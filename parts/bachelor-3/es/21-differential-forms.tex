\chapter{Formas diferenciales y teorema de Stokes}\label{ch:b3:forms}

Un teorema del análisis ha absorbido, en dos siglos, a todos los demás
de su especie: el teorema fundamental del cálculo, Green--Riemann
(demostrado en el volumen de segundo año), el teorema de la divergencia
de Gauss, el teorema del rotacional de Kelvin--Stokes --- cada uno dice
que la integral de cierta derivada sobre una región es igual a la
integral del objeto original sobre el
\omterm{def:b3:forms:boundary}{borde}. El lenguaje de las
\emph{\omterm{def:b3:forms:diffform}{formas diferenciales}} los
convierte en un solo enunciado,
$\int_M\dd\omega = \int_{\partial M}\omega$, y hace ese enunciado
demostrable de una sola vez. Este capítulo construye el lenguaje
honestamente --- álgebra multilineal alternada,
\omterm{def:b3:forms:d}{derivada exterior},
\omterm{def:b3:forms:pullback}{retrocesos},
\omterm{def:b3:forms:orientation}{orientación}, integración sobre las
subvariedades del~\cref{ch:b3:submanifolds} ---, demuestra el teorema de
Stokes y cobra los primeros cheques: los teoremas integrales clásicos,
el \omterm{def:b3:forms:winding}{índice} que en secreto gobernaba
el~\cref{ch:b3:residues} y, en el problema de fin de semana, el teorema
del punto fijo de Brouwer. En todo el capítulo, \emph{regular} significa
$\mathcal
C^\infty$; toda aplicación y toda forma son regulares salvo mención
expresa. Esto no cuesta ninguna generalidad que merezca la pena a este
nivel y deja las manos libres.

\section{Álgebra multilineal alternada}

\begin{definition}\label{def:b3:forms:alternating}
Sea $E$ un espacio vectorial real de dimensión $n$. Una \emph{forma
$k$-lineal alternada}\index{forma alternada} en $E$ es una aplicación
$\alpha\colon E^k \to \R$, lineal en cada variable, con
$\alpha(v_1, \dots, v_k) = 0$ siempre que dos argumentos sean iguales.
Su espacio se denota $\Lambda^k E^*$; por convenio, $\Lambda^0E^* = \R$.
La alternancia fuerza la antisimetría: intercambiar dos argumentos
cambia el signo (desarróllese
$\alpha(\dots, v + w, \dots, v + w, \dots) = 0$) y, más en general,
$\alpha(v_{\sigma(1)}, \dots, v_{\sigma(k)}) =
\varepsilon(\sigma)\,\alpha(v_1, \dots, v_k)$ para toda permutación
$\sigma$.
\end{definition}

\begin{example}
En $E = \R^n$: $\Lambda^1E^* = E^*$ es el
\omterm{def:b3:banach:operator}{espacio dual}; el determinante en la
base canónica es una \omterm{def:b3:forms:alternating}{forma alternada}
$n$-lineal, y la~\cref{prop:b3:forms:basis} mostrará que genera
$\Lambda^nE^*$ --- la razón profunda de que el determinante sea único
salvo escala. Para $k > n$, $\Lambda^kE^* = \{0\}$: $k$ vectores son
dependientes, y desarrollar uno según los demás anula $\alpha$ por
alternancia.
\end{example}

\begin{definition}\label{def:b3:forms:wedge}
Para $\ell_1, \dots, \ell_k \in E^*$, su \emph{producto
exterior}\index{producto exterior} es la
\omterm{def:b3:forms:alternating}{forma alternada} $k$-lineal
\[
(\ell_1 \wedge \dots \wedge \ell_k)(v_1, \dots, v_k)
= \det\bigl(\ell_i(v_j)\bigr)_{1 \leq i, j \leq k} .
\]
La alternancia y la multilinealidad son las del determinante en sus
columnas.
\end{definition}

\begin{proposition}[Base de $\Lambda^k$]\label{prop:b3:forms:basis}
Sea $(e_1, \dots, e_n)$ una base de $E$ con base
\omterm{def:b3:banach:operator}{dual} $(e_1^*, \dots, e_n^*)$. Las
formas
\[
e_I^* = e_{i_1}^* \wedge \dots \wedge e_{i_k}^*,
\qquad I = \{i_1 < \dots < i_k\} \subseteq \{1, \dots, n\},
\]
constituyen una base de $\Lambda^kE^*$; por tanto, $\dim\Lambda^kE^* =
\binom nk$. Explícitamente, $\alpha =
\sum_{\abs I = k}\alpha(e_{i_1}, \dots, e_{i_k})\,e_I^*$.
\end{proposition}

\begin{proof}
\emph{Generación.} Sean $\alpha \in \Lambda^kE^*$ y $\beta =
\sum_I\alpha(e_I)\,e_I^*$, donde $\alpha(e_I)$ abrevia
$\alpha(e_{i_1}, \dots, e_{i_k})$. Ambos miembros son $k$-lineales y
alternados, de modo que coinciden en cuanto coincidan en todas las
$k$-tuplas $(e_{j_1}, \dots, e_{j_k})$ con $j_1 < \dots <
j_k$ (la multilinealidad reduce a tuplas de vectores de la base, y la
alternancia a las estrictamente crecientes). Y $e_I^*(e_{j_1},
\dots, e_{j_k}) = \det(e_{i_r}^*(e_{j_s})) = \delta_{IJ}$: para $I = J$
la matriz es la identidad; para $I \neq J$ alguna fila es nula. Así,
$\beta(e_J) = \alpha(e_J)$ para todo $J$: $\beta = \alpha$.
\emph{Independencia.} Si $\sum_I c_Ie_I^* = 0$, evaluando en
$(e_{j_1}, \dots, e_{j_k})$ se obtiene $c_J = 0$.
\end{proof}

\begin{definition}\label{def:b3:forms:wedgegeneral}
El \omterm{def:b3:forms:wedge}{producto exterior} se extiende a una
aplicación bilineal $\Lambda^kE^* \times \Lambda^\ell E^* \to
\Lambda^{k+\ell}E^*$, determinada por la bilinealidad y por
$(e_I^*) \wedge (e_J^*) = e_I^* \wedge e_J^*$ (concaténese y reordénese;
el producto es $0$ si $I \cap J \neq
\varnothing$). Es asociativo y \emph{anticonmutativo graduado}:
\[
\beta \wedge \alpha = (-1)^{k\ell}\,\alpha \wedge \beta
\qquad (\alpha \in \Lambda^k, \ \beta \in \Lambda^\ell).
\]
\end{definition}

\begin{proof}[\omnameProof]
Ambas propiedades se comprueban en los elementos de la base y se
extienden por bilinealidad. Asociatividad: ambos parentizados de
$e_I^* \wedge e_J^* \wedge e_K^*$ coinciden con el \omterm{def:b3:forms:wedge}{producto exterior} de
la familia concatenada de $1$-formas, por la fórmula del determinante de
la~\cref{def:b3:forms:wedge} (desarrollo de Laplace por bloques). La
regla de los signos: pasar cada uno de los $\ell$ factores de $\beta$
por delante de los $k$ factores de $\alpha$ cuesta un signo por cada
transposición adyacente (un intercambio de dos filas del determinante),
y por tanto $(-1)^{k\ell}$ en total.
\end{proof}

\begin{proposition}[Retroceso, caso lineal]
\label{prop:b3:forms:linearpullback}
Una aplicación lineal $u\colon E \to F$ induce, para cada $k$, la
aplicación lineal $u^*\colon \Lambda^kF^* \to \Lambda^kE^*$,
$(u^*\alpha)(v_1, \dots, v_k) = \alpha(u(v_1), \dots,
u(v_k))$. Cumple $u^*(\alpha \wedge \beta) = u^*\alpha
\wedge u^*\beta$ y $(u \circ w)^* = w^* \circ u^*$. Además:
\begin{enumerate}
  \item Si $\dim E = n$ y $u\colon E \to E$, entonces en la recta
  $\Lambda^nE^*$: $u^*\alpha = (\det u)\,\alpha$. \item Si
  $\operatorname{rk}u < k$, entonces $u^* = 0$ en
        $\Lambda^kF^*$.
\end{enumerate}
\end{proposition}

\begin{proof}
Las identidades funtoriales son inmediatas de las definiciones (para la
regla del producto, compruébese en \omterm{def:b3:forms:wedge}{productos exteriores} de $1$-formas
con la fórmula del determinante --- $\det(\ell_i(uv_j)) =
\det((u^*\ell_i)(v_j))$ --- y extiéndase después bilinealmente). (1)
$u^*$ lleva el espacio unidimensional $\Lambda^nE^*$
(la~\cref{prop:b3:forms:basis}) en sí mismo, de modo que $u^*\alpha =
c\,\alpha$ con $c$ independiente de $\alpha \neq 0$; evaluando en
$\alpha = e_1^* \wedge \dots \wedge e_n^*$ y $(v_j) =
(e_j)$ se obtiene $c = \det(e_i^*(ue_j)) = \det u$. (2) Para
$v_1, \dots, v_k \in E$, los vectores $u(v_1), \dots,
u(v_k)$ están en la imagen de $u$, de dimensión $< k$: son linealmente
dependientes, y una \omterm{def:b3:forms:alternating}{forma alternada}
se anula sobre una familia dependiente (desarróllese el vector
dependiente según los demás).
\end{proof}

\section{Formas diferenciales y derivada exterior}

\begin{definition}\label{def:b3:forms:diffform}
Sea $U \subseteq \R^n$ abierto. Una \emph{$k$-forma
diferencial}\index{forma diferencial} en $U$ es una aplicación regular
$\omega\colon U \to \Lambda^k(\R^n)^*$; en la base de
la~\cref{prop:b3:forms:basis} (escribiendo $\dd x_i$ por
$e_i^*$),
\[
\omega = \sum_{\abs I = k} a_I\,\dd x_I,
\qquad
\dd x_I = \dd x_{i_1} \wedge \dots \wedge \dd x_{i_k},
\]
con coeficientes regulares $a_I \in \mathcal C^\infty(U)$. Su espacio es
$\Omega^k(U)$; $\Omega^0(U) = \mathcal
C^\infty(U)$. Una $0$-forma es una función; una $1$-forma es un campo de
formas lineales (por ejemplo, la diferencial $\dd f$ de una función);
una $n$-forma es $a\,\dd x_1\wedge\dots\wedge\dd
x_n$, el integrando natural del~\cref{ch:b3:product}.
\end{definition}

\begin{definition}[Derivada exterior]\label{def:b3:forms:d}
La \emph{derivada exterior}\index{derivada exterior} es la aplicación
lineal $\dd\colon \Omega^k(U) \to \Omega^{k+1}(U)$ definida por
\[
\dd\Bigl(\sum_I a_I\,\dd x_I\Bigr)
= \sum_I \dd a_I \wedge \dd x_I
= \sum_I\sum_{j=1}^n \frac{\partial a_I}{\partial
x_j}\,\dd x_j \wedge \dd x_I .
\]
En las $0$-formas es la diferencial usual.
\end{definition}

\begin{theorem}\label{thm:b3:forms:dsquare}
(a) $\dd(\omega \wedge \eta) = \dd\omega \wedge \eta +
(-1)^k\,\omega \wedge \dd\eta$ para $\omega \in \Omega^k$ (la
\emph{regla de Leibniz graduada}). (b) $\dd \circ \dd = 0$.\index{d
cuadrado nula@$\dd^2 = 0$}
\end{theorem}

\begin{proof}
(a) Por bilinealidad basta tratar $\omega = a\,\dd
x_I$, $\eta = b\,\dd x_J$. Entonces $\omega \wedge \eta =
ab\,\dd x_I \wedge \dd x_J$ y
\[
\dd(\omega\wedge\eta) = (b\,\dd a + a\,\dd b) \wedge \dd x_I
\wedge \dd x_J
= (\dd a \wedge \dd x_I) \wedge (b\,\dd x_J) +
a\,\dd b \wedge \dd x_I \wedge \dd x_J,
\]
y pasar la $1$-forma $\dd b$ por delante de los $k$ factores de $\dd
x_I$ cuesta $(-1)^k$ (la~\cref{def:b3:forms:wedgegeneral}): el segundo
término es $(-1)^k\,\omega \wedge \dd\eta$. (b) Para una función:
$\dd(\dd a) = \sum_{i,j}
\frac{\partial^2a}{\partial x_j\partial x_i}\,\dd x_j \wedge
\dd x_i$. El coeficiente es simétrico en $(i,j)$ por el teorema de
Schwarz sobre las derivadas parciales cruzadas (demostrado en el volumen
de segundo año; $a$ es $\mathcal C^\infty$), mientras que
$\dd x_j \wedge
\dd x_i$ es antisimétrico: emparejando los términos $(i,j)$ y $(j,i)$,
todo se cancela. Para $\omega = \sum
a_I\dd x_I$ general: $\dd\dd\omega = \sum\dd(\dd a_I \wedge \dd x_I)
= \sum(\dd\dd a_I\wedge\dd x_I - \dd a_I\wedge\dd(\dd x_I))$ por (a), y
ambos términos se anulan ($\dd(\dd x_I) = 0$, pues el coeficiente es
constante).
\end{proof}

\begin{definition}[Retroceso]\label{def:b3:forms:pullback}
Sea $\varphi\colon U \to V$ regular ($U \subseteq \R^m$,
$V \subseteq \R^n$ abiertos). El \emph{retroceso}\index{retroceso}
$\varphi^*\colon \Omega^k(V) \to \Omega^k(U)$ se define puntualmente por
el retroceso lineal a lo largo de la diferencial:
$(\varphi^*\omega)_x = (D\varphi(x))^*\,\omega_{\varphi(x)}$.
En concreto, $\varphi^*$ sustituye: $\varphi^*f = f \circ
\varphi$ en las funciones, $\varphi^*(\dd y_i) =
\dd\varphi_i = \sum_j\frac{\partial\varphi_i}{\partial
x_j}\dd x_j$ y $\varphi^*(a\,\dd y_{i_1}\wedge\dots\wedge
\dd y_{i_k}) =
(a\circ\varphi)\,\dd\varphi_{i_1}\wedge\dots\wedge
\dd\varphi_{i_k}$.
\end{definition}

\begin{theorem}\label{thm:b3:forms:pullback}
(a) $\varphi^*(\omega \wedge \eta) = \varphi^*\omega \wedge
\varphi^*\eta$ y $(\psi \circ \varphi)^* = \varphi^* \circ
\psi^*$. (b) $\varphi^*(\dd\omega) = \dd(\varphi^*\omega)$: la
\omterm{def:b3:forms:d}{derivada exterior} conmuta con toda sustitución
regular --- la identidad que la convierte en \emph{la} derivada de la
teoría. (c) Si $\varphi\colon U \to V$ es regular entre abiertos de
$\R^n$ y $\omega = a\,\dd y_1 \wedge \dots \wedge \dd
y_n$, entonces $\varphi^*\omega = (a \circ
\varphi)\,\det\bigl(D\varphi\bigr)\,\dd x_1 \wedge \dots
\wedge \dd x_n$.
\end{theorem}

\begin{proof}
(a) Enunciados puntuales sobre
\omterm{def:b3:forms:pullback}{retrocesos} lineales
(la~\cref{prop:b3:forms:linearpullback}), más la regla de la cadena
$D(\psi\circ\varphi)(x) = D\psi(\varphi(x))\,D\varphi(x)$.
(b) Para una $0$-forma $f$: $\varphi^*(\dd f) = \dd f \circ
D\varphi = \dd(f \circ \varphi)$ es la regla de la cadena. Para
$\omega = a\,\dd y_I$: usando (a), $\varphi^*\omega =
(a\circ\varphi)\,\dd\varphi_{i_1}\wedge\dots\wedge
\dd\varphi_{i_k}$, de modo que, por la regla de Leibniz
(el~\cref{thm:b3:forms:dsquare}(a)) y $\dd\dd\varphi_{i_r} =
0$:
\[
\dd(\varphi^*\omega) = \dd(a\circ\varphi) \wedge
\dd\varphi_{i_1}\wedge\dots\wedge\dd\varphi_{i_k}
= \varphi^*(\dd a) \wedge \varphi^*(\dd y_I)
= \varphi^*(\dd a \wedge \dd y_I) =
\varphi^*(\dd\omega).
\]
(c) Puntualmente esto es exactamente $u^*\alpha = (\det u)\alpha$ en las
formas de grado máximo (la~\cref{prop:b3:forms:linearpullback}(1) con
$u = D\varphi(x)$).
\end{proof}

\begin{example}[Coordenadas polares]\label{ex:b3:forms:polar}
Para $\varphi(r, \theta) = (r\cos\theta, r\sin\theta)$:
$\varphi^*\dd x = \cos\theta\,\dd r -
r\sin\theta\,\dd\theta$, $\varphi^*\dd y = \sin\theta\,\dd r
+ r\cos\theta\,\dd\theta$, de modo que
\[
\varphi^*(\dd x \wedge \dd y) =
(\cos\theta\,\dd r - r\sin\theta\,\dd\theta) \wedge
(\sin\theta\,\dd r + r\cos\theta\,\dd\theta)
= r\,\dd r \wedge \dd\theta,
\]
apareciendo el jacobiano del~\cref{ex:b3:product:polar} por pura álgebra
--- sin teoría de la \omterm{def:b3:measure:measure}{medida}.
El~\cref{exo:b3:forms:9} convierte esta observación en un enunciado:
para las integrales orientadas, la fórmula del cambio de variables
\emph{es} la fórmula del \omterm{def:b3:forms:pullback}{retroceso}.
\end{example}

\section{Formas cerradas y exactas; el lema de Poincaré}

\begin{definition}\label{def:b3:forms:closedexact}
$\omega \in \Omega^k(U)$ es \emph{cerrada}\index{forma cerrada} si
$\dd\omega = 0$, y \emph{exacta}\index{forma exacta} si
$\omega = \dd\eta$ para alguna $\eta \in \Omega^{k-1}(U)$ (una
\emph{primitiva} de $\omega$). Exacta $\Rightarrow$ cerrada por
$\dd^2 = 0$; el recíproco es una cuestión sobre la \emph{forma} de $U$.
\end{definition}

\begin{example}[La forma angular]\label{ex:b3:forms:angular}
En $U = \R^2 \setminus \{0\}$,
\[
\omega_\theta = \frac{x\,\dd y - y\,\dd x}{x^2 + y^2}
\]
es \omterm{def:b3:forms:closedexact}{cerrada} (cálculo directo:
el~\cref{exo:b3:forms:4}) pero no
\omterm{def:b3:forms:closedexact}{exacta}: su integral a lo largo de la
circunferencia unidad vale $2\pi \neq 0$, mientras que las integrales de
\omterm{def:b3:forms:closedexact}{formas exactas} a lo largo de curvas
\omterm{def:b3:forms:closedexact}{cerradas} se anulan
(la~\cref{prop:b3:forms:exactloop}). Localmente,
$\omega_\theta = \dd\theta$ para cualquier determinación regular
$\theta$ del ángulo polar --- de ahí el nombre y la obstrucción: no
existe tal determinación en todo $U$. Esta única forma gobierna el
\omterm{def:b3:forms:winding}{índice} (la~\cref{sec:b3:forms:winding})
y, a través de él, el teorema de los
\omterm{def:b3:residues:singularities}{residuos}
del~\cref{ch:b3:residues}.
\end{example}

\begin{theorem}[Lema de Poincaré]\label{thm:b3:forms:poincare}
Sea $U \subseteq \R^n$ abierto y \emph{estrellado} respecto de $0$. Toda
$k$-forma \omterm{def:b3:forms:closedexact}{cerrada} en $U$ ($k \geq 1$)
es \omterm{def:b3:forms:closedexact}{exacta}.\index{lema de Poincaré}
\end{theorem}

\begin{proof}
Construimos un \emph{operador de homotopía} lineal $h\colon
\Omega^k(U) \to \Omega^{k-1}(U)$ con
\begin{equation}\label{eq:b3:forms:homotopy}
\dd(h\omega) + h(\dd\omega) = \omega
\qquad (k \geq 1);
\end{equation}
si $\dd\omega = 0$, entonces $\omega = \dd(h\omega)$ y hemos terminado.
Para $\omega = \sum_I a_I\,\dd x_I$, póngase
\[
h\omega = \sum_{\abs I = k}\ \sum_{r=1}^{k}(-1)^{r-1}
\Bigl(\int_0^1 t^{k-1}a_I(tx)\,\dd t\Bigr)\,
x_{i_r}\,\dd x_{i_1}\wedge\dots\wedge
\widehat{\dd x_{i_r}}\wedge\dots\wedge\dd x_{i_k}
\]
(el acento circunflejo suprime un factor; las integrales son regulares
en $x$ por derivación bajo la integral,
el~\cref{thm:b3:lebesgue:paramdiff}, estando todas las derivadas
dominadas en los \omterm{def:b3:topology:compact}{compactos}). Comprobar
la~\eqref{eq:b3:forms:homotopy} es un cálculo que se hace una vez en la
vida, de modo que lo hacemos entero. Fíjese $I$ y tómese $\omega =
a\,\dd x_I$ (linealidad). Primero,
\[
\dd(h\omega) = k\Bigl(\int_0^1t^{k-1}a(tx)\dd t\Bigr)\dd x_I
+ \sum_{r=1}^k(-1)^{r-1}\sum_{j=1}^n
\Bigl(\int_0^1t^{k}\,\partial_ja(tx)\,\dd t\Bigr)
x_{i_r}\,\dd x_j\wedge\dd x_{I\setminus i_r} :
\]
el primer grupo recoge los términos en que $\dd$ golpea el factor
$x_{i_r}$ --- el \omterm{def:b3:forms:wedge}{producto exterior} $\dd x_{i_r}\wedge\dd
x_{I\setminus i_r}$ reconstruye $\dd x_I$ con un signo $(-1)^{r-1}$ que
cancela el prefactor, y los $k$ valores de $r$ dan el factor $k$ ---,
mientras que el segundo grupo recoge los términos en que $\dd$ golpea la
integral (la regla de la cadena saca $t\,\partial_ja(tx)$). A
continuación, $\dd\omega =
\sum_j\partial_ja\,\dd x_j\wedge\dd x_I$ y, aplicando la definición de
$h$ en grado $k+1$, con el \omterm{def:b3:forms:winding}{índice} $j$ ocupando la primera posición:
\[
h(\dd\omega) = \sum_{j=1}^n\Bigl(\int_0^1t^{k}
\partial_ja(tx)\dd t\Bigr)x_j\,\dd x_I
- \sum_{j=1}^n\sum_{r=1}^k(-1)^{r-1}
\Bigl(\int_0^1t^{k}\partial_ja(tx)\dd t\Bigr)
x_{i_r}\,\dd x_j\wedge\dd x_{I\setminus i_r} .
\]
Las sumas dobles se cancelan en $\dd(h\omega) + h(\dd\omega)$, que por
tanto vale
\[
\Bigl(\int_0^1\bigl(k\,t^{k-1}a(tx) +
t^k{\textstyle\sum_j}x_j\,\partial_ja(tx)\bigr)\dd t\Bigr)
\dd x_I
= \Bigl(\int_0^1\frac{\dd}{\dd t}\bigl(t^ka(tx)\bigr)\dd
t\Bigr)\dd x_I
= a(x)\,\dd x_I = \omega,
\]
por el teorema fundamental del cálculo. La condición de ser estrellado
entró donde tenía que entrar: $tx \in U$ para $t \in \intcc01$, de modo
que $a(tx)$ tiene sentido.
\end{proof}

\begin{remark}
Para $k = 1$ y $\omega = \sum_j a_j\dd x_j$, la primitiva es
$f(x) = \int_0^1\sum_ja_j(tx)\,x_j\,\dd t$ --- la integral de línea de
$\omega$ a lo largo del segmento $[0, x]$: el teorema es el «un campo
con jacobiana simétrica es un gradiente» de varias variables de segundo
año, ahora en todos los grados. La forma angular
(el~\cref{ex:b3:forms:angular}) muestra que la hipótesis sobre $U$ no es
decorativa: $\R^2\setminus\{0\}$ no es estrellado, y allí ser \omterm{def:b3:forms:closedexact}{cerrada} no
implica ser \omterm{def:b3:forms:closedexact}{exacta}. Lo que sobrevive en un
abierto general lo mide la
\emph{cohomología de de Rham} $H^k(U) =
\ker\dd/\operatorname{im}\dd$ --- véase el~\cref{exo:b3:forms:12} para
el primer cálculo no trivial.
\end{remark}

\section{Orientación e integración sobre subvariedades}

Integrar una $k$-forma exige territorio orientado de dimensión $k$.
Recuérdese del~\cref{ch:b3:submanifolds}
(el~\cref{thm:b3:submanifolds:characterizations}) que una
$k$-subvariedad $M \subseteq \R^n$ es localmente la imagen de una
parametrización regular $\gamma\colon V \to M \cap W$ ($V
\subseteq \R^k$ abierto, $\gamma$ un
\omterm{def:b3:topology:continuity}{homeomorfismo} sobre su imagen con
diferencial inyectiva).

\begin{definition}\label{def:b3:forms:orientation}
Una \emph{orientación}\index{orientación} de $M$ es la elección, para
cada $p \in M$, de una de las dos clases de orientación de bases del
\omterm{def:b3:submanifolds:tangent}{espacio tangente} $T_pM$, que sea
\emph{localmente coherente}: alrededor de cada punto hay una
parametrización $\gamma$ cuyo referencial de coordenadas
$(\partial_1\gamma, \dots,
\partial_k\gamma)$ está positivamente orientado en todo punto de su
dominio. Tales parametrizaciones se llaman \emph{directas}. $M$ es
\emph{orientable} si existe una orientación; la banda de Möbius muestra
que esto puede fallar. Todas las subvariedades de este capítulo están
orientadas.
\end{definition}

\begin{definition}[Integral de una forma]\label{def:b3:forms:integral}
Sean $M$ una $k$-subvariedad orientada y $\omega$ una $k$-forma definida
en un \omterm{def:b3:topology:topology}{entorno} de $M$, con
$\operatorname{supp}\omega \cap M$
\omterm{def:b3:topology:compact}{compacto}. (a) Si
$\operatorname{supp}\omega \cap M \subseteq
\gamma(V)$ para una única parametrización directa, póngase
\[
\int_M\omega = \int_V\gamma^*\omega
\]
--- siendo el miembro derecho la integral de Lebesgue en $V$
(el~\cref{ch:b3:product}) del coeficiente $g$ de
$\gamma^*\omega = g\,\dd u_1\wedge\dots\wedge\dd u_k$, que es
\omterm{def:b3:topology:continuity}{continuo} y de soporte
\omterm{def:b3:topology:compact}{compacto}. (b) En general, elíjanse un
número finito de parametrizaciones directas $\gamma_i(V_i)$ que recubran
el \omterm{def:b3:topology:compact}{compacto}
$\operatorname{supp}\omega \cap M$ y una
\omterm{lem:b3:forms:partition}{partición de la unidad} subordinada
$(\chi_i)$ (el~\cref{lem:b3:forms:partition}), y póngase $\int_M\omega =
\sum_i\int_M\chi_i\,\omega$, calculando cada término por (a). Para una
curva ($k = 1$) parametrizada por $\gamma\colon\intcc ab\to\R^n$
escribimos $\int_\gamma\omega = \int_a^b\gamma^*\omega$, sin exigir
inyectividad.
\end{definition}

\begin{lemma}[Consistencia]\label{lem:b3:forms:consistency}
La definición (a) no depende de la parametrización directa, y la
definición (b) no depende ni del recubrimiento ni de la
\omterm{lem:b3:forms:partition}{partición de la unidad}. Además, si
$\Phi$ es un difeomorfismo de
\omterm{def:b3:topology:topology}{entornos} de dos subvariedades
orientadas con $\Phi(M) = M'$, que lleva un referencial directo de $M$ a
un referencial directo de $M'$ en algún punto de cada componente de $M$,
entonces $\int_{M'}\omega =
\int_M\Phi^*\omega$.
\end{lemma}

\begin{proof}
(a) Sean $\gamma\colon V \to M$, $\delta\colon V' \to M$
parametrizaciones directas cuyas imágenes contienen
$\operatorname{supp}\omega\cap M$. La transición $\tau =
\delta^{-1}\circ\gamma$ es un difeomorfismo entre los abiertos
correspondientes de $V, V'$ (regularidad de las transiciones:
el~\cref{thm:b3:submanifolds:characterizations}, vía la descripción
local por grafos), y allí $\gamma = \delta\circ\tau$, de modo que
$\gamma^*\omega = \tau^*(\delta^*\omega)$
(el~\cref{thm:b3:forms:pullback}(a)). Escríbase $\delta^*\omega =
g\,\dd u_1\wedge\dots\wedge\dd u_k$; entonces
(\cref{thm:b3:forms:pullback}(c)) $\tau^*(\delta^*\omega) =
(g\circ\tau)\,\det(D\tau)\,\dd u_1\wedge\dots\wedge\dd u_k$.
Siendo directos ambos referenciales, $D\tau$ lleva una base positiva a
una base positiva: $\det D\tau > 0$, de modo que $\det D\tau =
\abs{\det D\tau}$ y el teorema del cambio de variables
(el~\cref{thm:b3:product:changeofvar}) da
$\int(g\circ\tau)\abs{\det D\tau} = \int g$: las dos integrales
coinciden. \emph{He aquí toda la razón de ser de la
\omterm{def:b3:forms:orientation}{orientación}}: sin el control del
signo, el jacobiano y su valor absoluto difieren y la integral queda mal
definida. (b) Si $(\chi_i)$ y $(\tilde\chi_j)$ son dos particiones
admisibles (recubrimientos incluidos), entonces, por (a) y la aditividad
finita, $\sum_i\int\chi_i\omega =
\sum_{i,j}\int\chi_i\tilde\chi_j\omega =
\sum_j\int\tilde\chi_j\omega$, siendo cada término doble calculable en
cualquiera de las dos cartas. La última afirmación: si $\gamma$ recorre
las parametrizaciones directas de $M$, entonces $\Phi\circ\gamma$
recorre las parametrizaciones directas de $M'$ (la comparación de
orientaciones es localmente constante
y está fijada en un punto por componente), y
$(\Phi\circ\gamma)^*\omega =
\gamma^*(\Phi^*\omega)$.
\end{proof}

\begin{lemma}[Particiones de la unidad, caso compacto]
\label{lem:b3:forms:partition}
Sean $K \subseteq \R^n$ \omterm{def:b3:topology:compact}{compacto} y
$W_1, \dots, W_m$ abiertos que
recubren $K$. Existen $\chi_1, \dots, \chi_m \in
\mathcal C^\infty(\R^n)$ con $0 \leq \chi_i \leq 1$,
$\operatorname{supp}\chi_i \subseteq W_i$
\omterm{def:b3:topology:compact}{compactos} y $\sum\chi_i = 1$ en un
\omterm{def:b3:topology:topology}{entorno} de $K$.\index{partición de la
unidad}
\end{lemma}

\begin{proof}
Cada $x \in K$ está en algún $W_{i(x)}$ con una bola
\omterm{def:b3:forms:closedexact}{cerrada}
$\bar B(x, 2r_x) \subseteq W_{i(x)}$; la \omterm{def:b3:topology:compact}{compacidad} extrae
$x_1, \dots, x_N$ con las bolas $B(x_s, r_{x_s})$ recubriendo $K$. Para
cada $s$, tómese una meseta $\theta_s \in \mathcal
C^\infty$, $0 \leq \theta_s \leq 1$, $\theta_s = 1$ en
$\bar B(x_s, r_{x_s})$, $\operatorname{supp}\theta_s
\subseteq B(x_s, 2r_{x_s})$ (regularícese el indicador de la bola de
radio $\frac32r_{x_s}$, el~\cref{thm:b3:lp:regularization}). Asígnese
cada $s$ a un \omterm{def:b3:forms:winding}{índice} $i(s)$ con $B(x_s, 2r_{x_s}) \subseteq W_{i(s)}$ y
póngase $\Theta_i = \sum_{i(s) = i}\theta_s$. En el
abierto
$\Omega_0 = \{\sum_j\Theta_j > \tfrac12\} \supseteq K$, las funciones
$\Theta_i/\sum_j\Theta_j$ hacen el trabajo, pero solo están definidas
allí; para globalizar, sea $\rho \in \mathcal
C^\infty(\R^n)$ tal que $\rho = 0$ donde $\sum_j\Theta_j
\geq 1$ y $\rho > 0$ donde $\sum_j\Theta_j \leq \tfrac12$ (regularícese
un corte adecuado de $1 - \sum_j\Theta_j$), y póngase
\[
\chi_i = \frac{\Theta_i}{\rho + \sum_j\Theta_j} .
\]
El denominador es en todas partes $> 0$ y vale $\sum_j\Theta_j$ en
$\{\sum_j\Theta_j \geq 1\}$, un
\omterm{def:b3:topology:topology}{entorno} abierto de $K$ (cada punto de
$K$ está en alguna bola $B(x_s, r_{x_s})$ donde $\theta_s = 1$); allí
$\sum_i\chi_i = 1$. Los soportes y las cotas son claros.
\end{proof}

\section{El teorema de Stokes}

\begin{definition}\label{def:b3:forms:boundary}
Una \emph{$k$-subvariedad con borde}\index{subvariedad con borde}
$M \subseteq \R^n$ es un conjunto recubierto por parametrizaciones
regulares de dos tipos: \emph{cartas
\omterm{def:b3:topology:interior}{interiores}}
$\gamma\colon V \to M \cap W$ con $V \subseteq \R^k$ abierto, y
\emph{cartas de borde} $\gamma\colon V \cap H^k \to M
\cap W$, donde $H^k = \{u \in \R^k : u_k \geq 0\}$ y $\gamma$ se
extiende de manera regular y con diferencial inyectiva al abierto $V$.
El \emph{borde} $\partial M$ es el conjunto de los puntos alcanzados en
$u_k = 0$; es una $(k-1)$-subvariedad sin borde, parametrizada por las
aplicaciones $u' \mapsto \gamma(u',
0)$. Una \omterm{def:b3:forms:orientation}{orientación} de $M$ induce
otra en $\partial M$ por la regla de la \emph{normal exterior en primer
lugar}: en $p \in \partial M$, una base $(w_1, \dots, w_{k-1})$ de
$T_p\partial M$ es positiva si y solo si $(\nu, w_1, \dots, w_{k-1})$ es
una base positiva de $T_pM$, donde
$\nu \in T_pM \setminus T_p\partial M$ apunta hacia fuera de $M$ (en una
carta de borde: $\nu =
-\partial_k\gamma$, salvo la suma de componentes tangenciales --- la
clase de \omterm{def:b3:forms:orientation}{orientación} no las ve).
\end{definition}

\begin{lemma}[Stokes en el semiespacio]
\label{lem:b3:forms:halfspace}
Sea $\eta$ una $(k-1)$-forma regular en $\R^k$ de soporte
\omterm{def:b3:topology:compact}{compacto}. Entonces
\[
\int_{H^k}\dd\eta = \int_{\partial H^k}\eta,
\]
donde $H^k$ lleva la \omterm{def:b3:forms:orientation}{orientación}
estándar de $\R^k$ y $\partial H^k = \{u_k = 0\} \cong \R^{k-1}$ la
inducida, que es $(-1)^k$ veces la
\omterm{def:b3:forms:orientation}{orientación} estándar de
$\R^{k-1}$.
\end{lemma}

\begin{proof}
Primero, la contabilidad de la
\omterm{def:b3:forms:orientation}{orientación}: la normal exterior a lo
largo de $\partial H^k$ es $-e_k$, y
\[
\det(-e_k, e_1, \dots, e_{k-1}) =
-\det(e_k, e_1, \dots, e_{k-1}) = -(-1)^{k-1}\det(e_1,
\dots, e_k) = (-1)^k :
\]
el referencial $(e_1, \dots, e_{k-1})$ de $\partial H^k$ es positivo
para la \omterm{def:b3:forms:orientation}{orientación} inducida
exactamente cuando $k$ es par, de donde la comparación enunciada. Por
linealidad, tómese $\eta
= f\,\dd u_1\wedge\dots\wedge\widehat{\dd
u_i}\wedge\dots\wedge\dd u_k$, $f \in \mathcal
C^\infty_c(\R^k)$; entonces $\dd\eta =
(-1)^{i-1}\,\partial_if\,\dd u_1\wedge\dots\wedge\dd u_k$ (llevar
$\dd u_i$ a su posición cuesta $i - 1$ intercambios). Dos casos, ambos
por Tonelli--Fubini (el~\cref{thm:b3:product:fubini}) y el teorema
fundamental del cálculo en una variable.

\emph{Caso $i < k$.} Integrando primero en $u_i$ sobre $\R$:
$\int_\R\partial_if\,\dd u_i = 0$ (soporte
\omterm{def:b3:topology:compact}{compacto}), de modo que
$\int_{H^k}\dd\eta = 0$. Y la restricción de $\eta$ a $\{u_k = 0\}$
contiene el factor $\dd u_k$, que se restringe a $0$ ($u_k$ es allí
constante): $\int_{\partial H^k}\eta =
0$ también.

\emph{Caso $i = k$.} Integrando primero en $u_k$ sobre
$\intco0\infty$:
\begin{align*}
\int_{H^k}\dd\eta &= (-1)^{k-1}\int_{\R^{k-1}}
\Bigl(\int_0^\infty\partial_kf\,\dd u_k\Bigr)\dd u' \\
&= (-1)^{k-1}\int_{\R^{k-1}}\bigl(0 - f(u', 0)\bigr)\dd u'
= (-1)^k\int_{\R^{k-1}}f(u',0)\,\dd u' .
\end{align*}
Del lado del \omterm{def:b3:forms:boundary}{borde}, $\eta$ se restringe
a $f(u',
0)\,\dd u_1\wedge\dots\wedge\dd u_{k-1}$ y, siendo la
\omterm{def:b3:forms:orientation}{orientación} inducida $(-1)^k$ veces
la estándar, $\int_{\partial H^k}\eta =
(-1)^k\int_{\R^{k-1}}f(u',0)\,\dd u'$. Los dos miembros coinciden.
\end{proof}

\begin{theorem}[Stokes]\label{thm:b3:forms:stokes}
Sean $M \subseteq \R^n$ una $k$-subvariedad orientada
compacta con
\omterm{def:b3:forms:boundary}{borde}, con $\partial M$ dotado de la
\omterm{def:b3:forms:orientation}{orientación} inducida, y $\omega$ una
$(k-1)$-forma regular en un \omterm{def:b3:topology:topology}{entorno}
de $M$. Entonces\index{teorema de Stokes}
\[
\int_M\dd\omega = \int_{\partial M}\omega .
\]
En particular, si $\partial M = \varnothing$:
$\int_M\dd\omega = 0$.
\end{theorem}

\begin{proof}
Recúbrase el \omterm{def:b3:topology:compact}{compacto} $M$ por un
número finito de imágenes de cartas directas
(\omterm{def:b3:topology:interior}{interiores} o de
\omterm{def:b3:forms:boundary}{borde}), y tómese una
\omterm{lem:b3:forms:partition}{partición de la unidad} $(\chi_i)$
subordinada a los abiertos
correspondientes $W_i$ de $\R^n$ (el~\cref{lem:b3:forms:partition}), con
$\sum\chi_i
= 1$ en un \omterm{def:b3:topology:topology}{entorno} de $M$. En ese
\omterm{def:b3:topology:topology}{entorno}, $\dd(\sum\chi_i) = 0$, de
modo que
\[
\int_M\dd\omega = \sum_i\int_M\dd(\chi_i\omega),
\qquad
\int_{\partial M}\omega =
\sum_i\int_{\partial M}\chi_i\omega :
\]
ambos miembros son aditivos, y basta demostrar el teorema para una forma
soportada en la imagen de una sola carta.

\emph{Carta \omterm{def:b3:topology:interior}{interior}.} Si
$\operatorname{supp}\omega \cap M
\subseteq \gamma(V)$ con $V$ abierto en $\R^k$: extiéndase $\eta =
\gamma^*\omega$ por cero a $\R^k$ (regular, de soporte
\omterm{def:b3:topology:compact}{compacto} en $V$) y aplíquese
el~\cref{lem:b3:forms:halfspace} con el soporte alejado de
$\partial H^k$ (trasládese $V$ al semiespacio superior abierto --- o
repítase sin más el cálculo del caso $i < k$ en todo $\R^k$):
$\int_M\dd\omega =
\int_{\R^k}\dd\eta = 0$, y $\int_{\partial M}\omega = 0$, pues $\omega$
se anula cerca de $\partial M$.

\emph{Carta de \omterm{def:b3:forms:boundary}{borde}.} Si
$\operatorname{supp}\omega\cap M
\subseteq \gamma(V \cap H^k)$: con $\eta = \gamma^*\omega$ extendida por
cero,
$\gamma^*(\dd\omega) = \dd\eta$
(el~\cref{thm:b3:forms:pullback}(b)), de modo que, por
\cref{def:b3:forms:integral}:
\[
\int_M\dd\omega = \int_{H^k}\dd\eta
\overset{\text{\cref{lem:b3:forms:halfspace}}}{=}
\int_{\partial H^k}\eta .
\]
Queda identificar el miembro derecho con $\int_{\partial
M}\omega$. El \omterm{def:b3:forms:boundary}{borde} está parametrizado
por $\beta(u') =
\gamma(u', 0)$, y $\beta^*\omega$ es la restricción de $\eta$ a
$\{u_k = 0\}$ (\omterm{def:b3:forms:pullback}{retroceso} por la
inclusión $u'
\mapsto (u', 0)$ compuesta con $\gamma$). La comparación de
orientaciones es el mismo $(-1)^k$ en
ambos miembros: el referencial
$(\partial_1\beta, \dots, \partial_{k-1}\beta)$ se sitúa en la
\omterm{def:b3:forms:orientation}{orientación} inducida de $\partial M$
con el signo $\det(-e_k, e_1, \dots, e_{k-1}) = (-1)^k$ calculado en la
carta (el vector exterior retrocede a $-e_k$), que es exactamente el
signo que relaciona la \omterm{def:b3:forms:orientation}{orientación}
inducida de $\partial H^k$ con la estándar de $\R^{k-1}$
(el~\cref{lem:b3:forms:halfspace}). Los dos convenios de signo se
cancelan:
$\int_{\partial H^k}\eta = \int_{\partial M}\omega$.
\end{proof}

\begin{example}[Los teoremas clásicos]
\label{ex:b3:forms:classical}
Sea $D \subseteq \R^2$ un dominio
\omterm{def:b3:topology:compact}{compacto} con
\omterm{def:b3:forms:boundary}{borde} regular, orientado de la manera
estándar. Para $\omega = P\,\dd x +
Q\,\dd y$: $\dd\omega = \bigl(\partial_xQ -
\partial_yP\bigr)\dd x\wedge\dd y$, y Stokes se lee
\[
\int_D\Bigl(\frac{\partial Q}{\partial x} -
\frac{\partial P}{\partial y}\Bigr)\dd x\,\dd y
= \oint_{\partial D}P\,\dd x + Q\,\dd y :
\]
\emph{Green--Riemann}, demostrado para dominios elementales en el
volumen de segundo año y ahora en su generalidad natural. En $\R^3$,
Stokes aplicado a la $2$-forma de \omterm{ex:b3:ode:planeclassification}{flujo} de un campo de vectores en un
dominio \omterm{def:b3:topology:compact}{compacto} da el \emph{teorema
de la divergencia} $\int_\Omega\operatorname{div}F =
\int_{\partial\Omega}\langle F, \nu\rangle\,\dd S$, y aplicado a una
$1$-forma en una superficie con \omterm{def:b3:forms:boundary}{borde}, el clásico teorema del rotacional
de \emph{Kelvin--Stokes}; el~\cref{exo:b3:forms:7} detalla ambos
diccionarios.
\end{example}

\section{El índice}\label{sec:b3:forms:winding}

\begin{definition}\label{def:b3:forms:winding}
Sea $\gamma\colon\intcc01\to\R^2\setminus\{a\}$ una curva
\omterm{def:b3:forms:closedexact}{cerrada} regular. Su
\emph{índice}\index{índice} alrededor de $a$ es
\[
\operatorname{Ind}_\gamma(a) =
\frac1{2\pi}\int_\gamma\omega_\theta^a,
\qquad
\omega_\theta^a = \frac{(x - a_1)\,\dd y - (y - a_2)\,\dd
x}{(x - a_1)^2 + (y - a_2)^2} .
\]
\end{definition}

\begin{proposition}\label{prop:b3:forms:windinginteger}
$\operatorname{Ind}_\gamma(a) \in \Z$; como función de $a$ es constante
en cada \omterm{def:b3:topology:components}{componente conexa} de
$\R^2\setminus\gamma(\intcc01)$ y nulo en la componente no acotada. Para
$\gamma(t) = a + r(\cos2\pi t, \sin2\pi t)$:
$\operatorname{Ind}_\gamma(a) = 1$.
\end{proposition}

\begin{proof}
Tómese $a = 0$ y escríbase $\gamma = (x, y)$, $\rho =
\norm\gamma > 0$, $\vartheta(t) =
\int_0^t\gamma^*\omega_\theta$, de modo que $\vartheta' =
\frac{xy' - yx'}{x^2 + y^2}$. En notación compleja, sea $u(t)
= \gamma(t)\,\rho(t)^{-1}\eu^{-\iu\vartheta(t)}$; entonces $\abs u = 1$
y un cálculo directo da
\[
\frac{u'}{u} = \frac{\gamma'}{\gamma} - \frac{\rho'}{\rho}
- \iu\vartheta'
= \frac{\bar\gamma\gamma'}{\abs\gamma^2} -
\frac{\rho'}{\rho} - \iu\,\frac{xy' -
yx'}{\rho^2}
= \frac{(xx' + yy')}{\rho^2} - \frac{\rho'}{\rho} = 0,
\]
puesto que $\bar\gamma\gamma' = (xx' + yy') + \iu(xy' - yx')$ y
$\rho\rho' = xx' + yy'$. Así, $u$ es constante: $\gamma(t) =
\rho(t)\,c\,\eu^{\iu\vartheta(t)}$ con $\abs c = 1$, y
$\gamma(1) = \gamma(0)$ con $\rho(1) = \rho(0)$ fuerza
$\eu^{\iu\vartheta(1)} = \eu^{\iu\vartheta(0)} = 1$:
$\vartheta(1) \in 2\pi\Z$, i.e.\
$\operatorname{Ind}_\gamma(0) \in \Z$. Como función de $a$ en el
complementario abierto de la curva
compacta, la integral que la define es
\omterm{def:b3:topology:continuity}{continua}
(el~\cref{thm:b3:lebesgue:paramcont}, con dominación en un
\omterm{def:b3:topology:topology}{entorno} de cada $a$); una función
\omterm{def:b3:topology:continuity}{continua} de valores enteros es
localmente constante, luego constante en cada componente. Para $\norm a$
grande, el integrando es $O(1/\norm a)$ uniformemente en $t$, de modo
que el \omterm{def:b3:forms:winding}{índice} tiende a $0$ y se anula en la componente no acotada. Para
la circunferencia: $\gamma^*\omega_\theta^a = 2\pi\,\dd t$ directamente.
\end{proof}

\begin{remark}
Con $\C \cong \R^2$, $\frac{\dd z}{z} = \frac{x\,\dd x +
y\,\dd y}{x^2 + y^2} + \iu\,\omega_\theta$, de modo que
$\operatorname{Ind}_\gamma(a) =
\frac1{2\iu\pi}\oint_\gamma\frac{\dd z}{z - a}$: este es el \omterm{def:b3:forms:winding}{índice}
del~\cref{ch:b3:residues}, y la~\cref{prop:b3:forms:windinginteger}
vuelve a demostrar su \omterm{def:b3:representations:character}{carácter} entero y su constancia local por medios
de variable real --- la mitad
\omterm{def:b3:topology:topology}{topológica} del teorema de los
\omterm{def:b3:residues:singularities}{residuos}, ahora apoyada en
Stokes.
\end{remark}

\begin{proposition}\label{prop:b3:forms:exactloop}
Si $\omega = \dd f$ es \omterm{def:b3:forms:closedexact}{exacta} en el
abierto $U$ y $\gamma\colon\intcc01\to U$ es una curva
\omterm{def:b3:forms:closedexact}{cerrada}, entonces
$\int_\gamma\omega = 0$.
\end{proposition}

\begin{proof}
$\int_\gamma\dd f = \int_0^1(f\circ\gamma)'(t)\,\dd t =
f(\gamma(1)) - f(\gamma(0)) = 0$ --- la regla de la cadena identifica
$\gamma^*(\dd f)$ con $(f\circ\gamma)'\,\dd t$.
\end{proof}

\begin{method}\label{met:b3:forms:method}
Calcular con formas: (1) mecanícese --- los \omterm{def:b3:forms:wedge}{productos exteriores} se
reordenan con signos, $\dd$ deriva los coeficientes generando nuevos
$\dd
x_j$ y los \omterm{def:b3:forms:pullback}{retrocesos} sustituyen;
confíese en el álgebra, que codifica todos los jacobianos. (2) Para
integrar una forma sobre una subvariedad: parametrícese directamente,
retrocédase la forma e intégrese el coeficiente; la
\omterm{def:b3:forms:orientation}{orientación} es la única trampa ---
compruébese un referencial. (3) Para demostrar una identidad integral,
búsquese una silueta de Stokes: ¿es el integrando
exacto?, ¿es el dominio un
\omterm{def:b3:forms:boundary}{borde}? (4) Para comparar integrales
sobre dos subvariedades «paralelas», aplíquese Stokes a la región
comprendida entre ellas (el argumento de deformación,
el~\cref{exo:b3:forms:10}). (5) Una integral no nula de una
\omterm{def:b3:forms:closedexact}{forma cerrada} certifica una
obstrucción \omterm{def:b3:topology:topology}{topológica} --- ninguna
primitiva, ninguna retracción, ninguna extensión sin ceros: así es como
el problema de fin de semana liquida las retracciones de la bola.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:forms:1}
En $\R^3$, sean $\omega = x\,\dd y - z\,\dd x$ y $\eta =
\dd x + y\,\dd z$. Calcúlense $\omega\wedge\eta$, $\dd\omega$, $\dd\eta$
y $\dd(\omega\wedge\eta)$, y verifíquese la regla de Leibniz graduada en
este ejemplo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:forms:2}
Identifíquense, en $\R^3$, las tres encarnaciones de $\dd$: para
$f \in \Omega^0$, $\dd f \leftrightarrow \nabla f$; para la forma de
trabajo $\omega_F = F_1\dd x + F_2\dd y + F_3\dd z$,
$\dd\omega_F \leftrightarrow \operatorname{curl}F$; para la forma de
\omterm{ex:b3:ode:planeclassification}{flujo} $\sigma_F = F_1\,\dd y\wedge\dd z + F_2\,\dd
z\wedge\dd x + F_3\,\dd x\wedge\dd y$, $\dd\sigma_F
\leftrightarrow \operatorname{div}F$. Dedúzcanse de $\dd^2 =
0$ las identidades $\operatorname{curl}\nabla f = 0$ y
$\operatorname{div}\operatorname{curl}F = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:forms:3}
Decídase si cada $1$-forma es \omterm{def:b3:forms:closedexact}{cerrada}
o \omterm{def:b3:forms:closedexact}{exacta} en su dominio, y calcúlese
una primitiva cuando exista: (a)
$(2xy + z^2)\,\dd x + x^2\,\dd y + 2xz\,\dd z$ en
$\R^3$;
(b) $\dfrac{x\,\dd x + y\,\dd y}{x^2 + y^2}$ en
$\R^2\setminus\{0\}$;
(c) $\dfrac{-y\,\dd x + x\,\dd y}{x^2 + y^2}$ en el semiplano
$\{x > 0\}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:forms:4}
(La forma angular) Verifíquese que $\omega_\theta$
(el~\cref{ex:b3:forms:angular}) es
\omterm{def:b3:forms:closedexact}{cerrada}; calcúlese
$\int_\gamma\omega_\theta$ para $\gamma$ la circunferencia de radio $r$
alrededor de $0$; conclúyase que $\omega_\theta$ no es
\omterm{def:b3:forms:closedexact}{exacta} en $\R^2\setminus\{0\}$ y que
$\R^2\setminus\{0\}$ no es estrellado respecto de ninguno de sus puntos
(dos vías: por el~\cref{thm:b3:forms:poincare} y directamente por la
geometría).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:forms:5}
(Forma de área de una hipersuperficie) Sea $M \subseteq \R^n$ una
hipersuperficie orientada compacta
cuya \omterm{def:b3:forms:orientation}{orientación} viene dada por un
campo normal unitario $\nu$ ($(w_1, \dots, w_{n-1})$ positiva si y solo
si $(\nu, w_1, \dots, w_{n-1})$ es positiva en $\R^n$). Demuéstrese que
la $(n-1)$-forma $\sigma_\nu =
\sum_i(-1)^{i-1}\nu_i\,\dd x_1\wedge\dots\wedge\widehat{\dd
x_i}\wedge\dots\wedge\dd x_n$ se restringe en $M$ a la \emph{forma de
área}: para una parametrización directa $\gamma$,
$\gamma^*\sigma_\nu = \sqrt{\det G}\,\dd u_1\wedge\dots
\wedge\dd u_{n-1}$, donde $G = ({}^t D\gamma)(D\gamma)$ es la matriz de
Gram. \emph{(Obsérvese que
$(\gamma^*\sigma_\nu)(e_1, \dots, e_{n-1}) = \det(\nu,
\partial_1\gamma, \dots, \partial_{n-1}\gamma)$ y elévese al cuadrado
este determinante.)} Calcúlese $\int_{S^2}\sigma_\nu = 4\pi$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:forms:6}
Calcúlese $\int_{S^2}\omega$ para $\omega = x\,\dd y\wedge\dd z
+ y\,\dd z\wedge\dd x + z\,\dd x\wedge\dd y$: (a) directamente en
coordenadas esféricas; (b) por Stokes en la bola unidad. Dedúzcase
$\operatorname{vol}(B^3) = \frac{4\pi}3$ a partir del área de $S^2$, y
generalícese:
$n\operatorname{vol}(B^n) = \operatorname{area}(S^{n-1})$,
en consonancia con el~\cref{thm:b3:product:ballvolume}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:forms:7}
(Los diccionarios) Dedúzcanse cuidadosamente
del~\cref{thm:b3:forms:stokes}: (a) el teorema de la divergencia en
$\R^3$ (combínense el~\cref{exo:b3:forms:2} y el~\cref{exo:b3:forms:5});
(b) el teorema de Kelvin--Stokes
$\int_S\langle\operatorname{curl}F, \nu\rangle\,\dd S =
\oint_{\partial S}\langle F, \tau\rangle\,\dd\ell$ para una superficie
orientada compacta con
\omterm{def:b3:forms:boundary}{borde} en $\R^3$. Compruébese que los
convenios de \omterm{def:b3:forms:orientation}{orientación} concuerdan
en la semiesfera superior limitada por el ecuador.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:forms:8}
(Identidades de Green) Para $u, v$ regulares en un
\omterm{def:b3:topology:topology}{entorno} de un dominio
\omterm{def:b3:topology:compact}{compacto} $\Omega \subseteq \R^n$ de
\omterm{def:b3:forms:boundary}{borde} regular, demuéstrese
\[
\int_\Omega\bigl(u\,\Delta v + \langle\nabla u, \nabla
v\rangle\bigr) = \int_{\partial\Omega}u\,\partial_\nu
v\,\dd S,
\qquad
\int_\Omega\bigl(u\,\Delta v - v\,\Delta u\bigr) =
\int_{\partial\Omega}\bigl(u\,\partial_\nu v -
v\,\partial_\nu u\bigr)\dd S .
\]
Dedúzcase: una \omterm{prop:b3:conformal:harmonicholo}{función armónica}
en $\Omega$ que se anula en $\partial\Omega$ es idénticamente nula, y
dos funciones armónicas con los
mismos valores en el \omterm{def:b3:forms:boundary}{borde} coinciden ---
unicidad en el problema de Dirichlet del~
\cref{ch:b3:conformal}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:forms:9}
(Cambio de variables, forma orientada) Sean $\varphi\colon U
\to V$ un difeomorfismo de abiertos de $\R^n$ con $\det
D\varphi > 0$, y $f$ \omterm{def:b3:topology:continuity}{continua} de
soporte \omterm{def:b3:topology:compact}{compacto} en $V$. Demuéstrese
que la identidad de \omterm{def:b3:forms:pullback}{retroceso}
$\int_U\varphi^*(f\,\dd x_1\wedge\dots\wedge\dd x_n) =
\int_Vf\,\dd x_1\wedge\dots\wedge\dd x_n$ es equivalente al teorema del
cambio de variables (el~\cref{thm:b3:product:changeofvar}) para tales
$\varphi$, y explíquese exactamente adónde fue a parar el valor absoluto
del jacobiano.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:forms:10}
(Deformación) Sean $\omega$ una $2$-forma
\emph{\omterm{def:b3:forms:closedexact}{cerrada}} en
$\R^3\setminus\{0\}$ y $S_r$ la esfera de radio $r$ centrada en $0$.
Demuéstrese que $\int_{S_r}\omega$ no depende de $r > 0$
\emph{(aplíquese Stokes a la corona esférica entre dos radios; atención
a las dos orientaciones del
\omterm{def:b3:forms:boundary}{borde})}. Aplíquese a la \emph{forma de
ángulo sólido}
\[
\omega = \frac{x\,\dd y\wedge\dd z + y\,\dd z\wedge\dd x +
z\,\dd x\wedge\dd y}{(x^2 + y^2 + z^2)^{3/2}} :
\]
compruébese que es \omterm{def:b3:forms:closedexact}{cerrada}, calcúlese
$\int_{S_r}\omega = 4\pi$ y conclúyase que es
\omterm{def:b3:forms:closedexact}{cerrada} pero no
\omterm{def:b3:forms:closedexact}{exacta} en $\R^3\setminus\{0\}$ --- la
hermana bidimensional de $\omega_\theta$ y el contenido geométrico de la
ley de Gauss en electrostática.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:forms:11}
Sea $\gamma$ una curva \omterm{def:b3:forms:closedexact}{cerrada}
regular en $\R^2\setminus\{0\}$ con $n = \operatorname{Ind}_\gamma(0)$.
Demuéstrese $\int_\gamma\omega = n\int_{S^1}\omega$ para \emph{toda}
$1$-forma \omterm{def:b3:forms:closedexact}{cerrada} $\omega$ en
$\R^2\setminus\{0\}$ \emph{(escríbase $\omega = c\,\omega_\theta
+ \dd f$ por el~\cref{exo:b3:forms:12})}. Interpretación: en el plano
punteado, el \omterm{def:b3:forms:winding}{índice} es la única
obstrucción a la anulación de los periodos.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:forms:12}
(Primer cálculo de de Rham) Demuéstrese que toda
\omterm{def:b3:forms:closedexact}{cerrada} $1$-forma $\omega$ en
$U = \R^2\setminus\{0\}$ se escribe de manera única
\[
\omega = c\,\omega_\theta + \dd f,
\qquad c = \frac1{2\pi}\int_{S^1}\omega,\quad f \in
\mathcal C^\infty(U) :
\]
defínase $f(p)$ integrando $\omega - c\,\omega_\theta$ a lo largo de un
\omterm{def:b3:topology:pathconnected}{camino} de $(1,0)$ a $p$ (un
tramo radial y después un arco de circunferencia), demuéstrese que el
resultado no depende de las elecciones precisamente porque el periodo de
$S^1$ se anula, y compruébese $\dd
f = \omega - c\,\omega_\theta$. Conclúyase:
$H^1(\R^2\setminus\{0\}) \cong \R$, generado por la forma angular.
\end{exercise}

\section{Problema: el teorema del punto fijo de Brouwer}

\begin{problem}[{Problema de fin de semana --- sin retracción, sin
escapatoria}]\label{pb:b3:forms:1} El teorema de Brouwer afirma que
\emph{toda \omterm{def:b3:topology:continuity}{aplicación continua} de
la bola unidad \omterm{def:b3:forms:closedexact}{cerrada}
$\bar B = \bar B^n \subseteq \R^n$ en sí misma tiene un punto fijo} ---
uno de los grandes teoremas de la matemática, con consecuencias desde la
teoría de juegos (equilibrios de Nash) hasta el análisis matricial. La
demostración con \omterm{def:b3:forms:diffform}{formas diferenciales} es la más limpia que se conoce:
Stokes muestra que la esfera no es un retracto de la bola, y todo lo
demás se sigue. En todo el problema, $S = S^{n-1} = \partial\bar B$,
$n \geq 2$ y
\[
\sigma = \sum_{i=1}^n(-1)^{i-1}x_i\,
\dd x_1\wedge\dots\wedge\widehat{\dd x_i}\wedge\dots\wedge
\dd x_n
\]
(el acento circunflejo suprime el factor).

\medskip
\noindent\textbf{Parte I --- El instrumento de \omterm{def:b3:measure:measure}{medida}.}
\begin{enumerate}
  \item Calcular $\dd\sigma$ y dedúzcase de Stokes
  (el~\cref{thm:b3:forms:stokes}) que $\int_S\sigma =
        n\operatorname{vol}(\bar B) > 0$, donde $S$ lleva la
  \omterm{def:b3:forms:orientation}{orientación} de
  \omterm{def:b3:forms:boundary}{borde} de la bola. \item Para $n = 2$ y
  $n = 3$, identifíquese la restricción de $\sigma$ a $S$ con las formas
  de longitud de arco y de área (el~\cref{exo:b3:forms:5} con
  $\nu(x) = x$) y recalcúlese $\int_S\sigma$ directamente. \item Sean
  $W \subseteq \R^N$ abierto y $\varphi\colon W
        \to \R^n$ regular con $\norm{\varphi(x)} = 1$ para todo
  $x \in W$. Demuéstrese que $\varphi^*(\dd\sigma) = 0$. \emph{(Derívese
  $\norm\varphi^2 = 1$: la imagen de $D\varphi(x)$ está en el hiperplano
  $\varphi(x)^\perp$, de dimensión $n - 1$; aplíquese después
        \cref{prop:b3:forms:linearpullback}(2).)}
  \item ¿Dónde está el fallo de la siguiente «demostración» de que
  $\int_S\sigma = 0$: «$S$ es compacta
  sin \omterm{def:b3:forms:boundary}{borde}, y $\sigma$ restringida a
  $S$ es una forma de grado máximo en ella, luego
  \omterm{def:b3:forms:closedexact}{cerrada}, luego $\int_S\sigma =
        \int_S\dd(\text{algo}) = 0$ por Stokes»? (Señálese la
  palabra equivocada.) \item Explíquese en un párrafo la estrategia de
  la Parte II: qué se integrará, sobre qué, y de dónde vendrá la
  contradicción.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- No hay retracción regular.} Supóngase, por
reducción al absurdo, que $r$ es una \emph{retracción regular} de la
bola sobre su esfera: $r$ es regular en un
\omterm{def:b3:topology:topology}{entorno} de $\bar B$,
$r(\bar B) \subseteq S$ y $r(x) = x$ para todo
$x \in S$.
\begin{enumerate}[resume]
  \item Justifíquese $\int_Sr^*\sigma = \int_S\sigma$ \emph{(en $S$, $r$
  se restringe a la identidad: si $\gamma$ es una parametrización
  directa de un trozo de $S$, entonces $r\circ\gamma = \gamma$)}. \item
  Usando Stokes en $\bar B$, demuéstrese $\int_Sr^*\sigma =
        \int_{\bar B}\dd(r^*\sigma)$. \item Demuéstrese
  $\dd(r^*\sigma) = r^*(\dd\sigma) = 0$ (pregunta 3 aplicada a
  $\varphi = r$) y conclúyase: \emph{no existe ninguna retracción
  regular $\bar B \to S$.} \item Resuélvase a mano el caso excluido
  $n = 1$: demuéstrese directamente que ninguna
  \omterm{def:b3:topology:continuity}{aplicación continua}
  $\intcc{-1}1 \to
        \{-1, 1\}$ fija ambos extremos, y nómbrese el teorema utilizado.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Brouwer regular.} Sea $g$ regular en un
\omterm{def:b3:topology:topology}{entorno} de $\bar B$ con $g(\bar B)
\subseteq \bar B$ y \emph{sin} ningún punto fijo en $\bar B$.
\begin{enumerate}[resume]
  \item Demuéstrese $\delta = \min_{x\in\bar B}\norm{g(x) - x} >
        0$. \item Para $x \in \bar B$, sea $u(x) = \frac{x -
        g(x)}{\norm{x - g(x)}}$ y sea $r(x) = x +
        t(x)\,u(x)$ la intersección del rayo $\{x + tu(x) : t \geq 0\}$
  con $S$. Resuélvase la ecuación de segundo grado $\norm{x + tu}^2 = 1$
  y obténgase
        \[
        t(x) = -\langle x, u(x)\rangle +
        \sqrt{1 - \norm x^2 + \langle x, u(x)\rangle^2}
        \;\geq\; 0 .
        \]
  \item Demuéstrese que el radicando es \emph{estrictamente} positivo en
  $\bar B$: si $1 - \norm x^2 + \langle x,
        u(x)\rangle^2 = 0$, entonces $\norm x = 1$ y $\langle
        x, u(x)\rangle = 0$, es decir, $\langle x, x -
        g(x)\rangle = 0$, o sea, $\langle x, g(x)\rangle =
        1$; por Cauchy--Schwarz con $\norm x = 1$, $\norm{g(x)} \leq 1$,
  esto fuerza $g(x) = x$ --- excluido. Dedúzcase que $r$ es regular en
  un \omterm{def:b3:topology:topology}{entorno} de $\bar B$. \item
  Demuéstrense $r(\bar B) \subseteq S$ y $r(x) = x$ para $x
        \in S$ \emph{(para $\norm x = 1$, compruébese $t(x) = 0$ usando
  $\langle x, u(x)\rangle \geq 0$, que a su vez se sigue de
  $\langle x, x - g(x)\rangle = 1 -
        \langle x, g(x)\rangle \geq 0$)}. Conclúyase con la Parte II:
  \emph{toda autoaplicación regular de $\bar B$ tiene un punto fijo.}
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Brouwer
\omterm{def:b3:topology:continuity}{continuo}.} Sea
$f\colon\bar B\to\bar B$ \omterm{def:b3:topology:continuity}{continua}
sin puntos fijos.
\begin{enumerate}[resume]
  \item Demuéstrese $\varepsilon = \min_{\bar B}\norm{f - x} > 0$ y
  prodúzcase una aplicación \emph{polinómica} $p\colon\R^n\to
        \R^n$ con $\sup_{\bar B}\norm{p - f} <
        \varepsilon/2$ (Stone--Weierstrass,
  el~\cref{thm:b3:complete:stoneweierstrass}, coordenada a coordenada
  --- justifíquese el paso de la aproximación escalar a la vectorial).
  \item La aplicación $p$ puede salirse de la bola; póngase
  $g = \frac{p}{1
        + \varepsilon/2}$. Demuéstrense $g(\bar B) \subseteq \bar B$ y
  $\sup_{\bar B}\norm{g - f} < \varepsilon$. \item Dedúzcase una
  contradicción con la Parte III y conclúyase: \emph{toda
  \omterm{def:b3:topology:continuity}{aplicación continua}
  $\bar B^n \to \bar B^n$ tiene un punto fijo.} \item Muéstrese con
  ejemplos que el teorema falla en: la bola abierta; la esfera $S$; una
  corona \omterm{def:b3:forms:closedexact}{cerrada}. ¿Qué propiedad de
  $\bar B$ pierde cada contraejemplo?
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos.}
\begin{enumerate}[resume]
  \item (Perron--Frobenius, existencia) Sean $A$ una matriz $n\times n$
  con todas sus entradas $> 0$ y
  $\Delta = \{x \in \R^n : x_i \geq 0,\ \sum x_i =
        1\}$. Demuéstrese que la aplicación $x \mapsto Ax/\norm{Ax}_1$
  está bien definida y es \omterm{def:b3:topology:continuity}{continua}
  en $\Delta$, que $\Delta$ es
  \omterm{def:b3:topology:continuity}{homeomorfo} a una bola
  \omterm{def:b3:forms:closedexact}{cerrada} de $\R^{n-1}$
  (\omterm{def:b3:topology:continuity}{homeomorfismo} radial desde un
  \omterm{def:b3:topology:compact}{compacto} convexo con
  \omterm{def:b3:topology:interior}{interior} no vacío en su envoltura
  afín) y conclúyase que $A$ tiene un vector propio de entradas
  estrictamente positivas y valor propio $> 0$. \item Dedúzcase que toda
  matriz estocástica de entradas positivas (con columnas que suman $1$)
  tiene un vector de probabilidad estacionario $\pi = A\pi$ --- el
  vector de tipo PageRank. (La unicidad también vale, pero exige otras
  herramientas.) \item (Bola peluda, preparación) Sea $v$ regular en un
  \omterm{def:b3:topology:topology}{entorno} de $S = S^{n-1}$ con
  $\langle v(x),
        x\rangle = 0$ y $\norm{v(x)} = 1$ para $x \in S$ (un \emph{campo
  tangente unitario}). Para $t \in \R$, póngase $F_t(x) = x + t\,v(x)$.
  Demuéstrese $\norm{F_t(x)} = \sqrt{1 + t^2}$ en $S$: $F_t$ lleva $S$
  dentro de la esfera $\sqrt{1+t^2}\,S$. \item Demuéstrese que
  $P(t) = \int_SF_t^*\sigma$ es un \emph{polinomio} en $t$ \emph{(cada
  función coeficiente de $F_t^*\sigma$ en una carta es polinómica en
  $t$, con coeficientes regulares en la variable de la carta; la
  integración es lineal)}. \item Demuéstrese que, para $\abs t$ pequeño,
  $F_t$ es un difeomorfismo de $S$ sobre $\sqrt{1+t^2}\,S$: inyectividad
  para $t\operatorname{Lip}(v) < 1$; difeomorfismo local por el teorema
  de la función inversa (el~\cref{thm:b3:submanifolds:ift} en cartas);
  imagen abierta y \omterm{def:b3:forms:closedexact}{cerrada} en la
  esfera de llegada, que es \omterm{def:b3:topology:connected}{conexa}.
  Dedúzcase, usando el~\cref{lem:b3:forms:consistency} y el reescalado
  $\sigma_{\lambda x} =
        \lambda^{n}\,\sigma_x$ bajo $x \mapsto \lambda x$ (compruébese),
  que
        \[
        P(t) = \pm(1 + t^2)^{n/2}\int_S\sigma,
        \qquad\text{con el signo } + \text{ para } t \text{ pequeño}
        \]
        (\omterm{def:b3:forms:orientation}{orientación} conservada por
        \omterm{def:b3:topology:continuity}{continuidad} desde $t = 0$).
        \item Conclúyase (Milnor): si $n$ es \emph{impar}, $(1 +
        t^2)^{n/2}$ no es un polinomio en $t$ y, sin embargo, coincide
        con el polinomio $P(t)/\int_S\sigma$ cerca de $0$ ---
        contradicción. Por tanto, \emph{las esferas de dimensión par
        $S^{n-1}$ ($n$ impar) no admiten ningún campo tangente unitario}
        y, normalizando y regularizando (convoluciónese componente a
        componente y proyéctese --- justifíquense ambos pasos), tampoco
        ningún campo tangente
        \omterm{def:b3:topology:continuity}{continuo} que no se anule:
        todo viento sobre la Tierra deja un punto en calma.

  \item (Las esferas impares se peinan sin dificultad) Exhíbase en
  $S^{2m-1}
        \subseteq \R^{2m} \cong \C^m$ un campo tangente unitario regular
  explícito: $v(x) = \iu x$ en notación compleja, es decir,
        \[
        v(x_1, y_1, \dots, x_m, y_m) =
        (-y_1, x_1, \dots, -y_m, x_m) .
        \]
        Verifíquense la tangencia y la longitud unidad, y conclúyase que
        la dicotomía de paridad de la pregunta 23 es óptima: una esfera
        es peinable exactamente cuando su dimensión es impar. ¿Dónde se
        rompe el argumento polinómico de la pregunta 22 para $n$ par?
        \item (Ceros a partir del comportamiento en el
        \omterm{def:b3:forms:boundary}{borde}) Sea $f \colon \bar
        B^n \to \R^n$ \omterm{def:b3:topology:continuity}{continua} con
        $\langle f(x),
        x\rangle \geq 0$ para todo $x \in S^{n-1}$. Demuéstrese que $f$
        se anula en algún punto de $\bar B^n$. \emph{(Si no,
        $g(x) = -f(x)/\norm{f(x)}$ lleva $\bar B$
        \omterm{def:b3:topology:continuity}{continuamente} dentro de
        $S \subseteq \bar B$; aplíquese Brouwer a $g$ y contradígase la
        hipótesis sobre el \omterm{def:b3:forms:boundary}{borde}.)}
        Dedúzcase el \emph{criterio de sobreyectividad}: una
        \omterm{def:b3:topology:continuity}{continua}
        $F\colon\R^n\to\R^n$ con
        $\frac{\langle F(x), x\rangle}{\norm x} \to
        +\infty$ cuando $\norm x \to \infty$ es sobreyectiva --- el
        antepasado en dimensión finita de los argumentos de coercividad
        del análisis no lineal.
\end{enumerate}
\end{problem}
