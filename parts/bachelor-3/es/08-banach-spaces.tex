\chapter{Espacios de Banach y los teoremas fundamentales}
\label{ch:b3:banach}

El análisis funcional estudia los espacios normados de dimensión
infinita a través de los operadores y las funcionales que viven sobre
ellos. Su descubrimiento fundacional es que la
\emph{\omterm{def:b3:complete:complete}{completitud}}, vía el teorema de
Baire, impone una fuerte uniformidad: las familias de operadores
puntualmente acotadas están acotadas en norma (Banach--Steinhaus), las
biyecciones \omterm{def:b3:topology:continuity}{continuas} tienen
inversa \omterm{def:b3:topology:continuity}{continua} (aplicación
abierta) y las gráficas detectan la
\omterm{def:b3:topology:continuity}{continuidad} (gráfica cerrada). El
otro pilar, \emph{Hahn--Banach}, no necesita
\omterm{def:b3:complete:complete}{completitud} alguna --- solo el lema
de Zorn --- y garantiza que los \omterm{def:b3:banach:operator}{espacios
duales} son lo bastante ricos como para ver todo vector. Este capítulo
demuestra los cuatro teoremas y los pone a prueba sobre los espacios de
sucesiones clásicos $\ell^p$, sobre cálculos concretos de
\omterm{def:b3:banach:operator}{duales} y sobre una aplicación
genuinamente sorprendente: existen funciones
\omterm{def:b3:topology:continuity}{continuas} $2\pi$-periódicas cuya
serie de Fourier \emph{diverge} en un punto --- lo que resuelve en
sentido negativo una cuestión que segundo año dejó abierta.

En todo el capítulo, $E, F$ son espacios normados sobre $K = \R$ o $\C$;
\emph{espacio de Banach}\index{espacio de Banach} significa espacio
normado \omterm{def:b3:complete:complete}{completo}.

\section{Operadores acotados; espacios de sucesiones}

\begin{definition}\label{def:b3:banach:operator}
$\mathcal L(E, F)$ denota el espacio de las aplicaciones lineales
\emph{acotadas} (= \omterm{def:b3:topology:continuity}{continuas},
segundo año) con la \emph{norma de operador}\index{norma de operador}
$\vertiii T = \sup_{\norm x \leq
1}\norm{Tx}$; es submultiplicativa: $\vertiii{ST} \leq
\vertiii S\,\vertiii T$. El \emph{dual}\index{espacio dual} es
$E' = \mathcal L(E, K)$.
\end{definition}

\begin{proposition}\label{prop:b3:banach:llcomplete}
Si $F$ es un espacio de Banach, también lo es $\mathcal L(E, F)$; en
particular, $E'$ siempre es un espacio de Banach.
\end{proposition}

\begin{proof}
Sea $(T_n)$ de Cauchy para $\vertiii\cdot$. Para cada $x$,
$\norm{T_nx - T_mx} \leq \vertiii{T_n - T_m}\norm x$: $(T_nx)$
es de Cauchy en $F$ y converge; llamemos $Tx$ a su límite. $T$ es lineal
(límites de identidades lineales); pasando al límite en
$\norm{T_nx - T_mx} \leq \varepsilon\norm x$ ($n, m \geq N$)
se obtiene $\norm{T_nx - Tx} \leq \varepsilon\norm x$: $T_n \to T$ en
\omterm{def:b3:banach:operator}{norma de operador}, y
$\vertiii T \leq \vertiii{T_N} +
\varepsilon < \infty$.
\end{proof}

\begin{definition}\label{def:b3:banach:lp}
Los espacios de sucesiones clásicos\index{espacios de sucesiones
$\ell^p$} (sobre $K$, indexados por $\N$):
\[
\ell^p = \Bigl\{x = (x_n) : \norm x_p = \Bigl(\sum_n
\abs{x_n}^p\Bigr)^{1/p} < \infty\Bigr\}\quad (1 \leq p <
\infty),
\]
\[
\ell^\infty = \{x : \norm x_\infty = \sup\abs{x_n} < \infty\},
\]
y $c_0 = \{x : x_n \to 0\}$ con $\norm\cdot_\infty$. Que $\norm\cdot_p$
sea una norma se sigue de la desigualdad de Minkowski, demostrada en el
caso discreto exactamente igual que en el~\cref{ch:b3:lp} (o sumando la
desigualdad de dimensión finita de segundo año). Todos son espacios de
Banach, y $c_0$ es un subespacio cerrado de
$\ell^\infty$ (\cref{exo:b3:banach:3}).
\end{definition}

\begin{proposition}[Serie de Neumann]\label{prop:b3:banach:neumann}
Sea $E$ de Banach y sea $T \in \mathcal L(E) = \mathcal L(E,E)$ con
$\vertiii T < 1$. Entonces $I - T$ es invertible en $\mathcal
L(E)$, con $(I - T)^{-1} = \sum_{n\geq0}T^n$ (convergente en
\omterm{def:b3:banach:operator}{norma de operador}). En consecuencia, el
conjunto de los operadores invertibles es abierto y la inversión es
\omterm{def:b3:topology:continuity}{continua} en él.\index{serie de
Neumann}
\end{proposition}

\begin{proof}
La serie converge absolutamente ($\vertiii{T^n} \leq \vertiii
T^n$, geométrica) en el espacio de Banach $\mathcal L(E)$
(\cref{prop:b3:banach:llcomplete};
el~\cref{exo:b3:complete:1}(b)). Telescopando, $(I - T)\sum_{n \leq
N}T^n = I - T^{N+1} \to I$, y análogamente por el otro lado. Para el
\omterm{def:b3:representations:character}{carácter} abierto: si $S$ es invertible y $\vertiii{H} <
1/\vertiii{S^{-1}}$, entonces $S + H = S(I + S^{-1}H)$ con
$\vertiii{S^{-1}H} < 1$: es invertible.
\omterm{def:b3:topology:continuity}{Continuidad} de la inversión: la
expresión en serie da $\vertiii{(S+H)^{-1} - S^{-1}} =
O(\vertiii H)$ localmente.
\end{proof}

\begin{example}[Una ecuación de Volterra, resuelta por Neumann]
\label{ex:b3:banach:volterraexample}
Sobre $E = \mathcal C(\intcc01)$, considérese la ecuación integral
\[
u(x) = 1 + \lambda\int_0^xu(t)\,\dd t,
\qquad\text{es decir}\qquad u = \mathbf 1 + \lambda Tu,
\quad (Tu)(x) = \int_0^xu .
\]
Aquí $\vertiii T \leq 1$, de modo que para $\abs\lambda < 1$ la
\omterm{prop:b3:banach:neumann}{serie de Neumann} se aplica
directamente: $u = (I - \lambda T)^{-1}\mathbf 1 =
\sum_n\lambda^nT^n\mathbf 1$. Calculando, $T^n\mathbf 1 =
\frac{x^n}{n!}$, luego
\[
u(x) = \sum_{n\geq0}\frac{(\lambda x)^n}{n!} =
\eu^{\lambda x} ,
\]
como confirma la derivación. Mejor aún: $\vertiii{T^n} \leq
\frac1{n!}$ (el núcleo iterado decrece factorialmente), de modo que
$\sum\lambda^nT^n$ converge para \emph{todo} $\lambda$ --- el operador
$I - \lambda T$ es invertible para todos los $\lambda
\in \C$, aunque $\vertiii{\lambda T} \geq 1$ acabe siendo grande: lo que
importa es el decrecimiento espectral de las potencias, no la primera
norma. Los operadores de Volterra llevan incorporado ese decrecimiento
factorial (el~\cref{exo:b3:complete:4}(a) explotaba exactamente esto), y
por eso los problemas de valores iniciales nunca padecen los fenómenos
de resonancia de los problemas de contorno
(\cref{ch:b3:spectral}).
\end{example}

\section{Hahn--Banach}

\begin{theorem}[Hahn--Banach, forma analítica]
\label{thm:b3:banach:hahnbanach}
Sean $E$ un espacio vectorial \emph{real}, $p \colon E \to \R$
\emph{sublineal} ($p(x + y) \leq p(x) + p(y)$ y $p(tx) =
tp(x)$ para $t \geq 0$), $F \subseteq E$ un subespacio y $f
\colon F \to \R$ lineal con $f \leq p$ sobre $F$. Entonces $f$ se
extiende a una aplicación lineal $\tilde f \colon E \to \R$ con
$\tilde f \leq p$ sobre $E$.\index{teorema de Hahn--Banach}
\end{theorem}

\begin{proof}
\emph{Extensión en un paso.} Sea $x_0 \notin F$; extendemos $f$ a
$F \oplus \R x_0$ eligiendo $\alpha = \tilde f(x_0)$ adecuadamente:
necesitamos que, para todos $y \in F$, $t > 0$,
\[
f(y) + t\alpha \leq p(y + tx_0)
\quad\text{y}\quad
f(y) - t\alpha \leq p(y - tx_0),
\]
lo que, tras dividir por $t$ (sublinealidad), se reduce a
\[
\sup_{v \in F}\ \bigl[f(v) - p(v - x_0)\bigr]
\;\leq\; \alpha \;\leq\;
\inf_{u \in F}\ \bigl[p(u + x_0) - f(u)\bigr].
\]
Un tal $\alpha$ existe si y solo si todo miembro izquierdo es $\leq$ que
todo miembro derecho: en efecto,
$f(v) + f(u) = f(u + v) \leq p(u + v) \leq
p(u + x_0) + p(v - x_0)$, es decir, $f(v) - p(v - x_0) \leq p(u +
x_0) - f(u)$.

\emph{Zorn.} Ordénense por extensión las extensiones de $f$ dominadas
por $p$ (pares: subespacio y funcional); una cadena tiene la unión por
cota superior; y un elemento maximal ha de estar definido en todo $E$,
pues en caso contrario la extensión en un paso contradiría la
maximalidad.
\end{proof}

\begin{corollary}\label{cor:b3:banach:hbnormed}
Sea $E$ un espacio normado ($K = \R$ o $\C$).
\begin{enumerate}
  \item Toda $f \in F'$ (con $F$ un subespacio) se extiende a $\tilde f
        \in E'$ con $\norm{\tilde f}_{E'} = \norm f_{F'}$. \item Para
  todo $x \neq 0$ existe $f \in E'$ con $\norm f
        = 1$ y $f(x) = \norm x$. En particular, $E'$ separa los puntos
  de $E$, y $\norm x = \sup_{\norm f \leq
        1}\abs{f(x)}$. \item Para un subespacio cerrado $F$ y
  $x \notin F$, existe $f
        \in E'$ que se anula en $F$ con $f(x) = d(x, F)$ y
        $\norm f \leq 1$.
\end{enumerate}
\end{corollary}

\begin{proof}
(1) Caso real: aplíquese el~\cref{thm:b3:banach:hahnbanach} con $p(x)
= \norm f_{F'}\,\norm x$ (sublineal); la extensión cumple
$\pm\tilde f(x) = \tilde f(\pm x) \leq p(x)$, luego
$\norm{\tilde f} \leq \norm f$, y $\geq$ es la restricción. Caso
complejo: sea $u = \operatorname{Re}f$, una funcional real con
$\abs u \leq \norm f\norm\cdot$; obsérvese que $f(x) = u(x) -
\iu\,u(\iu x)$ (compruébese sobre las partes real e imaginaria:
$\operatorname{Im}f(x) = -\operatorname{Re}f(\iu x)$). Extiéndase $u$ de
manera real-lineal con la misma cota y póngase $\tilde f(x) =
\tilde u(x) - \iu\tilde u(\iu x)$: es $\C$-lineal (comprobación directa
sobre la multiplicación por $\iu$) y extiende $f$; norma: para $x$ dado,
escríbase $\tilde f(x) = r\eu^{\iu\theta}$, y entonces
$\abs{\tilde f(x)}
= \tilde f(\eu^{-\iu\theta}x) = \tilde u(\eu^{-\iu\theta}x)
\leq \norm f\,\norm x$.

(2) Sobre $F = Kx$ defínase $f(tx) = t\norm x$: tiene norma $1$ sobre
$F$; extiéndase por (1). La fórmula de dualidad: $\leq$ es claro, y
$\geq$ por esta $f$.

(3) Sobre $F \oplus Kx$ defínase $f(y + tx) = t\,d(x, F)$; entonces,
para $t \neq 0$, $\norm{y + tx} = \abs t\,\norm{x + y/t} \geq \abs
t\,d(x, F) = \abs{f(y + tx)}$: $\norm f \leq 1$ sobre el subespacio;
extiéndase por (1).
\end{proof}

\begin{remark}\label{rem:b3:banach:bidual}
Por (2), la aplicación canónica $J \colon E \to E''$, $J(x)(f) =
f(x)$, es una isometría (el~\cref{exo:b3:banach:10}): todo espacio
normado se aloja dentro de su bidual. Los espacios con $J$ sobreyectiva
se llaman \emph{reflexivos}\index{espacio reflexivo}; el problema de fin
de semana muestra que $\ell^p$ ($1 < p < \infty$) es
\omterm{rem:b3:banach:bidual}{reflexivo} mientras que $\ell^1$ no lo es.
\end{remark}

\section{La trilogía de Baire}

\begin{theorem}[Banach--Steinhaus, acotación uniforme]
\label{thm:b3:banach:banachsteinhaus}
Sean $E$ un espacio de \emph{Banach}, $F$ normado y $(T_i)_{i\in
I} \subseteq \mathcal L(E, F)$ una familia con $\sup_i
\norm{T_ix} < \infty$ para todo $x \in E$. Entonces $\sup_i
\vertiii{T_i} < \infty$.\index{teorema de Banach--Steinhaus}
\end{theorem}

\begin{proof}
Los conjuntos $F_n = \{x : \sup_i\norm{T_ix} \leq n\}$ son cerrados
(intersecciones de preimágenes de bolas cerradas) y recubren $E$. Baire
(el~\cref{thm:b3:complete:baire}) da $n_0$ y una bola
$B(x_0, r) \subseteq F_{n_0}$. Para $\norm z < r$: $\norm{T_iz}
\leq \norm{T_i(x_0 + z)} + \norm{T_ix_0} \leq 2n_0$, luego
$\vertiii{T_i} \leq 2n_0/r$ para todo $i$.
\end{proof}

\begin{corollary}\label{cor:b3:banach:bslimits}
Si $E$ es de Banach y las $T_n \in \mathcal L(E,F)$ convergen
\emph{puntualmente} ($T_nx \to Tx$ para cada $x$), entonces $\sup_n
\vertiii{T_n} < \infty$, $T \in \mathcal L(E, F)$, y
$\vertiii T \leq \liminf \vertiii{T_n}$.
\end{corollary}

\begin{proof}
Las sucesiones convergentes están acotadas: acotación puntual;
Banach--Steinhaus acota las normas por cierto $M$; entonces $\norm{Tx}
= \lim\norm{T_nx} \leq M\norm x$ ($T$ es lineal por ser límite puntual),
y la cota más fina se obtiene haciendo $\liminf$ en
$\norm{T_nx} \leq \vertiii{T_n}\norm x$.
\end{proof}

\begin{theorem}[Series de Fourier divergentes]
\label{thm:b3:banach:fourierdiverge}
Existen funciones \omterm{def:b3:topology:continuity}{continuas}
$2\pi$-periódicas $f$ cuya serie de Fourier diverge en $0$:
$\sup_N\abs{S_N(f)(0)} =
\infty$. De hecho, tales $f$ forman un subconjunto denso de $\mathcal
C(S^1)$.\index{serie de Fourier, divergencia}
\end{theorem}

\begin{proof}
Trabajemos en $E = (\mathcal C(S^1), \norm\cdot_\infty)$, un espacio de
Banach, con las funcionales $\Lambda_N(f) = S_N(f)(0) =
\frac1{2\pi}\int_{-\pi}^{\pi}f(t)\,D_N(t)\,\dd t$ (núcleo de Dirichlet,
segundo año). Cada $\Lambda_N$ es
\omterm{def:b3:topology:continuity}{continua} con
\[
\norm{\Lambda_N} = \frac1{2\pi}\int_{-\pi}^\pi\abs{D_N(t)}\,\dd
t \;=\; L_N .
\]
($\leq$ es claro; $\geq$: tómese $f$
\omterm{def:b3:topology:continuity}{continua}, $\norm f_\infty
\leq 1$, que aproxime $\operatorname{sign}D_N$ --- el signo tiene un
número finito de saltos; suavizando cada salto en un intervalo de
longitud $\varepsilon$, la integral varía en $O(N\varepsilon)$.) Las
\emph{constantes de Lebesgue} $L_N$ tienden a infinito:
\[
L_N = \frac1{2\pi}\int_{-\pi}^{\pi}
\frac{\abs{\sin\bigl((N{+}\tfrac12)t\bigr)}}{\abs{\sin(t/2)}}
\,\dd t
\geq \frac{2}{\pi}\int_0^{\pi}
\frac{\abs{\sin\bigl((N{+}\tfrac12)t\bigr)}}{t}\,\dd t
= \frac{2}{\pi}\int_0^{(N+\frac12)\pi}\frac{\abs{\sin
u}}{u}\,\dd u
\]
(usando $\abs{\sin(t/2)} \leq t/2$ en $[0, \pi]$ y sustituyendo después
$u = (N + \tfrac12)t$). Cortando en arcos:
\[
\int_{(k-1)\pi}^{k\pi}\frac{\abs{\sin u}}u\,\dd u
\geq \frac1{k\pi}\int_{(k-1)\pi}^{k\pi}\abs{\sin u}\,\dd u =
\frac{2}{k\pi},
\qquad\text{luego}\qquad
L_N \geq \frac{4}{\pi^2}\sum_{k=1}^N\frac1k
\xrightarrow[N\to\infty]{} \infty .
\]
Si toda $f$ \omterm{def:b3:topology:continuity}{continua} cumpliera
$\sup_N
\abs{\Lambda_N(f)} < \infty$, Banach--Steinhaus obligaría a
$\sup_N\norm{\Lambda_N} < \infty$: contradicción. Luego alguna $f$ ---
de hecho, un \omterm{def:b3:topology:interior}{conjunto denso} y no
\omterm{rem:b3:complete:meagre}{magro} de $f$ (el complementario de
$\bigcup_M\{f: \sup_N\abs{\Lambda_Nf}\leq M\}$, unión numerable de
cerrados que, al no tener \omterm{def:b3:topology:interior}{interior}
por lo anterior aplicado en cualquier bola, es
\omterm{rem:b3:complete:meagre}{magro}) --- cumple
$\sup_N\abs{S_Nf(0)} =
\infty$.
\end{proof}

\begin{theorem}[Aplicación abierta]\label{thm:b3:banach:openmapping}
Sean $E, F$ espacios de Banach y sea $T \in \mathcal L(E, F)$
\emph{sobreyectivo}. Entonces $T$ es abierto: $T(B_E(0,1)) \supseteq
B_F(0, c)$ para cierto $c > 0$. En consecuencia, un operador acotado
biyectivo entre espacios de Banach tiene inversa acotada.\index{teorema
de la aplicación abierta}
\end{theorem}

\begin{proof}
Escribamos $B = B_E(0,1)$. La sobreyectividad da $F = \bigcup_n
\overline{T(nB)} = \bigcup_n n\,\overline{T(B)}$; Baire
(el~\cref{thm:b3:complete:baire}) da
\omterm{def:b3:topology:interior}{interior} a $\overline{T(B)}$: cierto
$B_F(y_0, 4c) \subseteq
\overline{T(B)}$. Recentrando en $0$:
para $\norm y < 4c$, tanto $y_0 + y$ como $y_0$ son límites de imágenes
$Tu_k$, $Tv_k$ con $u_k, v_k \in B$, luego $y = \lim T(u_k - v_k)$ con
$u_k - v_k
\in 2B$: $B_F(0, 4c) \subseteq \overline{T(2B)}$, es decir,
$B_F(0, 2c) \subseteq \overline{T(B)}$.

\emph{Eliminar la \omterm{def:b3:topology:interior}{clausura}} (aquí
interviene la \omterm{def:b3:complete:complete}{completitud} de $E$):
sea $\norm y < c$. Tómese $x_1 \in \frac12 B$ con $\norm{y -
Tx_1} < c/2$ ($B_F(0,2c) \subseteq \overline{T(B)}$ reescalado por
$\frac12$); inductivamente, $x_k \in 2^{-k}B$ con $\norm{y -
T(x_1 + \dots + x_k)} < c\,2^{-k}$. La serie $\sum x_k$ converge
absolutamente en el espacio de Banach $E$, hacia cierto $x \in B$ (de
norma $<
\sum 2^{-k} = 1$), y $Tx = y$ por
\omterm{def:b3:topology:continuity}{continuidad}: $B_F(0, c)
\subseteq T(B)$. El \omterm{def:b3:representations:character}{carácter} abierto de $T$ sobre abiertos arbitrarios
se sigue por traslación y homotecia; y para el corolario, que $T$ sea
abierta significa que $T^{-1}$ es
\omterm{def:b3:topology:continuity}{continua}.
\end{proof}

\begin{corollary}[Normas equivalentes]\label{cor:b3:banach:equivnorms}
Si un espacio vectorial es \omterm{def:b3:complete:complete}{completo}
para dos normas comparables ($\norm\cdot_a \leq C\norm\cdot_b$), las
normas son equivalentes.
\end{corollary}

\begin{proof}
La identidad $(E, \norm\cdot_b) \to (E, \norm\cdot_a)$ es acotada y
biyectiva entre espacios de Banach: su inversa es acotada.
\end{proof}

\begin{theorem}[Gráfica cerrada]\label{thm:b3:banach:closedgraph}
Sean $E, F$ de Banach y $T \colon E \to F$ lineal. Si la gráfica
$\Gamma = \{(x, Tx)\}$ es cerrada en $E \times F$ (es decir, si
$x_n \to x$ y $Tx_n \to y$ implican $y = Tx$), entonces $T$ es
acotado.\index{teorema de la gráfica cerrada}
\end{theorem}

\begin{proof}
$E \times F$ con $\norm{(x,y)} = \norm x + \norm y$ es de Banach;
$\Gamma$, subespacio cerrado, es de Banach. La proyección
$\pi_E\colon \Gamma \to E$ es acotada y biyectiva, de modo que su
inversa $x \mapsto (x, Tx)$ es acotada
(\cref{thm:b3:banach:openmapping}): $\norm{Tx} \leq \norm{(x,
Tx)} \leq C\norm x$.
\end{proof}

\begin{method}\label{met:b3:banach:trilogy}
Cuándo echar mano de cada teorema. \emph{Hahn--Banach}: para producir
una funcional con un comportamiento prescrito (que norme un vector, que
se anule en un subespacio, que extienda desde un subespacio) --- no hace
falta \omterm{def:b3:complete:complete}{completitud}.
\emph{Banach--Steinhaus}: para convertir información puntual en cotas
uniformes --- típicamente, para ver que una operación de paso al límite
es \omterm{def:b3:topology:continuity}{continua}, o bien (por
contrarrecíproco) para demostrar la \emph{divergencia} de algún
elemento, como con las series de Fourier. \emph{Aplicación abierta /
gráfica cerrada}: para obtener gratis la
\omterm{def:b3:topology:continuity}{continuidad} a partir de la
biyectividad \omterm{def:b3:galois:algebraic}{algebraica} o de una
propiedad de \omterm{def:b3:topology:interior}{clausura} de la gráfica
--- uso típico: comparar dos normas
\omterm{def:b3:complete:complete}{completas}, o demostrar la
\omterm{def:b3:topology:continuity}{continuidad} automática. Los tres
teoremas de Baire exigen la
\omterm{def:b3:complete:complete}{completitud} \emph{del espacio de
partida}; en caso contrario hay contraejemplos
(\cref{exo:b3:banach:7}).
\end{method}

\section{Espacios duales, en concreto}

\begin{theorem}\label{thm:b3:banach:duals}
Isométricamente: $(c_0)' \cong \ell^1$ y $(\ell^1)' \cong
\ell^\infty$, mediante el emparejamiento $\langle x, y\rangle = \sum_n
x_ny_n$.\index{duales de los espacios de sucesiones}
\end{theorem}

\begin{proof}
Demostramos $(c_0)' \cong \ell^1$; la segunda identificación es
el~\cref{exo:b3:banach:5}. A $y \in \ell^1$ asóciesele $\Lambda_y(x)
= \sum x_ny_n$ ($x \in c_0$): la serie converge absolutamente, con
$\abs{\Lambda_y(x)} \leq \norm x_\infty\norm y_1$, luego
$\norm{\Lambda_y} \leq \norm y_1$. Recíprocamente, sea $\Lambda \in
(c_0)'$; póngase $y_n = \Lambda(e_n)$ ($e_n$ son las sucesiones unidad).
Para $N$ cualquiera, evalúese en $x^{(N)} = \sum_{n \leq N}
\operatorname{sign}(\overline{y_n})\,e_n \in c_0$ (de norma $\leq
1$; en el caso complejo úsense factores unimodulares $\bar y_n/\abs
{y_n}$): $\Lambda(x^{(N)}) = \sum_{n\leq N}\abs{y_n} \leq
\norm\Lambda$. Luego $y \in \ell^1$ con $\norm y_1 \leq
\norm\Lambda$. Por último, $\Lambda = \Lambda_y$: ambas coinciden sobre
los $e_n$ y, por tanto, sobre las sucesiones finitas, que son densas en
$c_0$ (truncamiento:
$\norm{x - \sum_{n \leq N}x_ne_n}_\infty = \sup_{n > N}\abs{x_n}
\to 0$ precisamente porque $x_n \to 0$); y dos funcionales
\omterm{def:b3:topology:continuity}{continuas} que coinciden sobre un
\omterm{def:b3:topology:interior}{conjunto denso} son iguales. La
correspondencia es lineal, biyectiva e isométrica
($\norm{\Lambda_y} = \norm y_1$ a partir de las dos desigualdades).
\end{proof}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:banach:1}
Calcular las \omterm{def:b3:banach:operator}{normas de operador}: (a) de
los desplazamientos $S(x_1, x_2, \dots) = (0, x_1, x_2, \dots)$ y
$S^*(x_1, x_2, \dots) = (x_2, x_3, \dots)$ sobre $\ell^2$; (b) del
operador de multiplicación $M_a x = (a_nx_n)$ sobre $\ell^2$, para
$a \in \ell^\infty$; (c) de la funcional $\Lambda(f) = \int_0^{1/2}f -
\int_{1/2}^1f$ sobre $\mathcal C(\intcc01)$ --- demuéstrese que
$\norm\Lambda = 1$ y que la norma \emph{no} se alcanza.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:banach:2}
Sean $E$ de Banach, $T \in \mathcal L(E)$ invertible y $S$ con
$\vertiii{S - T} < 1/\vertiii{T^{-1}}$. Demostrar que $S$ es invertible
y estimar $\vertiii{S^{-1} - T^{-1}}$. Aplicación: si un sistema lineal
$Tx = b$ es \omterm{def:b3:groups:derived}{resoluble} con $T$
invertible, una perturbación suficientemente pequeña de $T$ lo mantiene
unívocamente \omterm{def:b3:groups:derived}{resoluble}, con una cota
cuantitativa del cambio de la solución.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:3}
(a) Demostrar que $\ell^1$, $\ell^\infty$ y $c_0$ son espacios de
Banach, y que $c_0$ es la \omterm{def:b3:topology:interior}{clausura} en
$\ell^\infty$ del espacio de las sucesiones finitas. (b) Demostrar que
$\ell^p \subseteq \ell^q$ con $\norm\cdot_q \leq
\norm\cdot_p$ para $1 \leq p \leq q \leq \infty$, y que la inclusión es
estricta.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:4}
Sean $F \subseteq E$ un subespacio cerrado y $x \notin F$. Usando
el~\cref{cor:b3:banach:hbnormed}, demostrar la fórmula de dualidad
\[
d(x, F) = \max\bigl\{\abs{f(x)} : f \in E',\ \norm f \leq 1,\
f\restriction_F = 0\bigr\}
\]
(obsérvese: un \emph{máximo}). Deducir que $F = \bigcap\{\ker f : f
\in E',\ f\restriction_F = 0\}$: los subespacios cerrados son
exactamente las intersecciones de núcleos de funcionales.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:5}
Demostrar que $(\ell^1)' \cong \ell^\infty$ isométricamente, siguiendo
el esquema del~\cref{thm:b3:banach:duals} (las sucesiones finitas son
densas en $\ell^1$). ¿Dónde se rompe el argumento para $(\ell^\infty)'$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:6}
Sea $E, F, G$ normado con $E$ de Banach, y sea $B \colon E \times
F \to G$ bilineal y \omterm{def:b3:topology:continuity}{continua} en
cada variable por separado. Demostrar que $B$ es (conjuntamente)
\omterm{def:b3:topology:continuity}{continua}: $\norm{B(x,y)} \leq
C\norm x\norm y$. \emph{(Aplíquese Banach--Steinhaus a la familia
$(B(\cdot, y))_{\norm y \leq 1}$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:7}
(a) Sobre $E = \mathcal C(\intcc01)$, compárense $\norm\cdot_\infty$ y
$\norm\cdot_1$: la identidad $(E, \norm\cdot_\infty) \to
(E, \norm\cdot_1)$ es acotada y biyectiva, pero su inversa no está
acotada. ¿Qué hipótesis del~\cref{cor:b3:banach:equivnorms} falla? (b)
Exhibir una aplicación lineal discontinua de un subespacio denso de
$\ell^2$ en $K$ (por ejemplo, sobre las sucesiones finitas) y explicar
por qué esto no contradice el teorema de la gráfica cerrada.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:banach:8}
(Hellinger--Toeplitz) Sea $T \colon \ell^2 \to \ell^2$ lineal (definido
en todo el espacio) y \emph{simétrico}: $\langle Tx,
y\rangle = \langle x, Ty\rangle$ para todos $x, y$, donde $\langle
x, y \rangle = \sum x_n\bar y_n$. Demostrar que $T$ es acotado.
\emph{(Gráfica cerrada: si $x_k \to x$ y $Tx_k \to z$, evalúese contra
$y$ arbitrario.)} Moraleja: los operadores simétricos no acotados ---
los hamiltonianos de la mecánica cuántica --- nunca pueden definirse en
todo el espacio.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:banach:9}
(Teorema de Pólya sobre cuadraturas) Para cada $n$, sea $\Lambda_n(f)
= \sum_{i=0}^{n} w_{i,n}f(x_{i,n})$ una fórmula de cuadratura sobre
$\mathcal C(\intcc01)$ ($x_{i,n} \in \intcc01$, $w_{i,n} \in
\R$). Demostrar que $\Lambda_n(f) \to \int_0^1f$ para \emph{toda} $f$
\omterm{def:b3:topology:continuity}{continua} si y solo si: (i)
$\Lambda_n(P) \to \int_0^1P$ para todo polinomio $P$ y (ii)
$\sup_n\sum_i\abs{w_{i,n}} <
\infty$. \emph{(Calcúlese $\norm{\Lambda_n}$; úsense Banach--Steinhaus y
Weierstrass.)} Compruébese que las fórmulas con pesos \emph{positivos}
exactas sobre las constantes cumplen (ii) automáticamente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:10}
Demostrar que $J \colon E \to E''$, $J(x)(f) = f(x)$, es una isometría
lineal (úsese el~\cref{cor:b3:banach:hbnormed}(2)) y que es sobreyectiva
cuando $\dim E < \infty$. Demostrar también que si $E'$ es separable,
también lo es $E$. \emph{(Tómense $x_n$ que casi normen una sucesión
densa de $E'$ y demuéstrese que su envoltura lineal cerrada es $E$,
mediante
\cref{cor:b3:banach:hbnormed}(3).)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:11}
(Espacios cociente) Sean $E$ un espacio de Banach y $F \subseteq
E$ un subespacio \emph{cerrado}. Sobre $E/F$ defínase
\[
\norm{\bar x} \;=\; d(x, F) = \inf_{y\in F}\norm{x - y} .
\]
(a) Demostrar que esto define bien una \emph{norma} sobre $E/F$ (¿dónde
interviene que $F$ sea cerrado?) y que la proyección $\pi
\colon E \to E/F$ cumple $\vertiii\pi \leq 1$ y lleva la bola unidad
abierta sobre la bola unidad abierta. (b) Demostrar que $E/F$ es
\omterm{def:b3:complete:complete}{completo}. \emph{(Úsese el criterio de
series del~\cref{exo:b3:complete:1}(b): dadas clases $\bar x_k$ con
$\sum\norm{\bar x_k} < \infty$, levántese cada una a $x_k \in E$ con
$\norm{x_k} \leq \norm{\bar x_k} + 2^{-k}$ y súmese en $E$.)} (c)
Calcúlese: para $E = c$ (sucesiones convergentes) y $F =
c_0$, demuéstrese que $c/c_0 \cong K$ isométricamente mediante
$\bar x \mapsto
\lim_nx_n$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:12}
(Proyecciones acotadas y subespacios complementados) Sean $E$ un espacio
de Banach y $P \colon E \to E$ lineal con $P^2 = P$ (una proyección
\omterm{def:b3:galois:algebraic}{algebraica}), $V = \operatorname{im}P$,
$W =
\ker P$. (a) Supóngase $P$ acotado. Demostrar que $V$ y $W$ son cerrados
y que $E = V \oplus W$ con la descomposición $x = Px + (x -
Px)$. (b) Recíprocamente, supóngase $E = V \oplus W$ con \emph{ambos}
$V,
W$ cerrados, y sea $P$ la proyección sobre $V$ paralelamente a $W$.
Demostrar que $P$ es acotado. \emph{(Gráfica cerrada: si $x_n \to x$ y
$Px_n \to z$, entonces $z \in V$, $x_n - Px_n \to x - z \in
W$, y la unicidad de la descomposición identifica $z =
Px$.)} (c) Deducir la \emph{equivalencia}: un subespacio $V$ admite una
proyección acotada si y solo si es cerrado y tiene un complemento
\omterm{def:b3:galois:algebraic}{algebraico} cerrado --- y obsérvese
(sin demostración) que existen subespacios cerrados sin esa propiedad
($c_0$ dentro de $\ell^\infty$ es el ejemplo clásico): los espacios de
Hilbert, donde $V^\perp$ siempre sirve (el~\cref{ch:b3:hilbert}), son la
excepción, no la regla.
\end{exercise}

\section{Problema: la dualidad de los espacios
\texorpdfstring{$\ell^p$}{lp}}

\begin{problem}[{Problema de fin de semana --- $(\ell^p)' = \ell^q$,
reflexividad y la rareza de $\ell^\infty$}]
\label{pb:b3:banach:1}
Fíjese $1 < p < \infty$ y sea $q$ el exponente conjugado,
$\frac1p + \frac1q = 1$. El emparejamiento en todo el problema es
$\langle x,
y\rangle = \sum_n x_ny_n$.

\medskip
\noindent\textbf{Parte I --- Hölder y Minkowski para sucesiones.}
\begin{enumerate}
  \item (Desigualdad de Young) Para $a, b \geq 0$, demostrar $ab \leq
        \frac{a^p}p + \frac{b^q}q$, usando la concavidad de $\log$ o
  estudiando $t \mapsto \frac{t^p}p + \frac1q - t$. \item (Hölder)
  Deducir: $\abs{\langle x, y\rangle} \leq
        \norm x_p\norm y_q$ para $x \in \ell^p$, $y \in \ell^q$;
  identificar el caso de igualdad. \item (Minkowski) Deducir la
  desigualdad triangular para $\norm\cdot_p$. \emph{(Escríbase
  $\abs{x_n + y_n}^p \leq
        \abs{x_n}\,\abs{x_n{+}y_n}^{p-1} +
        \abs{y_n}\,\abs{x_n{+}y_n}^{p-1}$ y aplíquese Hölder a cada
  término.)} \item Demostrar que $\ell^p$ es
  \omterm{def:b3:complete:complete}{completo} y que las sucesiones
  finitas son densas en él.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La dualidad $(\ell^p)' =
\ell^q$.}
\begin{enumerate}[resume]
  \item Para $y \in \ell^q$, demostrar que $\Lambda_y(x) = \langle
        x, y\rangle$ define $\Lambda_y \in (\ell^p)'$ con
  $\norm{\Lambda_y} \leq \norm y_q$ y, evaluando en
        $x_n = \abs{y_n}^{q-1}\operatorname{sign}(y_n)$
        (convenientemente truncada y normalizada), que
        $\norm{\Lambda_y} = \norm y_q$.
  \item Recíprocamente, dada $\Lambda \in (\ell^p)'$, póngase $y_n =
        \Lambda(e_n)$; demostrar que $y \in \ell^q$ con $\norm y_q \leq
        \norm\Lambda$ \emph{(evalúese en truncamientos como en la
  pregunta 5 y hágase crecer la longitud del truncamiento)} y concluir
  $\Lambda = \Lambda_y$: la aplicación $y \mapsto \Lambda_y$ es un
  isomorfismo isométrico $\ell^q \to (\ell^p)'$. \item Deducir que
  $\ell^p$ es \emph{\omterm{rem:b3:banach:bidual}{reflexivo}} para
  $1 < p <
        \infty$: componiendo las dos dualidades, todo elemento de
  $(\ell^p)''$ procede de $\ell^p$; verifíquese con cuidado que la
  compuesta es la $J$ canónica.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- $\ell^1$ y $\ell^\infty$ son animales
distintos.}
\begin{enumerate}[resume]
  \item Demostrar que $\ell^p$ ($1 \leq p < \infty$) y $c_0$ son
  separables, pero $\ell^\infty$ no lo es. \emph{(Las sucesiones
  indicadoras de los subconjuntos de $\N$, en cantidad no numerable,
  distan dos a dos $1$.)} \item Deducir del~\cref{exo:b3:banach:10} que
  $(\ell^1)'
        \cong \ell^\infty$ pero $(\ell^\infty)' \not\cong
        \ell^1$: $\ell^1$ \emph{no} es
  \omterm{rem:b3:banach:bidual}{reflexivo}. \emph{(Si $(\ell^\infty)'$
  fuese $\ell^1$, sería separable, lo que forzaría que $\ell^\infty$
  fuese separable.)} \item (Un límite de Banach, explícitamente) Sobre
  $\ell^\infty_\R$, sea $p(x) = \limsup_n \frac{x_1 + \dots + x_n}{n}$.
  Demostrar que $p$ es sublineal y que, sobre el subespacio $c$ de las
  sucesiones convergentes, $\mathrm{LIM}(x) = \lim x$ cumple
  $\mathrm{LIM} \leq p$. Extiéndase por Hahn--Banach a
  $\mathrm{LIM} \colon \ell^\infty_\R \to \R$ y demuéstrese que
  $\mathrm{LIM}$ es positiva ($x \geq 0 \Rightarrow
        \mathrm{LIM}(x) \geq 0$), invariante por desplazamiento,
        ($\mathrm{LIM}(x_2, x_3, \dots) = \mathrm{LIM}(x)$),
        extiende el límite y cumple $\liminf x \leq
        \mathrm{LIM}(x)\leq \limsup x$. \item Demostrar que una tal
        $\mathrm{LIM}$, vista en $(\ell^\infty)'$, \emph{no} es de la
        forma $\Lambda_y$ para ningún $y \in \ell^1$; concluir de nuevo
        $(\ell^\infty)'
        \neq \ell^1$. \emph{(Evalúese en las sucesiones unidad $e_n$ y
        después en la sucesión constante $1$.)} \item Evaluar
        $\mathrm{LIM}$ en $(0,1,0,1,\dots)$ y demostrar que no puede
        existir ninguna extensión \emph{multiplicativa} del límite
        invariante por desplazamiento \emph{(considérense $x =
        (0,1,0,1,\dots)$ y $x\cdot Sx$, donde $S$ es el
        desplazamiento)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Epílogo: por qué importa la reflexividad.}
\begin{enumerate}[resume]
  \item Usando el~\cref{cor:b3:banach:bslimits} y la pregunta 6,
  demostrar que toda sucesión acotada de $\ell^p$ ($1 < p <
        \infty$) tiene una subsucesión $(x^{(k)})$ que converge
  \emph{débilmente}: $\Lambda(x^{(k)})$ converge para toda
  $\Lambda \in (\ell^p)'$. \emph{(Extracción diagonal sobre las
  coordenadas, en cantidad numerable; identifíquese el límite débil en
  $\ell^p$ usando la acotación uniforme de las normas y Hölder.)}
  Demuéstrese con un ejemplo ($e_n$ en $\ell^1$, contra elementos bien
  elegidos de $\ell^\infty$) que esto falla en $\ell^1$: la \omterm{def:b3:topology:compact}{compacidad}
  débil es un privilegio de los \omterm{rem:b3:banach:bidual}{espacios
  reflexivos}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- La
\omterm{def:b3:topology:topology}{topología} débil en acción, y la
sorpresa de Schur.} Escribimos $x^{(k)} \rightharpoonup x$ en un espacio
normado $E$ (\emph{convergencia débil}) cuando
$\Lambda(x^{(k)}) \to \Lambda(x)$ para toda $\Lambda \in E'$.
\begin{enumerate}[resume]
  \item Complétese el censo:
  demuéstrese $(c_0)' \cong \ell^1$ isométricamente, con el esquema de
  las preguntas 5--6 (¿qué sustituye a las sucesiones de prueba?).
  Móntese la cadena $c_0 \to \ell^1 \to \ell^\infty \to \dots$ de
  \omterm{def:b3:banach:operator}{duales} sucesivos y señálese dónde
  falla la reflexividad. \item Demostrar que toda sucesión débilmente
  convergente de un espacio de Banach está acotada: véanse los $x^{(k)}$
  a través del encaje canónico $J$ como funcionales sobre $E'$ y
  aplíquese Banach--Steinhaus (el~\cref{thm:b3:banach:banachsteinhaus})
  --- ¿sobre qué espacio de Banach, y por qué se dispone allí de la
  \omterm{def:b3:complete:complete}{completitud}? \item Demostrar que,
  en $\ell^p$, $1 < p < \infty$: $x^{(k)}
        \rightharpoonup x$ si y solo si $\sup_k\norm{x^{(k)}}_p <
        \infty$ y $x^{(k)}_n \to x_n$ para toda coordenada $n$
  \emph{(una implicación usa Banach--Steinhaus a través del encaje
  canónico; para la otra, aproxímese $y
        \in \ell^q$ por sucesiones finitas)}. Deducir $e_k
        \rightharpoonup 0$ en $\ell^2$ mientras que $\norm{e_k}_2 =
        1$: los límites débiles pueden perder masa. \item Demostrar que
  la norma es débilmente semicontinua inferiormente:
  $x^{(k)} \rightharpoonup x$ implica $\norm x \leq
        \liminf_k\,\norm{x^{(k)}}$ \emph{(tómese una funcional que norme
  $x$, el~\cref{cor:b3:banach:hbnormed})}. \item (Radon--Riesz en
  $\ell^2$) Demostrar que, en $\ell^2$, la convergencia débil junto con
  la convergencia de las normas implica la convergencia en norma
  \emph{(desarróllese $\norm{x^{(k)} -
        x}_2^2$)}. Dese un contraejemplo del mismo enunciado sin la
  hipótesis sobre las normas. \item (Schur, paso 1) Sea
  $x^{(k)} \rightharpoonup 0$ en $\ell^1$ y supóngase, por reducción al
  absurdo, $\norm{x^{(k)}}_1 \geq \delta > 0$ a lo largo de una
  subsucesión. Demuéstrese primero que $x^{(k)}_n \to 0$ para cada $n$
  (¿qué funcionales?) y constrúyanse después recursivamente índices
  $k_1 < k_2 < \cdots$ y enteros $0 = N_0 < N_1 < N_2 < \cdots$ tales
  que la masa de $x^{(k_j)}$ se concentre en el bloque $B_j =
        \intoc{N_{j-1}}{N_j}$:
        \[
        \sum_{n \in B_j}\bigl|x^{(k_j)}_n\bigr| \geq
        \norm{x^{(k_j)}}_1 - \frac\delta{10} .
        \]
  \item (Schur, paso 2) Defínase $y \in \ell^\infty$ mediante $y_n =
        \operatorname{sign}\bigl(x^{(k_j)}_n\bigr)$ para $n \in
        B_j$. Demostrar que $\langle x^{(k_j)}, y\rangle \geq
        \norm{x^{(k_j)}}_1 - \frac{2\delta}{10} \geq
        \frac{8\delta}{10}$ y obtener una contradicción con
  $x^{(k)} \rightharpoonup 0$. Concluir el \emph{teorema de Schur}: en
  $\ell^1$, las sucesiones débilmente convergentes convergen en norma.
  \item Deducir que $(e_k)$ no tiene ninguna subsucesión débilmente
  convergente en $\ell^1$ (su único candidato a límite es $0$,
  coordenada a coordenada --- aplíquese después Schur), lo que recupera
  el fallo de \omterm{def:b3:topology:compact}{compacidad} débil de la pregunta 13; y resuélvase la
  aparente paradoja: en $\ell^1$, la convergencia débil y la
  convergencia en norma de \emph{sucesiones} coinciden y, sin embargo,
  las \emph{\omterm{def:b3:topology:topology}{topologías}} débil y de la norma difieren, y los acotados
  siguen sin ser débilmente \omterm{def:b3:topology:compact}{compactos}
  por sucesiones --- no hay contradicción, solo el fallo de la
  reflexividad. \item (Tabla de síntesis) Para
  $E \in \{c_0,\ \ell^1,\ \ell^p\
        (1{<}p{<}\infty),\ \ell^\infty\}$, tabúlense: el
  \omterm{def:b3:banach:operator}{dual}; la separabilidad; la
  reflexividad; si las sucesiones acotadas admiten subsucesiones
  débilmente convergentes; y una propiedad característica de cada
  espacio, justificada en una línea a partir de este problema.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Complementos: puntos más próximos,
convergencia promediada y el valor de un límite de Banach.}
\begin{enumerate}[resume]
  \item (Puntos más próximos: un dividendo de la reflexividad) Sean $F$
  un subespacio cerrado de $\ell^p$ ($1 < p < \infty$) y $x \in \ell^p$.
  Demostrar que $d = \operatorname{dist}(x,
        F)$ se \emph{alcanza}: extráigase de una sucesión minimizante
  una subsucesión débilmente convergente (pregunta 13), manténgase el
  límite débil dentro de $F$ construyendo, vía Hahn--Banach, una
  funcional que se anule en $F$ pero no en un punto exterior, y concluir
  con la pregunta 17. Demuéstrese después que el privilegio no es
  universal: en $c_0$, para $\Lambda(x) = \sum_n2^{-n}x_n$, demuéstrese
  que $\norm\Lambda = 1$ no se alcanza en la bola unidad, establézcase
  la fórmula de la distancia $\operatorname{dist}(x, \ker\Lambda) =
        \abs{\Lambda(x)}$ y dedúzcase que ningún $x \notin
        \ker\Lambda$ tiene punto más próximo en el hiperplano cerrado
  $\ker\Lambda$. \item (Banach--Saks en $\ell^2$) Sea $x^{(k)}
        \rightharpoonup 0$ en $\ell^2_{\R}$ con
  $\norm{x^{(k)}}_2 \leq C$. Constrúyase una subsucesión $(y_j)$ con
  $\abs{\langle y_i, y_j\rangle} \leq
        \frac1j$ para todos $i < j$, y dedúzcase
        \[
        \Bigl\lVert\frac{y_1 + \dots + y_m}m\Bigr\rVert_2^2
        \leq \frac{C^2 + 2}m \longrightarrow 0 :
        \]
        tras una extracción, las medias de Cesàro convergen en
        \emph{norma}. Compruébese en $(e_k)$, cuyas medias tienen norma
        $\frac1{\sqrt m}$: la convergencia débil, inútil para la propia
        sucesión (pregunta 16), se convierte en convergencia en norma
        para las medias. \item (El valor de un límite de Banach) Sea $L$
        un límite de Banach cualquiera (pregunta 10) y sea
        $A_mx = \frac1m(x + Sx + \dots
        + S^{m-1}x)$. Demostrar que $L(A_mx) = L(x)$ y $\liminf A_mx
        \leq L(x) \leq \limsup A_mx$; deducir que \emph{todos} los
        límites de Banach coinciden sobre las sucesiones periódicas, con
        valor la media sobre un período --- $\frac13$ sobre
        $(1, 0, 0, 1,
        0, 0, \dots)$, coherente con el $\frac12$ de la pregunta 12.
        Demuéstrese después que la coincidencia falla en general: para
        la sucesión por bloques $x$ igual a $1$ sobre
        $\intoc{3^{j-1}}{3^j}$ con $j$ par y a $0$ en el resto,
        demuéstrese que las medias de Cesàro oscilan entre $\leq
        \frac13$ y $\geq \frac23$, y constrúyanse dos límites de Banach
        $L_\pm$ con $L_-(x) \leq \frac13 < \frac23
        \leq L_+(x)$ \emph{(extiéndase desde $c \oplus \R x$ con los
        valores admisibles extremos $\pm$: compruébese que
        $\Lambda(y + tx) = \lim y + t\,p(x)$ está dominada por la
        funcional sublineal $p$ de la pregunta 10)}.
\end{enumerate}
\end{problem}
