\chapter{Representaciones de los grupos
finitos}\label{ch:b3:representations}

Para entender un grupo abstracto, hágase actuar sobre un espacio
vectorial y aplíquese el álgebra lineal --- valores propios, trazas,
productos escalares --- a la acción. Este programa, la \emph{teoría de
representaciones}, resulta
asombrosamente eficaz para los grupos finitos sobre $\C$: toda
\omterm{def:b3:representations:rep}{representación} se descompone en
\omterm{def:b3:representations:rep}{irreducibles} (Maschke), las
\omterm{def:b3:representations:rep}{irreducibles} quedan determinadas
por sus \emph{\omterm{def:b3:representations:character}{caracteres}}
(las trazas), y los
\omterm{def:b3:representations:character}{caracteres} cumplen relaciones
de ortogonalidad que vuelven mecánicos los cálculos. El capítulo
construye este cálculo y sus primeras obras maestras --- las
\omterm{def:b3:representations:table}{tablas de caracteres} de los
grupos pequeños ---, y el problema de fin de semana cosecha un teorema
que está muy lejos del alcance de la teoría de grupos desnuda: el
teorema $p^aq^b$ de Burnside. En todo el capítulo, $G$ es un grupo
finito y todos los espacios vectoriales son de dimensión finita sobre
$\C$.

\section{Representaciones, Maschke, Schur}

\begin{definition}\label{def:b3:representations:rep}
Una \emph{representación}\index{representación} de $G$ es un morfismo
$\rho \colon G \to GL(V)$ para cierto $\C$-espacio vectorial $V$;
$\dim V$ es su \emph{grado}. Un subespacio $W \subseteq V$ es
\emph{invariante} si $\rho(g)W \subseteq W$ para todo $g$; la
restricción hace de $W$ una \emph{subrepresentación}. $\rho$ es
\emph{irreducible}\index{representación irreducible} si $V \neq
0$ y sus únicos subespacios invariantes son $0$ y $V$. Un
\emph{morfismo} entre $(\rho, V)$ y $(\sigma, W)$ es una aplicación
lineal $f\colon V \to W$ con $f\rho(g) = \sigma(g)f$ para todo $g$
(equivarianza); los $f$ biyectivos son \emph{isomorfismos}.
\end{definition}

\begin{example}\label{ex:b3:representations:examples}
(a) Grado $1$: los morfismos $G \to \C^\times$. (b) La
\emph{representación regular}\index{representación regular}: $V = \C^G$
con base $(e_h)_{h \in G}$, $\rho(g)e_h = e_{gh}$; grado $\abs G$. (c)
Una acción por permutaciones de $G$ sobre un conjunto finito $X$ da la
\emph{\omterm{def:b3:representations:rep}{representación} de
permutación} sobre $\C^X$: $\rho(g)e_x =
e_{g\cdot x}$. (d) $S_n$ actúa sobre $\C^n$ permutando las coordenadas;
el hiperplano $\{\sum x_i = 0\}$ es invariante: la
\emph{\omterm{def:b3:representations:rep}{representación} estándar}, de
grado $n - 1$.
\end{example}

\begin{theorem}[Maschke]\label{thm:b3:representations:maschke}
Todo subespacio invariante $W$ de una
\omterm{def:b3:representations:rep}{representación} $(V, \rho)$ admite
un complemento invariante. En consecuencia, toda
\omterm{def:b3:representations:rep}{representación} es suma directa de
\omterm{def:b3:representations:rep}{irreducibles}
(\emph{semisimplicidad}).\index{teorema de Maschke}
\end{theorem}

\begin{proof}
Sea $p \colon V \to V$ una proyección \emph{cualquiera} de imagen $W$
(tómese un complemento cualquiera). Promediémosla sobre el grupo:
\[
\tilde p = \frac1{\abs G}\sum_{g \in G} \rho(g)\,p\,\rho(g)^{-1}.
\]
Cada término lleva $V$ dentro de $W$ ($W$ es invariante) y fija $W$
punto a punto: para $w \in W$, $\rho(g)^{-1}w \in W$, $p$ lo fija y
$\rho(g)$ lo devuelve --- de modo que $\tilde p$ es de nuevo una
proyección sobre $W$. Es equivariante: para $h \in G$,
$\rho(h)\tilde p\rho(h)^{-1}$ reindexa la misma suma. Por tanto, $\ker
\tilde p$ es un complemento invariante de $W$. Iterando sobre los
sumandos (dimensión finita) se descompone $V$ en
\omterm{def:b3:representations:rep}{irreducibles}.
\end{proof}

\begin{theorem}[Lema de Schur]\label{thm:b3:representations:schur}
Sea $f \colon V \to W$ un morfismo de
representaciones
\emph{\omterm{def:b3:representations:rep}{irreducibles}}. Entonces
$f = 0$ o bien $f$ es un isomorfismo; y si $(V,\rho) = (W,\sigma)$,
entonces $f = \lambda\,\mathrm{id}$ para cierto $\lambda \in \C$. Por
tanto, $\dim\operatorname{Hom}_G(V, W)$ vale $1$ si $V \cong W$ y $0$ en
caso contrario.\index{lema de Schur}
\end{theorem}

\begin{proof}
$\ker f$ y $\operatorname{im} f$ son invariantes (equivarianza), luego
cada uno es $0$ o el total: o bien $f = 0$, o bien $f$ es inyectivo y
con imagen total. Si $V = W$: $f$ tiene un valor propio $\lambda$ ($\C$
es \omterm{def:b3:galois:closure}{algebraicamente cerrado});
$f - \lambda\,\mathrm{id}$ es un morfismo no inyectivo $V \to V$, luego
$0$. Para el recuento de dimensiones cuando $V \cong W$: fijado un
isomorfismo $u$, todo morfismo $f$ da el endomorfismo
$u^{-1}f = \lambda\,\mathrm{id}$: $f =
\lambda u$.
\end{proof}

\section{Caracteres y ortogonalidad}

\begin{definition}\label{def:b3:representations:character}
El \emph{carácter}\index{carácter} de $(V, \rho)$ es
$\chi_\rho(g) = \operatorname{tr}\rho(g)$. Cumple
$\chi_\rho(e) = \dim V$, $\chi_\rho(hgh^{-1}) = \chi_\rho(g)$
(las trazas son invariantes por conjugación): los caracteres son
\emph{funciones de clase}\index{función de clase} --- elementos del
espacio $\mathcal{CF}(G)$ de las funciones constantes sobre las
\omterm{ex:b3:groups:actions}{clases de conjugación}, dotado del
producto escalar hermítico
\[
\langle \varphi, \psi\rangle = \frac1{\abs G}\sum_{g \in G}
\overline{\varphi(g)}\,\psi(g).
\]
\end{definition}

\begin{proposition}\label{prop:b3:representations:charbasics}
$\rho(g)$ es diagonalizable con valores propios raíces de la unidad;
$\chi_\rho(g^{-1}) = \overline{\chi_\rho(g)}$, y
$\abs{\chi_\rho(g)} \leq \chi_\rho(e)$ con igualdad si y solo si
$\rho(g)$ es un escalar. Los
\omterm{def:b3:representations:character}{caracteres} se suman sobre las
sumas directas: $\chi_{V
\oplus W} = \chi_V + \chi_W$.
\end{proposition}

\begin{proof}
$\rho(g)^N = \mathrm{id}$ para $N = \abs G$ (Lagrange): $\rho(g)$ anula
a $X^N - 1$ y se descompone con raíces simples, luego es diagonalizable
con valores propios $\lambda_i \in \mu_N$. Entonces
$\chi(g^{-1}) = \sum \lambda_i^{-1} = \sum\bar\lambda_i =
\overline{\chi(g)}$ y $\abs{\chi(g)} = \abs{\sum\lambda_i}
\leq \dim V$, con igualdad en la desigualdad triangular si y solo si
todos los $\lambda_i$ son iguales, es decir,
$\rho(g) = \lambda\,\mathrm{id}$. Aditividad sobre las sumas directas:
trazas por bloques.
\end{proof}

\begin{lemma}\label{lem:b3:representations:homtrace}
Sean $(V, \rho)$, $(W, \sigma)$ dos
representaciones. El operador de
promedio sobre $\operatorname{Hom}(W, V)$,
\[
c(A) = \frac1{\abs G}\sum_{g}\rho(g)\,A\,\sigma(g)^{-1},
\]
es una proyección sobre $\operatorname{Hom}_G(W, V)$, y para la
aplicación $\Phi_g \colon A \mapsto \rho(g)A\sigma(g)^{-1}$ se tiene
$\operatorname{tr}\Phi_g =
\chi_\rho(g)\,\overline{\chi_\sigma(g)}$.
\end{lemma}

\begin{proof}
$c(A)$ es equivariante (reindéxese la suma como en Maschke) y $c$ fija
las aplicaciones equivariantes (cada término vale $A$): $c$ es una
proyección de imagen $\operatorname{Hom}_G(W, V)$. Traza: en bases,
$\Phi_g(A) = BAC$ con $B = \rho(g)$, $C = \sigma(g)^{-1}$; sobre la base
$(E_{kl})$ de matrices, $BE_{kl}C = \sum_{m,n}
b_{mk}c_{ln}E_{mn}$, de modo que el coeficiente de $E_{kl}$ en
$\Phi_g(E_{kl})$ es $b_{kk}c_{ll}$: $\operatorname{tr}\Phi_g =
\sum_{k,l}b_{kk}c_{ll} = \operatorname{tr}(B)
\operatorname{tr}(C) = \chi_\rho(g)\chi_\sigma(g^{-1})$, y
$\chi_\sigma(g^{-1}) = \overline{\chi_\sigma(g)}$.
\end{proof}

\begin{theorem}[Primeras relaciones de ortogonalidad]
\label{thm:b3:representations:orthogonality}
Sean $\rho, \sigma$
\emph{\omterm{def:b3:representations:rep}{irreducibles}}.
Entonces\index{ortogonalidad de los caracteres}
\[
\langle\chi_\sigma, \chi_\rho\rangle =
\begin{cases}
1 & \text{si } \rho \cong \sigma,\\
0 & \text{en caso contrario:}
\end{cases}
\]
los \omterm{def:b3:representations:character}{caracteres}
\omterm{def:b3:representations:rep}{irreducibles} forman una familia
ortonormal en
$\mathcal{CF}(G)$.
\end{theorem}

\begin{proof}
La traza de una proyección es la dimensión de su imagen:
\[
\dim\operatorname{Hom}_G(W, V) = \operatorname{tr} c
= \frac1{\abs G}\sum_g \operatorname{tr}\Phi_g
= \frac1{\abs G}\sum_g \chi_\rho(g)\overline{\chi_\sigma(g)}
= \langle \chi_\sigma, \chi_\rho\rangle,
\]
y el lema de Schur evalúa el miembro de la izquierda como
$\delta_{\rho \cong
\sigma}$.
\end{proof}

\begin{corollary}\label{cor:b3:representations:multiplicity}
Descompóngase $V \cong \bigoplus_i V_i^{\oplus m_i}$ en
\omterm{def:b3:representations:rep}{irreducibles} distintas
($V_i \not\cong V_j$). Entonces $m_i = \langle
\chi_{V_i}, \chi_V\rangle$: las multiplicidades --- y por tanto la
\omterm{def:b3:representations:rep}{representación} salvo isomorfismo
--- quedan determinadas por el
\omterm{def:b3:representations:character}{carácter}. Además,
$\langle \chi_V, \chi_V\rangle = \sum_i
m_i^2$; en particular, $V$ es
\omterm{def:b3:representations:rep}{irreducible} si y solo si
$\langle\chi_V,
\chi_V\rangle = 1$.
\end{corollary}

\begin{proof}
$\chi_V = \sum m_i\chi_{V_i}$
(la~\cref{prop:b3:representations:charbasics}); tómense productos
escalares con cada $\chi_{V_i}$ y úsese la ortonormalidad. Dos
representaciones con el mismo
\omterm{def:b3:representations:character}{carácter} tienen las mismas
multiplicidades y, por tanto, son isomorfas.
\end{proof}

\begin{theorem}[La representación regular]
\label{thm:b3:representations:regular}
Sean $\chi_1, \dots, \chi_r$ los
\omterm{def:b3:representations:character}{caracteres}
\omterm{def:b3:representations:rep}{irreducibles} distintos, de grados
$n_i = \chi_i(e)$. La
\omterm{ex:b3:representations:examples}{representación regular} se
descompone con multiplicidades $m_i = n_i$; en consecuencia,
\[
\sum_{i=1}^{r} n_i^2 = \abs G,
\qquad
\sum_i n_i\chi_i(g) = 0 \quad (g \neq e).
\]
\end{theorem}

\begin{proof}
El \omterm{def:b3:representations:character}{carácter} regular:
$\chi_{\mathrm{reg}}(g) =
\#\{h : gh = h\}$, que vale $\abs G$ para $g = e$ y $0$ en los demás
casos. Luego $m_i = \langle \chi_i,\chi_{\mathrm{reg}}\rangle =
\frac1{\abs G}\overline{\chi_i(e)}\,\abs G = n_i$. Evaluando
$\chi_{\mathrm{reg}} = \sum n_i\chi_i$ en $e$ y en $g \neq e$ se
obtienen las dos identidades mostradas.
\end{proof}

\begin{theorem}\label{thm:b3:representations:numberirr}
Los \omterm{def:b3:representations:character}{caracteres}
\omterm{def:b3:representations:rep}{irreducibles} forman una \emph{base}
ortonormal de $\mathcal{CF}(G)$: el número de
representaciones \omterm{def:b3:representations:rep}{irreducibles} es
igual al número de \omterm{ex:b3:groups:actions}{clases de conjugación}
de $G$.
\end{theorem}

\begin{proof}
Solo queda la completitud: sea $f \in \mathcal{CF}(G)$ ortogonal a todo
$\chi_i$; veamos que $f = 0$. Para una
\omterm{def:b3:representations:rep}{representación} $(V,\rho)$, póngase
$T_{f,\rho} = \frac{1}{\abs G}
\sum_g \overline{f(g)}\,\rho(g)$. Es equivariante: para $h \in
G$,
\[
\rho(h)T_{f,\rho}\rho(h)^{-1} = \frac1{\abs G}\sum_g
\overline{f(g)}\rho(hgh^{-1})
= \frac1{\abs G}\sum_{g'}\overline{f(h^{-1}g'h)}\rho(g') =
T_{f,\rho}
\]
($f$ es una \omterm{def:b3:representations:character}{función de
clase}). Si $\rho$ es \omterm{def:b3:representations:rep}{irreducible}
de grado $n$, Schur da $T_{f,\rho} = \lambda\,\mathrm{id}$ con
\[
\lambda = \frac{\operatorname{tr}T_{f,\rho}}{n}
= \frac{1}{n\abs G}\sum_g \overline{f(g)}\,\chi_\rho(g)
= \frac1n\,\langle f, \chi_\rho\rangle = 0 .
\]
Así pues, $T_{f,\rho} = 0$ sobre toda
\omterm{def:b3:representations:rep}{irreducible} y, por tanto (sumas
directas), sobre \emph{toda}
\omterm{def:b3:representations:rep}{representación} --- en particular
sobre la regular. Aplíquese al vector de la base $e_e$:
$0 = T_{f,\mathrm{reg}}e_e =
\frac1{\abs G}\sum_g\overline{f(g)}e_g$, lo que obliga a que todo
$\overline{f(g)} = 0$. Así, la familia ortonormal $(\chi_i)$ genera
$\mathcal{CF}(G)$, cuya dimensión es el número de
\omterm{ex:b3:groups:actions}{clases de conjugación}.
\end{proof}

\begin{corollary}[Ortogonalidad por columnas]
\label{cor:b3:representations:column}
Para $g, h \in G$:
\[
\sum_{i=1}^{r}\overline{\chi_i(g)}\,\chi_i(h) =
\begin{cases}
\abs{Z_G(g)} & \text{si } g, h \text{ son conjugados},\\
0 & \text{en caso contrario.}
\end{cases}
\]
\end{corollary}

\begin{proof}
Sean $g_1, \dots, g_r$ representantes de las clases y $c_j$ los
cardinales de las clases. La matriz $U_{ij} =
\sqrt{c_j/\abs G}\;\chi_i(g_j)$ de tamaño $r \times r$ tiene las filas
ortonormales (el~\cref{thm:b3:representations:orthogonality} escrito por
clases: $\sum_j \frac{c_j}{\abs G}\chi_i(g_j)\overline{\chi_{i'}(g_j)} =
\delta_{ii'}$), es decir, $UU^* = I$; y una matriz cuadrada con $UU^* =
I$ cumple también $U^*U = I$: las columnas son ortonormales, lo que se
traduce en la identidad mostrada ($\abs G/c_j = \abs{Z_G(g_j)}$, la
relación órbita--estabilizador para la conjugación).
\end{proof}

\begin{proposition}[Caracteres de grado uno; elevación]
\label{prop:b3:representations:onedim}
(a) $G$ es abeliano si y solo si todas sus
representaciones \omterm{def:b3:representations:rep}{irreducibles}
tienen grado $1$; el número de
\omterm{def:b3:representations:character}{caracteres} de grado $1$ de un
$G$ cualquiera es $[G : D(G)]$ (son los
\omterm{def:b3:representations:character}{caracteres} de la
abelianización). (b) Si $N \trianglelefteq G$, las
representaciones \omterm{def:b3:representations:rep}{irreducibles} de
$G/N$ se elevan (componiendo con $G \to G/N$) exactamente a aquellas
representaciones \omterm{def:b3:representations:rep}{irreducibles} de
$G$ cuyo núcleo contiene $N$.
\end{proposition}

\begin{proof}
(a) Si $G$ es abeliano, cada clase es un punto: $r = \abs G$, y
$\sum n_i^2 = \abs G$ obliga a que todos los $n_i = 1$; recíprocamente,
si todos los $n_i = 1$, la
\omterm{ex:b3:representations:examples}{representación regular} es suma
de representaciones de dimensión uno, luego $\rho_{\mathrm{reg}}(G)$ es
simultáneamente diagonalizable y, por tanto, conmutativa, y
$\rho_{\mathrm{reg}}$ es fiel: $G$ es abeliano. Las
representaciones de grado $1$ son
morfismos $G \to \C^\times$ con destino abeliano: se factorizan a través
de $G^{\mathrm{ab}} = G/D(G)$ (el~\cref{exo:b3:groups:9}), y los
\omterm{def:b3:representations:character}{caracteres} distintos del
abeliano $G^{\mathrm{ab}}$ son $\abs{G^{\mathrm{ab}}}$ (tiene ese número
de clases, todas de grado $1$). (b) Componer con la proyección conserva
la irreducibilidad (los subespacios invariantes se corresponden), y una
\omterm{def:b3:representations:rep}{representación} trivial sobre $N$ se
factoriza a través del cociente (el~\cref{thm:b3:groups:firstiso}).
\end{proof}

\section{Tablas de caracteres}

\begin{definition}\label{def:b3:representations:table}
La \emph{tabla de caracteres}\index{tabla de caracteres} de $G$ es la
matriz $\bigl(\chi_i(g_j)\bigr)$ de tamaño $r \times r$: las filas están
indexadas por los \omterm{def:b3:representations:character}{caracteres}
\omterm{def:b3:representations:rep}{irreducibles} y las columnas por las
\omterm{ex:b3:groups:actions}{clases de conjugación} (con sus cardinales
indicados). Las filas son ortonormales para el producto ponderado y las
columnas son ortogonales (el~\cref{cor:b3:representations:column}): la
tabla está enormemente sobredeterminada, y eso es lo que la hace
calculable.
\end{definition}

\begin{example}[La tabla de $S_3$]\label{ex:b3:representations:s3}
Clases: $e$ (tamaño 1), las trasposiciones (3) y los $3$-ciclos (2);
luego hay $r
= 3$ \omterm{def:b3:representations:rep}{irreducibles}, de grados $n_i$
con $\sum n_i^2 = 6$: $1, 1,
2$. Grado $1$: la trivial $\mathbf 1$ y la signatura $\varepsilon$. La
última fila se sigue de la ortogonalidad por columnas (o de
$\chi_{\mathrm{std}} = \chi_{\mathrm{perm}} - \mathbf 1$):
\[
\begin{array}{c|ccc}
S_3 & e & (1\,2)\ [3] & (1\,2\,3)\ [2]\\
\hline
\mathbf 1 & 1 & 1 & 1\\
\varepsilon & 1 & -1 & 1\\
\chi_{\mathrm{std}} & 2 & 0 & -1
\end{array}
\]
Comprobación: $\langle\chi_{\mathrm{std}},\chi_{\mathrm{std}}\rangle =
\frac{1}{6}(4 + 0 + 2) = 1$:
\omterm{def:b3:representations:rep}{irreducible}.
\end{example}

\begin{example}[La tabla de $S_4$]\label{ex:b3:representations:s4}
Clases: $e$ [1], trasposiciones [6], dobles trasposiciones [3],
$3$-ciclos [8], $4$-ciclos [6]: cinco
\omterm{def:b3:representations:rep}{irreducibles}, $\sum n_i^2 =
24$ con dos de grado $1$ ($\mathbf 1, \varepsilon$; $[S_4 :
D(S_4)] = [S_4 : A_4] = 2$): grados $1, 1, 2, 3, 3$. La de grado $2$ se
eleva desde $S_4/V \cong S_3$
(la~\cref{prop:b3:representations:onedim}(b), con $V$ el grupo de
Klein); grado $3$: la
\omterm{def:b3:representations:rep}{representación} estándar y su
\omterm{def:b3:modules:torsion}{torsión} por
$\varepsilon$:
\[
\begin{array}{c|ccccc}
S_4 & e\,[1] & (1\,2)\,[6] & (1\,2)(3\,4)\,[3] &
(1\,2\,3)\,[8] & (1\,2\,3\,4)\,[6]\\
\hline
\mathbf 1 & 1 & 1 & 1 & 1 & 1\\
\varepsilon & 1 & -1 & 1 & 1 & -1\\
\chi_2 & 2 & 0 & 2 & -1 & 0\\
\chi_{\mathrm{std}} & 3 & 1 & -1 & 0 & -1\\
\varepsilon\chi_{\mathrm{std}} & 3 & -1 & -1 & 0 & 1
\end{array}
\]
($\chi_{\mathrm{std}}(g) = \operatorname{fix}(g) - 1$;
$\chi_2$ evalúa la tabla de $S_3$ sobre la imagen de cada clase módulo
$V$.) Todas las comprobaciones de ortogonalidad por filas y columnas
salen bien --- ejecutar dos de ellas es el calentamiento
del~\cref{exo:b3:representations:3}.
\end{example}

\begin{method}\label{met:b3:representations:table}
Para construir una \omterm{def:b3:representations:table}{tabla de
caracteres}: (1) lístense las \omterm{ex:b3:groups:actions}{clases de
conjugación} y sus cardinales; (2) cuéntense los
\omterm{def:b3:representations:character}{caracteres} de grado $1$
mediante $G/D(G)$ y escríbanse; (3) hállense los grados restantes a
partir de $\sum n_i^2 = \abs G$ (combinatoria de enteros pequeños); (4)
obténganse \omterm{def:b3:representations:rep}{irreducibles} baratas:
elévense desde cocientes, réstese $\mathbf 1$ de los
\omterm{def:b3:representations:character}{caracteres} de permutación
(compruébese $\langle\chi,\chi\rangle = 1$) y multiplíquense los
\omterm{def:b3:representations:character}{caracteres} conocidos por los
de grado $1$; (5) complétense las filas desconocidas por ortogonalidad
de columnas --- cada columna es ortogonal a las columnas ya completas, y
la columna $e$ contiene los grados. Verifíquese todo con un barrido
completo de ortogonalidad.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:representations:1}
(a) Demostrar que los
\omterm{def:b3:representations:character}{caracteres}
\omterm{def:b3:representations:rep}{irreducibles} de $\Z/n\Z$ son los
$\chi_k(\bar m) = \eu^{2\iu\pi km/n}$, $k = 0, \dots, n-1$, y escribir
la \omterm{def:b3:representations:table}{tabla de caracteres} de
$\Z/4\Z$. (b) Verificar en ella las dos relaciones de ortogonalidad ---
y reconocer la matriz: ¿dónde la ha visto antes este libro?
\end{exercise}

\section{Problema: el teorema \texorpdfstring{$p^aq^b$}{paqb} de
Burnside}

\begin{problem}[{Problema de fin de semana --- resolubilidad de los
grupos de
orden $p^aq^b$}]\label{pb:b3:representations:1} Burnside demostró en
1904 que todo grupo cuyo orden tiene a lo sumo dos factores primos es
\omterm{def:b3:groups:derived}{resoluble} --- un enunciado sobre grupos
abstractos cuyas únicas demostraciones conocidas durante medio siglo
pasaban por la teoría de
\omterm{def:b3:representations:character}{caracteres}. Este problema
construye la demostración completa, reuniendo el~\cref{ch:b3:groups}
(\omterm{def:b3:groups:derived}{resolubilidad}), el~\cref{ch:b3:modules}
($\Z$-módulos \omterm{def:b3:modules:free}{finitamente generados}) y el
presente capítulo. En todo él, $\chi_1, \dots, \chi_r$ son los
\omterm{def:b3:representations:character}{caracteres}
\omterm{def:b3:representations:rep}{irreducibles} de $G$,
$n_i = \chi_i(e)$.

\medskip
\noindent\textbf{Parte I --- Enteros
\omterm{def:b3:galois:algebraic}{algebraicos}.} Un \emph{entero
\omterm{def:b3:galois:algebraic}{algebraico}} es una raíz de un
polinomio \emph{mónico} de $\Z[X]$.
\begin{enumerate}
  \item Demostrar que $\alpha$ es un entero
  \omterm{def:b3:galois:algebraic}{algebraico} si y solo si el anillo
  $\Z[\alpha]$ es un $\Z$-módulo
  \omterm{def:b3:modules:free}{finitamente generado}. \item Deducir que
  los enteros \omterm{def:b3:galois:algebraic}{algebraicos} forman un
  subanillo de $\C$. \emph{(Si $\Z[\alpha], \Z[\beta]$ son
  \omterm{def:b3:modules:free}{finitamente generados}, también lo es
  $\Z[\alpha, \beta]$, y los submódulos de los $\Z$-módulos
  \omterm{def:b3:modules:free}{finitamente generados} son
  \omterm{def:b3:modules:free}{finitamente generados}, por
  el~\cref{thm:b3:modules:submodule} y un argumento de presentación ---
  o directamente: un submódulo de $\Z^n$ es libre de rango $\leq n$.)}
  \item Demostrar que un entero
  \omterm{def:b3:galois:algebraic}{algebraico} \emph{racional} es un
  entero. \emph{(Teorema de la raíz racional.)} \item Demostrar que todo
  valor \omterm{def:b3:representations:character}{carácter} $\chi(g)$ es
  un entero \omterm{def:b3:galois:algebraic}{algebraico}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Las relaciones de sumas de clases.}
Fíjense una $(V, \rho)$ \omterm{def:b3:representations:rep}{irreducible}
de grado $n$ y \omterm{def:b3:representations:character}{carácter}
$\chi$. Para una \omterm{ex:b3:groups:actions}{clase de conjugación}
$C$, sea $S_C = \sum_{g \in C}\rho(g) \in
\mathcal L(V)$.
\begin{enumerate}[resume]
  \item Demostrar que $S_C$ es equivariante y, por tanto, $S_C =
        \omega_C\,\mathrm{id}$ con
        \[
        \omega_C = \frac{\abs C\,\chi(g_C)}{n}
        \qquad (g_C \in C \text{ un representante cualquiera}).
        \]
  \item Demostrar que $S_CS_{C'} = \sum_{C''}
        a_{CC'C''}\,S_{C''}$, donde $a_{CC'C''} \in \N$ cuenta, para un
  $z \in C''$ fijo, los pares $(x, y) \in C \times
        C'$ con $xy = z$. Deducir que los $\omega_C$ cumplen
        $\omega_C\,\omega_{C'} = \sum_{C''}
        a_{CC'C''}\,\omega_{C''}$.
  \item Concluir que cada $\omega_C$ es un entero
  \omterm{def:b3:galois:algebraic}{algebraico}. \emph{(El $\Z$-módulo
  generado por $1$ y los $\omega_C$ es un anillo
  \omterm{def:b3:modules:free}{finitamente generado}; aplíquese el
  criterio de la pregunta 1 --- con más precisión, demuéstrese que
  $M = \Z\text{-gen}
        (1, (\omega_C)_C)$ cumple $\omega_{C_0} M \subseteq
        M$ y úsese un truco de determinantes o de Cayley--Hamilton, o
  bien el argumento de subanillo de la pregunta 2.)} \item Deducir la
  \emph{divisibilidad de
  \omterm{thm:b3:galois:finitefields}{Frobenius}}: $n_i$ divide a
  $\abs G$ para todo grado
  \omterm{def:b3:representations:rep}{irreducible}. \emph{(Calcúlese
  $\frac{\abs G}{n} = \frac{\abs G}{n}
        \langle\chi,\chi\rangle = \sum_C \omega_C\,
        \overline{\chi(g_C)}$: un entero
  \omterm{def:b3:galois:algebraic}{algebraico} que es racional.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El criterio de simplicidad de Burnside.}
\begin{enumerate}[resume]
  \item Sea $\chi$ \omterm{def:b3:representations:rep}{irreducible} de
  grado $n$ y sea $C$ una clase con $\gcd(\abs C, n) = 1$. Usando Bézout
  y las preguntas 4--7, demostrar que $\frac{\chi(g_C)}{n}$ es un entero
  \omterm{def:b3:galois:algebraic}{algebraico}. \item Supóngase además
  $0 < \abs{\chi(g_C)} < n$. Demostrar que esto es imposible: el entero
  \omterm{def:b3:galois:algebraic}{algebraico} $\alpha =
        \chi(g_C)/n$ tiene todos sus \emph{conjugados} --- los números
  $\alpha_\sigma = \frac1n\sigma(\chi(g_C))$ para
        $\sigma \in \operatorname{Gal}(\Q(\zeta_{\abs G})/\Q)$,
        cada uno de ellos promedio de $n$ raíces de la unidad --- de
        módulo
        $\leq 1$,
        de modo que el producto $N = \prod_\sigma\alpha_\sigma$ es un
        entero \omterm{def:b3:galois:algebraic}{algebraico} racional con
        $0 < \abs N < 1$ --- justifíquese cada afirmación, citando
        el~\cref{thm:b3:galois:cyclotomicirred} para el
        \omterm{def:b3:galois:galois}{grupo de Galois} y el hecho de que
        $\sigma$ permuta las raíces de la unidad. Concluir: o bien
        $\chi(g_C) = 0$, o bien $\rho(g_C)$ es escalar
        (la~\cref{prop:b3:representations:charbasics}). \item (Criterio
        de Burnside) Sea $C \neq \{e\}$ una
        \omterm{ex:b3:groups:actions}{clase de conjugación} de cardinal
        \emph{potencia de primo} $p^k > 1$, y supóngase que $G$ es
        simple no abeliano. La ortogonalidad de columnas entre la
        columna de $C$ y la de $e$ da
        $1 + \sum_{i \geq 2} n_i\chi_i(g_C) = 0$. Demostrar que algún
        $\chi_i$ no trivial con $p \nmid n_i$ cumple
        $\chi_i(g_C) \neq 0$ \emph{(en caso contrario, $\frac1p$ sería
        un entero \omterm{def:b3:galois:algebraic}{algebraico})}; por la
        pregunta 10, $\rho_i(g_C)$ es escalar; obténgase una
        contradicción con la simplicidad \emph{(el conjunto de los $g$
        con $\rho_i(g)$ escalar es un
        \omterm{def:b3:groups:normal}{subgrupo normal}; úsese la
        fidelidad del~\cref{exo:b3:representations:6})}. Concluir:
        \emph{ningún \omterm{def:b3:groups:simple}{grupo simple} no abeliano tiene una
        \omterm{ex:b3:groups:actions}{clase de conjugación} de cardinal
        potencia de primo $> 1$.}
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El teorema.}
\begin{enumerate}[resume]
  \item Sea $\abs G = p^aq^b$ con $a + b \geq 1$. Si $G$ es simple,
  demostrar que es abeliano: tómese $z \neq e$ en el
  \omterm{ex:b3:groups:actions}{centro} de un $q$-subgrupo de Sylow
  (el~\cref{thm:b3:groups:pfixed}) y considérese el cardinal de su
  \omterm{ex:b3:groups:actions}{clase de conjugación} $[G : Z_G(z)]$,
  potencia de $p$ (¿por qué?); aplíquese la pregunta 11. \item Concluir
  por inducción sobre $\abs G$: \emph{todo grupo de orden $p^aq^b$ es
  \omterm{def:b3:groups:derived}{resoluble}} (Burnside). ¿Por qué se
  rompe el argumento con tres primos --- y por qué tiene que romperse,
  dado
        $\abs{A_5} = 2^2\cdot3\cdot5$?
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- La
\omterm{def:b3:representations:table}{tabla de caracteres} de $A_5$.} El
grupo más pequeño al que el teorema de Burnside no llega merece su
retrato completo; todo lo que sigue usa solo este capítulo más
\cref{exo:b3:groups:11}.
\begin{enumerate}[resume]
  \item Recuérdense del~\cref{exo:b3:groups:11} las cinco
  \omterm{ex:b3:groups:actions}{clases de conjugación} de $A_5$:
  $\{e\}$, las $15$ dobles trasposiciones, los $20$ ciclos de longitud
  tres y dos clases de $12$ ciclos de longitud cinco cada una,
  representadas por $c =
        (1\,2\,3\,4\,5)$ y $c^2$. Explicar por qué los ciclos de
  longitud cinco se desdoblan en dos $A_5$-clases aunque formen una sola
  $S_5$-clase. \item Demostrar que los grados
  \omterm{def:b3:representations:rep}{irreducibles} de $A_5$ son
  exactamente $1, 3, 3, 4, 5$: úsese $\sum_in_i^2 = 60$ con $r = 5$
  clases, junto con el hecho de que $A_5$ es
  \omterm{prop:b3:galois:perfect}{perfecto} ($D(A_5) =
        A_5$), de modo que el
  \omterm{def:b3:representations:character}{carácter} trivial es su
  único \omterm{def:b3:representations:character}{carácter} lineal; elimínese después cualquier otro multiconjunto
  \emph{(escríbase $59$ como suma de cuatro cuadrados de enteros
  $\geq 2$: compruébese que solo ocurre de una manera con todos los
  sumandos grados plausibles)}. \item Sea $\pi$ el
  \omterm{def:b3:representations:character}{carácter} de permutación de
  $A_5$ sobre
        $\{1, \dots, 5\}$: $\pi(g) = \#\operatorname{Fix}(g)$,
        con valores $5, 1, 2, 0, 0$ en las cinco clases. Calcular
        $\langle\pi, \mathbf 1\rangle$ y $\langle\pi,
        \pi\rangle$, y deducir que $\chi_4 = \pi - \mathbf 1$ es
        \omterm{def:b3:representations:rep}{irreducible} de grado $4$,
        con valores $4, 0, 1, -1,
        -1$. \item El mismo juego sobre los $10$ pares no ordenados
        $\{i, j\}$: los recuentos de puntos fijos son $10, 2, 1, 0, 0$.
        Calcular $\langle\pi_{10}, \pi_{10}\rangle$, $\langle\pi_{10},
        \mathbf 1\rangle$ y $\langle\pi_{10}, \chi_4\rangle$, deducir la
        descomposición $\pi_{10} = \mathbf 1 + \chi_4
        + \chi_5$ y obtener la
        \omterm{def:b3:representations:rep}{irreducible} $\chi_5$ de
        grado $5$ con valores $5, 1, -1, 0, 0$. \item Las dos
        \omterm{def:b3:representations:rep}{irreducibles} restantes
        $\chi_2, \chi_3$ tienen grado $3$. La ortogonalidad de columnas
        (cada columna distinta de la identidad contra la columna de la
        identidad, y cada columna consigo misma) determina sus valores
        fuera de los ciclos de longitud cinco: demuéstrese que
        $\chi_2(g) = \chi_3(g) = -1$ sobre las dobles trasposiciones y
        $0$ sobre los ciclos de longitud tres. \item Sobre las clases de
        ciclos de longitud cinco, póngase $x = \chi_2(c)$ y $y =
        \chi_2(c^2)$; la simetría permite tomar $\chi_3(c) = y$,
        $\chi_3(c^2) = x$. A partir de la columna de $c$ emparejada con
        la columna de la identidad y con la columna de $c^2$, dedúzcanse
        $x + y = 1$ y $xy = -1$ (y compruébese el valor $x^2 +
        y^2 = 3$ que da la columna de $c$ consigo misma), de donde
        \[
        \{x, y\} = \Bigl\{\frac{1 + \sqrt5}2,\ \frac{1 -
        \sqrt5}2\Bigr\} :
        \]
        la razón áurea y su conjugada. Montar la
        \omterm{def:b3:representations:table}{tabla de caracteres}
        completa de $A_5$. \item Ejecutar las comprobaciones: la norma
        de la fila $\chi_2$ es $1$ (úsese
        $\varphi^2 + \bar\varphi^2 = 3$), $\langle\chi_2,
        \chi_3\rangle = 0$, y la divisibilidad de
        \omterm{thm:b3:galois:finitefields}{Frobenius} (pregunta 8) para
        los cinco grados. ¿Dónde se \emph{ve} en la tabla una diferencia
        con $S_5$, cuyos valores
        \omterm{def:b3:representations:character}{carácter} son todos
        enteros racionales? \item Deducir solo de la tabla que $A_5$ es
        simple: un \omterm{def:b3:groups:normal}{subgrupo normal} es una
        unión de \omterm{ex:b3:groups:actions}{clases de conjugación}
        que contiene $e$ y cuyo cardinal divide a $60$ --- compruébese
        que ninguna subsuma propia de $1 + 15 + 20 + 12 + 12$ que
        contenga el término $1$ divide a $60$. Contrástese con el
        criterio de la pregunta 11: verifíquese que ninguna clase de
        $A_5$ tiene cardinal potencia de primo $> 1$. \item (Coda
        icosaédrica) $A_5$ es el grupo de rotaciones del icosaedro, y
        las representaciones de
        grado $3$ son las dos acciones geométricas sobre $\R^3$.
        Verificar la identidad de trazas: una rotación de ángulo
        $\theta$ tiene traza $1 +
        2\cos\theta$, y $1 + 2\cos\frac{2\pi}5 =
        \frac{1+\sqrt5}2 = \varphi$. Explicar sin cálculo alguno por qué
        el otro \omterm{def:b3:representations:character}{carácter} de
        grado $3$ ha de llevar el valor conjugado: el
        \omterm{def:b3:galois:galois}{grupo de Galois} de
        $\Q(\sqrt5)/\Q$ actúa sobre toda la
        \omterm{def:b3:representations:table}{tabla de caracteres}
        (entrada a entrada), permutando los
        \omterm{def:b3:representations:character}{caracteres}
        \omterm{def:b3:representations:rep}{irreducibles}.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Complementos: una cota central y el
cuadrado tensorial.}
\begin{enumerate}[resume]
  \item (Más fino que una divisibilidad) Sea $\chi$
  \omterm{def:b3:representations:rep}{irreducible} de grado $n$.
  Demostrar que $\abs{\chi(z)} = n$ para todo $z \in Z(G)$ \emph{(lema
  de Schur: $\rho(z)$ es un escalar, de orden finito)} y deducir de
  $\langle\chi, \chi\rangle = 1$ la cota
        \[
        n^2 \leq [G : Z(G)] .
        \]
        Demostrar que los grupos no abelianos de orden $8$ tienen grados
        \omterm{def:b3:representations:rep}{irreducibles}
        $1, 1, 1, 1, 2$ \emph{(cinco
        \omterm{ex:b3:groups:actions}{clases de conjugación}; escríbase
        $8$ como suma de cinco cuadrados)} y alcanzan la igualdad
        $4 = [G : Z(G)]$; compruébese la cota en $A_5$, cuyo
        \omterm{ex:b3:groups:actions}{centro} es trivial. \item
        (Cuadrado tensorial de $\chi_2$) Para $g$ de orden finito,
        $\rho(g)$ es diagonalizable con valores propios raíces de la
        unidad; dedúzcanse las fórmulas de
        \omterm{def:b3:representations:character}{caracteres}
        \[
        \chi_{\operatorname{Sym}^2 V}(g)
        = \frac{\chi(g)^2 + \chi(g^2)}2,
        \qquad
        \chi_{\Lambda^2 V}(g)
        = \frac{\chi(g)^2 - \chi(g^2)}2 .
        \]
        Aplíquense a $\chi_2$ de $A_5$ \emph{(obsérvese que $g^2$
        recorre la clase de $c^2$ cuando $g$ recorre la de $c$, y
        recíprocamente)}: demuéstrese que $\Lambda^2\chi_2 =
        \chi_2$ y $\operatorname{Sym}^2\chi_2 = \mathbf 1 +
        \chi_5$, de donde
        \[
        \chi_2\otimes\chi_2 = \mathbf 1 + \chi_2 + \chi_5 .
        \]
        Interprétese $\Lambda^2\chi_2 = \chi_2$ geométricamente mediante
        el producto vectorial en $\R^3$. \item (Auditoría final de la
        tabla) Verificar numéricamente: la ortogonalidad de columnas
        entre las dos columnas de ciclos de longitud cinco
        ($\varphi\bar\varphi = -1$), el valor $5 =
        \abs{Z_{A_5}(c)}$ para la columna de $c$ consigo misma, y la
        anulación del
        \omterm{def:b3:representations:character}{carácter} regular
        $\sum_i
        n_i\chi_i(g) = 0$ en cada una de las cuatro columnas distintas
        de la identidad de la tabla.
\end{enumerate}
\end{problem}
