\chapter{Funciones holomorfas}\label{ch:b3:holomorphic}

La derivabilidad compleja parece una pequeña variación de la teoría real
--- un límite, un cociente. Es, en cambio, otro universo. Una función
derivable \emph{una vez} en un abierto de $\C$ es automáticamente
infinitamente derivable, analítica, determinada en todo un dominio por
sus valores cerca de un solo punto y sujeta a principios globales
rígidos (Liouville, módulo máximo). Todo ello brota de un milagro, el
teorema de Cauchy: la integral de una
\omterm{def:b3:holomorphic:holo}{función holomorfa} a lo largo de un
\omterm{def:b3:topology:pathconnected}{camino} cerrado en un dominio
estrellado es nula. Este capítulo demuestra el milagro (el argumento de
Goursat, sin suponer más regularidad que la derivabilidad), cosecha sus
consecuencias clásicas y salda un teorema que este libro viene usando a
crédito desde el~\cref{ch:b3:galois}: todo polinomio complejo no
constante tiene una raíz.

En todo el capítulo, $\Omega \subseteq \C$ es abierto y $D(a, r)$ denota
el disco abierto.

\section{Derivabilidad compleja}

\begin{definition}\label{def:b3:holomorphic:holo}
$f \colon \Omega \to \C$ es \emph{holomorfa}\index{función holomorfa} en
$\Omega$ si para todo $z_0 \in \Omega$ existe
\[
f'(z_0) = \lim_{h\to0}\frac{f(z_0 + h) - f(z_0)}{h}
\]
($h \in \C^*$). Sumas, productos, cocientes (con denominador no nulo) y
composiciones de funciones holomorfas son holomorfas, con las fórmulas
habituales (las demostraciones de primer y segundo año valen palabra por
palabra: solo usan operaciones de cuerpo y límites).
$\mathcal H(\Omega)$ denota el conjunto de las funciones holomorfas en
$\Omega$.
\end{definition}

\begin{proposition}[Cauchy--Riemann]
\label{prop:b3:holomorphic:cauchyriemann}
Escríbase $f(x + \iu y) = P(x,y) + \iu Q(x,y)$. Entonces $f$ es
\omterm{def:b3:holomorphic:holo}{holomorfa} en $z_0$ si y solo si $f$ es
$\R$-diferenciable en $z_0$ (como aplicación de dos variables reales)
y\index{ecuaciones de Cauchy--Riemann}
\[
\frac{\partial P}{\partial x} = \frac{\partial Q}{\partial y},
\qquad
\frac{\partial P}{\partial y} = -\frac{\partial Q}{\partial x}
\qquad \text{en } z_0 ;
\]
o, equivalentemente, si la diferencial real es la multiplicación por el
número complejo $f'(z_0)$.
\end{proposition}

\begin{proof}
La $\C$-derivabilidad dice que $f(z_0 + h) = f(z_0) + ch +
o(\abs h)$ con $c = f'(z_0)$: una diferencial $\R$-lineal que es la
multiplicación por $c = a + \iu b$, es decir, de matriz
$\bigl(\begin{smallmatrix} a & -b\\ b & a
\end{smallmatrix}\bigr)$ en la base $(1, \iu)$ --- exactamente las
relaciones exhibidas para las derivadas parciales. Recíprocamente, tal
diferencial es $\C$-lineal, y las definiciones de $o(\abs h)$ coinciden.
\end{proof}

\begin{example}\label{ex:b3:holomorphic:examples}
Los polinomios en $z$, las funciones racionales fuera de sus polos y ---
por el teorema de derivación término a término de las series de
potencias de segundo año, cuya demostración funciona idénticamente sobre
$\C$ --- toda suma de una serie de potencias $\sum a_n(z - a)^n$ dentro
de su disco de convergencia:
\omterm{def:b3:holomorphic:holo}{holomorfa}, con derivada $\sum na_n(z -
a)^{n-1}$ (mismo radio). En particular, $\exp z = \sum z^n/n!$ es entera
(\omterm{def:b3:holomorphic:holo}{holomorfa} en $\C$) con
$\exp' = \exp$. En cambio, $z \mapsto \bar z$, $\abs z$ y
$\operatorname{Re}z$ no son \omterm{def:b3:holomorphic:holo}{holomorfas}
en ningún punto (Cauchy--Riemann falla en todas partes): la
\omterm{def:b3:holomorphic:holo}{holomorfía} es rigidez que conserva
ángulos y orientación, no suavidad.
\end{example}

\section{Integrales de contorno}

\begin{definition}\label{def:b3:holomorphic:contour}
Un \emph{\omterm{def:b3:topology:pathconnected}{camino}} es una
aplicación $\mathcal C^1$ a trozos $\gamma \colon
\intcc ab \to \C$; es \emph{cerrado} si $\gamma(a) =
\gamma(b)$. Para $f$ \omterm{def:b3:topology:continuity}{continua} sobre
la imagen de $\gamma$:\index{integral de contorno}
\[
\int_\gamma f(z)\,\dd z =
\int_a^b f(\gamma(t))\,\gamma'(t)\,\dd t,
\qquad
\Bigl|\int_\gamma f\,\dd z\Bigr| \leq
\sup_{\gamma}\abs f\cdot\operatorname{length}(\gamma)
\]
(la \emph{desigualdad ML}; longitud $= \int_a^b\abs{\gamma'}$). La
integral es invariante por reparametrizaciones $\mathcal C^1$ crecientes
y cambia de signo al invertir la orientación.
\end{definition}

\begin{proposition}[Primitivas]\label{prop:b3:holomorphic:primitive}
Para $f$ \omterm{def:b3:topology:continuity}{continua} en $\Omega$, son
equivalentes: (i) $f$ tiene una \emph{primitiva}
$F \in \mathcal H(\Omega)$ ($F'
= f$); (ii) $\int_\gamma f\,\dd z = 0$ para todo
\omterm{def:b3:topology:pathconnected}{camino} cerrado $\gamma$
contenido en $\Omega$. En tal caso, $\int_\gamma f\,\dd z =
F(\gamma(b)) - F(\gamma(a))$ para todo
\omterm{def:b3:topology:pathconnected}{camino}.
\end{proposition}

\begin{proof}
(i)$\Rightarrow$: $\frac{\dd}{\dd t}F(\gamma(t)) =
F'(\gamma(t))\gamma'(t)$ (regla de la cadena, válida a trozos), de modo
que la integral se telescopa a la diferencia de los extremos; los
\omterm{def:b3:topology:pathconnected}{caminos} cerrados dan $0$.
(ii)$\Rightarrow$(i): fíjese $z_*$ en una
\omterm{def:b3:topology:components}{componente conexa}, defínase
$F(z) = \int_{\gamma_z}f\,\dd z$ a lo largo de cualquier
\omterm{def:b3:topology:pathconnected}{camino} de $z_*$ a $z$ (bien
definida: dos \omterm{def:b3:topology:pathconnected}{caminos} difieren
en un \omterm{def:b3:topology:pathconnected}{camino} cerrado); para $h$
pequeño, tomando el segmento de $z$ a
$z + h$,
\[
\frac{F(z + h) - F(z)}{h} - f(z)
= \frac1h\int_{[z, z+h]}\bigl(f(w) - f(z)\bigr)\dd w
\longrightarrow 0
\]
por la desigualdad ML y la
\omterm{def:b3:topology:continuity}{continuidad} de $f$ en $z$.
\end{proof}

\begin{definition}[Índice de un camino]\label{def:b3:holomorphic:index}
Para un \omterm{def:b3:topology:pathconnected}{camino} cerrado $\gamma$
y $z \notin
\operatorname{im}\gamma$, el \emph{índice}\index{número de
vueltas}\index{índice de un camino} es
\[
\operatorname{Ind}_\gamma(z) = \frac1{2\iu\pi}
\int_\gamma\frac{\dd w}{w - z} .
\]
Es un \emph{entero}: poniendo $\varphi(t) =
\int_a^t\frac{\gamma'(s)}{\gamma(s) - z}\dd s$, la función
$(\gamma(t) - z)\eu^{-\varphi(t)}$ tiene derivada nula (a trozos), luego
es constante; en $t = b$, $\eu^{\varphi(b)} =
\frac{\gamma(b) - z}{\gamma(a) - z} = 1$, de modo que $\varphi(b) \in
2\iu\pi\Z$. Como función de $z$, el índice es
\omterm{def:b3:topology:continuity}{continuo} en
$\C\setminus\operatorname{im}\gamma$ (convergencia dominada), luego
constante en cada \omterm{def:b3:topology:components}{componente
conexa}, y $0$ en la componente no acotada (ML: la integral tiende a $0$
cuando $z \to
\infty$). Para la circunferencia $\gamma(t) = a + r\eu^{\iu t}$, $t
\in \intcc0{2\pi}$: $\operatorname{Ind}_\gamma(z) = 1$ para $z
\in D(a,r)$ (calcúlese en $z = a$: $\frac1{2\iu\pi}
\int_0^{2\pi}\frac{r\iu\eu^{\iu t}}{r\eu^{\iu t}}\dd t = 1$; la
constancia hace el resto).
\end{definition}

\section{El teorema de Cauchy}

\begin{theorem}[Goursat]\label{thm:b3:holomorphic:goursat}
Sean $f \in \mathcal H(\Omega)$ y $T \subseteq \Omega$ un triángulo
macizo cerrado. Entonces $\int_{\partial T}f\,\dd z = 0$ (borde
recorrido una vez, en cualquier orientación).\index{teorema de Goursat}
\end{theorem}

\begin{proof}
Sea $I(T) = \int_{\partial T}f\,\dd z$. Uniendo los puntos medios de los
lados se parte $T$ en cuatro triángulos de tamaño mitad $T^{(1)},
\dots, T^{(4)}$, y las aristas \omterm{def:b3:topology:interior}{interiores} se cancelan por pares: $I(T) =
\sum_iI(T^{(i)})$. Elíjase entre ellos $T_1$ con
$\abs{I(T_1)} \geq \frac14\abs{I(T)}$, e itérese: una sucesión encajada
$T \supseteq T_1 \supseteq T_2 \supseteq\cdots$ con
\[
\abs{I(T_n)} \geq 4^{-n}\abs{I(T)},
\qquad
\operatorname{diam}T_n = 2^{-n}\operatorname{diam}T,
\quad
\operatorname{length}(\partial T_n) =
2^{-n}\operatorname{length}(\partial T).
\]
La intersección $\bigcap T_n$ es un único punto $z_0$
(\omterm{def:b3:topology:compact}{compactos} encajados de diámetros que
tienden a cero, el~\cref{thm:b3:topology:compactprops}(3)).
Derivabilidad en $z_0$: dado $\varepsilon$, para $n$ grande, sobre
$T_n$,
\[
f(z) = f(z_0) + f'(z_0)(z - z_0) + R(z),
\qquad \abs{R(z)} \leq \varepsilon\abs{z - z_0} \leq
\varepsilon\operatorname{diam}T_n .
\]
La parte afín tiene una primitiva: su integral sobre el $\partial T_n$
cerrado es nula (la~\cref{prop:b3:holomorphic:primitive}), y queda
\[
\abs{I(T_n)} = \Bigl|\int_{\partial T_n}R\Bigr|
\leq \varepsilon\operatorname{diam}(T_n)\,
\operatorname{length}(\partial T_n)
= \varepsilon\,4^{-n}\operatorname{diam}(T)
\operatorname{length}(\partial T) .
\]
Comparando con $\abs{I(T_n)} \geq 4^{-n}\abs{I(T)}$:
$\abs{I(T)} \leq \varepsilon\cdot\text{cte}$ para todo
$\varepsilon$: $I(T) = 0$.
\end{proof}

\begin{theorem}[Teorema de Cauchy, versión estrellada]
\label{thm:b3:holomorphic:cauchy}
Sea $\Omega$ \emph{estrellado} respecto de $c$ (todo segmento $[c, z]$,
$z \in \Omega$, está contenido en $\Omega$) --- por ejemplo, convexo.
Toda $f \in \mathcal H(\Omega)$ tiene una primitiva en $\Omega$; en
consecuencia, $\int_\gamma f\,\dd z = 0$ para \emph{todo}
\omterm{def:b3:topology:pathconnected}{camino} cerrado $\gamma$
contenido en $\Omega$.\index{teorema de Cauchy}
\end{theorem}

\begin{proof}
Defínase $F(z) = \int_{[c,z]}f\,\dd w$. Para $z, z + h \in
\Omega$ con $[z, z+h] \subseteq \Omega$ (cierto para $h$ pequeño), el
triángulo de vértices $c, z, z+h$ está contenido en $\Omega$ (por ser
estrellado: cada uno de sus puntos está en un segmento $[c,
w]$ con $w \in [z, z+h] \subseteq \Omega$); Goursat da
\[
F(z + h) - F(z) = \int_{[z, z+h]}f\,\dd w,
\]
y el cálculo del cociente incremental de
la~\cref{prop:b3:holomorphic:primitive} produce $F' = f$. La anulación
de las integrales sobre \omterm{def:b3:topology:pathconnected}{caminos}
cerrados se sigue de la misma proposición.
\end{proof}

\begin{theorem}[Fórmula integral de Cauchy]
\label{thm:b3:holomorphic:formula}
Sean $f \in \mathcal H(\Omega)$, $\bar D(a, r) \subseteq
\Omega$ y $C_r$ la circunferencia $\partial D(a,r)$ recorrida una vez en
sentido positivo. Entonces, para todo $z \in D(a, r)$:\index{fórmula
integral de Cauchy}
\[
f(z) = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - z}\,\dd w .
\]
\end{theorem}

\begin{proof}
Fíjese $z$ y defínase en $\Omega$
\[
g(w) = \begin{cases}
\dfrac{f(w) - f(z)}{w - z} & w \neq z,\\[4pt]
f'(z) & w = z :
\end{cases}
\]
$g$ es \omterm{def:b3:topology:continuity}{continua} en $\Omega$ y
\omterm{def:b3:holomorphic:holo}{holomorfa} fuera de $z$. Goursat vale
para $g$ en todo triángulo $T \subseteq \Omega'$, donde $\Omega'$ es un
disco ligeramente mayor que $\bar D(a,r)$ contenido en $\Omega$: si
$z \notin T$, directamente; si $z \in T$, pártase $T$ en triángulos
pequeños con $z$ por vértice más triángulos que evitan $z$; en un
triángulo de vértice $z$, la cota ML da
$\abs{\int_{\partial T'}g} \leq \sup_{T'}\abs
g\cdot\operatorname{length} \to 0$ cuando el triángulo se encoge, y las
piezas restantes se anulan por Goursat --- luego
$\int_{\partial T}g = 0$ en todos los casos. La demostración
del~\cref{thm:b3:holomorphic:cauchy} solo usó esta propiedad triangular:
$g$ tiene una primitiva en el convexo $\Omega'$, de modo que
$\int_{C_r}g = 0$, i.e.
\[
\frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - z}\dd w
= f(z)\,\frac{1}{2\iu\pi}\int_{C_r}\frac{\dd w}{w - z}
= f(z)\operatorname{Ind}_{C_r}(z) = f(z) .
\]
\end{proof}

\section{Analiticidad y su cascada}

\begin{theorem}[Holomorfa $=$ analítica]
\label{thm:b3:holomorphic:analytic}
Sean $f \in \mathcal H(\Omega)$ y $D(a, R) \subseteq \Omega$.
Entonces\index{analiticidad de las funciones holomorfas}
\[
f(z) = \sum_{n\geq0}c_n\,(z - a)^n \quad \text{en } D(a, R),
\qquad
c_n = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{(w -
a)^{n+1}}\,\dd w \ \ (0 < r < R),
\]
con coeficientes independientes de $r$. En consecuencia, $f$ es
infinitamente $\C$-derivable, $c_n = f^{(n)}(a)/n!$, y valen las
\emph{\omterm{thm:b3:holomorphic:analytic}{estimaciones de Cauchy}}:\index{estimaciones de Cauchy}
\[
\abs{c_n} \;\leq\; \frac{\sup_{\abs{w - a} =
r}\abs{f(w)}}{r^{n}} .
\]
\end{theorem}

\begin{proof}
Para $\abs{z - a} < r$: desarróllese el núcleo de Cauchy en la serie
geométrica
\[
\frac1{w - z} = \frac1{(w - a)\bigl(1 - \frac{z - a}{w -
a}\bigr)} = \sum_{n\geq0}\frac{(z - a)^n}{(w - a)^{n+1}},
\]
normalmente convergente en $w$ sobre $C_r$ ($\abs{\frac{z-a}{w-a}} =
\frac{\abs{z-a}}r < 1$): intégrese término a término contra
$\frac{f(w)}{2\iu\pi}$ (la convergencia uniforme justifica el
intercambio) y aplíquese el~\cref{thm:b3:holomorphic:formula}. Una serie
de potencias es infinitamente derivable con $c_n =
f^{(n)}(a)/n!$ (segundo año), lo que muestra además que los $c_n$ no
dependen de $r$. Las estimaciones: acótese la integral del coeficiente
por ML.
\end{proof}

\begin{example}[Las singularidades dictan los radios]
\label{ex:b3:holomorphic:radius}
¿Por qué la inocente función real $\frac1{1 + x^2}$ tiene en $x = 3$ una
serie de Taylor que converge solo para $\abs{x - 3} <
\sqrt{10}$, si nada va mal sobre la recta real? Porque el teorema
anterior iguala el radio de convergencia en $a$ a la distancia de $a$ al
punto más próximo donde falla la
\omterm{def:b3:holomorphic:holo}{holomorfía}. Aquí
$f(z) = \frac1{1 + z^2}$ es \omterm{def:b3:holomorphic:holo}{holomorfa}
exactamente en $\C\setminus\{\pm\iu\}$, de modo que el desarrollo en
$a = 3$ converge en el mayor disco que evita $\pm\iu$, de radio
$\abs{3 - \iu} = \sqrt{10}$ --- y no puede converger en uno mayor, pues
la suma extendería $f$ \omterm{def:b3:holomorphic:holo}{holomorfamente}
a un \omterm{def:b3:topology:topology}{entorno} de $\pm\iu$, donde
$\abs f \to \infty$. La teoría real ve el misterioso radio $\sqrt{10}$;
el plano complejo ve dos polos. Esta es la regla práctica: \emph{para
hallar un radio de convergencia, localícense las singularidades} --- por
ejemplo, la serie de Taylor de $\tan$ en $0$ tiene radio $\frac\pi2$
(ceros más próximos de $\cos$), y la función generatriz de Bernoulli
$\frac z{\eu^z - 1}$ (el~\cref{pb:b3:holomorphic:1}, Parte VI) tiene
radio $2\pi$ (ceros no nulos más próximos de $\eu^z - 1$:
$\pm2\iu\pi$).
\end{example}

\begin{corollary}[Liouville; d'Alembert--Gauss]
\label{cor:b3:holomorphic:liouville}
Toda función entera acotada es constante. En consecuencia, todo
polinomio no constante sobre $\C$ tiene una raíz: \emph{$\C$ es
\omterm{def:b3:galois:closure}{algebraicamente cerrado}}.\index{teorema
de Liouville}\index{teorema fundamental del álgebra}
\end{corollary}

\begin{proof}
Si $\abs f \leq M$ en $\C$: para todo $a$ y todo $r$, $\abs{c_1(a)}
= \abs{f'(a)} \leq M/r \to 0$: $f' \equiv 0$, y $f$ es constante (en el
\omterm{def:b3:topology:connected}{conexo} $\C$: derivada nula implica
localmente constante --- intégrese a lo largo de segmentos). Si $P$ no
tuviera raíz, $1/P$ sería entera y acotada ($\abs{P(z)} \to
\infty$ cuando $\abs z \to \infty$: el término dominante manda, de modo
que $\abs{1/P}$ es pequeño fuera de un disco grande y
\omterm{def:b3:topology:continuity}{continuo} en el disco
\omterm{def:b3:topology:compact}{compacto}): constante --- absurdo para
$P$ no constante. (El problema de fin de semana da una segunda
demostración, elemental, y las consecuencias
\omterm{def:b3:galois:algebraic}{algebraicas}.)
\end{proof}

\begin{theorem}[Los ceros son aislados; principio de identidad]
\label{thm:b3:holomorphic:identity}
Sean $\Omega$ \emph{\omterm{def:b3:topology:connected}{conexo}} y
$f \in \mathcal
H(\Omega)$, $f \not\equiv 0$. Entonces todo cero $a$ de $f$ tiene
\emph{orden} finito: $f(z) = (z - a)^m\,g(z)$ con $g \in
\mathcal H(\Omega)$, $g(a) \neq 0$, y los ceros de $f$ no tienen punto
de acumulación en $\Omega$. En consecuencia, si dos
funciones \omterm{def:b3:holomorphic:holo}{holomorfas} en $\Omega$
coinciden en un conjunto con un punto de acumulación en $\Omega$,
coinciden en todas partes.\index{principio de identidad}\index{ceros
aislados}
\end{theorem}

\begin{proof}
Sea $Z$ el conjunto de los puntos donde se anulan todas las derivadas de
$f$. $Z$ es cerrado (intersección de cerrados) y abierto: si todas las
$c_n = 0$ en $a$, el desarrollo en serie de potencias da $f
\equiv 0$ en un disco alrededor de $a$. Por
\omterm{def:b3:topology:connected}{conexión}: $Z =
\varnothing$ o $Z = \Omega$; lo segundo queda excluido por $f
\not\equiv 0$. Así, en un cero $a$ algún coeficiente es no nulo: sea $m$
mínimo con $c_m \neq 0$; entonces $f(z) = (z -
a)^m\sum_{n\geq m}c_n(z-a)^{n-m}$ en un disco, y la suma define $g$
\omterm{def:b3:holomorphic:holo}{holomorfa} cerca de $a$ con
$g(a) = c_m \neq 0$; extiéndase $g = f/(z-a)^m$ fuera de $a$
(\omterm{def:b3:holomorphic:holo}{holomorfa} allí). Como $g(a) \neq 0$ y
$g$ es \omterm{def:b3:topology:continuity}{continua}, $f$ no tiene
ningún otro cero en un \omterm{def:b3:topology:topology}{entorno} de
$a$: los ceros son aislados, y un conjunto de puntos aislados no tiene
punto de acumulación en $\Omega$ (un punto de acumulación de ceros es un
cero --- por \omterm{def:b3:topology:continuity}{continuidad} --- y no
sería aislado). Identidad: aplíquese a la diferencia, cuyo conjunto de
ceros tiene un punto de acumulación, lo que la fuerza a la rama
$Z = \Omega$.
\end{proof}

\begin{theorem}[Valor medio y módulo máximo]
\label{thm:b3:holomorphic:maximum}
Sea $f \in \mathcal H(\Omega)$.
\begin{enumerate}
  \item (Valor medio) Para $\bar D(a, r) \subseteq \Omega$:
        $f(a) = \frac1{2\pi}\int_0^{2\pi}f(a +
        r\eu^{\iu t})\,\dd t$.
  \item (Principio del máximo) Si $\Omega$ es
  \omterm{def:b3:topology:connected}{conexo} y $\abs f$ alcanza un
  máximo local en algún punto de $\Omega$, entonces $f$ es constante. En
  consecuencia, para $\Omega$ acotado y $f$
  \omterm{def:b3:topology:continuity}{continua} en $\bar\Omega$:
  $\sup_{\bar\Omega}\abs f =
        \sup_{\partial\Omega}\abs f$.\index{principio del máximo}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) es la fórmula de Cauchy en el \omterm{ex:b3:groups:actions}{centro}:
parametrícese $C_r$. (2) Digamos que $\abs f \leq \abs{f(a)}$ en
$\bar D(a, \rho)
\subseteq \Omega$. Si $f(a) = 0$, $f \equiv 0$ cerca de $a$. En caso
contrario, para $0 < r \leq \rho$, el valor medio da
\[
\abs{f(a)} \leq \frac1{2\pi}\int_0^{2\pi}\abs{f(a +
r\eu^{\iu t})}\,\dd t \leq \abs{f(a)} :
\]
el integrando \omterm{def:b3:topology:continuity}{continuo} no negativo
$\abs{f(a)} - \abs{f(a +
r\eu^{\iu t})}$ tiene media nula, luego se anula: $\abs f$ es
\emph{constante} $= \abs{f(a)} \ne 0$ en el disco. Una
\omterm{def:b3:holomorphic:holo}{función holomorfa} de módulo constante
no nulo en un disco es constante: derivando $P^2 + Q^2 = \text{cte}$
se obtiene $PP_x + QQ_x
= 0$ y $PP_y + QQ_y = 0$; sustituyendo las relaciones de Cauchy--Riemann
$P_y = -Q_x$, $Q_y = P_x$ en la segunda ecuación resulta el sistema
lineal
\[
P\,P_x + Q\,Q_x = 0, \qquad -P\,Q_x + Q\,P_x = 0,
\]
cuyo determinante es $P^2 + Q^2 \neq 0$: $P_x = Q_x = 0$, luego
$f' = P_x + \iu Q_x = 0$ en el disco: $f$ constante allí. El principio
de identidad propaga la constancia a todo $\Omega$. La forma con
frontera: $\abs f$ alcanza su supremo en el
\omterm{def:b3:topology:compact}{compacto} $\bar\Omega$; un máximo
\omterm{def:b3:topology:interior}{interior} hace $f$ constante, y el
supremo se alcanza en la frontera en todos los casos.
\end{proof}

\begin{theorem}[Teorema de convergencia de Weierstrass]
\label{thm:b3:holomorphic:weierstrassconv}
Si $f_n \in \mathcal H(\Omega)$ convergen a $f$ uniformemente en todo
\omterm{def:b3:topology:compact}{compacto} de $\Omega$, entonces
$f \in \mathcal
H(\Omega)$ y $f_n^{(k)} \to f^{(k)}$ uniformemente en los
\omterm{def:b3:topology:compact}{compactos}, para todo
$k$.\index{teorema de convergencia de Weierstrass}
\end{theorem}

\begin{proof}
$f$ es \omterm{def:b3:topology:continuity}{continua}. Para todo
triángulo cerrado $T \subseteq
\Omega$: $\int_{\partial T}f = \lim\int_{\partial T}f_n = 0$
(convergencia uniforme en el \omterm{def:b3:topology:compact}{compacto}
$\partial T$; Goursat para $f_n$). Por el argumento
del~\cref{thm:b3:holomorphic:cauchy}, $f$ tiene primitivas locales (los
discos son convexos; solo se usó la propiedad triangular), es decir,
$f = F'$ localmente con $F$ \omterm{def:b3:holomorphic:holo}{holomorfa};
$F$ es analítica (el~\cref{thm:b3:holomorphic:analytic}), luego también
lo es $f = F'$: \omterm{def:b3:holomorphic:holo}{holomorfa}. (Esto es el
\emph{teorema de Morera}\index{teorema de Morera}:
\omterm{def:b3:topology:continuity}{continua} con integrales
triangulares nulas implica \omterm{def:b3:holomorphic:holo}{holomorfa}.)
Derivadas: para $\bar D(a, 2r) \subseteq \Omega$ y $z \in \bar D(a, r)$,
la fórmula de Cauchy para las derivadas (derívese
el~\cref{thm:b3:holomorphic:formula} bajo la integral, o úsese la
fórmula de los coeficientes) da
\[
\abs{f_n'(z) - f'(z)} =
\Bigl|\frac{1}{2\iu\pi}\int_{C_{2r}}\frac{f_n(w) - f(w)}{(w -
z)^2}\,\dd w\Bigr|
\leq \frac{2r\,\sup_{C_{2r}}\abs{f_n - f}}{r^2} \to 0
\]
uniformemente en $\bar D(a,r)$; recúbrase un
\omterm{def:b3:topology:compact}{compacto} por un número finito de tales
discos, e itérese para los $k$ superiores.
\end{proof}

\begin{method}\label{met:b3:holomorphic:toolkit}
La caja de herramientas cotidiana. Para demostrar que una función es
\omterm{def:b3:holomorphic:holo}{holomorfa}: exhíbase como serie de
potencias, como composición, como límite uniforme sobre \omterm{def:b3:topology:compact}{compactos}
(el~\cref{thm:b3:holomorphic:weierstrassconv}) o como integral con
parámetro \omterm{def:b3:holomorphic:holo}{holomorfo}
(el~\cref{exo:b3:holomorphic:7} --- derívese bajo $\int$ o aplíquese
Morera--Fubini). Para demostrar identidades: demuéstrense en un segmento
o en un subdominio e invóquese el principio de identidad. Para acotar:
\omterm{thm:b3:holomorphic:analytic}{estimaciones de Cauchy} en la mayor
circunferencia disponible. Para demostrar constancia o inexistencia:
Liouville o el principio del máximo. Sépase siempre \emph{dónde} es
\omterm{def:b3:holomorphic:holo}{holomorfa} la función y \emph{qué}
discos caben en
$\Omega$.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:holomorphic:1}
(a) ¿En qué puntos son $z \mapsto \bar z$, $\abs z^2$ y
$\operatorname{Re}z$ derivables en sentido complejo?
¿\omterm{def:b3:holomorphic:holo}{Holomorfas} en algún
abierto? (b) Demostrar que
$P(x, y) = x^2 - y^2$ es la parte real de una
\omterm{def:b3:holomorphic:holo}{función holomorfa} en $\C$, hállese
explícitamente y determínense \emph{todas}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:holomorphic:2}
Calcúlese a partir de las definiciones: $\int_{C}z^n\,\dd z$ para todo
$n
\in \Z$, siendo $C$ la circunferencia unidad; $\int_\gamma\bar z\,\dd z$
a lo largo del segmento $[0, 1+\iu]$ y a lo largo del
\omterm{def:b3:topology:pathconnected}{camino} de dos segmentos que pasa
por $1$: conclúyase que $\bar z$ no tiene primitiva en ningún
\omterm{def:b3:topology:topology}{entorno} de estos
\omterm{def:b3:topology:pathconnected}{caminos}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:3}
(a) Demostrar que el logaritmo principal $\log z = \ln\abs z +
\iu\arg z$ ($\arg \in \intoo{-\pi}\pi$) es
\omterm{def:b3:holomorphic:holo}{holomorfo} en
$\C\setminus\intoc{-\infty}0$ con derivada $\frac1z$ \emph{(primitiva de
$\frac1z$ en el plano hendido, que es estrellado:
el~\cref{thm:b3:holomorphic:cauchy}; ajústese la constante)}. (b)
Demostrar que no existe ningún logaritmo
\omterm{def:b3:topology:continuity}{continuo} en $\C^*$ \emph{(la
obstrucción sin derivadas: el \omterm{def:b3:holomorphic:index}{índice}
de la circunferencia unidad)}. (c) Desarróllese $\log(1 + z)$ en serie
de potencias en $D(0,1)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:4}
(a) Sea $f$ entera con $\abs{f(z)} \leq C(1 + \abs
z)^{n}$. Demostrar que $f$ es un polinomio de grado $\leq n$
\emph{(\omterm{thm:b3:holomorphic:analytic}{estimaciones de Cauchy} en
circunferencias grandes)}. (b) Sea $f$ entera con $\operatorname{Re}f$
acotada superiormente. Demostrar que $f$ es constante \emph{(considérese
$\eu^{f}$)}. (c) Dedúzcase el «pequeño Picard para aplicaciones afines»:
una función entera que omite un semiplano es constante.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:5}
(a) Sea $f$ \omterm{def:b3:holomorphic:holo}{holomorfa} en un
\omterm{def:b3:topology:connected}{conexo} $\Omega \ni 0$ con
$f(\frac1n) = \frac1{n^2}$ para todo $n$ grande. Determínese $f$. (b)
¿Existe alguna \omterm{def:b3:holomorphic:holo}{holomorfa} $f$ en $\C^*$
que cumpla $f(\frac1n) =
\frac{(-1)^n}{n}$ para todo $n \geq 1$? Justifíquese. (c) Exhíbanse dos
funciones \omterm{def:b3:holomorphic:holo}{holomorfas} distintas en
$D(0,1)\sqcup D(3,1)$ que coincidan en $D(0,1)$: ¿dónde usa el principio
de identidad la \omterm{def:b3:topology:connected}{conexión}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:6}
Sea $f$ \omterm{def:b3:holomorphic:holo}{holomorfa} en el disco unidad
abierto $\mathbb D$ y \omterm{def:b3:topology:continuity}{continua} en
$\bar{\mathbb D}$, con $\abs f \equiv 1$ en la circunferencia frontera.
(a) Si $f$ no tiene ningún cero en $\mathbb D$, demuéstrese que $f$ es
constante \emph{(aplíquese el principio del máximo a $f$ y a $1/f$)}.
(b) Dese un ejemplo con un cero en el que $f$ no sea constante.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:7}
(\omterm{def:b3:holomorphic:holo}{Holomorfía} bajo la integral) Sea
$\mu$ una \omterm{def:b3:measure:measure}{medida} finita sobre un
espacio $X$ y $g \colon X\times\Omega \to \C$ con:
$g(x, \cdot) \in \mathcal H(\Omega)$ para cada $x$, $g$
\omterm{def:b3:lebesgue:measurable}{medible} en $x$, y
$\abs g \leq h(x)$ con $h$ \omterm{def:b3:lebesgue:l1}{integrable},
localmente uniformemente en $z$. Demuéstrese que
$G(z) = \int_Xg(x, z)\dd\mu(x)$ es
\omterm{def:b3:holomorphic:holo}{holomorfa} en $\Omega$. \emph{(Morera:
las integrales triangulares se anulan por Fubini y Goursat; la
\omterm{def:b3:topology:continuity}{continuidad}, por convergencia
dominada. Aplíquese después a $\Gamma(z) =
\int_0^\infty t^{z-1}\eu^{-t}\dd t$ en $\{\operatorname{Re}z >
0\}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:8}
(Gauss--Lucas) Sea $P \in \C[X]$ no constante. Demuéstrese que toda raíz
de $P'$ está en la envoltura convexa de las raíces de $P$.
\emph{(Escríbase $\frac{P'}{P} = \sum_k\frac{m_k}{z - a_k}$ en una raíz
$z$ de $P'$ que no sea raíz de $P$, tómense conjugados y léase una
combinación convexa.)} Ilústrese sobre
$P = z^3 - 1$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:9}
Sea $f$ entera y doblemente periódica: $f(z + 1) = f(z + \iu)
= f(z)$ para todo $z$. Demuéstrese que $f$ es constante. \emph{(Acótese
$f$ en el cuadrado fundamental, que es
\omterm{def:b3:topology:compact}{compacto}, y después en todas partes;
Liouville.)} Moraleja: las funciones elípticas no constantes han de
tener polos --- el tema del~\cref{ch:b3:residues}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:10}
(a) Demuéstrese que $P = \operatorname{Re}f$, para $f$
\omterm{def:b3:holomorphic:holo}{holomorfa}, cumple la propiedad del
valor medio $P(a) =
\frac1{2\pi}\int_0^{2\pi}P(a + r\eu^{\iu t})\dd t$ y es \emph{armónica}:
$\partial^2_{xx}P + \partial^2_{yy}P = 0$ (derívese Cauchy--Riemann;
úsese el~\cref{thm:b3:holomorphic:analytic} para la suavidad necesaria).
(b) Dedúzcase el principio del máximo para partes reales de
funciones \omterm{def:b3:holomorphic:holo}{holomorfas} en dominios
acotados.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:11}
(Reflexión de Schwarz) Sean $\Omega^+ = \{z : \abs z < 1,\
\operatorname{Im}z > 0\}$, $I = \intoo{-1}1$ y $f$
\omterm{def:b3:holomorphic:holo}{holomorfa} en $\Omega^+$,
\omterm{def:b3:topology:continuity}{continua} en $\Omega^+\cup I$,
\emph{con valores reales en $I$}. Defínase
\[
F(z) = \begin{cases} f(z) & z \in \Omega^+\cup I,\\
\overline{f(\bar z)} & \bar z \in \Omega^+ . \end{cases}
\]
(a) Demuéstrese que $F$ está bien definida y es
\omterm{def:b3:topology:continuity}{continua} en
$\Omega = \Omega^+\cup I\cup\Omega^-$, y
\omterm{def:b3:holomorphic:holo}{holomorfa} en $\Omega^\pm$ \emph{(para
$\Omega^-$: verifíquese Cauchy--Riemann para $\overline{f(\bar z)}$, o
desarróllese $f$ en serie de potencias local y conjúguense los
coeficientes)}. (b) Demuéstrese que $F$ es
\omterm{def:b3:holomorphic:holo}{holomorfa} en todo $\Omega$ por el
criterio de Morera: $\int_{\partial T}F = 0$ para todo triángulo
$T \subseteq \Omega$ \emph{(pártanse los triángulos en $I$ y sepárense
sus lados horizontales del eje en $\varepsilon$, usando la
\omterm{def:b3:topology:continuity}{continuidad} uniforme)}. (c)
Dedúzcase: una \omterm{def:b3:holomorphic:holo}{función holomorfa} en el
disco, real en un diámetro, cumple $f(\bar z) = \overline{f(z)}$; y una
\omterm{def:b3:holomorphic:holo}{función holomorfa} no constante no
puede tomar valores reales en ningún abierto no vacío de su dominio
(\omterm{def:b3:topology:connected}{conexo}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:12}
(La ecuación pitagórica compleja) Hállense todos los pares de funciones
\emph{enteras} con $f^2 + g^2 = 1$. (a) Demuéstrese que $h = f + \iu g$
es entera y no se anula, y que toda función entera sin ceros es
$\eu^{\varphi}$ para cierta $\varphi$ entera \emph{($h'/h$ es entera,
luego tiene una primitiva en el estrellado $\C$; ajústese la constante y
demuéstrese que $h\eu^{-\varphi}$ es constante)}. (b) Conclúyase
$f = \cos\varphi$, $g = \sin\varphi$ con $\varphi$ entera, y compruébese
el recíproco. ¿Cuáles son las soluciones enteras de $f^2 + g^2 = 0$?
\end{exercise}

\section{Problema: el teorema fundamental del álgebra, dos veces}

\begin{problem}[{Problema de fin de semana --- $\C$ es algebraicamente
cerrado: la demostración de d'Alembert, la de Liouville y la
cosecha}]\label{pb:b3:holomorphic:1} Sean
$P(z) = z^n + a_{n-1}z^{n-1} + \dots + a_0$, $n \geq 1$. Demostraremos
dos veces que $P$ tiene una raíz y recogeremos después lo que el álgebra
lleva esperando desde el~\cref{ch:b3:galois}.

\medskip
\noindent\textbf{Parte I --- Coercividad y mínimo.}
\begin{enumerate}
  \item Demostrar que $\abs{P(z)} \to +\infty$ cuando $\abs z \to
        \infty$: con precisión, $\abs{P(z)} \geq \frac12\abs z^n$ para
  $\abs z \geq R_0$ adecuados. \item Dedúzcase que $\abs P$ alcanza un
  mínimo global en $\C$: existe $z_0$ con $\abs{P(z_0)} = \inf_\C\abs P$
  \emph{(\omterm{def:b3:topology:compact}{compacidad} de un disco cerrado grande,
        \cref{cor:b3:topology:heineborel})}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El descenso de d'Alembert.} Supóngase, por
reducción al absurdo, que $P(z_0) \neq 0$.
\begin{enumerate}[resume]
  \item Desarróllese $Q(h) = P(z_0 + h)/P(z_0)$ como polinomio en $h$:
  $Q(h) = 1 + c_kh^k + h^{k+1}S(h)$ con $c_k \neq
        0$, $k \geq 1$, $S$ un polinomio. \item Elíjase la dirección de
  descenso: para $t > 0$ pequeño, póngase $h = t\,\omega$ donde
  $\omega^k = -1/c_k$ (¿por qué existe tal $\omega$? ---
  \emph{demuéstrese} la existencia de raíces $k$-ésimas de cualquier
  número complejo mediante la forma polar, independientemente del
  teorema que se está demostrando). Demuéstrese
        \[
        \abs{Q(t\omega)} \leq 1 - t^k + C\,t^{k+1}
        \]
        para $t$ pequeño, con una constante explícita $C$. \item
        Conclúyase $\abs{Q(t\omega)} < 1$ para $t$ pequeño --- en
        contradicción con la minimalidad de $\abs{P(z_0)}$. Luego
        $P(z_0) = 0$: \emph{todo polinomio complejo no constante tiene
        una raíz} (d'Alembert--Argand).
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El renglón de Liouville, desarrollado.}
\begin{enumerate}[resume]
  \item Escríbase con cuidado la demostración
  del~\cref{cor:b3:holomorphic:liouville}: si $P$ no tiene raíces,
  verifíquese que $1/P$ es entera y acotada (cuantifíquese usando la
  pregunta 1), luego constante, y conclúyase. Compárense las dos
  demostraciones: ¿qué ingredientes usa cada una? (La \omterm{def:b3:topology:compact}{compacidad} aparece
  en ambas --- ¿dónde?)
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La cosecha.}
\begin{enumerate}[resume]
  \item Demostrar que todo $P \in \C[X]$ de grado $n$ se escinde:
  $P = c\prod_{i}(X - \alpha_i)^{m_i}$ con $\sum m_i =
        n$ \emph{(inducción, división euclídea por $(X -
        \alpha)$)}. \item Demostrar que los polinomios
  \omterm{def:b3:rings:divisibility}{irreducibles} de $\R[X]$ son los de
  grado uno y los de grado dos con discriminante negativo
  \emph{(emparéjense las raíces conjugadas)}; dedúzcase que todo
  polinomio real de grado impar tiene una raíz real y dese una segunda
  demostración de este último hecho, de tipo ordinal (teorema del valor
  intermedio) --- comprobando que coinciden en $X^3 - X - 1$. \item
  Dedúzcanse las deudas que este libro puede ya saldar: (i) todo
  endomorfismo de un $\C$-espacio vectorial no nulo de dimensión finita
  tiene un valor propio, de modo que toda matriz compleja tiene una
  \omterm{thm:b3:modules:jordan}{forma de Jordan}
  (el~\cref{thm:b3:modules:jordan}); (ii) el cuerpo $\bar\Q$ de los
  números \omterm{def:b3:galois:algebraic}{algebraicos} usado en
  la~\cref{rem:b3:galois:closureexamples} es, en efecto, una
  \omterm{def:b3:galois:closure}{clausura algebraica} de $\Q$. \item
  (Final) Señálese con precisión dónde se rompería cada demostración
  sobre un cuerpo como $\Q(\iu)$: ¿qué pasos usan la existencia de
  raíces $k$-ésimas (pregunta 4) y cuáles la \omterm{def:b3:topology:compact}{compacidad} o la
  \omterm{def:b3:complete:complete}{completitud} (preguntas 2 y 6)?
  Conclúyase en cinco líneas: el teorema es genuinamente
  \emph{analítico} --- toda demostración invoca en algún punto la
  \omterm{def:b3:complete:complete}{completitud} o la
  \omterm{def:b3:topology:connected}{conexión} de $\R$ --- aunque su
  enunciado sea puramente \omterm{def:b3:galois:algebraic}{algebraico}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- La escala de rigidez de las funciones
enteras.} Liouville es el primer peldaño de una escala; subámosla.
\begin{enumerate}[resume]
  \item (\omterm{thm:b3:holomorphic:analytic}{Estimaciones de Cauchy}) A
  partir de la fórmula de Cauchy en la circunferencia de radio $r$
  centrada en $a$, demuéstrese
        \[
        \bigl|f^{(n)}(a)\bigr| \leq
        \frac{n!\,\sup_{\abs{z-a}=r}\abs f}{r^n} ,
        \]
        y recupérese Liouville como el caso $n = 1$, $r \to
        \infty$. \item (Crecimiento polinómico) Demuéstrese que una
        función entera $f$ con $\abs{f(z)} \leq A + B\abs z^m$ para todo
        $z$ es un polinomio de grado $\leq m$ \emph{(mátense los
        coeficientes de Taylor más allá de $m$ con la pregunta 11)}.
        \item (Parte real acotada) Demuéstrese que una función entera
        $f$ con $\operatorname{Re}f$ acotada superiormente es constante
        \emph{(aplíquese Liouville a $\eu^{f}$)}. \item (Doble
        periodicidad) Sea $f$ entera con $f(z + 1)
        = f(z)$ y $f(z + \iu) = f(z)$ para todo $z$. Demuéstrese que $f$
        es constante. Conclúyase: una función «elíptica» no constante ha
        de tener singularidades --- la razón histórica por la que los
        polos entran en el análisis complejo. \item (Imagen densa)
        Demuéstrese que la imagen de una función entera no constante es
        densa en $\C$: si $f(\C)$ evita un disco $D(a, r)$, entonces
        $\frac1{f - a}$ es entera y acotada. (Picard demostró que la
        imagen omite a lo sumo \emph{un punto}; la
        \omterm{ex:b3:lebesgue:gamma}{densidad} es el nivel que alcanzan
        nuestras herramientas.) \item (Propia $\Rightarrow$ polinomio)
        Supóngase que $f$ es entera y $\abs{f(z)} \to \infty$ cuando
        $\abs z \to
        \infty$. Demuéstrese: los ceros de $f$ son finitos en número
        ($z_1, \dots, z_p$, con multiplicidades $m_i$); el cociente
        $g = f/\prod(z - z_i)^{m_i}$ es entero y sin ceros; $1/g$ tiene
        crecimiento polinómico, luego (pregunta 12) es un polinomio,
        necesariamente constante (sin ceros); conclúyase que $f$ es un
        polinomio. Así pues, entre las funciones enteras, los polinomios
        son exactamente las propias --- $\eu^z$ falla la propiedad a lo
        largo de
        $\R_-$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Sombras armónicas y un valor medio de
Gauss.}
\begin{enumerate}[resume]
  \item Sea $f = u + \iu v$ \omterm{def:b3:holomorphic:holo}{holomorfa}
  en un abierto. Verifíquese que
  $u = \operatorname{Re}f$ cumple la propiedad del valor medio
        \[
        u(a) = \frac1{2\pi}\int_0^{2\pi}
        u\bigl(a + r\eu^{\iu\theta}\bigr)\,\dd\theta
        \]
        (parte real de la fórmula de Cauchy) y dedúzcase el principio
        del máximo para $u$ en un dominio acotado, con la misma
        demostración por \omterm{def:b3:topology:connected}{conexión}
        que para $\abs f$. \item (Valor medio de Gauss) Para $a \in \C$
        y $r > 0$ con $\abs a \neq r$, demuéstrese
        \[
        \frac1{2\pi}\int_0^{2\pi}
        \log\bigl|a - r\eu^{\iu\theta}\bigr|\,\dd\theta
        = \log\max\bigl(\abs a, r\bigr)
        \]
        \emph{(si $\abs a > r$: $z \mapsto \log\abs{a - z}$ es la parte
        real de un logaritmo \omterm{def:b3:holomorphic:holo}{holomorfo}
        en un \omterm{def:b3:topology:topology}{entorno} del disco
        cerrado --- ¿por qué existe uno? ---, de modo que se aplica la
        pregunta 17; si $\abs a < r$: factorícese
        $\abs{a - r\eu^{\iu\theta}} = r\,\abs{1 -
        \frac ar\eu^{-\iu\theta}}$ y reutilícese el primer caso)}. \item
        (\omterm{def:b3:measure:measure}{Medida} de Mahler) Para
        $P = c\prod_{i=1}^n(X -
        \alpha_i) \in \C[X]$, dedúzcase la \emph{fórmula de Jensen para
        polinomios}:
        \[
        \frac1{2\pi}\int_0^{2\pi}\log\bigl|P(\eu^{\iu\theta})
        \bigr|\,\dd\theta
        = \log\Bigl(\abs c\prod_{i}\max(1,
        \abs{\alpha_i})\Bigr) :
        \]
        la media geométrica de $\abs P$ sobre la circunferencia unidad
        lee las raíces exteriores al disco. Verifíquese en $P =
        X^2 - X$ y en $P = 2X - 1$. \item (Números de Bernoulli)
        Defínanse los coeficientes $B_n$ por
        $\frac{z}{\eu^z - 1} = \sum_{n\geq0}\frac{B_n}{n!}
        z^n$ cerca de $0$ (¿por qué es analítico el miembro izquierdo en
        $0$?). Dedúzcase la recurrencia
        $\sum_{k=0}^{n}\binom{n+1}kB_k = 0$ ($n \geq 1$) de
        $(\eu^z - 1)\cdot\frac z{\eu^z-1} = z$, calcúlense $B_0,
        \dots, B_6$ y demuéstrese $B_{2k+1} = 0$ para $k \geq 1$
        \emph{(la función $\frac z{\eu^z-1} + \frac z2$ es par)}. Estos
        números pondrán precio a cada $\zeta(2k)$ en
        \cref{ch:b3:residues}.
  \item (Realidad) Demuéstrese que una función entera que toma valores
  reales en $\R$ cumple $f(\bar z) =
        \overline{f(z)}$ en todas partes \emph{(compárense los
  coeficientes de Taylor en $0$, o aplíquese el principio de identidad a
  $z \mapsto \overline{f(\bar z)}$)}; dedúzcase de nuevo que las raíces
  no reales de los polinomios reales van por pares conjugados (el
  emparejamiento de la pregunta 8, demostrado ahora analíticamente).
  \item (La moraleja) Móntese la escala de rigidez: acotada
  $\Rightarrow$ constante; acotada polinómicamente $\Rightarrow$
  polinomio; propia $\Rightarrow$ polinomio; omite un disco
  $\Rightarrow$ constante; doblemente periódica $\Rightarrow$ constante.
  Contrástese en un breve párrafo con las funciones reales
  $\mathcal C^\infty$ (funciones meseta,
  el~\cref{thm:b3:lp:regularization}): ¿por qué la
  \omterm{def:b3:holomorphic:holo}{holomorfía}, condición puramente
  local, impone ley y orden globales?
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- Última cosecha.}
\begin{enumerate}[resume]
  \item (Desigualdad de Landau) Para $P = \sum_{k=0}^na_kX^k$,
  demuéstrese el valor medio $\frac1{2\pi}\int_0^{2\pi}
        \abs{P(\eu^{\iu\theta})}^2\dd\theta =
        \sum_k\abs{a_k}^2$ (ortogonalidad de los $\eu^{\iu k\theta}$) y,
  usando después la cota puntual $\log t \leq t - 1$ para comparar las
  medias de $\log\abs P^2$ y $\abs P^2$, dedúzcase de la pregunta 19 que
        \[
        \abs c\prod_{i}\max\bigl(1, \abs{\alpha_i}\bigr)
        \;\leq\;
        \Bigl(\sum_{k=0}^{n}\abs{a_k}^2\Bigr)^{1/2} :
        \]
        el producto de las raíces exteriores al disco unidad está
        controlado por el tamaño $\ell^2$ de los coeficientes (trátense
        las raíces \emph{sobre} la circunferencia aplicando la
        desigualdad a $P(rX)$ y haciendo $r \to 1$). Compruébese en
        $X^2 - X$. \item (Los números de Bernoulli crecen
        factorialmente) Demuéstrese que el radio de convergencia de
        $\sum\frac{B_n}{n!}z^n$ es \emph{exactamente} $2\pi$: al menos
        $2\pi$ porque $z/(\eu^z - 1)$ se extiende
        \omterm{def:b3:holomorphic:holo}{holomorfamente} a $D(0, 2\pi)$,
        y a lo sumo $2\pi$ porque, si no, la suma se mantendría acotada
        cerca de $2\pi\iu$, donde $\abs{z/(\eu^z-1)} \to \infty$.
        Dedúzcase
        \[
        \limsup_{k\to\infty}
        \Bigl(\frac{\abs{B_{2k}}}{(2k)!}\Bigr)^{1/2k}
        = \frac1{2\pi} :
        \]
        los números de Bernoulli crecen factorialmente. Admitiendo
        $B_{12}
        = -\frac{691}{2730}$ (la recurrencia de la pregunta 20, llevada
        más lejos), compárese $\abs{B_{12}} \approx 0.25311$ con la
        predicción más fina $2\,(2k)!/(2\pi)^{2k}
        \approx 0.25305$ en $k = 6$ --- cuatro cifras coincidentes de
        una ley asintótica que el~\cref{ch:b3:residues} demostrará
        exactamente, vía $\zeta(2k)$. \item (Las raíces se mueven
        \omterm{def:b3:topology:continuity}{continuamente}) Sean $(P_j)$
        mónicos de grado $n$ con coeficientes convergentes a los de $P$
        (mónico). Demuéstrese la \emph{cota de Cauchy}: toda raíz de un
        mónico $Q = X^n + \sum_{k<n}q_kX^k$ cumple
        $\abs\alpha \leq 1 + \max_k\abs{q_k}$; dedúzcase que las raíces
        de los $P_j$ permanecen en un
        \omterm{def:b3:topology:compact}{compacto} fijo y, extrayendo
        subsucesiones convergentes de vectores de raíces y pasando al
        límite en la factorización de la pregunta 7, que los
        multiconjuntos de raíces de $P_j$ convergen al de $P$.
        Demuéstrese, por último, que la
        \omterm{def:b3:topology:continuity}{continuidad} es lo mejor que
        puede decirse: para $P_\varepsilon = X^2 - 2X + 1 +
        \varepsilon$, una perturbación de tamaño $\varepsilon =
        10^{-4}$ desplaza la raíz doble $1$ en $10^{-2}$ --- exponente
        de Hölder $\frac1m$ en una raíz de multiplicidad $m$, jamás
        lipschitziano: numéricamente, las raíces múltiples cuestan la
        mitad de las cifras.
\end{enumerate}
\end{problem}
