\chapter{Medidas producto, Fubini, cambio de variable}
\label{ch:b3:product}

La teoría de Lebesgue unidimensional se convierte en cálculo
multidimensional gracias a dos teoremas. \emph{Tonelli--Fubini} dice que
las integrales sobre productos son integrales iteradas --- cortar en
rebanadas es lícito, en cualquier orden, bajo hipótesis que uno puede
comprobar de verdad. La \emph{fórmula del cambio de variable} transporta
integrales a lo largo de difeomorfismos $\mathcal C^1$, con el
determinante jacobiano como tipo de cambio del volumen; la demostramos
por completo, empezando por el caso lineal, donde explica qué \emph{es}
el determinante. Las aplicaciones se encadenan: la fórmula de las capas,
la convolución, las \omterm{ex:b3:product:polar}{coordenadas polares},
el volumen de la bola de $n$ dimensiones --- y, en el problema de fin de
semana, la fórmula de Stirling con un análisis honesto del error.

\section{$\sigma$-álgebras producto y medidas producto}

\begin{definition}\label{def:b3:product:sigma}
Para espacios \omterm{def:b3:lebesgue:measurable}{medibles}
$(X, \mathcal A)$ y $(Y, \mathcal B)$, la \emph{$\sigma$-álgebra
producto}\index{$\sigma$-álgebra producto}
$\mathcal A \otimes \mathcal B$ sobre $X \times
Y$ está generada por los \emph{rectángulos} $A \times B$ ($A \in
\mathcal A$, $B \in \mathcal B$) --- un $\pi$-sistema. Para $E
\subseteq X\times Y$ y $x \in X$, la \emph{sección} es $E_x
= \{y : (x,y) \in E\}$; y para una función $f$ sobre el producto,
$f_x = f(x, \cdot)$.
\end{definition}

\begin{proposition}\label{prop:b3:product:sections}
(a) Si $E \in \mathcal A\otimes\mathcal B$, toda sección $E_x
\in \mathcal B$ (y simétricamente); si $f$ es $\mathcal
A\otimes\mathcal B$-medible, toda $f_x$ es $\mathcal
B$-medible.
(b) $\mathcal B(\R^m)\otimes\mathcal B(\R^n) = \mathcal
B(\R^{m+n})$.
\end{proposition}

\begin{proof}
(a) Buenos conjuntos: $\{E : E_x \in \mathcal B\ \forall x\}$ es una
$\sigma$-álgebra (las secciones conmutan con complementos y uniones
numerables) que contiene los rectángulos. Para $f$:
$(f_x)^{-1}(B) = (f^{-1}(B))_x$.
(b) ($\subseteq$) Rectángulos de conjuntos de Borel: basta ver que las
cajas abiertas$\times$abiertas son de Borel en $\R^{m+n}$ (son abiertas)
y que los rectángulos de Borel generales son límites --- de nuevo,
buenos conjuntos: $\{A : A\times\R^n \in \mathcal B(\R^{m+n})\}$ es una
$\sigma$-álgebra que contiene los abiertos; interséquense dos de ellas.
($\supseteq$) Todo abierto de
$\R^{m+n}$ es unión numerable de cajas abiertas racionales $U\times V$:
está contenido en la $\sigma$-álgebra producto.
\end{proof}

\begin{theorem}[Medida producto]\label{thm:b3:product:existence}
Sean $(X, \mathcal A, \mu)$ y $(Y, \mathcal B, \nu)$
\emph{$\sigma$-finitas}. Para todo $E \in \mathcal
A\otimes\mathcal B$, la función $x \mapsto \nu(E_x)$ es
\omterm{def:b3:lebesgue:measurable}{medible}, y\index{medida producto}
\[
(\mu\otimes\nu)(E) = \int_X \nu(E_x)\,\dd\mu(x)
\]
define la única \omterm{def:b3:measure:measure}{medida} sobre
$\mathcal A\otimes\mathcal B$ con
$(\mu\otimes\nu)(A\times B) = \mu(A)\nu(B)$. Es $\sigma$-finita y
simétrica: la misma \omterm{def:b3:measure:measure}{medida} se obtiene
integrando las $x$-secciones contra $\nu$.
\end{theorem}

\begin{proof}
\emph{\omterm{def:b3:lebesgue:measurable}{Medibilidad} de
$x \mapsto \nu(E_x)$.} Sea primero $\nu$ finita. La clase $\mathcal D$
de los $E$ para los que la aplicación es
\omterm{def:b3:lebesgue:measurable}{medible} contiene los rectángulos
($\nu((A\times B)_x) =
\nu(B)\mathbf 1_A(x)$) y es un $\lambda$-sistema: para $E
\subseteq F$ en $\mathcal D$, $\nu((F\setminus E)_x) =
\nu(F_x) - \nu(E_x)$ (finitud); para $E_n \uparrow E$,
$\nu((E_n)_x) \uparrow \nu(E_x)$
(\omterm{def:b3:topology:continuity}{continuidad} por abajo), y los
límites monótonos de funciones
\omterm{def:b3:lebesgue:measurable}{medibles} son \omterm{def:b3:lebesgue:measurable}{medibles}. Los
rectángulos forman un $\pi$-sistema: Dynkin
(el~\cref{thm:b3:measure:dynkin}) da $\mathcal D = \mathcal
A\otimes\mathcal B$. Si $\nu$ es $\sigma$-finita, escríbase $Y =
\bigcup Y_k$, $Y_k \uparrow$, $\nu(Y_k) < \infty$: $\nu(E_x) =
\lim_k\nu_k(E_x)$ con las $\nu_k = \nu(\cdot\cap Y_k)$ finitas.

\emph{\omterm{def:b3:measure:measure}{Medida}.} La $\sigma$-aditividad
de $E \mapsto
\int\nu(E_x)\dd\mu$ se sigue del~\cref{cor:b3:lebesgue:additivity} (las
secciones de conjuntos disjuntos son disjuntas). Sobre los rectángulos
da $\mu(A)\nu(B)$. \emph{Unicidad}: dos candidatas coinciden sobre el
$\pi$-sistema de los rectángulos; la $\sigma$-finitud proporciona
rectángulos $X_k\times
Y_k \uparrow X\times Y$ de \omterm{def:b3:measure:measure}{medida}
finita: el~\cref{thm:b3:measure:uniqueness}. Simetría: la construcción
en el otro orden es también una \omterm{def:b3:measure:measure}{medida}
que coincide sobre los rectángulos --- única, luego la misma.
\end{proof}

\begin{definition}\label{def:b3:product:lebesgue}
La \emph{medida de Lebesgue sobre $\R^d$}\index{medida de Lebesgue sobre
$\R^d$} es $\lambda_d = \lambda\otimes\cdots\otimes\lambda$ ($d$
factores; la asociatividad de la construcción se comprueba sobre las
cajas y se propaga por unicidad). Es la única
\omterm{def:b3:measure:measure}{medida} de Borel que da a cada caja
$\prod\intoc{a_i}{b_i}$ su volumen $\prod(b_i - a_i)$; es invariante por
traslaciones (los trasladados coinciden sobre las cajas),
$\sigma$-finita y \omterm{def:b3:complete:complete}{completa} tras la
compleción de Carathéodory ---
escribimos $\lambda_d$ para la \omterm{def:b3:measure:measure}{medida}
completada e integramos en consecuencia.
\end{definition}

\section{Tonelli y Fubini}

\begin{theorem}[Tonelli]\label{thm:b3:product:tonelli}
Sean $\mu, \nu$ $\sigma$-finitas y $f \colon X\times Y \to [0,
+\infty]$ \omterm{def:b3:lebesgue:measurable}{medible}. Entonces
$x \mapsto \int_Y f_x\,\dd\nu$ es
\omterm{def:b3:lebesgue:measurable}{medible} y\index{teorema de Tonelli}
\[
\int_{X\times Y}f\,\dd(\mu\otimes\nu)
= \int_X\Bigl(\int_Y f(x,y)\,\dd\nu(y)\Bigr)\dd\mu(x)
= \int_Y\Bigl(\int_X f(x,y)\,\dd\mu(x)\Bigr)\dd\nu(y).
\]
\end{theorem}

\begin{proof}
La máquina estándar. Para $f = \mathbf 1_E$ esto es
el~\cref{thm:b3:product:existence} (y su forma simétrica). Por
linealidad vale para $f \geq 0$ simple. Para $f \geq
0$ general: tómense $s_n \nearrow f$ simples
(el~\cref{thm:b3:lebesgue:approximation}); entonces $\int_Y(s_n)_x
\dd\nu \nearrow \int_Y f_x\dd\nu$ para cada $x$ (convergencia monótona
en $Y$), de modo que los miembros izquierdos convergen por convergencia
monótona en $X$, mientras que
$\int s_n\,\dd(\mu\otimes\nu) \nearrow \int f$ por convergencia monótona
sobre el producto.
\end{proof}

\begin{theorem}[Fubini]\label{thm:b3:product:fubini}
Sean $\mu, \nu$ $\sigma$-finitas y $f \in L^1(\mu\otimes\nu)$. Entonces,
para $\mu$-casi todo $x$, la sección $f_x$ es $\nu$-integrable, la
función definida en casi todo punto $x \mapsto \int f_x\dd\nu$ es
\omterm{def:b3:lebesgue:l1}{integrable}, y las dos integrales iteradas
valen ambas $\int f\,\dd(\mu\otimes\nu)$.\index{teorema de Fubini}
\end{theorem}

\begin{proof}
Tonelli aplicado a $\abs f$ muestra que $\varphi(x) = \int\abs{f_x}
\dd\nu$ tiene integral finita y, por tanto, es finita en casi todo
punto: $f_x \in
L^1(\nu)$ para casi todo $x$. Descompóngase $f = f^+ - f^-$ (caso real;
el complejo, por componentes): Tonelli calcula cada integral iterada de
$f^\pm$ como $\int f^\pm\dd(\mu\otimes\nu) <
\infty$, y la diferencia, definida en casi todo punto, integra a la
diferencia. Simétricamente en el otro orden.
\end{proof}

\begin{method}\label{met:b3:product:fubini}
Para intercambiar dos integrales (o una integral y una suma, o dos
sumas): si el integrando es \emph{no negativo}, intercámbiese sin más
(Tonelli). En caso contrario, aplíquese primero Tonelli a $\abs f$ en el
orden que resulte más fácil de estimar; si el resultado es finito,
Fubini legitima el intercambio. Nunca se salte la comprobación de
$\abs f$: el integrando del~\cref{exo:b3:product:4} tiene dos integrales
iteradas con \emph{valores distintos}.
\end{method}

\begin{proposition}[Fórmula de las
capas]\label{prop:b3:product:layercake}
Para $f \geq 0$ \omterm{def:b3:lebesgue:measurable}{medible} sobre
$(X, \mathcal A, \mu)$ $\sigma$-finita:\index{fórmula de las capas}
\[
\int_X f\,\dd\mu = \int_0^{+\infty}\mu(\{f > t\})\,\dd t,
\qquad
\int_X f^p\,\dd\mu = p\int_0^{+\infty}t^{p-1}\mu(\{f >
t\})\,\dd t \quad (p \geq 1).
\]
\end{proposition}

\begin{proof}
Aplíquese Tonelli a $\mathbf 1_{\{(x,t) : 0 < t < f(x)\}}$ sobre $X
\times \intoo0{+\infty}$ (es
\omterm{def:b3:lebesgue:measurable}{medible}: se trata de
$\{(x,t): f(x) - t
> 0\}\cap\{t > 0\}$, una combinación de tipo Borel de la
\omterm{def:b3:lebesgue:measurable}{medible} $(x,t)\mapsto f(x) - t$):
integrando primero en $t$ se obtiene $\int f\,\dd\mu$; primero en $x$,
$\int_0^\infty\mu(f >
t)\dd t$. Para $f^p$: sustitúyase $t = s^p$ en $\int\mu(f^p > t)
\dd t$, es decir, aplíquese la primera fórmula a $f^p$ y cámbiese de
variable en la integral unidimensional ($\{f^p > s^p\} =
\{f > s\}$).
\end{proof}

\begin{theorem}[Convolución en $L^1$]\label{thm:b3:product:convolution}
Para $f, g \in L^1(\R^d, \lambda_d)$, la integral
\[
(f * g)(x) = \int_{\R^d} f(x - y)\,g(y)\,\dd y
\]
converge absolutamente para casi todo $x$, define $f * g \in
L^1(\R^d)$ con $\norm{f*g}_1 \leq \norm f_1\norm g_1$, y $*$ es
conmutativa y asociativa.\index{convolución}
\end{theorem}

\begin{proof}
$(x, y) \mapsto f(x-y)g(y)$ es
\omterm{def:b3:lebesgue:measurable}{medible} ($ (x,y)\mapsto x -
y$ es \omterm{def:b3:topology:continuity}{continua}; compóngase y
multiplíquese). Tonelli:
\[
\int\!\!\int \abs{f(x-y)}\abs{g(y)}\,\dd y\,\dd x
= \int\abs{g(y)}\Bigl(\int\abs{f(x - y)}\dd x\Bigr)\dd y
= \norm f_1\norm g_1 < \infty
\]
(invariancia por traslaciones de $\lambda_d$ en la integral \omterm{def:b3:topology:interior}{interior}).
Así, la integral doble es finita; Fubini da la convergencia absoluta en
casi todo punto y la cota de la norma $\norm{f*g}_1 \leq \norm
f_1\norm g_1$. Conmutatividad: sustitúyase $y \mapsto x - y$
(invariancia por traslaciones y por reflexión --- la invariancia por
reflexión vale sobre las cajas y, por unicidad, en todas partes).
Asociatividad: Tonelli--Fubini sobre una integral triple.
\end{proof}

\section{Cambio de variable}

\begin{theorem}[Cambio de variable lineal]
\label{thm:b3:product:linearchange}
Para $T \in GL_d(\R)$ y $A \in \mathcal B(\R^d)$:
$\lambda_d(T(A)) = \abs{\det T}\,\lambda_d(A)$; en consecuencia,
$\int f(y)\dd y = \abs{\det T}\int f(Tx)\,\dd x$ para $f \geq 0$ o
\omterm{def:b3:lebesgue:l1}{integrable}.\index{cambio de variable
lineal}
\end{theorem}

\begin{proof}
La \omterm{def:b3:measure:measure}{medida} $\mu_T(A) = \lambda_d(T(A))$
es una \omterm{def:b3:measure:measure}{medida} de Borel (los
\omterm{def:b3:topology:continuity}{homeomorfismos} conservan los
conjuntos de Borel, el~\cref{pb:b3:measure:1}), invariante por
traslaciones ($T(A + x) = T(A) + Tx$) y finita sobre la caja unidad: por
la caracterización de la \omterm{def:b3:measure:lebesgueouter}{medida de
Lebesgue} (el~\cref{exo:b3:measure:6}, cuya demostración vale
literalmente en $\R^d$ con cubos diádicos), $\mu_T = c(T)\lambda_d$ con
$c(T)
= \lambda_d(T(\intco01^d))$. La aplicación $T \mapsto c(T)$ es
multiplicativa ($c(ST) = c(S)c(T)$, componiendo), de modo que basta
calcular $c$ sobre generadores de $GL_d$: las matrices elementales.
Diagonal $\operatorname{diag}(a, 1, \dots, 1)$: lleva el cubo unidad a
una caja de volumen $\abs a$: $c = \abs a =
\abs\det$. Trasposición de coordenadas: permuta el cubo: $c
= 1 = \abs\det$. Transvección $T(x) = x + \alpha
x_2e_1$: la imagen del cubo unidad es un prisma cizallado; por Tonelli,
su \omterm{def:b3:measure:measure}{medida} es
$\int\lambda_1(\text{sección})\dd x_2
\cdots \dd x_d = 1$, pues cada $x_1$-sección es un intervalo de longitud
$1$: $c = 1 = \abs{\det}$. Toda matriz invertible es producto de estas
(eliminación gaussiana), y tanto $c$ como $\abs\det$ son
multiplicativos: $c(T) = \abs{\det T}$. La fórmula integral se sigue por
la máquina estándar (indicadores, simples, convergencia monótona).
\end{proof}

\begin{theorem}[Cambio de variable]\label{thm:b3:product:changeofvar}
Sean $U, V \subseteq \R^d$ abierto y $\Phi \colon U \to V$ un
difeomorfismo $\mathcal C^1$. Para toda $f \colon V
\to [0, +\infty]$ \omterm{def:b3:lebesgue:measurable}{medible} (o
$f \in L^1(V)$):\index{fórmula del cambio de variable}
\[
\int_V f(y)\,\dd y = \int_U
f\bigl(\Phi(x)\bigr)\,\abs{\det D\Phi(x)}\,\dd x .
\]
\end{theorem}

\begin{proof}
Escribamos $J(x) = \abs{\det D\Phi(x)}$. El corazón de la demostración
es la desigualdad
\[
\lambda_d\bigl(\Phi(A)\bigr) \leq \int_A J\,\dd\lambda_d
\qquad\text{para todo boreliano } A \subseteq U;
\tag{$*$}
\]
El paso 4 de más abajo eleva $(*)$ --- aplicada a $\Phi$ y a $\Phi^{-1}$
--- hasta la igualdad del teorema. Obsérvese que $\Phi^{-1}$ es él mismo
un difeomorfismo $\mathcal C^1$ con jacobiano
$\abs{\det D\Phi^{-1}(y)} = J(\Phi^{-1}y)^{-1}$ (regla de la cadena en
$\Phi\circ\Phi^{-1} = \mathrm{id}$).

\emph{Paso 1: $(*)$ para cubos, con un factor de distorsión.} Fíjese un
cubo cerrado $Q \subseteq U$ de \omterm{ex:b3:groups:actions}{centro}
$x_0$ y lado $2r$ (bola para la norma del supremo). Afirmación: para
todo $\varepsilon > 0$, si $\Phi$ es derivable en $Q$ con
$\norm{D\Phi(x) - D\Phi(x_0)} \leq \varepsilon$ sobre $Q$
(\omterm{def:b3:banach:operator}{norma de operador} para la norma del
supremo), entonces
\[
\Phi(Q) \subseteq \Phi(x_0) + D\Phi(x_0)\Bigl(\,\bigl(1 +
\varepsilon\norm{D\Phi(x_0)^{-1}}\bigr)\,(Q - x_0)\Bigr),
\]
porque, para $x \in Q$, la desigualdad del valor medio aplicada a
$\Phi(x) - \Phi(x_0) - D\Phi(x_0)(x - x_0)$ da
$\norm{\Phi(x) - \Phi(x_0) - D\Phi(x_0)(x-x_0)}_\infty \leq
\varepsilon\norm{x - x_0}_\infty \leq \varepsilon r$, y
$D\Phi(x_0)^{-1}$ absorbe ese defecto en un agrandamiento del cubo por
$\varepsilon\norm{D\Phi(x_0)^{-1}}\,r$. Por
el~\cref{thm:b3:product:linearchange},
\[
\lambda_d(\Phi(Q)) \leq \abs{\det D\Phi(x_0)}\,
\bigl(1 + \varepsilon\,C\bigr)^{d}\,\lambda_d(Q),
\qquad C = \sup_{Q}\norm{D\Phi(\cdot)^{-1}} .
\]

\emph{Paso 2: $(*)$ para cubos
\omterm{def:b3:topology:compact}{compactos}, por subdivisión.} Sean $Q
\subseteq U$ un cubo \omterm{def:b3:topology:compact}{compacto} y
$\varepsilon > 0$. Sobre $Q$, $D\Phi$ es uniformemente
\omterm{def:b3:topology:continuity}{continua} y $\norm{D\Phi^{-1}}$ está
acotada (\omterm{def:b3:topology:compact}{compacidad}); subdivídase $Q$ en $2^{kd}$ subcubos $Q_i$ lo
bastante pequeños como para que la oscilación de $D\Phi$ en cada uno sea
$\leq \varepsilon$. El paso 1 en cada subcubo (de
\omterm{ex:b3:groups:actions}{centro} $x_i$):
\[
\lambda_d(\Phi(Q)) \leq \sum_i\lambda_d(\Phi(Q_i))
\leq (1 + C\varepsilon)^d \sum_i \abs{\det
D\Phi(x_i)}\,\lambda_d(Q_i)
\leq (1 + C\varepsilon)^d\Bigl(\int_Q J + \varepsilon'\Bigr),
\]
el último paso porque $\sum_i\abs{\det D\Phi(x_i)}\mathbf
1_{Q_i} \to J$ uniformemente sobre $Q$
(\omterm{def:b3:topology:continuity}{continuidad} de $\det D\Phi$) ---
comparación con sumas de Riemann. Hágase $\varepsilon \to 0$: $(*)$ vale
para los cubos \omterm{def:b3:topology:compact}{compactos}.

\emph{Paso 3: $(*)$ para todo $A$ de Borel.} La función de conjunto $A
\mapsto \lambda_d(\Phi(A))$, sobre los subconjuntos de Borel de $U$, es
una \omterm{def:b3:measure:measure}{medida} ($\Phi$ es una biyección
sobre $V$ que conserva los conjuntos de Borel y la disjunción
numerable), y también lo es $A \mapsto \int_AJ$. Todo abierto de $U$ es
unión numerable de cubos diádicos
\omterm{def:b3:topology:compact}{compactos} casi disjuntos
(descomposición diádica estándar: tómense los cubos diádicos maximales
contenidos en el abierto), y ambas
\omterm{def:b3:measure:measure}{medidas} son aditivas a través de ellos
(las fronteras de los cubos son $\lambda_d$-nulas, y sus imágenes por
$\Phi$ son nulas por el paso 2 aplicado a recubrimientos finos de las
caras por cubos): $(*)$ pasa de los cubos a los
abiertos. Para $A$ de Borel general:
agótese $U$ mediante \omterm{def:b3:topology:compact}{compactos}
$K_m \uparrow U$ con $K_m \subseteq
\mathring K_{m+1}$ y fíjese $m$; sobre $\mathring K_{m+1}$, $J$ está
acotada por cierto $M_m$. Por regularidad exterior de $\lambda_d$
(demostración como en el~\cref{thm:b3:measure:regularity}, con cajas),
elíjanse abiertos $O_n$ con
$A \cap K_m \subseteq O_n
\subseteq \mathring K_{m+1}$ y $\lambda_d\bigl(O_n \setminus
(A\cap K_m)\bigr) \to 0$. Entonces
\[
\lambda_d\bigl(\Phi(A\cap K_m)\bigr) \leq
\lambda_d\bigl(\Phi(O_n)\bigr) \leq \int_{O_n}J
\leq \int_{A\cap K_m}J + M_m\,\lambda_d\bigl(O_n\setminus(A\cap
K_m)\bigr) \xrightarrow[n\to\infty]{} \int_{A\cap K_m}J .
\]
Hágase $m \to \infty$: \omterm{def:b3:topology:continuity}{continuidad}
por abajo a la izquierda, convergencia monótona a la derecha. Esto
establece $(*)$.

\emph{Paso 4: igualdad y fórmula integral.} Extendamos primero $(*)$ de
conjuntos a integrales: para toda $g
\geq 0$ \omterm{def:b3:lebesgue:measurable}{medible} sobre $V$,
\[
\int_V g(y)\,\dd y \leq \int_U g(\Phi(x))\,J(x)\,\dd x .
\tag{$**$}
\]
En efecto, para $g = \mathbf 1_B$ esto es $(*)$ con $A =
\Phi^{-1}(B)$; la linealidad lo extiende a $g$ simple, y la convergencia
monótona a toda $g \geq 0$ (la máquina estándar). Apliquemos ahora
$(**)$ dos veces: primero a $g$ y después --- para el difeomorfismo
$\Phi^{-1}$ --- a la función $x \mapsto g(\Phi(x))J(x)$:
\[
\int_Vg \leq \int_U g(\Phi(x))J(x)\dd x
\leq \int_V g(y)\,J(\Phi^{-1}y)\,\abs{\det
D\Phi^{-1}(y)}\,\dd y = \int_V g ,
\]
puesto que $J(\Phi^{-1}y)\abs{\det D\Phi^{-1}(y)} = \abs{\det\bigl(
D\Phi(\Phi^{-1}y)\,D\Phi^{-1}(y)\bigr)} = 1$ (regla de la cadena en
$\Phi\circ\Phi^{-1} = \mathrm{id}$). Todas las desigualdades son
igualdades: la fórmula vale para $g \geq 0$ y, por descomposición, para
las funciones $L^1$.
\end{proof}

\begin{example}[Coordenadas polares; la gaussiana de nuevo]
\label{ex:b3:product:polar}
$\Phi(r, \theta) = (r\cos\theta, r\sin\theta)$ es un difeomorfismo
$\mathcal
C^1$ de $\intoo0{+\infty}\times\intoo0{2\pi}$ sobre $\R^2$ menos una
semirrecta (conjunto nulo), con $\det D\Phi =
r$:\index{coordenadas polares}
\[
\int_{\R^2}f(x, y)\,\dd x\,\dd y
= \int_0^{2\pi}\!\!\int_0^{+\infty}
f(r\cos\theta, r\sin\theta)\,r\,\dd r\,\dd\theta .
\]
Para $f = \eu^{-x^2-y^2}$, Tonelli y esta fórmula dan
\[
G^2 = \Bigl(\int_\R \eu^{-x^2}\dd x\Bigr)^2
= \int_{\R^2}\eu^{-x^2-y^2}
= 2\pi\int_0^\infty r\eu^{-r^2}\dd r = \pi:
\]
la clásica demostración de dos líneas de $G = \sqrt\pi$, ahora
plenamente justificada (compárese con la demostración por parámetros
del~
\cref{pb:b3:lebesgue:1}).
\end{example}

\begin{theorem}[Volumen de la bola unidad]
\label{thm:b3:product:ballvolume}
Sea $v_d = \lambda_d(B(0,1))$ en $\R^d$. Entonces\index{volumen de la
bola unidad}
\[
v_d = \frac{\pi^{d/2}}{\Gamma\bigl(\frac d2 + 1\bigr)} :
\qquad
v_1 = 2,\quad v_2 = \pi,\quad v_3 = \tfrac{4\pi}3,\quad
v_4 = \tfrac{\pi^2}2,\ \dots
\]
\end{theorem}

\begin{proof}
Calcúlese $I = \int_{\R^d}\eu^{-\norm x_2^2}\dd\lambda_d$ de dos
maneras. Por Tonelli se factoriza: $I = G^d = \pi^{d/2}$. Por la fórmula
de las capas (la~\cref{prop:b3:product:layercake}) con $f = \eu^{-
\norm x^2}$, cuyos conjuntos de nivel son bolas: $\{f > t\} =
B\bigl(0, \sqrt{-\ln t}\bigr)$ para $0 < t < 1$, de
\omterm{def:b3:measure:measure}{medida} $v_d(-\ln t)^{d/2}$ (la
homotecia de razón $\rho$ escala $\lambda_d$ por $\rho^d$:
el~\cref{thm:b3:product:linearchange}), luego
\[
I = \int_0^1 v_d\,(-\ln t)^{d/2}\,\dd t
\overset{t = \eu^{-s}}{=}
v_d\int_0^\infty s^{d/2}\eu^{-s}\,\dd s
= v_d\,\Gamma\Bigl(\frac d2 + 1\Bigr).
\]
Iguálense. (Los valores: $\Gamma(\frac32) = \frac{\sqrt\pi}2$,
$\Gamma(2) = 1$, etc.) Obsérvese que $v_d \to 0$ cuando $d \to \infty$
--- el problema de fin de semana cuantifica a qué velocidad, mediante
Stirling.
\end{proof}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:product:1}
Sean $\mu$ la \omterm{def:b3:measure:measure}{medida} de conteo sobre
$(\intcc01, \mathcal
B(\intcc01))$ (que no es $\sigma$-finita) y $\lambda$ la
\omterm{def:b3:measure:lebesgueouter}{medida de Lebesgue}, y sea
$\Delta = \{(x,x)\}$ la diagonal de $\intcc01^2$. Demostrar que $\Delta$
es \omterm{def:b3:lebesgue:measurable}{medible} y calcular las dos
integrales iteradas de $\mathbf 1_\Delta$ contra $\lambda$ y $\mu$:
difieren. ¿Qué hipótesis del~\cref{thm:b3:product:tonelli} falla?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:product:2}
Justificar el intercambio y volver a obtener la integral de Dirichlet:
para
$A > 0$,
\[
\int_0^A\frac{\sin x}x\,\dd x
= \int_0^A\!\!\int_0^{+\infty}\eu^{-xy}\sin x\,\dd y\,\dd x
= \int_0^{+\infty}\!\!\int_0^A \eu^{-xy}\sin x\,\dd x\,\dd y,
\]
calcúlese la integral \omterm{def:b3:topology:interior}{interior} en forma cerrada y hágase $A \to
+\infty$ (domínese la integral en $y$) para obtener
$\int_0^\infty\frac{\sin x}x\dd x = \frac\pi2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:3}
(a) Demostrar que, para $f \geq 0$
\omterm{def:b3:lebesgue:measurable}{medible} y $\mu$ finita:
$\sum_{n\geq1}\mu(\{f \geq n\}) \leq \int f\,\dd\mu \leq
\mu(X) + \sum_{n\geq1}\mu(\{f\geq n\})$: la
\omterm{def:b3:lebesgue:l1}{integrabilidad} es la sumabilidad de las
\omterm{def:b3:measure:measure}{medidas} de las colas. (b) Deducir que
$f \in L^1(\mu)$ (con $\mu$ finita) si y solo si
$\sum_n\mu(\abs f \geq n) < \infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:4}
Para $f(x, y) = \dfrac{x^2 - y^2}{(x^2 + y^2)^2}$ sobre $\intoo01^2$,
demostrar
\[
\int_0^1\!\!\int_0^1 f\,\dd y\,\dd x = \frac\pi4,
\qquad
\int_0^1\!\!\int_0^1 f\,\dd x\,\dd y = -\frac\pi4
\]
\emph{(obsérvese que $f = \partial_y\bigl(\frac{y}{x^2+y^2}\bigr)$)}, y
comprobar directamente que $\int\!\int\abs f = +\infty$: la hipótesis de
\omterm{def:b3:lebesgue:l1}{integrabilidad} de Fubini no es decorativa.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:5}
(a) Calcular $\mathbf 1_{\intcc01} * \mathbf 1_{\intcc01}$
explícitamente (una función triangular) y la forma general de
$(\mathbf 1 * \mathbf 1 *
\mathbf 1)$. (b) Demostrar que $\operatorname{supp}(f * g) \subseteq
\overline{\operatorname{supp}f + \operatorname{supp}g}$. (c) Demostrar
que si $f \in L^1$ y $g$ está acotada y es
\omterm{def:b3:topology:continuity}{continua}, entonces $f * g$ es
\omterm{def:b3:topology:continuity}{continua}. \emph{(Convergencia
dominada mediante la \omterm{def:b3:topology:continuity}{continuidad} de
la traslación sobre la $g$ acotada.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:6}
(a) Demostrar que el símplex $\Delta_d = \{x \in \intco0\infty^d
: x_1 + \dots + x_d \leq 1\}$ tiene volumen $\frac1{d!}$
\emph{(inducción y Fubini)}. (b) Recuperar $v_2 = \pi$ y
$v_3 = \frac{4\pi}3$ del~\cref{thm:b3:product:ballvolume}, y demostrar
$\lambda_d(\text{elipsoide de semiejes } a_i) =
v_d\prod a_i$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:7}
¿Para qué $s > 0$ son finitas las siguientes integrales? Justifíquese
con \omterm{ex:b3:product:polar}{coordenadas polares}:
\[
\int_{B(0,1)\subseteq\R^2}\frac{\dd x\,\dd y}{(x^2 +
y^2)^{s}},
\qquad
\int_{\R^2\setminus B(0,1)}\frac{\dd x\,\dd y}{(x^2 +
y^2)^{s}} .
\]
Generalícese a $\R^d$ (los umbrales $s < d/2$ y $s > d/2$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:product:8}
(Beta--Gamma) Para $p, q > 0$, sea $B(p, q) = \int_0^1t^{p-1}(1
- t)^{q-1}\dd t$. Partiendo de $\Gamma(p)\Gamma(q)$ como integral doble,
sustitúyase $(x, y) = (uv,\, u(1 - v))$ (un difeomorfismo del cuadrante
abierto sobre $\intoo0\infty\times\intoo01$; calcúlese su jacobiano
$= u$) y concluir\index{función Beta}
\[
B(p, q) = \frac{\Gamma(p)\,\Gamma(q)}{\Gamma(p + q)} .
\]
Deducir $\int_0^{\pi/2}\sin^{2p-1}\theta\cos^{2q-1}\theta\,
\dd\theta = \frac12B(p,q)$ y el valor de las integrales de Wallis
$W_n = \int_0^{\pi/2}\sin^n$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:9}
(Fórmula de transferencia) Sea $T \colon (X, \mathcal A, \mu) \to (Y,
\mathcal B)$ \omterm{def:b3:lebesgue:measurable}{medible} y sea
$T_*\mu(B) = \mu(T^{-1}(B))$ la \emph{medida imagen}\index{medida
imagen}. Demostrar que, para toda $g \geq 0$
\omterm{def:b3:lebesgue:measurable}{medible} sobre $Y$:
\[
\int_Y g\,\dd(T_*\mu) = \int_X g\circ T\,\dd\mu
\]
(máquina estándar). Compárese después con
el~\cref{thm:b3:product:linearchange}: ¿qué información adicional lleva
la fórmula del cambio de variable que la fórmula abstracta de
transferencia no tiene? \emph{(La fórmula de transferencia nunca
identifica $T_*\mu$; el teorema del cambio de variable calcula
$T_*\lambda_d$ explícitamente como una
\omterm{def:b3:measure:measure}{medida} con
\omterm{ex:b3:lebesgue:gamma}{densidad}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:product:10}
(Momentos gaussianos) Usando \omterm{ex:b3:product:polar}{coordenadas
polares} y Fubini, calcúlense, para el peso gaussiano estándar sobre
$\R^d$:
\[
\int_{\R^d}\norm x_2^2\;\eu^{-\norm x_2^2}\,\dd x
\qquad\text{y}\qquad
\int_{\R^d}x_1^2\,\eu^{-\norm x^2_2}\,\dd x,
\]
compruébese la coherencia ($\norm x^2 = \sum x_i^2$) y dedúzcase el
segundo momento de la \omterm{def:b3:measure:measure}{medida}
$\pi^{-d/2}\eu^{-\norm x^2}\dd x$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:11}
(Gráfica e hipógrafo) Sea $f \colon \R^d \to \intco0\infty$
\omterm{def:b3:lebesgue:measurable}{medible}. (a) Demostrar que el
\emph{hipógrafo} $H = \{(x, y) \in
\R^d\times\R : 0 < y < f(x)\}$ es
\omterm{def:b3:lebesgue:measurable}{medible} en $\R^{d+1}$ con
\[
\lambda_{d+1}(H) = \int_{\R^d}f\,\dd\lambda_d :
\]
«la integral es el área bajo la gráfica», por fin un teorema.
\emph{(Secciones; Tonelli.)} (b) Demostrar que la gráfica
$\{(x, f(x)) : x \in \R^d\}$ es un conjunto nulo de $\R^{d+1}$. (c)
Deducir una demostración de dos líneas de que la esfera $S^{d-1}$ es
nula para Lebesgue en $\R^d$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:12}
(Una integral doble famosa) Usando la serie geométrica y Tonelli en
$\intoo01^2$, demostrar
\[
\int_0^1\!\!\int_0^1\frac{\dd x\,\dd y}{1 - xy}
= \sum_{n\geq1}\frac1{n^2} = \zeta(2),
\qquad
\int_0^1\!\!\int_0^1\frac{\dd x\,\dd y}{1 + xy}
= \sum_{n\geq1}\frac{(-1)^{n-1}}{n^2} = \frac{\zeta(2)}2 .
\]
(La segunda identidad de series: sepárense los índices pares e impares.)
Con $\zeta(2) = \frac{\pi^2}6$ (el~\cref{ch:b3:spectral}), dos
integrales de apariencia inocente valen $\frac{\pi^2}6$ y
$\frac{\pi^2}{12}$; ¿dónde hace exactamente su trabajo la hipótesis de
positividad de Tonelli?
\end{exercise}

\section{Problema: la fórmula de Stirling}

\begin{problem}[{Problema de fin de semana --- $n! \sim
\sqrt{2\pi n}\,(n/\eu)^n$, por convergencia dominada}]
\label{pb:b3:product:1}
La fórmula de Stirling gobierna todos los recuentos asintóticos de este
libro --- volúmenes de bolas, coeficientes binomiales, la forma local
del teorema central del límite. La demostramos a partir de la integral
$\Gamma$ (el~\cref{ex:b3:lebesgue:gamma}) con el \emph{método de
Laplace}, en su forma más limpia de convergencia dominada, y recogemos
después los dividendos.\index{fórmula de Stirling}

\medskip
\noindent\textbf{Parte I --- La fórmula.} Para $t > 0$,
$\Gamma(t + 1) = \int_0^\infty x^{t}\eu^{-x}\dd x$.
\begin{enumerate}
  \item Sustitúyase $x = t + \sqrt t\,u$ y demuéstrese
        \[
        \frac{\Gamma(t+1)}{t^{t}\eu^{-t}\sqrt t}
        = \int_{-\sqrt t}^{+\infty}
        \exp\Bigl(t\ln\Bigl(1 + \frac u{\sqrt
        t}\Bigr) - \sqrt t\,u\Bigr)\,\dd u
        \;=\;\int_\R g_t(u)\,\dd u,
        \]
        donde $g_t(u) = \exp\bigl(t\ln(1 + u/\sqrt t) -
        \sqrt t\,u\bigr)\mathbf 1_{u > -\sqrt t}$. \item Demostrar el
        límite puntual: para cada $u$ fijo, $g_t(u)
        \to \eu^{-u^2/2}$ cuando $t \to +\infty$ \emph{(desarróllese
        $\ln(1 + h)$ hasta el segundo orden)}. \item Dominación. Sea
        $\varphi(h) = \ln(1 + h) - h$, de modo que
        $g_t(u) = \exp\bigl(t\,\varphi(u/\sqrt t)\bigr)$ para
        $u > -\sqrt t$. Demostrar las dos cotas
        \[
        \varphi(h) \leq -\frac{h^2}4 \quad (-1 < h \leq 1),
        \qquad
        \varphi(h) \leq -c\,h \quad (h \geq 1),\ \ c = 1 -
        \ln 2 > 0
        \]
        \emph{(estúdiense $\varphi(h) + \frac{h^2}4$ y $\varphi(h)
        + ch$: calcúlense las derivadas y compruébese el signo en cada
        rango)}. Deducir, para $t \geq 1$:
        \[
        g_t(u) \leq \eu^{-u^2/4}\ \ (\abs u \leq \sqrt t),
        \qquad
        g_t(u) \leq \eu^{-cu}\ \ (u \geq \sqrt t),
        \]
        de modo que $g_t(u) \leq \eu^{-u^2/4} +
        \eu^{-cu}\,\mathbf 1_{u > 0}$: un dominador
        \omterm{def:b3:lebesgue:l1}{integrable} independiente de
        $t \geq 1$. \item Concluir con el teorema de la convergencia
        dominada y la integral gaussiana
        (el~\cref{ex:b3:product:polar}):
        \[
        \Gamma(t + 1) \;\sim\;
        \sqrt{2\pi t}\;\Bigl(\frac t\eu\Bigr)^{t}
        \qquad (t \to +\infty),
        \]
        y, en particular, $n! \sim \sqrt{2\pi
        n}\,(n/\eu)^n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Dividendos.}
\begin{enumerate}[resume]
  \item (Wallis) A partir del~\cref{exo:b3:product:8}, fórmulas del tipo
  $W_{2n} =
        \frac\pi2\binom{2n}n4^{-n}$: dedúzcase
  $\binom{2n}{n} \sim \frac{4^n}{\sqrt{\pi n}}$ de Stirling y
  contrástese con la recursión $W_{n} =
        \frac{n-1}nW_{n-2}$. \item (Los volúmenes de las bolas se
  desploman) Demostrar
        \[
        v_d = \frac{\pi^{d/2}}{\Gamma(\frac d2 + 1)}
        \;\sim\; \frac{1}{\sqrt{\pi d}}
        \Bigl(\frac{2\pi\eu}{d}\Bigr)^{d/2},
        \]
        de modo que $v_d \to 0$ más deprisa que cualquier sucesión
        geométrica; hállese la dimensión que maximiza $v_d$
        (numéricamente: $d =
        5$). \item (Concentración de la binomial --- anticipo
        del~\cref{ch:b3:clt}) Usando Stirling, demuéstrese la estimación
        local, para $k = n/2 + s\sqrt n/2$ con $s$ fijo y $n$ par:
        \[
        2^{-n}\binom{n}{k} \;\sim\;
        \sqrt{\frac{2}{\pi n}}\;\eu^{-s^2/2},
        \]
        el perfil gaussiano discreto: de Moivre--Laplace en embrión.
        \item ¿Dónde usó exactamente la demostración de la Parte I: (i)
        la convergencia monótona o la dominada; (ii) la integral
        gaussiana; (iii) las propiedades de invariancia de la
        \omterm{def:b3:measure:lebesgueouter}{medida de Lebesgue}? Una
        frase para cada una.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El término de error: Stirling con
barras.} Póngase $d_n = \ln n! - \bigl(n + \tfrac12\bigr)\ln n + n -
\ln\sqrt{2\pi}$, de modo que la Parte I dice $d_n \to 0$.
\begin{enumerate}[resume]
  \item Demostrar $d_n - d_{n+1} = \bigl(n +
        \tfrac12\bigr)\ln\bigl(1 + \tfrac1n\bigr) - 1$. \item Con
  $t = \frac1{2n+1}$, compruébese $\frac{n+1}n =
        \frac{1+t}{1-t}$ y desarróllese:
        \[
        d_n - d_{n+1} = \frac{t^2}3 + \frac{t^4}5 +
        \frac{t^6}7 + \cdots,
        \]
        y dedúzcanse las cotas por ambos lados
        \[
        \frac1{3(2n+1)^2} \;<\; d_n - d_{n+1} \;<\;
        \frac1{12n} - \frac1{12(n+1)} .
        \]
  \item Telescópese (usando $d_m \to 0$) y compruébese la agradable
  identidad \omterm{def:b3:galois:algebraic}{algebraica}
  $\frac1{3(2m+1)^2} >
        \frac1{12m+1} - \frac1{12(m+1)+1}$ para $m \geq 1$, para obtener
  el encaje clásico
        \[
        \sqrt{2\pi n}\Bigl(\frac n\eu\Bigr)^n
        \eu^{1/(12n+1)} \;<\; n! \;<\;
        \sqrt{2\pi n}\Bigl(\frac n\eu\Bigr)^n\eu^{1/(12n)} .
        \]
  \item Dos consecuencias: (a) el error relativo de la fórmula de
  Stirling es $< 10^{-6}$ en cuanto $n \geq 83\,334$; (b) estímese
  $100!$ con cuatro cifras significativas a mano a partir del encaje
  ($100! \approx 9.3326\cdot
        10^{157}$), y admírese uno un instante de la precisión de una
  fórmula asintótica en un $n$ muy finito.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La vía de Wallis: Stirling sin la
gaussiana.} Históricamente, la constante $\sqrt{2\pi}$ vino de Wallis,
no de Gauss; esta parte vuelve a demostrar Stirling con independencia de
las Partes I--II y, con ello, vuelve a demostrar la integral gaussiana.
Sea $W_n = \int_0^{\pi/2}\sin^n\theta\,
\dd\theta$.
\begin{enumerate}[resume]
  \item Establecer $W_n = \frac{n-1}nW_{n-2}$ (intégrese por partes),
  las formas cerradas
        \[
        W_{2n} = \frac\pi2\binom{2n}n4^{-n}, \qquad
        W_{2n+1} = \frac{4^n}{(2n+1)\binom{2n}n},
        \]
        y la identidad $W_nW_{n-1} = \frac\pi{2n}$. \item De la
        monotonía de $(W_n)$ dedúzcase $W_{2n}/W_{2n+1} \to 1$ y después
        \[
        W_{2n} \sim \frac12\sqrt{\frac\pi n}
        \qquad\text{y}\qquad
        \binom{2n}n4^{-n}\sqrt n \longrightarrow
        \frac1{\sqrt\pi} :
        \]
        el teorema de Wallis, obtenido sin Stirling. \item Demostrar,
        solo con el telescopado de la Parte III (sin necesidad del valor
        de la constante), que $e_n = \ln n! - (n +
        \frac12)\ln n + n$ converge a cierto límite $\ell$;
        equivalentemente, $n! \sim K\,n^{n+1/2}\eu^{-n}$ con $K =
        \eu^\ell > 0$ todavía sin identificar. \item Insértese esta
        asintótica en $\binom{2n}n4^{-n}\sqrt n$ e identifíquese, con la
        pregunta 14, el único valor posible: $K = \sqrt{2\pi}$. Móntese
        la lógica: las Partes III--IV juntas dan una segunda
        demostración \omterm{def:b3:complete:complete}{completa} de
        Stirling --- y, por tanto, recorriendo hacia atrás la
        sustitución de la Parte I, una evaluación independiente de
        $\int_\R\eu^{-u^2/2}\dd u = \sqrt{2\pi}$. Dos pilares, cada uno
        de los cuales sostiene al otro.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Últimos dividendos.}
\begin{enumerate}[resume]
  \item (El perfil local \omterm{def:b3:complete:complete}{completo}) Para enteros $\abs j \leq
        K\sqrt n$ (con $K$ fijo), demostrar
        \[
        \frac{\binom{2n}{n+j}}{\binom{2n}{n}} =
        \prod_{i=1}^{\abs j}\frac{n - i + 1}{n + i}
        = \exp\Bigl(-\frac{j^2}n +
        O\Bigl(\frac1{\sqrt n}\Bigr)\Bigr),
        \]
        uniformemente en $j$ \emph{(tómense logaritmos y úsese $\ln
        \frac{1-x}{1+y} = -(x + y) + O(x^2 + y^2)$)}. Esta es la versión
        por ambos lados de la pregunta 7 y la estimación exacta citada
        en el problema de fin de semana del~\cref{ch:b3:clt}. \item
        (Anticipo de Poisson) Demostrar con Stirling que
        $\eu^{-n}\dfrac{n^n}{n!} \sim \dfrac1{\sqrt{2\pi n}}$:
        la moda de una ley de Poisson de media grande $n$ tiene masa
        $\approx (2\pi n)^{-1/2}$, exactamente como predecirá el teorema
        central del límite. \item (Cocientes de Gamma) Para
        $a \in \intoo01$, demostrar
        $\dfrac{\Gamma(n + a)}{\Gamma(n)\,n^a} \to 1$ usando las cotas
        de pendientes por convexidad logarítmica
        del~\cref{pb:b3:lebesgue:1} (su pregunta 14), y extiéndase a
        todo $a > 0$ real mediante la ecuación funcional. (Esto es lo
        que significa «Stirling $\Gamma(t+1) \sim$» entre los enteros.)
        \item (Bolas, otra vez) A partir de
        $v_d = \pi^{d/2}/\Gamma(\frac d2
        + 1)$: tabúlense $v_1, \dots, v_7$ exactamente, compruébese la
        unimodalidad mediante $\frac{v_d}{v_{d-2}} = \frac{2\pi}d$
        (creciente mientras $d < 2\pi$, decreciente después) y
        demuéstrese la llamativa identidad generatriz
        \[
        \sum_{k\geq0}v_{2k}\,x^{2k} = \eu^{\pi x^2} :
        \]
        todos los volúmenes de bolas unidad de dimensión par
        empaquetados en una sola exponencial. \item (Asintótica de la
        entropía) Para $\alpha \in \intoo01$ fijo con $\alpha n \in \N$,
        dedúzcase de Stirling
        \[
        \binom{n}{\alpha n} \;\sim\;
        \frac{\eu^{n\,H(\alpha)}}
        {\sqrt{2\pi\,\alpha(1-\alpha)\,n}},
        \qquad
        H(\alpha) = -\alpha\ln\alpha -
        (1-\alpha)\ln(1-\alpha) :
        \]
        la tasa de crecimiento exponencial de los coeficientes
        binomiales es la \emph{entropía} $H$ --- compruébese que
        $\alpha =
        \frac12$ recupera la pregunta 5 y que $H(\alpha) <
        \ln2$ para $\alpha \neq \frac12$ (de modo que las binomiales
        descentradas son exponencialmente
        despreciables en $2^n$). \item (Áreas de superficies) El área de
        la esfera unidad $S^{d-1}$ es $s_{d-1} = d\,v_d$ (se demuestra
        como el~\cref{exo:b3:forms:6} en el capítulo de formas
        diferenciales; aquí tómese como definición). Tabúlense
        $s_0, \dots, s_6$, localícese el máximo ($d - 1 =
        6$, $s_6 = \frac{16\pi^3}{15} \approx 33.07$) y demuéstrese que
        $s_{d-1} \to 0$ también de manera supergeométrica --- las
        esferas de dimensión alta son, para cualquier vara de medir
        euclídea, evanescentemente pequeñas. \item (El primer término
        correctivo) Dedúzcase del encaje de la pregunta 11 que
        $d_n = \frac1{12n} +
        O\bigl(\frac1{n^2}\bigr)$, de donde
        \[
        n! = \sqrt{2\pi n}\,\Bigl(\frac
        n\eu\Bigr)^{n}\Bigl(1 + \frac1{12n} +
        O\Bigl(\frac1{n^2}\Bigr)\Bigr).
        \]
        Compruébese en $n = 10$: la fórmula desnuda da $3\,598\,696$
        (error relativo $8.3\cdot10^{-3}$) y la corregida $3\,628\,685$
        frente a $10! =
        3\,628\,800$ (error relativo $3.2\cdot10^{-5}$) --- un solo
        término de la serie compra dos cifras y media. \item (La mediana
        de $\Gamma$) Demostrar que
        \[
        \frac{1}{\Gamma(t+1)}
        \int_0^{t} x^{t}\eu^{-x}\,\dd x
        \;\longrightarrow\; \frac12
        \qquad (t \to +\infty) :
        \]
        asintóticamente, exactamente la mitad de la masa del integrando
        de $\Gamma$ está por debajo de su moda $x = t$. \emph{(Ejecútese
        la sustitución de la Parte I sobre la integral truncada; el
        dominador de la pregunta 3 ya está preparado.)} \item (Entropía,
        sin asintóticas) Para $\alpha \in
        \intoc0{\frac12}$, demostrar la cota, válida para \emph{todo}
        $n \geq 1$:
        \[
        \sum_{k=0}^{\lfloor\alpha n\rfloor}\binom nk
        \;\leq\; \eu^{n\,H(\alpha)} ,
        \]
        comparando la suma con $\sum_k\binom
        nk\lambda^{k-\alpha n}$ para el sesgo $\lambda =
        \frac{\alpha}{1-\alpha} \leq 1$. Compruébese que esta elección
        de $\lambda$ es óptima y reconcíliese con la pregunta 21: la
        tasa exponencial $H(\alpha)$ del enunciado asintótico se alcanza
        con una desigualdad de una línea, sin asintótica alguna.
\end{enumerate}
\end{problem}
