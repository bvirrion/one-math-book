\chapter{Espacios completos: Baire, Ascoli, Stone--Weierstrass}
\label{ch:b3:complete}

La \omterm{def:b3:complete:complete}{completitud} --- que toda sucesión
de Cauchy converja --- es la propiedad que permite al análisis
\emph{producir} objetos: puntos fijos de contracciones, sumas de series,
soluciones de ecuaciones obtenidas como límites. Este capítulo reúne las
tres grandes máquinas de existencia de la teoría métrica. El
\emph{teorema de Baire} muestra que un espacio
\omterm{def:b3:complete:complete}{completo} no puede ser unión numerable
de piezas despreciables, y conjura objetos
(¡funciones
\omterm{def:b3:topology:continuity}{continuas} \omterm{thm:b3:complete:nowherediff}{no derivables en ningún
punto}!) a partir de un razonamiento de puro sabor cardinal. El teorema
de \emph{Arzelà--Ascoli} identifica los subconjuntos
\omterm{def:b3:topology:compact}{compactos} de $\mathcal C(K)$ y es el
caballo de batalla de la \omterm{def:b3:topology:compact}{compacidad} en análisis --- el problema de fin
de semana lo usa para demostrar el teorema de existencia de Peano para
ecuaciones diferenciales. El teorema de \emph{Stone--Weierstrass}
muestra que los polinomios, y muchas otras familias, son densos en
$\mathcal C(K)$: la aproximación pasa a ser una verificación
\omterm{def:b3:galois:algebraic}{algebraica}. Por el \omterm{def:b3:topology:pathconnected}{camino} construimos
\omterm{thm:b3:complete:completion}{completados} y demostramos el
teorema de extensión para
aplicaciones uniformemente
\omterm{def:b3:topology:continuity}{continuas}, el pan de cada día del
\cref{ch:b3:lp,ch:b3:hilbert,ch:b3:fouriertransform}.

\section{Espacios completos, completados, extensiones}

\begin{definition}\label{def:b3:complete:complete}
Un espacio métrico es \emph{completo}\index{espacio métrico completo} si
toda sucesión de Cauchy converge (segundo año: $\R^n$ es completo;
$\mathcal C(\intcc01)$ con $\norm\cdot_\infty$ es completo). Un
subconjunto cerrado de un espacio completo es completo; un subconjunto
completo de un espacio métrico cualquiera es cerrado.
\end{definition}

\begin{proof}
Para ambos enunciados: una sucesión de Cauchy del cerrado $F$ converge
en $X$, y su límite, adherente a $F$, está en $F$; y una sucesión de un
\omterm{def:b3:complete:complete}{completo} $A$ convergente en $X$ es de
Cauchy, luego converge en $A$, y el límite es único.
\end{proof}

\begin{theorem}[Extensión de aplicaciones uniformemente continuas]
\label{thm:b3:complete:extension}
Sean $D \subseteq X$ denso, $Y$
\omterm{def:b3:complete:complete}{completo} y $f \colon D \to
Y$ uniformemente \omterm{def:b3:topology:continuity}{continua}. Entonces
$f$ se extiende de manera \emph{única} a una $\bar f \colon X \to Y$
\omterm{def:b3:topology:continuity}{continua}, y $\bar f$ es
uniformemente
\omterm{def:b3:topology:continuity}{continua}.\index{teorema de
extensión (continuidad uniforme)}
\end{theorem}

\begin{proof}
Unicidad: dos extensiones \omterm{def:b3:topology:continuity}{continuas}
coinciden en el denso $D$ y, por tanto, en todas partes (el conjunto de
coincidencia $\{g = h\}$ es cerrado: preimagen de la diagonal cerrada
por $x\mapsto(g(x), h(x))$). Existencia: para $x \in X$, tómese
$d_n \to x$, $d_n \in D$. La sucesión $(f(d_n))$ es de Cauchy: dado
$\varepsilon$, la \omterm{def:b3:topology:continuity}{continuidad}
uniforme proporciona $\delta$ con $d(u,v) < \delta \Rightarrow
d(f(u), f(v)) < \varepsilon$, y $(d_n)$ es de Cauchy. Defínase
$\bar f(x) = \lim f(d_n)$; el límite no depende de la sucesión elegida
(entrelácense dos de ellas). $\bar f$ extiende $f$ (sucesiones
constantes) y hereda el módulo de
\omterm{def:b3:topology:continuity}{continuidad}: si
$d(x, x') < \delta$, aproximando ambos por puntos de $D$ a distancia
$< \frac{\delta - d(x,x')}2$ se obtiene $d(\bar f(x), \bar
f(x')) \leq \varepsilon$ en el límite --- $\bar f$ es uniformemente
\omterm{def:b3:topology:continuity}{continua}.
\end{proof}

\begin{theorem}[Completado]\label{thm:b3:complete:completion}
Todo espacio métrico $X$ se sumerge isométricamente como subconjunto
denso de un \omterm{def:b3:complete:complete}{espacio métrico completo}
$\hat X$, único salvo isometría que fije $X$ punto a punto: su
\emph{completado}\index{completado de un espacio métrico}.
\end{theorem}

\begin{proof}
\emph{Existencia.} Sea $\mathcal X$ el conjunto de las sucesiones de
Cauchy de $X$, con la seudodistancia
\[
D\bigl((x_n), (y_n)\bigr) = \lim_n d(x_n, y_n),
\]
donde el límite existe porque $\abs{d(x_n, y_n) - d(x_m, y_m)}
\leq d(x_n, x_m) + d(y_n, y_m)$ hace de Cauchy la sucesión real. Sea
$\hat X = \mathcal X/{\sim}$, identificando las sucesiones a
$D$-distancia $0$; $D$ desciende a una distancia. Sumérjase $X$ mediante
las sucesiones constantes: es una isometría de imagen densa (una
sucesión de Cauchy queda $D$-aproximada por las constantes construidas
con sus propios términos: $D\bigl((x_n), (x_k)_{\rm
const}\bigr) = \lim_n d(x_n, x_k) \to 0$ cuando $k \to \infty$, por ser
de Cauchy). \omterm{def:b3:complete:complete}{Completitud} de $\hat X$:
sea $(\xi^k)$ de Cauchy en $\hat X$; por densidad, tómense $x_k \in X$
con $D(\xi^k, x_k)
\leq 2^{-k}$; entonces $(x_k)$ es de Cauchy en $X$ (desigualdad
triangular a través de los $\xi$), define un punto $\xi \in \hat
X$, y $D(\xi^k, \xi) \leq 2^{-k} + D(x_k, \xi) \to 0$ (la distancia de
la constante $x_k$ a la clase de $(x_j)_j$ es $\lim_j d(x_k, x_j)$,
pequeña para $k$ grande).

\emph{Unicidad}: dos \omterm{thm:b3:complete:completion}{completados}
$\hat X_1, \hat X_2$ contienen $X$ densamente; la identidad de $X$, que
es una isometría, es uniformemente
\omterm{def:b3:topology:continuity}{continua}, luego se extiende a
$\hat X_1 \to \hat X_2$ (el~\cref{thm:b3:complete:extension}), y sigue
siendo una isometría sobre un \omterm{def:b3:topology:interior}{conjunto
denso} y, por tanto, en todas partes; simétricamente en el otro sentido,
y las composiciones fijan el denso $X$: son las identidades.
\end{proof}

\begin{theorem}[Punto fijo de Banach]\label{thm:b3:complete:banach}
Sea $X$ \omterm{def:b3:complete:complete}{completo} y no vacío, y sea
$f \colon X \to X$ una \emph{contracción}:
$d(f(x), f(y)) \leq k\,d(x,y)$ con $k < 1$. Entonces $f$ tiene un único
punto fijo $x^*$, y toda \omterm{def:b3:groups:action}{órbita} converge
a él, con la velocidad explícita $d(x_n, x^*) \leq
\frac{k^n}{1-k}\,d(x_1, x_0)$.\index{teorema del punto fijo de Banach}
\end{theorem}

\begin{proof}
(Se demostró en segundo año; volvemos a registrar el argumento de dos
líneas por \omterm{def:b3:complete:complete}{completitud} de la exposición.) La
\omterm{def:b3:groups:action}{órbita} $x_{n+1} = f(x_n)$ cumple
$d(x_{n+1},
x_n) \leq k^nd(x_1, x_0)$, luego es de Cauchy (serie geométrica); su
límite $x^*$ es fijo (\omterm{def:b3:topology:continuity}{continuidad}
de $f$) y único, pues dos puntos fijos cumplirían $d \leq k\,d$. La
velocidad: súmese la cola geométrica.
\end{proof}

\begin{example}[Perturbar la identidad]
\label{ex:b3:complete:perturbident}
Sea $g \colon \R^d \to \R^d$ $k$-lipschitziana con $k < 1$. Entonces
$\varphi = \mathrm{id} + g$ es un
\emph{\omterm{def:b3:topology:continuity}{homeomorfismo} de $\R^d$ sobre
$\R^d$}. Inyectividad, con un módulo cuantitativo:
\[
\norm{\varphi(x) - \varphi(y)} \geq \norm{x - y} -
\norm{g(x) - g(y)} \geq (1 - k)\norm{x - y} .
\]
La sobreyectividad es el teorema del punto fijo: resolver
$\varphi(x) = y$ equivale a $x = y - g(x)$, y $x \mapsto y -
g(x)$ es una $k$-contracción del
\omterm{def:b3:complete:complete}{completo} $\R^d$ --- existe una única
solución $x = \psi(y)$ para todo $y$. La desigualdad mostrada hace la
inversa $\psi$ lipschitziana de constante $\frac1{1-k}$: es un
\omterm{def:b3:topology:continuity}{homeomorfismo}, con cotas explícitas
de ambos módulos. Este enunciado de apariencia inocua es el motor del
teorema de la función inversa (el~\cref{ch:b3:submanifolds}): cerca de
un punto donde $Df$ es invertible, $f$ \emph{es} una aplicación lineal
invertible más una pequeña perturbación lipschitziana, y el ejemplo de
hoy hace el resto. También cuantifica la robustez numérica: un sistema
perturbado por menos que el margen de la inversa sigue siendo
\omterm{def:b3:groups:derived}{resoluble}, y la solución se desplaza a
lo sumo $\frac{1}{1-k}$ veces la perturbación.
\end{example}

\section{El teorema de Baire}

\begin{theorem}[Baire]\label{thm:b3:complete:baire}
En un \omterm{def:b3:complete:complete}{espacio métrico completo}, toda
intersección numerable de abiertos
densos es densa. Equivalentemente: si $X = \bigcup_{n}F_n$ con cada
$F_n$ cerrado, entonces algún $F_n$ tiene
\omterm{def:b3:topology:interior}{interior} no vacío.\index{teorema de
Baire}
\end{theorem}

\begin{proof}
Sean $(U_n)$ abiertos densos y sea $B_0 = B(x_0, r_0)$ una bola abierta
cualquiera; encontraremos un punto de $\bigcap U_n$ en $B_0$.
Inductivamente: por ser $U_{n}$ denso y abierto, corta a la bola abierta
$B_{n-1}$ en un abierto, que contiene
una bola cerrada $\bar B(x_n, r_n)$ con $0 < r_n \leq r_{n-1}/2$ y
$\bar B(x_n, r_n) \subseteq
B_{n-1}\cap U_n$. Los \omterm{ex:b3:groups:actions}{centros} forman una
sucesión de Cauchy ($x_m \in
B_n$ para $m \geq n$, con radios $\to 0$); el límite $x$ está en todas
las $\bar B(x_n, r_n)$ (por ser cerradas), luego en todos los $U_n$ y en
$B_0$. Para la segunda formulación: si ningún $F_n$ tiene
\omterm{def:b3:topology:interior}{interior}, los $U_n = X \setminus F_n$
son abiertos y densos, y un punto de $\bigcap U_n$ escapa de
$\bigcup F_n = X$: absurdo.
\end{proof}

\begin{remark}\label{rem:b3:complete:meagre}
Vocabulario: un conjunto es \emph{denso en ninguna parte} si su
\omterm{def:b3:topology:interior}{clausura} tiene
\omterm{def:b3:topology:interior}{interior} vacío, y
\emph{magro}\index{conjunto magro} (de primera categoría) si es unión
numerable de \omterm{def:b3:topology:interior}{conjuntos densos} en
ninguna parte. Baire: \emph{un \omterm{def:b3:complete:complete}{espacio
métrico completo} no es \omterm{rem:b3:complete:meagre}{magro} en sí
mismo}, y el complementario de un
\omterm{rem:b3:complete:meagre}{conjunto magro} es denso.
«\omterm{rem:b3:complete:meagre}{Magro}» es una noción de pequeñez
ortogonal a la de medida (el~\cref{ch:b3:measure} producirá
\omterm{rem:b3:complete:meagre}{conjuntos magros} de medida total), y
los argumentos de Baire demuestran la \emph{existencia por abundancia}:
para exhibir un objeto sin la propiedad P, demuéstrese que los objetos
con P forman un \omterm{rem:b3:complete:meagre}{conjunto magro}.
\end{remark}

\begin{corollary}\label{cor:b3:complete:bairecors}
(a) $\R$ es no numerable. (b) $\Q$ no es intersección numerable de
abiertos de $\R$, y todo \omterm{def:b3:complete:complete}{espacio
métrico completo} no vacío sin puntos aislados es no numerable.
\end{corollary}

\begin{proof}
(a) $\R = \bigcup_{x}\{x\}$ sobre un conjunto numerable haría que algún
punto tuviera \omterm{def:b3:topology:interior}{interior}. (b) Si
$\Q = \bigcap_n V_n$ con $V_n$ abiertos (necesariamente densos, pues
$\supseteq \Q$), entonces los conjuntos $V_n$ y los complementarios
$\R\setminus\{q\}$, $q \in \Q$, forman una familia numerable de abiertos
densos de intersección vacía --- en contra de Baire. Y si $X$ es
\omterm{def:b3:complete:complete}{completo} sin puntos aislados y
numerable, $X = \bigcup_{x \in X}\{x\}$ lo exhibe como unión numerable
de cerrados de \omterm{def:b3:topology:interior}{interior} vacío (no hay
puntos aislados): Baire de nuevo.
\end{proof}

\begin{theorem}[Los monstruos de Weierstrass existen]
\label{thm:b3:complete:nowherediff}
Existen funciones \omterm{def:b3:topology:continuity}{continuas} sobre
$\intcc01$ que no son derivables en ningún punto. De hecho, el conjunto
de las $f \in \mathcal C(\intcc01)$ que tienen derivada (finita) aunque
sea en un solo punto es \omterm{rem:b3:complete:meagre}{magro} en
$(\mathcal C(\intcc01), \norm\cdot_\infty)$.\index{función no derivable
en ningún punto}
\end{theorem}

\begin{proof}
Para $n \geq 1$, póngase
\[
F_n = \Bigl\{f : \exists x \in \intcc01,\
\forall h \neq 0 \text{ con } x + h \in \intcc01,\
\abs{f(x+h) - f(x)} \leq n\abs h \Bigr\}.
\]
Si $f$ es derivable en $x$, entonces $f \in F_n$ para algún $n$: el
cociente $\abs{f(x+h)-f(x)}/\abs h$ está acotado para $\abs h
\leq \delta$ (derivabilidad: tiende a $\abs{f'(x)}$) y está acotado por
$2\norm f_\infty/\delta$ para $\abs h \geq \delta$. Así pues,
$\bigcup F_n$ contiene todas las funciones derivables en algún punto, y
basta demostrar que cada $F_n$ es cerrado con
\omterm{def:b3:topology:interior}{interior} vacío.

\emph{Cerrado}: sean $f_k \to f$ uniformemente, $f_k \in F_n$ con
testigos $x_k \to x$ (\omterm{def:b3:topology:compact}{compacidad}, tras una extracción). Para $h$ con
$x + h \in \intcc01$: elíjase $h_k \to h$ con $x_k + h_k
\in \intcc01$ (por ejemplo, $h_k = h + x - x_k$ truncado); entonces
$\abs{f(x + h) - f(x)} = \lim \abs{f_k(x_k + h_k) - f_k(x_k)}
\leq \lim n\abs{h_k} = n\abs h$, usando la convergencia uniforme y la
\omterm{def:b3:topology:continuity}{continuidad} de $f$ en los puntos
pertinentes: $f \in F_n$.

\emph{\omterm{def:b3:topology:interior}{Interior} vacío}: dados
$f \in F_n$ y $\varepsilon > 0$, buscamos $g$ con
$\norm{g - f}_\infty \leq \varepsilon$ y $g
\notin F_n$. Aproxímese primero $f$ hasta $\varepsilon/2$ por una
función afín a trozos $\varphi$
(\omterm{def:b3:topology:continuity}{continuidad} uniforme: interpólese
sobre una malla fina), de pendientes acotadas por cierto $M$. Añádase un
pequeño diente de sierra: $g = \varphi + \frac\varepsilon2\,s_N$, donde
$s_N(x)$ es el zigzag $\frac1N$-periódico de amplitud $1$ y pendiente
$\pm 2N$. En todo $x$, a un lado hay $h$ arbitrariamente pequeño para el
que el diente de sierra aporta la pendiente
$\pm 2N\cdot\frac\varepsilon2 =
\pm\varepsilon N$ sobre $[x, x+h]$: el cociente incremental de $g$
supera $\varepsilon N - M > n$ para $N$ grande. Luego $g \notin
F_n$, a cualquier distancia uniforme $\leq \varepsilon$ de $f$.
Conclusión: $\bigcup F_n$ es \omterm{rem:b3:complete:meagre}{magro}; por
Baire su complementario --- formado por
funciones \omterm{thm:b3:complete:nowherediff}{no derivables en ningún
punto} --- es denso en $\mathcal C(\intcc01)$: tales funciones existen
\emph{en abundancia}.
\end{proof}

\section{Arzelà--Ascoli}

En toda esta sección, $K$ es un espacio métrico
\emph{\omterm{def:b3:topology:compact}{compacto}} y $\mathcal
C(K) = \mathcal C(K, \R^d)$, con $\norm f_\infty = \sup_K
\norm{f(x)}$: un espacio \omterm{def:b3:complete:complete}{completo}
(los límites uniformes de \omterm{def:b3:topology:continuity}{continuas}
son \omterm{def:b3:topology:continuity}{continuos} --- segundo año).

\begin{definition}\label{def:b3:complete:equicontinuous}
Una familia $\mathcal F \subseteq \mathcal C(K)$ es
\emph{equicontinua}\index{equicontinuidad} si para todo
$\varepsilon > 0$ existe $\delta > 0$ tal que
\[
d(x, y) < \delta \implies \norm{f(x) - f(y)} < \varepsilon
\quad\text{para \emph{toda} } f \in \mathcal F
\]
(un mismo $\delta$ para toda la familia --- por ejemplo, cualquier
familia con una constante de Lipschitz común, o con un módulo de Hölder
común), y \emph{puntualmente acotada} si $\sup_{f}\norm{f(x)} < \infty$
para cada $x$.
\end{definition}

\begin{theorem}[Arzelà--Ascoli]\label{thm:b3:complete:ascoli}
Un subconjunto $\mathcal F \subseteq \mathcal C(K)$ es relativamente
\omterm{def:b3:topology:compact}{compacto} (tiene
\omterm{def:b3:topology:interior}{clausura}
compacta) si y solo si es
equicontinuo y puntualmente
acotado. En particular, toda \emph{sucesión}
\omterm{def:b3:complete:equicontinuous}{equicontinua} y puntualmente
acotada tiene una subsucesión uniformemente convergente.\index{teorema
de Arzelà--Ascoli}
\end{theorem}

\begin{proof}
($\Leftarrow$) Sea $(f_k)$ una sucesión de $\mathcal F$. Un espacio
métrico $K$ \omterm{def:b3:topology:compact}{compacto} es
\emph{separable}: para cada $n$, un número finito de bolas de radio
$\frac1n$ recubre $K$ (el~\cref{thm:b3:topology:metriccompact}); sus
\omterm{ex:b3:groups:actions}{centros} forman un
\omterm{def:b3:topology:interior}{conjunto denso} numerable
$D = \{x_1, x_2, \dots\}$. Por la acotación puntual y
Bolzano--Weierstrass, extráiganse sucesivamente subsucesiones que
converjan en $x_1$, después también en $x_2$, etc., y tómese la
subsucesión \emph{diagonal} $(g_j)$: converge en todo punto de $D$. La
\omterm{def:b3:complete:equicontinuous}{equicontinuidad} eleva esto a la
propiedad de Cauchy uniforme: dado $\varepsilon$, tómese $\delta$ como
en la definición, recúbrase $K$ por un número finito de bolas
$B(x_i, \delta)$ con $x_i \in D$ ($i \leq m$) y elíjase $J$ tan grande
que $\norm{g_j(x_i) - g_{j'}(x_i)} < \varepsilon$ para $j, j' \geq
J$, $i \leq m$. Para $x \in B(x_i, \delta)$ arbitrario:
\[
\norm{g_j(x) - g_{j'}(x)}
\leq \norm{g_j(x) - g_j(x_i)} + \norm{g_j(x_i) - g_{j'}(x_i)}
+ \norm{g_{j'}(x_i) - g_{j'}(x)} < 3\varepsilon .
\]
Así pues, $(g_j)$ es uniformemente de Cauchy y converge en el
\omterm{def:b3:complete:complete}{completo} $\mathcal C(K)$. Por tanto,
toda sucesión de $\mathcal F$ tiene una subsucesión convergente:
$\bar{\mathcal F}$ es \omterm{def:b3:topology:compact}{compacto}
(secuencialmente y, por el~\cref{thm:b3:topology:metriccompact}, sin
más).

($\Rightarrow$) Si $\bar{\mathcal F}$ es
\omterm{def:b3:topology:compact}{compacto}: la acotación puntual es
clara (la evaluación es \omterm{def:b3:topology:continuity}{continua}).
Para la \omterm{def:b3:complete:equicontinuous}{equicontinuidad},
recúbrase $\mathcal F$ por un número finito de bolas
$B(f_i, \varepsilon)$ de $\mathcal C(K)$; cada $f_i$ es uniformemente
\omterm{def:b3:topology:continuity}{continua} (Heine,
el~\cref{cor:b3:topology:heineborel}), lo que da un $\delta$ común para
$i \leq m$; y entonces, para $f \in B(f_i, \varepsilon)$ y $d(x,y)
< \delta$: $\norm{f(x)-f(y)} \leq 2\varepsilon +
\norm{f_i(x)-f_i(y)} < 3\varepsilon$.
\end{proof}

\begin{example}\label{ex:b3:complete:ascoliexamples}
La bola unidad cerrada de $\mathcal C(\intcc01)$ \emph{no} es
compacta ($f_n(x) = x^n$ no tiene
subsucesión uniformemente convergente: el límite puntual es discontinuo)
y, en efecto, $(x^n)$ no es
\omterm{def:b3:complete:equicontinuous}{equicontinua} en $1$. En cambio,
$\{f : \norm
f_\infty \leq 1,\ \operatorname{Lip}(f) \leq 1\}$ sí es
\omterm{def:b3:topology:compact}{compacto}: es acotado, $1$-equicontinuo
por Lipschitz, y cerrado. Ascoli explica \emph{por qué} falla la
\omterm{def:b3:topology:compact}{compacidad} en dimensión infinita (Riesz, segundo año) y qué hay que
añadir para restaurarla: un módulo uniforme de
\omterm{def:b3:topology:continuity}{continuidad}.
\end{example}

\section{Stone--Weierstrass}

\begin{lemma}[Dini]\label{lem:b3:complete:dini}
Sea $K$ \omterm{def:b3:topology:compact}{compacto} y sea $(f_n)$ una
sucesión \emph{monótona} de funciones reales
\omterm{def:b3:topology:continuity}{continuas} que converge
\emph{puntualmente} a una $f$
\omterm{def:b3:topology:continuity}{continua}. Entonces la convergencia
es uniforme.\index{teorema de Dini}
\end{lemma}

\begin{proof}
Digamos $f_n \uparrow f$; sea $g_n = f - f_n \downarrow 0$, que es
\omterm{def:b3:topology:continuity}{continua}. Dado $\varepsilon$, los
abiertos $U_n = \{g_n <
\varepsilon\}$ crecen y recubren $K$ (convergencia puntual); extráigase
un subrecubrimiento finito: $K = U_{n_0}$ para cierto $n_0$ (familia
creciente), es decir, $0 \leq g_n < \varepsilon$ en todas partes para
$n \geq n_0$.
\end{proof}

\begin{lemma}\label{lem:b3:complete:sqrt}
Existe una sucesión de \emph{polinomios} $u_n$ con $u_n(t) \to
\sqrt t$ uniformemente sobre $\intcc01$.
\end{lemma}

\begin{proof}
Defínanse $u_0 = 0$, $u_{n+1}(t) = u_n(t) + \frac12\bigl(t -
u_n(t)^2\bigr)$: son polinomios. Por inducción, $0 \leq u_n(t) \leq
\sqrt t$ sobre $\intcc01$: admitiéndolo para $n$,
\[
\sqrt t - u_{n+1}(t) = \bigl(\sqrt t - u_n(t)\bigr)
\Bigl(1 - \tfrac12\bigl(\sqrt t + u_n(t)\bigr)\Bigr) \geq 0,
\]
puesto que $\sqrt t + u_n \leq 2$; y $u_{n+1} \geq u_n \geq 0$. Así,
$(u_n(t))$ es no decreciente y está acotada por $\sqrt t$: converge
puntualmente, y el límite $\ell(t)$ cumple $\ell =
\ell + \frac12(t - \ell^2)$: $\ell(t) = \sqrt t$, que es
\omterm{def:b3:topology:continuity}{continua}. Dini
(el~\cref{lem:b3:complete:dini}) eleva la convergencia a uniforme.
\end{proof}

\begin{theorem}[Stone--Weierstrass, versión real]
\label{thm:b3:complete:stoneweierstrass}
Sea $K$ un \omterm{def:b3:topology:compact}{espacio compacto}
(\omterm{def:b3:topology:hausdorff}{de Hausdorff}) y sea
$\mathcal A \subseteq
\mathcal C(K, \R)$ una \emph{subálgebra} (estable por sumas, productos y
múltiplos escalares) que \emph{contiene las constantes} y \emph{separa
puntos} (para $x \neq y$, algún $f \in
\mathcal A$ cumple $f(x) \neq f(y)$). Entonces $\mathcal A$ es densa en
$(\mathcal C(K,\R), \norm\cdot_\infty)$.\index{teorema de
Stone--Weierstrass}
\end{theorem}

\begin{proof}
Sea $\bar{\mathcal A}$ la \omterm{def:b3:topology:interior}{clausura},
que sigue siendo un álgebra (los productos de límites uniformes sobre
acotados convergen).

\emph{Paso 1: $\bar{\mathcal A}$ es un retículo}, es decir, es estable
por $\max$ y $\min$. Como $\max(f,g) = \frac{f + g +
\abs{f-g}}2$ y $\min$ análogamente, basta con que $f \in
\bar{\mathcal A} \Rightarrow \abs f \in \bar{\mathcal A}$: con
$M = \norm f_\infty > 0$, $\abs f = M\sqrt{(f/M)^2}$, y
el~\cref{lem:b3:complete:sqrt} da polinomios $u_n$ con
$u_n\bigl((f/M)^2\bigr) \to \abs f/M$ uniformemente; y los polinomios en
elementos del álgebra (con término independiente: las constantes están)
permanecen en $\bar{\mathcal A}$.

\emph{Paso 2: interpolación en dos puntos.} Para $x \neq y$ y $a,
b \in \R$, algún $g \in \mathcal A$ cumple $g(x) = a$, $g(y) = b$:
tómese $h$ que separe $x, y$ y póngase $g = a + (b -
a)\frac{h - h(x)}{h(y) - h(x)}$.

\emph{Paso 3.} Sean $f \in \mathcal C(K)$, $\varepsilon > 0$. Para cada
par $x, y$ tómese $g_{x,y} \in \mathcal A$ con $g_{x,y}(x) = f(x)$,
$g_{x,y}(y) = f(y)$ (Paso 2; para $x = y$, tómese la función constante
$g_{x,x} = f(x)$). Fijemos $x$: para cada $y$, el
abierto $V_y = \{g_{x,y} < f +
\varepsilon\}$ contiene $y$; la \omterm{def:b3:topology:compact}{compacidad} extrae $y_1, \dots,
y_m$ con $K = \bigcup V_{y_j}$, y $h_x = \min_j g_{x, y_j}
\in \bar{\mathcal A}$ (Paso 1) cumple $h_x < f +
\varepsilon$ en todas partes y $h_x(x) = f(x)$. Hágase variar ahora $x$:
$W_x =
\{h_x > f - \varepsilon\}$ es abierto y contiene $x$; extráiganse $x_1,
\dots, x_l$ que recubran $K$, y $h = \max_i h_{x_i} \in
\bar{\mathcal A}$ cumple $f - \varepsilon < h < f +
\varepsilon$: $\norm{f - h}_\infty \leq \varepsilon$. Por tanto,
$f \in \bar{\mathcal A}$.
\end{proof}

\begin{corollary}\label{cor:b3:complete:weierstrass}
(a) (\emph{Weierstrass}) Los polinomios son densos en $\mathcal
C(\intcc ab, \R)$; los polinomios en $n$ variables son densos en
$\mathcal C(K, \R)$ para $K \subseteq \R^n$
\omterm{def:b3:topology:compact}{compacto}. (b) (\emph{Versión
compleja}) Si $\mathcal A \subseteq \mathcal
C(K, \C)$ es una subálgebra que contiene las constantes, separa puntos y
es \emph{estable por conjugación}, entonces es densa. (c) (\emph{Versión
trigonométrica}) Los polinomios trigonométricos
$\sum_{\abs n \leq N}c_n\eu^{\iu n t}$ son densos en el espacio de las
funciones \omterm{def:b3:topology:continuity}{continuas}
$2\pi$-periódicas, con $\norm\cdot_\infty$.\index{teorema de
aproximación de Weierstrass}
\end{corollary}

\begin{proof}
(a) Los polinomios forman un álgebra con constantes; las funciones
coordenadas separan los puntos de $\R^n$. (b) Las partes real e
imaginaria $\frac{f + \bar f}2$, $\frac{f - \bar f}{2\iu}$ de los
elementos de $\mathcal A$ forman un álgebra real
$\mathcal A_\R \subseteq \mathcal C(K, \R)$ con constantes, que separa
puntos ($f(x) \ne f(y)$ obliga a que $\operatorname{Re}f$ o
$\operatorname{Im}f$ separe). Aplíquese el teorema real y recombínese.
(c) Véanse las funciones $2\pi$-periódicas como $\mathcal C(S^1, \C)$
($S^1 = \R/2\pi\Z$, \omterm{def:b3:topology:compact}{compacto}:
el~\cref{exo:b3:topology:5}); el álgebra generada por
$\eu^{\iu t}, \eu^{-\iu t}$ y las constantes es estable por conjugación
y separa los puntos del círculo ($\eu^{\iu t}$ es inyectiva sobre él).
Aplíquese (b).
\end{proof}

\begin{remark}\label{rem:b3:complete:swremark}
La versión trigonométrica repara --- y generaliza enormemente --- la
laguna que dejó el capítulo de Fourier de segundo año: la densidad de
los polinomios trigonométricos en $(\mathcal C(S^1), \norm\cdot_2)$ se
sigue con mayor razón ($\norm\cdot_2 \leq \norm\cdot_\infty$ salvo la
constante de normalización), lo que hará del sistema de Fourier una
\emph{base} ortonormal en el~\cref{ch:b3:hilbert} y demostrará por fin
Parseval con toda generalidad.
\end{remark}

\section{Problema: el teorema de existencia de Peano}

\begin{problem}[{Problema de fin de semana --- existencia de soluciones
de
$x' = f(t, x)$ sin hipótesis de Lipschitz}]\label{pb:b3:complete:1}
Cauchy--Lipschitz (segundo año; redemostrado en el~\cref{ch:b3:ode})
exige que $f$ sea lipschitziana en $x$. Peano (1890): \emph{la
\omterm{def:b3:topology:continuity}{continuidad} de $f$ ya da la
existencia} --- aunque no la unicidad. Lo demostraremos con poligonales
de Euler y Ascoli. Marco: $f \colon R
\to \R^d$ \omterm{def:b3:topology:continuity}{continua} sobre el
rectángulo $R = \intcc{t_0 - a}{t_0
+ a}\times \bar B(x_0, b) \subseteq \R\times\R^d$, $M =
\sup_R\norm f$, y
\[
T = \min\bigl(a,\ b/M\bigr)
\qquad (\text{si } M = 0 \text{ el problema es trivial}).
\]

\medskip
\noindent\textbf{Parte I --- Poligonales de Euler.} Para $n \geq 1$,
subdivídase $[t_0, t_0 + T]$ mediante $t_k = t_0 + kT/n$ ($0 \leq k
\leq n$) y defínase $\varphi_n$ afín a trozos: $\varphi_n(t_0) = x_0$ y,
sobre $[t_k, t_{k+1}]$,
\[
\varphi_n(t) = \varphi_n(t_k) + (t -
t_k)\,f\bigl(t_k, \varphi_n(t_k)\bigr).
\]
\begin{enumerate}
  \item Demostrar por inducción que $\varphi_n$ está bien definida, con
  $\norm{\varphi_n(t) - x_0} \leq M(t - t_0) \leq b$ sobre
  $[t_0, t_0 + T]$ --- de modo que los puntos de evaluación permanecen
  en $R$. \emph{(Aquí es donde interviene $T \leq b/M$.)} \item
  Demostrar que cada $\varphi_n$ es $M$-lipschitziana. \item Deducir de
  Arzelà--Ascoli (el~\cref{thm:b3:complete:ascoli}) que alguna
  subsucesión $(\varphi_{n_j})$ converge uniformemente sobre
  $[t_0, t_0 +
        T]$ hacia cierta $\varphi$, ella misma $M$-lipschitziana con
        $\varphi(t_0) = x_0$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El límite resuelve la ecuación.} Defínase
el \emph{defecto} $\Delta_n(t) = \varphi_n'(t) -
f\bigl(t, \varphi_n(t)\bigr)$ en los puntos que no son de la malla.
\begin{enumerate}[resume]
  \item Demostrar que $f$ es uniformemente
  \omterm{def:b3:topology:continuity}{continua} sobre $R$ y deducir:
  para todo $\varepsilon > 0$ existe $n_0$ tal que, para $n \geq n_0$ y
  todo $t$ fuera de la malla, $\norm{\Delta_n(t)} \leq \varepsilon$.
  \emph{(Sobre $(t_k,
        t_{k+1})$, $\varphi_n'(t) = f(t_k, \varphi_n(t_k))$, y
  $(t, \varphi_n(t))$ dista menos de $(1 + M)\,T/n$ de
  $(t_k, \varphi_n(t_k))$.)} \item Establecer la forma integral: para
  todos $t$,
        \[
        \varphi_n(t) = x_0 + \int_{t_0}^{t}
        f\bigl(s, \varphi_n(s)\bigr)\dd s +
        \int_{t_0}^t \Delta_n(s)\,\dd s,
        \]
        donde el integrando central es
        \omterm{def:b3:topology:continuity}{continuo} a trozos. \item
        Pásese al límite a lo largo de $(n_j)$: demuéstrese
        $f(s, \varphi_{n_j}(s)) \to f(s, \varphi(s))$
        \emph{uniformemente} (de nuevo la
        \omterm{def:b3:topology:continuity}{continuidad} uniforme de
        $f$) y concluir
        \[
        \varphi(t) = x_0 + \int_{t_0}^t
        f\bigl(s, \varphi(s)\bigr)\dd s .
        \]
  \item Deducir que $\varphi$ es $\mathcal C^1$ sobre $[t_0, t_0 +
        T]$ y resuelve $x' = f(t, x)$, $x(t_0) = x_0$; extiéndase la
  construcción a $[t_0 - T, t_0]$ (inversión del tiempo). \emph{Este es
  el teorema de Peano.}
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La unicidad falla de verdad.}
Considérense $x' = 2\sqrt{\abs x}$, $x(0) = 0$, sobre $\R$.
\begin{enumerate}[resume]
  \item Comprobar que $f(x) = 2\sqrt{\abs x}$ es
  \omterm{def:b3:topology:continuity}{continua} pero no lipschitziana en
  ningún \omterm{def:b3:topology:topology}{entorno} de $0$. \item
  Comprobar que $x \equiv 0$ y que, para todo $c \geq 0$,
        \[
        x_c(t) = \begin{cases} 0 & t \leq c,\\ (t - c)^2 & t
        \geq c,\end{cases}
        \]
        son todas soluciones que pasan por $(0,0)$: un \omterm{def:b3:topology:continuity}{continuo} de
        soluciones distintas. \item ¿Dónde se rompe el argumento de la
        iteración de Picard (punto fijo de Banach) para esta $f$?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Límites del método.}
\begin{enumerate}[resume]
  \item Demostrar que el teorema de Peano falla en dimensión infinita:
  admitimos (o puede darse por bueno) el ejemplo clásico de Dieudonné en
  el espacio $c_0$ de las sucesiones que tienden a cero; en su lugar,
  demuéstrese el ingrediente de dimensión finita que allí falla: la bola
  unidad cerrada de $c_0$ (norma del supremo) no es
  compacta --- exhíbase una sucesión
  acotada sin subsucesión convergente y explíquese qué paso de la Parte
  I se rompe. \item Resúmase: ¿qué hipótesis dan existencia?, ¿cuáles
  existencia y unicidad? Enúnciense con precisión los dos teoremas
  (Peano; Cauchy--Lipschitz) uno junto al otro.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Osgood: unicidad más allá de Lipschitz.}
Sea $\omega\colon\intoo0\infty\to\intoo0\infty$
\omterm{def:b3:topology:continuity}{continua}, no decreciente, con
\[
\int_0^1\frac{\dd r}{\omega(r)} = +\infty
\qquad(\text{divergencia en } 0),
\]
y supóngase que $f$ cumple $\norm{f(t, x) - f(t, y)} \leq
\omega\bigl(\norm{x - y}\bigr)$ sobre $R$.
\begin{enumerate}[resume]
  \item Comprobar que $\omega(r) = Lr$ sirve (es lipschitziana), que
  $\omega(r) = r\log\frac1r$ (extendida por
  \omterm{def:b3:topology:continuity}{continuidad}, para $r$ pequeño)
  sirve aunque \emph{no} sea $O(r)$, y que $\omega(r) = 2\sqrt r$ no
  sirve. Calcúlese la integral en cada caso. \item Sean $x_1, x_2$
  soluciones de la ecuación sobre $[t_0, t_0 + T]$ con
  $x_1(t_0) = x_2(t_0)$, y sea $\delta(t) =
        \norm{x_1(t) - x_2(t)}$. Demostrar, solo a partir de las formas
  integrales, que para $t_0 \leq s \leq t$:
        \[
        \delta(t) \leq \delta(s) +
        \int_s^t\omega\bigl(\delta(v)\bigr)\dd v .
        \]
  \item (Teorema de Osgood) Supóngase $\delta(t_1) > 0$ para cierto
  $t_1$ y sea $\tau = \sup\{t \leq t_1 : \delta(t) =
        0\}$. Para $s \in \intoo\tau{t_1}$, póngase $u(t) = \delta(s)
        + \int_s^t\omega(\delta(v))\,\dd v$. Demostrar que $\delta \leq
        u$, $u' = \omega(\delta) \leq \omega(u)$, y deducir
        \[
        \int_{\delta(s)}^{u(t_1)}\frac{\dd r}{\omega(r)}
        \leq t_1 - s \leq t_1 - \tau .
        \]
        Hágase $s \downarrow \tau$ y obténgase una contradicción con la
        divergencia de la integral. Concluir: \emph{las soluciones con
        una misma condición inicial coinciden} --- unicidad bajo la
        condición de Osgood. \item Extráiganse las consecuencias: la
        unicidad de Cauchy--Lipschitz es el caso $\omega(r) = Lr$; la
        ecuación $x' =
        x\log\frac1{\abs x}$ (extendida por $0$ en $0$) tiene soluciones
        únicas aunque su miembro derecho no sea lipschitziano en $0$; y
        para $x' = 2\sqrt{\abs x}$, la convergencia de
        $\int_0\frac{\dd r}{2\sqrt r}$ es exactamente lo que permite a
        una solución abandonar $0$ en tiempo finito --- relaciónese el
        valor de la integral $\int_0^{h^2}\frac{\dd r}{2\sqrt r} = h$
        con el comportamiento de escape de $x_c$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Velocidades, esquemas, embudos.}
\begin{enumerate}[resume]
  \item (Lema integral de Grönwall) Sea $e, \eta \geq 0$
  \omterm{def:b3:topology:continuity}{continua} sobre $[t_0, t_0+T]$ con
  $e(t) \leq \int_{t_0}
        ^t\bigl(L\,e(s) + \eta(s)\bigr)\dd s$ para todo $t$. Demostrar
        \[
        e(t) \leq \frac{\sup\eta}{L}\bigl(\eu^{L(t - t_0)} -
        1\bigr)
        \]
        \emph{(póngase $G(t) = \int_{t_0}^t(Le + \eta)$, obsérvese que
        $G'
        \leq LG + \sup\eta$ y derívese
        $\eu^{-Lt}G(t)$)}.
  \item (Euler converge con una velocidad) Supóngase ahora que $f$ es
  $L$-lipschitziana en $x$ y $L'$-lipschitziana en $t$ sobre $R$.
  Combinando la cota del defecto de la pregunta 4 (hecha cuantitativa:
  $\norm{\Delta_n} \leq (L' + LM)T/n$) con la pregunta 17, demostrar
        \[
        \sup_{[t_0, t_0+T]}\norm{\varphi_n - \varphi}
        \;\leq\; \frac{(L' + LM)\,T}{n}\cdot
        \frac{\eu^{LT} - 1}{L} = O\Bigl(\frac1n\Bigr),
        \]
        donde $\varphi$ es \emph{la} solución: con datos lipschitzianos
        converge toda la sucesión, con una velocidad explícita --- sin
        necesidad de subsucesiones. ¿Por qué la unicidad eleva la
        convergencia por subsucesiones a convergencia plena incluso sin
        este cálculo? \item (El esquema elige) Para
        $x' = 2\sqrt{\abs x}$, $x(0) =
        0$: demostrar que toda poligonal de Euler es idénticamente nula,
        de modo que el esquema converge a la solución $x \equiv 0$;
        pero, arrancando en $x(0) = \varepsilon > 0$, converge (cuando
        $n \to \infty$ y después $\varepsilon \to 0$) a $t
        \mapsto t^2$, una solución \emph{distinta} que pasa por el
        origen. La no unicidad reaparece como sensibilidad del esquema
        numérico a las perturbaciones. \item Demostrar que el conjunto
        $\mathcal S$ de \emph{todas} las soluciones de $x' = f(t,x)$,
        $x(t_0) = x_0$, sobre $[t_0, t_0+T]$ (con valores en
        $\bar B(x_0, b)$) es no vacío (Parte II), uniformemente
        $M$-lipschitziano y cerrado en
        $\bigl(\mathcal C([t_0, t_0+T], \R^d),
        \norm\cdot_\infty\bigr)$; concluir con Ascoli que $\mathcal S$
        es \emph{\omterm{def:b3:topology:compact}{compacto}}. (El
        teorema de Kneser añade que $\mathcal S$ es
        \omterm{def:b3:topology:connected}{conexo}; no lo
        demostraremos.) \item Verificar el fenómeno de Kneser en el
        ejemplo: para $x' =
        2\sqrt{\abs x}$, $x(0) = 0$, sobre $[0, 1]$, demostrar que
        $\mathcal S = \{x_c : c \in \intcc0\infty\}$ con
        $x_\infty \equiv 0$ \emph{(para cualquier solución, póngase $c =
        \sup\{t : x(t) = 0\}$ e intégrese $(\sqrt x)' = 1$ sobre
        $\{x > 0\}$)}, que $c \mapsto x_c$ es
        \omterm{def:b3:topology:continuity}{continua} de $\intcc0\infty$
        (la \omterm{def:b3:topology:locallycompact}{compactificación por
        un punto}, es decir, con $c = \infty$ pegado como límite) en
        $\mathcal C([0,1])$, y concluir que $\mathcal S$ es, en efecto,
        \omterm{def:b3:topology:compact}{compacto} y
        \omterm{def:b3:topology:connected}{conexo} --- un embudo con
        forma de segmento. \item (Conjuntos alcanzables) Deducir de la
        pregunta 20 que, para cada $t$ fijo, el \emph{conjunto
        alcanzable} $\mathcal S(t) =
        \{x(t) : x \in \mathcal S\}$ es
        \omterm{def:b3:topology:compact}{compacto}; calcúlese para el
        ejemplo de la pregunta 21 y compruébese que también es
        \omterm{def:b3:topology:connected}{conexo}:
        $\mathcal S(t) = \intcc0{t^2}$ --- todo estado intermedio lo
        alcanza alguna solución.
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- Complementos: dependencia, optimalidad y
un esquema calculado a mano.}
\begin{enumerate}[resume]
  \item (Dependencia \omterm{def:b3:topology:continuity}{continua})
  Supóngase que $f$ es $L$-lipschitziana en $x$ sobre $R$ y sean $x, y$
  dos soluciones con valores iniciales $x_0, y_0$ en $t_0$. Adaptando la
  demostración de la pregunta 17 a la desigualdad $e(t) \leq \norm{x_0
        - y_0} + \int_{t_0}^tL\,e(s)\dd s$, demostrar
        \[
        \norm{x(t) - y(t)} \leq \norm{x_0 - y_0}\,
        \eu^{L(t - t_0)},
        \]
        y comprobar en $x' = Lx$ que la cota se alcanza: Grönwall es
        óptima. Deducir de nuevo la unicidad ($x_0 =
        y_0$) y que la aplicación de flujo $x_0 \mapsto x(t; x_0)$ es
        lipschitziana, de constante $\eu^{LT}$, allí donde está
        definida. \item (Osgood es óptima) Recíprocamente, sea $\omega$
        \omterm{def:b3:topology:continuity}{continua}, no decreciente,
        positiva en $\intoo0\infty$, con $\omega(r) \to 0$ cuando
        $r \to 0^+$, pero
        \[
        \int_0^1\frac{\dd r}{\omega(r)} < +\infty,
        \]
        y extiéndase $f(x) = \omega(\abs x)$ por $f(0) = 0$. Demostrar
        que $\Omega(x) = \int_0^x\frac{\dd r}{\omega(r)}$ es una
        biyección creciente de $\intoc0{1}$ sobre $\intoc0{\Omega(1)}$,
        que su inversa $g$ resuelve $g'
        = \omega(g)$ con $g(0^+) = 0$, y que $g$, extendida por $0$ para
        $t \leq 0$, es una solución $\mathcal C^1$ de $x' = f(x)$ que
        pasa por $(0, 0)$ y es distinta de $x \equiv
        0$ \emph{(para $g'(0) = 0$, acótese $\frac{g(t)}t$ por
        $\omega(g(t))$)}. Concluir: la hipótesis de divergencia de la
        pregunta 15 no es una comodidad, sino la frontera exacta de la
        unicidad; recupérese la Parte III a partir de
        $\omega(r) = 2\sqrt r$, $\Omega(x) = \sqrt x$.
  \item (Euler calculado a mano) Para $x' = x$, $x(0) = 1$ sobre
  $[0, 1]$: demostrar que la poligonal de Euler cumple
  $\varphi_n(1) = \bigl(1 + \frac1n\bigr)^n$. Demostrar el desarrollo
        \[
        \Bigl(1 + \frac1n\Bigr)^{n}
        = \eu\Bigl(1 - \frac1{2n} + O\Bigl(\frac1{n^2}\Bigr)
        \Bigr),
        \]
        de modo que el error en $t = 1$ es $\frac{\eu}{2n} +
        O(n^{-2})$: la velocidad $O(\frac1n)$ de la pregunta 18, con la
        constante exacta. Compruébese numéricamente para $n = 10$:
        $1.1^{10} = 2.59374$ frente a $\eu \approx 2.71828$, un error
        $0.12454$ que hay que comparar con $\frac{\eu}{20}
        \approx 0.13591$.
\end{enumerate}
\end{problem}
