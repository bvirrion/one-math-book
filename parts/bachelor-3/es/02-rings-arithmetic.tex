\chapter{Anillos y aritmética}\label{ch:b3:rings}

Los enteros ordinarios se factorizan de manera única en primos; los
polinomios sobre un cuerpo, también. ¿Son estos dos hechos un mismo
teorema? Este capítulo responde que sí y determina las hipótesis exactas
que hacen posible una «aritmética» en un anillo conmutativo: la cadena
\[
\text{euclídeo} \;\Longrightarrow\; \text{principal}
\;\Longrightarrow\; \text{factorial (DFU)},
\]
con todas las implicaciones demostradas y todos los recíprocos
refutados. La teoría se pone después a prueba allí donde demuestra su
valor: los enteros de Gauss $\Z[\iu]$ (que harán caer el teorema de los
dos cuadrados de Fermat en el problema de fin de semana), los anillos de
polinomios en varias variables (lema de Gauss, criterio de
\omterm{thm:b3:rings:criteria}{Eisenstein}) y los
\omterm{def:b3:rings:noetherian}{anillos noetherianos}, con el teorema
de la base de Hilbert como culminación. En todo el capítulo,
\emph{anillo} significa anillo conmutativo con unidad $1 \neq 0$; los
\omterm{def:b3:rings:ideal}{ideales} de $\Z$ y $K[X]$ vistos en segundo
año son nuestros dos ejemplos guía.

\section{Ideales, cocientes y el teorema de isomorfía}

\begin{definition}\label{def:b3:rings:ideal}
Un \emph{ideal}\index{ideal} $I$ de un anillo $A$ es un subgrupo aditivo
tal que $AI \subseteq I$. El \emph{anillo cociente}\index{anillo
cociente} $A/I$ es el \omterm{thm:b3:groups:quotient}{grupo cociente}
$(A, +)/I$ con la multiplicación $(a + I)(b + I) = ab + I$: está bien
definida, pues al cambiar $a$ por $a + x$ y $b$ por $b + y$
($x, y \in I$), $ab$ varía en $ay + xb + xy \in I$. La proyección
$\pi \colon A
\to A/I$ es un morfismo sobreyectivo de anillos con núcleo $I$, y los
núcleos de los morfismos de anillos son exactamente los ideales.
\end{definition}

\begin{theorem}[Primer teorema de
isomorfía]\label{thm:b3:rings:firstiso}
Si $f \colon A \to B$ es un morfismo de anillos, entonces $\bar f\colon
A/\ker f \to \operatorname{im} f$, $a + \ker f \mapsto f(a)$, es un
isomorfismo de anillos. Más en general, $f$ se factoriza a través de
$A/I$ para todo \omterm{def:b3:rings:ideal}{ideal} $I \subseteq \ker f$.
Los \omterm{def:b3:rings:ideal}{ideales} de $A/I$ son los $J/I$ con
$J \supseteq I$ \omterm{def:b3:rings:ideal}{ideal} de $A$ (teorema de
correspondencia).
\end{theorem}

\begin{proof}
Como para los grupos
(el~\cref{thm:b3:groups:firstiso,thm:b3:groups:correspondence}),
observando que todas las aplicaciones en juego respetan además los
productos: $\bar f$ está bien definida, es biyectiva sobre la imagen y
es multiplicativa; la correspondencia $J \mapsto J/I$,
$\bar J \mapsto \pi^{-1}(\bar J)$, conserva los
\omterm{def:b3:rings:ideal}{ideales} en ambos sentidos porque $\pi$ es
un morfismo sobreyectivo de anillos.
\end{proof}

\begin{definition}\label{def:b3:rings:primemaximal}
Sea $I \subsetneq A$ un \omterm{def:b3:rings:ideal}{ideal} propio. $I$
es \emph{primo}\index{ideal primo} si $ab \in I \Rightarrow a \in I$ o
$b \in I$; $I$ es \emph{maximal}\index{ideal maximal} si no hay ningún
\omterm{def:b3:rings:ideal}{ideal} estrictamente comprendido entre $I$ y
$A$.
\end{definition}

\begin{proposition}\label{prop:b3:rings:primemaximal}
$I$ es primo $\iff$ $A/I$ es un dominio de integridad; $I$ es
\omterm{def:b3:rings:primemaximal}{maximal} $\iff$ $A/I$ es un cuerpo.
En particular, los \omterm{def:b3:rings:primemaximal}{ideales maximales}
son primos.
\end{proposition}

\begin{proof}
Escribamos $\bar a$ para las clases en $A/I$. «$I$ primo» se traduce
literalmente por «$\bar a\bar b = 0 \Rightarrow \bar a = 0$ o $\bar b
= 0$», y $A/I \neq 0$ por $I \neq A$: esa es la definición de dominio.
Para la maximalidad, úsese el teorema de correspondencia: no hay
\omterm{def:b3:rings:ideal}{ideal} estrictamente comprendido entre $I$ y
$A$ $\iff$ $A/I$ no tiene más \omterm{def:b3:rings:ideal}{ideales} que
$0$ y él mismo $\iff$ $A/I$ es un cuerpo --- para este último paso: en
un cuerpo los únicos \omterm{def:b3:rings:ideal}{ideales} son $0$ y el
total (un \omterm{def:b3:rings:ideal}{ideal} que contenga $x \ne 0$
contiene $x^{-1}x = 1$); recíprocamente, si todo $x$ no nulo genera el
\omterm{def:b3:rings:ideal}{ideal} unidad, entonces $xy = 1$ para algún
$y$. Los cuerpos son dominios, luego los
\omterm{def:b3:rings:primemaximal}{ideales maximales} son primos.
\end{proof}

\begin{example}\label{ex:b3:rings:primemaximal}
En $\Z$: los \omterm{def:b3:rings:primemaximal}{ideales primos} son
$(0)$ y los $(p)$ con $p$ primo; los
\omterm{def:b3:rings:primemaximal}{maximales} son los $(p)$
($\Z/p\Z = \mathbb F_p$ es un cuerpo, $\Z/(0) = \Z$ no). En $K[X, Y]$:
$(X) \subsetneq (X, Y)$ son ambos primos ($K[X,Y]/(X) \cong K[Y]$, un
dominio; $K[X,Y]/(X,Y) \cong K$, un cuerpo), de modo que $(X)$ es primo
pero no \omterm{def:b3:rings:primemaximal}{maximal}.
\end{example}

Para garantizar que los \omterm{def:b3:rings:primemaximal}{ideales
maximales} \emph{existen} con total generalidad hace falta un principio
conjuntista. Un conjunto parcialmente ordenado es \emph{inductivo} si
todo subconjunto totalmente ordenado (\emph{cadena}) tiene una cota
superior.

\begin{theorem}[Lema de Zorn]\label{thm:b3:rings:zorn}
Todo conjunto parcialmente ordenado inductivo y no vacío tiene un
elemento \omterm{def:b3:rings:primemaximal}{maximal}.\index{lema de
Zorn}
\admitted
\end{theorem}

\begin{remark}\label{rem:b3:rings:zorn}
Esto no es un teorema de las matemáticas ordinarias, sino un
\emph{axioma}: sobre los axiomas básicos de Zermelo--Fraenkel de la
teoría de conjuntos equivale al axioma de elección («todo producto de
conjuntos no vacíos es no vacío»), que aceptamos en todo el libro.
Señalaremos cada uso. El análisis volverá a invocarlo (Hahn--Banach,
\cref{ch:b3:banach}).
\end{remark}

\begin{theorem}[Krull]\label{thm:b3:rings:krull}
Todo \omterm{def:b3:rings:ideal}{ideal} propio $I \subsetneq A$ está
\omterm{def:b3:rings:content}{contenido} en un \omterm{def:b3:rings:primemaximal}{ideal
maximal}.\index{teorema de Krull}
\end{theorem}

\begin{proof}
Ordénese por inclusión el conjunto $\mathcal E$ de los
\omterm{def:b3:rings:ideal}{ideales} propios que contienen $I$; es no
vacío ($I \in \mathcal E$). Una cadena $(J_\lambda)$ de $\mathcal E$
tiene por cota superior $J = \bigcup
J_\lambda$: es un \omterm{def:b3:rings:ideal}{ideal} (dos elementos
$a, b \in J$ cualesquiera están en un mismo $J_\lambda$ por ser una
cadena), propio ($1 \notin J_\lambda$ para todo $\lambda$) y contiene
$I$. El lema de Zorn da un elemento
\omterm{def:b3:rings:primemaximal}{maximal} de $\mathcal
E$, que es un \omterm{def:b3:rings:primemaximal}{ideal maximal} que
contiene $I$ (un \omterm{def:b3:rings:ideal}{ideal} propio estrictamente
mayor estaría en $\mathcal E$).
\end{proof}

\begin{theorem}[Teorema chino del resto]\label{thm:b3:rings:crt}
Sean $I_1, \dots, I_n$ \omterm{def:b3:rings:ideal}{ideales} de $A$
\emph{comaximales}\index{ideales comaximales} dos a dos ($I_k + I_l = A$
para $k \neq l$). Entonces
\[
A\Big/\bigcap_{k=1}^n I_k \;\xrightarrow{\;\sim\;}\;
\prod_{k=1}^n A/I_k,
\qquad
a \longmapsto (a + I_1, \dots, a + I_n),
\]
y además $\bigcap_k I_k = I_1 I_2 \cdots I_n$ (el
\omterm{def:b3:rings:ideal}{ideal} generado por los
productos).\index{teorema chino del resto}
\end{theorem}

\begin{proof}
La aplicación $f(a) = (a + I_k)_k$ es un morfismo de anillos con núcleo
$\bigcap I_k$; por el~\cref{thm:b3:rings:firstiso} basta demostrar la
sobreyectividad. Fijemos $k$; para cada $l \neq k$ escribamos $1 = u_l +
v_l$ con $u_l \in I_k$, $v_l \in I_l$ (comaximalidad). Entonces
\[
e_k = \prod_{l \neq k} v_l = \prod_{l\neq k}(1 - u_l)
\equiv 1 \pmod{I_k},
\qquad e_k \in I_l \ (l \neq k),
\]
luego $f(e_k) = (0, \dots, 1, \dots, 0)$; dado un objetivo $(a_k +
I_k)_k$, el elemento $\sum_k a_k e_k$ se aplica en él.

Productos frente a intersección: $I_1\cdots I_n \subseteq \bigcap I_k$
siempre. Recíprocamente, por inducción basta tratar el caso $n = 2$ (se
comprueba que $I_1$ y $I_2\cdots I_n$ son
\omterm{thm:b3:rings:crt}{comaximales}: multiplicando $1 = u_l + v_l$
sobre $l \geq 2$ se obtiene $1 \in I_1 + I_2\cdots
I_n$). Para $n = 2$: escríbase $1 = u + v$, $u \in I_1$, $v \in I_2$;
para $x \in I_1 \cap I_2$, $x = xu + xv \in I_2I_1 + I_1I_2 =
I_1I_2$.
\end{proof}

\begin{example}\label{ex:b3:rings:crtz}
En $\Z$ con $I_k = (m_k)$, $m_k$ coprimos dos a dos: $\Z/(m_1\cdots
m_n)\Z \cong \prod \Z/m_k\Z$ --- el teorema chino del resto de segundo
año. Restringiendo a las unidades: $(\Z/mn\Z)^\times \cong
(\Z/m\Z)^\times \times (\Z/n\Z)^\times$ para $\gcd(m,n)=1$, de donde la
multiplicatividad de la $\varphi$ de Euler
(\cref{exo:b3:rings:8}).
\end{example}

\section{Divisibilidad: euclídeo, principal, factorial}

\begin{definition}\label{def:b3:rings:divisibility}
Sea $A$ un dominio de integridad y $a, b \in A$. Decimos que $a$
\emph{divide} a $b$ ($a \mid b$) si $b \in (a) = aA$. Dos elementos $a,
b$ son \emph{asociados}\index{elementos asociados} si $a = ub$ con
$u \in A^\times$ (equivalentemente, $(a) = (b)$). Un elemento $p$ no
nulo y no unidad es:
\begin{itemize}
  \item \emph{irreducible}\index{elemento irreducible} si $p = ab$
  obliga a $a \in A^\times$ o $b \in A^\times$; \item
  \emph{primo}\index{elemento primo} si $p \mid ab$ obliga a $p
        \mid a$ o $p \mid b$ (es decir, si el
  \omterm{def:b3:rings:ideal}{ideal} $(p)$ es primo).
\end{itemize}
\end{definition}

\begin{proposition}\label{prop:b3:rings:primeirred}
En todo dominio, primo $\Rightarrow$
\omterm{def:b3:rings:divisibility}{irreducible}. El recíproco es falso
en general: en $\Z[\iu\sqrt 5] = \{a + \iu b\sqrt5 : a, b
\in \Z\}$, el elemento $2$ es
\omterm{def:b3:rings:divisibility}{irreducible} pero no primo.
\end{proposition}

\begin{proof}
Sea $p$ primo y $p = ab$. Entonces $p \mid ab$, digamos $p \mid a$:
$a = pc$, luego $p = pcb$ y, cancelando $p$ (¡es un dominio!), $cb =
1$: $b \in A^\times$.

En $\Z[\iu\sqrt5]$ se usa la norma $N(x + \iu y\sqrt 5) = x^2 +
5y^2$, que es multiplicativa (vale $\abs z^2$). Si $2 = ab$ con $a, b$
no unidades, entonces $4 = N(a)N(b)$ con $N(a), N(b) \neq
1$ (los elementos de norma $1$ son $\pm1$, las unidades), luego
$N(a) = 2$: imposible, $x^2 + 5y^2 = 2$ no tiene soluciones enteras. Por
tanto $2$ es \omterm{def:b3:rings:divisibility}{irreducible}. Pero
$2 \mid 6 = (1 + \iu\sqrt5)(1 - \iu\sqrt5)$ mientras que $2$ no divide a
ninguno de los dos factores ($\frac12 \pm \frac{\iu\sqrt5}2
\notin \Z[\iu\sqrt5]$): no es primo.
\end{proof}

\begin{definition}\label{def:b3:rings:pidufd}
Un dominio de integridad $A$ es:
\begin{itemize}
  \item \emph{euclídeo}\index{dominio euclídeo} si existe una aplicación
  $\nu \colon A \setminus \{0\} \to \N$ (una \emph{función euclídea})
  tal que, para todos $a, b$ con $b \ne 0$, existen $q, r$ con
  $a = bq + r$ y ($r = 0$ o $\nu(r) <
        \nu(b)$); \item \emph{principal} (un \emph{DIP}\index{dominio de
  ideales principales}) si todo \omterm{def:b3:rings:ideal}{ideal} es de
  la forma $(a)$; \item \emph{factorial} (un \emph{DFU}\index{dominio de
  factorización única}) si todo elemento no nulo y no unidad es un
  producto de \omterm{def:b3:rings:divisibility}{irreducibles}, único
  salvo el orden y salvo \omterm{def:b3:rings:divisibility}{asociados}.
\end{itemize}
\end{definition}

\begin{theorem}\label{thm:b3:rings:euclideanpid}
\omterm{def:b3:rings:pidufd}{Euclídeo} $\Rightarrow$ principal.
\end{theorem}

\begin{proof}
Sea $I \neq (0)$ un \omterm{def:b3:rings:ideal}{ideal} y sea
$b \in I \setminus\{0\}$ con $\nu(b)$ mínimo. Para $a \in I$, dividamos:
$a = bq + r$; entonces $r = a
- bq \in I$, y $\nu(r) < \nu(b)$ contradiría la minimalidad, luego
$r = 0$ y $a \in (b)$: $I = (b)$.
\end{proof}

\begin{example}\label{ex:b3:rings:euclidean}
$\Z$ (con $\nu = \abs\cdot$) y $K[X]$ (con $\nu = \deg$) son
\omterm{def:b3:rings:pidufd}{euclídeos} --- ambas divisiones se
establecieron en segundo año. También lo es $\Z[\iu]$, con $\nu = N$ la
norma cuadrática (\cref{exo:b3:rings:4}); la geometría de la
demostración está en la figura siguiente. Existe algún
\omterm{def:b3:rings:pidufd}{DIP} que no es
\omterm{def:b3:rings:pidufd}{euclídeo}, pero certificarlo es delicado
(el ejemplo estándar es $\Z\bigl[\frac{1+\iu\sqrt{19}}2\bigr]$); en
cambio, un \omterm{def:b3:rings:pidufd}{DFU} que no es
\omterm{def:b3:rings:pidufd}{DIP} es fácil: $K[X, Y]$
(\cref{exo:b3:rings:6}), o $\Z[X]$.
\end{example}

\begin{lemma}[Cadenas ascendentes de ideales principales]
\label{lem:b3:rings:acc}
En un \omterm{def:b3:rings:pidufd}{DIP}, toda sucesión creciente de
\omterm{def:b3:rings:ideal}{ideales} $I_1 \subseteq I_2
\subseteq \cdots$ es estacionaria.
\end{lemma}

\begin{proof}
$I = \bigcup_n I_n$ es un \omterm{def:b3:rings:ideal}{ideal} (la unión
es creciente), luego $I =
(a)$; el elemento $a$ está en algún $I_N$, y entonces $I = (a)
\subseteq I_N \subseteq I_n \subseteq I$ para $n \geq N$.
\end{proof}

\begin{lemma}[Bézout; lema de Euclides]\label{lem:b3:rings:bezout}
Sea $A$ un \omterm{def:b3:rings:pidufd}{DIP} y sean $a, b \in A$.
Entonces $(a) + (b) = (d)$ para cierto $d$, que es un \emph{máximo común
divisor}\index{máximo común divisor}: $d \mid a$, $d \mid b$, y todo
divisor común de $a,
b$ divide a $d$; además $d = au + bv$ para ciertos $u, v$ (Bézout). En
consecuencia, todo \omterm{def:b3:rings:divisibility}{elemento
irreducible} de un \omterm{def:b3:rings:pidufd}{DIP} es primo.
\end{lemma}

\begin{proof}
$(a) + (b)$ es un \omterm{def:b3:rings:ideal}{ideal}, luego $(d)$;
$a, b \in (d)$ da $d \mid
a, b$; y $d = au + bv \in (a) + (b)$. Un divisor común $c$ de $a, b$
divide a $au + bv = d$.

Euclides: sea $p$ \omterm{def:b3:rings:divisibility}{irreducible},
$p \mid ab$, $p \nmid a$. Un máximo común divisor $d$ de $p$ y $a$
divide a $p$, luego $d$ es una unidad o un asociado de $p$
(irreducibilidad); el caso asociado queda excluido por $p \nmid a$. Así
$1 = pu + av$, de donde $b = pub + abv$, y $p$ divide a ambos términos:
$p \mid b$.
\end{proof}

\begin{theorem}\label{thm:b3:rings:pidufd}
Principal $\Rightarrow$ factorial.
\end{theorem}

\begin{proof}
\emph{Existencia.} Supongamos que algún elemento $a$, no nulo y no
unidad, no admite factorización en
\omterm{def:b3:rings:divisibility}{irreducibles}. Entonces $a$ no es
\omterm{def:b3:rings:divisibility}{irreducible}: $a =
a_1b_1$ con ambos factores no unidades; al menos uno de ellos, digamos
$a_1$, tampoco admite factorización (un producto de dos elementos
factorizables es factorizable). Iterando, obtenemos $a = a_0, a_1, a_2,
\dots$, cada uno divisor propio del anterior y sin factorización, luego
$(a_0) \subsetneq (a_1) \subsetneq (a_2) \subsetneq \cdots$ --- las
inclusiones son estrictas porque $a_n = a_{n+1}c$ con $c$ no unidad
significa que $(a_n) = (a_{n+1})$ obligaría a $c \in A^\times$
(cancelando en un dominio). Esto contradice el~\cref{lem:b3:rings:acc}.

\emph{Unicidad.} Sea $p_1 \cdots p_r = q_1 \cdots q_s$ con todos los
factores \omterm{def:b3:rings:divisibility}{irreducibles} y $r \leq s$,
por inducción sobre $r$. El primo (el~\cref{lem:b3:rings:bezout}) $p_1$
divide al miembro de la derecha, luego divide a algún $q_j$;
renumerando, $j = 1$. Como $q_1$ es
\omterm{def:b3:rings:divisibility}{irreducible} y $p_1$ no es unidad,
$q_1 = u p_1$ con $u \in A^\times$: $p_1,
q_1$ son \omterm{def:b3:rings:divisibility}{asociados}. Cancelando
$p_1$: $p_2 \cdots p_r = (u q_2)
q_3\cdots q_s$, y se concluye por inducción ($r = 1$ obliga a $s = 1$:
una unidad por \omterm{def:b3:rings:divisibility}{irreducibles} no puede
ser $1$).
\end{proof}

\begin{remark}\label{rem:b3:rings:ufdgcd}
En un \omterm{def:b3:rings:pidufd}{DFU} existen los máximos comunes
divisores (tómense los exponentes mínimos en las factorizaciones) y vale
el lema de Euclides --- \omterm{def:b3:rings:divisibility}{irreducible}
$=$ primo (\cref{exo:b3:rings:2}) ---, pero Bézout puede fallar: en
$\Z[X]$, $\gcd(2, X) = 1$ y sin embargo $1 \neq 2U + XV$ (evalúese en
$X = 0$: $1 = 2U(0)$, imposible). Las identidades de Bézout son
patrimonio exclusivo de los \omterm{def:b3:rings:pidufd}{DIP}.
\end{remark}

\begin{example}[Un anillo sin factorización única]
\label{ex:b3:rings:nonufd}
Ninguna de las implicaciones \omterm{def:b3:rings:pidufd}{euclídeo}
$\Rightarrow$ \omterm{def:b3:rings:pidufd}{DIP} $\Rightarrow$
\omterm{def:b3:rings:pidufd}{DFU} es una equivalencia, y merece la pena
ver una vez, con todo detalle, cómo falla la última. En
\[
A = \Z[\iu\sqrt5] = \{a + \iu b\sqrt5 : a, b \in \Z\},
\qquad N(a + \iu b\sqrt5) = a^2 + 5b^2,
\]
la norma es multiplicativa y $N(z) = 1$ si y solo si $z \in
A^\times = \{\pm1\}$. Consideremos
\[
6 = 2 \cdot 3 = (1 + \iu\sqrt5)(1 - \iu\sqrt5).
\]
Los cuatro factores son
\omterm{def:b3:rings:divisibility}{irreducibles}: sus normas valen
$4, 9, 6,
6$, y una factorización propia $z = z_1z_2$ obligaría a
$N(z_1) \in \{2, 3\}$ --- pero $a^2 + 5b^2$ nunca es igual a $2$ ni a
$3$ ($b = 0$ deja los no cuadrados $2, 3$; $\abs b \geq
1$ da $\geq 5$). Sin embargo, $2$ no es asociado de ninguno de los dos
$1 \pm
\iu\sqrt5$ (normas $4 \neq 6$): dos factorizaciones genuinamente
distintas de $6$ en \omterm{def:b3:rings:divisibility}{irreducibles}.
Equivalentemente, aquí \omterm{def:b3:rings:divisibility}{irreducible}
$\neq$ primo: $2$ divide al producto $(1 +
\iu\sqrt5)(1 - \iu\sqrt5) = 6$ pero a ninguno de los dos factores (de
nuevo por las normas). La reparación del fallo en términos de \omterm{def:b3:rings:ideal}{ideales}
--- factorizar \emph{\omterm{def:b3:rings:ideal}{ideales}} en lugar de
elementos --- es el nacimiento de la teoría algebraica de números; a
nuestro nivel, el ejemplo calibra hasta qué punto son especiales los
anillos \omterm{def:b3:rings:pidufd}{euclídeos} $\Z$, $K[X]$, $\Z[\iu]$
de este capítulo.
\end{example}

\begin{method}\label{met:b3:rings:quotient}
Para identificar un \omterm{def:b3:rings:ideal}{anillo cociente} $A/I$,
búsquese un morfismo sobreyectivo $f \colon A \to B$ de núcleo $I$ e
invóquese el~\cref{thm:b3:rings:firstiso}; cuando $A = C[X]$ es un
anillo de polinomios, $f$ suele ser una evaluación. Así,
$\Z[X]/(X^2+1) \cong
\Z[\iu]$ (evaluar en $\iu$), $K[X,Y]/(Y - X^2) \cong K[X]$ (evaluar $Y$
en $X^2$), $\R[X]/(X^2+1) \cong \C$. Para ver que $I$ es primo o
\omterm{def:b3:rings:primemaximal}{maximal}, compruébese que el cociente
es un dominio o un cuerpo
(\cref{prop:b3:rings:primemaximal}).
\end{method}

\section{Polinomios sobre un DFU: Gauss y Eisenstein}

En toda esta sección, $A$ es un \omterm{def:b3:rings:pidufd}{DFU} con
cuerpo de fracciones $K$ (construido como el cuerpo de los cocientes
formales $a/b$, $b \neq 0$, exactamente igual que $\Q$ a partir de $\Z$;
esta construcción se hizo en segundo año para $\Q$ y se traslada
literalmente). Nuestro objetivo: la factorialidad pasa de $A$ a $A[X]$,
y la irreducibilidad sobre $A$ es esencialmente la irreducibilidad sobre
el cuerpo mayor $K$.

\begin{definition}\label{def:b3:rings:content}
El \emph{contenido}\index{contenido de un polinomio} $c(P)$ de un
$P \in A[X]$ no nulo es un máximo común divisor de sus coeficientes
(definido salvo unidades); $P$ es \emph{primitivo} si
$c(P) \in A^\times$. Todo $P
\in A[X]$ se escribe $P = c(P)\,P_1$ con $P_1$ primitivo, y todo
$P \in K[X]\setminus\{0\}$ se escribe $P = \lambda P_1$ con $\lambda
\in K^\times$ y $P_1 \in A[X]$ primitivo (quítense denominadores y
sáquese después el contenido).
\end{definition}

\begin{lemma}[Gauss]\label{lem:b3:rings:gauss}
El producto de dos polinomios primitivos de $A[X]$ es primitivo; en
consecuencia, $c(PQ) = c(P)c(Q)$ salvo unidades.\index{lema de Gauss}
\end{lemma}

\begin{proof}
Sean $P, Q$ primitivos y supongamos que algún
\omterm{def:b3:rings:divisibility}{irreducible} (= primo, por ser
\omterm{def:b3:rings:pidufd}{DFU}) $p$ divide todos los coeficientes de
$PQ$. Reduzcamos módulo $p$: en $(A/(p))[X]$, $\bar P \bar Q = 0$. Pero
$A/(p)$ es un dominio ($(p)$ primo), luego $(A/(p))[X]$ es un dominio
(los coeficientes directores se multiplican), lo que obliga a
$\bar P = 0$ o $\bar Q = 0$: $p$ divide todos los coeficientes de $P$ o
todos los de $Q$, en contra de la primitividad. Para la consecuencia,
escríbase $P = c(P)P_1$, $Q = c(Q)Q_1$: $PQ =
c(P)c(Q) P_1Q_1$ con $P_1Q_1$ primitivo.
\end{proof}

\begin{theorem}\label{thm:b3:rings:gaussufd}
Sea $A$ un \omterm{def:b3:rings:pidufd}{DFU} con cuerpo de fracciones
$K$.
\begin{enumerate}
  \item Un $P \in A[X]$ primitivo de grado $\geq 1$ es
  \omterm{def:b3:rings:divisibility}{irreducible} en $A[X]$ si y solo si
  es \omterm{def:b3:rings:divisibility}{irreducible} en $K[X]$. \item
  $A[X]$ es un \omterm{def:b3:rings:pidufd}{DFU}; sus
  \omterm{def:b3:rings:divisibility}{irreducibles} son los
  \omterm{def:b3:rings:divisibility}{irreducibles} de $A$ y los
  polinomios primitivos \omterm{def:b3:rings:divisibility}{irreducibles}
  sobre $K$. En particular $\Z[X]$ y, por inducción,
  $K[X_1, \dots, X_n]$ y $\Z[X_1, \dots, X_n]$, son
  \omterm{def:b3:rings:pidufd}{DFU}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) ($\Leftarrow$) Si $P = QR$ en $A[X]$ con $Q, R$ no unidades, ninguno
de los dos factores es constante (un factor constante de un polinomio
primitivo es una unidad), luego la factorización es propia en $K[X]$.
($\Rightarrow$) Supongamos $P = QR$ con $Q, R \in K[X]$ de grados
$\geq 1$. Escribamos $Q = \lambda Q_1$, $R = \mu R_1$ con $Q_1, R_1
\in A[X]$ primitivos: $P = \lambda\mu\, Q_1R_1$, y $Q_1R_1$ es primitivo
por Gauss. Tomando \omterm{def:b3:rings:content}{contenidos},
$\lambda\mu \in A^\times$ (ambos miembros tienen
\omterm{def:b3:rings:content}{contenido} unidad; formalmente,
$\lambda\mu = c(P) \in
A^\times$ salvo unidades y, en particular, $\lambda \mu \in A$): $P
= (\lambda\mu Q_1) R_1$ es una factorización propia en $A[X]$.

(2) Existencia: dado $P \neq 0$ no unidad, factorícese $P = c(P)P_1$,
factorícese $c(P)$ en \omterm{def:b3:rings:divisibility}{irreducibles}
de $A$ y factorícese $P_1$ en el \omterm{def:b3:rings:pidufd}{DFU}
$K[X]$ como $\prod Q_i$ con $Q_i \in K[X]$
\omterm{def:b3:rings:divisibility}{irreducibles}; escribiendo
$Q_i = \lambda_i R_i$ con $R_i \in A[X]$ primitivos (luego
\omterm{def:b3:rings:divisibility}{irreducibles} sobre $K$ y, por (1),
en $A[X]$), el producto $\prod \lambda_i$ es una unidad de $A$ como
antes, y $P_1 = u\prod
R_i$. Unicidad: compárense la parte constante y la parte polinómica de
una factorización; las constantes multiplican a $c(P)$ (Gauss), única
por la factorialidad de $A$; las partes polinómicas dan dos
factorizaciones en $K[X]$ del mismo polinomio, luego coinciden salvo
constantes de $K^\times$ (factorialidad de $K[X]$,
el~\cref{thm:b3:rings:pidufd}), y dos polinomios primitivos \omterm{def:b3:rings:divisibility}{asociados} en
$K[X]$ lo están en $A[X]$: si $R = \lambda R'$ con $R, R'$ primitivos y
$\lambda \in K^\times$, tomar \omterm{def:b3:rings:content}{contenidos}
obliga a $\lambda \in A^\times$.
\end{proof}

\begin{theorem}[Criterios de
irreducibilidad]\label{thm:b3:rings:criteria}
Sea $A$ un \omterm{def:b3:rings:pidufd}{DFU}, sea $K$ su cuerpo de
fracciones y sea $P = a_nX^n + \dots
+ a_0 \in A[X]$ primitivo de grado $n \geq 1$.
\begin{enumerate}
  \item (\emph{Reducción}) Si $p \in A$ es primo, $p \nmid a_n$ y la
  reducción $\bar P$ es \omterm{def:b3:rings:divisibility}{irreducible}
  en $(A/(p))[X]$, entonces $P$ es
  \omterm{def:b3:rings:divisibility}{irreducible} en $K[X]$ (y por tanto
  en
        $A[X]$).
  \item (\emph{Eisenstein}\index{criterio de Eisenstein}) Si algún primo
  $p$ cumple $p \nmid a_n$, $p \mid a_i$ para $0
        \leq i < n$ y $p^2 \nmid a_0$, entonces $P$ es
  \omterm{def:b3:rings:divisibility}{irreducible} en $K[X]$ (y por tanto
  en $A[X]$).
\end{enumerate}
\end{theorem}

\begin{proof}
Por el~\cref{thm:b3:rings:gaussufd}(1), una factorización propia sobre
$K$ da $P = QR$ con $Q, R \in A[X]$, $\deg Q, \deg R \geq 1$ (las
constantes quedan excluidas: serían unidades o estropearían la
primitividad).

(1) Redúzcase módulo $p$: $\bar P = \bar Q\bar R$ en $(A/(p))[X]$. Como
$p \nmid a_n$ y $\deg$ solo pueden bajar al reducir, $\deg \bar
Q = \deg Q \geq 1$ y $\deg\bar R = \deg R \geq 1$ (sus coeficientes
directores multiplican a $\bar a_n \neq 0$, luego ninguno baja):
$\bar P$ se factoriza propiamente --- contradicción.

(2) Redúzcase módulo $p$: $\bar Q \bar R = \bar P = \bar a_n X^n$ (todos
los coeficientes inferiores se anulan). En el dominio $(A/(p))[X]$, las
factorizaciones de $cX^n$ ($c \ne 0$) son en constantes y potencias
puras $c'X^k$: en efecto, si $\bar Q\bar R = \bar a_nX^n$ y, digamos,
$\bar Q$ tuviera un coeficiente no nulo en grado $< \deg\bar Q$, tomando
los términos no nulos de menor grado se obtiene
$\operatorname{val}(\bar Q\bar R) =
\operatorname{val}\bar Q + \operatorname{val}\bar R$ (dominio), que ha
de ser igual a $n = \deg\bar Q + \deg\bar R$, lo que obliga a
$\operatorname{val} = \deg$ en ambos: los dos son monomios. Como antes,
los grados no bajan, luego $Q$ y $R$ tienen sus términos constantes
$Q(0), R(0)$ divisibles por $p$ --- los dos, pues ambas reducciones son
monomios de grado $\geq 1$. Entonces $p^2 \mid
Q(0)R(0) = a_0$: contradicción.
\end{proof}

\begin{example}\label{ex:b3:rings:eisenstein}
$X^n - p$ es \omterm{def:b3:rings:divisibility}{irreducible} sobre $\Q$
para todo primo $p$ y todo $n
\geq 1$ (\omterm{thm:b3:rings:criteria}{Eisenstein} en $p$): hay
polinomios \omterm{def:b3:rings:divisibility}{irreducibles} de todos los
grados sobre $\Q$ --- en marcado contraste con $\C$ (grado $1$,
d'Alembert--Gauss, demostrado en el~\cref{ch:b3:holomorphic}) y con $\R$
(grados $1, 2$). El truco de \emph{trasladar} amplía el alcance de
\omterm{thm:b3:rings:criteria}{Eisenstein}: el \emph{polinomio
ciclotómico}\index{polinomio ciclotómico} $\Phi_p = X^{p-1} +
\dots + X + 1 = \frac{X^p - 1}{X - 1}$ de índice $p$ cumple
\[
\Phi_p(X + 1) = \frac{(X+1)^p - 1}{X}
= X^{p-1} + \binom{p}{1}X^{p-2} + \dots + \binom{p}{p-1},
\]
\omterm{thm:b3:rings:criteria}{Eisenstein} en $p$ ($p \mid \binom pk$
para $0 < k < p$, y $\binom{p}{p-1} = p \not\equiv 0 \bmod p^2$):
$\Phi_p(X+1)$, y por tanto $\Phi_p$, es
\omterm{def:b3:rings:divisibility}{irreducible} sobre $\Q$. Este es el
corazón algebraico de la historia del $17$-ágono que se cuenta en
el~\cref{ch:b3:galois}.
\end{example}

\begin{method}\label{met:b3:rings:irreducibility}
Para demostrar que $P \in \Z[X]$ es
\omterm{def:b3:rings:divisibility}{irreducible} sobre $\Q$: (i) hágase
$P$ primitivo; (ii) pruébese \omterm{thm:b3:rings:criteria}{Eisenstein},
sobre $P(X)$ y sobre las traslaciones $P(X \pm
1)$; (iii) pruébese a reducir módulo primos pequeños que no dividan al
coeficiente director --- basta la irreducibilidad módulo \emph{un} $p$,
y sobre $\mathbb F_p$ la irreducibilidad es una comprobación finita (no
tener raíces excluye los factores de grado $1$; después se ensayan los
finitos factores de cada grado $\leq \deg P/2$); (iv) si todo lo demás
falla, coeficientes indeterminados. Cuidado: que sea reducible módulo
todo $p$ \emph{no} implica que lo sea sobre $\Q$
(\cref{exo:b3:rings:11}).
\end{method}

\section{Anillos noetherianos}

\begin{definition}\label{def:b3:rings:noetherian}
Un anillo $A$ es \emph{noetheriano}\index{anillo noetheriano} si todo
\omterm{def:b3:rings:ideal}{ideal} de $A$ es finitamente generado.
\end{definition}

\begin{proposition}\label{prop:b3:rings:noethacc}
$A$ es \omterm{def:b3:rings:noetherian}{noetheriano} si y solo si toda
sucesión creciente de \omterm{def:b3:rings:ideal}{ideales} es
estacionaria (\emph{condición de cadena ascendente}), si y solo si toda
familia no vacía de \omterm{def:b3:rings:ideal}{ideales} tiene un
elemento \omterm{def:b3:rings:primemaximal}{maximal} (para la
inclusión).
\end{proposition}

\begin{proof}
\emph{(FG $\Rightarrow$ CCA)}: para una cadena $I_1 \subseteq I_2
\subseteq \cdots$, la unión $I$ es un
\omterm{def:b3:rings:ideal}{ideal}, generado por $x_1,
\dots, x_r$; todos los $x_i$ están en algún $I_N$, luego $I = I_N = I_n$
para $n \geq N$. \emph{(CCA $\Rightarrow$ elementos
\omterm{def:b3:rings:primemaximal}{maximales})}: si una familia no vacía
$\mathcal F$ no tuviera elemento
\omterm{def:b3:rings:primemaximal}{maximal}, tómese $I_1 \in
\mathcal F$ y después, inductivamente, $I_{n+1} \supsetneq I_n$ en
$\mathcal F$ (es posible porque $I_n$ no es
\omterm{def:b3:rings:primemaximal}{maximal}): una cadena infinita
estrictamente creciente. (Esto usa el axioma de elecciones dependientes,
una forma débil de elección que no discutiremos.) \emph{(Elementos
\omterm{def:b3:rings:primemaximal}{maximales} $\Rightarrow$ FG)}: dado
un \omterm{def:b3:rings:ideal}{ideal} $I$, la familia de los
\omterm{def:b3:rings:ideal}{ideales} finitamente generados \omterm{def:b3:rings:content}{contenidos} en
$I$ es no vacía ($(0)$); un elemento
\omterm{def:b3:rings:primemaximal}{maximal} $J = (x_1, \dots, x_r)$ ha
de ser igual a $I$: en caso contrario, añadiendo $x \in I \setminus J$ a
los generadores se obtendría un miembro estrictamente mayor de la
familia.
\end{proof}

\begin{theorem}[Teorema de la base de
Hilbert]\label{thm:b3:rings:hilbert}
Si $A$ es \omterm{def:b3:rings:noetherian}{noetheriano}, también lo es
$A[X]$. Por tanto, también lo son $A[X_1, \dots,
X_n]$ y todos sus cocientes.\index{teorema de la base de Hilbert}
\end{theorem}

\begin{proof}
Sea $I$ un \omterm{def:b3:rings:ideal}{ideal} de $A[X]$ y supongamos que
$I$ no es finitamente generado. Construyamos una sucesión:
$f_1 \in I \setminus \{0\}$ de grado mínimo y, inductivamente,
$f_{k+1} \in I \setminus (f_1,
\dots, f_k)$ de grado mínimo (el conjunto es no vacío por hipótesis).
Los grados $d_k = \deg f_k$ son no decrecientes (por la minimalidad de
cada elección: $f_{k+1}$ estaba disponible en el paso $k+1$; con más
precisión, $f_{k+1} \notin (f_1,\dots,f_k) \supseteq
(f_1, \dots, f_{k-1})$, luego $f_{k+1}$ competía en el paso $k$ y perdió
o empató: $d_{k+1} \geq d_k$). Sea $a_k \in A$ el coeficiente director
de $f_k$. La cadena de \omterm{def:b3:rings:ideal}{ideales}
$(a_1) \subseteq (a_1,
a_2) \subseteq \cdots$ se estabiliza: $a_{n+1} \in (a_1, \dots,
a_n)$ para algún $n$, digamos $a_{n+1} = \sum_{k\leq n} u_k a_k$.
Consideremos
\[
g = f_{n+1} - \sum_{k=1}^{n} u_k X^{\,d_{n+1} - d_k} f_k .
\]
Entonces $g \in I \setminus (f_1, \dots, f_n)$ (la suma está en el
\omterm{def:b3:rings:ideal}{ideal}, $f_{n+1}$ no), y sin embargo el
coeficiente de grado $d_{n+1}$ se cancela: $\deg g < d_{n+1}$, en contra
de la minimalidad de $d_{n+1} = \deg f_{n+1}$.

Iterando, $A[X_1, \dots, X_n] = (A[X_1, \dots, X_{n-1}])[X_n]$ es
\omterm{def:b3:rings:noetherian}{noetheriano}; un cociente $A/I$ es
\omterm{def:b3:rings:noetherian}{noetheriano} porque sus
\omterm{def:b3:rings:ideal}{ideales} $J/I$ se levantan a
\omterm{def:b3:rings:ideal}{ideales} de $A$ (correspondencia), donde un
número finito de generadores se proyecta sobre generadores.
\end{proof}

\begin{remark}\label{rem:b3:rings:noethmeaning}
La noetherianidad es el axioma de finitud de la geometría algebraica:
todo sistema de ecuaciones polinómicas en $n$ variables, por infinito
que sea, equivale a un número finito de ellas --- su conjunto de
soluciones queda definido por finitos polinomios. Los
\omterm{def:b3:rings:pidufd}{DIP} son
\omterm{def:b3:rings:noetherian}{noetherianos} (trivialmente);
$\Z[X_1, X_2, \dots]$ en infinitas variables no lo es
($(X_1) \subsetneq (X_1, X_2) \subsetneq \cdots$). También aparecen
anillos no \omterm{def:b3:rings:noetherian}{noetherianos} de manera
natural en análisis: las funciones continuas sobre $\intcc01$ forman uno
(\cref{exo:b3:rings:10}).
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:rings:1}
Identificar los cocientes: (a) $\Z[X]/(X^2 + 1) \cong \Z[\iu]$;
(b) $\R[X]/(X^2+1) \cong \C$; (c) $\mathbb F_2[X]/(X^2 + X + 1)$
es un cuerpo de $4$ elementos --- escríbase su tabla de multiplicar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:rings:2}
(a) Demostrar que en un \omterm{def:b3:rings:pidufd}{DFU} todo
\omterm{def:b3:rings:divisibility}{elemento irreducible} es primo. (b)
Demostrar que un dominio de integridad finito es un cuerpo. (c) Deducir
que en un anillo finito todo \omterm{def:b3:rings:primemaximal}{ideal
primo} es \omterm{def:b3:rings:primemaximal}{maximal}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:rings:3}
En $\Z[\iu\sqrt5]$: comprobar que $3$, $1 + \iu\sqrt5$ y $1 -
\iu\sqrt5$ son \omterm{def:b3:rings:divisibility}{irreducibles}, que
$9 = 3\cdot 3 = (2 +
\iu\sqrt5)(2 - \iu\sqrt5)$, y concluir de nuevo (tras
la~\cref{prop:b3:rings:primeirred}) que $\Z[\iu\sqrt5]$ no es un
\omterm{def:b3:rings:pidufd}{DFU}. ¿Dónde falla exactamente la unicidad?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:4}
(a) Demostrar que $\Z[\iu]$ es \omterm{def:b3:rings:pidufd}{euclídeo}
para la norma $N(x + \iu y) =
x^2 + y^2$: dados $a, b \neq 0$, elíjase $q \in \Z[\iu]$ lo más próximo
posible a $a/b \in \C$. (b) Determinar $\Z[\iu]^\times$. (c) Las mismas
preguntas para $\Z[\iu\sqrt2]$ y $N(x + \iu y\sqrt2) =
x^2 + 2y^2$. ¿Por qué falla el mismo argumento para
$\Z[\iu\sqrt5]$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:5}
Sea $A$ un anillo. (a) Demostrar que si $x$ es nilpotente ($x^n = 0$
para algún $n$), entonces $1 + x \in A^\times$. (b) Demostrar que si $A$
es un dominio, entonces $A[X]^\times = A^\times$; dar un contraejemplo
sobre $\Z/4\Z$. (c) Demostrar que un dominio no tiene más idempotentes
($e^2 = e$) que $0, 1$, ni más nilpotentes que $0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:6}
En $A = K[X, Y]$: (a) demostrar que el
\omterm{def:b3:rings:ideal}{ideal} $(X, Y)$ es
\omterm{def:b3:rings:primemaximal}{maximal} pero no principal --- de
modo que $K[X,Y]$ es un \omterm{def:b3:rings:pidufd}{DFU}
(el~\cref{thm:b3:rings:gaussufd}) que no es un
\omterm{def:b3:rings:pidufd}{DIP}; (b) identificar $K[X, Y]/(Y - X^2)$ y
$K[X,Y]/(XY - 1)$ como subanillos de funciones racionales; (c) ¿es
$(Y - X^2)$ primo?, ¿es \omterm{def:b3:rings:primemaximal}{maximal}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:7}
\omterm{def:b3:rings:divisibility}{Irreducible} o no sobre $\Q$:
$X^5 - 12X^3 + 36X - 12$;\quad $X^4 + X + 1$ \emph{(redúzcase módulo
$2$)};\quad $X^4 + 4$;\quad $\Phi_8 = X^4 + 1$ \emph{(trasládese
$1$)};\quad $X^3 - X - 1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:8}
(a) A partir del~\cref{thm:b3:rings:crt}, demostrar que la función de
Euler es multiplicativa sobre argumentos coprimos y que $\varphi(p^k) =
p^{k-1}(p-1)$; recuperar $\varphi(n) = n\prod_{p \mid n}(1 -
\frac1p)$. (b) Resolver: $x \equiv 2 \pmod 7$, $x \equiv 5 \pmod{11}$,
$x
\equiv 1 \pmod{13}$, exhibiendo los idempotentes $e_k$ de la
demostración del~\cref{thm:b3:rings:crt}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:rings:9}
(El nilradical) Sea $\operatorname{Nil}(A)$ el conjunto de los elementos
nilpotentes. (a) Demostrar que $\operatorname{Nil}(A)$ es un
\omterm{def:b3:rings:ideal}{ideal} \omterm{def:b3:rings:content}{contenido} en todo
\omterm{def:b3:rings:primemaximal}{ideal primo}. (b) Recíprocamente, sea
$a$ no nilpotente; usando el lema de Zorn sobre los
\omterm{def:b3:rings:ideal}{ideales} que evitan $S =
\{a^n : n \in \N\}$, constrúyase un
\omterm{def:b3:rings:primemaximal}{ideal primo} que no contenga $a$.
Concluir:
\[
\operatorname{Nil}(A) = \bigcap_{\mathfrak p \text{ primo}}
\mathfrak p .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:rings:10}
(a) Sea $A$ \omterm{def:b3:rings:noetherian}{noetheriano} y sea
$f \colon A \to A$ un morfismo sobreyectivo de anillos. Demostrar que
$f$ es inyectivo. \emph{(Considérese $\ker
f \subseteq \ker f^2 \subseteq \cdots$.)} (b) Demostrar que el anillo
$\mathcal C(\intcc01, \R)$ de las funciones continuas no es
\omterm{def:b3:rings:noetherian}{noetheriano}. \emph{(Considérese
$I_n = \{f : f = 0
\text{ en } \intcc0{1/n}\}$.)} (c) Demostrar que en un \emph{dominio}
\omterm{def:b3:rings:noetherian}{noetheriano} todo elemento no nulo y no
unidad es un producto (finito) de
\omterm{def:b3:rings:divisibility}{irreducibles} --- de modo que la no
factorialidad de $\Z[\iu\sqrt 5]$ es únicamente un fallo de
\emph{unicidad}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:rings:11}
Sea $P = X^4 + 1$. (a) Demostrar que $P$ es
\omterm{def:b3:rings:divisibility}{irreducible} sobre $\Q$
(\cref{exo:b3:rings:7}).
(b) Demostrar que $P$ es reducible módulo \emph{todo} primo $p$: trátese
$p = 2$; después, para $p$ impar, demuéstrese que $8 \mid p^2 - 1$ y
admítase por ahora (se demuestra en el~\cref{ch:b3:galois}) que el grupo
multiplicativo del cuerpo de $p^2$ elementos es cíclico, para concluir
que $P$ se descompone en dos factores cuadráticos módulo $p$;
explicítense cuando alguno de $-1$, $2$, $-2$ es un cuadrado módulo $p$,
y demuéstrese que siempre lo es alguno de ellos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:12}
(Los idempotentes descomponen los anillos) Un elemento $e$ de un anillo
conmutativo $A$ es \emph{idempotente} si $e^2 = e$. (a) Demostrar que si
$e$ es idempotente, también lo es $1 - e$, y que la aplicación
$x \mapsto (ex, (1-e)x)$ es un isomorfismo de anillos $A
\cong Ae \times A(1-e)$, donde $Ae$ es un anillo con unidad $e$. (b)
Hallar todos los idempotentes de un dominio y los de $\Z/12\Z$; exhibir
el isomorfismo $\Z/12\Z \cong \Z/4\Z \times \Z/3\Z$ nombrando sus dos
idempotentes no triviales. (c) Demostrar que la descomposición del
teorema chino del resto de $\Z/n\Z$ (\cref{ex:b3:rings:crtz})
corresponde exactamente a los idempotentes
$e_i \equiv 1 \bmod p_i^{a_i}$, $e_i \equiv 0$ módulo las demás
potencias de primos: los anillos se descomponen a lo largo de sus
idempotentes como los espacios se descomponen a lo largo de sus
proyecciones.
\end{exercise}

\section{Problema: el teorema de los dos cuadrados de Fermat}

\begin{problem}[{Problema de fin de semana --- sumas de dos cuadrados,
mediante
$\Z[\iu]$}]\label{pb:b3:rings:1}
¿Qué enteros son suma de dos cuadrados? La respuesta de Fermat (1640) es
una de las joyas de la aritmética; los enteros de Gauss convierten su
demostración en teoría de anillos. En todo el problema,
$N(x + \iu y) = x^2 + y^2$ denota la norma, $\Z[\iu]$ es
\omterm{def:b3:rings:pidufd}{euclídeo} (\cref{exo:b3:rings:4}) y, por
tanto, un \omterm{def:b3:rings:pidufd}{DIP} y un
\omterm{def:b3:rings:pidufd}{DFU}, y \emph{primo de Gauss} significa
\omterm{def:b3:rings:divisibility}{elemento primo} (= \omterm{def:b3:rings:divisibility}{irreducible}) de
$\Z[\iu]$.

\medskip
\noindent\textbf{Parte I --- Normas y primos de Gauss.}
\begin{enumerate}
  \item Comprobar $N(zw) = N(z)N(w)$, deducir de nuevo $\Z[\iu]^\times =
        \{\pm1, \pm\iu\}$ y demostrar la \emph{identidad de
  Brahmagupta}: un producto de dos sumas de dos cuadrados es suma de dos
  cuadrados. \item Demostrar que si $N(z)$ es un número primo, entonces
  $z$ es un primo de Gauss. \item Demostrar que todo primo de Gauss
  $\pi$ divide exactamente a un número primo $p$ \emph{(considérese
  $N(\pi) = \pi\bar\pi$)}, y que entonces $N(\pi) \in \{p, p^2\}$. \item
  Deducir la dicotomía: para cada primo $p$, o bien $p$ sigue siendo
  primo en $\Z[\iu]$ (y no existe ningún primo de Gauss de norma $p$), o
  bien $p = \pi\bar\pi$ con $\pi$ un primo de Gauss de norma $p$ --- y
  entonces $p = a^2 + b^2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El teorema de Wilson y $-1$ módulo
$p$.}
\begin{enumerate}[resume]
  \item Demostrar el \emph{teorema de Wilson}: para $p$ primo, $(p-1)!
        \equiv -1 \pmod p$. \emph{(Emparéjese cada resto con su inverso;
  ¿cuáles quedan emparejados consigo mismos?)} \item Sea $p$ un primo
  impar y $m = \frac{p-1}2$. Demostrar que
  $(m!)^2 \equiv (-1)^{m+1} \pmod p$ \emph{(en $(p-1)!$, sustitúyase
  cada factor $k > m$ por $-(p - k)$)}. \item Concluir: $-1$ es un
  cuadrado módulo $p$ si y solo si $p = 2$ o $p
        \equiv 1 \pmod 4$. \emph{(Para el «solo si»: si $x^2
        \equiv -1$, ¿cuál es el orden de $x$ en $(\Z/p\Z)^\times$ y qué
  dice Lagrange?)}
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La ley de descomposición.}
\begin{enumerate}[resume]
  \item Sea $p \equiv 1 \pmod 4$ y sea $x$ con $p \mid x^2 + 1 =
        (x + \iu)(x - \iu)$. Demostrar que $p$ \emph{no} es un primo de
  Gauss y concluir con la Parte I: $p = a^2 + b^2$. \item Sea
  $p \equiv 3 \pmod 4$. Demostrar directamente que $p$ no es suma de dos
  cuadrados \emph{(cuadrados módulo $4$)} y deducir que $p$ sigue siendo
  primo de Gauss. \item Resolver el caso $p = 2$: exhibir la
  factorización $2 =
        -\iu(1+\iu)^2$ y comprobar que $1 + \iu$ es un primo de Gauss.
  ($2$ es el único primo \emph{ramificado}: divisible por el cuadrado de
  un primo de Gauss salvo unidades.) \item Reunir la clasificación de
  los primos de Gauss, salvo unidades: $1 + \iu$; los enteros
  $p \equiv 3 \pmod 4$; los pares conjugados $\pi, \bar\pi$ de norma
  $p \equiv 1 \pmod
        4$. Comprobarla en $5 = (2+\iu)(2-\iu)$ y en $3$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El teorema de los dos cuadrados.}
\begin{enumerate}[resume]
  \item Demostrar la implicación directa: si en la factorización $n =
        \prod p_i^{\alpha_i}$ todo primo $\equiv 3 \pmod 4$ aparece con
  exponente par, entonces $n$ es suma de dos cuadrados.
  \emph{(Brahmagupta y las Partes II--III.)} \item Demostrar el
  recíproco: si $n = a^2 + b^2 = N(a + \iu b)$ y $q \equiv 3 \pmod 4$
  divide a $n$, demostrar que $q$, primo de Gauss, divide a $a + \iu b$
  o a $a - \iu b$, que de hecho divide tanto a $a$ como a $b$, y
  concluir por inducción sobre $n$ que el exponente de $q$ en $n$ es
  par. \item Enunciar el teorema final. ¿Cuáles de $2025$, $2026$,
  $2027$ son sumas de dos cuadrados? \emph{($2025 = 81 \cdot 25$;
  $2026 = 2 \cdot 1013$, $1013$ primo; $2027$ primo.)} \item (Epílogo)
  Demostrar que un primo $p \equiv 1 \pmod 4$ es suma de dos cuadrados
  de manera esencialmente única: si $p =
        a^2 + b^2 = c^2 + d^2$ (enteros positivos), entonces $\{a, b\}
        = \{c, d\}$. \emph{(Unicidad de la factorización en
        $\Z[\iu]$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Contar representaciones: la fórmula de
Jacobi y la serie de Leibniz.} Escribamos $r_2(n) = \#\{(a, b) \in
\Z^2 : a^2 + b^2 = n\}$ (pares ordenados, con signos y ceros incluidos)
y sea $\chi$ el carácter no trivial módulo $4$: $\chi(d) = +1$ si
$d \equiv 1$, $-1$ si $d \equiv 3 \pmod4$, $0$ si $d$ es par.
\begin{enumerate}[resume]
  \item (Calentamiento, por contraste) ¿Qué enteros son
  \emph{diferencias} de dos cuadrados? Demostrar: $n = a^2 - b^2$ con
  $a, b \in \Z$ si y solo si $n \not\equiv 2 \pmod 4$ --- sin teoría de
  anillos, y sin ninguna estructura comparable a lo que sigue. \item
  Demostrar que $r_2(n)$ es el número de $z \in \Z[\iu]$ con $N(z) = n$.
  Escribiendo $n = 2^{a}\prod_jp_j^{b_j}
        \prod_kq_k^{c_k}$ con $p_j \equiv 1$, $q_k \equiv 3
        \pmod4$, úsense la clasificación de la pregunta 11 y la
  factorización única para demostrar que tales $z$ existen si y solo si
  todos los $c_k$ son pares, y que en ese caso
        \[
        r_2(n) = 4\prod_j\,(b_j + 1) .
        \]
        \emph{(Recuento: $z = u\,(1+\iu)^{a}\prod_j\pi_j^{s_j}
        \bar\pi_j^{\,b_j - s_j}\prod_kq_k^{c_k/2}$ con $u$ una unidad y
        $0 \leq s_j \leq b_j$; ¿por qué es exhaustiva y sin repeticiones
        esta lista?)} \item Demostrar que $d \mapsto \chi(d)$ es
        completamente multiplicativa, deducir que $n \mapsto
        \sum_{d \mid n}\chi(d)$ es multiplicativa y calcularla sobre las
        potencias de primos: vale $1$ en $2^a$; $b + 1$ en $p^b$
        ($p \equiv 1$); y $1$ o $0$ en $q^c$ ($q \equiv 3$) según que
        $c$ sea par o impar. \item Concluir el \emph{teorema de Jacobi}:
        \[
        r_2(n) = 4\sum_{d \mid n}\chi(d) =
        4\bigl(d_1(n) - d_3(n)\bigr),
        \]
        donde $d_i(n)$ cuenta los divisores $\equiv i \pmod 4$.
        Comprobarlo en $n = 3, 5, 9, 25$ y listar las $16$
        representaciones de $65$. \item (El círculo) Demostrar que
        $\sum_{n \leq x}r_2(n)$ es el número de puntos de la red $\Z^2$
        en el disco cerrado de radio $\sqrt x$, y demostrar
        \[
        \sum_{n\leq x}r_2(n) = \pi x + O(\sqrt x)
        \]
        \emph{(cada punto de la red se apropia de un cuadrado unidad;
        compárense áreas, con el error alojado en una corona de anchura
        $O(1)$)}.
  \item (Leibniz, leído aritméticamente) Combínense las preguntas
  19--20:
        \[
        \sum_{d \leq x}\chi(d)\Bigl\lfloor\frac
        xd\Bigr\rfloor = \frac{\pi x}4 + O(\sqrt x),
        \]
        y dedúzcase --- eliminando con cuidado las partes enteras --- la
        serie de Leibniz
        \[
        1 - \frac13 + \frac15 - \frac17 + \dots = \frac\pi4 .
        \]
        La serie alternada de los inversos de los impares \emph{es} el
        exceso medio de divisores $\equiv 1$ sobre divisores $\equiv 3$:
        el análisis calculado por la aritmética. \item (¿Cuán raras son
        las sumas de dos cuadrados?) Demostrar que ningún entero
        $\equiv 3 \pmod 4$ es suma de dos cuadrados (de dos maneras:
        cuadrados módulo $4$, o el criterio de paridad de la pregunta
        17), de modo que al menos una cuarta parte de los enteros queda
        fuera; y demostrar que el promedio $\frac1x\sum_{n\leq
        x}r_2(n) \to \pi$ de la pregunta 20 es compatible con que los
        enteros representables tengan densidad $0$ --- exhíbanse enteros
        con un número anormalmente alto de representaciones (tómense
        productos de muchos primos $\equiv 1 \bmod 4$) para explicar
        cómo una proporción que tiende a cero puede sostener aun así un
        promedio positivo. (Landau demostró que la densidad verdadera
        decae como $1/\sqrt{\log x}$; eso queda fuera de nuestras
        herramientas, pero el mecanismo ya es visible.)
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Complementos: representaciones primitivas
y Pitágoras.}
\begin{enumerate}[resume]
  \item Diremos que una representación $n = a^2 + b^2$ es
  \emph{primitiva} si $\gcd(a, b) = 1$. Demostrar que $n \geq 1$ admite
  una representación primitiva si y solo si $4 \nmid n$ y ningún primo
  $q \equiv 3 \pmod 4$ divide a $n$. \emph{(Para la necesidad,
  reutilícense el descenso de la pregunta 13 y los cuadrados módulo $4$;
  para la suficiencia, constrúyase $z$ a partir de $1 + \iu$ y de los
  $\pi_j$ únicamente --- sin conjugados --- y explíquese por qué un
  factor primo común de $a$ y $b$ metería a la vez $\pi_j$ y
  $\bar\pi_j$, o $(1+\iu)^2$, en
        $z$.)}
  \item (Ternas pitagóricas) Sean $a^2 + b^2 = c^2$ con $a, b,
        c$ positivos, $\gcd(a, b) = 1$ y $b$ par. Demostrar que
  $a + \iu b$ y $a - \iu b$ son coprimos en $\Z[\iu]$ \emph{(un divisor
  primo de Gauss común dividiría a $2a$ y a $2b$, y $c$ es impar)},
  deducir de la factorización única que $a + \iu b = u(m + \iu n)^2$
  para cierta unidad $u$, y concluir la parametrización clásica: salvo
  intercambiar $a$ y $b$,
        \[
        a = m^2 - n^2, \qquad b = 2mn, \qquad c = m^2 + n^2,
        \]
        con $m > n \geq 1$ coprimos de paridades opuestas. Recupérense
        $(3, 4, 5)$ y $(21, 20, 29)$ a partir de $(m, n) =
        (2, 1)$ y $(5, 2)$. \item (Comprobación numérica) Tómese
        $x = 25$. Calcúlese $r_2(n)$ para $1 \leq n \leq 25$ con la
        fórmula de Jacobi, compruébese que los valores no nulos se dan
        exactamente en $n = 1,
        2, 4, 5, 8, 9, 10, 13, 16, 17, 18, 20, 25$, y que
        \[
        \sum_{n \leq 25} r_2(n) = 80
        = 4\sum_{d \leq 25}\chi(d)
        \Bigl\lfloor\frac{25}d\Bigr\rfloor .
        \]
        Compruébese que el disco cerrado de radio $5$ contiene $81$
        puntos de la red y compárese con $\pi x \approx 78.5$: el error
        queda holgadamente dentro del $O(\sqrt x)$ de la pregunta
        20.
\end{enumerate}
\end{problem}
