\chapter{Los espacios \texorpdfstring{$L^p$}{Lp}}\label{ch:b3:lp}

La integral de Lebesgue se construyó para el análisis; los espacios
$L^p$ son el lugar donde ese análisis vive. Son espacios de Banach
(Riesz--Fischer) --- los
\omterm{thm:b3:complete:completion}{completados} que, según
el~\cref{exo:b3:complete:1}, les faltaban a las funciones
\omterm{def:b3:topology:continuity}{continuas} --- y admiten una
tecnología de suavizado, la \emph{convolución contra
\omterm{def:b3:lp:mollifier}{núcleos regularizantes}}, que aproxima
cualquier función de $L^p$ por funciones $\mathcal C^\infty$. Este
capítulo demuestra las versiones integrales de Hölder y Minkowski, la
\omterm{def:b3:complete:complete}{completitud}, los teoremas de
\omterm{ex:b3:lebesgue:gamma}{densidad} y la máquina de regularización,
y termina con la geografía de inclusiones e interpolación de la escala
$L^p$. En todo el capítulo, $(X, \mathcal A, \mu)$ es un
\omterm{def:b3:measure:measure}{espacio de medida} y las funciones toman
valores complejos; sobre $\R^d$, la
\omterm{def:b3:measure:measure}{medida} es
$\lambda_d$.

\section{Definición; Hölder y Minkowski}

\begin{definition}\label{def:b3:lp:space}
Para $1 \leq p < \infty$, $\mathcal L^p(\mu)$ es el conjunto de las $f$
\omterm{def:b3:lebesgue:measurable}{medibles} con
$\norm f_p = \bigl(\int\abs
f^p\dd\mu\bigr)^{1/p} < \infty$, y $\mathcal L^\infty(\mu)$ el conjunto
de las $f$ acotadas fuera de un conjunto nulo, con $\norm
f_\infty$ el \emph{supremo esencial} --- el menor $M$ con
$\abs f \leq M$ en casi todo punto (el ínfimo se alcanza: interséquense
los conjuntos nulos correspondientes a $M + \frac1n$). Como
$\norm f_p = 0$ solo obliga a $f
= 0$ \emph{en casi todo punto} (el~\cref{exo:b3:lebesgue:5}), definimos
\[
L^p(\mu) = \mathcal L^p(\mu)/\{f = 0 \text{ en casi todo punto}\} :
\]
sus elementos son clases de funciones módulo conjuntos nulos, y
$\norm\cdot_p$ es una norma genuina sobre $L^p$.\index{espacio Lp@espacio
$L^p$}
\end{definition}

\begin{theorem}[Desigualdad de Hölder]\label{thm:b3:lp:holder}
Sean $1 \leq p, q \leq \infty$ con $\frac1p + \frac1q = 1$ (exponentes
conjugados). Para $f, g$
\omterm{def:b3:lebesgue:measurable}{medibles}:\index{desigualdad de
Hölder}
\[
\norm{fg}_1 \leq \norm f_p\,\norm g_q ,
\]
con igualdad (para $1 < p < \infty$, normas finitas y $f,g\ne0$) si y
solo si $\abs f^p$ y $\abs g^q$ son proporcionales en casi todo punto.
\end{theorem}

\begin{proof}
Los casos $\{p, q\} = \{1, \infty\}$ son directos ($\abs{fg} \leq
\norm g_\infty\abs f$ en casi todo punto). Sea $1 < p < \infty$;
normalícese $\norm f_p = \norm g_q = 1$ (homogeneidad; las normas nulas
o infinitas son triviales). La desigualdad de Young
$ab \leq \frac{a^p}p +
\frac{b^q}q$ ($a, b \geq 0$; concavidad de $\ln$, como en
el~\cref{pb:b3:banach:1}) da puntualmente $\abs{f g} \leq
\frac{\abs f^p}p + \frac{\abs g^q}q$; intégrese: $\norm{fg}_1
\leq \frac1p + \frac1q = 1$. La igualdad obliga a la igualdad en casi
todo punto en Young, es decir, $\abs f^p = \abs g^q$ en casi todo punto
(tras la normalización; deshaciéndola, la proporcionalidad).
\end{proof}

\begin{theorem}[Desigualdad de Minkowski]\label{thm:b3:lp:minkowski}
Para $1 \leq p \leq \infty$: $\norm{f + g}_p \leq \norm f_p +
\norm g_p$.\index{desigualdad de Minkowski}
\end{theorem}

\begin{proof}
$p = 1, \infty$: desigualdad triangular puntual (o en casi todo punto).
Para $1 <
p < \infty$, supóngase $\norm{f+g}_p < \infty$ (si no, úsese
$\abs{f+g}^p \leq 2^{p-1}(\abs f^p + \abs g^p)$, que se sigue de la
convexidad de $t^p$, para ver que el miembro izquierdo es finito cuando
lo es el derecho). Entonces
\[
\norm{f{+}g}_p^p \leq \int\abs f\,\abs{f{+}g}^{p-1} +
\int\abs g\,\abs{f{+}g}^{p-1}
\leq \bigl(\norm f_p + \norm g_p\bigr)\,
\bigl\|\abs{f{+}g}^{p-1}\bigr\|_q
\]
por Hölder, y $\norm{\abs{f+g}^{p-1}}_q =
\norm{f+g}_p^{p/q}$ porque $(p-1)q = p$; divídase por
$\norm{f+g}_p^{p/q}$ (si no es nulo; si lo es, es trivial) y úsese $p -
\frac pq = 1$.
\end{proof}

\section{Completitud y sus compañeros}

\begin{theorem}[Riesz--Fischer]\label{thm:b3:lp:complete}
Para $1 \leq p \leq \infty$, $L^p(\mu)$ es un espacio de Banach. Además,
toda sucesión convergente en $L^p$ tiene una subsucesión convergente
\emph{en casi todo punto} (con un dominador
en $L^p$ si $p < \infty$).\index{teorema de Riesz--Fischer}
\end{theorem}

\begin{proof}
$p = \infty$: una sucesión de Cauchy para $\norm\cdot_\infty$ es, fuera
de un único conjunto nulo (unión de una cantidad numerable),
uniformemente de Cauchy: converge uniformemente fuera de él; listo. Sea
$p < \infty$. Por el~\cref{exo:b3:complete:1}(b) basta sumar series
absolutamente convergentes: sea $\sum\norm{f_k}_p = M < \infty$. Póngase
$G_n = \sum_{k \leq n}\abs{f_k}$ y $G = \sum_k\abs{f_k}$ (puntualmente
en $[0,\infty]$): por Minkowski $\norm{G_n}_p \leq
M$, y la convergencia monótona ($G_n^p \nearrow G^p$) da
$\int G^p \leq M^p$: $G < \infty$ en casi todo punto, de modo que la
serie $\sum f_k(x)$ converge absolutamente para casi todo $x$; llámese
$S(x)$ a la suma (con cualquier valor sobre el conjunto nulo). Entonces
$\abs{S - \sum_{k\leq n}f_k}^p \leq (2G)^p
\in L^1$, y la convergencia dominada da
$\norm{S - \sum_{k\leq n}f_k}_p \to 0$: la serie converge en $L^p$.

El enunciado sobre subsucesiones: si $f_n \to f$ en $L^p$, tómense $n_k$
con $\norm{f_{n_{k+1}} - f_{n_k}}_p \leq 2^{-k}$; la serie
$\sum(f_{n_{k+1}} - f_{n_k})$ cae bajo el argumento anterior: converge
absolutamente en casi todo punto, dominada por una $G \in
L^p$, de modo que $f_{n_k} \to f_{n_1} + \sum(\cdots) $ en casi todo
punto, y ese límite ha de ser (un representante de) $f$ (ambos son
límites en $L^p$). El dominador: $\abs{f_{n_k}} \leq \abs{f_{n_1}} + G$.
\end{proof}

\begin{remark}\label{rem:b3:lp:modes}
La convergencia en $L^p$ no implica la convergencia en casi todo punto
(la sucesión de la \emph{máquina de escribir}, el~\cref{exo:b3:lp:3}),
ni al revés (jorobas fugitivas): los dos modos solo están ligados a
través de subsucesiones y de la dominación. Tener presentes los
contraejemplos del~\cref{exo:b3:lp:3} es la mejor vacuna.
\end{remark}

\section{Teoremas de densidad}

\begin{theorem}\label{thm:b3:lp:density}
Sea $1 \leq p < \infty$.
\begin{enumerate}
  \item Las \omterm{def:b3:lebesgue:simple}{funciones simples} (con
  soportes de \omterm{def:b3:measure:measure}{medida} finita) son densas
  en $L^p(\mu)$. \item En $L^p(\R^d)$, las funciones
  \omterm{def:b3:topology:continuity}{continuas} de soporte \omterm{def:b3:topology:compact}{compacto}
  $\mathcal C_c(\R^d)$ son densas. \item La traslación es
  \omterm{def:b3:topology:continuity}{continua} en $L^p(\R^d)$:
  escribiendo
        $\tau_hf = f(\cdot - h)$, $\norm{\tau_hf - f}_p \to 0$
        cuando $h \to 0$.
\end{enumerate}
Ninguno de los tres se cumple para $p = \infty$.
\end{theorem}

\begin{proof}
(1) Para $f \geq 0$: las $s_n \nearrow f$ diádicas
del~\cref{thm:b3:lebesgue:approximation} cumplen $\abs{f - s_n}^p
\leq f^p \in L^1$: convergencia dominada. (Cada $s_n \leq f$ está en
$L^p$, y sus conjuntos de nivel tienen
\omterm{def:b3:measure:measure}{medida} finita allí donde el valor es
positivo: $\mu(s_n \geq c) \leq c^{-p}\int f^p$.) Descompóngase una $f$
general en cuatro partes no negativas.

(2) Por (1) basta aproximar $\mathbf 1_A$ con $A$ de Borel y
$\lambda_d(A) < \infty$. La regularidad (demostración como en
el~\cref{thm:b3:measure:regularity}) da
\omterm{def:b3:topology:compact}{compacto} $K \subseteq A
\subseteq U$ abierto con $\lambda_d(U\setminus K) <
\varepsilon$; la función de Urysohn
\[
\varphi(x) = \frac{d(x, \R^d\setminus U)}{d(x, \R^d\setminus U)
+ d(x, K)}
\]
es \omterm{def:b3:topology:continuity}{continua}, vale $1$ sobre $K$ y
$0$ fuera de $U$, y puede tomarse de soporte \omterm{def:b3:topology:compact}{compacto} (redúzcase primero
$U$ a un abierto acotado). Entonces
$\norm{\mathbf 1_A - \varphi}_p^p \leq \lambda_d(U\setminus K)
< \varepsilon$.

(3) Para $g \in \mathcal C_c$: la
\omterm{def:b3:topology:continuity}{continuidad} uniforme da
$\norm{\tau_hg - g}_\infty \to 0$, con soportes en un
\omterm{def:b3:topology:compact}{compacto} fijo para $\abs h \leq 1$:
$\norm{\tau_hg - g}_p \to 0$. Para $f$ general: tómese
$g \in \mathcal C_c$ con $\norm{f - g}_p <
\varepsilon$; entonces $\norm{\tau_hf - f}_p \leq 2\norm{f - g}_p +
\norm{\tau_hg - g}_p$ (invariancia por traslaciones de la norma).

Para $p = \infty$: la aproximación uniforme de $\mathbf
1_{\intoo0\infty}$ por funciones
\omterm{def:b3:topology:continuity}{continuas} es imposible (el salto),
y $\norm{\tau_h\mathbf 1_{\intoo0\infty} - \mathbf
1_{\intoo0\infty}}_\infty = 1$ para $h \neq 0$.
\end{proof}

\section{Convolución y regularización}

\begin{theorem}[Desigualdad de Young]\label{thm:b3:lp:young}
Sean $1 \leq p \leq \infty$, $f \in L^1(\R^d)$, $g \in
L^p(\R^d)$. Entonces $f * g$ está definida en casi todo punto, pertenece
a $L^p$ y\index{desigualdad de Young}
\[
\norm{f * g}_p \leq \norm f_1\,\norm g_p .
\]
\end{theorem}

\begin{proof}
$p = \infty$: cota directa. $p = 1$:
el~\cref{thm:b3:product:convolution}. Sean $1 < p < \infty$ y $q$
conjugados. Descompóngase $\abs{f(y)} = \abs{f(y)}^{1/q}\cdot
\abs{f(y)}^{1/p}$ y aplíquese Hölder:
\[
\int\abs{f(y)}\,\abs{g(x{-}y)}\,\dd y
\leq \Bigl(\int\abs f\Bigr)^{1/q}
\Bigl(\int\abs{f(y)}\,\abs{g(x - y)}^p\,\dd y\Bigr)^{1/p} .
\]
Elévese a la potencia $p$ e intégrese en $x$; Tonelli sobre el segundo
factor da $\norm f_1^{p/q}\cdot\norm f_1\norm g_p^p$, es decir,
$\norm{f*g}_p^p \leq \norm f_1^{1 + p/q}\norm g_p^p =
(\norm f_1\norm g_p)^p$ --- y la finitud de la integral de Tonelli
justifica la convergencia absoluta en casi todo punto, igual que en
\cref{thm:b3:product:convolution}.
\end{proof}

\begin{definition}[Núcleos regularizantes]\label{def:b3:lp:mollifier}
La función\index{núcleo regularizante}\index{función meseta}
\[
\rho(x) = \begin{cases}
c\,\exp\Bigl(-\dfrac{1}{1 - \norm x^2}\Bigr) & \norm x < 1,\\
0 & \norm x \geq 1,
\end{cases}
\]
con $c$ normalizando $\int\rho = 1$, es $\mathcal C^\infty$ en $\R^d$:
la clave es que $t \mapsto \eu^{-1/t}\mathbf
1_{t>0}$ es $\mathcal C^\infty$ en $\R$, con todas sus derivadas nulas
en $0^+$, iguales a $0$ (cada derivada es $P(1/t)\eu^{-1/t}$ para cierto
polinomio $P$, que tiende a $0$; inducción). Para $\varepsilon > 0$,
póngase $\rho_\varepsilon(x) =
\varepsilon^{-d}\rho(x/\varepsilon)$: de soporte en $\bar B(0,
\varepsilon)$ y todavía de integral $1$.
\end{definition}

\begin{theorem}[Regularización]\label{thm:b3:lp:regularization}
Sean $1 \leq p < \infty$ y $f \in L^p(\R^d)$. Entonces:
\begin{enumerate}
  \item $f * \rho_\varepsilon \in \mathcal C^\infty(\R^d)$,
        con $\partial^\alpha(f * \rho_\varepsilon) = f *
        \partial^\alpha\rho_\varepsilon$; \item
        $\norm{f * \rho_\varepsilon - f}_p \to 0$ cuando
        $\varepsilon \to 0$;
  \item en consecuencia, $\mathcal C^\infty_c(\R^d)$ es denso en
        $L^p(\R^d)$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Derivación bajo el signo integral
(el~\cref{thm:b3:lebesgue:paramdiff}) en $x$: para $x$ en una bola $B$,
$\abs{\partial_{x_i}\rho_\varepsilon(x - y)} \leq
C_\varepsilon\,\mathbf 1_{K}(y)$ con $K$
\omterm{def:b3:topology:compact}{compacto} ($y$ a distancia
$\varepsilon$ de $B$), y $\abs f\,\mathbf 1_K \in L^1$ (Hölder contra
$\mathbf 1_K$): el teorema se aplica; itérese para las derivadas de
orden superior.

(2) Como $\int\rho_\varepsilon = 1$:
\[
(f * \rho_\varepsilon)(x) - f(x)
= \int \bigl(f(x - y) - f(x)\bigr)\rho_\varepsilon(y)\,\dd y ,
\]
y la desigualdad \emph{integral} de Minkowski --- o directamente: Hölder
o Jensen con la \omterm{def:b3:measure:measure}{medida} de probabilidad
$\rho_\varepsilon\dd y$, y Tonelli ---
\[
\norm{f*\rho_\varepsilon - f}_p^p
\leq \int\Bigl(\int\abs{f(x-y) -
f(x)}^p\dd x\Bigr)\rho_\varepsilon(y)\,\dd y
= \int \norm{\tau_yf - f}_p^p\;\rho_\varepsilon(y)\,\dd y
\]
(el paso intermedio: aplíquese la desigualdad de Jensen,
el~\cref{exo:b3:lp:10}, a la integral \omterm{def:b3:topology:interior}{interior} en $y$, y después
Tonelli). El integrando tiene soporte en $\norm y \leq \varepsilon$ y
tiende allí a $0$ uniformemente cuando $\varepsilon \to 0$
(el~\cref{thm:b3:lp:density}(3)): toda la expresión tiende a
$0$.

(3) Aproxímese $f$ por $g \in \mathcal C_c$
(el~\cref{thm:b3:lp:density}(2)) y después $g$ por $g *
\rho_\varepsilon \in \mathcal C_c^\infty$ (soporte
\omterm{def:b3:topology:compact}{compacto}: suma de soportes).
\end{proof}

\begin{example}[Regularizar $\abs x$, con velocidades]
\label{ex:b3:lp:mollifyexample}
Tómese $f(x) = \abs x$ en $\R$ (localmente $L^1$; el teorema se aplica
en toda ventana acotada) y un \omterm{def:b3:lp:mollifier}{núcleo
regularizante} simétrico $\rho_\varepsilon$. Entonces
\[
f_\varepsilon(x) = (f * \rho_\varepsilon)(x)
= \int\abs{x - y}\,\rho_\varepsilon(y)\,\dd y
\]
es $\mathcal C^\infty$; lejos del pico no pasa nada: para
$\abs x \geq \varepsilon$, $\abs{x - y}$ es lineal en $x$ sobre el
soporte de $\rho_\varepsilon$, luego $f_\varepsilon(x) =
\abs x$ \emph{exactamente} (la simetría anula la corrección). Cerca de
$0$, el suavizado cuesta exactamente
\[
0 \leq f_\varepsilon(0) = \int\abs
y\,\rho_\varepsilon(y)\,\dd y \leq \varepsilon,
\qquad
\norm{f_\varepsilon - f}_\infty \leq \varepsilon :
\]
el error de aproximación queda confinado al \omterm{def:b3:topology:topology}{entorno} de tamaño
$\varepsilon$ de la singularidad y es de su tamaño. Entretanto
$f_\varepsilon'' \geq 0$ en todas partes ($f$ es convexa, y la
convolución contra $\rho_\varepsilon
\geq 0$ conserva la convexidad), con $\int f_\varepsilon'' =
f_\varepsilon'(\infty) - f_\varepsilon'(-\infty) = 2$: la derivada
segunda es una joroba de masa $2$ comprimida en una anchura
$O(\varepsilon)$, de modo que $\norm{f_\varepsilon''}_\infty \gtrsim
\varepsilon^{-1}$. El suavizado es un trueque: error uniforme
$O(\varepsilon)$ frente a explosión de la derivada $O(\varepsilon^{-1})$
--- el tipo de cambio exacto que el análisis cuantitativo (desigualdades
de interpolación, el círculo de ideas del~\cref{pb:b3:lp:1}) formaliza.
\end{example}

\begin{corollary}[Lema fundamental del cálculo de
variaciones]\label{cor:b3:lp:fundlemma} Sea
$f \in L^1_{\mathrm{loc}}(\R^d)$
(\omterm{def:b3:lebesgue:l1}{integrable} sobre los
\omterm{def:b3:topology:compact}{compactos}) con $\int f\varphi = 0$
para toda $\varphi \in \mathcal
C^\infty_c(\R^d)$. Entonces $f = 0$ en casi todo punto.\index{lema
fundamental del cálculo de variaciones}
\end{corollary}

\begin{proof}
Fíjese una bola $B = B(0, R)$ y sea $g = f\mathbf 1_{B(0, R+1)}
\in L^1$. Para $x \in B$ y $\varepsilon < 1$: $(g *
\rho_\varepsilon)(x) = \int f(y)\rho_\varepsilon(x - y)\dd y =
0$, siendo la función de prueba $y \mapsto \rho_\varepsilon(x-y)
\in \mathcal C_c^\infty$. Pero $g * \rho_\varepsilon \to g$ en $L^1$
(el~\cref{thm:b3:lp:regularization}): $g = 0$ en casi todo punto de $B$;
agótese $\R^d$.
\end{proof}

\section{La geografía de los espacios \texorpdfstring{$L^p$}{Lp}}

\begin{proposition}\label{prop:b3:lp:inclusions}
(a) Si $\mu(X) < \infty$ y $1 \leq p \leq q \leq \infty$, entonces
$L^q \subseteq L^p$ con $\norm f_p \leq
\mu(X)^{\frac1p - \frac1q}\,\norm f_q$. (b) Sobre $\R^d$
(\omterm{def:b3:measure:measure}{medida} infinita) no hay inclusiones:
para $p \neq q$ hay funciones en $L^p\setminus L^q$. (c) (Interpolación)
Si $p < r < q$ y $\alpha \in \intoo01$ está definido por
$\frac1r = \frac\alpha p + \frac{1 - \alpha}q$, entonces
\[
\norm f_r \leq \norm f_p^{\alpha}\,\norm f_q^{1 - \alpha} ;
\]
en particular, $L^p \cap L^q \subseteq L^r$.
\end{proposition}

\begin{proof}
(a) Hölder con los exponentes $\frac qp$ y su conjugado:
$\int\abs f^p\cdot 1 \leq \norm{\abs f^p}_{q/p}\,\norm
1_{(q/p)'} = \norm f_q^p\,\mu(X)^{1 - p/q}$ ($q = \infty$
directamente). (b) Cerca de $0$ y cerca de $\infty$, las potencias
$x^{-\alpha}$ calibran: el~\cref{exo:b3:lp:2}. (c) Escríbase
$\abs f^r = \abs
f^{r\alpha}\abs f^{r(1-\alpha)}$ y aplíquese Hölder con el par conjugado
$\frac p{r\alpha}$, $\frac q{r(1-\alpha)}$ (conjugados precisamente por
la definición de $\alpha$):
$\int\abs f^r \leq \norm f_p^{r\alpha}\norm f_q^{r(1 -
\alpha)}$.
\end{proof}

\begin{method}\label{met:b3:lp:toolkit}
El instrumental de $L^p$, tal como se usa en todo lo que sigue: para
demostrar una identidad o una desigualdad para todo $f \in L^p$ ---
demuéstrese sobre una clase densa ($\mathcal C_c^\infty$ mediante
el~\cref{thm:b3:lp:regularization}) y extiéndase por
\omterm{def:b3:topology:continuity}{continuidad}
(el~\cref{thm:b3:complete:extension}, siendo ambos miembros
$L^p$-continuos); para demostrar $f = 0$, evalúese contra $\mathcal
C_c^\infty$ (el~\cref{cor:b3:lp:fundlemma}); para ganar regularidad,
convoluciónese; para cambiar de exponente, Hölder e interpolación. La
teoría de Fourier del~\cref{ch:b3:fouriertransform} es una larga
aplicación de este método.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:lp:1}
(a) Enunciar y demostrar la desigualdad de Cauchy--Schwarz en $L^2(\mu)$
como el caso $p = q = 2$ de Hölder. (b) Sobre un espacio de
\emph{probabilidad}, demostrar que $p \mapsto \norm f_p$ es no
decreciente. (c) ¿Cuándo es Hölder una igualdad para $p = 1$,
$q = \infty$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:lp:2}
¿Para qué $p \in \intco1\infty$ pertenecen las siguientes funciones a
$L^p$?
\[
x^{-1/2}\mathbf 1_{\intoo01},\qquad
x^{-1/2}\mathbf 1_{\intoo1\infty},\qquad
\frac{1}{x^{1/2}(1 + \abs{\ln x})}\ \text{en } \intoo01,
\qquad
\frac1{1 + \abs x}\ \text{en } \R .
\]
Conclúyase: en $\intoo01$, $p$ pequeño es más fácil; en $\intoo1\infty$,
$p$ grande es más fácil; y ningún $L^p$ contiene a otro en $\R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:3}
(La máquina de escribir) Enumérense los intervalos diádicos $I_1 =
\intcc01$, $I_2 = \intcc0{\frac12}$, $I_3 = \intcc{\frac12}1$,
$I_4 = \intcc0{\frac14}$, \dots\ y sea $f_n = \mathbf
1_{I_n}$. (a) Demostrar que $f_n \to 0$ en todo $L^p(\intcc01)$,
$p < \infty$, pero que $(f_n(x))$ diverge para \emph{todo}
$x \in \intcc01$. (b) Exhibir la subsucesión convergente en casi todo
punto que promete el~
\cref{thm:b3:lp:complete}.
(c) Recíprocamente, dese una sucesión que converja en casi todo punto
pero no en $L^1$, y otra que converja en $L^1$ pero en ningún $L^p$,
$p > 1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:4}
Sean $\mu(X) < \infty$ y $f \in L^\infty(\mu)$, $f \neq 0$. Demostrar
que $\norm f_p \to \norm f_\infty$ cuando $p \to \infty$. \emph{(Cota
superior por (a) de la~\cref{prop:b3:lp:inclusions}; cota inferior
integrando sobre $\{\abs f > \norm f_\infty -
\varepsilon\}$, de \omterm{def:b3:measure:measure}{medida} positiva.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:5}
(a) ¿Dónde usa exactamente la demostración
del~\cref{thm:b3:lp:density}(3) que $p < \infty$? (b) Demostrar que
$f \in L^\infty(\R)$ cumple $\norm{\tau_hf -
f}_\infty \to 0$ si y solo si $f$ tiene un representante uniformemente
\omterm{def:b3:topology:continuity}{continuo}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:6}
Sean $p, q$ conjugados, $f \in L^p(\R^d)$, $g \in
L^q(\R^d)$. Demostrar que $f * g$ está definida \emph{en todo punto}, es
acotada, con $\norm{f*g}_\infty \leq \norm f_p\norm g_q$, y
\emph{uniformemente \omterm{def:b3:topology:continuity}{continua}}.
\emph{(\omterm{def:b3:topology:continuity}{Continuidad} de la traslación
en $L^p$; trátese $p \in \{1, \infty\}$ aparte --- para $p =
\infty$, úsese la \omterm{def:b3:topology:continuity}{continuidad} de la
traslación sobre el factor $L^1$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:7}
Sea $f \in L^1_{\mathrm{loc}}(\intoo ab)$ con $\int f\varphi' = 0$ para
toda $\varphi \in \mathcal
C^\infty_c(\intoo ab)$. Demostrar que $f$ es igual en casi todo punto a
una constante. \emph{(Fíjese $\chi \in \mathcal C_c^\infty$ con
$\int\chi = 1$; toda $\psi \in \mathcal C_c^\infty$ con $\int\psi = 0$
es una $\varphi'$; escríbase una función de prueba general como
$\psi + (\int\psi)\chi$ y aplíquese el~\cref{cor:b3:lp:fundlemma} a
$f - c$ con $c = \int
f\chi$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:lp:8}
(Urysohn regular) Sean $K \subseteq U \subseteq \R^d$, $K$
\omterm{def:b3:topology:compact}{compacto} y $U$ abierto. Constrúyase
$\varphi \in \mathcal
C^\infty_c(\R^d)$ con $0 \leq \varphi \leq 1$, $\varphi = 1$ sobre $K$ y
$\operatorname{supp}\varphi \subseteq U$. \emph{(Regularícese el
indicador del \omterm{def:b3:topology:topology}{entorno} $K_\delta$ de tamaño $\delta$ de $K$ con
$\rho_{\delta/2}$, para $\delta$ pequeño.)} Dedúzcase un enunciado de
partición de la unidad $\mathcal C^\infty$ para un
\omterm{def:b3:topology:compact}{compacto} recubierto por un número
finito de abiertos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:9}
Usando la interpolación (la~\cref{prop:b3:lp:inclusions}(c)): (a)
demuéstrese que $L^1(\R)\cap L^\infty(\R) \subseteq L^p(\R)$ para todo
$p$, con $\norm f_p \leq \norm f_1^{1/p}\norm
f_\infty^{1 - 1/p}$; (b) demuéstrese que $f \mapsto \norm f_p$ es, para
$f$ fija, logarítmicamente convexa en $\frac1p$, y dese un ejemplo en el
que $f \in L^p$ exactamente para $p$ en un intervalo dado $(p_0, p_1)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:10}
(Jensen) Sean $\mu$ una \omterm{def:b3:measure:measure}{medida} de
\emph{probabilidad}, $f \in
L^1(\mu)$ real y $\Phi \colon \R \to \R$ convexa. Demostrar
\[
\Phi\Bigl(\int f\,\dd\mu\Bigr) \leq \int \Phi\circ
f\,\dd\mu
\]
\emph{(recta de apoyo de $\Phi$ en el punto $m = \int f$)}. Dedúzcanse
la desigualdad entre las medias aritmética y geométrica y la monotonía
de $p \mapsto \norm f_p$ del~\cref{exo:b3:lp:1}(b).\index{desigualdad de
Jensen}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:lp:11}
(Desigualdad de Young para la convolución) Sean $1 \leq p, q, r \leq
\infty$ con $\frac1p + \frac1q = 1 + \frac1r$, y $f \in
L^p(\R^d)$, $g \in L^q(\R^d)$. (a) Demostrar
$\norm{f * g}_r \leq \norm f_p\,\norm g_q$. \emph{(Escríbase, con los
exponentes conjugados deducidos de $p, q,
r$,
\[
\abs{f(y)g(x-y)} = \bigl(\abs f^p\abs g^q\bigr)^{1/r}
\cdot\abs f^{\,p(1/p - 1/r)}\cdot\abs g^{\,q(1/q - 1/r)},
\]
y aplíquese la desigualdad de Hölder con tres factores y exponentes $r$,
$\frac{pr}{r - p}$, $\frac{qr}{r - q}$; intégrese después en $x$ por
Tonelli.)} (b) Compruébense los tres casos especiales ya conocidos: $r =
\infty$ (Hölder, el~\cref{exo:b3:lp:6}); $q = 1$ (estabilidad $L^p$ de
la convolución por un núcleo \omterm{def:b3:lebesgue:l1}{integrable});
$p = q = 1$ ($L^1$ es un álgebra para la convolución,
\cref{thm:b3:product:convolution}).
(c) ¿Por qué no hay ninguna desigualdad con $\frac1p + \frac1q < 1 +
\frac1r$? \emph{(Evalúese en las dilataciones $f_\lambda(x) =
f(\lambda x)$ y compárense los cambios de escala de ambos miembros.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:12}
(Casos de igualdad) (a) En la desigualdad de Hölder
$\int\abs{fg} \leq \norm f_p\norm g_q$ ($1 < p < \infty$),
demuéstrese que hay igualdad si y solo si $\abs f^p$ y $\abs g^q$ son
proporcionales en casi todo punto. \emph{(Sígase el caso de igualdad de
la desigualdad de Young $ab \leq \frac{a^p}p + \frac{b^q}q$, que es $a^p
= b^q$.)} (b) En la desigualdad de Minkowski
$\norm{f + g}_p \leq \norm f_p
+ \norm g_p$ ($1 < p < \infty$), demuéstrese que la igualdad con $f,
g \neq 0$ obliga a $g = cf$ en casi todo punto con $c > 0$. (c)
Contrástese con $p = 1$ y $p = \infty$: descríbanse los casos de
igualdad (mucho más amplios) allí, con ejemplos.
\end{exercise}

\section{Problema: la desigualdad de Hardy}

\begin{problem}[{Problema de fin de semana --- la desigualdad de Hardy y
su constante óptima}]\label{pb:b3:lp:1} Para
$f \in L^p(\intoo0{+\infty})$, $1 < p < \infty$, defínase el
\emph{operador de Hardy}
\[
(Hf)(x) = \frac1x\int_0^x f(t)\,\dd t .
\]
La desigualdad de Hardy (1920) afirma que\index{desigualdad de Hardy}
\[
\norm{Hf}_p \;\leq\; \frac{p}{p-1}\,\norm f_p ,
\]
y que la constante $\frac p{p-1}$ es óptima y no se alcanza. Este
problema lo demuestra todo y extiende después el resultado a series.

\medskip
\noindent\textbf{Parte I --- La desigualdad.} Supóngase primero $f
\geq 0$ \omterm{def:b3:topology:continuity}{continua} de soporte
\omterm{def:b3:topology:compact}{compacto} en $\intoo0{+\infty}$, y sea
$F(x) = \int_0^xf$.
\begin{enumerate}
  \item Demostrar que $Hf \in L^p$: cerca de $0$, $F$ se anula en un
  \omterm{def:b3:topology:topology}{entorno} de $0$; cerca de $\infty$,
  $F$ está acotada, luego $(Hf)(x) = O(1/x)$, y $x \mapsto \frac1x$
  pertenece a $L^p(\intoo1{+\infty})$ para $p > 1$. \item Intégrese por
  partes para demostrar
        \[
        \int_0^\infty \Bigl(\frac Fx\Bigr)^p\dd x
        = \frac{p}{p-1}\int_0^\infty\Bigl(\frac
        Fx\Bigr)^{p-1}f(x)\,\dd x .
        \]
        \emph{(Derívese $x^{1-p}F^p$; los términos de frontera se anulan
        --- justifíquense ambos extremos.)} \item Aplíquese Hölder al
        miembro derecho y dedúzcase
        $\norm{Hf}_p \leq \frac p{p-1}\norm f_p$ para tales $f$. \item
        Extiéndase a todo $L^p$: para $f \geq 0$, constrúyanse $f_n$
        \omterm{def:b3:topology:continuity}{continuas} de soporte
        \omterm{def:b3:topology:compact}{compacto} en
        $\intoo0{+\infty}$, $0 \leq f_n \nearrow f$ a.e.\
        \emph{(trúnquese y aproxímese después monótonamente ---
        justifíquese la construcción)}; entonces $Hf_n \nearrow Hf$
        puntualmente (convergencia monótona dentro del promedio) y la
        convergencia monótona pasa la desigualdad al límite. Para $f$
        con signo o compleja, concluir con $\abs{Hf} \leq H\abs f$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Optimalidad.}
\begin{enumerate}[resume]
  \item Para $A > 1$, sea $f_A(t) = t^{-1/p}\,\mathbf
        1_{\intcc1A}(t)$. Calcúlese $\norm{f_A}_p^p = \ln A$ y, para
  $1 \leq x \leq A$,
        \[
        (Hf_A)(x) = \frac p{p-1}\;x^{-1/p}\,
        \bigl(1 - x^{-(1 - 1/p)}\bigr).
        \]
  \item Deducir $\liminf_{A\to\infty} \norm{Hf_A}_p/
        \norm{f_A}_p \geq \frac{p}{p-1}$ y concluir que la constante es
  óptima. \item Demostrar que la igualdad $\norm{Hf}_p = \frac
        p{p-1}\norm f_p$ con $f \neq 0$ es imposible. \emph{(Sígase el
  caso de igualdad de Hölder en la pregunta 3: forzaría un
  comportamiento de tipo $f = cx^{-1/p}$, que no está en $L^p$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La desigualdad discreta.}
\begin{enumerate}[resume]
  \item Para $g \geq 0$ no creciente sobre $(0,\infty)$ y $a_n = g(n)$,
  compárense $\sum a_n^p$ y $\int g^p$, y los promedios $H$ en
  consecuencia, para deducir de la Parte I la \emph{desigualdad discreta
  de Hardy}: para $a_n \geq 0$,
        \[
        \sum_{n\geq1}\Bigl(\frac{a_1 + \dots +
        a_n}{n}\Bigr)^{p}
        \;\leq\;
        \Bigl(\frac{p}{p-1}\Bigr)^{p}\,\sum_{n\geq1}a_n^p
        \]
        --- demuéstrese primero para $(a_n)$ no creciente mediante la
        comparación anterior, y redúzcase después el caso general al no
        creciente por reordenación \emph{(admítase, con una
        justificación de una línea, que ordenar $(a_n)$ de forma
        decreciente solo puede aumentar el miembro izquierdo dejando
        fijo el derecho)}. \item Deducir: si $\sum a_n^p < \infty$,
        entonces las medias de Cesàro de $(a_n)$ están de nuevo en
        $\ell^p$ --- y dese un ejemplo ($p = 2$) en el que
        $(a_n) \in \ell^2$ pero $a_n$ no sea sumable, y Hardy siga
        controlando las medias.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Epílogo.}
\begin{enumerate}[resume]
  \item Demostrar que la desigualdad de Hardy falla para $p = 1$: con
  $f = \mathbf 1_{\intcc01}$, calcúlese $Hf$ y obsérvese que
  $Hf \notin L^1$. ¿Dónde se rompe la demostración?
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- La función maximal y el teorema de
derivación de Lebesgue.} Hardy promedia desde el origen;
Hardy--Littlewood promedian \emph{alrededor de cada punto}. Para
$f \in L^1(\R)$, defínase
\[
Mf(x) = \sup_{r>0}\ \frac1{2r}\int_{x-r}^{x+r}\abs
f\,\dd\lambda .
\]
\begin{enumerate}[resume]
  \item (Vitali, versión finita) Sean $B_1, \dots, B_N$ intervalos
  abiertos. Demostrar que existe una subfamilia disjunta
  $B_{i_1}, \dots, B_{i_k}$ con $\bigcup_jB_j
        \subseteq \bigcup_l3B_{i_l}$, donde $3B$ denota el intervalo del
  mismo \omterm{ex:b3:groups:actions}{centro} y longitud triple
  \emph{(voraz: tómese repetidamente el intervalo más largo disjunto de
  los ya tomados)}. \item (Tipo débil $(1,1)$) Demostrar que, para todo
  $t > 0$,
        \[
        \lambda\bigl(\{Mf > t\}\bigr) \;\leq\;
        \frac3t\,\norm f_1 :
        \]
        cada $x$ con $Mf(x) > t$ posee un intervalo centrado $B_x$ con
        $\int_{B_x}\abs f > t\,\lambda(B_x)$; tómese un
        \omterm{def:b3:topology:compact}{compacto}
        $K \subseteq \{Mf > t\}$ (regularidad \omterm{def:b3:topology:interior}{interior}), recúbrase por
        un número finito de $B_x$, aplíquese la pregunta 11 y agótese.
        \item Calcúlese $M\mathbf 1_{\intcc01}(x)$ para $x > 1$ y
        dedúzcase que $Mf \notin L^1$ para todo $f \neq 0$
        ($Mf(x) \geq \frac c{\abs x}$ en el infinito): en $p =
        1$, la desigualdad débil de la pregunta 12 es el mejor enunciado
        posible. \item (Tipo fuerte para $p > 1$) Para $f \in L^p$:
        descompóngase $f =
        f\,\mathbf 1_{\abs f > t/2} + f\,\mathbf 1_{\abs f
        \leq t/2}$, obsérvese que $Mf \leq M\bigl(f\mathbf 1_{\abs
        f > t/2}\bigr) + \frac t2$, y combínese la pregunta 12 con la
        fórmula de las capas (la~\cref{prop:b3:product:layercake}) y
        Tonelli para demostrar
        \[
        \norm{Mf}_p^p \;\leq\;
        \frac{6p\,2^{p-1}}{p-1}\,\norm f_p^p .
        \]
        (La explosión cuando $p \downarrow 1$ es el fallo de la pregunta
        13, cuantificado.) \item (Teorema de derivación de Lebesgue)
        Demostrar que, para $f \in
        L^1(\R)$,
        \[
        \frac1{2r}\int_{x-r}^{x+r}f\,\dd\lambda
        \;\longrightarrow\; f(x)
        \qquad (r \to 0)\quad\text{para casi todo }x .
        \]
        \emph{(Claro para $f$
        \omterm{def:b3:topology:continuity}{continua}. En general,
        escríbase $f
        = g + h$ con $g$ \omterm{def:b3:topology:continuity}{continua}
        de soporte \omterm{def:b3:topology:compact}{compacto},
        $\norm h_1 < \varepsilon$
        (el~\cref{thm:b3:lp:density}); el conjunto donde
        $\limsup_{r\to0}$ de la oscilación promediada supera $\delta$
        está dentro de $\{Mh > \delta/2\} \cup \{\abs h
        > \delta/2\}$, de \omterm{def:b3:measure:measure}{medida}
        $O(\varepsilon/\delta)$; hágase $\varepsilon \to 0$ y después
        $\delta \to 0$ a lo largo de una sucesión.)} \item Dedúzcase:
        (a) casi todo punto es un \emph{punto de Lebesgue} de $f$; (b)
        para $f \in L^1$, la primitiva $F(x) = \int_0^xf$ es derivable
        en casi todo punto con $F' =
        f$ en casi todo punto --- la mitad integral del teorema
        fundamental del cálculo en el mundo de Lebesgue, que cierra el
        círculo abierto por la escalera (el~\cref{pb:b3:measure:1}), la
        cual mostraba que la otra mitad puede fallar. \item (Puntos de
        \omterm{ex:b3:lebesgue:gamma}{densidad}) Para $A \subseteq \R$
        \omterm{def:b3:lebesgue:measurable}{medible}, demuéstrese que
        casi todo $x \in A$ cumple
        $\frac{\lambda(A\cap\intcc{x-r}{x+r})}{2r} \to 1$:
        los conjuntos \omterm{def:b3:lebesgue:measurable}{medibles} son
        localmente llenos en casi todos sus puntos. Esbócese en dos
        líneas cómo esto da una demostración más del teorema de
        Steinhaus
        (\cref{exo:b3:measure:8}).
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Variaciones sobre el tema del promedio.}
\begin{enumerate}[resume]
  \item (Hardy con peso) Para $\alpha < p - 1$ y $f \geq 0$, demostrar
        \[
        \int_0^\infty\Bigl(\frac{F(x)}x\Bigr)^{p}
        x^{\alpha}\,\dd x \;\leq\;
        \Bigl(\frac{p}{p - 1 - \alpha}\Bigr)^{p}
        \int_0^\infty f(x)^p\,x^{\alpha}\,\dd x
        \]
        mediante la misma integración por partes, y compruébese que el
        caso límite $\alpha = p - 1$ está genuinamente prohibido
        (adáptese el contraejemplo de la pregunta 10). \item (El
        adjunto) Sea $H^*f(x) =
        \int_x^{\infty}\frac{f(t)}t\,\dd t$. Demostrar $\langle
        Hf, g\rangle = \langle f, H^*g\rangle$ para $f, g$ no negativas
        (Tonelli) y demostrar $\norm{H^*f}_p \leq p\,\norm f_p$
        \emph{(directamente por partes, o a partir de Hardy en el
        exponente conjugado por dualidad --- atención a qué exponente se
        lleva qué constante)}. \item (Una desigualdad de tipo Hilbert)
        Dedúzcase que, para $f \in L^p$ no negativas, $g \in L^q$:
        \[
        \int_0^\infty\!\!\int_0^\infty
        \frac{f(x)\,g(y)}{\max(x,y)}\,\dd x\,\dd y
        \;\leq\; (p + q)\,\norm f_p\,\norm g_q
        \]
        \emph{(sepárese según $y < x$ / $y \geq x$: cada mitad es un
        emparejamiento de una función contra la transformada de Hardy de
        la otra)}. \item (Optimalidad, caso discreto) Demostrar que la
        constante $\bigl(\frac p{p-1}\bigr)^p$ de la pregunta 8 también
        es óptima: evalúese en $a_n = n^{-1/p}\,\mathbf 1_{n \leq
        N}$, compárense ambos miembros con integrales y hágase $N
        \to \infty$ (el espejo discreto de la Parte II). \item
        (Síntesis) En este problema han aparecido tres operadores de
        promedio: el $H$ de Hardy, la media discreta de Cesàro y el
        operador maximal $M$. Enúnciese en una línea qué dice la
        acotación de cada uno, obsérvese que los tres fallan exactamente
        en $p = 1$ y explíquese por qué se trata del mismo fallo tres
        veces (la cola armónica
        $\frac1x$).
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- La desigualdad de Carleman, y cuán óptimo
es lo óptimo.}
\begin{enumerate}[resume]
  \item (Desigualdad de Carleman) Sea $a_n \geq 0$ con $\sum
        a_n < \infty$. Aplíquese la desigualdad discreta de Hardy de la
  pregunta 8 a $b_n = a_n^{1/p}$, úsese la desigualdad entre las medias
  aritmética y geométrica y hágase $p \to
        \infty$ (demuéstrese que $p \mapsto
        \bigl(\frac p{p-1}\bigr)^p$ decrece hacia $\eu$) para obtener
        \[
        \sum_{n\geq1}\bigl(a_1a_2\cdots a_n\bigr)^{1/n}
        \;\leq\; \eu\,\sum_{n\geq1}a_n :
        \]
        las medias geométricas de una sucesión sumable son sumables, a
        un coste de a lo sumo $\eu$. \item (La constante $\eu$ es
        óptima) Evalúese en $a_n =
        \frac1n\,\mathbf 1_{n\leq N}$: usando el encaje de Stirling
        del~\cref{pb:b3:product:1}, demuéstrese
        $(n!)^{-1/n} = \frac\eu n\bigl(1 +
        O\bigl(\frac{\ln n}n\bigr)\bigr)$, dedúzcase que ambos miembros
        de Carleman crecen como $\eu\ln N$ y concluir que ninguna
        constante menor que $\eu$ puede servir. (Obsérvese el patrón:
        los optimizadores de Hardy y de Carleman son ambos las
        sucesiones de tipo armónico que están justo fuera del espacio.)
        \item (¿Con qué lentitud se aproxima lo «óptimo»?) Tómese
        $p = 2$. Para $f = \mathbf 1_{\intcc01}$, calcúlese
        $\norm{Hf}_2/\norm f_2 = \sqrt2$, frente a la cota $2$. Para los
        casi optimizadores $f_A$ de la pregunta 5, demuéstrese la
        identidad exacta
        \[
        \norm{Hf_A}_2^2 = 4\ln A - 8 + \frac{8}{\sqrt A},
        \qquad\text{luego}\qquad
        \frac{\norm{Hf_A}_2^2}{\norm{f_A}_2^2}
        = 4 - \frac{8 - 8A^{-1/2}}{\ln A} .
        \]
        Evalúese en $A = \eu^{10}$ (cociente $\approx 1.790$) y
        coméntese: el supremo $2$ se aproxima solo a velocidad $1/\ln A$
        --- una constante óptima puede ser prácticamente invisible desde
        el punto de vista numérico.
\end{enumerate}
\end{problem}
