\chapter{La transformada de Fourier}\label{ch:b3:fouriertransform}

Las series de Fourier descomponen las señales periódicas en armónicos
discretos; la \emph{transformada} de Fourier hace lo mismo con las
señales sobre toda la recta, con un \omterm{def:b3:topology:continuity}{continuo} de frecuencias. Convierte
la derivación en multiplicación, la convolución en productos y las
gaussianas en gaussianas --- las razones por las que resuelve ecuaciones
diferenciales, mueve el procesamiento de señales y demostrará el teorema
central del límite en el~\cref{ch:b3:clt}. Este capítulo desarrolla la
teoría en $L^1$ (Riemann--Lebesgue, inversión, inyectividad), la
\omterm{def:b3:fouriertransform:schwartz}{clase de Schwartz} donde la
transformada es una biyección \omterm{prop:b3:galois:perfect}{perfecta},
y la teoría en $L^2$ (Plancherel: la transformada es, salvo una
constante, un operador unitario), con dos aplicaciones estelares: la
ecuación del calor, resuelta de principio a fin en el problema de fin de
semana, y la fórmula de sumación de Poisson. Convenio:
\[
\hat f(\xi) = \int_\R f(x)\,\eu^{-\iu\xi x}\,\dd x .
\]

\section{La transformada en \texorpdfstring{$L^1$}{L1}}

\begin{proposition}\label{prop:b3:fouriertransform:basic}
Para $f \in L^1(\R)$: $\hat f$ está bien definida, es acotada
($\norm{\hat f}_\infty \leq \norm f_1$) y
\omterm{def:b3:topology:continuity}{continua}, y:
\begin{enumerate}
  \item $\widehat{\tau_af}(\xi) = \eu^{-\iu a\xi}\hat f(\xi)$
        y $\widehat{\eu^{\iu ax}f}(\xi) = \hat f(\xi - a)$; \item
        $\widehat{f(\cdot/\lambda)}(\xi) = \lambda\hat
        f(\lambda\xi)$ para $\lambda > 0$; \item si $xf \in L^1$,
        entonces $\hat f$ es $\mathcal C^1$ con
        $(\hat f)'(\xi) = \widehat{(-\iu x)f}(\xi)$;
  \item si $f \in \mathcal C^1$ con $f' \in L^1$ (y $f \to
        0$ en $\pm\infty$, automático aquí), entonces
        $\widehat{f'}(\xi) = \iu\xi\hat f(\xi)$;
  \item $\widehat{f * g} = \hat f\,\hat g$ para $f, g \in L^1$.
\end{enumerate}
\end{proposition}

\begin{proof}
Acotación: $\abs{\hat f} \leq \int\abs f$.
\omterm{def:b3:topology:continuity}{Continuidad}: convergencia dominada
con dominador $\abs f$ (el~\cref{thm:b3:lebesgue:paramcont}). (1) y (2):
sustituciones (el~\cref{thm:b3:product:linearchange}). (3): derivación
bajo el signo integral, con dominador $\abs{xf}$
(el~\cref{thm:b3:lebesgue:paramdiff}). (4): primero, $f(x) = f(0)
+ \int_0^xf'$ tiene límite en $\pm\infty$ ($f' \in L^1$), que ha de ser
$0$ ($f \in L^1$); después intégrese por partes en $[-A,
A]$ y hágase $A \to \infty$. (5): Fubini, lícito porque
$(x,y)\mapsto f(x - y)g(y)\eu^{-\iu\xi x}$ es absolutamente
\omterm{def:b3:lebesgue:l1}{integrable}
(el~\cref{thm:b3:product:convolution}):
\[
\widehat{f*g}(\xi) = \iint f(x - y)g(y)\eu^{-\iu\xi(x - y)}
\eu^{-\iu\xi y}\dd x\,\dd y = \hat f(\xi)\,\hat g(\xi).
\]
\end{proof}

\begin{example}\label{ex:b3:fouriertransform:gaussian}
La gaussiana: para $a > 0$,\index{gaussiana, transformada de Fourier}
\[
\widehat{\eu^{-ax^2}}(\xi) =
\sqrt{\frac\pi a}\;\eu^{-\xi^2/4a} :
\]
por el~\cref{exo:b3:lebesgue:7} (el truco de la ecuación diferencial
$F' = -\frac\xi{2}F$, reescalado), o por (3):
$g = \widehat{\eu^{-ax^2}}$ cumple $g'(\xi) = -\frac{\xi}{2a}g(\xi)$
(intégrese por partes) y $g(0) = \sqrt{\pi/a}$. Las gaussianas son
puntos fijos de la transformada salvo cambios de escala --- la razón
profunda de que gobiernen el teorema central del límite.
\end{example}

\begin{theorem}[Riemann--Lebesgue]
\label{thm:b3:fouriertransform:riemannlebesgue}
Para $f \in L^1(\R)$: $\hat f(\xi) \to 0$ cuando $\abs\xi \to
\infty$. Así pues, $\widehat{\phantom f} \colon L^1 \to \mathcal
C_0(\R)$ (funciones \omterm{def:b3:topology:continuity}{continuas} que
se anulan en el infinito).\index{lema de Riemann--Lebesgue}
\end{theorem}

\begin{proof}
Para el indicador de un intervalo, $\widehat{\mathbf 1_{\intcc
ab}}(\xi) = \frac{\eu^{-\iu a\xi} - \eu^{-\iu b\xi}}{\iu\xi}
\to 0$; y de ahí para las funciones escalonadas. Las funciones
escalonadas son densas en $L^1$ (el~\cref{thm:b3:lp:density}(1) más la
aproximación de conjuntos de \omterm{def:b3:measure:measure}{medida}
finita por uniones finitas de intervalos, el~\cref{exo:b3:measure:7}),
y la transformada es \omterm{def:b3:topology:continuity}{continua} de
$\norm\cdot_\infty$ a $\norm\cdot_1$: para $\norm{f -
s}_1 < \varepsilon$, $\limsup_{\abs\xi\to\infty}\abs{\hat
f(\xi)} \leq \varepsilon$.
\end{proof}

\section{Inversión e inyectividad}

\begin{lemma}[Fórmula de multiplicación]
\label{lem:b3:fouriertransform:multiplication}
Para $f, g \in L^1(\R)$: $\displaystyle\int \hat f\,g =
\int f\,\hat g$.
\end{lemma}

\begin{proof}
Ambos miembros valen $\iint f(x)g(\xi)\eu^{-\iu x\xi}\dd
x\,\dd\xi$ (Tonelli--Fubini: la integral doble del valor absoluto es
$\norm f_1\norm g_1$).
\end{proof}

\begin{theorem}[Inversión]\label{thm:b3:fouriertransform:inversion}
Sea $f \in L^1(\R)$.\index{teorema de inversión de Fourier}
\begin{enumerate}
  \item (Sumabilidad gaussiana) Para todo $x$,
        \[
        (f * g_\varepsilon)(x) =
        \frac1{2\pi}\int_\R \hat f(\xi)\,
        \eu^{-\varepsilon\xi^2}\,\eu^{\iu x\xi}\,\dd\xi,
        \qquad\text{donde } g_\varepsilon(y) =
        \frac{1}{2\sqrt{\pi\varepsilon}}\,
        \eu^{-y^2/4\varepsilon},
        \]
        y $f * g_\varepsilon \to f$ en $L^1$ cuando $\varepsilon
        \to 0$. \item Si además $\hat f \in L^1$, entonces para casi
        todo
        $x$
        \[
        f(x) = \frac{1}{2\pi}\int_\R \hat
        f(\xi)\,\eu^{\iu x\xi}\,\dd\xi ,
        \]
        y $f$ tiene un representante
        \omterm{def:b3:topology:continuity}{continuo}. \item
        (Inyectividad) Si $\hat f = 0$, entonces $f = 0$ en casi todo
        punto.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Fíjese $x$ y aplíquese
el~\cref{lem:b3:fouriertransform:multiplication} a $f$ y a
$g(\xi) = \frac1{2\pi}\eu^{-\varepsilon\xi^2}\eu^{\iu x\xi}$:
por el~\cref{ex:b3:fouriertransform:gaussian} (con la regla de
modulación),
\[
\hat g(y) = \frac1{2\pi}\sqrt{\frac\pi\varepsilon}\,
\eu^{-(y - x)^2/4\varepsilon} = g_\varepsilon(x - y),
\]
luego $\frac1{2\pi}\int\hat f(\xi)\eu^{-\varepsilon\xi^2}
\eu^{\iu x\xi}\dd\xi = \int f(y)g_\varepsilon(x - y)\dd y =
(f*g_\varepsilon)(x)$. Las $g_\varepsilon$ forman una aproximación de la
identidad: $g_\varepsilon \geq 0$, $\int g_\varepsilon = 1$ (integral
gaussiana), concentrándose en $0$; la demostración
del~\cref{thm:b3:lp:regularization}(2) se aplica literalmente (solo se
usaron $\int g_\varepsilon = 1$ y la concentración: para la cola,
$\int_{\abs y > \delta}g_\varepsilon \to
0$): $\norm{f * g_\varepsilon - f}_1 \to 0$.

(2) Si $\hat f \in L^1$: el miembro derecho de (1) converge, por
convergencia dominada (dominador $\abs{\hat f}$), a
$\frac1{2\pi}\int\hat
f(\xi)\eu^{\iu x\xi}\dd\xi$ para \emph{todo} $x$, y esa función límite
es \omterm{def:b3:topology:continuity}{continua} (de nuevo, convergencia
dominada). Por otra parte, $f
* g_\varepsilon \to f$ en $L^1$, de modo que, a lo largo de una
subsucesión, en casi todo punto (el~\cref{thm:b3:lp:complete}): los dos
límites coinciden en casi todo punto.

(3) $\hat f = 0$ anula el miembro derecho de (1): $f *
g_\varepsilon = 0$ para todo $\varepsilon$, y $f *
g_\varepsilon \to f$ en $L^1$: $f = 0$ en casi todo punto.
\end{proof}

\section{La clase de Schwartz}

\begin{definition}\label{def:b3:fouriertransform:schwartz}
La \emph{clase de Schwartz}\index{clase de Schwartz} $\mathcal
S(\R)$ está formada por las funciones $\mathcal C^\infty$ $f$ con
$\sup_x\abs{x^m f^{(n)}(x)} < \infty$ para todos $m, n \geq 0$ (todas
las derivadas decaen más deprisa que cualquier potencia). Ejemplos:
$\eu^{-ax^2}$, $\mathcal C_c^\infty$. Claramente $\mathcal S
\subseteq L^p$ para todo $p$ (acótese por $C(1 + x^2)^{-1}$), y
$\mathcal S$ es estable por derivación, por multiplicación por
polinomios y por productos.
\end{definition}

\begin{theorem}\label{thm:b3:fouriertransform:schwartz}
La transformada de Fourier aplica $\mathcal S(\R)$ biyectivamente sobre
sí mismo, con inversa $\check g(x) = \frac1{2\pi}\int
g(\xi)\eu^{\iu x\xi}\dd\xi$.
\end{theorem}

\begin{proof}
Sea $f \in \mathcal S$. Iterando
la~\cref{prop:b3:fouriertransform:basic}(3), $(\hat f)^{(n)} =
\widehat{(-\iu x)^nf}$ (cada $x^kf \in L^1$); iterando (4) con
$h = (-\iu x)^nf \in \mathcal S$ (todas cuyas derivadas son
\omterm{def:b3:lebesgue:l1}{integrables}), $(\iu\xi)^m\hat h =
\widehat{h^{(m)}}$. Combinando,
\[
\abs{\xi^m\,(\hat f)^{(n)}(\xi)}
= \bigl|\widehat{\,h^{(m)}}(\xi)\bigr|
\leq \bigl\|\bigl((-\iu x)^nf\bigr)^{(m)}\bigr\|_1 < \infty
\]
uniformemente en $\xi$: $\hat f \in \mathcal S$. Como $\hat f \in L^1$,
la inversión (el~\cref{thm:b3:fouriertransform:inversion}(2)) vale en
todo punto (ambos miembros son
\omterm{def:b3:topology:continuity}{continuos}): $\check{\hat f} = f$, y
simétricamente $\widehat{\check g} = g$ (la transformada con acento es
$g \mapsto \frac1{2\pi}\hat g(-\cdot)$, que también conserva
$\mathcal S$): biyección.
\end{proof}

\section{Plancherel y \texorpdfstring{$L^2$}{L2}}

\begin{theorem}[Plancherel]\label{thm:b3:fouriertransform:plancherel}
Para $f \in L^1 \cap L^2(\R)$:\index{teorema de Plancherel}
\[
\norm{\hat f}_2^2 = 2\pi\,\norm f_2^2 .
\]
En consecuencia, $\widehat{\phantom f}$ se extiende de manera única a
una aplicación lineal \omterm{def:b3:topology:continuity}{continua}
$\mathcal F \colon L^2(\R) \to L^2(\R)$ con
$\norm{\mathcal Ff}_2 = \sqrt{2\pi}\norm f_2$; $\mathcal
F$ es biyectiva, con $\mathcal F^{-1} =
\frac1{2\pi}\,\mathcal F\circ\sigma$ donde $\sigma f =
f(-\cdot)$, y conserva los \omterm{def:b3:hilbert:inner}{productos
escalares} salvo el factor
$2\pi$.
\end{theorem}

\begin{proof}
Sean $f \in L^1\cap L^2$ y $h = f * \tilde f$ con
$\tilde f(x) = \overline{f(-x)}$. Entonces $h \in L^1$
(el~\cref{thm:b3:product:convolution}), $h$ es
\omterm{def:b3:topology:continuity}{continua} y acotada
(el~\cref{exo:b3:lp:6}: $f, \tilde f \in L^2$), $h(0) =
\int f\bar f = \norm f_2^2$ y $\hat h = \hat
f\,\widehat{\tilde f} = \hat f\,\overline{\hat f} =
\abs{\hat f}^2 \geq 0$ (calcúlese $\widehat{\tilde f} =
\overline{\hat f}$). Aplíquese
el~\cref{thm:b3:fouriertransform:inversion}(1) a $h$ en $x = 0$:
\[
(h * g_\varepsilon)(0) = \frac1{2\pi}\int \hat
h(\xi)\,\eu^{-\varepsilon\xi^2}\dd\xi .
\]
Cuando $\varepsilon \to 0$: el miembro izquierdo tiende a $h(0)$ ($h$ es
\omterm{def:b3:topology:continuity}{continua} y acotada:
$(h*g_\varepsilon)(0) - h(0) = \int(h(-y)
- h(0))g_\varepsilon(y)\dd y \to 0$, separando $y$ pequeños y grandes);
y el derecho crece hacia $\frac1{2\pi}\int\hat h$ por convergencia
monótona ($\hat h \geq 0$). Por tanto, $\frac1{2\pi}\int\abs{\hat f}^2 =
\norm f_2^2$, finito o no \emph{a priori} --- y finito, lo que demuestra
a la vez la pertenencia y la identidad.

Extensión: $L^1\cap L^2 \supseteq \mathcal C_c$ es denso en $L^2$
(el~\cref{thm:b3:lp:density}); la transformada es allí
$\sqrt{2\pi}$-isométrica, de modo que se extiende de manera única a una
isometría salvo constante $\mathcal F$ sobre $L^2$
(el~\cref{thm:b3:complete:extension}). La inversión para $\mathcal S$
(el~\cref{thm:b3:fouriertransform:schwartz}) se traslada por la misma
\omterm{ex:b3:lebesgue:gamma}{densidad} (ambos miembros son
$L^2$-continuos): $\mathcal
F\bigl(\frac1{2\pi}\mathcal F(\sigma f)\bigr) = f$ sobre $\mathcal S$ y,
por tanto, sobre $L^2$: biyectividad.
\omterm{def:b3:hilbert:inner}{Productos escalares}: polarización a
partir de la identidad de normas.
\end{proof}

\begin{theorem}[Sumación de Poisson]
\label{thm:b3:fouriertransform:poisson}
Sea $f \in \mathcal S(\R)$ (basta con $f$
\omterm{def:b3:topology:continuity}{continua} con $\abs{f} +
\abs{\hat f} \leq C(1 + \abs\cdot)^{-2}$). Entonces\index{fórmula de
sumación de Poisson}
\[
\sum_{n\in\Z} f(n) \;=\; \sum_{k\in\Z}\hat f(2\pi k) .
\]
\end{theorem}

\begin{proof}
Sea $F(x) = \sum_{n\in\Z}f(x + n)$: la serie converge normalmente sobre
los \omterm{def:b3:topology:compact}{compactos} (decaimiento de $f$), de
modo que $F$ es \omterm{def:b3:topology:continuity}{continua} y
$1$-periódica. Sus coeficientes de Fourier (período $1$:
$c_k(F) = \int_0^1F(t)\eu^{-2\iu\pi kt}\dd t$):
\[
c_k(F) = \sum_n\int_0^1 f(t + n)\,\eu^{-2\iu\pi kt}\dd t
= \int_\R f(t)\,\eu^{-2\iu\pi kt}\dd t = \hat f(2\pi k)
\]
(la convergencia normal justifica el intercambio; la fase es
$1$-periódica). La serie $\sum_k\abs{c_k(F)}$ converge (decaimiento de
$\hat f$), luego la serie de Fourier de $F$ converge normalmente; su
suma es una función \omterm{def:b3:topology:continuity}{continua} con
los mismos coeficientes de Fourier que $F$ y, por tanto, es igual a $F$
(inyectividad en el círculo: la diferencia tiene coeficientes nulos, y
el~\cref{thm:b3:hilbert:fourier} da cero en $L^2$ y, por
\omterm{def:b3:topology:continuity}{continuidad}, en todas partes).
Evalúese en $x = 0$.
\end{proof}

\begin{example}[La identidad theta]
\label{ex:b3:fouriertransform:theta}
Aplicando Poisson a $f(x) = \eu^{-\pi tx^2}$ ($t > 0$), cuya
transformada es $\hat f(\xi) = t^{-1/2}\eu^{-\xi^2/4\pi t}$
(el~\cref{ex:b3:fouriertransform:gaussian} con $a = \pi t$):
\[
\sum_{n\in\Z}\eu^{-\pi n^2t}
= \frac1{\sqrt t}\sum_{k\in\Z}\eu^{-\pi k^2/t} :
\]
la ecuación funcional de la función theta de Jacobi, clave para la
ecuación funcional de la $\zeta$ de Riemann --- y un acelerador numérico
espectacular: para $t$ pequeño, el miembro izquierdo converge con
lentitud y el derecho a toda velocidad.
\end{example}

\begin{method}\label{met:b3:fouriertransform:method}
Rangos de trabajo: $L^1$ --- transformada definida puntualmente, la
inversión necesita $\hat f \in L^1$; $\mathcal S$ --- todo es lícito,
demuéstrese aquí primero; $L^2$ --- transformada definida por
\omterm{ex:b3:lebesgue:gamma}{densidad} (¡no por la integral!), simetría
\omterm{prop:b3:galois:perfect}{perfecta}, contabilidad de
\omterm{thm:b3:hilbert:parseval}{Parseval}. Para calcular una
transformada: redúzcase a la tabla (indicador, exponencial, gaussiana)
mediante las reglas de la~\cref{prop:b3:fouriertransform:basic}; para
demostrar una identidad: establézcase en $\mathcal S$ (o en
$\mathcal C_c^\infty$) y extiéndase por
\omterm{ex:b3:lebesgue:gamma}{densidad} y
\omterm{def:b3:topology:continuity}{continuidad}
(el~\cref{met:b3:lp:toolkit}); para resolver una ecuación diferencial
lineal, ordinaria o en derivadas parciales, con coeficientes constantes:
transfórmese, divídase, inviértase.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:fouriertransform:1}
Calcular las transformadas de Fourier de: $\mathbf 1_{\intcc{-a}a}$;
$\eu^{-a\abs x}$ ($a > 0$); la función triangular $\max(0, 1 -
\abs x)$; $\frac1{x^2 + a^2}$ \emph{(úsese la inversión en la segunda)}.
Regístrese la tabla que va surgiendo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:fouriertransform:2}
Sea $f \in L^1$. Exprésense en términos de $\hat f$ las transformadas
de: $f(x - a)$, $f(x)\cos(bx)$, $f(ax + b)$, $\overline{f(-x)}$,
$(f * f)(x)$. Verifíquese cada regla sobre la gaussiana.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:3}
(a) Demostrar que $\mathbf 1_{\intcc{-1}1} * \mathbf
1_{\intcc{-1}1}$ tiene transformada $\bigl(\frac{2\sin\xi}\xi
\bigr)^2$ y dedúzcase
$\int_\R\bigl(\frac{\sin\xi}\xi\bigr)^2\dd\xi = \pi$ por Plancherel ---
o por inversión en $0$. Compárese
\cref{pb:b3:lebesgue:1}.
(b) Calcular $\int_\R\frac{\dd x}{(x^2+1)^2}$ mediante Plancherel
aplicado a $\eu^{-\abs x}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:4}
(Álgebra del \omterm{pb:b3:fouriertransform:1}{núcleo del calor}) Con
$g_t(x) =
\frac1{2\sqrt{\pi t}}\eu^{-x^2/4t}$: (a) compruébese
$\hat g_t(\xi) = \eu^{-t\xi^2}$; (b) dedúzcase la ley de semigrupo
$g_t * g_s = g_{t+s}$ sin ningún cálculo de integrales; (c) demuéstrese
$\norm{g_t}_1 = 1$ y $\norm{g_t}_2^2 =
(8\pi t)^{-1/2}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:5}
Demostrar que si $f \in L^1$ es par y real, $\hat f$ es par y real; y
que si $f$ es impar y real, $\hat f$ es impar y puramente imaginaria.
¿Qué calcula $\hat f(0)$? Dedúzcase que $f \geq
0$ obliga a $\norm{\hat f}_\infty = \hat f(0) = \int f$, e interprétese
para \omterm{ex:b3:lebesgue:gamma}{densidades} de probabilidad (el~\cref{ch:b3:clt}: una función
característica tiene módulo $\leq 1$, alcanzado en
$0$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:fouriertransform:6}
(No sobreyectividad) Demostrar que $\widehat{\phantom f}\colon L^1
\to \mathcal C_0$ es inyectiva y
\omterm{def:b3:topology:continuity}{continua} pero \emph{no}
sobreyectiva, en tres pasos. (i) Inyectividad
(el~\cref{thm:b3:fouriertransform:inversion}) y
\omterm{def:b3:topology:continuity}{continuidad}
($\norm{\hat f}_\infty \leq \norm f_1$), y $\mathcal C_0$ es un espacio
de Banach (cerrado en
$\norm\cdot_\infty$).
(ii) Si la aplicación fuera sobreyectiva, sería biyectiva, y el teorema
de la aplicación abierta (el~\cref{thm:b3:banach:openmapping}) daría una
constante $C$ con $\norm f_1 \leq C\norm{\hat
f}_\infty$ para toda $f \in L^1$. (iii) Contradígase esto con
$f_n(x) = \frac{\sin
x}{x}\cdot\frac{\sin(x/n)}{x/n}$: su transformada es (salvo constantes)
el trapecio de tipo convolución $\mathbf 1_{\intcc{-1}1} *
\mathbf 1_{\intcc{-1/n}{1/n}}$ --- demuéstrese que
$\norm{\hat f_n}_\infty \leq \pi$ uniformemente, mientras que
$\norm{f_n}_1 \geq c\ln n$ contando los arcos de $\frac{\abs{\sin x}}x$
en $[1, n]$ (donde el segundo factor está acotado inferiormente), como
en el~\cref{thm:b3:banach:fourierdiverge}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:7}
(Diccionario regularidad $\leftrightarrow$ decaimiento) Demostrar:
$f \in
L^1$ con $\hat f(\xi) = O(\abs\xi^{-k-1-\delta})$ para cierto
$\delta > 0$ implica que $f$ tiene un representante $\mathcal C^k$.
Recíprocamente, $f \in \mathcal C^k_c$ implica $\hat f(\xi) =
O(\abs\xi^{-k})$. Ilústrense ambos sentidos sobre la función triangular.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:fouriertransform:8}
(Desigualdad de Heisenberg) Para $f \in \mathcal S(\R)$ real con
$\norm f_2 = 1$, demostrar
\[
\Bigl(\int x^2f(x)^2\dd x\Bigr)\cdot
\Bigl(\frac1{2\pi}\int \xi^2\abs{\hat f(\xi)}^2\dd\xi\Bigr)
\;\geq\; \frac14 ,
\]
con igualdad para las gaussianas. \emph{(Escríbase $1 = \int f^2 =
-\int x\,(f^2)'$ por partes, acótese por Cauchy--Schwarz y conviértase
$\norm{f'}_2$ mediante Plancherel.)} Interpretación: una señal y su
espectro no pueden estar ambos concentrados.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:9}
Justifíquese en detalle el~\cref{ex:b3:fouriertransform:theta}
(hipótesis de Poisson para la gaussiana) y úsese la identidad para
evaluar $\sum_{n\in\Z}\eu^{-\pi n^2}$ con seis decimales usando tres
términos. ¿Cuántos términos de la serie \emph{definitoria} exigiría la
misma precisión en $t = 10^{-2}$, frente a la serie transformada?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:10}
(Funciones de banda limitada) Sea $f \in L^2(\R)$ con $\mathcal Ff$ de
soporte en $\intcc{-\pi}\pi$. Demostrar que $f$ tiene un representante
cada uno de cuyos valores puede recuperarse de las muestras: demuéstrese
la \emph{interpolación de Shannon} en los enteros,
\[
f(x) = \sum_{n\in\Z} f(n)\,
\frac{\sin\bigl(\pi(x - n)\bigr)}{\pi(x - n)}
\quad\text{en } L^2,
\]
desarrollando $\mathcal Ff$ en la base de Fourier de
$L^2(\intcc{-\pi}\pi)$ (el~\cref{thm:b3:hilbert:fourier}) y
transformando de vuelta término a término.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:11}
(La transformada como operador de orden cuatro) Sobre $\mathcal
S(\R)$, sea $\mathcal F f = \hat f$. (a) Usando la fórmula de inversión,
demuéstrese $(\mathcal F^2f)(x) =
2\pi\,f(-x)$ y dedúzcase $\mathcal F^4 = (2\pi)^2\,
\mathrm{id}$. (b) Dedúzcase que todo valor propio de $\mathcal F$ sobre
$\mathcal S$ pertenece a $\{\pm\sqrt{2\pi},
\pm\iu\sqrt{2\pi}\}$, y exhíbase una función propia para $+\sqrt{2\pi}$
\emph{(¿qué función de este capítulo es proporcional a su propia
transformada?)}. (c) Demostrar que las funciones pares cumplen
$\mathcal F^2f = 2\pi
f$ y las impares $\mathcal F^2f = -2\pi f$; prodúzcase una función
propia para el valor propio $-\iu\sqrt{2\pi}$ a partir de
$x\eu^{-x^2/2}$ calculando su transformada (derívese la transformada de
la gaussiana).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:12}
(Autocorrelación y lema de Wiener) Para $f \in L^2(\R)$ defínanse
$\tilde f(x) = \overline{f(-x)}$ y la \emph{autocorrelación}
$A_f = f * \tilde f$. (a) Demostrar que $A_f$ es una función acotada y
\omterm{def:b3:topology:continuity}{continua} con
$A_f(0) = \norm f_2^2 \geq \abs{A_f(x)}$ para todo $x$
(el~\cref{exo:b3:lp:6} y Cauchy--Schwarz). (b) Demostrar, primero para
$f \in L^1\cap L^2$, que $\widehat{A_f} = \abs{\hat f\,}^2 \geq 0$: la
autocorrelación tiene transformada no negativa --- los espectros de las
autocorrelaciones son espectros de potencia. (c) Dedúzcase la identidad
$\int_\R\abs{\hat
f(\xi)}^2\eu^{\iu x\xi}\,\dd\xi = 2\pi A_f(x)$ (inversión; justifíquese
su aplicabilidad cuando $\hat f \in L^2$ tiene $\abs{\hat f}^2 \in L^1$)
y evalúese para $f = \mathbf
1_{\intcc{-1/2}{1/2}}$ en $x = 0$: recupérese
$\int_\R\bigl(\frac{\sin u}u\bigr)^2\dd u = \pi$.
\end{exercise}

\section{Problema: la ecuación del calor en la recta}

\begin{problem}[{Problema de fin de semana --- $\partial_tu =
\partial^2_{xx}u$, resuelta de principio a fin}]
\label{pb:b3:fouriertransform:1}
El calor se propaga; la ecuación $\partial_tu = \partial_{xx}^2u$ dice
que su \omterm{ex:b3:lebesgue:gamma}{densidad} difunde a un ritmo dado
por la curvatura local del perfil de temperatura. Resolveremos el
problema de Cauchy en $\R$ --- dada $f$, hállese $u(t, x)$ para $t > 0$
con $u(0, \cdot) = f$ ---, demostraremos las notables propiedades de la
solución y veremos por qué el tiempo no puede invertirse. En todo el
problema, $g_t(x) = \frac1{2\sqrt{\pi t}}\eu^{-x^2/4t}$ es el
\emph{núcleo del calor}\index{núcleo del calor} y $u(t, \cdot) = g_t *
f$.

\medskip
\noindent\textbf{Parte I --- Deducción del núcleo.} Trabájese primero de
manera formal: supóngase que $u(t, \cdot) \in \mathcal S$ resuelve la
ecuación y sea $\hat u(t, \xi)$ la transformada en
$x$.
\begin{enumerate}
  \item Demostrar (formalmente) que $\partial_t\hat u = -\xi^2\hat u$,
  de donde $\hat u(t,\xi) = \eu^{-t\xi^2}\hat f(\xi)$, y reconózcase
  $u(t) = g_t * f$ (el~\cref{exo:b3:fouriertransform:4}). Esto motiva la
  \emph{definición} de $u$; todo lo demás se demuestra ya directamente,
  para $f \in \mathcal C_b(\R)$
  (\omterm{def:b3:topology:continuity}{continua} acotada) o $f \in L^p$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Verificación.}
\begin{enumerate}[resume]
  \item Demostrar que, para $t > 0$, $u(t, x) = \int
        g_t(x-y)f(y)\dd y$ está bien definida para $f \in \mathcal
        C_b$ y que $u$ es $\mathcal C^\infty$ en $(t, x)$ sobre
  $\intoo0\infty\times\R$ \emph{(derívese bajo el signo integral;
  domínense las derivadas gaussianas localmente uniformemente en
  $(t,x)$)}. \item Verificar $\partial_tg_t = \partial^2_{xx}g_t$ por
  cálculo directo y dedúzcase $\partial_tu =
        \partial^2_{xx}u$ para $t > 0$. \item (Condición inicial)
  Demostrar que, para $f \in \mathcal
        C_b$, $u(t, x) \to f(x)$ cuando $t \to 0^+$, uniformemente sobre
  los \omterm{def:b3:topology:compact}{compactos} \emph{(aproximación de
  la identidad: sepárese $\abs y
        \leq \delta$, $\abs y > \delta$)}; y para $f \in L^p$
  ($p < \infty$), demuéstrese $\norm{u(t) - f}_p \to 0$. \item
  (Suavizado instantáneo) Concluir: incluso para $f$ meramente
  \omterm{def:b3:topology:continuity}{continua} y acotada, la solución
  es $\mathcal C^\infty$ para todo $t > 0$ --- el calor borra al
  instante la aspereza. Calcúlese $u(t, \cdot)$ explícitamente para $f =
        \mathbf 1_{\intoo0\infty}$ (una función error) y esbócese su
  perfil para tres valores de $t$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Propiedades estructurales.}
\begin{enumerate}[resume]
  \item (Positividad y comparación) Si $f \geq 0$, entonces $u > 0$ para
  todo $t > 0$ (estrictamente, salvo que $f = 0$ en casi todo punto); si
  $f_1 \leq f_2$, entonces $u_1 \leq u_2$. Un punto frío se calienta al
  instante: coméntese. \item (Conservación) Para $f \in L^1$:
  $\int u(t, x)\dd x =
        \int f$ para todo $t$ \emph{(Tonelli)} --- el calor total se
  conserva. \item (Disipación) Para $f \in L^1\cap L^2$, demuéstrese
  mediante Plancherel que $t \mapsto \norm{u(t)}_2$ es no creciente,
  estrictamente salvo que $f = 0$, y calcúlese su límite cuando
  $t \to \infty$. Demuéstrese además
  $\norm{u(t)}_\infty \leq \frac{\norm
        f_1}{2\sqrt{\pi t}} \to 0$: el calor se esparce y muere. \item
  (Unicidad, clase $L^2$) Sea $u$ una solución con $u(t) \in L^2$ para
  todo $t$, $u \in \mathcal
        C^1(\intoo0\infty, L^2)$ en el sentido natural y $u(t) \to f$ en
  $L^2$ cuando $t\to0$; admitiendo que la transformada la convierte en
  $\partial_t\hat u =
        -\xi^2\hat u$ puntualmente en casi todo punto de $\xi$ para casi
  todo $t$ \emph{(justificado evaluando contra $\mathcal
        C_c^\infty$ en $\xi$ --- esbócese)}, demuéstrese
  $\hat u(t,\xi) = \eu^{-t\xi^2}\hat f(\xi)$ y, por tanto, la unicidad
  en esta clase.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La flecha del tiempo.}
\begin{enumerate}[resume]
  \item Demostrar que el problema \emph{retrógrado} está mal planteado:
  para que la solución exista en el instante $-s$ ($s > 0$) con dato $f$
  en el instante $0$ --- es decir, para que $f = g_s * h$ tenga solución
  $h \in L^2$ --- es necesario que $\eu^{s\xi^2}\hat f(\xi) \in L^2$:
  una condición de decaimiento extrema sobre $\hat f$. Exhíbase una $f
        \in L^2$ regular explícita para la que no exista solución
  retrógrada en ningún intervalo de tiempo: tómese la función con
  $\hat f(\xi) =
        \eu^{-\abs\xi}$ --- identifíquese $f$
  (el~\cref{exo:b3:fouriertransform:1}) y demuéstrese
  $\eu^{s\xi^2}\eu^{-\abs\xi} \notin L^2$ para todo $s >
        0$. \item (Suavizado frente a información) Explíquese en un
  párrafo breve, usando las preguntas 5, 9 y 10, por qué el semigrupo
  del calor $(f \mapsto g_t * f)_{t\geq0}$ es inyectivo pero no
  sobreyectivo sobre $L^2$, y por qué esto expresa la irreversibilidad
  de la difusión.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- El teorema del muestreo de Shannon.} Una
función $f \in L^2(\R)$ es \emph{de banda limitada} a $\Omega$ si
$\hat f = 0$ en casi todo punto fuera de $\intcc{-\Omega}\Omega$;
escribimos $PW_\Omega$ (espacio de Paley--Wiener) para tales funciones.
\begin{enumerate}[resume]
  \item Demostrar que toda $f \in PW_\Omega$ coincide en casi todo punto
  con la función $\mathcal C^\infty$
  $\frac1{2\pi}\int_{-\Omega}^{\Omega}\hat
        f(\xi)\eu^{\iu x\xi}\,\dd\xi$ (justifíquense la regularidad y la
  identificación), con todas las derivadas acotadas: la limitación de
  banda es una forma extrema de regularidad. A partir de ahora, $f$
  denota ese representante. \item Desarróllese
  $\hat f \in L^2(\intcc{-\Omega}\Omega)$ en la base de Fourier de ese
  intervalo e identifíquense los coeficientes como \emph{muestras} de
  $f$:
        \[
        \hat f(\xi) = \frac\pi\Omega\sum_{n\in\Z}
        f\Bigl(\frac{n\pi}\Omega\Bigr)\,
        \eu^{-\iu n\pi\xi/\Omega}
        \quad\text{en } L^2(\intcc{-\Omega}\Omega) .
        \]
  \item Dedúzcase el \emph{teorema del muestreo}: para $f \in
        PW_\Omega$,
        \[
        f(x) = \sum_{n\in\Z}f\Bigl(\frac{n\pi}
        \Omega\Bigr)\,\operatorname{sinc}(\Omega x - n\pi),
        \qquad \operatorname{sinc}t = \frac{\sin t}t,
        \]
        con convergencia en $L^2(\R)$ y uniforme en $\R$
        \emph{(insértese la serie de la pregunta 13 en la fórmula de
        inversión y calcúlese la integral elemental)}: una señal de
        banda limitada queda enteramente determinada por sus valores
        sobre una malla de paso $\pi/\Omega$ --- la frecuencia de
        Nyquist. \item Demostrar que las funciones $x \mapsto
        \operatorname{sinc}(\Omega x - n\pi)$, $n \in \Z$, forman una
        familia ortogonal en $L^2(\R)$ de norma constante
        $\sqrt{\pi/\Omega}$, y dedúzcase la identidad de energía
        $\norm f_2^2 =
        \frac\pi\Omega\sum_n\abs{f(n\pi/\Omega)}^2$. \item
        (Solapamiento) Exhíbase una $g \in PW_{2\Omega}$ no nula que se
        anule en todos los puntos de muestreo $\frac{n\pi}\Omega$
        \emph{(considérese $g(x) =
        \sin(\Omega x)\operatorname{sinc}(\Omega x)$ y compruébese su
        banda)}: muestrear por debajo de la frecuencia de Nyquist pierde
        información --- dos señales distintas pueden compartir todas sus
        muestras: el efecto estroboscópico de la rueda de carro,
        matematizado. \item (Grados de libertad) Usando las preguntas
        14--15, justifíquese la regla de ingeniería: una señal de banda
        limitada a $\Omega$ cuya energía está esencialmente concentrada
        en una ventana temporal de longitud $T$ queda descrita por
        aproximadamente $\frac{\Omega T}\pi$ muestras reales ---
        precísese «esencialmente» mediante la identidad de energía y la
        cola $\sum_{\abs{n\pi/\Omega} >
        T/2}$. \item (Comprobaciones de coherencia) Verifíquese el
        teorema del muestreo a mano sobre dos elementos de $PW_\Omega$:
        (a) $f =
        \operatorname{sinc}(\Omega\,\cdot)$, cuyas muestras son
        $\delta_{n0}$; (b) señales de banda estrecha de tipo
        $f(x) = \cos(\omega x)
        \operatorname{sinc}(\varepsilon x)$ --- con más precisión,
        demuéstrese que para $f \in
        PW_{\Omega'}$ con $\Omega' < \Omega$ la serie de frecuencia
        $\Omega$ también reconstruye $f$ (sobremuestrear es inocuo),
        sumergiendo $PW_
        {\Omega'} \subseteq PW_\Omega$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Incertidumbre, dos veces más.} La
desigualdad de Heisenberg (el~\cref{exo:b3:fouriertransform:8}) acota
\emph{cuánto} pueden concentrarse conjuntamente $f$ y $\hat f$; aquí van
su hermana de todo o nada y su saturación exacta.
\begin{enumerate}[resume]
  \item Sea $f \in L^1$ con $\operatorname{supp}f \subseteq
        \intcc{-A}A$. Demostrar que $\hat f$ es la suma de una serie de
  potencias convergente en todo punto:
        \[
        \hat f(\xi) = \sum_{k\geq0}\frac{(-\iu\xi)^k}{k!}
        \,m_k, \qquad m_k = \int_{-A}^{A}x^kf(x)\,\dd x,
        \quad \abs{m_k} \leq A^k\norm f_1
        \]
        \emph{(desarróllese $\eu^{-\iu\xi x}$ y justifíquese el
        intercambio por convergencia normal)}: la transformada es
        \emph{analítica real}, con radio de convergencia infinito en
        todo punto. \item Dedúzcase la dicotomía de soportes: una
        función analítica real que se anula en un intervalo abierto no
        vacío se anula idénticamente \emph{(el conjunto donde se anulan
        todas las derivadas es abierto y cerrado --- desarróllese el
        argumento de Taylor)}; concluir que ninguna $f$ no nula tiene a
        la vez $f$ y $\hat f$ de soporte \omterm{def:b3:topology:compact}{compacto}, y que $PW_\Omega$ no
        contiene ninguna función no nula de soporte \omterm{def:b3:topology:compact}{compacto} --- las
        señales de banda limitada duran para siempre, y las de duración
        limitada se filtran a todas las frecuencias. \item (Saturación
        de Heisenberg) Sobre la familia gaussiana $f
        = \eu^{-ax^2}$, calcúlense ambos factores de concentración y
        compruébese que el producto normalizado
        $\bigl(\int x^2\abs f^2\bigr)\bigl(\frac1{2\pi}\int
        \xi^2\abs{\hat f}^2\bigr)\big/\norm f_2^4$ vale $\frac14$ para
        \emph{todo} $a$ --- la familia de igualdad
        del~\cref{exo:b3:fouriertransform:8} en carne y hueso;
        explíquese mediante un argumento de escala por qué el producto
        ha de ser constante a lo largo de la familia. \item Explíquese
        la lectura física (\omterm{ex:b3:lebesgue:gamma}{densidades} de posición y momento de un estado
        cuántico; $\hbar$ en la normalización da $\sigma_x\sigma_p \geq
        \frac\hbar2$) y conéctese todo el capítulo: el suavizado
        instantáneo (Parte II), la irreversibilidad (Parte IV), el
        muestreo (Parte V), Heisenberg y la dicotomía de soportes son
        cinco expresiones de una misma ley --- el comportamiento de
        $\hat f$ en el infinito legisla lo que $f$ puede hacer en
        cualquier sitio.
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- El álgebra del núcleo y un ejemplo
\omterm{def:b3:groups:derived}{resoluble}.}
\begin{enumerate}[resume]
  \item (Semigrupo) Demostrar la \emph{identidad de Chapman--Kolmogorov}
  $g_t * g_s = g_{t+s}$ para $t, s > 0$ (mediante el teorema de la
  convolución y la inyectividad de la transformada en $L^1$) y dedúzcase
  $u(t + s) = g_s *
        u(t)$: evolucionar durante un tiempo $t + s$ es evolucionar
  durante $t$ y después durante $s$. Afínese la disipación de la
  pregunta 8: escribiendo $\norm{u(t)}_2^2 = \frac1{2\pi}\int
        \eu^{-2t\xi^2}\abs{\hat f(\xi)}^2\dd\xi$, demuéstrese por
  Cauchy--Schwarz que
        \[
        t \longmapsto \ln\,\norm{u(t)}_2
        \quad\text{es convexa en } \intoo0{+\infty} :
        \]
        la energía $L^2$ no solo decrece: decrece de manera
        logarítmicamente convexa. \item (A dónde va el calor) Sean
        $f \geq 0$, $f \in L^1$, con $\int x^2f(x)\dd x < \infty$.
        Demostrar que, para todo
        $t > 0$
        \[
        \int_\R x\,u(t, x)\,\dd x = \int_\R x f(x)\,\dd x,
        \qquad
        \int_\R x^2u(t, x)\,\dd x
        = \int_\R x^2f(x)\,\dd x + 2t\int_\R f :
        \]
        el \omterm{ex:b3:groups:actions}{centro} de calor nunca se mueve
        y la varianza crece \emph{linealmente} con el tiempo --- el
        escalado difusivo $x \sim \sqrt{2t}$, que habrá que releer
        cuando aparezca el movimiento browniano en
        el~\cref{ch:b3:probability}. \emph{(Calcúlense los dos primeros
        momentos de $g_t$ y úsese Tonelli sobre la convolución.)} \item
        (La gaussiana, resuelta de principio a fin) Para $f(x) =
        \eu^{-x^2}$, establézcase la forma cerrada
        \[
        u(t, x) = \frac1{\sqrt{1 + 4t}}\,
        \exp\Bigl(-\frac{x^2}{1 + 4t}\Bigr),
        \]
        y verifíquense sobre ella, a mano: la ecuación $\partial_tu
        = \partial^2_{xx}u$; la conservación $\int u(t) =
        \sqrt\pi$; la ley de disipación $\norm{u(t)}_2 =
        (\pi/2)^{1/4}(1 + 4t)^{-1/4}$ (compárese su decaimiento
        $t^{-1/4}$ con el decaimiento $t^{-1/2}$ de la norma del supremo
        de la pregunta 8); y el crecimiento exacto de la varianza de la
        pregunta 24. En $t = 6$: el pico ha caído al $\frac15$ de su
        altura inicial mientras el perfil es cinco veces más ancho ---
        el mismo calor, esparcido.
\end{enumerate}
\end{problem}
