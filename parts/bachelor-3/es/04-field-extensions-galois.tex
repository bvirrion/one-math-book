\chapter{Extensiones de cuerpos y teoría de Galois}\label{ch:b3:galois}

¿Puede resolverse toda ecuación por
\omterm{def:b3:galois:radical}{radicales}, como sugieren la fórmula de
la ecuación de segundo \omterm{def:b3:galois:extension}{grado} y las fórmulas de Cardano para la cúbica?
¿Puede trisecarse un ángulo con regla y compás? Ambas preguntas,
abiertas durante siglos, quedan respondidas --- negativamente --- por
una sola idea de Évariste Galois: asociar
a cada polinomio un \emph{grupo} finito de simetrías de sus raíces y
leer la respuesta en el grupo. Este capítulo construye el diccionario:
extensiones de cuerpos y
\omterm{def:b3:galois:extension}{grados},
\omterm{thm:b3:galois:splitting}{cuerpos de descomposición} y
\omterm{def:b3:galois:closure}{clausuras algebraicas},
\omterm{thm:b3:galois:finitefields}{cuerpos finitos} (una teoría
completa --- y la prometida ciclicidad de $\mathbb F_q^\times$),
separabilidad y, por fin, la
\omterm{thm:b3:galois:fundamental}{correspondencia de Galois} misma, con
demostraciones completas. La cosecha: la imposibilidad de las
construcciones clásicas, la estructura de los cuerpos ciclotómicos y la
imposibilidad de resolver por \omterm{def:b3:galois:radical}{radicales}
la ecuación de quinto \omterm{def:b3:galois:extension}{grado} --- la simplicidad de $A_5$
del~\cref{ch:b3:groups} dando en el blanco.

\section{Extensiones, grado, algebraicidad}

\begin{definition}\label{def:b3:galois:extension}
Una \emph{extensión de cuerpos}\index{extensión de cuerpos} $L/K$ es un
cuerpo $L$ que contiene $K$ como subcuerpo; entonces $L$ es un
$K$-espacio vectorial, y el \emph{grado}\index{grado de una extensión}
$[L:K]$ es su dimensión. La extensión es \emph{finita} si $[L:K] <
\infty$. La \emph{característica} de un cuerpo es el generador $\geq 0$
del núcleo de $\Z \to K$, $n \mapsto n\cdot 1$: es $0$ o un primo $p$;
en consecuencia, $K$ contiene un menor subcuerpo (el \emph{cuerpo
primo}) isomorfo a $\Q$ o a $\mathbb
F_p = \Z/p\Z$.
\end{definition}

\begin{theorem}[Fórmula de los grados]\label{thm:b3:galois:tower}
Si $K \subseteq L \subseteq M$, entonces $[M:K] = [M:L]\,[L:K]$: si
$(e_i)$ es una base de $L$ sobre $K$ y $(f_j)$ una base de $M$ sobre
$L$, entonces $(e_if_j)$ es una base de $M$ sobre $K$.\index{fórmula de
los grados}
\end{theorem}

\begin{proof}
Genera: $x \in M$ se escribe $x = \sum_j \lambda_jf_j$
($\lambda_j \in L$), y cada $\lambda_j = \sum_i \mu_{ij}e_i$
($\mu_{ij} \in K$): $x = \sum_{i,j}\mu_{ij}e_if_j$. Es independiente:
$\sum_{i,j}\mu_{ij}e_if_j = 0$ se reescribe $\sum_j (\sum_i
\mu_{ij}e_i)f_j = 0$; las sumas interiores están en $L$, luego se anulan
(los $f_j$ son independientes sobre $L$); entonces todos los
$\mu_{ij} = 0$ (los $e_i$ son independientes sobre $K$).
\end{proof}

\begin{definition}\label{def:b3:galois:algebraic}
Sean $L/K$ y $\alpha \in L$. Si algún $P \in K[X]$ no nulo cumple
$P(\alpha) = 0$, se dice que $\alpha$ es
\emph{algebraico}\index{elemento algebraico} sobre $K$; el generador
mónico $\pi_\alpha$ del \omterm{def:b3:rings:ideal}{ideal}
$\{P : P(\alpha) = 0\}$ de $K[X]$ es su \emph{polinomio
mínimo}\index{polinomio mínimo (teoría de cuerpos)}, un polinomio
\omterm{def:b3:rings:divisibility}{irreducible} ($\pi = QR$ con
$Q(\alpha) = 0$ obliga a que $R$ sea constante, por la minimalidad del
\omterm{def:b3:galois:extension}{grado}). En caso contrario, $\alpha$ es
\emph{trascendente}. Escribimos $K(\alpha)$ para el menor subcuerpo de
$L$ que contiene $K$ y $\alpha$, y $K[\alpha]$ para el menor subanillo.
\end{definition}

\begin{theorem}\label{thm:b3:galois:simple}
Si $\alpha$ es \omterm{def:b3:galois:algebraic}{algebraico} sobre $K$
con $d = \deg\pi_\alpha$, entonces
\[
K(\alpha) = K[\alpha] \cong K[X]/(\pi_\alpha),
\qquad [K(\alpha) : K] = d,
\]
con base $1, \alpha, \dots, \alpha^{d-1}$. Recíprocamente, si
$[L:K] < \infty$, todo $\alpha \in L$ es
\omterm{def:b3:galois:algebraic}{algebraico} de
\omterm{def:b3:galois:extension}{grado} $\deg \pi_\alpha$ divisor de
$[L:K]$.
\end{theorem}

\begin{proof}
La evaluación $K[X] \to L$, $P \mapsto P(\alpha)$, tiene imagen
$K[\alpha]$ y núcleo $(\pi_\alpha)$; como $\pi_\alpha$ es
\omterm{def:b3:rings:divisibility}{irreducible}, $K[X]/(\pi_\alpha)$ es
un \emph{cuerpo} (la~\cref{prop:b3:rings:primemaximal}: en el
\omterm{def:b3:rings:pidufd}{DIP} $K[X]$, un
\omterm{def:b3:rings:divisibility}{irreducible} genera un
\omterm{def:b3:rings:primemaximal}{ideal maximal}), luego $K[\alpha]$ es
un cuerpo que contiene $K$ y $\alpha$: coincide con $K(\alpha)$. Las
clases de $1, X, \dots, X^{d-1}$ forman una base del cociente (división
euclídea), de donde la base y el
\omterm{def:b3:galois:extension}{grado}. Recíprocamente, si $[L:K]
= n < \infty$: los $1, \alpha, \dots, \alpha^n$ son dependientes, lo que
da un polinomio anulador; y entonces $[K(\alpha):K] =
\deg\pi_\alpha$ divide a $n$ por la \omterm{thm:b3:galois:tower}{fórmula
de los grados}.
\end{proof}

\begin{corollary}\label{cor:b3:galois:algclosed}
Si $\alpha, \beta$ son \omterm{def:b3:galois:algebraic}{algebraicos}
sobre $K$, también lo son $\alpha \pm
\beta$, $\alpha\beta$, $\alpha/\beta$ ($\beta \ne 0$): los elementos de
$L$ \omterm{def:b3:galois:algebraic}{algebraicos} sobre $K$ forman un
subcuerpo de $L$. Además, la algebraicidad es transitiva:
\omterm{def:b3:galois:algebraic}{algebraico} sobre
\omterm{def:b3:galois:algebraic}{algebraico} es
\omterm{def:b3:galois:algebraic}{algebraico}.
\end{corollary}

\begin{proof}
$K(\alpha, \beta) = (K(\alpha))(\beta)$ es finita sobre $K(\alpha)$
($\beta$ es \omterm{def:b3:galois:algebraic}{algebraico} sobre
$K \subseteq K(\alpha)$) y $K(\alpha)/K$ es finita: por la
\omterm{thm:b3:galois:tower}{fórmula de los grados},
$[K(\alpha,\beta):K] <
\infty$, y todo elemento de $K(\alpha, \beta)$ --- incluidos los cuatro
citados --- es \omterm{def:b3:galois:algebraic}{algebraico}
(el~\cref{thm:b3:galois:simple}). Transitividad: si $\beta$ es
\omterm{def:b3:galois:algebraic}{algebraico} sobre $L$ y $L/K$ es
\omterm{def:b3:galois:algebraic}{algebraica}, los coeficientes
$c_0, \dots, c_{m-1}$ de $\pi_{\beta/L}$ generan una extensión finita
$F = K(c_0, \dots,
c_{m-1})$ de $K$ (\omterm{thm:b3:galois:tower}{fórmula de los grados}
reiterada), y $F(\beta)/F$ es finita: $[F(\beta):K] < \infty$, luego
$\beta$ es \omterm{def:b3:galois:algebraic}{algebraico} sobre
$K$.
\end{proof}

\begin{example}\label{ex:b3:galois:degrees}
$[\Q(\sqrt2):\Q] = 2$, $[\Q(\sqrt[3]2):\Q] = 3$ ($X^3 - 2$ es
\omterm{def:b3:rings:divisibility}{irreducible}:
\omterm{thm:b3:rings:criteria}{Eisenstein}), $[\Q(\zeta_p):\Q] = p - 1$
para $\zeta_p = \eu^{2\iu\pi/p}$ ($\Phi_p$ es
\omterm{def:b3:rings:divisibility}{irreducible},
el~\cref{ex:b3:rings:eisenstein}). La
\omterm{thm:b3:galois:tower}{fórmula de los grados} es ya un arma:
$\sqrt[3]2 \notin \Q(\sqrt2)$, pues $3 \nmid 2$.
\end{example}

\section{Cuerpos de descomposición; clausura algebraica}

\begin{theorem}[Cuerpos de
descomposición]\label{thm:b3:galois:splitting}
Sea $P \in K[X]$ no constante. Existe un \emph{cuerpo de
descomposición}\index{cuerpo de descomposición} de $P$ sobre $K$: una
extensión $L =
K(\alpha_1, \dots, \alpha_n)$ generada por raíces de $P$ en la que $P$
se descompone en factores lineales. Es único salvo $K$-isomorfismo, y
$[L:K] \leq (\deg P)!$.
\end{theorem}

\begin{proof}
\emph{Existencia}, por inducción sobre $\deg P$: tómese un factor
\omterm{def:b3:rings:divisibility}{irreducible} $Q$ de $P$; el cuerpo
$K_1 = K[X]/(Q)$ contiene la raíz $\alpha_1 = \bar X$ de $Q$ y, por
tanto, de $P$; escríbase $P = (X -
\alpha_1)P_1$ sobre $K_1$ y aplíquese la inducción a $P_1$ sobre $K_1$;
los \omterm{def:b3:galois:extension}{grados} multiplican a lo sumo
$n(n-1)\cdots = n!$.

La \emph{unicidad} se sigue del \emph{lema de extensión de
isomorfismos}, más fuerte: sean $\sigma \colon K \to K'$ un isomorfismo,
$P \in K[X]$, $P^\sigma$ el polinomio con los coeficientes trasladados y
$L, L'$ \omterm{thm:b3:galois:splitting}{cuerpos de descomposición} de
$P, P^\sigma$; entonces $\sigma$ se extiende a un isomorfismo
$L \to L'$. Inducción sobre $[L:K]$: si $P$ se descompone en $K$,
entonces $L = K$ y $L' = K'$ ($P^\sigma$ se descompone en $K'$, y $L'$
está generado por sus raíces). En caso contrario, elíjase una raíz
$\alpha \in L \setminus K$ de un factor
\omterm{def:b3:rings:divisibility}{irreducible} $Q$ de $P$ con
$\deg Q \geq 2$; $Q^\sigma$ es un factor
\omterm{def:b3:rings:divisibility}{irreducible} de $P^\sigma$, con una
raíz $\beta \in L'$; entonces
\[
K(\alpha) \cong K[X]/(Q) \xrightarrow{\ \sigma\ }
K'[X]/(Q^\sigma) \cong K'(\beta)
\]
extiende $\sigma$ con $\alpha \mapsto \beta$. Ahora $L$ es un
\omterm{thm:b3:galois:splitting}{cuerpo de descomposición} de $P$ sobre
$K(\alpha)$, y $L'$ lo es de $P^\sigma$ sobre $K'(\beta)$, con
$[L : K(\alpha)] < [L:K]$: la inducción extiende aún más hasta
$L \to L'$.
\end{proof}

\begin{definition}\label{def:b3:galois:closure}
Un cuerpo $\Omega$ es \emph{algebraicamente cerrado}\index{cuerpo
algebraicamente cerrado} si todo polinomio no constante de $\Omega[X]$
tiene una raíz en $\Omega$ (y por tanto se descompone). Una
\emph{clausura algebraica}\index{clausura algebraica} de $K$ es una
extensión \omterm{def:b3:galois:algebraic}{algebraica} $\bar K/K$ con
$\bar K$ \omterm{def:b3:galois:algebraic}{algebraicamente} cerrado.
\end{definition}

\begin{theorem}[Steinitz]\label{thm:b3:galois:steinitz}
Todo cuerpo $K$ tiene una \omterm{def:b3:galois:closure}{clausura
algebraica}, única salvo $K$-isomorfismo.
\end{theorem}

\begin{proof}
\emph{Existencia (construcción de Artin).} Sea $R = K[(X_f)_f]$ el
anillo de polinomios con una variable $X_f$ por cada $f \in K[X]$ mónico
no constante, y sea $I$ el \omterm{def:b3:rings:ideal}{ideal} generado
por todos los $f(X_f)$. $I$ es propio: una relación
$1 = \sum_{i=1}^r g_i\,
f_i(X_{f_i})$ involucra un número finito de polinomios; en un
\omterm{thm:b3:galois:splitting}{cuerpo de descomposición} común $E$ de
$f_1\cdots f_r$, tómense raíces $\alpha_i$ de $f_i$ y evalúese
$X_{f_i} \mapsto \alpha_i$ (las demás variables $\mapsto 0$): $1 = 0$,
absurdo. Sea $\mathfrak m \supseteq I$ maximal
(el~\cref{thm:b3:rings:krull}; Zorn) y sea $K_1 = R/\mathfrak
m$: una \omterm{def:b3:galois:extension}{extensión de cuerpos} de $K$ en
la que todo $f \in
K[X]$ no constante tiene una raíz, a saber $\bar X_f$, y que es
\omterm{def:b3:galois:algebraic}{algebraica} sobre $K$ (está generada
por los $\bar X_f$, cada uno
\omterm{def:b3:galois:algebraic}{algebraico}). Iterando:
$K \subseteq K_1 \subseteq K_2 \subseteq \cdots$, donde $K_{n+1}$ hace
con $K_n$ lo que $K_1$ hizo con $K$, y sea $\Omega =
\bigcup_n K_n$, que es un cuerpo. Todo $g \in \Omega[X]$ no constante
tiene sus finitos coeficientes en algún $K_n$; un factor
\omterm{def:b3:rings:divisibility}{irreducible} de $g$ sobre $K_n$ tiene
una raíz en $K_{n+1} \subseteq
\Omega$: $\Omega$ es \omterm{def:b3:galois:closure}{algebraicamente
cerrado} y \omterm{def:b3:galois:algebraic}{algebraico} sobre $K$ (cada
$K_n$ lo es, por transitividad, el~\cref{cor:b3:galois:algclosed}):
$\Omega$ es una \omterm{def:b3:galois:closure}{clausura algebraica}.

\emph{Unicidad.} Sean $\Omega, \Omega'$ dos
\omterm{def:b3:galois:closure}{clausuras algebraicas}. Considérese el
conjunto de los pares $(E, \tau)$ con $K
\subseteq E \subseteq \Omega$ y $\tau\colon E \to \Omega'$ una
$K$-inmersión, ordenado por extensión; es no vacío ($(K,
\mathrm{id})$) e inductivo (unión de una cadena), de modo que Zorn da un
$(E_0, \tau_0)$ maximal. Si $E_0 \neq \Omega$, tómese $\alpha \in
\Omega\setminus E_0$: $\pi_{\alpha/E_0}$ se transforma en un polinomio
sobre $\tau_0(E_0)$ que tiene una raíz $\beta$ en el
\omterm{def:b3:galois:closure}{algebraicamente cerrado} $\Omega'$, y
$\tau_0$ se extiende a $E_0(\alpha) \to
\Omega'$ ($\alpha \mapsto \beta$), en contra de la maximalidad. Existe,
pues, una $K$-inmersión $\tau \colon \Omega \to \Omega'$; su imagen,
isomorfa a $\Omega$, es \omterm{def:b3:galois:algebraic}{algebraicamente}
cerrada, y $\Omega'$ es \omterm{def:b3:galois:algebraic}{algebraico}
sobre ella: para $x \in \Omega'$, $\pi_{x/\tau(\Omega)}$ se descompone
sobre $\tau(\Omega)$, luego $x \in
\tau(\Omega)$. Así pues, $\tau$ es sobreyectiva: es un isomorfismo.
\end{proof}

\begin{remark}\label{rem:b3:galois:closureexamples}
Para $K = \Q$ puede evitarse la maquinaria transfinita: los números
\omterm{def:b3:galois:algebraic}{algebraicos}
$\bar\Q = \{z \in \C : z \text{ algebraic over
} \Q\}$ forman una \omterm{def:b3:galois:closure}{clausura algebraica}
--- un subcuerpo de $\C$ por el~\cref{cor:b3:galois:algclosed},
\omterm{def:b3:galois:closure}{algebraicamente cerrado} porque $\C$ lo
es (d'Alembert--Gauss, demostrado mediante análisis complejo en
el~\cref{ch:b3:holomorphic}) y porque las raíces de polinomios sobre
$\bar\Q$ son \omterm{def:b3:galois:algebraic}{algebraicas} sobre $\Q$
por transitividad.
\end{remark}

\section{Cuerpos finitos}

\begin{theorem}\label{thm:b3:galois:finitefields}
Sea $p$ primo, $n \geq 1$, $q = p^n$.
\begin{enumerate}
  \item Un \omterm{thm:b3:galois:finitefields}{cuerpo finito} tiene por
  cardinal una potencia de primo y, para cada $q$, existe exactamente un
  cuerpo $\mathbb F_q$ de $q$ elementos salvo isomorfismo: el
  \omterm{thm:b3:galois:splitting}{cuerpo de descomposición} de
  $X^q - X$ sobre $\mathbb F_p$.\index{cuerpo finito} \item El
  \emph{Frobenius}\index{endomorfismo de Frobenius} $F
        \colon x \mapsto x^p$ es un automorfismo de $\mathbb
        F_q$, y el grupo de automorfismos de $\mathbb F_q$ es cíclico de
  orden $n$, generado por $F$. \item $\mathbb F_{p^m}$ se sumerge en
  $\mathbb F_{p^n}$ si y solo si $m \mid
        n$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Un \omterm{thm:b3:galois:finitefields}{cuerpo finito} $E$ tiene
característica $p > 0$ y es un $\mathbb F_p$-espacio vectorial de
dimensión finita: $\abs E = p^n$. Su grupo multiplicativo tiene orden
$q - 1$, luego todo $x \in E$ cumple $x^q = x$: $E$ consta de $q$ raíces
de $X^q - X$ y, por tanto, es un \omterm{thm:b3:galois:splitting}{cuerpo
de descomposición} suyo sobre $\mathbb F_p$ --- lo que determina $E$
salvo isomorfismo (el~\cref{thm:b3:galois:splitting}). Recíprocamente,
en un \omterm{thm:b3:galois:splitting}{cuerpo de descomposición} $L$ de
$X^q - X$, el conjunto $E$ de sus raíces es un \emph{subcuerpo}:
$(x + y)^q = x^q + y^q$ iterando el sueño del principiante
$(a+b)^p = a^p + b^p$ ($p \mid \binom pk$), y $(xy)^q = x^qy^q$,
$(x^{-1})^q = (x^q)^{-1}$; y tiene exactamente $q$ elementos porque
$X^q - X$ es \omterm{def:b3:galois:separable}{separable}: su derivada es
$qX^{q-1} - 1 = -1$ (pues $p \mid q$), coprima con él, de modo que no
hay raíces múltiples. Así, $L = E$ tiene $q$ elementos.

(2) $F$ es un morfismo de cuerpos (sueño del principiante), inyectivo
(son cuerpos) y, por tanto, biyectivo sobre el conjunto finito
$\mathbb F_q$. Se tiene $F^n =
\mathrm{id}$ ($x^q = x$) y ninguna potencia menor es la identidad:
$F^m = \mathrm{id}$ significaría que los $q$ elementos son raíces de
$X^{p^m} - X$, lo que obliga a $p^m \geq q$. Así pues,
$\langle F\rangle$ es cíclico de orden $n$; y no hay más automorfismos,
por la cota $\abs{\operatorname{Aut}} \leq [\,\mathbb F_q :
\mathbb F_p\,] = n$ que se demuestra más abajo
(la~\cref{prop:b3:galois:embeddings} con $L = \mathbb F_q$, $K =
\mathbb F_p$: los automorfismos fijan el cuerpo primo).

(3) Si $\mathbb F_{p^m} \subseteq \mathbb F_{p^n}$, la
\omterm{thm:b3:galois:tower}{fórmula de los grados} da $p^n = (p^m)^d$:
$m \mid n$. Recíprocamente, si $m \mid n$, entonces
$p^m - 1 \mid p^n - 1$ (suma geométrica), luego $X^{p^m} - X$ divide a
$X^{p^n} - X$ (mismo argumento sobre los exponentes:
$X^{a} - 1 \mid X^{b}
- 1$ cuando $a \mid b$), y las raíces del primero dentro de
$\mathbb F_{p^n}$ forman el subcuerpo pedido, de cardinal $p^m$
(separabilidad como en (1)).
\end{proof}

\begin{theorem}[Ciclicidad]\label{thm:b3:galois:cyclic}
Todo subgrupo finito del grupo multiplicativo de un cuerpo es cíclico.
En particular, $\mathbb F_q^\times \cong
\Z/(q-1)\Z$.\index{grupo multiplicativo de un cuerpo finito}
\end{theorem}

\begin{proof}
Sea $G \leq K^\times$ finito. Por el teorema de estructura
(el~\cref{cor:b3:modules:abelian}), $G \cong \Z/d_1 \times \dots
\times \Z/d_s$ con $d_1 \mid \dots \mid d_s$. Todo $x \in G$ cumple
entonces $x^{d_s} = 1$; pero $X^{d_s} - 1$ tiene a lo sumo $d_s$ raíces
en el cuerpo $K$: $\abs G = d_1\cdots d_s \leq d_s$, lo que obliga a
$s = 1$: $G$ es cíclico.
\end{proof}

\begin{example}\label{ex:b3:galois:f8}
$\mathbb F_8 = \mathbb F_2[X]/(X^3 + X + 1)$: la cúbica no tiene raíces
en $\mathbb F_2$, luego es
\omterm{def:b3:rings:divisibility}{irreducible}. Escribiendo $\omega =
\bar X$: $\mathbb F_8^\times$ es cíclico de orden $7$, de modo que
\emph{todo} elemento $\neq 0, 1$ lo genera. Los subcuerpos de
$\mathbb F_{p^{12}}$ forman el retículo de divisores de $12$:
$\mathbb F_p, \mathbb F_{p^2}, \mathbb F_{p^3}, \mathbb F_{p^4},
\mathbb F_{p^6}, \mathbb F_{p^{12}}$ --- un primer caso completo de la
\omterm{thm:b3:galois:fundamental}{correspondencia de Galois}.
\end{example}

\section{Separabilidad e inmersiones}

\begin{definition}\label{def:b3:galois:separable}
Un polinomio $P \in K[X]$ es \emph{separable}\index{polinomio separable}
si no tiene raíces múltiples en un
\omterm{thm:b3:galois:splitting}{cuerpo de descomposición} ---
equivalentemente, $\gcd(P, P') = 1$ (una raíz múltiple es una raíz
común; recíprocamente, sobre el \omterm{thm:b3:galois:splitting}{cuerpo
de descomposición} una raíz común es múltiple; y el máximo común divisor
no cambia al hacer una \omterm{def:b3:galois:extension}{extensión de
cuerpos}, según el argumento del~\cref{cor:b3:modules:descent}). Un
\omterm{def:b3:galois:algebraic}{elemento algebraico} es separable si lo
es su \omterm{def:b3:galois:algebraic}{polinomio mínimo}; una extensión
$L/K$ es separable si lo son todos sus elementos.
\end{definition}

\begin{proposition}\label{prop:b3:galois:perfect}
Un $P \in K[X]$ \emph{\omterm{def:b3:rings:divisibility}{irreducible}}
es \omterm{def:b3:galois:separable}{separable} salvo que $P' = 0$, lo
que obliga a $\operatorname{char} K = p > 0$ y $P \in K[X^p]$. En
consecuencia, toda extensión
\omterm{def:b3:galois:algebraic}{algebraica} de un cuerpo de
característica $0$, y de un \omterm{thm:b3:galois:finitefields}{cuerpo
finito}, es \omterm{def:b3:galois:separable}{separable} (tales cuerpos
se llaman \emph{perfectos}\index{cuerpo perfecto}).
\end{proposition}

\begin{proof}
$\gcd(P, P')$ divide a $P$; si no es $1$, la irreducibilidad obliga a
$\gcd = P$ (salvo una constante), luego $P \mid P'$ con $\deg
P' < \deg P$: $P' = 0$. Escribiendo $P = \sum a_kX^k$: $ka_k = 0$ para
todo $k$, de modo que en característica $0$, $P$ es constante
(excluido); y en característica $p$, $a_k = 0$ salvo que $p \mid k$:
$P =
Q(X^p)$. Sobre un \omterm{thm:b3:galois:finitefields}{cuerpo finito},
todo elemento es una potencia $p$-ésima (el
\omterm{thm:b3:galois:finitefields}{Frobenius} es sobreyectivo), luego
$Q(X^p) = \sum b_k^p X^{pk} = (\sum
b_kX^k)^p$ no es \omterm{def:b3:rings:divisibility}{irreducible}: allí
tampoco puede ocurrir $P' = 0$ para $P$
\omterm{def:b3:rings:divisibility}{irreducible}.
\end{proof}

\begin{proposition}[Recuento de
inmersiones]\label{prop:b3:galois:embeddings}
Sea $L = K(\alpha_1, \dots, \alpha_r)$ finita sobre $K$ y sea
$\sigma \colon K \to \Omega$ una inmersión en un
\omterm{def:b3:galois:closure}{cuerpo algebraicamente cerrado}. Entonces
el número de extensiones de $\sigma$ a $L$ es a lo sumo $[L:K]$, con
igualdad si $L/K$ es \omterm{def:b3:galois:separable}{separable}. En
particular, $\abs{\operatorname{Aut}_K(L)} \leq [L:K]$.
\end{proposition}

\begin{proof}
Inducción sobre $[L:K]$ mediante pasos simples. Para $L = K(\alpha)$:
una extensión $\tau$ queda determinada por $\tau(\alpha)$, que ha de ser
una raíz en $\Omega$ de $\pi_\alpha^\sigma$; y recíprocamente cada una
de esas raíces da una extensión ($K(\alpha) \cong K[X]/(\pi_\alpha)$).
El número de extensiones es el número de raíces \emph{distintas} de
$\pi_\alpha^\sigma$ en $\Omega$: a lo sumo $\deg\pi_\alpha =
[K(\alpha):K]$, con igualdad si y solo si $\pi_\alpha$ es
\omterm{def:b3:galois:separable}{separable} (la separabilidad de
$\pi^\sigma$ y la de $\pi$ coinciden: $\sigma$ conserva el máximo común
divisor con la derivada). En general, factorícese $L =
K(\alpha_1)(\alpha_2, \dots)$: las extensiones de $\sigma$ a
$K(\alpha_1)$ son $\leq [K(\alpha_1):K]$, y cada una se extiende de
$\leq [L : K(\alpha_1)]$ maneras por inducción; multiplíquese
(\omterm{thm:b3:galois:tower}{fórmula de los grados}). En el caso
\omterm{def:b3:galois:separable}{separable}, ambos recuentos son
igualdades: los \omterm{def:b3:galois:algebraic}{polinomios mínimos}
sobre el cuerpo mayor $K(\alpha_1)$ dividen a los de $K$ y, por tanto,
siguen siendo \omterm{def:b3:galois:separable}{separables}.
\end{proof}

\begin{theorem}[Elemento primitivo]\label{thm:b3:galois:primitive}
Toda extensión finita \emph{\omterm{def:b3:galois:separable}{separable}}
es simple: $L =
K(\gamma)$ para algún $\gamma$.\index{teorema del elemento primitivo}
\end{theorem}

\begin{proof}
Si $K$ es finito, también lo es $L$, y sirve un generador $\gamma$ del
grupo cíclico $L^\times$ (el~\cref{thm:b3:galois:cyclic}). Sea $K$
infinito; por inducción basta tratar $L =
K(\alpha, \beta)$. Sea $n = [L:K]$; por
la~\cref{prop:b3:galois:embeddings} hay $n$ $K$-inmersiones distintas
$\sigma_1, \dots, \sigma_n \colon L \to \Omega$ ($\Omega$ una
\omterm{def:b3:galois:closure}{clausura algebraica}). El polinomio
\[
D(T)=\prod_{i<j}\bigl[\bigl(\sigma_i(\alpha)-\sigma_j(\alpha)
\bigr) + T\bigl(\sigma_i(\beta) - \sigma_j(\beta)\bigr)\bigr]
\]
no es idénticamente nulo: un factor se anula idénticamente solo si
$\sigma_i, \sigma_j$ coinciden tanto en $\alpha$ como en $\beta$ y, por
tanto, en $L$ --- excluido para $i \neq j$. Como $K$ es infinito, tómese
$c \in
K$ con $D(c) \ne 0$: entonces los $n$ elementos $\sigma_i(\alpha +
c\beta)$ son distintos dos a dos, luego $\gamma = \alpha + c\beta$ tiene
al menos $n$ conjugados distintos en $\Omega$, es decir,
$\deg \pi_\gamma \geq n$: $[K(\gamma):K] \geq n = [L:K]$ obliga a
$L = K(\gamma)$.
\end{proof}

\section{La correspondencia de Galois}

\begin{definition}\label{def:b3:galois:galois}
Una extensión finita $L/K$ es \emph{de Galois}\index{extensión de
Galois} si es el \omterm{thm:b3:galois:splitting}{cuerpo de
descomposición} de un polinomio
\emph{\omterm{def:b3:galois:separable}{separable}} sobre $K$. Su
\emph{grupo de Galois}\index{grupo de Galois} es
$\operatorname{Gal}(L/K) = \operatorname{Aut}_K(L)$, el grupo de los
automorfismos de cuerpo de $L$ que fijan $K$ punto a punto.
\end{definition}

\begin{proposition}\label{prop:b3:galois:galoisorder}
Si $L/K$ es \omterm{def:b3:galois:galois}{de Galois}, entonces
$\abs{\operatorname{Gal}(L/K)} =
[L:K]$; además, $L/F$ es \omterm{def:b3:galois:galois}{de Galois} para
todo cuerpo intermedio $K
\subseteq F \subseteq L$, y toda $F$-inmersión $L \to \Omega
\supseteq L$ tiene imagen $L$ (\emph{normalidad}).
\end{proposition}

\begin{proof}
Sea $L$ el cuerpo que descompone el \omterm{def:b3:galois:separable}{polinomio
separable} $P$ sobre $K$, y fijemos una
\omterm{def:b3:galois:closure}{clausura algebraica}
$\Omega \supseteq L$. $L/K$ es
\omterm{def:b3:galois:separable}{separable}: está generado por raíces de
$P$; la separabilidad de todo elemento se sigue del caso de igualdad de
más abajo, pero argumentemos directamente ---
la~\cref{prop:b3:galois:embeddings} aplicada a los generadores (raíces
del \omterm{def:b3:galois:separable}{separable} $P$, cuyos
\omterm{def:b3:galois:algebraic}{polinomios mínimos} dividen a $P$) da
exactamente $[L:K]$ extensiones de $K \hookrightarrow \Omega$ (en el
paso inductivo, el \omterm{def:b3:galois:algebraic}{polinomio mínimo} de
una raíz de $P$ sobre un cuerpo intermedio sigue dividiendo a $P$ y, por
tanto, es \omterm{def:b3:galois:separable}{separable}). Cada una de esas
inmersiones $\tau \colon L \to \Omega$ permuta las raíces de $P$ ($\tau$
fija los coeficientes), y $L$ está generado por ellas: $\tau(L) = L$.
Así pues, inmersiones $=$ automorfismos:
$\abs{\operatorname{Gal}(L/K)} = [L:K]$. Para $F$ intermedio: $L$ es
también el \omterm{thm:b3:galois:splitting}{cuerpo de descomposición} de
$P$ sobre $F$, y $P$ sigue siendo
\omterm{def:b3:galois:separable}{separable}: $L/F$ es
\omterm{def:b3:galois:galois}{de Galois}; el mismo argumento da la
normalidad sobre $F$.
\end{proof}

\begin{lemma}[Artin]\label{lem:b3:galois:artin}
Sea $G$ un grupo finito de automorfismos de un cuerpo $L$ y sea
$K = L^G = \{x : \sigma(x) = x\ \forall\sigma \in G\}$ su cuerpo fijo.
Entonces $[L : L^G] \leq \abs G$.
\end{lemma}

\begin{proof}
Sean $n = \abs G$, $G = \{\sigma_1, \dots, \sigma_n\}$, y supongamos que
$x_1, \dots, x_{n+1} \in L$ son linealmente independientes sobre $K$. El
sistema lineal homogéneo de $n$ ecuaciones con $n+1$ incógnitas $(c_j)$
sobre $L$,
\[
\sum_{j=1}^{n+1} c_j\,\sigma_i(x_j) = 0
\qquad (i = 1, \dots, n),
\]
tiene una solución no nula; elíjase una con el \emph{menor} número de
entradas no nulas, digamos $c_1, \dots, c_r \neq 0$ (renumerando),
$r \geq 2$ (una sola $c_j\sigma_i(x_j) = 0$ es imposible), normalizada
con $c_r = 1$. No todos los $c_j$ están en $K$: la ecuación
correspondiente a $\sigma_i =
\mathrm{id}$ contradiría la independencia; digamos $c_1 \notin K$, de
modo que $\tau(c_1) \ne c_1$ para algún $\tau \in G$. Apliquemos $\tau$
a todas las ecuaciones: como $\tau\sigma_i$ recorre $G$, el vector
$(\tau(c_j))_j$ es otra solución; restando, $(c_j -
\tau(c_j))_j$ es una solución con menos entradas no nulas (la entrada
$r$-ésima $1 - 1 = 0$ se anula, y la primera no) y no nula:
contradicción. Así pues, $n+1$ elementos cualesquiera son dependientes:
$[L:K]
\leq n$.
\end{proof}

\begin{theorem}[Teorema fundamental de la teoría de Galois]
\label{thm:b3:galois:fundamental}
Sea $L/K$ una \omterm{def:b3:galois:galois}{extensión de Galois} de
grupo $G =
\operatorname{Gal}(L/K)$.\index{correspondencia de Galois}
\begin{enumerate}
  \item $L^G = K$.
  \item Las aplicaciones $H \mapsto L^H$ y $F \mapsto
        \operatorname{Gal}(L/F)$ son biyecciones mutuamente inversas y
  que invierten las inclusiones entre los subgrupos de $G$ y los cuerpos
  intermedios $K \subseteq F \subseteq L$; además, $[L : L^H] = \abs H$
  y $[L^H : K] = [G : H]$. \item $H \trianglelefteq G$ si y solo si
  $L^H/K$ es \omterm{def:b3:galois:galois}{de Galois}, y entonces la
  restricción induce $\operatorname{Gal}(L^H/K) \cong
        G/H$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Claramente $K \subseteq L^G$. Recíprocamente, sea $\alpha \in L
\setminus K$; exhibiremos $\sigma \in G$ con $\sigma(\alpha) \ne
\alpha$. El \omterm{def:b3:galois:algebraic}{polinomio mínimo}
$\pi_\alpha$ sobre $K$ tiene \omterm{def:b3:galois:extension}{grado}
$\geq 2$ y es \omterm{def:b3:galois:separable}{separable} ($L/K$ es
\omterm{def:b3:galois:separable}{separable},
la~\cref{prop:b3:galois:galoisorder}), luego tiene otra raíz
$\beta \neq \alpha$ en una \omterm{def:b3:galois:closure}{clausura
algebraica} $\Omega \supseteq L$. Extiéndase la $K$-inmersión
$K(\alpha) \to \Omega$, $\alpha \mapsto
\beta$, a una inmersión $\tau\colon L \to \Omega$
(la~\cref{prop:b3:galois:embeddings}); por normalidad
(la~\cref{prop:b3:galois:galoisorder}), $\tau(L) = L$, luego $\tau \in
G$, $\beta = \tau(\alpha) \in L$ y $\tau(\alpha) \neq
\alpha$.

(2) Para un subgrupo $H$: $L/L^H$ es \omterm{def:b3:galois:galois}{de
Galois} (la~\cref{prop:b3:galois:galoisorder}) y
$\operatorname{Gal}(L/L^H) \supseteq H$ trivialmente, luego $[L:L^H] =
\abs{\operatorname{Gal}(L/L^H)} \geq \abs H$; el lema de Artin da
$[L:L^H] \leq \abs H$: hay igualdad, y $\operatorname{Gal}(L/L^H) = H$.
Para un cuerpo intermedio $F$: como $L/F$ es
\omterm{def:b3:galois:galois}{de Galois}, (1) aplicado a $L/F$ da
$L^{\operatorname{Gal}(L/F)} = F$. Las dos aplicaciones son mutuamente
inversas y es evidente que invierten las inclusiones.
\omterm{def:b3:galois:extension}{Grados}: $[L:L^H] = \abs H$ acaba de
demostrarse, y $[L^H:K] = [L:K]/[L:L^H] = \abs G/\abs H$.

(3) Para $\sigma \in G$ y $H \leq G$: $\sigma(L^H) =
L^{\sigma H\sigma^{-1}}$ (comprobación directa). Por la biyección,
$\sigma(L^H) = L^H$ para todo $\sigma$ si y solo si
$H \trianglelefteq G$. Ahora bien, si $H \trianglelefteq G$, póngase
$F = L^H$: todo $\sigma \in
G$ se restringe a un automorfismo de $F$, lo que da un morfismo $\rho
\colon G \to \operatorname{Aut}_K(F)$ de núcleo $\{\sigma :
\sigma\restriction_F = \mathrm{id}\} = \operatorname{Gal}(L/F) =
H$. Así, $G/H$ se sumerge en $\operatorname{Aut}_K(F)$, de donde
$\abs{\operatorname{Aut}_K(F)} \geq [G:H] = [F:K]$; la desigualdad
contraria vale siempre (la~\cref{prop:b3:galois:embeddings}):
$\abs{\operatorname{Aut}_K(F)} = [F:K]$ y $\rho$ es sobreyectivo. Queda
ver que $F/K$ es \omterm{def:b3:galois:galois}{de Galois}: $F$ es
\omterm{def:b3:galois:separable}{separable} sobre $K$ (dentro del
\omterm{def:b3:galois:separable}{separable} $L/K$) y $F = K(\gamma)$
(el~\cref{thm:b3:galois:primitive}); el polinomio
$\prod_{\sigma \in G/H}\bigl(X - \sigma(\gamma)\bigr)$ (producto sobre
las imágenes distintas, que están en $F$: $\sigma(F) =
L^{\sigma H\sigma^{-1}} = L^H = F$ por la normalidad de $H$) tiene los
coeficientes fijados por $G$ y, por tanto, en $K$ por (1): es un
\omterm{def:b3:galois:separable}{polinomio separable} de $K[X]$
descompuesto por $F$, y sus raíces generan $F$: $F/K$ es
\omterm{def:b3:galois:galois}{de Galois}. Recíprocamente, si $F = L^H$
con $F/K$ \omterm{def:b3:galois:galois}{de Galois}, la normalidad de $F$
(la~\cref{prop:b3:galois:galoisorder}, aplicada a las inmersiones
$F \to \Omega$ restringidas de elementos de $G$) da $\sigma(F) = F$ para
todo $\sigma \in G$, es decir, $H
\trianglelefteq G$.
\end{proof}

\section{Extensiones ciclotómicas}

\begin{definition}\label{def:b3:galois:cyclotomic}
Sean $n \geq 1$ y $\zeta_n = \eu^{2\iu\pi/n}$. El \emph{polinomio
ciclotómico}\index{polinomio ciclotómico} de índice $n$ es
$\Phi_n = \prod_{\gcd(k,n)=1,\ 1 \le k \le n} \bigl(X -
\zeta_n^k\bigr)$, de \omterm{def:b3:galois:extension}{grado}
$\varphi(n)$; agrupando las raíces de $X^n - 1$ según su orden exacto,
$X^n - 1 = \prod_{d \mid n}\Phi_d$, lo que muestra por inducción que
$\Phi_n \in \Z[X]$ (división euclídea de polinomios enteros mónicos).
\end{definition}

\begin{theorem}\label{thm:b3:galois:cyclotomicirred}
$\Phi_n$ es \omterm{def:b3:rings:divisibility}{irreducible} sobre $\Q$;
por tanto $[\Q(\zeta_n) : \Q] =
\varphi(n)$ y
\[
\operatorname{Gal}\bigl(\Q(\zeta_n)/\Q\bigr) \;\cong\;
(\Z/n\Z)^\times,
\qquad \sigma_a(\zeta_n) = \zeta_n^a .
\]
Así pues, la extensión $\Q(\zeta_n)/\Q$ es
\omterm{def:b3:galois:galois}{de Galois} con grupo \emph{abeliano}.
\end{theorem}

\begin{proof}
Sea $f = \pi_{\zeta_n}$, de modo que $\Phi_n = fg$ con $f, g \in \Z[X]$
mónicos (lema de Gauss, el~\cref{lem:b3:rings:gauss}: los contenidos se
multiplican y todos los polinomios son mónicos). \emph{Afirmación: si
$\zeta$ es una raíz de $f$ y $p \nmid n$ es primo, entonces $\zeta^p$ es
raíz de $f$.} En caso contrario, $\zeta^p$ sería raíz de $g$ (es una
raíz $n$-ésima primitiva de la unidad), luego $\zeta$ sería raíz de
$g(X^p)$, y $f
\mid g(X^p)$ en $\Z[X]$ (\omterm{def:b3:galois:algebraic}{polinomio
mínimo} y de nuevo Gauss). Redúzcase módulo $p$:
$\bar g(X^p) = \bar g(X)^p$ (el
\omterm{thm:b3:galois:finitefields}{Frobenius} sobre $\mathbb F_p[X]$:
coeficiente a coeficiente $a^p = a$, y sueño del principiante), luego
$\bar f \mid \bar g^{\,p}$: $\bar f$ y $\bar g$ comparten un factor
\omterm{def:b3:rings:divisibility}{irreducible} y
$\bar\Phi_n = \bar f\bar g$ tiene un factor múltiple. Entonces también
lo tiene $X^n - \bar 1$; pero su derivada $\bar nX^{n-1}$ es coprima con
él ($p \nmid n$, y $0$ no es raíz): contradicción.

Toda raíz primitiva $\zeta_n^k$ ($\gcd(k, n) = 1$) se obtiene de
$\zeta_n$ mediante potencias sucesivas de primos que no dividen a $n$
(factorícese $k$): la afirmación se propaga, de modo que toda raíz
primitiva es raíz de $f$: $f = \Phi_n$,
\omterm{def:b3:rings:divisibility}{irreducible}. En consecuencia,
$[\Q(\zeta_n):\Q] = \varphi(n)$, y $\Q(\zeta_n)$ es el
\omterm{thm:b3:galois:splitting}{cuerpo de descomposición} del
\omterm{def:b3:galois:separable}{separable} $X^n - 1$ (todas las raíces
son potencias de $\zeta_n$): es \omterm{def:b3:galois:galois}{de
Galois}. Un automorfismo $\sigma$ envía $\zeta_n$ a otra raíz primitiva
$\zeta_n^{a(\sigma)}$, y $\sigma
\mapsto a(\sigma)$ es un morfismo inyectivo en $(\Z/n\Z)^\times$; ambos
grupos tienen orden $\varphi(n)$: es un isomorfismo.
\end{proof}

\section{Regla y compás}

\begin{definition}\label{def:b3:galois:constructible}
Identifiquemos el plano con $\C$ y partamos de $\{0, 1\}$. Un punto es
\emph{construible}\index{número construible} si puede obtenerse mediante
un número finito de intersecciones de rectas que pasan por dos puntos ya
construidos y de circunferencias centradas en un punto construido con
radio igual a la distancia entre dos puntos construidos.
\end{definition}

\begin{theorem}[Wantzel]\label{thm:b3:galois:wantzel}
$z \in \C$ es \omterm{def:b3:galois:constructible}{construible} si y
solo si existe una torre $\Q = F_0
\subseteq F_1 \subseteq \dots \subseteq F_r$ con $[F_{i+1} :
F_i] = 2$ y $z \in F_r$. En particular, un
\omterm{def:b3:galois:constructible}{número construible} es
\omterm{def:b3:galois:algebraic}{algebraico} de
\omterm{def:b3:galois:extension}{grado} una potencia de $2$ sobre
$\Q$.\index{teorema de Wantzel}
\end{theorem}

\begin{proof}
($\Rightarrow$) Las coordenadas de la intersección de dos rectas que
pasan por puntos con coordenadas en un subcuerpo $F \subseteq \R$
resuelven un sistema lineal sobre $F$: permanecen en $F$. Las
intersecciones recta--circunferencia y circunferencia--circunferencia
llevan, tras eliminar la parte lineal (restando las dos ecuaciones de
las circunferencias se obtiene una recta), a una ecuación de segundo
\omterm{def:b3:galois:extension}{grado} sobre $F$: las nuevas coordenadas están en $F$ o en $F(\sqrt d)$
para cierto $d \in F$, $d > 0$. Por inducción, todo punto construido
tiene sus coordenadas en una torre de extensiones cuadráticas de $\Q$; y
$z = x + \iu y$ está también en una torre cuadrática (adjúntese $\iu$:
un paso cuadrático más). La consecuencia sobre el
\omterm{def:b3:galois:extension}{grado}: $[\Q(z):\Q]$ divide a
$[F_r : \Q] = 2^r$ (\omterm{thm:b3:galois:tower}{fórmula de los
grados}).

($\Leftarrow$) Los \omterm{def:b3:galois:constructible}{números
construibles} forman un cuerpo: sumas y diferencias mediante
paralelogramos (las paralelas son
\omterm{def:b3:galois:constructible}{construibles}: trácense dos veces
perpendiculares --- la perpendicular clásica por un punto usa una
circunferencia y dos arcos); productos y cocientes mediante las
configuraciones del teorema de Tales (dadas longitudes $a, b$,
constrúyanse $ab$ y $a/b$ con triángulos semejantes sobre dos
semirrectas). Y el cuerpo es cerrado por raíces cuadradas: para $a > 0$,
la circunferencia de diámetro $1 + a$ y la perpendicular en el punto de
unión se cortan a altura $\sqrt a$ (teorema de la altura en un triángulo
rectángulo); para un $w = \rho\eu^{\iu\theta}$ complejo, constrúyase
$\sqrt\rho$ y bisóquese $\theta$ (la bisección de un ángulo es una
construcción con compás). Las partes real e imaginaria de los elementos
de una torre cuadrática son, por tanto,
\omterm{def:b3:galois:constructible}{construibles} por inducción sobre
la torre: cada paso adjunta las raíces de una ecuación de segundo \omterm{def:b3:galois:extension}{grado},
expresables mediante operaciones de cuerpo y una raíz cuadrada de un
número ya construido (la fórmula cuadrática; en característica $0$).
\end{proof}

\begin{corollary}\label{cor:b3:galois:impossible}
Los tres problemas clásicos no tienen solución con regla y
compás:\index{regla y compás}
\begin{enumerate}
  \item \emph{Duplicación del cubo}: $\sqrt[3]2$ tiene
  \omterm{def:b3:galois:extension}{grado} $3$, que no es potencia de
  $2$. \item \emph{Trisección del ángulo}: trisecar $60^\circ$ exige
  $\cos 20^\circ$, raíz del
  \omterm{def:b3:rings:divisibility}{irreducible} $8X^3
        - 6X - 1$: \omterm{def:b3:galois:extension}{grado} $3$. \item
  \emph{Cuadratura del círculo}: $\sqrt\pi$ es trascendente (lo es $\pi$
  --- teorema de Lindemann, admitido aquí: su demostración pertenece a
  un curso de teoría de la trascendencia).
\end{enumerate}
Además, el $n$-ágono regular es
\omterm{def:b3:galois:constructible}{construible} si y solo si
$\varphi(n)$ es una potencia de $2$ (Gauss--Wantzel; el «si» usa el
método del problema de fin de semana, y el «solo si» es
el~\cref{thm:b3:galois:wantzel} aplicado a $\zeta_n$, de
\omterm{def:b3:galois:extension}{grado} $\varphi(n)$). Para $n = 7$:
$\varphi(7) = 6$: el heptágono regular es imposible; para $n = 17$:
$\varphi(17) = 16 = 2^4$: es
\omterm{def:b3:galois:constructible}{construible} --- y el problema de
fin de semana lo construye.
\end{corollary}

\begin{proof}
(1) $X^3 - 2$ es \omterm{def:b3:rings:divisibility}{irreducible}
(\omterm{thm:b3:rings:criteria}{Eisenstein}). (2) De $\cos
3\theta = 4\cos^3\theta - 3\cos\theta$ con $3\theta =
60^\circ$: $8c^3 - 6c = 1$ para $c = \cos 20^\circ$; la cúbica
$8X^3 - 6X - 1$ no tiene raíces racionales (los candidatos
$\pm1, \pm\frac
12, \pm\frac14, \pm\frac18$ fallan), luego es
\omterm{def:b3:rings:divisibility}{irreducible}:
\omterm{def:b3:galois:extension}{grado} $3$. Un ángulo general de
$60^\circ$ es \omterm{def:b3:galois:constructible}{construible}, de modo
que un trisector construiría $c$. (3) Si $\sqrt\pi$ fuera
\omterm{def:b3:galois:constructible}{construible} sería
\omterm{def:b3:galois:algebraic}{algebraico}, y también lo sería $\pi$.
Para el $n$-ágono: el \omterm{def:b3:galois:extension}{grado} de
$\zeta_n$ es $\varphi(n)$ (el~\cref{thm:b3:galois:cyclotomicirred}); la
necesidad se sigue de Wantzel; y para la suficiencia, el
\omterm{def:b3:galois:galois}{grupo de Galois}, abeliano de orden $2^m$,
admite una cadena de subgrupos de índice $2$ (todo $2$-grupo finito la
admite: el~\cref{exo:b3:groups:10}), cuyos cuerpos fijos forman una
torre cuadrática que termina en $\Q(\zeta_n)$
(el~\cref{thm:b3:galois:fundamental}); se concluye por
\cref{thm:b3:galois:wantzel}.
\end{proof}

\section{Resolubilidad por radicales}

\begin{definition}\label{def:b3:galois:radical}
Una extensión $L/K$ (característica $0$ en toda esta sección) es
\emph{radical} si existe una torre $K = F_0 \subseteq \dots
\subseteq F_r = L$ con $F_{i+1} = F_i(\alpha_i)$,
$\alpha_i^{n_i} \in F_i$: cada paso adjunta una raíz $n_i$-ésima. Un
polinomio $P \in K[X]$ es \emph{resoluble por radicales}\index{resoluble
por radicales} si su \omterm{thm:b3:galois:splitting}{cuerpo de
descomposición} está contenido en alguna extensión radical de $K$.
\end{definition}

\begin{lemma}\label{lem:b3:galois:cyclicsteps}
Supongamos que $K$ contiene una raíz $n$-ésima primitiva de la unidad
$\zeta$, es decir, un $\zeta$ de orden $n$ en $K^\times$, y sea
$a \in K^\times$. Entonces $K(\sqrt[n]a)/K$ es
\omterm{def:b3:galois:galois}{de Galois} con grupo \emph{cíclico}.
Recíprocamente --- no lo necesitaremos ---, toda extensión cíclica de
\omterm{def:b3:galois:extension}{grado} $n$ es de esta forma. Además,
$K(\zeta_n)/K$ es \omterm{def:b3:galois:galois}{de Galois} con grupo
\emph{abeliano}, para todo $K$ de característica $0$.
\end{lemma}

\begin{proof}
$X^n - a$ es \omterm{def:b3:galois:separable}{separable} ($\gcd$ con
$nX^{n-1}$: $a \neq 0$) y se descompone en $K(\alpha)$, $\alpha^n = a$:
sus raíces son los $\zeta^k\alpha \in K(\alpha)$. Luego $K(\alpha)/K$ es
\omterm{def:b3:galois:galois}{de Galois}; la aplicación
$\sigma \mapsto \sigma(\alpha)/\alpha \in \mu_n = \langle
\zeta\rangle$ es un morfismo inyectivo ($\sigma\tau(\alpha) =
\sigma(\tau(\alpha)/\alpha \cdot \alpha) =
\tau(\alpha)/\alpha\cdot\sigma(\alpha)$, pues el cociente está en $K$)
en un grupo cíclico: $\operatorname{Gal}$ es cíclico. El recíproco es la
teoría de Kummer, que no necesitaremos (véase la observación siguiente).
Para $K(\zeta_n)$: descompone el
\omterm{def:b3:galois:separable}{separable} $X^n -
1$, y $\sigma \mapsto a(\sigma)$ con $\sigma(\zeta_n) =
\zeta_n^{a(\sigma)}$ sumerge el grupo en el abeliano $(\Z/n\Z)^\times$
igual que en el~\cref{thm:b3:galois:cyclotomicirred} (la inyectividad
solo necesita que $\zeta_n$ genere las raíces de la unidad implicadas).
\end{proof}

\begin{theorem}[Galois]\label{thm:b3:galois:solvable}
Sea $K$ de característica $0$ y sea $P \in K[X]$ con
\omterm{thm:b3:galois:splitting}{cuerpo de descomposición} $L$. Si $P$
es \omterm{def:b3:galois:radical}{resoluble por radicales}, entonces
$\operatorname{Gal}(L/K)$ es un \omterm{def:b3:groups:derived}{grupo
resoluble}. (El recíproco también es cierto; no lo
necesitaremos.)\index{teorema de resolubilidad de Galois}
\end{theorem}

\begin{proof}
Paso 1: \emph{agrandar la torre \omterm{def:b3:galois:radical}{radical}
hasta hacerla \omterm{def:b3:galois:galois}{de Galois}.} Sea $L
\subseteq M$ con $M/K$ \omterm{def:b3:galois:radical}{radical}, de
exponentes de radicación $n_1, \dots,
n_r$ y $n = n_1\cdots n_r$. Adjuntemos primero $\zeta_n$: la torre
$K \subseteq K(\zeta_n) \subseteq M(\zeta_n)$ sigue siendo
\omterm{def:b3:galois:radical}{radical} ($\zeta_n$ es una raíz de la
unidad: un paso \omterm{def:b3:galois:radical}{radical},
$\zeta_n^n = 1$) y sus pasos posteriores al primero ocurren sobre
cuerpos que contienen las raíces de la unidad necesarias. Sustitúyase
después $M(\zeta_n)$ por el compuesto $N$ de todos los
$\sigma(M(\zeta_n))$, donde $\sigma$ recorre las (finitas)
$K$-inmersiones de $M(\zeta_n)$ en una
\omterm{def:b3:galois:closure}{clausura algebraica} fija: $N$ es el
\omterm{thm:b3:galois:splitting}{cuerpo de descomposición} del producto
de los \omterm{def:b3:galois:algebraic}{polinomios mínimos} de un
sistema generador (característica $0$: finita y
\omterm{def:b3:galois:separable}{separable}), luego $N/K$ es
\omterm{def:b3:galois:galois}{de Galois}; y $N$ es
\omterm{def:b3:galois:radical}{radical} sobre $K$: cada
$\sigma(M(\zeta_n))$ es \omterm{def:b3:galois:radical}{radical} sobre
$K$ (aplíquese $\sigma$ a una torre
\omterm{def:b3:galois:radical}{radical}), y un compuesto de extensiones
\omterm{def:b3:galois:radical}{radicales} es
\omterm{def:b3:galois:radical}{radical} (concaténense las torres: si
$F'/K$ es \omterm{def:b3:galois:radical}{radical} con torre que adjunta
$\beta_j$, los pasos de tipo $F''(\beta_j)$ siguen siendo
\omterm{def:b3:galois:radical}{radicales} sobre cualquier base mayor).

Paso 2: \emph{leer la \omterm{def:b3:groups:derived}{resolubilidad} en
la torre \omterm{def:b3:galois:galois}{de Galois}.} Supongamos, pues,
$L \subseteq N$ con $N/K$ \omterm{def:b3:galois:galois}{de Galois} y
\omterm{def:b3:galois:radical}{radical}, con torre $K
\subseteq K(\zeta_n) = E_0 \subseteq E_1 \subseteq \dots
\subseteq E_s = N$ y cada $E_{i+1} = E_i(\sqrt[n_i]{a_i})$ con
$\zeta_{n_i} \in E_0 \subseteq E_i$. Sean $G =
\operatorname{Gal}(N/K)$ y $G_i = \operatorname{Gal}(N/E_i)$: una cadena
decreciente $G \supseteq G_0 \supseteq G_1 \supseteq
\dots \supseteq G_s = \{e\}$. Cada $E_{i+1}/E_i$ es
\omterm{def:b3:galois:galois}{de Galois} con grupo cíclico
(el~\cref{lem:b3:galois:cyclicsteps}), luego, por el teorema fundamental
aplicado a la \omterm{def:b3:galois:galois}{extensión de Galois} $N/E_i$
(el~\cref{thm:b3:galois:fundamental}(3), con grupo ambiente $G_i$):
$G_{i+1} \trianglelefteq G_i$ con $G_i/G_{i+1} \cong
\operatorname{Gal}(E_{i+1}/E_i)$ cíclico. Análogamente, $E_0/K$ es
\omterm{def:b3:galois:galois}{de Galois} con grupo abeliano $G/G_0$
(el~\cref{lem:b3:galois:cyclicsteps}). La cadena exhibe $G$ como
\omterm{def:b3:groups:derived}{resoluble}
(la~\cref{prop:b3:groups:derived}). Por último,
$\operatorname{Gal}(L/K)$ es un \emph{cociente} de $G$: $L/K$ es
\omterm{def:b3:galois:galois}{de Galois} ($P$ es
\omterm{def:b3:galois:separable}{separable} en característica $0$) y la
restricción $G
\to \operatorname{Gal}(L/K)$ es sobreyectiva
(el~\cref{thm:b3:galois:fundamental}(3) con $H =
\operatorname{Gal}(N/L)$); y los cocientes de
\omterm{def:b3:groups:derived}{grupos resolubles} son
\omterm{def:b3:groups:derived}{resolubles}.
\end{proof}

\begin{corollary}[Insolubilidad de la ecuación de quinto grado]
\label{cor:b3:galois:quintic}
Hay polinomios de \omterm{def:b3:galois:extension}{grado} $5$ sobre $\Q$
que no son \omterm{def:b3:galois:radical}{resolubles por radicales}: por
ejemplo $X^5 - 4X + 2$, cuyo \omterm{def:b3:galois:galois}{grupo de
Galois} es $S_5$ (el~\cref{exo:b3:galois:11}), un grupo
no \omterm{def:b3:groups:derived}{resoluble}
(el~\cref{cor:b3:groups:snnotsolvable}). No puede existir ninguna
fórmula general por \omterm{def:b3:galois:radical}{radicales} para el
\omterm{def:b3:galois:extension}{grado} $\geq 5$.\index{ecuación de
quinto grado, insolubilidad}
\end{corollary}

\begin{remark}\label{rem:b3:galois:converse}
El recíproco del~\cref{thm:b3:galois:solvable} --- que un
\omterm{def:b3:galois:galois}{grupo de Galois}
\omterm{def:b3:groups:derived}{resoluble} implica la
\omterm{def:b3:groups:derived}{resolubilidad} por
\omterm{def:b3:galois:radical}{radicales} --- se demuestra descendiendo
por la \omterm{def:b3:groups:derived}{serie derivada} y viendo que toda
extensión cíclica (con suficientes raíces de la unidad) es
\omterm{def:b3:galois:radical}{radical}, mediante las \emph{resolventes
de Lagrange}; explica por qué los
\omterm{def:b3:galois:extension}{grados} $2, 3, 4$ tienen fórmulas:
$S_2, S_3, S_4$ son \omterm{def:b3:groups:derived}{resolubles}
(el~\cref{ex:b3:groups:solvableexamples}). Lo dejamos admitido a este
nivel; un tratamiento completo corresponde a un curso de máster, pero
el~\cref{exo:b3:galois:8} lo hace concreto para la cúbica.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b3:galois:1}
Demostrar que $[\Q(\sqrt2, \sqrt3):\Q] = 4$, que $\Q(\sqrt2 + \sqrt3) =
\Q(\sqrt2, \sqrt3)$, y calcular el
\omterm{def:b3:galois:algebraic}{polinomio mínimo} de $\sqrt2 + \sqrt3$
sobre $\Q$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:galois:2}
Sea $\alpha = \sqrt[3]2$. Demostrar que $\Q(\alpha)/\Q$ no es normal
(exhíbase una inmersión $\Q(\alpha) \to \C$ cuya imagen no sea
$\Q(\alpha)$), determinar el \omterm{thm:b3:galois:splitting}{cuerpo de
descomposición} $L$ de $X^3 -
2$ y $[L:\Q]$, y comprobar que $\operatorname{Aut}_\Q(\Q(\alpha)) =
\{\mathrm{id}\}$: para las extensiones que
no son \omterm{def:b3:galois:galois}{de Galois}, el grupo de automorfismos puede ser mucho menor que
el \omterm{def:b3:galois:extension}{grado}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:galois:3}
Construir $\mathbb F_9$ como $\mathbb F_3[X]/(X^2+1)$ y hallar un
generador de $\mathbb F_9^\times$. Listar los polinomios mónicos
\omterm{def:b3:rings:divisibility}{irreducibles} de
\omterm{def:b3:galois:extension}{grados} $1, 2, 3$ sobre $\mathbb F_2$,
y comprobar que $X^8 - X = X(X+1)(X^3+X+1)(X^3+X^2+1)$ sobre
$\mathbb F_2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:4}
(a) Hallar todas las raíces primitivas módulo $7$ y módulo $11$ (es
decir, los generadores de $\mathbb F_7^\times$,
$\mathbb F_{11}^\times$). (b) Demostrar que, para $p$ impar,
$x \in \mathbb F_p^\times$ es un cuadrado si y solo si $x^{(p-1)/2} = 1$
(criterio de Euler), y recuperar el criterio para $-1$
del~\cref{pb:b3:rings:1}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:5}
Demostrar que $\mathbb F_{p^m} \cap \mathbb F_{p^n} = \mathbb
F_{p^{\gcd(m,n)}}$ y $\mathbb F_{p^m}\mathbb F_{p^n} = \mathbb
F_{p^{\operatorname{lcm}(m,n)}}$ dentro de una
\omterm{def:b3:galois:closure}{clausura algebraica} fija
$\bar{\mathbb F}_p$, y describir
$\operatorname{Gal}(\mathbb F_{p^n}/\mathbb F_{p^m})$ para $m
\mid n$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:6}
Sea $I_d(q)$ el número de polinomios mónicos
\omterm{def:b3:rings:divisibility}{irreducibles} de
\omterm{def:b3:galois:extension}{grado} $d$ sobre $\mathbb F_q$.
Demostrar
\[
X^{q^n} - X \;=\; \prod_{d \mid n}\ \prod_{P \text{ irred.
monic, } \deg P = d} P ,
\qquad\text{luego}\qquad
q^n = \sum_{d \mid n} d\, I_d(q).
\]
Deducir $I_1, I_2, I_3, I_4$ explícitamente, y que $I_d(q) \geq 1$ para
todo $d$ (de modo que las extensiones $\mathbb F_{q^d}/\mathbb F_q$
existen como cocientes $\mathbb F_q[X]/(P)$ para todo $d$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:7}
Determinar $\operatorname{Gal}(\Q(\sqrt2,\sqrt3)/\Q)$ y el retículo
completo de cuerpos intermedios. La misma pregunta para el
\omterm{thm:b3:galois:splitting}{cuerpo de descomposición} de
$(X^2-2)(X^2-3)(X^2-6)$ --- ¿qué se observa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:8}
(La cúbica, resuelta por su grupo) Sea $P = X^3 + pX + q \in
\Q[X]$ \omterm{def:b3:rings:divisibility}{irreducible} con raíces
$x_1, x_2, x_3$ y \omterm{thm:b3:galois:splitting}{cuerpo de
descomposición} $L$. Sean $\delta = (x_1 - x_2)(x_1 - x_3)(x_2 - x_3)$ y
$\Delta = \delta^2 = -4p^3 - 27q^2$ (admítase esta identidad clásica o
compruébese desarrollando funciones simétricas). (a) Demostrar que
$\operatorname{Gal}(L/\Q) \cong A_3$ o $S_3$, según que $\Delta$ sea o
no un cuadrado en $\Q$. (b) Con $j = \zeta_3$, defínanse las resolventes
de Lagrange $u = x_1
+ jx_2 + j^2x_3$ y $v = x_1 + j^2x_2 + jx_3$. Demostrar que $u^3 + v^3
= -27q$ y $uv = -3p$, y despejar $u^3, v^3$: las fórmulas de Cardano
salen solas. ¿Dónde se ha usado la
\omterm{def:b3:groups:derived}{resolubilidad} de $S_3$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:9}
En $\Q(\zeta_5)$: demostrar que el único subcuerpo cuadrático es
$\Q(\sqrt5)$, mediante las sumas de Gauss
$\eta_0 = \zeta_5 + \zeta_5^4$, $\eta_1 = \zeta_5^2 + \zeta_5^3$:
calcúlense $\eta_0 + \eta_1$ y $\eta_0\eta_1$ y dedúzcase
$\cos\frac{2\pi}5 = \frac{\sqrt5 -
1}4$. Concluir que el pentágono regular es
\omterm{def:b3:galois:constructible}{construible}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:galois:10}
Sean $K = \mathbb F_p(S, T)$ (funciones racionales en dos
indeterminadas) y $L = K(S^{1/p}, T^{1/p})$. (a) Demostrar que
$[L:K] = p^2$ y que $\alpha^p \in K$ para todo
$\alpha \in L$.
(b) Deducir que $L/K$ \emph{no} es simple: no existe elemento primitivo
--- la inseparabilidad es fatal para
\cref{thm:b3:galois:primitive}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:galois:11}
Sean $P = X^5 - 4X + 2$ y $G$ su \omterm{def:b3:galois:galois}{grupo de
Galois} sobre $\Q$, que actúa sobre las $5$ raíces. (a) Demostrar que
$P$ es \omterm{def:b3:rings:divisibility}{irreducible} y deducir
$5 \mid \abs G$; concluir que $G$ contiene un $5$-ciclo (Cauchy,
\cref{thm:b3:groups:cauchy}).
(b) Demostrar, estudiando las variaciones de $x \mapsto x^5 - 4x +
2$, que $P$ tiene exactamente $3$ raíces reales; deducir que la
conjugación compleja se restringe a una trasposición de $G$. (c)
Demostrar que un subgrupo de $S_5$ que contiene una trasposición y un
$5$-ciclo es $S_5$ \emph{(conjúguese la trasposición por potencias del
ciclo)}. Concluir que $G \cong S_5$ y, con
el~\cref{thm:b3:galois:solvable}, que $P$ no es
\omterm{def:b3:galois:radical}{resoluble por radicales}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:galois:12}
(La cuártica diédrica) Sean $\alpha = \sqrt[4]2$ y $L =
\Q(\alpha, \iu)$, el \omterm{thm:b3:galois:splitting}{cuerpo de
descomposición} de $X^4 - 2$ sobre $\Q$. (a) Demostrar que
$[L : \Q] = 8$ y que $G =
\operatorname{Gal}(L/\Q)$ está generado por $\sigma\colon
\alpha \mapsto \iu\alpha,\ \iu \mapsto \iu$ y la conjugación compleja
$\tau$, con $\sigma^4 = \tau^2 = e$ y
$\tau\sigma\tau = \sigma^{-1}$: $G \cong D_4$.
(b) Listar el retículo de subgrupos de $D_4$ (diez subgrupos) y
emparejar cada uno con su cuerpo fijo; compruébese en particular que
$\Q(\sqrt2)$, $\Q(\iu)$, $\Q(\iu\sqrt2)$ son los tres subcuerpos
cuadráticos, y localícense $\Q(\alpha)$,
$\Q(\iu\alpha)$, $\Q(\sqrt2, \iu)$.
(c) ¿Qué cuerpos intermedios son \omterm{def:b3:galois:galois}{de
Galois} sobre $\Q$? Contrástese la respuesta con los
\omterm{def:b3:groups:normal}{subgrupos normales} de $D_4$ y explíquese
por qué $\Q(\alpha)/\Q$ falla mientras que $\Q(\sqrt2)/\Q$ funciona.
\end{exercise}

\section{Problema: Gauss y el 17-ágono regular}

\begin{problem}[{Problema de fin de semana --- constructibilidad del
17-ágono}]\label{pb:b3:galois:1} El 30 de marzo de 1796, Gauss, con
diecinueve años, demostró que el $17$-ágono regular es
\omterm{def:b3:galois:constructible}{construible} --- el primer avance
en la cuestión desde la Antigüedad. Reconstruimos su cálculo con las
herramientas de este capítulo. Póngase $\zeta = \eu^{2\iu\pi/17}$, $L =
\Q(\zeta)$, $G = \operatorname{Gal}(L/\Q)$.

\medskip
\noindent\textbf{Parte I --- El grupo y su filtración.}
\begin{enumerate}
  \item Justificar: $[L:\Q] = 16$, $G \cong (\Z/17\Z)^\times$, cíclico
  de orden $16$. Comprobar que $3$ es un generador de $(\Z/17\Z)^\times$
  \emph{(calcúlense las potencias de $3$ módulo $17$:
  $3, 9, 10, 13, 5, 15, 11, 16, \dots$)}. \item Sea $\sigma \in G$ con
  $\sigma(\zeta) = \zeta^3$, y $H_k = \langle \sigma^{2^k}\rangle$ para
  $k = 0, \dots,
        4$. Demostrar que $G = H_0 \supset H_1 \supset H_2 \supset
        H_3 \supset H_4 = \{e\}$ con cada índice $[H_k :
        H_{k+1}] = 2$, y que los cuerpos fijos $\Q = L_0
        \subset L_1 \subset L_2 \subset L_3 \subset L_4 = L$ forman una
  torre de extensiones cuadráticas. \item Concluir \emph{a priori},
  usando el~\cref{thm:b3:galois:wantzel}, que $\zeta$ --- y por tanto el
  $17$-ágono --- es \omterm{def:b3:galois:constructible}{construible}.
  El resto del problema hace explícita la torre.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Los períodos de longitud 8.} Definamos los
\emph{períodos de Gauss}
\[
\eta_0 = \sum_{k \text{ par}} \zeta^{3^k \bmod 17}
= \zeta^{1} + \zeta^{9} + \zeta^{13} + \zeta^{15} + \zeta^{16} +
\zeta^{8} + \zeta^{4} + \zeta^{2},
\qquad
\eta_1 = \sum_{k \text{ impar}} \zeta^{3^k \bmod 17}.
\]
\begin{enumerate}[resume]
  \item Demostrar que $\eta_0, \eta_1$ quedan fijos por $H_1$ y se
  intercambian por $\sigma$; deducir $\eta_0, \eta_1 \in L_1$ y que son
  las dos raíces de una ecuación de segundo \omterm{def:b3:galois:extension}{grado} sobre $\Q$. \item
  Calcular $\eta_0 + \eta_1 = -1$. Demostrar que $\eta_0\eta_1 = -4$
  \emph{(cada producto $\zeta^a\zeta^b$ es algún $\zeta^c$, $c \neq 0$;
  cuéntese cuántas veces aparece cada $c$, o arguméntese que el producto
  es un entero racional fijado por $G$, igual a la suma de los $64$
  productos, y úsese que cada resto no nulo aparece el mismo número de
  veces)}. \item Deducir $\eta_0 = \frac{-1 + \sqrt{17}}2$, $\eta_1 =
        \frac{-1-\sqrt{17}}2$ (identifíquese cuál es cuál numéricamente:
  $\eta_0 \approx 1.56$) y $L_1 =
        \Q(\sqrt{17})$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Períodos de longitud 4 y 2.} Definamos
\[
\beta_0 = \zeta + \zeta^{13} + \zeta^{16} + \zeta^{4},\quad
\beta_1 = \zeta^3 + \zeta^5 + \zeta^{14} + \zeta^{12},\quad
\beta_2 = \zeta^9 + \zeta^{15} + \zeta^{8} + \zeta^{2},\quad
\beta_3 = \zeta^{10} + \zeta^{11} + \zeta^{7} + \zeta^{6}.
\]
\begin{enumerate}[resume]
  \item Demostrar que $\beta_0 + \beta_2 = \eta_0$, $\beta_1 + \beta_3 =
        \eta_1$, y que $\beta_0, \beta_2$ quedan fijos por $H_2$ y se
  intercambian por $\sigma^2$. \item Calcular $\beta_0\beta_2 = -1$ y
  $\beta_1\beta_3 = -1$ \emph{(desarróllese: los dieciséis exponentes
  obtenidos recorren $1, \dots, 16$ exactamente una vez)}. \item Deducir
  $\beta_0 = \frac{\eta_0 + \sqrt{\eta_0^2 + 4}}2$ (compruébese el signo
  numéricamente: $\beta_0 \approx 2.05$) y la fórmula análoga para
  $\beta_1$; de ahí $L_2 =
        \Q(\beta_0)$, cuadrático sobre $L_1$. \item Sean
  $\gamma_0 = \zeta + \zeta^{16} =
        2\cos\frac{2\pi}{17}$ y $\gamma_1 = \zeta^{13} +
        \zeta^4$. Demostrar que $\gamma_0 + \gamma_1 = \beta_0$ y
  $\gamma_0\gamma_1 = \beta_1$, de modo que $\gamma_0 =
        \frac{\beta_0 + \sqrt{\beta_0^2 - 4\beta_1}}2$. \item Reunir la
  cadena de fórmulas que expresa $\cos\frac{2\pi}{17}$ mediante
  \omterm{def:b3:galois:radical}{radicales} cuadráticos anidados, y dar una comprobación decimal
  ($\cos\frac{2\pi}{17} \approx 0.93247$).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Epílogo.}
\begin{enumerate}[resume]
  \item ¿Dónde se ha usado exactamente que $17$ es un \emph{primo de
  Fermat} ($17 = 2^{2^2} + 1$)? Demostrar que, para un primo $p$, el
  $p$-ágono regular es \omterm{def:b3:galois:constructible}{construible}
  si y solo si $p
        = 2^{2^t} + 1$ para algún $t$ \emph{(si $p - 1 = 2^m$,
  demuéstrese que $m$ ha de ser a su vez una potencia de $2$)}. \item
  Deducir la lista completa de los $n$-ágonos regulares
  \omterm{def:b3:galois:constructible}{construibles} para $n \leq 20$,
  usando el criterio de Gauss--Wantzel
  del~\cref{cor:b3:galois:impossible}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Sumas de Gauss y reciprocidad cuadrática.}
Los períodos de la Parte II esconden un tesoro. Para un primo impar $p$,
el \emph{símbolo de Legendre} $\bigl(\frac
ap\bigr)$ vale $+1$ si $a$ es un cuadrado no nulo módulo $p$, $-1$ si no
lo es y $0$ si $p \mid a$; el~\cref{exo:b3:galois:4}(b) (criterio de
Euler) da $\bigl(\frac ap\bigr) \equiv
a^{(p-1)/2} \pmod p$, de donde la multiplicatividad. Escribamos $\zeta =
\eu^{2\iu\pi/p}$, $p^* = (-1)^{(p-1)/2}p$, y definamos la \emph{suma de
Gauss}
\[
g \;=\; \sum_{a=1}^{p-1}\Bigl(\frac ap\Bigr)\zeta^a .
\]
\begin{enumerate}[resume]
  \item Demostrar que $\sum_{a=1}^{p-1}\bigl(\frac ap\bigr) = 0$ (hay
  tantos cuadrados como no cuadrados) y probar la forma alternativa
  $g = \sum_{a=0}^{p-1}\zeta^{a^2}$ \emph{(cada cuadrado no nulo se
  alcanza dos veces, y $\sum_{a}\zeta^a = 0$)}. Para $p = 17$:
  relaciónese $g$ con los períodos de la Parte II --- demuéstrese que
  $g = \eta_0 - \eta_1$ \emph{(los cuadrados módulo $17$ son exactamente
  las potencias pares del generador $3$)}. \item Demostrar que
  $g^2 = p^*$: desarróllese
        \[
        g^2 = \sum_{a,b\neq0}\Bigl(\frac{ab}p\Bigr)
        \zeta^{a+b}
        = \sum_{c}\ \sum_{a \neq 0}\Bigl(\frac{a(c -
        a)}p\Bigr)\zeta^{c}
        \]
        (póngase $b = c - a$), sustitúyase $c - a = at$ para evaluar la
        suma interior como $\bigl(\frac{-1}p\bigr)(p - 1)$ si $c = 0$ y
        $-\bigl(\frac{-1}p\bigr)$ en caso contrario, y concluir con la
        pregunta 14. Compruébese numéricamente: para $p =
        17$, $(\eta_0 - \eta_1)^2 = 17$ (Parte II). \item Deducir
        $\sqrt{p^*} \in \Q(\zeta_p)$ y concluir que el \emph{único}
        subcuerpo cuadrático de $\Q(\zeta_p)$ es $\Q(\sqrt{p^*})$ ---
        único porque $\operatorname{Gal}(\Q(\zeta_p)/\Q)$ es cíclico
        (el~\cref{thm:b3:galois:cyclotomicirred}) y un grupo cíclico
        tiene exactamente un subgrupo de índice $2$. (Todo cuerpo
        cuadrático se sumerge en algún cuerpo ciclotómico --- este es el
        primer caso del teorema de Kronecker--Weber, cuya forma general
        queda muy lejos.) \item Sea ahora $q \neq p$ otro primo impar.
        Trabajando en el anillo $\Z[\zeta]$ módulo $q$, demostrar que
        \[
        g^q \equiv \Bigl(\frac qp\Bigr)\,g \pmod{q\Z[\zeta]}
        \]
        \emph{(sueño del principiante: $(x + y)^q \equiv x^q + y^q$
        módulo $q$ en todo anillo conmutativo; después $g^q \equiv
        \sum_a\bigl(\frac ap\bigr)^q\zeta^{aq}$, reindéxese $b =
        aq$ y sáquese $\bigl(\frac{q^{-1}}p\bigr) =
        \bigl(\frac qp\bigr)$)}. \item Por otra parte,
        $g^q = g\,(g^2)^{(q-1)/2} =
        g\,(p^*)^{(q-1)/2}$; usando el criterio de Euler módulo $q$,
        dedúzcase $g^q \equiv \bigl(\frac{p^*}q\bigr)g
        \pmod{q\Z[\zeta]}$ y después --- multiplicando las dos
        expresiones de $g^q$ por $g$ y usando $g^2 = p^*$, invertible
        módulo $q$ --- concluir
        \[
        \Bigl(\frac qp\Bigr) = \Bigl(\frac{p^*}q\Bigr) .
        \]
        \emph{(¿Por qué una congruencia entre los enteros $\pm p^*$
        módulo $q\Z[\zeta]$ implica su igualdad? Interséquese con
        $\Z$.)} \item Desarrollar $\bigl(\frac{p^*}q\bigr) =
        \bigl(\frac{-1}q\bigr)^{(p-1)/2}\bigl(\frac pq\bigr)$ y
        $\bigl(\frac{-1}q\bigr) = (-1)^{(q-1)/2}$ para obtener la
        \emph{ley de reciprocidad cuadrática}:
        \[
        \Bigl(\frac pq\Bigr)\Bigl(\frac qp\Bigr)
        = (-1)^{\frac{p-1}2\cdot\frac{q-1}2} .
        \]
        Compruébese en $(p, q) = (17, 3)$ listando los cuadrados módulo
        $17$ y módulo $3$, y úsese para decidir en tres líneas si
        $x^2 \equiv 219 \pmod{383}$ es
        \omterm{def:b3:groups:derived}{resoluble} ($383$ es primo,
        $219 = 3\cdot73$).
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Contar polinomios
\omterm{def:b3:rings:divisibility}{irreducibles}: el teorema de los
números primos de $\mathbb F_q[X]$.} Fíjese una potencia de primo $q$ y
sea $N_q(n)$ el número de polinomios \emph{mónicos
\omterm{def:b3:rings:divisibility}{irreducibles}} de
\omterm{def:b3:galois:extension}{grado} $n$ sobre $\mathbb F_q$;
recuérdense del~\cref{exo:b3:galois:6} la factorización de $X^{q^n} - X$
y la identidad $q^n = \sum_{d\mid n}d\,N_q(d)$, que ahora invertiremos,
reinterpretaremos y explotaremos.
\begin{enumerate}[resume]
  \item (Palabras) Diremos que una palabra $w \in \mathbb F_q^n$ es
  \emph{primitiva} si no es una potencia $u^{n/d} =
        u\cdots u$ de una palabra $u$ estrictamente más corta, y sea
  $A(d)$ el número de palabras primitivas de longitud $d$. Demostrar que
  toda palabra de longitud $n$ es, de manera única, potencia de una
  palabra primitiva de cierta longitud $d \mid n$, de modo que
  $q^n = \sum_{d \mid n}A(d)$; comparando con el~\cref{exo:b3:galois:6},
  concluir $A(d) = d\,N_q(d)$ para todo $d$ y explicar esta coincidencia
  mediante una biyección explícita: un elemento $\alpha \in
        \mathbb F_{q^d}$ de \omterm{def:b3:galois:extension}{grado} $d$
  tiene \omterm{def:b3:groups:action}{órbita}
  $(\alpha, \alpha^q, \dots, \alpha^{q^{d-1}})$ bajo el
  \omterm{thm:b3:galois:finitefields}{Frobenius} con exactamente $d$
  elementos distintos, y los elementos de
  \omterm{def:b3:galois:extension}{grado} $d$ se corresponden $d$ a uno
  con los \omterm{def:b3:rings:divisibility}{irreducibles} de
  \omterm{def:b3:galois:extension}{grado}
        $d$.
  \item Demostrar la \emph{fórmula de inversión de Möbius}: si $f(n) =
        \sum_{d\mid n}g(d)$ para todo $n$, entonces $g(n) =
        \sum_{d\mid n}\mu(d)\,f(n/d)$, donde $\mu$ es la función de
  Möbius ($\mu(m) = (-1)^{\#\text{prime
        factors}}$ si $m$ es libre de cuadrados y $0$ en caso contrario)
  \emph{(lema clave: $\sum_{d \mid m}\mu(d) = 0$ para $m >
        1$ --- emparéjense los divisores con y sin un factor primo
  fijo)}. Deducir
        \[
        N_q(n) = \frac1n\sum_{d \mid n}\mu(d)\,q^{n/d} .
        \]
  \item Demostrar que
  $N_q(n) \geq \frac1n\bigl(q^n - 2q^{n/2}\bigr) > 0$ para todo
  $n \geq 1$: una nueva demostración de que $\mathbb F_{q^n}$ existe
  para todo $n$. Interprétese el término principal: un polinomio mónico
  de \omterm{def:b3:galois:extension}{grado} $n$ tomado al azar es
  \omterm{def:b3:rings:divisibility}{irreducible} con probabilidad
  $\sim \frac1n$ --- el análogo
  \omterm{prop:b3:galois:perfect}{perfecto} del teorema de los números
  primos, con $\log x$ sustituido por $n$; compruébese numéricamente
  para $q = 2$, $n \leq 4$ (el~\cref{exo:b3:galois:6} da los recuentos).
  \item Demostrar el análogo multiplicativo de la pregunta 21:
        \[
        \prod_{\substack{\pi \text{ mónico irred.}\\ \deg\pi =
        n}}\pi \;=\; \prod_{d \mid n}
        \bigl(X^{q^d} - X\bigr)^{\mu(n/d)}
        \]
        \emph{(inversión de Möbius en el grupo abeliano de las funciones
        racionales no nulas)}; compruébese a mano para $q
        = 2$, $n = 2$: $(X^4 - X)/(X^2 - X) = X^2 + X + 1$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- Dos codas.}
\begin{enumerate}[resume]
  \item (El segundo suplemento) El método de la Parte V permite calcular
  también $\bigl(\frac2q\bigr)$. Sean $\omega = \eu^{2\iu\pi/8}$ y
  $g = \omega + \omega^{-1}$. Demostrar que $g^2 = 2$
  \emph{($\omega^2 = \iu$)}; después, para un primo impar $q$, demostrar
  en $\Z[\omega]$ módulo $q$ que
        \[
        g^q \equiv \omega^q + \omega^{-q} \pmod{q\Z[\omega]},
        \]
        y que el miembro de la derecha vale $g$ si $q \equiv \pm1
        \pmod 8$ y $-g$ si $q \equiv \pm3 \pmod 8$. Comparando con
        $g^q = g\,(g^2)^{(q-1)/2} \equiv
        \bigl(\frac2q\bigr)g$ como en la pregunta 18, concluir
        \[
        \Bigl(\frac2q\Bigr) = (-1)^{(q^2-1)/8},
        \]
        comprobando que $(q^2 - 1)/8$ es par exactamente cuando $q
        \equiv \pm1 \pmod 8$. Verifíquese: $2$ es un cuadrado módulo $7$
        y módulo $17$ ($3^2$ y $6^2$), y no lo es módulo $3$ ni módulo
        $5$.
  \item (La función zeta de $\mathbb F_q[X]$) Demostrar la identidad de
  series formales de potencias en $t$:
        \[
        \prod_{n \geq 1}\bigl(1 - t^n\bigr)^{-N_q(n)}
        = \frac1{1 - qt}
        \]
        \emph{(factorización única en
        \omterm{def:b3:rings:divisibility}{irreducibles} mónicos:
        desarróllese cada factor como serie geométrica y cuéntense los
        polinomios mónicos de \omterm{def:b3:galois:extension}{grado}
        $n$)}. Recupérese la identidad $q^m = \sum_{d \mid m}d\,N_q(d)$
        del~\cref{exo:b3:galois:6} tomando logaritmos. Compruébese a
        mano el coeficiente de $t^2$ para $q = 2$, y úsese la fórmula de
        la pregunta 21 para calcular $N_2(6) = 9$, verificando que
        $2^6 = 1\cdot2 + 2\cdot1 + 3\cdot2 +
        6\cdot9$.
\end{enumerate}
\end{problem}
