\chapter{फूरिये रूपांतर}\label{ch:b3:fouriertransform}

फूरिये श्रेणी आवर्ती संकेतों को विविक्त संनादों में विघटित करती
है; फूरिये \emph{रूपांतर} पूरी रेखा पर के संकेतों के लिए वही करता
है, और वह भी आवृत्तियों के एक सातत्य के साथ। वह अवकलन को गुणन में,
संवलन को गुणनफल में, और गाउसीय को गाउसीय में बदल देता है --- यही
कारण है कि वह अवकल समीकरण हल करता है, संकेत संसाधन चलाता है, और
\cref{ch:b3:clt} में केंद्रीय सीमा प्रमेय सिद्ध करेगा। यह अध्याय $L^1$ सिद्धांत
(रीमान--लेबेग, प्रतिलोमन, एकैकीपन), वह \omterm{def:b3:fouriertransform:schwartz}{श्वार्ट्स वर्ग} जहाँ रूपांतर
एक सिद्ध एकैकी आच्छादक प्रतिचित्रण है, और $L^2$ सिद्धांत (प्लांशरेल:
रूपांतर, किसी नियतांक तक, एक एकात्मक संकारक है) विकसित करता है,
साथ में दो प्रदर्शनीय अनुप्रयोग: ऊष्मा समीकरण, जिसे सप्ताहांत
समस्या आरंभ से अंत तक हल करती है, और प्वासों योग सूत्र। परिपाटी:
\[
\hat f(\xi) = \int_\R f(x)\,\eu^{-\iu\xi x}\,\dd x .
\]

\section{\texorpdfstring{$L^1$}{L1} पर रूपांतर}

\begin{proposition}\label{prop:b3:fouriertransform:basic}
$f \in L^1(\R)$ के लिए: $\hat f$ सुपरिभाषित, परिबद्ध ($\norm{\hat f}_\infty \leq \norm f_1$) और \omterm{def:b3:topology:continuity}{संतत} है, तथा:
\begin{enumerate}
  \item $\widehat{\tau_af}(\xi) = \eu^{-\iu a\xi}\hat f(\xi)$ और $\widehat{\eu^{\iu ax}f}(\xi) = \hat f(\xi - a)$;
  \item $\lambda > 0$ के लिए $\widehat{f(\cdot/\lambda)}(\xi) = \lambda\hat
        f(\lambda\xi)$;
  \item यदि $xf \in L^1$ हो, तो $\hat f$ $(\hat f)'(\xi) = \widehat{(-\iu x)f}(\xi)$ के साथ $\mathcal C^1$ है;
  \item यदि $f' \in L^1$ के साथ $f \in \mathcal C^1$ हो (और $\pm\infty$ पर $f \to
        0$, जो यहाँ स्वतः
        है), तो $\widehat{f'}(\xi) = \iu\xi\hat f(\xi)$;
  \item $f, g \in L^1$ के लिए $\widehat{f * g} = \hat f\,\hat g$।
\end{enumerate}
\end{proposition}

\begin{proof}
परिबद्धता: $\abs{\hat f} \leq \int\abs f$। \omterm{def:b3:topology:continuity}{सांतत्य}: प्रभावी फलन $\abs f$ के साथ प्रभावी अभिसरण
प्रमेय (\cref{thm:b3:lebesgue:paramcont})। (1), (2): प्रतिस्थापन (\cref{thm:b3:product:linearchange})। (3): समाकल के भीतर
अवकलन, प्रभावी फलन $\abs{xf}$ (\cref{thm:b3:lebesgue:paramdiff})। (4): पहले, $f(x) = f(0)
+ \int_0^xf'$ की $\pm\infty$ पर कोई
सीमा है ($f' \in L^1$), जो $0$ होनी ही चाहिए ($f \in L^1$); फिर $[-A,
A]$ पर खंडशः
समाकलन कीजिए और $A \to \infty$ जाने दीजिए। (5): फुबिनी, जो वैध है क्योंकि
$(x,y)\mapsto f(x - y)g(y)\eu^{-\iu\xi x}$ निरपेक्षतः \omterm{def:b3:lebesgue:l1}{समाकलनीय} है (\cref{thm:b3:product:convolution}):
\[
\widehat{f*g}(\xi) = \iint f(x - y)g(y)\eu^{-\iu\xi(x - y)}
\eu^{-\iu\xi y}\dd x\,\dd y = \hat f(\xi)\,\hat g(\xi).
\]
\end{proof}

\begin{example}\label{ex:b3:fouriertransform:gaussian}
गाउसीय: $a > 0$ के लिए,\index{गाउसीय, फूरिये रूपांतर}
\[
\widehat{\eu^{-ax^2}}(\xi) =
\sqrt{\frac\pi a}\;\eu^{-\xi^2/4a} :
\]
यह \cref{exo:b3:lebesgue:7} से मिलता है (अवकल समीकरण $F' = -\frac\xi{2}F$ की युक्ति, पुनःमापित), अथवा
(3) से: $g = \widehat{\eu^{-ax^2}}$ $g'(\xi) = -\frac{\xi}{2a}g(\xi)$ को संतुष्ट करता है (खंडशः समाकलन), और $g(0) = \sqrt{\pi/a}$।
गाउसीय मापन तक रूपांतर के स्थिर बिंदु हैं --- और यही गहरा कारण है
कि वे केंद्रीय सीमा प्रमेय पर राज करते हैं।
\end{example}

\begin{theorem}[रीमान--लेबेग]
\label{thm:b3:fouriertransform:riemannlebesgue}
$f \in L^1(\R)$ के लिए: $\abs\xi \to
\infty$ पर $\hat f(\xi) \to 0$। अतः $\widehat{\phantom f} \colon L^1 \to \mathcal
C_0(\R)$ (अनंत पर लुप्त होने वाले \omterm{def:b3:topology:continuity}{संतत}
फलन)।\index{रीमान--लेबेग प्रमेयिका}
\end{theorem}

\begin{proof}
किसी अंतराल के सूचक के लिए $\widehat{\mathbf 1_{\intcc
ab}}(\xi) = \frac{\eu^{-\iu a\xi} - \eu^{-\iu b\xi}}{\iu\xi}
\to 0$; अतः सोपान फलनों के लिए भी। सोपान
फलन $L^1$ में सघन हैं (\cref{thm:b3:lp:density}(1) और परिमित \omterm{def:b3:measure:measure}{माप} वाले समुच्चयों का
अंतरालों के परिमित सम्मिलनों से आसन्नन, \cref{exo:b3:measure:7}), और रूपांतर
$\norm\cdot_\infty$-$\norm\cdot_1$ \omterm{def:b3:topology:continuity}{संतत} है: $\norm{f -
s}_1 < \varepsilon$ के लिए $\limsup_{\abs\xi\to\infty}\abs{\hat
f(\xi)} \leq \varepsilon$।
\end{proof}

\section{प्रतिलोमन और एकैकीपन}

\begin{lemma}[गुणन सूत्र]
\label{lem:b3:fouriertransform:multiplication}
$f, g \in L^1(\R)$ के लिए: $\displaystyle\int \hat f\,g =
\int f\,\hat g$।
\end{lemma}

\begin{proof}
दोनों पक्ष $\iint f(x)g(\xi)\eu^{-\iu x\xi}\dd
x\,\dd\xi$ के बराबर हैं (टोनेली--फुबिनी: निरपेक्ष मान का
द्विक समाकल $\norm f_1\norm g_1$ है)।
\end{proof}

\begin{theorem}[प्रतिलोमन]\label{thm:b3:fouriertransform:inversion}
मान लीजिए $f \in L^1(\R)$।\index{फूरिये प्रतिलोमन प्रमेय}
\begin{enumerate}
  \item (गाउसीय योग्यता) प्रत्येक $x$ के लिए
        \[
        (f * g_\varepsilon)(x) =
        \frac1{2\pi}\int_\R \hat f(\xi)\,
        \eu^{-\varepsilon\xi^2}\,\eu^{\iu x\xi}\,\dd\xi,
        \qquad\text{जहाँ } g_\varepsilon(y) =
        \frac{1}{2\sqrt{\pi\varepsilon}}\,
        \eu^{-y^2/4\varepsilon},
        \]
        और $\varepsilon
        \to 0$ पर $L^1$ में $f * g_\varepsilon \to f$।
  \item यदि इसके अतिरिक्त $\hat f \in L^1$ हो, तो लगभग प्रत्येक $x$ के लिए
        \[
        f(x) = \frac{1}{2\pi}\int_\R \hat
        f(\xi)\,\eu^{\iu x\xi}\,\dd\xi ,
        \]
        और $f$ का कोई \omterm{def:b3:topology:continuity}{संतत} प्रतिनिधि होता है।
  \item (एकैकीपन) यदि $\hat f = 0$ हो, तो लगभग सर्वत्र $f = 0$।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) $x$ स्थिर कीजिए और $f$ तथा $g(\xi) = \frac1{2\pi}\eu^{-\varepsilon\xi^2}\eu^{\iu x\xi}$ पर \cref{lem:b3:fouriertransform:multiplication} लगाइए: \cref{ex:b3:fouriertransform:gaussian} से
(मॉडुलन नियम के साथ)
\[
\hat g(y) = \frac1{2\pi}\sqrt{\frac\pi\varepsilon}\,
\eu^{-(y - x)^2/4\varepsilon} = g_\varepsilon(x - y),
\]
अतः $\frac1{2\pi}\int\hat f(\xi)\eu^{-\varepsilon\xi^2}
\eu^{\iu x\xi}\dd\xi = \int f(y)g_\varepsilon(x - y)\dd y =
(f*g_\varepsilon)(x)$। $g_\varepsilon$ एक आसन्न तत्समक हैं: $g_\varepsilon \geq 0$, $\int g_\varepsilon = 1$ (गाउसीय समाकल), जो
$0$ पर संकेंद्रित होते हैं; और \cref{thm:b3:lp:regularization}(2) की उपपत्ति शब्दशः लागू
होती है (केवल $\int g_\varepsilon = 1$ और संकेंद्रण का उपयोग हुआ था: पुच्छ के लिए
$\int_{\abs y > \delta}g_\varepsilon \to
0$): अतः $\norm{f * g_\varepsilon - f}_1 \to 0$।

(2) यदि $\hat f \in L^1$ हो: (1) का दायाँ पक्ष प्रभावी अभिसरण प्रमेय (प्रभावी
फलन $\abs{\hat f}$) से \emph{प्रत्येक} $x$ के लिए $\frac1{2\pi}\int\hat
f(\xi)\eu^{\iu x\xi}\dd\xi$ पर अभिसरित होता है,
और यह सीमा फलन \omterm{def:b3:topology:continuity}{संतत} है (फिर प्रभावी अभिसरण प्रमेय)। दूसरी ओर
$L^1$ में $f
* g_\varepsilon \to f$, अतः किसी उपानुक्रम के अनुदिश लगभग सर्वत्र
(\cref{thm:b3:lp:complete}): इसलिए दोनों सीमाएँ लगभग सर्वत्र सहमत हैं।

(3) $\hat f = 0$ से (1) का दायाँ पक्ष लुप्त हो जाता है: सभी $\varepsilon$ के लिए
$f *
g_\varepsilon = 0$, और $L^1$ में $f *
g_\varepsilon \to f$: अतः लगभग सर्वत्र $f = 0$।
\end{proof}

\section{श्वार्ट्स वर्ग}

\begin{definition}\label{def:b3:fouriertransform:schwartz}
\emph{श्वार्ट्स वर्ग}\index{श्वार्ट्स वर्ग} $\mathcal
S(\R)$ उन $\mathcal C^\infty$ फलनों
$f$ से बना है जिनके लिए सभी $m, n \geq 0$ पर $\sup_x\abs{x^m f^{(n)}(x)} < \infty$ (अर्थात् सभी अवकलज
किसी भी घात से तेज \omterm{rem:b3:complete:meagre}{क्षीण} होते हैं)। उदाहरण: $\eu^{-ax^2}$, $\mathcal C_c^\infty$। स्पष्टतः
प्रत्येक $p$ के लिए $\mathcal S
\subseteq L^p$ ($C(1 + x^2)^{-1}$ से परिबद्ध कीजिए), और $\mathcal S$
अवकलजों, बहुपदों से गुणन, तथा गुणनफलों के अंतर्गत स्थिर है।
\end{definition}

\begin{theorem}\label{thm:b3:fouriertransform:schwartz}
फूरिये रूपांतर $\mathcal S(\R)$ को उसी पर एकैकी आच्छादक रूप से प्रतिचित्रित
करता है, जिसका प्रतिलोम $\check g(x) = \frac1{2\pi}\int
g(\xi)\eu^{\iu x\xi}\dd\xi$ है।
\end{theorem}

\begin{proof}
मान लीजिए $f \in \mathcal S$। \cref{prop:b3:fouriertransform:basic}(3) को दोहराने पर $(\hat f)^{(n)} =
\widehat{(-\iu x)^nf}$ (प्रत्येक $x^kf \in L^1$);
और $h = (-\iu x)^nf \in \mathcal S$ (जिसके सभी अवकलज \omterm{def:b3:lebesgue:l1}{समाकलनीय} हैं) के साथ (4) को दोहराने पर
$(\iu\xi)^m\hat h =
\widehat{h^{(m)}}$। इन्हें मिलाकर
\[
\abs{\xi^m\,(\hat f)^{(n)}(\xi)}
= \bigl|\widehat{\,h^{(m)}}(\xi)\bigr|
\leq \bigl\|\bigl((-\iu x)^nf\bigr)^{(m)}\bigr\|_1 < \infty
\]
और वह भी $\xi$ में एकसमान रूप से: अतः $\hat f \in \mathcal S$। चूँकि $\hat f \in L^1$,
प्रतिलोमन (\cref{thm:b3:fouriertransform:inversion}(2)) सर्वत्र लागू होता है (दोनों पक्ष \omterm{def:b3:topology:continuity}{संतत}):
अर्थात् $\check{\hat f} = f$, और सममित रूप से $\widehat{\check g} = g$ (जाँच रूपांतर $g \mapsto \frac1{2\pi}\hat g(-\cdot)$ है, जो
फिर $\mathcal S$ को सुरक्षित रखता है): अतः एकैकी आच्छादक।
\end{proof}

\section{प्लांशरेल और \texorpdfstring{$L^2$}{L2}}

\begin{theorem}[प्लांशरेल]\label{thm:b3:fouriertransform:plancherel}
$f \in L^1 \cap L^2(\R)$ के लिए:\index{प्लांशरेल की प्रमेय}
\[
\norm{\hat f}_2^2 = 2\pi\,\norm f_2^2 .
\]
फलस्वरूप $\widehat{\phantom f}$ $\norm{\mathcal Ff}_2 = \sqrt{2\pi}\norm f_2$ वाले किसी \omterm{def:b3:topology:continuity}{संतत} रैखिक प्रतिचित्रण $\mathcal F \colon L^2(\R) \to L^2(\R)$ तक
अद्वितीय रूप से विस्तारित हो जाता है; $\mathcal
F$ एकैकी आच्छादक है,
जिसमें $\sigma f =
f(-\cdot)$ के साथ $\mathcal F^{-1} =
\frac1{2\pi}\,\mathcal F\circ\sigma$, और वह अंतःगुणनों को गुणक $2\pi$ तक
सुरक्षित रखता है।
\end{theorem}

\begin{proof}
मान लीजिए $f \in L^1\cap L^2$ और $\tilde f(x) = \overline{f(-x)}$ के साथ $h = f * \tilde f$। तब $h \in L^1$ (\cref{thm:b3:product:convolution}), $h$
\omterm{def:b3:topology:continuity}{संतत} और परिबद्ध है (\cref{exo:b3:lp:6}: $f, \tilde f \in L^2$), $h(0) =
\int f\bar f = \norm f_2^2$, और $\hat h = \hat
f\,\widehat{\tilde f} = \hat f\,\overline{\hat f} =
\abs{\hat f}^2 \geq 0$ ($\widehat{\tilde f} =
\overline{\hat f}$ परिकलित
कीजिए)। $x = 0$ पर $h$ के लिए \cref{thm:b3:fouriertransform:inversion}(1) लगाइए:
\[
(h * g_\varepsilon)(0) = \frac1{2\pi}\int \hat
h(\xi)\,\eu^{-\varepsilon\xi^2}\dd\xi .
\]
$\varepsilon \to 0$ पर: बायाँ पक्ष $h(0)$ की ओर जाता है ($h$ \omterm{def:b3:topology:continuity}{संतत} परिबद्ध:
छोटे/बड़े $y$ में बाँटकर $(h*g_\varepsilon)(0) - h(0) = \int(h(-y)
- h(0))g_\varepsilon(y)\dd y \to 0$); और दायाँ पक्ष एकदिष्ट अभिसरण
प्रमेय ($\hat h \geq 0$) से बढ़कर $\frac1{2\pi}\int\hat h$ तक जाता है। अतः $\frac1{2\pi}\int\abs{\hat f}^2 =
\norm f_2^2$, जो
\emph{पूर्वतः} परिमित हो या न हो --- और वह परिमित है, जिससे सदस्यता
और सर्वसमिका दोनों सिद्ध हो जाती हैं।

विस्तार: $L^1\cap L^2 \supseteq \mathcal C_c$ $L^2$ में सघन है (\cref{thm:b3:lp:density}); रूपांतर वहाँ
$\sqrt{2\pi}$-समदूरिक है, अतः वह $L^2$ पर किसी नियतांक-तक-समदूरीयता $\mathcal F$
तक अद्वितीय रूप से विस्तारित हो जाता है (\cref{thm:b3:complete:extension})। $\mathcal S$ के लिए
प्रतिलोमन (\cref{thm:b3:fouriertransform:schwartz}) उसी सघनता से स्थानांतरित हो जाता है (दोनों पक्ष
$L^2$-संतत): अतः $\mathcal S$ पर $\mathcal
F\bigl(\frac1{2\pi}\mathcal F(\sigma f)\bigr) = f$, और इसलिए $L^2$ पर भी: एकैकी
आच्छादकता। \omterm{def:b3:hilbert:inner}{अंतःगुणन}: मानक सर्वसमिका से ध्रुवण।
\end{proof}

\begin{theorem}[प्वासों योग]
\label{thm:b3:fouriertransform:poisson}
मान लीजिए $f \in \mathcal S(\R)$ ($\abs{f} +
\abs{\hat f} \leq C(1 + \abs\cdot)^{-2}$ वाला \omterm{def:b3:topology:continuity}{संतत} $f$ पर्याप्त है)। तब\index{प्वासों
योग सूत्र}
\[
\sum_{n\in\Z} f(n) \;=\; \sum_{k\in\Z}\hat f(2\pi k) .
\]
\end{theorem}

\begin{proof}
मान लीजिए $F(x) = \sum_{n\in\Z}f(x + n)$: यह श्रेणी संहतों पर प्रसामान्य रूप से अभिसरित
होती है ($f$ का क्षय), अतः $F$ \omterm{def:b3:topology:continuity}{संतत} है, और वह $1$-आवर्ती है।
उसके फूरिये गुणांक (आवर्त $1$: $c_k(F) = \int_0^1F(t)\eu^{-2\iu\pi kt}\dd t$):
\[
c_k(F) = \sum_n\int_0^1 f(t + n)\,\eu^{-2\iu\pi kt}\dd t
= \int_\R f(t)\,\eu^{-2\iu\pi kt}\dd t = \hat f(2\pi k)
\]
(प्रसामान्य अभिसरण अदला-बदली को उचित ठहराता है; और कला
$1$-आवर्ती है)। श्रेणी $\sum_k\abs{c_k(F)}$ अभिसरित होती है ($\hat f$ का क्षय),
अतः $F$ की फूरिये श्रेणी प्रसामान्य रूप से अभिसरित होती है;
उसका योग एक ऐसा \omterm{def:b3:topology:continuity}{संतत} फलन है जिसके फूरिये गुणांक $F$ के ही हैं,
इसलिए वह $F$ के बराबर है (वृत्त पर एकैकीपन: अंतर के सभी गुणांक
शून्य हैं, और \cref{thm:b3:hilbert:fourier} $L^2$ में शून्य दे देता है, अतः \omterm{def:b3:topology:continuity}{सांतत्य} से
सर्वत्र)। अब $x = 0$ पर मान लीजिए।
\end{proof}

\begin{example}[थीटा सर्वसमिका]
\label{ex:b3:fouriertransform:theta}
प्वासों को $f(x) = \eu^{-\pi tx^2}$ ($t > 0$) पर लगाने से, जिसका रूपांतर $\hat f(\xi) = t^{-1/2}\eu^{-\xi^2/4\pi t}$ है
($a = \pi t$ के साथ \cref{ex:b3:fouriertransform:gaussian}):
\[
\sum_{n\in\Z}\eu^{-\pi n^2t}
= \frac1{\sqrt t}\sum_{k\in\Z}\eu^{-\pi k^2/t} :
\]
अर्थात् याकोबी के थीटा फलन का कार्यकीय समीकरण, जो रीमान के $\zeta$
के कार्यकीय समीकरण की कुंजी है --- और एक शानदार संख्यात्मक त्वरक
भी: छोटे $t$ के लिए बायाँ पक्ष धीरे अभिसरित होता है, और दायाँ
पक्ष बिजली की गति से।
\end{example}

\begin{method}\label{met:b3:fouriertransform:method}
कार्य-क्षेत्र: $L^1$ --- रूपांतर बिंदुशः परिभाषित, प्रतिलोमन के
लिए $\hat f \in L^1$ चाहिए; $\mathcal S$ --- यहाँ सब कुछ वैध है, पहले यहीं सिद्ध
कीजिए; $L^2$ --- रूपांतर सघनता से परिभाषित (समाकल से नहीं!), \omterm{def:b3:complete:complete}{पूर्ण}
सममिति, \omterm{thm:b3:hilbert:parseval}{पार्सेवाल} लेखा-जोखा। कोई रूपांतर परिकलित करने के लिए:
\cref{prop:b3:fouriertransform:basic} के नियमों से उसे सारणी (सूचक, चरघातांकी, गाउसीय) तक सिमटाइए;
कोई सर्वसमिका सिद्ध करने के लिए: उसे $\mathcal S$ (या $\mathcal C_c^\infty$) पर स्थापित
कीजिए और सघनता तथा \omterm{def:b3:topology:continuity}{सांतत्य} से बढ़ाइए (\cref{met:b3:lp:toolkit}); और अचर गुणांकों वाला
कोई रैखिक आंशिक या साधारण अवकल समीकरण हल करने के लिए: रूपांतर
लीजिए, भाग दीजिए, प्रतिलोम लीजिए।
\end{method}

इनके फूरिये रूपांतर परिकलित कीजिए: $\mathbf 1_{\intcc{-a}a}$; $\eu^{-a\abs x}$ ($a > 0$); तंबू फलन
$\max(0, 1 -
\abs x)$; $\frac1{x^2 + a^2}$ \emph{(दूसरे पर प्रतिलोमन का उपयोग कीजिए)}। बनती हुई
सारणी दर्ज कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:fouriertransform:2}
मान लीजिए $f \in L^1$। $\hat f$ के पदों में इनके रूपांतर व्यक्त कीजिए:
$f(x - a)$, $f(x)\cos(bx)$, $f(ax + b)$, $\overline{f(-x)}$, $(f * f)(x)$। प्रत्येक नियम को गाउसीय पर
सत्यापित कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:3}
(क) दर्शाइए कि $\mathbf 1_{\intcc{-1}1} * \mathbf
1_{\intcc{-1}1}$ का रूपांतर $\bigl(\frac{2\sin\xi}\xi
\bigr)^2$ है, और प्लांशरेल से ---
अथवा $0$ पर प्रतिलोमन से --- $\int_\R\bigl(\frac{\sin\xi}\xi\bigr)^2\dd\xi = \pi$ निकालिए। \cref{pb:b3:lebesgue:1} से तुलना
कीजिए।
(ख) $\eu^{-\abs x}$ पर प्लांशरेल लगाकर $\int_\R\frac{\dd x}{(x^2+1)^2}$ परिकलित कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:4}
(\omterm{pb:b3:fouriertransform:1}{ऊष्मा नाभिक} का बीजगणित) $g_t(x) =
\frac1{2\sqrt{\pi t}}\eu^{-x^2/4t}$ के साथ:
(क) $\hat g_t(\xi) = \eu^{-t\xi^2}$ सत्यापित कीजिए;
(ख) बिना किसी समाकल परिकलन के अर्धसमूह नियम $g_t * g_s = g_{t+s}$ निकालिए;
(ग) $\norm{g_t}_1 = 1$ और $\norm{g_t}_2^2 =
(8\pi t)^{-1/2}$ दर्शाइए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:5}
दर्शाइए कि यदि $f \in L^1$ सम और वास्तविक हो, तो $\hat f$ सम और वास्तविक
होता है; और यदि $f$ विषम और वास्तविक हो, तो $\hat f$ विषम और
विशुद्ध काल्पनिक। $\hat f(0)$ क्या परिकलित करता है? निष्कर्ष निकालिए कि
$f \geq
0$ से $\norm{\hat f}_\infty = \hat f(0) = \int f$ अनिवार्य हो जाता है, और प्रायिकता घनत्वों के लिए
इसकी व्याख्या कीजिए (\cref{ch:b3:clt}: किसी अभिलक्षणिक फलन का मापांक $\leq 1$
होता है, जो $0$ पर प्राप्त होता है)।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:fouriertransform:6}
(अनाच्छादकता) तीन चरणों में दर्शाइए कि $\widehat{\phantom f}\colon L^1
\to \mathcal C_0$ एकैकी और \omterm{def:b3:topology:continuity}{संतत} है, पर
आच्छादक \emph{नहीं}।
(क) एकैकीपन (\cref{thm:b3:fouriertransform:inversion}) और \omterm{def:b3:topology:continuity}{सांतत्य} ($\norm{\hat f}_\infty \leq \norm f_1$), तथा यह कि $\mathcal C_0$ एक बानाख
समष्टि है ($\norm\cdot_\infty$ में संवृत)।
(ख) यदि प्रतिचित्रण आच्छादक होता, तो वह एकैकी आच्छादक होता, और
विवृत प्रतिचित्रण प्रमेय (\cref{thm:b3:banach:openmapping}) ऐसा नियतांक $C$ देती जिसके लिए
सभी $f \in L^1$ पर $\norm f_1 \leq C\norm{\hat
f}_\infty$।
(ग) इसका खंडन $f_n(x) = \frac{\sin
x}{x}\cdot\frac{\sin(x/n)}{x/n}$ से कीजिए: उसका रूपांतर (नियतांकों तक) संवलन
$\mathbf 1_{\intcc{-1}1} *
\mathbf 1_{\intcc{-1/n}{1/n}}$ प्रकार का समलंब है --- दर्शाइए कि एकसमान रूप से $\norm{\hat f_n}_\infty \leq \pi$, जबकि
$[1, n]$ पर (जहाँ दूसरा गुणनखंड नीचे से परिबद्ध है) $\frac{\abs{\sin x}}x$ के मेहराब
गिनकर $\norm{f_n}_1 \geq c\ln n$, ठीक जैसे \cref{thm:b3:banach:fourierdiverge} में।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:7}
(चिकनाई $\leftrightarrow$ क्षय का शब्दकोश) सिद्ध कीजिए: किसी $\delta > 0$ के लिए
$\hat f(\xi) = O(\abs\xi^{-k-1-\delta})$ वाले $f \in
L^1$ से $f$ का कोई $\mathcal C^k$ प्रतिनिधि होना निकलता है।
विलोमतः $f \in \mathcal C^k_c$ से $\hat f(\xi) =
O(\abs\xi^{-k})$ निकलता है। दोनों दिशाओं को तंबू फलन पर
दर्शाइए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:fouriertransform:8}
(हाइजेनबर्ग की असमिका) $\norm f_2 = 1$ वाले वास्तविक $f \in \mathcal S(\R)$ के लिए सिद्ध
कीजिए
\[
\Bigl(\int x^2f(x)^2\dd x\Bigr)\cdot
\Bigl(\frac1{2\pi}\int \xi^2\abs{\hat f(\xi)}^2\dd\xi\Bigr)
\;\geq\; \frac14 ,
\]
जिसमें गाउसीय के लिए समता होती है। \emph{($1 = \int f^2 =
-\int x\,(f^2)'$ को खंडशः लिखिए,
कोशी--श्वार्ज़ से परिबद्ध कीजिए, और प्लांशरेल से $\norm{f'}_2$ में बदल
दीजिए।)} व्याख्या: कोई संकेत और उसका स्पेक्ट्रम दोनों एक साथ
संकेंद्रित नहीं हो सकते।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:9}
\cref{ex:b3:fouriertransform:theta} को विस्तार से उचित ठहराइए (गाउसीय के लिए प्वासों की
परिकल्पनाएँ), और इस सर्वसमिका से तीन पदों में $\sum_{n\in\Z}\eu^{-\pi n^2}$ का छह दशमलव
तक मान निकालिए। $t = 10^{-2}$ पर उतनी ही यथार्थता के लिए
\emph{परिभाषक} श्रेणी के कितने पद चाहिए होते, और रूपांतरित श्रेणी
के कितने?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:10}
(आवृत्ति-सीमित फलन) मान लीजिए $f \in L^2(\R)$ ऐसा है कि $\mathcal Ff$ $\intcc{-\pi}\pi$ में
आलंबित है। दर्शाइए कि $f$ का कोई ऐसा प्रतिनिधि है जिसका प्रत्येक
मान प्रतिदर्शों से पुनः प्राप्त किया जा सकता है: $L^2(\intcc{-\pi}\pi)$ (\cref{thm:b3:hilbert:fourier}) के
फूरिये आधार में $\mathcal Ff$ का प्रसार करके और पदशः प्रतिलोम रूपांतर लेकर
पूर्णांकों पर \emph{शैनन अंतर्वेशन}
\[
f(x) = \sum_{n\in\Z} f(n)\,
\frac{\sin\bigl(\pi(x - n)\bigr)}{\pi(x - n)}
\quad L^2 \text{ में},
\]
सिद्ध कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:11}
(चतुर्थ कोटि के संकारक के रूप में रूपांतर) $\mathcal
S(\R)$ पर मान लीजिए
$\mathcal F f = \hat f$।
(क) प्रतिलोमन सूत्र का उपयोग करके $(\mathcal F^2f)(x) =
2\pi\,f(-x)$ दर्शाइए, और $\mathcal F^4 = (2\pi)^2\,
\mathrm{id}$ निष्कर्ष
निकालिए।
(ख) निष्कर्ष निकालिए कि $\mathcal S$ पर $\mathcal F$ का प्रत्येक अभिलक्षणिक मान
$\{\pm\sqrt{2\pi},
\pm\iu\sqrt{2\pi}\}$ में है, और $+\sqrt{2\pi}$ के लिए एक अभिलक्षणिक फलन प्रस्तुत कीजिए
\emph{(इस अध्याय का कौन-सा फलन अपने ही रूपांतर के समानुपाती है?)}।
(ग) दर्शाइए कि सम फलन $\mathcal F^2f = 2\pi
f$ को और विषम फलन $\mathcal F^2f = -2\pi f$ को संतुष्ट करते
हैं; और $x\eu^{-x^2/2}$ से अभिलक्षणिक मान $-\iu\sqrt{2\pi}$ के लिए एक अभिलक्षणिक फलन
बनाइए, उसका रूपांतर परिकलित करके (गाउसीय के रूपांतर का अवकलन
कीजिए)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:12}
(स्वसहसंबंध और वीनर की प्रमेयिका) $f \in L^2(\R)$ के लिए $\tilde f(x) = \overline{f(-x)}$ और
\emph{स्वसहसंबंध} $A_f = f * \tilde f$ परिभाषित कीजिए।
(क) दर्शाइए कि $A_f$ एक परिबद्ध \omterm{def:b3:topology:continuity}{संतत} फलन है जिसके लिए सभी $x$ पर
$A_f(0) = \norm f_2^2 \geq \abs{A_f(x)}$ (\cref{exo:b3:lp:6} और कोशी--श्वार्ज़)।
(ख) पहले $f \in L^1\cap L^2$ के लिए दर्शाइए कि $\widehat{A_f} = \abs{\hat f\,}^2 \geq 0$: अर्थात् स्वसहसंबंध का
रूपांतर अऋणात्मक होता है --- स्वसहसंबंधों के स्पेक्ट्रम शक्ति
स्पेक्ट्रम होते हैं।
(ग) सर्वसमिका $\int_\R\abs{\hat
f(\xi)}^2\eu^{\iu x\xi}\,\dd\xi = 2\pi A_f(x)$ निष्कर्ष के रूप में निकालिए (प्रतिलोमन; जब
$\hat f \in L^2$ के लिए $\abs{\hat f}^2 \in L^1$ हो तब उसकी प्रयोज्यता उचित ठहराइए), और उसे
$x = 0$ पर $f = \mathbf
1_{\intcc{-1/2}{1/2}}$ के लिए परिकलित कीजिए: $\int_\R\bigl(\frac{\sin u}u\bigr)^2\dd u = \pi$ पुनः प्राप्त कीजिए।
\end{exercise}

\section{समस्या: रेखा पर ऊष्मा समीकरण}

\begin{problem}[{सप्ताहांत समस्या --- $\partial_tu =
\partial^2_{xx}u$, आरंभ से अंत तक हल}]
\label{pb:b3:fouriertransform:1}
ऊष्मा फैलती है; समीकरण $\partial_tu = \partial_{xx}^2u$ कहता है कि उसका \omterm{ex:b3:lebesgue:gamma}{घनत्व} ताप-रूपरेखा की
स्थानीय वक्रता से दी गई दर पर विसरित होता है। हम $\R$ पर कोशी
समस्या हल करते हैं --- $f$ दिया हो तो $u(0, \cdot) = f$ के साथ $t > 0$ के लिए
$u(t, x)$ ढूँढ़िए --- हल के उल्लेखनीय गुण सिद्ध करते हैं, और देखते हैं
कि समय उलटा क्यों नहीं जा सकता। सर्वत्र $g_t(x) = \frac1{2\sqrt{\pi t}}\eu^{-x^2/4t}$ \emph{ऊष्मा
नाभिक}\index{ऊष्मा नाभिक} है और $u(t, \cdot) = g_t *
f$।

\medskip
\noindent\textbf{भाग I --- नाभिक की व्युत्पत्ति।} पहले औपचारिक रूप
से कार्य कीजिए: मान लीजिए $u(t, \cdot) \in \mathcal S$ समीकरण को हल करता है, और मान
लीजिए $\hat u(t, \xi)$ $x$ में रूपांतर है।
\begin{enumerate}
  \item (औपचारिक रूप से) दर्शाइए $\partial_t\hat u = -\xi^2\hat u$, अतः $\hat u(t,\xi) = \eu^{-t\xi^2}\hat f(\xi)$, और $u(t) = g_t * f$
        पहचानिए (\cref{exo:b3:fouriertransform:4})। इससे $u$ की \emph{परिभाषा} प्रेरित होती
        है; अब सब कुछ सीधे सिद्ध होगा, $f \in \mathcal C_b(\R)$ (परिबद्ध \omterm{def:b3:topology:continuity}{संतत}) अथवा
        $f \in L^p$ के लिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- सत्यापन।}
\begin{enumerate}[resume]
  \item दर्शाइए कि $t > 0$ के लिए $f \in \mathcal
        C_b$ पर $u(t, x) = \int
        g_t(x-y)f(y)\dd y$ सुपरिभाषित है, और
        यह कि $\intoo0\infty\times\R$ पर $(t, x)$ में $u$ $\mathcal C^\infty$ है \emph{(समाकल के
        भीतर अवकलन कीजिए; गाउसीय अवकलजों को $(t,x)$ में स्थानीय रूप
        से एकसमान ढंग से प्रभावित कीजिए)}।
  \item सीधे परिकलन से $\partial_tg_t = \partial^2_{xx}g_t$ सत्यापित कीजिए, और $t > 0$ के लिए $\partial_tu =
        \partial^2_{xx}u$
        निष्कर्ष निकालिए।
  \item (आरंभिक शर्त) दर्शाइए कि $f \in \mathcal
        C_b$ के लिए $t \to 0^+$ पर संहतों पर
        एकसमान रूप से $u(t, x) \to f(x)$ \emph{(आसन्न तत्समक: $\abs y
        \leq \delta$, $\abs y > \delta$ में
        बाँटिए)}; और $f \in L^p$ ($p < \infty$) के लिए $\norm{u(t) - f}_p \to 0$ दर्शाइए।
  \item (तात्क्षणिक चिकनाई) निष्कर्ष निकालिए: केवल परिबद्ध \omterm{def:b3:topology:continuity}{संतत}
        $f$ के लिए भी हल प्रत्येक $t > 0$ पर $\mathcal C^\infty$ होता है ---
        ऊष्मा तत्काल खुरदरापन मिटा देती है। $f =
        \mathbf 1_{\intoo0\infty}$ के लिए $u(t, \cdot)$
        स्पष्ट परिकलित कीजिए (एक त्रुटि फलन) और $t$ के तीन
        मानों के लिए उसकी रूपरेखा खींचिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- संरचनात्मक गुण।}
\begin{enumerate}[resume]
  \item (धनात्मकता और तुलना) यदि $f \geq 0$ हो, तो सभी $t > 0$ के लिए
        $u > 0$ (और कड़ाई से, जब तक लगभग सर्वत्र $f = 0$ न हो); और
        यदि $f_1 \leq f_2$ हो, तो $u_1 \leq u_2$। कोई ठंडा स्थान तत्काल गर्म हो जाता
        है: टिप्पणी कीजिए।
  \item (संरक्षण) $f \in L^1$ के लिए: सभी $t$ पर $\int u(t, x)\dd x =
        \int f$ \emph{(टोनेली)}
        --- अर्थात् कुल ऊष्मा संरक्षित रहती है।
  \item (अपव्यय) $f \in L^1\cap L^2$ के लिए प्लांशरेल के माध्यम से दर्शाइए कि
        $t \mapsto \norm{u(t)}_2$ अनवर्धी है, और कड़ाई से जब तक $f = 0$ न हो, तथा $t \to \infty$
        पर उसकी सीमा परिकलित कीजिए। इसके अतिरिक्त $\norm{u(t)}_\infty \leq \frac{\norm
        f_1}{2\sqrt{\pi t}} \to 0$ भी
        दर्शाइए: ऊष्मा फैलती है और मर जाती है।
  \item (अद्वितीयता, $L^2$ वर्ग) मान लीजिए $u$ ऐसा हल है जिसके
        लिए सभी $t$ पर $u(t) \in L^2$, स्वाभाविक अर्थ में $u \in \mathcal
        C^1(\intoo0\infty, L^2)$, और
        $t\to0$ पर $L^2$ में $u(t) \to f$; और यह स्वीकार करते हुए कि
        रूपांतर उसे लगभग प्रत्येक $t$ के लिए $\xi$ में बिंदुशः
        लगभग सर्वत्र $\partial_t\hat u =
        -\xi^2\hat u$ में बदल देता है \emph{($\xi$ में $\mathcal
        C_c^\infty$
        के सापेक्ष परीक्षा लेकर उचित ठहराया गया --- इसकी रूपरेखा
        दीजिए)}, दर्शाइए कि $\hat u(t,\xi) = \eu^{-t\xi^2}\hat f(\xi)$, अतः इस वर्ग में अद्वितीयता।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- समय का तीर।}
\begin{enumerate}[resume]
  \item दर्शाइए कि \emph{पश्चगामी} समस्या दुर्विन्यस्त है: समय
        $0$ पर आँकड़े $f$ देकर समय $-s$ ($s > 0$) पर हल के
        अस्तित्व के लिए --- अर्थात् $f = g_s * h$ का कोई हल $h \in L^2$ होने के
        लिए --- यह आवश्यक है कि $\eu^{s\xi^2}\hat f(\xi) \in L^2$: अर्थात् $\hat f$ पर एक अत्यंत
        कठोर क्षय शर्त। ऐसा कोई स्पष्ट चिकना $f
        \in L^2$ प्रस्तुत कीजिए
        जिसके लिए किसी भी समय अंतराल पर कोई पश्चगामी हल विद्यमान
        न हो: $\hat f(\xi) =
        \eu^{-\abs\xi}$ वाला फलन लीजिए --- $f$ पहचानिए (\cref{exo:b3:fouriertransform:1}) और
        प्रत्येक $s >
        0$ के लिए $\eu^{s\xi^2}\eu^{-\abs\xi} \notin L^2$ दर्शाइए।
  \item (चिकनाई बनाम सूचना) प्रश्न 5, 9 और 10 का उपयोग करते हुए
        एक छोटे अनुच्छेद में समझाइए कि ऊष्मा अर्धसमूह $(f \mapsto g_t * f)_{t\geq0}$ $L^2$
        पर एकैकी क्यों है पर आच्छादक नहीं, और यह विसरण की
        अनुत्क्रमणीयता को कैसे व्यक्त करता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- शैनन की प्रतिचयन प्रमेय।} कोई फलन
$f \in L^2(\R)$ $\Omega$ तक \emph{आवृत्ति-सीमित} कहलाता है यदि $\intcc{-\Omega}\Omega$ के बाहर
लगभग सर्वत्र $\hat f = 0$ हो; ऐसे फलनों के लिए $PW_\Omega$ (पेली--वीनर समष्टि)
लिखिए।
\begin{enumerate}[resume]
  \item दर्शाइए कि प्रत्येक $f \in PW_\Omega$ लगभग सर्वत्र $\mathcal C^\infty$ फलन $\frac1{2\pi}\int_{-\Omega}^{\Omega}\hat
        f(\xi)\eu^{\iu x\xi}\,\dd\xi$ के
        बराबर है (चिकनाई और यह पहचान उचित ठहराइए), और उसके सभी
        अवकलज परिबद्ध होते हैं: अर्थात् आवृत्ति-सीमा नियमितता का
        एक चरम रूप है। आगे से $f$ इसी प्रतिनिधि को निरूपित करेगा।
  \item $\hat f \in L^2(\intcc{-\Omega}\Omega)$ का उस अंतराल के फूरिये आधार में प्रसार कीजिए और
        गुणांकों को $f$ के \emph{प्रतिदर्श} के रूप में पहचानिए:
        \[
        \hat f(\xi) = \frac\pi\Omega\sum_{n\in\Z}
        f\Bigl(\frac{n\pi}\Omega\Bigr)\,
        \eu^{-\iu n\pi\xi/\Omega}
        \quad L^2(\intcc{-\Omega}\Omega) \text{ में} .
        \]
  \item \emph{प्रतिचयन प्रमेय} निष्कर्ष के रूप में निकालिए: $f \in
        PW_\Omega$
        के लिए
        \[
        f(x) = \sum_{n\in\Z}f\Bigl(\frac{n\pi}
        \Omega\Bigr)\,\operatorname{sinc}(\Omega x - n\pi),
        \qquad \operatorname{sinc}t = \frac{\sin t}t,
        \]
        जिसमें अभिसरण $L^2(\R)$ में और $\R$ पर एकसमान रूप से होता है
        \emph{(प्रश्न 13 की श्रेणी को प्रतिलोमन सूत्र में डालिए
        और प्रारंभिक समाकल परिकलित कीजिए)}: अर्थात् कोई
        आवृत्ति-सीमित संकेत पूर्णतः चरण $\pi/\Omega$ की जालिका पर अपने
        मानों से निर्धारित हो जाता है --- नाइक्विस्ट दर।
  \item दर्शाइए कि फलन $x \mapsto
        \operatorname{sinc}(\Omega x - n\pi)$, $n \in \Z$, $L^2(\R)$ में एक लांबिक कुल बनाते
        हैं जिनका मानक अचर $\sqrt{\pi/\Omega}$ है, और ऊर्जा सर्वसमिका $\norm f_2^2 =
        \frac\pi\Omega\sum_n\abs{f(n\pi/\Omega)}^2$
        निष्कर्ष के रूप में निकालिए।
  \item (छद्मन) ऐसा कोई शून्येतर $g \in PW_{2\Omega}$ प्रस्तुत कीजिए जो प्रत्येक
        प्रतिदर्श बिंदु $\frac{n\pi}\Omega$ पर लुप्त हो जाए \emph{($g(x) =
        \sin(\Omega x)\operatorname{sinc}(\Omega x)$ पर
        विचार कीजिए और उसका आवृत्ति-परास जाँचिए)}: अर्थात्
        नाइक्विस्ट दर से नीचे प्रतिचयन सूचना खो देता है --- दो
        भिन्न संकेतों के सभी प्रतिदर्श एक हो सकते हैं: गणितीय रूप
        में वही स्ट्रोबोस्कोपी पहिया प्रभाव।
  \item (स्वातंत्र्य कोटियाँ) प्रश्न 14--15 का उपयोग करते हुए
        अभियांत्रिकी का नियम उचित ठहराइए: $\Omega$ तक आवृत्ति-सीमित
        और जिसकी ऊर्जा मूलतः लंबाई $T$ की समय-खिड़की में हो, ऐसा
        संकेत लगभग $\frac{\Omega T}\pi$ वास्तविक प्रतिदर्शों से वर्णित हो जाता है
        --- ``मूलतः'' को ऊर्जा सर्वसमिका और पुच्छ $\sum_{\abs{n\pi/\Omega} >
        T/2}$ के माध्यम
        से यथार्थ बनाइए।
  \item (संगति जाँच) प्रतिचयन प्रमेय $PW_\Omega$ के दो सदस्यों पर हाथ
        से सत्यापित कीजिए: (क) $f =
        \operatorname{sinc}(\Omega\,\cdot)$, जिसके प्रतिदर्श $\delta_{n0}$ हैं;
        (ख) $f(x) = \cos(\omega x)
        \operatorname{sinc}(\varepsilon x)$ प्रकार के संकीर्ण-पट्टी संकेत --- अधिक ठीक-ठीक,
        दर्शाइए कि $\Omega' < \Omega$ वाले $f \in
        PW_{\Omega'}$ के लिए $\Omega$-दर वाली श्रेणी भी
        $f$ का पुनर्निर्माण कर देती है (अति-प्रतिचयन हानिरहित
        है), और वह भी $PW_
        {\Omega'} \subseteq PW_\Omega$ को अंतःस्थापित करके।
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- अनिश्चितता, दो बार और।}
हाइजेनबर्ग की असमिका (\cref{exo:b3:fouriertransform:8}) यह परिबद्ध करती है कि $f$ और $\hat f$
एक साथ \emph{कितने} संकेंद्रित हो सकते हैं; यहाँ उसका सब-या-कुछ-नहीं
वाला सहोदर और उसकी यथार्थ संतृप्ति है।
\begin{enumerate}[resume]
  \item मान लीजिए $\operatorname{supp}f \subseteq
        \intcc{-A}A$ के साथ $f \in L^1$। दर्शाइए कि $\hat f$ सर्वत्र
        अभिसारी एक घात श्रेणी का योग है:
        \[
        \hat f(\xi) = \sum_{k\geq0}\frac{(-\iu\xi)^k}{k!}
        \,m_k, \qquad m_k = \int_{-A}^{A}x^kf(x)\,\dd x,
        \quad \abs{m_k} \leq A^k\norm f_1
        \]
        \emph{($\eu^{-\iu\xi x}$ का प्रसार कीजिए और प्रसामान्य अभिसरण से
        अदला-बदली उचित ठहराइए)}: अर्थात् रूपांतर
        \emph{वास्तविक-वैश्लेषिक} है, और प्रत्येक बिंदु पर उसकी
        अभिसरण त्रिज्या अनंत है।
  \item आलंब द्विभाजन निष्कर्ष के रूप में निकालिए: किसी अरिक्त
        विवृत अंतराल पर लुप्त होने वाला वास्तविक-वैश्लेषिक फलन
        सर्वथा लुप्त होता है \emph{(वह समुच्चय जहाँ सभी अवकलज
        लुप्त हैं विवृत और संवृत दोनों है --- टेलर तर्क विस्तार से
        लिखिए)}; निष्कर्ष निकालिए कि किसी भी शून्येतर $f$ के लिए
        $f$ और $\hat f$ दोनों \omterm{def:b3:topology:compact}{संहत} आलंब वाले नहीं हो सकते, और यह कि
        $PW_\Omega$ में कोई शून्येतर \omterm{def:b3:topology:compact}{संहत} आलंब वाला फलन नहीं है ---
        आवृत्ति-सीमित संकेत सदा बने रहते हैं, और समय-सीमित संकेत
        सभी आवृत्तियों में रिस जाते हैं।
  \item (हाइजेनबर्ग की संतृप्ति) गाउसीय कुल $f
        = \eu^{-ax^2}$ पर दोनों
        संकेंद्रण गुणक परिकलित कीजिए और सत्यापित कीजिए कि
        मानकीकृत गुणनफल $\bigl(\int x^2\abs f^2\bigr)\bigl(\frac1{2\pi}\int
        \xi^2\abs{\hat f}^2\bigr)\big/\norm f_2^4$ \emph{प्रत्येक} $a$ के लिए $\frac14$
        के बराबर है --- अर्थात् \cref{exo:b3:fouriertransform:8} का समता-कुल साक्षात रूप में;
        और किसी मापन तर्क से समझाइए कि इस कुल के अनुदिश यह गुणनफल
        अचर क्यों होना ही चाहिए।
  \item भौतिक पाठ समझाइए (किसी क्वांटम अवस्था के स्थिति/संवेग
        \omterm{ex:b3:lebesgue:gamma}{घनत्व}; मानकीकरण में $\hbar$ से $\sigma_x\sigma_p \geq
        \frac\hbar2$ मिलता है), और अध्याय
        भर संबंध जोड़िए: तात्क्षणिक चिकनाई (भाग II), अनुत्क्रमणीयता
        (भाग IV), प्रतिचयन (भाग V), हाइजेनबर्ग और आलंब द्विभाजन
        --- ये एक ही नियम की पाँच अभिव्यक्तियाँ हैं: अनंत पर
        $\hat f$ का व्यवहार यह तय कर देता है कि $f$ कहीं भी क्या
        कर सकता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग VII --- नाभिक का बीजगणित, और एक हल्य
उदाहरण।}
\begin{enumerate}[resume]
  \item (अर्धसमूह) $t, s > 0$ के लिए \emph{चैपमैन--कोलमोगोरोव
        सर्वसमिका} $g_t * g_s = g_{t+s}$ सिद्ध कीजिए (संवलन प्रमेय और $L^1$ पर
        रूपांतर के एकैकीपन से), और $u(t + s) = g_s *
        u(t)$ निष्कर्ष निकालिए: समय
        $t + s$ तक विकसित होना पहले $t$ तक, फिर $s$ तक विकसित
        होना ही है। प्रश्न 8 के अपव्यय को तीक्ष्ण कीजिए: $\norm{u(t)}_2^2 = \frac1{2\pi}\int
        \eu^{-2t\xi^2}\abs{\hat f(\xi)}^2\dd\xi$
        लिखकर कोशी--श्वार्ज़ से दर्शाइए कि
        \[
        t \longmapsto \ln\,\norm{u(t)}_2
        \quad \intoo0{+\infty} \text{ पर उत्तल है} :
        \]
        अर्थात् $L^2$ ऊर्जा केवल घटती ही नहीं, बल्कि लघुगणकीय
        उत्तल ढंग से घटती है।
  \item (ऊष्मा कहाँ जाती है) मान लीजिए $f \geq 0$, $f \in L^1$, जिसमें
        $\int x^2f(x)\dd x < \infty$। दर्शाइए कि सभी $t > 0$ के लिए
        \[
        \int_\R x\,u(t, x)\,\dd x = \int_\R x f(x)\,\dd x,
        \qquad
        \int_\R x^2u(t, x)\,\dd x
        = \int_\R x^2f(x)\,\dd x + 2t\int_\R f :
        \]
        अर्थात् ऊष्मा का \omterm{ex:b3:groups:actions}{केंद्र} कभी नहीं हिलता, और प्रसरण समय में
        \emph{रैखिक} रूप से बढ़ता है --- विसरणीय मापन $x \sim \sqrt{2t}$, जिसे
        \cref{ch:b3:probability} में ब्राउनी गति आने पर फिर से पढ़ना चाहिए।
        \emph{($g_t$ के पहले दो आघूर्ण परिकलित कीजिए और संवलन पर
        टोनेली का उपयोग कीजिए।)}
  \item (गाउसीय, आरंभ से अंत तक हल) $f(x) =
        \eu^{-x^2}$ के लिए संवृत रूप
        \[
        u(t, x) = \frac1{\sqrt{1 + 4t}}\,
        \exp\Bigl(-\frac{x^2}{1 + 4t}\Bigr),
        \]
        स्थापित कीजिए और उस पर हाथ से सत्यापित कीजिए: समीकरण
        $\partial_tu
        = \partial^2_{xx}u$; संरक्षण $\int u(t) =
        \sqrt\pi$; अपव्यय नियम $\norm{u(t)}_2 =
        (\pi/2)^{1/4}(1 + 4t)^{-1/4}$ (उसके $t^{-1/4}$ क्षय
        की तुलना प्रश्न 8 के $t^{-1/2}$ उच्चतम-मानक क्षय से कीजिए); और
        प्रश्न 24 की यथार्थ प्रसरण वृद्धि। $t = 6$ पर: शिखर गिरकर
        अपनी आरंभिक ऊँचाई का $\frac15$ रह जाता है जबकि रूपरेखा पाँच
        गुना चौड़ी हो जाती है --- वही ऊष्मा, फैली हुई।
\end{enumerate}
