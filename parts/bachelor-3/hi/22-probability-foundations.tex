\chapter{प्रायिकता: आधार और बृहत् संख्याओं का
नियम}\label{ch:b3:probability}

दूसरे वर्ष में प्रायिकता गणनीय समष्टियों पर खड़ी की गई थी; \omterm{def:b3:measure:measure}{माप}
सिद्धांत अब हर प्रतिबंध हटा देता है। \omterm{def:b3:probability:space}{प्रायिकता समष्टि} कुल द्रव्यमान
$1$ की \omterm{def:b3:measure:measure}{माप समष्टि} है, \omterm{def:b3:probability:space}{यादृच्छिक चर} \omterm{def:b3:lebesgue:measurable}{मापनीय} प्रतिचित्रण हैं, \omterm{def:b3:probability:space}{प्रत्याशा}
लेबेग समाकल है --- और तत्काल समूचा वैश्लेषिक शस्त्रागार (\cref{ch:b3:measure,ch:b3:lebesgue,ch:b3:product}) संयोग
पर लागू हो जाता है। यह अध्याय वह शब्दकोश स्थापित करता है, \omterm{def:b3:probability:independence}{स्वतंत्र}
यादृच्छिक चरों के अनंत अनुक्रम रचता है ($\intcc01$ पर, द्विआधारी अंकों से:
यादृच्छिकता \omterm{def:b3:measure:lebesgueouter}{लेबेग माप} के भीतर ही छिपी हुई है), बोरेल--कांतेली
प्रमेयिकाएँ और कोल्मोगोरोव का \omterm{thm:b3:probability:zeroone}{शून्य--एक नियम} सिद्ध करता है, अभिसरण
के प्रकारों को छाँटता है, और बृहत् संख्याओं का नियम सिद्ध करता है
--- वही प्रमेय जो आवृत्तियों को प्रायिकताओं की ओर अभिसरित कराती है
और सांख्यिकी को संभव बनाती है। सप्ताहांत समस्या प्रबल नियम की
एतेमादी वाली उपपत्ति उसके निश्चयात्मक $L^1$ रूप में देती है।

\section{शब्दकोश}

\begin{definition}\label{def:b3:probability:space}
\emph{प्रायिकता समष्टि}\index{प्रायिकता समष्टि} $\P(\Omega) =
1$ वाली \omterm{def:b3:measure:measure}{माप समष्टि}
$(\Omega, \mathcal A, \P)$ है; $\mathcal A$ के अवयव \emph{घटनाएँ} कहलाते हैं, और कोई गुण
\emph{लगभग निश्चित रूप से} लागू होता है यदि उसकी घटना की प्रायिकता
$1$ हो। \emph{यादृच्छिक चर}\index{यादृच्छिक चर} कोई \omterm{def:b3:lebesgue:measurable}{मापनीय}
प्रतिचित्रण $X \colon \Omega \to \R$ है (अथवा $\R^d$: यादृच्छिक सदिश); उसका \emph{बंटन
नियम}\index{यादृच्छिक चर का बंटन नियम} $\R$ पर अग्रसारित प्रायिकता
\omterm{def:b3:measure:measure}{माप} $\P_X =
X_*\P$ है (\cref{exo:b3:product:9}), जो \emph{बंटन फलन} $F_X(t) = \P(X \leq t)$ (\cref{exo:b3:measure:3}) से निर्धारित
होता है। $X$ का \emph{\omterm{ex:b3:lebesgue:gamma}{घनत्व}} $f$ है यदि $\P_X = f\,\dd\lambda$ हो; और वह
\emph{विविक्त} है यदि $\P_X$ दिराक द्रव्यमानों का कोई गणनीय संचय हो।
\emph{प्रत्याशा}\index{प्रत्याशा}
\[
\E[X] = \int_\Omega X\,\dd\P
\qquad (X \geq 0 \text{ अथवा } X \in L^1(\P)),
\]
है, और \emph{अंतरण प्रमेय} (\cref{exo:b3:product:9}) उसे बंटन नियम में परिकलित कर देती
है: $\E[g(X)] = \int_\R g\,\dd\P_X$ --- विविक्त स्थिति में $= \sum g(x_k)p_k$, और \omterm{ex:b3:lebesgue:gamma}{घनत्व} वाली स्थिति में
$= \int
g(x)f(x)\dd x$: अर्थात् दूसरे वर्ष के सूत्र, अब एक ही सिद्धांत की प्रमेयें।
$X \in L^2$ के लिए \emph{प्रसरण} $\V(X) =
\E[(X - \E X)^2] = \E[X^2] - (\E X)^2$ है।
\end{definition}

\begin{example}\label{ex:b3:probability:laws}
उल्लेखनीय मानक \omterm{def:b3:probability:space}{बंटन नियम} और उनके रूपांतर: बर्नूली $\mathcal B(p)$, द्विपद
$\mathcal B(n, p)$, ज्यामितीय, प्वासों $\mathcal P(\lambda)$ (विविक्त: दूसरे वर्ष की सारणियाँ मान्य
बनी रहती हैं); $\intcc01$ पर एकसमान (स्वयं \omterm{def:b3:measure:lebesgueouter}{लेबेग माप}); चरघातांकी $\mathcal E(\lambda)$
(\omterm{ex:b3:lebesgue:gamma}{घनत्व} $\lambda\eu^{-\lambda x}\mathbf 1_{x>0}$); और \emph{गाउसीय}\index{गाउसीय बंटन नियम} $\mathcal N(m, \sigma^2)$, जिसका
\omterm{ex:b3:lebesgue:gamma}{घनत्व} $\frac1{\sigma\sqrt{2\pi}}\exp\bigl(-\frac{(x -
m)^2}{2\sigma^2}\bigr)$ है --- यह \cref{pb:b3:lebesgue:1} से एक प्रायिकता \omterm{ex:b3:lebesgue:gamma}{घनत्व} है, जिसका माध्य
$m$ और प्रसरण $\sigma^2$ है (गाउसीय आघूर्ण, \cref{exo:b3:product:10})।
\end{example}

\begin{proposition}[मार्कोव और चेबिशेव]
\label{prop:b3:probability:markov}
$X \geq 0$ और $a > 0$ के लिए: $\P(X \geq a) \leq \frac{\E
X}{a}$; और $X \in L^2$ के लिए: $\P\bigl(\abs{X - \E X} \geq
a\bigr) \leq \frac{\V(X)}{a^2}$।\index{मार्कोव
असमिका}\index{चेबिशेव असमिका}
\end{proposition}

\begin{proof}
\cref{exo:b3:lebesgue:5}(क); और चेबिशेव $(X - \E X)^2$ पर लगाई गई मार्कोव ही है।
\end{proof}

\section{स्वातंत्र्य}

\begin{definition}\label{def:b3:probability:independence}
उप-$\sigma$-बीजगणित $\mathcal A_1, \dots, \mathcal A_n
\subseteq \mathcal A$ \emph{स्वतंत्र}\index{स्वातंत्र्य} कहलाते हैं
यदि सभी $A_i \in
\mathcal A_i$ के लिए $\P(A_1\cap\dots\cap A_n) = \prod\P(A_i)$ हो; घटनाएँ स्वतंत्र कहलाती हैं यदि
$\sigma$-बीजगणित $\{\varnothing, A_i, A_i^c, \Omega\}$ स्वतंत्र हों; और \omterm{def:b3:probability:space}{यादृच्छिक चर} $X_1, \dots, X_n$ स्वतंत्र
कहलाते हैं यदि $\sigma$-बीजगणित $\sigma(X_i) =
X_i^{-1}(\mathcal B(\R))$ स्वतंत्र हों। कोई अनंत कुल स्वतंत्र
कहलाता है यदि उसका प्रत्येक परिमित उपकुल स्वतंत्र हो।
\end{definition}

\begin{theorem}\label{thm:b3:probability:independence}
$X_1, \dots, X_n$ \omterm{def:b3:probability:independence}{स्वतंत्र} हैं तभी और केवल तभी जब सदिश $(X_1, \dots, X_n)$ का \omterm{def:b3:probability:space}{बंटन नियम} \omterm{thm:b3:product:existence}{गुणन
माप} $\P_{X_1}\otimes\cdots\otimes\P_{X_n}$ हो। उस स्थिति में $g_i \geq 0$ के लिए (अथवा ऐसे फलनों के लिए जिनके
गुणनफल \omterm{def:b3:lebesgue:l1}{समाकलनीय} हों):
\[
\E\Bigl[\prod_ig_i(X_i)\Bigr] = \prod_i\E[g_i(X_i)],
\]
विशेष रूप से, \omterm{def:b3:probability:independence}{स्वतंत्र} $L^2$ चरों के लिए $\E[XY] = \E X\,\E Y$ और $\V(X_1 + \dots +
X_n) = \sum\V(X_i)$।
\end{theorem}

\begin{proof}
यदि $X_i$ \omterm{def:b3:probability:independence}{स्वतंत्र} हों, तो दोनों प्रायिकता \omterm{def:b3:measure:measure}{माप} $\P_{(X_1,\dots,X_n)}$ और $\bigotimes\P_{X_i}$ बोरेल
समुच्चयों के सभी गुणनफलों $B_1\times\dots\times B_n$ पर सहमत होते हैं --- और ये $\mathcal B(\R^n)$ को
जनित करने वाली एक $\pi$-प्रणाली हैं (\cref{prop:b3:product:sections}(ख)) --- अतः वे सर्वत्र
सहमत हैं (\cref{thm:b3:measure:uniqueness})। विलोमतः, कोई गुणनफल \omterm{def:b3:probability:space}{बंटन नियम} सभी घटनाओं $\bigcap_iX_i^{-1}(B_i)$ को
गुणनखंडित कर देता है: अर्थात् \omterm{def:b3:probability:independence}{स्वातंत्र्य}। तब \omterm{def:b3:probability:space}{प्रत्याशा} सूत्र अंतरण
प्रमेय के माध्यम से टोनेली/फुबिनी (\cref{thm:b3:product:tonelli}) है; $\E[XY] = \E X\E Y$ स्थिति $g_i =
\mathrm{id}$ है,
और वर्ग का प्रसार प्रसरणों की योज्यता दे देता है (तिर्यक पद $\E[(X_i - \E X_i)(X_j - \E X_j)]
= 0$)।
\end{proof}

\begin{theorem}[स्वतंत्र अनुक्रमों का अस्तित्व]
\label{thm:b3:probability:existence}
$\bigl(\intcc01, \mathcal L, \lambda\bigr)$ पर \omterm{def:b3:probability:independence}{स्वतंत्र} यादृच्छिक चरों का ऐसा अनुक्रम $(U_n)_{n\geq1}$ विद्यमान है
जिसका प्रत्येक पद $\intcc01$ पर एकसमान है। परिणामतः, $\R$ पर किन्हीं भी
निर्धारित बंटन नियमों $(\mu_n)$ के लिए $\P_{X_n} = \mu_n$ वाले \omterm{def:b3:probability:independence}{स्वतंत्र} $(X_n)$ विद्यमान
हैं।\index{स्वतंत्र अनुक्रम, अस्तित्व}
\end{theorem}

\begin{proof}
\emph{अंक।} $\omega \in \intcc01$ के लिए मान लीजिए $(b_k(\omega))$ उसके द्विआधारी अंक हैं
($\omega = \sum b_k2^{-k}$; वह प्रसार चुनिए जो अंततः सर्वथा $1$ न हो --- यह अस्पष्टता
केवल एक गणनीय, अतः अकिंचन, समुच्चय से संबंधित है)। प्रत्येक $b_k$
\omterm{def:b3:probability:space}{यादृच्छिक चर} है ($\{b_k = 1\}$ द्विअंशी अंतरालों का परिमित संघ है) और सदिश
$(b_1, \dots, b_m)$ $\{0,1\}^m$ का प्रत्येक मान लंबाई $2^{-m}$ के किसी द्विअंशी अंतराल पर
लेता है: अतः $b_k$ \omterm{def:b3:probability:independence}{स्वतंत्र} बर्नूली$(\frac12)$ हैं।

\emph{पुनःसमूहन।} $\N^*$ को अनंत संख्या के असंयुक्त अनंत समुच्चयों
$(I_n)$ में विभाजित कीजिए (उदाहरणार्थ अभाज्य घातों से, या विकर्णों
से); मान लीजिए $(k^n_j)_j$ $I_n$ की गणना करता है, और
\[
U_n = \sum_{j\geq1} b_{k^n_j}\,2^{-j} .
\]
रखिए। प्रत्येक $U_n$ एकसमान है: उसके द्विआधारी अंक \omterm{def:b3:probability:independence}{स्वतंत्र} निष्पक्ष
बिट हैं, अतः प्रत्येक द्विअंशी अंतराल के लिए $\P(U_n \in [l2^{-m}, (l+1)2^{-m})) = 2^{-m}$, और द्विअंशी
अंतराल \omterm{def:b3:probability:space}{बंटन नियम} को निर्धारित कर देते हैं (\cref{thm:b3:measure:uniqueness})। $U_n$ \omterm{def:b3:probability:independence}{स्वतंत्र}
हैं: वे \omterm{def:b3:probability:independence}{स्वतंत्र} कुल $(b_k)$ के असंयुक्त खंडों के फलन हैं ---
औपचारिक रूप से, द्विअंशी $D_n$ के लिए घटनाएँ $\{U_n \in D_n\}$ असंयुक्त
समुच्चयों के परिमित संख्या के अंकों पर निर्भर करती हैं और गुणनखंडित
हो जाती हैं; और $\pi$-प्रणाली वाला तर्क इसे सभी बोरेल समुच्चयों तक
उन्नत कर देता है।

\emph{स्वेच्छ \omterm{def:b3:probability:space}{बंटन नियम}।} मान लीजिए $G_n(u) = \inf\{t : F_{\mu_n}(t)
\geq u\}$ (बंटन फलन $F_{\mu_n}$ का
\emph{क्वांटाइल फलन}); मुख्य तुल्यता $G_n(u) \leq t
\iff u \leq F_{\mu_n}(t)$ ($F$ की दक्षिण-सांतत्य
और एकदिष्टता) दिखा देती है कि $X_n = G_n(U_n)$ $\P(X_n \leq t) = \P(U_n \leq F_{\mu_n}(t)) =
F_{\mu_n}(t)$ के साथ \omterm{def:b3:lebesgue:measurable}{मापनीय} है: अर्थात्
\omterm{def:b3:probability:space}{बंटन नियम} $\mu_n$; और \omterm{def:b3:probability:independence}{स्वातंत्र्य} विरासत में मिल जाता है (\omterm{def:b3:probability:independence}{स्वतंत्र}
चरों के फलन, \cref{exo:b3:probability:3})।
\end{proof}

\begin{example}[जन्मदिन समस्या, ईमानदारी से]
\label{ex:b3:probability:birthday}
$N = 365$ दिनों पर \omterm{def:b3:probability:independence}{स्वतंत्र}, एकसमान जन्मदिनों वाले $n$ व्यक्तियों में
सभी जन्मदिनों के भिन्न होने की प्रायिकता पुनरावृत्त सप्रतिबंधन से
\[
p_n = \prod_{k=1}^{n-1}\Bigl(1 - \frac kN\Bigr),
\]
है (अथवा सीधे: कुल $N^n$ में से अनुकूल $N(N-1)\cdots(N - n + 1)$, और \omterm{def:b3:probability:independence}{स्वातंत्र्य} का
गुणनफल सूत्र इस गणना-तर्क को कठोर बना देता है)। लघुगणक लेकर और
$-\ln(1 - x) = x +
O(x^2)$ का उपयोग करके:
\[
\ln p_n = -\frac{n(n-1)}{2N} +
O\Bigl(\frac{n^3}{N^2}\Bigr),
\qquad\text{अतः}\qquad
p_n \approx \eu^{-n^2/2N} .
\]
पलटाव बिंदु $p_n = \frac12$ $n \approx
\sqrt{2N\ln2} \approx 1.18\sqrt N$ पर बैठता है: $N = 365$ के लिए $n = 23$ ($p_{23} = 0.4927$)।
दो शिक्षाएँ। पहली, $N$ पेटियों में $n$ वस्तुओं के बीच टकराव
$n \sim \sqrt N$ के मापक्रम पर प्रकट होते हैं, न कि $n \sim N$ पर --- यही वह
\emph{जन्मदिन मापन} है जो हैश टकरावों को और कूटलेखन में जन्मदिन
आक्रमणों की $\sqrt N$ लागत को नियंत्रित करता है। दूसरी, यह परिकलन एक
प्रतिमान है: $\binom n2$ युग्म-टकराव घटनाएँ \omterm{def:b3:probability:independence}{स्वतंत्र} नहीं हैं, फिर भी
उत्तर ऐसे व्यवहार करता है मानो वे \omterm{def:b3:probability:independence}{स्वतंत्र} हों ($\eu^{-\binom n2/N}$ ठीक-ठीक
स्वतंत्र-युग्मों वाली अनुमान-विधि है) --- यह प्वासों आसन्नन का पहला
उदाहरण है, जिसे \cref{ch:b3:clt} की सप्ताहांत समस्या में कठोर बनाया गया है
(ले काम असमिका)।
\end{example}

\section{बोरेल--कांतेली और शून्य--एक नियम}

\begin{theorem}[बोरेल--कांतेली]
\label{thm:b3:probability:borelcantelli}
मान लीजिए $(A_n)$ घटनाएँ हैं और $\limsup A_n = \bigcap_N
\bigcup_{n\geq N}A_n$ (``$A_n$ अनंत बार
घटित होती है'')।\index{बोरेल--कांतेली प्रमेयिकाएँ}
\begin{enumerate}
  \item यदि $\sum\P(A_n) < \infty$ हो, तो $\P(\limsup A_n) =
        0$।
  \item यदि $\sum\P(A_n) = \infty$ हो \emph{और $A_n$ \omterm{def:b3:probability:independence}{स्वतंत्र} हों}, तो $\P(\limsup A_n) = 1$।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) \cref{exo:b3:measure:4} है। (2): $N \leq M$ के लिए पूरकों का \omterm{def:b3:probability:independence}{स्वातंत्र्य} (\cref{exo:b3:probability:3})
\[
\P\Bigl(\bigcap_{n=N}^{M}A_n^c\Bigr) = \prod_{n=N}^M\bigl(1
- \P(A_n)\bigr) \leq
\exp\Bigl(-\sum_{n=N}^M\P(A_n)\Bigr) \xrightarrow[M \to
\infty]{} 0
\]
दे देता है ($1 - x \leq \eu^{-x}$; और यह श्रेणी अपसारी है)। अतः प्रत्येक $N$ के
लिए $\P\bigl(\bigcup_{n\geq N}A_n\bigr) = 1$, और $N$ पर ह्रासमान सर्वनिष्ठ की प्रायिकता फिर भी $1$
है (ऊपर से \omterm{def:b3:topology:continuity}{सांतत्य}, \cref{prop:b3:measure:basics})।
\end{proof}

\begin{theorem}[कोल्मोगोरोव का शून्य--एक नियम]
\label{thm:b3:probability:zeroone}
मान लीजिए $(X_n)$ \omterm{def:b3:probability:independence}{स्वतंत्र} हैं और $\mathcal T =
\bigcap_N\sigma(X_N, X_{N+1}, \dots)$ \emph{पुच्छ
$\sigma$-बीजगणित}\index{पुच्छ $\sigma$-बीजगणित} है (अर्थात् वे घटनाएँ जो
$X_n$ में से किसी भी परिमित संख्या के प्रति असंवेदी हैं: $\sum X_n$ का
अभिसरण, $\frac{S_n}n$ का अभिसरण, $\limsup$ के मान, \dots)। तब प्रत्येक $T \in \mathcal T$
के लिए $\P(T) \in \{0,
1\}$।\index{शून्य--एक नियम}
\end{theorem}

\begin{proof}
$N$ स्थिर कीजिए। $\sigma$-बीजगणित $\sigma(X_1, \dots, X_N)$ और $\sigma(X_{N+1}, \dots)$ \omterm{def:b3:probability:independence}{स्वतंत्र} हैं:
असंयुक्त खंडों पर निर्भर घटनाएँ जनक $\pi$-प्रणालियों पर गुणनखंडित
हो जाती हैं (एक ओर बेलन $\bigcap_{i\leq N}\{X_i \in B_i\}$, और दूसरी ओर बाद के चरों पर परिमित
प्रतिबंध), और दिन्किन (\cref{thm:b3:measure:dynkin}, दो बार लगाई गई, एक बार में एक ओर) उस
गुणनखंडन को बढ़ा देती है। कोई पुच्छ घटना $T$ प्रत्येक $N$ के
लिए $\sigma(X_{N+1}, \dots)$ में होती है: अतः $T$ प्रत्येक $\sigma(X_1, \dots, X_N)$ से \omterm{def:b3:probability:independence}{स्वतंत्र} है, और
इसलिए उनसे जनित $\sigma$-बीजगणित $\sigma(X_1, X_2, \dots)$ से भी (एक बार फिर दिन्किन:
$\sigma(X_1,\dots,X_N)$ का संघ उसे जनित करने वाली $\pi$-प्रणाली है)। पर $T \in \sigma(X_1, X_2, \dots)$ भी:
अतः $T$ \emph{स्वयं से} \omterm{def:b3:probability:independence}{स्वतंत्र} है, $\P(T) = \P(T\cap T) =
\P(T)^2$: इसलिए $\P(T) \in \{0, 1\}$।
\end{proof}

\section{अभिसरण के प्रकार}

\begin{definition}\label{def:b3:probability:modes}
$X_n \to X$ \emph{लगभग निश्चित रूप से} यदि $\P(X_n \to X) = 1$ हो; \emph{प्रायिकता में}
यदि प्रत्येक $\varepsilon > 0$ के लिए $\P(\abs{X_n - X} \geq \varepsilon)
\to 0$ हो; और \emph{$L^p$ में} यदि $\E\abs{X_n - X}^p \to 0$
हो।\index{यादृच्छिक चरों का अभिसरण}
\end{definition}

\begin{proposition}\label{prop:b3:probability:modes}
(क) लगभग निश्चित अभिसरण से प्रायिकता में अभिसरण निकलता है;
(ख) $L^p$ अभिसरण से प्रायिकता में अभिसरण निकलता है;
(ग) प्रायिकता में अभिसरण से \emph{किसी उपानुक्रम के अनुदिश} लगभग
निश्चित अभिसरण निकलता है;
(घ) और व्यापक रूप से कोई अन्य निहितार्थ लागू नहीं होता।
\end{proposition}

\begin{proof}
(क) लगभग निश्चित अभिसरण के अंतर्गत $\P(\abs{X_n - X} \geq \varepsilon) \leq
\P\bigl(\sup_{m\geq n}\abs{X_m - X} \geq \varepsilon\bigr)
\downarrow \P\bigl(\limsup\{\abs{X_m - X} \geq
\varepsilon\}\bigr) = 0$ (ऊपर से \omterm{def:b3:topology:continuity}{सांतत्य}; उच्च-सीमा
घटना अभिसरण को अपवर्जित कर देती है)। (ख) मार्कोव: $\P(\abs{X_n - X} \geq \varepsilon) \leq
\varepsilon^{-p}\,\E\abs{X_n - X}^p$। (ग) $\P(\abs{X_{n_k} - X} \geq 2^{-k}) \leq 2^{-k}$
वाले $n_k$ चुनिए; बोरेल--कांतेली (1) से अंततः, लगभग निश्चित रूप से,
$\abs{X_{n_k} - X} < 2^{-k}$ हो जाता है। (घ) $(\intcc01, \lambda)$ पर टंकण-यंत्र (\cref{exo:b3:lp:3}) $L^1$ में और
प्रायिकता में अभिसरित होता है पर कहीं भी बिंदुशः नहीं; $n\mathbf 1_{\intoo0{1/n}} \to 0$ लगभग
निश्चित रूप से, पर $L^1$ में नहीं; विवरण और शेष प्रतिउदाहरण \cref{exo:b3:probability:6}
में।
\end{proof}

\section{बृहत् संख्याओं का नियम}

सर्वत्र $(X_n)$ एक ही \omterm{def:b3:probability:space}{बंटन नियम} वाले \omterm{def:b3:probability:independence}{स्वतंत्र} चर हैं
(\emph{\omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित}), और $S_n = X_1 + \dots + X_n$।

\begin{theorem}[बृहत् संख्याओं का दुर्बल नियम]
\label{thm:b3:probability:wlln}
यदि $X_1 \in L^2$ हो, $m = \E X_1$ के साथ:
\[
\P\Bigl(\Bigl|\frac{S_n}{n} - m\Bigr| \geq
\varepsilon\Bigr) \leq
\frac{\V(X_1)}{n\,\varepsilon^2}
\xrightarrow[n\to\infty]{} 0 :
\]
तो $\frac{S_n}n \to m$ प्रायिकता में (और $L^2$ में भी)।\index{बृहत् संख्याओं का
नियम (दुर्बल)}
\end{theorem}

\begin{proof}
$\E\frac{S_n}n = m$ और $\V\bigl(\frac{S_n}n\bigr) =
\frac{n\V(X_1)}{n^2}$ (\cref{thm:b3:probability:independence}); फिर चेबिशेव।
\end{proof}

\begin{theorem}[बृहत् संख्याओं का प्रबल नियम]
\label{thm:b3:probability:slln}
यदि $X_1 \in L^1$ हो, तो\index{बृहत् संख्याओं का नियम (प्रबल)}
\[
\frac{S_n}{n} \xrightarrow[n\to\infty]{\text{लगभग निश्चित}} \E[X_1].
\]
हम इसे यहाँ प्रबलतर परिकल्पना $X_1 \in
L^4$ के अंतर्गत सिद्ध करते हैं;
व्यापक स्थिति ($L^1$: एतेमादी की उपपत्ति) सप्ताहांत समस्या है।
\end{theorem}

\begin{proof}[$\E X_1^4 < \infty$ के अंतर्गत उपपत्ति]
केंद्रीकरण ($X_i \mapsto X_i - m$) करके मान लीजिए $m = 0$। प्रसार कीजिए:
\[
\E[S_n^4] = \sum_{i,j,k,l}\E[X_iX_jX_kX_l]
= n\,\E[X_1^4] + 3n(n-1)\,\bigl(\E[X_1^2]\bigr)^2 \leq
C\,n^2 ,
\]
क्योंकि \omterm{def:b3:probability:independence}{स्वातंत्र्य} और केंद्रीकरण हर उस पद को मार देते हैं जिसमें
कोई एकाकी गुणनखंड हो ($\E[X_iX_jX_kX_l] =
\E[X_i]\E[\cdots] = 0$ जब तक सूचकांक युग्मित न हों: केवल
$n$ पद $i=j=k=l$ और दो भिन्न युग्मों वाले $3n(n-1)$ पद बचते हैं)।
मार्कोव:
\[
\P\Bigl(\Bigl|\frac{S_n}n\Bigr| \geq \varepsilon\Bigr)
= \P\bigl(S_n^4 \geq n^4\varepsilon^4\bigr)
\leq \frac{Cn^2}{n^4\varepsilon^4} =
\frac{C}{\varepsilon^4n^2},
\]
जो योगनीय है: अतः बोरेल--कांतेली (1) से प्रत्येक परिमेय $\varepsilon$ के
लिए अंततः, लगभग निश्चित रूप से, $\abs{S_n/n} < \varepsilon$; और $\varepsilon \in \Q_+^*$ पर सर्वनिष्ठ लेकर
(गणनीय संख्या की प्रायिकता-$1$ घटनाएँ): $S_n/n \to 0$ लगभग निश्चित रूप
से।
\end{proof}

\begin{example}[प्रबल नियम से क्या मिलता है]
\label{ex:b3:probability:sllnapps}
(क) \emph{आवृत्तियाँ}: \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित सिक्का-उछालों के लिए चित
की प्रेक्षित आवृत्ति लगभग निश्चित रूप से $p$ की ओर अभिसरित होती
है --- अर्थात् स्वयं प्रायिकता का आनुभविक औचित्य।
(ख) \emph{मोंते कार्लो}\index{मोंते कार्लो विधि}: $g \in
L^1(\intcc01)$ और \omterm{def:b3:probability:independence}{स्वतंत्र}
सम-बंटित एकसमान $(U_n)$ (\cref{thm:b3:probability:existence}) के लिए लगभग निश्चित रूप से $\frac1n\sum_{k\leq n}g(U_k) \to \int_0^1g$:
अर्थात् प्रतिचयन से समाकल, किसी भी विमा में, \cref{ch:b3:clt} में परिशुद्ध की
गई विमा-निरपेक्ष दर $\sim n^{-1/2}$ पर।
(ग) \emph{सामान्य संख्याएँ}: लगभग प्रत्येक वास्तविक संख्या के
द्विआधारी प्रसार में एकों की अनंतस्पर्शी आवृत्ति $\frac12$ होती है
(\cref{thm:b3:probability:existence} के अंक चरों पर प्रबल नियम लगाइए) --- यह बोरेल की प्रमेय है,
जो \emph{रोज़मर्रा} की संख्याओं के बारे में एक कथन है और \omterm{def:b3:measure:measure}{माप} से सिद्ध
होता है: \cref{pb:b3:probability:1} उसे सभी आधारों में पूरा कर देता है।
\end{example}

\begin{method}\label{met:b3:probability:toolkit}
यादृच्छिक अनुक्रमों के बारे में अनंतस्पर्शी कथनों के लिए कार्य-क्रम:
(1) \emph{क्या यह घटना कोई पुच्छ घटना है?} तब उसकी प्रायिकता $0$
या $1$ है (\cref{thm:b3:probability:zeroone}) और केवल यह तय करना शेष रहता है कि कौन-सी।
(2) \emph{लगभग निश्चित कथन सिद्ध करने के लिए}: बोरेल--कांतेली ---
जो भी आघूर्ण विद्यमान हों उन पर मार्कोव/चेबिशेव-प्रकार के परिबंधों
के माध्यम से ``बुरी'' घटनाओं की योगनीय प्रायिकताएँ; और \omterm{def:b3:probability:independence}{स्वातंत्र्य}
केवल विपरीत दिशा के लिए चाहिए। (3) \emph{उपानुक्रम + सैंडविच}: किसी
सुप्रबंध्य उपानुक्रम के अनुदिश अभिसरण सिद्ध कीजिए, और बीच के दोलन
को एकदिष्टता या उच्चिष्ठ असमिकाओं से नियंत्रित कीजिए --- यही
एतेमादी की उपपत्ति का कंकाल है। (4) बंटन-संबंधी सीमाओं के लिए \cref{ch:b3:clt}
की प्रतीक्षा कीजिए।
\end{method}

पाठ को अनंत बार धारण करता है: असंयुक्त खंडों पर बोरेल--कांतेली (2)

\end{exercise}

और जाँचिए कि उसके लिए \omterm{thm:b3:probability:zeroone}{शून्य--एक नियम} विफल हो सकता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:6}
$(\intcc01, \lambda)$ पर --- उपपत्तियों सहित --- ऐसे \omterm{def:b3:probability:space}{यादृच्छिक चर} प्रस्तुत कीजिए
कि:
(क) $X_n \to 0$ प्रायिकता में और प्रत्येक $L^p$ में, पर कहीं भी लगभग
निश्चित रूप से नहीं;
(ख) $X_n \to 0$ लगभग निश्चित रूप से, पर किसी भी $L^p$ में नहीं;
(ग) $X_n \to 0$ $L^1$ में, पर $L^2$ में नहीं;
(घ) और दर्शाइए: यदि $X_n \to X$ प्रायिकता में हो और $\abs{X_n}
\leq Y \in L^1$ हो, तो $X_n \to X$
$L^1$ में \emph{(उपानुक्रम + प्रभावी अभिसरण + उप-उपानुक्रम वाली
युक्ति)}।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:7}
कोई जनमत सर्वेक्षण किसी अज्ञात अनुपात $p$ का आकलन $n$ \omterm{def:b3:probability:independence}{स्वतंत्र}
प्रतिचयनों की आनुभविक आवृत्ति $\hat p_n$ से करता है।
(क) चेबिशेव: दर्शाइए कि $\P(\abs{\hat p_n - p} \geq \varepsilon)
\leq \frac1{4n\varepsilon^2}$ ($p(1-p) \leq \frac14$ का उपयोग कीजिए)।
(ख) इस परिबंध से कितने प्रतिचयन $\geq 95\%$ प्रायिकता के साथ $\leq 3\%$ त्रुटि
की प्रत्याभूति देते हैं? (\cref{ch:b3:clt} के माध्यम से वास्तविक उत्तर लगभग
$1070$ है: चेबिशेव ईमानदार है पर कच्चा।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:probability:8}
(बर्नस्टाइन) $f \in \mathcal C(\intcc01)$ के लिए बर्नस्टाइन बहुपद $B_nf(x) =
\sum_{k=0}^n\binom nkx^k(1-x)^{n-k}f\bigl(\frac kn\bigr)$ परिभाषित कीजिए।
(क) द्विपद $\mathcal B(n, x)$ वाले $S_n$ के लिए $B_nf(x) = \E\bigl[f\bigl(\frac
{S_n}n\bigr)\bigr]$ को पहचानिए।
(ख) सिद्ध कीजिए कि $\intcc01$ पर $B_nf \to f$ \emph{एकसमान रूप से}: $\{\abs{\frac{S_n}n - x} \leq \delta\}$ और
उसके पूरक पर विभाजन कीजिए, और एकसमान \omterm{def:b3:topology:continuity}{सांतत्य} तथा एकसमान परिबंध
$\V(\frac{S_n}n) \leq \frac1{4n}$ के साथ चेबिशेव का उपयोग कीजिए।
(ग) निष्कर्ष निकालिए: वाइरश्ट्रास आसन्नन प्रमेय (\cref{cor:b3:complete:weierstrass}) की एक
दूसरी, प्रायिकता-आधारित उपपत्ति, और साथ में \omterm{def:b3:topology:continuity}{सांतत्य} मापांक $\omega_f$
के लिए स्पष्ट दर
$\norm{B_nf - f}_\infty \leq \frac32\,\omega_f(n^{-1/2})$
--- कम से कम $O(\omega_f(n^{-1/2}))$ वाला रूप सिद्ध कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:probability:9}
(कूपन संग्राहक) $n$ प्रकारों के पत्ते प्रतिस्थापन के साथ एकसमान
रूप से निकाले जाते हैं; मान लीजिए $T_n$ वह प्राचय संख्या है जब तक
सभी प्रकार दिख न जाएँ।
(क) $T_n = \sum_{k=1}^{n}\tau_k$ लिखिए, जहाँ $\tau_k$ प्राचल $\frac{n - k + 1}n$ वाले ज्यामितीय हैं और
$\tau_k$ \omterm{def:b3:probability:independence}{स्वतंत्र} हैं, और $\E T_n = n\,H_n \sim n\ln n$ निकालिए ($H_n$ प्रसंवादी संख्या) तथा
$\V(T_n) \leq
\frac{\pi^2}6n^2$।
(ख) चेबिशेव: प्रायिकता में $\frac{T_n}{n\ln n} \to 1$।
(ग) बोरेल--कांतेली से परिष्कार कीजिए: $\beta > 1$ के लिए सीधे $\P(T_n >
\beta n\ln n) \leq n^{1 - \beta}$
दर्शाइए \emph{($\beta n\ln n$ प्राचयों के बाद किसी प्रकार के छूट जाने की घटना
पर संघ परिबंध, $1 - x \leq \eu^{-x}$ का उपयोग करते हुए)}, और निकालिए कि $n = 2^m$ के
अनुदिश, प्रत्येक $\beta > 2$ के लिए, लगभग निश्चित रूप से अंततः $T_n \leq \beta n\ln
n$।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:10}
अंक-रचना (\cref{thm:b3:probability:existence}) का उपयोग करते हुए:
(क) सीधे परिकलन से सत्यापित कीजिए कि $U = \sum b_{2k}2^{-k}$ (किसी एकसमान $\omega$ के
सम-सूचकांक वाले अंक) एकसमान है और $V = \sum b_{2k-1}2^{-k}$ से \omterm{def:b3:probability:independence}{स्वतंत्र} है;
(ख) इससे $\intcc01$ और $\intcc01^2$ के बीच माप-संरक्षी, अकिंचन-समुच्चयों-तक
\omterm{def:b3:lebesgue:measurable}{मापनीय} एकैकी आच्छादन निकालिए, और टिप्पणी कीजिए: एक ही एकसमान
यादृच्छिक संख्या दो (और गणनीय अनंत) \omterm{def:b3:probability:independence}{स्वतंत्र} संख्याएँ धारण करती है
--- इसकी तुलना पियानो वक्र (\cref{pb:b3:topology:1}) से कीजिए, जिसने आच्छादकता तो
प्राप्त की थी पर माप-संरक्षण या एकैकीपन नहीं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:11}
(अभिलेख) मान लीजिए $(X_n)_{n\geq1}$ \omterm{def:b3:topology:continuity}{संतत} बंटन फलन वाले \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित चर
हैं, और कहिए कि समय $n$ पर कोई \emph{अभिलेख} घटित होता है यदि
$X_n > \max(X_1, \dots, X_{n-1})$ हो (समय $1$ अभिलेख है)। मान लीजिए $R_n$ अभिलेख सूचक है।
(क) दर्शाइए कि $\P(R_n = 1) = \frac1n$ \emph{(सममिति से, $X_1, \dots, X_n$ के $n!$ क्रमों में
से प्रत्येक समान रूप से संभाव्य है और बराबरी की प्रायिकता $0$
है)}।
(ख) दर्शाइए कि $R_n$ \emph{\omterm{def:b3:probability:independence}{स्वतंत्र}} हैं \emph{(निर्धारित अभिलेख
स्थानों के अनुकूल क्रमों की गणना कीजिए, अथवा यह तर्क दीजिए कि
$X_1, \dots, X_{n-1}$ का सापेक्ष क्रम उनमें $X_n$ की कोटि से \omterm{def:b3:probability:independence}{स्वतंत्र} है)}।
(ग) बोरेल--कांतेली (\cref{thm:b3:probability:borelcantelli}, दोनों अर्ध भाग) से निकालिए कि लगभग
निश्चित रूप से अनंत अभिलेख घटित होते हैं, पर क्रमागत समयों $n, n+1$
पर अभिलेख अनंत बार किस प्रायिकता के साथ घटित होते हैं --- तय कीजिए
कि कौन-सी! --- और $\sum_n\P(R_n = 1, R_{n+1} = 1)$ परिकलित कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:12}
(सबसे लंबी चित लड़ी) कोई निष्पक्ष सिक्का अनंत बार उछालिए, और मान
लीजिए $L_n$ प्रथम $n$ उछालों के भीतर क्रमागत चितों की सबसे लंबी
लड़ी की लंबाई है।
(क) दर्शाइए कि प्रत्येक $\varepsilon > 0$ के लिए लगभग निश्चित रूप से अंततः $L_n \leq
(1 + \varepsilon)\log_2n$
\emph{(प्रथम $n$ उछालों में लंबाई $\ell$ की कोई लड़ी आरंभ होने की
प्रायिकता अधिक से अधिक $n2^{-\ell}$ है; $n =
2^k$ के अनुदिश बोरेल--कांतेली)}।
(ख) दर्शाइए कि लगभग निश्चित रूप से अंततः $L_n \geq (1 - \varepsilon)\log_2n$ \emph{(प्रथम $n$
उछालों को लंबाई $\ell =
\lceil(1 - \varepsilon)\log_2n\rceil$ के $\lfloor n/\ell\rfloor$ असंयुक्त खंडों में काटिए; ये खंड
\omterm{def:b3:probability:independence}{स्वतंत्र} हैं, प्रत्येक प्रायिकता $2^{-\ell}$ के साथ सर्वथा चित है, और
किसी के भी सर्वथा चित न होने की प्रायिकता अधिक से अधिक $\exp(-n2^{-\ell}/\ell)$ है;
फिर से $n = 2^k$ के अनुदिश योग कीजिए)}।
(ग) निष्कर्ष निकालिए कि लगभग निश्चित रूप से $\frac{L_n}{\log_2n} \to 1$: दस लाख निष्पक्ष
उछालों में लगभग $20$ चितों की लड़ी की अपेक्षा करनी चाहिए --- और
जिस आँकड़ा-समुच्चय में ऐसी लड़ी न हो वह संभवतः गढ़ा हुआ है।
\end{exercise}

\section{समस्या: प्रबल नियम की एतेमादी वाली उपपत्ति}

\begin{problem}[{सप्ताहांत समस्या --- स्वतंत्र सम-बंटित समाकलनीय
चरों के लिए बृहत् संख्याओं का प्रबल नियम}]
\label{pb:b3:probability:1}
कोल्मोगोरोव के प्रबल नियम --- \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित $X_n \in L^1$ के लिए लगभग
निश्चित रूप से $\frac{S_n}n \to \E X_1$ --- की बहुत समय तक केवल जटिल उपपत्तियाँ ही
थीं; 1981 में एन.~एतेमादी ने आश्चर्यजनक मितव्ययिता वाली एक उपपत्ति
खोजी, जो इस अध्याय से बाहर कुछ भी उपयोग नहीं करती (और \omterm{def:b3:probability:independence}{स्वातंत्र्य}
को दुर्बल करके युग्मशः \omterm{def:b3:probability:independence}{स्वातंत्र्य} तक भी ले आती है)। हम उसी का
अनुसरण करते हैं। मान लीजिए $(X_n)$ युग्मशः \omterm{def:b3:probability:independence}{स्वतंत्र}, सम-बंटित,
\omterm{def:b3:lebesgue:l1}{समाकलनीय} हैं; और $m = \E X_1$, $S_n = X_1 + \dots + X_n$।

\medskip
\noindent\textbf{भाग I --- न्यूनीकरण।}
\begin{enumerate}
  \item दर्शाइए कि केवल $X_n \geq 0$ की स्थिति देखना पर्याप्त है
        \emph{($X_n = X_n^+ - X_n^-$ को विभाजित कीजिए: जाँचिए कि दोनों अर्ध भाग
        पुनः युग्मशः \omterm{def:b3:probability:independence}{स्वतंत्र}, सम-बंटित और \omterm{def:b3:lebesgue:l1}{समाकलनीय} हैं)}। अब से
        $X_n \geq 0$ मान लीजिए।
  \item (कर्तन) मान लीजिए $Y_n = X_n\,\mathbf 1_{X_n \leq n}$ और $S_n^* = Y_1 + \dots + Y_n$। दर्शाइए कि
        \[
        \sum_{n\geq1}\P(X_n \neq Y_n) =
        \sum_{n\geq1}\P(X_1 > n) \leq \E[X_1] < \infty
        \]
        (\cref{exo:b3:product:3}), और बोरेल--कांतेली के माध्यम से निकालिए कि लगभग
        निश्चित रूप से $\frac{S_n - S_n^*}{n} \to 0$: अर्थात् केवल यह सिद्ध करना पर्याप्त
        है कि लगभग निश्चित रूप से $\frac{S^*_n}n \to m$।
  \item दर्शाइए कि $\E Y_n = \E\bigl[X_1\mathbf 1_{X_1\leq
        n}\bigr] \to m$ (एकदिष्ट अभिसरण), अतः $\frac1n\sum_{k\leq n}\E Y_k \to m$ (चेज़ारो):
        अर्थात् केवल यह सिद्ध करना पर्याप्त है कि लगभग निश्चित
        रूप से $\frac{S_n^* - \E S_n^*}{n} \to 0$।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- प्रसरण आकलन।}
\begin{enumerate}[resume]
  \item दर्शाइए कि
        \[
        \V(Y_n) \leq \E[Y_n^2] = \E\bigl[X_1^2\,\mathbf
        1_{X_1 \leq n}\bigr]
        \]
        और परत-केक सूत्र (\cref{prop:b3:product:layercake}) का उपयोग करते हुए मुख्य परिबंध
        \[
        \sum_{n\geq1}\frac{\V(Y_n)}{n^2}
        \leq \sum_{n\geq1}\frac1{n^2}\,
        \E\bigl[X_1^2\mathbf 1_{X_1\leq n}\bigr]
        \leq C\,\E[X_1] < \infty
        \]
        \emph{(योग और \omterm{def:b3:probability:space}{प्रत्याशा} की अदला-बदली कीजिए --- श्रेणियों
        के लिए टोनेली --- और आंतरिक आकलन $x^2\sum_{n\geq x}n^{-2} \leq 2x$ के लिए $\sum_{n \geq
        x}\frac1{n^2} \leq \frac2{\max(x,1)}$ को
        परिबद्ध कीजिए)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- ज्यामितीय उपानुक्रमों के अनुदिश
अभिसरण।} $\alpha > 1$ स्थिर कीजिए और मान लीजिए $k_j =
\lfloor\alpha^j\rfloor$।
\begin{enumerate}[resume]
  \item युग्मशः \omterm{def:b3:probability:independence}{स्वातंत्र्य} (प्रसरण जुड़ते हैं, \cref{thm:b3:probability:independence} --- जाँचिए
        कि प्रसरणों की योज्यता के लिए केवल युग्मशः \omterm{def:b3:probability:independence}{स्वातंत्र्य} ही
        चाहिए) और चेबिशेव का उपयोग करके प्रत्येक $\varepsilon > 0$ के लिए
        दर्शाइए:
        \[
        \sum_{j\geq1}\P\Bigl(\Bigl|
        \frac{S^*_{k_j} - \E S^*_{k_j}}{k_j}\Bigr| \geq
        \varepsilon\Bigr)
        \leq
        \frac1{\varepsilon^2}\sum_{j\geq1}\frac1{k_j^2}
        \sum_{n\leq k_j}\V(Y_n)
        = \frac1{\varepsilon^2}\sum_{n\geq1}\V(Y_n)
        \sum_{j\,:\,k_j\geq n}\frac1{k_j^2} .
        \]
  \item $\sum_{j : k_j \geq n}k_j^{-2} \leq
        \frac{C_\alpha}{n^2}$ दर्शाइए \emph{(ज्यामितीय श्रेणी; निम्नमान से सावधान:
        $\alpha^j \geq 2$-प्रकार की सावधानी के लिए $k_j \geq \frac{\alpha^j}2$)}, और प्रश्न 4 तथा
        बोरेल--कांतेली के साथ निष्कर्ष निकालिए:
        \[
        \frac{S^*_{k_j} - \E S^*_{k_j}}{k_j}
        \xrightarrow[j\to\infty]{\text{लगभग निश्चित}} 0,
        \qquad\text{अतः}\qquad
        \frac{S^*_{k_j}}{k_j} \to m \ \text{लगभग निश्चित}
        \]
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- सैंडविच और निष्कर्ष।}
\begin{enumerate}[resume]
  \item $k_j \leq n \leq k_{j+1}$ के लिए $S^*_n$ की एकदिष्टता (योज्य पद ऋणेतर हैं!) का
        उपयोग करके दर्शाइए कि
        \[
        \frac{k_j}{k_{j+1}}\,\frac{S^*_{k_j}}{k_j}
        \;\leq\; \frac{S^*_n}{n} \;\leq\;
        \frac{k_{j+1}}{k_j}\,\frac{S^*_{k_{j+1}}}{k_{j+1}},
        \]
        और लगभग निश्चित रूप से निकालिए:
        \[
        \frac m\alpha \leq \liminf\frac{S^*_n}n \leq
        \limsup\frac{S^*_n}n \leq \alpha\,m .
        \]
  \item किसी अनुक्रम के अनुदिश $\alpha \downarrow 1$ लीजिए और निष्कर्ष निकालिए कि
        लगभग निश्चित रूप से $\frac{S_n^*}n \to m$, अतः (भाग I) \emph{बृहत्
        संख्याओं का प्रबल नियम}:
        \[
        \boxed{\ \frac{S_n}{n}
        \xrightarrow[n\to\infty]{\text{लगभग निश्चित}} \E[X_1].\ }
        \]
  \item ठीक-ठीक कहाँ युग्मशः \omterm{def:b3:probability:independence}{स्वातंत्र्य} (पूर्ण स्वातंत्र्य के
        बजाय) पर्याप्त था? उन तीनों स्थानों की सूची बनाइए जहाँ
        स्वातंत्र्य-प्रकार की परिकल्पनाएँ लगाई गईं।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- लाभांश।}
\begin{enumerate}[resume]
  \item (बोरेल की सामान्य संख्याएँ) दर्शाइए कि $\lambda$-लगभग प्रत्येक
        $x \in \intcc01$ \emph{प्रत्येक आधार $b \geq 2$ में सामान्य} है: प्रत्येक
        अंक $0, \dots, b-1$ अनंतस्पर्शी आवृत्ति $\frac1b$ के साथ प्रकट होता है
        \emph{($b$ और कोई एक अंक स्थिर कीजिए, सूचक चरों पर प्रबल
        नियम लगाइए --- यह उचित ठहराइए कि किसी एकसमान चर के
        आधार-$b$ अंक \cref{thm:b3:probability:existence} की भाँति $\{0,\dots,b-1\}$ पर \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित
        एकसमान हैं --- फिर गणनीय संख्या की प्रायिकता-एक घटनाओं का
        सर्वनिष्ठ लीजिए)}। कोई एक स्पष्ट असामान्य संख्या प्रस्तुत
        कीजिए, और विचार कीजिए: यह प्रमेय लगभग सभी संख्याओं की
        सामान्यता का दावा करती है, फिर भी $\sqrt2$ या $\pi$ की
        सामान्यता सिद्ध करना आज भी खुला प्रश्न है।
  \item (\omterm{ex:b3:probability:sllnapps}{मोंते कार्लो}, प्रत्याभूत) $g \in L^1(\intcc01^d)$ के लिए \cref{ex:b3:probability:sllnapps}(ख) की विधि
        को पूरी तरह उचित ठहराइए: \cref{thm:b3:probability:existence} और \cref{exo:b3:probability:10} से $\intcc01^d$ पर
        \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित एकसमान प्रतिदर्श की रचना कीजिए, और
        बताइए कि प्रबल नियम क्या दे देता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- पूर्ण स्वातंत्र्य से क्या मिलता है:
उच्चिष्ठ असमिकाएँ और यादृच्छिक श्रेणियाँ।} एतेमादी केवल युग्मशः
\omterm{def:b3:probability:independence}{स्वातंत्र्य} खर्च करते हैं; शेष भाग \omterm{def:b3:complete:complete}{पूर्ण} (पारस्परिक) रूप का दोहन
करते हैं। मान लीजिए $(Z_n)$ $L^2$ वाले \omterm{def:b3:probability:independence}{स्वतंत्र} केंद्रित चर हैं और
$S_k = Z_1 + \dots + Z_k$ (यह एक नया संकेतन है, जिसका ऊपर वाले $X_n$ से कोई संबंध
नहीं)।
\begin{enumerate}[resume]
  \item (कोल्मोगोरोव की उच्चिष्ठ असमिका) $\varepsilon
        > 0$ के लिए सिद्ध कीजिए
        \[
        \P\Bigl(\max_{1\leq k\leq n}\abs{S_k} \geq
        \varepsilon\Bigr) \;\leq\;
        \frac1{\varepsilon^2}\sum_{k=1}^n\V(Z_k) :
        \]
        अर्थात् चेबिशेव की क़ीमत में उच्चिष्ठ मिल जाता है
        \emph{(घटना को उस पहले सूचकांक $k$ के अनुसार विभाजित
        कीजिए जिसके लिए $\abs{S_k} \geq \varepsilon$ हो; उस टुकड़े पर $S_n^2 \geq S_k^2 + 2S_k(S_n - S_k)$ लिखिए और
        गुटों $(Z_1, \dots, Z_k)$ तथा $(Z_{k+1}, \dots, Z_n)$ के \omterm{def:b3:probability:independence}{स्वातंत्र्य} का उपयोग कीजिए,
        \cref{thm:b3:probability:independence})}। उस चरण की ओर संकेत कीजिए जहाँ युग्मशः \omterm{def:b3:probability:independence}{स्वातंत्र्य}
        अब पर्याप्त नहीं रह जाता।
  \item (खिनचिन--कोल्मोगोरोव एक-श्रेणी प्रमेय) निकालिए: यदि $\sum_n\V(Z_n) < \infty$
        हो, तो $\sum_nZ_n$ लगभग निश्चित रूप से अभिसरित होती है
        \emph{(दर्शाइए कि लगभग निश्चित रूप से आंशिक योग कोई कोशी
        अनुक्रम बनाते हैं: $Z_{N+1}, \dots, Z_{N+m}$ पर लगाई गई उच्चिष्ठ असमिका में
        $m \to
        \infty$ लीजिए, फिर $N \to
        \infty$ लीजिए)}।
  \item (राडेमाखर श्रेणी) मान लीजिए $(\varepsilon_n)$ \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित चिह्न
        हैं, $\P(\varepsilon_n = \pm1) =
        \frac12$ (\cref{thm:b3:probability:existence}), और मान लीजिए $(x_n)$ वास्तविक संख्याएँ
        हैं। दर्शाइए कि $\sum_nx_n^2 < \infty$ होते ही $\sum_nx_n\varepsilon_n$ लगभग निश्चित रूप से
        अभिसरित हो जाती है; और यह भी दर्शाइए कि $(x_n)$ कुछ भी हो,
        $\sum_nx_n\varepsilon_n$ के अभिसरित होने की प्रायिकता $0$ या $1$ ही है
        (\cref{thm:b3:probability:zeroone})।
  \item विलोम, प्रारंभिक साधनों से। $T_n = \sum_{k\leq
        n}x_k\varepsilon_k$ और $s_n^2 = \sum_{k\leq
        n}x_k^2$ रखिए, और मान
        लीजिए $s_n \to \infty$।
        (क) \emph{पेली--ज़िगमुंड असमिका} सिद्ध कीजिए: $\E Z^2 < \infty$ और
        $0 < \theta <
        1$ वाले $Z \geq 0$ के लिए
        \[
        \P\bigl(Z > \theta\,\E Z\bigr) \;\geq\; (1 -
        \theta)^2\,\frac{(\E Z)^2}{\E Z^2}
        \]
        \emph{($\E Z$ को स्तर $\theta\E Z$ पर विभाजित कीजिए और ऊपरी टुकड़े
        पर कोशी--श्वार्ज़ लगाइए)}।
        (ख) दर्शाइए कि $\E T_n^4 \leq 3s_n^4$।
        (ग) $\P\bigl(\abs{T_n} > \frac{s_n}2\bigr)
        \geq \frac3{16}$ निकालिए और निष्कर्ष निकालिए कि $\sum_nx_n\varepsilon_n$ लगभग निश्चित
        रूप से अपसारित होती है; अतः द्विभाजन
        \[
        \sum_nx_n\varepsilon_n\ \text{लगभग निश्चित रूप से अभिसरित}
        \iff \sum_nx_n^2 < \infty .
        \]
  \item (यादृच्छिक प्रसंवादी श्रेणी) निष्कर्ष निकालिए कि $\sum_n\frac{\varepsilon_n}{n^s}$
        लगभग निश्चित रूप से अभिसरित होती है तभी और केवल तभी जब
        $s > \frac12$ हो। $\frac12 < s \leq
        1$ के लिए यह श्रेणी लगभग निश्चित रूप से
        अभिसरित होती है जबकि $\sum_nn^{-s} = \infty$: यादृच्छिक चिह्न वर्गमूल-सामर्थ्य
        वाला निरसन उत्पन्न करते हैं --- इसकी तुलना एकांतर श्रेणी
        $\sum_n\frac{(-1)^n}{n^s}$ से कीजिए, जो \emph{प्रत्येक} $s > 0$ के लिए अभिसरित
        होती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग VII --- संकेंद्रण: हॉफडिंग असमिका।} प्रबल
नियम कहता है कि $\frac{S_n}n \to m$; और संकेंद्रण असमिकाएँ बताती हैं कि
\emph{प्रत्येक स्थिर $n$ पर} कोई विचलन कितना असंभाव्य है।
\begin{enumerate}[resume]
  \item (हॉफडिंग प्रमेयिका)
        (क) दोनों श्रेणियों की पदशः तुलना करके दर्शाइए कि सभी
        $\lambda \in \R$ के लिए $\cosh\lambda \leq \eu^{\lambda^2/2}$।
        (ख) मान लीजिए $Z$ $a \leq Z \leq b$, $a < b$ वाला केंद्रित चर है।
        दर्शाइए कि
        \[
        \E\,\eu^{\lambda Z} \leq
        \exp\Bigl(\frac{\lambda^2(b - a)^2}8\Bigr)
        \]
        \emph{($\intcc ab$ पर $\eu^{\lambda z}$ को उसकी जीवा से परिबद्ध कीजिए,
        \omterm{def:b3:probability:space}{प्रत्याशा} लीजिए, और $p =
        \frac{-a}{b-a}$ तथा $t = \lambda(b - a)$ के साथ $\varphi(t)
        = -pt + \log(1 - p + p\eu^t)$ का अध्ययन
        कीजिए: दर्शाइए कि $\varphi(0) = \varphi'(0) = 0$ और $\varphi''
        \leq \frac14$)}।
  \item (हॉफडिंग असमिका) मान लीजिए $X_1, \dots, X_n$ $a_i \leq X_i \leq b_i$ और $S_n =
        X_1 + \dots + X_n$ वाले
        \omterm{def:b3:probability:independence}{स्वतंत्र} चर हैं। $t > 0$ के लिए सिद्ध कीजिए
        \[
        \P\bigl(S_n - \E S_n \geq t\bigr) \leq
        \exp\Bigl(\frac{-2t^2}{\sum_{i=1}^n(b_i -
        a_i)^2}\Bigr),
        \]
        और अधःपुच्छ के लिए भी वही परिबंध \emph{(चरघातांकी चेबिशेव:
        \omterm{def:b3:probability:independence}{स्वातंत्र्य} और प्रश्न 17 का उपयोग करके $\E\,\eu^{\lambda(S_n - \E S_n)}$ को परिबद्ध
        कीजिए, फिर $\lambda > 0$ पर अनुकूलन कीजिए)}।
  \item (प्रबल नियम, परिबद्ध स्थिति, दर के साथ) मान लीजिए $X_i$
        $\intcc ab$ में मान लेने वाले \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित चर हैं और $m
        = \E X_1$।
        दर्शाइए कि
        \[
        \P\Bigl(\Bigl|\frac{S_n}n - m\Bigr| \geq
        \varepsilon\Bigr) \leq
        2\exp\Bigl(\frac{-2n\varepsilon^2}{(b - a)^2}\Bigr)
        \]
        और बोरेल--कांतेली से लगभग निश्चित रूप से $\frac{S_n}n \to m$ पुनः प्राप्त
        कीजिए: यह परिबद्ध चरों के लिए प्रबल नियम की एक दूसरी
        उपपत्ति है --- कोई कर्तन नहीं, और प्रत्येक परिमित $n$ पर
        एक चरघातांकी दर, पर बदले में परिबद्ध योज्य पद और पूर्ण
        स्वातंत्र्य। एतेमादी की परिकल्पनाओं से इनकी तुलना कीजिए।
  \item (\omterm{ex:b3:probability:sllnapps}{मोंते कार्लो}, स्थिर $n$ पर प्रत्याभूत) मान लीजिए $g
        \colon \intcc01^d \to \intcc01$
        \omterm{def:b3:lebesgue:measurable}{मापनीय} है और $(U_k)$ प्रश्न 11 वाला \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित एकसमान
        प्रतिदर्श। $\varepsilon, \delta > 0$ दिया हो, तो दर्शाइए कि
        \[
        n \geq \frac{\log(2/\delta)}{2\varepsilon^2}
        \implies
        \P\Bigl(\Bigl|\frac1n\sum_{k=1}^ng(U_k) -
        \int g\,\dd\lambda_d\Bigr| \geq \varepsilon\Bigr)
        \leq \delta,
        \]
        और $\varepsilon =
        \delta = 10^{-2}$ के लिए देहली का मूल्यांकन कीजिए। यह परिबंध $d$
        को सम्मिलित नहीं करता: इसकी तुलना प्रश्न 11 से और नियतांकी
        जालकों से कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग VIII --- यादृच्छिक भ्रमण कितना बड़ा होता है?
पुनरावृत्त लघुगणक की ओर।} मान लीजिए $S_n = \varepsilon_1 +
\dots + \varepsilon_n$ \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित
निष्पक्ष चिह्नों से बना सरल यादृच्छिक भ्रमण है।
\begin{enumerate}[resume]
  \item (उप-गाउसीय पुच्छ) दर्शाइए कि $\E\,\eu^{\lambda S_n} =
        (\cosh\lambda)^n \leq \eu^{n\lambda^2/2}$ और $x > 0$ के लिए निकालिए
        \[
        \P(S_n \geq x) \leq \eu^{-x^2/(2n)},
        \qquad
        \P(\abs{S_n} \geq x) \leq 2\,\eu^{-x^2/(2n)} .
        \]
  \item बोरेल--कांतेली के माध्यम से निकालिए
        \[
        \limsup_{n\to\infty}\frac{\abs{S_n}}
        {\sqrt{2n\log n}} \leq 1 \quad\text{लगभग निश्चित}
        \]
        \emph{($\eta > 0$ के लिए $x =
        (1 + \eta)\sqrt{2n\log n}$ पर पुच्छ परिबंधों का योग कीजिए,
        फिर $\eta = \frac1p$ पर सर्वनिष्ठ लीजिए)}। विशेष रूप से यह भ्रमण
        किसी लघुगणकीय गुणक तक केंद्रीय सीमा मापक्रम $\sqrt n$ पर जीता
        है --- अर्थात् कच्चे परिबंध $\abs{S_n} \leq n$ से बहुत नीचे।
  \item द्विगुणन उपानुक्रम $n_j = 2^j$ के अनुदिश दर्शाइए
        \[
        \limsup_{j\to\infty}\frac{S_{n_j}}
        {\sqrt{2n_j\log\log n_j}} \leq 1 \quad\text{लगभग निश्चित},
        \]
        और विचार कीजिए: \emph{पुनरावृत्त लघुगणक का नियम} (खिनचिन;
        और व्यापक केंद्रित $L^2$ योज्य पदों के लिए
        हार्टमन--विंटनर) कहता है कि
        \[
        \limsup_{n\to\infty}\frac{S_n}
        {\sqrt{2n\log\log n}} = 1 \quad\text{लगभग निश्चित}
        \]
        ठीक-ठीक समझाइए कि अभी सिद्ध किए गए उपानुक्रम आकलन को इस
        कथन के ऊपरी अर्ध भाग से क्या पृथक करता है (प्रत्येक खंड के
        भीतर $\max_{n_j \leq n \leq
        n_{j+1}}S_n$ को नियंत्रित करना पड़ता है, जिसके लिए
        \emph{चरघातांकी} मापक्रम पर कोई उच्चिष्ठ असमिका चाहिए) और
        मात्रात्मक रूप से जाँचिए कि प्रश्न 12 की असमिका उस प्रयोजन
        के लिए बहुत दुर्बल है। निचला अर्ध भाग \omterm{def:b3:probability:independence}{स्वतंत्र} खंडों पर
        लगाई गई दूसरी बोरेल--कांतेली प्रमेयिका पर टिका है; और दोनों
        अर्ध भाग किसी समर्पित प्रायिकता पाठ्यक्रम के लिए ईमानदार
        तृतीय वर्ष की सामग्री हैं।

  \item (किसी परिमित वर्ग पर एकसमान विचलन) मान लीजिए $A_1,
        \dots, A_N$ किसी
        पुनरावर्तनीय प्रयोग की घटनाएँ हैं, और प्रत्येक प्रायिकता
        का आकलन $n$ \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित पुनरावृत्तियों पर उसकी
        आनुभविक आवृत्ति $\hat p_i$ से किया जाता है। हॉफडिंग असमिका को
        संघ परिबंध के साथ मिलाकर दर्शाइए
        \[
        \P\Bigl(\max_{i\leq N}\,\abs{\hat p_i - \P(A_i)} >
        \varepsilon\Bigr) \;\leq\; 2N\,\eu^{-2n\varepsilon^2},
        \]
        और प्रतिदर्श-आकार नियम निकालिए: $n \geq
        \frac{\ln(2N/\delta)}{2\varepsilon^2}$ इस बात की प्रत्याभूति
        देता है कि सभी $N$ आकलन एक साथ $\geq 1 -
        \delta$ प्रायिकता के साथ
        $\varepsilon$-यथार्थ हैं। $N = 10^6$, $\varepsilon =
        0.01$, $\delta = 0.05$ के लिए $n$ परिकलित
        कीजिए: अर्थात् एकसमानता की लघुगणकीय क़ीमत।
  \item (यादृच्छिक प्रसंवादी खिड़की) यादृच्छिक श्रेणी सिद्धांत के
        दोनों अर्ध भागों को मिलाकर दर्शाइए कि \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित
        चिह्नों $(\varepsilon_n)$ के लिए श्रेणी $\sum_n\frac{\varepsilon_n}{n^\alpha}$ लगभग निश्चित रूप से
        अभिसरित होती है यदि $\alpha > \frac12$ हो और लगभग निश्चित रूप से अपसारित
        होती है यदि $\alpha \leq \frac12$ हो; इसकी तुलना निरपेक्ष अभिसरण से कीजिए
        (जिसके लिए $\alpha > 1$ चाहिए): खिड़की $\alpha \in \intoc{\frac12}1$ पर अभिसरण वस्तुतः
        एक प्रायिकता-आधारित परिघटना है --- निरसन, आकार नहीं।
\end{enumerate}
\end{problem}
