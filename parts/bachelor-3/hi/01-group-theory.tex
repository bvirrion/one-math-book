\chapter{समूह सिद्धांत}\label{ch:b3:groups}

दूसरे वर्ष के खंड में समूह लेखा-जोखा के उपकरण के रूप में मिले थे:
लाग्राँज की प्रमेय, चक्रीय समूह, सममित समूह और उसका चिह्न। यह
अध्याय समूह सिद्धांत को एक \emph{विधि} में बदल देता है। इसकी चालक
शक्ति यह धारणा है कि एक समूह किसी समुच्चय पर \emph{\omterm{def:b3:groups:action}{क्रिया}} करता
है: कक्षाओं और स्थिर बिंदुओं की गणना से \omterm{cor:b3:groups:classeq}{वर्ग समीकरण}, कोशी की
प्रमेय और तीनों सिलो प्रमेय निकलते हैं --- परिमित समूह सिद्धांत
का मूल स्थानीय-से-वैश्विक सिद्धांत। इसके बाद हम समूहों को जोड़ना
(प्रत्यक्ष और \omterm{def:b3:groups:semidirect}{अर्ध-प्रत्यक्ष गुणन}) और उन्हें तोड़ना (\omterm{thm:b3:groups:jordanholder}{संघटन
श्रेणी}, \omterm{def:b3:groups:derived}{विलेय समूह}) सीखते हैं, और वह प्रमेय सिद्ध करते हैं जो
\cref{ch:b3:galois} में बहुपद समीकरणों के बारे में तीन सदी पुराने प्रश्न को बंद
कर देगी: \omterm{thm:b3:groups:ansimple}{एकांतर समूह} $A_n$ $n
\geq 5$ के लिए \omterm{def:b3:groups:simple}{सरल} है।

\section{भागफल समूह और तुल्याकारिता प्रमेय}

आगे सर्वत्र $G$ गुणनात्मक रूप में लिखा गया एक समूह है और $e$
उसका तत्समक। दूसरे वर्ष के खंड से स्मरण कीजिए: उपसमूह,
सहसमुच्चय $gH$, लाग्राँज की प्रमेय (परिमित $G$ के लिए $\abs G = [G:H]\,\abs H$),
किसी अवयव का क्रम, चक्रीय समूह, तथा सममित समूह $S_n$ और उसकी
चिह्न समाकारिता $\varepsilon \colon S_n \to \{\pm1\}$।

\begin{definition}\label{def:b3:groups:normal}
$G$ का उपसमूह $N$ \emph{प्रसामान्य}\index{प्रसामान्य उपसमूह}
कहलाता है (लिखा जाता है $N \trianglelefteq G$) जब प्रत्येक $g
\in G$ के लिए $gNg^{-1} = N$ हो
--- समतुल्य रूप से, जब बाएँ और दाएँ सहसमुच्चय एक ही हों:
सभी $g$ के लिए $gN =
Ng$।
\end{definition}

\begin{theorem}[भागफल समूह]\label{thm:b3:groups:quotient}
मान लीजिए $N \trianglelefteq G$। सहसमुच्चयों का समुच्चय $G/N$, गुणन $(gN)(hN) = ghN$ के साथ,
एक सुपरिभाषित समूह है --- \emph{भागफल समूह}\index{भागफल समूह}
--- और \emph{विहित प्रक्षेप} $\pi \colon G \to G/N$, $g \mapsto gN$, एक आच्छादक समाकारिता
है जिसकी अष्टि $N$ है। विलोमतः, किसी भी समूह समाकारिता की
अष्टि \omterm{def:b3:groups:normal}{प्रसामान्य} होती है: \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} ठीक अष्टियाँ ही हैं।
\end{theorem}

\begin{proof}
सारा मर्म सुपरिभाषितता में है। यदि $gN = g'N$ और $hN = h'N$ हों, तो
$n, m \in N$ के साथ $g' = gn$, $h' = hm$ लिखिए। तब $g'h' = gnhm =
gh\,(h^{-1}nh)\,m \in ghN$, क्योंकि प्रसामान्यता से
$h^{-1}nh \in N$: सहसमुच्चयों का गुणनफल प्रतिनिधियों पर निर्भर नहीं करता।
साहचर्य, तत्समक $eN = N$ और प्रतिलोम $(gN)^{-1} = g^{-1}N$ $G$ से विरासत में मिलते
हैं। स्पष्टतः $\pi$ एक आच्छादक समाकारिता है और $\pi(g) = N \iff g \in N$।

यदि $f \colon G \to H$ एक समाकारिता हो और $k \in \ker f$, तो $f(gkg^{-1}) = f(g)f(k)f(g)^{-1} = e$: अष्टियाँ \omterm{def:b3:groups:normal}{प्रसामान्य}
होती हैं।
\end{proof}

\begin{theorem}[सार्वत्रिक गुण; प्रथम तुल्याकारिता प्रमेय]
\label{thm:b3:groups:firstiso}
मान लीजिए $f \colon G \to H$ एक समाकारिता है और $N \trianglelefteq G$ ऐसा है कि $N \subseteq \ker f$। तब
$f = \bar f \circ \pi$ को संतुष्ट करने वाली एक अद्वितीय समाकारिता $\bar f \colon G/N
\to H$ विद्यमान
है। विशेष रूप से, $N = \ker
f$ लेने पर:\index{तुल्याकारिता प्रमेय}
\[
G/\ker f \;\xrightarrow{\;\sim\;}\; \operatorname{im} f,
\qquad gN \mapsto f(g).
\]
\end{theorem}

\begin{proof}
अद्वितीयता: $\bar f(gN)$ को $f(g)$ होना ही चाहिए। अस्तित्व: यदि $gN = g'N$ तो
$g^{-1}g' \in N \subseteq \ker f$, अतः $f(g) = f(g')$ और $\bar
f(gN) = f(g)$ सुपरिभाषित है; यह समाकारिता है क्योंकि
$f$ समाकारिता है। $N = \ker f$ के लिए: $\bar f$ एकैकी है, क्योंकि $\bar f(gN) = e$ का
अर्थ है $g \in \ker f$, अर्थात् $gN = N$; और इसका प्रतिबिंब वही है जो $f$ का
है।
\end{proof}

\begin{theorem}[द्वितीय और तृतीय तुल्याकारिता प्रमेय]
\label{thm:b3:groups:secondiso}
मान लीजिए $H \leq G$ और $N \trianglelefteq G$।
\begin{enumerate}
  \item $HN = \{hn : h \in H,\, n \in N\}$ एक उपसमूह है, $N
        \trianglelefteq HN$, $H \cap N \trianglelefteq H$, तथा
        \[
        H/(H \cap N) \;\cong\; HN/N .
        \]
  \item यदि इसके अतिरिक्त $N \subseteq K \trianglelefteq G$ हो, तो $K/N
        \trianglelefteq G/N$ और $(G/N)\big/(K/N) \cong G/K$।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) $HN$ एक उपसमूह है: $N$ की प्रसामान्यता का उपयोग करते हुए
$(hn)(h'n') = hh'\,(h'^{-1}nh')n' \in HN$ तथा $(hn)^{-1} = h^{-1}(hn^{-1}h^{-1}) \in HN$। $H \hookrightarrow HN \xrightarrow{\pi} HN/N$ का संयोजन कीजिए: यह समाकारिता आच्छादक है
($hnN = hN$) और इसकी अष्टि $\{h \in H : h \in
N\} = H \cap N$ है; अब \cref{thm:b3:groups:firstiso} लगाइए।

(2) प्रक्षेप $G/N \to G/K$, $gN \mapsto gK$, सुपरिभाषित है ($N \subseteq K$), आच्छादक है, और
उसकी अष्टि $K/N$ है; पुनः \cref{thm:b3:groups:firstiso} लगाइए।
\end{proof}

\begin{theorem}[संगति प्रमेय]\label{thm:b3:groups:correspondence}
मान लीजिए $N \trianglelefteq G$। प्रतिचित्रण $H \mapsto H/N$ $N$ को अंतर्विष्ट करने वाले
$G$ के उपसमूहों और $G/N$ के उपसमूहों के बीच एक एकैकी आच्छादक
संगति है, जो (दोनों दिशाओं में) अंतर्वेशन, सूचकांक और
प्रसामान्यता को सुरक्षित रखती है।
\end{theorem}

\begin{proof}
इसका प्रतिलोम $\bar H \mapsto \pi^{-1}(\bar H)$ है। दोनों प्रतिचित्रण उपसमूहों को उपसमूहों पर
भेजते हैं और परस्पर प्रतिलोम हैं: $\pi^{-1}(H/N) =
HN = H$, क्योंकि $N \subseteq H$, तथा $\pi$
की आच्छादकता से $\pi(\pi^{-1}(\bar H)) = \bar H$। अंतर्वेशन स्पष्टतः सुरक्षित रहते हैं;
$[G:H] = [G/N : H/N]$, क्योंकि $gH \mapsto (gN)(H/N)$ सहसमुच्चय समष्टियों के बीच एक सुपरिभाषित
एकैकी आच्छादक संगति है; और सभी $g$ के लिए $gHg^{-1} = H$ तभी और केवल तभी
जब सभी $gN$ के लिए $(gN)(H/N)(gN)^{-1} = H/N$ --- फिर से $\pi$ की आच्छादकता से।
\end{proof}

\begin{example}\label{ex:b3:groups:firstiso}
$\varepsilon \colon S_n \to \{\pm 1\}$ से $S_n/A_n \cong
\{\pm1\}$ मिलता है; $\det \colon GL_n(K) \to K^\times$ से $GL_n(K)/SL_n(K) \cong K^\times$; और $t \mapsto \eu^{2\iu\pi t}$ से $\R/\Z \cong \mathbb U$,
अर्थात् वृत्त समूह। व्यवहार में भागफल इसी प्रकार \emph{परिकलित}
किए जाते हैं: उपयुक्त अष्टि वाली एक आच्छादक समाकारिता ढूँढ़िए
--- यही प्रथम तुल्याकारिता प्रमेय का उपयोग है।
\end{example}

\begin{method}\label{met:b3:groups:normal}
$N \trianglelefteq G$ सिद्ध करने के लिए, सुघड़ता के घटते क्रम में: $G$ पर
परिभाषित किसी समाकारिता की अष्टि के रूप में $N$ को प्रस्तुत
कीजिए; सभी $g$ के लिए $gNg^{-1} \subseteq N$ की जाँच कीजिए (यह पर्याप्त है:
इसे $g^{-1}$ पर लगाकर और संयुग्मन करके विपरीत अंतर्वेशन मिल जाता
है); सत्यापित कीजिए कि $N$ संयुग्मन वर्गों का सम्मिलन है; अथवा
यह देखिए कि $[G:N] = 2$ (तब $gN = Ng$ अनिवार्य है --- \cref{exo:b3:groups:1})।
\end{method}

\section{समूह क्रियाएँ}

\begin{definition}\label{def:b3:groups:action}
किसी समुच्चय $X$ पर $G$ की एक \emph{क्रिया}\index{समूह
क्रिया} $X$ के एकैकी आच्छादक प्रतिचित्रणों के समूह में जाने
वाली एक समाकारिता $\varphi \colon G \to \mathfrak{S}(X)$ है; $\varphi(g)(x)$ के लिए $g \cdot x$ लिखा जाता है।
समतुल्य रूप से: एक ऐसा प्रतिचित्रण $G \times X \to X$ जिसके लिए $e \cdot x = x$ और
$g
\cdot (h \cdot x) = (gh) \cdot x$। $x$ की \emph{कक्षा}\index{कक्षा} $\mathcal O_x = \{g \cdot x : g \in G\}$ है, उसका
\emph{स्थायीकारी}\index{स्थायीकारी} उपसमूह $G_x = \{g :
g\cdot x = x\}$ है, और $X^G = \{x : \forall g,\ g \cdot x = x\}$
\emph{स्थिर बिंदुओं} का समुच्चय है। क्रिया \emph{संक्रमणीय}
कहलाती है यदि ठीक एक कक्षा हो, \emph{निष्ठ} यदि $\varphi$ एकैकी हो,
और \emph{मुक्त} यदि सभी स्थायीकारी तुच्छ हों।
\end{definition}

\begin{example}\label{ex:b3:groups:actions}
पाँच क्रियाएँ समूचे परिमित समूह सिद्धांत को चलाती हैं:
\begin{enumerate}
  \item \emph{बायाँ स्थानांतरण} $g \cdot x = gx$ द्वारा $G$ की अपने ही
        ऊपर \omterm{def:b3:groups:action}{क्रिया}: मुक्त और संक्रमणीय।
  \item \emph{संयुग्मन} $g \cdot x = gxg^{-1}$ द्वारा $G$ की अपने ही ऊपर
        \omterm{def:b3:groups:action}{क्रिया}: कक्षाएँ \emph{संयुग्मन वर्ग}\index{संयुग्मन
        वर्ग} हैं, \omterm{def:b3:groups:action}{स्थायीकारी} \emph{केंद्रक}
        \index{केंद्रक} $Z_G(x) = \{g : gx = xg\}$ हैं, और स्थिर
        बिंदु \emph{केंद्र}\index{केंद्र} $Z(G)$ हैं।
  \item सहसमुच्चय समष्टि $G/H$ पर $g \cdot xH = gxH$ द्वारा $G$ की \omterm{def:b3:groups:action}{क्रिया}:
        संक्रमणीय, जिसमें सहसमुच्चय $H$ का \omterm{def:b3:groups:action}{स्थायीकारी} $H$
        है। प्रत्येक संक्रमणीय \omterm{def:b3:groups:action}{क्रिया} इसी रूप की होती है
        (\cref{exo:b3:groups:8})।
  \item संयुग्मन द्वारा $G$ की अपने उपसमूहों के समुच्चय पर
        \omterm{def:b3:groups:action}{क्रिया}: $H$ का \omterm{def:b3:groups:action}{स्थायीकारी} \emph{प्रसामान्यक}
        \index{प्रसामान्यक} $N_G(H) =
        \{g : gHg^{-1} = H\}$ है, अर्थात् $G$ का सबसे बड़ा
        उपसमूह जिसमें $H$ \omterm{def:b3:groups:normal}{प्रसामान्य} है।
  \item $\intint{1}{n}$ पर $S_n$ की \omterm{def:b3:groups:action}{क्रिया}: सब उदाहरणों की जननी।
\end{enumerate}
\end{example}

\begin{theorem}[कक्षा--स्थायीकारी]\label{thm:b3:groups:orbitstab}
प्रतिचित्रण $gG_x \mapsto g \cdot x$ एक सुपरिभाषित एकैकी आच्छादक संगति $G/G_x \to \mathcal O_x$ है।
विशेष रूप से, परिमित $G$ के
लिए,\index{कक्षा--स्थायीकारी प्रमेय}
\[
\abs{\mathcal O_x} = [G : G_x] \quad\text{जो } \abs G \text{ का भाजक है,}
\]
और, चूँकि कक्षाएँ $X$ का विभाजन करती हैं (वे तुल्यता $x \sim y \iff y \in \mathcal O_x$ के
वर्ग हैं),
\[
\abs X = \sum_{i} \,[G : G_{x_i}]
\qquad (x_i\colon \text{प्रति कक्षा एक बिंदु}).
\]
\end{theorem}

\begin{proof}
सुपरिभाषित और एकैकी: $gG_x = hG_x \iff h^{-1}g \in G_x \iff
h^{-1}g \cdot x = x \iff g \cdot x = h \cdot x$; इस शृंखला को दोनों दिशाओं में
पढ़िए। आच्छादकता \omterm{def:b3:groups:action}{कक्षा} की परिभाषा ही है। गणना संबंधी कथन
लाग्राँज की प्रमेय और $X$ के कक्षाओं में विभाजन से निकलते हैं।
\end{proof}

\begin{corollary}[वर्ग समीकरण]\label{cor:b3:groups:classeq}
किसी परिमित समूह $G$ के लिए, एक से अधिक अवयव वाले प्रत्येक
\omterm{ex:b3:groups:actions}{संयुग्मन वर्ग} में से एक प्रतिनिधि $x_i$ चुनने
पर:\index{वर्ग समीकरण}
\[
\abs G = \abs{Z(G)} + \sum_i \,[G : Z_G(x_i)],
\qquad\text{प्रत्येक } [G : Z_G(x_i)] > 1 \text{, जो } \abs G \text{ का भाजक है}.
\]
\end{corollary}

\begin{proof}
संयुग्मन \omterm{def:b3:groups:action}{क्रिया} पर \cref{thm:b3:groups:orbitstab} लगाइए: एकल-अवयवी कक्षाएँ ठीक $Z(G)$ के
अवयव हैं।
\end{proof}

\begin{theorem}[$p$-समूहों के स्थिर बिंदु]\label{thm:b3:groups:pfixed}
मान लीजिए $p$ अभाज्य है। \emph{$p$-समूह}\index{पी-समूह@$p$-समूह}
वह परिमित समूह है जिसका क्रम $p$ की कोई घात हो। यदि कोई
$p$-समूह $G$ किसी परिमित समुच्चय $X$ पर \omterm{def:b3:groups:action}{क्रिया} करे, तो
\[
\abs{X^G} \equiv \abs X \pmod p .
\]
परिणाम: अतुच्छ $p$-समूह का \omterm{ex:b3:groups:actions}{केंद्र} अतुच्छ होता है, और क्रम
$p^2$ का प्रत्येक समूह आबेली होता है।
\end{theorem}

\begin{proof}
प्रत्येक \omterm{def:b3:groups:action}{कक्षा} की गणनीयता $[G:G_x]$ है, जो $p$ की एक घात है; यह
घात ठीक स्थिर बिंदुओं पर $1$ होती है और अन्यथा $p$ से
विभाज्य। कक्षाओं पर योग लेने से सर्वांगसमता मिल जाती है।
\omterm{ex:b3:groups:actions}{केंद्र} के लिए: $G$ की अपने ऊपर संयुग्मन \omterm{def:b3:groups:action}{क्रिया} में $X^G = Z(G)$ है,
अतः $\abs{Z(G)} \equiv
\abs G \equiv 0 \pmod p$, और $Z(G) \ni e$ से $\abs{Z(G)} \geq
p$ अनिवार्य हो जाता है। क्रम $p^2$:
यदि $Z(G) \neq G$ हो तो $\abs{Z(G)} = p$ और $G/Z(G)$ क्रम $p$ का चक्रीय समूह है,
जिससे $G$ का आबेली होना अनिवार्य हो जाता है (\cref{exo:b3:groups:2}) ---
विरोधाभास।
\end{proof}

\begin{theorem}[कोशी]\label{thm:b3:groups:cauchy}
यदि कोई अभाज्य $p$ $\abs G$ को विभाजित करे, तो $G$ में क्रम
$p$ का कोई अवयव विद्यमान है।\index{कोशी की प्रमेय}
\end{theorem}

\begin{proof}[उपपत्ति (मैके)]
मान लीजिए $X = \{(g_1, \dots, g_p) \in G^p : g_1 g_2 \cdots g_p = e\}$। $g_1, \dots, g_{p-1}$ को स्वतंत्र रूप से चुनने पर $g_p$ निर्धारित
हो जाता है: $\abs X =
\abs G^{p-1}$, जो $p$ से विभाज्य है। चक्रीय समूह $\Z/p\Z$
चक्रीय विस्थापन $(g_1, \dots, g_p) \mapsto (g_2, \dots, g_p,
g_1)$ द्वारा $X$ पर \omterm{def:b3:groups:action}{क्रिया} करता है --- यह
$X$ को सुरक्षित रखता है, क्योंकि $g_2 \cdots g_p g_1 =
g_1^{-1}(g_1 \cdots g_p)g_1 = e$। \cref{thm:b3:groups:pfixed} से, $\abs{X^{\Z/p\Z}} \equiv \abs X \equiv 0 \pmod p$।
स्थिर बिंदु वे अचर बहुक $(g, \dots, g)$ हैं जिनमें $g^p = e$; बहुक $(e,
\dots, e)$
उनमें से एक है, अतः वे कम से कम $p$ हैं, और इसलिए $g^p = e$ वाला
कम से कम एक $g \neq e$ है: उसका क्रम ठीक $p$ है।
\end{proof}

\begin{theorem}[केली]\label{thm:b3:groups:cayley}
क्रम $n$ का प्रत्येक समूह $S_n$ में अंतःस्थापित हो जाता
है।\index{केली की प्रमेय}
\end{theorem}

\begin{proof}
बायाँ स्थानांतरण $\varphi \colon G \to \mathfrak S(G) \cong S_n$ एक समाकारिता है; $\varphi(g) = \mathrm{id}$ से $g = ge = e$ अनिवार्य
हो जाता है: यह निष्ठ है।
\end{proof}

\begin{method}\label{met:b3:groups:counting}
स्थिर बिंदुओं की गणना परिमित समूह सिद्धांत की सार्वत्रिक पहली
चाल है। किसी वस्तु का \emph{अस्तित्व} सिद्ध करने के लिए (कोई
केंद्रीय अवयव, क्रम $p$ का अवयव, कोई \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह}, कोई
स्थिर बिंदु) एक सुचुने हुए समूह को एक सुचुने हुए परिमित समुच्चय
पर \omterm{def:b3:groups:action}{क्रिया} कराइए, फिर $p$ के सापेक्ष $\abs{X^G}$ और $\abs X$ की तुलना
कीजिए, अथवा कक्षाओं के आकारों को समूह के क्रम का भाजक बनने
दीजिए। \cref{thm:b3:groups:pfixed,thm:b3:groups:cauchy} की और नीचे दी गई तीनों सिलो प्रमेयों की उपपत्तियाँ
इसी एक विचार के पाँच रूपांतर हैं।
\end{method}

\section{सिलो प्रमेय}

लाग्राँज की प्रमेय कहती है कि किसी उपसमूह का क्रम $\abs G$ को
विभाजित करता है; इसका विलोम असत्य है ($A_4$, जिसका क्रम $12$
है, में क्रम $6$ का कोई उपसमूह नहीं है --- \cref{exo:b3:groups:1})। सिलो प्रमेय
अभाज्य घातों के लिए इस विलोम को बचा लेती हैं, और उनका गणना संबंधी
अंश \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} उत्पन्न करने का हमारा सबसे तीक्ष्ण सामान्य
उपकरण है।

\begin{definition}\label{def:b3:groups:sylow}
$p \nmid m$ के साथ $\abs G = p^a m$ लिखिए। $G$ का \emph{सिलो
$p$-उपसमूह}\index{सिलो उपसमूह} क्रम $p^a$ का उपसमूह है ---
अर्थात् संभवतम सबसे बड़े क्रम का $p$-उपसमूह। $G$ के सिलो
$p$-उपसमूहों की संख्या $n_p$ से निरूपित की जाती है।
\end{definition}

\begin{lemma}\label{lem:b3:groups:binom}
यदि $p \nmid m$ के साथ $\abs G = p^a m$ हो, तो $\dbinom{p^a m}{p^a}
\equiv m \pmod p$।
\end{lemma}

\begin{proof}
$\mathbb F_p[X]$ में, नौसिखिए का स्वप्न $(1+X)^p = 1 + X^p$ (गुणांक $\binom pk$, $0<k<p$, $p$
से विभाज्य हैं: $p$ $\frac{p!}{k!(p-k)!}$ के अंश को विभाजित करता है पर हर को
नहीं) पुनरावृत्ति से $(1+X)^{p^a} = 1 + X^{p^a}$ बन जाता है, जिससे
\[
(1+X)^{p^a m} = \bigl(1 + X^{p^a}\bigr)^m
= \sum_{k=0}^{m} \binom mk X^{k p^a} \quad \mathbb
F_p[X] \text{ में}.
\]
अब $X^{p^a}$ के गुणांक की पहचान कीजिए: बाईं ओर $\binom{p^a
m}{p^a} \bmod p$, दाईं ओर $\binom m1 = m$।
\end{proof}

\begin{theorem}[सिलो I: अस्तित्व]\label{thm:b3:groups:sylow1}
प्रत्येक अभाज्य $p$ के लिए $G$ के सिलो $p$-उपसमूह विद्यमान
हैं।
\end{theorem}

\begin{proof}[उपपत्ति (वीलांट)]
मान लीजिए $\Omega$ गणनीयता $p^a$ वाले $G$ के \emph{उपसमुच्चयों}
का समुच्चय है; $G$ बाएँ स्थानांतरण $g \cdot S = gS$ द्वारा $\Omega$ पर
\omterm{def:b3:groups:action}{क्रिया} करता है। \cref{lem:b3:groups:binom} से $\abs\Omega = \binom{p^a m}{p^a}
\equiv m \not\equiv 0 \pmod p$, अतः किसी \omterm{def:b3:groups:action}{कक्षा} $\mathcal O_S$ का आकार
$p$ से सह-अभाज्य है (यदि $p$ हर \omterm{def:b3:groups:action}{कक्षा} के आकार को विभाजित
करता, तो वह $\abs\Omega$ को भी विभाजित करता)। ऐसे ही किसी $S$ का
\omterm{def:b3:groups:action}{स्थायीकारी} $H = G_S$ लीजिए। चूँकि $[G : H] = \abs{\mathcal O_S}$ $p$ से सह-अभाज्य है और
$p^a \mid
\abs G = [G:H]\,\abs H$, हमें $p^a \mid \abs H$ मिलता है। विलोमतः, $s \in S$ को स्थिर कीजिए:
प्रतिचित्रण $H \to S$, $h \mapsto hs$, एकैकी है और $S$ में उतरता है क्योंकि
$hS = S$; अतः $\abs H \leq \abs S = p^a$। इसलिए $\abs H = p^a$।
\end{proof}

\begin{theorem}[सिलो II: प्रभुत्व और संयुग्मता]\label{thm:b3:groups:sylow2}
मान लीजिए $P$ एक सिलो $p$-उपसमूह है और $Q$ $G$ का कोई
भी $p$-उपसमूह। तब किसी $g \in G$ के लिए $Q \subseteq gPg^{-1}$। विशेष रूप से सभी
सिलो $p$-उपसमूह संयुग्म होते हैं, और $P \trianglelefteq G \iff
n_p = 1$।
\end{theorem}

\begin{proof}
$Q$ को गणनीयता $m
\not\equiv 0 \pmod p$ वाली सहसमुच्चय समष्टि $X = G/P$ पर \omterm{def:b3:groups:action}{क्रिया}
करने दीजिए। $p$-समूह $Q$ पर \cref{thm:b3:groups:pfixed} लगाने से $\abs{X^Q} \equiv m \not\equiv 0 \pmod p$: कोई स्थिर
सहसमुच्चय $gP$ है, अर्थात् $QgP = gP$, अर्थात् $g^{-1}Qg \subseteq
P$। यदि $Q$
स्वयं एक \omterm{def:b3:groups:sylow}{सिलो उपसमूह} हो, तो क्रमों की समानता $Q \subseteq gPg^{-1}$ को समता में
बदल देती है। अंततः $P
\trianglelefteq G$ तभी और केवल तभी जब उसके संयुग्म $\{gPg^{-1}\}$
--- जो ऊपर के अनुसार \emph{समस्त} सिलो $p$-उपसमूह हैं --- घटकर
$\{P\}$ रह जाएँ।
\end{proof}

\begin{theorem}[सिलो III: गणना]\label{thm:b3:groups:sylow3}
$n_p \equiv 1 \pmod p$, तथा $n_p = [G : N_G(P)]$, जो $m$ को विभाजित करता है।
\end{theorem}

\begin{proof}
मान लीजिए $\mathrm{Syl}_p$ सिलो $p$-उपसमूहों का समुच्चय है; $G$ उस पर
संयुग्मन द्वारा संक्रमणीय रूप से \omterm{def:b3:groups:action}{क्रिया} करता है (\cref{thm:b3:groups:sylow2}), और
$P$ का \omterm{def:b3:groups:action}{स्थायीकारी} \omterm{ex:b3:groups:actions}{प्रसामान्यक} $N_G(P) \supseteq P$ है: $n_p = [G :
N_G(P)]$, और $m = [G:P] = [G:N_G(P)]\,[N_G(P):P]$ से
$n_p \mid
m$ सिद्ध हो जाता है।

अब \omterm{def:b3:groups:action}{क्रिया} को $P$ तक सीमित कीजिए और स्थिर बिंदु गिनिए। यदि
$Q \in
\mathrm{Syl}_p$ $P$ द्वारा स्थिर हो, तो $P \subseteq N_G(Q)$; $P$ और $Q$ दोनों
समूह $N_G(Q)$ के सिलो $p$-उपसमूह हैं, अतः उसी में संयुग्म हैं
($N_G(Q)$ पर \cref{thm:b3:groups:sylow2} लगाइए); परंतु $Q \trianglelefteq N_G(Q)$, इसलिए वहाँ $Q$ का एकमात्र
संयुग्म वह स्वयं है: $P = Q$। इस प्रकार एकमात्र स्थिर बिंदु $P$
स्वयं है, और \cref{thm:b3:groups:pfixed} से $n_p = \abs{\mathrm{Syl}_p}
\equiv \abs{\mathrm{Syl}_p^P} = 1 \pmod p$ मिल जाता है।
\end{proof}

\begin{method}\label{met:b3:groups:sylowmethod}
दिए हुए क्रम $n = p^a m$ के समूह का विश्लेषण करने के लिए: $m$ के उन
भाजकों की सूची बनाइए जो $p$ के सापेक्ष $1$ के सर्वांगसम हैं
--- ये $n_p$ के उम्मीदवार हैं। यदि एकमात्र उम्मीदवार $1$ हो,
तो सिलो $p$-उपसमूह \omterm{def:b3:groups:normal}{प्रसामान्य} है। यदि $n_p > 1$ का छोटा होना
अनिवार्य हो, तो $\mathrm{Syl}_p$ पर संयुग्मन द्वारा \omterm{def:b3:groups:action}{क्रिया} कराकर छोटी अष्टि
वाली समाकारिता $G \to S_{n_p}$ प्राप्त कीजिए। और अवयव गिनिए: \emph{अभाज्य}
क्रम $p$ के भिन्न-भिन्न सिलो $p$-उपसमूह तुच्छ रूप से
प्रतिच्छेद करते हैं, अतः वे ठीक क्रम $p$ के $n_p(p-1)$ अवयव धारण
करते हैं; भिन्न अभाज्यों के लिए ये गणनाएँ अतिव्यापी होकर प्रायः
विरोधाभास पर विवश कर देती हैं (\cref{exo:b3:groups:7})।
\end{method}

\begin{example}\label{ex:b3:groups:pq}
मान लीजिए $\abs G = pq$ है, जहाँ $p < q$ अभाज्य हैं और $p \nmid q - 1$। तब $n_q \mid p$
और $n_q \equiv 1 \bmod q$ से $n_q = 1$ अनिवार्य हो जाता है (क्योंकि $p <
q$); $n_p \mid q$
और $n_p \equiv 1 \bmod p$ से $n_p = 1$ (क्योंकि $q
\not\equiv 1 \bmod p$)। मान लीजिए $P, Q$ दोनों
\omterm{def:b3:groups:normal}{प्रसामान्य} \omterm{def:b3:groups:sylow}{सिलो उपसमूह} हैं: $P
\cap Q = \{e\}$ (क्रम सह-अभाज्य हैं), अतः
$\abs{PQ} = pq$ (\cref{exo:b3:groups:4}) और नीचे दी गई \cref{prop:b3:groups:direct} से $G \cong P \times Q \cong \Z/p\Z
\times \Z/q\Z \cong \Z/pq\Z$। क्रम $15$, $33$,
$35$, \dots\ का प्रत्येक समूह चक्रीय है। अपवर्जित स्थिति $p \mid q - 1$
ठीक एक और समूह देती है, जो अनाबेली है --- सप्ताहांत समस्या
देखिए (\cref{pb:b3:groups:1})।
\end{example}

\begin{example}[एक पूर्ण सिलो गणना: $S_4$]
\label{ex:b3:groups:s4sylow}
आइए इस विधि को $G = S_4$, $\abs G = 24 = 2^3\cdot3$ पर चलाएँ।
\emph{सिलो $3$:} $n_3 \mid 8$, $n_3 \equiv 1 \bmod 3$, अतः $n_3
\in \{1, 4\}$; और चूँकि $\langle(123)\rangle$ तथा
$\langle(124)\rangle$ भिन्न हैं, $n_3 = 4$ --- अर्थात् चार उपसमूह $\langle(abc)\rangle$, प्रत्येक
$3$-अवयवी उपसमुच्चय $\{a, b, c\}$ के लिए एक, जो $8$ त्रि-चक्रों का
लेखा-जोखा पूरा कर देते हैं। सिलो~II से वे संयुग्म हैं, और यहाँ
संयुग्मन समाकारिता $S_4 \to
S_{\mathrm{Syl}_3} \cong S_4$ एक तुल्याकारिता है (उसकी अष्टि क्रम
$24/4 =
6$ वाले $N = N_G(\langle(123)\rangle)$ में अंतर्विष्ट है, और $S_3$ जैसे $N$ के भीतर
$S_4$ का \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} तुच्छ ही हो सकता है: वह चारों सिलो
उपसमूहों को स्थिर रखने वाले सम क्रमचयों से बनता, और ऐसा केवल
$e$ करता है)। \emph{सिलो $2$:} $n_2 \mid 3$, $n_2
\equiv 1 \bmod 2$: $n_2 \in \{1, 3\}$। उपसमूह
$D =
\langle(1234), (13)\rangle$ का क्रम $8$ है (एक द्विफलकी $D_4$: शीर्षों $1, 2, 3, 4$ वाले
वर्ग की सममितियाँ), वह \omterm{def:b3:groups:normal}{प्रसामान्य} नहीं है ($(12)(1234)(12) = (2134)$ एक भिन्न
$4$-चक्र उपसमूह जनित करता है), अतः $n_2 = 3$: $D_4$ की तीन
प्रतियाँ $4$ बिंदुओं को एक ``वर्ग'' में युग्मित करने के तीन
तरीकों के अनुरूप हैं। इस गणना का सार ध्यान देने योग्य है: $\abs{S_4} = 24$
में दोनों में से किसी भी \omterm{def:b3:groups:sylow}{सिलो उपसमूह} के अप्रसामान्य होने की
गुंजाइश बची रहती है, और दोनों ही अप्रसामान्य निकलते हैं ---
इसकी तुलना क्रम $12$ से कीजिए, जहाँ गणना उनमें से एक को
\omterm{def:b3:groups:normal}{प्रसामान्य} होने पर विवश कर देती है (\cref{pb:b3:groups:1} का भाग~IV)।
\end{example}

\section{गुणन: प्रत्यक्ष और अर्ध-प्रत्यक्ष}

\begin{proposition}[प्रत्यक्ष गुणन की पहचान]\label{prop:b3:groups:direct}
मान लीजिए $H, K \trianglelefteq G$ के साथ $H \cap K = \{e\}$ और $HK = G$। तब $(h,k) \mapsto hk$ एक तुल्याकारिता
$H \times K \to
G$ है।\index{प्रत्यक्ष गुणन}
\end{proposition}

\begin{proof}
$h \in H$, $k \in K$ के लिए \omterm{def:b3:groups:derived}{क्रमविनिमेयक} $hkh^{-1}k^{-1}$ $K$ में है ($K$ की
प्रसामान्यता का उपयोग करते हुए इसे $(hkh^{-1})k^{-1}$ के रूप में पढ़िए) और
$H$ में भी (इसे $h(kh^{-1}k^{-1})$ के रूप में पढ़िए): अतः वह $e$ है,
इसलिए $H$ और $K$ अवयवशः क्रमविनिमेय हैं और प्रतिचित्रण
समाकारिता है। यह आच्छादक है क्योंकि $HK = G$, और एकैकी है क्योंकि
$hk = e$ से $h = k^{-1} \in
H \cap K = \{e\}$ मिलता है।
\end{proof}

\emph{दोनों} गुणनखंडों की प्रसामान्यता ही सबसे अधिक बार विफल
होती है: $S_3
= \langle (1\,2\,3)\rangle \,\langle(1\,2)\rangle$ में दोनों गुणनखंड तुच्छ रूप से प्रतिच्छेद करते हैं
और मिलकर पूरा समूह जनित करते हैं, फिर भी $S_3 \not\cong \Z/3\Z \times
\Z/2\Z$। जब केवल एक ही
गुणनखंड \omterm{def:b3:groups:normal}{प्रसामान्य} हो, तब उपयुक्त धारणा यह है:

\begin{definition}\label{def:b3:groups:semidirect}
मान लीजिए $H$, $K$ समूह हैं और $\varphi \colon K \to
\operatorname{Aut}(H)$ एक समाकारिता है।
\emph{अर्ध-प्रत्यक्ष गुणन}\index{अर्ध-प्रत्यक्ष गुणन} $H \rtimes_\varphi K$
समुच्चय $H \times K$ है, जिस पर संक्रिया
\[
(h, k)\,(h', k') = \bigl(h\,\varphi(k)(h'),\; kk'\bigr).
\]
\end{definition}

\begin{proposition}\label{prop:b3:groups:semidirect}
$H \rtimes_\varphi K$ एक समूह है; $H \times \{e\}$ $H$ के तुल्याकारी एक \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह}
है और $\{e\} \times K$ $K$ के तुल्याकारी एक उपसमूह; वे तुच्छ रूप से
प्रतिच्छेद करते हैं और मिलकर पूरा समूह जनित करते हैं। विलोमतः,
यदि $N \trianglelefteq G$, $K \leq G$ और $N \cap K = \{e\}$ के साथ $G =
NK$ हो, तो $\varphi(k) = (n \mapsto
knk^{-1})$ के लिए
$G \cong N \rtimes_\varphi K$।
\end{proposition}

\begin{proof}
सीधा सत्यापन: साहचर्य $\varphi(kk') =
\varphi(k)\circ\varphi(k')$ पर और प्रत्येक $\varphi(k)$ के समाकारिता
होने पर सिमट जाता है; तत्समक $(e,e)$ है और $(h,k)^{-1} =
\bigl(\varphi(k^{-1})(h^{-1}), k^{-1}\bigr)$। प्रक्षेप $(h,k)
\mapsto k$
$K$ पर एक आच्छादक समाकारिता है जिसकी अष्टि $H \times \{e\}$ है, अतः वह
\omterm{def:b3:groups:normal}{प्रसामान्य} है। विलोम के लिए: प्रत्येक $g \in G$ $n \in N$, $k \in K$ के साथ
\emph{अद्वितीय} रूप से $nk$ लिखा जाता है (अस्तित्व: $G =
NK$;
अद्वितीयता: $nk = n'k'$ से $n'^{-1}n = k'k^{-1} \in N \cap
K$ मिलता है), और
\[
(nk)(n'k') = n\,(kn'k^{-1})\;kk'
\]
से पता चलता है कि $nk \mapsto (n, k)$ $G$ की संक्रिया को $N \rtimes_\varphi K$ की संक्रिया
पर ले जाता है।
\end{proof}

\begin{example}\label{ex:b3:groups:semidirectexamples}
(क) किसी सम $n$-भुज की $2n$ सममितियों का \emph{द्विफलकी
समूह}\index{द्विफलकी समूह} $D_n$ ($n \geq
3$): घूर्णन सूचकांक $2$
का एक \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} बनाते हैं, कोई भी परावर्तन एक पूरक जनित
करता है, और किसी घूर्णन का परावर्तन द्वारा संयुग्मन उसे उलट देता
है: $\varphi(1) = (x
\mapsto -x)$ के साथ $D_n \cong \Z/n\Z \rtimes_\varphi \Z/2\Z$।
(ख) एक रेखा का आफ़ीन समूह, $\{x \mapsto ax + b : a \in
K^\times,\, b \in K\} \cong K \rtimes K^\times$: स्थानांतरण \omterm{def:b3:groups:normal}{प्रसामान्य},
समरूपण एक पूरक।
(ग) $S_n \cong A_n \rtimes \Z/2\Z$ (पूरक: कोई भी स्थानांतरण)।
(घ) \omterm{pb:b3:groups:1}{क्वाटर्नियन समूह} $Q_8$ उचित उपसमूहों का \omterm{def:b3:groups:semidirect}{अर्ध-प्रत्यक्ष गुणन}
\emph{नहीं} है: प्रत्येक अतुच्छ उपसमूह $-1$ को अंतर्विष्ट करता
है (\cref{pb:b3:groups:1}), अतः कोई भी दो उचित उपसमूह तुच्छ रूप से प्रतिच्छेद
नहीं करते।
\end{example}

\section{विलेय समूह; \texorpdfstring{$A_n$}{An} की सरलता}

\begin{definition}\label{def:b3:groups:derived}
$x, y \in G$ का \emph{क्रमविनिमेयक}\index{क्रमविनिमेयक} $[x,y]
= xyx^{-1}y^{-1}$ है;
\emph{व्युत्पन्न उपसमूह}\index{व्युत्पन्न उपसमूह} $D(G)$ समस्त
क्रमविनिमेयकों से जनित उपसमूह है। \emph{व्युत्पन्न श्रेणी}
$D^0(G) = G$, $D^{i+1}(G) = D(D^i(G))$ है, और $G$ \emph{विलेय}\index{विलेय समूह} कहलाता
है यदि किसी $n$ के लिए $D^n(G) =
\{e\}$ हो।
\end{definition}

\begin{proposition}\label{prop:b3:groups:derived}
$D(G)$ \omterm{def:b3:groups:normal}{प्रसामान्य} है (वस्तुतः प्रत्येक स्वसमाकारिता के अंतर्गत
स्थिर), $G/D(G)$ आबेली है, और $N \trianglelefteq G$ के लिए: $G/N$ आबेली $\iff
D(G) \subseteq N$।
इसके अतिरिक्त $G$ \omterm{def:b3:groups:derived}{विलेय} है तभी और केवल तभी जब कोई ऐसी शृंखला
$G = G_0 \trianglerighteq G_1 \trianglerighteq \dots
\trianglerighteq G_n = \{e\}$ हो जिसमें प्रत्येक $G_{i+1} \trianglelefteq
G_i$ और प्रत्येक भागफल $G_i/G_{i+1}$ आबेली हो।
\omterm{def:b3:groups:derived}{विलेय} समूहों के उपसमूह और भागफल \omterm{def:b3:groups:derived}{विलेय} होते हैं; विलोमतः, यदि
$N$ और $G/N$ \omterm{def:b3:groups:derived}{विलेय} हों, तो $G$ भी \omterm{def:b3:groups:derived}{विलेय} है।
\end{proposition}

\begin{proof}
कोई स्वसमाकारिता $\alpha$ $[x,y]$ को $[\alpha x, \alpha y]$ पर भेजती है: वह
क्रमविनिमेयकों का क्रमचय करती है, अतः उनसे जनित उपसमूह को
सुरक्षित रखती है; संयुग्मन स्वसमाकारिताएँ हैं, जिससे प्रसामान्यता
सिद्ध हो जाती है। $G/D(G)$ में $\bar x\bar y \bar x^{-1}\bar y^{-1} = \overline{[x,y]} = \bar e$: भागफल आबेली है। यदि $G/N$
आबेली हो तो प्रत्येक $[x,y] \in
N$, अतः $D(G) \subseteq N$; विलोमतः यदि $D(G) \subseteq N$ हो तो
$G/N$ तृतीय तुल्याकारिता प्रमेय से आबेली $G/D(G)$ का भागफल होने के
कारण आबेली है।

यदि $G$ \omterm{def:b3:groups:derived}{विलेय} हो, तो \omterm{def:b3:groups:derived}{व्युत्पन्न श्रेणी} ऐसी ही एक शृंखला है।
विलोमतः, ऐसी शृंखला दी हो तो आगमन द्वारा $D^i(G) \subseteq G_i$: $G_i/G_{i+1}$ के आबेली
होने से $D(G_i) \subseteq G_{i+1}$ मिलता है, अतः $D^{i+1}(G) = D(D^i G)
\subseteq D(G_i) \subseteq G_{i+1}$; इसलिए $D^n(G) = \{e\}$।

वंशानुक्रम: $H \leq G$ के लिए $D^i(H) \subseteq D^i(G)$ (आगमन), और $D^i(G/N) = \pi(D^i(G))$, क्योंकि $\pi$
क्रमविनिमेयकों को क्रमविनिमेयकों पर आच्छादित करता है; इससे
उपसमूहों और भागफलों संबंधी कथन मिल जाते हैं। विस्तार: यदि $D^m(G/N) = \{e\}$
हो तो $D^m(G) \subseteq N$, और $D^n(N) = \{e\}$ से $D^{m+n}(G) = D^n(D^m(G)) \subseteq D^n(N)
= \{e\}$ मिलता है।
\end{proof}

\begin{example}\label{ex:b3:groups:solvableexamples}
आबेली समूह \omterm{def:b3:groups:derived}{विलेय} होते हैं। $p$-समूह क्रम पर आगमन द्वारा \omterm{def:b3:groups:derived}{विलेय}
होते हैं: $Z(G) \neq \{e\}$ और $G/Z(G)$ एक छोटा $p$-समूह है। $S_3$ और $S_4$
\omterm{def:b3:groups:derived}{विलेय} हैं: $S_4 \trianglerighteq A_4
\trianglerighteq V \trianglerighteq \{e\}$, जहाँ $V = \{e,
(1\,2)(3\,4), (1\,3)(2\,4), (1\,4)(2\,3)\}$ द्विक स्थानांतरणों का क्लाइन समूह
है ($S_4$ में \omterm{def:b3:groups:normal}{प्रसामान्य}: वह संयुग्मन वर्गों का सम्मिलन है),
और भागफल $\Z/2\Z$, $\Z/3\Z$, $V$ आबेली हैं। \cref{ch:b3:galois} में ``घात $n$
का व्यापक समीकरण मूलों द्वारा हल्य है'' का शाब्दिक \emph{अर्थ}
ही होगा ``$S_n$ एक \omterm{def:b3:groups:derived}{विलेय समूह} है''। इसी से अगली परिभाषा का महत्त्व
स्पष्ट होता है।
\end{example}

\begin{definition}\label{def:b3:groups:simple}
कोई समूह $G \neq \{e\}$ \emph{सरल}\index{सरल समूह} कहलाता है यदि उसके
एकमात्र \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} $\{e\}$ और $G$ हों। अनाबेली सरल समूह
\omterm{def:b3:groups:derived}{विलेय} नहीं होता: $D(G) \trianglelefteq G$ $\{e\}$ नहीं है (अन्यथा $G$ आबेली होता),
अतः $D(G) = G$ और \omterm{def:b3:groups:derived}{व्युत्पन्न श्रेणी} अचर रहती है। आबेली सरल समूह ठीक
$\Z/p\Z$ हैं, जहाँ $p$ अभाज्य है (कोई आबेली समूह सरल है तभी और
केवल तभी जब उसका कोई उचित अतुच्छ उपसमूह न हो, अर्थात् लाग्राँज से
वह अभाज्य क्रम का चक्रीय समूह हो)।
\end{definition}

\begin{lemma}\label{lem:b3:groups:threecycles}
$n \geq 3$ के लिए $A_n$ $3$-चक्रों से जनित है; और $n \geq 5$ के लिए
समस्त $3$-चक्र \emph{$A_n$ में} संयुग्म हैं।
\end{lemma}

\begin{proof}
$A_n$ का कोई अवयव सम संख्या में स्थानांतरणों का गुणनफल है; उन्हें
जोड़ों में बाँटिए और (दाएँ से बाएँ संयोजन लेते हुए)
\[
(a\,b)(c\,d) = (a\,c\,b)(a\,c\,d), \qquad
(a\,b)(b\,c) = (a\,b\,c), \qquad
(a\,b)(a\,b) = e
\]
का क्रमशः असंयुक्त, अतिव्यापी और समान जोड़ों के लिए उपयोग कीजिए:
स्थानांतरणों का प्रत्येक जोड़ा $3$-चक्रों का गुणनफल है।

संयुग्मता: $\sigma(a\,b\,c)\sigma^{-1} = (\sigma a\; \sigma b\;
\sigma c)$, अतः कोई भी दो $3$-चक्र किसी $\sigma \in
S_n$ द्वारा
संयुग्म हैं। यदि $\sigma$ विषम हो, तो उसके स्थान पर $\sigma' = \sigma (d\,e)$ लीजिए,
जहाँ $d, e$ $\{a, b, c\}$ के बाहर के दो बिंदु हैं --- वे विद्यमान हैं
क्योंकि $n \geq 5$; तब $\sigma'$ सम है और $\sigma'(a\,b\,c)
\sigma'^{-1} = \sigma(a\,b\,c)\sigma^{-1}$, क्योंकि $(d\,e)$ $(a\,b\,c)$ के
साथ क्रमविनिमेय है।
\end{proof}

\begin{theorem}[एकांतर समूह की सरलता]\label{thm:b3:groups:ansimple}
$n \geq 5$ के लिए $A_n$ \omterm{def:b3:groups:simple}{सरल} है।\index{एकांतर समूह}
\end{theorem}

\begin{proof}
मान लीजिए $N \trianglelefteq A_n$, $N \neq \{e\}$। \cref{lem:b3:groups:threecycles} से यह दिखाना पर्याप्त है कि
$N$ में \emph{एक} $3$-चक्र है: तब $A_n$ में $3$-चक्रों की
प्रसामान्यता और संयुग्मता समस्त $3$-चक्रों को $N$ में डाल
देती है, अतः $N = A_n$।

$\rho \in S_n$ के लिए मान लीजिए $F(\rho) = \{x : \rho(x) \neq x\}$ उसका \emph{आलंब} है और $f(\rho) = \abs{F(\rho)}$।
\emph{न्यूनतम} $f(\sigma)$ वाला $\sigma \in N
\setminus \{e\}$ चुनिए। ध्यान दीजिए कि किसी
अतुच्छ सम क्रमचय के लिए $f \geq 3$ होता है, और $\sigma \in A_n$ के लिए $f(\sigma) =
4$
असंभव है जब तक $\sigma$ एक द्विक स्थानांतरण न हो ($4$-चक्र विषम
होता है)। हम दिखाएँगे कि $\sigma$ एक $3$-चक्र है।

\emph{स्थिति क: $\sigma$ असंयुक्त स्थानांतरणों का गुणनफल है},
मान लीजिए $f(\sigma) \geq 4$ के साथ $\sigma = (a\,b)(c\,d)\cdots$। $e'
\notin \{a, b, c, d\}$ चुनिए (यह संभव है: $n \geq 5$),
$\tau =
(c\,d\,e')$ रखिए और
\[
\sigma' = \tau\sigma\tau^{-1}\,\sigma^{-1} \in N
\qquad (\tau\sigma\tau^{-1} \in N \text{, प्रसामान्यता से}).
\]
चूँकि $\sigma\tau^{-1}\sigma^{-1} = (\sigma c\;\sigma e'\;\sigma
d) = (d\;\sigma e'\;c)$ ($\sigma c = d$, $\sigma d = c$ का उपयोग करते हुए), हमें $\sigma' = (c\,d\,e')(d\;\sigma e'\;c)$ मिलता
है।

यदि $\sigma e' = e'$ (जो विशेष रूप से तब होता है जब $f(\sigma) =
4$, अर्थात् $\sigma = (a\,b)(c\,d)$):
तब $(d\,e'\,c) =
(c\,d\,e')$ और $\sigma' = (c\,d\,e')^2 = (c\,e'\,d)$, जो $N$ में स्थित एक $3$-चक्र है, जिसके
लिए $f(\sigma') = 3 < 4 \leq f(\sigma)$ --- यह न्यूनतमता का खंडन है।

यदि $\sigma e' \neq e'$: तब $\sigma e' \notin \{a, b, c, d,
e'\}$ ($\sigma$ $a,b$ और $c,d$ की अदला-बदली करता है,
और एकैकी $\sigma$ के साथ $e' \notin \{a,b,c,d\}$), अतः $\sigma$ छह बिंदुओं $a, b,
c, d, e', \sigma e'$ को
हिलाता है: $f(\sigma) \geq 6$। दूसरी ओर $\sigma'$, जो $\{c, d,
e', \sigma e'\}$ में आलंब वाले दो
$3$-चक्रों का गुणनफल है, $f(\sigma') \leq 4$ को संतुष्ट करता है; और $\sigma' \neq
e$,
क्योंकि $\sigma'(d) = \tau\sigma\tau^{-1}(c) = \tau\sigma(e') =
\sigma e' \neq d$ ($\tau$ $\sigma e' \notin \{c,d,e'\}$ को स्थिर रखता है)। अतः $f(\sigma') \leq 4 <
f(\sigma)$ के साथ
$\sigma' \in N \setminus\{e\}$: न्यूनतमता का खंडन हो जाता है।

\emph{स्थिति ख: $\sigma$ के किसी चक्र की लंबाई $\geq 3$ है}, मान
लीजिए $\sigma(a) = b$, जहाँ $a, b, c$ भिन्न हैं और $\sigma(b) = c$। यदि $\sigma$ ठीक यही
$3$-चक्र हो, तो काम पूरा हुआ। अन्यथा $f(\sigma) \geq 5$ ($\geq3$-चक्र के साथ
$f(\sigma) = 4$ वाली स्थिति विषम $4$-चक्र है, जो अपवर्जित है), अतः हम
$d, e' \in
F(\sigma) \setminus \{a, b, c\}$ चुन सकते हैं। $\tau = (c\,d\,e')$ रखिए और $\sigma' = \tau\sigma\tau^{-1}\sigma^{-1} \in N$। पहले की तरह $\sigma' = (c\,d\,e')\,(\sigma c\;\sigma e'\;\sigma d)$
केवल
\[
M = \{c, d, e'\} \cup \{\sigma c, \sigma d, \sigma e'\}
\subseteq F(\sigma)
\]
के बिंदुओं को हिलाता है (हिले हुए बिंदुओं के प्रतिबिंब भी हिलते
हैं: $\sigma(x) \ne x$ से $\sigma(\sigma x) \neq \sigma x$ मिलता है, क्योंकि $\sigma$ एकैकी है)। साथ ही
$b \notin M$: पाँच बिंदु $a, b, c, d, e'$ भिन्न हैं, अतः $b
\notin \{c, d, e'\}$; और $b \in \{\sigma c, \sigma d, \sigma
e'\}$ से $a \in \{c, d, e'\}$
अनिवार्य हो जाता ($\sigma a = b$ का उपयोग करते हुए $\sigma^{-1}$ लगाइए), जो असत्य
है। अतः $\sigma'$ $b$ को स्थिर रखता है, जबकि $\sigma$ $b$ को हिलाता
है; और $F(\sigma') \subseteq F(\sigma)$। अंततः $\sigma' \neq e$: $\sigma^{-1}(c) = b$, $\tau^{-1}(b) = b$, $\sigma(b) = c$, $\tau(c) = d$, अतः
$\sigma'(c) = d \neq c$। इस प्रकार $f(\sigma') \leq f(\sigma) - 1$ के साथ $\sigma' \in N
\setminus \{e\}$, जो न्यूनतमता का खंडन करता
है।

दोनों स्थितियाँ असंभव होने से $\sigma$ एक $3$-चक्र है।
\end{proof}

\begin{corollary}\label{cor:b3:groups:snnotsolvable}
$n \geq 5$ के लिए: $A_n$ और $S_n$ \omterm{def:b3:groups:derived}{विलेय} नहीं हैं, और $S_n$ के
एकमात्र \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} $\{e\}$, $A_n$ और $S_n$ हैं।
\end{corollary}

\begin{proof}
$A_n$ अनाबेली \omterm{def:b3:groups:simple}{सरल} है, अतः \omterm{def:b3:groups:derived}{विलेय} नहीं (\cref{def:b3:groups:simple}); और अविलेय उपसमूह
वाला समूह \omterm{def:b3:groups:derived}{विलेय} नहीं होता (\cref{prop:b3:groups:derived})। मान लीजिए $N
\trianglelefteq S_n$: तब $N \cap A_n \trianglelefteq A_n$
$\{e\}$ या $A_n$ के बराबर है। यदि $N \cap A_n = A_n$ हो, तो $A_n \subseteq N$ और सूचकांक से
$N \in \{A_n, S_n\}$। यदि $N \cap A_n = \{e\}$ हो, तो प्रक्षेप $S_n \to S_n/A_n \cong
\Z/2\Z$ का $N$ पर प्रतिबंध
एकैकी है, अतः $\abs N \leq 2$; और यदि $N = \{e, \sigma\}$ हो, तो प्रसामान्यता से $\sigma$
का \omterm{ex:b3:groups:actions}{संयुग्मन वर्ग} $\{\sigma\}$ के बराबर हो जाता है, अर्थात् $\sigma \in Z(S_n)$। परंतु
$n \geq 3$ के लिए $Z(S_n) = \{e\}$: यदि $\sigma \ne e$ $a$ को $b \neq a$ पर ले जाए, तो
$c
\notin \{a, b\}$ चुनिए; तब $(b\,c)\sigma(b\,c)^{-1}$ $a$ को $c
\ne b$ पर भेजता है, अतः वह $\sigma$
से भिन्न है। अतः $N = \{e\}$।
\end{proof}

\begin{theorem}[जॉर्डन--होल्डर]\label{thm:b3:groups:jordanholder}
प्रत्येक परिमित समूह $G \neq \{e\}$ एक \emph{संघटन
श्रेणी}\index{संघटन श्रेणी}
\[
\{e\} = G_0 \trianglelefteq G_1 \trianglelefteq \cdots
\trianglelefteq G_r = G,
\qquad G_{i}/G_{i-1} \text{ सरल},
\]
स्वीकार करता है, और \emph{संघटन गुणनखंडों} $G_i/G_{i-1}$ का बहुसमुच्चय,
तुल्याकारिता तक, चुनी गई श्रेणी पर निर्भर नहीं करता। कोई परिमित
समूह \omterm{def:b3:groups:derived}{विलेय} है तभी और केवल तभी जब उसके सभी संघटन गुणनखंड अभाज्य
क्रम के चक्रीय समूह हों।\index{जॉर्डन--होल्डर प्रमेय}
\end{theorem}

\begin{proof}
\emph{अस्तित्व}: $\abs G$ पर आगमन। यदि $G$ \omterm{def:b3:groups:simple}{सरल} हो, तो $\{e\} \trianglelefteq G$
लीजिए। अन्यथा कोई महत्तम उचित \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} $N$ चुनिए
(उपसमूह परिमित संख्या में हैं); संगति प्रमेय से $G/N$ \omterm{def:b3:groups:simple}{सरल} है
($G/N$ का कोई उचित अतुच्छ \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} $N$ और $G$ के
बीच कड़ाई से स्थित $G$ के \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} तक उठ जाता)।
$N$ की एक \omterm{thm:b3:groups:jordanholder}{संघटन श्रेणी} में $N \trianglelefteq G$ जोड़ दीजिए।

\emph{अद्वितीयता}: $\abs G$ पर आगमन; $G$ के \omterm{def:b3:groups:simple}{सरल} होने की स्थिति
स्पष्ट है। दो संघटन श्रेणियाँ लीजिए, जिनके उपांत्य पद $M \trianglelefteq G$ और
$N \trianglelefteq G$ हैं (अतः $G/M$, $G/N$ \omterm{def:b3:groups:simple}{सरल} हैं)। यदि $M = N$ हो, तो $M$ पर
आगमन लगाकर निष्कर्ष निकल आता है। अन्यथा $MN$, जो $G$ में
\omterm{def:b3:groups:normal}{प्रसामान्य} है और $M$ को कड़ाई से अंतर्विष्ट करता है, $G$ के
बराबर है ($M$ महत्तम \omterm{def:b3:groups:normal}{प्रसामान्य} है: कोई भी \omterm{def:b3:groups:normal}{प्रसामान्य} $M \subsetneq L
\subsetneq G$
\omterm{def:b3:groups:simple}{सरल} $G/M$ के एक उचित अतुच्छ \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} पर जाता)। द्वितीय
तुल्याकारिता प्रमेय से
\[
G/M = MN/M \cong N/(M \cap N),
\qquad
G/N = MN/N \cong M/(M \cap N).
\]
$K = M \cap N$ रखिए ($\trianglelefteq G$) और $K$ की एक \omterm{thm:b3:groups:jordanholder}{संघटन श्रेणी} स्थिर कीजिए। तब
$M$ पर दो संघटन श्रेणियाँ हैं: उसकी मूल श्रेणी, और $K$ की
श्रेणी के बाद $K \trianglelefteq
M$ (भागफल $M/K \cong G/N$ \omterm{def:b3:groups:simple}{सरल} है)। आगमन से ($M$ पर
लगाकर) $M$ की मूल श्रेणी के गुणनखंड $\{K \text{ के गुणनखंड}\} \cup \{G/N\}$ हैं; $N$ के लिए
भी वैसा ही। अतः $G$ की दोनों श्रेणियों के गुणनखंड
\[
\{K \text{ के गुणनखंड}\} \;\cup\; \{\,G/N,\; G/M\,\},
\]
हैं --- वही बहुसमुच्चय।

\omterm{def:b3:groups:derived}{विलेयता}: यदि सभी गुणनखंड $\Z/p_i\Z$ हों, तो श्रेणी आबेली भागफलों वाली
एक शृंखला है, अतः $G$ \omterm{def:b3:groups:derived}{विलेय} है (\cref{prop:b3:groups:derived})। विलोमतः, \omterm{def:b3:groups:derived}{विलेय समूह}
का संघटन गुणनखंड \omterm{def:b3:groups:derived}{विलेय} (किसी उपसमूह का भागफल) और \omterm{def:b3:groups:simple}{सरल} होता है;
और \omterm{def:b3:groups:derived}{विलेय} \omterm{def:b3:groups:simple}{सरल समूह} आबेली होता है ($D(G) \ne G$ से $D(G) = \{e\}$ अनिवार्य है),
अतः वह कोई $\Z/p\Z$ है।
\end{proof}

\begin{remark}\label{rem:b3:groups:jhmeaning}
जॉर्डन--होल्डर कहती है कि प्रत्येक परिमित समूह \omterm{def:b3:groups:simple}{सरल} समूहों से बना
है, और उसकी पुर्जों की सूची सुनिश्चित है --- समूहों का एक ऐसा
अंकगणित जिसमें \omterm{def:b3:groups:simple}{सरल समूह} अभाज्यों की भूमिका निभाते हैं, और जहाँ
साधारण गुणा की जगह यह प्रश्न ले लेता है कि \emph{पुर्जे किस
प्रकार जोड़े गए हैं} (विस्तार आँकड़े, जैसे \omterm{def:b3:groups:semidirect}{अर्ध-प्रत्यक्ष गुणन}
में)। परिमित \omterm{def:b3:groups:simple}{सरल} समूहों का वर्गीकरण --- चक्रीय $\Z/p\Z$, एकांतर
$A_{n \geq
5}$, ली प्रकार के सोलह कुल, और $26$ विरल समूह --- बीसवीं
शताब्दी के गणित के स्मारकों में से एक है; लगभग दस हजार शोध-पृष्ठों
में फैली उसकी उपपत्ति इस पाठ्यक्रम से बहुत परे है।
\end{remark}

\begin{exercise}[$\star$]\label{exo:b3:groups:3}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:4}
मान लीजिए $H, K$ किसी परिमित समूह $G$ के उपसमूह हैं।
(क) प्रतिचित्रण $H \times K
\to HK$, $(h,k) \mapsto hk$, के तंतुओं की गणना करके
\emph{गुणनफल सूत्र} $\abs{HK}\,\abs{H\cap K} =
\abs H\, \abs K$ सिद्ध कीजिए।
(ख) दर्शाइए कि $HK$ उपसमूह है तभी और केवल तभी जब $HK = KH$ (जो
दोनों में से एक के \omterm{def:b3:groups:normal}{प्रसामान्य} होने पर स्वतः सत्य है)।
(ग) यदि $H, K \trianglelefteq G$ और $H \cap K = \{e\}$ हों, तो दर्शाइए कि सभी $h \in H$, $k \in K$ के
लिए $hk = kh$।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:5}
(बर्नसाइड की गणना प्रमेयिका) कोई परिमित समूह $G$ किसी परिमित
समुच्चय $X$ पर \omterm{def:b3:groups:action}{क्रिया} करता है। समुच्चय $\{(g,x) : g\cdot x = x\}$ को दो प्रकार से
गिनकर दर्शाइए कि कक्षाओं की संख्या \emph{स्थिर बिंदुओं की औसत
संख्या} है:
\[
\#\{\text{कक्षाएँ}\} = \frac{1}{\abs G}\sum_{g \in G}
\abs{\operatorname{Fix}(g)},
\qquad \operatorname{Fix}(g) = \{x \in X : g \cdot x = x\},
\]
अनुप्रयोग: $\Z/p\Z$ ($p$ अभाज्य) $a$ उपलब्ध रंगों वाले $p$
मनकों के हारों पर घूर्णन द्वारा \omterm{def:b3:groups:action}{क्रिया} करता है; इससे फर्मा की
लघु प्रमेय $a^p \equiv a \pmod p$ निष्कर्ष के रूप में निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:8}
(क) मान लीजिए $H \leq G$ सूचकांक $n$ का है। दर्शाइए कि $G/H$ पर
$G$ की \omterm{def:b3:groups:action}{क्रिया} से एक समाकारिता $G \to S_n$ मिलती है, जिसकी अष्टि
$\bigcap_{g \in
G} gHg^{-1}$ $H$ में अंतर्विष्ट $G$ का सबसे बड़ा \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह}
है।
(ख) निष्कर्ष निकालिए: यदि $G$ परिमित हो और $p$ $\abs G$ का

उपसमूह \omterm{def:b3:groups:normal}{प्रसामान्य} है।
(ग) दर्शाइए कि किसी समुच्चय $X$ पर $G$ की प्रत्येक
संक्रमणीय \omterm{def:b3:groups:action}{क्रिया} किसी सहसमुच्चय समष्टि पर की \omterm{def:b3:groups:action}{क्रिया} के
तुल्याकारी है: क्रियाओं के साथ क्रमविनिमेय एक एकैकी आच्छादक संगति
$X
\to G/G_x$ विद्यमान है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:9}
(क) दर्शाइए कि $D(G)$ आबेली भागफल वाला $G$ का सबसे छोटा
\omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} है, और यह कि $G$ से किसी आबेली समूह में जाने
वाली प्रत्येक समाकारिता \emph{आबेलीकरण} $G^{\mathrm{ab}} = G/D(G)$ से होकर
अद्वितीय रूप से गुजरती है।
(ख) $n \geq 2$ के लिए $D(S_n)$ और $S_n^{\mathrm{ab}}$ ज्ञात कीजिए, तथा $D(Q_8)$ और
$Q_8^{\mathrm{ab}}$ भी।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:10}
मान लीजिए $G$ एक $p$-समूह है और $H \subsetneq G$ उसका उचित उपसमूह।
दर्शाइए कि $H \subsetneq N_G(H)$ (``\omterm{ex:b3:groups:actions}{प्रसामान्यक} बढ़ते हैं''), और इससे निष्कर्ष
निकालिए कि किसी $p$-समूह का प्रत्येक महत्तम उपसमूह सूचकांक
$p$ का \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} है। \emph{($Z(G) \neq \{e\}$ का उपयोग करते हुए
$\abs G$ पर आगमन: $Z(G) \subseteq H$ और $Z(G) \not\subseteq H$ की स्थितियों को अलग-अलग लीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:groups:11}
($A_5$ की सरलता, प्रत्यक्ष रूप से)
(क) दर्शाइए कि $A_5$ के संयुग्मन वर्गों की गणनीयताएँ $1$,
$15$, $20$, $12$, $12$ हैं। $5$-चक्रों के $S_5$-वर्ग के
विभाजन पर ध्यान दीजिए: किसी $5$-चक्र $\sigma$ के लिए $S_5$ और
$A_5$ में $\sigma$ के केंद्रकों की तुलना कीजिए।
(ख) निष्कर्ष निकालिए कि $A_5$ \omterm{def:b3:groups:simple}{सरल} है: कोई \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह}
संयुग्मन वर्गों का सम्मिलन होता है, उसमें $e$ होता है, और
उसकी गणनीयता $60$ को विभाजित करती है।
(ग) दर्शाइए कि क्रम $60$ के किसी \omterm{def:b3:groups:simple}{सरल समूह} के लिए अनिवार्यतः
$n_5 =
6$ होता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:12}

\section{समस्या: अधिकतम 15 क्रम वाले समूह}

\begin{problem}[{सप्ताहांत समस्या --- छोटे समूहों का
वर्गीकरण}]\label{pb:b3:groups:1}
लक्ष्य है तुल्याकारिता तक क्रम $\leq 15$ वाले समूहों का पूर्ण
वर्गीकरण, पूरी उपपत्तियों सहित। क्रम $1, 2, 3, 5,
7, 11, 13$ लाग्राँज से तय हो
जाते हैं (चक्रीय), और क्रम $4$ तथा $9$ \cref{thm:b3:groups:pfixed} से एवं नीचे
दिए गए $p^2$ के विश्लेषण से: शेष रह जाते हैं $6, 8, 10, 12, 14, 15$।

\medskip
\noindent\textbf{भाग I --- साधन।}
\begin{enumerate}
  \item दर्शाइए कि जिस समूह में प्रत्येक अवयव $x^2 =
        e$ को संतुष्ट
        करता है वह आबेली है; इससे निष्कर्ष निकालिए कि ऐसे परिमित
        समूह का क्रम $2^k$ होता है और वह $(\Z/2\Z)^k$ के तुल्याकारी है।
        \emph{(इसे $\mathbb F_2$ पर एक सदिश समष्टि की तरह देखिए।)}
  \item दर्शाइए कि क्रम $p^2$ का समूह $\Z/p^2\Z$ या $(\Z/p\Z)^2$ के
        तुल्याकारी होता है। तुल्याकारिता तक क्रम $8$ के आबेली
        समूहों की सूची बनाइए: $\Z/8\Z$, $\Z/4\Z \times
        \Z/2\Z$, $(\Z/2\Z)^3$ --- और \cref{ch:b3:modules}
        की संरचना प्रमेय के \emph{बिना} सिद्ध कीजिए कि यह सूची
        पूर्ण है और उसमें कोई पुनरुक्ति नहीं (किसी अवयव के महत्तम
        क्रम के अनुसार स्थितियाँ लीजिए)।
  \item मान लीजिए $\varphi, \varphi' \colon K \to \operatorname{Aut}(H)$ दो क्रियाएँ हैं। दर्शाइए कि यदि $\alpha \in \operatorname{Aut}(K)$
        के साथ $\varphi' = \varphi \circ
        \alpha$ हो, तो $H
        \rtimes_{\varphi} K \cong H \rtimes_{\varphi'} K$।
  \item $n = 3, 4, 5,
        7$ के लिए $\operatorname{Aut}(\Z/n\Z)$ स्पष्ट रूप से ज्ञात कीजिए, और $\operatorname{Aut}\bigl((\Z/2\Z)^2
        \bigr) \cong S_3$
        दर्शाइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- क्रम $2p$ ($6$, $10$, $14$)
और $pq$।}
\begin{enumerate}[resume]
  \item मान लीजिए $\abs G = 2p$, जहाँ $p$ विषम अभाज्य है। दर्शाइए कि
        $G$ में क्रम $p$ का एक \omterm{def:b3:groups:normal}{प्रसामान्य उपसमूह} $N = \langle r \rangle$ है
        और $N$ के बाहर क्रम $2$ का एक अवयव $s$।
  \item निष्कर्ष निकालिए कि $G \cong \Z/p\Z \rtimes_\varphi \Z/2\Z$, जहाँ $\varphi(1) \in \operatorname{Aut}(\Z/p\Z)$ एक स्वप्रतिलोमी
        अवयव है, और यह भी: $G \cong \Z/2p\Z$ या $G \cong
        D_p$; जाँचिए कि ये दोनों
        तुल्याकारी नहीं हैं। इससे क्रम $6$, $10$, $14$ तय
        हो जाते हैं।
  \item अधिक व्यापक रूप से, मान लीजिए $\abs G = pq$, जहाँ $p < q$ अभाज्य
        हैं। दर्शाइए कि यदि $p \nmid q-1$ हो तो $G$ चक्रीय है
        (\cref{ex:b3:groups:pq}), और यह कि यदि $p \mid q - 1$ हो तो $\Z/pq\Z$ के अतिरिक्त
        तुल्याकारिता तक \emph{ठीक एक} अनाबेली समूह $\Z/q\Z \rtimes \Z/p\Z$ होता है
        --- इसके लिए प्रश्न 3 का और इस तथ्य का उपयोग कीजिए कि
        $\operatorname{Aut}(\Z/q\Z) \cong
        (\Z/q\Z)^\times$ क्रम $q - 1$ का चक्रीय समूह है, जिसे यहाँ स्वीकार
        किया गया है और \cref{ch:b3:galois} में सिद्ध किया गया है ($\mathbb F_q^\times$ की
        चक्रीयता)। क्रम $15$ के लिए निष्कर्ष निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- क्रम $8$।} मान लीजिए $G$
क्रम $8$ का अनाबेली समूह है।
\begin{enumerate}[resume]
  \item दर्शाइए कि $G$ में क्रम $4$ का एक अवयव $r$ है
        (प्रश्न 1 का उपयोग कीजिए) और यह कि $N = \langle r\rangle$ \omterm{def:b3:groups:normal}{प्रसामान्य} है।
  \item मान लीजिए $s \notin N$। दर्शाइए कि $s^2 \in N$ (\cref{exo:b3:groups:1}(ख)), कि
        $srs^{-1} = r^{-1}$ (संयुग्मन के अंतर्गत $r$ के संभावित प्रतिबिंबों
        की जाँच कीजिए, जिनका क्रम 4 होना चाहिए, और $srs^{-1} = r$ को
        अपवर्जित कीजिए), और यह कि $s^2 \in \{e, r^2\}$ \emph{(यदि $s^2 = r$ या
        $r^3$ हो तो क्या होगा? और $s^2$ को $s$ के साथ
        क्रमविनिमेय क्यों होना चाहिए?)}।
  \item $s^2 = e$ की स्थिति में $G \cong D_4$ दर्शाइए।
  \item $s^2 = r^2$ की स्थिति में दर्शाइए कि गुणन सारणी पूर्णतः
        निर्धारित हो जाती है; प्राप्त समूह \emph{क्वाटर्नियन
        समूह}\index{क्वाटर्नियन समूह} $Q_8 =
        \{\pm 1, \pm \mathrm i, \pm \mathrm j, \pm \mathrm k\}$ है, $\mathrm i^2 = \mathrm j^2 = \mathrm k^2 = \mathrm i\,
        \mathrm j\,\mathrm k = -1$ ($r = \mathrm i$,
        $s =
        \mathrm j$ रखिए)। सत्यापित कीजिए कि $Q_8$ विद्यमान है,
        उदाहरणार्थ $GL_2(\C)$ के भीतर
        \[
        \mathrm i \mapsto \begin{pmatrix} \iu & 0\\ 0 & -\iu
        \end{pmatrix},
        \qquad
        \mathrm j \mapsto \begin{pmatrix} 0 & 1\\ -1 & 0
        \end{pmatrix}.
        \]
  \item दर्शाइए कि $Q_8$ का प्रत्येक अतुच्छ उपसमूह $-1$ को
        अंतर्विष्ट करता है; इससे निष्कर्ष निकालिए कि $Q_8$ का
        प्रत्येक उपसमूह \omterm{def:b3:groups:normal}{प्रसामान्य} है, कि $D_4 \not\cong Q_8$ (क्रम $2$ के
        अवयव गिनिए), और यह कि $Q_8$ दो उचित उपसमूहों का
        \omterm{def:b3:groups:semidirect}{अर्ध-प्रत्यक्ष गुणन} नहीं है।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- क्रम $12$।} मान लीजिए $\abs G = 12$,
$P_3
\in \mathrm{Syl}_3(G)$, $P_2 \in \mathrm{Syl}_2(G)$।
\begin{enumerate}[resume]
  \item दर्शाइए कि $n_3 \in \{1, 4\}$, $n_2 \in \{1, 3\}$, और यह कि $n_3
        = 4$ से $n_2 = 1$
        अनिवार्य हो जाता है \emph{(क्रम $3$ के अवयव गिनिए)}।
  \item मान लीजिए $n_3 = 4$। $\mathrm{Syl}_3$ पर संयुग्मन \omterm{def:b3:groups:action}{क्रिया} से $\rho \colon G \to S_4$
        मिलता है। दर्शाइए कि $\ker \rho$, जो प्रत्येक $N_G(P_3)$ में
        अंतर्विष्ट है और इसलिए जिसका क्रम $3$ को विभाजित करता
        है, तुच्छ है \emph{(उसका क्रम $3$ क्यों नहीं हो सकता?)};
        कि प्रतिबिंब, जो $S_4$ का क्रम $12$ वाला उपसमूह है,
        अनिवार्यतः $A_4$ है \emph{(सूचकांक 2 वाले उपसमूह
        \omterm{def:b3:groups:normal}{प्रसामान्य} होते हैं और सभी वर्गों को अंतर्विष्ट करते
        हैं --- \cref{exo:b3:groups:1}; $S_4$ में वर्ग गिनिए)}; और निष्कर्ष
        निकालिए कि $G \cong A_4$।
  \item मान लीजिए $n_3 = 1$, अतः $\varphi \colon P_2 \to \operatorname{Aut}(\Z/3\Z)
        \cong \Z/2\Z$ के साथ $G \cong \Z/3\Z \rtimes_\varphi P_2$। स्थितियों
        की गणना कीजिए: $\varphi$ तुच्छ होने पर $\Z/12\Z$ और $\Z/6\Z \times \Z/2\Z$
        मिलते हैं; आच्छादक $\varphi$ के साथ $P_2 =
        \Z/4\Z$ से
        \emph{द्विचक्रीय समूह} $\mathrm{Dic}_3 = \Z/3\Z \rtimes
        \Z/4\Z$ मिलता है; आच्छादक $\varphi$
        के साथ $P_2 = (\Z/2\Z)^2$ से, प्रश्न 3 की तुल्यता तक, एक ही समूह
        मिलता है --- दर्शाइए कि वह $D_6$ है, उदाहरणार्थ क्रम
        $6$ का एक अवयव और परावर्तन जैसा एक स्वप्रतिलोमी अवयव
        प्रस्तुत करके।
  \item दर्शाइए कि $\Z/12\Z$, $\Z/6\Z \times \Z/2\Z$, $D_6$, $A_4$, $\mathrm{Dic}_3$
        जोड़े-जोड़े में अतुल्याकारी हैं \emph{(क्रम $2$ के
        अवयव गिनिए, अथवा $n_3$ का उपयोग कीजिए)}। इससे क्रम
        $12$ तय हो जाता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- संश्लेषण।}
\begin{enumerate}[resume]
  \item वर्गीकरण सारणी संकलित कीजिए: प्रत्येक क्रम $n \leq
        15$ के लिए
        तुल्याकारिता तक समूहों की पूरी सूची, गणनाओं $1, 1, 1, 2, 1, 2, 1, 5, 2, 2, 1, 5, 1, 2, 1$ सहित।
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- आगे: क्रम $p^3$ के समूह, जहाँ
$p$ विषम है।} भाग III के क्रम-$8$ विश्लेषण का एक सुंदर
विषम-अभाज्य समरूप है, जिसमें एक सर्वथा नई परिघटना प्रकट होती है।
मान लीजिए $p$ विषम अभाज्य है और $G$ क्रम $p^3$ का अनाबेली
समूह।
\begin{enumerate}[resume]
  \item दर्शाइए कि $\abs{Z(G)} = p$, कि $G/Z(G) \cong
        (\Z/p\Z)^2$ \emph{(\omterm{ex:b3:groups:actions}{केंद्र} द्वारा चक्रीय
        भागफल आबेलीयता पर विवश कर देता है: \cref{exo:b3:groups:2})}, और कि
        $D(G) =
        Z(G)$ \emph{($D(G) \subseteq Z(G)$ के लिए इसका उपयोग कीजिए कि $G/Z(G)$
        आबेली है; समता के लिए, $G$ अनाबेली है और $D(G) \neq \{e\}$)}।
        निष्कर्ष निकालिए कि प्रत्येक \omterm{def:b3:groups:derived}{क्रमविनिमेयक} $[x, y] = xyx^{-1}y^{-1}$
        केंद्रीय है और उसका क्रम $p$ को विभाजित करता है।
  \item (मुख्य सर्वसमिका) मान लीजिए $x, y \in G$ और $z = [y, x]$ केंद्रीय
        है। $k$ पर आगमन द्वारा सिद्ध कीजिए:
        \[
        (xy)^k = x^k y^k z^{k(k-1)/2} .
        \]
        \emph{(प्रत्येक $y$ को प्रत्येक $x$ के पार ले
        जाइए; हर पारगमन एक केंद्रीय गुणनखंड $z$ की कीमत पर
        होता है।)}
  \item निष्कर्ष निकालिए कि विषम $p$ के लिए प्रतिचित्रण $\theta\colon x \mapsto
        x^p$
        $G$ से $Z(G)$ में जाने वाली एक समूह समाकारिता है
        \emph{($x^p$ केंद्रीय क्यों है? $z^{p(p-1)/2} = e$ के लिए $p$
        का विषम होना क्यों आवश्यक है?)}, और यह कि $G$ का
        घातांक $p$ या $p^2$ है, तथा ये दोनों स्थितियाँ इससे
        अलग होती हैं कि $\theta$ तुच्छ है या नहीं।
  \item (घातांक $p$) मान लीजिए प्रत्येक अवयव $x^p = e$ को संतुष्ट
        करता है। ऐसे $x, y$ चुनिए जिनके वर्ग $G/Z(G)$ को जनित करें
        और $z =
        [y, x]$ रखिए। दर्शाइए कि $z \neq e$, कि $G$ का प्रत्येक
        अवयव अद्वितीय रूप से $x^ay^bz^c$ है ($0 \leq a, b, c < p$), और यह कि गुणन
        पूर्णतः संबंधों $x^p = y^p = z^p = e$, $z$ केंद्रीय, $[y, x] =
        z$ से
        निर्धारित हो जाता है। सत्यापित कीजिए कि \emph{हाइजेनबर्ग
        समूह}
        \[
        H_p = \left\{
        \begin{pmatrix} 1 & a & c\\ 0 & 1 & b\\ 0 & 0 & 1
        \end{pmatrix} : a, b, c \in \mathbb F_p \right\}
        \subseteq GL_3(\mathbb F_p)
        \]
        इन संबंधों को साकार करता है और उसका घातांक $p$ है
        ($N^3 = 0$ और $p \geq 3$ का उपयोग करते हुए कड़ाई से ऊपरी
        त्रिभुजाकार $N$ के साथ $(I + N)^p$ परिकलित कीजिए): क्रम
        $p^3$ का प्रत्येक घातांक-$p$ अनाबेली समूह $H_p$ के
        तुल्याकारी है।
  \item (घातांक $p^2$) मान लीजिए किसी $r \in G$ का क्रम $p^2$ है,
        और $N = \langle r\rangle$ रखिए, जो \omterm{def:b3:groups:normal}{प्रसामान्य} है (सूचकांक $p$:
        \cref{exo:b3:groups:10})। दर्शाइए कि $s^p = e$ वाला कोई $s \notin N$ विद्यमान है
        \emph{(कोई भी $t \notin N$ लीजिए; प्रश्न 20 का उपयोग करके उसे
        सुधारिए: $\theta(t) = t^p \in Z(G) \subseteq N$ --- $Z(G) = \langle r^p\rangle$ को उचित ठहराइए --- और $s^p = e$
        को संतुष्ट करने वाला $s = tr^{a}$ लेकर $a$ चुनिए; $p$ का
        विषम होना कहाँ प्रयुक्त हुआ?)}। दर्शाइए कि $s$ को
        किसी घात से बदलने तक $srs^{-1} = r^{1+p}$, और निष्कर्ष निकालिए: क्रम
        $p^3$ और घातांक $p^2$ का ठीक \emph{एक} अनाबेली समूह
        है, अर्थात् $\varphi(1)\colon r \mapsto r^{1+p}$ के साथ $\Z/p^2\Z \rtimes_\varphi \Z/p\Z$ \emph{(प्रश्न 3 का
        उपयोग कीजिए; $\operatorname{Aut}(\Z/p^2\Z)$ क्रम $p(p-1)$ का चक्रीय समूह है, जिसे
        यहाँ स्वीकार किया गया है, अतः उसका क्रम $p$ का
        अद्वितीय उपसमूह है)}।
  \item गणना का निष्कर्ष निकालिए: विषम $p$ के लिए क्रम $p^3$
        के ठीक $5$ समूह हैं (तीन आबेली, दो अनाबेली), ठीक वैसे
        ही जैसे $p = 2$ के लिए --- परंतु दोनों अनाबेली समूह अब
        $D_4$ और $Q_8$ नहीं रहे। ठीक-ठीक बताइए कि विषम-$p$
        तर्क $p = 2$ के लिए कहाँ टूट जाता है: प्रश्न 19 की
        सर्वसमिका में $k = p = 2$ के लिए $z^{k(k-1)/2}$ $z^1 \neq
        e$ है, अतः वर्ग
        करना समाकारिता नहीं रह जाता --- और वस्तुतः $Q_8$ में
        क्रम $2$ का अद्वितीय अवयव है, जबकि घातांक-$4$
        $D_4$ में ऐसे पाँच अवयव हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग VII --- पूरक।}
\begin{enumerate}[resume]
  \item विषम $p$ के लिए क्रम $p^3$ के दोनों अनाबेली समूहों
        में क्रम $p$ के अवयव गिनिए: दर्शाइए कि $H_p$ में ठीक
        $p^3 - 1$ अवयव हैं, जबकि $M_p =
        \Z/p^2\Z \rtimes \Z/p\Z$ में ठीक $p^2 - 1$
        \emph{(प्रश्न 20 की समाकारिता $\theta$ का उपयोग कीजिए:
        पहले उसका प्रतिबिंब पहचानिए, फिर उसकी अष्टि का क्रम)}।
        $p = 3$ के लिए संख्यात्मक सत्यापन कीजिए: $26$ बनाम
        $8$। समझाइए कि इस प्रकार का कोई भी वर्ग-समाकारिता
        संबंधी तर्क $D_4$ को $Q_8$ से अलग क्यों नहीं कर सकता,
        और कौन-सी गणना उन्हें अलग करती है।
  \item किसी पूर्णांक $n \geq 1$ को \emph{चक्रीय} कहिए यदि क्रम
        $n$ का प्रत्येक समूह चक्रीय हो। दर्शाइए कि यदि किसी
        अभाज्य $p$ के लिए $p^2 \mid n$ हो, अथवा यदि $n$ के ऐसे
        अभाज्य भाजक $p < q$ हों जिनके लिए $p \mid q - 1$, तो $n$
        चक्रीय नहीं है \emph{(दोनों स्थितियों में क्रम $n$
        का एक अचक्रीय समूह प्रस्तुत कीजिए, दूसरी के लिए भाग II
        का उपयोग करते हुए)}। निष्कर्ष निकालिए कि $n$ के
        चक्रीय होने से $\gcd(n, \varphi(n)) = 1$ अनिवार्य हो जाता है, जहाँ $\varphi$
        ऑयलर का टोशेंट है; और प्रश्न 17 की सारणी से मिलान
        कीजिए: $n \leq 15$ में से एकमात्र समूह धारण करने वाले क्रम
        ठीक $n \in \{1, 2, 3, 5, 7, 11, 13, 15\}$ हैं, अर्थात् ठीक वे जिनके लिए $\gcd(n, \varphi(n)) = 1$।
\end{enumerate}
\end{problem}
