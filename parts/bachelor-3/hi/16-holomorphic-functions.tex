\chapter{समविश्लेषिक फलन}\label{ch:b3:holomorphic}

सम्मिश्र अवकलनीयता देखने में वास्तविक सिद्धांत का एक छोटा-सा
रूपांतर लगती है --- एक सीमा, एक भागफल। पर वह वस्तुतः एक भिन्न
ब्रह्मांड है। $\C$ के किसी विवृत उपसमुच्चय पर \emph{एक बार}
अवकलनीय फलन स्वतः अनंत बार अवकलनीय, वैश्लेषिक, किसी एक बिंदु के
निकट के अपने मानों से पूरे प्रांत पर निर्धारित, और कठोर वैश्विक
सिद्धांतों (ल्यूविल, उच्चतम मापांक) से बँधा होता है। यह सब एक ही
चमत्कार से बहता है, कोशी की प्रमेय: किसी तारकाकार प्रांत में
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक फलन} का किसी संवृत \omterm{def:b3:topology:pathconnected}{पथ} के अनुदिश समाकल लुप्त हो जाता है।
यह अध्याय उस चमत्कार को सिद्ध करता है (गुरसा का तर्क, जिसमें
अवकलनीयता से आगे कोई नियमितता नहीं मानी जाती), उसके शास्त्रीय
परिणाम बटोरता है, और उस प्रमेय का निपटारा कर देता है जिसे यह
पुस्तक \cref{ch:b3:galois} से उधार पर बरत रही है: प्रत्येक अचरेतर सम्मिश्र बहुपद
का कोई मूल होता है।

सर्वत्र $\Omega \subseteq \C$ विवृत है, और $D(a, r)$ विवृत चक्र को निरूपित करता है।

\section{सम्मिश्र अवकलनीयता}

\begin{definition}\label{def:b3:holomorphic:holo}
$f \colon \Omega \to \C$ $\Omega$ पर \emph{समविश्लेषिक}\index{समविश्लेषिक फलन} कहलाता है
यदि प्रत्येक $z_0 \in \Omega$ के लिए
\[
f'(z_0) = \lim_{h\to0}\frac{f(z_0 + h) - f(z_0)}{h}
\]
विद्यमान हो ($h \in \C^*$)। समविश्लेषिक फलनों के योग, गुणनफल, भागफल
(हर शून्येतर) और संयोजन समविश्लेषिक होते हैं, और वही सामान्य
सूत्र लागू होते हैं (पहले और दूसरे वर्ष की उपपत्तियाँ शब्दशः वैसी
ही हैं: वे केवल क्षेत्र संक्रियाओं और सीमाओं का उपयोग करती हैं)।
$\mathcal H(\Omega)$ $\Omega$ पर समविश्लेषिक फलनों का समुच्चय निरूपित करता है।
\end{definition}

\begin{proposition}[कोशी--रीमान]
\label{prop:b3:holomorphic:cauchyriemann}
$f(x + \iu y) = P(x,y) + \iu Q(x,y)$ लिखिए। तब $f$ $z_0$ पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है तभी और केवल तभी जब
$f$ $z_0$ पर $\R$-अवकलनीय हो (दो वास्तविक चरों के प्रतिचित्रण के
रूप में) और\index{कोशी--रीमान समीकरण}
\[
\frac{\partial P}{\partial x} = \frac{\partial Q}{\partial y},
\qquad
\frac{\partial P}{\partial y} = -\frac{\partial Q}{\partial x}
\qquad z_0 \text{ पर} ;
\]
समतुल्य रूप से, जब वास्तविक अवकलज सम्मिश्र संख्या $f'(z_0)$ से गुणन हो।
\end{proposition}

\begin{proof}
$\C$-अवकलनीयता कहती है कि $c = f'(z_0)$ के साथ $f(z_0 + h) = f(z_0) + ch +
o(\abs h)$: अर्थात् एक
$\R$-रैखिक अवकलज जो $c = a + \iu b$ से गुणन है, यानी जिसका आधार $(1, \iu)$ में
आव्यूह $\bigl(\begin{smallmatrix} a & -b\\ b & a
\end{smallmatrix}\bigr)$ है --- और यही आंशिक अवकलजों के लिए प्रदर्शित संबंध
हैं। विलोमतः ऐसा अवकलज $\C$-रैखिक होता है, और दोनों $o(\abs h)$
परिभाषाएँ मेल खा जाती हैं।
\end{proof}

\begin{example}\label{ex:b3:holomorphic:examples}
$z$ में बहुपद, अपने ध्रुवों के बाहर परिमेय फलन, और --- दूसरे वर्ष
की घात श्रेणियों के पदशः अवकलन प्रमेय से, जिसकी उपपत्ति $\C$ पर
भी वैसी ही काम करती है --- अपने अभिसरण चक्र के भीतर प्रत्येक घात
श्रेणी $\sum a_n(z - a)^n$ का योग: ये सब \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} हैं, जिनका अवकलज $\sum na_n(z -
a)^{n-1}$ है
(वही त्रिज्या)। विशेष रूप से $\exp z = \sum z^n/n!$ समग्र है ($\C$ पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक})
और $\exp' = \exp$। दूसरी ओर $z \mapsto \bar z$, $\abs z$, $\operatorname{Re}z$ कहीं भी \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} नहीं हैं
(कोशी--रीमान सर्वत्र विफल): \omterm{def:b3:holomorphic:holo}{समविश्लेषिकता} अभिविन्यास-और-कोण
सुरक्षित रखने वाली कठोरता है, चिकनाई नहीं।
\end{example}

\section{परिरेखा समाकल}

\begin{definition}\label{def:b3:holomorphic:contour}
\emph{\omterm{def:b3:topology:pathconnected}{पथ}} खंडशः $\mathcal C^1$ प्रतिचित्रण $\gamma \colon
\intcc ab \to \C$ है; वह \emph{संवृत} कहलाता
है यदि $\gamma(a) =
\gamma(b)$ हो। $\gamma$ के प्रतिबिंब पर \omterm{def:b3:topology:continuity}{संतत} $f$ के
लिए:\index{परिरेखा समाकल}
\[
\int_\gamma f(z)\,\dd z =
\int_a^b f(\gamma(t))\,\gamma'(t)\,\dd t,
\qquad
\Bigl|\int_\gamma f\,\dd z\Bigr| \leq
\sup_{\gamma}\abs f\cdot\operatorname{length}(\gamma)
\]
(\emph{ML असमिका}; लंबाई $= \int_a^b\abs{\gamma'}$)। यह समाकल बढ़ते हुए $\mathcal C^1$
पुनःप्राचलन के अंतर्गत अपरिवर्ती रहता है और अभिविन्यास उलटने पर
चिह्न बदल देता है।
\end{definition}

\begin{proposition}[आद्यंतर]\label{prop:b3:holomorphic:primitive}
$\Omega$ पर \omterm{def:b3:topology:continuity}{संतत} $f$ के लिए निम्नलिखित समतुल्य हैं: (क) $f$ का कोई
\emph{आद्यंतर} $F \in \mathcal H(\Omega)$ है ($F'
= f$); (ख) $\Omega$ में प्रत्येक संवृत \omterm{def:b3:topology:pathconnected}{पथ}
$\gamma$ के लिए $\int_\gamma f\,\dd z = 0$। उस स्थिति में प्रत्येक \omterm{def:b3:topology:pathconnected}{पथ} के लिए $\int_\gamma f\,\dd z =
F(\gamma(b)) - F(\gamma(a))$।
\end{proposition}

\begin{proof}
(क)$\Rightarrow$: $\frac{\dd}{\dd t}F(\gamma(t)) =
F'(\gamma(t))\gamma'(t)$ (शृंखला नियम, खंडशः वैध), अतः समाकल दूरबीनी होकर
सिरों के अंतर तक सिमट जाता है; और संवृत \omterm{def:b3:topology:pathconnected}{पथ} $0$ देते हैं।
(ख)$\Rightarrow$(क): किसी \omterm{def:b3:topology:components}{संबद्ध घटक} में $z_*$ स्थिर कीजिए, और $z_*$ से
$z$ तक किसी भी \omterm{def:b3:topology:pathconnected}{पथ} के अनुदिश $F(z) = \int_{\gamma_z}f\,\dd z$ परिभाषित कीजिए (यह सुपरिभाषित
है: दो \omterm{def:b3:topology:pathconnected}{पथ} किसी संवृत \omterm{def:b3:topology:pathconnected}{पथ} जितने ही भिन्न होते हैं); छोटे $h$ के
लिए $z$ से $z + h$ तक का रेखाखंड लेने पर,
\[
\frac{F(z + h) - F(z)}{h} - f(z)
= \frac1h\int_{[z, z+h]}\bigl(f(w) - f(z)\bigr)\dd w
\longrightarrow 0
\]
और यह ML असमिका तथा $z$ पर $f$ के \omterm{def:b3:topology:continuity}{सांतत्य} से।
\end{proof}

\begin{definition}[घूर्णन संख्या]\label{def:b3:holomorphic:index}
किसी संवृत \omterm{def:b3:topology:pathconnected}{पथ} $\gamma$ और $z \notin
\operatorname{im}\gamma$ के लिए \emph{सूचकांक}\index{घूर्णन
संख्या}\index{पथ का सूचकांक} है
\[
\operatorname{Ind}_\gamma(z) = \frac1{2\iu\pi}
\int_\gamma\frac{\dd w}{w - z} .
\]
यह एक \emph{पूर्णांक} है: $\varphi(t) =
\int_a^t\frac{\gamma'(s)}{\gamma(s) - z}\dd s$ रखने पर फलन $(\gamma(t) - z)\eu^{-\varphi(t)}$ का अवकलज (खंडशः)
शून्य होता है, अतः वह अचर है; और $t = b$ पर $\eu^{\varphi(b)} =
\frac{\gamma(b) - z}{\gamma(a) - z} = 1$, इसलिए $\varphi(b) \in
2\iu\pi\Z$। $z$
के फलन के रूप में सूचकांक $\C\setminus\operatorname{im}\gamma$ पर \omterm{def:b3:topology:continuity}{संतत} है (प्रभावी अभिसरण), अतः
प्रत्येक \omterm{def:b3:topology:components}{संबद्ध घटक} पर अचर, और अपरिबद्ध घटक पर $0$ (ML: $z \to
\infty$ पर
समाकल $0$ की ओर जाता है)। वृत्त $\gamma(t) = a + r\eu^{\iu t}$, $t
\in \intcc0{2\pi}$ के लिए: $z
\in D(a,r)$ के लिए
$\operatorname{Ind}_\gamma(z) = 1$ ($z = a$ पर परिकलित कीजिए: $\frac1{2\iu\pi}
\int_0^{2\pi}\frac{r\iu\eu^{\iu t}}{r\eu^{\iu t}}\dd t = 1$; शेष अचरता कर देती है)।
\end{definition}

\section{कोशी की प्रमेय}

\begin{theorem}[गुरसा]\label{thm:b3:holomorphic:goursat}
मान लीजिए $f \in \mathcal H(\Omega)$ और $T \subseteq \Omega$ कोई संवृत ठोस त्रिभुज है। तब $\int_{\partial T}f\,\dd z = 0$
(परिसीमा एक बार, किसी भी अभिविन्यास में)।\index{गुरसा की प्रमेय}
\end{theorem}

\begin{proof}
मान लीजिए $I(T) = \int_{\partial T}f\,\dd z$। भुजाओं के मध्य बिंदु जोड़ने से $T$ आधे आकार के
चार त्रिभुजों $T^{(1)},
\dots, T^{(4)}$ में बँट जाता है, और भीतरी किनारे जोड़ों में
निरस्त हो जाते हैं: $I(T) =
\sum_iI(T^{(i)})$। उनमें से $\abs{I(T_1)} \geq \frac14\abs{I(T)}$ वाला $T_1$ चुनिए, और
दोहराइए: इससे एक नेस्टेड अनुक्रम $T \supseteq T_1 \supseteq T_2 \supseteq\cdots$ मिलता है जिसमें
\[
\abs{I(T_n)} \geq 4^{-n}\abs{I(T)},
\qquad
\operatorname{diam}T_n = 2^{-n}\operatorname{diam}T,
\quad
\operatorname{length}(\partial T_n) =
2^{-n}\operatorname{length}(\partial T).
\]
प्रतिच्छेदन $\bigcap T_n$ एक ही बिंदु $z_0$ है (लुप्त होते व्यास वाले
नेस्टेड \omterm{def:b3:topology:compact}{संहत}, \cref{thm:b3:topology:compactprops}(3))। $z_0$ पर अवकलनीयता: $\varepsilon$ दिया हो, तो
बड़े $n$ के लिए $T_n$ पर
\[
f(z) = f(z_0) + f'(z_0)(z - z_0) + R(z),
\qquad \abs{R(z)} \leq \varepsilon\abs{z - z_0} \leq
\varepsilon\operatorname{diam}T_n .
\]
आफ़ीन भाग का कोई आद्यंतर है: संवृत $\partial T_n$ पर उसका समाकल लुप्त हो
जाता है (\cref{prop:b3:holomorphic:primitive}), जिससे
\[
\abs{I(T_n)} = \Bigl|\int_{\partial T_n}R\Bigr|
\leq \varepsilon\operatorname{diam}(T_n)\,
\operatorname{length}(\partial T_n)
= \varepsilon\,4^{-n}\operatorname{diam}(T)
\operatorname{length}(\partial T) .
\]
बचता है। $\abs{I(T_n)} \geq 4^{-n}\abs{I(T)}$ से तुलना करने पर: प्रत्येक $\varepsilon$ के लिए $\abs{I(T)} \leq \varepsilon\cdot\text{अचर}$:
अतः $I(T) = 0$।
\end{proof}

\begin{theorem}[कोशी की प्रमेय, तारकाकार रूप]
\label{thm:b3:holomorphic:cauchy}
मान लीजिए $\Omega$ $c$ के परितः \emph{तारकाकार} है (प्रत्येक
रेखाखंड $[c, z]$, $z \in \Omega$, $\Omega$ में है) --- उदाहरणार्थ उत्तल।
प्रत्येक $f \in \mathcal H(\Omega)$ का $\Omega$ पर कोई आद्यंतर होता है; फलस्वरूप $\Omega$
में \emph{प्रत्येक} संवृत \omterm{def:b3:topology:pathconnected}{पथ} $\gamma$ के लिए $\int_\gamma f\,\dd z = 0$।\index{कोशी की
प्रमेय}
\end{theorem}

\begin{proof}
$F(z) = \int_{[c,z]}f\,\dd w$ परिभाषित कीजिए। $[z, z+h] \subseteq \Omega$ वाले $z, z + h \in
\Omega$ के लिए (जो छोटे $h$ के
लिए सत्य है) शीर्षों $c, z, z+h$ वाला त्रिभुज $\Omega$ में होता है
(तारकाकारता: उसका प्रत्येक बिंदु $w \in [z, z+h] \subseteq \Omega$ वाले किसी रेखाखंड $[c,
w]$
पर होता है): अतः गुरसा
\[
F(z + h) - F(z) = \int_{[z, z+h]}f\,\dd w,
\]
दे देता है, और \cref{prop:b3:holomorphic:primitive} का अंतर-भागफल परिकलन $F' = f$ दे देता है।
संवृत-पथ समाकलों का लुप्त होना उसी प्रतिज्ञप्ति से निकलता है।
\end{proof}

\begin{theorem}[कोशी का समाकल सूत्र]
\label{thm:b3:holomorphic:formula}
मान लीजिए $f \in \mathcal H(\Omega)$, $\bar D(a, r) \subseteq
\Omega$, और $C_r$ वह वृत्त $\partial D(a,r)$ है जो एक बार
वामावर्त चलाया गया है। तब प्रत्येक $z \in D(a, r)$ के लिए:\index{कोशी
समाकल सूत्र}
\[
f(z) = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - z}\,\dd w .
\]
\end{theorem}

\begin{proof}
$z$ स्थिर कीजिए और $\Omega$ पर परिभाषित कीजिए
\[
g(w) = \begin{cases}
\dfrac{f(w) - f(z)}{w - z} & w \neq z,\\[4pt]
f'(z) & w = z :
\end{cases}
\]
$g$ $\Omega$ पर \omterm{def:b3:topology:continuity}{संतत} है और $z$ के बाहर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक}। $g$ के लिए
गुरसा प्रत्येक त्रिभुज $T \subseteq \Omega'$ पर लागू होता है, जहाँ $\Omega'$ $\Omega$ के
भीतर $\bar D(a,r)$ से थोड़ा बड़ा चक्र है: यदि $z \notin T$ हो तो सीधे; और यदि
$z \in T$ हो, तो $T$ को $z$ को शीर्ष रखने वाले छोटे त्रिभुजों तथा
$z$ से बचने वाले त्रिभुजों में बाँटिए; शीर्ष $z$ वाले त्रिभुज
पर त्रिभुज के सिकुड़ने के साथ ML परिबंध $\abs{\int_{\partial T'}g} \leq \sup_{T'}\abs
g\cdot\operatorname{length} \to 0$ दे देता है, और शेष
टुकड़े गुरसा से लुप्त हो जाते हैं --- अतः सभी स्थितियों में $\int_{\partial T}g = 0$।
\cref{thm:b3:holomorphic:cauchy} की उपपत्ति ने केवल इसी त्रिभुज गुण का उपयोग किया था: अतः
$g$ का उत्तल $\Omega'$ पर कोई आद्यंतर है, इसलिए $\int_{C_r}g = 0$, अर्थात्
\[
\frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - z}\dd w
= f(z)\,\frac{1}{2\iu\pi}\int_{C_r}\frac{\dd w}{w - z}
= f(z)\operatorname{Ind}_{C_r}(z) = f(z) .
\]
\end{proof}

\section{वैश्लेषिकता और उसका झरना}

\begin{theorem}[समविश्लेषिक $=$ वैश्लेषिक]
\label{thm:b3:holomorphic:analytic}
मान लीजिए $f \in \mathcal H(\Omega)$ और $D(a, R) \subseteq \Omega$। तब\index{समविश्लेषिक फलनों की
वैश्लेषिकता}
\[
f(z) = \sum_{n\geq0}c_n\,(z - a)^n \quad D(a, R) \text{ पर},
\qquad
c_n = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{(w -
a)^{n+1}}\,\dd w \ \ (0 < r < R),
\]
और गुणांक $r$ पर निर्भर नहीं करते। फलस्वरूप $f$ अनंत बार
$\C$-अवकलनीय है, $c_n = f^{(n)}(a)/n!$, और \emph{कोशी आकलन}\index{कोशी आकलन}
लागू होते हैं:
\[
\abs{c_n} \;\leq\; \frac{\sup_{\abs{w - a} =
r}\abs{f(w)}}{r^{n}} .
\]
\end{theorem}

\begin{proof}
$\abs{z - a} < r$ के लिए: कोशी नाभिक का गुणोत्तर श्रेणी में प्रसार कीजिए
\[
\frac1{w - z} = \frac1{(w - a)\bigl(1 - \frac{z - a}{w -
a}\bigr)} = \sum_{n\geq0}\frac{(z - a)^n}{(w - a)^{n+1}},
\]
जो $C_r$ पर $w$ में प्रसामान्य रूप से अभिसरित होती है ($\abs{\frac{z-a}{w-a}} =
\frac{\abs{z-a}}r < 1$):
$\frac{f(w)}{2\iu\pi}$ के सापेक्ष पदशः समाकलन कीजिए (एकसमान अभिसरण अदला-बदली को
उचित ठहराता है) और \cref{thm:b3:holomorphic:formula} लगाइए। कोई घात श्रेणी अनंत बार अवकलनीय
होती है, जिसमें $c_n =
f^{(n)}(a)/n!$ (दूसरा वर्ष), और इससे यह भी दिखता है कि
$c_n$ $r$ पर निर्भर नहीं करते। आकलन: गुणांक समाकल को ML से
परिबद्ध कीजिए।
\end{proof}

\begin{example}[एकलताएँ त्रिज्याएँ तय करती हैं]
\label{ex:b3:holomorphic:radius}
निर्दोष-सा वास्तविक फलन $\frac1{1 + x^2}$ की $x = 3$ पर टेलर श्रेणी केवल $\abs{x - 3} <
\sqrt{10}$
के लिए क्यों अभिसरित होती है, जबकि वास्तविक रेखा पर कहीं कुछ गलत
नहीं होता? क्योंकि ऊपर की प्रमेय $a$ पर अभिसरण त्रिज्या को
$a$ से उस निकटतम बिंदु की दूरी के बराबर कर देती है जहाँ
\omterm{def:b3:holomorphic:holo}{समविश्लेषिकता} विफल होती है। यहाँ $f(z) = \frac1{1 + z^2}$ ठीक $\C\setminus\{\pm\iu\}$ पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक}
है, अतः $a = 3$ पर प्रसार $\pm\iu$ से बचने वाले सबसे बड़े चक्र पर
अभिसरित होता है, जिसकी त्रिज्या $\abs{3 - \iu} = \sqrt{10}$ है --- और उससे बड़े पर
अभिसरित नहीं हो सकता, क्योंकि तब योग $f$ को $\pm\iu$ के किसी
\omterm{def:b3:topology:topology}{प्रतिवेश} तक \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} रूप से बढ़ा देता, जहाँ $\abs f \to \infty$। वास्तविक
सिद्धांत रहस्यमय त्रिज्या $\sqrt{10}$ देखता है; सम्मिश्र समतल दो ध्रुव
देखता है। व्यावहारिक नियम यही है: \emph{अभिसरण त्रिज्या ढूँढ़ने के
लिए एकलताएँ ढूँढ़िए} --- उदाहरणार्थ $0$ पर $\tan$ की टेलर श्रेणी
की त्रिज्या $\frac\pi2$ है ($\cos$ के निकटतम शून्य), और बर्नूली जनक फलन
$\frac z{\eu^z - 1}$ (\cref{pb:b3:holomorphic:1}, भाग VI) की त्रिज्या $2\pi$ है ($\eu^z - 1$ के निकटतम
शून्येतर शून्य: $\pm2\iu\pi$)।
\end{example}

\begin{corollary}[ल्यूविल; दालांबेर--गाउस]
\label{cor:b3:holomorphic:liouville}
कोई परिबद्ध समग्र फलन अचर होता है। फलस्वरूप $\C$ पर प्रत्येक
अचरेतर बहुपद का कोई मूल होता है: \emph{$\C$ \omterm{def:b3:galois:closure}{बीजीय रूप से संवृत}
है}।\index{ल्यूविल की प्रमेय}\index{बीजगणित की मूल प्रमेय}
\end{corollary}

\begin{proof}
यदि $\C$ पर $\abs f \leq M$ हो: प्रत्येक $a$ और $r$ के लिए $\abs{c_1(a)}
= \abs{f'(a)} \leq M/r \to 0$:
अतः $f' \equiv 0$, और $f$ अचर है (\omterm{def:b3:topology:connected}{संबद्ध} $\C$ पर: शून्य अवकलज से
स्थानीय अचरता निकलती है --- रेखाखंडों के अनुदिश समाकलन कीजिए)।
यदि $P$ का कोई मूल न होता, तो $1/P$ समग्र और परिबद्ध होता
($\abs z \to \infty$ पर $\abs{P(z)} \to
\infty$: अग्र पद प्रभावी होता है, अतः $\abs{1/P}$ किसी बड़े
चक्र के बाहर छोटा है और \omterm{def:b3:topology:compact}{संहत} चक्र पर \omterm{def:b3:topology:continuity}{संतत}): अतः अचर --- जो
अचरेतर $P$ के लिए निरर्थक है। (सप्ताहांत समस्या एक दूसरी,
प्रारंभिक उपपत्ति और \omterm{def:b3:galois:algebraic}{बीजीय} परिणाम देती है।)
\end{proof}

\begin{theorem}[शून्य विविक्त होते हैं; तत्समता प्रमेय]
\label{thm:b3:holomorphic:identity}
मान लीजिए $\Omega$ \emph{\omterm{def:b3:topology:connected}{संबद्ध}} है और $f \in \mathcal
H(\Omega)$, $f \not\equiv 0$। तब $f$ के
प्रत्येक शून्य $a$ का \emph{क्रम} परिमित होता है: $g \in
\mathcal H(\Omega)$, $g(a) \neq 0$
के साथ $f(z) = (z - a)^m\,g(z)$, और $f$ के शून्यों का $\Omega$ में कोई संचय बिंदु
नहीं होता। फलस्वरूप, यदि $\Omega$ पर दो \omterm{def:b3:holomorphic:holo}{समविश्लेषिक फलन} किसी ऐसे
समुच्चय पर सहमत हों जिसका $\Omega$ में कोई संचय बिंदु हो, तो वे
सर्वत्र सहमत होते हैं।\index{तत्समता प्रमेय}\index{विविक्त शून्य}
\end{theorem}

\begin{proof}
मान लीजिए $Z$ उन बिंदुओं का समुच्चय है जहाँ $f$ के सभी अवकलज
लुप्त हैं। $Z$ संवृत है (संवृत समुच्चयों का प्रतिच्छेदन) और
विवृत भी: यदि $a$ पर सभी $c_n = 0$ हों, तो घात श्रेणी प्रसार $a$
के चारों ओर किसी चक्र पर $f
\equiv 0$ बना देता है। \omterm{def:b3:topology:connected}{संबद्धता}: $Z =
\varnothing$ या
$Z = \Omega$; और दूसरा $f
\not\equiv 0$ से अपवर्जित है। अतः किसी शून्य $a$ पर कोई
गुणांक शून्येतर होता है: $c_m \neq 0$ वाला न्यूनतम $m$ लीजिए; तब किसी
चक्र पर $f(z) = (z -
a)^m\sum_{n\geq m}c_n(z-a)^{n-m}$, और उसका योग $a$ के निकट $g(a) = c_m \neq 0$ वाला \omterm{def:b3:holomorphic:holo}{समविश्लेषिक}
$g$ परिभाषित करता है; $g = f/(z-a)^m$ को $a$ के बाहर बढ़ा दीजिए (वहाँ
वह \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है)। चूँकि $g(a) \neq 0$ और $g$ \omterm{def:b3:topology:continuity}{संतत} है, $a$ के किसी
\omterm{def:b3:topology:topology}{प्रतिवेश} में $f$ का कोई दूसरा शून्य नहीं होता: अतः शून्य विविक्त
हैं, और विविक्त बिंदुओं के समुच्चय का $\Omega$ में कोई संचय बिंदु
नहीं होता (शून्यों का कोई संचय बिंदु \omterm{def:b3:topology:continuity}{सांतत्य} से स्वयं शून्य होता,
और विविक्त न होता)। तत्समता: अंतर पर लगाइए, जिसके शून्य समुच्चय
का कोई संचय बिंदु है, अतः वह $Z = \Omega$ शाखा में चला जाता है।
\end{proof}

\begin{theorem}[माध्य मान और उच्चतम मापांक]
\label{thm:b3:holomorphic:maximum}
मान लीजिए $f \in \mathcal H(\Omega)$।
\begin{enumerate}
  \item (माध्य मान) $\bar D(a, r) \subseteq \Omega$ के लिए: $f(a) = \frac1{2\pi}\int_0^{2\pi}f(a +
        r\eu^{\iu t})\,\dd t$।
  \item (उच्चतम सिद्धांत) यदि $\Omega$ \omterm{def:b3:topology:connected}{संबद्ध} हो और $\abs f$ $\Omega$ के
        किसी बिंदु पर कोई स्थानीय अधिकतम प्राप्त करे, तो $f$
        अचर है। फलस्वरूप, परिबद्ध $\Omega$ और $\bar\Omega$ पर \omterm{def:b3:topology:continuity}{संतत} $f$ के
        लिए: $\sup_{\bar\Omega}\abs f =
        \sup_{\partial\Omega}\abs f$।\index{उच्चतम सिद्धांत}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) \omterm{ex:b3:groups:actions}{केंद्र} पर कोशी का सूत्र है: $C_r$ का प्राचलन कीजिए।
(2) मान लीजिए $\bar D(a, \rho)
\subseteq \Omega$ पर $\abs f \leq \abs{f(a)}$। यदि $f(a) = 0$ हो, तो $a$ के निकट
$f \equiv 0$। अन्यथा, $0 < r \leq \rho$ के लिए माध्य मान
\[
\abs{f(a)} \leq \frac1{2\pi}\int_0^{2\pi}\abs{f(a +
r\eu^{\iu t})}\,\dd t \leq \abs{f(a)} :
\]
देता है; \omterm{def:b3:topology:continuity}{संतत} अऋणात्मक समाकल्य $\abs{f(a)} - \abs{f(a +
r\eu^{\iu t})}$ का माध्य शून्य है, अतः वह
लुप्त हो जाता है: इसलिए चक्र पर $\abs f$ \emph{अचर} $= \abs{f(a)} \ne 0$ है। किसी
चक्र पर अचर शून्येतर मापांक वाला \omterm{def:b3:holomorphic:holo}{समविश्लेषिक फलन} अचर होता है:
$P^2 + Q^2 = \text{अचर}$ का अवकलन $PP_x + QQ_x
= 0$ और $PP_y + QQ_y = 0$ देता है; दूसरे समीकरण में
कोशी--रीमान संबंध $P_y = -Q_x$, $Q_y = P_x$ प्रतिस्थापित करने पर रैखिक निकाय
\[
P\,P_x + Q\,Q_x = 0, \qquad -P\,Q_x + Q\,P_x = 0,
\]
मिलता है, जिसका सारणिक $P^2 + Q^2 \neq 0$ है: अतः $P_x = Q_x = 0$, इसलिए चक्र पर $f' = P_x + \iu Q_x = 0$:
यानी वहाँ $f$ अचर। तत्समता प्रमेय अचरता को पूरे $\Omega$ पर फैला
देती है। सीमांत रूप: $\abs f$ अपना उच्चतम \omterm{def:b3:topology:compact}{संहत} $\bar\Omega$ पर प्राप्त
करता है; और कोई आंतरिक अधिकतम $f$ को अचर बना देता है, अतः हर
स्थिति में उच्चतम परिसीमा पर प्राप्त होता है।
\end{proof}

\begin{theorem}[वाइरश्ट्रास अभिसरण प्रमेय]
\label{thm:b3:holomorphic:weierstrassconv}
यदि $f_n \in \mathcal H(\Omega)$ $\Omega$ के प्रत्येक \omterm{def:b3:topology:compact}{संहत} उपसमुच्चय पर एकसमान रूप से
$f$ पर अभिसरित हों, तो $f \in \mathcal
H(\Omega)$ और प्रत्येक $k$ के लिए संहतों पर
एकसमान रूप से $f_n^{(k)} \to f^{(k)}$।\index{वाइरश्ट्रास अभिसरण प्रमेय}
\end{theorem}

\begin{proof}
$f$ \omterm{def:b3:topology:continuity}{संतत} है। किसी भी संवृत त्रिभुज $T \subseteq
\Omega$ के लिए: $\int_{\partial T}f = \lim\int_{\partial T}f_n = 0$ (\omterm{def:b3:topology:compact}{संहत}
$\partial T$ पर एकसमान अभिसरण; $f_n$ के लिए गुरसा)। \cref{thm:b3:holomorphic:cauchy} के तर्क से
$f$ के स्थानीय आद्यंतर हैं (चक्र उत्तल हैं; केवल त्रिभुज गुण का
उपयोग हुआ था), अर्थात् \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} $F$ के साथ स्थानीय रूप से
$f = F'$; और $F$ वैश्लेषिक है (\cref{thm:b3:holomorphic:analytic}), अतः $f = F'$ भी: यानी
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक}। (यही \emph{मोरेरा की प्रमेय}\index{मोरेरा की प्रमेय}
है: \omterm{def:b3:topology:continuity}{संतत} और त्रिभुज समाकलों का लुप्त होना \omterm{def:b3:holomorphic:holo}{समविश्लेषिकता} दे देता
है।) अवकलज: $\bar D(a, 2r) \subseteq \Omega$ और $z \in \bar D(a, r)$ के लिए अवकलजों का कोशी सूत्र (समाकल के
भीतर \cref{thm:b3:holomorphic:formula} का अवकलन कीजिए, अथवा गुणांक सूत्र का उपयोग कीजिए)
\[
\abs{f_n'(z) - f'(z)} =
\Bigl|\frac{1}{2\iu\pi}\int_{C_{2r}}\frac{f_n(w) - f(w)}{(w -
z)^2}\,\dd w\Bigr|
\leq \frac{2r\,\sup_{C_{2r}}\abs{f_n - f}}{r^2} \to 0
\]
देता है, और वह भी $\bar D(a,r)$ पर एकसमान रूप से; किसी \omterm{def:b3:topology:compact}{संहत} को परिमित
संख्या के ऐसे चक्रों से आच्छादित कीजिए, और ऊँचे $k$ के लिए
दोहराइए।
\end{proof}

\begin{method}\label{met:b3:holomorphic:toolkit}
रोज का औजार-बक्सा। किसी फलन को \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} सिद्ध करने के लिए: उसे
किसी घात श्रेणी, संयोजन, स्थानीय एकसमान सीमा (\cref{thm:b3:holomorphic:weierstrassconv}), अथवा
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक} प्राचल वाले समाकल (\cref{exo:b3:holomorphic:7} --- $\int$ के भीतर अवकलन
कीजिए या मोरेरा--फुबिनी लगाइए) के रूप में प्रस्तुत कीजिए।
सर्वसमिकाएँ सिद्ध करने के लिए: उन्हें किसी रेखाखंड या उपप्रांत पर
सिद्ध कीजिए और तत्समता प्रमेय का आह्वान कीजिए। परिबद्ध करने के
लिए: उपलब्ध सबसे बड़े वृत्त पर \omterm{thm:b3:holomorphic:analytic}{कोशी आकलन}। अचरता या अनस्तित्व सिद्ध
करने के लिए: ल्यूविल अथवा उच्चतम सिद्धांत। सदा जानिए कि आपका फलन
\emph{कहाँ} \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है और $\Omega$ में \emph{कौन-से} चक्र समाते
हैं।
\end{method}

(क) $z \mapsto \bar z$, $\abs z^2$, $\operatorname{Re}z$ किन बिंदुओं पर सम्मिश्र-अवकलनीय हैं? किसी

है, उसे स्पष्ट रूप से ढूँढ़िए, और \emph{सभी} ऐसे फलन निर्धारित

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:holomorphic:2}
परिभाषाओं से परिकलित कीजिए: सभी $n
\in \Z$ के लिए $\int_{C}z^n\,\dd z$, जहाँ $C$
इकाई वृत्त है; तथा रेखाखंड $[0, 1+\iu]$ के अनुदिश और $1$ से होकर जाने
वाले दो-खंडी \omterm{def:b3:topology:pathconnected}{पथ} के अनुदिश $\int_\gamma\bar z\,\dd z$: निष्कर्ष निकालिए कि इन पथों के
किसी भी \omterm{def:b3:topology:topology}{प्रतिवेश} पर $\bar z$ का कोई आद्यंतर नहीं है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:3}

(ख) दर्शाइए कि $\C^*$ पर कोई \omterm{def:b3:topology:continuity}{संतत} लघुगणक विद्यमान नहीं है

(ग) $D(0,1)$ पर $\log(1 + z)$ का घात श्रेणी में प्रसार कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:4}
(क) मान लीजिए $f$ $\abs{f(z)} \leq C(1 + \abs
z)^{n}$ वाला समग्र फलन है। दर्शाइए कि $f$

(ग) ``आफ़ीन प्रतिचित्रणों के लिए लघु पिकार'' निष्कर्ष निकालिए:

\end{exercise}

$\Gamma(z) =

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:11}
(श्वार्ट्स परावर्तन) मान लीजिए $\Omega^+ = \{z : \abs z < 1,\
\operatorname{Im}z > 0\}$, $I = \intoo{-1}1$, और $f$ $\Omega^+$ पर
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है, $\Omega^+\cup I$ पर \omterm{def:b3:topology:continuity}{संतत}, और \emph{$I$ पर वास्तविक-मान
वाला}। परिभाषित कीजिए
\[
F(z) = \begin{cases} f(z) & z \in \Omega^+\cup I,\\
\overline{f(\bar z)} & \bar z \in \Omega^+ . \end{cases}
\]
(क) दर्शाइए कि $F$ $\Omega = \Omega^+\cup I\cup\Omega^-$ पर सुपरिभाषित और \omterm{def:b3:topology:continuity}{संतत} है, तथा $\Omega^\pm$ पर
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक} \emph{($\Omega^-$ के लिए: $\overline{f(\bar z)}$ के लिए कोशी--रीमान जाँचिए,
अथवा $f$ का स्थानीय घात श्रेणी में प्रसार कीजिए और गुणांकों के
संयुग्म लीजिए)}।
(ख) मोरेरा की कसौटी से दर्शाइए कि $F$ समूचे $\Omega$ पर
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है: प्रत्येक त्रिभुज $T \subseteq \Omega$ के लिए $\int_{\partial T}F = 0$ \emph{($I$
पर त्रिभुजों को बाँटिए और एकसमान \omterm{def:b3:topology:continuity}{सांतत्य} का उपयोग करते हुए उनकी
क्षैतिज भुजाओं को $\varepsilon$ से अक्ष से हटाइए)}।
(ग) निष्कर्ष निकालिए: चक्र पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} और किसी व्यास पर
वास्तविक कोई फलन $f(\bar z) = \overline{f(z)}$ को संतुष्ट करता है; और कोई अचरेतर
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक फलन} अपने (\omterm{def:b3:topology:connected}{संबद्ध}) प्रांत के किसी भी अरिक्त विवृत
उपसमुच्चय पर वास्तविक-मान वाला नहीं हो सकता।
\end{exercise}

\section{समस्या: बीजगणित की मूल प्रमेय, दो बार}

\begin{problem}[{सप्ताहांत समस्या --- $\C$ बीजीय रूप से संवृत है:
दालांबेर की उपपत्ति, ल्यूविल की उपपत्ति, और फसल}]\label{pb:b3:holomorphic:1}
मान लीजिए $P(z) = z^n + a_{n-1}z^{n-1} + \dots + a_0$, $n \geq 1$। हम दो बार सिद्ध करते हैं कि $P$ का कोई
मूल है, और फिर वह सब बटोरते हैं जिसकी बीजगणित \cref{ch:b3:galois} से प्रतीक्षा
कर रहा है।

\medskip
\noindent\textbf{भाग I --- प्रबलकारिता और न्यूनतम।}
\begin{enumerate}
  \item दर्शाइए कि $\abs z \to
        \infty$ पर $\abs{P(z)} \to +\infty$: ठीक-ठीक, उपयुक्त $\abs z \geq R_0$ के लिए
        $\abs{P(z)} \geq \frac12\abs z^n$।
  \item निष्कर्ष निकालिए कि $\abs P$ $\C$ पर एक वैश्विक न्यूनतम
        प्राप्त कर लेता है: ऐसा $z_0$ है जिसके लिए $\abs{P(z_0)} = \inf_\C\abs P$
        \emph{(किसी बड़े संवृत चक्र की संहतता, \cref{cor:b3:topology:heineborel})}।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दालांबेर का अवरोहण।} विरोधाभास के लिए
मान लीजिए $P(z_0) \neq 0$।
\begin{enumerate}[resume]
  \item $Q(h) = P(z_0 + h)/P(z_0)$ का $h$ में बहुपद के रूप में प्रसार कीजिए: $Q(h) = 1 + c_kh^k + h^{k+1}S(h)$,
        जहाँ $c_k \neq
        0$, $k \geq 1$, और $S$ कोई बहुपद।
  \item अवरोहण की दिशा चुनिए: छोटे $t > 0$ के लिए $h = t\,\omega$ रखिए, जहाँ
        $\omega^k = -1/c_k$ (ऐसा $\omega$ क्यों विद्यमान है? --- जिस प्रमेय को
        सिद्ध किया जा रहा है उससे स्वतंत्र रूप से, ध्रुवीय रूप के
        माध्यम से किसी भी सम्मिश्र संख्या के $k$-वें मूलों का
        अस्तित्व \emph{सिद्ध} कीजिए)। दर्शाइए कि छोटे $t$ के लिए
        किसी स्पष्ट नियतांक $C$ के साथ
        \[
        \abs{Q(t\omega)} \leq 1 - t^k + C\,t^{k+1}
        \]
  \item निष्कर्ष निकालिए कि छोटे $t$ के लिए $\abs{Q(t\omega)} < 1$ --- जो $\abs{P(z_0)}$
        की न्यूनतमता का खंडन है। अतः $P(z_0) = 0$: \emph{प्रत्येक अचरेतर
        सम्मिश्र बहुपद का कोई मूल होता है} (दालांबेर--आर्गां)।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- ल्यूविल की एक पंक्ति, पूरी।}
\begin{enumerate}[resume]
  \item \cref{cor:b3:holomorphic:liouville} की उपपत्ति सावधानी से लिखिए: यदि $P$ का कोई मूल न
        हो, तो सत्यापित कीजिए कि $1/P$ समग्र और परिबद्ध है
        (प्रश्न 1 का उपयोग करके मात्रात्मक बनाइए), अतः अचर, और
        निष्कर्ष निकालिए। दोनों उपपत्तियों की तुलना कीजिए:
        प्रत्येक किन अवयवों का उपयोग करती है? (संहतता दोनों में
        आती है --- कहाँ?)
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- फसल।}
\begin{enumerate}[resume]
  \item दर्शाइए कि घात $n$ का प्रत्येक $P \in \C[X]$ विभाजित हो जाता है:
        $\sum m_i =
        n$ के साथ $P = c\prod_{i}(X - \alpha_i)^{m_i}$ \emph{(आगमन, $(X -
        \alpha)$ से यूक्लिडीय
        विभाजन)}।
  \item दर्शाइए कि $\R[X]$ के \omterm{def:b3:rings:divisibility}{अखंडनीय} बहुपद रैखिक बहुपद और ऋणात्मक
        विविक्तकर वाले द्विघात हैं \emph{(संयुग्म मूलों को
        युग्मित कीजिए)}; इससे निष्कर्ष निकालिए कि विषम घात के
        प्रत्येक वास्तविक बहुपद का कोई वास्तविक मूल होता है, और
        उस अंतिम तथ्य की एक दूसरी, क्रम-सैद्धांतिक उपपत्ति दीजिए
        (मध्यवर्ती मान प्रमेय) --- और जाँचिए कि दोनों $X^3 - X - 1$ पर
        सहमत हैं।
  \item वे ऋण निकालिए जो यह पुस्तक अब चुका सकती है: (क) किसी
        अशून्य परिमित-विमीय $\C$-सदिश समष्टि की प्रत्येक
        अंतःसमाकारिता का कोई अभिलक्षणिक मान होता है, अतः प्रत्येक
        सम्मिश्र आव्यूह का \omterm{thm:b3:modules:jordan}{जॉर्डन रूप} होता है (\cref{thm:b3:modules:jordan}); (ख) \cref{rem:b3:galois:closureexamples}
        में प्रयुक्त \omterm{def:b3:galois:algebraic}{बीजीय} संख्याओं का क्षेत्र $\bar\Q$ वस्तुतः
        $\Q$ की एक \omterm{def:b3:galois:closure}{बीजीय संवृति} है।
  \item (समापन) ठीक-ठीक बताइए कि $\Q(\iu)$ जैसे किसी क्षेत्र पर
        प्रत्येक उपपत्ति कहाँ टूट जाती: कौन-से चरण $k$-वें मूलों
        के अस्तित्व (प्रश्न 4) का उपयोग करते हैं, और कौन-से संहतता
        या \omterm{def:b3:complete:complete}{पूर्णता} (प्रश्न 2 और 6) का? पाँच पंक्तियों में निष्कर्ष
        निकालिए: यह प्रमेय वास्तव में \emph{वैश्लेषिक} है ---
        प्रत्येक उपपत्ति कहीं न कहीं $\R$ की \omterm{def:b3:complete:complete}{पूर्णता} या \omterm{def:b3:topology:connected}{संबद्धता}
        का आह्वान करती है --- यद्यपि उसका कथन विशुद्ध \omterm{def:b3:galois:algebraic}{बीजीय} है।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- समग्र फलनों की कठोरता की सीढ़ी।}
ल्यूविल किसी सीढ़ी का पहला डंडा है; अब हम उस पर चढ़ते हैं।
\begin{enumerate}[resume]
  \item (\omterm{thm:b3:holomorphic:analytic}{कोशी आकलन}) $a$ के परितः त्रिज्या $r$ के वृत्त पर कोशी
        सूत्र से सिद्ध कीजिए
        \[
        \bigl|f^{(n)}(a)\bigr| \leq
        \frac{n!\,\sup_{\abs{z-a}=r}\abs f}{r^n} ,
        \]
        और $n = 1$, $r \to
        \infty$ की स्थिति के रूप में ल्यूविल पुनः प्राप्त
        कीजिए।
  \item (बहुपदीय वृद्धि) दर्शाइए कि सभी $z$ के लिए $\abs{f(z)} \leq A + B\abs z^m$ वाला
        कोई समग्र $f$ घात $\leq m$ का बहुपद होता है \emph{(प्रश्न
        11 से $m$ के आगे के टेलर गुणांक समाप्त कीजिए)}।
  \item (परिबद्ध वास्तविक भाग) दर्शाइए कि ऊपर से परिबद्ध $\operatorname{Re}f$
        वाला कोई समग्र $f$ अचर होता है \emph{($\eu^{f}$ पर ल्यूविल
        लगाइए)}।
  \item (द्विक आवर्तिता) मान लीजिए $f$ समग्र है, $f(z + 1)
        = f(z)$, और सभी
        $z$ के लिए $f(z + \iu) = f(z)$। दर्शाइए कि $f$ अचर है। निष्कर्ष
        निकालिए: किसी अचरेतर ``दीर्घवृत्तीय'' फलन की एकलताएँ
        होनी ही चाहिए --- यही वह ऐतिहासिक कारण है कि सम्मिश्र
        विश्लेषण में ध्रुव प्रवेश करते हैं।
  \item (सघन परिसर) दर्शाइए कि किसी अचरेतर समग्र फलन का परिसर
        $\C$ में सघन होता है: यदि $f(\C)$ किसी चक्र $D(a, r)$ को छोड़
        देता हो, तो $\frac1{f - a}$ समग्र और परिबद्ध होता। (पिकार ने सिद्ध
        किया कि परिसर अधिकतम \emph{एक बिंदु} छोड़ता है; सघनता वह
        स्तर है जहाँ तक हमारे औजार पहुँचते हैं।)
  \item (उचित $\Rightarrow$ बहुपद) मान लीजिए $f$ समग्र है और $\abs z \to
        \infty$ पर
        $\abs{f(z)} \to \infty$। दर्शाइए: $f$ के शून्य परिमित संख्या में हैं
        ($z_1, \dots, z_p$, बहुलताओं $m_i$ सहित); भागफल $g = f/\prod(z - z_i)^{m_i}$ समग्र और
        शून्य-रहित है; $1/g$ की वृद्धि बहुपदीय है, अतः (प्रश्न 12)
        वह बहुपद है, और शून्य-रहित होने के कारण अनिवार्यतः अचर;
        निष्कर्ष निकालिए कि $f$ एक बहुपद है। अतः समग्र फलनों में
        बहुपद ठीक वही हैं जो उचित हैं --- $\eu^z$ $\R_-$ के अनुदिश
        उचितता खो देता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- प्रसंवादी छायाएँ और गाउस का एक माध्य
मान।}
\begin{enumerate}[resume]
  \item मान लीजिए $f = u + \iu v$ किसी \omterm{def:b3:topology:topology}{विवृत समुच्चय} पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है।
        सत्यापित कीजिए कि $u = \operatorname{Re}f$ माध्य मान गुण
        \[
        u(a) = \frac1{2\pi}\int_0^{2\pi}
        u\bigl(a + r\eu^{\iu\theta}\bigr)\,\dd\theta
        \]
        को संतुष्ट करता है (कोशी सूत्र का वास्तविक भाग), और किसी
        परिबद्ध प्रांत पर $u$ के लिए उच्चतम सिद्धांत निष्कर्ष के
        रूप में निकालिए, उसी \omterm{def:b3:topology:connected}{संबद्धता} उपपत्ति से जो $\abs f$ के लिए
        थी।
  \item (गाउस का माध्य मान) $a \in \C$ और $\abs a \neq r$ वाले $r > 0$ के लिए
        सिद्ध कीजिए
        \[
        \frac1{2\pi}\int_0^{2\pi}
        \log\bigl|a - r\eu^{\iu\theta}\bigr|\,\dd\theta
        = \log\max\bigl(\abs a, r\bigr)
        \]
        \emph{(यदि $\abs a > r$: $z \mapsto \log\abs{a - z}$ संवृत चक्र के किसी \omterm{def:b3:topology:topology}{प्रतिवेश} पर
        किसी \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} लघुगणक का वास्तविक भाग है --- ऐसा
        क्यों विद्यमान है? --- अतः प्रश्न 17 लागू होता है; और यदि
        $\abs a < r$: $\abs{a - r\eu^{\iu\theta}} = r\,\abs{1 -
        \frac ar\eu^{-\iu\theta}}$ का गुणनखंडन कीजिए और पहली स्थिति फिर से
        प्रयोग कीजिए)}।
  \item (माह्लर \omterm{def:b3:measure:measure}{माप}) $P = c\prod_{i=1}^n(X -
        \alpha_i) \in \C[X]$ के लिए \emph{बहुपदों के लिए येंसन का
        सूत्र} निकालिए:
        \[
        \frac1{2\pi}\int_0^{2\pi}\log\bigl|P(\eu^{\iu\theta})
        \bigr|\,\dd\theta
        = \log\Bigl(\abs c\prod_{i}\max(1,
        \abs{\alpha_i})\Bigr) :
        \]
        अर्थात् इकाई वृत्त पर $\abs P$ का गुणोत्तर माध्य चक्र के बाहर
        के मूल पढ़ लेता है। $P =
        X^2 - X$ पर और $P = 2X - 1$ पर सत्यापित कीजिए।
  \item (बर्नूली संख्याएँ) $0$ के निकट $\frac{z}{\eu^z - 1} = \sum_{n\geq0}\frac{B_n}{n!}
        z^n$ से गुणांक $B_n$
        परिभाषित कीजिए (बायाँ पक्ष $0$ पर वैश्लेषिक क्यों है?)।
        $(\eu^z - 1)\cdot\frac z{\eu^z-1} = z$ से पुनरावर्तन $\sum_{k=0}^{n}\binom{n+1}kB_k = 0$ ($n \geq 1$) निकालिए, $B_0,
        \dots, B_6$ परिकलित
        कीजिए, और $k \geq 1$ के लिए $B_{2k+1} = 0$ दर्शाइए \emph{(फलन $\frac z{\eu^z-1} + \frac z2$ सम
        है)}। ये संख्याएँ \cref{ch:b3:residues} में प्रत्येक $\zeta(2k)$ का मोल तय
        करेंगी।
  \item (वास्तविकता) दर्शाइए कि $\R$ पर वास्तविक मान लेने वाला
        कोई समग्र फलन सर्वत्र $f(\bar z) =
        \overline{f(z)}$ को संतुष्ट करता है
        \emph{($0$ पर टेलर गुणांकों की तुलना कीजिए, अथवा $z \mapsto \overline{f(\bar z)}$
        पर तत्समता प्रमेय लगाइए)}; इससे पुनः निष्कर्ष निकालिए कि
        वास्तविक बहुपदों के अवास्तविक मूल संयुग्मी युग्मों में
        आते हैं (प्रश्न 8 की युग्मन, वैश्लेषिक रूप से पुनः
        सिद्ध)।
  \item (सार) कठोरता की सीढ़ी संकलित कीजिए: परिबद्ध $\Rightarrow$ अचर;
        बहुपदीय रूप से परिबद्ध $\Rightarrow$ बहुपद; उचित $\Rightarrow$ बहुपद; किसी
        चक्र को छोड़ने वाला $\Rightarrow$ अचर; द्विक आवर्ती $\Rightarrow$ अचर।
        एक छोटे अनुच्छेद में वास्तविक $\mathcal C^\infty$ फलनों (\omterm{def:b3:lp:mollifier}{उभार फलन},
        \cref{thm:b3:lp:regularization}) से तुलना कीजिए: \omterm{def:b3:holomorphic:holo}{समविश्लेषिकता}, जो विशुद्ध स्थानीय
        शर्त है, वैश्विक विधि-व्यवस्था क्यों थोप देती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग VII --- अंतिम फसल।}
\begin{enumerate}[resume]
  \item (लांडाउ की असमिका) $P = \sum_{k=0}^na_kX^k$ के लिए माध्य मान $\frac1{2\pi}\int_0^{2\pi}
        \abs{P(\eu^{\iu\theta})}^2\dd\theta =
        \sum_k\abs{a_k}^2$ सिद्ध
        कीजिए ($\eu^{\iu k\theta}$ की लांबिकता), फिर बिंदुशः परिबंध $\log t \leq t - 1$ का
        उपयोग करके $\log\abs P^2$ और $\abs P^2$ के माध्यों की तुलना कीजिए, और
        प्रश्न 19 से निकालिए
        \[
        \abs c\prod_{i}\max\bigl(1, \abs{\alpha_i}\bigr)
        \;\leq\;
        \Bigl(\sum_{k=0}^{n}\abs{a_k}^2\Bigr)^{1/2} :
        \]
        अर्थात् इकाई चक्र के बाहर के मूलों का गुणनफल गुणांकों के
        $\ell^2$ आकार से नियंत्रित होता है (वृत्त \emph{पर} के मूल
        $P(rX)$ पर असमिका लगाकर और $r \to 1$ जाने देकर सँभालिए)। इसे
        $X^2 - X$ पर जाँचिए।
  \item (बर्नूली संख्याएँ क्रमगुणित की तरह बढ़ती हैं) दर्शाइए कि
        $\sum\frac{B_n}{n!}z^n$ की अभिसरण त्रिज्या \emph{ठीक} $2\pi$ है: कम से कम
        $2\pi$ इसलिए कि $z/(\eu^z - 1)$ $D(0, 2\pi)$ तक \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} रूप से बढ़ जाता
        है, और अधिकतम $2\pi$ इसलिए कि अन्यथा योग $2\pi\iu$ के निकट
        परिबद्ध रहता, जहाँ $\abs{z/(\eu^z-1)} \to \infty$। इससे निकालिए
        \[
        \limsup_{k\to\infty}
        \Bigl(\frac{\abs{B_{2k}}}{(2k)!}\Bigr)^{1/2k}
        = \frac1{2\pi} :
        \]
        अर्थात् बर्नूली संख्याएँ क्रमगुणित की तरह बढ़ती हैं।
        $B_{12}
        = -\frac{691}{2730}$ स्वीकार करते हुए (प्रश्न 20 का पुनरावर्तन, आगे
        चलाया गया), $\abs{B_{12}} \approx 0.25311$ की तुलना $k = 6$ पर तीक्ष्णतर भविष्यवाणी
        $2\,(2k)!/(2\pi)^{2k}
        \approx 0.25305$ से कीजिए --- किसी स्पर्शोन्मुख नियम के चार मिलते
        हुए अंक, जिसे \cref{ch:b3:residues} $\zeta(2k)$ के माध्यम से ठीक-ठीक सिद्ध
        करेगा।
  \item (मूल संततता से हिलते हैं) मान लीजिए $(P_j)$ घात $n$ के
        मोनिक बहुपद हैं जिनके गुणांक (मोनिक) $P$ के गुणांकों पर
        अभिसरित होते हैं। \emph{कोशी परिबंध} सिद्ध कीजिए: किसी
        मोनिक $Q = X^n + \sum_{k<n}q_kX^k$ का प्रत्येक मूल $\abs\alpha \leq 1 + \max_k\abs{q_k}$ को संतुष्ट करता है;
        इससे निष्कर्ष निकालिए कि $P_j$ के मूल किसी स्थिर \omterm{def:b3:topology:compact}{संहत}
        समुच्चय में रहते हैं, और मूल सदिशों के अभिसारी उपानुक्रम
        निकालकर तथा प्रश्न 7 के गुणनखंडन में सीमा लेकर, यह भी कि
        $P_j$ के मूल बहुसमुच्चय $P$ के मूल बहुसमुच्चय पर
        अभिसरित होते हैं। अंततः दर्शाइए कि इससे अधिक कुछ नहीं कहा
        जा सकता: $P_\varepsilon = X^2 - 2X + 1 +
        \varepsilon$ के लिए $\varepsilon =
        10^{-4}$ आकार का विक्षोभ द्विक मूल
        $1$ को $10^{-2}$ जितना हिला देता है --- अर्थात् किसी
        $m$-गुणी मूल पर होल्डर घातांक $\frac1m$, कभी लिप्शिट्ज़ नहीं:
        सांख्यिक दृष्टि से, बहुगुणी मूल आधे अंक खा जाते हैं।
\end{enumerate}
\end{problem}
