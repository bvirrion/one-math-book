\chapter{अनुरूप प्रतिचित्रण और रीमान प्रतिचित्रण प्रमेय}
\label{ch:b3:conformal}

दो प्रांतों के बीच कोई \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} एकैकी आच्छादक प्रतिचित्रण
समूचे सम्मिश्र विश्लेषण को एक से दूसरे तक ले जाता है: ऐसे
प्रतिचित्रण --- \emph{अनुरूप}, क्योंकि वे कोण सुरक्षित रखते हैं
--- \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} संसार की तुल्याकारिताएँ हैं। यह अध्याय जहाँ
वर्गीकरण संभव है वहाँ उनका वर्गीकरण करता है (चक्र, समतल: कुंजी
श्वार्ट्स की प्रमेयिका है, आश्चर्यजनक शक्ति वाली एक असमिका),
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक} कुलों का संहतता सिद्धांत बनाता है (मोंतेल), और इस
विषय की गहनतम अस्तित्व प्रमेय सिद्ध करता है: $\C$ का
\emph{प्रत्येक} \omterm{def:b3:conformal:simplyconnected}{सरल संबद्ध} उचित उपप्रांत, चाहे उसकी परिसीमा कितनी
ही दंतुरित क्यों न हो, इकाई चक्र के \omterm{def:b3:conformal:conformal}{अनुरूप तुल्य} होता है। हम
प्रसंवादी फलनों और \omterm{thm:b3:conformal:poisson}{प्वासों नाभिक} पर समाप्त करते हैं, और चक्र पर
दिरिक्ले समस्या हल करते हैं --- अनुरूप ज्यामिति का वैश्लेषिक
प्रतिफल। सर्वत्र $\mathbb D = D(0,1)$ और $\mathbb H = \{\operatorname{Im}z >
0\}$।

\section{अनुरूप प्रतिचित्रण; मोबियस रूपांतरण}

\begin{definition}\label{def:b3:conformal:conformal}
विवृत समुच्चयों के बीच \emph{अनुरूप प्रतिचित्रण}\index{अनुरूप
प्रतिचित्रण} कोई \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} एकैकी आच्छादक प्रतिचित्रण है; उसका
प्रतिलोम स्वतः \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} होता है (\cref{cor:b3:residues:openmapping})। दो प्रांत \emph{अनुरूप
तुल्य} कहलाते हैं यदि ऐसा प्रतिचित्रण विद्यमान हो; $\operatorname{Aut}(\Omega)$ अनुरूप
स्व-प्रतिचित्रणों का समूह निरूपित करता है। जहाँ $f' \neq 0$ --- और एकैकी
$f$ के लिए सर्वत्र (\cref{cor:b3:residues:openmapping} की उपपत्ति) --- वहाँ अवकलज $f'(z) \neq 0$ से गुणन
है: अर्थात् एक समरूपता, अतः अनुरूप प्रतिचित्रण वक्रों के बीच के
कोण सुरक्षित रखते हैं, अभिविन्यास सहित।
\end{definition}

\begin{example}[मोबियस रूपांतरण]
\label{ex:b3:conformal:mobius}
$\bigl(\begin{smallmatrix}a & b\\ c &
d\end{smallmatrix}\bigr) \in GL_2(\C)$ के लिए \emph{मोबियस रूपांतरण}\index{मोबियस रूपांतरण} $z \mapsto
\frac{az + b}{cz + d}$
$\C\setminus\{-d/c\}$ से $\C\setminus\{a/c\}$ पर अनुरूप है (प्रतिलोम भी उसी प्रकार का, प्रतिलोम
आव्यूह से; और संयोजन आव्यूह गुणनफल के अनुरूप है)। \emph{केली
प्रतिचित्रण}\index{केली रूपांतर}
\[
\varphi(z) = \frac{z - \iu}{z + \iu}
\]
$\mathbb H$ को $\mathbb D$ पर अनुरूप रूप से भेजता है: वस्तुतः $\abs{z
- \iu} < \abs{z + \iu}$ ठीक तब जब
$z$ $-\iu$ की अपेक्षा $\iu$ के अधिक निकट हो, अर्थात् $\operatorname{Im}z > 0$; और
प्रतिलोम $w \mapsto \iu\frac{1 + w}{1 - w}$ है। मोबियस प्रतिचित्रण वृत्तों-और-रेखाओं के कुल को
अपने आप पर भेजते हैं (\cref{exo:b3:conformal:1})।
\end{example}

\begin{example}[ज़ूकोव्स्की प्रतिचित्रण]\label{ex:b3:conformal:joukowski}
मोबियस से आगे, शास्त्रीय अनुप्रयुक्त गणित का सबसे उपयोगी \omterm{def:b3:conformal:conformal}{अनुरूप
प्रतिचित्रण} है
\[
J(z) = \frac12\Bigl(z + \frac1z\Bigr) .
\]
इकाई चक्र के बाह्य भाग $\Omega = \{\abs z > 1\}$ पर $J$ एकैकी है: $J(z) = J(w)$ से $(z - w)(1 -
\frac1{zw}) = 0$ और
$\abs{zw} > 1$ मिलते हैं। उसका अवकलज $J'(z) =
\frac12(1 - z^{-2})$ केवल $z = \pm1$ पर, यानी परिसीमा पर,
लुप्त होता है: अतः $J$ $\Omega$ से उसके प्रतिबिंब पर अनुरूप तुल्यता
है, और वह प्रतिबिंब $\C\setminus\intcc{-1}1$ है --- इकाई वृत्त स्वयं दो-से-एक ढंग से
रेखाखंड पर मुड़ जाता है ($J(\eu^{\iu\theta}) = \cos\theta$)। इस प्रकार किसी \emph{रेखाखंड} का
बाह्य भाग, जो चिकनी परिसीमा रहित एक कटा समतल है, अनुरूप दृष्टि से
किसी चक्र का बाह्य भाग ही है: कोने अनुरूप तुल्यता में बाधा नहीं
हैं, केवल परिसीमा की चिकनाई में हैं। $\pm1$ से होकर जाने वाले पर
अकेंद्रित वृत्तों के प्रतिबिंब वायुपंख के आकार के वक्र होते हैं,
और $J$ को मोबियस प्रतिचित्रणों के साथ जोड़ने से किसी बेलन के
चारों ओर का प्रवाह --- जो हाथ से परिकलनीय है --- किसी पंख के
चारों ओर के प्रवाह पर पहुँच जाता है: बीसवीं शताब्दी के पूर्वार्ध
में यह उदाहरण \emph{ही} वायुगतिकी था। यह चेबिशेव का द्वार भी है:
$J$ $z \mapsto z^n$ को चेबिशेव बहुपद $T_n$ (\cref{pb:b3:hilbert:1}, भाग V) से संयुग्मित कर
देता है, क्योंकि $z = \eu^{\iu\theta}$ होने पर $J(z^n) = \cos(n\theta)$।
\end{example}

\section{श्वार्ट्स की प्रमेयिका और स्वसमाकारिता समूह}

\begin{theorem}[श्वार्ट्स प्रमेयिका]\label{thm:b3:conformal:schwarz}
मान लीजिए $f \colon \mathbb D \to \mathbb D$ $f(0) = 0$ के साथ \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है। तब\index{श्वार्ट्स
प्रमेयिका}
\[
\abs{f(z)} \leq \abs z \ \ (z \in \mathbb D)
\qquad\text{तथा}\qquad \abs{f'(0)} \leq 1 ,
\]
और यदि किसी एक $z_0 \neq 0$ के लिए $\abs{f(z_0)} = \abs{z_0}$ हो, अथवा $\abs{f'(0)} = 1$ हो, तो $f(z) = \eu^{\iu\theta}z$ कोई
घूर्णन है।
\end{theorem}

\begin{proof}
$g(z) = f(z)/z$ $\mathbb D$ तक \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} रूप से बढ़ जाता है ($0$ पर एकलता
विलोपनीय है: $g$ $0$ के निकट परिबद्ध है, \cref{thm:b3:residues:riemanncw}; और $0$ पर उसका
मान $f'(0)$ है)। $\abs z = r < 1$ पर: $\abs g \leq \frac1r$, अतः उच्चतम सिद्धांत (\cref{thm:b3:holomorphic:maximum}) से $\bar D(0,r)$
पर $\abs g \leq \frac1r$; अब $r \to 1$ लीजिए: $\mathbb D$ पर $\abs
g \leq 1$, जो दोनों असमिकाएँ हैं।
किसी आंतरिक बिंदु पर समता $\abs g$ को कोई आंतरिक अधिकतम प्राप्त करा
देती है: अतः $g$ मापांक $1$ का अचर है।
\end{proof}

\begin{theorem}[चक्र की स्वसमाकारिताएँ]
\label{thm:b3:conformal:autdisc}
$a \in \mathbb D$ के लिए \emph{ब्लाश्के गुणनखंड}\index{ब्लाश्के गुणनखंड}
\[
\varphi_a(z) = \frac{z - a}{1 - \bar a z}
\]
$\mathbb D$ की ऐसी स्वसमाकारिता है जो $a$ और $0$ की अदला-बदली करती है,
और $\varphi_a^{-1} = \varphi_{-a}$। $\mathbb D$ की प्रत्येक स्वसमाकारिता अद्वितीय $\theta
\in \R/2\pi\Z$, $a \in \mathbb D$ के लिए
$\eu^{\iu\theta}\varphi_a$ होती है।
\end{theorem}

\begin{proof}
$\abs z = 1$ पर: $\abs{1 - \bar az} = \abs{\bar z}\abs{1 -
\bar az} = \abs{\bar z - \bar a\abs z^2} = \abs{\bar z - \bar
a} = \abs{z - a}$, अतः वहाँ $\abs{\varphi_a} = 1$; उच्चतम सिद्धांत से $\varphi_a(\mathbb D) \subseteq \bar{\mathbb
D}$, और विवृतता
प्रतिबिंब को $\mathbb D$ में रख देती है। बीजगणितीय सर्वसमिका $\varphi_{-a}\circ\varphi_a = \mathrm{id}$ (सीधा
परिकलन) एकैकी आच्छादकता दिखा देती है। अब मान लीजिए $f \in
\operatorname{Aut}(\mathbb D)$ और $a = f^{-1}(0)$:
तब $g =
f\circ\varphi_{-a}$ $0$ को स्थिर रखने वाली स्वसमाकारिता है। $g$ पर और $g^{-1}$
पर श्वार्ट्स लगाइए: $\abs{g(z)} \leq \abs z$ और $\abs{g^{-1}(w)} \leq \abs w$, अतः $\abs{g(z)} = \abs z$: यानी घूर्णन, $g = \eu^{\iu\theta}\,\mathrm{id}$,
अर्थात् $f =
\eu^{\iu\theta}\varphi_a$। अद्वितीयता: $f'$ प्रकार के मान-निर्धारण से $a = f^{-1}(0)$
और $\theta$ (अथवा $f(0) =
-\eu^{\iu\theta}a$ और एक और मान से)।
\end{proof}

\begin{theorem}[समतल की स्वसमाकारिताएँ]
\label{thm:b3:conformal:autplane}
$\operatorname{Aut}(\C) = \{z \mapsto az + b : a \in \C^*,\ b
\in \C\}$। फलस्वरूप $\C$ और $\mathbb D$ \omterm{def:b3:conformal:conformal}{अनुरूप तुल्य} नहीं हैं।
\end{theorem}

\begin{proof}
मान लीजिए $f \in \operatorname{Aut}(\C)$ और $\C^*$ पर $g(z) =
f(1/z)$ पर विचार कीजिए: यह $0$ पर
\omterm{def:b3:residues:singularities}{विविक्त एकलता} वाला \omterm{def:b3:holomorphic:holo}{समविश्लेषिक फलन} है। यदि वह सारभूत होती, तो
काज़ोराती--वाइरश्ट्रास (\cref{thm:b3:residues:riemanncw}) $g\bigl(D(0,\varepsilon)\setminus\{0\}\bigr)$ को सघन बना देती, जबकि $f(D(0, 1))$
विवृत है और उससे असंयुक्त ($f$ एकैकी: दोनों समुच्चय असंयुक्त
समुच्चयों के प्रतिबिंब हैं) --- जो किसी \omterm{def:b3:topology:interior}{सघन समुच्चय} और किसी अरिक्त
\omterm{def:b3:topology:topology}{विवृत समुच्चय} के लिए असंभव है। अतः $g$ के लिए $0$ ध्रुव है या
विलोपनीय, अर्थात् $\abs{f(z)}$ की वृद्धि अधिकतम बहुपदीय है: इसलिए $f$
कोई बहुपद है (\cref{exo:b3:holomorphic:4}(क))। एकैकीपन घात $1$ अनिवार्य कर देता है:
उच्चतर घात के बहुपद का या तो कहीं $f - c$ का बहुगुणी मूल होता है
($f'$ लुप्त होता है) या कई भिन्न पूर्वप्रतिबिंब (दालांबेर--गाउस,
\cref{pb:b3:holomorphic:1}); दोनों ही स्थितियों में एकैकीपन विफल हो जाता है। अंततः कोई
अनुरूप $\C \to
\mathbb D$ परिबद्ध समग्र फलन होता: अचर (ल्यूविल) --- अतः कोई
तुल्यता नहीं।
\end{proof}

\section{मोंतेल की प्रमेय}

\begin{theorem}[मोंतेल]\label{thm:b3:conformal:montel}
मान लीजिए $\mathcal F \subseteq \mathcal H(\Omega)$ \emph{स्थानीय परिबद्ध} है: अर्थात् प्रत्येक बिंदु
का कोई ऐसा \omterm{def:b3:topology:topology}{प्रतिवेश} है जिस पर $\sup_{f\in
\mathcal F}\sup\abs f < \infty$। तब $\mathcal F$ के प्रत्येक अनुक्रम
का कोई ऐसा उपानुक्रम है जो $\Omega$ के समस्त \omterm{def:b3:topology:compact}{संहत} उपसमुच्चयों पर
एकसमान रूप से (किसी \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} सीमा पर) अभिसरित होता
है।\index{मोंतेल की प्रमेय}
\end{theorem}

\begin{proof}
स्थानीय \omterm{def:b3:complete:equicontinuous}{समसांतत्य}: यदि सभी $f \in \mathcal F$ के लिए $D(a, 2r)
\subseteq \Omega$ पर $\abs f \leq M$ हो, तो
$z, z' \in D(a, r)$ के लिए कोशी सूत्र
\[
\abs{f(z) - f(z')} = \frac{\abs{z - z'}}{2\pi}
\Bigl|\int_{C_{2r}}\frac{f(w)\,\dd w}{(w-z)(w-z')}\Bigr|
\leq \frac{\abs{z - z'}\,2\pi\cdot2r\,M}{2\pi\,r^2}
= \frac{2M}{r}\,\abs{z - z'} :
\]
देता है, अर्थात् एक एकसमान लिप्शिट्ज़ परिबंध। $\Omega$ को संहतों
$K_m$ से निःशेष कीजिए; प्रत्येक $K_m$ परिमित संख्या के ऐसे चक्रों
से आच्छादित होता है, अतः $\mathcal F$ $K_m$ पर एकसमान परिबद्ध और \omterm{def:b3:complete:equicontinuous}{समसंतत}
है: आर्ज़ेला--आस्कोली (\cref{thm:b3:complete:ascoli}) $K_m$ पर एकसमान रूप से अभिसारी कोई
उपानुक्रम निकाल देता है; अब $m$ पर विकर्णन कीजिए। सीमा \cref{thm:b3:holomorphic:weierstrassconv} से
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है।
\end{proof}

\section{रीमान प्रतिचित्रण प्रमेय}

\begin{definition}\label{def:b3:conformal:simplyconnected}
कोई विवृत \omterm{def:b3:topology:connected}{संबद्ध} $\Omega \subseteq \C$ \emph{सरल संबद्ध}\index{सरल संबद्ध प्रांत}
कहलाता है (समजातता के अर्थ में, जो हमारे सभी प्रयोजनों के लिए
पर्याप्त है) यदि $\Omega$ में प्रत्येक चक्रक $\gamma$ और प्रत्येक $w \notin \Omega$
के लिए $\operatorname{Ind}_\gamma(w) = 0$ हो --- अर्थात् ``$\Omega$ का कोई चक्रक किसी छिद्र को
नहीं घेरता''। वैश्विक कोशी प्रमेय (\cref{thm:b3:residues:globalcauchy}) और \cref{prop:b3:holomorphic:primitive} से ऐसे $\Omega$
पर \emph{प्रत्येक \omterm{def:b3:holomorphic:holo}{समविश्लेषिक फलन} का कोई आद्यंतर होता है}; अतः
प्रत्येक शून्य-रहित $f \in \mathcal H(\Omega)$ का कोई \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} लघुगणक होता है
($\exp\circ$($f'/f$ का आद्यंतर), जिसे किसी अचर से समायोजित किया गया हो,
क्योंकि $(f\eu^{-L})' = 0$) और \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} $n$-वें मूल $\eu^{L/n}$
भी।\index{समविश्लेषिक लघुगणक}
\end{definition}

\begin{theorem}[रीमान प्रतिचित्रण प्रमेय]
\label{thm:b3:conformal:rmt}
$\Omega
\neq \varnothing$ वाला प्रत्येक \omterm{def:b3:conformal:simplyconnected}{सरल संबद्ध} विवृत $\Omega \subsetneq \C$ $\mathbb D$ के \omterm{def:b3:conformal:conformal}{अनुरूप तुल्य}
है; और $z_0 \in \Omega$ दिया हो, तो $f(z_0) = 0$ तथा $f'(z_0) >
0$ वाला एक अद्वितीय अनुरूप
$f \colon
\Omega \to \mathbb D$ है।\index{रीमान प्रतिचित्रण प्रमेय}
\end{theorem}

\begin{proof}
\emph{चरण 0: कुल अरिक्त है।} $b \notin \Omega$ चुनिए: $z - b$ $\Omega$ पर
शून्य-रहित है, अतः उसका कोई \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} वर्गमूल $h$ है ($h^2 = z - b$)।
$h$ एकैकी है ($h(z) =
h(z')$ का वर्ग $z = z'$ है), और यदि $w \in h(\Omega)$ हो तो
$-w
\notin h(\Omega)$ ($h(z) = -h(z')$ का वर्ग भी $z = z'$ है, जिससे $w = -w = 0$ मिलता है, जो असंभव
है क्योंकि $h$ शून्य-रहित है)। चूँकि $h(\Omega)$ विवृत है, उसमें कोई
चक्र $D(h(z_0), \rho)$ समाता है; तब $D(-h(z_0), \rho) \cap h(\Omega) = \varnothing$, अर्थात् प्रत्येक $z \in \Omega$ के लिए
$\abs{h(z) + h(z_0)} \geq \rho$। अतः
\[
g(z) = \frac{\rho}{2\,\bigl(h(z) + h(z_0)\bigr)}
\]
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक} और एकैकी है (कोई मोबियस प्रतिचित्रण एकैकी $h$ के
साथ संयोजित), जिसमें $\abs g \leq \frac12 < 1$। $g(z_0)$ को $0$ पर ले जाने के लिए
किसी \omterm{thm:b3:conformal:autdisc}{ब्लाश्के गुणनखंड} (\cref{thm:b3:conformal:autdisc}) के साथ संयोजित करने पर कुल
\[
\mathcal F = \{f \colon \Omega \to \mathbb D \text{
समविश्लेषिक, एकैकी, } f(z_0) = 0\}
\]
अरिक्त हो जाता है।

\emph{चरण 1: एक चरम अवयव।} मान लीजिए $s =
\sup_{\mathcal F}\abs{f'(z_0)} \in \intoc0{+\infty}$ ($> 0$: सदस्य एकैकी
हैं, अतः $f'(z_0) \neq 0$)। $\abs{f_n'(z_0)} \to s$ वाले $f_n \in
\mathcal F$ लीजिए: कुल $1$ से परिबद्ध है,
अतः मोंतेल (\cref{thm:b3:conformal:montel}) संहतों पर एकसमान रूप से $f_n \to f$ निकाल देता है;
$f$ \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है, $f(z_0) = 0$, $\abs{f'(z_0)} = s$ (अवकलजों के लिए \cref{thm:b3:holomorphic:weierstrassconv}), और
विशेष रूप से $f$ अचरेतर है; $f$ हुर्विट्स (\cref{exo:b3:residues:8}(ख)) से एकैकी
है, और $f(\Omega) \subseteq \bar{\mathbb D}$, अतः $\subseteq
\mathbb D$ (विवृत प्रतिचित्रण)। इसलिए $f \in \mathcal F$ उच्चतम
प्राप्त कर लेता है: $s < \infty$।

\emph{चरण 2: चरम प्रतिचित्रण आच्छादक है।} मान लीजिए $a \in
\mathbb D\setminus f(\Omega)$।
ब्लाश्के परिवहन $\varphi_a\circ f$ \omterm{def:b3:conformal:simplyconnected}{सरल संबद्ध} $\Omega$ पर शून्य-रहित है: अतः उसका
कोई \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} वर्गमूल $F$ है ($F(\Omega) \subseteq \mathbb D$ के साथ, क्योंकि $\abs F^2 =
\abs{\varphi_a\circ f} < 1$),
जो एकैकी है (वर्ग भेद कर देते हैं)। मानकीकृत कीजिए: $G = \varphi_{F(z_0)}\circ F \in
\mathcal F$। उलटने
पर: $f = \varphi_{-a}\circ s_2 \circ
\varphi_{-F(z_0)}\circ G$, जहाँ $s_2(w) = w^2$; और प्रतिचित्रण $\Psi
= \varphi_{-a}\circ s_2\circ\varphi_{-F(z_0)} \colon \mathbb D
\to \mathbb D$ $\Psi(0) = f(z_0) = 0$ के साथ
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है और कोई घूर्णन \emph{नहीं} है (वह एकैकी नहीं है:
$s_2$ नहीं है)। श्वार्ट्स की प्रमेयिका (कड़ी स्थिति): $\abs{\Psi'(0)} < 1$, और
शृंखला नियम $f = \Psi\circ G$ से $\abs{f'(z_0)} =
\abs{\Psi'(0)}\,\abs{G'(z_0)} < \abs{G'(z_0)}$ --- जो महत्तमता का खंडन है (ध्यान
दीजिए कि $G \in \mathcal F$)। अतः $f$ आच्छादक है: अर्थात् एक अनुरूप तुल्यता।

\emph{चरण 3: मानकीकरण और अद्वितीयता।} $f'(z_0) > 0$ बनाने के लिए $f$ को
$\eu^{-\iu\arg f'(z_0)}$ से गुणा कीजिए (यह $\mathcal F$ में ही रहता है)। यदि $f_1, f_2$ दोनों
काम करें, तो $\psi =
f_2\circ f_1^{-1} \in \operatorname{Aut}(\mathbb D)$ $\psi'(0) = f_2'(z_0)/f_1'(z_0) > 0$ के साथ $0$ को स्थिर रखता है; \cref{thm:b3:conformal:autdisc} से
$\psi$ $\eu^{\iu\theta} > 0$ वाला कोई घूर्णन $\eu^{\iu\theta}$ है: अतः $\psi =
\mathrm{id}$।
\end{proof}

\begin{remark}\label{rem:b3:conformal:rmtscope}
यह प्रमेय आश्चर्यजनक विस्तार वाला विशुद्ध अस्तित्व कथन है: एक
वर्ग, एक अर्ध-समतल, किसी दरार का पूरक, दो स्पर्श करते वृत्तों के
बीच का क्षेत्र, किसी फ्रैक्टल परिसीमा वाला प्रांत --- सब $\mathbb D$ के
अनुरूप एक ही हैं। यह जो नहीं देती: कोई सूत्र (स्पष्ट प्रतिचित्रण
अपवाद हैं: \cref{exo:b3:conformal:5}), परिसीमा व्यवहार (उसे एक गहरा सिद्धांत ---
कैराथियोडोरी की प्रमेय --- सँभालता है), अथवा $\C$ तक या बहुसंबद्ध
प्रांतों तक विस्तार की अद्वितीयता: वलय $\{1 < \abs z < 2\}$ अनुरूप दृष्टि से
कोई छिद्रित चक्र \emph{नहीं} है, और भिन्न त्रिज्या अनुपात वाले
वलय परस्पर अतुल्य हैं (जो सचमुच कठिनतर तथ्य है)।
\end{remark}

\section{प्रसंवादी फलन और प्वासों नाभिक}

\begin{proposition}\label{prop:b3:conformal:harmonicholo}
मान लीजिए $\Omega$ \omterm{def:b3:conformal:simplyconnected}{सरल संबद्ध} है और $u \colon \Omega \to \R$
प्रसंवादी\index{प्रसंवादी फलन} ($\Delta u = u_{xx} + u_{yy} = 0$ के साथ $\mathcal C^2$)। तब किसी
\omterm{def:b3:holomorphic:holo}{समविश्लेषिक} $F$ के लिए $u =
\operatorname{Re}F$, जो किसी काल्पनिक अचर तक अद्वितीय
है। फलस्वरूप $u$ $\mathcal C^\infty$ है, माध्य मान गुण को संतुष्ट करता है, और
उच्चतम सिद्धांत का पालन करता है (अचर न हो तो कोई कड़ा आंतरिक
चरम-मान नहीं)।
\end{proposition}

\begin{proof}
$g = u_x - \iu u_y$ \omterm{def:b3:topology:continuity}{संतत} आंशिक अवकलजों के साथ \omterm{prop:b3:holomorphic:cauchyriemann}{कोशी--रीमान समीकरण} संतुष्ट करता है
($P = u_x$, $Q = -u_y$: $P_x = u_{xx} = -u_{yy} = Q_y$ और $P_y = u_{xy} = u_{yx} = -Q_x$): अतः $g$ \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है (\cref{prop:b3:holomorphic:cauchyriemann};
और $\R$-अवकलनीयता $\mathcal C^1$ से निकलती है)। मान लीजिए $F_0$ कोई
आद्यंतर है (सरल \omterm{def:b3:topology:connected}{संबद्धता}, \cref{def:b3:conformal:simplyconnected}); तब $\operatorname{Re}F_0$ का प्रवणक $(u_x, u_y)$ है
($F_0' = g$ $F_0$ के लिए कोशी--रीमान के माध्यम से ठीक वही खुलता है),
अतः $u - \operatorname{Re}F_0$ अचर है ($\Omega$ \omterm{def:b3:topology:connected}{संबद्ध}): अब $F = F_0 + c$ समायोजित कीजिए। गुण
\cref{thm:b3:holomorphic:maximum} और \cref{exo:b3:holomorphic:10} से स्थानांतरित हो जाते हैं (स्वयं $u$ पर लगाए गए
उच्चतम सिद्धांत के लिए वहीं की तरह $\eu^{F}$ का उपयोग कीजिए)।
\end{proof}

\begin{theorem}[प्वासों सूत्र; चक्र पर दिरिक्ले
समस्या]\label{thm:b3:conformal:poisson}
$0 \leq r < 1$ के लिए \emph{प्वासों नाभिक}\index{प्वासों नाभिक}
\[
P_r(\theta) = \sum_{n\in\Z}r^{\abs n}\eu^{\iu n\theta}
= \frac{1 - r^2}{1 - 2r\cos\theta + r^2} \;>\; 0 .
\]
परिभाषित कीजिए। मान लीजिए $g \colon \partial\mathbb D \to \R$ \omterm{def:b3:topology:continuity}{संतत} है, और $z = r\eu^{\iu\varphi} \in \mathbb D$ के लिए
\[
u(z) = \frac1{2\pi}\int_0^{2\pi}
P_r(\varphi - t)\,g(\eu^{\iu t})\,\dd t .
\]
रखिए। तब $u$ $\mathbb D$ पर प्रसंवादी है और परिसीमा मानों $g$ के
साथ $\bar{\mathbb D}$ तक संततता से बढ़ जाता है: और वही ऐसा अद्वितीय \omterm{prop:b3:conformal:harmonicholo}{प्रसंवादी
फलन} है।\index{दिरिक्ले समस्या}
\end{theorem}

\begin{proof}
\emph{नाभिक सर्वसमिकाएँ}: दो गुणोत्तर श्रेणियों का योग लेने पर
\[
\sum_{n\in\Z}r^{\abs n}\eu^{\iu n\theta}
= \operatorname{Re}\frac{1 + r\eu^{\iu\theta}}{1 -
r\eu^{\iu\theta}}
= \frac{1 - r^2}{\abs{1 - r\eu^{\iu\theta}}^2},
\]
जो प्रदर्शित भागफल है; धनात्मकता स्पष्ट है, और $\frac1{2\pi}\int_0^{2\pi}P_r = 1$ (केवल $n = 0$
बचता है)।

\emph{प्रसंवादिता}: $z = r\eu^{\iu\varphi}$ के साथ
\[
u(z) = \operatorname{Re}\biggl[\frac1{2\pi}\int_0^{2\pi}
\frac{\eu^{\iu t} + z}{\eu^{\iu t} - z}\,
g(\eu^{\iu t})\,\dd t\biggr],
\]
(कोष्ठक में रखे नाभिक का वास्तविक भाग $P_r(\varphi - t)$ है: परिकलित कीजिए),
और कोष्ठक $\mathbb D$ पर $z$ में \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है (\cref{exo:b3:holomorphic:7}): अतः $u$
किसी \omterm{def:b3:holomorphic:holo}{समविश्लेषिक फलन} का वास्तविक भाग है, इसलिए प्रसंवादी।

\emph{परिसीमा मान}: $r \to 1^-$ पर $P_r(\cdot)$ एक आसन्न तत्समक है: द्रव्यमान
$1$, और $\delta \leq
\abs\theta \leq \pi$ के लिए एकसमान रूप से $P_r(\theta) \leq \frac{1 - r^2}{1 -
2r\cos\delta + r^2} \to 0$। मानक विभाजन ($\eu^{\iu\varphi_0}$ के
निकट $g$ का \omterm{def:b3:topology:continuity}{सांतत्य}, अन्यत्र परिबद्धता) $r\eu^{\iu\varphi} \to
\eu^{\iu\varphi_0}$ पर $u(r\eu^{\iu\varphi}) \to
g(\eu^{\iu\varphi_0})$ दे देता
है, और वह भी परिसीमा बिंदु में एकसमान रूप से: अतः विस्तार \omterm{def:b3:topology:continuity}{संतत}
है। \emph{अद्वितीयता}: दो हलों का अंतर $\mathbb D$ पर प्रसंवादी है,
\omterm{def:b3:topology:interior}{संवृति} पर \omterm{def:b3:topology:continuity}{संतत}, और परिसीमा पर शून्य: अतः (अंतर के $\pm$ पर लगाए
गए) उच्चतम सिद्धांत से वह लुप्त हो जाता है।
\end{proof}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:5}
(क) दर्शाइए कि कोई \omterm{def:b3:conformal:conformal}{अनुरूप प्रतिचित्रण} $\C \to \mathbb D$ या $\C \to

(ख) निम्नलिखित में से कौन-से $\mathbb D$ के \omterm{def:b3:conformal:conformal}{अनुरूप तुल्य} हैं? \cref{thm:b3:conformal:rmt} से

\emph{(अंतिम दो के लिए: छिद्रित चक्र का कोई अनुरूप प्रतिबिंब

\end{exercise}

\section{समस्या: क्षेत्रफल प्रमेय और क्योबे की चतुर्थांश प्रमेय}

\begin{problem}[{सप्ताहांत समस्या --- किसी एकमानी प्रतिचित्रण को
कितना ढकना ही पड़ता है?}]\label{pb:b3:conformal:1}
\emph{एकमानी} फलन कोई \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} एकैकी प्रतिचित्रण होता है। चक्र
पर मानकीकृत एकमानी फलन,
\[
\mathcal S = \bigl\{f \in \mathcal H(\mathbb D) \text{
एकैकी},\ f(z) = z + a_2z^2 + a_3z^3 + \cdots\bigr\},
\]
कठोरता से बँधे होते हैं: हम बीबरबाख की असमिका $\abs{a_2} \leq 2$ सिद्ध करते
हैं और उससे \emph{क्योबे चतुर्थांश प्रमेय} निकालते हैं: किसी भी
$f \in \mathcal S$ का प्रतिबिंब चक्र $D(0, \frac14)$ को अंतर्विष्ट करता है --- अनुरूप
ज्यामिति का तीक्ष्ण सार्वत्रिक नियतांक।\index{क्योबे चतुर्थांश
प्रमेय}\index{क्षेत्रफल प्रमेय}

\medskip
\noindent\textbf{भाग I --- क्षेत्रफल प्रमेय।} मान लीजिए $g(w) = w
+ b_0 + \frac{b_1}w + \frac{b_2}{w^2} + \cdots$
$\{\abs w > 1\}$ पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} और \emph{एकैकी} है।
\begin{enumerate}
  \item $\rho > 1$ के लिए मान लीजिए $A_\rho$ \omterm{def:b3:topology:compact}{संहत} समुच्चय $K_\rho = \C\setminus
        g(\{\abs w > \rho\})$ का
        क्षेत्रफल (\omterm{def:b3:measure:lebesgueouter}{लेबेग माप}) है, अर्थात् चिकने जॉर्डन वक्र $g(C_\rho)$
        से घिरा क्षेत्र। दूसरे वर्ष के खंड की ग्रीन--रीमान प्रमेय
        के क्षेत्रफल सूत्र का उपयोग करते हुए --- घिरा क्षेत्रफल
        धनात्मक अभिविन्यास वाली परिसीमा पर $\frac1{2\iu}
        \oint\bar\zeta\,\dd\zeta$ है --- दर्शाइए
        कि
        \[
        A_\rho = \frac{1}{2\iu}\int_{C_\rho}
        \overline{g(w)}\,g'(w)\,\dd w
        = \pi\Bigl(\rho^2 -
        \sum_{n\geq1}n\,\abs{b_n}^2\rho^{-2n}\Bigr) :
        \]
        और $C_\rho$ पर $\bar g$ तथा $g'$ की \omterm{thm:b3:residues:laurent}{लोरां श्रेणी} प्रतिस्थापित
        करके पदशः समाकलन कीजिए (प्रसामान्य अभिसरण; केवल
        आवृत्ति-शून्य गुणनफल बचते हैं)।
  \item $\rho \to 1^+$ लीजिए और \emph{क्षेत्रफल प्रमेय} निष्कर्ष के रूप
        में निकालिए:
        \[
        \sum_{n\geq1}n\,\abs{b_n}^2 \;\leq\; 1 .
        \]
        विशेष रूप से $\abs{b_1} \leq 1$। $\abs{b_1}
        = 1$ कब होता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- बीबरबाख का $\abs{a_2} \leq
2$।} मान लीजिए $f = z + a_2z^2 + \cdots \in \mathcal S$।
\begin{enumerate}[resume]
  \item दर्शाइए कि $f(z^2)/z^2$ $\mathbb D$ पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} और शून्य-रहित है,
        और $\varphi(0) = 1$ के साथ कोई \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} वर्गमूल $\varphi$ स्वीकार करता
        है; $h(z) =
        z\varphi(z^2)$ रखिए, जिससे $h(z)^2 = f(z^2)$। दर्शाइए कि $h$ $\mathbb
        D$ पर
        प्रसार $h(z) = z + \frac{a_2}2z^3 +
        \cdots$ वाला एक \emph{विषम} एकमानी फलन है।
        \emph{(एकैकीपन: $h(z)^2 = h(z')^2$ से $z^2 = z'^2$ अनिवार्य हो जाता है; पूरा
        करने के लिए विषमता का उपयोग कीजिए।)}
  \item $\{\abs w > 1\}$ पर $g(w) = 1/h(1/w) = w -
        \frac{a_2}{2w} + \cdots$ पर क्षेत्रफल प्रमेय लगाइए (एकमानिता और
        प्रसार सत्यापित कीजिए), और $\abs{a_2} \leq 2$ निष्कर्ष निकालिए।
  \item दर्शाइए कि \emph{क्योबे फलन}
        \[
        k(z) = \frac{z}{(1 - z)^2} = \sum_{n\geq1}n\,z^n
        \]
        $\mathcal S$ में है, उसका $a_2 = 2$ है, और वह $\mathbb D$ को $\C\setminus\intoc{-\infty}{-\frac14}$ पर
        भेजता है \emph{($k = \frac14\bigl[\bigl(\frac{1+z}{1 -
        z}\bigr)^2 - 1\bigr]$ लिखिए और प्रतिबिंबों का अनुसरण
        कीजिए)}: अतः आगे आने वाली सभी असमिकाएँ तीक्ष्ण हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- चतुर्थांश प्रमेय।}
\begin{enumerate}[resume]
  \item मान लीजिए $f \in \mathcal S$ और $c \notin f(\mathbb D)$। दर्शाइए कि
        \[
        F(z) = \frac{c\,f(z)}{c - f(z)}
        \]
        $\mathcal S$ में है, और उसका द्वितीय गुणांक परिकलित कीजिए:
        $A_2 = a_2 + \frac1c$।
  \item $f$ और $F$ दोनों पर बीबरबाख लगाइए: निष्कर्ष निकालिए
        $\abs{\frac1c} \leq \abs{A_2} + \abs{a_2} \leq 4$, अर्थात् $\abs c \geq \frac14$। \emph{प्रत्येक छोड़े गए मान का
        मापांक $\geq \frac14$ होता है}: अतः $f(\mathbb D) \supseteq
        D(0, \frac14)$ --- क्योबे की चतुर्थांश
        प्रमेय। क्योबे फलन पर तीक्ष्णता जाँचिए।
  \item मात्रात्मक रीमान-प्रतिचित्रण आकलन निकालिए: यदि $\varphi \colon \Omega \to \mathbb D$
        $z_0$ पर \cref{thm:b3:conformal:rmt} का रीमान प्रतिचित्रण हो, तो
        \[
        \frac{d\bigl(z_0, \partial\Omega\bigr)}{4}
        \;\leq\; \frac1{\varphi'(z_0)} \;\leq\;
        4\,d\bigl(z_0, \partial\Omega\bigr)
        \]
        --- कम से कम बाईं असमिका उपयुक्त रूप से मानकीकृत $\varphi^{-1}$ पर
        क्योबे लगाकर सिद्ध कीजिए, और दाईं चक्र $D(z_0, d) \subseteq \Omega$ पर $\varphi$ पर
        श्वार्ट्स लगाकर।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- परिप्रेक्ष्य।}
\begin{enumerate}[resume]
  \item बीबरबाख ने (1916) सभी $n$ के लिए $\abs{a_n} \leq n$ की परिकल्पना की
        थी, जिसमें समता केवल क्योबे फलन के घूर्णनों के लिए; और
        दे ब्रांज ने 1985 में उसे सिद्ध कर दिया। क्योबे फलन और
        उसके घूर्णनों $\eu^{-\iu\theta}k(\eu^{\iu\theta}z)$ के लिए इस परिकल्पना को हाथ से
        सत्यापित कीजिए। फिर प्रश्न 4 के प्रसार को एक पद और आगे
        बढ़ाइए: $h(z) = z + \frac{a_2}2z^3 + c_5z^5 + \cdots$ लिखकर दर्शाइए कि $c_5 = \frac{a_3}2 - \frac{a_2^2}8$ और
        \[
        g(w) = w - \frac{a_2}{2}\,w^{-1} +
        \Bigl(\frac{3a_2^2}8 - \frac{a_3}2\Bigr)w^{-3} +
        \cdots ,
        \]
        अतः क्षेत्रफल प्रमेय परिष्कृत असमिका $\bigl|\frac{a_2}2\bigr|^2 + 3\bigl|\frac{3a_2^2}8 -
        \frac{a_3}2\bigr|^2 \leq 1$ दे देती है।
        इसे क्योबे फलन पर जाँचिए ($a_2 = 2$, $a_3 = 3$)।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- विरूपण प्रमेय।} बीबरबाख की असमिका,
जिसे स्वसमाकारिताओं द्वारा चक्र भर घुमाया गया हो, $f'$ को
सर्वत्र नियंत्रित कर देती है। $f \in \mathcal
S$ स्थिर कीजिए।
\begin{enumerate}[resume]
  \item (क्योबे रूपांतर) $z_0 \in \mathbb D$ के लिए $\varphi(z) = \frac{z + z_0}{1 + \bar z_0z}$ लीजिए, जो $\varphi(0) = z_0$ वाली
        चक्र स्वसमाकारिता (\cref{thm:b3:conformal:autdisc}) है। दर्शाइए कि
        \[
        F(z) = \frac{f(\varphi(z)) - f(z_0)}
        {f'(z_0)\,\bigl(1 - \abs{z_0}^2\bigr)}
        \]
        $\mathcal S$ में है \emph{(एकमानिता विरासत में मिलती है; $\varphi'(0) = 1 - \abs{z_0}^2$
        परिकलित कीजिए और मानकीकरण जाँचिए; स्मरण कीजिए कि एकैकी
        $f$ के लिए $f' \neq 0$, \cref{def:b3:conformal:conformal})}।
  \item $F$ का द्वितीय गुणांक $A_2 = \frac12F''(0)$ परिकलित कीजिए:
        \[
        A_2 = \frac12\Bigl[\bigl(1 - \abs{z_0}^2\bigr)
        \frac{f''(z_0)}{f'(z_0)} - 2\bar z_0\Bigr] ,
        \]
        और बीबरबाख (प्रश्न 4) से $z =
        r\eu^{\iu\theta}$ के लिए \emph{मूल असमिका}
        निकालिए:
        \[
        \Bigl|\,z\,\frac{f''(z)}{f'(z)} -
        \frac{2r^2}{1 - r^2}\Bigr|
        \;\leq\; \frac{4r}{1 - r^2} .
        \]
  \item वास्तविक भाग निकालिए:
        \[
        \frac{2r^2 - 4r}{1 - r^2} \;\leq\;
        \operatorname{Re}\Bigl(z\,\frac{f''(z)}{f'(z)}\Bigr)
        \;\leq\; \frac{2r^2 + 4r}{1 - r^2} .
        \]
  \item दर्शाइए कि $z = t\eu^{\iu\theta}$ पर $\frac{\dd}{\dd t}\log\bigl|
        f'(t\eu^{\iu\theta})\bigr| = \frac1t
        \operatorname{Re}\bigl(z\frac{f''(z)}{f'(z)}\bigr)$ \emph{(किसी वास्तविक चर के
        अलुप्त $\mathcal C^1$ फलन $g$ के लिए $\frac{\dd}{\dd t}\log\abs g =
        \operatorname{Re}(g'/g)$)}, और किरण के अनुदिश
        प्रश्न 12 के परिबंधों का समाकलन करके \emph{विरूपण प्रमेय}
        प्राप्त कीजिए:
        \[
        \frac{1 - r}{(1 + r)^3} \;\leq\; \abs{f'(z)}
        \;\leq\; \frac{1 + r}{(1 - r)^3},
        \qquad \abs z = r .
        \]
  \item \emph{वृद्धि प्रमेय} निकालिए:
        \[
        \frac{r}{(1 + r)^2} \;\leq\; \abs{f(z)} \;\leq\;
        \frac{r}{(1 - r)^2},
        \qquad \abs z = r
        \]
        \emph{(ऊपरी परिबंध: रेखाखंड $[0, z]$ पर $f'$ का समाकलन
        कीजिए; निचला परिबंध: यदि $\abs{f(z)} < \frac14$ हो, तो प्रश्न 7 से
        रेखाखंड $[0, f(z)]$ $f(\mathbb D)$ में होता है; उसे $f^{-1}$ से पीछे खींचिए
        --- जो \cref{cor:b3:residues:openmapping} से \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है --- और $\abs{\dd\zeta} \geq
        \dd\abs\zeta$ का उपयोग करते
        हुए $\abs{f(z)} = \int_\gamma\abs{f'(\zeta)}
        \,\abs{\dd\zeta} \geq \int_0^r\frac{1 - t}{(1 +
        t)^3}\,\dd t$ को परिबद्ध कीजिए)}।
  \item सत्यापित कीजिए कि क्योबे फलन चारों परिबंधों में समता
        प्राप्त कर लेता है, ऊपरी वालों के लिए $z = r$ पर और निचले
        वालों के लिए $z = -r$ पर: अर्थात् $k$ एक साथ $\mathcal S$ का सबसे
        अधिक फैलाने वाला और, प्रतिध्रुव पर, सबसे अधिक सिकोड़ने
        वाला सदस्य है।
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- चरम कठोरता।} अब तक की प्रत्येक
असमिका में समता क्योबे फलन को घूर्णन तक पहचान देती है। हम इसे
सिद्ध करते हैं, फिर फसल काटते हैं।
\begin{enumerate}[resume]
  \item मान लीजिए $f \in \mathcal S$ के लिए $\abs{a_2} = 2$। प्रश्न 2--4 में समता का
        पीछा कीजिए: क्षेत्रफल प्रमेय $g(w) = w + b_0 + \eu^{\iu\alpha}/w$ अनिवार्य कर देती है;
        $h$ की विषमता $g$ को विषम बना देती है, अतः $b_0 = 0$;
        उलटकर $h$ ढूँढ़िए, फिर $f$, और निष्कर्ष निकालिए कि
        \[
        f(z) = \eu^{-\iu\theta}k\bigl(\eu^{\iu\theta}z\bigr)
        \quad\text{जहाँ } \eu^{\iu\theta} =
        -\eu^{\iu\alpha} :
        \]
        अर्थात् $\abs{a_2} = 2$ वाले $\mathcal S$ के एकमात्र सदस्य क्योबे फलन के
        घूर्णन हैं।
  \item दर्शाइए कि यदि $f \in \mathcal S$ ठीक $\abs c = \frac14$ वाला कोई मान $c$ छोड़
        दे, तो $f$ क्योबे फलन का कोई घूर्णन है, और $c =
        -\eu^{-\iu\theta}/4$ पहचानिए
        \emph{(प्रश्न 7 की शृंखला $4 = \abs{1/c} = \abs{A_2 - a_2}
        \leq \abs{A_2} + \abs{a_2} \leq 4$ में समता का पीछा कीजिए)}:
        अर्थात् चतुर्थांश प्रमेय का नियतांक केवल उसी चरम कुल से
        प्राप्त होता है।
  \item (सस्ते में गुणांक) वृद्धि प्रमेय को वृत्त $\abs z = 1 - \frac1n$ पर कोशी
        आकलनों (\cref{thm:b3:holomorphic:analytic}) के साथ मिलाकर सिद्ध कीजिए
        \[
        \abs{a_n} \;\leq\; \eu\,n^2 \qquad (n \geq 2) .
        \]
        (दे ब्रांज, 1985: $\abs{a_n} \leq n$; गुणक $\eu n$ प्रारंभिक औजारों की
        कीमत है।)
  \item (उपचक्रों का आच्छादन) दर्शाइए कि प्रत्येक $0 < r <
        1$ के लिए
        \[
        f\bigl(D(0, r)\bigr) \supseteq
        D\Bigl(0, \frac{r}{(1 + r)^2}\Bigr),
        \]
        जो क्योबे फलन के लिए तीक्ष्ण है, और $r \to 1^-$ के रूप में
        चतुर्थांश प्रमेय पुनः प्राप्त कीजिए। \emph{(विवृत
        प्रतिबिंब $f(D(0,r))$ के परिसीमा बिंदु $f(\partial D(0,r))$ पर होते हैं, अतः
        प्रश्न 14 से उनका मापांक $\geq
        \frac{r}{(1+r)^2}$ होता है; और $0$ से किसी
        छोटे मापांक वाले छूटे बिंदु तक का कोई रेखाखंड उस परिसीमा
        को पार करता।)}
  \item (प्रत्येक बिंदु पर क्योबे) मान लीजिए $f$ $\mathbb D$ पर एकमानी
        है, और आवश्यक नहीं कि मानकीकृत भी, तथा $z_0
        \in \mathbb D$। सिद्ध
        कीजिए
        \[
        \tfrac14\bigl(1 - \abs{z_0}^2\bigr)\abs{f'(z_0)}
        \;\leq\;
        d\bigl(f(z_0), \partial f(\mathbb D)\bigr)
        \;\leq\;
        \bigl(1 - \abs{z_0}^2\bigr)\abs{f'(z_0)}
        \]
        \emph{(बाईं ओर: प्रश्न 10 के क्योबे रूपांतर पर चतुर्थांश
        प्रमेय; दाईं ओर: $\psi^{-1}\circ\hat g$ पर श्वार्ट्स (\cref{thm:b3:conformal:schwarz}), जहाँ $\hat g(w) =
        f^{-1}\bigl(f(z_0) + dw\bigr)$,
        $d$ दूरी है, और $\psi$ $0$ को $z_0$ पर भेजने वाली कोई
        चक्र स्वसमाकारिता)}। $\partial f(\mathbb D)$ अरिक्त क्यों है?
  \item प्रश्न 20 को $z_0 = r \in
        \intoo01$ पर $f = k$ पर जाँचिए: $d\bigl(k(r), \partial k(\mathbb
        D)\bigr) = \frac{(1+r)^2}{4(1-r)^2}$ परिकलित
        कीजिए और सत्यापित कीजिए कि बाईं असमिका एक \emph{समता} है:
        अर्थात् क्योबे फलन $\intoo01$ के प्रत्येक बिंदु पर अपनी ही
        प्रमेय को संतृप्त कर देता है।
  \item (सार) दस पंक्तियों में: क्षेत्रफल प्रमेय के मूल में
        कौन-सा एक सिद्धांत है, और बीबरबाख, चतुर्थांश प्रमेय,
        विरूपण, वृद्धि और आच्छादन उससे कैसे निकलते हैं? \cref{thm:b3:conformal:schwarz} के
        श्वार्ट्स--पिक संसार से तुलना कीजिए: दोनों में एक ही
        आंतरिक असमिका पूरी ज्यामिति को कठोर कर देती है, और चरम
        अवयव घूर्णन तक अद्वितीय होते हैं।

\end{enumerate}
\medskip
\noindent\textbf{भाग VII --- संहतता, प्रतिलोम, और एक वास्तविकता
जाँच।}
\begin{enumerate}[resume]
  \item दर्शाइए कि वर्ग $\mathcal S$ स्थानीय एकसमान अभिसरण के लिए
        \emph{\omterm{def:b3:topology:compact}{संहत}} है: वृद्धि प्रमेय से वह स्थानीय परिबद्ध है,
        अतः प्रसामान्य (\cref{thm:b3:conformal:montel}); और $\mathcal S$ के सदस्यों की कोई
        स्थानीय एकसमान सीमा पुनः $\mathcal S$ में होती है
        \emph{(मानकीकरण अवकलजों के वाइरश्ट्रास अभिसरण से सीमा तक
        चले जाते हैं; और एकैकीपन हुर्विट्स \cref{exo:b3:residues:8} से बचा रहता है,
        क्योंकि सीमा अचरेतर है)}। बीबरबाख जैसी चरम समस्याओं के
        लिए यह क्यों महत्त्वपूर्ण है?
  \item $f \in \mathcal S$ के लिए $0$ के निकट परिभाषित $g = f^{-1}$ लीजिए। $g(w) = w - a_2w^2 + O(w^3)$
        दर्शाइए, अतः प्रतिलोम का द्वितीय गुणांक भी वही तीक्ष्ण
        परिबंध $\abs{A_2} = \abs{a_2} \leq 2$ मानता है, जिसमें समता ठीक घूर्णित क्योबे
        फलनों के लिए होती है।
  \item निर्धारित कीजिए कि किन $a \in \C$ के लिए बहुपद $f(z)
        = z + az^2$ $\mathcal S$ में
        है: दर्शाइए कि $f$ $\mathbb D$ पर एकैकी है तभी और केवल तभी जब
        $\abs a \leq \frac12$ \emph{($f(z_1) - f(z_2)$ का गुणनखंडन कीजिए)}। निष्कर्ष निकालिए:
        द्वितीय घात के बहुपदों के लिए वास्तविक गुणांक परिबंध
        $\frac12$ है, जो बीबरबाख के $2$ से चार गुना छोटा है ---
        अर्थात् $\mathcal S$ के चरम अवयव सचमुच अबीजीय वस्तुएँ हैं, और
        कोई बहुपद उनके पास तक नहीं पहुँचता।
\end{enumerate}
\end{problem}
