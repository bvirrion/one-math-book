\chapter{गुणन माप, फुबिनी, चर परिवर्तन}
\label{ch:b3:product}

एक-विमीय लेबेग सिद्धांत दो प्रमेयों के माध्यम से बहु-विमीय कलन बन
जाता है। \emph{टोनेली--फुबिनी} कहती है कि गुणनों पर समाकल
पुनरावृत्त समाकल होते हैं --- अर्थात् काटना वैध है, किसी भी क्रम
में, और वह भी ऐसी परिकल्पनाओं के अंतर्गत जिन्हें सचमुच जाँचा जा
सकता है। \emph{चर परिवर्तन सूत्र} समाकलों को $\mathcal C^1$ अवकल समरूपताओं के
अनुदिश ले जाता है, जिसमें याकोबीय सारणिक आयतन की विनिमय दर का काम
करता है; हम उसे पूरी तरह सिद्ध करते हैं, और आरंभ रैखिक स्थिति से
करते हैं, जहाँ वह बता देता है कि सारणिक \emph{है} क्या। अनुप्रयोग
झरने की तरह गिरते हैं: परत-केक सूत्र, संवलन, \omterm{ex:b3:product:polar}{ध्रुवीय निर्देशांक},
$n$-गोले का आयतन --- और सप्ताहांत समस्या में, ईमानदार त्रुटि
विश्लेषण सहित स्टर्लिंग का सूत्र।

\section{गुणन $\sigma$-बीजगणित और गुणन माप}

\begin{definition}\label{def:b3:product:sigma}
\omterm{def:b3:lebesgue:measurable}{मापनीय} समष्टियों $(X, \mathcal A)$, $(Y, \mathcal B)$ के लिए $X \times
Y$ पर \emph{गुणन
$\sigma$-बीजगणित}\index{गुणन $\sigma$-बीजगणित} $\mathcal A \otimes \mathcal B$ \emph{आयतों} $A \times B$
($A \in
\mathcal A$, $B \in \mathcal B$) से जनित होता है --- जो एक $\pi$-प्रणाली है। $E
\subseteq X\times Y$ और
$x \in X$ के लिए \emph{परिच्छेद} $E_x
= \{y : (x,y) \in E\}$ है; और गुणन पर किसी फलन $f$ के
लिए $f_x = f(x, \cdot)$।
\end{definition}

\begin{proposition}\label{prop:b3:product:sections}
(क) यदि $E \in \mathcal A\otimes\mathcal B$ हो, तो प्रत्येक परिच्छेद $E_x
\in \mathcal B$ (और सममित रूप से भी);
और यदि $f$ $\mathcal
A\otimes\mathcal B$-मापनीय हो, तो प्रत्येक $f_x$ $\mathcal
B$-मापनीय है।
(ख) $\mathcal B(\R^m)\otimes\mathcal B(\R^n) = \mathcal
B(\R^{m+n})$।
\end{proposition}

\begin{proof}
(क) अच्छे समुच्चय: $\{E : E_x \in \mathcal B\ \forall x\}$ एक $\sigma$-बीजगणित है (परिच्छेद पूरकों और
गणनीय सम्मिलनों के साथ क्रमविनिमेय हैं) जिसमें आयत हैं। $f$ के
लिए: $(f_x)^{-1}(B) = (f^{-1}(B))_x$।
(ख) ($\subseteq$) बोरेल समुच्चयों के आयत: इतना ही पर्याप्त है कि
विवृत$\times$विवृत बक्से $\R^{m+n}$ में बोरेल हों (वे विवृत हैं) और यह कि
व्यापक बोरेल आयत सीमाएँ हों --- फिर अच्छे समुच्चय: $\{A : A\times\R^n \in \mathcal B(\R^{m+n})\}$ विवृत
समुच्चयों को अंतर्विष्ट करने वाला $\sigma$-बीजगणित है; ऐसे दो का
प्रतिच्छेदन लीजिए।
($\supseteq$) $\R^{m+n}$ का प्रत्येक \omterm{def:b3:topology:topology}{विवृत समुच्चय} परिमेय विवृत बक्सों $U\times V$ का
गणनीय सम्मिलन है: अतः गुणन $\sigma$-बीजगणित में अंतर्विष्ट है।
\end{proof}

\begin{theorem}[गुणन माप]\label{thm:b3:product:existence}
मान लीजिए $(X, \mathcal A, \mu)$ और $(Y, \mathcal B, \nu)$ \emph{$\sigma$-परिमित} हैं। प्रत्येक $E \in \mathcal
A\otimes\mathcal B$ के
लिए फलन $x \mapsto \nu(E_x)$ \omterm{def:b3:lebesgue:measurable}{मापनीय} है, और\index{गुणन माप}
\[
(\mu\otimes\nu)(E) = \int_X \nu(E_x)\,\dd\mu(x)
\]
$\mathcal A\otimes\mathcal B$ पर $(\mu\otimes\nu)(A\times B) = \mu(A)\nu(B)$ वाला अद्वितीय \omterm{def:b3:measure:measure}{माप} परिभाषित करता है। वह $\sigma$-परिमित
और सममित है: $\nu$ के सापेक्ष $x$-परिच्छेदों का समाकलन करने पर
वही \omterm{def:b3:measure:measure}{माप} मिलता है।
\end{theorem}

\begin{proof}
\emph{$x \mapsto \nu(E_x)$ की \omterm{def:b3:lebesgue:measurable}{मापनीयता}।} पहले मान लीजिए $\nu$ परिमित है। जिन $E$
के लिए यह प्रतिचित्रण \omterm{def:b3:lebesgue:measurable}{मापनीय} है उनका वर्ग $\mathcal D$ आयतों को अंतर्विष्ट
करता है ($\nu((A\times B)_x) =
\nu(B)\mathbf 1_A(x)$) और एक $\lambda$-प्रणाली है: $\mathcal D$ में $E
\subseteq F$ के लिए $\nu((F\setminus E)_x) =
\nu(F_x) - \nu(E_x)$
(परिमितता); और $E_n \uparrow E$ के लिए $\nu((E_n)_x) \uparrow \nu(E_x)$ (नीचे से \omterm{def:b3:topology:continuity}{सांतत्य}), तथा \omterm{def:b3:lebesgue:measurable}{मापनीय}
फलनों की एकदिष्ट सीमाएँ \omterm{def:b3:lebesgue:measurable}{मापनीय} होती हैं। आयत एक $\pi$-प्रणाली
बनाते हैं: अतः डिन्किन (\cref{thm:b3:measure:dynkin}) से $\mathcal D = \mathcal
A\otimes\mathcal B$। यदि $\nu$ $\sigma$-परिमित हो,
तो $Y =
\bigcup Y_k$, $Y_k \uparrow$, $\nu(Y_k) < \infty$ लिखिए: तब परिमित $\nu_k = \nu(\cdot\cap Y_k)$ के साथ $\nu(E_x) =
\lim_k\nu_k(E_x)$।

\emph{\omterm{def:b3:measure:measure}{माप}।} $E \mapsto
\int\nu(E_x)\dd\mu$ की $\sigma$-योज्यता \cref{cor:b3:lebesgue:additivity} से निकलती है (असंयुक्त
समुच्चयों के परिच्छेद असंयुक्त होते हैं)। आयतों पर वह $\mu(A)\nu(B)$ दे
देती है। \emph{अद्वितीयता}: दो उम्मीदवार आयतों की $\pi$-प्रणाली
पर सहमत हैं; और $\sigma$-परिमितता परिमित \omterm{def:b3:measure:measure}{माप} वाले आयत $X_k\times
Y_k \uparrow X\times Y$ दे देती है:
\cref{thm:b3:measure:uniqueness}। सममिति: दूसरे क्रम वाली रचना भी एक \omterm{def:b3:measure:measure}{माप} है जो आयतों पर सहमत
है --- और वह अद्वितीय है, अतः वही है।
\end{proof}

\begin{definition}\label{def:b3:product:lebesgue}
\emph{$\R^d$ पर लेबेग माप}\index{$\R^d$ पर लेबेग माप} $\lambda_d = \lambda\otimes\cdots\otimes\lambda$ है ($d$
गुणनखंड; रचना की साहचर्यता बक्सों पर जाँची जाती है और अद्वितीयता
से फैल जाती है)। यह वह अद्वितीय बोरेल \omterm{def:b3:measure:measure}{माप} है जो प्रत्येक बक्से
$\prod\intoc{a_i}{b_i}$ को उसका आयतन $\prod(b_i - a_i)$ देता है; वह स्थानांतरण-अपरिवर्ती और
$\sigma$-परिमित है, तथा कैराथियोडोरी \omterm{thm:b3:complete:completion}{पूर्णन} के बाद \omterm{def:b3:complete:complete}{पूर्ण} --- \omterm{def:b3:complete:complete}{पूर्ण}
किए गए \omterm{def:b3:measure:measure}{माप} को हम $\lambda_d$ लिखते हैं और उसी के अनुसार समाकलन करते हैं।
\end{definition}

\section{टोनेली और फुबिनी}

\begin{theorem}[टोनेली]\label{thm:b3:product:tonelli}
$\mu, \nu$ $\sigma$-परिमित हैं, $f \colon X\times Y \to [0,
+\infty]$ \omterm{def:b3:lebesgue:measurable}{मापनीय} है। तब $x \mapsto \int_Y f_x\,\dd\nu$ \omterm{def:b3:lebesgue:measurable}{मापनीय} है
और\index{टोनेली की प्रमेय}
\[
\int_{X\times Y}f\,\dd(\mu\otimes\nu)
= \int_X\Bigl(\int_Y f(x,y)\,\dd\nu(y)\Bigr)\dd\mu(x)
= \int_Y\Bigl(\int_X f(x,y)\,\dd\mu(x)\Bigr)\dd\nu(y).
\]
\end{theorem}

\begin{proof}
वही मानक मशीन। $f = \mathbf 1_E$ के लिए यह \cref{thm:b3:product:existence} है (और उसका सममित रूप)।
रैखिकता से यह सरल $f \geq 0$ के लिए लागू होता है। व्यापक $f \geq
0$ के लिए:
सरल $s_n \nearrow f$ लीजिए (\cref{thm:b3:lebesgue:approximation}); तब प्रत्येक $x$ के लिए $\int_Y(s_n)_x
\dd\nu \nearrow \int_Y f_x\dd\nu$ ($Y$
में एकदिष्ट अभिसरण प्रमेय), अतः बाएँ पद $X$ में एकदिष्ट अभिसरण
प्रमेय से अभिसरित होते हैं, जबकि गुणन पर एकदिष्ट अभिसरण प्रमेय से
$\int s_n\,\dd(\mu\otimes\nu) \nearrow \int f$।
\end{proof}

\begin{theorem}[फुबिनी]\label{thm:b3:product:fubini}
$\mu, \nu$ $\sigma$-परिमित हैं, $f \in L^1(\mu\otimes\nu)$। तब $\mu$-लगभग प्रत्येक $x$ के
लिए परिच्छेद $f_x$ $\nu$-समाकलनीय है, लगभग-सर्वत्र-परिभाषित फलन
$x \mapsto \int f_x\dd\nu$ \omterm{def:b3:lebesgue:l1}{समाकलनीय} है, और दोनों पुनरावृत्त समाकल $\int f\,\dd(\mu\otimes\nu)$ के बराबर
होते हैं।\index{फुबिनी की प्रमेय}
\end{theorem}

\begin{proof}
$\abs f$ पर लगाया गया टोनेली दिखा देता है कि $\varphi(x) = \int\abs{f_x}
\dd\nu$ का समाकल परिमित
है, अतः वह लगभग सर्वत्र परिमित है: लगभग प्रत्येक $x$ के लिए
$f_x \in
L^1(\nu)$। $f = f^+ - f^-$ को बाँटिए (वास्तविक स्थिति; सम्मिश्र स्थिति घटकों
से): टोनेली $f^\pm$ के प्रत्येक पुनरावृत्त समाकल को $\int f^\pm\dd(\mu\otimes\nu) <
\infty$ के रूप में
परिकलित कर देता है, और लगभग-सर्वत्र-परिभाषित अंतर का समाकल उसी
अंतर के बराबर आता है। दूसरे क्रम के लिए सममित रूप से।
\end{proof}

\begin{method}\label{met:b3:product:fubini}
दो समाकलों की (या किसी समाकल और योग की, या दो योगों की) अदला-बदली
करने के लिए: यदि समाकल्य \emph{अऋणात्मक} हो, तो स्वतंत्र रूप से
अदला-बदली कीजिए (टोनेली)। अन्यथा पहले $\abs f$ पर टोनेली लगाइए, और
वह भी उस क्रम में जिसमें आकलन आसान हो; यदि परिणाम परिमित निकले, तो
फुबिनी अदला-बदली को वैध कर देती है। $\abs f$ की जाँच कभी मत छोड़िए:
\cref{exo:b3:product:4} के समाकल्य के दोनों पुनरावृत्त समाकलों के \emph{मान भिन्न}
हैं।
\end{method}

\begin{proposition}[परत-केक]\label{prop:b3:product:layercake}
$(X, \mathcal A, \mu)$ $\sigma$-परिमित पर \omterm{def:b3:lebesgue:measurable}{मापनीय} $f \geq 0$ के लिए:\index{परत-केक सूत्र}
\[
\int_X f\,\dd\mu = \int_0^{+\infty}\mu(\{f > t\})\,\dd t,
\qquad
\int_X f^p\,\dd\mu = p\int_0^{+\infty}t^{p-1}\mu(\{f >
t\})\,\dd t \quad (p \geq 1).
\]
\end{proposition}

\begin{proof}
$X
\times \intoo0{+\infty}$ पर $\mathbf 1_{\{(x,t) : 0 < t < f(x)\}}$ पर टोनेली लगाइए (यह \omterm{def:b3:lebesgue:measurable}{मापनीय} है: वह $\{(x,t): f(x) - t
> 0\}\cap\{t > 0\}$ है, अर्थात्
\omterm{def:b3:lebesgue:measurable}{मापनीय} $(x,t)\mapsto f(x) - t$ का बोरेल प्रकार का संचय): पहले $t$ में समाकलन करने
पर $\int f\,\dd\mu$ मिलता है; और पहले $x$ में करने पर $\int_0^\infty\mu(f >
t)\dd t$। $f^p$ के लिए:
$\int\mu(f^p > t)
\dd t$ में $t = s^p$ प्रतिस्थापित कीजिए, अर्थात् पहला सूत्र $f^p$ पर
लगाइए और एक-विमीय समाकल में चर बदलिए ($\{f^p > s^p\} =
\{f > s\}$)।
\end{proof}

\begin{theorem}[$L^1$ पर संवलन]\label{thm:b3:product:convolution}
$f, g \in L^1(\R^d, \lambda_d)$ के लिए समाकल
\[
(f * g)(x) = \int_{\R^d} f(x - y)\,g(y)\,\dd y
\]
लगभग प्रत्येक $x$ के लिए निरपेक्षतः अभिसरित होता है, $\norm{f*g}_1 \leq \norm f_1\norm g_1$ वाला
$f * g \in
L^1(\R^d)$ परिभाषित करता है, और $*$ क्रमविनिमेय तथा साहचर्य
है।\index{संवलन}
\end{theorem}

\begin{proof}
$(x, y) \mapsto f(x-y)g(y)$ \omterm{def:b3:lebesgue:measurable}{मापनीय} है ($(x,y)\mapsto x -
y$ \omterm{def:b3:topology:continuity}{संतत} है; संयोजन और गुणन कीजिए)। टोनेली:
\[
\int\!\!\int \abs{f(x-y)}\abs{g(y)}\,\dd y\,\dd x
= \int\abs{g(y)}\Bigl(\int\abs{f(x - y)}\dd x\Bigr)\dd y
= \norm f_1\norm g_1 < \infty
\]
(भीतरी समाकल में $\lambda_d$ की स्थानांतरण अपरिवर्तिता)। अतः द्विक
समाकल परिमित है; और फुबिनी लगभग सर्वत्र निरपेक्ष अभिसरण तथा मानक
परिबंध $\norm{f*g}_1 \leq \norm
f_1\norm g_1$ दे देती है। क्रमविनिमेयता: $y \mapsto x - y$ प्रतिस्थापित कीजिए
(स्थानांतरण और परावर्तन अपरिवर्तिता --- परावर्तन अपरिवर्तिता
बक्सों पर लागू होती है, अतः अद्वितीयता से सर्वत्र)। साहचर्यता:
त्रिक समाकल पर टोनेली--फुबिनी।
\end{proof}

\section{चर परिवर्तन}

\begin{theorem}[रैखिक चर परिवर्तन]
\label{thm:b3:product:linearchange}
$T \in GL_d(\R)$ और $A \in \mathcal B(\R^d)$ के लिए: $\lambda_d(T(A)) = \abs{\det T}\,\lambda_d(A)$; फलस्वरूप $f \geq 0$ अथवा \omterm{def:b3:lebesgue:l1}{समाकलनीय} के लिए
$\int f(y)\dd y = \abs{\det T}\int f(Tx)\,\dd x$।\index{रैखिक चर परिवर्तन}
\end{theorem}

\begin{proof}
\omterm{def:b3:measure:measure}{माप} $\mu_T(A) = \lambda_d(T(A))$ एक बोरेल \omterm{def:b3:measure:measure}{माप} है (समस्थितिकताएँ बोरेल समुच्चय सुरक्षित
रखती हैं, \cref{pb:b3:measure:1}), स्थानांतरण-अपरिवर्ती है ($T(A + x) = T(A) + Tx$), और इकाई बक्से
पर परिमित: अतः \omterm{def:b3:measure:lebesgueouter}{लेबेग माप} के अभिलक्षण (\cref{exo:b3:measure:6}, जिसकी उपपत्ति
द्विआधारी घनों के साथ $\R^d$ में शब्दशः काम करती है) से $c(T)
= \lambda_d(T(\intco01^d))$ के
साथ $\mu_T = c(T)\lambda_d$। प्रतिचित्रण $T \mapsto c(T)$ गुणनात्मक है (संयोजन से $c(ST) = c(S)c(T)$), अतः
$GL_d$ के जनकों पर $c$ परिकलित करना पर्याप्त है: अर्थात् प्रारंभिक
आव्यूहों पर। विकर्ण $\operatorname{diag}(a, 1, \dots, 1)$: यह इकाई घन को आयतन $\abs a$ वाले बक्से पर
भेजता है: $c = \abs a =
\abs\det$। निर्देशांकों का स्थानांतरण: यह घन का क्रमचय कर
देता है: $c
= 1 = \abs\det$। अपरूपण $T(x) = x + \alpha
x_2e_1$: इकाई घन का प्रतिबिंब एक तिरछा प्रिज्म
है; टोनेली से उसका \omterm{def:b3:measure:measure}{माप} $\int\lambda_1(\text{परिच्छेद})\dd x_2
\cdots \dd x_d = 1$ है, क्योंकि प्रत्येक $x_1$-परिच्छेद
लंबाई $1$ का अंतराल है: $c = 1 = \abs{\det}$। प्रत्येक व्युत्क्रमणीय आव्यूह
इन्हीं का गुणनफल है (गाउस निराकरण), और $c$ तथा $\abs\det$ दोनों
गुणनात्मक हैं: अतः $c(T) = \abs{\det T}$। समाकल सूत्र मानक मशीन से निकलता है
(सूचक, सरल, एकदिष्ट अभिसरण प्रमेय)।
\end{proof}

\begin{theorem}[चर परिवर्तन]\label{thm:b3:product:changeofvar}
मान लीजिए $U, V \subseteq \R^d$ विवृत हैं और $\Phi \colon U \to V$ एक $\mathcal C^1$ अवकल समरूपता है।
प्रत्येक \omterm{def:b3:lebesgue:measurable}{मापनीय} $f \colon V
\to [0, +\infty]$ (अथवा $f \in L^1(V)$) के लिए:\index{चर परिवर्तन
सूत्र}
\[
\int_V f(y)\,\dd y = \int_U
f\bigl(\Phi(x)\bigr)\,\abs{\det D\Phi(x)}\,\dd x .
\]
\end{theorem}

\begin{proof}
$J(x) = \abs{\det D\Phi(x)}$ लिखिए। उपपत्ति का हृदय असमिका
\[
\lambda_d\bigl(\Phi(A)\bigr) \leq \int_A J\,\dd\lambda_d
\qquad\text{प्रत्येक बोरेल } A \subseteq U \text{ के लिए};
\tag{$*$}
\]
है। नीचे का चरण 4 $(*)$ को --- जो $\Phi$ और $\Phi^{-1}$ दोनों पर लगाया
जाता है --- प्रमेय की समता तक उठा देता है। ध्यान दीजिए कि $\Phi^{-1}$
स्वयं याकोबीय $\abs{\det D\Phi^{-1}(y)} = J(\Phi^{-1}y)^{-1}$ वाली एक $\mathcal C^1$ अवकल समरूपता है ($\Phi\circ\Phi^{-1} = \mathrm{id}$ पर
शृंखला नियम)।

\emph{चरण 1: विरूपण गुणक सहित घनों के लिए $(*)$।} \omterm{ex:b3:groups:actions}{केंद्र} $x_0$ और
भुजा $2r$ वाला कोई संवृत घन $Q \subseteq U$ (उच्चतम-मानक गोला) स्थिर
कीजिए। दावा: प्रत्येक $\varepsilon > 0$ के लिए, यदि $\Phi$ $Q$ पर अवकलनीय हो
और $Q$ पर $\norm{D\Phi(x) - D\Phi(x_0)} \leq \varepsilon$ हो (उच्चतम-मानक के लिए \omterm{def:b3:banach:operator}{संकारक मानक}), तो
\[
\Phi(Q) \subseteq \Phi(x_0) + D\Phi(x_0)\Bigl(\,\bigl(1 +
\varepsilon\norm{D\Phi(x_0)^{-1}}\bigr)\,(Q - x_0)\Bigr),
\]
क्योंकि $x \in Q$ के लिए $\Phi(x) - \Phi(x_0) - D\Phi(x_0)(x - x_0)$ पर लगाई गई माध्य मान असमिका $\norm{\Phi(x) - \Phi(x_0) - D\Phi(x_0)(x-x_0)}_\infty \leq
\varepsilon\norm{x - x_0}_\infty \leq \varepsilon r$ देती
है, और $D\Phi(x_0)^{-1}$ इस दोष को घन के $\varepsilon\norm{D\Phi(x_0)^{-1}}\,r$-विस्तार में समेट लेता है।
\cref{thm:b3:product:linearchange} से
\[
\lambda_d(\Phi(Q)) \leq \abs{\det D\Phi(x_0)}\,
\bigl(1 + \varepsilon\,C\bigr)^{d}\,\lambda_d(Q),
\qquad C = \sup_{Q}\norm{D\Phi(\cdot)^{-1}} .
\]

\emph{चरण 2: उपविभाजन द्वारा \omterm{def:b3:topology:compact}{संहत} घनों के लिए $(*)$।} मान लीजिए
$Q
\subseteq U$ कोई \omterm{def:b3:topology:compact}{संहत} घन है और $\varepsilon > 0$। $Q$ पर $D\Phi$ एकसमान \omterm{def:b3:topology:continuity}{संतत} है और
$\norm{D\Phi^{-1}}$ परिबद्ध (संहतता); $Q$ को $2^{kd}$ उपघनों $Q_i$ में इतना छोटा
उपविभाजित कीजिए कि प्रत्येक पर $D\Phi$ का दोलन $\leq \varepsilon$ हो। प्रत्येक
उपघन (\omterm{ex:b3:groups:actions}{केंद्र} $x_i$) पर चरण 1:
\[
\lambda_d(\Phi(Q)) \leq \sum_i\lambda_d(\Phi(Q_i))
\leq (1 + C\varepsilon)^d \sum_i \abs{\det
D\Phi(x_i)}\,\lambda_d(Q_i)
\leq (1 + C\varepsilon)^d\Bigl(\int_Q J + \varepsilon'\Bigr),
\]
और अंतिम चरण इसलिए कि $Q$ पर एकसमान रूप से $\sum_i\abs{\det D\Phi(x_i)}\mathbf
1_{Q_i} \to J$ ($\det D\Phi$ का
\omterm{def:b3:topology:continuity}{सांतत्य}) --- रीमान-योग तुलना। अब $\varepsilon \to 0$ लीजिए: अतः \omterm{def:b3:topology:compact}{संहत} घनों के लिए
$(*)$ लागू है।

\emph{चरण 3: समस्त बोरेल $A$ के लिए $(*)$।} $U$ के बोरेल
उपसमुच्चयों पर समुच्चय फलन $A
\mapsto \lambda_d(\Phi(A))$ एक \omterm{def:b3:measure:measure}{माप} है ($\Phi$ $V$ पर एकैकी
आच्छादक है और बोरेल समुच्चयों तथा गणनीय असंयुक्तता को सुरक्षित
रखता है), और $A \mapsto \int_AJ$ भी। $U$ का प्रत्येक विवृत उपसमुच्चय लगभग
असंयुक्त द्विआधारी \omterm{def:b3:topology:compact}{संहत} घनों का गणनीय सम्मिलन है (मानक द्विआधारी
विघटन: \omterm{def:b3:topology:topology}{विवृत समुच्चय} में अंतर्विष्ट महत्तम द्विआधारी घन लीजिए),
और दोनों \omterm{def:b3:measure:measure}{माप} उन पर योज्य हैं (घनों की परिसीमाएँ $\lambda_d$-शून्य हैं,
और उनके $\Phi$-प्रतिबिंब फलकों के पतले घन-आवरणों पर चरण 2 लगाने से
शून्य हैं): अतः $(*)$ घनों से विवृत समुच्चयों तक पहुँच जाता है।
व्यापक बोरेल $A$: $K_m \subseteq
\mathring K_{m+1}$ वाले संहतों $K_m \uparrow U$ से $U$ को निःशेष
कीजिए और $m$ स्थिर कीजिए; $\mathring K_{m+1}$ पर $J$ किसी $M_m$ से परिबद्ध
है। $\lambda_d$ की बाह्य नियमितता से (उपपत्ति \cref{thm:b3:measure:regularity} जैसी, बक्सों के
साथ) $A \cap K_m \subseteq O_n
\subseteq \mathring K_{m+1}$ और $\lambda_d\bigl(O_n \setminus
(A\cap K_m)\bigr) \to 0$ वाले \omterm{def:b3:topology:topology}{विवृत समुच्चय} $O_n$ चुनिए। तब
\[
\lambda_d\bigl(\Phi(A\cap K_m)\bigr) \leq
\lambda_d\bigl(\Phi(O_n)\bigr) \leq \int_{O_n}J
\leq \int_{A\cap K_m}J + M_m\,\lambda_d\bigl(O_n\setminus(A\cap
K_m)\bigr) \xrightarrow[n\to\infty]{} \int_{A\cap K_m}J .
\]
अब $m \to \infty$ लीजिए: बाईं ओर नीचे से \omterm{def:b3:topology:continuity}{सांतत्य}, दाईं ओर एकदिष्ट अभिसरण
प्रमेय। इससे $(*)$ स्थापित हो जाता है।

\emph{चरण 4: समता और समाकल सूत्र।} पहले $(*)$ को समुच्चयों से
समाकलों तक बढ़ाइए: $V$ पर प्रत्येक \omterm{def:b3:lebesgue:measurable}{मापनीय} $g
\geq 0$ के लिए
\[
\int_V g(y)\,\dd y \leq \int_U g(\Phi(x))\,J(x)\,\dd x .
\tag{$**$}
\]
वस्तुतः, $g = \mathbf 1_B$ के लिए यह $A =
\Phi^{-1}(B)$ के साथ $(*)$ है; रैखिकता इसे सरल
$g$ तक बढ़ा देती है, और एकदिष्ट अभिसरण प्रमेय समस्त $g \geq 0$ तक
(मानक मशीन)। अब $(**)$ को दो बार लगाइए: पहले $g$ पर, फिर ---
अवकल समरूपता $\Phi^{-1}$ के लिए --- फलन $x \mapsto g(\Phi(x))J(x)$ पर:
\[
\int_Vg \leq \int_U g(\Phi(x))J(x)\dd x
\leq \int_V g(y)\,J(\Phi^{-1}y)\,\abs{\det
D\Phi^{-1}(y)}\,\dd y = \int_V g ,
\]
क्योंकि $J(\Phi^{-1}y)\abs{\det D\Phi^{-1}(y)} = \abs{\det\bigl(
D\Phi(\Phi^{-1}y)\,D\Phi^{-1}(y)\bigr)} = 1$ ($\Phi\circ\Phi^{-1} = \mathrm{id}$ पर शृंखला नियम)। अतः सभी असमिकाएँ समताएँ
हैं: सूत्र $g \geq 0$ के लिए लागू है, और विघटन से $L^1$ फलनों के लिए
भी।
\end{proof}

\begin{example}[ध्रुवीय निर्देशांक; गाउसीय पुनः]
\label{ex:b3:product:polar}
$\Phi(r, \theta) = (r\cos\theta, r\sin\theta)$ $\intoo0{+\infty}\times\intoo0{2\pi}$ से $\R^2$ में से एक अर्ध-रेखा (शून्य समुच्चय) हटाकर
मिलने वाली समष्टि पर $\det D\Phi =
r$ वाली $\mathcal
C^1$ अवकल समरूपता
है:\index{ध्रुवीय निर्देशांक}
\[
\int_{\R^2}f(x, y)\,\dd x\,\dd y
= \int_0^{2\pi}\!\!\int_0^{+\infty}
f(r\cos\theta, r\sin\theta)\,r\,\dd r\,\dd\theta .
\]
$f = \eu^{-x^2-y^2}$ के लिए टोनेली और यही सूत्र
\[
G^2 = \Bigl(\int_\R \eu^{-x^2}\dd x\Bigr)^2
= \int_{\R^2}\eu^{-x^2-y^2}
= 2\pi\int_0^\infty r\eu^{-r^2}\dd r = \pi:
\]
देते हैं --- $G = \sqrt\pi$ की शास्त्रीय दो-पंक्ति उपपत्ति, जो अब पूर्णतः
उचित ठहरा दी गई है (\cref{pb:b3:lebesgue:1} की प्राचल उपपत्ति से तुलना कीजिए)।
\end{example}

\begin{theorem}[इकाई गोले का आयतन]
\label{thm:b3:product:ballvolume}
मान लीजिए $\R^d$ में $v_d = \lambda_d(B(0,1))$। तब\index{इकाई गोले का आयतन}
\[
v_d = \frac{\pi^{d/2}}{\Gamma\bigl(\frac d2 + 1\bigr)} :
\qquad
v_1 = 2,\quad v_2 = \pi,\quad v_3 = \tfrac{4\pi}3,\quad
v_4 = \tfrac{\pi^2}2,\ \dots
\]
\end{theorem}

\begin{proof}
$I = \int_{\R^d}\eu^{-\norm x_2^2}\dd\lambda_d$ को दो बार परिकलित कीजिए। टोनेली से वह गुणनखंडित हो जाता है:
$I = G^d = \pi^{d/2}$। और परत-केक सूत्र (\cref{prop:b3:product:layercake}) से $f = \eu^{-
\norm x^2}$ के साथ, जिसके स्तर
समुच्चय गोले हैं: $0 < t < 1$ के लिए $\{f > t\} =
B\bigl(0, \sqrt{-\ln t}\bigr)$, जिसका \omterm{def:b3:measure:measure}{माप} $v_d(-\ln t)^{d/2}$ है ($\rho$
से प्रसार $\lambda_d$ को $\rho^d$ से मापित करता है: \cref{thm:b3:product:linearchange}), अतः
\[
I = \int_0^1 v_d\,(-\ln t)^{d/2}\,\dd t
\overset{t = \eu^{-s}}{=}
v_d\int_0^\infty s^{d/2}\eu^{-s}\,\dd s
= v_d\,\Gamma\Bigl(\frac d2 + 1\Bigr).
\]
अब बराबर कीजिए। (मान: $\Gamma(\frac32) = \frac{\sqrt\pi}2$, $\Gamma(2) = 1$, इत्यादि।) ध्यान दीजिए कि
$d \to \infty$ पर $v_d \to 0$ --- सप्ताहांत समस्या स्टर्लिंग के माध्यम से बताती
है कि कितनी तेजी से।
\end{proof}

\begin{exercise}[$\star\star\star$]\label{exo:b3:product:8}
(बीटा--गामा) $p, q > 0$ के लिए मान लीजिए $B(p, q) = \int_0^1t^{p-1}(1

चतुर्थांश की $\intoo0\infty\times\intoo01$ पर एक अवकल समरूपता है; उसका याकोबीय $= u$
परिकलित कीजिए) और निष्कर्ष निकालिए\index{बीटा फलन}

\]
इससे $\int_0^{\pi/2}\sin^{2p-1}\theta\cos^{2q-1}\theta\,
\dd\theta = \frac12B(p,q)$ और वालिस समाकलों $W_n = \int_0^{\pi/2}\sin^n$ का मान निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:9}
(स्थानांतरण सूत्र) मान लीजिए $T \colon (X, \mathcal A, \mu) \to (Y,
\mathcal B)$ \omterm{def:b3:lebesgue:measurable}{मापनीय} है और $T_*\mu(B) = \mu(T^{-1}(B))$
\emph{अग्रप्रेषित माप}\index{अग्रप्रेषित माप}। दर्शाइए कि $Y$
पर प्रत्येक \omterm{def:b3:lebesgue:measurable}{मापनीय} $g \geq 0$ के लिए
\[
\int_Y g\,\dd(T_*\mu) = \int_X g\circ T\,\dd\mu
\]
(मानक मशीन)। फिर \cref{thm:b3:product:linearchange} से तुलना कीजिए: चर परिवर्तन सूत्र वह
कौन-सी अतिरिक्त सूचना देता है जो अमूर्त स्थानांतरण सूत्र नहीं
देता? \emph{(स्थानांतरण सूत्र $T_*\mu$ की पहचान कभी नहीं करता; चर
परिवर्तन प्रमेय $T_*\lambda_d$ को \omterm{ex:b3:lebesgue:gamma}{घनत्व} \omterm{def:b3:measure:measure}{माप} के रूप में स्पष्ट परिकलित कर
देती है।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:product:10}
(गाउसीय आघूर्ण) ध्रुवीय निर्देशांकों और फुबिनी का उपयोग करके
$\R^d$ पर मानक गाउसीय भार के लिए परिकलित कीजिए:
\[
\int_{\R^d}\norm x_2^2\;\eu^{-\norm x_2^2}\,\dd x
\qquad\text{तथा}\qquad
\int_{\R^d}x_1^2\,\eu^{-\norm x^2_2}\,\dd x,
\]
संगति जाँचिए ($\norm x^2 = \sum x_i^2$), और \omterm{def:b3:measure:measure}{माप} $\pi^{-d/2}\eu^{-\norm x^2}\dd x$ का द्वितीय आघूर्ण निष्कर्ष के
रूप में निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:11}
(आलेख और अधोआलेख) मान लीजिए $f \colon \R^d \to \intco0\infty$ \omterm{def:b3:lebesgue:measurable}{मापनीय} है।
(क) दर्शाइए कि \emph{अधोआलेख} $H = \{(x, y) \in
\R^d\times\R : 0 < y < f(x)\}$ $\R^{d+1}$ में \omterm{def:b3:lebesgue:measurable}{मापनीय} है और
\[
\lambda_{d+1}(H) = \int_{\R^d}f\,\dd\lambda_d :
\]
अर्थात् ``समाकल आलेख के नीचे का क्षेत्रफल है'', जो अंततः एक प्रमेय
बन गया। \emph{(परिच्छेद; टोनेली।)}
(ख) दर्शाइए कि आलेख $\{(x, f(x)) : x \in \R^d\}$ $\R^{d+1}$ का शून्य समुच्चय है।
(ग) इससे दो पंक्तियों में सिद्ध कीजिए कि गोलक $S^{d-1}$ $\R^d$ में
लेबेग-शून्य है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:12}
(एक प्रसिद्ध द्विक समाकल) $\intoo01^2$ पर गुणोत्तर श्रेणी और टोनेली का
उपयोग करके सिद्ध कीजिए
\[
\int_0^1\!\!\int_0^1\frac{\dd x\,\dd y}{1 - xy}
= \sum_{n\geq1}\frac1{n^2} = \zeta(2),
\qquad
\int_0^1\!\!\int_0^1\frac{\dd x\,\dd y}{1 + xy}
= \sum_{n\geq1}\frac{(-1)^{n-1}}{n^2} = \frac{\zeta(2)}2 .
\]
(दूसरी श्रेणी सर्वसमिका: सम और विषम सूचकांक अलग कीजिए।) $\zeta(2) = \frac{\pi^2}6$
(\cref{ch:b3:spectral}) के साथ, दो निर्दोष दिखने वाले समाकलों के मान $\frac{\pi^2}6$ और
$\frac{\pi^2}{12}$ निकल आते हैं; टोनेली की धनात्मकता परिकल्पना ठीक कहाँ अपना
काम करती है?
\end{exercise}

\section{समस्या: स्टर्लिंग का सूत्र}

\begin{problem}[{सप्ताहांत समस्या --- $n! \sim
\sqrt{2\pi n}\,(n/\eu)^n$, प्रभावी अभिसरण से}]
\label{pb:b3:product:1}
स्टर्लिंग का सूत्र इस पुस्तक की प्रत्येक स्पर्शोन्मुख गणना पर
शासन करता है --- गोलों के आयतन, द्विपद गुणांक, केंद्रीय सीमा
प्रमेय का स्थानीय रूप। हम उसे $\Gamma$ समाकल (\cref{ex:b3:lebesgue:gamma}) से
\emph{लाप्लास विधि} द्वारा, उसके स्वच्छतम प्रभावी-अभिसरण रूप में
सिद्ध करते हैं, और फिर लाभांश बटोरते हैं।\index{स्टर्लिंग का
सूत्र}

\medskip
\noindent\textbf{भाग I --- सूत्र।} $t > 0$ के लिए $\Gamma(t + 1) = \int_0^\infty x^{t}\eu^{-x}\dd x$।
\begin{enumerate}
  \item $x = t + \sqrt t\,u$ प्रतिस्थापित कीजिए और दर्शाइए
        \[
        \frac{\Gamma(t+1)}{t^{t}\eu^{-t}\sqrt t}
        = \int_{-\sqrt t}^{+\infty}
        \exp\Bigl(t\ln\Bigl(1 + \frac u{\sqrt
        t}\Bigr) - \sqrt t\,u\Bigr)\,\dd u
        \;=\;\int_\R g_t(u)\,\dd u,
        \]
        जहाँ $g_t(u) = \exp\bigl(t\ln(1 + u/\sqrt t) -
        \sqrt t\,u\bigr)\mathbf 1_{u > -\sqrt t}$।
  \item बिंदुशः सीमा दर्शाइए: प्रत्येक स्थिर $u$ के लिए $t \to +\infty$
        पर $g_t(u)
        \to \eu^{-u^2/2}$ \emph{($\ln(1 + h)$ का द्वितीय कोटि तक प्रसार कीजिए)}।
  \item प्रभुत्व। मान लीजिए $\varphi(h) = \ln(1 + h) - h$, जिससे $u > -\sqrt t$ के लिए $g_t(u) = \exp\bigl(t\,\varphi(u/\sqrt t)\bigr)$।
        दोनों परिबंध सिद्ध कीजिए
        \[
        \varphi(h) \leq -\frac{h^2}4 \quad (-1 < h \leq 1),
        \qquad
        \varphi(h) \leq -c\,h \quad (h \geq 1),\ \ c = 1 -
        \ln 2 > 0
        \]
        \emph{($\varphi(h) + \frac{h^2}4$ और $\varphi(h)
        + ch$ का अध्ययन कीजिए: अवकलज परिकलित
        कीजिए और प्रत्येक परास पर चिह्न जाँचिए)}। $t \geq 1$ के लिए
        निष्कर्ष निकालिए:
        \[
        g_t(u) \leq \eu^{-u^2/4}\ \ (\abs u \leq \sqrt t),
        \qquad
        g_t(u) \leq \eu^{-cu}\ \ (u \geq \sqrt t),
        \]
        जिससे $g_t(u) \leq \eu^{-u^2/4} +
        \eu^{-cu}\,\mathbf 1_{u > 0}$: अर्थात् $t \geq 1$ से स्वतंत्र एक \omterm{def:b3:lebesgue:l1}{समाकलनीय}
        प्रभावी फलन।
  \item प्रभावी अभिसरण प्रमेय और गाउसीय समाकल (\cref{ex:b3:product:polar}) से निष्कर्ष
        निकालिए:
        \[
        \Gamma(t + 1) \;\sim\;
        \sqrt{2\pi t}\;\Bigl(\frac t\eu\Bigr)^{t}
        \qquad (t \to +\infty),
        \]
        और विशेष रूप से $n! \sim \sqrt{2\pi
        n}\,(n/\eu)^n$।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- लाभांश।}
\begin{enumerate}[resume]
  \item (वालिस) \cref{exo:b3:product:8}, $W_{2n} =
        \frac\pi2\binom{2n}n4^{-n}$ प्रकार के सूत्रों से: स्टर्लिंग से
        $\binom{2n}{n} \sim \frac{4^n}{\sqrt{\pi n}}$ निकालिए, और उसे पुनरावर्तन $W_{n} =
        \frac{n-1}nW_{n-2}$ के सापेक्ष
        जाँचिए।
  \item (गोलों के आयतन ढह जाते हैं) दर्शाइए
        \[
        v_d = \frac{\pi^{d/2}}{\Gamma(\frac d2 + 1)}
        \;\sim\; \frac{1}{\sqrt{\pi d}}
        \Bigl(\frac{2\pi\eu}{d}\Bigr)^{d/2},
        \]
        अतः $v_d \to 0$, और वह भी किसी भी गुणोत्तर अनुक्रम से तेज;
        $v_d$ को अधिकतम करने वाली विमा ज्ञात कीजिए (संख्यात्मक
        रूप से: $d =
        5$)।
  \item (द्विपद का संकेंद्रण --- \cref{ch:b3:clt} की एक झलक) स्टर्लिंग का
        उपयोग करके स्थानीय आकलन दर्शाइए, स्थिर $s$ वाले $k = n/2 + s\sqrt n/2$
        और सम $n$ के लिए:
        \[
        2^{-n}\binom{n}{k} \;\sim\;
        \sqrt{\frac{2}{\pi n}}\;\eu^{-s^2/2},
        \]
        अर्थात् विविक्त गाउसीय रूपरेखा: भ्रूण रूप में
        दे म्वाव्र--लाप्लास।
  \item भाग I की उपपत्ति ने ठीक कहाँ उपयोग किया: (क) एकदिष्ट या
        प्रभावी अभिसरण प्रमेय; (ख) गाउसीय समाकल; (ग) \omterm{def:b3:measure:lebesgueouter}{लेबेग माप}
        के अपरिवर्तिता गुण? प्रत्येक के लिए एक-एक वाक्य।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- त्रुटि पद: सीमाओं सहित स्टर्लिंग।}
$d_n = \ln n! - \bigl(n + \tfrac12\bigr)\ln n + n -
\ln\sqrt{2\pi}$ रखिए, जिससे भाग I कहता है कि $d_n \to 0$।
\begin{enumerate}[resume]
  \item दर्शाइए $d_n - d_{n+1} = \bigl(n +
        \tfrac12\bigr)\ln\bigl(1 + \tfrac1n\bigr) - 1$।
  \item $t = \frac1{2n+1}$ के साथ $\frac{n+1}n =
        \frac{1+t}{1-t}$ सत्यापित कीजिए और प्रसार कीजिए:
        \[
        d_n - d_{n+1} = \frac{t^2}3 + \frac{t^4}5 +
        \frac{t^6}7 + \cdots,
        \]
        तथा द्विपक्षीय परिबंध निकालिए
        \[
        \frac1{3(2n+1)^2} \;<\; d_n - d_{n+1} \;<\;
        \frac1{12n} - \frac1{12(n+1)} .
        \]
  \item दूरबीनी योग लीजिए ($d_m \to 0$ का उपयोग करते हुए) और $m \geq 1$ के
        लिए सुखद बीजगणितीय सर्वसमिका $\frac1{3(2m+1)^2} >
        \frac1{12m+1} - \frac1{12(m+1)+1}$ जाँचिए, जिससे
        शास्त्रीय कोष्ठन
        \[
        \sqrt{2\pi n}\Bigl(\frac n\eu\Bigr)^n
        \eu^{1/(12n+1)} \;<\; n! \;<\;
        \sqrt{2\pi n}\Bigl(\frac n\eu\Bigr)^n\eu^{1/(12n)} .
        \]
        मिल जाए।
  \item दो परिणाम: (क) $n \geq 83\,334$ होते ही स्टर्लिंग के सूत्र की
        सापेक्ष त्रुटि $< 10^{-6}$ हो जाती है; (ख) कोष्ठन से $100!$ का
        हाथ से चार सार्थक अंकों तक आकलन कीजिए ($100! \approx 9.3326\cdot
        10^{157}$), और एक बहुत
        ही परिमित $n$ पर किसी स्पर्शोन्मुख सूत्र की यथार्थता पर
        क्षण भर विस्मित होइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- वालिस मार्ग: गाउसीय के बिना
स्टर्लिंग।} ऐतिहासिक रूप से नियतांक $\sqrt{2\pi}$ वालिस से आया था,
गाउस से नहीं; यह भाग स्टर्लिंग को भाग I और II से स्वतंत्र रूप से
पुनः सिद्ध करता है, और इस प्रकार गाउसीय समाकल को भी पुनः सिद्ध कर
देता है। मान लीजिए $W_n = \int_0^{\pi/2}\sin^n\theta\,
\dd\theta$।
\begin{enumerate}[resume]
  \item $W_n = \frac{n-1}nW_{n-2}$ स्थापित कीजिए (खंडशः समाकलन), संवृत रूप
        \[
        W_{2n} = \frac\pi2\binom{2n}n4^{-n}, \qquad
        W_{2n+1} = \frac{4^n}{(2n+1)\binom{2n}n},
        \]
        और सर्वसमिका $W_nW_{n-1} = \frac\pi{2n}$ भी।
  \item $(W_n)$ की एकदिष्टता से $W_{2n}/W_{2n+1} \to 1$ निकालिए, फिर
        \[
        W_{2n} \sim \frac12\sqrt{\frac\pi n}
        \qquad\text{तथा}\qquad
        \binom{2n}n4^{-n}\sqrt n \longrightarrow
        \frac1{\sqrt\pi} :
        \]
        अर्थात् वालिस की प्रमेय, जो स्टर्लिंग के बिना प्राप्त हुई।
  \item केवल भाग III के दूरबीनी योग से (नियतांक के मान की कोई
        आवश्यकता नहीं) दर्शाइए कि $e_n = \ln n! - (n +
        \frac12)\ln n + n$ किसी सीमा $\ell$ पर
        अभिसरित होता है; समतुल्य रूप से $n! \sim K\,n^{n+1/2}\eu^{-n}$, जिसमें $K =
        \eu^\ell > 0$ अभी
        अज्ञात है।
  \item इस स्पर्शोन्मुखता को $\binom{2n}n4^{-n}\sqrt n$ में डालिए और प्रश्न 14 का
        उपयोग करके एकमात्र संभव मान पहचानिए: $K = \sqrt{2\pi}$। तर्क को
        संकलित कीजिए: भाग III और IV मिलकर स्टर्लिंग की एक पूरी
        दूसरी उपपत्ति देते हैं --- और इसलिए, भाग I का प्रतिस्थापन
        उलटा चलाकर, $\int_\R\eu^{-u^2/2}\dd u = \sqrt{2\pi}$ का एक स्वतंत्र मूल्यांकन भी। दो स्तंभ,
        जिनमें से कोई भी दूसरे को थाम सकता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- अंतिम लाभांश।}
\begin{enumerate}[resume]
  \item (पूरी स्थानीय रूपरेखा) पूर्णांक $\abs j \leq
        K\sqrt n$ ($K$ स्थिर) के
        लिए दर्शाइए
        \[
        \frac{\binom{2n}{n+j}}{\binom{2n}{n}} =
        \prod_{i=1}^{\abs j}\frac{n - i + 1}{n + i}
        = \exp\Bigl(-\frac{j^2}n +
        O\Bigl(\frac1{\sqrt n}\Bigr)\Bigr),
        \]
        और वह भी $j$ में एकसमान रूप से \emph{(लघुगणक लीजिए और
        $\ln
        \frac{1-x}{1+y} = -(x + y) + O(x^2 + y^2)$ का उपयोग कीजिए)}। यह प्रश्न 7 का द्विपक्षीय रूप है
        और \cref{ch:b3:clt} की सप्ताहांत समस्या में उद्धृत यथार्थ आकलन भी।
  \item (प्वासों की एक झलक) स्टर्लिंग से दर्शाइए कि $\eu^{-n}\dfrac{n^n}{n!} \sim \dfrac1{\sqrt{2\pi n}}$:
        अर्थात् बड़े माध्य $n$ वाले प्वासों नियम का बहुलक $\approx (2\pi n)^{-1/2}$
        द्रव्यमान धारण करता है, ठीक वैसे ही जैसा केंद्रीय सीमा
        प्रमेय बताएगी।
  \item (गामा अनुपात) $a \in \intoo01$ के लिए \cref{pb:b3:lebesgue:1} (वहाँ प्रश्न 14) की
        लघुगणकीय-उत्तलता ढाल परिबंधों का उपयोग करके $\dfrac{\Gamma(n + a)}{\Gamma(n)\,n^a} \to 1$ सिद्ध
        कीजिए, और कार्यकीय समीकरण से इसे प्रत्येक वास्तविक $a > 0$
        तक बढ़ाइए। (पूर्णांकों के बीच ``$\Gamma(t+1) \sim$ स्टर्लिंग'' का यही
        अर्थ है।)
  \item (गोले, फिर से) $v_d = \pi^{d/2}/\Gamma(\frac d2
        + 1)$ से: $v_1, \dots, v_7$ यथार्थ रूप से सारणीबद्ध
        कीजिए, $\frac{v_d}{v_{d-2}} = \frac{2\pi}d$ के माध्यम से एकबहुलकता सत्यापित कीजिए ($d < 2\pi$
        तक बढ़ता हुआ, उसके बाद घटता हुआ), और उल्लेखनीय जनक
        सर्वसमिका
        \[
        \sum_{k\geq0}v_{2k}\,x^{2k} = \eu^{\pi x^2} :
        \]
        सिद्ध कीजिए --- समस्त सम-विमीय इकाई गोलों के आयतन एक ही
        चरघातांकी में बँधे हुए।
  \item (एन्ट्रॉपी स्पर्शोन्मुखता) $\alpha n \in \N$ वाले स्थिर $\alpha \in \intoo01$ के लिए
        स्टर्लिंग से निकालिए
        \[
        \binom{n}{\alpha n} \;\sim\;
        \frac{\eu^{n\,H(\alpha)}}
        {\sqrt{2\pi\,\alpha(1-\alpha)\,n}},
        \qquad
        H(\alpha) = -\alpha\ln\alpha -
        (1-\alpha)\ln(1-\alpha) :
        \]
        अर्थात् द्विपद गुणांकों की चरघातांकी वृद्धि दर
        \emph{एन्ट्रॉपी} $H$ है --- जाँचिए कि $\alpha =
        \frac12$ से प्रश्न
        5 पुनः मिल जाता है, और यह कि $\alpha \neq \frac12$ के लिए $H(\alpha) <
        \ln2$ (अतः
        \omterm{ex:b3:groups:actions}{केंद्र} से हटे द्विपद $2^n$ में चरघातांकी रूप से नगण्य होते
        हैं)।
  \item (पृष्ठ क्षेत्रफल) इकाई गोलक $S^{d-1}$ का क्षेत्रफल $s_{d-1} = d\,v_d$ है
        (जो अवकल रूपों वाले अध्याय में \cref{exo:b3:forms:6} के रूप में सिद्ध हुआ
        है; यहाँ उसे परिभाषा मान लीजिए)। $s_0, \dots, s_6$ सारणीबद्ध कीजिए,
        महत्तम वाला ढूँढ़िए ($d - 1 =
        6$, $s_6 = \frac{16\pi^3}{15} \approx 33.07$), और दर्शाइए कि $s_{d-1} \to 0$
        भी अति-गुणोत्तर दर से होता है --- उच्च-विमीय गोलक,
        प्रत्येक यूक्लिडीय मापदंड से, लुप्तप्राय रूप से छोटे हैं।
  \item (पहला संशोधन पद) प्रश्न 11 के कोष्ठन से निकालिए कि
        $d_n = \frac1{12n} +
        O\bigl(\frac1{n^2}\bigr)$, अतः
        \[
        n! = \sqrt{2\pi n}\,\Bigl(\frac
        n\eu\Bigr)^{n}\Bigl(1 + \frac1{12n} +
        O\Bigl(\frac1{n^2}\Bigr)\Bigr).
        \]
        $n = 10$ पर सत्यापित कीजिए: नंगा सूत्र $3\,598\,696$ देता है
        (सापेक्ष त्रुटि $8.3\cdot10^{-3}$), और संशोधित सूत्र $10! =
        3\,628\,800$ के सामने
        $3\,628\,685$ (सापेक्ष त्रुटि $3.2\cdot10^{-5}$) --- श्रेणी का एक पद ढाई
        अंक खरीद लेता है।
  \item ($\Gamma$ की माध्यिका) दर्शाइए कि स्पर्शोन्मुख रूप से
        \[
        \frac{1}{\Gamma(t+1)}
        \int_0^{t} x^{t}\eu^{-x}\,\dd x
        \;\longrightarrow\; \frac12
        \qquad (t \to +\infty) :
        \]
        अर्थात् $\Gamma$ के समाकल्य का ठीक आधा द्रव्यमान उसके बहुलक
        $x = t$ के नीचे बैठता है। \emph{(छँटे हुए समाकल पर भाग I का
        प्रतिस्थापन चलाइए; प्रश्न 3 का प्रभावी फलन पहले से
        उपलब्ध है।)}
  \item (एन्ट्रॉपी, बिना स्पर्शोन्मुखता के) $\alpha \in
        \intoc0{\frac12}$ के लिए
        \emph{प्रत्येक} $n \geq 1$ पर वैध परिबंध
        \[
        \sum_{k=0}^{\lfloor\alpha n\rfloor}\binom nk
        \;\leq\; \eu^{n\,H(\alpha)} ,
        \]
        सिद्ध कीजिए, और वह भी झुकाव $\lambda =
        \frac{\alpha}{1-\alpha} \leq 1$ के लिए योग की $\sum_k\binom
        nk\lambda^{k-\alpha n}$ से
        तुलना करके। जाँचिए कि $\lambda$ का यह चयन इष्टतम है, और प्रश्न
        21 से मेल कराइए: स्पर्शोन्मुख कथन की चरघातांकी दर $H(\alpha)$
        बिना किसी स्पर्शोन्मुखता के, एक पंक्ति की असमिका से ही
        प्राप्त हो जाती है।
\end{enumerate}
\end{problem}
