\chapter{\texorpdfstring{$\R^n$}{Rn} के उपबहुविध}
\label{ch:b3:submanifolds}

गोलक, टोरस, घूर्णन समूह: ज्यामिति और यांत्रिकी के स्वाभाविक निवास
सदिश समष्टियाँ नहीं, बल्कि \emph{ऐसे वक्रित समुच्चय हैं जो पास से
समतल दिखते हैं}। यह अध्याय उस वाक्यांश को ठीक-ठीक अर्थ देता है ---
$\R^n$ के उपबहुविध --- और उन पर काम करने का कलन भी। नींव है प्रतिलोम
फलन प्रमेय, जिसे यहाँ बानाख स्थिर बिंदु से सिद्ध किया गया है; और
शेष सब कुछ निर्देशांकों का परिवर्तन ही है: उपबहुविध के चार तुल्य
वर्णन (स्थानीय सीधाकरण, स्तर समुच्चय, आलेख, प्राचलन), स्पर्श
समष्टियाँ, और लाग्राँज गुणकों से प्रतिबंधित इष्टतमीकरण --- जो,
विदाई प्रदर्शन के रूप में, सममित आव्यूहों के लिए स्पेक्ट्रमी
प्रमेय को ज्यामिति की तीन पंक्तियों में पुनः सिद्ध कर देता है।
सप्ताहांत समस्या घूर्णन समूह $SO(3)$ और उसका क्वाटर्नियनी द्विक आवरण
बनाती है: बीजगणित (\cref{ch:b3:groups} का $Q_8$, बड़ा होकर) ज्यामिति से मिलता हुआ।

\section{प्रतिलोम फलन प्रमेय}

\begin{theorem}[प्रतिलोम फलन प्रमेय]
\label{thm:b3:submanifolds:ift}
मान लीजिए $U \subseteq \R^n$ विवृत है, $f \colon U \to \R^n$ वर्ग $\mathcal C^1$ का है, और $a \in U$ ऐसा कि
$Df(a)$ व्युत्क्रमणीय हो। तब ऐसे \omterm{def:b3:topology:topology}{विवृत समुच्चय} $V \ni a$, $W \ni f(a)$ हैं कि
$f \colon V \to W$ $\mathcal C^1$ प्रतिलोम वाला एकैकी आच्छादक प्रतिचित्रण है,
और\index{प्रतिलोम फलन प्रमेय}
\[
D(f^{-1})(y) = \bigl(Df(f^{-1}(y))\bigr)^{-1}
\qquad (y \in W).
\]
यदि $f$ $\mathcal C^k$ हो, तो $f^{-1}$ भी।
\end{theorem}

\begin{proof}
मानकीकरण: $f$ के स्थान पर $x \mapsto Df(a)^{-1}\bigl(f(a +
x) - f(a)\bigr)$ रखने पर हम $a = 0$, $f(0) = 0$, $Df(0) =
I$ मान
सकते हैं (व्यापक कथन आफ़ीन एकैकी आच्छादक प्रतिचित्रणों के साथ
संयोजन से निकलता है)। $f(x) = x + g(x)$ लिखिए: $Dg(0) = 0$, और $Dg$ के \omterm{def:b3:topology:continuity}{सांतत्य} से
$r > 0$ ऐसा चुनिए कि $\bar B(0, r)$ पर $\vertiii{Dg(x)} \leq \frac12$ हो; तब माध्य मान असमिका वहाँ
$\norm{g(x) - g(x')} \leq \frac12
\norm{x - x'}$ दे देती है।

\emph{किसी \omterm{def:b3:topology:topology}{प्रतिवेश} पर एकैकी आच्छादकता।} $y \in B(0,
\frac r2)$ के लिए $f(x) = y$ हल करने
का अर्थ है $\Phi_y(x) = y - g(x)$ का कोई स्थिर बिंदु ढूँढ़ना; $\Phi_y$ $\bar B(0, r)$ को उसी में
भेजता है ($\norm{\Phi_y(x)} \leq \norm y + \frac12\norm x \leq
r$) और $\frac12$-लिप्शिट्ज़ है: अतः बानाख (\cref{thm:b3:complete:banach}) एक
अद्वितीय हल $x = \varphi(y) \in \bar B(0,r)$ दे देता है। इसके अतिरिक्त $f$ $\bar B(0,r)$ पर एकैकी
है:
\[
\norm{f(x) - f(x')} \geq \norm{x - x'} - \norm{g(x) - g(x')}
\geq \tfrac12\norm{x - x'} .
\tag{$*$}
\]
$W = B(0, \frac r2)$ और $V = f^{-1}(W)\cap B(0, r)$ रखिए: ये विवृत हैं (\omterm{def:b3:topology:continuity}{सांतत्य}), और $f \colon V \to W$ एकैकी आच्छादक।

\emph{प्रतिलोम का \omterm{def:b3:topology:continuity}{सांतत्य} और अवकलनीयता।} $(*)$ कहता है कि $\varphi = f^{-1}$
$2$-लिप्शिट्ज़ है। $y_0 = f(x_0)
\in W$ स्थिर कीजिए; $A = Df(x_0)$ की व्युत्क्रमणीयता
($I$ से उसकी दूरी $\leq \frac12$ है: न्यूमन, \cref{prop:b3:banach:neumann}) और $f$ की अवकलनीयता
$y_0$ के निकट $y = f(x)$ के लिए
\[
\varphi(y) - \varphi(y_0) - A^{-1}(y - y_0)
= -A^{-1}\bigl(f(x) - f(x_0) - A(x - x_0)\bigr)
= -A^{-1}\,o(\norm{x - x_0}) = o(\norm{y - y_0}),
\]
दे देती हैं, जहाँ $\norm{x - x_0} \leq 2\norm{y - y_0}$ को बदलने के लिए $(*)$ का उपयोग हुआ: अतः
$\varphi$ $y_0$ पर प्रतिलोम अवकलज के साथ अवकलनीय है। $y \mapsto D\varphi(y) =
Df(\varphi(y))^{-1}$ का \omterm{def:b3:topology:continuity}{सांतत्य}:
\omterm{def:b3:topology:continuity}{संतत} प्रतिचित्रणों का संयोजन (प्रतिलोमन \omterm{def:b3:topology:continuity}{संतत} है, \cref{prop:b3:banach:neumann}): अतः
$\varphi \in \mathcal C^1$; और उसी सूत्र को बार-बार लगाने पर $\mathcal C^k$।
\end{proof}

\begin{theorem}[अंतर्निहित फलन प्रमेय]
\label{thm:b3:submanifolds:implicit}
मान लीजिए $F \colon U \subseteq \R^p\times\R^q \to \R^q$ $(a, b)$ के निकट $\mathcal C^1$ है, $F(a,b) = 0$, और मान लीजिए आंशिक
अवकलज $D_yF(a,b) \in \mathcal L(\R^q)$ व्युत्क्रमणीय है। तब ऐसे \omterm{def:b3:topology:topology}{प्रतिवेश} $A \ni a$, $B \ni b$ और ऐसा
$\mathcal C^1$ प्रतिचित्रण $\psi \colon A \to B$ हैं कि
\[
\bigl\{(x, y)\in A\times B : F(x,y) = 0\bigr\}
= \{(x, \psi(x)) : x \in A\},
\]
और $D\psi(x) = -D_yF(x, \psi(x))^{-1}\,D_xF(x,
\psi(x))$।\index{अंतर्निहित फलन प्रमेय}
\end{theorem}

\begin{proof}
$\Theta(x, y) = (x,
F(x,y))$ पर \cref{thm:b3:submanifolds:ift} लगाइए: $(a,b)$ पर उसका अवकलज, जो व्युत्क्रमणीय विकर्ण
खंडों $I$ और $D_yF$ के साथ खंड-त्रिभुजाकार है, व्युत्क्रमणीय है।
स्थानीय प्रतिलोम का रूप $\Theta^{-1}(x, z) = (x, h(x,
z))$ होता है; $\psi(x) = h(x, 0)$ रखिए: तब स्थानीय रूप
से $F(x, y) = 0$ तभी और केवल तभी जब $\Theta(x,y) = (x, 0)$, अर्थात् तभी और केवल तभी जब
$y = \psi(x)$। सूत्र: शृंखला नियम से $F(x, \psi(x)) = 0$ का अवकलन कीजिए।
\end{proof}

\section{उपबहुविध: चार परिभाषाएँ}

\begin{theorem}[तुल्य अभिलक्षण]
\label{thm:b3:submanifolds:characterizations}
मान लीजिए $M \subseteq \R^n$, $d \in \{0, \dots, n\}$, और $k \geq
1$। प्रत्येक बिंदु $a \in M$ के लिए
निम्नलिखित समतुल्य हैं (और $M$ \emph{वर्ग $\mathcal C^k$ का $d$-विमीय
उपबहुविध}\index{उपबहुविध} कहलाता है यदि वे प्रत्येक $a
\in M$ पर लागू
हों):
\begin{enumerate}
  \item (\emph{सीधाकरण}) किसी विवृत $\Omega \ni a$ से किसी विवृत $\Omega' \subseteq \R^n$ पर
        कोई $\mathcal C^k$ अवकल समरूपता $\Phi$ ऐसी है कि
        \[
        \Phi(M\cap\Omega) = \Omega' \cap
        \bigl(\R^d\times\{0\}\bigr).
        \]
  \item (\emph{स्तर समुच्चय}) किसी विवृत $\Omega \ni a$ पर $M\cap\Omega = F^{-1}(0)$ वाला कोई
        $\mathcal C^k$ \emph{अवगाहन} $F \colon \Omega \to \R^{n-d}$ (अर्थात् $DF(x)$ आच्छादक) है।
  \item (\emph{आलेख}) निर्देशांकों के क्रमचय तक, $M$ स्थानीय
        रूप से किसी $\mathcal C^k$ प्रतिचित्रण $\psi
        \colon A \subseteq \R^d \to \R^{n-d}$ का आलेख है।
  \item (\emph{प्राचलन}) कोई $\mathcal C^k$ \emph{निमज्जन} $\varphi \colon A \subseteq \R^d \to
        \R^n$ ($D\varphi(u)$
        एकैकी) ऐसा है कि $A$ विवृत हो और किसी विवृत $\Omega \ni a$ के
        लिए $\varphi$ $A$ से $M \cap
        \Omega$ पर \omterm{def:b3:topology:continuity}{समस्थितिकता} हो।
\end{enumerate}
\end{theorem}

\begin{proof}
(1)$\Rightarrow$(2): $F = (\Phi_{d+1}, \dots, \Phi_n)$ ($\Phi$ के अंतिम निर्देशांक): यह अवगाहन है
($D\Phi$ व्युत्क्रमणीय)।
(2)$\Rightarrow$(3): $DF(a)$ आच्छादक: अतः याकोबीय का कोई $q \times q$ उपसारणिक
व्युत्क्रमणीय है ($q = n - d$); निर्देशांकों के क्रमचय के बाद $D_yF(a)$
व्युत्क्रमणीय है, और अंतर्निहित फलन प्रमेय (\cref{thm:b3:submanifolds:implicit}) $M$ को
स्थानीय रूप से आलेख $y = \psi(x)$ के रूप में व्यक्त कर देती है।
(3)$\Rightarrow$(4): $\varphi(x) = (x, \psi(x))$: यह एक निमज्जन है (अवकलज $\bigl(\begin{smallmatrix}I\\
D\psi\end{smallmatrix}\bigr)$ एकैकी), और
आलेख पर \omterm{def:b3:topology:continuity}{समस्थितिकता} (प्रतिलोम: प्रक्षेप, जो \omterm{def:b3:topology:continuity}{संतत} है)।
(4)$\Rightarrow$(1): मान लीजिए $\varphi(u_0) = a$; $\operatorname{im}D\varphi(u_0)$ को किसी पूरक $E$ ($\dim E = n - d$) से
पूरा कीजिए और $A\times E$ पर $\Theta(u, v) = \varphi(u) + v$ परिभाषित कीजिए: $D\Theta(u_0, 0)$ एकैकी आच्छादक
है (प्रतिबिंब में $\operatorname{im}D\varphi(u_0)$ और $E$ हैं), अतः $\Theta$ एक स्थानीय अवकल
समरूपता है (\cref{thm:b3:submanifolds:ift}); और उसका प्रतिलोम $\Phi$ सीधा कर देता है:
$a$ के निकट $M$ के बिंदु ठीक $\varphi(u) = \Theta(u, 0)$ हैं --- और इसके लिए (4) की
\omterm{def:b3:topology:continuity}{समस्थितिकता} परिकल्पना यह सुनिश्चित करती है कि छोटे $\Omega$ के लिए
$M\cap\Omega$ में कोई अन्य पर्त न हो (छोटे $v \neq 0$ वाले $a$ के निकट के
$\varphi(u') + v = m \in
M$ को अपवर्जित करना होगा: \omterm{def:b3:topology:continuity}{समस्थितिकता} गुण से $u_0$ के निकट किसी
$u''$ के लिए $m = \varphi(u'')$, और $\Theta$ की स्थानीय एकैकीयता $v = 0$ अनिवार्य
कर देती है)। तब सिकोड़ने तक $\Phi(M\cap\Omega) = (A\times\{0\})
\cap\Phi(\Omega)$।
\end{proof}

\begin{example}\label{ex:b3:submanifolds:examples}
गोलक $S^{n-1} = \{\norm x^2 = 1\}$: $\R^n\setminus\{0\}$ ($DF(x) = 2x^{\mathsf T} \ne 0$) पर अवगाहन $F(x) = \norm x_2^2 - 1$ का स्तर समुच्चय: अतः
विमा $n - 1$ का $\mathcal C^\infty$ उपबहुविध। $\R^3$ में टोरस: $\bigl(\sqrt{x^2 + y^2} -
R\bigr)^2 + z^2 - r^2$ का स्तर
समुच्चय ($0 < r < R$)। शंकु $\{x^2 + y^2 =
z^2\}$ $0$ पर उपबहुविध \emph{नहीं} है
(\cref{exo:b3:submanifolds:1})। आव्यूह समूह: $SL_n$ और $O_n$ $M_n(\R)$ के उपबहुविध हैं
(\cref{exo:b3:submanifolds:5,exo:b3:submanifolds:6}) --- यही ली सिद्धांत का प्रस्थान बिंदु है।
\end{example}

\section{स्पर्श समष्टियाँ}

\begin{definition}\label{def:b3:submanifolds:tangent}
मान लीजिए $M$ कोई $d$-उपबहुविध है और $a \in M$। \emph{स्पर्श
समष्टि}\index{स्पर्श समष्टि} $T_aM$ $\gamma(0) = a$ वाले $\mathcal C^1$ वक्रों $\gamma \colon
\intoo{-\varepsilon}\varepsilon \to M$
के वेग सदिशों $\gamma'(0)$ का समुच्चय है।
\end{definition}

\begin{proposition}\label{prop:b3:submanifolds:tangent}
$T_aM$ $\R^n$ की एक $d$-विमीय सदिश उपसमष्टि है, और:
\begin{enumerate}
  \item यदि अवगाहन $F$ के साथ स्थानीय रूप से $M = F^{-1}(0)$ हो: $T_aM = \ker DF(a)$;
  \item यदि $M$ निमज्जन $\varphi$ ($\varphi(u_0) = a$) से प्राचलित हो: $T_aM =
        \operatorname{im}D\varphi(u_0)$।
\end{enumerate}
\end{proposition}

\begin{proof}
$M$ के वक्र $F(\gamma(t)) = 0$ संतुष्ट करते हैं; $0$ पर शृंखला नियम $DF(a)\gamma'(0) = 0$
देता है: अतः $T_aM \subseteq \ker DF(a)$। विलोमतः, सीधाकरण (\cref{thm:b3:submanifolds:%
characterizations}(1)) $\R^d\times\{0\}$ की रेखाओं
को $M$ के वक्रों पर ले जाता है: अर्थात् किसी $d$-विमीय
उपसमष्टि का प्रत्येक सदिश साकार हो जाता है; और विमाओं की तुलना
($\dim\ker DF(a) = n - (n - d) = d$) (1) में समता अनिवार्य कर देती है, तथा वही परिवहन तर्क (2)
दे देता है ($A$ की सरल रेखाओं पर $D\varphi(u_0)$ लगाइए; फिर विमाएँ)।
\end{proof}

\begin{theorem}[लाग्राँज गुणक]
\label{thm:b3:submanifolds:lagrange}
मान लीजिए $M = F^{-1}(0)$, जहाँ $F = (F_1, \dots, F_q) \colon \Omega
\to \R^q$ कोई $\mathcal C^1$ अवगाहन है, और मान लीजिए
$f \colon
\Omega \to \R$ $\mathcal C^1$ है। यदि प्रतिबंध $f\restriction_M$ का $a \in M$ पर कोई स्थानीय
चरम-मान हो, तो ऐसे अद्वितीय वास्तविक $\lambda_1, \dots, \lambda_q$ (\emph{\omterm{thm:b3:submanifolds:lagrange}{लाग्राँज
गुणक}}) हैं कि\index{लाग्राँज गुणक}
\[
\nabla f(a) = \lambda_1\nabla F_1(a) + \dots +
\lambda_q\nabla F_q(a) .
\]
\end{theorem}

\begin{proof}
$a$ से होकर जाने वाले $M$ के प्रत्येक वक्र $\gamma$ के लिए:
$t \mapsto
f(\gamma(t))$ का $0$ पर स्थानीय चरम-मान होता है, अतः $0 =
\frac{\dd}{\dd t}f(\gamma(t))\big|_0 = \langle\nabla f(a),
\gamma'(0)\rangle$: इसलिए $\nabla f(a) \perp T_aM = \ker DF(a)$
(\cref{prop:b3:submanifolds:tangent})। अब $\ker DF(a)^\perp
= \operatorname{im}DF(a)^{\mathsf T}$ (दूसरे वर्ष की कोटि/लांबिकता सर्वसमिका, अथवा
परिमित विमा में \cref{exo:b3:hilbert:8}: $(\ker T)^\perp = \operatorname{im}T^*$), जो प्रवणकों $\nabla F_i(a)$ से फैला है ---
और वे स्वतंत्र हैं, क्योंकि $DF(a)$ आच्छादक है: अतः गुणक विद्यमान
और अद्वितीय हैं।
\end{proof}

\begin{example}[स्पेक्ट्रमी प्रमेय, ज्यामितीय रूप से]
\label{ex:b3:submanifolds:rayleigh}
मान लीजिए $A$ कोई वास्तविक सममित $n\times n$ आव्यूह है, और गोलक
$S^{n-1}$ पर $f(x) = \langle Ax, x\rangle$ को अधिकतम कीजिए (जो \omterm{def:b3:topology:compact}{संहत} है: अतः अधिकतम किसी
$v_1$ पर प्राप्त होता है)। $F(x) = \norm x^2 - 1$ के साथ लाग्राँज: $\nabla f = 2Ax$ और $\nabla F =
2x$
से $Av_1 = \lambda_1v_1$ मिलता है --- अर्थात् एक अभिलक्षणिक सदिश, जिसमें $\lambda_1 = \max_{S^{n-1}}\langle Ax, x\rangle$।
अब $A$ को $v_1^\perp$ पर प्रतिबंधित कीजिए (जो अपरिवर्ती है: $\langle Av, v_1\rangle =
\langle v, Av_1\rangle = \lambda_1\langle v, v_1\rangle = 0$)
और दोहराइए: इससे अभिलक्षणिक सदिशों का प्रसामान्य लांबिक आधार मिल
जाता है। दूसरे वर्ष की स्पेक्ट्रमी प्रमेय, विशुद्ध इष्टतमीकरण से
पुनः सिद्ध --- और इसी तर्क की अनंत-विमीय छाया ने \cref{lem:b3:spectral:existence} सिद्ध किया
था।
\end{example}

\begin{method}\label{met:b3:submanifolds:method}
किसी समुच्चय को उपबहुविध सिद्ध करने के लिए: उसे स्थानीय रूप से
$F^{-1}(0)$ के रूप में प्रस्तुत कीजिए जहाँ $DF$ \emph{उस समुच्चय पर}
आच्छादक हो (सबसे सामान्य मार्ग), अथवा किसी आलेख के रूप में। उसकी
विमा और \omterm{def:b3:submanifolds:tangent}{स्पर्श समष्टि} परिकलित करने के लिए: $d = n - q$ और $T_a = \ker DF(a)$। उस पर
इष्टतमीकरण के लिए: लाग्राँज --- पर पहले सदा संहतता (या
प्रबलकारिता) जाँचिए, ताकि कोई चरम-मान विद्यमान हो जिस पर प्रमेय
लगाई जा सके; और स्मरण रखिए कि गुणक समीकरण केवल \emph{आवश्यक} है:
अतः सभी क्रांतिक बिंदु इकट्ठे कीजिए, फिर मानों की तुलना कीजिए।
आव्यूह समूहों के लिए तत्समक पर वक्रों का अवकलन करके स्पर्श
समष्टियाँ पहचानिए।
\end{method}

\section{समस्या: \texorpdfstring{$SO(3)$}{SO(3)} और क्वाटर्नियन}

\begin{problem}[{सप्ताहांत समस्या --- घूर्णन, समूह $S^3$, और
द्विक आवरण}]\label{pb:b3:submanifolds:1}
क्वाटर्नियन $\mathbb H = \{t + x\mathrm i + y\mathrm j +
z\mathrm k\}$ --- वह बीजगणित जिसके इकाई समूह में \cref{pb:b3:groups:1} का
$Q_8$ है --- त्रि-विमीय घूर्णनों का दो बार प्राचलन कर देते हैं:
``किसी इकाई क्वाटर्नियन से संयुग्मन'' प्रतिचित्रण अष्टि $\{\pm1\}$
वाली एक आच्छादक समाकारिता $S^3 \to
SO(3)$ है। हम सब कुछ बनाते हैं।
स्मरण/परिभाषा: गुणन $\R$-द्विरैखिक है, जिसमें $\mathrm i^2 = \mathrm j^2 = \mathrm k^2 =
\mathrm{ijk} = -1$; $q = t + x\mathrm i +
y\mathrm j + z\mathrm k$ का
संयुग्म $\bar q = t - x\mathrm i -
y\mathrm j - z\mathrm k$ है; और $N(q) = q\bar q = t^2 + x^2 + y^2
+ z^2$।

\medskip
\noindent\textbf{भाग I --- बीजगणित $\mathbb H$ और समूह $S^3$।}
\begin{enumerate}
  \item सत्यापित कीजिए कि $\mathbb H$ \omterm{ex:b3:groups:actions}{केंद्र} $\R$ वाला एक साहचर्य
        $\R$-बीजगणित है, कि $\overline{pq} =
        \bar q\,\bar p$, और यह कि $N(pq) = N(p)N(q)$ \emph{(एक साफ
        मार्ग: $q$ को $q =
        \alpha + \beta\mathrm j$ के साथ $2\times2$ सम्मिश्र आव्यूह $\bigl(\begin{smallmatrix}\alpha & \beta\\
        -\bar\beta & \bar\alpha\end{smallmatrix}\bigr)$
        के रूप में निरूपित कीजिए, और $\det$ का उपयोग कीजिए)}।
  \item निष्कर्ष निकालिए कि प्रत्येक $q \neq 0$ व्युत्क्रमणीय है
        ($q^{-1} =
        \bar q/N(q)$): अतः $\mathbb H$ एक (अक्रमविनिमेय) क्षेत्र है, और
        $S^3 = \{N(q) = 1\}$ एक समूह --- तथा $\R^4$ का \omterm{def:b3:topology:compact}{संहत} $3$-उपबहुविध भी
        (\cref{ex:b3:submanifolds:examples})।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- घूर्णन समाकारिता।} $\R^3$ की पहचान
\emph{शुद्ध क्वाटर्नियनों} $P = \{x\mathrm i +
y\mathrm j + z\mathrm k\}$ से कीजिए, और $q \in S^3$ के लिए $\rho_q(v) = q\,v\,\bar q$
परिभाषित कीजिए।
\begin{enumerate}[resume]
  \item दर्शाइए कि $\rho_q$ $P$ को $P$ में भेजता है \emph{(शुद्ध
        क्वाटर्नियन वे हैं जिनके लिए $\bar v = -v$)}, $\R$-रैखिक है,
        मानक सुरक्षित रखता है, और यह कि $\rho
        \colon q \mapsto \rho_q$ एक समूह समाकारिता
        $S^3
        \to O(3)$ है।
  \item अष्टि परिकलित कीजिए: $\rho_q = \mathrm{id}$ तभी और केवल तभी जब $q$
        $\mathrm i, \mathrm j, \mathrm k$ के साथ क्रमविनिमेय हो, अर्थात् तभी और केवल तभी जब
        $q \in \R\cap S^3 = \{\pm1\}$।
  \item $u \in P$, $N(u) = 1$ के साथ $q = \cos\frac\theta2 + \sin\frac\theta2\,u$ लिखिए ($q \in S^3$ के लिए यह सदा संभव
        क्यों है?)। दर्शाइए कि $\rho_q$ $u$ को स्थिर रखता है और
        समतल $u^\perp\cap P$ पर कोण $\theta$ के घूर्णन की तरह क्रिया करता है
        \emph{(लांबिक शुद्ध इकाइयों के लिए $uw = -wu$ का उपयोग करते
        हुए $w \perp u$ के लिए $\rho_q(w)$ परिकलित कीजिए --- यह सर्वसमिका
        गुणन सारणी से, अथवा $uw + wu =
        -2\langle u, w\rangle$ से सिद्ध कीजिए)}।
  \item निष्कर्ष निकालिए: $\operatorname{im}\rho \subseteq SO(3)$ (प्रत्येक $\rho_q$ ऊपर परिकलित अक्ष
        और कोण वाला घूर्णन है --- सारणिक $+1$ \omterm{def:b3:topology:connected}{संबद्ध} $S^3$ पर
        $q
        \mapsto \det\rho_q$ के \omterm{def:b3:topology:continuity}{सांतत्य} से, अथवा सीधे), और $\rho$ $SO(3)$ पर
        \emph{आच्छादक} है: $\R^3$ के प्रत्येक घूर्णन का कोई अक्ष
        होता है (सिद्ध कीजिए: $\det = 1$ वाले किसी वास्तविक $3\times3$
        लांबिक आव्यूह का अभिलक्षणिक मान $1$ होता है ---
        अभिलक्षणिक बहुपद पर विचार कीजिए) और इसलिए वह कोई $\rho_q$
        है। सार:
        \[
        SO(3) \;\cong\; S^3/\{\pm 1\} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- उपबहुविध के रूप में $SO(3)$;
रोद्रीग।}
\begin{enumerate}[resume]
  \item दर्शाइए कि $SO(3)$ $M_3(\R)$ का \omterm{def:b3:topology:compact}{संहत} $3$-विमीय उपबहुविध है,
        जिसमें $T_ISO(3) =$ प्रतिसममित आव्यूह हैं (\cref{exo:b3:submanifolds:5}; और सारणिक शर्त
        घटकों में से कुछ का चयन कर लेती है)।
  \item $u \in \R^3$ ($A_uv = u\wedge v$, सदिश गुणन) से \omterm{def:b3:topology:connected}{संबद्ध} प्रतिसममित आव्यूह
        $A_u$ के लिए \emph{रोद्रीग का सूत्र} सिद्ध कीजिए:
        \[
        \eu^{\theta A_u} = I + \sin\theta\,A_u + (1 -
        \cos\theta)\,A_u^2 \qquad (\norm u = 1)
        \]
        \emph{($A_u^3 = -A_u$ से: चरघातांकी श्रेणी को $A_u$ की घातों के
        अनुदिश बाँटिए)}, और उसे अक्ष $u$ तथा कोण $\theta$ के घूर्णन
        के रूप में पहचानिए। निष्कर्ष निकालिए कि $\exp$ प्रतिसममित
        आव्यूहों को $SO(3)$ \emph{पर} भेजता है।
  \item दोनों प्राचलनों को जोड़िए: दर्शाइए कि $q(t) = \cos\frac t2 +
        \sin\frac t2\,u$ के साथ $t
        \mapsto \rho_{q(t)}$
        घूर्णनों का एक-प्राचल समूह है जिसका $t = 0$ पर अवकलज $A_u$
        है --- अर्थात् क्वाटर्नियनी और \omterm{thm:b3:ode:matrixexp}{आव्यूह चरघातांकी} क्रमशः
        आधी और पूरी गति से वही कहानी कहते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- द्विक आवरण, अनुभव से।}
\begin{enumerate}[resume]
  \item दर्शाइए कि \omterm{def:b3:topology:pathconnected}{पथ} $q(t) = \cos\frac t2 + \sin\frac
        t2\,\mathrm k$, $t \in \intcc0{2\pi}$, $SO(3)$ में एक फंदा है (उसका
        प्रतिबिंब $\rho_{q(t)}$ तत्समक पर लौट आता है) जिसका क्वाटर्नियनी
        उठान फंदा \emph{नहीं} है: $q(2\pi) = -q(0)$। $t = 4\pi$ तक जारी रखने पर
        उठान बंद हो जाता है। एक छोटे अनुच्छेद में समझाइए कि यह
        क्या कहता है: $2\pi$ घूर्णन को संततता से पूर्ववत नहीं किया
        जा सकता जबकि $4\pi$ घूर्णन को किया जा सकता है (पट्टा
        युक्ति), क्योंकि $SO(3)$ के फंदे उसके द्विक आवरण $S^3$ में
        पकड़े जाते हैं।
  \item व्यावहारिक लाभांश भी निकालिए: घूर्णनों का संयोजन =
        क्वाटर्नियनों का गुणन ($9$ के बदले $4$ गुणनों जितने
        आँकड़े, और लांबिकता से कोई बहाव नहीं) --- इसे $\mathrm i$ और
        $\mathrm j$ के परितः दो चौथाई-घूर्णनों के संयोजन पर सत्यापित
        कीजिए: गुणनफल का अक्ष और कोण परिकलित कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- स्पष्ट आव्यूह: ऑयलर--रोद्रीग।} $q = a + b\mathrm i + c\mathrm j +
d\mathrm k \in S^3$
लिखिए, जिससे $a^2 + b^2 + c^2 + d^2 = 1$।
\begin{enumerate}[resume]
  \item गुणन सारणी से $\rho_q(\mathrm i)$ पूरा परिकलित कीजिए; फिर चक्रीय
        प्रतिस्थापन $\mathrm i \to \mathrm j \to \mathrm k
        \to \mathrm i$, $(b, c, d) \to (c, d, b)$ से $\rho_q(\mathrm
        j)$ और $\rho_q(\mathrm k)$ प्राप्त कीजिए
        \emph{(इसे उचित ठहराइए: $\mathrm i, \mathrm j,
        \mathrm k$ का चक्रीकरण $\mathbb
        H$ की किसी
        स्वसमाकारिता तक बढ़ जाता है, क्योंकि परिभाषक संबंध चक्रीय
        रूप से सममित हैं)}। निष्कर्ष निकालिए कि आधार $(\mathrm i, \mathrm j, \mathrm k)$ में
        $\rho_q$ का आव्यूह \emph{ऑयलर--रोद्रीग आव्यूह}
        \[
        R_q = \begin{pmatrix}
        a^2 + b^2 - c^2 - d^2 & 2(bc - ad) & 2(bd + ac)\\
        2(bc + ad) & a^2 - b^2 + c^2 - d^2 & 2(cd - ab)\\
        2(bd - ac) & 2(cd + ab) & a^2 - b^2 - c^2 + d^2
        \end{pmatrix}.
        \]
        है।
  \item (किसी घूर्णन को उलटकर पढ़ना) प्रश्न 5 और 8 के संकेतन में
        दर्शाइए कि
        \[
        \operatorname{tr}R_q = 4a^2 - 1 = 1 + 2\cos\theta,
        \qquad
        \tfrac12\bigl(R_q - R_q^{\mathsf T}\bigr) =
        \sin\theta\,A_u,
        \]
        इससे किसी घूर्णन आव्यूह $R$ से $\pm q$ पुनः प्राप्त करने
        की कोई कलनविधि निकालिए: कोण अनुरेख से; अक्ष प्रतिसममित
        भाग से जब $0 < \theta < \pi$; और जब $\theta = \pi$ हो तब सर्वसमिका $R
        + I = 2\,uu^{\mathsf T}$ सिद्ध
        करके उसका उपयोग कीजिए।
  \item प्रश्न 11 के गुणनफल $q =
        \frac12(1 + \mathrm i + \mathrm j + \mathrm k)$ के लिए $R_q$ का मान निकालिए: एक
        क्रमचय आव्यूह प्रकट होता है। उस घूर्णन को पहचानिए और
        प्रश्न 11 में मिले अक्ष तथा कोण से मेल कराइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- $S^3$ के भीतर: $SU(2)$, \omterm{ex:b3:groups:actions}{संयुग्मन वर्ग},
चरघातांकी।}
\begin{enumerate}[resume]
  \item दर्शाइए कि प्रश्न 1 का आव्यूह \omterm{def:b3:representations:rep}{निरूपण} (उसे $\Phi$ कहिए)
        $S^3$ से \emph{विशेष एकात्मक समूह}
        \[
        SU(2) = \bigl\{U \in M_2(\C) : U^*U = I,\ \det U =
        1\bigr\}
        \]
        पर एक समूह तुल्याकारिता तक प्रतिबंधित हो जाता है
        \emph{(आच्छादकता के लिए किसी व्यापक $2\times2$ सम्मिश्र आव्यूह
        के लिए समीकरण $U^{-1} = U^*$ और $\det U = 1$ लिखिए)}।
  \item दर्शाइए कि वास्तविक भाग $S^3$ पर एक संयुग्मन अपरिवर्ती है
        --- सभी $p \in S^3$ के लिए $\operatorname{Re}(pq\bar p) =
        \operatorname{Re}q$ --- और विलोमतः, कि समान
        वास्तविक भाग वाले दो इकाई क्वाटर्नियन $S^3$ में संयुग्म
        होते हैं \emph{(इसे एक इकाई शुद्ध अक्ष को दूसरे पर ले
        जाने तक सिमटाइए, जो भाग II दे देता है)}। $S^3$ के संयुग्मन
        वर्गों का ज्यामितीय वर्णन कीजिए; उन्हें $SU(2)$ में अनूदित
        कीजिए (अनुरेख के स्तर समुच्चय); और $\rho$ से प्रक्षेपित
        कीजिए: दो घूर्णन $SO(3)$ में संयुग्म हैं तभी और केवल तभी जब
        उनका कोण $\theta \in \intcc0\pi$ एक ही हो।
  \item $\mathbb H$ पर $\exp$ को चरघातांकी श्रेणी से परिभाषित कीजिए;
        $\abs q =
        \sqrt{N(q)}$ के लिए $\abs{pq} = \abs p\,\abs q$ का उपयोग करते हुए निरपेक्ष अभिसरण
        जाँचिए। किसी इकाई शुद्ध $u$ और $\theta
        \in \R$ के लिए दर्शाइए
        \[
        \exp(\theta u) = \cos\theta + \sin\theta\,u ,
        \]
        इससे निष्कर्ष निकालिए कि $\exp$ अधिसमतल $P$ को $S^3$ पर
        भेजता है, और जाँचिए कि $\rho_{\exp(su)} =
        \eu^{2sA_u}$: अर्थात् प्रश्न 9 की
        अर्ध-कोण परिघटना फिर से।
  \item शुद्ध क्वाटर्नियनों $v, w$ के लिए गुणन नियम $vw = -\langle v, w\rangle + v\wedge w$ सिद्ध
        कीजिए, अतः \omterm{def:b3:groups:derived}{क्रमविनिमेयक} सर्वसमिका $vw - wv = 2\,v\wedge w$; और प्रश्न 8 के
        आव्यूहों के लिए $[A_v, A_w] = A_{v\wedge w}$ भी सिद्ध कीजिए। निष्कर्ष निकालिए कि
        वक्रों $t \mapsto \exp(tv)$ के अनुदिश $1$ पर $\rho$ का अवकलज $P$ से
        प्रतिसममित आव्यूहों पर रैखिक तुल्याकारिता $v \mapsto 2A_v$ है, और यह
        कि वह क्वाटर्नियन \omterm{def:b3:groups:derived}{क्रमविनिमेयक} को आव्यूह \omterm{def:b3:groups:derived}{क्रमविनिमेयक} पर
        ले जाता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग VII --- वैश्विक संरचना।}
\begin{enumerate}[resume]
  \item (कोई \omterm{def:b3:topology:continuity}{संतत} खंड नहीं) मान लीजिए $s \colon SO(3) \to
        S^3$ $\rho \circ s =
        \operatorname{id}$ के साथ \omterm{def:b3:topology:continuity}{संतत} है।
        प्रश्न 10 के फंदे $R(t) =
        \rho_{q(t)}$ के लिए $t \in \intcc0{2\pi}$ हेतु $\varepsilon(t) =
        s(R(t))\,q(t)^{-1}$ रखिए।
        दर्शाइए कि $\varepsilon$ $\{\pm1\}$ में मान लेते हुए \omterm{def:b3:topology:continuity}{संतत} है, और
        विरोधाभास निकालिए: अर्थात् प्रत्येक घूर्णन को निरूपित
        करने वाले इकाई क्वाटर्नियन का कोई \omterm{def:b3:topology:continuity}{संतत} वैश्विक चयन विद्यमान
        नहीं है।
  \item (गोला मॉडल) मान लीजिए $\bar B \subseteq \R^3$ त्रिज्या $\pi$ का संवृत गोला
        है और $E(0) = I$ के साथ $E(v) = \eu^{A_v}$। दर्शाइए कि $E$ $\bar B$ को $SO(3)$
        पर भेजता है, विवृत गोले पर एकैकी है, और परिसीमा गोलक पर
        ठीक प्रतिध्रुवों की पहचान कर देता है: $E(\pi u) = E(-\pi u) = 2uu^{\mathsf T} - I$, और कोई अन्य
        संपात नहीं। अतः $SO(3)$ प्रतिध्रुवी परिसीमा बिंदुओं को
        जोड़ा हुआ गोला है --- अर्थात् प्रक्षेपी समष्टि $\mathbb{RP}^3$ ---
        और कोई व्यास प्रश्न 10 का असंकोच्य फंदा बन जाता है।
  \item दर्शाइए कि $\rho_p\rho_q\rho_p^{-1} = \rho_{pq\bar
        p}$; कि $SO(3)$ के स्वप्रतिलोमी अवयव ($R^2 = I$
        वाले $R \neq
        I$) ठीक अर्ध-घूर्णन $\rho_w$ हैं जहाँ $w$ कोई इकाई
        शुद्ध क्वाटर्नियन है; और यह कि $SO(3)$ का \omterm{ex:b3:groups:actions}{केंद्र} तुच्छ है।
  \item दर्शाइए कि प्रत्येक घूर्णन दो अर्ध-घूर्णनों का गुणनफल है:
        $q = \cos\frac\theta2 +
        \sin\frac\theta2\,u$ के लिए कोई इकाई शुद्ध $w \perp
        u$ चुनिए, जाँचिए कि $w' = qw$
        पुनः कोई इकाई शुद्ध क्वाटर्नियन है, और $\rho_q =
        \rho_{w'}\rho_w$ सत्यापित
        कीजिए। दोनों अक्ष कहाँ हैं, और उनके बीच कौन-सा कोण है?
  \item संस्थितिक सार निकालिए: $SO(3)$ \omterm{def:b3:topology:compact}{संहत} और \omterm{def:b3:topology:pathconnected}{पथ-संबद्ध} है (दो
        उपपत्तियाँ दीजिए: $\rho$ के अंतर्गत $S^3$ का \omterm{def:b3:topology:continuity}{संतत}
        प्रतिबिंब; और $\exp$ का प्रतिबिंब), जबकि $O(3)$ के ठीक दो
        \omterm{def:b3:topology:components}{संबद्ध घटक} हैं, और प्रत्येक $SO(3)$ का समस्थितिक है।

  \item (एक संयोजन, तीन प्रकार से) मान लीजिए $R_1$ $z$-अक्ष के
        परितः $\frac\pi2$ का घूर्णन है और $R_2$ $x$-अक्ष के परितः
        $\frac\pi2$ का। $R_2R_1$ का अक्ष और कोण परिकलित कीजिए: (क) दोनों
        $3\times3$ आव्यूहों को गुणा करके और अनुरेख/प्रतिसममित भाग (भाग
        V) का उपयोग करके; (ख) संगत इकाई क्वाटर्नियनों $q_2q_1$ को
        गुणा करके। जाँचिए कि दोनों उत्तर मेल खाते हैं: कोण $\frac{2\pi}3$,
        अक्ष $\frac1{\sqrt3}(1, -1, 1)$।
  \item (\omterm{ex:b3:conformal:mobius}{केली रूपांतर}) प्रतिसममित $K$ के लिए दर्शाइए कि $I + K$
        व्युत्क्रमणीय है और
        \[
        C(K) = (I - K)(I + K)^{-1} \in SO(n),
        \]
        जिसमें $-1$ $C(K)$ का अभिलक्षणिक मान कभी नहीं होता;
        दर्शाइए कि $K \mapsto C(K)$ प्रतिसममित आव्यूहों से $\{R \in SO(n) : -1 \notin
        \operatorname{Sp}R\}$ पर एक एकैकी
        आच्छादक संगति है, जिसका प्रतिलोम $R \mapsto (I -
        R)(I + R)^{-1}$ है। ($SO(n)$ का एक
        परिमेय चार्ट, जो \cref{exo:b3:submanifolds:11} के अबीजीय $\exp$ का सहचर है।)
\end{enumerate}
\end{problem}
