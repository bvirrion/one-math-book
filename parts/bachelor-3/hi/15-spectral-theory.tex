\chapter{संहत संकारक और स्पेक्ट्रमी प्रमेय}
\label{ch:b3:spectral}

विकर्णन परिमित-विमीय रैखिक बीजगणित का मुकुट-रत्न है: किसी सममित
आव्यूह का अभिलक्षणिक सदिशों का प्रसामान्य लांबिक आधार होता है।
अनंत विमा में यह व्यापक परिबद्ध \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} संकारकों के लिए विफल हो
जाता है --- $L^2(\intcc01)$ पर $x$ से गुणन का तो कोई अभिलक्षणिक मान ही नहीं
है (\cref{exo:b3:spectral:6}) --- पर उन संकारकों के लिए वह \omterm{def:b3:complete:complete}{पूर्ण} रूप में बचा रहता है जो
\emph{लगभग परिमित-विमीय} हैं: अर्थात् \omterm{def:b3:spectral:compact}{संहत} संकारकों के लिए। \omterm{def:b3:spectral:compact}{संहत}
\omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} संकारकों के लिए स्पेक्ट्रमी प्रमेय अनुप्रयुक्त कार्यकीय
विश्लेषण की सबसे अधिक प्रयुक्त प्रमेय है: वह समाकल समीकरणों का
विकर्णन कर देती है, \omterm{thm:b3:spectral:fredholm}{फ्रेडहोम विकल्प} चलाती है, और (सप्ताहांत समस्या)
कंपमान डोरी हल कर देती है, जिससे विशुद्ध संकारक सिद्धांत से फूरिये
विश्लेषण का ज्या आधार निकल आता है --- और विदाई उपहार के रूप में
किसी अनुरेख सूत्र से ऑयलर का $\zeta(2) = \frac{\pi^2}6$ भी झड़ पड़ता है। सर्वत्र $H$
$\C$ (या $\R$; कथन अनुकूलित हो जाते हैं) पर एक \omterm{def:b3:hilbert:inner}{हिल्बर्ट समष्टि} है,
और संकारक परिबद्ध हैं।

\section{संहत संकारक}

\begin{definition}\label{def:b3:spectral:compact}
$T \in \mathcal L(E, F)$ ($E, F$ बानाख) \emph{संहत}\index{संहत संकारक} कहलाता है यदि
इकाई गोले का प्रतिबिंब $T(B)$ $F$ में सापेक्षतः संहत हो ---
समतुल्य रूप से, यदि प्रत्येक परिबद्ध अनुक्रम $(x_n)$ का कोई ऐसा
उपानुक्रम हो जिसके लिए $(Tx_{n_k})$ अभिसारी हो। परिमित कोटि के संकारक संहत
होते हैं (परिमित विमा में परिबद्ध समुच्चय); और किसी अनंत-विमीय
समष्टि का तत्समक कभी नहीं (रीस की प्रमेय, दूसरा वर्ष)।
\end{definition}

\begin{proposition}\label{prop:b3:spectral:ideal}
\omterm{def:b3:spectral:compact}{संहत संकारक} $\mathcal K(E, F)$ $\mathcal L(E, F)$ की एक संवृत उपसमष्टि बनाते हैं, और एक
द्विपक्षीय \omterm{def:b3:rings:ideal}{आदर्श} भी: $S$ \omterm{def:b3:spectral:compact}{संहत} $\Rightarrow$ परिबद्ध $A, B$ के लिए $AS$ और
$SB$ \omterm{def:b3:spectral:compact}{संहत}। इसके अतिरिक्त, किसी \omterm{def:b3:hilbert:inner}{हिल्बर्ट समष्टि} में प्रत्येक \omterm{def:b3:spectral:compact}{संहत
संकारक} परिमित कोटि के संकारकों की मानक-सीमा होता है।
\end{proposition}

\begin{proof}
उपसमष्टि: अनुक्रम वाले अभिलक्षण से स्पष्ट। \omterm{def:b3:rings:ideal}{आदर्श}: परिबद्ध
प्रतिचित्रण अभिसारी अनुक्रमों को अभिसारी पर और परिबद्ध को परिबद्ध
पर भेजते हैं। संवृतता: मान लीजिए $T_n \to T$, जहाँ $T_n$ \omterm{def:b3:spectral:compact}{संहत} हैं, और
$1$ से परिबद्ध $(x_k)$; कोई विकर्ण निष्कर्षण प्रत्येक $n$ के लिए
$(T_nx_{k_j})_j$ को अभिसारी बना देता है; तब $(Tx_{k_j})$ कोशी है, क्योंकि
\[
\norm{Tx_{k_j} - Tx_{k_l}} \leq 2\vertiii{T - T_n} +
\norm{T_nx_{k_j} - T_nx_{k_l}} ,
\]
और यहाँ पहले $n$ चुनिए, फिर सूचकांक। हिल्बर्ट समष्टियों में
आसन्नन: मान लीजिए $T$ \omterm{def:b3:spectral:compact}{संहत} है, अतः $K = \overline{T(B)}$ \omterm{def:b3:spectral:compact}{संहत}; $\varepsilon$ दिया हो तो
$K$ को परिमित संख्या के गोलों $B(y_i, \varepsilon)$ से आच्छादित कीजिए और मान लीजिए
$P$ $V = \operatorname{Vect}(y_1, \dots, y_m)$ (जो संवृत है: परिमित-विमीय) पर लांबिक प्रक्षेप है। तब
$PT$ की कोटि परिमित है, और $\norm x \leq 1$ के लिए: $\norm{Tx - y_i} <
\varepsilon$ वाला $y_i$ चुनने पर
\[
\norm{Tx - PTx} \leq \norm{Tx - y_i} + \norm{P(y_i - Tx)}
\leq 2\varepsilon
\]
($y_i = Py_i$; $\vertiii P \leq 1$): अतः $\vertiii{T - PT} \leq
2\varepsilon$।
\end{proof}

\begin{example}\label{ex:b3:spectral:examples}
(क) $\ell^2$ पर विकर्ण संकारक: $T(x_n) = (\lambda_nx_n)$ \omterm{def:b3:spectral:compact}{संहत} है तभी और केवल तभी जब $\lambda_n \to 0$
(\cref{exo:b3:spectral:2})।
(ख) $\mathcal C(\intcc01)$ पर नाभिक संकारक: आस्कोली से \omterm{def:b3:spectral:compact}{संहत} (\cref{exo:b3:complete:7})।
(ग) \emph{हिल्बर्ट--श्मिट संकारक}\index{हिल्बर्ट--श्मिट संकारक}:
$k \in L^2(\intcc01^2)$ के लिए
\[
(T_kf)(x) = \int_0^1k(x, y)\,f(y)\,\dd y
\]
$L^2(\intcc01)$ पर एक \omterm{def:b3:spectral:compact}{संहत संकारक} परिभाषित करता है जिसके लिए
$\vertiii{T_k} \leq \norm k_{L^2}$
(\cref{exo:b3:spectral:4}: $k$ के आधार प्रसार को छाँटने पर $T_k$ परिमित कोटि के
संकारकों की सीमा के रूप में दिख जाता है)।
\end{example}

\section{स्वसंलग्न संकारक}

\begin{definition}\label{def:b3:spectral:selfadjoint}
$T \in \mathcal L(H)$ \emph{स्वसंलग्न}\index{स्वसंलग्न संकारक} कहलाता है यदि $T = T^*$
(\cref{exo:b3:hilbert:8}), अर्थात् सभी $x, y$ के लिए $\langle Tx, y\rangle = \langle x, Ty\rangle$। तब प्रत्येक $x$ के लिए
$\langle x, Tx\rangle \in \R$ (वह अपने ही संयुग्म के बराबर है)।
\end{definition}

\begin{proposition}\label{prop:b3:spectral:sanorm}
\omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} $T$ के लिए:\index{संख्यात्मक त्रिज्या}
\[
\vertiii T = \sup_{\norm x \leq 1}\ \abs{\langle x,
Tx\rangle} .
\]
$T$ के अभिलक्षणिक मान वास्तविक होते हैं, और भिन्न अभिलक्षणिक
मानों के अभिलक्षणिक सदिश लांबिक होते हैं।
\end{proposition}

\begin{proof}
मान लीजिए $M$ वह उच्चतम है; कोशी--श्वार्ज़ से $M \leq \vertiii T$। विलोमतः,
ध्रुवण प्रकार की सर्वसमिका
\[
\langle x{+}y, T(x{+}y)\rangle - \langle x{-}y,
T(x{-}y)\rangle = 4\operatorname{Re}\langle y, Tx\rangle
\]
(प्रसार कीजिए; \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्नता} से क्रॉस पद $\langle y, Tx\rangle + \langle x,
Ty\rangle = 2\operatorname{Re}\langle y, Tx\rangle$) समांतर चतुर्भुज नियम
के साथ
\[
4\operatorname{Re}\langle y, Tx\rangle \leq
M\bigl(\norm{x{+}y}^2 + \norm{x{-}y}^2\bigr) = 2M\bigl(\norm
x^2 + \norm y^2\bigr).
\]
दे देती है। $Tx \neq 0$ वाले $\norm x = 1$ के लिए $y = Tx/\norm{Tx}$ लीजिए: $4\norm{Tx} \leq 4M$। अतः $\vertiii T \leq M$।
अभिलक्षणिक मान: $Tx = \lambda x$, और $x \ne 0$ से $\lambda\norm x^2 = \langle
x, Tx\rangle \in \R$ मिलता है। लांबिकता:
वास्तविक $\lambda \neq \mu$ के साथ $\lambda\langle x,
y\rangle = \langle Tx, y\rangle = \langle x, Ty\rangle =
\mu\langle x, y\rangle$।
\end{proof}

\section{स्पेक्ट्रमी प्रमेय}

\begin{lemma}[किसी चरम अभिलक्षणिक मान का अस्तित्व]
\label{lem:b3:spectral:existence}
मान लीजिए $T \neq 0$ \omterm{def:b3:spectral:compact}{संहत} और \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} है। तब $\vertiii T$ या $-\vertiii T$ $T$ का
अभिलक्षणिक मान होता है।
\end{lemma}

\begin{proof}
\cref{prop:b3:spectral:sanorm} से $\langle x_n, Tx_n\rangle \to \mu$ वाले इकाई सदिश $x_n$ चुनिए, जहाँ $\abs\mu =
\vertiii T > 0$ (चिह्न स्थिर
करने के लिए किसी उपानुक्रम पर जाइए)। तब
\[
\norm{Tx_n - \mu x_n}^2
= \norm{Tx_n}^2 - 2\mu\langle x_n, Tx_n\rangle + \mu^2
\leq \vertiii T^2 - 2\mu\langle x_n, Tx_n\rangle + \mu^2
\longrightarrow 2\mu^2 - 2\mu\cdot\mu = 0 .
\]
संहतता से कोई उपानुक्रम $Tx_{n_k} \to y$; तब $\mu
x_{n_k} = Tx_{n_k} - (Tx_{n_k} - \mu x_{n_k}) \to y$, अतः $x_{n_k} \to x = y/\mu$, जो एक इकाई सदिश
है, और \omterm{def:b3:topology:continuity}{सांतत्य} $Tx = \mu x$ दे देता है।
\end{proof}

\begin{theorem}[संहत स्वसंलग्न संकारकों के लिए स्पेक्ट्रमी
प्रमेय]\label{thm:b3:spectral:spectral}
मान लीजिए $T$ किसी \omterm{def:b3:hilbert:inner}{हिल्बर्ट समष्टि} $H$ पर \omterm{def:b3:spectral:compact}{संहत} \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न
संकारक} है।\index{स्पेक्ट्रमी प्रमेय (संहत स्वसंलग्न)}
\begin{enumerate}
  \item $H$ $T$ के अभिलक्षणिक सदिशों का एक प्रसामान्य लांबिक
        निकाय $(e_n)_{n \in N}$ ($N$ परिमित या गणनीय) स्वीकार करता है,
        जिसके अभिलक्षणिक मान $(\lambda_n)$ वास्तविक और शून्येतर हैं, और
        जिसके लिए
        \[
        Tx = \sum_{n\in N}\lambda_n\,\langle e_n, x\rangle\,
        e_n \qquad (x \in H),
        \]
        तथा $H = \ker T \,\oplus^\perp\,
        \overline{\operatorname{Vect}}(e_n : n \in N)$।
  \item यदि $N$ अनंत हो, तो $\lambda_n \to 0$; प्रत्येक $\delta > 0$ के लिए केवल
        परिमित संख्या के $n$ $\abs{\lambda_n} \geq \delta$ को संतुष्ट करते हैं, और
        प्रत्येक अभिलक्षणिक समष्टि $\ker(T - \lambda)$, $\lambda \neq 0$, परिमित-विमीय है।
  \item $\ker T$ के किसी प्रसामान्य लांबिक आधार से $(e_n)$ को पूरा
        करने पर, जब $H$ पृथक्करणीय हो, $H$ का अभिलक्षणिक
        सदिशों से बना एक प्रसामान्य लांबिक आधार मिल जाता है: अतः
        $T$ विकर्णित हो जाता है।
\end{enumerate}
\end{theorem}

\begin{proof}
पहले (2)। यदि अनंत संख्या के प्रसामान्य लांबिक अभिलक्षणिक सदिशों
$x_k$ के लिए $\abs{\lambda_{(k)}} \geq \delta$ होता, तो (लांबिकता, पाइथागोरस) $\norm{Tx_k - Tx_l}^2 =
\lambda_{(k)}^2 + \lambda_{(l)}^2 \geq 2\delta^2$: अर्थात्
$(Tx_k)$ का कोई अभिसारी उपानुक्रम नहीं, जो परिबद्ध $(x_k)$ पर $T$ की
संहतता का खंडन है। इससे प्रत्येक $\delta$ के लिए $\intoo{-\delta}\delta$ के बाहर के
अभिलक्षणिक मानों की कुल बहुलता परिमित हो जाती है; और गणनीयता तथा
$\lambda_n \to 0$ इसी से निकलते हैं।

(1) मान लीजिए $H_0$ शून्येतर अभिलक्षणिक मानों वाले \emph{समस्त}
अभिलक्षणिक सदिशों का संवृत विस्तार है, जिन्हें ((2) से और प्रत्येक
परिमित-विमीय अभिलक्षणिक समष्टि के भीतर ग्राम--श्मिट से; अभिलक्षणिक
समष्टियों के बीच लांबिकता \cref{prop:b3:spectral:sanorm} से) अभिलक्षणिक मानों $\lambda_n \neq 0$ वाले
किसी प्रसामान्य लांबिक निकाय $(e_n)$ में व्यवस्थित किया गया है।
$T$ $H_0$ को $H_0$ में और $H_0^\perp$ को $H_0^\perp$ में भी भेजता है: $y
\perp H_0$
और अभिलक्षणिक सदिश $e$ के लिए $\langle e, Ty\rangle =
\langle Te, y\rangle = \lambda\langle e, y\rangle = 0$। प्रतिबंध $T' = T\restriction_{H_0^\perp}$ \omterm{def:b3:hilbert:inner}{हिल्बर्ट
समष्टि} $H_0^\perp$ पर \omterm{def:b3:spectral:compact}{संहत} \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} है; यदि $T' \neq
0$ हो, तो \cref{lem:b3:spectral:existence} $H_0^\perp$
के भीतर शून्येतर अभिलक्षणिक मान वाला $T$ का कोई अभिलक्षणिक सदिश
बना देता --- जो असंभव है, क्योंकि ऐसे सदिश $H_0$ में रहते हैं। अतः
$T' = 0$: इसलिए $H_0^\perp \subseteq \ker T$। विलोमतः प्रत्येक $e_n$ $\ker T \perp$ ($\langle e_n, z\rangle = \frac1{\lambda_n}\langle
Te_n, z\rangle = \frac1{\lambda_n}\langle e_n, Tz\rangle =
0$): $\ker T \subseteq H_0^\perp$,
जिससे $\ker T =
H_0^\perp$ और लांबिक विघटन मिल जाते हैं। प्रसार: $x = z + \sum_nc_ne_n$ के लिए
($z \in \ker T$, $c_n = \langle
e_n, x\rangle$; $H_0$ पर \cref{thm:b3:hilbert:parseval}(1)), $T$ का \omterm{def:b3:topology:continuity}{सांतत्य} $Tx = \sum_nc_n\lambda_ne_n$ दे देता
है।

(3) $\ker T$, जो किसी पृथक्करणीय समष्टि की संवृत उपसमष्टि है,
पृथक्करणीय है: अतः उसका कोई प्रसामान्य लांबिक आधार है (\cref{prop:b3:hilbert:gramschmidt}); और
(1) के विघटन से सम्मिलन $H$ का प्रसामान्य लांबिक आधार बन जाता है।
\end{proof}

\begin{theorem}[फ्रेडहोम विकल्प]
\label{thm:b3:spectral:fredholm}
मान लीजिए $T$ \omterm{def:b3:spectral:compact}{संहत} \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} है और $\lambda \in \R\setminus
\{0\}$।\index{फ्रेडहोम विकल्प}
\begin{enumerate}
  \item यदि $\lambda$ अभिलक्षणिक मान न हो, तो $T - \lambda I$ परिबद्ध प्रतिलोम
        के साथ एकैकी आच्छादक है: प्रत्येक $f$ के लिए समीकरण
        $Tx - \lambda x = f$ का ठीक एक हल है, जो $f$ पर संततता से निर्भर करता
        है।
  \item यदि $\lambda$ अभिलक्षणिक मान हो, तो $Tx - \lambda x = f$ हल्य है तभी और केवल
        तभी जब $f \perp \ker(T - \lambda I)$, और हल उस (परिमित-विमीय) अष्टि तक अद्वितीय
        होता है।
\end{enumerate}
\end{theorem}

\begin{proof}
\cref{thm:b3:spectral:spectral} ($z, w \in \ker T$) के अनुदिश $x = z + \sum c_ne_n$ और $f = w + \sum d_ne_n$ का विघटन कीजिए। समीकरण इस
रूप में पढ़ा जाता है
\[
-\lambda z = w, \qquad (\lambda_n - \lambda)\,c_n = d_n\
(n \in N).
\]
(1) $\lambda \notin \{\lambda_n\}\cup\{0\}$: स्पेक्ट्रमी प्रमेय के (2) से $\inf_n\abs{\lambda_n - \lambda} = \delta >
0$ (अभिलक्षणिक मान
केवल $0 \neq \lambda$ पर संचित होते हैं)। हल कीजिए: $z = -w/\lambda$, $c_n = d_n/(\lambda_n - \lambda)$, जिसमें
$\sum\abs{c_n}^2 \leq \delta^{-2}\sum\abs{d_n}^2$: अतः $\norm x \leq C\norm f$ वाला एक अद्वितीय हल।
(2) किसी परिमित समुच्चय $F$ में $n$ के लिए $\lambda = \lambda_{n}$: तब $n \in F$ के
लिए $(\lambda_n - \lambda)c_n = d_n$ की हल्यता के लिए $d_n = 0$ चाहिए, अर्थात् $f \perp e_n$ ($n \in F$),
अर्थात् $f
\perp \ker(T - \lambda I)$; और तब $c_n$, $n \in F$, स्वतंत्र रहते हैं।
\end{proof}

\begin{example}\label{ex:b3:spectral:minxy}
$L^2(\intcc01)$ पर मान लीजिए $Tf(x) = \int_0^1\min(x, y)f(y)\dd y$: यह वास्तविक सममित नाभिक वाला
\omterm{ex:b3:spectral:examples}{हिल्बर्ट--श्मिट संकारक} है: अतः \omterm{def:b3:spectral:compact}{संहत} और \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न}। $Tf = \lambda f$ हल करना:
संबंध $\bigl(Tf\bigr)(x) = \int_0^xyf(y)\dd y +
x\int_x^1f(y)\dd y$ दिखाता है कि $u = Tf$ $u'' = -f$ को संतुष्ट करता है (दो बार
अवकलन, जो \omterm{def:b3:topology:continuity}{संतत} $f$ के लिए वैध है, और $f \in L^2$ के लिए $Tf$ \omterm{def:b3:topology:continuity}{संतत} है:
प्रभावी अभिसरण), जिसमें $u(0)
= 0$ और $u'(1) = 0$। अतः अभिलक्षणिक फलन $\lambda u'' =
-u$,
$u(0) = 0$, $u'(1) = 0$ को हल करते हैं:
\[
u_n(x) = \sin\Bigl(\bigl(n + \tfrac12\bigr)\pi x\Bigr),
\qquad
\lambda_n = \frac{1}{\bigl(n + \frac12\bigr)^2\pi^2}
\quad (n \geq 0),
\]
और स्पेक्ट्रमी प्रमेय कहती है --- बिना किसी फूरिये सिद्धांत के ---
कि मानकीकरण के बाद ये ज्या $L^2(\intcc01)$ का प्रसामान्य लांबिक आधार बनाते
हैं ($T$ की अष्टि $0$ है: $Tf = 0$ से दो अवकलनों द्वारा लगभग
सर्वत्र $f = 0$ अनिवार्य हो जाता है)। सप्ताहांत समस्या कंपमान डोरी
के लिए इसी विचार-मंडल को चलाती है और अनुरेख से $\zeta(2)$ निकाल लेती है।
\end{example}

\begin{method}\label{met:b3:spectral:method}
कोई समाकल या अवकल समीकरण दिया हो: (1) उसे किसी समाकल संकारक $K$
के साथ $(I - \lambda K)u = f$ या $Ku = \lambda u$ के रूप में लिखिए; (2) $K$ की संहतता जाँचिए
(हिल्बर्ट--श्मिट नाभिक, अथवा आस्कोली) और यदि संभव हो तो
\omterm{def:b3:spectral:selfadjoint}{स्वसंलग्नता} भी (सममित वास्तविक नाभिक); (3) स्पेक्ट्रमी प्रमेय से
विकर्णन कीजिए अथवा हल्यता के लिए \omterm{thm:b3:spectral:fredholm}{फ्रेडहोम विकल्प} का आह्वान कीजिए;
(4) अभिलक्षणिक आधार में अस्तित्व, अद्वितीयता, स्थायित्व और हलों के
श्रेणी सूत्र पढ़ लीजिए। अवकल संकारक अपरिबद्ध होते हैं, पर उनके
\emph{प्रतिलोम} (ग्रीन संकारक) \omterm{def:b3:spectral:compact}{संहत} होते हैं: अतः पहले सदा
प्रतिलोम लीजिए।
\end{method}

\begin{exercise}[$\star\star$]\label{exo:b3:spectral:7}

(क) दर्शाइए कि $V$ \omterm{def:b3:spectral:compact}{संहत} है (नाभिक $\mathbf

\begin{exercise}[$\star\star\star$]\label{exo:b3:spectral:8}
(कूरां--फिशर) मान लीजिए $T$ \omterm{def:b3:spectral:compact}{संहत}, \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} और \emph{धनात्मक}
है, जिसके अभिलक्षणिक मान $\mu_1 \geq \mu_2 \geq \dots
> 0$ हैं (बहुलता सहित दोहराए गए,
अभिलक्षणिक सदिश $e_1, e_2,
\dots$)। दर्शाइए:
\[
\mu_{k} = \max_{\substack{V \subseteq H \\ \dim V = k}}\
\min_{\substack{x \in V\\ \norm x = 1}}\ \langle x, Tx\rangle
= \min_{\substack{W \subseteq H\\ \operatorname{codim}W =
k-1}}\ \max_{\substack{x\in W\\ \norm x = 1}}\ \langle x,
Tx\rangle .
\]
\emph{($V = \operatorname{Vect}(e_1,\dots,e_k)$ पर परीक्षा लीजिए; ऊपरी परिबंध के लिए किसी भी $V$
का $\operatorname{Vect}(e_k, e_{k+1}, \dots)$ प्रकार की समष्टियों से प्रतिच्छेदन लीजिए: विमा गिनने से
कोई शून्येतर प्रतिच्छेदन अनिवार्य हो जाता है।)} इससे निष्कर्ष
निकालिए कि अभिलक्षणिक मान $T$ पर एकदिष्ट रूप से निर्भर करते हैं
($T \leq S
\Rightarrow \mu_k(T) \leq \mu_k(S)$)।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spectral:10}

(ग) टिप्पणी: अस्वसंलग्न, असंहत संकारकों के लिए अभिलक्षणिक मानों का

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spectral:11}
(वर्गमूल) मान लीजिए $T$ किसी \omterm{def:b3:hilbert:inner}{हिल्बर्ट समष्टि} $H$ पर \omterm{def:b3:spectral:compact}{संहत},
\omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न}, धनात्मक ($\langle x, Tx\rangle \geq 0$) है, जिसका स्पेक्ट्रमी विघटन $Tx = \sum_n\mu_n\langle e_n,
x\rangle e_n$
($\mu_n > 0$) है।
(क) $Sx = \sum_n\sqrt{\mu_n}\,\langle e_n, x\rangle
e_n$ परिभाषित कीजिए; दर्शाइए कि $S$ \omterm{def:b3:spectral:compact}{संहत}, \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} और
धनात्मक है, $S^2
= T$ के साथ।
(ख) अद्वितीयता सिद्ध कीजिए: $R^2 = T$ वाला कोई भी \omterm{def:b3:spectral:compact}{संहत} धनात्मक
\omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} $R$ $T$ की अभिलक्षणिक समष्टियों को सुरक्षित रखता है
\emph{($RT = R^3 = TR$: $R$ $T$ के साथ क्रमविनिमेय है, अतः $R(\ker(T - \mu)) \subseteq \ker(T - \mu)$)}, और
$\ker(T - \mu)$ पर $R$ किसी परिमित-विमीय समष्टि पर $\mu\,\mathrm{id}$ का वर्ग देने वाला
धनात्मक संकारक है: वहाँ उसका विकर्णन कीजिए और प्रत्येक अभिलक्षणिक
समष्टि पर $R =
\sqrt\mu\,\mathrm{id}$ निष्कर्ष निकालिए, अतः $R = S$।
(ग) \cref{pb:b3:spectral:1} के डोरी संकारक $G$ के लिए $\sqrt G$ परिकलित कीजिए: ज्या
आधार पर किस नाभिक के अभिलक्षणिक मान $\frac1{n\pi}$ हैं? ($\sqrt G$ को नाभिकों
की $L^2$-सीमा के रूप में व्यक्त कीजिए; किसी संवृत रूप की आवश्यकता
नहीं।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spectral:12}
(एकल मान विघटन) मान लीजिए $T \in \mathcal L(H)$ \emph{\omterm{def:b3:spectral:compact}{संहत}} है, और आवश्यक नहीं कि
\omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} भी।
(क) दर्शाइए कि $T^*T$ \omterm{def:b3:spectral:compact}{संहत}, \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} और धनात्मक है; मान लीजिए
$(e_n)$ $T^*Te_n = s_n^2e_n$, $s_n > 0$ (\emph{एकल मान}) वाले अभिलक्षणिक सदिशों का
कोई प्रसामान्य लांबिक कुल है, जिसे $\ker(T^*T) = \ker T$ से पूरा किया गया है (इस
समता को सिद्ध कीजिए)।
(ख) $f_n = \frac{Te_n}{s_n}$ रखिए; दर्शाइए कि $(f_n)$ प्रसामान्य लांबिक है, और
\emph{एकल मान विघटन} स्थापित कीजिए:
\[
Tx = \sum_n s_n\,\langle e_n, x\rangle\,f_n
\qquad (x \in H),
\]
जिसमें अभिसरण $H$ में होता है।
(ग) निष्कर्ष निकालिए: $\vertiii T = \max_ns_n$; $T$ परिमित कोटि के संकारकों की
मानक-सीमा है (जिससे हिल्बर्ट समष्टियों के लिए \cref{prop:b3:spectral:ideal} का विलोम पुनः
सिद्ध हो जाता है); और \cref{exo:b3:spectral:7} के वोल्तेरा संकारक $V$ के लिए, जिसका
कोई अभिलक्षणिक मान नहीं है, समझाइए कि फिर भी एकल मान विघटन क्यों
विद्यमान है और उसके अवयव क्या हैं ($V^*V$ को किसी डोरी-प्रकार के
नाभिक संकारक के रूप में पहचानिए --- उसके अभिलक्षणिक मान स्पष्ट
परिकलित करना \cref{exo:b3:spectral:9} का क्षेत्र है)।
\end{exercise}

\section{समस्या: कंपमान डोरी और \texorpdfstring{$\zeta(2)$}{ZETA(2)}}

\begin{problem}[{सप्ताहांत समस्या --- ग्रीन संकारक, ज्या आधार,
और एक अनुरेख सूत्र}]\label{pb:b3:spectral:1}
हम स्थिर सिरों वाली कंपमान डोरी की अभिलक्षणिक मान समस्या ---
$-u'' = \nu u$, $u(0) = u(1) = 0$ --- संकारक सिद्धांत से हल करते हैं, बिना किसी फूरिये
परिकलन के ज्या प्रसामान्य लांबिक आधार प्राप्त करते हैं, और ग्रीन
संकारक के \emph{अनुरेख} के दो व्यंजकों की तुलना करके $\zeta(2)$ का मान
निकालते हैं। $L^2(\intcc01)$ पर परिभाषित कीजिए
\[
(Gf)(x) = \int_0^1 g(x,y)\,f(y)\,\dd y,
\qquad
g(x, y) = \min(x,y)\,\bigl(1 - \max(x,y)\bigr).
\]

\medskip
\noindent\textbf{भाग I --- ग्रीन संकारक।}
\begin{enumerate}
  \item दर्शाइए कि $g$ \omterm{def:b3:topology:continuity}{संतत} और सममित है, $0 \leq g
        \leq \frac14$ के साथ, और यह कि
        $G$ \omterm{def:b3:spectral:compact}{संहत} तथा \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} है (\cref{ex:b3:spectral:examples}(ग))।
  \item \omterm{def:b3:topology:continuity}{संतत} $f$ के लिए दर्शाइए कि $u = Gf$ \[
        -u'' = f, \qquad u(0) = u(1) = 0
        \] के साथ $\mathcal
        C^2$ है
        \emph{($u(x) = (1-x)\int_0^xyf(y)\dd y +
        x\int_x^1(1-y)f(y)\dd y$ लिखिए और दो बार अवकलन कीजिए)}। विलोमतः, यदि
        $u(0) = u(1)
        = 0$ के साथ $u \in \mathcal C^2$ हो, तो $G(-u'') = u$: अर्थात् $G$ डोरी संकारक
        का प्रतिलोम है।
  \item $\ker G = \{0\}$ दर्शाइए \emph{(यदि $f \in
        L^2$ के साथ $Gf = 0$ हो: \omterm{def:b3:topology:continuity}{संतत}
        $\varphi$ के सापेक्ष परीक्षा लीजिए, सममिति/फुबिनी से $G$ को
        $\varphi$ पर स्थानांतरित कीजिए, और मूल प्रमेयिका \cref{cor:b3:lp:fundlemma} का
        उपयोग कीजिए --- अथवा नियमितीकरण कीजिए)}, और यह भी कि $G$
        एक धनात्मक संकारक है: $\langle f, Gf\rangle \geq 0$। \emph{(\omterm{def:b3:topology:continuity}{संतत} $f$ के लिए:
        खंडशः, $u
        = Gf$ के साथ $\langle f, Gf\rangle = \int_0^1 (u')^2$; सघनता से निष्कर्ष निकालिए।)}
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- विकर्णन: ज्या आधार।}
\begin{enumerate}[resume]
  \item दर्शाइए कि अभिलक्षणिक मान $\lambda \ne 0$ वाले $G$ के अभिलक्षणिक
        फलन, अदिशों तक, $-\lambda u'' = u$, $u(0) = u(1) = 0$ के हल हैं \emph{(किसी
        अभिलक्षणिक फलन का कोई \omterm{def:b3:topology:continuity}{संतत} प्रतिनिधि होता है --- $f \in L^2$ के
        लिए $Gf$ \omterm{def:b3:topology:continuity}{संतत} है, क्यों? --- अतः प्रश्न 2 को बार-बार
        लगाकर वह $\mathcal C^2$ है)}।
  \item सीमांत मान समस्या हल कीजिए: $G$ के अभिलक्षणिक मान
        $\lambda_n = \frac1{n^2\pi^2}$ ($n \geq 1$) हैं, जिनके मानकीकृत अभिलक्षणिक फलन $e_n(x) =
        \sqrt2\,\sin(n\pi x)$
        हैं; जाँच के तौर पर सीधे समाकलन से प्रसामान्य लांबिकता
        सत्यापित कीजिए।
  \item \cref{thm:b3:spectral:spectral} और प्रश्न 3 से निष्कर्ष निकालिए कि $\bigl(\sqrt2\sin(n\pi x)\bigr)_{n\geq1}$ $L^2(\intcc01)$ का
        एक प्रसामान्य लांबिक \emph{आधार} है --- न स्टोन--वाइरश्ट्रास
        चाहिए, न फूरिये श्रेणी। $f(x) = x(1-x)$ का इस आधार में प्रसार कीजिए
        और उसके लिए \omterm{thm:b3:hilbert:parseval}{पार्सेवाल} लिखिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- अनुरेख सूत्र और $\zeta(2)$।}
\begin{enumerate}[resume]
  \item दोनों सर्वसमिकाएँ सिद्ध कीजिए
        \[
        \langle e_n, Ge_n\rangle = \lambda_n
        \quad\text{तथा}\quad
        \sum_{n\geq1}\lambda_n = \int_0^1 g(x,x)\,\dd x .
        \]
        दूसरी (\emph{अनुरेख सूत्र}) के लिए: स्थिर $x$ पर $g(x, \cdot)$
        का आधार $(e_n)$ में प्रसार कीजिए --- दर्शाइए कि गुणांक
        $\lambda_ne_n(x)$ हैं, अतः $L^2$ में $g(x, \cdot) = \sum_n\lambda_ne_n(x)\,e_n$। यहाँ ज्या स्पष्ट हैं:
        सीधे सत्यापित कीजिए कि $\sum_n\lambda_ne_n(x)e_n(y)$ वर्ग पर \emph{एकसमान} रूप से
        अभिसरित होता है ($\sum
        \frac2{n^2\pi^2}$ से तुलना कीजिए), अतः उसका योग \omterm{def:b3:topology:continuity}{संतत}
        है और, प्रत्येक $x$ के लिए $y$ में वही $L^2$-प्रसार
        रखने के कारण, वह सर्वत्र $g(x,y)$ के बराबर है। अब $y = x$ रखिए
        और पदशः समाकलन कीजिए।
  \item $\int_0^1g(x,x)\dd x = \int_0^1x(1-x)\dd x =
        \frac16$ परिकलित कीजिए, और निष्कर्ष निकालिए
        \[
        \sum_{n\geq1}\frac{1}{n^2\pi^2} = \frac16,
        \qquad\text{अर्थात्}\qquad
        \boxed{\ \zeta(2) = \frac{\pi^2}{6}\ } :
        \]
        अर्थात् किसी संकारक अनुरेख से ऑयलर का योग।
  \item $\zeta(2)$ को तीसरे प्रकार से पुनः निकालिए: ज्या आधार में अचर
        फलन $\mathbf 1$ पर \omterm{thm:b3:hilbert:parseval}{पार्सेवाल} लगाइए, $\sum_{n \text{ विषम}}\frac1{n^2}$ परिकलित कीजिए, और
        निष्कर्ष निकालिए। फिर क्रियाविधियों की तुलना कीजिए: प्रश्न
        7--8 का अनुरेख तर्क किस अर्थ में ``पूरे नाभिक पर एक साथ
        लगाया गया \omterm{thm:b3:hilbert:parseval}{पार्सेवाल}'' है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- डोरी कंपन करती है।}
\begin{enumerate}[resume]
  \item (चर पृथक्करण, संश्लेषित) $f \in
        L^2$ के लिए
        \[
        u(t, x) = \sum_{n\geq1}\;c_n\,
        \cos(n\pi t)\,\sqrt2\sin(n\pi x),
        \qquad c_n = \langle e_n, f\rangle .
        \]
        परिभाषित कीजिए। दर्शाइए कि श्रेणी प्रत्येक $t$ के लिए
        $L^2(\intcc01)$ में अभिसरित होती है, कि $t\mapsto u(t, \cdot)$ $L^2$ में \omterm{def:b3:topology:continuity}{संतत} है, और
        यह कि परिमित संख्या के $e_n$ के विस्तार में आने वाले
        $f$ के लिए वह तरंग समीकरण $\partial_t^2u =
        \partial_x^2u$ को $u(0) = f$, $\partial_tu(0) = 0$ और
        स्थिर सिरों के साथ हल करता है। अभिलक्षणिक मान $n^2\pi^2$
        वर्गित आवृत्तियाँ हैं: डोरी के संनाद --- एक अनुच्छेद में
        \cref{thm:b3:spectral:spectral} की सांगीतिक व्याख्या दीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- विचरण लाभांश: घात विधि, वेल स्थायित्व,
और $\pi$ पर एक कठोर परिबंध।} मान लीजिए $A$ ऐसा \omterm{def:b3:spectral:compact}{संहत} \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न}
\emph{धनात्मक} संकारक है जिसके अभिलक्षणिक मान $\mu_1 \geq \mu_2 \geq \cdots > 0$ और प्रसामान्य
लांबिक अभिलक्षणिक सदिश $(u_n)$ हैं; $R(x) = \frac{\langle x,
Ax\rangle}{\norm x^2}$। न्यूनतम--उच्चतम सूत्र
\cref{exo:b3:spectral:8} हैं; यहाँ हम उन्हें भुनाते हैं।
\begin{enumerate}[resume]
  \item (घात विधि) $x \neq 0$ के लिए $m_p =
        \sum_n\mu_n^p\abs{\langle u_n, x\rangle}^2$ लिखिए। कोशी--श्वार्ज़ से
        $m_pm_{p+2} \geq m_{p+1}^2$ दर्शाइए, शृंखला
        \[
        R(x) \;\leq\; \frac{\langle Ax, Ax\rangle}{\langle x,
        Ax\rangle} \;\leq\; R(Ax) \;\leq\; \mu_1,
        \]
        निकालिए, और सिद्ध कीजिए कि यदि $\langle u_1, x\rangle \neq 0$ हो, तो $R(A^kx) \to \mu_1$: अर्थात्
        किसी भी प्ररूपी सदिश पर संकारक को बार-बार लगाने से सबसे
        बड़ा अभिलक्षणिक मान परिकलित हो जाता है --- सांख्यिक
        विश्लेषण की घात विधि, प्रमाणित।
  \item (वेल स्थायित्व) \omterm{def:b3:spectral:compact}{संहत} \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} धनात्मक $A, B$ के लिए \cref{exo:b3:spectral:8}
        से निष्कर्ष निकालिए
        \[
        \abs{\mu_n(A) - \mu_n(B)} \;\leq\; \vertiii{A - B}
        \qquad\text{प्रत्येक } n \text{ के लिए} :
        \]
        अर्थात् पूरा स्पेक्ट्रम \omterm{def:b3:banach:operator}{संकारक मानक} में $1$-लिप्शिट्ज़ है
        --- बड़े सममित निकायों के अभिलक्षणिक मान आसन्ननों से,
        प्रत्याभूत त्रुटि के साथ, परिकलित किए जा सकते हैं।
  \item $G$ पर परीक्षण फलन $u(x) = x(1-x)$ के साथ रेली परिबंध लगाइए:
        $-w'' = u$, $w(0) =
        w(1) = 0$ हल करके $Gu = w = \frac{x(1-x)(1 + x -
        x^2)}{12}$ प्राप्त कीजिए,
        \[
        \norm u_2^2 = \frac1{30}, \qquad
        \langle u, Gu\rangle = \frac1{12}\Bigl(\frac1{30} +
        \frac1{140}\Bigr) = \frac{17}{5040},
        \qquad R(u) = \frac{17}{168},
        \]
        परिकलित कीजिए, और \emph{कठोर} परिबंध $\frac1{\pi^2}
        = \lambda_1 \geq \frac{17}{168}$ निकालिए,
        अर्थात् $\pi \leq
        \sqrt{168/17} < 3.1437$।
  \item उसी परीक्षण फलन पर प्रश्न 11 की शृंखला का एक चरण: $\int_0^1(x - x^2)^4\dd x =
        \frac1{630}$
        का उपयोग करके
        \[
        \norm{Gu}_2^2 = \frac1{144}\Bigl(\frac1{30} +
        \frac2{140} + \frac1{630}\Bigr) = \frac{31}{90720},
        \qquad
        \frac{\langle Gu, Gu\rangle}{\langle u, Gu\rangle} =
        \frac{31}{306},
        \]
        परिकलित कीजिए और $\pi \leq \sqrt{306/31} < 3.1419$ निष्कर्ष निकालिए: दो समाकल, चार
        सही अंक। (प्रत्येक अगली पुनरावृत्ति यथार्थता को लगभग वर्ग
        कर देती है: अभिलक्षणिक सदिश अंतराल $\lambda_1/\lambda_2 = 4$ ही गुणोत्तर
        अभिसरण चलाता है।)
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- $G^2$ का अनुरेख, और $\zeta(4)$।}
\begin{enumerate}[resume]
  \item दर्शाइए कि $\sum_n\lambda_n^2 =
        \iint_{\intcc01^2}g(x,y)^2\,\dd x\,\dd y$ \emph{($L^2(\intcc01^2)$ के गुणन आधार $(e_m(x)e_n(y))_{m,n}$ पर $g$
        का प्रसार कीजिए --- जो एक \omterm{def:b3:hilbert:onb}{हिल्बर्ट आधार} है, तुलना \cref{exo:b3:spectral:5}
        --- और वर्ग पर \omterm{thm:b3:hilbert:parseval}{पार्सेवाल} लगाइए; प्रश्न 7 गुणांक पहचान देता
        है)}।
  \item द्विक समाकल परिकलित कीजिए:
        \[
        \iint g^2 = 2\int_0^1(1-x)^2\Bigl(\int_0^x
        y^2\,\dd y\Bigr)\dd x
        = \frac23\int_0^1x^3(1-x)^2\,\dd x = \frac1{90} .
        \]
  \item $\zeta(4) = \dfrac{\pi^4}{90}$ निष्कर्ष निकालिए; बिना परिकलन के समझाइए कि उच्चतर
        घातों $G^k$ के अनुरेख सभी $k \geq 1$ के लिए $\zeta(2k) \in \pi^{2k}\,\Q$ कैसे दे देते
        हैं, और विषम मान $\zeta(3),
        \zeta(5), \dots$ संरचनात्मक रूप से इस मशीन की पहुँच
        से बाहर क्यों हैं।
  \item (नीचे से $\pi$) $\lambda_1^2 \leq
        \sum_n\lambda_n^2 = \frac1{90}$ से $\pi \geq
        90^{1/4} > 3.080$ निकालिए, और प्रश्न 14 के
        साथ मिलाकर द्विपक्षीय निर्णय
        \[
        3.080 \;<\; \pi \;<\; 3.1419,
        \]
        संकलित कीजिए, जो पूरी तरह कंपमान डोरी के अंकगणित से मिला
        है। यदि उच्चतर अनुरेख $(\operatorname{tr}G^{2k})^{-1/4k}$ का उपयोग किया जाए, तो कौन-सा
        पक्ष तेजी से अभिसरित होता है, और क्यों?
\end{enumerate}

\medskip
\noindent\textbf{भाग VII --- प्रणोदन और अनुनाद।} $\nu \in \R$ स्थिर
कीजिए और प्रणोदित डोरी $-u'' - \nu u =
f$, $u(0) = u(1) = 0$ पर विचार कीजिए, जिसमें $f \in L^2$
और $c_n = \langle
e_n, f\rangle$।
\begin{enumerate}[resume]
  \item मान लीजिए $\nu \notin \{n^2\pi^2 : n \geq 1\}$। दर्शाइए कि
        \[
        u = \sum_{n\geq1}\frac{c_n}{n^2\pi^2 - \nu}\,e_n
        \]
        $L^2$ में और $\intcc01$ पर एकसमान रूप से अभिसरित होता है
        \emph{($(c_n)$ और $\sum n^{-4}$ की पुच्छों के बीच कोशी--श्वार्ज़,
        $\norm{e_n}_\infty =
        \sqrt2$ के साथ)}, और यह कि वह $u = Gf + \nu Gu$ को संतुष्ट करता है ---
        अर्थात् \omterm{thm:b3:spectral:fredholm}{फ्रेडहोम विकल्प} (\cref{thm:b3:spectral:fredholm}) का स्पष्ट-निर्देशांक रूप,
        अद्वितीयता सहित।
  \item मान लीजिए $\nu = m^2\pi^2$। दर्शाइए कि $u = Gf + \nu
        Gu$ का हल है तभी और केवल
        तभी जब $c_m = 0$, और वह $e_m$ के गुणज जोड़ने तक अद्वितीय है।
        भौतिक पाठ: किसी झूले को ठीक उसकी अपनी आवृत्ति पर धकेलना।
  \item $\nu < \pi^2$ के लिए दर्शाइए कि हल संकारक $R_\nu\colon f \mapsto u$ $L^2$ पर मानक
        $\frac1{\pi^2 - \nu}$ के साथ परिबद्ध, \omterm{def:b3:spectral:compact}{संहत}, \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} और धनात्मक है: अतः
        पूरा स्पेक्ट्रमी विश्लेषण फिर से आरंभ हो जाता है, केवल
        $\nu$ से खिसका हुआ।
  \item (संश्लेषण) इस समस्या का शब्दकोश संकलित कीजिए: अभिलक्षणिक
        मान $\leftrightarrow$ वर्गित आवृत्ति (संनाद); अनुरेख $\leftrightarrow$ $\zeta(2)$;
        हिल्बर्ट--श्मिट मानक $\leftrightarrow$ $\zeta(4)$; \omterm{thm:b3:spectral:fredholm}{फ्रेडहोम विकल्प} $\leftrightarrow$
        अनुनाद; न्यूनतम--उच्चतम $\leftrightarrow$ विचरण परिबंध (एक ही बहुपद से
        $\pi <
        3.1437$)। एक समाकल संकारक, और विश्लेषण के पाँच अध्याय भुना
        लिए गए।
\end{enumerate}

\medskip
\noindent\textbf{भाग VIII --- अंतिम तीन प्रतिध्वनियाँ।}
\begin{enumerate}[resume]
  \item ($\zeta(6)$, मुफ्त में) प्रश्न 6 की \omterm{thm:b3:hilbert:parseval}{पार्सेवाल} सर्वसमिका ने
        $\sum_{n\text{ विषम}}n^{-6} =
        \frac{\pi^6}{960}$ दिया था। $\zeta(6)$ को विषम और सम $n$ में बाँटिए और
        बिना किसी नए समाकल के
        \[
        \zeta(6) = \frac{\pi^6}{945},
        \]
        निष्कर्ष निकालिए: प्रश्न 17 की मशीन ($G^3$ के अनुरेख) यही
        मान देती, पर एक पुनरावृत्त नाभिक की कीमत पर --- यहाँ किसी
        एक सुचुने हुए फलन पर \omterm{thm:b3:hilbert:parseval}{पार्सेवाल} सस्ता मार्ग है।
  \item (मूल अवस्था धनात्मक होती है) मान लीजिए $A$ $L^2(\intcc01)$ पर
        $\intoo01^2$ पर \omterm{def:b3:topology:continuity}{संतत} सममित नाभिक $k > 0$ से दिया गया \omterm{def:b3:spectral:compact}{संहत} \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न}
        धनात्मक संकारक है, जिसका सबसे बड़ा अभिलक्षणिक मान $\mu_1$ है।
        दर्शाइए: (क) रेली भागफल का कोई भी अधिकतमकारी $\mu_1$-अभिलक्षणिक
        फलन होता है; (ख) यदि $u$ ऐसा एक हो, तो $\langle\abs u, A\abs u\rangle \geq \langle u,
        Au\rangle$, और यदि
        $u$ धनात्मक \omterm{def:b3:measure:measure}{माप} वाले समुच्चयों पर दोनों चिह्न लेता हो तो
        \emph{कड़ी} असमिका --- अतः $u$ का चिह्न लगभग सर्वत्र अचर
        होता है, और $u = \mu_1^{-1}Au$ $\intoo01$ पर कभी लुप्त नहीं होता; (ग) $\mu_1$
        एक \emph{सरल} अभिलक्षणिक मान है। $G$ पर प्रत्येक दावा
        सत्यापित कीजिए: $e_1 = \sqrt2\sin(\pi x) > 0$, और प्रत्येक $e_n$, $n
        \geq 2$, $e_1$ के
        लांबिक होने के कारण चिह्न बदलना ही चाहिए (और वह बदलता है:
        $n - 1$ आंतरिक शून्य)।
  \item (स्पेक्ट्रम से दूरी, और अनुनाद की कीमत) $\nu \notin \{n^2\pi^2\}$ के लिए
        दर्शाइए कि प्रश्न 19 का हल संकारक $R_\nu$ परिबद्ध,
        \omterm{def:b3:spectral:selfadjoint}{स्वसंलग्न} और \omterm{def:b3:spectral:compact}{संहत} है, जिसमें
        \[
        \vertiii{R_\nu} =
        \frac1{\min_{n\geq1}\,\abs{n^2\pi^2 - \nu}}
        = \frac1{\operatorname{dist}\bigl(\nu,
        \{n^2\pi^2\}\bigr)},
        \]
        और यह मानक निकटतम विधा पर प्राप्त होता है। फिर प्रश्न 20
        के झूले को मात्रात्मक रूप दीजिए: $\nu = (1 - \varepsilon)\pi^2$ पर $f = e_1$ से प्रणोदन
        $u =
        \frac{1}{\varepsilon\pi^2}\,e_1$ उत्पन्न करता है, अर्थात् स्थैतिक अनुक्रिया $Ge_1 =
        \frac1{\pi^2}e_1$ का
        $\frac1\varepsilon$ गुना प्रवर्धन --- और मूल आवृत्ति से एक प्रतिशत नीचे
        ($\varepsilon = 10^{-2}$) डोरी सौ गुना ऊँचा उत्तर देती है।
\end{enumerate}
\end{problem}
