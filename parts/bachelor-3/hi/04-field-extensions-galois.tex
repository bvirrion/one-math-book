\chapter{क्षेत्र विस्तार और गाल्वा सिद्धांत}\label{ch:b3:galois}

क्या प्रत्येक समीकरण मूलकों द्वारा हल किया जा सकता है, जैसा
द्विघात सूत्र और कार्दानो के त्रिघात सूत्र सुझाते हैं? क्या
पैमाने और परकार से किसी कोण का त्रिविभाजन संभव है? सदियों से खुले
पड़े ये दोनों प्रश्न एवारिस्त \omterm{def:b3:galois:galois}{गाल्वा} के एक ही विचार से हल हो
जाते हैं --- और उत्तर नकारात्मक है: प्रत्येक बहुपद के साथ उसके
मूलों की सममितियों का एक परिमित \emph{समूह} जोड़िए, और उत्तर उसी
समूह में पढ़ लीजिए। यह अध्याय वह शब्दकोश बनाता है: \omterm{def:b3:galois:extension}{क्षेत्र
विस्तार} और घातें, \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} और \omterm{def:b3:galois:algebraic}{बीजीय} संवृतियाँ, \omterm{thm:b3:galois:finitefields}{परिमित
क्षेत्र} (एक पूर्ण सिद्धांत --- और $\mathbb F_q^\times$ की वह चक्रीयता जिसका वचन
दिया गया था), वियोज्यता, और फिर पूरी उपपत्तियों सहित स्वयं \omterm{thm:b3:galois:fundamental}{गाल्वा
संगति}। फसल यह है: शास्त्रीय रचनाओं की असंभवता, वृत्तखंडी क्षेत्रों
की संरचना, और पंचघात समीकरण की मूलकों द्वारा अहल्यता --- \cref{ch:b3:groups} की
$A_5$ की सरलता अपने लक्ष्य पर जा लगती है।

\section{विस्तार, घात, बीजीयता}

\begin{definition}\label{def:b3:galois:extension}
कोई \emph{क्षेत्र विस्तार}\index{क्षेत्र विस्तार} $L/K$ एक ऐसा
क्षेत्र $L$ है जो $K$ को उपक्षेत्र के रूप में अंतर्विष्ट करता
है; तब $L$ एक $K$-सदिश समष्टि है, और
\emph{घात}\index{विस्तार की घात} $[L:K]$ उसकी विमा है। विस्तार
\emph{परिमित} कहलाता है यदि $[L:K] <
\infty$ हो। किसी क्षेत्र का
\emph{अभिलक्षण} $\Z \to K$, $n \mapsto n\cdot 1$, की अष्टि का जनक $\geq 0$ है: वह $0$
होता है या कोई अभाज्य $p$; तदनुसार $K$ में $\Q$ या $\mathbb
F_p = \Z/p\Z$ के
तुल्याकारी एक सबसे छोटा उपक्षेत्र (\emph{अभाज्य क्षेत्र}) होता
है।
\end{definition}

\begin{theorem}[मीनार नियम]\label{thm:b3:galois:tower}
यदि $K \subseteq L \subseteq M$ हो, तो $[M:K] = [M:L]\,[L:K]$: यदि $(e_i)$ $K$ पर $L$ का आधार हो और $(f_j)$
$L$ पर $M$ का आधार, तो $(e_if_j)$ $K$ पर $M$ का आधार
है।\index{मीनार नियम}
\end{theorem}

\begin{proof}
जनकता: $x \in M$ $x = \sum_j \lambda_jf_j$ लिखा जाता है ($\lambda_j \in L$), और प्रत्येक $\lambda_j = \sum_i \mu_{ij}e_i$ ($\mu_{ij} \in K$):
अतः $x = \sum_{i,j}\mu_{ij}e_if_j$। स्वतंत्रता: $\sum_{i,j}\mu_{ij}e_if_j = 0$ को $\sum_j (\sum_i
\mu_{ij}e_i)f_j = 0$ के रूप में लिखिए; भीतरी योग
$L$ में हैं, अतः वे लुप्त हो जाते हैं ($f_j$ $L$ पर स्वतंत्र);
तब सभी $\mu_{ij} = 0$ ($e_i$ $K$ पर स्वतंत्र)।
\end{proof}

\begin{definition}\label{def:b3:galois:algebraic}
मान लीजिए $L/K$ और $\alpha \in L$। यदि किसी शून्येतर $P \in K[X]$ के लिए $P(\alpha) = 0$ हो,
तो $\alpha$ $K$ पर \emph{बीजीय}\index{बीजीय अवयव} कहलाता है; $K[X]$
के \omterm{def:b3:rings:ideal}{आदर्श} $\{P : P(\alpha) = 0\}$ का मोनिक जनक $\pi_\alpha$ उसका \emph{न्यूनतम
बहुपद}\index{न्यूनतम बहुपद (क्षेत्र सिद्धांत)} कहलाता है, जो एक
\omterm{def:b3:rings:divisibility}{अखंडनीय} बहुपद है ($Q(\alpha) = 0$ के साथ $\pi = QR$ होने पर \omterm{def:b3:galois:extension}{घात} की न्यूनतमता से
$R$ का अचर होना अनिवार्य हो जाता है)। अन्यथा $\alpha$
\emph{अबीजीय} कहलाता है। $K$ और $\alpha$ को अंतर्विष्ट करने वाले
$L$ के सबसे छोटे उपक्षेत्र के लिए हम $K(\alpha)$ लिखते हैं, और सबसे
छोटे उपवलय के लिए $K[\alpha]$।
\end{definition}

\begin{theorem}\label{thm:b3:galois:simple}
यदि $\alpha$ $K$ पर \omterm{def:b3:galois:algebraic}{बीजीय} हो और $d = \deg\pi_\alpha$ हो, तो
\[
K(\alpha) = K[\alpha] \cong K[X]/(\pi_\alpha),
\qquad [K(\alpha) : K] = d,
\]
जिसका आधार $1, \alpha, \dots, \alpha^{d-1}$ है। विलोमतः, यदि $[L:K] < \infty$ हो, तो प्रत्येक $\alpha \in L$
\omterm{def:b3:galois:algebraic}{बीजीय} है और उसकी \omterm{def:b3:galois:extension}{घात} $\deg \pi_\alpha$ $[L:K]$ को विभाजित करती है।
\end{theorem}

\begin{proof}
मान-निर्धारण $K[X] \to L$, $P \mapsto P(\alpha)$, का प्रतिबिंब $K[\alpha]$ और अष्टि $(\pi_\alpha)$ है;
चूँकि $\pi_\alpha$ \omterm{def:b3:rings:divisibility}{अखंडनीय} है, $K[X]/(\pi_\alpha)$ एक \emph{क्षेत्र} है (\cref{prop:b3:rings:primemaximal}: \omterm{def:b3:rings:pidufd}{PID}
$K[X]$ में \omterm{def:b3:rings:divisibility}{अखंडनीय अवयव} \omterm{def:b3:rings:primemaximal}{महत्तम आदर्श} जनित करता है), अतः $K[\alpha]$ $K$
और $\alpha$ को अंतर्विष्ट करने वाला क्षेत्र है: वह $K(\alpha)$ के बराबर है।
यूक्लिडीय विभाजन से $1, X, \dots, X^{d-1}$ के वर्ग भागफल का आधार बनाते हैं, जिससे
आधार और \omterm{def:b3:galois:extension}{घात} दोनों मिल जाते हैं। विलोमतः यदि $[L:K]
= n < \infty$ हो: $1, \alpha, \dots, \alpha^n$ आश्रित
हैं, जिससे एक विनाशक बहुपद मिलता है; और तब \omterm{thm:b3:galois:tower}{मीनार नियम} से $[K(\alpha):K] =
\deg\pi_\alpha$
$n$ को विभाजित करता है।
\end{proof}

\begin{corollary}\label{cor:b3:galois:algclosed}
यदि $\alpha, \beta$ $K$ पर \omterm{def:b3:galois:algebraic}{बीजीय} हों, तो $\alpha \pm
\beta$, $\alpha\beta$, $\alpha/\beta$ ($\beta \ne 0$) भी
\omterm{def:b3:galois:algebraic}{बीजीय} हैं: अर्थात् $L$ के वे अवयव जो $K$ पर \omterm{def:b3:galois:algebraic}{बीजीय} हैं $L$
का एक उपक्षेत्र बनाते हैं। इसके अतिरिक्त बीजीयता संक्रमणीय है:
\omterm{def:b3:galois:algebraic}{बीजीय} पर \omterm{def:b3:galois:algebraic}{बीजीय} \omterm{def:b3:galois:algebraic}{बीजीय} ही होता है।
\end{corollary}

\begin{proof}
$K(\alpha, \beta) = (K(\alpha))(\beta)$ $K(\alpha)$ पर परिमित है ($\beta$ $K \subseteq K(\alpha)$ पर \omterm{def:b3:galois:algebraic}{बीजीय}) और $K(\alpha)/K$ परिमित
है: \omterm{thm:b3:galois:tower}{मीनार नियम} से $[K(\alpha,\beta):K] <
\infty$, और $K(\alpha, \beta)$ का प्रत्येक अवयव --- जिनमें ये
चारों सम्मिलित हैं --- \omterm{def:b3:galois:algebraic}{बीजीय} है (\cref{thm:b3:galois:simple})। संक्रमणीयता: यदि $\beta$
$L$ पर \omterm{def:b3:galois:algebraic}{बीजीय} हो और $L/K$ \omterm{def:b3:galois:algebraic}{बीजीय} हो, तो $\pi_{\beta/L}$ के गुणांक $c_0, \dots, c_{m-1}$
$K$ का कोई परिमित विस्तार $F = K(c_0, \dots,
c_{m-1})$ जनित करते हैं (\omterm{thm:b3:galois:tower}{मीनार नियम} को
बार-बार लगाइए), और $F(\beta)/F$ परिमित है: अतः $[F(\beta):K] < \infty$, इसलिए $\beta$
$K$ पर \omterm{def:b3:galois:algebraic}{बीजीय} है।
\end{proof}

\begin{example}\label{ex:b3:galois:degrees}
$[\Q(\sqrt2):\Q] = 2$, $[\Q(\sqrt[3]2):\Q] = 3$ ($X^3 - 2$ \omterm{def:b3:rings:divisibility}{अखंडनीय} है: \omterm{thm:b3:rings:criteria}{आइज़ेनश्टाइन}), और $\zeta_p = \eu^{2\iu\pi/p}$ के लिए
$[\Q(\zeta_p):\Q] = p - 1$ ($\Phi_p$ \omterm{def:b3:rings:divisibility}{अखंडनीय} है, \cref{ex:b3:rings:eisenstein})। \omterm{thm:b3:galois:tower}{मीनार नियम} पहले से ही एक
हथियार है: $\sqrt[3]2 \notin \Q(\sqrt2)$, क्योंकि $3 \nmid 2$।
\end{example}

\section{विभाजन क्षेत्र; बीजीय संवृति}

\begin{theorem}[विभाजन क्षेत्र]\label{thm:b3:galois:splitting}
मान लीजिए $P \in K[X]$ अनचर है। $K$ पर $P$ का एक \emph{विभाजन
क्षेत्र}\index{विभाजन क्षेत्र} विद्यमान है: अर्थात् $P$ के
मूलों से जनित ऐसा विस्तार $L =
K(\alpha_1, \dots, \alpha_n)$ जिसमें $P$ रैखिक गुणनखंडों में
विभाजित हो जाता है। वह $K$-तुल्याकारिता तक अद्वितीय है, और
$[L:K] \leq (\deg P)!$।
\end{theorem}

\begin{proof}
\emph{अस्तित्व}, $\deg P$ पर आगमन द्वारा: $P$ का कोई \omterm{def:b3:rings:divisibility}{अखंडनीय}
गुणनखंड $Q$ चुनिए; क्षेत्र $K_1 = K[X]/(Q)$ में $Q$ का, अतः $P$ का
भी, मूल $\alpha_1 = \bar X$ होता है; $K_1$ पर $P = (X -
\alpha_1)P_1$ लिखिए और $K_1$ पर $P_1$
पर आगमन लगाइए; घातें गुणा होकर अधिकतम $n(n-1)\cdots = n!$ तक पहुँचती हैं।

\emph{अद्वितीयता} अधिक प्रबल \emph{तुल्याकारिता विस्तार
प्रमेयिका} से निकलती है: मान लीजिए $\sigma \colon K \to K'$ एक तुल्याकारिता है,
$P \in K[X]$, $P^\sigma$ प्रतिचित्रित गुणांकों वाला बहुपद, और $L, L'$ $P, P^\sigma$
के \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र}; तब $\sigma$ किसी तुल्याकारिता $L \to L'$ तक विस्तारित
हो जाता है। $[L:K]$ पर आगमन: यदि $P$ $K$ में विभाजित हो जाए,
तो $L = K$, और $L' = K'$ ($P^\sigma$ $K'$ में विभाजित होता है, और $L'$
उसके मूलों से जनित है)। अन्यथा $\deg Q \geq 2$ वाले $P$ के किसी \omterm{def:b3:rings:divisibility}{अखंडनीय}
गुणनखंड $Q$ का मूल $\alpha \in L \setminus K$ चुनिए; $Q^\sigma$ $P^\sigma$ का \omterm{def:b3:rings:divisibility}{अखंडनीय}
गुणनखंड है, जिसका कोई मूल $\beta \in L'$ है; तब
\[
K(\alpha) \cong K[X]/(Q) \xrightarrow{\ \sigma\ }
K'[X]/(Q^\sigma) \cong K'(\beta)
\]
$\sigma$ को $\alpha \mapsto \beta$ के साथ विस्तारित करता है। अब $L$ $K(\alpha)$ पर
$P$ का \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} है, और $L'$ $K'(\beta)$ पर $P^\sigma$ का, जिनमें
$[L : K(\alpha)] < [L:K]$: आगमन इसे आगे $L \to L'$ तक विस्तारित कर देता है।
\end{proof}

\begin{definition}\label{def:b3:galois:closure}
कोई क्षेत्र $\Omega$ \emph{बीजीय रूप से संवृत}\index{बीजीय रूप से
संवृत क्षेत्र} कहलाता है यदि $\Omega[X]$ के प्रत्येक अनचर बहुपद का
$\Omega$ में कोई मूल हो (और इसलिए वह विभाजित हो जाए)। $K$ की कोई
\emph{बीजीय संवृति}\index{बीजीय संवृति} ऐसा \omterm{def:b3:galois:algebraic}{बीजीय} विस्तार $\bar K/K$
है जिसमें $\bar K$ \omterm{def:b3:galois:algebraic}{बीजीय} रूप से संवृत हो।
\end{definition}

\begin{theorem}[श्टाइनित्स]\label{thm:b3:galois:steinitz}
प्रत्येक क्षेत्र $K$ की एक \omterm{def:b3:galois:closure}{बीजीय संवृति} होती है, जो
$K$-तुल्याकारिता तक अद्वितीय है।
\end{theorem}

\begin{proof}
\emph{अस्तित्व (आर्टिन की रचना)।} मान लीजिए $R = K[(X_f)_f]$ वह बहुपद वलय
है जिसमें प्रत्येक अनचर मोनिक $f \in K[X]$ के लिए एक चर $X_f$ है, और
$I$ समस्त $f(X_f)$ से जनित \omterm{def:b3:rings:ideal}{आदर्श}। $I$ उचित है: कोई भी संबंध
$1 = \sum_{i=1}^r g_i\,
f_i(X_{f_i})$ परिमित संख्या में बहुपदों को समेटता है; $f_1\cdots f_r$ के किसी
उभयनिष्ठ \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} $E$ में $f_i$ के मूल $\alpha_i$ चुनकर
$X_{f_i} \mapsto \alpha_i$ का मान लीजिए (शेष चर $\mapsto 0$): तब $1 = 0$, जो निरर्थक है। अब
$\mathfrak m \supseteq I$ को महत्तम लीजिए (\cref{thm:b3:rings:krull}; ज़ोर्न) और $K_1 = R/\mathfrak
m$ रखिए: यह $K$
का ऐसा \omterm{def:b3:galois:extension}{क्षेत्र विस्तार} है जिसमें प्रत्येक अनचर $f \in
K[X]$ का कोई मूल
है, अर्थात् $\bar X_f$, और जो $K$ पर \omterm{def:b3:galois:algebraic}{बीजीय} है (वह $\bar X_f$ से जनित है,
और उनमें से प्रत्येक \omterm{def:b3:galois:algebraic}{बीजीय} है)। इसे दोहराइए: $K \subseteq K_1 \subseteq K_2 \subseteq \cdots$, जहाँ $K_{n+1}$
$K_n$ के साथ वही करता है जो $K_1$ ने $K$ के साथ किया था, और
$\Omega =
\bigcup_n K_n$ रखिए, जो एक क्षेत्र है। किसी भी अनचर $g \in \Omega[X]$ के परिमित
गुणांक किसी न किसी $K_n$ में होते हैं; $K_n$ पर $g$ के किसी
\omterm{def:b3:rings:divisibility}{अखंडनीय} गुणनखंड का $K_{n+1} \subseteq
\Omega$ में मूल होता है: अतः $\Omega$ \omterm{def:b3:galois:closure}{बीजीय रूप से
संवृत} है, और $K$ पर \omterm{def:b3:galois:algebraic}{बीजीय} (संक्रमणीयता से प्रत्येक $K_n$ \omterm{def:b3:galois:algebraic}{बीजीय}
है, \cref{cor:b3:galois:algclosed}): इस प्रकार $\Omega$ एक \omterm{def:b3:galois:closure}{बीजीय संवृति} है।

\emph{अद्वितीयता।} मान लीजिए $\Omega, \Omega'$ दो \omterm{def:b3:galois:algebraic}{बीजीय} संवृतियाँ हैं।
युग्मों $(E, \tau)$ के उस समुच्चय पर विचार कीजिए जिनमें $K
\subseteq E \subseteq \Omega$ है और
$\tau\colon E \to \Omega'$ एक $K$-अंतःस्थापन है, और जिसे विस्तार से क्रमित किया गया
है; वह अरिक्त है ($(K,
\mathrm{id})$) और आगमनात्मक (किसी शृंखला का सम्मिलन),
अतः ज़ोर्न से कोई महत्तम $(E_0, \tau_0)$ मिलता है। यदि $E_0 \neq \Omega$ हो, तो $\alpha \in
\Omega\setminus E_0$
चुनिए: $\pi_{\alpha/E_0}$ $\tau_0(E_0)$ पर ऐसे बहुपद पर जाता है जिसका \omterm{def:b3:galois:closure}{बीजीय रूप से
संवृत} $\Omega'$ में कोई मूल $\beta$ है, और $\tau_0$ $E_0(\alpha) \to
\Omega'$ तक विस्तारित हो
जाता है ($\alpha \mapsto \beta$), जो महत्तमता का खंडन है। अतः कोई
$K$-अंतःस्थापन $\tau \colon \Omega \to \Omega'$ विद्यमान है; उसका प्रतिबिंब, जो $\Omega$ के
तुल्याकारी है, \omterm{def:b3:galois:closure}{बीजीय रूप से संवृत} है, और $\Omega'$ उस पर \omterm{def:b3:galois:algebraic}{बीजीय} है:
$x \in \Omega'$ के लिए $\pi_{x/\tau(\Omega)}$ $\tau(\Omega)$ पर विभाजित हो जाता है, अतः $x \in
\tau(\Omega)$। इस
प्रकार $\tau$ आच्छादक है: अर्थात् एक तुल्याकारिता।
\end{proof}

\begin{remark}\label{rem:b3:galois:closureexamples}
$K = \Q$ के लिए पारलौकिक यंत्रावली से बचा जा सकता है: \omterm{def:b3:galois:algebraic}{बीजीय} संख्याएँ
$\bar\Q = \{z \in \C : z \text{, } \Q \text{ पर
बीजीय}\}$ एक \omterm{def:b3:galois:closure}{बीजीय संवृति} बनाती हैं --- \cref{cor:b3:galois:algclosed} से यह $\C$ का
उपक्षेत्र है, और यह \omterm{def:b3:galois:closure}{बीजीय रूप से संवृत} है क्योंकि $\C$ संवृत है
(दालांबेर--गाउस, जिसे \cref{ch:b3:holomorphic} में सम्मिश्र विश्लेषण द्वारा सिद्ध
किया गया है) और $\bar\Q$ पर बहुपदों के मूल संक्रमणीयता से $\Q$ पर
\omterm{def:b3:galois:algebraic}{बीजीय} होते हैं।
\end{remark}

\section{परिमित क्षेत्र}

\begin{theorem}\label{thm:b3:galois:finitefields}
मान लीजिए $p$ अभाज्य है, $n \geq 1$, $q = p^n$।
\begin{enumerate}
  \item किसी \omterm{thm:b3:galois:finitefields}{परिमित क्षेत्र} की गणनीयता कोई अभाज्य \omterm{def:b3:galois:extension}{घात} होती है,
        और प्रत्येक $q$ के लिए तुल्याकारिता तक ठीक एक क्षेत्र
        $\mathbb F_q$ है जिसमें $q$ अवयव हैं: अर्थात् $\mathbb F_p$ पर $X^q - X$
        का \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र}।\index{परिमित क्षेत्र}
  \item \emph{फ्रोबेनियस}\index{फ्रोबेनियस अंतःसमाकारिता} $F
        \colon x \mapsto x^p$
        $\mathbb
        F_q$ की एक स्वसमाकारिता है, और $\mathbb F_q$ का स्वसमाकारिता
        समूह $F$ से जनित, क्रम $n$ का चक्रीय समूह है।
  \item $\mathbb F_{p^m}$ $\mathbb F_{p^n}$ में अंतःस्थापित होता है तभी और केवल तभी जब
        $m \mid
        n$।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) किसी \omterm{thm:b3:galois:finitefields}{परिमित क्षेत्र} $E$ का अभिलक्षण $p > 0$ होता है और वह
परिमित विमा की $\mathbb F_p$-सदिश समष्टि है: अतः $\abs E = p^n$। उसके गुणनात्मक
समूह का क्रम $q - 1$ है, अतः प्रत्येक $x \in E$ $x^q = x$ को संतुष्ट करता
है: इस प्रकार $E$ $X^q - X$ के $q$ मूलों से बना है, और इसलिए
$\mathbb F_p$ पर उसका \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} है --- जिससे $E$ तुल्याकारिता तक
निर्धारित हो जाता है (\cref{thm:b3:galois:splitting})। विलोमतः, $X^q - X$ के किसी \omterm{thm:b3:galois:splitting}{विभाजन
क्षेत्र} $L$ में उसके मूलों का समुच्चय $E$ एक
\emph{उपक्षेत्र} है: नौसिखिए के स्वप्न $(a+b)^p = a^p + b^p$ ($p \mid \binom pk$) को दोहराने
से $(x + y)^q = x^q + y^q$, तथा $(xy)^q = x^qy^q$, $(x^{-1})^q = (x^q)^{-1}$; और उसमें ठीक $q$ अवयव हैं क्योंकि
$X^q - X$ \omterm{def:b3:galois:separable}{वियोज्य} है: उसका अवकलज $qX^{q-1} - 1 = -1$ है (क्योंकि $p \mid q$), जो उसके
साथ सह-अभाज्य है, अतः कोई पुनरावृत्त मूल नहीं। इस प्रकार $L = E$
में $q$ अवयव हैं।

(2) $F$ एक क्षेत्र समाकारिता है (नौसिखिए का स्वप्न), और एकैकी
है (क्षेत्र), अतः परिमित $\mathbb F_q$ पर एकैकी आच्छादक। $F^n =
\mathrm{id}$ ($x^q = x$),
और उससे छोटी कोई \omterm{def:b3:galois:extension}{घात} तत्समक नहीं है: $F^m = \mathrm{id}$ का अर्थ होगा कि
$q$ अवयव $X^{p^m} - X$ के मूल हैं, जिससे $p^m \geq q$ अनिवार्य हो जाता है।
अतः $\langle F\rangle$ क्रम $n$ का चक्रीय समूह है; और अन्य कोई स्वसमाकारिता
नहीं है, क्योंकि नीचे सिद्ध परिबंध $\abs{\operatorname{Aut}} \leq [\,\mathbb F_q :
\mathbb F_p\,] = n$ लागू होता है (\cref{prop:b3:galois:embeddings},
जहाँ $L = \mathbb F_q$, $K =
\mathbb F_p$: स्वसमाकारिताएँ अभाज्य क्षेत्र को स्थिर रखती
हैं)।

(3) यदि $\mathbb F_{p^m} \subseteq \mathbb F_{p^n}$ हो, तो \omterm{thm:b3:galois:tower}{मीनार नियम} से $p^n = (p^m)^d$: अतः $m \mid n$। विलोमतः यदि
$m \mid n$ हो, तो $p^m - 1 \mid p^n - 1$ (गुणोत्तर योग), अतः $X^{p^m} - X$ $X^{p^n} - X$ को विभाजित
करता है (घातांकों पर वही तर्क: $a \mid b$ होने पर $X^{a} - 1 \mid X^{b}
- 1$), और $\mathbb F_{p^n}$ के
भीतर पहले वाले के मूल अभीष्ट उपक्षेत्र बनाते हैं, जिसकी गणनीयता
$p^m$ है ((1) की तरह वियोज्यता से)।
\end{proof}

\begin{theorem}[चक्रीयता]\label{thm:b3:galois:cyclic}
किसी क्षेत्र के गुणनात्मक समूह का प्रत्येक परिमित उपसमूह चक्रीय
होता है। विशेष रूप से $\mathbb F_q^\times \cong
\Z/(q-1)\Z$।\index{परिमित क्षेत्र का गुणनात्मक
समूह}
\end{theorem}

\begin{proof}
मान लीजिए $G \leq K^\times$ परिमित है। संरचना प्रमेय (\cref{cor:b3:modules:abelian}) से $d_1 \mid \dots \mid d_s$ के
साथ $G \cong \Z/d_1 \times \dots
\times \Z/d_s$। तब प्रत्येक $x \in G$ $x^{d_s} = 1$ को संतुष्ट करता है; परंतु
$X^{d_s} - 1$ के क्षेत्र $K$ में अधिकतम $d_s$ मूल होते हैं: अतः $\abs G = d_1\cdots d_s \leq d_s$,
जिससे $s = 1$ अनिवार्य हो जाता है: $G$ चक्रीय है।
\end{proof}

\begin{example}\label{ex:b3:galois:f8}
$\mathbb F_8 = \mathbb F_2[X]/(X^3 + X + 1)$: यह त्रिघात बहुपद $\mathbb F_2$ में कोई मूल नहीं रखता, अतः \omterm{def:b3:rings:divisibility}{अखंडनीय}
है। $\omega =
\bar X$ लिखने पर: $\mathbb F_8^\times$ क्रम $7$ का चक्रीय समूह है, अतः
\emph{प्रत्येक} अवयव $\neq 0, 1$ उसे जनित करता है। $\mathbb F_{p^{12}}$ के उपक्षेत्र
$12$ का भाजक जालक बनाते हैं: $\mathbb F_p, \mathbb F_{p^2}, \mathbb F_{p^3}, \mathbb F_{p^4},
\mathbb F_{p^6}, \mathbb F_{p^{12}}$ --- \omterm{thm:b3:galois:fundamental}{गाल्वा संगति} का पहला,
पूर्ण दृष्टांत।
\end{example}

\section{वियोज्यता और अंतःस्थापन}

\begin{definition}\label{def:b3:galois:separable}
कोई बहुपद $P \in K[X]$ \emph{वियोज्य}\index{वियोज्य बहुपद} कहलाता है
यदि उसके \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} में उसका कोई पुनरावृत्त मूल न हो ---
समतुल्य रूप से यदि $\gcd(P, P') = 1$ (पुनरावृत्त मूल उभयनिष्ठ मूल होता है;
विलोमतः, \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} पर उभयनिष्ठ मूल पुनरावृत्त होता है; और
महत्तम समापवर्तक \omterm{def:b3:galois:extension}{क्षेत्र विस्तार} से नहीं बदलता, \cref{cor:b3:modules:descent} का तर्क)।
कोई \omterm{def:b3:galois:algebraic}{बीजीय अवयव} वियोज्य कहलाता है यदि उसका \omterm{def:b3:galois:algebraic}{न्यूनतम बहुपद} वियोज्य
हो; और कोई विस्तार $L/K$ वियोज्य कहलाता है यदि उसके सभी अवयव
वियोज्य हों।
\end{definition}

\begin{proposition}\label{prop:b3:galois:perfect}
कोई \emph{\omterm{def:b3:rings:divisibility}{अखंडनीय}} $P \in K[X]$ \omterm{def:b3:galois:separable}{वियोज्य} होता है, सिवाय तब जब $P' = 0$ हो,
जिससे $\operatorname{char} K = p > 0$ और $P \in K[X^p]$ अनिवार्य हो जाते हैं। फलस्वरूप अभिलक्षण
$0$ वाले क्षेत्र का, और किसी \omterm{thm:b3:galois:finitefields}{परिमित क्षेत्र} का, प्रत्येक \omterm{def:b3:galois:algebraic}{बीजीय}
विस्तार \omterm{def:b3:galois:separable}{वियोज्य} होता है (ऐसे क्षेत्र \emph{पूर्ण}\index{पूर्ण
क्षेत्र} कहलाते हैं)।
\end{proposition}

\begin{proof}
$\gcd(P, P')$ $P$ को विभाजित करता है; यदि वह $1$ न हो, तो अखंडनीयता
से (किसी अचर तक) $\gcd = P$ अनिवार्य हो जाता है, अतः $\deg
P' < \deg P$ के साथ
$P \mid P'$: $P' = 0$। $P = \sum a_kX^k$ लिखने पर: सभी $k$ के लिए $ka_k = 0$, अतः
अभिलक्षण $0$ में $P$ अचर होता (जो अपवर्जित है); और अभिलक्षण
$p$ में $a_k = 0$, सिवाय तब जब $p \mid k$: अर्थात् $P =
Q(X^p)$। किसी \omterm{thm:b3:galois:finitefields}{परिमित
क्षेत्र} पर प्रत्येक अवयव कोई $p$-वीं \omterm{def:b3:galois:extension}{घात} होता है (\omterm{thm:b3:galois:finitefields}{फ्रोबेनियस}
आच्छादक है), अतः $Q(X^p) = \sum b_k^p X^{pk} = (\sum
b_kX^k)^p$ \omterm{def:b3:rings:divisibility}{अखंडनीय} नहीं होता: वहाँ भी \omterm{def:b3:rings:divisibility}{अखंडनीय} $P$
के लिए $P' = 0$ नहीं हो सकता।
\end{proof}

\begin{proposition}[अंतःस्थापनों की गणना]\label{prop:b3:galois:embeddings}
मान लीजिए $L = K(\alpha_1, \dots, \alpha_r)$ $K$ पर परिमित है, और $\sigma \colon K \to \Omega$ किसी \omterm{def:b3:galois:closure}{बीजीय रूप से
संवृत क्षेत्र} में एक अंतःस्थापन। तब $\sigma$ के $L$ तक विस्तारों
की संख्या अधिकतम $[L:K]$ है, और यदि $L/K$ \omterm{def:b3:galois:separable}{वियोज्य} हो तो समता होती
है। विशेष रूप से $\abs{\operatorname{Aut}_K(L)} \leq [L:K]$।
\end{proposition}

\begin{proof}
सरल चरणों के माध्यम से $[L:K]$ पर आगमन। $L = K(\alpha)$ के लिए: कोई विस्तार
$\tau$ $\tau(\alpha)$ से निर्धारित होता है, जिसे $\pi_\alpha^\sigma$ का $\Omega$ में मूल
होना चाहिए; विलोमतः ऐसा प्रत्येक मूल एक विस्तार देता है ($K(\alpha) \cong K[X]/(\pi_\alpha)$)।
विस्तारों की संख्या $\Omega$ में $\pi_\alpha^\sigma$ के \emph{भिन्न} मूलों की
संख्या है: अधिकतम $\deg\pi_\alpha =
[K(\alpha):K]$, और समता तभी और केवल तभी जब $\pi_\alpha$
\omterm{def:b3:galois:separable}{वियोज्य} हो ($\pi^\sigma$ और $\pi$ की वियोज्यता एक ही बात है: अवकलज के
साथ महत्तम समापवर्तक $\sigma$ से सुरक्षित रहता है)। व्यापक स्थिति
में $L =
K(\alpha_1)(\alpha_2, \dots)$ का गुणनखंडन कीजिए: $\sigma$ के $K(\alpha_1)$ तक विस्तार $\leq [K(\alpha_1):K]$
हैं, और आगमन से उनमें से प्रत्येक $\leq [L : K(\alpha_1)]$ प्रकार से आगे विस्तारित
होता है; \omterm{thm:b3:galois:tower}{मीनार नियम} से इन्हें गुणा कीजिए। \omterm{def:b3:galois:separable}{वियोज्य} स्थिति में
दोनों गणनाएँ समताएँ हैं: बड़े क्षेत्र $K(\alpha_1)$ पर \omterm{def:b3:galois:algebraic}{न्यूनतम बहुपद}
$K$ पर के बहुपदों को विभाजित करते हैं, अतः \omterm{def:b3:galois:separable}{वियोज्य} बने रहते
हैं।
\end{proof}

\begin{theorem}[आद्य अवयव]\label{thm:b3:galois:primitive}
प्रत्येक परिमित \emph{\omterm{def:b3:galois:separable}{वियोज्य}} विस्तार सरल होता है: किसी $\gamma$ के
लिए $L =
K(\gamma)$।\index{आद्य अवयव प्रमेय}
\end{theorem}

\begin{proof}
यदि $K$ परिमित हो, तो $L$ भी परिमित है, और चक्रीय समूह
$L^\times$ (\cref{thm:b3:galois:cyclic}) का कोई जनक $\gamma$ काम कर देता है। अब मान लीजिए
$K$ अनंत है; आगमन से केवल $L =
K(\alpha, \beta)$ की स्थिति देखनी पर्याप्त है।
मान लीजिए $n = [L:K]$; \cref{prop:b3:galois:embeddings} से $n$ भिन्न $K$-अंतःस्थापन $\sigma_1, \dots, \sigma_n \colon L \to \Omega$
हैं ($\Omega$ एक \omterm{def:b3:galois:closure}{बीजीय संवृति})। बहुपद
\[
D(T)=\prod_{i<j}\bigl[\bigl(\sigma_i(\alpha)-\sigma_j(\alpha)
\bigr) + T\bigl(\sigma_i(\beta) - \sigma_j(\beta)\bigr)\bigr]
\]
सर्वथा शून्य नहीं है: कोई गुणनखंड सर्वथा लुप्त तभी होगा जब
$\sigma_i, \sigma_j$ $\alpha$ और $\beta$ दोनों पर, अतः $L$ पर, सहमत हों --- जो
$i \neq j$ के लिए अपवर्जित है। चूँकि $K$ अनंत है, $D(c) \ne 0$ वाला $c \in
K$
चुनिए: तब $\sigma_i(\alpha +
c\beta)$ ये $n$ अवयव जोड़े-जोड़े में भिन्न हैं, अतः
$\gamma = \alpha + c\beta$ के $\Omega$ में कम से कम $n$ भिन्न संयुग्म हैं, अर्थात्
$\deg \pi_\gamma \geq n$: और $[K(\gamma):K] \geq n = [L:K]$ से $L = K(\gamma)$ अनिवार्य हो जाता है।
\end{proof}

\section{गाल्वा संगति}

\begin{definition}\label{def:b3:galois:galois}
कोई परिमित विस्तार $L/K$ \emph{गाल्वा}\index{गाल्वा विस्तार}
कहलाता है यदि वह $K$ पर किसी \emph{\omterm{def:b3:galois:separable}{वियोज्य}} बहुपद का \omterm{thm:b3:galois:splitting}{विभाजन
क्षेत्र} हो। उसका \emph{गाल्वा समूह}\index{गाल्वा समूह} $\operatorname{Gal}(L/K) = \operatorname{Aut}_K(L)$ है,
अर्थात् $L$ की उन क्षेत्र स्वसमाकारिताओं का समूह जो $K$ को
बिंदुशः स्थिर रखती हैं।
\end{definition}

\begin{proposition}\label{prop:b3:galois:galoisorder}
यदि $L/K$ \omterm{def:b3:galois:galois}{गाल्वा} हो, तो $\abs{\operatorname{Gal}(L/K)} =
[L:K]$; इसके अतिरिक्त प्रत्येक मध्यवर्ती
क्षेत्र $K
\subseteq F \subseteq L$ के लिए $L/F$ \omterm{def:b3:galois:galois}{गाल्वा} है, और प्रत्येक
$F$-अंतःस्थापन $L \to \Omega
\supseteq L$ का प्रतिबिंब $L$ होता है
(\emph{प्रसामान्यता})।
\end{proposition}

\begin{proof}
मान लीजिए $L$ $K$ पर \omterm{def:b3:galois:separable}{वियोज्य} $P$ को विभाजित करता है, और
कोई \omterm{def:b3:galois:closure}{बीजीय संवृति} $\Omega \supseteq L$ स्थिर कीजिए। $L/K$ \omterm{def:b3:galois:separable}{वियोज्य} है: वह $P$
के मूलों से जनित है; प्रत्येक अवयव की वियोज्यता नीचे की समता
स्थिति से निकलती है, पर सीधे तर्क करें --- जनकों पर (जो \omterm{def:b3:galois:separable}{वियोज्य}
$P$ के मूल हैं और जिनके \omterm{def:b3:galois:algebraic}{न्यूनतम बहुपद} $P$ को विभाजित करते
हैं) \cref{prop:b3:galois:embeddings} लगाने से $K \hookrightarrow \Omega$ के ठीक $[L:K]$ विस्तार मिलते हैं (आगमन
चरण में किसी मध्यवर्ती क्षेत्र पर $P$ के मूल का \omterm{def:b3:galois:algebraic}{न्यूनतम बहुपद}
अब भी $P$ को विभाजित करता है, अतः \omterm{def:b3:galois:separable}{वियोज्य} है)। ऐसा प्रत्येक
अंतःस्थापन $\tau \colon L \to \Omega$ $P$ के मूलों का क्रमचय करता है ($\tau$ गुणांक
स्थिर रखता है), और $L$ उन्हीं से जनित है: अतः $\tau(L) = L$। इसलिए
अंतःस्थापन $=$ स्वसमाकारिताएँ हैं: $\abs{\operatorname{Gal}(L/K)} = [L:K]$। मध्यवर्ती $F$ के
लिए: $L$ $F$ पर भी $P$ का \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} है, और $P$
\omterm{def:b3:galois:separable}{वियोज्य} बना रहता है: अतः $L/F$ \omterm{def:b3:galois:galois}{गाल्वा} है; और वही तर्क $F$ पर
प्रसामान्यता दे देता है।
\end{proof}

\begin{lemma}[आर्टिन]\label{lem:b3:galois:artin}
मान लीजिए $G$ किसी क्षेत्र $L$ की स्वसमाकारिताओं का परिमित
समूह है और $K = L^G = \{x : \sigma(x) = x\ \forall\sigma \in G\}$ उसका स्थिर क्षेत्र। तब $[L : L^G] \leq \abs G$।
\end{lemma}

\begin{proof}
मान लीजिए $n = \abs G$, $G = \{\sigma_1, \dots, \sigma_n\}$, और मान लीजिए $x_1, \dots, x_{n+1} \in L$ $K$ पर रैखिकतः
स्वतंत्र हैं। $L$ पर $n+1$ अज्ञातों $(c_j)$ में $n$ समीकरणों
का समघात रैखिक निकाय
\[
\sum_{j=1}^{n+1} c_j\,\sigma_i(x_j) = 0
\qquad (i = 1, \dots, n),
\]
कोई शून्येतर हल रखता है; ऐसा हल चुनिए जिसमें \emph{सबसे कम}
शून्येतर प्रविष्टियाँ हों, मान लीजिए $c_1, \dots, c_r \neq 0$ (पुनःक्रमांकन के
बाद), $r \geq 2$ (एकमात्र $c_j\sigma_i(x_j) = 0$ असंभव है), और $c_r = 1$ मानकीकृत। सभी
$c_j$ $K$ में नहीं हो सकते: अन्यथा $\sigma_i =
\mathrm{id}$ का समीकरण स्वतंत्रता
का खंडन करता; मान लीजिए $c_1 \notin K$, अतः किसी $\tau \in G$ के लिए $\tau(c_1) \ne c_1$। अब
सभी समीकरणों पर $\tau$ लगाइए: चूँकि $\tau\sigma_i$ $G$ में विचरता है,
सदिश $(\tau(c_j))_j$ एक और हल है; घटाने पर $(c_j -
\tau(c_j))_j$ ऐसा हल है जिसमें कम
शून्येतर प्रविष्टियाँ हैं ($r$-वीं प्रविष्टि $1 - 1 = 0$ लुप्त हो
जाती है, पहली नहीं) और जो शून्य नहीं है: विरोधाभास। अतः कोई भी
$n+1$ अवयव आश्रित होते हैं: $[L:K]
\leq n$।
\end{proof}

\begin{theorem}[गाल्वा सिद्धांत की मूल प्रमेय]
\label{thm:b3:galois:fundamental}
मान लीजिए $L/K$ समूह $G =
\operatorname{Gal}(L/K)$ वाला \omterm{def:b3:galois:galois}{गाल्वा विस्तार} है।\index{गाल्वा
संगति}
\begin{enumerate}
  \item $L^G = K$।
  \item प्रतिचित्रण $H \mapsto L^H$ और $F \mapsto
        \operatorname{Gal}(L/F)$ $G$ के उपसमूहों और
        मध्यवर्ती क्षेत्रों $K \subseteq F \subseteq L$ के बीच परस्पर प्रतिलोम,
        अंतर्वेशन-उलटने वाली एकैकी आच्छादक संगतियाँ हैं; इसके
        अतिरिक्त $[L : L^H] = \abs H$ और $[L^H : K] = [G : H]$।
  \item $H \trianglelefteq G$ तभी और केवल तभी जब $L^H/K$ \omterm{def:b3:galois:galois}{गाल्वा} हो, और तब प्रतिबंध
        से $\operatorname{Gal}(L^H/K) \cong
        G/H$ प्रेरित होता है।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) स्पष्टतः $K \subseteq L^G$। विलोमतः मान लीजिए $\alpha \in L
\setminus K$; हम $\sigma(\alpha) \ne
\alpha$ वाला
$\sigma \in G$ प्रस्तुत करते हैं। $K$ पर \omterm{def:b3:galois:algebraic}{न्यूनतम बहुपद} $\pi_\alpha$ की \omterm{def:b3:galois:extension}{घात}
$\geq 2$ है और वह \omterm{def:b3:galois:separable}{वियोज्य} है ($L/K$ \omterm{def:b3:galois:separable}{वियोज्य}, \cref{prop:b3:galois:galoisorder}), अतः किसी
\omterm{def:b3:galois:closure}{बीजीय संवृति} $\Omega \supseteq L$ में उसका कोई दूसरा मूल $\beta \neq \alpha$ है।
$K$-अंतःस्थापन $K(\alpha) \to \Omega$, $\alpha \mapsto
\beta$, को किसी अंतःस्थापन $\tau\colon L \to \Omega$ तक
विस्तारित कीजिए (\cref{prop:b3:galois:embeddings}); प्रसामान्यता (\cref{prop:b3:galois:galoisorder}) से $\tau(L) = L$, अतः
$\tau \in
G$, $\beta = \tau(\alpha) \in L$, और $\tau(\alpha) \neq
\alpha$।

(2) किसी उपसमूह $H$ के लिए: $L/L^H$ \omterm{def:b3:galois:galois}{गाल्वा} है (\cref{prop:b3:galois:galoisorder}), और तुच्छ
रूप से $\operatorname{Gal}(L/L^H) \supseteq H$, अतः $[L:L^H] =
\abs{\operatorname{Gal}(L/L^H)} \geq \abs H$; आर्टिन की प्रमेयिका $[L:L^H] \leq \abs H$ देती है:
इसलिए समता, और $\operatorname{Gal}(L/L^H) = H$। किसी मध्यवर्ती क्षेत्र $F$ के लिए:
$L/F$ \omterm{def:b3:galois:galois}{गाल्वा} होने से $L/F$ पर (1) लगाकर $L^{\operatorname{Gal}(L/F)} = F$ मिलता है। दोनों
प्रतिचित्रण परस्पर प्रतिलोम हैं; और वे स्पष्टतः अंतर्वेशन उलट
देते हैं। घातें: $[L:L^H] = \abs H$ अभी सिद्ध हुआ, और $[L^H:K] = [L:K]/[L:L^H] = \abs G/\abs H$।

(3) $\sigma \in G$ और $H \leq G$ के लिए: $\sigma(L^H) =
L^{\sigma H\sigma^{-1}}$ (सीधी जाँच)। उस एकैकी आच्छादक
संगति से, सभी $\sigma$ के लिए $\sigma(L^H) = L^H$ तभी और केवल तभी जब $H \trianglelefteq G$। अब
यदि $H \trianglelefteq G$ हो, तो $F = L^H$ रखिए: प्रत्येक $\sigma \in
G$ $F$ की किसी
स्वसमाकारिता तक प्रतिबंधित हो जाता है, जिससे अष्टि $\{\sigma :
\sigma\restriction_F = \mathrm{id}\} = \operatorname{Gal}(L/F) =
H$ वाली
समाकारिता $\rho
\colon G \to \operatorname{Aut}_K(F)$ मिलती है। अतः $G/H$ $\operatorname{Aut}_K(F)$ में अंतःस्थापित हो
जाता है, जिससे $\abs{\operatorname{Aut}_K(F)} \geq [G:H] = [F:K]$; और विपरीत असमिका सदैव लागू होती है
(\cref{prop:b3:galois:embeddings}): अतः $\abs{\operatorname{Aut}_K(F)} = [F:K]$ और $\rho$ आच्छादक है। अब यह देखना शेष है कि
$F/K$ \omterm{def:b3:galois:galois}{गाल्वा} है: $F$ $K$ पर \omterm{def:b3:galois:separable}{वियोज्य} है (\omterm{def:b3:galois:separable}{वियोज्य} $L/K$ के
भीतर), और $F = K(\gamma)$ (\cref{thm:b3:galois:primitive}); बहुपद $\prod_{\sigma \in G/H}\bigl(X - \sigma(\gamma)\bigr)$ (गुणन भिन्न प्रतिबिंबों
पर, जो $F$ में होते हैं: $H$ की प्रसामान्यता से $\sigma(F) =
L^{\sigma H\sigma^{-1}} = L^H = F$) के
गुणांक $G$ से स्थिर रहते हैं, अतः (1) से वे $K$ में हैं: यह
$K[X]$ का \omterm{def:b3:galois:separable}{वियोज्य बहुपद} है जिसे $F$ विभाजित करता है, और उसके
मूल $F$ को जनित करते हैं: अतः $F/K$ \omterm{def:b3:galois:galois}{गाल्वा} है। विलोमतः, यदि
$F/K$ के \omterm{def:b3:galois:galois}{गाल्वा} होते हुए $F = L^H$ हो, तो $F$ की प्रसामान्यता
(\cref{prop:b3:galois:galoisorder}, जिसे $G$ के अवयवों से प्रतिबंधित अंतःस्थापनों $F \to \Omega$ पर
लगाया गया है) से सभी $\sigma \in G$ के लिए $\sigma(F) = F$, अर्थात् $H
\trianglelefteq G$।
\end{proof}

\section{वृत्तखंडी विस्तार}

\begin{definition}\label{def:b3:galois:cyclotomic}
मान लीजिए $n \geq 1$ और $\zeta_n = \eu^{2\iu\pi/n}$। $n$-वाँ \emph{वृत्तखंडी
बहुपद}\index{वृत्तखंडी बहुपद} $\Phi_n = \prod_{\gcd(k,n)=1,\ 1 \le k \le n} \bigl(X -
\zeta_n^k\bigr)$ है, जिसकी \omterm{def:b3:galois:extension}{घात} $\varphi(n)$ है;
$X^n - 1$ के मूलों को उनके ठीक क्रम के अनुसार समूहबद्ध करने पर
$X^n - 1 = \prod_{d \mid n}\Phi_d$, जिससे आगमन द्वारा $\Phi_n \in \Z[X]$ सिद्ध हो जाता है (मोनिक पूर्णांक
बहुपदों का यूक्लिडीय विभाजन)।
\end{definition}

\begin{theorem}\label{thm:b3:galois:cyclotomicirred}
$\Phi_n$ $\Q$ पर \omterm{def:b3:rings:divisibility}{अखंडनीय} है; अतः $[\Q(\zeta_n) : \Q] =
\varphi(n)$ और
\[
\operatorname{Gal}\bigl(\Q(\zeta_n)/\Q\bigr) \;\cong\;
(\Z/n\Z)^\times,
\qquad \sigma_a(\zeta_n) = \zeta_n^a .
\]
इस प्रकार विस्तार $\Q(\zeta_n)/\Q$ \emph{आबेली} समूह वाला \omterm{def:b3:galois:galois}{गाल्वा विस्तार}
है।
\end{theorem}

\begin{proof}
मान लीजिए $f = \pi_{\zeta_n}$, अतः मोनिक $f, g \in \Z[X]$ के साथ $\Phi_n = fg$ (गाउस की प्रमेयिका
\cref{lem:b3:rings:gauss}: अंतर्वस्तुएँ गुणा होती हैं, और सभी बहुपद मोनिक हैं)।
\emph{दावा: यदि $\zeta$ $f$ का मूल हो और $p \nmid n$ अभाज्य हो, तो
$\zeta^p$ $f$ का मूल है।} अन्यथा $\zeta^p$ $g$ का मूल है (वह इकाई
का आद्य $n$-वाँ मूल है), अतः $\zeta$ $g(X^p)$ का मूल है, और $\Z[X]$
में $f
\mid g(X^p)$ (\omterm{def:b3:galois:algebraic}{न्यूनतम बहुपद}, फिर पुनः गाउस)। $p$ के सापेक्ष अपचयन
कीजिए: $\bar g(X^p) = \bar g(X)^p$ ($\mathbb F_p[X]$ पर \omterm{thm:b3:galois:finitefields}{फ्रोबेनियस}: गुणांकशः $a^p = a$, और नौसिखिए का
स्वप्न), अतः $\bar f \mid \bar g^{\,p}$: इस प्रकार $\bar f$ और $\bar g$ में एक उभयनिष्ठ
\omterm{def:b3:rings:divisibility}{अखंडनीय} गुणनखंड है, और $\bar\Phi_n = \bar f\bar g$ में कोई पुनरावृत्त गुणनखंड है। तब
$X^n - \bar 1$ में भी होगा; परंतु उसका अवकलज $\bar nX^{n-1}$ उसके साथ सह-अभाज्य है
($p \nmid n$, और $0$ मूल नहीं है): विरोधाभास।

प्रत्येक आद्य मूल $\zeta_n^k$ ($\gcd(k, n) = 1$) $\zeta_n$ से उन अभाज्य घातों द्वारा
प्राप्त होता है जो $n$ को विभाजित नहीं करतीं ($k$ का
गुणनखंडन कीजिए): दावा आगे बढ़ता जाता है, अतः प्रत्येक आद्य मूल
$f$ का मूल है: इसलिए $f = \Phi_n$, जो \omterm{def:b3:rings:divisibility}{अखंडनीय} है। फलस्वरूप $[\Q(\zeta_n):\Q] = \varphi(n)$, और
$\Q(\zeta_n)$ \omterm{def:b3:galois:separable}{वियोज्य} $X^n - 1$ का \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} है (सभी मूल $\zeta_n$ की घातें
हैं): अतः \omterm{def:b3:galois:galois}{गाल्वा}। कोई स्वसमाकारिता $\sigma$ $\zeta_n$ को किसी दूसरे
आद्य मूल $\zeta_n^{a(\sigma)}$ पर भेजती है, और $\sigma
\mapsto a(\sigma)$ $(\Z/n\Z)^\times$ में एक एकैकी
समाकारिता है; दोनों समूहों का क्रम $\varphi(n)$ है: अतः तुल्याकारिता।
\end{proof}

\section{पैमाना और परकार}

\begin{definition}\label{def:b3:galois:constructible}
समतल की पहचान $\C$ से कीजिए; $\{0, 1\}$ से आरंभ कीजिए। कोई बिंदु
\emph{रचनीय}\index{रचनीय संख्या} कहलाता है यदि वह पहले से रचे गए
दो बिंदुओं से जाती रेखाओं और किसी रचे गए बिंदु पर केंद्रित उन
वृत्तों के, जिनकी त्रिज्या दो रचे गए बिंदुओं की दूरी हो, परिमित
संख्या के प्रतिच्छेदनों से प्राप्त हो सके।
\end{definition}

\begin{theorem}[वांत्ज़ेल]\label{thm:b3:galois:wantzel}
$z \in \C$ \omterm{def:b3:galois:constructible}{रचनीय} है तभी और केवल तभी जब $[F_{i+1} :
F_i] = 2$ और $z \in F_r$ वाली कोई मीनार
$\Q = F_0
\subseteq F_1 \subseteq \dots \subseteq F_r$ हो। विशेष रूप से, कोई \omterm{def:b3:galois:constructible}{रचनीय संख्या} \omterm{def:b3:galois:algebraic}{बीजीय} होती है और $\Q$
पर उसकी \omterm{def:b3:galois:extension}{घात} $2$ की कोई \omterm{def:b3:galois:extension}{घात} होती है।\index{वांत्ज़ेल की प्रमेय}
\end{theorem}

\begin{proof}
($\Rightarrow$) किसी उपक्षेत्र $F \subseteq \R$ में निर्देशांक रखने वाले बिंदुओं से
जाती दो रेखाओं के प्रतिच्छेदन के निर्देशांक $F$ पर एक रैखिक
निकाय हल करते हैं: वे $F$ में ही रहते हैं। रेखा--वृत्त और
वृत्त--वृत्त प्रतिच्छेदन, रैखिक अंश हटाने के बाद (दोनों वृत्त
समीकरण घटाने से एक रेखा मिलती है), $F$ पर एक द्विघात समीकरण
पर पहुँचते हैं: अतः नए निर्देशांक $F$ में होते हैं या किसी
$d \in F$, $d > 0$ के लिए $F(\sqrt d)$ में। आगमन से प्रत्येक रचे गए बिंदु के
निर्देशांक $\Q$ के द्विघात विस्तारों की किसी मीनार में होते
हैं; और $z = x + \iu y$ भी किसी द्विघात मीनार में होता है ($\iu$ जोड़िए:
एक और द्विघात चरण)। \omterm{def:b3:galois:extension}{घात} संबंधी परिणाम: \omterm{thm:b3:galois:tower}{मीनार नियम} से $[\Q(z):\Q]$
$[F_r : \Q] = 2^r$ को विभाजित करता है।

($\Leftarrow$) \omterm{def:b3:galois:constructible}{रचनीय} संख्याएँ एक क्षेत्र बनाती हैं: योग और अंतर
समांतर चतुर्भुजों से (समांतर रेखाएँ \omterm{def:b3:galois:constructible}{रचनीय} हैं: दो बार लंब गिराइए
और उठाइए --- किसी बिंदु से जाता शास्त्रीय लंब एक वृत्त और दो
चापों से बनता है); गुणनफल और भागफल थेल्स के अंतःखंड विन्यासों से
(दी हुई लंबाइयों $a, b$ से दो किरणों पर समरूप त्रिभुजों द्वारा
$ab$ और $a/b$ रचिए)। और यह क्षेत्र वर्गमूल के अंतर्गत संवृत है:
$a > 0$ के लिए, व्यास $1 + a$ वाला वृत्त और संधि बिंदु पर खींचा गया
लंब ऊँचाई $\sqrt a$ पर मिलते हैं (समकोण त्रिभुज में शीर्षलंब का
गुणोत्तर-माध्य संबंध); और किसी सम्मिश्र $w = \rho\eu^{\iu\theta}$ के लिए $\sqrt\rho$ रचकर
$\theta$ का अर्धक बनाइए (कोण का अर्धकन परकार से होने वाली रचना है)।
अतः किसी द्विघात मीनार के सदस्यों के वास्तविक और काल्पनिक भाग
मीनार पर आगमन द्वारा \omterm{def:b3:galois:constructible}{रचनीय} होते हैं: प्रत्येक चरण किसी द्विघात
के मूल जोड़ता है, जो क्षेत्र संक्रियाओं और पहले से रची गई किसी
संख्या के एक वर्गमूल से व्यक्त हो जाते हैं (द्विघात सूत्र;
अभिलक्षण $0$ में)।
\end{proof}

\begin{corollary}\label{cor:b3:galois:impossible}
तीनों शास्त्रीय समस्याएँ पैमाने और परकार से अहल्य हैं:\index{पैमाना
और परकार}
\begin{enumerate}
  \item \emph{घन का द्विगुणन}: $\sqrt[3]2$ की \omterm{def:b3:galois:extension}{घात} $3$ है, जो $2$
        की \omterm{def:b3:galois:extension}{घात} नहीं है।
  \item \emph{कोण का त्रिविभाजन}: $60^\circ$ के त्रिविभाजन के लिए
        $\cos 20^\circ$ चाहिए, जो \omterm{def:b3:rings:divisibility}{अखंडनीय} $8X^3
        - 6X - 1$ का मूल है: \omterm{def:b3:galois:extension}{घात} $3$।
  \item \emph{वृत्त का वर्गीकरण}: $\sqrt\pi$ अबीजीय है ($\pi$ अबीजीय
        है --- लिंडेमान की प्रमेय, जिसे यहाँ स्वीकार किया गया
        है: उसकी उपपत्ति अबीजीयता सिद्धांत के पाठ्यक्रम की वस्तु
        है)।
\end{enumerate}
साथ ही, सम $n$-भुज \omterm{def:b3:galois:constructible}{रचनीय} है तभी और केवल तभी जब $\varphi(n)$ $2$ की
कोई \omterm{def:b3:galois:extension}{घात} हो (गाउस--वांत्ज़ेल; ``तभी'' वाला भाग सप्ताहांत समस्या
की विधि से आता है, और ``केवल तभी'' वाला \omterm{def:b3:galois:extension}{घात} $\varphi(n)$ के $\zeta_n$ पर
\cref{thm:b3:galois:wantzel} लगाने से)। $n = 7$ के लिए: $\varphi(7) = 6$: सम सप्तभुज असंभव है; और
$n = 17$ के लिए: $\varphi(17) = 16 = 2^4$: \omterm{def:b3:galois:constructible}{रचनीय} --- सप्ताहांत समस्या उसकी रचना करती
है।
\end{corollary}

\begin{proof}
(1) $X^3 - 2$ \omterm{def:b3:rings:divisibility}{अखंडनीय} है (\omterm{thm:b3:rings:criteria}{आइज़ेनश्टाइन})। (2) $3\theta =
60^\circ$ वाले $\cos
3\theta = 4\cos^3\theta - 3\cos\theta$ से:
$c = \cos 20^\circ$ के लिए $8c^3 - 6c = 1$; त्रिघात $8X^3 - 6X - 1$ का कोई परिमेय मूल नहीं है
(उम्मीदवार $\pm1, \pm\frac
12, \pm\frac14, \pm\frac18$ विफल हो जाते हैं), अतः वह \omterm{def:b3:rings:divisibility}{अखंडनीय} है: \omterm{def:b3:galois:extension}{घात}
$3$। कोई भी सामान्य $60^\circ$ कोण \omterm{def:b3:galois:constructible}{रचनीय} है, अतः कोई त्रिविभाजक
$c$ की रचना कर देता। (3) यदि $\sqrt\pi$ \omterm{def:b3:galois:constructible}{रचनीय} होता तो वह \omterm{def:b3:galois:algebraic}{बीजीय}
होता, और इसलिए $\pi$ भी। $n$-भुज संबंधी कथन: $\zeta_n$ की \omterm{def:b3:galois:extension}{घात}
$\varphi(n)$ है (\cref{thm:b3:galois:cyclotomicirred}); आवश्यकता वांत्ज़ेल से निकलती है; और
पर्याप्तता के लिए, क्रम $2^m$ का आबेली \omterm{def:b3:galois:galois}{गाल्वा समूह} सूचकांक-$2$
उपसमूहों की एक शृंखला स्वीकार करता है (कोई भी परिमित $2$-समूह
करता है: \cref{exo:b3:groups:10}), जिनके स्थिर क्षेत्र $\Q(\zeta_n)$ पर समाप्त होने वाली
एक द्विघात मीनार बनाते हैं (\cref{thm:b3:galois:fundamental}); अब \cref{thm:b3:galois:wantzel} से निष्कर्ष
निकालिए।
\end{proof}

\section{मूलकों द्वारा हल्यता}

\begin{definition}\label{def:b3:galois:radical}
कोई विस्तार $L/K$ (इस पूरे अनुभाग में अभिलक्षण $0$)
\emph{मूलक} कहलाता है यदि $F_{i+1} = F_i(\alpha_i)$, $\alpha_i^{n_i} \in F_i$ वाली कोई मीनार $K = F_0 \subseteq \dots
\subseteq F_r = L$ हो:
अर्थात् प्रत्येक चरण कोई $n_i$-वाँ मूल जोड़ता हो। कोई बहुपद
$P \in K[X]$ \emph{मूलकों द्वारा हल्य}\index{मूलकों द्वारा हल्य} कहलाता
है यदि उसका \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} $K$ के किसी मूलक विस्तार में
अंतर्विष्ट हो।
\end{definition}

\begin{lemma}\label{lem:b3:galois:cyclicsteps}
मान लीजिए $K$ में इकाई का कोई आद्य $n$-वाँ मूल $\zeta$ है,
अर्थात् $K^\times$ में क्रम $n$ का $\zeta$, और $a \in K^\times$। तब $K(\sqrt[n]a)/K$
\emph{चक्रीय} समूह वाला \omterm{def:b3:galois:galois}{गाल्वा विस्तार} है। विलोमतः --- जिसकी नीचे
आवश्यकता नहीं --- \omterm{def:b3:galois:extension}{घात} $n$ का प्रत्येक चक्रीय विस्तार इसी रूप
का होता है। इसके अतिरिक्त अभिलक्षण $0$ वाले किसी भी $K$ के
लिए $K(\zeta_n)/K$ \emph{आबेली} समूह वाला \omterm{def:b3:galois:galois}{गाल्वा विस्तार} है।
\end{lemma}

\begin{proof}
$X^n - a$ \omterm{def:b3:galois:separable}{वियोज्य} है ($nX^{n-1}$ के साथ $\gcd$: $a \neq 0$) और $K(\alpha)$, $\alpha^n = a$ में
विभाजित हो जाता है: उसके मूल $\zeta^k\alpha \in K(\alpha)$ हैं। अतः $K(\alpha)/K$ \omterm{def:b3:galois:galois}{गाल्वा} है;
प्रतिचित्रण $\sigma \mapsto \sigma(\alpha)/\alpha \in \mu_n = \langle
\zeta\rangle$ एक एकैकी समाकारिता है ($\sigma\tau(\alpha) =
\sigma(\tau(\alpha)/\alpha \cdot \alpha) =
\tau(\alpha)/\alpha\cdot\sigma(\alpha)$, क्योंकि भागफल
$K$ में है) जो किसी चक्रीय समूह में जाती है: अतः $\operatorname{Gal}$ चक्रीय
है। विलोम कुम्मर सिद्धांत है, जिसकी हमें आवश्यकता नहीं (नीचे की
टिप्पणी देखिए)। $K(\zeta_n)$ के लिए: वह \omterm{def:b3:galois:separable}{वियोज्य} $X^n -
1$ को विभाजित करता
है, और $\sigma(\zeta_n) =
\zeta_n^{a(\sigma)}$ के साथ $\sigma \mapsto a(\sigma)$ समूह को \cref{thm:b3:galois:cyclotomicirred} की तरह आबेली $(\Z/n\Z)^\times$ में
अंतःस्थापित कर देता है (एकैकीपन के लिए केवल इतना चाहिए कि $\zeta_n$
संबंधित इकाई-मूलों को जनित करे)।
\end{proof}

\begin{theorem}[गाल्वा]\label{thm:b3:galois:solvable}
मान लीजिए $K$ का अभिलक्षण $0$ है और $P \in K[X]$ का \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र}
$L$ है। यदि $P$ \omterm{def:b3:galois:radical}{मूलकों द्वारा हल्य} हो, तो $\operatorname{Gal}(L/K)$ एक \omterm{def:b3:groups:derived}{विलेय
समूह} है। (विलोम भी सत्य है; हमें उसकी आवश्यकता नहीं
होगी।)\index{गाल्वा की हल्यता प्रमेय}
\end{theorem}

\begin{proof}
चरण 1: \emph{\omterm{def:b3:galois:radical}{मूलक} मीनार को बड़ा करके \omterm{def:b3:galois:galois}{गाल्वा} बनाइए।} मान लीजिए
$M/K$ \omterm{def:b3:galois:radical}{मूलक} होते हुए $L
\subseteq M$, जिसके मूल घातांक $n_1, \dots,
n_r$ हैं और
$n = n_1\cdots n_r$। पहले $\zeta_n$ जोड़िए: मीनार $K \subseteq K(\zeta_n) \subseteq M(\zeta_n)$ अब भी \omterm{def:b3:galois:radical}{मूलक} है ($\zeta_n$
इकाई का मूल है: एक \omterm{def:b3:galois:radical}{मूलक} चरण, $\zeta_n^n = 1$), और पहले चरण के बाद उसके सभी
चरण उन क्षेत्रों पर होते हैं जिनमें अभीष्ट इकाई-मूल विद्यमान
हैं। इसके बाद $M(\zeta_n)$ के स्थान पर समस्त $\sigma(M(\zeta_n))$ का संयुक्त क्षेत्र
$N$ रखिए, जहाँ $\sigma$ किसी स्थिर \omterm{def:b3:galois:closure}{बीजीय संवृति} में $M(\zeta_n)$ के
(परिमित संख्या के) $K$-अंतःस्थापनों में विचरता है: $N$ किसी
जनक समुच्चय के न्यूनतम बहुपदों के गुणनफल का \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} है
(अभिलक्षण $0$: परिमित और \omterm{def:b3:galois:separable}{वियोज्य}), अतः $N/K$ \omterm{def:b3:galois:galois}{गाल्वा} है; और
$N$ $K$ पर \omterm{def:b3:galois:radical}{मूलक} है: प्रत्येक $\sigma(M(\zeta_n))$ $K$ पर \omterm{def:b3:galois:radical}{मूलक} है (किसी
\omterm{def:b3:galois:radical}{मूलक} मीनार पर $\sigma$ लगाइए), और \omterm{def:b3:galois:radical}{मूलक} विस्तारों का संयुक्त क्षेत्र
\omterm{def:b3:galois:radical}{मूलक} होता है (मीनारों को जोड़ दीजिए: यदि $F'/K$ $\beta_j$ जोड़ने वाली
मीनार के साथ \omterm{def:b3:galois:radical}{मूलक} हो, तो $F''(\beta_j)$ प्रकार के चरण किसी भी बड़े आधार
पर \omterm{def:b3:galois:radical}{मूलक} बने रहते हैं)।

चरण 2: \emph{\omterm{def:b3:galois:galois}{गाल्वा} मीनार में हल्यता पढ़िए।} अतः मान लीजिए
$L \subseteq N$, जहाँ $N/K$ \omterm{def:b3:galois:galois}{गाल्वा} और \omterm{def:b3:galois:radical}{मूलक} है और उसकी मीनार $K
\subseteq K(\zeta_n) = E_0 \subseteq E_1 \subseteq \dots
\subseteq E_s = N$ है
जिसमें प्रत्येक $E_{i+1} = E_i(\sqrt[n_i]{a_i})$ के लिए $\zeta_{n_i} \in E_0 \subseteq E_i$। मान लीजिए $G =
\operatorname{Gal}(N/K)$ और $G_i = \operatorname{Gal}(N/E_i)$:
एक घटती हुई शृंखला $G \supseteq G_0 \supseteq G_1 \supseteq
\dots \supseteq G_s = \{e\}$। प्रत्येक $E_{i+1}/E_i$ चक्रीय समूह वाला \omterm{def:b3:galois:galois}{गाल्वा}
है (\cref{lem:b3:galois:cyclicsteps}), अतः \omterm{def:b3:galois:galois}{गाल्वा विस्तार} $N/E_i$ पर मूल प्रमेय लगाने से
(\cref{thm:b3:galois:fundamental}(3), जिसमें परिवेशी समूह $G_i$ है): $G_{i+1} \trianglelefteq G_i$, जहाँ $G_i/G_{i+1} \cong
\operatorname{Gal}(E_{i+1}/E_i)$
चक्रीय है। इसी प्रकार $E_0/K$ आबेली समूह $G/G_0$ वाला \omterm{def:b3:galois:galois}{गाल्वा} है
(\cref{lem:b3:galois:cyclicsteps})। यह शृंखला $G$ को \omterm{def:b3:groups:derived}{विलेय} के रूप में प्रस्तुत कर देती है
(\cref{prop:b3:groups:derived})। अंततः $\operatorname{Gal}(L/K)$ $G$ का एक \emph{भागफल} है: $L/K$ \omterm{def:b3:galois:galois}{गाल्वा}
है (अभिलक्षण $0$ में $P$ \omterm{def:b3:galois:separable}{वियोज्य}) और प्रतिबंध $G
\to \operatorname{Gal}(L/K)$ आच्छादक
है ($H =
\operatorname{Gal}(N/L)$ के साथ \cref{thm:b3:galois:fundamental}(3)); और \omterm{def:b3:groups:derived}{विलेय} समूहों के भागफल \omterm{def:b3:groups:derived}{विलेय} होते
हैं।
\end{proof}

\begin{corollary}[पंचघात की अहल्यता]
\label{cor:b3:galois:quintic}
$\Q$ पर \omterm{def:b3:galois:extension}{घात} $5$ के ऐसे बहुपद हैं जो \omterm{def:b3:galois:radical}{मूलकों द्वारा हल्य} नहीं
हैं: उदाहरणार्थ $X^5 - 4X + 2$, जिसका \omterm{def:b3:galois:galois}{गाल्वा समूह} $S_5$ है (\cref{exo:b3:galois:11}) और जो
अविलेय समूह है (\cref{cor:b3:groups:snnotsolvable})। \omterm{def:b3:galois:extension}{घात} $\geq 5$ के लिए मूलकों में कोई व्यापक
सूत्र संभव नहीं है।\index{पंचघात, अहल्यता}
\end{corollary}

\begin{remark}\label{rem:b3:galois:converse}
\cref{thm:b3:galois:solvable} का विलोम --- कि \omterm{def:b3:groups:derived}{विलेय} \omterm{def:b3:galois:galois}{गाल्वा समूह} से मूलकों द्वारा हल्यता
निकलती है --- \omterm{def:b3:groups:derived}{व्युत्पन्न श्रेणी} में उतरकर और यह दिखाकर सिद्ध होता
है कि (पर्याप्त इकाई-मूलों सहित) प्रत्येक चक्रीय विस्तार \omterm{def:b3:galois:radical}{मूलक}
होता है; इसका साधन हैं \emph{लाग्राँज विलायक}। इससे यह भी स्पष्ट
हो जाता है कि \omterm{def:b3:galois:extension}{घात} $2, 3, 4$ के लिए सूत्र क्यों हैं: $S_2, S_3, S_4$ \omterm{def:b3:groups:derived}{विलेय} हैं
(\cref{ex:b3:groups:solvableexamples})। इस स्तर पर हम उसे स्वीकृत छोड़ देते हैं; उसका पूरा
विवेचन स्नातकोत्तर पाठ्यक्रम की वस्तु है, पर \cref{exo:b3:galois:8} उसे त्रिघात
के लिए मूर्त रूप दे देता है।
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:6}
मान लीजिए $I_d(q)$ $\mathbb F_q$ पर \omterm{def:b3:galois:extension}{घात} $d$ के मोनिक \omterm{def:b3:rings:divisibility}{अखंडनीय} बहुपदों की

X^{q^n} - X \;=\; \prod_{d \mid n}\ \prod_{P \text{ अखंडनीय

\qquad\text{अतः}\qquad
q^n = \sum_{d \mid n} d\, I_d(q).
\]
इससे $I_1, I_2, I_3, I_4$ स्पष्ट रूप से निकालिए, और प्रत्येक $d$ के लिए
$I_d(q) \geq 1$ (अतः विस्तार $\mathbb F_{q^d}/\mathbb F_q$ सभी $d$ के लिए भागफलों $\mathbb F_q[X]/(P)$ के रूप
में विद्यमान हैं)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:8}
(त्रिघात, अपने समूह से हल) मान लीजिए $P = X^3 + pX + q \in
\Q[X]$ \omterm{def:b3:rings:divisibility}{अखंडनीय} है, जिसके मूल
$x_1, x_2, x_3$ हैं और \omterm{thm:b3:galois:splitting}{विभाजन क्षेत्र} $L$। मान लीजिए $\delta = (x_1 - x_2)(x_1 - x_3)(x_2 - x_3)$ और $\Delta = \delta^2 = -4p^3 - 27q^2$
(इस शास्त्रीय सर्वसमिका को स्वीकार कीजिए अथवा सममित फलनों का
प्रसार करके सत्यापित कीजिए)।
(क) दर्शाइए कि $\Delta$ $\Q$ में वर्ग है या नहीं, इसके अनुसार
$\operatorname{Gal}(L/\Q) \cong A_3$ या $S_3$।
(ख) $j = \zeta_3$ के साथ लाग्राँज विलायक $u = x_1
+ jx_2 + j^2x_3$ और $v = x_1 + j^2x_2 + jx_3$ परिभाषित कीजिए।

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:galois:11}
मान लीजिए $P = X^5 - 4X + 2$ और $\Q$ पर उसका \omterm{def:b3:galois:galois}{गाल्वा समूह} $G$ है, जो $5$
मूलों पर क्रिया करता है।
(क) दर्शाइए कि $P$ \omterm{def:b3:rings:divisibility}{अखंडनीय} है, और इससे $5 \mid \abs G$ निष्कर्ष
निकालिए; निष्कर्ष निकालिए कि $G$ में कोई $5$-चक्र है (कोशी,
\cref{thm:b3:groups:cauchy})।
(ख) $x \mapsto x^5 - 4x +
2$ के उतार-चढ़ाव का अध्ययन करके दर्शाइए कि $P$ के ठीक
$3$ वास्तविक मूल हैं; इससे निष्कर्ष निकालिए कि सम्मिश्र
संयुग्मन $G$ में किसी स्थानांतरण तक प्रतिबंधित हो जाता है।
(ग) दर्शाइए कि $S_5$ का ऐसा उपसमूह जिसमें कोई स्थानांतरण और कोई
$5$-चक्र हो, $S_5$ ही होता है \emph{(स्थानांतरण को चक्र की
घातों से संयुग्मित कीजिए)}। $G \cong S_5$ निष्कर्ष निकालिए और \cref{thm:b3:galois:solvable} के
साथ यह भी कि $P$ \omterm{def:b3:galois:radical}{मूलकों द्वारा हल्य} नहीं है।
\end{exercise}

(ख) $D_4$ का उपसमूह जालक सूचीबद्ध कीजिए (दस उपसमूह) और प्रत्येक

\section{समस्या: गाउस और सम 17-भुज}

\begin{problem}[{सप्ताहांत समस्या --- 17-भुज की रचनीयता}]\label{pb:b3:galois:1}
30 मार्च 1796 को उन्नीस वर्ष के गाउस ने दिखाया कि सम $17$-भुज
\omterm{def:b3:galois:constructible}{रचनीय} है --- प्राचीन काल के बाद इस प्रश्न पर पहली प्रगति। हम इस
अध्याय के साधनों से उनका परिकलन पुनर्निर्मित करते हैं। $\zeta = \eu^{2\iu\pi/17}$,
$L =
\Q(\zeta)$, $G = \operatorname{Gal}(L/\Q)$ रखिए।

\medskip
\noindent\textbf{भाग I --- समूह और उसका निस्यंदन।}
\begin{enumerate}
  \item उचित ठहराइए: $[L:\Q] = 16$, $G \cong (\Z/17\Z)^\times$, जो क्रम $16$ का चक्रीय
        समूह है। सत्यापित कीजिए कि $3$ $(\Z/17\Z)^\times$ का जनक है
        \emph{($17$ के सापेक्ष $3$ की घातें परिकलित कीजिए:
        $3, 9, 10, 13, 5, 15, 11, 16, \dots$)}।
  \item मान लीजिए $\sigma(\zeta) = \zeta^3$ के साथ $\sigma \in G$, और $k = 0, \dots,
        4$ के लिए $H_k = \langle \sigma^{2^k}\rangle$।
        दर्शाइए कि $G = H_0 \supset H_1 \supset H_2 \supset
        H_3 \supset H_4 = \{e\}$, जिसमें प्रत्येक सूचकांक $[H_k :
        H_{k+1}] = 2$ है,
        और यह कि स्थिर क्षेत्र $\Q = L_0
        \subset L_1 \subset L_2 \subset L_3 \subset L_4 = L$ द्विघात विस्तारों की एक
        मीनार बनाते हैं।
  \item \cref{thm:b3:galois:wantzel} का उपयोग करके \emph{पूर्वतः} निष्कर्ष निकालिए कि
        $\zeta$ --- और इसलिए $17$-भुज --- \omterm{def:b3:galois:constructible}{रचनीय} है। शेष समस्या
        इस मीनार को स्पष्ट रूप से बनाती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- लंबाई 8 के आवर्त।} \emph{गाउस आवर्त}
परिभाषित कीजिए:
\[
\eta_0 = \sum_{k \text{ सम}} \zeta^{3^k \bmod 17}
= \zeta^{1} + \zeta^{9} + \zeta^{13} + \zeta^{15} + \zeta^{16} +
\zeta^{8} + \zeta^{4} + \zeta^{2},
\qquad
\eta_1 = \sum_{k \text{ विषम}} \zeta^{3^k \bmod 17}.
\]
\begin{enumerate}[resume]
  \item दर्शाइए कि $\eta_0, \eta_1$ $H_1$ से स्थिर रहते हैं और $\sigma$ से
        अदल-बदल जाते हैं; इससे $\eta_0, \eta_1 \in L_1$ निष्कर्ष निकालिए और यह भी
        कि वे $\Q$ पर किसी द्विघात के दो मूल हैं।
  \item $\eta_0 + \eta_1 = -1$ परिकलित कीजिए। $\eta_0\eta_1 = -4$ दर्शाइए \emph{(प्रत्येक
        गुणनफल $\zeta^a\zeta^b$ कोई $\zeta^c$ है, $c \neq 0$; गिनिए कि प्रत्येक
        $c$ कितनी बार आता है, अथवा यह तर्क दीजिए कि गुणनफल
        $G$ से स्थिर एक परिमेय पूर्णांक है, जो सभी $64$
        गुणनफलों के योग के बराबर है, और यह कि प्रत्येक शून्येतर
        शेष समान बार आता है)}।
  \item $\eta_0 = \frac{-1 + \sqrt{17}}2$, $\eta_1 =
        \frac{-1-\sqrt{17}}2$ निष्कर्ष निकालिए (संख्यात्मक रूप से पहचानिए
        कि कौन कौन है: $\eta_0 \approx 1.56$), और $L_1 =
        \Q(\sqrt{17})$।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- लंबाई 4 और 2 के आवर्त।} परिभाषित
कीजिए
\[
\beta_0 = \zeta + \zeta^{13} + \zeta^{16} + \zeta^{4},\quad
\beta_1 = \zeta^3 + \zeta^5 + \zeta^{14} + \zeta^{12},\quad
\beta_2 = \zeta^9 + \zeta^{15} + \zeta^{8} + \zeta^{2},\quad
\beta_3 = \zeta^{10} + \zeta^{11} + \zeta^{7} + \zeta^{6}.
\]
\begin{enumerate}[resume]
  \item दर्शाइए कि $\beta_0 + \beta_2 = \eta_0$, $\beta_1 + \beta_3 =
        \eta_1$, और यह कि $\beta_0, \beta_2$ $H_2$ से
        स्थिर रहते हैं और $\sigma^2$ से अदल-बदल जाते हैं।
  \item $\beta_0\beta_2 = -1$ और $\beta_1\beta_3 = -1$ परिकलित कीजिए \emph{(प्रसार कीजिए:
        प्राप्त सोलह घातांक $1, \dots, 16$ को ठीक एक-एक बार आच्छादित
        करते हैं)}।
  \item $\beta_0 = \frac{\eta_0 + \sqrt{\eta_0^2 + 4}}2$ निष्कर्ष निकालिए (चिह्न की संख्यात्मक जाँच कीजिए:
        $\beta_0 \approx 2.05$) और $\beta_1$ के लिए तत्समरूप सूत्र; अतः $L_2 =
        \Q(\beta_0)$, जो
        $L_1$ पर द्विघात है।
  \item मान लीजिए $\gamma_0 = \zeta + \zeta^{16} =
        2\cos\frac{2\pi}{17}$ और $\gamma_1 = \zeta^{13} +
        \zeta^4$। दर्शाइए कि $\gamma_0 + \gamma_1 = \beta_0$ और $\gamma_0\gamma_1 = \beta_1$,
        जिससे $\gamma_0 =
        \frac{\beta_0 + \sqrt{\beta_0^2 - 4\beta_1}}2$।
  \item $\cos\frac{2\pi}{17}$ को नेस्टेड वर्गमूलों से व्यक्त करने वाली सूत्रों
        की शृंखला संकलित कीजिए, और दशमलव जाँच दीजिए ($\cos\frac{2\pi}{17} \approx 0.93247$)।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- उपसंहार।}
\begin{enumerate}[resume]
  \item तर्क में ठीक कहाँ इसका उपयोग हुआ कि $17$ एक \emph{फर्मा
        अभाज्य} है ($17 = 2^{2^2} + 1$)? दर्शाइए कि किसी अभाज्य $p$ के
        लिए सम $p$-भुज \omterm{def:b3:galois:constructible}{रचनीय} है तभी और केवल तभी जब किसी
        $t$ के लिए $p
        = 2^{2^t} + 1$ हो \emph{(यदि $p - 1 = 2^m$ हो, तो दर्शाइए
        कि $m$ स्वयं $2$ की कोई \omterm{def:b3:galois:extension}{घात} होना चाहिए)}।
  \item \cref{cor:b3:galois:impossible} की गाउस--वांत्ज़ेल कसौटी का उपयोग करके $n \leq 20$ तक
        \omterm{def:b3:galois:constructible}{रचनीय} सम $n$-भुजों की पूरी सूची निकालिए।
\end{enumerate}

\noindent\textbf{भाग V --- गाउस योग और द्विघात पारस्परिकता।}
भाग II के आवर्तों में एक निधि छिपी है। किसी विषम अभाज्य $p$
के लिए \emph{लजांद्र संकेत} $\bigl(\frac
ap\bigr)$ $+1$ है यदि $a$ $p$
के सापेक्ष शून्येतर वर्ग हो, $-1$ यदि न हो, और $0$ यदि
$p \mid a$; \cref{exo:b3:galois:4}(ख) (ऑयलर की कसौटी) से $\bigl(\frac ap\bigr) \equiv
a^{(p-1)/2} \pmod p$ मिलता है, जिससे
गुणनात्मकता निकलती है। $\zeta =
\eu^{2\iu\pi/p}$, $p^* = (-1)^{(p-1)/2}p$ लिखिए और \emph{गाउस योग}
परिभाषित कीजिए
\[
g \;=\; \sum_{a=1}^{p-1}\Bigl(\frac ap\Bigr)\zeta^a .
\]
\begin{enumerate}[resume]
  \item दर्शाइए कि $\sum_{a=1}^{p-1}\bigl(\frac ap\bigr) = 0$ (वर्ग और अवर्ग बराबर संख्या में हैं),
        और वैकल्पिक रूप $g = \sum_{a=0}^{p-1}\zeta^{a^2}$ सिद्ध कीजिए \emph{(प्रत्येक
        शून्येतर वर्ग दो बार आता है, और $\sum_{a}\zeta^a = 0$)}। $p = 17$ के लिए:
        $g$ का भाग II के आवर्तों से संबंध जोड़िए --- दर्शाइए
        कि $g = \eta_0 - \eta_1$ \emph{($17$ के सापेक्ष वर्ग ठीक जनक $3$
        की सम घातें हैं)}।
  \item $g^2 = p^*$ सिद्ध कीजिए: \[
        g^2 = \sum_{a,b\neq0}\Bigl(\frac{ab}p\Bigr)
        \zeta^{a+b}
        = \sum_{c}\ \sum_{a \neq 0}\Bigl(\frac{a(c -
        a)}p\Bigr)\zeta^{c}
        \] का प्रसार कीजिए ($b = c - a$
        रखिए), भीतरी योग का मान निकालने के लिए $c - a = at$ प्रतिस्थापित
        कीजिए, जिससे वह $c = 0$ के लिए $\bigl(\frac{-1}p\bigr)(p - 1)$ और अन्यथा $-\bigl(\frac{-1}p\bigr)$
        निकले, और प्रश्न 14 से निष्कर्ष निकालिए। संख्यात्मक जाँच
        कीजिए: $p =
        17$ के लिए $(\eta_0 - \eta_1)^2 = 17$ (भाग II)।
  \item $\sqrt{p^*} \in \Q(\zeta_p)$ निष्कर्ष निकालिए, और यह कि $\Q(\zeta_p)$ का \emph{एकमात्र}
        द्विघात उपक्षेत्र $\Q(\sqrt{p^*})$ है --- एकमात्र इसलिए कि $\operatorname{Gal}(\Q(\zeta_p)/\Q)$
        चक्रीय है (\cref{thm:b3:galois:cyclotomicirred}) और चक्रीय समूह में सूचकांक $2$ का
        ठीक एक उपसमूह होता है। (प्रत्येक द्विघात क्षेत्र किसी न
        किसी वृत्तखंडी क्षेत्र में अंतःस्थापित हो जाता है --- यह
        क्रोनेकर--वेबर प्रमेय की पहली स्थिति है, जिसका व्यापक रूप
        बहुत आगे पड़ता है।)
  \item अब मान लीजिए $q \neq p$ एक और विषम अभाज्य है। वलय $\Z[\zeta]$ में
        $q$ के सापेक्ष कार्य करते हुए सिद्ध कीजिए
        \[
        g^q \equiv \Bigl(\frac qp\Bigr)\,g \pmod{q\Z[\zeta]}
        \]
        \emph{(नौसिखिए का स्वप्न: किसी भी क्रमविनिमेय वलय में
        $q$ के सापेक्ष $(x + y)^q \equiv x^q + y^q$; फिर $g^q \equiv
        \sum_a\bigl(\frac ap\bigr)^q\zeta^{aq}$, $b =
        aq$ का
        पुनःसूचकांकन कीजिए और $\bigl(\frac{q^{-1}}p\bigr) =
        \bigl(\frac qp\bigr)$ बाहर निकालिए)}।
  \item दूसरी ओर $g^q = g\,(g^2)^{(q-1)/2} =
        g\,(p^*)^{(q-1)/2}$; $q$ के सापेक्ष ऑयलर की कसौटी का उपयोग
        करके $g^q \equiv \bigl(\frac{p^*}q\bigr)g
        \pmod{q\Z[\zeta]}$ निष्कर्ष निकालिए, और फिर --- $g^q$ के दोनों
        व्यंजकों को $g$ से गुणा करके तथा $q$ के सापेक्ष
        व्युत्क्रमणीय $g^2 = p^*$ का उपयोग करके ---
        \[
        \Bigl(\frac qp\Bigr) = \Bigl(\frac{p^*}q\Bigr) .
        \]
        निष्कर्ष निकालिए \emph{($q\Z[\zeta]$ के सापेक्ष पूर्णांकों
        $\pm p^*$ के बीच सर्वांगसमता से उनकी समता क्यों निकलती है?
        $\Z$ से प्रतिच्छेदन लीजिए।)}
  \item $\bigl(\frac{p^*}q\bigr) =
        \bigl(\frac{-1}q\bigr)^{(p-1)/2}\bigl(\frac pq\bigr)$ और $\bigl(\frac{-1}q\bigr) = (-1)^{(q-1)/2}$ को खोलकर \emph{द्विघात पारस्परिकता नियम}
        प्राप्त कीजिए:
        \[
        \Bigl(\frac pq\Bigr)\Bigl(\frac qp\Bigr)
        = (-1)^{\frac{p-1}2\cdot\frac{q-1}2} .
        \]
        $17$ के सापेक्ष और $3$ के सापेक्ष वर्ग सूचीबद्ध
        करके इसे $(p, q) = (17, 3)$ पर सत्यापित कीजिए, और इसका उपयोग करके तीन
        पंक्तियों में तय कीजिए कि $x^2 \equiv 219 \pmod{383}$ हल्य है या नहीं ($383$
        अभाज्य है, $219 = 3\cdot73$)।
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- \omterm{def:b3:rings:divisibility}{अखंडनीय} बहुपदों की गणना: $\mathbb F_q[X]$ की
अभाज्य संख्या प्रमेय।} कोई अभाज्य \omterm{def:b3:galois:extension}{घात} $q$ स्थिर कीजिए और मान
लीजिए $N_q(n)$ $\mathbb F_q$ पर \omterm{def:b3:galois:extension}{घात} $n$ के \emph{मोनिक \omterm{def:b3:rings:divisibility}{अखंडनीय}}
बहुपदों की संख्या है; \cref{exo:b3:galois:6} से $X^{q^n} - X$ का गुणनखंडन और सर्वसमिका
$q^n = \sum_{d\mid n}d\,N_q(d)$ स्मरण कीजिए, जिसे अब हम प्रतिलोमित करेंगे, नई दृष्टि से
पढ़ेंगे और काम में लाएँगे।
\begin{enumerate}[resume]
  \item (शब्द) किसी शब्द $w \in \mathbb F_q^n$ को \emph{आद्य} कहिए यदि वह किसी
        कड़ाई से छोटे शब्द $u$ की \omterm{def:b3:galois:extension}{घात} $u^{n/d} =
        u\cdots u$ न हो, और मान
        लीजिए $A(d)$ लंबाई $d$ के आद्य शब्दों की संख्या है।
        दर्शाइए कि लंबाई $n$ का प्रत्येक शब्द किसी लंबाई
        $d \mid n$ के आद्य शब्द की अद्वितीय रूप से \omterm{def:b3:galois:extension}{घात} है, अतः
        $q^n = \sum_{d \mid n}A(d)$; \cref{exo:b3:galois:6} से तुलना करके निष्कर्ष निकालिए कि प्रत्येक
        $d$ के लिए $A(d) = d\,N_q(d)$, और इस संयोग को एक स्पष्ट एकैकी
        आच्छादक संगति से समझाइए: \omterm{def:b3:galois:extension}{घात} $d$ के किसी अवयव $\alpha \in
        \mathbb F_{q^d}$
        की \omterm{thm:b3:galois:finitefields}{फ्रोबेनियस} \omterm{def:b3:groups:action}{कक्षा} $(\alpha, \alpha^q, \dots, \alpha^{q^{d-1}})$ में ठीक $d$ भिन्न अवयव
        होते हैं, और \omterm{def:b3:galois:extension}{घात} $d$ के अवयव \omterm{def:b3:galois:extension}{घात} $d$ के \omterm{def:b3:rings:divisibility}{अखंडनीय}
        बहुपदों के साथ $d$-से-एक संगति में हैं।
  \item \emph{मोबियस प्रतिलोमन सूत्र} सिद्ध कीजिए: यदि सभी $n$
        के लिए $f(n) =
        \sum_{d\mid n}g(d)$ हो, तो $g(n) =
        \sum_{d\mid n}\mu(d)\,f(n/d)$, जहाँ $\mu$ मोबियस फलन है
        ($\mu(m) = (-1)^{\#\text{अभाज्य
        गुणनखंड}}$ यदि $m$ वर्ग-मुक्त हो, अन्यथा $0$)
        \emph{(मुख्य प्रमेयिका: $m >
        1$ के लिए $\sum_{d \mid m}\mu(d) = 0$ --- भाजकों
        को किसी स्थिर अभाज्य गुणनखंड सहित और उसके बिना युग्मित
        कीजिए)}। इससे निकालिए
        \[
        N_q(n) = \frac1n\sum_{d \mid n}\mu(d)\,q^{n/d} .
        \]
  \item दर्शाइए कि प्रत्येक $n \geq 1$ के लिए $N_q(n) \geq \frac1n\bigl(q^n - 2q^{n/2}\bigr) > 0$: यह इस बात की
        नई उपपत्ति है कि सभी $n$ के लिए $\mathbb F_{q^n}$ विद्यमान है।
        अग्र पद की व्याख्या कीजिए: \omterm{def:b3:galois:extension}{घात} $n$ का कोई यादृच्छिक
        मोनिक बहुपद $\sim \frac1n$ प्रायिकता से \omterm{def:b3:rings:divisibility}{अखंडनीय} होता है --- यह
        अभाज्य संख्या प्रमेय का सटीक समरूप है, जिसमें $\log x$ के
        स्थान पर $n$ आ गया है; $q = 2$, $n \leq 4$ के लिए
        संख्यात्मक सत्यापन कीजिए (\cref{exo:b3:galois:6} में गणनाएँ दी हैं)।
  \item प्रश्न 21 का गुणनात्मक सहचर सिद्ध कीजिए:
        \[
        \prod_{\substack{\pi \text{ मोनिक अखंडनीय}\\ \deg\pi =
        n}}\pi \;=\; \prod_{d \mid n}
        \bigl(X^{q^d} - X\bigr)^{\mu(n/d)}
        \]
        \emph{(शून्येतर परिमेय फलनों के आबेली समूह में मोबियस
        प्रतिलोमन)}; $q
        = 2$, $n = 2$ के लिए इसे हाथ से सत्यापित
        कीजिए: $(X^4 - X)/(X^2 - X) = X^2 + X + 1$।
\end{enumerate}

\medskip
\noindent\textbf{भाग VII --- दो समापन।}
\begin{enumerate}[resume]
  \item (द्वितीय अनुपूरक) भाग V की विधि $\bigl(\frac2q\bigr)$ भी परिकलित कर देती
        है। मान लीजिए $\omega = \eu^{2\iu\pi/8}$ और $g = \omega + \omega^{-1}$। दर्शाइए कि $g^2 = 2$
        \emph{($\omega^2 = \iu$)}; फिर किसी विषम अभाज्य $q$ के लिए
        $\Z[\omega]$ में $q$ के सापेक्ष सिद्ध कीजिए कि
        \[
        g^q \equiv \omega^q + \omega^{-q} \pmod{q\Z[\omega]},
        \]
        और यह कि दाईं ओर $q \equiv \pm1
        \pmod 8$ होने पर $g$ और $q \equiv \pm3 \pmod 8$ होने पर
        $-g$ के बराबर है। प्रश्न 18 की तरह $g^q = g\,(g^2)^{(q-1)/2} \equiv
        \bigl(\frac2q\bigr)g$ से तुलना करके
        निष्कर्ष निकालिए
        \[
        \Bigl(\frac2q\Bigr) = (-1)^{(q^2-1)/8},
        \]
        और जाँचिए कि $(q^2 - 1)/8$ ठीक तभी सम है जब $q
        \equiv \pm1 \pmod 8$। सत्यापित
        कीजिए: $2$ $7$ के सापेक्ष और $17$ के सापेक्ष
        वर्ग है ($3^2$ और $6^2$), पर $3$ के सापेक्ष नहीं,
        न $5$ के सापेक्ष।
  \item ($\mathbb F_q[X]$ का ज़ीटा फलन) $t$ में औपचारिक \omterm{def:b3:galois:extension}{घात} श्रेणियों की
        सर्वसमिका सिद्ध कीजिए:
        \[
        \prod_{n \geq 1}\bigl(1 - t^n\bigr)^{-N_q(n)}
        = \frac1{1 - qt}
        \]
        \emph{(मोनिक \omterm{def:b3:rings:divisibility}{अखंडनीय} बहुपदों में अद्वितीय गुणनखंडन:
        प्रत्येक गुणनखंड को गुणोत्तर श्रेणी के रूप में फैलाइए और
        \omterm{def:b3:galois:extension}{घात} $n$ के मोनिक बहुपद गिनिए)}। लघुगणक लेकर \cref{exo:b3:galois:6} की
        सर्वसमिका $q^m = \sum_{d \mid m}d\,N_q(d)$ पुनः प्राप्त कीजिए। $q = 2$ के लिए
        $t^2$ का गुणांक हाथ से जाँचिए, और प्रश्न 21 के सूत्र से
        $N_2(6) = 9$ परिकलित करके $2^6 = 1\cdot2 + 2\cdot1 + 3\cdot2 +
        6\cdot9$ सत्यापित कीजिए।
\end{enumerate}
\end{problem}
