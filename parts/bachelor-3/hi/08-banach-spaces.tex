\chapter{बानाख समष्टियाँ और मूल प्रमेय}
\label{ch:b3:banach}

कार्यकीय विश्लेषण अनंत-विमीय मानकित समष्टियों का अध्ययन उन पर रहने
वाले संकारकों और फलनिकों के माध्यम से करता है। उसकी संस्थापक खोज
यह है कि \emph{\omterm{def:b3:complete:complete}{पूर्णता}}, बेयर की प्रमेय के द्वारा, प्रबल एकरूपता
पर विवश कर देती है: संकारकों के बिंदुशः परिबद्ध कुल मानक में
परिबद्ध होते हैं (बानाख--श्टाइनहाउस), \omterm{def:b3:topology:continuity}{संतत} एकैकी आच्छादक
प्रतिचित्रणों के प्रतिलोम \omterm{def:b3:topology:continuity}{संतत} होते हैं (विवृत प्रतिचित्रण), और
आलेख \omterm{def:b3:topology:continuity}{सांतत्य} पकड़ लेते हैं (संवृत आलेख)। दूसरा स्तंभ,
\emph{हान--बानाख}, \omterm{def:b3:complete:complete}{पूर्णता} की माँग बिलकुल नहीं करता --- केवल
ज़ोर्न की प्रमेयिका की --- और यह सुनिश्चित करता है कि द्वैत
समष्टियाँ इतनी समृद्ध हों कि प्रत्येक सदिश को देख सकें। यह अध्याय
चारों प्रमेय सिद्ध करता है और उन्हें शास्त्रीय अनुक्रम समष्टियों
$\ell^p$ पर, मूर्त द्वैत परिकलनों पर, और एक सचमुच चौंकाने वाले
अनुप्रयोग पर परखता है: ऐसे \omterm{def:b3:topology:continuity}{संतत} $2\pi$-आवर्ती फलन विद्यमान हैं जिनकी
फूरिये श्रेणी किसी बिंदु पर \emph{अपसरित} हो जाती है --- जिससे
दूसरे वर्ष का छूटा हुआ प्रश्न नकारात्मक रूप में हल हो जाता है।

सर्वत्र $E, F$ $K = \R$ या $\C$ पर मानकित समष्टियाँ हैं; \emph{बानाख
समष्टि}\index{बानाख समष्टि} से अभिप्राय है \omterm{def:b3:complete:complete}{पूर्ण} मानकित समष्टि।

\section{परिबद्ध संकारक; अनुक्रम समष्टियाँ}

\begin{definition}\label{def:b3:banach:operator}
$\mathcal L(E, F)$ \emph{परिबद्ध} (= \omterm{def:b3:topology:continuity}{संतत}, दूसरा वर्ष) रैखिक प्रतिचित्रणों की
समष्टि को निरूपित करता है, जिस पर \emph{संकारक मानक}\index{संकारक
मानक} $\vertiii T = \sup_{\norm x \leq
1}\norm{Tx}$ है; वह उपगुणात्मक है: $\vertiii{ST} \leq
\vertiii S\,\vertiii T$। \emph{द्वैत
समष्टि}\index{द्वैत समष्टि} $E' = \mathcal L(E, K)$ है।
\end{definition}

\begin{proposition}\label{prop:b3:banach:llcomplete}
यदि $F$ एक बानाख समष्टि हो, तो $\mathcal L(E, F)$ भी बानाख समष्टि है; विशेष
रूप से $E'$ सदैव बानाख समष्टि है।
\end{proposition}

\begin{proof}
मान लीजिए $(T_n)$ $\vertiii\cdot$ के लिए कोशी है। प्रत्येक $x$ के लिए $\norm{T_nx - T_mx} \leq \vertiii{T_n - T_m}\norm x$:
अतः $(T_nx)$ $F$ में कोशी है और अभिसरित होता है; उसकी सीमा को $Tx$
कहिए। $T$ रैखिक है (रैखिक सर्वसमिकाओं की सीमाएँ); और $\norm{T_nx - T_mx} \leq \varepsilon\norm x$
($n, m \geq N$) में सीमा लेने पर $\norm{T_nx - Tx} \leq \varepsilon\norm x$ मिलता है: अतः \omterm{def:b3:banach:operator}{संकारक मानक} में $T_n \to T$,
और $\vertiii T \leq \vertiii{T_N} +
\varepsilon < \infty$।
\end{proof}

\begin{definition}\label{def:b3:banach:lp}
शास्त्रीय अनुक्रम समष्टियाँ\index{अनुक्रम समष्टियाँ $\ell^p$} ($K$
पर, $\N$ से सूचकित):
\[
\ell^p = \Bigl\{x = (x_n) : \norm x_p = \Bigl(\sum_n
\abs{x_n}^p\Bigr)^{1/p} < \infty\Bigr\}\quad (1 \leq p <
\infty),
\]
\[
\ell^\infty = \{x : \norm x_\infty = \sup\abs{x_n} < \infty\},
\]
और $\norm\cdot_\infty$ के साथ $c_0 = \{x : x_n \to 0\}$। $\norm\cdot_p$ का मानक होना मिन्कोव्स्की असमिका से
निकलता है, जो विविक्त स्थिति में ठीक वैसे ही सिद्ध होती है जैसे
\cref{ch:b3:lp} में (अथवा दूसरे वर्ष की परिमित-विमीय असमिका का योग लेकर)।
ये सभी बानाख समष्टियाँ हैं, और $c_0$ $\ell^\infty$ की एक संवृत उपसमष्टि है
(\cref{exo:b3:banach:3})।
\end{definition}

\begin{proposition}[न्यूमन श्रेणी]\label{prop:b3:banach:neumann}
मान लीजिए $E$ बानाख है और $\vertiii T < 1$ के साथ $T \in \mathcal L(E) = \mathcal L(E,E)$। तब $I - T$ $\mathcal
L(E)$ में
व्युत्क्रमणीय है और $(I - T)^{-1} = \sum_{n\geq0}T^n$ (\omterm{def:b3:banach:operator}{संकारक मानक} में अभिसारी)। फलस्वरूप
व्युत्क्रमणीय संकारकों का समुच्चय विवृत है, और उस पर प्रतिलोमन
\omterm{def:b3:topology:continuity}{संतत} है।\index{न्यूमन श्रेणी}
\end{proposition}

\begin{proof}
श्रेणी बानाख $\mathcal L(E)$ में निरपेक्षतः अभिसरित होती है ($\vertiii{T^n} \leq \vertiii
T^n$,
गुणोत्तर) (\cref{prop:b3:banach:llcomplete}; \cref{exo:b3:complete:1}(ख))। दूरबीनी योग से $(I - T)\sum_{n \leq
N}T^n = I - T^{N+1} \to I$, और इसी प्रकार
दूसरी ओर भी। विवृतता के लिए: यदि $S$ व्युत्क्रमणीय हो और $\vertiii{H} <
1/\vertiii{S^{-1}}$,
तो $\vertiii{S^{-1}H} < 1$ के साथ $S + H = S(I + S^{-1}H)$: अतः व्युत्क्रमणीय। प्रतिलोमन का \omterm{def:b3:topology:continuity}{सांतत्य}:
श्रेणी वाला व्यंजक स्थानीय रूप से $\vertiii{(S+H)^{-1} - S^{-1}} =
O(\vertiii H)$ दे देता है।
\end{proof}

\begin{example}[न्यूमन से हल किया गया एक वोल्तेरा समीकरण]
\label{ex:b3:banach:volterraexample}
$E = \mathcal C(\intcc01)$ पर समाकल समीकरण
\[
u(x) = 1 + \lambda\int_0^xu(t)\,\dd t,
\qquad\text{अर्थात्}\qquad u = \mathbf 1 + \lambda Tu,
\quad (Tu)(x) = \int_0^xu .
\]
पर विचार कीजिए। यहाँ $\vertiii T \leq 1$, अतः $\abs\lambda < 1$ के लिए \omterm{prop:b3:banach:neumann}{न्यूमन श्रेणी} सीधे
लागू होती है: $u = (I - \lambda T)^{-1}\mathbf 1 =
\sum_n\lambda^nT^n\mathbf 1$। परिकलन करने पर $T^n\mathbf 1 =
\frac{x^n}{n!}$, अतः
\[
u(x) = \sum_{n\geq0}\frac{(\lambda x)^n}{n!} =
\eu^{\lambda x} ,
\]
जैसा अवकलन से पुष्ट हो जाता है। इससे भी अच्छी बात: $\vertiii{T^n} \leq
\frac1{n!}$
(पुनरावृत्त नाभिक क्रमगुणित की दर से सिकुड़ता है), अतः $\sum\lambda^nT^n$
\emph{प्रत्येक} $\lambda$ के लिए अभिसरित होती है --- संकारक $I - \lambda T$
समस्त $\lambda
\in \C$ के लिए व्युत्क्रमणीय है, यद्यपि अंततः $\vertiii{\lambda T} \geq 1$: महत्त्व
घातों के स्पेक्ट्रमी क्षय का है, पहले मानक का नहीं। वोल्तेरा
संकारकों में यह क्रमगुणित क्षय अंतर्निहित होता है (\cref{exo:b3:complete:4}(क) ने ठीक
इसी का उपयोग किया था), और इसीलिए आरंभिक मान समस्याएँ सीमांत मान
समस्याओं की अनुनाद परिघटनाओं से कभी ग्रस्त नहीं होतीं (\cref{ch:b3:spectral})।
\end{example}

\section{हान--बानाख}

\begin{theorem}[हान--बानाख, वैश्लेषिक रूप]
\label{thm:b3:banach:hahnbanach}
मान लीजिए $E$ एक \emph{वास्तविक} सदिश समष्टि है, $p \colon E \to \R$
\emph{उपरैखिक} है ($p(x + y) \leq p(x) + p(y)$ और $t \geq 0$ के लिए $p(tx) =
tp(x)$), $F \subseteq E$ एक उपसमष्टि
है और $f
\colon F \to \R$ ऐसा रैखिक फलनिक जिसके लिए $F$ पर $f \leq p$। तब $f$ किसी
ऐसे रैखिक $\tilde f \colon E \to \R$ तक विस्तारित हो जाता है जिसके लिए $E$ पर
$\tilde f \leq p$।\index{हान--बानाख प्रमेय}
\end{theorem}

\begin{proof}
\emph{एक-चरण विस्तार।} मान लीजिए $x_0 \notin F$; हम $\alpha = \tilde f(x_0)$ को ठीक से चुनकर
$f$ को $F \oplus \R x_0$ तक विस्तारित करते हैं: हमें सभी $y \in F$, $t > 0$ के
लिए चाहिए
\[
f(y) + t\alpha \leq p(y + tx_0)
\quad\text{तथा}\quad
f(y) - t\alpha \leq p(y - tx_0),
\]
जो $t$ से भाग देने पर (उपरैखिकता) घटकर
\[
\sup_{v \in F}\ \bigl[f(v) - p(v - x_0)\bigr]
\;\leq\; \alpha \;\leq\;
\inf_{u \in F}\ \bigl[p(u + x_0) - f(u)\bigr].
\]
हो जाता है। ऐसा $\alpha$ विद्यमान है तभी और केवल तभी जब प्रत्येक
बायाँ पद प्रत्येक दाएँ पद से $\leq$ हो: और वस्तुतः $f(v) + f(u) = f(u + v) \leq p(u + v) \leq
p(u + x_0) + p(v - x_0)$, अर्थात्
$f(v) - p(v - x_0) \leq p(u +
x_0) - f(u)$।

\emph{ज़ोर्न।} $p$ से प्रभावित $f$ के विस्तारों (युग्म:
उपसमष्टि, फलनिक) को विस्तार से क्रमित कीजिए; किसी शृंखला का ऊपरी
परिबंध उसका सम्मिलन है; और कोई महत्तम अवयव पूरे $E$ पर
परिभाषित होना ही चाहिए, अन्यथा एक-चरण विस्तार महत्तमता का खंडन कर
देता है।
\end{proof}

\begin{corollary}\label{cor:b3:banach:hbnormed}
मान लीजिए $E$ एक मानकित समष्टि है ($K = \R$ या $\C$)।
\begin{enumerate}
  \item प्रत्येक $f \in F'$ (जहाँ $F$ एक उपसमष्टि है) $\norm{\tilde f}_{E'} = \norm f_{F'}$ वाले
        किसी $\tilde f
        \in E'$ तक विस्तारित हो जाता है।
  \item प्रत्येक $x \neq 0$ के लिए $\norm f
        = 1$ और $f(x) = \norm x$ वाला कोई $f \in E'$ है।
        विशेष रूप से $E'$ $E$ के बिंदुओं को पृथक करता है, और
        $\norm x = \sup_{\norm f \leq
        1}\abs{f(x)}$।
  \item किसी संवृत उपसमष्टि $F$ और $x \notin F$ के लिए ऐसा $f
        \in E'$ है
        जो $F$ पर लुप्त हो जाता है और जिसके लिए $f(x) = d(x, F)$ तथा
        $\norm f \leq 1$।
\end{enumerate}
\end{corollary}

\begin{proof}
(1) वास्तविक स्थिति: $p(x)
= \norm f_{F'}\,\norm x$ (जो उपरैखिक है) के साथ \cref{thm:b3:banach:hahnbanach} लगाइए;
विस्तार $\pm\tilde f(x) = \tilde f(\pm x) \leq p(x)$ को संतुष्ट करता है, अतः $\norm{\tilde f} \leq \norm f$, और $\geq$ प्रतिबंध
है। सम्मिश्र स्थिति: मान लीजिए $u = \operatorname{Re}f$, जो $\abs u \leq \norm f\norm\cdot$ वाला वास्तविक
फलनिक है; ध्यान दीजिए कि $f(x) = u(x) -
\iu\,u(\iu x)$ (वास्तविक और काल्पनिक भागों पर
जाँचिए: $\operatorname{Im}f(x) = -\operatorname{Re}f(\iu x)$)। $u$ को उसी परिबंध के साथ वास्तविक-रैखिक ढंग से
विस्तारित कीजिए और $\tilde f(x) =
\tilde u(x) - \iu\tilde u(\iu x)$ रखिए: यह $\C$-रैखिक है ($\iu$ से गुणन
पर सीधी जाँच) और $f$ का विस्तार है; मानक के लिए: दिए हुए $x$
के लिए $\tilde f(x) = r\eu^{\iu\theta}$ लिखिए, तब $\abs{\tilde f(x)}
= \tilde f(\eu^{-\iu\theta}x) = \tilde u(\eu^{-\iu\theta}x)
\leq \norm f\,\norm x$।

(2) $F = Kx$ पर $f(tx) = t\norm x$ परिभाषित कीजिए: $F$ पर मानक $1$; अब (1)
से विस्तारित कीजिए। द्वैत सूत्र: $\leq$ स्पष्ट है, और इस $f$ से
$\geq$।

(3) $F \oplus Kx$ पर $f(y + tx) = t\,d(x, F)$ परिभाषित कीजिए; तब $t \neq 0$ के लिए $\norm{y + tx} = \abs t\,\norm{x + y/t} \geq \abs
t\,d(x, F) = \abs{f(y + tx)}$: अतः
उपसमष्टि पर $\norm f \leq 1$; अब (1) से विस्तारित कीजिए।
\end{proof}

\begin{remark}\label{rem:b3:banach:bidual}
(2) से विहित प्रतिचित्रण $J \colon E \to E''$, $J(x)(f) =
f(x)$, एक समदूरीयता है (\cref{exo:b3:banach:10}):
अर्थात् प्रत्येक मानकित समष्टि अपने द्विद्वैत के भीतर बैठती है।
जिन समष्टियों के लिए $J$ आच्छादक है वे \emph{स्वद्वैती}\index{स्वद्वैती
समष्टि} कहलाती हैं; सप्ताहांत समस्या दिखाती है कि $\ell^p$ ($1 < p < \infty$)
\omterm{rem:b3:banach:bidual}{स्वद्वैती} है जबकि $\ell^1$ नहीं।
\end{remark}

\section{बेयर त्रयी}

\begin{theorem}[बानाख--श्टाइनहाउस, एकसमान परिबद्धता]
\label{thm:b3:banach:banachsteinhaus}
मान लीजिए $E$ एक \emph{बानाख} समष्टि है, $F$ मानकित, और
$(T_i)_{i\in
I} \subseteq \mathcal L(E, F)$ ऐसा कुल कि प्रत्येक $x \in E$ के लिए $\sup_i
\norm{T_ix} < \infty$। तब
$\sup_i
\vertiii{T_i} < \infty$।\index{बानाख--श्टाइनहाउस प्रमेय}
\end{theorem}

\begin{proof}
समुच्चय $F_n = \{x : \sup_i\norm{T_ix} \leq n\}$ संवृत हैं (संवृत गोलों के पूर्वप्रतिबिंबों के
प्रतिच्छेदन) और $E$ को आच्छादित करते हैं। बेयर (\cref{thm:b3:complete:baire}) $n_0$
और कोई गोला $B(x_0, r) \subseteq F_{n_0}$ दे देती है। $\norm z < r$ के लिए: $\norm{T_iz}
\leq \norm{T_i(x_0 + z)} + \norm{T_ix_0} \leq 2n_0$, अतः प्रत्येक
$i$ के लिए $\vertiii{T_i} \leq 2n_0/r$।
\end{proof}

\begin{corollary}\label{cor:b3:banach:bslimits}
यदि $E$ बानाख हो और $T_n \in \mathcal L(E,F)$ \emph{बिंदुशः} अभिसरित हों (प्रत्येक
$x$ के लिए $T_nx \to Tx$), तो $\sup_n
\vertiii{T_n} < \infty$, $T \in \mathcal L(E, F)$, और $\vertiii T \leq \liminf \vertiii{T_n}$।
\end{corollary}

\begin{proof}
अभिसारी अनुक्रम परिबद्ध होते हैं: अतः बिंदुशः परिबद्धता;
बानाख--श्टाइनहाउस मानकों को किसी $M$ से परिबद्ध कर देती है; तब
$\norm{Tx}
= \lim\norm{T_nx} \leq M\norm x$ ($T$ बिंदुशः सीमा के रूप में रैखिक है), और $\norm{T_nx} \leq \vertiii{T_n}\norm x$ में
$\liminf$ तक जाने से तीक्ष्णतर परिबंध मिल जाता है।
\end{proof}

\begin{theorem}[अपसारी फूरिये श्रेणी]
\label{thm:b3:banach:fourierdiverge}
ऐसे \omterm{def:b3:topology:continuity}{संतत} $2\pi$-आवर्ती फलन $f$ विद्यमान हैं जिनकी फूरिये श्रेणी
$0$ पर अपसरित होती है: $\sup_N\abs{S_N(f)(0)} =
\infty$। वस्तुतः ऐसे $f$ $\mathcal
C(S^1)$ का एक
सघन उपसमुच्चय बनाते हैं।\index{फूरिये श्रेणी, अपसरण}
\end{theorem}

\begin{proof}
बानाख समष्टि $E = (\mathcal C(S^1), \norm\cdot_\infty)$ में फलनिकों $\Lambda_N(f) = S_N(f)(0) =
\frac1{2\pi}\int_{-\pi}^{\pi}f(t)\,D_N(t)\,\dd t$ (दिरिक्ले नाभिक, दूसरा वर्ष)
के साथ कार्य कीजिए। प्रत्येक $\Lambda_N$ \omterm{def:b3:topology:continuity}{संतत} है और
\[
\norm{\Lambda_N} = \frac1{2\pi}\int_{-\pi}^\pi\abs{D_N(t)}\,\dd
t \;=\; L_N .
\]
($\leq$ स्पष्ट है; $\geq$ के लिए: $\norm f_\infty
\leq 1$ वाला \omterm{def:b3:topology:continuity}{संतत} $f$ लीजिए जो
$\operatorname{sign}D_N$ का आसन्नन करे --- चिह्न में परिमित संख्या के उछाल हैं;
प्रत्येक उछाल को लंबाई $\varepsilon$ के अंतराल पर चिकना करने से समाकल
$O(N\varepsilon)$ जितना बदलता है।)
\emph{लेबेग नियतांक} $L_N$ अनंत की ओर जाते हैं:
\[
L_N = \frac1{2\pi}\int_{-\pi}^{\pi}
\frac{\abs{\sin\bigl((N{+}\tfrac12)t\bigr)}}{\abs{\sin(t/2)}}
\,\dd t
\geq \frac{2}{\pi}\int_0^{\pi}
\frac{\abs{\sin\bigl((N{+}\tfrac12)t\bigr)}}{t}\,\dd t
= \frac{2}{\pi}\int_0^{(N+\frac12)\pi}\frac{\abs{\sin
u}}{u}\,\dd u
\]
($[0, \pi]$ पर $\abs{\sin(t/2)} \leq t/2$ का उपयोग करके, फिर $u = (N + \tfrac12)t$ प्रतिस्थापित करके)।
मेहराबों में काटने पर:
\[
\int_{(k-1)\pi}^{k\pi}\frac{\abs{\sin u}}u\,\dd u
\geq \frac1{k\pi}\int_{(k-1)\pi}^{k\pi}\abs{\sin u}\,\dd u =
\frac{2}{k\pi},
\qquad\text{अतः}\qquad
L_N \geq \frac{4}{\pi^2}\sum_{k=1}^N\frac1k
\xrightarrow[N\to\infty]{} \infty .
\]
यदि प्रत्येक \omterm{def:b3:topology:continuity}{संतत} $f$ के लिए $\sup_N
\abs{\Lambda_N(f)} < \infty$ होता, तो
बानाख--श्टाइनहाउस $\sup_N\norm{\Lambda_N} < \infty$ पर विवश कर देती: विरोधाभास। अतः कोई
$f$ --- वस्तुतः $f$ का एक अक्षीण, \omterm{def:b3:topology:interior}{सघन समुच्चय} ($\bigcup_M\{f: \sup_N\abs{\Lambda_Nf}\leq M\}$ का
पूरक, जो संवृत समुच्चयों का गणनीय सम्मिलन है और जिसका, उपर्युक्त
तर्क किसी भी गोले में लगाने से, कोई \omterm{def:b3:topology:interior}{अंतःभाग} न होने के कारण वह
\omterm{rem:b3:complete:meagre}{क्षीण} है) --- $\sup_N\abs{S_Nf(0)} =
\infty$ रखता है।
\end{proof}

\begin{theorem}[विवृत प्रतिचित्रण]\label{thm:b3:banach:openmapping}
मान लीजिए $E, F$ बानाख समष्टियाँ हैं और $T \in \mathcal L(E, F)$ \emph{आच्छादक} है।
तब $T$ विवृत है: किसी $c > 0$ के लिए $T(B_E(0,1)) \supseteq
B_F(0, c)$। फलस्वरूप बानाख
समष्टियों के बीच किसी एकैकी आच्छादक परिबद्ध संकारक का प्रतिलोम
परिबद्ध होता है।\index{विवृत प्रतिचित्रण प्रमेय}
\end{theorem}

\begin{proof}
$B = B_E(0,1)$ लिखिए। आच्छादकता से $F = \bigcup_n
\overline{T(nB)} = \bigcup_n n\,\overline{T(B)}$; और बेयर (\cref{thm:b3:complete:baire}) $\overline{T(B)}$ को
\omterm{def:b3:topology:interior}{अंतःभाग} दे देती है: किसी $B_F(y_0, 4c) \subseteq
\overline{T(B)}$ के लिए। $0$ पर पुनःकेंद्रित
कीजिए: $\norm y < 4c$ के लिए $y_0 + y$ और $y_0$ दोनों $u_k, v_k \in B$ वाले प्रतिबिंबों
$Tu_k$, $Tv_k$ की सीमाएँ हैं, अतः $u_k - v_k
\in 2B$ वाला $y = \lim T(u_k - v_k)$: अर्थात् $B_F(0, 4c) \subseteq \overline{T(2B)}$,
यानी $B_F(0, 2c) \subseteq \overline{T(B)}$।

\emph{\omterm{def:b3:topology:interior}{संवृति} हटाना} (यहीं $E$ की \omterm{def:b3:complete:complete}{पूर्णता} प्रवेश करती है): मान
लीजिए $\norm y < c$। $\norm{y -
Tx_1} < c/2$ वाला $x_1 \in \frac12 B$ चुनिए ($\frac12$ से मापित $B_F(0,2c) \subseteq \overline{T(B)}$);
आगमन से $\norm{y -
T(x_1 + \dots + x_k)} < c\,2^{-k}$ वाला $x_k \in 2^{-k}B$। श्रेणी $\sum x_k$ बानाख $E$ में निरपेक्षतः
अभिसरित होकर $x \in B$ (मानक $<
\sum 2^{-k} = 1$) पर जाती है, और \omterm{def:b3:topology:continuity}{सांतत्य} से $Tx = y$:
अतः $B_F(0, c)
\subseteq T(B)$। मनमाने विवृत समुच्चयों पर $T$ की विवृतता स्थानांतरण
और मापन से निकलती है; और उपप्रमेय के लिए, $T$ की विवृतता का अर्थ
है कि $T^{-1}$ \omterm{def:b3:topology:continuity}{संतत} है।
\end{proof}

\begin{corollary}[तुल्य मानक]\label{cor:b3:banach:equivnorms}
यदि कोई सदिश समष्टि दो तुलनीय मानकों ($\norm\cdot_a \leq C\norm\cdot_b$) के लिए \omterm{def:b3:complete:complete}{पूर्ण} हो, तो
वे मानक तुल्य होते हैं।
\end{corollary}

\begin{proof}
तत्समक $(E, \norm\cdot_b) \to (E, \norm\cdot_a)$ बानाख समष्टियों के बीच परिबद्ध और एकैकी आच्छादक है:
अतः उसका प्रतिलोम परिबद्ध है।
\end{proof}

\begin{theorem}[संवृत आलेख]\label{thm:b3:banach:closedgraph}
मान लीजिए $E, F$ बानाख हैं और $T \colon E \to F$ रैखिक। यदि आलेख $\Gamma = \{(x, Tx)\}$ $E \times F$
में संवृत हो (अर्थात् $x_n \to x$ और $Tx_n \to y$ से $y = Tx$ निकलता हो), तो
$T$ परिबद्ध है।\index{संवृत आलेख प्रमेय}
\end{theorem}

\begin{proof}
$\norm{(x,y)} = \norm x + \norm y$ के साथ $E \times F$ बानाख है; और संवृत उपसमष्टि $\Gamma$ भी बानाख है।
प्रक्षेप $\pi_E\colon \Gamma \to E$ परिबद्ध और एकैकी आच्छादक है, अतः उसका प्रतिलोम
$x \mapsto (x, Tx)$ परिबद्ध है (\cref{thm:b3:banach:openmapping}): इसलिए $\norm{Tx} \leq \norm{(x,
Tx)} \leq C\norm x$।
\end{proof}

\begin{method}\label{met:b3:banach:trilogy}
कौन-सी प्रमेय कब उठानी है। \emph{हान--बानाख}: निर्धारित व्यवहार
वाला कोई फलनिक बनाने के लिए (किसी सदिश का मानक निकालना, किसी
उपसमष्टि पर लुप्त होना, किसी उपसमष्टि से विस्तार करना) --- यहाँ
\omterm{def:b3:complete:complete}{पूर्णता} की आवश्यकता नहीं। \emph{बानाख--श्टाइनहाउस}: बिंदुशः सूचना
को एकसमान परिबंधों में बदलने के लिए --- प्रायः यह दिखाने के लिए
कि कोई सीमा संक्रिया \omterm{def:b3:topology:continuity}{संतत} है, अथवा (प्रतिधनात्मक रूप में) किसी
अवयव के लिए \emph{अपसरण} सिद्ध करने के लिए, जैसा फूरिये श्रेणी के
लिए। \emph{विवृत प्रतिचित्रण / संवृत आलेख}: बीजगणितीय एकैकी
आच्छादकता से अथवा आलेख के किसी \omterm{def:b3:topology:interior}{संवृति} गुण से मुफ्त में \omterm{def:b3:topology:continuity}{सांतत्य}
पाने के लिए --- प्ररूपी उपयोग: दो \omterm{def:b3:complete:complete}{पूर्ण} मानकों की तुलना, या
स्वतः \omterm{def:b3:topology:continuity}{सांतत्य} सिद्ध करना। तीनों बेयर प्रमेयों को \emph{स्रोत की}
\omterm{def:b3:complete:complete}{पूर्णता} चाहिए; अन्यथा प्रतिउदाहरण हैं (\cref{exo:b3:banach:7})।
\end{method}

\section{द्वैत समष्टियाँ, मूर्त रूप में}

\begin{theorem}\label{thm:b3:banach:duals}
समदूरिक रूप से: $(c_0)' \cong \ell^1$ और $(\ell^1)' \cong
\ell^\infty$, युग्मन $\langle x, y\rangle = \sum_n
x_ny_n$ के
माध्यम से।\index{अनुक्रम समष्टियों के द्वैत}
\end{theorem}

\begin{proof}
हम $(c_0)' \cong \ell^1$ सिद्ध करते हैं; दूसरी पहचान \cref{exo:b3:banach:5} है। $y \in \ell^1$ के साथ
$\Lambda_y(x)
= \sum x_ny_n$ ($x \in c_0$) जोड़िए: यह निरपेक्षतः अभिसारी है और $\abs{\Lambda_y(x)} \leq \norm x_\infty\norm y_1$, अतः
$\norm{\Lambda_y} \leq \norm y_1$। विलोमतः मान लीजिए $\Lambda \in
(c_0)'$; $y_n = \Lambda(e_n)$ रखिए ($e_n$ इकाई अनुक्रम
हैं)। किसी भी $N$ के लिए $x^{(N)} = \sum_{n \leq N}
\operatorname{sign}(\overline{y_n})\,e_n \in c_0$ की परीक्षा लीजिए (मानक $\leq
1$;
सम्मिश्र स्थिति में इकाई-मापांक गुणनखंड $\bar y_n/\abs
{y_n}$ का उपयोग कीजिए):
$\Lambda(x^{(N)}) = \sum_{n\leq N}\abs{y_n} \leq
\norm\Lambda$। अतः $\norm y_1 \leq
\norm\Lambda$ के साथ $y \in \ell^1$। अंततः $\Lambda = \Lambda_y$: दोनों $e_n$ पर सहमत
हैं, अतः परिमित अनुक्रमों पर भी, जो $c_0$ में सघन हैं (छँटाई:
$\norm{x - \sum_{n \leq N}x_ne_n}_\infty = \sup_{n > N}\abs{x_n}
\to 0$ ठीक इसलिए कि $x_n \to 0$); और \omterm{def:b3:topology:interior}{सघन समुच्चय} पर सहमत \omterm{def:b3:topology:continuity}{संतत} फलनिक
बराबर होते हैं। यह संगति रैखिक, एकैकी आच्छादक और समदूरिक है
(दोनों असमिकाओं से $\norm{\Lambda_y} = \norm y_1$)।
\end{proof}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:5}
\cref{thm:b3:banach:duals} की योजना का अनुसरण करते हुए समदूरिक रूप से $(\ell^1)' \cong \ell^\infty$ सिद्ध
कीजिए (परिमित अनुक्रम $\ell^1$ में सघन हैं)। $(\ell^\infty)'$ के लिए तर्क कहाँ
टूट जाता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:6}
मान लीजिए $E, F, G$ मानकित हैं और $E$ बानाख, तथा $B \colon E \times

लगाइए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:7}
(क) $E = \mathcal C(\intcc01)$ पर $\norm\cdot_\infty$ और $\norm\cdot_1$ की तुलना कीजिए: तत्समक $(E, \norm\cdot_\infty) \to

और एकैकी आच्छादक है पर उसका प्रतिलोम अपरिबद्ध। \cref{cor:b3:banach:equivnorms} की कौन-सी

(ख) $\ell^2$ की किसी सघन उपसमष्टि से $K$ में जाता कोई असंतत रैखिक

\begin{exercise}[$\star\star\star$]\label{exo:b3:banach:8}

परिभाषित नहीं किए जा सकते।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:banach:9}
(संख्यात्मक समाकलन पर पोल्या की प्रमेय) प्रत्येक $n$ के लिए मान
लीजिए $\Lambda_n(f)
= \sum_{i=0}^{n} w_{i,n}f(x_{i,n})$ $\mathcal C(\intcc01)$ पर एक संख्यात्मक समाकलन नियम है ($x_{i,n} \in \intcc01$, $w_{i,n} \in

दर्शाइए कि \emph{प्रत्येक} \omterm{def:b3:topology:continuity}{संतत} $f$ के लिए $\Lambda_n(f) \to \int_0^1f$ होता है तभी
और केवल तभी जब: (क) प्रत्येक बहुपद $P$ के लिए $\Lambda_n(P) \to \int_0^1P$, और (ख)

\infty$। \emph{($\norm{\Lambda_n}$ परिकलित कीजिए; बानाख--श्टाइनहाउस और

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:10}
दर्शाइए कि $J \colon E \to E''$, $J(x)(f) = f(x)$, एक रैखिक समदूरीयता है (\cref{cor:b3:banach:hbnormed}(2) का उपयोग

के किसी सघन अनुक्रम का लगभग मानक निकालने वाले $x_n$ चुनिए और
\cref{cor:b3:banach:hbnormed}(3) के माध्यम से दिखाइए कि उनका संवृत विस्तार $E$ है।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:11}
(भागफल समष्टियाँ) मान लीजिए $E$ एक बानाख समष्टि है और $F \subseteq

\[
\norm{\bar x} \;=\; d(x, F) = \inf_{y\in F}\norm{x - y} .
\]
(क) दर्शाइए कि यह $E/F$ पर एक सुपरिभाषित \emph{मानक} है ($F$ की
संवृतता कहाँ प्रवेश करती है?), और यह कि प्रक्षेप $\pi
\colon E \to E/F$ $\vertiii\pi \leq 1$
रखता है और विवृत इकाई गोले को विवृत इकाई गोले पर आच्छादित कर देता
है।
(ख) दर्शाइए कि $E/F$ \omterm{def:b3:complete:complete}{पूर्ण} है। \emph{(\cref{exo:b3:complete:1}(ख) की श्रेणी कसौटी का
उपयोग कीजिए: $\sum\norm{\bar x_k} < \infty$ वाले वर्ग $\bar x_k$ दिए हों, तो प्रत्येक को $\norm{x_k} \leq \norm{\bar x_k} + 2^{-k}$
वाले $x_k \in E$ तक उठाइए और $E$ में योग लीजिए।)}
(ग) परिकलित कीजिए: $E = c$ (अभिसारी अनुक्रम) और $F =
c_0$ के लिए
दर्शाइए कि $\bar x \mapsto
\lim_nx_n$ के माध्यम से समदूरिक रूप से $c/c_0 \cong K$।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:banach:12}
(परिबद्ध प्रक्षेप और पूरकित उपसमष्टियाँ) मान लीजिए $E$ एक बानाख
समष्टि है और $P \colon E \to E$ $P^2 = P$ वाला रैखिक प्रतिचित्रण (एक बीजगणितीय
प्रक्षेप), $V = \operatorname{im}P$, $W =
\ker P$।
(क) मान लीजिए $P$ परिबद्ध है। दर्शाइए कि $V$ और $W$ संवृत
हैं और विघटन $x = Px + (x -
Px)$ के साथ $E = V \oplus W$।
(ख) विलोमतः, मान लीजिए $E = V \oplus W$ है जहाँ $V,
W$ \emph{दोनों} संवृत
हैं, और मान लीजिए $P$ $W$ के अनुदिश $V$ पर प्रक्षेप है।
दर्शाइए कि $P$ परिबद्ध है। \emph{(संवृत आलेख: यदि $x_n \to x$ और
$Px_n \to z$ हो, तो $z \in V$, $x_n - Px_n \to x - z \in
W$, और विघटन की अद्वितीयता $z =
Px$ की पहचान
कर देती है।)}
(ग) \emph{तुल्यता} निष्कर्ष के रूप में निकालिए: कोई उपसमष्टि
$V$ परिबद्ध प्रक्षेप स्वीकार करती है तभी और केवल तभी जब वह
संवृत हो और उसका कोई संवृत बीजगणितीय पूरक हो --- और (बिना
उपपत्ति के) ध्यान दीजिए कि इस गुण के बिना संवृत उपसमष्टियाँ
विद्यमान हैं ($\ell^\infty$ के भीतर $c_0$ शास्त्रीय उदाहरण है): हिल्बर्ट
समष्टियाँ, जहाँ $V^\perp$ सदैव काम कर जाता है (\cref{ch:b3:hilbert}), अपवाद हैं,
नियम नहीं।
\end{exercise}

\section{समस्या: \texorpdfstring{$\ell^p$}%
{lp} समष्टियों की द्वैतता}

\begin{problem}[{सप्ताहांत समस्या --- $(\ell^p)' = \ell^q$, स्वद्वैतिता, और
$\ell^\infty$ की विचित्रता}]
\label{pb:b3:banach:1}
$1 < p < \infty$ स्थिर कीजिए और मान लीजिए $q$ संयुग्मी घातांक है, $\frac1p + \frac1q = 1$।
सर्वत्र युग्मन $\langle x,
y\rangle = \sum_n x_ny_n$ है।

\medskip
\noindent\textbf{भाग I --- अनुक्रमों के लिए होल्डर और
मिन्कोव्स्की।}
\begin{enumerate}
  \item (यंग की असमिका) $a, b \geq 0$ के लिए $\log$ की अवतलता का उपयोग
        करके अथवा $t \mapsto \frac{t^p}p + \frac1q - t$ का अध्ययन करके $ab \leq
        \frac{a^p}p + \frac{b^q}q$ दर्शाइए।
  \item (होल्डर) निष्कर्ष निकालिए: $x \in \ell^p$, $y \in \ell^q$ के लिए $\abs{\langle x, y\rangle} \leq
        \norm x_p\norm y_q$;
        और समता स्थिति पहचानिए।
  \item (मिन्कोव्स्की) $\norm\cdot_p$ के लिए त्रिभुज असमिका निष्कर्ष के
        रूप में निकालिए। \emph{($\abs{x_n + y_n}^p \leq
        \abs{x_n}\,\abs{x_n{+}y_n}^{p-1} +
        \abs{y_n}\,\abs{x_n{+}y_n}^{p-1}$ लिखिए और प्रत्येक पद पर
        होल्डर लगाइए।)}
  \item सिद्ध कीजिए कि $\ell^p$ \omterm{def:b3:complete:complete}{पूर्ण} है और परिमित अनुक्रम उसमें
        सघन हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- द्वैतता $(\ell^p)' =
\ell^q$।}
\begin{enumerate}[resume]
  \item $y \in \ell^q$ के लिए दर्शाइए कि $\Lambda_y(x) = \langle
        x, y\rangle$ $\norm{\Lambda_y} \leq \norm y_q$ वाला $\Lambda_y \in (\ell^p)'$ परिभाषित
        करता है, और
        $x_n = \abs{y_n}^{q-1}\operatorname{sign}(y_n)$
        पर (उपयुक्त रूप से छाँटकर और मानकीकृत करके) परीक्षा लेकर
        यह भी कि $\norm{\Lambda_y} = \norm y_q$।
  \item विलोमतः, $\Lambda \in (\ell^p)'$ दिया हो तो $y_n =
        \Lambda(e_n)$ रखिए; दर्शाइए कि $\norm y_q \leq
        \norm\Lambda$
        के साथ $y \in \ell^q$ \emph{(प्रश्न 5 की तरह छँटाइयों पर परीक्षा
        लीजिए और छँटाई की लंबाई बढ़ने दीजिए)}, और निष्कर्ष
        निकालिए $\Lambda = \Lambda_y$: अर्थात् प्रतिचित्रण $y \mapsto \Lambda_y$ एक समदूरिक
        तुल्याकारिता $\ell^q \to (\ell^p)'$ है।
  \item निष्कर्ष निकालिए कि $1 < p <
        \infty$ के लिए $\ell^p$
        \emph{\omterm{rem:b3:banach:bidual}{स्वद्वैती}} है: दोनों द्वैतताओं का संयोजन करने पर
        $(\ell^p)''$ का प्रत्येक अवयव $\ell^p$ से आता है; ध्यान से
        सत्यापित कीजिए कि यह संयोजन विहित $J$ ही है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- $\ell^1$ और $\ell^\infty$ भिन्न प्राणी हैं।}
\begin{enumerate}[resume]
  \item दर्शाइए कि $\ell^p$ ($1 \leq p < \infty$) और $c_0$ पृथक्करणीय हैं, पर
        $\ell^\infty$ नहीं। \emph{($\N$ के उपसमुच्चयों के अगणनीय संख्या
        के सूचक अनुक्रम जोड़े-जोड़े में दूरी $1$ पर हैं।)}
  \item \cref{exo:b3:banach:10} से निष्कर्ष निकालिए कि $(\ell^1)'
        \cong \ell^\infty$ पर $(\ell^\infty)' \not\cong
        \ell^1$: अर्थात्
        $\ell^1$ \omterm{rem:b3:banach:bidual}{स्वद्वैती} \emph{नहीं} है। \emph{(यदि $(\ell^\infty)'$ $\ell^1$
        होता, तो वह पृथक्करणीय होता, जिससे $\ell^\infty$ का पृथक्करणीय
        होना अनिवार्य हो जाता।)}
  \item (एक बानाख सीमा, स्पष्ट रूप से) $\ell^\infty_\R$ पर मान लीजिए $p(x) = \limsup_n \frac{x_1 + \dots + x_n}{n}$।
        दर्शाइए कि $p$ उपरैखिक है, और यह कि अभिसारी अनुक्रमों
        की उपसमष्टि $c$ पर $\mathrm{LIM}(x) = \lim x$ $\mathrm{LIM} \leq p$ को संतुष्ट करता है।
        हान--बानाख से $\mathrm{LIM} \colon \ell^\infty_\R \to \R$ तक विस्तारित कीजिए और दर्शाइए:
        $\mathrm{LIM}$ धनात्मक है ($x \geq 0 \Rightarrow
        \mathrm{LIM}(x) \geq 0$), विस्थापन-अपरिवर्ती है
        ($\mathrm{LIM}(x_2, x_3, \dots) = \mathrm{LIM}(x)$), सीमा का विस्तार है, और $\liminf x \leq
        \mathrm{LIM}(x)\leq \limsup x$ को संतुष्ट करता है।
  \item दर्शाइए कि $(\ell^\infty)'$ में देखा गया ऐसा $\mathrm{LIM}$ किसी भी $y \in \ell^1$ के
        लिए $\Lambda_y$ के रूप का \emph{नहीं} है; इससे पुनः $(\ell^\infty)'
        \neq \ell^1$
        निष्कर्ष निकालिए। \emph{(पहले इकाई अनुक्रमों $e_n$ पर मान
        लीजिए, फिर अचर अनुक्रम $1$ पर।)}
  \item $(0,1,0,1,\dots)$ पर $\mathrm{LIM}$ का मान निकालिए, और दर्शाइए कि सीमा का कोई
        विस्थापन-अपरिवर्ती \emph{गुणनात्मक} विस्तार विद्यमान नहीं
        हो सकता \emph{($x =
        (0,1,0,1,\dots)$ और $x\cdot Sx$ पर विचार कीजिए, जहाँ $S$
        विस्थापन है)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- उपसंहार: स्वद्वैतिता क्यों
महत्त्वपूर्ण है।}
\begin{enumerate}[resume]
  \item \cref{cor:b3:banach:bslimits} और प्रश्न 6 का उपयोग करके दर्शाइए कि $\ell^p$ ($1 < p <
        \infty$)
        के प्रत्येक परिबद्ध अनुक्रम का कोई ऐसा उपानुक्रम $(x^{(k)})$ है
        जो \emph{दुर्बल} रूप से अभिसरित होता है: अर्थात् प्रत्येक
        $\Lambda \in (\ell^p)'$ के लिए $\Lambda(x^{(k)})$ अभिसरित होता है। \emph{(गणनीय संख्या
        के निर्देशांकों पर विकर्ण निष्कर्षण; मानकों की एकसमान
        परिबद्धता और होल्डर का उपयोग करके $\ell^p$ में दुर्बल सीमा
        पहचानिए।)} उदाहरण से दर्शाइए ($\ell^1$ में $e_n$, $\ell^\infty$ के
        सुचुने हुए अवयवों के सापेक्ष) कि $\ell^1$ में यह विफल हो
        जाता है: दुर्बल संहतता \omterm{rem:b3:banach:bidual}{स्वद्वैती} समष्टियों का विशेषाधिकार
        है।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- दुर्बल \omterm{def:b3:topology:topology}{संस्थिति} काम पर, और शूर का
आश्चर्य।} किसी मानकित समष्टि $E$ में $x^{(k)} \rightharpoonup x$ (\emph{दुर्बल
अभिसरण}) तब लिखिए जब प्रत्येक $\Lambda \in E'$ के लिए $\Lambda(x^{(k)}) \to \Lambda(x)$ हो।
\begin{enumerate}[resume]
  \item गणना पूरी कीजिए: प्रश्न 5--6 की योजना से समदूरिक रूप से
        $(c_0)' \cong \ell^1$ दर्शाइए (परीक्षण अनुक्रमों का स्थान क्या लेता है?)।
        क्रमागत द्वैतों की शृंखला $c_0 \to \ell^1 \to \ell^\infty \to \dots$ संकलित कीजिए और चिह्नित
        कीजिए कि स्वद्वैतिता कहाँ विफल होती है।
  \item दर्शाइए कि किसी बानाख समष्टि का प्रत्येक दुर्बल अभिसारी
        अनुक्रम परिबद्ध होता है: $x^{(k)}$ को विहित अंतःस्थापन $J$
        के माध्यम से $E'$ पर फलनिकों के रूप में देखिए और
        बानाख--श्टाइनहाउस (\cref{thm:b3:banach:banachsteinhaus}) लगाइए --- किस बानाख समष्टि पर,
        और वहाँ \omterm{def:b3:complete:complete}{पूर्णता} क्यों उपलब्ध है?
  \item दर्शाइए कि $\ell^p$ में $1 < p < \infty$: $x^{(k)}
        \rightharpoonup x$ तभी और केवल तभी जब
        $\sup_k\norm{x^{(k)}}_p <
        \infty$ हो और प्रत्येक निर्देशांक $n$ के लिए $x^{(k)}_n \to x_n$ हो
        \emph{(एक दिशा विहित अंतःस्थापन से बानाख--श्टाइनहाउस का
        उपयोग करती है; दूसरी के लिए $y
        \in \ell^q$ का परिमित अनुक्रमों से
        आसन्नन कीजिए)}। इससे $\ell^2$ में $e_k
        \rightharpoonup 0$ निष्कर्ष निकालिए,
        जबकि $\norm{e_k}_2 =
        1$: दुर्बल सीमाएँ द्रव्यमान खो सकती हैं।
  \item दर्शाइए कि मानक दुर्बल रूप से अधोअर्धसंतत है: $x^{(k)} \rightharpoonup x$ से
        $\norm x \leq
        \liminf_k\,\norm{x^{(k)}}$ निकलता है \emph{($x$ के लिए कोई मानक निकालने वाला
        फलनिक चुनिए, \cref{cor:b3:banach:hbnormed})}।
  \item ($\ell^2$ में रादों--रीस) दर्शाइए कि $\ell^2$ में दुर्बल अभिसरण
        के साथ मानकों का अभिसरण मिलकर मानक अभिसरण दे देता है
        \emph{($\norm{x^{(k)} -
        x}_2^2$ का प्रसार कीजिए)}। मानक परिकल्पना के बिना
        उसी कथन का प्रतिउदाहरण दीजिए।
  \item (शूर, चरण 1) मान लीजिए $\ell^1$ में $x^{(k)} \rightharpoonup 0$ और, विरोधाभास के
        लिए, मान लीजिए किसी उपानुक्रम के अनुदिश $\norm{x^{(k)}}_1 \geq \delta > 0$। पहले
        दर्शाइए कि प्रत्येक $n$ के लिए $x^{(k)}_n \to 0$ (कौन-से फलनिक?),
        फिर पुनरावर्ती रूप से ऐसे सूचकांक $k_1 < k_2 < \cdots$ और पूर्णांक
        $0 = N_0 < N_1 < N_2 < \cdots$ बनाइए कि $x^{(k_j)}$ का द्रव्यमान खंड $B_j =
        \intoc{N_{j-1}}{N_j}$ पर संकेंद्रित
        हो जाए:
        \[
        \sum_{n \in B_j}\bigl|x^{(k_j)}_n\bigr| \geq
        \norm{x^{(k_j)}}_1 - \frac\delta{10} .
        \]
  \item (शूर, चरण 2) $n \in
        B_j$ के लिए $y_n =
        \operatorname{sign}\bigl(x^{(k_j)}_n\bigr)$ से $y \in \ell^\infty$ परिभाषित कीजिए।
        दर्शाइए कि $\langle x^{(k_j)}, y\rangle \geq
        \norm{x^{(k_j)}}_1 - \frac{2\delta}{10} \geq
        \frac{8\delta}{10}$ और $x^{(k)} \rightharpoonup 0$ के साथ विरोधाभास निकालिए।
        \emph{शूर की प्रमेय} निष्कर्ष के रूप में निकालिए: $\ell^1$
        में दुर्बल अभिसारी अनुक्रम मानक में अभिसरित होते हैं।
  \item निष्कर्ष निकालिए कि $\ell^1$ में $(e_k)$ का कोई दुर्बल अभिसारी
        उपानुक्रम नहीं है (उसकी एकमात्र संभावित सीमा निर्देशांकशः
        $0$ है --- अब शूर लगाइए), जिससे प्रश्न 13 की दुर्बल
        संहतता की विफलता पुनः प्राप्त हो जाती है; और स्पष्ट
        विरोधाभास सुलझाइए: $\ell^1$ में \emph{अनुक्रमों} का दुर्बल
        और मानक अभिसरण एक ही बात है, फिर भी दुर्बल और मानक
        \emph{संस्थितियाँ} भिन्न हैं और परिबद्ध समुच्चय अब भी
        दुर्बल अनुक्रमीय \omterm{def:b3:topology:compact}{संहत} नहीं होते --- इसमें कोई विरोधाभास
        नहीं, केवल स्वद्वैतिता की विफलता है।
  \item (संश्लेषण सारणी) $E \in \{c_0,\ \ell^1,\ \ell^p\
        (1{<}p{<}\infty),\ \ell^\infty\}$ के लिए सारणीबद्ध कीजिए: द्वैत;
        पृथक्करणीयता; स्वद्वैतिता; क्या परिबद्ध अनुक्रम दुर्बल
        अभिसारी उपानुक्रम स्वीकार करते हैं; और प्रत्येक समष्टि
        का एक विशिष्ट गुण, जिसे इस समस्या से एक पंक्ति में उचित
        ठहराया गया हो।
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- पूरक: निकटतम बिंदु, औसतीकृत अभिसरण,
और किसी बानाख सीमा का मान।}
\begin{enumerate}[resume]
  \item (निकटतम बिंदु: स्वद्वैतिता का लाभांश) मान लीजिए $F$
        $\ell^p$ ($1 < p < \infty$) की संवृत उपसमष्टि है और $x \in \ell^p$। दर्शाइए कि
        $d = \operatorname{dist}(x,
        F)$ \emph{प्राप्त हो जाता है}: किसी न्यूनीकारी अनुक्रम
        से कोई दुर्बल अभिसारी उपानुक्रम निकालिए (प्रश्न 13),
        हान--बानाख से ऐसा फलनिक बनाकर जो $F$ पर लुप्त हो पर
        उसके बाहर के किसी बिंदु पर नहीं, दुर्बल सीमा को $F$ के
        भीतर रखिए, और प्रश्न 17 से निष्कर्ष निकालिए। फिर दिखाइए
        कि यह विशेषाधिकार सार्वत्रिक नहीं है: $c_0$ में, $\Lambda(x) = \sum_n2^{-n}x_n$ के
        लिए, सिद्ध कीजिए कि इकाई गोले पर $\norm\Lambda = 1$ प्राप्त नहीं होता,
        दूरी सूत्र $\operatorname{dist}(x, \ker\Lambda) =
        \abs{\Lambda(x)}$ स्थापित कीजिए, और निष्कर्ष निकालिए कि
        संवृत अधिसमतल $\ker\Lambda$ में किसी भी $x \notin
        \ker\Lambda$ का कोई निकटतम बिंदु
        नहीं है।
  \item ($\ell^2$ में बानाख--साक्स) मान लीजिए $\ell^2_{\R}$ में $x^{(k)}
        \rightharpoonup 0$ और
        $\norm{x^{(k)}}_2 \leq C$। ऐसा उपानुक्रम $(y_j)$ बनाइए जिसके लिए सभी $i < j$ पर
        $\abs{\langle y_i, y_j\rangle} \leq
        \frac1j$ हो, और निष्कर्ष निकालिए
        \[
        \Bigl\lVert\frac{y_1 + \dots + y_m}m\Bigr\rVert_2^2
        \leq \frac{C^2 + 2}m \longrightarrow 0 :
        \]
        अर्थात् निष्कर्षण के बाद चेज़ारो माध्य \emph{मानक} में
        अभिसरित होते हैं। $(e_k)$ पर जाँचिए, जिसके माध्यों का मानक
        $\frac1{\sqrt m}$ है: दुर्बल अभिसरण, जो स्वयं अनुक्रम के लिए निरर्थक
        था (प्रश्न 16), औसतों के लिए मानक अभिसरण बन जाता है।
  \item (किसी बानाख सीमा का मान) मान लीजिए $L$ कोई बानाख सीमा
        है (प्रश्न 10) और $A_mx = \frac1m(x + Sx + \dots
        + S^{m-1}x)$। दर्शाइए कि $L(A_mx) = L(x)$ और $\liminf A_mx
        \leq L(x) \leq \limsup A_mx$; इससे
        निष्कर्ष निकालिए कि \emph{सभी} बानाख सीमाएँ आवर्ती
        अनुक्रमों पर सहमत होती हैं, और उनका मान एक आवर्त पर का
        माध्य होता है --- $(1, 0, 0, 1,
        0, 0, \dots)$ पर $\frac13$, जो प्रश्न 12 के $\frac12$
        के अनुरूप है। फिर दिखाइए कि व्यापक रूप में सहमति विफल हो
        जाती है: उस खंड अनुक्रम $x$ के लिए जो सम $j$ के लिए
        $\intoc{3^{j-1}}{3^j}$ पर $1$ है और अन्यत्र $0$, दर्शाइए कि चेज़ारो
        माध्य $\leq
        \frac13$ और $\geq \frac23$ के बीच दोलन करते हैं, और $L_-(x) \leq \frac13 < \frac23
        \leq L_+(x)$ वाली
        दो बानाख सीमाएँ $L_\pm$ बनाइए \emph{($c \oplus \R x$ से चरम स्वीकार्य
        मानों $\pm$ के साथ विस्तार कीजिए: जाँचिए कि $\Lambda(y + tx) = \lim y + t\,p(x)$ प्रश्न
        10 के उपरैखिक $p$ से प्रभावित है)}।
\end{enumerate}
\end{problem}
