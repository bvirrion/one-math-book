\chapter{साधारण अवकल समीकरण}\label{ch:b3:ode}

दूसरे वर्ष ने रैखिक अवकल समीकरण हल किए थे और कोशी--लिप्शिट्ज़
प्रमेय का कथन दिया था; यह अध्याय उसे सिद्ध करता है --- और वह भी
दो बार: बानाख स्थिर बिंदु से अस्तित्व और अद्वितीयता, तथा महत्तम
हलों के सिद्धांत और संहतों-से-पलायन प्रमेय से वैश्विक संरचना।
इसके बाद रैखिक सिद्धांत ईमानदार नींव पर फिर से खड़ा किया जाता है
(\omterm{thm:b3:ode:linearstructure}{विलायक}, \omterm{thm:b3:ode:linearstructure}{व्रोन्स्कियन}, \omterm{thm:b3:ode:matrixexp}{आव्यूह चरघातांकी}, दुआमेल), और अध्याय का
उत्तरार्ध \emph{गुणात्मक} सिद्धांत खोलता है --- प्रवाह,
साम्यावस्थाएँ, \omterm{thm:b3:ode:lyapunov}{ल्यापुनोव फलन}, और \omterm{thm:b3:ode:linearization}{रैखिकीकरण से स्थायित्व}: अर्थात्
उन हलों को समझना जिन्हें कोई कभी परिकलित नहीं करेगा। सप्ताहांत
समस्या में लोलक चिरंतन अध्ययन-विषय है। सर्वत्र $U \subseteq \R\times\R^d$ विवृत है और
$f \colon U \to \R^d$ \omterm{def:b3:topology:continuity}{संतत}; $x' = f(t, x)$ का \emph{हल} ऐसा $\mathcal C^1$ प्रतिचित्रण $x \colon I \to \R^d$ ($I$ कोई
अंतराल) है जिसका आलेख $U$ में हो और जो समीकरण संतुष्ट करे।

\section{कोशी--लिप्शिट्ज़}

\begin{definition}\label{def:b3:ode:lipschitz}
$f$ \emph{$x$ में स्थानीय लिप्शिट्ज़}\index{स्थानीय लिप्शिट्ज़}
कहलाता है यदि $U$ के प्रत्येक बिंदु का कोई \omterm{def:b3:topology:topology}{प्रतिवेश} $V$ और कोई
नियतांक $L$ ऐसा हो कि $(t, x_1), (t, x_2) \in V$ के लिए $\norm{f(t, x_1) - f(t, x_2)} \leq L\norm{x_1 -
x_2}$। यदि $f$ $\mathcal
C^1$ हो (अथवा
केवल $\partial_xf$ विद्यमान और \omterm{def:b3:topology:continuity}{संतत} हो), तो वह $x$ में स्थानीय लिप्शिट्ज़
है: किसी \omterm{def:b3:topology:compact}{संहत} उत्तल \omterm{def:b3:topology:topology}{प्रतिवेश} पर $L =
\sup\vertiii{\partial_xf}$ के साथ माध्य मान असमिका।
\end{definition}

\begin{theorem}[कोशी--लिप्शिट्ज़, स्थानीय]
\label{thm:b3:ode:cauchylipschitz}
मान लीजिए $f$ \omterm{def:b3:topology:continuity}{संतत} और $x$ में \omterm{def:b3:ode:lipschitz}{स्थानीय लिप्शिट्ज़} है, और $(t_0,
x_0) \in U$।
तब ऐसा $T > 0$ है कि कोशी समस्या
\[
x' = f(t, x), \qquad x(t_0) = x_0
\]
का $[t_0 - T, t_0 +
T]$ पर ठीक एक हल है।\index{कोशी--लिप्शिट्ज़ प्रमेय}
\end{theorem}

\begin{proof}
$Q = [t_0 - a, t_0 + a]\times\bar
B(x_0, b) \subseteq U$ वाला $a, b > 0$ चुनिए, जिस पर $\norm f \leq M$ और $f$ $x$ में
$L$-लिप्शिट्ज़ हो। कोई $\mathcal C^1$ फलन हल है तभी और केवल तभी जब वह समाकल
समीकरण
\[
x(t) = x_0 + \int_{t_0}^{t}f\bigl(s, x(s)\bigr)\,\dd s
\]
संतुष्ट करे (कलन की मूल प्रमेय, दोनों दिशाओं में)। मान लीजिए
$T =
\min\bigl(a, \frac bM, \frac1{2L}\bigr)$, $I = [t_0 - T, t_0
+ T]$, और
\[
\mathcal E = \{x \in \mathcal C(I, \R^d) : \norm{x(t) - x_0}
\leq b\ I \text{ पर}\},
\]
जो बानाख समष्टि $(\mathcal C(I, \R^d),
\norm\cdot_\infty)$ का संवृत उपसमुच्चय है: अतः \omterm{def:b3:complete:complete}{पूर्ण} (\cref{def:b3:complete:complete})।
$\Phi(x)(t) = x_0 +
\int_{t_0}^tf(s, x(s))\dd s$ परिभाषित कीजिए: $x \in \mathcal E$ के लिए
$\norm{\Phi(x)(t) - x_0} \leq M\abs{t - t_0} \leq MT \leq b$
--- अर्थात् $\Phi$ $\mathcal E$ को उसी पर भेजता है --- और $x, y \in
\mathcal E$ के लिए:
\[
\norm{\Phi(x)(t) - \Phi(y)(t)} \leq \Bigl|\int_{t_0}^t
L\,\norm{x(s) - y(s)}\,\dd s\Bigr| \leq LT\,\norm{x -
y}_\infty \leq \tfrac12\norm{x - y}_\infty :
\]
अर्थात् एक संकुचन। बानाख स्थिर बिंदु (\cref{thm:b3:complete:banach}) $\mathcal E$ में एक अद्वितीय
स्थिर बिंदु दे देता है: अस्तित्व, और $\bar B(x_0, b)$ में रहने वाले हलों के
बीच अद्वितीयता --- पर $I$ पर कोई भी हल वहीं रहता है (जब तक आलेख
$Q$ में रहता है तब तक $\norm{x(t) - x_0} \leq M\abs{t - t_0} \leq b$, जो एक \omterm{def:b3:topology:continuity}{सांतत्य} तर्क है): अतः $I$ पर
अद्वितीयता।
\end{proof}

\begin{lemma}[ग्रोनवाल]\label{lem:b3:ode:gronwall}
मान लीजिए $u \colon I \to \intco0\infty$ \omterm{def:b3:topology:continuity}{संतत} है, $t_0 \in
I$, और मान लीजिए\index{ग्रोनवाल की
प्रमेयिका}
\[
u(t) \leq \alpha + \beta\,\Bigl|\int_{t_0}^{t}u(s)\,\dd
s\Bigr| \qquad (t \in I)
\]
जहाँ $\alpha \geq 0$, $\beta > 0$। तब $I$ पर $u(t) \leq
\alpha\,\eu^{\beta\abs{t - t_0}}$।
\end{lemma}

\begin{proof}
$t \geq t_0$ के लिए: मान लीजिए $v(t) = \alpha +
\beta\int_{t_0}^tu(s)\dd s$, अतः $u \leq v$, $v' = \beta u
\leq \beta v$, और $(v\eu^{-\beta(t - t_0)})' \leq 0$: इसलिए
$v(t)
\leq v(t_0)\eu^{\beta(t-t_0)} = \alpha\eu^{\beta(t - t_0)}$। $t \leq t_0$ के लिए वही $\tilde u(t) = u(2t_0 -
t)$ पर लगाइए।
\end{proof}

\begin{corollary}[अद्वितीयता और संतत निर्भरता]
\label{cor:b3:ode:uniqueness}
\cref{thm:b3:ode:cauchylipschitz} की परिकल्पनाओं के अंतर्गत, $x' = f(t,x)$ के जो दो हल किसी एक बिंदु पर
सहमत हों वे अपनी उभयनिष्ठ परिभाषा-अंतराल पर सहमत होते हैं।
मात्रात्मक रूप से, यदि $x, y$ ऐसे दो हल हों जिनके आलेख उस क्षेत्र
में हों जहाँ $f$ $x$ में $L$-लिप्शिट्ज़ है, तो
\[
\norm{x(t) - y(t)} \leq \norm{x(t_0) -
y(t_0)}\,\eu^{L\abs{t - t_0}} .
\]
\end{corollary}

\begin{proof}
आकलन: $u = \norm{x - y}$ $u(t) \leq u(t_0)
+ L\abs{\int_{t_0}^tu}$ को संतुष्ट करता है (समाकल समीकरण घटाइए); अब
ग्रोनवाल। वैश्विक अद्वितीयता: सहमति समुच्चय $\{t : x(t)
= y(t)\}$ उभयनिष्ठ अंतराल
में संवृत, अरिक्त और विवृत भी है --- किसी भी सहमति बिंदु के चारों
ओर उभयनिष्ठ आलेख के किसी \omterm{def:b3:topology:compact}{संहत} टुकड़े को परिमित संख्या के लिप्शिट्ज़
बक्सों से आच्छादित कीजिए और प्रत्येक पर $u(t_1) = 0$ के साथ यह आकलन
लगाइए: तब स्थानीय रूप से $x \equiv y$। और किसी अंतराल का अरिक्त विवृत
संवृत उपसमुच्चय समूचा अंतराल ही होता है।
\end{proof}

\section{महत्तम हल}

\begin{theorem}[महत्तम हल; संहतों से पलायन]
\label{thm:b3:ode:maximal}
मान लीजिए $f$ \omterm{def:b3:topology:continuity}{संतत} और $x$ में \omterm{def:b3:ode:lipschitz}{स्थानीय लिप्शिट्ज़} है।
\begin{enumerate}
  \item प्रत्येक कोशी समस्या का एक अद्वितीय \emph{महत्तम} हल
        $x \colon \intoo{T_-}{T_+} \to \R^d$ होता है: $(t_0, x_0)$ से होकर जाने वाला हर दूसरा हल उसी का
        प्रतिबंध है। अंतराल विवृत होता है।
  \item (\emph{पलायन}) प्रत्येक \omterm{def:b3:topology:compact}{संहत} $K \subseteq U$ के लिए ऐसा $\varepsilon > 0$ है कि
        सभी $t \in \intoo{T_+ - \varepsilon}{T_+}$ के लिए $(t, x(t)) \notin
        K$ (और $T_-$ पर सममित रूप से):
        \emph{अर्थात् किसी \omterm{thm:b3:ode:maximal}{महत्तम हल} का आलेख अंततः $U$ के
        प्रत्येक \omterm{def:b3:topology:compact}{संहत} उपसमुच्चय को छोड़ देता है}। विशेष रूप से
        $U = \R\times\R^d$ और $T_+ < +\infty$ के लिए: $t \to T_+^-$ पर $\norm{x(t)} \to +\infty$
        (\emph{विस्फोट})।\index{महत्तम हल}\index{संहतों से
        पलायन}\index{विस्फोट}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) मान लीजिए $\mathcal S$ $(t_0, x_0)$ से होकर जाने वाले समस्त हलों का समुच्चय
है; \cref{cor:b3:ode:uniqueness} से कोई भी दो अपने अंतरालों के प्रतिच्छेदन पर सहमत होते
हैं, अतः वे जुड़ जाते हैं: $J =
\bigcup_{y \in \mathcal S}I_y$ पर $t$ पर परिभाषित किसी भी
$y$ के लिए $x(t) = y(t)$ रखिए: यह एक सुपरिभाषित हल है, जो स्पष्टतः
महत्तम और अद्वितीय है। $J$ विवृत है: किसी सिरे पर परिभाषित हल
को उसी सिरे पर \cref{thm:b3:ode:cauchylipschitz} से आगे बढ़ाया जा सकता।

(2) मान लीजिए दावा $T_+$ पर विफल होता है: तब $(t_n, x(t_n)) \in K$ वाले $t_n \to
T_+$ हैं;
ध्यान दीजिए कि इससे $T_+ <
\infty$ अनिवार्य हो जाता है, अथवा यदि $T_+ = \infty$ हो
तो सिद्ध करने को कुछ है ही नहीं ($K$ समय में परिबद्ध है)। अतः
मान लीजिए $T_+ < \infty$। संहतता: $K$ के किसी \omterm{def:b3:topology:topology}{प्रतिवेश} के समस्त कोशी
आँकड़ों के लिए एकसमान नियतांक $M, L, a,
b$ काम करते हैं --- मूर्त रूप में,
$K$ को स्थानीय प्रमेय की उपपत्ति जैसे परिमित संख्या के बक्सों
$Q_i$ से आच्छादित कीजिए और मान लीजिए $T^* > 0$ संगत अस्तित्व समयों का
न्यूनतम है: तब $K$ का कोई भी कोशी आँकड़ा ऐसा हल जन्माता है जो
अपने आरंभिक समय के आगे कम से कम $T^*$ तक जीता है। इसे $t_n > T_+ -
T^*/2$ के साथ
$(t_n, x(t_n))$ पर लगाने से $x$ $T_+$ के पार बढ़ जाता है (विस्तार
अद्वितीयता से $x$ के साथ सहमत होता है, फिर उसे आगे बढ़ा देता
है): जो महत्तमता का विरोधाभास है। अतः आलेख $T_+$ से पहले $K$ को
निश्चित रूप से छोड़ देता है। $U = \R\times\R^d$ के लिए: यदि $\norm{x(t)}\not\to\infty$ हो, तो कोई
अनुक्रम $t_n \to T_+$ $(t_n, x(t_n))$ को \omterm{def:b3:topology:compact}{संहत} $[t_0, T_+]\times\bar B(0, R)$ में रख देता: जो अपवर्जित है।
\end{proof}

\begin{corollary}[रैखिक वृद्धि के अंतर्गत वैश्विक अस्तित्व]
\label{cor:b3:ode:globallinear}
यदि $U = I\times\R^d$ ($I$ विवृत अंतराल) हो और \omterm{def:b3:topology:continuity}{संतत} $\alpha, \beta$ के साथ $\norm{f(t, x)}
\leq \alpha(t)\norm x + \beta(t)$, तो
प्रत्येक \omterm{thm:b3:ode:maximal}{महत्तम हल} समूचे $I$ पर परिभाषित होता है।
\end{corollary}

\begin{proof}
किसी \omterm{def:b3:topology:compact}{संहत} $[t_0, T] \subseteq I$ पर: $\norm{x(t)} \leq
\norm{x_0} + \int_{t_0}^t(\alpha\norm x + \beta)$, अतः ग्रोनवाल से (जहाँ $\alpha, \beta$ वहाँ $A, B$
से परिबद्ध है) $\norm{x(t)} \leq (\norm{x_0} + B(T - t_0))\eu^{A(T - t_0)}$: अर्थात् परिबद्ध। यदि $T_+ < \sup I$ हो, तो आलेख
$T_+$ के निकट $I\times\R^d$ के किसी \omterm{def:b3:topology:compact}{संहत} में रहता: जो पलायन (\cref{thm:b3:ode:maximal}) का
खंडन है।
\end{proof}

\section{रैखिक निकाय}

इस पूरे अनुभाग में $A \colon I \to M_d(\R)$ और $b
\colon I \to \R^d$ \omterm{def:b3:topology:continuity}{संतत} हैं; और निकाय $x' = A(t)x
+ b(t)$ है ---
रैखिक वृद्धि: अतः सभी \omterm{thm:b3:ode:maximal}{महत्तम हल} समूचे $I$ पर जीते हैं (\cref{cor:b3:ode:globallinear})।

\begin{theorem}[संरचना]\label{thm:b3:ode:linearstructure}
समघात निकाय $x' = A(t)x$ के हल एक $d$-विमीय सदिश समष्टि $S_H$ बनाते
हैं; और प्रत्येक $t_0$ के लिए मान-निर्धारण $x \mapsto x(t_0)$ एक तुल्याकारिता
$S_H \to
\R^d$ है। \emph{विलायक}\index{रैखिक निकाय का विलायक} $R(t, s) \in GL_d(\R)$, जो
इस प्रकार परिभाषित है कि $t \mapsto R(t,
s)v$ $s$ पर मान $v$ वाला हल है,
\[
R(s,s) = I,\quad R(t, u)R(u, s) = R(t, s),\quad
\partial_tR(t,s) = A(t)R(t,s),
\]
संतुष्ट करता है, और असमघात समस्या \emph{दुआमेल के
सूत्र}\index{दुआमेल का सूत्र} से हल हो जाती है:
\[
x(t) = R(t, t_0)\,x_0 + \int_{t_0}^{t}R(t, s)\,b(s)\,\dd s .
\]
अंततः \emph{व्रोन्स्कियन}\index{व्रोन्स्कियन} $w(t) = \det
R(t, s)$ ल्यूविल के
सूत्र $w' =
\operatorname{tr}A(t)\,w$ का पालन करता है, अतः $w(t) =
\exp\bigl(\int_s^t\operatorname{tr}A\bigr) > 0$।
\end{theorem}

\begin{proof}
समीकरण की रैखिकता हलों को एक सदिश समष्टि बना देती है;
मान-निर्धारण रैखिक है, एकैकी (अद्वितीयता: $t_0$ पर लुप्त होने
वाला हल $\equiv 0$ है) और आच्छादक (अस्तित्व): अतः विमा $d$। \omterm{thm:b3:ode:linearstructure}{विलायक}
के गुण अद्वितीयता को ही दोहराते हैं (प्रत्येक सर्वसमिका के दोनों
पक्ष एक ही कोशी समस्या हल करते हैं); और व्युत्क्रमणीयता $R(s,t)R(t,s) = I$ से।
दुआमेल: सूत्र का अवकलन कीजिए --- $x'(t) = A(t)R(t,t_0)x_0 + R(t,t)b(t) +
\int_{t_0}^tA(t)R(t,s)b(s)\dd s = A(t)x(t) + b(t)$ (समाकल के भीतर अवकलन वैध
है: समाकल्य $t$ में $\mathcal C^1$ है और $(t,s)$ में उसका अवकलज \omterm{def:b3:topology:continuity}{संतत}; अथवा
समाकल समीकरण से सत्यापित कीजिए)। ल्यूविल: $w(t + h) = \det\bigl(R(t+h, t)\bigr)w(t)$ और $R(t + h, t) =
I + hA(t) + o(h)$ (समाकल
समीकरण से), अतः $\det = 1
+ h\operatorname{tr}A(t) + o(h)$ ($I$ पर $\det$ का प्रसार): इसलिए $w'(t) = \operatorname{tr}A(t)\,w(t)$; अब
अदिश रैखिक अवकल समीकरण का समाकलन कीजिए।
\end{proof}

\begin{theorem}[आव्यूह चरघातांकी]\label{thm:b3:ode:matrixexp}
$A \in M_d(\C)$ के लिए श्रेणी $\eu^{A} =
\sum_{n\geq0}\frac{A^n}{n!}$ अभिसरित होती है (किसी भी उपगुणात्मक
मानक में, निरपेक्षतः), और $AB = BA$ होने पर $\eu^{A+B} = \eu^A\eu^B$, तथा $t \mapsto \eu^{tA}$ अचर
निकाय का \omterm{thm:b3:ode:linearstructure}{विलायक} है: $R(t,s) = \eu^{(t-s)A}$; वह $\mathcal
C^\infty$ है, जिसमें $\frac{\dd}{\dd t}\eu^{tA} = A\eu^{tA}$। इसके
अतिरिक्त: यदि $A$ के प्रत्येक अभिलक्षणिक मान $\lambda$ के लिए $\operatorname{Re}\lambda < -\alpha < 0$
हो, तो $t \geq 0$ पर $\vertiii{\eu^{tA}}
\leq C\eu^{-\alpha t}$।\index{आव्यूह चरघातांकी}
\end{theorem}

\begin{proof}
अभिसरण: $\vertiii{A^n/n!} \leq \vertiii A^n/n!$, जो योग्य है (बानाख बीजगणित $M_d$ में \cref{exo:b3:complete:1}(ख))।
क्रमविनिमेय $A, B$ के लिए: दोनों निरपेक्षतः अभिसारी श्रेणियों का
कोशी गुणनफल, द्विपद प्रमेय (जो $AB = BA$ होने पर वैध है) के माध्यम से
पुनर्विन्यस्त होकर $\sum_n\frac{(A+B)^n}{n!}$ बन जाता है। अवकलनीयता, सीधे: $\frac{\eu^{(t+h)A} - \eu^{tA}}h = \eu^{tA}\frac{\eu^{hA} -
I}{h} \to \eu^{tA}A$,
क्योंकि $\norm{\frac{\eu^{hA} - I}h - A}
\leq \sum_{n\geq2}\frac{\abs h^{n-1}\vertiii A^n}{n!} =
O(h)$। अतः $t \mapsto \eu^{(t - s)A}v$ $R(t,s)v$ को परिभाषित करने वाली कोशी समस्या
हल कर देता है। स्पेक्ट्रमी परिबंध: \omterm{thm:b3:modules:jordan}{जॉर्डन रूप} (\cref{thm:b3:modules:jordan}) से $A = P(D + N)P^{-1}$,
जहाँ $D$ अभिलक्षणिक मान धारण करने वाला विकर्ण है, $N$
शून्यभावी, और $DN =
ND$। तब $\eu^{tA} = P\,\eu^{tD}\eu^{tN}P^{-1}$, जिसमें किसी $\delta > 0$ ($t \geq 0$) के लिए
$\vertiii{\eu^{tD}} \leq \eu^{-(\alpha + \delta)t}$ और $t$ में $\eu^{tN}$ बहुपद (शून्यभाविता): अतः गुणनफल $\leq C\eu^{-\alpha t}$
है ($\eu^{-\delta t}$ बहुपद को हरा देता है)।
\end{proof}

\begin{example}[समतल, वर्गीकृत]
\label{ex:b3:ode:planeclassification}
व्युत्क्रमणीय $A \in M_2(\R)$ के साथ $x' = Ax$ के लिए $0$ के निकट प्रावस्था
चित्र $\tau = \operatorname{tr}A$ और $\delta = \det A$ से, अभिलक्षणिक मानों $\lambda_\pm
= \frac{\tau \pm \sqrt{\tau^2 - 4\delta}}2$ के माध्यम से, तय
होता है:
\begin{itemize}
  \item $\delta < 0$: विपरीत चिह्नों वाले वास्तविक अभिलक्षणिक मान ---
        एक \emph{काठी}; दो प्रक्षेपपथ भीतर आते हैं, दो बाहर जाते
        हैं, और शेष सब पास से गुजर जाते हैं। सदा अस्थायी।
  \item $\delta > 0$, $\tau^2 \geq 4\delta$: एक ही चिह्न वाले वास्तविक अभिलक्षणिक मान
        ($= \operatorname{sign}\tau$) --- एक \emph{ग्रंथि}, जो \omterm{def:b3:ode:stability}{स्थायी} है तभी और केवल
        तभी जब $\tau < 0$; प्रक्षेपपथ मंद अभिलक्षणिक दिशा के स्पर्शी
        होते हैं।
  \item $\delta > 0$, $\tau^2 < 4\delta$, $\tau \neq 0$: सम्मिश्र संयुग्मी अभिलक्षणिक मान
        $\frac\tau2 \pm
        \iu\omega$ --- एक \emph{सर्पिल} (नाभि), जो \omterm{def:b3:ode:stability}{स्थायी} है तभी और
        केवल तभी जब $\tau < 0$; और हल आवर्तकाल $\frac{2\pi}\omega$ वाले $\eu^{\tau t/2}\times$ घूर्णन
        हैं।
  \item $\tau = 0$, $\delta > 0$: विशुद्ध काल्पनिक अभिलक्षणिक मान --- एक
        \emph{\omterm{ex:b3:groups:actions}{केंद्र}}: संवृत कक्षाएँ (दीर्घवृत्त), और स्पर्शोन्मुख
        स्थायित्व के बिना स्थायित्व, ठीक वही सीमांत स्थिति जिसे
        \cref{thm:b3:ode:linearization} अरैखिक निकायों के लिए तय नहीं कर सकती (लोलक की
        निचली साम्यावस्था, \cref{pb:b3:ode:1}, यहीं बैठती है)।
\end{itemize}
परिसीमा परवलय $\tau^2 = 4\delta$ पर अपकर्षित ग्रंथियाँ होती हैं (जॉर्डन खंड:
एक ही स्पर्श दिशा वाले प्रक्षेपपथ)। सब कुछ दो संख्याओं में पढ़
लिया जाता है --- और इसीलिए किसी भी समतलीय प्रावस्था चित्र से पहले
पहली प्रतिक्रिया $\operatorname{tr}$ और $\det$ परिकलित करना ही होती है;
उदाहरणार्थ $A = \bigl(\begin{smallmatrix}0 & 1\\ -1 &
-c\end{smallmatrix}\bigr)$ (अवमंदित दोलक): $\delta = 1 >
0$, $\tau = -c$: अतः $0 < c < 2$ के लिए
\omterm{def:b3:ode:stability}{स्थायी} सर्पिल, और $c \geq 2$ के लिए \omterm{def:b3:ode:stability}{स्थायी} ग्रंथि --- एक ही दृष्टि में
अल्प-अवमंदन बनाम अति-अवमंदन।
\end{example}

\section{प्रवाह, साम्यावस्थाएँ, स्थायित्व}

अब \emph{स्वायत्त} समीकरण $x' = F(x)$ पर विचार कीजिए, जिसमें $F
\colon \Omega \to \R^d$
विवृत $\Omega
\subseteq \R^d$ पर \omterm{def:b3:ode:lipschitz}{स्थानीय लिप्शिट्ज़} है। $x(0) = x_0$ वाले \omterm{thm:b3:ode:maximal}{महत्तम हल} के लिए
$\varphi_t(x_0) = x(t)$ लिखिए (\emph{सदिश क्षेत्र का प्रवाह}\index{सदिश क्षेत्र का
प्रवाह}); स्वायत्तता जहाँ परिभाषित हो वहाँ समूह गुण $\varphi_{t+s} =
\varphi_t\circ\varphi_s$ दे देती
है (दोनों पक्ष समय $s$ पर एक ही समस्या हल करते हैं)।

\begin{definition}\label{def:b3:ode:stability}
\emph{साम्यावस्था} वह बिंदु $\bar x$ है जिसके लिए $F(\bar x) =
0$ (अतः $\varphi_t(\bar x) = \bar x$)।
वह \emph{स्थायी}\index{स्थायित्व (ल्यापुनोव)} कहलाती है यदि
प्रत्येक $\varepsilon > 0$ के लिए ऐसा $\delta > 0$ हो कि $\norm{x_0 - \bar x} < \delta$ से यह निकले कि हल सभी
$t \geq 0$ के लिए विद्यमान है और $\norm{\varphi_t(x_0) - \bar
x} < \varepsilon$; और \emph{स्पर्शोन्मुख रूप से
स्थायी} यदि इसके अतिरिक्त $\bar x$ के निकट सभी $x_0$ के लिए $\varphi_t(x_0) \to \bar x$।
\end{definition}

\begin{theorem}[ल्यापुनोव फलन]\label{thm:b3:ode:lyapunov}
मान लीजिए $\bar x$ कोई साम्यावस्था है और $V \colon \mathcal V \to
\R$ $\bar x$ के किसी
\omterm{def:b3:topology:topology}{प्रतिवेश} पर ऐसा $\mathcal C^1$ फलन है कि:\index{ल्यापुनोव फलन}
\[
V(\bar x) = 0,\qquad V(x) > 0 \text{ जब } x \neq \bar x,
\qquad \dot V(x) = \nabla V(x)\cdot F(x) \leq 0 .
\]
तब $\bar x$ \omterm{def:b3:ode:stability}{स्थायी} है। और यदि इसके अतिरिक्त $\bar
x$ के बाहर $\dot V < 0$ हो,
तो $\bar x$ स्पर्शोन्मुख रूप से \omterm{def:b3:ode:stability}{स्थायी} है।
\end{theorem}

\begin{proof}
किसी हल के अनुदिश $\frac{\dd}{\dd t}V(x(t)) = \dot V(x(t))
\leq 0$: अतः $V$ घटता है। $\varepsilon$ दिया हो (इतना
छोटा कि $\bar B(\bar x, \varepsilon) \subseteq \mathcal V$), तो
$m = \min\{V(x) : \norm{x - \bar x} = \varepsilon\} > 0$
लीजिए (संहतता, धनात्मकता) और $B(\bar x, \delta)$ पर $V < m$ वाला $\delta < \varepsilon$ चुनिए
(\omterm{def:b3:topology:continuity}{सांतत्य})। $B(\bar x, \delta)$ में आरंभ होने वाले किसी हल के लिए बाद के सभी
समयों पर $V(x(t)) < m$ रहता है, अतः वह गोलक $\norm{x -
\bar x} = \varepsilon$ (जहाँ $V \geq m$) तक कभी
नहीं पहुँच सकता: वह गोले में ही रहता है --- और तब सभी $t \geq 0$ के
लिए विद्यमान रहता है: हल \omterm{def:b3:topology:compact}{संहत} $\bar B$ में बना रहता है, अतः \cref{thm:b3:ode:maximal}(2)
(\omterm{thm:b3:ode:maximal}{संहतों से पलायन}) $T_+ = +\infty$ अनिवार्य कर देती है। यही स्थायित्व है।

स्पर्शोन्मुख स्थिति: मान लीजिए $x(t)$ $B(\bar x, \delta)$ में आरंभ होता है; तब
$V(x(t))$ घटकर किसी $c \geq 0$ तक जाता है। यदि $c > 0$ हो: प्रक्षेपपथ $K = \{x \in \bar B(\bar x,
\varepsilon): V(x) \geq c\}$
में रहता है, जो एक ऐसा \omterm{def:b3:topology:compact}{संहत} समुच्चय है जो $\bar x$ के किसी \omterm{def:b3:topology:topology}{प्रतिवेश}
को अपवर्जित कर देता है ($V$ $V(\bar x)
= 0 < c$ के साथ \omterm{def:b3:topology:continuity}{संतत} है)। $K$ पर फलन
$\dot V$ \omterm{def:b3:topology:continuity}{संतत} और कड़ाई से ऋणात्मक है, अतः संहतता से $\mu = \max_K\dot V < 0$; तब $V(x(t)) \leq V(x(0)) + \mu t \to
-\infty$:
निरर्थक, इसलिए $V \geq 0$। अतः $c = 0$, और $x(t) \to \bar x$ (गोले के भीतर $\bar x$ से
दूरी $\geq
\rho$ वाले बिंदुओं के लिए $V \geq m_\rho >
0$)।
\end{proof}

\begin{theorem}[रैखिकीकरण से स्थायित्व]
\label{thm:b3:ode:linearization}
मान लीजिए $F$ $\mathcal C^1$ है, $F(\bar x) = 0$, $A = DF(\bar x)$। यदि $A$ के प्रत्येक
अभिलक्षणिक मान के लिए $\operatorname{Re}\lambda <
0$ हो, तो $\bar x$ स्पर्शोन्मुख रूप से
\omterm{def:b3:ode:stability}{स्थायी} है।\index{रैखिकीकरण से स्थायित्व}
\end{theorem}

\begin{proof}
$\bar x$ को $0$ पर स्थानांतरित कीजिए और $g(x) = o(\norm x)$ ($\mathcal C^1$ अवकलनीयता) के
साथ $F(x) = Ax + g(x)$ लिखिए। $\vertiii{\eu^{tA}} \leq C\eu^{-\alpha t}$ वाला $\alpha > 0$ चुनिए ($t \geq 0$; \cref{thm:b3:ode:matrixexp}) और $\norm x \leq
r$
के लिए $\norm{g(x)} \leq \frac{\alpha}{2C}\norm x$ वाला $r > 0$। $b(s) = g(x(s))$ के साथ दुआमेल:
\[
x(t) = \eu^{tA}x_0 + \int_0^t\eu^{(t-s)A}g(x(s))\,\dd s,
\]
जो तब तक वैध है जब तक $\norm{x(s)} \leq r$। तब $u(t) =
\eu^{\alpha t}\norm{x(t)}$
\[
u(t) \leq C\norm{x_0} +
\int_0^{t}C\,\frac{\alpha}{2C}\,u(s)\,\dd s ,
\]
संतुष्ट करता है, अतः ग्रोनवाल $u(t) \leq C\norm{x_0}\eu^{\alpha t/2}$ देता है, अर्थात् $\norm{x(t)} \leq C\norm{x_0}\eu^{-\alpha t/2}$। यदि
$\norm{x_0} < r/C$ हो, तो यह पूर्व परिबंध सभी $t$ के लिए $\norm{x(t)} <
r$ बनाए रखता है
(एक \omterm{def:b3:topology:continuity}{सांतत्य}/उन्नयन तर्क: कड़े आकलन को देखते हुए जिन समयों पर
$\norm x \leq r$ है उनका समुच्चय $\intco0{T_+}$ में विवृत और संवृत दोनों है), हल
वैश्विक हो जाता है, और वह $0$ पर चरघातांकी दर से अभिसरित होता
है: अर्थात् स्पर्शोन्मुख स्थायित्व।
\end{proof}

\begin{method}\label{met:b3:ode:strategy}
किसी अवकल समीकरण का सामना होने पर: (1) अस्तित्व/अद्वितीयता ---
\omterm{def:b3:ode:lipschitz}{स्थानीय लिप्शिट्ज़} जाँचिए (प्रायः $\mathcal C^1$); (2) वैश्विकता ---
रैखिक वृद्धि, परिबद्धता, अथवा किसी \omterm{thm:b3:ode:lyapunov}{ल्यापुनोव फलन} या प्रथम समाकल
से कोई अपरिवर्ती \omterm{def:b3:topology:compact}{संहत}; और यदि यह न मिले, तो विस्फोट का संदेह
कीजिए और अदिश व्यंग्यचित्र $x' = x^2$ पर परखिए; (3) रैखिक निकाय ---
\omterm{thm:b3:ode:linearstructure}{विलायक}, दुआमेल, और अचर गुणांकों के लिए $A$ की अभिलक्षणिक
संरचना; (4) गुणात्मक प्रश्न --- साम्यावस्थाएँ, रैखिकीकरण, और
किसी \omterm{thm:b3:ode:lyapunov}{ल्यापुनोव फलन} (यांत्रिक निकाय होने पर ऊर्जा) या ऐसे प्रथम
समाकल की खोज जिसके स्तर समुच्चय प्रक्षेपपथों को फँसा लें।
सप्ताहांत समस्या पूरी विधि को लोलक पर चलाकर दिखाती है।
\end{method}

\begin{exercise}[$\star$]\label{exo:b3:ode:1}
स्पष्ट रूप से हल कीजिए और महत्तम अंतराल निर्धारित कीजिए:
(क) $x' = x^2$, $x(0) = 1$; (ख) $x' = 1 + x^2$, $x(0) = 0$; (ग) $x' = x(1-x)$, $x(0) = \frac12$। प्रत्येक
उत्तर का \cref{thm:b3:ode:maximal}(2) और \cref{cor:b3:ode:globallinear} से मेल कराइए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:ode:2}
मान लीजिए $x, y$ $x' = f(t,x)$ को हल करते हैं, जहाँ $f$ $\R\times\R^d$ पर $x$
में वैश्विक रूप से $L$-लिप्शिट्ज़ है।
(क) $\norm{x(t) - y(t)} \leq \norm{x(0) -
y(0)}\eu^{L\abs t}$ सिद्ध कीजिए, और उदाहरण से (रैखिक!) दर्शाइए कि गुणक
$\eu^{L\abs t}$ प्राप्त हो जाता है।
(ख) निष्कर्ष निकालिए कि \omterm{ex:b3:ode:planeclassification}{प्रवाह प्रतिचित्रण} $x_0 \mapsto x(t; x_0)$ \omterm{def:b3:topology:continuity}{संतत} है, और
परिबद्ध समुच्चयों पर वस्तुतः लिप्शिट्ज़।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:3}
दर्शाइए कि निम्नलिखित में से प्रत्येक के सभी \omterm{thm:b3:ode:maximal}{महत्तम हल} $\R$ पर
वैश्विक हैं, और उचित प्रमेय उद्धृत कीजिए:
(क) $x' = \sin(tx)$;
(ख) $x' = \frac{t\,x}{1 + x^2}$;
(ग) \omterm{def:b3:topology:continuity}{संतत} $q$ के साथ $x'' + q(t)x = 0$ (प्रथम-कोटि निकाय में बदलिए);
(घ) \omterm{def:b3:topology:continuity}{संतत} और परिबद्ध $A$ के साथ $x' = A(t)x$ --- और $\norm{x(t)}$ पर ग्रोनवाल
परिबंध दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:4}
(क) $A = \bigl(\begin{smallmatrix}0 & -1\\ 1 &
0\end{smallmatrix}\bigr)$, $\bigl(\begin{smallmatrix}\lambda & 1\\ 0 &
\lambda\end{smallmatrix}\bigr)$, और $\bigl(\begin{smallmatrix}0 & 1\\ 1 &

(ख) प्रणोदित दोलक $x'' + x = \cos(\omega t)$ को (निकाय रूप में) दुआमेल से हल कीजिए,

1$ के लिए: अनुनाद धर्मनिरपेक्ष पद $t\sin t$ के रूप में

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:5}
अदिश समीकरण $x'' + p(t)x' + q(t)x = 0$ के लिए:
(क) दर्शाइए कि दो हलों का \omterm{thm:b3:ode:linearstructure}{व्रोन्स्कियन} $w = x_1x_2' - x_1'x_2$ $w' = -p\,w$ (आबेल) को
संतुष्ट करता है, और निष्कर्ष निकालिए कि जिन दो हलों के लिए कहीं
$w \neq 0$ हो वे एक आधार बनाते हैं।
(ख) कोई एक अलुप्त हल $x_1$ दिया हो, तो \emph{कोटि अपचयन} से
सामान्य हल ढूँढ़िए: $x_2 = x_1\int
\frac{\eu^{-\int p}}{x_1^2}$ रखिए और सत्यापित कीजिए। इसे $x_1(t) = t^2$ के
साथ $\intoo0{+\infty}$ पर $t^2x'' -
2x = 0$ पर लगाइए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:6}
(लॉजिस्टिक) $x' = x(1 - x)$ के लिए: समस्त साम्यावस्थाएँ और उनका स्थायित्व
निर्धारित कीजिए (\cref{thm:b3:ode:linearization} से और सीधे भी); दर्शाइए कि $x(0) \in \intoo01$ वाला
प्रत्येक हल बढ़ता हुआ और वैश्विक है, जिसकी $\mp\infty$ पर सीमाएँ $0$
और $1$ हैं; और स्पष्ट रूप से हल करके पुष्टि कीजिए। अधिक व्यापक
रूप से दर्शाइए कि अदिश स्वायत्त हल एकदिष्ट होते हैं, और निष्कर्ष
निकालिए: विमा $1$ में कोई अचरेतर आवर्ती हल नहीं होता।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:7}
(प्रथम समाकल) मान लीजिए $H \colon \Omega \to \R$ $\mathcal
C^1$ है और समतलीय हैमिल्टनी निकाय
$x' =
\partial_yH$, $y' = -\partial_xH$ पर विचार कीजिए।
(क) दर्शाइए कि $H$ हलों के अनुदिश अचर है।
(ख) $H = \frac{y^2}2 + \frac{x^4}4$ के लिए: दर्शाइए कि सभी हल वैश्विक और परिबद्ध हैं, और
मूल बिंदु \omterm{def:b3:ode:stability}{स्थायी} है (ल्यापुनोव: $H$), यद्यपि रैखिकीकरण ($\bigl(\begin{smallmatrix}0&1\\0&0\end{smallmatrix}\bigr)$)
स्पर्शोन्मुख रूप से \omterm{def:b3:ode:stability}{स्थायी} \emph{नहीं} है: अर्थात् रैखिकीकरण
अनिर्णायक हो सकता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:8}
(अवमंदित लोलक) $x'' + cx' + \sin x = 0$, $c > 0$; निकाय: $x' = y$, $y' = -\sin x - cy$।
(क) दर्शाइए कि $V(x, y) = \frac{y^2}2 + 1 - \cos x$ $\dot
V = -cy^2 \leq 0$ को संतुष्ट करता है: अतः मूल बिंदु
\omterm{def:b3:ode:stability}{स्थायी} है।
(ख) $\dot V$ पूरे अक्ष $y = 0$ पर लुप्त हो जाता है: अतः ल्यापुनोव की
कड़ी कसौटी विफल हो जाती है। फिर भी रैखिकीकरण (\cref{thm:b3:ode:linearization}) से
स्पर्शोन्मुख स्थायित्व सिद्ध कीजिए: $(0,0)$ पर रैखिकीकृत आव्यूह के
अभिलक्षणिक मान परिकलित कीजिए और प्रत्येक $c > 0$ के लिए $\operatorname{Re} < 0$
जाँचिए।
(ग) साम्यावस्था $(\pi, 0)$ पर क्या होता है? रैखिकीकरण परिकलित कीजिए
और निष्कर्ष निकालिए (एक अभिलक्षणिक मान धनात्मक: अस्थायित्व ---
आप अस्थायित्व के कथन का अनौपचारिक उपयोग कर सकते हैं अथवा रैखिक
निकाय का कोई स्पष्ट भागता हुआ हल प्रस्तुत कर सकते हैं)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:9}
(अद्वितीयता की सीमा) $\alpha \in \intoo01$ के लिए दर्शाइए कि समस्या $x' = \abs x^{\alpha}$, $x(0) = 0$
के अनंत हल हैं (\cref{pb:b3:complete:1}, भाग III को अनुकूलित कीजिए)। इसके विपरीत
दर्शाइए कि $\alpha = 1$ (अर्थात् $x' = \abs
x$) के लिए $0$ से होकर जाने वाला
हल अद्वितीय है, और ठीक-ठीक बताइए कि \cref{thm:b3:ode:cauchylipschitz} की कौन-सी परिकल्पना इन
दोनों स्थितियों में भेद करती है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:10}
(पूर्व परिबंध हलों को फँसा लेते हैं) मान लीजिए $F \colon \R^d \to \R^d$ \omterm{def:b3:ode:lipschitz}{स्थानीय
लिप्शिट्ज़} है और $\norm x \geq R$ होने पर $\langle F(x), x\rangle \leq 0$।
(क) दर्शाइए कि संवृत गोला $\bar B(0, R)$ धनात्मक रूप से अपरिवर्ती है:
भीतर से आरंभ होने वाले हल $t \geq
0$ के लिए भीतर ही रहते हैं।
\emph{(यदि $\norm{x(t_2)} > R$ हो, तो $\norm{x(t_1)} = R$ वाला अंतिम समय $t_1 < t_2$ लीजिए और
$\intcc{t_1}{t_2}$ पर $\frac{\dd}{\dd t}\norm{x(t)}^2$ का अध्ययन कीजिए।)}
(ख) गोले के आँकड़ों के लिए वैश्विक अग्रगामी अस्तित्व निष्कर्ष के
रूप में निकालिए। फिर प्रवणक निकाय $x' = -\nabla G(x)$, $G \in
\mathcal C^2$ लीजिए, जहाँ $\norm x \to
\infty$
पर $G(x) \to +\infty$: दर्शाइए कि हलों के अनुदिश $G$ घटता है, कि प्रत्येक
हल (परिबद्ध) अधोस्तर समुच्चय $\{G \leq
G(x_0)\}$ में रहता है, और वैश्विक
अग्रगामी अस्तित्व निष्कर्ष निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:11}
(तुलना से विस्फोट) $x' = x^2 + t^2$, $x(0) =
1$ पर विचार कीजिए।
(क) दर्शाइए कि \omterm{thm:b3:ode:maximal}{महत्तम हल} किसी $\intco0{T_+}$ पर विद्यमान है जिसमें $T_+ < \infty$:
$y' = y^2$, $y(0) = 1$ से तुलना कीजिए \emph{(अभीष्ट तुलना प्रमेयिका सिद्ध
कीजिए: यदि $x' \geq
F(x)$ और $x(0) \geq y(0)$ के साथ $y' = F(y)$ हो, तो जहाँ दोनों जीते हैं
वहाँ $x \geq
y$)}, और $T_+ \leq 1$ निष्कर्ष निकालिए।
(ख) $T_+$ को नीचे से परिबद्ध कीजिए: $\intcc01$ पर $x' \leq x^2 +
1$; $z(0) =
1$ वाले
अधिहल $z' = z^2 + 1$ से तुलना कीजिए, जो $z(t) = \tan\bigl(t + \frac\pi4\bigr)$ से हल हो जाता है, और $T_+ \geq \frac\pi4$
निष्कर्ष निकालिए।
(ग) $\frac\pi4 \leq T_+ \leq 1$ संकलित कीजिए और सार बाँधिए: दायाँ पक्ष अधिरैखिक रूप से
बढ़ता हो तो वैश्विक अस्तित्व मर जाता है (\cref{exo:b3:ode:3} इसका प्रतिबिंदु
है), और सीमा $\int^{\infty}\frac{\dd s}{F(s)}$ के अभिसरण पर पड़ती है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:12}
(श्टुर्म की तुलना प्रमेय) मान लीजिए $q_1 \leq q_2$ किसी अंतराल $I$ पर
\omterm{def:b3:topology:continuity}{संतत} हैं, और $u \neq 0$ $u''
+ q_1u = 0$ को तथा $v \neq 0$ $v'' + q_2v = 0$ को हल करते हैं।
(क) \omterm{thm:b3:ode:linearstructure}{व्रोन्स्कियन} सर्वसमिका स्थापित कीजिए: $W = uv' - u'v$ के साथ $W' = (q_1 - q_2)\,uv$।
(ख) (श्टुर्म) दर्शाइए कि $u$ के दो क्रमागत शून्यों $a < b$ के बीच
या तो $v$ $\intoo ab$ में कहीं लुप्त होता है, या वहाँ $q_1 = q_2$ और $v \propto u$
\emph{($\intoo ab$ पर $u > 0$ और $v > 0$ भी मान लीजिए; (क) का $a$ से
$b$ तक समाकलन कीजिए और सीमांत पदों $W(a),
W(b)$ के चिह्न देखिए)}।
(ग) निष्कर्ष निकालिए: $q \geq m > 0$ वाले $u'' + qu = 0$ के हल लंबाई $\pi/\sqrt m$ के
प्रत्येक अंतराल में कम से कम एक बार लुप्त होते हैं ($v'' + mv = 0$ से
तुलना कीजिए); और $q \leq 0$ वाले हल $\R$ पर अधिकतम एक बार। दोनों को
$q \equiv \pm1$ पर परखिए।
\end{exercise}

\section{समस्या: लोलक, पूर्णतः हल}

\begin{problem}[{सप्ताहांत समस्या --- दोलन, घूर्णन, विभाजक, और
आवर्तकाल}]\label{pb:b3:ode:1}
लोलक समीकरण $x'' = -\sin x$ --- निकाय के रूप में: $\R^2$ पर $x' =
y$, $y' = -\sin x$ ---
गतिकी की ड्रोसोफिला है: लिखने में सरल, प्रारंभिक सूत्रों से हल
करने में असंभव, फिर भी गुणात्मक विधि से पूर्णतः बोधगम्य। मान
लीजिए $E(x, y) = \frac{y^2}2 - \cos x$ (\emph{ऊर्जा})।

\medskip
\noindent\textbf{भाग I --- वैश्विक संरचना।}
\begin{enumerate}
  \item दर्शाइए कि सभी \omterm{thm:b3:ode:maximal}{महत्तम हल} वैश्विक ($\R$-परिभाषित) हैं:
        $\dot E = 0$ और \cref{thm:b3:ode:maximal} का उपयोग कीजिए। साम्यावस्थाएँ: $(k\pi, 0)$;
        उनके रैखिकीकरणों का वर्गीकरण कीजिए (सम $k$ के लिए
        केंद्र-प्रकार, विषम $k$ के लिए काठी)।
  \item दर्शाइए कि प्रक्षेपपथ स्तर समुच्चयों $\{E = E_0\}$ में अंतर्विष्ट
        हैं, और उन्हें $E_0 \in \intco{-1}{+\infty}$ के मान के अनुसार खींचिए/वर्णित
        कीजिए: $E_0 = -1$ (साम्यावस्थाएँ), $-1 < E_0 < 1$ ($(2k\pi, 0)$ के चारों ओर
        संवृत वक्र), $E_0 = 1$ ($(\pm\pi, 0)$ से होकर जाता विभाजक), $E_0 > 1$
        ($x$ पर आलेख: घूर्णन)।
  \item सिद्ध कीजिए कि निचली साम्यावस्था $(0,0)$ \omterm{def:b3:ode:stability}{स्थायी} है पर
        स्पर्शोन्मुख रूप से \omterm{def:b3:ode:stability}{स्थायी} \emph{नहीं}। \emph{($V = E + 1$ के
        साथ ल्यापुनोव; अस्पर्शोन्मुखता: ऊर्जा संरक्षण कक्षाओं को
        मूल बिंदु से दूर स्तर वक्रों पर फँसा देता है।)}
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दोलन और उनका आवर्तकाल।} $-1 < E_0 < 1$ स्थिर
कीजिए और $a \in
\intoo0\pi$ (आयाम) के साथ $E_0 = -\cos a$ लिखिए।
\begin{enumerate}[resume]
  \item दर्शाइए कि $x(0) = a$, $y(0) = 0$ वाला हल दोलन करता है: $x(t) \in \intcc{-a}a$, और
        \omterm{def:b3:groups:action}{कक्षा} संवृत वक्र $y^2 = 2(\cos x - \cos a)$ है। यह उचित ठहराइए कि हल आवर्ती
        है: \omterm{def:b3:groups:action}{कक्षा} साम्यावस्थाओं से रहित एक \omterm{def:b3:topology:compact}{संहत} वक्र है जिसे
        नीचे से परिबद्ध चाल पर पार किया जाता है --- इसे एक तर्क
        बनाइए (हल परिमित समय में अपने आरंभिक बिंदु पर लौट आता
        है, और फिर अद्वितीयता आवर्तिता अनिवार्य कर देती है)।
  \item आवर्तकाल सूत्र
        \[
        T(a) = 4\int_0^{a}\frac{\dd x}{\sqrt{2(\cos x -
        \cos a)}}
        \]
        स्थापित कीजिए \emph{(किसी चौथाई \omterm{def:b3:groups:action}{कक्षा} पर $y = \frac{\dd x}{\dd t} >
        0$ और चर
        पृथक कीजिए; $x = a$ पर अनुचित अभिसरण उचित ठहराइए)}।
  \item (छोटे दोलन) $\sin\frac x2 =
        \sin\frac a2\,\sin\varphi$ प्रतिस्थापित कीजिए और
        \[
        T(a) = 4\int_0^{\pi/2}\frac{\dd\varphi}{\sqrt{1 -
        k^2\sin^2\varphi}},
        \qquad k = \sin\frac a2
        \]
        दर्शाइए (एक \omterm{def:b3:complete:complete}{पूर्ण} दीर्घवृत्तीय समाकल)। प्रभावी अभिसरण से
        निष्कर्ष निकालिए कि $a \to 0^+$ पर $T(a) \to 2\pi$: अर्थात् संनादी सीमा,
        जो आयाम से स्वतंत्र है --- गैलीलियो की सन्निकट समकालिकता,
        और उसका यथार्थ संशोधन $T(a) = 2\pi\bigl(1 + \frac{k^2}4 +
        O(k^4)\bigr)$ (समाकल्य का प्रसार कीजिए और
        प्रसामान्य अभिसरण से उचित ठहराते हुए पदशः समाकलन कीजिए)।
  \item दर्शाइए कि $a \to \pi^-$ पर $T(a) \to +\infty$ \emph{($k \to 1$ होने पर $\varphi =
        \frac\pi2$ के
        निकट समाकल्य को नीचे से परिबद्ध कीजिए, अथवा एकदिष्ट
        अभिसरण लगाइए)}: अर्थात् विभाजक के पास पहुँचते हुए लोलक
        बिना किसी सीमा के धीमा होता जाता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- विभाजक।}
\begin{enumerate}[resume]
  \item $E_0 = 1$ के लिए ऊपरी शाखा पर $y = 2\cos\frac x2$: चर पृथक कीजिए और
        समाकलन करके स्पष्ट हल
        \[
        x(t) = 4\arctan\bigl(\eu^{t}\bigr) - \pi
        \]
        ढूँढ़िए ($x(0) = 0$, $y(0) = 2$ के साथ)। सीधे सत्यापित कीजिए कि वह
        लोलक समीकरण हल करता है, और $t \to
        \pm\infty$ पर उसकी तथा $y(t)$ की
        सीमाएँ परिकलित कीजिए।
  \item निष्कर्ष निकालिए: विभाजक \omterm{def:b3:groups:action}{कक्षा} काठी $(-\pi, 0)$ ($t \to -\infty$ पर) को
        काठी $(\pi, 0)$ ($t \to +\infty$ पर) से जोड़ती है, पर किसी तक भी परिमित
        समय में नहीं पहुँचती --- जो अद्वितीयता के अनुरूप है (किसी
        काठी तक परिमित समय में पहुँचना \cref{cor:b3:ode:uniqueness} का खंडन क्यों
        करता?)।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- घूर्णन, और पूरा चित्र।}
\begin{enumerate}[resume]
  \item $E_0 > 1$ के लिए: दर्शाइए कि $y$ कभी लुप्त नहीं होता,
        $x$ कड़ाई से एकदिष्ट और वैश्विक है, $x(t) \to
        \pm\infty$, और $t \mapsto y(t)$
        आवर्तकाल
        \[
        \tau(E_0) = \int_{-\pi}^{\pi}
        \frac{\dd x}{\sqrt{2(E_0 + \cos x)}} .
        \]
        के साथ आवर्ती है।
  \item पूरा प्रावस्था चित्र (अध्याय का चित्र) संकलित कीजिए और
        उसकी प्रत्येक विशेषता को पूरी तरह उचित ठहराइए, तथा विधि
        का दस-पंक्ति सार लिखिए: ऊर्जा, स्तर समुच्चय, संहतता,
        अद्वितीयता --- अध्याय की प्रत्येक प्रमेय कहाँ आई? हमें
        कहीं भी सामान्य हल के किसी सूत्र की आवश्यकता पड़ी?
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- सूक्ष्मदर्शी के नीचे आवर्तकाल फलन।}
\begin{enumerate}[resume]
  \item वालिस आघूर्ण
        \[
        W_n = \int_0^{\pi/2}\sin^{2n}\varphi\,\dd\varphi
        = \frac\pi2\cdot\frac{(2n)!}{4^n\,(n!)^2}
        \]
        आगमन से (खंडशः समाकलन) सिद्ध कीजिए, प्रश्न 6 के समाकल्य
        का द्विपद श्रेणी में प्रसार कीजिए, और पदशः समाकलन उचित
        ठहराकर पूरी श्रेणी
        \[
        T(a) = 2\pi\sum_{n\geq0}
        \Bigl(\frac{(2n)!}{4^n(n!)^2}\Bigr)^{\!2}k^{2n}
        = 2\pi\Bigl(1 + \frac{k^2}4 + \frac{9k^4}{64}
        + O(k^6)\Bigr),
        \qquad k = \sin\frac a2 .
        \]
        प्राप्त कीजिए।
  \item आयाम में बदलिए:
        \[
        T(a) = 2\pi\Bigl(1 + \frac{a^2}{16} +
        \frac{11\,a^4}{3072} + O(a^6)\Bigr)
        \]
        \emph{($\sin\frac a2$ का प्रसार प्रतिस्थापित कीजिए और पद इकट्ठे
        कीजिए)}। समकालिकता कोटि $a^2$ पर विफल हो जाती है, और वह
        विफलता अब कोटि $a^4$ तक मात्रात्मक हो गई है।
  \item दर्शाइए कि $a \mapsto T(a)$ $\intoo0\pi$ पर \omterm{def:b3:topology:continuity}{संतत} और कड़ाई से बढ़ता हुआ है,
        और प्रश्न 6--7 के साथ निष्कर्ष निकालिए कि $T$ $\intoo0\pi$ से
        $\intoo{2\pi}{+\infty}$ पर एक एकैकी आच्छादक संगति है: अर्थात् प्रत्येक
        अधिक्रांतिक आवर्तकाल ठीक एक ही आयाम से साकार होता है।
  \item (घड़ीसाज़ का अंकगणित) लुप्त होते आयाम पर नियमित किया गया
        लोलक \omterm{def:b3:rings:ideal}{आदर्श} समय रखता है; दर्शाइए कि आयाम $a$ पर चलाने से
        वह \omterm{def:b3:rings:ideal}{आदर्श} समय के अंश $\frac{a^2}{16} + O(a^4)$ जितना पिछड़ जाता है, और $a = 0.2$
        रेडियन के लिए बहाव परिकलित कीजिए: लगभग $216$ सेकंड प्रति
        दिन। (हाइगेंस के चक्रज गाल और निर्गम तंत्रों के छोटे
        अचर आयाम, दोनों इसी संख्या के उत्तर हैं।)
  \item प्रश्न 10 के घूर्णन आवर्तकाल $\tau$ पर लौटिए: दर्शाइए कि
        $\tau$ $\intoo1{+\infty}$ पर कड़ाई से घटता हुआ है, कि $E_0 \to 1^+$ पर $\tau(E_0) \to +\infty$
        (एकदिष्ट अभिसरण), और यह कि $E_0 \to
        +\infty$ पर $\sqrt{2E_0}\,\tau(E_0) \to 2\pi$ (प्रभावी
        अभिसरण): अर्थात् तेज चक्करदार घूर्णन स्पर्शोन्मुख रूप से
        कोणीय चाल $\sqrt{2E_0}$ का मुक्त घूर्णन है।
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- विधि का निर्यात: लोत्का--वोल्तेरा।}
लोलक का नुस्खा --- प्रथम समाकल, \omterm{def:b3:topology:compact}{संहत} स्तर वक्र, अद्वितीयता ---
किसी पारितंत्र को भी हल कर देता है। $\alpha, \beta, \gamma, \delta > 0$ स्थिर कीजिए और विवृत
चतुर्थांश $Q = \intoo0{+\infty}\times\intoo0{+\infty}$ पर
\[
x' = x\,(\alpha - \beta y), \qquad
y' = y\,(\delta x - \gamma)
\]
पर विचार कीजिए ($x$ शिकार, $y$ परभक्षी)।
\begin{enumerate}[resume]
  \item दर्शाइए कि $Q$ अपरिवर्ती है --- अक्ष स्पष्ट परिकलनीय
        कक्षाओं के सम्मिलन हैं जिन्हें कोई हल पार नहीं कर सकता
        (\cref{cor:b3:ode:uniqueness}) --- और यह कि $Q$ में एकमात्र साम्यावस्था $(x_*, y_*) =
        \bigl(\frac\gamma\delta, \frac\alpha\beta\bigr)$
        है।
  \item दर्शाइए कि
        \[
        H(x, y) = \delta x - \gamma\ln x + \beta y -
        \alpha\ln y
        \]
        एक प्रथम समाकल है, कि $H = f(x) + g(y)$, जहाँ $f, g$ $\intoo0{+\infty}$ पर कड़ाई से
        उत्तल और उचित है और उसके न्यूनतम $x_*$, $y_*$ पर हैं, और
        निष्कर्ष निकालिए कि $Q$ में सभी \omterm{thm:b3:ode:maximal}{महत्तम हल} वैश्विक हैं।
  \item दर्शाइए कि $h > h_* = H(x_*, y_*)$ के लिए स्तर समुच्चय $\{H = h\}\cap Q$ साम्यावस्था के
        चारों ओर एक संवृत वक्र है: किसी \omterm{def:b3:topology:compact}{संहत} अंतराल $\intcc{x_-}{x_+} \ni x_*$ पर दो
        \omterm{def:b3:topology:continuity}{संतत} शाखाएँ $y_\pm(x)$, जो सिरों पर जुड़ी हों --- अर्थात् लोलक
        के अंडाकारों का समरूप।
  \item सिद्ध कीजिए कि $Q$ में प्रत्येक असाम्य \omterm{def:b3:groups:action}{कक्षा} आवर्ती है:
        रेखाओं $x = x_*$ और $y = y_*$ से कटे चारों क्षेत्रों में
        वामावर्त परिभ्रमण स्थापित कीजिए, प्रत्येक चाप के पार करने
        के समय को अभिसारी वर्गमूल एकलता वाले किसी समाकल से परिबद्ध
        कीजिए (प्रश्न 5 की तरह), और अद्वितीयता से समाप्त कीजिए
        (प्रश्न 4 की तरह)।
  \item (वोल्तेरा का औसतों का नियम) यदि $T$ ऐसी किसी \omterm{def:b3:groups:action}{कक्षा} का
        आवर्तकाल हो, तो दर्शाइए कि
        \[
        \frac1T\int_0^Tx(t)\,\dd t = \frac\gamma\delta,
        \qquad
        \frac1T\int_0^Ty(t)\,\dd t = \frac\alpha\beta :
        \]
        अर्थात् समय-औसत साम्यावस्था मानों के बराबर होते हैं, चाहे
        आयाम कुछ भी हो \emph{(एक आवर्त पर $(\ln x)' =
        \alpha - \beta y$ का समाकलन कीजिए)}।
  \item (मछली पकड़ने का विरोधाभास) दोनों प्रजातियों की दर $\varepsilon \in \intoo0\alpha$
        से कटाई कीजिए: निकाय अपना रूप बनाए रखता है, जिसमें $\alpha$
        और $\gamma$ के स्थान पर $\alpha - \varepsilon$ और $\gamma +
        \varepsilon$ आ जाते हैं। औसत
        जनसंख्याओं का क्या होता है? दांकोना के प्रेक्षण
        (1914--1918) की व्याख्या कीजिए: युद्ध के दौरान जब
        एड्रियाटिक में मछली पकड़ना घटा, तब पकड़ में परभक्षियों
        (शार्क) का अनुपात \emph{बढ़ गया} --- और यह भी कि मध्यम
        मछली पकड़ना शिकार को क्यों लाभ पहुँचाता है।
  \item दस-पंक्ति सार लिखिए: प्रत्येक चरण को अध्याय की कौन-सी
        प्रमेय चलाती है, लोलक की ऊर्जा का स्थान क्या लेता है, और
        दोनों में से किसी निकाय को किसी प्रारंभिक संवृत-रूप हल की
        आवश्यकता क्यों नहीं पड़ी --- और वह स्वीकार्य भी क्यों
        नहीं।

  \item (चाल का मॉडुलन) घूर्णन व्यवस्था $E_0 > 1$ में दर्शाइए कि
        $y = x'$ $\sqrt{2(E_0 - 1)}$ ($x \equiv \pi \bmod 2\pi$ पर) और $\sqrt{2(E_0 + 1)}$ ($x \equiv 0$ पर) के बीच दोलन
        करता है, कि एक आवर्त पर $y$ का समय-औसत ठीक $\frac{2\pi}{\tau(E_0)}$ है,
        और यह कि $E_0
        \to \infty$ पर मॉडुलन अनुपात $\sqrt{\frac{E_0 + 1}{E_0 - 1}} \to 1$: अर्थात् तेज घूर्णन
        स्पर्शोन्मुख रूप से एकसमान होता है।
  \item (घूर्णन आवर्तकाल की एकदिष्टता) दर्शाइए कि $\tau(E_0)$ $\mathcal C^1$ है
        और $\intoo1{+\infty}$ पर कड़ाई से घटता हुआ \emph{(समाकल चिह्न के भीतर
        अवकलन कीजिए, प्रत्येक $\intco{1 + \delta}\infty$ पर प्रभुत्व के साथ)}, जिसमें
        $E_0 \to 1^+$ पर $\tau \to \infty$ और $E_0 \to \infty$ पर $\tau
        \to 0$। ऊर्जा अक्ष के अनुदिश
        लोलक का पूरा द्विशाखन चित्र संकलित कीजिए: $E_0 = -1$ पर
        साम्यावस्थाएँ, $-1 < E_0 < 1$ पर $2\pi$ से $\infty$ तक बढ़ते आवर्तकाल
        वाले दोलन, $E_0 = 1$ पर विभाजक, और उसके आगे $\infty$ से $0$ तक
        घटते आवर्तकाल वाले घूर्णन।
\end{enumerate}
\end{problem}
