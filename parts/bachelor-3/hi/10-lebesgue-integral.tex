\chapter{लेबेग समाकल}\label{ch:b3:lebesgue}

रीमान का समाकल \emph{प्रांत} को छोटे अंतरालों में काटता है;
लेबेग का \emph{परिसर} को काटता है: $f$ का समाकलन करने के लिए
समुच्चयों $\{f > t\}$ का \omterm{def:b3:measure:measure}{माप} लीजिए। यह परिवर्तन देखने में निर्दोष लगता
है और है क्रांतिकारी। सीमाएँ और समाकल, जो रीमान सिद्धांत में सदा
लड़ते रहते थे (एकसमान अभिसरण चाहिए!), तीन अभिसरण प्रमेयों से मेल
कर लेते हैं --- एकदिष्ट अभिसरण, फतू, प्रभावी अभिसरण --- जिनकी
परिकल्पनाएँ लगभग लज्जाजनक रूप से दुर्बल हैं। यह अध्याय किसी भी \omterm{def:b3:measure:measure}{माप
समष्टि} $(X, \mathcal A,
\mu)$ पर समाकल की रचना करता है, तीनों प्रमेय सिद्ध करता है,
रीमान के समाकल के साथ ठीक-ठीक संबंध तय करता है (कोई परिबद्ध फलन
रीमान-समाकलनीय है तभी और केवल तभी जब वह \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र} \omterm{def:b3:topology:continuity}{संतत} हो), और
प्राचल-निर्भर समाकलों के अवकलन को औद्योगिक रूप दे देता है --- वही
तकनीक जिससे सप्ताहांत समस्या $\int_0^\infty\frac{\sin x}x\,\dd x$ और $\int_\R \eu^{-x^2}\dd x$ परिकलित करती है।

\section{मापनीय फलन}

\begin{definition}\label{def:b3:lebesgue:measurable}
मान लीजिए $(X, \mathcal A)$, $(Y, \mathcal B)$ मापनीय समष्टियाँ हैं। $f \colon X \to Y$
\emph{मापनीय}\index{मापनीय फलन} कहलाता है यदि प्रत्येक $B \in
\mathcal B$ के
लिए $f^{-1}(B) \in \mathcal A$ हो। वास्तविक (या $[-\infty,+\infty]$-मान वाले) फलनों के लिए $Y = \R$ अपना
बोरेल $\sigma$-बीजगणित धारण करता है, और सभी $t \in \R$ के लिए $f^{-1}(\intoo t{+\infty}) = \{f > t\} \in
\mathcal A$ जाँच
लेना पर्याप्त है: अच्छे समुच्चय $\{B :
f^{-1}(B) \in \mathcal A\}$ जनक किरणों को अंतर्विष्ट
करने वाला $\sigma$-बीजगणित बनाते हैं (पूर्वप्रतिबिंब समुच्चय
संक्रियाओं के साथ क्रमविनिमेय हैं) (\cref{def:b3:measure:borel}, \cref{met:b3:measure:goodsets})।
\end{definition}

\begin{proposition}\label{prop:b3:lebesgue:stability}
(क) \omterm{def:b3:lebesgue:measurable}{मापनीय} प्रतिचित्रणों के संयोजन \omterm{def:b3:lebesgue:measurable}{मापनीय} होते हैं; और \omterm{def:b3:topology:continuity}{संतत
प्रतिचित्रण} बोरेल-मापनीय होते हैं।
(ख) यदि $f, g \colon X \to \R$ \omterm{def:b3:lebesgue:measurable}{मापनीय} हों, तो $f + g$, $fg$, $\max(f,g)$, $\abs f$, $\lambda f$ भी।
(ग) यदि $(f_n)$ $[-\infty, +\infty]$ में मान लेने वाले \omterm{def:b3:lebesgue:measurable}{मापनीय फलन} हों, तो $\sup_nf_n$,
$\inf_nf_n$, $\limsup f_n$, $\liminf f_n$ \omterm{def:b3:lebesgue:measurable}{मापनीय} हैं; और यदि बिंदुशः $f_n \to f$ हो, तो $f$
\omterm{def:b3:lebesgue:measurable}{मापनीय} है।
\end{proposition}

\begin{proof}
(क) $(g\circ f)^{-1}(B) = f^{-1}(g^{-1}(B))$; और \omterm{def:b3:topology:continuity}{सांतत्य} जनक विवृत समुच्चयों के माध्यम से \omterm{def:b3:lebesgue:measurable}{मापनीयता} दे
देता है (\cref{pb:b3:measure:1}, प्रश्न 10, व्यापक रूप में)।
(ख) $(f, g) \colon X \to \R^2$ $\R^2$ के बोरेल $\sigma$-बीजगणित के लिए \omterm{def:b3:lebesgue:measurable}{मापनीय} है --- विवृत
बक्सों पर जाँचिए, जो जनक हैं ($\R^2$ के \omterm{def:b3:topology:topology}{विवृत समुच्चय} परिमेय बक्सों
के गणनीय सम्मिलन हैं): $(f,g)^{-1}(U\times V) = f^{-1}(U)\cap g^{-1}(V)$ --- और $+, \times, \max$ \omterm{def:b3:topology:continuity}{संतत} $\R^2 \to \R$ हैं: अब संयोजन
कीजिए।
(ग) $\{\sup f_n > t\} = \bigcup_n\{f_n > t\}$; $\inf = -\sup(-)$; $\limsup = \inf_N\sup_{n \geq N}$; और कोई बिंदुशः सीमा अपना ही $\limsup$ होती है।
\end{proof}

\begin{definition}\label{def:b3:lebesgue:simple}
\emph{सरल फलन}\index{सरल फलन} परिमित संख्या के मान लेने वाला
\omterm{def:b3:lebesgue:measurable}{मापनीय फलन} है: $s = \sum_{i=1}^n
c_i\,\mathbf 1_{A_i}$, जहाँ $A_i \in \mathcal A$ असंयुक्त हैं और $c_i \geq
0$ (अऋणात्मक
सिद्धांत के लिए)। उसका समाकल है
\[
\int s\,\dd\mu = \sum_i c_i\,\mu(A_i) \in [0, +\infty]
\]
(परिपाटी $0\cdot\infty = 0$); और यह मान \omterm{def:b3:representations:rep}{निरूपण} पर निर्भर नहीं करता (दो
विभाजनों को परिष्कृत कीजिए)।
\end{definition}

\begin{theorem}[सरल फलनों से आसन्नन]
\label{thm:b3:lebesgue:approximation}
प्रत्येक \omterm{def:b3:lebesgue:measurable}{मापनीय} $f \colon X \to [0, +\infty]$ सरल फलनों के किसी \emph{बढ़ते} अनुक्रम की
बिंदुशः सीमा है:
\[
s_n = \sum_{k=1}^{n2^n} \frac{k-1}{2^n}\,
\mathbf 1_{\{\frac{k-1}{2^n} \leq f < \frac k{2^n}\}}
+ n\,\mathbf 1_{\{f \geq n\}} \nearrow f .
\]
\end{theorem}

\begin{proof}
प्रत्येक $s_n$ सरल है (समुच्चय बोरेल समुच्चयों के पूर्वप्रतिबिंब
हैं)। एकदिष्टता: $n$ से $n+1$ पर जाने से प्रत्येक द्विआधारी स्तर
दो में बँट जाता है और सौंपा गया मान कभी घटता नहीं ($\frac{k-1}{2^n} \leq f(x) < \frac k{2^n}$ वाले किसी
बिंदु को या तो $\frac{2k-2}{2^{n+1}}$ मिलता है या $\frac{2k-1}{2^{n+1}}$, और दोनों $\geq
\frac{k-1}{2^n}$ हैं; ऊपरी
सीमा $n$ भी उठती है)। अभिसरण: यदि $f(x) < \infty$ हो, तो $n > f(x)$ के लिए हमें
$f(x) - s_n(x) \leq
2^{-n}$ मिलता है; और यदि $f(x) = \infty$ हो, तो $s_n(x) = n \to \infty$।
\end{proof}

\section{समाकल और अभिसरण प्रमेय}

\begin{definition}\label{def:b3:lebesgue:integral}
\omterm{def:b3:lebesgue:measurable}{मापनीय} $f \geq 0$ के लिए:
\[
\int f \,\dd\mu = \sup\Bigl\{\int s\,\dd\mu : s \text{ सरल},
\ 0 \leq s \leq f\Bigr\} \in [0, +\infty].
\]
रचना से यह $f$ में एकदिष्ट है, और सरल स्थिति का विस्तार है (सरल
$f$ के लिए उच्चतम $f$ पर प्राप्त हो जाता है: उभयनिष्ठ
परिष्करणों के माध्यम से सरल समाकलों की तुलना)।
\end{definition}

\begin{theorem}[एकदिष्ट अभिसरण, बेप्पो लेवी]
\label{thm:b3:lebesgue:mct}
यदि बिंदुशः $0 \leq f_n \nearrow f$ हो (\omterm{def:b3:lebesgue:measurable}{मापनीय}), तो
\[
\int f_n\,\dd\mu \nearrow \int f\,\dd\mu .
\]
\index{एकदिष्ट अभिसरण प्रमेय}
\end{theorem}

\begin{proof}
$f$ \omterm{def:b3:lebesgue:measurable}{मापनीय} है (\cref{prop:b3:lebesgue:stability}(ग)) और $\int f_n$ बढ़कर किसी $L \leq \int f$ तक जाता है
(एकदिष्टता)। विलोमतः, कोई सरल $s = \sum c_i\mathbf 1_{A_i} \leq f$ और $\theta \in (0,1)$ स्थिर कीजिए; समुच्चय
$E_n = \{f_n \geq \theta s\}$ \omterm{def:b3:lebesgue:measurable}{मापनीय} हैं और बढ़कर $X$ तक जाते हैं (जहाँ $s(x) > 0$: $f(x)
\geq s(x) > \theta s(x)$, अतः
अंततः $f_n(x) \geq \theta
s(x)$; और जहाँ $s(x) = 0$: तुच्छ रूप से)। तब नीचे से \omterm{def:b3:topology:continuity}{सांतत्य}
(\cref{prop:b3:measure:basics}(ग)) से
\[
\int f_n \geq \int_{E_n}\theta s\,\dd\mu
= \theta\sum_i c_i\,\mu(A_i \cap E_n)
\xrightarrow[n\to\infty]{} \theta\sum_ic_i\,\mu(A_i)
= \theta\int s
\]
अतः सभी $\theta < 1$ और सभी सरल $s
\leq f$ के लिए $L \geq \theta\int s$: इसलिए $L \geq \int f$।
\end{proof}

\begin{corollary}\label{cor:b3:lebesgue:additivity}
\omterm{def:b3:lebesgue:measurable}{मापनीय} $f, g \geq 0$ और $c \geq 0$ के लिए: $\int(f + g) =
\int f + \int g$ और $\int cf = c\int f$; और अऋणात्मक \omterm{def:b3:lebesgue:measurable}{मापनीय}
फलनों की किसी श्रेणी के लिए $\int\sum_nf_n =
\sum_n\int f_n$।
\end{corollary}

\begin{proof}
सरल फलनों के लिए योज्यता किसी उभयनिष्ठ परिष्करण पर एक परिकलन है।
व्यापक स्थिति में $s_n \nearrow f$, $t_n \nearrow g$ लीजिए (\cref{thm:b3:lebesgue:approximation}): $s_n + t_n \nearrow f +
g$, और एकदिष्ट
अभिसरण प्रमेय योज्यता को सीमा तक ले जाती है। श्रेणी वाला कथन आंशिक
योगों पर लगाई गई एकदिष्ट अभिसरण प्रमेय है।
\end{proof}

\begin{theorem}[फतू की प्रमेयिका]\label{thm:b3:lebesgue:fatou}
\omterm{def:b3:lebesgue:measurable}{मापनीय} $f_n \geq 0$ के लिए:
\[
\int \liminf_n f_n \,\dd\mu \;\leq\; \liminf_n \int
f_n\,\dd\mu .
\]
\index{फतू की प्रमेयिका}
\end{theorem}

\begin{proof}
मान लीजिए $g_N = \inf_{n\geq N}f_n$: यह \omterm{def:b3:lebesgue:measurable}{मापनीय} है, $0 \leq g_N \nearrow
\liminf f_n$, और प्रत्येक $n \geq N$ के लिए
$g_N \leq f_n$, अतः $\int g_N \leq \inf_{n \geq N}\int f_n$। बाईं ओर एकदिष्ट अभिसरण प्रमेय लगाइए: $\int\liminf f_n = \lim_N\int g_N \leq
\lim_N\inf_{n\geq N}\int f_n = \liminf\int f_n$।
\end{proof}

\begin{definition}\label{def:b3:lebesgue:l1}
कोई \omterm{def:b3:lebesgue:measurable}{मापनीय} $f \colon X \to \R$ (या $\C$) \emph{समाकलनीय}\index{समाकलनीय फलन}
कहलाता है यदि $\int\abs
f\,\dd\mu < \infty$ हो; तब $\int f = \int f^+ - \int f^-$ (धनात्मक और ऋणात्मक भाग;
सम्मिश्र स्थिति में वास्तविक और काल्पनिक भाग)। समाकल समाकलनीय
फलनों पर रैखिक है (विघटित कीजिए और धनात्मक भागों को पुनः जोड़िए;
सम्मिश्र स्थिति वास्तविक स्थिति तक सिमट जाती है) और $\abs{\int f} \leq
\int\abs f$ को
संतुष्ट करता है (वास्तविक स्थिति: $\pm\int f = \int(\pm f) \leq
\int\abs f$; सम्मिश्र स्थिति: समाकल
को वास्तविक बनाने के लिए किसी इकाई-मापांक अचर से गुणा कीजिए)। कोई
गुण \emph{लगभग सर्वत्र}\index{लगभग सर्वत्र} लागू होता है यदि वह
केवल किसी $\mu$-शून्य समुच्चय पर विफल हो; किसी शून्य समुच्चय पर
$f$ को बदलने से कोई समाकल नहीं बदलता (अंतर $\infty\cdot\mathbf
1_N$ से प्रभावित है,
जिसका समाकल $0$ है)।
\end{definition}

\begin{theorem}[प्रभावी अभिसरण]\label{thm:b3:lebesgue:dct}
मान लीजिए \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र} $f_n \to f$, और किसी स्थिर \emph{\omterm{def:b3:lebesgue:l1}{समाकलनीय}}
$g$ के लिए \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र} $\abs{f_n} \leq g$। तब $f$ \omterm{def:b3:lebesgue:l1}{समाकलनीय} है और
\[
\int f_n\,\dd\mu \longrightarrow \int f\,\dd\mu,
\qquad\text{वस्तुतः}\quad \int\abs{f_n - f}\,\dd\mu \to 0 .
\]
\index{प्रभावी अभिसरण प्रमेय}
\end{theorem}

\begin{proof}
परिकल्पनाओं को बिंदुशः बनाने के लिए एक शून्य समुच्चय छोड़ दीजिए।
$\abs f
\leq g$: अतः $f$ \omterm{def:b3:lebesgue:l1}{समाकलनीय} है। फलन $h_n = 2g - \abs{f_n
- f} \geq 0$ $\liminf h_n = 2g$ को संतुष्ट करते हैं;
फतू से
\[
\int 2g \leq \liminf\int\bigl(2g - \abs{f_n - f}\bigr)
= \int 2g - \limsup\int\abs{f_n - f},
\]
अतः $\limsup\int\abs{f_n - f} \leq 0$ (घटाव वैध है: $\int 2g < \infty$)। अंततः $\abs{\int f_n - \int f} \leq
\int\abs{f_n - f} \to 0$।
\end{proof}

\begin{method}\label{met:b3:lebesgue:convergence}
$\lim_n\int f_n$ का सामना होने पर क्रम से आज़माइए --- (1) क्या अनुक्रम
एकदिष्ट है (अथवा अऋणात्मक पदों की श्रेणी)? तब एकदिष्ट अभिसरण
प्रमेय, \omterm{def:b3:lebesgue:l1}{समाकलनीयता} की कोई आवश्यकता नहीं। (2) क्या कोई एक ही
\omterm{def:b3:lebesgue:l1}{समाकलनीय} प्रभावी फलन $g \geq \abs{f_n}$ है, जो मोटे परिबंधों से मिल जाए
(आकलनों को ``$\sup_n$'' कर दीजिए)? तब प्रभावी अभिसरण प्रमेय। (3)
न प्रभुत्व, न एकदिष्टता? तब भी फतू एक ओर को परिबद्ध कर देती है,
और समता वास्तव में विफल हो सकती है: भागती हुई उभार $f_n = n\mathbf
1_{\intoo0{1/n}}$ के लिए
$\int f_n = 1$ है पर \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र} $f_n \to 0$। प्रभुत्व ठीक वही है जो द्रव्यमान
को अनंत की ओर --- ऊर्ध्वाधर या क्षैतिज --- भागने से रोकता है।
\end{method}

\section{रीमान बनाम लेबेग}

\begin{theorem}[लेबेग की कसौटी]
\label{thm:b3:lebesgue:riemann}
मान लीजिए $f \colon \intcc ab \to \R$ \emph{परिबद्ध} है। तब $f$ रीमान-समाकलनीय है तभी
और केवल तभी जब $f$ $\lambda$-लगभग सर्वत्र \omterm{def:b3:topology:continuity}{संतत} हो; और उस स्थिति में
$f$ लेबेग-समाकलनीय है और दोनों समाकल एक ही होते हैं।\index{रीमान
बनाम लेबेग समाकल}
\end{theorem}

\begin{proof}
किसी उपविभाजन $\sigma = (a = x_0 < \dots < x_N = b)$ के लिए मान लीजिए $L_\sigma$ और $U_\sigma$ वे सोपान फलन
हैं जो प्रत्येक $\intoo{x_{i-1}}{x_i}$ पर क्रमशः $m_i = \inf_{[x_{i-1}, x_i]}f$ और $M_i = \sup$ के बराबर हैं;
दारबू योग उन्हीं के समाकल हैं (सोपान फलनों पर रीमान और लेबेग सहमत
हैं, और दोनों $\sum
m_i\Delta x_i$ देते हैं)। उपविभाजनों का ऐसा अनुक्रम $\sigma_n$
लीजिए जिसमें प्रत्येक पिछले का परिष्करण हो, जिनकी सूक्ष्मता $\to 0$
हो, और जिनके दारबू योग $f$ के अधो और उच्च दारबू समाकलों पर
अभिसरित हों। परिष्करण $L_{\sigma_n}$ को अनवरोही और $U_{\sigma_n}$ को अनवर्धी बना
देते हैं, समस्त विभाजन बिंदुओं के गणनीय समुच्चय $D$ के बाहर
बिंदुशः; सीमाओं को $\ell$ और $u$ कहिए (जो \omterm{def:b3:lebesgue:measurable}{मापनीय} हैं, \cref{prop:b3:lebesgue:stability})।
$x \notin
D$ के लिए, $x$ को अंतर्विष्ट करने वाले विवृत $\sigma_n$-अंतराल के
लिए $I_n(x)$ लिखने पर: $\ell(x) = \sup_n\inf_{I_n(x)}f$ और $u(x) =
\inf_n\sup_{I_n(x)}f$; और चूँकि सूक्ष्मताएँ $0$
तक सिकुड़ती हैं, ये $x$ पर $f$ के अधो और उच्च \emph{आवरक} हैं
--- $u(x) - \ell(x)$ $x$ पर $f$ का दोलन है --- अतः \emph{$\ell(x) = u(x)$ तभी और
केवल तभी जब $f$ $x$ पर \omterm{def:b3:topology:continuity}{संतत} हो}। एकदिष्ट/प्रभावी अभिसरण
प्रमेय से (परिबद्ध, परिमित अंतराल):
\[
\int_{\intcc ab}\ell\,\dd\lambda = \lim_n\int L_{\sigma_n}
= \underline{\int}f,
\qquad
\int_{\intcc ab}u\,\dd\lambda = \overline{\int}f .
\]
$f$ रीमान-समाकलनीय $\iff$ $\underline\int f =
\overline\int f$ $\iff$ $\int(u - \ell) = 0$ $\iff$ \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र}
$u = \ell$ ($u - \ell \geq 0$; \cref{exo:b3:lebesgue:5}) $\iff$ $f$ \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र} \omterm{def:b3:topology:continuity}{संतत}। उस स्थिति में
\omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र} $\ell =
u$ के साथ $\ell \leq f \leq u$: अतः $f$ \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र} \omterm{def:b3:lebesgue:measurable}{मापनीय}
$\ell$ के बराबर है, और इसलिए $\int f\,\dd\lambda = \int\ell\,\dd\lambda = \underline\int f =
\int_a^bf$ के साथ लेबेग-मापनीय है ($\lambda$ की
\omterm{def:b3:complete:complete}{पूर्णता})।
\end{proof}

\begin{example}\label{ex:b3:lebesgue:riemannexamples}
$\mathbf 1_\Q$ कहीं भी \omterm{def:b3:topology:continuity}{संतत} नहीं है: अतः रीमान-समाकलनीय नहीं --- पर लेबेग
की दृष्टि में तुच्छ: $\int\mathbf 1_\Q\,\dd\lambda =
\lambda(\Q) = 0$। टोमे का फलन (परिमेय $\frac pq$ पर $\frac1q$,
अन्यत्र $0$) ठीक अपरिमेयों पर \omterm{def:b3:topology:continuity}{संतत} है: अतः रीमान-समाकलनीय, और
उसका समाकल $0$ है। और \emph{अनुचित} रीमान समाकल एक भिन्न धारणा
हैं: $\int_0^\infty\frac{\sin x}x\,\dd x$ $\int_0^A$ की सीमा के रूप में अभिसरित होता है (सप्ताहांत
समस्या उसे $= \frac\pi2$ परिकलित करती है), परंतु $\frac{\sin x}x \notin L^1(\intoo0{+\infty})$: निरपेक्ष समाकल
हरात्मक श्रेणी की तरह अपसरित होता है (\cref{exo:b3:lebesgue:6})। लेबेग का सिद्धांत
सप्रतिबंध अभिसरण छोड़कर दृढ़ सीमा प्रमेय ले लेता है।
\end{example}

\section{प्राचल वाले समाकल}

आगे सर्वत्र $(X, \mathcal A, \mu)$ एक \omterm{def:b3:measure:measure}{माप समष्टि} है, $T$ एक मापीय समष्टि (प्राचल),
और $f \colon T \times X \to \C$ ऐसा है कि प्रत्येक $t$ के लिए $f(t, \cdot)$ \omterm{def:b3:lebesgue:l1}{समाकलनीय} है;
$F(t) = \int_X
f(t, x)\,\dd\mu(x)$ रखिए।

\begin{theorem}[सांतत्य]\label{thm:b3:lebesgue:paramcont}
मान लीजिए: लगभग प्रत्येक $x$ के लिए $t \mapsto f(t,x)$ $t_0$ पर \omterm{def:b3:topology:continuity}{संतत} है, और
कोई ऐसा \omterm{def:b3:lebesgue:l1}{समाकलनीय} $g$ है कि $t_0$ के किसी \omterm{def:b3:topology:topology}{प्रतिवेश} के सभी $t$
और लगभग प्रत्येक $x$ के लिए $\abs{f(t,x)} \leq
g(x)$ हो। तब $F$ $t_0$ पर \omterm{def:b3:topology:continuity}{संतत}
है।\index{प्राचल समाकल}
\end{theorem}

\begin{proof}
किसी भी अनुक्रम $t_n \to t_0$ के लिए: \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र} $f(t_n, \cdot) \to f(t_0,
\cdot)$, और $g$ से
प्रभावित: अतः प्रभावी अभिसरण प्रमेय $F(t_n) \to F(t_0)$ दे देती है; और मापीय
समष्टियों में अनुक्रमीय \omterm{def:b3:topology:continuity}{सांतत्य} पर्याप्त है (\cref{rem:b3:topology:sequences})।
\end{proof}

\begin{theorem}[समाकल के भीतर अवकलन]
\label{thm:b3:lebesgue:paramdiff}
मान लीजिए $T$ $\R$ का कोई विवृत अंतराल है। मान लीजिए: लगभग
प्रत्येक $x$ के लिए $t \mapsto f(t,x)$ $T$ पर अवकलनीय है, और
\[
\Bigl|\frac{\partial f}{\partial t}(t, x)\Bigr| \leq g(x)
\quad \text{सभी } t \in T \text{ के लिए},\ \text{लगभग प्रत्येक } x,
\]
जहाँ $g$ \omterm{def:b3:lebesgue:l1}{समाकलनीय} है। तब $F$ $T$ पर अवकलनीय है और $F'(t)
= \int_X \frac{\partial f}{\partial t}(t, x)\,\dd\mu(x)$।
\end{theorem}

\begin{proof}
$t$ और $h_n \to 0$ स्थिर कीजिए: अंतर भागफल
\[
\varphi_n(x) = \frac{f(t + h_n, x) - f(t, x)}{h_n}
\longrightarrow \frac{\partial f}{\partial t}(t,x)
\quad\text{लगभग सर्वत्र},
\]
और माध्य मान असमिका $\abs{\varphi_n(x)} \leq
\sup_{s}\abs{\partial_tf(s,x)} \leq g(x)$ को परिबद्ध कर देती है: अतः प्रभावी
अभिसरण प्रमेय लागू होती है, और $\frac{F(t + h_n) - F(t)}{h_n} = \int\varphi_n \to
\int\partial_t f(t, \cdot)$।
\end{proof}

\begin{example}[गामा फलन]\label{ex:b3:lebesgue:gamma}
$t > 0$ के लिए मान लीजिए\index{गामा फलन}
\[
\Gamma(t) = \int_0^{+\infty} x^{t-1}\eu^{-x}\,\dd x .
\]
यह समाकल अभिसरित होता है: $0$ के निकट $x^{t-1}$ \omterm{def:b3:lebesgue:l1}{समाकलनीय} है ($t >
0$);
और अनंत पर $x^{t-1}\eu^{-x} \leq C\eu^{-x/2}$। खंडशः समाकलन ($[\varepsilon, A]$ पर, फिर एकदिष्ट अभिसरण
प्रमेय से सीमाएँ) कार्यकीय समीकरण $\Gamma(t + 1) =
t\,\Gamma(t)$ दे देता है, जिससे $\Gamma(n+1) = n!$:
अर्थात् क्रमगुणित का अंतर्वेशन। प्रत्येक $\intcc ab \subseteq \intoo0{+\infty}$ पर $\partial_t\bigl(x^{t-1}\eu^{-x}\bigr) = \ln
x\cdot x^{t-1}\eu^{-x}$ \omterm{def:b3:lebesgue:l1}{समाकलनीय}
$\abs{\ln x}(x^{a-1} +
x^{b-1})\eu^{-x}$ से प्रभावित है: अतः $\Gamma$ $\mathcal C^1$ है, और आगमन से $\mathcal C^\infty$, जिसमें
$\Gamma^{(k)}(t) =
\int_0^\infty(\ln x)^kx^{t-1}\eu^{-x}\dd x$। मान $\Gamma(\frac12) = \sqrt\pi$ छद्म वेश में गाउसीय समाकल है (\cref{pb:b3:lebesgue:1})।
\end{example}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lebesgue:11}
(शेफे की प्रमेयिका) मान लीजिए $f_n, f \geq 0$ \omterm{def:b3:lebesgue:l1}{समाकलनीय} हैं, \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र}
$f_n \to f$, और $\int f_n \to \int f$।
(क) दर्शाइए कि $\int\abs{f_n - f} \to 0$। \emph{($g_n = (f - f_n)^+ \leq f$ पर प्रभावी अभिसरण लगाइए, और
$\int\abs{f_n - f} = 2\int g_n - \int(f - f_n)$ लिखिए।)}
(ख) उदाहरण से दर्शाइए कि परिकल्पना $\int f_n \to \int
f$ हटाई नहीं जा सकती
(कोई सरकती या संकेंद्रित होती उभार), और यह कि निरपेक्ष-मान नियंत्रण
के बिना चिह्नित $f_n$ के लिए निष्कर्ष विफल हो जाता है: $f_n = n\mathbf 1_{\intoc0{1/n}} -
n\mathbf 1_{\intoc{-1/n}0}$ के
लिए \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र} $f_n \to 0$, $\int f_n
= 0 \to 0$, फिर भी $\int\abs{f_n} = 2$।
(ग) अनुप्रयोग (\omterm{ex:b3:lebesgue:gamma}{घनत्व}): यदि प्रायिकता \omterm{ex:b3:lebesgue:gamma}{घनत्व} \omterm{def:b3:lebesgue:l1}{लगभग सर्वत्र} $p_n
\to p$
हों, तो स्वतः $\int\abs{p_n - p} \to 0$: घनत्वों का बिंदुशः अभिसरण $L^1$ अभिसरण होता
है --- मुफ्त में अभिसरण का उन्नयन।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lebesgue:12}
एकदिष्ट/प्रभावी अभिसरण प्रमेय के माध्यम से पूरे औचित्य के साथ
शास्त्रीय सीमाएँ:
\[
\text{(क)}\ \lim_{n\to\infty}\int_0^n\Bigl(1 -
\frac xn\Bigr)^n\eu^{x/2}\,\dd x, \qquad
\text{(ख)}\ \lim_{n\to\infty}\int_0^1\frac{n\,x^{n-1}}{1 +
x}\,\dd x,
\]
\[
\text{(ग)}\ \lim_{n\to\infty}\int_0^\infty
\frac{\dd x}{(1 + x/n)^n\,x^{1/n}} .
\]
\emph{((क) के लिए: स्थिर $x$ पर $(1 - x/n)^n \nearrow \eu^{-x}$ --- एकदिष्टता $\log$ से
सिद्ध कीजिए; (ख) के लिए खंडशः समाकलन कीजिए अथवा $x = u^{1/n}$ प्रतिस्थापित
करके किसी सीमांत संकेंद्रण को पहचानिए; (ग) के लिए $x =
1$ पर बाँटकर
सभी $n \geq 2$ के लिए वैध कोई \omterm{def:b3:lebesgue:l1}{समाकलनीय} प्रभावी फलन ढूँढ़िए।)}
\end{exercise}

\section{समस्या: दो प्रसिद्ध समाकल}

\begin{problem}[{सप्ताहांत समस्या --- गाउसीय समाकल और दिरिक्ले का
समाकल, केवल प्राचलों से}]
\label{pb:b3:lebesgue:1}
दो समाकल अनुप्रयुक्त विश्लेषण पर राज करते हैं:
\[
G = \int_{-\infty}^{+\infty}\eu^{-x^2}\dd x = \sqrt\pi,
\qquad
D = \int_0^{+\infty}\frac{\sin x}{x}\,\dd x = \frac\pi2
\]
(दूसरा अनुचित समाकल के रूप में, \cref{ex:b3:lebesgue:riemannexamples})। हम दोनों को केवल इस
अध्याय के औजारों से सिद्ध करते हैं।

\medskip
\noindent\textbf{भाग I --- गाउसीय।} $t \geq 0$ के लिए रखिए
\[
A(t) = \Bigl(\int_0^t\eu^{-x^2}\dd x\Bigr)^{2},
\qquad
B(t) = \int_0^1\frac{\eu^{-t^2(1 + x^2)}}{1 + x^2}\,\dd x .
\]
\begin{enumerate}
  \item उचित ठहराइए कि $A$ और $B$ $\intoo0{+\infty}$ पर $\mathcal C^1$ हैं, $A'$
        और $B'$ परिकलित कीजिए, और $A'(t) + B'(t) = 0$ दर्शाइए। \emph{($B'$
        में $u =
        tx$ प्रतिस्थापित कीजिए।)}
  \item $A(0) + B(0)$ और $\lim_{t\to+\infty}(A + B)(t)$ परिकलित कीजिए --- $B$ में समाकल के भीतर
        सीमा को उचित ठहराइए।
  \item $\int_0^\infty \eu^{-x^2}\dd x =
        \frac{\sqrt\pi}2$ निष्कर्ष निकालिए, अतः $G = \sqrt\pi$, और $\Gamma(\tfrac12) = \sqrt\pi$ भी \emph{($\Gamma(\frac12)$
        में $x =
        u^2$ प्रतिस्थापित कीजिए)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दिरिक्ले का समाकल।} $t
\geq 0$ के लिए
रखिए
\[
F(t) = \int_0^{+\infty}\eu^{-tx}\,\frac{\sin x}{x}\,\dd x .
\]
\begin{enumerate}[resume]
  \item दर्शाइए कि $F(t)$ को परिभाषित करने वाला समाकल प्रत्येक
        $t > 0$ के लिए लेबेग समाकल के रूप में अभिसरित होता है, और
        $t = 0$ के लिए अनुचित समाकल के रूप में; दर्शाइए कि $D =
        \lim_{A\to\infty}\int_0^A\frac{\sin x}x\dd x$
        विद्यमान है \emph{($[\pi, A]$ पर खंडशः समाकलन कीजिए)}।
  \item दर्शाइए कि $F$ $\intoo0{+\infty}$ पर $\mathcal C^1$ है, जिसमें
        \[
        F'(t) = -\int_0^{+\infty}\eu^{-tx}\sin x\,\dd x
        = -\frac{1}{1 + t^2}
        \]
        \emph{(प्रत्येक $t_0 >
        0$ के लिए $[t_0, \infty)$ पर प्रभुत्व; अंतिम
        समाकल दो बार खंडशः समाकलन से अथवा सम्मिश्र चरघातांकियों
        से)}।
  \item दर्शाइए कि $t \to +\infty$ पर $F(t) \to 0$, और $\intoo0{+\infty}$ पर $F(t)
        = \frac\pi2 - \arctan t$ निष्कर्ष
        निकालिए।
  \item नाजुक बिंदु: $D = \lim_{t\to0^+}F(t)$। इसे पुच्छ के एकसमान नियंत्रण से सिद्ध
        कीजिए: $0 \leq t \leq 1$ और $A \geq \pi$ के लिए खंडशः समाकलन करके दर्शाइए कि
        \[
        \Bigl|\int_A^{+\infty}\eu^{-tx}\frac{\sin
        x}x\,\dd x\Bigr| \leq \frac{C}{A}
        \]
        जहाँ $C$ $t$ से स्वतंत्र है \emph{($\frac{\eu^{-tx}}x$ का अवकलन
        कीजिए और $\abs{\cos}$ को $1$ से परिबद्ध कीजिए; ध्यान दीजिए कि
        $\sup_{u\geq0}u\eu^{-u} < 1$ के साथ $t\eu^{-tx} \leq 1/x\cdot(tx\eu^{-tx})$)}; फिर $F(t) - D$ को $[0, A]$ (जहाँ $t \to 0$ पर
        प्रभावी अभिसरण प्रमेय लागू होती है) और $[A, \infty)$ में बाँटिए।
  \item निष्कर्ष निकालिए: $D = \frac\pi2$।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- लाभांश।}
\begin{enumerate}[resume]
  \item $\int_0^{+\infty}\frac{\sin^2x}{x^2}\,\dd x$ परिकलित कीजिए \emph{(खंडशः समाकलन कीजिए और $\sin
        2x = 2\sin x\cos x$ के
        माध्यम से $D$ तक सिमटाइए)}।
  \item $\int_0^{+\infty}\frac{1 -
        \cos x}{x^2}\,\dd x$ परिकलित कीजिए, और दोनों परिणामों की संगति जाँचिए।
  \item $a > 0$ के लिए $\int_0^{+\infty}\frac{\sin(ax)}x\dd x$ और $\int_{-\infty}^{+\infty}\eu^{-ax^2}\dd x$ परिकलित कीजिए, और मापन नियम
        दर्ज कीजिए (वे \cref{ch:b3:fouriertransform} के श्रमिक बनेंगे)।
  \item ठीक-ठीक समझाइए कि $D$ को $t = 0$ पर सीधे प्रभावी अभिसरण
        प्रमेय से क्यों नहीं निपटाया जा सकता था ($[0,1]\times[0,\infty)$ पर कोई
        \omterm{def:b3:lebesgue:l1}{समाकलनीय} प्रभावी फलन नहीं), और प्रश्न 7 का पुच्छ-विभाजन
        उसका ईमानदार विकल्प क्यों है --- यह प्रतिरूप (``पुच्छों की
        एकसमान \omterm{def:b3:lebesgue:l1}{समाकलनीयता}'') विश्लेषण भर बार-बार लौटता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- बोर और मोलेरुप के अनुसार \omterm{ex:b3:lebesgue:gamma}{गामा फलन}।}
फलन $\Gamma$ (\cref{ex:b3:lebesgue:gamma}) $\Gamma(1) = 1$ और $\Gamma(x+1) = x\Gamma(x)$ को संतुष्ट करता है --- पर ऐसे
अनंत अन्य फलन भी करते हैं (किसी भी $1$-आवर्ती डगमगाहट से गुणा
कर दीजिए)। उत्तलता की एक शर्त $\Gamma$ को अद्वितीय रूप से बाँध देती
है, और तब उसकी गहरी सर्वसमिकाएँ यंत्रवत झड़ पड़ती हैं। किसी अंतराल
पर कोई धनात्मक फलन $f$ \emph{लघुगणकीय उत्तल} कहलाता है यदि
$\log f$ उत्तल हो।
\begin{enumerate}[resume]
  \item दर्शाइए कि लघुगणकीय उत्तलता से उत्तलता निकलती है, कि
        लघुगणकीय उत्तल फलनों के गुणनफल और उनके आफ़ीन प्रतिचित्रणों
        के साथ संयोजन लघुगणकीय उत्तल होते हैं, और --- दो-फलन वाली
        होल्डर असमिका $\int\abs{uv} \leq
        \bigl(\int\abs u^p\bigr)^{1/p}\bigl(\int\abs
        v^q\bigr)^{1/q}$ के माध्यम से, जो यंग की असमिका से
        सीधे सिद्ध होती है --- कि $\Gamma$ $\intoo0\infty$ पर लघुगणकीय उत्तल
        है।
  \item (ढाल प्रमेयिका) मान लीजिए $g$ $\intoo0\infty$ पर उत्तल है और
        प्रत्येक पूर्णांक $n
        \geq 1$ के लिए $g(n+1) - g(n) = \log n$। $x \in \intoc01$ और $n \geq 2$ के
        लिए $[n-1, n]$, $[n, n+x]$ और $[n, n+1]$ पर $g$ की ढालों की तुलना
        कीजिए, और निष्कर्ष निकालिए
        \[
        x\log(n-1) \;\leq\; g(n + x) - g(n) \;\leq\; x\log n .
        \]
  \item (बोर--मोलेरुप) मान लीजिए $f > 0$ $f(1) = 1$, $f(x+1) = xf(x)$ को संतुष्ट
        करता है और $\log f$ उत्तल है। पुनरावर्तन को $f(n + x) = x(x+1)\cdots(x + n -
        1)\,f(x)$ और $f(n) = (n-1)!$
        में खोलकर प्रश्न 14 से निष्कर्ष निकालिए कि $x \in \intoc01$ के लिए
        \[
        f(x) = \lim_{n\to\infty}
        \frac{n!\,n^x}{x(x+1)\cdots(x+n)} :
        \]
        अर्थात् $f$ अद्वितीय है, इसलिए $f = \Gamma$, और \emph{गाउस का
        सीमा सूत्र} लागू होता है (पुनरावर्तन से समस्त $x > 0$ तक
        विस्तारित कीजिए)।
  \item \emph{बीटा फलन} $B(x, y) =
        \int_0^1t^{x-1}(1-t)^{y-1}\,\dd t$ ($x, y > 0$) परिभाषित कीजिए। अभिसरण,
        पुनरावर्तन $B(x+1, y) =
        \frac{x}{x+y}\,B(x, y)$ (खंडशः समाकलन कीजिए), और $B(1, y) = \frac1y$ सिद्ध
        कीजिए।
  \item दर्शाइए कि $x \mapsto B(x, y)$ लघुगणकीय उत्तल है (फिर होल्डर), और
        \[
        f(x) = \frac{B(x, y)\,\Gamma(x + y)}{\Gamma(y)}
        \]
        पर बोर--मोलेरुप लगाकर \emph{ऑयलर का सूत्र} निष्कर्ष के रूप
        में निकालिए: $B(x, y) =
        \dfrac{\Gamma(x)\Gamma(y)}{\Gamma(x+y)}$ --- और कहीं कोई द्विक समाकल नहीं।
  \item $B(\frac12, \frac12)$ सीधे परिकलित कीजिए ($t
        = \sin^2\theta$ प्रतिस्थापित कीजिए) और
        $\Gamma(\frac12) =
        \sqrt\pi$ निष्कर्ष निकालिए: भाग I का गाउसीय समाकल, विशुद्ध
        उत्तलता से पुनः प्राप्त। दोनों उपपत्तियों की तुलना एक-एक
        वाक्य में कीजिए।
  \item (लजांद्र द्विगुणन) दर्शाइए कि
        \[
        g(x) = \frac{2^{x-1}}{\sqrt\pi}\,
        \Gamma\Bigl(\frac x2\Bigr)
        \Gamma\Bigl(\frac{x+1}2\Bigr)
        \]
        बोर--मोलेरुप की तीनों परिकल्पनाएँ संतुष्ट करता है, और
        $g = \Gamma$ निष्कर्ष निकालिए, अर्थात् समस्त $z > 0$ के लिए $\Gamma(2z) =
        \frac{2^{2z-1}}{\sqrt\pi}\,\Gamma(z)\,\Gamma(z +
        \tfrac12)$।
  \item संवृत रूप $\Gamma\bigl(n + \tfrac12\bigr)
        = \dfrac{(2n)!}{4^n\,n!}\sqrt\pi$ निष्कर्ष के रूप में निकालिए, और बड़े
        पूर्णांकों के आसपास $\log\Gamma$ पर ढाल प्रमेयिका लगाकर
        स्पर्शोन्मुखता
        \[
        \frac{\Gamma(n + \frac12)}{\Gamma(n)\,\sqrt n}
        \longrightarrow 1 .
        \]
        सिद्ध कीजिए।
  \item पिछले दोनों प्रश्नों को मिलाकर केंद्रीय द्विपद
        स्पर्शोन्मुखता
        \[
        \binom{2n}{n} \sim \frac{4^n}{\sqrt{\pi n}},
        \]
        निकालिए, और $n = 10$ के लिए संख्यात्मक सत्यापन कीजिए ($4^{10}/\sqrt{10\pi} \approx
        187079$
        के सामने $\binom{20}{10} =
        184756$: अनुपात $\approx 0.988$)।
  \item (संश्लेषण) नियतांक $\sqrt\pi$ अब तक गाउसीय समाकल के रूप में
        (भाग I), $B(\frac12,
        \frac12)$ के रूप में (प्रश्न 18), और द्विगुणन के भीतर
        (प्रश्न 19) प्रकट हो चुका है; और केंद्रीय द्विपद आकलन
        स्टर्लिंग (\cref{ch:b3:product} की सप्ताहांत समस्या) तथा
        दे म्वाव्र--लाप्लास दोनों की प्रत्याशा करता है। संबंधों का
        नक्शा बनाइए: कौन-से कथन किनके तुल्य हैं, और प्रत्येक
        तकनीक --- समाकल के भीतर अवकलन बनाम उत्तलता --- वह क्या
        देती है जो दूसरी नहीं दे सकती?
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- तीन और लाभांश।}
\begin{enumerate}[resume]
  \item (बीटा से वालिस) $p > -1$ के लिए $t =
        \sin^2\theta$ प्रतिस्थापित करके
        दर्शाइए
        \[
        W_p = \int_0^{\pi/2}\sin^p\theta\,\dd\theta
        = \frac12\,B\Bigl(\frac{p+1}2, \frac12\Bigr),
        \]
        और बीटा पुनरावर्तन (प्रश्न 16) से निष्कर्ष निकालिए कि
        पूर्णांक $n \geq
        0$ के लिए $W_{n+2} = \frac{n+1}{n+2}\,W_n$। $W_{2n}$ और $W_{2n+1}$ संवृत रूप में
        परिकलित कीजिए, दबाव से $W_{2n+1}/W_{2n} \to 1$ दर्शाइए, और \emph{वालिस के
        गुणनफल}
        \[
        \frac\pi2 = \lim_{n\to\infty}
        \prod_{k=1}^{n}\frac{4k^2}{4k^2 - 1} .
        \]
        पर पहुँचिए।
  \item (गाउसीय की आवृत्ति से भेंट) $b \in \R$ के लिए रखिए
        \[
        \Phi(b) = \int_{-\infty}^{+\infty}
        \eu^{-x^2}\cos(2bx)\,\dd x .
        \]
        दर्शाइए कि $\Phi$ $\R$ पर $\mathcal C^1$ है, कि खंडशः समाकलन से
        अवकल समीकरण $\Phi'(b) = -2b\,\Phi(b)$ मिलता है, और निष्कर्ष निकालिए
        \[
        \Phi(b) = \sqrt\pi\,\eu^{-b^2} :
        \]
        अर्थात् इस रूपांतर के अंतर्गत गाउसीय अपने आप को पुनः
        उत्पन्न कर लेता है --- यही वह एक सर्वसमिका है जिस पर
        \cref{ch:b3:fouriertransform} चलेगा।
  \item (फ्रुलानी का समाकल) $0 < a < b$ के लिए प्राचल $a$ में अवकलन
        करके दर्शाइए
        \[
        \int_0^{+\infty}
        \frac{\eu^{-ax} - \eu^{-bx}}{x}\,\dd x
        = \log\frac ba ,
        \]
        (प्रत्येक $\intco{a_0}{+\infty}$, $a_0 > 0$ पर प्रभुत्व उचित ठहराइए, और $a \to b$
        जाने देकर नियतांक पहचानिए)। समाकल्य को $x = 0$ पर अपनी
        विलोपनीय एकलता की आवश्यकता ठीक कहाँ पड़ती है?
\end{enumerate}
\end{problem}
