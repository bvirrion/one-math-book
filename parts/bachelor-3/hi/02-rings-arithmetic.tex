\chapter{वलय और अंकगणित}\label{ch:b3:rings}

साधारण पूर्णांकों का अभाज्यों में गुणनखंडन अद्वितीय होता है; किसी
क्षेत्र पर बहुपदों का भी। क्या ये दोनों तथ्य एक ही प्रमेय हैं? यह
अध्याय उत्तर देता है: हाँ; और वह ठीक-ठीक उन परिकल्पनाओं को खोज
निकालता है जो किसी क्रमविनिमेय वलय में ``अंकगणित'' को संभव बनाती
हैं --- शृंखला
\[
\text{यूक्लिडीय} \;\Longrightarrow\; \text{मुख्य}
\;\Longrightarrow\; \text{गुणनखंडनीय (UFD)},
\]
जिसमें सभी निहितार्थ सिद्ध किए गए हैं और सभी विलोम खंडित। फिर
सिद्धांत को वहाँ परखा जाता है जहाँ वह अपनी उपयोगिता सिद्ध करता है:
गाउसीय पूर्णांक $\Z[\iu]$ (जो सप्ताहांत समस्या में फर्मा की दो-वर्ग
प्रमेय को तोड़ देंगे), अनेक चरों के बहुपद वलय (गाउस की प्रमेयिका,
\omterm{thm:b3:rings:criteria}{आइज़ेनश्टाइन} कसौटी), और \omterm{def:b3:rings:noetherian}{नोएथरीय वलय}, जिनका शिखर है हिल्बर्ट की
आधार प्रमेय। सर्वत्र \emph{वलय} से अभिप्राय है इकाई $1 \neq 0$ वाला
क्रमविनिमेय वलय; दूसरे वर्ष के खंड के $\Z$ और $K[X]$ के \omterm{def:b3:rings:ideal}{आदर्श} हमारे
दो मार्गदर्शक उदाहरण हैं।

\section{आदर्श, भागफल और तुल्याकारिता प्रमेय}

\begin{definition}\label{def:b3:rings:ideal}
किसी वलय $A$ का \emph{आदर्श}\index{आदर्श} $I$ एक ऐसा योगात्मक
उपसमूह है जिसके लिए $AI \subseteq I$। \emph{भागफल वलय}\index{भागफल वलय}
$A/I$ \omterm{thm:b3:groups:quotient}{भागफल समूह} $(A, +)/I$ है, जिस पर गुणन $(a + I)(b + I) = ab + I$ है: यह सुपरिभाषित है,
क्योंकि $a$ को $a + x$ और $b$ को $b + y$ में बदलने पर ($x, y \in I$) $ab$
में $ay + xb + xy \in I$ का परिवर्तन होता है। प्रक्षेप $\pi \colon A
\to A/I$ एक आच्छादक वलय
समाकारिता है जिसकी अष्टि $I$ है, और वलय समाकारिताओं की अष्टियाँ
ठीक आदर्श ही हैं।
\end{definition}

\begin{theorem}[प्रथम तुल्याकारिता प्रमेय]\label{thm:b3:rings:firstiso}
यदि $f \colon A \to B$ एक वलय समाकारिता हो, तो $\bar f\colon
A/\ker f \to \operatorname{im} f$, $a + \ker f \mapsto f(a)$, एक वलय
तुल्याकारिता है। अधिक व्यापक रूप से, किसी भी \omterm{def:b3:rings:ideal}{आदर्श} $I \subseteq \ker f$ के लिए
$f$ $A/I$ से होकर गुजरता है। $A/I$ के \omterm{def:b3:rings:ideal}{आदर्श} ठीक $J/I$ हैं, जहाँ
$J \supseteq I$ $A$ का \omterm{def:b3:rings:ideal}{आदर्श} है (संगति प्रमेय)।
\end{theorem}

\begin{proof}
समूहों की तरह ही (\cref{thm:b3:groups:firstiso,%
thm:b3:groups:correspondence}), यह ध्यान रखते हुए कि यहाँ दिखने वाले सभी
प्रतिचित्रण गुणनफलों का भी आदर करते हैं: $\bar f$ सुपरिभाषित है,
प्रतिबिंब पर एकैकी आच्छादक है और गुणनात्मक है; तथा संगति $J \mapsto J/I$,
$\bar J \mapsto \pi^{-1}(\bar J)$ दोनों दिशाओं में आदर्शों को सुरक्षित रखती है, क्योंकि $\pi$
एक आच्छादक वलय समाकारिता है।
\end{proof}

\begin{definition}\label{def:b3:rings:primemaximal}
मान लीजिए $I \subsetneq A$ एक उचित \omterm{def:b3:rings:ideal}{आदर्श} है। $I$
\emph{अभाज्य}\index{अभाज्य आदर्श} कहलाता है यदि $ab \in I \Rightarrow a \in I$ या $b \in I$;
और $I$ \emph{महत्तम}\index{महत्तम आदर्श} कहलाता है यदि $I$
और $A$ के बीच कड़ाई से कोई \omterm{def:b3:rings:ideal}{आदर्श} न हो।
\end{definition}

\begin{proposition}\label{prop:b3:rings:primemaximal}
$I$ अभाज्य है $\iff$ $A/I$ एक पूर्णांकीय प्रांत है; $I$ \omterm{def:b3:rings:primemaximal}{महत्तम} है
$\iff$ $A/I$ एक क्षेत्र है। विशेष रूप से \omterm{def:b3:rings:primemaximal}{महत्तम आदर्श} अभाज्य होते
हैं।
\end{proposition}

\begin{proof}
$A/I$ के वर्गों के लिए $\bar a$ लिखिए। ``$I$ अभाज्य'' का शब्दशः
अनुवाद है ``$\bar a\bar b = 0 \Rightarrow \bar a = 0$ या $\bar b
= 0$'', और $A/I \neq 0$ का $I \neq A$: यही प्रांत की
परिभाषा है। महत्तमता के लिए संगति प्रमेय का उपयोग कीजिए: $I$
और $A$ के बीच कड़ाई से कोई \omterm{def:b3:rings:ideal}{आदर्श} नहीं $\iff$ $A/I$ में $0$ और
स्वयं के अतिरिक्त कोई \omterm{def:b3:rings:ideal}{आदर्श} नहीं $\iff$ $A/I$ एक क्षेत्र है ---
अंतिम पद के लिए: किसी क्षेत्र में एकमात्र \omterm{def:b3:rings:ideal}{आदर्श} $0$ और समूचा
क्षेत्र हैं ($x \ne 0$ को अंतर्विष्ट करने वाला \omterm{def:b3:rings:ideal}{आदर्श} $x^{-1}x = 1$ को भी
अंतर्विष्ट करता है); विलोमतः यदि प्रत्येक शून्येतर $x$ इकाई
\omterm{def:b3:rings:ideal}{आदर्श} जनित करे, तो किसी $y$ के लिए $xy = 1$। क्षेत्र प्रांत होते
हैं, अतः \omterm{def:b3:rings:primemaximal}{महत्तम आदर्श} अभाज्य होते हैं।
\end{proof}

\begin{example}\label{ex:b3:rings:primemaximal}
$\Z$ में: \omterm{def:b3:rings:primemaximal}{अभाज्य आदर्श} $(0)$ और $(p)$ हैं, जहाँ $p$ अभाज्य है;
\omterm{def:b3:rings:primemaximal}{महत्तम आदर्श} $(p)$ हैं ($\Z/p\Z = \mathbb F_p$ एक क्षेत्र है, $\Z/(0) = \Z$ नहीं)। $K[X, Y]$
में: $(X) \subsetneq (X, Y)$ दोनों अभाज्य हैं ($K[X,Y]/(X) \cong K[Y]$, एक प्रांत; $K[X,Y]/(X,Y) \cong K$, एक क्षेत्र),
अतः $(X)$ अभाज्य है पर \omterm{def:b3:rings:primemaximal}{महत्तम} नहीं।
\end{example}

पूर्ण व्यापकता में \omterm{def:b3:rings:primemaximal}{महत्तम} आदर्शों के \emph{अस्तित्व} की गारंटी के
लिए हमें एक समुच्चय-सैद्धांतिक सिद्धांत चाहिए। कोई आंशिक क्रमित
समुच्चय \emph{आगमनात्मक} कहलाता है यदि उसके प्रत्येक पूर्णतः
क्रमित उपसमुच्चय (\emph{शृंखला}) का कोई ऊपरी परिबंध हो।

\begin{theorem}[ज़ोर्न की प्रमेयिका]\label{thm:b3:rings:zorn}
प्रत्येक अरिक्त आगमनात्मक आंशिक क्रमित समुच्चय में कोई \omterm{def:b3:rings:primemaximal}{महत्तम}
अवयव होता है।\index{ज़ोर्न की प्रमेयिका}
\admitted
\end{theorem}

\begin{remark}\label{rem:b3:rings:zorn}
यह साधारण गणित की प्रमेय नहीं, बल्कि एक \emph{स्वयंसिद्ध} है:
समुच्चय सिद्धांत के मूल ज़र्मेलो--फ्रेंकेल स्वयंसिद्धों पर यह
चयन स्वयंसिद्ध (``अरिक्त समुच्चयों का प्रत्येक गुणन अरिक्त होता
है'') के तुल्य है, जिसे हम इस पुस्तक भर स्वीकार करते हैं।
प्रत्येक उपयोग को हम रेखांकित करेंगे। विश्लेषण में यह पुनः आएगा
(हान--बानाख, \cref{ch:b3:banach})।
\end{remark}

\begin{theorem}[क्रुल]\label{thm:b3:rings:krull}
प्रत्येक उचित \omterm{def:b3:rings:ideal}{आदर्श} $I \subsetneq A$ किसी \omterm{def:b3:rings:primemaximal}{महत्तम आदर्श} में अंतर्विष्ट
होता है।\index{क्रुल की प्रमेय}
\end{theorem}

\begin{proof}
$I$ को अंतर्विष्ट करने वाले उचित आदर्शों के समुच्चय $\mathcal E$ को
अंतर्वेशन से क्रमित कीजिए; वह अरिक्त है ($I \in \mathcal E$)। $\mathcal E$ की किसी
शृंखला $(J_\lambda)$ का ऊपरी परिबंध $J = \bigcup
J_\lambda$ है: यह एक \omterm{def:b3:rings:ideal}{आदर्श} है (पूर्ण
क्रमबद्धता से कोई भी $a, b \in J$ किसी एक ही $J_\lambda$ में स्थित होते हैं),
उचित है (सभी $\lambda$ के लिए $1 \notin J_\lambda$), और $I$ को अंतर्विष्ट करता
है। ज़ोर्न की प्रमेयिका से $\mathcal
E$ का कोई \omterm{def:b3:rings:primemaximal}{महत्तम} अवयव मिलता है, जो
$I$ को अंतर्विष्ट करने वाला \omterm{def:b3:rings:primemaximal}{महत्तम आदर्श} है (उससे कड़ाई से
ऊपर का कोई उचित \omterm{def:b3:rings:ideal}{आदर्श} $\mathcal E$ में ही होता)।
\end{proof}

\begin{theorem}[चीनी शेषफल प्रमेय]\label{thm:b3:rings:crt}
मान लीजिए $I_1, \dots, I_n$ $A$ के जोड़े-जोड़े में
\emph{सह-महत्तम}\index{सह-महत्तम आदर्श} \omterm{def:b3:rings:ideal}{आदर्श} हैं ($k \neq l$ के लिए
$I_k + I_l = A$)। तब
\[
A\Big/\bigcap_{k=1}^n I_k \;\xrightarrow{\;\sim\;}\;
\prod_{k=1}^n A/I_k,
\qquad
a \longmapsto (a + I_1, \dots, a + I_n),
\]
और इसके अतिरिक्त $\bigcap_k I_k = I_1 I_2 \cdots I_n$ (गुणनफलों से जनित
\omterm{def:b3:rings:ideal}{आदर्श})।\index{चीनी शेषफल प्रमेय}
\end{theorem}

\begin{proof}
प्रतिचित्रण $f(a) = (a + I_k)_k$ एक वलय समाकारिता है जिसकी अष्टि $\bigcap I_k$ है;
\cref{thm:b3:rings:firstiso} से केवल आच्छादकता सिद्ध करना शेष रह जाता है। $k$ को स्थिर
कीजिए; प्रत्येक $l \neq k$ के लिए सह-महत्तमता से $u_l \in I_k$, $v_l \in I_l$ के साथ
$1 = u_l +
v_l$ लिखिए। तब
\[
e_k = \prod_{l \neq k} v_l = \prod_{l\neq k}(1 - u_l)
\equiv 1 \pmod{I_k},
\qquad e_k \in I_l \ (l \neq k),
\]
अतः $f(e_k) = (0, \dots, 1, \dots, 0)$; और कोई लक्ष्य $(a_k +
I_k)_k$ दिया हो, तो अवयव $\sum_k a_k e_k$ उसी पर
जाता है।

गुणनफल बनाम प्रतिच्छेदन: $I_1\cdots I_n \subseteq \bigcap I_k$ सदैव। विलोमतः, आगमन से केवल
$n = 2$ की स्थिति देखनी पर्याप्त है (जाँच लीजिए कि $I_1$ और $I_2\cdots I_n$
\omterm{thm:b3:rings:crt}{सह-महत्तम} हैं: $l \geq 2$ पर $1 = u_l + v_l$ का गुणन करने से $1 \in I_1 + I_2\cdots
I_n$ मिलता है)।
$n = 2$ के लिए: $1 = u + v$, $u \in I_1$, $v \in I_2$ लिखिए; तब $x \in I_1 \cap I_2$ के लिए $x = xu + xv \in I_2I_1 + I_1I_2 =
I_1I_2$।
\end{proof}

\begin{example}\label{ex:b3:rings:crtz}
$\Z$ में, जहाँ $I_k = (m_k)$ है और $m_k$ जोड़े-जोड़े में सह-अभाज्य हैं:
$\Z/(m_1\cdots
m_n)\Z \cong \prod \Z/m_k\Z$ --- यही दूसरे वर्ष के खंड की चीनी शेषफल प्रमेय है। इकाइयों
तक सीमित करने पर $\gcd(m,n)=1$ के लिए $(\Z/mn\Z)^\times \cong
(\Z/m\Z)^\times \times (\Z/n\Z)^\times$, जिससे ऑयलर के $\varphi$ की
गुणनात्मकता निकलती है (\cref{exo:b3:rings:8})।
\end{example}

\section{विभाज्यता: यूक्लिडीय, मुख्य, गुणनखंडनीय}

\begin{definition}\label{def:b3:rings:divisibility}
मान लीजिए $A$ एक पूर्णांकीय प्रांत है और $a, b \in A$। हम कहते हैं कि
$a$ $b$ को \emph{विभाजित} करता है ($a \mid b$) यदि $b \in (a) = aA$ हो। अवयव
$a,
b$ \emph{सहचारी}\index{सहचारी अवयव} कहलाते हैं यदि $u \in A^\times$ के
साथ $a = ub$ हो (समतुल्य रूप से $(a) = (b)$)। कोई शून्येतर अनैकिक अवयव
$p$:
\begin{itemize}
  \item \emph{अखंडनीय}\index{अखंडनीय अवयव} कहलाता है यदि $p = ab$
        से $a \in A^\times$ या $b \in A^\times$ अनिवार्य हो जाए;
  \item \emph{अभाज्य}\index{अभाज्य अवयव} कहलाता है यदि $p \mid ab$ से
        $p
        \mid a$ या $p \mid b$ अनिवार्य हो जाए (अर्थात् \omterm{def:b3:rings:ideal}{आदर्श} $(p)$
        अभाज्य हो)।
\end{itemize}
\end{definition}

\begin{proposition}\label{prop:b3:rings:primeirred}
किसी भी प्रांत में, अभाज्य $\Rightarrow$ \omterm{def:b3:rings:divisibility}{अखंडनीय}। विलोम सामान्यतः असत्य
है: $\Z[\iu\sqrt 5] = \{a + \iu b\sqrt5 : a, b
\in \Z\}$ में अवयव $2$ \omterm{def:b3:rings:divisibility}{अखंडनीय} है पर अभाज्य नहीं।
\end{proposition}

\begin{proof}
मान लीजिए $p$ अभाज्य है और $p = ab$। तब $p \mid ab$, मान लीजिए $p \mid a$:
$a = pc$, अतः $p = pcb$, और $p$ को निरस्त करने पर (प्रांत!) $cb =
1$
मिलता है: $b \in A^\times$।

$\Z[\iu\sqrt5]$ में मानक $N(x + \iu y\sqrt 5) = x^2 +
5y^2$ का उपयोग कीजिए, जो गुणनात्मक है (वह $\abs z^2$
है)। यदि अनैकिक $a, b$ के साथ $2 = ab$ हो, तो $N(a), N(b) \neq
1$ के साथ $4 = N(a)N(b)$
(मानक-$1$ अवयव $\pm1$ हैं, अर्थात् इकाइयाँ), अतः $N(a) = 2$:
असंभव, क्योंकि $x^2 + 5y^2 = 2$ का कोई पूर्णांक हल नहीं है। अतः $2$
\omterm{def:b3:rings:divisibility}{अखंडनीय} है। परंतु $2 \mid 6 = (1 + \iu\sqrt5)(1 - \iu\sqrt5)$, जबकि $2$ किसी भी गुणनखंड को विभाजित
नहीं करता ($\frac12 \pm \frac{\iu\sqrt5}2
\notin \Z[\iu\sqrt5]$): अभाज्य नहीं।
\end{proof}

\begin{definition}\label{def:b3:rings:pidufd}
कोई पूर्णांकीय प्रांत $A$:
\begin{itemize}
  \item \emph{यूक्लिडीय}\index{यूक्लिडीय प्रांत} कहलाता है यदि
        कोई ऐसा प्रतिचित्रण $\nu \colon A \setminus \{0\} \to \N$ (एक \emph{यूक्लिडीय फलन}) हो
        कि $b \ne 0$ वाले सभी $a, b$ के लिए ऐसे $q, r$ विद्यमान हों
        जिनके लिए $a = bq + r$ तथा ($r = 0$ या $\nu(r) <
        \nu(b)$);
  \item \emph{मुख्य} (एक \emph{PID}\index{मुख्य आदर्श प्रांत})
        कहलाता है यदि प्रत्येक \omterm{def:b3:rings:ideal}{आदर्श} $(a)$ के रूप का हो;
  \item \emph{गुणनखंडनीय} (एक \emph{UFD}\index{अद्वितीय
        गुणनखंडन प्रांत}) कहलाता है यदि प्रत्येक शून्येतर अनैकिक
        अवयव \omterm{def:b3:rings:divisibility}{अखंडनीय} अवयवों का गुणनफल हो, और वह भी क्रम तथा
        सहचारियों तक अद्वितीय रूप से।
\end{itemize}
\end{definition}

\begin{theorem}\label{thm:b3:rings:euclideanpid}
\omterm{def:b3:rings:pidufd}{यूक्लिडीय} $\Rightarrow$ मुख्य।
\end{theorem}

\begin{proof}
मान लीजिए $I \neq (0)$ एक \omterm{def:b3:rings:ideal}{आदर्श} है और $b \in I \setminus\{0\}$ ऐसा अवयव जिसके लिए $\nu(b)$
न्यूनतम है। $a \in I$ के लिए भाग दीजिए: $a = bq + r$; तब $r = a
- bq \in I$, और $\nu(r) < \nu(b)$
न्यूनतमता का खंडन करता, अतः $r = 0$ और $a \in (b)$: $I = (b)$।
\end{proof}

\begin{example}\label{ex:b3:rings:euclidean}
$\Z$ ($\nu = \abs\cdot$ के साथ) और $K[X]$ ($\nu = \deg$ के साथ) \omterm{def:b3:rings:pidufd}{यूक्लिडीय} हैं ---
दूसरे वर्ष के खंड में दोनों विभाजन सिद्ध हो चुके हैं। $\Z[\iu]$ भी
\omterm{def:b3:rings:pidufd}{यूक्लिडीय} है, जिसमें $\nu = N$ वर्ग-मानक है (\cref{exo:b3:rings:4}); उपपत्ति की
ज्यामिति नीचे के चित्र में है। ऐसा \omterm{def:b3:rings:pidufd}{PID} जो \omterm{def:b3:rings:pidufd}{यूक्लिडीय} न हो,
विद्यमान तो है पर उसका प्रमाणन नाजुक है (मानक उदाहरण $\Z\bigl[\frac{1+\iu\sqrt{19}}2\bigr]$ है);
ऐसा \omterm{def:b3:rings:pidufd}{UFD} जो \omterm{def:b3:rings:pidufd}{PID} न हो, आसान है: $K[X, Y]$ (\cref{exo:b3:rings:6}), अथवा $\Z[X]$।
\end{example}

\begin{lemma}[मुख्य आदर्शों की आरोही शृंखलाएँ]
\label{lem:b3:rings:acc}
किसी \omterm{def:b3:rings:pidufd}{PID} में आदर्शों का प्रत्येक बढ़ता हुआ अनुक्रम $I_1 \subseteq I_2
\subseteq \cdots$ अंततः
अचर हो जाता है।
\end{lemma}

\begin{proof}
$I = \bigcup_n I_n$ एक \omterm{def:b3:rings:ideal}{आदर्श} है (सम्मिलन बढ़ता हुआ है), अतः $I =
(a)$; अवयव $a$
किसी $I_N$ में स्थित है, और तब $n \geq N$ के लिए $I = (a)
\subseteq I_N \subseteq I_n \subseteq I$।
\end{proof}

\begin{lemma}[बेज़ू; यूक्लिड की प्रमेयिका]\label{lem:b3:rings:bezout}
मान लीजिए $A$ एक \omterm{def:b3:rings:pidufd}{PID} है और $a, b \in A$। तब किसी $d$ के लिए $(a) + (b) = (d)$,
जो एक \emph{महत्तम समापवर्तक}\index{महत्तम समापवर्तक} है:
$d \mid a$, $d \mid b$, और $a,
b$ का प्रत्येक उभयनिष्ठ भाजक $d$ को
विभाजित करता है; इसके अतिरिक्त किन्हीं $u, v$ के लिए $d = au + bv$
(बेज़ू)। फलस्वरूप किसी \omterm{def:b3:rings:pidufd}{PID} का प्रत्येक \omterm{def:b3:rings:divisibility}{अखंडनीय अवयव} अभाज्य होता
है।
\end{lemma}

\begin{proof}
$(a) + (b)$ एक \omterm{def:b3:rings:ideal}{आदर्श} है, अतः $(d)$; $a, b \in (d)$ से $d \mid
a, b$ मिलता है; और $d = au + bv \in (a) + (b)$।
$a, b$ का कोई उभयनिष्ठ भाजक $c$ $au + bv = d$ को विभाजित करता है।

यूक्लिड: मान लीजिए $p$ \omterm{def:b3:rings:divisibility}{अखंडनीय} है, $p \mid ab$, $p \nmid a$। $p$ और
$a$ का \omterm{def:b3:rings:primemaximal}{महत्तम} समापवर्तक $d$ $p$ को विभाजित करता है, अतः
अखंडनीयता से $d$ या तो इकाई है या $p$ का \omterm{def:b3:rings:divisibility}{सहचारी}; \omterm{def:b3:rings:divisibility}{सहचारी} होना
$p \nmid a$ से अपवर्जित है। अतः $1 = pu + av$, जिससे $b = pub + abv$, और $p$ दोनों
पदों को विभाजित करता है: $p \mid b$।
\end{proof}

\begin{theorem}\label{thm:b3:rings:pidufd}
मुख्य $\Rightarrow$ गुणनखंडनीय।
\end{theorem}

\begin{proof}
\emph{अस्तित्व।} मान लीजिए किसी शून्येतर अनैकिक $a$ का
\omterm{def:b3:rings:divisibility}{अखंडनीय} अवयवों में गुणनखंडन नहीं है। तब $a$ \omterm{def:b3:rings:divisibility}{अखंडनीय} नहीं है:
$a =
a_1b_1$, जहाँ दोनों गुणनखंड अनैकिक हैं; उनमें से कम से कम एक --- मान
लीजिए $a_1$ --- का भी कोई गुणनखंडन नहीं है (दो गुणनखंडनीय अवयवों
का गुणनफल गुणनखंडनीय होता है)। इसे दोहराते जाने पर हमें $a = a_0, a_1, a_2,
\dots$
मिलते हैं, जिनमें प्रत्येक पिछले का उचित भाजक है और किसी का
गुणनखंडन नहीं है, अतः $(a_0) \subsetneq (a_1) \subsetneq (a_2) \subsetneq \cdots$ --- अंतर्वेशन कड़े हैं, क्योंकि
अनैकिक $c$ के साथ $a_n = a_{n+1}c$ का अर्थ है कि $(a_n) = (a_{n+1})$ से $c \in A^\times$ अनिवार्य
हो जाता (प्रांत में निरस्त कीजिए)। यह \cref{lem:b3:rings:acc} का खंडन करता है।

\emph{अद्वितीयता।} मान लीजिए $p_1 \cdots p_r = q_1 \cdots q_s$, जहाँ सभी गुणनखंड \omterm{def:b3:rings:divisibility}{अखंडनीय} हैं
और $r \leq s$; $r$ पर आगमन कीजिए। अभाज्य (\cref{lem:b3:rings:bezout}) $p_1$ दाईं ओर को
विभाजित करता है, अतः किसी $q_j$ को विभाजित करता है; पुनःक्रमांकन
से $j = 1$। चूँकि $q_1$ \omterm{def:b3:rings:divisibility}{अखंडनीय} है और $p_1$ इकाई नहीं, $u \in A^\times$ के
साथ $q_1 = u p_1$: अर्थात् $p_1,
q_1$ \omterm{def:b3:rings:divisibility}{सहचारी} हैं। $p_1$ को निरस्त कीजिए:
$p_2 \cdots p_r = (u q_2)
q_3\cdots q_s$, और आगमन से निष्कर्ष निकालिए ($r = 1$ से $s = 1$ अनिवार्य हो
जाता है: इकाई और \omterm{def:b3:rings:divisibility}{अखंडनीय} अवयवों का गुणनफल $1$ नहीं हो सकता)।
\end{proof}

\begin{remark}\label{rem:b3:rings:ufdgcd}
किसी \omterm{def:b3:rings:pidufd}{UFD} में \omterm{def:b3:rings:primemaximal}{महत्तम} समापवर्तक विद्यमान होते हैं (गुणनखंडनों में
न्यूनतम घातांक लीजिए) और यूक्लिड की प्रमेयिका भी लागू होती है ---
\omterm{def:b3:rings:divisibility}{अखंडनीय} $=$ अभाज्य (\cref{exo:b3:rings:2}) --- पर बेज़ू विफल हो सकता है:
$\Z[X]$ में $\gcd(2, X) = 1$, फिर भी $1 \neq 2U + XV$ ($X = 0$ पर मान लीजिए: $1 = 2U(0)$, जो
असंभव है)। बेज़ू सर्वसमिकाएँ \omterm{def:b3:rings:pidufd}{PID} की अनन्य संपत्ति हैं।
\end{remark}

\begin{example}[अद्वितीय गुणनखंडन के बिना एक वलय]
\label{ex:b3:rings:nonufd}
\omterm{def:b3:rings:pidufd}{यूक्लिडीय} $\Rightarrow$ \omterm{def:b3:rings:pidufd}{PID} $\Rightarrow$ \omterm{def:b3:rings:pidufd}{UFD} में से कोई भी निहितार्थ तुल्यता
नहीं है, और अंतिम की विफलता को एक बार पूरे विस्तार से देखना
सार्थक है।
\[
A = \Z[\iu\sqrt5] = \{a + \iu b\sqrt5 : a, b \in \Z\},
\qquad N(a + \iu b\sqrt5) = a^2 + 5b^2,
\]
में मानक गुणनात्मक है और $N(z) = 1$ तभी और केवल तभी जब $z \in
A^\times = \{\pm1\}$। अब
\[
6 = 2 \cdot 3 = (1 + \iu\sqrt5)(1 - \iu\sqrt5).
\]
पर विचार कीजिए। चारों गुणनखंड \omterm{def:b3:rings:divisibility}{अखंडनीय} हैं: उनके मानक $4, 9, 6,
6$ हैं,
और कोई उचित गुणनखंडन $z = z_1z_2$ $N(z_1) \in \{2, 3\}$ पर विवश कर देता --- परंतु
$a^2 + 5b^2$ कभी भी $2$ या $3$ के बराबर नहीं होता ($b = 0$ से
अवर्ग $2, 3$ बचते हैं; $\abs b \geq
1$ से $\geq 5$ मिलता है)। फिर भी $2$
$1 \pm
\iu\sqrt5$ में से किसी का भी \omterm{def:b3:rings:divisibility}{सहचारी} नहीं है (मानक $4 \neq 6$): अर्थात्
$6$ के \omterm{def:b3:rings:divisibility}{अखंडनीय} अवयवों में दो सर्वथा भिन्न गुणनखंडन। समतुल्य
रूप से, यहाँ \omterm{def:b3:rings:divisibility}{अखंडनीय} $\neq$ अभाज्य: $2$ गुणनफल $(1 +
\iu\sqrt5)(1 - \iu\sqrt5) = 6$ को
विभाजित करता है पर किसी गुणनखंड को नहीं (फिर मानक देखिए)। इस
विफलता की आदर्श-सैद्धांतिक मरम्मत --- अवयवों के स्थान पर
\emph{आदर्शों} का गुणनखंडन --- बीजीय संख्या सिद्धांत का जन्म है;
हमारे स्तर पर यह उदाहरण यह आँकने में सहायक है कि इस अध्याय के
\omterm{def:b3:rings:pidufd}{यूक्लिडीय} वलय $\Z$, $K[X]$, $\Z[\iu]$ वास्तव में कितने विशिष्ट हैं।
\end{example}

\begin{method}\label{met:b3:rings:quotient}
किसी \omterm{def:b3:rings:ideal}{भागफल वलय} $A/I$ को पहचानने के लिए अष्टि $I$ वाली कोई
आच्छादक समाकारिता $f \colon A \to B$ खोजिए और \cref{thm:b3:rings:firstiso} लगाइए; जब $A = C[X]$ कोई बहुपद
वलय हो, तब $f$ प्रायः कोई मान-निर्धारण होता है। इस प्रकार
$\Z[X]/(X^2+1) \cong
\Z[\iu]$ ($\iu$ पर मान लीजिए), $K[X,Y]/(Y - X^2) \cong K[X]$ ($X^2$ पर $Y$ का मान
लीजिए), $\R[X]/(X^2+1) \cong \C$। $I$ को अभाज्य या \omterm{def:b3:rings:primemaximal}{महत्तम} दिखाने के लिए यह
दिखाइए कि भागफल एक प्रांत है या एक क्षेत्र (\cref{prop:b3:rings:primemaximal})।
\end{method}

\section{किसी UFD पर बहुपद: गाउस और आइज़ेनश्टाइन}

इस पूरे अनुभाग में $A$ भिन्न क्षेत्र $K$ वाला एक \omterm{def:b3:rings:pidufd}{UFD} है
(यह क्षेत्र औपचारिक भागफलों $a/b$, $b \neq 0$, के क्षेत्र के रूप
में ठीक वैसे ही रचा जाता है जैसे $\Z$ से $\Q$; दूसरे वर्ष के
खंड में यह रचना $\Q$ के लिए की गई थी, और वह शब्दशः यहाँ भी
लागू होती है)। हमारा लक्ष्य: गुणनखंडनीयता $A$ से $A[X]$ तक
पहुँच जाती है, और $A$ पर अखंडनीयता मूलतः बड़े क्षेत्र $K$ पर
अखंडनीयता ही है।

\begin{definition}\label{def:b3:rings:content}
किसी शून्येतर $P \in A[X]$ की \emph{अंतर्वस्तु}\index{बहुपद की
अंतर्वस्तु} $c(P)$ उसके गुणांकों का एक \omterm{def:b3:rings:primemaximal}{महत्तम} समापवर्तक है (जो
किसी इकाई तक परिभाषित है); $P$ \emph{आद्य} कहलाता है यदि
$c(P) \in A^\times$ हो। प्रत्येक $P
\in A[X]$ $P_1$ आद्य के साथ $P = c(P)\,P_1$ के रूप में
लिखा जाता है, और प्रत्येक $P \in K[X]\setminus\{0\}$ $\lambda
\in K^\times$ तथा $P_1 \in A[X]$ आद्य के साथ
$P = \lambda P_1$ के रूप में (हर हटाइए, फिर अंतर्वस्तु बाहर निकालिए)।
\end{definition}

\begin{lemma}[गाउस]\label{lem:b3:rings:gauss}
$A[X]$ के दो आद्य बहुपदों का गुणनफल आद्य होता है; फलस्वरूप इकाइयों
तक $c(PQ) = c(P)c(Q)$।\index{गाउस की प्रमेयिका}
\end{lemma}

\begin{proof}
मान लीजिए $P, Q$ आद्य हैं और कोई \omterm{def:b3:rings:divisibility}{अखंडनीय} (= अभाज्य, क्योंकि \omterm{def:b3:rings:pidufd}{UFD})
$p$ $PQ$ के सभी गुणांकों को विभाजित करता है। $p$ के सापेक्ष
अपचयन कीजिए: $(A/(p))[X]$ में $\bar P \bar Q = 0$। परंतु $A/(p)$ एक प्रांत है ($(p)$
अभाज्य), अतः $(A/(p))[X]$ भी प्रांत है (अग्र गुणांक आपस में गुणा होते
हैं), जिससे $\bar P = 0$ या $\bar Q = 0$ अनिवार्य हो जाता है: अर्थात् $p$
$P$ के सभी गुणांकों को या $Q$ के सभी गुणांकों को विभाजित
करता है, जो आद्यता का खंडन है। परिणाम के लिए $P = c(P)P_1$, $Q = c(Q)Q_1$ लिखिए:
$P_1Q_1$ आद्य के साथ $PQ =
c(P)c(Q) P_1Q_1$।
\end{proof}

\begin{theorem}\label{thm:b3:rings:gaussufd}
मान लीजिए $A$ भिन्न क्षेत्र $K$ वाला एक \omterm{def:b3:rings:pidufd}{UFD} है।
\begin{enumerate}
  \item घात $\geq 1$ वाला कोई आद्य $P \in A[X]$ $A[X]$ में \omterm{def:b3:rings:divisibility}{अखंडनीय} है तभी
        और केवल तभी जब वह $K[X]$ में \omterm{def:b3:rings:divisibility}{अखंडनीय} हो।
  \item $A[X]$ एक \omterm{def:b3:rings:pidufd}{UFD} है; उसके \omterm{def:b3:rings:divisibility}{अखंडनीय अवयव} $A$ के \omterm{def:b3:rings:divisibility}{अखंडनीय अवयव}
        तथा $K$ पर \omterm{def:b3:rings:divisibility}{अखंडनीय} आद्य बहुपद हैं। विशेष रूप से $\Z[X]$,
        और आगमन से $K[X_1, \dots, X_n]$ तथा $\Z[X_1, \dots, X_n]$, \omterm{def:b3:rings:pidufd}{UFD} हैं।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) ($\Leftarrow$) यदि अनैकिक $Q, R$ के साथ $A[X]$ में $P = QR$ हो, तो कोई
भी गुणनखंड अचर नहीं है (आद्य बहुपद का अचर गुणनखंड इकाई होता है),
अतः गुणनखंडन $K[X]$ में उचित है।
($\Rightarrow$) मान लीजिए $P = QR$, जहाँ $Q, R \in K[X]$ की घातें $\geq 1$ हैं। $Q_1, R_1
\in A[X]$
आद्य के साथ $Q = \lambda Q_1$, $R = \mu R_1$ लिखिए: $P = \lambda\mu\, Q_1R_1$, और गाउस से $Q_1R_1$ आद्य है।
\omterm{def:b3:rings:content}{अंतर्वस्तु} लेने पर $\lambda\mu \in A^\times$ (दोनों ओर इकाई \omterm{def:b3:rings:content}{अंतर्वस्तु} है; औपचारिक
रूप से इकाइयों तक $\lambda\mu = c(P) \in
A^\times$, और विशेष रूप से $\lambda \mu \in A$): अतः $P
= (\lambda\mu Q_1) R_1$
$A[X]$ में एक उचित गुणनखंडन है।

(2) अस्तित्व: कोई अनैकिक $P \neq 0$ दिया हो, तो $P = c(P)P_1$ गुणनखंडित
कीजिए, $c(P)$ को $A$ के \omterm{def:b3:rings:divisibility}{अखंडनीय} अवयवों में गुणनखंडित कीजिए, और
$P_1$ को \omterm{def:b3:rings:pidufd}{UFD} $K[X]$ में $Q_i \in K[X]$ \omterm{def:b3:rings:divisibility}{अखंडनीय} के साथ $\prod Q_i$ के रूप में
गुणनखंडित कीजिए; $R_i \in A[X]$ आद्य (अतः $K$ पर \omterm{def:b3:rings:divisibility}{अखंडनीय}, और (1) से
$A[X]$ में भी) के साथ $Q_i = \lambda_i R_i$ लिखने पर, गुणनफल $\prod \lambda_i$ पहले की तरह
$A$ की इकाई है, और $P_1 = u\prod
R_i$। अद्वितीयता: किसी गुणनखंडन के अचर भाग
और बहुपद भाग की तुलना कीजिए; अचर गुणा होकर $c(P)$ देते हैं (गाउस),
जो $A$ की गुणनखंडनीयता से अद्वितीय है; बहुपद भाग एक ही बहुपद
के $K[X]$ में दो गुणनखंडन देते हैं, अतः वे $K^\times$ के अचरों तक मेल
खाते हैं ($K[X]$ की गुणनखंडनीयता, \cref{thm:b3:rings:pidufd}), और $K[X]$ में \omterm{def:b3:rings:divisibility}{सहचारी}
मिलान करने वाले आद्य बहुपद $A[X]$ में भी \omterm{def:b3:rings:divisibility}{सहचारी} होते हैं: यदि
$R, R'$ आद्य और $\lambda \in K^\times$ के साथ $R = \lambda R'$ हो, तो \omterm{def:b3:rings:content}{अंतर्वस्तु} लेने पर
$\lambda \in A^\times$ अनिवार्य हो जाता है।
\end{proof}

\begin{theorem}[अखंडनीयता की कसौटियाँ]\label{thm:b3:rings:criteria}
मान लीजिए $A$ एक \omterm{def:b3:rings:pidufd}{UFD} है, $K$ उसका भिन्न क्षेत्र, और $P = a_nX^n + \dots
+ a_0 \in A[X]$
घात $n \geq 1$ का आद्य बहुपद।
\begin{enumerate}
  \item (\emph{अपचयन}) यदि $p \in A$ अभाज्य हो, $p \nmid a_n$, और अपचयन
        $\bar P$ $(A/(p))[X]$ में \omterm{def:b3:rings:divisibility}{अखंडनीय} हो, तो $P$ $K[X]$ में (अतः
        $A[X]$ में भी) \omterm{def:b3:rings:divisibility}{अखंडनीय} है।
  \item (\emph{आइज़ेनश्टाइन}\index{आइज़ेनश्टाइन कसौटी}) यदि कोई
        अभाज्य $p$ $p \nmid a_n$, $0
        \leq i < n$ के लिए $p \mid a_i$, तथा $p^2 \nmid a_0$ को
        संतुष्ट करे, तो $P$ $K[X]$ में (अतः $A[X]$ में भी)
        \omterm{def:b3:rings:divisibility}{अखंडनीय} है।
\end{enumerate}
\end{theorem}

\begin{proof}
\cref{thm:b3:rings:gaussufd}(1) से, $K$ पर कोई उचित गुणनखंडन $Q, R \in A[X]$, $\deg Q, \deg R \geq 1$ के साथ
$P = QR$ देता है (अचर अपवर्जित हैं: वे या तो इकाई होते या आद्यता
भंग करते)।

(1) $p$ के सापेक्ष अपचयन कीजिए: $(A/(p))[X]$ में $\bar P = \bar Q\bar R$। चूँकि अपचयन
के अंतर्गत $p \nmid a_n$ और $\deg$ केवल घट सकते हैं, अतः $\deg \bar
Q = \deg Q \geq 1$ और $\deg\bar R = \deg R \geq 1$
(उनके अग्र गुणांक गुणा होकर $\bar a_n \neq 0$ देते हैं, अतः कोई नहीं घटता):
$\bar P$ का उचित गुणनखंडन हो जाता है --- विरोधाभास।

(2) $p$ के सापेक्ष अपचयन कीजिए: $\bar Q \bar R = \bar P = \bar a_n X^n$ (सभी निम्न गुणांक समाप्त
हो जाते हैं)। प्रांत $(A/(p))[X]$ में $cX^n$ ($c \ne 0$) के गुणनखंडन अचरों
और शुद्ध घातों $c'X^k$ में ही होते हैं: वस्तुतः यदि $\bar Q\bar R = \bar a_nX^n$ हो और
मान लीजिए $\bar Q$ का घात $< \deg\bar Q$ में कोई शून्येतर गुणांक हो, तो
न्यूनतम शून्येतर पद लीजिए: $\operatorname{val}(\bar Q\bar R) =
\operatorname{val}\bar Q + \operatorname{val}\bar R$ (प्रांत), जो $n = \deg\bar Q + \deg\bar R$ के बराबर
होना चाहिए, जिससे दोनों के लिए $\operatorname{val} = \deg$ अनिवार्य हो जाता है: दोनों
एकपदी हैं। ऊपर की तरह घातें नहीं घटतीं, अतः $Q$ और $R$ के
अचर पद $Q(0), R(0)$ $p$ से विभाज्य हैं --- दोनों, क्योंकि दोनों
अपचयन घात $\geq 1$ के एकपदी हैं। तब $p^2 \mid
Q(0)R(0) = a_0$: विरोधाभास।
\end{proof}

\begin{example}\label{ex:b3:rings:eisenstein}
प्रत्येक अभाज्य $p$ और $n
\geq 1$ के लिए $X^n - p$ $\Q$ पर \omterm{def:b3:rings:divisibility}{अखंडनीय} है
($p$ पर \omterm{thm:b3:rings:criteria}{आइज़ेनश्टाइन}): अर्थात् $\Q$ पर प्रत्येक घात के \omterm{def:b3:rings:divisibility}{अखंडनीय}
बहुपद हैं --- यह $\C$ (घात $1$, दालांबेर--गाउस, \cref{ch:b3:holomorphic} में
सिद्ध) और $\R$ (घात $1, 2$) से एकदम विपरीत है। \emph{विस्थापन}
की युक्ति \omterm{thm:b3:rings:criteria}{आइज़ेनश्टाइन} की पहुँच बढ़ा देती है: $p$-वाँ
\emph{वृत्तखंडी बहुपद}\index{वृत्तखंडी बहुपद} $\Phi_p = X^{p-1} +
\dots + X + 1 = \frac{X^p - 1}{X - 1}$ इस प्रकार है
\[
\Phi_p(X + 1) = \frac{(X+1)^p - 1}{X}
= X^{p-1} + \binom{p}{1}X^{p-2} + \dots + \binom{p}{p-1},
\]
और $p$ पर \omterm{thm:b3:rings:criteria}{आइज़ेनश्टाइन} लागू होता है ($0 < k < p$ के लिए $p \mid \binom pk$, तथा
$\binom{p}{p-1} = p \not\equiv 0 \bmod p^2$): अतः $\Phi_p(X+1)$, और इसलिए $\Phi_p$, $\Q$ पर \omterm{def:b3:rings:divisibility}{अखंडनीय} है। यही
\cref{ch:b3:galois} में सुनाई गई $17$-भुज की कथा का बीजीय मर्म है।
\end{example}

\begin{method}\label{met:b3:rings:irreducibility}
$\Q$ पर $P \in \Z[X]$ को \omterm{def:b3:rings:divisibility}{अखंडनीय} सिद्ध करने के लिए: (क) $P$ को आद्य
बनाइए; (ख) $P(X)$ पर और विस्थापनों $P(X \pm
1)$ पर \omterm{thm:b3:rings:criteria}{आइज़ेनश्टाइन} आज़माइए;
(ग) अग्र गुणांक को न विभाजित करने वाले छोटे अभाज्यों के सापेक्ष
अपचयन आज़माइए --- \emph{एक} $p$ के सापेक्ष अखंडनीयता पर्याप्त
है, और $\mathbb F_p$ पर अखंडनीयता एक परिमित जाँच है (मूल न होना घात-$1$
गुणनखंडों को अपवर्जित कर देता है; फिर प्रत्येक घात $\leq \deg P/2$ के
परिमित गुणनखंडों की जाँच कीजिए); (घ) और यदि सब कुछ विफल हो जाए,
तो अनिर्धारित गुणांक। सावधान: प्रत्येक $p$ के सापेक्ष
खंडनीयता का अर्थ $\Q$ पर खंडनीयता \emph{नहीं} है (\cref{exo:b3:rings:11})।
\end{method}

\section{नोएथरीय वलय}

\begin{definition}\label{def:b3:rings:noetherian}
कोई वलय $A$ \emph{नोएथरीय}\index{नोएथरीय वलय} कहलाता है यदि
$A$ का प्रत्येक \omterm{def:b3:rings:ideal}{आदर्श} परिमिततः जनित हो।
\end{definition}

\begin{proposition}\label{prop:b3:rings:noethacc}
$A$ \omterm{def:b3:rings:noetherian}{नोएथरीय} है तभी और केवल तभी जब आदर्शों का प्रत्येक बढ़ता हुआ
अनुक्रम अंततः अचर हो जाए (\emph{आरोही शृंखला शर्त}), और तभी और
केवल तभी जब आदर्शों के प्रत्येक अरिक्त कुल में (अंतर्वेशन के
सापेक्ष) कोई \omterm{def:b3:rings:primemaximal}{महत्तम} अवयव हो।
\end{proposition}

\begin{proof}
\emph{(परिमिततः जनित $\Rightarrow$ आरोही शृंखला शर्त)}: किसी शृंखला
$I_1 \subseteq I_2
\subseteq \cdots$ के लिए सम्मिलन $I$ एक \omterm{def:b3:rings:ideal}{आदर्श} है, जो $x_1,
\dots, x_r$ से जनित है;
सभी $x_i$ किसी एक $I_N$ में स्थित हैं, अतः $n \geq N$ के लिए $I = I_N = I_n$।
\emph{(आरोही शृंखला शर्त $\Rightarrow$ \omterm{def:b3:rings:primemaximal}{महत्तम} अवयव)}: यदि किसी अरिक्त
कुल $\mathcal F$ में कोई \omterm{def:b3:rings:primemaximal}{महत्तम} अवयव न हो, तो $I_1 \in
\mathcal F$ चुनिए, फिर आगमन से
$\mathcal F$ में $I_{n+1} \supsetneq I_n$ (यह संभव है क्योंकि $I_n$ \omterm{def:b3:rings:primemaximal}{महत्तम} नहीं है): एक
अनंत कड़ाई से बढ़ती हुई शृंखला। (यहाँ आश्रित चयनों का स्वयंसिद्ध
प्रयुक्त होता है, जो चयन का एक दुर्बल रूप है और जिसकी हम चिंता
नहीं करते।)
\emph{(\omterm{def:b3:rings:primemaximal}{महत्तम} अवयव $\Rightarrow$ परिमिततः जनित)}: कोई \omterm{def:b3:rings:ideal}{आदर्श} $I$ दिया
हो, तो $I$ में अंतर्विष्ट परिमिततः जनित आदर्शों का कुल अरिक्त
है ($(0)$); उसका कोई \omterm{def:b3:rings:primemaximal}{महत्तम} अवयव $J = (x_1, \dots, x_r)$ अनिवार्यतः $I$ के
बराबर होगा: अन्यथा जनकों में $x \in I \setminus J$ जोड़ देने से उस कुल का कड़ाई
से बड़ा सदस्य बन जाता।
\end{proof}

\begin{theorem}[हिल्बर्ट की आधार प्रमेय]\label{thm:b3:rings:hilbert}
यदि $A$ \omterm{def:b3:rings:noetherian}{नोएथरीय} हो, तो $A[X]$ भी \omterm{def:b3:rings:noetherian}{नोएथरीय} है। अतः $A[X_1, \dots,
X_n]$ भी, और
उनका प्रत्येक भागफल भी।\index{हिल्बर्ट की आधार प्रमेय}
\end{theorem}

\begin{proof}
मान लीजिए $I$ $A[X]$ का एक \omterm{def:b3:rings:ideal}{आदर्श} है, और मान लीजिए $I$
परिमिततः जनित नहीं है। एक अनुक्रम बनाइए: न्यूनतम घात का $f_1 \in I \setminus \{0\}$,
और आगमन से न्यूनतम घात का $f_{k+1} \in I \setminus (f_1,
\dots, f_k)$ (परिकल्पना से यह समुच्चय अरिक्त
है)। घातें $d_k = \deg f_k$ अनवरोही हैं (प्रत्येक चयन की न्यूनतमता से:
$f_{k+1}$ चरण $k+1$ पर उपलब्ध था \dots ठीक-ठीक, $f_{k+1} \notin (f_1,\dots,f_k) \supseteq
(f_1, \dots, f_{k-1})$, अतः $f_{k+1}$
चरण $k$ पर प्रतिस्पर्धा में था और हार गया या बराबरी पर रहा:
$d_{k+1} \geq d_k$)। मान लीजिए $a_k \in A$ $f_k$ का अग्र गुणांक है। आदर्शों की
शृंखला $(a_1) \subseteq (a_1,
a_2) \subseteq \cdots$ स्थिर हो जाती है: किसी $n$ के लिए $a_{n+1} \in (a_1, \dots,
a_n)$, मान
लीजिए $a_{n+1} = \sum_{k\leq n} u_k a_k$। अब
\[
g = f_{n+1} - \sum_{k=1}^{n} u_k X^{\,d_{n+1} - d_k} f_k .
\]
पर विचार कीजिए। तब $g \in I \setminus (f_1, \dots, f_n)$ (योग \omterm{def:b3:rings:ideal}{आदर्श} में है, $f_{n+1}$ नहीं), फिर भी
घात $d_{n+1}$ का गुणांक निरस्त हो जाता है: $\deg g < d_{n+1}$, जो $d_{n+1} = \deg f_{n+1}$ की
न्यूनतमता का खंडन है।

इसे दोहराने पर $A[X_1, \dots, X_n] = (A[X_1, \dots, X_{n-1}])[X_n]$ \omterm{def:b3:rings:noetherian}{नोएथरीय} है; और कोई भागफल $A/I$ \omterm{def:b3:rings:noetherian}{नोएथरीय} है
क्योंकि उसके \omterm{def:b3:rings:ideal}{आदर्श} $J/I$ $A$ के आदर्शों तक उठ जाते हैं
(संगति), जहाँ परिमित संख्या में जनक जनकों पर प्रक्षेपित हो जाते
हैं।
\end{proof}

\begin{remark}\label{rem:b3:rings:noethmeaning}
नोएथरीयता बीजीय ज्यामिति का परिमितता स्वयंसिद्ध है: $n$ चरों
में बहुपद समीकरणों का कोई भी निकाय, चाहे कितना ही अनंत क्यों न
हो, उनमें से परिमित संख्या के तुल्य होता है --- उसका हल-समुच्चय
परिमित संख्या के बहुपदों से काटा जाता है। \omterm{def:b3:rings:pidufd}{PID} तुच्छ रूप से
\omterm{def:b3:rings:noetherian}{नोएथरीय} होते हैं; अनंत चरों में $\Z[X_1, X_2, \dots]$ \omterm{def:b3:rings:noetherian}{नोएथरीय} नहीं है ($(X_1) \subsetneq (X_1, X_2) \subsetneq \cdots$)।
अनोएथरीय वलय विश्लेषण में भी स्वाभाविक रूप से मिलते हैं: $\intcc01$
पर संतत फलन ऐसा ही एक वलय बनाते हैं (\cref{exo:b3:rings:10})।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b3:rings:1}
भागफलों को पहचानिए: (क) $\Z[X]/(X^2 + 1) \cong \Z[\iu]$; (ख) $\R[X]/(X^2+1) \cong \C$; (ग) $\mathbb F_2[X]/(X^2 + X + 1)$ $4$
अवयवों वाला क्षेत्र है --- उसकी गुणन सारणी लिखिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:rings:2}
(क) दर्शाइए कि किसी \omterm{def:b3:rings:pidufd}{UFD} में प्रत्येक \omterm{def:b3:rings:divisibility}{अखंडनीय अवयव} अभाज्य होता है।
(ख) दर्शाइए कि परिमित पूर्णांकीय प्रांत एक क्षेत्र होता है।
(ग) निष्कर्ष निकालिए कि किसी परिमित वलय में प्रत्येक \omterm{def:b3:rings:primemaximal}{अभाज्य आदर्श}
\omterm{def:b3:rings:primemaximal}{महत्तम} होता है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:rings:3}
$\Z[\iu\sqrt5]$ में: जाँचिए कि $3$, $1 + \iu\sqrt5$ और $1 -
\iu\sqrt5$ \omterm{def:b3:rings:divisibility}{अखंडनीय} हैं, कि
$9 = 3\cdot 3 = (2 +
\iu\sqrt5)(2 - \iu\sqrt5)$, और (\cref{prop:b3:rings:primeirred} के बाद) पुनः निष्कर्ष निकालिए कि $\Z[\iu\sqrt5]$ \omterm{def:b3:rings:pidufd}{UFD}
नहीं है। अद्वितीयता ठीक कहाँ विफल होती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:4}
(क) दर्शाइए कि $\Z[\iu]$ मानक $N(x + \iu y) =
x^2 + y^2$ के सापेक्ष \omterm{def:b3:rings:pidufd}{यूक्लिडीय} है: $a, b \neq 0$
दिए हों, तो $a/b \in \C$ के निकटतम $q \in \Z[\iu]$ चुनिए।
(ख) $\Z[\iu]^\times$ ज्ञात कीजिए।
(ग) $\Z[\iu\sqrt2]$ और $N(x + \iu y\sqrt2) =
x^2 + 2y^2$ के लिए वही प्रश्न। $\Z[\iu\sqrt5]$ के लिए यही तर्क
क्यों विफल हो जाता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:5}
मान लीजिए $A$ एक वलय है। (क) दर्शाइए कि यदि $x$ शून्यभावी
हो (किसी $n$ के लिए $x^n = 0$), तो $1 + x \in A^\times$। (ख) दर्शाइए कि यदि
$A$ एक प्रांत हो, तो $A[X]^\times = A^\times$; $\Z/4\Z$ पर एक प्रतिउदाहरण दीजिए।
(ग) दर्शाइए कि किसी प्रांत में $0, 1$ के अतिरिक्त कोई वर्गसम अवयव
($e^2 = e$) नहीं होता, और $0$ के अतिरिक्त कोई शून्यभावी अवयव नहीं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:6}
$A = K[X, Y]$ में: (क) दर्शाइए कि \omterm{def:b3:rings:ideal}{आदर्श} $(X, Y)$ \omterm{def:b3:rings:primemaximal}{महत्तम} है पर मुख्य नहीं
--- अतः $K[X,Y]$ एक ऐसा \omterm{def:b3:rings:pidufd}{UFD} है (\cref{thm:b3:rings:gaussufd}) जो \omterm{def:b3:rings:pidufd}{PID} नहीं है; (ख) $K[X, Y]/(Y - X^2)$
और $K[X,Y]/(XY - 1)$ को परिमेय फलनों के उपवलयों के रूप में पहचानिए; (ग) क्या
$(Y - X^2)$ अभाज्य है? \omterm{def:b3:rings:primemaximal}{महत्तम}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:7}
$\Q$ पर \omterm{def:b3:rings:divisibility}{अखंडनीय} है या नहीं: $X^5 - 12X^3 + 36X - 12$;\quad $X^4 + X + 1$ \emph{($2$ के
सापेक्ष अपचयन कीजिए)};\quad $X^4 + 4$;\quad $\Phi_8 = X^4 + 1$ \emph{($1$ से
विस्थापित कीजिए)};\quad $X^3 - X - 1$।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:8}
(क) \cref{thm:b3:rings:crt} से सिद्ध कीजिए कि ऑयलर का फलन सह-अभाज्य कोटियों पर
गुणनात्मक है और यह कि $\varphi(p^k) =
p^{k-1}(p-1)$; $\varphi(n) = n\prod_{p \mid n}(1 -

(ख) हल कीजिए: $x \equiv 2 \pmod 7$, $x \equiv 5 \pmod{11}$, $x

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:rings:9}
(शून्यमूलक) मान लीजिए $\operatorname{Nil}(A)$ शून्यभावी अवयवों का समुच्चय है।
(क) दर्शाइए कि $\operatorname{Nil}(A)$ एक \omterm{def:b3:rings:ideal}{आदर्श} है जो प्रत्येक \omterm{def:b3:rings:primemaximal}{अभाज्य आदर्श} में
अंतर्विष्ट है। (ख) विलोमतः, मान लीजिए $a$ अशून्यभावी है;
$S =
\{a^n : n \in \N\}$ से बचने वाले आदर्शों पर ज़ोर्न की प्रमेयिका लगाकर ऐसा
\omterm{def:b3:rings:primemaximal}{अभाज्य आदर्श} बनाइए जिसमें $a$ न हो। निष्कर्ष निकालिए:
\[
\operatorname{Nil}(A) = \bigcap_{\mathfrak p \text{ अभाज्य}}
\mathfrak p .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:rings:10}
(क) मान लीजिए $A$ \omterm{def:b3:rings:noetherian}{नोएथरीय} है और $f \colon A \to A$ एक आच्छादक वलय
समाकारिता। दर्शाइए कि $f$ एकैकी है। \emph{($\ker
f \subseteq \ker f^2 \subseteq \cdots$ पर विचार
कीजिए।)}
(ख) दर्शाइए कि संतत फलनों का वलय $\mathcal C(\intcc01, \R)$ \omterm{def:b3:rings:noetherian}{नोएथरीय} नहीं है।
\emph{($I_n = \{f : f = 0
\intcc0{1/n} \text{ पर}\}$ पर विचार कीजिए।)}
(ग) दर्शाइए कि किसी \omterm{def:b3:rings:noetherian}{नोएथरीय} \emph{प्रांत} में प्रत्येक शून्येतर
अनैकिक अवयव \omterm{def:b3:rings:divisibility}{अखंडनीय} अवयवों का (परिमित) गुणनफल होता है --- अतः
$\Z[\iu\sqrt 5]$ की अगुणनखंडनीयता केवल \emph{अद्वितीयता} की विफलता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:rings:11}
मान लीजिए $P = X^4 + 1$।
(क) दर्शाइए कि $P$ $\Q$ पर \omterm{def:b3:rings:divisibility}{अखंडनीय} है (\cref{exo:b3:rings:7})।
(ख) दर्शाइए कि $P$ \emph{प्रत्येक} अभाज्य $p$ के सापेक्ष
खंडनीय है: $p = 2$ की स्थिति लीजिए; फिर विषम $p$ के लिए दर्शाइए
कि $8 \mid p^2 - 1$ और फिलहाल यह स्वीकार कीजिए (\cref{ch:b3:galois} में सिद्ध) कि $p^2$
अवयवों वाले क्षेत्र का गुणनात्मक समूह चक्रीय है, जिससे निष्कर्ष
निकले कि $p$ के सापेक्ष $P$ दो द्विघात गुणनखंडों में विभाजित
हो जाता है; जब $-1$, $2$, $-2$ में से कोई एक $p$ के सापेक्ष
वर्ग हो तब उन्हें स्पष्ट रूप से लिखिए, और दर्शाइए कि उनमें से एक
सदैव वर्ग होता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:12}
(वर्गसम अवयव वलयों को तोड़ते हैं) किसी क्रमविनिमेय वलय $A$ का
अवयव $e$ \emph{वर्गसम} कहलाता है यदि $e^2 = e$ हो।
(क) दर्शाइए कि यदि $e$ वर्गसम हो, तो $1 - e$ भी वर्गसम है, और यह
कि प्रतिचित्रण $x \mapsto (ex, (1-e)x)$ एक वलय तुल्याकारिता $A
\cong Ae \times A(1-e)$ है, जहाँ $Ae$
इकाई $e$ वाला वलय है।
(ख) किसी प्रांत के, और $\Z/12\Z$ के, समस्त वर्गसम अवयव ज्ञात कीजिए;
तुल्याकारिता $\Z/12\Z \cong \Z/4\Z \times \Z/3\Z$ के दो अतुच्छ वर्गसम अवयव बताकर उसे प्रस्तुत
कीजिए।
(ग) दर्शाइए कि $\Z/n\Z$ का चीनी शेषफल विघटन (\cref{ex:b3:rings:crtz}) ठीक उन वर्गसम
अवयवों $e_i \equiv 1 \bmod p_i^{a_i}$, $e_i \equiv 0$ के अनुरूप है जो अन्य अभाज्य घातों के सापेक्ष
लिए गए हैं: वलय अपने वर्गसम अवयवों के अनुदिश उसी प्रकार विघटित
होते हैं जैसे समष्टियाँ प्रक्षेपों के अनुदिश।
\end{exercise}

\section{समस्या: फर्मा की दो-वर्ग प्रमेय}

\begin{problem}[{सप्ताहांत समस्या --- दो वर्गों के योग, $\Z[\iu]$ के
माध्यम से}]\label{pb:b3:rings:1}
कौन-से पूर्णांक दो वर्गों के योग हैं? फर्मा का उत्तर (1640)
अंकगणित के रत्नों में से एक है; गाउसीय पूर्णांक उसकी उपपत्ति को
वलय सिद्धांत में बदल देते हैं। सर्वत्र $N(x + \iu y) = x^2 + y^2$ मानक को निरूपित
करता है, $\Z[\iu]$ \omterm{def:b3:rings:pidufd}{यूक्लिडीय} है (\cref{exo:b3:rings:4}), अतः \omterm{def:b3:rings:pidufd}{PID} और \omterm{def:b3:rings:pidufd}{UFD} भी, और
\emph{गाउसीय अभाज्य} से अभिप्राय है $\Z[\iu]$ का अभाज्य (= \omterm{def:b3:rings:divisibility}{अखंडनीय})
अवयव।

\medskip
\noindent\textbf{भाग I --- मानक और गाउसीय अभाज्य।}
\begin{enumerate}
  \item $N(zw) = N(z)N(w)$ सत्यापित कीजिए, $\Z[\iu]^\times =
        \{\pm1, \pm\iu\}$ पुनः निष्कर्ष के रूप में
        निकालिए, और \emph{ब्रह्मगुप्त सर्वसमिका} सिद्ध कीजिए:
        दो वर्गों के दो योगों का गुणनफल दो वर्गों का योग होता है।
  \item दर्शाइए कि यदि $N(z)$ एक अभाज्य संख्या हो, तो $z$ एक
        गाउसीय अभाज्य है।
  \item दर्शाइए कि प्रत्येक गाउसीय अभाज्य $\pi$ ठीक एक अभाज्य
        संख्या $p$ को विभाजित करता है \emph{($N(\pi) = \pi\bar\pi$ पर विचार
        कीजिए)}, और तब $N(\pi) \in \{p, p^2\}$।
  \item द्विभाजन का निष्कर्ष निकालिए: प्रत्येक अभाज्य $p$ के
        लिए या तो $p$ $\Z[\iu]$ में अभाज्य बना रहता है (और मानक
        $p$ का कोई गाउसीय अभाज्य नहीं होता), अथवा $p = \pi\bar\pi$, जहाँ
        $\pi$ मानक $p$ का गाउसीय अभाज्य है --- और तब $p = a^2 + b^2$।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- विल्सन की प्रमेय और $p$ के सापेक्ष
$-1$।}
\begin{enumerate}[resume]
  \item \emph{विल्सन की प्रमेय} सिद्ध कीजिए: अभाज्य $p$ के लिए
        $(p-1)!
        \equiv -1 \pmod p$। \emph{(प्रत्येक शेष को उसके प्रतिलोम के साथ
        युग्मित कीजिए; कौन-से अपने ही साथ युग्मित होते हैं?)}
  \item मान लीजिए $p$ एक विषम अभाज्य है और $m = \frac{p-1}2$। दर्शाइए कि
        $(m!)^2 \equiv (-1)^{m+1} \pmod p$ \emph{($(p-1)!$ में प्रत्येक गुणनखंड $k > m$ के स्थान
        पर $-(p - k)$ रखिए)}।
  \item निष्कर्ष निकालिए: $-1$ $p$ के सापेक्ष वर्ग है तभी और
        केवल तभी जब $p = 2$ या $p
        \equiv 1 \pmod 4$। \emph{(``केवल तभी'' के लिए:
        यदि $x^2
        \equiv -1$ हो, तो $(\Z/p\Z)^\times$ में $x$ का क्रम क्या है, और
        लाग्राँज क्या कहती है?)}
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- विभाजन नियम।}
\begin{enumerate}[resume]
  \item मान लीजिए $p \equiv 1 \pmod 4$, और $p \mid x^2 + 1 =
        (x + \iu)(x - \iu)$ के साथ $x$। दर्शाइए कि
        $p$ गाउसीय अभाज्य \emph{नहीं} है, और भाग I से निष्कर्ष
        निकालिए: $p = a^2 + b^2$।
  \item मान लीजिए $p \equiv 3 \pmod 4$। सीधे दर्शाइए कि $p$ दो वर्गों का योग
        नहीं है \emph{($4$ के सापेक्ष वर्ग)}, और इससे निष्कर्ष
        निकालिए कि $p$ गाउसीय अभाज्य बना रहता है।
  \item $p = 2$ की स्थिति तय कीजिए: गुणनखंडन $2 =
        -\iu(1+\iu)^2$ प्रस्तुत कीजिए
        और जाँचिए कि $1 + \iu$ एक गाउसीय अभाज्य है। ($2$ एकमात्र
        \emph{शाखित} अभाज्य है: वह किसी इकाई तक किसी गाउसीय
        अभाज्य के वर्ग से विभाज्य है।)
  \item इकाइयों तक गाउसीय अभाज्यों का वर्गीकरण संकलित कीजिए:
        $1 + \iu$; पूर्णांक $p \equiv 3 \pmod 4$; और मानक $p \equiv 1 \pmod
        4$ के संयुग्मी युग्म
        $\pi, \bar\pi$। इसे $5 = (2+\iu)(2-\iu)$ पर और $3$ पर सत्यापित कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- दो-वर्ग प्रमेय।}
\begin{enumerate}[resume]
  \item सीधा अर्धांश सिद्ध कीजिए: यदि गुणनखंडन $n =
        \prod p_i^{\alpha_i}$ में
        प्रत्येक अभाज्य $\equiv 3 \pmod 4$ सम घातांक के साथ आए, तो $n$ दो
        वर्गों का योग है। \emph{(ब्रह्मगुप्त + भाग II और III।)}
  \item विलोम सिद्ध कीजिए: यदि $n = a^2 + b^2 = N(a + \iu b)$ हो और $q \equiv 3 \pmod 4$ $n$ को
        विभाजित करे, तो दर्शाइए कि गाउसीय अभाज्य $q$ $a + \iu b$ या
        $a - \iu b$ को विभाजित करता है, कि वस्तुतः वह $a$ और $b$
        दोनों को विभाजित करता है, और $n$ पर आगमन से निष्कर्ष
        निकालिए कि $n$ में $q$ का घातांक सम है।
  \item अंतिम प्रमेय का कथन दीजिए। $2025$, $2026$, $2027$ में से
        कौन-से दो वर्गों के योग हैं? \emph{($2025 = 81 \cdot 25$; $2026 = 2 \cdot 1013$, $1013$
        अभाज्य; $2027$ अभाज्य।)}
  \item (उपसंहार) दर्शाइए कि कोई अभाज्य $p \equiv 1 \pmod 4$ मूलतः अद्वितीय रूप
        से दो वर्गों का योग होता है: यदि $p =
        a^2 + b^2 = c^2 + d^2$ (धन पूर्णांक) हो,
        तो $\{a, b\}
        = \{c, d\}$। \emph{($\Z[\iu]$ में गुणनखंडन की अद्वितीयता।)}
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- निरूपणों की गणना: याकोबी का सूत्र और
लाइब्नित्स की श्रेणी।} $r_2(n) = \#\{(a, b) \in
\Z^2 : a^2 + b^2 = n\}$ लिखिए (क्रमित युग्म, चिह्नों और
शून्यों सहित), और मान लीजिए $\chi$ $4$ के सापेक्ष अतुच्छ वर्ण
है: $d \equiv 1$ हो तो $\chi(d) = +1$, $d \equiv 3 \pmod4$ हो तो $-1$, और $d$ सम हो तो
$0$।
\begin{enumerate}[resume]
  \item (तुलना के लिए एक प्रारंभिक प्रश्न) कौन-से पूर्णांक दो
        वर्गों के \emph{अंतर} हैं? दर्शाइए: $a, b \in \Z$ के साथ $n = a^2 - b^2$
        तभी और केवल तभी जब $n \not\equiv 2 \pmod 4$ --- इसमें वलय सिद्धांत की कोई
        आवश्यकता नहीं, और आगे आने वाली संरचना जैसा कुछ भी नहीं।
  \item दर्शाइए कि $r_2(n)$ $N(z) = n$ वाले $z \in \Z[\iu]$ की संख्या है। $p_j \equiv 1$,
        $q_k \equiv 3
        \pmod4$ के साथ $n = 2^{a}\prod_jp_j^{b_j}
        \prod_kq_k^{c_k}$ लिखकर, प्रश्न 11 के वर्गीकरण और
        अद्वितीय गुणनखंडन का उपयोग करके दर्शाइए: ऐसे $z$ तभी
        और केवल तभी विद्यमान हैं जब सभी $c_k$ सम हों, और उस
        स्थिति में
        \[
        r_2(n) = 4\prod_j\,(b_j + 1) .
        \]
        \emph{(गणना: $z = u\,(1+\iu)^{a}\prod_j\pi_j^{s_j}
        \bar\pi_j^{\,b_j - s_j}\prod_kq_k^{c_k/2}$, जहाँ $u$ एक इकाई है और $0 \leq s_j \leq b_j$;
        यह सूची निःशेष और पुनरुक्ति-रहित क्यों है?)}
  \item दर्शाइए कि $d \mapsto \chi(d)$ पूर्णतः गुणनात्मक है, इससे निष्कर्ष
        निकालिए कि $n \mapsto
        \sum_{d \mid n}\chi(d)$ गुणनात्मक है, और उसे अभाज्य घातों पर
        परिकलित कीजिए: वह $2^a$ पर $1$ है; $p^b$ पर $b + 1$ है
        ($p \equiv 1$); और $q^c$ पर ($q \equiv 3$) $c$ के सम या विषम होने
        के अनुसार $1$ या $0$।
  \item \emph{याकोबी की प्रमेय} का निष्कर्ष निकालिए:
        \[
        r_2(n) = 4\sum_{d \mid n}\chi(d) =
        4\bigl(d_1(n) - d_3(n)\bigr),
        \]
        जहाँ $d_i(n)$ भाजकों $\equiv i \pmod 4$ की गणना करता है। $n = 3, 5, 9, 25$ पर
        सत्यापित कीजिए, और $65$ के $16$ निरूपण सूचीबद्ध
        कीजिए।
  \item (वृत्त) दर्शाइए कि $\sum_{n \leq x}r_2(n)$ त्रिज्या $\sqrt x$ के संवृत चक्र
        में $\Z^2$ के जालक बिंदुओं की संख्या है, और सिद्ध कीजिए
        \[
        \sum_{n\leq x}r_2(n) = \pi x + O(\sqrt x)
        \]
        \emph{(प्रत्येक जालक बिंदु एक इकाई वर्ग का स्वामी है;
        क्षेत्रफलों की तुलना कीजिए, त्रुटि चौड़ाई $O(1)$ के एक
        वलयाकार क्षेत्र में रहती है)}।
  \item (लाइब्नित्स, अंकगणितीय दृष्टि से) प्रश्न 19--20 को
        मिलाइए:
        \[
        \sum_{d \leq x}\chi(d)\Bigl\lfloor\frac
        xd\Bigr\rfloor = \frac{\pi x}4 + O(\sqrt x),
        \]
        और --- पूर्णांक भागों को सावधानी से हटाकर --- लाइब्नित्स
        की श्रेणी निकालिए
        \[
        1 - \frac13 + \frac15 - \frac17 + \dots = \frac\pi4 .
        \]
        विषम व्युत्क्रमों की एकांतर श्रेणी \emph{वस्तुतः} भाजकों
        $\equiv 1$ का भाजकों $\equiv 3$ पर औसत आधिक्य है: विश्लेषण, जिसे
        अंकगणित ने परिकलित किया।
  \item (दो वर्गों के योग कितने विरल हैं?) दर्शाइए कि कोई भी
        पूर्णांक $\equiv 3 \pmod 4$ दो वर्गों का योग नहीं है (दो तरीके:
        $4$ के सापेक्ष वर्ग, अथवा प्रश्न 17 की सम-विषम कसौटी),
        अतः कम से कम एक-चौथाई पूर्णांक छूट जाते हैं; और दर्शाइए
        कि प्रश्न 20 का औसत $\frac1x\sum_{n\leq
        x}r_2(n) \to \pi$ निरूपणीय पूर्णांकों के घनत्व
        $0$ होने के अनुकूल है --- असामान्य रूप से बहुत अधिक
        निरूपण वाले पूर्णांक प्रस्तुत करके ($\equiv 1 \bmod 4$ जैसे अनेक
        अभाज्यों के गुणनफल लीजिए) समझाइए कि शून्य की ओर जाता हुआ
        अनुपात भी धनात्मक औसत कैसे धारण कर सकता है। (लांडाउ ने
        सिद्ध किया कि वास्तविक घनत्व $1/\sqrt{\log x}$ की तरह क्षीण होता है;
        वह हमारे साधनों से परे है, पर उसकी क्रियाविधि अब दिखाई
        देने लगी है।)
\end{enumerate}

\medskip
\noindent\textbf{भाग VI --- पूरक: आद्य निरूपण और पाइथागोरस।}
\begin{enumerate}[resume]
  \item किसी निरूपण $n = a^2 + b^2$ को \emph{आद्य} कहिए यदि $\gcd(a, b) = 1$ हो।
        दर्शाइए कि $n \geq 1$ आद्य निरूपण स्वीकार करता है तभी और केवल
        तभी जब $4 \nmid n$ हो और कोई अभाज्य $q \equiv 3 \pmod 4$ $n$ को विभाजित न
        करे। \emph{(आवश्यकता के लिए प्रश्न 13 का अवरोहण और
        $4$ के सापेक्ष वर्ग पुनः प्रयोग कीजिए; पर्याप्तता के
        लिए $z$ को केवल $1 + \iu$ और $\pi_j$ से बनाइए --- संयुग्मी
        बिना --- और समझाइए कि $a$ तथा $b$ का कोई उभयनिष्ठ
        अभाज्य गुणनखंड $\pi_j$ और $\bar\pi_j$ दोनों को, अथवा $(1+\iu)^2$ को,
        $z$ में क्यों धकेल देता।)}
  \item (पाइथागोरसीय त्रिक) मान लीजिए $a^2 + b^2 = c^2$, जहाँ $a, b,
        c$ धनात्मक
        हैं, $\gcd(a, b) = 1$ और $b$ सम है। दर्शाइए कि $a + \iu b$ और $a - \iu b$
        $\Z[\iu]$ में सह-अभाज्य हैं \emph{(कोई उभयनिष्ठ गाउसीय
        अभाज्य भाजक $2a$ और $2b$ को विभाजित करता, जबकि $c$
        विषम है)}, अद्वितीय गुणनखंडन से निष्कर्ष निकालिए कि
        किसी इकाई $u$ के लिए $a + \iu b = u(m + \iu n)^2$, और शास्त्रीय प्राचलन
        निकालिए: $a$ और $b$ की अदला-बदली तक,
        \[
        a = m^2 - n^2, \qquad b = 2mn, \qquad c = m^2 + n^2,
        \]
        जहाँ $m > n \geq 1$ विपरीत सम-विषमता वाले सह-अभाज्य हैं। $(m, n) =
        (2, 1)$
        और $(5, 2)$ से $(3, 4, 5)$ तथा $(21, 20, 29)$ पुनः प्राप्त कीजिए।
  \item (संख्यात्मक सत्यापन) $x = 25$ लीजिए। याकोबी के सूत्र से
        $1 \leq n \leq 25$ के लिए $r_2(n)$ परिकलित कीजिए, जाँचिए कि शून्येतर मान
        ठीक $n = 1,
        2, 4, 5, 8, 9, 10, 13, 16, 17, 18, 20, 25$ पर आते हैं, और यह भी कि
        \[
        \sum_{n \leq 25} r_2(n) = 80
        = 4\sum_{d \leq 25}\chi(d)
        \Bigl\lfloor\frac{25}d\Bigr\rfloor .
        \]
        सत्यापित कीजिए कि त्रिज्या $5$ के संवृत चक्र में $81$
        जालक बिंदु हैं, और उसकी तुलना $\pi x \approx 78.5$ से कीजिए: त्रुटि
        प्रश्न 20 के $O(\sqrt x)$ के भीतर ही रहती है।
\end{enumerate}
\end{problem}
