\section*{الفصل \ref{ch:b3:residues} --- متسلسلات لوران
ومبرهنة البواقي}

\begin{solution}{exo:b3:residues:1}
$\frac{\sin z}z = 1 - \frac{z^2}6 + \cdots$: قابل للإزالة، \omterm{def:b3:residues:singularities}{والباقي} $0$. $\frac{\eu^z - 1}{z^2} = \frac1z + \frac12 +
\frac z6 + \cdots$: قطب بسيط، \omterm{def:b3:residues:singularities}{والباقي} $1$.
$\frac1{z(z-1)^2}$: قطب بسيط عند $0$ \omterm{def:b3:residues:singularities}{بباقٍ} $\frac1{(0-1)^2} = 1$؛ وقطب مضاعف عند $1$ \omterm{def:b3:residues:singularities}{بباقٍ}
$\frac{\dd}{\dd z}\bigl(\frac1z\bigr)\big|_{z=1} = -1$.
$\frac{\cos z}{z^3} = \frac1{z^3} - \frac1{2z} + \cdots$: قطب من الرتبة $3$، \omterm{def:b3:residues:singularities}{والباقي} $-\frac12$. $\eu^{1/z} = \sum_{n\geq0}
\frac{z^{-n}}{n!}$: جوهري،
\omterm{def:b3:residues:singularities}{والباقي} $1$. و $\frac1{\sin z}$: أقطاب بسيطة عند $n\pi$؛
\omterm{def:b3:residues:singularities}{والباقي} $\frac1{\cos 0} = 1$ عند $0$، و $\frac1{\cos\pi} = -1$ عند $\pi$ (\cref{met:b3:residues:compute}، أي
$g/h'$).
\end{solution}

\begin{solution}{exo:b3:residues:2}
مع $z = \eu^{\iu t}$ و $\cos t = \frac{z + z^{-1}}2$ و $\dd t
= \frac{\dd z}{\iu z}$:
\[
\int_0^{2\pi}\frac{\dd t}{a + \cos t}
= \oint_{\abs z = 1}\frac{2\,\dd z}{\iu\,(z^2 + 2az + 1)} .
\]
ويحقق الجذران $z_\pm = -a \pm \sqrt{a^2 - 1}$ أن $z_+z_- =
1$ مع $\abs{z_+} < 1 < \abs{z_-}$؛ \omterm{def:b3:residues:singularities}{والباقي} عند $z_+$ هو
$\frac{1}{z_+ - z_-} = \frac1{2\sqrt{a^2-1}}$، ومنه يكون التكامل $\frac2\iu\cdot2\iu\pi\cdot\frac1{2\sqrt{a^2-1}} =
\frac{2\pi}{\sqrt{a^2-1}}$. وحين $a \to 1^+$ ينفجر (إذ
تبلغ الدالة المكامَلة ذروتها عند $t = \pi$)؛ وحين
$a \to \infty$ يسلك مثل $\frac{2\pi}a$، وهو يوافق $\int\frac{\dd t}a$.
\end{solution}

\begin{solution}{exo:b3:residues:3}
التكامل الأول: الأقطاب العليا للمقدار $\frac{z^2}{1+z^6}$ عند $p =
\eu^{\iu\pi/6}, \iu, \eu^{5\iu\pi/6}$؛ وعند
كلٍّ منها، $\operatorname{Res} = \frac{p^2}{6p^5} = \frac{p^3}{6p^6} =
-\frac{p^3}6$، ويأخذ $p^3$ القيم $\iu, -\iu, \iu$: فمجموع
البواقي $-\frac{\iu}{6}$. والقوس $O(R^{-4})\cdot O(R) \to 0$:
\[
\int_\R\frac{x^2\,\dd x}{1 + x^6} =
2\iu\pi\Bigl(-\frac \iu6\Bigr) = \frac\pi3 .
\]
والثاني: قطب مضاعف عند $\iu$ للمقدار $\frac1{(1+z^2)^2} =
\frac1{(z-\iu)^2(z+\iu)^2}$:
\[
\operatorname{Res} = \frac{\dd}{\dd z}\,(z +
\iu)^{-2}\Big|_{z=\iu} = \frac{-2}{(2\iu)^3} =
\frac{-2}{-8\iu} = -\frac\iu4,
\qquad
\int_\R\frac{\dd x}{(1+x^2)^2} =
2\iu\pi\cdot\Bigl(-\frac\iu4\Bigr) = \frac\pi2 ,
\]
وهو متسق مع \cref{exo:b3:fouriertransform:3}(b).
\end{solution}

\begin{solution}{exo:b3:residues:4}
نغلق $\frac{\eu^{\iu tz}}{z^2 + a^2}$ في نصف المستوي العلوي (حيث $t \geq 0$): فهناك
$\abs{\eu^{\iu tz}} =
\eu^{-t\operatorname{Im}z} \leq 1$، ومنه يسهم القوس بالمقدار $O(R^{-2})\cdot O(R) \to 0$. والقطب الوحيد المحاط
$\iu a$ بسيط وباقيه $\frac{\eu^{-at}}{2\iu a}$:
\[
\int_\R\frac{\eu^{\iu tx}}{x^2 + a^2}\dd x =
\frac{\pi}{a}\,\eu^{-at},
\qquad\text{ومنه}\qquad
\int_\R\frac{\cos(tx)}{x^2+a^2}\dd x = \frac\pi
a\,\eu^{-a\abs t}
\]
(بالجزء الحقيقي؛ وهو زوجي في $t$). وهذا هو نظير القلب للمقدار
$\widehat{\eu^{-a\abs x}} = \frac{2a}{a^2+\xi^2}$ (\cref{exo:b3:fouriertransform:1}): فالحسابان يؤكد كلٌّ منهما الآخر عبر
\cref{thm:b3:fouriertransform:inversion}.
\end{solution}

\begin{solution}{exo:b3:residues:5}
بالكسور الجزئية: $f = \frac1{z-2} - \frac1{z-1}$. وعلى $\abs z < 1$ (بتايلور): $f = \sum_{n\geq0}\bigl(1 - 2^{-n-1}\bigr)z^n$.
وعلى $1 < \abs z < 2$: $\frac1{z-2} =
-\sum_{n\geq0}\frac{z^n}{2^{n+1}}$ و $-\frac1{z-1} =
-\sum_{n\geq1}z^{-n}$: أي متسلسلة ذات طرفين
فعلًا. وعلى $\abs z > 2$: $f = \sum_{n\geq1}\bigl(2^{n-1} -
1\bigr)z^{-n}$. وتختلف الثلاثة لأن نشرات
لوران مرتبطة بحلقات \emph{التيجان} لا بالنقاط: فلكل منطقة نشراتها
الهندسية الخاصة. ومعاملا $z^{-1}$ ($-1$ و $0$ على التوالي)
تكاملان على دوائر تحيط \emph{بالشذوذات الداخلية}، لا بواقٍ عند
$0$ (إذ $f$ \omterm{def:b3:holomorphic:holo}{هولومورفية} عند $0$): فمن أجل $1 < \abs z < 2$ يكون
المعامل $-1$ هو $\operatorname{Res}(f, 1)$؛ ومن أجل $\abs z > 2$ يكون المعامل $0$
هو $\operatorname{Res}(f,1) +
\operatorname{Res}(f,2) = -1 + 1$. وأما بواقي $f$: فهي $-1$ عند $1$ و $+1$ عند $2$.
\end{solution}

\begin{solution}{exo:b3:residues:6}
(a) \omterm{thm:b3:residues:laurent}{لمتسلسلة لوران} $\sum_nz^{-n}/n!$ عددٌ لانهائي من الحدود السالبة: أي
جوهري. وبحل $\eu^{1/z} = w$ (حيث $w \neq
0$): $\frac1z = \log\abs w + \iu\arg w + 2\iu\pi k$، ومنه
\[
z_k = \frac1{\log\abs w + \iu\arg w + 2\iu\pi k}
\xrightarrow[k\to\infty]{} 0 :
\]
فتُبلَغ كل قيمة غير معدومة عددًا لانهائيًّا من المرات بجوار $0$
--- وهو أقوى من الكثافة.
(b) على امتداد $z = 1/x$ حين $x \to +\infty$: $\eu^x \to \infty$؛ وعلى
امتداد $z = \iu/y$: الطويلة $1$. والقطب يتطلب $\abs{f(z)} \to \infty$ \emph{على
امتداد كل اقتراب}: وهنا لا توجد النهاية ببساطة، ولا حتى في
$[0, +\infty]$.
\end{solution}

\begin{solution}{exo:b3:residues:7}
(a) على $\abs z = 1$: $\abs{z^7 + z - 1} \leq 3 < 4 =
\abs{-4z^3}$. وبروشيه مع $f = -4z^3$ و $g = z^7 + z - 1$:
ثلاثة جذور في القرص.
(b) على $\abs z = 1$: $\abs{z^4 + 1} \leq 2 < 5 = \abs{5z}$: أي جذر واحد في $\abs z < 1$. وعلى
$\abs z = 2$: $\abs{5z + 1} \leq
11 < 16 = \abs{z^4}$: أي أربعة جذور في $\abs z < 2$. ومنه ثلاثة
جذور في التاج.
(c) من أجل $P = z^n + a_{n-1}z^{n-1} + \dots$: على $\abs z = R >
1 + \sum\abs{a_k}$، $\abs{P - z^n} \leq
\bigl(\sum\abs{a_k}\bigr)R^{n-1} < R^n = \abs{z^n}$: ومنه للمقدار $P$ عدد $n$ من
الجذور بالضبط في $D(0, R)$ --- أي دالمبير--غاوس بالترافق، بالعدّ
المحض.
\end{solution}

\begin{solution}{exo:b3:residues:8}
(a) لنفترض أن $f(a) = 0$ مع $f \not\equiv 0$: نختار $r$ بحيث
تكون $f$ عديمة الجذور على الدائرة $C = \partial D(a, r)$ (بعزل الجذور) و $m = \min_C\abs f > 0$.
وحسب \cref{thm:b3:holomorphic:weierstrassconv}، $f_n \to f$ و $f_n' \to f'$ بانتظام على $C$؛
ومن أجل $n$ كبير، $\abs{f_n}
\geq m/2$ على $C$، ومنه
\[
\frac1{2\iu\pi}\int_C\frac{f_n'}{f_n}
\longrightarrow \frac1{2\iu\pi}\int_C\frac{f'}{f} \geq 1
\]
(إذ تعدّ النهاية الجذرَ $a$؛ والتقارب لأن البسوط تتقارب بانتظام
والمقامات محدودة من الأسفل بانتظام). والطرف الأيسر عددٌ صحيح
يعدّ جذور $f_n$ في القرص: فيجب أن يكون $\geq 1$ في النهاية ---
وهو يناقض انعدام الجذور. ومنه تكون $f$ عديمة الجذور.

(b) نثبّت $w \in \Omega$ ونطبّق (a) على \omterm{def:b3:topology:topology}{المجموعة المفتوحة}
المترابطة $\Omega\setminus\{w\}$ (إذ نزع نقطة من جزء مفتوح مترابط من $\C$ يحفظ
الترابط) على $g_n(z) = f_n(z) -
f_n(w)$، وهي عديمة الجذور هناك بالتباين، ومتقاربة
إلى $g = f
- f(w)$. فإذا كانت $f$ غير ثابتة، فإن $g \not\equiv 0$ على
$\Omega\setminus\{w\}$، ومنه تكون $g$ عديمة الجذور هناك: أي $f(z) \neq
f(w)$ من أجل كل
$z \neq w$. وبما أن $w$ كيفية، تكون $f$ متباينة.
\end{solution}

\begin{solution}{exo:b3:residues:9}
تتألف حافة القطاع من $[0, R]$ ومن القوس $A_R$ ومن نصف المستقيم
$\eu^{2\iu\pi/n}[0, R]$ معكوسًا. ويقع في الداخل القطب الوحيد $p = \eu^{\iu\pi/n}$ للمقدار
$\frac1{1+z^n}$، وباقيه $\frac1{np^{n-1}} = \frac{p}{np^n} = -\frac pn$. وعلى نصف المستقيم الراجع، يعطي
$z = \eu^{2\iu\pi/n}x$ أن $z^n = x^n$ و $\dd z = \eu^{2\iu\pi/n}\dd x$؛ والقوس $O(R^{-n})\cdot O(R) \to 0$. ومنه
\[
\bigl(1 - \eu^{2\iu\pi/n}\bigr)
\int_0^\infty\frac{\dd x}{1 + x^n}
= 2\iu\pi\Bigl(-\frac{\eu^{\iu\pi/n}}n\Bigr),
\quad\text{ومنه}\quad
\int_0^\infty\frac{\dd x}{1+x^n}
= \frac{2\iu\pi}{n\,\bigl(\eu^{\iu\pi/n} -
\eu^{-\iu\pi/n}\bigr)} = \frac{\pi}{n\sin(\pi/n)} .
\]
ومن أجل $n = 2$: $\frac\pi{2\sin(\pi/2)} = \frac\pi2 =
[\arctan]_0^\infty$. وحين $n \to \infty$: تؤول القيمة إلى
$1$، وبالفعل تؤول الدالة المكامَلة إلى $\mathbf 1_{\intco01}$ (بهيمنة التقارب
المهيمن $\min(1, x^{-2})$ من أجل $n \geq 2$).
\end{solution}

\begin{solution}{exo:b3:residues:10}
على $\abs z = R$ كبير، $\abs f \leq C R^{-2}$: $\abs{\oint_{C_R}
f} \leq 2\pi R\cdot CR^{-2} \to 0$. لكن من أجل $R$ وراء جميع
الأقطاب، تعطي مبرهنة البواقي أن $\oint_{C_R}f =
2\iu\pi\sum_{\text{كل}p}\operatorname{Res}(f, p)$: أي إن المجموع الكلي
منعدم. ومن أجل $f = \frac1{z(z-1)(z-2)}$: البواقي $\frac1{(-1)(-2)} = \frac12$ عند $0$، و $\frac1{1\cdot(-1)} = -1$ عند $1$،
و $\frac1{2\cdot1} = \frac12$ عند $2$ --- ومجموعها $0$ كما تنبّأنا، و
\[
\frac1{z(z-1)(z-2)} = \frac{1/2}{z} - \frac1{z - 1} +
\frac{1/2}{z-2} :
\]
فالبواقي \emph{هي} معاملات الكسور الجزئية، وتوفّر متطابقة
المجموع المعدوم فحصَ اتساق مجانيًّا (أو تعيّن المعامل الأخير من
الباقين).
\end{solution}

\begin{solution}{exo:b3:residues:11}
على ثقب المفتاح، بالتعيين المختار: يكون فوق الشق مباشرةً
$\log z = \ln x$؛ وتحته مباشرةً $\log z =
\ln x + 2\iu\pi$. وتعطي القطع الأربع
\[
\Bigl(1 - \eu^{2\iu\pi(a-1)}\Bigr)\int_\varepsilon^R
\frac{x^{a-1}}{1+x}\dd x + \int_{C_R} + \int_{C_\varepsilon}
= 2\iu\pi\operatorname{Res}(f, -1) .
\]
وأما الأقواس: $\abs{f} \leq \frac{R^{a-1}}{R - 1}$ على $C_R$ وطوله $2\pi R$: فالإسهام $O(R^{a-1}) \to 0$
(لأن $a < 1$)؛ و $\abs f \leq \frac{\varepsilon^{a-1}}{1 - \varepsilon}$ على $C_\varepsilon$ وطوله $2\pi\varepsilon$:
$O(\varepsilon^a)
\to 0$ (لأن $a > 0$). وأما \omterm{def:b3:residues:singularities}{الباقي}: فعند $z = -1 = \eu^{\iu\pi}$، $\operatorname{Res} = \eu^{(a-1)\iu\pi}$. ومنه
\[
I(a) = \frac{2\iu\pi\,\eu^{\iu\pi(a-1)}}{1 -
\eu^{2\iu\pi(a-1)}}
= \frac{2\iu\pi}{\eu^{-\iu\pi(a-1)} - \eu^{\iu\pi(a-1)}}
= \frac{\pi}{-\sin(\pi(a-1))} = \frac{\pi}{\sin\pi a} .
\]
وأما انعكاس غاما: $B(a, 1-a) =
\int_0^1t^{a-1}(1-t)^{-a}\dd t$؛ ويحوّله التعويض $t =
\frac x{1+x}$ و $1 - t = \frac1{1+x}$
و $\dd t =
\frac{\dd x}{(1+x)^2}$ إلى $\int_0^\infty
\frac{x^{a-1}}{1+x}\dd x = I(a)$، وتعطي صيغة أولير
$B(a, 1-a) = \Gamma(a)\Gamma(1-a)/\Gamma(1)$
(\cref{pb:b3:lebesgue:1}) أن $\Gamma(a)\Gamma(1-a) = \frac\pi{\sin\pi a}$ --- وعلى وجه الخصوص $\Gamma(\tfrac12) = \sqrt\pi$ مرة أخرى.

\end{solution}

\begin{solution}{exo:b3:residues:12}
(a) على $\abs z = 1$: $\abs{z^4} = 1 < 7 \leq \abs{8z + 1}$ ($\abs{8z} - 1 = 7$): فللمقدار $P$ من الجذور في
$\mathbb D$ ما للمقدار $8z + 1$، أي جذرٌ \emph{واحد} (عند
$-\frac18$).

(b) على $\abs z = 3$: $\abs{8z + 1} \leq 25 < 81 =
\abs{z^4}$: وتعطي روشيه مقابل $z^4$ جميعَ
الجذور الأربعة في $\abs z < 3$، ومنه $4 - 1 = 3$ جذور في التاج
$1 <
\abs z < 3$. وأما التحديد الأدق: فجذرٌ بطويلة $\abs z = r \geq 2.1$ كان سيحقق
$r^4 = \abs{8z + 1} \leq 8r + 1$، لكن $r^4 -
8r - 1$ متزايد من أجل $r \geq 2$ ويساوي $19.45 -
16.8 - 1 = 1.65 > 0$
عند $r = 2.1$: وهو مستحيل. ومنه تقع الجذور الخارجية الثلاثة في
$1 < \abs z < 2.1$. (وعدديًّا: جذر حقيقي بجوار $-1.95$ وزوج مرافق بجوار
$1.04 \pm
1.73\iu$، طويلته $2.02$ --- ولهذا يجب أن تخفق محاولة روشيه عند
نصف القطر $2$ بالضبط: إذ تتطلب المبرهنة هيمنةً أكيدة، والجذور
تجلس خارجه بقليل.)

(c) لا جذور على $\iu\R$: $\operatorname{Re}P(\iu t) = t^4 +
1 \geq 1$. وأما الجذور في نصف المستوي
الأيمن: فنستعمل مبدأ العمدة على حافة نصف القرص $\{\abs z \leq
R,\ \operatorname{Re}z \geq 0\}$. وعلى
القوس الكبير، يدور $\arg P
\approx \arg z^4$ بمقدار $4\cdot\pi = 2\pi\cdot2$ (إذ يمتدّ القوس على
الزاوية $\pi$). وعلى امتداد المحور التخيّلي من $\iu R$ نزولًا
إلى $-\iu R$: يبقى $P(\iu t) = (t^4 + 1) + 8\iu t$ في نصف المستوي الأيمن ($\operatorname{Re} > 0$)، ومنه
تتغيّر $\arg P$ داخل $\intoo{-\pi/2}{\pi/2}$ وترجع بتغيّر صافٍ $\to 0$ حين
$R \to \infty$ (إذ الطرفان كلاهما $\approx \arg t^4 = 0$). فمجموع اللف:
$\frac{4\pi + 0}
{2\pi} = 2$: أي \emph{جذران} في نصف المستوي الأيمن --- وهو متسق مع
العدديات: إذ للزوج المرافق $\approx
1.04 \pm 1.73\iu$ جزء حقيقي موجب، والجذران
الحقيقيان $\approx -0.125$ و $\approx -1.96$ سالبان.
\end{solution}

\begin{solution}{pb:b3:residues:1}
\textbf{1.} للمقدار $\sin(\pi z)$ جذور بسيطة عند $\Z$ بالضبط
(إذ $\sin\pi z = 0$ إذا وفقط إذا كان $z \in \Z$، و $(\sin\pi z)' = \pi\cos\pi
z \neq 0$
هناك)، و $\cos(\pi n) \neq 0$: ومنه توجد للمقدار $\pi\cot(\pi z) =
\pi\cos(\pi z)/\sin(\pi z)$ أقطاب بسيطة عند $\Z$ مع
\[
\operatorname{Res}(\pi\cot\pi z,\ n)
= \frac{\pi\cos(\pi n)}{\pi\cos(\pi n)} = 1
\]
(بقاعدة $g/h'$، \cref{met:b3:residues:compute}).

\textbf{2.} المقدار $\frac{\sin(\pi z)}{\pi z} = 1 - \frac{(\pi z)^2}6
+ \frac{(\pi z)^4}{120} - \cdots$ هولومورفي وغير معدوم بجوار $0$:
ومنه يكون مقلوبه هولومورفيًّا (\cref{def:b3:holomorphic:holo}: خارج القسمة)،
بمتسلسلة $1 +
\frac{(\pi z)^2}{6} + \frac{7(\pi z)^4}{360} + \cdots$ (بالتعيين: بمعاملات من نمط $(1 - u)^{-1}$
انطلاقًا من $u =
\frac{(\pi z)^2}6 - \frac{(\pi z)^4}{120}$: فمعامل $z^4$ هو $\frac1{36} - \frac1{120} = \frac{7}{360}$). ثم نضرب في
$\cos(\pi z) = 1 - \frac{(\pi z)^2}2 + \frac{(\pi
z)^4}{24} - \cdots$ ونقسم على $z$:
\[
\pi\cot(\pi z) = \frac1z\Bigl[1 + \pi^2z^2\Bigl(\frac16 -
\frac12\Bigr) + \pi^4z^4\Bigl(\frac7{360} - \frac1{12} +
\frac1{24}\Bigr)\Bigr] + \cdots
= \frac1z - \frac{\pi^2}3\,z - \frac{\pi^4}{45}\,z^3 -
\cdots
\]
($\frac7{360} - \frac{30}{360} + \frac{15}{360} =
-\frac8{360} = -\frac1{45}$).

\textbf{3.} الأضلاع الشاقولية $z = \pm(N + \frac12) + \iu y$: بدورية $\cot$ بالدور
$\pi$، $\cot(\pi z) = \cot(\pm\frac\pi2
+ \iu\pi y) = -\tan(\iu\pi y) = -\iu\tanh(\pi y)$، وطويلته $\leq 1$. والأضلاع الأفقية $z = x \pm \iu(N + \frac12)$: من
$\abs{\cot(a + \iu b)}^2 = \frac{\cos^2a +
\sinh^2b}{\sin^2a + \sinh^2b} \leq \frac{1 +
\sinh^2b}{\sinh^2b} = \coth^2 b$،
\[
\abs{\cot(\pi z)} \leq \coth\bigl(\pi(N + \tfrac12)\bigr)
\leq \coth(\pi/2) < 1.1 .
\]
والحدّان كلاهما $\leq 2$.

\textbf{4.} نطبّق \cref{thm:b3:residues:residue} على $F(z) =
\pi\cot(\pi z)f(z)$ على $C_N$ (حيث $N$
وراء جميع أقطاب $f$):
\[
\frac1{2\iu\pi}\oint_{C_N}F = \sum_{n=-N}^{N}f(n) +
\sum_p\operatorname{Res}(F, p),
\]
إذ تسهم الأقطاب الصحيحة بالمقدار $f(n)$ (السؤال 1؛ و $f$
\omterm{def:b3:holomorphic:holo}{هولومورفية} هناك). وعلى $C_N$: $\abs F \leq 2\pi\cdot
C\abs z^{-2} \leq 2\pi C N^{-2}$، والمحيط $8(N +
\frac12)$: ومنه
يكون التكامل $O(1/N) \to 0$. ونجعل $N \to
\infty$: فنجد صيغة الجمع
المعروضة.

\textbf{5.} التناقص $\abs f = O(\abs z^{-2})$ ألغى \omterm{def:b3:holomorphic:contour}{تكامل المسار} \emph{وجعل}
$\sum\abs{f(n)}$ متقاربة. ومن أجل $f(z) = \frac1{z + \frac12}$: تتلسكب المجاميع
المتناظرة $\sum_{-N}^N\frac1{n + \frac12}$ إلى $0$ (إذ يتلاشى الحدّان $n$ و $-n - 1$)،
ويكون الطرف الأيمن $-\operatorname{Res}\bigl(\frac{\pi\cot\pi z}{z + \frac12},
-\frac12\bigr) = -\pi\cot(-\frac\pi2) = 0$: أي متسقًا --- لكن $\sum\abs{f(n)}$
تتباعد؛ فالطريقة لا تحسب سوى النهاية المتناظرة (أي القيمة
الرئيسية).

\textbf{6.} $g(z) = \frac{\pi\cot(\pi z)}{z^2}$: أقطاب عند الأعداد الصحيحة غير المعدومة
ببواقٍ $\frac1{n^2}$، وعند $0$ حيث، حسب السؤال 2،
\[
g(z) = \frac1{z^3} - \frac{\pi^2}{3z} - \frac{\pi^4}{45}z -
\cdots :
\qquad \operatorname{Res}(g, 0) = -\frac{\pi^2}3 .
\]
ويؤول \omterm{def:b3:holomorphic:contour}{تكامل المسار} على $C_N$ إلى $0$ كما في السؤال 4 ($\abs{g} = O(N^{-2})$
على $C_N$). ومنه $0 =
\sum_{n\neq0}\frac1{n^2} - \frac{\pi^2}3$: أي $2\zeta(2) =
\frac{\pi^2}3$، $\zeta(2) = \frac{\pi^2}6$.

\textbf{7.} $g(z) = \frac{\pi\cot(\pi z)}{z^4} = \frac1{z^5}
- \frac{\pi^2}{3z^3} - \frac{\pi^4}{45z} - \cdots$: \omterm{def:b3:residues:singularities}{الباقي} عند $0$ يساوي $-\frac{\pi^4}{45}$، و $0 =
2\zeta(4) - \frac{\pi^4}{45}$:
ومنه $\zeta(4) =
\frac{\pi^4}{90}$.

\textbf{8.} مع $g = \pi\cot(\pi z)/z^{2k}$: يكون \omterm{def:b3:residues:singularities}{الباقي} عند $0$ هو المعامل
$a_{2k-1}$ للمقدار $z^{2k-1}$ في نشر $\pi\cot(\pi z)$، ويعطي
انعدام \omterm{def:b3:holomorphic:contour}{تكامل المسار} $2\zeta(2k) + a_{2k-1} = 0$: أي $\zeta(2k) = -a_{2k-1}/2$، وهو مضاعف ناطق للمقدار
$\pi^{2k}$ لأن معاملات ظل التمام كذلك. وتعطي خطوةُ قسمة إضافية
$a_5 =
-\frac{2\pi^6}{945}$، ومنه $\zeta(6) = \frac{\pi^6}{945}$. وأما من أجل $\zeta(3)$: فالنواة الطبيعية
$g = \pi\cot(\pi z)/z^3$ تنتج $\sum_{n\neq0}\frac1{n^3} = 0$ بالفردية --- فتبرهن الطريقة على $0 = 0$
وتكون عمياء بنيويًّا عن قيم زيتا الفردية (إذ لا صيغة مغلقة
معروفة للمقدار $\zeta(3)$؛ ولاطباعيته، أبيري 1978، لزمت أفكار
مختلفة كليًّا).

\textbf{9.} للمقدار $F(z) = \frac{\pi\cot(\pi z)}{(z - w)(z + w)}$ أقطاب عند الأعداد الصحيحة (ببواقٍ
$\frac1{n^2 - w^2}$، ولاحظ الإشارة: $f(n) = \frac1{(n-w)(n+w)} = \frac1{n^2 -
w^2}$) وأقطاب بسيطة عند $\pm w$
ببواقٍ $\frac{\pi\cot(\pm\pi w)}{\pm2w} = \frac{\pi\cot(\pi
w)}{2w}$ لكلٍّ ($\cot$ فردية). وتعطي حجة السؤال 4
($\abs f = O(\abs z^{-2})$)
\[
\sum_{n\in\Z}\frac1{n^2 - w^2} + \frac{\pi\cot(\pi w)}{w} =
0,
\qquad\text{أي}\qquad
\pi\cot(\pi w) = \frac1w + \sum_{n\geq1}\frac{2w}{w^2 -
n^2},
\]
(إذ الحدّ $n = 0$ هو $-\frac1{w^2}$؛ وأعد تجميع $\pm n$). وأما
التقارب الناظمي على متراصات $\C\setminus\Z$: فمن أجل $\abs w \leq
R$
و $n \geq 2R$، $\abs{\frac{2w}{w^2 - n^2}} \leq
\frac{2R}{n^2 - R^2} \leq \frac{8R}{3n^2}$.

\textbf{10.} من أجل $\abs w \leq r < 1$: $\frac{2w}{w^2 - n^2} =
-\frac{2w}{n^2}\cdot\frac1{1 - w^2/n^2} =
-2\sum_{k\geq0}\frac{w^{2k+1}}{n^{2k+2}}$، مع $\abs{\text{الحدود}} \leq 2r^{2k+1}/n^{2k+2}$، وهي قابلة
للجمع على $(n, k)$: فتعيد فوبيني للمتسلسلات ترتيب
\[
\pi\cot(\pi w) = \frac1w -
2\sum_{k\geq0}\zeta(2k+2)\,w^{2k+1} .
\]
وبالمطابقة مع السؤال 2: $-2\zeta(2) = -\frac{\pi^2}3$ و $-2\zeta(4) = -\frac{\pi^4}{45}$ --- أي القيم نفسها.
وثلاثة طرق إلى $\frac{\pi^2}6$: \omterm{thm:b3:hilbert:parseval}{بارسيفال} (بمتسلسلات فورييه)،
وأثر مؤثر غرين للوتر، ونشرا ظل التمام؛ وأن يتوافق مجموعٌ على
الترددات، وأثرُ مؤثر، وتكاملُ مسار ليس صدفة --- فكلٌّ منها وجه
للمتطابقة الطيفية نفسها.

\textbf{11.} نثبّت $R \geq 1$ وليكن $n_0$ أصغر عدد صحيح
$\geq 2R$. ومن أجل $\abs z \leq R$ و $n \geq n_0$: $\abs{z^2/n^2} \leq \frac14$،
ومنه $1 - z^2/n^2 \in
\bar D(1, \frac14)$، حيث يكون اللوغاريتم الرئيسي هولومورفيًّا، و
\[
\Bigl|\log\Bigl(1 - \frac{z^2}{n^2}\Bigr)\Bigr|
\leq \sum_{k\geq1}\frac1k\,\Bigl|\frac{z^2}{n^2}\Bigr|^k
\leq \frac{\abs{z^2/n^2}}{1 - \abs{z^2/n^2}}
\leq \frac{2R^2}{n^2} :
\]
ويتقارب المجموع $S(z) = \sum_{n\geq n_0}\log(1 - z^2/n^2)$ تقاربًا ناظميًّا على $\bar D(0, R)$،
بمجاميع جزئية \omterm{def:b3:holomorphic:holo}{هولومورفية} $S_N$ ومع $\abs{S_N} \leq 2R^2\zeta(2)$ بانتظام. وبما أن
$\abs{\eu^a - \eu^b} \leq
\eu^{\max(\abs a,\abs b)}\abs{a - b}$ (بحدّ القيمة المتوسطة على القطعة)، تتقارب جداءات الذيل
$\prod_{n_0\leq n\leq N} =
\eu^{S_N}$ بانتظام على $\bar D(0, R)$ إلى $\eu^S$ العديم الجذور.
وبالضرب في كثير الحدود الثابت $z\prod_{n<n_0}(1 - z^2/n^2)$: يكون $P_N \to P$ بانتظام
على $\bar D(0, R)$، وتجعل \cref{thm:b3:holomorphic:weierstrassconv} الدالةَ $P$ \omterm{def:b3:holomorphic:holo}{هولومورفية}
هناك؛ وبكون $R$ كيفيًّا، تكون $P$ صحيحة. وعلى $\bar D(0, R)$
تكون جذور $P$ هي جذور المعامل الحدودي --- أي الأعداد الصحيحة
التي طويلتها $\leq R$، وكلٌّ منها بسيط (إذ توجد للمقدار $(1 - z/n)(1 + z/n)$ جذور بسيطة
متمايزة، ولا جذور للمقدار $\eu^S$): ومنه تكون مجموعة جذور $P$ هي
$\Z$، وكلها بسيطة.

\textbf{12.} بالاشتقاق اللوغاريتمي للجداء المنتهي، بعيدًا عن
جذوره:
\[
\frac{P_N'(z)}{P_N(z)} = \frac1z +
\sum_{n=1}^{N}\frac{-2z/n^2}{1 - z^2/n^2}
= \frac1z + \sum_{n=1}^{N}\frac{2z}{z^2 - n^2} .
\]
وعلى متراصة $K \subseteq \C\setminus\Z$: $P_N \to P$ و $P_N' \to P'$ بانتظام
(\cref{thm:b3:holomorphic:weierstrassconv})، و $\min_K
\abs P > 0$ (إذ لا تنعدم $P$ إلا على $\Z$)، ومنه
في النهاية $\abs{P_N} \geq \frac12\min_K\abs P$ و $P_N'/P_N \to
P'/P$ بانتظام على $K$. ويتقارب الطرف
الأوسط إلى
$\frac1z + \sum_{n\geq1}\frac{2z}{z^2-n^2} = \pi\cot(\pi z)$
حسب السؤال 9: ومنه $P'/P = \pi\cot(\pi z)$ على $\C\setminus\Z$.

\textbf{13.} الدالتان $\sin(\pi z)$ و $P$ صحيحتان بمجموعة
الجذور نفسها $\Z$، وكلها بسيطة (السؤالان 1 و11). وبجوار
$m \in \Z$ نكتب $\sin(\pi z) = (z - m)\,\sigma(z)$ و $P(z) = (z - m)\,\psi(z)$ مع $\sigma, \psi$
هولومورفيتين وغير منعدمتين عند $m$ (بتحليل متسلسلة القوى):
فيمتدّ $Q =
\sin(\pi z)/P = \sigma/\psi$ هولومورفيًّا وعديمَ الجذور عبر كل عدد صحيح،
وهو عديم الجذور على $\C\setminus\Z$ بوصفه خارج قسمة دالتين
عديمتَي الجذور. وهناك،
\[
\frac{Q'}{Q} = \frac{(\sin\pi z)'}{\sin\pi z} -
\frac{P'}{P} = \pi\cot(\pi z) - \pi\cot(\pi z) = 0 ,
\]
ومنه تنعدم الدالة الصحيحة $Q'$ على $\C\setminus\Z$، ومن ثَمّ في
كل مكان بالاتصال: فتكون $Q$ ثابتة. وحين $z \to
0$: $\sin(\pi z)/z \to \pi$
و $P(z)/z \to 1$، ومنه $Q =
\pi$:
\[
\sin(\pi z) = \pi z\prod_{n\geq1}
\Bigl(1 - \frac{z^2}{n^2}\Bigr) .
\]

\textbf{14.} عند $z = \frac12$: $1 = \sin\frac\pi2 =
\frac\pi2\prod_{n\geq1}\bigl(1 - \frac1{4n^2}\bigr) =
\frac\pi2\prod\frac{4n^2-1}{4n^2}$، ومنه
\[
\frac\pi2 = \prod_{n\geq1}\frac{4n^2}{4n^2 - 1}
= \lim_{N\to\infty}\prod_{n=1}^N
\frac{(2n)(2n)}{(2n-1)(2n+1)}
= \lim_{N\to\infty}
\frac{2\cdot2\cdot4\cdot4\cdots(2N)(2N)}
{1\cdot3\cdot3\cdot5\cdots(2N-1)(2N+1)} :
\]
أي جداء واليس، نتيجةً من سطر واحد لتحليل أولير.

\textbf{15.} من أجل $\abs z \leq r < 1$ يقع كل عامل في $D(1, r^2) \subseteq D(1, 1)$، ومنه
$P(z)/z = \exp\bigl(
\sum_{n\geq1}\log(1 - z^2/n^2)\bigr)$: فكل جداء جزئي هو أُسّي مجموع جزئي، ويمرّ الطرفان إلى
النهاية باتصال $\exp$. والمتسلسلة المزدوجة
\[
\sum_{n\geq1}\log\Bigl(1 - \frac{z^2}{n^2}\Bigr)
= -\sum_{n\geq1}\sum_{k\geq1}\frac{z^{2k}}{k\,n^{2k}}
= -\sum_{k\geq1}\frac{\zeta(2k)}k\,z^{2k}
= -\zeta(2)\,z^2 + O(z^4)
\]
تُعاد ترتيبها بفوبيني للمتسلسلات: $\sum_{n,k}
\frac{r^{2k}}{kn^{2k}} \leq \sum_k\zeta(2k)r^{2k} \leq
\zeta(2)\frac{r^2}{1-r^2} < \infty$. ومنه
\[
P(z) = z\,\exp\bigl(-\zeta(2)z^2 + O(z^4)\bigr)
= z - \zeta(2)\,z^3 + O(z^5) ,
\]
ويقارن السؤال 13 هذا مع $\sin(\pi z) = \pi z -
\frac{\pi^3}6z^3 + O(z^5)$: أي $\pi\zeta(2) = \frac{\pi^3}6$، أي $\zeta(2) = \frac{\pi^2}6$. فالوجه
الجمعي (السؤال 10) والوجه الضربي يحسبان العدد نفسه.

\textbf{16.} على متراصة $K \subseteq \C\setminus\Z$ تتقارب المجاميع الجزئية $S_N = \frac1z + \sum_{n\leq N}\bigl(
\frac1{z-n} + \frac1{z+n}\bigr)$
(السؤال 9، بحدود معاد تجميعها على الصورة $\frac{2z}{z^2-n^2} = \frac1{z-n} +
\frac1{z+n}$) بانتظام إلى
$\pi\cot(\pi z)$، ومنه تعطي \cref{thm:b3:holomorphic:weierstrassconv} أن $S_N' \to
(\pi\cot\pi z)' = -\pi^2/\sin^2(\pi z)$ بانتظام على
$K$. وبما أن $S_N' = -\sum_{\abs n\leq N}(z - n)^{-2}$:
\[
\frac{\pi^2}{\sin^2(\pi z)} =
\sum_{n\in\Z}\frac1{(z - n)^2} ,
\]
والتقارب ناظمي على متراصات $\C\setminus\Z$ (إذ الحدود
$O(n^{-2})$).

\textbf{17.} عند $z = \frac12$ يكون الطرف الأيسر $\pi^2$؛ وعلى
اليمين، $(\frac12 - n)^2 = \frac{(2n-1)^2}4$ مع $2n -
1$ يجول على جميع الأعداد الصحيحة
الفردية مرة واحدة بالضبط حين يجول $n$ على $\Z$:
\[
\pi^2 = \sum_{n\in\Z}\frac{4}{(2n-1)^2}
= 8\sum_{m\geq0}\frac1{(2m+1)^2},
\qquad
\sum_{m\geq0}\frac1{(2m+1)^2} = \frac{\pi^2}8 .
\]
وبالشطر حسب التماثل: $\zeta(2) = \frac{\pi^2}8 +
\sum_{n\geq1}\frac1{(2n)^2} = \frac{\pi^2}8 +
\frac{\zeta(2)}4$، ومنه $\frac34\zeta(2) = \frac{\pi^2}8$ و $\zeta(2) = \frac{\pi^2}6$ مرة أخرى.

\textbf{18.} مع $c = \cot\theta$ و $\cot(2\theta) =
\frac{c^2-1}{2c}$: $\cot\theta - 2\cot(2\theta) = c -
\frac{c^2-1}c = \frac1c = \tan\theta$. ومنه
$\pi\tan(\pi z)
= \pi\cot(\pi z) - 2\pi\cot(2\pi z)$، ويعطي السؤال 9 عند $z$ وعند $2z$
\[
\pi\cot(\pi z) = \frac1z +
\sum_{n\geq1}\frac{2z}{z^2 - n^2},
\qquad
2\pi\cot(2\pi z) = \frac1z +
\sum_{n\geq1}\frac{8z}{4z^2 - n^2} .
\]
والمتسلسلتان متقاربتان تقاربًا مطلقًا عند كل $z$ مثبَّت خارج
الأقطاب، ومنه يمكن إعادة تجميع الفرق كما نشاء: ففي المتسلسلة
الثانية تعطي الحدود الزوجية $n = 2m$ المقدارَ $\frac{8z}{4z^2 -
4m^2} = \frac{2z}{z^2 - m^2}$ وتلغي
المتسلسلة الأولى كليًّا، فيبقى
\[
\pi\tan(\pi z) = -\sum_{m\geq0}\frac{8z}{4z^2 - (2m+1)^2}
= \sum_{m\geq0}\frac{8z}{(2m+1)^2 - 4z^2} ,
\]
ناظميًّا على المتراصات التي تتفادى $\frac12 + \Z$ (إذ الحدود
$O(m^{-2})$).

\textbf{19.} من أجل $\abs z \leq r < \frac12$:
\[
\frac{8z}{(2m+1)^2 - 4z^2} = \frac{8z}{(2m+1)^2}
\sum_{k\geq0}\Bigl(\frac{4z^2}{(2m+1)^2}\Bigr)^{k},
\]
مع $\sum_{m,k}8r\,(4r^2)^k(2m+1)^{-2k-2} < \infty$ لأن $4r^2 < 1$: فتعيد فوبيني ترتيب المجموع المزدوج
إلى
\[
\pi\tan(\pi z) =
\sum_{k\geq0}8\cdot4^k\,\lambda(2k+2)\,z^{2k+1} .
\]
ومقابل $\pi\tan(\pi z) = \pi^2z + \frac{\pi^4}3z^3 +
O(z^5)$: يعطي معامل $z$ أن $8\lambda(2) = \pi^2$ --- أي السؤال 17 من
جديد --- ويعطي معامل $z^3$ أن $32\lambda(4) = \frac{\pi^4}3$، أي $\lambda(4) =
\frac{\pi^4}{96}$. وبإزالة
المقامات الزوجية، $\lambda(4) = \zeta(4) - 2^{-4}\zeta(4) =
\frac{15}{16}\zeta(4)$: أي $\zeta(4) = \frac{16}{15}\cdot
\frac{\pi^4}{96} = \frac{\pi^4}{90}$، وهو يوافق السؤال 7.

\textbf{20.} $\cot\frac\theta2 - \cot\theta =
\frac{\cos\frac\theta2\,\sin\theta -
\cos\theta\,\sin\frac\theta2}{\sin\frac\theta2\,\sin\theta}
= \frac{\sin\frac\theta2}{\sin\frac\theta2\,\sin\theta} =
\frac1{\sin\theta}$، والبسط هو $\sin(\theta -
\frac\theta2)$. ومع $\theta = \pi z$،
يُقرأ السؤال 9 عند $\frac z2$ على الصورة $\pi\cot\frac{\pi z}2 = \frac2z +
\sum_{n\geq1}\frac{4z}{z^2 - 4n^2}$، ومنه
\[
\frac{\pi}{\sin(\pi z)}
= \pi\cot\frac{\pi z}2 - \pi\cot(\pi z)
= \frac1z + \sum_{n\geq1}\frac{4z}{z^2 - 4n^2}
- \sum_{n\geq1}\frac{2z}{z^2 - n^2} .
\]
ويسمح التقارب المطلق بإعادة التجميع حسب التماثل: إذ يسهم
$n
= 2m$ الزوجي في المتسلسلة المطروحة بالمقدار $\frac{2z}{z^2-4m^2}$، فيبقى
$+\frac{2z}{z^2-4m^2}$ من المجموع الأول، بينما ينجو $n$ الفردي بالإشارة $-$:
\[
\frac{\pi}{\sin(\pi z)} = \frac1z +
\sum_{n\geq1}(-1)^n\,\frac{2z}{z^2 - n^2} .
\]
وفحص الإشارات: $\operatorname{Res}\bigl(\pi/\sin(\pi z),
n\bigr) = \pi/(\pi\cos\pi n) = (-1)^n$ (\cref{met:b3:residues:compute})، ويحمل $(-1)^n\frac{2z}{z^2 -
n^2} = \frac{(-1)^n}{z-n} + \frac{(-1)^n}{z+n}$ ذلك \omterm{def:b3:residues:singularities}{الباقي}
بالضبط عند $\pm n$.

\textbf{21.} من أجل $\abs z \leq r < 1$، بنشر كل حدّ كما في السؤال 19
($(-1)^n\frac{2z}{z^2-n^2} =
(-1)^{n-1}\frac{2z}{n^2}\sum_k\frac{z^{2k}}{n^{2k}}$) وبتطبيق فوبيني:
\[
\frac{\pi}{\sin(\pi z)} = \frac1z +
2\sum_{k\geq0}\eta(2k+2)\,z^{2k+1},
\qquad
\eta(s) = \sum_{n\geq1}\frac{(-1)^{n-1}}{n^s} .
\]
وأما جهة تايلور: فيعطي $\sin(\pi z) = \pi z(1 - \frac{(\pi z)^2}6 +
O(z^4))$ أن $\frac\pi{\sin\pi z} = \frac1z +
\frac{\pi^2}6z + O(z^3)$: أي $2\eta(2) = \frac{\pi^2}6$، ومنه
$\eta(2) = \frac{\pi^2}{12}$. وأما الاتساق: $\eta(2) =
\zeta(2) - 2\sum_n(2n)^{-2} = (1 - 2^{1-2})\zeta(2) =
\frac{\zeta(2)}2 = \frac{\pi^2}{12}$.

\textbf{22.} عند $z = \frac12$، يعطي السؤال 20
\[
\pi = 2 + \sum_{n\geq1}(-1)^n\frac{1}{\frac14 - n^2}
= 2 + 4\sum_{n\geq1}\frac{(-1)^{n-1}}{(2n-1)(2n+1)} .
\]
ومع $\frac1{(2n-1)(2n+1)} = \frac12\bigl(\frac1{2n-1} -
\frac1{2n+1}\bigr)$، وبتقارب المتسلسلتين المتناوبتين (بمحك لايبنتز)،
ينشطر المجموع إلى $\frac12\bigl[L -
(1 - L)\bigr] = L - \frac12$، حيث $L = 1 - \frac13 +
\frac15 - \cdots$ و $\frac13 - \frac15 + \frac17 - \cdots
= 1 - L$. ومنه $\pi = 2 + 4(L - \frac12) = 4L$:
\[
\frac\pi4 = 1 - \frac13 + \frac15 - \frac17 + \cdots
\]
والعبرة: أن كل نواة \omterm{def:b3:residues:singularities}{مرومورفية} بأقطاب على متتالية حسابية وببواقٍ
مقرَّرة. فظل التمام يضع \omterm{def:b3:residues:singularities}{الباقي} $1$ عند كل عدد صحيح ويجمع
$f(n)$؛ ومشتقته تربّع الأقطاب وتسعّر $\lambda(2)$؛ والظل يحرّك
الأقطاب إلى $\frac12 + \Z$ ويسعّر المقامات الفردية؛
و $\pi/\sin$ يبقي الأقطاب الصحيحة لكنه يناوب البواقي
$(-1)^n$، ومن هنا المتسلسلات المتناوبة. والنوى من الرتبة
الأولى كلها دوال فردية: فمزاوجة $n$ مع $-n$ تضاعف معاملات
القوى الزوجية وتفني الفردية، ومنه ينسكب $\zeta(2k)$ آليًّا
بينما لا يظهر $\zeta(3)$ أبدًا. فالعمى تماثلٌ، لا قصور تقنية.

\textbf{23.} مع $w = 2\iu\pi z$:
\[
\iu\pi z + \frac{2\iu\pi z}{\eu^{2\iu\pi z} - 1}
= \iu\pi z\,\frac{\eu^{2\iu\pi z} + 1}{\eu^{2\iu\pi z} - 1}
= \iu\pi z\,\frac{\eu^{\iu\pi z} +
\eu^{-\iu\pi z}}{\eu^{\iu\pi z} - \eu^{-\iu\pi z}}
= \pi z\,\frac{\cos\pi z}{\sin\pi z} = \pi z\cot(\pi z) .
\]
ومنه، باستعمال الدالة المولّدة عند $w = 2\iu\pi z$ (مع إلغاء
$B_1 = -\frac12$ للحدّ $\iu\pi z$، وانعدام $B$ الفردية بعدها):
\[
\pi z\cot(\pi z) = \sum_{k\geq0}\frac{B_{2k}}{(2k)!}
(2\iu\pi z)^{2k}
= 1 + \sum_{k\geq1}(-1)^k\frac{(2\pi)^{2k}B_{2k}}{(2k)!}
z^{2k} .
\]
ومن جهة أخرى يعطي نشر الكسور الجزئية (الجزء الرابع) أن $\pi z\cot(\pi z) = 1 - 2\sum_{k\geq1}\zeta(2k)z^{2k}$
(بنشر كل $\frac{2z^2}{z^2 - n^2} =
-2\sum_k\frac{z^{2k}}{n^{2k}}$ والجمع على $n$، إذ يبرّر التقارب الناظمي
التبديلَ من أجل $\abs z < 1$). وبمقارنة المعاملات: $-2\zeta(2k) =
(-1)^k\frac{(2\pi)^{2k}B_{2k}}{(2k)!}$، وهي
الصيغة المذكورة.

\textbf{24.} $k = 1$: $\frac{(2\pi)^2}{2\cdot2}\cdot\frac16
= \frac{\pi^2}6$. و $k = 2$: $-\frac{(2\pi)^4}{2\cdot24}
\cdot\bigl(-\frac1{30}\bigr) = \frac{16\pi^4}{48\cdot30} =
\frac{\pi^4}{90}$. و $k = 3$:
$\frac{(2\pi)^6}{2\cdot720}
\cdot\frac1{42} = \frac{64\pi^6}{1440\cdot42} =
\frac{\pi^6}{945}$.

\textbf{25.} نكتب $C(z) = \pi z\cot(\pi z) = 1 -
2\sum_{k\geq1}\zeta(2k)z^{2k}$ (السؤال 23). وبالاشتقاق المباشر
للمقدار $C = \pi z\cot(\pi z)$، باستعمال $(\cot u)'
= -1 - \cot^2u$:
\[
zC'(z) = \pi z\cot(\pi z) - \pi^2z^2\bigl(1 +
\cot^2(\pi z)\bigr) = C - \pi^2z^2 - C^2 .
\]
والآن ننشر الطرفين في قوى $z^2$. الطرف الأيسر: $\sum_k(-4k)\,\zeta(2k)\,z^{2k}$. والطرف
الأيمن: بتربيع المتسلسلة،
\[
C - C^2 = 2\sum_{k\geq1}\zeta(2k)z^{2k}
- 4\sum_{k\geq2}\Bigl(\sum_{j=1}^{k-1}\zeta(2j)
\zeta(2k-2j)\Bigr)z^{2k},
\]
والحدّ $-\pi^2z^2 = -6\zeta(2)z^2$ لا يعدّل سوى $k =
1$. وبمقارنة معاملات $z^{2k}$
من أجل $k \geq 2$:
\[
-4k\,\zeta(2k) = 2\,\zeta(2k) -
4\sum_{j=1}^{k-1}\zeta(2j)\,\zeta(2k-2j),
\]
أي $\bigl(k + \frac12\bigr)\zeta(2k) =
\sum_{j=1}^{k-1}\zeta(2j)\zeta(2k-2j)$. (وعند $k = 1$ تُقرأ المتطابقة $-4\zeta(2) = 2\zeta(2) - 6\zeta(2)$: أي فحص
اتساق، لا معلومة جديدة.) والتطبيقات: $k =
2$: $\frac52\zeta(4) = \zeta(2)^2 = \frac{\pi^4}{36}$، ومنه
$\zeta(4) = \frac{\pi^4}{90}$؛ و $k = 3$: $\frac72\zeta(6) =
2\zeta(2)\zeta(4) = \frac{\pi^6}{270}$، ومنه $\zeta(6) =
\frac{2}{7}\cdot\frac{\pi^6}{270} = \frac{\pi^6}{945}$. فبذرة متسامية واحدة
($\zeta(2) = \frac{\pi^2}6$)، وجبرٌ محض يولّد كل قيمة زيتا زوجية.
\end{solution}
