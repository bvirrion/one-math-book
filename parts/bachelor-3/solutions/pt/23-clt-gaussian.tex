\section*{Capítulo \ref{ch:b3:clt} --- Funções características e o teorema central do limite}

\begin{solution}{exo:b3:clt:1}
Uniforme ligado $\intcc{-1}1$: $\varphi(\xi) =
\frac12\int_{-1}^1\eu^{\iu\xi x}\dd x =
\frac{\sin\xi}{\xi}$ (igual a $1$ no $\xi = 0$).
Exponencial $\mathcal E(\lambda)$: $\varphi(\xi) =
\lambda\int_0^\infty\eu^{(\iu\xi - \lambda)x}\dd x =
\frac{\lambda}{\lambda - \iu\xi}$ (a antiderivada
desaparece em $+\infty $ desde $\operatorname{Re}(\iu\xi -
\lambda) < 0$). Poisson $\mathcal P(\lambda)$: pelo
teorema de transferência para \omterm{def:b3:probability:space}{leis} discretas;
\[
\varphi(\xi) = \sum_{k\geq0}\eu^{\iu\xi
k}\,\eu^{-\lambda}\frac{\lambda^k}{k!}
= \eu^{-\lambda}\exp\bigl(\lambda\eu^{\iu\xi}\bigr)
= \exp\bigl(\lambda(\eu^{\iu\xi} - 1)\bigr).
\]
Binômio $\mathcal B(n, p)$: uma soma de $ n $ \omterm{def:b3:probability:independence}{independente}
Variáveis de Bernoulli, cada uma com cf $1 - p + p\eu^{\iu\xi}$,
então $\varphi(\xi) = \bigl(1 - p + p\eu^{\iu\xi}\bigr)^n $
(\cref{prop:b3:clt:cfbasics}(b)). Aditividade de Poisson: se $ X
\sim \mathcal P(\lambda)$, $ Y \sim \mathcal P(\mu)$ são
\omterm{def:b3:probability:independence}{independente},
\[
\varphi_{X+Y}(\xi) = \eu^{\lambda(\eu^{\iu\xi}-1)}
\eu^{\mu(\eu^{\iu\xi}-1)} =
\eu^{(\lambda+\mu)(\eu^{\iu\xi}-1)},
\]
o cf de $\mathcal P(\lambda + \mu)$; injetividade
(\cref{thm:b3:clt:injectivity}) identifica a \omterm{def:b3:probability:space}{lei}.
\end{solution}

\begin{solution}{exo:b3:clt:2}
(um) $\overline{\varphi_X(\xi)} = \E\eu^{-\iu\xi X} =
\varphi_{-X}(\xi)$. Então $\varphi_X $ é real se e somente se
$\varphi_X = \varphi_{-X}$, se e somente se (injetividade,
\cref{thm:b3:clt:injectivity}) $ X $ e $-X $ tem o mesmo
\omterm{def:b3:probability:space}{lei}.
(b) Escreva $\varphi_X(\xi_0) = \eu^{\iu\theta}$. Então
\[
\E\bigl[1 - \cos(\xi_0X - \theta)\bigr]
= 1 - \operatorname{Re}\bigl(\eu^{-\iu\theta}
\varphi_X(\xi_0)\bigr) = 1 - 1 = 0 .
\]
O integrando é não negativo, então $\cos(\xi_0X - \theta) = 1$
quase certamente (uma variável não negativa com zero \omterm{def:b3:probability:space}{esperança}
desaparece como), ou seja,\ $\xi_0X - \theta \in 2\pi\Z $ como.: $ X $
toma seus valores na progressão aritmética
$\frac{\theta}{\xi_0} + \frac{2\pi}{\xi_0}\Z $ quase certamente.
Se $ X $ tem um \omterm{ex:b3:lebesgue:gamma}{densidade}, este conjunto contável é nulo de Lebesgue,
então carrega probabilidade $0$ --- contradição; portanto
$\abs{\varphi_X(\xi)} < 1$ para cada $\xi \neq 0$.
\end{solution}

\begin{solution}{exo:b3:clt:3}
Por independência e \cref{prop:b3:clt:cfbasics}:
\[
\varphi_{X+Y}(\xi) = \eu^{\iu m_1\xi - \sigma_1^2\xi^2/2}\,
\eu^{\iu m_2\xi - \sigma_2^2\xi^2/2}
= \eu^{\iu(m_1+m_2)\xi - (\sigma_1^2+\sigma_2^2)\xi^2/2},
\]
o cf de $\mathcal N(m_1 + m_2, \sigma_1^2 + \sigma_2^2)$;
conclui a injetividade. A estabilidade sob aplicações afins é a
regra afim ($ aX + b \sim \mathcal N(am_1 + b,
a^2\sigma_1^2)$, permitindo o caso degenerado $ a = 0$) e
estabilidade sob \omterm{def:b3:probability:independence}{independente} somas segue por indução no
cálculo acima. Para \emph{dependente} \omterm{def:b3:clt:gaussianvector}{Gaussianos} a soma
não precisa ser \omterm{def:b3:clt:gaussianvector}{Gaussiano}: em \cref{exo:b3:clt:9}, $ X $ e $ Y =
\varepsilon X $ cada um é padrão \omterm{def:b3:clt:gaussianvector}{Gaussiano} mas $ X + Y $
desaparece com probabilidade $\frac12$ sem ser assim\
zero, então não é \omterm{def:b3:clt:gaussianvector}{Gaussiano}.
\end{solution}

\begin{solution}{exo:b3:clt:4}
(um) \emph{Implicação direta.} Deixar $ t $ seja um continuidade ponto
de $ F_X $ e $\delta > 0$. Pegue o \omterm{def:b3:topology:continuity}{contínuo} rampas
$ f^-$ ($= 1$ sobre $\intoc{-\infty}{t-\delta}$, $0$ de $ t $
ligado, afim entre) e $ f^+$ ($= 1$ sobre $\intoc{-\infty}t $,
$0$ de $ t + \delta $ ligado, afim entre); então $ f^- \leq
\mathbf 1_{\intoc{-\infty}t} \leq f^+$, então
\[
\E f^-(X_n) \leq F_{X_n}(t) \leq \E f^+(X_n),
\]
e os termos externos convergem para $\E f^\pm(X)$, eles mesmos
espremido entre $ F_X(t - \delta)$ e $ F_X(t + \delta)$.
Deixando $ n \to \infty $ então $\delta \to 0$ e usando
continuidade de $ F_X $ no $ t $: $ F_{X_n}(t) \to F_X(t)$.

\emph{Conversar.} Deixar $ f $ ser limitado \omterm{def:b3:topology:continuity}{contínuo}, $ M =
\sup\abs f $, $\varepsilon > 0$. O continuidade pontos de
$ F_X $ são \omterm{def:b3:topology:interior}{densos} ($ F_X $ tem no \omterm{def:b3:rings:primemaximal}{máximo} muitos saltos contáveis), então
escolher continuidade pontos $ a < b $ com $ F_X(a) < \varepsilon $
e $1 - F_X(b) < \varepsilon $. No compactar $\intcc ab $
a função $ f $ é uniformemente \omterm{def:b3:topology:continuity}{contínuo}: escolher continuidade
pontos $ a = t_0 < t_1 < \dots < t_m = b $ de $ F_X $ com
oscilação de $ f $ no \omterm{def:b3:rings:primemaximal}{máximo} $\varepsilon $ em cada
$\intoc{t_{j-1}}{t_j}$ e definir $ g = \sum_j
f(t_j)\,\mathbf 1_{\intoc{t_{j-1}}{t_j}}$. Então $\abs{f - g}
\leq \varepsilon $ sobre $\intoc ab $, $\abs g \leq M $, e para
$ T = X_n $ ou $ X $:
\[
\bigl|\E f(T) - \E g(T)\bigr| \leq \varepsilon +
2M\bigl(F_T(a) + 1 - F_T(b)\bigr).
\]
Além disso $\E g(X_n) = \sum_j f(t_j)\bigl(F_{X_n}(t_j) -
F_{X_n}(t_{j-1})\bigr) \to \E g(X)$ (uma soma finita de
termos convergentes, todos $ t_j $ ser continuidade pontos) e
$ F_{X_n}(a) \to F_X(a) < \varepsilon $, $1 - F_{X_n}(b) \to 1
- F_X(b) < \varepsilon $. Montagem:
$\limsup_n\abs{\E f(X_n) - \E f(X)} \leq 2\varepsilon +
8M\varepsilon $; deixar $\varepsilon \to 0$.

(b) A função de distribuição da constante $ c $ é
$\mathbf 1_{\intco c\infty}$, \omterm{def:b3:topology:continuity}{contínuo} exceto em $ c $. Para
$\varepsilon > 0$, os pontos $ c - \varepsilon $ e $ c +
\frac\varepsilon2$ são continuidade pontos, então
\[
\P(\abs{X_n - c} > \varepsilon) \leq F_{X_n}(c -
\varepsilon) + 1 - F_{X_n}\Bigl(c + \frac\varepsilon2\Bigr)
\longrightarrow 0 + 1 - 1 = 0 .
\]
\end{solution}

\begin{solution}{exo:b3:clt:5}
Deixar $ z_n = p_n(\eu^{\iu\xi} - 1)$, então $\varphi_{X_n}(\xi) =
(1 + z_n)^n $ (\cref{exo:b3:clt:1}) e $\abs{z_n} \leq 2p_n
\to 0$ (observação $ p_n = \frac{np_n}n \to 0$). Ambos $1 + z_n $ e
$\eu^{z_n}$ tem módulo no \omterm{def:b3:rings:primemaximal}{máximo} $1$: $\abs{1 + z_n} =
\abs{(1 - p_n) + p_n\eu^{\iu\xi}} \leq 1$ pelo triângulo
desigualdade, e $\abs{\eu^{z_n}} = \eu^{p_n(\cos\xi - 1)}
\leq 1$. A desigualdade telescópica $\abs{a^n - b^n} \leq
n\abs{a - b}$ (prova de \cref{thm:b3:clt:clt}) e o poder
encadernado em série $\abs{\eu^z - 1 - z} \leq
\abs z^2\eu^{\abs z}$ dar
\[
\bigl|(1 + z_n)^n - \eu^{nz_n}\bigr| \leq n\bigl|1 + z_n -
\eu^{z_n}\bigr| \leq n\,\abs{z_n}^2\,\eu^{\abs{z_n}} \leq
4\eu^2\,np_n^2 = 4\eu^2\,(np_n)\,p_n \longrightarrow 0 .
\]
Desde $ nz_n = np_n(\eu^{\iu\xi} - 1) \to
\lambda(\eu^{\iu\xi} - 1)$, concluímos $\varphi_{X_n}(\xi)
\to \exp\bigl(\lambda(\eu^{\iu\xi} - 1)\bigr)$ para cada
$\xi $: o cf de $\mathcal P(\lambda)$ e euévy
(\cref{thm:b3:clt:levy}) dá $ X_n \Rightarrow \mathcal
P(\lambda)$. Numericamente: $\P\bigl(\mathcal B(100, 0.02) =
0\bigr) = 0.98^{100} = \eu^{100\ln 0.98} \approx
\eu^{-2.020} \approx 0.1326$, enquanto $\P\bigl(\mathcal P(2) =
0\bigr) = \eu^{-2} \approx 0.1353$: dois por cento de diferença
já neste grosseiro $ n $.
\end{solution}

\begin{solution}{exo:b3:clt:6}
(a) Um lançamento tem média $\frac72$ e variação
$\frac{35}{12}$, então $ S $ tem significado $3500$, variação
$\frac{35000}{12} \approx 2916.7$ e desvio padrão
$\approx 54.0$. Pela CLT,
\[
\P(S > 3600) = \P\Bigl(\frac{S - 3500}{54.0} > 1.85\Bigr)
\approx 1 - \Phi(1.85) \approx 0.032 :
\]
sobre um $3\%$ chance.
(b) $ S \sim \mathcal B(100, \frac12)$: média $50$, padrão
desvio $5$. Com o continuidade correção,
\[
\P(45 \leq S \leq 55) \approx
\Phi\Bigl(\frac{55.5 - 50}{5}\Bigr) -
\Phi\Bigl(\frac{44.5 - 50}{5}\Bigr) = 2\Phi(1.1) - 1
\approx 0.729,
\]
contra o valor \omterm{def:b3:forms:closedexact}{exato} $0.7287$; sem a correção,
$2\Phi(1) - 1 \approx 0.683$, perdendo quase cinco pontos. O
a correção é importante porque $ S $ é uma variável de rede: o
átomo $\P(S = k)$ é bem aproximado pelo \omterm{def:b3:clt:gaussianvector}{Gaussiano} massa
de $\intcc{k - \frac12}{k + \frac12}$ e recortando o
intervalo nos inteiros $45$ e $55$ descarta meio átomo
em cada extremidade.
\end{solution}

\begin{solution}{exo:b3:clt:7}
As variáveis $ g(U_k)$ são i.i.d.\ (\omterm{def:b3:lebesgue:measurable}{mensurável} imagens de
i.i.d.\ variáveis), quadrado \omterm{def:b3:lebesgue:l1}{integrável}, com média $ I $
(teorema de transferência, \cref{exo:b3:product:9}) e variância
$\sigma^2$. Se $\sigma > 0$, \cref{thm:b3:clt:clt} aplicado
para eles é exatamente a convergência declarada
\[
\sqrt n\,\Bigl(\frac1n\sum_{k\leq n}g(U_k) - I\Bigr) =
\frac{\sum_{k\leq n}\bigl(g(U_k) - I\bigr)}{\sqrt n}
\Longrightarrow \mathcal N(0, \sigma^2)
\]
(se $\sigma = 0$, $ g $ é como constante e o lado esquerdo
lado desaparece de forma idêntica). Por isso
$\P\bigl(\abs{\frac1n\sum g(U_k) - I} \leq
1.96\,\sigma/\sqrt n\bigr) \to 0.95$: a barra de erro $\pm
1.96\,\sigma/\sqrt n $ vê a dimensão $ d $ somente através
a constante $\sigma $, nunca através da taxa em $ n $. O
regra do ponto médio com $ n $ nós em dimensão $ d $ tem malha
$ n^{-1/d}$ e erro de pedido $ n^{-2/d}$ para $\mathcal C^2$
integrandos. Monte Carlode $ n^{-1/2}$ decai mais rápido do que
$ n^{-2/d}$ exatamente quando $\frac12 > \frac2d $, ou seja\ $ d > 4$:
da dimensão $5$ ligado, a amostragem aleatória bate assintoticamente
a grade --- a maldição da dimensionalidade poupa
métodos probabilísticos, razão pela qual \omterm{ex:b3:probability:sllnapps}{Monte Carlo} regras
integração de alta dimensão.
\end{solution}

\begin{solution}{exo:b3:clt:8}
\emph{Soma.} Para fixo $\xi $:
\[
\bigl|\E\eu^{\iu\xi(X_n+Y_n)} -
\eu^{\iu\xi c}\,\E\eu^{\iu\xi X_n}\bigr|
= \bigl|\E\bigl[\eu^{\iu\xi X_n}\bigl(\eu^{\iu\xi Y_n} -
\eu^{\iu\xi c}\bigr)\bigr]\bigr|
\leq \E\bigl|\eu^{\iu\xi(Y_n - c)} - 1\bigr| .
\]
Dividir no evento $\{\abs{Y_n - c} \leq \delta\}$: lá,
$\abs{\eu^{\iu\xi(Y_n-c)} - 1} \leq \abs\xi\,\delta $ (o
o acorde é mais curto que o arco); o complemento contribui
no \omterm{def:b3:rings:primemaximal}{máximo} $2\,\P(\abs{Y_n - c} > \delta) \to 0$. Daí o
$\limsup $ é $\leq \abs\xi\,\delta $ para cada $\delta > 0$:
a diferença tende a $0$. Desde $\E\eu^{\iu\xi X_n} \to
\varphi_X(\xi)$, obtemos $\varphi_{X_n+Y_n}(\xi) \to
\eu^{\iu\xi c}\varphi_X(\xi) = \varphi_{X+c}(\xi)$ e
euévy (\cref{thm:b3:clt:levy}) rendimentos $ X_n + Y_n
\Rightarrow X + c $.

\emph{Produto.} Primeiro, $ cX_n \Rightarrow cX $:
$\varphi_{cX_n}(\xi) = \varphi_{X_n}(c\xi) \to
\varphi_X(c\xi) = \varphi_{cX}(\xi)$. Próximo, $(Y_n - c)X_n
\to 0$ em probabilidade: as \omterm{def:b3:probability:space}{leis} do $ X_n $ são apertados (seu
cfs convergem para um cf; veja a etapa de aperto de
\cref{thm:b3:clt:levy}), assim dado $\varepsilon > 0$ escolha $ M $
com $\P(\abs{X_n} > M) \leq \varepsilon $ para todos $ n $; então
\[
\P\bigl(\abs{(Y_n - c)X_n} > \varepsilon\bigr) \leq
\P(\abs{X_n} > M) + \P\Bigl(\abs{Y_n - c} >
\frac{\varepsilon}{M}\Bigr) \leq \varepsilon + o(1) .
\]
Escrita $ Y_nX_n = cX_n + (Y_n - c)X_n $ e aplicando a soma
parte (cuja prova só foi usada $ Y_n' := (Y_n - c)X_n \to 0$ em
probabilidade, com constante $0$): $ Y_nX_n \Rightarrow cX $.

\emph{Aplicativo.} Pela \omterm{def:b3:probability:space}{lei} forte dos grandes números
(\cref{thm:b3:probability:slln}), $\hat p_n \to p $ como, então
por continuidade $\hat\sigma_n = \sqrt{\hat p_n(1 - \hat p_n)}
\to \sigma = \sqrt{p(1 - p)} > 0$ como, portanto
$\frac{\sigma}{\hat\sigma_n} \to 1$ em probabilidade. Slutsky
atualizações de regras de produto $\frac{S_n - np}{\sigma\sqrt n}
\Rightarrow \mathcal N(0,1)$ para $\frac{S_n -
np}{\hat\sigma_n\sqrt n} = \frac{\sigma}{\hat\sigma_n}\cdot
\frac{S_n - np}{\sigma\sqrt n} \Rightarrow \mathcal N(0,1)$:
o \emph{utilizável} intervalo de confiança $\hat p_n \pm
1.96\,\hat\sigma_n/\sqrt n $, construído apenas a partir dos dados,
mantém sua assintótica $95\%$ nível.
\end{solution}

\begin{solution}{exo:b3:clt:9}
(a) Dividindo a \omterm{def:b3:probability:space}{esperança} sobre os dois valores de
$\varepsilon $ (independência): para Borel $ B $, $\P(Y \in B) =
\frac12\P(X \in B) + \frac12\P(-X \in B) = \P(X \in B)$,
desde $-X \sim X $ ($\mathcal N(0,1)$ é simétrico): $ Y \sim
\mathcal N(0,1)$. E $\operatorname{Cov}(X, Y) =
\E[\varepsilon X^2] = \E[\varepsilon]\,\E[X^2] = 0 \cdot 1 =
0$.
(b) $\abs Y = \abs X $, então $\P(\abs X \leq 1,\ \abs Y \geq 2)
= 0$ enquanto $\P(\abs X \leq 1)\,\P(\abs Y \geq 2) > 0$: não
\omterm{def:b3:probability:independence}{independente}. Se $(X, Y)$ eram um Vetor gaussiano, $ X + Y =
(1 + \varepsilon)X $ seria um verdadeiro \omterm{def:b3:clt:gaussianvector}{Gaussiano} variável
(\cref{def:b3:clt:gaussianvector} com $ t = (1,1)$); mas
$\P(X + Y = 0) = \P(\varepsilon = -1) = \frac12$, enquanto um
\omterm{def:b3:clt:gaussianvector}{Gaussiano} variável tem um átomo apenas se for uma constante
--- e $ X + Y $ é igual $2X \neq 0$ como em
$\{\varepsilon = 1\}$. Contradição: $(X, Y)$ não é
\omterm{def:b3:clt:gaussianvector}{Gaussiano}.
(c) Cada marginal é \omterm{def:b3:clt:gaussianvector}{Gaussiano} e a covariância desaparece,
ainda independência falha --- porque o \emph{par} não é
conjuntamente \omterm{def:b3:clt:gaussianvector}{Gaussiano}. \cref{thm:b3:clt:gaussianvector}(2) não pode
ser enfraquecido para ``\omterm{def:b3:clt:gaussianvector}{Gaussiano} marginais''.
\end{solution}

\begin{solution}{exo:b3:clt:10}
(um) \cref{exo:b3:fouriertransform:1} calcula
$\widehat{\eu^{-\abs\cdot}}(\xi) = \frac{2}{1 + \xi^2}$;
ambos os lados sendo \omterm{def:b3:lebesgue:l1}{integrável}, Inversão de Fourier
(\cref{thm:b3:fouriertransform:inversion}) transforma isso
ao redor:
\[
\int_\R\eu^{\iu\xi x}\,\frac{\dd x}{\pi(1 + x^2)} =
\eu^{-\abs\xi},
\]
que é exatamente $\varphi_X(\xi)$ para uma variável Cauchy $ X $.
(b) Por independência, $\varphi_{S_n}(\xi) =
\bigl(\eu^{-\abs\xi}\bigr)^n = \eu^{-n\abs\xi}$, então
$\varphi_{S_n/n}(\xi) = \varphi_{S_n}(\xi/n) =
\eu^{-\abs\xi}$: a média empírica $\frac{S_n}n $ é de novo
Cauchy padrão para cada $ n $ (injetividade). A média
nunca se concentra: suas flutuações no tempo $10^6$ são
aqueles de uma única observação.
(c) $\E\abs{X_1} = \frac2\pi\int_0^\infty\frac{x\,\dd x}{1 +
x^2} = +\infty $: a \omterm{def:b3:probability:space}{lei} Cauchy não é \omterm{def:b3:lebesgue:l1}{integrável}, então o
\omterm{def:b3:probability:space}{lei} forte dos grandes números
(\cref{thm:b3:probability:slln}) não se aplica, e a CLT
(que precisa de uma variação finita) ainda menos. Aqui está o seu
as conclusões falham genuinamente, e não apenas as suas provas.
Verificação de consistência: $\varphi(\xi) = \eu^{-\abs\xi}$ não é
diferenciável em $0$, como \cref{prop:b3:clt:cfbasics}(c)
ler contrapositivamente prevê para um não \omterm{def:b3:lebesgue:l1}{integrável}
variável.
\end{solution}

\begin{solution}{exo:b3:clt:11}
(um) $\varphi_{aX_1 + bX_2}(\xi) =
\eu^{-a\abs\xi}\eu^{-b\abs\xi} = \eu^{-(a+b)\abs\xi} =
\varphi_{(a+b)X_1}(\xi)$ (independência e
\cref{exo:b3:clt:10}); a injetividade identifica as \omterm{def:b3:probability:space}{leis}.

(b) $\varphi_{aX_1+bX_2}(\xi) = \eu^{-a^2\xi^2/2}
\eu^{-b^2\xi^2/2} = \eu^{-(a^2+b^2)\xi^2/2}$: a \omterm{def:b3:probability:space}{lei} de
$\sqrt{a^2+b^2}\,X_1$.

(c) Se $\varphi_X(\xi) = \eu^{-c\abs\xi^\alpha}$, então
$ S_n = X_1 + \dots + X_n $ tem $\varphi_{S_n} =
\eu^{-cn\abs\xi^\alpha}$, e $ S_n/n^{1/\alpha}$ tem
$\varphi(\xi) = \eu^{-c\abs\xi^\alpha}$ novamente: \omterm{def:b3:forms:closedexact}{exato}
auto-reprodução sob o $ n^{1/\alpha}$ dimensionamento ---
$\sqrt n $ para o \omterm{def:b3:clt:gaussianvector}{Gaussiano} ($\alpha = 2$), $ n $ em si para
Cauchy ($\alpha = 1$, \cref{exo:b3:clt:10}(b)). Uma soma de
variáveis i.i.d.\ só podem convergir (após afins
normalização) para uma \omterm{def:b3:probability:space}{lei} que seja \omterm{def:b3:ode:stability}{estável} sob tal
convoluções; a CLT diz que a variância finita força o
\omterm{def:b3:clt:gaussianvector}{Gaussiano} bacia, e a bacia de Cauchy é reservada para \omterm{def:b3:probability:space}{leis}
com caudas tão pesadas que $\E X^2 = \infty $ e até mesmo $\E\abs
X = \infty $ --- por exemplo\ somas das próprias variáveis de Cauchy.
A universalidade possui diversas ilhas, indexadas pela cauda
expoente $\alpha \in \intoc02$.
\end{solution}

\begin{solution}{exo:b3:clt:12}
(a) Os indicadores $\mathbf 1_{X_k \leq t}$ são i.i.d.\
Bernoulli do parâmetro $ p = F(t)$: a soma deles $ nF_n(t)$ é
binômio $\mathcal B(n, p)$; a \omterm{def:b3:probability:space}{lei} forte dá $ F_n(t)
\to p $ a.s., e a CLT (\cref{thm:b3:clt:clt}) aplicado a
os mesmos indicadores (variância $ p(1-p)$) fornece o declarado
\omterm{def:b3:clt:gaussianvector}{Gaussiano} limite.

(b) $ p(1 - p)$ é \omterm{def:b3:rings:primemaximal}{máximo} em $ p = \frac12$, ou seja, onde
$ F(t) = \frac12$: no \emph{mediana}. Estimando cauda
probabilidades é assintoticamente fácil (variância $\to 0$ como
$ p \to 0, 1$); a região mediana carrega o maior
ruído estatístico --- a curva empírica oscila mais em
seu meio.

(c) Para \omterm{def:b3:topology:continuity}{contínuo} $ F $, as variáveis $ U_k = F(X_k)$ são
i.i.d.\ uniforme em $\intoo01$ (\cref{exo:b3:probability:1}),
e monotonicidade de $ F $ dá, escrevendo $ G_n $ para o
função de distribuição empírica do $ U_k $:
\[
\sup_{t\in\R}\,\abs{F_n(t) - F(t)}
= \sup_{u \in \operatorname{im}F}\,\abs{G_n(u) - u}
= \sup_{u\in\intcc01}\abs{G_n(u) - u} :
\]
a primeira igualdade porque $\{X_k \leq t\} = \{U_k \leq
F(t)\}$ até eventos nulos (monotonicidade; desigualdade estrita
pode falhar apenas nas partes planas do $ F $, onde ambos os lados
permanecem inalterados), e o segundo porque um \omterm{def:b3:topology:continuity}{contínuo} $ F $,
fugindo de $0$ a $1$, atinge todos os valores de
$\intoo01$ (teorema do valor intermediário) e os pontos finais
não adicione nada ($ G_n(0) - 0 = 0$ e $ G_n(1) - 1 = 0$). O
lado direito envolve apenas uniformes: uma \omterm{def:b3:probability:space}{lei}
para todos $ F $ --- então uma única tabela de valores críticos (que
da distribuição de Kolmogorov) testes \emph{qualquer} \omterm{def:b3:topology:continuity}{contínuo}
modelo em relação aos dados.
\end{solution}

\begin{solution}{pb:b3:clt:1}
\textbf{1.} A família $(X_1, \dots, X_n, N_1, \dots, N_n)$
é \omterm{def:b3:probability:independence}{independente}: os dois blocos são \omterm{def:b3:probability:independence}{independente} de cada
outro por construção e cada bloco é i.i.d. $ W_i $ é um
função \omterm{def:b3:lebesgue:measurable}{mensurável} das variáveis $(X_j)_{j<i}$ e
$(N_j)_{j>i}$ apenas, todos distintos de $ X_i $ e $ N_i $: por
o princípio da coalizão
(\cref{thm:b3:probability:independence}), $ W_i $ é
\omterm{def:b3:probability:independence}{independente} do par $(X_i, N_i)$. As decomposições
$ H_i = W_i + \frac{X_i}{\sqrt n}$ e $ H_{i-1} = W_i +
\frac{N_i}{\sqrt n}$ são imediatos das definições:
passando de $ H_i $ a $ H_{i-1}$ troca o pedido único
$ X_i $ para $ N_i $.

\textbf{2.} Taylor-Lagrange sob encomenda $3$: há $ c $
entre $ w $ e $ w + h $ com $ f(w + h) = f(w) + f'(w)h +
\frac12f''(w)h^2 + \frac16f'''(c)h^3$, e $\abs{f'''(c)}
\leq M_3$ dá o limite.

\textbf{3.} Subtraindo as duas expansões no comum
ponto \omterm{def:b3:topology:basis}{base} $ w = W_i $:
\[
f(H_i) - f(H_{i-1}) = f'(W_i)\,\frac{X_i - N_i}{\sqrt n} +
\frac{f''(W_i)}{2}\,\frac{X_i^2 - N_i^2}{n} + R_i,
\qquad
\abs{R_i} \leq \frac{M_3}{6}\cdot
\frac{\abs{X_i}^3 + \abs{N_i}^3}{n^{3/2}} .
\]
Pegar expectativas. Pela pergunta 1, $ f'(W_i)$ e $ f''(W_i)$
são \omterm{def:b3:probability:independence}{independente} de $(X_i, N_i)$, então o misto expectativas
fator:
\begin{align*}
\E\Bigl[f'(W_i)\,\frac{X_i - N_i}{\sqrt n}\Bigr] &=
\E\bigl[f'(W_i)\bigr]\,\frac{\E X_i - \E N_i}{\sqrt n} = 0,
\\
\E\Bigl[f''(W_i)\,\frac{X_i^2 - N_i^2}{n}\Bigr] &=
\E\bigl[f''(W_i)\bigr]\,\frac{1 - 1}{n} = 0 :
\end{align*}
os dois primeiros momentos $ X_i $ e $ N_i $ \emph{corresponder}e
apenas o restante sobrevive:
\[
\bigl|\E f(H_i) - \E f(H_{i-1})\bigr| \leq \E\abs{R_i} \leq
\frac{M_3}{6}\cdot\frac{\beta + \gamma}{n^{3/2}} .
\]
O \omterm{def:b3:clt:gaussianvector}{Gaussiano} terceiro momento: $\gamma = \E\abs{N_1}^3 =
2\int_0^\infty x^3\,\frac{\eu^{-x^2/2}}{\sqrt{2\pi}}\,\dd x
= \frac{2}{\sqrt{2\pi}}\int_0^\infty 2u\,\eu^{-u}\dd u =
\frac{4}{\sqrt{2\pi}} = \frac{2\sqrt2}{\sqrt\pi}$
(substituição $ u = x^2/2$, então $\Gamma(2) = 1$).

\textbf{4.} Telescópico $\E f(T_n) - \E f(G_n) =
\sum_{i=1}^n\bigl(\E f(H_i) - \E f(H_{i-1})\bigr)$ e
aplicando a questão 3 a cada um dos $ n $ termos:
\[
\bigl|\E f(T_n) - \E f(G_n)\bigr| \leq
n \cdot \frac{M_3(\beta + \gamma)}{6\,n^{3/2}} =
\frac{M_3\,(\beta + \gamma)}{6\,\sqrt n} .
\]

\textbf{5.} $ G_n $ é $\mathcal N(0,1)$ \emph{exatamente} para
cada $ n $ (uma soma normalizada de \omterm{def:b3:probability:independence}{independente} padrão
\omterm{def:b3:clt:gaussianvector}{Gaussianos}, \cref{exo:b3:clt:3}), então $\E f(G_n) = \E f(N)$
e a pergunta 4 é lida $\abs{\E f(T_n) - \E f(N)} \leq
\frac{M_3(\beta+\gamma)}{6\sqrt n} \to 0$ para $ f \in
\mathcal C^3_b $. \emph{Atualizar.} Deixar $ f $ ser limitado
\omterm{def:b3:topology:continuity}{contínuo}, $ M = \sup\abs f $, $\varepsilon > 0$. Escolher $ A
\geq 1$ com $\frac1{A^2} \leq \varepsilon $: Chebyshev com
$\V(T_n) = 1$ dá $\P(\abs{T_n} > A) \leq \varepsilon $
para todos $ n $, e da mesma forma $\P(\abs N > A) \leq \varepsilon $.
Deixar $\chi $ ser $\mathcal C^\infty $ com
$\mathbf 1_{\intcc{-A}A} \leq \chi \leq
\mathbf 1_{\intcc{-A-1}{A+1}}$ (um planalto suave, construído por
apaziguador $\mathbf 1_{\intcc{-A-\frac12}{A+\frac12}}$,
\cref{thm:b3:lp:regularization}); $ g = f\chi $ é \omterm{def:b3:topology:continuity}{contínuo}
com compactar suporte, portanto uniformemente \omterm{def:b3:topology:continuity}{contínuo}, então é
apaziguamento $ g_\eta = g * \rho_\eta $ é $\mathcal
C^\infty $ com derivadas limitadas de todas as ordens e
$\norm{g - g_\eta}_\infty \leq \varepsilon $ para $\eta $
pequeno o suficiente. Para $ T = T_n $ ou $ N $, desde $ f = g $ sobre
$\intcc{-A}A $ e $\abs{f - g} \leq 2M $ em todos os lugares:
\[
\bigl|\E f(T) - \E g_\eta(T)\bigr| \leq
\E\abs{(f - g)(T)} + \norm{g - g_\eta}_\infty
\leq 2M\,\P(\abs T > A) + \varepsilon \leq
(2M + 1)\,\varepsilon .
\]
Combinando com $\E g_\eta(T_n) \to \E g_\eta(N)$ (pergunta
4 se aplica: $ g_\eta \in \mathcal C^3_b $):
\[
\limsup_n\;\bigl|\E f(T_n) - \E f(N)\bigr| \leq
2(2M + 1)\,\varepsilon ,
\]
e $\varepsilon $ foi arbitrário: $\E f(T_n) \to \E f(N)$ para
cada limite \omterm{def:b3:topology:continuity}{contínuo} $ f $, ou seja\ $ T_n \Rightarrow
\mathcal N(0,1)$.

\textbf{6.} Distribuição idêntica inserida apenas através de um
frase: ``$ X_i $ e $ N_i $ tem os mesmos dois primeiros
momentos''. Então deixe $ X_1, \dots, X_n $ ser \omterm{def:b3:probability:independence}{independente},
centrado, com variações $\sigma_i^2$ e terço finito
momentos, $ s_n^2 = \sum_i\sigma_i^2 > 0$, e pegue $ N_i \sim
\mathcal N(0, \sigma_i^2)$ \omterm{def:b3:probability:independence}{independente} de tudo.
Defina os híbridos com normalização $ s_n $: $ H_i =
\frac1{s_n}(\sum_{j\leq i}X_j + \sum_{j>i}N_j)$. No
$ i $-ésima troca, $\E X_i = \E N_i = 0$ e $\E X_i^2 = \E N_i^2
= \sigma_i^2$ mate novamente o $ f'$ e $ f''$ termos, e o
resto dá (usando $\E\abs{N_i}^3 = \sigma_i^3\gamma $ por
escala):
\[
\bigl|\E f(H_i) - \E f(H_{i-1})\bigr| \leq
\frac{M_3}{6\,s_n^3}\bigl(\E\abs{X_i}^3 +
\sigma_i^3\gamma\bigr) .
\]
Telescópico:
\[
\Bigl|\E f\Bigl(\frac{X_1 + \dots + X_n}{s_n}\Bigr) -
\E f(N)\Bigr| \leq \frac{M_3}{6\,s_n^3}\sum_{i=1}^n
\Bigl(\E\abs{X_i}^3 + \V(X_i)^{3/2}\,\gamma\Bigr) .
\]
Desde $\sigma_i^3 = (\E X_i^2)^{3/2} \leq \E\abs{X_i}^3$
(a desigualdade média de potência, ou seja, \ Jensen para $ t \mapsto
t^{3/2}$ aplicado a $ X_i^2$), o lado direito é no \omterm{def:b3:rings:primemaximal}{máximo}
$\frac{M_3(1 + \gamma)}{6}\cdot
\frac{\sum_i\E\abs{X_i}^3}{s_n^3}$: sob \emph{Liapunov
condição} $\frac1{s_n^3}\sum_i\E\abs{X_i}^3 \to 0$, o
somas normalizadas convergem na \omterm{def:b3:probability:space}{lei} para $\mathcal N(0,1)$ ---
a CLT sem distribuição idêntica.

\textbf{7.} Deixar $\rho \in \mathcal C^\infty_c(\intoo01)$
com $\int\rho = 1$ e definir $\psi(x) = \int_x^1\rho(s)\dd
s $: $\psi $ é $\mathcal C^\infty $, não crescente, $\psi = 1$
sobre $\R_-$, $\psi = 0$ sobre $\intco1\infty $; deixar $ K =
\norm{\psi'''}_\infty $. Para $ t \in \R $ e $\delta > 0$
definir $\psi_\delta(x) = \psi\bigl(\frac{x - t}\delta\bigr)$
e $\tilde\psi_\delta(x) = \psi\bigl(\frac{x - t}\delta +
1\bigr)$: estes são $\mathcal C^3_b $ com terceira derivada
limitado por $ K/\delta^3$, e
\[
\mathbf 1_{\intoc{-\infty}{t-\delta}} \leq
\tilde\psi_\delta \leq \mathbf 1_{\intoc{-\infty}t} \leq
\psi_\delta \leq \mathbf 1_{\intoc{-\infty}{t+\delta}} .
\]
Limite superior: pela questão 4 aplicada a $\psi_\delta $ (com
$ M_3 = K/\delta^3$),
\[
\P(T_n \leq t) \leq \E\psi_\delta(T_n) \leq
\E\psi_\delta(N) + \frac{K(\beta +
\gamma)}{6\,\delta^3\sqrt n}
\leq \Phi(t + \delta) + \frac{K(\beta +
\gamma)}{6\,\delta^3\sqrt n}
\leq \Phi(t) + \frac{\delta}{\sqrt{2\pi}} +
\frac{K(\beta + \gamma)}{6\,\delta^3\sqrt n},
\]
porque $\Phi $ é Lipschitz com constante
$\frac1{\sqrt{2\pi}}$ (isso é \omterm{ex:b3:lebesgue:gamma}{densidade} é limitado por
$\frac1{\sqrt{2\pi}}$). O limite inferior simétrico via
$\tilde\psi_\delta $ fornece a estimativa de dois termos
\[
\sup_{t\in\R}\,\bigl|\P(T_n \leq t) - \Phi(t)\bigr| \leq
\frac{K(\beta + \gamma)}{6}\cdot\frac{1}{\delta^3\sqrt n} +
\frac{\delta}{\sqrt{2\pi}}
\qquad(\delta > 0\ \text{arbitrary}).
\]
Os dois termos se equilibram quando $\delta^{-3}n^{-1/2} \asymp
\delta $, ou seja\ $\delta = n^{-1/8}$: ambos são então
$ O(n^{-1/8})$, uma taxa uniforme explícita válida para cada
$ n $. (A taxa \omterm{def:b3:rings:ideal}{ideal} de Berry--Esseen $ C\beta/\sqrt n $
requer a desigualdade de suavização de Fourier; trocando negociações
nitidez para completo elementaridade.)

\textbf{8.} Para $ X_i = 2B_i - 1$ (sinais justos): centrado,
variação $1$, e $\abs{X_i} = 1$ então $\beta = 1$. Pergunta 7
então limites $\sup_t\abs{\P(\frac{S_n}{\sqrt n} \leq t) -
\Phi(t)}$ explícita e uniformemente para \emph{todo} finito
$ n $ --- uma asserção global e não assintótica sobre o
função de distribuição. A estimativa local de
\cref{pb:b3:product:1}, pergunta 7, fornece em vez disso o \omterm{def:b3:forms:closedexact}{exato}
assintótica de um átomo individual, $\P(S_{2n} = 2k) \sim
\frac{\eu^{-k^2/n}}{\sqrt{\pi n}}$: resolve
probabilidades de tamanho $ n^{-1/2}$, muito abaixo da questão 7
$ n^{-1/8}$ resolução, mas é pontual, assintótica
(nenhum erro explícito em fixo $ n $) e vinculado a isso
\omterm{def:b3:probability:space}{lei} particular da rede. Precisão local versus global
uniformidade: os dois métodos são complementares e somando
a estimativa local acima $ k \in \intint{a\sqrt n}{b\sqrt n}$
recupera de Moivre-Laplace em intervalos --- com um mais nítido
taxa, mas apenas para esta \omterm{def:b3:probability:space}{lei}.

\textbf{9.} O argumento da troca não usou nada sobre a \omterm{def:b3:probability:space}{lei}
do $ X_i $ além $\E X_i = 0$, $\E X_i^2 = 1$ e o
finitude de $\E\abs{X_i}^3$: se tivéssemos substituído o \omterm{def:b3:clt:gaussianvector}{Gaussianos}
$ N_i $ por qualquer outra família i.i.d.\ com os mesmos dois primeiros
momentos e terceiro momento finito, a mesma telescopia seria
vinculado $\abs{\E f(\text{sum}_X) - \E f(\text{sum}_Y)}$ por
$ O(n^{-1/2})$ para cada suave $ f $. Estatísticas suaves de
grande \omterm{def:b3:probability:independence}{independente} as somas são, portanto, \emph{universal}: até
um erro quantificado, eles dependem apenas da \omterm{def:b3:probability:space}{lei} dos summands
através de dois números. Este é o princípio da invariância:
provar um teorema limite para a \omterm{def:b3:probability:space}{lei} mais computável (a
\omterm{def:b3:clt:gaussianvector}{Gaussiano}, onde tudo é \omterm{def:b3:forms:closedexact}{exato}), depois transfira-o para
todas as \omterm{def:b3:probability:space}{leis} por troca. O mesmo esquema --- com somas substituídas
por funcionais mais elaborados --- impulsiona o semicírculo de Wigner
\omterm{def:b3:probability:space}{lei} para matrizes aleatórias, a universalidade das raízes de matrizes aleatórias
polinômios e grande parte da probabilidade moderna; o \omterm{ex:b3:groups:actions}{centro}
o teorema do limite é sua primeira e mais \omterm{def:b3:groups:simple}{simples} instância.

\textbf{10.} $ Z $ é uma função de Borel de $(N_{i+1}, \dots,
N_n)$ apenas, enquanto $ A = W_i + \theta h - Z $ e $ h $ são
funções das variáveis restantes do \omterm{def:b3:probability:independence}{independente}
família $(X_1, \dots, X_n, N_1, \dots, N_n)$: pelo
princípio de coalizão (\cref{thm:b3:probability:independence}),
$ Z $ é \omterm{def:b3:probability:independence}{independente} de $(A, h)$. Como soma do \omterm{def:b3:probability:independence}{independente}
$ N_j/\sqrt n \sim \mathcal N(0, \frac1n)$, $ Z \sim \mathcal
N(0, s^2)$ com $ s^2 = \frac{n-i}n $ (\cref{exo:b3:clt:3}),
com \omterm{ex:b3:lebesgue:gamma}{densidade} limitado por $\frac1{s\sqrt{2\pi}}$. A \omterm{def:b3:probability:space}{lei} de
$((A, h), Z)$ é o produto das duas \omterm{def:b3:probability:space}{leis} marginais, então
Tonelli (transferência) congela o primeiro bloco: com $ G(a) =
\E\abs{g(a + Z)} = \int\abs{g(a + z)}\,\varphi_s(z)\,\dd z
\leq \frac{\norm g_{L^1}}{s\sqrt{2\pi}}$ para cada $ a $,
\[
\E\bigl[\abs h^3\abs{g(A + Z)}\bigr] =
\E\bigl[\abs h^3\,G(A)\bigr] \leq
\frac{\norm g_{L^1}}{s\sqrt{2\pi}}\,\E\abs h^3
= \sqrt{\frac{n}{2\pi(n-i)}}\,\norm g_{L^1}\,\E\abs h^3 .
\]

\textbf{11.} A forma integral da fórmula de Taylor segue
integrando $ f(w + h) - f(w) =
h\int_0^1f'(w + \theta h)\,\dd\theta $ por partes duas vezes em
$\theta $. Tirando expectativas no $ i $-ª troca, o
pedidos $0, 1, 2$ cancelar exatamente como na questão 3, e o
dois restos (para $ h = X_i/\sqrt n $
e $ N_i/\sqrt n $) são limitados, por $ i \leq n - 1$, por
questão 10 com $ g = f'''$:
\[
\bigl|\E f(H_i) - \E f(H_{i-1})\bigr| \leq
\int_0^1\frac{(1-\theta)^2}2\,\dd\theta\;
\sqrt{\frac{n}{2\pi(n-i)}}\,\norm{f'''}_{L^1}
\frac{\beta + \gamma}{n^{3/2}}
= \frac{\beta + \gamma}{6\,n^{3/2}}
\sqrt{\frac{n}{2\pi(n-i)}}\,\norm{f'''}_{L^1} .
\]
Resumindo, com $\sum_{i=1}^{n-1}\sqrt{\frac n{n-i}} =
\sqrt n\sum_{m=1}^{n-1}m^{-1/2} \leq 2n $ e adicionando
a pergunta 3 está vinculada à última troca ($ i = n $, não \omterm{def:b3:clt:gaussianvector}{Gaussiano}
esquerda):
\[
\bigl|\E f(T_n) - \E f(G_n)\bigr| \leq
\frac{\beta + \gamma}{3\sqrt{2\pi}}\cdot
\frac{\norm{f'''}_{L^1}}{\sqrt n} +
\frac{M_3(\beta + \gamma)}{6\,n^{3/2}} .
\]
Rampas: $\psi_\delta'''(x) =
\delta^{-3}\psi'''\bigl(\frac{x - t}\delta\bigr)$, então
$ M_3 = K\delta^{-3}$ com $ K = \norm{\psi'''}_\infty $ e
$\norm{\psi_\delta'''}_{L^1} = \delta^{-2}
\norm{\psi'''}_{L^1} = K_1\delta^{-2}$ (substituição). O
sanduíche da questão 7 então dá
\[
\sup_t\,\bigl|\P(T_n \leq t) - \Phi(t)\bigr| \leq
\frac{K_1(\beta + \gamma)}{3\sqrt{2\pi}}\cdot
\frac1{\delta^2\sqrt n} +
\frac{K(\beta + \gamma)}{6}\cdot
\frac1{\delta^3n^{3/2}} + \frac\delta{\sqrt{2\pi}} .
\]
No $\delta = n^{-1/6}$ o primeiro e o terceiro termos são
$ O(n^{-1/6})$ e o do meio $ O(n^{-1})$: um uniforme
taxa $ O(n^{-1/6})$, estritamente melhor que a pergunta 7
$ n^{-1/8}$ --- o \omterm{def:b3:clt:gaussianvector}{Gaussiano} metade do híbrido fez o
suavização extra.

\textbf{12.} $\E N_1^3 = 0$ (integrando ímpar) e
integração por partes dá $\E N_1^4 = 3\,\E N_1^2 = 3$
($\int x^3\cdot x\varphi(x)\dd x =
3\int x^2\varphi $). Para $ f $ de classe $\mathcal C^4$ com
derivativos limitados, expanda cada swap para quarta ordem: o
termos de terceira ordem carregam o fator $\E X_i^3 - \E N_i^3 =
0$ (independência os fatora como na questão 3), então apenas
o resto de quarta ordem
$\int_0^1\frac{(1-\theta)^3}6f^{(4)}(w + \theta
h)h^4\dd\theta $ sobrevive, com $\int_0^1
\frac{(1-\theta)^3}6\dd\theta = \frac1{24}$ e $\E h^4 =
\beta_4n^{-2}$ ou $3n^{-2}$. Pergunta 10 (com $ g =
f^{(4)}$) limita as trocas $ i \leq n - 1$, e somando como
na questão 11:
\[
\bigl|\E f(T_n) - \E f(G_n)\bigr| \leq
\frac{\beta_4 + 3}{12\sqrt{2\pi}}\cdot
\frac{\norm{f^{(4)}}_{L^1}}{n} +
\frac{M_4(\beta_4 + 3)}{24\,n^2} .
\]
Com $\norm{\psi_\delta^{(4)}}_{L^1} = K_2\delta^{-3}$ e
$ M_4 = K'\delta^{-4}$, o limite da função de distribuição
torna-se $ C\delta^{-3}n^{-1} + C'\delta^{-4}n^{-2} +
\frac\delta{\sqrt{2\pi}}$; no $\delta = n^{-1/4}$ o
termos externos são $ O(n^{-1/4})$ e o meio $ O(n^{-1})$:
avaliar $ O(n^{-1/4})$.

\textbf{13.} Com $ k $ momentos combinados os sobreviventes
restante por swap é de ordem $\E\abs h^{k+1} \asymp
n^{-(k+1)/2}$; o \omterm{def:b3:clt:gaussianvector}{Gaussiano} oculto encargos vinculados
$\norm{f^{(k+1)}}_{L^1}$ e a soma dos swaps contribui
o fator $2n $, dando $\asymp\norm{f^{(k+1)}}_{L^1}\,
n^{-(k-1)/2}$ para suave $ f $. Custo das rampas
$\norm{\psi_\delta^{(k+1)}}_{L^1} \asymp \delta^{-k}$, então
o erro da função de distribuição é $\asymp
\delta^{-k}n^{-(k-1)/2} + \delta $, equilibrado em $\delta =
n^{-(k-1)/(2k+2)}$: avaliar $ n^{-(k-1)/(2k+2)}$, que é
$ n^{-1/6}$ para $ k = 2$, $ n^{-1/4}$ para $ k = 3$, e tende
para $ n^{-1/2}$ apenas como $ k \to \infty $ --- mas $ k \geq 4$
forçaria $\E X_1^4 = 3$ e além, ou seja, uma \omterm{def:b3:probability:space}{lei} que
já imita o \omterm{def:b3:clt:gaussianvector}{Gaussiano}. A saturação é
estrutural: troca de acréscimos $ n $ erros de troca \emph{em
valor absoluto}, renunciando a qualquer cancelamento entre
trocas. A prova de Fourier compara funções características,
onde aparecem os erros com suas fases oscilantes;
A suavização da desigualdade de Esseen converte $\abs{\varphi_{T_n}
- \varphi_N}$, integrado contra $\frac{\dd\xi}{\abs\xi}$,
em uma função de distribuição vinculada apenas ao logarítmico
custo e entrega Berry--Esseen's $ C\beta n^{-1/2}$ de
três momentos. Otimidade de negociações de reposição para
robustez --- e, como mostra a Parte VII, para portabilidade.

\textbf{14.} $\Sigma $ é simétrico positivo semidefinido;
com $\Sigma = PDP^{\mathsf T}$ ($ P $ ortogonal, $ D \geq 0$
diagonal, \cref{exo:b3:submanifolds:8}), o simétrico $ C =
P\sqrt DP^{\mathsf T}$ satisfaz $ C^2 = \Sigma $. Para qualquer $ t
\in \R^2$, $\langle t, CZ\rangle = \langle Ct, Z\rangle =
(Ct)_1Z^1 + (Ct)_2Z^2$ é uma combinação linear de
\omterm{def:b3:probability:independence}{independente} \omterm{def:b3:clt:gaussianvector}{Gaussianos}, por isso \omterm{def:b3:clt:gaussianvector}{Gaussiano}
(\cref{exo:b3:clt:3}): $ N = CZ $ é um Vetor gaussiano; é
significa é $0$ e sua covariância $\E[NN^{\mathsf T}] =
C\,\E[ZZ^{\mathsf T}]\,C^{\mathsf T} = CC^{\mathsf T} =
\Sigma $. Momentos: $\norm N^3 \leq (\abs{N_1} +
\abs{N_2})^3 \leq 4(\abs{N_1}^3 + \abs{N_2}^3)$ (convexidade
de $ x^3$ sobre $\R_+$), e cada coordenada é um real
\omterm{def:b3:clt:gaussianvector}{Gaussiano} com momentos de todas as ordens
(\cref{exo:b3:product:10}): $\gamma' < \infty $. Finalmente
cada $\langle t, G_n\rangle = \frac1{\sqrt
n}\sum_i\langle t, N_i\rangle $ é uma soma normalizada de
eu.i.d.\ $\mathcal N(0, t^{\mathsf T}\Sigma t)$, portanto
exatamente $\mathcal N(0, t^{\mathsf T}\Sigma t)$: $ G_n $ é um
Vetor gaussiano com média $0$ e covariância $\Sigma $ e
sua \omterm{def:b3:probability:space}{lei} é $\mathcal N(0, \Sigma)$
(\cref{def:b3:clt:gaussianvector}: a \omterm{def:b3:probability:space}{lei} é determinada
por esses dados).

\textbf{15.} Deixar $\phi(t) = f(w + th)$, $ t \in \intcc01$:
$\phi $ é $\mathcal C^3$ com
\[
\phi'''(t) = \sum_{j,k,l\in\{1,2\}}\partial_{jkl}f(w +
th)\,h_jh_kh_l, \qquad \abs{\phi'''(t)} \leq
M_3\Bigl(\sum_j\abs{h_j}\Bigr)^3 = M_3(\abs{h_1} +
\abs{h_2})^3 .
\]
Taylor-Lagrange sob encomenda $3$ para $\phi $ entre $0$ e
$1$ dá a primeira desigualdade; Cauchy-Schwarz dá
$\abs{h_1} + \abs{h_2} \leq \sqrt2\norm h $, de onde
constante $\frac{2\sqrt2M_3}6 = \frac{\sqrt2M_3}3$.

\textbf{16.} Definir $ H_i $ e $ W_i $ como na pergunta 1, agora
em $\R^2$; o argumento da coalizão permanece inalterado. No
$ i $-ésima troca, os termos de primeira ordem fornecem
$\sum_j\E[\partial_jf(W_i)]\,(\E X_{i,j} - \E N_{i,j})/
\sqrt n = 0$ e os termos de segunda ordem dão
$\frac1{2n}\sum_{j,k}\E[\partial_{jk}f(W_i)]\,(\Sigma_{jk}
- \Sigma_{jk}) = 0$: correspondência de médias e covariâncias. Pergunta
15 limita os dois restos:
\[
\bigl|\E f(H_i) - \E f(H_{i-1})\bigr| \leq
\frac{\sqrt2M_3}3\cdot\frac{\E\norm{X_i}^3 +
\E\norm{N_i}^3}{n^{3/2}}
= \frac{\sqrt2M_3(\beta' + \gamma')}{3\,n^{3/2}},
\]
e telescópica sobre o $ n $ trocas:
\[
\Bigl|\E f\Bigl(\frac{S_n}{\sqrt n}\Bigr) - \E
f(G_n)\Bigr| \leq \frac{\sqrt2\,M_3(\beta' +
\gamma')}{3\sqrt n},
\qquad G_n \sim \mathcal N(0, \Sigma)\ \text{exactly} .
\]
Atualizar: $\E\norm{T_n}^2 = \E\norm{X_1}^2 =
\operatorname{tr}\Sigma $ (termos cruzados desaparecem por
independência e centralização), então $\P(\norm{T_n} > A) \leq
\operatorname{tr}\Sigma/A^2$, e da mesma forma para $ N $:
aperto. Dado um limite \omterm{def:b3:topology:continuity}{contínuo} $ f $ e $\varepsilon
> 0$, multiplique por um platô suave $\chi $ igual a $1$ ligado
a bola de raio $ A $ e apoiado no raio $ A + 1$
(acalmar um indicador em $\R^2$,
\cref{thm:b3:lp:regularization}); $ g = f\chi $ é uniformemente
\omterm{def:b3:topology:continuity}{contínuo} com compactar suporte, então é bidimensional
apaziguamento $ g_\eta $ é $\mathcal C^\infty $ com limitado
derivadas de todas as ordens e $\norm{g - g_\eta}_\infty
\leq \varepsilon $ para pequeno $\eta $. Os três-$\varepsilon $
cadeia da pergunta 5 então transfere literalmente: $\E f(T_n)
\to \E f(N)$ para cada limite \omterm{def:b3:topology:continuity}{contínuo} $ f \colon \R^2
\to \R $. Isso é \cref{thm:b3:clt:multiclt} para $ d = 2$,
agora comprovado --- a troca evita o bidimensional
eutodo teorema que o capítulo havia deixado admitia.

\textbf{17.} Para limitado \omterm{def:b3:topology:continuity}{contínuo} $ g \colon \R \to \R $,
o aplicação $ x \mapsto g(\langle t, x\rangle)$ é limitado
\omterm{def:b3:topology:continuity}{contínuo} sobre $\R^2$, então a questão 16 dá $\E g(\langle
t, T_n\rangle) \to \E g(\langle t, N\rangle)$: cada
a projeção converge na distribuição, e $\langle t,
N\rangle \sim \mathcal N(0, t^{\mathsf T}\Sigma t)$. (Este
é a direção fácil de Cramér--Wold: convergência conjunta
implica convergência de todas as imagens lineares.) Aplicação:
$ V_i = (\xi_i, \xi_i^2 - 1)$ são vetores i.i.d.\ centrados
($\E\xi_1^2 = 1$), com entradas de covariância $\V(\xi_1) =
1$, $\operatorname{Cov}(\xi_1, \xi_1^2 - 1) = \E\xi_1^3$
e $\V(\xi_1^2 - 1) = \E\xi_1^4 - 1$; o terceiro momento
$\E\norm{V_1}^3 \leq 4\bigl(\E\abs{\xi_1}^3 +
\E\abs{\xi_1^2 - 1}^3\bigr)$ é finito quando $\xi_1 \in
L^6$. A questão 16 produz a junta exibida \omterm{def:b3:clt:gaussianvector}{Gaussiano}
limite, e \cref{thm:b3:clt:gaussianvector}(2): os dois
coordenadas limite são \omterm{def:b3:probability:independence}{independente} exatamente quando o
covariância $\E\xi_1^3$ desaparece --- para \omterm{def:b3:probability:space}{leis} simétricas,
média empírica e variância empírica dissociadas
assintoticamente.

\textbf{18.} Escrever $ g(x) - g(\theta) = (g'(\theta) +
\eta(x))(x - \theta)$ onde $\eta(x) = \frac{g(x) -
g(\theta)}{x - \theta} - g'(\theta)$ para $ x \neq \theta $
e $\eta(\theta) = 0$: diferenciabilidade em $\theta $
significa precisamente $\eta(x) \to 0$ como $ x \to \theta $.
\emph{Etapa 1:} $\hat\theta_n \to \theta $ em probabilidade:
para $\varepsilon > 0$ e qualquer $ A > 0$, eventualmente
$\varepsilon\sqrt n \geq A $, então $\P(\abs{\hat\theta_n -
\theta} > \varepsilon) \leq \P(\abs{\sqrt
n(\hat\theta_n - \theta)} > A) \to \P(\sigma\abs N > A)$
(funções de distribuição convergem no continuidade pontos
$\pm A $), e o lado direito tende a $0$ como $ A \to
\infty $. \emph{Etapa 2:} $\eta(\hat\theta_n) \to 0$ em
probabilidade: dada $\varepsilon' > 0$, escolha $\delta $ com
$\abs\eta \leq \varepsilon'$ sobre $\abs{x - \theta} \leq
\delta $; então $\P(\abs{\eta(\hat\theta_n)} > \varepsilon')
\leq \P(\abs{\hat\theta_n - \theta} > \delta) \to 0$.
\emph{Etapa 3:}
\[
\sqrt n\bigl(g(\hat\theta_n) - g(\theta)\bigr) =
g'(\theta)\,\sqrt n(\hat\theta_n - \theta) +
\eta(\hat\theta_n)\cdot\sqrt n(\hat\theta_n - \theta) .
\]
Pela regra do produto de Slutsky (\cref{exo:b3:clt:8}, com o
sequência $\eta(\hat\theta_n) \to 0$ em probabilidade e o
cunhado convergente $\sqrt n(\hat\theta_n - \theta)$), o
segundo mandato converge legalmente para $0\cdot\mathcal N(0,
\sigma^2) = 0$, daí para $0$ em probabilidade
(\cref{exo:b3:clt:4}(b)); o primeiro converge legalmente para
$ g'(\theta)\mathcal N(0, \sigma^2)$ (Slutsky novamente, ou o
regra afim para funções características); A soma de Slutsky
regra os reúne: o limite é $\mathcal N(0,
g'(\theta)^2\sigma^2)$.

\textbf{19.} (a) A CLT dá $\sqrt n(\bar X_n - \mu)
\Rightarrow \mathcal N(0, \sigma^2)$; o método delta
com $ g(x) = x^2$, $ g'(\mu) = 2\mu $, dá $\sqrt n(\bar
X_n^2 - \mu^2) \Rightarrow \mathcal N(0, 4\mu^2\sigma^2)$
--- degenerar (limite $0$) quando $\mu = 0$. Nesse caso
a flutuação vive uma escala acima: $ n\bar X_n^2 =
(\sqrt n\,\bar X_n)^2$, e para $ t > 0$
\[
\P\bigl(n\bar X_n^2 \leq t\bigr) = \P\bigl(-\sqrt t \leq
\sqrt n\,\bar X_n \leq \sqrt t\bigr) \longrightarrow
\Phi\Bigl(\frac{\sqrt t}\sigma\Bigr) -
\Phi\Bigl(-\frac{\sqrt t}\sigma\Bigr) = \P(\sigma^2N^2
\leq t) :
\]
$ n\bar X_n^2 \Rightarrow \sigma^2N^2$, o quadrado de um
\omterm{def:b3:clt:gaussianvector}{Gaussiano} (uma \omterm{def:b3:probability:space}{lei} do qui-quadrado) --- quando o primeiro
a derivada morre, o termo de segunda ordem de Taylor dita
um não gaussiano limite. (b) Aqui $\sqrt n(\hat p_n - p)
\Rightarrow \mathcal N(0, p(1 - p))$ e $ g(p) =
\arcsin\sqrt p $ tem $ g'(p) = \frac1{2\sqrt{p(1 - p)}}$, então
$ g'(p)^2\,p(1 - p) = \frac14$: o limite é $\mathcal
N(0, \frac14)$ para cada $ p \in \intoo01$. No
$\arcsin $ escalar o assintótico $95\%$ barra de erro é $\pm
\frac{0.98}{\sqrt n}$, conhecido antecipadamente --- enquanto em
\cref{ex:b3:clt:confidence} a largura envolvia o desconhecido
$\sigma = \sqrt{p(1-p)}$, para ser o pior caso $\frac12$
ou estimado: a transformação \emph{estabiliza} o
variação.

\textbf{20.} Deixar $ A^* = \{k : \mu(\{k\}) > \nu(\{k\})\}$
e $\Delta_k = \mu(\{k\}) - \nu(\{k\})$, então
$\sum_k\Delta_k = 0$. Para qualquer $ A \subseteq \N $: $\mu(A) -
\nu(A) = \sum_{k\in A}\Delta_k \leq \sum_{k\in
A^*}\Delta_k $, com igualdade em $ A = A^*$; e desde o
partes positivas e negativas de $(\Delta_k)$ tem igual
massa total, $\sum_{A^*}\Delta_k =
\frac12\sum_k\abs{\Delta_k}$. Troca $\mu, \nu $ alças
o sinal: $ d_{\mathrm{TV}}(\mu, \nu) =
\frac12\sum_k\abs{\Delta_k}$. Acoplamento: para qualquer $ A $,
\[
\mu(A) - \nu(A) = \E\bigl[\mathbf 1_A(X) - \mathbf
1_A(Y)\bigr] = \E\bigl[(\mathbf 1_A(X) - \mathbf
1_A(Y))\,\mathbf 1_{X\neq Y}\bigr] \leq \P(X \neq Y),
\]
e assumir o supremo $ A $.

\textbf{21.} As duas \omterm{def:b3:probability:space}{leis} cobram: $ k = 0$: $1 - p $ contra
$\eu^{-p}$, com $\eu^{-p} > 1 - p $; $ k = 1$: $ p $ contra
$ p\,\eu^{-p} < p $; $ k \geq 2$: $0$ contra o Poisson
restante $1 - \eu^{-p} - p\eu^{-p} \geq 0$. Por isso
\[
d_{\mathrm{TV}} = \tfrac12\bigl[(\eu^{-p} - 1 + p) + (p -
p\eu^{-p}) + (1 - \eu^{-p} - p\eu^{-p})\bigr] =
\tfrac12\bigl(2p - 2p\eu^{-p}\bigr) = p(1 - \eu^{-p}),
\]
e $1 - \eu^{-p} \leq p $ dá o limite $ p^2$.

\textbf{22.} Escrever $ H_{i-1} = W_i + X_i $ e $ H_i = W_i +
Y_i $ com $ W_i = \sum_{j<i}Y_j + \sum_{j>i}X_j $,
\omterm{def:b3:probability:independence}{independente} do par $(X_i, Y_i)$ (coalizões). Para $ A
\subseteq \N $, condicionando os muitos valores contáveis
por independência,
\[
\P(H_{i-1} \in A) = \sum_{k\geq0}\P(X_i = k)\,\P(W_i + k
\in A),
\]
e da mesma forma para $ H_i $ com $ Y_i $. Subtraindo, com $ c_k
= \P(W_i + k \in A) \in \intcc01$ e $\Delta_k = \P(X_i =
k) - \P(Y_i = k)$ de soma zero:
\[
\abs{\P(H_{i-1} \in A) - \P(H_i \in A)} =
\Bigl|\sum_k\Delta_k\bigl(c_k - \tfrac12\bigr)\Bigr| \leq
\tfrac12\sum_k\abs{\Delta_k} =
d_{\mathrm{TV}}\bigl(\mathcal B(1, p_i), \mathcal
P(p_i)\bigr) .
\]
Telescópica de $ H_0 = S $ a $ H_n = \sum_iY_i \sim
\mathcal P(\lambda)$ (\cref{exo:b3:clt:1}, iterado) e
usando a questão 21:
\[
\abs{\P(S \in A) - \P(\mathcal P(\lambda) \in A)} \leq
\sum_{i=1}^np_i\bigl(1 - \eu^{-p_i}\bigr) \leq
\sum_{i=1}^np_i^2
\qquad\text{for every } A :
\]
A desigualdade de Le Cam. (O limite de acoplamento da questão 20 dá uma
rota alternativa: acoplar cada par em um uniforme
variável para que $\P(X_i \neq Y_i) \leq p_i^2$ e vinculado
$\P(S \neq \sum Y_i)$; a troca não precisa de construção em
tudo.)

\textbf{23.} (a) Com $ p_i = \frac\lambda n $:
$ d_{\mathrm{TV}}(\text{law of }S, \mathcal P(\lambda))
\leq \frac{\lambda^2}n $. Isso aguça
\cref{exo:b3:clt:5} três vezes: um erro explícito em
cada finito $ n $, uniformidade sobre todos os eventos $ A $ de uma vez
(não um intervalo de cada vez) e não há necessidade de igual
$ p_i $ --- apenas $\sum_ip_i^2$ pequeno, por exemplo\ $\sum p_i^2 \leq
\lambda\max_ip_i $: \emph{muitos eventos raros, nenhum
dominante}. (b) Aqui $ n = 500$, $ p_i = \frac1{500}$,
$\lambda = 1$: o modelo de Poisson erra no \omterm{def:b3:rings:primemaximal}{máximo}
$500\cdot\frac1{500^2} = 0.002$ em cada evento; em
nomeadamente, tomando $ A = \{0\}$,
\[
\P(\text{no letter astray}) = \Bigl(1 -
\frac1{500}\Bigr)^{500},
\qquad
\Bigl|\P(\text{no letter astray}) - \eu^{-1}\Bigr| \leq
0.002,
\]
então a resposta é $\eu^{-1} \approx 0.368$ até um
garantido $0.002$ (a verdadeira discrepância é sobre
$4\cdot10^{-4}$). (c) O
O problema se encerra em um método com dois regimes. Quando $ n $
contribuições comparáveis, cada uma com variação
$\frac1n $, correspondência \emph{dois} momentos contra o
\omterm{def:b3:clt:gaussianvector}{Gaussiano} comete erros de troca $ o(\frac1n)$ cada um: as somas vão
\omterm{def:b3:clt:gaussianvector}{Gaussiano} --- com Taylor como ferramenta de comparação local.
Quando $ n $ contribuições são indicadores de probabilidade
$ p_i $, correspondendo ao \emph{média} contra um átomo de Poisson
faz com que cada troca custe $ p_i^2$: contagens de eventos raros vão
Poisson --- com variação total como o local \omterm{def:b3:forms:closedexact}{exato}
comparação. Os mesmos híbridos, o mesmo telescópio, locais diferentes
estimativa: a substituição é uma estratégia, não um teorema, e
o \omterm{def:b3:clt:gaussianvector}{Gaussiano} e os limites de Poisson são seus dois mais antigos
dividendos.

\textbf{24.} A CLT dá $\sqrt n(\bar X_n - \mu)
\Rightarrow \mathcal N(0, \sigma^2)$, e $ g(x) = \ln x $ é
diferenciável em $\mu > 0$ com $ g'(\mu) = \frac1\mu $: o
rendimentos do método delta (Parte VI) $\sqrt n(\ln\bar X_n - \ln\mu)
\Rightarrow \mathcal N(0, \sigma^2/\mu^2)$. Desenrolando o
intervalo $\abs{\ln\bar X_n - \ln\mu} \leq
\frac{1.96\,\sigma}{\mu\sqrt n}$ por exponenciação:
\[
\mu \in \bar X_n\cdot
\eu^{\pm1.96\,\sigma/(\mu\sqrt n)}
\qquad\text{with asymptotic probability } 95\%
\]
(na prática $\sigma/\mu $ é substituído por seu empírico
versão, Slutsky como em \cref{exo:b3:clt:8}). O
intervalo multiplicativo é o natural quando os dados
são positivos com erros proporcionais ao seu tamanho ---
rendimentos, concentrações, meias-vidas: quantidades que vivem
em uma escala logarítmica, onde intervalos aditivos simétricos poderiam
até cruzar zero.

\textbf{25.} $\E[(X - p)^3] = (1-p)^3p + (-p)^3(1 - p) =
p(1-p)\bigl[(1-p)^2 - p^2\bigr] = p(1-p)(1 - 2p)$. No
análise de troca (Parte IV), o principal termo de erro após
combinar dois momentos carrega o \emph{assinado} terceiro
momento: para $ p < \frac12$ é positivo (a \omterm{def:b3:probability:space}{lei} inclina-se
direita: raras grandes excursões acima da média), e o
a aproximação \omterm{def:b3:groups:normal}{normal} perde sistematicamente a massa ---
subestimando a cauda esquerda curta e superestimando a
certo --- com um erro de ordem $ n^{-1/2}$; no $ p =
\frac12$ o terceiro momento desaparece, os jogos de Bernoulli
o \omterm{def:b3:clt:gaussianvector}{Gaussiano} para terceira ordem, e a taxa melhora (Parte
questão do momento combinado de IV). Numericamente: $\P(S = 0) =
0.9^{20} = 0.1216$, enquanto o \omterm{def:b3:clt:gaussianvector}{Gaussiano}
$\mathcal N(2, 1.8)$ dá $\Phi\bigl(\frac{0.5 -
2}{\sqrt{1.8}}\bigr) = \Phi(-1.118) \approx 0.132$: o
curva \omterm{def:b3:groups:normal}{normal}, ignorando a parede em $0$ e do
inclinação para a direita, coloca muita massa na parte inferior --- o
sinal previsto do erro, visível em $ n = 20$.
