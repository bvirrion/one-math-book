\section*{अध्याय \ref{ch:b3:clt} --- अभिलक्षणिक फलन और
केंद्रीय सीमा प्रमेय}

\begin{solution}{exo:b3:clt:1}
$\intcc{-1}1$ पर एकसमान: $\varphi(\xi) =
\frac12\int_{-1}^1\eu^{\iu\xi x}\dd x =
\frac{\sin\xi}{\xi}$ ($\xi = 0$ पर $1$ के बराबर)।
चरघातांकी $\mathcal E(\lambda)$: $\varphi(\xi) =
\lambda\int_0^\infty\eu^{(\iu\xi - \lambda)x}\dd x =
\frac{\lambda}{\lambda - \iu\xi}$ (आद्यंतर $+\infty$ पर
लुप्त हो जाता है, क्योंकि $\operatorname{Re}(\iu\xi -
\lambda) < 0$)। प्वासों $\mathcal P(\lambda)$: विविक्त
बंटन नियमों के लिए अंतरण प्रमेय से,
\[
\varphi(\xi) = \sum_{k\geq0}\eu^{\iu\xi
k}\,\eu^{-\lambda}\frac{\lambda^k}{k!}
= \eu^{-\lambda}\exp\bigl(\lambda\eu^{\iu\xi}\bigr)
= \exp\bigl(\lambda(\eu^{\iu\xi} - 1)\bigr).
\]
द्विपद $\mathcal B(n, p)$: यह $n$ \omterm{def:b3:probability:independence}{स्वतंत्र} बर्नूली चरों का
योग है, जिनमें से प्रत्येक का \omterm{def:b3:clt:cf}{अभिलक्षणिक फलन} $1 - p +
p\eu^{\iu\xi}$ है, अतः $\varphi(\xi) = \bigl(1 - p +
p\eu^{\iu\xi}\bigr)^n$
(\cref{prop:b3:clt:cfbasics}(ख))। प्वासों योज्यता: यदि $X
\sim \mathcal P(\lambda)$, $Y \sim \mathcal P(\mu)$
\omterm{def:b3:probability:independence}{स्वतंत्र} हों, तो
\[
\varphi_{X+Y}(\xi) = \eu^{\lambda(\eu^{\iu\xi}-1)}
\eu^{\mu(\eu^{\iu\xi}-1)} =
\eu^{(\lambda+\mu)(\eu^{\iu\xi}-1)},
\]
जो $\mathcal P(\lambda + \mu)$ का \omterm{def:b3:clt:cf}{अभिलक्षणिक फलन} है; और
एकैकीपन (\cref{thm:b3:clt:injectivity}) \omterm{def:b3:probability:space}{बंटन नियम} की पहचान
कर देता है।
\end{solution}

\begin{solution}{exo:b3:clt:2}
(क) $\overline{\varphi_X(\xi)} = \E\eu^{-\iu\xi X} =
\varphi_{-X}(\xi)$। अतः $\varphi_X$ वास्तविक है तभी और केवल
तभी जब $\varphi_X = \varphi_{-X}$, अर्थात् (एकैकीपन,
\cref{thm:b3:clt:injectivity}) तभी और केवल तभी जब $X$ और
$-X$ का \omterm{def:b3:probability:space}{बंटन नियम} एक ही हो।
(ख) $\varphi_X(\xi_0) = \eu^{\iu\theta}$ लिखिए। तब
\[
\E\bigl[1 - \cos(\xi_0X - \theta)\bigr]
= 1 - \operatorname{Re}\bigl(\eu^{-\iu\theta}
\varphi_X(\xi_0)\bigr) = 1 - 1 = 0 .
\]
समाकल्य अऋणात्मक है, अतः लगभग निश्चित रूप से
$\cos(\xi_0X - \theta) = 1$ (शून्य \omterm{def:b3:probability:space}{प्रत्याशा} वाला अऋणात्मक
चर लगभग निश्चित रूप से लुप्त हो जाता है), अर्थात् लगभग
निश्चित रूप से $\xi_0X - \theta \in 2\pi\Z$: इसलिए $X$ लगभग
निश्चित रूप से समांतर श्रेढ़ी
$\frac{\theta}{\xi_0} + \frac{2\pi}{\xi_0}\Z$ में मान लेता
है। यदि $X$ का कोई \omterm{ex:b3:lebesgue:gamma}{घनत्व} हो, तो यह गणनीय समुच्चय लेबेग-अकिंचन
है, अतः उसकी प्रायिकता $0$ है --- विरोधाभास; इसलिए प्रत्येक
$\xi \neq 0$ के लिए $\abs{\varphi_X(\xi)} < 1$।
\end{solution}

\begin{solution}{exo:b3:clt:3}
\omterm{def:b3:probability:independence}{स्वातंत्र्य} और \cref{prop:b3:clt:cfbasics} से:
\[
\varphi_{X+Y}(\xi) = \eu^{\iu m_1\xi - \sigma_1^2\xi^2/2}\,
\eu^{\iu m_2\xi - \sigma_2^2\xi^2/2}
= \eu^{\iu(m_1+m_2)\xi - (\sigma_1^2+\sigma_2^2)\xi^2/2},
\]
जो $\mathcal N(m_1 + m_2, \sigma_1^2 + \sigma_2^2)$ का
\omterm{def:b3:clt:cf}{अभिलक्षणिक फलन} है; एकैकीपन निष्कर्ष निकाल देता है। आफ़ीन
प्रतिचित्रणों के अंतर्गत स्थायित्व आफ़ीन नियम ही है ($aX + b
\sim \mathcal N(am_1 + b, a^2\sigma_1^2)$, जिसमें अपभ्रष्ट
स्थिति $a = 0$ भी सम्मिलित है), और \omterm{def:b3:probability:independence}{स्वतंत्र} योगों के अंतर्गत
स्थायित्व उपर्युक्त परिकलन पर आगमन से निकलता है।
\emph{आश्रित} \omterm{def:b3:clt:gaussianvector}{गाउसीय} चरों का योग \omterm{def:b3:clt:gaussianvector}{गाउसीय} होना आवश्यक नहीं है:
\cref{exo:b3:clt:9} में $X$ और $Y = \varepsilon X$ दोनों
मानक \omterm{def:b3:clt:gaussianvector}{गाउसीय} हैं, पर $X + Y$ प्रायिकता $\frac12$ के साथ लुप्त
हो जाता है और फिर भी लगभग निश्चित रूप से शून्य नहीं है, अतः
वह \omterm{def:b3:clt:gaussianvector}{गाउसीय} नहीं है।
\end{solution}

\begin{solution}{exo:b3:clt:4}
(क) \emph{सीधा निहितार्थ।} मान लीजिए $t$ $F_X$ का कोई \omterm{def:b3:topology:continuity}{सांतत्य}
बिंदु है और $\delta > 0$। \omterm{def:b3:topology:continuity}{संतत} ढाल
$f^-$ ($\intoc{-\infty}{t-\delta}$ पर $= 1$, $t$ से आगे $0$,
बीच में आफ़ीन) और $f^+$ ($\intoc{-\infty}t$ पर $= 1$, $t +
\delta$ से आगे $0$, बीच में आफ़ीन) लीजिए; तब $f^- \leq
\mathbf 1_{\intoc{-\infty}t} \leq f^+$, अतः
\[
\E f^-(X_n) \leq F_{X_n}(t) \leq \E f^+(X_n),
\]
और बाहरी पद $\E f^\pm(X)$ की ओर अभिसरित होते हैं, जो स्वयं
$F_X(t - \delta)$ और $F_X(t + \delta)$ के बीच दबे हुए हैं।
पहले $n \to \infty$, फिर $\delta \to 0$ लीजिए और $t$ पर
$F_X$ के \omterm{def:b3:topology:continuity}{सांतत्य} का उपयोग कीजिए: $F_{X_n}(t) \to F_X(t)$।

\emph{विलोम।} मान लीजिए $f$ परिबद्ध \omterm{def:b3:topology:continuity}{संतत} है, $M =
\sup\abs f$, $\varepsilon > 0$। $F_X$ के \omterm{def:b3:topology:continuity}{सांतत्य} बिंदु सघन
हैं ($F_X$ के उछाल अधिक से अधिक गणनीय हैं), अतः ऐसे \omterm{def:b3:topology:continuity}{सांतत्य}
बिंदु $a < b$ चुनिए जिनके लिए $F_X(a) < \varepsilon$ और
$1 - F_X(b) < \varepsilon$। \omterm{def:b3:topology:compact}{संहत} $\intcc ab$ पर फलन $f$
एकसमान रूप से \omterm{def:b3:topology:continuity}{संतत} है: $F_X$ के ऐसे \omterm{def:b3:topology:continuity}{सांतत्य} बिंदु
$a = t_0 < t_1 < \dots < t_m = b$ चुनिए जिन पर प्रत्येक
$\intoc{t_{j-1}}{t_j}$ में $f$ का दोलन अधिक से अधिक
$\varepsilon$ हो, और $g = \sum_j
f(t_j)\,\mathbf 1_{\intoc{t_{j-1}}{t_j}}$ रखिए। तब
$\intoc ab$ पर $\abs{f - g}
\leq \varepsilon$, $\abs g \leq M$, और $T = X_n$ अथवा $X$
के लिए:
\[
\bigl|\E f(T) - \E g(T)\bigr| \leq \varepsilon +
2M\bigl(F_T(a) + 1 - F_T(b)\bigr).
\]
इसके अतिरिक्त $\E g(X_n) = \sum_j f(t_j)\bigl(F_{X_n}(t_j) -
F_{X_n}(t_{j-1})\bigr) \to \E g(X)$ (अभिसारी पदों का परिमित
योग, क्योंकि सभी $t_j$ \omterm{def:b3:topology:continuity}{सांतत्य} बिंदु हैं), तथा
$F_{X_n}(a) \to F_X(a) < \varepsilon$ और $1 - F_{X_n}(b) \to
1 - F_X(b) < \varepsilon$। सब जोड़ने पर:
$\limsup_n\abs{\E f(X_n) - \E f(X)} \leq 2\varepsilon +
8M\varepsilon$; अब $\varepsilon \to 0$ लीजिए।

(ख) अचर $c$ का बंटन फलन $\mathbf 1_{\intco c\infty}$ है, जो
$c$ के अतिरिक्त सर्वत्र \omterm{def:b3:topology:continuity}{संतत} है। $\varepsilon > 0$ के लिए
बिंदु $c - \varepsilon$ और $c +
\frac\varepsilon2$ \omterm{def:b3:topology:continuity}{सांतत्य} बिंदु हैं, अतः
\[
\P(\abs{X_n - c} > \varepsilon) \leq F_{X_n}(c -
\varepsilon) + 1 - F_{X_n}\Bigl(c + \frac\varepsilon2\Bigr)
\longrightarrow 0 + 1 - 1 = 0 .
\]
\end{solution}

\begin{solution}{exo:b3:clt:5}
मान लीजिए $z_n = p_n(\eu^{\iu\xi} - 1)$, ताकि
$\varphi_{X_n}(\xi) = (1 + z_n)^n$ (\cref{exo:b3:clt:1}) और
$\abs{z_n} \leq 2p_n \to 0$ (ध्यान दीजिए कि $p_n =
\frac{np_n}n \to 0$)। $1 + z_n$ और $\eu^{z_n}$ दोनों का
मापांक अधिक से अधिक $1$ है: त्रिभुज असमिका से $\abs{1 + z_n}
= \abs{(1 - p_n) + p_n\eu^{\iu\xi}} \leq 1$, और
$\abs{\eu^{z_n}} = \eu^{p_n(\cos\xi - 1)}
\leq 1$। दूरबीनी असमिका $\abs{a^n - b^n} \leq
n\abs{a - b}$ (\cref{thm:b3:clt:clt} की उपपत्ति) और घात
श्रेणी परिबंध $\abs{\eu^z - 1 - z} \leq
\abs z^2\eu^{\abs z}$ देते हैं
\[
\bigl|(1 + z_n)^n - \eu^{nz_n}\bigr| \leq n\bigl|1 + z_n -
\eu^{z_n}\bigr| \leq n\,\abs{z_n}^2\,\eu^{\abs{z_n}} \leq
4\eu^2\,np_n^2 = 4\eu^2\,(np_n)\,p_n \longrightarrow 0 .
\]
चूँकि $nz_n = np_n(\eu^{\iu\xi} - 1) \to
\lambda(\eu^{\iu\xi} - 1)$, इसलिए प्रत्येक $\xi$ के लिए
$\varphi_{X_n}(\xi)
\to \exp\bigl(\lambda(\eu^{\iu\xi} - 1)\bigr)$: यह $\mathcal
P(\lambda)$ का \omterm{def:b3:clt:cf}{अभिलक्षणिक फलन} है, और लेवी
(\cref{thm:b3:clt:levy}) से $X_n \Rightarrow \mathcal
P(\lambda)$। संख्यात्मक रूप से: $\P\bigl(\mathcal B(100,
0.02) = 0\bigr) = 0.98^{100} = \eu^{100\ln 0.98} \approx
\eu^{-2.020} \approx 0.1326$, जबकि $\P\bigl(\mathcal P(2) =
0\bigr) = \eu^{-2} \approx 0.1353$: इस कच्चे $n$ पर भी
अंतर दो प्रतिशत का ही है।
\end{solution}

\begin{solution}{exo:b3:clt:6}
(क) एक फेंक का माध्य $\frac72$ और प्रसरण
$\frac{35}{12}$ है, अतः $S$ का माध्य $3500$, प्रसरण
$\frac{35000}{12} \approx 2916.7$ और मानक विचलन
$\approx 54.0$। केंद्रीय सीमा प्रमेय से,
\[
\P(S > 3600) = \P\Bigl(\frac{S - 3500}{54.0} > 1.85\Bigr)
\approx 1 - \Phi(1.85) \approx 0.032 :
\]
अर्थात् लगभग $3\%$ संभावना।
(ख) $S \sim \mathcal B(100, \frac12)$: माध्य $50$, मानक
विचलन $5$। \omterm{def:b3:topology:continuity}{सांतत्य} संशोधन के साथ,
\[
\P(45 \leq S \leq 55) \approx
\Phi\Bigl(\frac{55.5 - 50}{5}\Bigr) -
\Phi\Bigl(\frac{44.5 - 50}{5}\Bigr) = 2\Phi(1.1) - 1
\approx 0.729,
\]
जबकि यथार्थ मान $0.7287$ है; और संशोधन के बिना
$2\Phi(1) - 1 \approx 0.683$, जो लगभग पाँच बिंदु दूर है।
संशोधन इसलिए महत्त्व रखता है कि $S$ जालक चर है: परमाणु $\P(S
= k)$ का अच्छा आसन्नन $\intcc{k - \frac12}{k + \frac12}$ का
\omterm{def:b3:clt:gaussianvector}{गाउसीय} द्रव्यमान करता है, और अंतराल को पूर्णांकों $45$ तथा
$55$ पर काट देने से दोनों सिरों पर आधा-आधा परमाणु छूट जाता
है।
\end{solution}

\begin{solution}{exo:b3:clt:7}
चर $g(U_k)$ \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित (\omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित चरों के
\omterm{def:b3:lebesgue:measurable}{मापनीय} प्रतिबिंब) और वर्ग-समाकलनीय हैं, जिनका माध्य $I$
(अंतरण प्रमेय, \cref{exo:b3:product:9}) और प्रसरण
$\sigma^2$ है। यदि $\sigma > 0$ हो, तो उन पर लगाई गई
\cref{thm:b3:clt:clt} ठीक कथित अभिसरण है:
\[
\sqrt n\,\Bigl(\frac1n\sum_{k\leq n}g(U_k) - I\Bigr) =
\frac{\sum_{k\leq n}\bigl(g(U_k) - I\bigr)}{\sqrt n}
\Longrightarrow \mathcal N(0, \sigma^2)
\]
(और $\sigma = 0$ होने पर $g$ लगभग निश्चित रूप से अचर है और
बायाँ पक्ष सर्वथा लुप्त हो जाता है)। अतः
$\P\bigl(\abs{\frac1n\sum g(U_k) - I} \leq
1.96\,\sigma/\sqrt n\bigr) \to 0.95$: अर्थात् त्रुटि-दंड
$\pm 1.96\,\sigma/\sqrt n$ विमा $d$ को केवल अचर $\sigma$ के
माध्यम से देखता है, $n$ में दर के माध्यम से कभी नहीं। विमा
$d$ में $n$ नोडों वाले मध्यबिंदु नियम का जाल $n^{-1/d}$ है
और $\mathcal C^2$ समाकल्यों के लिए उसकी त्रुटि $n^{-2/d}$
कोटि की। \omterm{ex:b3:probability:sllnapps}{मोंते कार्लो} का $n^{-1/2}$ $n^{-2/d}$ से तेज़ी से
घटता है ठीक तब जब $\frac12 > \frac2d$, अर्थात् $d > 4$:
विमा $5$ से आगे यादृच्छिक प्रतिचयन अनंतस्पर्शी रूप से जालक
को हरा देता है --- विमा का अभिशाप प्रायिकता-आधारित विधियों
को छोड़ देता है, और यही कारण है कि उच्च-विमीय समाकलन पर
\omterm{ex:b3:probability:sllnapps}{मोंते कार्लो} का राज है।
\end{solution}

\begin{solution}{exo:b3:clt:8}
\emph{योग।} स्थिर $\xi$ के लिए:
\[
\bigl|\E\eu^{\iu\xi(X_n+Y_n)} -
\eu^{\iu\xi c}\,\E\eu^{\iu\xi X_n}\bigr|
= \bigl|\E\bigl[\eu^{\iu\xi X_n}\bigl(\eu^{\iu\xi Y_n} -
\eu^{\iu\xi c}\bigr)\bigr]\bigr|
\leq \E\bigl|\eu^{\iu\xi(Y_n - c)} - 1\bigr| .
\]
घटना $\{\abs{Y_n - c} \leq \delta\}$ पर विभाजन कीजिए: वहाँ
$\abs{\eu^{\iu\xi(Y_n-c)} - 1} \leq \abs\xi\,\delta$ (जीवा
चाप से छोटी होती है); और पूरक अधिक से अधिक
$2\,\P(\abs{Y_n - c} > \delta) \to 0$ का योगदान करता है। अतः
प्रत्येक $\delta > 0$ के लिए $\limsup$ $\leq
\abs\xi\,\delta$ है: इसलिए अंतर $0$ की ओर जाता है। चूँकि
$\E\eu^{\iu\xi X_n} \to
\varphi_X(\xi)$, इसलिए $\varphi_{X_n+Y_n}(\xi) \to
\eu^{\iu\xi c}\varphi_X(\xi) = \varphi_{X+c}(\xi)$, और लेवी
(\cref{thm:b3:clt:levy}) से $X_n + Y_n
\Rightarrow X + c$।

\emph{गुणनफल।} पहले, $cX_n \Rightarrow cX$:
$\varphi_{cX_n}(\xi) = \varphi_{X_n}(c\xi) \to
\varphi_X(c\xi) = \varphi_{cX}(\xi)$। इसके बाद, प्रायिकता
में $(Y_n - c)X_n \to 0$: $X_n$ के \omterm{def:b3:probability:space}{बंटन नियम} सुसंहत हैं
(उनके \omterm{def:b3:clt:cf}{अभिलक्षणिक फलन} किसी \omterm{def:b3:clt:cf}{अभिलक्षणिक फलन} की ओर अभिसरित होते
हैं; \cref{thm:b3:clt:levy} का सुसंहतता चरण देखिए), अतः
$\varepsilon > 0$ दिया हो तो ऐसा $M$ चुनिए जिसके साथ सभी $n$
के लिए $\P(\abs{X_n} > M) \leq \varepsilon$; तब
\[
\P\bigl(\abs{(Y_n - c)X_n} > \varepsilon\bigr) \leq
\P(\abs{X_n} > M) + \P\Bigl(\abs{Y_n - c} >
\frac{\varepsilon}{M}\Bigr) \leq \varepsilon + o(1) .
\]
$Y_nX_n = cX_n + (Y_n - c)X_n$ लिखिए और योग वाला भाग लगाइए
(जिसकी उपपत्ति में केवल इतना ही उपयोग हुआ कि प्रायिकता में
$Y_n' := (Y_n - c)X_n \to 0$, अचर $0$ के साथ): अतः $Y_nX_n
\Rightarrow cX$।

\emph{अनुप्रयोग।} बृहत् संख्याओं के प्रबल नियम
(\cref{thm:b3:probability:slln}) से लगभग निश्चित रूप से
$\hat p_n \to p$, अतः \omterm{def:b3:topology:continuity}{सांतत्य} से लगभग निश्चित रूप से
$\hat\sigma_n = \sqrt{\hat p_n(1 - \hat p_n)}
\to \sigma = \sqrt{p(1 - p)} > 0$, इसलिए प्रायिकता में
$\frac{\sigma}{\hat\sigma_n} \to 1$। स्लुत्स्की का गुणनफल
नियम $\frac{S_n - np}{\sigma\sqrt n}
\Rightarrow \mathcal N(0,1)$ को $\frac{S_n -
np}{\hat\sigma_n\sqrt n} = \frac{\sigma}{\hat\sigma_n}\cdot
\frac{S_n - np}{\sigma\sqrt n} \Rightarrow \mathcal N(0,1)$
तक उन्नत कर देता है: अर्थात् केवल आँकड़ों से बना
\emph{उपयोग्य} विश्वास अंतराल $\hat p_n \pm
1.96\,\hat\sigma_n/\sqrt n$ अपना अनंतस्पर्शी $95\%$ स्तर
बनाए रखता है।
\end{solution}

\begin{solution}{pb:b3:clt:1}
\textbf{1.} कुल $(X_1, \dots, X_n, N_1, \dots, N_n)$
\omterm{def:b3:probability:independence}{स्वतंत्र} है: दोनों खंड रचना से एक-दूसरे से \omterm{def:b3:probability:independence}{स्वतंत्र} हैं और
प्रत्येक खंड \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित है। $W_i$ केवल चरों
$(X_j)_{j<i}$ और $(N_j)_{j>i}$ का \omterm{def:b3:lebesgue:measurable}{मापनीय फलन} है, जो सब $X_i$
और $N_i$ से भिन्न हैं: अतः गुट सिद्धांत
(\cref{thm:b3:probability:independence}) से $W_i$ युग्म
$(X_i, N_i)$ से \omterm{def:b3:probability:independence}{स्वतंत्र} है। विभाजन
$H_i = W_i + \frac{X_i}{\sqrt n}$ और $H_{i-1} = W_i +
\frac{N_i}{\sqrt n}$ परिभाषाओं से तत्काल निकलते हैं: $H_i$
से $H_{i-1}$ तक जाने में केवल एक योज्य पद $X_i$ की जगह $N_i$
आ जाता है।

\textbf{2.} कोटि $3$ पर टेलर--लाग्रांज: $w$ और $w + h$ के
बीच कोई ऐसा $c$ है जिसके साथ $f(w + h) = f(w) + f'(w)h +
\frac12f''(w)h^2 + \frac16f'''(c)h^3$, और $\abs{f'''(c)}
\leq M_3$ वह परिबंध दे देता है।

\textbf{3.} उभयनिष्ठ आधार बिंदु $w = W_i$ पर दोनों प्रसारों
को घटाने पर:
\[
f(H_i) - f(H_{i-1}) = f'(W_i)\,\frac{X_i - N_i}{\sqrt n} +
\frac{f''(W_i)}{2}\,\frac{X_i^2 - N_i^2}{n} + R_i,
\qquad
\abs{R_i} \leq \frac{M_3}{6}\cdot
\frac{\abs{X_i}^3 + \abs{N_i}^3}{n^{3/2}} .
\]
अब \omterm{def:b3:probability:space}{प्रत्याशा} लीजिए। प्रश्न 1 से $f'(W_i)$ और $f''(W_i)$
$(X_i, N_i)$ से \omterm{def:b3:probability:independence}{स्वतंत्र} हैं, अतः मिश्रित प्रत्याशाएँ
गुणनखंडित हो जाती हैं:
\begin{align*}
\E\Bigl[f'(W_i)\,\frac{X_i - N_i}{\sqrt n}\Bigr] &=
\E\bigl[f'(W_i)\bigr]\,\frac{\E X_i - \E N_i}{\sqrt n} = 0,
\\
\E\Bigl[f''(W_i)\,\frac{X_i^2 - N_i^2}{n}\Bigr] &=
\E\bigl[f''(W_i)\bigr]\,\frac{1 - 1}{n} = 0 :
\end{align*}
$X_i$ और $N_i$ के पहले दो आघूर्ण \emph{मेल खाते} हैं, और
केवल शेषफल बचता है:
\[
\bigl|\E f(H_i) - \E f(H_{i-1})\bigr| \leq \E\abs{R_i} \leq
\frac{M_3}{6}\cdot\frac{\beta + \gamma}{n^{3/2}} .
\]
\omterm{def:b3:clt:gaussianvector}{गाउसीय} तृतीय आघूर्ण: $\gamma = \E\abs{N_1}^3 =
2\int_0^\infty x^3\,\frac{\eu^{-x^2/2}}{\sqrt{2\pi}}\,\dd x
= \frac{2}{\sqrt{2\pi}}\int_0^\infty 2u\,\eu^{-u}\dd u =
\frac{4}{\sqrt{2\pi}} = \frac{2\sqrt2}{\sqrt\pi}$
(प्रतिस्थापन $u = x^2/2$, फिर $\Gamma(2) = 1$)।

\textbf{4.} $\E f(T_n) - \E f(G_n) =
\sum_{i=1}^n\bigl(\E f(H_i) - \E f(H_{i-1})\bigr)$ का
दूरबीनी योग लीजिए और $n$ में से प्रत्येक पद पर प्रश्न 3
लगाइए:
\[
\bigl|\E f(T_n) - \E f(G_n)\bigr| \leq
n \cdot \frac{M_3(\beta + \gamma)}{6\,n^{3/2}} =
\frac{M_3\,(\beta + \gamma)}{6\,\sqrt n} .
\]

\textbf{5.} $G_n$ प्रत्येक $n$ के लिए \emph{ठीक-ठीक}
$\mathcal N(0,1)$ है (\omterm{def:b3:probability:independence}{स्वतंत्र} मानक \omterm{def:b3:clt:gaussianvector}{गाउसीय} चरों का मानकीकृत
योग, \cref{exo:b3:clt:3}), अतः $\E f(G_n) = \E f(N)$ और
प्रश्न 4 का पाठ यह है कि $f \in
\mathcal C^3_b$ के लिए $\abs{\E f(T_n) - \E f(N)} \leq
\frac{M_3(\beta+\gamma)}{6\sqrt n} \to 0$। \emph{उन्नयन।}
मान लीजिए $f$ परिबद्ध \omterm{def:b3:topology:continuity}{संतत} है, $M = \sup\abs f$,
$\varepsilon > 0$। $\frac1{A^2} \leq \varepsilon$ वाला कोई
$A \geq 1$ चुनिए: $\V(T_n) = 1$ के साथ चेबिशेव सभी $n$ के
लिए $\P(\abs{T_n} > A) \leq \varepsilon$ देती है, और वैसे ही
$\P(\abs N > A) \leq \varepsilon$। कोई ऐसा $\mathcal
C^\infty$ फलन $\chi$ लीजिए जिसके लिए
$\mathbf 1_{\intcc{-A}A} \leq \chi \leq
\mathbf 1_{\intcc{-A-1}{A+1}}$ (एक चिकना पठार, जो
$\mathbf 1_{\intcc{-A-\frac12}{A+\frac12}}$ के मृदुकरण से
बनता है, \cref{thm:b3:lp:regularization}); तब $g = f\chi$
\omterm{def:b3:topology:compact}{संहत} आलंब वाला \omterm{def:b3:topology:continuity}{संतत} फलन है, अतः एकसमान रूप से \omterm{def:b3:topology:continuity}{संतत}, इसलिए
पर्याप्त छोटे $\eta$ के लिए उसका मृदुकरण $g_\eta = g *
\rho_\eta$ $\mathcal C^\infty$ है, जिसके सभी कोटियों के
अवकलज परिबद्ध हैं और $\norm{g - g_\eta}_\infty \leq
\varepsilon$। $T = T_n$ अथवा $N$ के लिए, चूँकि
$\intcc{-A}A$ पर $f = g$ और सर्वत्र $\abs{f - g} \leq 2M$:
\[
\bigl|\E f(T) - \E g_\eta(T)\bigr| \leq
\E\abs{(f - g)(T)} + \norm{g - g_\eta}_\infty
\leq 2M\,\P(\abs T > A) + \varepsilon \leq
(2M + 1)\,\varepsilon .
\]
इसे $\E g_\eta(T_n) \to \E g_\eta(N)$ के साथ मिलाने पर
(प्रश्न 4 लागू है: $g_\eta \in \mathcal C^3_b$):
\[
\limsup_n\;\bigl|\E f(T_n) - \E f(N)\bigr| \leq
2(2M + 1)\,\varepsilon ,
\]
और $\varepsilon$ मनमाना था: अतः प्रत्येक परिबद्ध \omterm{def:b3:topology:continuity}{संतत} $f$ के
लिए $\E f(T_n) \to \E f(N)$, अर्थात् $T_n \Rightarrow
\mathcal N(0,1)$।

\textbf{6.} सम बंटन केवल एक ही वाक्य से भीतर आया: ``$X_i$ और
$N_i$ के पहले दो आघूर्ण एक ही हैं''। अतः मान लीजिए $X_1,
\dots, X_n$ \omterm{def:b3:probability:independence}{स्वतंत्र} और केंद्रित हैं, जिनके प्रसरण
$\sigma_i^2$ और तृतीय आघूर्ण परिमित हैं, $s_n^2 =
\sum_i\sigma_i^2 > 0$, और $N_i \sim
\mathcal N(0, \sigma_i^2)$ सबसे \omterm{def:b3:probability:independence}{स्वतंत्र} लीजिए। संकरों को
मानकीकरण $s_n$ के साथ परिभाषित कीजिए: $H_i =
\frac1{s_n}(\sum_{j\leq i}X_j + \sum_{j>i}N_j)$। $i$-वीं
अदला-बदली में $\E X_i = \E N_i = 0$ और $\E X_i^2 = \E N_i^2
= \sigma_i^2$ फिर से $f'$ और $f''$ वाले पद मार देते हैं, और
शेषफल देता है (मापन से $\E\abs{N_i}^3 = \sigma_i^3\gamma$ का
उपयोग करते हुए):
\[
\bigl|\E f(H_i) - \E f(H_{i-1})\bigr| \leq
\frac{M_3}{6\,s_n^3}\bigl(\E\abs{X_i}^3 +
\sigma_i^3\gamma\bigr) .
\]
दूरबीनी योग लेने पर:
\[
\Bigl|\E f\Bigl(\frac{X_1 + \dots + X_n}{s_n}\Bigr) -
\E f(N)\Bigr| \leq \frac{M_3}{6\,s_n^3}\sum_{i=1}^n
\Bigl(\E\abs{X_i}^3 + \V(X_i)^{3/2}\,\gamma\Bigr) .
\]
चूँकि $\sigma_i^3 = (\E X_i^2)^{3/2} \leq \E\abs{X_i}^3$
(घात--माध्य असमिका, अर्थात् $X_i^2$ पर लगाई गई $t \mapsto
t^{3/2}$ वाली जेनसन), इसलिए दायाँ पक्ष अधिक से अधिक
$\frac{M_3(1 + \gamma)}{6}\cdot
\frac{\sum_i\E\abs{X_i}^3}{s_n^3}$ है: अतः \emph{ल्यापुनोव
की शर्त} $\frac1{s_n^3}\sum_i\E\abs{X_i}^3 \to 0$ के
अंतर्गत मानकीकृत योग \omterm{def:b3:probability:space}{बंटन नियम} में $\mathcal N(0,1)$ की ओर
अभिसरित होते हैं --- अर्थात् सम बंटन के बिना केंद्रीय सीमा
प्रमेय।

\textbf{7.} मान लीजिए $\rho \in \mathcal
C^\infty_c(\intoo01)$ के लिए $\int\rho = 1$, और $\psi(x) =
\int_x^1\rho(s)\dd s$ रखिए: $\psi$ $\mathcal C^\infty$ और
अवर्धमान है, $\R_-$ पर $\psi = 1$ और $\intco1\infty$ पर
$\psi = 0$; और $K = \norm{\psi'''}_\infty$ लीजिए। $t \in \R$
तथा $\delta > 0$ के लिए
$\psi_\delta(x) = \psi\bigl(\frac{x - t}\delta\bigr)$
और $\tilde\psi_\delta(x) = \psi\bigl(\frac{x - t}\delta +
1\bigr)$ परिभाषित कीजिए: ये $\mathcal C^3_b$ हैं जिनका
तृतीय अवकलज $K/\delta^3$ से परिबद्ध है, और
\[
\mathbf 1_{\intoc{-\infty}{t-\delta}} \leq
\tilde\psi_\delta \leq \mathbf 1_{\intoc{-\infty}t} \leq
\psi_\delta \leq \mathbf 1_{\intoc{-\infty}{t+\delta}} .
\]
ऊपरी परिबंध: $\psi_\delta$ पर लगाए गए प्रश्न 4 से ($M_3 =
K/\delta^3$ के साथ),
\[
\P(T_n \leq t) \leq \E\psi_\delta(T_n) \leq
\E\psi_\delta(N) + \frac{K(\beta +
\gamma)}{6\,\delta^3\sqrt n}
\leq \Phi(t + \delta) + \frac{K(\beta +
\gamma)}{6\,\delta^3\sqrt n}
\leq \Phi(t) + \frac{\delta}{\sqrt{2\pi}} +
\frac{K(\beta + \gamma)}{6\,\delta^3\sqrt n},
\]
क्योंकि $\Phi$ अचर $\frac1{\sqrt{2\pi}}$ के साथ लिप्शिट्ज़
है (उसका \omterm{ex:b3:lebesgue:gamma}{घनत्व} $\frac1{\sqrt{2\pi}}$ से परिबद्ध है)।
$\tilde\psi_\delta$ के माध्यम से सममित निचला परिबंध द्विपद
आकलन दे देता है:
\[
\sup_{t\in\R}\,\bigl|\P(T_n \leq t) - \Phi(t)\bigr| \leq
\frac{K(\beta + \gamma)}{6}\cdot\frac{1}{\delta^3\sqrt n} +
\frac{\delta}{\sqrt{2\pi}}
\qquad(\delta > 0\ \text{मनमाना}).
\]
दोनों पद तब संतुलित होते हैं जब $\delta^{-3}n^{-1/2} \asymp
\delta$, अर्थात् $\delta = n^{-1/8}$: तब दोनों $O(n^{-1/8})$
हैं, जो प्रत्येक $n$ के लिए मान्य एक स्पष्ट एकसमान दर है।
(अनुकूलतम बेरी--एसेन दर $C\beta/\sqrt n$ के लिए फूरिये
मृदुकरण असमिका चाहिए; अदला-बदली तीक्ष्णता का सौदा \omterm{def:b3:complete:complete}{पूर्ण}
प्रारंभिकता से करती है।)

\textbf{8.} $X_i = 2B_i - 1$ (निष्पक्ष चिह्न) के लिए: वे
केंद्रित हैं, प्रसरण $1$, और $\abs{X_i} = 1$ अतः $\beta =
1$। तब प्रश्न 7 $\sup_t\abs{\P(\frac{S_n}{\sqrt n} \leq t) -
\Phi(t)}$ को \emph{प्रत्येक} परिमित $n$ के लिए स्पष्ट और
एकसमान रूप से परिबद्ध कर देता है --- अर्थात् बंटन फलन के
बारे में एक वैश्विक, अनंतस्पर्शी-इतर कथन।
\cref{pb:b3:product:1}, प्रश्न 7 का स्थानीय आकलन इसके बजाय
किसी एक परमाणु का यथार्थ अनंतस्पर्शी रूप देता है:
$\P(S_{2n} = 2k) \sim
\frac{\eu^{-k^2/n}}{\sqrt{\pi n}}$: वह $n^{-1/2}$ आकार की
प्रायिकताएँ सुलझा लेता है, जो प्रश्न 7 के $n^{-1/8}$
विभेदन से बहुत नीचे है, पर वह बिंदुशः तथा अनंतस्पर्शी है
(स्थिर $n$ पर कोई स्पष्ट त्रुटि नहीं) और इसी विशेष जालक
\omterm{def:b3:probability:space}{बंटन नियम} से बँधा है। स्थानीय परिशुद्धता बनाम वैश्विक
एकसमानता: दोनों विधियाँ पूरक हैं, और स्थानीय आकलन का $k \in
\intint{a\sqrt n}{b\sqrt n}$ पर योग लेने से अंतरालों पर दे
म्वाव्र--लाप्लास पुनः प्राप्त हो जाती है --- तीक्ष्णतर दर के
साथ, पर केवल इसी \omterm{def:b3:probability:space}{बंटन नियम} के लिए।

\textbf{9.} अदला-बदली तर्क ने $X_i$ के \omterm{def:b3:probability:space}{बंटन नियम} के बारे में
$\E X_i = 0$, $\E X_i^2 = 1$ और $\E\abs{X_i}^3$ की परिमितता
से आगे कुछ भी उपयोग नहीं किया: यदि हमने \omterm{def:b3:clt:gaussianvector}{गाउसीय} चरों $N_i$ के
स्थान पर उन्हीं पहले दो आघूर्णों और परिमित तृतीय आघूर्ण वाला
कोई अन्य \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित कुल रखा होता, तो वही दूरबीनी योग
प्रत्येक चिकने $f$ के लिए $\abs{\E f(\text{योग}_X) - \E
f(\text{योग}_Y)}$ को $O(n^{-1/2})$ से परिबद्ध कर देता। अतः
बड़े \omterm{def:b3:probability:independence}{स्वतंत्र} योगों की चिकनी सांख्यिकी \emph{सार्वत्रिक} हैं:
किसी मात्रात्मक त्रुटि तक वे योज्य पदों के \omterm{def:b3:probability:space}{बंटन नियम} पर केवल
दो संख्याओं के माध्यम से निर्भर हैं। यही निश्चरता सिद्धांत
है: किसी सीमा प्रमेय को सबसे परिकलनीय \omterm{def:b3:probability:space}{बंटन नियम} के लिए सिद्ध
कीजिए (\omterm{def:b3:clt:gaussianvector}{गाउसीय}, जहाँ सब कुछ यथार्थ है), फिर अदला-बदली से उसे
सभी बंटन नियमों तक ले जाइए। यही योजना --- जिसमें योगों के
स्थान पर अधिक जटिल फलनक आ जाते हैं --- यादृच्छिक आव्यूहों के
लिए विग्नर का अर्धवृत्त नियम, यादृच्छिक बहुपदों के मूलों की
सार्वत्रिकता और आधुनिक प्रायिकता का बहुत कुछ चलाती है; और
केंद्रीय सीमा प्रमेय उसका पहला तथा सरलतम उदाहरण है।

\textbf{10.} $Z$ केवल $(N_{i+1}, \dots, N_n)$ का बोरेल फलन
है, जबकि $A = W_i + \theta h - Z$ और $h$ \omterm{def:b3:probability:independence}{स्वतंत्र} कुल
$(X_1, \dots, X_n, N_1, \dots, N_n)$ के शेष चरों के फलन हैं:
अतः गुट सिद्धांत (\cref{thm:b3:probability:independence})
से $Z$ $(A, h)$ से \omterm{def:b3:probability:independence}{स्वतंत्र} है। \omterm{def:b3:probability:independence}{स्वतंत्र}
$N_j/\sqrt n \sim \mathcal N(0, \frac1n)$ के योग के रूप में
$Z \sim \mathcal N(0, s^2)$, जहाँ $s^2 = \frac{n-i}n$
(\cref{exo:b3:clt:3}), और उसका \omterm{ex:b3:lebesgue:gamma}{घनत्व}
$\frac1{s\sqrt{2\pi}}$ से परिबद्ध है। $((A, h), Z)$ का \omterm{def:b3:probability:space}{बंटन
नियम} दोनों उपांत बंटन नियमों का गुणनफल है, अतः टोनेली
(अंतरण) पहले खंड को जमा देता है: प्रत्येक $a$ के लिए $G(a) =
\E\abs{g(a + Z)} = \int\abs{g(a + z)}\,\varphi_s(z)\,\dd z
\leq \frac{\norm g_{L^1}}{s\sqrt{2\pi}}$ के साथ,
\[
\E\bigl[\abs h^3\abs{g(A + Z)}\bigr] =
\E\bigl[\abs h^3\,G(A)\bigr] \leq
\frac{\norm g_{L^1}}{s\sqrt{2\pi}}\,\E\abs h^3
= \sqrt{\frac{n}{2\pi(n-i)}}\,\norm g_{L^1}\,\E\abs h^3 .
\]

\textbf{11.} टेलर सूत्र का समाकल रूप $f(w + h) - f(w) =
h\int_0^1f'(w + \theta h)\,\dd\theta$ का $\theta$ में दो बार
खंडशः समाकलन करने से निकलता है। $i$-वीं अदला-बदली में
\omterm{def:b3:probability:space}{प्रत्याशा} लेने पर कोटि $0, 1, 2$ ठीक वैसे ही निरस्त हो जाती
हैं जैसे प्रश्न 3 में, और दोनों शेषफल ($h = X_i/\sqrt n$
तथा $N_i/\sqrt n$ के लिए) $i \leq n - 1$ पर $g = f'''$ वाले
प्रश्न 10 से परिबद्ध हो जाते हैं:
\[
\bigl|\E f(H_i) - \E f(H_{i-1})\bigr| \leq
\int_0^1\frac{(1-\theta)^2}2\,\dd\theta\;
\sqrt{\frac{n}{2\pi(n-i)}}\,\norm{f'''}_{L^1}
\frac{\beta + \gamma}{n^{3/2}}
= \frac{\beta + \gamma}{6\,n^{3/2}}
\sqrt{\frac{n}{2\pi(n-i)}}\,\norm{f'''}_{L^1} .
\]
$\sum_{i=1}^{n-1}\sqrt{\frac n{n-i}} =
\sqrt n\sum_{m=1}^{n-1}m^{-1/2} \leq 2n$ के साथ योग लेकर, और
अंतिम अदला-बदली ($i = n$, कोई \omterm{def:b3:clt:gaussianvector}{गाउसीय} शेष नहीं) के लिए प्रश्न
3 का परिबंध जोड़कर:
\[
\bigl|\E f(T_n) - \E f(G_n)\bigr| \leq
\frac{\beta + \gamma}{3\sqrt{2\pi}}\cdot
\frac{\norm{f'''}_{L^1}}{\sqrt n} +
\frac{M_3(\beta + \gamma)}{6\,n^{3/2}} .
\]
ढाल: $\psi_\delta'''(x) =
\delta^{-3}\psi'''\bigl(\frac{x - t}\delta\bigr)$, अतः
$K = \norm{\psi'''}_\infty$ के साथ $M_3 = K\delta^{-3}$ और
$\norm{\psi_\delta'''}_{L^1} = \delta^{-2}
\norm{\psi'''}_{L^1} = K_1\delta^{-2}$ (प्रतिस्थापन)। तब
प्रश्न 7 का सैंडविच देता है
\[
\sup_t\,\bigl|\P(T_n \leq t) - \Phi(t)\bigr| \leq
\frac{K_1(\beta + \gamma)}{3\sqrt{2\pi}}\cdot
\frac1{\delta^2\sqrt n} +
\frac{K(\beta + \gamma)}{6}\cdot
\frac1{\delta^3n^{3/2}} + \frac\delta{\sqrt{2\pi}} .
\]
$\delta = n^{-1/6}$ पर पहला और तीसरा पद $O(n^{-1/6})$ हैं
और बीच वाला $O(n^{-1})$: अर्थात् एकसमान दर $O(n^{-1/6})$,
जो प्रश्न 7 के $n^{-1/8}$ से कड़ाई से बेहतर है --- संकर के
\omterm{def:b3:clt:gaussianvector}{गाउसीय} आधे भाग ने अतिरिक्त मृदुकरण कर दिया।

\textbf{12.} $\E N_1^3 = 0$ (विषम समाकल्य), और खंडशः समाकलन
से $\E N_1^4 = 3\,\E N_1^2 = 3$
($\int x^3\cdot x\varphi(x)\dd x =
3\int x^2\varphi$)। परिबद्ध अवकलजों वाले $\mathcal C^4$ वर्ग
के $f$ के लिए प्रत्येक अदला-बदली का चतुर्थ कोटि तक प्रसार
कीजिए: तृतीय कोटि के पद गुणनखंड $\E X_i^3 - \E N_i^3 =
0$ धारण करते हैं (\omterm{def:b3:probability:independence}{स्वातंत्र्य} उन्हें प्रश्न 3 की भाँति
गुणनखंडित कर देता है), अतः केवल चतुर्थ कोटि का शेषफल
$\int_0^1\frac{(1-\theta)^3}6f^{(4)}(w + \theta
h)h^4\dd\theta$ बचता है, जिसमें $\int_0^1
\frac{(1-\theta)^3}6\dd\theta = \frac1{24}$ और $\E h^4 =
\beta_4n^{-2}$ अथवा $3n^{-2}$। प्रश्न 10 ($g =
f^{(4)}$ के साथ) अदला-बदलियों $i \leq n - 1$ को परिबद्ध
करता है, और प्रश्न 11 की भाँति योग लेने पर:
\[
\bigl|\E f(T_n) - \E f(G_n)\bigr| \leq
\frac{\beta_4 + 3}{12\sqrt{2\pi}}\cdot
\frac{\norm{f^{(4)}}_{L^1}}{n} +
\frac{M_4(\beta_4 + 3)}{24\,n^2} .
\]
$\norm{\psi_\delta^{(4)}}_{L^1} = K_2\delta^{-3}$ और
$M_4 = K'\delta^{-4}$ के साथ बंटन-फलन परिबंध
$C\delta^{-3}n^{-1} + C'\delta^{-4}n^{-2} +
\frac\delta{\sqrt{2\pi}}$ हो जाता है; और $\delta = n^{-1/4}$
पर बाहरी पद $O(n^{-1/4})$ तथा बीच वाला $O(n^{-1})$: अर्थात्
दर $O(n^{-1/4})$।

\textbf{13.} $k$ मिलित आघूर्णों के साथ प्रति अदला-बदली बचा
हुआ शेषफल $\E\abs h^{k+1} \asymp
n^{-(k+1)/2}$ कोटि का होता है; छिपे-गाउसीय परिबंध की क़ीमत
$\norm{f^{(k+1)}}_{L^1}$ है और अदला-बदलियों पर योग गुणक $2n$
जोड़ देता है, जिससे चिकने $f$ के लिए
$\asymp\norm{f^{(k+1)}}_{L^1}\,
n^{-(k-1)/2}$ मिलता है। ढालों की क़ीमत
$\norm{\psi_\delta^{(k+1)}}_{L^1} \asymp \delta^{-k}$ है,
अतः बंटन-फलन त्रुटि $\asymp
\delta^{-k}n^{-(k-1)/2} + \delta$ है, जो $\delta =
n^{-(k-1)/(2k+2)}$ पर संतुलित होती है: अर्थात् दर
$n^{-(k-1)/(2k+2)}$, जो $k = 2$ के लिए $n^{-1/6}$ और $k = 3$
के लिए $n^{-1/4}$ है, और $n^{-1/2}$ तक केवल $k \to \infty$
पर पहुँचती है --- पर $k \geq 4$ $\E X_1^4 = 3$ और उससे आगे
अनिवार्य कर देता, अर्थात् ऐसा \omterm{def:b3:probability:space}{बंटन नियम} जो पहले ही \omterm{def:b3:clt:gaussianvector}{गाउसीय}
की नक़ल कर रहा हो। यह संतृप्ति संरचनात्मक है: अदला-बदली $n$
अदला-बदली-त्रुटियों को \emph{निरपेक्ष मान में} जोड़ती है,
और अदला-बदलियों के बीच के हर निरसन का त्याग कर देती है।
फूरिये उपपत्ति अभिलक्षणिक फलनों की तुलना करती है, जहाँ
त्रुटियाँ अपने दोलनशील कलाओं के साथ प्रकट होती हैं; एसेन की
मृदुकरण असमिका $\frac{\dd\xi}{\abs\xi}$ के विरुद्ध समाकलित
$\abs{\varphi_{T_n}
- \varphi_N}$ को केवल लघुगणकीय क़ीमत पर बंटन-फलन परिबंध में
बदल देती है, और तीन आघूर्णों से बेरी--एसेन का $C\beta
n^{-1/2}$ दे देती है। प्रतिस्थापन अनुकूलता का सौदा
प्रबलता से करता है --- और, जैसा भाग VII दिखाता है,
वहनीयता से भी।

\textbf{14.} $\Sigma$ सममित धनात्मक अर्ध-निश्चित है; और
$\Sigma = PDP^{\mathsf T}$ ($P$ लांबिक, $D \geq 0$ विकर्ण,
\cref{exo:b3:submanifolds:8}) के साथ सममित $C =
P\sqrt DP^{\mathsf T}$ $C^2 = \Sigma$ संतुष्ट करता है। किसी
भी $t \in \R^2$ के लिए $\langle t, CZ\rangle = \langle Ct,
Z\rangle = (Ct)_1Z^1 + (Ct)_2Z^2$ \omterm{def:b3:probability:independence}{स्वतंत्र} \omterm{def:b3:clt:gaussianvector}{गाउसीय} चरों का
रैखिक संचय है, अतः \omterm{def:b3:clt:gaussianvector}{गाउसीय}
(\cref{exo:b3:clt:3}): इसलिए $N = CZ$ \omterm{def:b3:clt:gaussianvector}{गाउसीय सदिश} है; उसका
माध्य $0$ है और सहप्रसरण $\E[NN^{\mathsf T}] =
C\,\E[ZZ^{\mathsf T}]\,C^{\mathsf T} = CC^{\mathsf T} =
\Sigma$। आघूर्ण: $\norm N^3 \leq (\abs{N_1} +
\abs{N_2})^3 \leq 4(\abs{N_1}^3 + \abs{N_2}^3)$ ($\R_+$ पर
$x^3$ की उत्तलता), और प्रत्येक निर्देशांक वास्तविक \omterm{def:b3:clt:gaussianvector}{गाउसीय} है
जिसके सभी कोटियों के आघूर्ण हैं
(\cref{exo:b3:product:10}): अतः $\gamma' < \infty$। अंत में
प्रत्येक $\langle t, G_n\rangle = \frac1{\sqrt
n}\sum_i\langle t, N_i\rangle$ \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित
$\mathcal N(0, t^{\mathsf T}\Sigma t)$ चरों का मानकीकृत योग
है, अतः ठीक-ठीक $\mathcal N(0, t^{\mathsf T}\Sigma t)$:
इसलिए $G_n$ माध्य $0$ और सहप्रसरण $\Sigma$ वाला \omterm{def:b3:clt:gaussianvector}{गाउसीय सदिश}
है, और उसका \omterm{def:b3:probability:space}{बंटन नियम} $\mathcal N(0, \Sigma)$
(\cref{def:b3:clt:gaussianvector}: \omterm{def:b3:probability:space}{बंटन नियम} इन्हीं आँकड़ों
से निर्धारित होता है)।

\textbf{15.} मान लीजिए $\phi(t) = f(w + th)$, $t \in
\intcc01$: तब $\phi$ $\mathcal C^3$ है, जिसमें
\[
\phi'''(t) = \sum_{j,k,l\in\{1,2\}}\partial_{jkl}f(w +
th)\,h_jh_kh_l, \qquad \abs{\phi'''(t)} \leq
M_3\Bigl(\sum_j\abs{h_j}\Bigr)^3 = M_3(\abs{h_1} +
\abs{h_2})^3 .
\]
$0$ और $1$ के बीच $\phi$ के लिए कोटि $3$ पर
टेलर--लाग्रांज पहली असमिका दे देता है; और कोशी--श्वार्ज़ से
$\abs{h_1} + \abs{h_2} \leq \sqrt2\norm h$, जिससे अचर
$\frac{2\sqrt2M_3}6 = \frac{\sqrt2M_3}3$ निकलता है।

\textbf{16.} $H_i$ और $W_i$ को प्रश्न 1 की भाँति परिभाषित
कीजिए, अब $\R^2$ में; गुट तर्क अपरिवर्तित है। $i$-वीं
अदला-बदली में प्रथम कोटि के पद
$\sum_j\E[\partial_jf(W_i)]\,(\E X_{i,j} - \E N_{i,j})/
\sqrt n = 0$ देते हैं और द्वितीय कोटि के पद
$\frac1{2n}\sum_{j,k}\E[\partial_{jk}f(W_i)]\,(\Sigma_{jk}
- \Sigma_{jk}) = 0$: माध्य और सहप्रसरण मेल खाते हैं। प्रश्न
15 दोनों शेषफलों को परिबद्ध कर देता है:
\[
\bigl|\E f(H_i) - \E f(H_{i-1})\bigr| \leq
\frac{\sqrt2M_3}3\cdot\frac{\E\norm{X_i}^3 +
\E\norm{N_i}^3}{n^{3/2}}
= \frac{\sqrt2M_3(\beta' + \gamma')}{3\,n^{3/2}},
\]
और $n$ अदला-बदलियों पर दूरबीनी योग लेने पर:
\[
\Bigl|\E f\Bigl(\frac{S_n}{\sqrt n}\Bigr) - \E
f(G_n)\Bigr| \leq \frac{\sqrt2\,M_3(\beta' +
\gamma')}{3\sqrt n},
\qquad G_n \sim \mathcal N(0, \Sigma)\ \text{ठीक-ठीक} .
\]
उन्नयन: $\E\norm{T_n}^2 = \E\norm{X_1}^2 =
\operatorname{tr}\Sigma$ (\omterm{def:b3:probability:independence}{स्वातंत्र्य} और केंद्रण से मिश्रित
पद लुप्त हो जाते हैं), अतः $\P(\norm{T_n} > A) \leq
\operatorname{tr}\Sigma/A^2$, और वैसे ही $N$ के लिए:
अर्थात् सुसंहतता। कोई परिबद्ध \omterm{def:b3:topology:continuity}{संतत} $f$ और $\varepsilon
> 0$ दिया हो, तो त्रिज्या $A$ के गोले पर $1$ के बराबर और
त्रिज्या $A + 1$ में आलंबित किसी चिकने पठार $\chi$ से गुणा
कीजिए ($\R^2$ में किसी सूचक का मृदुकरण,
\cref{thm:b3:lp:regularization}); तब $g = f\chi$ \omterm{def:b3:topology:compact}{संहत} आलंब
वाला एकसमान रूप से \omterm{def:b3:topology:continuity}{संतत} फलन है, अतः उसका द्वि-विमीय मृदुकरण
$g_\eta$ $\mathcal C^\infty$ है, जिसके सभी कोटियों के अवकलज
परिबद्ध हैं और छोटे $\eta$ के लिए $\norm{g - g_\eta}_\infty
\leq \varepsilon$। तब प्रश्न 5 की तीन-$\varepsilon$ शृंखला
शब्दशः स्थानांतरित हो जाती है: प्रत्येक परिबद्ध \omterm{def:b3:topology:continuity}{संतत} $f
\colon \R^2 \to \R$ के लिए $\E f(T_n)
\to \E f(N)$। यह $d = 2$ के लिए \cref{thm:b3:clt:multiclt}
है, जो अब सिद्ध हो गई --- अदला-बदली उस द्वि-विमीय लेवी
प्रमेय को किनारे कर देती है जिसे अध्याय ने अस्वीकृत छोड़
दिया था।

\textbf{17.} परिबद्ध \omterm{def:b3:topology:continuity}{संतत} $g \colon \R \to \R$ के लिए
प्रतिचित्रण $x \mapsto g(\langle t, x\rangle)$ $\R^2$ पर
परिबद्ध \omterm{def:b3:topology:continuity}{संतत} है, अतः प्रश्न 16 से $\E g(\langle
t, T_n\rangle) \to \E g(\langle t, N\rangle)$: अर्थात्
प्रत्येक प्रक्षेप \omterm{def:b3:probability:space}{बंटन नियम} में अभिसरित होता है, और
$\langle t, N\rangle \sim \mathcal N(0, t^{\mathsf T}\Sigma
t)$। (यह क्रामेर--वोल्ड की सरल दिशा है: संयुक्त अभिसरण से
सभी रैखिक प्रतिबिंबों का अभिसरण निकलता है।) अनुप्रयोग:
$V_i = (\xi_i, \xi_i^2 - 1)$ \omterm{def:b3:probability:independence}{स्वतंत्र} सम-बंटित केंद्रित सदिश
हैं ($\E\xi_1^2 = 1$), जिनकी सहप्रसरण प्रविष्टियाँ
$\V(\xi_1) = 1$, $\operatorname{Cov}(\xi_1, \xi_1^2 - 1) =
\E\xi_1^3$ और $\V(\xi_1^2 - 1) = \E\xi_1^4 - 1$ हैं; और
तृतीय आघूर्ण $\E\norm{V_1}^3 \leq 4\bigl(\E\abs{\xi_1}^3 +
\E\abs{\xi_1^2 - 1}^3\bigr)$ तब परिमित है जब $\xi_1 \in
L^6$। प्रश्न 16 कथित संयुक्त \omterm{def:b3:clt:gaussianvector}{गाउसीय} सीमा दे देता है, और
\cref{thm:b3:clt:gaussianvector}(2): सीमा के दोनों
निर्देशांक ठीक तब \omterm{def:b3:probability:independence}{स्वतंत्र} हैं जब सहप्रसरण $\E\xi_1^3$ लुप्त
हो --- अर्थात् सममित बंटन नियमों के लिए आनुभविक माध्य और
आनुभविक प्रसरण अनंतस्पर्शी रूप से विसंयोजित हो जाते हैं।

\textbf{18.} $g(x) - g(\theta) = (g'(\theta) +
\eta(x))(x - \theta)$ लिखिए, जहाँ $x \neq \theta$ के लिए
$\eta(x) = \frac{g(x) - g(\theta)}{x - \theta} - g'(\theta)$
और $\eta(\theta) = 0$: $\theta$ पर अवकलनीयता का अर्थ ठीक
यही है कि $x \to \theta$ पर $\eta(x) \to 0$।
\emph{चरण 1:} प्रायिकता में $\hat\theta_n \to \theta$:
$\varepsilon > 0$ और किसी भी $A > 0$ के लिए अंततः
$\varepsilon\sqrt n \geq A$, अतः $\P(\abs{\hat\theta_n -
\theta} > \varepsilon) \leq \P(\abs{\sqrt
n(\hat\theta_n - \theta)} > A) \to \P(\sigma\abs N > A)$
(बंटन फलन \omterm{def:b3:topology:continuity}{सांतत्य} बिंदुओं $\pm A$ पर अभिसरित होते हैं), और
$A \to \infty$ पर दायाँ पक्ष $0$ की ओर जाता है।
\emph{चरण 2:} प्रायिकता में $\eta(\hat\theta_n) \to 0$:
$\varepsilon' > 0$ दिया हो, तो ऐसा $\delta$ चुनिए जिसके साथ
$\abs{x - \theta} \leq \delta$ पर $\abs\eta \leq
\varepsilon'$; तब $\P(\abs{\eta(\hat\theta_n)} >
\varepsilon') \leq \P(\abs{\hat\theta_n - \theta} > \delta)
\to 0$। \emph{चरण 3:}
\[
\sqrt n\bigl(g(\hat\theta_n) - g(\theta)\bigr) =
g'(\theta)\,\sqrt n(\hat\theta_n - \theta) +
\eta(\hat\theta_n)\cdot\sqrt n(\hat\theta_n - \theta) .
\]
स्लुत्स्की के गुणनफल नियम (\cref{exo:b3:clt:8}, जिसमें
अनुक्रम $\eta(\hat\theta_n) \to 0$ प्रायिकता में और
बंटन-अभिसारी $\sqrt n(\hat\theta_n - \theta)$) से दूसरा पद
\omterm{def:b3:probability:space}{बंटन नियम} में $0\cdot\mathcal N(0,
\sigma^2) = 0$ की ओर अभिसरित होता है, अतः प्रायिकता में $0$
की ओर (\cref{exo:b3:clt:4}(ख)); और पहला \omterm{def:b3:probability:space}{बंटन नियम} में
$g'(\theta)\mathcal N(0, \sigma^2)$ की ओर (फिर स्लुत्स्की,
अथवा अभिलक्षणिक फलनों का आफ़ीन नियम); और स्लुत्स्की का योग
नियम उन्हें जोड़ देता है: सीमा $\mathcal N(0,
g'(\theta)^2\sigma^2)$ है।

\textbf{19.} (क) केंद्रीय सीमा प्रमेय $\sqrt n(\bar X_n -
\mu) \Rightarrow \mathcal N(0, \sigma^2)$ देती है; और $g(x)
= x^2$, $g'(\mu) = 2\mu$ वाली डेल्टा विधि $\sqrt n(\bar
X_n^2 - \mu^2) \Rightarrow \mathcal N(0, 4\mu^2\sigma^2)$
--- जो $\mu = 0$ पर अपभ्रष्ट (सीमा $0$) हो जाती है। उस
स्थिति में उतार-चढ़ाव एक मापक्रम ऊपर रहता है: $n\bar X_n^2 =
(\sqrt n\,\bar X_n)^2$, और $t > 0$ के लिए
\[
\P\bigl(n\bar X_n^2 \leq t\bigr) = \P\bigl(-\sqrt t \leq
\sqrt n\,\bar X_n \leq \sqrt t\bigr) \longrightarrow
\Phi\Bigl(\frac{\sqrt t}\sigma\Bigr) -
\Phi\Bigl(-\frac{\sqrt t}\sigma\Bigr) = \P(\sigma^2N^2
\leq t) :
\]
अर्थात् $n\bar X_n^2 \Rightarrow \sigma^2N^2$, किसी \omterm{def:b3:clt:gaussianvector}{गाउसीय}
का वर्ग (``काई-वर्ग'' \omterm{def:b3:probability:space}{बंटन नियम}) --- जब पहला अवकलज मर जाता
है, तब टेलर का द्वितीय कोटि पद कोई अगाउसीय सीमा तय कर देता
है। (ख) यहाँ $\sqrt n(\hat p_n - p)
\Rightarrow \mathcal N(0, p(1 - p))$ और $g(p) =
\arcsin\sqrt p$ का $g'(p) = \frac1{2\sqrt{p(1 - p)}}$, अतः
$g'(p)^2\,p(1 - p) = \frac14$: इसलिए प्रत्येक $p \in
\intoo01$ के लिए सीमा $\mathcal N(0, \frac14)$ है।
$\arcsin$ मापक्रम पर अनंतस्पर्शी $95\%$ त्रुटि-दंड $\pm
\frac{0.98}{\sqrt n}$ है, जो पहले से ज्ञात है --- जबकि
\cref{ex:b3:clt:confidence} में चौड़ाई में अज्ञात
$\sigma = \sqrt{p(1-p)}$ आता था, जिसे या तो $\frac12$ से
निकृष्टतम स्थिति में बाँधना पड़ता था या आकलित करना पड़ता था:
अर्थात् यह रूपांतर प्रसरण को \emph{स्थिर} कर देता है।

\textbf{20.} मान लीजिए $A^* = \{k : \mu(\{k\}) >
\nu(\{k\})\}$ और $\Delta_k = \mu(\{k\}) - \nu(\{k\})$, अतः
$\sum_k\Delta_k = 0$। किसी भी $A \subseteq \N$ के लिए:
$\mu(A) - \nu(A) = \sum_{k\in A}\Delta_k \leq \sum_{k\in
A^*}\Delta_k$, जिसमें $A = A^*$ पर समता है; और चूँकि
$(\Delta_k)$ के धनात्मक तथा ऋणात्मक भागों का कुल द्रव्यमान
बराबर है, इसलिए $\sum_{A^*}\Delta_k =
\frac12\sum_k\abs{\Delta_k}$। $\mu, \nu$ की अदला-बदली चिह्न
सँभाल लेती है: अतः $d_{\mathrm{TV}}(\mu, \nu) =
\frac12\sum_k\abs{\Delta_k}$। युग्मन: किसी भी $A$ के लिए
\[
\mu(A) - \nu(A) = \E\bigl[\mathbf 1_A(X) - \mathbf
1_A(Y)\bigr] = \E\bigl[(\mathbf 1_A(X) - \mathbf
1_A(Y))\,\mathbf 1_{X\neq Y}\bigr] \leq \P(X \neq Y),
\]
और $A$ पर उच्चतम लीजिए।

\textbf{21.} दोनों \omterm{def:b3:probability:space}{बंटन नियम} इस प्रकार द्रव्यमान देते हैं:
$k = 0$: $1 - p$ बनाम $\eu^{-p}$, जहाँ $\eu^{-p} > 1 - p$;
$k = 1$: $p$ बनाम $p\,\eu^{-p} < p$; $k \geq 2$: $0$ बनाम
प्वासों शेष $1 - \eu^{-p} - p\eu^{-p} \geq 0$। अतः
\[
d_{\mathrm{TV}} = \tfrac12\bigl[(\eu^{-p} - 1 + p) + (p -
p\eu^{-p}) + (1 - \eu^{-p} - p\eu^{-p})\bigr] =
\tfrac12\bigl(2p - 2p\eu^{-p}\bigr) = p(1 - \eu^{-p}),
\]
और $1 - \eu^{-p} \leq p$ से परिबंध $p^2$ मिलता है।

\textbf{22.} $W_i = \sum_{j<i}Y_j + \sum_{j>i}X_j$ के साथ
$H_{i-1} = W_i + X_i$ और $H_i = W_i +
Y_i$ लिखिए, जहाँ $W_i$ युग्म $(X_i, Y_i)$ से \omterm{def:b3:probability:independence}{स्वतंत्र} है
(गुट)। $A \subseteq \N$ के लिए \omterm{def:b3:probability:independence}{स्वातंत्र्य} से गणनीय संख्या
के मानों पर प्रतिबंधन करने पर,
\[
\P(H_{i-1} \in A) = \sum_{k\geq0}\P(X_i = k)\,\P(W_i + k
\in A),
\]
और वैसे ही $H_i$ के लिए $Y_i$ के साथ। घटाने पर, $c_k
= \P(W_i + k \in A) \in \intcc01$ और शून्य योग वाले
$\Delta_k = \P(X_i = k) - \P(Y_i = k)$ के साथ:
\[
\abs{\P(H_{i-1} \in A) - \P(H_i \in A)} =
\Bigl|\sum_k\Delta_k\bigl(c_k - \tfrac12\bigr)\Bigr| \leq
\tfrac12\sum_k\abs{\Delta_k} =
d_{\mathrm{TV}}\bigl(\mathcal B(1, p_i), \mathcal
P(p_i)\bigr) .
\]
$H_0 = S$ से $H_n = \sum_iY_i \sim
\mathcal P(\lambda)$ तक दूरबीनी योग लेकर
(\cref{exo:b3:clt:1}, दोहराया गया) और प्रश्न 21 का उपयोग
करके:
\[
\abs{\P(S \in A) - \P(\mathcal P(\lambda) \in A)} \leq
\sum_{i=1}^np_i\bigl(1 - \eu^{-p_i}\bigr) \leq
\sum_{i=1}^np_i^2
\qquad\text{प्रत्येक } A \text{ के लिए} :
\]
अर्थात् ले काम असमिका। (प्रश्न 20 का युग्मन परिबंध एक
वैकल्पिक मार्ग देता है: प्रत्येक युग्म को एक ही एकसमान चर पर
युग्मित कीजिए ताकि $\P(X_i \neq Y_i) \leq p_i^2$ हो और
$\P(S \neq \sum Y_i)$ परिबद्ध कीजिए; अदला-बदली को कोई रचना
चाहिए ही नहीं।)

\textbf{23.} (क) $p_i = \frac\lambda n$ के साथ:
$d_{\mathrm{TV}}(S \text{ का बंटन नियम}, \mathcal
P(\lambda)) \leq \frac{\lambda^2}n$। यह
\cref{exo:b3:clt:5} को तीन प्रकार से तीक्ष्ण कर देता है:
प्रत्येक परिमित $n$ पर स्पष्ट त्रुटि, एक ही बार में सभी
घटनाओं $A$ पर एकसमानता (एक-एक अंतराल पर नहीं), और समान $p_i$
की कोई आवश्यकता नहीं --- केवल $\sum_ip_i^2$ का छोटा होना,
उदाहरणार्थ $\sum p_i^2 \leq
\lambda\max_ip_i$: अर्थात् \emph{बहुत सी विरल घटनाएँ, कोई भी
प्रभावी नहीं}। (ख) यहाँ $n = 500$, $p_i = \frac1{500}$,
$\lambda = 1$: प्वासों प्रतिरूप की त्रुटि प्रत्येक घटना पर
अधिक से अधिक $500\cdot\frac1{500^2} = 0.002$ है; विशेष रूप
से $A = \{0\}$ लेने पर,
\[
\P(\text{कोई पत्र नहीं भटका}) = \Bigl(1 -
\frac1{500}\Bigr)^{500},
\qquad
\Bigl|\P(\text{कोई पत्र नहीं भटका}) - \eu^{-1}\Bigr| \leq
0.002,
\]
अतः उत्तर प्रत्याभूत $0.002$ तक $\eu^{-1} \approx 0.368$ है
(वास्तविक विसंगति लगभग $4\cdot10^{-4}$ है)। (ग) समस्या दो
व्यवस्थाओं वाली एक ही विधि पर समाप्त होती है। जब $n$ तुलनीय
योगदान प्रत्येक प्रसरण $\frac1n$ धारण करते हैं, तब \omterm{def:b3:clt:gaussianvector}{गाउसीय} के
विरुद्ध \emph{दो} आघूर्णों का मिलान प्रत्येक अदला-बदली की
त्रुटि को $o(\frac1n)$ बना देता है: तब योग \omterm{def:b3:clt:gaussianvector}{गाउसीय} हो जाते
हैं --- और स्थानीय तुलना का साधन टेलर है। और जब $n$ योगदान
प्रायिकता $p_i$ वाले सूचक होते हैं, तब किसी प्वासों परमाणु
के विरुद्ध \emph{माध्य} का मिलान प्रत्येक अदला-बदली की
क़ीमत $p_i^2$ बना देता है: तब विरल घटनाओं की गणनाएँ प्वासों
हो जाती हैं --- और स्थानीय तुलना का यथार्थ साधन कुल विचरण
है। वही संकर, वही दूरबीन, भिन्न स्थानीय आकलन: प्रतिस्थापन
कोई प्रमेय नहीं, एक रणनीति है, और \omterm{def:b3:clt:gaussianvector}{गाउसीय} तथा प्वासों सीमाएँ
उसके दो सबसे पुराने लाभांश हैं।

\textbf{24.} केंद्रीय सीमा प्रमेय $\sqrt n(\bar X_n - \mu)
\Rightarrow \mathcal N(0, \sigma^2)$ देती है, और $g(x) = \ln
x$ $\mu > 0$ पर अवकलनीय है जिसका $g'(\mu) = \frac1\mu$: अतः
डेल्टा विधि (भाग VI) $\sqrt n(\ln\bar X_n - \ln\mu)
\Rightarrow \mathcal N(0, \sigma^2/\mu^2)$ दे देती है।
अंतराल $\abs{\ln\bar X_n - \ln\mu} \leq
\frac{1.96\,\sigma}{\mu\sqrt n}$ को चरघातांक लेकर खोलने पर:
\[
\mu \in \bar X_n\cdot
\eu^{\pm1.96\,\sigma/(\mu\sqrt n)}
\qquad\text{अनंतस्पर्शी प्रायिकता } 95\% \text{ के साथ}
\]
(व्यवहार में $\sigma/\mu$ के स्थान पर उसका आनुभविक रूप रखा
जाता है, \cref{exo:b3:clt:8} की भाँति स्लुत्स्की से)। गुणक
अंतराल तब स्वाभाविक है जब आँकड़े धनात्मक हों और त्रुटियाँ
उनके आकार के समानुपाती --- आय, सांद्रता, अर्ध-आयु: अर्थात्
वे राशियाँ जो लघुगणकीय मापक्रम पर बसती हैं, जहाँ सममित
योगात्मक अंतराल शून्य को पार तक कर सकते हैं।

\textbf{25.} $\E[(X - p)^3] = (1-p)^3p + (-p)^3(1 - p) =
p(1-p)\bigl[(1-p)^2 - p^2\bigr] = p(1-p)(1 - 2p)$।
अदला-बदली विश्लेषण (भाग IV) में, दो आघूर्णों के मिलान के बाद
बची प्रमुख त्रुटि \emph{चिह्न सहित} तृतीय आघूर्ण धारण करती
है: $p < \frac12$ के लिए वह धनात्मक है (\omterm{def:b3:probability:space}{बंटन नियम} दाईं ओर
झुकता है: माध्य से ऊपर विरल बड़े उत्क्रमण), और प्रसामान्य
आसन्नन द्रव्यमान को व्यवस्थित रूप से ग़लत जगह रखता है ---
छोटी बाईं पुच्छ को कम और दाईं को अधिक आँकते हुए --- जिसकी
त्रुटि $n^{-1/2}$ कोटि की है; और $p =
\frac12$ पर तृतीय आघूर्ण लुप्त हो जाता है, बर्नूली \omterm{def:b3:clt:gaussianvector}{गाउसीय} से
तृतीय कोटि तक मेल खा जाती है, और दर सुधर जाती है (भाग IV का
मिलित-आघूर्ण प्रश्न)। संख्यात्मक रूप से: $\P(S = 0) =
0.9^{20} = 0.1216$, जबकि \omterm{def:b3:clt:gaussianvector}{गाउसीय}
$\mathcal N(2, 1.8)$ $\Phi\bigl(\frac{0.5 -
2}{\sqrt{1.8}}\bigr) = \Phi(-1.118) \approx 0.132$ देती है:
अर्थात् प्रसामान्य वक्र, जो $0$ पर की दीवार से और दाईं ओर की
विषमता से अनभिज्ञ है, नीचे बहुत अधिक द्रव्यमान रख देता है ---
ठीक वही त्रुटि-चिह्न जिसकी भविष्यवाणी की गई थी, और वह $n =
20$ पर ही दिख जाता है।
\end{solution}
