\section*{अध्याय \ref{ch:b3:complete} --- पूर्ण समष्टियाँ:
बेयर, आस्कोली, स्टोन--वाइरश्ट्रास}

\begin{solution}{exo:b3:complete:1}
(क) $m \geq n$ के लिए $f_n - f_m$ लंबाई $\frac1n$ के किसी अंतराल के बाहर लुप्त हो जाता
है और $1$ से परिबद्ध है: $\norm{f_n - f_m}_1
\leq \frac1n \to 0$: अतः कोशी। यदि \omterm{def:b3:topology:continuity}{संतत} $f$ के साथ $\norm\cdot_1$ में
$f_n \to f$ हो: तो स्थिर $\alpha <
\frac12$ के लिए $\int_0^{\alpha}\abs f = \lim \int_0^\alpha\abs{f -
f_n + f_n} \leq \lim\bigl(\norm{f - f_n}_1 + 0\bigr) = 0$ (बड़े $n$ के लिए वहाँ $f_n \equiv 0$), अतः
$[0, \frac12)$ पर $f \equiv 0$ (\omterm{def:b3:topology:continuity}{सांतत्य}); और इसी प्रकार $(\frac12, 1]$ पर $f \equiv 1$: कोई \omterm{def:b3:topology:continuity}{संतत} फलन
दोनों नहीं कर सकता। अतः यह समष्टि अपूर्ण है --- उसका \omterm{thm:b3:complete:completion}{पूर्णन} $L^1$
है, जो \cref{ch:b3:lp} में बनाया गया है।

(ख) मान लीजिए $(x_n)$ कोशी है; $\norm{x_{n_{k+1}} - x_{n_k}} \leq 2^{-k}$ वाले $n_1 < n_2 < \cdots$ चुनिए। श्रेणी $\sum_k (x_{n_{k+1}} - x_{n_k})$
निरपेक्ष रूप से अभिसरित होती है, अतः अभिसरित होती है; उसके आंशिक
योग $x_{n_{k+1}} - x_{n_1}$ हैं, इसलिए $(x_{n_k})$ अभिसरित होता है, और अभिसारी उपानुक्रम वाला
कोई कोशी अनुक्रम अभिसरित होता है।
\end{solution}

\begin{solution}{exo:b3:complete:2}
(क) मान लीजिए $E$ \omterm{def:b3:galois:algebraic}{बीजीय} आधार $(e_n)_{n\in\N}$ के साथ \omterm{def:b3:complete:complete}{पूर्ण} है, और $F_n = \operatorname{Vect}(e_1, \dots, e_n)$ लीजिए:
यह संवृत है (परिमित-विमीय उपसमष्टियाँ \omterm{def:b3:complete:complete}{पूर्ण}, अतः संवृत होती हैं ---
दूसरा वर्ष), और उसका \omterm{def:b3:topology:interior}{अंतःभाग} रिक्त है: यदि $B(x, r) \subseteq F_n$ हो, तो $v \notin F_n$ लीजिए;
तब $x + \frac{r}{2\norm v}v \in B(x,r)
\setminus F_n$, जो असंगत है। पर $E = \bigcup F_n$ (प्रत्येक सदिश कोई परिमित संचय है):
यह बेयर (\cref{thm:b3:complete:baire}) का खंडन करता है। समष्टि $\R[X]$ का गणनीय आधार $(X^n)$ है,
अतः कोई भी मानक उसे \omterm{def:b3:complete:complete}{पूर्ण} नहीं बना सकता।

(ख) $\delta > 0$ और कोई अरिक्त विवृत गोला $B$ स्थिर कीजिए; हम $B$ में
$\Omega_\delta = \{x : \operatorname{osc}_xf <
\delta\}$ का कोई बिंदु ढूँढ़ते हैं, जहाँ $\operatorname{osc}_xf = \inf_{V \ni
x}\operatorname{diam} f(V)$। ($\Omega_\delta$ विवृत है: यदि किसी
विवृत $V \ni x$ के लिए $\operatorname{diam}f(V) < \delta$ हो, तो प्रत्येक $y \in V$ का दोलन $< \delta$ होता है।)
समुच्चय
\[
F_N = \bigl\{x : \abs{f_n(x) - f_m(x)} \leq \tfrac\delta3\
\ \forall\, n, m \geq N \bigr\}
\]
संवृत हैं (संवृत समुच्चयों के पूर्व-प्रतिबिंबों के प्रतिच्छेद) और
$\R$ को आच्छादित करते हैं (बिंदुशः अभिसरण $(f_n(x))$ को कोशी बना देता है)।
\omterm{def:b3:complete:complete}{पूर्ण} $\bar B$ के भीतर बेयर लगाने पर: कोई $F_N \cap
\bar B$ किसी गोले $B' = B(x_0, \rho)$ को धारण
करता है। $m \to
\infty$ लेने पर: $B'$ पर $\abs{f_N - f} \leq \frac\delta3$। $x_0$ पर $f_N$ की \omterm{def:b3:topology:continuity}{सांतत्य} से
$B'' \ni x_0$ तक सिकोड़िए जहाँ $\abs{f_N - f_N(x_0)} \leq \frac\delta3$; तब $x \in B''$ के लिए
\[
\abs{f(x) - f(x_0)} \leq \abs{f - f_N}(x) + \abs{f_N(x) -
f_N(x_0)} + \abs{f_N - f}(x_0) \leq \delta:
\]
$\operatorname{diam} f(B'') \leq 2\delta$, अतः $x_0 \in
\Omega_{3\delta} \cap B$। इस प्रकार प्रत्येक $\Omega_\delta$ विवृत और सघन है; और बेयर
से $\bigcap_k\Omega_{1/k}$ --- अर्थात् \omterm{def:b3:topology:continuity}{सांतत्य} बिंदुओं का समुच्चय --- सघन है।
\end{solution}

\begin{solution}{exo:b3:complete:3}
(क) $f(x) = \frac12(x + \frac ax)$ $[\sqrt a, \infty)$ को स्वयं पर भेजता है (समांतर माध्य--गुणोत्तर माध्य:
$f(x) \geq \sqrt{x \cdot \frac ax} = \sqrt a$), और वहाँ $f'(x) = \frac12(1 - \frac a{x^2}) \in [0, \frac12)$: अर्थात् किसी संवृत (\omterm{def:b3:complete:complete}{पूर्ण}) समुच्चय का
$\frac12$-लिपशित्ज़ संकुचन। स्थिर बिंदु: $x = f(x) \iff x^2 = a$: $x^* = \sqrt a$। दर (\cref{thm:b3:complete:banach}): $d(x_n, \sqrt a) \leq
2^{-n+1}\,d(x_1, x_0)$।
$a = 2$, $x_0 = 2$ के लिए: $d(x_1, x_0) =
\frac12$, अतः $n = 40$ $2^{-40} < 10^{-12}$ की प्रत्याभूति देता है।
(वास्तव में न्यूटन की विधि द्विघातीय रूप से अभिसरित होती है: मुट्ठी
भर पुनरावृत्तियाँ पर्याप्त हैं; संकुचन आकलन निराशावादी है पर मुफ़्त।)

(ख) $f_m(x) = m + e\sin x$ \omterm{def:b3:complete:complete}{पूर्ण} $\R$ पर $e < 1$ के साथ $e$-लिपशित्ज़ है: अतः अद्वितीय
स्थिर बिंदु $x(m)$। दो प्राचलों के लिए:
\[
\abs{x(m) - x(m')} = \abs{f_m(x(m)) - f_{m'}(x(m'))}
\leq \abs{m - m'} + e\abs{x(m) - x(m')},
\]
अतः $\abs{x(m) - x(m')} \leq \frac{\abs{m - m'}}{1 - e}$: अर्थात् $m$ में लिपशित्ज़ तक।
\end{solution}

\begin{solution}{exo:b3:complete:4}
(क) मान लीजिए $x^*$ $f^p$ का अद्वितीय स्थिर बिंदु है। तब $f^p(f(x^*)) = f(f^p(x^*)) = f(x^*)$: अतः
$f(x^*)$ $f^p$ का स्थिर बिंदु है, इसलिए $f(x^*) = x^*$। $f$ का कोई स्थिर बिंदु
$f^p$ का भी होता है: अद्वितीयता अंतरित हो जाती है। $T$ के लिए: आगमन
से $\abs{T^pf(x)}
\leq \norm f_\infty x^p/p!$ (प्रत्येक समाकलन एक गुणक $\frac xk$ जोड़ता है), अतः $\norm{T^p} \leq a^p/p!$।
प्रतिचित्रण $S(u) = g +
\lambda Tu$ $S^p(u) - S^p(v) = \lambda^pT^p(u - v)$ संतुष्ट करता है, जिसका मानक $\leq \abs\lambda^pa^p/p!\,\norm{u - v} \to 0$ है: अतः
कोई $S^p$ संकुचन है, और $S$ का कोई अद्वितीय स्थिर बिंदु है: अर्थात्
वोल्तेरा समीकरण $u = g + \lambda Tu$ \emph{प्रत्येक} $\lambda$ के लिए अद्वितीय रूप से
हल-योग्य है।

(ख) $\varphi(x) = d(x, f(x))$ \omterm{def:b3:topology:compact}{संहत} $K$ पर \omterm{def:b3:topology:continuity}{संतत} है: अतः वह किसी $x_0$ पर अपना निम्निष्ठ
प्राप्त कर लेता है। यदि $f(x_0) \neq x_0$ हो, तो $\varphi(f(x_0)) = d(f(x_0), f^2(x_0)) < d(x_0, f(x_0)) =
\min\varphi$: असंगत। अद्वितीयता: दो
स्थिर बिंदु $x \ne y$ $d(x,y) = d(f(x), f(y)) < d(x,y)$ दे देते। संहतता के बिना: $[1, \infty)$ पर $f(x) = x + \frac1x$
$x < y$ के लिए $f(y) - f(x) = (y - x)\bigl(1 - \frac1{xy}\bigr) <
y - x$ संतुष्ट करता है, फिर भी सदैव $f(x) > x$: कोई स्थिर
बिंदु नहीं --- कड़ाई से दूरी घटना किसी एकसमान संकुचन गुणक से दुर्बल
है।
\end{solution}

(ख) $f$, जो कोई समदूरीयता है, एकसमान रूप से \omterm{def:b3:topology:continuity}{संतत} है: अतः वह $F\colon X \to Y$
(\cref{thm:b3:complete:extension}) तक बढ़ जाती है और समदूरीय बनी रहती है (संबंध $d(F(x), F(x')) = d(x, x')$ युग्मों

इसे दो पूर्णनों में सघन रूप से बैठे $D = X$ पर लगाइए।

$\{\norm f_\infty \leq 1, \norm{f'}_\infty \leq 1\}$: माध्य मान असमिका कुल को $1$-लिपशित्ज़ बना देती है, अतः
\omterm{def:b3:complete:equicontinuous}{समसंतत}; और परिबद्ध: इसलिए आस्कोली से सापेक्षतः \omterm{def:b3:topology:compact}{संहत}। (\omterm{def:b3:topology:compact}{संहत} नहीं:

संवरण मानक $\leq 1$ वाले $1$-लिपशित्ज़ फलन हैं।)

$\{\operatorname{Lip}(f) \leq 1, f(0) = 0\}$: \omterm{def:b3:complete:equicontinuous}{समसंतत}; बिंदुशः परिबद्ध ($\abs{f(x)} \leq x \leq 1$); और एकसमान सीमाओं के अंतर्गत
संवृत (लिपशित्ज़ असमिका और $0$ पर मान सीमाओं तक चले जाते हैं): अतः
\omterm{def:b3:topology:compact}{संहत}।
\end{solution}

\begin{solution}{exo:b3:complete:9}

\textbf{13.} $\omega(r) = Lr$: $\int_0^1\frac{\dd r}{Lr} =
+\infty$: योग्य। $\omega(r) = r\log\frac1r$ ($0$ के निकट): $r \to 0$ पर
$\int\frac{\dd r}{r\log(1/r)} = \bigl[-\log\log\frac1r\bigr]
\to +\infty$: योग्य --- फिर भी $\omega(r)/r =
\log\frac1r \to \infty$: लिपशित्ज़ नहीं। $\omega(r) = 2\sqrt

\textbf{14.} दोनों समाकल रूप घटाइए:

\textbf{15.} $\tau$ सुपरिभाषित है ($\delta(t_0) = 0$) जिसमें $\delta(\tau) = 0$ (\omterm{def:b3:topology:continuity}{सांतत्य}) और
$\intoc\tau{t_1}$ पर $\delta > 0$। $s \in \intoo\tau{t_1}$ स्थिर कीजिए। तब $u$ $\mathcal C^1$ है, $[s, t_1]$ पर
$u(s) = \delta(s) > 0$, $u \geq \delta$ (प्रश्न 14), और $u' = \omega(\delta) \leq
\omega(u)$ ($\omega$ अनह्रासमान, $u > 0$)। भाग
दीजिए और समाकलन कीजिए:
\[
\int_{\delta(s)}^{u(t_1)}\frac{\dd r}{\omega(r)}
= \int_s^{t_1}\frac{u'(v)}{\omega(u(v))}\,\dd v \leq t_1 - s
\leq t_1 - \tau .
\]
यद्यपि $u$ $s$ पर निर्भर करता है, परिबंध $u(t_1) \geq
\delta(t_1)$ प्रत्येक $s$

की ओर जाता है (तब $\delta(s) \to \delta(\tau) = 0$, और समाकल $0$ पर अपसारित होता है): इससे
परिबद्ध दाएँ पक्ष का खंडन हो जाता है। अतः $\delta \equiv 0$: अद्वितीयता।

\textbf{16.} लिपशित्ज़ $\omega = Lr$ है: अद्वितीयता पुनः प्राप्त। $x' = x\log\frac1{\abs x}$ के
लिए: दायाँ पक्ष $0$ के निकट ऑस्गुड मापांक $\omega(r) = r\log\frac1r$ संतुष्ट करता है
($x \mapsto x\log\frac1x$ पर माध्य मान असमिका, जिसका अवकलज $\log\frac1x - 1$ अपरिबद्ध है ---
लिपशित्ज़ विफल, ऑस्गुड लागू): अतः अद्वितीय हल; और ध्यान दीजिए कि
$x \equiv 0$ उनमें से एक है, इसलिए कोई अन्य हल $0$ को छू नहीं सकता। $\omega(r) = 2\sqrt r$
के लिए: $\int_0^{h^2}\frac{\dd r}{2\sqrt r} = h$ --- $0$ से ऊँचाई $h^2$ तक चढ़ने का ``ऑस्गुड बजट''
ठीक समय $h$ है, और वस्तुतः $x_c(c + h) = h^2$: हल ठीक उतना समय $h$ वही करने
में बिताता है जितना अभिसारी समाकल अनुमति देता है। समाकल का अपसरण
$0$ को परिमित समय में छोड़ने की असंभवता है; और अभिसरण ही पलायन का
मार्ग।

\textbf{17.} $G(t) = \int_{t_0}^t(Le(s) + \eta(s))\dd s$ $G' = Le + \eta \leq LG + \sup\eta$ के साथ $\mathcal C^1$ है (परिकल्पना $e \leq G$)। तब
$\bigl(\eu^{-Lt}(G +
\tfrac{\sup\eta}L)\bigr)' = \eu^{-Lt}(G' - LG - \sup\eta)
\leq 0$: कोष्ठक घटता है, अतः $G(t) + \frac{\sup\eta}L
\leq \eu^{L(t - t_0)}\bigl(G(t_0) + \frac{\sup\eta}L\bigr) =
\eu^{L(t-t_0)}\frac{\sup\eta}L$, अर्थात् $e(t) \leq G(t) \leq
\frac{\sup\eta}L(\eu^{L(t-t_0)} - 1)$।

\textbf{18.} मात्रात्मक दोष: $(t_k, t_{k+1})$ पर
$\Delta_n(t) = f(t_k, \varphi_n(t_k)) - f(t, \varphi_n(t))$
जिसमें $\abs{t - t_k} \leq T/n$ और $\norm{\varphi_n(t) -
\varphi_n(t_k)} \leq MT/n$, अतः $\norm{\Delta_n} \leq L'T/n +
LMT/n$। $\varphi_n$ (प्रश्न 5) और $\varphi$ की समाकल
सर्वसमिकाएँ घटाकर और लिपशित्ज़ परिबंध का उपयोग करके:
\[
\norm{\varphi_n(t) - \varphi(t)} \leq
\int_{t_0}^t L\,\norm{\varphi_n - \varphi}(s)\,\dd s +
\int_{t_0}^t\norm{\Delta_n(s)}\,\dd s,
\]
और $\eta = \norm{\Delta_n}$ के साथ प्रश्न 17 कहा गया $O(1/n)$ परिबंध दे देता है, जिसमें
$\varphi$ कोशी--लिपशित्ज़ (अथवा ऑस्गुड) से अद्वितीय है। दरों के बिना भी:
समपरिबद्ध, सम-लिपशित्ज़ $(\varphi_n)$ के प्रत्येक उपानुक्रम का कोई उप-उपानुक्रम
\emph{किसी} हल पर अभिसरित होता है (आस्कोली + भाग II), जिसे अद्वितीयता
$\varphi$ होने पर बाध्य कर देती है: और वह अनुक्रम अभिसरित होता है जिसके
सभी उपानुक्रमों के उप-उपानुक्रमों की सीमा एक ही हो।

\textbf{19.} $\varphi_n(t_k) = 0$ से: ढाल $2\sqrt0 =
0$ $\varphi_n(t_{k+1}) = 0$ देती है; आगमन: $\varphi_n
\equiv 0$, जो
शून्य हल पर अभिसरित होता है। $\varepsilon > 0$ से: $\varepsilon' < \varepsilon$ वाले $[\varepsilon', \infty)$ पर फलन $2\sqrt x$
लिपशित्ज़ है, अतः प्रश्न 18 लागू होता है और ऑयलर $(0, \varepsilon)$ से होकर जाते
अद्वितीय हल पर अभिसरित होता है, अर्थात् $x(t) =
(t + \sqrt\varepsilon)^2$ पर (जाँचिए: $x' = 2(t + \sqrt
\varepsilon) = 2\sqrt x$)।
$\varepsilon \to 0$ पर $[0,1]$ पर एकसमान रूप से $(t +
\sqrt\varepsilon)^2 \to t^2$: अतः द्विक सीमा $x_0(t) = t^2$ पर उतरती
है, $0$ पर नहीं। प्रारंभिक आँकड़े का स्वेच्छ रूप से छोटा विक्षोभ
योजना को एक हल से दूसरे पर मोड़ देता है: अद्वितीयता का अभाव, संख्यात्मक
अस्थिरता के रूप में पढ़ा गया।

\textbf{20.} अरिक्त: भाग II। प्रत्येक हल $\norm{x'} = \norm{f(t, x)} \leq M$ संतुष्ट करता है:
$\mathcal S$ एकसमान रूप से $M$-लिपशित्ज़ और एकसमान रूप से परिबद्ध है (मान
$\bar B(x_0, b)$ में)। संवृत: यदि एकसमान रूप से $x_n \in \mathcal S \to x$ हो, तो $x_n(t) = x_0 +
\int_{t_0}^tf(s, x_n(s))\dd s$ में सीमा
लीजिए (\omterm{def:b3:topology:compact}{संहत} $R$ पर $f$ की एकसमान \omterm{def:b3:topology:continuity}{सांतत्य} से समाकल्य एकसमान रूप
से अभिसरित होते हैं): $x \in \mathcal S$। आस्कोली: $\mathcal S$ $\mathcal C([t_0, t_0+T],
\R^d)$ का कोई संवृत,
परिबद्ध, \omterm{def:b3:complete:equicontinuous}{समसंतत} उपसमुच्चय है: अतः \omterm{def:b3:topology:compact}{संहत}।

\textbf{21.} कोई भी हल $x(0) = 0$ के साथ अनह्रासमान ($x' = 2\sqrt{\abs
x} \geq 0$) है, अतः
$x \geq 0$। मान लीजिए $c =
\sup\{t \in [0,1] : x(t) = 0\}$ (यदि $x \equiv 0$ हो तो संभवतः $c = \infty$, जिस दशा में
$x = x_\infty$)। $t > c$ के लिए: $x
> 0$ (एकदिष्टता और $c$ की परिभाषा), और वहाँ
$(\sqrt
x)' = \frac{x'}{2\sqrt x} = 1$, अतः $\sqrt{x(t)} = t - c$ ($c$ पर \omterm{def:b3:topology:continuity}{सांतत्य}): $x = x_c$। $c \mapsto
x_c$ की \omterm{def:b3:topology:continuity}{सांतत्य}: $c, c' \in [0, 1]$
के लिए $\sup_{[0,1]}\abs{(t - c)_+^2 -
(t - c')_+^2} \leq 2\abs{c - c'}$ (प्रतिचित्रण $c \mapsto
(t-c)_+^2$ $t \in [0,1]$ में एकसमान रूप से $2$-लिपशित्ज़
है), और चूँकि प्रत्येक $c \geq 1$ के लिए $[0,1]$ पर $x_c \equiv 0$, अतः कुल $\mathcal S = \{x_c : c \in [0,1]\}$
तक सिमट जाता है ($x_1 = 0 = x_\infty$ के साथ)। इसलिए $\mathcal S$ \omterm{def:b3:topology:compact}{संहत} \omterm{def:b3:topology:connected}{संबद्ध} $[0,1]$ का
\omterm{def:b3:topology:continuity}{संतत प्रतिचित्रण} $c
\mapsto x_c$ के अंतर्गत प्रतिबिंब है: अतः \omterm{def:b3:topology:compact}{संहत} और \omterm{def:b3:topology:connected}{संबद्ध}।
हलों की कीप $t^2$ (तात्कालिक पलायन) से $0$ (चिरंतन विश्राम) तक
चलता एक \omterm{def:b3:topology:continuity}{संतत} रेखाखंड है।

\textbf{22.} मान-निर्धारण $\operatorname{ev}_t\colon
\mathcal C([0,1]) \to \R$, $x \mapsto x(t)$, \omterm{def:b3:topology:continuity}{संतत} है ($\abs{x(t) - y(t)} \leq \norm{x - y}_\infty$), अतः $\mathcal
S(t) = \operatorname{ev}_t(\mathcal S)$
किसी \omterm{def:b3:topology:compact}{संहत} का \omterm{def:b3:topology:continuity}{संतत} प्रतिबिंब है: \omterm{def:b3:topology:compact}{संहत} --- और किसी \omterm{def:b3:topology:connected}{संबद्ध} समुच्चय का
भी: \omterm{def:b3:topology:connected}{संबद्ध}। उदाहरण के लिए: $x_c(t) = (t - c)_+^2$, जैसे-जैसे $c$ $[0, 1]$ पर चलता है,
$c = 0$ पर $t^2$ से लेकर $c \geq t$ पर $0$ तक के सभी मान \omterm{def:b3:topology:continuity}{संतत} रूप से
झाड़ देता है: $\mathcal S(t) =
\intcc0{t^2}$। प्रत्येक क्षण पर कीप का अनुप्रस्थ काट एक पूरा
रेखाखंड है: विश्राम और अधिकतम पलायन के बीच प्रत्येक समझौता किसी
वास्तविक हल से साकार होता है।

\textbf{23.} समाकल रूप $x(t) = x_0 +
\int_{t_0}^tf(s, x(s))\dd s$ और $y$ के लिए उसका समरूप घटाकर, तथा
$e(t) = \norm{x(t) - y(t)}$, $d = \norm{x_0 - y_0}$ रखकर:
\[
e(t) \leq d + \int_{t_0}^tL\,e(s)\dd s = G(t).
\]
तब $G(t_0) = d$ और $G' = Le \leq LG$, अतः $\bigl(\eu^{-L(t
- t_0)}G\bigr)' \leq 0$ और $G(t) \leq d\,\eu^{L(t - t_0)}$; इसलिए $e(t) \leq d\,\eu^{L(t-t_0)}$। तीक्ष्णता:
$x' =
Lx$ के लिए $x_0$ और $y_0$ से होकर जाते हल $x_0\eu^{L(t -
t_0)}$ और $y_0\eu^{L(t-t_0)}$ हैं, जिनकी
दूरी ठीक $d\,\eu^{L(t-t_0)}$ है। $d = 0$ के साथ $e \equiv 0$: कोशी--लिपशित्ज़ अद्वितीयता,
दो पंक्तियों में पुनः व्युत्पन्न। और स्थिर $t \in [t_0, t_0 + T]$ के लिए $\norm{x(t; x_0) - x(t;
y_0)} \leq \eu^{LT}\norm{x_0 - y_0}$: प्रवाह
प्रारंभिक शर्त में लिपशित्ज़ है --- नियतांकी निर्भरता, एक नियंत्रित
चरघातांकी क़ीमत पर।

\textbf{24.} $\intoc01$ पर $\Omega$ सुपरिभाषित है (परिकल्पना से समाकल $0$
पर अभिसरित होता है), $\Omega' = \frac1\omega > 0$ के साथ $\mathcal C^1$: अर्थात् $\intoc0{\Omega(1)}$ पर कोई वर्धमान
एकैकी आच्छादन, जिसमें $x \to 0^+$ पर $\Omega(x) \to 0$। उसका प्रतिलोम $g \colon \intoc0{\Omega(1)} \to \intoc01$ $\mathcal C^1$ है
जिसमें
\[
g'(t) = \frac1{\Omega'(g(t))} = \omega\bigl(g(t)\bigr) > 0,
\qquad g(t) \xrightarrow[t \to 0^+]{} 0 .
\]
$g$ को $t \leq 0$ पर $0$ रखकर बढ़ाइए: \omterm{def:b3:topology:continuity}{सांतत्य} स्पष्ट है, और $t = 0$ पर,
$t > 0$ के लिए,
\[
\frac{g(t)}t = \frac1t\int_0^tg'(s)\dd s
= \frac1t\int_0^t\omega\bigl(g(s)\bigr)\dd s
\leq \omega\bigl(g(t)\bigr) \xrightarrow[t\to0^+]{} 0
\]
($\omega$ अनह्रासमान, $g$ वर्धमान, $\omega(0^+) = 0$): अतः $g'(0) = 0 = f(g(0))$, और $0$ के
दोनों ओर $g' = f(g)$ लागू है। इसलिए $x \equiv 0$ और $g$ $(0, 0)$ से होकर जाते दो
भिन्न $\mathcal
C^1$ हल हैं: अर्थात् जब $\int_0\frac{\dd r}
{\omega(r)}$ अभिसरित होता है तब अद्वितीयता
विफल हो जाती है --- और प्रश्न 15 का अपसरण ठीक वही सीमांत है। $\omega(r) = 2\sqrt
r$
के लिए: $\Omega(x) = \int_0^x\frac{\dd r}{2\sqrt r} = \sqrt x$, $g(t) = t^2$, और समय स्थानांतरण भाग III का समूचा कुल $x_c$
दे देते हैं।

\textbf{25.} चरण $h = \frac1n$ के साथ: $\varphi_n(t_{k+1}) =
\varphi_n(t_k) + h\,\varphi_n(t_k) = (1 + h)\varphi_n(t_k)$, अतः $n$ चरणों के बाद $\varphi_n(1) = (1 + \frac1n)^n$।
प्रसार:
\[
n\log\Bigl(1 + \frac1n\Bigr)
= n\Bigl(\frac1n - \frac1{2n^2} + O\Bigl(\frac1{n^3}\Bigr)
\Bigr) = 1 - \frac1{2n} + O\Bigl(\frac1{n^2}\Bigr),
\]
और चरघातांक लेने पर $(1 + \frac1n)^n = \eu\,\eu^{-1/(2n) +
O(n^{-2})} = \eu\bigl(1 - \frac1{2n} + O(n^{-2})\bigr)$। अतः $t = 1$ पर त्रुटि $\eu - \varphi_n(1) =
\frac{\eu}{2n} + O(n^{-2})$ है: यही प्रश्न
18 का $O(\frac1n)$ है, यहाँ अपने यथार्थ अचर $\frac\eu2$ के साथ। संख्यात्मक रूप
से, $n =
10$: $1.1^{10} = 2.5937424601$ ($1.1^2 = 1.21$, $1.1^4 =
1.4641$, $1.1^8 = 2.14358881$, गुणा $1.21$), और $\eu -
2.59374 \approx 0.12454$, जबकि
अनंतस्पर्शी भविष्यवाणी $\frac{\eu}{20} \approx 0.13591$ है: अर्थात् $O(n^{-2})$ संशोधन के भीतर सहमति,
जिसका अग्रणी पद यहाँ भविष्यवाणी को प्रेक्षित मान की ओर नीचे लाता
है।
