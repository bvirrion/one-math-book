\begin{solution}{exo:b3:holomorphic:1}
(क) $\bar z$: $P = x$, $Q = -y$, अतः $P_x = 1 \neq -1 = Q_y$: कहीं भी $\C$-अवकलनीय नहीं। $\abs z^2$: $P = x^2 + y^2$,
$Q =
0$: कोशी--रीमान $2x = 0 = 2y$ माँगते हैं: अतः केवल $0$ पर अवकलनीय --- और
कहीं भी \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} नहीं (कोई \omterm{def:b3:topology:topology}{विवृत समुच्चय} नहीं)। $\operatorname{Re}z$: $P_x = 1 \neq 0 = Q_y$: कहीं
नहीं।

(ख) $x^2 - y^2 = \operatorname{Re}(z^2)$: $f(z) = z^2$ काम करता है। सभी हल: यदि \omterm{def:b3:topology:connected}{संबद्ध} $\C$ पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक}
$f, g$ के साथ $\operatorname{Re}f = \operatorname{Re}g$ हो, तो $h = f -
g$ का $\operatorname{Re}h = 0$ है; और कोशी--रीमान $h' =
P_x + \iu Q_x = 0 - \iu P_y = 0$ दे देते
हैं: अतः $h$ कोई काल्पनिक अचर है। उत्तर: $f(z) = z^2 + \iu c$, $c \in \R$।
\end{solution}

\begin{solution}{exo:b3:holomorphic:2}
इकाई वृत्त $\gamma(t) = \eu^{\iu t}$ पर:
\[
\int_C z^n\,\dd z = \int_0^{2\pi}\eu^{\iu nt}\,\iu\eu^{\iu
t}\dd t = \iu\int_0^{2\pi}\eu^{\iu(n+1)t}\dd t
= \begin{cases} 2\iu\pi & n = -1,\\ 0 & n \neq -1.\end{cases}
\]
$\bar z$ के लिए: $[0, 1+\iu]$ के अनुदिश $\gamma(t) = t(1 + \iu)$: $\int_0^1 t(1 - \iu)(1 + \iu)\dd t = \int_0^12t\,\dd t = 1$। $0 \to 1 \to 1 + \iu$ के अनुदिश: $\int_0^1t\dd t +
\int_0^1(1 - \iu t)\,\iu\,\dd t = \frac12 + \iu + \frac12 = 1
+ \iu$। एक ही
अंत-बिंदुओं के बीच भिन्न मान: अतः \cref{prop:b3:holomorphic:primitive} से $\bar z$ का दोनों पथों को धारण
करने वाले किसी भी \omterm{def:b3:topology:topology}{विवृत समुच्चय} पर कोई आद्यंतर नहीं है।
\end{solution}

\begin{solution}{exo:b3:holomorphic:7}
$G$ की \omterm{def:b3:topology:continuity}{सांतत्य}: प्रभावी फलन $h$ (स्थानीय एकसमान परिबंध) के साथ
प्रभावी अभिसरण। \omterm{def:b3:holomorphic:holo}{समविश्लेषिकता} मोरेरा से (\cref{thm:b3:holomorphic:weierstrassconv} के भीतर स्थापित):
किसी ऐसे चक्र में जहाँ $\abs g \leq h$ हो, किसी संवृत त्रिभुज $T$ के लिए
\[
\int_{\partial T}G(z)\,\dd z =
\int_X\Bigl(\int_{\partial T}g(x, z)\,\dd z\Bigr)\dd\mu(x) =
0,
\]
जिसमें अदला-बदली फुबिनी ($\int_X\int_{\partial T}\abs
g \leq \operatorname{length}(\partial T)\int h < \infty$) से और आंतरिक लोप गुरसा से है। $\Gamma$
के लिए: पट्टी $a
\leq \operatorname{Re}z \leq b$ ($0 < a \leq b$) पर $\abs{t^{z-1}\eu^{-t}} = t^{\operatorname{Re}z-1}\eu^{-t} \leq
(t^{a-1} + t^{b-1})\eu^{-t}$, जो $\intoo0{+\infty}$ पर \omterm{def:b3:lebesgue:l1}{समाकलनीय} है: अतः
$\Gamma$ $\{\operatorname{Re} z > 0\}$ पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है (\omterm{def:b3:measure:measure}{माप} केवल $\sigma$-परिमित है, पर तर्क को
केवल \omterm{def:b3:lebesgue:l1}{समाकलनीय} प्रभावी फलन चाहिए)। तत्समता प्रमेय से फलनीय समीकरण
$\Gamma(z + 1) = z\Gamma(z)$, जो $\intoo0{+\infty}$ (\cref{ex:b3:lebesgue:gamma}) पर सिद्ध हुआ, समूचे अर्ध-समतल पर लागू है।
\end{solution}

\begin{solution}{exo:b3:holomorphic:8}
$P = c\prod_k(X - a_k)^{m_k}$ लिखिए (\cref{pb:b3:holomorphic:1})। मान लीजिए $P'(z) = 0$। यदि $P(z) = 0$ हो, तो $z$ $a_k$
में से एक है: अतः उत्तल आवरण में। अन्यथा लघुगणकीय अवकलज
\[
0 = \frac{P'(z)}{P(z)} = \sum_k\frac{m_k}{z - a_k}
= \sum_k m_k\,\frac{\bar z - \bar a_k}{\abs{z - a_k}^2} ;
\]
देता है; संयुग्मन करने पर $w_k = m_k/\abs{z -
a_k}^2 > 0$ के साथ $\sum_kw_k(z - a_k) = 0$: $z = \sum_k\frac{w_k}{\sum w}\,a_k$, जो मूलों का कोई
उत्तल संचय है। $P = z^3 - 1$ के लिए: मूल इकाई के घनमूल, और द्विक मूल $0$
के साथ $P' = 3z^2$ --- अर्थात् समबाहु त्रिभुज का \omterm{ex:b3:groups:actions}{केंद्रक}।
\end{solution}

\begin{solution}{exo:b3:holomorphic:11}

\overline{f(\bar z)} = \sum_n\bar c_n\,(z - z_0)^n,

(ग) चक्र पर $G(z) = \overline{f(\bar z)}$ (क) के परिकलन से \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है और व्यास पर $f$

\begin{solution}{exo:b3:holomorphic:12}
(क) $1 = f^2 + g^2 = (f + \iu g)(f - \iu g)$, अतः $h = f +
\iu g$ कभी लुप्त नहीं होता (उसके सहगुणक को फटना पड़ता)।
शून्य-रहित समग्र $h$ के लिए: $h'/h$ समग्र है, और $\C$ तारकाकार है,
अतः उसका कोई आद्यंतर $\varphi_0$ है (\cref{thm:b3:holomorphic:cauchy}); तब $\bigl(h\eu^{-\varphi_0}\bigr)' = \eu^{-\varphi_0}(h' -
h\varphi_0') = 0$: $c \neq 0$ के साथ $h = c\,\eu^{\varphi_0}$,
और किसी अचर $\log c$ ($c$ के किसी सम्मिश्र लघुगणक) को $\varphi = \varphi_0
+ \log c$ में सोख
लेने पर: $h =
\eu^{\varphi}$।

(ख) $h = \eu^{\varphi}$ और $h^{-1} = f - \iu g =
\eu^{-\varphi}$ के साथ:
\[
f = \frac{\eu^{\varphi} + \eu^{-\varphi}}2,
\qquad
g = \frac{\eu^{\varphi} - \eu^{-\varphi}}{2\iu} .
\]
समग्र $\psi = -\iu\varphi$ के साथ $\varphi = \iu\psi$ लिखने पर ये $f = \cos\psi$, $g = \sin\psi$ के रूप में पढ़े
जाते हैं: अतः समग्र हल ठीक समग्र $\psi$ वाले युग्म $(\cos\psi,
\sin\psi)$ हैं, और विलोम
सर्वसमिका $\cos^2 + \sin^2 = 1$ है। $f^2 + g^2 = 0$ के लिए: पूर्णांकीय प्रांत $\mathcal H(\C)$ में $(f + \iu g)(f - \iu g) = 0$
($\C$ \omterm{def:b3:topology:connected}{संबद्ध}: शून्य भाजक तत्समता प्रमेय का उल्लंघन करते): अतः स्वेच्छ
समग्र $f$ के साथ $g = \pm\iu
f$।
\end{solution}

\begin{solution}{pb:b3:holomorphic:1}
\textbf{1.} $\abs z \geq 1$ के लिए:
\[
\abs{P(z)} \geq \abs z^n\Bigl(1 -
\frac{\abs{a_{n-1}}}{\abs z} - \dots -
\frac{\abs{a_0}}{\abs z^n}\Bigr)
\geq \abs z^n\Bigl(1 - \frac{A}{\abs z}\Bigr),
\qquad A = \sum_k\abs{a_k} :
\]
$\abs z \geq R_0 = \max(1, 2A)$, $\abs{P(z)} \geq
\frac12\abs z^n \to \infty$ के लिए।

\textbf{2.} $\frac12R^n \geq
\abs{P(0)}$ वाला $R \geq R_0$ चुनिए। \omterm{def:b3:topology:compact}{संहत} $\bar D(0, R)$ पर \omterm{def:b3:topology:continuity}{संतत} $\abs P$ अपना
निम्निष्ठ किसी $z_0$ पर प्राप्त कर लेता है; और बाहर $\abs P
\geq \frac12R^n \geq \abs{P(0)} \geq \abs{P(z_0)}$: अतः
निम्निष्ठ वैश्विक है।

\textbf{3.} $Q(h) = P(z_0 + h)/P(z_0)$ $Q(0) = 1$ के साथ $h$ में कोई बहुपद है; वह अचरेतर है
($P$ है), अतः अचर पद से आगे कोई गुणांक शून्येतर है: $c_k \neq 0$, $S \in \C[X]$
के साथ न्यूनतम $k \geq 1$ वाला $Q(h) = 1 +
c_kh^k + h^{k+1}S(h)$।

\textbf{4.} मूल: किसी भी $w = \rho\eu^{\iu\varphi} \neq 0$ का $k$-वाँ मूल $\rho^{1/k}\eu^{\iu\varphi/k}$ है, जहाँ $\rho^{1/k}$
$\intco0\infty$ पर $t \mapsto t^k$ पर लगाई गई मध्यवर्ती मान प्रमेय से विद्यमान है ---
कोई चक्रीयता नहीं। $\omega^k = -1/c_k$ वाला $\omega$ चुनिए। तब
\[
Q(t\omega) = 1 - t^k + t^{k+1}\,\omega^{k+1}S(t\omega),
\qquad
\abs{Q(t\omega)} \leq 1 - t^k + C\,t^{k+1}
\quad (0 < t \leq 1),
\]
$C = \abs\omega^{k+1}\sup_{\abs h \leq
\abs\omega}\abs{S(h)}$ के साथ (ध्यान दीजिए कि $\intcc01$ पर $1 - t^k \geq 0$)।

\textbf{5.} $0 < t < \min(1, 1/C)$ के लिए: $\abs{Q(t\omega)}
\leq 1 - t^k(1 - Ct) < 1$, अर्थात् $\abs{P(z_0 + t\omega)} <
\abs{P(z_0)}$ --- जो वैश्विक
न्यूनतमता का खंडन करता है। अतः $P(z_0) = 0$: दालांबेर--आर्गां की उपपत्ति
\omterm{def:b3:complete:complete}{पूर्ण} है।

\textbf{6.} ल्यूविल वाला रूप: यदि $P$ कभी लुप्त न हो, तो $1/P$
समग्र है; प्रश्न 1 से $\bar D(0, R_0)$ के बाहर $\abs{1/P} \leq 2R_0^{-n}$, और $1/P$ उस \omterm{def:b3:topology:compact}{संहत} चक्र पर
\omterm{def:b3:topology:continuity}{संतत} है, अतः वहाँ भी परिबद्ध: इसलिए परिबद्ध समग्र, अतः अचर (\cref{cor:b3:holomorphic:liouville}),
जिससे $P$ अचर हो जाता: असंगत। सामग्री: दालांबेर संहतता (निम्निष्ठ
का अस्तित्व) और $k$-वें मूलों के ध्रुवीय-रूप अस्तित्व का उपयोग
करते हैं; और ल्यूविल समूचे कोशी उपकरण (गुरसा --- जो स्वयं एक नेस्टेड
संहतों का तर्क है --- और \omterm{thm:b3:holomorphic:analytic}{कोशी आकलन}) के साथ वही प्रबलकारिता। संवृत
चक्रों की संहतता ही उभयनिष्ठ, अपरिहार्य गिरी है।

\textbf{7.} यदि $\deg P \geq 1$ हो, तो कोई मूल $\alpha$ चुनिए (प्रश्न 5); भाग
दीजिए: $\deg Q = n - 1$ के साथ $P = (X - \alpha)Q + P(\alpha) = (X -
\alpha)Q$; और आगमन कीजिए। बराबर गुणनखंडों को समूहित
करने पर: $P = c\prod_i(X - \alpha_i)^{m_i}$, $\sum m_i = n$।

\textbf{8.} वास्तविक $P$ के लिए: $P(\bar\alpha) =
\overline{P(\alpha)} = 0$, और बहुलताएँ सहमत होती हैं
(गुणनखंडन का संयुग्मन लीजिए): अतः अवास्तविक मूल युग्मों में आते हैं
और $(X - \alpha)(X - \bar\alpha) = X^2 -
2\operatorname{Re}(\alpha)X + \abs\alpha^2$ का योगदान देते हैं, जो विविक्तकर $< 0$ वाला कोई वास्तविक
द्विघात है। अतः अखंडनीयों की कही गई सूची, और विषम घात का कोई वास्तविक
बहुपद, जिसके अवास्तविक मूलों की संख्या सम है, कोई वास्तविक मूल रखता
ही है। सीधी उपपत्ति: $x \to \pm\infty$ पर $P(x) \to \pm\infty$ (विषम घात, कहिए धनात्मक अग्र
गुणांक), अतः $P$ चिह्न बदलता है, और मध्यवर्ती मान प्रमेय लागू होती
है। $X^3 - X - 1$ के लिए: दोनों तर्क एकमात्र वास्तविक मूल $\approx 1.3247$ (और एक संयुग्म
युग्म) दे देते हैं।

\textbf{9.} (क) $\chi_u \in \C[X]$ अचरेतर है: अतः उसका कोई मूल $\lambda$ है, और $\det(u - \lambda\,\mathrm{id}) = 0$
कोई अभिलक्षणिक सदिश दे देता है; तब \cref{thm:b3:modules:jordan} की प्रारंभिक-भाजक मशीनरी
किसी भी सम्मिश्र आव्यूह पर लागू होती है, जिसमें $\chi$ सदैव विभाजित
होता है। (ख) मान लीजिए $P \in \bar\Q[X]$ अचरेतर है। $\C$ पर बहुपद के रूप में
उसका कोई मूल $z
\in \C$ है; $z$ $\bar\Q$ पर \omterm{def:b3:galois:algebraic}{बीजीय} है, अतः संक्रामिता (\cref{cor:b3:galois:algclosed})
से $\Q$ पर भी, इसलिए $z \in
\bar\Q$: अर्थात् $\bar\Q$ पर प्रत्येक अचरेतर बहुपद
का $\bar\Q$ में कोई मूल है।

\textbf{10.} $\Q(\iu)$ पर प्रश्न 4 पहले ही विफल हो जाता है: $k$-वें
मूलों का होना आवश्यक नहीं (कोई $\sqrt2$ नहीं), और मूल मान भी लें तो
प्रश्न 2 विफल हो जाता है --- $\Q$ में \omterm{def:b3:complete:complete}{पूर्णता} न होने से कोई
न्यूनतमकारी अनुक्रम अभिसरित नहीं होता; और ल्यूविल के मार्ग में गुरसा
के नेस्टेड \omterm{def:b3:topology:compact}{संहत} त्रिभुजों का $\Q(\iu)$-बिंदुओं पर प्रतिच्छेद रिक्त होता
है। दोनों उपपत्तियाँ $\R$ की \omterm{def:b3:complete:complete}{पूर्णता} (समतुल्य रूप से, परिबद्ध
एकदिष्ट अभिसरण के माध्यम से क्रम-पूर्णता) खा जाती हैं; और \omterm{def:b3:topology:connected}{संबद्धता}
ध्रुवीय रूप के पीछे की मध्यवर्ती मान प्रमेय को चलाती है। कथन ``$\C$
\omterm{def:b3:galois:closure}{बीजीय रूप से संवृत} है'' बीजगणित है; पर उसकी प्रत्येक ज्ञात उपपत्ति
$\R$ की परिभाषा के माध्यम से तस्करी किया गया विश्लेषण है।

\textbf{11.} $a$ पर \cref{thm:b3:holomorphic:analytic} से $f^{(n)}(a) = n!\,c_n$, जिसमें
\[
c_n = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{(w -
a)^{n+1}}\,\dd w,
\qquad
\abs{c_n} \leq \frac{2\pi r}{2\pi}\cdot
\frac{\sup_{C_r}\abs f}{r^{n+1}}
= \frac{\sup_{C_r}\abs f}{r^{n}} :
\]
अर्थात् $n!$ से गुणा करने के बाद दिखाए गए रूप में \omterm{thm:b3:holomorphic:analytic}{कोशी आकलन}। यदि
$\C$ पर $\abs f \leq M$ हो: तो प्रत्येक $a$ और प्रत्येक $r$ के लिए $r \to
\infty$
पर $\abs{f'(a)} \leq M/r \to 0$, अतः $f' \equiv 0$ और $f$ \omterm{def:b3:topology:connected}{संबद्ध} $\C$ पर अचर है --- अर्थात्
ल्यूविल पुनः प्राप्त।

\textbf{12.} $0$ पर $f = \sum_kc_kz^k$ का प्रसार कीजिए (त्रिज्या $\infty$)। $k > m$

\sum_{k\leq m}c_kz^k$ अधिक से अधिक घात $m$
का बहुपद है।

\textbf{13.} $g = \eu^f$ $\abs g =
\eu^{\operatorname{Re}f} \leq \eu^M$ के साथ समग्र है: अतः प्रश्न 11 से अचर। तब
शून्य-रहित $g$ के साथ $0 = g' = f'g$: $f' = 0$ और $f$ अचर है।

\textbf{14.} \omterm{def:b3:topology:compact}{संहत} संवृत इकाई वर्ग $K$ पर $M = \sup_K\abs f$ लीजिए। प्रत्येक

को अपरिवर्तित छोड़ देती है: अतः समूचे $\C$ पर $\abs f \leq M$, और प्रश्न 11
$f$ को अचर बना देता है। इसलिए जालक के अंतर्गत निश्चर कोई अचरेतर
फलन समग्र नहीं हो सकता: शास्त्रीय सिद्धांत के दीर्घवृत्तीय फलनों को
ध्रुव ढोने ही पड़ते हैं --- \cref{ch:b3:residues} का ऐतिहासिक द्वार।

\textbf{15.} यदि $f(\C)$ चक्र $D(a, r)$ छोड़ देता हो, तो सभी $z$ के लिए
$\abs{f(z) - a} \geq r$, अतः $g = 1/(f - a)$ $\abs g \leq 1/r$ के साथ समग्र है: ल्यूविल से अचर, इसलिए $f$
अचर। प्रतिधनात्मक रूप में: किसी अचरेतर समग्र फलन का परास प्रत्येक
चक्र से मिलता है --- अर्थात् वह $\C$ में सघन है।

\textbf{16.} $D(0, R)$ के बाहर $\abs f \geq 1$ वाला $R$ चुनिए। $f$ के शून्य
\omterm{def:b3:topology:compact}{संहत} $\bar D(0, R)$ में हैं; और यदि वे अनंत होते, तो वे वहीं संचित होते, और
\cref{thm:b3:holomorphic:identity} $f
\equiv 0$ अनिवार्य कर देता --- असंभव। उन्हें बहुलताओं $m_1, \dots, m_p$ के साथ
$z_1, \dots, z_p$ कहिए, $M = \sum m_i$ रखिए और $\Pi(z) = \prod_i(z - z_i)^{m_i}$। घात श्रेणी में से प्रत्येक शून्य को
गुणनखंडित करने पर $g = f/\Pi$ समग्र और शून्य-रहित है। $\abs z \geq \max(R, 2\max_i\abs{z_i})$ के लिए: $\abs{z - z_i}
\leq 2\abs z$
और $\abs f \geq 1$, अतः $\abs{1/g} =
\abs\Pi/\abs f \leq 2^M\abs z^M$; और शेष \omterm{def:b3:topology:compact}{संहत} चक्र पर $1/g$ \omterm{def:b3:topology:continuity}{संतत}, अतः परिबद्ध है:
इसलिए सर्वत्र $\abs{1/g} \leq A +
B\abs z^M$। प्रश्न 12 से $1/g$ कोई बहुपद है; और वह शून्य-रहित
है, अतः प्रश्न 7 से कोई शून्येतर अचर $c$: इसलिए $f = \frac1c\Pi$ कोई बहुपद
है। विलोमतः प्रश्न 1 प्रत्येक अचरेतर बहुपद को यथोचित बना देता है।
और $\eu^z$ ईमानदारी से अपवर्जित है: $\R_-$ के अनुदिश $\abs{\eu^z} = \eu^x \to 0$ जबकि $\abs z \to \infty$।

\textbf{17.} \omterm{ex:b3:groups:actions}{केंद्र} पर \cref{thm:b3:holomorphic:formula} का प्राचलन कीजिए: $w = a + r\eu^{\iu\theta}$, $\dd w = \iu
r\eu^{\iu\theta}\dd\theta$ के साथ
\[
f(a) = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - a}\,\dd w
= \frac1{2\pi}\int_0^{2\pi}
f\bigl(a + r\eu^{\iu\theta}\bigr)\,\dd\theta ;
\]
और वास्तविक भाग लेने पर $u$ का माध्य मान गुण मिल जाता है। यदि
$u$ \omterm{def:b3:topology:connected}{संबद्ध} $\Omega$ के किसी आंतरिक बिंदु पर उच्चिष्ठ प्राप्त कर ले:
तो $\abs{\eu^f} = \eu^u$ आंतरिक उच्चिष्ठ प्राप्त कर लेता है, अतः \cref{thm:b3:holomorphic:maximum}(2) से $\eu^f$
अचर है, और $u =
\log\abs{\eu^f}$ भी। परिसीमा तक \omterm{def:b3:topology:continuity}{सांतत्य} वाले किसी परिबद्ध प्रांत पर
$\sup_{\bar\Omega}u =
\sup_{\partial\Omega}u$, ठीक वैसे ही जैसे $\abs f$ के लिए।

\textbf{18.} \emph{स्थिति $\abs a > r$।} $r < R
< \abs a$ वाला $R$ चुनिए: उत्तल चक्र
$D(0, R)$ पर फलन $a - z$ \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} और शून्य-रहित है, और $z \mapsto -1/(a - z)$ का वहाँ
कोई आद्यंतर $L$ है (\cref{thm:b3:holomorphic:cauchy}); अचर समायोजित करने के बाद $\bigl(\eu^{-L}(a -
z)\bigr)' = \eu^{-L}\bigl(-L'\,(a - z) - 1\bigr) = 0$ $\eu^L = a - z$
दे देता है: अर्थात् कोई \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} लघुगणक विद्यमान है, और $\log\abs{a - z} = \operatorname{Re}L(z)$।
प्रश्न 17 का माध्य मान गुण $0$ पर, त्रिज्या $r$:
\[
\frac1{2\pi}\int_0^{2\pi}
\log\bigl|a - r\eu^{\iu\theta}\bigr|\,\dd\theta
= \operatorname{Re}L(0) = \log\abs a .
\]
\emph{स्थिति $\abs a < r$।} $a - r\eu^{\iu\theta} =
-r\eu^{\iu\theta}\bigl(1 - \frac ar\eu^{-\iu\theta}\bigr)$ से माध्य $\log r$ और साथ में $\log\abs{1 -
\frac ar\eu^{-\iu\theta}}$ के माध्य
के बराबर है। प्रतिस्थापन $\theta \mapsto
2\pi - \theta$, फिर --- $\frac ar =
\rho\eu^{\iu\varphi}$, $\rho < 1$ लिखते हुए, और
$a = 0$ स्थिति तुच्छ होने के कारण --- विस्थापन $\theta \mapsto \theta - \varphi$ (दोनों किसी आवर्त
पर माध्य सुरक्षित रखते हैं) इसे $\log\abs{1 - \rho\eu^{\iu\theta}}$ के माध्य में बदल देते हैं:
अर्थात् $(a, r) = (1, \rho)$ के साथ पहली स्थिति, जो $\log 1 = 0$ देती है। कुल: दोनों
स्थितियों में $\log r = \log\max(\abs a, r)$।

\textbf{19.} $\log\abs{P(\eu^{\iu\theta})} = \log\abs c +
\sum_i\log\abs{\alpha_i - \eu^{\iu\theta}}$, जिसमें प्रत्येक मूल अपनी बहुलता के अनुसार दोहराया
गया है; $\theta$ में औसत लेकर और इकाई वृत्त से बाहर के प्रत्येक मूल पर
$r = 1$ के साथ प्रश्न 18 लगाने पर $\log\bigl(\abs c\prod_i\max(1,
\abs{\alpha_i})\bigr)$ मिलता है। जाँच। $P = 2X - 1 = 2(X -
\frac12)$ के लिए
सूत्र $\log 2$ की भविष्यवाणी करता है; और सीधे, $\abs{2\eu^{\iu\theta} - 1} = 2\abs{\frac12 -
\eu^{\iu\theta}}$ तथा $\log\abs{\frac12 -
\eu^{\iu\theta}}$ का माध्य
$\log\max(\frac12, 1) = 0$: अतः माध्य $\log
2$। $P = X^2 - X = X(X - 1)$ के लिए मूल $1$ वृत्त \emph{पर} बैठता
है; और सूत्र $0$ की भविष्यवाणी करता है। सीधे, $\log\abs{\eu^{\iu\theta}}$ का माध्य $0$
है, और $\abs{\eu^{\iu\theta} - 1} = 2\abs{\sin\frac\theta2}$ के साथ:
\[
\frac1{2\pi}\int_0^{2\pi}
\log\Bigl(2\sin\frac\theta2\Bigr)\dd\theta
= \frac1\pi\int_0^\pi\log(2\sin u)\,\dd u
= \log 2 + \frac J\pi,
\qquad J = \int_0^\pi\log\sin u\,\dd u .
\]
प्रतिस्थापन $u = 2v$ और $\sin 2v = 2\sin v\cos v$ $J = \pi\log2 + 2\int_0^{\pi/2}\log\sin +
2\int_0^{\pi/2}\log\cos = \pi\log 2 + 2J$ दे देते हैं ($\sin$ की सममितियों से
प्रत्येक आधा $J/2$ के बराबर है), अतः $J = -\pi\log 2$ (अनुचित समाकल अभिसरित होते
हैं, क्योंकि $\log\sin$ अंत-बिंदुओं पर \omterm{def:b3:lebesgue:l1}{समाकलनीय} है): अर्थात् माध्य $\log2 - \log2 = 0$।
सूत्र वृत्त पर के मूलों को झेल जाता है।

\textbf{20.} $\frac{\eu^z - 1}z =
\sum_{j\geq0}\frac{z^j}{(j+1)!}$ समग्र है और $0$ पर $1$ के बराबर: अतः उसका
व्युत्क्रम $0$ के निकट \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} है (वस्तुतः $\abs z <
2\pi$ पर, क्योंकि
$\eu^z - 1$ के निकटतम अन्य शून्य $\pm2\iu\pi$ हैं), इसलिए $\frac z{\eu^z-1}$ $0$ पर वैश्लेषिक
है। दोनों श्रेणियों को गुणा करके और $(\frac{\eu^z-1}z)\cdot(\frac
z{\eu^z-1}) = 1$ में $z^n$, $n \geq 1$, का गुणांक
पढ़ने पर:
\[
\sum_{k=0}^{n}\frac{B_k}{k!\,(n+1-k)!} = 0
\quad\Longleftrightarrow\quad
\sum_{k=0}^{n}\binom{n+1}{k}B_k = 0 .
\]
क्रमशः: $B_0 = 1$, $B_1 = -\frac12$, $B_2 = \frac16$, $B_3 = 0$, $B_4 = -\frac1{30}$, $B_5 = 0$, $B_6 =
\frac1{42}$। समता: $F(z) = \frac z{\eu^z-1} +
\frac z2$
के साथ
\[
F(-z) = \frac{-z}{\eu^{-z} - 1} - \frac z2
= \frac{z\,\eu^z}{\eu^z - 1} - \frac z2
= z + \frac{z}{\eu^z - 1} - \frac z2 = F(z) :
\]
$F$ सम है, अतः $k \geq 1$ के लिए $B_{2k+1} = 0$ (अकेला विषम गुणांक $B_1$ $+\frac z2$
ने सोख लिया था)। आगे का संकेत: $\cot w = \iu + \frac{2\iu}{\eu^{2\iu w}-1}$ $w\cot w = 1 + \sum_{k\geq1}\frac{B_{2k}}{(2k)!}(2\iu
w)^{2k}$ देता है, अतः कोटांजेंट
के लोरां गुणांक --- और इसलिए, \cref{ch:b3:residues} से, प्रत्येक $\zeta(2k)$ --- बर्नूली
संख्याओं से आँके जाते हैं:
\[
\zeta(2k) = (-1)^{k+1}\,\frac{(2\pi)^{2k}B_{2k}}{2\,(2k)!} .
\]

\textbf{21.} $f = \sum_nc_nz^n$ लिखिए (त्रिज्या $\infty$); तब $g(z) = \overline{f(\bar z)} = \sum_n\bar c_nz^n$ समग्र है। $\R$
पर: $g(x) = \overline{f(x)} = f(x)$, अतः $g$ और $f$ \omterm{def:b3:topology:connected}{संबद्ध} $\C$ में संचय बिंदुओं वाले किसी
समुच्चय पर सहमत हैं: \cref{thm:b3:holomorphic:identity} $g \equiv f$ दे देती है, अर्थात् $f(\bar z) = \overline{f(z)}$ (समतुल्य
रूप से: सभी $c_n$ वास्तविक हैं)। किसी वास्तविक बहुपद $P$ के लिए:
$P(\bar\alpha) = \overline{P(\alpha)} = 0$, और वही सर्वसमिका वास्तविक अवकलजों $P', P'', \dots$ पर लगाने से बहुलताएँ
सुरक्षित रहती हैं: अतः अवास्तविक मूल युग्मबद्ध हो जाते हैं ---
प्रश्न 8 का युग्मन, वैश्लेषिक रूप से पुनः सिद्ध।

\textbf{22.} जोड़ी गई सीढ़ी: परिबद्ध $\Rightarrow$ अचर (11); $A + B\abs z^m$ से प्रभावित
$\Rightarrow$ बहुपद (12); ऊपर से परिबद्ध वास्तविक भाग $\Rightarrow$ अचर (13);
द्विक-आवर्ती $\Rightarrow$ अचर (14); कोई चक्र छोड़ता परास $\Rightarrow$ अचर (15);
यथोचित $\Rightarrow$ बहुपद (16)। प्रत्येक पायदान कोशी का सूत्र है: किसी
बिंदु पर मान एक वृत्तीय औसत है, अतः सभी टेलर गुणांक बड़े वृत्तों पर
$f$ के आकार से आँके जाते हैं, और कोई वृद्धि-सीमा गुणांकों को थोक
में मार देती है। वास्तविक $\mathcal
C^\infty$ फलनों पर ऐसा कुछ भी बंधन नहीं है:
कोई \omterm{def:b3:lp:mollifier}{उभार फलन} (\cref{thm:b3:lp:regularization}) परिबद्ध, \omterm{def:b3:topology:compact}{संहत} आलंबित और उन्मुक्त रूप से अचरेतर
होता है, और आलंब के बाहर किसी भी बिंदु पर उसके सभी अवकलज लुप्त हो
जाते हैं जबकि फलन आसपास कहीं भी लुप्त नहीं होता। चिकनाई भिन्न बिंदुओं
के अवकलजों को बिलकुल नहीं जोड़ती; और \omterm{def:b3:holomorphic:holo}{समविश्लेषिकता} प्रत्येक अवकलज
को किसी दूरस्थ वृत्त पर के एक ही समाकल से बाँध देती है। एक स्थानीय
शर्त और एक वैश्विक मुखबिर --- इसीलिए समग्र फलन नियम-क़ानून मानते
हैं।

\textbf{23.} $\abs{P(\eu^{\iu\theta})}^2 = \sum_{k,l}a_k\bar a_l
\eu^{\iu(k-l)\theta}$ का प्रसार करके औसत लेने पर प्रत्येक पद $k \neq
l$ मर
जाता है: अतः माध्य $\sum_k\abs{a_k}^2 =: N$ है। पहले मान लीजिए कि $P$ का इकाई वृत्त
पर कोई शून्य नहीं है, अतः $\theta \mapsto
\log\abs{P(\eu^{\iu\theta})}$ \omterm{def:b3:topology:continuity}{संतत} है। परिबंध $\log
t \leq t - 1$ को $t = \abs P^2/N$ पर
लगाकर औसत लेने पर
\[
\frac1{2\pi}\int_0^{2\pi}\log\abs{P(\eu^{\iu\theta})}^2
\dd\theta - \log N
\;\leq\; \frac1N\cdot N - 1 = 0,
\]
अतः $\abs P$ का गुणोत्तर माध्य अधिक से अधिक $\sqrt N$ है; और प्रश्न 19 उस
गुणोत्तर माध्य की पहचान $\abs c\prod_i\max(1, \abs{\alpha_i})$ से कर देता है: यही लांडाउ की असमिका
है। वृत्त पर के मूल: प्रत्येक $\abs{\alpha_i}$ से भिन्न $r > 1$ चुनिए; बहुपद
$P(rX)$, जिसके मूल $\alpha_i/r$ इकाई वृत्त से बाहर हैं और गुणांक $a_kr^k$, असमिका
संतुष्ट करता है; और दोनों पक्ष $r$ में \omterm{def:b3:topology:continuity}{संतत} हैं, अतः $r \to 1^+$ लेने पर
व्यापक स्थिति मिल जाती है। $X^2 -
X$ पर: मूल $0$ और $1$, अतः बायाँ
पक्ष $1$ है, और दायाँ पक्ष $\sqrt{1 + 1} = \sqrt2$: सत्य, खुली जगह के साथ।

\textbf{24.} समग्र $g(z) =
\sum_{k\geq0}\frac{z^k}{(k+1)!}$, $g(0) = 1$ के साथ $\eu^z - 1 = z\,g(z)$ लिखिए। चूँकि ठीक $2\pi\iu\Z$
पर $\eu^z = 1$, अतः $g$ का $D(0, 2\pi)$ में कोई शून्य नहीं है ($0 < \abs z < 2\pi$ के लिए
क्योंकि $\eu^z - 1
\neq 0$, और $0$ पर $g(0) = 1$ से), इसलिए $h = 1/g$ $D(0,2\pi)$ पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक}
है और $0$ पर उसकी टेलर श्रेणी --- जो परिभाषा से $\sum\frac{B_n}{n!}z^n$ है --- समूचे
चक्र पर अभिसरित होती है: $\rho \geq 2\pi$। यदि $\rho > 2\pi$ होता, तो योग $S$ $D(0,\rho)$
पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} होता, और वह $0 < \abs z < 2\pi$ पर $z
\mapsto z/(\eu^z - 1)$ से सहमत है; दोनों \omterm{def:b3:topology:connected}{संबद्ध}
\omterm{def:b3:topology:topology}{विवृत समुच्चय} $D(0,\rho) \setminus
2\pi\iu\Z$ पर \omterm{def:b3:holomorphic:holo}{समविश्लेषिक} हैं, अतः तत्समता प्रमेय से वे वहाँ
सहमत होते। पर $z \to 2\pi\iu$ पर $\abs{z/(\eu^z - 1)} \to \infty$ (अंश $\to 2\pi$, हर $\to 0$) जबकि $S$ $2\pi\iu$
पर \omterm{def:b3:topology:continuity}{संतत} है: विरोधाभास। अतः ठीक $\rho = 2\pi$, और आदमार का सूत्र $\limsup_n\abs{B_n/n!}^{1/n} = \frac1{2\pi}$ दे देता
है; और $n = 3$ से आगे विषम गुणांक शून्य होने के कारण $\limsup$ सम \omterm{def:b3:holomorphic:index}{सूचकांक}
ढोते हैं, जो $n = 2k$ के साथ कहा गया सूत्र है। $k = 6$ पर संख्यात्मक रूप
से: $(2\pi)^{12} \approx
3.7858\cdot10^9$ और $2\cdot12! = 958\,003\,200$, अतः $2\,(2k)!/(2\pi)^{2k} \approx 0.25305$, जबकि $\abs{B_{12}} = \frac{691}{2730} \approx 0.25311$। अनुपात $1.00025$ दिखाए गए
अंकों तक ठीक $\zeta(12)$ है: \cref{ch:b3:residues} का अवशेष-कलन सूत्र $\zeta(2k) =
(-1)^{k+1}\frac{(2\pi)^{2k}B_{2k}}{2\,(2k)!}$ गुणक $2$ और
उस नन्ही अधिकता दोनों को समझा देता है।

\textbf{25.} कोशी परिबंध: यदि $M =
\max_k\abs{q_k}$ के साथ $\abs z > 1 + M$ हो, तो
\[
\Bigl|\sum_{k<n}q_kz^k\Bigr|
\leq M\,\frac{\abs z^n - 1}{\abs z - 1}
< \frac{M}{\abs z - 1}\,\abs z^n \leq \abs z^n,
\]
अतः $\abs{Q(z)} > 0$: इसलिए सभी मूल $\overline D(0, 1+M)$ में हैं। $P_j$ के गुणांक अभिसरित होते
हैं, अतः किसी $M$ से परिबद्ध हैं: इसलिए सभी $P_j$ के (और $P$ के)
सभी मूल \omterm{def:b3:topology:compact}{संहत} $K = \overline D(0, 1 + M)$ में हैं। मान लीजिए $v_j \in K^n$ $P_j$ के मूलों को बहुलता
सहित सूचीबद्ध करने वाला सदिश है (प्रश्न 7)। $(v_j)$ के प्रत्येक उपानुक्रम
का कोई आगे का उपानुक्रम किसी $(\beta_1, \dots, \beta_n)$ पर अभिसरित होता है; और $\prod_i(X - \alpha_i^{(j)})$ के
गुणांक चिह्न तक $v_j$ के प्रारंभिक सममित फलन हैं --- जो \omterm{def:b3:topology:continuity}{संतत} हैं
--- अतः उस उपानुक्रम के अनुदिश वे $\prod_i(X - \beta_i)$ के गुणांकों पर अभिसरित होते
हैं; पर परिकल्पना से वे $P$ के गुणांकों पर अभिसरित होते हैं, इसलिए
$\prod_i(X -
\beta_i) = P$: अर्थात् $(v_j)$ की प्रत्येक उपानुक्रमिक सीमा $P$ के मूल सदिश
का कोई क्रमचय है। यदि मिलान दूरी $\delta_j = \min_\sigma\max_i\,
\abs{\alpha_i^{(j)} - \alpha_{\sigma(i)}}$ $0$ की ओर न जाती, तो कोई
उपानुक्रम $\delta_j \geq \varepsilon$ बनाए रखता जबकि उसके मूल सदिश $P$ के मूलों के किसी
क्रमचय पर अभिसरित होते --- जिससे उसके अनुदिश $\delta_j \to 0$ अनिवार्य हो जाता:
विरोधाभास। अतः मूल बहुसमुच्चय अभिसरित होते हैं। तीक्ष्णता: $P_\varepsilon = (X - 1)^2 + \varepsilon$ के
मूल $1 \pm
\iu\sqrt\varepsilon$ हैं: द्विक मूल $\sqrt\varepsilon$ भर हट जाता है, उदाहरणार्थ $\varepsilon =
10^{-4}$ के
लिए $10^{-2}$ भर। व्यापक रूप से, यदि $\alpha$ कोई $m$-गुणा मूल हो, तो
$\alpha$ के निकट $\abs{P(z)} \asymp \abs{z -
\alpha}^m$, अतः आकार-$\varepsilon$ का कोई विक्षोभ मूलों के गुच्छ को
लगभग $\varepsilon^{1/m}$ भर विस्थापित कर देता है: अर्थात् घातांक $\frac1m$ की होल्डर
\omterm{def:b3:topology:continuity}{सांतत्य}, और इससे बेहतर नहीं --- और इसीलिए किसी द्विक मूल के निकट कोई
संख्यात्मक हलकर्ता काम के आधे ही अंक बचा पाता है।
\end{solution}
