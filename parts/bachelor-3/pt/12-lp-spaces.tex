\chapter{Os espaços \texorpdfstring{$L^p$}{Lp}}\label{ch:b3:lp}

A integral de Lebesgue foi construída para a análise; os espaços
$L^p$ são onde essa análise vive. Eles são espaços de Banach
(Riesz--Fischer) --- os \omterm{thm:b3:complete:completion}{completamentos} que o~\cref{exo:b3:complete:1} mostrou
faltarem às funções \omterm{def:b3:topology:continuity}{contínuas} --- e sustentam uma tecnologia de
suavização, a \emph{convolução com aproximações da identidade}, que
aproxima qualquer função $L^p$ por funções $\mathcal C^\infty$. Este capítulo
demonstra as versões integrais de Hölder e Minkowski, a \omterm{def:b3:complete:complete}{completude}, os
teoremas de \omterm{ex:b3:lebesgue:gamma}{densidade} e a máquina de regularização, encerrando com a
geografia de inclusões e interpolações da escala $L^p$. Ao longo de
todo o capítulo, $(X, \mathcal A, \mu)$ é um \omterm{def:b3:measure:measure}{espaço de medida} e as funções tomam valores
complexos; em $\R^d$, a \omterm{def:b3:measure:measure}{medida} é $\lambda_d$.

\section{Definição; Hölder e Minkowski}

\begin{definition}\label{def:b3:lp:space}
Para $1 \leq p < \infty$, $\mathcal L^p(\mu)$ é o conjunto das \omterm{def:b3:lebesgue:measurable}{funções mensuráveis} $f$ com $\norm f_p = \bigl(\int\abs
f^p\dd\mu\bigr)^{1/p} < \infty$, e
$\mathcal L^\infty(\mu)$ é o conjunto das $f$ limitadas fora de um conjunto nulo, com
$\norm
f_\infty$ o \emph{supremo essencial} --- o menor $M$ com $\abs f \leq M$ q.t.p.\ (o
ínfimo é atingido: intersecte os conjuntos nulos para $M + \frac1n$). Como $\norm f_p = 0$
apenas força $f
= 0$ \emph{q.t.p.} (\cref{exo:b3:lebesgue:5}), definimos
\[
L^p(\mu) = \mathcal L^p(\mu)/\{f = 0 \text{ q.t.p.}\} :
\]
seus elementos são classes de funções módulo conjuntos nulos, e $\norm\cdot_p$ é
uma norma genuína em $L^p$.\index{espaço Lp@espaço $L^p$}
\end{definition}

\begin{theorem}[Desigualdade de Hölder]\label{thm:b3:lp:holder}
Sejam $1 \leq p, q \leq \infty$ com $\frac1p + \frac1q = 1$ (expoentes conjugados). Para $f, g$
\omterm{def:b3:lebesgue:measurable}{mensuráveis}:\index{desigualdade de Hölder}
\[
\norm{fg}_1 \leq \norm f_p\,\norm g_q ,
\]
com igualdade (para $1 < p < \infty$, normas finitas, $f,g\ne0$) se, e somente se, $\abs f^p$ e
$\abs g^q$ são proporcionais q.t.p.
\end{theorem}

\begin{proof}
Os casos $\{p, q\} = \{1, \infty\}$ são diretos ($\abs{fg} \leq
\norm g_\infty\abs f$ q.t.p.). Seja $1 < p < \infty$; normalize $\norm f_p = \norm g_q = 1$
(homogeneidade; normas nulas ou infinitas são triviais). A desigualdade
de Young $ab \leq \frac{a^p}p +
\frac{b^q}q$ ($a, b \geq 0$; concavidade de $\ln$, como no~\cref{pb:b3:banach:1}) dá
pontualmente $\abs{f g} \leq
\frac{\abs f^p}p + \frac{\abs g^q}q$; integre: $\norm{fg}_1
\leq \frac1p + \frac1q = 1$. A igualdade força a igualdade q.t.p.\
em Young, isto é, $\abs f^p = \abs g^q$ q.t.p.\ (após a normalização; desfazendo-a,
proporcionalidade).
\end{proof}

\begin{theorem}[Desigualdade de Minkowski]\label{thm:b3:lp:minkowski}
Para $1 \leq p \leq \infty$: $\norm{f + g}_p \leq \norm f_p +
\norm g_p$.\index{desigualdade de Minkowski}
\end{theorem}

\begin{proof}
$p = 1, \infty$: desigualdade triangular pontual ou q.t.p. Para $1 <
p < \infty$, suponha
$\norm{f+g}_p < \infty$ (do contrário, use $\abs{f+g}^p \leq 2^{p-1}(\abs f^p + \abs g^p)$, que vem da convexidade de $t^p$, para
ver que o membro esquerdo é finito quando o direito o é). Então
\[
\norm{f{+}g}_p^p \leq \int\abs f\,\abs{f{+}g}^{p-1} +
\int\abs g\,\abs{f{+}g}^{p-1}
\leq \bigl(\norm f_p + \norm g_p\bigr)\,
\bigl\|\abs{f{+}g}^{p-1}\bigr\|_q
\]
por Hölder, e $\norm{\abs{f+g}^{p-1}}_q =
\norm{f+g}_p^{p/q}$ pois $(p-1)q = p$; divida por $\norm{f+g}_p^{p/q}$ (se não for nulo; caso
contrário é trivial) e use $p -
\frac pq = 1$.
\end{proof}

\section{Completude e suas companheiras}

\begin{theorem}[Riesz--Fischer]\label{thm:b3:lp:complete}
Para $1 \leq p \leq \infty$, $L^p(\mu)$ é um espaço de Banach. Além disso, toda sequência
convergente em $L^p$ tem uma subsequência que converge \emph{quase em
toda parte} (com um dominador $L^p$ no caso $p < \infty$).\index{teorema de
Riesz--Fischer}
\end{theorem}

\begin{proof}
$p = \infty$: uma sequência de Cauchy para $\norm\cdot_\infty$ é, fora de um único conjunto
nulo (reunião de enumeráveis outros), uniformemente de Cauchy: ela
converge uniformemente ali fora; pronto. Seja $p < \infty$.
Pelo~\cref{exo:b3:complete:1}(b), basta somar séries absolutamente convergentes:
seja $\sum\norm{f_k}_p = M < \infty$. Ponha $G_n = \sum_{k \leq n}\abs{f_k}$ e $G = \sum_k\abs{f_k}$ (pontualmente em $[0,\infty]$): por Minkowski
$\norm{G_n}_p \leq
M$, e a convergência monótona ($G_n^p \nearrow G^p$) dá $\int G^p \leq M^p$: $G < \infty$ q.t.p., de modo que
a série $\sum f_k(x)$ converge absolutamente para q.t.p.\ $x$; chame a soma de
$S(x)$ (com valor qualquer no conjunto nulo). Então $\abs{S - \sum_{k\leq n}f_k}^p \leq (2G)^p
\in L^1$, e a
convergência dominada dá $\norm{S - \sum_{k\leq n}f_k}_p \to 0$: a série converge em $L^p$.

Quanto ao enunciado sobre subsequências: se $f_n \to f$ em $L^p$, escolha
$n_k$ com $\norm{f_{n_{k+1}} - f_{n_k}}_p \leq 2^{-k}$; a série $\sum(f_{n_{k+1}} - f_{n_k})$ cai no argumento anterior: ela converge
absolutamente q.t.p., dominada por um $G \in
L^p$, de modo que $f_{n_k} \to f_{n_1} + \sum(\cdots) $ q.t.p., e
esse limite q.t.p.\ tem de ser (um representante de) $f$ (ambos são
limites $L^p$). O dominador: $\abs{f_{n_k}} \leq \abs{f_{n_1}} + G$.
\end{proof}

\begin{remark}\label{rem:b3:lp:modes}
A convergência em $L^p$ não implica a convergência q.t.p.\ (a
sequência \emph{máquina de escrever}, \cref{exo:b3:lp:3}), nem
reciprocamente (ondas que fogem): os dois modos só se ligam por
subsequências e dominação. Ter em mente os contraexemplos
do~\cref{exo:b3:lp:3} é a melhor vacina.
\end{remark}

\section{Teoremas de densidade}

\begin{theorem}\label{thm:b3:lp:density}
Seja $1 \leq p < \infty$.
\begin{enumerate}
  \item As \omterm{def:b3:lebesgue:simple}{funções simples} (com suportes de \omterm{def:b3:measure:measure}{medida} finita) são densas
        em $L^p(\mu)$.
  \item Em $L^p(\R^d)$, as funções \omterm{def:b3:topology:continuity}{contínuas} de suporte \omterm{def:b3:topology:compact}{compacto} $\mathcal C_c(\R^d)$ são
        densas.
  \item A translação é \omterm{def:b3:topology:continuity}{contínua} em $L^p(\R^d)$: escrevendo $\tau_hf = f(\cdot - h)$, $\norm{\tau_hf - f}_p \to 0$ quando
        $h \to 0$.
\end{enumerate}
Nenhuma das três vale para $p = \infty$.
\end{theorem}

\begin{proof}
(1) Para $f \geq 0$: as $s_n \nearrow f$ diádicas do~\cref{thm:b3:lebesgue:approximation} satisfazem $\abs{f - s_n}^p
\leq f^p \in L^1$:
convergência dominada. (Cada $s_n \leq f$ está em $L^p$, e seus conjuntos
de nível têm \omterm{def:b3:measure:measure}{medida} finita onde o valor é positivo: $\mu(s_n \geq c) \leq c^{-p}\int f^p$.) Separe uma
$f$ geral em quatro partes não negativas.

(2) Por (1), basta aproximar $\mathbf 1_A$, com $A$ boreliano e $\lambda_d(A) < \infty$. A
regularidade (demonstração como no~\cref{thm:b3:measure:regularity}) dá $K \subseteq A
\subseteq U$ \omterm{def:b3:topology:compact}{compacto} e
aberto com $\lambda_d(U\setminus K) <
\varepsilon$; a função de Urysohn
\[
\varphi(x) = \frac{d(x, \R^d\setminus U)}{d(x, \R^d\setminus U)
+ d(x, K)}
\]
é \omterm{def:b3:topology:continuity}{contínua}, vale $1$ em $K$ e $0$ fora de $U$, e pode ser
tomada de suporte \omterm{def:b3:topology:compact}{compacto} (encolha primeiro $U$ a um aberto
limitado). Então $\norm{\mathbf 1_A - \varphi}_p^p \leq \lambda_d(U\setminus K)
< \varepsilon$.

(3) Para $g \in \mathcal C_c$: a \omterm{def:b3:topology:continuity}{continuidade} uniforme dá $\norm{\tau_hg - g}_\infty \to 0$, com suportes em um
\omterm{def:b3:topology:compact}{compacto} fixo para $\abs h \leq 1$: $\norm{\tau_hg - g}_p \to 0$. Para $f$ geral: tome $g \in \mathcal C_c$ com
$\norm{f - g}_p <
\varepsilon$; então $\norm{\tau_hf - f}_p \leq 2\norm{f - g}_p +
\norm{\tau_hg - g}_p$ (invariância da norma por translação).

Para $p = \infty$: a aproximação uniforme de $\mathbf
1_{\intoo0\infty}$ por funções \omterm{def:b3:topology:continuity}{contínuas} é
impossível (o salto), e $\norm{\tau_h\mathbf 1_{\intoo0\infty} - \mathbf
1_{\intoo0\infty}}_\infty = 1$ para $h \neq 0$.
\end{proof}

\section{Convolução e regularização}

\begin{theorem}[Desigualdade de Young]\label{thm:b3:lp:young}
Sejam $1 \leq p \leq \infty$, $f \in L^1(\R^d)$, $g \in
L^p(\R^d)$. Então $f * g$ está definida q.t.p.,
pertence a $L^p$ e\index{desigualdade de Young}
\[
\norm{f * g}_p \leq \norm f_1\,\norm g_p .
\]
\end{theorem}

\begin{proof}
$p = \infty$: cota direta. $p = 1$: \cref{thm:b3:product:convolution}. Sejam $1 < p < \infty$ e $q$
conjugado. Separe $\abs{f(y)} = \abs{f(y)}^{1/q}\cdot
\abs{f(y)}^{1/p}$ e aplique Hölder:
\[
\int\abs{f(y)}\,\abs{g(x{-}y)}\,\dd y
\leq \Bigl(\int\abs f\Bigr)^{1/q}
\Bigl(\int\abs{f(y)}\,\abs{g(x - y)}^p\,\dd y\Bigr)^{1/p} .
\]
Eleve à $p$-ésima potência e integre em $x$; Tonelli no segundo
fator dá $\norm f_1^{p/q}\cdot\norm f_1\norm g_p^p$, isto é, $\norm{f*g}_p^p \leq \norm f_1^{1 + p/q}\norm g_p^p =
(\norm f_1\norm g_p)^p$ --- e a finitude da integral de Tonelli
justifica a convergência absoluta q.t.p., como no~\cref{thm:b3:product:convolution}.
\end{proof}

\begin{definition}[Aproximações da identidade]\label{def:b3:lp:mollifier}
A função\index{aproximação da identidade}\index{função de corte}
\[
\rho(x) = \begin{cases}
c\,\exp\Bigl(-\dfrac{1}{1 - \norm x^2}\Bigr) & \norm x < 1,\\
0 & \norm x \geq 1,
\end{cases}
\]
com $c$ normalizando $\int\rho = 1$, é $\mathcal C^\infty$ em $\R^d$: o ponto é que
$t \mapsto \eu^{-1/t}\mathbf
1_{t>0}$ é $\mathcal C^\infty$ em $\R$, com todas as derivadas em $0^+$ iguais a
$0$ (cada derivada é $P(1/t)\eu^{-1/t}$ para um polinômio $P$, e isso tende a
$0$; indução). Para $\varepsilon > 0$, ponha $\rho_\varepsilon(x) =
\varepsilon^{-d}\rho(x/\varepsilon)$: com suporte em $\bar B(0,
\varepsilon)$ e ainda
de integral $1$.
\end{definition}

\begin{theorem}[Regularização]\label{thm:b3:lp:regularization}
Sejam $1 \leq p < \infty$ e $f \in L^p(\R^d)$. Então:
\begin{enumerate}
  \item $f * \rho_\varepsilon \in \mathcal C^\infty(\R^d)$, com $\partial^\alpha(f * \rho_\varepsilon) = f *
        \partial^\alpha\rho_\varepsilon$;
  \item $\norm{f * \rho_\varepsilon - f}_p \to 0$ quando $\varepsilon \to 0$;
  \item consequentemente, $\mathcal C^\infty_c(\R^d)$ é denso em $L^p(\R^d)$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Derivação sob o sinal de integral (\cref{thm:b3:lebesgue:paramdiff}) em $x$: para
$x$ em uma bola $B$, $\abs{\partial_{x_i}\rho_\varepsilon(x - y)} \leq
C_\varepsilon\,\mathbf 1_{K}(y)$ com $K$ \omterm{def:b3:topology:compact}{compacto} ($y$ a
distância no máximo $\varepsilon$ de $B$), e $\abs f\,\mathbf 1_K \in L^1$ (Hölder contra $\mathbf 1_K$): o
teorema se aplica; itere para as derivadas de ordem superior.

(2) Como $\int\rho_\varepsilon = 1$:
\[
(f * \rho_\varepsilon)(x) - f(x)
= \int \bigl(f(x - y) - f(x)\bigr)\rho_\varepsilon(y)\,\dd y ,
\]
e a desigualdade \emph{integral} de Minkowski --- ou, diretamente,
Hölder/Jensen com a \omterm{def:b3:measure:measure}{medida} de probabilidade $\rho_\varepsilon\dd y$ e Tonelli ---
\[
\norm{f*\rho_\varepsilon - f}_p^p
\leq \int\Bigl(\int\abs{f(x-y) -
f(x)}^p\dd x\Bigr)\rho_\varepsilon(y)\,\dd y
= \int \norm{\tau_yf - f}_p^p\;\rho_\varepsilon(y)\,\dd y
\]
(a etapa do meio: aplique a desigualdade de Jensen, \cref{exo:b3:lp:10}, à
integral interna em $y$ e, depois, Tonelli). O integrando tem
suporte em $\norm y \leq \varepsilon$ e tende ali a $0$ uniformemente quando $\varepsilon \to 0$
(\cref{thm:b3:lp:density}(3)): a expressão inteira tende a $0$.

(3) Aproxime $f$ por $g \in \mathcal C_c$ (\cref{thm:b3:lp:density}(2)) e, depois, $g$
por $g *
\rho_\varepsilon \in \mathcal C_c^\infty$ (suporte \omterm{def:b3:topology:compact}{compacto}: soma dos suportes).
\end{proof}

\begin{example}[Regularizando $\abs x$, com taxas]
\label{ex:b3:lp:mollifyexample}
Tome $f(x) = \abs x$ em $\R$ (localmente $L^1$; o teorema se aplica em
toda janela limitada) e uma \omterm{def:b3:lp:mollifier}{aproximação da identidade} simétrica $\rho_\varepsilon$.
Então
\[
f_\varepsilon(x) = (f * \rho_\varepsilon)(x)
= \int\abs{x - y}\,\rho_\varepsilon(y)\,\dd y
\]
é $\mathcal C^\infty$; longe do bico, nada acontece: para $\abs x \geq \varepsilon$, $\abs{x - y}$ é linear em
$x$ no suporte de $\rho_\varepsilon$, de modo que $f_\varepsilon(x) =
\abs x$ \emph{exatamente} (a
simetria mata a correção). Perto de $0$, a suavização custa
precisamente
\[
0 \leq f_\varepsilon(0) = \int\abs
y\,\rho_\varepsilon(y)\,\dd y \leq \varepsilon,
\qquad
\norm{f_\varepsilon - f}_\infty \leq \varepsilon :
\]
o erro de aproximação fica confinado à \omterm{def:b3:topology:topology}{vizinhança} de raio $\varepsilon$ da
singularidade e é da ordem dela. Enquanto isso, $f_\varepsilon'' \geq 0$ em toda parte
($f$ é convexa, e a convolução com $\rho_\varepsilon
\geq 0$ preserva a convexidade),
com $\int f_\varepsilon'' =
f_\varepsilon'(\infty) - f_\varepsilon'(-\infty) = 2$: a segunda derivada é uma onda de massa $2$ espremida em
largura $O(\varepsilon)$, de modo que $\norm{f_\varepsilon''}_\infty \gtrsim
\varepsilon^{-1}$. Suavizar é uma troca: erro uniforme
$O(\varepsilon)$ contra explosão da derivada $O(\varepsilon^{-1})$ --- exatamente a taxa de câmbio
que a análise quantitativa (desigualdades de interpolação, o círculo de
ideias do~\cref{pb:b3:lp:1}) formaliza.
\end{example}

\begin{corollary}[Lema fundamental do cálculo das
variações]\label{cor:b3:lp:fundlemma}
Seja $f \in L^1_{\mathrm{loc}}(\R^d)$ (\omterm{def:b3:lebesgue:l1}{integrável} nos \omterm{def:b3:topology:compact}{compactos}) com $\int f\varphi = 0$ para toda $\varphi \in \mathcal
C^\infty_c(\R^d)$. Então
$f = 0$ q.t.p.\index{lema fundamental do cálculo das variações}
\end{corollary}

\begin{proof}
Fixe uma bola $B = B(0, R)$ e ponha $g = f\mathbf 1_{B(0, R+1)}
\in L^1$. Para $x \in B$ e $\varepsilon < 1$: $(g *
\rho_\varepsilon)(x) = \int f(y)\rho_\varepsilon(x - y)\dd y =
0$, sendo a
função de teste $y \mapsto \rho_\varepsilon(x-y)
\in \mathcal C_c^\infty$. Mas $g * \rho_\varepsilon \to g$ em $L^1$ (\cref{thm:b3:lp:regularization}):
$g = 0$ q.t.p.\ em $B$; exaura $\R^d$.
\end{proof}

\section{A geografia dos \texorpdfstring{$L^p$}{Lp}}

\begin{proposition}\label{prop:b3:lp:inclusions}
(a) Se $\mu(X) < \infty$ e $1 \leq p \leq q \leq \infty$, então $L^q \subseteq L^p$ com $\norm f_p \leq
\mu(X)^{\frac1p - \frac1q}\,\norm f_q$.
(b) Em $\R^d$ (\omterm{def:b3:measure:measure}{medida} infinita) não há inclusão alguma: para $p \neq q$
existem funções em $L^p\setminus L^q$.
(c) (Interpolação) Se $p < r < q$ e $\alpha \in \intoo01$ é definido por $\frac1r = \frac\alpha p + \frac{1 - \alpha}q$, então
\[
\norm f_r \leq \norm f_p^{\alpha}\,\norm f_q^{1 - \alpha} ;
\]
em particular, $L^p \cap L^q \subseteq L^r$.
\end{proposition}

\begin{proof}
(a) Hölder com os expoentes $\frac qp$ e seu conjugado: $\int\abs f^p\cdot 1 \leq \norm{\abs f^p}_{q/p}\,\norm
1_{(q/p)'} = \norm f_q^p\,\mu(X)^{1 - p/q}$ ($q = \infty$
diretamente). (b) Perto de $0$ e perto de $\infty$, as potências
$x^{-\alpha}$ calibram: \cref{exo:b3:lp:2}. (c) Escreva $\abs f^r = \abs
f^{r\alpha}\abs f^{r(1-\alpha)}$ e aplique Hölder com o
par conjugado $\frac p{r\alpha}$, $\frac q{r(1-\alpha)}$ (conjugados precisamente pela definição de
$\alpha$): $\int\abs f^r \leq \norm f_p^{r\alpha}\norm f_q^{r(1 -
\alpha)}$.
\end{proof}

\begin{method}\label{met:b3:lp:toolkit}
A caixa de ferramentas $L^p$, tal como usada em tudo o que segue:
para demonstrar uma identidade ou desigualdade para todo $f \in L^p$ ---
demonstre-a em uma classe densa ($\mathcal C_c^\infty$, via o~\cref{thm:b3:lp:regularization}) e estenda
por \omterm{def:b3:topology:continuity}{continuidade} (\cref{thm:b3:complete:extension}, sendo ambos os lados \omterm{def:b3:topology:continuity}{contínuos} em
$L^p$); para demonstrar que $f = 0$, teste contra $\mathcal
C_c^\infty$
(\cref{cor:b3:lp:fundlemma}); para ganhar regularidade, convolua; para trocar
expoentes, Hölder e interpolação. A teoria de Fourier do~\cref{ch:b3:fouriertransform} é uma longa aplicação desse método.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:lp:1}
(a) Enuncie e demonstre a desigualdade de Cauchy--Schwarz em $L^2(\mu)$ como
o caso $p = q = 2$ de Hölder.
(b) Em um espaço de \emph{probabilidade}, mostre que $p \mapsto \norm f_p$ é não
decrescente.
(c) Quando Hölder é uma igualdade para $p = 1$, $q = \infty$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:lp:2}
Para quais $p \in \intco1\infty$ as funções seguintes pertencem a $L^p$?
\[
x^{-1/2}\mathbf 1_{\intoo01},\qquad
x^{-1/2}\mathbf 1_{\intoo1\infty},\qquad
\frac{1}{x^{1/2}(1 + \abs{\ln x})}\ \text{em } \intoo01,
\qquad
\frac1{1 + \abs x}\ \text{em } \R .
\]
Conclua: em $\intoo01$, $p$ pequeno é mais fácil; em $\intoo1\infty$, $p$ grande
é mais fácil; e nenhum $L^p$ contém outro em $\R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:3}
(A máquina de escrever) Enumere os intervalos diádicos $I_1 =
\intcc01$, $I_2 = \intcc0{\frac12}$,
$I_3 = \intcc{\frac12}1$, $I_4 = \intcc0{\frac14}$, \dots\ e ponha $f_n = \mathbf
1_{I_n}$.
(a) Mostre que $f_n \to 0$ em todo $L^p(\intcc01)$, $p < \infty$, mas que $(f_n(x))$ diverge para
\emph{todo} $x \in \intcc01$.
(b) Exiba a subsequência convergente q.t.p.\ prometida
pelo~\cref{thm:b3:lp:complete}.
(c) Reciprocamente, dê uma sequência que converge q.t.p.\ mas não em
$L^1$, e outra que converge em $L^1$ mas em nenhum $L^p$ com
$p > 1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:4}
Sejam $\mu(X) < \infty$ e $f \in L^\infty(\mu)$, $f \neq 0$. Mostre que $\norm f_p \to \norm f_\infty$ quando $p \to \infty$. \emph{(Cota
superior por (a) da~\cref{prop:b3:lp:inclusions}; cota inferior integrando em $\{\abs f > \norm f_\infty -
\varepsilon\}$,
de \omterm{def:b3:measure:measure}{medida} positiva.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:5}
(a) Onde exatamente a demonstração do~\cref{thm:b3:lp:density}(3) usa $p < \infty$?
(b) Mostre que $f \in L^\infty(\R)$ satisfaz $\norm{\tau_hf -
f}_\infty \to 0$ se, e somente se, $f$ tem um
representante uniformemente \omterm{def:b3:topology:continuity}{contínuo}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:6}
Sejam $p, q$ conjugados, $f \in L^p(\R^d)$, $g \in
L^q(\R^d)$. Mostre que $f * g$ está
definida \emph{em toda parte}, é limitada, com $\norm{f*g}_\infty \leq \norm f_p\norm g_q$, e
\emph{uniformemente \omterm{def:b3:topology:continuity}{contínua}}. \emph{(\omterm{def:b3:topology:continuity}{Continuidade} da translação em
$L^p$; trate $p \in \{1, \infty\}$ à parte --- para $p =
\infty$, use a \omterm{def:b3:topology:continuity}{continuidade} da
translação no fator $L^1$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:7}
Seja $f \in L^1_{\mathrm{loc}}(\intoo ab)$ com $\int f\varphi' = 0$ para toda $\varphi \in \mathcal
C^\infty_c(\intoo ab)$. Mostre que $f$ é q.t.p.\ igual
a uma constante. \emph{(Fixe $\chi \in \mathcal C_c^\infty$ com $\int\chi = 1$; toda $\psi \in \mathcal C_c^\infty$ com $\int\psi = 0$ é uma
$\varphi'$; escreva uma função de teste geral como $\psi + (\int\psi)\chi$ e aplique
o~\cref{cor:b3:lp:fundlemma} a $f - c$ com $c = \int
f\chi$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:lp:8}
(Urysohn suave) Sejam $K \subseteq U \subseteq \R^d$, com $K$ \omterm{def:b3:topology:compact}{compacto} e $U$ aberto.
Construa $\varphi \in \mathcal
C^\infty_c(\R^d)$ com $0 \leq \varphi \leq 1$, $\varphi = 1$ em $K$, $\operatorname{supp}\varphi \subseteq U$. \emph{(Regularize a
indicadora da \omterm{def:b3:topology:topology}{vizinhança} de raio $\delta$, $K_\delta$, de $K$ com
$\rho_{\delta/2}$, para $\delta$ pequeno.)} Deduza um enunciado de partição da
unidade $\mathcal C^\infty$ para um \omterm{def:b3:topology:compact}{compacto} coberto por finitos abertos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:9}
Usando a interpolação (\cref{prop:b3:lp:inclusions}(c)):
(a) mostre que $L^1(\R)\cap L^\infty(\R) \subseteq L^p(\R)$ para todo $p$, com $\norm f_p \leq \norm f_1^{1/p}\norm
f_\infty^{1 - 1/p}$;
(b) mostre que $f \mapsto \norm f_p$ é, para $f$ fixada, log-convexa em $\frac1p$, e dê
um exemplo em que $f \in L^p$ exatamente para $p$ em um intervalo dado
$(p_0, p_1)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:10}
(Jensen) Sejam $\mu$ uma \omterm{def:b3:measure:measure}{medida} de \emph{probabilidade}, $f \in
L^1(\mu)$ real
e $\Phi \colon \R \to \R$ convexa. Mostre que
\[
\Phi\Bigl(\int f\,\dd\mu\Bigr) \leq \int \Phi\circ
f\,\dd\mu
\]
\emph{(reta de suporte de $\Phi$ no ponto $m = \int f$)}. Deduza a
desigualdade entre as médias aritmética e geométrica e a monotonicidade
de $p \mapsto \norm f_p$ do~\cref{exo:b3:lp:1}(b).\index{desigualdade de Jensen}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:lp:11}
(Desigualdade de Young para a convolução) Sejam $1 \leq p, q, r \leq
\infty$ com $\frac1p + \frac1q = 1 + \frac1r$, e
$f \in
L^p(\R^d)$, $g \in L^q(\R^d)$.
(a) Demonstre $\norm{f * g}_r \leq \norm f_p\,\norm g_q$. \emph{(Escreva, para expoentes conjugados
deduzidos de $p, q,
r$,
\[
\abs{f(y)g(x-y)} = \bigl(\abs f^p\abs g^q\bigr)^{1/r}
\cdot\abs f^{\,p(1/p - 1/r)}\cdot\abs g^{\,q(1/q - 1/r)},
\]
e aplique a desigualdade de Hölder com três fatores, de expoentes
$r$, $\frac{pr}{r - p}$, $\frac{qr}{r - q}$; depois integre em $x$ por Tonelli.)}
(b) Confira os três casos particulares já conhecidos: $r =
\infty$ (Hölder,
\cref{exo:b3:lp:6}); $q = 1$ (estabilidade $L^p$ da convolução por um
núcleo \omterm{def:b3:lebesgue:l1}{integrável}); $p = q = 1$ ($L^1$ é uma álgebra de convolução,
\cref{thm:b3:product:convolution}).
(c) Por que não há desigualdade com $\frac1p + \frac1q < 1 +
\frac1r$? \emph{(Teste em dilatações
$f_\lambda(x) =
f(\lambda x)$ e compare os reescalamentos dos dois lados.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lp:12}
(Casos de igualdade) (a) Na desigualdade de Hölder $\int\abs{fg} \leq \norm f_p\norm g_q$ ($1 < p < \infty$), mostre
que a igualdade vale se, e somente se, $\abs f^p$ e $\abs g^q$ são proporcionais
q.t.p. \emph{(Acompanhe o caso de igualdade da desigualdade de Young
$ab \leq \frac{a^p}p + \frac{b^q}q$, que é $a^p
= b^q$.)}
(b) Na desigualdade de Minkowski $\norm{f + g}_p \leq \norm f_p
+ \norm g_p$ ($1 < p < \infty$), mostre que a igualdade
com $f,
g \neq 0$ força $g = cf$ q.t.p.\ com $c > 0$.
(c) Contraste com $p = 1$ e $p = \infty$: descreva os casos de igualdade
(muito maiores) que ali ocorrem, com exemplos.
\end{exercise}

\section{Problema: a desigualdade de Hardy}

\begin{problem}[{Problema de fim de semana --- a desigualdade de Hardy
e sua constante ótima}]\label{pb:b3:lp:1}
Para $f \in L^p(\intoo0{+\infty})$, $1 < p < \infty$, defina o \emph{operador de Hardy}
\[
(Hf)(x) = \frac1x\int_0^x f(t)\,\dd t .
\]
A desigualdade de Hardy (1920) afirma\index{desigualdade de Hardy}
\[
\norm{Hf}_p \;\leq\; \frac{p}{p-1}\,\norm f_p ,
\]
e a constante $\frac p{p-1}$ é ótima e não é atingida. Este problema demonstra
tudo e, em seguida, estende o resultado a séries.

\medskip
\noindent\textbf{Parte I --- A desigualdade.} Suponha primeiro $f
\geq 0$
\omterm{def:b3:topology:continuity}{contínua} de suporte \omterm{def:b3:topology:compact}{compacto} em $\intoo0{+\infty}$, e ponha $F(x) = \int_0^xf$.
\begin{enumerate}
  \item Mostre que $Hf \in L^p$: perto de $0$, $F$ se anula em uma
        \omterm{def:b3:topology:topology}{vizinhança} de $0$; perto de $\infty$, $F$ é limitada,
        de modo que $(Hf)(x) = O(1/x)$, e $x \mapsto \frac1x$ pertence a $L^p(\intoo1{+\infty})$ para $p > 1$.
  \item Integre por partes para mostrar
        \[
        \int_0^\infty \Bigl(\frac Fx\Bigr)^p\dd x
        = \frac{p}{p-1}\int_0^\infty\Bigl(\frac
        Fx\Bigr)^{p-1}f(x)\,\dd x .
        \]
        \emph{(Derive $x^{1-p}F^p$; os termos de fronteira se anulam ---
        justifique nas duas pontas.)}
  \item Aplique Hölder ao membro direito e deduza $\norm{Hf}_p \leq \frac p{p-1}\norm f_p$ para tais
        $f$.
  \item Estenda a todo o $L^p$: para $f \geq 0$, construa $f_n$
        \omterm{def:b3:topology:continuity}{contínuas} de suporte \omterm{def:b3:topology:compact}{compacto} em $\intoo0{+\infty}$, com $0 \leq f_n \nearrow f$ q.t.p.\
        \emph{(trunque e aproxime monotonamente --- justifique a
        construção)}; então $Hf_n \nearrow Hf$ pontualmente (convergência monótona
        dentro da média) e a convergência monótona passa a desigualdade
        ao limite. Para $f$ com sinal ou complexa, conclua com
        $\abs{Hf} \leq H\abs f$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Otimalidade.}
\begin{enumerate}[resume]
  \item Para $A > 1$, ponha $f_A(t) = t^{-1/p}\,\mathbf
        1_{\intcc1A}(t)$. Calcule $\norm{f_A}_p^p = \ln A$ e, para $1 \leq x \leq A$,
        \[
        (Hf_A)(x) = \frac p{p-1}\;x^{-1/p}\,
        \bigl(1 - x^{-(1 - 1/p)}\bigr).
        \]
  \item Deduza $\liminf_{A\to\infty} \norm{Hf_A}_p/
        \norm{f_A}_p \geq \frac{p}{p-1}$ e conclua que a constante é ótima.
  \item Mostre que a igualdade $\norm{Hf}_p = \frac
        p{p-1}\norm f_p$ com $f \neq 0$ é impossível.
        \emph{(Acompanhe o caso de igualdade de Hölder na questão 3:
        ela forçaria um comportamento do tipo $f = cx^{-1/p}$, que não está em
        $L^p$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A desigualdade discreta.}
\begin{enumerate}[resume]
  \item Para $g \geq 0$ não crescente em $(0,\infty)$ e $a_n = g(n)$, compare $\sum a_n^p$ e $\int g^p$,
        e as médias de $H$ correspondentes, para deduzir da Parte I
        a \emph{desigualdade discreta de Hardy}: para $a_n \geq 0$,
        \[
        \sum_{n\geq1}\Bigl(\frac{a_1 + \dots +
        a_n}{n}\Bigr)^{p}
        \;\leq\;
        \Bigl(\frac{p}{p-1}\Bigr)^{p}\,\sum_{n\geq1}a_n^p
        \]
        --- demonstre-a primeiro para $(a_n)$ não crescente, pela
        comparação acima, e reduza depois o caso geral ao não crescente
        por rearranjo \emph{(admita, com uma justificativa de uma
        linha, que ordenar $(a_n)$ de maneira decrescente só pode
        aumentar o membro esquerdo, deixando o direito inalterado)}.
  \item Deduza: se $\sum a_n^p < \infty$, então as médias de Cesàro de $(a_n)$ estão
        de novo em $\ell^p$ --- e dê um exemplo ($p = 2$) em que
        $(a_n) \in \ell^2$ mas $a_n$ não é somável, e ainda assim Hardy controla
        as médias.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Epílogo.}
\begin{enumerate}[resume]
  \item Mostre que a desigualdade de Hardy falha para $p = 1$: com
        $f = \mathbf 1_{\intcc01}$, calcule $Hf$ e observe que $Hf \notin L^1$. Onde a demonstração
        se quebra?
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- A função maximal e o teorema de
diferenciação de Lebesgue.} Hardy faz médias a partir da origem;
Hardy--Littlewood fazem médias \emph{em torno de cada ponto}. Para
$f \in L^1(\R)$, defina
\[
Mf(x) = \sup_{r>0}\ \frac1{2r}\int_{x-r}^{x+r}\abs
f\,\dd\lambda .
\]
\begin{enumerate}[resume]
  \item (Vitali, versão finita) Sejam $B_1, \dots, B_N$ intervalos abertos. Mostre
        que existe uma subfamília disjunta $B_{i_1}, \dots, B_{i_k}$ com $\bigcup_jB_j
        \subseteq \bigcup_l3B_{i_l}$, em que
        $3B$ denota o intervalo de mesmo \omterm{ex:b3:groups:actions}{centro} e comprimento triplo
        \emph{(guloso: escolha repetidamente o intervalo mais longo
        disjunto dos já escolhidos)}.
  \item (Tipo fraco $(1,1)$) Mostre que, para todo $t > 0$,
        \[
        \lambda\bigl(\{Mf > t\}\bigr) \;\leq\;
        \frac3t\,\norm f_1 :
        \]
        cada $x$ com $Mf(x) > t$ é dono de um intervalo centrado $B_x$
        com $\int_{B_x}\abs f > t\,\lambda(B_x)$; tome um \omterm{def:b3:topology:compact}{compacto} $K \subseteq \{Mf > t\}$ (regularidade \omterm{def:b3:topology:interior}{interior}),
        cubra-o por um número finito de $B_x$, aplique a questão 11
        e exaura.
  \item Calcule $M\mathbf 1_{\intcc01}(x)$ para $x > 1$ e deduza que $Mf \notin L^1$ para todo $f \neq 0$
        ($Mf(x) \geq \frac c{\abs x}$ no infinito): em $p =
        1$, a desigualdade fraca da
        questão 12 é o melhor enunciado possível.
  \item (Tipo forte para $p > 1$) Para $f \in L^p$: separe $f =
        f\,\mathbf 1_{\abs f > t/2} + f\,\mathbf 1_{\abs f
        \leq t/2}$, observe
        que $Mf \leq M\bigl(f\mathbf 1_{\abs
        f > t/2}\bigr) + \frac t2$ e combine a questão 12 com a fórmula do bolo de
        camadas (\cref{prop:b3:product:layercake}) e Tonelli para demonstrar
        \[
        \norm{Mf}_p^p \;\leq\;
        \frac{6p\,2^{p-1}}{p-1}\,\norm f_p^p .
        \]
        (A explosão quando $p \downarrow 1$ é o fracasso da questão 13,
        quantificado.)
  \item (Teorema de diferenciação de Lebesgue) Demonstre: para $f \in
        L^1(\R)$,
        \[
        \frac1{2r}\int_{x-r}^{x+r}f\,\dd\lambda
        \;\longrightarrow\; f(x)
        \qquad (r \to 0)\quad\text{para q.t.p.\ }x .
        \]
        \emph{(Claro para $f$ \omterm{def:b3:topology:continuity}{contínua}. No caso geral, escreva
        $f
        = g + h$, com $g$ \omterm{def:b3:topology:continuity}{contínua} de suporte \omterm{def:b3:topology:compact}{compacto} e $\norm h_1 < \varepsilon$
        (\cref{thm:b3:lp:density}); o conjunto em que o $\limsup_{r\to0}$ da oscilação média
        ultrapassa $\delta$ está contido em $\{Mh > \delta/2\} \cup \{\abs h
        > \delta/2\}$, de \omterm{def:b3:measure:measure}{medida} $O(\varepsilon/\delta)$;
        faça $\varepsilon \to 0$ e, depois, $\delta \to 0$ ao longo de uma sequência.)}
  \item Deduza: (a) quase todo ponto é um \emph{ponto de Lebesgue} de
        $f$; (b) para $f \in L^1$, a primitiva $F(x) = \int_0^xf$ é derivável q.t.p.\
        com $F' =
        f$ q.t.p.\ --- a metade integral do teorema
        fundamental do cálculo no mundo de Lebesgue, fechando o círculo
        aberto pela escada (\cref{pb:b3:measure:1}), que mostrou que a outra
        metade pode falhar.
  \item (Pontos de \omterm{ex:b3:lebesgue:gamma}{densidade}) Para $A \subseteq \R$ \omterm{def:b3:lebesgue:measurable}{mensurável}, mostre que quase
        todo $x \in A$ satisfaz $\frac{\lambda(A\cap\intcc{x-r}{x+r})}{2r} \to 1$: os conjuntos \omterm{def:b3:lebesgue:measurable}{mensuráveis} são
        localmente cheios em quase todos os seus pontos. Esboce, em
        duas linhas, como isso fornece mais uma demonstração do teorema
        de Steinhaus (\cref{exo:b3:measure:8}).
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Variações sobre o tema da média.}
\begin{enumerate}[resume]
  \item (Hardy com peso) Para $\alpha < p - 1$ e $f \geq 0$, mostre que
        \[
        \int_0^\infty\Bigl(\frac{F(x)}x\Bigr)^{p}
        x^{\alpha}\,\dd x \;\leq\;
        \Bigl(\frac{p}{p - 1 - \alpha}\Bigr)^{p}
        \int_0^\infty f(x)^p\,x^{\alpha}\,\dd x
        \]
        pela mesma integração por partes, e verifique que o caso limite
        $\alpha = p - 1$ é genuinamente proibido (adapte o contraexemplo da questão
        10).
  \item (O adjunto) Seja $H^*f(x) =
        \int_x^{\infty}\frac{f(t)}t\,\dd t$. Mostre que $\langle
        Hf, g\rangle = \langle f, H^*g\rangle$ para $f, g$ não
        negativas (Tonelli) e demonstre $\norm{H^*f}_p \leq p\,\norm f_p$ \emph{(diretamente por
        partes, ou a partir de Hardy no expoente conjugado por
        dualidade --- atenção a qual expoente recebe qual constante)}.
  \item (Uma desigualdade do tipo Hilbert) Deduza que, para $f \in L^p$ não
        negativas, $g \in L^q$:
        \[
        \int_0^\infty\!\!\int_0^\infty
        \frac{f(x)\,g(y)}{\max(x,y)}\,\dd x\,\dd y
        \;\leq\; (p + q)\,\norm f_p\,\norm g_q
        \]
        \emph{(separe ao longo de $y < x$ / $y \geq x$: cada metade é um
        emparelhamento de uma função contra a transformada de Hardy da
        outra)}.
  \item (Otimalidade, caso discreto) Mostre que a constante $\bigl(\frac p{p-1}\bigr)^p$ da
        questão 8 também é ótima: teste em $a_n = n^{-1/p}\,\mathbf 1_{n \leq
        N}$, compare os dois lados
        com integrais e faça $N
        \to \infty$ (o espelho discreto da Parte II).
  \item (Síntese) Três operadores de média apareceram neste problema:
        o $H$ de Hardy, a média discreta de Cesàro e o operador
        maximal $M$. Enuncie em uma linha o que a limitação de cada
        um diz, observe que os três falham exatamente em $p = 1$ e
        explique por que se trata do mesmo fracasso três vezes (a cauda
        harmônica $\frac1x$).
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- A desigualdade de Carleman, e quão
ótimo é ótimo.}
\begin{enumerate}[resume]
  \item (Desigualdade de Carleman) Seja $a_n \geq 0$ com $\sum
        a_n < \infty$. Aplique a
        desigualdade discreta de Hardy da questão 8 a $b_n = a_n^{1/p}$, use a
        desigualdade entre as médias aritmética e geométrica e faça
        $p \to
        \infty$ (mostre que $p \mapsto
        \bigl(\frac p{p-1}\bigr)^p$ decresce até $\eu$) para obter
        \[
        \sum_{n\geq1}\bigl(a_1a_2\cdots a_n\bigr)^{1/n}
        \;\leq\; \eu\,\sum_{n\geq1}a_n :
        \]
        as médias geométricas de uma sequência somável são somáveis, ao
        custo de no máximo $\eu$.
  \item (A constante $\eu$ é ótima) Teste $a_n =
        \frac1n\,\mathbf 1_{n\leq N}$: usando o
        enquadramento de Stirling do~\cref{pb:b3:product:1}, mostre que $(n!)^{-1/n} = \frac\eu n\bigl(1 +
        O\bigl(\frac{\ln n}n\bigr)\bigr)$,
        deduza que ambos os lados de Carleman crescem como $\eu\ln N$ e
        conclua que nenhuma constante menor que $\eu$ pode
        funcionar. (Observe o padrão: os otimizadores de Hardy e de
        Carleman são ambos as sequências de tipo harmônico que por
        pouco deixam de estar no espaço.)
  \item (Com que lentidão se aproxima o ``ótimo''?) Tome $p = 2$.
        Para $f = \mathbf 1_{\intcc01}$, calcule $\norm{Hf}_2/\norm f_2 = \sqrt2$, contra a cota $2$. Para os
        quase-otimizadores $f_A$ da questão 5, demonstre a
        identidade exata
        \[
        \norm{Hf_A}_2^2 = 4\ln A - 8 + \frac{8}{\sqrt A},
        \qquad\text{logo}\qquad
        \frac{\norm{Hf_A}_2^2}{\norm{f_A}_2^2}
        = 4 - \frac{8 - 8A^{-1/2}}{\ln A} .
        \]
        Avalie em $A = \eu^{10}$ (razão $\approx 1.790$) e comente: o supremo $2$ é
        aproximado apenas à velocidade $1/\ln A$ --- uma constante ótima
        pode ser quase invisível numericamente.
\end{enumerate}
\end{problem}
