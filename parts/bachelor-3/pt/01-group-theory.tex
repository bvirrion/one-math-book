\chapter{Teoria dos grupos}\label{ch:b3:groups}

No volume do segundo ano, os grupos não passavam de instrumentos de
contagem: teorema de Lagrange, grupos cíclicos, grupo simétrico e sua
assinatura. Este capítulo transforma a teoria dos grupos em um
\emph{método}. O motor é a noção de grupo que \emph{age} sobre um
conjunto: contar \omterm{def:b3:groups:action}{órbitas} e pontos fixos produz a \omterm{cor:b3:groups:classeq}{equação das classes}, o
teorema de Cauchy e os três teoremas de Sylow --- o princípio
local-global fundamental da teoria dos grupos finitos. Aprendemos em
seguida a montar grupos (\omterm{prop:b3:groups:direct}{produtos diretos} e semidiretos) e a
desmontá-los (\omterm{thm:b3:groups:jordanholder}{séries de composição}, \omterm{def:b3:groups:derived}{grupos solúveis}), e demonstramos o
teorema que, no~\cref{ch:b3:galois}, encerrará uma questão de três séculos
sobre equações polinomiais: o \omterm{thm:b3:groups:ansimple}{grupo alternado} $A_n$ é \omterm{def:b3:groups:simple}{simples} para
$n
\geq 5$.

\section{Grupos quocientes e teoremas de isomorfismo}

Ao longo de todo o capítulo, $G$ é um grupo escrito
multiplicativamente e $e$ é seu elemento neutro. Recordamos do volume
do segundo ano: subgrupos, classes laterais $gH$, teorema de Lagrange
($\abs G = [G:H]\,\abs H$ para $G$ finito), a ordem de um elemento, os grupos cíclicos e o
grupo simétrico $S_n$ com seu morfismo assinatura $\varepsilon \colon S_n \to \{\pm1\}$.

\begin{definition}\label{def:b3:groups:normal}
Um subgrupo $N$ de $G$ é \emph{normal}\index{subgrupo normal}
(escreve-se $N \trianglelefteq G$) quando $gNg^{-1} = N$ para todo $g
\in G$ --- de maneira
equivalente, quando as classes laterais à esquerda e à direita
coincidem: $gN =
Ng$ para todo $g$.
\end{definition}

\begin{theorem}[Grupo quociente]\label{thm:b3:groups:quotient}
Seja $N \trianglelefteq G$. O conjunto $G/N$ das classes laterais, munido da
multiplicação $(gN)(hN) = ghN$, é um grupo bem definido, o \emph{grupo
quociente}\index{grupo quociente}, e a \emph{projeção canônica} $\pi \colon G \to G/N$,
$g \mapsto gN$, é um morfismo sobrejetor de núcleo $N$. Reciprocamente, todo
núcleo de um morfismo de grupos é \omterm{def:b3:groups:normal}{normal}: os \omterm{def:b3:groups:normal}{subgrupos normais} são
exatamente os núcleos.
\end{theorem}

\begin{proof}
Tudo está em verificar que a multiplicação está bem definida. Se $gN = g'N$ e
$hN = h'N$, escreva $g' = gn$, $h' = hm$ com $n, m \in N$. Então $g'h' = gnhm =
gh\,(h^{-1}nh)\,m \in ghN$ pois $h^{-1}nh \in N$ por
normalidade: o produto de classes laterais não depende dos
representantes. A associatividade, o elemento neutro $eN = N$ e os
inversos $(gN)^{-1} = g^{-1}N$ são herdados de $G$. É claro que $\pi$ é um morfismo
sobrejetor e que $\pi(g) = N \iff g \in N$.

Se $f \colon G \to H$ é um morfismo e $k \in \ker f$, então $f(gkg^{-1}) = f(g)f(k)f(g)^{-1} = e$: os núcleos são normais.
\end{proof}

\begin{theorem}[Propriedade universal; primeiro teorema de isomorfismo]
\label{thm:b3:groups:firstiso}
Sejam $f \colon G \to H$ um morfismo e $N \trianglelefteq G$ com $N \subseteq \ker f$. Existe um único morfismo $\bar f \colon G/N
\to H$
tal que $f = \bar f \circ \pi$. Em particular, tomando $N = \ker
f$:\index{teoremas de
isomorfismo}
\[
G/\ker f \;\xrightarrow{\;\sim\;}\; \operatorname{im} f,
\qquad gN \mapsto f(g).
\]
\end{theorem}

\begin{proof}
Unicidade: $\bar f(gN)$ tem necessariamente de valer $f(g)$. Existência: se
$gN = g'N$, então $g^{-1}g' \in N \subseteq \ker f$, logo $f(g) = f(g')$ e $\bar
f(gN) = f(g)$ está bem definido; trata-se de um
morfismo porque $f$ o é. Para $N = \ker f$: $\bar f$ é injetor, pois
$\bar f(gN) = e$ significa $g \in \ker f$, isto é, $gN = N$; sua imagem é a de $f$.
\end{proof}

\begin{theorem}[Segundo e terceiro teoremas de isomorfismo]
\label{thm:b3:groups:secondiso}
Sejam $H \leq G$ e $N \trianglelefteq G$.
\begin{enumerate}
  \item $HN = \{hn : h \in H,\, n \in N\}$ é um subgrupo, $N
        \trianglelefteq HN$, $H \cap N \trianglelefteq H$, e
        \[
        H/(H \cap N) \;\cong\; HN/N .
        \]
  \item Se, além disso, $N \subseteq K \trianglelefteq G$, então $K/N
        \trianglelefteq G/N$ e $(G/N)\big/(K/N) \cong G/K$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) $HN$ é um subgrupo: $(hn)(h'n') = hh'\,(h'^{-1}nh')n' \in HN$ e $(hn)^{-1} = h^{-1}(hn^{-1}h^{-1}) \in HN$, usando a normalidade de $N$.
Componha $H \hookrightarrow HN \xrightarrow{\pi} HN/N$: esse morfismo é sobrejetor ($hnN = hN$) e tem núcleo $\{h \in H : h \in
N\} = H \cap N$;
aplique o~\cref{thm:b3:groups:firstiso}.

(2) A projeção $G/N \to G/K$, $gN \mapsto gK$, está bem definida ($N \subseteq K$), é sobrejetora e tem
núcleo $K/N$; aplique novamente o~\cref{thm:b3:groups:firstiso}.
\end{proof}

\begin{theorem}[Teorema da correspondência]\label{thm:b3:groups:correspondence}
Seja $N \trianglelefteq G$. A aplicação $H \mapsto H/N$ é uma bijeção entre os subgrupos de $G$
que contêm $N$ e os subgrupos de $G/N$, que preserva inclusões,
índices e normalidade (nos dois sentidos).
\end{theorem}

\begin{proof}
Sua inversa é $\bar H \mapsto \pi^{-1}(\bar H)$. Ambas as aplicações levam subgrupos em subgrupos e
são mutuamente inversas: $\pi^{-1}(H/N) =
HN = H$ pois $N \subseteq H$, e $\pi(\pi^{-1}(\bar H)) = \bar H$ pela sobrejetividade de
$\pi$. As inclusões são claramente preservadas; $[G:H] = [G/N : H/N]$ porque $gH \mapsto (gN)(H/N)$ é
uma bijeção bem definida entre os espaços de classes laterais; e $gHg^{-1} = H$
para todo $g$ se, e somente se, $(gN)(H/N)(gN)^{-1} = H/N$ para todo $gN$, de novo
pela sobrejetividade de $\pi$.
\end{proof}

\begin{example}\label{ex:b3:groups:firstiso}
$\varepsilon \colon S_n \to \{\pm 1\}$ dá $S_n/A_n \cong
\{\pm1\}$; $\det \colon GL_n(K) \to K^\times$ dá $GL_n(K)/SL_n(K) \cong K^\times$; $t \mapsto \eu^{2\iu\pi t}$ dá $\R/\Z \cong \mathbb U$, o grupo do círculo. O
primeiro teorema de isomorfismo é a maneira como os quocientes são
\emph{calculados} na prática: encontre uma sobrejeção com o núcleo
certo.
\end{example}

\begin{method}\label{met:b3:groups:normal}
Para demonstrar que $N \trianglelefteq G$, em ordem decrescente de elegância: exiba
$N$ como núcleo de um morfismo definido em $G$; verifique $gNg^{-1} \subseteq N$
para todo $g$ (isso basta: aplicando a $g^{-1}$ e conjugando
obtém-se a inclusão inversa); verifique que $N$ é uma reunião de
\omterm{ex:b3:groups:actions}{classes de conjugação}; ou observe que $[G:N] = 2$ (então $gN = Ng$ é forçado ---
\cref{exo:b3:groups:1}).
\end{method}

\section{Ações de grupo}

\begin{definition}\label{def:b3:groups:action}
Uma \emph{ação}\index{ação de grupo} de $G$ sobre um conjunto
$X$ é um morfismo $\varphi \colon G \to \mathfrak{S}(X)$ no grupo das bijeções de $X$;
escreve-se $g \cdot x$ em lugar de $\varphi(g)(x)$. De maneira equivalente: uma
aplicação $G \times X \to X$ com $e \cdot x = x$ e $g
\cdot (h \cdot x) = (gh) \cdot x$. A \emph{órbita}\index{órbita} de
$x$ é $\mathcal O_x = \{g \cdot x : g \in G\}$, seu \emph{estabilizador}\index{estabilizador} é o
subgrupo $G_x = \{g :
g\cdot x = x\}$, e $X^G = \{x : \forall g,\ g \cdot x = x\}$ é o conjunto dos \emph{pontos fixos}. A ação é
\emph{transitiva} se há exatamente uma órbita, \emph{fiel} se $\varphi$ é
injetor, \emph{livre} se todos os estabilizadores são triviais.
\end{definition}

\begin{example}\label{ex:b3:groups:actions}
Cinco ações governam toda a teoria dos grupos finitos:
\begin{enumerate}
  \item $G$ sobre si mesmo por \emph{translação à esquerda} $g \cdot x = gx$:
        livre e transitiva.
  \item $G$ sobre si mesmo por \emph{conjugação} $g \cdot x = gxg^{-1}$: as \omterm{def:b3:groups:action}{órbitas}
        são as \emph{classes de conjugação}\index{classe de
        conjugação}, os \omterm{def:b3:groups:action}{estabilizadores} são os
        \emph{centralizadores}\index{centralizador} $Z_G(x) = \{g : gx = xg\}$, e os pontos
        fixos formam o \emph{centro}\index{centro} $Z(G)$.
  \item $G$ sobre o espaço de classes laterais $G/H$ por $g \cdot xH = gxH$:
        transitiva, com \omterm{def:b3:groups:action}{estabilizador} da classe $H$ igual a
        $H$. Toda \omterm{def:b3:groups:action}{ação} transitiva é dessa forma
        (\cref{exo:b3:groups:8}).
  \item $G$ sobre o conjunto de seus subgrupos por conjugação: o
        \omterm{def:b3:groups:action}{estabilizador} de $H$ é o \emph{normalizador}\index{normalizador}
        $N_G(H) =
        \{g : gHg^{-1} = H\}$, o maior subgrupo de $G$ no qual $H$ é \omterm{def:b3:groups:normal}{normal}.
  \item $S_n$ sobre $\intint{1}{n}$: a mãe de todos os exemplos.
\end{enumerate}
\end{example}

\begin{theorem}[Órbita--estabilizador]\label{thm:b3:groups:orbitstab}
A aplicação $gG_x \mapsto g \cdot x$ é uma bijeção bem definida $G/G_x \to \mathcal O_x$. Em particular, para
$G$ finito,\index{teorema órbita--estabilizador}
\[
\abs{\mathcal O_x} = [G : G_x] \quad\text{divide } \abs G,
\]
e, como as \omterm{def:b3:groups:action}{órbitas} formam uma partição de $X$ (elas são as classes
da equivalência $x \sim y \iff y \in \mathcal O_x$),
\[
\abs X = \sum_{i} \,[G : G_{x_i}]
\qquad (x_i\colon \text{um ponto por órbita}).
\]
\end{theorem}

\begin{proof}
Boa definição e injetividade: $gG_x = hG_x \iff h^{-1}g \in G_x \iff
h^{-1}g \cdot x = x \iff g \cdot x = h \cdot x$; basta ler a cadeia nos dois
sentidos. A sobrejetividade é a própria definição de \omterm{def:b3:groups:action}{órbita}. Os
enunciados de contagem decorrem do teorema de Lagrange e da partição de
$X$ em \omterm{def:b3:groups:action}{órbitas}.
\end{proof}

\begin{corollary}[Equação das classes]\label{cor:b3:groups:classeq}
Para um grupo finito $G$, escolhendo um representante $x_i$ em
cada \omterm{ex:b3:groups:actions}{classe de conjugação} com mais de um elemento:\index{equação das
classes}
\[
\abs G = \abs{Z(G)} + \sum_i \,[G : Z_G(x_i)],
\qquad\text{cada } [G : Z_G(x_i)] > 1 \text{ que divide } \abs G.
\]
\end{corollary}

\begin{proof}
Aplique o~\cref{thm:b3:groups:orbitstab} à \omterm{def:b3:groups:action}{ação} por conjugação: as \omterm{def:b3:groups:action}{órbitas} unitárias
são exatamente os elementos de $Z(G)$.
\end{proof}

\begin{theorem}[Pontos fixos dos $p$-grupos]\label{thm:b3:groups:pfixed}
Seja $p$ primo. Um \emph{$p$-grupo}\index{p-grupo@$p$-grupo}
é um grupo finito cuja ordem é uma potência de $p$. Se um
$p$-grupo $G$ age sobre um conjunto finito $X$, então
\[
\abs{X^G} \equiv \abs X \pmod p .
\]
Consequências: um $p$-grupo não trivial tem \omterm{ex:b3:groups:actions}{centro} não trivial, e
todo grupo de ordem $p^2$ é abeliano.
\end{theorem}

\begin{proof}
Cada \omterm{def:b3:groups:action}{órbita} tem cardinalidade $[G:G_x]$, uma potência de $p$; essa
potência vale $1$ exatamente nos pontos fixos e é divisível por
$p$ nos demais casos. Somando sobre as \omterm{def:b3:groups:action}{órbitas} obtém-se a
congruência. Quanto ao \omterm{ex:b3:groups:actions}{centro}: a \omterm{def:b3:groups:action}{ação} de $G$ sobre si mesmo por
conjugação tem $X^G = Z(G)$, logo $\abs{Z(G)} \equiv
\abs G \equiv 0 \pmod p$, e $Z(G) \ni e$ força $\abs{Z(G)} \geq
p$. Ordem
$p^2$: se $Z(G) \neq G$, então $\abs{Z(G)} = p$ e $G/Z(G)$ é cíclico de ordem
$p$, o que força $G$ abeliano (\cref{exo:b3:groups:2}) --- contradição.
\end{proof}

\begin{theorem}[Cauchy]\label{thm:b3:groups:cauchy}
Se um primo $p$ divide $\abs G$, então $G$ contém um
elemento de ordem $p$.\index{teorema de Cauchy}
\end{theorem}

\begin{proof}[Demonstração (McKay)]
Seja $X = \{(g_1, \dots, g_p) \in G^p : g_1 g_2 \cdots g_p = e\}$. Escolher $g_1, \dots, g_{p-1}$ livremente determina $g_p$: $\abs X =
\abs G^{p-1}$, que é
divisível por $p$. O grupo cíclico $\Z/p\Z$ age sobre $X$
por permutação circular $(g_1, \dots, g_p) \mapsto (g_2, \dots, g_p,
g_1)$ --- o que preserva $X$, pois $g_2 \cdots g_p g_1 =
g_1^{-1}(g_1 \cdots g_p)g_1 = e$.
Pelo~\cref{thm:b3:groups:pfixed}, $\abs{X^{\Z/p\Z}} \equiv \abs X \equiv 0 \pmod p$. Os pontos fixos são as $(g, \dots, g)$ constantes com
$g^p = e$; a $(e,
\dots, e)$ é um deles, de modo que há pelo menos $p$ deles e,
portanto, pelo menos um $g \neq e$ com $g^p = e$: sua ordem é exatamente
$p$.
\end{proof}

\begin{theorem}[Cayley]\label{thm:b3:groups:cayley}
Todo grupo de ordem $n$ mergulha em $S_n$.\index{teorema de
Cayley}
\end{theorem}

\begin{proof}
A translação à esquerda $\varphi \colon G \to \mathfrak S(G) \cong S_n$ é um morfismo; $\varphi(g) = \mathrm{id}$ força $g = ge = e$: ela é
fiel.
\end{proof}

\begin{method}\label{met:b3:groups:counting}
Contar pontos fixos é o lance de abertura universal da teoria dos
grupos finitos. Para demonstrar que algo \emph{existe} (um elemento
central, um elemento de ordem $p$, um \omterm{def:b3:groups:normal}{subgrupo normal}, um ponto
fixo), faça um grupo bem escolhido agir sobre um conjunto finito bem
escolhido e compare então $\abs{X^G}$ com $\abs X$ módulo $p$, ou
deixe os tamanhos das \omterm{def:b3:groups:action}{órbitas} dividirem a ordem do grupo. As
demonstrações do~\cref{thm:b3:groups:pfixed,thm:b3:groups:cauchy} e dos três teoremas de Sylow abaixo são
cinco variações sobre essa única ideia.
\end{method}

\section{Os teoremas de Sylow}

O teorema de Lagrange afirma que a ordem de um subgrupo divide
$\abs G$; a recíproca é falsa ($A_4$, de ordem $12$, não
tem subgrupo de ordem $6$ --- \cref{exo:b3:groups:1}). Os teoremas de Sylow
salvam a recíproca para potências de primos, e sua cláusula de contagem
é a ferramenta geral mais fina de que dispomos para produzir \omterm{def:b3:groups:normal}{subgrupos
normais}.

\begin{definition}\label{def:b3:groups:sylow}
Escreva $\abs G = p^a m$ com $p \nmid m$. Um \emph{$p$-subgrupo de
Sylow}\index{subgrupo de Sylow} de $G$ é um subgrupo de ordem
$p^a$ --- um $p$-subgrupo da maior ordem concebível. O número
de $p$-subgrupos de Sylow de $G$ é denotado por $n_p$.
\end{definition}

\begin{lemma}\label{lem:b3:groups:binom}
Se $\abs G = p^a m$ com $p \nmid m$, então $\dbinom{p^a m}{p^a}
\equiv m \pmod p$.
\end{lemma}

\begin{proof}
Em $\mathbb F_p[X]$, o sonho do calouro $(1+X)^p = 1 + X^p$ (os coeficientes $\binom pk$, $0<k<p$,
são divisíveis por $p$: $p$ divide o numerador de $\frac{p!}{k!(p-k)!}$, mas não
o denominador) itera-se em $(1+X)^{p^a} = 1 + X^{p^a}$, de onde
\[
(1+X)^{p^a m} = \bigl(1 + X^{p^a}\bigr)^m
= \sum_{k=0}^{m} \binom mk X^{k p^a} \quad\text{em } \mathbb
F_p[X].
\]
Identifique o coeficiente de $X^{p^a}$: à esquerda $\binom{p^a
m}{p^a} \bmod p$, à direita $\binom m1 = m$.
\end{proof}

\begin{theorem}[Sylow I: existência]\label{thm:b3:groups:sylow1}
Para todo primo $p$, os $p$-subgrupos de Sylow de $G$
existem.
\end{theorem}

\begin{proof}[Demonstração (Wielandt)]
Seja $\Omega$ o conjunto dos \emph{subconjuntos} de $G$ de
cardinalidade $p^a$; $G$ age sobre $\Omega$ por translação
à esquerda $g \cdot S = gS$. Pelo~\cref{lem:b3:groups:binom}, $\abs\Omega = \binom{p^a m}{p^a}
\equiv m \not\equiv 0 \pmod p$, logo alguma \omterm{def:b3:groups:action}{órbita} $\mathcal O_S$
tem tamanho primo com $p$ (se $p$ dividisse o tamanho de toda
\omterm{def:b3:groups:action}{órbita}, dividiria $\abs\Omega$). Seja $H = G_S$ o \omterm{def:b3:groups:action}{estabilizador} de um tal $S$.
Como $[G : H] = \abs{\mathcal O_S}$ é primo com $p$ e $p^a \mid
\abs G = [G:H]\,\abs H$, obtemos $p^a \mid \abs H$. Reciprocamente,
fixe $s \in S$: a aplicação $H \to S$, $h \mapsto hs$, é injetora e toma valores em
$S$ porque $hS = S$; logo $\abs H \leq \abs S = p^a$. Portanto $\abs H = p^a$.
\end{proof}

\begin{theorem}[Sylow II: dominação e conjugação]\label{thm:b3:groups:sylow2}
Sejam $P$ um $p$-subgrupo de Sylow e $Q$ um $p$-subgrupo
qualquer de $G$. Então $Q \subseteq gPg^{-1}$ para algum $g \in G$. Em particular, todos
os $p$-subgrupos de Sylow são conjugados, e $P \trianglelefteq G \iff
n_p = 1$.
\end{theorem}

\begin{proof}
Faça $Q$ agir sobre o espaço de classes laterais $X = G/P$, de
cardinalidade $m
\not\equiv 0 \pmod p$. Pelo~\cref{thm:b3:groups:pfixed} aplicado ao $p$-grupo
$Q$, $\abs{X^Q} \equiv m \not\equiv 0 \pmod p$: existe uma classe lateral fixa $gP$, isto é, $QgP = gP$,
ou seja, $g^{-1}Qg \subseteq
P$. Se $Q$ é, ele próprio, um \omterm{def:b3:groups:sylow}{subgrupo de Sylow}, a
igualdade das ordens transforma $Q \subseteq gPg^{-1}$ em uma igualdade. Por fim, $P
\trianglelefteq G$
se, e somente se, seus conjugados $\{gPg^{-1}\}$ --- que, pelo que precede, são
\emph{todos} os $p$-subgrupos de Sylow --- se reduzem a $\{P\}$.
\end{proof}

\begin{theorem}[Sylow III: contagem]\label{thm:b3:groups:sylow3}
$n_p \equiv 1 \pmod p$, e $n_p = [G : N_G(P)]$, que divide $m$.
\end{theorem}

\begin{proof}
Seja $\mathrm{Syl}_p$ o conjunto dos $p$-subgrupos de Sylow; $G$ age sobre
ele transitivamente por conjugação (\cref{thm:b3:groups:sylow2}), com \omterm{def:b3:groups:action}{estabilizador}
de $P$ igual ao \omterm{ex:b3:groups:actions}{normalizador} $N_G(P) \supseteq P$: $n_p = [G :
N_G(P)]$, e $m = [G:P] = [G:N_G(P)]\,[N_G(P):P]$ mostra que $n_p \mid
m$.

Restrinja agora a \omterm{def:b3:groups:action}{ação} a $P$ e conte os pontos fixos. Se $Q \in
\mathrm{Syl}_p$ é
fixado por $P$, então $P \subseteq N_G(Q)$; $P$ e $Q$ são ambos
$p$-subgrupos de Sylow do grupo $N_G(Q)$ e, portanto, nele
conjugados (\cref{thm:b3:groups:sylow2} aplicado a $N_G(Q)$); mas $Q \trianglelefteq N_G(Q)$, de modo
que $Q$ é ali seu único conjugado: $P = Q$. Assim, o único
ponto fixo é o próprio $P$, e o~\cref{thm:b3:groups:pfixed} dá $n_p = \abs{\mathrm{Syl}_p}
\equiv \abs{\mathrm{Syl}_p^P} = 1 \pmod p$.
\end{proof}

\begin{method}\label{met:b3:groups:sylowmethod}
Para analisar um grupo de ordem dada $n = p^a m$: liste os divisores de
$m$ congruentes a $1$ módulo $p$ --- são esses os
candidatos a $n_p$. Se o único candidato for $1$, o
$p$-subgrupo de Sylow é \omterm{def:b3:groups:normal}{normal}. Se $n_p > 1$ for obrigado a ser
pequeno, faça o grupo agir por conjugação sobre $\mathrm{Syl}_p$ para obter um
morfismo $G \to S_{n_p}$ de núcleo pequeno. E conte elementos: $p$-subgrupos
de Sylow distintos de ordem \emph{prima} $p$ intersectam-se
trivialmente, de modo que contêm $n_p(p-1)$ elementos de ordem exatamente
$p$; as contagens para primos diferentes, quando se superpõem,
costumam forçar uma contradição (\cref{exo:b3:groups:7}).
\end{method}

\begin{example}\label{ex:b3:groups:pq}
Seja $\abs G = pq$ com $p < q$ primos e $p \nmid q - 1$. Então $n_q \mid p$ e $n_q \equiv 1 \bmod q$ forçam
$n_q = 1$ (pois $p <
q$); $n_p \mid q$ e $n_p \equiv 1 \bmod p$ forçam $n_p = 1$ (pois $q
\not\equiv 1 \bmod p$).
Sejam $P, Q$ os dois \omterm{def:b3:groups:sylow}{subgrupos de Sylow} normais: $P
\cap Q = \{e\}$ (ordens
coprimas), logo $\abs{PQ} = pq$ (\cref{exo:b3:groups:4}) e $G \cong P \times Q \cong \Z/p\Z
\times \Z/q\Z \cong \Z/pq\Z$ pela~\cref{prop:b3:groups:direct}
abaixo. Todo grupo de ordem $15$, $33$, $35$, \dots\ é
cíclico. O caso excluído $p \mid q - 1$ produz exatamente um grupo a mais, não
abeliano --- veja o problema de fim de semana (\cref{pb:b3:groups:1}).
\end{example}

\begin{example}[Um censo de Sylow completo: $S_4$]
\label{ex:b3:groups:s4sylow}
Apliquemos o método a $G = S_4$, $\abs G = 24 = 2^3\cdot3$. \emph{Sylow $3$:} $n_3 \mid 8$, $n_3 \equiv 1 \bmod 3$,
logo $n_3
\in \{1, 4\}$; como $\langle(123)\rangle$ e $\langle(124)\rangle$ são distintos, $n_3 = 4$ --- os quatro
subgrupos $\langle(abc)\rangle$, um para cada subconjunto de $3$ elementos $\{a, b, c\}$,
que respondem pelos $8$ ciclos de comprimento três. Por
Sylow~II eles são conjugados, e o morfismo de conjugação $S_4 \to
S_{\mathrm{Syl}_3} \cong S_4$ é aqui
um isomorfismo (seu núcleo está contido em $N = N_G(\langle(123)\rangle)$, de ordem $24/4 =
6$, e um
\omterm{def:b3:groups:normal}{subgrupo normal} de $S_4$ dentro de um $N$ do tipo $S_3$ tem
de ser trivial: ele seria formado por permutações pares fixando os
quatro \omterm{def:b3:groups:sylow}{subgrupos de Sylow}, e só $e$ o faz). \emph{Sylow $2$:}
$n_2 \mid 3$, $n_2
\equiv 1 \bmod 2$: $n_2 \in \{1, 3\}$. O subgrupo $D =
\langle(1234), (13)\rangle$ tem ordem $8$ (um diedral
$D_4$: as simetrias do quadrado de vértices $1, 2, 3, 4$) e não é \omterm{def:b3:groups:normal}{normal}
($(12)(1234)(12) = (2134)$ gera um outro subgrupo gerado por um $4$-ciclo), logo $n_2 = 3$:
as três cópias de $D_4$ correspondem às três maneiras de agrupar os
$4$ pontos em um ``quadrado''. Note a moral do censo: $\abs{S_4} = 24$ deixa
espaço para que qualquer um dos dois \omterm{def:b3:groups:sylow}{subgrupos de Sylow} deixe de ser
\omterm{def:b3:groups:normal}{normal}, e ambos deixam --- compare com a ordem $12$, em que a
contagem força um deles a ser \omterm{def:b3:groups:normal}{normal} (Parte~IV do~\cref{pb:b3:groups:1}).
\end{example}

\section{Produtos, diretos e semidiretos}

\begin{proposition}[Reconhecer um produto direto]\label{prop:b3:groups:direct}
Sejam $H, K \trianglelefteq G$ com $H \cap K = \{e\}$ e $HK = G$. Então $(h,k) \mapsto hk$ é um isomorfismo
$H \times K \to
G$.\index{produto direto}
\end{proposition}

\begin{proof}
Para $h \in H$, $k \in K$, o \omterm{def:b3:groups:derived}{comutador} $hkh^{-1}k^{-1}$ pertence a $K$ (leia-o como
$(hkh^{-1})k^{-1}$, usando a normalidade de $K$) e a $H$ (leia-o como $h(kh^{-1}k^{-1})$):
ele vale $e$, logo $H$ e $K$ comutam elemento a elemento e
a aplicação é um morfismo. Ela é sobrejetora pois $HK = G$, e
injetora pois $hk = e$ dá $h = k^{-1} \in
H \cap K = \{e\}$.
\end{proof}

A normalidade de \emph{ambos} os fatores é o que falha com mais
frequência: em $S_3
= \langle (1\,2\,3)\rangle \,\langle(1\,2)\rangle$ os dois fatores intersectam-se trivialmente e
geram o grupo, e no entanto $S_3 \not\cong \Z/3\Z \times
\Z/2\Z$. A noção correta quando apenas um dos
fatores é \omterm{def:b3:groups:normal}{normal}:

\begin{definition}\label{def:b3:groups:semidirect}
Sejam $H$, $K$ grupos e $\varphi \colon K \to
\operatorname{Aut}(H)$ um morfismo. O \emph{produto
semidireto}\index{produto semidireto} $H \rtimes_\varphi K$ é o conjunto $H \times K$ munido de
\[
(h, k)\,(h', k') = \bigl(h\,\varphi(k)(h'),\; kk'\bigr).
\]
\end{definition}

\begin{proposition}\label{prop:b3:groups:semidirect}
$H \rtimes_\varphi K$ é um grupo; $H \times \{e\}$ é um \omterm{def:b3:groups:normal}{subgrupo normal} isomorfo a $H$ e $\{e\} \times K$
é um subgrupo isomorfo a $K$; eles intersectam-se trivialmente e
geram o grupo. Reciprocamente, se $G =
NK$ com $N \trianglelefteq G$, $K \leq G$ e $N \cap K = \{e\}$,
então $G \cong N \rtimes_\varphi K$ para $\varphi(k) = (n \mapsto
knk^{-1})$.
\end{proposition}

\begin{proof}
Verificação direta: a associatividade reduz-se a $\varphi(kk') =
\varphi(k)\circ\varphi(k')$ e ao fato de
cada $\varphi(k)$ ser um morfismo; o elemento neutro é $(e,e)$ e $(h,k)^{-1} =
\bigl(\varphi(k^{-1})(h^{-1}), k^{-1}\bigr)$. A
projeção $(h,k)
\mapsto k$ é um morfismo sobre $K$ de núcleo $H \times \{e\}$, que é
portanto \omterm{def:b3:groups:normal}{normal}. Para a recíproca: todo $g \in G$ escreve-se de maneira
\emph{única} como $nk$ com $n \in N$, $k \in K$ (existência: $G =
NK$;
unicidade: $nk = n'k'$ dá $n'^{-1}n = k'k^{-1} \in N \cap
K$), e
\[
(nk)(n'k') = n\,(kn'k^{-1})\;kk'
\]
mostra que $nk \mapsto (n, k)$ transporta a lei de $G$ para a de $N \rtimes_\varphi K$.
\end{proof}

\begin{example}\label{ex:b3:groups:semidirectexamples}
(a) O \emph{grupo diedral}\index{grupo diedral} $D_n$ ($n \geq
3$) das
$2n$ simetrias de um $n$-ágono regular: as rotações formam um
\omterm{def:b3:groups:normal}{subgrupo normal} de índice $2$, qualquer reflexão gera um
complemento, e conjugar uma rotação por uma reflexão inverte-a: $D_n \cong \Z/n\Z \rtimes_\varphi \Z/2\Z$
com $\varphi(1) = (x
\mapsto -x)$.
(b) O grupo afim de uma reta, $\{x \mapsto ax + b : a \in
K^\times,\, b \in K\} \cong K \rtimes K^\times$: translações normais, homotetias
como complemento.
(c) $S_n \cong A_n \rtimes \Z/2\Z$ (complemento: qualquer transposição).
(d) O \omterm{pb:b3:groups:1}{grupo dos quatérnios} $Q_8$ \emph{não} é um \omterm{def:b3:groups:semidirect}{produto semidireto}
de subgrupos próprios: todo subgrupo não trivial contém $-1$
(\cref{pb:b3:groups:1}), de modo que dois subgrupos próprios nunca se
intersectam trivialmente.
\end{example}

\section{Grupos solúveis; simplicidade de \texorpdfstring{$A_n$}{An}}

\begin{definition}\label{def:b3:groups:derived}
O \emph{comutador}\index{comutador} de $x, y \in G$ é $[x,y]
= xyx^{-1}y^{-1}$; o \emph{subgrupo
derivado}\index{subgrupo derivado} $D(G)$ é o subgrupo gerado por
todos os comutadores. A \emph{série derivada} é $D^0(G) = G$, $D^{i+1}(G) = D(D^i(G))$, e $G$
é \emph{solúvel}\index{grupo solúvel} se $D^n(G) =
\{e\}$ para algum $n$.
\end{definition}

\begin{proposition}\label{prop:b3:groups:derived}
$D(G)$ é \omterm{def:b3:groups:normal}{normal} (na verdade, estável por todo automorfismo),
$G/D(G)$ é abeliano e, para $N \trianglelefteq G$: $G/N$ abeliano $\iff
D(G) \subseteq N$. Além
disso, $G$ é \omterm{def:b3:groups:derived}{solúvel} se, e somente se, existe uma cadeia $G = G_0 \trianglerighteq G_1 \trianglerighteq \dots
\trianglerighteq G_n = \{e\}$ com
cada $G_{i+1} \trianglelefteq
G_i$ e cada quociente $G_i/G_{i+1}$ abeliano. Subgrupos e quocientes de
\omterm{def:b3:groups:derived}{grupos solúveis} são \omterm{def:b3:groups:derived}{solúveis}; reciprocamente, se $N$ e $G/N$ são
\omterm{def:b3:groups:derived}{solúveis}, $G$ também é.
\end{proposition}

\begin{proof}
Um automorfismo $\alpha$ leva $[x,y]$ em $[\alpha x, \alpha y]$: ele permuta os
\omterm{def:b3:groups:derived}{comutadores} e preserva, portanto, o subgrupo que eles geram; as
conjugações são automorfismos, donde a normalidade. Em $G/D(G)$,
$\bar x\bar y \bar x^{-1}\bar y^{-1} = \overline{[x,y]} = \bar e$: o quociente é abeliano. Se $G/N$ é abeliano, então todo
$[x,y] \in
N$, logo $D(G) \subseteq N$; reciprocamente, se $D(G) \subseteq N$, então $G/N$, quociente
do abeliano $G/D(G)$ pelo terceiro teorema de isomorfismo, é
abeliano.

Se $G$ é \omterm{def:b3:groups:derived}{solúvel}, a \omterm{def:b3:groups:derived}{série derivada} é uma tal cadeia. Reciprocamente,
dada uma cadeia, $D^i(G) \subseteq G_i$ por indução: $G_i/G_{i+1}$ abeliano dá $D(G_i) \subseteq G_{i+1}$, logo $D^{i+1}(G) = D(D^i G)
\subseteq D(G_i) \subseteq G_{i+1}$;
portanto $D^n(G) = \{e\}$.

Hereditariedade: $D^i(H) \subseteq D^i(G)$ para $H \leq G$ (indução), e $D^i(G/N) = \pi(D^i(G))$ pois $\pi$
aplica \omterm{def:b3:groups:derived}{comutadores} sobre \omterm{def:b3:groups:derived}{comutadores}; isso dá os enunciados relativos a
subgrupos e quocientes. Extensão: se $D^m(G/N) = \{e\}$, então $D^m(G) \subseteq N$, e $D^n(N) = \{e\}$ dá
$D^{m+n}(G) = D^n(D^m(G)) \subseteq D^n(N)
= \{e\}$.
\end{proof}

\begin{example}\label{ex:b3:groups:solvableexamples}
Os grupos abelianos são \omterm{def:b3:groups:derived}{solúveis}. Os $p$-grupos são \omterm{def:b3:groups:derived}{solúveis}, por
indução sobre a ordem: $Z(G) \neq \{e\}$ e $G/Z(G)$ é um $p$-grupo menor.
$S_3$ e $S_4$ são \omterm{def:b3:groups:derived}{solúveis}: $S_4 \trianglerighteq A_4
\trianglerighteq V \trianglerighteq \{e\}$, onde $V = \{e,
(1\,2)(3\,4), (1\,3)(2\,4), (1\,4)(2\,3)\}$ é o grupo de
Klein das transposições duplas (\omterm{def:b3:groups:normal}{normal} em $S_4$: uma reunião de
\omterm{ex:b3:groups:actions}{classes de conjugação}), com quocientes abelianos $\Z/2\Z$,
$\Z/3\Z$, $V$. No~\cref{ch:b3:galois}, ``a equação geral de grau
$n$ é \omterm{def:b3:groups:derived}{solúvel} por radicais'' \emph{significará} literalmente
``$S_n$ é um \omterm{def:b3:groups:derived}{grupo solúvel}''. Daí a importância da próxima
definição.
\end{example}

\begin{definition}\label{def:b3:groups:simple}
Um grupo $G \neq \{e\}$ é \emph{simples}\index{grupo simples} se seus únicos
\omterm{def:b3:groups:normal}{subgrupos normais} são $\{e\}$ e $G$. Um grupo simples não
abeliano não é \omterm{def:b3:groups:derived}{solúvel}: $D(G) \trianglelefteq G$ não é $\{e\}$ (do contrário $G$
seria abeliano), logo $D(G) = G$ e a \omterm{def:b3:groups:derived}{série derivada} é constante. Os grupos
simples abelianos são exatamente os $\Z/p\Z$, com $p$ primo (um
grupo abeliano é simples se, e somente se, não tem subgrupo próprio não
trivial, se, e somente se, é cíclico de ordem prima, por Lagrange).
\end{definition}

\begin{lemma}\label{lem:b3:groups:threecycles}
Para $n \geq 3$, $A_n$ é gerado pelos $3$-ciclos; para $n \geq 5$, todos
os $3$-ciclos são conjugados \emph{em $A_n$}.
\end{lemma}

\begin{proof}
Um elemento de $A_n$ é um produto de um número par de
transposições; agrupe-as duas a duas e use (compondo da direita para a
esquerda)
\[
(a\,b)(c\,d) = (a\,c\,b)(a\,c\,d), \qquad
(a\,b)(b\,c) = (a\,b\,c), \qquad
(a\,b)(a\,b) = e
\]
para pares disjuntos, com um ponto em comum e iguais, respectivamente:
cada par de transposições é um produto de $3$-ciclos.

Conjugação: $\sigma(a\,b\,c)\sigma^{-1} = (\sigma a\; \sigma b\;
\sigma c)$, de modo que dois $3$-ciclos quaisquer são
conjugados por algum $\sigma \in
S_n$. Se $\sigma$ é ímpar, substitua-o por
$\sigma' = \sigma (d\,e)$, onde $d, e$ são dois pontos fora de $\{a, b, c\}$ --- eles existem
pois $n \geq 5$; então $\sigma'$ é par e $\sigma'(a\,b\,c)
\sigma'^{-1} = \sigma(a\,b\,c)\sigma^{-1}$, já que $(d\,e)$ comuta com
$(a\,b\,c)$.
\end{proof}

\begin{theorem}[Simplicidade do grupo alternado]\label{thm:b3:groups:ansimple}
$A_n$ é \omterm{def:b3:groups:simple}{simples} para $n \geq 5$.\index{grupo alternado}
\end{theorem}

\begin{proof}
Sejam $N \trianglelefteq A_n$, $N \neq \{e\}$. Pelo~\cref{lem:b3:groups:threecycles}, basta mostrar que $N$
contém \emph{um} $3$-ciclo: a normalidade e a conjugação dos
$3$-ciclos em $A_n$ colocam então todos os $3$-ciclos em
$N$, logo $N = A_n$.

Para $\rho \in S_n$, seja $F(\rho) = \{x : \rho(x) \neq x\}$ seu \emph{suporte} e $f(\rho) = \abs{F(\rho)}$. Escolha $\sigma \in N
\setminus \{e\}$ com
$f(\sigma)$ \emph{mínimo}. Note que uma permutação par não trivial tem $f \geq 3$,
e que $f(\sigma) =
4$ é impossível para $\sigma \in A_n$ a menos que $\sigma$ seja uma
transposição dupla (um $4$-ciclo é ímpar). Mostremos que
$\sigma$ é um $3$-ciclo.

\emph{Caso A: $\sigma$ é um produto de transposições disjuntas},
digamos $\sigma = (a\,b)(c\,d)\cdots$ com $f(\sigma) \geq 4$. Tome $e'
\notin \{a, b, c, d\}$ (é possível: $n \geq 5$), ponha $\tau =
(c\,d\,e')$ e
\[
\sigma' = \tau\sigma\tau^{-1}\,\sigma^{-1} \in N
\qquad (\tau\sigma\tau^{-1} \in N \text{ por normalidade}).
\]
Como $\sigma\tau^{-1}\sigma^{-1} = (\sigma c\;\sigma e'\;\sigma
d) = (d\;\sigma e'\;c)$ (usando $\sigma c = d$, $\sigma d = c$), obtemos $\sigma' = (c\,d\,e')(d\;\sigma e'\;c)$.

Se $\sigma e' = e'$ (o que ocorre em particular quando $f(\sigma) =
4$, isto é, $\sigma = (a\,b)(c\,d)$):
então $(d\,e'\,c) =
(c\,d\,e')$ e $\sigma' = (c\,d\,e')^2 = (c\,e'\,d)$, um $3$-ciclo pertencente a $N$, com $f(\sigma') = 3 < 4 \leq f(\sigma)$
--- o que contradiz a minimalidade.

Se $\sigma e' \neq e'$: então $\sigma e' \notin \{a, b, c, d,
e'\}$ ($\sigma$ troca $a,b$ e $c,d$, e
$e' \notin \{a,b,c,d\}$ com $\sigma$ injetora), de modo que $\sigma$ move os seis
pontos $a, b,
c, d, e', \sigma e'$: $f(\sigma) \geq 6$. Por outro lado, $\sigma'$, um produto de dois
$3$-ciclos com suportes em $\{c, d,
e', \sigma e'\}$, satisfaz $f(\sigma') \leq 4$; e $\sigma' \neq
e$, pois
$\sigma'(d) = \tau\sigma\tau^{-1}(c) = \tau\sigma(e') =
\sigma e' \neq d$ ($\tau$ fixa $\sigma e' \notin \{c,d,e'\}$). Assim $\sigma' \in N \setminus\{e\}$ com $f(\sigma') \leq 4 <
f(\sigma)$: a minimalidade é
contradita.

\emph{Caso B: algum ciclo de $\sigma$ tem comprimento
$\geq 3$}, digamos $\sigma(a) = b$, $\sigma(b) = c$ com $a, b, c$ distintos. Se
$\sigma$ é exatamente esse $3$-ciclo, terminamos. Caso
contrário, $f(\sigma) \geq 5$ (o caso $f(\sigma) = 4$ com um ciclo de comprimento
$\geq3$ é o $4$-ciclo ímpar, excluído), de modo que podemos
escolher $d, e' \in
F(\sigma) \setminus \{a, b, c\}$. Ponha $\tau = (c\,d\,e')$ e $\sigma' = \tau\sigma\tau^{-1}\sigma^{-1} \in N$. Como antes, $\sigma' = (c\,d\,e')\,(\sigma c\;\sigma e'\;\sigma d)$ move apenas
pontos de
\[
M = \{c, d, e'\} \cup \{\sigma c, \sigma d, \sigma e'\}
\subseteq F(\sigma)
\]
(as imagens de pontos movidos são movidas: $\sigma(x) \ne x$ implica $\sigma(\sigma x) \neq \sigma x$, sendo
$\sigma$ injetora). Além disso, $b \notin M$: os cinco pontos $a, b, c, d, e'$ são
distintos, logo $b
\notin \{c, d, e'\}$; e $b \in \{\sigma c, \sigma d, \sigma
e'\}$ forçaria $a \in \{c, d, e'\}$ (aplique $\sigma^{-1}$, usando
$\sigma a = b$), o que é falso. Portanto $\sigma'$ fixa $b$, ao passo que
$\sigma$ move $b$; e $F(\sigma') \subseteq F(\sigma)$. Por fim, $\sigma' \neq e$: $\sigma^{-1}(c) = b$, $\tau^{-1}(b) = b$,
$\sigma(b) = c$, $\tau(c) = d$, logo $\sigma'(c) = d \neq c$. Assim $\sigma' \in N
\setminus \{e\}$ com $f(\sigma') \leq f(\sigma) - 1$, contradizendo a
minimalidade.

Sendo ambos os casos impossíveis, $\sigma$ é um $3$-ciclo.
\end{proof}

\begin{corollary}\label{cor:b3:groups:snnotsolvable}
Para $n \geq 5$: $A_n$ e $S_n$ não são \omterm{def:b3:groups:derived}{solúveis}, e os únicos
\omterm{def:b3:groups:normal}{subgrupos normais} de $S_n$ são $\{e\}$, $A_n$ e $S_n$.
\end{corollary}

\begin{proof}
$A_n$ é \omterm{def:b3:groups:simple}{simples} não abeliano e, portanto, não é \omterm{def:b3:groups:derived}{solúvel}
(\cref{def:b3:groups:simple}); um grupo que contém um subgrupo não \omterm{def:b3:groups:derived}{solúvel} não é
\omterm{def:b3:groups:derived}{solúvel} (\cref{prop:b3:groups:derived}). Seja $N
\trianglelefteq S_n$: então $N \cap A_n \trianglelefteq A_n$ vale $\{e\}$ ou
$A_n$. Se $N \cap A_n = A_n$, então $A_n \subseteq N$ e $N \in \{A_n, S_n\}$ pelo índice. Se $N \cap A_n = \{e\}$, a
restrição a $N$ da projeção $S_n \to S_n/A_n \cong
\Z/2\Z$ é injetora, logo $\abs N \leq 2$; se $N = \{e, \sigma\}$,
a normalidade faz com que a \omterm{ex:b3:groups:actions}{classe de conjugação} de $\sigma$ seja
igual a $\{\sigma\}$, isto é, $\sigma \in Z(S_n)$. Mas $Z(S_n) = \{e\}$ para $n \geq 3$: se $\sigma \ne e$ leva
$a$ em $b \neq a$, escolha $c
\notin \{a, b\}$; então $(b\,c)\sigma(b\,c)^{-1}$ leva $a$ em $c
\ne b$, de
modo que difere de $\sigma$. Logo $N = \{e\}$.
\end{proof}

\begin{theorem}[Jordan--Hölder]\label{thm:b3:groups:jordanholder}
Todo grupo finito $G \neq \{e\}$ admite uma \emph{série de
composição}\index{série de composição}
\[
\{e\} = G_0 \trianglelefteq G_1 \trianglelefteq \cdots
\trianglelefteq G_r = G,
\qquad G_{i}/G_{i-1} \text{ simples},
\]
e o multiconjunto dos \emph{fatores de composição} $G_i/G_{i-1}$, a menos de
isomorfismo, não depende da série escolhida. Um grupo finito é \omterm{def:b3:groups:derived}{solúvel}
se, e somente se, todos os seus fatores de composição são cíclicos de
ordem prima.\index{teorema de Jordan--Hölder}
\end{theorem}

\begin{proof}
\emph{Existência}: indução sobre $\abs G$. Se $G$ é \omterm{def:b3:groups:simple}{simples},
tome $\{e\} \trianglelefteq G$. Caso contrário, escolha um \omterm{def:b3:groups:normal}{subgrupo normal} próprio maximal
$N$ (há uma quantidade finita de subgrupos); $G/N$ é \omterm{def:b3:groups:simple}{simples}
pelo teorema da correspondência (um \omterm{def:b3:groups:normal}{subgrupo normal} próprio não trivial
de $G/N$ se levantaria a um \omterm{def:b3:groups:normal}{subgrupo normal} de $G$ estritamente
entre $N$ e $G$). Acrescente $N \trianglelefteq G$ a uma \omterm{thm:b3:groups:jordanholder}{série de composição} de
$N$.

\emph{Unicidade}: indução sobre $\abs G$, sendo claro o caso
$G$ \omterm{def:b3:groups:simple}{simples}. Tome duas \omterm{thm:b3:groups:jordanholder}{séries de composição}, de penúltimos termos
$M \trianglelefteq G$ e $N \trianglelefteq G$ (de modo que $G/M$ e $G/N$ são \omterm{def:b3:groups:simple}{simples}). Se
$M = N$, conclua por indução aplicada a $M$. Caso contrário,
$MN$, \omterm{def:b3:groups:normal}{normal} em $G$ e contendo estritamente $M$, vale
$G$ ($M$ é \omterm{def:b3:groups:normal}{normal} maximal: qualquer $M \subsetneq L
\subsetneq G$ \omterm{def:b3:groups:normal}{normal} se aplicaria
sobre um \omterm{def:b3:groups:normal}{subgrupo normal} próprio não trivial do \omterm{def:b3:groups:simple}{grupo simples}
$G/M$). O segundo teorema de isomorfismo dá
\[
G/M = MN/M \cong N/(M \cap N),
\qquad
G/N = MN/N \cong M/(M \cap N).
\]
Ponha $K = M \cap N$ ($\trianglelefteq G$) e fixe uma \omterm{thm:b3:groups:jordanholder}{série de composição} de $K$. Então
$M$ carrega duas \omterm{thm:b3:groups:jordanholder}{séries de composição}: a sua original e a série de
$K$ seguida de $K \trianglelefteq
M$ (o quociente $M/K \cong G/N$ é \omterm{def:b3:groups:simple}{simples}). Por indução
(aplicada a $M$), os fatores da série original de $M$ são $\{\text{fatores de } K\} \cup \{G/N\}$;
o mesmo vale para $N$. Logo as duas séries de $G$ têm por
fatores
\[
\{\text{fatores de } K\} \;\cup\; \{\,G/N,\; G/M\,\},
\]
o mesmo multiconjunto.

\omterm{def:b3:groups:derived}{Solubilidade}: se todos os fatores são $\Z/p_i\Z$, a série é uma cadeia de
quocientes abelianos, logo $G$ é \omterm{def:b3:groups:derived}{solúvel} (\cref{prop:b3:groups:derived}).
Reciprocamente, um fator de composição de um \omterm{def:b3:groups:derived}{grupo solúvel} é \omterm{def:b3:groups:derived}{solúvel}
(quociente de um subgrupo) e \omterm{def:b3:groups:simple}{simples}; um \omterm{def:b3:groups:simple}{grupo simples} \omterm{def:b3:groups:derived}{solúvel} é
abeliano ($D(G) \ne G$ força $D(G) = \{e\}$), logo é algum $\Z/p\Z$.
\end{proof}

\begin{remark}\label{rem:b3:groups:jhmeaning}
Jordan--Hölder diz que todo grupo finito é construído a partir de
\omterm{def:b3:groups:simple}{grupos simples}, com uma lista de peças bem definida --- uma aritmética
dos grupos em que os \omterm{def:b3:groups:simple}{grupos simples} são os primos e em que \emph{o modo
como as peças são coladas} (os dados de extensão, como no \omterm{def:b3:groups:semidirect}{produto
semidireto}) substitui a mera multiplicação. A classificação dos \omterm{def:b3:groups:simple}{grupos
simples} finitos --- os cíclicos $\Z/p\Z$, os alternados $A_{n \geq
5}$,
dezesseis famílias de tipo de Lie e $26$ grupos esporádicos --- é
um dos monumentos da matemática do século XX; sua demonstração,
espalhada por cerca de dez mil páginas de revistas, está muito além
deste curso.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:groups:1}
(a) Mostre que todo subgrupo de índice $2$ é \omterm{def:b3:groups:normal}{normal}.
(b) Mostre que, se $[G : H] = 2$, então $x^2 \in H$ para todo $x \in
G$.
(c) Deduza que $A_4$ não tem subgrupo de ordem $6$: a recíproca
de Lagrange é falsa. \emph{(Conte os quadrados dos $3$-ciclos.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:groups:2}
Mostre que, se $G/Z(G)$ é cíclico, então $G$ é abeliano. Deduza
novamente que todo grupo de ordem $p^2$ é abeliano e exiba, para
cada primo $p$, um grupo não abeliano de ordem $p^3$.
\emph{(Pense nas matrizes triangulares superiores com diagonal unitária
sobre $\mathbb F_p$.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:groups:3}
(a) Mostre que $\operatorname{Aut}(\Z/n\Z) \cong (\Z/n\Z)^\times$.
(b) Mostre que os automorfismos internos $\iota_g \colon x \mapsto
gxg^{-1}$ formam um \omterm{def:b3:groups:normal}{subgrupo
normal} $\operatorname{Inn}(G)
\trianglelefteq \operatorname{Aut}(G)$, com $\operatorname{Inn}(G) \cong G/Z(G)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:4}
Sejam $H, K$ subgrupos de um grupo finito $G$.
(a) Demonstre a \emph{fórmula do produto} $\abs{HK}\,\abs{H\cap K} =
\abs H\, \abs K$, contando as fibras da
aplicação $H \times K
\to HK$, $(h,k) \mapsto hk$.
(b) Mostre que $HK$ é um subgrupo se, e somente se, $HK = KH$
(automático quando um dos dois é \omterm{def:b3:groups:normal}{normal}).
(c) Se $H, K \trianglelefteq G$ e $H \cap K = \{e\}$, mostre que $hk = kh$ para todos $h \in H$, $k \in K$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:5}
(Lema de contagem de Burnside) Um grupo finito $G$ age sobre um
conjunto finito $X$. Mostre que o número de \omterm{def:b3:groups:action}{órbitas} é o
\emph{número médio de pontos fixos}:
\[
\#\{\text{órbitas}\} = \frac{1}{\abs G}\sum_{g \in G}
\abs{\operatorname{Fix}(g)},
\qquad \operatorname{Fix}(g) = \{x \in X : g \cdot x = x\},
\]
contando o conjunto $\{(g,x) : g\cdot x = x\}$ de duas maneiras. Aplicação: $\Z/p\Z$
($p$ primo) age por rotação sobre colares de $p$ contas com
$a$ cores disponíveis; deduza o pequeno teorema de Fermat $a^p \equiv a \pmod p$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:groups:6}
Usando os teoremas de Sylow, mostre que todo grupo de ordem $15$ é
cíclico e que todo grupo de ordem $45$ é abeliano.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:7}
Mostre que nenhum grupo de ordem $30$, e nenhum de ordem $56$,
é \omterm{def:b3:groups:simple}{simples}. \emph{(Para $30$: se $n_3 \neq 1$ e $n_5 \neq 1$, conte os elementos
de ordens $3$ e $5$. Para $56$: conte os elementos de ordem
$7$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:8}
(a) Seja $H \leq G$ de índice $n$. Mostre que a \omterm{def:b3:groups:action}{ação} de $G$ sobre
$G/H$ fornece um morfismo $G \to S_n$ cujo núcleo $\bigcap_{g \in
G} gHg^{-1}$ é o maior
\omterm{def:b3:groups:normal}{subgrupo normal} de $G$ contido em $H$.
(b) Deduza: se $G$ é finito e $p$ é o \emph{menor} divisor
primo de $\abs G$, todo subgrupo de índice $p$ é \omterm{def:b3:groups:normal}{normal}.
(c) Mostre que toda \omterm{def:b3:groups:action}{ação} transitiva de $G$ sobre um conjunto
$X$ é isomorfa à \omterm{def:b3:groups:action}{ação} sobre um espaço de classes laterais: existe
uma bijeção $X
\to G/G_x$ que comuta com as ações.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:9}
(a) Mostre que $D(G)$ é o menor \omterm{def:b3:groups:normal}{subgrupo normal} de $G$ com
quociente abeliano, e que todo morfismo de $G$ em um grupo abeliano
se fatora de maneira única pela \emph{abelianização} $G^{\mathrm{ab}} = G/D(G)$.
(b) Calcule $D(S_n)$ e $S_n^{\mathrm{ab}}$ para $n \geq 2$, bem como $D(Q_8)$ e
$Q_8^{\mathrm{ab}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:10}
Sejam $G$ um $p$-grupo e $H \subsetneq G$ um subgrupo próprio. Mostre que
$H \subsetneq N_G(H)$ (``os \omterm{ex:b3:groups:actions}{normalizadores} crescem'') e deduza que todo subgrupo
maximal de um $p$-grupo é \omterm{def:b3:groups:normal}{normal} de índice $p$.
\emph{(Indução sobre $\abs G$, usando $Z(G) \neq \{e\}$: trate separadamente
$Z(G) \subseteq H$ e $Z(G) \not\subseteq H$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:groups:11}
(Simplicidade de $A_5$, na prática)
(a) Mostre que as \omterm{ex:b3:groups:actions}{classes de conjugação} de $A_5$ têm cardinalidades
$1$, $15$, $20$, $12$, $12$. Preste atenção à
cisão da $S_5$-classe dos $5$-ciclos: para um $5$-ciclo
$\sigma$, compare os \omterm{ex:b3:groups:actions}{centralizadores} de $\sigma$ em $S_5$ e
em $A_5$.
(b) Deduza que $A_5$ é \omterm{def:b3:groups:simple}{simples}: um \omterm{def:b3:groups:normal}{subgrupo normal} é uma reunião de
\omterm{ex:b3:groups:actions}{classes de conjugação}, contém $e$ e tem cardinalidade que divide
$60$.
(c) Mostre que um \omterm{def:b3:groups:simple}{grupo simples} de ordem $60$ tem necessariamente
$n_5 =
6$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:groups:12}
(Os \omterm{ex:b3:groups:actions}{normalizadores} dos \omterm{def:b3:groups:sylow}{subgrupos de Sylow} são autonormalizantes) Sejam
$P$ um $p$-subgrupo de Sylow de um grupo finito $G$ e
$H =
N_G(P)$.
(a) Mostre que $P$ é o \emph{único} $p$-subgrupo de Sylow de
$H$.
(b) Deduza $N_G(H) = H$. \emph{(Para $g \in N_G(H)$: $gPg^{-1}$ é um $p$-subgrupo de
Sylow de $H$, logo $gPg^{-1} = P$.)}
(c) Conclua que nenhum \omterm{ex:b3:groups:actions}{normalizador} de \omterm{def:b3:groups:sylow}{subgrupo de Sylow} está contido
em um \omterm{def:b3:groups:normal}{subgrupo normal} próprio de $G$, e que um subgrupo maximal
contendo $N_G(P)$ é autonormalizante.
\end{exercise}

\section{Problema: os grupos de ordem no máximo 15}

\begin{problem}[{Problema de fim de semana --- classificação dos grupos
pequenos}]\label{pb:b3:groups:1}
O objetivo é uma classificação completa, com demonstrações integrais,
dos grupos de ordem $\leq 15$ a menos de isomorfismo. As ordens $1, 2, 3, 5,
7, 11, 13$ são
resolvidas por Lagrange (cíclicos), e as ordens $4$ e $9$
pelo~\cref{thm:b3:groups:pfixed} somado à análise de $p^2$ feita abaixo: restam
$6, 8, 10, 12, 14, 15$.

\medskip
\noindent\textbf{Parte I --- Ferramentas.}
\begin{enumerate}
  \item Mostre que um grupo em que todo elemento satisfaz $x^2 =
        e$ é
        abeliano; deduza que um tal grupo finito tem ordem $2^k$ e
        é isomorfo a $(\Z/2\Z)^k$. \emph{(Veja-o como um espaço vetorial sobre
        $\mathbb F_2$.)}
  \item Mostre que um grupo de ordem $p^2$ é isomorfo a $\Z/p^2\Z$ ou
        $(\Z/p\Z)^2$. Liste os grupos abelianos de ordem $8$ a menos de
        isomorfismo: $\Z/8\Z$, $\Z/4\Z \times
        \Z/2\Z$, $(\Z/2\Z)^3$ --- demonstre que a
        lista é completa e sem repetições \emph{sem} usar o teorema de
        estrutura do~\cref{ch:b3:modules} (discuta segundo a ordem máxima de
        um elemento).
  \item Sejam $\varphi, \varphi' \colon K \to \operatorname{Aut}(H)$ duas ações. Mostre que, se $\varphi' = \varphi \circ
        \alpha$ com $\alpha \in \operatorname{Aut}(K)$, então
        $H
        \rtimes_{\varphi} K \cong H \rtimes_{\varphi'} K$.
  \item Determine $\operatorname{Aut}(\Z/n\Z)$ para $n = 3, 4, 5,
        7$ explicitamente e mostre que $\operatorname{Aut}\bigl((\Z/2\Z)^2
        \bigr) \cong S_3$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Ordens $2p$ ($6$, $10$,
$14$) e $pq$.}
\begin{enumerate}[resume]
  \item Seja $\abs G = 2p$ com $p$ primo ímpar. Mostre que $G$ tem um
        \omterm{def:b3:groups:normal}{subgrupo normal} $N = \langle r \rangle$ de ordem $p$ e um elemento $s$ de
        ordem $2$ fora de $N$.
  \item Deduza $G \cong \Z/p\Z \rtimes_\varphi \Z/2\Z$, onde $\varphi(1) \in \operatorname{Aut}(\Z/p\Z)$ é uma involução, e conclua: $G \cong \Z/2p\Z$ ou
        $G \cong
        D_p$; verifique que esses dois grupos não são isomorfos. Isso
        resolve as ordens $6$, $10$, $14$.
  \item Mais geralmente, seja $\abs G = pq$ com $p < q$ primos. Mostre que,
        se $p \nmid q-1$, então $G$ é cíclico (\cref{ex:b3:groups:pq}), e que, se
        $p \mid q - 1$, existe, além de $\Z/pq\Z$, \emph{exatamente um} grupo não
        abeliano $\Z/q\Z \rtimes \Z/p\Z$ a menos de isomorfismo --- use a questão 3 e o
        fato de que $\operatorname{Aut}(\Z/q\Z) \cong
        (\Z/q\Z)^\times$ é cíclico de ordem $q - 1$, admitido aqui
        e demonstrado no~\cref{ch:b3:galois} (ciclicidade de $\mathbb F_q^\times$). Conclua
        para a ordem $15$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Ordem $8$.} Seja $G$ não
abeliano de ordem $8$.
\begin{enumerate}[resume]
  \item Mostre que $G$ tem um elemento $r$ de ordem $4$
        (use a questão 1) e que $N = \langle r\rangle$ é \omterm{def:b3:groups:normal}{normal}.
  \item Seja $s \notin N$. Mostre que $s^2 \in N$ (\cref{exo:b3:groups:1}(b)), que $srs^{-1} = r^{-1}$
        (examine as imagens possíveis de $r$ por conjugação, que
        devem ter ordem 4, e exclua $srs^{-1} = r$), e que $s^2 \in \{e, r^2\}$ \emph{(o que
        acontece se $s^2 = r$ ou $r^3$? e por que $s^2$ deve
        comutar com $s$?)}.
  \item No caso $s^2 = e$, mostre que $G \cong D_4$.
  \item No caso $s^2 = r^2$, mostre que a tabela de multiplicação fica
        inteiramente determinada; o grupo resultante é o \emph{grupo
        dos quatérnios}\index{grupo dos quatérnios} $Q_8 =
        \{\pm 1, \pm \mathrm i, \pm \mathrm j, \pm \mathrm k\}$, $\mathrm i^2 = \mathrm j^2 = \mathrm k^2 = \mathrm i\,
        \mathrm j\,\mathrm k = -1$
        (ponha $r = \mathrm i$, $s =
        \mathrm j$). Verifique que $Q_8$ existe, por
        exemplo dentro de $GL_2(\C)$ via
        \[
        \mathrm i \mapsto \begin{pmatrix} \iu & 0\\ 0 & -\iu
        \end{pmatrix},
        \qquad
        \mathrm j \mapsto \begin{pmatrix} 0 & 1\\ -1 & 0
        \end{pmatrix}.
        \]
  \item Mostre que todo subgrupo não trivial de $Q_8$ contém
        $-1$; deduza que todo subgrupo de $Q_8$ é \omterm{def:b3:groups:normal}{normal}, que
        $D_4 \not\cong Q_8$ (conte os elementos de ordem $2$) e que $Q_8$
        não é um \omterm{def:b3:groups:semidirect}{produto semidireto} de dois subgrupos próprios.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Ordem $12$.} Sejam $\abs G = 12$, $P_3
\in \mathrm{Syl}_3(G)$,
$P_2 \in \mathrm{Syl}_2(G)$.
\begin{enumerate}[resume]
  \item Mostre que $n_3 \in \{1, 4\}$, $n_2 \in \{1, 3\}$, e que $n_3
        = 4$ força $n_2 = 1$
        \emph{(conte os elementos de ordem $3$)}.
  \item Suponha $n_3 = 4$. A \omterm{def:b3:groups:action}{ação} por conjugação sobre $\mathrm{Syl}_3$ dá $\rho \colon G \to S_4$.
        Mostre que $\ker \rho$, contido em todo $N_G(P_3)$ e portanto de ordem
        divisora de $3$, é trivial \emph{(por que ele não pode ter
        ordem $3$?)}; que a imagem, subgrupo de ordem $12$ de
        $S_4$, é necessariamente $A_4$ \emph{(os subgrupos de
        índice 2 são normais e contêm todos os quadrados ---
        \cref{exo:b3:groups:1}; conte os quadrados em $S_4$)}; e conclua
        que $G \cong A_4$.
  \item Suponha $n_3 = 1$, de modo que $G \cong \Z/3\Z \rtimes_\varphi P_2$ com $\varphi \colon P_2 \to \operatorname{Aut}(\Z/3\Z)
        \cong \Z/2\Z$. Enumere os
        casos: $\varphi$ trivial dá $\Z/12\Z$ e $\Z/6\Z \times \Z/2\Z$; $P_2 =
        \Z/4\Z$ com $\varphi$
        sobrejetor dá o \emph{grupo diciclico} $\mathrm{Dic}_3 = \Z/3\Z \rtimes
        \Z/4\Z$; $P_2 = (\Z/2\Z)^2$ com
        $\varphi$ sobrejetor dá, a menos da equivalência da questão 3, um
        único grupo --- mostre que ele é $D_6$, por exemplo
        exibindo um elemento de ordem $6$ e uma involução do tipo
        reflexão.
  \item Mostre que $\Z/12\Z$, $\Z/6\Z \times \Z/2\Z$, $D_6$, $A_4$, $\mathrm{Dic}_3$ são
        dois a dois não isomorfos \emph{(conte os elementos de ordem
        $2$, ou use $n_3$)}. Isso resolve a ordem $12$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Síntese.}
\begin{enumerate}[resume]
  \item Monte a tabela de classificação: para cada ordem $n \leq
        15$, a
        lista completa dos grupos a menos de isomorfismo, com as
        contagens $1, 1, 1, 2, 1, 2, 1, 5, 2, 2, 1, 5, 1, 2, 1$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Além: os grupos de ordem $p^3$, com
$p$ ímpar.} A análise da ordem $8$ da Parte III tem um belo
análogo para primos ímpares, com um fenômeno genuinamente novo. Sejam
$p$ um primo ímpar e $G$ não abeliano de ordem $p^3$.
\begin{enumerate}[resume]
  \item Mostre que $\abs{Z(G)} = p$, que $G/Z(G) \cong
        (\Z/p\Z)^2$ \emph{(um quociente cíclico pelo
        \omterm{ex:b3:groups:actions}{centro} força a abelianidade: \cref{exo:b3:groups:2})} e que $D(G) =
        Z(G)$
        \emph{(para $D(G) \subseteq Z(G)$, use que $G/Z(G)$ é abeliano; para a
        igualdade, $G$ é não abeliano e $D(G) \neq \{e\}$)}. Deduza que todo
        \omterm{def:b3:groups:derived}{comutador} $[x, y] = xyx^{-1}y^{-1}$ é central e de ordem divisora de $p$.
  \item (A identidade-chave) Sejam $x, y \in G$ e $z = [y, x]$, central. Demonstre
        por indução sobre $k$:
        \[
        (xy)^k = x^k y^k z^{k(k-1)/2} .
        \]
        \emph{(Faça cada $y$ atravessar cada $x$; cada
        cruzamento custa um fator central $z$.)}
  \item Deduza que, para $p$ ímpar, a aplicação $\theta\colon x \mapsto
        x^p$ é um
        morfismo de grupos de $G$ em $Z(G)$ \emph{(por que
        $x^p$ é central? por que $z^{p(p-1)/2} = e$ exige $p$ ímpar?)}, e
        conclua que $G$ tem expoente $p$ ou $p^2$, os
        dois casos sendo distinguidos pela trivialidade ou não de
        $\theta$.
  \item (Expoente $p$) Suponha que todo elemento satisfaça $x^p = e$.
        Escolha $x, y$ cujas classes geram $G/Z(G)$ e ponha
        $z =
        [y, x]$. Mostre que $z \neq e$, que todo elemento de $G$ se
        escreve de maneira única como $x^ay^bz^c$ ($0 \leq a, b, c < p$) e que a
        multiplicação fica inteiramente determinada pelas relações
        $x^p = y^p = z^p = e$, $z$ central, $[y, x] =
        z$. Verifique que o \emph{grupo
        de Heisenberg}
        \[
        H_p = \left\{
        \begin{pmatrix} 1 & a & c\\ 0 & 1 & b\\ 0 & 0 & 1
        \end{pmatrix} : a, b, c \in \mathbb F_p \right\}
        \subseteq GL_3(\mathbb F_p)
        \]
        realiza essas relações e tem expoente $p$ (calcule $(I + N)^p$
        com $N$ estritamente triangular superior, usando $N^3 = 0$ e
        $p \geq 3$): todo grupo não abeliano de ordem $p^3$ e expoente
        $p$ é isomorfo a $H_p$.
  \item (Expoente $p^2$) Suponha que algum $r \in G$ tenha ordem
        $p^2$ e ponha $N = \langle r\rangle$, \omterm{def:b3:groups:normal}{normal} (índice $p$:
        \cref{exo:b3:groups:10}). Mostre que existe $s \notin N$ com $s^p = e$
        \emph{(tome qualquer $t \notin N$; usando a questão 20, corrija-o:
        $\theta(t) = t^p \in Z(G) \subseteq N$ --- justifique $Z(G) = \langle r^p\rangle$ --- e escolha $a$ com $s = tr^{a}$
        satisfazendo $s^p = e$; onde se usa que $p$ é ímpar?)}.
        Mostre que $srs^{-1} = r^{1+p}$ a menos de substituir $s$ por uma
        potência, e conclua: existe exatamente \emph{um} grupo não
        abeliano de ordem $p^3$ e expoente $p^2$, a saber,
        $\Z/p^2\Z \rtimes_\varphi \Z/p\Z$ com $\varphi(1)\colon r \mapsto r^{1+p}$ \emph{(use a questão 3; $\operatorname{Aut}(\Z/p^2\Z)$ é cíclico de
        ordem $p(p-1)$, admitido aqui, de modo que tem um único
        subgrupo de ordem $p$)}.
  \item Conclua a contagem: para $p$ ímpar há exatamente $5$
        grupos de ordem $p^3$ (três abelianos, dois não
        abelianos), tal como para $p = 2$ --- mas os dois não
        abelianos já não são $D_4$ e $Q_8$. Localize
        exatamente onde o argumento para $p$ ímpar falha quando
        $p = 2$: na identidade da questão 19, $z^{k(k-1)/2}$ para $k = p = 2$
        vale $z^1 \neq
        e$, de modo que elevar ao quadrado não é um morfismo
        --- e, de fato, $Q_8$ tem um único elemento de ordem
        $2$, ao passo que $D_4$, de expoente $4$, tem
        cinco.
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- Complementos.}
\begin{enumerate}[resume]
  \item Para $p$ ímpar, conte os elementos de ordem $p$ em
        cada um dos dois grupos não abelianos de ordem $p^3$:
        mostre que $H_p$ tem exatamente $p^3 - 1$ deles, enquanto
        $M_p =
        \Z/p^2\Z \rtimes \Z/p\Z$ tem exatamente $p^2 - 1$ \emph{(use o morfismo
        $\theta$ da questão 20: identifique sua imagem e, em
        seguida, a ordem de seu núcleo)}. Verifique numericamente para
        $p = 3$: $26$ contra $8$. Explique por que
        nenhum argumento desse tipo, baseado no morfismo de elevar ao
        quadrado, pode separar $D_4$ de $Q_8$, e qual
        contagem os separa.
  \item Diga que um inteiro $n \geq 1$ é \emph{cíclico} se todo grupo de
        ordem $n$ é cíclico. Mostre que, se $p^2 \mid n$ para algum
        primo $p$, ou se $n$ tem divisores primos $p < q$
        com $p \mid q - 1$, então $n$ não é cíclico \emph{(em cada caso,
        exiba um grupo não cíclico de ordem $n$, usando a Parte II
        no segundo)}. Deduza que $n$ cíclico força $\gcd(n, \varphi(n)) = 1$, onde
        $\varphi$ é a função totiente de Euler, e confira com a tabela da
        questão 17: entre $n \leq 15$, as ordens que comportam um único
        grupo são exatamente $n \in \{1, 2, 3, 5, 7, 11, 13, 15\}$, precisamente aquelas com $\gcd(n, \varphi(n)) = 1$.
\end{enumerate}
\end{problem}
