\chapter{Subvariedades de \texorpdfstring{$\R^n$}{Rn}}
\label{ch:b3:submanifolds}

Esferas, toros, grupos de rotações: os habitats naturais da geometria e
da mecânica não são espaços vetoriais, mas \emph{conjuntos curvos que,
de perto, parecem planos}. Este capítulo dá a essa frase um sentido
preciso --- as subvariedades de $\R^n$ --- e o cálculo para trabalhar
sobre elas. O alicerce é o teorema da função inversa, aqui demonstrado
pelo ponto fixo de Banach; todo o resto é mudança de coordenadas: as
quatro descrições equivalentes de uma subvariedade (retificação local,
conjuntos de nível, gráficos, parametrizações), os \omterm{def:b3:submanifolds:tangent}{espaços tangentes} e a
otimização com restrições por \omterm{thm:b3:submanifolds:lagrange}{multiplicadores de Lagrange} --- que, como
demonstração de despedida, redemonstra o teorema espectral das matrizes
simétricas em três linhas de geometria. O problema de fim de semana
constrói o grupo das rotações $SO(3)$ e seu recobrimento duplo
quaterniônico: a álgebra (o $Q_8$ do~\cref{ch:b3:groups}, crescido)
encontrando a geometria.

\section{O teorema da função inversa}

\begin{theorem}[Teorema da função inversa]
\label{thm:b3:submanifolds:ift}
Sejam $U \subseteq \R^n$ aberto, $f \colon U \to \R^n$ de classe $\mathcal C^1$ e $a \in U$ com $Df(a)$ inversível.
Então existem abertos $V \ni a$, $W \ni f(a)$ tais que $f \colon V \to W$ é uma bijeção com inversa
$\mathcal C^1$, e\index{teorema da função inversa}
\[
D(f^{-1})(y) = \bigl(Df(f^{-1}(y))\bigr)^{-1}
\qquad (y \in W).
\]
Se $f$ é $\mathcal C^k$, então $f^{-1}$ também é.
\end{theorem}

\begin{proof}
Normalize: substituindo $f$ por $x \mapsto Df(a)^{-1}\bigl(f(a +
x) - f(a)\bigr)$, podemos supor $a = 0$, $f(0) = 0$,
$Df(0) =
I$ (o enunciado geral decorre compondo com as bijeções afins).
Escreva $f(x) = x + g(x)$: $Dg(0) = 0$ e, pela \omterm{def:b3:topology:continuity}{continuidade} de $Dg$, escolha $r > 0$
com $\vertiii{Dg(x)} \leq \frac12$ em $\bar B(0, r)$; a desigualdade do valor médio dá $\norm{g(x) - g(x')} \leq \frac12
\norm{x - x'}$ ali.

\emph{Bijetividade sobre uma \omterm{def:b3:topology:topology}{vizinhança}.} Para $y \in B(0,
\frac r2)$, resolver $f(x) = y$
significa encontrar um ponto fixo de $\Phi_y(x) = y - g(x)$; $\Phi_y$ leva $\bar B(0, r)$ nele
mesmo ($\norm{\Phi_y(x)} \leq \norm y + \frac12\norm x \leq
r$) e é $\frac12$-lipschitziana: Banach (\cref{thm:b3:complete:banach}) dá uma única
solução $x = \varphi(y) \in \bar B(0,r)$. Além disso, $f$ é injetora em $\bar B(0,r)$:
\[
\norm{f(x) - f(x')} \geq \norm{x - x'} - \norm{g(x) - g(x')}
\geq \tfrac12\norm{x - x'} .
\tag{$*$}
\]
Ponha $W = B(0, \frac r2)$ e $V = f^{-1}(W)\cap B(0, r)$: abertos (\omterm{def:b3:topology:continuity}{continuidade}), com $f \colon V \to W$ bijetora.

\emph{\omterm{def:b3:topology:continuity}{Continuidade} e derivabilidade da inversa.} $(*)$ diz que $\varphi = f^{-1}$ é
$2$-lipschitziana. Fixe $y_0 = f(x_0)
\in W$; a inversibilidade de $A = Df(x_0)$ (sua distância
a $I$ é $\leq \frac12$: Neumann, \cref{prop:b3:banach:neumann}) e a derivabilidade de $f$
dão, para $y = f(x)$ perto de $y_0$:
\[
\varphi(y) - \varphi(y_0) - A^{-1}(y - y_0)
= -A^{-1}\bigl(f(x) - f(x_0) - A(x - x_0)\bigr)
= -A^{-1}\,o(\norm{x - x_0}) = o(\norm{y - y_0}),
\]
usando $(*)$ para converter $\norm{x - x_0} \leq 2\norm{y - y_0}$: $\varphi$ é derivável em $y_0$ com a
diferencial inversa. \omterm{def:b3:topology:continuity}{Continuidade} de $y \mapsto D\varphi(y) =
Df(\varphi(y))^{-1}$: composição de \omterm{def:b3:topology:continuity}{aplicações
contínuas} (a inversão é \omterm{def:b3:topology:continuity}{contínua}, \cref{prop:b3:banach:neumann}): $\varphi \in \mathcal C^1$; e repetir a
mesma fórmula (bootstrapping) dá $\mathcal C^k$.
\end{proof}

\begin{theorem}[Teorema da função implícita]
\label{thm:b3:submanifolds:implicit}
Seja $F \colon U \subseteq \R^p\times\R^q \to \R^q$ de classe $\mathcal C^1$ perto de $(a, b)$, com $F(a,b) = 0$, e suponha que a
diferencial parcial $D_yF(a,b) \in \mathcal L(\R^q)$ seja inversível. Então existem \omterm{def:b3:topology:topology}{vizinhanças}
$A \ni a$, $B \ni b$ e uma aplicação $\psi \colon A \to B$ de classe $\mathcal C^1$ com
\[
\bigl\{(x, y)\in A\times B : F(x,y) = 0\bigr\}
= \{(x, \psi(x)) : x \in A\},
\]
e $D\psi(x) = -D_yF(x, \psi(x))^{-1}\,D_xF(x,
\psi(x))$.\index{teorema da função implícita}
\end{theorem}

\begin{proof}
Aplique o~\cref{thm:b3:submanifolds:ift} a $\Theta(x, y) = (x,
F(x,y))$: sua diferencial em $(a,b)$, triangular
por blocos com blocos diagonais inversíveis $I$ e $D_yF$, é
inversível. A inversa local tem a forma $\Theta^{-1}(x, z) = (x, h(x,
z))$; ponha $\psi(x) = h(x, 0)$: então $F(x, y) = 0$ se,
e somente se, $\Theta(x,y) = (x, 0)$, se, e somente se, $y = \psi(x)$, localmente. A fórmula:
derive $F(x, \psi(x)) = 0$ pela regra da cadeia.
\end{proof}

\section{Subvariedades: quatro definições}

\begin{theorem}[Caracterizações equivalentes]
\label{thm:b3:submanifolds:characterizations}
Sejam $M \subseteq \R^n$, $d \in \{0, \dots, n\}$ e $k \geq
1$. São equivalentes, em cada ponto $a \in M$ (e $M$
é uma \emph{subvariedade de dimensão $d$ e classe
$\mathcal C^k$}\index{subvariedade} se valem em todo $a
\in M$):
\begin{enumerate}
  \item (\emph{Retificação}) Existe um difeomorfismo $\Phi$ $\mathcal C^k$ de
        um aberto $\Omega \ni a$ sobre um aberto $\Omega' \subseteq \R^n$ com
        \[
        \Phi(M\cap\Omega) = \Omega' \cap
        \bigl(\R^d\times\{0\}\bigr).
        \]
  \item (\emph{Conjunto de nível}) Existe uma \emph{submersão} $F \colon \Omega \to \R^{n-d}$ de
        classe $\mathcal C^k$ (isto é, com $DF(x)$ sobrejetora) em um aberto
        $\Omega \ni a$ com $M\cap\Omega = F^{-1}(0)$.
  \item (\emph{Gráfico}) A menos de permutar coordenadas, $M$ é
        localmente o gráfico de uma aplicação $\psi
        \colon A \subseteq \R^d \to \R^{n-d}$ de classe $\mathcal C^k$.
  \item (\emph{Parametrização}) Existe uma \emph{imersão} $\varphi \colon A \subseteq \R^d \to
        \R^n$ de
        classe $\mathcal C^k$ (com $D\varphi(u)$ injetora) com $A$ aberto, sendo $\varphi$
        um \omterm{def:b3:topology:continuity}{homeomorfismo} de $A$ sobre $M \cap
        \Omega$ para algum aberto $\Omega \ni a$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1)$\Rightarrow$(2): $F = (\Phi_{d+1}, \dots, \Phi_n)$ (as últimas coordenadas de $\Phi$): uma submersão
($D\Phi$ é inversível).
(2)$\Rightarrow$(3): $DF(a)$ é sobrejetora: algum menor $q \times q$ da jacobiana é
inversível ($q = n - d$); após permutar coordenadas, $D_yF(a)$ é inversível, e o
teorema da função implícita (\cref{thm:b3:submanifolds:implicit}) exprime $M$ localmente
como gráfico $y = \psi(x)$.
(3)$\Rightarrow$(4): $\varphi(x) = (x, \psi(x))$: uma imersão (diferencial $\bigl(\begin{smallmatrix}I\\
D\psi\end{smallmatrix}\bigr)$ injetora), um
\omterm{def:b3:topology:continuity}{homeomorfismo} sobre o gráfico (inversa: a projeção, \omterm{def:b3:topology:continuity}{contínua}).
(4)$\Rightarrow$(1): seja $\varphi(u_0) = a$; complete $\operatorname{im}D\varphi(u_0)$ por um suplementar $E$ ($\dim E = n - d$)
e defina $\Theta(u, v) = \varphi(u) + v$ em $A\times E$: $D\Theta(u_0, 0)$ é bijetora (a imagem contém $\operatorname{im}D\varphi(u_0)$ e
$E$), de modo que $\Theta$ é um difeomorfismo local
(\cref{thm:b3:submanifolds:ift}); sua inversa $\Phi$ retifica: perto de $a$, os
pontos de $M$ são exatamente os $\varphi(u) = \Theta(u, 0)$ --- e, para isso, a hipótese
de \omterm{def:b3:topology:continuity}{homeomorfismo} em (4) garante que $M\cap\Omega$, para $\Omega$ pequeno,
não contém outras folhas ($\varphi(u') + v = m \in
M$ perto de $a$ com $v \neq 0$ pequeno deve
ser excluído: $m = \varphi(u'')$ para algum $u''$ perto de $u_0$ pela
propriedade de \omterm{def:b3:topology:continuity}{homeomorfismo}, e a injetividade local de $\Theta$
força $v = 0$). Então $\Phi(M\cap\Omega) = (A\times\{0\})
\cap\Phi(\Omega)$, a menos de encolher.
\end{proof}

\begin{example}\label{ex:b3:submanifolds:examples}
A esfera $S^{n-1} = \{\norm x^2 = 1\}$: conjunto de nível da submersão $F(x) = \norm x_2^2 - 1$ em $\R^n\setminus\{0\}$ ($DF(x) = 2x^{\mathsf T} \ne 0$): uma
subvariedade $\mathcal C^\infty$ de dimensão $n - 1$. O toro em $\R^3$:
conjunto de nível de $\bigl(\sqrt{x^2 + y^2} -
R\bigr)^2 + z^2 - r^2$ ($0 < r < R$). O cone $\{x^2 + y^2 =
z^2\}$ \emph{não} é uma
subvariedade em $0$ (\cref{exo:b3:submanifolds:1}). Grupos de matrizes: $SL_n$
e $O_n$ são subvariedades de $M_n(\R)$ (\cref{exo:b3:submanifolds:5,exo:b3:submanifolds:6}) --- o ponto de
partida da teoria de Lie.
\end{example}

\section{Espaços tangentes}

\begin{definition}\label{def:b3:submanifolds:tangent}
Sejam $M$ uma $d$-subvariedade e $a \in M$. O \emph{espaço
tangente}\index{espaço tangente} $T_aM$ é o conjunto dos vetores
velocidade $\gamma'(0)$ das curvas $\gamma \colon
\intoo{-\varepsilon}\varepsilon \to M$ de classe $\mathcal C^1$ com $\gamma(0) = a$.
\end{definition}

\begin{proposition}\label{prop:b3:submanifolds:tangent}
$T_aM$ é um subespaço vetorial de $\R^n$ de dimensão $d$, e:
\begin{enumerate}
  \item se $M = F^{-1}(0)$ localmente com $F$ submersão: $T_aM = \ker DF(a)$;
  \item se $M$ é parametrizada pela imersão $\varphi$ ($\varphi(u_0) = a$): $T_aM =
        \operatorname{im}D\varphi(u_0)$.
\end{enumerate}
\end{proposition}

\begin{proof}
As curvas em $M$ satisfazem $F(\gamma(t)) = 0$; a regra da cadeia em $0$ dá
$DF(a)\gamma'(0) = 0$: $T_aM \subseteq \ker DF(a)$. Reciprocamente, a retificação (\cref{thm:b3:submanifolds:%
characterizations}(1))
transporta retas de $\R^d\times\{0\}$ em curvas de $M$: todo vetor de um
subespaço de dimensão $d$ é realizado; comparar dimensões ($\dim\ker DF(a) = n - (n - d) = d$)
força a igualdade em (1), e o mesmo argumento de transporte dá (2)
($D\varphi(u_0)$ aplicada a retas em $A$; dimensões, de novo).
\end{proof}

\begin{theorem}[Multiplicadores de Lagrange]
\label{thm:b3:submanifolds:lagrange}
Sejam $M = F^{-1}(0)$ com $F = (F_1, \dots, F_q) \colon \Omega
\to \R^q$ uma submersão $\mathcal C^1$, e seja $f \colon
\Omega \to \R$ de classe $\mathcal C^1$.
Se a restrição $f\restriction_M$ tem extremo local em $a \in M$, então existem reais
únicos $\lambda_1, \dots, \lambda_q$ (os \emph{\omterm{thm:b3:submanifolds:lagrange}{multiplicadores de Lagrange}})
com\index{multiplicadores de Lagrange}
\[
\nabla f(a) = \lambda_1\nabla F_1(a) + \dots +
\lambda_q\nabla F_q(a) .
\]
\end{theorem}

\begin{proof}
Para toda curva $\gamma$ em $M$ por $a$: $t \mapsto
f(\gamma(t))$ tem extremo
local em $0$, de modo que $0 =
\frac{\dd}{\dd t}f(\gamma(t))\big|_0 = \langle\nabla f(a),
\gamma'(0)\rangle$: $\nabla f(a) \perp T_aM = \ker DF(a)$ (\cref{prop:b3:submanifolds:tangent}). Ora, $\ker DF(a)^\perp
= \operatorname{im}DF(a)^{\mathsf T}$
(a identidade de posto/ortogonalidade do segundo ano, ou
o~\cref{exo:b3:hilbert:8} em dimensão finita: $(\ker T)^\perp = \operatorname{im}T^*$), que é gerado pelos
gradientes $\nabla F_i(a)$ --- independentes, pois $DF(a)$ é sobrejetora: os
multiplicadores existem e são únicos.
\end{proof}

\begin{example}[O teorema espectral, geometricamente]
\label{ex:b3:submanifolds:rayleigh}
Seja $A$ uma matriz real simétrica $n\times n$ e maximize $f(x) = \langle Ax, x\rangle$ na esfera
$S^{n-1}$ (compacta: o máximo é atingido, em algum $v_1$). Lagrange com
$F(x) = \norm x^2 - 1$: $\nabla f = 2Ax$ e $\nabla F =
2x$ dão $Av_1 = \lambda_1v_1$ --- um autovetor, com $\lambda_1 = \max_{S^{n-1}}\langle Ax, x\rangle$. Restrinja
$A$ a $v_1^\perp$ (invariante: $\langle Av, v_1\rangle =
\langle v, Av_1\rangle = \lambda_1\langle v, v_1\rangle = 0$) e itere: uma base ortonormal de
autovetores. O teorema espectral do segundo ano, redemonstrado por pura
otimização --- e a sombra em dimensão infinita do mesmo argumento
demonstrou o~\cref{lem:b3:spectral:existence}.
\end{example}

\begin{method}\label{met:b3:submanifolds:method}
Para demonstrar que um conjunto é subvariedade: exiba-o localmente como
$F^{-1}(0)$ com $DF$ sobrejetora \emph{no conjunto} (a rota mais comum),
ou como gráfico. Para calcular sua dimensão e seu \omterm{def:b3:submanifolds:tangent}{espaço tangente}:
$d = n - q$ e $T_a = \ker DF(a)$. Para otimizar nele: Lagrange --- verifique sempre antes a
\omterm{def:b3:topology:compact}{compacidade} (ou a coercividade), de modo que exista um extremo ao qual o
teorema possa se aplicar, e lembre que a equação dos multiplicadores é
apenas \emph{necessária}: reúna todos os pontos críticos e, depois,
compare valores. Para grupos de matrizes, derive curvas na identidade
para identificar \omterm{def:b3:submanifolds:tangent}{espaços tangentes}.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:submanifolds:1}
(a) Verifique que os conjuntos seguintes são subvariedades $\mathcal C^\infty$ e dê
suas dimensões: $S^{n-1}$; o hiperboloide $\{x^2 + y^2 - z^2 = 1\}$; o toro do~\cref{ex:b3:submanifolds:examples}.
(b) Mostre que o cone $C = \{x^2 + y^2 = z^2\} \subseteq
\R^3$ não é uma $2$-subvariedade em $0$:
determine o número de \omterm{def:b3:topology:components}{componentes conexas} de $\bigl(C\setminus\{0\}\bigr)\cap B(0,\varepsilon)$ e compare com o
número que uma retificação (\cref{thm:b3:submanifolds:characterizations}(1)) forçaria para um plano
menos um ponto.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:submanifolds:2}
Calcule os \omterm{def:b3:submanifolds:tangent}{espaços tangentes}:
(a) $T_aS^{n-1}$ para $a$ qualquer (resposta: $a^\perp$);
(b) o plano tangente ao toro do~\cref{ex:b3:submanifolds:examples} em um ponto arbitrário do
equador externo $\{z = 0,\ x^2 + y^2 = (R + r)^2\}$;
(c) a reta tangente à hélice $\varphi(t) = (\cos t,
\sin t, t)$ em $\varphi(t_0)$, verificando
a~\cref{prop:b3:submanifolds:tangent}(2).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:3}
Seja $f(x, y) = (x^2 - y^2,\ 2xy)$ (isto é, $z \mapsto z^2$).
(a) Em que pontos o~\cref{thm:b3:submanifolds:ift} se aplica?
(b) Mostre que $f$ é local mas não globalmente inversível em $\R^2\setminus\{0\}$, e
exiba explicitamente as duas inversas locais definidas em uma \omterm{def:b3:topology:topology}{vizinhança}
de $(1, 0)$ (os dois ramos da raiz quadrada).
(c) Mesma discussão para a aplicação de \omterm{ex:b3:product:polar}{coordenadas polares} $(r,
\theta) \mapsto (r\cos\theta, r\sin\theta)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:4}
(Fólio) Sejam $F(x,y) = x^3 + y^3 - 3xy$ e $\mathcal C =
F^{-1}(0)$.
(a) Mostre que, perto de todo ponto de $\mathcal C$ distinto da origem, $\mathcal C$ é
uma $1$-subvariedade, localmente um gráfico em $x$ ou em $y$
(qual, e onde?).
(b) Calcule a reta tangente em $(\frac32, \frac32)$.
(c) O que acontece na origem? (Dois ramos se cruzam: exiba duas curvas
$\mathcal C^1$ em $\mathcal C$ por $0$ com velocidades independentes e conclua que
não existe retificação.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:5}
Seja $F(M) = M^{\mathsf T}M$ de $M_n(\R)$ no espaço $S_n$ das matrizes simétricas.
(a) Mostre que $DF(M)(H) = M^{\mathsf T}H + H^{\mathsf T}M$ e que $DF(M)$ é sobrejetora sobre $S_n$ em
todo $M \in O_n$ \emph{(dado $S \in S_n$, tente $H = \frac12MS$)}.
(b) Conclua que $O_n = F^{-1}(I)$ é uma subvariedade $\mathcal C^\infty$ compacta de dimensão
$\frac{n(n-1)}2$, com $T_IO_n = \{H : H^{\mathsf T} = -H\}$, as matrizes antissimétricas.
(c) Mostre que $\eu^{tH} \in O_n$ para toda $H$ antissimétrica: as direções
tangentes se integram em curvas dentro do grupo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:6}
(a) Mostre que $\det \colon M_n(\R) \to \R$ tem diferencial $D\det(M)(H) = \operatorname{tr}\bigl(\operatorname{com}(M)
^{\mathsf T}H\bigr)$, não nula em todo $M \in SL_n$.
(b) Conclua que $SL_n(\R)$ é uma subvariedade de dimensão $n^2 - 1$ com $T_ISL_n = \{H : \operatorname{tr}H = 0\}$.
(c) $GL_n(\R)$ é uma subvariedade? De que dimensão?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:7}
Por \omterm{thm:b3:submanifolds:lagrange}{multiplicadores de Lagrange}:
(a) encontre os extremos de $f(x,y) = xy$ no círculo $x^2 +
y^2 = 1$;
(b) mostre que, entre todos os vetores de probabilidade $(p_1, \dots,
p_n)$ (positivos
e de soma $1$), a entropia $-\sum
p_i\ln p_i$ é maximizada exatamente na
distribuição uniforme;
(c) encontre o ponto da elipse $\{x^2/4 + y^2 = 1\}$ mais próximo de $(1, 0)$ e
verifique geometricamente a equação dos multiplicadores (alinhamento das
normais).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:submanifolds:8}
Escreva o~\cref{ex:b3:submanifolds:rayleigh} por extenso: demonstre por indução que uma
matriz real simétrica admite base ortonormal de autovetores, sendo
$\lambda_1 \geq \dots \geq \lambda_n$ os máximos com restrição sucessivos do quociente de Rayleigh.
Deduza então as fórmulas de Courant--Fischer do~\cref{exo:b3:spectral:8} em
dimensão finita diretamente dessa construção.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:submanifolds:9}
(Desigualdade de Hadamard) Para $M \in GL_n(\R)$ com colunas $c_1, \dots, c_n$:
\[
\abs{\det M} \;\leq\; \prod_{i=1}^n\norm{c_i}_2 ,
\]
com igualdade se, e somente se, as colunas são ortogonais.
\emph{(Reduza a colunas de norma $1$ por reescalamento; maximize
$\det$ no produto \omterm{def:b3:topology:compact}{compacto} de esferas $(S^{n-1})^n$; em um maximizador,
Lagrange em cada coluna separadamente dá $\nabla_{c_i}\det =
\lambda_ic_i$, e $\nabla_{c_i}\det$ é a $i$-ésima
coluna de $\operatorname{com}(M)$: deduza que $M^{\mathsf T}M$ é diagonal, logo $= I$, logo
$\det = \pm1$.)} Leitura geométrica: o volume de um paralelepípedo é no máximo o
produto dos comprimentos de suas arestas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:10}
Perto de quais de seus pontos o círculo $S^1$ é um gráfico $y =
\psi(x)$? E
um gráfico $x = \chi(y)$? Verifique explicitamente a caracterização por gráficos
(\cref{thm:b3:submanifolds:%
characterizations}(3)) em $(1, 0)$ e explique em uma frase por que
\emph{alguma} permutação de coordenadas é sempre suficiente, mas nenhuma
única serve sempre.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:11}
(O grupo ortogonal como subvariedade, quantitativamente)
(a) Mostre que $O_n = \{M : M^{\mathsf T}M = I\}$ é \omterm{def:b3:topology:compact}{compacto}: limitado (cada coluna é um vetor
unitário, de modo que $\norm M \leq
\sqrt n$ para a norma euclidiana de matrizes) e
fechado.
(b) Mostre que seu \omterm{def:b3:submanifolds:tangent}{espaço tangente} em $I$ é o espaço das matrizes
antissimétricas, de dimensão $\frac{n(n-1)}2$, e em um $A \in O_n$ geral: $T_AO_n = \{AK : K^{\mathsf T} =
-K\}$.
(c) Deduza que a aplicação $t \mapsto A\exp(tK)$ é, para cada $K$ antissimétrica, uma
curva em $O_n$ por $A$ com velocidade $AK$ \emph{(verifique
$\exp(tK) \in O_n$ usando $\exp(X)^{\mathsf T} = \exp(X^{\mathsf T})$ e $\exp(-X)\exp(X) = I$)}: todo vetor tangente é realizado por uma
curva explícita, sem precisar do teorema da função implícita.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:12}
(Pontos críticos da distância) Sejam $M \subseteq \R^n$ uma subvariedade e $p \notin M$.
Mostre que, se $x_0 \in M$ minimiza a distância a $p$ (um tal ponto existe
quando $M$ é fechada e não vazia --- por quê?), então
\[
p - x_0 \;\perp\; T_{x_0}M
\]
\emph{(derive $t \mapsto \norm{\gamma(t) - p}^2$ ao longo de curvas em $M$)}. Deduza: o ponto
mais próximo em uma esfera está na semirreta que passa pelo \omterm{ex:b3:groups:actions}{centro}; e
use a condição para calcular a distância de $p = (2, 0)$ à parábola $y = x^2$
(reduza a uma cúbica e resolva-a numericamente com três algarismos).
\end{exercise}

\section{Problema: \texorpdfstring{$SO(3)$}{SO(3)} e os
quatérnios}

\begin{problem}[{Problema de fim de semana --- rotações, o grupo
$S^3$ e o recobrimento duplo}]\label{pb:b3:submanifolds:1}
Os quatérnios $\mathbb H = \{t + x\mathrm i + y\mathrm j +
z\mathrm k\}$ --- a álgebra cujo grupo de unidades contém o
$Q_8$ do~\cref{pb:b3:groups:1} --- parametrizam as rotações
tridimensionais duas vezes: a aplicação ``conjugar por um quatérnio
unitário'' é um morfismo sobrejetor $S^3 \to
SO(3)$ de núcleo $\{\pm1\}$. Construímos
tudo. Recorde/defina: a multiplicação é $\R$-bilinear com $\mathrm i^2 = \mathrm j^2 = \mathrm k^2 =
\mathrm{ijk} = -1$; o
conjugado de $q = t + x\mathrm i +
y\mathrm j + z\mathrm k$ é $\bar q = t - x\mathrm i -
y\mathrm j - z\mathrm k$; $N(q) = q\bar q = t^2 + x^2 + y^2
+ z^2$.

\medskip
\noindent\textbf{Parte I --- A álgebra $\mathbb H$ e o grupo $S^3$.}
\begin{enumerate}
  \item Verifique que $\mathbb H$ é uma $\R$-álgebra associativa de \omterm{ex:b3:groups:actions}{centro}
        $\R$, que $\overline{pq} =
        \bar q\,\bar p$ e que $N(pq) = N(p)N(q)$ \emph{(uma rota limpa:
        represente $q$ pela matriz complexa $2\times2$ $\bigl(\begin{smallmatrix}\alpha & \beta\\
        -\bar\beta & \bar\alpha\end{smallmatrix}\bigr)$, $q =
        \alpha + \beta\mathrm j$, e use
        $\det$)}.
  \item Deduza que todo $q \neq 0$ é inversível ($q^{-1} =
        \bar q/N(q)$): $\mathbb H$ é um corpo (não
        comutativo), e $S^3 = \{N(q) = 1\}$ é um grupo --- e uma $3$-subvariedade
        compacta de $\R^4$ (\cref{ex:b3:submanifolds:examples}).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O morfismo de rotação.} Identifique
$\R^3$ com os \emph{quatérnios puros} $P = \{x\mathrm i +
y\mathrm j + z\mathrm k\}$ e, para $q \in S^3$, defina
$\rho_q(v) = q\,v\,\bar q$.
\begin{enumerate}[resume]
  \item Mostre que $\rho_q$ leva $P$ em $P$ \emph{(os
        quatérnios puros são aqueles com $\bar v = -v$)}, é $\R$-linear,
        preserva a norma, e que $\rho
        \colon q \mapsto \rho_q$ é um morfismo de grupos $S^3
        \to O(3)$.
  \item Calcule o núcleo: $\rho_q = \mathrm{id}$ se, e somente se, $q$ comuta com
        $\mathrm i, \mathrm j, \mathrm k$, se, e somente se, $q \in \R\cap S^3 = \{\pm1\}$.
  \item Escreva $q = \cos\frac\theta2 + \sin\frac\theta2\,u$ com $u \in P$, $N(u) = 1$ (por que isso é sempre possível
        para $q \in S^3$?). Mostre que $\rho_q$ fixa $u$ e, no plano
        $u^\perp\cap P$, age como a rotação de ângulo $\theta$ \emph{(calcule
        $\rho_q(w)$ para $w \perp u$ usando $uw = -wu$ para unitários puros ortogonais ---
        demonstre essa identidade a partir da tabela de multiplicação,
        ou de $uw + wu =
        -2\langle u, w\rangle$)}.
  \item Conclua: $\operatorname{im}\rho \subseteq SO(3)$ (cada $\rho_q$ é uma rotação de eixo e ângulo
        como calculados --- determinante $+1$ por \omterm{def:b3:topology:continuity}{continuidade} de
        $q
        \mapsto \det\rho_q$ no \omterm{def:b3:topology:connected}{conexo} $S^3$, ou diretamente), e $\rho$ é
        \emph{sobrejetor} sobre $SO(3)$: toda rotação de $\R^3$
        tem eixo (demonstre: uma matriz ortogonal real $3\times3$ com $\det = 1$
        tem autovalor $1$ --- considere o polinômio característico) e,
        portanto, é algum $\rho_q$. Em resumo:
        \[
        SO(3) \;\cong\; S^3/\{\pm 1\} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- $SO(3)$ como subvariedade;
Rodrigues.}
\begin{enumerate}[resume]
  \item Mostre que $SO(3)$ é uma subvariedade compacta de dimensão
        $3$ de $M_3(\R)$ com $T_ISO(3) =$ as matrizes antissimétricas
        (\cref{exo:b3:submanifolds:5}; a condição sobre o determinante seleciona uma
        reunião de componentes).
  \item Para a matriz antissimétrica $A_u$ associada a $u \in \R^3$ ($A_uv = u\wedge v$,
        o produto vetorial), demonstre a \emph{fórmula de Rodrigues}:
        \[
        \eu^{\theta A_u} = I + \sin\theta\,A_u + (1 -
        \cos\theta)\,A_u^2 \qquad (\norm u = 1)
        \]
        \emph{(a partir de $A_u^3 = -A_u$: separe a série exponencial segundo as
        potências de $A_u$)}, e identifique-a como a rotação de eixo
        $u$ e ângulo $\theta$. Deduza que $\exp$ leva as
        matrizes antissimétricas \emph{sobre} $SO(3)$.
  \item Relacione as duas parametrizações: mostre que $t
        \mapsto \rho_{q(t)}$ com $q(t) = \cos\frac t2 +
        \sin\frac t2\,u$ é
        um grupo a um parâmetro de rotações cuja derivada em $t = 0$
        é $A_u$ --- as exponenciais quaterniônica e matricial contam
        a mesma história, a meia velocidade e a velocidade plena,
        respectivamente.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O recobrimento duplo, sentido.}
\begin{enumerate}[resume]
  \item Mostre que o \omterm{def:b3:topology:pathconnected}{caminho} $q(t) = \cos\frac t2 + \sin\frac
        t2\,\mathrm k$, $t \in \intcc0{2\pi}$, é um laço em $SO(3)$ (sua
        imagem $\rho_{q(t)}$ volta à identidade) cujo levantamento quaterniônico
        \emph{não} é um laço: $q(2\pi) = -q(0)$. Continuar até $t = 4\pi$ fecha o
        levantamento. Explique em um parágrafo curto o que isso diz: uma
        rotação de $2\pi$ não pode ser desfeita \omterm{def:b3:topology:continuity}{continuamente}, mas
        uma de $4\pi$ pode (o truque do cinto), porque os laços de
        $SO(3)$ são detectados em seu recobrimento duplo $S^3$.
  \item Deduza também o dividendo prático: compor rotações =
        multiplicar quatérnios (o equivalente a $4$ multiplicações de
        dados em vez de $9$, sem deriva da ortogonalidade) ---
        verifique na composição de duas rotações de um quarto de volta
        em torno de $\mathrm i$ e de $\mathrm j$: calcule o eixo e o ângulo do
        produto.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- A matriz explícita: Euler--Rodrigues.}
Escreva $q = a + b\mathrm i + c\mathrm j +
d\mathrm k \in S^3$, de modo que $a^2 + b^2 + c^2 + d^2 = 1$.
\begin{enumerate}[resume]
  \item Calcule $\rho_q(\mathrm i)$ por extenso a partir da tabela de multiplicação;
        obtenha em seguida $\rho_q(\mathrm
        j)$ e $\rho_q(\mathrm k)$ pela substituição cíclica $\mathrm i \to \mathrm j \to \mathrm k
        \to \mathrm i$,
        $(b, c, d) \to (c, d, b)$ \emph{(justifique-a: ciclar $\mathrm i, \mathrm j,
        \mathrm k$ se estende a um
        automorfismo de $\mathbb
        H$, pois as relações de definição são
        ciclicamente simétricas)}. Conclua que a matriz de $\rho_q$
        na base $(\mathrm i, \mathrm j, \mathrm k)$ é a \emph{matriz de Euler--Rodrigues}
        \[
        R_q = \begin{pmatrix}
        a^2 + b^2 - c^2 - d^2 & 2(bc - ad) & 2(bd + ac)\\
        2(bc + ad) & a^2 - b^2 + c^2 - d^2 & 2(cd - ab)\\
        2(bd - ac) & 2(cd + ab) & a^2 - b^2 - c^2 + d^2
        \end{pmatrix}.
        \]
  \item (Lendo uma rotação de trás para diante) Mostre que
        \[
        \operatorname{tr}R_q = 4a^2 - 1 = 1 + 2\cos\theta,
        \qquad
        \tfrac12\bigl(R_q - R_q^{\mathsf T}\bigr) =
        \sin\theta\,A_u,
        \]
        na notação das questões 5 e 8. Deduza um algoritmo que recupera
        $\pm q$ a partir de uma matriz de rotação $R$: o ângulo
        pelo traço; o eixo pela parte antissimétrica quando $0 < \theta < \pi$; e,
        quando $\theta = \pi$, demonstre e use a identidade $R
        + I = 2\,uu^{\mathsf T}$.
  \item Avalie $R_q$ para o produto $q =
        \frac12(1 + \mathrm i + \mathrm j + \mathrm k)$ da questão 11: aparece uma
        matriz de permutação. Identifique a rotação e reconcilie com o
        eixo e o ângulo encontrados na questão 11.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Dentro de $S^3$: $SU(2)$, \omterm{ex:b3:groups:actions}{classes
de conjugação}, exponenciais.}
\begin{enumerate}[resume]
  \item Mostre que a \omterm{def:b3:representations:rep}{representação} matricial da questão 1
        (chame-a $\Phi$) se restringe a um isomorfismo de grupos de
        $S^3$ sobre o \emph{grupo especial unitário}
        \[
        SU(2) = \bigl\{U \in M_2(\C) : U^*U = I,\ \det U =
        1\bigr\}
        \]
        \emph{(para a sobrejetividade, escreva as equações $U^{-1} = U^*$ e $\det U = 1$
        para uma matriz complexa $2\times2$ geral)}.
  \item Mostre que a parte real é um invariante de conjugação em
        $S^3$ --- $\operatorname{Re}(pq\bar p) =
        \operatorname{Re}q$ para todo $p \in S^3$ --- e, reciprocamente, que
        dois quatérnios unitários de mesma parte real são conjugados em
        $S^3$ \emph{(reduza a levar um eixo unitário puro sobre
        outro, o que a Parte II fornece)}. Descreva geometricamente as
        \omterm{ex:b3:groups:actions}{classes de conjugação} de $S^3$; traduza para $SU(2)$
        (conjuntos de nível do traço); e projete por $\rho$: duas
        rotações são conjugadas em $SO(3)$ se, e somente se, têm o
        mesmo ângulo $\theta \in \intcc0\pi$.
  \item Defina $\exp$ em $\mathbb H$ pela série exponencial; verifique a
        convergência absoluta, usando $\abs{pq} = \abs p\,\abs q$ para $\abs q =
        \sqrt{N(q)}$. Mostre, para
        $u$ puro unitário e $\theta
        \in \R$,
        \[
        \exp(\theta u) = \cos\theta + \sin\theta\,u ,
        \]
        deduza que $\exp$ leva o hiperplano $P$ sobre $S^3$, e
        verifique que $\rho_{\exp(su)} =
        \eu^{2sA_u}$: de novo o fenômeno do meio ângulo da questão
        9.
  \item Para quatérnios puros $v, w$, demonstre a regra do produto
        $vw = -\langle v, w\rangle + v\wedge w$, e daí a identidade do \omterm{def:b3:groups:derived}{comutador} $vw - wv = 2\,v\wedge w$; demonstre também
        $[A_v, A_w] = A_{v\wedge w}$ para as matrizes da questão 8. Conclua que a derivada de
        $\rho$ em $1$ ao longo das curvas $t \mapsto \exp(tv)$ é o isomorfismo
        linear $v \mapsto 2A_v$ de $P$ sobre as matrizes antissimétricas, e que
        ele transporta o \omterm{def:b3:groups:derived}{comutador} de quatérnios ao \omterm{def:b3:groups:derived}{comutador} de
        matrizes.
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- Estrutura global.}
\begin{enumerate}[resume]
  \item (Não há seção \omterm{def:b3:topology:continuity}{contínua}) Suponha $s \colon SO(3) \to
        S^3$ \omterm{def:b3:topology:continuity}{contínua} com $\rho \circ s =
        \operatorname{id}$. Para o
        laço $R(t) =
        \rho_{q(t)}$ da questão 10, ponha $\varepsilon(t) =
        s(R(t))\,q(t)^{-1}$ para $t \in \intcc0{2\pi}$. Mostre que
        $\varepsilon$ é \omterm{def:b3:topology:continuity}{contínua} com valores em $\{\pm1\}$ e obtenha uma contradição:
        não existe escolha global \omterm{def:b3:topology:continuity}{contínua} de um quatérnio unitário que
        represente cada rotação.
  \item (O modelo da bola) Sejam $\bar B \subseteq \R^3$ a bola fechada de raio
        $\pi$ e $E(v) = \eu^{A_v}$, com $E(0) = I$. Mostre que $E$ leva $\bar B$
        sobre $SO(3)$, é injetora na bola aberta e, na esfera de
        fronteira, identifica exatamente os antípodas: $E(\pi u) = E(-\pi u) = 2uu^{\mathsf T} - I$, sem
        nenhuma outra coincidência. Assim, $SO(3)$ é a bola com os
        pontos antípodas da fronteira colados --- o espaço projetivo
        $\mathbb{RP}^3$ --- e um diâmetro se torna o laço não contrátil da questão
        10.
  \item Mostre que $\rho_p\rho_q\rho_p^{-1} = \rho_{pq\bar
        p}$; que as involuções de $SO(3)$ (os $R \neq
        I$ com
        $R^2 = I$) são exatamente as meias-voltas $\rho_w$ com $w$
        quatérnio puro unitário; e que o \omterm{ex:b3:groups:actions}{centro} de $SO(3)$ é trivial.
  \item Mostre que toda rotação é produto de duas meias-voltas: para
        $q = \cos\frac\theta2 +
        \sin\frac\theta2\,u$, escolha um puro unitário $w \perp
        u$, verifique que $w' = qw$ é de
        novo um quatérnio puro unitário, e confirme $\rho_q =
        \rho_{w'}\rho_w$. Onde ficam os
        dois eixos, e que ângulo eles formam?
  \item Conclua o resumo \omterm{def:b3:topology:topology}{topológico}: $SO(3)$ é \omterm{def:b3:topology:compact}{compacto} e \omterm{def:b3:topology:pathconnected}{conexo por
        caminhos} (dê duas demonstrações: imagem \omterm{def:b3:topology:continuity}{contínua} de $S^3$ por
        $\rho$; imagem de $\exp$), ao passo que $O(3)$ tem
        exatamente duas \omterm{def:b3:topology:components}{componentes conexas}, cada uma \omterm{def:b3:topology:continuity}{homeomorfa} a
        $SO(3)$.

  \item (Uma composição, de três maneiras) Sejam $R_1$ a rotação de
        $\frac\pi2$ em torno do eixo $z$ e $R_2$ a rotação de $\frac\pi2$ em
        torno do eixo $x$. Calcule o eixo e o ângulo de $R_2R_1$:
        (i) multiplicando as duas matrizes $3\times3$ e usando o traço e a
        parte antissimétrica (Parte V); (ii) multiplicando os quatérnios
        unitários correspondentes $q_2q_1$. Verifique que as duas
        respostas coincidem: ângulo $\frac{2\pi}3$, eixo $\frac1{\sqrt3}(1, -1, 1)$.
  \item (A \omterm{ex:b3:conformal:mobius}{transformada de Cayley}) Para $K$ antissimétrica, mostre
        que $I + K$ é inversível e
        \[
        C(K) = (I - K)(I + K)^{-1} \in SO(n),
        \]
        com $-1$ nunca sendo autovalor de $C(K)$; mostre que
        $K \mapsto C(K)$ é uma bijeção das matrizes antissimétricas sobre $\{R \in SO(n) : -1 \notin
        \operatorname{Sp}R\}$, com
        inversa $R \mapsto (I -
        R)(I + R)^{-1}$. (Uma carta racional de $SO(n)$, companheira da
        $\exp$ transcendente do~\cref{exo:b3:submanifolds:11}.)
\end{enumerate}
\end{problem}
