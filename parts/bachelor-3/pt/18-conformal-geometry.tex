\chapter{Aplicações conformes e o teorema da aplicação de Riemann}
\label{ch:b3:conformal}

Uma bijeção \omterm{def:b3:holomorphic:holo}{holomorfa} entre dois domínios transporta toda a análise
complexa de um para o outro: tais aplicações --- \emph{conformes},
porque preservam ângulos --- são os isomorfismos do mundo \omterm{def:b3:holomorphic:holo}{holomorfo}.
Este capítulo as classifica onde a classificação é possível (o disco, o
plano: o lema de Schwarz é a chave, uma desigualdade de poder
espantoso), constrói a teoria de \omterm{def:b3:topology:compact}{compacidade} das famílias \omterm{def:b3:holomorphic:holo}{holomorfas}
(Montel) e demonstra o teorema de existência mais profundo do assunto:
\emph{todo} subdomínio próprio \omterm{def:b3:conformal:simplyconnected}{simplesmente conexo} de $\C$, por mais
recortada que seja sua fronteira, é \omterm{def:b3:conformal:conformal}{conformemente} equivalente ao disco
unitário. Encerramos com as \omterm{prop:b3:conformal:harmonicholo}{funções harmônicas} e o \omterm{thm:b3:conformal:poisson}{núcleo de Poisson},
resolvendo o problema de Dirichlet no disco --- o dividendo analítico da
geometria conforme. Ao longo de todo o capítulo, $\mathbb D = D(0,1)$ e $\mathbb H = \{\operatorname{Im}z >
0\}$.

\section{Aplicações conformes; transformações de Möbius}

\begin{definition}\label{def:b3:conformal:conformal}
Uma \emph{aplicação conforme}\index{aplicação conforme} (ou
biholomorfismo) entre abertos é uma bijeção \omterm{def:b3:holomorphic:holo}{holomorfa}; sua inversa é
automaticamente \omterm{def:b3:holomorphic:holo}{holomorfa} (\cref{cor:b3:residues:openmapping}). Dois domínios são
\emph{conformemente equivalentes} se existe uma tal aplicação; $\operatorname{Aut}(\Omega)$
denota o grupo das autoaplicações conformes. Onde $f' \neq 0$ --- em toda
parte, para $f$ injetora (demonstração do~\cref{cor:b3:residues:openmapping}) --- a
diferencial é a multiplicação por $f'(z) \neq 0$: uma semelhança, de modo que as
aplicações conformes preservam ângulos entre curvas, inclusive a
orientação.
\end{definition}

\begin{example}[Transformações de Möbius]
\label{ex:b3:conformal:mobius}
Para $\bigl(\begin{smallmatrix}a & b\\ c &
d\end{smallmatrix}\bigr) \in GL_2(\C)$, a \emph{transformação de Möbius}\index{transformação de
Möbius} $z \mapsto
\frac{az + b}{cz + d}$ é conforme de $\C\setminus\{-d/c\}$ sobre $\C\setminus\{a/c\}$ (a inversa é do mesmo tipo,
vinda da matriz inversa; a composição corresponde ao produto de
matrizes). A \emph{aplicação de Cayley}\index{transformada de Cayley}
\[
\varphi(z) = \frac{z - \iu}{z + \iu}
\]
leva $\mathbb H$ \omterm{def:b3:conformal:conformal}{conformemente} sobre $\mathbb D$: com efeito, $\abs{z
- \iu} < \abs{z + \iu}$ exatamente quando
$z$ está mais perto de $\iu$ do que de $-\iu$, isto é, $\operatorname{Im}z > 0$; a
inversa é $w \mapsto \iu\frac{1 + w}{1 - w}$. As aplicações de Möbius levam a família de círculos e
retas nela mesma (\cref{exo:b3:conformal:1}).
\end{example}

\begin{example}[A aplicação de Joukowski]\label{ex:b3:conformal:joukowski}
Além de Möbius, a \omterm{def:b3:conformal:conformal}{aplicação conforme} mais útil da matemática aplicada
clássica é
\[
J(z) = \frac12\Bigl(z + \frac1z\Bigr) .
\]
No exterior $\Omega = \{\abs z > 1\}$ do disco unitário, $J$ é injetora: $J(z) = J(w)$ dá $(z - w)(1 -
\frac1{zw}) = 0$ e
$\abs{zw} > 1$. Sua derivada $J'(z) =
\frac12(1 - z^{-2})$ só se anula em $z = \pm1$, na fronteira: $J$ é uma
equivalência conforme de $\Omega$ sobre sua imagem, que é $\C\setminus\intcc{-1}1$ --- o
próprio círculo unitário é dobrado dois a um sobre o segmento ($J(\eu^{\iu\theta}) = \cos\theta$).
Assim, o exterior de um \emph{segmento}, um plano com um corte sem
fronteira suave, é \omterm{def:b3:conformal:conformal}{conformemente} o exterior de um disco: os cantos não
são obstáculo à equivalência conforme, apenas à regularidade na
fronteira. As imagens de círculos que passam por $\pm1$ mas
descentrados são curvas em forma de perfil de asa, e compor $J$ com
aplicações de Möbius transportou o escoamento em torno de um cilindro
--- calculável à mão --- para o escoamento em torno de uma asa: durante
a primeira metade do século XX, este exemplo \emph{era} a aerodinâmica.
Ele é também a porta para Chebyshev: $J$ conjuga $z \mapsto z^n$ ao polinômio de
Chebyshev $T_n$ (\cref{pb:b3:hilbert:1}, Parte V), pois $J(z^n) = \cos(n\theta)$ quando $z = \eu^{\iu\theta}$.
\end{example}

\section{O lema de Schwarz e os grupos de automorfismos}

\begin{theorem}[Lema de Schwarz]\label{thm:b3:conformal:schwarz}
Seja $f \colon \mathbb D \to \mathbb D$ \omterm{def:b3:holomorphic:holo}{holomorfa} com $f(0) = 0$. Então\index{lema de Schwarz}
\[
\abs{f(z)} \leq \abs z \ \ (z \in \mathbb D)
\qquad\text{e}\qquad \abs{f'(0)} \leq 1 ,
\]
e, se $\abs{f(z_0)} = \abs{z_0}$ para um $z_0 \neq 0$, ou se $\abs{f'(0)} = 1$, então $f(z) = \eu^{\iu\theta}z$ é uma rotação.
\end{theorem}

\begin{proof}
$g(z) = f(z)/z$ se estende \omterm{def:b3:holomorphic:holo}{holomorficamente} a $\mathbb D$ (a singularidade em $0$ é
removível: $g$ é limitada perto de $0$, \cref{thm:b3:residues:riemanncw}; seu valor
em $0$ é $f'(0)$). Em $\abs z = r < 1$: $\abs g \leq \frac1r$, de modo que, pelo princípio do
máximo (\cref{thm:b3:holomorphic:maximum}), $\abs g \leq \frac1r$ em $\bar D(0,r)$; faça $r \to 1$: $\abs
g \leq 1$ em $\mathbb D$, o que
dá as duas desigualdades. A igualdade em um ponto \omterm{def:b3:topology:interior}{interior} faz
$\abs g$ atingir um máximo \omterm{def:b3:topology:interior}{interior}: $g$ é constante de módulo
$1$.
\end{proof}

\begin{theorem}[Automorfismos do disco]
\label{thm:b3:conformal:autdisc}
Para $a \in \mathbb D$, o \emph{fator de Blaschke}\index{fator de Blaschke}
\[
\varphi_a(z) = \frac{z - a}{1 - \bar a z}
\]
é um automorfismo de $\mathbb D$ que troca $a$ e $0$, com $\varphi_a^{-1} = \varphi_{-a}$. Todo
automorfismo de $\mathbb D$ é $\eu^{\iu\theta}\varphi_a$ para $\theta
\in \R/2\pi\Z$, $a \in \mathbb D$ únicos.
\end{theorem}

\begin{proof}
Em $\abs z = 1$: $\abs{1 - \bar az} = \abs{\bar z}\abs{1 -
\bar az} = \abs{\bar z - \bar a\abs z^2} = \abs{\bar z - \bar
a} = \abs{z - a}$, de modo que $\abs{\varphi_a} = 1$ ali; pelo princípio do máximo, $\varphi_a(\mathbb D) \subseteq \bar{\mathbb
D}$, e
o \omterm{def:b3:representations:character}{caráter} aberto coloca a imagem em $\mathbb D$. A identidade \omterm{def:b3:galois:algebraic}{algébrica} $\varphi_{-a}\circ\varphi_a = \mathrm{id}$
(cálculo direto) mostra a bijetividade. Sejam agora $f \in
\operatorname{Aut}(\mathbb D)$ e $a = f^{-1}(0)$: $g =
f\circ\varphi_{-a}$ é
um automorfismo que fixa $0$. Schwarz aplicado a $g$ e a
$g^{-1}$: $\abs{g(z)} \leq \abs z$ e $\abs{g^{-1}(w)} \leq \abs w$, de modo que $\abs{g(z)} = \abs z$: uma rotação, $g = \eu^{\iu\theta}\,\mathrm{id}$, isto é,
$f =
\eu^{\iu\theta}\varphi_a$. Unicidade: $a = f^{-1}(0)$ e $\theta$ a partir de uma avaliação do tipo
$f'$ (ou de $f(0) =
-\eu^{\iu\theta}a$ e mais um valor).
\end{proof}

\begin{theorem}[Automorfismos do plano]
\label{thm:b3:conformal:autplane}
$\operatorname{Aut}(\C) = \{z \mapsto az + b : a \in \C^*,\ b
\in \C\}$. Consequentemente, $\C$ e $\mathbb D$ não são \omterm{def:b3:conformal:conformal}{conformemente
equivalentes}.
\end{theorem}

\begin{proof}
Seja $f \in \operatorname{Aut}(\C)$ e considere $g(z) =
f(1/z)$ em $\C^*$: uma \omterm{def:b3:holomorphic:holo}{função holomorfa} com
\omterm{def:b3:residues:singularities}{singularidade isolada} em $0$. Se ela fosse essencial,
Casorati--Weierstrass (\cref{thm:b3:residues:riemanncw}) tornaria $g\bigl(D(0,\varepsilon)\setminus\{0\}\bigr)$ denso, ao passo
que $f(D(0, 1))$ é aberto e disjunto dele ($f$ é injetora: os dois conjuntos
são imagens de conjuntos disjuntos) --- impossível para um \omterm{def:b3:topology:interior}{conjunto
denso} e um aberto não vazio. Logo $0$ é polo ou removível para
$g$, isto é, $\abs{f(z)}$ tem no máximo crescimento polinomial: $f$ é um
polinômio (\cref{exo:b3:holomorphic:4}(a)). A injetividade força o grau $1$: um
polinômio de grau maior tem, em algum ponto, uma raiz múltipla de
$f - c$ ($f'$ se anula) ou várias pré-imagens distintas
(d'Alembert--Gauss, \cref{pb:b3:holomorphic:1}); de um jeito ou de outro, a
injetividade falha. Por fim, uma \omterm{def:b3:conformal:conformal}{aplicação conforme} $\C \to
\mathbb D$ seria uma
função inteira limitada: constante (Liouville) --- não há equivalência.
\end{proof}

\section{O teorema de Montel}

\begin{theorem}[Montel]\label{thm:b3:conformal:montel}
Seja $\mathcal F \subseteq \mathcal H(\Omega)$ \emph{localmente limitada}: todo ponto tem uma \omterm{def:b3:topology:topology}{vizinhança}
sobre a qual $\sup_{f\in
\mathcal F}\sup\abs f < \infty$. Então toda sequência de $\mathcal F$ tem uma subsequência
que converge uniformemente em todos os \omterm{def:b3:topology:compact}{compactos} de $\Omega$ (para
um limite \omterm{def:b3:holomorphic:holo}{holomorfo}).\index{teorema de Montel}
\end{theorem}

\begin{proof}
\omterm{def:b3:complete:equicontinuous}{Equicontinuidade} local: se $\abs f \leq M$ em $D(a, 2r)
\subseteq \Omega$ para todo $f \in \mathcal F$, a fórmula de
Cauchy dá, para $z, z' \in D(a, r)$,
\[
\abs{f(z) - f(z')} = \frac{\abs{z - z'}}{2\pi}
\Bigl|\int_{C_{2r}}\frac{f(w)\,\dd w}{(w-z)(w-z')}\Bigr|
\leq \frac{\abs{z - z'}\,2\pi\cdot2r\,M}{2\pi\,r^2}
= \frac{2M}{r}\,\abs{z - z'} :
\]
uma cota lipschitziana uniforme. Exaura $\Omega$ por \omterm{def:b3:topology:compact}{compactos}
$K_m$; cada $K_m$ é coberto por um número finito de tais discos,
de modo que $\mathcal F$ é uniformemente limitada e \omterm{def:b3:complete:equicontinuous}{equicontínua} em $K_m$:
Arzelà--Ascoli (\cref{thm:b3:complete:ascoli}) extrai uma subsequência que converge
uniformemente em $K_m$; diagonalize em $m$. O limite é \omterm{def:b3:holomorphic:holo}{holomorfo}
pelo~\cref{thm:b3:holomorphic:weierstrassconv}.
\end{proof}

\section{O teorema da aplicação de Riemann}

\begin{definition}\label{def:b3:conformal:simplyconnected}
Um aberto \omterm{def:b3:topology:connected}{conexo} $\Omega \subseteq \C$ é \emph{simplesmente conexo}\index{domínio
simplesmente conexo} (no sentido homológico, suficiente para todos os
nossos propósitos) se $\operatorname{Ind}_\gamma(w) = 0$ para todo ciclo $\gamma$ em $\Omega$
e todo $w \notin \Omega$ --- ``nenhum ciclo de $\Omega$ contorna um buraco''.
Pelo teorema de Cauchy global (\cref{thm:b3:residues:globalcauchy}) e pela~\cref{prop:b3:holomorphic:primitive},
em um tal $\Omega$ \emph{toda \omterm{def:b3:holomorphic:holo}{função holomorfa} tem primitiva}; logo,
toda $f \in \mathcal H(\Omega)$ sem zeros tem logaritmo \omterm{def:b3:holomorphic:holo}{holomorfo} ($\exp\circ$ (primitiva de
$f'/f$), ajustada por uma constante, pois $(f\eu^{-L})' = 0$) e raízes
$n$-ésimas \omterm{def:b3:holomorphic:holo}{holomorfas} $\eu^{L/n}$.\index{logaritmo holomorfo}
\end{definition}

\begin{theorem}[Teorema da aplicação de Riemann]
\label{thm:b3:conformal:rmt}
Todo aberto \omterm{def:b3:conformal:simplyconnected}{simplesmente conexo} $\Omega \subsetneq \C$, com $\Omega
\neq \varnothing$, é \omterm{def:b3:conformal:conformal}{conformemente}
equivalente a $\mathbb D$; dado $z_0 \in \Omega$, existe uma única \omterm{def:b3:conformal:conformal}{aplicação conforme}
$f \colon
\Omega \to \mathbb D$ com $f(z_0) = 0$ e $f'(z_0) >
0$.\index{teorema da aplicação de Riemann}
\end{theorem}

\begin{proof}
\emph{Etapa 0: a família é não vazia.} Tome $b \notin \Omega$: $z - b$ é sem
zeros em $\Omega$, de modo que tem raiz quadrada \omterm{def:b3:holomorphic:holo}{holomorfa} $h$
($h^2 = z - b$). $h$ é injetora ($h(z) =
h(z')$ elevado ao quadrado dá $z = z'$),
e se $w \in h(\Omega)$, então $-w
\notin h(\Omega)$ ($h(z) = -h(z')$ também elevado ao quadrado dá
$z = z'$, o que dá $w = -w = 0$, impossível pois $h$ é sem zeros). Como
$h(\Omega)$ é aberto, ele contém um disco $D(h(z_0), \rho)$; então $D(-h(z_0), \rho) \cap h(\Omega) = \varnothing$, isto é, $\abs{h(z) + h(z_0)} \geq \rho$
para todo $z \in \Omega$. Logo
\[
g(z) = \frac{\rho}{2\,\bigl(h(z) + h(z_0)\bigr)}
\]
é \omterm{def:b3:holomorphic:holo}{holomorfa}, injetora (uma aplicação de Möbius composta com a injetora
$h$), com $\abs g \leq \frac12 < 1$. Compondo com um \omterm{thm:b3:conformal:autdisc}{fator de Blaschke}
(\cref{thm:b3:conformal:autdisc}) para levar $g(z_0)$ a $0$, a família
\[
\mathcal F = \{f \colon \Omega \to \mathbb D \text{
holomorphic, injective, } f(z_0) = 0\}
\]
é não vazia.

\emph{Etapa 1: um elemento extremal.} Seja $s =
\sup_{\mathcal F}\abs{f'(z_0)} \in \intoc0{+\infty}$ ($> 0$: seus membros
são injetores, de modo que $f'(z_0) \neq 0$). Tome $f_n \in
\mathcal F$ com $\abs{f_n'(z_0)} \to s$: a família é
limitada por $1$, de modo que Montel (\cref{thm:b3:conformal:montel}) extrai $f_n \to f$
uniformemente nos \omterm{def:b3:topology:compact}{compactos}; $f$ é \omterm{def:b3:holomorphic:holo}{holomorfa}, $f(z_0) = 0$, $\abs{f'(z_0)} = s$
(\cref{thm:b3:holomorphic:weierstrassconv} para as derivadas) e, em particular, $f$ não é
constante; $f$ é injetora por Hurwitz (\cref{exo:b3:residues:8}(b)), e $f(\Omega) \subseteq \bar{\mathbb D}$,
logo $\subseteq
\mathbb D$ (aplicação aberta). Assim $f \in \mathcal F$ atinge o supremo: $s < \infty$.

\emph{Etapa 2: a aplicação extremal é sobrejetora.} Suponha $a \in
\mathbb D\setminus f(\Omega)$. O
transporte de Blaschke $\varphi_a\circ f$ é sem zeros no \omterm{def:b3:conformal:simplyconnected}{simplesmente conexo}
$\Omega$: ele tem raiz quadrada \omterm{def:b3:holomorphic:holo}{holomorfa} $F$ (com $F(\Omega) \subseteq \mathbb D$, pois
$\abs F^2 =
\abs{\varphi_a\circ f} < 1$), injetora (os quadrados distinguem). Normalize: $G = \varphi_{F(z_0)}\circ F \in
\mathcal F$.
Desfazendo: $f = \varphi_{-a}\circ s_2 \circ
\varphi_{-F(z_0)}\circ G$, em que $s_2(w) = w^2$; a aplicação $\Psi
= \varphi_{-a}\circ s_2\circ\varphi_{-F(z_0)} \colon \mathbb D
\to \mathbb D$ é \omterm{def:b3:holomorphic:holo}{holomorfa} com $\Psi(0) = f(z_0) = 0$
e \emph{não} é uma rotação (não é injetora: $s_2$ não é). Lema de
Schwarz (caso estrito): $\abs{\Psi'(0)} < 1$, e a regra da cadeia $f = \Psi\circ G$ dá $\abs{f'(z_0)} =
\abs{\Psi'(0)}\,\abs{G'(z_0)} < \abs{G'(z_0)}$ ---
contradizendo a maximalidade (note $G \in \mathcal F$). Logo $f$ é sobrejetora:
uma equivalência conforme.

\emph{Etapa 3: normalização e unicidade.} Multiplique $f$ por $\eu^{-\iu\arg f'(z_0)}$
para que $f'(z_0) > 0$ (isso permanece em $\mathcal F$). Se $f_1, f_2$ servem ambas, $\psi =
f_2\circ f_1^{-1} \in \operatorname{Aut}(\mathbb D)$
fixa $0$ com $\psi'(0) = f_2'(z_0)/f_1'(z_0) > 0$; pelo~\cref{thm:b3:conformal:autdisc}, $\psi$ é uma rotação
$\eu^{\iu\theta}$ com $\eu^{\iu\theta} > 0$: $\psi =
\mathrm{id}$.
\end{proof}

\begin{remark}\label{rem:b3:conformal:rmtscope}
O teorema é um enunciado de existência pura, de alcance espantoso: um
quadrado, um semiplano, o complementar de um corte, a região entre dois
círculos tangentes, um domínio de fronteira fractal --- todos
\omterm{def:b3:conformal:conformal}{conformemente} idênticos a $\mathbb D$. O que ele não dá: fórmula alguma (as
aplicações explícitas são a exceção: \cref{exo:b3:conformal:5}), comportamento na
fronteira (uma teoria mais profunda --- o teorema de Carathéodory ---
cuida disso) nem unicidade de extensão a $\C$ ou domínios
multiplamente \omterm{def:b3:topology:connected}{conexos}: o anel $\{1 < \abs z < 2\}$ \emph{não} é \omterm{def:b3:conformal:conformal}{conformemente} um disco
perfurado, e anéis com razões de raios diferentes são inequivalentes
(um fato genuinamente mais difícil).
\end{remark}

\section{Funções harmônicas e o núcleo de Poisson}

\begin{proposition}\label{prop:b3:conformal:harmonicholo}
Sejam $\Omega$ \omterm{def:b3:conformal:simplyconnected}{simplesmente conexo} e $u \colon \Omega \to \R$
harmônica\index{função harmônica} ($\mathcal C^2$ com $\Delta u = u_{xx} + u_{yy} = 0$). Então $u =
\operatorname{Re}F$ para
uma $F$ \omterm{def:b3:holomorphic:holo}{holomorfa}, única a menos de constante imaginária.
Consequentemente, $u$ é $\mathcal C^\infty$, satisfaz a propriedade do valor médio
e obedece ao princípio do máximo (sem extremo \omterm{def:b3:topology:interior}{interior} estrito, salvo se
for constante).
\end{proposition}

\begin{proof}
$g = u_x - \iu u_y$ satisfaz as \omterm{prop:b3:holomorphic:cauchyriemann}{equações de Cauchy--Riemann} ($P = u_x$, $Q = -u_y$: $P_x = u_{xx} = -u_{yy} = Q_y$ e
$P_y = u_{xy} = u_{yx} = -Q_x$) com parciais \omterm{def:b3:topology:continuity}{contínuas}: $g$ é \omterm{def:b3:holomorphic:holo}{holomorfa} (\cref{prop:b3:holomorphic:cauchyriemann}; a
$\R$-derivabilidade decorre de $\mathcal C^1$). Seja $F_0$ uma primitiva
(\omterm{def:b3:topology:connected}{conexidade} simples, \cref{def:b3:conformal:simplyconnected}); então $\operatorname{Re}F_0$ tem gradiente $(u_x, u_y)$
($F_0' = g$ se desdobra exatamente nisso via Cauchy--Riemann para
$F_0$), de modo que $u - \operatorname{Re}F_0$ é constante ($\Omega$ \omterm{def:b3:topology:connected}{conexo}): ajuste
$F = F_0 + c$. As propriedades se transferem do~\cref{thm:b3:holomorphic:maximum} e
do~\cref{exo:b3:holomorphic:10} (para o princípio do máximo aplicado à própria
$u$, use $\eu^{F}$ como lá).
\end{proof}

\begin{theorem}[Fórmula de Poisson; problema de Dirichlet no
disco]\label{thm:b3:conformal:poisson}
Para $0 \leq r < 1$, defina o \emph{núcleo de Poisson}\index{núcleo de Poisson}
\[
P_r(\theta) = \sum_{n\in\Z}r^{\abs n}\eu^{\iu n\theta}
= \frac{1 - r^2}{1 - 2r\cos\theta + r^2} \;>\; 0 .
\]
Seja $g \colon \partial\mathbb D \to \R$ \omterm{def:b3:topology:continuity}{contínua} e ponha, para $z = r\eu^{\iu\varphi} \in \mathbb D$,
\[
u(z) = \frac1{2\pi}\int_0^{2\pi}
P_r(\varphi - t)\,g(\eu^{\iu t})\,\dd t .
\]
Então $u$ é harmônica em $\mathbb D$ e se estende \omterm{def:b3:topology:continuity}{continuamente} a $\bar{\mathbb D}$ com
valores de fronteira $g$: é a única \omterm{prop:b3:conformal:harmonicholo}{função harmônica} com essa
propriedade.\index{problema de Dirichlet}
\end{theorem}

\begin{proof}
\emph{Identidades do núcleo}: somando duas séries geométricas,
\[
\sum_{n\in\Z}r^{\abs n}\eu^{\iu n\theta}
= \operatorname{Re}\frac{1 + r\eu^{\iu\theta}}{1 -
r\eu^{\iu\theta}}
= \frac{1 - r^2}{\abs{1 - r\eu^{\iu\theta}}^2},
\]
que é o quociente exibido; a positividade é clara, e $\frac1{2\pi}\int_0^{2\pi}P_r = 1$ (só $n = 0$
sobrevive).

\emph{Harmonicidade}: com $z = r\eu^{\iu\varphi}$,
\[
u(z) = \operatorname{Re}\biggl[\frac1{2\pi}\int_0^{2\pi}
\frac{\eu^{\iu t} + z}{\eu^{\iu t} - z}\,
g(\eu^{\iu t})\,\dd t\biggr],
\]
(o núcleo entre colchetes tem parte real $P_r(\varphi - t)$: calcule), e o colchete é
\omterm{def:b3:holomorphic:holo}{holomorfo} em $z$ em $\mathbb D$ (\cref{exo:b3:holomorphic:7}): $u$ é a parte real de
uma \omterm{def:b3:holomorphic:holo}{função holomorfa}, logo é harmônica.

\emph{Valores de fronteira}: $P_r(\cdot)$ é uma \omterm{def:b3:lp:mollifier}{aproximação da identidade}
quando $r \to 1^-$: massa $1$ e, para $\delta \leq
\abs\theta \leq \pi$, $P_r(\theta) \leq \frac{1 - r^2}{1 -
2r\cos\delta + r^2} \to 0$ uniformemente. A
separação padrão (\omterm{def:b3:topology:continuity}{continuidade} de $g$ perto de $\eu^{\iu\varphi_0}$, limitação nos
demais pontos) dá $u(r\eu^{\iu\varphi}) \to
g(\eu^{\iu\varphi_0})$ quando $r\eu^{\iu\varphi} \to
\eu^{\iu\varphi_0}$, uniformemente no ponto de
fronteira: a extensão é \omterm{def:b3:topology:continuity}{contínua}. \emph{Unicidade}: a diferença de duas
soluções é harmônica em $\mathbb D$, \omterm{def:b3:topology:continuity}{contínua} no \omterm{def:b3:topology:interior}{fecho} e nula na fronteira:
pelo princípio do máximo (aplicado a $\pm$ a diferença), ela se
anula.
\end{proof}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:conformal:1}
(a) Verifique que a \omterm{ex:b3:conformal:mobius}{aplicação de Cayley} $\varphi(z) = \frac{z -
\iu}{z + \iu}$ é uma bijeção $\mathbb H \to \mathbb D$ com a
inversa indicada, e calcule as imagens de $\iu$, $0$, $1$,
$\infty$ (limite).
(b) Mostre que $z \mapsto 1/z$ leva círculos e retas em círculos e retas.
\emph{(Escreva sua equação comum $\alpha\abs z^2 + \bar\beta z + \beta\bar z + \gamma = 0$, $\alpha, \gamma \in \R$.)} Deduza o mesmo para
todas as aplicações de Möbius.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:conformal:2}
Seja $f \colon \mathbb D \to \mathbb D$ \omterm{def:b3:holomorphic:holo}{holomorfa}.
(a) Se $f(0) = 0$ e $f(a) = a$ para algum $a \neq 0$, mostre que $f
= \mathrm{id}$.
(b) Se $f$ é um automorfismo com dois pontos fixos distintos em
$\mathbb D$, mostre que $f = \mathrm{id}$ \emph{(conjugue por um \omterm{thm:b3:conformal:autdisc}{fator de Blaschke} para
reduzir a (a))}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:3}
(Schwarz--Pick) Para $f\colon \mathbb D \to
\mathbb D$ \omterm{def:b3:holomorphic:holo}{holomorfa}, demonstre
\[
\frac{\abs{f'(z)}}{1 - \abs{f(z)}^2} \;\leq\;
\frac{1}{1 - \abs z^2}
\qquad (z \in \mathbb D),
\]
com igualdade (em um ponto e, portanto, em toda parte) se, e somente
se, $f \in
\operatorname{Aut}(\mathbb D)$. \emph{(Aplique Schwarz a $\varphi_{f(z)}\circ
f\circ\varphi_{-z}$.)} Interpretação: as
autoaplicações \omterm{def:b3:holomorphic:holo}{holomorfas} contraem a métrica hiperbólica.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:4}
Encontre equivalências conformes explícitas:
(a) a faixa $\{0 < \operatorname{Im}z < \pi\} \to \mathbb H$;
(b) o quadrante $\{\operatorname{Re}z > 0,
\operatorname{Im}z > 0\} \to \mathbb H$;
(c) o semidisco $\mathbb D\cap\mathbb H \to$ para um quadrante e, depois, $\to \mathbb H$;
(d) $\mathbb D \to \mathbb D$ levando $\frac12$ em $0$ com derivada positiva ali.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:5}
(a) Mostre que não existe \omterm{def:b3:conformal:conformal}{aplicação conforme} $\C \to \mathbb D$ nem $\C \to
\mathbb H$, e nenhuma
$\mathbb D \to \C$.
(b) Quais dos seguintes são \omterm{def:b3:conformal:conformal}{conformemente equivalentes} a $\mathbb D$?
Justifique via o~\cref{thm:b3:conformal:rmt} ou por uma obstrução: um quadrado;
$\C\setminus\intoc{-\infty}0$; $\mathbb D\setminus\{0\}$; $\{1 < \abs z < 2\}$. \emph{(Para os dois últimos: uma imagem conforme do
disco perfurado se estenderia sobre a perfuração pelo~\cref{thm:b3:residues:riemanncw}(1)
--- desenvolva isso.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:6}
Seja $\mathcal F = \{f \in \mathcal H(\mathbb D) : f(0) = 1,\
\operatorname{Re}f > 0\}$.
(a) Mostre que $\mathcal F$ é localmente limitada. \emph{(Componha com a
aplicação do tipo Cayley $w \mapsto \frac{w -
1}{w + 1}$ que leva o semiplano direito em $\mathbb D$ e
aplique Schwarz.)}
(b) Deduza a \emph{cota de Herglotz}: $\abs{f(z)} \leq
\frac{1 + \abs z}{1 - \abs z}$ para $f \in \mathcal F$, com as
possibilidades de igualdade.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:conformal:7}
Onde a demonstração do~\cref{thm:b3:conformal:rmt} usa cada hipótese? Acompanhe: (i)
a \omterm{def:b3:topology:connected}{conexidade} simples (duas vezes); (ii) $\Omega \neq \C$; (iii) a \omterm{def:b3:topology:connected}{conexidade}.
Mostre então que o teorema \emph{falha} para $\Omega = \C$ e para o anel,
apontando que passo da demonstração se quebra em cada caso.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:8}
Resolva o problema de Dirichlet em $\mathbb D$ para os dados de fronteira:
(a) $g(\eu^{\iu t}) = \cos t$; (b) $g(\eu^{\iu t}) =
\cos^2 t$; (c) $g = \mathbf 1_{\text{semicírculo superior}}$ --- para (c), calcule $u(0)$ e
interprete pela propriedade do valor médio. \emph{(Expanda $g$ em
série de Fourier e use a série de $P_r$: $u(r\eu^{\iu\varphi}) = \sum_n c_n(g)
r^{\abs n}\eu^{\iu n\varphi}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:conformal:9}
(Harnack) Seja $u \geq 0$ harmônica em $\mathbb D$. Demonstre, para $\abs z = r < 1$:
\[
\frac{1 - r}{1 + r}\,u(0) \;\leq\; u(z) \;\leq\;
\frac{1 + r}{1 - r}\,u(0)
\]
\emph{(limite o \omterm{thm:b3:conformal:poisson}{núcleo de Poisson} entre $\frac{1-r}{1+r}$ e $\frac{1+r}{1-r}$; aplique a
\omterm{def:b3:representations:rep}{representação} em discos ligeiramente menores e passe ao limite)}.
Deduza: uma \omterm{prop:b3:conformal:harmonicholo}{função harmônica} em $\C$ limitada inferiormente é
constante.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:10}
Usando a invariância conforme da harmonicidade ($u\circ f$ é harmônica quando
$u$ é harmônica e $f$ é \omterm{def:b3:holomorphic:holo}{holomorfa} --- demonstre-o localmente via
a~\cref{prop:b3:conformal:harmonicholo}), resolva o problema de Dirichlet no semiplano
superior com dados de fronteira $\mathbf 1_{\intoo{-\infty}0}$: mostre que
\[
u(x + \iu y) = \frac1\pi\,\arg(x + \iu y)
\qquad (\arg \in \intoo0\pi \text{ em } \mathbb H)
\]
é harmônica em $\mathbb H$ (parte imaginária de um \omterm{def:b3:conformal:simplyconnected}{logaritmo holomorfo}) com
os limites de fronteira exigidos em todo $x \neq
0$, e transporte-a para o
disco por Cayley para redemonstrar o~\cref{exo:b3:conformal:8}(c).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:11}
(Pontos fixos e iteração no disco) Seja $f\colon\mathbb
D \to \mathbb D$ \omterm{def:b3:holomorphic:holo}{holomorfa}.
(a) Mostre que, se $f$ tem \emph{dois} pontos fixos distintos, então
$f = \mathrm{id}$ \emph{(leve um deles a $0$ por um automorfismo e aplique o caso
de igualdade de Schwarz)}.
(b) Suponha $f(0) = 0$ e $f$ não sendo uma rotação. Mostre que os iterados
$f^{\circ n} \to 0$ uniformemente em todo \omterm{def:b3:topology:compact}{compacto} $\bar D(0, r)$, $r < 1$ \emph{(Schwarz dá
$\abs{f(z)}
\leq c_r\abs z$ em $\bar D(0,r)$ com $c_r < 1$ --- justifique essa constante estrita pelo
princípio do máximo aplicado a $f(z)/z$)}.
(c) Ilustre com $f(z) = \frac{z^2 + z}2$: pontos fixos e a taxa de convergência da \omterm{def:b3:groups:action}{órbita}
de $z_0 =
\frac12$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:12}
(Conjugadas harmônicas, concretamente) Seja $u(x, y) = x^3 - 3xy^2
+ 2y$.
(a) Verifique que $u$ é harmônica em $\R^2$ e encontre todas as
conjugadas harmônicas $v$ (isto é, com $u + \iu v$ \omterm{def:b3:holomorphic:holo}{holomorfa}) integrando
as \omterm{prop:b3:holomorphic:cauchyriemann}{equações de Cauchy--Riemann}; identifique $f(z) =
u + \iu v$ como um polinômio em
$z$.
(b) Mostre que, em um aberto \emph{estrelado}, toda \omterm{prop:b3:conformal:harmonicholo}{função harmônica}
admite conjugada harmônica, única a menos de constante aditiva
\emph{(a $1$-forma $-u_y\,\dd x +
u_x\,\dd y$ é fechada; o maquinário de primitivas
do~\cref{thm:b3:holomorphic:cauchy}, ou o lema de Poincaré do~\cref{ch:b3:forms})}.
(c) Dê o contraexemplo padrão em $\C^*$: $u = \ln\abs z$ não tem conjugada
global --- relacione com a forma angular e o índice de rotação
(\cref{ch:b3:forms}).
\end{exercise}

\section{Problema: o teorema da área e o teorema do quarto de Koebe}

\begin{problem}[{Problema de fim de semana --- quanto uma aplicação
univalente tem de cobrir?}]\label{pb:b3:conformal:1}
Uma função \emph{univalente} é uma injeção \omterm{def:b3:holomorphic:holo}{holomorfa}. As funções
univalentes normalizadas no disco,
\[
\mathcal S = \bigl\{f \in \mathcal H(\mathbb D) \text{
injective},\ f(z) = z + a_2z^2 + a_3z^3 + \cdots\bigr\},
\]
são rigidamente restringidas: demonstramos a desigualdade de Bieberbach
$\abs{a_2} \leq 2$ e deduzimos o \emph{teorema do quarto de Koebe}: a imagem de
qualquer $f \in \mathcal S$ contém o disco $D(0, \frac14)$ --- a constante universal exata da
geometria conforme.\index{teorema do quarto de Koebe}\index{teorema da
área}

\medskip
\noindent\textbf{Parte I --- O teorema da área.} Seja $g(w) = w
+ b_0 + \frac{b_1}w + \frac{b_2}{w^2} + \cdots$ \omterm{def:b3:holomorphic:holo}{holomorfa} e
\emph{injetora} em $\{\abs w > 1\}$.
\begin{enumerate}
  \item Para $\rho > 1$, seja $A_\rho$ a área (\omterm{def:b3:measure:lebesgueouter}{medida de Lebesgue}) do
        \omterm{def:b3:topology:compact}{compacto} $K_\rho = \C\setminus
        g(\{\abs w > \rho\})$, a região limitada pela curva de Jordan suave
        $g(C_\rho)$. Usando a fórmula da área do teorema de Green--Riemann do
        volume do segundo ano --- a área englobada é $\frac1{2\iu}
        \oint\bar\zeta\,\dd\zeta$ ao longo da
        fronteira orientada positivamente ---, mostre que
        \[
        A_\rho = \frac{1}{2\iu}\int_{C_\rho}
        \overline{g(w)}\,g'(w)\,\dd w
        = \pi\Bigl(\rho^2 -
        \sum_{n\geq1}n\,\abs{b_n}^2\rho^{-2n}\Bigr) :
        \]
        substitua as \omterm{thm:b3:residues:laurent}{séries de Laurent} de $\bar g$ e $g'$ em
        $C_\rho$ e integre termo a termo (convergência normal; só os
        produtos de frequência zero sobrevivem).
  \item Faça $\rho \to 1^+$ e conclua o \emph{teorema da área}:
        \[
        \sum_{n\geq1}n\,\abs{b_n}^2 \;\leq\; 1 .
        \]
        Em particular, $\abs{b_1} \leq 1$. Quando $\abs{b_1}
        = 1$?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O $\abs{a_2} \leq
2$ de Bieberbach.} Seja $f = z + a_2z^2 + \cdots \in \mathcal S$.
\begin{enumerate}[resume]
  \item Mostre que $f(z^2)/z^2$ é \omterm{def:b3:holomorphic:holo}{holomorfa} e sem zeros em $\mathbb D$ e admite raiz
        quadrada \omterm{def:b3:holomorphic:holo}{holomorfa} $\varphi$ com $\varphi(0) = 1$; ponha $h(z) =
        z\varphi(z^2)$, de modo que
        $h(z)^2 = f(z^2)$. Mostre que $h$ é uma função univalente \emph{ímpar} em
        $\mathbb
        D$ com expansão $h(z) = z + \frac{a_2}2z^3 +
        \cdots$. \emph{(Injetividade: $h(z)^2 = h(z')^2$ força $z^2 = z'^2$;
        use a imparidade para concluir.)}
  \item Aplique o teorema da área a $g(w) = 1/h(1/w) = w -
        \frac{a_2}{2w} + \cdots$ em $\{\abs w > 1\}$ (verifique a
        univalência e a expansão) e conclua $\abs{a_2} \leq 2$.
  \item Mostre que a \emph{função de Koebe}
        \[
        k(z) = \frac{z}{(1 - z)^2} = \sum_{n\geq1}n\,z^n
        \]
        pertence a $\mathcal S$, tem $a_2 = 2$ e leva $\mathbb D$ sobre $\C\setminus\intoc{-\infty}{-\frac14}$
        \emph{(escreva $k = \frac14\bigl[\bigl(\frac{1+z}{1 -
        z}\bigr)^2 - 1\bigr]$ e acompanhe as imagens)}: todas as
        desigualdades que virão são exatas.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O teorema do quarto.}
\begin{enumerate}[resume]
  \item Sejam $f \in \mathcal S$ e $c \notin f(\mathbb D)$. Mostre que
        \[
        F(z) = \frac{c\,f(z)}{c - f(z)}
        \]
        pertence a $\mathcal S$ e calcule seu segundo coeficiente: $A_2 = a_2 + \frac1c$.
  \item Aplique Bieberbach a $f$ e a $F$: conclua $\abs{\frac1c} \leq \abs{A_2} + \abs{a_2} \leq 4$, isto é,
        $\abs c \geq \frac14$. \emph{Todo valor omitido tem módulo $\geq \frac14$}: $f(\mathbb D) \supseteq
        D(0, \frac14)$ --- o
        teorema do quarto de Koebe. Verifique a exatidão na função de
        Koebe.
  \item Deduza uma estimativa quantitativa para a aplicação de Riemann:
        se $\varphi \colon \Omega \to \mathbb D$ é a aplicação de Riemann do~\cref{thm:b3:conformal:rmt} em $z_0$,
        então
        \[
        \frac{d\bigl(z_0, \partial\Omega\bigr)}{4}
        \;\leq\; \frac1{\varphi'(z_0)} \;\leq\;
        4\,d\bigl(z_0, \partial\Omega\bigr)
        \]
        --- demonstre ao menos a desigualdade da esquerda aplicando
        Koebe a $\varphi^{-1}$ convenientemente normalizada, e a da direita por
        Schwarz aplicado a $\varphi$ no disco $D(z_0, d) \subseteq \Omega$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Perspectiva.}
\begin{enumerate}[resume]
  \item Bieberbach conjecturou (1916) que $\abs{a_n} \leq n$ para todo $n$, com
        igualdade apenas para as rotações da função de Koebe; de
        Branges demonstrou isso em 1985. Verifique a conjectura à mão
        para a função de Koebe e suas rotações $\eu^{-\iu\theta}k(\eu^{\iu\theta}z)$. Depois, leve a
        expansão da questão 4 um termo adiante: escrevendo $h(z) = z + \frac{a_2}2z^3 + c_5z^5 + \cdots$, mostre
        que $c_5 = \frac{a_3}2 - \frac{a_2^2}8$ e
        \[
        g(w) = w - \frac{a_2}{2}\,w^{-1} +
        \Bigl(\frac{3a_2^2}8 - \frac{a_3}2\Bigr)w^{-3} +
        \cdots ,
        \]
        de modo que o teorema da área fornece a desigualdade refinada
        $\bigl|\frac{a_2}2\bigr|^2 + 3\bigl|\frac{3a_2^2}8 -
        \frac{a_3}2\bigr|^2 \leq 1$. Confira-a na função de Koebe ($a_2 = 2$, $a_3 = 3$).
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- O teorema da distorção.} A desigualdade de
Bieberbach, transportada pelo disco por automorfismos, controla $f'$
em toda parte. Fixe $f \in \mathcal
S$.
\begin{enumerate}[resume]
  \item (Transformada de Koebe) Para $z_0 \in \mathbb D$, seja $\varphi(z) = \frac{z + z_0}{1 + \bar z_0z}$, um automorfismo
        do disco (\cref{thm:b3:conformal:autdisc}) com $\varphi(0) = z_0$. Mostre que
        \[
        F(z) = \frac{f(\varphi(z)) - f(z_0)}
        {f'(z_0)\,\bigl(1 - \abs{z_0}^2\bigr)}
        \]
        pertence a $\mathcal S$ \emph{(a univalência é herdada; calcule $\varphi'(0) = 1 - \abs{z_0}^2$ e
        verifique a normalização; lembre que $f' \neq 0$ para $f$ injetora,
        \cref{def:b3:conformal:conformal})}.
  \item Calcule o segundo coeficiente $A_2 = \frac12F''(0)$ de $F$:
        \[
        A_2 = \frac12\Bigl[\bigl(1 - \abs{z_0}^2\bigr)
        \frac{f''(z_0)}{f'(z_0)} - 2\bar z_0\Bigr] ,
        \]
        e deduza de Bieberbach (questão 4), para $z =
        r\eu^{\iu\theta}$, a
        \emph{desigualdade fundamental}:
        \[
        \Bigl|\,z\,\frac{f''(z)}{f'(z)} -
        \frac{2r^2}{1 - r^2}\Bigr|
        \;\leq\; \frac{4r}{1 - r^2} .
        \]
  \item Extraia a parte real:
        \[
        \frac{2r^2 - 4r}{1 - r^2} \;\leq\;
        \operatorname{Re}\Bigl(z\,\frac{f''(z)}{f'(z)}\Bigr)
        \;\leq\; \frac{2r^2 + 4r}{1 - r^2} .
        \]
  \item Mostre que $\frac{\dd}{\dd t}\log\bigl|
        f'(t\eu^{\iu\theta})\bigr| = \frac1t
        \operatorname{Re}\bigl(z\frac{f''(z)}{f'(z)}\bigr)$ em $z = t\eu^{\iu\theta}$ \emph{(para uma função $g$ $\mathcal C^1$
        que não se anula, de uma variável real, $\frac{\dd}{\dd t}\log\abs g =
        \operatorname{Re}(g'/g)$)}, e integre as
        cotas da questão 12 ao longo do raio para obter o \emph{teorema
        da distorção}:
        \[
        \frac{1 - r}{(1 + r)^3} \;\leq\; \abs{f'(z)}
        \;\leq\; \frac{1 + r}{(1 - r)^3},
        \qquad \abs z = r .
        \]
  \item Deduza o \emph{teorema do crescimento}:
        \[
        \frac{r}{(1 + r)^2} \;\leq\; \abs{f(z)} \;\leq\;
        \frac{r}{(1 - r)^2},
        \qquad \abs z = r
        \]
        \emph{(cota superior: integre $f'$ no segmento $[0, z]$;
        cota inferior: se $\abs{f(z)} < \frac14$, o segmento $[0, f(z)]$ está em $f(\mathbb D)$ pela
        questão 7; puxe-o de volta por $f^{-1}$ --- \omterm{def:b3:holomorphic:holo}{holomorfa}
        pelo~\cref{cor:b3:residues:openmapping} --- e limite $\abs{f(z)} = \int_\gamma\abs{f'(\zeta)}
        \,\abs{\dd\zeta} \geq \int_0^r\frac{1 - t}{(1 +
        t)^3}\,\dd t$, usando $\abs{\dd\zeta} \geq
        \dd\abs\zeta$)}.
  \item Verifique que a função de Koebe atinge a igualdade nas quatro
        cotas, em $z = r$ para as superiores e em $z = -r$ para as
        inferiores: $k$ é simultaneamente o membro mais expansivo e,
        no antípoda, o mais contrativo de $\mathcal S$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Rigidez extremal.} Em todas as
desigualdades vistas até aqui, a igualdade identifica a função de Koebe
a menos de rotação. Demonstramos isso e, depois, colhemos.
\begin{enumerate}[resume]
  \item Suponha que $f \in \mathcal S$ tenha $\abs{a_2} = 2$. Persiga a igualdade pelas
        questões 2--4: o teorema da área força $g(w) = w + b_0 + \eu^{\iu\alpha}/w$; a imparidade de
        $h$ torna $g$ ímpar, de modo que $b_0 = 0$; inverta para
        achar $h$ e, depois, $f$, e conclua que
        \[
        f(z) = \eu^{-\iu\theta}k\bigl(\eu^{\iu\theta}z\bigr)
        \quad\text{com } \eu^{\iu\theta} =
        -\eu^{\iu\alpha} :
        \]
        as rotações da função de Koebe são os únicos membros de $\mathcal S$
        com $\abs{a_2} = 2$.
  \item Mostre que, se $f \in \mathcal S$ omite um valor $c$ com exatamente
        $\abs c = \frac14$, então $f$ é uma rotação da função de Koebe, e
        identifique $c =
        -\eu^{-\iu\theta}/4$ \emph{(acompanhe a igualdade pela cadeia $4 = \abs{1/c} = \abs{A_2 - a_2}
        \leq \abs{A_2} + \abs{a_2} \leq 4$
        da questão 7)}: a constante do teorema do quarto só é atingida
        pela família extremal.
  \item (Coeficientes a preço baixo) Combine o teorema do crescimento
        com as \omterm{thm:b3:holomorphic:analytic}{estimativas de Cauchy} (\cref{thm:b3:holomorphic:analytic}) no círculo $\abs z = 1 - \frac1n$
        para demonstrar
        \[
        \abs{a_n} \;\leq\; \eu\,n^2 \qquad (n \geq 2) .
        \]
        (De Branges, 1985: $\abs{a_n} \leq n$; o fator $\eu n$ é o preço das
        ferramentas elementares.)
  \item (Cobertura de subdiscos) Mostre que, para todo $0 < r <
        1$,
        \[
        f\bigl(D(0, r)\bigr) \supseteq
        D\Bigl(0, \frac{r}{(1 + r)^2}\Bigr),
        \]
        exato para a função de Koebe, e recupere o teorema do quarto
        quando $r \to 1^-$. \emph{(Os pontos de fronteira da imagem aberta
        $f(D(0,r))$ estão em $f(\partial D(0,r))$, logo têm módulo $\geq
        \frac{r}{(1+r)^2}$ pela questão 14; um
        segmento de $0$ a um ponto omitido de módulo menor teria de
        cruzar essa fronteira.)}
  \item (Koebe em todo ponto) Sejam $f$ univalente em $\mathbb D$, não
        necessariamente normalizada, e $z_0
        \in \mathbb D$. Demonstre
        \[
        \tfrac14\bigl(1 - \abs{z_0}^2\bigr)\abs{f'(z_0)}
        \;\leq\;
        d\bigl(f(z_0), \partial f(\mathbb D)\bigr)
        \;\leq\;
        \bigl(1 - \abs{z_0}^2\bigr)\abs{f'(z_0)}
        \]
        \emph{(à esquerda: o teorema do quarto aplicado à transformada
        de Koebe da questão 10; à direita: Schwarz
        (\cref{thm:b3:conformal:schwarz}) aplicado a $\psi^{-1}\circ\hat g$, em que $\hat g(w) =
        f^{-1}\bigl(f(z_0) + dw\bigr)$, com $d$ a
        distância e $\psi$ um automorfismo do disco que leva $0$
        em $z_0$)}. Por que $\partial f(\mathbb D)$ é não vazio?
  \item Confira a questão 20 em $f = k$ em $z_0 = r \in
        \intoo01$: calcule $d\bigl(k(r), \partial k(\mathbb
        D)\bigr) = \frac{(1+r)^2}{4(1-r)^2}$ e
        verifique que a desigualdade da esquerda é uma \emph{igualdade}:
        a função de Koebe satura seu próprio teorema em todo ponto de
        $\intoo01$.
  \item (A moral) Em dez linhas: que princípio único está por trás do
        teorema da área, e como Bieberbach, o teorema do quarto, a
        distorção, o crescimento e a cobertura decorrem todos dele?
        Compare com o mundo de Schwarz--Pick do~\cref{thm:b3:conformal:schwarz}: em
        ambos, uma única desigualdade \omterm{def:b3:topology:interior}{interior} enrijece a geometria
        inteira, e os extremais são únicos a menos de rotação.

\end{enumerate}
\medskip
\noindent\textbf{Parte VII --- \omterm{def:b3:topology:compact}{Compacidade}, inversas e um teste de
realidade.}
\begin{enumerate}[resume]
  \item Mostre que a classe $\mathcal S$ é \emph{compacta} para a convergência
        localmente uniforme: ela é localmente limitada pelo teorema do
        crescimento, logo normal (\cref{thm:b3:conformal:montel}); e um limite
        localmente uniforme de membros de $\mathcal S$ está de novo em $\mathcal S$
        \emph{(as normalizações passam ao limite pela convergência de
        Weierstrass das derivadas; a injetividade sobrevive por
        Hurwitz, \cref{exo:b3:residues:8}, sendo o limite não constante)}. Por que
        isso importa para problemas extremais como o de Bieberbach?
  \item Para $f \in \mathcal S$, seja $g = f^{-1}$, definida perto de $0$. Mostre que
        $g(w) = w - a_2w^2 + O(w^3)$, de modo que o segundo coeficiente da inversa obedece à
        mesma cota exata $\abs{A_2} = \abs{a_2} \leq 2$, com igualdade exatamente para as funções
        de Koebe rotacionadas.
  \item Determine para quais $a \in \C$ o polinômio $f(z)
        = z + az^2$ pertence a $\mathcal S$:
        mostre que $f$ é injetora em $\mathbb D$ se, e somente se, $\abs a \leq \frac12$
        \emph{(fatore $f(z_1) - f(z_2)$)}. Conclua: para polinômios de grau dois, a
        cota verdadeira dos coeficientes é $\frac12$, quatro vezes menor do
        que o $2$ de Bieberbach --- os extremais de $\mathcal S$ são objetos
        genuinamente transcendentes, e nenhum polinômio chega perto.
\end{enumerate}
\end{problem}
