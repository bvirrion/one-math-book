\chapter{Teoria da medida}\label{ch:b3:measure}

Qual é o comprimento de um subconjunto de $\R$? A resposta ingênua
--- atribuir a todo conjunto um comprimento invariante por translação
que estenda o dos intervalos --- é \emph{impossível}: a construção de
Vitali, no fim deste capítulo, produz um conjunto sem comprimento
coerente. A teoria da \omterm{def:b3:measure:measure}{medida} é a retirada disciplinada: restringimos a
atenção a uma classe rica de conjuntos \emph{mensuráveis}, sobre a qual
existe, e é única, uma noção de comprimento enumeravelmente aditiva. As
recompensas são imensas --- a integral de Lebesgue (\cref{ch:b3:lebesgue}), os
espaços $L^p$ da análise funcional e toda a probabilidade moderna
(\cref{ch:b3:probability}) assentam-se sobre os três teoremas aqui demonstrados: o
lema de unicidade de Dynkin, o teorema de extensão de Carathéodory e a
existência da \omterm{def:b3:measure:lebesgueouter}{medida de Lebesgue}.

\section{$\sigma$-álgebras}

\begin{definition}\label{def:b3:measure:sigmaalgebra}
Uma \emph{$\sigma$-álgebra}\index{sigma-álgebra@$\sigma$-álgebra}
sobre um conjunto $X$ é uma família $\mathcal A$ de subconjuntos que contém
$\varnothing$ e é estável por complementação e por reuniões
\emph{enumeráveis} (logo, por interseções enumeráveis e por diferenças
de conjuntos, e contém $X$). O par $(X, \mathcal A)$ é um \emph{espaço
mensurável}; os membros de $\mathcal A$ são os \emph{conjuntos mensuráveis}.
Para uma família $\mathcal E$ de subconjuntos, $\sigma(\mathcal E)$ denota a menor
$\sigma$-álgebra que contém $\mathcal E$ (a interseção de todas elas --- uma
interseção de $\sigma$-álgebras também o é).
\end{definition}

\begin{definition}\label{def:b3:measure:borel}
A \emph{$\sigma$-álgebra de Borel}\index{sigma-álgebra de Borel@%
$\sigma$-álgebra de Borel} de um espaço \omterm{def:b3:topology:topology}{topológico} é $\mathcal B(X) = \sigma(\{\text{abertos}\})$. Em
$\R$: $\mathcal B(\R)$ é gerada também pelos intervalos abertos, pelos
intervalos fechados, pelas semirretas $(-\infty, a]$ e pelas semirretas com
extremidades racionais (\cref{exo:b3:measure:1}) --- cada família gera os abertos
por operações enumeráveis; por exemplo, todo aberto de $\R$ é uma
reunião enumerável de intervalos abertos com dados racionais.
\end{definition}

\begin{definition}\label{def:b3:measure:dynkin}
Um \emph{$\pi$-sistema}\index{pi-sistema@$\pi$-sistema} é uma
família estável por interseções finitas. Um
\emph{$\lambda$-sistema}\index{lambda-sistema@$\lambda$-sistema} (classe de
Dynkin) é uma família $\mathcal D$ com: $X \in \mathcal
D$; $A, B \in \mathcal D$, $A \subseteq B$ $\Rightarrow$ $B
\setminus A \in \mathcal D$; e $A_n \uparrow A$,
$A_n \in
\mathcal D$ $\Rightarrow$ $A \in \mathcal D$.
\end{definition}

\begin{theorem}[Lema $\pi$--$\lambda$ de Dynkin]
\label{thm:b3:measure:dynkin}
Se um $\lambda$-sistema $\mathcal D$ contém um $\pi$-sistema $\mathcal P$, então
$\mathcal D \supseteq
\sigma(\mathcal P)$.\index{lema de Dynkin}
\end{theorem}

\begin{proof}
Seja $\mathcal D_0$ o menor $\lambda$-sistema que contém $\mathcal P$ (interseção de todos
eles); basta mostrar que $\mathcal D_0$ é uma $\sigma$-álgebra, pois então
$\sigma(\mathcal P) \subseteq \mathcal D_0 \subseteq \mathcal
D$. Um $\lambda$-sistema estável por interseções finitas \emph{é} uma
$\sigma$-álgebra: complementos ($X \setminus A = X
\setminus A$ com $A \subseteq X$), reuniões finitas
($A \cup B =
X \setminus ((X\setminus A)\cap(X\setminus B))$) e reuniões enumeráveis via $\bigcup_{k \leq n}A_k \uparrow \bigcup_kA_k$. Demonstramos, portanto, que
$\mathcal D_0$ é um $\pi$-sistema, em duas etapas. Seja
\[
\mathcal D_1 = \{A \in \mathcal D_0 : A \cap P \in \mathcal
D_0 \ \forall P \in \mathcal P\}.
\]
$\mathcal D_1$ é um $\lambda$-sistema (os três axiomas se verificam intersectando com
$P$: por exemplo, $(B\setminus A)\cap P
= (B \cap P)\setminus(A \cap P)$, uma diferença própria dentro de $\mathcal D_0$) e
contém $\mathcal P$ (que é $\pi$-sistema): $\mathcal D_1 = \mathcal D_0$. Seja agora
\[
\mathcal D_2 = \{A \in \mathcal D_0 : A \cap D \in \mathcal
D_0\ \forall D \in \mathcal D_0\}.
\]
Pela etapa anterior, $\mathcal D_2 \supseteq \mathcal P$; e $\mathcal D_2$ é um $\lambda$-sistema pela mesma
verificação: $\mathcal D_2 = \mathcal D_0$, o que diz exatamente que $\mathcal D_0$ é estável por
interseções.
\end{proof}

\section{Medidas}

\begin{definition}\label{def:b3:measure:measure}
Uma \emph{medida}\index{medida} em $(X, \mathcal A)$ é uma aplicação $\mu \colon \mathcal A \to [0, +\infty]$ com $\mu(\varnothing)
= 0$
que é \emph{$\sigma$-aditiva}: para $(A_n)_{n\in\N}$ dois a dois disjuntos,
\[
\mu\Bigl(\bigsqcup_n A_n\Bigr) = \sum_n \mu(A_n).
\]
$(X, \mathcal A, \mu)$ é um \emph{espaço de medida}; $\mu$ é \emph{finita} se $\mu(X) < \infty$, é
uma \emph{medida de probabilidade} se $\mu(X) = 1$, e é
\emph{$\sigma$-finita} se $X$ é reunião enumerável de conjuntos de
medida finita. Exemplos: a medida de contagem em $(\N, \mathcal P(\N))$; a massa de Dirac
$\delta_a(A)
= \mathbf 1_{a \in A}$; e, objeto deste capítulo, a \omterm{def:b3:measure:lebesgueouter}{medida de Lebesgue}.
\end{definition}

\begin{proposition}\label{prop:b3:measure:basics}
Seja $\mu$ uma \omterm{def:b3:measure:measure}{medida}. (a) Monotonicidade: $A \subseteq B
\Rightarrow \mu(A) \leq \mu(B)$. (b) Subaditividade
enumerável: $\mu(\bigcup A_n) \leq \sum\mu(A_n)$. (c) \omterm{def:b3:topology:continuity}{Continuidade} por baixo: $A_n \uparrow A \Rightarrow \mu(A_n) \to \mu(A)$. (d) \omterm{def:b3:topology:continuity}{Continuidade}
por cima: $A_n \downarrow A$ \emph{com $\mu(A_1) < \infty$} $\Rightarrow \mu(A_n) \to \mu(A)$.
\end{proposition}

\begin{proof}
(a) $B = A \sqcup (B\setminus A)$. (b) Disjuntifique: os $B_n = A_n
\setminus \bigcup_{k<n}A_k$ são disjuntos com a mesma
reunião, e $\mu(B_n) \leq \mu(A_n)$. (c) $A = \bigsqcup_n (A_n
\setminus A_{n-1})$ ($A_0 = \varnothing$): as somas parciais de $\sum\mu(A_n\setminus A_{n-1})$ valem
$\mu(A_n)$. (d) Aplique (c) a $A_1 \setminus A_n \uparrow A_1 \setminus A$ e subtraia de $\mu(A_1)$ --- a finitude torna
a subtração legítima. Contraexemplo sem ela: $A_n = [n, \infty)$ para a \omterm{def:b3:measure:lebesgueouter}{medida de
Lebesgue}: $A_n \downarrow \varnothing$, mas $\mu(A_n) =
\infty$.
\end{proof}

\begin{theorem}[Unicidade]\label{thm:b3:measure:uniqueness}
Sejam $\mu, \nu$ \omterm{def:b3:measure:measure}{medidas} em $\sigma(\mathcal P)$ e $\mathcal
P$ um $\pi$-sistema, com $\mu = \nu$ em
$\mathcal P$. Se existem conjuntos $P_k \in \mathcal P$ com $P_k \uparrow X$ e $\mu(P_k) < \infty$, então $\mu = \nu$ em todo
$\sigma(\mathcal P)$.
\end{theorem}

\begin{proof}
Fixe $k$ e considere as \omterm{def:b3:measure:measure}{medidas} finitas $\mu_k(A) = \mu(A \cap
P_k)$ e $\nu_k(A) = \nu(A \cap P_k)$ em $\sigma(\mathcal
P)$: elas
coincidem em $\mathcal P$, pois $P \cap P_k \in \mathcal
P$ ($\pi$-sistema), e atribuem a $X$
o mesmo valor finito $\mu(P_k)$. A classe $\mathcal D = \{A : \mu_k(A) =
\nu_k(A)\}$ é um $\lambda$-sistema: $X \in \mathcal D$; as
diferenças próprias, por subtração (valores finitos); e os limites
crescentes, por \omterm{def:b3:topology:continuity}{continuidade} por baixo (\cref{prop:b3:measure:basics}(c)). Ela contém
o $\pi$-sistema $\mathcal P$, de modo que Dynkin (\cref{thm:b3:measure:dynkin}) dá $\mathcal D \supseteq
\sigma(\mathcal P)$:
$\mu_k = \nu_k$ em toda parte. Por fim, para $A \in \sigma(\mathcal P)$ qualquer, a \omterm{def:b3:topology:continuity}{continuidade} por
baixo ao longo de $A
\cap P_k \uparrow A$ dá $\mu(A) = \lim_k\mu_k(A) =
\lim_k\nu_k(A) = \nu(A)$.
\end{proof}

\section{Medidas exteriores e o teorema de Carathéodory}

\begin{definition}\label{def:b3:measure:outer}
Uma \emph{medida exterior}\index{medida exterior} em $X$ é uma
aplicação $\mu^* \colon \mathcal P(X) \to [0, \infty]$ com $\mu^*(\varnothing) = 0$, monótona e enumeravelmente subaditiva. Um
conjunto $A$ é \emph{$\mu^*$-mensurável} (Carathéodory) se ele
reparte aditivamente todo conjunto:
\[
\mu^*(E) = \mu^*(E \cap A) + \mu^*(E \setminus A)
\qquad \text{para todo } E \subseteq X
\]
($\leq$ vale sempre, por subaditividade; o conteúdo está em
$\geq$).
\end{definition}

\begin{theorem}[Carathéodory]\label{thm:b3:measure:caratheodory}
Os conjuntos $\mu^*$-mensuráveis formam uma $\sigma$-álgebra
$\mathcal
M$, e $\mu^*\restriction_{\mathcal M}$ é uma \omterm{def:b3:measure:measure}{medida}. Além disso, todo conjunto com $\mu^*(N) = 0$ pertence
a $\mathcal M$ (a \omterm{def:b3:measure:measure}{medida} é \emph{\omterm{def:b3:complete:complete}{completa}}).\index{teorema de Carathéodory}
\end{theorem}

\begin{proof}
$\mathcal M$ contém $\varnothing$ e é estável por complementação (a condição de
definição é simétrica em $A$ e $X\setminus A$). \emph{Reuniões finitas}:
sejam $A, B \in \mathcal
M$ e $E$ arbitrário; repartindo $E$ por $A$ e,
depois, cada pedaço por $B$:
\[
\mu^*(E) = \mu^*(E\cap A\cap B) + \mu^*(E\cap A\setminus B) +
\mu^*(E\cap B\setminus A) + \mu^*(E\setminus(A\cup B)).
\]
Os três primeiros pedaços cobrem $E \cap (A \cup B)$, de modo que a subaditividade dá
$\mu^*(E) \geq \mu^*(E\cap(A\cup B)) +
\mu^*(E\setminus(A\cup B))$: $A \cup B \in \mathcal M$. Por indução, obtêm-se as reuniões finitas; com os
complementos, ficam disponíveis as manipulações de disjunção finita.

\emph{Aditividade em $\mathcal M$}: para $A, B \in
\mathcal M$ disjuntos e $E$ qualquer:
$\mu^*(E\cap(A\sqcup B)) = \mu^*(E\cap
A) + \mu^*(E \cap B)$ (repartindo por $A$); por indução,
\[
\mu^*\Bigl(E \cap \bigsqcup_{k\leq n}A_k\Bigr)
= \sum_{k\leq n}\mu^*(E\cap A_k).
\tag{$*$}
\]

\emph{Reuniões enumeráveis}: sejam $(A_k) \subseteq \mathcal M$ disjuntos (o que basta, por
disjuntificação dentro da álgebra $\mathcal M$), $A = \bigsqcup A_k$ e $E$ arbitrário.
Usando $\bigsqcup_{k\leq n}A_k \in \mathcal M$ e a monotonicidade:
\[
\mu^*(E) = \mu^*\Bigl(E\cap\bigsqcup_{k\leq n}A_k\Bigr) +
\mu^*\Bigl(E\setminus\bigsqcup_{k\leq n}A_k\Bigr)
\geq \sum_{k \leq n}\mu^*(E\cap A_k) + \mu^*(E\setminus A)
\]
por ($*$). Faça $n \to \infty$ e use a subaditividade enumerável ao
contrário:
\[
\mu^*(E) \geq \sum_{k}\mu^*(E\cap A_k) + \mu^*(E\setminus A)
\geq \mu^*(E \cap A) + \mu^*(E\setminus A) \geq \mu^*(E):
\]
todas as desigualdades são igualdades. Isso demonstra tanto $A \in
\mathcal M$ quanto,
tomando $E = A$, a aditividade enumerável de $\mu^*$ em $\mathcal M$.

\emph{Conjuntos nulos}: se $\mu^*(N) = 0$, então, para $E$ qualquer, $\mu^*(E \cap N) + \mu^*(E\setminus N) \leq 0 + \mu^*(E)$:
$N \in \mathcal M$.
\end{proof}

\section{A medida de Lebesgue em \texorpdfstring{$\R$}{R}}

\begin{definition}\label{def:b3:measure:lebesgueouter}
A \emph{medida exterior de Lebesgue}\index{medida de Lebesgue} de $A
\subseteq \R$
é
\[
\lambda^*(A) = \inf\Bigl\{\sum_{n} (b_n - a_n) :
A \subseteq \bigcup_n \intoo{a_n}{b_n}\Bigr\}
\]
(coberturas enumeráveis por intervalos abertos).
\end{definition}

\begin{lemma}\label{lem:b3:measure:outerbasics}
$\lambda^*$ é uma \omterm{def:b3:measure:outer}{medida exterior}, invariante por translações, e $\lambda^*(I) = \ell(I)$ (o
comprimento) para todo intervalo $I$.
\end{lemma}

\begin{proof}
\omterm{def:b3:measure:outer}{Medida exterior}: $\varnothing$ é coberto por intervalos arbitrariamente
pequenos; a monotonicidade é clara; subaditividade: dadas coberturas de
cada $A_n$ a menos de $\varepsilon 2^{-n}$ do ínfimo, sua reunião cobre $\bigcup A_n$ com
comprimento total $\leq \sum
\lambda^*(A_n) + \varepsilon$. Invariância por translação: translade as
coberturas.

Comprimento: basta tratar $I = \intcc ab$ (os outros tipos diferem pelas
extremidades, que têm \omterm{def:b3:measure:outer}{medida exterior} $0$: cubra-as por intervalos
minúsculos; depois, comprima por comparações do tipo $\intcc{a+\varepsilon}{b -
\varepsilon} \subseteq \intoo ab$). $\lambda^*(\intcc ab) \leq b - a$:
cubra por $\intoo{a-\varepsilon}{b+\varepsilon}$. Reciprocamente, seja $\intcc
ab \subseteq \bigcup_n\intoo{a_n}{b_n}$: por \emph{\omterm{def:b3:topology:compact}{compacidade}}
(Borel--Lebesgue, \cref{thm:b3:topology:metriccompact}), um número finito de intervalos
basta, digamos $I_1, \dots, I_N$. Mostramos que $\sum_{k\leq N}(b_k -
a_k) \geq b - a$ por indução sobre $N$:
escolha $I_{k_1} \ni a$; se $b_{k_1} > b$, terminamos ($b_{k_1} - a_{k_1} > b - a$); do contrário, o segmento
$\intcc{b_{k_1}}b$ é coberto pelos $N - 1$ intervalos restantes, e a indução dá
$\sum_{k \neq k_1}(b_k - a_k)
\geq b - b_{k_1}$, ao passo que $b_{k_1} - a_{k_1} > b_{k_1} - a$: some.
\end{proof}

\begin{theorem}[Medida de Lebesgue]\label{thm:b3:measure:lebesgue}
Todo boreliano de $\R$ é $\lambda^*$-mensurável. A restrição $\lambda$ de
$\lambda^*$ à $\sigma$-álgebra $\mathcal L = \mathcal M_{\lambda^*} \supseteq \mathcal B(\R)$ (a \emph{$\sigma$-álgebra de
Lebesgue}) é a única \omterm{def:b3:measure:measure}{medida} em $\mathcal B(\R)$ que atribui a cada intervalo seu
comprimento; ela é invariante por translações e $\sigma$-finita.
\end{theorem}

\begin{proof}
Pelo~\cref{thm:b3:measure:caratheodory}, basta mostrar que cada semirreta $A = \intoo{-\infty}c$ é
$\lambda^*$-mensurável (as semirretas geram $\mathcal B$, \cref{def:b3:measure:borel}). Sejam
$E \subseteq \R$ com $\lambda^*(E) < \infty$ e $\bigcup I_n \supseteq E$ uma cobertura com $\sum\ell(I_n) \leq \lambda^*(E) + \varepsilon$. Cada $I_n$ se
reparte nos dois intervalos $I_n' = I_n \cap A$ e $I_n'' = I_n\setminus A$ (um intervalo menos uma
semirreta é um intervalo) com $\ell(I_n') + \ell(I_n'') = \ell(I_n)$; os $I_n'$ cobrem $E \cap A$ e os
$I_n''$ cobrem $E \setminus A$ (aumente cada um até um intervalo aberto de
comprimento $\ell +
\varepsilon2^{-n}$ para permanecer dentro da definição), de modo que
\[
\lambda^*(E\cap A) + \lambda^*(E\setminus A)
\leq \sum_n\bigl(\ell(I_n') + \ell(I_n'')\bigr) + 2\varepsilon
\leq \lambda^*(E) + 3\varepsilon .
\]
Unicidade: duas \omterm{def:b3:measure:measure}{medidas} que coincidem com o comprimento no
$\pi$-sistema dos intervalos $\intoc ab$ (finitas neles) coincidem em
$\sigma(\text{intervalos}) = \mathcal B$ pelo~\cref{thm:b3:measure:uniqueness} com $P_k = \intoc{-k}k$. $\sigma$-finitude: $\R = \bigcup(-k, k]$.
\end{proof}

\begin{theorem}[Regularidade]\label{thm:b3:measure:regularity}
Para todo $A \in \mathcal L$:\index{regularidade da medida de Lebesgue}
\[
\lambda(A) = \inf\{\lambda(U) : U \supseteq A \text{ aberto}\}
= \sup\{\lambda(K) : K \subseteq A \text{ compacto}\}.
\]
\end{theorem}

\begin{proof}
\emph{Exterior}: uma cobertura $\bigcup I_n$ com $\sum\ell(I_n) \leq
\lambda(A) + \varepsilon$ é um aberto $U \supseteq A$ com
$\lambda(U) \leq \lambda(A) + \varepsilon$ (subaditividade); se $\lambda(A) = \infty$, o enunciado é trivial.
\emph{\omterm{def:b3:topology:interior}{Interior}}: primeiro suponha $A$ limitado, $A \subseteq [-M, M]$. Escolha um
aberto $U \supseteq ([-M,M]\setminus A)$ com $\lambda(U) \leq
\lambda([-M,M]\setminus A) + \varepsilon$; então $K = [-M,
M]\setminus U$ é \omterm{def:b3:topology:compact}{compacto}, $K \subseteq A$, e
\[
\lambda(K) = \lambda([-M,M]) - \lambda([-M,M]\cap U)
\geq \lambda([-M,M]) - \bigl(\lambda([-M,M]) - \lambda(A) +
\varepsilon\bigr) = \lambda(A) - \varepsilon .
\]
Para $A$ geral: $\lambda(A) = \lim_M\lambda(A\cap[-M,M])$ (\omterm{def:b3:topology:continuity}{continuidade} por baixo) e aplique o caso
limitado por dentro.
\end{proof}

\begin{example}\label{ex:b3:measure:cantor}
O conjunto de Cantor (\cref{exo:b3:topology:10}) satisfaz $\lambda(C) = 0$: $C \subseteq C_n$, reunião de
$2^n$ intervalos de comprimento $3^{-n}$, de modo que $\lambda(C) \leq (2/3)^n \to 0$. Um
conjunto nulo não enumerável --- a cardinalidade não enxerga a \omterm{def:b3:measure:measure}{medida}.
Reciprocamente, os \emph{conjuntos de Cantor gordos}
(\cref{exo:b3:measure:5}) são nunca densos e têm \omterm{def:b3:measure:measure}{medida} positiva: a \omterm{def:b3:topology:topology}{topologia}
tampouco enxerga a \omterm{def:b3:measure:measure}{medida}. O problema de fim de semana leva essa
interação à sua conclusão marcante: existem conjuntos
Lebesgue-mensuráveis que não são borelianos.
\end{example}

\begin{theorem}[Vitali]\label{thm:b3:measure:vitali}
Não existe \omterm{def:b3:measure:measure}{medida} sobre \emph{todos} os subconjuntos de $\R$ que
seja invariante por translações e atribua a cada intervalo seu
comprimento. Em particular, $\mathcal L \neq \mathcal P(\R)$: existem conjuntos não
mensuráveis.\index{conjunto não mensurável de Vitali}
\end{theorem}

\begin{proof}
Suponha que $\mu$ fosse uma tal \omterm{def:b3:measure:measure}{medida}. Em $\intcc01$, considere a
equivalência $x \sim y \iff x - y \in \Q$; pelo \emph{axioma da escolha}, tome um
representante em $\intcc01$ por classe: um conjunto $V$. Para $q \in \Q\cap\intcc{-1}1$, os
transladados $V + q$ são dois a dois disjuntos (dois pontos de
$V$ que diferissem por um racional seriam equivalentes e, no
entanto, representantes distintos) e
\[
\intcc01 \subseteq \bigsqcup_{q \in \Q\cap\intcc{-1}1}(V + q)
\subseteq \intcc{-1}2 :
\]
a primeira inclusão porque todo $x \in \intcc01$ difere de seu representante
$v$ por um racional $q = x - v \in
\intcc{-1}1$. A monotonicidade e a
$\sigma$-aditividade dão
\[
1 \leq \sum_{q}\mu(V + q) \leq 3,
\qquad\text{com } \mu(V + q) = \mu(V) \text{ para todo } q .
\]
Uma soma infinita da constante $\mu(V)$ vale $0$ ou
$\infty$: as duas cotas não podem valer simultaneamente. Logo, não
existe uma tal $\mu$ --- e $V
\notin \mathcal L$, pois $\lambda$ em $\mathcal L$ tem todas as
propriedades usadas.
\end{proof}

\begin{remark}\label{rem:b3:measure:banachtarski}
Em $\R^3$ o fracasso é mais dramático: o paradoxo de
Banach--Tarski decompõe uma bola em cinco pedaços que se remontam, por
rotações e translações, em \emph{duas} bolas do mesmo raio --- de modo
que não existe sequer um volume finitamente aditivo e invariante por
rotações sobre todos os subconjuntos de $\R^3$. Os pedaços são,
claro, não mensuráveis. A mensurabilidade não é cautela burocrática; é
a fronteira da coerência.
\end{remark}

\begin{method}\label{met:b3:measure:goodsets}
O \emph{princípio dos bons conjuntos}: para demonstrar que todos os
conjuntos de $\sigma(\mathcal E)$ têm uma propriedade, mostre que os bons conjuntos
formam uma $\sigma$-álgebra (ou um $\lambda$-sistema, se a propriedade
for de natureza medida-teórica e $\mathcal E$ for um $\pi$-sistema --- e
então use Dynkin) contendo $\mathcal E$. Quase toda demonstração deste
capítulo e do próximo é um caso particular. Para demonstrar que duas
\omterm{def:b3:measure:measure}{medidas} são iguais: verifique-o em um $\pi$-sistema gerador, mais a
$\sigma$-finitude (\cref{thm:b3:measure:uniqueness}). Para construir uma \omterm{def:b3:measure:measure}{medida}:
construa uma \omterm{def:b3:measure:outer}{medida exterior} por coberturas e invoque Carathéodory.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:measure:1}
(a) Mostre que $\{A \subseteq X : A$ ou $X\setminus A$ é enumerável$\}$ é uma
$\sigma$-álgebra: a gerada pelos conjuntos unitários.
(b) Mostre que $\mathcal B(\R)$ é gerada por cada uma das famílias: intervalos
abertos; intervalos fechados; semirretas $\intoc{-\infty}a$; semirretas com $a \in \Q$.
(c) A família das \emph{reuniões finitas disjuntas de intervalos}
$\intoc ab$ é uma $\sigma$-álgebra? É uma álgebra (estável por
complementação e reuniões finitas)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:measure:2}
(a) Demonstre o princípio da inclusão--exclusão para uma \omterm{def:b3:measure:measure}{medida} finita:
$\mu(A\cup
B) = \mu(A) + \mu(B) - \mu(A\cap B)$, e a versão com três conjuntos.
(b) Dê um exemplo mostrando que a \omterm{def:b3:topology:continuity}{continuidade} por cima
(\cref{prop:b3:measure:basics}(d)) falha sem a hipótese de finitude.
(c) Mostre que um conjunto enumerável tem \omterm{def:b3:measure:lebesgueouter}{medida de Lebesgue} nula.
Deduza $\lambda(\Q) = 0$ e $\lambda(\intcc01\setminus\Q) =
1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:3}
Sejam $\mu, \nu$ \omterm{def:b3:measure:measure}{medidas} de probabilidade em $\mathcal B(\R)$ com $\mu(\intoc{-\infty}t) = \nu(\intoc{-\infty}t)$ para todo $t \in \R$.
Mostre que $\mu = \nu$. (Isso faz da \emph{função de distribuição} $F(t) = \mu(\intoc{-\infty}t)$ um
invariante \omterm{def:b3:complete:complete}{completo} --- o fundamento do~\cref{ch:b3:probability}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:4}
(Borel--Cantelli, versão de \omterm{def:b3:measure:measure}{medida}) Sejam $(A_n)$ mensuráveis com
$\sum_n\mu(A_n) < \infty$, e $\limsup A_n =
\bigcap_N\bigcup_{n\geq N}A_n$ (os pontos que pertencem a uma infinidade de $A_n$).
Mostre que $\mu(\limsup A_n) = 0$. Aplicação: para quase todo $x \in \intcc01$, apenas um número
finito de $n$ satisfaz $\abs{x - p/q_n} \leq 4^{-n}$ para o $n$-ésimo racional
$p/q_n$ de uma enumeração de $\Q\cap\intcc01$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:5}
(Conjunto de Cantor gordo) Repita a construção de Cantor em $\intcc01$, mas,
na etapa $n$, retire de cada um dos $2^{n-1}$ intervalos apenas um
intervalo aberto \emph{centrado} de comprimento $4^{-n}$. Mostre que
o $K = \bigcap K_n$ resultante é \omterm{def:b3:topology:compact}{compacto}, tem \omterm{def:b3:topology:interior}{interior} vazio (nenhum intervalo
sobrevive) e
\[
\lambda(K) = 1 - \sum_{n\geq1}2^{n-1}4^{-n} = \tfrac12 :
\]
um conjunto nunca denso de \omterm{def:b3:measure:measure}{medida} $\frac12$. Deduza um subconjunto
\emph{\omterm{rem:b3:complete:meagre}{magro}} de $\intcc01$ de \omterm{def:b3:measure:measure}{medida} total $1$ e um subconjunto aberto
denso de \omterm{def:b3:measure:measure}{medida} $< \varepsilon$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:6}
Seja $\mu$ uma \omterm{def:b3:measure:measure}{medida} em $\mathcal B(\R)$, invariante por translações, com
$c = \mu(\intoc01) < \infty$. Mostre que $\mu =
c\,\lambda$ em $\mathcal B(\R)$. \emph{(Calcule $\mu$ nos intervalos
diádicos dividindo $\intoc01$ em $2^n$ transladados e invoque em seguida
o~\cref{thm:b3:measure:uniqueness}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:7}
(Aproximação) Sejam $A \in \mathcal L$ com $\lambda(A) <
\infty$ e $\varepsilon > 0$. Mostre que existe uma
reunião \emph{finita} de intervalos $B$ com $\lambda(A\,\triangle\,B) <
\varepsilon$ ($\triangle$ = diferença
simétrica). \emph{(Regularidade: comprima $K \subseteq A \subseteq U$ e use a estrutura do
aberto $U$ como reunião enumerável de intervalos, mais a \omterm{def:b3:topology:compact}{compacidade}
de $K$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:measure:8}
(Steinhaus) Seja $A \in \mathcal L$ com $\lambda(A) > 0$. Mostre que $A - A = \{x - y : x, y \in A\}$ contém um intervalo
em torno de $0$. \emph{(Reduza a $\lambda(A) < \infty$; por regularidade no estilo
do~\cref{exo:b3:measure:7}, encontre um intervalo $I$ com $\lambda(A \cap I) > \frac34\ell(I)$; então, para
$\abs t <
\frac12\ell(I)$, os conjuntos $A\cap I$ e $(A\cap I) + t$ estão ambos em um intervalo de
comprimento $\frac32\ell(I)$ e têm \omterm{def:b3:measure:measure}{medida} total $> \frac32\ell(I)$: eles têm de se
intersectar.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:measure:9}
Mostre que todo $A \in \mathcal L$ com $\lambda(A) > 0$ contém um subconjunto não mensurável.
\emph{(Intersecte $A$ com os transladados $V + q$ do conjunto de
Vitali: se todos os $A \cap (V+q)$ fossem mensuráveis, cada um seria nulo pelo
argumento do~\cref{thm:b3:measure:vitali} --- Steinhaus (\cref{exo:b3:measure:8}) ajuda: um
conjunto mensurável de \omterm{def:b3:measure:measure}{medida} positiva dentro de $V + q$ daria a
$(V+q) - (V+q) \supseteq$ um intervalo, contradizendo o fato de que esse conjunto de
diferenças encontra $\Q$ apenas em $0$; conclua por
subaditividade.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:10}
Mostre que $A \subseteq \R$ com $\lambda^*(A) < \infty$ é Lebesgue-mensurável se, e somente se, para
todo $\varepsilon > 0$ existe um aberto $U \supseteq A$ com $\lambda^*(U \setminus A) <
\varepsilon$, e se, e somente se, existe
um conjunto $G_\delta$ $G \supseteq A$ com $\lambda^*(G\setminus A) = 0$. (Assim, os conjuntos de Lebesgue são
os borelianos a menos de conjuntos nulos.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:11}
(\omterm{def:b3:topology:continuity}{Continuidade} ao longo de limites monótonos, e sua precisão)
(a) Mostre que, para conjuntos mensuráveis, $\mu(\liminf A_n) \leq
\liminf\mu(A_n)$ (Fatou para
conjuntos) e que, se $\mu\bigl(\bigcup A_n\bigr) < \infty$, também $\limsup\mu(A_n) \leq \mu(\limsup A_n)$.
(b) Exiba, para a \omterm{def:b3:measure:lebesgueouter}{medida de Lebesgue} em $\R$, uma sequência com
$\mu(A_n) = 1$ para todo $n$ e, no entanto, $\mu(\limsup A_n) = 0$: a hipótese de finitude na
segunda desigualdade não é decorativa.
(c) Deduza: se $\sum\mu(A_n) < \infty$, então $\mu(\limsup A_n) = 0$ (Borel--Cantelli de novo), e, se os
$A_n$ crescem ou decrescem (com $\mu(A_1) < \infty$ no caso decrescente), $\mu(\lim A_n) = \lim\mu(A_n)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:measure:12}
(Teorema de Egorov) Sejam $\mu(X) < \infty$ e $f_n \to f$ pontualmente, todas mensuráveis
(a valores reais). Para $k, N \geq 1$, ponha
\[
E_{k,N} = \bigcap_{n \geq N}\Bigl\{x : \abs{f_n(x) - f(x)}
\leq \tfrac1k\Bigr\} .
\]
(a) Mostre que, para $k$ fixado, $E_{k,N} \nearrow X$ quando $N \to
\infty$, e deduza a
existência de $N_k$ com $\mu(X \setminus E_{k,N_k})
\leq \varepsilon2^{-k}$.
(b) Conclua o \emph{teorema de Egorov}: para todo $\varepsilon
> 0$ existe um
conjunto mensurável $A$ com $\mu(X\setminus A) \leq
\varepsilon$ tal que $f_n \to f$ \emph{uniformemente
em $A$} --- a convergência pontual é convergência uniforme fora de
um conjunto arbitrariamente pequeno.
(c) Mostre que o teorema falha em $(\R, \lambda)$: as ondas móveis $f_n = \mathbf 1_{\intcc n{n+1}}$ convergem
pontualmente para $0$, mas não uniformemente no complementar de
nenhum conjunto de \omterm{def:b3:measure:measure}{medida} finita. Onde (a) usou $\mu(X) < \infty$?
\end{exercise}

\section{Problema: a escada de Cantor--Vitali e um conjunto mensurável
que não é boreliano}

\begin{problem}[{Problema de fim de semana --- a escada do diabo, e
$\mathcal B(\R) \subsetneq \mathcal L$}]
\label{pb:b3:measure:1}
Construímos a \emph{função de Cantor--Vitali} (a escada do diabo),
usamo-la para transportar \omterm{def:b3:measure:measure}{medida} de maneira patológica e concluímos com
um teorema que nenhum argumento leve fornece: existem conjuntos
Lebesgue-mensuráveis que não são borelianos. Notação: $C$ é o
conjunto de Cantor, $C_n$ sua $n$-ésima etapa ($2^n$
intervalos de comprimento $3^{-n}$), e todo $x \in C$ tem dígitos
ternários $x = \sum 2b_n3^{-n}$, $b_n \in \{0,1\}$ (\cref{exo:b3:topology:10}).

\medskip
\noindent\textbf{Parte I --- A escada.} Defina $c_0(x) = x$ e $c_{n+1}$ a partir de
$c_n$ por
\[
c_{n+1}(x) = \begin{cases}
\tfrac12\,c_n(3x) & 0 \leq x \leq \tfrac13,\\[2pt]
\tfrac12 & \tfrac13 \leq x \leq \tfrac23,\\[2pt]
\tfrac12 + \tfrac12\,c_n(3x - 2) & \tfrac23 \leq x \leq 1.
\end{cases}
\]
\begin{enumerate}
  \item Mostre que cada $c_n$ é \omterm{def:b3:topology:continuity}{contínua}, não decrescente, com $c_n(0) = 0$,
        $c_n(1) = 1$, e que $\norm{c_{n+1} - c_n}_\infty \leq
        \tfrac12\norm{c_n - c_{n-1}}_\infty$.
  \item Deduza que $(c_n)$ converge uniformemente para uma função
        $c$ \omterm{def:b3:topology:continuity}{contínua} e não decrescente com $c(0) = 0$, $c(1) = 1$ (\emph{a}
        função de Cantor--Vitali), que satisfaz as mesmas relações
        autossemelhantes que os $c_{n+1}$ acima.
  \item Mostre que $c$ é constante em cada \omterm{def:b3:topology:components}{componente conexa} de
        $\intcc01\setminus C$ e que, para $x = \sum_n
        2b_n3^{-n} \in C$, $c(x) = \sum_n b_n2^{-n}$: a escada lê em binário os dígitos
        de Cantor (a função $g$ do~\cref{pb:b3:topology:1}, tornada monótona e
        global).
  \item Deduza que $c$ é derivável, com $c' = 0$, em todo ponto
        de $\intcc01\setminus C$: $c' = 0$ \emph{$\lambda$-quase em toda parte}
        (\cref{ex:b3:measure:cantor}). Conclua que o teorema fundamental do cálculo
        falha para $c$:
        \[
        c(1) - c(0) = 1 \neq 0 = \int_0^1 c'(t)\,\dd t
        \]
        (a integral sendo tomada sobre o conjunto de \omterm{def:b3:measure:measure}{medida} total em
        que $c' = 0$; antecipando o~\cref{ch:b3:lebesgue}, os conjuntos nulos
        não afetam integrais). Que hipótese do teorema fundamental
        $\mathcal C^1$ é violada?
  \item Mostre que $c(C) = \intcc01$: o conjunto nulo $C$ é levado \emph{sobre}
        um conjunto de \omterm{def:b3:measure:measure}{medida} total.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O \omterm{def:b3:topology:continuity}{homeomorfismo} torto.} Seja $h(x) = \frac{c(x) + x}{2}$.
\begin{enumerate}[resume]
  \item Mostre que $h \colon \intcc01 \to \intcc01$ é um \omterm{def:b3:topology:continuity}{homeomorfismo} (estritamente crescente,
        \omterm{def:b3:topology:continuity}{contínuo}, sobrejetor).
  \item Mostre que $\lambda\bigl(h(\intcc01\setminus C)\bigr) =
        \tfrac12$: em cada lacuna de comprimento $\ell$,
        $h$ age como uma aplicação afim de inclinação $\tfrac12$, e as
        lacunas têm comprimento total $1$.
  \item Deduza que $\lambda\bigl(h(C)\bigr) = \tfrac12$: a imagem \omterm{def:b3:topology:continuity}{homeomorfa} de um conjunto nulo pode
        ter \omterm{def:b3:measure:measure}{medida} positiva. (Em que ponto isso contraria a intuição
        ingênua sobre ``tamanho''?)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Um conjunto mensurável que não é
boreliano.}
\begin{enumerate}[resume]
  \item Pelo~\cref{exo:b3:measure:9}, escolha um $W
        \subseteq h(C)$ não mensurável. Mostre que
        $Z = h^{-1}(W) \subseteq C$ é Lebesgue-mensurável. \emph{(É subconjunto de um conjunto
        nulo; \omterm{def:b3:complete:complete}{completude}, \cref{thm:b3:measure:caratheodory}.)}
  \item Mostre que a pré-imagem de um boreliano por uma \omterm{def:b3:topology:continuity}{aplicação
        contínua} é boreliana. \emph{(Princípio dos bons conjuntos:
        $\{B : h^{-1}(B) \in \mathcal B\}$ é uma $\sigma$-álgebra que contém os abertos ---
        atenção ao sentido da aplicação.)}
  \item Conclua que $Z$ \emph{não} é boreliano: se fosse, $W = (h^{-1})^{-1}(Z)$
        seria boreliano (aplique a questão 10 à \omterm{def:b3:topology:continuity}{aplicação contínua}
        $h^{-1}$), logo mensurável --- contradição. Portanto,
        \[
        \boxed{\ \mathcal B(\R) \subsetneq \mathcal L\ }
        \]
        e a \omterm{def:b3:complete:complete}{completude} amplia genuinamente o mundo boreliano.
  \item Exiba uma função Lebesgue-mensurável $g$ e uma função
        \omterm{def:b3:topology:continuity}{contínua} $\varphi$ tais que $g \circ
        \varphi$ não seja Lebesgue-mensurável: a
        mensurabilidade, ao contrário da \omterm{def:b3:topology:continuity}{continuidade}, não se compõe.
        \emph{(Tome $g =
        \mathbf 1_Z$ e $\varphi = h^{-1}$, antecipando a definição de funções
        mensuráveis do~\cref{ch:b3:lebesgue}: as pré-imagens de borelianos são
        conjuntos de Lebesgue. Onde é preciso cuidado quanto a qual
        $\sigma$-álgebra se usa no alvo?)}
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Epílogo.}
\begin{enumerate}[resume]
  \item Ordene por inclusão estrita as classes seguintes e justifique
        cada estrita com um exemplo deste capítulo ou de seu problema:
        conjuntos enumeráveis; borelianos nulos; conjuntos nulos de
        Lebesgue; borelianos; conjuntos de Lebesgue; conjuntos
        arbitrários.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- A \omterm{def:b3:measure:measure}{medida} de Cantor: massa em um conjunto
nulo.} A escada é a função de distribuição de uma \omterm{def:b3:measure:measure}{medida} notável, que
construímos agora com as ferramentas do próprio capítulo.
\begin{enumerate}[resume]
  \item (Lebesgue--Stieltjes, existência) Seja $F\colon\R\to\R$ não decrescente,
        \omterm{def:b3:topology:continuity}{contínua} e limitada. Em intervalos semiabertos, defina $\rho\bigl(\intoc ab\bigr) = F(b) -
        F(a)$ e,
        para $A \subseteq \R$,
        \[
        \mu_F^*(A) = \inf\Bigl\{\sum_k\bigl(F(b_k) -
        F(a_k)\bigr) : A \subseteq
        \bigcup_k\intoc{a_k}{b_k}\Bigr\} .
        \]
        Mostre que $\mu_F^*$ é uma \omterm{def:b3:measure:outer}{medida exterior} e que $\mu_F^*\bigl(\intoc ab\bigr) = F(b) - F(a)$ \emph{(imite
        o argumento de \omterm{def:b3:topology:compact}{compacidade} do~\cref{thm:b3:measure:lebesgue}, aumentando cada
        $\intoc{a_k}{b_k}$ até um intervalo aberto a um custo em $F$ de $\leq \varepsilon2^{-k}$ ---
        onde se usa a \omterm{def:b3:topology:continuity}{continuidade} de $F$?)}.
  \item Mostre que todo boreliano é $\mu_F^*$-mensurável no sentido de
        Carathéodory \emph{(como no caso de Lebesgue, basta testar
        semirretas; siga a demonstração da aplicação
        do~\cref{thm:b3:measure:caratheodory})}, de modo que $\mu_F = \mu_F^*$ restrita a $\mathcal B(\R)$ é uma
        \omterm{def:b3:measure:measure}{medida} com $\mu_F(\intoc ab) = F(b) - F(a)$: a \emph{\omterm{def:b3:measure:measure}{medida} de Lebesgue--Stieltjes} de
        $F$.
  \item Aplique isso à escada ($F = c$ estendida por $0$ em
        $\R_-$ e por $1$ em $\intco1\infty$): a \emph{\omterm{def:b3:measure:measure}{medida} de Cantor}
        $\mu$. Mostre que $\mu(\R) = 1$, que toda lacuna do conjunto de
        Cantor é $\mu$-nula ($c$ é ali constante) e conclua
        \[
        \mu(C) = 1, \qquad \lambda(C) = 0 :
        \]
        $\mu$ e $\lambda$ vivem em suportes disjuntos ($C$ e seu
        complementar). Duas \omterm{def:b3:measure:measure}{medidas} nessa situação são ditas
        \emph{mutuamente singulares}, o que se escreve $\mu \perp
        \lambda$.
  \item Mostre que $\mu$ não tem átomos: $\mu(\{x\}) = 0$ para todo $x$
        \emph{(\omterm{def:b3:topology:continuity}{continuidade} de $c$)}. Uma \omterm{def:b3:measure:measure}{medida} de probabilidade sem
        átomos, carregada por um \omterm{def:b3:topology:compact}{compacto} de \omterm{def:b3:measure:lebesgueouter}{medida de Lebesgue} nula:
        compare com as únicas \omterm{def:b3:measure:measure}{medidas} vistas até aqui.
  \item (Lançamentos de moeda disfarçados) Para uma palavra $(\varepsilon_1,
        \dots, \varepsilon_m) \in \{0,1\}^m$, seja
        $C_{\varepsilon}$ o conjunto dos $x \in C$ cujos dígitos ternários satisfazem
        $b_i(x) = \varepsilon_i$ para $i \leq m$ (uma das $2^m$ peças de Cantor de
        profundidade $m$). Mostre que $\mu(C_\varepsilon) = 2^{-m}$ \emph{(a escada sobe
        $2^{-m}$ ao atravessar essa peça: use a questão 3 da Parte
        I)}. A \omterm{def:b3:measure:measure}{medida} de Cantor é a lei de uma sequência infinita de
        lançamentos de uma moeda honesta lidos em ternário --- o~\cref{ch:b3:probability} tornará isso exato.
  \item Demonstre a autossemelhança: para todo boreliano $A$,
        \[
        \mu(A) = \tfrac12\,\mu(3A) +
        \tfrac12\,\mu(3A - 2) ,
        \]
        em que $3A - 2 = \{3x - 2 : x \in A\}$ \emph{(verifique-a nos intervalos geradores $\intoc ab$
        via as relações autossemelhantes de $c$ e invoque em seguida
        a unicidade, \cref{thm:b3:measure:uniqueness})}.
  \item Mostre que a reflexão $s(x) = 1 - x$ preserva $\mu$: $\mu(s(A)) = \mu(A)$ \emph{(via
        $c(1 - x) = 1 -
        c(x)$, que decorre da simetria da construção --- demonstre-a)}.
  \item Calcule os dois primeiros momentos de $\mu$, isto é, de um
        ponto aleatório $X$ de lei $\mu$ (as integrais podem ser
        tratadas como limites de somas sobre as peças de profundidade
        $m$, antecipando o~\cref{ch:b3:lebesgue}): a simetria dá $\int x\,\dd\mu = \frac12$, e a
        autossemelhança dá
        \[
        \int x^2\,\dd\mu = \frac38,
        \qquad\text{logo}\qquad
        \operatorname{Var}(X) = \frac18 .
        \]
        Compare com a lei uniforme em $\intcc01$ (variância $\frac1{12}$): a massa de
        Cantor, empurrada para as bordas, espalha-se \emph{mais}.
  \item Mostre que o \emph{suporte} (\omterm{def:b3:topology:topology}{topológico}) de $\mu$ --- o
        menor fechado de \omterm{def:b3:measure:measure}{medida} total --- é exatamente $C$.
  \item (Síntese) A escada $c$ é \omterm{def:b3:topology:continuity}{contínua} e não decrescente e, ainda
        assim, falha o teorema fundamental do cálculo (Parte I); a
        \omterm{def:b3:measure:measure}{medida} $\mu_c$ é uma probabilidade, sem átomos, singular em
        relação a $\lambda$. Explique em um parágrafo curto como essas são
        duas faces de um mesmo fenômeno e enuncie a moral geral: as
        funções não decrescentes correspondem a \omterm{def:b3:measure:measure}{medidas} ($F \leftrightarrow \mu_F$), a
        derivabilidade q.t.p.\ corresponde à ``parte absolutamente
        \omterm{def:b3:topology:continuity}{contínua}'', e $c$ é a testemunha padrão de que uma $F$
        \omterm{def:b3:topology:continuity}{contínua} pode não carregar parte absolutamente \omterm{def:b3:topology:continuity}{contínua}
        \emph{alguma}.
  \item (O módulo de \omterm{def:b3:topology:continuity}{continuidade} exato) Seja $s =
        \frac{\ln 2}{\ln 3}$. Mostre que $c$
        é hölderiana de expoente $s$:
        \[
        \abs{c(x) - c(y)} \leq 4\,\abs{x - y}^{s}
        \qquad (x, y \in \intcc01),
        \]
        e que nenhum expoente $t > s$ pode funcionar, nem mesmo
        localmente. Deduza a forma medida-teórica: para todo $x$ e
        todo $r \in \intoc01$,
        \[
        \mu\bigl(\intcc{x - r}{x + r}\bigr) \leq 8\,r^{s} .
        \]
        \emph{(Compare uma grade triádica de profundidade $m$ com a
        escala de $\abs{x - y}$; a questão 18 dá a subida ao longo de cada
        peça. O expoente $s$ é a dimensão \omterm{def:b3:topology:hausdorff}{de Hausdorff} de $C$,
        como dirão cursos posteriores.)}
  \item (A autossemelhança caracteriza $\mu$) Demonstre a recíproca
        da questão 19: se $\nu$ é uma \omterm{def:b3:measure:measure}{medida} de probabilidade em
        $\mathcal B(\R)$ carregada por $\intcc01$ e satisfaz
        \[
        \nu(A) = \tfrac12\,\nu(3A) + \tfrac12\,\nu(3A - 2)
        \qquad (A \in \mathcal B(\R)),
        \]
        então $\nu = \mu$. \emph{(Itere a relação $m$ vezes para espalhar
        $\nu$ sobre as $2^m$ peças de Cantor de profundidade
        $m$, estime $\nu(\intoc ab)$ contra a contagem de peças dentro de
        $\intoc ab$, e faça $m \to \infty$; termine com o~\cref{thm:b3:measure:uniqueness}.)}
\end{enumerate}
\end{problem}
