\chapter{A transformada de Fourier}\label{ch:b3:fouriertransform}

As séries de Fourier decompõem sinais periódicos em harmônicos
discretos; a \emph{transformada} de Fourier faz o mesmo com sinais em
toda a reta, com um \omterm{def:b3:topology:continuity}{contínuo} de frequências. Ela converte derivação em
multiplicação, convolução em produto e gaussianas em gaussianas --- as
razões pelas quais resolve equações diferenciais, move o processamento
de sinais e vai demonstrar o teorema central do limite no~\cref{ch:b3:clt}. Este capítulo desenvolve a teoria $L^1$
(Riemann--Lebesgue, inversão, injetividade), a \omterm{def:b3:fouriertransform:schwartz}{classe de Schwartz}, em
que a transformada é uma bijeção \omterm{prop:b3:galois:perfect}{perfeita}, e a teoria $L^2$
(Plancherel: a transformada é, a menos de uma constante, um operador
unitário), com duas aplicações de vitrine: a equação do calor, resolvida
de ponta a ponta no problema de fim de semana, e a fórmula de soma de
Poisson. Convenção:
\[
\hat f(\xi) = \int_\R f(x)\,\eu^{-\iu\xi x}\,\dd x .
\]

\section{A transformada em \texorpdfstring{$L^1$}{L1}}

\begin{proposition}\label{prop:b3:fouriertransform:basic}
Para $f \in L^1(\R)$: $\hat f$ está bem definida, é limitada ($\norm{\hat f}_\infty \leq \norm f_1$) e \omterm{def:b3:topology:continuity}{contínua}, e:
\begin{enumerate}
  \item $\widehat{\tau_af}(\xi) = \eu^{-\iu a\xi}\hat f(\xi)$ e $\widehat{\eu^{\iu ax}f}(\xi) = \hat f(\xi - a)$;
  \item $\widehat{f(\cdot/\lambda)}(\xi) = \lambda\hat
        f(\lambda\xi)$ para $\lambda > 0$;
  \item se $xf \in L^1$, então $\hat f$ é $\mathcal C^1$ com $(\hat f)'(\xi) = \widehat{(-\iu x)f}(\xi)$;
  \item se $f \in \mathcal C^1$ com $f' \in L^1$ (e $f \to
        0$ em $\pm\infty$, o que aqui é automático),
        então $\widehat{f'}(\xi) = \iu\xi\hat f(\xi)$;
  \item $\widehat{f * g} = \hat f\,\hat g$ para $f, g \in L^1$.
\end{enumerate}
\end{proposition}

\begin{proof}
Limitação: $\abs{\hat f} \leq \int\abs f$. \omterm{def:b3:topology:continuity}{Continuidade}: convergência dominada com dominador
$\abs f$ (\cref{thm:b3:lebesgue:paramcont}). (1), (2): substituições
(\cref{thm:b3:product:linearchange}). (3): derivação sob o sinal de integral, com dominador
$\abs{xf}$ (\cref{thm:b3:lebesgue:paramdiff}). (4): primeiro, $f(x) = f(0)
+ \int_0^xf'$ tem limite em $\pm\infty$ ($f' \in L^1$), que
tem de ser $0$ ($f \in L^1$); depois, integre por partes em $[-A,
A]$ e faça
$A \to \infty$. (5): Fubini, legítimo porque $(x,y)\mapsto f(x - y)g(y)\eu^{-\iu\xi x}$ é absolutamente \omterm{def:b3:lebesgue:l1}{integrável}
(\cref{thm:b3:product:convolution}):
\[
\widehat{f*g}(\xi) = \iint f(x - y)g(y)\eu^{-\iu\xi(x - y)}
\eu^{-\iu\xi y}\dd x\,\dd y = \hat f(\xi)\,\hat g(\xi).
\]
\end{proof}

\begin{example}\label{ex:b3:fouriertransform:gaussian}
A gaussiana: para $a > 0$,\index{gaussiana, transformada de Fourier}
\[
\widehat{\eu^{-ax^2}}(\xi) =
\sqrt{\frac\pi a}\;\eu^{-\xi^2/4a} :
\]
pelo~\cref{exo:b3:lebesgue:7} (o truque da EDO $F' = -\frac\xi{2}F$, reescalado), ou por (3):
$g = \widehat{\eu^{-ax^2}}$ satisfaz $g'(\xi) = -\frac{\xi}{2a}g(\xi)$ (integre por partes), com $g(0) = \sqrt{\pi/a}$. As gaussianas são
pontos fixos da transformada a menos de reescalamento --- a razão
profunda de governarem o teorema central do limite.
\end{example}

\begin{theorem}[Riemann--Lebesgue]
\label{thm:b3:fouriertransform:riemannlebesgue}
Para $f \in L^1(\R)$: $\hat f(\xi) \to 0$ quando $\abs\xi \to
\infty$. Assim, $\widehat{\phantom f} \colon L^1 \to \mathcal
C_0(\R)$ (funções \omterm{def:b3:topology:continuity}{contínuas} que se
anulam no infinito).\index{lema de Riemann--Lebesgue}
\end{theorem}

\begin{proof}
Para a indicadora de um intervalo, $\widehat{\mathbf 1_{\intcc
ab}}(\xi) = \frac{\eu^{-\iu a\xi} - \eu^{-\iu b\xi}}{\iu\xi}
\to 0$; logo, para funções escada. As
funções escada são densas em $L^1$ (\cref{thm:b3:lp:density}(1) mais a
aproximação de conjuntos de \omterm{def:b3:measure:measure}{medida} finita por reuniões finitas de
intervalos, \cref{exo:b3:measure:7}), e a transformada é \omterm{def:b3:topology:continuity}{contínua} de $\norm\cdot_\infty$ em
$\norm\cdot_1$: para $\norm{f -
s}_1 < \varepsilon$, $\limsup_{\abs\xi\to\infty}\abs{\hat
f(\xi)} \leq \varepsilon$.
\end{proof}

\section{Inversão e injetividade}

\begin{lemma}[Fórmula de multiplicação]
\label{lem:b3:fouriertransform:multiplication}
Para $f, g \in L^1(\R)$: $\displaystyle\int \hat f\,g =
\int f\,\hat g$.
\end{lemma}

\begin{proof}
Ambos os lados valem $\iint f(x)g(\xi)\eu^{-\iu x\xi}\dd
x\,\dd\xi$ (Tonelli--Fubini: a integral dupla do valor
absoluto é $\norm f_1\norm g_1$).
\end{proof}

\begin{theorem}[Inversão]\label{thm:b3:fouriertransform:inversion}
Seja $f \in L^1(\R)$.\index{teorema de inversão de Fourier}
\begin{enumerate}
  \item (Somabilidade gaussiana) Para todo $x$,
        \[
        (f * g_\varepsilon)(x) =
        \frac1{2\pi}\int_\R \hat f(\xi)\,
        \eu^{-\varepsilon\xi^2}\,\eu^{\iu x\xi}\,\dd\xi,
        \qquad\text{em que } g_\varepsilon(y) =
        \frac{1}{2\sqrt{\pi\varepsilon}}\,
        \eu^{-y^2/4\varepsilon},
        \]
        e $f * g_\varepsilon \to f$ em $L^1$ quando $\varepsilon
        \to 0$.
  \item Se, além disso, $\hat f \in L^1$, então, para quase todo $x$,
        \[
        f(x) = \frac{1}{2\pi}\int_\R \hat
        f(\xi)\,\eu^{\iu x\xi}\,\dd\xi ,
        \]
        e $f$ tem um representante \omterm{def:b3:topology:continuity}{contínuo}.
  \item (Injetividade) Se $\hat f = 0$, então $f = 0$ q.t.p.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Fixe $x$ e aplique o~\cref{lem:b3:fouriertransform:multiplication} a $f$ e a $g(\xi) = \frac1{2\pi}\eu^{-\varepsilon\xi^2}\eu^{\iu x\xi}$:
pelo~\cref{ex:b3:fouriertransform:gaussian} (com a regra de modulação),
\[
\hat g(y) = \frac1{2\pi}\sqrt{\frac\pi\varepsilon}\,
\eu^{-(y - x)^2/4\varepsilon} = g_\varepsilon(x - y),
\]
de modo que $\frac1{2\pi}\int\hat f(\xi)\eu^{-\varepsilon\xi^2}
\eu^{\iu x\xi}\dd\xi = \int f(y)g_\varepsilon(x - y)\dd y =
(f*g_\varepsilon)(x)$. As $g_\varepsilon$ formam uma \omterm{def:b3:lp:mollifier}{aproximação da identidade}: $g_\varepsilon \geq 0$,
$\int g_\varepsilon = 1$ (integral gaussiana), concentrando-se em $0$; a demonstração
do~\cref{thm:b3:lp:regularization}(2) se aplica literalmente (só foram usados $\int g_\varepsilon = 1$ e a
concentração: para a cauda, $\int_{\abs y > \delta}g_\varepsilon \to
0$): $\norm{f * g_\varepsilon - f}_1 \to 0$.

(2) Se $\hat f \in L^1$: o membro direito de (1) converge, por convergência
dominada (dominador $\abs{\hat f}$), a $\frac1{2\pi}\int\hat
f(\xi)\eu^{\iu x\xi}\dd\xi$ para \emph{todo} $x$, e essa função
limite é \omterm{def:b3:topology:continuity}{contínua} (convergência dominada, de novo). Por outro lado,
$f
* g_\varepsilon \to f$ em $L^1$, de modo que, ao longo de uma subsequência, q.t.p.\
(\cref{thm:b3:lp:complete}): os dois limites coincidem q.t.p.

(3) $\hat f = 0$ anula o membro direito de (1): $f *
g_\varepsilon = 0$ para todo $\varepsilon$, e $f *
g_\varepsilon \to f$
em $L^1$: $f = 0$ q.t.p.
\end{proof}

\section{A classe de Schwartz}

\begin{definition}\label{def:b3:fouriertransform:schwartz}
A \emph{classe de Schwartz}\index{classe de Schwartz} $\mathcal
S(\R)$ é formada
pelas funções $\mathcal C^\infty$ $f$ com $\sup_x\abs{x^m f^{(n)}(x)} < \infty$ para todos $m, n \geq 0$ (todas as
derivadas decaem mais rápido do que qualquer potência). Exemplos:
$\eu^{-ax^2}$, $\mathcal C_c^\infty$. Claramente $\mathcal S
\subseteq L^p$ para todo $p$ (limite por $C(1 + x^2)^{-1}$), e
$\mathcal S$ é estável por derivação, por multiplicação por polinômios e por
produtos.
\end{definition}

\begin{theorem}\label{thm:b3:fouriertransform:schwartz}
A transformada de Fourier leva $\mathcal S(\R)$ bijetivamente sobre si mesma, com
inversa $\check g(x) = \frac1{2\pi}\int
g(\xi)\eu^{\iu x\xi}\dd\xi$.
\end{theorem}

\begin{proof}
Seja $f \in \mathcal S$. Iterando o~\cref{prop:b3:fouriertransform:basic}(3), $(\hat f)^{(n)} =
\widehat{(-\iu x)^nf}$ (cada $x^kf \in L^1$); iterando
(4) com $h = (-\iu x)^nf \in \mathcal S$ (cujas derivadas são todas \omterm{def:b3:lebesgue:l1}{integráveis}), $(\iu\xi)^m\hat h =
\widehat{h^{(m)}}$. Combinando,
\[
\abs{\xi^m\,(\hat f)^{(n)}(\xi)}
= \bigl|\widehat{\,h^{(m)}}(\xi)\bigr|
\leq \bigl\|\bigl((-\iu x)^nf\bigr)^{(m)}\bigr\|_1 < \infty
\]
uniformemente em $\xi$: $\hat f \in \mathcal S$. Como $\hat f \in L^1$, a inversão
(\cref{thm:b3:fouriertransform:inversion}(2)) vale em toda parte (ambos os lados são \omterm{def:b3:topology:continuity}{contínuos}):
$\check{\hat f} = f$, e simetricamente $\widehat{\check g} = g$ (a transformada conjugada é $g \mapsto \frac1{2\pi}\hat g(-\cdot)$, que
também preserva $\mathcal S$): eis a bijeção.
\end{proof}

\section{Plancherel e \texorpdfstring{$L^2$}{L2}}

\begin{theorem}[Plancherel]\label{thm:b3:fouriertransform:plancherel}
Para $f \in L^1 \cap L^2(\R)$:\index{teorema de Plancherel}
\[
\norm{\hat f}_2^2 = 2\pi\,\norm f_2^2 .
\]
Consequentemente, $\widehat{\phantom f}$ se estende de maneira única a uma aplicação
linear \omterm{def:b3:topology:continuity}{contínua} $\mathcal F \colon L^2(\R) \to L^2(\R)$ com $\norm{\mathcal Ff}_2 = \sqrt{2\pi}\norm f_2$; $\mathcal
F$ é bijetora, com $\mathcal F^{-1} =
\frac1{2\pi}\,\mathcal F\circ\sigma$ em que
$\sigma f =
f(-\cdot)$, e preserva \omterm{def:b3:hilbert:inner}{produtos internos} a menos do fator $2\pi$.
\end{theorem}

\begin{proof}
Sejam $f \in L^1\cap L^2$ e $h = f * \tilde f$ com $\tilde f(x) = \overline{f(-x)}$. Então $h \in L^1$ (\cref{thm:b3:product:convolution}), $h$ é
\omterm{def:b3:topology:continuity}{contínua} e limitada (\cref{exo:b3:lp:6}: $f, \tilde f \in L^2$), $h(0) =
\int f\bar f = \norm f_2^2$, e $\hat h = \hat
f\,\widehat{\tilde f} = \hat f\,\overline{\hat f} =
\abs{\hat f}^2 \geq 0$ (calcule
$\widehat{\tilde f} =
\overline{\hat f}$). Aplique o~\cref{thm:b3:fouriertransform:inversion}(1) a $h$ em $x = 0$:
\[
(h * g_\varepsilon)(0) = \frac1{2\pi}\int \hat
h(\xi)\,\eu^{-\varepsilon\xi^2}\dd\xi .
\]
Quando $\varepsilon \to 0$: o membro esquerdo tende a $h(0)$ ($h$ é \omterm{def:b3:topology:continuity}{contínua} e
limitada: $(h*g_\varepsilon)(0) - h(0) = \int(h(-y)
- h(0))g_\varepsilon(y)\dd y \to 0$, separando $y$ pequenos e grandes); o membro direito
cresce a $\frac1{2\pi}\int\hat h$ por convergência monótona ($\hat h \geq 0$). Logo $\frac1{2\pi}\int\abs{\hat f}^2 =
\norm f_2^2$, finito ou
não \emph{a priori} --- e finito, o que demonstra tanto a pertinência
quanto a identidade.

Extensão: $L^1\cap L^2 \supseteq \mathcal C_c$ é denso em $L^2$ (\cref{thm:b3:lp:density}); a transformada é
ali $\sqrt{2\pi}$-isométrica, de modo que se estende de maneira única a uma
isometria a menos da constante $\mathcal F$ em $L^2$ (\cref{thm:b3:complete:extension}). A
inversão para $\mathcal S$ (\cref{thm:b3:fouriertransform:schwartz}) se transfere pela mesma \omterm{ex:b3:lebesgue:gamma}{densidade}
(ambos os lados são \omterm{def:b3:topology:continuity}{contínuos} em $L^2$): $\mathcal
F\bigl(\frac1{2\pi}\mathcal F(\sigma f)\bigr) = f$ em $\mathcal S$ e, portanto,
em $L^2$: eis a bijetividade. \omterm{def:b3:hilbert:inner}{Produtos internos}: polarização a
partir da identidade de normas.
\end{proof}

\begin{theorem}[Soma de Poisson]
\label{thm:b3:fouriertransform:poisson}
Seja $f \in \mathcal S(\R)$ (basta $f$ \omterm{def:b3:topology:continuity}{contínua} com $\abs{f} +
\abs{\hat f} \leq C(1 + \abs\cdot)^{-2}$). Então\index{fórmula de
soma de Poisson}
\[
\sum_{n\in\Z} f(n) \;=\; \sum_{k\in\Z}\hat f(2\pi k) .
\]
\end{theorem}

\begin{proof}
Ponha $F(x) = \sum_{n\in\Z}f(x + n)$: a série converge normalmente nos \omterm{def:b3:topology:compact}{compactos} (decaimento de
$f$), de modo que $F$ é \omterm{def:b3:topology:continuity}{contínua}, e ela é $1$-periódica. Seus
coeficientes de Fourier (período $1$: $c_k(F) = \int_0^1F(t)\eu^{-2\iu\pi kt}\dd t$):
\[
c_k(F) = \sum_n\int_0^1 f(t + n)\,\eu^{-2\iu\pi kt}\dd t
= \int_\R f(t)\,\eu^{-2\iu\pi kt}\dd t = \hat f(2\pi k)
\]
(a convergência normal justifica a troca; a fase é $1$-periódica). A
série $\sum_k\abs{c_k(F)}$ converge (decaimento de $\hat f$), de modo que a série de
Fourier de $F$ converge normalmente; sua soma é uma função \omterm{def:b3:topology:continuity}{contínua}
com os mesmos coeficientes de Fourier que $F$ e, portanto, é igual a
$F$ (injetividade no círculo: a diferença tem coeficientes nulos, e
o~\cref{thm:b3:hilbert:fourier} dá zero em $L^2$ e, por \omterm{def:b3:topology:continuity}{continuidade}, em toda
parte). Avalie em $x = 0$.
\end{proof}

\begin{example}[A identidade teta]
\label{ex:b3:fouriertransform:theta}
Aplicando Poisson a $f(x) = \eu^{-\pi tx^2}$ ($t > 0$), cuja transformada é $\hat f(\xi) = t^{-1/2}\eu^{-\xi^2/4\pi t}$
(\cref{ex:b3:fouriertransform:gaussian} com $a = \pi t$):
\[
\sum_{n\in\Z}\eu^{-\pi n^2t}
= \frac1{\sqrt t}\sum_{k\in\Z}\eu^{-\pi k^2/t} :
\]
a equação funcional da função teta de Jacobi, chave da equação
funcional da $\zeta$ de Riemann --- e um acelerador numérico
espetacular: para $t$ pequeno, o membro esquerdo converge devagar e
o direito converge com enorme rapidez.
\end{example}

\begin{method}\label{met:b3:fouriertransform:method}
Faixas de trabalho: $L^1$ --- transformada definida pontualmente, e
a inversão exige $\hat f \in L^1$; $\mathcal S$ --- tudo é legítimo, demonstre primeiro
aqui; $L^2$ --- transformada definida por \omterm{ex:b3:lebesgue:gamma}{densidade} (não pela
integral!), simetria \omterm{prop:b3:galois:perfect}{perfeita}, contabilidade de \omterm{thm:b3:hilbert:parseval}{Parseval}. Para calcular
uma transformada: reduza à tabela (indicadora, exponencial, gaussiana)
pelas regras do~\cref{prop:b3:fouriertransform:basic}; para demonstrar uma identidade:
estabeleça-a em $\mathcal S$ (ou em $\mathcal C_c^\infty$) e estenda por \omterm{ex:b3:lebesgue:gamma}{densidade} e
\omterm{def:b3:topology:continuity}{continuidade} (\cref{met:b3:lp:toolkit}); para resolver uma EDP ou EDO linear de
coeficientes constantes: transforme, divida, inverta.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:fouriertransform:1}
Calcule as transformadas de Fourier de: $\mathbf 1_{\intcc{-a}a}$; $\eu^{-a\abs x}$ ($a > 0$); a
função-tenda $\max(0, 1 -
\abs x)$; $\frac1{x^2 + a^2}$ \emph{(use a inversão na segunda)}. Registre a
tabela que vai surgindo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:fouriertransform:2}
Seja $f \in L^1$. Exprima em termos de $\hat f$ as transformadas de:
$f(x - a)$, $f(x)\cos(bx)$, $f(ax + b)$, $\overline{f(-x)}$, $(f * f)(x)$. Verifique cada regra na gaussiana.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:3}
(a) Mostre que $\mathbf 1_{\intcc{-1}1} * \mathbf
1_{\intcc{-1}1}$ tem transformada $\bigl(\frac{2\sin\xi}\xi
\bigr)^2$ e deduza $\int_\R\bigl(\frac{\sin\xi}\xi\bigr)^2\dd\xi = \pi$ por
Plancherel --- ou por inversão em $0$. Compare com
o~\cref{pb:b3:lebesgue:1}.
(b) Calcule $\int_\R\frac{\dd x}{(x^2+1)^2}$ via Plancherel aplicado a $\eu^{-\abs x}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:4}
(Álgebra do \omterm{pb:b3:fouriertransform:1}{núcleo do calor}) Com $g_t(x) =
\frac1{2\sqrt{\pi t}}\eu^{-x^2/4t}$:
(a) verifique $\hat g_t(\xi) = \eu^{-t\xi^2}$;
(b) deduza a lei de semigrupo $g_t * g_s = g_{t+s}$ sem nenhum cálculo de integral;
(c) mostre $\norm{g_t}_1 = 1$ e $\norm{g_t}_2^2 =
(8\pi t)^{-1/2}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:5}
Mostre que, se $f \in L^1$ é par e real, $\hat f$ é par e real; e que, se
$f$ é ímpar e real, $\hat f$ é ímpar e puramente imaginária. O
que $\hat f(0)$ calcula? Deduza que $f \geq
0$ força $\norm{\hat f}_\infty = \hat f(0) = \int f$ e interprete para
\omterm{ex:b3:lebesgue:gamma}{densidades} de probabilidade (\cref{ch:b3:clt}: uma função característica
tem módulo $\leq 1$, atingido em $0$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:fouriertransform:6}
(Não sobrejetividade) Mostre que $\widehat{\phantom f}\colon L^1
\to \mathcal C_0$ é injetora e \omterm{def:b3:topology:continuity}{contínua}, mas
\emph{não} é sobrejetora, em três etapas.
(i) Injetividade (\cref{thm:b3:fouriertransform:inversion}) e \omterm{def:b3:topology:continuity}{continuidade} ($\norm{\hat f}_\infty \leq \norm f_1$), e $\mathcal C_0$ é um
espaço de Banach (fechado em $\norm\cdot_\infty$).
(ii) Se a aplicação fosse sobrejetora, ela seria bijetora, e o teorema
da aplicação aberta (\cref{thm:b3:banach:openmapping}) daria uma constante $C$ com
$\norm f_1 \leq C\norm{\hat
f}_\infty$ para toda $f \in L^1$.
(iii) Contradiga isso com $f_n(x) = \frac{\sin
x}{x}\cdot\frac{\sin(x/n)}{x/n}$: sua transformada é (a menos de
constantes) o trapézio do tipo convolução $\mathbf 1_{\intcc{-1}1} *
\mathbf 1_{\intcc{-1/n}{1/n}}$ --- mostre que $\norm{\hat f_n}_\infty \leq \pi$
uniformemente, enquanto $\norm{f_n}_1 \geq c\ln n$, contando os arcos de $\frac{\abs{\sin x}}x$ em $[1, n]$
(onde o segundo fator é limitado inferiormente), como
no~\cref{thm:b3:banach:fourierdiverge}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:7}
(O dicionário regularidade $\leftrightarrow$ decaimento) Demonstre: $f \in
L^1$ com $\hat f(\xi) = O(\abs\xi^{-k-1-\delta})$
para algum $\delta > 0$ implica que $f$ tem um representante $\mathcal C^k$.
Reciprocamente, $f \in \mathcal C^k_c$ implica $\hat f(\xi) =
O(\abs\xi^{-k})$. Ilustre as duas direções na
função-tenda.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:fouriertransform:8}
(Desigualdade de Heisenberg) Para $f \in \mathcal S(\R)$ real com $\norm f_2 = 1$, demonstre
\[
\Bigl(\int x^2f(x)^2\dd x\Bigr)\cdot
\Bigl(\frac1{2\pi}\int \xi^2\abs{\hat f(\xi)}^2\dd\xi\Bigr)
\;\geq\; \frac14 ,
\]
com igualdade para as gaussianas. \emph{(Escreva $1 = \int f^2 =
-\int x\,(f^2)'$ por partes,
limite por Cauchy--Schwarz e converta $\norm{f'}_2$ por Plancherel.)}
Interpretação: um sinal e seu espectro não podem estar ambos
concentrados.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:9}
Justifique o~\cref{ex:b3:fouriertransform:theta} em detalhe (hipóteses de Poisson para a
gaussiana) e use a identidade para avaliar $\sum_{n\in\Z}\eu^{-\pi n^2}$ com seis casas
decimais usando três termos. Quantos termos da série \emph{original}
seriam necessários para a mesma precisão em $t = 10^{-2}$, comparado à série
transformada?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:10}
(Funções de banda limitada) Seja $f \in L^2(\R)$ com $\mathcal Ff$ suportada em $\intcc{-\pi}\pi$.
Mostre que $f$ tem um representante cujos valores são todos
recuperáveis a partir de amostras: demonstre a \emph{interpolação de
Shannon} nos inteiros,
\[
f(x) = \sum_{n\in\Z} f(n)\,
\frac{\sin\bigl(\pi(x - n)\bigr)}{\pi(x - n)}
\quad\text{em } L^2,
\]
expandindo $\mathcal Ff$ na base de Fourier de $L^2(\intcc{-\pi}\pi)$ (\cref{thm:b3:hilbert:fourier}) e
transformando de volta termo a termo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:11}
(A transformada como operador de ordem quatro) Em $\mathcal
S(\R)$, seja $\mathcal F f = \hat f$.
(a) Usando a fórmula de inversão, mostre $(\mathcal F^2f)(x) =
2\pi\,f(-x)$ e deduza $\mathcal F^4 = (2\pi)^2\,
\mathrm{id}$.
(b) Deduza que todo autovalor de $\mathcal F$ em $\mathcal S$ pertence a $\{\pm\sqrt{2\pi},
\pm\iu\sqrt{2\pi}\}$ e exiba
uma autofunção para $+\sqrt{2\pi}$ \emph{(qual função deste capítulo é
proporcional à sua própria transformada?)}.
(c) Mostre que as funções pares satisfazem $\mathcal F^2f = 2\pi
f$ e as ímpares $\mathcal F^2f = -2\pi f$;
produza uma autofunção para o autovalor $-\iu\sqrt{2\pi}$ a partir de $x\eu^{-x^2/2}$,
calculando sua transformada (derive a transformada da gaussiana).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:12}
(Autocorrelação e lema de Wiener) Para $f \in L^2(\R)$, defina $\tilde f(x) = \overline{f(-x)}$ e a
\emph{autocorrelação} $A_f = f * \tilde f$.
(a) Mostre que $A_f$ é uma função \omterm{def:b3:topology:continuity}{contínua} limitada com $A_f(0) = \norm f_2^2 \geq \abs{A_f(x)}$ para
todo $x$ (\cref{exo:b3:lp:6} e Cauchy--Schwarz).
(b) Mostre, primeiro para $f \in L^1\cap L^2$, que $\widehat{A_f} = \abs{\hat f\,}^2 \geq 0$: a autocorrelação tem
transformada não negativa --- os espectros de autocorrelações são
espectros de potência.
(c) Deduza a identidade $\int_\R\abs{\hat
f(\xi)}^2\eu^{\iu x\xi}\,\dd\xi = 2\pi A_f(x)$ (inversão; justifique sua aplicabilidade
quando $\hat f \in L^2$ tem $\abs{\hat f}^2 \in L^1$) e avalie-a para $f = \mathbf
1_{\intcc{-1/2}{1/2}}$ em $x = 0$: recupere
$\int_\R\bigl(\frac{\sin u}u\bigr)^2\dd u = \pi$.
\end{exercise}

\section{Problema: a equação do calor na reta}

\begin{problem}[{Problema de fim de semana --- $\partial_tu =
\partial^2_{xx}u$, resolvida de
ponta a ponta}]
\label{pb:b3:fouriertransform:1}
O calor se espalha; a equação $\partial_tu = \partial_{xx}^2u$ diz que sua \omterm{ex:b3:lebesgue:gamma}{densidade} se difunde a
uma taxa dada pela curvatura local do perfil de temperatura. Resolvemos
o problema de Cauchy em $\R$ --- dado $f$, encontrar $u(t, x)$ para
$t > 0$ com $u(0, \cdot) = f$ ---, demonstramos as propriedades notáveis da
solução e vemos por que o tempo não pode ser invertido. Ao longo de todo
o problema, $g_t(x) = \frac1{2\sqrt{\pi t}}\eu^{-x^2/4t}$ é o \emph{núcleo do calor}\index{núcleo do calor} e
$u(t, \cdot) = g_t *
f$.

\medskip
\noindent\textbf{Parte I --- Deduzindo o núcleo.} Trabalhe primeiro
formalmente: suponha que $u(t, \cdot) \in \mathcal S$ resolva a equação e seja $\hat u(t, \xi)$ a
transformada em $x$.
\begin{enumerate}
  \item Mostre (formalmente) que $\partial_t\hat u = -\xi^2\hat u$, logo $\hat u(t,\xi) = \eu^{-t\xi^2}\hat f(\xi)$, e reconheça $u(t) = g_t * f$
        (\cref{exo:b3:fouriertransform:4}). Isso motiva a \emph{definição} de $u$;
        tudo é agora demonstrado diretamente, para $f \in \mathcal C_b(\R)$ (\omterm{def:b3:topology:continuity}{contínua} e
        limitada) ou $f \in L^p$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Verificação.}
\begin{enumerate}[resume]
  \item Mostre que, para $t > 0$, $u(t, x) = \int
        g_t(x-y)f(y)\dd y$ está bem definida para $f \in \mathcal
        C_b$,
        e que $u$ é $\mathcal C^\infty$ em $(t, x)$ sobre $\intoo0\infty\times\R$ \emph{(derive
        sob o sinal de integral; domine as derivadas gaussianas
        localmente de maneira uniforme em $(t,x)$)}.
  \item Verifique $\partial_tg_t = \partial^2_{xx}g_t$ por cálculo direto e deduza $\partial_tu =
        \partial^2_{xx}u$ para
        $t > 0$.
  \item (Condição inicial) Mostre que, para $f \in \mathcal
        C_b$, $u(t, x) \to f(x)$ quando $t \to 0^+$,
        uniformemente nos \omterm{def:b3:topology:compact}{compactos} \emph{(\omterm{def:b3:lp:mollifier}{aproximação da identidade}:
        separe $\abs y
        \leq \delta$, $\abs y > \delta$)}; para $f \in L^p$ ($p < \infty$), mostre $\norm{u(t) - f}_p \to 0$.
  \item (Suavização instantânea) Conclua: mesmo para $f$ apenas
        \omterm{def:b3:topology:continuity}{contínua} e limitada, a solução é $\mathcal C^\infty$ para todo $t > 0$ ---
        o calor apaga instantaneamente a aspereza. Calcule $u(t, \cdot)$
        explicitamente para $f =
        \mathbf 1_{\intoo0\infty}$ (uma função erro) e esboce seu perfil
        para três valores de $t$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Propriedades estruturais.}
\begin{enumerate}[resume]
  \item (Positividade e comparação) Se $f \geq 0$, então $u > 0$ para
        todo $t > 0$ (estritamente, salvo se $f = 0$ q.t.p.); se
        $f_1 \leq f_2$, então $u_1 \leq u_2$. Um ponto frio se aquece instantaneamente:
        comente.
  \item (Conservação) Para $f \in L^1$: $\int u(t, x)\dd x =
        \int f$ para todo $t$ \emph{(Tonelli)}
        --- o calor total se conserva.
  \item (Dissipação) Para $f \in L^1\cap L^2$, mostre via Plancherel que $t \mapsto \norm{u(t)}_2$ é não
        crescente, estritamente salvo se $f = 0$, e calcule seu
        limite quando $t \to \infty$. Mostre, além disso, que $\norm{u(t)}_\infty \leq \frac{\norm
        f_1}{2\sqrt{\pi t}} \to 0$: o calor se
        espalha e morre.
  \item (Unicidade, classe $L^2$) Seja $u$ uma solução com $u(t) \in L^2$
        para todo $t$, $u \in \mathcal
        C^1(\intoo0\infty, L^2)$ no sentido natural e $u(t) \to f$ em $L^2$
        quando $t\to0$; admitindo que a transformada a converte em
        $\partial_t\hat u =
        -\xi^2\hat u$ pontualmente q.t.p.\ em $\xi$ para q.t.p.\ $t$
        \emph{(justificado testando contra $\mathcal
        C_c^\infty$ em $\xi$ --- esboce
        isso)}, mostre que $\hat u(t,\xi) = \eu^{-t\xi^2}\hat f(\xi)$ e, portanto, a unicidade nessa classe.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A flecha do tempo.}
\begin{enumerate}[resume]
  \item Mostre que o problema \emph{retrógrado} é mal posto: para que a
        solução exista no instante $-s$ ($s > 0$) com dados
        $f$ no instante $0$ --- isto é, para que $f = g_s * h$ tenha
        solução $h \in L^2$ ---, é necessário que $\eu^{s\xi^2}\hat f(\xi) \in L^2$: uma condição extrema
        de decaimento sobre $\hat f$. Exiba uma $f
        \in L^2$ suave
        explícita para a qual não existe solução retrógrada em
        intervalo de tempo algum: tome a função com $\hat f(\xi) =
        \eu^{-\abs\xi}$ ---
        identifique $f$ (\cref{exo:b3:fouriertransform:1}) e mostre que $\eu^{s\xi^2}\eu^{-\abs\xi} \notin L^2$ para todo
        $s >
        0$.
  \item (Suavização e informação) Explique em um parágrafo curto,
        usando as questões 5, 9 e 10, por que o semigrupo do calor
        $(f \mapsto g_t * f)_{t\geq0}$ é injetor mas não sobrejetor em $L^2$, e por que isso
        exprime a irreversibilidade da difusão.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- O teorema da amostragem de Shannon.} Uma
função $f \in L^2(\R)$ é \emph{de banda limitada} a $\Omega$ se $\hat f = 0$ q.t.p.\
fora de $\intcc{-\Omega}\Omega$; escrevemos $PW_\Omega$ (espaço de Paley--Wiener) para essas
funções.
\begin{enumerate}[resume]
  \item Mostre que toda $f \in PW_\Omega$ coincide q.t.p.\ com a função $\mathcal C^\infty$ dada
        por $\frac1{2\pi}\int_{-\Omega}^{\Omega}\hat
        f(\xi)\eu^{\iu x\xi}\,\dd\xi$ (justifique a regularidade e a identificação), com
        todas as derivadas limitadas: a limitação de banda é uma forma
        extrema de regularidade. Daqui em diante, $f$ denota esse
        representante.
  \item Expanda $\hat f \in L^2(\intcc{-\Omega}\Omega)$ na base de Fourier desse intervalo e identifique
        os coeficientes como \emph{amostras} de $f$:
        \[
        \hat f(\xi) = \frac\pi\Omega\sum_{n\in\Z}
        f\Bigl(\frac{n\pi}\Omega\Bigr)\,
        \eu^{-\iu n\pi\xi/\Omega}
        \quad\text{em } L^2(\intcc{-\Omega}\Omega) .
        \]
  \item Deduza o \emph{teorema da amostragem}: para $f \in
        PW_\Omega$,
        \[
        f(x) = \sum_{n\in\Z}f\Bigl(\frac{n\pi}
        \Omega\Bigr)\,\operatorname{sinc}(\Omega x - n\pi),
        \qquad \operatorname{sinc}t = \frac{\sin t}t,
        \]
        com convergência em $L^2(\R)$ e uniforme em $\R$ \emph{(injete a
        série da questão 13 na fórmula de inversão e calcule a integral
        elementar)}: um sinal de banda limitada fica inteiramente
        determinado por seus valores em uma grade de passo $\pi/\Omega$ --- a
        taxa de Nyquist.
  \item Mostre que as funções $x \mapsto
        \operatorname{sinc}(\Omega x - n\pi)$, $n \in \Z$, formam uma família ortogonal
        em $L^2(\R)$ de norma constante $\sqrt{\pi/\Omega}$, e deduza a identidade de
        energia $\norm f_2^2 =
        \frac\pi\Omega\sum_n\abs{f(n\pi/\Omega)}^2$.
  \item (Aliasing) Exiba uma $g \in PW_{2\Omega}$ não nula que se anula em todo ponto
        de amostragem $\frac{n\pi}\Omega$ \emph{(considere $g(x) =
        \sin(\Omega x)\operatorname{sinc}(\Omega x)$ e verifique sua
        banda)}: amostrar abaixo da taxa de Nyquist perde informação
        --- dois sinais diferentes podem partilhar todas as amostras: o
        efeito estroboscópico da roda de carroça, matematizado.
  \item (Graus de liberdade) Usando as questões 14--15, justifique a
        regra de engenharia: um sinal de banda limitada a $\Omega$
        cuja energia é essencialmente carregada por uma janela temporal
        de comprimento $T$ é descrito por aproximadamente $\frac{\Omega T}\pi$
        amostras reais --- torne ``essencialmente'' preciso por meio da
        identidade de energia e da cauda $\sum_{\abs{n\pi/\Omega} >
        T/2}$.
  \item (Verificações de coerência) Confira o teorema da amostragem à
        mão em dois membros de $PW_\Omega$: (a) $f =
        \operatorname{sinc}(\Omega\,\cdot)$, cujas amostras são
        $\delta_{n0}$; (b) sinais de banda estreita do tipo $f(x) = \cos(\omega x)
        \operatorname{sinc}(\varepsilon x)$ --- mais
        precisamente, mostre que, para $f \in
        PW_{\Omega'}$ com $\Omega' < \Omega$, a série na taxa
        $\Omega$ também reconstrói $f$ (sobreamostrar é
        inofensivo), mergulhando $PW_
        {\Omega'} \subseteq PW_\Omega$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Incerteza, mais duas vezes.} A
desigualdade de Heisenberg (\cref{exo:b3:fouriertransform:8}) limita \emph{quanto} $f$
e $\hat f$ podem estar conjuntamente concentradas; eis sua irmã do
tipo tudo-ou-nada e sua saturação exata.
\begin{enumerate}[resume]
  \item Seja $f \in L^1$ com $\operatorname{supp}f \subseteq
        \intcc{-A}A$. Mostre que $\hat f$ é a soma de uma
        série de potências convergente em toda parte:
        \[
        \hat f(\xi) = \sum_{k\geq0}\frac{(-\iu\xi)^k}{k!}
        \,m_k, \qquad m_k = \int_{-A}^{A}x^kf(x)\,\dd x,
        \quad \abs{m_k} \leq A^k\norm f_1
        \]
        \emph{(expanda $\eu^{-\iu\xi x}$ e justifique a troca pela convergência
        normal)}: a transformada é \emph{real-analítica}, com raio de
        convergência infinito em todo ponto.
  \item Deduza a dicotomia dos suportes: uma função real-analítica que
        se anula em um intervalo aberto não vazio se anula
        identicamente \emph{(o conjunto em que todas as derivadas se
        anulam é aberto e fechado --- desenvolva o argumento de
        Taylor)}; conclua que nenhuma $f$ não nula tem $f$ e
        $\hat f$ ambas de suporte \omterm{def:b3:topology:compact}{compacto}, e que $PW_\Omega$ não contém
        função não nula de suporte \omterm{def:b3:topology:compact}{compacto} --- os sinais de banda
        limitada duram para sempre, e os sinais de duração limitada
        vazam para todas as frequências.
  \item (Saturação de Heisenberg) Na família gaussiana $f
        = \eu^{-ax^2}$, calcule os
        dois fatores de concentração e verifique que o produto
        normalizado $\bigl(\int x^2\abs f^2\bigr)\bigl(\frac1{2\pi}\int
        \xi^2\abs{\hat f}^2\bigr)\big/\norm f_2^4$ vale $\frac14$ para \emph{todo} $a$ --- a
        família de igualdade do~\cref{exo:b3:fouriertransform:8} em carne e osso; explique
        por um argumento de reescalamento por que o produto tem de ser
        constante ao longo da família.
  \item Explique a leitura física (\omterm{ex:b3:lebesgue:gamma}{densidades} de posição e de momento
        de um estado quântico; o $\hbar$ da normalização dá $\sigma_x\sigma_p \geq
        \frac\hbar2$) e
        faça a conexão ao longo do capítulo: a suavização instantânea
        (Parte II), a irreversibilidade (Parte IV), a amostragem (Parte
        V), Heisenberg e a dicotomia dos suportes são cinco expressões
        de uma mesma lei --- o comportamento de $\hat f$ no infinito
        legisla o que $f$ pode fazer em qualquer lugar.
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- A álgebra do núcleo e um exemplo
resolúvel.}
\begin{enumerate}[resume]
  \item (Semigrupo) Demonstre a \emph{identidade de
        Chapman--Kolmogorov} $g_t * g_s = g_{t+s}$ para $t, s > 0$ (via o teorema da
        convolução e a injetividade da transformada em $L^1$) e
        deduza $u(t + s) = g_s *
        u(t)$: evoluir por um tempo $t + s$ é evoluir por
        $t$ e, depois, por $s$. Refine a dissipação da questão 8:
        escrevendo $\norm{u(t)}_2^2 = \frac1{2\pi}\int
        \eu^{-2t\xi^2}\abs{\hat f(\xi)}^2\dd\xi$, mostre por Cauchy--Schwarz que
        \[
        t \longmapsto \ln\,\norm{u(t)}_2
        \quad\text{é convexa em } \intoo0{+\infty} :
        \]
        a energia $L^2$ não apenas decresce: ela decresce de maneira
        log-convexa.
  \item (Para onde vai o calor) Sejam $f \geq 0$, $f \in L^1$, com $\int x^2f(x)\dd x < \infty$. Mostre que,
        para todo $t > 0$,
        \[
        \int_\R x\,u(t, x)\,\dd x = \int_\R x f(x)\,\dd x,
        \qquad
        \int_\R x^2u(t, x)\,\dd x
        = \int_\R x^2f(x)\,\dd x + 2t\int_\R f :
        \]
        o \omterm{ex:b3:groups:actions}{centro} do calor nunca se move, e a variância cresce
        \emph{linearmente} no tempo --- o reescalamento difusivo $x \sim \sqrt{2t}$, a
        ser relido quando o movimento browniano aparecer no~\cref{ch:b3:probability}. \emph{(Calcule os dois primeiros momentos de
        $g_t$ e use Tonelli na convolução.)}
  \item (A gaussiana, resolvida de ponta a ponta) Para $f(x) =
        \eu^{-x^2}$, estabeleça
        a forma fechada
        \[
        u(t, x) = \frac1{\sqrt{1 + 4t}}\,
        \exp\Bigl(-\frac{x^2}{1 + 4t}\Bigr),
        \]
        e verifique nela, à mão: a equação $\partial_tu
        = \partial^2_{xx}u$; a conservação $\int u(t) =
        \sqrt\pi$; a
        lei de dissipação $\norm{u(t)}_2 =
        (\pi/2)^{1/4}(1 + 4t)^{-1/4}$ (compare seu decaimento $t^{-1/4}$ com o
        decaimento $t^{-1/2}$ da norma do supremo, questão 8); e o
        crescimento exato da variância da questão 24. Em $t = 6$: o
        pico caiu a $\frac15$ de sua altura inicial, enquanto o perfil está
        cinco vezes mais largo --- o mesmo calor, espalhado.
\end{enumerate}
\end{problem}
