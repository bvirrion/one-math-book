\chapter{Espaços completos: Baire, Ascoli, Stone--Weierstrass}
\label{ch:b3:complete}

A \omterm{def:b3:complete:complete}{completude} --- toda sequência de Cauchy converge --- é a propriedade
que permite à análise \emph{produzir} objetos: pontos fixos de
contrações, somas de séries, soluções de equações obtidas como limites.
Este capítulo reúne as três grandes máquinas de existência da teoria
métrica. O \emph{teorema de Baire} mostra que um espaço \omterm{def:b3:complete:complete}{completo} não
pode ser reunião enumerável de pedaços desprezíveis e conjura objetos
(funções \omterm{def:b3:topology:continuity}{contínuas} em parte alguma deriváveis!) a partir de um
raciocínio de puro tipo cardinal. \emph{Arzelà--Ascoli} identifica os
subconjuntos \omterm{def:b3:topology:compact}{compactos} de $\mathcal C(K)$ e é o cavalo de batalha da análise em
matéria de \omterm{def:b3:topology:compact}{compacidade} --- o problema de fim de semana o usa para
demonstrar o teorema de existência de Peano para equações diferenciais.
\emph{Stone--Weierstrass} mostra que os polinômios, e muito mais, são
densos em $\mathcal C(K)$: a aproximação torna-se uma verificação \omterm{def:b3:galois:algebraic}{algébrica}. Pelo
\omterm{def:b3:topology:pathconnected}{caminho}, construímos \omterm{thm:b3:complete:completion}{completamentos} e demonstramos o teorema de
extensão das aplicações uniformemente \omterm{def:b3:topology:continuity}{contínuas}, o pão de cada dia do~\cref{ch:b3:lp,ch:b3:hilbert,ch:b3:fouriertransform}.

\section{Espaços completos, completamentos, extensões}

\begin{definition}\label{def:b3:complete:complete}
Um espaço métrico é \emph{completo}\index{espaço métrico completo} se
toda sequência de Cauchy converge (segundo ano: $\R^n$ é completo;
$\mathcal C(\intcc01)$ com $\norm\cdot_\infty$ é completo). Um subconjunto fechado de um espaço completo é
completo; um subconjunto completo de um espaço métrico qualquer é
fechado.
\end{definition}

\begin{proof}
Quanto às duas afirmações: uma sequência de Cauchy do fechado $F$
converge em $X$, e seu limite, aderente a $F$, está em $F$; uma
sequência de um $A$ \omterm{def:b3:complete:complete}{completo} que converge em $X$ é de Cauchy,
logo converge em $A$, e os limites são únicos.
\end{proof}

\begin{theorem}[Extensão das aplicações uniformemente contínuas]
\label{thm:b3:complete:extension}
Sejam $D \subseteq X$ denso, $Y$ \omterm{def:b3:complete:complete}{completo} e $f \colon D \to
Y$ uniformemente \omterm{def:b3:topology:continuity}{contínua}. Então
$f$ se estende de maneira \emph{única} a uma \omterm{def:b3:topology:continuity}{aplicação contínua}
$\bar f \colon X \to Y$, e $\bar f$ é uniformemente \omterm{def:b3:topology:continuity}{contínua}.\index{teorema de extensão
(continuidade uniforme)}
\end{theorem}

\begin{proof}
Unicidade: duas extensões \omterm{def:b3:topology:continuity}{contínuas} coincidem no denso $D$ e,
portanto, em toda parte (o conjunto de coincidência $\{g = h\}$ é fechado:
pré-imagem da diagonal fechada por $x\mapsto(g(x), h(x))$). Existência: para $x \in X$ tome
$d_n \to x$, $d_n \in D$. A sequência $(f(d_n))$ é de Cauchy: dado $\varepsilon$, a \omterm{def:b3:topology:continuity}{continuidade}
uniforme fornece $\delta$ com $d(u,v) < \delta \Rightarrow
d(f(u), f(v)) < \varepsilon$, e $(d_n)$ é de Cauchy. Defina
$\bar f(x) = \lim f(d_n)$; o limite não depende da sequência escolhida (entrelace duas
delas). $\bar f$ estende $f$ (sequências constantes) e herda o
módulo de \omterm{def:b3:topology:continuity}{continuidade}: se $d(x, x') < \delta$, aproximar ambos por pontos de $D$ a
distância $< \frac{\delta - d(x,x')}2$ dá $d(\bar f(x), \bar
f(x')) \leq \varepsilon$ no limite --- $\bar f$ é uniformemente
\omterm{def:b3:topology:continuity}{contínua}.
\end{proof}

\begin{theorem}[Completamento]\label{thm:b3:complete:completion}
Todo espaço métrico $X$ mergulha isometricamente como subconjunto
denso de um \omterm{def:b3:complete:complete}{espaço métrico completo} $\hat X$, único a menos de uma
isometria que fixa $X$ ponto a ponto: seu
\emph{completamento}\index{completamento de um espaço métrico}.
\end{theorem}

\begin{proof}
\emph{Existência.} Seja $\mathcal X$ o conjunto das sequências de Cauchy de
$X$, com a pseudodistância
\[
D\bigl((x_n), (y_n)\bigr) = \lim_n d(x_n, y_n),
\]
cujo limite existe porque $\abs{d(x_n, y_n) - d(x_m, y_m)}
\leq d(x_n, x_m) + d(y_n, y_m)$ torna a sequência real de Cauchy. Ponha
$\hat X = \mathcal X/{\sim}$, identificando as sequências a $D$-distância $0$; $D$
desce a uma distância. Mergulhe $X$ pelas sequências constantes: uma
isometria de imagem densa (uma sequência de Cauchy é
$D$-aproximada pelas constantes construídas sobre seus próprios
termos: $D\bigl((x_n), (x_k)_{\rm
const}\bigr) = \lim_n d(x_n, x_k) \to 0$ quando $k \to \infty$, por ser de Cauchy). \omterm{def:b3:complete:complete}{Completude} de $\hat X$:
seja $(\xi^k)$ de Cauchy em $\hat X$; por densidade, escolha $x_k \in X$ com
$D(\xi^k, x_k)
\leq 2^{-k}$; então $(x_k)$ é de Cauchy em $X$ (desigualdade triangular
através dos $\xi$), define um ponto $\xi \in \hat
X$, e $D(\xi^k, \xi) \leq 2^{-k} + D(x_k, \xi) \to 0$ (a distância da
constante $x_k$ à classe de $(x_j)_j$ é $\lim_j d(x_k, x_j)$, pequena para $k$ grande).

\emph{Unicidade}: dois \omterm{thm:b3:complete:completion}{completamentos} $\hat X_1, \hat X_2$ contêm $X$ densamente; a
identidade de $X$, uma isometria, é uniformemente \omterm{def:b3:topology:continuity}{contínua}, logo se
estende a $\hat X_1 \to \hat X_2$ (\cref{thm:b3:complete:extension}), ainda uma isometria em um \omterm{def:b3:topology:interior}{conjunto
denso} e, portanto, em toda parte; simetricamente no outro sentido, e as
compostas fixam o denso $X$: elas são as identidades.
\end{proof}

\begin{theorem}[Ponto fixo de Banach]\label{thm:b3:complete:banach}
Sejam $X$ \omterm{def:b3:complete:complete}{completo} e não vazio e $f \colon X \to X$ uma \emph{contração}: $d(f(x), f(y)) \leq k\,d(x,y)$ com
$k < 1$. Então $f$ tem um único ponto fixo $x^*$, e toda \omterm{def:b3:groups:action}{órbita}
converge para ele, com a taxa explícita $d(x_n, x^*) \leq
\frac{k^n}{1-k}\,d(x_1, x_0)$.\index{teorema do ponto
fixo de Banach}
\end{theorem}

\begin{proof}
(O segundo ano demonstrou isto; registramos de novo o argumento de duas
linhas para que o texto se baste.) A \omterm{def:b3:groups:action}{órbita} $x_{n+1} = f(x_n)$ satisfaz $d(x_{n+1},
x_n) \leq k^nd(x_1, x_0)$, logo é
de Cauchy (série geométrica); seu limite $x^*$ é fixo (\omterm{def:b3:topology:continuity}{continuidade}
de $f$) e único, pois dois pontos fixos satisfazem $d \leq k\,d$. A taxa: some
a cauda geométrica.
\end{proof}

\begin{example}[Perturbando a identidade]
\label{ex:b3:complete:perturbident}
Seja $g \colon \R^d \to \R^d$ $k$-lipschitziana com $k < 1$. Então $\varphi = \mathrm{id} + g$ é um
\emph{\omterm{def:b3:topology:continuity}{homeomorfismo} de $\R^d$ sobre $\R^d$}. Injetividade, com um
módulo quantitativo:
\[
\norm{\varphi(x) - \varphi(y)} \geq \norm{x - y} -
\norm{g(x) - g(y)} \geq (1 - k)\norm{x - y} .
\]
A sobrejetividade é o teorema do ponto fixo: resolver $\varphi(x) = y$ significa
$x = y - g(x)$, e $x \mapsto y -
g(x)$ é uma $k$-contração do espaço \omterm{def:b3:complete:complete}{completo} $\R^d$ ---
existe uma única solução $x = \psi(y)$ para cada $y$. A desigualdade exibida
torna a inversa $\psi$ lipschitziana de constante $\frac1{1-k}$: um
\omterm{def:b3:topology:continuity}{homeomorfismo}, com cotas explícitas para ambos os módulos. Esse
enunciado de aparência inofensiva é o motor dentro do teorema da função
inversa (\cref{ch:b3:submanifolds}): perto de um ponto em que $Df$ é
inversível, $f$ \emph{é} uma aplicação linear inversível mais uma
pequena perturbação lipschitziana, e o exemplo de hoje faz o resto. Ele
também quantifica a robustez numérica: um sistema perturbado por menos
do que a margem da inversa permanece \omterm{def:b3:groups:derived}{solúvel}, com a solução deslocando-se
no máximo $\frac{1}{1-k}$ vezes a perturbação.
\end{example}

\section{O teorema de Baire}

\begin{theorem}[Baire]\label{thm:b3:complete:baire}
Em um \omterm{def:b3:complete:complete}{espaço métrico completo}, uma interseção enumerável de abertos
densos é densa. Equivalentemente: se $X = \bigcup_{n}F_n$ com cada $F_n$ fechado,
então algum $F_n$ tem \omterm{def:b3:topology:interior}{interior} não vazio.\index{teorema de Baire}
\end{theorem}

\begin{proof}
Sejam $(U_n)$ abertos densos e $B_0 = B(x_0, r_0)$ uma bola aberta qualquer;
encontramos um ponto de $\bigcap U_n$ em $B_0$. Indutivamente: sendo
$U_{n}$ denso e aberto, ele encontra a bola aberta $B_{n-1}$ em um
aberto, que contém uma bola fechada $\bar B(x_n, r_n)$ com $0 < r_n \leq r_{n-1}/2$ e $\bar B(x_n, r_n) \subseteq
B_{n-1}\cap U_n$. Os \omterm{ex:b3:groups:actions}{centros}
formam uma sequência de Cauchy ($x_m \in
B_n$ para $m \geq n$, com raios $\to 0$);
o limite $x$ está em toda $\bar B(x_n, r_n)$ (por serem fechadas), logo em todo
$U_n$ e em $B_0$. Quanto à segunda forma: se nenhum $F_n$ tem
\omterm{def:b3:topology:interior}{interior}, os $U_n = X \setminus F_n$ são abertos e densos, e um ponto de $\bigcap U_n$ escapa a
$\bigcup F_n = X$: absurdo.
\end{proof}

\begin{remark}\label{rem:b3:complete:meagre}
Vocabulário: um conjunto é \emph{nunca denso} se seu \omterm{def:b3:topology:interior}{fecho} tem \omterm{def:b3:topology:interior}{interior}
vazio, e \emph{magro}\index{conjunto magro} (de primeira categoria) se
é uma reunião enumerável de conjuntos nunca densos. Baire: \emph{um
\omterm{def:b3:complete:complete}{espaço métrico completo} não é \omterm{rem:b3:complete:meagre}{magro} em si mesmo}, e o complementar de
um \omterm{rem:b3:complete:meagre}{conjunto magro} é denso. ``\omterm{rem:b3:complete:meagre}{Magro}'' é uma noção de pequenez ortogonal à
de medida (o~\cref{ch:b3:measure} produzirá \omterm{rem:b3:complete:meagre}{conjuntos magros} de medida total),
e os argumentos de Baire demonstram \emph{existência por abundância}:
para exibir um objeto sem a propriedade P, mostre que os objetos com P
formam um \omterm{rem:b3:complete:meagre}{conjunto magro}.
\end{remark}

\begin{corollary}\label{cor:b3:complete:bairecors}
(a) $\R$ é não enumerável. (b) $\Q$ não é uma interseção
enumerável de abertos de $\R$, e um \omterm{def:b3:complete:complete}{espaço métrico completo} não
vazio sem pontos isolados é não enumerável.
\end{corollary}

\begin{proof}
(a) $\R = \bigcup_{x}\{x\}$ sobre um conjunto enumerável faria algum conjunto unitário ter
\omterm{def:b3:topology:interior}{interior}. (b) Se $\Q = \bigcap_n V_n$ com $V_n$ abertos (necessariamente densos,
pois $\supseteq \Q$), então os conjuntos $V_n$ e os complementares $\R\setminus\{q\}$,
$q \in \Q$, formam uma família enumerável de abertos densos com interseção
vazia --- contradizendo Baire. Se $X$ é \omterm{def:b3:complete:complete}{completo}, sem pontos
isolados e enumerável, $X = \bigcup_{x \in X}\{x\}$ o exibe como reunião enumerável de fechados
de \omterm{def:b3:topology:interior}{interior} vazio (ausência de pontos isolados): Baire de novo.
\end{proof}

\begin{theorem}[Os monstros de Weierstrass existem]
\label{thm:b3:complete:nowherediff}
Existem funções \omterm{def:b3:topology:continuity}{contínuas} em $\intcc01$ que não são deriváveis em ponto
algum. Mais ainda: o conjunto das $f \in \mathcal C(\intcc01)$ que têm derivada (finita) em ao
menos um ponto é \omterm{rem:b3:complete:meagre}{magro} em $(\mathcal C(\intcc01), \norm\cdot_\infty)$.\index{função em parte alguma derivável}
\end{theorem}

\begin{proof}
Para $n \geq 1$, ponha
\[
F_n = \Bigl\{f : \exists x \in \intcc01,\
\forall h \neq 0 \text{ com } x + h \in \intcc01,\
\abs{f(x+h) - f(x)} \leq n\abs h \Bigr\}.
\]
Se $f$ é derivável em $x$, então $f \in F_n$ para algum $n$: o
quociente $\abs{f(x+h)-f(x)}/\abs h$ é limitado para $\abs h
\leq \delta$ (derivabilidade: ele tende a
$\abs{f'(x)}$) e limitado por $2\norm f_\infty/\delta$ para $\abs h \geq \delta$. Assim $\bigcup F_n$ contém todas as
funções deriváveis em algum ponto, e basta mostrar que cada $F_n$ é
fechado e tem \omterm{def:b3:topology:interior}{interior} vazio.

\emph{Fechado}: seja $f_k \to f$ uniformemente, com $f_k \in F_n$ e testemunhas $x_k \to x$
(\omterm{def:b3:topology:compact}{compacidade}, após extração). Para $h$ com $x + h \in \intcc01$: escolha $h_k \to h$ com
$x_k + h_k
\in \intcc01$ (por exemplo $h_k = h + x - x_k$ truncado); então $\abs{f(x + h) - f(x)} = \lim \abs{f_k(x_k + h_k) - f_k(x_k)}
\leq \lim n\abs{h_k} = n\abs h$, usando a convergência
uniforme e a \omterm{def:b3:topology:continuity}{continuidade} de $f$ nos pontos relevantes: $f \in F_n$.

\emph{\omterm{def:b3:topology:interior}{Interior} vazio}: dados $f \in F_n$ e $\varepsilon > 0$, encontramos $g$ com
$\norm{g - f}_\infty \leq \varepsilon$ e $g
\notin F_n$. Primeiro, aproxime $f$ a menos de $\varepsilon/2$ por uma função
$\varphi$ afim por partes (\omterm{def:b3:topology:continuity}{continuidade} uniforme: interpole em uma grade
fina), de inclinações limitadas por algum $M$. Some um pequeno dente
de serra: $g = \varphi + \frac\varepsilon2\,s_N$, em que $s_N(x)$ é o zigue-zague $\frac1N$-periódico de
amplitude $1$ e inclinação $\pm 2N$. Em todo $x$, de um dos
lados existe $h$ arbitrariamente pequeno com o dente de serra
contribuindo com inclinação $\pm 2N\cdot\frac\varepsilon2 =
\pm\varepsilon N$ em $[x, x+h]$: o quociente de diferenças de
$g$ ultrapassa $\varepsilon N - M > n$ para $N$ grande. Logo $g \notin
F_n$, a qualquer
distância uniforme $\leq \varepsilon$ de $f$. Conclusão: $\bigcup F_n$ é \omterm{rem:b3:complete:meagre}{magro}; por
Baire, seu complementar --- formado por \omterm{thm:b3:complete:nowherediff}{funções em parte alguma
deriváveis} --- é denso em $\mathcal C(\intcc01)$: tais funções existem \emph{em
abundância}.
\end{proof}

\section{Arzelà--Ascoli}

Ao longo de toda esta seção, $K$ é um espaço métrico
\emph{\omterm{def:b3:topology:compact}{compacto}} e $\mathcal
C(K) = \mathcal C(K, \R^d)$, com $\norm f_\infty = \sup_K
\norm{f(x)}$: um espaço \omterm{def:b3:complete:complete}{completo} (limites
uniformes de \omterm{def:b3:topology:continuity}{contínuas} são \omterm{def:b3:topology:continuity}{contínuas} --- segundo ano).

\begin{definition}\label{def:b3:complete:equicontinuous}
Uma família $\mathcal F \subseteq \mathcal C(K)$ é \emph{equicontínua}\index{equicontinuidade} se,
para todo $\varepsilon > 0$, existe $\delta > 0$ tal que
\[
d(x, y) < \delta \implies \norm{f(x) - f(y)} < \varepsilon
\quad\text{para \emph{toda} } f \in \mathcal F
\]
(um único $\delta$ para toda a família --- por exemplo, qualquer
família com uma constante de Lipschitz comum, ou um módulo de Hölder
comum), e é \emph{pontualmente limitada} se $\sup_{f}\norm{f(x)} < \infty$ para cada $x$.
\end{definition}

\begin{theorem}[Arzelà--Ascoli]\label{thm:b3:complete:ascoli}
Um subconjunto $\mathcal F \subseteq \mathcal C(K)$ é relativamente \omterm{def:b3:topology:compact}{compacto} (tem \omterm{def:b3:topology:interior}{fecho} \omterm{def:b3:topology:compact}{compacto}) se,
e somente se, é \omterm{def:b3:complete:equicontinuous}{equicontínuo} e pontualmente limitado. Em particular,
toda \emph{sequência} \omterm{def:b3:complete:equicontinuous}{equicontínua} e pontualmente limitada tem uma
subsequência uniformemente convergente.\index{teorema de
Arzelà--Ascoli}
\end{theorem}

\begin{proof}
($\Leftarrow$) Seja $(f_k)$ uma sequência de $\mathcal F$. O \omterm{def:b3:topology:compact}{compacto} métrico
$K$ é \emph{separável}: para cada $n$, um número finito de bolas
de raio $\frac1n$ cobre $K$ (\cref{thm:b3:topology:metriccompact}); seus \omterm{ex:b3:groups:actions}{centros} formam um
conjunto enumerável denso $D = \{x_1, x_2, \dots\}$. Pela limitação pontual e por
Bolzano--Weierstrass, extraia sucessivamente subsequências que
convergem em $x_1$, depois também em $x_2$, etc., e tome a
subsequência \emph{diagonal} $(g_j)$: ela converge em todo ponto de
$D$. A \omterm{def:b3:complete:equicontinuous}{equicontinuidade} eleva isso a uniformemente de Cauchy: dado
$\varepsilon$, tome $\delta$ como na definição, cubra $K$ por um número
finito de bolas $B(x_i, \delta)$ com $x_i \in D$ ($i \leq m$) e escolha $J$ tão grande que
$\norm{g_j(x_i) - g_{j'}(x_i)} < \varepsilon$ para $j, j' \geq
J$, $i \leq m$. Para $x \in B(x_i, \delta)$ arbitrário:
\[
\norm{g_j(x) - g_{j'}(x)}
\leq \norm{g_j(x) - g_j(x_i)} + \norm{g_j(x_i) - g_{j'}(x_i)}
+ \norm{g_{j'}(x_i) - g_{j'}(x)} < 3\varepsilon .
\]
Logo $(g_j)$ é uniformemente de Cauchy e converge no espaço \omterm{def:b3:complete:complete}{completo}
$\mathcal C(K)$. Assim, toda sequência de $\mathcal F$ tem subsequência convergente:
$\bar{\mathcal F}$ é \omterm{def:b3:topology:compact}{compacto} (sequencialmente e, portanto, pelo~\cref{thm:b3:topology:metriccompact}).

($\Rightarrow$) Se $\bar{\mathcal F}$ é \omterm{def:b3:topology:compact}{compacto}: a limitação pontual é clara (a avaliação é
\omterm{def:b3:topology:continuity}{contínua}). Quanto à \omterm{def:b3:complete:equicontinuous}{equicontinuidade}, cubra $\mathcal F$ por um número finito
de bolas $B(f_i, \varepsilon)$ de raio $\mathcal C(K)$; cada $f_i$ é uniformemente \omterm{def:b3:topology:continuity}{contínua}
(Heine, \cref{cor:b3:topology:heineborel}), o que dá um $\delta$ comum para $i \leq m$;
então, para $f \in B(f_i, \varepsilon)$ e $d(x,y)
< \delta$: $\norm{f(x)-f(y)} \leq 2\varepsilon +
\norm{f_i(x)-f_i(y)} < 3\varepsilon$.
\end{proof}

\begin{example}\label{ex:b3:complete:ascoliexamples}
A bola unitária fechada de $\mathcal C(\intcc01)$ \emph{não} é compacta ($f_n(x) = x^n$ não tem
subsequência uniformemente convergente: o limite pontual é
descontínuo) e, com efeito, $(x^n)$ não é \omterm{def:b3:complete:equicontinuous}{equicontínua} em $1$.
Em contraste, $\{f : \norm
f_\infty \leq 1,\ \operatorname{Lip}(f) \leq 1\}$ é \omterm{def:b3:topology:compact}{compacto}: limitado, \omterm{def:b3:complete:equicontinuous}{equicontínuo} por serem todas
$1$-lipschitzianas, e fechado. Ascoli explica \emph{por que} a
\omterm{def:b3:topology:compact}{compacidade} falha em dimensão infinita (Riesz, segundo ano) e o que
acrescentar para restaurá-la: um módulo de \omterm{def:b3:topology:continuity}{continuidade} uniforme.
\end{example}

\section{Stone--Weierstrass}

\begin{lemma}[Dini]\label{lem:b3:complete:dini}
Sejam $K$ \omterm{def:b3:topology:compact}{compacto} e $(f_n)$ uma sequência \emph{monótona} de
funções reais \omterm{def:b3:topology:continuity}{contínuas} que converge \emph{pontualmente} para uma
função \omterm{def:b3:topology:continuity}{contínua} $f$. Então a convergência é uniforme.\index{teorema
de Dini}
\end{lemma}

\begin{proof}
Digamos $f_n \uparrow f$; ponha $g_n = f - f_n \downarrow 0$, \omterm{def:b3:topology:continuity}{contínua}. Dado $\varepsilon$, os abertos $U_n = \{g_n <
\varepsilon\}$
crescem e cobrem $K$ (convergência pontual); extraia uma
subcobertura finita: $K = U_{n_0}$ para algum $n_0$ (família crescente),
isto é, $0 \leq g_n < \varepsilon$ em toda parte para $n \geq n_0$.
\end{proof}

\begin{lemma}\label{lem:b3:complete:sqrt}
Existe uma sequência de \emph{polinômios} $u_n$ com $u_n(t) \to
\sqrt t$
uniformemente em $\intcc01$.
\end{lemma}

\begin{proof}
Defina $u_0 = 0$, $u_{n+1}(t) = u_n(t) + \frac12\bigl(t -
u_n(t)^2\bigr)$: são polinômios. Por indução, $0 \leq u_n(t) \leq
\sqrt t$ em $\intcc01$:
admitindo isso para $n$,
\[
\sqrt t - u_{n+1}(t) = \bigl(\sqrt t - u_n(t)\bigr)
\Bigl(1 - \tfrac12\bigl(\sqrt t + u_n(t)\bigr)\Bigr) \geq 0,
\]
pois $\sqrt t + u_n \leq 2$; e $u_{n+1} \geq u_n \geq 0$. Assim $(u_n(t))$ é não decrescente e limitada por $\sqrt t$:
ela converge pontualmente, e o limite $\ell(t)$ satisfaz $\ell =
\ell + \frac12(t - \ell^2)$: $\ell(t) = \sqrt t$, que é
\omterm{def:b3:topology:continuity}{contínua}. Dini (\cref{lem:b3:complete:dini}) eleva a convergência a uniforme.
\end{proof}

\begin{theorem}[Stone--Weierstrass, versão real]
\label{thm:b3:complete:stoneweierstrass}
Sejam $K$ um \omterm{def:b3:topology:compact}{espaço compacto} (\omterm{def:b3:topology:hausdorff}{de Hausdorff}) e $\mathcal A \subseteq
\mathcal C(K, \R)$ uma
\emph{subálgebra} (estável por somas, produtos e múltiplos escalares)
que \emph{contém as constantes} e \emph{separa pontos} (para $x \neq y$,
algum $f \in
\mathcal A$ satisfaz $f(x) \neq f(y)$). Então $\mathcal A$ é densa em
$(\mathcal C(K,\R), \norm\cdot_\infty)$.\index{teorema de Stone--Weierstrass}
\end{theorem}

\begin{proof}
Seja $\bar{\mathcal A}$ o \omterm{def:b3:topology:interior}{fecho}, ainda uma álgebra (produtos de limites uniformes em
conjuntos limitados convergem).

\emph{Etapa 1: $\bar{\mathcal A}$ é um reticulado}, isto é, estável por $\max$ e
$\min$. Como $\max(f,g) = \frac{f + g +
\abs{f-g}}2$, e analogamente para $\min$, basta que $f \in
\bar{\mathcal A} \Rightarrow \abs f \in \bar{\mathcal A}$:
com $M = \norm f_\infty > 0$, $\abs f = M\sqrt{(f/M)^2}$, e o~\cref{lem:b3:complete:sqrt} dá polinômios $u_n$ com $u_n\bigl((f/M)^2\bigr) \to \abs f/M$
uniformemente; polinômios em membros da álgebra (com termo constante:
as constantes lá estão) permanecem em $\bar{\mathcal A}$.

\emph{Etapa 2: interpolação em dois pontos.} Para $x \neq y$ e $a,
b \in \R$, algum
$g \in \mathcal A$ satisfaz $g(x) = a$, $g(y) = b$: tome $h$ que separe $x, y$ e ponha
$g = a + (b -
a)\frac{h - h(x)}{h(y) - h(x)}$.

\emph{Etapa 3.} Sejam $f \in \mathcal C(K)$, $\varepsilon > 0$. Para cada par $x, y$ escolha
$g_{x,y} \in \mathcal A$ com $g_{x,y}(x) = f(x)$, $g_{x,y}(y) = f(y)$ (Etapa 2; para $x = y$ tome a função constante
$g_{x,x} = f(x)$). Fixe $x$: para cada $y$, o aberto $V_y = \{g_{x,y} < f +
\varepsilon\}$ contém $y$; a
\omterm{def:b3:topology:compact}{compacidade} extrai $y_1, \dots,
y_m$ com $K = \bigcup V_{y_j}$, e $h_x = \min_j g_{x, y_j}
\in \bar{\mathcal A}$ (Etapa 1) satisfaz $h_x < f +
\varepsilon$ em
toda parte, além de $h_x(x) = f(x)$. Agora varie $x$: $W_x =
\{h_x > f - \varepsilon\}$ é aberto e contém
$x$; extraia $x_1,
\dots, x_l$ cobrindo $K$, e $h = \max_i h_{x_i} \in
\bar{\mathcal A}$ satisfaz $f - \varepsilon < h < f +
\varepsilon$: $\norm{f - h}_\infty \leq \varepsilon$.
Logo $f \in \bar{\mathcal A}$.
\end{proof}

\begin{corollary}\label{cor:b3:complete:weierstrass}
(a) (\emph{Weierstrass}) Os polinômios são densos em $\mathcal
C(\intcc ab, \R)$; os
polinômios em $n$ variáveis são densos em $\mathcal C(K, \R)$ para $K \subseteq \R^n$ \omterm{def:b3:topology:compact}{compacto}.
(b) (\emph{Versão complexa}) Se $\mathcal A \subseteq \mathcal
C(K, \C)$ é uma subálgebra que contém as
constantes, separa pontos e é \emph{estável por conjugação}, então ela
é densa.
(c) (\emph{Versão trigonométrica}) Os polinômios trigonométricos $\sum_{\abs n \leq N}c_n\eu^{\iu n t}$
são densos no espaço das funções \omterm{def:b3:topology:continuity}{contínuas} $2\pi$-periódicas com
$\norm\cdot_\infty$.\index{teorema de aproximação de Weierstrass}
\end{corollary}

\begin{proof}
(a) Os polinômios formam uma álgebra com constantes; as funções
coordenadas separam os pontos de $\R^n$.
(b) As partes real e imaginária $\frac{f + \bar f}2$, $\frac{f - \bar f}{2\iu}$ dos membros de $\mathcal A$ formam
uma álgebra real $\mathcal A_\R \subseteq \mathcal C(K, \R)$ com constantes; ela separa pontos ($f(x) \ne f(y)$ força
$\operatorname{Re}f$ ou $\operatorname{Im}f$ a separar). Aplique o teorema real e recombine.
(c) Veja as funções $2\pi$-periódicas como $\mathcal C(S^1, \C)$ ($S^1 = \R/2\pi\Z$, \omterm{def:b3:topology:compact}{compacto}:
\cref{exo:b3:topology:5}); a álgebra gerada por $\eu^{\iu t}, \eu^{-\iu t}$ e pelas constantes é estável
por conjugação e separa os pontos do círculo ($\eu^{\iu t}$ é injetora nele).
Aplique (b).
\end{proof}

\begin{remark}\label{rem:b3:complete:swremark}
A versão trigonométrica repara, e generaliza amplamente, a lacuna
deixada no capítulo de Fourier do segundo ano: a densidade dos
polinômios trigonométricos em $(\mathcal C(S^1), \norm\cdot_2)$ decorre a fortiori ($\norm\cdot_2 \leq \norm\cdot_\infty$ a menos da
constante de normalização), o que fará do sistema de Fourier uma
\emph{base} ortonormal no~\cref{ch:b3:hilbert}, demonstrando enfim Parseval em
plena generalidade.
\end{remark}

contração pode se perder: em $X = [1, +\infty)$ (\omterm{def:b3:complete:complete}{completo}, não \omterm{def:b3:topology:compact}{compacto}), $f(x) = x

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:complete:5}

\section{Problema: o teorema de existência de Peano}

\begin{problem}[{Problema de fim de semana --- existência de soluções
de $x' = f(t, x)$ sem Lipschitz}]\label{pb:b3:complete:1}
Cauchy--Lipschitz (segundo ano; redemonstrado no~\cref{ch:b3:ode}) exige
$f$ lipschitziana em $x$. Peano (1890): \emph{a \omterm{def:b3:topology:continuity}{continuidade} de
$f$ já garante a existência} --- embora não a unicidade.
Demonstramo-lo com poligonais de Euler e Ascoli. Cenário: $f \colon R
\to \R^d$
\omterm{def:b3:topology:continuity}{contínua} no retângulo $R = \intcc{t_0 - a}{t_0
+ a}\times \bar B(x_0, b) \subseteq \R\times\R^d$, $M =
\sup_R\norm f$, e
\[
T = \min\bigl(a,\ b/M\bigr)
\qquad (\text{se } M = 0 \text{ o problema é trivial}).
\]

\medskip
\noindent\textbf{Parte I --- Poligonais de Euler.} Para $n \geq 1$,
subdivida $[t_0, t_0 + T]$ por $t_k = t_0 + kT/n$ ($0 \leq k
\leq n$) e defina $\varphi_n$ de maneira afim por
partes: $\varphi_n(t_0) = x_0$ e, em $[t_k, t_{k+1}]$,
\[
\varphi_n(t) = \varphi_n(t_k) + (t -
t_k)\,f\bigl(t_k, \varphi_n(t_k)\bigr).
\]
\begin{enumerate}
  \item Mostre por indução que $\varphi_n$ está bem definida, com $\norm{\varphi_n(t) - x_0} \leq M(t - t_0) \leq b$ em
        $[t_0, t_0 + T]$ --- de modo que os pontos de avaliação permanecem em
        $R$. \emph{(É aqui que $T \leq b/M$ entra.)}
  \item Mostre que cada $\varphi_n$ é $M$-lipschitziana.
  \item Deduza de Arzelà--Ascoli (\cref{thm:b3:complete:ascoli}) que uma subsequência
        $(\varphi_{n_j})$ converge uniformemente em $[t_0, t_0 +
        T]$ para alguma $\varphi$, ela
        própria $M$-lipschitziana e com $\varphi(t_0) = x_0$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O limite resolve a equação.} Defina o
\emph{defeito} $\Delta_n(t) = \varphi_n'(t) -
f\bigl(t, \varphi_n(t)\bigr)$ nos pontos que não são da grade.
\begin{enumerate}[resume]
  \item Mostre que $f$ é uniformemente \omterm{def:b3:topology:continuity}{contínua} em $R$ e deduza:
        para todo $\varepsilon > 0$ existe $n_0$ tal que, para $n \geq n_0$ e todo
        $t$ fora da grade, $\norm{\Delta_n(t)} \leq \varepsilon$. \emph{(Em $(t_k,
        t_{k+1})$, $\varphi_n'(t) = f(t_k, \varphi_n(t_k))$, e $(t, \varphi_n(t))$
        está a distância no máximo $(1 + M)\,T/n$ de $(t_k, \varphi_n(t_k))$.)}
  \item Estabeleça a forma integral: para todo $t$,
        \[
        \varphi_n(t) = x_0 + \int_{t_0}^{t}
        f\bigl(s, \varphi_n(s)\bigr)\dd s +
        \int_{t_0}^t \Delta_n(s)\,\dd s,
        \]
        sendo o integrando do meio \omterm{def:b3:topology:continuity}{contínuo} por partes.
  \item Passe ao limite ao longo de $(n_j)$: mostre que $f(s, \varphi_{n_j}(s)) \to f(s, \varphi(s))$
        \emph{uniformemente} (\omterm{def:b3:topology:continuity}{continuidade} uniforme de $f$, de
        novo), e conclua
        \[
        \varphi(t) = x_0 + \int_{t_0}^t
        f\bigl(s, \varphi(s)\bigr)\dd s .
        \]
  \item Deduza que $\varphi$ é $\mathcal C^1$ em $[t_0, t_0 +
        T]$ e resolve $x' = f(t, x)$, $x(t_0) = x_0$; estenda
        a construção a $[t_0 - T, t_0]$ (inversão do tempo). \emph{Este é o teorema
        de Peano.}
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A unicidade falha de verdade.}
Considere $x' = 2\sqrt{\abs x}$, $x(0) = 0$, em $\R$.
\begin{enumerate}[resume]
  \item Verifique que $f(x) = 2\sqrt{\abs x}$ é \omterm{def:b3:topology:continuity}{contínua} mas não é lipschitziana em
        \omterm{def:b3:topology:topology}{vizinhança} alguma de $0$.
  \item Verifique que $x \equiv 0$ e, para todo $c \geq 0$,
        \[
        x_c(t) = \begin{cases} 0 & t \leq c,\\ (t - c)^2 & t
        \geq c,\end{cases}
        \]
        são todas soluções por $(0,0)$: um \omterm{def:b3:topology:continuity}{contínuo} de soluções
        distintas.
  \item Onde exatamente a iteração de Picard (ponto fixo de Banach) se
        quebra para esse $f$?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Limites do método.}
\begin{enumerate}[resume]
  \item Mostre que o teorema de Peano falha em dimensão infinita:
        admitimos (ou você pode aceitar de boa-fé) o exemplo clássico
        de Dieudonné no espaço $c_0$ das sequências nulas; em vez
        disso, demonstre o ingrediente de dimensão finita que ali
        falha: a bola unitária fechada de $c_0$ (norma do supremo)
        não é compacta --- exiba uma sequência limitada sem
        subsequência convergente e explique qual etapa da Parte I se
        quebra.
  \item Resuma: que hipóteses dão existência? e existência mais
        unicidade? Enuncie com precisão os dois teoremas (Peano;
        Cauchy--Lipschitz) lado a lado.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Osgood: unicidade além de Lipschitz.}
Seja $\omega\colon\intoo0\infty\to\intoo0\infty$ \omterm{def:b3:topology:continuity}{contínua}, não decrescente, com
\[
\int_0^1\frac{\dd r}{\omega(r)} = +\infty
\qquad(\text{divergência em } 0),
\]
e suponha que $f$ satisfaça $\norm{f(t, x) - f(t, y)} \leq
\omega\bigl(\norm{x - y}\bigr)$ em $R$.
\begin{enumerate}[resume]
  \item Verifique que $\omega(r) = Lr$ serve (é lipschitziana), que $\omega(r) = r\log\frac1r$
        (estendida por \omterm{def:b3:topology:continuity}{continuidade}, para $r$ pequeno) serve, embora
        \emph{não} seja $O(r)$, e que $\omega(r) = 2\sqrt r$ não serve. Calcule a
        integral em cada caso.
  \item Sejam $x_1, x_2$ soluções da equação em $[t_0, t_0 + T]$ com $x_1(t_0) = x_2(t_0)$, e ponha
        $\delta(t) =
        \norm{x_1(t) - x_2(t)}$. Mostre, apenas a partir das formas integrais, que para
        $t_0 \leq s \leq t$:
        \[
        \delta(t) \leq \delta(s) +
        \int_s^t\omega\bigl(\delta(v)\bigr)\dd v .
        \]
  \item (Teorema de Osgood) Suponha $\delta(t_1) > 0$ para algum $t_1$, e seja
        $\tau = \sup\{t \leq t_1 : \delta(t) =
        0\}$. Para $s \in \intoo\tau{t_1}$, ponha $u(t) = \delta(s)
        + \int_s^t\omega(\delta(v))\,\dd v$. Mostre que $\delta \leq
        u$, $u' = \omega(\delta) \leq \omega(u)$, e deduza
        \[
        \int_{\delta(s)}^{u(t_1)}\frac{\dd r}{\omega(r)}
        \leq t_1 - s \leq t_1 - \tau .
        \]
        Faça $s \downarrow \tau$ e obtenha uma contradição com a divergência da
        integral. Conclua: \emph{soluções que passam por uma mesma
        condição inicial coincidem} --- unicidade sob a condição de
        Osgood.
  \item Tire as consequências: a unicidade de Cauchy--Lipschitz é o
        caso $\omega(r) = Lr$; a equação $x' =
        x\log\frac1{\abs x}$ (estendida por $0$ em $0$)
        tem soluções únicas embora seu lado direito não seja
        lipschitziano em $0$; e, para $x' = 2\sqrt{\abs x}$, a convergência de $\int_0\frac{\dd r}{2\sqrt r}$
        é exatamente o que permite a uma solução deixar $0$ em tempo
        finito --- compare o valor da integral $\int_0^{h^2}\frac{\dd r}{2\sqrt r} = h$ com o comportamento
        de fuga de $x_c$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Taxas, esquemas, funis.}
\begin{enumerate}[resume]
  \item (Lema integral de Grönwall) Seja $e, \eta \geq 0$ \omterm{def:b3:topology:continuity}{contínua} em $[t_0, t_0+T]$ com
        $e(t) \leq \int_{t_0}
        ^t\bigl(L\,e(s) + \eta(s)\bigr)\dd s$ para todo $t$. Mostre que
        \[
        e(t) \leq \frac{\sup\eta}{L}\bigl(\eu^{L(t - t_0)} -
        1\bigr)
        \]
        \emph{(ponha $G(t) = \int_{t_0}^t(Le + \eta)$, note que $G'
        \leq LG + \sup\eta$ e derive $\eu^{-Lt}G(t)$)}.
  \item (Euler converge com uma taxa) Suponha agora $f$
        $L$-lipschitziana em $x$ e $L'$-lipschitziana em
        $t$ em $R$. Combinando a cota do defeito da questão 4
        (tornada quantitativa: $\norm{\Delta_n} \leq (L' + LM)T/n$) com a questão 17, demonstre
        \[
        \sup_{[t_0, t_0+T]}\norm{\varphi_n - \varphi}
        \;\leq\; \frac{(L' + LM)\,T}{n}\cdot
        \frac{\eu^{LT} - 1}{L} = O\Bigl(\frac1n\Bigr),
        \]
        em que $\varphi$ é \emph{a} solução: com dados lipschitzianos, a
        sequência inteira converge, com uma taxa explícita --- sem
        precisar de subsequências. Por que a unicidade eleva a
        convergência de subsequências à convergência plena, mesmo sem
        esse cálculo?
  \item (O esquema escolhe) Para $x' = 2\sqrt{\abs x}$, $x(0) =
        0$: mostre que toda poligonal
        de Euler é identicamente nula, de modo que o esquema converge
        para a solução $x \equiv 0$; mas, iniciado em $x(0) = \varepsilon > 0$, ele converge
        (quando $n \to \infty$ e, em seguida, $\varepsilon \to 0$) para $t
        \mapsto t^2$, uma solução
        \emph{diferente} pela origem. A não unicidade ressurge como
        sensibilidade do esquema numérico a perturbações.
  \item Mostre que o conjunto $\mathcal S$ de \emph{todas} as soluções de
        $x' = f(t,x)$, $x(t_0) = x_0$ em $[t_0, t_0+T]$ (com valores em $\bar B(x_0, b)$) é não vazio (Parte
        II), uniformemente $M$-lipschitziano e fechado em $\bigl(\mathcal C([t_0, t_0+T], \R^d),
        \norm\cdot_\infty\bigr)$;
        conclua por Ascoli que $\mathcal S$ é \emph{\omterm{def:b3:topology:compact}{compacto}}. (O teorema de
        Kneser acrescenta que $\mathcal S$ é \omterm{def:b3:topology:connected}{conexo}; não o demonstraremos.)
  \item Verifique o fenômeno de Kneser no exemplo: para $x' =
        2\sqrt{\abs x}$, $x(0) = 0$, em
        $[0, 1]$, mostre que $\mathcal S = \{x_c : c \in \intcc0\infty\}$ com $x_\infty \equiv 0$ \emph{(para uma solução
        qualquer, ponha $c =
        \sup\{t : x(t) = 0\}$ e integre $(\sqrt x)' = 1$ em $\{x > 0\}$)}, que $c \mapsto x_c$ é
        \omterm{def:b3:topology:continuity}{contínua} de $\intcc0\infty$ (\omterm{def:b3:topology:locallycompact}{compactificação por um ponto}, isto é, com
        $c = \infty$ colado como limite) em $\mathcal C([0,1])$, e conclua que $\mathcal S$ é de fato
        \omterm{def:b3:topology:compact}{compacto} e \omterm{def:b3:topology:connected}{conexo} --- um funil em forma de segmento.
  \item (Conjuntos atingíveis) Deduza da questão 20 que, para cada
        $t$ fixado, o \emph{conjunto atingível} $\mathcal S(t) =
        \{x(t) : x \in \mathcal S\}$ é \omterm{def:b3:topology:compact}{compacto};
        calcule-o para o exemplo da questão 21 e verifique que ele é
        também \omterm{def:b3:topology:connected}{conexo}: $\mathcal S(t) = \intcc0{t^2}$ --- todo estado intermediário é atingido
        por alguma solução.
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- Complementos: dependência, otimalidade
e um esquema calculado à mão.}
\begin{enumerate}[resume]
  \item (Dependência \omterm{def:b3:topology:continuity}{contínua}) Suponha $f$ $L$-lipschitziana em
        $x$ em $R$ e sejam $x, y$ duas soluções com valores
        iniciais $x_0, y_0$ em $t_0$. Adaptando a demonstração da questão
        17 à desigualdade $e(t) \leq \norm{x_0
        - y_0} + \int_{t_0}^tL\,e(s)\dd s$, mostre que
        \[
        \norm{x(t) - y(t)} \leq \norm{x_0 - y_0}\,
        \eu^{L(t - t_0)},
        \]
        e verifique em $x' = Lx$ que a cota é atingida: Grönwall é ótimo.
        Deduza de novo a unicidade ($x_0 =
        y_0$) e que a aplicação de fluxo
        $x_0 \mapsto x(t; x_0)$ é lipschitziana, de constante $\eu^{LT}$, onde quer que esteja
        definida.
  \item (Osgood é ótimo) Reciprocamente, seja $\omega$ \omterm{def:b3:topology:continuity}{contínua},
        não decrescente, positiva em $\intoo0\infty$, com $\omega(r) \to 0$ quando $r \to 0^+$, mas
        \[
        \int_0^1\frac{\dd r}{\omega(r)} < +\infty,
        \]
        e estenda $f(x) = \omega(\abs x)$ por $f(0) = 0$. Mostre que $\Omega(x) = \int_0^x\frac{\dd r}{\omega(r)}$ é uma bijeção
        crescente de $\intoc0{1}$ sobre $\intoc0{\Omega(1)}$, que sua inversa $g$ resolve
        $g'
        = \omega(g)$ com $g(0^+) = 0$, e que $g$, estendida por $0$ para $t \leq 0$,
        é uma solução $\mathcal C^1$ de $x' = f(x)$ por $(0, 0)$ distinta de $x \equiv
        0$
        \emph{(para $g'(0) = 0$, limite $\frac{g(t)}t$ por $\omega(g(t))$)}. Conclua: a hipótese
        de divergência da questão 15 não é uma comodidade, mas a
        fronteira exata da unicidade; recupere a Parte III a partir de
        $\omega(r) = 2\sqrt r$, $\Omega(x) = \sqrt x$.
  \item (Euler calculado à mão) Para $x' = x$, $x(0) = 1$ em $[0, 1]$:
        mostre que a poligonal de Euler satisfaz $\varphi_n(1) = \bigl(1 + \frac1n\bigr)^n$. Demonstre a
        expansão
        \[
        \Bigl(1 + \frac1n\Bigr)^{n}
        = \eu\Bigl(1 - \frac1{2n} + O\Bigl(\frac1{n^2}\Bigr)
        \Bigr),
        \]
        de modo que o erro em $t = 1$ é $\frac{\eu}{2n} +
        O(n^{-2})$: a taxa $O(\frac1n)$ da
        questão 18, com a constante exata. Confira numericamente para
        $n = 10$: $1.1^{10} = 2.59374$ contra $\eu \approx 2.71828$, um erro $0.12454$ a comparar com
        $\frac{\eu}{20}
        \approx 0.13591$.
\end{enumerate}
\end{problem}
