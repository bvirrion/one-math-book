\chapter{Equações diferenciais ordinárias}\label{ch:b3:ode}

O segundo ano resolveu equações diferenciais lineares e enunciou o
teorema de Cauchy--Lipschitz; este capítulo o demonstra --- e duas
vezes: existência e unicidade pelo ponto fixo de Banach, estrutura
global pela teoria das \omterm{thm:b3:ode:maximal}{soluções maximais} e pelo teorema da \omterm{thm:b3:ode:maximal}{fuga dos
compactos}. A teoria linear é então reconstruída sobre fundamentos
honestos (\omterm{thm:b3:ode:linearstructure}{resolvente}, \omterm{thm:b3:ode:linearstructure}{wronskiano}, \omterm{thm:b3:ode:matrixexp}{exponencial de matriz}, Duhamel), e a
segunda metade do capítulo abre a teoria \emph{qualitativa} --- \omterm{ex:b3:ode:planeclassification}{fluxos},
equilíbrios, funções de Lyapunov e \omterm{thm:b3:ode:linearization}{estabilidade por linearização}: como
compreender soluções que jamais serão calculadas. O pêndulo, no problema
de fim de semana, é o estudo de caso eterno. Ao longo de todo o
capítulo, $U \subseteq \R\times\R^d$ é aberto e $f \colon U \to \R^d$ é \omterm{def:b3:topology:continuity}{contínua}; uma \emph{solução} de $x' = f(t, x)$ é
uma aplicação $\mathcal C^1$ $x \colon I \to \R^d$ (com $I$ intervalo) de gráfico em $U$ que
satisfaz a equação.

\section{Cauchy--Lipschitz}

\begin{definition}\label{def:b3:ode:lipschitz}
$f$ é \emph{localmente lipschitziana em $x$}\index{localmente
lipschitziana} se todo ponto de $U$ tem uma \omterm{def:b3:topology:topology}{vizinhança} $V$ e uma
constante $L$ com $\norm{f(t, x_1) - f(t, x_2)} \leq L\norm{x_1 -
x_2}$ para $(t, x_1), (t, x_2) \in V$. Se $f$ é $\mathcal
C^1$ (ou apenas se
$\partial_xf$ existe e é \omterm{def:b3:topology:continuity}{contínua}), ela é localmente lipschitziana em $x$: em
uma \omterm{def:b3:topology:topology}{vizinhança} compacta convexa, pela desigualdade do valor médio com
$L =
\sup\vertiii{\partial_xf}$.
\end{definition}

\begin{theorem}[Cauchy--Lipschitz, local]
\label{thm:b3:ode:cauchylipschitz}
Sejam $f$ \omterm{def:b3:topology:continuity}{contínua} e \omterm{def:b3:ode:lipschitz}{localmente lipschitziana} em $x$, e $(t_0,
x_0) \in U$.
Existe $T > 0$ tal que o problema de Cauchy
\[
x' = f(t, x), \qquad x(t_0) = x_0
\]
tem exatamente uma solução em $[t_0 - T, t_0 +
T]$.\index{teorema de
Cauchy--Lipschitz}
\end{theorem}

\begin{proof}
Escolha $a, b > 0$ com $Q = [t_0 - a, t_0 + a]\times\bar
B(x_0, b) \subseteq U$, no qual $\norm f \leq M$ e $f$ é $L$-lipschitziana em
$x$. Uma função $\mathcal C^1$ é solução se, e somente se, satisfaz a equação
integral
\[
x(t) = x_0 + \int_{t_0}^{t}f\bigl(s, x(s)\bigr)\,\dd s
\]
(teorema fundamental do cálculo, nos dois sentidos). Sejam $T =
\min\bigl(a, \frac bM, \frac1{2L}\bigr)$, $I = [t_0 - T, t_0
+ T]$, e
\[
\mathcal E = \{x \in \mathcal C(I, \R^d) : \norm{x(t) - x_0}
\leq b\ \text{em } I\},
\]
um subconjunto fechado do espaço de Banach $(\mathcal C(I, \R^d),
\norm\cdot_\infty)$: \omterm{def:b3:complete:complete}{completo}
(\cref{def:b3:complete:complete}). Defina $\Phi(x)(t) = x_0 +
\int_{t_0}^tf(s, x(s))\dd s$: para $x \in \mathcal E$, $\norm{\Phi(x)(t) - x_0} \leq M\abs{t - t_0} \leq MT \leq b$ --- $\Phi$ leva $\mathcal E$
nele mesmo --- e, para $x, y \in
\mathcal E$:
\[
\norm{\Phi(x)(t) - \Phi(y)(t)} \leq \Bigl|\int_{t_0}^t
L\,\norm{x(s) - y(s)}\,\dd s\Bigr| \leq LT\,\norm{x -
y}_\infty \leq \tfrac12\norm{x - y}_\infty :
\]
uma contração. O ponto fixo de Banach (\cref{thm:b3:complete:banach}) dá um único ponto
fixo em $\mathcal E$: existência, e unicidade entre as soluções que permanecem
em $\bar B(x_0, b)$ --- mas toda solução em $I$ ali permanece ($\norm{x(t) - x_0} \leq M\abs{t - t_0} \leq b$ enquanto o
gráfico ficar em $Q$, por um argumento de \omterm{def:b3:topology:continuity}{continuidade}): unicidade em
$I$.
\end{proof}

\begin{lemma}[Grönwall]\label{lem:b3:ode:gronwall}
Sejam $u \colon I \to \intco0\infty$ \omterm{def:b3:topology:continuity}{contínua}, $t_0 \in
I$, e suponha\index{lema de Grönwall}
\[
u(t) \leq \alpha + \beta\,\Bigl|\int_{t_0}^{t}u(s)\,\dd
s\Bigr| \qquad (t \in I)
\]
com $\alpha \geq 0$, $\beta > 0$. Então $u(t) \leq
\alpha\,\eu^{\beta\abs{t - t_0}}$ em $I$.
\end{lemma}

\begin{proof}
Para $t \geq t_0$: ponha $v(t) = \alpha +
\beta\int_{t_0}^tu(s)\dd s$, de modo que $u \leq v$, $v' = \beta u
\leq \beta v$, e $(v\eu^{-\beta(t - t_0)})' \leq 0$: $v(t)
\leq v(t_0)\eu^{\beta(t-t_0)} = \alpha\eu^{\beta(t - t_0)}$. Para $t \leq t_0$,
aplique o mesmo a $\tilde u(t) = u(2t_0 -
t)$.
\end{proof}

\begin{corollary}[Unicidade e dependência contínua]
\label{cor:b3:ode:uniqueness}
Sob as hipóteses do~\cref{thm:b3:ode:cauchylipschitz}, duas soluções de $x' = f(t,x)$ que coincidem
em um ponto coincidem em seu intervalo comum de definição.
Quantitativamente, se $x, y$ são duas soluções cujos gráficos estão em
uma região onde $f$ é $L$-lipschitziana em $x$, então
\[
\norm{x(t) - y(t)} \leq \norm{x(t_0) -
y(t_0)}\,\eu^{L\abs{t - t_0}} .
\]
\end{corollary}

\begin{proof}
A estimativa: $u = \norm{x - y}$ satisfaz $u(t) \leq u(t_0)
+ L\abs{\int_{t_0}^tu}$ (subtraia as equações integrais);
Grönwall. Unicidade global: o conjunto de coincidência $\{t : x(t)
= y(t)\}$ é fechado no
intervalo comum, não vazio e aberto --- em torno de qualquer ponto de
coincidência, cubra um pedaço \omterm{def:b3:topology:compact}{compacto} do gráfico comum por um número
finito de caixas lipschitzianas e aplique a estimativa com $u(t_1) = 0$ em cada
uma: localmente $x \equiv y$. Um subconjunto não vazio, aberto e fechado de um
intervalo é o intervalo todo.
\end{proof}

\section{Soluções maximais}

\begin{theorem}[Soluções maximais; fuga dos compactos]
\label{thm:b3:ode:maximal}
Suponha $f$ \omterm{def:b3:topology:continuity}{contínua} e \omterm{def:b3:ode:lipschitz}{localmente lipschitziana} em $x$.
\begin{enumerate}
  \item Todo problema de Cauchy tem uma única solução \emph{maximal}
        $x \colon \intoo{T_-}{T_+} \to \R^d$: toda outra solução por $(t_0, x_0)$ é sua restrição. O intervalo é
        aberto.
  \item (\emph{Fuga}) Para todo \omterm{def:b3:topology:compact}{compacto} $K \subseteq U$ existe $\varepsilon > 0$ tal que $(t, x(t)) \notin
        K$
        para todo $t \in \intoo{T_+ - \varepsilon}{T_+}$ (e simetricamente em $T_-$): \emph{o gráfico de
        uma \omterm{thm:b3:ode:maximal}{solução maximal} acaba por deixar todo \omterm{def:b3:topology:compact}{compacto} de $U$}.
        Em particular, para $U = \R\times\R^d$ e $T_+ < +\infty$: $\norm{x(t)} \to +\infty$ quando $t \to T_+^-$
        (\emph{explosão}).\index{solução maximal}\index{fuga dos
        compactos}\index{explosão}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Seja $\mathcal S$ o conjunto de todas as soluções por $(t_0, x_0)$;
pelo~\cref{cor:b3:ode:uniqueness}, duas quaisquer coincidem na interseção de seus
intervalos, de modo que se colam: em $J =
\bigcup_{y \in \mathcal S}I_y$, defina $x(t) = y(t)$ para qualquer
$y$ definida em $t$: uma solução bem definida, evidentemente
maximal e única. $J$ é aberto: uma solução definida em uma
extremidade poderia ser prolongada pelo~\cref{thm:b3:ode:cauchylipschitz} nessa
extremidade.

(2) Suponha que a afirmação falhe em $T_+$: existem $t_n \to
T_+$ com $(t_n, x(t_n)) \in K$;
note que isso força $T_+ <
\infty$ ou, se $T_+ = \infty$, nada há a demonstrar ($K$ é
limitado em tempo). Seja, pois, $T_+ < \infty$. \omterm{def:b3:topology:compact}{Compacidade}: constantes uniformes
$M, L, a,
b$ servem para todos os dados de Cauchy em uma \omterm{def:b3:topology:topology}{vizinhança} de $K$
--- concretamente, cubra $K$ por um número finito de caixas
$Q_i$ como na demonstração do teorema local e seja $T^* > 0$ o mínimo dos
tempos de existência correspondentes: todo dado de Cauchy em $K$
lança uma solução que vive pelo menos $T^*$ além de seu instante
inicial. Aplicando isso em $(t_n, x(t_n))$ com $t_n > T_+ -
T^*/2$, estende-se $x$ além de
$T_+$ (a extensão coincide com $x$ por unicidade e, depois, a
prolonga): contradição com a maximalidade. Logo o gráfico deixa
$K$ definitivamente antes de $T_+$. Para $U = \R\times\R^d$: se $\norm{x(t)}\not\to\infty$, uma
sequência $t_n \to T_+$ manteria $(t_n, x(t_n))$ no \omterm{def:b3:topology:compact}{compacto} $[t_0, T_+]\times\bar B(0, R)$: excluído.
\end{proof}

\begin{corollary}[Existência global sob crescimento linear]
\label{cor:b3:ode:globallinear}
Se $U = I\times\R^d$ (com $I$ intervalo aberto) e $\norm{f(t, x)}
\leq \alpha(t)\norm x + \beta(t)$ com $\alpha, \beta$ \omterm{def:b3:topology:continuity}{contínuas},
então toda \omterm{thm:b3:ode:maximal}{solução maximal} está definida em todo $I$.
\end{corollary}

\begin{proof}
Em um \omterm{def:b3:topology:compact}{compacto} $[t_0, T] \subseteq I$: $\norm{x(t)} \leq
\norm{x_0} + \int_{t_0}^t(\alpha\norm x + \beta)$, de modo que, por Grönwall (com $\alpha, \beta$
limitadas por $A, B$ ali), $\norm{x(t)} \leq (\norm{x_0} + B(T - t_0))\eu^{A(T - t_0)}$: limitada. Se $T_+ < \sup I$, o gráfico
permanece em um \omterm{def:b3:topology:compact}{compacto} de $I\times\R^d$ perto de $T_+$: contradiz a fuga
(\cref{thm:b3:ode:maximal}).
\end{proof}

\section{Sistemas lineares}

Ao longo de toda esta seção, $A \colon I \to M_d(\R)$ e $b
\colon I \to \R^d$ são \omterm{def:b3:topology:continuity}{contínuas}; o sistema é
$x' = A(t)x
+ b(t)$ --- crescimento linear: todas as \omterm{thm:b3:ode:maximal}{soluções maximais} vivem em todo
$I$ (\cref{cor:b3:ode:globallinear}).

\begin{theorem}[Estrutura]\label{thm:b3:ode:linearstructure}
As soluções do sistema homogêneo $x' = A(t)x$ formam um espaço vetorial
$S_H$ de dimensão $d$; para cada $t_0$, a avaliação $x \mapsto x(t_0)$ é
um isomorfismo $S_H \to
\R^d$. O \emph{resolvente}\index{resolvente de um sistema
linear} $R(t, s) \in GL_d(\R)$, definido por: $t \mapsto R(t,
s)v$ é a solução de valor $v$ em
$s$, satisfaz
\[
R(s,s) = I,\quad R(t, u)R(u, s) = R(t, s),\quad
\partial_tR(t,s) = A(t)R(t,s),
\]
e o problema não homogêneo é resolvido pela \emph{fórmula de
Duhamel}\index{fórmula de Duhamel}:
\[
x(t) = R(t, t_0)\,x_0 + \int_{t_0}^{t}R(t, s)\,b(s)\,\dd s .
\]
Por fim, o \emph{wronskiano}\index{wronskiano} $w(t) = \det
R(t, s)$ obedece à fórmula
de Liouville $w' =
\operatorname{tr}A(t)\,w$, de modo que $w(t) =
\exp\bigl(\int_s^t\operatorname{tr}A\bigr) > 0$.
\end{theorem}

\begin{proof}
A linearidade da equação faz das soluções um espaço vetorial; a
avaliação é linear, injetora (unicidade: uma solução que se anula em
$t_0$ é $\equiv 0$) e sobrejetora (existência): dimensão $d$. As
propriedades do \omterm{thm:b3:ode:linearstructure}{resolvente} reafirmam a unicidade (ambos os lados de cada
identidade resolvem o mesmo problema de Cauchy); a inversibilidade vem
de $R(s,t)R(t,s) = I$. Duhamel: derive a fórmula --- $x'(t) = A(t)R(t,t_0)x_0 + R(t,t)b(t) +
\int_{t_0}^tA(t)R(t,s)b(s)\dd s = A(t)x(t) + b(t)$ (a derivação sob a
integral é legítima: o integrando é $\mathcal C^1$ em $t$, com derivada
\omterm{def:b3:topology:continuity}{contínua} em $(t,s)$; ou verifique pela equação integral). Liouville:
$w(t + h) = \det\bigl(R(t+h, t)\bigr)w(t)$ e $R(t + h, t) =
I + hA(t) + o(h)$ (da equação integral), de modo que $\det = 1
+ h\operatorname{tr}A(t) + o(h)$ (expansão de
$\det$ em $I$): $w'(t) = \operatorname{tr}A(t)\,w(t)$; integre a EDO linear escalar.
\end{proof}

\begin{theorem}[Exponencial de matriz]\label{thm:b3:ode:matrixexp}
Para $A \in M_d(\C)$, a série $\eu^{A} =
\sum_{n\geq0}\frac{A^n}{n!}$ converge (absolutamente, em qualquer norma
submultiplicativa), $\eu^{A+B} = \eu^A\eu^B$ sempre que $AB = BA$, e $t \mapsto \eu^{tA}$ é o \omterm{thm:b3:ode:linearstructure}{resolvente} do
sistema de coeficientes constantes: $R(t,s) = \eu^{(t-s)A}$; ela é $\mathcal
C^\infty$ com $\frac{\dd}{\dd t}\eu^{tA} = A\eu^{tA}$. Além
disso: se $\operatorname{Re}\lambda < -\alpha < 0$ para todo autovalor $\lambda$ de $A$, então $\vertiii{\eu^{tA}}
\leq C\eu^{-\alpha t}$ para
$t \geq 0$.\index{exponencial de matriz}
\end{theorem}

\begin{proof}
Convergência: $\vertiii{A^n/n!} \leq \vertiii A^n/n!$, somável (\cref{exo:b3:complete:1}(b) na álgebra de Banach
$M_d$). Para $A, B$ que comutam: o produto de Cauchy das duas
séries absolutamente convergentes se rearranja, via o teorema binomial
(válido quando $AB = BA$), em $\sum_n\frac{(A+B)^n}{n!}$. Derivabilidade, diretamente: $\frac{\eu^{(t+h)A} - \eu^{tA}}h = \eu^{tA}\frac{\eu^{hA} -
I}{h} \to \eu^{tA}A$,
pois $\norm{\frac{\eu^{hA} - I}h - A}
\leq \sum_{n\geq2}\frac{\abs h^{n-1}\vertiii A^n}{n!} =
O(h)$. Logo $t \mapsto \eu^{(t - s)A}v$ resolve o problema de Cauchy que define $R(t,s)v$.
Cota espectral: pela \omterm{thm:b3:modules:jordan}{forma de Jordan} (\cref{thm:b3:modules:jordan}), $A = P(D + N)P^{-1}$ com
$D$ diagonal carregando os autovalores, $N$ nilpotente e $DN =
ND$.
Então $\eu^{tA} = P\,\eu^{tD}\eu^{tN}P^{-1}$ com $\vertiii{\eu^{tD}} \leq \eu^{-(\alpha + \delta)t}$ para algum $\delta > 0$ ($t \geq 0$) e $\eu^{tN}$ polinomial em
$t$ (nilpotência): o produto é $\leq C\eu^{-\alpha t}$ (um polinômio é vencido por
$\eu^{-\delta t}$).
\end{proof}

\begin{example}[O plano, classificado]
\label{ex:b3:ode:planeclassification}
Para $x' = Ax$ com $A \in M_2(\R)$ inversível, o retrato de fase perto de $0$ é
decidido por $\tau = \operatorname{tr}A$ e $\delta = \det A$, através dos autovalores $\lambda_\pm
= \frac{\tau \pm \sqrt{\tau^2 - 4\delta}}2$:
\begin{itemize}
  \item $\delta < 0$: autovalores reais de sinais opostos --- uma
        \emph{sela}; duas trajetórias entram, duas saem, e todas as
        demais passam ao largo. Sempre instável.
  \item $\delta > 0$, $\tau^2 \geq 4\delta$: autovalores reais de mesmo sinal ($= \operatorname{sign}\tau$) --- um
        \emph{nó}, \omterm{def:b3:ode:stability}{estável} se, e somente se, $\tau < 0$; as trajetórias são
        tangentes à direção própria lenta.
  \item $\delta > 0$, $\tau^2 < 4\delta$, $\tau \neq 0$: autovalores complexos conjugados $\frac\tau2 \pm
        \iu\omega$ ---
        uma \emph{espiral} (foco), \omterm{def:b3:ode:stability}{estável} se, e somente se, $\tau < 0$; as
        soluções são rotações $\eu^{\tau t/2}\times$ de período $\frac{2\pi}\omega$.
  \item $\tau = 0$, $\delta > 0$: autovalores puramente imaginários --- um
        \emph{\omterm{ex:b3:groups:actions}{centro}}: \omterm{def:b3:groups:action}{órbitas} fechadas (elipses), estabilidade sem
        estabilidade assintótica, exatamente o caso limite que
        o~\cref{thm:b3:ode:linearization} não consegue decidir para sistemas não
        lineares (o equilíbrio inferior do pêndulo, \cref{pb:b3:ode:1}, fica
        aqui).
\end{itemize}
A parábola de fronteira $\tau^2 = 4\delta$ carrega os nós degenerados (blocos de
Jordan: trajetórias com uma única direção tangente). Tudo se lê em dois
números --- e é por isso que o primeiro reflexo diante de um retrato de
fase plano é calcular $\operatorname{tr}$ e $\det$; por exemplo, $A = \bigl(\begin{smallmatrix}0 & 1\\ -1 &
-c\end{smallmatrix}\bigr)$ (oscilador
amortecido): $\delta = 1 >
0$, $\tau = -c$: espiral \omterm{def:b3:ode:stability}{estável} para $0 < c < 2$, nó \omterm{def:b3:ode:stability}{estável} para
$c \geq 2$ --- subamortecimento contra superamortecimento, em um relance.
\end{example}

\section{Fluxos, equilíbrios, estabilidade}

Considere agora a equação \emph{autônoma} $x' = F(x)$, com $F
\colon \Omega \to \R^d$ \omterm{def:b3:ode:lipschitz}{localmente
lipschitziana} no aberto $\Omega
\subseteq \R^d$. Escreva $\varphi_t(x_0) = x(t)$ para a \omterm{thm:b3:ode:maximal}{solução maximal} com
$x(0) = x_0$ (o \emph{fluxo}\index{fluxo de um campo de vetores}); a autonomia
dá a propriedade de grupo $\varphi_{t+s} =
\varphi_t\circ\varphi_s$ onde definida (ambos os lados resolvem o
mesmo problema no instante $s$).

\begin{definition}\label{def:b3:ode:stability}
Um \emph{equilíbrio} é um ponto $\bar x$ com $F(\bar x) =
0$ (de modo que
$\varphi_t(\bar x) = \bar x$). Ele é \emph{estável}\index{estabilidade (Lyapunov)} se, para
todo $\varepsilon > 0$, existe $\delta > 0$ tal que $\norm{x_0 - \bar x} < \delta$ implica que a solução existe para
todo $t \geq 0$ com $\norm{\varphi_t(x_0) - \bar
x} < \varepsilon$; é \emph{assintoticamente estável} se, além disso,
$\varphi_t(x_0) \to \bar x$ para todo $x_0$ perto de $\bar x$.
\end{definition}

\begin{theorem}[Funções de Lyapunov]\label{thm:b3:ode:lyapunov}
Sejam $\bar x$ um equilíbrio e $V \colon \mathcal V \to
\R$ uma função $\mathcal C^1$ em uma
\omterm{def:b3:topology:topology}{vizinhança} de $\bar x$ com:\index{função de Lyapunov}
\[
V(\bar x) = 0,\qquad V(x) > 0 \text{ para } x \neq \bar x,
\qquad \dot V(x) = \nabla V(x)\cdot F(x) \leq 0 .
\]
Então $\bar x$ é \omterm{def:b3:ode:stability}{estável}. Se, além disso, $\dot V < 0 $ fora de $\bar
x$,
então $\bar x$ é assintoticamente \omterm{def:b3:ode:stability}{estável}.
\end{theorem}

\begin{proof}
Ao longo de uma solução, $\frac{\dd}{\dd t}V(x(t)) = \dot V(x(t))
\leq 0$: $V$ decresce. Dado $\varepsilon$ (pequeno o
bastante para que $\bar B(\bar x, \varepsilon) \subseteq \mathcal V$), ponha $m = \min\{V(x) : \norm{x - \bar x} = \varepsilon\} > 0$ (\omterm{def:b3:topology:compact}{compacidade}, positividade) e
escolha $\delta < \varepsilon$ com $V < m$ em $B(\bar x, \delta)$ (\omterm{def:b3:topology:continuity}{continuidade}). Uma solução que
parte de $B(\bar x, \delta)$ satisfaz $V(x(t)) < m$ em todos os instantes posteriores, de modo
que nunca pode atingir a esfera $\norm{x -
\bar x} = \varepsilon$ (onde $V \geq m$): ela permanece na
bola --- e então existe para todo $t \geq 0$: a solução fica no \omterm{def:b3:topology:compact}{compacto}
$\bar B$, de modo que o~\cref{thm:b3:ode:maximal}(2) (\omterm{thm:b3:ode:maximal}{fuga dos compactos}) força
$T_+ = +\infty$. Eis a estabilidade.

Caso assintótico: seja $x(t)$ partindo de $B(\bar x, \delta)$; $V(x(t))$ decresce até
algum $c \geq 0$. Se $c > 0$: a trajetória permanece em $K = \{x \in \bar B(\bar x,
\varepsilon): V(x) \geq c\}$, um \omterm{def:b3:topology:compact}{compacto}
que exclui uma \omterm{def:b3:topology:topology}{vizinhança} de $\bar x$ ($V$ é \omterm{def:b3:topology:continuity}{contínua} com $V(\bar x)
= 0 < c$).
Em $K$, a função $\dot V$ é \omterm{def:b3:topology:continuity}{contínua} e estritamente negativa,
logo $\mu = \max_K\dot V < 0$ (\omterm{def:b3:topology:compact}{compacidade}); então $V(x(t)) \leq V(x(0)) + \mu t \to
-\infty$: absurdo, $V \geq 0$. Assim $c = 0$
e $x(t) \to \bar x$ (os pontos a distância $\geq
\rho$ de $\bar x$ dentro da bola têm
$V \geq m_\rho >
0$).
\end{proof}

\begin{theorem}[Estabilidade por linearização]
\label{thm:b3:ode:linearization}
Sejam $F$ de classe $\mathcal C^1$, $F(\bar x) = 0$, $A = DF(\bar x)$. Se todo autovalor de
$A$ tem $\operatorname{Re}\lambda <
0$, então $\bar x$ é assintoticamente
\omterm{def:b3:ode:stability}{estável}.\index{estabilidade por linearização}
\end{theorem}

\begin{proof}
Translade $\bar x$ para $0$ e escreva $F(x) = Ax + g(x)$ com $g(x) = o(\norm x)$
(derivabilidade $\mathcal C^1$). Tome $\alpha > 0$ com $\vertiii{\eu^{tA}} \leq C\eu^{-\alpha t}$ ($t \geq 0$; \cref{thm:b3:ode:matrixexp}) e
$r > 0$ com $\norm{g(x)} \leq \frac{\alpha}{2C}\norm x$ para $\norm x \leq
r$. Duhamel com $b(s) = g(x(s))$:
\[
x(t) = \eu^{tA}x_0 + \int_0^t\eu^{(t-s)A}g(x(s))\,\dd s,
\]
válido enquanto $\norm{x(s)} \leq r$. Então $u(t) =
\eu^{\alpha t}\norm{x(t)}$ satisfaz
\[
u(t) \leq C\norm{x_0} +
\int_0^{t}C\,\frac{\alpha}{2C}\,u(s)\,\dd s ,
\]
de modo que Grönwall dá $u(t) \leq C\norm{x_0}\eu^{\alpha t/2}$, isto é, $\norm{x(t)} \leq C\norm{x_0}\eu^{-\alpha t/2}$. Se $\norm{x_0} < r/C$, a cota a priori
mantém $\norm{x(t)} <
r$ para todo $t$ (um argumento de \omterm{def:b3:topology:continuity}{continuidade} e
bootstrapping: o conjunto dos instantes em que $\norm x \leq r$ é aberto e fechado
em $\intco0{T_+}$, dada a estimativa estrita), a solução é global e converge a
$0$ exponencialmente: estabilidade assintótica.
\end{proof}

\begin{method}\label{met:b3:ode:strategy}
Diante de uma EDO: (1) existência e unicidade --- verifique se é
\omterm{def:b3:ode:lipschitz}{localmente lipschitziana} (em geral $\mathcal C^1$); (2) globalidade ---
crescimento linear, limitação, ou um \omterm{def:b3:topology:compact}{compacto} invariante via \omterm{thm:b3:ode:lyapunov}{função de
Lyapunov} ou integral primeira; na falta disso, suspeite de explosão e
teste na caricatura escalar $x' = x^2$; (3) sistemas lineares ---
\omterm{thm:b3:ode:linearstructure}{resolvente}, Duhamel e, para coeficientes constantes, a estrutura
espectral de $A$; (4) questões qualitativas --- equilíbrios,
linearize e cace uma \omterm{thm:b3:ode:lyapunov}{função de Lyapunov} (a energia, quando o sistema é
mecânico) ou uma integral primeira cujos conjuntos de nível aprisionem
as trajetórias. O problema de fim de semana percorre o método inteiro no
pêndulo.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:ode:1}
Resolva explicitamente e determine o intervalo maximal:
(a) $x' = x^2$, $x(0) = 1$; (b) $x' = 1 + x^2$, $x(0) = 0$; (c) $x' = x(1-x)$, $x(0) = \frac12$. Reconcilie cada
resposta com o~\cref{thm:b3:ode:maximal}(2) e com o~\cref{cor:b3:ode:globallinear}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:ode:2}
Sejam $x, y$ soluções de $x' = f(t,x)$ com $f$ globalmente
$L$-lipschitziana em $x$ em $\R\times\R^d$.
(a) Demonstre $\norm{x(t) - y(t)} \leq \norm{x(0) -
y(0)}\eu^{L\abs t}$ e mostre com um exemplo (linear!) que o fator $\eu^{L\abs t}$
é atingido.
(b) Deduza que a aplicação de \omterm{ex:b3:ode:planeclassification}{fluxo} $x_0 \mapsto x(t; x_0)$ é \omterm{def:b3:topology:continuity}{contínua} e, de fato,
lipschitziana nos conjuntos limitados.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:3}
Mostre que cada uma das seguintes tem todas as \omterm{thm:b3:ode:maximal}{soluções maximais} globais
em $\R$, citando o teorema adequado:
(a) $x' = \sin(tx)$;
(b) $x' = \frac{t\,x}{1 + x^2}$;
(c) $x'' + q(t)x = 0$ com $q$ \omterm{def:b3:topology:continuity}{contínua} (converta a um sistema de primeira
ordem);
(d) $x' = A(t)x$ com $A$ \omterm{def:b3:topology:continuity}{contínua} e limitada --- e dê a cota de Grönwall
para $\norm{x(t)}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:4}
(a) Calcule $\eu^{tA}$ para $A = \bigl(\begin{smallmatrix}0 & -1\\ 1 &
0\end{smallmatrix}\bigr)$, $\bigl(\begin{smallmatrix}\lambda & 1\\ 0 &
\lambda\end{smallmatrix}\bigr)$ e $\bigl(\begin{smallmatrix}0 & 1\\ 1 &
0\end{smallmatrix}\bigr)$.
(b) Resolva o oscilador forçado $x'' + x = \cos(\omega t)$ por Duhamel (na forma de
sistema), para $\omega \neq 1$ e para $\omega =
1$: a ressonância aparece como o termo
secular $t\sin t$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:5}
Para a equação escalar $x'' + p(t)x' + q(t)x = 0$:
(a) Mostre que o \omterm{thm:b3:ode:linearstructure}{wronskiano} $w = x_1x_2' - x_1'x_2$ de duas soluções satisfaz $w' = -p\,w$
(Abel) e deduza que duas soluções com $w \neq 0$ em algum ponto formam uma
base.
(b) Dada uma solução $x_1$ que não se anula, encontre a solução
geral por \emph{redução de ordem}: ponha $x_2 = x_1\int
\frac{\eu^{-\int p}}{x_1^2}$ e verifique. Aplique a
$t^2x'' -
2x = 0$ em $\intoo0{+\infty}$ com $x_1(t) = t^2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:6}
(Logística) Para $x' = x(1 - x)$: determine todos os equilíbrios e sua
estabilidade (pelo~\cref{thm:b3:ode:linearization} e diretamente); mostre que toda
solução com $x(0) \in \intoo01$ é crescente, global, com limites $0$ e $1$ em
$\mp\infty$; e resolva explicitamente para confirmar. Mostre, mais
geralmente, que as soluções escalares autônomas são monótonas e conclua:
não há soluções periódicas não constantes em dimensão $1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:7}
(Integrais primeiras) Seja $H \colon \Omega \to \R$ de classe $\mathcal
C^1$ e considere o sistema
hamiltoniano plano $x' =
\partial_yH$, $y' = -\partial_xH$.
(a) Mostre que $H$ é constante ao longo das soluções.
(b) Para $H = \frac{y^2}2 + \frac{x^4}4$: mostre que todas as soluções são globais e limitadas, e
que a origem é \omterm{def:b3:ode:stability}{estável} (Lyapunov: $H$), embora a linearização
($\bigl(\begin{smallmatrix}0&1\\0&0\end{smallmatrix}\bigr)$) \emph{não} seja assintoticamente \omterm{def:b3:ode:stability}{estável}: a linearização pode
ser inconclusiva.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:8}
(Pêndulo amortecido) $x'' + cx' + \sin x = 0$, $c > 0$; sistema: $x' = y$, $y' = -\sin x - cy$.
(a) Mostre que $V(x, y) = \frac{y^2}2 + 1 - \cos x$ satisfaz $\dot
V = -cy^2 \leq 0$: a origem é \omterm{def:b3:ode:stability}{estável}.
(b) $\dot V$ se anula em todo o eixo $y = 0$: o critério estrito
de Lyapunov falha. Demonstre a estabilidade assintótica assim mesmo,
por linearização (\cref{thm:b3:ode:linearization}): calcule os autovalores da matriz
linearizada em $(0,0)$ e verifique $\operatorname{Re} < 0$ para todo $c > 0$.
(c) O que acontece no equilíbrio $(\pi, 0)$? Calcule a linearização e conclua
(um autovalor positivo: instabilidade --- você pode usar o enunciado de
instabilidade informalmente ou produzir uma solução explícita do sistema
linear que escapa).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:9}
(Fronteira da unicidade) Para $\alpha \in \intoo01$, mostre que o problema $x' = \abs x^{\alpha}$, $x(0) = 0$
tem infinitas soluções (adapte o~\cref{pb:b3:complete:1}, Parte III). Mostre, ao
contrário, que para $\alpha = 1$ (isto é, $x' = \abs
x$) a solução por $0$ é única,
e identifique com precisão qual hipótese do~\cref{thm:b3:ode:cauchylipschitz} distingue os
dois casos.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:10}
(Cotas a priori aprisionam soluções) Seja $F \colon \R^d \to \R^d$ \omterm{def:b3:ode:lipschitz}{localmente
lipschitziana} com $\langle F(x), x\rangle \leq 0$ sempre que $\norm x \geq R$.
(a) Mostre que a bola fechada $\bar B(0, R)$ é positivamente invariante: as
soluções que partem de dentro ali permanecem para $t \geq
0$. \emph{(Se
$\norm{x(t_2)} > R$, considere o último instante $t_1 < t_2$ com $\norm{x(t_1)} = R$ e estude $\frac{\dd}{\dd t}\norm{x(t)}^2$ em
$\intcc{t_1}{t_2}$.)}
(b) Deduza a existência global para a frente com dados na bola. Trate
em seguida o sistema gradiente $x' = -\nabla G(x)$, $G \in
\mathcal C^2$ com $G(x) \to +\infty$ quando $\norm x \to
\infty$:
mostre que $G$ decresce ao longo das soluções, que cada solução
permanece no conjunto de subnível (limitado) $\{G \leq
G(x_0)\}$, e conclua a
existência global para a frente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:11}
(Explosão por comparação) Considere $x' = x^2 + t^2$, $x(0) =
1$.
(a) Mostre que a \omterm{thm:b3:ode:maximal}{solução maximal} existe em algum $\intco0{T_+}$ com $T_+ < \infty$:
compare com $y' = y^2$, $y(0) = 1$ \emph{(demonstre o lema de comparação de que
precisa: se $x' \geq
F(x)$ e $y' = F(y)$ com $x(0) \geq y(0)$, então $x \geq
y$ onde ambas vivem)}, e
deduza $T_+ \leq 1$.
(b) Limite $T_+$ por baixo: em $\intcc01$, $x' \leq x^2 +
1$; compare com a
supersolução $z' = z^2 + 1$, $z(0) =
1$, resolvida por $z(t) = \tan\bigl(t + \frac\pi4\bigr)$, e conclua $T_+ \geq \frac\pi4$.
(c) Reúna $\frac\pi4 \leq T_+ \leq 1$ e formule a moral: o crescimento superlinear do membro
direito é o que mata a existência global (sendo o~\cref{exo:b3:ode:3} o
contraponto), estando a fronteira na convergência de $\int^{\infty}\frac{\dd s}{F(s)}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:12}
(Teorema de comparação de Sturm) Sejam $q_1 \leq q_2$ \omterm{def:b3:topology:continuity}{contínuas} em um intervalo
$I$, e sejam $u \neq 0$ solução de $u''
+ q_1u = 0$ e $v \neq 0$ solução de $v'' + q_2v = 0$.
(a) Estabeleça a identidade do \omterm{thm:b3:ode:linearstructure}{wronskiano}: com $W = uv' - u'v$, $W' = (q_1 - q_2)\,uv$.
(b) (Sturm) Mostre que, entre dois zeros consecutivos $a < b$ de
$u$, ou $v$ se anula em algum ponto de $\intoo ab$, ou $q_1 = q_2$ e $v \propto u$
ali \emph{(suponha $u > 0$ em $\intoo ab$ e $v > 0$ também; integre (a)
de $a$ a $b$ e examine os sinais dos termos de fronteira $W(a),
W(b)$)}.
(c) Deduza: as soluções de $u'' + qu = 0$ com $q \geq m > 0$ se anulam ao menos uma vez em
todo intervalo de comprimento $\pi/\sqrt m$ (compare com $v'' + mv = 0$); as soluções com
$q \leq 0$ se anulam no máximo uma vez em $\R$. Teste ambas em $q \equiv \pm1$.
\end{exercise}

\section{Problema: o pêndulo, completamente resolvido}

\begin{problem}[{Problema de fim de semana --- oscilações, rotações,
separatriz e o período}]\label{pb:b3:ode:1}
A equação do pêndulo $x'' = -\sin x$ --- como sistema: $x' =
y$, $y' = -\sin x$ em
$\R^2$ --- é a drosófila da dinâmica: simples de escrever,
impossível de resolver por fórmulas elementares e, ainda assim,
inteiramente compreensível pelo método qualitativo. Seja $E(x, y) = \frac{y^2}2 - \cos x$ (a
\emph{energia}).

\medskip
\noindent\textbf{Parte I --- Estrutura global.}
\begin{enumerate}
  \item Mostre que todas as \omterm{thm:b3:ode:maximal}{soluções maximais} são globais (definidas em
        $\R$): use $\dot E = 0$ e o~\cref{thm:b3:ode:maximal}. Equilíbrios: $(k\pi, 0)$;
        classifique suas linearizações (do tipo \omterm{ex:b3:groups:actions}{centro} para $k$ par,
        sela para $k$ ímpar).
  \item Mostre que as trajetórias estão contidas nos conjuntos de nível
        $\{E = E_0\}$, e esboce-as ou descreva-as pelo valor de $E_0 \in \intco{-1}{+\infty}$: $E_0 = -1$
        (equilíbrios), $-1 < E_0 < 1$ (curvas fechadas em torno de $(2k\pi, 0)$), $E_0 = 1$
        (a separatriz por $(\pm\pi, 0)$), $E_0 > 1$ (gráficos sobre $x$:
        rotações).
  \item Demonstre que o equilíbrio inferior $(0,0)$ é \omterm{def:b3:ode:stability}{estável} mas
        \emph{não} assintoticamente \omterm{def:b3:ode:stability}{estável}. \emph{(Lyapunov com $V = E + 1$;
        não assintótico: a conservação de energia aprisiona as \omterm{def:b3:groups:action}{órbitas}
        em curvas de nível afastadas da origem.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Oscilações e seu período.} Fixe $-1 < E_0 < 1$ e
escreva $E_0 = -\cos a$ com $a \in
\intoo0\pi$ (a amplitude).
\begin{enumerate}[resume]
  \item Mostre que a solução com $x(0) = a$, $y(0) = 0$ oscila: $x(t) \in \intcc{-a}a$, e que a
        \omterm{def:b3:groups:action}{órbita} é a curva fechada $y^2 = 2(\cos x - \cos a)$. Justifique que a solução é
        periódica: a \omterm{def:b3:groups:action}{órbita} é uma curva compacta sem equilíbrios,
        percorrida com velocidade limitada inferiormente --- transforme
        isso em um argumento (a solução volta a seu ponto inicial em
        tempo finito e, então, a unicidade força a periodicidade).
  \item Estabeleça a fórmula do período
        \[
        T(a) = 4\int_0^{a}\frac{\dd x}{\sqrt{2(\cos x -
        \cos a)}}
        \]
        \emph{(em um quarto de \omterm{def:b3:groups:action}{órbita}, $y = \frac{\dd x}{\dd t} >
        0$ e separe variáveis;
        justifique a convergência imprópria em $x = a$)}.
  \item (Pequenas oscilações) Substitua $\sin\frac x2 =
        \sin\frac a2\,\sin\varphi$ e mostre que
        \[
        T(a) = 4\int_0^{\pi/2}\frac{\dd\varphi}{\sqrt{1 -
        k^2\sin^2\varphi}},
        \qquad k = \sin\frac a2
        \]
        (uma integral elíptica \omterm{def:b3:complete:complete}{completa}). Deduza por convergência
        dominada que $T(a) \to 2\pi$ quando $a \to 0^+$: o limite harmônico,
        independente da amplitude --- o isocronismo aproximado de
        Galileu, com sua correção exata $T(a) = 2\pi\bigl(1 + \frac{k^2}4 +
        O(k^4)\bigr)$ (expanda o integrando e
        integre termo a termo, justificando pela convergência normal).
  \item Mostre que $T(a) \to +\infty$ quando $a \to \pi^-$ \emph{(limite o integrando por
        baixo perto de $\varphi =
        \frac\pi2$ quando $k \to 1$, ou aplique a convergência
        monótona)}: ao se aproximar da separatriz, o pêndulo desacelera
        sem limite.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A separatriz.}
\begin{enumerate}[resume]
  \item Para $E_0 = 1$, $y = 2\cos\frac x2$ no ramo superior: separe variáveis e integre
        para encontrar a solução explícita
        \[
        x(t) = 4\arctan\bigl(\eu^{t}\bigr) - \pi
        \]
        (com $x(0) = 0$, $y(0) = 2$). Verifique diretamente que ela resolve a
        equação do pêndulo e calcule seus limites e os limites de
        $y(t)$ quando $t \to
        \pm\infty$.
  \item Conclua: a \omterm{def:b3:groups:action}{órbita} separatriz liga a sela $(-\pi, 0)$ (quando $t \to -\infty$) à
        sela $(\pi, 0)$ (quando $t \to +\infty$), mas não alcança nenhuma delas em tempo
        finito --- coerentemente com a unicidade (por que alcançar uma
        sela em tempo finito contradiria o~\cref{cor:b3:ode:uniqueness}?).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Rotações e o quadro \omterm{def:b3:complete:complete}{completo}.}
\begin{enumerate}[resume]
  \item Para $E_0 > 1$: mostre que $y$ nunca se anula, que $x$ é
        estritamente monótona e global com $x(t) \to
        \pm\infty$, e que $t \mapsto y(t)$ é periódica
        de período
        \[
        \tau(E_0) = \int_{-\pi}^{\pi}
        \frac{\dd x}{\sqrt{2(E_0 + \cos x)}} .
        \]
  \item Monte o retrato de fase \omterm{def:b3:complete:complete}{completo} (a figura do capítulo), com
        justificativa integral de cada característica, e escreva um
        resumo de dez linhas do método: energia, conjuntos de nível,
        \omterm{def:b3:topology:compact}{compacidade}, unicidade --- como cada teorema do capítulo
        entrou. Em que momento precisamos de uma fórmula para a solução
        geral?
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- A função período sob o microscópio.}
\begin{enumerate}[resume]
  \item Demonstre os momentos de Wallis
        \[
        W_n = \int_0^{\pi/2}\sin^{2n}\varphi\,\dd\varphi
        = \frac\pi2\cdot\frac{(2n)!}{4^n\,(n!)^2}
        \]
        por indução (integre por partes), expanda o integrando da
        questão 6 pela série binomial e justifique a integração termo a
        termo para obter a série \omterm{def:b3:complete:complete}{completa}
        \[
        T(a) = 2\pi\sum_{n\geq0}
        \Bigl(\frac{(2n)!}{4^n(n!)^2}\Bigr)^{\!2}k^{2n}
        = 2\pi\Bigl(1 + \frac{k^2}4 + \frac{9k^4}{64}
        + O(k^6)\Bigr),
        \qquad k = \sin\frac a2 .
        \]
  \item Converta para a amplitude:
        \[
        T(a) = 2\pi\Bigl(1 + \frac{a^2}{16} +
        \frac{11\,a^4}{3072} + O(a^6)\Bigr)
        \]
        \emph{(substitua a expansão de $\sin\frac a2$ e agrupe)}. O isocronismo
        falha na ordem $a^2$, e a falha está agora quantificada até
        a ordem $a^4$.
  \item Mostre que $a \mapsto T(a)$ é \omterm{def:b3:topology:continuity}{contínua} e estritamente crescente em $\intoo0\pi$, e
        conclua, com as questões 6--7, que $T$ é uma bijeção de $\intoo0\pi$
        sobre $\intoo{2\pi}{+\infty}$: todo período supercrítico é realizado por
        exatamente uma amplitude.
  \item (Aritmética do relojoeiro) Um pêndulo regulado a amplitude
        infinitesimal marca o tempo \omterm{def:b3:rings:ideal}{ideal}; mostre que, operando a
        amplitude $a$, ele atrasa a fração $\frac{a^2}{16} + O(a^4)$ do tempo \omterm{def:b3:rings:ideal}{ideal}, e
        calcule a deriva para $a = 0.2$ rad: cerca de $216$ segundos por
        dia. (As bochechas cicloidais de Huygens e as pequenas
        amplitudes constantes dos escapamentos são ambas respostas a
        esse número.)
  \item Volte ao período de rotação $\tau$ da questão 10: mostre que
        $\tau$ é estritamente decrescente em $\intoo1{+\infty}$, que $\tau(E_0) \to +\infty$ quando
        $E_0 \to 1^+$ (convergência monótona) e que $\sqrt{2E_0}\,\tau(E_0) \to 2\pi$ quando $E_0 \to
        +\infty$
        (convergência dominada): o giro rápido é assintoticamente uma
        rotação livre de velocidade angular $\sqrt{2E_0}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- O método exportado: Lotka--Volterra.} A
receita do pêndulo --- integral primeira, curvas de nível compactas,
unicidade --- resolve um ecossistema. Fixe $\alpha, \beta, \gamma, \delta > 0$ e considere, no
quadrante aberto $Q = \intoo0{+\infty}\times\intoo0{+\infty}$,
\[
x' = x\,(\alpha - \beta y), \qquad
y' = y\,(\delta x - \gamma)
\]
($x$ presas, $y$ predadores).
\begin{enumerate}[resume]
  \item Mostre que $Q$ é invariante --- os eixos são reuniões de
        \omterm{def:b3:groups:action}{órbitas}, explicitamente calculáveis, que nenhuma solução pode
        cruzar (\cref{cor:b3:ode:uniqueness}) --- e que o único equilíbrio em $Q$
        é $(x_*, y_*) =
        \bigl(\frac\gamma\delta, \frac\alpha\beta\bigr)$.
  \item Mostre que
        \[
        H(x, y) = \delta x - \gamma\ln x + \beta y -
        \alpha\ln y
        \]
        é uma integral primeira, que $H = f(x) + g(y)$ com $f, g$ estritamente
        convexas e próprias em $\intoo0{+\infty}$ com mínimos em $x_*$, $y_*$,
        e deduza que todas as \omterm{thm:b3:ode:maximal}{soluções maximais} em $Q$ são globais.
  \item Mostre que, para $h > h_* = H(x_*, y_*)$, o conjunto de nível $\{H = h\}\cap Q$ é uma curva
        fechada em torno do equilíbrio: dois ramos \omterm{def:b3:topology:continuity}{contínuos} $y_\pm(x)$ sobre
        um intervalo \omterm{def:b3:topology:compact}{compacto} $\intcc{x_-}{x_+} \ni x_*$, colados nas extremidades --- o
        análogo dos ovais do pêndulo.
  \item Demonstre que toda \omterm{def:b3:groups:action}{órbita} não estacionária em $Q$ é
        periódica: estabeleça a circulação anti-horária pelas quatro
        regiões recortadas pelas retas $x = x_*$ e $y = y_*$, limite o tempo de
        travessia de cada arco por uma integral com singularidade de
        raiz quadrada convergente (como na questão 5) e feche com a
        unicidade (como na questão 4).
  \item (Lei das médias de Volterra) Se $T$ é o período de uma tal
        \omterm{def:b3:groups:action}{órbita}, mostre que
        \[
        \frac1T\int_0^Tx(t)\,\dd t = \frac\gamma\delta,
        \qquad
        \frac1T\int_0^Ty(t)\,\dd t = \frac\alpha\beta :
        \]
        as médias temporais são iguais aos valores de equilíbrio,
        qualquer que seja a amplitude \emph{(integre $(\ln x)' =
        \alpha - \beta y$ ao longo de um
        período)}.
  \item (O paradoxo da pesca) Colha as duas espécies à taxa $\varepsilon \in \intoo0\alpha$: o
        sistema mantém sua forma, com $\alpha - \varepsilon$ e $\gamma +
        \varepsilon$ no lugar de
        $\alpha$ e $\gamma$. O que acontece com as populações
        médias? Explique a observação de d'Ancona (1914--1918): quando
        a pesca no Adriático diminuiu durante a guerra, a proporção de
        predadores (tubarões) na captura \emph{aumentou} --- e por que a
        pesca moderada favorece as presas.
  \item Escreva a moral em dez linhas: que teoremas do capítulo movem
        cada etapa, o que substitui a energia do pêndulo, e por que
        nenhum dos dois sistemas precisou --- ou admite --- uma solução
        elementar em forma fechada.

  \item (Modulação da velocidade) No regime de rotação $E_0 > 1$, mostre que
        $y = x'$ oscila entre $\sqrt{2(E_0 - 1)}$ (em $x \equiv \pi \bmod 2\pi$) e $\sqrt{2(E_0 + 1)}$ (em $x \equiv 0$),
        que a média temporal de $y$ ao longo de um período vale
        exatamente $\frac{2\pi}{\tau(E_0)}$, e que a razão de modulação $\sqrt{\frac{E_0 + 1}{E_0 - 1}} \to 1$ quando $E_0
        \to \infty$:
        a rotação rápida é assintoticamente uniforme.
  \item (Monotonicidade do período de rotação) Mostre que $\tau(E_0)$ é $\mathcal C^1$
        e estritamente decrescente em $\intoo1{+\infty}$ \emph{(derive sob o sinal de
        integral, com dominação em todo $\intco{1 + \delta}\infty$)}, com $\tau \to \infty$ quando $E_0 \to 1^+$ e
        $\tau
        \to 0$ quando $E_0 \to \infty$. Monte o quadro \omterm{def:b3:complete:complete}{completo} de bifurcação do
        pêndulo ao longo do eixo da energia: equilíbrios em $E_0 = -1$,
        librações com período crescendo de $2\pi$ a $\infty$ em
        $-1 < E_0 < 1$, a separatriz em $E_0 = 1$ e as rotações com período
        decrescendo de $\infty$ a $0$ além dela.
\end{enumerate}
\end{problem}
