\chapter{Probabilidade: fundamentos e a lei dos grandes
números}\label{ch:b3:probability}

O segundo ano construiu a probabilidade em espaços enumeráveis; a teoria
da \omterm{def:b3:measure:measure}{medida} agora retira toda restrição. Um \omterm{def:b3:probability:space}{espaço de probabilidade} é um
\omterm{def:b3:measure:measure}{espaço de medida} de massa total $1$, as \omterm{def:b3:probability:space}{variáveis aleatórias} são
aplicações \omterm{def:b3:lebesgue:measurable}{mensuráveis}, a \omterm{def:b3:probability:space}{esperança} é a integral de Lebesgue --- e, de
imediato, todo o arsenal analítico (\cref{ch:b3:measure,ch:b3:lebesgue,ch:b3:product}) se aplica ao acaso.
Este capítulo instala o dicionário, constrói sequências infinitas de
\omterm{def:b3:probability:space}{variáveis aleatórias} \omterm{def:b3:probability:independence}{independentes} (em $\intcc01$, a partir de dígitos
binários: o acaso está escondido dentro da \omterm{def:b3:measure:lebesgueouter}{medida de Lebesgue}),
demonstra os lemas de Borel--Cantelli e a \omterm{thm:b3:probability:zeroone}{lei zero--um} de Kolmogorov,
organiza os modos de convergência e demonstra a \omterm{def:b3:probability:space}{lei} dos grandes números
--- o teorema que faz as frequências convergirem às probabilidades e
torna a estatística possível. O problema de fim de semana dá a
demonstração de Etemadi da \omterm{def:b3:probability:space}{lei} forte em sua forma definitiva $L^1$.

\section{O dicionário}

\begin{definition}\label{def:b3:probability:space}
Um \emph{espaço de probabilidade}\index{espaço de probabilidade} é um
\omterm{def:b3:measure:measure}{espaço de medida} $(\Omega, \mathcal A, \P)$ com $\P(\Omega) =
1$; os elementos de $\mathcal A$ são \emph{eventos}, e
uma propriedade vale \emph{quase certamente} (q.c.) se seu evento tem
probabilidade $1$. Uma \emph{variável aleatória}\index{variável
aleatória} é uma aplicação \omterm{def:b3:lebesgue:measurable}{mensurável} $X \colon \Omega \to \R$ (ou em $\R^d$: um vetor
aleatório); sua \emph{lei}\index{lei de uma variável aleatória} é a
\omterm{def:b3:measure:measure}{medida} de probabilidade imagem $\P_X =
X_*\P$ em $\R$ (\cref{exo:b3:product:9}),
determinada pela \emph{função de distribuição} $F_X(t) = \P(X \leq t)$
(\cref{exo:b3:measure:3}). $X$ tem \emph{\omterm{ex:b3:lebesgue:gamma}{densidade}} $f$ se $\P_X = f\,\dd\lambda$; ela é
\emph{discreta} se $\P_X$ é uma combinação enumerável de massas de
Dirac. A \emph{esperança}\index{esperança} é
\[
\E[X] = \int_\Omega X\,\dd\P
\qquad (X \geq 0 \text{ ou } X \in L^1(\P)),
\]
e o \emph{teorema de transferência} (\cref{exo:b3:product:9}) a calcula na lei:
$\E[g(X)] = \int_\R g\,\dd\P_X$ --- $= \sum g(x_k)p_k$ no caso discreto, $= \int
g(x)f(x)\dd x$ no caso com \omterm{ex:b3:lebesgue:gamma}{densidade}: as fórmulas
do segundo ano, agora teoremas de uma única teoria. A \emph{variância} é
$\V(X) =
\E[(X - \E X)^2] = \E[X^2] - (\E X)^2$ para $X \in L^2$.
\end{definition}

\begin{example}\label{ex:b3:probability:laws}
As \omterm{def:b3:probability:space}{leis} padrão e suas transformadas notáveis: Bernoulli $\mathcal B(p)$, binomial
$\mathcal B(n, p)$, geométrica, Poisson $\mathcal P(\lambda)$ (discretas: as tabelas do segundo ano
continuam válidas); uniforme em $\intcc01$ (a própria \omterm{def:b3:measure:lebesgueouter}{medida de Lebesgue});
exponencial $\mathcal E(\lambda)$ (\omterm{ex:b3:lebesgue:gamma}{densidade} $\lambda\eu^{-\lambda x}\mathbf 1_{x>0}$); e a \emph{gaussiana}\index{lei
gaussiana} $\mathcal N(m, \sigma^2)$, de \omterm{ex:b3:lebesgue:gamma}{densidade} $\frac1{\sigma\sqrt{2\pi}}\exp\bigl(-\frac{(x -
m)^2}{2\sigma^2}\bigr)$ --- uma \omterm{ex:b3:lebesgue:gamma}{densidade} de probabilidade
pelo~\cref{pb:b3:lebesgue:1}, de média $m$ e variância $\sigma^2$ (momentos
gaussianos, \cref{exo:b3:product:10}).
\end{example}

\begin{proposition}[Markov e Chebyshev]
\label{prop:b3:probability:markov}
Para $X \geq 0$ e $a > 0$: $\P(X \geq a) \leq \frac{\E
X}{a}$; para $X \in L^2$: $\P\bigl(\abs{X - \E X} \geq
a\bigr) \leq \frac{\V(X)}{a^2}$.\index{desigualdade de
Markov}\index{desigualdade de Chebyshev}
\end{proposition}

\begin{proof}
\cref{exo:b3:lebesgue:5}(a); Chebyshev é Markov aplicado a $(X - \E X)^2$.
\end{proof}

\section{Independência}

\begin{definition}\label{def:b3:probability:independence}
Sub-$\sigma$-álgebras $\mathcal A_1, \dots, \mathcal A_n
\subseteq \mathcal A$ são
\emph{independentes}\index{independência} se $\P(A_1\cap\dots\cap A_n) = \prod\P(A_i)$ para todos $A_i \in
\mathcal A_i$;
eventos são independentes se as $\sigma$-álgebras $\{\varnothing, A_i, A_i^c, \Omega\}$ o forem;
\omterm{def:b3:probability:space}{variáveis aleatórias} $X_1, \dots, X_n$, se as $\sigma$-álgebras $\sigma(X_i) =
X_i^{-1}(\mathcal B(\R))$ o forem. Uma
família infinita é independente se toda subfamília finita o for.
\end{definition}

\begin{theorem}\label{thm:b3:probability:independence}
$X_1, \dots, X_n$ são \omterm{def:b3:probability:independence}{independentes} se, e somente se, a \omterm{def:b3:probability:space}{lei} do vetor $(X_1, \dots, X_n)$ é a \omterm{thm:b3:product:existence}{medida
produto} $\P_{X_1}\otimes\cdots\otimes\P_{X_n}$. Nesse caso, para $g_i \geq 0$ (ou tais que os produtos sejam
\omterm{def:b3:lebesgue:l1}{integráveis}):
\[
\E\Bigl[\prod_ig_i(X_i)\Bigr] = \prod_i\E[g_i(X_i)],
\]
em particular, $\E[XY] = \E X\,\E Y$ e $\V(X_1 + \dots +
X_n) = \sum\V(X_i)$ para variáveis $L^2$ \omterm{def:b3:probability:independence}{independentes}.
\end{theorem}

\begin{proof}
Se as $X_i$ são \omterm{def:b3:probability:independence}{independentes}, as duas \omterm{def:b3:measure:measure}{medidas} de probabilidade
$\P_{(X_1,\dots,X_n)}$ e $\bigotimes\P_{X_i}$ coincidem em todos os produtos $B_1\times\dots\times B_n$ de borelianos --- um
$\pi$-sistema que gera $\mathcal B(\R^n)$ (\cref{prop:b3:product:sections}(b)) ---, logo em toda
parte (\cref{thm:b3:measure:uniqueness}). Reciprocamente, uma \omterm{def:b3:probability:space}{lei} produto fatoriza todos os
eventos $\bigcap_iX_i^{-1}(B_i)$: \omterm{def:b3:probability:independence}{independência}. A fórmula da \omterm{def:b3:probability:space}{esperança} é então
Tonelli/Fubini (\cref{thm:b3:product:tonelli}) através do teorema de transferência;
$\E[XY] = \E X\E Y$ é o caso $g_i =
\mathrm{id}$, e expandir o quadrado dá a aditividade das
variâncias (os termos cruzados $\E[(X_i - \E X_i)(X_j - \E X_j)]
= 0$).
\end{proof}

\begin{theorem}[Existência de sequências independentes]
\label{thm:b3:probability:existence}
Em $\bigl(\intcc01, \mathcal L, \lambda\bigr)$ existe uma sequência $(U_n)_{n\geq1}$ de \omterm{def:b3:probability:space}{variáveis aleatórias}
\omterm{def:b3:probability:independence}{independentes}, cada uma uniforme em $\intcc01$. Consequentemente, para \omterm{def:b3:probability:space}{leis}
prescritas $(\mu_n)$ em $\R$ quaisquer, existem $(X_n)$ \omterm{def:b3:probability:independence}{independentes}
com $\P_{X_n} = \mu_n$.\index{sequências independentes, existência}
\end{theorem}

\begin{proof}
\emph{Dígitos.} Para $\omega \in \intcc01$, sejam $(b_k(\omega))$ seus dígitos binários ($\omega = \sum b_k2^{-k}$;
escolha a expansão que não termina em uma sequência de $1$ --- a
ambiguidade envolve apenas um conjunto enumerável, logo nulo). Cada
$b_k$ é \omterm{def:b3:probability:space}{variável aleatória} ($\{b_k = 1\}$ é uma reunião finita de intervalos
diádicos), e o vetor $(b_1, \dots, b_m)$ assume cada valor de $\{0,1\}^m$ em um intervalo
diádico de comprimento $2^{-m}$: os $b_k$ são Bernoulli$(\frac12)$
\omterm{def:b3:probability:independence}{independentes}.

\emph{Reagrupamento.} Separe $\N^*$ em infinitos conjuntos infinitos
disjuntos $(I_n)$ (por exemplo, por potências de primos ou por
diagonais); seja $(k^n_j)_j$ uma enumeração de $I_n$ e ponha
\[
U_n = \sum_{j\geq1} b_{k^n_j}\,2^{-j} .
\]
Cada $U_n$ é uniforme: seus dígitos binários são bits \omterm{def:b3:probability:independence}{independentes} e
honestos, de modo que $\P(U_n \in [l2^{-m}, (l+1)2^{-m})) = 2^{-m}$ para todo intervalo diádico, e os intervalos
diádicos determinam a \omterm{def:b3:probability:space}{lei} (\cref{thm:b3:measure:uniqueness}). Os $U_n$ são
\omterm{def:b3:probability:independence}{independentes}: são funções de blocos disjuntos da família \omterm{def:b3:probability:independence}{independente}
$(b_k)$ --- formalmente, os eventos $\{U_n \in D_n\}$ para $D_n$ diádicos
dependem de finitos dígitos de conjuntos disjuntos e se fatorizam; o
argumento do $\pi$-sistema eleva isso a todos os borelianos.

\emph{\omterm{def:b3:probability:space}{Leis} arbitrárias.} Seja $G_n(u) = \inf\{t : F_{\mu_n}(t)
\geq u\}$ (a \emph{função quantil} da função
de distribuição $F_{\mu_n}$); a equivalência-chave $G_n(u) \leq t
\iff u \leq F_{\mu_n}(t)$ (\omterm{def:b3:topology:continuity}{continuidade} à
direita de $F$, monotonicidade) mostra que $X_n = G_n(U_n)$ é \omterm{def:b3:lebesgue:measurable}{mensurável} com
$\P(X_n \leq t) = \P(U_n \leq F_{\mu_n}(t)) =
F_{\mu_n}(t)$: \omterm{def:b3:probability:space}{lei} $\mu_n$; a \omterm{def:b3:probability:independence}{independência} é herdada (funções de variáveis
\omterm{def:b3:probability:independence}{independentes}, \cref{exo:b3:probability:3}).
\end{proof}

\begin{example}[O problema dos aniversários, honestamente]
\label{ex:b3:probability:birthday}
Entre $n$ pessoas com aniversários \omterm{def:b3:probability:independence}{independentes} e uniformes em
$N = 365$ dias, a probabilidade de que todos os aniversários difiram é
\[
p_n = \prod_{k=1}^{n-1}\Bigl(1 - \frac kN\Bigr),
\]
por condicionamentos sucessivos (ou diretamente: os $N(N-1)\cdots(N - n + 1)$ casos
favoráveis sobre o total $N^n$, um argumento de contagem que a
fórmula do produto da \omterm{def:b3:probability:independence}{independência} torna rigoroso). Tomando logaritmos
e usando $-\ln(1 - x) = x +
O(x^2)$:
\[
\ln p_n = -\frac{n(n-1)}{2N} +
O\Bigl(\frac{n^3}{N^2}\Bigr),
\qquad\text{logo}\qquad
p_n \approx \eu^{-n^2/2N} .
\]
O ponto de virada $p_n = \frac12$ fica em $n \approx
\sqrt{2N\ln2} \approx 1.18\sqrt N$: para $N = 365$, $n = 23$ ($p_{23} = 0.4927$).
Duas morais. Primeira: as colisões entre $n$ itens em $N$ caixas
aparecem na escala $n \sim \sqrt N$, e não $n \sim N$ --- o \emph{escalamento do
aniversário}, que governa as colisões de hash e o custo $\sqrt N$ dos ataques
de aniversário em criptografia. Segunda: o cálculo é um modelo: os
$\binom n2$ eventos de colisão de pares não são \omterm{def:b3:probability:independence}{independentes} e, no entanto, a
resposta se comporta como se fossem ($\eu^{-\binom n2/N}$ é exatamente a heurística de
pares \omterm{def:b3:probability:independence}{independentes}) --- uma primeira instância da aproximação de
Poisson, tornada rigorosa no problema de fim de semana do~\cref{ch:b3:clt} (desigualdade de Le Cam).
\end{example}

\section{Borel--Cantelli e a lei zero--um}

\begin{theorem}[Borel--Cantelli]
\label{thm:b3:probability:borelcantelli}
Sejam $(A_n)$ eventos e $\limsup A_n = \bigcap_N
\bigcup_{n\geq N}A_n$ (``$A_n$ ocorre infinitas
vezes'').\index{lemas de Borel--Cantelli}
\begin{enumerate}
  \item Se $\sum\P(A_n) < \infty$, então $\P(\limsup A_n) =
        0$.
  \item Se $\sum\P(A_n) = \infty$ \emph{e os $A_n$ são \omterm{def:b3:probability:independence}{independentes}}, então $\P(\limsup A_n) = 1$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) é o~\cref{exo:b3:measure:4}. (2): para $N \leq M$, a \omterm{def:b3:probability:independence}{independência} dos
complementares (\cref{exo:b3:probability:3}) dá
\[
\P\Bigl(\bigcap_{n=N}^{M}A_n^c\Bigr) = \prod_{n=N}^M\bigl(1
- \P(A_n)\bigr) \leq
\exp\Bigl(-\sum_{n=N}^M\P(A_n)\Bigr) \xrightarrow[M \to
\infty]{} 0
\]
($1 - x \leq \eu^{-x}$; a série diverge). Logo $\P\bigl(\bigcup_{n\geq N}A_n\bigr) = 1$ para todo $N$, e a interseção
decrescente em $N$ continua tendo probabilidade $1$ (\omterm{def:b3:topology:continuity}{continuidade}
por cima, \cref{prop:b3:measure:basics}).
\end{proof}

\begin{theorem}[Lei zero--um de Kolmogorov]
\label{thm:b3:probability:zeroone}
Sejam $(X_n)$ \omterm{def:b3:probability:independence}{independentes} e $\mathcal T =
\bigcap_N\sigma(X_N, X_{N+1}, \dots)$ a \emph{$\sigma$-álgebra
caudal}\index{$\sigma$-álgebra caudal} (eventos insensíveis a
qualquer número finito dos $X_n$: convergência de $\sum X_n$, de $\frac{S_n}n$,
valores de $\limsup$, \dots). Então todo $T \in \mathcal T$ tem $\P(T) \in \{0,
1\}$.\index{lei
zero--um}
\end{theorem}

\begin{proof}
Fixe $N$. As $\sigma$-álgebras $\sigma(X_1, \dots, X_N)$ e $\sigma(X_{N+1}, \dots)$ são \omterm{def:b3:probability:independence}{independentes}:
eventos que dependem de blocos disjuntos se fatorizam nos
$\pi$-sistemas geradores (cilindros $\bigcap_{i\leq N}\{X_i \in B_i\}$ e condições finitas sobre
as variáveis posteriores, respectivamente), e Dynkin
(\cref{thm:b3:measure:dynkin}, aplicado duas vezes, um lado de cada vez) estende a
fatoração. Um evento caudal $T$ está em $\sigma(X_{N+1}, \dots)$ para todo $N$:
$T$ é \omterm{def:b3:probability:independence}{independente} de todo $\sigma(X_1, \dots, X_N)$, logo da $\sigma$-álgebra que
eles geram, $\sigma(X_1, X_2, \dots)$ (Dynkin mais uma vez: a reunião dos $\sigma(X_1,\dots,X_N)$ é um
$\pi$-sistema que a gera). Mas $T \in \sigma(X_1, X_2, \dots)$ também: $T$ é \omterm{def:b3:probability:independence}{independente}
\emph{de si mesmo}, $\P(T) = \P(T\cap T) =
\P(T)^2$: $\P(T) \in \{0, 1\}$.
\end{proof}

\section{Modos de convergência}

\begin{definition}\label{def:b3:probability:modes}
$X_n \to X$ \emph{quase certamente} se $\P(X_n \to X) = 1$; \emph{em probabilidade} se $\P(\abs{X_n - X} \geq \varepsilon)
\to 0$
para todo $\varepsilon > 0$; \emph{em $L^p$} se $\E\abs{X_n - X}^p \to 0$.\index{convergência de
variáveis aleatórias}
\end{definition}

\begin{proposition}\label{prop:b3:probability:modes}
(a) a convergência q.c.\ implica a convergência em probabilidade;
(b) a convergência $L^p$ implica a convergência em probabilidade;
(c) a convergência em probabilidade implica a convergência q.c.\
\emph{ao longo de uma subsequência};
(d) nenhuma outra implicação vale em geral.
\end{proposition}

\begin{proof}
(a) $\P(\abs{X_n - X} \geq \varepsilon) \leq
\P\bigl(\sup_{m\geq n}\abs{X_m - X} \geq \varepsilon\bigr)
\downarrow \P\bigl(\limsup\{\abs{X_m - X} \geq
\varepsilon\}\bigr) = 0$ sob convergência q.c.\ (\omterm{def:b3:topology:continuity}{continuidade} por cima; o evento
limsup exclui a convergência). (b) Markov: $\P(\abs{X_n - X} \geq \varepsilon) \leq
\varepsilon^{-p}\,\E\abs{X_n - X}^p$. (c) Tome $n_k$ com
$\P(\abs{X_{n_k} - X} \geq 2^{-k}) \leq 2^{-k}$; Borel--Cantelli (1) faz $\abs{X_{n_k} - X} < 2^{-k}$ a partir de certo índice, q.c. (d) A
máquina de escrever (\cref{exo:b3:lp:3}) em $(\intcc01, \lambda)$ converge em $L^1$ e em
probabilidade, mas em ponto algum; $n\mathbf 1_{\intoo0{1/n}} \to 0$ q.c.\ mas não em $L^1$; os
detalhes e os demais contraexemplos estão no~\cref{exo:b3:probability:6}.
\end{proof}

\section{A lei dos grandes números}

Ao longo de toda esta seção, $(X_n)$ são \omterm{def:b3:probability:independence}{independentes} de mesma \omterm{def:b3:probability:space}{lei}
(\emph{i.i.d.}), com $S_n = X_1 + \dots + X_n$.

\begin{theorem}[Lei fraca dos grandes números]
\label{thm:b3:probability:wlln}
Se $X_1 \in L^2$, com $m = \E X_1$:
\[
\P\Bigl(\Bigl|\frac{S_n}{n} - m\Bigr| \geq
\varepsilon\Bigr) \leq
\frac{\V(X_1)}{n\,\varepsilon^2}
\xrightarrow[n\to\infty]{} 0 :
\]
$\frac{S_n}n \to m$ em probabilidade (e em $L^2$).\index{lei dos grandes números
(fraca)}
\end{theorem}

\begin{proof}
$\E\frac{S_n}n = m$ e $\V\bigl(\frac{S_n}n\bigr) =
\frac{n\V(X_1)}{n^2}$ (\cref{thm:b3:probability:independence}); Chebyshev.
\end{proof}

\begin{theorem}[Lei forte dos grandes números]
\label{thm:b3:probability:slln}
Se $X_1 \in L^1$, então\index{lei dos grandes números (forte)}
\[
\frac{S_n}{n} \xrightarrow[n\to\infty]{\text{q.c.}} \E[X_1].
\]
Demonstramo-la aqui sob a hipótese mais forte $X_1 \in
L^4$; o caso geral
($L^1$: a demonstração de Etemadi) é o problema de fim de semana.
\end{theorem}

\begin{proof}[Demonstração sob $\E X_1^4 < \infty$]
Centrando ($X_i \mapsto X_i - m$), suponha $m = 0$. Expanda:
\[
\E[S_n^4] = \sum_{i,j,k,l}\E[X_iX_jX_kX_l]
= n\,\E[X_1^4] + 3n(n-1)\,\bigl(\E[X_1^2]\bigr)^2 \leq
C\,n^2 ,
\]
pois a \omterm{def:b3:probability:independence}{independência} e a centragem matam todo termo que contenha um
fator isolado ($\E[X_iX_jX_kX_l] =
\E[X_i]\E[\cdots] = 0$, a menos que os índices se emparelhem: os únicos
sobreviventes são os $n$ termos $i=j=k=l$ e os $3n(n-1)$ termos com dois
pares distintos). Markov:
\[
\P\Bigl(\Bigl|\frac{S_n}n\Bigr| \geq \varepsilon\Bigr)
= \P\bigl(S_n^4 \geq n^4\varepsilon^4\bigr)
\leq \frac{Cn^2}{n^4\varepsilon^4} =
\frac{C}{\varepsilon^4n^2},
\]
somável: Borel--Cantelli (1) dá, para cada racional $\varepsilon$, que $\abs{S_n/n} < \varepsilon$ a
partir de certo índice, q.c.; intersectando em $\varepsilon \in \Q_+^*$ (uma quantidade
enumerável de eventos de probabilidade $1$): $S_n/n \to 0$ q.c.
\end{proof}

\begin{example}[O que a lei forte compra]
\label{ex:b3:probability:sllnapps}
(a) \emph{Frequências}: para lançamentos de moeda i.i.d., a frequência
observada de caras converge q.c.\ a $p$ --- a justificativa empírica
da própria probabilidade.
(b) \emph{Monte Carlo}\index{método de Monte Carlo}: para $g \in
L^1(\intcc01)$ e
$(U_n)$ i.i.d.\ uniformes (\cref{thm:b3:probability:existence}), $\frac1n\sum_{k\leq n}g(U_k) \to \int_0^1g$ q.c.: integrais por
amostragem, em qualquer dimensão, à taxa $\sim n^{-1/2}$ \omterm{def:b3:probability:independence}{independente} da dimensão,
tornada precisa no~\cref{ch:b3:clt}.
(c) \emph{Números normais}: quase todo número real tem, em sua expansão
binária, frequência assintótica $\frac12$ de uns (aplique a \omterm{def:b3:probability:space}{lei} forte às
variáveis dígito do~\cref{thm:b3:probability:existence}) --- o teorema de Borel, uma
afirmação sobre os números de \emph{todo} dia demonstrada por \omterm{def:b3:measure:measure}{medida}; o~\cref{pb:b3:probability:1} a \omterm{def:b3:complete:complete}{completa} em todas as bases.
\end{example}

\begin{method}\label{met:b3:probability:toolkit}
A ordem de trabalho para enunciados assintóticos sobre sequências
aleatórias: (1) \emph{o evento é caudal?} Então sua probabilidade é
$0$ ou $1$ (\cref{thm:b3:probability:zeroone}), e resta apenas decidir qual. (2)
\emph{Para demonstrar enunciados q.c.}: Borel--Cantelli --- somabilidade
das probabilidades dos eventos ``ruins'', via cotas do tipo
Markov/Chebyshev sobre os momentos que existirem; a \omterm{def:b3:probability:independence}{independência} só é
necessária na direção recíproca. (3) \emph{Subsequência + sanduíche}:
demonstre a convergência ao longo de uma subsequência tratável e
controle a oscilação intermediária por monotonicidade ou desigualdades
maximais --- o esqueleto da demonstração de Etemadi. (4) Para limites em
distribuição, espere o~\cref{ch:b3:clt}.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:probability:1}
(a) Seja $X$ com função de distribuição $F$ \omterm{def:b3:topology:continuity}{contínua} e
estritamente crescente. Mostre que $F(X)$ é uniforme em $\intcc01$ e que
$G(U) \sim F$ para $U$ uniforme, $G = F^{-1}$: simulação por inversão.
(b) Calcule a função de distribuição e a \omterm{ex:b3:lebesgue:gamma}{densidade} de $X^2$ para
$X$ uniforme em $\intcc{-1}1$, e de $-\frac1\lambda\ln
U$ para $U$ uniforme em $\intoo01$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:probability:2}

P(\lambda)$ e geométrica

(b) Mostre que uma \omterm{def:b3:probability:space}{variável aleatória} positiva $T$ com $\P(T > t)

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:3}
(a) Mostre que, se $X_1, \dots, X_n$ são \omterm{def:b3:probability:independence}{independentes} e as $f_i$ são funções

\begin{exercise}[$\star\star$]\label{exo:b3:probability:4}
(a) (O macaco infinito) Uma sequência i.i.d.\ de teclas uniformes em um

(b) (Corridas) Para bits honestos i.i.d., seja $R_n$ o comprimento da

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:5}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:6}
Em $(\intcc01, \lambda)$, exiba --- com demonstrações --- \omterm{def:b3:probability:space}{variáveis aleatórias} tais que:

(b) $X_n \to 0$ q.c.\ mas em nenhum $L^p$;
(c) $X_n \to 0$ em $L^1$ mas não em $L^2$;
(d) e mostre: se $X_n \to X$ em probabilidade e $\abs{X_n}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:7}
Uma pesquisa de opinião estima uma proporção desconhecida $p$ pela
frequência empírica $\hat p_n$ de $n$ sorteios \omterm{def:b3:probability:independence}{independentes}.
(a) Chebyshev: mostre que $\P(\abs{\hat p_n - p} \geq \varepsilon)
\leq \frac1{4n\varepsilon^2}$ (use $p(1-p) \leq \frac14$).
(b) Quantos sorteios garantem um erro $\leq 3\%$ com probabilidade $\geq 95\%$ por
essa cota? (A resposta verdadeira, via o~\cref{ch:b3:clt}, é cerca de
$1070$: Chebyshev é honesto, mas grosseiro.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:probability:8}
(Bernstein) Para $f \in \mathcal C(\intcc01)$, defina o polinômio de Bernstein $B_nf(x) =
\sum_{k=0}^n\binom nkx^k(1-x)^{n-k}f\bigl(\frac kn\bigr)$.
(a) Reconheça $B_nf(x) = \E\bigl[f\bigl(\frac
{S_n}n\bigr)\bigr]$ para $S_n$ binomial $\mathcal B(n, x)$.
(b) Demonstre $B_nf \to f$ \emph{uniformemente} em $\intcc01$: separe em $\{\abs{\frac{S_n}n - x} \leq \delta\}$ e seu
complementar, usando a \omterm{def:b3:topology:continuity}{continuidade} uniforme e Chebyshev com a cota
uniforme $\V(\frac{S_n}n) \leq \frac1{4n}$.
(c) Conclua: uma segunda demonstração, probabilística, do teorema de
aproximação de Weierstrass (\cref{cor:b3:complete:weierstrass}), com a taxa explícita $\norm{B_nf - f}_\infty \leq \frac32\,\omega_f(n^{-1/2})$
para o módulo de \omterm{def:b3:topology:continuity}{continuidade} $\omega_f$ --- demonstre ao menos a forma
$O(\omega_f(n^{-1/2}))$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:probability:9}
(Colecionador de figurinhas) Figurinhas de $n$ tipos são sorteadas
uniformemente com reposição; seja $T_n$ o número de sorteios até que
todos os tipos apareçam.
(a) Escreva $T_n = \sum_{k=1}^{n}\tau_k$ com $\tau_k$ geométricas de parâmetro $\frac{n - k + 1}n$, os
$\tau_k$ \omterm{def:b3:probability:independence}{independentes}, e deduza $\E T_n = n\,H_n \sim n\ln n$ (com $H_n$ o número
harmônico) e $\V(T_n) \leq
\frac{\pi^2}6n^2$.
(b) Chebyshev: $\frac{T_n}{n\ln n} \to 1$ em probabilidade.
(c) Refine com Borel--Cantelli: mostre diretamente que $\P(T_n >
\beta n\ln n) \leq n^{1 - \beta}$ para $\beta > 1$
\emph{(cota da união sobre o evento de algum tipo faltar após $\beta n\ln n$
sorteios, usando $1 - x \leq \eu^{-x}$)} e deduza que, ao longo de $n = 2^m$, q.c.\ $T_n \leq \beta n\ln
n$ a
partir de certo índice, para todo $\beta > 2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:10}
Usando a construção por dígitos (\cref{thm:b3:probability:existence}):
(a) verifique por cálculo direto que $U = \sum b_{2k}2^{-k}$ (os dígitos de índice par de

(b) deduza uma bijeção \omterm{def:b3:lebesgue:measurable}{mensurável} a menos de conjuntos nulos entre $\intcc01$

curva de Peano (\cref{pb:b3:topology:1}), que obteve sobrejetividade, mas não
preservação de \omterm{def:b3:measure:measure}{medida} nem injetividade.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:11}
(Recordes) Sejam $(X_n)_{n\geq1}$ i.i.d.\ com função de distribuição \omterm{def:b3:topology:continuity}{contínua}, e
diga que ocorre um \emph{recorde} no instante $n$ se $X_n > \max(X_1, \dots, X_{n-1})$ (o
instante $1$ é recorde). Seja $R_n$ a indicadora de recorde.
(a) Mostre que $\P(R_n = 1) = \frac1n$ \emph{(por simetria, cada uma das $n!$
ordenações de $X_1, \dots, X_n$ é igualmente provável, e os empates têm probabilidade
$0$)}.
(b) Mostre que os $R_n$ são \emph{\omterm{def:b3:probability:independence}{independentes}} \emph{(conte as
ordenações compatíveis com posições de recorde prescritas, ou argumente
que a ordem relativa de $X_1, \dots, X_{n-1}$ é \omterm{def:b3:probability:independence}{independente} do posto de $X_n$ entre
eles)}.
(c) Deduza de Borel--Cantelli (\cref{thm:b3:probability:borelcantelli}, as duas metades) que
ocorrem infinitos recordes q.c., mas que recordes em instantes
consecutivos $n, n+1$ ocorrem infinitas vezes com probabilidade ---
decida qual! --- e calcule $\sum_n\P(R_n = 1, R_{n+1} = 1)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:12}
(A corrida de caras mais longa) Lance uma moeda honesta infinitas vezes
e seja $L_n$ o comprimento da maior corrida de caras consecutivas
entre os primeiros $n$ lançamentos.
(a) Mostre que, para todo $\varepsilon > 0$, q.c.\ $L_n \leq
(1 + \varepsilon)\log_2n$ a partir de certo índice
\emph{(a probabilidade de alguma corrida de comprimento $\ell$
começar entre os primeiros $n$ lançamentos é no máximo $n2^{-\ell}$;
Borel--Cantelli ao longo de $n =
2^k$)}.
(b) Mostre que q.c.\ $L_n \geq (1 - \varepsilon)\log_2n$ a partir de certo índice \emph{(reparta os
primeiros $n$ lançamentos em $\lfloor n/\ell\rfloor$ blocos disjuntos de comprimento
$\ell =
\lceil(1 - \varepsilon)\log_2n\rceil$; os blocos são \omterm{def:b3:probability:independence}{independentes}, cada um todo de caras com
probabilidade $2^{-\ell}$, e a probabilidade de nenhum ser todo de caras é no
máximo $\exp(-n2^{-\ell}/\ell)$; some ao longo de $n = 2^k$, de novo)}.
(c) Conclua que $\frac{L_n}{\log_2n} \to 1$ q.c.: em um milhão de lançamentos honestos deve-se
esperar uma corrida de cerca de $20$ caras --- e um conjunto de dados
sem ela provavelmente foi fabricado.
\end{exercise}

\section{Problema: a demonstração de Etemadi da lei forte}

\begin{problem}[{Problema de fim de semana --- a lei forte dos grandes
números para variáveis i.i.d.\ integráveis}]
\label{pb:b3:probability:1}
A \omterm{def:b3:probability:space}{lei} forte de Kolmogorov --- $\frac{S_n}n \to \E X_1$ q.c.\ para $X_n \in L^1$ i.i.d.\ --- teve
por muito tempo apenas demonstrações intrincadas; em 1981, N.~Etemadi
encontrou uma de economia notável, que não usa nada além deste capítulo
(e chega a enfraquecer a \omterm{def:b3:probability:independence}{independência} para \omterm{def:b3:probability:independence}{independência} duas a duas).
Vamos segui-la. Sejam $(X_n)$ \omterm{def:b3:probability:independence}{independentes} duas a duas, de mesma
\omterm{def:b3:probability:space}{lei}, \omterm{def:b3:lebesgue:l1}{integráveis}; $m = \E X_1$, $S_n = X_1 + \dots + X_n$.

\medskip
\noindent\textbf{Parte I --- Reduções.}
\begin{enumerate}
  \item Mostre que basta tratar $X_n \geq 0$ \emph{(separe $X_n = X_n^+ - X_n^-$: verifique que
        as duas metades são de novo i.i.d.\ \omterm{def:b3:lebesgue:l1}{integráveis} e \omterm{def:b3:probability:independence}{independentes}
        duas a duas)}. Suponha daqui em diante $X_n \geq 0$.
  \item (Truncamento) Sejam $Y_n = X_n\,\mathbf 1_{X_n \leq n}$ e $S_n^* = Y_1 + \dots + Y_n$. Mostre que
        \[
        \sum_{n\geq1}\P(X_n \neq Y_n) =
        \sum_{n\geq1}\P(X_1 > n) \leq \E[X_1] < \infty
        \]
        (\cref{exo:b3:product:3}) e deduza, via Borel--Cantelli, que $\frac{S_n - S_n^*}{n} \to 0$ q.c.:
        basta demonstrar $\frac{S^*_n}n \to m$ q.c.
  \item Mostre que $\E Y_n = \E\bigl[X_1\mathbf 1_{X_1\leq
        n}\bigr] \to m$ (convergência monótona), logo $\frac1n\sum_{k\leq n}\E Y_k \to m$ (Cesàro):
        basta demonstrar $\frac{S_n^* - \E S_n^*}{n} \to 0$ q.c.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A estimativa de variância.}
\begin{enumerate}[resume]
  \item Mostre que
        \[
        \V(Y_n) \leq \E[Y_n^2] = \E\bigl[X_1^2\,\mathbf
        1_{X_1 \leq n}\bigr]
        \]
        e, usando o bolo de camadas (\cref{prop:b3:product:layercake}), a cota-chave
        \[
        \sum_{n\geq1}\frac{\V(Y_n)}{n^2}
        \leq \sum_{n\geq1}\frac1{n^2}\,
        \E\bigl[X_1^2\mathbf 1_{X_1\leq n}\bigr]
        \leq C\,\E[X_1] < \infty
        \]
        \emph{(troque a soma com a \omterm{def:b3:probability:space}{esperança} --- Tonelli para séries ---
        e limite $\sum_{n \geq
        x}\frac1{n^2} \leq \frac2{\max(x,1)}$ para a estimativa interna $x^2\sum_{n\geq x}n^{-2} \leq 2x$)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Convergência ao longo de subsequências
geométricas.} Fixe $\alpha > 1$ e ponha $k_j =
\lfloor\alpha^j\rfloor$.
\begin{enumerate}[resume]
  \item Usando a \omterm{def:b3:probability:independence}{independência} duas a duas (as variâncias se somam,
        \cref{thm:b3:probability:independence} --- verifique que a aditividade das variâncias só
        precisa da \omterm{def:b3:probability:independence}{independência} duas a duas) e Chebyshev, mostre que,
        para todo $\varepsilon > 0$:
        \[
        \sum_{j\geq1}\P\Bigl(\Bigl|
        \frac{S^*_{k_j} - \E S^*_{k_j}}{k_j}\Bigr| \geq
        \varepsilon\Bigr)
        \leq
        \frac1{\varepsilon^2}\sum_{j\geq1}\frac1{k_j^2}
        \sum_{n\leq k_j}\V(Y_n)
        = \frac1{\varepsilon^2}\sum_{n\geq1}\V(Y_n)
        \sum_{j\,:\,k_j\geq n}\frac1{k_j^2} .
        \]
  \item Mostre que $\sum_{j : k_j \geq n}k_j^{-2} \leq
        \frac{C_\alpha}{n^2}$ \emph{(série geométrica; cuidado com a parte
        inteira: $k_j \geq \frac{\alpha^j}2$ para o cuidado do tipo $\alpha^j \geq 2$)} e conclua, com a
        questão 4 e Borel--Cantelli:
        \[
        \frac{S^*_{k_j} - \E S^*_{k_j}}{k_j}
        \xrightarrow[j\to\infty]{\text{q.c.}} 0,
        \qquad\text{logo}\qquad
        \frac{S^*_{k_j}}{k_j} \to m \ \text{q.c.}
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Sanduíche e conclusão.}
\begin{enumerate}[resume]
  \item Para $k_j \leq n \leq k_{j+1}$, use a monotonicidade de $S^*_n$ (as parcelas são
        não negativas!) para mostrar
        \[
        \frac{k_j}{k_{j+1}}\,\frac{S^*_{k_j}}{k_j}
        \;\leq\; \frac{S^*_n}{n} \;\leq\;
        \frac{k_{j+1}}{k_j}\,\frac{S^*_{k_{j+1}}}{k_{j+1}},
        \]
        e deduza, q.c.:
        \[
        \frac m\alpha \leq \liminf\frac{S^*_n}n \leq
        \limsup\frac{S^*_n}n \leq \alpha\,m .
        \]
  \item Faça $\alpha \downarrow 1$ ao longo de uma sequência e conclua $\frac{S_n^*}n \to m$ q.c., logo
        (Parte I) a \emph{\omterm{def:b3:probability:space}{lei} forte dos grandes números}:
        \[
        \boxed{\ \frac{S_n}{n}
        \xrightarrow[n\to\infty]{\text{q.c.}} \E[X_1].\ }
        \]
  \item Onde exatamente a \omterm{def:b3:probability:independence}{independência} duas a duas (em vez da
        \omterm{def:b3:probability:independence}{independência} plena) foi suficiente? Liste os três pontos em que
        hipóteses do tipo \omterm{def:b3:probability:independence}{independência} foram invocadas.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos.}
\begin{enumerate}[resume]
  \item (Números normais de Borel) Mostre que $\lambda$-quase todo $x \in \intcc01$ é
        \emph{normal em toda base} $b \geq 2$: cada dígito $0, \dots, b-1$ aparece com
        frequência assintótica $\frac1b$ \emph{(fixe $b$ e um dígito,
        aplique a \omterm{def:b3:probability:space}{lei} forte às variáveis indicadoras --- justifique que
        os dígitos na base $b$ de uma variável uniforme são i.i.d.\
        uniformes em $\{0,\dots,b-1\}$, como no~\cref{thm:b3:probability:existence} --- e intersecte os
        enumeráveis eventos de probabilidade um)}. Exiba um número
        explicitamente não normal e reflita: o teorema afirma a
        normalidade de quase todos os números e, no entanto, demonstrar
        a normalidade de $\sqrt2$ ou de $\pi$ continua em aberto.
  \item (\omterm{ex:b3:probability:sllnapps}{Monte Carlo}, com garantia) Justifique por completo o método
        do~\cref{ex:b3:probability:sllnapps}(b) para $g \in L^1(\intcc01^d)$: construa a amostra uniforme
        i.i.d.\ em $\intcc01^d$ a partir do~\cref{thm:b3:probability:existence} e do~\cref{exo:b3:probability:10},
        e enuncie o que a \omterm{def:b3:probability:space}{lei} forte entrega.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- O que a \omterm{def:b3:probability:independence}{independência} plena compra:
desigualdades maximais e séries aleatórias.} Etemadi gasta apenas a
\omterm{def:b3:probability:independence}{independência} duas a duas; as partes restantes exploram a versão plena
(mútua). Sejam $(Z_n)$ variáveis \omterm{def:b3:probability:independence}{independentes} centradas de $L^2$
e $S_k = Z_1 + \dots + Z_k$ (notação nova, sem relação com os $X_n$ acima).
\begin{enumerate}[resume]
  \item (Desigualdade maximal de Kolmogorov) Para $\varepsilon
        > 0$, demonstre
        \[
        \P\Bigl(\max_{1\leq k\leq n}\abs{S_k} \geq
        \varepsilon\Bigr) \;\leq\;
        \frac1{\varepsilon^2}\sum_{k=1}^n\V(Z_k) :
        \]
        o preço de Chebyshev compra o máximo \emph{(reparta o evento
        segundo o primeiro índice $k$ com $\abs{S_k} \geq \varepsilon$; nessa peça, escreva
        $S_n^2 \geq S_k^2 + 2S_k(S_n - S_k)$ e use a \omterm{def:b3:probability:independence}{independência} das coalizões $(Z_1, \dots, Z_k)$ e $(Z_{k+1}, \dots, Z_n)$,
        \cref{thm:b3:probability:independence})}. Aponte o passo em que a \omterm{def:b3:probability:independence}{independência} duas a
        duas já não bastaria.
  \item (Teorema de uma série de Khinchin--Kolmogorov) Deduza: se
        $\sum_n\V(Z_n) < \infty$, então $\sum_nZ_n$ converge quase certamente \emph{(mostre que,
        q.c., as somas parciais formam uma sequência de Cauchy: faça
        $m \to
        \infty$ na desigualdade maximal aplicada a $Z_{N+1}, \dots, Z_{N+m}$ e, depois, faça
        $N \to
        \infty$)}.
  \item (Séries de Rademacher) Sejam $(\varepsilon_n)$ sinais i.i.d., $\P(\varepsilon_n = \pm1) =
        \frac12$
        (\cref{thm:b3:probability:existence}), e sejam $(x_n)$ números reais. Mostre que
        $\sum_nx_n\varepsilon_n$ converge q.c.\ assim que $\sum_nx_n^2 < \infty$; mostre também que, quaisquer
        que sejam os $(x_n)$, a probabilidade de $\sum_nx_n\varepsilon_n$ convergir é
        $0$ ou $1$ (\cref{thm:b3:probability:zeroone}).
  \item A recíproca, de maneira elementar. Ponha $T_n = \sum_{k\leq
        n}x_k\varepsilon_k$ e $s_n^2 = \sum_{k\leq
        n}x_k^2$, e suponha
        $s_n \to \infty$.
        (a) Demonstre a \emph{desigualdade de Paley--Zygmund}: para
        $Z \geq 0$ com $\E Z^2 < \infty$ e $0 < \theta <
        1$,
        \[
        \P\bigl(Z > \theta\,\E Z\bigr) \;\geq\; (1 -
        \theta)^2\,\frac{(\E Z)^2}{\E Z^2}
        \]
        \emph{(separe $\E Z$ no nível $\theta\E Z$ e aplique
        Cauchy--Schwarz à peça superior)}.
        (b) Mostre que $\E T_n^4 \leq 3s_n^4$.
        (c) Deduza $\P\bigl(\abs{T_n} > \frac{s_n}2\bigr)
        \geq \frac3{16}$ e conclua que $\sum_nx_n\varepsilon_n$ diverge q.c.; donde a
        dicotomia
        \[
        \sum_nx_n\varepsilon_n\ \text{converge q.c.}
        \iff \sum_nx_n^2 < \infty .
        \]
  \item (Série harmônica aleatória) Conclua que $\sum_n\frac{\varepsilon_n}{n^s}$ converge q.c.\ se,
        e somente se, $s > \frac12$. Para $\frac12 < s \leq
        1$, a série converge q.c.\ enquanto
        $\sum_nn^{-s} = \infty$: sinais aleatórios produzem cancelamento de intensidade
        raiz quadrada --- compare com a série alternada $\sum_n\frac{(-1)^n}{n^s}$, que
        converge para \emph{todo} $s > 0$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- Concentração: a desigualdade de
Hoeffding.} A \omterm{def:b3:probability:space}{lei} forte diz que $\frac{S_n}n \to m$; as desigualdades de concentração
dizem quão improvável é um desvio \emph{em cada $n$ fixado}.
\begin{enumerate}[resume]
  \item (Lema de Hoeffding)
        (a) Mostre que $\cosh\lambda \leq \eu^{\lambda^2/2}$ para todo $\lambda \in \R$, comparando as duas séries
        termo a termo.
        (b) Seja $Z$ centrada com $a \leq Z \leq b$, $a < b$. Mostre que
        \[
        \E\,\eu^{\lambda Z} \leq
        \exp\Bigl(\frac{\lambda^2(b - a)^2}8\Bigr)
        \]
        \emph{(limite $\eu^{\lambda z}$ em $\intcc ab$ por sua corda, tome \omterm{def:b3:probability:space}{esperanças} e
        estude $\varphi(t)
        = -pt + \log(1 - p + p\eu^t)$ com $p =
        \frac{-a}{b-a}$ e $t = \lambda(b - a)$: mostre que $\varphi(0) = \varphi'(0) = 0$ e $\varphi''
        \leq \frac14$)}.
  \item (Desigualdade de Hoeffding) Sejam $X_1, \dots, X_n$ \omterm{def:b3:probability:independence}{independentes} com $a_i \leq X_i \leq b_i$
        e $S_n =
        X_1 + \dots + X_n$. Demonstre, para $t > 0$,
        \[
        \P\bigl(S_n - \E S_n \geq t\bigr) \leq
        \exp\Bigl(\frac{-2t^2}{\sum_{i=1}^n(b_i -
        a_i)^2}\Bigr),
        \]
        e a mesma cota para a cauda inferior \emph{(Chebyshev
        exponencial: limite $\E\,\eu^{\lambda(S_n - \E S_n)}$ usando a \omterm{def:b3:probability:independence}{independência} e a questão 17
        e, depois, otimize em $\lambda > 0$)}.
  \item (A \omterm{def:b3:probability:space}{lei} forte, caso limitado, com taxa) Sejam as $X_i$
        i.i.d.\ com valores em $\intcc ab$ e $m
        = \E X_1$. Mostre que
        \[
        \P\Bigl(\Bigl|\frac{S_n}n - m\Bigr| \geq
        \varepsilon\Bigr) \leq
        2\exp\Bigl(\frac{-2n\varepsilon^2}{(b - a)^2}\Bigr)
        \]
        e recupere $\frac{S_n}n \to m$ q.c.\ por Borel--Cantelli: uma segunda
        demonstração da \omterm{def:b3:probability:space}{lei} forte para variáveis limitadas --- sem
        truncamento, com taxa exponencial em cada $n$ finito, mas com
        parcelas limitadas e \omterm{def:b3:probability:independence}{independência} plena. Compare as hipóteses
        com as de Etemadi.
  \item (\omterm{ex:b3:probability:sllnapps}{Monte Carlo}, com garantia em $n$ fixado) Sejam $g
        \colon \intcc01^d \to \intcc01$
        \omterm{def:b3:lebesgue:measurable}{mensurável} e $(U_k)$ a amostra uniforme i.i.d.\ da questão
        11. Dado $\varepsilon, \delta > 0$, mostre que
        \[
        n \geq \frac{\log(2/\delta)}{2\varepsilon^2}
        \implies
        \P\Bigl(\Bigl|\frac1n\sum_{k=1}^ng(U_k) -
        \int g\,\dd\lambda_d\Bigr| \geq \varepsilon\Bigr)
        \leq \delta,
        \]
        e avalie o limiar para $\varepsilon =
        \delta = 10^{-2}$. A cota não envolve $d$: compare
        com a questão 11 e com as grades determinísticas.
\end{enumerate}

\medskip
\noindent\textbf{Parte VIII --- Qual é o tamanho de um passeio
aleatório? Rumo ao logaritmo iterado.} Seja $S_n = \varepsilon_1 +
\dots + \varepsilon_n$ o passeio aleatório
simples construído a partir de sinais honestos i.i.d.
\begin{enumerate}[resume]
  \item (Caudas sub-gaussianas) Mostre que $\E\,\eu^{\lambda S_n} =
        (\cosh\lambda)^n \leq \eu^{n\lambda^2/2}$ e deduza, para
        $x > 0$,
        \[
        \P(S_n \geq x) \leq \eu^{-x^2/(2n)},
        \qquad
        \P(\abs{S_n} \geq x) \leq 2\,\eu^{-x^2/(2n)} .
        \]
  \item Deduza, via Borel--Cantelli,
        \[
        \limsup_{n\to\infty}\frac{\abs{S_n}}
        {\sqrt{2n\log n}} \leq 1 \quad\text{q.c.}
        \]
        \emph{(para $\eta > 0$, some as cotas de cauda em $x =
        (1 + \eta)\sqrt{2n\log n}$ e, depois,
        intersecte em $\eta = \frac1p$)}. Em particular, o passeio vive na escala do
        teorema central do limite, $\sqrt n$, a menos de um fator logarítmico
        --- muito abaixo da cota grosseira $\abs{S_n} \leq n$.
  \item Ao longo da subsequência de dobramento $n_j = 2^j$, mostre que
        \[
        \limsup_{j\to\infty}\frac{S_{n_j}}
        {\sqrt{2n_j\log\log n_j}} \leq 1 \quad\text{q.c.},
        \]
        e reflita: a \emph{\omterm{def:b3:probability:space}{lei} do logaritmo iterado} (Khinchin;
        Hartman--Wintner para parcelas $L^2$ centradas gerais) afirma
        que
        \[
        \limsup_{n\to\infty}\frac{S_n}
        {\sqrt{2n\log\log n}} = 1 \quad\text{q.c.}
        \]
        Explique com precisão o que separa a estimativa em subsequência
        que acabamos de demonstrar da metade superior desse enunciado (é
        preciso controlar $\max_{n_j \leq n \leq
        n_{j+1}}S_n$ dentro de cada bloco, o que exige uma
        desigualdade maximal em escala \emph{exponencial}) e verifique
        quantitativamente que a desigualdade da questão 12 é fraca demais
        para esse fim. A metade inferior repousa sobre o segundo lema de
        Borel--Cantelli aplicado a blocos \omterm{def:b3:probability:independence}{independentes}; as duas metades
        são material honesto de terceiro ano para um curso dedicado de
        probabilidade.

  \item (Desvio uniforme sobre uma classe finita) Sejam $A_1,
        \dots, A_N$ eventos de
        um experimento repetível, e estime cada probabilidade por sua
        frequência empírica $\hat p_i$ em $n$ repetições i.i.d.
        Combinando a desigualdade de Hoeffding com uma cota da união,
        mostre que
        \[
        \P\Bigl(\max_{i\leq N}\,\abs{\hat p_i - \P(A_i)} >
        \varepsilon\Bigr) \;\leq\; 2N\,\eu^{-2n\varepsilon^2},
        \]
        e deduza a regra de tamanho de amostra: $n \geq
        \frac{\ln(2N/\delta)}{2\varepsilon^2}$ garante que as
        $N$ estimativas sejam simultaneamente $\varepsilon$-precisas com
        probabilidade $\geq 1 -
        \delta$. Calcule $n$ para $N = 10^6$, $\varepsilon =
        0.01$, $\delta = 0.05$: o
        preço logarítmico da uniformidade.
  \item (A janela harmônica aleatória) Combinando as duas metades da
        teoria de séries aleatórias, mostre que, para sinais i.i.d.\
        $(\varepsilon_n)$, a série $\sum_n\frac{\varepsilon_n}{n^\alpha}$ converge q.c.\ se $\alpha > \frac12$ e diverge q.c.\ se
        $\alpha \leq \frac12$; contraste com a convergência absoluta (que exige $\alpha > 1$):
        na janela $\alpha \in \intoc{\frac12}1$, a convergência é um fenômeno genuinamente
        probabilístico --- cancelamento, não tamanho.
\end{enumerate}
\end{problem}
