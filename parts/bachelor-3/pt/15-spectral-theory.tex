\chapter{Operadores compactos e o teorema espectral}
\label{ch:b3:spectral}

A diagonalização é a joia da coroa da álgebra linear em dimensão
finita: uma matriz simétrica tem uma base ortonormal de autovetores. Em
dimensão infinita isso falha, em geral, para \omterm{def:b3:spectral:selfadjoint}{operadores autoadjuntos}
limitados --- a multiplicação por $x$ em $L^2(\intcc01)$ \emph{não} tem autovalor
algum (\cref{exo:b3:spectral:6}) ---, mas sobrevive, em forma \omterm{prop:b3:galois:perfect}{perfeita}, para os
operadores que são \emph{quase de dimensão finita}: os \omterm{def:b3:spectral:compact}{compactos}. O
teorema espectral para \omterm{def:b3:spectral:compact}{operadores compactos} \omterm{def:b3:spectral:selfadjoint}{autoadjuntos} é o teorema
mais usado da análise funcional aplicada: ele diagonaliza equações
integrais, move a \omterm{thm:b3:spectral:fredholm}{alternativa de Fredholm} e (problema de fim de semana)
resolve a corda vibrante, produzindo a base de senos da análise de
Fourier a partir de pura teoria de operadores --- com o $\zeta(2) = \frac{\pi^2}6$ de Euler
caindo de uma fórmula de traço como brinde de despedida. Ao longo de
todo o capítulo, $H$ é um \omterm{def:b3:hilbert:inner}{espaço de Hilbert} sobre $\C$ (ou $\R$;
os enunciados se adaptam), e os operadores são limitados.

\section{Operadores compactos}

\begin{definition}\label{def:b3:spectral:compact}
$T \in \mathcal L(E, F)$ (com $E, F$ de Banach) é \emph{compacto}\index{operador
compacto} se a imagem $T(B)$ da bola unitária é relativamente
compacta em $F$ --- equivalentemente, se toda sequência limitada
$(x_n)$ tem uma subsequência com $(Tx_{n_k})$ convergente. Os operadores de
posto finito são compactos (conjuntos limitados em dimensão finita); a
identidade de um espaço de dimensão infinita nunca é (teorema de Riesz,
segundo ano).
\end{definition}

\begin{proposition}\label{prop:b3:spectral:ideal}
Os \omterm{def:b3:spectral:compact}{operadores compactos} $\mathcal K(E, F)$ formam um subespaço fechado de $\mathcal L(E, F)$ e um
\omterm{def:b3:rings:ideal}{ideal} bilateral: $S$ \omterm{def:b3:spectral:compact}{compacto} $\Rightarrow$ $AS$ e $SB$ \omterm{def:b3:spectral:compact}{compactos}, para
$A, B$ limitados. Além disso, em um \omterm{def:b3:hilbert:inner}{espaço de Hilbert}, todo \omterm{def:b3:spectral:compact}{operador
compacto} é limite em \omterm{def:b3:banach:operator}{norma de operadores} de posto finito.
\end{proposition}

\begin{proof}
Subespaço: claro pela caracterização por sequências. \omterm{def:b3:rings:ideal}{Ideal}: aplicações
limitadas levam sequências convergentes em convergentes e limitadas em
limitadas. Fechamento: seja $T_n \to T$ com $T_n$ \omterm{def:b3:spectral:compact}{compactos}, e $(x_k)$
limitada por $1$; uma extração diagonal torna $(T_nx_{k_j})_j$ convergente para
todo $n$; então $(Tx_{k_j})$ é de Cauchy, pois
\[
\norm{Tx_{k_j} - Tx_{k_l}} \leq 2\vertiii{T - T_n} +
\norm{T_nx_{k_j} - T_nx_{k_l}} ,
\]
escolhendo primeiro $n$ e depois os índices. Aproximação em \omterm{def:b3:hilbert:inner}{espaços
de Hilbert}: seja $T$ \omterm{def:b3:spectral:compact}{compacto}, com $K = \overline{T(B)}$ \omterm{def:b3:spectral:compact}{compacto}; dado $\varepsilon$, cubra
$K$ por um número finito de bolas $B(y_i, \varepsilon)$ e seja $P$ a projeção
ortogonal sobre $V = \operatorname{Vect}(y_1, \dots, y_m)$ (fechado: tem dimensão finita). Então $PT$ tem
posto finito e, para $\norm x \leq 1$: tomando $y_i$ com $\norm{Tx - y_i} <
\varepsilon$,
\[
\norm{Tx - PTx} \leq \norm{Tx - y_i} + \norm{P(y_i - Tx)}
\leq 2\varepsilon
\]
($y_i = Py_i$; $\vertiii P \leq 1$): $\vertiii{T - PT} \leq
2\varepsilon$.
\end{proof}

\begin{example}\label{ex:b3:spectral:examples}
(a) Operadores diagonais em $\ell^2$: $T(x_n) = (\lambda_nx_n)$ é \omterm{def:b3:spectral:compact}{compacto} se, e somente
se, $\lambda_n \to 0$ (\cref{exo:b3:spectral:2}).
(b) Operadores com núcleo em $\mathcal C(\intcc01)$: \omterm{def:b3:spectral:compact}{compactos} por Ascoli
(\cref{exo:b3:complete:7}).
(c) \emph{Operadores de Hilbert--Schmidt}\index{operador de
Hilbert--Schmidt}: para $k \in L^2(\intcc01^2)$,
\[
(T_kf)(x) = \int_0^1k(x, y)\,f(y)\,\dd y
\]
define um \omterm{def:b3:spectral:compact}{operador compacto} em $L^2(\intcc01)$ com $\vertiii{T_k} \leq \norm k_{L^2}$ (\cref{exo:b3:spectral:4}: truncar a
expansão de $k$ na base exibe $T_k$ como limite de operadores de
posto finito).
\end{example}

\section{Operadores autoadjuntos}

\begin{definition}\label{def:b3:spectral:selfadjoint}
$T \in \mathcal L(H)$ é \emph{autoadjunto}\index{operador autoadjunto} se $T = T^*$
(\cref{exo:b3:hilbert:8}), isto é, se $\langle Tx, y\rangle = \langle x, Ty\rangle$ para todos $x, y$. Então $\langle x, Tx\rangle \in \R$ para
todo $x$ (por ser igual a seu conjugado).
\end{definition}

\begin{proposition}\label{prop:b3:spectral:sanorm}
Para $T$ \omterm{def:b3:spectral:selfadjoint}{autoadjunto}:\index{raio numérico}
\[
\vertiii T = \sup_{\norm x \leq 1}\ \abs{\langle x,
Tx\rangle} .
\]
Os autovalores de $T$ são reais, e autovetores \omterm{def:b3:rings:divisibility}{associados} a
autovalores distintos são ortogonais.
\end{proposition}

\begin{proof}
Seja $M$ o supremo; $M \leq \vertiii T$ por Cauchy--Schwarz. Reciprocamente, a
identidade do tipo polarização
\[
\langle x{+}y, T(x{+}y)\rangle - \langle x{-}y,
T(x{-}y)\rangle = 4\operatorname{Re}\langle y, Tx\rangle
\]
(expanda; os termos cruzados $\langle y, Tx\rangle + \langle x,
Ty\rangle = 2\operatorname{Re}\langle y, Tx\rangle$ por \omterm{def:b3:spectral:selfadjoint}{autoadjunção}) dá, junto com a lei
do paralelogramo,
\[
4\operatorname{Re}\langle y, Tx\rangle \leq
M\bigl(\norm{x{+}y}^2 + \norm{x{-}y}^2\bigr) = 2M\bigl(\norm
x^2 + \norm y^2\bigr).
\]
Para $\norm x = 1$ com $Tx \neq 0$, tome $y = Tx/\norm{Tx}$: $4\norm{Tx} \leq 4M$. Logo $\vertiii T \leq M$. Autovalores: $Tx = \lambda x$,
$x \ne 0$ dá $\lambda\norm x^2 = \langle
x, Tx\rangle \in \R$. Ortogonalidade: $\lambda\langle x,
y\rangle = \langle Tx, y\rangle = \langle x, Ty\rangle =
\mu\langle x, y\rangle$ com $\lambda \neq \mu$ reais.
\end{proof}

\section{O teorema espectral}

\begin{lemma}[Existência de um autovalor extremo]
\label{lem:b3:spectral:existence}
Seja $T \neq 0$ \omterm{def:b3:spectral:compact}{compacto} e \omterm{def:b3:spectral:selfadjoint}{autoadjunto}. Então $\vertiii T$ ou $-\vertiii T$ é autovalor de
$T$.
\end{lemma}

\begin{proof}
Pela~\cref{prop:b3:spectral:sanorm}, tome vetores unitários $x_n$ com $\langle x_n, Tx_n\rangle \to \mu$, em que
$\abs\mu =
\vertiii T > 0$ (passe a uma subsequência para fixar o sinal). Então
\[
\norm{Tx_n - \mu x_n}^2
= \norm{Tx_n}^2 - 2\mu\langle x_n, Tx_n\rangle + \mu^2
\leq \vertiii T^2 - 2\mu\langle x_n, Tx_n\rangle + \mu^2
\longrightarrow 2\mu^2 - 2\mu\cdot\mu = 0 .
\]
Por \omterm{def:b3:topology:compact}{compacidade}, uma subsequência $Tx_{n_k} \to y$; então $\mu
x_{n_k} = Tx_{n_k} - (Tx_{n_k} - \mu x_{n_k}) \to y$, de modo que $x_{n_k} \to x = y/\mu$, um
vetor unitário, e a \omterm{def:b3:topology:continuity}{continuidade} dá $Tx = \mu x$.
\end{proof}

\begin{theorem}[Teorema espectral para operadores compactos
autoadjuntos]\label{thm:b3:spectral:spectral}
Seja $T$ um \omterm{def:b3:spectral:compact}{operador compacto} \omterm{def:b3:spectral:selfadjoint}{autoadjunto} em um \omterm{def:b3:hilbert:inner}{espaço de Hilbert}
$H$.\index{teorema espectral (compacto autoadjunto)}
\begin{enumerate}
  \item $H$ admite um sistema ortonormal $(e_n)_{n \in N}$ ($N$ finito ou
        enumerável) de autovetores de $T$, com autovalores reais não
        nulos $(\lambda_n)$, tal que
        \[
        Tx = \sum_{n\in N}\lambda_n\,\langle e_n, x\rangle\,
        e_n \qquad (x \in H),
        \]
        e $H = \ker T \,\oplus^\perp\,
        \overline{\operatorname{Vect}}(e_n : n \in N)$.
  \item Se $N$ é infinito, $\lambda_n \to 0$; para cada $\delta > 0$, apenas um número
        finito de $n$ satisfaz $\abs{\lambda_n} \geq \delta$, e cada autoespaço $\ker(T - \lambda)$, $\lambda \neq 0$,
        tem dimensão finita.
  \item Completando $(e_n)$ com uma base ortonormal de $\ker T$
        obtém-se, quando $H$ é separável, uma base ortonormal de
        $H$ formada por autovetores: $T$ está diagonalizado.
\end{enumerate}
\end{theorem}

\begin{proof}
Primeiro (2). Se infinitos autovetores ortonormais $x_k$ tivessem
$\abs{\lambda_{(k)}} \geq \delta$: $\norm{Tx_k - Tx_l}^2 =
\lambda_{(k)}^2 + \lambda_{(l)}^2 \geq 2\delta^2$ (ortogonalidade, Pitágoras): não haveria subsequência
convergente de $(Tx_k)$, contradizendo a \omterm{def:b3:topology:compact}{compacidade} de $T$ na
sequência limitada $(x_k)$. Isso limita por um número finito, para
cada $\delta$, a multiplicidade total dos autovalores fora de $\intoo{-\delta}\delta$;
a enumerabilidade e $\lambda_n \to 0$ decorrem.

(1) Seja $H_0$ o subespaço fechado gerado por \emph{todos} os
autovetores de autovalor não nulo, organizados (por (2) e por
Gram--Schmidt dentro de cada autoespaço de dimensão finita, com
ortogonalidade entre autoespaços vinda da~\cref{prop:b3:spectral:sanorm}) em um sistema
ortonormal $(e_n)$ de autovalores $\lambda_n \neq 0$. $T$ leva $H_0$ em
$H_0$ e também $H_0^\perp$ em $H_0^\perp$: para $y
\perp H_0$ e $e$ autovetor, $\langle e, Ty\rangle =
\langle Te, y\rangle = \lambda\langle e, y\rangle = 0$.
A restrição $T' = T\restriction_{H_0^\perp}$ é compacta \omterm{def:b3:spectral:selfadjoint}{autoadjunta} no \omterm{def:b3:hilbert:inner}{espaço de Hilbert} $H_0^\perp$; se
$T' \neq
0$, o~\cref{lem:b3:spectral:existence} produziria um autovetor de $T$ com autovalor
não nulo dentro de $H_0^\perp$ --- impossível, pois tais vetores vivem em
$H_0$. Logo $T' = 0$: $H_0^\perp \subseteq \ker T$. Reciprocamente, $\ker T \perp$ a todo
$e_n$ ($\langle e_n, z\rangle = \frac1{\lambda_n}\langle
Te_n, z\rangle = \frac1{\lambda_n}\langle e_n, Tz\rangle =
0$): $\ker T \subseteq H_0^\perp$, donde $\ker T =
H_0^\perp$ e a decomposição ortogonal. A
expansão: para $x = z + \sum_nc_ne_n$ ($z \in \ker T$, $c_n = \langle
e_n, x\rangle$; \cref{thm:b3:hilbert:parseval}(1) em $H_0$), a
\omterm{def:b3:topology:continuity}{continuidade} de $T$ dá $Tx = \sum_nc_n\lambda_ne_n$.

(3) $\ker T$, subespaço fechado de um espaço separável, é separável:
ele tem base ortonormal (\cref{prop:b3:hilbert:gramschmidt}); a reunião é base ortonormal
de $H$ pela decomposição de (1).
\end{proof}

\begin{theorem}[Alternativa de Fredholm]
\label{thm:b3:spectral:fredholm}
Sejam $T$ \omterm{def:b3:spectral:compact}{compacto} \omterm{def:b3:spectral:selfadjoint}{autoadjunto} e $\lambda \in \R\setminus
\{0\}$.\index{alternativa de
Fredholm}
\begin{enumerate}
  \item Se $\lambda$ não é autovalor, então $T - \lambda I$ é bijetor com inversa
        limitada: para toda $f$, a equação $Tx - \lambda x = f$ tem exatamente uma
        solução, que depende \omterm{def:b3:topology:continuity}{continuamente} de $f$.
  \item Se $\lambda$ é autovalor, $Tx - \lambda x = f$ é \omterm{def:b3:groups:derived}{solúvel} se, e somente se, $f \perp \ker(T - \lambda I)$, e
        a solução é única a menos desse núcleo (de dimensão finita).
\end{enumerate}
\end{theorem}

\begin{proof}
Decomponha $x = z + \sum c_ne_n$ e $f = w + \sum d_ne_n$ conforme o~\cref{thm:b3:spectral:spectral} ($z, w \in \ker T$). A equação se
escreve
\[
-\lambda z = w, \qquad (\lambda_n - \lambda)\,c_n = d_n\
(n \in N).
\]
(1) $\lambda \notin \{\lambda_n\}\cup\{0\}$: por (2) do teorema espectral, $\inf_n\abs{\lambda_n - \lambda} = \delta >
0$ (os autovalores só se
acumulam em $0 \neq \lambda$). Resolva: $z = -w/\lambda$, $c_n = d_n/(\lambda_n - \lambda)$, com $\sum\abs{c_n}^2 \leq \delta^{-2}\sum\abs{d_n}^2$: uma única solução,
com $\norm x \leq C\norm f$.
(2) $\lambda = \lambda_{n}$ para $n$ em um conjunto finito $F$: a \omterm{def:b3:groups:derived}{solubilidade} de
$(\lambda_n - \lambda)c_n = d_n$ para $n \in F$ exige $d_n = 0$, isto é, $f \perp e_n$ ($n \in F$), ou seja, $f
\perp \ker(T - \lambda I)$; os
$c_n$, $n \in F$, ficam então livres.
\end{proof}

\begin{example}\label{ex:b3:spectral:minxy}
Em $L^2(\intcc01)$, seja $Tf(x) = \int_0^1\min(x, y)f(y)\dd y$: um \omterm{ex:b3:spectral:examples}{operador de Hilbert--Schmidt} de núcleo real e
simétrico, portanto \omterm{def:b3:spectral:compact}{compacto} e \omterm{def:b3:spectral:selfadjoint}{autoadjunto}. Resolvendo $Tf = \lambda f$: a relação
$\bigl(Tf\bigr)(x) = \int_0^xyf(y)\dd y +
x\int_x^1f(y)\dd y$ mostra que $u = Tf$ satisfaz $u'' = -f$ (duas derivações,
legítimas para $f$ \omterm{def:b3:topology:continuity}{contínua}, e $Tf$ é \omterm{def:b3:topology:continuity}{contínua} para $f \in L^2$:
convergência dominada), com $u(0)
= 0$ e $u'(1) = 0$. Assim, as autofunções
resolvem $\lambda u'' =
-u$, $u(0) = 0$, $u'(1) = 0$:
\[
u_n(x) = \sin\Bigl(\bigl(n + \tfrac12\bigr)\pi x\Bigr),
\qquad
\lambda_n = \frac{1}{\bigl(n + \frac12\bigr)^2\pi^2}
\quad (n \geq 0),
\]
e o teorema espectral afirma --- sem nenhuma teoria de Fourier --- que
esses senos formam, após normalização, uma base ortonormal de $L^2(\intcc01)$ (o
núcleo de $T$ é $0$: $Tf = 0$ força, pelas duas derivações,
$f = 0$ q.t.p.). O problema de fim de semana percorre o mesmo
círculo de ideias para a corda vibrante e extrai $\zeta(2)$ do traço.
\end{example}

\begin{method}\label{met:b3:spectral:method}
Dada uma equação integral ou diferencial: (1) reescreva-a como $(I - \lambda K)u = f$ ou
$Ku = \lambda u$ com $K$ um operador integral; (2) verifique que $K$ é
\omterm{def:b3:spectral:compact}{compacto} (núcleo de Hilbert--Schmidt, ou Ascoli) e, se possível,
\omterm{def:b3:spectral:selfadjoint}{autoadjunto} (núcleo real simétrico); (3) diagonalize com o teorema
espectral ou invoque a \omterm{thm:b3:spectral:fredholm}{alternativa de Fredholm} para a \omterm{def:b3:groups:derived}{solubilidade}; (4)
leia existência, unicidade, estabilidade e fórmulas em série para as
soluções na base de autovetores. Os operadores diferenciais são
ilimitados, mas suas \emph{inversas} (operadores de Green) são
compactas: inverta sempre primeiro.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b3:spectral:1}
(a) Mostre que um operador limitado de imagem de dimensão finita é
\omterm{def:b3:spectral:compact}{compacto}.
(b) Mostre que a identidade de um espaço normado é compacta se, e
somente se, a dimensão é finita (Riesz, segundo ano). Deduza que um
\omterm{def:b3:spectral:compact}{operador compacto} em um espaço de dimensão infinita nunca é inversível
com inversa limitada.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:spectral:2}
Seja $T(x_1, x_2, \dots) = (\lambda_1x_1, \lambda_2x_2, \dots)$ em $\ell^2$, com $(\lambda_n)$ limitada.
(a) Mostre que $\vertiii T = \sup\abs{\lambda_n}$.
(b) Mostre que $T$ é \omterm{def:b3:spectral:compact}{compacto} se, e somente se, $\lambda_n \to 0$. \emph{(Para
$\Leftarrow$, trunque; para $\Rightarrow$, teste em $(e_n)$.)}
(c) Quando $T$ é \omterm{def:b3:spectral:selfadjoint}{autoadjunto}? Verifique o teorema espectral por
inspeção nesse caso.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spectral:3}
Dê os detalhes da propriedade de \omterm{def:b3:rings:ideal}{ideal} (\cref{prop:b3:spectral:ideal}): se $S$ é
\omterm{def:b3:spectral:compact}{compacto} e $A, B$ são limitados, então $ASB$ é \omterm{def:b3:spectral:compact}{compacto}. Deduza
que, se $ST = TS = I$ para algum $S$ limitado e $\dim H = \infty$, então $T$ não é
\omterm{def:b3:spectral:compact}{compacto} --- e reconcilie com o~\cref{exo:b3:spectral:1}(b).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spectral:4}
(Hilbert--Schmidt) Sejam $k \in L^2(\intcc01^2)$ e $(e_n)$ uma \omterm{def:b3:hilbert:onb}{base hilbertiana} de
$L^2(\intcc01)$.
(a) Mostre que $\vertiii{T_k} \leq \norm k_{L^2}$ \emph{(Cauchy--Schwarz na variável $y$ e,
depois, Tonelli)}.
(b) Expanda $k(x,y) = \sum_{m,n}c_{mn}e_m(x)\overline{e_n(y)}$ em $L^2$ do quadrado (justifique que os produtos
formam ali uma \omterm{def:b3:hilbert:onb}{base hilbertiana}) e mostre que truncar a soma dá
operadores de posto finito que convergem a $T_k$ em \omterm{def:b3:banach:operator}{norma de
operador}: $T_k$ é \omterm{def:b3:spectral:compact}{compacto}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spectral:5}
Seja $T$ \omterm{def:b3:spectral:selfadjoint}{autoadjunto} com $\langle x, Tx\rangle \geq 0$ para todo $x$ (operador
\emph{positivo}).
(a) Mostre que os autovalores são $\geq 0$ e que $\vertiii T =
\sup_{\norm x\leq1}\langle x, Tx\rangle$.
(b) Demonstre a desigualdade de Cauchy--Schwarz generalizada $\abs{\langle x, Ty\rangle}^2 \leq \langle x, Tx\rangle\langle
y, Ty\rangle$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spectral:6}
Em $L^2(\intcc01)$, seja $(Mf)(x) = x\,f(x)$.
(a) Mostre que $M$ é limitado e \omterm{def:b3:spectral:selfadjoint}{autoadjunto}, com $\vertiii M =
1$, mas
\emph{não} tem autovalor algum.
(b) Mostre que $M$ não é \omterm{def:b3:spectral:compact}{compacto} \emph{(exiba uma sequência
limitada cuja imagem não tem subsequência convergente, por exemplo
indicadoras normalizadas de intervalos que encolhem perto de $1$ ---
ou invoque o teorema espectral)}.
(c) Onde a demonstração do~\cref{lem:b3:spectral:existence} se quebra para $M$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spectral:7}
(Volterra) Em $L^2(\intcc01)$, seja $Vf(x) = \int_0^xf(y)\dd
y$.
(a) Mostre que $V$ é \omterm{def:b3:spectral:compact}{compacto} (Hilbert--Schmidt de núcleo $\mathbf
1_{y < x}$) mas
não é \omterm{def:b3:spectral:selfadjoint}{autoadjunto}; calcule $V^*$.
(b) Mostre que $V$ não tem autovalor não nulo. \emph{(De $Vf =
\lambda f$:
$f$ tem um representante \omterm{def:b3:topology:continuity}{contínuo}, depois é $\mathcal C^1$ e resolve $\lambda f' = f$,
$f(0) = 0$.)}
(c) Conclua que só a \omterm{def:b3:topology:compact}{compacidade} não fornece autovetores: a \omterm{def:b3:spectral:selfadjoint}{autoadjunção}
no~\cref{thm:b3:spectral:spectral} é essencial.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spectral:8}
(Courant--Fischer) Seja $T$ \omterm{def:b3:spectral:compact}{compacto}, \omterm{def:b3:spectral:selfadjoint}{autoadjunto} e \emph{positivo},
com autovalores $\mu_1 \geq \mu_2 \geq \dots
> 0$ (repetidos conforme a multiplicidade, com
autovetores $e_1, e_2,
\dots$). Mostre que:
\[
\mu_{k} = \max_{\substack{V \subseteq H \\ \dim V = k}}\
\min_{\substack{x \in V\\ \norm x = 1}}\ \langle x, Tx\rangle
= \min_{\substack{W \subseteq H\\ \operatorname{codim}W =
k-1}}\ \max_{\substack{x\in W\\ \norm x = 1}}\ \langle x,
Tx\rangle .
\]
\emph{(Teste $V = \operatorname{Vect}(e_1,\dots,e_k)$; para a cota superior, intersecte um $V$ qualquer
com espaços do tipo $\operatorname{Vect}(e_k, e_{k+1}, \dots)$: a contagem de dimensões força uma interseção
não nula.)} Deduza que os autovalores dependem monotonamente de $T$
($T \leq S
\Rightarrow \mu_k(T) \leq \mu_k(S)$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spectral:9}
Usando o~\cref{thm:b3:spectral:fredholm} para $Tf(x) =
\int_0^1\min(x,y)f(y)\dd y$ (\cref{ex:b3:spectral:minxy}): para quais $\lambda \in \R$ a
equação integral
\[
f(x) - \lambda\int_0^1\min(x,y)\,f(y)\,\dd y = g(x)
\]
tem solução única $f \in L^2$ para toda $g \in L^2$? O que acontece nos valores
excepcionais?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spectral:10}
Seja $S$ o deslocamento em $\ell^2$ (\cref{exo:b3:banach:1}).
(a) Mostre que $S$ não tem autovalores, ao passo que todo $\lambda$ com
$\abs\lambda < 1$ é autovalor de $S^*$ (encontre explicitamente os autovetores,
sequências geométricas).
(b) Nem $S$ nem $S^*$ é \omterm{def:b3:spectral:compact}{compacto}: verifique testando em
$(e_n)$ ao estilo do~\cref{exo:b3:spectral:2}.
(c) Comente: para operadores não \omterm{def:b3:spectral:selfadjoint}{autoadjuntos} e não \omterm{def:b3:spectral:compact}{compactos}, a
paisagem de autovalores pode ser qualquer coisa, do vazio a um disco
inteiro --- a noção que sobrevive é o \emph{espectro}, estudado em um
curso posterior.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:spectral:11}
(Raízes quadradas) Seja $T$ \omterm{def:b3:spectral:compact}{compacto}, \omterm{def:b3:spectral:selfadjoint}{autoadjunto} e positivo ($\langle x, Tx\rangle \geq 0$)
em um \omterm{def:b3:hilbert:inner}{espaço de Hilbert} $H$, com decomposição espectral $Tx = \sum_n\mu_n\langle e_n,
x\rangle e_n$
($\mu_n > 0$).
(a) Defina $Sx = \sum_n\sqrt{\mu_n}\,\langle e_n, x\rangle
e_n$; mostre que $S$ é \omterm{def:b3:spectral:compact}{compacto}, \omterm{def:b3:spectral:selfadjoint}{autoadjunto} e positivo,
com $S^2
= T$.
(b) Demonstre a unicidade: todo $R$ \omterm{def:b3:spectral:compact}{compacto} positivo \omterm{def:b3:spectral:selfadjoint}{autoadjunto}
com $R^2 = T$ preserva os autoespaços de $T$ \emph{($RT = R^3 = TR$: $R$ comuta
com $T$, logo $R(\ker(T - \mu)) \subseteq \ker(T - \mu)$)} e, em $\ker(T - \mu)$, $R$ é um operador positivo cujo
quadrado é $\mu\,\mathrm{id}$ em um espaço de dimensão finita: diagonalize-o ali e
conclua $R =
\sqrt\mu\,\mathrm{id}$ em cada autoespaço, logo $R = S$.
(c) Calcule $\sqrt G$ para o operador da corda $G$ do~\cref{pb:b3:spectral:1}: que
núcleo tem autovalores $\frac1{n\pi}$ na base de senos? (Exprima $\sqrt G$ como
limite em $L^2$ de núcleos; não se exige forma fechada.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:spectral:12}
(Decomposição em valores singulares) Seja $T \in \mathcal L(H)$ \emph{\omterm{def:b3:spectral:compact}{compacto}}, não
necessariamente \omterm{def:b3:spectral:selfadjoint}{autoadjunto}.
(a) Mostre que $T^*T$ é \omterm{def:b3:spectral:compact}{compacto}, \omterm{def:b3:spectral:selfadjoint}{autoadjunto} e positivo; seja
$(e_n)$ uma família ortonormal de autovetores com $T^*Te_n = s_n^2e_n$, $s_n > 0$ (os
\emph{valores singulares}), completada por $\ker(T^*T) = \ker T$ (demonstre essa
igualdade).
(b) Ponha $f_n = \frac{Te_n}{s_n}$; mostre que $(f_n)$ é ortonormal e estabeleça a
\emph{decomposição em valores singulares}:
\[
Tx = \sum_n s_n\,\langle e_n, x\rangle\,f_n
\qquad (x \in H),
\]
com convergência em $H$.
(c) Deduza: $\vertiii T = \max_ns_n$; $T$ é limite em \omterm{def:b3:banach:operator}{norma de operadores} de posto
finito (redemonstrando a recíproca da~\cref{prop:b3:spectral:ideal} para \omterm{def:b3:hilbert:inner}{espaços de
Hilbert}); e, para o operador de Volterra $V$ do~\cref{exo:b3:spectral:7}, que
não tem autovalores, explique por que a decomposição em valores
singulares ainda assim existe e quais são seus ingredientes
(identifique $V^*V$ como um operador de núcleo do tipo corda ---
calcular seus autovalores explicitamente é o território
do~\cref{exo:b3:spectral:9}).
\end{exercise}

\section{Problema: a corda vibrante e
\texorpdfstring{$\zeta(2)$}{zeta(2)}}

\begin{problem}[{Problema de fim de semana --- o operador de Green, a
base de senos e uma fórmula de traço}]\label{pb:b3:spectral:1}
Resolvemos o problema de autovalores da corda vibrante de extremidades
fixas --- $-u'' = \nu u$, $u(0) = u(1) = 0$ --- por teoria de operadores, obtemos a base
ortonormal de senos sem nenhum cálculo de Fourier e avaliamos $\zeta(2)$
comparando duas expressões para o \emph{traço} do operador de Green.
Defina, em $L^2(\intcc01)$,
\[
(Gf)(x) = \int_0^1 g(x,y)\,f(y)\,\dd y,
\qquad
g(x, y) = \min(x,y)\,\bigl(1 - \max(x,y)\bigr).
\]

\medskip
\noindent\textbf{Parte I --- O operador de Green.}
\begin{enumerate}
  \item Mostre que $g$ é \omterm{def:b3:topology:continuity}{contínuo} e simétrico, com $0 \leq g
        \leq \frac14$, e que
        $G$ é \omterm{def:b3:spectral:compact}{compacto} e \omterm{def:b3:spectral:selfadjoint}{autoadjunto} (\cref{ex:b3:spectral:examples}(c)).
  \item Para $f$ \omterm{def:b3:topology:continuity}{contínua}, mostre que $u = Gf$ é $\mathcal
        C^2$ com
        \[
        -u'' = f, \qquad u(0) = u(1) = 0
        \]
        \emph{(escreva $u(x) = (1-x)\int_0^xyf(y)\dd y +
        x\int_x^1(1-y)f(y)\dd y$ e derive duas vezes)}. Reciprocamente, se
        $u \in \mathcal C^2$ com $u(0) = u(1)
        = 0$, então $G(-u'') = u$: $G$ inverte o operador da
        corda.
  \item Mostre que $\ker G = \{0\}$ \emph{(se $Gf = 0$ com $f \in
        L^2$: teste contra
        $\varphi$ \omterm{def:b3:topology:continuity}{contínua}, transfira $G$ por simetria/Fubini para
        $\varphi$ e use o lema fundamental \cref{cor:b3:lp:fundlemma} --- ou
        regularize)}, e que $G$ é um operador positivo: $\langle f, Gf\rangle \geq 0$.
        \emph{(Para $f$ \omterm{def:b3:topology:continuity}{contínua}: $\langle f, Gf\rangle = \int_0^1 (u')^2$ com $u
        = Gf$, por partes;
        conclua por \omterm{ex:b3:lebesgue:gamma}{densidade}.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Diagonalização: a base de senos.}
\begin{enumerate}[resume]
  \item Mostre que as autofunções de $G$ com autovalor $\lambda \ne 0$ são, a
        menos de escalares, as soluções de $-\lambda u'' = u$, $u(0) = u(1) = 0$ \emph{(uma
        autofunção tem representante \omterm{def:b3:topology:continuity}{contínuo} --- $Gf$ é \omterm{def:b3:topology:continuity}{contínua}
        para $f \in L^2$, por quê? ---, logo é $\mathcal C^2$ por bootstrapping da
        questão 2)}.
  \item Resolva o problema de valores de contorno: os autovalores de
        $G$ são $\lambda_n = \frac1{n^2\pi^2}$ ($n \geq 1$), com autofunções normalizadas $e_n(x) =
        \sqrt2\,\sin(n\pi x)$;
        verifique a ortonormalidade por integração direta, como teste de
        sanidade.
  \item Conclua do~\cref{thm:b3:spectral:spectral} e da questão 3 que $\bigl(\sqrt2\sin(n\pi x)\bigr)_{n\geq1}$ é uma
        \emph{base} ortonormal de $L^2(\intcc01)$ --- sem Stone--Weierstrass e sem
        séries de Fourier. Expanda $f(x) = x(1-x)$ nessa base e escreva \omterm{thm:b3:hilbert:parseval}{Parseval}
        para ela.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A fórmula de traço e $\zeta(2)$.}
\begin{enumerate}[resume]
  \item Demonstre as duas identidades
        \[
        \langle e_n, Ge_n\rangle = \lambda_n
        \quad\text{e}\quad
        \sum_{n\geq1}\lambda_n = \int_0^1 g(x,x)\,\dd x .
        \]
        Para a segunda (a \emph{fórmula de traço}): expanda $g(x, \cdot)$, para
        $x$ fixado, na base $(e_n)$ --- mostre que os
        coeficientes são $\lambda_ne_n(x)$, de modo que $g(x, \cdot) = \sum_n\lambda_ne_n(x)\,e_n$ em $L^2$. Aqui os
        senos são explícitos: verifique diretamente que $\sum_n\lambda_ne_n(x)e_n(y)$ converge
        \emph{uniformemente} no quadrado (compare com $\sum
        \frac2{n^2\pi^2}$), de modo
        que sua soma é \omterm{def:b3:topology:continuity}{contínua} e, tendo as mesmas expansões $L^2$
        em $y$ para cada $x$, é igual a $g(x,y)$ em toda
        parte. Ponha $y = x$ e integre termo a termo.
  \item Calcule $\int_0^1g(x,x)\dd x = \int_0^1x(1-x)\dd x =
        \frac16$ e conclua
        \[
        \sum_{n\geq1}\frac{1}{n^2\pi^2} = \frac16,
        \qquad\text{isto é}\qquad
        \boxed{\ \zeta(2) = \frac{\pi^2}{6}\ } :
        \]
        a soma de Euler a partir de um traço de operador.
  \item Redemonstre $\zeta(2)$ de uma terceira maneira: aplique \omterm{thm:b3:hilbert:parseval}{Parseval} na
        base de senos à função constante $\mathbf 1$, calcule $\sum_{n \text{ ímpar}}\frac1{n^2}$ e conclua.
        Compare então os mecanismos: em que sentido o argumento de
        traço das questões 7--8 é ``\omterm{thm:b3:hilbert:parseval}{Parseval} aplicado ao núcleo inteiro
        de uma só vez''?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A corda vibra.}
\begin{enumerate}[resume]
  \item (Separação de variáveis, sintetizada) Para $f \in
        L^2$, defina
        \[
        u(t, x) = \sum_{n\geq1}\;c_n\,
        \cos(n\pi t)\,\sqrt2\sin(n\pi x),
        \qquad c_n = \langle e_n, f\rangle .
        \]
        Mostre que a série converge em $L^2(\intcc01)$ para cada $t$, que
        $t\mapsto u(t, \cdot)$ é \omterm{def:b3:topology:continuity}{contínua} com valores em $L^2$, e que, para $f$ no
        subespaço gerado por um número finito de $e_n$, ela resolve
        a equação da onda $\partial_t^2u =
        \partial_x^2u$ com $u(0) = f$, $\partial_tu(0) = 0$, de extremidades fixas.
        Os autovalores $n^2\pi^2$ são as frequências ao quadrado: os
        harmônicos da corda --- explique em um parágrafo a
        interpretação musical do~\cref{thm:b3:spectral:spectral}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos variacionais: o método da
potência, a estabilidade de Weyl e uma cota rigorosa para $\pi$.}
Seja $A$ um \omterm{def:b3:spectral:compact}{operador compacto} \omterm{def:b3:spectral:selfadjoint}{autoadjunto} \emph{positivo} de
autovalores $\mu_1 \geq \mu_2 \geq \cdots > 0$ e autovetores ortonormais $(u_n)$; $R(x) = \frac{\langle x,
Ax\rangle}{\norm x^2}$. As
fórmulas de mín--máx são o~\cref{exo:b3:spectral:8}; aqui as gastamos.
\begin{enumerate}[resume]
  \item (Método da potência) Para $x \neq 0$, escreva $m_p =
        \sum_n\mu_n^p\abs{\langle u_n, x\rangle}^2$. Mostre que $m_pm_{p+2} \geq m_{p+1}^2$
        (Cauchy--Schwarz), deduza a cadeia
        \[
        R(x) \;\leq\; \frac{\langle Ax, Ax\rangle}{\langle x,
        Ax\rangle} \;\leq\; R(Ax) \;\leq\; \mu_1,
        \]
        e demonstre que, se $\langle u_1, x\rangle \neq 0$, então $R(A^kx) \to \mu_1$: iterar o operador em um
        vetor genérico qualquer calcula o autovalor máximo --- o método
        da potência da análise numérica, certificado.
  \item (Estabilidade de Weyl) Para $A, B$ \omterm{def:b3:spectral:compact}{compactos} \omterm{def:b3:spectral:selfadjoint}{autoadjuntos}
        positivos, deduza do~\cref{exo:b3:spectral:8} que
        \[
        \abs{\mu_n(A) - \mu_n(B)} \;\leq\; \vertiii{A - B}
        \qquad\text{para todo } n :
        \]
        o espectro inteiro é $1$-lipschitziano na \omterm{def:b3:banach:operator}{norma de operador}
        --- os autovalores de grandes sistemas simétricos podem ser
        calculados a partir de aproximações com erro garantido.
  \item Aplique a cota de Rayleigh a $G$ com a função de teste
        $u(x) = x(1-x)$: resolva $-w'' = u$, $w(0) =
        w(1) = 0$ para obter $Gu = w = \frac{x(1-x)(1 + x -
        x^2)}{12}$, calcule
        \[
        \norm u_2^2 = \frac1{30}, \qquad
        \langle u, Gu\rangle = \frac1{12}\Bigl(\frac1{30} +
        \frac1{140}\Bigr) = \frac{17}{5040},
        \qquad R(u) = \frac{17}{168},
        \]
        e conclua a cota \emph{rigorosa} $\frac1{\pi^2}
        = \lambda_1 \geq \frac{17}{168}$, isto é, $\pi \leq
        \sqrt{168/17} < 3.1437$.
  \item Um passo da cadeia da questão 11, na mesma função de teste:
        usando $\int_0^1(x - x^2)^4\dd x =
        \frac1{630}$, calcule
        \[
        \norm{Gu}_2^2 = \frac1{144}\Bigl(\frac1{30} +
        \frac2{140} + \frac1{630}\Bigr) = \frac{31}{90720},
        \qquad
        \frac{\langle Gu, Gu\rangle}{\langle u, Gu\rangle} =
        \frac{31}{306},
        \]
        e conclua $\pi \leq \sqrt{306/31} < 3.1419$: duas integrais, quatro algarismos corretos.
        (Cada iteração adicional aproximadamente eleva a precisão ao
        quadrado: a lacuna entre autovetores $\lambda_1/\lambda_2 = 4$ comanda a
        convergência geométrica.)
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- O traço de $G^2$ e $\zeta(4)$.}
\begin{enumerate}[resume]
  \item Mostre que $\sum_n\lambda_n^2 =
        \iint_{\intcc01^2}g(x,y)^2\,\dd x\,\dd y$ \emph{(expanda $g$ na base produto $(e_m(x)e_n(y))_{m,n}$ de
        $L^2(\intcc01^2)$ --- uma \omterm{def:b3:hilbert:onb}{base hilbertiana}, cf.\ \cref{exo:b3:spectral:5} --- e aplique
        \omterm{thm:b3:hilbert:parseval}{Parseval} no quadrado; a questão 7 identifica os coeficientes)}.
  \item Calcule a integral dupla:
        \[
        \iint g^2 = 2\int_0^1(1-x)^2\Bigl(\int_0^x
        y^2\,\dd y\Bigr)\dd x
        = \frac23\int_0^1x^3(1-x)^2\,\dd x = \frac1{90} .
        \]
  \item Conclua $\zeta(4) = \dfrac{\pi^4}{90}$; explique, sem cálculo, como os traços de
        potências superiores $G^k$ produzem $\zeta(2k) \in \pi^{2k}\,\Q$ para todo $k \geq 1$, e
        por que os valores ímpares $\zeta(3),
        \zeta(5), \dots$ estão estruturalmente fora do
        alcance dessa máquina.
  \item ($\pi$ por baixo) De $\lambda_1^2 \leq
        \sum_n\lambda_n^2 = \frac1{90}$ deduza $\pi \geq
        90^{1/4} > 3.080$, e reúna com a
        questão 14 o veredito bilateral
        \[
        3.080 \;<\; \pi \;<\; 3.1419,
        \]
        obtido inteiramente da aritmética da corda vibrante. Qual lado
        converge mais rápido se usarmos traços superiores $(\operatorname{tr}G^{2k})^{-1/4k}$, e por
        quê?
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- Forçamento e ressonância.} Fixe $\nu \in \R$ e
considere a corda forçada $-u'' - \nu u =
f$, $u(0) = u(1) = 0$, com $f \in L^2$ e $c_n = \langle
e_n, f\rangle$.
\begin{enumerate}[resume]
  \item Suponha $\nu \notin \{n^2\pi^2 : n \geq 1\}$. Mostre que
        \[
        u = \sum_{n\geq1}\frac{c_n}{n^2\pi^2 - \nu}\,e_n
        \]
        converge em $L^2$ e uniformemente em $\intcc01$
        \emph{(Cauchy--Schwarz entre $(c_n)$ e as caudas de $\sum n^{-4}$,
        com $\norm{e_n}_\infty =
        \sqrt2$)}, e que ela satisfaz $u = Gf + \nu Gu$ --- a forma em
        coordenadas explícitas da \omterm{thm:b3:spectral:fredholm}{alternativa de Fredholm}
        (\cref{thm:b3:spectral:fredholm}), com unicidade.
  \item Suponha $\nu = m^2\pi^2$. Mostre que $u = Gf + \nu
        Gu$ tem solução se, e somente se,
        $c_m = 0$, única a menos de somar múltiplos de $e_m$. Leitura
        física: empurrar um balanço exatamente em sua própria
        frequência.
  \item Para $\nu < \pi^2$, mostre que o operador solução $R_\nu\colon f \mapsto u$ é limitado em
        $L^2$ com norma $\frac1{\pi^2 - \nu}$, \omterm{def:b3:spectral:compact}{compacto}, \omterm{def:b3:spectral:selfadjoint}{autoadjunto} e positivo:
        toda a análise espectral recomeça, deslocada de $\nu$.
  \item (Síntese) Compile o dicionário deste problema: autovalor $\leftrightarrow$
        frequência ao quadrado (harmônicos); traço $\leftrightarrow$ $\zeta(2)$; norma de
        Hilbert--Schmidt $\leftrightarrow$ $\zeta(4)$; \omterm{thm:b3:spectral:fredholm}{alternativa de Fredholm} $\leftrightarrow$
        ressonância; mín--máx $\leftrightarrow$ cotas variacionais ($\pi <
        3.1437$ a partir de
        um único polinômio). Um operador integral, cinco capítulos de
        análise convertidos em dinheiro.
\end{enumerate}

\medskip
\noindent\textbf{Parte VIII --- Três últimos ecos.}
\begin{enumerate}[resume]
  \item ($\zeta(6)$, de graça) A identidade de \omterm{thm:b3:hilbert:parseval}{Parseval} da questão 6 deu
        $\sum_{n\text{ ímpar}}n^{-6} =
        \frac{\pi^6}{960}$. Separe $\zeta(6)$ em $n$ ímpares e pares e conclua
        \[
        \zeta(6) = \frac{\pi^6}{945},
        \]
        sem nenhuma integral nova: a máquina da questão 17 (traços de
        $G^3$) teria produzido o mesmo valor ao custo de um núcleo
        iterado --- \omterm{thm:b3:hilbert:parseval}{Parseval} em uma única função bem escolhida é aqui a
        rota mais barata.
  \item (O estado fundamental é positivo) Seja $A$ um \omterm{def:b3:spectral:compact}{operador
        compacto} \omterm{def:b3:spectral:selfadjoint}{autoadjunto} positivo em $L^2(\intcc01)$ dado por um núcleo
        \omterm{def:b3:topology:continuity}{contínuo} simétrico $k > 0$ em $\intoo01^2$, com maior autovalor
        $\mu_1$. Mostre: (a) todo maximizador do quociente de
        Rayleigh é uma autofunção de $\mu_1$; (b) se $u$ é uma
        delas, então $\langle\abs u, A\abs u\rangle \geq \langle u,
        Au\rangle$, com desigualdade \emph{estrita} se $u$
        assume os dois sinais em conjuntos de \omterm{def:b3:measure:measure}{medida} positiva --- de
        modo que $u$ tem sinal constante q.t.p., e $u = \mu_1^{-1}Au$ nunca se
        anula em $\intoo01$; (c) $\mu_1$ é um autovalor \emph{simples}.
        Verifique cada afirmação em $G$: $e_1 = \sqrt2\sin(\pi x) > 0$, e cada $e_n$,
        $n
        \geq 2$, sendo ortogonal a $e_1$, tem de mudar de sinal (e
        muda: $n - 1$ zeros \omterm{def:b3:topology:interior}{interiores}).
  \item (Distância ao espectro e o preço da ressonância) Para $\nu \notin \{n^2\pi^2\}$,
        mostre que o operador solução $R_\nu$ da questão 19 é
        limitado, \omterm{def:b3:spectral:selfadjoint}{autoadjunto} e \omterm{def:b3:spectral:compact}{compacto}, com
        \[
        \vertiii{R_\nu} =
        \frac1{\min_{n\geq1}\,\abs{n^2\pi^2 - \nu}}
        = \frac1{\operatorname{dist}\bigl(\nu,
        \{n^2\pi^2\}\bigr)},
        \]
        sendo a norma atingida no modo mais próximo. Quantifique então o
        balanço da questão 20: forçar com $f = e_1$ em $\nu = (1 - \varepsilon)\pi^2$ produz $u =
        \frac{1}{\varepsilon\pi^2}\,e_1$,
        uma amplificação por $\frac1\varepsilon$ da resposta estática $Ge_1 =
        \frac1{\pi^2}e_1$ --- a um
        por cento abaixo da fundamental ($\varepsilon = 10^{-2}$), a corda responde cem
        vezes mais alto.
\end{enumerate}
\end{problem}
