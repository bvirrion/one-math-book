\chapter{Funções holomorfas}\label{ch:b3:holomorphic}

A derivabilidade complexa parece uma pequena variação da teoria real
--- um limite, um quociente. É, em vez disso, outro universo. Uma
função derivável \emph{uma única vez} em um aberto de $\C$ é
automaticamente infinitamente derivável, analítica, determinada em todo
um domínio por seus valores perto de um único ponto e restringida por
princípios globais rígidos (Liouville, módulo máximo). Tudo isso decorre
de um milagre, o teorema de Cauchy: a integral de uma \omterm{def:b3:holomorphic:holo}{função holomorfa}
ao longo de um \omterm{def:b3:topology:pathconnected}{caminho} fechado em um domínio estrelado se anula. Este
capítulo demonstra o milagre (argumento de Goursat, sem supor
regularidade além da derivabilidade), colhe suas consequências
clássicas e liquida um teorema que este livro usa a crédito desde o~\cref{ch:b3:galois}: todo polinômio complexo não constante tem raiz.

Ao longo de todo o capítulo, $\Omega \subseteq \C$ é aberto, e $D(a, r)$ denota o disco aberto.

\section{Derivabilidade complexa}

\begin{definition}\label{def:b3:holomorphic:holo}
$f \colon \Omega \to \C$ é \emph{holomorfa}\index{função holomorfa} em $\Omega$ se, para
todo $z_0 \in \Omega$,
\[
f'(z_0) = \lim_{h\to0}\frac{f(z_0 + h) - f(z_0)}{h}
\]
existe ($h \in \C^*$). Somas, produtos, quocientes (com denominadores não
nulos) e composições de funções holomorfas são holomorfos, com as
fórmulas usuais (as demonstrações do primeiro e do segundo ano valem
literalmente: elas só usam as operações de corpo e limites). $\mathcal H(\Omega)$ denota
o conjunto das funções holomorfas em $\Omega$.
\end{definition}

\begin{proposition}[Cauchy--Riemann]
\label{prop:b3:holomorphic:cauchyriemann}
Escreva $f(x + \iu y) = P(x,y) + \iu Q(x,y)$. Então $f$ é \omterm{def:b3:holomorphic:holo}{holomorfa} em $z_0$ se, e somente se,
$f$ é $\R$-derivável em $z_0$ (como aplicação de duas variáveis
reais) e\index{equações de Cauchy--Riemann}
\[
\frac{\partial P}{\partial x} = \frac{\partial Q}{\partial y},
\qquad
\frac{\partial P}{\partial y} = -\frac{\partial Q}{\partial x}
\qquad \text{em } z_0 ;
\]
equivalentemente, a diferencial real é a multiplicação pelo número
complexo $f'(z_0)$.
\end{proposition}

\begin{proof}
A $\C$-derivabilidade diz que $f(z_0 + h) = f(z_0) + ch +
o(\abs h)$ com $c = f'(z_0)$: uma diferencial
$\R$-linear que é a multiplicação por $c = a + \iu b$, isto é, que tem matriz
$\bigl(\begin{smallmatrix} a & -b\\ b & a
\end{smallmatrix}\bigr)$ na base $(1, \iu)$ --- exatamente as relações exibidas para as parciais.
Reciprocamente, uma tal diferencial é $\C$-linear, e as definições de
$o(\abs h)$ coincidem.
\end{proof}

\begin{example}\label{ex:b3:holomorphic:examples}
Os polinômios em $z$, as funções racionais fora de seus polos e ---
pelo teorema de derivação termo a termo para séries de potências do
segundo ano, cuja demonstração funciona identicamente sobre $\C$ ---
toda soma de uma série de potências $\sum a_n(z - a)^n$ dentro de seu disco de
convergência: \omterm{def:b3:holomorphic:holo}{holomorfos}, com derivada $\sum na_n(z -
a)^{n-1}$ (mesmo raio). Em
particular, $\exp z = \sum z^n/n!$ é inteira (\omterm{def:b3:holomorphic:holo}{holomorfa} em $\C$) com $\exp' = \exp$. Por outro
lado, $z \mapsto \bar z$, $\abs z$ e $\operatorname{Re}z$ não são \omterm{def:b3:holomorphic:holo}{holomorfas} em parte alguma
(Cauchy--Riemann falha em todo ponto): a \omterm{def:b3:holomorphic:holo}{holomorfia} é uma rigidez de
preservação de ângulos e de orientação, não uma regularidade.
\end{example}

\section{Integrais de contorno}

\begin{definition}\label{def:b3:holomorphic:contour}
Um \emph{\omterm{def:b3:topology:pathconnected}{caminho}} é uma aplicação $\gamma \colon
\intcc ab \to \C$ de classe $\mathcal C^1$ por partes; ele é
\emph{fechado} se $\gamma(a) =
\gamma(b)$. Para $f$ \omterm{def:b3:topology:continuity}{contínua} na imagem de
$\gamma$:\index{integral de contorno}
\[
\int_\gamma f(z)\,\dd z =
\int_a^b f(\gamma(t))\,\gamma'(t)\,\dd t,
\qquad
\Bigl|\int_\gamma f\,\dd z\Bigr| \leq
\sup_{\gamma}\abs f\cdot\operatorname{length}(\gamma)
\]
(a \emph{desigualdade ML}; comprimento $= \int_a^b\abs{\gamma'}$). A integral é invariante
por reparametrização $\mathcal C^1$ crescente e muda de sinal na inversão de
orientação.
\end{definition}

\begin{proposition}[Primitivas]\label{prop:b3:holomorphic:primitive}
Para $f$ \omterm{def:b3:topology:continuity}{contínua} em $\Omega$, são equivalentes: (i) $f$ tem
uma \emph{primitiva} $F \in \mathcal H(\Omega)$ ($F'
= f$); (ii) $\int_\gamma f\,\dd z = 0$ para todo \omterm{def:b3:topology:pathconnected}{caminho}
fechado $\gamma$ em $\Omega$. Nesse caso, $\int_\gamma f\,\dd z =
F(\gamma(b)) - F(\gamma(a))$ para todo \omterm{def:b3:topology:pathconnected}{caminho}.
\end{proposition}

\begin{proof}
(i)$\Rightarrow$: $\frac{\dd}{\dd t}F(\gamma(t)) =
F'(\gamma(t))\gamma'(t)$ (regra da cadeia, válida por partes), de modo que a
integral telescopa até a diferença nas extremidades; \omterm{def:b3:topology:pathconnected}{caminhos} fechados
dão $0$. (ii)$\Rightarrow$(i): fixe $z_*$ em uma \omterm{def:b3:topology:components}{componente conexa} e defina
$F(z) = \int_{\gamma_z}f\,\dd z$ ao longo de um \omterm{def:b3:topology:pathconnected}{caminho} qualquer de $z_*$ a $z$ (bem definida:
dois \omterm{def:b3:topology:pathconnected}{caminhos} diferem por um \omterm{def:b3:topology:pathconnected}{caminho} fechado); para $h$ pequeno,
tomando o segmento de $z$ a $z + h$,
\[
\frac{F(z + h) - F(z)}{h} - f(z)
= \frac1h\int_{[z, z+h]}\bigl(f(w) - f(z)\bigr)\dd w
\longrightarrow 0
\]
pela desigualdade ML e pela \omterm{def:b3:topology:continuity}{continuidade} de $f$ em $z$.
\end{proof}

\begin{definition}[Índice de rotação]\label{def:b3:holomorphic:index}
Para um \omterm{def:b3:topology:pathconnected}{caminho} fechado $\gamma$ e $z \notin
\operatorname{im}\gamma$, o \emph{índice}\index{índice
de rotação}\index{índice de um caminho} é
\[
\operatorname{Ind}_\gamma(z) = \frac1{2\iu\pi}
\int_\gamma\frac{\dd w}{w - z} .
\]
Ele é um \emph{inteiro}: pondo $\varphi(t) =
\int_a^t\frac{\gamma'(s)}{\gamma(s) - z}\dd s$, a função $(\gamma(t) - z)\eu^{-\varphi(t)}$ tem derivada nula
(por partes), logo é constante; em $t = b$, $\eu^{\varphi(b)} =
\frac{\gamma(b) - z}{\gamma(a) - z} = 1$, de modo que $\varphi(b) \in
2\iu\pi\Z$.
Como função de $z$, o índice é \omterm{def:b3:topology:continuity}{contínuo} em $\C\setminus\operatorname{im}\gamma$ (convergência
dominada), logo constante em cada \omterm{def:b3:topology:components}{componente conexa}, e vale $0$ na
componente ilimitada (ML: a integral tende a $0$ quando $z \to
\infty$). Para o
círculo $\gamma(t) = a + r\eu^{\iu t}$, $t
\in \intcc0{2\pi}$: $\operatorname{Ind}_\gamma(z) = 1$ para $z
\in D(a,r)$ (calcule em $z = a$: $\frac1{2\iu\pi}
\int_0^{2\pi}\frac{r\iu\eu^{\iu t}}{r\eu^{\iu t}}\dd t = 1$; a
constância faz o resto).
\end{definition}

\section{O teorema de Cauchy}

\begin{theorem}[Goursat]\label{thm:b3:holomorphic:goursat}
Sejam $f \in \mathcal H(\Omega)$ e $T \subseteq \Omega$ um triângulo sólido fechado. Então $\int_{\partial T}f\,\dd z = 0$ (a fronteira
percorrida uma vez, em qualquer orientação).\index{teorema de Goursat}
\end{theorem}

\begin{proof}
Seja $I(T) = \int_{\partial T}f\,\dd z$. Ligando os pontos médios dos lados, $T$ se reparte em
quatro triângulos de tamanho metade $T^{(1)},
\dots, T^{(4)}$, e as arestas internas se
cancelam aos pares: $I(T) =
\sum_iI(T^{(i)})$. Escolha entre eles um $T_1$ com $\abs{I(T_1)} \geq \frac14\abs{I(T)}$, e
itere: uma sequência encaixada $T \supseteq T_1 \supseteq T_2 \supseteq\cdots$ com
\[
\abs{I(T_n)} \geq 4^{-n}\abs{I(T)},
\qquad
\operatorname{diam}T_n = 2^{-n}\operatorname{diam}T,
\quad
\operatorname{length}(\partial T_n) =
2^{-n}\operatorname{length}(\partial T).
\]
A interseção $\bigcap T_n$ é um único ponto $z_0$ (\omterm{def:b3:topology:compact}{compactos} encaixados de
diâmetros que tendem a zero, \cref{thm:b3:topology:compactprops}(3)). Derivabilidade em
$z_0$: dado $\varepsilon$, para $n$ grande, em $T_n$,
\[
f(z) = f(z_0) + f'(z_0)(z - z_0) + R(z),
\qquad \abs{R(z)} \leq \varepsilon\abs{z - z_0} \leq
\varepsilon\operatorname{diam}T_n .
\]
A parte afim tem primitiva: sua integral no fechado $\partial T_n$ se anula
(\cref{prop:b3:holomorphic:primitive}), restando
\[
\abs{I(T_n)} = \Bigl|\int_{\partial T_n}R\Bigr|
\leq \varepsilon\operatorname{diam}(T_n)\,
\operatorname{length}(\partial T_n)
= \varepsilon\,4^{-n}\operatorname{diam}(T)
\operatorname{length}(\partial T) .
\]
Comparando com $\abs{I(T_n)} \geq 4^{-n}\abs{I(T)}$: $\abs{I(T)} \leq \varepsilon\cdot\text{const}$ para todo $\varepsilon$: $I(T) = 0$.
\end{proof}

\begin{theorem}[Teorema de Cauchy, versão estrelada]
\label{thm:b3:holomorphic:cauchy}
Seja $\Omega$ \emph{estrelado} em relação a $c$ (todo segmento
$[c, z]$, $z \in \Omega$, está em $\Omega$) --- por exemplo, convexo. Toda
$f \in \mathcal H(\Omega)$ tem primitiva em $\Omega$; consequentemente, $\int_\gamma f\,\dd z = 0$ para
\emph{todo} \omterm{def:b3:topology:pathconnected}{caminho} fechado $\gamma$ em $\Omega$.\index{teorema
de Cauchy}
\end{theorem}

\begin{proof}
Defina $F(z) = \int_{[c,z]}f\,\dd w$. Para $z, z + h \in
\Omega$ com $[z, z+h] \subseteq \Omega$ (o que vale para $h$ pequeno), o
triângulo de vértices $c, z, z+h$ está em $\Omega$ (por ser estrelado:
cada um de seus pontos está em um segmento $[c,
w]$ com $w \in [z, z+h] \subseteq \Omega$):
Goursat dá
\[
F(z + h) - F(z) = \int_{[z, z+h]}f\,\dd w,
\]
e o cálculo com quocientes de diferenças da~\cref{prop:b3:holomorphic:primitive} fornece
$F' = f$. O anulamento das integrais em \omterm{def:b3:topology:pathconnected}{caminhos} fechados decorre da
mesma proposição.
\end{proof}

\begin{theorem}[Fórmula integral de Cauchy]
\label{thm:b3:holomorphic:formula}
Sejam $f \in \mathcal H(\Omega)$, $\bar D(a, r) \subseteq
\Omega$ e $C_r$ o círculo $\partial D(a,r)$ percorrido uma vez no
sentido anti-horário. Então, para todo $z \in D(a, r)$:\index{fórmula integral de
Cauchy}
\[
f(z) = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - z}\,\dd w .
\]
\end{theorem}

\begin{proof}
Fixe $z$ e defina em $\Omega$
\[
g(w) = \begin{cases}
\dfrac{f(w) - f(z)}{w - z} & w \neq z,\\[4pt]
f'(z) & w = z :
\end{cases}
\]
$g$ é \omterm{def:b3:topology:continuity}{contínua} em $\Omega$ e \omterm{def:b3:holomorphic:holo}{holomorfa} fora de $z$. Goursat
vale para $g$ em todo triângulo $T \subseteq \Omega'$, em que $\Omega'$ é um disco
ligeiramente maior do que $\bar D(a,r)$ contido em $\Omega$: se $z \notin T$,
diretamente; se $z \in T$, reparta $T$ em pequenos triângulos que têm
$z$ por vértice mais triângulos que evitam $z$; em um triângulo
de vértice $z$, a cota ML dá $\abs{\int_{\partial T'}g} \leq \sup_{T'}\abs
g\cdot\operatorname{length} \to 0$ à \omterm{def:b3:measure:measure}{medida} que o triângulo encolhe,
e as demais peças se anulam por Goursat --- logo $\int_{\partial T}g = 0$ em todos os
casos. A demonstração do~\cref{thm:b3:holomorphic:cauchy} usou apenas essa propriedade
triangular: $g$ tem primitiva no convexo $\Omega'$, de modo que $\int_{C_r}g = 0$,
isto é,
\[
\frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - z}\dd w
= f(z)\,\frac{1}{2\iu\pi}\int_{C_r}\frac{\dd w}{w - z}
= f(z)\operatorname{Ind}_{C_r}(z) = f(z) .
\]
\end{proof}

\section{A analiticidade e sua cascata}

\begin{theorem}[Holomorfa $=$ analítica]
\label{thm:b3:holomorphic:analytic}
Sejam $f \in \mathcal H(\Omega)$ e $D(a, R) \subseteq \Omega$. Então\index{analiticidade das funções holomorfas}
\[
f(z) = \sum_{n\geq0}c_n\,(z - a)^n \quad \text{em } D(a, R),
\qquad
c_n = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{(w -
a)^{n+1}}\,\dd w \ \ (0 < r < R),
\]
sendo os coeficientes independentes de $r$. Consequentemente,
$f$ é infinitamente $\C$-derivável, $c_n = f^{(n)}(a)/n!$, e valem as
\emph{estimativas de Cauchy}\index{estimativas de Cauchy}:
\[
\abs{c_n} \;\leq\; \frac{\sup_{\abs{w - a} =
r}\abs{f(w)}}{r^{n}} .
\]
\end{theorem}

\begin{proof}
Para $\abs{z - a} < r$: expanda o núcleo de Cauchy na série geométrica
\[
\frac1{w - z} = \frac1{(w - a)\bigl(1 - \frac{z - a}{w -
a}\bigr)} = \sum_{n\geq0}\frac{(z - a)^n}{(w - a)^{n+1}},
\]
normalmente convergente em $w$ sobre $C_r$ ($\abs{\frac{z-a}{w-a}} =
\frac{\abs{z-a}}r < 1$): integre termo
a termo contra $\frac{f(w)}{2\iu\pi}$ (a convergência uniforme justifica a troca) e
aplique o~\cref{thm:b3:holomorphic:formula}. Uma série de potências é infinitamente
derivável com $c_n =
f^{(n)}(a)/n!$ (segundo ano), o que também mostra que os $c_n$
não dependem de $r$. As estimativas: limite a integral do
coeficiente por ML.
\end{proof}

\begin{example}[As singularidades ditam os raios]
\label{ex:b3:holomorphic:radius}
Por que a inocente função real $\frac1{1 + x^2}$ tem, em $x = 3$, uma série de
Taylor que converge apenas para $\abs{x - 3} <
\sqrt{10}$, se nada dá errado sobre a reta
real? Porque o teorema acima faz o raio de convergência em $a$ ser
igual à distância de $a$ ao ponto mais próximo em que a \omterm{def:b3:holomorphic:holo}{holomorfia}
falha. Aqui, $f(z) = \frac1{1 + z^2}$ é \omterm{def:b3:holomorphic:holo}{holomorfa} exatamente em $\C\setminus\{\pm\iu\}$, de modo que a
expansão em $a = 3$ converge no maior disco que evita $\pm\iu$,
de raio $\abs{3 - \iu} = \sqrt{10}$ --- e não pode convergir em um maior, pois a soma
estenderia $f$ \omterm{def:b3:holomorphic:holo}{holomorficamente} a uma \omterm{def:b3:topology:topology}{vizinhança} de $\pm\iu$,
onde $\abs f \to \infty$. A teoria real vê o misterioso raio $\sqrt{10}$; o plano complexo
vê dois polos. Eis a regra prática: \emph{para achar um raio de
convergência, localize as singularidades} --- por exemplo, a série de
Taylor de $\tan$ em $0$ tem raio $\frac\pi2$ (os zeros mais próximos
de $\cos$), e a função geradora de Bernoulli $\frac z{\eu^z - 1}$
(\cref{pb:b3:holomorphic:1}, Parte VI) tem raio $2\pi$ (os zeros não nulos mais
próximos de $\eu^z - 1$: $\pm2\iu\pi$).
\end{example}

\begin{corollary}[Liouville; d'Alembert--Gauss]
\label{cor:b3:holomorphic:liouville}
Uma função inteira limitada é constante. Consequentemente, todo
polinômio não constante sobre $\C$ tem raiz: \emph{$\C$ é
\omterm{def:b3:galois:closure}{algebricamente fechado}}.\index{teorema de Liouville}\index{teorema
fundamental da álgebra}
\end{corollary}

\begin{proof}
Se $\abs f \leq M$ em $\C$: para todos $a$ e $r$, $\abs{c_1(a)}
= \abs{f'(a)} \leq M/r \to 0$: $f' \equiv 0$, e
$f$ é constante (no \omterm{def:b3:topology:connected}{conexo} $\C$: derivada nula implica localmente
constante --- integre ao longo de segmentos). Se $P$ não tivesse
raiz, $1/P$ seria inteira e limitada ($\abs{P(z)} \to
\infty$ quando $\abs z \to \infty$: o termo
líder domina, de modo que $\abs{1/P}$ é pequeno fora de um disco grande, e a
função é \omterm{def:b3:topology:continuity}{contínua} no disco \omterm{def:b3:topology:compact}{compacto}): seria constante --- absurdo para
$P$ não constante. (O problema de fim de semana dá uma segunda
demonstração, elementar, e as consequências \omterm{def:b3:galois:algebraic}{algébricas}.)
\end{proof}

\begin{theorem}[Os zeros são isolados; teorema da identidade]
\label{thm:b3:holomorphic:identity}
Sejam $\Omega$ \emph{\omterm{def:b3:topology:connected}{conexo}} e $f \in \mathcal
H(\Omega)$, $f \not\equiv 0$. Então todo zero $a$
de $f$ tem \emph{ordem} finita: $f(z) = (z - a)^m\,g(z)$ com $g \in
\mathcal H(\Omega)$, $g(a) \neq 0$, e os zeros
de $f$ não têm ponto de acumulação em $\Omega$. Consequentemente,
se duas \omterm{def:b3:holomorphic:holo}{funções holomorfas} em $\Omega$ coincidem em um conjunto com
ponto de acumulação em $\Omega$, elas coincidem em toda
parte.\index{teorema da identidade}\index{zeros isolados}
\end{theorem}

\begin{proof}
Seja $Z$ o conjunto dos pontos em que todas as derivadas de $f$
se anulam. $Z$ é fechado (interseção de fechados) e aberto: se todas
as $c_n = 0$ se anulam em $a$, a expansão em série de potências faz
$f
\equiv 0$ em um disco em torno de $a$. \omterm{def:b3:topology:connected}{Conexidade}: $Z =
\varnothing$ ou $Z = \Omega$; o
segundo caso está excluído por $f
\not\equiv 0$. Assim, em um zero $a$, algum
coeficiente é não nulo: seja $m$ mínimo com $c_m \neq 0$; então $f(z) = (z -
a)^m\sum_{n\geq m}c_n(z-a)^{n-m}$ em um
disco, e a soma define $g$ \omterm{def:b3:holomorphic:holo}{holomorfa} perto de $a$ com $g(a) = c_m \neq 0$;
estenda $g = f/(z-a)^m$ fora de $a$ (ali é \omterm{def:b3:holomorphic:holo}{holomorfa}). Como $g(a) \neq 0$ e $g$ é
\omterm{def:b3:topology:continuity}{contínua}, $f$ não tem outro zero em uma \omterm{def:b3:topology:topology}{vizinhança} de $a$: os
zeros são isolados, e um conjunto de pontos isolados não tem ponto de
acumulação em $\Omega$ (um ponto de acumulação de zeros é zero ---
\omterm{def:b3:topology:continuity}{continuidade} --- e não seria isolado). Identidade: aplique à diferença,
cujo conjunto de zeros tem ponto de acumulação, o que a força no ramo
$Z = \Omega$.
\end{proof}

\begin{theorem}[Valor médio e módulo máximo]
\label{thm:b3:holomorphic:maximum}
Seja $f \in \mathcal H(\Omega)$.
\begin{enumerate}
  \item (Valor médio) Para $\bar D(a, r) \subseteq \Omega$: $f(a) = \frac1{2\pi}\int_0^{2\pi}f(a +
        r\eu^{\iu t})\,\dd t$.
  \item (Princípio do máximo) Se $\Omega$ é \omterm{def:b3:topology:connected}{conexo} e $\abs f$
        atinge um máximo local em algum ponto de $\Omega$, então
        $f$ é constante. Consequentemente, para $\Omega$ limitado
        e $f$ \omterm{def:b3:topology:continuity}{contínua} em $\bar\Omega$: $\sup_{\bar\Omega}\abs f =
        \sup_{\partial\Omega}\abs f$.\index{princípio do máximo}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) é a fórmula de Cauchy no \omterm{ex:b3:groups:actions}{centro}: parametrize $C_r$.
(2) Digamos $\abs f \leq \abs{f(a)}$ em $\bar D(a, \rho)
\subseteq \Omega$. Se $f(a) = 0$, $f \equiv 0$ perto de $a$. Caso
contrário, para $0 < r \leq \rho$, o valor médio dá
\[
\abs{f(a)} \leq \frac1{2\pi}\int_0^{2\pi}\abs{f(a +
r\eu^{\iu t})}\,\dd t \leq \abs{f(a)} :
\]
o integrando \omterm{def:b3:topology:continuity}{contínuo} e não negativo $\abs{f(a)} - \abs{f(a +
r\eu^{\iu t})}$ tem média nula, logo se
anula: $\abs f$ é \emph{constante} $= \abs{f(a)} \ne 0$ no disco. Uma \omterm{def:b3:holomorphic:holo}{função
holomorfa} de módulo constante não nulo em um disco é constante: derivar
$P^2 + Q^2 = \text{const}$ dá $PP_x + QQ_x
= 0$ e $PP_y + QQ_y = 0$; substituindo as relações de Cauchy--Riemann
$P_y = -Q_x$, $Q_y = P_x$ na segunda equação, obtém-se o sistema linear
\[
P\,P_x + Q\,Q_x = 0, \qquad -P\,Q_x + Q\,P_x = 0,
\]
cujo determinante é $P^2 + Q^2 \neq 0$: $P_x = Q_x = 0$, de modo que $f' = P_x + \iu Q_x = 0$ no disco: $f$ é
ali constante. O teorema da identidade espalha a constância a todo
$\Omega$. A forma com fronteira: $\abs f$ atinge seu supremo no
\omterm{def:b3:topology:compact}{compacto} $\bar\Omega$; um máximo \omterm{def:b3:topology:interior}{interior} torna $f$ constante, e o supremo
é atingido na fronteira em todos os casos.
\end{proof}

\begin{theorem}[Teorema de convergência de Weierstrass]
\label{thm:b3:holomorphic:weierstrassconv}
Se $f_n \in \mathcal H(\Omega)$ convergem para $f$ uniformemente em todo \omterm{def:b3:topology:compact}{compacto} de
$\Omega$, então $f \in \mathcal
H(\Omega)$ e $f_n^{(k)} \to f^{(k)}$ uniformemente nos \omterm{def:b3:topology:compact}{compactos}, para todo
$k$.\index{teorema de convergência de Weierstrass}
\end{theorem}

\begin{proof}
$f$ é \omterm{def:b3:topology:continuity}{contínua}. Para todo triângulo fechado $T \subseteq
\Omega$: $\int_{\partial T}f = \lim\int_{\partial T}f_n = 0$
(convergência uniforme no \omterm{def:b3:topology:compact}{compacto} $\partial T$; Goursat para $f_n$). Pelo
argumento do~\cref{thm:b3:holomorphic:cauchy}, $f$ tem primitivas locais (os discos
são convexos; só a propriedade triangular foi usada), isto é,
$f = F'$ localmente com $F$ \omterm{def:b3:holomorphic:holo}{holomorfa}; $F$ é analítica
(\cref{thm:b3:holomorphic:analytic}), logo $f = F'$ também: é \omterm{def:b3:holomorphic:holo}{holomorfa}. (Esse é o
\emph{teorema de Morera}\index{teorema de Morera}: \omterm{def:b3:topology:continuity}{contínua} com
integrais triangulares nulas implica \omterm{def:b3:holomorphic:holo}{holomorfa}.) Derivadas: para $\bar D(a, 2r) \subseteq \Omega$ e
$z \in \bar D(a, r)$, a fórmula de Cauchy para as derivadas (derive
o~\cref{thm:b3:holomorphic:formula} sob a integral, ou use a fórmula dos coeficientes) dá
\[
\abs{f_n'(z) - f'(z)} =
\Bigl|\frac{1}{2\iu\pi}\int_{C_{2r}}\frac{f_n(w) - f(w)}{(w -
z)^2}\,\dd w\Bigr|
\leq \frac{2r\,\sup_{C_{2r}}\abs{f_n - f}}{r^2} \to 0
\]
uniformemente em $\bar D(a,r)$; cubra um \omterm{def:b3:topology:compact}{compacto} por um número finito de tais
discos e itere para $k$ maiores.
\end{proof}

\begin{method}\label{met:b3:holomorphic:toolkit}
A caixa de ferramentas do dia a dia. Para demonstrar que uma função é
\omterm{def:b3:holomorphic:holo}{holomorfa}: exiba-a como série de potências, como composição, como
limite localmente uniforme (\cref{thm:b3:holomorphic:weierstrassconv}) ou como integral com
parâmetro \omterm{def:b3:holomorphic:holo}{holomorfo} (\cref{exo:b3:holomorphic:7} --- derive sob $\int$ ou aplique
Morera--Fubini). Para demonstrar identidades: demonstre-as em um
segmento ou subdomínio e invoque o teorema da identidade. Para limitar:
\omterm{thm:b3:holomorphic:analytic}{estimativas de Cauchy} no maior círculo disponível. Para demonstrar
constância ou inexistência: Liouville ou o princípio do máximo. Saiba
sempre \emph{onde} sua função é \omterm{def:b3:holomorphic:holo}{holomorfa} e \emph{quais} discos cabem em
$\Omega$.
\end{method}

\begin{exercise}[$\star$]\label{exo:b3:holomorphic:2}
Calcule a partir das definições: $\int_{C}z^n\,\dd z$ para todo $n
\in \Z$, com $C$ o
círculo unitário; $\int_\gamma\bar z\,\dd z$ ao longo do segmento $[0, 1+\iu]$ e ao longo do \omterm{def:b3:topology:pathconnected}{caminho}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:3}
(a) Mostre que o logaritmo principal $\log z = \ln\abs z +

(b) Mostre que não existe logaritmo \omterm{def:b3:topology:continuity}{contínuo} em $\C^*$ \emph{(a

(c) Expanda $\log(1 + z)$ em série de potências em $D(0,1)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:4}
(a) Seja $f$ inteira com $\abs{f(z)} \leq C(1 + \abs
z)^{n}$. Mostre que $f$ é um polinômio de
grau $\leq n$ \emph{(\omterm{thm:b3:holomorphic:analytic}{estimativas de Cauchy} em círculos grandes)}.
(b) Seja $f$ inteira com $\operatorname{Re}f$ limitada superiormente. Mostre que
$f$ é constante \emph{(considere $\eu^{f}$)}.
(c) Deduza o ``pequeno Picard para aplicações afins'': uma função
inteira que omite um semiplano é constante.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:5}
(a) Seja $f$ \omterm{def:b3:holomorphic:holo}{holomorfa} em um \omterm{def:b3:topology:connected}{conexo} $\Omega \ni 0$ com $f(\frac1n) = \frac1{n^2}$ para todo $n$
grande. Determine $f$.
(b) Existe alguma $f$ \omterm{def:b3:holomorphic:holo}{holomorfa} em $\C^*$ com $f(\frac1n) =

$n \geq 1$? Justifique.
(c) Exiba duas \omterm{def:b3:holomorphic:holo}{funções holomorfas} distintas em $D(0,1)\sqcup D(3,1)$ que coincidem em
$D(0,1)$: onde o teorema da identidade usa a \omterm{def:b3:topology:connected}{conexidade}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:6}
Seja $f$ \omterm{def:b3:holomorphic:holo}{holomorfa} no disco unitário aberto $\mathbb D$, \omterm{def:b3:topology:continuity}{contínua} em

(a) Se $f$ não tem zero em $\mathbb D$, mostre que $f$ é constante
\emph{(aplique o princípio do máximo a $f$ e a $1/f$)}.
(b) Dê um exemplo com um zero em que $f$ não é constante.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:7}
(\omterm{def:b3:holomorphic:holo}{Holomorfia} sob a integral) Sejam $\mu$ uma \omterm{def:b3:measure:measure}{medida} finita em um
espaço $X$ e $g \colon X\times\Omega \to \C$ com: $g(x, \cdot) \in \mathcal H(\Omega)$ para cada $x$, $g$ \omterm{def:b3:lebesgue:measurable}{mensurável}
em $x$, e $\abs g \leq h(x)$ com $h$ \omterm{def:b3:lebesgue:l1}{integrável}, localmente uniformemente em
$z$. Mostre que $G(z) = \int_Xg(x, z)\dd\mu(x)$ é \omterm{def:b3:holomorphic:holo}{holomorfa} em $\Omega$. \emph{(Morera: as
integrais triangulares se anulam por Fubini e Goursat; a \omterm{def:b3:topology:continuity}{continuidade},
por convergência dominada. Aplique em seguida a $\Gamma(z) =
\int_0^\infty t^{z-1}\eu^{-t}\dd t$ em $\{\operatorname{Re}z >
0\}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:8}
(Gauss--Lucas) Seja $P \in \C[X]$ não constante. Mostre que toda raiz de
$P'$ está na envoltória convexa das raízes de $P$. \emph{(Escreva
$\frac{P'}{P} = \sum_k\frac{m_k}{z - a_k}$ em uma raiz $z$ de $P'$ que não seja raiz de $P$, tome
conjugados e leia uma combinação convexa.)} Ilustre em $P = z^3 - 1$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:9}
Seja $f$ inteira e duplamente periódica: $f(z + 1) = f(z + \iu)
= f(z)$ para todo $z$.
Mostre que $f$ é constante. \emph{(Limite $f$ no quadrado
fundamental \omterm{def:b3:topology:compact}{compacto} e, depois, em toda parte; Liouville.)} Moral: as
funções elípticas não constantes têm de ter polos --- o tema do~\cref{ch:b3:residues}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:10}
(a) Mostre que $P = \operatorname{Re}f$, para $f$ \omterm{def:b3:holomorphic:holo}{holomorfa}, satisfaz a propriedade do
valor médio $P(a) =
\frac1{2\pi}\int_0^{2\pi}P(a + r\eu^{\iu t})\dd t$ e é \emph{harmônica}: $\partial^2_{xx}P + \partial^2_{yy}P = 0$ (derive Cauchy--Riemann;
use o~\cref{thm:b3:holomorphic:analytic} para a regularidade necessária).
(b) Deduza o princípio do máximo para partes reais de \omterm{def:b3:holomorphic:holo}{funções
holomorfas} em domínios limitados.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:11}
(Reflexão de Schwarz) Sejam $\Omega^+ = \{z : \abs z < 1,\
\operatorname{Im}z > 0\}$, $I = \intoo{-1}1$ e $f$ \omterm{def:b3:holomorphic:holo}{holomorfa} em $\Omega^+$,
\omterm{def:b3:topology:continuity}{contínua} em $\Omega^+\cup I$ e \emph{a valores reais em $I$}. Defina
\[
F(z) = \begin{cases} f(z) & z \in \Omega^+\cup I,\\
\overline{f(\bar z)} & \bar z \in \Omega^+ . \end{cases}
\]
(a) Mostre que $F$ está bem definida e é \omterm{def:b3:topology:continuity}{contínua} em $\Omega = \Omega^+\cup I\cup\Omega^-$ e
\omterm{def:b3:holomorphic:holo}{holomorfa} em $\Omega^\pm$ \emph{(para $\Omega^-$: verifique Cauchy--Riemann para
$\overline{f(\bar z)}$, ou expanda $f$ em séries de potências locais e conjugue os
coeficientes)}.
(b) Mostre que $F$ é \omterm{def:b3:holomorphic:holo}{holomorfa} em todo $\Omega$ pelo critério de
Morera: $\int_{\partial T}F = 0$ para todo triângulo $T \subseteq \Omega$ \emph{(reparta os triângulos em
$I$ e empurre seus lados horizontais para fora do eixo por $\varepsilon$,
usando a \omterm{def:b3:topology:continuity}{continuidade} uniforme)}.
(c) Deduza: uma \omterm{def:b3:holomorphic:holo}{função holomorfa} no disco, real sobre um diâmetro,
satisfaz $f(\bar z) = \overline{f(z)}$; e uma \omterm{def:b3:holomorphic:holo}{função holomorfa} não constante não pode ser a
valores reais em nenhum aberto não vazio de seu domínio (\omterm{def:b3:topology:connected}{conexo}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:12}
(A equação pitagórica complexa) Encontre todos os pares de funções
\emph{inteiras} com $f^2 + g^2 = 1$.
(a) Mostre que $h = f + \iu g$ é inteira e sem zeros, e que toda função inteira
sem zeros é $\eu^{\varphi}$ para alguma $\varphi$ inteira \emph{($h'/h$ é inteira,
logo tem primitiva no estrelado $\C$; ajuste a constante e mostre
que $h\eu^{-\varphi}$ é constante)}.
(b) Conclua $f = \cos\varphi$, $g = \sin\varphi$ com $\varphi$ inteiras, e verifique a recíproca.
Quais são as soluções inteiras de $f^2 + g^2 = 0$?
\end{exercise}

\section{Problema: o teorema fundamental da álgebra, duas vezes}

\begin{problem}[{Problema de fim de semana --- $\C$ é
algebricamente fechado: a demonstração de d'Alembert, a de Liouville e
a colheita}]\label{pb:b3:holomorphic:1}
Sejam $P(z) = z^n + a_{n-1}z^{n-1} + \dots + a_0$, $n \geq 1$. Demonstramos duas vezes que $P$ tem raiz e, em
seguida, recolhemos o que a álgebra espera desde o~\cref{ch:b3:galois}.

\medskip
\noindent\textbf{Parte I --- Coercividade e o mínimo.}
\begin{enumerate}
  \item Mostre que $\abs{P(z)} \to +\infty$ quando $\abs z \to
        \infty$: mais precisamente, $\abs{P(z)} \geq \frac12\abs z^n$ para
        $\abs z \geq R_0$ adequado.
  \item Deduza que $\abs P$ atinge um mínimo global em $\C$:
        existe $z_0$ com $\abs{P(z_0)} = \inf_\C\abs P$ \emph{(\omterm{def:b3:topology:compact}{compacidade} de um disco fechado
        grande, \cref{cor:b3:topology:heineborel})}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A descida de d'Alembert.} Suponha, por
absurdo, $P(z_0) \neq 0$.
\begin{enumerate}[resume]
  \item Expanda $Q(h) = P(z_0 + h)/P(z_0)$ como polinômio em $h$: $Q(h) = 1 + c_kh^k + h^{k+1}S(h)$ com $c_k \neq
        0$, $k \geq 1$ e
        $S$ polinômio.
  \item Escolha a direção de descida: para $t > 0$ pequeno, ponha
        $h = t\,\omega$ com $\omega^k = -1/c_k$ (por que existe um tal $\omega$? ---
        \emph{demonstre} a existência de raízes $k$-ésimas de
        qualquer número complexo pela forma polar, independentemente do
        teorema em demonstração). Mostre que
        \[
        \abs{Q(t\omega)} \leq 1 - t^k + C\,t^{k+1}
        \]
        para $t$ pequeno, com uma constante $C$ explícita.
  \item Conclua $\abs{Q(t\omega)} < 1$ para $t$ pequeno --- contradizendo a
        minimalidade de $\abs{P(z_0)}$. Logo $P(z_0) = 0$: \emph{todo polinômio complexo
        não constante tem raiz} (d'Alembert--Argand).
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A linha única de Liouville, por
extenso.}
\begin{enumerate}[resume]
  \item Escreva com cuidado a demonstração do~\cref{cor:b3:holomorphic:liouville}: se
        $P$ não tem raiz, verifique que $1/P$ é inteira e
        limitada (quantifique, usando a questão 1), logo constante, e
        conclua. Compare as duas demonstrações: que ingredientes cada
        uma usa? (A \omterm{def:b3:topology:compact}{compacidade} aparece em ambas --- onde?)
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A colheita.}
\begin{enumerate}[resume]
  \item Mostre que todo $P \in \C[X]$ de grau $n$ se decompõe: $P = c\prod_{i}(X - \alpha_i)^{m_i}$ com
        $\sum m_i =
        n$ \emph{(indução, divisão euclidiana por $(X -
        \alpha)$)}.
  \item Mostre que os polinômios irredutíveis de $\R[X]$ são os de
        grau um e as quadráticas de discriminante negativo \emph{(pareie
        raízes conjugadas)}; deduza que todo polinômio real de grau
        ímpar tem raiz real e dê uma segunda demonstração desse último
        fato, de natureza ordinal (teorema do valor intermediário) ---
        verificando que ambas concordam em $X^3 - X - 1$.
  \item Deduza as dívidas que este livro pode enfim pagar: (i) todo
        endomorfismo de um $\C$-espaço vetorial não nulo de dimensão
        finita tem autovalor, de modo que toda matriz complexa tem
        \omterm{thm:b3:modules:jordan}{forma de Jordan} (\cref{thm:b3:modules:jordan}); (ii) o corpo $\bar\Q$ dos
        números \omterm{def:b3:galois:algebraic}{algébricos} usado na~\cref{rem:b3:galois:closureexamples} é de fato um \omterm{def:b3:galois:closure}{fecho
        algébrico} de $\Q$.
  \item (Final) Localize onde cada demonstração se quebraria sobre um
        corpo como $\Q(\iu)$: que passos usam a existência de raízes
        $k$-ésimas (questão 4) e quais usam \omterm{def:b3:topology:compact}{compacidade} ou \omterm{def:b3:complete:complete}{completude}
        (questões 2 e 6)? Conclua em cinco linhas: o teorema é
        genuinamente \emph{analítico} --- toda demonstração invoca em
        algum ponto a \omterm{def:b3:complete:complete}{completude} ou a \omterm{def:b3:topology:connected}{conexidade} de $\R$ --- ainda
        que seu enunciado seja puramente \omterm{def:b3:galois:algebraic}{algébrico}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- A escada de rigidez das funções
inteiras.} Liouville é o primeiro degrau de uma escada; vamos subi-la.
\begin{enumerate}[resume]
  \item (\omterm{thm:b3:holomorphic:analytic}{Estimativas de Cauchy}) A partir da fórmula de Cauchy no
        círculo de raio $r$ em torno de $a$, demonstre
        \[
        \bigl|f^{(n)}(a)\bigr| \leq
        \frac{n!\,\sup_{\abs{z-a}=r}\abs f}{r^n} ,
        \]
        e recupere Liouville como o caso $n = 1$, $r \to
        \infty$.
  \item (Crescimento polinomial) Mostre que uma $f$ inteira com
        $\abs{f(z)} \leq A + B\abs z^m$ para todo $z$ é um polinômio de grau $\leq m$
        \emph{(mate os coeficientes de Taylor além de $m$ com a
        questão 11)}.
  \item (Parte real limitada) Mostre que uma $f$ inteira com $\operatorname{Re}f$
        limitada superiormente é constante \emph{(aplique Liouville a
        $\eu^{f}$)}.
  \item (Dupla periodicidade) Seja $f$ inteira com $f(z + 1)
        = f(z)$ e $f(z + \iu) = f(z)$ para
        todo $z$. Mostre que $f$ é constante. Conclua: uma função
        ``elíptica'' não constante tem de ter singularidades --- a razão
        histórica pela qual os polos entram na análise complexa.
  \item (Imagem densa) Mostre que a imagem de uma função inteira não
        constante é densa em $\C$: se $f(\C)$ não encontra um
        disco $D(a, r)$, então $\frac1{f - a}$ é inteira e limitada. (Picard
        demonstrou que a imagem omite no máximo \emph{um ponto}; a
        \omterm{ex:b3:lebesgue:gamma}{densidade} é o nível que nossas ferramentas alcançam.)
  \item (Própria $\Rightarrow$ polinômio) Suponha $f$ inteira com $\abs{f(z)} \to \infty$
        quando $\abs z \to
        \infty$. Mostre: os zeros de $f$ são em número finito
        ($z_1, \dots, z_p$, com multiplicidades $m_i$); o quociente $g = f/\prod(z - z_i)^{m_i}$ é
        inteiro e sem zeros; $1/g$ tem crescimento polinomial, logo
        (questão 12) é um polinômio, necessariamente constante (sem
        zeros); conclua que $f$ é um polinômio. Assim, entre as
        funções inteiras, os polinômios são exatamente as próprias ---
        $\eu^z$ falha a propriedade ao longo de $\R_-$.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Sombras harmônicas e um valor médio de
Gauss.}
\begin{enumerate}[resume]
  \item Seja $f = u + \iu v$ \omterm{def:b3:holomorphic:holo}{holomorfa} em um aberto. Verifique que $u = \operatorname{Re}f$ satisfaz
        a propriedade do valor médio
        \[
        u(a) = \frac1{2\pi}\int_0^{2\pi}
        u\bigl(a + r\eu^{\iu\theta}\bigr)\,\dd\theta
        \]
        (parte real da fórmula de Cauchy) e deduza o princípio do
        máximo para $u$ em um domínio limitado, com a mesma
        demonstração por \omterm{def:b3:topology:connected}{conexidade} usada para $\abs f$.
  \item (Valor médio de Gauss) Para $a \in \C$ e $r > 0$ com $\abs a \neq r$,
        demonstre
        \[
        \frac1{2\pi}\int_0^{2\pi}
        \log\bigl|a - r\eu^{\iu\theta}\bigr|\,\dd\theta
        = \log\max\bigl(\abs a, r\bigr)
        \]
        \emph{(se $\abs a > r$: $z \mapsto \log\abs{a - z}$ é a parte real de um logaritmo \omterm{def:b3:holomorphic:holo}{holomorfo}
        em uma \omterm{def:b3:topology:topology}{vizinhança} do disco fechado --- por que existe um? ---, de
        modo que a questão 17 se aplica; se $\abs a < r$: fatore $\abs{a - r\eu^{\iu\theta}} = r\,\abs{1 -
        \frac ar\eu^{-\iu\theta}}$ e
        reutilize o primeiro caso)}.
  \item (\omterm{def:b3:measure:measure}{Medida} de Mahler) Para $P = c\prod_{i=1}^n(X -
        \alpha_i) \in \C[X]$, deduza a \emph{fórmula de Jensen
        para polinômios}:
        \[
        \frac1{2\pi}\int_0^{2\pi}\log\bigl|P(\eu^{\iu\theta})
        \bigr|\,\dd\theta
        = \log\Bigl(\abs c\prod_{i}\max(1,
        \abs{\alpha_i})\Bigr) :
        \]
        a média geométrica de $\abs P$ no círculo unitário lê as
        raízes de fora do disco. Verifique em $P =
        X^2 - X$ e em $P = 2X - 1$.
  \item (Números de Bernoulli) Defina os coeficientes $B_n$ por
        $\frac{z}{\eu^z - 1} = \sum_{n\geq0}\frac{B_n}{n!}
        z^n$ perto de $0$ (por que o membro esquerdo é analítico em
        $0$?). Deduza a recorrência $\sum_{k=0}^{n}\binom{n+1}kB_k = 0$ ($n \geq 1$) de $(\eu^z - 1)\cdot\frac z{\eu^z-1} = z$, calcule
        $B_0,
        \dots, B_6$ e mostre que $B_{2k+1} = 0$ para $k \geq 1$ \emph{(a função $\frac z{\eu^z-1} + \frac z2$ é par)}.
        Esses números vão precificar todo $\zeta(2k)$ no~\cref{ch:b3:residues}.
  \item (Realidade) Mostre que uma função inteira que toma valores
        reais em $\R$ satisfaz $f(\bar z) =
        \overline{f(z)}$ em toda parte \emph{(compare os
        coeficientes de Taylor em $0$, ou aplique o teorema da
        identidade a $z \mapsto \overline{f(\bar z)}$)}; deduza de novo que as raízes não reais de
        polinômios reais vêm em pares conjugados (o pareamento da
        questão 8, redemonstrado analiticamente).
  \item (A moral) Monte a escada de rigidez: limitada $\Rightarrow$ constante;
        limitada polinomialmente $\Rightarrow$ polinômio; própria $\Rightarrow$
        polinômio; omite um disco $\Rightarrow$ constante; duplamente periódica
        $\Rightarrow$ constante. Contraste em um parágrafo curto com as funções
        reais $\mathcal C^\infty$ (funções de corte, \cref{thm:b3:lp:regularization}): por que a
        \omterm{def:b3:holomorphic:holo}{holomorfia}, uma condição puramente local, impõe lei e ordem
        globais?
\end{enumerate}

\medskip
\noindent\textbf{Parte VII --- Última colheita.}
\begin{enumerate}[resume]
  \item (Desigualdade de Landau) Para $P = \sum_{k=0}^na_kX^k$, demonstre o valor médio
        $\frac1{2\pi}\int_0^{2\pi}
        \abs{P(\eu^{\iu\theta})}^2\dd\theta =
        \sum_k\abs{a_k}^2$ (ortogonalidade das $\eu^{\iu k\theta}$) e, depois, usando a cota
        pontual $\log t \leq t - 1$ para comparar as médias de $\log\abs P^2$ e $\abs P^2$, deduza da
        questão 19 que
        \[
        \abs c\prod_{i}\max\bigl(1, \abs{\alpha_i}\bigr)
        \;\leq\;
        \Bigl(\sum_{k=0}^{n}\abs{a_k}^2\Bigr)^{1/2} :
        \]
        o produto das raízes fora do disco unitário é controlado pelo
        tamanho $\ell^2$ dos coeficientes (trate as raízes
        \emph{sobre} o círculo aplicando a desigualdade a $P(rX)$ e
        fazendo $r \to 1$). Confira em $X^2 - X$.
  \item (Os números de Bernoulli crescem fatorialmente) Mostre que o
        raio de convergência de $\sum\frac{B_n}{n!}z^n$ é \emph{exatamente} $2\pi$: ao
        menos $2\pi$ porque $z/(\eu^z - 1)$ se estende \omterm{def:b3:holomorphic:holo}{holomorficamente} a
        $D(0, 2\pi)$, e no máximo $2\pi$ porque, do contrário, a soma
        permaneceria limitada perto de $2\pi\iu$, onde $\abs{z/(\eu^z-1)} \to \infty$. Deduza
        \[
        \limsup_{k\to\infty}
        \Bigl(\frac{\abs{B_{2k}}}{(2k)!}\Bigr)^{1/2k}
        = \frac1{2\pi} :
        \]
        os números de Bernoulli crescem fatorialmente. Admitindo $B_{12}
        = -\frac{691}{2730}$
        (a recorrência da questão 20, levada adiante), compare $\abs{B_{12}} \approx 0.25311$ com
        a previsão mais fina $2\,(2k)!/(2\pi)^{2k}
        \approx 0.25305$ em $k = 6$ --- quatro algarismos
        coincidentes de uma lei assintótica que o~\cref{ch:b3:residues}
        demonstrará exatamente, via $\zeta(2k)$.
  \item (As raízes se movem \omterm{def:b3:topology:continuity}{continuamente}) Sejam $(P_j)$ mônicos de
        grau $n$ com coeficientes convergindo aos de um $P$
        (mônico). Demonstre a \emph{cota de Cauchy}: toda raiz de um
        mônico $Q = X^n + \sum_{k<n}q_kX^k$ satisfaz $\abs\alpha \leq 1 + \max_k\abs{q_k}$; deduza que as raízes dos $P_j$
        permanecem em um \omterm{def:b3:topology:compact}{compacto} fixo e, extraindo subsequências
        convergentes de vetores de raízes e passando ao limite na
        fatoração da questão 7, que os multiconjuntos de raízes de
        $P_j$ convergem ao de $P$. Mostre, por fim, que a
        \omterm{def:b3:topology:continuity}{continuidade} é o máximo que se pode afirmar: para $P_\varepsilon = X^2 - 2X + 1 +
        \varepsilon$, uma
        perturbação de tamanho $\varepsilon =
        10^{-4}$ move a raiz dupla $1$ de $10^{-2}$
        --- expoente de Hölder $\frac1m$ em uma raiz de multiplicidade
        $m$, nunca lipschitziano: numericamente, raízes múltiplas
        custam metade dos algarismos.
\end{enumerate}
\end{problem}
