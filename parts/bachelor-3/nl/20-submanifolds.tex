\chapter{Deelvariëteiten van \texorpdfstring{$\R^n$}{Rn}}
\label{ch:b3:submanifolds}

Sferen, tori, rotatiegroepen: de natuurlijke leefomgevingen van
de meetkunde en de mechanica zijn geen vectorruimten maar
\emph{gekromde verzamelingen die van dichtbij vlak lijken}. Dit
hoofdstuk geeft die uitdrukking een precieze betekenis ---
deelvariëteiten van $\R^n$ --- en de calculus om erop te werken.
Het fundament is de inverse-functiestelling, hier bewezen met het
vaste punt van Banach; al de rest is coördinaten verwisselen: de
vier gelijkwaardige beschrijvingen van een deelvariëteit (lokaal
rechttrekken, niveauverzamelingen, grafieken, parametriseringen),
\omterm{def:b3:submanifolds:tangent}{raakruimten}, en optimaliseren onder nevenvoorwaarden met de
\omterm{thm:b3:submanifolds:lagrange}{multiplicatoren van Lagrange} --- die, als afscheidsdemonstratie,
de spectraalstelling voor symmetrische matrices in drie regels
meetkunde herbewijst. De weekendopgave bouwt de rotatiegroep
$SO(3)$ en haar quaternionische dubbele overdekking: algebra (de
$Q_8$ van \cref{ch:b3:groups}, volwassen geworden) die de
meetkunde ontmoet.

\section{De inverse-functiestelling}

\begin{theorem}[Inverse-functiestelling]
\label{thm:b3:submanifolds:ift}
Zij $U \subseteq \R^n$ open, $f \colon U \to \R^n$ van klasse
$\mathcal C^1$, en $a \in U$ met $Df(a)$ inverteerbaar. Dan zijn
er \omterm{def:b3:topology:topology}{open verzamelingen} $V \ni a$ en $W \ni f(a)$ zodanig dat $f
\colon V \to W$ een bijectie is met een $\mathcal C^1$-inverse,
en\index{inverse-functiestelling}
\[
D(f^{-1})(y) = \bigl(Df(f^{-1}(y))\bigr)^{-1}
\qquad (y \in W).
\]
Is $f$ $\mathcal C^k$, dan is $f^{-1}$ dat ook.
\end{theorem}

\begin{proof}
Normeer: door $f$ te vervangen door $x \mapsto
Df(a)^{-1}\bigl(f(a + x) - f(a)\bigr)$ mogen we aannemen dat $a =
0$, $f(0) = 0$ en $Df(0) = I$ (de algemene uitspraak volgt door
met de affiene bijecties samen te stellen). Schrijf $f(x) = x +
g(x)$: dan is $Dg(0) = 0$, en kies wegens de \omterm{def:b3:topology:continuity}{continuïteit} van
$Dg$ een $r > 0$ met $\vertiii{Dg(x)} \leq \frac12$ op $\bar B(0,
r)$; de middelwaardeongelijkheid geeft daar $\norm{g(x) - g(x')}
\leq \frac12 \norm{x - x'}$.

\emph{Bijectiviteit op een \omterm{def:b3:topology:topology}{omgeving}.} Voor $y \in B(0, \frac
r2)$ betekent $f(x) = y$ oplossen dat je een vast punt van
$\Phi_y(x) = y - g(x)$ zoekt; $\Phi_y$ beeldt $\bar B(0, r)$ in
zichzelf af ($\norm{\Phi_y(x)} \leq \norm y + \frac12\norm x \leq
r$) en is $\frac12$-lipschitz: Banach
(\cref{thm:b3:complete:banach}) geeft een unieke oplossing $x =
\varphi(y) \in \bar B(0,r)$. Bovendien is $f$ injectief op $\bar
B(0,r)$:
\[
\norm{f(x) - f(x')} \geq \norm{x - x'} - \norm{g(x) - g(x')}
\geq \tfrac12\norm{x - x'} .
\tag{$*$}
\]
Stel $W = B(0, \frac r2)$ en $V = f^{-1}(W)\cap B(0, r)$: open
(\omterm{def:b3:topology:continuity}{continuïteit}), met $f \colon V \to W$ bijectief.

\emph{\omterm{def:b3:topology:continuity}{Continuïteit} en differentieerbaarheid van de inverse.}
$(*)$ zegt dat $\varphi = f^{-1}$ $2$-lipschitz is. Leg $y_0 =
f(x_0) \in W$ vast; de inverteerbaarheid van $A = Df(x_0)$ (haar
afstand tot $I$ is $\leq \frac12$: Neumann,
\cref{prop:b3:banach:neumann}) en de differentieerbaarheid van
$f$ geven voor $y = f(x)$ nabij $y_0$
\[
\varphi(y) - \varphi(y_0) - A^{-1}(y - y_0)
= -A^{-1}\bigl(f(x) - f(x_0) - A(x - x_0)\bigr)
= -A^{-1}\,o(\norm{x - x_0}) = o(\norm{y - y_0}),
\]
met $(*)$ om $\norm{x - x_0} \leq 2\norm{y - y_0}$ om te zetten:
dus is $\varphi$ differentieerbaar in $y_0$ met de inverse
differentiaal. \omterm{def:b3:topology:continuity}{Continuïteit} van $y \mapsto D\varphi(y) =
Df(\varphi(y))^{-1}$: een samenstelling van \omterm{def:b3:topology:continuity}{continue
afbeeldingen} (het inverteren is \omterm{def:b3:topology:continuity}{continu},
\cref{prop:b3:banach:neumann}): dus $\varphi \in \mathcal C^1$;
en dezelfde formule als hefboom geeft $\mathcal C^k$.
\end{proof}

\begin{theorem}[Impliciete-functiestelling]
\label{thm:b3:submanifolds:implicit}
Zij $F \colon U \subseteq \R^p\times\R^q \to \R^q$ $\mathcal C^1$
nabij $(a, b)$, met $F(a,b) = 0$, en neem aan dat de partiële
differentiaal $D_yF(a,b) \in \mathcal L(\R^q)$ inverteerbaar is.
Dan zijn er \omterm{def:b3:topology:topology}{omgevingen} $A \ni a$ en $B \ni b$ en een $\mathcal
C^1$-afbeelding $\psi \colon A \to B$ met
\[
\bigl\{(x, y)\in A\times B : F(x,y) = 0\bigr\}
= \{(x, \psi(x)) : x \in A\},
\]
en $D\psi(x) = -D_yF(x, \psi(x))^{-1}\,D_xF(x,
\psi(x))$.\index{impliciete-functiestelling}
\end{theorem}

\begin{proof}
Pas \cref{thm:b3:submanifolds:ift} toe op $\Theta(x, y) = (x,
F(x,y))$: haar differentiaal in $(a,b)$, blokdriehoekig met
inverteerbare diagonaalblokken $I$ en $D_yF$, is inverteerbaar.
De lokale inverse heeft de vorm $\Theta^{-1}(x, z) = (x, h(x,
z))$; stel $\psi(x) = h(x, 0)$: dan is lokaal $F(x, y) = 0$ dan
en slechts dan als $\Theta(x,y) = (x, 0)$, dat wil zeggen als $y
= \psi(x)$. De formule: differentieer $F(x, \psi(x)) = 0$ met de
kettingregel.
\end{proof}

\section{Deelvariëteiten: vier definities}

\begin{theorem}[Gelijkwaardige karakteriseringen]
\label{thm:b3:submanifolds:characterizations}
Zij $M \subseteq \R^n$, $d \in \{0, \dots, n\}$ en $k \geq 1$. De
volgende uitspraken zijn gelijkwaardig, voor elk punt $a \in M$
(en $M$ is een \emph{$d$-dimensionale deelvariëteit van klasse
$\mathcal C^k$}\index{deelvariëteit} als ze in elke $a \in M$
gelden):
\begin{enumerate}
  \item (\emph{Rechttrekken}) Er is een $\mathcal
        C^k$-diffeomorfisme $\Phi$ van een open $\Omega \ni a$ op
        een open $\Omega' \subseteq \R^n$ met
        \[
        \Phi(M\cap\Omega) = \Omega' \cap
        \bigl(\R^d\times\{0\}\bigr).
        \]
  \item (\emph{Niveauverzameling}) Er is een $\mathcal
        C^k$-\emph{submersie} $F \colon \Omega \to \R^{n-d}$
        (dat wil zeggen $DF(x)$ surjectief) op een open $\Omega
        \ni a$ met $M\cap\Omega = F^{-1}(0)$.
  \item (\emph{Grafiek}) Op een permutatie van de coördinaten na
        is $M$ lokaal de grafiek van een $\mathcal
        C^k$-afbeelding $\psi \colon A \subseteq \R^d \to
        \R^{n-d}$.
  \item (\emph{Parametrisering}) Er is een $\mathcal
        C^k$-\emph{immersie} $\varphi \colon A \subseteq \R^d \to
        \R^n$ ($D\varphi(u)$ injectief) met $A$ open, waarbij
        $\varphi$ een \omterm{def:b3:topology:continuity}{homeomorfisme} is van $A$ op $M \cap \Omega$
        voor een open $\Omega \ni a$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1)$\Rightarrow$(2): $F = (\Phi_{d+1}, \dots, \Phi_n)$ (de
laatste coördinaten van $\Phi$): een submersie ($D\Phi$ is
inverteerbaar).
(2)$\Rightarrow$(3): $DF(a)$ is surjectief, dus is een
$q\times q$-minor van de jacobiaan inverteerbaar ($q = n - d$);
na permutatie van de coördinaten is $D_yF(a)$ inverteerbaar, en
de impliciete-functiestelling
(\cref{thm:b3:submanifolds:implicit}) drukt $M$ lokaal uit als
grafiek $y = \psi(x)$.
(3)$\Rightarrow$(4): $\varphi(x) = (x, \psi(x))$: een immersie
(de differentiaal $\bigl(\begin{smallmatrix}I\\
D\psi\end{smallmatrix}\bigr)$ is injectief), en een
\omterm{def:b3:topology:continuity}{homeomorfisme} op de grafiek (de inverse is de projectie,
\omterm{def:b3:topology:continuity}{continu}).
(4)$\Rightarrow$(1): zij $\varphi(u_0) = a$; vul
$\operatorname{im}D\varphi(u_0)$ aan met een complement $E$
($\dim E = n - d$) en definieer $\Theta(u, v) = \varphi(u) + v$
op $A\times E$: $D\Theta(u_0, 0)$ is bijectief (het beeld bevat
$\operatorname{im}D\varphi(u_0)$ en $E$), dus is $\Theta$ een
lokaal diffeomorfisme (\cref{thm:b3:submanifolds:ift}); haar
inverse $\Phi$ trekt recht: nabij $a$ zijn de punten van $M$
precies de $\varphi(u) = \Theta(u, 0)$ --- daarvoor waarborgt de
hypothese van het \omterm{def:b3:topology:continuity}{homeomorfisme} in (4) dat $M\cap\Omega$ voor
kleine $\Omega$ geen andere bladen bevat ($\varphi(u') + v = m
\in M$ dicht bij $a$ met kleine $v \neq 0$ moet worden
uitgesloten: $m = \varphi(u'')$ voor een $u''$ nabij $u_0$ wegens
de homeomorfismeeigenschap, en de lokale injectiviteit van
$\Theta$ dwingt $v = 0$ af). Dan is $\Phi(M\cap\Omega) =
(A\times\{0\})\cap\Phi(\Omega)$, na eventueel te verkleinen.
\end{proof}

\begin{example}\label{ex:b3:submanifolds:examples}
De sfeer $S^{n-1} = \{\norm x^2 = 1\}$: de niveauverzameling van
de submersie $F(x) = \norm x_2^2 - 1$ op $\R^n\setminus\{0\}$
($DF(x) = 2x^{\mathsf T} \ne 0$): een $\mathcal
C^\infty$-deelvariëteit van dimensie $n - 1$. De torus in $\R^3$:
de niveauverzameling van $\bigl(\sqrt{x^2 + y^2} - R\bigr)^2 +
z^2 - r^2$ ($0 < r < R$). De kegel $\{x^2 + y^2 = z^2\}$ is in
$0$ \emph{geen} deelvariëteit (\cref{exo:b3:submanifolds:1}).
Matrixgroepen: $SL_n$ en $O_n$ zijn deelvariëteiten van
$M_n(\R)$ (\cref{exo:b3:submanifolds:5,exo:b3:submanifolds:6})
--- het startpunt van de lietheorie.
\end{example}

\section{Raakruimten}

\begin{definition}\label{def:b3:submanifolds:tangent}
Zij $M$ een $d$-deelvariëteit en $a \in M$. De
\emph{raakruimte}\index{raakruimte} $T_aM$ is de verzameling
snelheidsvectoren $\gamma'(0)$ van $\mathcal C^1$-krommen $\gamma
\colon \intoo{-\varepsilon}\varepsilon \to M$ met $\gamma(0) =
a$.
\end{definition}

\begin{proposition}\label{prop:b3:submanifolds:tangent}
$T_aM$ is een $d$-dimensionale deelvectorruimte van $\R^n$, en:
\begin{enumerate}
  \item is $M = F^{-1}(0)$ lokaal met $F$ een submersie, dan is
        $T_aM = \ker DF(a)$;
  \item is $M$ geparametriseerd door de immersie $\varphi$ (met
        $\varphi(u_0) = a$), dan is $T_aM =
        \operatorname{im}D\varphi(u_0)$.
\end{enumerate}
\end{proposition}

\begin{proof}
Krommen in $M$ voldoen aan $F(\gamma(t)) = 0$; de kettingregel in
$0$ geeft $DF(a)\gamma'(0) = 0$: dus $T_aM \subseteq \ker DF(a)$.
Omgekeerd vervoert het rechttrekken (\cref{thm:b3:submanifolds:%
characterizations}(1)) rechten van $\R^d\times\{0\}$ naar krommen
van $M$: elke vector van een $d$-dimensionale deelruimte wordt
verwezenlijkt; de dimensies vergelijken ($\dim\ker DF(a) = n - (n
- d) = d$) dwingt de gelijkheid in (1) af, en hetzelfde
transportargument geeft (2) ($D\varphi(u_0)$ toegepast op rechten
in $A$; opnieuw de dimensies).
\end{proof}

\begin{theorem}[Multiplicatoren van Lagrange]
\label{thm:b3:submanifolds:lagrange}
Zij $M = F^{-1}(0)$ met $F = (F_1, \dots, F_q) \colon \Omega \to
\R^q$ een $\mathcal C^1$-submersie, en zij $f \colon \Omega \to
\R$ $\mathcal C^1$. Heeft de beperking $f\restriction_M$ een
lokaal extremum in $a \in M$, dan zijn er unieke reële getallen
$\lambda_1, \dots, \lambda_q$ (de \emph{\omterm{thm:b3:submanifolds:lagrange}{multiplicatoren van
Lagrange}}) met\index{multiplicatoren van Lagrange}
\[
\nabla f(a) = \lambda_1\nabla F_1(a) + \dots +
\lambda_q\nabla F_q(a) .
\]
\end{theorem}

\begin{proof}
Voor elke kromme $\gamma$ in $M$ door $a$ heeft $t \mapsto
f(\gamma(t))$ een lokaal extremum in $0$, dus is $0 =
\frac{\dd}{\dd t}f(\gamma(t))\big|_0 = \langle\nabla f(a),
\gamma'(0)\rangle$: dus $\nabla f(a) \perp T_aM = \ker DF(a)$
(\cref{prop:b3:submanifolds:tangent}). Nu is $\ker DF(a)^\perp =
\operatorname{im}DF(a)^{\mathsf T}$ (de identiteit tussen rang en
orthogonaliteit uit het volume van bachelorjaar 2, of
\cref{exo:b3:hilbert:8} in eindige dimensie: $(\ker T)^\perp =
\operatorname{im}T^*$), die door de gradiënten $\nabla F_i(a)$
wordt opgespannen --- onafhankelijk, want $DF(a)$ is surjectief:
de multiplicatoren bestaan en zijn uniek.
\end{proof}

\begin{example}[De spectraalstelling, meetkundig]
\label{ex:b3:submanifolds:rayleigh}
Zij $A$ een reële symmetrische $n\times n$-matrix, en
maximaliseer $f(x) = \langle Ax, x\rangle$ op de sfeer $S^{n-1}$
(\omterm{def:b3:topology:compact}{compact}: het maximum wordt bereikt, in een $v_1$). Lagrange met
$F(x) = \norm x^2 - 1$: $\nabla f = 2Ax$ en $\nabla F = 2x$ geven
$Av_1 = \lambda_1v_1$ --- een eigenvector, met $\lambda_1 =
\max_{S^{n-1}}\langle Ax, x\rangle$. Beperk $A$ tot $v_1^\perp$
(invariant: $\langle Av, v_1\rangle = \langle v, Av_1\rangle =
\lambda_1\langle v, v_1\rangle = 0$) en itereer: een orthonormale
basis van eigenvectoren. De spectraalstelling uit het volume van
bachelorjaar 2, herbewezen met pure optimalisatie --- en de
oneindigdimensionale schaduw van hetzelfde argument bewees
\cref{lem:b3:spectral:existence}.
\end{example}

\begin{method}\label{met:b3:submanifolds:method}
Om aan te tonen dat een verzameling een deelvariëteit is: geef
haar lokaal als $F^{-1}(0)$ met $DF$ surjectief \emph{op de
verzameling} (de gebruikelijkste weg), of als grafiek. Om haar
dimensie en \omterm{def:b3:submanifolds:tangent}{raakruimte} te berekenen: $d = n - q$ en $T_a = \ker
DF(a)$. Om erop te optimaliseren: Lagrange --- ga altijd eerst de
\omterm{def:b3:topology:compact}{compactheid} (of de coërciviteit) na, zodat er een extremum
bestaat waarop de stelling kan slaan, en onthoud dat de
vergelijking met de multiplicatoren enkel \emph{noodzakelijk} is:
verzamel alle kritieke punten en vergelijk daarna de waarden.
Voor matrixgroepen differentieer je krommen in de eenheid om de
\omterm{def:b3:submanifolds:tangent}{raakruimten} te bepalen.
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b3:submanifolds:1}
(a) Ga na dat de volgende $\mathcal C^\infty$-deelvariëteiten
zijn en geef hun dimensies: $S^{n-1}$; de hyperboloïde $\{x^2 +
y^2 - z^2 = 1\}$; de torus van
\cref{ex:b3:submanifolds:examples}.
(b) Toon aan dat de kegel $C = \{x^2 + y^2 = z^2\} \subseteq
\R^3$ in $0$ geen $2$-deelvariëteit is: bepaal het aantal
\omterm{def:b3:topology:components}{samenhangscomponenten} van
$\bigl(C\setminus\{0\}\bigr)\cap B(0,\varepsilon)$, en vergelijk
met het aantal dat een rechttrekking
(\cref{thm:b3:submanifolds:characterizations}(1)) voor een vlak
min een punt zou afdwingen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:submanifolds:2}
Bereken de \omterm{def:b3:submanifolds:tangent}{raakruimten}:
(a) $T_aS^{n-1}$ voor willekeurige $a$ (antwoord: $a^\perp$);
(b) het raakvlak aan de torus van
\cref{ex:b3:submanifolds:examples} in een willekeurig punt van de
buitenevenaar $\{z = 0,\ x^2 + y^2 = (R + r)^2\}$;
(c) de raaklijn aan de schroeflijn $\varphi(t) = (\cos t, \sin t,
t)$ in $\varphi(t_0)$, met een controle van
\cref{prop:b3:submanifolds:tangent}(2).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:3}
Zij $f(x, y) = (x^2 - y^2,\ 2xy)$ (dat wil zeggen $z \mapsto
z^2$).
(a) In welke punten is \cref{thm:b3:submanifolds:ift} van
toepassing?
(b) Toon aan dat $f$ op $\R^2\setminus\{0\}$ lokaal maar niet
globaal inverteerbaar is, en geef expliciet de twee lokale
inversen die op een \omterm{def:b3:topology:topology}{omgeving} van $(1, 0)$ gedefinieerd zijn (de
twee takken van de vierkantswortel).
(c) Dezelfde bespreking voor de afbeelding van de
\omterm{ex:b3:product:polar}{poolcoördinaten} $(r, \theta) \mapsto (r\cos\theta,
r\sin\theta)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:4}
(Folium) Zij $F(x,y) = x^3 + y^3 - 3xy$ en $\mathcal C =
F^{-1}(0)$.
(a) Toon aan dat $\mathcal C$ nabij elk punt van $\mathcal C$
behalve de oorsprong een $1$-deelvariëteit is, lokaal een grafiek
in $x$ of in $y$ (welke, waar?).
(b) Bereken de raaklijn in $(\frac32, \frac32)$.
(c) Wat gebeurt er in de oorsprong? (Twee takken kruisen elkaar:
geef twee $\mathcal C^1$-krommen in $\mathcal C$ door $0$ met
onafhankelijke snelheden, en besluit dat er geen rechttrekking
bestaat.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:5}
Zij $F(M) = M^{\mathsf T}M$ van $M_n(\R)$ naar de ruimte $S_n$
van de symmetrische matrices.
(a) Toon aan dat $DF(M)(H) = M^{\mathsf T}H + H^{\mathsf T}M$ en
dat $DF(M)$ in elke $M \in O_n$ surjectief is op $S_n$
\emph{(probeer bij gegeven $S \in S_n$ $H = \frac12MS$)}.
(b) Besluit dat $O_n = F^{-1}(I)$ een \omterm{def:b3:topology:compact}{compacte} $\mathcal
C^\infty$-deelvariëteit van dimensie $\frac{n(n-1)}2$ is, met
$T_IO_n = \{H : H^{\mathsf T} = -H\}$, de antisymmetrische
matrices.
(c) Toon aan dat $\eu^{tH} \in O_n$ voor elke antisymmetrische
$H$: de raakrichtingen integreren tot krommen in de groep.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:6}
(a) Toon aan dat $\det \colon M_n(\R) \to \R$ de differentiaal

^{\mathsf T}H\bigr)$ heeft, die in elke $M \in SL_n$ niet nul
is.
(b) Besluit dat $SL_n(\R)$ een deelvariëteit van dimensie $n^2 -
1$ is met $T_ISL_n = \{H : \operatorname{tr}H = 0\}$.
(c) Is $GL_n(\R)$ een deelvariëteit? Van welke dimensie?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:7}
Met de \omterm{thm:b3:submanifolds:lagrange}{multiplicatoren van Lagrange}:
(a) zoek de extrema van $f(x,y) = xy$ op de cirkel $x^2 + y^2 =
1$;
(b) toon aan dat onder alle kansvectoren $(p_1, \dots, p_n)$

(c) zoek het punt van de ellips $\{x^2/4 + y^2 = 1\}$ dat het

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:submanifolds:8}
Schrijf \cref{ex:b3:submanifolds:rayleigh} voluit uit: bewijs met
inductie dat een reële symmetrische matrix een orthonormale basis
van eigenvectoren toelaat, waarbij $\lambda_1 \geq \dots \geq
\lambda_n$ de opeenvolgende maxima onder nevenvoorwaarden van het
quotiënt van Rayleigh zijn. Leid daarna de formules van
Courant--Fischer uit \cref{exo:b3:spectral:8} in eindige dimensie
rechtstreeks uit deze constructie af.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:submanifolds:9}
(Ongelijkheid van Hadamard) Voor $M \in GL_n(\R)$ met kolommen
$c_1, \dots, c_n$ geldt
\[
\abs{\det M} \;\leq\; \prod_{i=1}^n\norm{c_i}_2 ,
\]
met gelijkheid dan en slechts dan als de kolommen orthogonaal
zijn. \emph{(Herleid door te schalen tot kolommen met norm $1$;
maximaliseer $\det$ op het \omterm{def:b3:topology:compact}{compacte} product van sferen
$(S^{n-1})^n$; in een maximalisator geeft Lagrange in elke kolom
afzonderlijk $\nabla_{c_i}\det = \lambda_ic_i$, en
$\nabla_{c_i}\det$ is de $i$-de kolom van
$\operatorname{com}(M)$: leid af dat $M^{\mathsf T}M$ diagonaal
is, dus $= I$, dus $\det = \pm1$.)} Meetkundige lezing: het
volume van een parallellepipedum is hoogstens het product van
zijn ribbelengten.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:10}
Nabij welke van haar punten is de cirkel $S^1$ een grafiek $y =
\psi(x)$? En een grafiek $x = \chi(y)$? Ga de karakterisering met
grafieken (\cref{thm:b3:submanifolds:%
characterizations}(3)) expliciet na in $(1, 0)$, en leg in één
zin uit waarom \emph{een} permutatie van de coördinaten altijd
volstaat maar geen enkele vaste permutatie altijd werkt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:11}
(De orthogonale groep als deelvariëteit, kwantitatief)
(a) Toon aan dat $O_n = \{M : M^{\mathsf T}M = I\}$ \omterm{def:b3:topology:compact}{compact} is:
begrensd (elke kolom is een eenheidsvector, dus $\norm M \leq
\sqrt n$ voor de euclidische matrixnorm) en gesloten.
(b) Toon aan dat haar \omterm{def:b3:submanifolds:tangent}{raakruimte} in $I$ de ruimte van de
antisymmetrische matrices is, van dimensie $\frac{n(n-1)}2$, en
in een algemene $A \in O_n$: $T_AO_n = \{AK : K^{\mathsf T} =
-K\}$.
(c) Leid af dat $t \mapsto A\exp(tK)$ voor elke antisymmetrische
$K$ een kromme in $O_n$ door $A$ is met snelheid $AK$ \emph{(ga
$\exp(tK) \in O_n$ na met $\exp(X)^{\mathsf T} =
\exp(X^{\mathsf T})$ en $\exp(-X)\exp(X) = I$)}: elke raakvector
wordt door een expliciete kromme verwezenlijkt, zonder dat er een
impliciete-functiestelling nodig is.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:submanifolds:12}
(Kritieke punten van de afstand) Zij $M \subseteq \R^n$ een
deelvariëteit en $p \notin M$. Toon aan dat als $x_0 \in M$ de
afstand tot $p$ minimaliseert (zo'n punt bestaat als $M$ gesloten
en niet leeg is --- waarom?), dan
\[
p - x_0 \;\perp\; T_{x_0}M
\]
\emph{(differentieer $t \mapsto \norm{\gamma(t) - p}^2$ langs
krommen in $M$)}. Leid af: het dichtstbijzijnde punt op een sfeer
ligt op de straal door het middelpunt; en gebruik de voorwaarde
om de afstand van $p = (2, 0)$ tot de parabool $y = x^2$ te
berekenen (herleid tot een derdegraadsvergelijking en los die
numeriek op tot drie cijfers).
\end{exercise}

\section{Probleem: \texorpdfstring{$SO(3)$}{SO(3)} en de
quaternionen}

\begin{problem}[{Weekendopgave --- rotaties, de groep $S^3$ en de
dubbele overdekking}]\label{pb:b3:submanifolds:1}
De quaternionen $\mathbb H = \{t + x\mathrm i + y\mathrm j +
z\mathrm k\}$ --- de algebra waarvan de eenhedengroep de $Q_8$
van \cref{pb:b3:groups:1} bevat --- parametriseren de
driedimensionale rotaties dubbel: de afbeelding ``vervoegen met
een eenheidsquaternion'' is een surjectief morfisme $S^3 \to
SO(3)$ met kern $\{\pm1\}$. We bouwen alles op. Ter herinnering
en als definitie: de vermenigvuldiging is $\R$-bilineair met
$\mathrm i^2 = \mathrm j^2 = \mathrm k^2 = \mathrm{ijk} = -1$; de
toegevoegde van $q = t + x\mathrm i + y\mathrm j + z\mathrm k$ is
$\bar q = t - x\mathrm i - y\mathrm j - z\mathrm k$; en $N(q) =
q\bar q = t^2 + x^2 + y^2 + z^2$.

\medskip
\noindent\textbf{Deel I --- De algebra $\mathbb H$ en de groep
$S^3$.}
\begin{enumerate}
  \item Ga na dat $\mathbb H$ een associatieve $\R$-algebra is
        met \omterm{ex:b3:groups:actions}{centrum} $\R$, dat $\overline{pq} = \bar q\,\bar p$,
        en dat $N(pq) = N(p)N(q)$ \emph{(één nette weg: stel $q$
        voor als de complexe $2\times2$-matrix
        $\bigl(\begin{smallmatrix}\alpha & \beta\\
        -\bar\beta & \bar\alpha\end{smallmatrix}\bigr)$ met $q =
        \alpha + \beta\mathrm j$, en gebruik $\det$)}.
  \item Leid af dat elke $q \neq 0$ inverteerbaar is ($q^{-1} =
        \bar q/N(q)$): $\mathbb H$ is een (niet-commutatief)
        lichaam, en $S^3 = \{N(q) = 1\}$ is een groep --- en een
        \omterm{def:b3:topology:compact}{compacte} $3$-deelvariëteit van $\R^4$
        (\cref{ex:b3:submanifolds:examples}).
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het rotatiemorfisme.}
Identificeer $\R^3$ met de \emph{pure quaternionen} $P =
\{x\mathrm i + y\mathrm j + z\mathrm k\}$, en definieer voor $q
\in S^3$ $\rho_q(v) = q\,v\,\bar q$.
\begin{enumerate}[resume]
  \item Toon aan dat $\rho_q$ $P$ in $P$ afbeeldt \emph{(de pure
        quaternionen zijn die met $\bar v = -v$)}, dat ze
        $\R$-lineair is en de norm behoudt, en dat $\rho \colon q
        \mapsto \rho_q$ een groepsmorfisme $S^3 \to O(3)$ is.
  \item Bereken de kern: $\rho_q = \mathrm{id}$ dan en slechts
        dan als $q$ met $\mathrm i, \mathrm j, \mathrm k$
        commuteert, dat wil zeggen $q \in \R\cap S^3 =
        \{\pm1\}$.
  \item Schrijf $q = \cos\frac\theta2 + \sin\frac\theta2\,u$ met
        $u \in P$ en $N(u) = 1$ (waarom kan dat voor $q \in S^3$
        altijd?). Toon aan dat $\rho_q$ $u$ vastlaat en op het
        vlak $u^\perp\cap P$ werkt als de rotatie over hoek
        $\theta$ \emph{(bereken $\rho_q(w)$ voor $w \perp u$ met
        $uw = -wu$ voor orthogonale pure eenheden --- bewijs die
        identiteit met de vermenigvuldigingstabel, of uit $uw +
        wu = -2\langle u, w\rangle$)}.
  \item Besluit: $\operatorname{im}\rho \subseteq SO(3)$ (elke
        $\rho_q$ is een rotatie met de berekende as en hoek ---
        determinant $+1$ wegens de \omterm{def:b3:topology:continuity}{continuïteit} van $q \mapsto
        \det\rho_q$ op het \omterm{def:b3:topology:connected}{samenhangende} $S^3$, of rechtstreeks),
        en $\rho$ is \emph{surjectief} op $SO(3)$: elke rotatie
        van $\R^3$ heeft een as (bewijs: een reële orthogonale
        $3\times3$-matrix met $\det = 1$ heeft eigenwaarde $1$
        --- beschouw de karakteristieke veelterm) en is dus een
        $\rho_q$. Samengevat:
        \[
        SO(3) \;\cong\; S^3/\{\pm 1\} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- $SO(3)$ als deelvariëteit;
Rodrigues.}
\begin{enumerate}[resume]
  \item Toon aan dat $SO(3)$ een \omterm{def:b3:topology:compact}{compacte} $3$-dimensionale
        deelvariëteit van $M_3(\R)$ is met $T_ISO(3) =$ de
        antisymmetrische matrices
        (\cref{exo:b3:submanifolds:5}; de determinantvoorwaarde
        selecteert een vereniging van componenten).
  \item Bewijs voor de antisymmetrische matrix $A_u$ die bij $u
        \in \R^3$ hoort ($A_uv = u\wedge v$, het uitwendig
        product) de \emph{formule van Rodrigues}:
        \[
        \eu^{\theta A_u} = I + \sin\theta\,A_u + (1 -
        \cos\theta)\,A_u^2 \qquad (\norm u = 1)
        \]
        \emph{(uit $A_u^3 = -A_u$: splits de exponentiële reeks
        langs de machten van $A_u$)}, en identificeer haar als de
        rotatie met as $u$ over hoek $\theta$. Leid af dat $\exp$
        de antisymmetrische matrices \emph{op} $SO(3)$ afbeeldt.
  \item Verbind de twee parametriseringen: toon aan dat $t
        \mapsto \rho_{q(t)}$ met $q(t) = \cos\frac t2 +
        \sin\frac t2\,u$ een eenparametergroep van rotaties is
        waarvan de afgeleide in $t = 0$ gelijk is aan $A_u$ ---
        de quaternionische en de \omterm{thm:b3:ode:matrixexp}{matrixexponentiële} vertellen
        hetzelfde verhaal, respectievelijk op halve en op volle
        snelheid.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De dubbele overdekking, gevoeld.}
\begin{enumerate}[resume]
  \item Toon aan dat de weg $q(t) = \cos\frac t2 + \sin\frac
        t2\,\mathrm k$, $t \in \intcc0{2\pi}$, een lus in
        $SO(3)$ is (haar beeld $\rho_{q(t)}$ keert terug naar de
        identiteit) waarvan de quaternionische lift \emph{geen}
        lus is: $q(2\pi) = -q(0)$. Doorgaan tot $t = 4\pi$ sluit
        de lift. Leg in een korte alinea uit wat dat zegt: een
        rotatie over $2\pi$ is niet \omterm{def:b3:topology:continuity}{continu} ongedaan te maken en
        een over $4\pi$ wel (de riemtruc), omdat de lussen van
        $SO(3)$ in haar dubbele overdekking $S^3$ worden
        opgespoord.
  \item Leid ook het praktische dividend af: het samenstellen van
        rotaties = het vermenigvuldigen van quaternionen ($4$
        vermenigvuldigingen aan gegevens in plaats van $9$, geen
        afdrijven van de orthogonaliteit) --- ga dat na op de
        samenstelling van twee kwartslagen om $\mathrm i$ en
        $\mathrm j$: bereken de as en de hoek van het product.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- De expliciete matrix:
Euler--Rodrigues.} Schrijf $q = a + b\mathrm i + c\mathrm j +
d\mathrm k \in S^3$, zodat $a^2 + b^2 + c^2 + d^2 = 1$.
\begin{enumerate}[resume]
  \item Bereken $\rho_q(\mathrm i)$ \omterm{def:b3:complete:complete}{volledig} met de
        vermenigvuldigingstabel; verkrijg vervolgens
        $\rho_q(\mathrm j)$ en $\rho_q(\mathrm k)$ met de
        cyclische substitutie $\mathrm i \to \mathrm j \to
        \mathrm k \to \mathrm i$, $(b, c, d) \to (c, d, b)$
        \emph{(verantwoord dat: het cyclisch verwisselen van
        $\mathrm i, \mathrm j, \mathrm k$ breidt uit tot een
        automorfisme van $\mathbb H$, omdat de definiërende
        betrekkingen cyclisch symmetrisch zijn)}. Besluit dat de
        matrix van $\rho_q$ in de basis $(\mathrm i, \mathrm j,
        \mathrm k)$ de \emph{matrix van Euler--Rodrigues} is:
        \[
        R_q = \begin{pmatrix}
        a^2 + b^2 - c^2 - d^2 & 2(bc - ad) & 2(bd + ac)\\
        2(bc + ad) & a^2 - b^2 + c^2 - d^2 & 2(cd - ab)\\
        2(bd - ac) & 2(cd + ab) & a^2 - b^2 - c^2 + d^2
        \end{pmatrix}.
        \]
  \item (Een rotatie achterstevoren lezen) Toon aan dat
        \[
        \operatorname{tr}R_q = 4a^2 - 1 = 1 + 2\cos\theta,
        \qquad
        \tfrac12\bigl(R_q - R_q^{\mathsf T}\bigr) =
        \sin\theta\,A_u,
        \]
        in de notatie van de vragen 5 en 8. Leid een algoritme af
        dat $\pm q$ uit een rotatiematrix $R$ terugwint: de hoek
        uit het spoor; de as uit het antisymmetrische deel als $0
        < \theta < \pi$; en bewijs en gebruik voor $\theta = \pi$
        de identiteit $R + I = 2\,uu^{\mathsf T}$.
  \item Evalueer $R_q$ voor het product $q = \frac12(1 + \mathrm
        i + \mathrm j + \mathrm k)$ uit vraag 11: er verschijnt
        een permutatiematrix. Identificeer de rotatie en verzoen
        dat met de as en de hoek uit vraag 11.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Binnen $S^3$: $SU(2)$,
\omterm{ex:b3:groups:actions}{conjugatieklassen}, exponentiëlen.}
\begin{enumerate}[resume]
  \item Toon aan dat de matrixvoorstelling van vraag 1 (noem haar
        $\Phi$) beperkt tot een groepsisomorfisme van $S^3$ op de
        \emph{speciale unitaire groep}
        \[
        SU(2) = \bigl\{U \in M_2(\C) : U^*U = I,\ \det U =
        1\bigr\}
        \]
        \emph{(voor de surjectiviteit: schrijf de vergelijkingen
        $U^{-1} = U^*$ en $\det U = 1$ uit voor een algemene
        complexe $2\times2$-matrix)}.
  \item Toon aan dat het reële deel een conjugatie-invariant is
        op $S^3$ --- $\operatorname{Re}(pq\bar p) =
        \operatorname{Re}q$ voor alle $p \in S^3$ --- en
        omgekeerd dat twee eenheidsquaternionen met hetzelfde
        reële deel in $S^3$ geconjugeerd zijn \emph{(herleid tot
        het verplaatsen van de ene pure eenheidsas naar de
        andere, wat Deel II levert)}. Beschrijf de
        \omterm{ex:b3:groups:actions}{conjugatieklassen} van $S^3$ meetkundig; vertaal naar
        $SU(2)$ (niveauverzamelingen van het spoor); en projecteer
        met $\rho$: twee rotaties zijn geconjugeerd in $SO(3)$
        dan en slechts dan als ze dezelfde hoek $\theta \in
        \intcc0\pi$ hebben.
  \item Definieer $\exp$ op $\mathbb H$ met de exponentiële
        reeks; ga de absolute convergentie na met $\abs{pq} =
        \abs p\,\abs q$ voor $\abs q = \sqrt{N(q)}$. Toon voor
        een pure eenheid $u$ en $\theta \in \R$ aan dat
        \[
        \exp(\theta u) = \cos\theta + \sin\theta\,u ,
        \]
        leid af dat $\exp$ het hypervlak $P$ op $S^3$ afbeeldt,
        en ga na dat $\rho_{\exp(su)} = \eu^{2sA_u}$: opnieuw het
        halvehoekverschijnsel van vraag 9.
  \item Bewijs voor pure quaternionen $v, w$ de productregel $vw
        = -\langle v, w\rangle + v\wedge w$, en daaruit de
        commutatoridentiteit $vw - wv = 2\,v\wedge w$; bewijs ook
        $[A_v, A_w] = A_{v\wedge w}$ voor de matrices van vraag
        8. Besluit dat de afgeleide van $\rho$ in $1$ langs de
        krommen $t \mapsto \exp(tv)$ het lineaire isomorfisme $v
        \mapsto 2A_v$ is van $P$ op de antisymmetrische matrices,
        en dat ze de \omterm{def:b3:groups:derived}{commutator} van de quaternionen omzet in de
        \omterm{def:b3:groups:derived}{commutator} van de matrices.
\end{enumerate}

\medskip
\noindent\textbf{Deel VII --- Globale structuur.}
\begin{enumerate}[resume]
  \item (Geen \omterm{def:b3:topology:continuity}{continue} sectie) Stel dat $s \colon SO(3) \to S^3$
        \omterm{def:b3:topology:continuity}{continu} is met $\rho \circ s = \operatorname{id}$. Stel
        voor de lus $R(t) = \rho_{q(t)}$ van vraag 10
        $\varepsilon(t) = s(R(t))\,q(t)^{-1}$ voor $t \in
        \intcc0{2\pi}$. Toon aan dat $\varepsilon$ \omterm{def:b3:topology:continuity}{continu} is met
        waarden in $\{\pm1\}$, en leid een tegenspraak af: er is
        geen \omterm{def:b3:topology:continuity}{continue} globale keuze van een eenheidsquaternion
        dat elke rotatie voorstelt.
  \item (Het balmodel) Zij $\bar B \subseteq \R^3$ de gesloten
        bal met straal $\pi$ en $E(v) = \eu^{A_v}$, met $E(0) =
        I$. Toon aan dat $E$ $\bar B$ op $SO(3)$ afbeeldt, dat ze
        injectief is op de open bal, en dat ze op de randsfeer
        precies de antipoden identificeert: $E(\pi u) = E(-\pi u)
        = 2uu^{\mathsf T} - I$, zonder andere samenvallingen.
        $SO(3)$ is dus de bal met de antipodale randpunten aan
        elkaar gelijmd --- de projectieve ruimte
        $\mathbb{RP}^3$ --- en een middellijn wordt de niet
        samentrekbare lus van vraag 10.
  \item Toon aan dat $\rho_p\rho_q\rho_p^{-1} = \rho_{pq\bar
        p}$; dat de involuties van $SO(3)$ (de $R \neq I$ met
        $R^2 = I$) precies de halve slagen $\rho_w$ zijn met $w$
        een puur eenheidsquaternion; en dat het \omterm{ex:b3:groups:actions}{centrum} van
        $SO(3)$ triviaal is.
  \item Toon aan dat elke rotatie een product van twee halve
        slagen is: kies voor $q = \cos\frac\theta2 +
        \sin\frac\theta2\,u$ een pure eenheid $w \perp u$, ga na
        dat $w' = qw$ opnieuw een puur eenheidsquaternion is, en
        controleer $\rho_q = \rho_{w'}\rho_w$. Waar liggen de
        twee assen, en welke hoek maken ze?
  \item Besluit met de topologische samenvatting: $SO(3)$ is
        \omterm{def:b3:topology:compact}{compact} en \omterm{def:b3:topology:pathconnected}{wegsamenhangend} (geef twee bewijzen: het
        \omterm{def:b3:topology:continuity}{continue} beeld van $S^3$ onder $\rho$; het beeld van
        $\exp$), terwijl $O(3)$ precies twee \omterm{def:b3:topology:components}{samenhangscomponenten}
        heeft, elk \omterm{def:b3:topology:continuity}{homeomorf} met $SO(3)$.

  \item (Eén samenstelling, op drie manieren) Zij $R_1$ de
        rotatie over $\frac\pi2$ om de $z$-as en $R_2$ de rotatie
        over $\frac\pi2$ om de $x$-as. Bereken de as en de hoek
        van $R_2R_1$: (i) door de twee $3\times3$-matrices te
        vermenigvuldigen en het spoor en het antisymmetrische
        deel te gebruiken (Deel V); (ii) door de bijbehorende
        eenheidsquaternionen $q_2q_1$ te vermenigvuldigen.
        Controleer dat de twee antwoorden overeenstemmen: hoek
        $\frac{2\pi}3$, as $\frac1{\sqrt3}(1, -1, 1)$.
  \item (De \omterm{ex:b3:conformal:mobius}{cayleytransformatie}) Toon voor antisymmetrische $K$
        aan dat $I + K$ inverteerbaar is en dat
        \[
        C(K) = (I - K)(I + K)^{-1} \in SO(n),
        \]
        waarbij $-1$ nooit een eigenwaarde van $C(K)$ is; toon
        aan dat $K \mapsto C(K)$ een bijectie is van de
        antisymmetrische matrices op $\{R \in SO(n) : -1 \notin
        \operatorname{Sp}R\}$, met inverse $R \mapsto (I -
        R)(I + R)^{-1}$. (Een rationale kaart van $SO(n)$,
        metgezel van de transcendente $\exp$ uit
        \cref{exo:b3:submanifolds:11}.)
\end{enumerate}
\end{problem}
