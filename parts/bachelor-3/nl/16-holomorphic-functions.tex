\chapter{Holomorfe functies}\label{ch:b3:holomorphic}

Complexe differentieerbaarheid lijkt een kleine variatie op de
reële theorie --- één limiet, één quotiënt. In werkelijkheid is
het een ander heelal. Een functie die op een open deel van $\C$
\emph{één keer} differentieerbaar is, is automatisch oneindig
vaak differentieerbaar, analytisch, op een heel gebied bepaald
door haar waarden nabij één enkel punt, en gebonden aan starre
globale beginselen (Liouville, maximum van de modulus). Dat alles
vloeit voort uit één wonder, de stelling van Cauchy: de integraal
van een \omterm{def:b3:holomorphic:holo}{holomorfe functie} langs een gesloten weg in een stervormig
gebied verdwijnt. Dit hoofdstuk bewijst dat wonder (het argument
van Goursat, zonder enige regulariteit boven op de
differentieerbaarheid aan te nemen), oogst haar klassieke
gevolgen, en vereffent een stelling die dit boek sinds
\cref{ch:b3:galois} op krediet gebruikt: elke niet-constante
complexe veelterm heeft een wortel.

Overal is $\Omega \subseteq \C$ open, en duidt $D(a, r)$ de open
schijf aan.

\section{Complexe differentieerbaarheid}

\begin{definition}\label{def:b3:holomorphic:holo}
$f \colon \Omega \to \C$ heet \emph{holomorf}\index{holomorfe
functie} op $\Omega$ als voor elke $z_0 \in \Omega$
\[
f'(z_0) = \lim_{h\to0}\frac{f(z_0 + h) - f(z_0)}{h}
\]
bestaat ($h \in \C^*$). Sommen, producten, quotiënten (met
noemers $\neq 0$) en samenstellingen van holomorfe functies zijn
holomorf, met de gebruikelijke formules (de bewijzen uit de
volumes van bachelorjaar 1 en 2 gaan woordelijk door: ze
gebruiken enkel lichaamsbewerkingen en limieten). $\mathcal
H(\Omega)$ duidt de verzameling holomorfe functies op $\Omega$
aan.
\end{definition}

\begin{proposition}[Cauchy--Riemann]
\label{prop:b3:holomorphic:cauchyriemann}
Schrijf $f(x + \iu y) = P(x,y) + \iu Q(x,y)$. Dan is $f$
\omterm{def:b3:holomorphic:holo}{holomorf} in $z_0$ dan en slechts dan als $f$ in $z_0$
$\R$-differentieerbaar is (als afbeelding van twee reële
veranderlijken) en\index{vergelijkingen van Cauchy--Riemann}
\[
\frac{\partial P}{\partial x} = \frac{\partial Q}{\partial y},
\qquad
\frac{\partial P}{\partial y} = -\frac{\partial Q}{\partial x}
\qquad \text{in } z_0 ;
\]
gelijkwaardig: de reële differentiaal is de vermenigvuldiging met
het complexe getal $f'(z_0)$.
\end{proposition}

\begin{proof}
$\C$-differentieerbaarheid zegt $f(z_0 + h) = f(z_0) + ch +
o(\abs h)$ met $c = f'(z_0)$: een $\R$-lineaire differentiaal die
de vermenigvuldiging met $c = a + \iu b$ is, dus met matrix
$\bigl(\begin{smallmatrix} a & -b\\ b & a
\end{smallmatrix}\bigr)$ in de basis $(1, \iu)$ --- precies de
weergegeven betrekkingen voor de partiële afgeleiden. Omgekeerd
is zo'n differentiaal $\C$-lineair, en de definities met $o(\abs
h)$ komen overeen.
\end{proof}

\begin{example}\label{ex:b3:holomorphic:examples}
Veeltermen in $z$, rationale functies buiten hun polen, en ---
volgens de stelling over term voor term differentiëren van
machtreeksen uit het volume van bachelorjaar 2, waarvan het
bewijs over $\C$ identiek werkt --- elke som van een machtreeks
$\sum a_n(z - a)^n$ binnen haar convergentieschijf: \omterm{def:b3:holomorphic:holo}{holomorf}, met
afgeleide $\sum na_n(z - a)^{n-1}$ (dezelfde straal). In het
bijzonder is $\exp z = \sum z^n/n!$ geheel (\omterm{def:b3:holomorphic:holo}{holomorf} op $\C$) met
$\exp' = \exp$. Anderzijds zijn $z \mapsto \bar z$, $\abs z$ en
$\operatorname{Re}z$ nergens \omterm{def:b3:holomorphic:holo}{holomorf} (Cauchy--Riemann faalt
overal): \omterm{def:b3:holomorphic:holo}{holomorfie} is starheid die oriëntatie en hoeken behoudt,
geen gladheid.
\end{example}

\section{Contourintegralen}

\begin{definition}\label{def:b3:holomorphic:contour}
Een \emph{weg} is een stuksgewijs $\mathcal C^1$-afbeelding
$\gamma \colon \intcc ab \to \C$; hij heet \emph{gesloten} als
$\gamma(a) = \gamma(b)$. Voor \omterm{def:b3:topology:continuity}{continue} $f$ op het beeld van
$\gamma$:\index{contourintegraal}
\[
\int_\gamma f(z)\,\dd z =
\int_a^b f(\gamma(t))\,\gamma'(t)\,\dd t,
\qquad
\Bigl|\int_\gamma f\,\dd z\Bigr| \leq
\sup_{\gamma}\abs f\cdot\operatorname{length}(\gamma)
\]
(de \emph{ML-ongelijkheid}; de lengte is $\int_a^b
\abs{\gamma'}$). De integraal is invariant onder een stijgende
$\mathcal C^1$-herparametrisering en wisselt van teken bij
omkering van de oriëntatie.
\end{definition}

\begin{proposition}[Primitieven]\label{prop:b3:holomorphic:primitive}
Voor \omterm{def:b3:topology:continuity}{continue} $f$ op $\Omega$ zijn de volgende uitspraken
gelijkwaardig: (i) $f$ heeft een \emph{primitieve} $F \in
\mathcal H(\Omega)$ ($F' = f$); (ii) $\int_\gamma f\,\dd z = 0$
voor elke gesloten weg $\gamma$ in $\Omega$. In dat geval is
$\int_\gamma f\,\dd z = F(\gamma(b)) - F(\gamma(a))$ voor elke
weg.
\end{proposition}

\begin{proof}
(i)$\Rightarrow$: $\frac{\dd}{\dd t}F(\gamma(t)) =
F'(\gamma(t))\gamma'(t)$ (kettingregel, stuksgewijs geldig), dus
telescopeert de integraal tot het verschil van de eindpunten;
gesloten wegen geven $0$. (ii)$\Rightarrow$(i): leg $z_*$ in een
\omterm{def:b3:topology:components}{samenhangscomponent} vast en definieer $F(z) = \int_{\gamma_z}f\,
\dd z$ langs een willekeurige weg van $z_*$ naar $z$ (goed
gedefinieerd: twee wegen verschillen een gesloten weg); voor
kleine $h$ is, met het lijnstuk van $z$ naar $z + h$,
\[
\frac{F(z + h) - F(z)}{h} - f(z)
= \frac1h\int_{[z, z+h]}\bigl(f(w) - f(z)\bigr)\dd w
\longrightarrow 0
\]
volgens de ML-ongelijkheid en de \omterm{def:b3:topology:continuity}{continuïteit} van $f$ in $z$.
\end{proof}

\begin{definition}[Windingsgetal]\label{def:b3:holomorphic:index}
Voor een gesloten weg $\gamma$ en $z \notin
\operatorname{im}\gamma$ is de \emph{index}\index{windingsgetal}
\index{index van een weg}
\[
\operatorname{Ind}_\gamma(z) = \frac1{2\iu\pi}
\int_\gamma\frac{\dd w}{w - z} .
\]
Het is een \emph{geheel getal}: stel $\varphi(t) =
\int_a^t\frac{\gamma'(s)}{\gamma(s) - z}\dd s$, dan heeft de
functie $(\gamma(t) - z)\eu^{-\varphi(t)}$ (stuksgewijs)
afgeleide nul en is ze dus constant; in $t = b$ is
$\eu^{\varphi(b)} = \frac{\gamma(b) - z}{\gamma(a) - z} = 1$, dus
$\varphi(b) \in 2\iu\pi\Z$. Als functie van $z$ is de index
\omterm{def:b3:topology:continuity}{continu} op $\C\setminus\operatorname{im}\gamma$ (gedomineerde
convergentie), dus constant op elke \omterm{def:b3:topology:components}{samenhangscomponent}, en $0$
op de onbegrensde component (ML: de integraal gaat naar $0$ als
$z \to \infty$). Voor de cirkel $\gamma(t) = a + r\eu^{\iu t}$,
$t \in \intcc0{2\pi}$, is $\operatorname{Ind}_\gamma(z) = 1$ voor
$z \in D(a,r)$ (bereken in $z = a$: $\frac1{2\iu\pi}
\int_0^{2\pi}\frac{r\iu\eu^{\iu t}}{r\eu^{\iu t}}\dd t = 1$; de
constantheid doet de rest).
\end{definition}

\section{De stelling van Cauchy}

\begin{theorem}[Goursat]\label{thm:b3:holomorphic:goursat}
Zij $f \in \mathcal H(\Omega)$ en $T \subseteq \Omega$ een
gesloten volle driehoek. Dan is $\int_{\partial T}f\,\dd z = 0$
(de rand eenmaal doorlopen, in willekeurige
oriëntatie).\index{stelling van Goursat}
\end{theorem}

\begin{proof}
Zij $I(T) = \int_{\partial T}f\,\dd z$. De middens van de zijden
verbinden splitst $T$ in vier half zo grote driehoeken
$T^{(1)}, \dots, T^{(4)}$, en de binnenste zijden vallen
paarsgewijs weg: $I(T) = \sum_iI(T^{(i)})$. Kies daaronder een
$T_1$ met $\abs{I(T_1)} \geq \frac14\abs{I(T)}$ en itereer: een
genestelde rij $T \supseteq T_1 \supseteq T_2 \supseteq\cdots$
met
\[
\abs{I(T_n)} \geq 4^{-n}\abs{I(T)},
\qquad
\operatorname{diam}T_n = 2^{-n}\operatorname{diam}T,
\quad
\operatorname{length}(\partial T_n) =
2^{-n}\operatorname{length}(\partial T).
\]
De doorsnede $\bigcap T_n$ is één enkel punt $z_0$ (genestelde
\omterm{def:b3:topology:compact}{compacte} verzamelingen met diameters naar nul,
\cref{thm:b3:topology:compactprops}(3)). Differentieerbaarheid in
$z_0$: bij gegeven $\varepsilon$ is voor grote $n$, op $T_n$,
\[
f(z) = f(z_0) + f'(z_0)(z - z_0) + R(z),
\qquad \abs{R(z)} \leq \varepsilon\abs{z - z_0} \leq
\varepsilon\operatorname{diam}T_n .
\]
Het affiene deel heeft een primitieve: zijn integraal over de
gesloten $\partial T_n$ verdwijnt
(\cref{prop:b3:holomorphic:primitive}), zodat
\[
\abs{I(T_n)} = \Bigl|\int_{\partial T_n}R\Bigr|
\leq \varepsilon\operatorname{diam}(T_n)\,
\operatorname{length}(\partial T_n)
= \varepsilon\,4^{-n}\operatorname{diam}(T)
\operatorname{length}(\partial T) .
\]
Vergelijken met $\abs{I(T_n)} \geq 4^{-n}\abs{I(T)}$ geeft
$\abs{I(T)} \leq \varepsilon\cdot\text{constante}$ voor elke
$\varepsilon$: dus $I(T) = 0$.
\end{proof}

\begin{theorem}[Stelling van Cauchy, stervormige versie]
\label{thm:b3:holomorphic:cauchy}
Zij $\Omega$ \emph{stervormig} ten opzichte van $c$ (elk lijnstuk
$[c, z]$ met $z \in \Omega$ ligt in $\Omega$) --- bijvoorbeeld
convex. Elke $f \in \mathcal H(\Omega)$ heeft een primitieve op
$\Omega$; bijgevolg is $\int_\gamma f\,\dd z = 0$ voor
\emph{elke} gesloten weg $\gamma$ in $\Omega$.\index{stelling van
Cauchy}
\end{theorem}

\begin{proof}
Definieer $F(z) = \int_{[c,z]}f\,\dd w$. Voor $z, z + h \in
\Omega$ met $[z, z+h] \subseteq \Omega$ (waar voor kleine $h$)
ligt de driehoek met hoekpunten $c, z, z+h$ in $\Omega$ (de
stervormigheid: elk van zijn punten ligt op een lijnstuk $[c,
w]$ met $w \in [z, z+h] \subseteq \Omega$): Goursat geeft
\[
F(z + h) - F(z) = \int_{[z, z+h]}f\,\dd w,
\]
en de berekening met het differentiequotiënt uit
\cref{prop:b3:holomorphic:primitive} levert $F' = f$. Dat de
integralen over gesloten wegen verdwijnen, volgt uit dezelfde
propositie.
\end{proof}

\begin{theorem}[Integraalformule van Cauchy]
\label{thm:b3:holomorphic:formula}
Zij $f \in \mathcal H(\Omega)$, $\bar D(a, r) \subseteq \Omega$
en $C_r$ de cirkel $\partial D(a,r)$, eenmaal tegen de wijzers
van de klok in doorlopen. Dan is voor elke $z \in D(a,
r)$:\index{integraalformule van Cauchy}
\[
f(z) = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - z}\,\dd w .
\]
\end{theorem}

\begin{proof}
Leg $z$ vast en definieer op $\Omega$
\[
g(w) = \begin{cases}
\dfrac{f(w) - f(z)}{w - z} & w \neq z,\\[4pt]
f'(z) & w = z :
\end{cases}
\]
$g$ is \omterm{def:b3:topology:continuity}{continu} op $\Omega$ en \omterm{def:b3:holomorphic:holo}{holomorf} buiten $z$. Goursat geldt
voor $g$ op elke driehoek $T \subseteq \Omega'$, waarbij
$\Omega'$ een schijf is die iets groter is dan $\bar D(a,r)$ en
in $\Omega$ ligt: is $z \notin T$, rechtstreeks; is $z \in T$,
splits $T$ dan in kleine driehoeken met $z$ als hoekpunt plus
driehoeken die $z$ mijden; op een driehoek met hoekpunt $z$ geeft
de ML-afschatting $\abs{\int_{\partial T'}g} \leq \sup_{T'}\abs
g\cdot\operatorname{length} \to 0$ terwijl de driehoek krimpt, en
de overige stukken verdwijnen volgens Goursat --- dus is
$\int_{\partial T}g = 0$ in alle gevallen. Het bewijs van
\cref{thm:b3:holomorphic:cauchy} gebruikte enkel die
driehoekseigenschap: $g$ heeft dus een primitieve op de convexe
$\Omega'$, zodat $\int_{C_r}g = 0$, dat wil zeggen
\[
\frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - z}\dd w
= f(z)\,\frac{1}{2\iu\pi}\int_{C_r}\frac{\dd w}{w - z}
= f(z)\operatorname{Ind}_{C_r}(z) = f(z) .
\]
\end{proof}

\section{Analyticiteit en haar waterval}

\begin{theorem}[Holomorf $=$ analytisch]
\label{thm:b3:holomorphic:analytic}
Zij $f \in \mathcal H(\Omega)$ en $D(a, R) \subseteq \Omega$. Dan
is\index{analyticiteit van holomorfe functies}
\[
f(z) = \sum_{n\geq0}c_n\,(z - a)^n \quad \text{op } D(a, R),
\qquad
c_n = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{(w -
a)^{n+1}}\,\dd w \ \ (0 < r < R),
\]
waarbij de coëfficiënten niet van $r$ afhangen. Bijgevolg is $f$
oneindig vaak $\C$-differentieerbaar, is $c_n = f^{(n)}(a)/n!$,
en gelden de \emph{afschattingen van
Cauchy}\index{afschattingen van Cauchy}:
\[
\abs{c_n} \;\leq\; \frac{\sup_{\abs{w - a} =
r}\abs{f(w)}}{r^{n}} .
\]
\end{theorem}

\begin{proof}
Voor $\abs{z - a} < r$: ontwikkel de kern van Cauchy in de
meetkundige reeks
\[
\frac1{w - z} = \frac1{(w - a)\bigl(1 - \frac{z - a}{w -
a}\bigr)} = \sum_{n\geq0}\frac{(z - a)^n}{(w - a)^{n+1}},
\]
normaal convergent in $w$ op $C_r$ (want $\abs{\frac{z-a}{w-a}} =
\frac{\abs{z-a}}r < 1$): integreer term voor term tegen
$\frac{f(w)}{2\iu\pi}$ (de uniforme convergentie rechtvaardigt
het verwisselen) en pas \cref{thm:b3:holomorphic:formula} toe.
Een machtreeks is oneindig vaak differentieerbaar met $c_n =
f^{(n)}(a)/n!$ (volume van bachelorjaar 2), wat ook toont dat de
$c_n$ niet van $r$ afhangen. De afschattingen: begrens de
integraal voor de coëfficiënt met ML.
\end{proof}

\begin{example}[Singulariteiten dicteren stralen]
\label{ex:b3:holomorphic:radius}
Waarom heeft de onschuldige reële functie $\frac1{1 + x^2}$ in $x
= 3$ een taylorreeks die enkel convergeert voor $\abs{x - 3} <
\sqrt{10}$, terwijl er op de reële lijn niets misgaat? Omdat de
bovenstaande stelling de convergentiestraal in $a$ gelijkmaakt
aan de afstand van $a$ tot het dichtstbijzijnde punt waar de
\omterm{def:b3:holomorphic:holo}{holomorfie} faalt. Hier is $f(z) = \frac1{1 + z^2}$ precies
\omterm{def:b3:holomorphic:holo}{holomorf} op $\C\setminus\{\pm\iu\}$, dus convergeert de
ontwikkeling in $a = 3$ op de grootste schijf die $\pm\iu$
mijdt, met straal $\abs{3 - \iu} = \sqrt{10}$ --- en ze kan niet
op een grotere convergeren, want de som zou $f$ dan \omterm{def:b3:holomorphic:holo}{holomorf}
uitbreiden tot een \omterm{def:b3:topology:topology}{omgeving} van $\pm\iu$, waar $\abs f \to
\infty$. De reële theorie ziet de raadselachtige straal
$\sqrt{10}$; het complexe vlak ziet twee polen. Dat is de
praktische regel: \emph{om een convergentiestraal te vinden,
lokaliseer de singulariteiten} --- zo heeft de taylorreeks van
$\tan$ in $0$ straal $\frac\pi2$ (de dichtstbijzijnde nulpunten
van $\cos$), en de genererende functie van Bernoulli $\frac
z{\eu^z - 1}$ (\cref{pb:b3:holomorphic:1}, Deel VI) straal $2\pi$
(de dichtstbijzijnde nulpunten $\neq 0$ van $\eu^z - 1$:
$\pm2\iu\pi$).
\end{example}

\begin{corollary}[Liouville; d'Alembert--Gauss]
\label{cor:b3:holomorphic:liouville}
Een begrensde gehele functie is constant. Bijgevolg heeft elke
niet-constante veelterm over $\C$ een wortel: \emph{$\C$ is
\omterm{def:b3:galois:algebraic}{algebraïsch} afgesloten}.\index{stelling van
Liouville}\index{hoofdstelling van de algebra}
\end{corollary}

\begin{proof}
Is $\abs f \leq M$ op $\C$, dan is voor elke $a$ en $r$
$\abs{c_1(a)} = \abs{f'(a)} \leq M/r \to 0$: dus $f' \equiv 0$,
en $f$ is constant (op het \omterm{def:b3:topology:connected}{samenhangende} $\C$: een afgeleide nul
impliceert lokaal constant --- integreer langs lijnstukken). Had
$P$ geen wortel, dan zou $1/P$ geheel en begrensd zijn ($\abs{P(z)}
\to \infty$ als $\abs z \to \infty$: de kopterm domineert, dus is
$\abs{1/P}$ klein buiten een grote schijf en \omterm{def:b3:topology:continuity}{continu} op de
\omterm{def:b3:topology:compact}{compacte} schijf): constant --- absurd voor een niet-constante
$P$. (De weekendopgave geeft een tweede, elementair bewijs en de
\omterm{def:b3:galois:algebraic}{algebraïsche} gevolgen.)
\end{proof}

\begin{theorem}[Nulpunten zijn geïsoleerd; identiteitsstelling]
\label{thm:b3:holomorphic:identity}
Zij $\Omega$ \emph{\omterm{def:b3:topology:connected}{samenhangend}} en $f \in \mathcal H(\Omega)$
met $f \not\equiv 0$. Dan heeft elk nulpunt $a$ van $f$ een
eindige \emph{orde}: $f(z) = (z - a)^m\,g(z)$ met $g \in \mathcal
H(\Omega)$ en $g(a) \neq 0$, en de nulpunten van $f$ hebben geen
ophopingspunt in $\Omega$. Bijgevolg vallen twee \omterm{def:b3:holomorphic:holo}{holomorfe
functies} op $\Omega$ die samenvallen op een verzameling met een
ophopingspunt in $\Omega$, overal
samen.\index{identiteitsstelling}\index{geïsoleerde nulpunten}
\end{theorem}

\begin{proof}
Zij $Z$ de verzameling punten waar alle afgeleiden van $f$
verdwijnen. $Z$ is gesloten (doorsnede van gesloten
verzamelingen) en open: zijn alle $c_n$ in $a$ nul, dan maakt de
machtreeksontwikkeling $f \equiv 0$ op een schijf rond $a$.
\omterm{def:b3:topology:connected}{Samenhang}: $Z = \varnothing$ of $Z = \Omega$; het tweede is
uitgesloten door $f \not\equiv 0$. In een nulpunt $a$ is dus een
coëfficiënt $\neq 0$: zij $m$ minimaal met $c_m \neq 0$; dan is
$f(z) = (z - a)^m\sum_{n\geq m}c_n(z-a)^{n-m}$ op een schijf, en
de som definieert een $g$ die \omterm{def:b3:holomorphic:holo}{holomorf} is nabij $a$ met $g(a) =
c_m \neq 0$; breid $g = f/(z-a)^m$ uit buiten $a$ (daar
\omterm{def:b3:holomorphic:holo}{holomorf}). Omdat $g(a) \neq 0$ en $g$ \omterm{def:b3:topology:continuity}{continu} is, heeft $f$ geen
ander nulpunt in een \omterm{def:b3:topology:topology}{omgeving} van $a$: de nulpunten zijn
geïsoleerd, en een verzameling geïsoleerde punten heeft geen
ophopingspunt in $\Omega$ (een ophopingspunt van nulpunten is een
nulpunt --- \omterm{def:b3:topology:continuity}{continuïteit} --- en zou niet geïsoleerd zijn).
Identiteit: pas dit toe op het verschil, waarvan de
nulpuntenverzameling een ophopingspunt heeft, wat het in de tak
$Z = \Omega$ dwingt.
\end{proof}

\begin{theorem}[Gemiddelde waarde en maximum van de modulus]
\label{thm:b3:holomorphic:maximum}
Zij $f \in \mathcal H(\Omega)$.
\begin{enumerate}
  \item (Gemiddelde waarde) Voor $\bar D(a, r) \subseteq \Omega$:
        $f(a) = \frac1{2\pi}\int_0^{2\pi}f(a + r\eu^{\iu t})\,\dd
        t$.
  \item (Maximumprincipe) Is $\Omega$ \omterm{def:b3:topology:connected}{samenhangend} en bereikt
        $\abs f$ in een punt van $\Omega$ een lokaal maximum, dan
        is $f$ constant. Bijgevolg is voor begrensde $\Omega$ en
        $f$ \omterm{def:b3:topology:continuity}{continu} op $\bar\Omega$ $\sup_{\bar\Omega}\abs f =
        \sup_{\partial\Omega}\abs f$.\index{maximumprincipe}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) is de formule van Cauchy in het middelpunt: parametriseer
$C_r$.
(2) Stel $\abs f \leq \abs{f(a)}$ op $\bar D(a, \rho) \subseteq
\Omega$. Is $f(a) = 0$, dan is $f \equiv 0$ nabij $a$. Anders
geeft voor $0 < r \leq \rho$ de gemiddelde waarde
\[
\abs{f(a)} \leq \frac1{2\pi}\int_0^{2\pi}\abs{f(a +
r\eu^{\iu t})}\,\dd t \leq \abs{f(a)} :
\]
de \omterm{def:b3:topology:continuity}{continue} niet-negatieve integrand $\abs{f(a)} - \abs{f(a +
r\eu^{\iu t})}$ heeft gemiddelde nul en verdwijnt dus: $\abs f$
is \emph{constant} $= \abs{f(a)} \ne 0$ op de schijf. Een
\omterm{def:b3:holomorphic:holo}{holomorfe functie} met constante modulus $\neq 0$ op een schijf is
constant: $P^2 + Q^2 = \text{constante}$ differentiëren geeft
$PP_x + QQ_x = 0$ en $PP_y + QQ_y = 0$; de betrekkingen van
Cauchy--Riemann $P_y = -Q_x$, $Q_y = P_x$ invoeren in de tweede
vergelijking levert het lineaire stelsel
\[
P\,P_x + Q\,Q_x = 0, \qquad -P\,Q_x + Q\,P_x = 0,
\]
waarvan de determinant $P^2 + Q^2 \neq 0$ is: dus $P_x = Q_x =
0$, zodat $f' = P_x + \iu Q_x = 0$ op de schijf en $f$ daar
constant is. De identiteitsstelling verspreidt de constantheid
over heel $\Omega$. De randvorm: $\abs f$ bereikt haar supremum
op de \omterm{def:b3:topology:compact}{compacte} $\bar\Omega$; een inwendig maximum maakt $f$
constant, en het supremum wordt in elk geval op de rand bereikt.
\end{proof}

\begin{theorem}[Convergentiestelling van Weierstrass]
\label{thm:b3:holomorphic:weierstrassconv}
Convergeren $f_n \in \mathcal H(\Omega)$ naar $f$ uniform op elke
\omterm{def:b3:topology:compact}{compacte} deelverzameling van $\Omega$, dan is $f \in \mathcal
H(\Omega)$ en convergeert $f_n^{(k)} \to f^{(k)}$ uniform op
\omterm{def:b3:topology:compact}{compacte} verzamelingen, voor elke
$k$.\index{convergentiestelling van Weierstrass}
\end{theorem}

\begin{proof}
$f$ is \omterm{def:b3:topology:continuity}{continu}. Voor elke gesloten driehoek $T \subseteq \Omega$
is $\int_{\partial T}f = \lim\int_{\partial T}f_n = 0$ (uniforme
convergentie op de \omterm{def:b3:topology:compact}{compacte} $\partial T$; Goursat voor $f_n$).
Volgens het argument van \cref{thm:b3:holomorphic:cauchy} heeft
$f$ lokale primitieven (schijven zijn convex; enkel de
driehoekseigenschap werd gebruikt), dat wil zeggen $f = F'$
lokaal met $F$ \omterm{def:b3:holomorphic:holo}{holomorf}; $F$ is analytisch
(\cref{thm:b3:holomorphic:analytic}), en dus ook $f = F'$:
\omterm{def:b3:holomorphic:holo}{holomorf}. (Dat is de \emph{stelling van Morera}\index{stelling
van Morera}: \omterm{def:b3:topology:continuity}{continu} met verdwijnende driehoeksintegralen
impliceert \omterm{def:b3:holomorphic:holo}{holomorf}.) Afgeleiden: voor $\bar D(a, 2r) \subseteq
\Omega$ en $z \in \bar D(a, r)$ geeft de formule van Cauchy voor
afgeleiden (differentieer \cref{thm:b3:holomorphic:formula} onder
de integraal, of gebruik de coëfficiëntformule)
\[
\abs{f_n'(z) - f'(z)} =
\Bigl|\frac{1}{2\iu\pi}\int_{C_{2r}}\frac{f_n(w) - f(w)}{(w -
z)^2}\,\dd w\Bigr|
\leq \frac{2r\,\sup_{C_{2r}}\abs{f_n - f}}{r^2} \to 0
\]
uniform op $\bar D(a,r)$; overdek een \omterm{def:b3:topology:compact}{compacte} verzameling met
eindig veel zulke schijven en itereer voor hogere $k$.
\end{proof}

\begin{method}\label{met:b3:holomorphic:toolkit}
De dagelijkse gereedschapskist. Om aan te tonen dat een functie
\omterm{def:b3:holomorphic:holo}{holomorf} is: geef haar als machtreeks, als samenstelling, als
lokaal uniforme limiet
(\cref{thm:b3:holomorphic:weierstrassconv}) of als integraal met
\omterm{def:b3:holomorphic:holo}{holomorfe} parameter (\cref{exo:b3:holomorphic:7} ---
differentieer onder $\int$ of pas Morera en Fubini toe). Om
identiteiten te bewijzen: bewijs ze op een lijnstuk of
deelgebied en roep de identiteitsstelling in. Om af te schatten:
de \omterm{thm:b3:holomorphic:analytic}{afschattingen van Cauchy} op de grootste beschikbare cirkel. Om
constantheid of niet-bestaan te bewijzen: Liouville of het
maximumprincipe. Weet altijd \emph{waar} je functie \omterm{def:b3:holomorphic:holo}{holomorf} is
en \emph{welke} schijven in $\Omega$ passen.
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b3:holomorphic:1}
(a) In welke punten zijn $z \mapsto \bar z$, $\abs z^2$ en
$\operatorname{Re}z$ complex differentieerbaar? En \omterm{def:b3:holomorphic:holo}{holomorf} op
een \omterm{def:b3:topology:topology}{open verzameling}?
(b) Toon aan dat $P(x, y) = x^2 - y^2$ het reële deel is van een
\omterm{def:b3:holomorphic:holo}{holomorfe functie} op $\C$, geef er expliciet een, en bepaal ze
\emph{alle}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:holomorphic:2}
Bereken vanuit de definities: $\int_{C}z^n\,\dd z$ voor alle $n
\in \Z$, met $C$ de eenheidscirkel; $\int_\gamma\bar z\,\dd z$
langs het lijnstuk $[0, 1+\iu]$ en langs de weg van twee
lijnstukken door $1$: besluit dat $\bar z$ op geen enkele
\omterm{def:b3:topology:topology}{omgeving} van deze wegen een primitieve heeft.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:3}
(a) Toon aan dat de hoofdlogaritme $\log z = \ln\abs z + \iu\arg
z$ ($\arg \in \intoo{-\pi}\pi$) \omterm{def:b3:holomorphic:holo}{holomorf} is op
$\C\setminus\intoc{-\infty}0$ met afgeleide $\frac1z$
\emph{(primitieve van $\frac1z$ op het stervormige gesneden vlak:
\cref{thm:b3:holomorphic:cauchy}; leg de constante vast)}.
(b) Toon aan dat er op $\C^*$ geen \omterm{def:b3:topology:continuity}{continue} logaritme bestaat
\emph{(de obstructie zonder afgeleiden: de \omterm{def:b3:holomorphic:index}{index} van de
eenheidscirkel)}.
(c) Ontwikkel $\log(1 + z)$ in een machtreeks op $D(0,1)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:4}
(a) Zij $f$ geheel met $\abs{f(z)} \leq C(1 + \abs z)^{n}$. Toon
aan dat $f$ een veelterm van graad $\leq n$ is \emph{(de
\omterm{thm:b3:holomorphic:analytic}{afschattingen van Cauchy} op grote cirkels)}.
(b) Zij $f$ geheel met $\operatorname{Re}f$ naar boven begrensd.
Toon aan dat $f$ constant is \emph{(beschouw $\eu^{f}$)}.
(c) Leid ``de kleine Picard voor affiene afbeeldingen'' af: een
gehele functie die een halfvlak overslaat, is constant.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:5}
(a) Zij $f$ \omterm{def:b3:holomorphic:holo}{holomorf} op een \omterm{def:b3:topology:connected}{samenhangende} $\Omega \ni 0$ met
$f(\frac1n) = \frac1{n^2}$ voor alle grote $n$. Bepaal $f$.
(b) Bestaat er een \omterm{def:b3:holomorphic:holo}{holomorfe} $f$ op $\C^*$ met $f(\frac1n) =
\frac{(-1)^n}{n}$ voor alle $n \geq 1$? Verantwoord.
(c) Geef twee verschillende \omterm{def:b3:holomorphic:holo}{holomorfe functies} op $D(0,1)\sqcup
D(3,1)$ die op $D(0,1)$ samenvallen: waar gebruikt de
identiteitsstelling de \omterm{def:b3:topology:connected}{samenhang}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:6}
Zij $f$ \omterm{def:b3:holomorphic:holo}{holomorf} op de open eenheidsschijf $\mathbb D$, \omterm{def:b3:topology:continuity}{continu}
op $\bar{\mathbb D}$, met $\abs f \equiv 1$ op de randcirkel.
(a) Heeft $f$ geen nulpunt in $\mathbb D$, toon dan aan dat $f$
constant is \emph{(pas het maximumprincipe toe op $f$ en op
$1/f$)}.
(b) Geef een voorbeeld met een nulpunt waarin $f$ niet constant
is.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:7}
(\omterm{def:b3:holomorphic:holo}{Holomorfie} onder de integraal) Zij $\mu$ een eindige \omterm{def:b3:measure:measure}{maat} op een
ruimte $X$ en $g \colon X\times\Omega \to \C$ zodanig dat $g(x,
\cdot) \in \mathcal H(\Omega)$ voor elke $x$, $g$ \omterm{def:b3:lebesgue:measurable}{meetbaar} is in
$x$, en $\abs g \leq h(x)$ met $h$ \omterm{def:b3:lebesgue:l1}{integreerbaar}, lokaal uniform
in $z$. Toon aan dat $G(z) = \int_Xg(x, z)\dd\mu(x)$ \omterm{def:b3:holomorphic:holo}{holomorf} is
op $\Omega$. \emph{(Morera: de driehoeksintegralen verdwijnen
volgens Fubini en Goursat; de \omterm{def:b3:topology:continuity}{continuïteit} volgens de gedomineerde
convergentie. Pas dat daarna toe op $\Gamma(z) = \int_0^\infty
t^{z-1}\eu^{-t}\dd t$ op $\{\operatorname{Re}z > 0\}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:8}
(Gauss--Lucas) Zij $P \in \C[X]$ niet-constant. Toon aan dat elke
wortel van $P'$ in het convexe omhulsel van de wortels van $P$
ligt. \emph{(Schrijf $\frac{P'}{P} = \sum_k\frac{m_k}{z - a_k}$
in een wortel $z$ van $P'$ die geen wortel van $P$ is, neem
toegevoegden, en lees er een convexe combinatie uit af.)}
Illustreer op $P = z^3 - 1$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:9}
Zij $f$ geheel en dubbelperiodiek: $f(z + 1) = f(z + \iu) = f(z)$
voor alle $z$. Toon aan dat $f$ constant is. \emph{(Begrens $f$
op het \omterm{def:b3:topology:compact}{compacte} fundamentaalvierkant, dus overal; Liouville.)}
Moraal: niet-constante elliptische functies moeten polen hebben
--- het thema van \cref{ch:b3:residues}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:10}
(a) Toon aan dat $P = \operatorname{Re}f$ van een \omterm{def:b3:holomorphic:holo}{holomorfe} $f$
voldoet aan de eigenschap van de gemiddelde waarde $P(a) =
\frac1{2\pi}\int_0^{2\pi}P(a + r\eu^{\iu t})\dd t$ en
\emph{harmonisch} is: $\partial^2_{xx}P + \partial^2_{yy}P = 0$
(differentieer Cauchy--Riemann; gebruik
\cref{thm:b3:holomorphic:analytic} voor de nodige gladheid).
(b) Leid het maximumprincipe af voor reële delen van \omterm{def:b3:holomorphic:holo}{holomorfe
functies} op begrensde gebieden.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:11}
(Spiegeling van Schwarz) Zij $\Omega^+ = \{z : \abs z < 1,\
\operatorname{Im}z > 0\}$, $I = \intoo{-1}1$, en $f$ \omterm{def:b3:holomorphic:holo}{holomorf} op
$\Omega^+$, \omterm{def:b3:topology:continuity}{continu} op $\Omega^+\cup I$ en \emph{reëelwaardig op
$I$}. Definieer
\[
F(z) = \begin{cases} f(z) & z \in \Omega^+\cup I,\\
\overline{f(\bar z)} & \bar z \in \Omega^+ . \end{cases}
\]
(a) Toon aan dat $F$ goed gedefinieerd en \omterm{def:b3:topology:continuity}{continu} is op $\Omega =
\Omega^+\cup I\cup\Omega^-$, en \omterm{def:b3:holomorphic:holo}{holomorf} op $\Omega^\pm$
\emph{(voor $\Omega^-$: ga Cauchy--Riemann na voor
$\overline{f(\bar z)}$, of ontwikkel $f$ in lokale machtreeksen
en neem de toegevoegde coëfficiënten)}.
(b) Toon met het criterium van Morera aan dat $F$ \omterm{def:b3:holomorphic:holo}{holomorf} is op
heel $\Omega$: $\int_{\partial T}F = 0$ voor elke driehoek $T
\subseteq \Omega$ \emph{(splits driehoeken langs $I$ en duw hun
horizontale zijden met $\varepsilon$ van de as weg, met de
uniforme \omterm{def:b3:topology:continuity}{continuïteit})}.
(c) Leid af: een \omterm{def:b3:holomorphic:holo}{holomorfe functie} op de schijf die reëel is op
een middellijn, voldoet aan $f(\bar z) = \overline{f(z)}$; en een
niet-constante \omterm{def:b3:holomorphic:holo}{holomorfe functie} kan op geen enkele niet-lege
open deelverzameling van haar (\omterm{def:b3:topology:connected}{samenhangende}) gebied reëelwaardig
zijn.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:12}
(De complexe vergelijking van Pythagoras) Zoek alle paren
\emph{gehele} functies met $f^2 + g^2 = 1$.
(a) Toon aan dat $h = f + \iu g$ geheel is en nergens nul, en dat
elke gehele functie zonder nulpunten $\eu^{\varphi}$ is voor een
gehele $\varphi$ \emph{($h'/h$ is geheel en heeft dus een
primitieve op het stervormige $\C$; pas de constante aan en toon
aan dat $h\eu^{-\varphi}$ constant is)}.
(b) Besluit $f = \cos\varphi$ en $g = \sin\varphi$ met $\varphi$
geheel, en ga de omkering na. Wat zijn de gehele oplossingen van
$f^2 + g^2 = 0$?
\end{exercise}

\section{Probleem: de hoofdstelling van de algebra, tweemaal}

\begin{problem}[{Weekendopgave --- $\C$ is algebraïsch
afgesloten: het bewijs van d'Alembert, dat van Liouville, en de
oogst}]\label{pb:b3:holomorphic:1}
Zij $P(z) = z^n + a_{n-1}z^{n-1} + \dots + a_0$ met $n \geq 1$.
We bewijzen tweemaal dat $P$ een wortel heeft, en verzamelen
daarna waar de algebra sinds \cref{ch:b3:galois} op wacht.

\medskip
\noindent\textbf{Deel I --- Coërciviteit en het minimum.}
\begin{enumerate}
  \item Toon aan dat $\abs{P(z)} \to +\infty$ als $\abs z \to
        \infty$: preciezer, dat $\abs{P(z)} \geq \frac12\abs z^n$
        voor $\abs z \geq R_0$ geschikt.
  \item Leid af dat $\abs P$ op $\C$ een globaal minimum bereikt:
        er is een $z_0$ met $\abs{P(z_0)} = \inf_\C\abs P$
        \emph{(\omterm{def:b3:topology:compact}{compactheid} van een grote gesloten schijf,
        \cref{cor:b3:topology:heineborel})}.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De afdaling van d'Alembert.} Stel,
om tot een tegenspraak te komen, dat $P(z_0) \neq 0$.
\begin{enumerate}[resume]
  \item Ontwikkel $Q(h) = P(z_0 + h)/P(z_0)$ als veelterm in $h$:
        $Q(h) = 1 + c_kh^k + h^{k+1}S(h)$ met $c_k \neq 0$, $k
        \geq 1$ en $S$ een veelterm.
  \item Kies de richting van de afdaling: stel voor kleine $t >
        0$ $h = t\,\omega$ met $\omega^k = -1/c_k$ (waarom
        bestaat zo'n $\omega$? --- \emph{bewijs} het bestaan van
        $k$-de machtswortels van elk complex getal met de
        poolvorm, onafhankelijk van de stelling die we bewijzen).
        Toon aan dat
        \[
        \abs{Q(t\omega)} \leq 1 - t^k + C\,t^{k+1}
        \]
        voor kleine $t$, met een expliciete constante $C$.
  \item Besluit $\abs{Q(t\omega)} < 1$ voor kleine $t$ --- in
        tegenspraak met de minimaliteit van $\abs{P(z_0)}$. Dus
        is $P(z_0) = 0$: \emph{elke niet-constante complexe
        veelterm heeft een wortel} (d'Alembert--Argand).
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Het eenregelige bewijs van
Liouville, voluit.}
\begin{enumerate}[resume]
  \item Schrijf het bewijs van
        \cref{cor:b3:holomorphic:liouville} zorgvuldig uit: heeft
        $P$ geen wortel, ga dan na dat $1/P$ geheel en begrensd
        is (kwantificeer, met vraag 1), dus constant, en besluit.
        Vergelijk de twee bewijzen: welke ingrediënten gebruikt
        elk? (De \omterm{def:b3:topology:compact}{compactheid} duikt in beide op --- waar?)
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De oogst.}
\begin{enumerate}[resume]
  \item Toon aan dat elke $P \in \C[X]$ van graad $n$ splitst: $P
        = c\prod_{i}(X - \alpha_i)^{m_i}$ met $\sum m_i = n$
        \emph{(inductie, euclidische deling door $(X -
        \alpha)$)}.
  \item Toon aan dat de irreducibele veeltermen van $\R[X]$ de
        lineaire zijn en de kwadratische met negatieve
        discriminant \emph{(paar de toegevoegde wortels)}; leid
        af dat elke reële veelterm van oneven graad een reële
        wortel heeft, en geef van dat laatste feit een tweede,
        ordetheoretisch bewijs (tussenwaardestelling) --- ga na
        dat ze op $X^3 - X - 1$ overeenstemmen.
  \item Leid de schulden af die dit boek nu kan afbetalen: (i)
        elk endomorfisme van een eindigdimensionale
        $\C$-vectorruimte $\neq 0$ heeft een eigenwaarde, zodat
        elke complexe matrix een \omterm{thm:b3:modules:jordan}{jordanvorm} heeft
        (\cref{thm:b3:modules:jordan}); (ii) het lichaam
        $\bar\Q$ van de \omterm{def:b3:galois:algebraic}{algebraïsche} getallen dat in
        \cref{rem:b3:galois:closureexamples} werd gebruikt, is
        wel degelijk een \omterm{def:b3:galois:closure}{algebraïsche afsluiting} van $\Q$.
  \item (Finale) Wijs precies aan waar elk bewijs zou breken over
        een lichaam als $\Q(\iu)$: welke stappen gebruiken het
        bestaan van $k$-de machtswortels (vraag 4), en welke de
        \omterm{def:b3:topology:compact}{compactheid} of de \omterm{def:b3:complete:complete}{volledigheid} (vragen 2 en 6)? Besluit
        in vijf regels: de stelling is werkelijk
        \emph{analytisch} --- elk bewijs roept ergens de
        \omterm{def:b3:complete:complete}{volledigheid} of de \omterm{def:b3:topology:connected}{samenhang} van $\R$ in --- ook al is
        haar uitspraak zuiver \omterm{def:b3:galois:algebraic}{algebraïsch}.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- De starheidsladder van de gehele
functies.} Liouville is de eerste sport van een ladder; we
beklimmen hem.
\begin{enumerate}[resume]
  \item (\omterm{thm:b3:holomorphic:analytic}{Afschattingen van Cauchy}) Bewijs met de formule van
        Cauchy op de cirkel met straal $r$ rond $a$ dat
        \[
        \bigl|f^{(n)}(a)\bigr| \leq
        \frac{n!\,\sup_{\abs{z-a}=r}\abs f}{r^n} ,
        \]
        en vind Liouville terug als het geval $n = 1$, $r \to
        \infty$.
  \item (Polynomiale groei) Toon aan dat een gehele $f$ met
        $\abs{f(z)} \leq A + B\abs z^m$ voor alle $z$ een
        veelterm van graad $\leq m$ is \emph{(dood de
        taylorcoëfficiënten voorbij $m$ met vraag 11)}.
  \item (Begrensd reëel deel) Toon aan dat een gehele $f$ met
        $\operatorname{Re}f$ naar boven begrensd constant is
        \emph{(pas Liouville toe op $\eu^{f}$)}.
  \item (Dubbele periodiciteit) Zij $f$ geheel met $f(z + 1) =
        f(z)$ en $f(z + \iu) = f(z)$ voor alle $z$. Toon aan dat
        $f$ constant is. Besluit: een niet-constante
        ``elliptische'' functie moet singulariteiten hebben ---
        de historische reden waarom polen de complexe analyse
        binnenkomen.
  \item (\omterm{def:b3:topology:interior}{Dicht} beeld) Toon aan dat het beeld van een
        niet-constante gehele functie \omterm{def:b3:topology:interior}{dicht} ligt in $\C$: mist
        $f(\C)$ een schijf $D(a, r)$, dan is $\frac1{f - a}$
        geheel en begrensd. (Picard bewees dat het beeld hoogstens
        \emph{één punt} mist; de \omterm{ex:b3:lebesgue:gamma}{dichtheid} is het niveau dat ons
        gereedschap haalt.)
  \item (Eigenlijk $\Rightarrow$ veelterm) Stel dat $f$ geheel is
        en $\abs{f(z)} \to \infty$ als $\abs z \to \infty$. Toon
        aan: de nulpunten van $f$ zijn eindig in aantal ($z_1,
        \dots, z_p$, met multipliciteiten $m_i$); het quotiënt $g
        = f/\prod(z - z_i)^{m_i}$ is geheel en zonder nulpunten;
        $1/g$ heeft polynomiale groei, is dus (vraag 12) een
        veelterm, noodzakelijk constant (zonder nulpunten);
        besluit dat $f$ een veelterm is. Onder de gehele functies
        zijn de veeltermen dus precies de eigenlijke --- $\eu^z$
        faalt in eigenlijkheid langs $\R_-$.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Harmonische schaduwen en een
gemiddelde van Gauss.}
\begin{enumerate}[resume]
  \item Zij $f = u + \iu v$ \omterm{def:b3:holomorphic:holo}{holomorf} op een \omterm{def:b3:topology:topology}{open verzameling}. Ga
        na dat $u = \operatorname{Re}f$ voldoet aan de eigenschap
        van de gemiddelde waarde
        \[
        u(a) = \frac1{2\pi}\int_0^{2\pi}
        u\bigl(a + r\eu^{\iu\theta}\bigr)\,\dd\theta
        \]
        (het reële deel van de formule van Cauchy), en leid het
        maximumprincipe voor $u$ op een begrensd gebied af, met
        hetzelfde samenhangsbewijs als voor $\abs f$.
  \item (Gemiddelde waarde van Gauss) Bewijs voor $a \in \C$ en
        $r > 0$ met $\abs a \neq r$ dat
        \[
        \frac1{2\pi}\int_0^{2\pi}
        \log\bigl|a - r\eu^{\iu\theta}\bigr|\,\dd\theta
        = \log\max\bigl(\abs a, r\bigr)
        \]
        \emph{(is $\abs a > r$, dan is $z \mapsto \log\abs{a -
        z}$ het reële deel van een \omterm{def:b3:holomorphic:holo}{holomorfe} logaritme op een
        \omterm{def:b3:topology:topology}{omgeving} van de gesloten schijf --- waarom bestaat die?
        --- zodat vraag 17 van toepassing is; is $\abs a < r$,
        splits dan $\abs{a - r\eu^{\iu\theta}} = r\,\abs{1 -
        \frac ar\eu^{-\iu\theta}}$ af en hergebruik het eerste
        geval)}.
  \item (\omterm{def:b3:measure:measure}{Maat} van Mahler) Leid voor $P = c\prod_{i=1}^n(X -
        \alpha_i) \in \C[X]$ de \emph{formule van Jensen voor
        veeltermen} af:
        \[
        \frac1{2\pi}\int_0^{2\pi}\log\bigl|P(\eu^{\iu\theta})
        \bigr|\,\dd\theta
        = \log\Bigl(\abs c\prod_{i}\max(1,
        \abs{\alpha_i})\Bigr) :
        \]
        het meetkundig gemiddelde van $\abs P$ op de
        eenheidscirkel leest de wortels buiten de schijf af. Ga
        dat na op $P = X^2 - X$ en op $P = 2X - 1$.
  \item (Getallen van Bernoulli) Definieer de coëfficiënten $B_n$
        door $\frac{z}{\eu^z - 1} = \sum_{n\geq0}\frac{B_n}{n!}
        z^n$ nabij $0$ (waarom is het linkerlid analytisch in
        $0$?). Leid uit $(\eu^z - 1)\cdot\frac z{\eu^z-1} = z$ de
        recursie $\sum_{k=0}^{n}\binom{n+1}kB_k = 0$ ($n \geq 1$)
        af, bereken $B_0, \dots, B_6$, en toon aan dat $B_{2k+1}
        = 0$ voor $k \geq 1$ \emph{(de functie $\frac
        z{\eu^z-1} + \frac z2$ is even)}. Deze getallen zullen in
        \cref{ch:b3:residues} elke $\zeta(2k)$ beprijzen.
  \item (Realiteit) Toon aan dat een gehele functie die op $\R$
        reële waarden aanneemt, overal voldoet aan $f(\bar z) =
        \overline{f(z)}$ \emph{(vergelijk de taylorcoëfficiënten
        in $0$, of pas de identiteitsstelling toe op $z \mapsto
        \overline{f(\bar z)}$)}; leid opnieuw af dat de
        niet-reële wortels van reële veeltermen in toegevoegde
        paren komen (de paring van vraag 8, analytisch
        herbewezen).
  \item (De moraal) Stel de starheidsladder samen: begrensd
        $\Rightarrow$ constant; polynomiaal begrensd $\Rightarrow$
        veelterm; eigenlijk $\Rightarrow$ veelterm; een schijf
        missend $\Rightarrow$ constant; dubbelperiodiek
        $\Rightarrow$ constant. Contrasteer in een korte alinea
        met de reële $\mathcal C^\infty$-functies (\omterm{def:b3:lp:mollifier}{bultfuncties},
        \cref{thm:b3:lp:regularization}): waarom legt de
        \omterm{def:b3:holomorphic:holo}{holomorfie}, een zuiver lokale voorwaarde, een globale wet
        en orde op?
\end{enumerate}

\medskip
\noindent\textbf{Deel VII --- Laatste oogst.}
\begin{enumerate}[resume]
  \item (Ongelijkheid van Landau) Bewijs voor $P =
        \sum_{k=0}^na_kX^k$ de gemiddelde waarde
        $\frac1{2\pi}\int_0^{2\pi}
        \abs{P(\eu^{\iu\theta})}^2\dd\theta =
        \sum_k\abs{a_k}^2$ (orthogonaliteit van de $\eu^{\iu
        k\theta}$), en leid daarna, met de puntsgewijze
        afschatting $\log t \leq t - 1$ om de gemiddelden van
        $\log\abs P^2$ en $\abs P^2$ te vergelijken, uit vraag 19
        af dat
        \[
        \abs c\prod_{i}\max\bigl(1, \abs{\alpha_i}\bigr)
        \;\leq\;
        \Bigl(\sum_{k=0}^{n}\abs{a_k}^2\Bigr)^{1/2} :
        \]
        het product van de wortels buiten de eenheidsschijf wordt
        beheerst door de $\ell^2$-grootte van de coëfficiënten
        (behandel wortels \emph{op} de cirkel door de
        ongelijkheid op $P(rX)$ toe te passen en $r \to 1$ te
        laten gaan). Controleer haar op $X^2 - X$.
  \item (De getallen van Bernoulli groeien faculteitsgewijs) Toon
        aan dat de convergentiestraal van $\sum\frac{B_n}{n!}z^n$
        \emph{precies} $2\pi$ is: minstens $2\pi$ omdat $z/(\eu^z
        - 1)$ zich \omterm{def:b3:holomorphic:holo}{holomorf} uitbreidt tot $D(0, 2\pi)$, hoogstens
        $2\pi$ omdat de som anders begrensd zou blijven nabij
        $2\pi\iu$, waar $\abs{z/(\eu^z-1)} \to \infty$. Leid af
        dat
        \[
        \limsup_{k\to\infty}
        \Bigl(\frac{\abs{B_{2k}}}{(2k)!}\Bigr)^{1/2k}
        = \frac1{2\pi} :
        \]
        de getallen van Bernoulli groeien faculteitsgewijs. Neem
        $B_{12} = -\frac{691}{2730}$ zonder bewijs aan (de
        recursie van vraag 20, verder doorgezet) en vergelijk
        $\abs{B_{12}} \approx 0.25311$ met de scherpere
        voorspelling $2\,(2k)!/(2\pi)^{2k} \approx 0.25305$ bij
        $k = 6$ --- vier overeenkomende cijfers van een
        asymptotische wet die \cref{ch:b3:residues} exact zal
        bewijzen, via $\zeta(2k)$.
  \item (Wortels bewegen \omterm{def:b3:topology:continuity}{continu}) Zij $(P_j)$ monisch van graad
        $n$ met coëfficiënten die naar die van de (monische) $P$
        convergeren. Bewijs de \emph{grens van Cauchy}: elke
        wortel van een monische $Q = X^n + \sum_{k<n}q_kX^k$
        voldoet aan $\abs\alpha \leq 1 + \max_k\abs{q_k}$; leid
        af dat de wortels van de $P_j$ in een vaste \omterm{def:b3:topology:compact}{compacte}
        verzameling blijven en, door convergente deelrijen van
        wortelvectoren te nemen en in de ontbinding van vraag 7
        naar de limiet over te gaan, dat de multiverzamelingen
        van wortels van $P_j$ naar die van $P$ convergeren. Toon
        ten slotte aan dat \omterm{def:b3:topology:continuity}{continuïteit} het beste is wat je kunt
        zeggen: voor $P_\varepsilon = X^2 - 2X + 1 + \varepsilon$
        verplaatst een verstoring ter grootte $\varepsilon =
        10^{-4}$ de dubbele wortel $1$ over $10^{-2}$ --- een
        hölderexponent $\frac1m$ in een $m$-voudige wortel, nooit
        lipschitz: numeriek kosten meervoudige wortels de helft
        van de cijfers.
\end{enumerate}
\end{problem}
