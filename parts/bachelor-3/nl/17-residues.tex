\chapter{Laurentreeksen en de residustelling}
\label{ch:b3:residues}

Wat gebeurt er met een \omterm{def:b3:holomorphic:holo}{holomorfe functie} nabij een punt waar ze
\emph{niet} gedefinieerd is? Het antwoord is een volledige
driedeling --- ophefbaar punt, pool of essentiële singulariteit
--- af te lezen van een tweezijdige machtreeks, de
laurentontwikkeling. Eén coëfficiënt van die ontwikkeling, het
\emph{\omterm{def:b3:residues:singularities}{residu}}, beheerst elke \omterm{def:b3:holomorphic:contour}{contourintegraal} rond de
singulariteit: de residustelling zet moeilijke bepaalde
integralen om in eindige algebra, telt de nulpunten van functies
(argumentprincipe, Rouché) en bewijst de open-afbeeldingsstelling.
We versterken eerst de stelling van Cauchy van stervormige
gebieden tot haar definitieve, homologievrije vorm --- het
elegante argument van Dixon --- zodat alle contouren met
windingsgetal nul rond het complement beschikbaar worden.

\section{De globale stelling van Cauchy}

Een \emph{cykel}\index{cykel} $\Gamma$ is een eindige formele som
van gesloten wegen $\gamma_1, \dots, \gamma_m$; integralen en
indices langs $\Gamma$ zijn de bijbehorende sommen, en
$\operatorname{im}\Gamma = \bigcup\operatorname{im}\gamma_j$.

\begin{theorem}[Cauchy, globale vorm]
\label{thm:b3:residues:globalcauchy}
Zij $\Omega \subseteq \C$ open, $f \in \mathcal H(\Omega)$ en
$\Gamma$ een cykel in $\Omega$ zodanig dat
\[
\operatorname{Ind}_\Gamma(w) = 0
\qquad\text{voor elke } w \notin \Omega .
\]
Dan is voor alle $z \in \Omega\setminus\operatorname{im}\Gamma$
\[
\frac1{2\iu\pi}\int_\Gamma\frac{f(w)}{w - z}\,\dd w =
\operatorname{Ind}_\Gamma(z)\,f(z),
\qquad\text{en}\qquad
\int_\Gamma f(w)\,\dd w = 0 .
\]
\index{stelling van Cauchy (globaal)}
\end{theorem}

\begin{proof}[Bewijs (Dixon)]
Definieer $g \colon \Omega\times\Omega \to \C$ door
\[
g(z, w) = \begin{cases}
\dfrac{f(w) - f(z)}{w - z} & w \neq z,\\[4pt]
f'(z) & w = z .
\end{cases}
\]
\emph{$g$ is \omterm{def:b3:topology:continuity}{continu}}: buiten de diagonaal is dat duidelijk.
Nabij een diagonaalpunt $(a, a)$ ontwikkel je $f$ in een
machtreeks in $a$ (\cref{thm:b3:holomorphic:analytic}): $f(w) -
f(z) = \sum_{n\geq1}c_n\bigl((w-a)^n - (z-a)^n\bigr)$, en elke
term door $w - z$ delen (ontbinding van $u^n - v^n$) geeft voor
$z, w \in D(a, r)$
\[
g(z, w) =
\sum_{n\geq1}c_n\sum_{j=0}^{n-1}(w-a)^{\,j}(z-a)^{\,n-1-j},
\]
ook geldig op de diagonaal (elke binnenste som wordt
$n(z-a)^{n-1}$, met som $f'(z)$). Voor kleine $r$ convergeert de
reeks uniform op $D(a,r)^2$ ($\abs{\text{term}} \leq
n\abs{c_n}r^{n-1}$, sommeerbaar binnen de straal): de som is dus
\omterm{def:b3:topology:continuity}{continu}.

Stel $h(z) = \frac1{2\iu\pi}\int_\Gamma g(z, w)\,\dd w$ op
$\Omega$: \omterm{def:b3:topology:continuity}{continu} (uniforme \omterm{def:b3:topology:continuity}{continuïteit} van $g$ op \omterm{def:b3:topology:compact}{compacte}
verzamelingen) en \omterm{def:b3:holomorphic:holo}{holomorf} --- volgens Morera (het criterium in
\cref{thm:b3:holomorphic:weierstrassconv}): voor een driehoek $T
\subseteq \Omega$ geeft Fubini $\int_{\partial T}h =
\frac1{2\iu\pi}\int_\Gamma\bigl(\int_{\partial T}g(z, w)\dd
z\bigr)\dd w = 0$, waarbij de binnenste integraal verdwijnt
omdat $z \mapsto g(z, w)$ \omterm{def:b3:holomorphic:holo}{holomorf} is op $\Omega$ (in $z = w$ is
de singulariteit ophefbaar: $g$ is daar \omterm{def:b3:topology:continuity}{continu} en elders
\omterm{def:b3:holomorphic:holo}{holomorf} --- het uitbreidingsargument uit het bewijs van
\cref{thm:b3:holomorphic:formula}).

Definieer op de \omterm{def:b3:topology:topology}{open verzameling} $\Omega' = \{z \notin
\operatorname{im}\Gamma : \operatorname{Ind}_\Gamma(z) = 0\}$
$h_1(z) = \frac1{2\iu\pi}\int_\Gamma\frac{f(w)}{w - z}\dd w$:
\omterm{def:b3:holomorphic:holo}{holomorf} op $\Omega'$ (Morera, of differentiëren onder de
integraal). Voor $z \in \Omega\cap\Omega'$ is
\[
h(z) = \frac1{2\iu\pi}\int_\Gamma\frac{f(w)}{w-z}\dd w
- f(z)\operatorname{Ind}_\Gamma(z) = h_1(z) .
\]
Volgens de hypothese is $\Omega\cup\Omega' = \C$ ($w \notin
\Omega \Rightarrow \operatorname{Ind}_\Gamma(w) = 0$), dus lijmen
$h$ en $h_1$ tot een gehele functie $H$. Omdat de onbegrensde
component van het complement van $\operatorname{im}\Gamma$ in
$\Omega'$ ligt en $h_1(z) \to 0$ als $\abs z \to \infty$
(ML-afschatting), is $H$ begrensd en gaat ze naar $0$: Liouville
(\cref{cor:b3:holomorphic:liouville}) geeft $H \equiv 0$. Dus is
$h \equiv 0$ op $\Omega$, en dat is de integraalformule. Pas haar
voor vaste $a \in \Omega\setminus\operatorname{im}\Gamma$ toe op
$\tilde f(w) = (w - a)f(w)$ in $z = a$:
\[
\frac1{2\iu\pi}\int_\Gamma f(w)\dd w =
\frac1{2\iu\pi}\int_\Gamma \frac{\tilde f(w)}{w - a}\dd w =
\operatorname{Ind}_\Gamma(a)\,\tilde f(a) = 0 .
\]
\end{proof}

\section{Laurentreeksen en geïsoleerde singulariteiten}

\begin{theorem}[Laurentontwikkeling]\label{thm:b3:residues:laurent}
Zij $f$ \omterm{def:b3:holomorphic:holo}{holomorf} op de ring $A = \{r < \abs{z - a} < R\}$ ($0
\leq r < R \leq \infty$). Dan is\index{laurentreeks}
\[
f(z) = \sum_{n\in\Z}c_n\,(z - a)^n
\qquad\text{op } A,
\qquad
c_n = \frac1{2\iu\pi}\int_{C_\rho}\frac{f(w)}{(w -
a)^{n+1}}\,\dd w
\]
voor elke $r < \rho < R$ (onafhankelijk van $\rho$), waarbij de
twee halve reeksen normaal convergeren op \omterm{def:b3:topology:compact}{compacte} deelringen. De
ontwikkeling is uniek.
\end{theorem}

\begin{proof}
Leg $r < \rho_1 < \abs{z - a} < \rho_2 < R$ vast en zij $\Gamma =
C_{\rho_2} - C_{\rho_1}$ (de buitenste tegen de wijzers van de
klok in, de binnenste met de wijzers mee): een cykel in $A$ met
$\operatorname{Ind}_\Gamma(w) = 0$ voor alle $w \notin A$ (punten
binnen de kleine schijf: $1 - 1 = 0$; buiten de grote: $0 - 0$).
Volgens \cref{thm:b3:residues:globalcauchy} geeft
$\operatorname{Ind}_\Gamma(z) = 1 - 0 = 1$
\[
f(z) = \frac1{2\iu\pi}\int_{C_{\rho_2}}\frac{f(w)}{w - z}\dd
w - \frac1{2\iu\pi}\int_{C_{\rho_1}}\frac{f(w)}{w - z}\dd w .
\]
Ontwikkel de eerste kern als in
\cref{thm:b3:holomorphic:analytic} (machten van $\frac{z - a}{w -
a}$, met modulus $< 1$): het niet-negatieve deel
$\sum_{n\geq0}c_n(z-a)^n$. Ontwikkel in de tweede de andere kant
op: $\frac{-1}{w - z} = \frac1{(z-a)(1 - \frac{w - a}{z - a})} =
\sum_{m\geq0}\frac{(w-a)^m}{(z - a)^{m+1}}$, normaal convergent
op $C_{\rho_1}$: het negatieve deel $\sum_{n\leq-1}c_n(z-a)^n$
met de vermelde coëfficiënten (index $n = -m-1$). Onafhankelijkheid
van $\rho$: de integralen voor de coëfficiënt over $C_{\rho}$ en
$C_{\rho'}$ verschillen een $\int_\Gamma$ over een cykel met
index nul van de op $A$ \omterm{def:b3:holomorphic:holo}{holomorfe} $\frac{f(w)}{(w-a)^{n+1}}$:
nul, opnieuw volgens \cref{thm:b3:residues:globalcauchy}.
Uniciteit: integreer $\sum c_n(z-a)^n$ tegen $(z - a)^{-m-1}$
term voor term over $C_\rho$ (normale convergentie): alleen $n =
m$ overleeft.
\end{proof}

\begin{definition}\label{def:b3:residues:singularities}
Is $f$ \omterm{def:b3:holomorphic:holo}{holomorf} op een geperforeerde schijf $D(a, R)\setminus
\{a\}$, ontwikkel dan volgens Laurent ($r = 0$). Drie elkaar
uitsluitende gevallen:\index{geïsoleerde singulariteit}
\begin{itemize}
  \item alle $c_n = 0$ voor $n < 0$: een \emph{\omterm{thm:b3:residues:riemanncw}{ophefbare
        singulariteit}} (de niet-negatieve reeks breidt $f$
        \omterm{def:b3:holomorphic:holo}{holomorf} uit tot $a$);
  \item $c_n \neq 0$ voor eindig veel, maar minstens één, $n <
        0$: een \emph{pool} van orde $m = -\min\{n : c_n \neq
        0\}$; gelijkwaardig is $f = g/(z-a)^m$ met $g$ \omterm{def:b3:holomorphic:holo}{holomorf}
        en $g(a) \neq 0$; gelijkwaardig gaat $\abs{f(z)} \to
        \infty$ als $z\to a$;
  \item oneindig veel negatieve $c_n \neq 0$: een
        \emph{essentiële singulariteit}.
\end{itemize}
Het \emph{residu}\index{residu} is $\operatorname{Res}(f, a) =
c_{-1}$. Een functie die \omterm{def:b3:holomorphic:holo}{holomorf} is op $\Omega$ min een
verzameling polen, heet \emph{meromorf}\index{meromorfe functie}
op $\Omega$.
\end{definition}

\begin{theorem}[Riemann; Casorati--Weierstrass]
\label{thm:b3:residues:riemanncw}
Zij $f$ \omterm{def:b3:holomorphic:holo}{holomorf} op $D(a,R)\setminus\{a\}$.
\begin{enumerate}
  \item (Riemann) Is $f$ \emph{begrensd} nabij $a$, dan is de
        singulariteit ophefbaar.\index{ophefbare singulariteit}
  \item (Casorati--Weierstrass) Is $a$ essentieel, dan ligt
        $f\bigl(D(a,\varepsilon)\setminus\{a\}\bigr)$ \omterm{def:b3:topology:interior}{dicht} in
        $\C$ voor elke $\varepsilon$.
        \index{stelling van Casorati--Weierstrass}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Voor $n < 0$ en $\rho \to 0$: $\abs{c_n} \leq
M\rho^{-n-1}\cdot\rho\cdot\rho^{-\,n}\dots$ volgens ML op
$C_\rho$: $\abs{c_n} \leq \frac{1}{2\pi}\,2\pi\rho\cdot
M\rho^{-(n+1)} = M\rho^{-n} \to 0$ (want $-n > 0$): alle
negatieve coëfficiënten verdwijnen.
(2) Werd een waarde $b$ niet benaderd, dan is $\abs{f - b} \geq
\delta$ nabij $a$, zodat $g = 1/(f - b)$ \omterm{def:b3:holomorphic:holo}{holomorf} en begrensd is
nabij $a$: ophefbaar volgens (1), en $g$ breidt uit met waarde
$c$. Is $c \neq 0$, dan is $f = b + 1/g$ begrensd nabij $a$:
ophefbaar --- uitgesloten. Is $c = 0$, dan heeft $g$ in $a$ een
nulpunt van eindige orde $m$ (\cref{thm:b3:holomorphic:identity};
$g \not\equiv 0$), en heeft $f = b + 1/g$ een pool van orde $m$:
opnieuw uitgesloten.
\end{proof}

\section{De residustelling}

\begin{theorem}[Residustelling]\label{thm:b3:residues:residue}
Zij $\Omega$ open, $S \subseteq \Omega$ eindig, $f \in \mathcal
H(\Omega\setminus S)$, en $\Gamma$ een cykel in $\Omega\setminus
S$ met $\operatorname{Ind}_\Gamma(w) = 0$ voor elke $w \notin
\Omega$. Dan is\index{residustelling}
\[
\frac{1}{2\iu\pi}\int_\Gamma f(z)\,\dd z
= \sum_{a\in S}\operatorname{Ind}_\Gamma(a)\,
\operatorname{Res}(f, a) .
\]
\end{theorem}

\begin{proof}
Zij voor elke $a \in S$ $P_a(z) = \sum_{n\leq-1}c_n^{(a)}(z -
a)^n$ het hoofddeel van $f$ in $a$: een reeks die op
$\C\setminus\{a\}$ convergeert (haar straal in $1/(z-a)$ is
oneindig: de laurentstaart convergeert voor alle kleine $\abs
{z-a}$ en dus, als machtreeks in $(z-a)^{-1}$, overal) en daar
\omterm{def:b3:holomorphic:holo}{holomorf} is. Dan heeft $g = f - \sum_{a\in S}P_a$ in elk punt van
$S$ een \omterm{thm:b3:residues:riemanncw}{ophefbare singulariteit} (haar laurentontwikkeling in $a$
heeft geen negatief deel: de andere $P_{a'}$ zijn \omterm{def:b3:holomorphic:holo}{holomorf} in
$a$), zodat $g$ zich \omterm{def:b3:holomorphic:holo}{holomorf} uitbreidt tot $\Omega$, en
\cref{thm:b3:residues:globalcauchy} geeft $\int_\Gamma g = 0$.
Rest nog elke $P_a$ te integreren: term voor term (normale
convergentie op de \omterm{def:b3:topology:compact}{compacte} $\operatorname{im}\Gamma$, die $a$
mijdt),
\[
\frac1{2\iu\pi}\int_\Gamma(z - a)^n\,\dd z = 0 \ (n \leq -2:
\text{primitieve } \tfrac{(z-a)^{n+1}}{n+1}),
\qquad
\frac1{2\iu\pi}\int_\Gamma\frac{\dd z}{z - a} =
\operatorname{Ind}_\Gamma(a),
\]
dus $\frac1{2\iu\pi}\int_\Gamma P_a =
c_{-1}^{(a)}\operatorname{Ind}_\Gamma(a)$. Sommeer over $a$.
\end{proof}

\begin{method}[Residuen berekenen]\label{met:b3:residues:compute}
Enkelvoudige pool: $\operatorname{Res}(f, a) = \lim_{z\to a}(z -
a)f(z)$; voor $f = g/h$ met $g(a) \neq 0$, $h(a) = 0$ en $h'(a)
\neq 0$ is $\operatorname{Res} = g(a)/h'(a)$. Pool van orde $m$:
$\operatorname{Res}(f, a) = \frac1{(m-1)!}\lim_{z\to
a}\bigl((z-a)^mf(z)\bigr)^{(m-1)}$. Essentiële singulariteiten:
ontwikkel en lees $c_{-1}$ af (bijvoorbeeld uit bekende reeksen).
Ga altijd na welke polen de contour werkelijk omsluit, en met
welke index.
\end{method}

\begin{example}[De vier klassieke integraaltypen]
\label{ex:b3:residues:integrals}
(a) \emph{Rationaal over $\R$}: sluit voor $\int_\R\frac{\dd x}{1
+ x^4}$ af met een grote halve cirkel $S_R$ in het bovenste
halfvlak: de integrand is daar $O(R^{-4})$, dus $\int_{S_R} \to
0$ (ML), en de residustelling met de polen $\eu^{\iu\pi/4}$ en
$\eu^{3\iu\pi/4}$ (enkelvoudig, met \omterm{def:b3:residues:singularities}{residu} $\frac1{4z^3} =
\frac{z}{4z^4} = -\frac z4$ in een pool) geeft
\[
\int_\R\frac{\dd x}{1 + x^4}
= 2\iu\pi\Bigl(-\frac{\eu^{\iu\pi/4}}4 -
\frac{\eu^{3\iu\pi/4}}4\Bigr)
= \frac{\pi}{\sqrt2} .
\]
(b) \emph{Fouriertype}: voor $t \geq 0$ is
$\int_\R\frac{\eu^{\iu tx}}{1 + x^2}\dd x =
2\iu\pi\operatorname{Res}\bigl(\tfrac{\eu^{\iu tz}}{1+z^2},
\iu\bigr) = 2\iu\pi\frac{\eu^{-t}}{2\iu} = \pi\eu^{-t}$ --- de
bovenste halve cirkel werkt omdat daar $\abs{\eu^{\iu tz}} =
\eu^{-t\operatorname{Im}z} \leq 1$; reële delen nemen geeft
$\int_\R\frac{\cos(tx)}{1+x^2}\dd x = \pi\eu^{-\abs t}$, wat de
zonder bewijs aangenomen formule van \cref{exo:b3:lebesgue:10}
vereffent.
(c) \emph{Goniometrisch over een periode}: substitueer $z =
\eu^{\iu t}$, $\cos t = \frac{z + z^{-1}}2$ en $\dd t = \frac{\dd
z}{\iu z}$: $\int_0^{2\pi}\frac{\dd t}{a + \cos t}$ ($a > 1$)
wordt een residutelling binnen de eenheidscirkel
(\cref{exo:b3:residues:2}).
(d) \emph{Reeksen}: paar $f$ met $\pi\cot(\pi z)$, waarvan de
polen de gehele getallen zijn met \omterm{def:b3:residues:singularities}{residu} $1$: zo sommeert de
weekendopgave $\sum n^{-2}$ en $\sum n^{-4}$.
\end{example}

\section{Het argumentprincipe en de stelling van Rouché}

\begin{theorem}[Argumentprincipe]
\label{thm:b3:residues:argument}
Zij $f$ \omterm{def:b3:residues:singularities}{meromorf} op $\Omega$, met nulpunten $z_j$ (van orde
$m_j$) en polen $p_k$ (van orde $\mu_k$), en $\gamma$ een
gesloten weg in $\Omega$ die ze alle mijdt, met
$\operatorname{Ind}_\gamma = 0$ buiten $\Omega$. Dan is
\index{argumentprincipe}
\[
\frac1{2\iu\pi}\int_\gamma\frac{f'(z)}{f(z)}\,\dd z
= \sum_j m_j\operatorname{Ind}_\gamma(z_j)
- \sum_k \mu_k\operatorname{Ind}_\gamma(p_k)
\]
(slechts eindig veel termen zijn $\neq 0$). Voor een enkelvoudige
contour tegen de wijzers van de klok in telt de integraal de
nulpunten min de polen binnenin, met multipliciteit --- en is ze
gelijk aan het windingsgetal van de beeldweg $f\circ\gamma$ rond
$0$.
\end{theorem}

\begin{proof}
Nabij een nulpunt van orde $m$: $f = (z-a)^mg$ met $g(a) \neq 0$,
dus $\frac{f'}f = \frac m{z - a} + \frac{g'}g$ met de tweede term
\omterm{def:b3:holomorphic:holo}{holomorf} nabij $a$: een enkelvoudige pool met \omterm{def:b3:residues:singularities}{residu} $m$. Nabij
een pool van orde $\mu$ geeft $f = (z-a)^{-\mu}g$ \omterm{def:b3:residues:singularities}{residu} $-\mu$.
Elders is $\frac{f'}f$ \omterm{def:b3:holomorphic:holo}{holomorf}. (De nulpunten en polen met index
$\neq 0$ liggen in een \omterm{def:b3:topology:compact}{compact} gebied dat door $\gamma$ wordt
omsloten; volgens de identiteitsstelling zijn ze daar eindig in
aantal, want $f \not\equiv 0$.) Pas
\cref{thm:b3:residues:residue} toe. De laatste opmerking:
$\frac1{2\iu\pi}\int_\gamma\frac{f'}f =
\frac1{2\iu\pi}\int_{f\circ\gamma}\frac{\dd w}w =
\operatorname{Ind}_{f\circ\gamma}(0)$ (substitueer $w =
f(\gamma(t))$).
\end{proof}

\begin{theorem}[Rouché]\label{thm:b3:residues:rouche}
Zij $f, g$ \omterm{def:b3:holomorphic:holo}{holomorf} op $\Omega$, en $\gamma$ een gesloten weg met
$\operatorname{Ind}_\gamma \in \{0,1\}$, nul buiten $\Omega$ (een
enkelvoudige contour). Is\index{stelling van Rouché}
\[
\abs{g(z)} < \abs{f(z)} \qquad \text{op }
\operatorname{im}\gamma,
\]
dan hebben $f$ en $f + g$ evenveel nulpunten (met
multipliciteit) in het gebied $\{\operatorname{Ind}_\gamma =
1\}$.
\end{theorem}

\begin{proof}
Voor $t \in \intcc01$ heeft $f_t = f + tg$ geen nulpunt op
$\operatorname{im}\gamma$ (want $\abs{f_t} \geq \abs f - \abs g >
0$), dus is
\[
N(t) = \frac1{2\iu\pi}\int_\gamma
\frac{f_t'(z)}{f_t(z)}\,\dd z
\]
goed gedefinieerd; ze telt de nulpunten in het omsloten gebied
(\cref{thm:b3:residues:argument}; geen polen). $N$ is \omterm{def:b3:topology:continuity}{continu} in
$t$ (de integrand is gezamenlijk \omterm{def:b3:topology:continuity}{continu}, de noemers zijn uniform
van onderen begrensd --- gedomineerde convergentie) en neemt
gehele waarden aan: dus constant. Bijgevolg $N(0) = N(1)$.
\end{proof}

\begin{corollary}[Open-afbeeldingsstelling]
\label{cor:b3:residues:openmapping}
Een niet-constante \omterm{def:b3:holomorphic:holo}{holomorfe functie} op een \omterm{def:b3:topology:connected}{samenhangende} \omterm{def:b3:topology:topology}{open
verzameling} is een open afbeelding. In het bijzonder geldt
(opnieuw) het maximumprincipe, en heeft een \omterm{def:b3:holomorphic:holo}{holomorfe} bijectie
een \omterm{def:b3:holomorphic:holo}{holomorfe} inverse.\index{open-afbeeldingsstelling
(holomorf)}
\end{corollary}

\begin{proof}
Zij $f(a) = b$; $f - b$ heeft in $a$ een nulpunt van eindige orde
$m \geq 1$ (identiteitsstelling: $f \not\equiv b$). Kies $r$
zodat $f - b$ geen nulpunt heeft op $\bar D(a, r)\setminus\{a\}$
(\omterm{thm:b3:holomorphic:identity}{geïsoleerde nulpunten}) en zij $\delta = \min_{\abs{z - a} =
r}\abs{f(z) - b} > 0$. Voor $\abs{w - b} < \delta$ is op de
cirkel $\abs{(b - w)} < \delta \leq \abs{f - b}$, dus zegt Rouché
($f - b$ tegenover de constante $b - w$) dat $f - w$ precies $m$
nulpunten in $D(a, r)$ heeft: elke zulke $w$ wordt bereikt ---
$f(D(a,r)) \supseteq D(b, \delta)$: open. Maximumprincipe: een
inwendig maximum van $\abs f$ is onmogelijk voor een
niet-constante $f$, want haar beeld rond $f(a)$ bevat punten met
grotere modulus. Inverse: een \omterm{def:b3:holomorphic:holo}{holomorfe} bijectie $f$ is open, dus
is $f^{-1}$ \omterm{def:b3:topology:continuity}{continu}; het nulpunt van $f - f(a)$ in $a$ is
enkelvoudig ($m \geq 2$ zou $m$ originelen van nabije waarden
geven --- verschillende, want $f'$ verdwijnt enkel in
geïsoleerde punten, zodat nabij $a$ de $m$ nulpunten van $f - w$
voor generieke kleine $w$ enkelvoudig en verschillend zijn: in
tegenspraak met de injectiviteit); dan is $f'(a) \neq 0$ en
convergeert het differentiequotiënt van $f^{-1}$:
$\bigl(f^{-1}\bigr)'(b) = 1/f'(a)$.
\end{proof}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b3:residues:1}
Klasseer de singulariteit in $0$ en bereken het \omterm{def:b3:residues:singularities}{residu}:
\[
\frac{\sin z}{z},\qquad
\frac{\eu^z - 1}{z^2},\qquad
\frac{1}{z(z-1)^2},\qquad
\frac{\cos z}{z^3},\qquad
\eu^{1/z},\qquad
\frac1{\sin z} .
\]
Geef ook het \omterm{def:b3:residues:singularities}{residu} van de derde in $z = 1$ en van de laatste in
$z = \pi$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:residues:2}
Bereken voor $a > 1$, via $z = \eu^{\iu t}$:
\[
\int_0^{2\pi}\frac{\dd t}{a + \cos t}
= \frac{2\pi}{\sqrt{a^2 - 1}} .
\]
Ga het limietgedrag bij $a \to 1^+$ en $a \to \infty$ na.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:3}
Bereken met halvecirkelcontouren, met verantwoording van de
afschattingen op de boog:
\[
\int_\R\frac{x^2}{1 + x^6}\,\dd x = \frac\pi3,
\qquad
\int_\R\frac{\dd x}{(1 + x^2)^{2}} = \frac\pi2
\quad\text{(waarmee \cref{exo:b3:fouriertransform:3} opnieuw
volgt)}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:4}
Bewijs voor $t \geq 0$ en $a > 0$:
\[
\int_\R\frac{\cos(tx)}{x^2 + a^2}\,\dd x =
\frac{\pi}{a}\,\eu^{-at},
\]
en leid met inversie de fouriergetransformeerde van $x \mapsto
\eu^{-a\abs x}$ af --- vergelijk met
\cref{exo:b3:fouriertransform:1}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:5}
Ontwikkel $f(z) = \dfrac1{(z-1)(z-2)}$ in een \omterm{thm:b3:residues:laurent}{laurentreeks} in elk
van de drie gebieden $\abs z < 1$, $1 < \abs z < 2$ en $\abs z >
2$. Waarom verschillen de drie ontwikkelingen? Leg uit waarom de
coëfficiënt van $z^{-1}$ in de tweede en de derde ontwikkeling
\emph{geen} \omterm{def:b3:residues:singularities}{residu} van $f$ in $0$ is ($f$ heeft daar geen
singulariteit), en bereken de werkelijke \omterm{def:b3:residues:singularities}{residuen} van $f$, in $1$
en in $2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:6}
(a) Toon aan dat $\eu^{1/z}$ in $0$ een essentiële singulariteit
heeft en ga Casorati--Weierstrass met de hand na: los $\eu^{1/z}
= w$ expliciet op voor elke $w \neq 0$ en geef oplossingen
willekeurig dicht bij $0$.
(b) Toon aan dat $\abs{\eu^{1/z}}$ op elke geperforeerde \omterm{def:b3:topology:topology}{omgeving}
van $0$ onbegrensd is en dat $\eu^{1/z}$ toch geen pool heeft:
welke limiet faalt?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:7}
Tel met Rouché:
(a) de nulpunten van $z^7 - 4z^3 + z - 1$ in $\abs z < 1$;
(b) de nulpunten van $z^4 + 5z + 1$ in $\abs z < 1$ en in $1 <
\abs z < 2$;
(c) bewijs d'Alembert--Gauss opnieuw: een monische veelterm van
graad $n$ heeft $n$ nulpunten in een grote schijf
\emph{(vergelijk met $z^n$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:residues:8}
(Hurwitz) Zij $f_n \to f$ uniform op \omterm{def:b3:topology:compact}{compacte} verzamelingen, met
$f_n \in \mathcal H(\Omega)$, $\Omega$ \omterm{def:b3:topology:connected}{samenhangend} en $f
\not\equiv 0$.
(a) Toon aan dat als alle $f_n$ nulpuntvrij zijn, $f$ dat ook is.
\emph{(Is $f(a) = 0$: het argumentprincipe op een kleine cirkel
rond $a$, en \cref{thm:b3:holomorphic:weierstrassconv} om in
$\int f_n'/f_n$ naar de limiet over te gaan.)}
(b) Toon aan dat als alle $f_n$ injectief zijn, $f$ injectief of
constant is. \emph{(Pas (a) toe op $z \mapsto f_n(z) - f_n(w)$ op
$\Omega\setminus\{w\}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:residues:9}
Integreer voor $n \geq 2$ $\frac1{1 + z^n}$ over de rand van de
sector $\{0 \leq \arg z \leq \frac{2\pi}n,\ \abs z \leq R\}$ en
leid af dat
\[
\int_0^{+\infty}\frac{\dd x}{1 + x^n} =
\frac{\pi}{n\,\sin(\pi/n)} .
\]
Controleer $n = 2$ tegen $\arctan$, en de limiet $n \to \infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:10}
Zij $f$ een rationale functie met $\deg(\text{noemer}) \geq
\deg(\text{teller}) + 2$. Toon aan dat de som van \emph{alle}
\omterm{def:b3:residues:singularities}{residuen} van $f$ nul is \emph{(integreer over steeds grotere
cirkels)}. Gebruik dat om de splitsing in partieelbreuken van
$\frac1{z(z-1)(z-2)}$ opnieuw te berekenen zonder lineaire
algebra.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:residues:11}
(Het sleutelgat: de reflectie-integraal van Euler) Bereken voor
$0 < a < 1$
\[
I(a) = \int_0^{\infty}\frac{x^{a-1}}{1 + x}\,\dd x =
\frac{\pi}{\sin(\pi a)}
\]
door $f(z) = \frac{z^{a-1}}{1+z} = \frac{\eu^{(a-1)\log z}}{1 +
z}$ (de logaritme gesneden langs $\R_+$, met $\arg z \in
\intoo0{2\pi}$) te integreren over de sleutelgatcontour: naar
buiten langs de bovenkant van de snede van $\varepsilon$ tot
$R$, rond $C_R$, terug onder de snede, rond $C_\varepsilon$.
Verantwoord: de twee rechte stukken verschillen de factor
$\eu^{2\iu\pi(a-1)}$, de bijdragen van de cirkels verdwijnen
($R^{a-1}\cdot R \to 0$ en $\varepsilon^{a-1}\cdot\varepsilon \to
0$), en de enige pool $z = -1$ heeft \omterm{def:b3:residues:singularities}{residu} $\eu^{\iu\pi(a -
1)}$. Leid ook $\Gamma(a)\Gamma(1 - a) = \frac\pi{\sin\pi a}$ af
\emph{(schrijf $\Gamma(a)\Gamma(1-a) = B(a, 1-a)$ volgens
\cref{pb:b3:lebesgue:1} en substitueer $t = \frac{x}{1+x}$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:12}
(Nulpunten tellen met het argumentprincipe, numeriek) Zij $P(z) =
z^4 + 8z + 1$.
(a) Hoeveel nulpunten in de eenheidsschijf? \emph{(Rouché tegen
$8z + 1$.)}
(b) Hoeveel in de ring $1 < \abs z < 3$? \emph{(Rouché tegen
$z^4$ op $\abs z = 3$.)} Scherp aan: toon aan dat elk nulpunt
modulus $< 2.1$ heeft.
(c) Hoeveel in het rechterhalfvlak? \emph{(Tel eerst op $\abs z =
2$; volg daarna het beeld van de imaginaire as: $P(\iu t) = t^4 +
1 + 8\iu t$ heeft overal een positief reëel deel, dus geen
nulpunten op de as, en de argumentvariatie erlangs is
berekenbaar --- besluit met een grote halve schijf.)}
\end{exercise}

\section{Probleem: \texorpdfstring{$\zeta(2k)$}{zeta(2k)} met de
cotangens}

\begin{problem}[{Weekendopgave --- $\sum n^{-2k}$ sommeren met
\omterm{def:b3:residues:singularities}{residuen}}]\label{pb:b3:residues:1}
De residustelling sommeert reeksen: een rationale functie paren
met $\pi\cot(\pi z)$, waarvan de polen in de gehele getallen
liggen, maakt van $\sum_{n}f(n)$ een residutelling. We bewijzen
de methode en berekenen $\zeta(2) = \frac{\pi^2}{6}$ en $\zeta(4)
= \frac{\pi^4}{90}$ --- de waarden die in het volume van
bachelorjaar 2 met fourierreeksen en in \cref{ch:b3:spectral} met
operatorsporen werden gevonden, nu met contourintegratie.

\medskip
\noindent\textbf{Deel I --- De cotangenskern.}
\begin{enumerate}
  \item Toon aan dat $\pi\cot(\pi z)$ \omterm{def:b3:residues:singularities}{meromorf} is op $\C$, met
        enkelvoudige polen precies in $z = n \in \Z$, elk met
        \omterm{def:b3:residues:singularities}{residu} $1$ \emph{(bereken $\lim_{z\to n}(z-n)\pi\cot\pi
        z$)}.
  \item Bereken het begin van de laurentontwikkeling in $0$:
        \[
        \pi\cot(\pi z) = \frac1z - \frac{\pi^2}{3}\,z -
        \frac{\pi^4}{45}\,z^3 + O(z^5) ,
        \]
        door de machtreeks van $\cos$ door die van $\sin$ te
        delen (verantwoord de deling: $\frac{\sin\pi z}{\pi z}$
        is \omterm{def:b3:holomorphic:holo}{holomorf} en $\neq 0$ nabij $0$, dus is haar omgekeerde
        \omterm{def:b3:holomorphic:holo}{holomorf}; identificeer de coëfficiënten tot en met orde
        $3$).
  \item Zij $C_N$ de rand van het vierkant met hoekpunten
        $(\pm1\pm\iu)(N + \frac12)$. Toon aan dat
        $\abs{\cot(\pi z)} \leq 2$ op $C_N$ voor elke $N \geq 1$.
        \emph{(Op de verticale zijden is $\cot(\pi(\pm(N +
        \frac12) + \iu y)) = \mp\tan(\iu\pi y)$, met modulus
        $\abs{\tanh(\pi y)} \leq 1$; op de horizontale zijden,
        waar $\abs y = N + \frac12$, schat je
        $\abs{\cot(\pi(x\pm\iu y))} \leq \coth(\pi y) \leq
        \coth(\pi/2) < 1.1$ af.)}
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De sommatiestelling.}
\begin{enumerate}[resume]
  \item Zij $f$ rationaal en \omterm{def:b3:holomorphic:holo}{holomorf} in de gehele getallen, met
        $\deg(\text{noemer}) \geq \deg(\text{teller}) + 2$.
        Bewijs met de residustelling op $C_N$ en de afschatting
        van vraag 3:
        \[
        \lim_{N\to\infty}\ \sum_{n = -N}^{N} f(n)
        = -\sum_{p\ \text{pool van}\ f}
        \operatorname{Res}\bigl(\pi\cot(\pi z)f(z),\,p\bigr).
        \]
  \item Waar heeft het argument de graadvoorwaarde nodig? Toon
        met een voorbeeld (neem $f(z) = 1/(z + \frac12)$) aan dat
        bij trager verval de symmetrische limiet nog kan bestaan
        terwijl de tweezijdige reeks divergeert --- en dat de
        formule dan de \emph{hoofdwaarde} berekent.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De waarden.}
\begin{enumerate}[resume]
  \item Pas de methode toe op $f(z) = 1/z^2$: hier ligt de pool
        van $f$ \emph{in} een geheel getal, dus voer je het
        argument rechtstreeks uit --- integreer $g(z) =
        \frac{\pi\cot(\pi z)}{z^2}$ over $C_N$, toon aan dat de
        integraal naar $0$ gaat, en bereken
        $\operatorname{Res}(g, 0)$ met vraag 2. Besluit:
        \[
        2\sum_{n\geq1}\frac1{n^2} = \frac{\pi^2}{3},
        \qquad \zeta(2) = \frac{\pi^2}6 .
        \]
  \item Idem met $g(z) = \frac{\pi\cot(\pi z)}{z^4}$: bereken
        $\operatorname{Res}(g, 0)$ en leid $\zeta(4) =
        \frac{\pi^4}{90}$ af.
  \item Verklaar het algemene patroon: voor elke $k \geq 1$ is
        $\zeta(2k)$ gelijk aan $-\frac12$ maal de coëfficiënt van
        $z^{2k-1}$ in de laurentontwikkeling van $\pi\cot(\pi z)$
        in $0$ --- een rationaal veelvoud van $\pi^{2k}$. Bereken
        $\zeta(6)$ door de deling van vraag 2 één stap verder te
        duwen. Wat zegt de methode over $\zeta(3)$ --- en waarom
        zegt ze niets?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De partieelbreuksplitsing van de
cotangens.}
\begin{enumerate}[resume]
  \item Leg $w \in \C\setminus\Z$ vast en pas de methode van
        Deel II toe op $f(z) = \dfrac{1}{(z - w)(z + w)}$ ---
        merk op dat $\pi\cot(\pi z)f(z)$ nu extra enkelvoudige
        polen in $\pm w$ heeft, waarvan de \omterm{def:b3:residues:singularities}{residuen} mee moeten
        tellen. Leid de partieelbreuksplitsing
        \[
        \pi\cot(\pi w) = \frac1w +
        \sum_{n\geq1}\frac{2w}{w^2 - n^2}
        \]
        af, waarbij de reeks normaal convergeert op \omterm{def:b3:topology:compact}{compacte}
        deelverzamelingen van $\C\setminus\Z$.
  \item Vind uit deze ontwikkeling, door elke term in machten van
        $w$ te ontwikkelen (verantwoord het verwisselen),
        \emph{dezelfde} laurentcoëfficiënten terug als in vraag 2
        --- de cirkel sluit zich: de formule van Euler
        $\sum\frac1{n^2} = \frac{\pi^2}6$ is de coëfficiënt van
        $w$ in de twee gezichten van de cotangens. Vergelijk met
        het bewijs met fourierreeksen (volume van bachelorjaar 2)
        en dat met het spoor (\cref{pb:b3:spectral:1}): drie
        theorieën, één getal.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Het product van Euler voor de
sinus.} De ontwikkeling van vraag 9 is de logaritmische afgeleide
van een oneindig product; we bewijzen nu eerlijk de ontbinding
die Euler in 1734 vond.
\begin{enumerate}[resume]
  \item Stel voor $N \geq 1$ $P_N(z) = z\prod_{n=1}^{N}\bigl(1 -
        \frac{z^2}{n^2}\bigr)$. Toon aan dat $P_N$ uniform op
        elke schijf $\bar D(0, R)$ convergeert naar een gehele
        functie $P$ waarvan de nulpunten precies de gehele
        getallen zijn, alle enkelvoudig. \emph{(Schrijf voor $n
        \geq 2R$ de factor als $\exp\log(1 - z^2/n^2)$ met de
        hoofdlogaritme van \cref{exo:b3:holomorphic:3}, schat
        $\abs{\log(1+u)} \leq 2\abs u$ af voor $\abs u \leq
        \frac12$ met de reeks, en exponentieer de normaal
        convergente som van logaritmen; de eindig veel resterende
        factoren vormen een veelterm. Besluit met
        \cref{thm:b3:holomorphic:weierstrassconv}.)}
  \item Toon aan dat op $\C\setminus\Z$
        \[
        \frac{P'(z)}{P(z)} = \frac1z +
        \sum_{n\geq1}\frac{2z}{z^2 - n^2} = \pi\cot(\pi z)
        \]
        \emph{(differentieer de eindige producten, ga met
        \cref{thm:b3:holomorphic:weierstrassconv} en de
        nulpuntvrijheid van $P$ buiten $\Z$ naar de limiet, en
        citeer vraag 9)}.
  \item Toon aan dat $Q = \sin(\pi z)/P(z)$ zich uitbreidt tot
        een nulpuntvrije gehele functie met $Q' = 0$, en besluit
        tot het \emph{product van Euler}:
        \[
        \sin(\pi z) = \pi z\prod_{n\geq1}
        \Bigl(1 - \frac{z^2}{n^2}\Bigr)
        \qquad (z \in \C) .
        \]
  \item (Wallis, 1655) Evalueer in $z = \frac12$:
        \[
        \frac\pi2 = \prod_{n\geq1}\frac{4n^2}{4n^2 - 1}
        = \lim_{N\to\infty}
        \frac{2\cdot2\cdot4\cdot4\cdots(2N)(2N)}
        {1\cdot3\cdot3\cdot5\cdots(2N-1)(2N+1)} .
        \]
  \item Ontwikkel voor $\abs z < 1$ de logaritme van het product
        als dubbele reeks (verantwoord de herschikking) en vind
        $\zeta(2) = \frac{\pi^2}6$ terug door de coëfficiënt van
        $z^3$ te vergelijken in $\sin(\pi z) = \pi z -
        \frac{\pi^3}6z^3 + \cdots$ --- het gezicht van Eulers
        getal aan de kant van het product.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Zusterkernen.} De cotangens heeft
zussen; elk beprijst haar eigen familie reeksen.
\begin{enumerate}[resume]
  \item Differentieer de ontwikkeling van vraag 9 term voor term
        (verantwoord met \cref{thm:b3:holomorphic:weierstrassconv})
        om, normaal op \omterm{def:b3:topology:compact}{compacte} deelverzamelingen van
        $\C\setminus\Z$, te verkrijgen
        \[
        \frac{\pi^2}{\sin^2(\pi z)} =
        \sum_{n\in\Z}\frac1{(z - n)^2} .
        \]
  \item Evalueer in $z = \frac12$:
        $\sum_{m\geq0}\frac1{(2m+1)^2} = \frac{\pi^2}8$; vind
        $\zeta(2)$ nog eens terug door de gehele getallen naar
        pariteit te splitsen.
  \item Ga de verdubbelingsidentiteit $\tan\theta = \cot\theta -
        2\cot(2\theta)$ na en leid af dat
        \[
        \pi\tan(\pi z) =
        \sum_{m\geq0}\frac{8z}{(2m+1)^2 - 4z^2} ,
        \]
        normaal op \omterm{def:b3:topology:compact}{compacte} verzamelingen die $\frac12 + \Z$
        mijden.
  \item Ontwikkel rond $0$ ($\abs z < \frac12$; opnieuw Fubini):
        met $\lambda(s) = \sum_{m\geq0}(2m+1)^{-s}$ is
        \[
        \pi\tan(\pi z) =
        \sum_{k\geq0}8\cdot4^k\,\lambda(2k+2)\,z^{2k+1} ;
        \]
        vergelijk met $\tan u = u + \frac{u^3}3 + O(u^5)$ om
        $\lambda(2) = \frac{\pi^2}8$ terug te vinden en
        $\lambda(4) = \frac{\pi^4}{96}$ te krijgen, en controleer
        daarna $\zeta(4) = \frac{\pi^4}{90}$ via $\lambda(4) = (1
        - 2^{-4})\,\zeta(4)$.
  \item Ga $\frac1{\sin\theta} = \cot\frac\theta2 - \cot\theta$
        na en leid af dat
        \[
        \frac{\pi}{\sin(\pi z)} = \frac1z +
        \sum_{n\geq1}(-1)^n\,\frac{2z}{z^2 - n^2} .
        \]
        Controleer de tekens tegen de \omterm{def:b3:residues:singularities}{residuen} van $\pi/\sin(\pi
        z)$ in de gehele getallen.
  \item Lees de coëfficiënt van $z$ af: $\eta(2) =
        \sum_{n\geq1}\frac{(-1)^{n-1}}{n^2} = \frac{\pi^2}{12}$,
        en bevestig de consistentie $\eta(2) = (1 -
        2^{1-2})\,\zeta(2)$.
  \item (Finale) Evalueer de ontwikkeling van vraag 20 in $z =
        \frac12$ en leid de \emph{formule van Leibniz} af:
        \[
        \frac\pi4 = 1 - \frac13 + \frac15 - \frac17 +
        \cdots
        \]
        Sluit af met een korte alinea: één kern per rekenkunde
        --- welke kern beprijst welke familie reeksen, en waarom
        zijn ze structureel allemaal blind voor $\zeta(3)$?

\end{enumerate}
\medskip
\noindent\textbf{Deel VII --- De volledige prijslijst: de
getallen van Bernoulli.}
\begin{enumerate}[resume]
  \item Combineer de partieelbreuksplitsing van $\pi z\cot(\pi
        z)$ met de genererende functie van de getallen van
        Bernoulli ($\frac{w}{\eu^w - 1} = \sum_n\frac{B_n}{n!}
        w^n$, \cref{pb:b3:holomorphic:1}, Deel VI): leid uit
        \[
        \pi z\cot(\pi z) = \iu\pi z +
        \frac{2\iu\pi z}{\eu^{2\iu\pi z} - 1}
        \]
        (bewijs eerst die identiteit) de gesloten vorm
        \[
        \zeta(2k) = (-1)^{k+1}\,
        \frac{(2\pi)^{2k}\,B_{2k}}{2\,(2k)!}
        \qquad (k \geq 1)
        \]
        af.
  \item Toets de formule aan $B_2 = \frac16$, $B_4 =
        -\frac1{30}$ en $B_6 = \frac1{42}$: vind $\zeta(2) =
        \frac{\pi^2}6$ en $\zeta(4) = \frac{\pi^4}{90}$ terug, en
        bereken $\zeta(6) = \frac{\pi^6}{945}$.
  \item (Recursie van Euler) Ontwikkel beide leden van
        $\bigl(z\cot z\bigr)' = \cot z - z(1 + \cot^2z)$ --- of
        kwadrateer de cotangensreeks rechtstreeks --- om
        \[
        \Bigl(k + \frac12\Bigr)\zeta(2k) =
        \sum_{j=1}^{k-1}\zeta(2j)\,\zeta(2k - 2j)
        \qquad (k \geq 2)
        \]
        te bewijzen, en ga na dat ze $\zeta(4)$ uit $\zeta(2)$ en
        $\zeta(6)$ uit $\zeta(2), \zeta(4)$ berekent --- alle
        even zetawaarden uit het ene zaadje $\frac{\pi^2}6$,
        zonder één nieuwe integratie.
\end{enumerate}
\end{problem}
