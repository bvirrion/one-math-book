\chapter{Hilbertruimten}\label{ch:b3:hilbert}

Een \omterm{def:b3:hilbert:inner}{hilbertruimte} is een banachruimte waarvan de norm van een
\omterm{def:b3:hilbert:inner}{inwendig product} komt --- en die ene extra structuur herstelt in
oneindige dimensie zo goed als de hele euclidische meetkunde:
orthogonale projecties bestaan, elke \omterm{def:b3:topology:continuity}{continue} functionaal is een
\omterm{def:b3:hilbert:inner}{inwendig product} tegen een vaste vector (Riesz), en orthonormale
bases ontwikkelen elke vector in een convergente reeks met een
pythagoreïsche boekhouding (\omterm{thm:b3:hilbert:parseval}{Parseval}). Het hoogtepunt van dit
hoofdstuk is een ingeloste schuld: het trigonometrische stelsel
is een orthonormale basis van $L^2$, zodat de identiteit van
\omterm{thm:b3:hilbert:parseval}{Parseval} voor \emph{elke} kwadratisch \omterm{def:b3:lebesgue:l1}{integreerbare functie}
geldt --- de uitspraak die het volume van bachelorjaar 2 alleen
voor stuksgewijs $\mathcal C^1$-functies kon bewijzen. We
eindigen met Lax--Milgram, het werkpaard van de variationele
aanpak van differentiaalvergelijkingen.

Overal is $H$ een vectorruimte over $K = \R$ of $\C$.

\section{Inwendige producten; de projectiestelling}

\begin{definition}\label{def:b3:hilbert:inner}
Een \emph{inwendig product}\index{inwendig product} is een
afbeelding $\langle \cdot,\cdot\rangle \colon H\times H \to K$
die lineair is in de tweede veranderlijke, met $\langle y,
x\rangle = \overline{\langle x, y\rangle}$ en $\langle x,
x\rangle > 0$ voor $x \neq 0$. Ze induceert de norm $\norm x =
\langle x, x\rangle^{1/2}$, de \emph{ongelijkheid van
Cauchy--Schwarz} $\abs{\langle x, y\rangle} \leq \norm x\norm y$
(het bewijs uit het volume van bachelorjaar 2 --- de
discriminant --- blijft ongewijzigd) en de
\emph{parallellogramwet}
\[
\norm{x + y}^2 + \norm{x - y}^2 = 2\norm x^2 + 2\norm y^2 .
\]
Een \emph{hilbertruimte}\index{hilbertruimte} is een ruimte met
inwendig product die \omterm{def:b3:complete:complete}{volledig} is voor deze norm. Voorbeelden:
$\ell^2$ (\cref{pb:b3:banach:1}) en, het fundamentele geval,
$L^2(\mu)$ met $\langle f, g\rangle = \int\bar fg\,\dd\mu$ ---
\omterm{def:b3:complete:complete}{volledig} volgens Riesz--Fischer (\cref{thm:b3:lp:complete}); het
inwendig product is eindig volgens Cauchy--Schwarz ($=$ Hölder
bij $p = q = 2$).
\end{definition}

\begin{theorem}[Projectie op een gesloten convexe verzameling]
\label{thm:b3:hilbert:projection}
Zij $C \neq \varnothing$ een gesloten \emph{convexe} deelruimte
van de \omterm{def:b3:hilbert:inner}{hilbertruimte} $H$ en $x \in H$. Er is een unieke $p_C(x)
\in C$ met\index{projectiestelling}
\[
\norm{x - p_C(x)} = d(x, C),
\]
gekarakteriseerd door: $\operatorname{Re}\langle x - p_C(x),\ c -
p_C(x)\rangle \leq 0$ voor alle $c \in C$. De afbeelding $p_C$ is
$1$-lipschitz.
\end{theorem}

\begin{proof}
Zij $d = d(x, C)$ en $(c_n) \subseteq C$ met $\norm{x - c_n} \to
d$. Parallellogram op $x - c_n$ en $x - c_m$:
\[
\norm{c_n - c_m}^2 = 2\norm{x - c_n}^2 + 2\norm{x - c_m}^2 -
4\,\bigl\|x - \tfrac{c_n + c_m}2\bigr\|^2
\leq 2\norm{x{-}c_n}^2 + 2\norm{x{-}c_m}^2 - 4d^2
\]
(de convexiteit legt het midden in $C$): het rechterlid gaat naar
$0$, dus is $(c_n)$ cauchy, en haar limiet $p \in C$ (gesloten)
bereikt $d$. Uniciteit: twee minimalisatoren geven met dezelfde
identiteit $\norm{p - p'}^2 \leq 2d^2 + 2d^2 - 4d^2 = 0$.

Karakterisering: voor $c \in C$ en $t \in \intoc01$ ligt de
vector $p + t(c - p)$ in $C$, dus
\[
d^2 \leq \norm{x - p - t(c-p)}^2
= d^2 - 2t\operatorname{Re}\langle x - p, c - p\rangle +
t^2\norm{c-p}^2 ;
\]
deel door $t$ en laat $t \to 0^+$: $\operatorname{Re}\langle x -
p, c - p\rangle \leq 0$. Omgekeerd geeft deze ongelijkheid
$\norm{x - c}^2 = \norm{x - p}^2 - 2\operatorname{Re}\langle x -
p, c - p\rangle + \norm{p - c}^2 \geq \norm{x-p}^2$.
Lipschitz: tel voor $x, y$ met projecties $p, q$ de twee
variationele ongelijkheden op (met $c = q$ respectievelijk $c =
p$): $\operatorname{Re}\langle x - y - (p - q), p - q\rangle \geq
0$, dus $\norm{p - q}^2 \leq \operatorname{Re}\langle x - y, p -
q\rangle \leq \norm{x - y}\norm{p - q}$.
\end{proof}

\begin{theorem}[Orthogonale ontbinding]
\label{thm:b3:hilbert:decomposition}
Zij $F$ een \emph{gesloten deelruimte} van $H$. Dan is $p_F$
lineair, is $x - p_F(x) \perp F$ voor alle $x$, en
geldt\index{orthogonaal complement}
\[
H = F \oplus F^\perp,
\qquad F^\perp = \{y : \langle y, f\rangle = 0\ \forall f\in
F\},
\qquad (F^\perp)^\perp = F .
\]
Voor een algemene deelruimte is $(F^\perp)^\perp = \bar F$; in
het bijzonder ligt $F$ \omterm{def:b3:topology:interior}{dicht} dan en slechts dan als $F^\perp =
\{0\}$.
\end{theorem}

\begin{proof}
Voor een deelruimte dwingt de variationele karakterisering met $c
= p_F(x) \pm f$ ($f \in F$, beide tekens, en $\iu f$ in het
complexe geval) af dat $\langle x - p_F(x), f\rangle = 0$: het
residu staat loodrecht op $F$. Ontbinding $x = p_F(x) + (x -
p_F(x))$ met $F \cap F^\perp = \{0\}$ (want $\langle y, y \rangle
= 0$); de lineariteit van $p_F$ volgt uit de uniciteit van
dergelijke ontbindingen (beide leden zijn er lineair in). Altijd
is $(F^\perp)^\perp \supseteq F$; omgekeerd, staat $x$ loodrecht
op $F^\perp$, schrijf dan $x = f + g$: dan is $g = x - f \in
F^\perp$ en $\langle g, g\rangle = \langle x, g\rangle - \langle
f, g\rangle = 0$, dus $x = f \in F$. Voor een algemene deelruimte
$F$ is $F^\perp = \bar F^{\,\perp}$ (\omterm{def:b3:topology:continuity}{continuïteit} van het
\omterm{def:b3:hilbert:inner}{inwendig product}), zodat $(F^\perp)^\perp = \bar F$ volgens het
gesloten geval; en \omterm{ex:b3:lebesgue:gamma}{dichtheid} geldt dan en slechts dan als $\bar F
= H$, dat wil zeggen $F^\perp = 0$.
\end{proof}

\begin{theorem}[Representatiestelling van
Riesz]\label{thm:b3:hilbert:riesz}
Voor elke \omterm{def:b3:topology:continuity}{continue} lineaire functionaal $\varphi \in H'$ is er
een unieke $a \in H$ met\index{representatiestelling van Riesz}
\[
\varphi(x) = \langle a, x\rangle \quad (x \in H),
\qquad \norm\varphi_{H'} = \norm a .
\]
\end{theorem}

\begin{proof}
Is $\varphi = 0$, neem dan $a = 0$. Anders is $F = \ker\varphi$
een gesloten echte deelruimte; kies $u \in F^\perp$ met $\norm u
= 1$ (\cref{thm:b3:hilbert:decomposition}: $F^\perp \neq 0$ omdat
$F \neq H$). Voor elke $x$ ligt de vector $\varphi(x)u -
\varphi(u)x$ in $\ker\varphi$, dus $\perp u$:
\[
0 = \langle u, \varphi(x)u - \varphi(u)x\rangle
= \varphi(x) - \varphi(u)\langle u, x\rangle :
\qquad \varphi(x) = \langle
\overline{\varphi(u)}\,u,\ x\rangle .
\]
Dus voldoet $a = \overline{\varphi(u)}u$. Uniciteit: $\langle a -
a', x\rangle = 0$ voor alle $x$, test met $x = a - a'$. Normen:
$\abs{\varphi(x)} \leq \norm a\norm x$ (Cauchy--Schwarz), met
gelijkheid bij $x = a$.
\end{proof}

\begin{example}[Een projectie, tot het einde berekend]
\label{ex:b3:hilbert:projectionexample}
Wat is in $H = L^2(\intcc01)$ de beste benadering van $f(x) =
x^2$ door een affiene functie? De deelruimte $F =
\operatorname{Vect}(1, x)$ is gesloten (eindigdimensionaal), en
$p_F(f) = a + bx$ wordt gekarakteriseerd door de orthogonaliteit
van het residu op $1$ en op $x$:
\[
\int_0^1(x^2 - a - bx)\,\dd x = 0,
\qquad
\int_0^1x\,(x^2 - a - bx)\,\dd x = 0,
\]
dat wil zeggen $\frac13 = a + \frac b2$ en $\frac14 = \frac a2 +
\frac b3$: dus $a = -\frac16$ en $b = 1$. Bijgevolg is $p_F(x^2)
= x - \frac16$, en de fout is
\[
d(f, F)^2 = \int_0^1\Bigl(x^2 - x + \frac16\Bigr)^2\dd x =
\frac1{180},
\qquad d(f, F) = \frac1{6\sqrt5} .
\]
Twee opmerkingen die het onthouden waard zijn. Ten eerste is de
berekening niets anders dan een $2\times2$-stelsel --- de
\emph{normaalvergelijkingen}; voor de monoombasis is hun matrix
$\bigl(\frac1{i+j+1}\bigr)$ de berucht slecht geconditioneerde
hilbertmatrix, en eerst orthogonaliseren (legendreveeltermen,
\cref{pb:b3:hilbert:1}) is het geneesmiddel. Ten tweede is de
beste \emph{uniforme} benadering van $x^2$ door affiene functies
een andere ($x - \frac18$, wegens de gelijkmatige oscillatie):
elke norm heeft haar eigen meetkunde, en alleen de hilbertiaanse
antwoordt met een lineair stelsel.
\end{example}

\section{Orthonormale bases}

\begin{definition}\label{def:b3:hilbert:onb}
Een familie $(e_i)_{i\in I}$ heet \emph{orthonormaal} als
$\langle e_i, e_j\rangle = \delta_{ij}$, en een
\emph{hilbertbasis}\index{hilbertbasis} (orthonormale basis) als
bovendien haar eindige lineaire combinaties \omterm{def:b3:topology:interior}{dicht} liggen in $H$
(de familie is \emph{totaal}). We behandelen het aftelbare geval
$I = \N$, dat via Gram--Schmidt elke \emph{separabele} $H$ dekt
(\cref{prop:b3:hilbert:gramschmidt}).
\end{definition}

\begin{theorem}[Bessel, Parseval]\label{thm:b3:hilbert:parseval}
Zij $(e_n)_{n\in\N}$ orthonormaal in $H$, en $c_n(x) = \langle
e_n, x\rangle$.
\begin{enumerate}
  \item (Bessel) $\sum_n\abs{c_n(x)}^2 \leq \norm x^2$, en de
        reeks $\sum_nc_n(x)e_n$ convergeert in $H$, met som
        $p_F(x)$, waarbij $F =
        \overline{\operatorname{Vect}}(e_n)$.
  \item De volgende uitspraken zijn gelijkwaardig: (i) $(e_n)$ is
        een \omterm{def:b3:hilbert:onb}{hilbertbasis}; (ii) $x = \sum_nc_n(x)e_n$ voor elke
        $x$; (iii) \emph{Parseval}\index{identiteit van
        Parseval}: $\norm x^2 = \sum_n\abs{c_n(x)}^2$ voor elke
        $x$; (iv) de enige vector die loodrecht staat op alle
        $e_n$ is $0$.
  \item Is $(e_n)$ een \omterm{def:b3:hilbert:onb}{hilbertbasis}, dan is $x \mapsto
        (c_n(x))_n$ een isometrisch isomorfisme $H \to \ell^2$
        (\emph{elke} oneindigdimensionale separabele
        \omterm{def:b3:hilbert:inner}{hilbertruimte} ``is'' $\ell^2$), en is $\langle x,
        y\rangle = \sum_n\overline{c_n(x)}c_n(y)$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Voor eindige $N$ is $x - \sum_{n\leq N}c_ne_n \perp e_k$ ($k
\leq N$), zodat Pythagoras $\norm x^2 = \sum_{n\leq
N}\abs{c_n}^2 + \norm{x - \sum_{n\leq N}c_ne_n}^2$ geeft: dat is
Bessel. De partiële sommen $S_N = \sum_{n\leq N}c_ne_n$ zijn
cauchy: $\norm{S_N - S_M}^2 = \sum_{M<n\leq N}\abs{c_n}^2$, de
staart van een convergente reeks; de limiet ligt in $F$, en $x -
\lim S_N$ staat loodrecht op elke $e_k$ (\omterm{def:b3:topology:continuity}{continuïteit}), dus
$\perp F$: wegens de uniciteit van de orthogonale ontbinding is
$\lim S_N = p_F(x)$.

(2) (i)$\Rightarrow$(ii): $F = H$, dus $p_F = \mathrm{id}$.
(ii)$\Rightarrow$(iii): Pythagoras in de limiet ($\norm{S_N}^2 =
\sum_{n \leq N}\abs{c_n}^2 \to \norm x^2$).
(iii)$\Rightarrow$(iv): staat $x$ loodrecht op alle $e_n$, dan is
$\norm x^2 = 0$. (iv)$\Rightarrow$(i): $F^\perp = \{0\}$
(loodrecht staan op alle $e_n$ is loodrecht staan op $F$), dus
ligt $F$ \omterm{def:b3:topology:interior}{dicht} volgens
\cref{thm:b3:hilbert:decomposition}; maar $F$ is als \omterm{def:b3:topology:interior}{afsluiting}
al gesloten: dus $F = H$.

(3) De afbeelding is lineair, isometrisch volgens (iii) (dus
injectief) en surjectief: is $(c_n) \in \ell^2$ gegeven, dan
convergeert de reeks $\sum c_ne_n$ (cauchy als in (1)) naar een
origineel. De formule voor het \omterm{def:b3:hilbert:inner}{inwendig product} is polarisatie
uit (iii), of een rechtstreekse limietberekening.
\end{proof}

\begin{proposition}[Gram--Schmidt]\label{prop:b3:hilbert:gramschmidt}
Zij $(x_n)$ een lineair onafhankelijke rij. Inductief $\tilde e_n
= x_n - \sum_{k<n}\langle e_k, x_n\rangle e_k$ en $e_n = \tilde
e_n/\norm{\tilde e_n}$ stellen levert een orthonormale $(e_n)$ op
met dezelfde eindige opspansels: $\operatorname{Vect}(e_1, \dots,
e_n) = \operatorname{Vect}(x_1, \dots, x_n)$. Bijgevolg heeft
elke separabele \omterm{def:b3:hilbert:inner}{hilbertruimte} (een ruimte met een aftelbare
\omterm{def:b3:topology:interior}{dichte} deelverzameling) een
\omterm{def:b3:hilbert:onb}{hilbertbasis}.\index{gram-schmidt-orthonormalisatie}
\end{proposition}

\begin{proof}
Inductie: per constructie is $\tilde e_n \perp e_k$ ($k < n$), en
$\tilde e_n \neq 0$ wegens de onafhankelijkheid; de opspansels
komen in elke stap overeen (driehoekige basisverandering). Voor
een separabele $H$: haal uit een \omterm{def:b3:topology:interior}{dichte} rij een lineair
onafhankelijke deelfamilie met \omterm{def:b3:topology:interior}{dicht} opspansel (gooi elke vector
weg die in het opspansel van zijn voorgangers ligt --- het
opspansel verandert niet) en orthonormaliseer: het resultaat is
totaal.
\end{proof}

\begin{theorem}[Het trigonometrische stelsel; Parseval, eindelijk]
\label{thm:b3:hilbert:fourier}
In $L^2(\intcc{-\pi}\pi)$ met $\langle f, g\rangle =
\frac1{2\pi}\int_{-\pi}^\pi \bar fg$ is de familie $e_n(t) =
\eu^{\iu nt}$, $n \in \Z$, een \omterm{def:b3:hilbert:onb}{hilbertbasis}. Bijgevolg geldt
voor \emph{elke} $f \in L^2$ --- in het bijzonder elke
stuksgewijs \omterm{def:b3:topology:continuity}{continue} $2\pi$-periodieke $f$ --- met $c_n(f) =
\frac1{2\pi}\int_{-\pi}^{\pi}f(t)\eu^{-\iu nt}\dd t$:
\[
f = \sum_{n\in\Z}c_n(f)\,\eu^{\iu nt} \ \ \text{in } L^2,
\qquad
\frac1{2\pi}\int_{-\pi}^{\pi}\abs f^2 =
\sum_{n\in\Z}\abs{c_n(f)}^2 .
\]
Dit bewijst in volle algemeenheid de identiteit van \omterm{thm:b3:hilbert:parseval}{Parseval} die
het volume van bachelorjaar 2 zonder bewijs
aannam.\index{identiteit van Parseval (algemeen)}
\end{theorem}

\begin{proof}
De orthonormaliteit is een rechtstreekse berekening (volume van
bachelorjaar 2). Totaliteit: zij $f \in L^2$ loodrecht op alle
$e_n$, dat wil zeggen met verdwijnende fouriercoëfficiënten. De
\omterm{def:b3:topology:continuity}{continue} $2\pi$-periodieke functies liggen \omterm{def:b3:topology:interior}{dicht} in
$L^2(\intcc{-\pi}\pi)$: inderdaad ligt $\mathcal
C_c(\intoo{-\pi}\pi)$ \omterm{def:b3:topology:interior}{dicht} (\cref{thm:b3:lp:density}(2)) en
zulke functies breiden periodiek en \omterm{def:b3:topology:continuity}{continu} uit.
Trigonometrische veeltermen liggen
$\norm\cdot_\infty$-dicht in de \omterm{def:b3:topology:continuity}{continue} periodieke functies
(Stone--Weierstrass, \cref{cor:b3:complete:weierstrass}(c)), en
$\norm\cdot_2 \leq \norm\cdot_\infty$: dus liggen
trigonometrische veeltermen \omterm{def:b3:topology:interior}{dicht} in $L^2$. Maar $f$ staat
loodrecht op elke trigonometrische veelterm, dus loodrecht op een
\omterm{def:b3:topology:interior}{dichte} deelruimte: $f \in (\text{dicht})^\perp = \{0\}$
(\cref{thm:b3:hilbert:decomposition}). Criterium (iv) van
\cref{thm:b3:hilbert:parseval} besluit; (ii) en (iii) vertalen
zich in de weergegeven formules (na herindexering van het
aftelbare $\Z$; de tweezijdige reeks convergeert
onvoorwaardelijk --- de partiële sommen over elke uitputtende
familie convergeren, volgens het staartargument in $\ell^2$).
\end{proof}

\begin{theorem}[Lax--Milgram]\label{thm:b3:hilbert:laxmilgram}
Zij $H$ een reële \omterm{def:b3:hilbert:inner}{hilbertruimte} en $a \colon H\times H \to \R$
bilineair, \emph{\omterm{def:b3:topology:continuity}{continu}} ($\abs{a(u,v)} \leq M\norm u\norm v$)
en \emph{coërcief} ($a(u, u) \geq \alpha\norm u^2$, met $\alpha >
0$). Dan is er voor elke $\varphi \in H'$ een unieke $u \in H$
met\index{stelling van Lax--Milgram}
\[
a(u, v) = \varphi(v) \qquad \text{voor alle } v \in H .
\]
\end{theorem}

\begin{proof}
Voor vaste $u$ is $v \mapsto a(u, v)$ \omterm{def:b3:topology:continuity}{continu} lineair: Riesz
geeft een unieke $Au \in H$ met $a(u,v) = \langle Au, v\rangle$;
$A$ is lineair met $\norm{Au} \leq M\norm u$ (uniciteit van de
vertegenwoordigers, dan de afschatting). Coërciviteit:
$\alpha\norm u^2 \leq a(u,u) = \langle Au, u\rangle \leq
\norm{Au}\norm u$, dus $\norm{Au} \geq \alpha\norm u$: $A$ is
injectief met gesloten beeld (een cauchyrij $Au_n$ in het beeld
dwingt af dat $(u_n)$ cauchy is). Het beeld ligt \omterm{def:b3:topology:interior}{dicht}: staat $w$
loodrecht op $\operatorname{im}A$, dan is $0 = \langle Aw,
w\rangle \geq \alpha\norm w^2$. Gesloten en \omterm{def:b3:topology:interior}{dicht}: dus is $A$
bijectief. Zij $\varphi$ gegeven, $f$ haar vertegenwoordiger
(Riesz) en $u = A^{-1}f$: dan is $a(u, v) = \langle f, v\rangle =
\varphi(v)$, en dat op unieke wijze ($a(u - u', \cdot) = 0$ plus
coërciviteit).
\end{proof}

\begin{remark}\label{rem:b3:hilbert:variational}
Is $a$ symmetrisch, dan is de oplossing van Lax--Milgram de
unieke minimalisator van de \emph{energie} $J(v) = \frac12a(v,v)
- \varphi(v)$ (\cref{exo:b3:hilbert:9}): het bestaan van
oplossingen van variatieproblemen in één klap. Toegepast op
geschikte functieruimten (de sobolevruimten van een latere
cursus) lost dit randwaardeproblemen voor
differentiaalvergelijkingen op --- de moderne toegangspoort tot
de partiële differentiaalvergelijkingen.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b3:hilbert:1}
(a) Bewijs de polarisatie-identiteiten (reëel: $4\langle x,
y\rangle = \norm{x+y}^2 - \norm{x-y}^2$; complex: de versie met
vier termen).
(b) Toon aan dat $\norm\cdot_1$ op $L^1(\intcc01)$ en
$\norm\cdot_\infty$ op $\mathcal C(\intcc01)$ de
parallellogramwet schenden: deze normen komen van geen enkel
\omterm{def:b3:hilbert:inner}{inwendig product}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:hilbert:2}
In $H = L^2(\intcc01)$ (reëel):
(a) bereken de projectie van $f$ op de deelruimte van de
constante functies, en interpreteer;
(b) bereken de projectie op $\{g : g = 0 \text{ b.o.\ op }
\intcc0{1/2}\}$;
(c) bereken $d\bigl(x \mapsto x,\ \operatorname{Vect}(\mathbf
1)\bigr)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hilbert:3}
(a) Toon voor een deelruimte $F$ aan: $F$ ligt \omterm{def:b3:topology:interior}{dicht} $\iff$
$F^\perp = \{0\}$, en geef in $\ell^2$ een voorbeeld van een
\emph{echte} \omterm{def:b3:topology:interior}{dichte} deelruimte (dus $F^\perp = 0$ zonder dat $F =
H$: de ontbindingsstelling heeft $F$ gesloten werkelijk nodig).
(b) Toon aan dat uit $x_n \to x$ en $y_n \to y$ in norm volgt dat
$\langle x_n, y_n\rangle \to \langle x, y\rangle$, en wijs twee
plaatsen aan waar het hoofdstuk deze \omterm{def:b3:topology:continuity}{continuïteit} gebruikte.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hilbert:4}
Pas Gram--Schmidt toe op $1, x, x^2$ in $L^2(\intcc{-1}1)$
(\omterm{def:b3:measure:lebesgueouter}{lebesguemaat}): verkrijg de eerste drie genormeerde
\emph{legendreveeltermen} en ga na dat ze samenvallen met
$\sqrt{n + \frac12}\,P_n$ voor de rodriguesveeltermen $P_n$ van
\cref{pb:b3:hilbert:1}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hilbert:5}
Pas \omterm{thm:b3:hilbert:parseval}{Parseval} (\cref{thm:b3:hilbert:fourier}) toe op $f(t) = t$ en
$f(t) = t^2$ op $\intcc{-\pi}\pi$ --- nu voor deze functies
rechtmatig (\omterm{def:b3:topology:continuity}{continu}, maar vroeger vroeg de identiteit om
zorgvuldigheid met stuksgewijs $\mathcal C^1$ bij de sprong aan
de rand): vind opnieuw
\[
\sum_{n\geq1}\frac1{n^2} = \frac{\pi^2}6,
\qquad
\sum_{n\geq1}\frac1{n^4} = \frac{\pi^4}{90} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hilbert:6}
(a) Zoek $a \in L^2(\intcc01)$ met $\int_0^{1/2}f = \langle a,
f\rangle$ voor alle $f$; bereken $\norm\varphi$ voor deze
functionaal.
(b) Toon aan dat de evaluatie $f \mapsto f(\frac12)$,
gedefinieerd op de deelruimte $\mathcal C(\intcc01) \subseteq
L^2(\intcc01)$, \emph{niet} \omterm{def:b3:topology:continuity}{continu} is voor $\norm\cdot_2$: er
bestaat geen vertegenwoordiger van Riesz (evaluatie is geen
$L^2$-begrip).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hilbert:7}
Zij $H$ separabel met \omterm{def:b3:hilbert:onb}{hilbertbasis} $(e_n)$, en $(x_k)$ een
begrensde rij.
(a) Toon aan dat een deelrij \emph{zwak} convergeert: er is een
$x$ met $\langle y, x_{k_j}\rangle \to \langle y, x\rangle$ voor
elke $y \in H$. \emph{(Diagonaalextractie op de coëfficiënten
$\langle e_n, x_k\rangle$; stel $x$ samen met Bessel en de
uniforme begrensdheid van de normen.)}
(b) Toon aan dat $e_n \rightharpoonup 0$ terwijl $\norm{e_n} =
1$: zwakke limieten kunnen norm verliezen. Toon in (a) aan dat
$\norm x \leq \liminf\norm{x_{k_j}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hilbert:8}
(Toegevoegde operatoren) Toon voor $T \in \mathcal L(H)$ aan dat
er een unieke $T^* \in \mathcal L(H)$ is met $\langle Tx,
y\rangle = \langle x, T^*y\rangle$ (Riesz), en dat $\vertiii{T^*}
= \vertiii T$. Bereken de toegevoegde van de verschuiving $S$ op
$\ell^2$ en bewijs $\ker T^* = (\operatorname{im}T)^\perp$ ---
leid daaruit $\overline{\operatorname{im}T} = (\ker T^*)^\perp$
af.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hilbert:9}
Zij $a$ als in Lax--Milgram en bovendien \emph{symmetrisch}. Toon
aan dat $u$ de vergelijking $a(u, \cdot) = \varphi$ oplost dan en
slechts dan als $u$ de functionaal $J(v) = \frac12a(v, v) -
\varphi(v)$ minimaliseert, en dat het minimum in precies één punt
wordt bereikt. \emph{(Kwadraat afsplitsen: $J(u + w) - J(u) =
\frac12a(w,w) \geq \frac\alpha2\norm w^2$.)} Toepassing: leid de
projectiestelling voor gesloten deelruimten opnieuw af uit
Lax--Milgram.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hilbert:10}
(Het stelsel van Haar) Stel op $\intcc01$ $h_{0} = \mathbf 1$,
en voor $n = 2^j + k$ ($j \geq 0$, $0 \leq k < 2^j$):
\[
h_n = 2^{j/2}\Bigl(\mathbf 1_{[k2^{-j},\,(k +
\frac12)2^{-j})} - \mathbf 1_{[(k+\frac12)2^{-j},\,(k+1)2^{-j})}
\Bigr).
\]
Toon aan dat $(h_n)_{n\geq0}$ orthonormaal en totaal is in
$L^2(\intcc01)$. \emph{(Orthogonaliteit: disjuncte of
geneste dragers; totaliteit: de eindige opspansels bevatten alle
dyadische trapfuncties, en die liggen \omterm{def:b3:topology:interior}{dicht} --- via
\cref{thm:b3:lp:density}(1) en dyadische benadering van
intervallen.)} Het stelsel van Haar is de voorvader van de
wavelets.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:hilbert:11}
(Orthogonale projecties, gekarakteriseerd) Zij $H$ een
\omterm{def:b3:hilbert:inner}{hilbertruimte} en $P \in \mathcal L(H)$ met $P^2 = P$ en $P \neq
0$. Toon de gelijkwaardigheid aan van:
(i) $P$ is de orthogonale projectie op $\operatorname{im}P$;
(ii) $P = P^*$ (\cref{exo:b3:hilbert:8});
(iii) $\vertiii P = 1$.
\emph{(Voor (iii) $\Rightarrow$ (i): had een $x \in (\ker
P)^\perp$ $Px \neq x$, beschouw dan $x + t(Px - x)$ --- of
rechtstreeks: ontwikkel voor $u \in \operatorname{im}P$ en $v \in
\ker P$ de ongelijkheid $\norm{P(u + tv)}^2 \leq \norm{u +
tv}^2$ voor alle $t \in \R$ en besluit dat $\langle u, v\rangle =
0$.)}
Geef een niet-orthogonale projectie op $\R^2$ en bereken haar
norm.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:hilbert:12}
(Ergodenstelling van von Neumann) Zij $U \in \mathcal L(H)$
\emph{unitair} ($U^*U = UU^* = I$), $F = \ker(U - I)$ de vaste
ruimte, $P$ de orthogonale projectie op $F$, en $A_n =
\frac1n\sum_{k=0}^{n-1}U^k$.
(a) Toon aan dat $\ker(U - I) = \ker(U^* - I)$ \emph{(uit
$\norm{Ux - x}^2 = 2\norm x^2 - 2\operatorname{Re}\langle Ux,
x\rangle$ en de unitariteit)}, en leid
$\overline{\operatorname{im}(U - I)} = F^\perp$ af.
(b) Toon aan dat $A_nx \to x$ voor $x \in F$ en $A_nx \to 0$ voor
$x \in \operatorname{im}(U - I)$ \emph{(telescoperen)}, en
vervolgens voor $x \in \overline{\operatorname{im}(U - I)}$
(uniforme grens $\vertiii{A_n} \leq 1$).
(c) Besluit: $A_nx \to Px$ voor \emph{elke} $x \in H$ ---
tijdsgemiddelden convergeren naar de projectie op de
invarianten.
(d) Werk dat uit voor $H = L^2(\R/\Z)$ en $Uf = f(\cdot +
\alpha)$ met $\alpha$ irrationaal: identificeer $F$ (gebruik
fourierreeksen, \cref{thm:b3:hilbert:fourier}) en leid af dat
$\frac1n\sum_{k<n}f(x + k\alpha) \to \int_0^1f$ in $L^2$: de
gelijkverdeling in $L^2$ van irrationale rotaties.
\end{exercise}

\section{Probleem: orthogonale veeltermen}

\begin{problem}[{Weekendopgave --- Legendre, Hermite en de
kwadratuur van Gauss}]\label{pb:b3:hilbert:1}
Zij $I \subseteq \R$ een interval en $w > 0$ een \omterm{def:b3:topology:continuity}{continu}
\emph{gewicht} op het inwendige van $I$ zodat $\int_I \abs
t^nw(t)\dd t < \infty$ voor alle $n$; we werken in $H = L^2(I,
w\,\dd\lambda)$ met $\langle f, g\rangle = \int_I \bar fg\,w$.
Gram--Schmidt toegepast op $1, t, t^2, \dots$ levert de
\emph{\omterm{pb:b3:hilbert:1}{orthogonale veeltermen}} $(p_n)$ voor $w$ op (monische
normering: $p_n = t^n + \cdots$).\index{orthogonale veeltermen}

\medskip
\noindent\textbf{Deel I --- Algemene theorie.}
\begin{enumerate}
  \item Toon aan dat $p_n$ loodrecht staat op elke veelterm van
        graad $< n$, en dat $(p_0, \dots, p_n)$ een basis van
        $\R_n[t]$ is.
  \item (Driepuntsrecursie) Toon aan dat er reële getallen $a_n,
        b_n$ bestaan met
        \[
        p_{n+1}(t) = (t - a_n)\,p_n(t) - b_n\,p_{n-1}(t),
        \qquad b_n = \frac{\norm{p_n}^2}{\norm{p_{n-1}}^2} >
        0 .
        \]
        \emph{(Ontwikkel $t\,p_n$ in de basis $(p_k)_{k \leq
        n+1}$ en dood coëfficiënten met de orthogonaliteit,
        gebruikmakend van $\langle tp_n, p_k\rangle = \langle
        p_n, tp_k\rangle$.)}
  \item (Wortels) Toon aan dat $p_n$ $n$ \emph{verschillende}
        wortels heeft, alle inwendig in $I$. \emph{(Zij $t_1 <
        \dots < t_m$ de inwendige tekenwisselingen van $p_n$; is
        $m < n$, test $p_n$ dan tegen $\prod_{i\leq m}(t - t_i)$
        en weerspreek de orthogonaliteit.)}
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Legendre ($I = \intcc{-1}1$, $w =
1$).} Definieer $P_n(t) = \frac{1}{2^nn!}\,\frac{\dd^n}{\dd
t^n}\bigl[(t^2 - 1)^n\bigr]$ (Rodrigues).
\begin{enumerate}[resume]
  \item Toon aan dat $\deg P_n = n$ met kopcoëfficiënt
        $\frac{(2n)!}{2^n(n!)^2}$, en, door $n$ maal partieel te
        integreren, dat $\langle P_n, Q\rangle = 0$ voor elke
        veelterm $Q$ van graad $< n$: de $P_n$ zijn (op de
        normering na) de \omterm{pb:b3:hilbert:1}{orthogonale veeltermen} voor $w = 1$.
  \item Bereken $\norm{P_n}_2^2 = \frac{2}{2n+1}$
        \emph{(integreer $n$ maal partieel tegen zichzelf en
        herleid tot een bèta- of wallisintegraal,
        \cref{exo:b3:product:8})}.
  \item Toon aan dat de genormeerde legendreveeltermen een
        \omterm{def:b3:hilbert:onb}{hilbertbasis} van $L^2(\intcc{-1}1)$ vormen
        \emph{(Weierstrass, \cref{cor:b3:complete:weierstrass},
        plus de \omterm{ex:b3:lebesgue:gamma}{dichtheid} van $\mathcal C$ in $L^2$)}, en
        ontwikkel $f(t) = \abs t$ tot graad $2$: bereken de beste
        kwadratische $L^2$-benadering van $\abs t$.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Hermite ($I = \R$, $w(t) =
\eu^{-t^2}$).} Definieer $H_n(t) =
(-1)^n\eu^{t^2}\frac{\dd^n}{\dd t^n}\eu^{-t^2}$.
\begin{enumerate}[resume]
  \item Toon aan dat $H_n$ een veelterm van graad $n$ is met
        kopcoëfficiënt $2^n$, dat $H_{n+1} = 2tH_n - H_n'$, en
        dat $\langle H_m, H_n\rangle_w = \delta_{mn}\,2^nn!
        \sqrt\pi$ \emph{(opnieuw partieel integreren)}.
  \item Toon aan dat de familie van Hermite totaal is in
        $L^2(\R, \eu^{-t^2}\dd t)$, waarbij je één resultaat uit
        \cref{ch:b3:fouriertransform} zonder bewijs mag
        aannemen: heeft $g \in L^1(\R)$ $\int g(t)\eu^{-\iu\xi
        t}\dd t = 0$ voor alle $\xi$, dan is $g = 0$ b.o.
        \emph{(Voor $f$ loodrecht op alle $H_n$, dus loodrecht op
        alle veeltermen: toon aan dat $z \mapsto \int
        f(t)\eu^{-t^2}\eu^{-\iu zt}\dd t$ goed gedefinieerd is,
        ontwikkel de exponentiële functie in een reeks,
        rechtvaardig het verwisselen met dominantie en besluit
        dat de fouriergetransformeerde van $f\eu^{-t^2}$
        verdwijnt.)}
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De kwadratuur van Gauss.} Leg $n$
vast, zij $t_1 < \dots < t_n$ de wortels van $p_n$ (Deel I), en
definieer de gewichten $w_i = \int_I \ell_i(t)\,w(t)\dd t$,
waarbij $\ell_i$ de lagrange-interpolatiebasisveeltermen in de
$t_i$ zijn.
\begin{enumerate}[resume]
  \item Toon aan dat de kwadratuurregel $Q(f) = \sum_iw_if(t_i)$
        exact is op alle veeltermen van graad $\leq n - 1$
        (interpolatie), en in feite --- het wonder --- op alle
        veeltermen van graad $\leq 2n - 1$: schrijf $P = qp_n +
        r$ en gebruik de orthogonaliteit op het quotiënt $q$.
  \item Toon aan dat de gewichten positief zijn \emph{(pas de
        regel toe op $\ell_i^2$, van graad $2n - 2$)}, en leid
        uit de stelling van Polya (\cref{exo:b3:banach:9}) af
        dat de kwadratuur van Gauss convergeert: $Q_n(f) \to
        \int_I fw$ voor elke \omterm{def:b3:topology:continuity}{continue} $f$ op een \omterm{def:b3:topology:compact}{compacte} $I$.
  \item Bereken voor $n = 2$, $I = \intcc{-1}1$, $w = 1$ de
        knopen $\pm\frac1{\sqrt3}$ en de gewichten $1, 1$, en ga
        de exactheid op $1, t, t^2, t^3$ met de hand na.
        Vergelijk met de trapeziumregel in dezelfde twee
        evaluatiepunten.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Chebyshev: de veeltermen die het
best oscilleren.} Nu is $I = \intcc{-1}1$ en $w(t) =
\frac1{\sqrt{1 - t^2}}$.
\begin{enumerate}[resume]
  \item Toon aan dat $T_n(\cos\theta) = \cos n\theta$ een
        veelterm $T_n$ van graad $n$ definieert (leid $T_{n+1} =
        2t\,T_n - T_{n-1}$ af uit een goniometrische identiteit),
        met kopcoëfficiënt $2^{n-1}$ voor $n \geq 1$; en dat de
        substitutie $t = \cos\theta$ geeft
        \[
        \langle T_m, T_n\rangle_w =
        \int_0^\pi\cos m\theta\,\cos n\theta\,\dd\theta
        = 0 \ (m \neq n), \qquad
        \norm{T_0}_w^2 = \pi,\ \ \norm{T_n}_w^2 = \frac\pi2 :
        \]
        de $T_n$ zijn de \omterm{pb:b3:hilbert:1}{orthogonale veeltermen} voor dit gewicht,
        en chebyshevontwikkelingen \emph{zijn} vermomde
        fouriercosinusreeksen.
  \item Lokaliseer expliciet de $n$ wortels $t_k =
        \cos\frac{(2k-1)\pi}{2n}$ en de $n + 1$ extrema $s_j =
        \cos\frac{j\pi}n$ van $T_n$ op $\intcc{-1}1$, waar
        $T_n(s_j) = (-1)^j$: de grafiek \emph{oscilleert
        gelijkmatig} tussen $\pm1$.
  \item (Minimax) Toon aan dat onder alle \emph{monische}
        veeltermen van graad $n$ de veelterm $2^{1-n}T_n$ de
        kleinste supremumnorm op $\intcc{-1}1$ heeft, namelijk
        $2^{1-n}$ --- en dat ze de unieke minimalisator is.
        \emph{(Had een monische $P$ $\sup\abs P < 2^{1-n}$, dan
        zou het verschil $2^{1-n}T_n - P$, van graad $\leq n-1$,
        in de $n+1$ punten van gelijkmatige oscillatie van teken
        wisselen.)}
  \item Toepassing op interpolatie: voor knopen $t_1 < \dots <
        t_n$ in $\intcc{-1}1$ bevat de fout van de
        lagrange-interpolatie van een $\mathcal C^n$-functie de
        factor $\omega(t) = \prod_i(t - t_i)$. Toon aan dat de
        keuze van de chebyshevwortels als knopen
        $\sup_{\intcc{-1}1}\abs\omega$ minimaliseert, en geef de
        resulterende grens $\norm{f - L_nf}_\infty \leq
        \frac{\norm{f^{(n)}}_\infty} {2^{n-1}\,n!}$ ---
        vergelijk met gelijkmatig verdeelde knopen (vermeld het
        verschijnsel van Runge als waarschuwing).
  \item Ga na dat $\abs{T_n'(\pm1)} = n^2$ \emph{(differentieer
        $T_n(\cos\theta) = \cos n\theta$ en neem de limieten
        $\theta \to 0, \pi$)}: veeltermen die op $\intcc{-1}1$
        door $1$ begrensd zijn, kunnen aan de rand een afgeleide
        zo groot als $n^2$ hebben (de ongelijkheid van Markov
        zegt dat het niet groter kan --- enkel de uitspraak).
        Waar in het interval is de grens voor de afgeleide
        slechts $O(n)$?
  \item (Kwadratuur van Chebyshev--Gauss) Toon aan dat de
        gaussregel voor het gewicht $w$ in de $n$
        chebyshevwortels \emph{gelijke} gewichten $w_i = \frac\pi
        n$ heeft \emph{(exactheid op $T_0, \dots, T_{n-1}$ plus
        de goniometrische sommen
        $\sum_{k=1}^n\cos\bigl(j\tfrac{(2k-1)\pi}{2n}\bigr) = 0$
        voor $1 \leq j \leq n - 1$)}: de meest uniforme van alle
        kwadraturen. Schrijf haar uit voor $n = 3$.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Christoffel--Darboux, verstrengeling
en de jacobimatrix.} Terug naar een algemeen gewicht; $h_k =
\norm{p_k}^2$ (met monische $p_k$) en $b_k = h_k/h_{k-1}$.
\begin{enumerate}[resume]
  \item (Kleinste norm) Toon aan dat onder alle \emph{monische}
        veeltermen van graad $n$ de orthogonale $p_n$ de unieke
        veelterm van minimale $L^2(w)$-norm is --- herken de
        minimalisering als een orthogonale projectie op
        $\R_{n-1}[t]$ (\cref{thm:b3:hilbert:projection}, of de
        eindigdimensionale projectie uit het volume van
        bachelorjaar 2). De minimaxeigenschap van vraag 14 is
        dezelfde uitspraak met $L^\infty$ in de plaats van
        $L^2$: dezelfde held, twee normen.
  \item (Christoffel--Darboux) Bewijs met inductie naar $n$,
        gebruikmakend van de driepuntsrecursie, de identiteit
        \[
        \sum_{k=0}^{n}\frac{p_k(x)\,p_k(y)}{h_k}
        = \frac{p_{n+1}(x)\,p_n(y) -
        p_n(x)\,p_{n+1}(y)}{h_n\,(x - y)}
        \qquad (x \neq y),
        \]
        en haar samenvloeiende vorm ($y \to x$):
        $\sum_{k\leq n}\frac{p_k(x)^2}{h_k} =
        \frac{p_{n+1}'(x)p_n(x) - p_n'(x)p_{n+1}(x)}{h_n}$.
  \item Leid af dat $p_n$ en $p_{n+1}$ geen gemeenschappelijke
        wortel hebben, en dat in elke wortel $x_0$ van $p_{n+1}$
        geldt $p_n(x_0)\,p_{n+1}'(x_0) > 0$. Besluit tot de
        \emph{verstrengeling} van de wortels: tussen twee
        opeenvolgende wortels van $p_{n+1}$ ligt precies één
        wortel van $p_n$.
  \item (Jacobimatrix) Zij $J_n$ de symmetrische tridiagonale
        $n\times n$-matrix met diagonaal $a_0, \dots, a_{n-1}$ en
        nevendiagonale elementen $\sqrt{b_1}, \dots,
        \sqrt{b_{n-1}}$. Toon met inductie aan dat $\det(tI_n -
        J_n) = p_n(t)$, zodat de wortels van $p_n$ de
        eigenwaarden van een reële symmetrische matrix zijn ---
        wat in één regel opnieuw bewijst dat ze reëel zijn, en
        (met de verstrengeling hierboven) de \omterm{pb:b3:hilbert:1}{orthogonale
        veeltermen} aan de spectrale wereld van
        \cref{ch:b3:spectral} bindt.
  \item (Synthese) Stel het woordenboek samen voor de drie
        klassieke families (Legendre, Hermite, Chebyshev):
        interval, gewicht, definiërende formule,
        driepuntsrecursie, norm en de natuurlijke habitat van
        elk (kwadratuur en benadering op \omterm{def:b3:topology:compact}{compacte} verzamelingen;
        gaussische analyse; minimax- en
        fouriercosinusmethoden). Eén zin over wat de algemene
        theorie (Delen I en VI) heeft opgeleverd wat geen enkele
        afzonderlijke berekening kon geven.
\end{enumerate}

\medskip
\noindent\textbf{Deel VII --- De foutterm, en de kern achter de
gewichten.} Hier is $I$ \omterm{def:b3:topology:compact}{compact} en $f \in \mathcal C^{2n}(I)$.
\begin{enumerate}[resume]
  \item (Foutformule van Gauss) Zij $Hf$ de
        hermite-interpolant van graad $\leq 2n - 1$ die $f$ en
        $f'$ in de knopen $t_1, \dots, t_n$ vastlegt (bewijs
        haar bestaan en de puntsgewijze fout
        \[
        f(t) - Hf(t) =
        \frac{f^{(2n)}(\xi_t)}{(2n)!}\;p_n(t)^2
        \]
        met het gebruikelijke argument met een hulpfunctie).
        Leid, door deze identiteit tegen $w$ te integreren en in
        te klemmen tussen de extrema van $f^{(2n)}$, af dat
        \[
        \int_I f\,w - Q_n(f)
        = \frac{f^{(2n)}(\xi)}{(2n)!}\,h_n
        \qquad\text{voor een } \xi \in I,
        \]
        met $h_n = \norm{p_n}^2$ als in Deel VI: de kwadratuur
        van Gauss vergist zich met één $2n$-de afgeleide,
        gewogen met de gekwadrateerde norm van de monische
        orthogonale veelterm.
  \item (De gewichten zijn christoffelwaarden) Bewijs met de
        reproducerende kern $K_n(x, y) = \sum_{k=0}^{n-1}
        \frac{p_k(x)p_k(y)}{h_k}$ van $\R_{n-1}[t]$ en de
        exactheid van $Q_n$ tot graad $2n - 2$ dat
        \[
        w_i \;=\;
        \Bigl(\,\sum_{k=0}^{n-1}
        \frac{p_k(t_i)^2}{h_k}\Bigr)^{\!-1} :
        \]
        elk gewicht is de waarde in zijn eigen knoop van de
        \emph{functie van Christoffel} --- opnieuw de
        positiviteit van de gewichten (vraag 10), nu met een
        exacte formule. Ga na dat ze $w_1 = w_2 = 1$ teruggeeft
        voor $n = 2$, $I = \intcc{-1}1$, $w = 1$.
  \item (Alles klopt op één integraal) Bereken voor het
        chebyshevgewicht en $n = 3$ knopen beide leden van
        \[
        \int_{-1}^{1}\frac{t^6}{\sqrt{1 - t^2}}\,\dd t
        = \frac{5\pi}{16},
        \qquad
        Q_3(t^6) = \frac{9\pi}{32},
        \]
        zodat de kwadratuurfout precies $\frac{\pi}{32}$ is; ga
        vervolgens na dat de foutformule van vraag 23 precies
        deze waarde voorspelt (hier is $f^{(6)} = 6!$ constant,
        en $h_3 = \norm{2^{-2}T_3}_w^2 = \frac\pi{32}$): theorie
        en berekening stemmen tot op het laatste cijfer overeen.
\end{enumerate}
\end{problem}
