\chapter{De fouriertransformatie}\label{ch:b3:fouriertransform}

Fourierreeksen ontbinden periodieke signalen in discrete
harmonischen; de fourier\emph{transformatie} doet hetzelfde voor
signalen op de hele lijn, met een continuüm aan frequenties. Ze
zet differentiëren om in vermenigvuldigen, convolutie in
producten, en gaussische functies in gaussische functies --- de
redenen waarom ze differentiaalvergelijkingen oplost, de
signaalverwerking aandrijft en in \cref{ch:b3:clt} de centrale
limietstelling zal bewijzen. Dit hoofdstuk ontwikkelt de
$L^1$-theorie (Riemann--Lebesgue, inversie, injectiviteit), de
\omterm{def:b3:fouriertransform:schwartz}{klasse van Schwartz} waarin de transformatie een volmaakte
bijectie is, en de $L^2$-theorie (Plancherel: de transformatie is
op een constante na een unitaire operator), met twee
paradepaardjes als toepassing: de warmtevergelijking, in de
weekendopgave van begin tot eind opgelost, en de sommatieformule
van Poisson. Afspraak:
\[
\hat f(\xi) = \int_\R f(x)\,\eu^{-\iu\xi x}\,\dd x .
\]

\section{De transformatie op \texorpdfstring{$L^1$}{L1}}

\begin{proposition}\label{prop:b3:fouriertransform:basic}
Voor $f \in L^1(\R)$ is $\hat f$ goed gedefinieerd, begrensd
($\norm{\hat f}_\infty \leq \norm f_1$) en \omterm{def:b3:topology:continuity}{continu}, en geldt:
\begin{enumerate}
  \item $\widehat{\tau_af}(\xi) = \eu^{-\iu a\xi}\hat f(\xi)$ en
        $\widehat{\eu^{\iu ax}f}(\xi) = \hat f(\xi - a)$;
  \item $\widehat{f(\cdot/\lambda)}(\xi) = \lambda\hat
        f(\lambda\xi)$ voor $\lambda > 0$;
  \item is $xf \in L^1$, dan is $\hat f$ $\mathcal C^1$ met
        $(\hat f)'(\xi) = \widehat{(-\iu x)f}(\xi)$;
  \item is $f \in \mathcal C^1$ met $f' \in L^1$ (en $f \to 0$ in
        $\pm\infty$, hier automatisch), dan is
        $\widehat{f'}(\xi) = \iu\xi\hat f(\xi)$;
  \item $\widehat{f * g} = \hat f\,\hat g$ voor $f, g \in L^1$.
\end{enumerate}
\end{proposition}

\begin{proof}
Begrensdheid: $\abs{\hat f} \leq \int\abs f$. \omterm{def:b3:topology:continuity}{Continuïteit}:
gedomineerde convergentie met dominant $\abs f$
(\cref{thm:b3:lebesgue:paramcont}). (1), (2): substituties
(\cref{thm:b3:product:linearchange}). (3): differentiëren onder
de integraal, met dominant $\abs{xf}$
(\cref{thm:b3:lebesgue:paramdiff}). (4): eerst heeft $f(x) = f(0)
+ \int_0^xf'$ een limiet in $\pm\infty$ (want $f' \in L^1$), die
$0$ moet zijn (want $f \in L^1$); integreer daarna partieel op
$[-A, A]$ en laat $A \to \infty$ gaan. (5): Fubini, geoorloofd
omdat $(x,y)\mapsto f(x - y)g(y)\eu^{-\iu\xi x}$ absoluut
\omterm{def:b3:lebesgue:l1}{integreerbaar} is (\cref{thm:b3:product:convolution}):
\[
\widehat{f*g}(\xi) = \iint f(x - y)g(y)\eu^{-\iu\xi(x - y)}
\eu^{-\iu\xi y}\dd x\,\dd y = \hat f(\xi)\,\hat g(\xi).
\]
\end{proof}

\begin{example}\label{ex:b3:fouriertransform:gaussian}
De gaussische functie: voor $a > 0$ is\index{gaussische functie,
fouriergetransformeerde}
\[
\widehat{\eu^{-ax^2}}(\xi) =
\sqrt{\frac\pi a}\;\eu^{-\xi^2/4a} :
\]
volgens \cref{exo:b3:lebesgue:7} (de truc met de
differentiaalvergelijking $F' = -\frac\xi{2}F$, herschaald), of
volgens (3): $g = \widehat{\eu^{-ax^2}}$ voldoet aan $g'(\xi) =
-\frac{\xi}{2a}g(\xi)$ (partieel integreren) met $g(0) =
\sqrt{\pi/a}$. Gaussische functies zijn op een schaalfactor na
vaste punten van de transformatie --- de diepe reden waarom ze de
centrale limietstelling beheersen.
\end{example}

\begin{theorem}[Riemann--Lebesgue]
\label{thm:b3:fouriertransform:riemannlebesgue}
Voor $f \in L^1(\R)$ geldt $\hat f(\xi) \to 0$ als $\abs\xi \to
\infty$. Dus beeldt $\widehat{\phantom f}$ $L^1$ af in $\mathcal
C_0(\R)$ (de \omterm{def:b3:topology:continuity}{continue} functies die in oneindig
verdwijnen).\index{lemma van Riemann--Lebesgue}
\end{theorem}

\begin{proof}
Voor de indicator van een interval is $\widehat{\mathbf 1_{\intcc
ab}}(\xi) = \frac{\eu^{-\iu a\xi} - \eu^{-\iu b\xi}}{\iu\xi} \to
0$; dus ook voor trapfuncties. Trapfuncties liggen \omterm{def:b3:topology:interior}{dicht} in $L^1$
(\cref{thm:b3:lp:density}(1) plus de benadering van
verzamelingen van eindige \omterm{def:b3:measure:measure}{maat} door eindige verenigingen van
intervallen, \cref{exo:b3:measure:7}), en de transformatie is
\omterm{def:b3:topology:continuity}{continu} van $\norm\cdot_1$ naar $\norm\cdot_\infty$: is $\norm{f
- s}_1 < \varepsilon$, dan is
$\limsup_{\abs\xi\to\infty}\abs{\hat f(\xi)} \leq \varepsilon$.
\end{proof}

\section{Inversie en injectiviteit}

\begin{lemma}[Vermenigvuldigingsformule]
\label{lem:b3:fouriertransform:multiplication}
Voor $f, g \in L^1(\R)$ is $\displaystyle\int \hat f\,g = \int
f\,\hat g$.
\end{lemma}

\begin{proof}
Beide leden zijn gelijk aan $\iint f(x)g(\xi)\eu^{-\iu x\xi}\dd
x\,\dd\xi$ (Tonelli--Fubini: de dubbele integraal van de absolute
waarde is $\norm f_1\norm g_1$).
\end{proof}

\begin{theorem}[Inversie]\label{thm:b3:fouriertransform:inversion}
Zij $f \in L^1(\R)$.\index{inversiestelling van Fourier}
\begin{enumerate}
  \item (Gaussische sommeerbaarheid) Voor elke $x$ geldt
        \[
        (f * g_\varepsilon)(x) =
        \frac1{2\pi}\int_\R \hat f(\xi)\,
        \eu^{-\varepsilon\xi^2}\,\eu^{\iu x\xi}\,\dd\xi,
        \qquad\text{met } g_\varepsilon(y) =
        \frac{1}{2\sqrt{\pi\varepsilon}}\,
        \eu^{-y^2/4\varepsilon},
        \]
        en $f * g_\varepsilon \to f$ in $L^1$ als $\varepsilon
        \to 0$.
  \item Is bovendien $\hat f \in L^1$, dan is voor bijna elke
        $x$
        \[
        f(x) = \frac{1}{2\pi}\int_\R \hat
        f(\xi)\,\eu^{\iu x\xi}\,\dd\xi ,
        \]
        en heeft $f$ een \omterm{def:b3:topology:continuity}{continue} vertegenwoordiger.
  \item (Injectiviteit) Is $\hat f = 0$, dan is $f = 0$ b.o.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Leg $x$ vast en pas
\cref{lem:b3:fouriertransform:multiplication} toe op $f$ en
$g(\xi) = \frac1{2\pi}\eu^{-\varepsilon\xi^2}\eu^{\iu x\xi}$:
volgens \cref{ex:b3:fouriertransform:gaussian} (met de
modulatieregel) is
\[
\hat g(y) = \frac1{2\pi}\sqrt{\frac\pi\varepsilon}\,
\eu^{-(y - x)^2/4\varepsilon} = g_\varepsilon(x - y),
\]
dus $\frac1{2\pi}\int\hat f(\xi)\eu^{-\varepsilon\xi^2} \eu^{\iu
x\xi}\dd\xi = \int f(y)g_\varepsilon(x - y)\dd y =
(f*g_\varepsilon)(x)$. De $g_\varepsilon$ vormen een benaderende
eenheid: $g_\varepsilon \geq 0$, $\int g_\varepsilon = 1$ (de
gaussische integraal), geconcentreerd rond $0$; het bewijs van
\cref{thm:b3:lp:regularization}(2) gaat woordelijk door (er werd
enkel $\int g_\varepsilon = 1$ en de concentratie gebruikt: voor
de staart is $\int_{\abs y > \delta}g_\varepsilon \to 0$): dus
$\norm{f * g_\varepsilon - f}_1 \to 0$.

(2) Is $\hat f \in L^1$, dan convergeert het rechterlid van (1),
volgens de gedomineerde convergentie (dominant $\abs{\hat f}$),
voor \emph{elke} $x$ naar $\frac1{2\pi}\int\hat f(\xi)\eu^{\iu
x\xi}\dd\xi$, en die limietfunctie is \omterm{def:b3:topology:continuity}{continu} (opnieuw
gedomineerde convergentie). Anderzijds is $f * g_\varepsilon \to
f$ in $L^1$, dus langs een deelrij b.o.\
(\cref{thm:b3:lp:complete}): de twee limieten vallen b.o.\ samen.

(3) $\hat f = 0$ laat het rechterlid van (1) verdwijnen: dus $f *
g_\varepsilon = 0$ voor alle $\varepsilon$, terwijl $f *
g_\varepsilon \to f$ in $L^1$: bijgevolg $f = 0$ b.o.
\end{proof}

\section{De klasse van Schwartz}

\begin{definition}\label{def:b3:fouriertransform:schwartz}
De \emph{klasse van Schwartz}\index{klasse van Schwartz}
$\mathcal S(\R)$ bestaat uit de $\mathcal C^\infty$-functies $f$
met $\sup_x\abs{x^m f^{(n)}(x)} < \infty$ voor alle $m, n \geq 0$
(alle afgeleiden dalen sneller dan elke macht). Voorbeelden:
$\eu^{-ax^2}$, $\mathcal C_c^\infty$. Duidelijk is $\mathcal S
\subseteq L^p$ voor elke $p$ (afschatten met $C(1 + x^2)^{-1}$),
en $\mathcal S$ is stabiel onder afgeleiden, vermenigvuldiging
met veeltermen en producten.
\end{definition}

\begin{theorem}\label{thm:b3:fouriertransform:schwartz}
De fouriertransformatie beeldt $\mathcal S(\R)$ bijectief op
zichzelf af, met inverse $\check g(x) = \frac1{2\pi}\int
g(\xi)\eu^{\iu x\xi}\dd\xi$.
\end{theorem}

\begin{proof}
Zij $f \in \mathcal S$. Iteratie van
\cref{prop:b3:fouriertransform:basic}(3) geeft $(\hat f)^{(n)} =
\widehat{(-\iu x)^nf}$ (elke $x^kf$ ligt in $L^1$); iteratie van
(4) met $h = (-\iu x)^nf \in \mathcal S$ (waarvan alle afgeleiden
\omterm{def:b3:lebesgue:l1}{integreerbaar} zijn) geeft $(\iu\xi)^m\hat h =
\widehat{h^{(m)}}$. Samen:
\[
\abs{\xi^m\,(\hat f)^{(n)}(\xi)}
= \bigl|\widehat{\,h^{(m)}}(\xi)\bigr|
\leq \bigl\|\bigl((-\iu x)^nf\bigr)^{(m)}\bigr\|_1 < \infty
\]
uniform in $\xi$: dus $\hat f \in \mathcal S$. Omdat $\hat f \in
L^1$, geldt de inversie
(\cref{thm:b3:fouriertransform:inversion}(2)) overal (beide leden
zijn \omterm{def:b3:topology:continuity}{continu}): $\check{\hat f} = f$, en symmetrisch
$\widehat{\check g} = g$ (de inverse transformatie is $g \mapsto
\frac1{2\pi}\hat g(-\cdot)$, en behoudt eveneens $\mathcal S$):
een bijectie.
\end{proof}

\section{Plancherel en \texorpdfstring{$L^2$}{L2}}

\begin{theorem}[Plancherel]\label{thm:b3:fouriertransform:plancherel}
Voor $f \in L^1 \cap L^2(\R)$ is\index{stelling van Plancherel}
\[
\norm{\hat f}_2^2 = 2\pi\,\norm f_2^2 .
\]
Bijgevolg breidt $\widehat{\phantom f}$ zich op unieke wijze uit
tot een \omterm{def:b3:topology:continuity}{continue} lineaire afbeelding $\mathcal F \colon L^2(\R)
\to L^2(\R)$ met $\norm{\mathcal Ff}_2 = \sqrt{2\pi}\norm f_2$;
$\mathcal F$ is bijectief, met $\mathcal F^{-1} =
\frac1{2\pi}\,\mathcal F\circ\sigma$ waarbij $\sigma f =
f(-\cdot)$, en ze behoudt de inwendige producten op de factor
$2\pi$ na.
\end{theorem}

\begin{proof}
Zij $f \in L^1\cap L^2$ en $h = f * \tilde f$ met $\tilde f(x) =
\overline{f(-x)}$. Dan is $h \in L^1$
(\cref{thm:b3:product:convolution}), is $h$ \omterm{def:b3:topology:continuity}{continu} en begrensd
(\cref{exo:b3:lp:6}: $f, \tilde f \in L^2$), is $h(0) = \int f\bar
f = \norm f_2^2$, en is $\hat h = \hat f\,\widehat{\tilde f} =
\hat f\,\overline{\hat f} = \abs{\hat f}^2 \geq 0$ (bereken
$\widehat{\tilde f} = \overline{\hat f}$). Pas
\cref{thm:b3:fouriertransform:inversion}(1) toe op $h$ in $x =
0$:
\[
(h * g_\varepsilon)(0) = \frac1{2\pi}\int \hat
h(\xi)\,\eu^{-\varepsilon\xi^2}\dd\xi .
\]
Als $\varepsilon \to 0$: het linkerlid gaat naar $h(0)$ ($h$
\omterm{def:b3:topology:continuity}{continu} en begrensd: $(h*g_\varepsilon)(0) - h(0) = \int(h(-y) -
h(0))g_\varepsilon(y)\dd y \to 0$ door kleine en grote $y$ te
splitsen); het rechterlid stijgt naar $\frac1{2\pi}\int\hat h$
volgens de monotone convergentie (want $\hat h \geq 0$).
Bijgevolg is $\frac1{2\pi}\int\abs{\hat f}^2 = \norm f_2^2$,
\emph{a priori} eindig of niet --- en dus eindig, wat zowel het
lidmaatschap als de identiteit bewijst.

Uitbreiding: $L^1\cap L^2 \supseteq \mathcal C_c$ ligt \omterm{def:b3:topology:interior}{dicht} in
$L^2$ (\cref{thm:b3:lp:density}); daar is de transformatie
$\sqrt{2\pi}$-isometrisch, dus breidt ze zich op unieke wijze uit
tot een op een constante na isometrische $\mathcal F$ op $L^2$
(\cref{thm:b3:complete:extension}). De inversie op $\mathcal S$
(\cref{thm:b3:fouriertransform:schwartz}) draagt met dezelfde
\omterm{ex:b3:lebesgue:gamma}{dichtheid} over (beide leden zijn $L^2$-continu): $\mathcal
F\bigl(\frac1{2\pi}\mathcal F(\sigma f)\bigr) = f$ op $\mathcal
S$, dus op $L^2$: bijectiviteit. Inwendige producten: polarisatie
uit de normidentiteit.
\end{proof}

\begin{theorem}[Sommatieformule van Poisson]
\label{thm:b3:fouriertransform:poisson}
Zij $f \in \mathcal S(\R)$ (een \omterm{def:b3:topology:continuity}{continue} $f$ met $\abs{f} +
\abs{\hat f} \leq C(1 + \abs\cdot)^{-2}$ volstaat). Dan
is\index{sommatieformule van Poisson}
\[
\sum_{n\in\Z} f(n) \;=\; \sum_{k\in\Z}\hat f(2\pi k) .
\]
\end{theorem}

\begin{proof}
Zij $F(x) = \sum_{n\in\Z}f(x + n)$: de reeks convergeert normaal
op \omterm{def:b3:topology:compact}{compacte} verzamelingen (het verval van $f$), dus is $F$
\omterm{def:b3:topology:continuity}{continu}, en ze is $1$-periodiek. Haar fouriercoëfficiënten
(periode $1$: $c_k(F) = \int_0^1F(t)\eu^{-2\iu\pi kt}\dd t$):
\[
c_k(F) = \sum_n\int_0^1 f(t + n)\,\eu^{-2\iu\pi kt}\dd t
= \int_\R f(t)\,\eu^{-2\iu\pi kt}\dd t = \hat f(2\pi k)
\]
(de normale convergentie rechtvaardigt het verwisselen; de fase
is $1$-periodiek). De reeks $\sum_k\abs{c_k(F)}$ convergeert (het
verval van $\hat f$), dus convergeert de fourierreeks van $F$
normaal; haar som is een \omterm{def:b3:topology:continuity}{continue} functie met dezelfde
fouriercoëfficiënten als $F$, en is dus gelijk aan $F$
(injectiviteit op de cirkel: het verschil heeft nulcoëfficiënten,
en \cref{thm:b3:hilbert:fourier} geeft nul in $L^2$, dus wegens
de \omterm{def:b3:topology:continuity}{continuïteit} overal). Evalueer in $x = 0$.
\end{proof}

\begin{example}[De identiteit van de thètafunctie]
\label{ex:b3:fouriertransform:theta}
Poisson toepassen op $f(x) = \eu^{-\pi tx^2}$ ($t > 0$), waarvan
de getransformeerde $\hat f(\xi) = t^{-1/2}\eu^{-\xi^2/4\pi t}$
is (\cref{ex:b3:fouriertransform:gaussian} met $a = \pi t$),
geeft
\[
\sum_{n\in\Z}\eu^{-\pi n^2t}
= \frac1{\sqrt t}\sum_{k\in\Z}\eu^{-\pi k^2/t} :
\]
de functionaalvergelijking van de thètafunctie van Jacobi,
sleutel tot de functionaalvergelijking van de $\zeta$ van Riemann
--- en een spectaculaire numerieke versneller: voor kleine $t$
convergeert het linkerlid traag en het rechterlid razendsnel.
\end{example}

\begin{method}\label{met:b3:fouriertransform:method}
Werkgebieden: $L^1$ --- de transformatie is puntsgewijs
gedefinieerd, en de inversie vereist $\hat f \in L^1$; $\mathcal
S$ --- alles is geoorloofd, bewijs hier eerst; $L^2$ --- de
transformatie is per \omterm{ex:b3:lebesgue:gamma}{dichtheid} gedefinieerd (niet met de
integraal!), volmaakt symmetrisch, met de boekhouding van
\omterm{thm:b3:hilbert:parseval}{Parseval}. Om een getransformeerde te berekenen: herleid met de
regels van \cref{prop:b3:fouriertransform:basic} tot de tabel
(indicator, exponentiële functie, gaussische functie); om een
identiteit te bewijzen: vestig haar op $\mathcal S$ (of $\mathcal
C_c^\infty$) en breid uit met \omterm{ex:b3:lebesgue:gamma}{dichtheid} en \omterm{def:b3:topology:continuity}{continuïteit}
(\cref{met:b3:lp:toolkit}); om een lineaire partiële of gewone
differentiaalvergelijking met constante coëfficiënten op te
lossen: transformeer, deel, inverteer.
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b3:fouriertransform:1}
Bereken de fouriergetransformeerden van: $\mathbf
1_{\intcc{-a}a}$; $\eu^{-a\abs x}$ ($a > 0$); de tentfunctie
$\max(0, 1 - \abs x)$; $\frac1{x^2 + a^2}$ \emph{(gebruik de
inversie op de tweede)}. Noteer de tabel die zo ontstaat.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:fouriertransform:2}
Zij $f \in L^1$. Druk in $\hat f$ de getransformeerden uit van:
$f(x - a)$, $f(x)\cos(bx)$, $f(ax + b)$, $\overline{f(-x)}$, $(f
* f)(x)$. Ga elke regel na op de gaussische functie.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:3}
(a) Toon aan dat $\mathbf 1_{\intcc{-1}1} * \mathbf
1_{\intcc{-1}1}$ als getransformeerde $\bigl(\frac{2\sin\xi}\xi
\bigr)^2$ heeft, en leid met Plancherel ---
of met inversie in $0$ --- af dat
$\int_\R\bigl(\frac{\sin\xi}\xi\bigr)^2\dd\xi = \pi$. Vergelijk
\cref{pb:b3:lebesgue:1}.
(b) Bereken $\int_\R\frac{\dd x}{(x^2+1)^2}$ met Plancherel
toegepast op $\eu^{-\abs x}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:4}
(Algebra van de \omterm{pb:b3:fouriertransform:1}{warmtekern}) Met $g_t(x) = \frac1{2\sqrt{\pi
t}}\eu^{-x^2/4t}$:
(a) ga na dat $\hat g_t(\xi) = \eu^{-t\xi^2}$;
(b) leid de halfgroepswet $g_t * g_s = g_{t+s}$ af zonder enige
integraalberekening;
(c) toon aan dat $\norm{g_t}_1 = 1$ en $\norm{g_t}_2^2 = (8\pi
t)^{-1/2}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:5}
Toon aan dat $\hat f$ even en reëel is als $f \in L^1$ even en
reëel is, en oneven en zuiver imaginair als $f$ oneven en reëel
is. Wat berekent $\hat f(0)$? Leid af dat $f \geq 0$ afdwingt dat
$\norm{\hat f}_\infty = \hat f(0) = \int f$, en interpreteer dat
voor kansdichtheden (\cref{ch:b3:clt}: een karakteristieke
functie heeft modulus $\leq 1$, bereikt in $0$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:fouriertransform:6}
(Niet-surjectiviteit) Toon in drie stappen aan dat
$\widehat{\phantom f}\colon L^1 \to \mathcal C_0$ injectief en
\omterm{def:b3:topology:continuity}{continu} is, maar \emph{niet} surjectief.
(i) Injectiviteit (\cref{thm:b3:fouriertransform:inversion}) en
\omterm{def:b3:topology:continuity}{continuïteit} ($\norm{\hat f}_\infty \leq \norm f_1$), en
$\mathcal C_0$ is een banachruimte (gesloten in
$\norm\cdot_\infty$).
(ii) Was de afbeelding surjectief, dan was ze bijectief, en zou
de open-afbeeldingsstelling
(\cref{thm:b3:banach:openmapping}) een constante $C$ geven met
$\norm f_1 \leq C\norm{\hat f}_\infty$ voor alle $f \in L^1$.
(iii) Weerspreek dat met $f_n(x) = \frac{\sin
x}{x}\cdot\frac{\sin(x/n)}{x/n}$: haar getransformeerde is (op
constanten na) een trapezium van het type $\mathbf
1_{\intcc{-1}1} * \mathbf 1_{\intcc{-1/n}{1/n}}$ --- toon aan dat
$\norm{\hat f_n}_\infty \leq \pi$ uniform, terwijl $\norm{f_n}_1
\geq c\ln n$ door de bogen van $\frac{\abs{\sin x}}x$ op $[1, n]$
te tellen (waar de tweede factor van onderen begrensd is), als in
\cref{thm:b3:banach:fourierdiverge}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:7}
(Woordenboek gladheid $\leftrightarrow$ verval) Bewijs: is $f \in
L^1$ met $\hat f(\xi) = O(\abs\xi^{-k-1-\delta})$ voor een
$\delta > 0$, dan heeft $f$ een $\mathcal
C^k$-vertegenwoordiger. Omgekeerd impliceert $f \in \mathcal
C^k_c$ dat $\hat f(\xi) = O(\abs\xi^{-k})$. Illustreer beide
richtingen op de tentfunctie.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:fouriertransform:8}
(Ongelijkheid van Heisenberg) Bewijs voor reële $f \in \mathcal
S(\R)$ met $\norm f_2 = 1$ dat
\[
\Bigl(\int x^2f(x)^2\dd x\Bigr)\cdot
\Bigl(\frac1{2\pi}\int \xi^2\abs{\hat f(\xi)}^2\dd\xi\Bigr)
\;\geq\; \frac14 ,
\]
met gelijkheid voor gaussische functies. \emph{(Schrijf $1 = \int
f^2 = -\int x\,(f^2)'$ met partiële integratie, schat af met
Cauchy--Schwarz en zet $\norm{f'}_2$ om met Plancherel.)}
Interpretatie: een signaal en zijn spectrum kunnen niet allebei
geconcentreerd zijn.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:9}
Rechtvaardig \cref{ex:b3:fouriertransform:theta} in detail (de
hypothesen van Poisson voor de gaussische functie), en gebruik de
identiteit om $\sum_{n\in\Z}\eu^{-\pi n^2}$ met drie termen tot
op zes decimalen te bepalen. Hoeveel termen van de
\emph{definiërende} reeks zou dezelfde nauwkeurigheid vergen bij
$t = 10^{-2}$, tegenover de getransformeerde reeks?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:10}
(Bandbegrensde functies) Zij $f \in L^2(\R)$ met $\mathcal Ff$
met drager in $\intcc{-\pi}\pi$. Toon aan dat $f$ een
vertegenwoordiger heeft waarvan elke waarde uit monsters kan
worden teruggewonnen: bewijs de \emph{interpolatie van Shannon}
in de gehele getallen,
\[
f(x) = \sum_{n\in\Z} f(n)\,
\frac{\sin\bigl(\pi(x - n)\bigr)}{\pi(x - n)}
\quad\text{in } L^2,
\]
door $\mathcal Ff$ te ontwikkelen in de fourierbasis van
$L^2(\intcc{-\pi}\pi)$ (\cref{thm:b3:hilbert:fourier}) en term
voor term terug te transformeren.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:11}
(De transformatie als operator van orde vier) Stel op $\mathcal
S(\R)$ $\mathcal F f = \hat f$.
(a) Toon met de inversieformule aan dat $(\mathcal F^2f)(x) =
2\pi\,f(-x)$, en leid $\mathcal F^4 = (2\pi)^2\, \mathrm{id}$
af.
(b) Leid af dat elke eigenwaarde van $\mathcal F$ op $\mathcal S$
in $\{\pm\sqrt{2\pi}, \pm\iu\sqrt{2\pi}\}$ ligt, en geef een
eigenfunctie voor $+\sqrt{2\pi}$ \emph{(welke functie uit dit
hoofdstuk is evenredig met haar eigen getransformeerde?)}.
(c) Toon aan dat even functies voldoen aan $\mathcal F^2f = 2\pi
f$ en oneven aan $\mathcal F^2f = -2\pi f$; maak uit
$x\eu^{-x^2/2}$ een eigenfunctie voor de eigenwaarde
$-\iu\sqrt{2\pi}$ door haar getransformeerde te berekenen
(differentieer de getransformeerde van de gaussische functie).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:fouriertransform:12}
(Autocorrelatie en het lemma van Wiener) Definieer voor $f \in
L^2(\R)$ $\tilde f(x) = \overline{f(-x)}$ en de
\emph{autocorrelatie} $A_f = f * \tilde f$.
(a) Toon aan dat $A_f$ een begrensde \omterm{def:b3:topology:continuity}{continue} functie is met
$A_f(0) = \norm f_2^2 \geq \abs{A_f(x)}$ voor alle $x$
(\cref{exo:b3:lp:6} en Cauchy--Schwarz).
(b) Toon, eerst voor $f \in L^1\cap L^2$, aan dat
$\widehat{A_f} = \abs{\hat f\,}^2 \geq 0$: de autocorrelatie
heeft een niet-negatieve getransformeerde --- de spectra van
autocorrelaties zijn vermogensspectra.
(c) Leid de identiteit $\int_\R\abs{\hat f(\xi)}^2\eu^{\iu
x\xi}\,\dd\xi = 2\pi A_f(x)$ af (inversie; rechtvaardig haar
toepasbaarheid wanneer $\hat f \in L^2$ voldoet aan $\abs{\hat
f}^2 \in L^1$), en evalueer haar voor $f = \mathbf
1_{\intcc{-1/2}{1/2}}$ in $x = 0$: vind opnieuw
$\int_\R\bigl(\frac{\sin u}u\bigr)^2\dd u = \pi$.
\end{exercise}

\section{Probleem: de warmtevergelijking op de lijn}

\begin{problem}[{Weekendopgave --- $\partial_tu =
\partial^2_{xx}u$, van begin tot eind opgelost}]
\label{pb:b3:fouriertransform:1}
Warmte verspreidt zich; de vergelijking $\partial_tu =
\partial_{xx}^2u$ zegt dat haar \omterm{ex:b3:lebesgue:gamma}{dichtheid} diffundeert met een
snelheid die door de lokale kromming van het temperatuurprofiel
wordt gegeven. We lossen het beginwaardeprobleem op $\R$ op ---
gegeven $f$, zoek $u(t, x)$ voor $t > 0$ met $u(0, \cdot) = f$
--- bewijzen de opmerkelijke eigenschappen van de oplossing, en
zien waarom de tijd niet omkeerbaar is. Overal is $g_t(x) =
\frac1{2\sqrt{\pi t}}\eu^{-x^2/4t}$ de
\emph{warmtekern}\index{warmtekern} en $u(t, \cdot) = g_t * f$.

\medskip
\noindent\textbf{Deel I --- De kern afleiden.} Werk eerst
formeel: stel dat $u(t, \cdot) \in \mathcal S$ de vergelijking
oplost, en zij $\hat u(t, \xi)$ de getransformeerde in $x$.
\begin{enumerate}
  \item Toon (formeel) aan dat $\partial_t\hat u = -\xi^2\hat u$,
        dus $\hat u(t,\xi) = \eu^{-t\xi^2}\hat f(\xi)$, en herken
        $u(t) = g_t * f$ (\cref{exo:b3:fouriertransform:4}). Dat
        motiveert de \emph{definitie} van $u$; alles wordt nu
        rechtstreeks bewezen, voor $f \in \mathcal C_b(\R)$
        (begrensd \omterm{def:b3:topology:continuity}{continu}) of $f \in L^p$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Verificatie.}
\begin{enumerate}[resume]
  \item Toon aan dat voor $t > 0$ de uitdrukking $u(t, x) = \int
        g_t(x-y)f(y)\dd y$ goed gedefinieerd is voor $f \in
        \mathcal C_b$, en dat $u$ $\mathcal C^\infty$ is in $(t,
        x)$ op $\intoo0\infty\times\R$ \emph{(differentieer onder
        de integraal; domineer de gaussische afgeleiden lokaal
        uniform in $(t,x)$)}.
  \item Ga met een rechtstreekse berekening $\partial_tg_t =
        \partial^2_{xx}g_t$ na, en leid $\partial_tu =
        \partial^2_{xx}u$ af voor $t > 0$.
  \item (Beginvoorwaarde) Toon aan dat voor $f \in \mathcal C_b$
        $u(t, x) \to f(x)$ als $t \to 0^+$, uniform op \omterm{def:b3:topology:compact}{compacte}
        verzamelingen \emph{(benaderende eenheid: splits $\abs y
        \leq \delta$ en $\abs y > \delta$)}; toon voor $f \in
        L^p$ ($p < \infty$) aan dat $\norm{u(t) - f}_p \to 0$.
  \item (Ogenblikkelijk gladstrijken) Besluit: zelfs voor louter
        begrensde \omterm{def:b3:topology:continuity}{continue} $f$ is de oplossing $\mathcal C^\infty$
        voor elke $t > 0$ --- warmte wist ruwheid ogenblikkelijk
        uit. Bereken $u(t, \cdot)$ expliciet voor $f = \mathbf
        1_{\intoo0\infty}$ (een foutfunctie) en schets haar
        profiel voor drie waarden van $t$.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Structurele eigenschappen.}
\begin{enumerate}[resume]
  \item (Positiviteit en vergelijking) Is $f \geq 0$, dan is $u >
        0$ voor alle $t > 0$ (strikt, tenzij $f = 0$ b.o.); is
        $f_1 \leq f_2$, dan is $u_1 \leq u_2$. Een koude plek
        warmt ogenblikkelijk op: becommentarieer.
  \item (Behoud) Voor $f \in L^1$: $\int u(t, x)\dd x = \int f$
        voor alle $t$ \emph{(Tonelli)} --- de totale warmte
        blijft behouden.
  \item (Dissipatie) Toon voor $f \in L^1\cap L^2$ met Plancherel
        aan dat $t \mapsto \norm{u(t)}_2$ niet-stijgend is,
        strikt dalend tenzij $f = 0$, en bereken haar limiet als
        $t \to \infty$. Toon bovendien $\norm{u(t)}_\infty \leq
        \frac{\norm f_1}{2\sqrt{\pi t}} \to 0$ aan: warmte
        verspreidt zich en sterft.
  \item (Uniciteit, klasse $L^2$) Zij $u$ een oplossing met $u(t)
        \in L^2$ voor alle $t$, $u \in \mathcal
        C^1(\intoo0\infty, L^2)$ in de natuurlijke zin en $u(t)
        \to f$ in $L^2$ als $t\to0$; aannemend dat de
        transformatie haar omzet in $\partial_t\hat u =
        -\xi^2\hat u$ puntsgewijs b.o.\ in $\xi$ voor b.o.\ $t$
        \emph{(gerechtvaardigd door te testen tegen $\mathcal
        C_c^\infty$ in $\xi$ --- schets dat)}, toon aan dat $\hat
        u(t,\xi) = \eu^{-t\xi^2}\hat f(\xi)$, en dus de uniciteit
        in deze klasse.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De pijl van de tijd.}
\begin{enumerate}[resume]
  \item Toon aan dat het \emph{achterwaartse} probleem slecht
        gesteld is: opdat de oplossing op tijdstip $-s$ ($s > 0$)
        met gegeven $f$ op tijdstip $0$ zou bestaan --- dat wil
        zeggen opdat $f = g_s * h$ een oplossing $h \in L^2$ zou
        hebben --- is het nodig dat $\eu^{s\xi^2}\hat f(\xi) \in
        L^2$: een extreme vervalvoorwaarde op $\hat f$. Geef een
        expliciete gladde $f \in L^2$ waarvoor op geen enkel
        tijdsinterval een achterwaartse oplossing bestaat: neem
        de functie met $\hat f(\xi) = \eu^{-\abs\xi}$ ---
        identificeer $f$ (\cref{exo:b3:fouriertransform:1}) en
        toon aan dat $\eu^{s\xi^2}\eu^{-\abs\xi} \notin L^2$ voor
        elke $s > 0$.
  \item (Gladstrijken tegenover informatie) Leg in een korte
        alinea uit, met de vragen 5, 9 en 10, waarom de
        warmtehalfgroep $(f \mapsto g_t * f)_{t\geq0}$ injectief
        maar niet surjectief is op $L^2$, en waarom dat de
        onomkeerbaarheid van de diffusie uitdrukt.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- De bemonsteringsstelling van
Shannon.} Een functie $f \in L^2(\R)$ heet
\emph{bandbegrensd} tot $\Omega$ als $\hat f = 0$ b.o.\ buiten
$\intcc{-\Omega}\Omega$; noteer $PW_\Omega$ (de ruimte van
Paley--Wiener) voor deze functies.
\begin{enumerate}[resume]
  \item Toon aan dat elke $f \in PW_\Omega$ b.o.\ samenvalt met de
        $\mathcal C^\infty$-functie
        $\frac1{2\pi}\int_{-\Omega}^{\Omega}\hat f(\xi)\eu^{\iu
        x\xi}\,\dd\xi$ (rechtvaardig de gladheid en de
        identificatie), met alle afgeleiden begrensd:
        bandbegrenzing is een extreme vorm van regulariteit.
        Vanaf nu duidt $f$ deze vertegenwoordiger aan.
  \item Ontwikkel $\hat f \in L^2(\intcc{-\Omega}\Omega)$ in de
        fourierbasis van dat interval en herken de coëfficiënten
        als \emph{monsters} van $f$:
        \[
        \hat f(\xi) = \frac\pi\Omega\sum_{n\in\Z}
        f\Bigl(\frac{n\pi}\Omega\Bigr)\,
        \eu^{-\iu n\pi\xi/\Omega}
        \quad\text{in } L^2(\intcc{-\Omega}\Omega) .
        \]
  \item Leid de \emph{bemonsteringsstelling} af: voor $f \in
        PW_\Omega$ is
        \[
        f(x) = \sum_{n\in\Z}f\Bigl(\frac{n\pi}
        \Omega\Bigr)\,\operatorname{sinc}(\Omega x - n\pi),
        \qquad \operatorname{sinc}t = \frac{\sin t}t,
        \]
        met convergentie in $L^2(\R)$ en uniform op $\R$
        \emph{(steek de reeks van vraag 13 in de inversieformule
        en bereken de elementaire integraal)}: een bandbegrensd
        signaal ligt \omterm{def:b3:complete:complete}{volledig} vast door zijn waarden op een
        rooster met stap $\pi/\Omega$ --- de nyquistsnelheid.
  \item Toon aan dat de functies $x \mapsto
        \operatorname{sinc}(\Omega x - n\pi)$, $n \in \Z$, een
        orthogonale familie in $L^2(\R)$ vormen met constante
        norm $\sqrt{\pi/\Omega}$, en leid de energie-identiteit
        $\norm f_2^2 = \frac\pi\Omega\sum_n\abs{f(n\pi/\Omega)}^2$
        af.
  \item (Aliasing) Geef een $g \neq 0$ in $PW_{2\Omega}$ die in
        elk monsterpunt $\frac{n\pi}\Omega$ verdwijnt
        \emph{(beschouw $g(x) = \sin(\Omega
        x)\operatorname{sinc}(\Omega x)$ en ga haar band na)}:
        bemonsteren onder de nyquistsnelheid verliest informatie
        --- twee verschillende signalen kunnen al hun monsters
        delen: het stroboscopische wagenwieleffect,
        gewiskundigd.
  \item (Vrijheidsgraden) Rechtvaardig met de vragen 14 en 15 de
        vuistregel van de ingenieur: een tot $\Omega$
        bandbegrensd signaal waarvan de energie in wezen door een
        tijdvenster van lengte $T$ wordt gedragen, wordt door
        ongeveer $\frac{\Omega T}\pi$ reële monsters beschreven
        --- maak ``in wezen'' precies met de energie-identiteit
        en de staart $\sum_{\abs{n\pi/\Omega} > T/2}$.
  \item (Consistentiecontroles) Ga de bemonsteringsstelling met
        de hand na op twee leden van $PW_\Omega$: (a) $f =
        \operatorname{sinc}(\Omega\,\cdot)$, waarvan de monsters
        $\delta_{n0}$ zijn; (b) smalbandige signalen van het type
        $f(x) = \cos(\omega x) \operatorname{sinc}(\varepsilon
        x)$ --- toon preciezer aan dat voor $f \in PW_{\Omega'}$
        met $\Omega' < \Omega$ ook de reeks met snelheid $\Omega$
        $f$ reconstrueert (overbemonstering is onschadelijk),
        door $PW_ {\Omega'} \subseteq PW_\Omega$ in te bedden.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Onzekerheid, nog tweemaal.} De
ongelijkheid van Heisenberg (\cref{exo:b3:fouriertransform:8})
begrenst \emph{hoe} geconcentreerd $f$ en $\hat f$ samen kunnen
zijn; hier volgen haar alles-of-niets-zusje en haar exacte
verzadiging.
\begin{enumerate}[resume]
  \item Zij $f \in L^1$ met $\operatorname{supp}f \subseteq
        \intcc{-A}A$. Toon aan dat $\hat f$ de som is van een
        overal convergente machtreeks:
        \[
        \hat f(\xi) = \sum_{k\geq0}\frac{(-\iu\xi)^k}{k!}
        \,m_k, \qquad m_k = \int_{-A}^{A}x^kf(x)\,\dd x,
        \quad \abs{m_k} \leq A^k\norm f_1
        \]
        \emph{(ontwikkel $\eu^{-\iu\xi x}$ en rechtvaardig het
        verwisselen met de normale convergentie)}: de
        transformatie is \emph{reëel-analytisch}, met in elk punt
        een oneindige convergentiestraal.
  \item Leid de dichotomie van de dragers af: een
        reëel-analytische functie die op een niet-leeg open
        interval verdwijnt, verdwijnt identiek \emph{(de
        verzameling waar alle afgeleiden verdwijnen is open en
        gesloten --- werk het taylorargument uit)}; besluit dat
        geen enkele $f \neq 0$ zowel $f$ als $\hat f$ met
        \omterm{def:b3:topology:compact}{compacte} drager heeft, en dat $PW_\Omega$ geen enkele
        functie $\neq 0$ met \omterm{def:b3:topology:compact}{compacte} drager bevat ---
        bandbegrensde signalen duren eeuwig, en tijdbegrensde
        signalen lekken in alle frequenties.
  \item (Verzadiging van Heisenberg) Bereken op de gaussische
        familie $f = \eu^{-ax^2}$ beide concentratiefactoren en
        ga na dat het genormeerde product $\bigl(\int x^2\abs
        f^2\bigr)\bigl(\frac1{2\pi}\int \xi^2\abs{\hat
        f}^2\bigr)\big/\norm f_2^4$ voor \emph{elke} $a$ gelijk
        is aan $\frac14$ --- de gelijkheidsfamilie van
        \cref{exo:b3:fouriertransform:8} in levenden lijve; leg
        met een schalingsargument uit waarom het product langs de
        familie constant moet zijn.
  \item Geef de fysische lezing (dichtheden van positie en impuls
        van een kwantumtoestand; $\hbar$ in de normering geeft
        $\sigma_x\sigma_p \geq \frac\hbar2$), en leg de
        verbanden door het hoofdstuk heen: ogenblikkelijk
        gladstrijken (Deel II), onomkeerbaarheid (Deel IV),
        bemonstering (Deel V), Heisenberg en de dichotomie van de
        dragers zijn vijf uitdrukkingen van één wet --- het
        gedrag van $\hat f$ op oneindig schrijft voor wat $f$
        waar dan ook mag doen.
\end{enumerate}

\medskip
\noindent\textbf{Deel VII --- De algebra van de kern, en één
\omterm{def:b3:groups:derived}{oplosbaar} voorbeeld.}
\begin{enumerate}[resume]
  \item (Halfgroep) Bewijs de \emph{identiteit van
        Chapman--Kolmogorov} $g_t * g_s = g_{t+s}$ voor $t, s >
        0$ (via de convolutiestelling en de injectiviteit van de
        transformatie op $L^1$), en leid $u(t + s) = g_s * u(t)$
        af: evolueren gedurende $t + s$ is evolueren gedurende
        $t$, en daarna gedurende $s$. Scherp de dissipatie van
        vraag 8 aan: schrijf $\norm{u(t)}_2^2 = \frac1{2\pi}\int
        \eu^{-2t\xi^2}\abs{\hat f(\xi)}^2\dd\xi$ en toon met
        Cauchy--Schwarz aan dat
        \[
        t \longmapsto \ln\,\norm{u(t)}_2
        \quad\text{convex is op } \intoo0{+\infty} :
        \]
        de $L^2$-energie daalt niet alleen, ze daalt op een
        log-convexe manier.
  \item (Waar de warmte heen gaat) Zij $f \geq 0$, $f \in L^1$,
        met $\int x^2f(x)\dd x < \infty$. Toon aan dat voor alle
        $t > 0$
        \[
        \int_\R x\,u(t, x)\,\dd x = \int_\R x f(x)\,\dd x,
        \qquad
        \int_\R x^2u(t, x)\,\dd x
        = \int_\R x^2f(x)\,\dd x + 2t\int_\R f :
        \]
        het zwaartepunt van de warmte beweegt nooit, en de
        variantie groeit \emph{lineair} in de tijd --- de
        diffusieve schaling $x \sim \sqrt{2t}$, te herlezen
        wanneer de brownse beweging in
        \cref{ch:b3:probability} opduikt. \emph{(Bereken de
        eerste twee momenten van $g_t$ en gebruik Tonelli op de
        convolutie.)}
  \item (De gaussische functie, van begin tot eind opgelost)
        Vestig voor $f(x) = \eu^{-x^2}$ de gesloten vorm
        \[
        u(t, x) = \frac1{\sqrt{1 + 4t}}\,
        \exp\Bigl(-\frac{x^2}{1 + 4t}\Bigr),
        \]
        en ga er met de hand op na: de vergelijking
        $\partial_tu = \partial^2_{xx}u$; het behoud $\int u(t) =
        \sqrt\pi$; de dissipatiewet $\norm{u(t)}_2 =
        (\pi/2)^{1/4}(1 + 4t)^{-1/4}$ (vergelijk haar verval als
        $t^{-1/4}$ met het verval als $t^{-1/2}$ van de
        supremumnorm uit vraag 8); en de exacte
        variantiegroei van vraag 24. Bij $t = 6$: de piek is
        gezakt tot $\frac15$ van zijn beginhoogte, terwijl het
        profiel vijf keer breder is --- dezelfde warmte,
        uitgespreid.
\end{enumerate}
\end{problem}
