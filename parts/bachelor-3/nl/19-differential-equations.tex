\chapter{Gewone differentiaalvergelijkingen}\label{ch:b3:ode}

Het volume van bachelorjaar 2 loste lineaire
differentiaalvergelijkingen op en formuleerde de stelling van
Cauchy--Lipschitz; dit hoofdstuk bewijst haar --- en dat
tweemaal: bestaan en uniciteit met het vaste punt van Banach, de
globale structuur met de theorie van de \omterm{thm:b3:ode:maximal}{maximale oplossingen} en
de stelling over het \omterm{thm:b3:ode:maximal}{ontsnappen uit compacte verzamelingen}. De
lineaire theorie wordt daarna op eerlijke fundamenten herbouwd
(\omterm{thm:b3:ode:linearstructure}{resolvente}, \omterm{thm:b3:ode:linearstructure}{wronskiaan}, \omterm{thm:b3:ode:matrixexp}{matrixexponentiële}, Duhamel), en de
tweede helft van het hoofdstuk opent de \emph{kwalitatieve}
theorie --- \omterm{ex:b3:ode:planeclassification}{stromingen}, evenwichten, \omterm{thm:b3:ode:lyapunov}{lyapunovfuncties} en
\omterm{thm:b3:ode:linearization}{stabiliteit door linearisatie}: hoe je oplossingen begrijpt die je
nooit zult berekenen. De slinger is in de weekendopgave de
eeuwige gevalstudie. Overal is $U \subseteq \R\times\R^d$ open en
is $f \colon U \to \R^d$ \omterm{def:b3:topology:continuity}{continu}; een \emph{oplossing} van $x' =
f(t, x)$ is een $\mathcal C^1$-afbeelding $x \colon I \to \R^d$
($I$ een interval) met grafiek in $U$ die aan de vergelijking
voldoet.

\section{Cauchy--Lipschitz}

\begin{definition}\label{def:b3:ode:lipschitz}
$f$ heet \emph{lokaal lipschitz in $x$}\index{lokaal lipschitz}
als elk punt van $U$ een \omterm{def:b3:topology:topology}{omgeving} $V$ en een constante $L$ heeft
met $\norm{f(t, x_1) - f(t, x_2)} \leq L\norm{x_1 - x_2}$ voor
$(t, x_1), (t, x_2) \in V$. Is $f$ $\mathcal C^1$ (of bestaat
enkel $\partial_xf$ en is die \omterm{def:b3:topology:continuity}{continu}), dan is ze lokaal
lipschitz in $x$: op een \omterm{def:b3:topology:compact}{compacte} convexe \omterm{def:b3:topology:topology}{omgeving} met de
middelwaardeongelijkheid en $L = \sup\vertiii{\partial_xf}$.
\end{definition}

\begin{theorem}[Cauchy--Lipschitz, lokaal]
\label{thm:b3:ode:cauchylipschitz}
Zij $f$ \omterm{def:b3:topology:continuity}{continu} en \omterm{def:b3:ode:lipschitz}{lokaal lipschitz} in $x$, en $(t_0, x_0) \in
U$. Er is een $T > 0$ zodanig dat het beginwaardeprobleem
\[
x' = f(t, x), \qquad x(t_0) = x_0
\]
precies één oplossing heeft op $[t_0 - T, t_0 +
T]$.\index{stelling van Cauchy--Lipschitz}
\end{theorem}

\begin{proof}
Kies $a, b > 0$ met $Q = [t_0 - a, t_0 + a]\times\bar B(x_0, b)
\subseteq U$, waarop $\norm f \leq M$ en $f$ $L$-lipschitz is in
$x$. Een $\mathcal C^1$-functie is een oplossing dan en slechts
dan als ze voldoet aan de integraalvergelijking
\[
x(t) = x_0 + \int_{t_0}^{t}f\bigl(s, x(s)\bigr)\,\dd s
\]
(de hoofdstelling van de integraalrekening, in beide richtingen).
Zij $T = \min\bigl(a, \frac bM, \frac1{2L}\bigr)$, $I = [t_0 - T,
t_0 + T]$ en
\[
\mathcal E = \{x \in \mathcal C(I, \R^d) : \norm{x(t) - x_0}
\leq b\ \text{op } I\},
\]
een gesloten deelverzameling van de banachruimte $(\mathcal C(I,
\R^d), \norm\cdot_\infty)$: \omterm{def:b3:complete:complete}{volledig}
(\cref{def:b3:complete:complete}). Definieer $\Phi(x)(t) = x_0 +
\int_{t_0}^tf(s, x(s))\dd s$: voor $x \in \mathcal E$ is
$\norm{\Phi(x)(t) - x_0} \leq M\abs{t - t_0} \leq MT \leq b$ ---
$\Phi$ beeldt $\mathcal E$ in zichzelf af --- en voor $x, y \in
\mathcal E$:
\[
\norm{\Phi(x)(t) - \Phi(y)(t)} \leq \Bigl|\int_{t_0}^t
L\,\norm{x(s) - y(s)}\,\dd s\Bigr| \leq LT\,\norm{x -
y}_\infty \leq \tfrac12\norm{x - y}_\infty :
\]
een contractie. Het vaste punt van Banach
(\cref{thm:b3:complete:banach}) geeft een uniek vast punt in
$\mathcal E$: het bestaan, en de uniciteit onder de oplossingen
die in $\bar B(x_0, b)$ blijven --- maar elke oplossing op $I$
blijft daar ($\norm{x(t) - x_0} \leq M\abs{t - t_0} \leq b$
zolang de grafiek in $Q$ blijft, een continuïteitsargument): dus
uniciteit op $I$.
\end{proof}

\begin{lemma}[Grönwall]\label{lem:b3:ode:gronwall}
Zij $u \colon I \to \intco0\infty$ \omterm{def:b3:topology:continuity}{continu}, $t_0 \in I$, en stel
dat\index{lemma van Grönwall}
\[
u(t) \leq \alpha + \beta\,\Bigl|\int_{t_0}^{t}u(s)\,\dd
s\Bigr| \qquad (t \in I)
\]
met $\alpha \geq 0$ en $\beta > 0$. Dan is $u(t) \leq
\alpha\,\eu^{\beta\abs{t - t_0}}$ op $I$.
\end{lemma}

\begin{proof}
Voor $t \geq t_0$: zij $v(t) = \alpha + \beta\int_{t_0}^tu(s)\dd
s$, dan is $u \leq v$, $v' = \beta u \leq \beta v$ en $(v\eu^{-
\beta(t - t_0)})' \leq 0$: dus $v(t) \leq v(t_0)\eu^{\beta(t-t_0)}
= \alpha\eu^{\beta(t - t_0)}$. Voor $t \leq t_0$ pas je hetzelfde
toe op $\tilde u(t) = u(2t_0 - t)$.
\end{proof}

\begin{corollary}[Uniciteit en continue afhankelijkheid]
\label{cor:b3:ode:uniqueness}
Onder de hypothesen van \cref{thm:b3:ode:cauchylipschitz} vallen
twee oplossingen van $x' = f(t,x)$ die in één punt samenvallen,
samen op hun gemeenschappelijke definitie-interval. Kwantitatief:
zijn $x, y$ twee oplossingen met grafieken in een gebied waar $f$
$L$-lipschitz is in $x$, dan is
\[
\norm{x(t) - y(t)} \leq \norm{x(t_0) -
y(t_0)}\,\eu^{L\abs{t - t_0}} .
\]
\end{corollary}

\begin{proof}
De afschatting: $u = \norm{x - y}$ voldoet aan $u(t) \leq u(t_0)
+ L\abs{\int_{t_0}^tu}$ (trek de integraalvergelijkingen af);
Grönwall. Globale uniciteit: de verzameling waar ze samenvallen,
$\{t : x(t) = y(t)\}$, is gesloten in het gemeenschappelijke
interval, niet leeg en open --- overdek rond een punt van
samenvallen een \omterm{def:b3:topology:compact}{compact} stuk van de gemeenschappelijke grafiek
met eindig veel lipschitzblokken en pas op elk de afschatting toe
met $u(t_1) = 0$: lokaal is $x \equiv y$. Een niet-lege open en
gesloten deelverzameling van een interval is alles.
\end{proof}

\section{Maximale oplossingen}

\begin{theorem}[Maximale oplossingen; ontsnappen uit compacte
verzamelingen]
\label{thm:b3:ode:maximal}
Neem $f$ \omterm{def:b3:topology:continuity}{continu} en \omterm{def:b3:ode:lipschitz}{lokaal lipschitz} in $x$ aan.
\begin{enumerate}
  \item Elk beginwaardeprobleem heeft een unieke
        \emph{maximale} oplossing $x \colon \intoo{T_-}{T_+} \to
        \R^d$: elke andere oplossing door $(t_0, x_0)$ is haar
        beperking. Het interval is open.
  \item (\emph{Ontsnapping}) Voor elke \omterm{def:b3:topology:compact}{compacte} $K \subseteq U$
        is er een $\varepsilon > 0$ zodat $(t, x(t)) \notin K$
        voor alle $t \in \intoo{T_+ - \varepsilon}{T_+}$ (en
        symmetrisch bij $T_-$): \emph{de grafiek van een \omterm{thm:b3:ode:maximal}{maximale
        oplossing} verlaat uiteindelijk elke \omterm{def:b3:topology:compact}{compacte}
        deelverzameling van $U$}. In het bijzonder gaat voor $U =
        \R\times\R^d$ en $T_+ < +\infty$ $\norm{x(t)} \to
        +\infty$ als $t \to T_+^-$
        (\emph{explosie}).\index{maximale oplossing}\index{
        ontsnappen uit compacte verzamelingen}\index{explosie}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Zij $\mathcal S$ de verzameling van alle oplossingen door
$(t_0, x_0)$; volgens \cref{cor:b3:ode:uniqueness} vallen elke
twee samen op de doorsnede van hun intervallen, dus lijmen ze:
definieer op $J = \bigcup_{y \in \mathcal S}I_y$ $x(t) = y(t)$
voor een willekeurige $y$ die in $t$ gedefinieerd is: een goed
gedefinieerde oplossing, kennelijk maximaal en uniek. $J$ is
open: een oplossing die in een eindpunt gedefinieerd is, zou daar
volgens \cref{thm:b3:ode:cauchylipschitz} verlengd kunnen
worden.

(2) Stel dat de bewering faalt bij $T_+$: er zijn $t_n \to T_+$
met $(t_n, x(t_n)) \in K$; merk op dat dat $T_+ < \infty$
afdwingt, of dat er, als $T_+ = \infty$, niets te bewijzen valt
($K$ is begrensd in de tijd). Zij dus $T_+ < \infty$.
\omterm{def:b3:topology:compact}{Compactheid}: uniforme constanten $M, L, a, b$ werken voor alle
begingegevens in een \omterm{def:b3:topology:topology}{omgeving} van $K$ --- concreet: overdek $K$
met eindig veel blokken $Q_i$ als in het bewijs van de lokale
stelling en zij $T^* > 0$ het minimum van de bijbehorende
bestaanstijden: elk begingegeven in $K$ lanceert een oplossing
die minstens $T^*$ voorbij haar begintijd leeft. Dat toepassen in
$(t_n, x(t_n))$ met $t_n > T_+ - T^*/2$ breidt $x$ voorbij $T_+$
uit (de uitbreiding valt wegens de uniciteit met $x$ samen, en
verlengt haar dan): in tegenspraak met de maximaliteit. Dus
verlaat de grafiek $K$ definitief vóór $T_+$. Voor $U =
\R\times\R^d$: gaat $\norm{x(t)}\not\to\infty$, dan houdt een rij
$t_n \to T_+$ $(t_n, x(t_n))$ in de \omterm{def:b3:topology:compact}{compacte} $[t_0, T_+]\times\bar
B(0, R)$: uitgesloten.
\end{proof}

\begin{corollary}[Globaal bestaan bij lineaire groei]
\label{cor:b3:ode:globallinear}
Is $U = I\times\R^d$ ($I$ een open interval) en $\norm{f(t, x)}
\leq \alpha(t)\norm x + \beta(t)$ met $\alpha, \beta$ \omterm{def:b3:topology:continuity}{continu},
dan is elke \omterm{thm:b3:ode:maximal}{maximale oplossing} op heel $I$ gedefinieerd.
\end{corollary}

\begin{proof}
Op een \omterm{def:b3:topology:compact}{compacte} $[t_0, T] \subseteq I$ is $\norm{x(t)} \leq
\norm{x_0} + \int_{t_0}^t(\alpha\norm x + \beta)$, dus geeft
Grönwall (met $\alpha, \beta$ daar begrensd door $A, B$)
$\norm{x(t)} \leq (\norm{x_0} + B(T - t_0))\eu^{A(T - t_0)}$:
begrensd. Is $T_+ < \sup I$, dan blijft de grafiek nabij $T_+$ in
een \omterm{def:b3:topology:compact}{compacte} deelverzameling van $I\times\R^d$: in tegenspraak
met de ontsnapping (\cref{thm:b3:ode:maximal}).
\end{proof}

\section{Lineaire stelsels}

Overal in deze paragraaf zijn $A \colon I \to M_d(\R)$ en $b
\colon I \to \R^d$ \omterm{def:b3:topology:continuity}{continu}; het stelsel is $x' = A(t)x + b(t)$
--- lineaire groei: alle \omterm{thm:b3:ode:maximal}{maximale oplossingen} leven op heel $I$
(\cref{cor:b3:ode:globallinear}).

\begin{theorem}[Structuur]\label{thm:b3:ode:linearstructure}
De oplossingen van het homogene stelsel $x' = A(t)x$ vormen een
$d$-dimensionale vectorruimte $S_H$; voor elke $t_0$ is de
evaluatie $x \mapsto x(t_0)$ een isomorfisme $S_H \to \R^d$. De
\emph{resolvente}\index{resolvente van een lineair stelsel} $R(t,
s) \in GL_d(\R)$, gedefinieerd doordat $t \mapsto R(t, s)v$ de
oplossing is met waarde $v$ in $s$, voldoet aan
\[
R(s,s) = I,\quad R(t, u)R(u, s) = R(t, s),\quad
\partial_tR(t,s) = A(t)R(t,s),
\]
en het inhomogene probleem wordt opgelost door de \emph{formule
van Duhamel}\index{formule van Duhamel}:
\[
x(t) = R(t, t_0)\,x_0 + \int_{t_0}^{t}R(t, s)\,b(s)\,\dd s .
\]
Ten slotte gehoorzaamt de \emph{wronskiaan}\index{wronskiaan}
$w(t) = \det R(t, s)$ aan de formule van Liouville $w' =
\operatorname{tr}A(t)\,w$, dus $w(t) =
\exp\bigl(\int_s^t\operatorname{tr}A\bigr) > 0$.
\end{theorem}

\begin{proof}
De lineariteit van de vergelijking maakt de oplossingen een
vectorruimte; de evaluatie is lineair, injectief (uniciteit: een
oplossing die in $t_0$ verdwijnt, is $\equiv 0$) en surjectief
(bestaan): dus dimensie $d$. De eigenschappen van de \omterm{thm:b3:ode:linearstructure}{resolvente}
herformuleren de uniciteit (beide leden van elke identiteit
lossen hetzelfde beginwaardeprobleem op); de inverteerbaarheid
komt uit $R(s,t)R(t,s) = I$. Duhamel: differentieer de formule
--- $x'(t) = A(t)R(t,t_0)x_0 + R(t,t)b(t) +
\int_{t_0}^tA(t)R(t,s)b(s)\dd s = A(t)x(t) + b(t)$ (het
differentiëren onder de integraal is geoorloofd: de integrand is
$\mathcal C^1$ in $t$ met een \omterm{def:b3:topology:continuity}{continue} afgeleide in $(t,s)$; of
ga na met de integraalvergelijking). Liouville: $w(t + h) =
\det\bigl(R(t+h, t)\bigr)w(t)$ en $R(t + h, t) = I + hA(t) +
o(h)$ (uit de integraalvergelijking), dus $\det = 1 +
h\operatorname{tr}A(t) + o(h)$ (ontwikkeling van $\det$ in $I$):
$w'(t) = \operatorname{tr}A(t)\,w(t)$; integreer de scalaire
lineaire vergelijking.
\end{proof}

\begin{theorem}[Matrixexponentiële]\label{thm:b3:ode:matrixexp}
Voor $A \in M_d(\C)$ convergeert de reeks $\eu^{A} =
\sum_{n\geq0}\frac{A^n}{n!}$ (absoluut, in elke
submultiplicatieve norm), is $\eu^{A+B} = \eu^A\eu^B$ zodra $AB =
BA$, en is $t \mapsto \eu^{tA}$ de \omterm{thm:b3:ode:linearstructure}{resolvente} van het constante
stelsel: $R(t,s) = \eu^{(t-s)A}$; ze is $\mathcal C^\infty$ met
$\frac{\dd}{\dd t}\eu^{tA} = A\eu^{tA}$. Bovendien: is
$\operatorname{Re}\lambda < -\alpha < 0$ voor elke eigenwaarde
$\lambda$ van $A$, dan is $\vertiii{\eu^{tA}} \leq
C\eu^{-\alpha t}$ voor $t \geq 0$.\index{matrixexponentiële}
\end{theorem}

\begin{proof}
Convergentie: $\vertiii{A^n/n!} \leq \vertiii A^n/n!$,
sommeerbaar (\cref{exo:b3:complete:1}(b) in de banachalgebra
$M_d$). Voor commuterende $A, B$: het cauchyproduct van de twee
absoluut convergente reeksen herschikt zich, via het
binomium (geldig zodra $AB = BA$), tot
$\sum_n\frac{(A+B)^n}{n!}$. Differentieerbaarheid, rechtstreeks:
$\frac{\eu^{(t+h)A} - \eu^{tA}}h = \eu^{tA}\frac{\eu^{hA} - I}{h}
\to \eu^{tA}A$, want $\norm{\frac{\eu^{hA} - I}h - A} \leq
\sum_{n\geq2}\frac{\abs h^{n-1}\vertiii A^n}{n!} = O(h)$.
Bijgevolg lost $t \mapsto \eu^{(t - s)A}v$ het
beginwaardeprobleem op dat $R(t,s)v$ definieert. Spectrale
afschatting: volgens de \omterm{thm:b3:modules:jordan}{jordanvorm}
(\cref{thm:b3:modules:jordan}) is $A = P(D + N)P^{-1}$ met $D$
diagonaal met de eigenwaarden, $N$ nilpotent en $DN = ND$. Dan is
$\eu^{tA} = P\,\eu^{tD}\eu^{tN}P^{-1}$ met $\vertiii{\eu^{tD}}
\leq \eu^{-(\alpha + \delta)t}$ voor een $\delta > 0$ ($t \geq
0$) en $\eu^{tN}$ polynomiaal in $t$ (nilpotentie): het product
is $\leq C\eu^{-\alpha t}$ (een veelterm wordt door
$\eu^{-\delta t}$ verslagen).
\end{proof}

\begin{example}[Het vlak, geklasseerd]
\label{ex:b3:ode:planeclassification}
Voor $x' = Ax$ met $A \in M_2(\R)$ inverteerbaar wordt het
faseportret nabij $0$ beslist door $\tau = \operatorname{tr}A$ en
$\delta = \det A$, via de eigenwaarden $\lambda_\pm = \frac{\tau
\pm \sqrt{\tau^2 - 4\delta}}2$:
\begin{itemize}
  \item $\delta < 0$: reële eigenwaarden met tegengesteld teken
        --- een \emph{zadel}; twee \omterm{def:b3:groups:action}{banen} komen binnen, twee gaan
        weg, alle andere vliegen voorbij. Altijd instabiel.
  \item $\delta > 0$, $\tau^2 \geq 4\delta$: reële eigenwaarden
        met hetzelfde teken ($= \operatorname{sign}\tau$) --- een
        \emph{knoop}, \omterm{def:b3:ode:stability}{stabiel} dan en slechts dan als $\tau < 0$;
        de \omterm{def:b3:groups:action}{banen} raken de trage eigenrichting.
  \item $\delta > 0$, $\tau^2 < 4\delta$, $\tau \neq 0$:
        toegevoegd complexe eigenwaarden $\frac\tau2 \pm
        \iu\omega$ --- een \emph{spiraal} (focus), \omterm{def:b3:ode:stability}{stabiel} dan en
        slechts dan als $\tau < 0$; de oplossingen zijn
        $\eu^{\tau t/2}$ maal rotaties met periode
        $\frac{2\pi}\omega$.
  \item $\tau = 0$, $\delta > 0$: zuiver imaginaire eigenwaarden
        --- een \emph{\omterm{ex:b3:groups:actions}{centrum}}: gesloten \omterm{def:b3:groups:action}{banen} (ellipsen),
        stabiliteit zonder asymptotische stabiliteit, precies het
        grensgeval dat \cref{thm:b3:ode:linearization} voor
        niet-lineaire stelsels niet kan beslissen (het onderste
        evenwicht van de slinger, \cref{pb:b3:ode:1}, zit hier).
\end{itemize}
De grensparabool $\tau^2 = 4\delta$ draagt de ontaarde knopen
(jordanblokken: \omterm{def:b3:groups:action}{banen} met één enkele raakrichting). Alles wordt
van twee getallen afgelezen --- daarom is de eerste reflex vóór
elk vlak faseportret $\operatorname{tr}$ en $\det$ te berekenen;
zo geeft $A = \bigl(\begin{smallmatrix}0 & 1\\ -1 &
-c\end{smallmatrix}\bigr)$ (de gedempte oscillator): $\delta = 1
> 0$, $\tau = -c$: een stabiele spiraal voor $0 < c < 2$, een
stabiele knoop voor $c \geq 2$ --- onderdemping tegenover
overdemping, in één oogopslag.
\end{example}

\section{Stromingen, evenwichten, stabiliteit}

Beschouw nu de \emph{autonome} vergelijking $x' = F(x)$, met $F
\colon \Omega \to \R^d$ \omterm{def:b3:ode:lipschitz}{lokaal lipschitz} op de open $\Omega
\subseteq \R^d$. Schrijf $\varphi_t(x_0) = x(t)$ voor de \omterm{thm:b3:ode:maximal}{maximale
oplossing} met $x(0) = x_0$ (de \emph{stroming}\index{stroming van
een vectorveld}); de autonomie geeft de groepseigenschap
$\varphi_{t+s} = \varphi_t\circ\varphi_s$ waar ze gedefinieerd is
(beide leden lossen op tijdstip $s$ hetzelfde probleem op).

\begin{definition}\label{def:b3:ode:stability}
Een \emph{evenwicht} is een punt $\bar x$ met $F(\bar x) = 0$
(zodat $\varphi_t(\bar x) = \bar x$). Het heet
\emph{stabiel}\index{stabiliteit (Lyapunov)} als er voor elke
$\varepsilon > 0$ een $\delta > 0$ is zodat $\norm{x_0 - \bar x}
< \delta$ impliceert dat de oplossing voor alle $t \geq 0$
bestaat met $\norm{\varphi_t(x_0) - \bar x} < \varepsilon$; en
\emph{asymptotisch stabiel} als bovendien $\varphi_t(x_0) \to
\bar x$ voor alle $x_0$ nabij $\bar x$.
\end{definition}

\begin{theorem}[Lyapunovfuncties]\label{thm:b3:ode:lyapunov}
Zij $\bar x$ een evenwicht en $V \colon \mathcal V \to \R$ een
$\mathcal C^1$-functie op een \omterm{def:b3:topology:topology}{omgeving} van $\bar x$
met:\index{lyapunovfunctie}
\[
V(\bar x) = 0,\qquad V(x) > 0 \text{ voor } x \neq \bar x,
\qquad \dot V(x) = \nabla V(x)\cdot F(x) \leq 0 .
\]
Dan is $\bar x$ \omterm{def:b3:ode:stability}{stabiel}. Is bovendien $\dot V < 0$ buiten $\bar
x$, dan is $\bar x$ asymptotisch \omterm{def:b3:ode:stability}{stabiel}.
\end{theorem}

\begin{proof}
Langs een oplossing is $\frac{\dd}{\dd t}V(x(t)) = \dot V(x(t))
\leq 0$: $V$ daalt. Zij bij gegeven $\varepsilon$ (klein genoeg
dat $\bar B(\bar x, \varepsilon) \subseteq \mathcal V$) $m =
\min\{V(x) : \norm{x - \bar x} = \varepsilon\} > 0$ (\omterm{def:b3:topology:compact}{compactheid},
positiviteit), en kies $\delta < \varepsilon$ met $V < m$ op
$B(\bar x, \delta)$ (\omterm{def:b3:topology:continuity}{continuïteit}). Een oplossing die in $B(\bar
x, \delta)$ start, heeft $V(x(t)) < m$ voor alle latere
tijdstippen, en kan dus nooit de sfeer $\norm{x - \bar x} =
\varepsilon$ bereiken (waar $V \geq m$): ze blijft in de bal ---
en bestaat dan voor alle $t \geq 0$: de oplossing blijft in de
\omterm{def:b3:topology:compact}{compacte} $\bar B$, dus dwingt \cref{thm:b3:ode:maximal}(2) (het
\omterm{thm:b3:ode:maximal}{ontsnappen uit compacte verzamelingen}) $T_+ = +\infty$ af.
Stabiliteit.

Het asymptotische geval: zij $x(t)$ startend in $B(\bar x,
\delta)$; $V(x(t))$ daalt naar een $c \geq 0$. Is $c > 0$, dan
blijft de \omterm{def:b3:groups:action}{baan} in $K = \{x \in \bar B(\bar x, \varepsilon): V(x)
\geq c\}$, een \omterm{def:b3:topology:compact}{compacte} verzameling die een \omterm{def:b3:topology:topology}{omgeving} van $\bar x$
uitsluit ($V$ is \omterm{def:b3:topology:continuity}{continu} met $V(\bar x) = 0 < c$). Op $K$ is de
functie $\dot V$ \omterm{def:b3:topology:continuity}{continu} en strikt negatief, dus is $\mu =
\max_K\dot V < 0$ (\omterm{def:b3:topology:compact}{compactheid}); dan is $V(x(t)) \leq V(x(0)) +
\mu t \to -\infty$: absurd, want $V \geq 0$. Dus $c = 0$, en
$x(t) \to \bar x$ (punten op afstand $\geq \rho$ van $\bar x$
binnen de bal hebben $V \geq m_\rho > 0$).
\end{proof}

\begin{theorem}[Stabiliteit door linearisatie]
\label{thm:b3:ode:linearization}
Zij $F$ $\mathcal C^1$, $F(\bar x) = 0$ en $A = DF(\bar x)$.
Heeft elke eigenwaarde van $A$ $\operatorname{Re}\lambda < 0$,
dan is $\bar x$ asymptotisch \omterm{def:b3:ode:stability}{stabiel}.\index{stabiliteit door
linearisatie}
\end{theorem}

\begin{proof}
Verschuif $\bar x$ naar $0$ en schrijf $F(x) = Ax + g(x)$ met
$g(x) = o(\norm x)$ ($\mathcal C^1$-differentieerbaarheid). Kies
$\alpha > 0$ met $\vertiii{\eu^{tA}} \leq C\eu^{-\alpha t}$ ($t
\geq 0$; \cref{thm:b3:ode:matrixexp}) en $r > 0$ met
$\norm{g(x)} \leq \frac{\alpha}{2C}\norm x$ voor $\norm x \leq
r$. Duhamel met $b(s) = g(x(s))$:
\[
x(t) = \eu^{tA}x_0 + \int_0^t\eu^{(t-s)A}g(x(s))\,\dd s,
\]
geldig zolang $\norm{x(s)} \leq r$. Dan voldoet $u(t) =
\eu^{\alpha t}\norm{x(t)}$ aan
\[
u(t) \leq C\norm{x_0} +
\int_0^{t}C\,\frac{\alpha}{2C}\,u(s)\,\dd s ,
\]
dus geeft Grönwall $u(t) \leq C\norm{x_0}\eu^{\alpha t/2}$, dat
wil zeggen $\norm{x(t)} \leq C\norm{x_0}\eu^{-\alpha t/2}$. Is
$\norm{x_0} < r/C$, dan houdt de a-priorigrens $\norm{x(t)} < r$
voor alle $t$ (een argument met \omterm{def:b3:topology:continuity}{continuïteit} en hefboomwerking:
de verzameling tijdstippen waar $\norm x \leq r$ is open en
gesloten in $\intco0{T_+}$ gegeven de strikte afschatting), is de
oplossing globaal, en convergeert ze exponentieel naar $0$:
asymptotische stabiliteit.
\end{proof}

\begin{method}\label{met:b3:ode:strategy}
Tegenover een differentiaalvergelijking: (1) bestaan en
uniciteit --- ga \omterm{def:b3:ode:lipschitz}{lokaal lipschitz} na (meestal $\mathcal C^1$);
(2) globaliteit --- lineaire groei, begrensdheid, of een
invariante \omterm{def:b3:topology:compact}{compacte} verzameling via een \omterm{thm:b3:ode:lyapunov}{lyapunovfunctie} of een
eerste integraal; lukt dat niet, verdenk dan explosie en test op
de scalaire karikatuur $x' = x^2$; (3) lineaire stelsels ---
\omterm{thm:b3:ode:linearstructure}{resolvente}, Duhamel, en voor constante coëfficiënten de
eigenstructuur van $A$; (4) kwalitatieve vragen --- evenwichten,
lineariseer, en jaag op een \omterm{thm:b3:ode:lyapunov}{lyapunovfunctie} (de energie, als het
stelsel mechanisch is) of op een eerste integraal waarvan de
niveauverzamelingen de \omterm{def:b3:groups:action}{banen} vangen. De weekendopgave loopt de
hele methode door met de slinger.
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b3:ode:1}
Los expliciet op en bepaal het maximale interval:
(a) $x' = x^2$, $x(0) = 1$; (b) $x' = 1 + x^2$, $x(0) = 0$;
(c) $x' = x(1-x)$, $x(0) = \frac12$. Verzoen elk antwoord met
\cref{thm:b3:ode:maximal}(2) en
\cref{cor:b3:ode:globallinear}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:ode:2}
Zij $x, y$ oplossingen van $x' = f(t,x)$ met $f$ globaal
$L$-lipschitz in $x$ op $\R\times\R^d$.
(a) Bewijs $\norm{x(t) - y(t)} \leq \norm{x(0) -
y(0)}\eu^{L\abs t}$, en toon met een voorbeeld (lineair!) aan dat
de factor $\eu^{L\abs t}$ bereikt wordt.
(b) Leid af dat de stromingsafbeelding $x_0 \mapsto x(t; x_0)$
\omterm{def:b3:topology:continuity}{continu} is, en zelfs lipschitz op begrensde verzamelingen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:3}
Toon aan dat elk van de volgende alleen \omterm{thm:b3:ode:maximal}{maximale oplossingen}
heeft die globaal zijn op $\R$, met vermelding van de juiste
stelling:
(a) $x' = \sin(tx)$;
(b) $x' = \frac{t\,x}{1 + x^2}$;
(c) $x'' + q(t)x = 0$ met $q$ \omterm{def:b3:topology:continuity}{continu} (zet om in een stelsel van
eerste orde);
(d) $x' = A(t)x$ met $A$ \omterm{def:b3:topology:continuity}{continu} en begrensd --- en geef de grens
van Grönwall op $\norm{x(t)}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:4}
(a) Bereken $\eu^{tA}$ voor
$A = \bigl(\begin{smallmatrix}0 & -1\\ 1 &
0\end{smallmatrix}\bigr)$,
$\bigl(\begin{smallmatrix}\lambda & 1\\ 0 &
\lambda\end{smallmatrix}\bigr)$ en
$\bigl(\begin{smallmatrix}0 & 1\\ 1 &
0\end{smallmatrix}\bigr)$.
(b) Los de aangedreven oscillator $x'' + x = \cos(\omega t)$ op
met Duhamel (in stelselvorm), voor $\omega \neq 1$ en $\omega =
1$: de resonantie duikt op als de seculiere term $t\sin t$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:5}
Voor de scalaire vergelijking $x'' + p(t)x' + q(t)x = 0$:
(a) Toon aan dat de \omterm{thm:b3:ode:linearstructure}{wronskiaan} $w = x_1x_2' - x_1'x_2$ van twee
oplossingen voldoet aan $w' = -p\,w$ (Abel), en leid af dat twee
oplossingen met ergens $w \neq 0$ een basis vormen.
(b) Zoek, gegeven één nergens verdwijnende oplossing $x_1$, de
algemene oplossing met \emph{ordeverlaging}: stel $x_2 =
x_1\int \frac{\eu^{-\int p}}{x_1^2}$ en ga na. Pas toe op $t^2x''
- 2x = 0$ op $\intoo0{+\infty}$ met $x_1(t) = t^2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:6}
(Logistisch) Voor $x' = x(1 - x)$: bepaal alle evenwichten en hun
stabiliteit (met \cref{thm:b3:ode:linearization} en
rechtstreeks); toon aan dat elke oplossing met $x(0) \in
\intoo01$ stijgend en globaal is, met limieten $0$ en $1$ in
$\mp\infty$; en los expliciet op ter bevestiging. Toon
algemener aan dat scalaire autonome oplossingen monotoon zijn, en
besluit: geen niet-constante periodieke oplossingen in dimensie
$1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:7}
(Eerste integralen) Zij $H \colon \Omega \to \R$ $\mathcal C^1$
en beschouw het vlakke hamiltoniaanse stelsel $x' = \partial_yH$,
$y' = -\partial_xH$.
(a) Toon aan dat $H$ constant is langs de oplossingen.
(b) Voor $H = \frac{y^2}2 + \frac{x^4}4$: toon aan dat alle
oplossingen globaal en begrensd zijn, en dat de oorsprong \omterm{def:b3:ode:stability}{stabiel}
is (Lyapunov: $H$) hoewel de linearisatie
($\bigl(\begin{smallmatrix}0&1\\0&0\end{smallmatrix}\bigr)$)
\emph{niet} asymptotisch \omterm{def:b3:ode:stability}{stabiel} is: linearisatie kan
onbeslissend zijn.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:8}
(Gedempte slinger) $x'' + cx' + \sin x = 0$ met $c > 0$; als
stelsel: $x' = y$, $y' = -\sin x - cy$.
(a) Toon aan dat $V(x, y) = \frac{y^2}2 + 1 - \cos x$ voldoet aan
$\dot V = -cy^2 \leq 0$: de oorsprong is \omterm{def:b3:ode:stability}{stabiel}.
(b) $\dot V$ verdwijnt op de hele as $y = 0$: het strikte
criterium van Lyapunov faalt. Bewijs de asymptotische stabiliteit
toch, met linearisatie (\cref{thm:b3:ode:linearization}): bereken
de eigenwaarden van de gelineariseerde matrix in $(0,0)$ en
controleer $\operatorname{Re} < 0$ voor elke $c > 0$.
(c) Wat gebeurt er in het evenwicht $(\pi, 0)$? Bereken de
linearisatie en besluit (één positieve eigenwaarde: instabiliteit
--- je mag de instabiliteitsuitspraak informeel gebruiken of een
expliciete ontsnappende oplossing van het lineaire stelsel
geven).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:9}
(De grens van de uniciteit) Toon voor $\alpha \in \intoo01$ aan
dat het probleem $x' = \abs x^{\alpha}$, $x(0) = 0$ oneindig veel
oplossingen heeft (pas \cref{pb:b3:complete:1}, Deel III, aan).
Toon daarentegen aan dat voor $\alpha = 1$ (dat wil zeggen $x' =
\abs x$) de oplossing door $0$ uniek is, en wijs precies aan
welke hypothese van \cref{thm:b3:ode:cauchylipschitz} de twee
gevallen onderscheidt.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:10}
(A-priorigrenzen vangen oplossingen) Zij $F \colon \R^d \to \R^d$
\omterm{def:b3:ode:lipschitz}{lokaal lipschitz} met $\langle F(x), x\rangle \leq 0$ zodra
$\norm x \geq R$.
(a) Toon aan dat de gesloten bal $\bar B(0, R)$ positief
invariant is: oplossingen die binnen starten, blijven binnen voor
$t \geq 0$. \emph{(Is $\norm{x(t_2)} > R$, beschouw dan het
laatste tijdstip $t_1 < t_2$ met $\norm{x(t_1)} = R$ en bestudeer
$\frac{\dd}{\dd t}\norm{x(t)}^2$ op $\intcc{t_1}{t_2}$.)}
(b) Leid het globale voorwaartse bestaan af voor gegevens in de
bal. Behandel daarna het gradiëntstelsel $x' = -\nabla G(x)$ met
$G \in \mathcal C^2$ en $G(x) \to +\infty$ als $\norm x \to
\infty$: toon aan dat $G$ daalt langs de oplossingen, dat elke
oplossing in de (begrensde) sublevelverzameling $\{G \leq
G(x_0)\}$ blijft, en besluit tot globaal voorwaarts bestaan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:11}
(Explosie door vergelijking) Beschouw $x' = x^2 + t^2$, $x(0) =
1$.
(a) Toon aan dat de \omterm{thm:b3:ode:maximal}{maximale oplossing} op een $\intco0{T_+}$ met
$T_+ < \infty$ bestaat: vergelijk met $y' = y^2$, $y(0) = 1$
\emph{(bewijs het vergelijkingslemma dat je nodig hebt: is $x'
\geq F(x)$ en $y' = F(y)$ met $x(0) \geq y(0)$, dan is $x \geq y$
waar beide leven)}, en leid $T_+ \leq 1$ af.
(b) Begrens $T_+$ van onderen: op $\intcc01$ is $x' \leq x^2 +
1$; vergelijk met de superoplossing $z' = z^2 + 1$, $z(0) = 1$,
opgelost door $z(t) = \tan\bigl(t + \frac\pi4\bigr)$, en besluit
$T_+ \geq \frac\pi4$.
(c) Stel $\frac\pi4 \leq T_+ \leq 1$ samen en formuleer de
moraal: superlineaire groei van het rechterlid is wat het globale
bestaan doodt (met \cref{exo:b3:ode:3} als tegenhanger), waarbij
de grens de convergentie van $\int^{\infty}\frac{\dd s}{F(s)}$
is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:12}
(Vergelijkingsstelling van Sturm) Zij $q_1 \leq q_2$ \omterm{def:b3:topology:continuity}{continu} op
een interval $I$, en zij $u \neq 0$ een oplossing van $u'' + q_1u
= 0$ en $v \neq 0$ een oplossing van $v'' + q_2v = 0$.
(a) Vestig de identiteit voor de \omterm{thm:b3:ode:linearstructure}{wronskiaan}: met $W = uv' -
u'v$ is $W' = (q_1 - q_2)\,uv$.
(b) (Sturm) Toon aan dat tussen twee opeenvolgende nulpunten $a <
b$ van $u$ ofwel $v$ ergens in $\intoo ab$ verdwijnt, ofwel $q_1
= q_2$ en $v \propto u$ daar \emph{(neem $u > 0$ op $\intoo ab$
aan, en ook $v > 0$; integreer (a) van $a$ tot $b$ en bekijk de
tekens van de randtermen $W(a), W(b)$)}.
(c) Leid af: oplossingen van $u'' + qu = 0$ met $q \geq m > 0$
verdwijnen minstens één keer in elk interval van lengte
$\pi/\sqrt m$ (vergelijk met $v'' + mv = 0$); oplossingen met $q
\leq 0$ verdwijnen hoogstens één keer op $\R$. Test beide op $q
\equiv \pm1$.
\end{exercise}

\section{Probleem: de slinger, volledig opgelost}

\begin{problem}[{Weekendopgave --- oscillaties, rotaties,
separatrix en de periode}]\label{pb:b3:ode:1}
De slingervergelijking $x'' = -\sin x$ --- als stelsel: $x' = y$,
$y' = -\sin x$ op $\R^2$ --- is de fruitvlieg van de dynamica:
eenvoudig op te schrijven, onmogelijk met elementaire formules op
te lossen, en toch \omterm{def:b3:complete:complete}{volledig} te begrijpen met de kwalitatieve
methode. Zij $E(x, y) = \frac{y^2}2 - \cos x$ (de
\emph{energie}).

\medskip
\noindent\textbf{Deel I --- Globale structuur.}
\begin{enumerate}
  \item Toon aan dat alle \omterm{thm:b3:ode:maximal}{maximale oplossingen} globaal zijn (op
        $\R$ gedefinieerd): gebruik $\dot E = 0$ en
        \cref{thm:b3:ode:maximal}. Evenwichten: $(k\pi, 0)$;
        klasseer hun linearisaties (van centrumtype voor even
        $k$, zadel voor oneven $k$).
  \item Toon aan dat de \omterm{def:b3:groups:action}{banen} in de niveauverzamelingen $\{E =
        E_0\}$ liggen, en schets of beschrijf ze naar de waarde
        van $E_0 \in \intco{-1}{+\infty}$: $E_0 = -1$
        (evenwichten), $-1 < E_0 < 1$ (gesloten krommen rond
        $(2k\pi, 0)$), $E_0 = 1$ (de separatrix door $(\pm\pi,
        0)$), $E_0 > 1$ (grafieken boven $x$: rotaties).
  \item Bewijs dat het onderste evenwicht $(0,0)$ \omterm{def:b3:ode:stability}{stabiel} is maar
        \emph{niet} asymptotisch \omterm{def:b3:ode:stability}{stabiel}. \emph{(Lyapunov met $V
        = E + 1$; niet asymptotisch: het behoud van energie vangt
        de \omterm{def:b3:groups:action}{banen} op niveaukrommen weg van de oorsprong.)}
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Oscillaties en hun periode.} Leg $-1
< E_0 < 1$ vast en schrijf $E_0 = -\cos a$ met $a \in \intoo0\pi$
(de amplitude).
\begin{enumerate}[resume]
  \item Toon aan dat de oplossing met $x(0) = a$, $y(0) = 0$
        oscilleert: $x(t) \in \intcc{-a}a$, en de \omterm{def:b3:groups:action}{baan} is de
        gesloten kromme $y^2 = 2(\cos x - \cos a)$. Verantwoord
        dat de oplossing periodiek is: de \omterm{def:b3:groups:action}{baan} is een \omterm{def:b3:topology:compact}{compacte}
        kromme zonder evenwichten, doorlopen met een van onderen
        begrensde snelheid --- maak daar een argument van (de
        oplossing keert in eindige tijd naar haar beginpunt
        terug, en de uniciteit dwingt dan de periodiciteit af).
  \item Vestig de periodeformule
        \[
        T(a) = 4\int_0^{a}\frac{\dd x}{\sqrt{2(\cos x -
        \cos a)}}
        \]
        \emph{(op een kwart \omterm{def:b3:groups:action}{baan} is $y = \frac{\dd x}{\dd t} >
        0$; scheid de veranderlijken en verantwoord de
        oneigenlijke convergentie in $x = a$)}.
  \item (Kleine oscillaties) Substitueer $\sin\frac x2 =
        \sin\frac a2\,\sin\varphi$ en toon aan dat
        \[
        T(a) = 4\int_0^{\pi/2}\frac{\dd\varphi}{\sqrt{1 -
        k^2\sin^2\varphi}},
        \qquad k = \sin\frac a2
        \]
        (een volledige elliptische integraal). Leid met de
        gedomineerde convergentie af dat $T(a) \to 2\pi$ als $a
        \to 0^+$: de harmonische limiet, onafhankelijk van de
        amplitude --- het benaderende isochronisme van Galilei,
        met zijn exacte correctie $T(a) = 2\pi\bigl(1 +
        \frac{k^2}4 + O(k^4)\bigr)$ (ontwikkel de integrand en
        integreer term voor term, verantwoord met de normale
        convergentie).
  \item Toon aan dat $T(a) \to +\infty$ als $a \to \pi^-$
        \emph{(schat de integrand van onderen af nabij $\varphi =
        \frac\pi2$ als $k \to 1$, of pas de monotone convergentie
        toe)}: naderend tot de separatrix vertraagt de slinger
        onbegrensd.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De separatrix.}
\begin{enumerate}[resume]
  \item Voor $E_0 = 1$ is $y = 2\cos\frac x2$ op de bovenste
        tak: scheid de veranderlijken en integreer om de
        expliciete oplossing
        \[
        x(t) = 4\arctan\bigl(\eu^{t}\bigr) - \pi
        \]
        te vinden (met $x(0) = 0$, $y(0) = 2$). Ga rechtstreeks
        na dat ze de slingervergelijking oplost, en bereken haar
        limieten en die van $y(t)$ als $t \to \pm\infty$.
  \item Besluit: de \omterm{def:b3:groups:action}{baan} van de separatrix verbindt het zadel
        $(-\pi, 0)$ (als $t \to -\infty$) met het zadel $(\pi,
        0)$ (als $t \to +\infty$), maar bereikt geen van beide in
        eindige tijd --- in overeenstemming met de uniciteit
        (waarom zou een zadel in eindige tijd bereiken in
        tegenspraak zijn met \cref{cor:b3:ode:uniqueness}?).
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Rotaties, en het volledige beeld.}
\begin{enumerate}[resume]
  \item Voor $E_0 > 1$: toon aan dat $y$ nooit verdwijnt, dat $x$
        strikt monotoon en globaal is met $x(t) \to \pm\infty$,
        en dat $t \mapsto y(t)$ periodiek is met periode
        \[
        \tau(E_0) = \int_{-\pi}^{\pi}
        \frac{\dd x}{\sqrt{2(E_0 + \cos x)}} .
        \]
  \item Stel het volledige faseportret samen (de figuur van dit
        hoofdstuk) met een volledige verantwoording van elk
        kenmerk, en schrijf een samenvatting van tien regels over
        de methode: energie, niveauverzamelingen, \omterm{def:b3:topology:compact}{compactheid},
        uniciteit --- hoe elke stelling van het hoofdstuk
        binnenkwam. Waar hadden we ooit een formule voor de
        algemene oplossing nodig?
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- De periodefunctie onder de
microscoop.}
\begin{enumerate}[resume]
  \item Bewijs de momenten van Wallis
        \[
        W_n = \int_0^{\pi/2}\sin^{2n}\varphi\,\dd\varphi
        = \frac\pi2\cdot\frac{(2n)!}{4^n\,(n!)^2}
        \]
        met inductie (partieel integreren), ontwikkel de
        integrand van vraag 6 met de binomiaalreeks, en
        verantwoord het term voor term integreren om de volledige
        reeks
        \[
        T(a) = 2\pi\sum_{n\geq0}
        \Bigl(\frac{(2n)!}{4^n(n!)^2}\Bigr)^{\!2}k^{2n}
        = 2\pi\Bigl(1 + \frac{k^2}4 + \frac{9k^4}{64}
        + O(k^6)\Bigr),
        \qquad k = \sin\frac a2
        \]
        te verkrijgen.
  \item Zet om naar de amplitude:
        \[
        T(a) = 2\pi\Bigl(1 + \frac{a^2}{16} +
        \frac{11\,a^4}{3072} + O(a^6)\Bigr)
        \]
        \emph{(vul de ontwikkeling van $\sin\frac a2$ in en
        verzamel)}. Het isochronisme faalt in orde $a^2$, en dat
        falen is nu tot in orde $a^4$ gekwantificeerd.
  \item Toon aan dat $a \mapsto T(a)$ \omterm{def:b3:topology:continuity}{continu} en strikt stijgend
        is op $\intoo0\pi$, en besluit met de vragen 6 en 7 dat
        $T$ een bijectie is van $\intoo0\pi$ op
        $\intoo{2\pi}{+\infty}$: elke superkritische periode
        wordt door precies één amplitude verwezenlijkt.
  \item (Rekenkunde van de klokkenmaker) Een slinger die bij
        verdwijnende amplitude is geregeld, houdt ideale tijd;
        toon aan dat ze bij amplitude $a$ achterloopt met de
        fractie $\frac{a^2}{16} + O(a^4)$ van de ideale tijd, en
        bereken de afwijking voor $a = 0.2$ rad: ongeveer $216$
        seconden per dag. (De cycloïdale wangen van Huygens en de
        kleine constante amplitudes van echappementen zijn beide
        antwoorden op dit getal.)
  \item Keer terug naar de rotatieperiode $\tau$ uit vraag 10:
        toon aan dat $\tau$ strikt dalend is op
        $\intoo1{+\infty}$, dat $\tau(E_0) \to +\infty$ als $E_0
        \to 1^+$ (monotone convergentie), en dat
        $\sqrt{2E_0}\,\tau(E_0) \to 2\pi$ als $E_0 \to +\infty$
        (gedomineerde convergentie): snel rondwentelen is
        asymptotisch vrije rotatie met hoeksnelheid
        $\sqrt{2E_0}$.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- De methode uitgevoerd:
Lotka--Volterra.} Het recept van de slinger --- eerste integraal,
\omterm{def:b3:topology:compact}{compacte} niveaukrommen, uniciteit --- lost een ecosysteem op. Leg
$\alpha, \beta, \gamma, \delta > 0$ vast en beschouw op het open
kwadrant $Q = \intoo0{+\infty}\times\intoo0{+\infty}$
\[
x' = x\,(\alpha - \beta y), \qquad
y' = y\,(\delta x - \gamma)
\]
($x$ de prooi, $y$ de roofdieren).
\begin{enumerate}[resume]
  \item Toon aan dat $Q$ invariant is --- de assen zijn
        verenigingen van \omterm{def:b3:groups:action}{banen}, expliciet berekenbaar, die geen
        enkele oplossing mag kruisen
        (\cref{cor:b3:ode:uniqueness}) --- en dat het enige
        evenwicht in $Q$ $(x_*, y_*) = \bigl(\frac\gamma\delta,
        \frac\alpha\beta\bigr)$ is.
  \item Toon aan dat
        \[
        H(x, y) = \delta x - \gamma\ln x + \beta y -
        \alpha\ln y
        \]
        een eerste integraal is, dat $H = f(x) + g(y)$ met $f, g$
        strikt convex en eigenlijk op $\intoo0{+\infty}$ met
        minima in $x_*$ en $y_*$, en leid af dat alle \omterm{thm:b3:ode:maximal}{maximale
        oplossingen} in $Q$ globaal zijn.
  \item Toon aan dat voor $h > h_* = H(x_*, y_*)$ de
        niveauverzameling $\{H = h\}\cap Q$ een gesloten kromme
        rond het evenwicht is: twee \omterm{def:b3:topology:continuity}{continue} takken $y_\pm(x)$
        boven een \omterm{def:b3:topology:compact}{compact} interval $\intcc{x_-}{x_+} \ni x_*$,
        aan de uiteinden aan elkaar gelijmd --- het analogon van
        de ovalen van de slinger.
  \item Bewijs dat elke \omterm{def:b3:groups:action}{baan} in $Q$ die geen evenwicht is,
        periodiek is: vestig de circulatie tegen de wijzers van
        de klok in door de vier gebieden die door de rechten $x =
        x_*$ en $y = y_*$ worden afgesneden, begrens de tijd om
        elke boog te doorlopen met een integraal met een
        convergente vierkantswortelsingulariteit (als in vraag
        5), en sluit af met de uniciteit (als in vraag 4).
  \item (Wet van de gemiddelden van Volterra) Is $T$ de periode
        van zo'n \omterm{def:b3:groups:action}{baan}, toon dan aan dat
        \[
        \frac1T\int_0^Tx(t)\,\dd t = \frac\gamma\delta,
        \qquad
        \frac1T\int_0^Ty(t)\,\dd t = \frac\alpha\beta :
        \]
        de tijdsgemiddelden zijn gelijk aan de evenwichtswaarden,
        wat de amplitude ook is \emph{(integreer $(\ln x)' =
        \alpha - \beta y$ over één periode)}.
  \item (De visserijparadox) Oogst beide soorten met snelheid
        $\varepsilon \in \intoo0\alpha$: het stelsel behoudt haar
        vorm met $\alpha - \varepsilon$ en $\gamma + \varepsilon$
        in de plaats van $\alpha$ en $\gamma$. Wat gebeurt er met
        de gemiddelde populaties? Verklaar de waarneming van
        d'Ancona (1914--1918): toen de visserij op de Adriatische
        Zee tijdens de oorlog afnam, \emph{steeg} het aandeel van
        de roofdieren (haaien) in de vangst --- en waarom
        gematigde visserij de prooi bevoordeelt.
  \item Schrijf de moraal van tien regels: welke stellingen van
        het hoofdstuk elke stap aandrijven, wat de energie van de
        slinger vervangt, en waarom geen van beide stelsels een
        elementaire gesloten oplossing nodig had --- of toelaat.

  \item (Snelheidsmodulatie) Toon in het rotatieregime $E_0 > 1$
        aan dat $y = x'$ oscilleert tussen $\sqrt{2(E_0 - 1)}$
        (in $x \equiv \pi \bmod 2\pi$) en $\sqrt{2(E_0 + 1)}$ (in
        $x \equiv 0$), dat het tijdsgemiddelde van $y$ over één
        periode precies $\frac{2\pi}{\tau(E_0)}$ is, en dat de
        modulatieverhouding $\sqrt{\frac{E_0 + 1}{E_0 - 1}} \to
        1$ als $E_0 \to \infty$: snelle rotatie is asymptotisch
        uniform.
  \item (Monotonie van de rotatieperiode) Toon aan dat
        $\tau(E_0)$ $\mathcal C^1$ en strikt dalend is op
        $\intoo1{+\infty}$ \emph{(differentieer onder het
        integraalteken, met dominantie op elke $\intco{1 +
        \delta}\infty$)}, met $\tau \to \infty$ als $E_0 \to 1^+$
        en $\tau \to 0$ als $E_0 \to \infty$. Stel het volledige
        bifurcatiebeeld van de slinger langs de energieas samen:
        evenwichten bij $E_0 = -1$, libraties met een periode die
        van $2\pi$ tot $\infty$ stijgt op $-1 < E_0 < 1$, de
        separatrix bij $E_0 = 1$, en daarbuiten rotaties met een
        periode die van $\infty$ tot $0$ daalt.
\end{enumerate}
\end{problem}
