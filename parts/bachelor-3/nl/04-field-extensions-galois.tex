\chapter{Lichaamsuitbreidingen en Galoistheorie}\label{ch:b3:galois}

Is elke vergelijking \omterm{def:b3:galois:radical}{oplosbaar door worteltrekking}, zoals de
abc-formule en de formules van Cardano voor de derde \omterm{def:b3:galois:extension}{graad} doen
vermoeden? Kun je een hoek in drieën delen met liniaal en passer?
Beide vragen, eeuwenlang open, worden --- ontkennend --- beantwoord
door één idee van Évariste \omterm{def:b3:galois:galois}{Galois}: hang aan elke veelterm een
eindige \emph{groep} symmetrieën van haar wortels, en lees het
antwoord van die groep af. Dit hoofdstuk bouwt het woordenboek:
\omterm{def:b3:galois:extension}{lichaamsuitbreidingen} en graden, splitsingslichamen en
\omterm{def:b3:galois:closure}{algebraïsche afsluitingen}, eindige lichamen (een volledige theorie
--- met de beloofde cycliciteit van $\mathbb F_q^\times$),
separabiliteit, en dan de \omterm{thm:b3:galois:fundamental}{Galoiscorrespondentie} zelf, met volledige
bewijzen. De oogst: de onmogelijkheid van de klassieke
constructies, de structuur van de cyclotomische lichamen, en de
onoplosbaarheid van de vijfdegraadsvergelijking door
worteltrekking --- de enkelvoudigheid van $A_5$ uit
\cref{ch:b3:groups} die haar doel treft.

\section{Uitbreidingen, graad, algebraïciteit}

\begin{definition}\label{def:b3:galois:extension}
Een \emph{lichaamsuitbreiding}\index{lichaamsuitbreiding} $L/K$ is
een lichaam $L$ dat $K$ als deellichaam bevat; $L$ is dan een
$K$-vectorruimte, en de \emph{graad}\index{graad van een
uitbreiding} $[L:K]$ is de dimensie daarvan. De uitbreiding heet
\emph{eindig} als $[L:K] < \infty$. De \emph{karakteristiek} van
een lichaam is de voortbrenger $\geq 0$ van de kern van $\Z \to
K$, $n \mapsto n\cdot 1$: die is $0$ of een priemgetal $p$;
dienovereenkomstig bevat $K$ een kleinste deellichaam
(\emph{priemlichaam}) isomorf met $\Q$ of met $\mathbb F_p =
\Z/p\Z$.
\end{definition}

\begin{theorem}[Torenwet]\label{thm:b3:galois:tower}
Is $K \subseteq L \subseteq M$, dan is $[M:K] = [M:L]\,[L:K]$: is
$(e_i)$ een basis van $L$ over $K$ en $(f_j)$ een basis van $M$
over $L$, dan is $(e_if_j)$ een basis van $M$ over
$K$.\index{torenwet}
\end{theorem}

\begin{proof}
Voortbrengend: een $x \in M$ schrijft zich als $x = \sum_j
\lambda_jf_j$ met $\lambda_j \in L$, en elke $\lambda_j = \sum_i
\mu_{ij}e_i$ met $\mu_{ij} \in K$: dus $x =
\sum_{i,j}\mu_{ij}e_if_j$. Onafhankelijk: $\sum_{i,j}\mu_{ij}
e_if_j = 0$ herschrijft zich als $\sum_j (\sum_i \mu_{ij}e_i)f_j =
0$; de binnenste sommen liggen in $L$ en verdwijnen dus (de $f_j$
zijn onafhankelijk over $L$); daarna zijn alle $\mu_{ij} = 0$ (de
$e_i$ zijn onafhankelijk over $K$).
\end{proof}

\begin{definition}\label{def:b3:galois:algebraic}
Zij $L/K$ en $\alpha \in L$. Is er een $P \in K[X]$ met $P \neq 0$
en $P(\alpha) = 0$, dan heet $\alpha$
\emph{algebraïsch}\index{algebraïsch element} over $K$; de
monische voortbrenger $\pi_\alpha$ van het \omterm{def:b3:rings:ideal}{ideaal} $\{P :
P(\alpha) = 0\}$ van $K[X]$ heet haar
\emph{minimaalveelterm}\index{minimaalveelterm
(lichaamstheorie)}, een irreducibele veelterm ($\pi = QR$ met
$Q(\alpha) = 0$ dwingt $R$ constant af wegens de minimaliteit van
de \omterm{def:b3:galois:extension}{graad}). Anders heet $\alpha$ \emph{transcendent}. We schrijven
$K(\alpha)$ voor het kleinste deellichaam van $L$ dat $K$ en
$\alpha$ bevat, en $K[\alpha]$ voor de kleinste zulke deelring.
\end{definition}

\begin{theorem}\label{thm:b3:galois:simple}
Is $\alpha$ \omterm{def:b3:galois:algebraic}{algebraïsch} over $K$ met $d = \deg\pi_\alpha$, dan is
\[
K(\alpha) = K[\alpha] \cong K[X]/(\pi_\alpha),
\qquad [K(\alpha) : K] = d,
\]
met basis $1, \alpha, \dots, \alpha^{d-1}$. Omgekeerd is, zodra
$[L:K] < \infty$, elke $\alpha \in L$ \omterm{def:b3:galois:algebraic}{algebraïsch} van een \omterm{def:b3:galois:extension}{graad}
$\deg \pi_\alpha$ die $[L:K]$ deelt.
\end{theorem}

\begin{proof}
De evaluatie $K[X] \to L$, $P \mapsto P(\alpha)$, heeft beeld
$K[\alpha]$ en kern $(\pi_\alpha)$; omdat $\pi_\alpha$
\omterm{def:b3:rings:divisibility}{irreducibel} is, is $K[X]/(\pi_\alpha)$ een \emph{lichaam}
(\cref{prop:b3:rings:primemaximal}: in het \omterm{def:b3:rings:pidufd}{hoofdideaaldomein}
$K[X]$ brengt een \omterm{def:b3:rings:divisibility}{irreducibel element} een \omterm{def:b3:rings:primemaximal}{maximaal ideaal} voort),
dus is $K[\alpha]$ een lichaam dat $K$ en $\alpha$ bevat: het is
gelijk aan $K(\alpha)$. De klassen van $1, X, \dots, X^{d-1}$
vormen een basis van het quotiënt (euclidische deling), waaruit de
basis en de \omterm{def:b3:galois:extension}{graad} volgen. Omgekeerd, is $[L:K] = n < \infty$, dan
zijn $1, \alpha, \dots, \alpha^n$ afhankelijk, wat een
annihilerende veelterm levert; en dan deelt $[K(\alpha):K] =
\deg\pi_\alpha$ het getal $n$ volgens de \omterm{thm:b3:galois:tower}{torenwet}.
\end{proof}

\begin{corollary}\label{cor:b3:galois:algclosed}
Zijn $\alpha$ en $\beta$ \omterm{def:b3:galois:algebraic}{algebraïsch} over $K$, dan ook $\alpha \pm
\beta$, $\alpha\beta$ en $\alpha/\beta$ (voor $\beta \ne 0$): de
elementen van $L$ die \omterm{def:b3:galois:algebraic}{algebraïsch} zijn over $K$ vormen een
deellichaam van $L$. Bovendien is algebraïciteit transitief:
\omterm{def:b3:galois:algebraic}{algebraïsch} over \omterm{def:b3:galois:algebraic}{algebraïsch} is \omterm{def:b3:galois:algebraic}{algebraïsch}.
\end{corollary}

\begin{proof}
$K(\alpha, \beta) = (K(\alpha))(\beta)$ is eindig over
$K(\alpha)$ ($\beta$ is \omterm{def:b3:galois:algebraic}{algebraïsch} over $K \subseteq K(\alpha)$)
en $K(\alpha)/K$ is eindig: volgens de \omterm{thm:b3:galois:tower}{torenwet} is
$[K(\alpha,\beta):K] < \infty$, en is elk element van $K(\alpha,
\beta)$ --- de vier genoemde inbegrepen --- \omterm{def:b3:galois:algebraic}{algebraïsch}
(\cref{thm:b3:galois:simple}). Transitiviteit: is $\beta$
\omterm{def:b3:galois:algebraic}{algebraïsch} over $L$ en $L/K$ \omterm{def:b3:galois:algebraic}{algebraïsch}, dan brengen de
coëfficiënten $c_0, \dots, c_{m-1}$ van $\pi_{\beta/L}$ een
eindige uitbreiding $F = K(c_0, \dots, c_{m-1})$ van $K$ voort
(\omterm{thm:b3:galois:tower}{torenwet}, herhaald), en $F(\beta)/F$ is eindig: dus
$[F(\beta):K] < \infty$ en is $\beta$ \omterm{def:b3:galois:algebraic}{algebraïsch} over $K$.
\end{proof}

\begin{example}\label{ex:b3:galois:degrees}
$[\Q(\sqrt2):\Q] = 2$, $[\Q(\sqrt[3]2):\Q] = 3$ ($X^3 - 2$ is
\omterm{def:b3:rings:divisibility}{irreducibel}: \omterm{thm:b3:rings:criteria}{Eisenstein}), $[\Q(\zeta_p):\Q] = p - 1$ voor
$\zeta_p = \eu^{2\iu\pi/p}$ ($\Phi_p$ is \omterm{def:b3:rings:divisibility}{irreducibel},
\cref{ex:b3:rings:eisenstein}). De \omterm{thm:b3:galois:tower}{torenwet} is nu al een wapen:
$\sqrt[3]2 \notin \Q(\sqrt2)$, want $3 \nmid 2$.
\end{example}

\section{Splitsingslichamen; algebraïsche afsluiting}

\begin{theorem}[Splitsingslichamen]\label{thm:b3:galois:splitting}
Zij $P \in K[X]$ niet-constant. Dan bestaat er een
\emph{splitsingslichaam}\index{splitsingslichaam} van $P$ over
$K$: een uitbreiding $L = K(\alpha_1, \dots, \alpha_n)$
voortgebracht door wortels van $P$, waarin $P$ in lineaire
factoren uiteenvalt. Het is uniek op $K$-isomorfie na, en $[L:K]
\leq (\deg P)!$.
\end{theorem}

\begin{proof}
\emph{Bestaan}, met inductie naar $\deg P$: kies een irreducibele
factor $Q$ van $P$; het lichaam $K_1 = K[X]/(Q)$ bevat de wortel
$\alpha_1 = \bar X$ van $Q$, en dus van $P$; schrijf $P = (X -
\alpha_1)P_1$ over $K_1$ en pas de inductiehypothese toe op $P_1$
over $K_1$; de graden vermenigvuldigen tot hoogstens
$n(n-1)\cdots = n!$.

\emph{Eenduidigheid} volgt uit het sterkere \emph{lemma over het
voortzetten van isomorfismen}: zij $\sigma \colon K \to K'$ een
isomorfisme, $P \in K[X]$, $P^\sigma$ de veelterm met de
afgebeelde coëfficiënten, en $L, L'$ splitsingslichamen van $P$
respectievelijk $P^\sigma$; dan zet $\sigma$ zich voort tot een
isomorfisme $L \to L'$. Inductie naar $[L:K]$: valt $P$ al in $K$
uiteen, dan is $L = K$ en $L' = K'$ ($P^\sigma$ valt uiteen in
$K'$, en $L'$ wordt door haar wortels voortgebracht). Kies anders
een wortel $\alpha \in L \setminus K$ van een irreducibele factor
$Q$ van $P$ met $\deg Q \geq 2$; dan is $Q^\sigma$ een
irreducibele factor van $P^\sigma$, met een wortel $\beta \in
L'$, en
\[
K(\alpha) \cong K[X]/(Q) \xrightarrow{\ \sigma\ }
K'[X]/(Q^\sigma) \cong K'(\beta)
\]
zet $\sigma$ voort met $\alpha \mapsto \beta$. Nu is $L$ een
\omterm{thm:b3:galois:splitting}{splitsingslichaam} van $P$ over $K(\alpha)$ en $L'$ er een van
$P^\sigma$ over $K'(\beta)$, met $[L : K(\alpha)] < [L:K]$: de
inductie zet verder voort tot $L \to L'$.
\end{proof}

\begin{definition}\label{def:b3:galois:closure}
Een lichaam $\Omega$ heet \emph{algebraïsch
gesloten}\index{algebraïsch gesloten lichaam} als elke
niet-constante veelterm van $\Omega[X]$ een wortel in $\Omega$
heeft (en er dus in uiteenvalt). Een \emph{algebraïsche
afsluiting}\index{algebraïsche afsluiting} van $K$ is een
\omterm{def:b3:galois:algebraic}{algebraïsche} uitbreiding $\bar K/K$ met $\bar K$ \omterm{def:b3:galois:algebraic}{algebraïsch}
gesloten.
\end{definition}

\begin{theorem}[Steinitz]\label{thm:b3:galois:steinitz}
Elk lichaam $K$ heeft een \omterm{def:b3:galois:closure}{algebraïsche afsluiting}, uniek op
$K$-isomorfie na.
\end{theorem}

\begin{proof}
\emph{Bestaan (de constructie van Artin).} Zij $R = K[(X_f)_f]$ de
veeltermring met één veranderlijke $X_f$ per niet-constante
monische $f \in K[X]$, en $I$ het \omterm{def:b3:rings:ideal}{ideaal} voortgebracht door alle
$f(X_f)$. $I$ is een echt \omterm{def:b3:rings:ideal}{ideaal}: een betrekking $1 =
\sum_{i=1}^r g_i\, f_i(X_{f_i})$ betreft eindig veel veeltermen;
kies in een gemeenschappelijk \omterm{thm:b3:galois:splitting}{splitsingslichaam} $E$ van $f_1\cdots
f_r$ wortels $\alpha_i$ van $f_i$ en evalueer $X_{f_i} \mapsto
\alpha_i$ (de overige veranderlijken $\mapsto 0$): dan is $1 = 0$,
absurd. Zij $\mathfrak m \supseteq I$ maximaal
(\cref{thm:b3:rings:krull}; Zorn) en $K_1 = R/\mathfrak m$: een
\omterm{def:b3:galois:extension}{lichaamsuitbreiding} van $K$ waarin elke niet-constante $f \in
K[X]$ een wortel heeft, namelijk $\bar X_f$, en die \omterm{def:b3:galois:algebraic}{algebraïsch} is
over $K$ (ze wordt voortgebracht door de $\bar X_f$, elk
\omterm{def:b3:galois:algebraic}{algebraïsch}). Herhaal: $K \subseteq K_1 \subseteq K_2 \subseteq
\cdots$, waarbij $K_{n+1}$ met $K_n$ doet wat $K_1$ met $K$ deed,
en stel $\Omega = \bigcup_n K_n$, een lichaam. Een niet-constante
$g \in \Omega[X]$ heeft haar eindig veel coëfficiënten in een
zekere $K_n$; een irreducibele factor van $g$ over $K_n$ heeft een
wortel in $K_{n+1} \subseteq \Omega$: dus is $\Omega$ \omterm{def:b3:galois:closure}{algebraïsch
gesloten}, en \omterm{def:b3:galois:algebraic}{algebraïsch} over $K$ (elke $K_n$ is dat, en dan
transitiviteit, \cref{cor:b3:galois:algclosed}): $\Omega$ is een
\omterm{def:b3:galois:closure}{algebraïsche afsluiting}.

\emph{Eenduidigheid.} Zij $\Omega, \Omega'$ twee \omterm{def:b3:galois:closure}{algebraïsche
afsluitingen}. Beschouw de verzameling paren $(E, \tau)$ met $K
\subseteq E \subseteq \Omega$ en $\tau\colon E \to \Omega'$ een
$K$-inbedding, geordend naar voortzetting; ze is niet leeg ($(K,
\mathrm{id})$) en inductief (vereniging van een keten), zodat Zorn
een maximale $(E_0, \tau_0)$ levert. Was $E_0 \neq \Omega$, kies
dan $\alpha \in \Omega\setminus E_0$: $\pi_{\alpha/E_0}$ gaat over
in een veelterm over $\tau_0(E_0)$ die in het \omterm{def:b3:galois:closure}{algebraïsch gesloten}
$\Omega'$ een wortel $\beta$ heeft, en $\tau_0$ zet zich voort tot
$E_0(\alpha) \to \Omega'$ ($\alpha \mapsto \beta$), in strijd met
de maximaliteit. Er is dus een $K$-inbedding $\tau \colon \Omega
\to \Omega'$; haar beeld, isomorf met $\Omega$, is \omterm{def:b3:galois:closure}{algebraïsch
gesloten}, en $\Omega'$ is er \omterm{def:b3:galois:algebraic}{algebraïsch} over: voor $x \in
\Omega'$ valt $\pi_{x/\tau(\Omega)}$ over $\tau(\Omega)$ uiteen,
dus $x \in \tau(\Omega)$. Bijgevolg is $\tau$ surjectief: een
isomorfisme.
\end{proof}

\begin{remark}\label{rem:b3:galois:closureexamples}
Voor $K = \Q$ kan men de transfiniete machinerie vermijden: de
\omterm{def:b3:galois:algebraic}{algebraïsche} getallen $\bar\Q = \{z \in \C : z \text{ algebraïsch
over } \Q\}$ vormen een \omterm{def:b3:galois:closure}{algebraïsche afsluiting} --- een
deellichaam van $\C$ volgens \cref{cor:b3:galois:algclosed},
\omterm{def:b3:galois:closure}{algebraïsch gesloten} omdat $\C$ het is (d'Alembert--Gauss,
bewezen met complexe analyse in \cref{ch:b3:holomorphic}), en
wortels van veeltermen over $\bar\Q$ zijn wegens de
transitiviteit \omterm{def:b3:galois:algebraic}{algebraïsch} over $\Q$.
\end{remark}

\section{Eindige lichamen}

\begin{theorem}\label{thm:b3:galois:finitefields}
Zij $p$ priem, $n \geq 1$ en $q = p^n$.
\begin{enumerate}
  \item Een \omterm{thm:b3:galois:finitefields}{eindig lichaam} heeft als kardinaliteit een
        priemmacht, en voor elke $q$ is er op isomorfie na precies
        één lichaam $\mathbb F_q$ met $q$ elementen: het
        \omterm{thm:b3:galois:splitting}{splitsingslichaam} van $X^q - X$ over $\mathbb
        F_p$.\index{eindig lichaam}
  \item De \emph{Frobenius}\index{frobeniusendomorfisme} $F \colon
        x \mapsto x^p$ is een automorfisme van $\mathbb F_q$, en
        de automorfismegroep van $\mathbb F_q$ is cyclisch van
        orde $n$, voortgebracht door $F$.
  \item $\mathbb F_{p^m}$ bedt zich in $\mathbb F_{p^n}$ in dan en
        slechts dan als $m \mid n$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Een \omterm{thm:b3:galois:finitefields}{eindig lichaam} $E$ heeft karakteristiek $p > 0$ en is een
eindigdimensionale $\mathbb F_p$-vectorruimte: $\abs E = p^n$.
Haar multiplicatieve groep heeft orde $q - 1$, dus voldoet elke $x
\in E$ aan $x^q = x$: $E$ bestaat uit $q$ wortels van $X^q - X$ en
is er dus een \omterm{thm:b3:galois:splitting}{splitsingslichaam} van over $\mathbb F_p$ --- wat $E$
op isomorfie na vastlegt (\cref{thm:b3:galois:splitting}).
Omgekeerd is in een \omterm{thm:b3:galois:splitting}{splitsingslichaam} $L$ van $X^q - X$ de
verzameling $E$ van haar wortels een \emph{deellichaam}: $(x +
y)^q = x^q + y^q$ door de droom van de eerstejaars $(a+b)^p = a^p
+ b^p$ te herhalen ($p \mid \binom pk$), en $(xy)^q = x^qy^q$,
$(x^{-1})^q = (x^q)^{-1}$; het heeft precies $q$ elementen omdat
$X^q - X$ \omterm{def:b3:galois:separable}{separabel} is: haar afgeleide is $qX^{q-1} - 1 = -1$
(want $p \mid q$), onderling ondeelbaar met haar, dus zijn er geen
meervoudige wortels. Bijgevolg heeft $L = E$ precies $q$
elementen.

(2) $F$ is een lichaamsmorfisme (droom van de eerstejaars) en
injectief (lichamen), dus bijectief op het eindige $\mathbb F_q$.
Er geldt $F^n = \mathrm{id}$ (want $x^q = x$), en geen kleinere
macht is de identiteit: $F^m = \mathrm{id}$ zou betekenen dat alle
$q$ elementen wortels van $X^{p^m} - X$ zijn, wat $p^m \geq q$
afdwingt. Dus is $\langle F\rangle$ cyclisch van orde $n$; en er
zijn geen andere automorfismen, wegens de hieronder bewezen grens
$\abs{\operatorname{Aut}} \leq [\,\mathbb F_q : \mathbb F_p\,] =
n$ (\cref{prop:b3:galois:embeddings} met $L = \mathbb F_q$ en $K =
\mathbb F_p$: automorfismen houden het priemlichaam vast).

(3) Is $\mathbb F_{p^m} \subseteq \mathbb F_{p^n}$, dan geeft de
\omterm{thm:b3:galois:tower}{torenwet} $p^n = (p^m)^d$: dus $m \mid n$. Is omgekeerd $m \mid n$,
dan is $p^m - 1 \mid p^n - 1$ (meetkundige som), zodat $X^{p^m} -
X$ de veelterm $X^{p^n} - X$ deelt (hetzelfde argument op de
exponenten: $X^{a} - 1 \mid X^{b} - 1$ zodra $a \mid b$), en
vormen de wortels van de eerste binnen $\mathbb F_{p^n}$ het
gevraagde deellichaam, van kardinaliteit $p^m$ (separabiliteit als
in (1)).
\end{proof}

\begin{theorem}[Cycliciteit]\label{thm:b3:galois:cyclic}
Elke eindige deelgroep van de multiplicatieve groep van een
lichaam is cyclisch. In het bijzonder is $\mathbb F_q^\times \cong
\Z/(q-1)\Z$.\index{multiplicatieve groep van een eindig lichaam}
\end{theorem}

\begin{proof}
Zij $G \leq K^\times$ eindig. Volgens de structuurstelling
(\cref{cor:b3:modules:abelian}) is $G \cong \Z/d_1 \times \dots
\times \Z/d_s$ met $d_1 \mid \dots \mid d_s$. Elke $x \in G$
voldoet dan aan $x^{d_s} = 1$; maar $X^{d_s} - 1$ heeft hoogstens
$d_s$ wortels in het lichaam $K$: dus $\abs G = d_1\cdots d_s \leq
d_s$, wat $s = 1$ afdwingt: $G$ is cyclisch.
\end{proof}

\begin{example}\label{ex:b3:galois:f8}
$\mathbb F_8 = \mathbb F_2[X]/(X^3 + X + 1)$: de derdegraadsveelterm
heeft geen wortel in $\mathbb F_2$ en is dus \omterm{def:b3:rings:divisibility}{irreducibel}. Met
$\omega = \bar X$ is $\mathbb F_8^\times$ cyclisch van orde $7$,
zodat \emph{elk} element $\neq 0, 1$ voortbrengt. De deellichamen
van $\mathbb F_{p^{12}}$ vormen het delertralie van $12$: $\mathbb
F_p, \mathbb F_{p^2}, \mathbb F_{p^3}, \mathbb F_{p^4}, \mathbb
F_{p^6}, \mathbb F_{p^{12}}$ --- een eerste, volledig geval van de
\omterm{thm:b3:galois:fundamental}{Galoiscorrespondentie}.
\end{example}

\section{Separabiliteit en inbeddingen}

\begin{definition}\label{def:b3:galois:separable}
Een veelterm $P \in K[X]$ heet \emph{separabel}\index{separabele
veelterm} als ze in een \omterm{thm:b3:galois:splitting}{splitsingslichaam} geen meervoudige wortel
heeft --- gelijkwaardig: $\gcd(P, P') = 1$ (een meervoudige wortel
is een gemeenschappelijke wortel; omgekeerd is over het
\omterm{thm:b3:galois:splitting}{splitsingslichaam} een gemeenschappelijke wortel meervoudig; en de
\omterm{lem:b3:rings:bezout}{grootste gemene deler} verandert niet onder \omterm{def:b3:galois:extension}{lichaamsuitbreiding},
zie het argument bij \cref{cor:b3:modules:descent}). Een
\omterm{def:b3:galois:algebraic}{algebraïsch element} heet separabel als haar \omterm{def:b3:galois:algebraic}{minimaalveelterm} het
is, en een uitbreiding $L/K$ heet separabel als al haar elementen
het zijn.
\end{definition}

\begin{proposition}\label{prop:b3:galois:perfect}
Een \emph{irreducibele} $P \in K[X]$ is \omterm{def:b3:galois:separable}{separabel}, tenzij $P' =
0$, wat $\operatorname{char} K = p > 0$ en $P \in K[X^p]$
afdwingt. Bijgevolg is elke \omterm{def:b3:galois:algebraic}{algebraïsche} uitbreiding van een
lichaam van karakteristiek $0$, en van een \omterm{thm:b3:galois:finitefields}{eindig lichaam},
\omterm{def:b3:galois:separable}{separabel} (zulke lichamen heten \emph{perfect}\index{perfect
lichaam}).
\end{proposition}

\begin{proof}
$\gcd(P, P')$ deelt $P$; is die niet $1$, dan dwingt de
irreducibiliteit $\gcd = P$ af (op een constante na), dus $P \mid
P'$ met $\deg P' < \deg P$: bijgevolg $P' = 0$. Schrijf $P = \sum
a_kX^k$: dan is $ka_k = 0$ voor alle $k$, zodat $P$ in
karakteristiek $0$ constant is (uitgesloten), en in karakteristiek
$p$ geldt $a_k = 0$ tenzij $p \mid k$: dus $P = Q(X^p)$. Over een
\omterm{thm:b3:galois:finitefields}{eindig lichaam} is elk element een $p$-de macht (\omterm{thm:b3:galois:finitefields}{Frobenius} is
surjectief), zodat $Q(X^p) = \sum b_k^p X^{pk} = (\sum b_kX^k)^p$
niet \omterm{def:b3:rings:divisibility}{irreducibel} is: ook daar kan $P' = 0$ voor irreducibele $P$
niet voorkomen.
\end{proof}

\begin{proposition}[Inbeddingen tellen]\label{prop:b3:galois:embeddings}
Zij $L = K(\alpha_1, \dots, \alpha_r)$ eindig over $K$, en $\sigma
\colon K \to \Omega$ een inbedding in een \omterm{def:b3:galois:closure}{algebraïsch gesloten
lichaam}. Dan is het aantal voortzettingen van $\sigma$ tot $L$
hoogstens $[L:K]$, met gelijkheid als $L/K$ \omterm{def:b3:galois:separable}{separabel} is. In het
bijzonder is $\abs{\operatorname{Aut}_K(L)} \leq [L:K]$.
\end{proposition}

\begin{proof}
Inductie naar $[L:K]$ via enkelvoudige stappen. Voor $L =
K(\alpha)$ ligt een voortzetting $\tau$ vast door $\tau(\alpha)$,
dat een wortel in $\Omega$ van $\pi_\alpha^\sigma$ moet zijn; en
omgekeerd geeft elke zulke wortel één voortzetting ($K(\alpha)
\cong K[X]/(\pi_\alpha)$). Het aantal voortzettingen is dus het
aantal \emph{verschillende} wortels van $\pi_\alpha^\sigma$ in
$\Omega$: hoogstens $\deg\pi_\alpha = [K(\alpha):K]$, met
gelijkheid precies wanneer $\pi_\alpha$ \omterm{def:b3:galois:separable}{separabel} is (de
separabiliteit van $\pi^\sigma$ en van $\pi$ komen overeen: $\sigma$
bewaart de \omterm{lem:b3:rings:bezout}{grootste gemene deler} met de afgeleide). In het
algemeen splitsen we $L = K(\alpha_1)(\alpha_2, \dots)$: er zijn
$\leq [K(\alpha_1):K]$ voortzettingen van $\sigma$ tot
$K(\alpha_1)$, en elk daarvan zet zich per inductie op $\leq [L :
K(\alpha_1)]$ manieren voort; vermenigvuldig (\omterm{thm:b3:galois:tower}{torenwet}). In het
separabele geval zijn beide tellingen gelijkheden: de
\omterm{def:b3:galois:algebraic}{minimaalveeltermen} over het grotere lichaam $K(\alpha_1)$ delen
die over $K$ en blijven dus \omterm{def:b3:galois:separable}{separabel}.
\end{proof}

\begin{theorem}[Primitief element]\label{thm:b3:galois:primitive}
Elke eindige \emph{separabele} uitbreiding is enkelvoudig: $L =
K(\gamma)$ voor zekere $\gamma$.\index{stelling van het primitieve
element}
\end{theorem}

\begin{proof}
Is $K$ eindig, dan is $L$ dat ook, en volstaat een voortbrenger
$\gamma$ van de cyclische groep $L^\times$
(\cref{thm:b3:galois:cyclic}). Zij $K$ oneindig; per inductie
volstaat het geval $L = K(\alpha, \beta)$. Zij $n = [L:K]$;
volgens \cref{prop:b3:galois:embeddings} zijn er $n$
verschillende $K$-inbeddingen $\sigma_1, \dots, \sigma_n \colon L
\to \Omega$ (met $\Omega$ een \omterm{def:b3:galois:closure}{algebraïsche afsluiting}). De
veelterm
\[
D(T)=\prod_{i<j}\bigl[\bigl(\sigma_i(\alpha)-\sigma_j(\alpha)
\bigr) + T\bigl(\sigma_i(\beta) - \sigma_j(\beta)\bigr)\bigr]
\]
is niet identiek nul: een factor verdwijnt identiek alleen als
$\sigma_i$ en $\sigma_j$ zowel op $\alpha$ als op $\beta$
samenvallen, en dus op $L$ --- uitgesloten voor $i \neq j$. Omdat
$K$ oneindig is, kies $c \in K$ met $D(c) \ne 0$: dan zijn de $n$
elementen $\sigma_i(\alpha + c\beta)$ paarsgewijs verschillend,
zodat $\gamma = \alpha + c\beta$ minstens $n$ verschillende
toegevoegden in $\Omega$ heeft, dat wil zeggen $\deg \pi_\gamma
\geq n$: en $[K(\gamma):K] \geq n = [L:K]$ dwingt $L = K(\gamma)$
af.
\end{proof}

\section{De Galoiscorrespondentie}

\begin{definition}\label{def:b3:galois:galois}
Een eindige uitbreiding $L/K$ heet
\emph{Galois}\index{Galoisuitbreiding} als ze het
\omterm{thm:b3:galois:splitting}{splitsingslichaam} is van een \emph{separabele} veelterm over $K$.
Haar \emph{Galoisgroep}\index{Galoisgroep} is
$\operatorname{Gal}(L/K) = \operatorname{Aut}_K(L)$, de groep
lichaamsautomorfismen van $L$ die $K$ puntsgewijs vasthouden.
\end{definition}

\begin{proposition}\label{prop:b3:galois:galoisorder}
Is $L/K$ \omterm{def:b3:galois:galois}{Galois}, dan is $\abs{\operatorname{Gal}(L/K)} = [L:K]$;
bovendien is $L/F$ \omterm{def:b3:galois:galois}{Galois} voor elk tussenliggend lichaam $K
\subseteq F \subseteq L$, en heeft elke $F$-inbedding $L \to
\Omega \supseteq L$ als beeld $L$ (\emph{normaliteit}).
\end{proposition}

\begin{proof}
Zij $L$ het \omterm{thm:b3:galois:splitting}{splitsingslichaam} van de separabele $P$ over $K$, en
leg een \omterm{def:b3:galois:closure}{algebraïsche afsluiting} $\Omega \supseteq L$ vast. $L/K$
is \omterm{def:b3:galois:separable}{separabel}: ze wordt voortgebracht door wortels van $P$; de
separabiliteit van elk element volgt uit het gelijkheidsgeval
hieronder, maar laten we rechtstreeks redeneren ---
\cref{prop:b3:galois:embeddings}, toegepast op de voortbrengers
(wortels van de separabele $P$, waarvan de \omterm{def:b3:galois:algebraic}{minimaalveeltermen} $P$
delen), levert precies $[L:K]$ voortzettingen van $K
\hookrightarrow \Omega$ (in de inductiestap deelt de
\omterm{def:b3:galois:algebraic}{minimaalveelterm} van een wortel van $P$ over een tussenliggend
lichaam nog steeds $P$ en is ze dus \omterm{def:b3:galois:separable}{separabel}). Elke zulke
inbedding $\tau \colon L \to \Omega$ permuteert de wortels van $P$
($\tau$ houdt de coëfficiënten vast), en $L$ wordt door die
wortels voortgebracht: dus $\tau(L) = L$. Inbeddingen zijn
bijgevolg automorfismen: $\abs{\operatorname{Gal}(L/K)} = [L:K]$.
Voor een tussenliggend $F$: $L$ is ook het \omterm{thm:b3:galois:splitting}{splitsingslichaam} van
$P$ over $F$, en $P$ blijft \omterm{def:b3:galois:separable}{separabel}, dus is $L/F$ \omterm{def:b3:galois:galois}{Galois};
hetzelfde argument geeft de normaliteit over $F$.
\end{proof}

\begin{lemma}[Artin]\label{lem:b3:galois:artin}
Zij $G$ een eindige groep automorfismen van een lichaam $L$ en $K
= L^G = \{x : \sigma(x) = x\ \forall\sigma \in G\}$ haar vaste
lichaam. Dan is $[L : L^G] \leq \abs G$.
\end{lemma}

\begin{proof}
Zij $n = \abs G$ en $G = \{\sigma_1, \dots, \sigma_n\}$, en stel
dat $x_1, \dots, x_{n+1} \in L$ lineair onafhankelijk zijn over
$K$. Het homogene lineaire stelsel van $n$ vergelijkingen in $n+1$
onbekenden $(c_j)$ over $L$,
\[
\sum_{j=1}^{n+1} c_j\,\sigma_i(x_j) = 0
\qquad (i = 1, \dots, n),
\]
heeft een oplossing $\neq 0$; kies er een met de \emph{minste}
ingangen $\neq 0$, zeg $c_1, \dots, c_r \neq 0$ (na hernummering),
met $r \geq 2$ (één enkele $c_j\sigma_i(x_j) = 0$ is onmogelijk)
en genormaliseerd tot $c_r = 1$. Niet alle $c_j$ liggen in $K$:
de vergelijking voor $\sigma_i = \mathrm{id}$ zou de
onafhankelijkheid tegenspreken; zeg $c_1 \notin K$, zodat
$\tau(c_1) \ne c_1$ voor zekere $\tau \in G$. Pas $\tau$ toe op
alle vergelijkingen: omdat $\tau\sigma_i$ heel $G$ doorloopt, is
de vector $(\tau(c_j))_j$ opnieuw een oplossing; door af te
trekken is $(c_j - \tau(c_j))_j$ een oplossing met minder
ingangen $\neq 0$ (de $r$-de ingang $1 - 1 = 0$ verdwijnt, de
eerste niet) en niet nul: tegenspraak. Elke $n+1$ elementen zijn
dus afhankelijk: $[L:K] \leq n$.
\end{proof}

\begin{theorem}[Hoofdstelling van de Galoistheorie]
\label{thm:b3:galois:fundamental}
Zij $L/K$ een \omterm{def:b3:galois:galois}{Galoisuitbreiding} met groep $G =
\operatorname{Gal}(L/K)$.\index{Galoiscorrespondentie}
\begin{enumerate}
  \item $L^G = K$.
  \item De afbeeldingen $H \mapsto L^H$ en $F \mapsto
        \operatorname{Gal}(L/F)$ zijn elkaars inverse,
        inclusieomkerende bijecties tussen de deelgroepen van $G$
        en de tussenliggende lichamen $K \subseteq F \subseteq L$;
        bovendien is $[L : L^H] = \abs H$ en $[L^H : K] = [G :
        H]$.
  \item $H \trianglelefteq G$ dan en slechts dan als $L^H/K$
        \omterm{def:b3:galois:galois}{Galois} is, en dan induceert beperking $\operatorname{Gal}
        (L^H/K) \cong G/H$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Uiteraard is $K \subseteq L^G$. Zij omgekeerd $\alpha \in L
\setminus K$; we geven een $\sigma \in G$ met $\sigma(\alpha) \ne
\alpha$. De \omterm{def:b3:galois:algebraic}{minimaalveelterm} $\pi_\alpha$ over $K$ heeft \omterm{def:b3:galois:extension}{graad}
$\geq 2$ en is \omterm{def:b3:galois:separable}{separabel} ($L/K$ is \omterm{def:b3:galois:separable}{separabel},
\cref{prop:b3:galois:galoisorder}), dus heeft ze in een
\omterm{def:b3:galois:closure}{algebraïsche afsluiting} $\Omega \supseteq L$ nog een wortel $\beta
\neq \alpha$. Zet de $K$-inbedding $K(\alpha) \to \Omega$, $\alpha
\mapsto \beta$, voort tot een inbedding $\tau\colon L \to \Omega$
(\cref{prop:b3:galois:embeddings}); wegens de normaliteit
(\cref{prop:b3:galois:galoisorder}) is $\tau(L) = L$, dus $\tau
\in G$, $\beta = \tau(\alpha) \in L$ en $\tau(\alpha) \neq
\alpha$.

(2) Voor een deelgroep $H$: $L/L^H$ is \omterm{def:b3:galois:galois}{Galois}
(\cref{prop:b3:galois:galoisorder}), en $\operatorname{Gal}(L/L^H)
\supseteq H$ is triviaal, dus $[L:L^H] =
\abs{\operatorname{Gal}(L/L^H)} \geq \abs H$; het lemma van Artin
geeft $[L:L^H] \leq \abs H$: dus gelijkheid, en
$\operatorname{Gal}(L/L^H) = H$. Voor een tussenliggend lichaam
$F$ geeft ``$L/F$ \omterm{def:b3:galois:galois}{Galois}'' dat $L^{\operatorname{Gal}(L/F)} = F$,
via (1) toegepast op $L/F$. De twee afbeeldingen zijn dus elkaars
inverse; dat ze inclusies omkeren is evident. Graden: $[L:L^H] =
\abs H$ is zojuist bewezen, en $[L^H:K] = [L:K]/[L:L^H] = \abs
G/\abs H$.

(3) Voor $\sigma \in G$ en $H \leq G$ is $\sigma(L^H) = L^{\sigma
H\sigma^{-1}}$ (rechtstreeks na te gaan). Wegens de bijectie geldt
$\sigma(L^H) = L^H$ voor alle $\sigma$ dan en slechts dan als $H
\trianglelefteq G$. Is nu $H \trianglelefteq G$, stel dan $F =
L^H$: elke $\sigma \in G$ beperkt zich tot een automorfisme van
$F$, wat een morfisme $\rho \colon G \to \operatorname{Aut}_K(F)$
geeft met kern $\{\sigma : \sigma\restriction_F = \mathrm{id}\} =
\operatorname{Gal}(L/F) = H$. Dus bedt $G/H$ zich in
$\operatorname{Aut}_K(F)$ in, waaruit
$\abs{\operatorname{Aut}_K(F)} \geq [G:H] = [F:K]$; de omgekeerde
ongelijkheid geldt altijd (\cref{prop:b3:galois:embeddings}): dus
$\abs{\operatorname{Aut}_K(F)} = [F:K]$ en is $\rho$ surjectief.
Rest nog dat $F/K$ \omterm{def:b3:galois:galois}{Galois} is: $F$ is \omterm{def:b3:galois:separable}{separabel} over $K$ (het ligt
binnen de separabele $L/K$) en $F = K(\gamma)$
(\cref{thm:b3:galois:primitive}); de veelterm $\prod_{\sigma \in
G/H}\bigl(X - \sigma(\gamma)\bigr)$ (product over de verschillende
beelden, die in $F$ liggen: $\sigma(F) = L^{\sigma H\sigma^{-1}} =
L^H = F$ wegens de normaliteit van $H$) heeft coëfficiënten die
door $G$ worden vastgehouden en dus volgens (1) in $K$ liggen:
het is een \omterm{def:b3:galois:separable}{separabele veelterm} van $K[X]$ die door $F$ wordt
gesplitst, en haar wortels brengen $F$ voort: $F/K$ is \omterm{def:b3:galois:galois}{Galois}. Is
omgekeerd $F = L^H$ met $F/K$ \omterm{def:b3:galois:galois}{Galois}, dan geeft de normaliteit van
$F$ (\cref{prop:b3:galois:galoisorder}, toegepast op inbeddingen
$F \to \Omega$ die van elementen van $G$ afkomstig zijn) dat
$\sigma(F) = F$ voor alle $\sigma \in G$, dat wil zeggen $H
\trianglelefteq G$.
\end{proof}

\section{Cyclotomische uitbreidingen}

\begin{definition}\label{def:b3:galois:cyclotomic}
Zij $n \geq 1$ en $\zeta_n = \eu^{2\iu\pi/n}$. De $n$-de
\emph{cyclotomische veelterm}\index{cyclotomische veelterm} is
$\Phi_n = \prod_{\gcd(k,n)=1,\ 1 \le k \le n} \bigl(X -
\zeta_n^k\bigr)$, van \omterm{def:b3:galois:extension}{graad} $\varphi(n)$; groeperen we de wortels
van $X^n - 1$ naar hun exacte orde, dan is $X^n - 1 = \prod_{d
\mid n}\Phi_d$, wat inductief laat zien dat $\Phi_n \in \Z[X]$
(euclidische deling van monische gehele veeltermen).
\end{definition}

\begin{theorem}\label{thm:b3:galois:cyclotomicirred}
$\Phi_n$ is \omterm{def:b3:rings:divisibility}{irreducibel} over $\Q$; bijgevolg is $[\Q(\zeta_n) :
\Q] = \varphi(n)$ en
\[
\operatorname{Gal}\bigl(\Q(\zeta_n)/\Q\bigr) \;\cong\;
(\Z/n\Z)^\times,
\qquad \sigma_a(\zeta_n) = \zeta_n^a .
\]
De uitbreiding $\Q(\zeta_n)/\Q$ is dus \omterm{def:b3:galois:galois}{Galois} met een
\emph{abelse} groep.
\end{theorem}

\begin{proof}
Zij $f = \pi_{\zeta_n}$, zodat $\Phi_n = fg$ met $f, g \in \Z[X]$
monisch (het lemma van Gauss, \cref{lem:b3:rings:gauss}: de
inhouden vermenigvuldigen, alle veeltermen zijn monisch).
\emph{Bewering: is $\zeta$ een wortel van $f$ en $p \nmid n$
priem, dan is $\zeta^p$ een wortel van $f$.} Anders is $\zeta^p$
een wortel van $g$ (het is een primitieve $n$-de eenheidswortel),
dus is $\zeta$ een wortel van $g(X^p)$ en $f \mid g(X^p)$ in
$\Z[X]$ (\omterm{def:b3:galois:algebraic}{minimaalveelterm}, en dan opnieuw Gauss). Reduceer modulo
$p$: $\bar g(X^p) = \bar g(X)^p$ (\omterm{thm:b3:galois:finitefields}{Frobenius} op $\mathbb F_p[X]$:
per coëfficiënt is $a^p = a$, plus de droom van de eerstejaars),
dus $\bar f \mid \bar g^{\,p}$: $\bar f$ en $\bar g$ delen een
irreducibele factor, en $\bar\Phi_n = \bar f\bar g$ heeft dus een
meervoudige factor. Dan geldt dat ook voor $X^n - \bar 1$; maar de
afgeleide $\bar nX^{n-1}$ is er onderling ondeelbaar mee ($p \nmid
n$, en $0$ is geen wortel): tegenspraak.

Elke primitieve wortel $\zeta_n^k$ (met $\gcd(k, n) = 1$) wordt
uit $\zeta_n$ verkregen door achtereenvolgens priemmachten te
nemen die $n$ niet delen (ontbind $k$): de bewering plant zich
voort, dus is elke primitieve wortel een wortel van $f$, waaruit
$f = \Phi_n$, \omterm{def:b3:rings:divisibility}{irreducibel}. Bijgevolg is $[\Q(\zeta_n):\Q] =
\varphi(n)$, en is $\Q(\zeta_n)$ het \omterm{thm:b3:galois:splitting}{splitsingslichaam} van de
separabele $X^n - 1$ (alle wortels zijn machten van $\zeta_n$):
\omterm{def:b3:galois:galois}{Galois}. Een automorfisme $\sigma$ stuurt $\zeta_n$ naar een andere
primitieve wortel $\zeta_n^{a(\sigma)}$, en $\sigma \mapsto
a(\sigma)$ is een injectief morfisme naar $(\Z/n\Z)^\times$; beide
groepen hebben orde $\varphi(n)$: dus een isomorfisme.
\end{proof}

\section{Liniaal en passer}

\begin{definition}\label{def:b3:galois:constructible}
Vereenzelvig het vlak met $\C$ en begin met $\{0, 1\}$. Een punt
heet \emph{construeerbaar}\index{construeerbaar getal} als het te
verkrijgen is door eindig veel snijdingen van rechten door twee
reeds geconstrueerde punten en cirkels met een geconstrueerd punt
als middelpunt en als straal de afstand tussen twee
geconstrueerde punten.
\end{definition}

\begin{theorem}[Wantzel]\label{thm:b3:galois:wantzel}
Een $z \in \C$ is \omterm{def:b3:galois:constructible}{construeerbaar} dan en slechts dan als er een
toren $\Q = F_0 \subseteq F_1 \subseteq \dots \subseteq F_r$
bestaat met $[F_{i+1} : F_i] = 2$ en $z \in F_r$. In het bijzonder
is een \omterm{def:b3:galois:constructible}{construeerbaar getal} \omterm{def:b3:galois:algebraic}{algebraïsch} van een \omterm{def:b3:galois:extension}{graad} die een
macht van $2$ is over $\Q$.\index{stelling van Wantzel}
\end{theorem}

\begin{proof}
($\Rightarrow$) De coördinaten van het snijpunt van twee rechten
door punten met coördinaten in een deellichaam $F \subseteq \R$
lossen een lineair stelsel over $F$ op: ze blijven in $F$.
Snijdingen rechte--cirkel en cirkel--cirkel leiden, na eliminatie
van het lineaire deel (de twee cirkelvergelijkingen van elkaar
aftrekken geeft een rechte), tot een kwadratische vergelijking
over $F$: de nieuwe coördinaten liggen in $F$ of in $F(\sqrt d)$
voor zekere $d \in F$ met $d > 0$. Per inductie heeft elk
geconstrueerd punt coördinaten in een toren van kwadratische
uitbreidingen van $\Q$; en ook $z = x + \iu y$ ligt in zo'n toren
(voeg $\iu$ toe: nog één kwadratische stap). Het gevolg voor de
\omterm{def:b3:galois:extension}{graad}: $[\Q(z):\Q]$ deelt $[F_r : \Q] = 2^r$ (\omterm{thm:b3:galois:tower}{torenwet}).

($\Leftarrow$) De construeerbare getallen vormen een lichaam:
sommen en verschillen met parallellogrammen (parallellen zijn
\omterm{def:b3:galois:constructible}{construeerbaar}: laat tweemaal een loodlijn neer en richt ze weer
op --- de klassieke loodlijn door een punt gebruikt één cirkel en
twee bogen); producten en quotiënten met de configuraties van
Thales (construeer bij gegeven lengten $a, b$ de lengten $ab$ en
$a/b$ met gelijkvormige driehoeken op twee halfrechten). En het
lichaam is gesloten onder worteltrekking: voor $a > 0$ snijden de
cirkel met diameter $1 + a$ en de loodlijn in het aansluitpunt
elkaar op hoogte $\sqrt a$ (de hoogtestelling in een rechthoekige
driehoek); en voor een complexe $w = \rho\eu^{\iu\theta}$
construeer je $\sqrt\rho$ en deel je $\theta$ in tweeën (een hoek
halveren is een passerconstructie). Reële en imaginaire delen van
elementen van een kwadratische toren zijn dus per inductie op de
toren \omterm{def:b3:galois:constructible}{construeerbaar}: elke stap voegt wortels van een kwadratische
vergelijking toe, uit te drukken met lichaamsbewerkingen en één
vierkantswortel van een reeds geconstrueerd getal (de abc-formule;
in karakteristiek $0$).
\end{proof}

\begin{corollary}\label{cor:b3:galois:impossible}
De drie klassieke problemen zijn niet \omterm{def:b3:groups:derived}{oplosbaar} met liniaal en
passer:\index{liniaal en passer}
\begin{enumerate}
  \item \emph{Verdubbeling van de kubus}: $\sqrt[3]2$ heeft \omterm{def:b3:galois:extension}{graad}
        $3$, geen macht van $2$.
  \item \emph{Driedeling van de hoek}: een hoek van $60^\circ$ in
        drieën delen vergt $\cos 20^\circ$, een wortel van de
        irreducibele $8X^3 - 6X - 1$: \omterm{def:b3:galois:extension}{graad} $3$.
  \item \emph{Kwadratuur van de cirkel}: $\sqrt\pi$ is
        transcendent (want $\pi$ is het --- de stelling van
        Lindemann, hier zonder bewijs aangenomen: haar bewijs
        hoort in een cursus transcendentietheorie).
\end{enumerate}
Verder is de regelmatige $n$-hoek \omterm{def:b3:galois:constructible}{construeerbaar} dan en slechts
dan als $\varphi(n)$ een macht van $2$ is (Gauss--Wantzel; het
``als'' gebruikt de methode van de weekendopgave, het ``slechts
dan'' is \cref{thm:b3:galois:wantzel} toegepast op $\zeta_n$, van
\omterm{def:b3:galois:extension}{graad} $\varphi(n)$). Voor $n = 7$ is $\varphi(7) = 6$: de
regelmatige zevenhoek is onmogelijk; voor $n = 17$ is $\varphi(17)
= 16 = 2^4$: \omterm{def:b3:galois:constructible}{construeerbaar} --- en de weekendopgave construeert
haar.
\end{corollary}

\begin{proof}
(1) $X^3 - 2$ is \omterm{def:b3:rings:divisibility}{irreducibel} (\omterm{thm:b3:rings:criteria}{Eisenstein}). (2) Uit $\cos 3\theta =
4\cos^3\theta - 3\cos\theta$ met $3\theta = 60^\circ$ volgt $8c^3
- 6c = 1$ voor $c = \cos 20^\circ$; de derdegraadsveelterm $8X^3 -
6X - 1$ heeft geen rationale wortel (de kandidaten $\pm1,
\pm\frac 12, \pm\frac14, \pm\frac18$ falen) en is dus
\omterm{def:b3:rings:divisibility}{irreducibel}: \omterm{def:b3:galois:extension}{graad} $3$. Een hoek van $60^\circ$ is in het algemeen
\omterm{def:b3:galois:constructible}{construeerbaar}, dus zou een driedeler $c$ construeren. (3) Was
$\sqrt\pi$ \omterm{def:b3:galois:constructible}{construeerbaar}, dan was het \omterm{def:b3:galois:algebraic}{algebraïsch}, en dus ook
$\pi$. De uitspraak over de $n$-hoek: de \omterm{def:b3:galois:extension}{graad} van $\zeta_n$ is
$\varphi(n)$ (\cref{thm:b3:galois:cyclotomicirred}); de
noodzakelijkheid volgt uit Wantzel; en voor de toereikendheid
bezit de \omterm{def:b3:galois:galois}{Galoisgroep}, abels van orde $2^m$, een keten deelgroepen
van index $2$ (een eindige $2$-groep heeft die:
\cref{exo:b3:groups:10}), waarvan de vaste lichamen een
kwadratische toren vormen die eindigt in $\Q(\zeta_n)$
(\cref{thm:b3:galois:fundamental}); besluit met
\cref{thm:b3:galois:wantzel}.
\end{proof}

\section{Oplosbaarheid door worteltrekking}

\begin{definition}\label{def:b3:galois:radical}
Een uitbreiding $L/K$ (in deze paragraaf overal in karakteristiek
$0$) heet \emph{radicaal} als er een toren $K = F_0 \subseteq
\dots \subseteq F_r = L$ is met $F_{i+1} = F_i(\alpha_i)$ en
$\alpha_i^{n_i} \in F_i$: elke stap voegt een $n_i$-de wortel toe.
Een veelterm $P \in K[X]$ heet \emph{oplosbaar door
worteltrekking}\index{oplosbaar door worteltrekking} als haar
\omterm{thm:b3:galois:splitting}{splitsingslichaam} in een radicale uitbreiding van $K$ ligt.
\end{definition}

\begin{lemma}\label{lem:b3:galois:cyclicsteps}
Bevat $K$ een primitieve $n$-de eenheidswortel $\zeta$, dat wil
zeggen een $\zeta$ van orde $n$ in $K^\times$, en is $a \in
K^\times$, dan is $K(\sqrt[n]a)/K$ \omterm{def:b3:galois:galois}{Galois} met een \emph{cyclische}
groep. Omgekeerd --- hieronder niet nodig --- is elke cyclische
uitbreiding van \omterm{def:b3:galois:extension}{graad} $n$ van deze vorm. Bovendien is
$K(\zeta_n)/K$ \omterm{def:b3:galois:galois}{Galois} met een \emph{abelse} groep, voor elk $K$
van karakteristiek $0$.
\end{lemma}

\begin{proof}
$X^n - a$ is \omterm{def:b3:galois:separable}{separabel} (\omterm{lem:b3:rings:bezout}{grootste gemene deler} met $nX^{n-1}$: $a
\neq 0$) en valt uiteen in $K(\alpha)$ met $\alpha^n = a$: haar
wortels zijn de $\zeta^k\alpha \in K(\alpha)$. Dus is
$K(\alpha)/K$ \omterm{def:b3:galois:galois}{Galois}; en de afbeelding $\sigma \mapsto
\sigma(\alpha)/\alpha \in \mu_n = \langle \zeta\rangle$ is een
injectief morfisme ($\sigma\tau(\alpha) = \sigma(\tau(\alpha)/
\alpha \cdot \alpha) = \tau(\alpha)/\alpha\cdot\sigma(\alpha)$,
want het quotiënt ligt in $K$) naar een cyclische groep: dus is
$\operatorname{Gal}$ cyclisch. De omkering is Kummertheorie, die
we niet nodig hebben (zie de opmerking hieronder). Voor
$K(\zeta_n)$: het splitst de separabele $X^n - 1$, en $\sigma
\mapsto a(\sigma)$ met $\sigma(\zeta_n) = \zeta_n^{a(\sigma)}$
bedt de groep in het abelse $(\Z/n\Z)^\times$ in, net als in
\cref{thm:b3:galois:cyclotomicirred} (voor de injectiviteit is
alleen nodig dat $\zeta_n$ de betrokken eenheidswortels
voortbrengt).
\end{proof}

\begin{theorem}[Galois]\label{thm:b3:galois:solvable}
Zij $K$ van karakteristiek $0$ en $P \in K[X]$ met
\omterm{thm:b3:galois:splitting}{splitsingslichaam} $L$. Is $P$ \omterm{def:b3:galois:radical}{oplosbaar door worteltrekking}, dan
is $\operatorname{Gal}(L/K)$ een \omterm{def:b3:groups:derived}{oplosbare groep}. (Het omgekeerde
geldt eveneens; we hebben het niet nodig.)\index{oplosbaarheidsstelling
van Galois}
\end{theorem}

\begin{proof}
Stap 1: \emph{vergroot de \omterm{def:b3:galois:radical}{radicale} toren tot een Galoistoren.} Zij
$L \subseteq M$ met $M/K$ \omterm{def:b3:galois:radical}{radicaal}, met wortelexponenten $n_1,
\dots, n_r$ en $n = n_1\cdots n_r$. Voeg eerst $\zeta_n$ toe: de
toren $K \subseteq K(\zeta_n) \subseteq M(\zeta_n)$ is nog steeds
\omterm{def:b3:galois:radical}{radicaal} ($\zeta_n$ is een eenheidswortel: een \omterm{def:b3:galois:radical}{radicale} stap, want
$\zeta_n^n = 1$), en haar stappen voorbij de eerste spelen zich af
boven lichamen die de benodigde eenheidswortels bevatten. Vervang
vervolgens $M(\zeta_n)$ door het samenstel $N$ van alle
$\sigma(M(\zeta_n))$, waarbij $\sigma$ de (eindig veel)
$K$-inbeddingen van $M(\zeta_n)$ in een vaste \omterm{def:b3:galois:closure}{algebraïsche
afsluiting} doorloopt: $N$ is het \omterm{thm:b3:galois:splitting}{splitsingslichaam} van het product
van de \omterm{def:b3:galois:algebraic}{minimaalveeltermen} van een voortbrengende verzameling (in
karakteristiek $0$: eindig en \omterm{def:b3:galois:separable}{separabel}), dus is $N/K$ \omterm{def:b3:galois:galois}{Galois}; en
$N$ is \omterm{def:b3:galois:radical}{radicaal} over $K$: elke $\sigma(M(\zeta_n))$ is \omterm{def:b3:galois:radical}{radicaal}
over $K$ (pas $\sigma$ toe op een \omterm{def:b3:galois:radical}{radicale} toren), en een
samenstel van \omterm{def:b3:galois:radical}{radicale} uitbreidingen is \omterm{def:b3:galois:radical}{radicaal} (zet de torens
achter elkaar: is $F'/K$ \omterm{def:b3:galois:radical}{radicaal} met een toren die $\beta_j$
toevoegt, dan blijven stappen van het type $F''(\beta_j)$ \omterm{def:b3:galois:radical}{radicaal}
boven elke grotere basis).

Stap 2: \emph{lees de \omterm{def:b3:groups:derived}{oplosbaarheid} van de Galoistoren af.} Neem
dus aan dat $L \subseteq N$ met $N/K$ \omterm{def:b3:galois:galois}{Galois} en \omterm{def:b3:galois:radical}{radicaal}, met
toren $K \subseteq K(\zeta_n) = E_0 \subseteq E_1 \subseteq \dots
\subseteq E_s = N$, waarbij elke $E_{i+1} =
E_i(\sqrt[n_i]{a_i})$ met $\zeta_{n_i} \in E_0 \subseteq E_i$. Zij
$G = \operatorname{Gal}(N/K)$ en $G_i = \operatorname{Gal}(N/E_i)$:
een dalende keten $G \supseteq G_0 \supseteq G_1 \supseteq \dots
\supseteq G_s = \{e\}$. Elke $E_{i+1}/E_i$ is \omterm{def:b3:galois:galois}{Galois} met
cyclische groep (\cref{lem:b3:galois:cyclicsteps}), dus geeft de
hoofdstelling, toegepast op de \omterm{def:b3:galois:galois}{Galoisuitbreiding} $N/E_i$
(\cref{thm:b3:galois:fundamental}(3), met omgevende groep $G_i$):
$G_{i+1} \trianglelefteq G_i$ met $G_i/G_{i+1} \cong
\operatorname{Gal}(E_{i+1}/E_i)$ cyclisch. Net zo is $E_0/K$
\omterm{def:b3:galois:galois}{Galois} met abelse groep $G/G_0$
(\cref{lem:b3:galois:cyclicsteps}). Die keten toont $G$ als
\omterm{def:b3:groups:derived}{oplosbaar} (\cref{prop:b3:groups:derived}). Ten slotte is
$\operatorname{Gal}(L/K)$ een \emph{quotiënt} van $G$: $L/K$ is
\omterm{def:b3:galois:galois}{Galois} ($P$ is \omterm{def:b3:galois:separable}{separabel} in karakteristiek $0$) en de beperking $G
\to \operatorname{Gal}(L/K)$ is surjectief
(\cref{thm:b3:galois:fundamental}(3) met $H =
\operatorname{Gal}(N/L)$); en quotiënten van \omterm{def:b3:groups:derived}{oplosbare groepen}
zijn \omterm{def:b3:groups:derived}{oplosbaar}.
\end{proof}

\begin{corollary}[Onoplosbaarheid van de vijfdegraadsvergelijking]
\label{cor:b3:galois:quintic}
Er zijn veeltermen van \omterm{def:b3:galois:extension}{graad} $5$ over $\Q$ die niet \omterm{def:b3:groups:derived}{oplosbaar} zijn
door worteltrekking: bijvoorbeeld $X^5 - 4X + 2$, waarvan de
\omterm{def:b3:galois:galois}{Galoisgroep} $S_5$ is (\cref{exo:b3:galois:11}), een niet-oplosbare
groep (\cref{cor:b3:groups:snnotsolvable}). Voor \omterm{def:b3:galois:extension}{graad} $\geq 5$
kan er dus geen algemene formule met worteltrekkingen
bestaan.\index{vijfdegraadsvergelijking, onoplosbaarheid}
\end{corollary}

\begin{remark}\label{rem:b3:galois:converse}
Het omgekeerde van \cref{thm:b3:galois:solvable} --- een oplosbare
\omterm{def:b3:galois:galois}{Galoisgroep} impliceert \omterm{def:b3:groups:derived}{oplosbaarheid} door worteltrekking ---
bewijst men door de afgeleide reeks af te dalen en aan te tonen
dat elke cyclische uitbreiding (met genoeg eenheidswortels)
\omterm{def:b3:galois:radical}{radicaal} is, met behulp van de \emph{resolventen van Lagrange};
het verklaart waarom de graden $2, 3, 4$ wél formules hebben:
$S_2, S_3, S_4$ zijn \omterm{def:b3:groups:derived}{oplosbaar}
(\cref{ex:b3:groups:solvableexamples}). Wij nemen het op dit
niveau zonder bewijs aan; een volledige behandeling hoort in een
mastercursus, maar \cref{exo:b3:galois:8} maakt het concreet voor
de derde \omterm{def:b3:galois:extension}{graad}.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b3:galois:1}
Toon aan dat $[\Q(\sqrt2, \sqrt3):\Q] = 4$ en dat $\Q(\sqrt2 +
\sqrt3) = \Q(\sqrt2, \sqrt3)$, en bereken de \omterm{def:b3:galois:algebraic}{minimaalveelterm} van
$\sqrt2 + \sqrt3$ over $\Q$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:galois:2}
Zij $\alpha = \sqrt[3]2$. Toon aan dat $\Q(\alpha)/\Q$ niet
normaal is (geef een inbedding $\Q(\alpha) \to \C$ waarvan het
beeld niet $\Q(\alpha)$ is), bepaal het \omterm{thm:b3:galois:splitting}{splitsingslichaam} $L$ van
$X^3 - 2$ en $[L:\Q]$, en ga na dat
$\operatorname{Aut}_\Q(\Q(\alpha)) = \{\mathrm{id}\}$: bij
\omterm{def:b3:galois:galois}{niet-Galoisuitbreidingen} kan de automorfismegroep veel kleiner
zijn dan de \omterm{def:b3:galois:extension}{graad}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:galois:3}
Construeer $\mathbb F_9$ als $\mathbb F_3[X]/(X^2+1)$ en vind een
voortbrenger van $\mathbb F_9^\times$. Geef de monische
irreducibele veeltermen van \omterm{def:b3:galois:extension}{graad} $1, 2, 3$ over $\mathbb F_2$, en
ga na dat $X^8 - X = X(X+1)(X^3+X+1)(X^3+X^2+1)$ over $\mathbb
F_2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:4}
(a) Bepaal alle primitieve wortels modulo $7$ en modulo $11$ (dat
wil zeggen de voortbrengers van $\mathbb F_7^\times$ en $\mathbb
F_{11}^\times$).
(b) Toon aan dat voor oneven $p$ een $x \in \mathbb F_p^\times$
een kwadraat is dan en slechts dan als $x^{(p-1)/2} = 1$ (het
criterium van Euler), en vind zo het criterium voor $-1$ uit
\cref{pb:b3:rings:1} terug.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:5}
Toon aan dat binnen een vaste \omterm{def:b3:galois:closure}{algebraïsche afsluiting}
$\bar{\mathbb F}_p$ geldt: $\mathbb F_{p^m} \cap \mathbb F_{p^n} =
\mathbb F_{p^{\gcd(m,n)}}$ en $\mathbb F_{p^m}\mathbb F_{p^n} =
\mathbb F_{p^{\operatorname{lcm}(m,n)}}$, en beschrijf
$\operatorname{Gal}(\mathbb F_{p^n}/\mathbb F_{p^m})$ voor $m \mid
n$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:6}
Zij $I_d(q)$ het aantal monische irreducibele veeltermen van \omterm{def:b3:galois:extension}{graad}
$d$ over $\mathbb F_q$. Bewijs
\[
X^{q^n} - X \;=\; \prod_{d \mid n}\ \prod_{P \text{ irred.
monisch, } \deg P = d} P ,
\qquad\text{en dus}\qquad
q^n = \sum_{d \mid n} d\, I_d(q).
\]
Leid $I_1, I_2, I_3, I_4$ expliciet af, en toon aan dat $I_d(q)
\geq 1$ voor elke $d$ (zodat de uitbreidingen $\mathbb
F_{q^d}/\mathbb F_q$ voor alle $d$ bestaan als quotiënten $\mathbb
F_q[X]/(P)$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:7}
Bepaal $\operatorname{Gal}(\Q(\sqrt2,\sqrt3)/\Q)$ en het volledige
tralie van tussenliggende lichamen. Dezelfde vraag voor het
\omterm{thm:b3:galois:splitting}{splitsingslichaam} van $(X^2-2)(X^2-3)(X^2-6)$ --- wat valt je op?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:8}
(De derdegraadsvergelijking, opgelost door haar groep) Zij $P =
X^3 + pX + q \in \Q[X]$ \omterm{def:b3:rings:divisibility}{irreducibel} met wortels $x_1, x_2, x_3$ en
\omterm{thm:b3:galois:splitting}{splitsingslichaam} $L$. Stel $\delta = (x_1 - x_2)(x_1 - x_3)(x_2 -
x_3)$ en $\Delta = \delta^2 = -4p^3 - 27q^2$ (neem die klassieke
identiteit zonder bewijs aan of ga ze na door de symmetrische
functies uit te werken).
(a) Toon aan dat $\operatorname{Gal}(L/\Q) \cong A_3$ of $S_3$,
naargelang $\Delta$ wel of niet een kwadraat in $\Q$ is.
(b) Definieer met $j = \zeta_3$ de resolventen van Lagrange $u =
x_1 + jx_2 + j^2x_3$ en $v = x_1 + j^2x_2 + jx_3$. Toon aan dat
$u^3 + v^3 = -27q$ en $uv = -3p$, en los $u^3$ en $v^3$ op: de
formules van Cardano rollen eruit. Waar is de \omterm{def:b3:groups:derived}{oplosbaarheid} van
$S_3$ gebruikt?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:9}
Toon in $\Q(\zeta_5)$ aan dat het enige kwadratische deellichaam
$\Q(\sqrt5)$ is, via de Gauss-sommen $\eta_0 = \zeta_5 +
\zeta_5^4$ en $\eta_1 = \zeta_5^2 + \zeta_5^3$: bereken $\eta_0 +
\eta_1$ en $\eta_0\eta_1$, en leid af dat $\cos\frac{2\pi}5 =
\frac{\sqrt5 - 1}4$. Besluit dat de regelmatige vijfhoek
\omterm{def:b3:galois:constructible}{construeerbaar} is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:galois:10}
Zij $K = \mathbb F_p(S, T)$ (rationale functies in twee
onbepaalden) en $L = K(S^{1/p}, T^{1/p})$.
(a) Toon aan dat $[L:K] = p^2$ en dat $\alpha^p \in K$ voor elke
$\alpha \in L$.
(b) Leid af dat $L/K$ \emph{niet} enkelvoudig is: er bestaat geen
primitief element --- inseparabiliteit is fataal voor
\cref{thm:b3:galois:primitive}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:galois:11}
Zij $P = X^5 - 4X + 2$ en $G$ haar \omterm{def:b3:galois:galois}{Galoisgroep} over $\Q$, werkend
op de $5$ wortels.
(a) Toon aan dat $P$ \omterm{def:b3:rings:divisibility}{irreducibel} is en leid af dat $5 \mid \abs
G$; besluit dat $G$ een $5$-cykel bevat (Cauchy,
\cref{thm:b3:groups:cauchy}).
(b) Toon, door het verloop van $x \mapsto x^5 - 4x + 2$ te
bestuderen, aan dat $P$ precies $3$ reële wortels heeft; leid af
dat de complexe toevoeging zich in $G$ tot een transpositie
beperkt.
(c) Toon aan dat een deelgroep van $S_5$ die een transpositie en
een $5$-cykel bevat, gelijk is aan $S_5$ \emph{(conjugeer de
transpositie met machten van de cykel)}. Besluit dat $G \cong S_5$
en, met \cref{thm:b3:galois:solvable}, dat $P$ niet \omterm{def:b3:groups:derived}{oplosbaar} is
door worteltrekking.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:galois:12}
(De diëdrale vierdegraadsvergelijking) Zij $\alpha = \sqrt[4]2$ en
$L = \Q(\alpha, \iu)$, het \omterm{thm:b3:galois:splitting}{splitsingslichaam} van $X^4 - 2$ over
$\Q$.
(a) Toon aan dat $[L : \Q] = 8$ en dat $G =
\operatorname{Gal}(L/\Q)$ wordt voortgebracht door
$\sigma\colon \alpha \mapsto \iu\alpha,\ \iu \mapsto \iu$ en de
complexe toevoeging $\tau$, met $\sigma^4 = \tau^2 = e$ en
$\tau\sigma\tau = \sigma^{-1}$: dus $G \cong D_4$.
(b) Geef het deelgroepentralie van $D_4$ (tien deelgroepen) en
koppel elke deelgroep aan haar vaste lichaam; ga in het bijzonder
na dat $\Q(\sqrt2)$, $\Q(\iu)$ en $\Q(\iu\sqrt2)$ de drie
kwadratische deellichamen zijn, en lokaliseer $\Q(\alpha)$,
$\Q(\iu\alpha)$ en $\Q(\sqrt2, \iu)$.
(c) Welke tussenliggende lichamen zijn \omterm{def:b3:galois:galois}{Galois} over $\Q$? Vergelijk
je antwoord met de \omterm{def:b3:groups:normal}{normaaldelers} van $D_4$, en verklaar waarom
$\Q(\alpha)/\Q$ faalt terwijl $\Q(\sqrt2)/\Q$ slaagt.
\end{exercise}

\section{Probleem: Gauss en de regelmatige 17-hoek}

\begin{problem}[{Weekendopgave --- construeerbaarheid van de
17-hoek}]\label{pb:b3:galois:1}
Op 30 maart 1796 toonde de negentienjarige Gauss aan dat de
regelmatige $17$-hoek \omterm{def:b3:galois:constructible}{construeerbaar} is --- de eerste vooruitgang
op die vraag sinds de oudheid. We doen zijn berekening over met
het gereedschap van dit hoofdstuk. Stel $\zeta =
\eu^{2\iu\pi/17}$, $L = \Q(\zeta)$ en $G =
\operatorname{Gal}(L/\Q)$.

\medskip
\noindent\textbf{Deel I --- De groep en haar filtratie.}
\begin{enumerate}
  \item Verantwoord: $[L:\Q] = 16$ en $G \cong (\Z/17\Z)^\times$,
        cyclisch van orde $16$. Ga na dat $3$ een voortbrenger van
        $(\Z/17\Z)^\times$ is \emph{(bereken de machten van $3$
        modulo $17$: $3, 9, 10, 13, 5, 15, 11, 16, \dots$)}.
  \item Zij $\sigma \in G$ met $\sigma(\zeta) = \zeta^3$, en $H_k
        = \langle \sigma^{2^k}\rangle$ voor $k = 0, \dots, 4$.
        Toon aan dat $G = H_0 \supset H_1 \supset H_2 \supset H_3
        \supset H_4 = \{e\}$ met telkens index $[H_k : H_{k+1}] =
        2$, en dat de vaste lichamen $\Q = L_0 \subset L_1 \subset
        L_2 \subset L_3 \subset L_4 = L$ een toren van
        kwadratische uitbreidingen vormen.
  \item Besluit \emph{a priori}, met
        \cref{thm:b3:galois:wantzel}, dat $\zeta$ --- en dus de
        $17$-hoek --- \omterm{def:b3:galois:constructible}{construeerbaar} is. De rest van het probleem
        maakt de toren expliciet.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De perioden van lengte 8.}
Definieer de \emph{Gauss-perioden}
\[
\eta_0 = \sum_{k \text{ even}} \zeta^{3^k \bmod 17}
= \zeta^{1} + \zeta^{9} + \zeta^{13} + \zeta^{15} + \zeta^{16} +
\zeta^{8} + \zeta^{4} + \zeta^{2},
\qquad
\eta_1 = \sum_{k \text{ oneven}} \zeta^{3^k \bmod 17}.
\]
\begin{enumerate}[resume]
  \item Toon aan dat $\eta_0$ en $\eta_1$ door $H_1$ worden
        vastgehouden en door $\sigma$ worden verwisseld; leid af
        dat $\eta_0, \eta_1 \in L_1$ en dat ze de twee wortels
        zijn van een kwadratische veelterm over $\Q$.
  \item Bereken $\eta_0 + \eta_1 = -1$. Toon aan dat
        $\eta_0\eta_1 = -4$ \emph{(elk product $\zeta^a\zeta^b$ is
        een zekere $\zeta^c$ met $c \neq 0$; tel hoe vaak elke $c$
        voorkomt, of redeneer dat het product een geheel getal is
        dat door $G$ wordt vastgehouden, gelijk aan de som over
        alle $64$ producten, en gebruik dat elke rest $\neq 0$
        even vaak optreedt)}.
  \item Leid af dat $\eta_0 = \frac{-1 + \sqrt{17}}2$ en $\eta_1 =
        \frac{-1-\sqrt{17}}2$ (bepaal numeriek welke welke is:
        $\eta_0 \approx 1.56$), en dat $L_1 = \Q(\sqrt{17})$.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Perioden van lengte 4 en 2.}
Definieer
\[
\beta_0 = \zeta + \zeta^{13} + \zeta^{16} + \zeta^{4},\quad
\beta_1 = \zeta^3 + \zeta^5 + \zeta^{14} + \zeta^{12},\quad
\beta_2 = \zeta^9 + \zeta^{15} + \zeta^{8} + \zeta^{2},\quad
\beta_3 = \zeta^{10} + \zeta^{11} + \zeta^{7} + \zeta^{6}.
\]
\begin{enumerate}[resume]
  \item Toon aan dat $\beta_0 + \beta_2 = \eta_0$ en $\beta_1 +
        \beta_3 = \eta_1$, en dat $\beta_0$ en $\beta_2$ door
        $H_2$ worden vastgehouden en door $\sigma^2$ worden
        verwisseld.
  \item Bereken $\beta_0\beta_2 = -1$ en $\beta_1\beta_3 = -1$
        \emph{(werk uit: de zestien verkregen exponenten
        doorlopen $1, \dots, 16$ precies één keer)}.
  \item Leid af dat $\beta_0 = \frac{\eta_0 + \sqrt{\eta_0^2 +
        4}}2$ (controleer het teken numeriek: $\beta_0 \approx
        2.05$) en de overeenkomstige formule voor $\beta_1$; dus
        $L_2 = \Q(\beta_0)$, kwadratisch over $L_1$.
  \item Zij $\gamma_0 = \zeta + \zeta^{16} =
        2\cos\frac{2\pi}{17}$ en $\gamma_1 = \zeta^{13} +
        \zeta^4$. Toon aan dat $\gamma_0 + \gamma_1 = \beta_0$ en
        $\gamma_0\gamma_1 = \beta_1$, zodat $\gamma_0 =
        \frac{\beta_0 + \sqrt{\beta_0^2 - 4\beta_1}}2$.
  \item Stel de keten formules samen die $\cos\frac{2\pi}{17}$
        door geneste vierkantswortels uitdrukt, en geef een
        decimale controle ($\cos\frac{2\pi}{17} \approx
        0.93247$).
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Slotwoord.}
\begin{enumerate}[resume]
  \item Waar precies gebruikte het argument dat $17$ een
        \emph{Fermat-priemgetal} is ($17 = 2^{2^2} + 1$)? Toon aan
        dat voor een priemgetal $p$ de regelmatige $p$-hoek
        \omterm{def:b3:galois:constructible}{construeerbaar} is dan en slechts dan als $p = 2^{2^t} + 1$
        voor zekere $t$ \emph{(toon aan dat uit $p - 1 = 2^m$
        volgt dat $m$ zelf een macht van $2$ moet zijn)}.
  \item Leid de volledige lijst construeerbare regelmatige
        $n$-hoeken voor $n \leq 20$ af, met het criterium van
        Gauss--Wantzel uit \cref{cor:b3:galois:impossible}.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Gauss-sommen en kwadratische
reciprociteit.} De perioden uit Deel II verbergen een schat. Voor
een oneven priemgetal $p$ is het \emph{legendresymbool}
$\bigl(\frac ap\bigr)$ gelijk aan $+1$ als $a$ een kwadraat $\neq
0$ modulo $p$ is, $-1$ als het dat niet is, en $0$ als $p \mid
a$; \cref{exo:b3:galois:4}(b) (het criterium van Euler) geeft
$\bigl(\frac ap\bigr) \equiv a^{(p-1)/2} \pmod p$, waaruit de
multiplicativiteit volgt. Schrijf $\zeta = \eu^{2\iu\pi/p}$, $p^*
= (-1)^{(p-1)/2}p$, en definieer de \emph{Gauss-som}
\[
g \;=\; \sum_{a=1}^{p-1}\Bigl(\frac ap\Bigr)\zeta^a .
\]
\begin{enumerate}[resume]
  \item Toon aan dat $\sum_{a=1}^{p-1}\bigl(\frac ap\bigr) = 0$
        (er zijn evenveel kwadraten als niet-kwadraten), en bewijs
        de alternatieve vorm $g = \sum_{a=0}^{p-1}\zeta^{a^2}$
        \emph{(elk kwadraat $\neq 0$ wordt tweemaal geraakt, en
        $\sum_{a}\zeta^a = 0$)}. Voor $p = 17$: breng $g$ in
        verband met de perioden van Deel II --- toon aan dat $g =
        \eta_0 - \eta_1$ \emph{(de kwadraten modulo $17$ zijn
        precies de even machten van de voortbrenger $3$)}.
  \item Bewijs $g^2 = p^*$: werk
        \[
        g^2 = \sum_{a,b\neq0}\Bigl(\frac{ab}p\Bigr)
        \zeta^{a+b}
        = \sum_{c}\ \sum_{a \neq 0}\Bigl(\frac{a(c -
        a)}p\Bigr)\zeta^{c}
        \]
        uit (stel $b = c - a$), substitueer $c - a = at$ om de
        binnenste som te evalueren als $\bigl(\frac{-1}p\bigr)(p -
        1)$ voor $c = 0$ en als $-\bigl(\frac{-1}p\bigr)$ anders,
        en besluit met vraag 14. Controleer numeriek: voor $p =
        17$ is $(\eta_0 - \eta_1)^2 = 17$ (Deel II).
  \item Leid af dat $\sqrt{p^*} \in \Q(\zeta_p)$, en besluit dat
        het \emph{enige} kwadratische deellichaam van
        $\Q(\zeta_p)$ gelijk is aan $\Q(\sqrt{p^*})$ --- enig,
        omdat $\operatorname{Gal}(\Q(\zeta_p)/\Q)$ cyclisch is
        (\cref{thm:b3:galois:cyclotomicirred}) en een cyclische
        groep precies één deelgroep van index $2$ heeft. (Elk
        kwadratisch lichaam bedt zich in een cyclotomisch lichaam
        in --- het eerste geval van de stelling van
        Kronecker--Weber, waarvan de algemene vorm nog ver voor
        ons ligt.)
  \item Zij nu $q \neq p$ een ander oneven priemgetal. Bewijs,
        werkend in de ring $\Z[\zeta]$ modulo $q$, dat
        \[
        g^q \equiv \Bigl(\frac qp\Bigr)\,g \pmod{q\Z[\zeta]}
        \]
        \emph{(droom van de eerstejaars: $(x + y)^q \equiv x^q +
        y^q$ modulo $q$ in elke commutatieve ring; vervolgens $g^q
        \equiv \sum_a\bigl(\frac ap\bigr)^q\zeta^{aq}$,
        herindexeer $b = aq$ en haal $\bigl(\frac{q^{-1}}p\bigr) =
        \bigl(\frac qp\bigr)$ naar buiten)}.
  \item Anderzijds is $g^q = g\,(g^2)^{(q-1)/2} =
        g\,(p^*)^{(q-1)/2}$; leid met het criterium van Euler
        modulo $q$ af dat $g^q \equiv \bigl(\frac{p^*}q\bigr)g
        \pmod{q\Z[\zeta]}$, en besluit --- door de twee
        uitdrukkingen voor $g^q$ met $g$ te vermenigvuldigen en
        $g^2 = p^*$ te gebruiken, dat inverteerbaar is modulo $q$
        --- dat
        \[
        \Bigl(\frac qp\Bigr) = \Bigl(\frac{p^*}q\Bigr) .
        \]
        \emph{(Waarom impliceert een congruentie tussen de gehele
        getallen $\pm p^*$ modulo $q\Z[\zeta]$ hun gelijkheid?
        Snijd met $\Z$.)}
  \item Werk $\bigl(\frac{p^*}q\bigr) =
        \bigl(\frac{-1}q\bigr)^{(p-1)/2}\bigl(\frac pq\bigr)$ en
        $\bigl(\frac{-1}q\bigr) = (-1)^{(q-1)/2}$ uit om de
        \emph{wet van de kwadratische reciprociteit} te
        verkrijgen:
        \[
        \Bigl(\frac pq\Bigr)\Bigl(\frac qp\Bigr)
        = (-1)^{\frac{p-1}2\cdot\frac{q-1}2} .
        \]
        Controleer haar op $(p, q) = (17, 3)$ door de kwadraten
        modulo $17$ en modulo $3$ op te schrijven, en gebruik haar
        om in drie regels te beslissen of $x^2 \equiv 219
        \pmod{383}$ \omterm{def:b3:groups:derived}{oplosbaar} is ($383$ is priem, $219 =
        3\cdot73$).
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Irreducibele veeltermen tellen: de
priemgetalstelling van $\mathbb F_q[X]$.} Leg een priemmacht $q$
vast en zij $N_q(n)$ het aantal \emph{monische irreducibele}
veeltermen van \omterm{def:b3:galois:extension}{graad} $n$ over $\mathbb F_q$; herinner uit
\cref{exo:b3:galois:6} de ontbinding van $X^{q^n} - X$ en de
identiteit $q^n = \sum_{d\mid n}d\,N_q(d)$, die we nu inverteren,
herinterpreteren en uitbuiten.
\begin{enumerate}[resume]
  \item (Woorden) Noem een woord $w \in \mathbb F_q^n$
        \emph{primitief} als het geen macht $u^{n/d} = u\cdots u$
        van een strikt korter woord $u$ is, en zij $A(d)$ het
        aantal primitieve woorden van lengte $d$. Toon aan dat elk
        woord van lengte $n$ op precies één manier een macht is
        van een primitief woord van zekere lengte $d \mid n$,
        zodat $q^n = \sum_{d \mid n}A(d)$; besluit door
        vergelijking met \cref{exo:b3:galois:6} dat $A(d) =
        d\,N_q(d)$ voor elke $d$, en verklaar dat samenvallen met
        een expliciete bijectie: een $\alpha \in \mathbb F_{q^d}$
        van \omterm{def:b3:galois:extension}{graad} $d$ heeft een frobeniusbaan $(\alpha, \alpha^q,
        \dots, \alpha^{q^{d-1}})$ van precies $d$ verschillende
        elementen, en de elementen van \omterm{def:b3:galois:extension}{graad} $d$ corresponderen
        $d$-op-één met de irreducibele veeltermen van \omterm{def:b3:galois:extension}{graad} $d$.
  \item Bewijs de \emph{inversieformule van Möbius}: is $f(n) =
        \sum_{d\mid n}g(d)$ voor alle $n$, dan is $g(n) =
        \sum_{d\mid n}\mu(d)\,f(n/d)$, waarbij $\mu$ de
        möbiusfunctie is ($\mu(m) = (-1)^{\#\text{priemfactoren}}$
        als $m$ kwadraatvrij is, en $0$ anders) \emph{(sleutellemma:
        $\sum_{d \mid m}\mu(d) = 0$ voor $m > 1$ --- paar de
        delers met en zonder een vaste priemfactor)}. Leid af dat
        \[
        N_q(n) = \frac1n\sum_{d \mid n}\mu(d)\,q^{n/d} .
        \]
  \item Toon aan dat $N_q(n) \geq \frac1n\bigl(q^n -
        2q^{n/2}\bigr) > 0$ voor elke $n \geq 1$: een nieuw bewijs
        dat $\mathbb F_{q^n}$ voor alle $n$ bestaat. Interpreteer
        de hoofdterm: een toevallig gekozen monische veelterm van
        \omterm{def:b3:galois:extension}{graad} $n$ is \omterm{def:b3:rings:divisibility}{irreducibel} met kans $\sim \frac1n$ --- het
        volmaakte analogon van de priemgetalstelling, met $n$ in
        plaats van $\log x$; controleer het numeriek voor $q = 2$
        en $n \leq 4$ (\cref{exo:b3:galois:6} geeft de
        aantallen).
  \item Bewijs de multiplicatieve tegenhanger van vraag 21:
        \[
        \prod_{\substack{\pi \text{ monisch irred.}\\ \deg\pi =
        n}}\pi \;=\; \prod_{d \mid n}
        \bigl(X^{q^d} - X\bigr)^{\mu(n/d)}
        \]
        \emph{(möbiusinversie in de abelse groep van de rationale
        functies $\neq 0$)}; ga haar met de hand na voor $q = 2$
        en $n = 2$: $(X^4 - X)/(X^2 - X) = X^2 + X + 1$.
\end{enumerate}

\medskip
\noindent\textbf{Deel VII --- Twee slotstukken.}
\begin{enumerate}[resume]
  \item (Het tweede supplement) De methode van Deel V berekent ook
        $\bigl(\frac2q\bigr)$. Zij $\omega = \eu^{2\iu\pi/8}$ en
        $g = \omega + \omega^{-1}$. Toon aan dat $g^2 = 2$
        \emph{(want $\omega^2 = \iu$)}; bewijs vervolgens voor een
        oneven priemgetal $q$ in $\Z[\omega]$ modulo $q$ dat
        \[
        g^q \equiv \omega^q + \omega^{-q} \pmod{q\Z[\omega]},
        \]
        en dat het rechterlid gelijk is aan $g$ als $q \equiv \pm1
        \pmod 8$ en aan $-g$ als $q \equiv \pm3 \pmod 8$. Besluit,
        door met $g^q = g\,(g^2)^{(q-1)/2} \equiv
        \bigl(\frac2q\bigr)g$ te vergelijken als in vraag 18, dat
        \[
        \Bigl(\frac2q\Bigr) = (-1)^{(q^2-1)/8},
        \]
        en ga na dat $(q^2 - 1)/8$ even is precies wanneer $q
        \equiv \pm1 \pmod 8$. Controle: $2$ is een kwadraat modulo
        $7$ en modulo $17$ ($3^2$ en $6^2$), maar niet modulo $3$
        en niet modulo $5$.
  \item (De zètafunctie van $\mathbb F_q[X]$) Bewijs de identiteit
        van formele machtreeksen in $t$:
        \[
        \prod_{n \geq 1}\bigl(1 - t^n\bigr)^{-N_q(n)}
        = \frac1{1 - qt}
        \]
        \emph{(eenduidige ontbinding in monische irreducibele
        veeltermen: ontwikkel elke factor als meetkundige reeks en
        tel de monische veeltermen van \omterm{def:b3:galois:extension}{graad} $n$)}. Vind de
        identiteit $q^m = \sum_{d \mid m}d\,N_q(d)$ uit
        \cref{exo:b3:galois:6} terug door logaritmen te nemen.
        Controleer de coëfficiënt van $t^2$ met de hand voor $q =
        2$, en bereken met de formule van vraag 21 dat $N_2(6) =
        9$, waarmee $2^6 = 1\cdot2 + 2\cdot1 + 3\cdot2 + 6\cdot9$
        klopt.
\end{enumerate}
\end{problem}
