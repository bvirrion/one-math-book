\chapter{Conforme afbeeldingen en de afbeeldingsstelling van
Riemann}
\label{ch:b3:conformal}

Een \omterm{def:b3:holomorphic:holo}{holomorfe} bijectie tussen twee gebieden vervoert de hele
complexe analyse van het ene naar het andere: zulke afbeeldingen
--- \emph{conform}, omdat ze hoeken behouden --- zijn de
isomorfismen van de \omterm{def:b3:holomorphic:holo}{holomorfe} wereld. Dit hoofdstuk klasseert ze
waar klassering mogelijk is (de schijf, het vlak: het lemma van
Schwarz is de sleutel, een ongelijkheid van verbluffende kracht),
bouwt de compactheidstheorie van \omterm{def:b3:holomorphic:holo}{holomorfe} families (Montel) en
bewijst de diepste bestaansstelling van het vak: \emph{elk}
\omterm{def:b3:conformal:simplyconnected}{enkelvoudig samenhangend} echt deelgebied van $\C$ is, hoe grillig
zijn rand ook is, conform gelijkwaardig met de eenheidsschijf. We
sluiten af met \omterm{prop:b3:conformal:harmonicholo}{harmonische functies} en de \omterm{thm:b3:conformal:poisson}{poissonkern}, waarmee we
het dirichletprobleem op de schijf oplossen --- de analytische
opbrengst van de \omterm{def:b3:conformal:conformal}{conforme} meetkunde. Overal is $\mathbb D =
D(0,1)$ en $\mathbb H = \{\operatorname{Im}z > 0\}$.

\section{Conforme afbeeldingen; möbiustransformaties}

\begin{definition}\label{def:b3:conformal:conformal}
Een \emph{conforme afbeelding}\index{conforme afbeelding} (of
biholomorfisme) tussen \omterm{def:b3:topology:topology}{open verzamelingen} is een \omterm{def:b3:holomorphic:holo}{holomorfe}
bijectie; haar inverse is automatisch \omterm{def:b3:holomorphic:holo}{holomorf}
(\cref{cor:b3:residues:openmapping}). Twee gebieden heten
\emph{conform gelijkwaardig} als zo'n afbeelding bestaat;
$\operatorname{Aut}(\Omega)$ duidt de groep conforme
zelfafbeeldingen aan. Waar $f' \neq 0$ --- overal, voor injectieve
$f$ (het bewijs van \cref{cor:b3:residues:openmapping}) --- is de
differentiaal de vermenigvuldiging met $f'(z) \neq 0$: een
gelijkvormigheid, zodat conforme afbeeldingen de hoeken tussen
krommen behouden, met inbegrip van de oriëntatie.
\end{definition}

\begin{example}[Möbiustransformaties]
\label{ex:b3:conformal:mobius}
Voor $\bigl(\begin{smallmatrix}a & b\\ c &
d\end{smallmatrix}\bigr) \in GL_2(\C)$ is de
\emph{möbiustransformatie}\index{möbiustransformatie} $z \mapsto
\frac{az + b}{cz + d}$ conform van $\C\setminus\{-d/c\}$ op
$\C\setminus\{a/c\}$ (met een inverse van hetzelfde type, uit de
inverse matrix; de samenstelling stemt overeen met het
matrixproduct). De \emph{cayleytransformatie}\index{
cayleytransformatie}
\[
\varphi(z) = \frac{z - \iu}{z + \iu}
\]
beeldt $\mathbb H$ conform af op $\mathbb D$: inderdaad is
$\abs{z - \iu} < \abs{z + \iu}$ precies wanneer $z$ dichter bij
$\iu$ ligt dan bij $-\iu$, dat wil zeggen $\operatorname{Im}z >
0$; de inverse is $w \mapsto \iu\frac{1 + w}{1 - w}$.
Möbiusafbeeldingen sturen de familie cirkels-en-rechten naar
zichzelf (\cref{exo:b3:conformal:1}).
\end{example}

\begin{example}[De afbeelding van Joukowski]
\label{ex:b3:conformal:joukowski}
Voorbij Möbius is de nuttigste \omterm{def:b3:conformal:conformal}{conforme afbeelding} van de
klassieke toegepaste wiskunde
\[
J(z) = \frac12\Bigl(z + \frac1z\Bigr) .
\]
Op het uitwendige $\Omega = \{\abs z > 1\}$ van de
eenheidsschijf is $J$ injectief: $J(z) = J(w)$ geeft $(z - w)(1 -
\frac1{zw}) = 0$ en $\abs{zw} > 1$. Haar afgeleide $J'(z) =
\frac12(1 - z^{-2})$ verdwijnt enkel in $z = \pm1$, op de rand:
$J$ is een \omterm{def:b3:conformal:conformal}{conforme} gelijkwaardigheid van $\Omega$ op haar beeld,
namelijk $\C\setminus\intcc{-1}1$ --- de eenheidscirkel zelf
wordt twee-op-één op het lijnstuk gevouwen
($J(\eu^{\iu\theta}) = \cos\theta$). Het uitwendige van een
\emph{lijnstuk}, een gesneden vlak zonder gladde rand, is dus
conform het uitwendige van een schijf: hoeken zijn geen
belemmering voor \omterm{def:b3:conformal:conformal}{conforme} gelijkwaardigheid, alleen voor de
gladheid van de rand. Beelden van cirkels door $\pm1$ maar met
verschoven middelpunt zijn krommen in de vorm van een vleugelprofiel,
en $J$ samenstellen met möbiusafbeeldingen bracht de stroming om
een cilinder --- met de hand berekenbaar --- over naar de
stroming om een vleugel: in de eerste helft van de twintigste eeuw
\emph{was} dit voorbeeld de aerodynamica. Het is ook de deur naar
Chebyshev: $J$ vervoegt $z \mapsto z^n$ tot de chebyshevveelterm
$T_n$ (\cref{pb:b3:hilbert:1}, Deel V), want $J(z^n) =
\cos(n\theta)$ als $z = \eu^{\iu\theta}$.
\end{example}

\section{Het lemma van Schwarz en automorfismegroepen}

\begin{theorem}[Lemma van Schwarz]\label{thm:b3:conformal:schwarz}
Zij $f \colon \mathbb D \to \mathbb D$ \omterm{def:b3:holomorphic:holo}{holomorf} met $f(0) = 0$.
Dan is\index{lemma van Schwarz}
\[
\abs{f(z)} \leq \abs z \ \ (z \in \mathbb D)
\qquad\text{en}\qquad \abs{f'(0)} \leq 1 ,
\]
en is $\abs{f(z_0)} = \abs{z_0}$ voor één $z_0 \neq 0$, of
$\abs{f'(0)} = 1$, dan is $f(z) = \eu^{\iu\theta}z$ een rotatie.
\end{theorem}

\begin{proof}
$g(z) = f(z)/z$ breidt zich \omterm{def:b3:holomorphic:holo}{holomorf} uit tot $\mathbb D$ (de
singulariteit in $0$ is ophefbaar: $g$ is begrensd nabij $0$,
\cref{thm:b3:residues:riemanncw}; haar waarde in $0$ is $f'(0)$).
Op $\abs z = r < 1$ is $\abs g \leq \frac1r$, dus volgens het
maximumprincipe (\cref{thm:b3:holomorphic:maximum}) is $\abs g
\leq \frac1r$ op $\bar D(0,r)$; laat $r \to 1$ gaan: $\abs g \leq
1$ op $\mathbb D$, wat beide ongelijkheden is. Gelijkheid in een
inwendig punt laat $\abs g$ een inwendig maximum bereiken: $g$ is
dan constant met modulus $1$.
\end{proof}

\begin{theorem}[Automorfismen van de schijf]
\label{thm:b3:conformal:autdisc}
Voor $a \in \mathbb D$ is de
\emph{blaschkefactor}\index{blaschkefactor}
\[
\varphi_a(z) = \frac{z - a}{1 - \bar a z}
\]
een automorfisme van $\mathbb D$ dat $a$ en $0$ verwisselt, met
$\varphi_a^{-1} = \varphi_{-a}$. Elk automorfisme van $\mathbb D$
is $\eu^{\iu\theta}\varphi_a$ voor unieke $\theta \in \R/2\pi\Z$
en $a \in \mathbb D$.
\end{theorem}

\begin{proof}
Op $\abs z = 1$: $\abs{1 - \bar az} = \abs{\bar z}\abs{1 - \bar
az} = \abs{\bar z - \bar a\abs z^2} = \abs{\bar z - \bar a} =
\abs{z - a}$, dus is daar $\abs{\varphi_a} = 1$; volgens het
maximumprincipe is $\varphi_a(\mathbb D) \subseteq \bar{\mathbb
D}$, en de openheid legt het beeld in $\mathbb D$. De
\omterm{def:b3:galois:algebraic}{algebraïsche} identiteit $\varphi_{-a}\circ\varphi_a =
\mathrm{id}$ (rechtstreekse berekening) toont de bijectiviteit.
Zij nu $f \in \operatorname{Aut}(\mathbb D)$ en $a = f^{-1}(0)$:
dan is $g = f\circ\varphi_{-a}$ een automorfisme dat $0$ vastlaat.
Schwarz toegepast op $g$ en op $g^{-1}$: $\abs{g(z)} \leq \abs z$
en $\abs{g^{-1}(w)} \leq \abs w$, dus $\abs{g(z)} = \abs z$: een
rotatie, $g = \eu^{\iu\theta}\,\mathrm{id}$, dat wil zeggen $f =
\eu^{\iu\theta}\varphi_a$. Uniciteit: $a = f^{-1}(0)$, en
$\theta$ uit een evaluatie van het type $f'$ (of uit $f(0) =
-\eu^{\iu\theta}a$ en nog één waarde).
\end{proof}

\begin{theorem}[Automorfismen van het vlak]
\label{thm:b3:conformal:autplane}
$\operatorname{Aut}(\C) = \{z \mapsto az + b : a \in \C^*,\ b \in
\C\}$. Bijgevolg zijn $\C$ en $\mathbb D$ niet conform
gelijkwaardig.
\end{theorem}

\begin{proof}
Zij $f \in \operatorname{Aut}(\C)$ en beschouw $g(z) = f(1/z)$ op
$\C^*$: een \omterm{def:b3:holomorphic:holo}{holomorfe functie} met een \omterm{def:b3:residues:singularities}{geïsoleerde singulariteit}
in $0$. Was ze essentieel, dan zou Casorati--Weierstrass
(\cref{thm:b3:residues:riemanncw})
$g\bigl(D(0,\varepsilon)\setminus\{0\}\bigr)$ \omterm{def:b3:topology:interior}{dicht} maken,
terwijl $f(D(0, 1))$ open is en er disjunct van ($f$ is
injectief: de twee verzamelingen zijn beelden van disjuncte
verzamelingen) --- onmogelijk voor een \omterm{def:b3:topology:interior}{dichte verzameling} en een
niet-lege \omterm{def:b3:topology:topology}{open verzameling}. Dus is $0$ voor $g$ een pool of
ophefbaar, dat wil zeggen $\abs{f(z)}$ heeft hoogstens
polynomiale groei: $f$ is een veelterm
(\cref{exo:b3:holomorphic:4}(a)). De injectiviteit dwingt graad
$1$ af: een veelterm van hogere graad heeft ergens ofwel een
meervoudige wortel van $f - c$ ($f'$ verdwijnt) ofwel meerdere
verschillende originelen (d'Alembert--Gauss,
\cref{pb:b3:holomorphic:1}); in beide gevallen faalt de
injectiviteit. Ten slotte zou een \omterm{def:b3:conformal:conformal}{conforme} $\C \to \mathbb D$ een
begrensde gehele functie zijn: constant (Liouville) --- dus geen
gelijkwaardigheid.
\end{proof}

\section{De stelling van Montel}

\begin{theorem}[Montel]\label{thm:b3:conformal:montel}
Zij $\mathcal F \subseteq \mathcal H(\Omega)$ \emph{lokaal
begrensd}: elk punt heeft een \omterm{def:b3:topology:topology}{omgeving} waarop $\sup_{f\in
\mathcal F}\sup\abs f < \infty$. Dan heeft elke rij uit $\mathcal
F$ een deelrij die uniform convergeert op alle \omterm{def:b3:topology:compact}{compacte}
deelverzamelingen van $\Omega$ (naar een \omterm{def:b3:holomorphic:holo}{holomorfe}
limiet).\index{stelling van Montel}
\end{theorem}

\begin{proof}
Lokale gelijkmatige \omterm{def:b3:topology:continuity}{continuïteit}: is $\abs f \leq M$ op $D(a, 2r)
\subseteq \Omega$ voor alle $f \in \mathcal F$, dan geeft de
formule van Cauchy voor $z, z' \in D(a, r)$
\[
\abs{f(z) - f(z')} = \frac{\abs{z - z'}}{2\pi}
\Bigl|\int_{C_{2r}}\frac{f(w)\,\dd w}{(w-z)(w-z')}\Bigr|
\leq \frac{\abs{z - z'}\,2\pi\cdot2r\,M}{2\pi\,r^2}
= \frac{2M}{r}\,\abs{z - z'} :
\]
een uniforme lipschitzafschatting. Put $\Omega$ uit met \omterm{def:b3:topology:compact}{compacte}
$K_m$; elke $K_m$ wordt door eindig veel zulke schijven overdekt,
dus is $\mathcal F$ uniform begrensd en \omterm{def:b3:complete:equicontinuous}{equicontinu} op $K_m$:
Arzelà--Ascoli (\cref{thm:b3:complete:ascoli}) haalt er een
deelrij uit die uniform convergeert op $K_m$; diagonaliseer over
$m$. De limiet is \omterm{def:b3:holomorphic:holo}{holomorf} volgens
\cref{thm:b3:holomorphic:weierstrassconv}.
\end{proof}

\section{De afbeeldingsstelling van Riemann}

\begin{definition}\label{def:b3:conformal:simplyconnected}
Een open \omterm{def:b3:topology:connected}{samenhangende} $\Omega \subseteq \C$ heet
\emph{enkelvoudig samenhangend}\index{enkelvoudig samenhangend
gebied} (in de homologische zin, wat voor al onze doeleinden
volstaat) als $\operatorname{Ind}_\gamma(w) = 0$ voor elke cykel
$\gamma$ in $\Omega$ en elke $w \notin \Omega$ --- ``geen cykel
van $\Omega$ omsluit een gat''. Volgens de globale stelling van
Cauchy (\cref{thm:b3:residues:globalcauchy}) en
\cref{prop:b3:holomorphic:primitive} heeft op zo'n $\Omega$
\emph{elke \omterm{def:b3:holomorphic:holo}{holomorfe functie} een primitieve}; bijgevolg heeft
elke nulpuntvrije $f \in \mathcal H(\Omega)$ een \omterm{def:b3:holomorphic:holo}{holomorfe}
logaritme ($\exp\circ$(primitieve van $f'/f$), aangepast met een
constante, want $(f\eu^{-L})' = 0$) en \omterm{def:b3:holomorphic:holo}{holomorfe} $n$-de
machtswortels $\eu^{L/n}$.\index{holomorfe logaritme}
\end{definition}

\begin{theorem}[Afbeeldingsstelling van Riemann]
\label{thm:b3:conformal:rmt}
Elke \omterm{def:b3:conformal:simplyconnected}{enkelvoudig samenhangende} open $\Omega \subsetneq \C$ met
$\Omega \neq \varnothing$ is conform gelijkwaardig met $\mathbb
D$; bij gegeven $z_0 \in \Omega$ is er een unieke \omterm{def:b3:conformal:conformal}{conforme} $f
\colon \Omega \to \mathbb D$ met $f(z_0) = 0$ en $f'(z_0) >
0$.\index{afbeeldingsstelling van Riemann}
\end{theorem}

\begin{proof}
\emph{Stap 0: de familie is niet leeg.} Kies $b \notin \Omega$:
$z - b$ is nulpuntvrij op $\Omega$, dus heeft ze een \omterm{def:b3:holomorphic:holo}{holomorfe}
vierkantswortel $h$ ($h^2 = z - b$). $h$ is injectief ($h(z) =
h(z')$ kwadrateert tot $z = z'$), en ligt $w \in h(\Omega)$, dan
$-w \notin h(\Omega)$ ($h(z) = -h(z')$ kwadrateert eveneens tot
$z = z'$, wat $w = -w = 0$ geeft, onmogelijk omdat $h$
nulpuntvrij is). Omdat $h(\Omega)$ open is, bevat ze een schijf
$D(h(z_0), \rho)$; dan is $D(-h(z_0), \rho) \cap h(\Omega) =
\varnothing$, dat wil zeggen $\abs{h(z) + h(z_0)} \geq \rho$ voor
elke $z \in \Omega$. Bijgevolg is
\[
g(z) = \frac{\rho}{2\,\bigl(h(z) + h(z_0)\bigr)}
\]
\omterm{def:b3:holomorphic:holo}{holomorf} en injectief (een möbiusafbeelding samengesteld met de
injectieve $h$), met $\abs g \leq \frac12 < 1$. Samenstellen met
een \omterm{thm:b3:conformal:autdisc}{blaschkefactor} (\cref{thm:b3:conformal:autdisc}) om $g(z_0)$
naar $0$ te brengen maakt de familie
\[
\mathcal F = \{f \colon \Omega \to \mathbb D \text{
holomorf, injectief, } f(z_0) = 0\}
\]
niet leeg.

\emph{Stap 1: een extremaal element.} Zij $s =
\sup_{\mathcal F}\abs{f'(z_0)} \in \intoc0{+\infty}$ ($> 0$: de
leden zijn injectief, dus $f'(z_0) \neq 0$). Neem $f_n \in
\mathcal F$ met $\abs{f_n'(z_0)} \to s$: de familie is door $1$
begrensd, dus haalt Montel (\cref{thm:b3:conformal:montel}) er
een $f_n \to f$ uit die uniform convergeert op \omterm{def:b3:topology:compact}{compacte}
verzamelingen; $f$ is \omterm{def:b3:holomorphic:holo}{holomorf}, $f(z_0) = 0$ en $\abs{f'(z_0)} =
s$ (\cref{thm:b3:holomorphic:weierstrassconv} voor de
afgeleiden), in het bijzonder is $f$ niet constant; $f$ is
injectief volgens Hurwitz (\cref{exo:b3:residues:8}(b)), en
$f(\Omega) \subseteq \bar{\mathbb D}$, dus $\subseteq \mathbb D$
(open afbeelding). Dus bereikt $f \in \mathcal F$ het supremum:
en $s < \infty$.

\emph{Stap 2: de extremale afbeelding is surjectief.} Stel dat $a
\in \mathbb D\setminus f(\Omega)$. Het blaschketransport
$\varphi_a\circ f$ is nulpuntvrij op het \omterm{def:b3:conformal:simplyconnected}{enkelvoudig
samenhangende} $\Omega$: ze heeft een \omterm{def:b3:holomorphic:holo}{holomorfe} vierkantswortel
$F$ (met $F(\Omega) \subseteq \mathbb D$, want $\abs F^2 =
\abs{\varphi_a\circ f} < 1$), injectief (kwadraten
onderscheiden). Normeer: $G = \varphi_{F(z_0)}\circ F \in
\mathcal F$. Terugdraaien: $f = \varphi_{-a}\circ s_2 \circ
\varphi_{-F(z_0)}\circ G$ met $s_2(w) = w^2$; de afbeelding $\Psi
= \varphi_{-a}\circ s_2\circ\varphi_{-F(z_0)} \colon \mathbb D
\to \mathbb D$ is \omterm{def:b3:holomorphic:holo}{holomorf} met $\Psi(0) = f(z_0) = 0$ en is
\emph{geen} rotatie (ze is niet injectief: $s_2$ is dat niet).
Lemma van Schwarz (het strikte geval): $\abs{\Psi'(0)} < 1$, en
de kettingregel $f = \Psi\circ G$ geeft $\abs{f'(z_0)} =
\abs{\Psi'(0)}\,\abs{G'(z_0)} < \abs{G'(z_0)}$ --- in tegenspraak
met de maximaliteit (merk op dat $G \in \mathcal F$). Dus is $f$
surjectief: een \omterm{def:b3:conformal:conformal}{conforme} gelijkwaardigheid.

\emph{Stap 3: normering en uniciteit.} Vermenigvuldig $f$ met
$\eu^{-\iu\arg f'(z_0)}$ om $f'(z_0) > 0$ te maken (dat blijft in
$\mathcal F$). Voldoen $f_1$ en $f_2$ allebei, dan laat $\psi =
f_2\circ f_1^{-1} \in \operatorname{Aut}(\mathbb D)$ $0$ vast met
$\psi'(0) = f_2'(z_0)/f_1'(z_0) > 0$; volgens
\cref{thm:b3:conformal:autdisc} is $\psi$ een rotatie
$\eu^{\iu\theta}$ met $\eu^{\iu\theta} > 0$: dus $\psi =
\mathrm{id}$.
\end{proof}

\begin{remark}\label{rem:b3:conformal:rmtscope}
De stelling is een zuivere bestaansuitspraak van verbluffende
reikwijdte: een vierkant, een halfvlak, het complement van een
snede, het gebied tussen twee rakende cirkels, een gebied met
fractale rand --- alle conform identiek met $\mathbb D$. Wat ze
niet geeft: enige formule (expliciete afbeeldingen zijn de
uitzondering: \cref{exo:b3:conformal:5}), gedrag aan de rand (een
diepere theorie --- de stelling van Carathéodory --- behandelt
dat), of uniciteit van uitbreiding tot $\C$ of tot meervoudig
\omterm{def:b3:topology:connected}{samenhangende} gebieden: de ring $\{1 < \abs z < 2\}$ is
\emph{niet} conform een geperforeerde schijf, en ringen met
verschillende straalverhoudingen zijn niet gelijkwaardig (een
werkelijk moeilijker feit).
\end{remark}

\section{Harmonische functies en de poissonkern}

\begin{proposition}\label{prop:b3:conformal:harmonicholo}
Zij $\Omega$ \omterm{def:b3:conformal:simplyconnected}{enkelvoudig samenhangend} en $u \colon \Omega \to \R$
harmonisch\index{harmonische functie} ($\mathcal C^2$ met $\Delta
u = u_{xx} + u_{yy} = 0$). Dan is $u = \operatorname{Re}F$ voor
een \omterm{def:b3:holomorphic:holo}{holomorfe} $F$, uniek op een imaginaire constante na.
Bijgevolg is $u$ $\mathcal C^\infty$, voldoet ze aan de
eigenschap van de gemiddelde waarde en gehoorzaamt ze aan het
maximumprincipe (geen strikt inwendig extremum, tenzij ze
constant is).
\end{proposition}

\begin{proof}
$g = u_x - \iu u_y$ voldoet aan de \omterm{prop:b3:holomorphic:cauchyriemann}{vergelijkingen van
Cauchy--Riemann} ($P = u_x$, $Q = -u_y$: $P_x = u_{xx} = -u_{yy} =
Q_y$ en $P_y = u_{xy} = u_{yx} = -Q_x$) met \omterm{def:b3:topology:continuity}{continue} partiële
afgeleiden: dus is $g$ \omterm{def:b3:holomorphic:holo}{holomorf}
(\cref{prop:b3:holomorphic:cauchyriemann}; de
$\R$-differentieerbaarheid volgt uit $\mathcal C^1$). Zij $F_0$
een primitieve (enkelvoudige \omterm{def:b3:topology:connected}{samenhang},
\cref{def:b3:conformal:simplyconnected}); dan heeft
$\operatorname{Re}F_0$ gradiënt $(u_x, u_y)$ ($F_0' = g$ vertaalt
zich daar precies in via Cauchy--Riemann voor $F_0$), zodat $u -
\operatorname{Re}F_0$ constant is ($\Omega$ \omterm{def:b3:topology:connected}{samenhangend}): pas
aan tot $F = F_0 + c$. De eigenschappen dragen over uit
\cref{thm:b3:holomorphic:maximum} en
\cref{exo:b3:holomorphic:10} (gebruik voor het maximumprincipe
voor $u$ zelf $\eu^{F}$, zoals daar).
\end{proof}

\begin{theorem}[Formule van Poisson; het dirichletprobleem op de
schijf]\label{thm:b3:conformal:poisson}
Definieer voor $0 \leq r < 1$ de
\emph{poissonkern}\index{poissonkern}
\[
P_r(\theta) = \sum_{n\in\Z}r^{\abs n}\eu^{\iu n\theta}
= \frac{1 - r^2}{1 - 2r\cos\theta + r^2} \;>\; 0 .
\]
Zij $g \colon \partial\mathbb D \to \R$ \omterm{def:b3:topology:continuity}{continu}, en stel voor $z
= r\eu^{\iu\varphi} \in \mathbb D$
\[
u(z) = \frac1{2\pi}\int_0^{2\pi}
P_r(\varphi - t)\,g(\eu^{\iu t})\,\dd t .
\]
Dan is $u$ harmonisch op $\mathbb D$ en breidt ze zich \omterm{def:b3:topology:continuity}{continu}
uit tot $\bar{\mathbb D}$ met randwaarden $g$: de enige zulke
\omterm{prop:b3:conformal:harmonicholo}{harmonische functie}.\index{dirichletprobleem}
\end{theorem}

\begin{proof}
\emph{Identiteiten voor de kern}: twee meetkundige reeksen
sommeren geeft
\[
\sum_{n\in\Z}r^{\abs n}\eu^{\iu n\theta}
= \operatorname{Re}\frac{1 + r\eu^{\iu\theta}}{1 -
r\eu^{\iu\theta}}
= \frac{1 - r^2}{\abs{1 - r\eu^{\iu\theta}}^2},
\]
dat het weergegeven quotiënt is; de positiviteit is duidelijk, en
$\frac1{2\pi}\int_0^{2\pi}P_r = 1$ (enkel $n = 0$ overleeft).

\emph{Harmoniciteit}: met $z = r\eu^{\iu\varphi}$ is
\[
u(z) = \operatorname{Re}\biggl[\frac1{2\pi}\int_0^{2\pi}
\frac{\eu^{\iu t} + z}{\eu^{\iu t} - z}\,
g(\eu^{\iu t})\,\dd t\biggr],
\]
(de kern tussen haakjes heeft reëel deel $P_r(\varphi - t)$:
reken na), en het haakje is \omterm{def:b3:holomorphic:holo}{holomorf} in $z$ op $\mathbb D$
(\cref{exo:b3:holomorphic:7}): dus is $u$ het reële deel van een
\omterm{def:b3:holomorphic:holo}{holomorfe functie}, en dus harmonisch.

\emph{Randwaarden}: $P_r(\cdot)$ is een benaderende eenheid als
$r \to 1^-$: massa $1$, en voor $\delta \leq \abs\theta \leq \pi$
is $P_r(\theta) \leq \frac{1 - r^2}{1 - 2r\cos\delta + r^2} \to
0$ uniform. De standaardsplitsing (\omterm{def:b3:topology:continuity}{continuïteit} van $g$ nabij
$\eu^{\iu\varphi_0}$, begrensdheid elders) geeft
$u(r\eu^{\iu\varphi}) \to g(\eu^{\iu\varphi_0})$ als
$r\eu^{\iu\varphi} \to \eu^{\iu\varphi_0}$, uniform in het
randpunt: dus is de uitbreiding \omterm{def:b3:topology:continuity}{continu}. \emph{Uniciteit}: het
verschil van twee oplossingen is harmonisch op $\mathbb D$,
\omterm{def:b3:topology:continuity}{continu} op de \omterm{def:b3:topology:interior}{afsluiting} en nul op de rand: volgens het
maximumprincipe (toegepast op $\pm$ het verschil) verdwijnt het.
\end{proof}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b3:conformal:1}
(a) Ga na dat de \omterm{ex:b3:conformal:mobius}{cayleytransformatie} $\varphi(z) = \frac{z -
\iu}{z + \iu}$ een bijectie $\mathbb H \to \mathbb D$ is met de
vermelde inverse, en bereken de beelden van $\iu$, $0$, $1$ en
$\infty$ (als limiet).
(b) Toon aan dat $z \mapsto 1/z$ cirkels en rechten op cirkels en
rechten afbeeldt. \emph{(Schrijf hun gemeenschappelijke
vergelijking $\alpha\abs z^2 + \bar\beta z + \beta\bar z + \gamma
= 0$, met $\alpha, \gamma \in \R$.)} Leid hetzelfde af voor alle
möbiusafbeeldingen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:conformal:2}
Zij $f \colon \mathbb D \to \mathbb D$ \omterm{def:b3:holomorphic:holo}{holomorf}.
(a) Is $f(0) = 0$ en $f(a) = a$ voor een $a \neq 0$, toon dan aan
dat $f = \mathrm{id}$.
(b) Is $f$ een automorfisme met twee verschillende vaste punten
in $\mathbb D$, toon dan aan dat $f = \mathrm{id}$ \emph{(vervoeg
met een \omterm{thm:b3:conformal:autdisc}{blaschkefactor} om tot (a) te herleiden)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:3}
(Schwarz--Pick) Bewijs voor \omterm{def:b3:holomorphic:holo}{holomorfe} $f\colon \mathbb D \to
\mathbb D$ dat
\[
\frac{\abs{f'(z)}}{1 - \abs{f(z)}^2} \;\leq\;
\frac{1}{1 - \abs z^2}
\qquad (z \in \mathbb D),
\]
met gelijkheid (in één punt, dus overal) dan en slechts dan als
$f \in \operatorname{Aut}(\mathbb D)$. \emph{(Pas Schwarz toe op
$\varphi_{f(z)}\circ f\circ\varphi_{-z}$.)} Interpretatie:
\omterm{def:b3:holomorphic:holo}{holomorfe} zelfafbeeldingen krimpen de hyperbolische metriek.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:4}
Zoek expliciete \omterm{def:b3:conformal:conformal}{conforme} gelijkwaardigheden:
(a) de strook $\{0 < \operatorname{Im}z < \pi\} \to \mathbb H$;
(b) het kwadrant $\{\operatorname{Re}z > 0, \operatorname{Im}z >
0\} \to \mathbb H$;
(c) de halve schijf $\mathbb D\cap\mathbb H \to$ een kwadrant, en
dan $\to \mathbb H$;
(d) $\mathbb D \to \mathbb D$ die $\frac12$ naar $0$ stuurt met
daar een positieve afgeleide.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:5}
(a) Toon aan dat er geen \omterm{def:b3:conformal:conformal}{conforme afbeelding} $\C \to \mathbb D$
of $\C \to \mathbb H$ bestaat, en geen enkele $\mathbb D \to \C$.
(b) Welke van de volgende zijn conform gelijkwaardig met $\mathbb
D$? Verantwoord met \cref{thm:b3:conformal:rmt} of met een
obstructie: een vierkant; $\C\setminus\intoc{-\infty}0$;
$\mathbb D\setminus\{0\}$; $\{1 < \abs z < 2\}$. \emph{(Voor de

zich volgens \cref{thm:b3:residues:riemanncw}(1) over de
perforatie uitbreiden --- werk dat uit.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:6}
Zij $\mathcal F = \{f \in \mathcal H(\mathbb D) : f(0) = 1,\
\operatorname{Re}f > 0\}$.
(a) Toon aan dat $\mathcal F$ lokaal begrensd is. \emph{(Stel

+ 1}$, die het rechterhalfvlak naar $\mathbb D$ stuurt, en pas

(b) Leid de \emph{grens van Herglotz} af: $\abs{f(z)} \leq
\frac{1 + \abs z}{1 - \abs z}$ voor $f \in \mathcal F$, met de

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:conformal:7}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:8}

(a) $g(\eu^{\iu t}) = \cos t$; (b) $g(\eu^{\iu t}) = \cos^2 t$;
(c) $g = \mathbf 1_{\text{bovenste halve cirkel}}$ --- bereken

gemiddelde waarde. \emph{(Ontwikkel $g$ in een fourierreeks en

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:conformal:9}
(Harnack) Zij $u \geq 0$ harmonisch op $\mathbb D$. Bewijs voor

\[
\frac{1 - r}{1 + r}\,u(0) \;\leq\; u(z) \;\leq\;
\frac{1 + r}{1 - r}\,u(0)
\]
\emph{(schat de \omterm{thm:b3:conformal:poisson}{poissonkern} af tussen $\frac{1-r}{1+r}$ en

op $\C$ die van onderen begrensd is, is constant.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:10}
Los met de \omterm{def:b3:conformal:conformal}{conforme} invariantie van de harmoniciteit ($u\circ f$

u(x + \iu y) = \frac1\pi\,\arg(x + \iu y)
\qquad (\arg \in \intoo0\pi \text{ op } \mathbb H)
\]
harmonisch is op $\mathbb H$ (het imaginaire deel van een
\omterm{def:b3:conformal:simplyconnected}{holomorfe logaritme}) met de vereiste randlimieten in elke $x \neq

\cref{exo:b3:conformal:8}(c) opnieuw af te leiden.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:11}
(Vaste punten en iteratie in de schijf) Zij $f\colon\mathbb D \to
\mathbb D$ \omterm{def:b3:holomorphic:holo}{holomorf}.

(b) Stel dat $f(0) = 0$ en dat $f$ geen rotatie is. Toon aan dat

(c) Illustreer met $f(z) = \frac{z^2 + z}2$: de vaste punten, en

\end{exercise}

(b) Toon aan dat op een \emph{stervormige} \omterm{def:b3:topology:topology}{open verzameling} elke

\end{exercise}

\section{Probleem: de oppervlaktestelling en de kwartstelling van
Koebe}

\begin{problem}[{Weekendopgave --- hoeveel moet een univalente
afbeelding overdekken?}]\label{pb:b3:conformal:1}
Een \emph{univalente} functie is een \omterm{def:b3:holomorphic:holo}{holomorfe} injectie. De
genormeerde univalente functies op de schijf,
\[
\mathcal S = \bigl\{f \in \mathcal H(\mathbb D) \text{
injectief},\ f(z) = z + a_2z^2 + a_3z^3 + \cdots\bigr\},
\]
zijn star aan banden gelegd: we bewijzen de ongelijkheid van
Bieberbach $\abs{a_2} \leq 2$ en leiden de \emph{kwartstelling
van Koebe} af: het beeld van elke $f \in \mathcal S$ bevat de
schijf $D(0, \frac14)$ --- de scherpe universele constante van de
\omterm{def:b3:conformal:conformal}{conforme} meetkunde.\index{kwartstelling van
Koebe}\index{oppervlaktestelling}

\medskip
\noindent\textbf{Deel I --- De oppervlaktestelling.} Zij $g(w) =
w + b_0 + \frac{b_1}w + \frac{b_2}{w^2} + \cdots$ \omterm{def:b3:holomorphic:holo}{holomorf} en
\emph{injectief} op $\{\abs w > 1\}$.
\begin{enumerate}
  \item Zij voor $\rho > 1$ $A_\rho$ de oppervlakte (\omterm{def:b3:measure:lebesgueouter}{lebesguemaat})
        van de \omterm{def:b3:topology:compact}{compacte} verzameling $K_\rho = \C\setminus
        g(\{\abs w > \rho\})$, het gebied omsloten door de gladde
        jordankromme $g(C_\rho)$. Toon met de oppervlakteformule
        uit de stelling van Green--Riemann van het volume van
        bachelorjaar 2 --- de omsloten oppervlakte is
        $\frac1{2\iu} \oint\bar\zeta\,\dd\zeta$ langs de positief
        georiënteerde rand --- aan dat
        \[
        A_\rho = \frac{1}{2\iu}\int_{C_\rho}
        \overline{g(w)}\,g'(w)\,\dd w
        = \pi\Bigl(\rho^2 -
        \sum_{n\geq1}n\,\abs{b_n}^2\rho^{-2n}\Bigr) :
        \]
        vul de \omterm{thm:b3:residues:laurent}{laurentreeksen} van $\bar g$ en $g'$ op $C_\rho$ in
        en integreer term voor term (normale convergentie; alleen
        de producten met frequentie nul overleven).
  \item Laat $\rho \to 1^+$ gaan en besluit tot de
        \emph{oppervlaktestelling}:
        \[
        \sum_{n\geq1}n\,\abs{b_n}^2 \;\leq\; 1 .
        \]
        In het bijzonder is $\abs{b_1} \leq 1$. Wanneer is
        $\abs{b_1} = 1$?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- $\abs{a_2} \leq 2$ van Bieberbach.}
Zij $f = z + a_2z^2 + \cdots \in \mathcal S$.
\begin{enumerate}[resume]
  \item Toon aan dat $f(z^2)/z^2$ \omterm{def:b3:holomorphic:holo}{holomorf} en nulpuntvrij is op
        $\mathbb D$ en een \omterm{def:b3:holomorphic:holo}{holomorfe} vierkantswortel $\varphi$
        toelaat met $\varphi(0) = 1$; stel $h(z) =
        z\varphi(z^2)$, zodat $h(z)^2 = f(z^2)$. Toon aan dat $h$
        een \emph{oneven} univalente functie op $\mathbb D$ is
        met ontwikkeling $h(z) = z + \frac{a_2}2z^3 + \cdots$.
        \emph{(Injectiviteit: $h(z)^2 = h(z')^2$ dwingt $z^2 =
        z'^2$ af; gebruik de oneven pariteit om het af te
        maken.)}
  \item Pas de oppervlaktestelling toe op $g(w) = 1/h(1/w) = w -
        \frac{a_2}{2w} + \cdots$ op $\{\abs w > 1\}$ (ga de
        univalentie en de ontwikkeling na) en besluit $\abs{a_2}
        \leq 2$.
  \item Toon aan dat de \emph{functie van Koebe}
        \[
        k(z) = \frac{z}{(1 - z)^2} = \sum_{n\geq1}n\,z^n
        \]
        tot $\mathcal S$ behoort, $a_2 = 2$ heeft, en $\mathbb D$
        op $\C\setminus\intoc{-\infty}{-\frac14}$ afbeeldt
        \emph{(schrijf $k = \frac14\bigl[\bigl(\frac{1+z}{1 -
        z}\bigr)^2 - 1\bigr]$ en volg de beelden)}: alle
        ongelijkheden die volgen zijn scherp.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De kwartstelling.}
\begin{enumerate}[resume]
  \item Zij $f \in \mathcal S$ en $c \notin f(\mathbb D)$. Toon
        aan dat
        \[
        F(z) = \frac{c\,f(z)}{c - f(z)}
        \]
        tot $\mathcal S$ behoort, en bereken haar tweede
        coëfficiënt: $A_2 = a_2 + \frac1c$.
  \item Pas Bieberbach toe op zowel $f$ als $F$: besluit
        $\abs{\frac1c} \leq \abs{A_2} + \abs{a_2} \leq 4$, dat
        wil zeggen $\abs c \geq \frac14$. \emph{Elke overgeslagen
        waarde heeft modulus $\geq \frac14$}: dus $f(\mathbb D)
        \supseteq D(0, \frac14)$ --- de kwartstelling van Koebe.
        Ga de scherpte na op de functie van Koebe.
  \item Leid een kwantitatieve schatting voor de
        riemannafbeelding af: is $\varphi \colon \Omega \to
        \mathbb D$ de riemannafbeelding van
        \cref{thm:b3:conformal:rmt} in $z_0$, dan is
        \[
        \frac{d\bigl(z_0, \partial\Omega\bigr)}{4}
        \;\leq\; \frac1{\varphi'(z_0)} \;\leq\;
        4\,d\bigl(z_0, \partial\Omega\bigr)
        \]
        --- bewijs minstens de linkerongelijkheid door Koebe toe
        te passen op een geschikt genormeerde $\varphi^{-1}$, en
        de rechter met Schwarz toegepast op $\varphi$ op de
        schijf $D(z_0, d) \subseteq \Omega$.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Perspectief.}
\begin{enumerate}[resume]
  \item Bieberbach vermoedde (1916) dat $\abs{a_n} \leq n$ voor
        alle $n$, met gelijkheid enkel voor rotaties van de
        functie van Koebe; de Branges bewees het in 1985. Ga het
        vermoeden met de hand na voor de functie van Koebe en
        haar rotaties $\eu^{-\iu\theta}k(\eu^{\iu\theta}z)$. Duw
        daarna de ontwikkeling van vraag 4 één term verder: toon
        met $h(z) = z + \frac{a_2}2z^3 + c_5z^5 + \cdots$ aan dat
        $c_5 = \frac{a_3}2 - \frac{a_2^2}8$ en
        \[
        g(w) = w - \frac{a_2}{2}\,w^{-1} +
        \Bigl(\frac{3a_2^2}8 - \frac{a_3}2\Bigr)w^{-3} +
        \cdots ,
        \]
        zodat de oppervlaktestelling de verfijnde ongelijkheid
        $\bigl|\frac{a_2}2\bigr|^2 + 3\bigl|\frac{3a_2^2}8 -
        \frac{a_3}2\bigr|^2 \leq 1$ oplevert. Controleer haar op
        de functie van Koebe ($a_2 = 2$, $a_3 = 3$).
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- De vervormingsstelling.} De
ongelijkheid van Bieberbach, met automorfismen rond de schijf
vervoerd, beheerst $f'$ overal. Leg $f \in \mathcal S$ vast.
\begin{enumerate}[resume]
  \item (Transformatie van Koebe) Zij voor $z_0 \in \mathbb D$
        $\varphi(z) = \frac{z + z_0}{1 + \bar z_0z}$, een
        automorfisme van de schijf
        (\cref{thm:b3:conformal:autdisc}) met $\varphi(0) = z_0$.
        Toon aan dat
        \[
        F(z) = \frac{f(\varphi(z)) - f(z_0)}
        {f'(z_0)\,\bigl(1 - \abs{z_0}^2\bigr)}
        \]
        tot $\mathcal S$ behoort \emph{(de univalentie wordt
        geërfd; bereken $\varphi'(0) = 1 - \abs{z_0}^2$ en
        controleer de normering; herinner je dat $f' \neq 0$ voor
        injectieve $f$, \cref{def:b3:conformal:conformal})}.
  \item Bereken de tweede coëfficiënt $A_2 = \frac12F''(0)$ van
        $F$:
        \[
        A_2 = \frac12\Bigl[\bigl(1 - \abs{z_0}^2\bigr)
        \frac{f''(z_0)}{f'(z_0)} - 2\bar z_0\Bigr] ,
        \]
        en leid uit Bieberbach (vraag 4) voor $z =
        r\eu^{\iu\theta}$ de \emph{fundamentele ongelijkheid} af:
        \[
        \Bigl|\,z\,\frac{f''(z)}{f'(z)} -
        \frac{2r^2}{1 - r^2}\Bigr|
        \;\leq\; \frac{4r}{1 - r^2} .
        \]
  \item Haal het reële deel eruit:
        \[
        \frac{2r^2 - 4r}{1 - r^2} \;\leq\;
        \operatorname{Re}\Bigl(z\,\frac{f''(z)}{f'(z)}\Bigr)
        \;\leq\; \frac{2r^2 + 4r}{1 - r^2} .
        \]
  \item Toon aan dat $\frac{\dd}{\dd t}\log\bigl|
        f'(t\eu^{\iu\theta})\bigr| = \frac1t
        \operatorname{Re}\bigl(z\frac{f''(z)}{f'(z)}\bigr)$ in $z
        = t\eu^{\iu\theta}$ \emph{(voor een nergens verdwijnende
        $\mathcal C^1$-functie $g$ van een reële veranderlijke is
        $\frac{\dd}{\dd t}\log\abs g = \operatorname{Re}(g'/g)$)},
        en integreer de grenzen van vraag 12 langs de straal om
        de \emph{vervormingsstelling} te verkrijgen:
        \[
        \frac{1 - r}{(1 + r)^3} \;\leq\; \abs{f'(z)}
        \;\leq\; \frac{1 + r}{(1 - r)^3},
        \qquad \abs z = r .
        \]
  \item Leid de \emph{groeistelling} af:
        \[
        \frac{r}{(1 + r)^2} \;\leq\; \abs{f(z)} \;\leq\;
        \frac{r}{(1 - r)^2},
        \qquad \abs z = r
        \]
        \emph{(bovengrens: integreer $f'$ op het lijnstuk $[0,
        z]$; ondergrens: is $\abs{f(z)} < \frac14$, dan ligt het
        lijnstuk $[0, f(z)]$ volgens vraag 7 in $f(\mathbb D)$;
        trek het terug met $f^{-1}$ --- \omterm{def:b3:holomorphic:holo}{holomorf} volgens
        \cref{cor:b3:residues:openmapping} --- en schat
        $\abs{f(z)} = \int_\gamma\abs{f'(\zeta)}
        \,\abs{\dd\zeta} \geq \int_0^r\frac{1 - t}{(1 +
        t)^3}\,\dd t$ af, met $\abs{\dd\zeta} \geq
        \dd\abs\zeta$)}.
  \item Ga na dat de functie van Koebe in alle vier de grenzen
        gelijkheid bereikt, in $z = r$ voor de bovenste en in $z
        = -r$ voor de onderste: $k$ is tegelijk het meest
        uitrekkende en, op het tegenoverliggende punt, het meest
        samentrekkende lid van $\mathcal S$.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Extremale starheid.} In elke
ongelijkheid tot nu toe identificeert de gelijkheid de functie
van Koebe op een rotatie na. We bewijzen dat, en oogsten daarna.
\begin{enumerate}[resume]
  \item Stel dat $f \in \mathcal S$ $\abs{a_2} = 2$ heeft. Jaag
        de gelijkheid door de vragen 2 tot 4: de
        oppervlaktestelling dwingt $g(w) = w + b_0 +
        \eu^{\iu\alpha}/w$ af; de oneven pariteit van $h$ maakt
        $g$ oneven, dus $b_0 = 0$; inverteer om $h$ te vinden en
        dan $f$, en besluit dat
        \[
        f(z) = \eu^{-\iu\theta}k\bigl(\eu^{\iu\theta}z\bigr)
        \quad\text{met } \eu^{\iu\theta} =
        -\eu^{\iu\alpha} :
        \]
        de rotaties van de functie van Koebe zijn de enige leden
        van $\mathcal S$ met $\abs{a_2} = 2$.
  \item Toon aan dat als $f \in \mathcal S$ een waarde $c$
        overslaat met precies $\abs c = \frac14$, $f$ een rotatie
        van de functie van Koebe is, en identificeer $c =
        -\eu^{-\iu\theta}/4$ \emph{(volg de gelijkheid door de
        keten $4 = \abs{1/c} = \abs{A_2 - a_2} \leq \abs{A_2} +
        \abs{a_2} \leq 4$ van vraag 7)}: de constante van de
        kwartstelling wordt enkel door de extremale familie
        bereikt.
  \item (Coëfficiënten voor een prikje) Combineer de groeistelling
        met de \omterm{thm:b3:holomorphic:analytic}{afschattingen van Cauchy}
        (\cref{thm:b3:holomorphic:analytic}) op de cirkel $\abs z
        = 1 - \frac1n$ om
        \[
        \abs{a_n} \;\leq\; \eu\,n^2 \qquad (n \geq 2)
        \]
        te bewijzen. (De Branges, 1985: $\abs{a_n} \leq n$; de
        factor $\eu n$ is de prijs van elementair gereedschap.)
  \item (Overdekking van deelschijven) Toon aan dat voor elke $0
        < r < 1$
        \[
        f\bigl(D(0, r)\bigr) \supseteq
        D\Bigl(0, \frac{r}{(1 + r)^2}\Bigr),
        \]
        scherp voor de functie van Koebe, en vind de
        kwartstelling terug als $r \to 1^-$. \emph{(Randpunten van
        het open beeld $f(D(0,r))$ liggen op $f(\partial
        D(0,r))$, en hebben dus volgens vraag 14 modulus $\geq
        \frac{r}{(1+r)^2}$; een lijnstuk van $0$ naar een gemist
        punt met kleinere modulus zou die rand moeten kruisen.)}
  \item (Koebe in elk punt) Zij $f$ univalent op $\mathbb D$,
        niet noodzakelijk genormeerd, en $z_0 \in \mathbb D$.
        Bewijs
        \[
        \tfrac14\bigl(1 - \abs{z_0}^2\bigr)\abs{f'(z_0)}
        \;\leq\;
        d\bigl(f(z_0), \partial f(\mathbb D)\bigr)
        \;\leq\;
        \bigl(1 - \abs{z_0}^2\bigr)\abs{f'(z_0)}
        \]
        \emph{(links: de kwartstelling toegepast op de
        transformatie van Koebe uit vraag 10; rechts: Schwarz
        (\cref{thm:b3:conformal:schwarz}) toegepast op
        $\psi^{-1}\circ\hat g$, met $\hat g(w) = f^{-1}\bigl(f(z_0)
        + dw\bigr)$, $d$ de afstand, en $\psi$ een automorfisme
        van de schijf dat $0$ naar $z_0$ stuurt)}. Waarom is
        $\partial f(\mathbb D)$ niet leeg?
  \item Controleer vraag 20 op $f = k$ in $z_0 = r \in
        \intoo01$: bereken $d\bigl(k(r), \partial k(\mathbb
        D)\bigr) = \frac{(1+r)^2}{4(1-r)^2}$ en ga na dat de
        linkerongelijkheid een \emph{gelijkheid} is: de functie
        van Koebe verzadigt haar eigen stelling in elk punt van
        $\intoo01$.
  \item (De moraal) In tien regels: welk enkel beginsel ligt aan
        de oppervlaktestelling ten grondslag, en hoe vloeien
        Bieberbach, de kwartstelling, de vervorming, de groei en
        de overdekking er alle uit voort? Vergelijk met de wereld
        van Schwarz--Pick uit \cref{thm:b3:conformal:schwarz}: in
        beide maakt één inwendige ongelijkheid de hele meetkunde
        star, en zijn de extremalen uniek op een rotatie na.

\end{enumerate}
\medskip
\noindent\textbf{Deel VII --- \omterm{def:b3:topology:compact}{Compactheid}, inversen en een
werkelijkheidstoets.}
\begin{enumerate}[resume]
  \item Toon aan dat de klasse $\mathcal S$ \emph{\omterm{def:b3:topology:compact}{compact}} is
        voor lokaal uniforme convergentie: ze is lokaal begrensd
        volgens de groeistelling, dus normaal
        (\cref{thm:b3:conformal:montel}); en een lokaal uniforme
        limiet van leden van $\mathcal S$ ligt opnieuw in
        $\mathcal S$ \emph{(de normeringen gaan naar de limiet
        wegens de convergentie van de afgeleiden volgens
        Weierstrass; de injectiviteit overleeft volgens Hurwitz,
        \cref{exo:b3:residues:8}, want de limiet is niet
        constant)}. Waarom is dat van belang voor extremale
        problemen zoals dat van Bieberbach?
  \item Zij voor $f \in \mathcal S$ $g = f^{-1}$, gedefinieerd
        nabij $0$. Toon aan dat $g(w) = w - a_2w^2 + O(w^3)$,
        zodat de tweede coëfficiënt van de inverse aan dezelfde
        scherpe grens $\abs{A_2} = \abs{a_2} \leq 2$ voldoet, met
        gelijkheid precies voor de gedraaide functies van Koebe.
  \item Bepaal voor welke $a \in \C$ de veelterm $f(z) = z + az^2$
        tot $\mathcal S$ behoort: toon aan dat $f$ injectief is
        op $\mathbb D$ dan en slechts dan als $\abs a \leq
        \frac12$ \emph{(ontbind $f(z_1) - f(z_2)$)}. Besluit:
        voor veeltermen van graad twee is de werkelijke grens
        voor de coëfficiënt $\frac12$, vier keer kleiner dan de
        $2$ van Bieberbach --- de extremalen van $\mathcal S$
        zijn werkelijk transcendente objecten, en geen enkele
        veelterm komt in de buurt.
\end{enumerate}
\end{problem}
