\chapter{Productmaten, Fubini, variabelensubstitutie}
\label{ch:b3:product}

De eendimensionale theorie van Lebesgue wordt meerdimensionale
analyse door twee stellingen. \emph{Tonelli--Fubini} zegt dat
integralen over producten herhaalde integralen zijn --- in beide
volgorden mag je in plakken snijden, onder hypothesen die je
werkelijk kunt controleren. De \emph{\omterm{thm:b3:product:changeofvar}{formule voor
variabelensubstitutie}} vervoert integralen langs
$\mathcal C^1$-diffeomorfismen, met de jacobiaandeterminant als
wisselkoers voor het volume; we bewijzen haar \omterm{def:b3:complete:complete}{volledig}, te
beginnen bij het lineaire geval, waar ze verklaart wat de
determinant \emph{is}. De toepassingen buitelen over elkaar heen:
de laagjesformule, de convolutie, de \omterm{ex:b3:product:polar}{poolcoördinaten}, het volume
van de $n$-bal --- en, in de weekendopgave, de formule van
Stirling met een eerlijke foutanalyse.

\section{Product-$\sigma$-algebra's en productmaten}

\begin{definition}\label{def:b3:product:sigma}
Voor \omterm{def:b3:lebesgue:measurable}{meetbare} ruimten $(X, \mathcal A)$ en $(Y, \mathcal B)$ is de
\emph{product-$\sigma$-algebra}\index{product-$\sigma$-algebra}
$\mathcal A \otimes \mathcal B$ op $X \times Y$ voortgebracht door
de \emph{rechthoeken} $A \times B$ (met $A \in \mathcal A$ en $B
\in \mathcal B$) --- een $\pi$-systeem. Voor $E \subseteq X\times
Y$ en $x \in X$ is de \emph{sectie} $E_x = \{y : (x,y) \in E\}$;
en voor een functie $f$ op het product is $f_x = f(x, \cdot)$.
\end{definition}

\begin{proposition}\label{prop:b3:product:sections}
(a) Is $E \in \mathcal A\otimes\mathcal B$, dan is elke sectie
$E_x \in \mathcal B$ (en symmetrisch); en is $f$ $\mathcal
A\otimes\mathcal B$-meetbaar, dan is elke $f_x$ $\mathcal
B$-meetbaar.
(b) $\mathcal B(\R^m)\otimes\mathcal B(\R^n) = \mathcal
B(\R^{m+n})$.
\end{proposition}

\begin{proof}
(a) Goede verzamelingen: $\{E : E_x \in \mathcal B\ \forall x\}$
is een $\sigma$-algebra (secties verdragen zich met complementen
en aftelbare verenigingen) die de rechthoeken bevat. En voor $f$:
$(f_x)^{-1}(B) = (f^{-1}(B))_x$.
(b) ($\subseteq$) Rechthoeken van borelverzamelingen: het volstaat
dat blokken open$\times$open borel zijn in $\R^{m+n}$ (ze zijn
open) en dat algemene borelrechthoeken limieten zijn --- opnieuw
goede verzamelingen: $\{A : A\times\R^n \in \mathcal
B(\R^{m+n})\}$ is een $\sigma$-algebra die de \omterm{def:b3:topology:topology}{open verzamelingen}
bevat; snijd er twee van dat type. ($\supseteq$) Elke \omterm{def:b3:topology:topology}{open
verzameling} van $\R^{m+n}$ is een aftelbare vereniging van
rationale open blokken $U\times V$: dus bevat in de
product-$\sigma$-algebra.
\end{proof}

\begin{theorem}[Productmaat]\label{thm:b3:product:existence}
Zij $(X, \mathcal A, \mu)$ en $(Y, \mathcal B, \nu)$
\emph{$\sigma$-eindig}. Voor elke $E \in \mathcal
A\otimes\mathcal B$ is de functie $x \mapsto \nu(E_x)$ \omterm{def:b3:lebesgue:measurable}{meetbaar},
en\index{productmaat}
\[
(\mu\otimes\nu)(E) = \int_X \nu(E_x)\,\dd\mu(x)
\]
definieert de unieke \omterm{def:b3:measure:measure}{maat} op $\mathcal A\otimes\mathcal B$ met
$(\mu\otimes\nu)(A\times B) = \mu(A)\nu(B)$. Ze is
$\sigma$-eindig en symmetrisch: dezelfde \omterm{def:b3:measure:measure}{maat} komt eruit als je de
$x$-secties tegen $\nu$ integreert.
\end{theorem}

\begin{proof}
\emph{\omterm{def:b3:lebesgue:measurable}{Meetbaarheid} van $x \mapsto \nu(E_x)$.} Zij $\nu$ eerst
eindig. De klasse $\mathcal D$ van de $E$ waarvoor de afbeelding
\omterm{def:b3:lebesgue:measurable}{meetbaar} is, bevat de rechthoeken ($\nu((A\times B)_x) =
\nu(B)\mathbf 1_A(x)$) en is een $\lambda$-systeem: voor $E
\subseteq F$ in $\mathcal D$ is $\nu((F\setminus E)_x) = \nu(F_x)
- \nu(E_x)$ (eindigheid); en voor $E_n \uparrow E$ is
$\nu((E_n)_x) \uparrow \nu(E_x)$ (\omterm{def:b3:topology:continuity}{continuïteit} van onderen),
terwijl monotone limieten van \omterm{def:b3:lebesgue:measurable}{meetbare functies} \omterm{def:b3:lebesgue:measurable}{meetbaar} zijn. De
rechthoeken vormen een $\pi$-systeem: Dynkin
(\cref{thm:b3:measure:dynkin}) geeft $\mathcal D = \mathcal
A\otimes\mathcal B$. Is $\nu$ $\sigma$-eindig, schrijf dan $Y =
\bigcup Y_k$ met $Y_k \uparrow$ en $\nu(Y_k) < \infty$: dan is
$\nu(E_x) = \lim_k\nu_k(E_x)$ met $\nu_k = \nu(\cdot\cap Y_k)$
eindig.

\emph{\omterm{def:b3:measure:measure}{Maat}.} De $\sigma$-additiviteit van $E \mapsto
\int\nu(E_x)\dd\mu$ volgt uit
\cref{cor:b3:lebesgue:additivity} (secties van disjuncte
verzamelingen zijn disjunct). Op rechthoeken geeft ze
$\mu(A)\nu(B)$. \emph{Eenduidigheid}: twee kandidaten vallen samen
op het $\pi$-systeem van de rechthoeken; en de
$\sigma$-eindigheid levert rechthoeken $X_k\times Y_k \uparrow
X\times Y$ van eindige \omterm{def:b3:measure:measure}{maat}:
\cref{thm:b3:measure:uniqueness}. Symmetrie: de constructie in de
andere volgorde is eveneens een \omterm{def:b3:measure:measure}{maat} die op de rechthoeken
samenvalt --- uniek, dus dezelfde.
\end{proof}

\begin{definition}\label{def:b3:product:lebesgue}
De \emph{lebesguemaat op $\R^d$}\index{lebesguemaat op $\R^d$} is
$\lambda_d = \lambda\otimes\cdots\otimes\lambda$ ($d$ factoren; de
associativiteit van de constructie ga je na op blokken en plant
zich voort door de eenduidigheid). Zij is de unieke borelmaat die
aan elk blok $\prod\intoc{a_i}{b_i}$ zijn volume $\prod(b_i -
a_i)$ toekent; ze is translatie-invariant (translaten vallen samen
op blokken), $\sigma$-eindig, en \omterm{def:b3:complete:complete}{volledig} na \omterm{thm:b3:complete:completion}{vervollediging}
volgens Carathéodory --- we schrijven $\lambda_d$ voor de
vervolledigde \omterm{def:b3:measure:measure}{maat} en integreren dienovereenkomstig.
\end{definition}

\section{Tonelli en Fubini}

\begin{theorem}[Tonelli]\label{thm:b3:product:tonelli}
Zij $\mu, \nu$ $\sigma$-eindig en $f \colon X\times Y \to [0,
+\infty]$ \omterm{def:b3:lebesgue:measurable}{meetbaar}. Dan is $x \mapsto \int_Y f_x\,\dd\nu$ \omterm{def:b3:lebesgue:measurable}{meetbaar}
en\index{stelling van Tonelli}
\[
\int_{X\times Y}f\,\dd(\mu\otimes\nu)
= \int_X\Bigl(\int_Y f(x,y)\,\dd\nu(y)\Bigr)\dd\mu(x)
= \int_Y\Bigl(\int_X f(x,y)\,\dd\mu(x)\Bigr)\dd\nu(y).
\]
\end{theorem}

\begin{proof}
De standaardmachine. Voor $f = \mathbf 1_E$ is dit
\cref{thm:b3:product:existence} (met haar symmetrische vorm). Door
de lineariteit geldt het voor elementaire $f \geq 0$. En voor
algemene $f \geq 0$: neem elementaire $s_n \nearrow f$
(\cref{thm:b3:lebesgue:approximation}); dan is $\int_Y(s_n)_x
\dd\nu \nearrow \int_Y f_x\dd\nu$ voor elke $x$ (monotone
convergentie in $Y$), zodat de linkerleden met de monotone
convergentie in $X$ convergeren, terwijl $\int s_n\,
\dd(\mu\otimes\nu) \nearrow \int f$ door de monotone convergentie
op het product.
\end{proof}

\begin{theorem}[Fubini]\label{thm:b3:product:fubini}
Zij $\mu, \nu$ $\sigma$-eindig en $f \in L^1(\mu\otimes\nu)$. Dan
is voor $\mu$-bijna elke $x$ de sectie $f_x$ $\nu$-integreerbaar,
is de \omterm{def:b3:lebesgue:l1}{bijna overal} gedefinieerde functie $x \mapsto \int
f_x\dd\nu$ \omterm{def:b3:lebesgue:l1}{integreerbaar}, en zijn beide herhaalde integralen
gelijk aan $\int f\,\dd(\mu\otimes\nu)$.\index{stelling van
Fubini}
\end{theorem}

\begin{proof}
Tonelli toegepast op $\abs f$ laat zien dat $\varphi(x) =
\int\abs{f_x} \dd\nu$ een eindige integraal heeft en dus \omterm{def:b3:lebesgue:l1}{bijna
overal} eindig is: $f_x \in L^1(\nu)$ voor bijna elke $x$. Splits
$f = f^+ - f^-$ (reëel geval; complex per component): Tonelli
berekent elke herhaalde integraal van $f^\pm$ als $\int
f^\pm\dd(\mu\otimes\nu) < \infty$, en het \omterm{def:b3:lebesgue:l1}{bijna overal}
gedefinieerde verschil integreert tot het verschil. Symmetrisch
voor de andere volgorde.
\end{proof}

\begin{method}\label{met:b3:product:fubini}
Om twee integralen te verwisselen (of een integraal en een som, of
twee sommen): is de integrand \emph{niet-negatief}, verwissel dan
vrijelijk (Tonelli). Zo niet, pas dan eerst Tonelli toe op $\abs
f$, in de volgorde die het makkelijkst af te schatten is; is het
resultaat eindig, dan legitimeert Fubini de verwisseling. Sla de
controle op $\abs f$ nooit over: de integrand van
\cref{exo:b3:product:4} heeft twee herhaalde integralen met
\emph{verschillende waarden}.
\end{method}

\begin{proposition}[Laagjesformule]\label{prop:b3:product:layercake}
Voor \omterm{def:b3:lebesgue:measurable}{meetbare} $f \geq 0$ op een $\sigma$-eindige $(X, \mathcal A,
\mu)$ geldt\index{laagjesformule}
\[
\int_X f\,\dd\mu = \int_0^{+\infty}\mu(\{f > t\})\,\dd t,
\qquad
\int_X f^p\,\dd\mu = p\int_0^{+\infty}t^{p-1}\mu(\{f >
t\})\,\dd t \quad (p \geq 1).
\]
\end{proposition}

\begin{proof}
Pas Tonelli toe op $\mathbf 1_{\{(x,t) : 0 < t < f(x)\}}$ op $X
\times \intoo0{+\infty}$ (\omterm{def:b3:lebesgue:measurable}{meetbaar}: het is $\{(x,t): f(x) - t >
0\}\cap\{t > 0\}$, een borelachtige combinatie van de \omterm{def:b3:lebesgue:measurable}{meetbare}
$(x,t)\mapsto f(x) - t$): eerst in $t$ integreren geeft $\int
f\,\dd\mu$; eerst in $x$ geeft $\int_0^\infty\mu(f > t)\dd t$.
Voor $f^p$: substitueer $t = s^p$ in $\int\mu(f^p > t)\dd t$, dat
wil zeggen pas de eerste formule toe op $f^p$ en substitueer in de
eendimensionale integraal ($\{f^p > s^p\} = \{f > s\}$).
\end{proof}

\begin{theorem}[Convolutie op $L^1$]\label{thm:b3:product:convolution}
Voor $f, g \in L^1(\R^d, \lambda_d)$ convergeert de integraal
\[
(f * g)(x) = \int_{\R^d} f(x - y)\,g(y)\,\dd y
\]
absoluut voor bijna elke $x$, definieert ze een $f * g \in
L^1(\R^d)$ met $\norm{f*g}_1 \leq \norm f_1\norm g_1$, en is $*$
commutatief en associatief.\index{convolutie}
\end{theorem}

\begin{proof}
$(x, y) \mapsto f(x-y)g(y)$ is \omterm{def:b3:lebesgue:measurable}{meetbaar} ($(x,y)\mapsto x - y$ is
\omterm{def:b3:topology:continuity}{continu}; stel samen en vermenigvuldig). Tonelli:
\[
\int\!\!\int \abs{f(x-y)}\abs{g(y)}\,\dd y\,\dd x
= \int\abs{g(y)}\Bigl(\int\abs{f(x - y)}\dd x\Bigr)\dd y
= \norm f_1\norm g_1 < \infty
\]
(de translatie-invariantie van $\lambda_d$ in de binnenste
integraal). De dubbele integraal is dus eindig; Fubini geeft de
absolute convergentie voor bijna elke $x$ en de normgrens
$\norm{f*g}_1 \leq \norm f_1\norm g_1$. Commutativiteit:
substitueer $y \mapsto x - y$ (invariantie onder translatie en
spiegeling --- de spiegelingsinvariantie geldt op blokken en dus
overal wegens de eenduidigheid). Associativiteit:
Tonelli--Fubini op een drievoudige integraal.
\end{proof}

\section{Variabelensubstitutie}

\begin{theorem}[Lineaire variabelensubstitutie]
\label{thm:b3:product:linearchange}
Voor $T \in GL_d(\R)$ en $A \in \mathcal B(\R^d)$ is
$\lambda_d(T(A)) = \abs{\det T}\,\lambda_d(A)$; bijgevolg is $\int
f(y)\dd y = \abs{\det T}\int f(Tx)\,\dd x$ voor $f \geq 0$ of $f$
\omterm{def:b3:lebesgue:l1}{integreerbaar}.\index{lineaire variabelensubstitutie}
\end{theorem}

\begin{proof}
De \omterm{def:b3:measure:measure}{maat} $\mu_T(A) = \lambda_d(T(A))$ is een borelmaat
(\omterm{def:b3:topology:continuity}{homeomorfismen} bewaren borelverzamelingen,
\cref{pb:b3:measure:1}), translatie-invariant ($T(A + x) = T(A) +
Tx$) en eindig op het eenheidsblok: volgens de karakterisering van
de \omterm{def:b3:measure:lebesgueouter}{lebesguemaat} (\cref{exo:b3:measure:6}, waarvan het bewijs
letterlijk in $\R^d$ werkt met dyadische kubussen) is $\mu_T =
c(T)\lambda_d$ met $c(T) = \lambda_d(T(\intco01^d))$. De
afbeelding $T \mapsto c(T)$ is multiplicatief ($c(ST) = c(S)c(T)$,
door samen te stellen), dus volstaat het $c$ op voortbrengers van
$GL_d$ te berekenen: de elementaire matrices. Diagonaal
$\operatorname{diag}(a, 1, \dots, 1)$: beeldt de eenheidskubus af
op een blok van volume $\abs a$, dus $c = \abs a = \abs\det$.
Verwisseling van coördinaten: permuteert de kubus, dus $c = 1 =
\abs\det$. Transvectie $T(x) = x + \alpha x_2e_1$: het beeld van
de eenheidskubus is een scheefgetrokken prisma; volgens Tonelli is
de \omterm{def:b3:measure:measure}{maat} ervan $\int\lambda_1(\text{sectie})\dd x_2 \cdots \dd
x_d = 1$, want elke $x_1$-sectie is een interval van lengte $1$:
dus $c = 1 = \abs{\det}$. Elke inverteerbare matrix is een product
van deze (gauss-eliminatie), en zowel $c$ als $\abs\det$ is
multiplicatief: dus $c(T) = \abs{\det T}$. De integraalformule
volgt met de standaardmachine (indicatoren, \omterm{def:b3:lebesgue:simple}{elementaire functies},
monotone convergentie).
\end{proof}

\begin{theorem}[Variabelensubstitutie]\label{thm:b3:product:changeofvar}
Zij $U, V \subseteq \R^d$ open en $\Phi \colon U \to V$ een
$\mathcal C^1$-diffeomorfisme. Voor elke \omterm{def:b3:lebesgue:measurable}{meetbare} $f \colon V \to
[0, +\infty]$ (of $f \in L^1(V)$) geldt\index{formule voor
variabelensubstitutie}
\[
\int_V f(y)\,\dd y = \int_U
f\bigl(\Phi(x)\bigr)\,\abs{\det D\Phi(x)}\,\dd x .
\]
\end{theorem}

\begin{proof}
Schrijf $J(x) = \abs{\det D\Phi(x)}$. Het hart van het bewijs is
de ongelijkheid
\[
\lambda_d\bigl(\Phi(A)\bigr) \leq \int_A J\,\dd\lambda_d
\qquad\text{voor elke borelverzameling } A \subseteq U;
\tag{$*$}
\]
stap 4 hieronder tilt $(*)$ --- toegepast op zowel $\Phi$ als
$\Phi^{-1}$ --- op tot de gelijkheid van de stelling. Merk op dat
$\Phi^{-1}$ zelf een $\mathcal C^1$-diffeomorfisme is met
jacobiaan $\abs{\det D\Phi^{-1}(y)} = J(\Phi^{-1}y)^{-1}$
(kettingregel op $\Phi\circ\Phi^{-1} = \mathrm{id}$).

\emph{Stap 1: $(*)$ voor kubussen, met een vervormingsfactor.} Leg
een gesloten kubus $Q \subseteq U$ vast met middelpunt $x_0$ en
zijde $2r$ (een bal voor de supnorm). Bewering: voor elke
$\varepsilon > 0$ geldt, als $\Phi$ differentieerbaar is op $Q$
met $\norm{D\Phi(x) - D\Phi(x_0)} \leq \varepsilon$ op $Q$
(\omterm{def:b3:banach:operator}{operatornorm} voor de supnorm), dat
\[
\Phi(Q) \subseteq \Phi(x_0) + D\Phi(x_0)\Bigl(\,\bigl(1 +
\varepsilon\norm{D\Phi(x_0)^{-1}}\bigr)\,(Q - x_0)\Bigr),
\]
want voor $x \in Q$ geeft de middelwaardeongelijkheid toegepast op
$\Phi(x) - \Phi(x_0) - D\Phi(x_0)(x - x_0)$ dat $\norm{\Phi(x) -
\Phi(x_0) - D\Phi(x_0)(x-x_0)}_\infty \leq \varepsilon\norm{x -
x_0}_\infty \leq \varepsilon r$, en trekt $D\Phi(x_0)^{-1}$ dat
gebrek terug tot een vergroting van de kubus met
$\varepsilon\norm{D\Phi(x_0)^{-1}}\,r$. Volgens
\cref{thm:b3:product:linearchange} is dan
\[
\lambda_d(\Phi(Q)) \leq \abs{\det D\Phi(x_0)}\,
\bigl(1 + \varepsilon\,C\bigr)^{d}\,\lambda_d(Q),
\qquad C = \sup_{Q}\norm{D\Phi(\cdot)^{-1}} .
\]

\emph{Stap 2: $(*)$ voor \omterm{def:b3:topology:compact}{compacte} kubussen, door onderverdeling.}
Zij $Q \subseteq U$ een \omterm{def:b3:topology:compact}{compacte} kubus en $\varepsilon > 0$. Op
$Q$ is $D\Phi$ uniform \omterm{def:b3:topology:continuity}{continu} en is $\norm{D\Phi^{-1}}$ begrensd
(\omterm{def:b3:topology:compact}{compactheid}); verdeel $Q$ in $2^{kd}$ deelkubussen $Q_i$ die
klein genoeg zijn opdat de oscillatie van $D\Phi$ op elk ervan
$\leq \varepsilon$ is. Stap 1 op elke deelkubus (met middelpunt
$x_i$) geeft
\[
\lambda_d(\Phi(Q)) \leq \sum_i\lambda_d(\Phi(Q_i))
\leq (1 + C\varepsilon)^d \sum_i \abs{\det
D\Phi(x_i)}\,\lambda_d(Q_i)
\leq (1 + C\varepsilon)^d\Bigl(\int_Q J + \varepsilon'\Bigr),
\]
waarbij de laatste stap geldt omdat $\sum_i\abs{\det
D\Phi(x_i)}\mathbf 1_{Q_i} \to J$ uniform op $Q$ (\omterm{def:b3:topology:continuity}{continuïteit} van
$\det D\Phi$) --- een vergelijking met riemannsommen. Laat
$\varepsilon \to 0$ gaan: $(*)$ geldt voor \omterm{def:b3:topology:compact}{compacte} kubussen.

\emph{Stap 3: $(*)$ voor alle borelverzamelingen $A$.} De
verzamelingsfunctie $A \mapsto \lambda_d(\Phi(A))$ op de
borel-deelverzamelingen van $U$ is een \omterm{def:b3:measure:measure}{maat} ($\Phi$ is een
bijectie op $V$ die borelverzamelingen en aftelbare disjunctheid
bewaart), en $A \mapsto \int_AJ$ ook. Elke open deelverzameling
van $U$ is een aftelbare vereniging van bijna disjuncte dyadische
\omterm{def:b3:topology:compact}{compacte} kubussen (de standaard dyadische ontbinding: neem de
maximale dyadische kubussen binnen de \omterm{def:b3:topology:topology}{open verzameling}), en beide
\omterm{def:b3:measure:measure}{maten} zijn daarover additief (de randen van kubussen zijn
$\lambda_d$-nul, en hun $\Phi$-beelden zijn nul volgens stap 2,
toegepast op dunne kubusoverdekkingen van de zijvlakken): $(*)$
gaat dus van kubussen over op \omterm{def:b3:topology:topology}{open verzamelingen}. Algemene
borelverzameling $A$: put $U$ uit met \omterm{def:b3:topology:compact}{compacte} $K_m \uparrow U$
waarbij $K_m \subseteq \mathring K_{m+1}$, en leg $m$ vast; op
$\mathring K_{m+1}$ is $J$ begrensd door zekere $M_m$. Kies met de
uitwendige regulariteit van $\lambda_d$ (bewijs als in
\cref{thm:b3:measure:regularity}, met blokken) \omterm{def:b3:topology:topology}{open verzamelingen}
$O_n$ met $A \cap K_m \subseteq O_n \subseteq \mathring K_{m+1}$
en $\lambda_d\bigl(O_n \setminus (A\cap K_m)\bigr) \to 0$. Dan is
\[
\lambda_d\bigl(\Phi(A\cap K_m)\bigr) \leq
\lambda_d\bigl(\Phi(O_n)\bigr) \leq \int_{O_n}J
\leq \int_{A\cap K_m}J + M_m\,\lambda_d\bigl(O_n\setminus(A\cap
K_m)\bigr) \xrightarrow[n\to\infty]{} \int_{A\cap K_m}J .
\]
Laat $m \to \infty$ gaan: links de \omterm{def:b3:topology:continuity}{continuïteit} van onderen,
rechts de monotone convergentie. Daarmee is $(*)$ vastgesteld.

\emph{Stap 4: gelijkheid en de integraalformule.} Breid $(*)$
eerst uit van verzamelingen naar integralen: voor elke \omterm{def:b3:lebesgue:measurable}{meetbare} $g
\geq 0$ op $V$ is
\[
\int_V g(y)\,\dd y \leq \int_U g(\Phi(x))\,J(x)\,\dd x .
\tag{$**$}
\]
Immers, voor $g = \mathbf 1_B$ is dat $(*)$ met $A =
\Phi^{-1}(B)$; de lineariteit breidt het uit tot elementaire $g$,
en de monotone convergentie tot alle $g \geq 0$ (de
standaardmachine). Pas nu $(**)$ tweemaal toe: eerst op $g$, en
daarna --- voor het diffeomorfisme $\Phi^{-1}$ --- op de functie
$x \mapsto g(\Phi(x))J(x)$:
\[
\int_Vg \leq \int_U g(\Phi(x))J(x)\dd x
\leq \int_V g(y)\,J(\Phi^{-1}y)\,\abs{\det
D\Phi^{-1}(y)}\,\dd y = \int_V g ,
\]
want $J(\Phi^{-1}y)\abs{\det D\Phi^{-1}(y)} = \abs{\det\bigl(
D\Phi(\Phi^{-1}y)\,D\Phi^{-1}(y)\bigr)} = 1$ (kettingregel op
$\Phi\circ\Phi^{-1} = \mathrm{id}$). Alle ongelijkheden zijn dus
gelijkheden: de formule geldt voor $g \geq 0$, en voor
$L^1$-functies door ontbinding.
\end{proof}

\begin{example}[Poolcoördinaten; opnieuw de gaussische integraal]
\label{ex:b3:product:polar}
$\Phi(r, \theta) = (r\cos\theta, r\sin\theta)$ is een $\mathcal
C^1$-diffeomorfisme van $\intoo0{+\infty}\times\intoo0{2\pi}$ op
$\R^2$ zonder een halfrechte (een nulverzameling), met $\det D\Phi
= r$:\index{poolcoördinaten}
\[
\int_{\R^2}f(x, y)\,\dd x\,\dd y
= \int_0^{2\pi}\!\!\int_0^{+\infty}
f(r\cos\theta, r\sin\theta)\,r\,\dd r\,\dd\theta .
\]
Voor $f = \eu^{-x^2-y^2}$ geven Tonelli en deze formule
\[
G^2 = \Bigl(\int_\R \eu^{-x^2}\dd x\Bigr)^2
= \int_{\R^2}\eu^{-x^2-y^2}
= 2\pi\int_0^\infty r\eu^{-r^2}\dd r = \pi:
\]
het klassieke bewijs van twee regels voor $G = \sqrt\pi$, nu
\omterm{def:b3:complete:complete}{volledig} verantwoord (vergelijk het parameterbewijs van
\cref{pb:b3:lebesgue:1}).
\end{example}

\begin{theorem}[Volume van de eenheidsbal]
\label{thm:b3:product:ballvolume}
Zij $v_d = \lambda_d(B(0,1))$ in $\R^d$. Dan is\index{volume van
de eenheidsbal}
\[
v_d = \frac{\pi^{d/2}}{\Gamma\bigl(\frac d2 + 1\bigr)} :
\qquad
v_1 = 2,\quad v_2 = \pi,\quad v_3 = \tfrac{4\pi}3,\quad
v_4 = \tfrac{\pi^2}2,\ \dots
\]
\end{theorem}

\begin{proof}
Bereken $I = \int_{\R^d}\eu^{-\norm x_2^2}\dd\lambda_d$ op twee
manieren. Volgens Tonelli valt hij uiteen: $I = G^d = \pi^{d/2}$.
En volgens de laagjesformule (\cref{prop:b3:product:layercake})
met $f = \eu^{-\norm x^2}$, waarvan de niveauverzamelingen ballen
zijn: $\{f > t\} = B\bigl(0, \sqrt{-\ln t}\bigr)$ voor $0 < t <
1$, van \omterm{def:b3:measure:measure}{maat} $v_d(-\ln t)^{d/2}$ (uitrekken met $\rho$ schaalt
$\lambda_d$ met $\rho^d$: \cref{thm:b3:product:linearchange}), dus
\[
I = \int_0^1 v_d\,(-\ln t)^{d/2}\,\dd t
\overset{t = \eu^{-s}}{=}
v_d\int_0^\infty s^{d/2}\eu^{-s}\,\dd s
= v_d\,\Gamma\Bigl(\frac d2 + 1\Bigr).
\]
Stel gelijk. (De waarden: $\Gamma(\frac32) = \frac{\sqrt\pi}2$,
$\Gamma(2) = 1$, enzovoort.) Merk op dat $v_d \to 0$ als $d \to
\infty$ --- de weekendopgave kwantificeert hoe snel, via
Stirling.
\end{proof}

bereken de binnenste integraal in gesloten vorm en laat $A \to

\begin{exercise}[$\star\star\star$]\label{exo:b3:product:8}
(Bèta en gamma) Zij voor $p, q > 0$ de waarde $B(p, q) =
\int_0^1t^{p-1}(1 - t)^{q-1}\dd t$. Vertrek van
$\Gamma(p)\Gamma(q)$ als dubbele integraal, substitueer $(x, y) =

$\intoo0\infty\times\intoo01$; bereken de jacobiaan $= u$) en
besluit\index{bètafunctie}
\[
B(p, q) = \frac{\Gamma(p)\,\Gamma(q)}{\Gamma(p + q)} .
\]
Leid af dat $\int_0^{\pi/2}\sin^{2p-1}\theta\cos^{2q-1}\theta\,

wallisintegralen $W_n = \int_0^{\pi/2}\sin^n$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:9}

\[
\int_Y g\,\dd(T_*\mu) = \int_X g\circ T\,\dd\mu
\]
(de standaardmachine). Vergelijk dat vervolgens met
\cref{thm:b3:product:linearchange}: welke extra informatie draagt
de \omterm{thm:b3:product:changeofvar}{formule voor variabelensubstitutie} die de abstracte
overdrachtsformule niet draagt? \emph{(De overdrachtsformule
identificeert $T_*\mu$ nooit; de substitutiestelling berekent
$T_*\lambda_d$ expliciet als een \omterm{def:b3:measure:measure}{maat} met \omterm{ex:b3:lebesgue:gamma}{dichtheid}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:product:10}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:11}

\]

(c) Leid een bewijs van twee regels af dat de sfeer $S^{d-1}$

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:product:12}
(Een beroemde dubbele integraal) Bewijs met de meetkundige reeks
en Tonelli op $\intoo01^2$ dat
\[
\int_0^1\!\!\int_0^1\frac{\dd x\,\dd y}{1 - xy}
= \sum_{n\geq1}\frac1{n^2} = \zeta(2),
\qquad
\int_0^1\!\!\int_0^1\frac{\dd x\,\dd y}{1 + xy}

Met $\zeta(2) = \frac{\pi^2}6$ (\cref{ch:b3:spectral}) leveren

\end{exercise}

\section{Probleem: de formule van Stirling}

\begin{problem}[{Weekendopgave --- $n! \sim
\sqrt{2\pi n}\,(n/\eu)^n$, met gedomineerde convergentie}]
\label{pb:b3:product:1}
De formule van Stirling beheerst elke asymptotische telling in dit
boek --- balvolumes, binomiaalcoëfficiënten, de lokale vorm van de
centrale limietstelling. We bewijzen haar uit de
$\Gamma$-integraal (\cref{ex:b3:lebesgue:gamma}) met de
\emph{methode van Laplace}, in haar netste vorm met gedomineerde
convergentie, en oogsten daarna de dividenden.\index{formule van
Stirling}

\medskip
\noindent\textbf{Deel I --- De formule.} Voor $t > 0$ is
$\Gamma(t + 1) = \int_0^\infty x^{t}\eu^{-x}\dd x$.
\begin{enumerate}
  \item Substitueer $x = t + \sqrt t\,u$ en toon aan dat
        \[
        \frac{\Gamma(t+1)}{t^{t}\eu^{-t}\sqrt t}
        = \int_{-\sqrt t}^{+\infty}
        \exp\Bigl(t\ln\Bigl(1 + \frac u{\sqrt
        t}\Bigr) - \sqrt t\,u\Bigr)\,\dd u
        \;=\;\int_\R g_t(u)\,\dd u,
        \]
        met $g_t(u) = \exp\bigl(t\ln(1 + u/\sqrt t) - \sqrt
        t\,u\bigr)\mathbf 1_{u > -\sqrt t}$.
  \item Toon de puntsgewijze limiet aan: voor elke vaste $u$ is
        $g_t(u) \to \eu^{-u^2/2}$ als $t \to +\infty$
        \emph{(ontwikkel $\ln(1 + h)$ tot de tweede orde)}.
  \item Dominatie. Zij $\varphi(h) = \ln(1 + h) - h$, zodat
        $g_t(u) = \exp\bigl(t\,\varphi(u/\sqrt t)\bigr)$ voor $u >
        -\sqrt t$. Bewijs de twee grenzen
        \[
        \varphi(h) \leq -\frac{h^2}4 \quad (-1 < h \leq 1),
        \qquad
        \varphi(h) \leq -c\,h \quad (h \geq 1),\ \ c = 1 -
        \ln 2 > 0
        \]
        \emph{(bestudeer $\varphi(h) + \frac{h^2}4$ en $\varphi(h)
        + ch$: bereken de afgeleiden en controleer het teken op
        elk bereik)}. Leid daaruit af, voor $t \geq 1$:
        \[
        g_t(u) \leq \eu^{-u^2/4}\ \ (\abs u \leq \sqrt t),
        \qquad
        g_t(u) \leq \eu^{-cu}\ \ (u \geq \sqrt t),
        \]
        zodat $g_t(u) \leq \eu^{-u^2/4} + \eu^{-cu}\,\mathbf 1_{u
        > 0}$: een \omterm{def:b3:lebesgue:l1}{integreerbare} dominant die niet van $t \geq 1$
        afhangt.
  \item Besluit met de stelling van de gedomineerde convergentie
        en de gaussische integraal (\cref{ex:b3:product:polar}):
        \[
        \Gamma(t + 1) \;\sim\;
        \sqrt{2\pi t}\;\Bigl(\frac t\eu\Bigr)^{t}
        \qquad (t \to +\infty),
        \]
        en in het bijzonder $n! \sim \sqrt{2\pi n}\,(n/\eu)^n$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Dividenden.}
\begin{enumerate}[resume]
  \item (Wallis) Uit \cref{exo:b3:product:8} komen formules van
        het type $W_{2n} = \frac\pi2\binom{2n}n4^{-n}$: leid
        $\binom{2n}{n} \sim \frac{4^n}{\sqrt{\pi n}}$ uit Stirling
        af, en controleer het tegen de recursie $W_{n} =
        \frac{n-1}nW_{n-2}$.
  \item (Balvolumes storten in) Toon aan dat
        \[
        v_d = \frac{\pi^{d/2}}{\Gamma(\frac d2 + 1)}
        \;\sim\; \frac{1}{\sqrt{\pi d}}
        \Bigl(\frac{2\pi\eu}{d}\Bigr)^{d/2},
        \]
        zodat $v_d \to 0$ sneller dan elke meetkundige rij; en
        zoek de dimensie die $v_d$ maximaliseert (numeriek: $d =
        5$).
  \item (Concentratie van de binomiale wet --- een voorproefje van
        \cref{ch:b3:clt}) Toon met Stirling de lokale schatting
        aan, voor $k = n/2 + s\sqrt n/2$ met $s$ vast en $n$ even:
        \[
        2^{-n}\binom{n}{k} \;\sim\;
        \sqrt{\frac{2}{\pi n}}\;\eu^{-s^2/2},
        \]
        het discrete gaussische profiel: de Moivre--Laplace in de
        kiem.
  \item Waar precies gebruikte het bewijs van Deel I: (i) de
        monotone of de gedomineerde convergentie; (ii) de
        gaussische integraal; (iii) de invariantie-eigenschappen
        van de \omterm{def:b3:measure:lebesgueouter}{lebesguemaat}? Telkens één zin.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De foutterm: Stirling met marges.}
Stel $d_n = \ln n! - \bigl(n + \tfrac12\bigr)\ln n + n -
\ln\sqrt{2\pi}$, zodat Deel I zegt dat $d_n \to 0$.
\begin{enumerate}[resume]
  \item Toon aan dat $d_n - d_{n+1} = \bigl(n +
        \tfrac12\bigr)\ln\bigl(1 + \tfrac1n\bigr) - 1$.
  \item Ga met $t = \frac1{2n+1}$ na dat $\frac{n+1}n =
        \frac{1+t}{1-t}$, ontwikkel:
        \[
        d_n - d_{n+1} = \frac{t^2}3 + \frac{t^4}5 +
        \frac{t^6}7 + \cdots,
        \]
        en leid de tweezijdige grenzen af
        \[
        \frac1{3(2n+1)^2} \;<\; d_n - d_{n+1} \;<\;
        \frac1{12n} - \frac1{12(n+1)} .
        \]
  \item Telescopeer (met $d_m \to 0$) en ga de aangename
        \omterm{def:b3:galois:algebraic}{algebraïsche} identiteit $\frac1{3(2m+1)^2} >
        \frac1{12m+1} - \frac1{12(m+1)+1}$ voor $m \geq 1$ na, om
        de klassieke insluiting te verkrijgen:
        \[
        \sqrt{2\pi n}\Bigl(\frac n\eu\Bigr)^n
        \eu^{1/(12n+1)} \;<\; n! \;<\;
        \sqrt{2\pi n}\Bigl(\frac n\eu\Bigr)^n\eu^{1/(12n)} .
        \]
  \item Twee gevolgen: (a) de relatieve fout van de formule van
        Stirling is $< 10^{-6}$ zodra $n \geq 83\,334$; (b) schat
        $100!$ met de hand tot op vier significante cijfers uit de
        insluiting ($100! \approx 9.3326\cdot 10^{157}$), en sta
        even stil bij de nauwkeurigheid van een asymptotische
        formule bij een zeer eindige $n$.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De route van Wallis: Stirling zonder
de gaussische integraal.} Historisch kwam de constante
$\sqrt{2\pi}$ van Wallis en niet van Gauss; dit deel bewijst
Stirling opnieuw, onafhankelijk van de Delen I--II, en bewijst
daarmee ook de gaussische integraal opnieuw. Zij $W_n =
\int_0^{\pi/2}\sin^n\theta\, \dd\theta$.
\begin{enumerate}[resume]
  \item Stel $W_n = \frac{n-1}nW_{n-2}$ vast (partiële
        integratie), evenals de gesloten vormen
        \[
        W_{2n} = \frac\pi2\binom{2n}n4^{-n}, \qquad
        W_{2n+1} = \frac{4^n}{(2n+1)\binom{2n}n},
        \]
        en de identiteit $W_nW_{n-1} = \frac\pi{2n}$.
  \item Leid uit de monotonie van $(W_n)$ af dat
        $W_{2n}/W_{2n+1} \to 1$, en daarna
        \[
        W_{2n} \sim \frac12\sqrt{\frac\pi n}
        \qquad\text{en}\qquad
        \binom{2n}n4^{-n}\sqrt n \longrightarrow
        \frac1{\sqrt\pi} :
        \]
        de stelling van Wallis, verkregen zonder Stirling.
  \item Toon met uitsluitend het telescoperen uit Deel III (zonder
        de waarde van de constante) aan dat $e_n = \ln n! - (n +
        \frac12)\ln n + n$ naar een zekere limiet $\ell$
        convergeert; gelijkwaardig: $n! \sim K\,n^{n+1/2}\eu^{-n}$
        met $K = \eu^\ell > 0$ nog onbepaald.
  \item Vul die asymptotiek in $\binom{2n}n4^{-n}\sqrt n$ in en
        bepaal met vraag 14 de enige mogelijke waarde: $K =
        \sqrt{2\pi}$. Stel de logica samen: de Delen III--IV geven
        samen een \omterm{def:b3:complete:complete}{volledig} tweede bewijs van Stirling --- en
        daarmee, door de substitutie van Deel I achterstevoren te
        lopen, een onafhankelijke berekening van
        $\int_\R\eu^{-u^2/2}\dd u = \sqrt{2\pi}$. Twee pijlers,
        die elkaar allebei kunnen dragen.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Laatste dividenden.}
\begin{enumerate}[resume]
  \item (Het volledige lokale profiel) Toon voor gehele $\abs j
        \leq K\sqrt n$ (met $K$ vast) aan dat
        \[
        \frac{\binom{2n}{n+j}}{\binom{2n}{n}} =
        \prod_{i=1}^{\abs j}\frac{n - i + 1}{n + i}
        = \exp\Bigl(-\frac{j^2}n +
        O\Bigl(\frac1{\sqrt n}\Bigr)\Bigr),
        \]
        uniform in $j$ \emph{(neem logaritmen en gebruik $\ln
        \frac{1-x}{1+y} = -(x + y) + O(x^2 + y^2)$)}. Dat is de
        tweezijdige versie van vraag 7 en de precieze schatting
        die in de weekendopgave van \cref{ch:b3:clt} wordt
        aangehaald.
  \item (Een voorproefje van Poisson) Toon met Stirling aan dat
        $\eu^{-n}\dfrac{n^n}{n!} \sim \dfrac1{\sqrt{2\pi n}}$: de
        modus van een poissonwet met groot gemiddelde $n$ draagt
        massa $\approx (2\pi n)^{-1/2}$, precies zoals de centrale
        limietstelling zal voorspellen.
  \item (Verhoudingen van gammawaarden) Bewijs voor $a \in
        \intoo01$ dat $\dfrac{\Gamma(n + a)}{\Gamma(n)\,n^a} \to
        1$, met de hellingsgrenzen uit de logaritmische
        convexiteit van \cref{pb:b3:lebesgue:1} (vraag 14 daar),
        en breid uit tot elke reële $a > 0$ met de
        functionaalvergelijking. (Dat is wat ``$\Gamma(t+1) \sim$
        Stirling'' tussen de gehele getallen betekent.)
  \item (Ballen, nogmaals) Uit $v_d = \pi^{d/2}/\Gamma(\frac d2 +
        1)$: tabelleer $v_1, \dots, v_7$ exact, ga de
        unimodaliteit na via $\frac{v_d}{v_{d-2}} = \frac{2\pi}d$
        (stijgend zolang $d < 2\pi$, daarna dalend), en bewijs de
        treffende genererende identiteit
        \[
        \sum_{k\geq0}v_{2k}\,x^{2k} = \eu^{\pi x^2} :
        \]
        alle volumes van eenheidsballen in even dimensie, in één
        exponentiële functie verpakt.
  \item (Entropie-asymptotiek) Leid voor vaste $\alpha \in
        \intoo01$ met $\alpha n \in \N$ uit Stirling af dat
        \[
        \binom{n}{\alpha n} \;\sim\;
        \frac{\eu^{n\,H(\alpha)}}
        {\sqrt{2\pi\,\alpha(1-\alpha)\,n}},
        \qquad
        H(\alpha) = -\alpha\ln\alpha -
        (1-\alpha)\ln(1-\alpha) :
        \]
        de exponentiële groeisnelheid van binomiaalcoëfficiënten
        is de \emph{entropie} $H$ --- ga na dat $\alpha =
        \frac12$ vraag 5 teruggeeft, en dat $H(\alpha) < \ln2$
        voor $\alpha \neq \frac12$ (zodat binomiaalcoëfficiënten
        buiten het midden exponentieel verwaarloosbaar zijn ten
        opzichte van $2^n$).
  \item (Oppervlakten) De oppervlakte van de eenheidssfeer
        $S^{d-1}$ is $s_{d-1} = d\,v_d$ (bewezen als
        \cref{exo:b3:forms:6} in het hoofdstuk over
        differentiaalvormen; neem het hier als definitie).
        Tabelleer $s_0, \dots, s_6$, lokaliseer de grootste ($d -
        1 = 6$, $s_6 = \frac{16\pi^3}{15} \approx 33.07$), en toon
        aan dat ook $s_{d-1} \to 0$ supermeetkundig --- naar elke
        euclidische maatstaf zijn hoogdimensionale sferen
        verdwijnend klein.
  \item (De eerste correctieterm) Leid uit de insluiting van vraag
        11 af dat $d_n = \frac1{12n} + O\bigl(\frac1{n^2}\bigr)$,
        en dus
        \[
        n! = \sqrt{2\pi n}\,\Bigl(\frac
        n\eu\Bigr)^{n}\Bigl(1 + \frac1{12n} +
        O\Bigl(\frac1{n^2}\Bigr)\Bigr).
        \]
        Ga het na bij $n = 10$: de kale formule geeft
        $3\,598\,696$ (relatieve fout $8.3\cdot10^{-3}$), de
        gecorrigeerde $3\,628\,685$ tegenover $10! =
        3\,628\,800$ (relatieve fout $3.2\cdot10^{-5}$) --- één
        term van de reeks koopt tweeënhalf cijfer.
  \item (De mediaan van $\Gamma$) Toon aan dat
        \[
        \frac{1}{\Gamma(t+1)}
        \int_0^{t} x^{t}\eu^{-x}\,\dd x
        \;\longrightarrow\; \frac12
        \qquad (t \to +\infty) :
        \]
        asymptotisch ligt precies de helft van de massa van de
        $\Gamma$-integrand onder haar modus $x = t$. \emph{(Voer
        de substitutie van Deel I uit op de afgeknotte integraal;
        de dominant van vraag 3 ligt al klaar.)}
  \item (Entropie, niet-asymptotisch) Bewijs voor $\alpha \in
        \intoc0{\frac12}$ de grens, geldig voor \emph{elke} $n
        \geq 1$:
        \[
        \sum_{k=0}^{\lfloor\alpha n\rfloor}\binom nk
        \;\leq\; \eu^{n\,H(\alpha)} ,
        \]
        door de som met $\sum_k\binom nk\lambda^{k-\alpha n}$ te
        vergelijken voor de kanteling $\lambda =
        \frac{\alpha}{1-\alpha} \leq 1$. Ga na dat die keuze van
        $\lambda$ optimaal is, en breng het in overeenstemming met
        vraag 21: de exponentiële snelheid $H(\alpha)$ van de
        asymptotische uitspraak wordt bereikt door een
        ongelijkheid van één regel, geheel zonder asymptotiek.
\end{enumerate}
\end{problem}
