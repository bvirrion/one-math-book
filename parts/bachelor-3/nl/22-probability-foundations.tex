\chapter{Kansrekening: fundamenten en de wet van de grote
aantallen}\label{ch:b3:probability}

Het volume van bachelorjaar 2 bouwde de kansrekening op
aftelbare ruimten; de maattheorie neemt nu elke beperking weg.
Een \omterm{def:b3:probability:space}{kansruimte} is een maatruimte met totale massa $1$,
\omterm{def:b3:probability:space}{toevalsveranderlijken} zijn \omterm{def:b3:lebesgue:measurable}{meetbare} afbeeldingen, de
\omterm{def:b3:probability:space}{verwachtingswaarde} is de lebesgue-integraal --- en meteen is het
hele analytische arsenaal
(\cref{ch:b3:measure,ch:b3:lebesgue,ch:b3:product}) op het toeval
van toepassing. Dit hoofdstuk installeert het woordenboek,
construeert oneindige rijen \omterm{def:b3:probability:independence}{onafhankelijke} \omterm{def:b3:probability:space}{toevalsveranderlijken}
(op $\intcc01$, uit binaire cijfers: het toeval zit verstopt in
de \omterm{def:b3:measure:lebesgueouter}{lebesguemaat}), bewijst de lemma's van Borel--Cantelli en de
\omterm{thm:b3:probability:zeroone}{nul-een-wet} van Kolmogorov, sorteert de convergentiewijzen uit,
en bewijst de wet van de grote aantallen --- de stelling die
frequenties naar kansen laat convergeren en de statistiek
mogelijk maakt. De weekendopgave geeft het bewijs van Etemadi van
de sterke wet in haar definitieve $L^1$-vorm.

\section{Het woordenboek}

\begin{definition}\label{def:b3:probability:space}
Een \emph{kansruimte}\index{kansruimte} is een maatruimte
$(\Omega, \mathcal A, \P)$ met $\P(\Omega) = 1$; de elementen van
$\mathcal A$ heten \emph{gebeurtenissen}, en een eigenschap geldt
\emph{bijna zeker} (b.z.) als haar gebeurtenis kans $1$ heeft.
Een \emph{toevalsveranderlijke}\index{toevalsveranderlijke} is
een \omterm{def:b3:lebesgue:measurable}{meetbare} afbeelding $X \colon \Omega \to \R$ (of $\R^d$: een
toevalsvector); haar \emph{verdeling}\index{verdeling van een
toevalsveranderlijke} is de beeldkansmaat $\P_X = X_*\P$ op $\R$
(\cref{exo:b3:product:9}), vastgelegd door de
\emph{verdelingsfunctie} $F_X(t) = \P(X \leq t)$
(\cref{exo:b3:measure:3}). $X$ heeft \emph{\omterm{ex:b3:lebesgue:gamma}{dichtheid}} $f$ als
$\P_X = f\,\dd\lambda$; ze heet \emph{discreet} als $\P_X$ een
aftelbare combinatie van diracmassa's is. De
\emph{verwachtingswaarde}\index{verwachtingswaarde} is
\[
\E[X] = \int_\Omega X\,\dd\P
\qquad (X \geq 0 \text{ of } X \in L^1(\P)),
\]
en de \emph{overdrachtsformule} (\cref{exo:b3:product:9})
berekent haar in de verdeling: $\E[g(X)] = \int_\R g\,\dd\P_X$
--- $= \sum g(x_k)p_k$ in het discrete geval, $= \int
g(x)f(x)\dd x$ in het geval met een \omterm{ex:b3:lebesgue:gamma}{dichtheid}: de formules uit
het volume van bachelorjaar 2, nu stellingen van één theorie. De
\emph{variantie} is $\V(X) = \E[(X - \E X)^2] = \E[X^2] - (\E
X)^2$ voor $X \in L^2$.
\end{definition}

\begin{example}\label{ex:b3:probability:laws}
De standaardverdelingen en hun opmerkelijke transformaties:
bernoulli $\mathcal B(p)$, binomiaal $\mathcal B(n, p)$,
meetkundig, poisson $\mathcal P(\lambda)$ (discreet: de tabellen
uit het volume van bachelorjaar 2 blijven geldig); uniform op
$\intcc01$ (de \omterm{def:b3:measure:lebesgueouter}{lebesguemaat} zelf); exponentieel $\mathcal
E(\lambda)$ (\omterm{ex:b3:lebesgue:gamma}{dichtheid} $\lambda\eu^{-\lambda x}\mathbf
1_{x>0}$); de \emph{gaussische verdeling}\index{gaussische
verdeling} $\mathcal N(m, \sigma^2)$ met \omterm{ex:b3:lebesgue:gamma}{dichtheid}
$\frac1{\sigma\sqrt{2\pi}}\exp\bigl(-\frac{(x -
m)^2}{2\sigma^2}\bigr)$ --- een kansdichtheid volgens
\cref{pb:b3:lebesgue:1}, met gemiddelde $m$ en variantie
$\sigma^2$ (de gaussische momenten,
\cref{exo:b3:product:10}).
\end{example}

\begin{proposition}[Markov en Chebyshev]
\label{prop:b3:probability:markov}
Voor $X \geq 0$ en $a > 0$ is $\P(X \geq a) \leq \frac{\E X}{a}$;
voor $X \in L^2$ is $\P\bigl(\abs{X - \E X} \geq a\bigr) \leq
\frac{\V(X)}{a^2}$.\index{ongelijkheid van
Markov}\index{ongelijkheid van Chebyshev}
\end{proposition}

\begin{proof}
\cref{exo:b3:lebesgue:5}(a); Chebyshev is Markov toegepast op $(X
- \E X)^2$.
\end{proof}

\section{Onafhankelijkheid}

\begin{definition}\label{def:b3:probability:independence}
Deel-$\sigma$-algebra's $\mathcal A_1, \dots, \mathcal A_n
\subseteq \mathcal A$ heten
\emph{onafhankelijk}\index{onafhankelijkheid} als
$\P(A_1\cap\dots\cap A_n) = \prod\P(A_i)$ voor alle $A_i \in
\mathcal A_i$; gebeurtenissen zijn onafhankelijk als de
$\sigma$-algebra's $\{\varnothing, A_i, A_i^c, \Omega\}$ dat
zijn; en \omterm{def:b3:probability:space}{toevalsveranderlijken} $X_1, \dots, X_n$ als de
$\sigma$-algebra's $\sigma(X_i) = X_i^{-1}(\mathcal B(\R))$ dat
zijn. Een oneindige familie is onafhankelijk als elke eindige
deelfamilie dat is.
\end{definition}

\begin{theorem}\label{thm:b3:probability:independence}
$X_1, \dots, X_n$ zijn \omterm{def:b3:probability:independence}{onafhankelijk} dan en slechts dan als de
\omterm{def:b3:probability:space}{verdeling} van de vector $(X_1, \dots, X_n)$ de \omterm{thm:b3:product:existence}{productmaat}
$\P_{X_1}\otimes\cdots\otimes\P_{X_n}$ is. In dat geval is voor
$g_i \geq 0$ (of zodanig dat de producten \omterm{def:b3:lebesgue:l1}{integreerbaar} zijn)
\[
\E\Bigl[\prod_ig_i(X_i)\Bigr] = \prod_i\E[g_i(X_i)],
\]
in het bijzonder $\E[XY] = \E X\,\E Y$ en $\V(X_1 + \dots + X_n)
= \sum\V(X_i)$ voor \omterm{def:b3:probability:independence}{onafhankelijke} veranderlijken in $L^2$.
\end{theorem}

\begin{proof}
Zijn de $X_i$ \omterm{def:b3:probability:independence}{onafhankelijk}, dan vallen de twee kansmaten
$\P_{(X_1,\dots,X_n)}$ en $\bigotimes\P_{X_i}$ samen op alle
producten $B_1\times\dots\times B_n$ van borelverzamelingen ---
een $\pi$-systeem dat $\mathcal B(\R^n)$ voortbrengt
(\cref{prop:b3:product:sections}(b)) --- en dus overal
(\cref{thm:b3:measure:uniqueness}). Omgekeerd ontbindt een
productverdeling alle gebeurtenissen $\bigcap_iX_i^{-1}(B_i)$:
\omterm{def:b3:probability:independence}{onafhankelijkheid}. De formule voor de \omterm{def:b3:probability:space}{verwachtingswaarde} is dan
Tonelli of Fubini (\cref{thm:b3:product:tonelli}) via de
overdrachtsformule; $\E[XY] = \E X\E Y$ is het geval $g_i =
\mathrm{id}$, en het kwadraat ontwikkelen geeft de additiviteit
van de varianties (de kruistermen $\E[(X_i - \E X_i)(X_j - \E
X_j)] = 0$).
\end{proof}

\begin{theorem}[Bestaan van onafhankelijke rijen]
\label{thm:b3:probability:existence}
Op $\bigl(\intcc01, \mathcal L, \lambda\bigr)$ bestaat er een rij
$(U_n)_{n\geq1}$ \omterm{def:b3:probability:independence}{onafhankelijke} \omterm{def:b3:probability:space}{toevalsveranderlijken}, elk
uniform op $\intcc01$. Bijgevolg bestaan er voor elke
voorgeschreven \omterm{def:b3:probability:space}{verdelingen} $(\mu_n)$ op $\R$ \omterm{def:b3:probability:independence}{onafhankelijke}
$(X_n)$ met $\P_{X_n} = \mu_n$.\index{onafhankelijke rijen,
bestaan}
\end{theorem}

\begin{proof}
\emph{Cijfers.} Zij voor $\omega \in \intcc01$ $(b_k(\omega))$
haar binaire cijfers ($\omega = \sum b_k2^{-k}$; kies de
ontwikkeling die niet op louter enen eindigt --- de
dubbelzinnigheid betreft slechts een aftelbare, dus
verwaarloosbare verzameling). Elke $b_k$ is een
\omterm{def:b3:probability:space}{toevalsveranderlijke} ($\{b_k = 1\}$ is een eindige vereniging van
dyadische intervallen) en de vector $(b_1, \dots, b_m)$ neemt
elke waarde in $\{0,1\}^m$ aan op een dyadisch interval van
lengte $2^{-m}$: de $b_k$ zijn \omterm{def:b3:probability:independence}{onafhankelijk} en
bernoulli$(\frac12)$.

\emph{Hergroeperen.} Splits $\N^*$ in oneindig veel disjuncte
oneindige verzamelingen $(I_n)$ (bijvoorbeeld met priemmachten,
of met diagonalen); zij $(k^n_j)_j$ een opsomming van $I_n$ en
stel
\[
U_n = \sum_{j\geq1} b_{k^n_j}\,2^{-j} .
\]
Elke $U_n$ is uniform: haar binaire cijfers zijn \omterm{def:b3:probability:independence}{onafhankelijke}
eerlijke bits, dus $\P(U_n \in [l2^{-m}, (l+1)2^{-m})) = 2^{-m}$
voor elk dyadisch interval, en de dyadische intervallen leggen de
\omterm{def:b3:probability:space}{verdeling} vast (\cref{thm:b3:measure:uniqueness}). De $U_n$ zijn
\omterm{def:b3:probability:independence}{onafhankelijk}: het zijn functies van disjuncte blokken van de
\omterm{def:b3:probability:independence}{onafhankelijke} familie $(b_k)$ --- formeel hangen de
gebeurtenissen $\{U_n \in D_n\}$ voor dyadische $D_n$ af van
eindig veel cijfers uit disjuncte verzamelingen, en ontbinden ze;
het argument met het $\pi$-systeem tilt dat op tot alle
borelverzamelingen.

\emph{Willekeurige \omterm{def:b3:probability:space}{verdelingen}.} Zij $G_n(u) = \inf\{t :
F_{\mu_n}(t) \geq u\}$ (de \emph{kwantielfunctie} van de
verdelingsfunctie $F_{\mu_n}$); de sleutelgelijkwaardigheid
$G_n(u) \leq t \iff u \leq F_{\mu_n}(t)$ (rechtscontinuïteit van
$F$, monotonie) toont dat $X_n = G_n(U_n)$ \omterm{def:b3:lebesgue:measurable}{meetbaar} is met
$\P(X_n \leq t) = \P(U_n \leq F_{\mu_n}(t)) = F_{\mu_n}(t)$: dus
\omterm{def:b3:probability:space}{verdeling} $\mu_n$; de \omterm{def:b3:probability:independence}{onafhankelijkheid} wordt geërfd (functies
van \omterm{def:b3:probability:independence}{onafhankelijke} veranderlijken,
\cref{exo:b3:probability:3}).
\end{proof}

\begin{example}[Het verjaardagsprobleem, eerlijk]
\label{ex:b3:probability:birthday}
Onder $n$ mensen met \omterm{def:b3:probability:independence}{onafhankelijke}, uniform over $N = 365$ dagen
verdeelde verjaardagen is de kans dat alle verjaardagen
verschillen
\[
p_n = \prod_{k=1}^{n-1}\Bigl(1 - \frac kN\Bigr),
\]
door herhaald te conditioneren (of rechtstreeks: de gunstige
$N(N-1)\cdots(N - n + 1)$ op het totaal $N^n$, een telargument
dat de productformule van de \omterm{def:b3:probability:independence}{onafhankelijkheid} streng maakt).
Logaritmen nemen en $-\ln(1 - x) = x + O(x^2)$ gebruiken geeft
\[
\ln p_n = -\frac{n(n-1)}{2N} +
O\Bigl(\frac{n^3}{N^2}\Bigr),
\qquad\text{dus}\qquad
p_n \approx \eu^{-n^2/2N} .
\]
Het kantelpunt $p_n = \frac12$ ligt bij $n \approx
\sqrt{2N\ln2} \approx 1.18\sqrt N$: voor $N = 365$ is $n = 23$
($p_{23} = 0.4927$). Twee moralen. Ten eerste duiken botsingen
tussen $n$ voorwerpen in $N$ dozen op op de schaal $n \sim \sqrt
N$, niet $n \sim N$ --- de \emph{verjaardagsschaling} die de
botsingen van hashfuncties en de kostprijs $\sqrt N$ van
verjaardagsaanvallen in de cryptografie regeert. Ten tweede is de
berekening een sjabloon: de $\binom n2$ gebeurtenissen ``dit paar
botst'' zijn niet \omterm{def:b3:probability:independence}{onafhankelijk}, en toch gedraagt het antwoord
zich alsof ze het wel waren ($\eu^{-\binom n2/N}$ is precies de
heuristiek met \omterm{def:b3:probability:independence}{onafhankelijke} paren) --- een eerste geval van de
poissonbenadering die in de weekendopgave van \cref{ch:b3:clt}
streng wordt gemaakt (de ongelijkheid van Le Cam).
\end{example}

\section{Borel--Cantelli en de nul-een-wet}

\begin{theorem}[Borel--Cantelli]
\label{thm:b3:probability:borelcantelli}
Zij $(A_n)$ gebeurtenissen en $\limsup A_n = \bigcap_N
\bigcup_{n\geq N}A_n$ (``$A_n$ treedt oneindig vaak
op'').\index{lemma's van Borel--Cantelli}
\begin{enumerate}
  \item Is $\sum\P(A_n) < \infty$, dan is $\P(\limsup A_n) = 0$.
  \item Is $\sum\P(A_n) = \infty$ \emph{en zijn de $A_n$
        \omterm{def:b3:probability:independence}{onafhankelijk}}, dan is $\P(\limsup A_n) = 1$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) is \cref{exo:b3:measure:4}. (2): voor $N \leq M$ geeft de
\omterm{def:b3:probability:independence}{onafhankelijkheid} van de complementen
(\cref{exo:b3:probability:3})
\[
\P\Bigl(\bigcap_{n=N}^{M}A_n^c\Bigr) = \prod_{n=N}^M\bigl(1
- \P(A_n)\bigr) \leq
\exp\Bigl(-\sum_{n=N}^M\P(A_n)\Bigr) \xrightarrow[M \to
\infty]{} 0
\]
($1 - x \leq \eu^{-x}$; de reeks divergeert). Dus is
$\P\bigl(\bigcup_{n\geq N}A_n\bigr) = 1$ voor elke $N$, en heeft
de dalende doorsnede over $N$ nog steeds kans $1$ (\omterm{def:b3:topology:continuity}{continuïteit}
van boven, \cref{prop:b3:measure:basics}).
\end{proof}

\begin{theorem}[Nul-een-wet van Kolmogorov]
\label{thm:b3:probability:zeroone}
Zij $(X_n)$ \omterm{def:b3:probability:independence}{onafhankelijk} en $\mathcal T = \bigcap_N\sigma(X_N,
X_{N+1}, \dots)$ de \emph{staart-$\sigma$-algebra}\index{staart-$
\sigma$-algebra} (gebeurtenissen die ongevoelig zijn voor elk
eindig aantal van de $X_n$: de convergentie van $\sum X_n$, van
$\frac{S_n}n$, waarden van $\limsup$'s, \dots). Dan heeft elke $T
\in \mathcal T$ $\P(T) \in \{0, 1\}$.\index{nul-een-wet}
\end{theorem}

\begin{proof}
Leg $N$ vast. De $\sigma$-algebra's $\sigma(X_1, \dots, X_N)$ en
$\sigma(X_{N+1}, \dots)$ zijn \omterm{def:b3:probability:independence}{onafhankelijk}: gebeurtenissen die
van disjuncte blokken afhangen, ontbinden op de voortbrengende
$\pi$-systemen (cilinders $\bigcap_{i\leq N}\{X_i \in B_i\}$
respectievelijk eindige voorwaarden op latere veranderlijken), en
Dynkin (\cref{thm:b3:measure:dynkin}, tweemaal toegepast, telkens
aan één kant) breidt de ontbinding uit. Een staartgebeurtenis $T$
ligt voor elke $N$ in $\sigma(X_{N+1}, \dots)$: dus is $T$
\omterm{def:b3:probability:independence}{onafhankelijk} van elke $\sigma(X_1, \dots, X_N)$, en dus van de
$\sigma$-algebra die ze voortbrengen, $\sigma(X_1, X_2, \dots)$
(nog eens Dynkin: de vereniging van de $\sigma(X_1,\dots,X_N)$ is
een $\pi$-systeem dat haar voortbrengt). Maar $T \in \sigma(X_1,
X_2, \dots)$ eveneens: $T$ is \omterm{def:b3:probability:independence}{onafhankelijk} \emph{van zichzelf},
dus $\P(T) = \P(T\cap T) = \P(T)^2$: $\P(T) \in \{0, 1\}$.
\end{proof}

\section{Convergentiewijzen}

\begin{definition}\label{def:b3:probability:modes}
$X_n \to X$ \emph{bijna zeker} als $\P(X_n \to X) = 1$; \emph{in
kans} als $\P(\abs{X_n - X} \geq \varepsilon) \to 0$ voor elke
$\varepsilon > 0$; \emph{in $L^p$} als $\E\abs{X_n - X}^p \to
0$.\index{convergentie van toevalsveranderlijken}
\end{definition}

\begin{proposition}\label{prop:b3:probability:modes}
(a) b.z.\ convergentie impliceert convergentie in kans;
(b) $L^p$-convergentie impliceert convergentie in kans;
(c) convergentie in kans impliceert b.z.\ convergentie \emph{langs
een deelrij};
(d) geen enkele andere implicatie geldt in het algemeen.
\end{proposition}

\begin{proof}
(a) $\P(\abs{X_n - X} \geq \varepsilon) \leq
\P\bigl(\sup_{m\geq n}\abs{X_m - X} \geq \varepsilon\bigr)
\downarrow \P\bigl(\limsup\{\abs{X_m - X} \geq
\varepsilon\}\bigr) = 0$ bij b.z.\ convergentie (\omterm{def:b3:topology:continuity}{continuïteit} van
boven; de limsup-gebeurtenis sluit convergentie uit). (b) Markov:
$\P(\abs{X_n - X} \geq \varepsilon) \leq
\varepsilon^{-p}\,\E\abs{X_n - X}^p$. (c) Kies $n_k$ met
$\P(\abs{X_{n_k} - X} \geq 2^{-k}) \leq 2^{-k}$;
Borel--Cantelli (1) maakt dat uiteindelijk b.z.\ $\abs{X_{n_k} -
X} < 2^{-k}$. (d) De schrijfmachine (\cref{exo:b3:lp:3}) op
$(\intcc01, \lambda)$ convergeert in $L^1$ en in kans maar
nergens puntsgewijs; $n\mathbf 1_{\intoo0{1/n}} \to 0$ b.z.\ maar
niet in $L^1$; de details en de overige tegenvoorbeelden staan in
\cref{exo:b3:probability:6}.
\end{proof}

\section{De wet van de grote aantallen}

Overal zijn $(X_n)$ \omterm{def:b3:probability:independence}{onafhankelijk} met dezelfde \omterm{def:b3:probability:space}{verdeling}
(\emph{i.i.d.}), en $S_n = X_1 + \dots + X_n$.

\begin{theorem}[Zwakke wet van de grote aantallen]
\label{thm:b3:probability:wlln}
Is $X_1 \in L^2$, met $m = \E X_1$, dan is
\[
\P\Bigl(\Bigl|\frac{S_n}{n} - m\Bigr| \geq
\varepsilon\Bigr) \leq
\frac{\V(X_1)}{n\,\varepsilon^2}
\xrightarrow[n\to\infty]{} 0 :
\]
$\frac{S_n}n \to m$ in kans (en in $L^2$).\index{wet van de grote
aantallen (zwak)}
\end{theorem}

\begin{proof}
$\E\frac{S_n}n = m$ en $\V\bigl(\frac{S_n}n\bigr) =
\frac{n\V(X_1)}{n^2}$
(\cref{thm:b3:probability:independence}); Chebyshev.
\end{proof}

\begin{theorem}[Sterke wet van de grote aantallen]
\label{thm:b3:probability:slln}
Is $X_1 \in L^1$, dan is\index{wet van de grote aantallen
(sterk)}
\[
\frac{S_n}{n} \xrightarrow[n\to\infty]{\text{b.z.}} \E[X_1].
\]
We bewijzen dat hier onder de sterkere hypothese $X_1 \in L^4$;
het algemene geval ($L^1$: het bewijs van Etemadi) is de
weekendopgave.
\end{theorem}

\begin{proof}[Bewijs onder $\E X_1^4 < \infty$]
Na centreren ($X_i \mapsto X_i - m$) mogen we $m = 0$ aannemen.
Ontwikkel:
\[
\E[S_n^4] = \sum_{i,j,k,l}\E[X_iX_jX_kX_l]
= n\,\E[X_1^4] + 3n(n-1)\,\bigl(\E[X_1^2]\bigr)^2 \leq
C\,n^2 ,
\]
want de \omterm{def:b3:probability:independence}{onafhankelijkheid} en het centreren doden elke term met
een geïsoleerde factor ($\E[X_iX_jX_kX_l] = \E[X_i]\E[\cdots] =
0$ tenzij de indices paren vormen: de enige overlevenden zijn de
$n$ termen met $i=j=k=l$ en de $3n(n-1)$ termen met twee
verschillende paren). Markov:
\[
\P\Bigl(\Bigl|\frac{S_n}n\Bigr| \geq \varepsilon\Bigr)
= \P\bigl(S_n^4 \geq n^4\varepsilon^4\bigr)
\leq \frac{Cn^2}{n^4\varepsilon^4} =
\frac{C}{\varepsilon^4n^2},
\]
sommeerbaar: Borel--Cantelli (1) geeft voor elke rationale
$\varepsilon$ dat uiteindelijk b.z.\ $\abs{S_n/n} < \varepsilon$;
doorsnijden over $\varepsilon \in \Q_+^*$ (aftelbaar veel
gebeurtenissen met kans $1$) geeft $S_n/n \to 0$ b.z.
\end{proof}

\begin{example}[Wat de sterke wet oplevert]
\label{ex:b3:probability:sllnapps}
(a) \emph{Frequenties}: voor i.i.d.\ muntworpen convergeert de
waargenomen frequentie van kop b.z.\ naar $p$ --- de empirische
rechtvaardiging van de kansrekening zelf.
(b) \emph{Monte carlo}\index{monte-carlomethode}: voor $g \in
L^1(\intcc01)$ en $(U_n)$ i.i.d.\ uniform
(\cref{thm:b3:probability:existence}) is
$\frac1n\sum_{k\leq n}g(U_k) \to \int_0^1g$ b.z.: integralen door
te bemonsteren, in elke dimensie, met de van de dimensie
\omterm{def:b3:probability:independence}{onafhankelijke} snelheid $\sim n^{-1/2}$ die in \cref{ch:b3:clt}
precies wordt gemaakt.
(c) \emph{Normale getallen}: bijna elk reëel getal heeft in zijn
binaire ontwikkeling asymptotische frequentie $\frac12$ aan enen
(pas de sterke wet toe op de cijferveranderlijken van
\cref{thm:b3:probability:existence}) --- de stelling van Borel,
een uitspraak over de meest \emph{alledaagse} getallen bewezen
met \omterm{def:b3:measure:measure}{maat}: \cref{pb:b3:probability:1} maakt haar in alle bases
af.
\end{example}

\begin{method}\label{met:b3:probability:toolkit}
De werkvolgorde voor asymptotische uitspraken over toevallige
rijen: (1) \emph{Is de gebeurtenis een staartgebeurtenis?} Dan is
haar kans $0$ of $1$ (\cref{thm:b3:probability:zeroone}) en hoef
je enkel te beslissen welke van beide. (2) \emph{Om b.z.\
uitspraken te bewijzen}: Borel--Cantelli --- sommeerbare kansen
voor de ``slechte'' gebeurtenissen, via afschattingen van het
type Markov of Chebyshev op de momenten die bestaan; de
\omterm{def:b3:probability:independence}{onafhankelijkheid} is enkel voor de omgekeerde richting nodig. (3)
\emph{Deelrij plus insluiting}: bewijs de convergentie langs een
hanteerbare deelrij en beheers de oscillatie ertussen met
monotonie of maximaalongelijkheden --- het skelet van het bewijs
van Etemadi. (4) Voor limieten in \omterm{def:b3:probability:space}{verdeling}: wacht op
\cref{ch:b3:clt}.
\end{method}

\begin{exercise}[$\star$]\label{exo:b3:probability:1}
(a) Zij $X$ met een \omterm{def:b3:topology:continuity}{continue} strikt stijgende verdelingsfunctie
$F$. Toon aan dat $F(X)$ uniform is op $\intcc01$, en dat $G(U)
\sim F$ voor $U$ uniform en $G = F^{-1}$: simulatie door
inversie.
(b) Bereken de verdelingsfunctie en de \omterm{ex:b3:lebesgue:gamma}{dichtheid} van $X^2$ voor
$X$ uniform op $\intcc{-1}1$, en van $-\frac1\lambda\ln U$ voor
$U$ uniform op $\intoo01$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:probability:2}
(a) Bereken het gemiddelde en de variantie van de
poissonverdeling $\mathcal P(\lambda)$ en van de meetkundige
\omterm{def:b3:probability:space}{verdeling} met de overdrachtsformule.
(b) Toon aan dat een positieve \omterm{def:b3:probability:space}{toevalsveranderlijke} $T$ met $\P(T
> t) > 0$ voor alle $t$ de \emph{geheugenloosheid} $\P(T > t + s
\mid T > t) = \P(T > s)$ voor alle $s, t \geq 0$ bezit dan en
slechts dan als $T$ exponentieel is. \emph{(De overlevingsfunctie
voldoet aan de functionaalvergelijking van Cauchy; de monotonie
vervangt de \omterm{def:b3:topology:continuity}{continuïteit}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:3}
(a) Toon aan dat als $X_1, \dots, X_n$ \omterm{def:b3:probability:independence}{onafhankelijk} zijn en de
$f_i$ borelfuncties, de $f_i(X_i)$ \omterm{def:b3:probability:independence}{onafhankelijk} zijn.
(b) Toon aan dat gebeurtenissen $A_1, \dots, A_n$ \omterm{def:b3:probability:independence}{onafhankelijk}
zijn dan en slechts dan als hun complementen dat zijn, en dan en
slechts dan als de indicatoren $\mathbf 1_{A_i}$ \omterm{def:b3:probability:independence}{onafhankelijke}
\omterm{def:b3:probability:space}{toevalsveranderlijken} zijn.
(c) (Paarsgewijs is zwakker) Twee eerlijke munten: $A = $ de
eerste is kop, $B = $ de tweede is kop, $C = $ de twee stemmen
overeen. Toon aan dat $A, B, C$ paarsgewijs \omterm{def:b3:probability:independence}{onafhankelijk} zijn
maar niet \omterm{def:b3:probability:independence}{onafhankelijk}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:4}
(a) (De oneindige aap) Een i.i.d.\ rij uniforme toetsaanslagen op
een eindig alfabet bevat b.z.\ elke eindige tekst oneindig vaak:
bewijs dat met Borel--Cantelli (2) op disjuncte blokken.
(b) (Reeksen) Zij voor i.i.d.\ eerlijke bits $R_n$ de lengte van
de reeks enen die op plaats $n$ begint. Toon aan dat b.z.\ $R_n
\geq (1+\varepsilon)\log_2n$ slechts eindig vaak, en $R_n \geq
\log_2 n$ oneindig vaak \emph{(beide helften van
Borel--Cantelli; ga voor de tweede over op disjuncte blokken om
\omterm{def:b3:probability:independence}{onafhankelijkheid} te winnen)}: de langste reeks in de eerste $n$
cijfers groeit als $\log_2n$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:5}
Zij $(X_n)$ \omterm{def:b3:probability:independence}{onafhankelijk}.
(a) Toon aan dat de convergentiestraal van $\sum X_n z^n$ b.z.\
een constante is (mogelijk $0$ of $\infty$).
(b) Toon aan dat $\P(\sum X_n \text{ convergeert}) \in \{0, 1\}$
en $\P(S_n/n \to m) \in \{0,1\}$.
(c) Geef een gebeurtenis over $(X_n)$ die \emph{geen}
staartgebeurtenis is, en ga na dat de \omterm{thm:b3:probability:zeroone}{nul-een-wet} er kan falen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:6}
Geef op $(\intcc01, \lambda)$ --- met bewijzen ---
\omterm{def:b3:probability:space}{toevalsveranderlijken} zodat:
(a) $X_n \to 0$ in kans en in elke $L^p$, maar nergens b.z.;
(b) $X_n \to 0$ b.z.\ maar in geen enkele $L^p$;
(c) $X_n \to 0$ in $L^1$ maar niet in $L^2$;
(d) en toon aan: convergeert $X_n \to X$ in kans met $\abs{X_n}
\leq Y \in L^1$, dan is $X_n \to X$ in $L^1$ \emph{(deelrijen +
gedomineerde convergentie + de truc met de deeldeelrij)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:7}
Een opiniepeiling schat een onbekende verhouding $p$ met de
empirische frequentie $\hat p_n$ van $n$ \omterm{def:b3:probability:independence}{onafhankelijke}
trekkingen.
(a) Chebyshev: toon aan dat $\P(\abs{\hat p_n - p} \geq
\varepsilon) \leq \frac1{4n\varepsilon^2}$ (gebruik $p(1-p) \leq
\frac14$).
(b) Hoeveel trekkingen waarborgen met deze grens een fout $\leq
3\%$ met kans $\geq 95\%$? (Het ware antwoord, via
\cref{ch:b3:clt}, is ongeveer $1070$: Chebyshev is eerlijk maar
grof.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:probability:8}
(Bernstein) Definieer voor $f \in \mathcal C(\intcc01)$ de
bernsteinveelterm $B_nf(x) =
\sum_{k=0}^n\binom nkx^k(1-x)^{n-k}f\bigl(\frac kn\bigr)$.
(a) Herken $B_nf(x) = \E\bigl[f\bigl(\frac {S_n}n\bigr)\bigr]$
voor $S_n$ binomiaal $\mathcal B(n, x)$.
(b) Bewijs dat $B_nf \to f$ \emph{uniform} op $\intcc01$: splits
op $\{\abs{\frac{S_n}n - x} \leq \delta\}$ en haar complement,
met de uniforme \omterm{def:b3:topology:continuity}{continuïteit} en Chebyshev met de uniforme grens
$\V(\frac{S_n}n) \leq \frac1{4n}$.
(c) Besluit: een tweede, kanstheoretisch bewijs van de
benaderingsstelling van Weierstrass
(\cref{cor:b3:complete:weierstrass}), met de expliciete snelheid
$\norm{B_nf - f}_\infty \leq \frac32\,\omega_f(n^{-1/2})$ voor de
continuïteitsmodulus $\omega_f$ --- bewijs minstens de vorm
$O(\omega_f(n^{-1/2}))$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:probability:9}
(Plaatjesverzamelaar) Kaarten van $n$ soorten worden uniform met
teruglegging getrokken; zij $T_n$ het aantal trekkingen tot alle
soorten gezien zijn.
(a) Schrijf $T_n = \sum_{k=1}^{n}\tau_k$ met $\tau_k$ meetkundig
met parameter $\frac{n - k + 1}n$ en de $\tau_k$ \omterm{def:b3:probability:independence}{onafhankelijk},
en leid af dat $\E T_n = n\,H_n \sim n\ln n$ ($H_n$ het
harmonische getal) en $\V(T_n) \leq \frac{\pi^2}6n^2$.
(b) Chebyshev: $\frac{T_n}{n\ln n} \to 1$ in kans.
(c) Verfijn met Borel--Cantelli: toon rechtstreeks aan dat $\P(T_n
> \beta n\ln n) \leq n^{1 - \beta}$ voor $\beta > 1$ \emph{(de
somgrens op de gebeurtenis dat een soort na $\beta n\ln n$
trekkingen wordt gemist, met $1 - x \leq \eu^{-x}$)}, en leid af
dat langs $n = 2^m$ b.z.\ uiteindelijk $T_n \leq \beta n\ln n$
voor elke $\beta > 2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:10}
Met de constructie met cijfers
(\cref{thm:b3:probability:existence}):
(a) ga met een rechtstreekse berekening na dat $U = \sum
b_{2k}2^{-k}$ (de cijfers met even index van een uniforme
$\omega$) uniform is en onafhankelijk van $V = \sum
b_{2k-1}2^{-k}$;
(b) leid een \omterm{def:b3:lebesgue:measurable}{meetbare} bijectie op nulverzamelingen na af tussen
$\intcc01$ en $\intcc01^2$ die de \omterm{def:b3:measure:measure}{maat} behoudt, en
becommentarieer: één uniform toevalsgetal bevat er twee (en
aftelbaar veel) \omterm{def:b3:probability:independence}{onafhankelijke} --- vergelijk met de peanokromme
(\cref{pb:b3:topology:1}), die wel surjectiviteit haalde maar
geen maatbehoud of injectiviteit.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:11}
(Records) Zij $(X_n)_{n\geq1}$ i.i.d.\ met \omterm{def:b3:topology:continuity}{continue}
verdelingsfunctie, en zeg dat er op tijdstip $n$ een
\emph{record} optreedt als $X_n > \max(X_1, \dots, X_{n-1})$
(tijdstip $1$ is een record). Zij $R_n$ de recordindicator.
(a) Toon aan dat $\P(R_n = 1) = \frac1n$ \emph{(wegens de
symmetrie is elk van de $n!$ ordeningen van $X_1, \dots, X_n$
even waarschijnlijk, en gelijke waarden hebben kans $0$)}.
(b) Toon aan dat de $R_n$ \emph{\omterm{def:b3:probability:independence}{onafhankelijk}} zijn \emph{(tel de
ordeningen die met voorgeschreven recordposities verenigbaar
zijn, of beargumenteer dat de onderlinge volgorde van $X_1,
\dots, X_{n-1}$ \omterm{def:b3:probability:independence}{onafhankelijk} is van de rang van $X_n$ ertussen)}.
(c) Leid met Borel--Cantelli
(\cref{thm:b3:probability:borelcantelli}, beide helften) af dat
er b.z.\ oneindig veel records optreden, maar dat records op
opeenvolgende tijdstippen $n, n+1$ oneindig vaak optreden met
kans --- beslis welke! --- en bereken $\sum_n\P(R_n = 1, R_{n+1}
= 1)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:12}
(De langste reeks kop) Werp oneindig vaak met een eerlijke munt,
en zij $L_n$ de lengte van de langste reeks opeenvolgende koppen
binnen de eerste $n$ worpen.
(a) Toon aan dat voor elke $\varepsilon > 0$ b.z.\ uiteindelijk
$L_n \leq (1 + \varepsilon)\log_2n$ \emph{(de kans dat er onder
de eerste $n$ worpen een reeks van lengte $\ell$ begint, is
hoogstens $n2^{-\ell}$; Borel--Cantelli langs $n = 2^k$)}.
(b) Toon aan dat b.z.\ uiteindelijk $L_n \geq (1 -
\varepsilon)\log_2n$ \emph{(hak de eerste $n$ worpen in
$\lfloor n/\ell\rfloor$ disjuncte blokken van lengte $\ell =
\lceil(1 - \varepsilon)\log_2n\rceil$; de blokken zijn
\omterm{def:b3:probability:independence}{onafhankelijk}, elk \omterm{def:b3:complete:complete}{volledig} kop met kans $2^{-\ell}$, en de kans
dat geen enkel blok \omterm{def:b3:complete:complete}{volledig} kop is, is hoogstens
$\exp(-n2^{-\ell}/\ell)$; sommeer opnieuw langs $n = 2^k$)}.
(c) Besluit $\frac{L_n}{\log_2n} \to 1$ b.z.: in een miljoen
eerlijke worpen mag je een reeks van ongeveer $20$ koppen
verwachten --- en een gegevensverzameling zonder zo'n reeks is
waarschijnlijk verzonnen.
\end{exercise}

\section{Probleem: het bewijs van Etemadi van de sterke wet}

\begin{problem}[{Weekendopgave --- de sterke wet van de grote
aantallen voor i.i.d.\ integreerbare veranderlijken}]
\label{pb:b3:probability:1}
De sterke wet van Kolmogorov --- $\frac{S_n}n \to \E X_1$ b.z.\
voor i.i.d.\ $X_n \in L^1$ --- had lang enkel ingewikkelde
bewijzen; in 1981 vond N.~Etemadi er een van opvallende zuinigheid,
dat niets voorbij dit hoofdstuk gebruikt (en de
\omterm{def:b3:probability:independence}{onafhankelijkheid} zelfs verzwakt tot paarsgewijze
\omterm{def:b3:probability:independence}{onafhankelijkheid}). Wij volgen het. Zij $(X_n)$ paarsgewijs
\omterm{def:b3:probability:independence}{onafhankelijk}, identiek verdeeld en \omterm{def:b3:lebesgue:l1}{integreerbaar}; $m = \E X_1$
en $S_n = X_1 + \dots + X_n$.

\medskip
\noindent\textbf{Deel I --- Herleidingen.}
\begin{enumerate}
  \item Toon aan dat het volstaat $X_n \geq 0$ te behandelen
        \emph{(splits $X_n = X_n^+ - X_n^-$: ga na dat de twee
        helften opnieuw paarsgewijs \omterm{def:b3:probability:independence}{onafhankelijk}, identiek
        verdeeld en \omterm{def:b3:lebesgue:l1}{integreerbaar} zijn)}. Neem voortaan $X_n \geq
        0$ aan.
  \item (Afknotting) Zij $Y_n = X_n\,\mathbf 1_{X_n \leq n}$ en
        $S_n^* = Y_1 + \dots + Y_n$. Toon aan dat
        \[
        \sum_{n\geq1}\P(X_n \neq Y_n) =
        \sum_{n\geq1}\P(X_1 > n) \leq \E[X_1] < \infty
        \]
        (\cref{exo:b3:product:3}), en leid met Borel--Cantelli af
        dat $\frac{S_n - S_n^*}{n} \to 0$ b.z.: het volstaat dus
        $\frac{S^*_n}n \to m$ b.z.\ te bewijzen.
  \item Toon aan dat $\E Y_n = \E\bigl[X_1\mathbf 1_{X_1\leq
        n}\bigr] \to m$ (monotone convergentie), en dus
        $\frac1n\sum_{k\leq n}\E Y_k \to m$ (Cesàro): het volstaat
        dus $\frac{S_n^* - \E S_n^*}{n} \to 0$ b.z.\ te bewijzen.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De variantieafschatting.}
\begin{enumerate}[resume]
  \item Toon aan dat
        \[
        \V(Y_n) \leq \E[Y_n^2] = \E\bigl[X_1^2\,\mathbf
        1_{X_1 \leq n}\bigr]
        \]
        en, met de laagjesformule
        (\cref{prop:b3:product:layercake}), de sleutelgrens
        \[
        \sum_{n\geq1}\frac{\V(Y_n)}{n^2}
        \leq \sum_{n\geq1}\frac1{n^2}\,
        \E\bigl[X_1^2\mathbf 1_{X_1\leq n}\bigr]
        \leq C\,\E[X_1] < \infty
        \]
        \emph{(verwissel de som en de \omterm{def:b3:probability:space}{verwachtingswaarde} ---
        Tonelli voor reeksen --- en schat $\sum_{n \geq
        x}\frac1{n^2} \leq \frac2{\max(x,1)}$ af voor de
        inwendige schatting $x^2\sum_{n\geq x}n^{-2} \leq 2x$)}.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Convergentie langs meetkundige
deelrijen.} Leg $\alpha > 1$ vast en zij $k_j =
\lfloor\alpha^j\rfloor$.
\begin{enumerate}[resume]
  \item Toon met de paarsgewijze \omterm{def:b3:probability:independence}{onafhankelijkheid} (de varianties
        tellen op, \cref{thm:b3:probability:independence} --- ga
        na dat de additiviteit van de varianties enkel
        paarsgewijze \omterm{def:b3:probability:independence}{onafhankelijkheid} vergt) en Chebyshev aan
        dat voor elke $\varepsilon > 0$
        \[
        \sum_{j\geq1}\P\Bigl(\Bigl|
        \frac{S^*_{k_j} - \E S^*_{k_j}}{k_j}\Bigr| \geq
        \varepsilon\Bigr)
        \leq
        \frac1{\varepsilon^2}\sum_{j\geq1}\frac1{k_j^2}
        \sum_{n\leq k_j}\V(Y_n)
        = \frac1{\varepsilon^2}\sum_{n\geq1}\V(Y_n)
        \sum_{j\,:\,k_j\geq n}\frac1{k_j^2} .
        \]
  \item Toon aan dat $\sum_{j : k_j \geq n}k_j^{-2} \leq
        \frac{C_\alpha}{n^2}$ \emph{(meetkundige reeksen; pas op
        met het gehele deel: zorg voor iets als $k_j \geq
        \frac{\alpha^j}2$ zodra $\alpha^j \geq 2$)}, en besluit
        met vraag 4 en Borel--Cantelli:
        \[
        \frac{S^*_{k_j} - \E S^*_{k_j}}{k_j}
        \xrightarrow[j\to\infty]{\text{b.z.}} 0,
        \qquad\text{en dus}\qquad
        \frac{S^*_{k_j}}{k_j} \to m \ \text{b.z.}
        \]
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Insluiting en besluit.}
\begin{enumerate}[resume]
  \item Toon voor $k_j \leq n \leq k_{j+1}$ met de monotonie van
        $S^*_n$ (niet-negatieve termen!) aan dat
        \[
        \frac{k_j}{k_{j+1}}\,\frac{S^*_{k_j}}{k_j}
        \;\leq\; \frac{S^*_n}{n} \;\leq\;
        \frac{k_{j+1}}{k_j}\,\frac{S^*_{k_{j+1}}}{k_{j+1}},
        \]
        en leid b.z.\ af:
        \[
        \frac m\alpha \leq \liminf\frac{S^*_n}n \leq
        \limsup\frac{S^*_n}n \leq \alpha\,m .
        \]
  \item Laat $\alpha \downarrow 1$ gaan langs een rij en besluit
        $\frac{S_n^*}n \to m$ b.z., en dus (Deel I) de
        \emph{sterke wet van de grote aantallen}:
        \[
        \boxed{\ \frac{S_n}{n}
        \xrightarrow[n\to\infty]{\text{b.z.}} \E[X_1].\ }
        \]
  \item Waar precies volstond de paarsgewijze (in plaats van de
        volledige) \omterm{def:b3:probability:independence}{onafhankelijkheid}? Som de drie plaatsen op
        waar hypothesen van het onafhankelijkheidstype werden
        ingeroepen.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Dividenden.}
\begin{enumerate}[resume]
  \item (De normale getallen van Borel) Toon aan dat
        $\lambda$-bijna elke $x \in \intcc01$ \emph{normaal is in
        elke basis} $b \geq 2$: elk cijfer $0, \dots, b-1$
        verschijnt met asymptotische frequentie $\frac1b$
        \emph{(leg $b$ en een cijfer vast, pas de sterke wet toe
        op de indicatorveranderlijken --- verantwoord dat de
        cijfers in basis $b$ van een uniforme veranderlijke
        i.i.d.\ uniform op $\{0,\dots,b-1\}$ zijn, als in
        \cref{thm:b3:probability:existence} --- en snijd daarna
        de aftelbaar vele gebeurtenissen met kans één)}. Geef één
        expliciet niet-normaal getal, en bedenk: de stelling
        beweert de normaliteit van bijna alle getallen, en toch
        blijft de normaliteit van $\sqrt2$ of $\pi$ bewijzen een
        open probleem.
  \item (\omterm{ex:b3:probability:sllnapps}{Monte carlo}, gewaarborgd) Verantwoord de methode van
        \cref{ex:b3:probability:sllnapps}(b) \omterm{def:b3:complete:complete}{volledig} voor $g \in
        L^1(\intcc01^d)$: construeer de i.i.d.\ uniforme
        steekproef op $\intcc01^d$ met
        \cref{thm:b3:probability:existence} en
        \cref{exo:b3:probability:10}, en formuleer wat de sterke
        wet oplevert.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Wat volledige \omterm{def:b3:probability:independence}{onafhankelijkheid}
oplevert: maximaalongelijkheden en toevallige reeksen.} Etemadi
geeft enkel paarsgewijze \omterm{def:b3:probability:independence}{onafhankelijkheid} uit; de resterende
delen buiten de volledige (onderlinge) versie uit. Zij $(Z_n)$
\omterm{def:b3:probability:independence}{onafhankelijke} gecentreerde veranderlijken in $L^2$ en $S_k = Z_1
+ \dots + Z_k$ (een nieuwe notatie, los van de $X_n$ hierboven).
\begin{enumerate}[resume]
  \item (Maximaalongelijkheid van Kolmogorov) Bewijs voor
        $\varepsilon > 0$ dat
        \[
        \P\Bigl(\max_{1\leq k\leq n}\abs{S_k} \geq
        \varepsilon\Bigr) \;\leq\;
        \frac1{\varepsilon^2}\sum_{k=1}^n\V(Z_k) :
        \]
        de prijs van Chebyshev koopt het maximum \emph{(deel de
        gebeurtenis op naar de eerste index $k$ met $\abs{S_k}
        \geq \varepsilon$; schrijf op dat stuk $S_n^2 \geq S_k^2
        + 2S_k(S_n - S_k)$ en gebruik de \omterm{def:b3:probability:independence}{onafhankelijkheid} van de
        coalities $(Z_1, \dots, Z_k)$ en $(Z_{k+1}, \dots, Z_n)$,
        \cref{thm:b3:probability:independence})}. Wijs de stap
        aan waar paarsgewijze \omterm{def:b3:probability:independence}{onafhankelijkheid} niet langer zou
        volstaan.
  \item (Eenreeksstelling van Khinchin--Kolmogorov) Leid af: is
        $\sum_n\V(Z_n) < \infty$, dan convergeert $\sum_nZ_n$
        bijna zeker \emph{(toon aan dat de partiële sommen b.z.\
        een cauchyrij vormen: laat $m \to \infty$ gaan in de
        maximaalongelijkheid toegepast op $Z_{N+1}, \dots,
        Z_{N+m}$, en daarna $N \to \infty$)}.
  \item (Rademacherreeksen) Zij $(\varepsilon_n)$ i.i.d.\ tekens
        met $\P(\varepsilon_n = \pm1) = \frac12$
        (\cref{thm:b3:probability:existence}), en zij $(x_n)$ een
        rij reële getallen. Toon aan dat
        $\sum_nx_n\varepsilon_n$ b.z.\ convergeert zodra
        $\sum_nx_n^2 < \infty$; toon ook aan dat, wat $(x_n)$ ook
        is, de kans dat $\sum_nx_n\varepsilon_n$ convergeert $0$
        of $1$ is (\cref{thm:b3:probability:zeroone}).
  \item De omkering, elementair. Stel $T_n = \sum_{k\leq
        n}x_k\varepsilon_k$ en $s_n^2 = \sum_{k\leq n}x_k^2$, en
        neem aan dat $s_n \to \infty$.
        (a) Bewijs de \emph{ongelijkheid van Paley--Zygmund}:
        voor $Z \geq 0$ met $\E Z^2 < \infty$ en $0 < \theta <
        1$ is
        \[
        \P\bigl(Z > \theta\,\E Z\bigr) \;\geq\; (1 -
        \theta)^2\,\frac{(\E Z)^2}{\E Z^2}
        \]
        \emph{(splits $\E Z$ op het niveau $\theta\E Z$ en pas
        Cauchy--Schwarz toe op het bovenste stuk)}.
        (b) Toon aan dat $\E T_n^4 \leq 3s_n^4$.
        (c) Leid $\P\bigl(\abs{T_n} > \frac{s_n}2\bigr) \geq
        \frac3{16}$ af en besluit dat $\sum_nx_n\varepsilon_n$
        b.z.\ divergeert; en dus de dichotomie
        \[
        \sum_nx_n\varepsilon_n\ \text{convergeert b.z.}
        \iff \sum_nx_n^2 < \infty .
        \]
  \item (Toevallige harmonische reeks) Besluit dat
        $\sum_n\frac{\varepsilon_n}{n^s}$ b.z.\ convergeert dan
        en slechts dan als $s > \frac12$. Voor $\frac12 < s \leq
        1$ convergeert de reeks b.z.\ terwijl $\sum_nn^{-s} =
        \infty$: toevallige tekens brengen opheffing ter sterkte
        van een vierkantswortel voort --- vergelijk met de
        alternerende reeks $\sum_n\frac{(-1)^n}{n^s}$, die voor
        \emph{elke} $s > 0$ convergeert.
\end{enumerate}

\medskip
\noindent\textbf{Deel VII --- Concentratie: de ongelijkheid van
Hoeffding.} De sterke wet zegt dat $\frac{S_n}n \to m$;
concentratieongelijkheden zeggen hoe onwaarschijnlijk een
afwijking is \emph{bij elke vaste $n$}.
\begin{enumerate}[resume]
  \item (Lemma van Hoeffding)
        (a) Toon aan dat $\cosh\lambda \leq \eu^{\lambda^2/2}$
        voor alle $\lambda \in \R$, door de twee reeksen term
        voor term te vergelijken.
        (b) Zij $Z$ gecentreerd met $a \leq Z \leq b$ en $a < b$.
        Toon aan dat
        \[
        \E\,\eu^{\lambda Z} \leq
        \exp\Bigl(\frac{\lambda^2(b - a)^2}8\Bigr)
        \]
        \emph{(begrens $\eu^{\lambda z}$ op $\intcc ab$ met haar
        koorde, neem \omterm{def:b3:probability:space}{verwachtingswaarden}, en bestudeer
        $\varphi(t) = -pt + \log(1 - p + p\eu^t)$ met $p =
        \frac{-a}{b-a}$ en $t = \lambda(b - a)$: toon aan dat
        $\varphi(0) = \varphi'(0) = 0$ en $\varphi'' \leq
        \frac14$)}.
  \item (Ongelijkheid van Hoeffding) Zij $X_1, \dots, X_n$
        \omterm{def:b3:probability:independence}{onafhankelijk} met $a_i \leq X_i \leq b_i$ en $S_n = X_1 +
        \dots + X_n$. Bewijs voor $t > 0$
        \[
        \P\bigl(S_n - \E S_n \geq t\bigr) \leq
        \exp\Bigl(\frac{-2t^2}{\sum_{i=1}^n(b_i -
        a_i)^2}\Bigr),
        \]
        en dezelfde grens voor de onderste staart
        \emph{(exponentiële Chebyshev: begrens $\E\,\eu^{\lambda(S_n
        - \E S_n)}$ met de \omterm{def:b3:probability:independence}{onafhankelijkheid} en vraag 17, en
        optimaliseer daarna over $\lambda > 0$)}.
  \item (De sterke wet, begrensd geval, met een snelheid) Zij de
        $X_i$ i.i.d.\ met waarden in $\intcc ab$ en $m = \E X_1$.
        Toon aan dat
        \[
        \P\Bigl(\Bigl|\frac{S_n}n - m\Bigr| \geq
        \varepsilon\Bigr) \leq
        2\exp\Bigl(\frac{-2n\varepsilon^2}{(b - a)^2}\Bigr)
        \]
        en vind $\frac{S_n}n \to m$ b.z.\ terug met
        Borel--Cantelli: een tweede bewijs van de sterke wet voor
        begrensde veranderlijken --- geen afknotting, een
        exponentiële snelheid bij elke eindige $n$, maar
        begrensde termen en volledige \omterm{def:b3:probability:independence}{onafhankelijkheid}.
        Vergelijk de hypothesen met die van Etemadi.
  \item (\omterm{ex:b3:probability:sllnapps}{Monte carlo}, gewaarborgd bij vaste $n$) Zij $g \colon
        \intcc01^d \to \intcc01$ \omterm{def:b3:lebesgue:measurable}{meetbaar} en $(U_k)$ de i.i.d.\
        uniforme steekproef uit vraag 11. Toon voor gegeven
        $\varepsilon, \delta > 0$ aan dat
        \[
        n \geq \frac{\log(2/\delta)}{2\varepsilon^2}
        \implies
        \P\Bigl(\Bigl|\frac1n\sum_{k=1}^ng(U_k) -
        \int g\,\dd\lambda_d\Bigr| \geq \varepsilon\Bigr)
        \leq \delta,
        \]
        en evalueer de drempel voor $\varepsilon = \delta =
        10^{-2}$. De grens bevat $d$ niet: vergelijk met vraag 11
        en met deterministische roosters.
\end{enumerate}

\medskip
\noindent\textbf{Deel VIII --- Hoe groot is een toevalswandeling?
Op weg naar de geïtereerde logaritme.} Zij $S_n = \varepsilon_1 +
\dots + \varepsilon_n$ de eenvoudige toevalswandeling, opgebouwd
uit i.i.d.\ eerlijke tekens.
\begin{enumerate}[resume]
  \item (Subgaussische staarten) Toon aan dat $\E\,\eu^{\lambda
        S_n} = (\cosh\lambda)^n \leq \eu^{n\lambda^2/2}$ en leid
        voor $x > 0$ af dat
        \[
        \P(S_n \geq x) \leq \eu^{-x^2/(2n)},
        \qquad
        \P(\abs{S_n} \geq x) \leq 2\,\eu^{-x^2/(2n)} .
        \]
  \item Leid met Borel--Cantelli af dat
        \[
        \limsup_{n\to\infty}\frac{\abs{S_n}}
        {\sqrt{2n\log n}} \leq 1 \quad\text{b.z.}
        \]
        \emph{(sommeer voor $\eta > 0$ de staartgrenzen in $x =
        (1 + \eta)\sqrt{2n\log n}$, en snijd daarna over $\eta =
        \frac1p$)}. In het bijzonder leeft de wandeling op de
        schaal $\sqrt n$ van de centrale limietstelling, op een
        logaritmische factor na --- ver onder de grove grens
        $\abs{S_n} \leq n$.
  \item Toon langs de verdubbelende deelrij $n_j = 2^j$ aan dat
        \[
        \limsup_{j\to\infty}\frac{S_{n_j}}
        {\sqrt{2n_j\log\log n_j}} \leq 1 \quad\text{b.z.},
        \]
        en bedenk: de \emph{wet van de geïtereerde logaritme}
        (Khinchin; Hartman--Wintner voor algemene gecentreerde
        termen in $L^2$) zegt dat
        \[
        \limsup_{n\to\infty}\frac{S_n}
        {\sqrt{2n\log\log n}} = 1 \quad\text{b.z.}
        \]
        Leg precies uit wat de zonet bewezen schatting langs een
        deelrij scheidt van de bovenste helft van die uitspraak
        (men moet $\max_{n_j \leq n \leq n_{j+1}}S_n$ binnen elk
        blok beheersen, wat een maximaalongelijkheid op de
        \emph{exponentiële} schaal vergt) en ga kwantitatief na
        dat de ongelijkheid van vraag 12 daarvoor te zwak is. De
        onderste helft berust op het tweede lemma van
        Borel--Cantelli toegepast op \omterm{def:b3:probability:independence}{onafhankelijke} blokken;
        beide helften zijn eerlijk materiaal van bachelorjaar 3
        voor een aparte cursus kansrekening.

  \item (Uniforme afwijking over een eindige klasse) Zij $A_1,
        \dots, A_N$ gebeurtenissen in een herhaalbaar experiment,
        en schat elke kans met haar empirische frequentie $\hat
        p_i$ over $n$ i.i.d.\ herhalingen. Toon door de
        ongelijkheid van Hoeffding met een somgrens te combineren
        aan dat
        \[
        \P\Bigl(\max_{i\leq N}\,\abs{\hat p_i - \P(A_i)} >
        \varepsilon\Bigr) \;\leq\; 2N\,\eu^{-2n\varepsilon^2},
        \]
        en leid de regel voor de steekproefgrootte af: $n \geq
        \frac{\ln(2N/\delta)}{2\varepsilon^2}$ waarborgt dat alle
        $N$ schattingen tegelijk $\varepsilon$-nauwkeurig zijn
        met kans $\geq 1 - \delta$. Bereken $n$ voor $N = 10^6$,
        $\varepsilon = 0.01$ en $\delta = 0.05$: de logaritmische
        prijs van de uniformiteit.
  \item (Het toevallige harmonische venster) Toon door de twee
        helften van de theorie van de toevallige reeksen te
        combineren aan dat voor i.i.d.\ tekens $(\varepsilon_n)$
        de reeks $\sum_n\frac{\varepsilon_n}{n^\alpha}$ b.z.\
        convergeert als $\alpha > \frac12$ en b.z.\ divergeert
        als $\alpha \leq \frac12$; contrasteer met de absolute
        convergentie (die $\alpha > 1$ vergt): op het venster
        $\alpha \in \intoc{\frac12}1$ is de convergentie een
        werkelijk kanstheoretisch verschijnsel --- opheffing,
        geen grootte.
\end{enumerate}
\end{problem}
