\chapter{De Lebesgue-integraal}\label{ch:b3:lebesgue}

De integraal van Riemann snijdt het \emph{domein} in kleine
intervallen; die van Lebesgue snijdt het \emph{beeld}: om $f$ te
integreren, meet je de verzamelingen $\{f > t\}$. De wijziging
oogt onschuldig en is revolutionair. Limieten en integralen, in de
theorie van Riemann eeuwig kibbelend (uniforme convergentie
vereist!), worden verzoend door drie convergentiestellingen ---
monotone convergentie, Fatou, gedomineerde convergentie ---
waarvan de hypothesen bijna gênant zwak zijn. Dit hoofdstuk bouwt
de integraal op een willekeurige maatruimte $(X, \mathcal A,
\mu)$, bewijst de drie stellingen, beslecht de precieze
verhouding met de integraal van Riemann (een begrensde functie is
\omterm{def:b3:lebesgue:l1}{riemann-integreerbaar} dan en slechts dan als ze \omterm{def:b3:lebesgue:l1}{bijna overal}
\omterm{def:b3:topology:continuity}{continu} is) en industrialiseert het differentiëren van integralen
met een parameter --- de techniek waarmee de weekendopgave
$\int_0^\infty\frac{\sin x}x\,\dd x$ en $\int_\R \eu^{-x^2}\dd x$
berekent.

\section{Meetbare functies}

\begin{definition}\label{def:b3:lebesgue:measurable}
Zij $(X, \mathcal A)$ en $(Y, \mathcal B)$ meetbare ruimten. Dan
heet $f \colon X \to Y$ \emph{meetbaar}\index{meetbare functie}
als $f^{-1}(B) \in \mathcal A$ voor elke $B \in \mathcal B$. Voor
reële (of $[-\infty,+\infty]$-waardige) functies draagt $Y = \R$
haar Borel-$\sigma$-algebra, en volstaat het $f^{-1}(\intoo
t{+\infty}) = \{f > t\} \in \mathcal A$ voor alle $t \in \R$ na te
gaan: de goede verzamelingen $\{B : f^{-1}(B) \in \mathcal A\}$
vormen een $\sigma$-algebra (originelen verdragen zich met
verzamelingsbewerkingen) die de voortbrengende halfrechten bevat
(\cref{def:b3:measure:borel},
\cref{met:b3:measure:goodsets}).
\end{definition}

\begin{proposition}\label{prop:b3:lebesgue:stability}
(a) Samenstellingen van \omterm{def:b3:lebesgue:measurable}{meetbare} afbeeldingen zijn \omterm{def:b3:lebesgue:measurable}{meetbaar};
\omterm{def:b3:topology:continuity}{continue afbeeldingen} zijn \omterm{def:b3:lebesgue:measurable}{borel-meetbaar}.
(b) Zijn $f, g \colon X \to \R$ \omterm{def:b3:lebesgue:measurable}{meetbaar}, dan ook $f + g$, $fg$,
$\max(f,g)$, $\abs f$ en $\lambda f$.
(c) Zijn de $(f_n)$ \omterm{def:b3:lebesgue:measurable}{meetbaar} met waarden in $[-\infty, +\infty]$,
dan zijn $\sup_nf_n$, $\inf_nf_n$, $\limsup f_n$ en $\liminf f_n$
\omterm{def:b3:lebesgue:measurable}{meetbaar}; en convergeert $f_n \to f$ puntsgewijs, dan is $f$
\omterm{def:b3:lebesgue:measurable}{meetbaar}.
\end{proposition}

\begin{proof}
(a) $(g\circ f)^{-1}(B) = f^{-1}(g^{-1}(B))$; en de \omterm{def:b3:topology:continuity}{continuïteit}
geeft de \omterm{def:b3:lebesgue:measurable}{meetbaarheid} via de voortbrengende \omterm{def:b3:topology:topology}{open verzamelingen}
(\cref{pb:b3:measure:1}, vraag 10, in algemene vorm).
(b) $(f, g) \colon X \to \R^2$ is \omterm{def:b3:lebesgue:measurable}{meetbaar} voor de
Borel-$\sigma$-algebra van $\R^2$ --- ga het na op de open blokken,
die voortbrengen (de \omterm{def:b3:topology:topology}{open verzamelingen} van $\R^2$ zijn aftelbare
verenigingen van rationale blokken): $(f,g)^{-1}(U\times V) =
f^{-1}(U)\cap g^{-1}(V)$ --- en $+$, $\times$ en $\max$ zijn
\omterm{def:b3:topology:continuity}{continu} van $\R^2$ naar $\R$: stel samen.
(c) $\{\sup f_n > t\} = \bigcup_n\{f_n > t\}$; $\inf = -\sup(-)$;
$\limsup = \inf_N\sup_{n \geq N}$; en een puntsgewijze limiet is
haar eigen $\limsup$.
\end{proof}

\begin{definition}\label{def:b3:lebesgue:simple}
Een \emph{elementaire functie}\index{elementaire functie} is een
\omterm{def:b3:lebesgue:measurable}{meetbare functie} met eindig veel waarden: $s = \sum_{i=1}^n
c_i\,\mathbf 1_{A_i}$ met $A_i \in \mathcal A$ disjunct en $c_i
\geq 0$ (voor de niet-negatieve theorie). Haar integraal is
\[
\int s\,\dd\mu = \sum_i c_i\,\mu(A_i) \in [0, +\infty]
\]
(met de afspraak $0\cdot\infty = 0$); de waarde hangt niet van de
voorstelling af (verfijn twee partities).
\end{definition}

\begin{theorem}[Benadering door elementaire functies]
\label{thm:b3:lebesgue:approximation}
Elke \omterm{def:b3:lebesgue:measurable}{meetbare} $f \colon X \to [0, +\infty]$ is de puntsgewijze
limiet van een \emph{stijgende} rij \omterm{def:b3:lebesgue:simple}{elementaire functies}:
\[
s_n = \sum_{k=1}^{n2^n} \frac{k-1}{2^n}\,
\mathbf 1_{\{\frac{k-1}{2^n} \leq f < \frac k{2^n}\}}
+ n\,\mathbf 1_{\{f \geq n\}} \nearrow f .
\]
\end{theorem}

\begin{proof}
Elke $s_n$ is elementair (de verzamelingen zijn originelen van
borelverzamelingen). Monotonie: van $n$ naar $n+1$ gaan splitst
elk dyadisch niveau in tweeën en verlaagt de toegekende waarde
nooit (een punt met $\frac{k-1}{2^n} \leq f(x) < \frac k{2^n}$
krijgt $\frac{2k-2}{2^{n+1}}$ of $\frac{2k-1}{2^{n+1}}$, allebei
$\geq \frac{k-1}{2^n}$; en de afkapwaarde $n$ stijgt mee).
Convergentie: is $f(x) < \infty$, dan is voor $n > f(x)$ het
verschil $f(x) - s_n(x) \leq 2^{-n}$; en is $f(x) = \infty$, dan is
$s_n(x) = n \to \infty$.
\end{proof}

\section{De integraal en de convergentiestellingen}

\begin{definition}\label{def:b3:lebesgue:integral}
Voor \omterm{def:b3:lebesgue:measurable}{meetbare} $f \geq 0$:
\[
\int f \,\dd\mu = \sup\Bigl\{\int s\,\dd\mu : s \text{
elementair},\ 0 \leq s \leq f\Bigr\} \in [0, +\infty].
\]
Ze is per constructie monotoon in $f$, en zet het elementaire
geval voort (voor elementaire $f$ wordt het supremum in $f$
bereikt: vergelijk elementaire integralen via gemeenschappelijke
verfijningen).
\end{definition}

\begin{theorem}[Monotone convergentie, Beppo Levi]
\label{thm:b3:lebesgue:mct}
Is $0 \leq f_n \nearrow f$ puntsgewijs (\omterm{def:b3:lebesgue:measurable}{meetbaar}), dan is
\[
\int f_n\,\dd\mu \nearrow \int f\,\dd\mu .
\]
\index{stelling van de monotone convergentie}
\end{theorem}

\begin{proof}
$f$ is \omterm{def:b3:lebesgue:measurable}{meetbaar} (\cref{prop:b3:lebesgue:stability}(c)) en $\int
f_n$ stijgt naar een zekere $L \leq \int f$ (monotonie). Leg
omgekeerd een elementaire $s = \sum c_i\mathbf 1_{A_i} \leq f$
vast en een $\theta \in (0,1)$; de verzamelingen $E_n = \{f_n \geq
\theta s\}$ zijn \omterm{def:b3:lebesgue:measurable}{meetbaar} en stijgen naar $X$ (waar $s(x) > 0$:
$f(x) \geq s(x) > \theta s(x)$, dus vanaf zeker moment $f_n(x)
\geq \theta s(x)$; en waar $s(x) = 0$: triviaal). Dan is
\[
\int f_n \geq \int_{E_n}\theta s\,\dd\mu
= \theta\sum_i c_i\,\mu(A_i \cap E_n)
\xrightarrow[n\to\infty]{} \theta\sum_ic_i\,\mu(A_i)
= \theta\int s
\]
wegens de \omterm{def:b3:topology:continuity}{continuïteit} van onderen
(\cref{prop:b3:measure:basics}(c)). Dus $L \geq \theta\int s$ voor
alle $\theta < 1$ en alle elementaire $s \leq f$: $L \geq \int f$.
\end{proof}

\begin{corollary}\label{cor:b3:lebesgue:additivity}
Voor \omterm{def:b3:lebesgue:measurable}{meetbare} $f, g \geq 0$ en $c \geq 0$ geldt $\int(f + g) =
\int f + \int g$ en $\int cf = c\int f$; en voor een reeks
niet-negatieve \omterm{def:b3:lebesgue:measurable}{meetbare functies} is $\int\sum_nf_n =
\sum_n\int f_n$.
\end{corollary}

\begin{proof}
Voor \omterm{def:b3:lebesgue:simple}{elementaire functies} is de additiviteit een berekening op een
gemeenschappelijke verfijning. In het algemeen neem je $s_n
\nearrow f$ en $t_n \nearrow g$
(\cref{thm:b3:lebesgue:approximation}): dan is $s_n + t_n \nearrow
f + g$, en de monotone convergentie draagt de additiviteit over op
de limiet. De uitspraak over de reeks is de monotone convergentie
toegepast op de partiële sommen.
\end{proof}

\begin{theorem}[Lemma van Fatou]\label{thm:b3:lebesgue:fatou}
Voor \omterm{def:b3:lebesgue:measurable}{meetbare} $f_n \geq 0$ geldt
\[
\int \liminf_n f_n \,\dd\mu \;\leq\; \liminf_n \int
f_n\,\dd\mu .
\]
\index{lemma van Fatou}
\end{theorem}

\begin{proof}
Zij $g_N = \inf_{n\geq N}f_n$: \omterm{def:b3:lebesgue:measurable}{meetbaar}, met $0 \leq g_N \nearrow
\liminf f_n$ en $g_N \leq f_n$ voor elke $n \geq N$, zodat $\int
g_N \leq \inf_{n \geq N}\int f_n$. Pas de monotone convergentie
toe op het linkerlid: $\int\liminf f_n = \lim_N\int g_N \leq
\lim_N\inf_{n\geq N}\int f_n = \liminf\int f_n$.
\end{proof}

\begin{definition}\label{def:b3:lebesgue:l1}
Een \omterm{def:b3:lebesgue:measurable}{meetbare} $f \colon X \to \R$ (of $\C$) heet
\emph{integreerbaar}\index{integreerbare functie} als
$\int\abs f\,\dd\mu < \infty$; dan is $\int f = \int f^+ - \int
f^-$ (positief en negatief deel; reëel en imaginair deel in het
complexe geval). De integraal is lineair op de integreerbare
functies (ontbind en zet weer samen op de positieve delen; het
complexe geval herleidt tot het reële) en voldoet aan $\abs{\int
f} \leq \int\abs f$ (reëel geval: $\pm\int f = \int(\pm f) \leq
\int\abs f$; complex geval: vermenigvuldig met een constante van
modulus $1$ om de integraal reëel te maken). Een eigenschap geldt
\emph{bijna overal}\index{bijna overal} als ze alleen op een
$\mu$-nulverzameling faalt; en $f$ op een nulverzameling wijzigen
verandert geen enkele integraal (het verschil wordt gedomineerd
door $\infty\cdot\mathbf 1_N$, met integraal $0$).
\end{definition}

\begin{theorem}[Gedomineerde convergentie]\label{thm:b3:lebesgue:dct}
Zij $f_n \to f$ \omterm{def:b3:lebesgue:l1}{bijna overal}, met $\abs{f_n} \leq g$ \omterm{def:b3:lebesgue:l1}{bijna overal}
voor een vaste \emph{\omterm{def:b3:lebesgue:l1}{integreerbare}} $g$. Dan is $f$ \omterm{def:b3:lebesgue:l1}{integreerbaar}
en
\[
\int f_n\,\dd\mu \longrightarrow \int f\,\dd\mu,
\qquad\text{en zelfs}\quad \int\abs{f_n - f}\,\dd\mu \to 0 .
\]
\index{stelling van de gedomineerde convergentie}
\end{theorem}

\begin{proof}
Laat een nulverzameling weg om de hypothesen puntsgewijs te maken.
Uit $\abs f \leq g$ volgt dat $f$ \omterm{def:b3:lebesgue:l1}{integreerbaar} is. De functies
$h_n = 2g - \abs{f_n - f} \geq 0$ voldoen aan $\liminf h_n = 2g$;
Fatou geeft
\[
\int 2g \leq \liminf\int\bigl(2g - \abs{f_n - f}\bigr)
= \int 2g - \limsup\int\abs{f_n - f},
\]
dus $\limsup\int\abs{f_n - f} \leq 0$ (het aftrekken is geoorloofd:
$\int 2g < \infty$). Ten slotte is $\abs{\int f_n - \int f} \leq
\int\abs{f_n - f} \to 0$.
\end{proof}

\begin{method}\label{met:b3:lebesgue:convergence}
Sta je voor $\lim_n\int f_n$, probeer dan op volgorde: (1) is de
rij monotoon (of is het een reeks met niet-negatieve termen)? Dan
de monotone convergentie, zonder enige integreerbaarheidseis. (2)
Is er één \omterm{def:b3:lebesgue:l1}{integreerbare} dominant $g \geq \abs{f_n}$, gevonden met
grove afschattingen (neem het ``$\sup_n$'' van de schattingen)?
Dan de gedomineerde convergentie. (3) Geen dominant en geen
monotonie? Dan begrenst Fatou nog steeds één kant, en kan de
gelijkheid werkelijk falen: de ontsnappende bult $f_n = n\mathbf
1_{\intoo0{1/n}}$ heeft $\int f_n = 1$ terwijl $f_n \to 0$ \omterm{def:b3:lebesgue:l1}{bijna
overal}. Dominatie is precies wat verbiedt dat massa naar oneindig
ontsnapt, verticaal of horizontaal.
\end{method}

\section{Riemann tegenover Lebesgue}

\begin{theorem}[Criterium van Lebesgue]
\label{thm:b3:lebesgue:riemann}
Zij $f \colon \intcc ab \to \R$ \emph{begrensd}. Dan is $f$
\omterm{def:b3:lebesgue:l1}{riemann-integreerbaar} dan en slechts dan als $f$
$\lambda$-bijna overal \omterm{def:b3:topology:continuity}{continu} is; in dat geval is $f$
\omterm{def:b3:lebesgue:l1}{lebesgue-integreerbaar} en vallen de twee integralen
samen.\index{Riemann tegenover Lebesgue}
\end{theorem}

\begin{proof}
Zij voor een verdeling $\sigma = (a = x_0 < \dots < x_N = b)$ de
trapfuncties $L_\sigma$ en $U_\sigma$ op elke
$\intoo{x_{i-1}}{x_i}$ gelijk aan $m_i = \inf_{[x_{i-1}, x_i]}f$
respectievelijk $M_i = \sup$; de darbouxsommen zijn hun
integralen (Riemann en Lebesgue vallen op trapfuncties samen en
geven beide $\sum m_i\Delta x_i$). Neem een rij verdelingen
$\sigma_n$, elk een verfijning van de vorige, met maaswijdte $\to
0$ en met darbouxsommen die naar de onderste en bovenste
darbouxintegraal van $f$ convergeren. Door de verfijningen is
$L_{\sigma_n}$ niet-dalend en $U_{\sigma_n}$ niet-stijgend,
puntsgewijs buiten de aftelbare verzameling $D$ van alle
deelpunten; noem de limieten $\ell$ en $u$ (\omterm{def:b3:lebesgue:measurable}{meetbaar},
\cref{prop:b3:lebesgue:stability}). Voor $x \notin D$, met
$I_n(x)$ het open $\sigma_n$-interval dat $x$ bevat: $\ell(x) =
\sup_n\inf_{I_n(x)}f$ en $u(x) = \inf_n\sup_{I_n(x)}f$; en omdat
de maaswijdten naar $0$ krimpen, zijn dat de onderste en bovenste
\emph{omhullenden} van $f$ in $x$ --- $u(x) - \ell(x)$ is de
oscillatie van $f$ in $x$ --- zodat \emph{$\ell(x) = u(x)$ precies
wanneer $f$ in $x$ \omterm{def:b3:topology:continuity}{continu} is}. Met de monotone en de
gedomineerde convergentie (begrensd, eindig interval):
\[
\int_{\intcc ab}\ell\,\dd\lambda = \lim_n\int L_{\sigma_n}
= \underline{\int}f,
\qquad
\int_{\intcc ab}u\,\dd\lambda = \overline{\int}f .
\]
Dus: $f$ \omterm{def:b3:lebesgue:l1}{riemann-integreerbaar} $\iff$ $\underline\int f =
\overline\int f$ $\iff$ $\int(u - \ell) = 0$ $\iff$ $u = \ell$
\omterm{def:b3:lebesgue:l1}{bijna overal} ($u - \ell \geq 0$; \cref{exo:b3:lebesgue:5}) $\iff$
$f$ \omterm{def:b3:lebesgue:l1}{bijna overal} \omterm{def:b3:topology:continuity}{continu}. In dat geval is $\ell \leq f \leq u$ met
$\ell = u$ \omterm{def:b3:lebesgue:l1}{bijna overal}: $f$ is \omterm{def:b3:lebesgue:l1}{bijna overal} gelijk aan de
\omterm{def:b3:lebesgue:measurable}{meetbare} $\ell$ en dus \omterm{def:b3:lebesgue:measurable}{lebesgue-meetbaar} (\omterm{def:b3:complete:complete}{volledigheid} van
$\lambda$), met $\int f\,\dd\lambda = \int\ell\,\dd\lambda =
\underline\int f = \int_a^bf$.
\end{proof}

\begin{example}\label{ex:b3:lebesgue:riemannexamples}
$\mathbf 1_\Q$ is nergens \omterm{def:b3:topology:continuity}{continu}: niet \omterm{def:b3:lebesgue:l1}{riemann-integreerbaar} ---
maar voor Lebesgue triviaal: $\int\mathbf 1_\Q\,\dd\lambda =
\lambda(\Q) = 0$. De functie van Thomae ($\frac1q$ in de rationale
getallen $\frac pq$ en $0$ elders) is precies in de irrationale
punten \omterm{def:b3:topology:continuity}{continu}: \omterm{def:b3:lebesgue:l1}{riemann-integreerbaar} met integraal $0$. En
\emph{oneigenlijke} riemannintegralen zijn een ander begrip:
$\int_0^\infty\frac{\sin x}x\,\dd x$ convergeert als limiet van
$\int_0^A$ (de weekendopgave berekent hem als $\frac\pi2$), maar
$\frac{\sin x}x \notin L^1(\intoo0{+\infty})$: de absolute
integraal divergeert als de harmonische reeks
(\cref{exo:b3:lebesgue:6}). De theorie van Lebesgue ruilt
voorwaardelijke convergentie in voor robuuste limietstellingen.
\end{example}

\section{Integralen met parameters}

Overal is $(X, \mathcal A, \mu)$ een maatruimte, $T$ een metrische
ruimte (de parameter) en $f \colon T \times X \to \C$ met $f(t,
\cdot)$ \omterm{def:b3:lebesgue:l1}{integreerbaar} voor elke $t$; stel $F(t) = \int_X f(t,
x)\,\dd\mu(x)$.

\begin{theorem}[Continuïteit]\label{thm:b3:lebesgue:paramcont}
Stel dat $t \mapsto f(t,x)$ voor bijna elke $x$ \omterm{def:b3:topology:continuity}{continu} is in
$t_0$, en dat er een \omterm{def:b3:lebesgue:l1}{integreerbare} $g$ is met $\abs{f(t,x)} \leq
g(x)$ voor alle $t$ in een \omterm{def:b3:topology:topology}{omgeving} van $t_0$ en bijna elke $x$.
Dan is $F$ \omterm{def:b3:topology:continuity}{continu} in $t_0$.\index{parameterintegralen}
\end{theorem}

\begin{proof}
Voor elke rij $t_n \to t_0$ is $f(t_n, \cdot) \to f(t_0, \cdot)$
\omterm{def:b3:lebesgue:l1}{bijna overal}, gedomineerd door $g$: de gedomineerde convergentie
geeft $F(t_n) \to F(t_0)$; en in metrische ruimten volstaat de
rijcontinuïteit (\cref{rem:b3:topology:sequences}).
\end{proof}

\begin{theorem}[Differentiëren onder de integraal]
\label{thm:b3:lebesgue:paramdiff}
Zij $T$ een open interval van $\R$. Stel dat $t \mapsto f(t,x)$
voor bijna elke $x$ differentieerbaar is op $T$, met
\[
\Bigl|\frac{\partial f}{\partial t}(t, x)\Bigr| \leq g(x)
\quad \text{voor alle } t \in T \text{ en bijna elke } x,
\]
met $g$ \omterm{def:b3:lebesgue:l1}{integreerbaar}. Dan is $F$ differentieerbaar op $T$ met
$F'(t) = \int_X \frac{\partial f}{\partial t}(t, x)\,\dd\mu(x)$.
\end{theorem}

\begin{proof}
Leg $t$ vast en zij $h_n \to 0$: de differentiequotiënten
\[
\varphi_n(x) = \frac{f(t + h_n, x) - f(t, x)}{h_n}
\longrightarrow \frac{\partial f}{\partial t}(t,x)
\quad\text{bijna overal},
\]
en de middelwaardeongelijkheid begrenst $\abs{\varphi_n(x)} \leq
\sup_{s}\abs{\partial_tf(s,x)} \leq g(x)$: de gedomineerde
convergentie is van toepassing, en $\frac{F(t + h_n) - F(t)}{h_n}
= \int\varphi_n \to \int\partial_t f(t, \cdot)$.
\end{proof}

\begin{example}[De gammafunctie]\label{ex:b3:lebesgue:gamma}
Stel voor $t > 0$\index{gammafunctie}
\[
\Gamma(t) = \int_0^{+\infty} x^{t-1}\eu^{-x}\,\dd x .
\]
De integraal convergeert: nabij $0$ is $x^{t-1}$ \omterm{def:b3:lebesgue:l1}{integreerbaar}
($t > 0$); en in het oneindige is $x^{t-1}\eu^{-x} \leq
C\eu^{-x/2}$. Partiële integratie (op $[\varepsilon, A]$, met
daarna limieten via de monotone convergentie) geeft de
functionaalvergelijking $\Gamma(t + 1) = t\,\Gamma(t)$, waaruit
$\Gamma(n+1) = n!$: de faculteit geïnterpoleerd. Op elke $\intcc
ab \subseteq \intoo0{+\infty}$ wordt
$\partial_t\bigl(x^{t-1}\eu^{-x}\bigr) = \ln x\cdot
x^{t-1}\eu^{-x}$ gedomineerd door $\abs{\ln x}(x^{a-1} +
x^{b-1})\eu^{-x}$, \omterm{def:b3:lebesgue:l1}{integreerbaar}: dus is $\Gamma$ $\mathcal C^1$
en, per inductie, $\mathcal C^\infty$, met $\Gamma^{(k)}(t) =
\int_0^\infty(\ln x)^kx^{t-1}\eu^{-x}\dd x$. De waarde
$\Gamma(\frac12) = \sqrt\pi$ is de gaussische integraal in
vermomming (\cref{pb:b3:lebesgue:1}).
\end{example}

zich als $\frac2{\pi k}$ --- een harmonische reeks: $\frac{\sin
x}x \notin L^1$. \omterm{def:b3:lebesgue:l1}{Lebesgue-integreerbaarheid} is absolute
\omterm{def:b3:lebesgue:l1}{integreerbaarheid}.}
\end{omfigure}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b3:lebesgue:1}
(a) Toon aan dat een monotone functie $\R \to \R$ \omterm{def:b3:lebesgue:measurable}{borel-meetbaar}
is, en dat een afgeleide (van een overal differentieerbare
functie) \omterm{def:b3:lebesgue:measurable}{borel-meetbaar} is.
(b) Toon aan dat $f \colon X \to \R$ \omterm{def:b3:lebesgue:measurable}{meetbaar} is dan en slechts
dan als $\{f > q\} \in \mathcal A$ voor elke \emph{rationale}
$q$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:lebesgue:2}
Bereken, met volledige verantwoording:
\[
\lim_{n\to\infty}\int_0^{+\infty}
\frac{\cos x}{(1 + x/n)^{n}}\,\dd x,
\qquad
\lim_{n\to\infty}\int_0^1 \frac{n\,x^{n-1}}{1 + x}\,\dd x .
\]
\emph{(Voor de tweede: substitueer $u = x^n$ vóór het
domineren.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lebesgue:3}
(a) Geef een strikte ongelijkheid in het lemma van Fatou.
(b) Geef op drie manieren een $f_n \to 0$ puntsgewijs met $\int
f_n = 1$: ontsnapping in hoogte, in breedte, en naar oneindig.
Welke ene hypothese van de gedomineerde convergentie schendt elk
van hen?
(c) Toon aan dat men in het lemma van Fatou de $\liminf$ aan geen
van beide kanten door $\limsup$ kan vervangen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lebesgue:4}
(a) Toon aan dat $\displaystyle\int_0^{+\infty}\frac{x}{\eu^x -
1}\,\dd x = \sum_{n\geq1}\frac1{n^2} = \frac{\pi^2}6$
\emph{(ontwikkel $\frac1{\eu^x - 1}$ in een meetkundige reeks en
integreer term voor term --- welke stelling staat dat toe?)}.
(b) (De droom van de tweedejaars) Toon aan dat
$\displaystyle\int_0^1 x^{-x}\,\dd x = \sum_{n\geq1}n^{-n}$.
\emph{(Schrijf $x^{-x} = \eu^{-x\ln x} = \sum_k\frac{(-x\ln
x)^k}{k!}$ en bereken $\int_0^1(-x\ln x)^k\dd x$ door $x =
\eu^{-u/(k+1)}$ te substitueren en $\Gamma$ te herkennen.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lebesgue:5}
(a) Toon aan dat een \omterm{def:b3:lebesgue:measurable}{meetbare} $f \geq 0$ met $\int f\,\dd\mu = 0$
voldoet aan $f = 0$ \omterm{def:b3:lebesgue:l1}{bijna overal}. \emph{(Beschouw $\{f \geq 1/n\}$
en de ongelijkheid van Markov: $\mu(\{f \geq a\}) \leq \frac1a\int
f$ --- bewijs die.)}
(b) Toon aan dat een \omterm{def:b3:lebesgue:l1}{integreerbare} $f$ \omterm{def:b3:lebesgue:l1}{bijna overal} eindig is.
(c) Toon aan dat uit $\int_A f\,\dd\mu = 0$ voor \emph{elke}
\omterm{def:b3:lebesgue:measurable}{meetbare} $A$ volgt dat $f = 0$ \omterm{def:b3:lebesgue:l1}{bijna overal}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lebesgue:6}
(a) Pas \cref{thm:b3:lebesgue:riemann} toe om de
\omterm{def:b3:lebesgue:l1}{riemann-integreerbaarheid} te beslissen van: $\mathbf 1_\Q$; de
functie van Thomae; en $\mathbf 1_K$ voor $K$ een dikke
cantorverzameling (\cref{exo:b3:measure:5}).
(b) Toon aan dat $\int_1^{+\infty}\abs{\frac{\sin x}x}\,\dd x =
+\infty$, terwijl $\lim_{A\to\infty}\int_1^A\frac{\sin x}x\,\dd x$
bestaat \emph{(partiële integratie)}: oneigenlijke convergentie
zonder \omterm{def:b3:lebesgue:l1}{integreerbaarheid}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lebesgue:7}
Verantwoord dat $F(t) = \int_0^{+\infty}\eu^{-x^2}\cos(tx)\,\dd x$
$\mathcal C^1$ is op $\R$ en voldoet aan $F'(t) = -\frac t2F(t)$
\emph{(partiële integratie)}; leid af dat $F(t) =
F(0)\,\eu^{-t^2/4}$. (Met $F(0) = \frac{\sqrt\pi}2$ uit de
weekendopgave: de gaussische functie is in wezen haar eigen
fouriergetransformeerde --- \cref{ch:b3:fouriertransform} zal dat
systematiseren.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:lebesgue:8}
(Frullani) Zij $0 < a < b$. Toon aan dat
\[
\int_0^{+\infty}\frac{\eu^{-ax} - \eu^{-bx}}{x}\,\dd x =
\ln\frac ba,
\]
door de integrand als $\int_a^b \eu^{-xt}\,\dd t$ te schrijven en
de verwisseling te verantwoorden met de niet-negatieve theorie
(\cref{cor:b3:lebesgue:additivity} in \omterm{def:b3:topology:continuity}{continue} vorm --- loop op
Tonelli vooruit, of snijd $[a,b]$ in $n$ gelijke stukken en ga
naar de limiet).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lebesgue:9}
Zij $f \geq 0$ \omterm{def:b3:lebesgue:measurable}{meetbaar} op $(X, \mathcal A, \mu)$. Toon aan dat
$\nu(A) = \int_A f\,\dd\mu$ een \omterm{def:b3:measure:measure}{maat} definieert (met
\emph{dichtheid}\index{dichtheid van een maat} $f$ ten opzichte
van $\mu$), en dat $\int g\,\dd\nu = \int gf\,\dd\mu$ voor alle
\omterm{def:b3:lebesgue:measurable}{meetbare} $g \geq 0$ \emph{(bewijs het voor indicatorfuncties, dan
voor \omterm{def:b3:lebesgue:simple}{elementaire functies}, en dan met de monotone convergentie ---
de standaardmachine)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:lebesgue:10}
(Een falen in de trant van Weierstrass) Definieer $f(t) =
\int_0^{+\infty}\frac{\sin(tx)}{x(1 + x^2)}\,\dd x$.
(a) Toon aan dat $f$ goed gedefinieerd en \omterm{def:b3:topology:continuity}{continu} is op $\R$, en
$\mathcal C^1$ met $f'(t) = \int_0^\infty\frac{\cos(tx)}{1 +
x^2}\dd x$ voor elke $t$ --- maar dat \emph{nogmaals} onder de
integraal differentiëren ongeoorloofd is.
(b) Neem aan dat $f'(t) = \frac\pi2\eu^{-t}$ voor $t > 0$ (bewezen
in \cref{ch:b3:residues}): wat is dan
$\int_0^\infty\frac{x\sin(tx)}{1+x^2}\dd x$ voor $t > 0$, en
waarom bevestigt die formule het falen uit (a)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lebesgue:11}
(Lemma van Scheffé) Zij $f_n, f \geq 0$ \omterm{def:b3:lebesgue:l1}{integreerbaar} met $f_n \to
f$ \omterm{def:b3:lebesgue:l1}{bijna overal} en $\int f_n \to \int f$.
(a) Toon aan dat $\int\abs{f_n - f} \to 0$. \emph{(Pas de
gedomineerde convergentie toe op $g_n = (f - f_n)^+ \leq f$ en
schrijf $\int\abs{f_n - f} = 2\int g_n - \int(f - f_n)$.)}
(b) Toon met een voorbeeld aan dat de hypothese $\int f_n \to \int
f$ niet gemist kan worden (een schuivende of samentrekkende bult),
en dat de conclusie voor $f_n$ met wisselend teken zonder controle
op de absolute waarde faalt: $f_n = n\mathbf 1_{\intoc0{1/n}} -
n\mathbf 1_{\intoc{-1/n}0}$ heeft $f_n \to 0$ \omterm{def:b3:lebesgue:l1}{bijna overal} en
$\int f_n = 0 \to 0$, en toch $\int\abs{f_n} = 2$.
(c) Toepassing (dichtheden): convergeren kansdichtheden $p_n \to
p$ \omterm{def:b3:lebesgue:l1}{bijna overal}, dan is automatisch $\int\abs{p_n - p} \to 0$:
puntsgewijze convergentie van dichtheden is $L^1$-convergentie ---
een gratis opwaardering van de convergentie.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:lebesgue:12}
Klassieke limieten, met volledige verantwoording via de monotone
en de gedomineerde convergentie:
\[
\text{(a)}\ \lim_{n\to\infty}\int_0^n\Bigl(1 -
\frac xn\Bigr)^n\eu^{x/2}\,\dd x, \qquad
\text{(b)}\ \lim_{n\to\infty}\int_0^1\frac{n\,x^{n-1}}{1 +
x}\,\dd x,
\]
\[
\text{(c)}\ \lim_{n\to\infty}\int_0^\infty
\frac{\dd x}{(1 + x/n)^n\,x^{1/n}} .
\]
\emph{(Voor (a): $(1 - x/n)^n \nearrow \eu^{-x}$ voor vaste $x$ ---
bewijs de monotonie via $\log$; voor (b): integreer partieel of
substitueer $x = u^{1/n}$ en herken een concentratie bij de rand;
voor (c): zoek een \omterm{def:b3:lebesgue:l1}{integreerbare} dominant die voor alle $n \geq 2$
geldt, door bij $x = 1$ te splitsen.)}
\end{exercise}

\section{Probleem: twee beroemde integralen}

\begin{problem}[{Weekendopgave --- de gaussische integraal en de
integraal van Dirichlet, met parameters alleen}]
\label{pb:b3:lebesgue:1}
Twee integralen beheersen de toegepaste analyse:
\[
G = \int_{-\infty}^{+\infty}\eu^{-x^2}\dd x = \sqrt\pi,
\qquad
D = \int_0^{+\infty}\frac{\sin x}{x}\,\dd x = \frac\pi2
\]
(de tweede als oneigenlijke integraal,
\cref{ex:b3:lebesgue:riemannexamples}). We bewijzen beide met
uitsluitend het gereedschap van dit hoofdstuk.

\medskip
\noindent\textbf{Deel I --- De gaussische integraal.} Stel voor $t
\geq 0$
\[
A(t) = \Bigl(\int_0^t\eu^{-x^2}\dd x\Bigr)^{2},
\qquad
B(t) = \int_0^1\frac{\eu^{-t^2(1 + x^2)}}{1 + x^2}\,\dd x .
\]
\begin{enumerate}
  \item Verantwoord dat $A$ en $B$ $\mathcal C^1$ zijn op
        $\intoo0{+\infty}$ en bereken $A'$ en $B'$; toon aan dat
        $A'(t) + B'(t) = 0$. \emph{(Substitueer in $B'$: $u =
        tx$.)}
  \item Bereken $A(0) + B(0)$ en $\lim_{t\to+\infty}(A + B)(t)$
        --- verantwoord de limiet onder de integraal in $B$.
  \item Besluit dat $\int_0^\infty \eu^{-x^2}\dd x =
        \frac{\sqrt\pi}2$, dus $G = \sqrt\pi$, en leid af dat
        $\Gamma(\tfrac12) = \sqrt\pi$ \emph{(substitueer $x = u^2$
        in $\Gamma(\frac12)$)}.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De integraal van Dirichlet.} Stel
voor $t \geq 0$
\[
F(t) = \int_0^{+\infty}\eu^{-tx}\,\frac{\sin x}{x}\,\dd x .
\]
\begin{enumerate}[resume]
  \item Toon aan dat de integraal die $F(t)$ definieert voor elke
        $t > 0$ als lebesgue-integraal convergeert, en voor $t =
        0$ als oneigenlijke integraal; toon aan dat $D =
        \lim_{A\to\infty}\int_0^A\frac{\sin x}x\dd x$ bestaat
        \emph{(partiële integratie op $[\pi, A]$)}.
  \item Toon aan dat $F$ $\mathcal C^1$ is op $\intoo0{+\infty}$
        met
        \[
        F'(t) = -\int_0^{+\infty}\eu^{-tx}\sin x\,\dd x
        = -\frac{1}{1 + t^2}
        \]
        \emph{(dominatie op $[t_0, \infty)$ voor elke $t_0 > 0$;
        de laatste integraal met twee partiële integraties of met
        complexe exponentiëlen)}.
  \item Toon aan dat $F(t) \to 0$ als $t \to +\infty$, en leid af
        dat $F(t) = \frac\pi2 - \arctan t$ op $\intoo0{+\infty}$.
  \item Het delicate punt: $D = \lim_{t\to0^+}F(t)$. Bewijs het
        met uniforme controle op de staart: toon met partiële
        integratie aan dat voor $0 \leq t \leq 1$ en $A \geq \pi$
        \[
        \Bigl|\int_A^{+\infty}\eu^{-tx}\frac{\sin
        x}x\,\dd x\Bigr| \leq \frac{C}{A}
        \]
        met $C$ onafhankelijk van $t$ \emph{(differentieer
        $\frac{\eu^{-tx}}x$ en begrens $\abs{\cos}$ door $1$; merk
        op dat $t\eu^{-tx} \leq 1/x\cdot(tx\eu^{-tx})$ met
        $\sup_{u\geq0}u\eu^{-u} < 1$)}; splits daarna $F(t) - D$
        in $[0, A]$ (waar de gedomineerde convergentie werkt als
        $t \to 0$) en $[A, \infty)$.
  \item Besluit: $D = \frac\pi2$.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Dividenden.}
\begin{enumerate}[resume]
  \item Bereken $\int_0^{+\infty}\frac{\sin^2x}{x^2}\,\dd x$
        \emph{(integreer partieel en herleid tot $D$ via $\sin 2x
        = 2\sin x\cos x$)}.
  \item Bereken $\int_0^{+\infty}\frac{1 - \cos x}{x^2}\,\dd x$,
        en controleer of de twee resultaten met elkaar stroken.
  \item Bereken voor $a > 0$ de integralen
        $\int_0^{+\infty}\frac{\sin(ax)}x\dd x$ en
        $\int_{-\infty}^{+\infty}\eu^{-ax^2}\dd x$, en noteer de
        schaalregels (ze worden de werkpaarden van
        \cref{ch:b3:fouriertransform}).
  \item Leg precies uit waarom $D$ niet rechtstreeks met de
        gedomineerde convergentie in $t = 0$ behandeld kon worden
        (er is geen \omterm{def:b3:lebesgue:l1}{integreerbare} dominant op
        $[0,1]\times[0,\infty)$), en waarom het splitsen van de
        staart in vraag 7 het eerlijke alternatief is --- dat
        patroon (``uniforme \omterm{def:b3:lebesgue:l1}{integreerbaarheid} van de staarten'')
        keert overal in de analyse terug.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De gammafunctie volgens Bohr en
Mollerup.} De functie $\Gamma$ (\cref{ex:b3:lebesgue:gamma})
voldoet aan $\Gamma(1) = 1$ en $\Gamma(x+1) = x\Gamma(x)$ --- maar
dat doen oneindig veel andere functies ook (vermenigvuldig met een
willekeurige $1$-periodieke golf). Eén convexiteitsvoorwaarde legt
$\Gamma$ eenduidig vast, en dan volgen haar diepere identiteiten
mechanisch. Een positieve functie $f$ op een interval heet
\emph{logaritmisch convex} als $\log f$ convex is.
\begin{enumerate}[resume]
  \item Toon aan dat logaritmisch convex convex impliceert, dat
        producten van logaritmisch convexe functies en hun
        samenstellingen met affiene afbeeldingen logaritmisch
        convex zijn, en --- via de ongelijkheid van Hölder voor
        twee functies, $\int\abs{uv} \leq
        \bigl(\int\abs u^p\bigr)^{1/p}\bigl(\int\abs
        v^q\bigr)^{1/q}$, rechtstreeks uit de ongelijkheid van
        Young bewezen --- dat $\Gamma$ logaritmisch convex is op
        $\intoo0\infty$.
  \item (Hellingslemma) Zij $g$ convex op $\intoo0\infty$ met
        $g(n+1) - g(n) = \log n$ voor elk geheel getal $n \geq 1$.
        Vergelijk voor $x \in \intoc01$ en $n \geq 2$ de hellingen
        van $g$ over $[n-1, n]$, $[n, n+x]$ en $[n, n+1]$, en leid
        af dat
        \[
        x\log(n-1) \;\leq\; g(n + x) - g(n) \;\leq\; x\log n .
        \]
  \item (Bohr--Mollerup) Zij $f > 0$ met $f(1) = 1$, $f(x+1) =
        xf(x)$ en $\log f$ convex. Rol de recursie af tot $f(n +
        x) = x(x+1)\cdots(x + n - 1)\,f(x)$ en $f(n) = (n-1)!$, en
        leid met vraag 14 af dat voor $x \in \intoc01$
        \[
        f(x) = \lim_{n\to\infty}
        \frac{n!\,n^x}{x(x+1)\cdots(x+n)} :
        \]
        $f$ is uniek, dus $f = \Gamma$, en de \emph{limietformule
        van Gauss} geldt (breid met de recursie uit tot alle $x >
        0$).
  \item Definieer de \emph{bètafunctie} $B(x, y) =
        \int_0^1t^{x-1}(1-t)^{y-1}\,\dd t$ (met $x, y > 0$).
        Bewijs de convergentie, de recursie $B(x+1, y) =
        \frac{x}{x+y}\,B(x, y)$ (partiële integratie), en $B(1, y)
        = \frac1y$.
  \item Toon aan dat $x \mapsto B(x, y)$ logaritmisch convex is
        (opnieuw Hölder), en pas Bohr--Mollerup toe op
        \[
        f(x) = \frac{B(x, y)\,\Gamma(x + y)}{\Gamma(y)}
        \]
        om de \emph{formule van Euler} te besluiten: $B(x, y) =
        \dfrac{\Gamma(x)\Gamma(y)}{\Gamma(x+y)}$ --- zonder ook
        maar één dubbele integraal.
  \item Bereken $B(\frac12, \frac12)$ rechtstreeks (substitueer $t
        = \sin^2\theta$) en leid af dat $\Gamma(\frac12) =
        \sqrt\pi$: de gaussische integraal van Deel I, uit zuivere
        convexiteit teruggevonden. Vergelijk de twee bewijzen in
        telkens één zin.
  \item (Verdubbeling van Legendre) Toon aan dat
        \[
        g(x) = \frac{2^{x-1}}{\sqrt\pi}\,
        \Gamma\Bigl(\frac x2\Bigr)
        \Gamma\Bigl(\frac{x+1}2\Bigr)
        \]
        aan de drie hypothesen van Bohr--Mollerup voldoet, en
        besluit dat $g = \Gamma$, dat wil zeggen $\Gamma(2z) =
        \frac{2^{2z-1}}{\sqrt\pi}\,\Gamma(z)\,\Gamma(z +
        \tfrac12)$ voor alle $z > 0$.
  \item Leid de gesloten vorm $\Gamma\bigl(n + \tfrac12\bigr) =
        \dfrac{(2n)!}{4^n\,n!}\sqrt\pi$ af, en bewijs met het
        hellingslemma toegepast op $\log\Gamma$ rond grote gehele
        getallen de asymptotiek
        \[
        \frac{\Gamma(n + \frac12)}{\Gamma(n)\,\sqrt n}
        \longrightarrow 1 .
        \]
  \item Combineer de laatste twee vragen tot de asymptotiek van de
        centrale binomiaalcoëfficiënt
        \[
        \binom{2n}{n} \sim \frac{4^n}{\sqrt{\pi n}},
        \]
        en controleer numeriek voor $n = 10$ ($\binom{20}{10} =
        184756$, tegenover $4^{10}/\sqrt{10\pi} \approx 187079$:
        verhouding $\approx 0.988$).
  \item (Synthese) De constante $\sqrt\pi$ is nu verschenen als de
        gaussische integraal (Deel I), als $B(\frac12, \frac12)$
        (vraag 18) en binnen de verdubbeling (vraag 19); en de
        schatting van de centrale binomiaalcoëfficiënt loopt
        vooruit op zowel Stirling (de weekendopgave van
        \cref{ch:b3:product}) als de Moivre--Laplace. Breng de
        verbanden in kaart: welke uitspraken zijn met welke
        gelijkwaardig, en wat draagt elke techniek ---
        differentiëren onder de integraal tegenover convexiteit
        --- bij dat de andere niet kan?
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Nog drie dividenden.}
\begin{enumerate}[resume]
  \item (Wallis, via de bètafunctie) Substitueer voor $p > -1$ de
        variabele $t = \sin^2\theta$ om aan te tonen dat
        \[
        W_p = \int_0^{\pi/2}\sin^p\theta\,\dd\theta
        = \frac12\,B\Bigl(\frac{p+1}2, \frac12\Bigr),
        \]
        en leid uit de bètarecursie (vraag 16) af dat $W_{n+2} =
        \frac{n+1}{n+2}\,W_n$ voor gehele $n \geq 0$. Bereken
        $W_{2n}$ en $W_{2n+1}$ in gesloten vorm, toon met
        insluiting aan dat $W_{2n+1}/W_{2n} \to 1$, en besluit met
        het \emph{product van Wallis}
        \[
        \frac\pi2 = \lim_{n\to\infty}
        \prod_{k=1}^{n}\frac{4k^2}{4k^2 - 1} .
        \]
  \item (De gaussische functie ontmoet een frequentie) Stel voor
        $b \in \R$
        \[
        \Phi(b) = \int_{-\infty}^{+\infty}
        \eu^{-x^2}\cos(2bx)\,\dd x .
        \]
        Toon aan dat $\Phi$ $\mathcal C^1$ is op $\R$, dat een
        partiële integratie de differentiaalvergelijking $\Phi'(b)
        = -2b\,\Phi(b)$ oplevert, en besluit
        \[
        \Phi(b) = \sqrt\pi\,\eu^{-b^2} :
        \]
        de gaussische functie reproduceert zichzelf onder deze
        transformatie --- de ene identiteit waarop
        \cref{ch:b3:fouriertransform} zal draaien.
  \item (De integraal van Frullani) Toon voor $0 < a < b$ aan dat
        \[
        \int_0^{+\infty}
        \frac{\eu^{-ax} - \eu^{-bx}}{x}\,\dd x
        = \log\frac ba ,
        \]
        door naar de parameter $a$ te differentiëren (verantwoord
        de dominatie op elke $\intco{a_0}{+\infty}$ met $a_0 > 0$,
        en bepaal de constante door $a \to b$ te laten gaan). Waar
        precies heeft de integrand haar ophefbare singulariteit in
        $x = 0$ nodig?
\end{enumerate}
\end{problem}
