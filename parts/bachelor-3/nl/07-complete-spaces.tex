\chapter{Volledige ruimten: Baire, Ascoli, Stone--Weierstrass}
\label{ch:b3:complete}

\omterm{def:b3:complete:complete}{Volledigheid} --- elke Cauchyrij convergeert --- is de eigenschap
waarmee de analyse objecten \emph{voortbrengt}: vaste punten van
contracties, sommen van reeksen, oplossingen van vergelijkingen
verkregen als limiet. Dit hoofdstuk brengt de drie grote
bestaansmachines van de metrische theorie samen. De \emph{stelling
van Baire} laat zien dat een volledige ruimte geen aftelbare
vereniging van verwaarloosbare stukken kan zijn, en tovert
objecten tevoorschijn (\omterm{def:b3:topology:continuity}{continue} \omterm{thm:b3:complete:nowherediff}{nergens differentieerbare
functies}!) uit een redenering die niet veel meer dan tellen is.
\emph{Arzelà--Ascoli} bepaalt de \omterm{def:b3:topology:compact}{compacte} deelverzamelingen van
$\mathcal C(K)$ en is het compactheidswerkpaard van de analyse ---
de weekendopgave bewijst er de bestaansstelling van Peano voor
differentiaalvergelijkingen mee. \emph{Stone--Weierstrass} toont
aan dat veeltermen, en veel meer, \omterm{def:b3:topology:interior}{dicht} liggen in $\mathcal C(K)$:
benaderen wordt een \omterm{def:b3:galois:algebraic}{algebraïsche} controle. Onderweg construeren we
\omterm{thm:b3:complete:completion}{vervollediging} en bewijzen we de voortzettingsstelling voor
uniform \omterm{def:b3:topology:continuity}{continue afbeeldingen}, het dagelijks brood van
\cref{ch:b3:lp,ch:b3:hilbert,ch:b3:fouriertransform}.

\section{Volledige ruimten, vervollediging, voortzetting}

\begin{definition}\label{def:b3:complete:complete}
Een metrische ruimte heet \emph{volledig}\index{volledige
metrische ruimte} als elke Cauchyrij convergeert (bachelorjaar 2:
$\R^n$ is volledig; $\mathcal C(\intcc01)$ met
$\norm\cdot_\infty$ is volledig). Een gesloten deelverzameling van
een volledige ruimte is volledig; en een volledige deelverzameling
van een willekeurige metrische ruimte is gesloten.
\end{definition}

\begin{proof}
Voor de twee uitspraken: een Cauchyrij uit de gesloten $F$
convergeert in $X$, en haar limiet, adherent aan $F$, ligt in $F$;
en een in $X$ convergerende rij uit een volledige $A$ is Cauchy,
convergeert dus in $A$, en limieten zijn uniek.
\end{proof}

\begin{theorem}[Voortzetting van uniform continue afbeeldingen]
\label{thm:b3:complete:extension}
Zij $D \subseteq X$ \omterm{def:b3:topology:interior}{dicht}, $Y$ \omterm{def:b3:complete:complete}{volledig}, en $f \colon D \to Y$
uniform \omterm{def:b3:topology:continuity}{continu}. Dan zet $f$ zich op \emph{precies één} manier
voort tot een \omterm{def:b3:topology:continuity}{continue} $\bar f \colon X \to Y$, en $\bar f$ is
uniform \omterm{def:b3:topology:continuity}{continu}.\index{voortzettingsstelling (uniforme
continuïteit)}
\end{theorem}

\begin{proof}
Eenduidigheid: twee \omterm{def:b3:topology:continuity}{continue} voortzettingen vallen samen op de
\omterm{def:b3:topology:interior}{dichte} $D$ en dus overal (de verzameling $\{g = h\}$ is gesloten:
het origineel van de gesloten diagonaal onder $x\mapsto(g(x),
h(x))$). Bestaan: kies voor $x \in X$ een rij $d_n \to x$ met $d_n
\in D$. De rij $(f(d_n))$ is Cauchy: bij gegeven $\varepsilon$
levert de uniforme \omterm{def:b3:topology:continuity}{continuïteit} een $\delta$ met $d(u,v) < \delta
\Rightarrow d(f(u), f(v)) < \varepsilon$, en $(d_n)$ is Cauchy.
Definieer $\bar f(x) = \lim f(d_n)$; de limiet hangt niet van de
gekozen rij af (verweef er twee). $\bar f$ zet $f$ voort
(constante rijen) en erft de continuïteitsmodulus: is $d(x, x') <
\delta$, dan geeft benadering van beide door punten van $D$ op
afstand $< \frac{\delta - d(x,x')}2$ in de limiet $d(\bar f(x),
\bar f(x')) \leq \varepsilon$ --- $\bar f$ is uniform \omterm{def:b3:topology:continuity}{continu}.
\end{proof}

\begin{theorem}[Vervollediging]\label{thm:b3:complete:completion}
Elke metrische ruimte $X$ bedt zich isometrisch in als \omterm{def:b3:topology:interior}{dichte}
deelverzameling van een \omterm{def:b3:complete:complete}{volledige metrische ruimte} $\hat X$, uniek
op een isometrie na die $X$ puntsgewijs vasthoudt: haar
\emph{vervollediging}\index{vervollediging van een metrische
ruimte}.
\end{theorem}

\begin{proof}
\emph{Bestaan.} Zij $\mathcal X$ de verzameling Cauchyrijen van
$X$, met de pseudo-afstand
\[
D\bigl((x_n), (y_n)\bigr) = \lim_n d(x_n, y_n),
\]
waarbij de limiet bestaat omdat $\abs{d(x_n, y_n) - d(x_m, y_m)}
\leq d(x_n, x_m) + d(y_n, y_m)$ de reële rij Cauchy maakt. Stel
$\hat X = \mathcal X/{\sim}$, waarbij rijen op $D$-afstand $0$
vereenzelvigd worden; $D$ daalt af tot een afstand. Bed $X$ in via
de constante rijen: een isometrie met \omterm{def:b3:topology:interior}{dicht} beeld (een Cauchyrij
wordt in $D$ benaderd door de constanten gebouwd op haar eigen
termen: $D\bigl((x_n), (x_k)_{\rm const}\bigr) = \lim_n d(x_n,
x_k) \to 0$ als $k \to \infty$, wegens de Cauchy-eigenschap).
\omterm{def:b3:complete:complete}{Volledigheid} van $\hat X$: zij $(\xi^k)$ Cauchy in $\hat X$; kies
wegens de dichtheid $x_k \in X$ met $D(\xi^k, x_k) \leq 2^{-k}$;
dan is $(x_k)$ Cauchy in $X$ (driehoeksongelijkheid via de $\xi$),
definieert het een punt $\xi \in \hat X$, en is $D(\xi^k, \xi)
\leq 2^{-k} + D(x_k, \xi) \to 0$ (de afstand van de constante
$x_k$ tot de klasse van $(x_j)_j$ is $\lim_j d(x_k, x_j)$, klein
voor grote $k$).

\emph{Eenduidigheid}: twee \omterm{thm:b3:complete:completion}{vervolledigingen} $\hat X_1$ en $\hat
X_2$ bevatten $X$ \omterm{def:b3:topology:interior}{dicht}; de identiteit van $X$, een isometrie, is
uniform \omterm{def:b3:topology:continuity}{continu} en zet zich dus voort tot $\hat X_1 \to \hat X_2$
(\cref{thm:b3:complete:extension}), nog steeds een isometrie op
een \omterm{def:b3:topology:interior}{dichte verzameling} en dus overal; symmetrisch in de andere
richting, en de samenstellingen houden de \omterm{def:b3:topology:interior}{dichte} $X$ vast: het
zijn de identiteiten.
\end{proof}

\begin{theorem}[Vast punt van Banach]\label{thm:b3:complete:banach}
Zij $X$ \omterm{def:b3:complete:complete}{volledig} en niet-leeg, en $f \colon X \to X$ een
\emph{contractie}: $d(f(x), f(y)) \leq k\,d(x,y)$ met $k < 1$. Dan
heeft $f$ precies één vast punt $x^*$, en convergeert elke \omterm{def:b3:groups:action}{baan}
ernaartoe, met de expliciete snelheid $d(x_n, x^*) \leq
\frac{k^n}{1-k}\,d(x_1, x_0)$.\index{vastepuntstelling van
Banach}
\end{theorem}

\begin{proof}
(Bachelorjaar 2 bewees dit; we noteren het argument van twee
regels opnieuw, zodat dit hoofdstuk op zichzelf staat.) De \omterm{def:b3:groups:action}{baan}
$x_{n+1} = f(x_n)$ voldoet aan $d(x_{n+1}, x_n) \leq k^nd(x_1,
x_0)$ en is dus Cauchy (meetkundige reeks); haar limiet $x^*$ is
vast (\omterm{def:b3:topology:continuity}{continuïteit} van $f$) en uniek, want twee vaste punten
voldoen aan $d \leq k\,d$. De snelheid: sommeer de meetkundige
staart.
\end{proof}

\begin{example}[De identiteit verstoren]
\label{ex:b3:complete:perturbident}
Zij $g \colon \R^d \to \R^d$ $k$-lipschitz met $k < 1$. Dan is
$\varphi = \mathrm{id} + g$ een \emph{\omterm{def:b3:topology:continuity}{homeomorfisme} van $\R^d$
op $\R^d$}. Injectiviteit, met een kwantitatieve modulus:
\[
\norm{\varphi(x) - \varphi(y)} \geq \norm{x - y} -
\norm{g(x) - g(y)} \geq (1 - k)\norm{x - y} .
\]
Surjectiviteit is de vastepuntstelling: $\varphi(x) = y$
oplossen betekent $x = y - g(x)$, en $x \mapsto y - g(x)$ is
een $k$-contractie van het volledige $\R^d$ --- voor elke $y$
bestaat er precies één oplossing $x = \psi(y)$. De getoonde
ongelijkheid maakt de inverse $\psi$ lipschitz met constante
$\frac1{1-k}$: een \omterm{def:b3:topology:continuity}{homeomorfisme}, met expliciete grenzen op
beide moduli. Deze onschuldig ogende uitspraak is de motor
binnen de inverse-functiestelling
(\cref{ch:b3:submanifolds}): nabij een punt waar $Df$
inverteerbaar is, \emph{is} $f$ een inverteerbare lineaire
afbeelding plus een kleine lipschitz-verstoring, en het
voorbeeld van vandaag doet de rest. Het kwantificeert ook de
numerieke robuustheid: een stelsel dat minder wordt verstoord
dan de marge van de inverse, blijft \omterm{def:b3:groups:derived}{oplosbaar}, en de oplossing
verschuift met hoogstens $\frac{1}{1-k}$ maal de verstoring.
\end{example}

\section{De stelling van Baire}

\begin{theorem}[Baire]\label{thm:b3:complete:baire}
In een \omterm{def:b3:complete:complete}{volledige metrische ruimte} is een aftelbare doorsnede van
\omterm{def:b3:topology:interior}{dichte} \omterm{def:b3:topology:topology}{open verzamelingen} \omterm{def:b3:topology:interior}{dicht}. Gelijkwaardig: is $X =
\bigcup_{n}F_n$ met elke $F_n$ gesloten, dan heeft een zekere
$F_n$ een niet-leeg inwendige.\index{stelling van Baire}
\end{theorem}

\begin{proof}
Zij $(U_n)$ \omterm{def:b3:topology:interior}{dicht} en open, en $B_0 = B(x_0, r_0)$ een
willekeurige open bal; we zoeken een punt van $\bigcap U_n$ in
$B_0$. Inductief: omdat $U_{n}$ \omterm{def:b3:topology:interior}{dicht} en open is, snijdt het de
open bal $B_{n-1}$ in een \omterm{def:b3:topology:topology}{open verzameling}, die een gesloten bal
$\bar B(x_n, r_n)$ bevat met $0 < r_n \leq r_{n-1}/2$ en $\bar
B(x_n, r_n) \subseteq B_{n-1}\cap U_n$. De middelpunten vormen een
Cauchyrij ($x_m \in B_n$ voor $m \geq n$, en de stralen gaan naar
$0$); de limiet $x$ ligt in elke $\bar B(x_n, r_n)$
(geslotenheid), en dus in elke $U_n$ en in $B_0$. Voor de tweede
vorm: heeft geen enkele $F_n$ een inwendige, dan zijn de $U_n = X
\setminus F_n$ open en \omterm{def:b3:topology:interior}{dicht}, en ontsnapt een punt van $\bigcap
U_n$ aan $\bigcup F_n = X$: absurd.
\end{proof}

\begin{remark}\label{rem:b3:complete:meagre}
Woordenschat: een verzameling heet \emph{nergens \omterm{def:b3:topology:interior}{dicht}} als haar
\omterm{def:b3:topology:interior}{afsluiting} een leeg inwendige heeft, en \emph{mager}\index{magere
verzameling} (van de eerste categorie) als ze een aftelbare
vereniging van nergens \omterm{def:b3:topology:interior}{dichte verzamelingen} is. Baire zegt: \emph{een
\omterm{def:b3:complete:complete}{volledige metrische ruimte} is niet \omterm{rem:b3:complete:meagre}{mager} in zichzelf}, en het
complement van een \omterm{rem:b3:complete:meagre}{magere verzameling} is \omterm{def:b3:topology:interior}{dicht}. ``\omterm{rem:b3:complete:meagre}{Mager}'' is een
begrip van kleinheid dat loodrecht op de maat staat
(\cref{ch:b3:measure} zal \omterm{rem:b3:complete:meagre}{magere verzamelingen} van volle maat
opleveren), en baireargumenten bewijzen \emph{bestaan door
overvloed}: om één object zonder eigenschap P aan te wijzen, toon
je aan dat de objecten mét P een \omterm{rem:b3:complete:meagre}{magere verzameling} vormen.
\end{remark}

\begin{corollary}\label{cor:b3:complete:bairecors}
(a) $\R$ is overaftelbaar. (b) $\Q$ is geen aftelbare doorsnede
van open deelverzamelingen van $\R$, en een niet-lege \omterm{def:b3:complete:complete}{volledige
metrische ruimte} zonder geïsoleerde punten is overaftelbaar.
\end{corollary}

\begin{proof}
(a) Was $\R = \bigcup_{x}\{x\}$ over een aftelbare verzameling,
dan had een zeker singleton een inwendige. (b) Was $\Q =
\bigcap_n V_n$ met $V_n$ open (noodzakelijk \omterm{def:b3:topology:interior}{dicht}, want $\supseteq
\Q$), dan vormen de $V_n$ en de complementen $\R\setminus\{q\}$
met $q \in \Q$ een aftelbare familie \omterm{def:b3:topology:interior}{dichte} \omterm{def:b3:topology:topology}{open verzamelingen} met
lege doorsnede --- in strijd met Baire. Is $X$ \omterm{def:b3:complete:complete}{volledig} zonder
geïsoleerde punten en aftelbaar, dan toont $X = \bigcup_{x \in
X}\{x\}$ haar als een aftelbare vereniging gesloten verzamelingen
met leeg inwendige (geen geïsoleerde punten): opnieuw Baire.
\end{proof}

\begin{theorem}[De monsters van Weierstrass bestaan]
\label{thm:b3:complete:nowherediff}
Er bestaan \omterm{def:b3:topology:continuity}{continue} functies op $\intcc01$ die in geen enkel punt
differentieerbaar zijn. Sterker: de verzameling $f \in \mathcal
C(\intcc01)$ die zelfs maar in één punt een (eindige) afgeleide
hebben, is \omterm{rem:b3:complete:meagre}{mager} in $(\mathcal C(\intcc01),
\norm\cdot_\infty)$.\index{nergens differentieerbare functie}
\end{theorem}

\begin{proof}
Stel voor $n \geq 1$
\[
F_n = \Bigl\{f : \exists x \in \intcc01,\
\forall h \neq 0 \text{ met } x + h \in \intcc01,\
\abs{f(x+h) - f(x)} \leq n\abs h \Bigr\}.
\]
Is $f$ differentieerbaar in $x$, dan is $f \in F_n$ voor zekere
$n$: het quotiënt $\abs{f(x+h)-f(x)}/\abs h$ is begrensd voor
$\abs h \leq \delta$ (differentieerbaarheid: het gaat naar
$\abs{f'(x)}$) en begrensd door $2\norm f_\infty/\delta$ voor
$\abs h \geq \delta$. Dus bevat $\bigcup F_n$ alle functies die
ergens differentieerbaar zijn, en volstaat het aan te tonen dat
elke $F_n$ gesloten is met leeg inwendige.

\emph{Gesloten}: zij $f_k \to f$ uniform met $f_k \in F_n$ en
getuigen $x_k \to x$ (\omterm{def:b3:topology:compact}{compactheid}, na extractie). Voor $h$ met $x
+ h \in \intcc01$: kies $h_k \to h$ met $x_k + h_k \in \intcc01$
(bijvoorbeeld $h_k = h + x - x_k$, afgeknipt); dan is $\abs{f(x +
h) - f(x)} = \lim \abs{f_k(x_k + h_k) - f_k(x_k)} \leq \lim
n\abs{h_k} = n\abs h$, met de uniforme convergentie en de
\omterm{def:b3:topology:continuity}{continuïteit} van $f$ in de betrokken punten: dus $f \in F_n$.

\emph{Leeg inwendige}: gegeven $f \in F_n$ en $\varepsilon > 0$
zoeken we een $g$ met $\norm{g - f}_\infty \leq \varepsilon$ en $g
\notin F_n$. Benader eerst $f$ tot op $\varepsilon/2$ door een
stuksgewijs affiene $\varphi$ (uniforme \omterm{def:b3:topology:continuity}{continuïteit}: interpoleer
op een fijn rooster), met hellingen begrensd door zekere $M$. Tel
er een kleine zaagtand bij op: $g = \varphi +
\frac\varepsilon2\,s_N$, waarbij $s_N(x)$ de $\frac1N$-periodieke
zigzag met amplitude $1$ en helling $\pm 2N$ is. In elke $x$ is er
aan één kant een willekeurig kleine $h$ waarbij de zaagtand op
$[x, x+h]$ een helling $\pm 2N\cdot\frac\varepsilon2 =
\pm\varepsilon N$ bijdraagt: het differentiequotiënt van $g$
overschrijdt $\varepsilon N - M > n$ voor grote $N$. Dus is $g
\notin F_n$, op willekeurig kleine uniforme afstand $\leq
\varepsilon$ van $f$. Besluit: $\bigcup F_n$ is \omterm{rem:b3:complete:meagre}{mager}, en volgens
Baire is haar complement --- bestaande uit \omterm{thm:b3:complete:nowherediff}{nergens
differentieerbare functies} --- \omterm{def:b3:topology:interior}{dicht} in $\mathcal C(\intcc01)$:
zulke functies bestaan \emph{in overvloed}.
\end{proof}

\section{Arzelà--Ascoli}

Overal is $K$ een \emph{\omterm{def:b3:topology:compact}{compacte}} metrische ruimte en $\mathcal
C(K) = \mathcal C(K, \R^d)$, met $\norm f_\infty = \sup_K
\norm{f(x)}$: een volledige ruimte (uniforme limieten van \omterm{def:b3:topology:continuity}{continue}
functies zijn \omterm{def:b3:topology:continuity}{continu} --- bachelorjaar 2).

\begin{definition}\label{def:b3:complete:equicontinuous}
Een familie $\mathcal F \subseteq \mathcal C(K)$ heet
\emph{equicontinu}\index{equicontinuïteit} als er voor elke
$\varepsilon > 0$ een $\delta > 0$ is zodanig dat
\[
d(x, y) < \delta \implies \norm{f(x) - f(y)} < \varepsilon
\quad\text{voor \emph{alle} } f \in \mathcal F
\]
(één $\delta$ voor de hele familie --- bijvoorbeeld elke familie
met een gemeenschappelijke lipschitzconstante of een
gemeenschappelijke höldermodulus), en \emph{puntsgewijs begrensd}
als $\sup_{f}\norm{f(x)} < \infty$ voor elke $x$.
\end{definition}

\begin{theorem}[Arzelà--Ascoli]\label{thm:b3:complete:ascoli}
Een deelverzameling $\mathcal F \subseteq \mathcal C(K)$ is
relatief \omterm{def:b3:topology:compact}{compact} (heeft \omterm{def:b3:topology:compact}{compacte} \omterm{def:b3:topology:interior}{afsluiting}) dan en slechts dan
als ze \omterm{def:b3:complete:equicontinuous}{equicontinu} en puntsgewijs begrensd is. In het bijzonder
heeft elke \omterm{def:b3:complete:equicontinuous}{equicontinue}, puntsgewijs begrensde \emph{rij} een
uniform convergente deelrij.\index{stelling van
Arzelà--Ascoli}
\end{theorem}

\begin{proof}
($\Leftarrow$) Zij $(f_k)$ een rij uit $\mathcal F$. Een \omterm{def:b3:topology:compact}{compacte}
metrische $K$ is \emph{separabel}: voor elke $n$ overdekken
eindig veel ballen met straal $\frac1n$ de ruimte $K$
(\cref{thm:b3:topology:metriccompact}); hun middelpunten vormen
een aftelbare \omterm{def:b3:topology:interior}{dichte verzameling} $D = \{x_1, x_2, \dots\}$. Haal
met de puntsgewijze begrensdheid en Bolzano--Weierstrass
achtereenvolgens deelrijen tevoorschijn die in $x_1$ convergeren,
daarna ook in $x_2$, enzovoort, en neem de
\emph{diagonaaldeelrij} $(g_j)$: die convergeert in elk punt van
$D$. De \omterm{def:b3:complete:equicontinuous}{equicontinuïteit} tilt dat op tot uniform Cauchy: kies bij
gegeven $\varepsilon$ een $\delta$ als in de definitie, overdek
$K$ met eindig veel ballen $B(x_i, \delta)$ met $x_i \in D$ ($i
\leq m$), en kies $J$ zo groot dat $\norm{g_j(x_i) -
g_{j'}(x_i)} < \varepsilon$ voor $j, j' \geq J$ en $i \leq m$.
Voor willekeurige $x \in B(x_i, \delta)$:
\[
\norm{g_j(x) - g_{j'}(x)}
\leq \norm{g_j(x) - g_j(x_i)} + \norm{g_j(x_i) - g_{j'}(x_i)}
+ \norm{g_{j'}(x_i) - g_{j'}(x)} < 3\varepsilon .
\]
Dus is $(g_j)$ uniform Cauchy en convergeert ze in het volledige
$\mathcal C(K)$. Elke rij uit $\mathcal F$ heeft dus een
convergente deelrij: $\bar{\mathcal F}$ is (rijcompact, en dus
volgens \cref{thm:b3:topology:metriccompact}) \omterm{def:b3:topology:compact}{compact}.

($\Rightarrow$) Is $\bar{\mathcal F}$ \omterm{def:b3:topology:compact}{compact}, dan is de
puntsgewijze begrensdheid duidelijk (evalueren is \omterm{def:b3:topology:continuity}{continu}). Voor
de \omterm{def:b3:complete:equicontinuous}{equicontinuïteit}: overdek $\mathcal F$ met eindig veel ballen
$B(f_i, \varepsilon)$ van $\mathcal C(K)$; elke $f_i$ is uniform
\omterm{def:b3:topology:continuity}{continu} (Heine, \cref{cor:b3:topology:heineborel}), wat een
gemeenschappelijke $\delta$ geeft voor $i \leq m$; en dan is voor
$f \in B(f_i, \varepsilon)$ en $d(x,y) < \delta$:
$\norm{f(x)-f(y)} \leq 2\varepsilon + \norm{f_i(x)-f_i(y)} <
3\varepsilon$.
\end{proof}

\begin{example}\label{ex:b3:complete:ascoliexamples}
De gesloten eenheidsbal van $\mathcal C(\intcc01)$ is \emph{niet}
\omterm{def:b3:topology:compact}{compact} ($f_n(x) = x^n$ heeft geen uniform convergente deelrij:
de puntsgewijze limiet is discontinu), en inderdaad is $(x^n)$
niet \omterm{def:b3:complete:equicontinuous}{equicontinu} in $1$. Daartegenover is $\{f : \norm f_\infty
\leq 1,\ \operatorname{Lip}(f) \leq 1\}$ wél \omterm{def:b3:topology:compact}{compact}: begrensd,
\omterm{def:b3:complete:equicontinuous}{equicontinu} met lipschitzconstante $1$, en gesloten. Ascoli
verklaart \emph{waarom} de \omterm{def:b3:topology:compact}{compactheid} in oneindige dimensie
faalt (Riesz, bachelorjaar 2) en wat je moet toevoegen om haar te
herstellen: een uniforme continuïteitsmodulus.
\end{example}

\section{Stone--Weierstrass}

\begin{lemma}[Dini]\label{lem:b3:complete:dini}
Zij $K$ \omterm{def:b3:topology:compact}{compact} en $(f_n)$ een \emph{monotone} rij \omterm{def:b3:topology:continuity}{continue} reële
functies die \emph{puntsgewijs} naar een \omterm{def:b3:topology:continuity}{continue} $f$
convergeert. Dan is de convergentie uniform.\index{stelling van
Dini}
\end{lemma}

\begin{proof}
Zeg $f_n \uparrow f$; stel $g_n = f - f_n \downarrow 0$, \omterm{def:b3:topology:continuity}{continu}.
Bij gegeven $\varepsilon$ stijgen de \omterm{def:b3:topology:topology}{open verzamelingen} $U_n =
\{g_n < \varepsilon\}$ en overdekken ze $K$ (puntsgewijze
convergentie); haal er een eindige deeloverdekking uit: $K =
U_{n_0}$ voor zekere $n_0$ (stijgende familie), dat wil zeggen $0
\leq g_n < \varepsilon$ overal voor $n \geq n_0$.
\end{proof}

\begin{lemma}\label{lem:b3:complete:sqrt}
Er is een rij \emph{veeltermen} $u_n$ met $u_n(t) \to \sqrt t$
uniform op $\intcc01$.
\end{lemma}

\begin{proof}
Definieer $u_0 = 0$ en $u_{n+1}(t) = u_n(t) + \frac12\bigl(t -
u_n(t)^2\bigr)$: veeltermen. Per inductie is $0 \leq u_n(t) \leq
\sqrt t$ op $\intcc01$: aangenomen voor $n$, is
\[
\sqrt t - u_{n+1}(t) = \bigl(\sqrt t - u_n(t)\bigr)
\Bigl(1 - \tfrac12\bigl(\sqrt t + u_n(t)\bigr)\Bigr) \geq 0,
\]
want $\sqrt t + u_n \leq 2$; en $u_{n+1} \geq u_n \geq 0$. Dus is
$(u_n(t))$ niet-dalend en begrensd door $\sqrt t$: ze convergeert
puntsgewijs, en de limiet $\ell(t)$ voldoet aan $\ell = \ell +
\frac12(t - \ell^2)$, dus $\ell(t) = \sqrt t$, \omterm{def:b3:topology:continuity}{continu}. Dini
(\cref{lem:b3:complete:dini}) tilt dat op tot uniform.
\end{proof}

\begin{theorem}[Stone--Weierstrass, reële versie]
\label{thm:b3:complete:stoneweierstrass}
Zij $K$ een \omterm{def:b3:topology:compact}{compacte} (hausdorffse) ruimte en $\mathcal A \subseteq
\mathcal C(K, \R)$ een \emph{deelalgebra} (gesloten onder sommen,
producten en scalaire veelvouden) die \emph{de constanten bevat}
en \emph{de punten scheidt} (voor $x \neq y$ is er een $f \in
\mathcal A$ met $f(x) \neq f(y)$). Dan ligt $\mathcal A$ \omterm{def:b3:topology:interior}{dicht} in
$(\mathcal C(K,\R), \norm\cdot_\infty)$.\index{stelling van
Stone--Weierstrass}
\end{theorem}

\begin{proof}
Zij $\bar{\mathcal A}$ de \omterm{def:b3:topology:interior}{afsluiting}, opnieuw een algebra
(producten van uniforme limieten convergeren op begrensde
verzamelingen).

\emph{Stap 1: $\bar{\mathcal A}$ is een tralie}, dat wil zeggen
gesloten onder $\max$ en $\min$. Omdat $\max(f,g) = \frac{f + g +
\abs{f-g}}2$ en net zo voor $\min$, volstaat het dat uit $f \in
\bar{\mathcal A}$ volgt $\abs f \in \bar{\mathcal A}$: met $M =
\norm f_\infty > 0$ is $\abs f = M\sqrt{(f/M)^2}$, en
\cref{lem:b3:complete:sqrt} geeft veeltermen $u_n$ met
$u_n\bigl((f/M)^2\bigr) \to \abs f/M$ uniform; en veeltermen in
leden van de algebra (met constante term: de constanten zitten
erin) blijven in $\bar{\mathcal A}$.

\emph{Stap 2: interpolatie in twee punten.} Voor $x \neq y$ en $a,
b \in \R$ is er een $g \in \mathcal A$ met $g(x) = a$ en $g(y) =
b$: neem een $h$ die $x$ en $y$ scheidt en stel $g = a + (b -
a)\frac{h - h(x)}{h(y) - h(x)}$.

\emph{Stap 3.} Zij $f \in \mathcal C(K)$ en $\varepsilon > 0$.
Kies voor elk paar $x, y$ een $g_{x,y} \in \mathcal A$ met
$g_{x,y}(x) = f(x)$ en $g_{x,y}(y) = f(y)$ (stap 2; voor $x = y$
neem je de constante functie $g_{x,x} = f(x)$). Leg $x$ vast: voor
elke $y$ bevat de \omterm{def:b3:topology:topology}{open verzameling} $V_y = \{g_{x,y} < f +
\varepsilon\}$ het punt $y$; de \omterm{def:b3:topology:compact}{compactheid} levert $y_1, \dots,
y_m$ met $K = \bigcup V_{y_j}$, en $h_x = \min_j g_{x, y_j} \in
\bar{\mathcal A}$ (stap 1) voldoet overal aan $h_x < f +
\varepsilon$, met $h_x(x) = f(x)$. Varieer nu $x$: $W_x = \{h_x >
f - \varepsilon\}$ is open en bevat $x$; haal er $x_1, \dots, x_l$
uit die $K$ overdekken, en $h = \max_i h_{x_i} \in \bar{\mathcal
A}$ voldoet aan $f - \varepsilon < h < f + \varepsilon$: dus
$\norm{f - h}_\infty \leq \varepsilon$. Bijgevolg is $f \in
\bar{\mathcal A}$.
\end{proof}

\begin{corollary}\label{cor:b3:complete:weierstrass}
(a) (\emph{Weierstrass}) De veeltermen liggen \omterm{def:b3:topology:interior}{dicht} in $\mathcal
C(\intcc ab, \R)$; en de veeltermen in $n$ veranderlijken liggen
\omterm{def:b3:topology:interior}{dicht} in $\mathcal C(K, \R)$ voor \omterm{def:b3:topology:compact}{compacte} $K \subseteq \R^n$.
(b) (\emph{Complexe versie}) Is $\mathcal A \subseteq \mathcal
C(K, \C)$ een deelalgebra die de constanten bevat, de punten
scheidt en \emph{gesloten is onder toevoeging}, dan ligt ze \omterm{def:b3:topology:interior}{dicht}.
(c) (\emph{Trigonometrische versie}) De trigonometrische
veeltermen $\sum_{\abs n \leq N}c_n\eu^{\iu n t}$ liggen \omterm{def:b3:topology:interior}{dicht} in
de ruimte van \omterm{def:b3:topology:continuity}{continue} $2\pi$-periodieke functies met
$\norm\cdot_\infty$.\index{benaderingsstelling van Weierstrass}
\end{corollary}

\begin{proof}
(a) De veeltermen vormen een algebra met de constanten; en de
coördinaatfuncties scheiden de punten van $\R^n$.
(b) De reële en imaginaire delen $\frac{f + \bar f}2$ en $\frac{f
- \bar f}{2\iu}$ van leden van $\mathcal A$ vormen een reële
algebra $\mathcal A_\R \subseteq \mathcal C(K, \R)$ met de
constanten; ze scheidt de punten (uit $f(x) \ne f(y)$ volgt dat
$\operatorname{Re}f$ of $\operatorname{Im}f$ scheidt). Pas de
reële stelling toe en zet weer samen.
(c) Beschouw de $2\pi$-periodieke functies als $\mathcal C(S^1,
\C)$ (met $S^1 = \R/2\pi\Z$, \omterm{def:b3:topology:compact}{compact}:
\cref{exo:b3:topology:5}); de algebra voortgebracht door
$\eu^{\iu t}$, $\eu^{-\iu t}$ en de constanten is gesloten onder
toevoeging en scheidt de punten van de cirkel ($\eu^{\iu t}$ is er
injectief op). Pas (b) toe.
\end{proof}

\begin{remark}\label{rem:b3:complete:swremark}
De trigonometrische versie repareert, en veralgemeent enorm, het
gat dat het fourierhoofdstuk van bachelorjaar 2 openliet: de
dichtheid van de trigonometrische veeltermen in $(\mathcal C(S^1),
\norm\cdot_2)$ volgt er a fortiori uit ($\norm\cdot_2 \leq
\norm\cdot_\infty$ op de normeringsconstante na), wat van het
fouriersysteem in \cref{ch:b3:hilbert} een orthonormale
\emph{basis} zal maken en Parseval eindelijk in volle
algemeenheid zal bewijzen.
\end{remark}

\section{Probleem: de bestaansstelling van Peano}

\begin{problem}[{Weekendopgave --- bestaan van oplossingen van $x'
= f(t, x)$ zonder Lipschitz}]\label{pb:b3:complete:1}
Cauchy--Lipschitz (bachelorjaar 2; opnieuw bewezen in
\cref{ch:b3:ode}) eist dat $f$ lipschitz is in $x$. Peano (1890):
\emph{de \omterm{def:b3:topology:continuity}{continuïteit} van $f$ levert al het bestaan} --- zij het
niet de eenduidigheid. Wij bewijzen het met eulerpolygonen en
Ascoli. Kader: $f \colon R \to \R^d$ is \omterm{def:b3:topology:continuity}{continu} op de rechthoek $R
= \intcc{t_0 - a}{t_0 + a}\times \bar B(x_0, b) \subseteq
\R\times\R^d$, $M = \sup_R\norm f$, en
\[
T = \min\bigl(a,\ b/M\bigr)
\qquad (\text{is } M = 0 \text{, dan is het probleem triviaal}).
\]

\medskip
\noindent\textbf{Deel I --- Eulerpolygonen.} Verdeel voor $n \geq
1$ het interval $[t_0, t_0 + T]$ door $t_k = t_0 + kT/n$ ($0 \leq
k \leq n$) en definieer $\varphi_n$ stuksgewijs affien:
$\varphi_n(t_0) = x_0$ en, op $[t_k, t_{k+1}]$,
\[
\varphi_n(t) = \varphi_n(t_k) + (t -
t_k)\,f\bigl(t_k, \varphi_n(t_k)\bigr).
\]
\begin{enumerate}
  \item Toon met inductie aan dat $\varphi_n$ goed gedefinieerd
        is, met $\norm{\varphi_n(t) - x_0} \leq M(t - t_0) \leq b$
        op $[t_0, t_0 + T]$ --- zodat de evaluatiepunten in $R$
        blijven. \emph{(Hier komt $T \leq b/M$ binnen.)}
  \item Toon aan dat elke $\varphi_n$ $M$-lipschitz is.
  \item Leid uit Arzelà--Ascoli
        (\cref{thm:b3:complete:ascoli}) af dat een deelrij
        $(\varphi_{n_j})$ uniform op $[t_0, t_0 + T]$ naar een
        zekere $\varphi$ convergeert, zelf $M$-lipschitz met
        $\varphi(t_0) = x_0$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De limiet lost de vergelijking op.}
Definieer het \emph{defect} $\Delta_n(t) = \varphi_n'(t) -
f\bigl(t, \varphi_n(t)\bigr)$ in de punten buiten het rooster.
\begin{enumerate}[resume]
  \item Toon aan dat $f$ uniform \omterm{def:b3:topology:continuity}{continu} is op $R$, en leid af:
        voor elke $\varepsilon > 0$ is er een $n_0$ zodanig dat
        voor $n \geq n_0$ en elke $t$ buiten het rooster
        $\norm{\Delta_n(t)} \leq \varepsilon$. \emph{(Op $(t_k,
        t_{k+1})$ is $\varphi_n'(t) = f(t_k, \varphi_n(t_k))$, en
        ligt $(t, \varphi_n(t))$ op afstand hoogstens $(1 +
        M)\,T/n$ van $(t_k, \varphi_n(t_k))$.)}
  \item Stel de integraalvorm op: voor alle $t$ is
        \[
        \varphi_n(t) = x_0 + \int_{t_0}^{t}
        f\bigl(s, \varphi_n(s)\bigr)\dd s +
        \int_{t_0}^t \Delta_n(s)\,\dd s,
        \]
        waarbij de middelste integrand stuksgewijs \omterm{def:b3:topology:continuity}{continu} is.
  \item Ga langs $(n_j)$ tot de limiet over: toon aan dat $f(s,
        \varphi_{n_j}(s)) \to f(s, \varphi(s))$ \emph{uniform}
        (opnieuw de uniforme \omterm{def:b3:topology:continuity}{continuïteit} van $f$), en besluit
        \[
        \varphi(t) = x_0 + \int_{t_0}^t
        f\bigl(s, \varphi(s)\bigr)\dd s .
        \]
  \item Leid af dat $\varphi$ $\mathcal C^1$ is op $[t_0, t_0 +
        T]$ en $x' = f(t, x)$ met $x(t_0) = x_0$ oplost; en breid
        de constructie uit tot $[t_0 - T, t_0]$ (tijdomkering).
        \emph{Dat is de stelling van Peano.}
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De eenduidigheid faalt werkelijk.}
Beschouw $x' = 2\sqrt{\abs x}$ met $x(0) = 0$, op $\R$.
\begin{enumerate}[resume]
  \item Ga na dat $f(x) = 2\sqrt{\abs x}$ \omterm{def:b3:topology:continuity}{continu} is maar op geen
        enkele \omterm{def:b3:topology:topology}{omgeving} van $0$ lipschitz.
  \item Ga na dat $x \equiv 0$ en, voor elke $c \geq 0$,
        \[
        x_c(t) = \begin{cases} 0 & t \leq c,\\ (t - c)^2 & t
        \geq c,\end{cases}
        \]
        allemaal oplossingen door $(0,0)$ zijn: een continuüm van
        verschillende oplossingen.
  \item Waar loopt het argument met de picarditeratie (het vaste
        punt van Banach) vast voor deze $f$?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Grenzen van de methode.}
\begin{enumerate}[resume]
  \item Toon aan dat de stelling van Peano faalt in oneindige
        dimensie: we nemen het klassieke voorbeeld van Dieudonné
        in de ruimte $c_0$ van nulrijen zonder bewijs aan; bewijs
        in plaats daarvan het eindigdimensionale ingrediënt dat
        daar faalt: de gesloten eenheidsbal van $c_0$ (supnorm) is
        niet \omterm{def:b3:topology:compact}{compact} --- geef een begrensde rij zonder convergente
        deelrij, en leg uit welke stap van Deel I breekt.
  \item Vat samen: welke hypothesen geven bestaan? En welke
        bestaan én eenduidigheid? Formuleer de twee stellingen
        (Peano; Cauchy--Lipschitz) nauwkeurig naast elkaar.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Osgood: eenduidigheid voorbij
Lipschitz.} Zij $\omega\colon\intoo0\infty\to\intoo0\infty$
\omterm{def:b3:topology:continuity}{continu} en niet-dalend, met
\[
\int_0^1\frac{\dd r}{\omega(r)} = +\infty
\qquad(\text{divergentie in } 0),
\]
en stel dat $f$ op $R$ voldoet aan $\norm{f(t, x) - f(t, y)} \leq
\omega\bigl(\norm{x - y}\bigr)$.
\begin{enumerate}[resume]
  \item Ga na dat $\omega(r) = Lr$ voldoet (Lipschitz), dat
        $\omega(r) = r\log\frac1r$ (voor kleine $r$, \omterm{def:b3:topology:continuity}{continu}
        voortgezet) voldoet hoewel het \emph{geen} $O(r)$ is, en
        dat $\omega(r) = 2\sqrt r$ niet voldoet. Bereken de
        integraal in elk geval.
  \item Zij $x_1, x_2$ oplossingen op $[t_0, t_0 + T]$ met
        $x_1(t_0) = x_2(t_0)$, en $\delta(t) = \norm{x_1(t) -
        x_2(t)}$. Toon uitsluitend met de integraalvormen aan dat
        voor $t_0 \leq s \leq t$
        \[
        \delta(t) \leq \delta(s) +
        \int_s^t\omega\bigl(\delta(v)\bigr)\dd v .
        \]
  \item (Stelling van Osgood) Stel $\delta(t_1) > 0$ voor zekere
        $t_1$, en zij $\tau = \sup\{t \leq t_1 : \delta(t) = 0\}$.
        Stel voor $s \in \intoo\tau{t_1}$: $u(t) = \delta(s) +
        \int_s^t\omega(\delta(v))\,\dd v$. Toon aan dat $\delta
        \leq u$ en $u' = \omega(\delta) \leq \omega(u)$, en leid
        af dat
        \[
        \int_{\delta(s)}^{u(t_1)}\frac{\dd r}{\omega(r)}
        \leq t_1 - s \leq t_1 - \tau .
        \]
        Laat $s \downarrow \tau$ gaan en leid een tegenspraak af
        met de divergentie van de integraal. Besluit:
        \emph{oplossingen door een gemeenschappelijke
        beginvoorwaarde vallen samen} --- eenduidigheid onder de
        voorwaarde van Osgood.
  \item Trek de gevolgen: de eenduidigheid van Cauchy--Lipschitz
        is het geval $\omega(r) = Lr$; de vergelijking $x' =
        x\log\frac1{\abs x}$ (voortgezet met $0$ in $0$) heeft
        eenduidige oplossingen hoewel haar rechterlid in $0$ niet
        lipschitz is; en voor $x' = 2\sqrt{\abs x}$ is de
        convergentie van $\int_0\frac{\dd r}{2\sqrt r}$ precies
        wat een oplossing toestaat om $0$ in eindige tijd te
        verlaten --- breng de waarde van de integraal
        $\int_0^{h^2}\frac{\dd r}{2\sqrt r} = h$ in verband met
        het ontsnappingsgedrag van $x_c$.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Snelheden, schema's, trechters.}
\begin{enumerate}[resume]
  \item (Integraallemma van Grönwall) Zij $e, \eta \geq 0$ \omterm{def:b3:topology:continuity}{continu}
        op $[t_0, t_0+T]$ met $e(t) \leq \int_{t_0}
        ^t\bigl(L\,e(s) + \eta(s)\bigr)\dd s$ voor alle $t$. Toon
        aan dat
        \[
        e(t) \leq \frac{\sup\eta}{L}\bigl(\eu^{L(t - t_0)} -
        1\bigr)
        \]
        \emph{(stel $G(t) = \int_{t_0}^t(Le + \eta)$, merk op dat
        $G' \leq LG + \sup\eta$, en differentieer
        $\eu^{-Lt}G(t)$)}.
  \item (Euler convergeert met een snelheid) Stel nu dat $f$
        $L$-lipschitz is in $x$ en $L'$-lipschitz in $t$ op $R$.
        Bewijs, door de defectgrens van vraag 4 (kwantitatief
        gemaakt: $\norm{\Delta_n} \leq (L' + LM)T/n$) met vraag 17
        te combineren, dat
        \[
        \sup_{[t_0, t_0+T]}\norm{\varphi_n - \varphi}
        \;\leq\; \frac{(L' + LM)\,T}{n}\cdot
        \frac{\eu^{LT} - 1}{L} = O\Bigl(\frac1n\Bigr),
        \]
        waarbij $\varphi$ \emph{de} oplossing is: met
        lipschitzgegevens convergeert de hele rij, met een
        expliciete snelheid --- geen deelrijen nodig. Waarom tilt
        de eenduidigheid convergentie langs deelrijen ook zonder
        deze berekening op tot volledige convergentie?
  \item (Het schema kiest) Toon voor $x' = 2\sqrt{\abs x}$ met
        $x(0) = 0$ aan dat elke eulerpolygoon identiek nul is,
        zodat het schema naar de oplossing $x \equiv 0$
        convergeert; maar gestart in $x(0) = \varepsilon > 0$
        convergeert het (eerst $n \to \infty$, dan $\varepsilon
        \to 0$) naar $t \mapsto t^2$, een \emph{andere} oplossing
        door de oorsprong. De niet-eenduidigheid duikt weer op als
        gevoeligheid van het numerieke schema voor verstoringen.
  \item Toon aan dat de verzameling $\mathcal S$ van \emph{alle}
        oplossingen van $x' = f(t,x)$ met $x(t_0) = x_0$ op $[t_0,
        t_0+T]$ (met waarden in $\bar B(x_0, b)$) niet leeg is
        (Deel II), uniform $M$-lipschitz, en gesloten in
        $\bigl(\mathcal C([t_0, t_0+T], \R^d),
        \norm\cdot_\infty\bigr)$; besluit met Ascoli dat $\mathcal
        S$ \emph{\omterm{def:b3:topology:compact}{compact}} is. (De stelling van Kneser voegt toe
        dat $\mathcal S$ \omterm{def:b3:topology:connected}{samenhangend} is; die bewijzen we niet.)
  \item Ga het verschijnsel van Kneser na op het voorbeeld: toon
        voor $x' = 2\sqrt{\abs x}$ met $x(0) = 0$ op $[0, 1]$ aan
        dat $\mathcal S = \{x_c : c \in \intcc0\infty\}$ met
        $x_\infty \equiv 0$ \emph{(neem voor een willekeurige
        oplossing $c = \sup\{t : x(t) = 0\}$ en integreer $(\sqrt
        x)' = 1$ op $\{x > 0\}$)}, dat $c \mapsto x_c$ \omterm{def:b3:topology:continuity}{continu} is
        van $\intcc0\infty$ (de \omterm{def:b3:topology:locallycompact}{eenpuntscompactificatie}, met $c =
        \infty$ als limiet aangelijmd) naar $\mathcal C([0,1])$,
        en besluit dat $\mathcal S$ inderdaad \omterm{def:b3:topology:compact}{compact} en
        \omterm{def:b3:topology:connected}{samenhangend} is --- een trechter in de vorm van een
        lijnstuk.
  \item (Bereikbare verzamelingen) Leid uit vraag 20 af dat voor
        elke vaste $t$ de \emph{bereikbare verzameling} $\mathcal
        S(t) = \{x(t) : x \in \mathcal S\}$ \omterm{def:b3:topology:compact}{compact} is; bereken
        haar voor het voorbeeld van vraag 21 en ga na dat ze ook
        \omterm{def:b3:topology:connected}{samenhangend} is: $\mathcal S(t) = \intcc0{t^2}$ --- elke
        tussenliggende toestand wordt door een zekere oplossing
        bereikt.
\end{enumerate}

\medskip
\noindent\textbf{Deel VII --- Aanvullingen: afhankelijkheid,
optimaliteit, en een schema met de hand berekend.}
\begin{enumerate}[resume]
  \item (\omterm{def:b3:topology:continuity}{Continue} afhankelijkheid) Stel dat $f$ op $R$
        $L$-lipschitz is in $x$, en zij $x, y$ twee oplossingen
        met beginwaarden $x_0$ en $y_0$ in $t_0$. Pas het bewijs
        van vraag 17 aan de ongelijkheid $e(t) \leq \norm{x_0 -
        y_0} + \int_{t_0}^tL\,e(s)\dd s$ aan en toon aan dat
        \[
        \norm{x(t) - y(t)} \leq \norm{x_0 - y_0}\,
        \eu^{L(t - t_0)},
        \]
        en ga op $x' = Lx$ na dat de grens bereikt wordt: Grönwall
        is scherp. Leid opnieuw de eenduidigheid af ($x_0 = y_0$),
        en dat de stromingsafbeelding $x_0 \mapsto x(t; x_0)$
        lipschitz is met constante $\eu^{LT}$, overal waar ze
        gedefinieerd is.
  \item (Osgood is optimaal) Zij omgekeerd $\omega$ \omterm{def:b3:topology:continuity}{continu},
        niet-dalend en positief op $\intoo0\infty$, met $\omega(r)
        \to 0$ als $r \to 0^+$ maar
        \[
        \int_0^1\frac{\dd r}{\omega(r)} < +\infty,
        \]
        en zet $f(x) = \omega(\abs x)$ voort met $f(0) = 0$. Toon
        aan dat $\Omega(x) = \int_0^x\frac{\dd r}{\omega(r)}$ een
        stijgende bijectie van $\intoc0{1}$ op $\intoc0{\Omega(1)}$
        is, dat haar inverse $g$ de vergelijking $g' = \omega(g)$
        met $g(0^+) = 0$ oplost, en dat $g$, met $0$ voortgezet
        voor $t \leq 0$, een $\mathcal C^1$-oplossing van $x' =
        f(x)$ door $(0, 0)$ is die van $x \equiv 0$ verschilt
        \emph{(voor $g'(0) = 0$: schat $\frac{g(t)}t$ af met
        $\omega(g(t))$)}. Besluit: de divergentiehypothese van
        vraag 15 is geen gemak maar de exacte grens van de
        eenduidigheid; vind Deel III terug uit $\omega(r) = 2\sqrt
        r$ en $\Omega(x) = \sqrt x$.
  \item (Euler met de hand berekend) Toon voor $x' = x$ met $x(0)
        = 1$ op $[0, 1]$ aan dat de eulerpolygoon voldoet aan
        $\varphi_n(1) = \bigl(1 + \frac1n\bigr)^n$. Bewijs de
        ontwikkeling
        \[
        \Bigl(1 + \frac1n\Bigr)^{n}
        = \eu\Bigl(1 - \frac1{2n} + O\Bigl(\frac1{n^2}\Bigr)
        \Bigr),
        \]
        zodat de fout in $t = 1$ gelijk is aan $\frac{\eu}{2n} +
        O(n^{-2})$: de snelheid $O(\frac1n)$ van vraag 18, met de
        exacte constante. Controleer numeriek voor $n = 10$:
        $1.1^{10} = 2.59374$ tegenover $\eu \approx 2.71828$, een
        fout $0.12454$, te vergelijken met $\frac{\eu}{20} \approx
        0.13591$.
\end{enumerate}
\end{problem}
