\chapter{الاحتمالات: الأسس وقانون الأعداد
الكبيرة}\label{ch:b3:probability}

بنت السنة الجامعية 2 الاحتمالات على الفضاءات القابلة للعدّ؛ وترفع
نظرية \omterm{def:b3:measure:measure}{القياس} الآن كل قيد. \omterm{def:b3:probability:space}{فالفضاء الاحتمالي} فضاءٌ قياسي كتلته
الكلية $1$، والمتغيّرات العشوائية تطبيقات قابلة \omterm{def:b3:measure:measure}{للقياس}، \omterm{def:b3:probability:space}{والأمل}
هو تكامل لوبيغ --- وفورًا تنطبق الترسانة التحليلية كلها
(\cref{ch:b3:measure,ch:b3:lebesgue,ch:b3:product}) على الصدفة. ويركّب هذا الفصل المعجم، ويبني متتاليات
لانهائية من المتغيّرات العشوائية \omterm{def:b3:probability:independence}{المستقلة} (على $\intcc01$،
انطلاقًا من الأرقام الثنائية: فالعشوائية مختبئة داخل \omterm{def:b3:measure:lebesgueouter}{قياس
لوبيغ})، ويبرهن على مبرهنتَي بوريل--كانتيلي المساعدتين وعلى قانون
كولموغوروف صفر--واحد، ويرتّب أنماط التقارب، ويبرهن على قانون
الأعداد الكبيرة --- أي المبرهنة التي تجعل التواترات تتقارب إلى
الاحتمالات وتجعل الإحصاء ممكنًا. وتعطي مسألة نهاية الأسبوع
برهان إتِمادي على القانون القوي في صيغته النهائية $L^1$.

\section{المعجم}

\begin{definition}\label{def:b3:probability:space}
\emph{الفضاء الاحتمالي}\index{فضاء احتمالي} فضاءٌ قياسي $(\Omega, \mathcal A, \P)$
مع $\P(\Omega) =
1$؛ وتسمّى عناصر $\mathcal A$ \emph{حوادث}، ويقال إن
خاصيةً تصح \emph{شبه أكيد} إذا كان احتمال حادثتها $1$.
و\emph{المتغيّر العشوائي}\index{متغيّر عشوائي} تطبيقٌ قابل
\omterm{def:b3:measure:measure}{للقياس} $X \colon \Omega \to \R$ (أو إلى $\R^d$: أي متجهة عشوائية)؛ و\emph{قانونه}
\index{قانون متغيّر عشوائي} هو \omterm{def:b3:measure:measure}{القياس} الاحتمالي المدفوع $\P_X =
X_*\P$
على $\R$ (\cref{exo:b3:product:9})، معيَّنًا بواسطة \emph{دالة التوزيع} $F_X(t) = \P(X \leq t)$
(\cref{exo:b3:measure:3}). ويقبل $X$ \emph{كثافة} $f$ إذا كان $\P_X = f\,\dd\lambda$؛
ويكون \emph{متقطّعًا} إذا كان $\P_X$ تركيبةً قابلة للعدّ من كتل
ديراك. و\emph{الأمل}\index{أمل} هو
\[
\E[X] = \int_\Omega X\,\dd\P
\qquad (X \geq 0 \text{ أو } X \in L^1(\P)),
\]
وتحسبه \emph{مبرهنة النقل} (\cref{exo:b3:product:9}) في القانون: $\E[g(X)] = \int_\R g\,\dd\P_X$ ---
أي $= \sum g(x_k)p_k$ في الحالة المتقطّعة، و $= \int
g(x)f(x)\dd x$ في حالة الكثافة: أي صيغ
السنة الجامعية 2، وقد صارت مبرهنات نظرية واحدة. و\emph{التباين}
هو $\V(X) =
\E[(X - \E X)^2] = \E[X^2] - (\E X)^2$ من أجل $X \in L^2$.
\end{definition}

\begin{example}\label{ex:b3:probability:laws}
القوانين المعيارية وتحويلاتها الجديرة بالذكر: برنولي
$\mathcal B(p)$، والثنائي $\mathcal B(n, p)$، والهندسي، وبواسون $\mathcal P(\lambda)$
(وهي متقطّعة: وتبقى جداول السنة الجامعية 2 صحيحة)؛ والمنتظم
على $\intcc01$ (أي \omterm{def:b3:measure:lebesgueouter}{قياس لوبيغ} نفسه)؛ والأُسّي $\mathcal E(\lambda)$ (بكثافة
$\lambda\eu^{-\lambda x}\mathbf 1_{x>0}$)؛ و\emph{القانون الغاوسي}\index{قانون غاوسي} $\mathcal N(m, \sigma^2)$
بكثافة $\frac1{\sigma\sqrt{2\pi}}\exp\bigl(-\frac{(x -
m)^2}{2\sigma^2}\bigr)$ --- وهي كثافة احتمالية حسب \cref{pb:b3:lebesgue:1}، بمتوسط $m$
وتباين $\sigma^2$ (العزوم الغاوسية، \cref{exo:b3:product:10}).
\end{example}

\begin{proposition}[ماركوف وتشيبيشيف]
\label{prop:b3:probability:markov}
من أجل $X \geq 0$ و $a > 0$: $\P(X \geq a) \leq \frac{\E
X}{a}$؛ ومن أجل $X \in L^2$:
$\P\bigl(\abs{X - \E X} \geq
a\bigr) \leq \frac{\V(X)}{a^2}$.\index{متراجحة ماركوف}\index{متراجحة تشيبيشيف}
\end{proposition}

\begin{proof}
\cref{exo:b3:lebesgue:5}(a)؛ وتشيبيشيف هي ماركوف مطبَّقة على $(X - \E X)^2$.
\end{proof}

\section{الاستقلال}

\begin{definition}\label{def:b3:probability:independence}
تكون الجبور الجزئية من النمط $\sigma$ $\mathcal A_1, \dots, \mathcal A_n
\subseteq \mathcal A$
\emph{مستقلة}\index{استقلال} إذا كان $\P(A_1\cap\dots\cap A_n) = \prod\P(A_i)$ من أجل كل $A_i \in
\mathcal A_i$؛
وتكون الحوادث مستقلة إذا كانت الجبور من النمط $\sigma$ $\{\varnothing, A_i, A_i^c, \Omega\}$
كذلك؛ وتكون المتغيّرات العشوائية $X_1, \dots, X_n$ مستقلة إذا كانت الجبور
من النمط $\sigma$ $\sigma(X_i) =
X_i^{-1}(\mathcal B(\R))$ كذلك. وتكون عائلة لانهائية مستقلة إذا
كانت كل عائلة جزئية منتهية منها مستقلة.
\end{definition}

\begin{theorem}\label{thm:b3:probability:independence}
تكون $X_1, \dots, X_n$ \omterm{def:b3:probability:independence}{مستقلة} إذا وفقط إذا كان قانون المتجهة $(X_1, \dots, X_n)$ هو
\omterm{thm:b3:product:existence}{القياس الجدائي} $\P_{X_1}\otimes\cdots\otimes\P_{X_n}$. وفي تلك الحالة، من أجل $g_i \geq 0$ (أو
بحيث تكون الجداءات \omterm{def:b3:lebesgue:l1}{قابلة للمكاملة}):
\[
\E\Bigl[\prod_ig_i(X_i)\Bigr] = \prod_i\E[g_i(X_i)],
\]
وعلى وجه الخصوص $\E[XY] = \E X\,\E Y$ و $\V(X_1 + \dots +
X_n) = \sum\V(X_i)$ من أجل متغيّرات \omterm{def:b3:probability:independence}{مستقلة} من
$L^2$.
\end{theorem}

\begin{proof}
إذا كانت $X_i$ \omterm{def:b3:probability:independence}{مستقلة}، توافق القياسان الاحتماليان $\P_{(X_1,\dots,X_n)}$
و $\bigotimes\P_{X_i}$ على جميع الجداءات $B_1\times\dots\times B_n$ للمجموعات البوريلية --- وهي
نظام من النمط $\pi$ يولّد $\mathcal B(\R^n)$ (\cref{prop:b3:product:sections}(b)) --- ومن ثَمّ
في كل مكان (\cref{thm:b3:measure:uniqueness}). وبالعكس، يحلّل قانونٌ جدائي جميع
الحوادث $\bigcap_iX_i^{-1}(B_i)$: أي \omterm{def:b3:probability:independence}{الاستقلال}. وصيغة \omterm{def:b3:probability:space}{الأمل} هي عندئذٍ
تونيلي/فوبيني (\cref{thm:b3:product:tonelli}) عبر مبرهنة النقل؛ و $\E[XY] = \E X\E Y$ هي الحالة
$g_i =
\mathrm{id}$، ويعطي نشر المربّع جمعيةَ التباينات (إذ الحدود المتقاطعة
$\E[(X_i - \E X_i)(X_j - \E X_j)]
= 0$).
\end{proof}

\begin{theorem}[وجود المتتاليات المستقلة]
\label{thm:b3:probability:existence}
على $\bigl(\intcc01, \mathcal L, \lambda\bigr)$ توجد متتالية $(U_n)_{n\geq1}$ من المتغيّرات
العشوائية \omterm{def:b3:probability:independence}{المستقلة}، كلٌّ منها منتظم على $\intcc01$. ومن ثَمّ،
من أجل أي قوانين مقرَّرة $(\mu_n)$ على $\R$ توجد متغيّرات
\omterm{def:b3:probability:independence}{مستقلة} $(X_n)$ تحقق $\P_{X_n} = \mu_n$.\index{المتتاليات المستقلة، الوجود}
\end{theorem}

\begin{proof}
\emph{الأرقام.} من أجل $\omega \in \intcc01$، لتكن $(b_k(\omega))$ أرقامَه
الثنائية ($\omega = \sum b_k2^{-k}$؛ ونختار النشر الذي لا ينتهي بأرقام $1$ كلها
--- فالالتباس لا يعني سوى مجموعة قابلة للعدّ، ومن ثَمّ معدومة).
وكل $b_k$ \omterm{def:b3:probability:space}{متغيّر عشوائي} (إذ $\{b_k = 1\}$ اتحادٌ منتهٍ لفترات
ثنائية)، وتأخذ المتجهة $(b_1, \dots, b_m)$ كل قيمة في $\{0,1\}^m$ على فترة
ثنائية طولها $2^{-m}$: ومنه تكون $b_k$ \omterm{def:b3:probability:independence}{مستقلة} برنولية $(\frac12)$.

\emph{إعادة التجميع.} نشطر $\N^*$ إلى عدد لانهائي من المجموعات
اللانهائية المنفصلة $(I_n)$ (مثلًا بقوى الأعداد الأولية، أو
بالأقطار)؛ ولتعدّد $(k^n_j)_j$ المجموعةَ $I_n$ ونضع
\[
U_n = \sum_{j\geq1} b_{k^n_j}\,2^{-j} .
\]
فيكون كل $U_n$ منتظمًا: إذ أرقامه الثنائية بتّاتٌ عادلة \omterm{def:b3:probability:independence}{مستقلة}،
ومنه $\P(U_n \in [l2^{-m}, (l+1)2^{-m})) = 2^{-m}$ من أجل كل فترة ثنائية، والفترات الثنائية تعيّن
القانون (\cref{thm:b3:measure:uniqueness}). و $U_n$ \omterm{def:b3:probability:independence}{مستقلة}: فهي دوال لكتل منفصلة من
العائلة \omterm{def:b3:probability:independence}{المستقلة} $(b_k)$ --- وشكليًّا، تتعلق الحوادث $\{U_n \in D_n\}$ من
أجل $D_n$ ثنائية بعدد منتهٍ من الأرقام من مجموعات منفصلة،
وتتحلّل؛ وترقّي حجةُ النظام من النمط $\pi$ ذلك إلى جميع
المجموعات البوريلية.

\emph{القوانين الكيفية.} لتكن $G_n(u) = \inf\{t : F_{\mu_n}(t)
\geq u\}$ (أي \emph{دالة الرباعي}
لدالة التوزيع $F_{\mu_n}$)؛ ويبيّن التكافؤ المفتاحي $G_n(u) \leq t
\iff u \leq F_{\mu_n}(t)$
(بالاتصال من اليمين للدالة $F$، وبالرتابة) أن $X_n = G_n(U_n)$
قابل \omterm{def:b3:measure:measure}{للقياس} مع $\P(X_n \leq t) = \P(U_n \leq F_{\mu_n}(t)) =
F_{\mu_n}(t)$: أي بالقانون $\mu_n$؛ \omterm{def:b3:probability:independence}{والاستقلال} موروث
(إذ هي دوال لمتغيّرات \omterm{def:b3:probability:independence}{مستقلة}، \cref{exo:b3:probability:3}).
\end{proof}

\begin{example}[مسألة أعياد الميلاد، بأمانة]
\label{ex:b3:probability:birthday}
بين $n$ من الأشخاص بأعياد ميلاد \omterm{def:b3:probability:independence}{مستقلة} منتظمة على $N = 365$
يومًا، يكون احتمال اختلاف جميع أعياد الميلاد
\[
p_n = \prod_{k=1}^{n-1}\Bigl(1 - \frac kN\Bigr),
\]
بالشرطنة المتتالية (أو مباشرةً: بقسمة الحالات المواتية $N(N-1)\cdots(N - n + 1)$
على المجموع $N^n$، وهي حجة عدّ تجعلها صيغةُ جداء \omterm{def:b3:probability:independence}{الاستقلال}
صارمة). وبأخذ اللوغاريتمات وباستعمال $-\ln(1 - x) = x +
O(x^2)$:
\[
\ln p_n = -\frac{n(n-1)}{2N} +
O\Bigl(\frac{n^3}{N^2}\Bigr),
\qquad\text{ومنه}\qquad
p_n \approx \eu^{-n^2/2N} .
\]
وتقع نقطة الانقلاب $p_n = \frac12$ عند $n \approx
\sqrt{2N\ln2} \approx 1.18\sqrt N$: فمن أجل
$N = 365$، $n = 23$ ($p_{23} = 0.4927$). وعبرتان. أولًا، تظهر التصادمات
بين $n$ من العناصر في $N$ من الصناديق عند السلّم
$n \sim \sqrt N$، لا عند $n \sim N$ --- وهو \emph{تحجيم أعياد
الميلاد} الذي يحكم تصادمات التجزئة وكلفةَ $\sqrt N$ لهجمات
أعياد الميلاد في التعمية. وثانيًا، الحساب قالبٌ: فحوادث تصادم
الأزواج البالغة $\binom n2$ ليست \omterm{def:b3:probability:independence}{مستقلة}، ومع ذلك يسلك الجواب
كأنها \omterm{def:b3:probability:independence}{مستقلة} (إذ $\eu^{-\binom n2/N}$ هو بالضبط الحدس القائم على \omterm{def:b3:probability:independence}{استقلال}
الأزواج) --- وهي أول حالة لتقريب بواسون، مصوغةً بصرامة في مسألة
نهاية الأسبوع في \cref{ch:b3:clt} (متراجحة لوكام).
\end{example}

\section{بوريل--كانتيلي وقانون صفر--واحد}

\begin{theorem}[بوريل--كانتيلي]
\label{thm:b3:probability:borelcantelli}
لتكن $(A_n)$ حوادث ولتكن $\limsup A_n = \bigcap_N
\bigcup_{n\geq N}A_n$ («تقع $A_n$ عددًا لانهائيًّا من
المرات»).\index{المبرهنتان المساعدتان لبوريل--كانتيلي}
\begin{enumerate}
  \item إذا كان $\sum\P(A_n) < \infty$، فإن $\P(\limsup A_n) =
        0$.
  \item وإذا كان $\sum\P(A_n) = \infty$ \emph{وكانت $A_n$ \omterm{def:b3:probability:independence}{مستقلة}}، فإن $\P(\limsup A_n) = 1$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) هي \cref{exo:b3:measure:4}. و(2): من أجل $N \leq M$، يعطي \omterm{def:b3:probability:independence}{استقلال}
المتممات (\cref{exo:b3:probability:3})
\[
\P\Bigl(\bigcap_{n=N}^{M}A_n^c\Bigr) = \prod_{n=N}^M\bigl(1
- \P(A_n)\bigr) \leq
\exp\Bigl(-\sum_{n=N}^M\P(A_n)\Bigr) \xrightarrow[M \to
\infty]{} 0
\]
($1 - x \leq \eu^{-x}$؛ إذ تتباعد المتسلسلة). ومنه $\P\bigl(\bigcup_{n\geq N}A_n\bigr) = 1$ من أجل كل $N$،
ويبقى للتقاطع المتناقص على $N$ الاحتمالُ $1$ (بالاتصال من
الأعلى، \cref{prop:b3:measure:basics}).
\end{proof}

\begin{theorem}[قانون كولموغوروف صفر--واحد]
\label{thm:b3:probability:zeroone}
لتكن $(X_n)$ \omterm{def:b3:probability:independence}{مستقلة} ولتكن $\mathcal T =
\bigcap_N\sigma(X_N, X_{N+1}, \dots)$ \emph{الجبر الذيلي من النمط
$\sigma$}\index{جبر ذيلي من النمط سيغما@الجبر الذيلي من النمط
$\sigma$} (أي الحوادث غير الحساسة لأي عدد منتهٍ من $X_n$:
تقارب $\sum X_n$، وتقارب $\frac{S_n}n$، وقيم النهايات العليا،
\dots). عندئذٍ يحقق كل $T \in \mathcal T$ أن $\P(T) \in \{0,
1\}$.\index{قانون
صفر--واحد}
\end{theorem}

\begin{proof}
نثبّت $N$. الجبران من النمط $\sigma$ $\sigma(X_1, \dots, X_N)$ و $\sigma(X_{N+1}, \dots)$ مستقلان:
إذ تتحلّل الحوادث المتعلقة بكتل منفصلة على الأنظمة المولِّدة من
النمط $\pi$ (الأسطوانات $\bigcap_{i\leq N}\{X_i \in B_i\}$، وعلى التوالي الشروط المنتهية
على المتغيّرات اللاحقة)، وتمدّد دينكين (\cref{thm:b3:measure:dynkin}، مطبَّقة
مرتين، جهةً في كل مرة) التحليلَ. وتقع حادثة ذيلية $T$ في $\sigma(X_{N+1}, \dots)$
من أجل كل $N$: فتكون $T$ \omterm{def:b3:probability:independence}{مستقلة} عن كل $\sigma(X_1, \dots, X_N)$، ومن ثَمّ عن
الجبر من النمط $\sigma$ الذي تولّده، أي $\sigma(X_1, X_2, \dots)$ (بدينكين مرة
أخرى: إذ اتحاد $\sigma(X_1,\dots,X_N)$ نظامٌ من النمط $\pi$ يولّده). لكن
$T \in \sigma(X_1, X_2, \dots)$ أيضًا: فتكون $T$ \omterm{def:b3:probability:independence}{مستقلة} \emph{عن نفسها}، أي $\P(T) = \P(T\cap T) =
\P(T)^2$:
ومنه $\P(T) \in \{0, 1\}$.
\end{proof}

\section{أنماط التقارب}

\begin{definition}\label{def:b3:probability:modes}
يكون $X_n \to X$ \emph{شبه أكيد} إذا كان $\P(X_n \to X) = 1$؛
و\emph{احتماليًّا} إذا كان $\P(\abs{X_n - X} \geq \varepsilon)
\to 0$ من أجل كل $\varepsilon > 0$؛ و\emph{في
$L^p$} إذا كان $\E\abs{X_n - X}^p \to 0$.\index{تقارب المتغيّرات العشوائية}
\end{definition}

\begin{proposition}\label{prop:b3:probability:modes}
(a) يستلزم التقارب شبه الأكيد التقاربَ الاحتمالي؛
(b) ويستلزم التقارب في $L^p$ التقاربَ الاحتمالي؛
(c) ويستلزم التقارب الاحتمالي التقاربَ شبه الأكيد \emph{على
امتداد متتالية جزئية}؛
(d) ولا يصح أي استلزام آخر عمومًا.
\end{proposition}

\begin{proof}
(a) $\P(\abs{X_n - X} \geq \varepsilon) \leq
\P\bigl(\sup_{m\geq n}\abs{X_m - X} \geq \varepsilon\bigr)
\downarrow \P\bigl(\limsup\{\abs{X_m - X} \geq
\varepsilon\}\bigr) = 0$ تحت التقارب شبه الأكيد (بالاتصال من الأعلى؛ إذ
تستبعد حادثة $\limsup$ التقاربَ). (b) بماركوف: $\P(\abs{X_n - X} \geq \varepsilon) \leq
\varepsilon^{-p}\,\E\abs{X_n - X}^p$. (c)
نختار $n_k$ تحقق $\P(\abs{X_{n_k} - X} \geq 2^{-k}) \leq 2^{-k}$؛ وتجعل بوريل--كانتيلي (1) $\abs{X_{n_k} - X} < 2^{-k}$ في
النهاية، شبه أكيد. (d) تتقارب الآلة الكاتبة (\cref{exo:b3:lp:3}) على
$(\intcc01, \lambda)$ في $L^1$ واحتماليًّا لكنها لا تتقارب نقطةً نقطة في أي
مكان؛ و $n\mathbf 1_{\intoo0{1/n}} \to 0$ شبه أكيد لا في $L^1$؛ والتفاصيل والأمثلة
المضادة الباقية في \cref{exo:b3:probability:6}.
\end{proof}

\section{قانون الأعداد الكبيرة}

في كل ما يلي، تكون $(X_n)$ \omterm{def:b3:probability:independence}{مستقلة} بالقانون نفسه (أي
\emph{\omterm{def:b3:probability:independence}{مستقلة} متماثلة التوزيع})، مع $S_n = X_1 + \dots + X_n$.

\begin{theorem}[قانون الأعداد الكبيرة الضعيف]
\label{thm:b3:probability:wlln}
إذا كان $X_1 \in L^2$، مع $m = \E X_1$:
\[
\P\Bigl(\Bigl|\frac{S_n}{n} - m\Bigr| \geq
\varepsilon\Bigr) \leq
\frac{\V(X_1)}{n\,\varepsilon^2}
\xrightarrow[n\to\infty]{} 0 :
\]
$\frac{S_n}n \to m$ احتماليًّا (وفي $L^2$).\index{قانون الأعداد الكبيرة
(الضعيف)}
\end{theorem}

\begin{proof}
$\E\frac{S_n}n = m$ و $\V\bigl(\frac{S_n}n\bigr) =
\frac{n\V(X_1)}{n^2}$ (\cref{thm:b3:probability:independence})؛ ثم تشيبيشيف.
\end{proof}

\begin{theorem}[قانون الأعداد الكبيرة القوي]
\label{thm:b3:probability:slln}
إذا كان $X_1 \in L^1$، فإن\index{قانون الأعداد الكبيرة (القوي)}
\[
\frac{S_n}{n} \xrightarrow[n\to\infty]{\text{شبه أكيد}} \E[X_1].
\]
ونبرهن عليه هنا تحت الفرضية الأقوى $X_1 \in
L^4$؛ وأما الحالة العامة
($L^1$: أي برهان إتِمادي) فهي مسألة نهاية الأسبوع.
\end{theorem}

\begin{proof}[البرهان تحت الفرضية $\E X_1^4 < \infty$]
بالتوسيط ($X_i \mapsto X_i - m$)، نفترض $m = 0$. وننشر:
\[
\E[S_n^4] = \sum_{i,j,k,l}\E[X_iX_jX_kX_l]
= n\,\E[X_1^4] + 3n(n-1)\,\bigl(\E[X_1^2]\bigr)^2 \leq
C\,n^2 ,
\]
إذ يلغي \omterm{def:b3:probability:independence}{الاستقلال} والتوسيط كل حدّ يحتوي عاملًا منعزلًا (إذ
$\E[X_iX_jX_kX_l] =
\E[X_i]\E[\cdots] = 0$ ما لم تتزاوج الأدلّة: فلا ينجو سوى الحدود $n$ التي
$i=j=k=l$ والحدود $3n(n-1)$ ذات الزوجين المتمايزين). وبماركوف:
\[
\P\Bigl(\Bigl|\frac{S_n}n\Bigr| \geq \varepsilon\Bigr)
= \P\bigl(S_n^4 \geq n^4\varepsilon^4\bigr)
\leq \frac{Cn^2}{n^4\varepsilon^4} =
\frac{C}{\varepsilon^4n^2},
\]
وهي قابلة للجمع: فتعطي بوريل--كانتيلي (1)، من أجل كل
$\varepsilon$ ناطق، أن $\abs{S_n/n} < \varepsilon$ في النهاية، شبه أكيد؛ وبالتقاطع
على $\varepsilon \in \Q_+^*$ (وهي عدد قابل للعدّ من الحوادث ذات الاحتمال $1$):
$S_n/n \to 0$ شبه أكيد.
\end{proof}

\begin{example}[ما يشتريه القانون القوي]
\label{ex:b3:probability:sllnapps}
(a) \emph{التواترات}: من أجل رميات قطعة نقدية \omterm{def:b3:probability:independence}{مستقلة} متماثلة
التوزيع، يتقارب التواتر المرصود للوجه شبه أكيد إلى $p$ --- أي
التبرير التجريبي للاحتمال نفسه.
(b) \emph{مونتي كارلو}\index{طريقة مونتي كارلو}: من أجل $g \in
L^1(\intcc01)$
ومن أجل $(U_n)$ \omterm{def:b3:probability:independence}{مستقلة} متماثلة منتظمة (\cref{thm:b3:probability:existence})، $\frac1n\sum_{k\leq n}g(U_k) \to \int_0^1g$
شبه أكيد: أي التكاملات بالمعاينة، في أي بُعد، بالمعدل المستقل
عن البُعد $\sim n^{-1/2}$ المضبوط في \cref{ch:b3:clt}.
(c) \emph{الأعداد الناظمية}: لكل عدد حقيقي تقريبًا، في نشره
الثنائي، تواترٌ مقارب $\frac12$ من الآحاد (بتطبيق القانون القوي
على متغيّرات الأرقام في \cref{thm:b3:probability:existence}) --- أي مبرهنة بوريل، وهي
عبارة عن أعداد \emph{كل} يوم مبرهَنٌ عليها \omterm{def:b3:measure:measure}{بالقياس}: ويكملها
\cref{pb:b3:probability:1} في جميع الأسس.
\end{example}

\begin{method}\label{met:b3:probability:toolkit}
الترتيب العملي للعبارات المقاربة عن المتتاليات العشوائية:
(1) \emph{هل الحادثة ذيلية؟} فاحتمالها عندئذٍ $0$ أو $1$
(\cref{thm:b3:probability:zeroone}) ولا يبقى إلا أن نقرّر أيهما. (2) \emph{ولتبرهن
على عبارات شبه أكيدة}: بوريل--كانتيلي --- باحتمالات قابلة للجمع
للحوادث «السيئة»، عبر حدود من نمط ماركوف أو تشيبيشيف على
العزوم المتاحة أيًّا كانت؛ ولا يلزم \omterm{def:b3:probability:independence}{الاستقلال} إلا في الاتجاه
العكسي. (3) \emph{المتتالية الجزئية مع الحصر}: برهن على التقارب
على امتداد متتالية جزئية قابلة للمعالجة، وتحكم في التذبذب
بينها بالرتابة أو بمتراجحات عظمى --- وهو هيكل برهان إتِمادي.
(4) ومن أجل النهايات التوزيعية، انتظر \cref{ch:b3:clt}.
\end{method}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:4}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:6}
على $(\intcc01, \lambda)$، أبرز --- مع البراهين --- متغيّرات عشوائية بحيث:

\begin{exercise}[$\star\star$]\label{exo:b3:probability:7}
يقدّر استطلاعُ رأي نسبةً مجهولة $p$ بالتواتر التجريبي
$\hat p_n$ لعدد $n$ من السحوب \omterm{def:b3:probability:independence}{المستقلة}.
(a) بتشيبيشيف: برهن على $\P(\abs{\hat p_n - p} \geq \varepsilon)
\leq \frac1{4n\varepsilon^2}$ (باستعمال $p(1-p) \leq \frac14$).
(b) وكم سحبًا يضمن خطأً $\leq 3\%$ باحتمال $\geq 95\%$ بهذا
الحدّ؟ (والجواب الحقيقي، عبر \cref{ch:b3:clt}، نحو $1070$: فتشيبيشيف
أمينة لكنها خشنة.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:probability:8}
(برنشتاين) من أجل $f \in \mathcal C(\intcc01)$ نعرّف كثير حدود برنشتاين $B_nf(x) =
\sum_{k=0}^n\binom nkx^k(1-x)^{n-k}f\bigl(\frac kn\bigr)$.
(a) تعرّف على $B_nf(x) = \E\bigl[f\bigl(\frac
{S_n}n\bigr)\bigr]$ من أجل $S_n$ ثنائي $\mathcal B(n, x)$.
(b) برهن على أن $B_nf \to f$ \emph{بانتظام} على $\intcc01$:
بالشطر على $\{\abs{\frac{S_n}n - x} \leq \delta\}$ ومتممتها، باستعمال الاتصال المنتظم
وتشيبيشيف مع الحدّ المنتظم $\V(\frac{S_n}n) \leq \frac1{4n}$.
(c) اخلص: برهانًا ثانيًا احتماليًّا على مبرهنة فايرشتراس في
التقريب (\cref{cor:b3:complete:weierstrass})، بالمعدل الصريح
$\norm{B_nf - f}_\infty \leq \frac32\,\omega_f(n^{-1/2})$
من أجل معامل الاتصال $\omega_f$ --- وبرهن على الأقل على الصيغة
$O(\omega_f(n^{-1/2}))$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:probability:9}
(جامع القسائم) تُسحب بطاقات من $n$ نوعًا بانتظام مع الإرجاع؛
وليكن $T_n$ عددَ السحوب حتى تُرى جميع الأنواع.
(a) اكتب $T_n = \sum_{k=1}^{n}\tau_k$ مع $\tau_k$ هندسيًّا بالوسيط $\frac{n - k + 1}n$،
و $\tau_k$ \omterm{def:b3:probability:independence}{مستقلة}، واستنتج $\E T_n = n\,H_n \sim n\ln n$ ($H_n$ العددُ التوافقي)
و $\V(T_n) \leq
\frac{\pi^2}6n^2$.
(b) بتشيبيشيف: $\frac{T_n}{n\ln n} \to 1$ احتماليًّا.
(c) وحدّد أكثر ببوريل--كانتيلي: برهن مباشرةً على $\P(T_n >
\beta n\ln n) \leq n^{1 - \beta}$ من أجل
$\beta > 1$ \emph{(بحدّ الاتحاد على حادثة تفويت نوع ما بعد
$\beta n\ln n$ سحبة، باستعمال $1 - x \leq \eu^{-x}$)}، واستنتج أنه على امتداد
$n = 2^m$، يكون شبه أكيد $T_n \leq \beta n\ln
n$ في النهاية، من أجل كل
$\beta > 2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:11}
(الأرقام القياسية) لتكن $(X_n)_{n\geq1}$ \omterm{def:b3:probability:independence}{مستقلة} متماثلة بدالة
توزيع متصلة، ولنقل إن \emph{رقمًا قياسيًّا} يقع عند الزمن $n$
إذا كان $X_n > \max(X_1, \dots, X_{n-1})$ (والزمن $1$ رقم قياسي). وليكن $R_n$ الدالةَ
المميّزة للرقم القياسي.
(a) برهن على $\P(R_n = 1) = \frac1n$ \emph{(بالتناظر، إذ يكون كلٌّ من ترتيبات
$X_1, \dots, X_n$ البالغة $n!$ متساوي الاحتمال والتعادلات ذات احتمال
$0$)}.
(b) برهن على أن $R_n$ \emph{\omterm{def:b3:probability:independence}{مستقلة}} \emph{(بعدّ الترتيبات
المتوافقة مع مواضع أرقام قياسية مقرَّرة، أو بالحجة القائلة إن
الترتيب النسبي للمتغيّرات $X_1, \dots, X_{n-1}$ مستقل عن رتبة $X_n$ بينها)}.
(c) استنتج من بوريل--كانتيلي (\cref{thm:b3:probability:borelcantelli}، بشقّيها) أن عددًا
لانهائيًّا من الأرقام القياسية يقع شبه أكيد، لكن الأرقام
القياسية عند زمنين متتاليين $n, n+1$ تقع عددًا لانهائيًّا من
المرات باحتمال --- قرّر أيهما! --- واحسب $\sum_n\P(R_n = 1, R_{n+1} = 1)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:probability:12}
(أطول سلسلة وجوه) نرمي قطعة نقدية عادلة عددًا لانهائيًّا من
المرات، وليكن $L_n$ طولَ أطول سلسلة وجوه متتالية ضمن الرميات
$n$ الأولى.
(a) برهن على أنه من أجل كل $\varepsilon > 0$، يكون شبه أكيد $L_n \leq
(1 + \varepsilon)\log_2n$ في
النهاية \emph{(إذ احتمال أن تبدأ سلسلة طولها $\ell$ ضمن
الرميات $n$ الأولى هو على الأكثر $n2^{-\ell}$؛ وبوريل--كانتيلي
على امتداد $n =
2^k$)}.
(b) برهن على أنه شبه أكيد $L_n \geq (1 - \varepsilon)\log_2n$ في النهاية \emph{(بتقطيع
الرميات $n$ الأولى إلى $\lfloor n/\ell\rfloor$ كتلة منفصلة طولها $\ell =
\lceil(1 - \varepsilon)\log_2n\rceil$؛
والكتل \omterm{def:b3:probability:independence}{مستقلة}، وكلٌّ منها كلها وجوه باحتمال $2^{-\ell}$،
واحتمال ألا تكون أي منها كلها وجوه هو على الأكثر $\exp(-n2^{-\ell}/\ell)$؛ ثم
اجمع على امتداد $n = 2^k$ من جديد)}.
(c) اخلص إلى $\frac{L_n}{\log_2n} \to 1$ شبه أكيد: أي إنه في مليون رمية عادلة يجب
توقّع سلسلة نحو $20$ وجهًا --- ومجموعةُ بيانات بلا واحدة منها
مصطنعةٌ على الأرجح.
\end{exercise}

\section{مسألة: برهان إتِمادي على القانون القوي}

\begin{problem}[{مسألة نهاية الأسبوع --- قانون الأعداد الكبيرة
القوي من أجل متغيّرات مستقلة متماثلة قابلة للمكاملة}]
\label{pb:b3:probability:1}
لم يكن لقانون كولموغوروف القوي --- $\frac{S_n}n \to \E X_1$ شبه أكيد من أجل
$X_n \in L^1$ \omterm{def:b3:probability:independence}{مستقلة} متماثلة --- طويلًا سوى براهين معقّدة؛
وفي 1981 وجد ن.~إتِمادي برهانًا لافت الاقتصاد، لا يستعمل شيئًا
وراء هذا الفصل (بل يضعّف \omterm{def:b3:probability:independence}{الاستقلال} إلى \omterm{def:b3:probability:independence}{استقلال} مثنى مثنى).
ونتتبّعه. لتكن $(X_n)$ \omterm{def:b3:probability:independence}{مستقلة} مثنى مثنى، متماثلة التوزيع،
\omterm{def:b3:lebesgue:l1}{قابلة للمكاملة}؛ و $m = \E X_1$، $S_n = X_1 + \dots + X_n$.

\medskip
\noindent\textbf{الجزء الأول --- الإرجاعات.}
\begin{enumerate}
  \item برهن على أنه يكفي معالجة $X_n \geq 0$ \emph{(بالشطر
        $X_n = X_n^+ - X_n^-$: وتحقق من أن الشقّين مستقلان مثنى مثنى
        ومتماثلان وقابلان للمكاملة من جديد)}. ولنفترض من الآن
        فصاعدًا أن $X_n \geq 0$.
  \item (البتر) لتكن $Y_n = X_n\,\mathbf 1_{X_n \leq n}$ ولتكن $S_n^* = Y_1 + \dots + Y_n$. برهن على
        \[
        \sum_{n\geq1}\P(X_n \neq Y_n) =
        \sum_{n\geq1}\P(X_1 > n) \leq \E[X_1] < \infty
        \]
        (\cref{exo:b3:product:3})، واستنتج ببوريل--كانتيلي أن $\frac{S_n - S_n^*}{n} \to 0$ شبه
        أكيد: فيكفي أن نبرهن على $\frac{S^*_n}n \to m$ شبه أكيد.
  \item برهن على $\E Y_n = \E\bigl[X_1\mathbf 1_{X_1\leq
        n}\bigr] \to m$ (بالتقارب الرتيب)، ومن ثَمّ على
        $\frac1n\sum_{k\leq n}\E Y_k \to m$ (بتشيزارو): فيكفي أن نبرهن على $\frac{S_n^* - \E S_n^*}{n} \to 0$ شبه
        أكيد.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- تقدير التباين.}
\begin{enumerate}[resume]
  \item برهن على
        \[
        \V(Y_n) \leq \E[Y_n^2] = \E\bigl[X_1^2\,\mathbf
        1_{X_1 \leq n}\bigr]
        \]
        وعلى الحدّ المفتاحي، باستعمال كعكة الطبقات
        (\cref{prop:b3:product:layercake})،
        \[
        \sum_{n\geq1}\frac{\V(Y_n)}{n^2}
        \leq \sum_{n\geq1}\frac1{n^2}\,
        \E\bigl[X_1^2\mathbf 1_{X_1\leq n}\bigr]
        \leq C\,\E[X_1] < \infty
        \]
        \emph{(بتبديل المجموع \omterm{def:b3:probability:space}{والأمل} --- بتونيلي للمتسلسلات
        --- وبحدّ $\sum_{n \geq
        x}\frac1{n^2} \leq \frac2{\max(x,1)}$ من أجل التقدير الداخلي $x^2\sum_{n\geq x}n^{-2} \leq 2x$)}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- التقارب على امتداد
المتتاليات الجزئية الهندسية.} نثبّت $\alpha > 1$ ولتكن $k_j =
\lfloor\alpha^j\rfloor$.
\begin{enumerate}[resume]
  \item باستعمال \omterm{def:b3:probability:independence}{الاستقلال} المثنى (إذ تُجمع التباينات،
        \cref{thm:b3:probability:independence} --- وتحقق من أن جمعية التباينات لا تحتاج
        إلا إلى \omterm{def:b3:probability:independence}{الاستقلال} المثنى) وتشيبيشيف، برهن من أجل كل
        $\varepsilon > 0$ على:
        \[
        \sum_{j\geq1}\P\Bigl(\Bigl|
        \frac{S^*_{k_j} - \E S^*_{k_j}}{k_j}\Bigr| \geq
        \varepsilon\Bigr)
        \leq
        \frac1{\varepsilon^2}\sum_{j\geq1}\frac1{k_j^2}
        \sum_{n\leq k_j}\V(Y_n)
        = \frac1{\varepsilon^2}\sum_{n\geq1}\V(Y_n)
        \sum_{j\,:\,k_j\geq n}\frac1{k_j^2} .
        \]
  \item برهن على $\sum_{j : k_j \geq n}k_j^{-2} \leq
        \frac{C_\alpha}{n^2}$ \emph{(بالمتسلسلة الهندسية؛ واحذر
        الجزء الصحيح: $k_j \geq \frac{\alpha^j}2$ من أجل عناية من نمط $\alpha^j \geq 2$)}،
        واخلص مع السؤال 4 وبوريل--كانتيلي:
        \[
        \frac{S^*_{k_j} - \E S^*_{k_j}}{k_j}
        \xrightarrow[j\to\infty]{\text{شبه أكيد}} 0,
        \qquad\text{ومنه}\qquad
        \frac{S^*_{k_j}}{k_j} \to m \ \text{شبه أكيد}
        \]
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الحصر والخاتمة.}
\begin{enumerate}[resume]
  \item من أجل $k_j \leq n \leq k_{j+1}$، استعمل رتابة $S^*_n$ (فالحدود غير
        سالبة!) لتبرهن على
        \[
        \frac{k_j}{k_{j+1}}\,\frac{S^*_{k_j}}{k_j}
        \;\leq\; \frac{S^*_n}{n} \;\leq\;
        \frac{k_{j+1}}{k_j}\,\frac{S^*_{k_{j+1}}}{k_{j+1}},
        \]
        واستنتج، شبه أكيد:
        \[
        \frac m\alpha \leq \liminf\frac{S^*_n}n \leq
        \limsup\frac{S^*_n}n \leq \alpha\,m .
        \]
  \item اجعل $\alpha \downarrow 1$ على امتداد متتالية واخلص إلى $\frac{S_n^*}n \to m$ شبه
        أكيد، ومن ثَمّ (بالجزء الأول) إلى \emph{قانون الأعداد
        الكبيرة القوي}:
        \[
        \boxed{\ \frac{S_n}{n}
        \xrightarrow[n\to\infty]{\text{شبه أكيد}} \E[X_1].\ }
        \]
  \item وأين بالضبط كفى \omterm{def:b3:probability:independence}{الاستقلال} المثنى (بدل \omterm{def:b3:probability:independence}{الاستقلال}
        الكامل)؟ اذكر المواضع الثلاثة التي استُدعيت فيها
        فرضيات من نمط \omterm{def:b3:probability:independence}{الاستقلال}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- العوائد.}
\begin{enumerate}[resume]
  \item (أعداد بوريل الناظمية) برهن على أن كل $x \in \intcc01$
        تقريبًا بالمعنى $\lambda$ \emph{ناظمي في كل أساس}
        $b \geq 2$: أي إن كل رقم $0, \dots, b-1$ يظهر بتواتر
        مقارب $\frac1b$ \emph{(ثبّت $b$ ورقمًا، وطبّق القانون
        القوي على متغيّرات الدوال المميّزة --- وبرّر أن أرقام
        الأساس $b$ لمتغيّر منتظم \omterm{def:b3:probability:independence}{مستقلة} متماثلة منتظمة على
        $\{0,\dots,b-1\}$ كما في \cref{thm:b3:probability:existence} --- ثم قاطع الحوادث القابلة للعدّ
        ذات الاحتمال واحد)}. وأبرز عددًا واحدًا صريحًا غير
        ناظمي، وتأمّل: أن المبرهنة تؤكد ناظمية جميع الأعداد
        تقريبًا، ومع ذلك يبقى برهان ناظمية $\sqrt2$ أو $\pi$
        مفتوحًا.
  \item (\omterm{ex:b3:probability:sllnapps}{مونتي كارلو}، مضمونًا) برّر كاملًا طريقةَ \cref{ex:b3:probability:sllnapps}(b)
        من أجل $g \in L^1(\intcc01^d)$: ابنِ العيّنة \omterm{def:b3:probability:independence}{المستقلة} المتماثلة
        المنتظمة على $\intcc01^d$ من \cref{thm:b3:probability:existence} و \cref{exo:b3:probability:10}،
        وصُغ ما يسلّمه القانون القوي.
\end{enumerate}

\medskip
\noindent\textbf{الجزء السادس --- ما يشتريه \omterm{def:b3:probability:independence}{الاستقلال} الكامل:
المتراجحات العظمى والمتسلسلات العشوائية.} ينفق إتِمادي
الاستقلالَ المثنى وحده؛ وتستثمر بقية الأجزاء الصيغة الكاملة
(المتبادلة). لتكن $(Z_n)$ متغيّرات \omterm{def:b3:probability:independence}{مستقلة} موسَّطة من $L^2$
ولتكن $S_k = Z_1 + \dots + Z_k$ (وهو اصطلاح جديد، لا علاقة له بالمتغيّرات $X_n$
أعلاه).
\begin{enumerate}[resume]
  \item (متراجحة كولموغوروف العظمى) من أجل $\varepsilon
        > 0$ برهن على
        \[
        \P\Bigl(\max_{1\leq k\leq n}\abs{S_k} \geq
        \varepsilon\Bigr) \;\leq\;
        \frac1{\varepsilon^2}\sum_{k=1}^n\V(Z_k) :
        \]
        فيشتري ثمنُ تشيبيشيف القيمةَ العظمى \emph{(بتجزئة
        الحادثة حسب أول دليل $k$ يحقق $\abs{S_k} \geq \varepsilon$؛ وعلى تلك القطعة
        اكتب $S_n^2 \geq S_k^2 + 2S_k(S_n - S_k)$ واستعمل \omterm{def:b3:probability:independence}{استقلال} الائتلافين $(Z_1, \dots, Z_k)$ و
        $(Z_{k+1}, \dots, Z_n)$، \cref{thm:b3:probability:independence})}. وبيّن الخطوة التي لم يعد فيها
        \omterm{def:b3:probability:independence}{الاستقلال} المثنى كافيًا.
  \item (مبرهنة خينتشين--كولموغوروف في المتسلسلة الواحدة)
        استنتج: أنه إذا كان $\sum_n\V(Z_n) < \infty$، تقاربت $\sum_nZ_n$ شبه
        أكيد \emph{(برهن على أن المجاميع الجزئية تشكّل شبه
        أكيد متتالية كوشية: اجعل $m \to
        \infty$ في المتراجحة العظمى
        مطبَّقةً على $Z_{N+1}, \dots, Z_{N+m}$، ثم اجعل $N \to
        \infty$)}.
  \item (متسلسلات رادماخر) لتكن $(\varepsilon_n)$ إشارات \omterm{def:b3:probability:independence}{مستقلة} متماثلة،
        $\P(\varepsilon_n = \pm1) =
        \frac12$ (\cref{thm:b3:probability:existence})، ولتكن $(x_n)$ أعدادًا حقيقية.
        برهن على أن $\sum_nx_n\varepsilon_n$ تتقارب شبه أكيد بمجرد أن يكون
        $\sum_nx_n^2 < \infty$؛ وبرهن أيضًا على أنه، أيًّا كانت $(x_n)$، يكون
        احتمال تقارب $\sum_nx_n\varepsilon_n$ إما $0$ وإما $1$ (\cref{thm:b3:probability:zeroone}).
  \item العكس، ابتدائيًّا. نضع $T_n = \sum_{k\leq
        n}x_k\varepsilon_k$ و $s_n^2 = \sum_{k\leq
        n}x_k^2$، ونفترض
        $s_n \to \infty$.
        (a) برهن على \emph{متراجحة بيلي--زيغموند}: أنه من أجل
        $Z \geq 0$ يحقق $\E Z^2 < \infty$ و $0 < \theta <
        1$،
        \[
        \P\bigl(Z > \theta\,\E Z\bigr) \;\geq\; (1 -
        \theta)^2\,\frac{(\E Z)^2}{\E Z^2}
        \]
        \emph{(اشطر $\E Z$ عند المستوى $\theta\E Z$ وطبّق
        كوشي--شوارتز على القطعة العليا)}.
        (b) برهن على $\E T_n^4 \leq 3s_n^4$.
        (c) استنتج $\P\bigl(\abs{T_n} > \frac{s_n}2\bigr)
        \geq \frac3{16}$ واخلص إلى أن $\sum_nx_n\varepsilon_n$ تتباعد شبه
        أكيد؛ ومن ثَمّ الثنائية
        \[
        \sum_nx_n\varepsilon_n\ \text{ يتقارب شبه أكيد}
        \iff \sum_nx_n^2 < \infty .
        \]
  \item (المتسلسلة التوافقية العشوائية) اخلص إلى أن $\sum_n\frac{\varepsilon_n}{n^s}$
        تتقارب شبه أكيد إذا وفقط إذا كان $s > \frac12$. ومن
        أجل $\frac12 < s \leq
        1$ تتقارب المتسلسلة شبه أكيد بينما $\sum_nn^{-s} = \infty$:
        أي إن الإشارات العشوائية تنتج تلاشيًا بقوة الجذر
        التربيعي --- وقارن ذلك بالمتسلسلة المتناوبة $\sum_n\frac{(-1)^n}{n^s}$،
        وهي تتقارب من أجل \emph{كل} $s > 0$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء السابع --- التركّز: متراجحة هوفدنغ.}
يقول القانون القوي إن $\frac{S_n}n \to m$؛ وتقول متراجحات التركّز كم يكون
انحرافٌ ما غير محتمل \emph{عند كل $n$ مثبَّت}.
\begin{enumerate}[resume]
  \item (المبرهنة المساعدة لهوفدنغ)
        (a) برهن على $\cosh\lambda \leq \eu^{\lambda^2/2}$ من أجل كل $\lambda \in \R$،
        بمقارنة المتسلسلتين حدًّا حدًّا.
        (b) ليكن $Z$ موسَّطًا مع $a \leq Z \leq b$، حيث $a < b$. برهن على
        \[
        \E\,\eu^{\lambda Z} \leq
        \exp\Bigl(\frac{\lambda^2(b - a)^2}8\Bigr)
        \]
        \emph{(حُدّ $\eu^{\lambda z}$ على $\intcc ab$ بوترها، وخذ
        الآمال، وادرس $\varphi(t)
        = -pt + \log(1 - p + p\eu^t)$ مع $p =
        \frac{-a}{b-a}$ و $t = \lambda(b - a)$: برهن على
        $\varphi(0) = \varphi'(0) = 0$ وعلى $\varphi''
        \leq \frac14$)}.
  \item (متراجحة هوفدنغ) لتكن $X_1, \dots, X_n$ \omterm{def:b3:probability:independence}{مستقلة} مع $a_i \leq X_i \leq b_i$ ولتكن
        $S_n =
        X_1 + \dots + X_n$. برهن، من أجل $t > 0$، على
        \[
        \P\bigl(S_n - \E S_n \geq t\bigr) \leq
        \exp\Bigl(\frac{-2t^2}{\sum_{i=1}^n(b_i -
        a_i)^2}\Bigr),
        \]
        وعلى الحدّ نفسه من أجل الذيل السفلي \emph{(بتشيبيشيف
        الأُسّية: حُدّ $\E\,\eu^{\lambda(S_n - \E S_n)}$ باستعمال \omterm{def:b3:probability:independence}{الاستقلال} والسؤال 17،
        ثم أمثِل على $\lambda > 0$)}.
  \item (القانون القوي، الحالة المحدودة، بمعدل) لتكن $X_i$
        \omterm{def:b3:probability:independence}{مستقلة} متماثلة بقيم في $\intcc ab$ ومع $m
        = \E X_1$. برهن
        على
        \[
        \P\Bigl(\Bigl|\frac{S_n}n - m\Bigr| \geq
        \varepsilon\Bigr) \leq
        2\exp\Bigl(\frac{-2n\varepsilon^2}{(b - a)^2}\Bigr)
        \]
        واستعد $\frac{S_n}n \to m$ شبه أكيد ببوريل--كانتيلي: أي برهانًا
        ثانيًا على القانون القوي من أجل المتغيّرات المحدودة ---
        دون بتر، بمعدل أُسّي عند كل $n$ منتهٍ، لكن بحدود محدودة
        وباستقلال كامل. وقارن الفرضيات بفرضيات إتِمادي.
  \item (\omterm{ex:b3:probability:sllnapps}{مونتي كارلو}، مضمونًا عند $n$ مثبَّت) لتكن $g
        \colon \intcc01^d \to \intcc01$
        قابلة \omterm{def:b3:measure:measure}{للقياس} ولتكن $(U_k)$ العيّنة \omterm{def:b3:probability:independence}{المستقلة} المتماثلة
        المنتظمة في السؤال 11. وبإعطاء $\varepsilon, \delta > 0$، برهن على
        \[
        n \geq \frac{\log(2/\delta)}{2\varepsilon^2}
        \implies
        \P\Bigl(\Bigl|\frac1n\sum_{k=1}^ng(U_k) -
        \int g\,\dd\lambda_d\Bigr| \geq \varepsilon\Bigr)
        \leq \delta,
        \]
        وقيّم العتبة من أجل $\varepsilon =
        \delta = 10^{-2}$. والحدّ لا يتضمّن $d$:
        فقارنه بالسؤال 11 وبالشبكات الحتمية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثامن --- كم يكبر مشي عشوائي؟ نحو
اللوغاريتم المكرَّر.} لتكن $S_n = \varepsilon_1 +
\dots + \varepsilon_n$ المشيَ العشوائي البسيط
المبني من إشارات عادلة \omterm{def:b3:probability:independence}{مستقلة} متماثلة.
\begin{enumerate}[resume]
  \item (ذيول تحت غاوسية) برهن على $\E\,\eu^{\lambda S_n} =
        (\cosh\lambda)^n \leq \eu^{n\lambda^2/2}$ واستنتج، من أجل
        $x > 0$،
        \[
        \P(S_n \geq x) \leq \eu^{-x^2/(2n)},
        \qquad
        \P(\abs{S_n} \geq x) \leq 2\,\eu^{-x^2/(2n)} .
        \]
  \item استنتج، ببوريل--كانتيلي،
        \[
        \limsup_{n\to\infty}\frac{\abs{S_n}}
        {\sqrt{2n\log n}} \leq 1 \quad\text{شبه أكيد}
        \]
        \emph{(من أجل $\eta > 0$، اجمع حدود الذيل عند $x =
        (1 + \eta)\sqrt{2n\log n}$،
        ثم قاطع على $\eta = \frac1p$)}. وعلى وجه الخصوص يعيش
        المشي على سلّم مبرهنة النهاية المركزية $\sqrt n$ إلى
        غاية عامل لوغاريتمي --- أي أدنى بكثير من الحدّ الخشن
        $\abs{S_n} \leq n$.
  \item وعلى امتداد المتتالية الجزئية المضاعِفة $n_j = 2^j$،
        برهن على
        \[
        \limsup_{j\to\infty}\frac{S_{n_j}}
        {\sqrt{2n_j\log\log n_j}} \leq 1 \quad\text{شبه أكيد},
        \]
        وتأمّل: أن \emph{قانون اللوغاريتم المكرَّر} (خينتشين؛
        وهارتمان--وينتنر من أجل حدود موسَّطة عامة من $L^2$)
        ينص على أن
        \[
        \limsup_{n\to\infty}\frac{S_n}
        {\sqrt{2n\log\log n}} = 1 \quad\text{شبه أكيد}
        \]
        واشرح بالضبط ما يفصل تقدير المتتالية الجزئية المبرهَن
        عليه للتوّ عن الشقّ الأعلى من هذه العبارة (إذ يجب
        التحكم في $\max_{n_j \leq n \leq
        n_{j+1}}S_n$ داخل كل كتلة، وهو ما يتطلب متراجحةً
        عظمى على السلّم \emph{الأُسّي}) وتحقق كمّيًّا من أن
        متراجحة السؤال 12 أضعف من أن تفي بذلك الغرض. ويقوم
        الشقّ الأدنى على المبرهنة المساعدة الثانية
        لبوريل--كانتيلي مطبَّقةً على كتل \omterm{def:b3:probability:independence}{مستقلة}؛ والشقّان معًا
        مادةٌ أمينة من مواد السنة الثالثة من أجل مقرر احتمالات
        مخصَّص.

  \item (الانحراف المنتظم على صنف منتهٍ) لتكن $A_1,
        \dots, A_N$ حوادث في
        تجربة قابلة للتكرار، ولنقدّر كل احتمال بتواتره
        التجريبي $\hat p_i$ على $n$ من التكرارات \omterm{def:b3:probability:independence}{المستقلة}
        المتماثلة. وبضمّ متراجحة هوفدنغ إلى حدّ الاتحاد، برهن
        على
        \[
        \P\Bigl(\max_{i\leq N}\,\abs{\hat p_i - \P(A_i)} >
        \varepsilon\Bigr) \;\leq\; 2N\,\eu^{-2n\varepsilon^2},
        \]
        واستنتج قاعدة حجم العيّنة: أن $n \geq
        \frac{\ln(2N/\delta)}{2\varepsilon^2}$ يضمن دقة
        $\varepsilon$ لجميع التقديرات $N$ في آن واحد باحتمال
        $\geq 1 -
        \delta$. واحسب $n$ من أجل $N = 10^6$ و $\varepsilon =
        0.01$ و
        $\delta = 0.05$: أي الثمن اللوغاريتمي للانتظام.
  \item (النافذة التوافقية العشوائية) بضمّ شقّي نظرية
        المتسلسلات العشوائية، برهن على أنه من أجل إشارات
        \omterm{def:b3:probability:independence}{مستقلة} متماثلة $(\varepsilon_n)$ تتقارب المتسلسلة $\sum_n\frac{\varepsilon_n}{n^\alpha}$ شبه
        أكيد إذا كان $\alpha > \frac12$ وتتباعد شبه أكيد إذا كان $\alpha \leq \frac12$؛
        وقابل ذلك بالتقارب المطلق (الذي يتطلب $\alpha > 1$):
        أي إنه على النافذة $\alpha \in \intoc{\frac12}1$، يكون التقارب ظاهرةً
        احتمالية فعلًا --- تلاشيًا لا حجمًا.
\end{enumerate}
\end{problem}
