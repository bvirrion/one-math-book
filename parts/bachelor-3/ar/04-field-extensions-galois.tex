\chapter{امتدادات الحقول ونظرية غالوا}\label{ch:b3:galois}

هل يمكن حل كل معادلة بالجذور، كما توحي الصيغة التربيعية وصيغ
كاردانو للمعادلات التكعيبية؟ وهل يمكن تثليث زاوية \omterm{cor:b3:galois:impossible}{بالمسطرة
والفرجار}؟ كلا السؤالين، اللذين ظلا مفتوحين قرونًا، يجدان جوابهما
--- بالنفي --- في فكرة واحدة لإيفاريست غالوا: أن نُلحق بكل كثير
حدود \emph{زمرةً} منتهية من تناظرات جذوره، ثم نقرأ الجواب في
الزمرة. ويبني هذا الفصل ذلك القاموس: امتدادات الحقول ودرجاتها،
وحقول الانشطار والغلوق الجبرية، والحقول المنتهية (نظرية كاملة ---
مع الدائرية الموعودة للزمرة $\mathbb F_q^\times$)، والقابلية للفصل، ثم \omterm{thm:b3:galois:fundamental}{تقابل
غالوا} نفسه، ببراهين تامة. وحصادنا: استحالة الإنشاءات الكلاسيكية،
وبنية الحقول الدائرية، واستحالة حل معادلة \omterm{def:b3:galois:extension}{الدرجة} الخامسة بالجذور
--- إذ تصيب بساطة $A_5$ الواردة في \cref{ch:b3:groups} هدفها.

\section{الامتدادات والدرجة والجبرية}

\begin{definition}\label{def:b3:galois:extension}
\emph{امتداد الحقول}\index{امتداد حقول} $L/K$ هو حقل $L$ يحتوي
$K$ بوصفه حقلًا جزئيًا؛ وعندئذٍ يكون $L$ فضاءً متجهيًا على $K$،
وتكون \emph{الدرجة}\index{درجة امتداد} $[L:K]$ هي بُعده. ويكون
الامتداد \emph{منتهيًا} إذا كان $[L:K] <
\infty$. و\emph{مميّز} حقلٍ هو
المولّد $\geq 0$ لنواة التطبيق $\Z \to K$ المعطى بالعلاقة $n \mapsto n\cdot 1$: وهو
$0$ أو عدد أولي $p$؛ وبالمقابل يحتوي $K$ على أصغر حقل جزئي
(\emph{الحقل الأولي}) مماثل للحقل $\Q$ أو للحقل $\mathbb
F_p = \Z/p\Z$.
\end{definition}

\begin{theorem}[مبرهنة البرج]\label{thm:b3:galois:tower}
إذا كان $K \subseteq L \subseteq M$، فإن $[M:K] = [M:L]\,[L:K]$: فإذا كان $(e_i)$ أساسًا للحقل $L$ على $K$
و $(f_j)$ أساسًا للحقل $M$ على $L$، فإن $(e_if_j)$ أساس للحقل $M$ على
$K$.\index{مبرهنة البرج}
\end{theorem}

\begin{proof}
التوليد: يُكتب $x \in M$ على الصورة $x = \sum_j \lambda_jf_j$ (حيث $\lambda_j \in L$)، ويُكتب
كل $\lambda_j = \sum_i \mu_{ij}e_i$ (حيث $\mu_{ij} \in K$): ومنه $x = \sum_{i,j}\mu_{ij}e_if_j$. الاستقلال: تُعاد
كتابة $\sum_{i,j}\mu_{ij}e_if_j = 0$ على الصورة $\sum_j (\sum_i
\mu_{ij}e_i)f_j = 0$؛ والمجاميع الداخلية في $L$، ومنه
فهي تنعدم (لأن $f_j$ مستقلة على $L$)؛ ثم تكون جميع المعاملات
$\mu_{ij} = 0$ (لأن $e_i$ مستقلة على $K$).
\end{proof}

\begin{definition}\label{def:b3:galois:algebraic}
ليكن $L/K$ وليكن $\alpha \in L$. إذا وُجد كثير حدود غير معدوم
$P \in K[X]$ يحقق $P(\alpha) = 0$، قلنا إن $\alpha$
\emph{جبري}\index{عنصر جبري} على $K$؛ ويكون المولّد الواحدي
$\pi_\alpha$ \omterm{def:b3:rings:ideal}{للمثالي} $\{P : P(\alpha) = 0\}$ من $K[X]$ هو \emph{كثير حدوده
الأدنى}\index{كثير الحدود الأدنى (نظرية الحقول)}، وهو كثير حدود
غير قابل للاختزال (إذ يفرض $\pi = QR$ مع $Q(\alpha) = 0$ أن يكون
$R$ ثابتًا، بأصغرية \omterm{def:b3:galois:extension}{الدرجة}). وإلا فإن $\alpha$ \emph{متسامٍ}.
ونكتب $K(\alpha)$ للدلالة على أصغر حقل جزئي من $L$ يحتوي $K$
و $\alpha$، و $K[\alpha]$ للدلالة على أصغر حلقة جزئية.
\end{definition}

\begin{theorem}\label{thm:b3:galois:simple}
إذا كان $\alpha$ جبريًا على $K$ مع $d = \deg\pi_\alpha$، فإن
\[
K(\alpha) = K[\alpha] \cong K[X]/(\pi_\alpha),
\qquad [K(\alpha) : K] = d,
\]
بالأساس $1, \alpha, \dots, \alpha^{d-1}$. وبالعكس، إذا كان $[L:K] < \infty$، فإن كل
$\alpha \in L$ \omterm{def:b3:galois:algebraic}{جبري} ودرجته $\deg \pi_\alpha$ تقسم $[L:K]$.
\end{theorem}

\begin{proof}
التقييم $K[X] \to L$ المعطى بالعلاقة $P \mapsto P(\alpha)$ صورته $K[\alpha]$ ونواته
$(\pi_\alpha)$؛ وبما أن $\pi_\alpha$ غير قابل للاختزال، فإن
$K[X]/(\pi_\alpha)$ \emph{حقل} (\cref{prop:b3:rings:primemaximal}: ففي \omterm{def:b3:rings:pidufd}{حلقة المثاليات الرئيسية}
$K[X]$، يولّد العنصرُ غير القابل للاختزال مثاليًا أعظميًا)، ومنه
فإن $K[\alpha]$ حقل يحتوي $K$ و $\alpha$: أي إنه يساوي
$K(\alpha)$. وتشكّل أصناف $1, X, \dots, X^{d-1}$ أساسًا \omterm{def:b3:rings:ideal}{لحلقة القسمة} (بالقسمة
الإقليدية)، ومن هنا يأتي الأساس \omterm{def:b3:galois:extension}{والدرجة}. وبالعكس، إذا كان $[L:K]
= n < \infty$:
فإن $1, \alpha, \dots, \alpha^n$ مرتبطة، فتعطي كثير حدود مُبيدًا؛ ثم يقسم $[K(\alpha):K] =
\deg\pi_\alpha$ العددَ
$n$ حسب مبرهنة البرج.
\end{proof}

\begin{corollary}\label{cor:b3:galois:algclosed}
إذا كان $\alpha, \beta$ جبريين على $K$، فكذلك $\alpha \pm
\beta$
و $\alpha\beta$ و $\alpha/\beta$ (حيث $\beta \ne 0$): أي إن عناصر
$L$ الجبرية على $K$ تشكّل حقلًا جزئيًا من $L$. وعلاوة على ذلك،
الجبرية متعدّية: \omterm{def:b3:galois:algebraic}{فالجبري} على \omterm{def:b3:galois:algebraic}{الجبري} جبريٌّ.
\end{corollary}

\begin{proof}
الامتداد $K(\alpha, \beta) = (K(\alpha))(\beta)$ منتهٍ على $K(\alpha)$ (لأن $\beta$ \omterm{def:b3:galois:algebraic}{جبري} على
$K \subseteq K(\alpha)$) والامتداد $K(\alpha)/K$ منتهٍ: ومنه حسب مبرهنة البرج
$[K(\alpha,\beta):K] <
\infty$، ويكون كل عنصر من $K(\alpha, \beta)$ --- ومنها العناصر الأربعة
المذكورة --- جبريًا (\cref{thm:b3:galois:simple}). أما التعدّي: فإذا كان $\beta$
جبريًا على $L$ وكان $L/K$ جبريًا، فإن معاملات $c_0, \dots, c_{m-1}$ لكثير
الحدود $\pi_{\beta/L}$ تولّد امتدادًا منتهيًا $F = K(c_0, \dots,
c_{m-1})$ للحقل $K$
(بتكرار مبرهنة البرج)، ويكون $F(\beta)/F$ منتهيًا: ومنه $[F(\beta):K] < \infty$،
فيكون $\beta$ جبريًا على $K$.
\end{proof}

\begin{example}\label{ex:b3:galois:degrees}
$[\Q(\sqrt2):\Q] = 2$، $[\Q(\sqrt[3]2):\Q] = 3$ (لأن $X^3 - 2$ غير قابل للاختزال: \omterm{thm:b3:rings:criteria}{أيزنشتاين})،
$[\Q(\zeta_p):\Q] = p - 1$ من أجل $\zeta_p = \eu^{2\iu\pi/p}$ (لأن $\Phi_p$ غير قابل للاختزال،
\cref{ex:b3:rings:eisenstein}). ومبرهنة البرج سلاحٌ فعّال أصلًا: $\sqrt[3]2 \notin \Q(\sqrt2)$، لأن
$3 \nmid 2$.
\end{example}

\section{حقول الانشطار؛ الغلق الجبري}

\begin{theorem}[حقول الانشطار]\label{thm:b3:galois:splitting}
ليكن $P \in K[X]$ غير ثابت. يوجد \emph{حقل انشطار}\index{حقل
انشطار} لكثير الحدود $P$ على $K$: أي امتداد $L =
K(\alpha_1, \dots, \alpha_n)$ مولَّد بجذور
$P$ ينشطر فيه $P$ إلى عوامل من \omterm{def:b3:galois:extension}{الدرجة} الأولى. وهو وحيد إلى حدود
التماثل على $K$، و $[L:K] \leq (\deg P)!$.
\end{theorem}

\begin{proof}
\emph{الوجود}، بالتراجع على $\deg P$: نختار عاملًا $Q$ غير قابل
للاختزال من $P$؛ فيحتوي الحقل $K_1 = K[X]/(Q)$ على الجذر $\alpha_1 = \bar X$
لكثير الحدود $Q$، ومنه للحدود $P$؛ ونكتب $P = (X -
\alpha_1)P_1$ على $K_1$ ونطبّق التراجع على
$P_1$ على $K_1$؛ وتتضارب الدرجات لتعطي $n(n-1)\cdots = n!$ على الأكثر.

\emph{الوحدانية} تنتج من العبارة الأقوى، وهي \emph{مبرهنة تمديد
التماثل المساعدة}: ليكن $\sigma \colon K \to K'$ تماثلًا، وليكن $P \in K[X]$،
وليكن $P^\sigma$ كثير الحدود ذا المعاملات المرسَلة، وليكن
$L, L'$ حقلَي انشطار للحدود $P, P^\sigma$؛ عندئذٍ يمتد $\sigma$
إلى تماثل $L \to L'$. بالتراجع على $[L:K]$: إذا انشطر $P$ في
$K$، فإن $L = K$ و $L' = K'$ (لأن $P^\sigma$ ينشطر في $K'$ ولأن
$L'$ مولَّد بجذوره). وإلا فلنختر جذرًا $\alpha \in L \setminus K$ لعامل $Q$ غير قابل
للاختزال من $P$ مع $\deg Q \geq 2$؛ فيكون $Q^\sigma$ عاملًا غير
قابل للاختزال من $P^\sigma$، وله جذر $\beta \in L'$؛ وعندئذٍ
\[
K(\alpha) \cong K[X]/(Q) \xrightarrow{\ \sigma\ }
K'[X]/(Q^\sigma) \cong K'(\beta)
\]
يمدّد $\sigma$ مع $\alpha \mapsto \beta$. والآن يكون $L$ \omterm{thm:b3:galois:splitting}{حقل انشطار} للحدود $P$ على
$K(\alpha)$، ويكون $L'$ \omterm{thm:b3:galois:splitting}{حقل انشطار} للحدود $P^\sigma$ على
$K'(\beta)$، مع $[L : K(\alpha)] < [L:K]$: فيمدّد التراجع أكثر إلى $L \to L'$.
\end{proof}

\begin{definition}\label{def:b3:galois:closure}
يكون الحقل $\Omega$ \emph{مغلقًا جبريًا}\index{حقل مغلق جبريا}
إذا كان لكل كثير حدود غير ثابت من $\Omega[X]$ جذرٌ في $\Omega$
(ومنه ينشطر). و\emph{الغلق الجبري}\index{غلق جبري} للحقل $K$ هو
امتداد \omterm{def:b3:galois:algebraic}{جبري} $\bar K/K$ يكون فيه $\bar K$ مغلقًا جبريًا.
\end{definition}

\begin{theorem}[شتاينيتز]\label{thm:b3:galois:steinitz}
لكل حقل $K$ \omterm{def:b3:galois:closure}{غلق جبري}، وحيدٌ إلى حدود التماثل على $K$.
\end{theorem}

\begin{proof}
\emph{الوجود (بناء أرتين).} لتكن $R = K[(X_f)_f]$ حلقة كثيرات
الحدود بمتغيّر $X_f$ واحد لكل كثير حدود واحدي غير ثابت
$f \in K[X]$، وليكن $I$ \omterm{def:b3:rings:ideal}{المثالي} المولَّد بجميع العناصر $f(X_f)$.
\omterm{def:b3:rings:ideal}{والمثالي} $I$ فعلي: إذ إن علاقة $1 = \sum_{i=1}^r g_i\,
f_i(X_{f_i})$ تتضمن عددًا منتهيًا من
كثيرات الحدود؛ وفي \omterm{thm:b3:galois:splitting}{حقل انشطار} مشترك $E$ للجداء $f_1\cdots f_r$
نختار جذورًا $\alpha_i$ للعناصر $f_i$ ونقيّم $X_{f_i} \mapsto \alpha_i$ (والمتغيّرات
الأخرى $\mapsto 0$): فنجد $1 = 0$، وهو سخف. وليكن $\mathfrak m \supseteq I$ أعظميًا
(\cref{thm:b3:rings:krull}؛ زورن) ولتكن $K_1 = R/\mathfrak
m$: وهو امتداد حقلي للحقل $K$ يكون فيه
لكل كثير حدود غير ثابت $f \in
K[X]$ جذرٌ، هو $\bar X_f$، ويكون جبريًا
على $K$ (لأنه مولَّد بالعناصر $\bar X_f$، وكلٌّ منها \omterm{def:b3:galois:algebraic}{جبري}).
ونكرّر: $K \subseteq K_1 \subseteq K_2 \subseteq \cdots$، حيث يفعل $K_{n+1}$ بالحقل $K_n$ ما فعله $K_1$ بالحقل $K$،
ولتكن $\Omega =
\bigcup_n K_n$، وهي حقل. وأي كثير حدود غير ثابت $g \in \Omega[X]$ له معاملاته
المنتهية العدد في $K_n$ ما؛ ولعامل غير قابل للاختزال من $g$ على
$K_n$ جذرٌ في $K_{n+1} \subseteq
\Omega$: ومنه فإن $\Omega$ مغلق جبريًا \omterm{def:b3:galois:algebraic}{وجبري} على
$K$ (لأن كل $K_n$ كذلك، بالتعدّي، \cref{cor:b3:galois:algclosed}): أي إن $\Omega$
\omterm{def:b3:galois:closure}{غلق جبري}.

\emph{الوحدانية.} ليكن $\Omega, \Omega'$ غلقين جبريين. لننظر في مجموعة
الأزواج $(E, \tau)$ حيث $K
\subseteq E \subseteq \Omega$ ويكون $\tau\colon E \to \Omega'$ غمرًا على $K$،
مرتّبةً بالتمديد؛ فهي غير خالية ($(K,
\mathrm{id})$) واستقرائية (لأن اتحاد
سلسلة كذلك)، ومنه يعطي زورن عنصرًا أعظميًا $(E_0, \tau_0)$. فإذا
كان $E_0 \neq \Omega$، فلنختر $\alpha \in
\Omega\setminus E_0$: عندئذٍ يُرسَل $\pi_{\alpha/E_0}$ إلى كثير حدود
على $\tau_0(E_0)$ له جذر $\beta$ في الحقل $\Omega'$ المغلق
جبريًا، ويمتد $\tau_0$ إلى $E_0(\alpha) \to
\Omega'$ ($\alpha \mapsto \beta$)، وهذا يناقض
الأعظمية. إذن يوجد غمر على $K$ هو $\tau \colon \Omega \to \Omega'$؛ وصورته، المماثلة
للحقل $\Omega$، مغلقة جبريًا، ويكون $\Omega'$ جبريًا عليها: فمن أجل
$x \in \Omega'$، ينشطر $\pi_{x/\tau(\Omega)}$ على $\tau(\Omega)$، ومنه $x \in
\tau(\Omega)$.
إذن $\tau$ شامل: أي إنه تماثل.
\end{proof}

\begin{remark}\label{rem:b3:galois:closureexamples}
من أجل $K = \Q$ يمكن تجنّب الآلة اللامنتهية: فالأعداد الجبرية
$\bar\Q = \{z \in \C : z \text{ جبري على } \Q\}$ تشكّل غلقًا جبريًا --- وهو حقل جزئي من $\C$ حسب
\cref{cor:b3:galois:algclosed}، ومغلق جبريًا لأن $\C$ كذلك (دالمبير--غاوس،
المُبرهَن عليها بالتحليل العقدي في \cref{ch:b3:holomorphic})، ولأن جذور كثيرات
الحدود على $\bar\Q$ جبرية على $\Q$ بالتعدّي.
\end{remark}

\section{الحقول المنتهية}

\begin{theorem}\label{thm:b3:galois:finitefields}
ليكن $p$ أوليًا وليكن $n \geq 1$ و $q = p^n$.
\begin{enumerate}
  \item عدد عناصر أي حقل منتهٍ قوةٌ لعدد أولي، ومن أجل كل $q$
        يوجد حقل واحد بالضبط $\mathbb F_q$ من $q$ عنصرًا إلى
        حدود التماثل: وهو \omterm{thm:b3:galois:splitting}{حقل الانشطار} لكثير الحدود $X^q - X$
        على $\mathbb F_p$.\index{حقل منته}
  \item \emph{تشاكل فروبينيوس}\index{تشاكل فروبينيوس الذاتي}
        $F
        \colon x \mapsto x^p$ تماثل ذاتي للحقل $\mathbb
        F_q$، وتكون زمرة التماثلات
        الذاتية للحقل $\mathbb F_q$ دائرية رتبتها $n$ ويولّدها $F$.
  \item ينغمر $\mathbb F_{p^m}$ في $\mathbb F_{p^n}$ إذا وفقط إذا كان $m \mid
        n$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) لكل حقل منتهٍ $E$ مميّزٌ $p > 0$ وهو فضاء متجهي منتهي البُعد
على $\mathbb F_p$: ومنه $\abs E = p^n$. ورتبة زمرته الضربية
$q - 1$، ومنه يحقق كل $x \in E$ العلاقة $x^q = x$: أي إن $E$
يتألف من $q$ جذرًا لكثير الحدود $X^q - X$، ومنه فهو \omterm{thm:b3:galois:splitting}{حقل انشطار}
له على $\mathbb F_p$ --- وهذا يحدّد $E$ إلى حدود التماثل
(\cref{thm:b3:galois:splitting}). وبالعكس، ففي \omterm{thm:b3:galois:splitting}{حقل انشطار} $L$ لكثير الحدود
$X^q - X$، تكون مجموعة جذوره $E$ \emph{حقلًا جزئيًا}: إذ إن
$(x + y)^q = x^q + y^q$ بتكرار حلم المبتدئ $(a+b)^p = a^p + b^p$ ($p \mid \binom pk$)، و $(xy)^q = x^qy^q$، و $(x^{-1})^q = (x^q)^{-1}$؛
وعدد عناصره $q$ بالضبط لأن $X^q - X$ قابل للفصل: فمشتقه $qX^{q-1} - 1 = -1$
(لأن $p \mid q$)، وهو أولي معه، ومنه لا جذور مكرّرة. إذن
$L = E$ وعدد عناصره $q$.

(2) التطبيق $F$ تشاكل حقول (بحلم المبتدئ)، ومتباين (لأنهما
حقلان)، ومنه فهو تقابلي على $\mathbb F_q$ المنتهي. ولدينا
$F^n =
\mathrm{id}$ (لأن $x^q = x$)، ولا تساوي أي قوة أصغر التطابقَ: إذ
يعني $F^m = \mathrm{id}$ أن جميع العناصر $q$ جذورٌ لكثير الحدود
$X^{p^m} - X$، وهذا يفرض $p^m \geq q$. إذن $\langle F\rangle$ دائرية رتبتها
$n$؛ ولا توجد تماثلات ذاتية أخرى، وذلك بالحد $\abs{\operatorname{Aut}} \leq [\,\mathbb F_q :
\mathbb F_p\,] = n$ المُبرهَن
عليه أدناه (\cref{prop:b3:galois:embeddings} مع $L = \mathbb F_q$ و $K =
\mathbb F_p$: فالتماثلات
الذاتية تُثبّت الحقل الأولي).

(3) إذا كان $\mathbb F_{p^m} \subseteq \mathbb F_{p^n}$، أعطت مبرهنة البرج أن $p^n = (p^m)^d$: أي
$m \mid n$. وبالعكس، إذا كان $m \mid n$، فإن $p^m - 1 \mid p^n - 1$ (بمجموع
هندسي)، ومنه يقسم $X^{p^m} - X$ كثيرَ الحدود $X^{p^n} - X$
(بالحجّة نفسها على الأُسس: $X^{a} - 1 \mid X^{b}
- 1$ حين $a \mid b$)، وتشكّل جذور
الأول داخل $\mathbb F_{p^n}$ الحقلَ الجزئي المطلوب، وعدد عناصره $p^m$
(بالقابلية للفصل كما في (1)).
\end{proof}

\begin{theorem}[الدائرية]\label{thm:b3:galois:cyclic}
كل زمرة جزئية منتهية من الزمرة الضربية لحقل هي زمرة دائرية. وعلى
وجه الخصوص $\mathbb F_q^\times \cong
\Z/(q-1)\Z$.\index{الزمرة الضربية لحقل منته}
\end{theorem}

\begin{proof}
لتكن $G \leq K^\times$ منتهية. حسب مبرهنة البنية (\cref{cor:b3:modules:abelian})، لدينا
$G \cong \Z/d_1 \times \dots
\times \Z/d_s$ مع $d_1 \mid \dots \mid d_s$. عندئذٍ يحقق كل $x \in G$ العلاقة
$x^{d_s} = 1$؛ لكن لكثير الحدود $X^{d_s} - 1$ عدد جذور لا يتجاوز
$d_s$ في الحقل $K$: ومنه $\abs G = d_1\cdots d_s \leq d_s$، وهذا يفرض $s = 1$: أي إن $G$
دائرية.
\end{proof}

\begin{example}\label{ex:b3:galois:f8}
$\mathbb F_8 = \mathbb F_2[X]/(X^3 + X + 1)$: إذ ليس لكثير الحدود التكعيبي جذر في $\mathbb F_2$، ومنه
فهو غير قابل للاختزال. وبكتابة $\omega =
\bar X$: تكون $\mathbb F_8^\times$ دائرية
رتبتها $7$، ومنه فإن \emph{كل} عنصر $\neq 0, 1$ يولّدها. وتشكّل
الحقول الجزئية من $\mathbb F_{p^{12}}$ شبكةَ قواسم العدد $12$: $\mathbb F_p, \mathbb F_{p^2}, \mathbb F_{p^3}, \mathbb F_{p^4},
\mathbb F_{p^6}, \mathbb F_{p^{12}}$ ---
وهذه أول حالة كاملة من \omterm{thm:b3:galois:fundamental}{تقابل غالوا}.
\end{example}

\section{القابلية للفصل والغمور}

\begin{definition}\label{def:b3:galois:separable}
يكون كثير الحدود $P \in K[X]$ \emph{قابلًا للفصل}\index{كثير حدود
قابل للفصل} إذا لم يكن له جذر مكرّر في \omterm{thm:b3:galois:splitting}{حقل انشطار} --- وهذا يكافئ
$\gcd(P, P') = 1$ (إذ إن الجذر المكرّر جذر مشترك؛ وبالعكس، فإن الجذر
المشترك مكرّرٌ على \omterm{thm:b3:galois:splitting}{حقل الانشطار}؛ ولا يتغيّر $\gcd$
بامتداد الحقل، بحجّة \cref{cor:b3:modules:descent}). ويكون \omterm{def:b3:galois:algebraic}{عنصر جبري} قابلًا للفصل
إذا كان \omterm{def:b3:galois:algebraic}{كثير حدوده الأدنى} كذلك؛ ويكون الامتداد $L/K$ قابلًا
للفصل إذا كانت جميع عناصره كذلك.
\end{definition}

\begin{proposition}\label{prop:b3:galois:perfect}
يكون كثير الحدود \emph{غير القابل للاختزال} $P \in K[X]$ \omterm{def:b3:galois:separable}{قابلًا
للفصل} إلا إذا كان $P' = 0$، وهذا يفرض $\operatorname{char} K = p > 0$ و $P \in K[X^p]$.
ومن نتائج ذلك أن كل امتداد \omterm{def:b3:galois:algebraic}{جبري} لحقل مميّزه $0$، ولحقل منتهٍ،
قابلٌ للفصل (وتُسمّى هذه الحقول \emph{كاملة}\index{حقل كامل}).
\end{proposition}

\begin{proof}
يقسم $\gcd(P, P')$ كثيرَ الحدود $P$؛ فإذا لم يكن $1$، فرضت عدمُ
قابلية الاختزال أن $\gcd = P$ (إلى حدود ثابت)، ومنه $P \mid P'$
مع $\deg
P' < \deg P$: أي $P' = 0$. وبكتابة $P = \sum a_kX^k$: نجد $ka_k = 0$ من أجل
كل $k$، ومنه يكون $P$ ثابتًا في المميّز $0$ (وهو مستبعَد)؛ وفي
المميّز $p$ يكون $a_k = 0$ إلا إذا كان $p \mid k$: أي $P =
Q(X^p)$.
وعلى حقل منتهٍ، يكون كل عنصر قوةً نونية من الرتبة $p$ (لأن
فروبينيوس شامل)، ومنه فإن $Q(X^p) = \sum b_k^p X^{pk} = (\sum
b_kX^k)^p$ ليس غير قابل للاختزال: أي إن
$P' = 0$ لا يمكن أن يحدث من أجل $P$ غير قابل للاختزال هناك
أيضًا.
\end{proof}

\begin{proposition}[إحصاء الغمور]\label{prop:b3:galois:embeddings}
ليكن $L = K(\alpha_1, \dots, \alpha_r)$ منتهيًا على $K$، وليكن $\sigma \colon K \to \Omega$ غمرًا في حقل مغلق
جبريًا. عندئذٍ لا يتجاوز عدد تمديدات $\sigma$ إلى $L$ العددَ
$[L:K]$، مع التساوي إذا كان $L/K$ \omterm{def:b3:galois:separable}{قابلًا للفصل}. وعلى وجه الخصوص
$\abs{\operatorname{Aut}_K(L)} \leq [L:K]$.
\end{proposition}

\begin{proof}
بالتراجع على $[L:K]$ عبر خطوات بسيطة. من أجل $L = K(\alpha)$:
يتحدّد التمديد $\tau$ بالقيمة $\tau(\alpha)$، التي يجب أن تكون
جذرًا في $\Omega$ لكثير الحدود $\pi_\alpha^\sigma$؛ وبالعكس، يعطي كل جذر من
هذا النوع تمديدًا واحدًا ($K(\alpha) \cong K[X]/(\pi_\alpha)$). وعدد التمديدات هو عدد الجذور
\emph{المتمايزة} لكثير الحدود $\pi_\alpha^\sigma$ في $\Omega$: وهو لا يتجاوز
$\deg\pi_\alpha =
[K(\alpha):K]$، مع التساوي إذا وفقط إذا كان $\pi_\alpha$ \omterm{def:b3:galois:separable}{قابلًا للفصل}
(وقابلية فصل $\pi^\sigma$ و $\pi$ متوافقتان: لأن \omterm{lem:b3:rings:bezout}{القاسم المشترك
الأكبر} مع المشتق محفوظ بالتطبيق $\sigma$). وفي العموم، نفكّك $L =
K(\alpha_1)(\alpha_2, \dots)$:
فيكون عدد تمديدات $\sigma$ إلى $K(\alpha_1)$ هو $\leq [K(\alpha_1):K]$، ويمتد
كلٌّ منها بعدد $\leq [L : K(\alpha_1)]$ من الطرق بالتراجع؛ ثم نضرب (بمبرهنة البرج). وفي
الحالة القابلة للفصل يكون العددان متساويين: لأن كثيرات الحدود
الدنيا على الحقل الأكبر $K(\alpha_1)$ تقسم مثيلاتها على $K$،
ومنه تبقى قابلة للفصل.
\end{proof}

\begin{theorem}[العنصر البدائي]\label{thm:b3:galois:primitive}
كل امتداد منتهٍ \emph{قابل للفصل} هو امتداد بسيط: أي $L =
K(\gamma)$ من
أجل $\gamma$ ما.\index{مبرهنة العنصر البدائي}
\end{theorem}

\begin{proof}
إذا كان $K$ منتهيًا، فكذلك $L$، ويفي مولّد $\gamma$ للزمرة
الدائرية $L^\times$ (\cref{thm:b3:galois:cyclic}) بالغرض. وليكن $K$ لامنتهيًا؛
يكفي بالتراجع أن نعالج $L =
K(\alpha, \beta)$. وليكن $n = [L:K]$؛ فحسب
\cref{prop:b3:galois:embeddings} يوجد $n$ من الغمور المتمايزة على $K$ $\sigma_1, \dots, \sigma_n \colon L \to \Omega$
(حيث $\Omega$ \omterm{def:b3:galois:closure}{غلق جبري}). وكثير الحدود
\[
D(T)=\prod_{i<j}\bigl[\bigl(\sigma_i(\alpha)-\sigma_j(\alpha)
\bigr) + T\bigl(\sigma_i(\beta) - \sigma_j(\beta)\bigr)\bigr]
\]
ليس معدومًا تطابقيًا: إذ لا ينعدم عامل تطابقيًا إلا إذا توافق
$\sigma_i, \sigma_j$ على كلٍّ من $\alpha$ و $\beta$، ومنه على $L$ --- وهذا
مستبعَد من أجل $i \neq j$. وبما أن $K$ لامنتهٍ، فلنختر $c \in
K$
يحقق $D(c) \ne 0$: عندئذٍ تكون العناصر $n$ $\sigma_i(\alpha +
c\beta)$ متمايزة مثنى
مثنى، ومنه فإن للعنصر $\gamma = \alpha + c\beta$ على الأقل $n$ من المقترنات المتمايزة
في $\Omega$، أي $\deg \pi_\gamma \geq n$: ويفرض $[K(\gamma):K] \geq n = [L:K]$ أن $L = K(\gamma)$.
\end{proof}

\section{تقابل غالوا}

\begin{definition}\label{def:b3:galois:galois}
يكون الامتداد المنتهي $L/K$ \emph{غالوايًّا}\index{امتداد غالوا}
إذا كان \omterm{thm:b3:galois:splitting}{حقل الانشطار} لكثير حدود \emph{قابل للفصل} على $K$.
و\emph{زمرة غالوا}\index{زمرة غالوا} له هي $\operatorname{Gal}(L/K) = \operatorname{Aut}_K(L)$، أي زمرة
التماثلات الذاتية للحقل $L$ التي تُثبّت $K$ نقطةً نقطة.
\end{definition}

\begin{proposition}\label{prop:b3:galois:galoisorder}
إذا كان $L/K$ \omterm{def:b3:galois:galois}{غالوايًّا}، فإن $\abs{\operatorname{Gal}(L/K)} =
[L:K]$؛ وعلاوة على ذلك يكون $L/F$
\omterm{def:b3:galois:galois}{غالوايًّا} من أجل كل حقل وسيط $K
\subseteq F \subseteq L$، ويكون لكل غمر على $F$
$L \to \Omega
\supseteq L$ الصورةُ $L$ (\emph{الناظمية}).
\end{proposition}

\begin{proof}
ليكن $L$ \omterm{thm:b3:galois:splitting}{حقل انشطار} لكثير الحدود $P$ القابل للفصل على $K$،
ولنثبّت غلقًا جبريًا $\Omega \supseteq L$. الامتداد $L/K$ قابل للفصل: فهو
مولَّد بجذور $P$؛ وتنتج قابلية فصل كل عنصر من حالة التساوي
أدناه، لكن لنحاجج مباشرةً --- تعطي \cref{prop:b3:galois:embeddings} مطبَّقةً على
المولِّدات (وهي جذور لكثير الحدود $P$ القابل للفصل، وكثيرات
حدودها الدنيا تقسم $P$) عددَ $[L:K]$ من تمديدات $K \hookrightarrow \Omega$ بالضبط
(ففي خطوة التراجع، يبقى كثير الحدود الأدنى لجذرٍ من جذور $P$ على
حقل وسيط قاسمًا للحدود $P$، ومنه فهو قابل للفصل). ويبدّل كل غمر من
هذا النوع $\tau \colon L \to \Omega$ جذورَ $P$ فيما بينها (لأن $\tau$ يُثبّت
المعاملات)، ويكون $L$ مولَّدًا بها: أي $\tau(L) = L$. ومنه فإن
الغمور $=$ التماثلات الذاتية: أي $\abs{\operatorname{Gal}(L/K)} = [L:K]$. وأما من أجل حقل وسيط
$F$: فإن $L$ \omterm{thm:b3:galois:splitting}{حقل انشطار} للحدود $P$ على $F$ أيضًا، ويبقى $P$ \omterm{def:b3:galois:separable}{قابلًا
للفصل}: ومنه $L/F$ غالوايّ؛ وتعطي الحجّة نفسها الناظمية على $F$.
\end{proof}

\begin{lemma}[أرتين]\label{lem:b3:galois:artin}
لتكن $G$ زمرة منتهية من التماثلات الذاتية لحقل $L$ وليكن $K = L^G = \{x : \sigma(x) = x\ \forall\sigma \in G\}$
حقلَها الصامد. عندئذٍ $[L : L^G] \leq \abs G$.
\end{lemma}

\begin{proof}
ليكن $n = \abs G$ و $G = \{\sigma_1, \dots, \sigma_n\}$، ولنفترض أن العناصر $x_1, \dots, x_{n+1} \in L$ مستقلة
خطيًا على $K$. للجملة الخطية المتجانسة المؤلَّفة من $n$ معادلة
بعدد $n+1$ من المجاهيل $(c_j)$ على $L$،
\[
\sum_{j=1}^{n+1} c_j\,\sigma_i(x_j) = 0
\qquad (i = 1, \dots, n),
\]
حلٌّ غير معدوم؛ ونختار حلًّا عدد عناصره غير المعدومة
\emph{أصغري}، وليكن $c_1, \dots, c_r \neq 0$ (بإعادة الترقيم)، مع $r \geq 2$
(لأن حلًّا وحيدًا $c_j\sigma_i(x_j) = 0$ مستحيل)، ومنظَّمًا بالشرط $c_r = 1$. ولا
تقع جميع العناصر $c_j$ في $K$: إذ إن المعادلة الخاصة بالمقدار $\sigma_i =
\mathrm{id}$
ستناقض الاستقلال؛ وليكن مثلًا $c_1 \notin K$، ومنه $\tau(c_1) \ne c_1$ من
أجل $\tau \in G$ ما. نطبّق $\tau$ على جميع المعادلات: وبما أن
$\tau\sigma_i$ يمسح $G$، فإن المتجهة $(\tau(c_j))_j$ حلٌّ آخر؛
وبالطرح، يكون $(c_j -
\tau(c_j))_j$ حلًّا بعدد أقل من العناصر غير المعدومة
(إذ ينعدم العنصر النوني من الرتبة $r$ لأن $1 - 1 = 0$، ولا
ينعدم الأول) وهو غير معدوم: وهو تناقض. إذن كل $n+1$ عنصرًا
مرتبطة: أي $[L:K]
\leq n$.
\end{proof}

\begin{theorem}[المبرهنة الأساسية في نظرية غالوا]
\label{thm:b3:galois:fundamental}
ليكن $L/K$ امتدادًا \omterm{def:b3:galois:galois}{غالوايًّا} زمرته $G =
\operatorname{Gal}(L/K)$.\index{تقابل غالوا}
\begin{enumerate}
  \item $L^G = K$.
  \item التطبيقان $H \mapsto L^H$ و $F \mapsto
        \operatorname{Gal}(L/F)$ تقابلان عكسيان
        أحدهما للآخر، وهما يعكسان الاحتواء، بين الزمر الجزئية من
        $G$ والحقول الوسيطة $K \subseteq F \subseteq L$؛ وعلاوة على ذلك $[L : L^H] = \abs H$
        و $[L^H : K] = [G : H]$.
  \item يكون $H \trianglelefteq G$ إذا وفقط إذا كان $L^H/K$ \omterm{def:b3:galois:galois}{غالوايًّا}، وعندئذٍ
        يُحدث القصر $\operatorname{Gal}(L^H/K) \cong
        G/H$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) من الواضح أن $K \subseteq L^G$. وبالعكس، ليكن $\alpha \in L
\setminus K$؛ سنُظهر
$\sigma \in G$ يحقق $\sigma(\alpha) \ne
\alpha$. درجة كثير الحدود الأدنى $\pi_\alpha$
على $K$ هي $\geq 2$ وهو قابل للفصل (لأن $L/K$ قابل للفصل،
\cref{prop:b3:galois:galoisorder})، ومنه فله جذر آخر $\beta \neq \alpha$ في \omterm{def:b3:galois:closure}{غلق جبري} $\Omega \supseteq L$.
نمدّد الغمر على $K$ $K(\alpha) \to \Omega$، $\alpha \mapsto
\beta$، إلى غمر $\tau\colon L \to \Omega$
(\cref{prop:b3:galois:embeddings})؛ وبالناظمية (\cref{prop:b3:galois:galoisorder}) يكون $\tau(L) = L$،
ومنه $\tau \in
G$، $\beta = \tau(\alpha) \in L$، و $\tau(\alpha) \neq
\alpha$.

(2) من أجل زمرة جزئية $H$: يكون $L/L^H$ \omterm{def:b3:galois:galois}{غالوايًّا} (\cref{prop:b3:galois:galoisorder})،
ويكون $\operatorname{Gal}(L/L^H) \supseteq H$ بداهةً، ومنه $[L:L^H] =
\abs{\operatorname{Gal}(L/L^H)} \geq \abs H$؛ وتعطي مبرهنة أرتين المساعدة
أن $[L:L^H] \leq \abs H$: أي التساوي، ومنه $\operatorname{Gal}(L/L^H) = H$. ومن أجل حقل وسيط $F$:
يعطي كون $L/F$ \omterm{def:b3:galois:galois}{غالوايًّا} أن $L^{\operatorname{Gal}(L/F)} = F$ حسب (1) مطبَّقةً على $L/F$.
فالتطبيقان عكسيان أحدهما للآخر؛ ومن الواضح أنهما يعكسان
الاحتواءات. وأما الدرجات: فقد بُرهن على $[L:L^H] = \abs H$ للتو، و $[L^H:K] = [L:K]/[L:L^H] = \abs G/\abs H$.

(3) من أجل $\sigma \in G$ و $H \leq G$: لدينا $\sigma(L^H) =
L^{\sigma H\sigma^{-1}}$ (بتحقّق
مباشر). وبالتقابل، يكون $\sigma(L^H) = L^H$ من أجل كل $\sigma$ إذا وفقط إذا
كان $H \trianglelefteq G$. والآن، إذا كان $H \trianglelefteq G$، فلنضع $F = L^H$: عندئذٍ
يُقصَر كل $\sigma \in
G$ إلى تماثل ذاتي للحقل $F$، فيعطي تشاكلًا $\rho
\colon G \to \operatorname{Aut}_K(F)$
نواته $\{\sigma :
\sigma\restriction_F = \mathrm{id}\} = \operatorname{Gal}(L/F) =
H$. ومنه ينغمر $G/H$ في $\operatorname{Aut}_K(F)$، وينتج $\abs{\operatorname{Aut}_K(F)} \geq [G:H] = [F:K]$؛
والمتراجحة العكسية صحيحة دائمًا (\cref{prop:b3:galois:embeddings}): ومنه $\abs{\operatorname{Aut}_K(F)} = [F:K]$
ويكون $\rho$ شاملًا. ويبقى أن نرى أن $F/K$ غالوايّ: فإن $F$
قابل للفصل على $K$ (لوقوعه داخل $L/K$ القابل للفصل)، و
$F = K(\gamma)$ (\cref{thm:b3:galois:primitive})؛ ولكثير الحدود $\prod_{\sigma \in G/H}\bigl(X - \sigma(\gamma)\bigr)$ (والجداء
على الصور المتمايزة، وهي تقع في $F$: لأن $\sigma(F) =
L^{\sigma H\sigma^{-1}} = L^H = F$ بناظمية $H$)
معاملاتٌ يُثبّتها $G$، ومنه فهي في $K$ حسب (1): أي إنه \omterm{def:b3:galois:separable}{كثير حدود
قابل للفصل} من $K[X]$ ينشطر في $F$، وجذوره تولّد $F$: ومنه فإن
$F/K$ غالوايّ. وبالعكس، إذا كان $F = L^H$ مع $F/K$ غالوايّ،
أعطت ناظمية $F$ (\cref{prop:b3:galois:galoisorder}، مطبَّقةً على الغمور
$F \to \Omega$ المقصورة من عناصر $G$) أن $\sigma(F) = F$ من أجل
كل $\sigma \in G$، أي $H
\trianglelefteq G$.
\end{proof}

\section{الامتدادات الدائرية}

\begin{definition}\label{def:b3:galois:cyclotomic}
ليكن $n \geq 1$ وليكن $\zeta_n = \eu^{2\iu\pi/n}$. كثير الحدود
\emph{الدائري}\index{كثير حدود دائري} النوني هو $\Phi_n = \prod_{\gcd(k,n)=1,\ 1 \le k \le n} \bigl(X -
\zeta_n^k\bigr)$، ودرجته
$\varphi(n)$؛ وبتجميع جذور $X^n - 1$ حسب الرتبة المضبوطة نجد
$X^n - 1 = \prod_{d \mid n}\Phi_d$، وهذا يبيّن بالتراجع أن $\Phi_n \in \Z[X]$ (بالقسمة الإقليدية
لكثيرات الحدود الصحيحة الواحدية).
\end{definition}

\begin{theorem}\label{thm:b3:galois:cyclotomicirred}
كثير الحدود $\Phi_n$ غير قابل للاختزال على $\Q$؛ ومنه $[\Q(\zeta_n) : \Q] =
\varphi(n)$ و
\[
\operatorname{Gal}\bigl(\Q(\zeta_n)/\Q\bigr) \;\cong\;
(\Z/n\Z)^\times,
\qquad \sigma_a(\zeta_n) = \zeta_n^a .
\]
ومن ثَمّ فإن الامتداد $\Q(\zeta_n)/\Q$ غالوايّ وزمرته
\emph{أبيلية}.
\end{theorem}

\begin{proof}
ليكن $f = \pi_{\zeta_n}$، ومنه $\Phi_n = fg$ حيث $f, g \in \Z[X]$ واحديان
(بمبرهنة غاوس المساعدة \cref{lem:b3:rings:gauss}: إذ تتضارب المحتويات، وجميع
كثيرات الحدود واحدية). \emph{الادعاء: إذا كان $\zeta$ جذرًا
للحدود $f$ وكان $p \nmid n$ أوليًا، فإن $\zeta^p$ جذر للحدود $f$.} وإلا
كان $\zeta^p$ جذرًا للحدود $g$ (لأنه جذر بدائي للوحدة من الرتبة $n$)،
ومنه يكون $\zeta$ جذرًا للمقدار $g(X^p)$، ويكون $f
\mid g(X^p)$ في $\Z[X]$
(بكثير الحدود الأدنى، ثم غاوس مرة أخرى). نختزل بترديد $p$:
$\bar g(X^p) = \bar g(X)^p$ (بفروبينيوس على $\mathbb F_p[X]$: أي $a^p = a$ معاملًا
معاملًا، وحلم المبتدئ)، ومنه $\bar f \mid \bar g^{\,p}$: أي إن $\bar f$ و $\bar g$
يشتركان في عامل غير قابل للاختزال، ويكون للمقدار $\bar\Phi_n = \bar f\bar g$ عامل مكرّر.
وعندئذٍ يكون للمقدار $X^n - \bar 1$ عامل مكرّر أيضًا؛ لكن مشتقه
$\bar nX^{n-1}$ أولي معه (لأن $p \nmid n$ ولأن $0$ ليس جذرًا):
وهو تناقض.

ويُحصَل على كل جذر بدائي $\zeta_n^k$ (حيث $\gcd(k, n) = 1$) من
$\zeta_n$ بقوى أولية متتالية لا تقسم $n$ (بتفكيك $k$): فينتشر
الادعاء، ويكون كل جذر بدائي جذرًا للحدود $f$: أي $f = \Phi_n$، وهو
غير قابل للاختزال. ومن ثَمّ $[\Q(\zeta_n):\Q] = \varphi(n)$، ويكون $\Q(\zeta_n)$ حقلَ
الانشطار \omterm{def:b3:galois:separable}{لكثير الحدود القابل للفصل} $X^n - 1$ (لأن جميع الجذور
قوى للعنصر $\zeta_n$): أي إنه غالوايّ. ويرسل التماثل الذاتي $\sigma$
العنصرَ $\zeta_n$ إلى جذر بدائي آخر $\zeta_n^{a(\sigma)}$، ويكون $\sigma
\mapsto a(\sigma)$
تشاكلًا متباينًا إلى $(\Z/n\Z)^\times$؛ ورتبة الزمرتين $\varphi(n)$: أي
إنه تماثل.
\end{proof}

\section{المسطرة والفرجار}

\begin{definition}\label{def:b3:galois:constructible}
نطابق المستوي مع $\C$؛ وننطلق من $\{0, 1\}$. نقول عن نقطة إنها
\emph{قابلة للإنشاء}\index{عدد قابل للإنشاء} إذا أمكن الحصول
عليها بعدد منتهٍ من تقاطعات المستقيمات المارّة بنقطتين مُنشأتين
والدوائر التي مركزها نقطة مُنشأة ونصف قطرها مسافة بين نقطتين
مُنشأتين.
\end{definition}

\begin{theorem}[فانتزل]\label{thm:b3:galois:wantzel}
يكون $z \in \C$ قابلًا للإنشاء إذا وفقط إذا وُجد برج $\Q = F_0
\subseteq F_1 \subseteq \dots \subseteq F_r$ مع
$[F_{i+1} :
F_i] = 2$ و $z \in F_r$. وعلى وجه الخصوص، يكون \omterm{def:b3:galois:constructible}{العدد القابل
للإنشاء} جبريًا ودرجته قوةٌ للعدد $2$ على $\Q$.\index{مبرهنة
فانتزل}
\end{theorem}

\begin{proof}
($\Rightarrow$) إحداثيات تقاطع مستقيمين مارّين بنقاط إحداثياتها
في حقل جزئي $F \subseteq \R$ تحل جملة خطية على $F$: فهي تبقى في
$F$. أما تقاطع مستقيم مع دائرة أو دائرة مع دائرة فيؤدي، بعد حذف
الجزء الخطي (إذ يعطي طرح معادلتَي الدائرتين مستقيمًا)، إلى
معادلة من \omterm{def:b3:galois:extension}{الدرجة} الثانية على $F$: فتقع الإحداثيات الجديدة في $F$
أو في $F(\sqrt d)$ من أجل $d \in F$ ما مع $d > 0$. وبالتراجع،
تقع إحداثيات كل نقطة مُنشأة في برج من الامتدادات التربيعية
للحقل $\Q$؛ ويقع $z = x + \iu y$ في برج تربيعي أيضًا (بضم $\iu$:
أي خطوة تربيعية إضافية). ونتيجة \omterm{def:b3:galois:extension}{الدرجة}: يقسم $[\Q(z):\Q]$
العددَ $[F_r : \Q] = 2^r$ (بمبرهنة البرج).

($\Leftarrow$) تشكّل الأعداد \omterm{def:b3:galois:constructible}{القابلة للإنشاء} حقلًا: فالمجاميع
والفروق بمتوازيات الأضلاع (والمتوازيات \omterm{def:b3:galois:constructible}{قابلة للإنشاء}: بإنزال
العمودين ورفعهما مرتين --- فالعمود الكلاسيكي المارّ بنقطة يستعمل
دائرة واحدة وقوسين)؛ والجداءات والخوارج بأشكال طاليس التناسبية
(فبإعطاء طولين $a, b$ نُنشئ $ab$ و $a/b$ بمثلثات متشابهة على
شعاعين). والحقل مغلق تحت الجذور التربيعية: فمن أجل $a > 0$،
تتقاطع الدائرة ذات القطر $1 + a$ والعمود عند نقطة الوصل على
الارتفاع $\sqrt a$ (بعلاقة الارتفاع والوسط الهندسي في مثلث
قائم)؛ ومن أجل عدد عقدي $w = \rho\eu^{\iu\theta}$، نُنشئ $\sqrt\rho$ ونُنصّف
$\theta$ (وتنصيف الزاوية إنشاءٌ بالفرجار). ومنه فإن الجزأين
الحقيقي والتخيلي لعناصر برج تربيعي قابلان للإنشاء بالتراجع على
البرج: إذ تضم كل خطوة جذورَ معادلة من \omterm{def:b3:galois:extension}{الدرجة} الثانية، وهي
معبَّر عنها بعمليات الحقل وبجذر تربيعي واحد لعدد مُنشأ أصلًا
(بالصيغة التربيعية؛ في المميّز $0$).
\end{proof}

\begin{corollary}\label{cor:b3:galois:impossible}
المسائل الكلاسيكية الثلاث غير \omterm{def:b3:groups:derived}{قابلة للحل} \omterm{cor:b3:galois:impossible}{بالمسطرة
والفرجار}:\index{المسطرة والفرجار}
\begin{enumerate}
  \item \emph{تضعيف المكعب}: درجة $\sqrt[3]2$ هي $3$، وليست قوة
        للعدد $2$.
  \item \emph{تثليث الزاوية}: يتطلب تثليث $60^\circ$ العددَ
        $\cos 20^\circ$، وهو جذر لكثير الحدود غير القابل
        للاختزال $8X^3
        - 6X - 1$: أي من \omterm{def:b3:galois:extension}{الدرجة} $3$.
  \item \emph{تربيع الدائرة}: العدد $\sqrt\pi$ متسامٍ (لأن $\pi$
        كذلك --- مبرهنة لينديمان، نقبلها هنا: فبرهانها يخص مقرّرًا
        في نظرية التسامي).
\end{enumerate}
كذلك، يكون المضلع المنتظم ذو $n$ ضلعًا قابلًا للإنشاء إذا وفقط
إذا كان $\varphi(n)$ قوةً للعدد $2$ (غاوس--فانتزل؛ ويستعمل شقّ
«إذا» طريقةَ مسألة نهاية الأسبوع، وشقّ «فقط إذا» هو \cref{thm:b3:galois:wantzel}
مطبَّقةً على $\zeta_n$ ذي \omterm{def:b3:galois:extension}{الدرجة} $\varphi(n)$). ومن أجل $n = 7$:
$\varphi(7) = 6$: فالمضلع المنتظم ذو سبعة أضلاع مستحيل؛ ومن أجل
$n = 17$: $\varphi(17) = 16 = 2^4$: فهو قابل للإنشاء --- وتُنشئه مسألة نهاية
الأسبوع.
\end{corollary}

\begin{proof}
(1) كثير الحدود $X^3 - 2$ غير قابل للاختزال (\omterm{thm:b3:rings:criteria}{أيزنشتاين}). (2) من
$\cos
3\theta = 4\cos^3\theta - 3\cos\theta$ مع $3\theta =
60^\circ$ نجد $8c^3 - 6c = 1$ من أجل $c = \cos 20^\circ$؛ وليس
لكثير الحدود التكعيبي $8X^3 - 6X - 1$ جذر ناطق (فالمرشّحون
$\pm1, \pm\frac
12, \pm\frac14, \pm\frac18$ يخفقون)، ومنه فهو غير قابل للاختزال: أي من \omterm{def:b3:galois:extension}{الدرجة} $3$.
والزاوية $60^\circ$ العامة \omterm{def:b3:galois:constructible}{قابلة للإنشاء}، ومنه فإن مثلِّثًا لها
سيُنشئ $c$. (3) لو كان $\sqrt\pi$ قابلًا للإنشاء لكان جبريًا،
ومنه $\pi$ أيضًا. وأما عبارة المضلع: فدرجة $\zeta_n$ هي
$\varphi(n)$ (\cref{thm:b3:galois:cyclotomicirred})؛ وينتج اللزوم من فانتزل؛ وأما
الكفاية، فإن \omterm{def:b3:galois:galois}{زمرة غالوا}، وهي أبيلية رتبتها $2^m$، تقبل سلسلة من
الزمر الجزئية ذات الدليل $2$ (فكل زمرة $2$ منتهية تقبل ذلك:
\cref{exo:b3:groups:10})، وتشكّل حقولها الصامدة برجًا تربيعيًا ينتهي عند
$\Q(\zeta_n)$ (\cref{thm:b3:galois:fundamental})؛ ونخلص من \cref{thm:b3:galois:wantzel}.
\end{proof}

\section{القابلية للحل بالجذور}

\begin{definition}\label{def:b3:galois:radical}
يكون الامتداد $L/K$ (والمميّز $0$ في كل هذا القسم)
\emph{جذريًّا} إذا وُجد برج $K = F_0 \subseteq \dots
\subseteq F_r = L$ مع $F_{i+1} = F_i(\alpha_i)$، $\alpha_i^{n_i} \in F_i$: أي إن
كل خطوة تضم جذرًا نونيًا من الرتبة $n_i$. ويكون كثير الحدود
$P \in K[X]$ \emph{قابلًا للحل بالجذور}\index{قابل للحل
بالجذور} إذا كان حقل انشطاره محتوًى في امتداد جذريّ ما للحقل $K$.
\end{definition}

\begin{lemma}\label{lem:b3:galois:cyclicsteps}
ليكن $K$ حاويًا على جذر بدائي للوحدة من الرتبة $n$ هو $\zeta$،
أي $\zeta$ رتبته $n$ في $K^\times$، وليكن $a \in K^\times$.
عندئذٍ يكون $K(\sqrt[n]a)/K$ \omterm{def:b3:galois:galois}{غالوايًّا} وزمرته \emph{دائرية}.
وبالعكس --- وهو ما لا نحتاجه أدناه --- كل امتداد دائري من
\omterm{def:b3:galois:extension}{الدرجة} $n$ من هذا الشكل. وعلاوة على ذلك، يكون $K(\zeta_n)/K$
\omterm{def:b3:galois:galois}{غالوايًّا} وزمرته \emph{أبيلية}، من أجل أي $K$ مميّزه $0$.
\end{lemma}

\begin{proof}
كثير الحدود $X^n - a$ قابل للفصل (لأن قاسمه المشترك الأكبر مع
$nX^{n-1}$ يساوي $1$: إذ $a \neq 0$) وينشطر في $K(\alpha)$ حيث
$\alpha^n = a$: فجذوره هي $\zeta^k\alpha \in K(\alpha)$. ومنه فإن $K(\alpha)/K$
غالوايّ؛ والتطبيق $\sigma \mapsto \sigma(\alpha)/\alpha \in \mu_n = \langle
\zeta\rangle$ تشاكل متباين (لأن $\sigma\tau(\alpha) =
\sigma(\tau(\alpha)/\alpha \cdot \alpha) =
\tau(\alpha)/\alpha\cdot\sigma(\alpha)$، إذ يقع
الخارج في $K$)، إلى زمرة دائرية: ومنه فإن $\operatorname{Gal}$ دائرية. وأما
العكس فهو نظرية كومر، ولن نحتاج إليها (انظر الملاحظة أدناه).
ومن أجل $K(\zeta_n)$: فهو ينشطر \omterm{def:b3:galois:separable}{لكثير الحدود القابل للفصل}
$X^n -
1$، ويغمر $\sigma \mapsto a(\sigma)$ مع $\sigma(\zeta_n) =
\zeta_n^{a(\sigma)}$ الزمرةَ في الزمرة الأبيلية
$(\Z/n\Z)^\times$ كما في \cref{thm:b3:galois:cyclotomicirred} (إذ لا يحتاج التباين إلا إلى أن
يولّد $\zeta_n$ جذورَ الوحدة المعنية).
\end{proof}

\begin{theorem}[غالوا]\label{thm:b3:galois:solvable}
ليكن $K$ حقلًا مميّزه $0$ وليكن $P \in K[X]$ حقلُ انشطاره $L$.
إذا كان $P$ \omterm{def:b3:galois:radical}{قابلًا للحل بالجذور}، فإن $\operatorname{Gal}(L/K)$ \omterm{def:b3:groups:derived}{زمرة قابلة للحل}.
(والعكس صحيح أيضًا؛ ولن نحتاج إليه.)\index{مبرهنة غالوا في
القابلية للحل}
\end{theorem}

\begin{proof}
الخطوة 1: \emph{توسيع البرج الجذريّ إلى برج غالوايّ.} ليكن
$L
\subseteq M$ حيث $M/K$ جذريّ، بأُسس جذرية $n_1, \dots,
n_r$ و $n = n_1\cdots n_r$. نضم
أولًا $\zeta_n$: فيبقى البرج $K \subseteq K(\zeta_n) \subseteq M(\zeta_n)$ \omterm{def:b3:galois:radical}{جذريًّا} ($\zeta_n$ جذر
للوحدة: أي خطوة جذرية، $\zeta_n^n = 1$)، وتحدث خطواته بعد الأولى
فوق حقول تحتوي على جذور الوحدة اللازمة. ثم نستبدل بالحقل $M(\zeta_n)$
المركّبَ $N$ لجميع الحقول $\sigma(M(\zeta_n))$، حيث يمسح $\sigma$ الغمورَ
على $K$ (وعددها منتهٍ) للحقل $M(\zeta_n)$ في \omterm{def:b3:galois:closure}{غلق جبري} مثبَّت:
فيكون $N$ حقلَ الانشطار لجداء كثيرات الحدود الدنيا لمجموعة
مولِّدة (والمميّز $0$: فهي منتهية وقابلة للفصل)، ومنه فإن $N/K$
غالوايّ؛ ويكون $N$ \omterm{def:b3:galois:radical}{جذريًّا} على $K$: إذ إن كل $\sigma(M(\zeta_n))$ جذريّ على
$K$ (بتطبيق $\sigma$ على برج جذريّ)، ومركّب الامتدادات الجذرية
جذريّ (بضم الأبراج: فإذا كان $F'/K$ \omterm{def:b3:galois:radical}{جذريًّا} ببرج يضم العناصر
$\beta_j$، بقيت الخطوات من نمط $F''(\beta_j)$ جذرية على أي
قاعدة أكبر).

الخطوة 2: \emph{قراءة القابلية للحل في برج غالوا.} لنفترض إذن
$L \subseteq N$ حيث $N/K$ غالوايّ وجذريّ ببرج $K
\subseteq K(\zeta_n) = E_0 \subseteq E_1 \subseteq \dots
\subseteq E_s = N$، وكل
$E_{i+1} = E_i(\sqrt[n_i]{a_i})$ مع $\zeta_{n_i} \in E_0 \subseteq E_i$. ولتكن $G =
\operatorname{Gal}(N/K)$ و $G_i = \operatorname{Gal}(N/E_i)$: فنحصل على سلسلة
متناقصة $G \supseteq G_0 \supseteq G_1 \supseteq
\dots \supseteq G_s = \{e\}$. وكل امتداد $E_{i+1}/E_i$ غالوايّ وزمرته دائرية
(\cref{lem:b3:galois:cyclicsteps})، ومنه، بتطبيق المبرهنة الأساسية على الامتداد
الغالوايّ $N/E_i$ (\cref{thm:b3:galois:fundamental}(3)، بالزمرة المحيطة $G_i$):
$G_{i+1} \trianglelefteq G_i$ مع $G_i/G_{i+1} \cong
\operatorname{Gal}(E_{i+1}/E_i)$ دائرية. وبالمثل يكون $E_0/K$ \omterm{def:b3:galois:galois}{غالوايًّا}
وزمرته الأبيلية $G/G_0$ (\cref{lem:b3:galois:cyclicsteps}). وتُظهر السلسلة أن $G$
\omterm{def:b3:groups:derived}{قابلة للحل} (\cref{prop:b3:groups:derived}). وأخيرًا، فإن $\operatorname{Gal}(L/K)$ \emph{زمرة
قسمة} للزمرة $G$: إذ إن $L/K$ غالوايّ (لأن $P$ قابل للفصل في المميّز
$0$) والقصر $G
\to \operatorname{Gal}(L/K)$ شامل (\cref{thm:b3:galois:fundamental}(3) مع $H =
\operatorname{Gal}(N/L)$)؛ وزمر
القسمة \omterm{def:b3:groups:derived}{لزمرة قابلة للحل} قابلةٌ للحل.
\end{proof}

\begin{corollary}[استحالة حل معادلة الدرجة الخامسة]
\label{cor:b3:galois:quintic}
توجد كثيرات حدود من \omterm{def:b3:galois:extension}{الدرجة} $5$ على $\Q$ غير \omterm{def:b3:groups:derived}{قابلة للحل} بالجذور:
مثلًا $X^5 - 4X + 2$، الذي \omterm{def:b3:galois:galois}{زمرة غالوا} له هي $S_5$
(\cref{exo:b3:galois:11})، وهي زمرة غير \omterm{def:b3:groups:derived}{قابلة للحل} (\cref{cor:b3:groups:snnotsolvable}). فلا يمكن
أن توجد صيغة عامة بالجذور من أجل \omterm{def:b3:galois:extension}{الدرجة} $\geq 5$.\index{معادلة
الدرجة الخامسة، استحالة حلها}
\end{corollary}

\begin{remark}\label{rem:b3:galois:converse}
أما عكس \cref{thm:b3:galois:solvable} --- أي إن \omterm{def:b3:galois:galois}{زمرة غالوا} \omterm{def:b3:groups:derived}{القابلة للحل} تستلزم
القابلية للحل بالجذور --- فيُبرهَن عليه بالنزول على \omterm{def:b3:groups:derived}{السلسلة
المشتقة} وبإظهار أن كل امتداد دائري (مع ما يكفي من جذور الوحدة)
جذريٌّ، عبر \emph{مُحلِّلات لاغرانج}؛ وهو يفسّر لماذا توجد صيغ
للدرجات $2, 3, 4$: لأن $S_2, S_3, S_4$ \omterm{def:b3:groups:derived}{قابلة للحل}
(\cref{ex:b3:groups:solvableexamples}). ونقبله في هذا المستوى؛ فالمعالجة الكاملة تخص
مقرّرًا في الماجستير، لكن \cref{exo:b3:galois:8} يجعله ملموسًا في حالة
المعادلة التكعيبية.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b3:galois:1}
برهن على $[\Q(\sqrt2, \sqrt3):\Q] = 4$، وعلى أن $\Q(\sqrt2 + \sqrt3) =
\Q(\sqrt2, \sqrt3)$، واحسب كثير الحدود الأدنى
للعنصر $\sqrt2 + \sqrt3$ على $\Q$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:galois:2}
ليكن $\alpha = \sqrt[3]2$. برهن على أن $\Q(\alpha)/\Q$ ليس ناظميًا (بإظهار
غمر $\Q(\alpha) \to \C$ صورته ليست $\Q(\alpha)$)، وعيّن \omterm{thm:b3:galois:splitting}{حقل الانشطار} $L$
لكثير الحدود $X^3 -
2$ \omterm{def:b3:galois:extension}{والدرجة} $[L:\Q]$، وتحقق من $\operatorname{Aut}_\Q(\Q(\alpha)) =
\{\mathrm{id}\}$: فمن
أجل الامتدادات غير الغالوايّة، يمكن لزمرة التماثلات الذاتية أن
تكون أصغر بكثير من \omterm{def:b3:galois:extension}{الدرجة}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:galois:3}
أنشئ $\mathbb F_9$ على الصورة $\mathbb F_3[X]/(X^2+1)$ وجد مولّدًا للزمرة $\mathbb F_9^\times$.
واسرد كثيرات الحدود الواحدية غير القابلة للاختزال من الدرجات
$1, 2, 3$ على $\mathbb F_2$، وتحقق من $X^8 - X = X(X+1)(X^3+X+1)(X^3+X^2+1)$ على $\mathbb F_2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:4}
(a) جد جميع الجذور البدائية بترديد $7$ وبترديد $11$ (أي مولّدات
الزمرتين $\mathbb F_7^\times$ و $\mathbb F_{11}^\times$).
(b) برهن على أنه من أجل $p$ فردي، يكون $x \in \mathbb F_p^\times$ مربّعًا إذا وفقط
إذا كان $x^{(p-1)/2} = 1$ (محك أويلر)، واستعد محك العدد $-1$ الوارد
في \cref{pb:b3:rings:1}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:5}
برهن على أن $\mathbb F_{p^m} \cap \mathbb F_{p^n} = \mathbb
F_{p^{\gcd(m,n)}}$ و $\mathbb F_{p^m}\mathbb F_{p^n} = \mathbb
F_{p^{\operatorname{lcm}(m,n)}}$ داخل \omterm{def:b3:galois:closure}{غلق جبري} مثبَّت $\bar{\mathbb F}_p$،
وصِف $\operatorname{Gal}(\mathbb F_{p^n}/\mathbb F_{p^m})$ من أجل $m
\mid n$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:6}
ليكن $I_d(q)$ عدد كثيرات الحدود الواحدية غير القابلة للاختزال من
\omterm{def:b3:galois:extension}{الدرجة} $d$ على $\mathbb F_q$. برهن على
\[
X^{q^n} - X \;=\; \prod_{d \mid n}\ \prod_{P \text{ واحدي غير قابل للاختزال، } \deg P = d} P ,
\qquad\text{ومنه}\qquad
q^n = \sum_{d \mid n} d\, I_d(q).
\]
واستنتج $I_1, I_2, I_3, I_4$ صراحةً، وأن $I_d(q) \geq 1$ من أجل كل $d$ (ومنه
توجد الامتدادات $\mathbb F_{q^d}/\mathbb F_q$ بوصفها حواصل قسمة $\mathbb F_q[X]/(P)$ من أجل كل
$d$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:7}
عيّن $\operatorname{Gal}(\Q(\sqrt2,\sqrt3)/\Q)$ والشبكة الكاملة للحقول الوسيطة. والسؤال نفسه من أجل
\omterm{thm:b3:galois:splitting}{حقل الانشطار} لكثير الحدود $(X^2-2)(X^2-3)(X^2-6)$ --- ماذا تلاحظ؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:8}
(المعادلة التكعيبية، محلولةً بزمرتها) ليكن $P = X^3 + pX + q \in
\Q[X]$ غير قابل
للاختزال وجذوره $x_1, x_2, x_3$ وحقل انشطاره $L$. وليكن $\delta = (x_1 - x_2)(x_1 - x_3)(x_2 - x_3)$
و $\Delta = \delta^2 = -4p^3 - 27q^2$ (اقبل هذه المتطابقة الكلاسيكية أو تحقق منها بنشر
الدوال المتناظرة).
(a) برهن على أن $\operatorname{Gal}(L/\Q) \cong A_3$ أو $S_3$، بحسب كون $\Delta$ مربّعًا في
$\Q$ أو عدمه.
(b) بوضع $j = \zeta_3$، عرّف مُحلِّلَي لاغرانج $u = x_1
+ jx_2 + j^2x_3$ و $v = x_1 + j^2x_2 + jx_3$.
برهن على $u^3 + v^3
= -27q$ و $uv = -3p$، ثم حل بدلالة $u^3, v^3$: فتسقط
صيغ كاردانو. وأين استُعملت قابلية $S_3$ للحل؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:galois:9}
في $\Q(\zeta_5)$: برهن على أن الحقل الجزئي التربيعي الوحيد هو
$\Q(\sqrt5)$، وذلك بواسطة مجموعَي غاوس $\eta_0 = \zeta_5 + \zeta_5^4$ و $\eta_1 = \zeta_5^2 + \zeta_5^3$: احسب
$\eta_0 + \eta_1$ و $\eta_0\eta_1$، واستنتج $\cos\frac{2\pi}5 = \frac{\sqrt5 -
1}4$. واخلص إلى أن المخمّس
المنتظم قابل للإنشاء.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:galois:10}
ليكن $K = \mathbb F_p(S, T)$ (الدوال الكسرية بمتغيّرين غير محدَّدين) وليكن
$L = K(S^{1/p}, T^{1/p})$.
(a) برهن على أن $[L:K] = p^2$ وأن $\alpha^p \in K$ من أجل كل
$\alpha \in L$.
(b) استنتج أن $L/K$ \emph{ليس} بسيطًا: فلا يوجد عنصر بدائي ---
إذ إن عدم القابلية للفصل قاتلة لما ورد في \cref{thm:b3:galois:primitive}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:galois:11}
ليكن $P = X^5 - 4X + 2$ ولتكن $G$ \omterm{def:b3:galois:galois}{زمرة غالوا} له على $\Q$، فاعلةً على
الجذور $5$.
(a) برهن على أن $P$ غير قابل للاختزال، واستنتج $5 \mid \abs G$؛
ثم اخلص إلى أن $G$ تحتوي على دورة خماسية (كوشي، \cref{thm:b3:groups:cauchy}).
(b) برهن، بدراسة تغيّرات $x \mapsto x^5 - 4x +
2$، على أن لكثير الحدود $P$ ثلاثة جذور
حقيقية بالضبط؛ واستنتج أن الاقتران العقدي يُقصَر إلى مبادلة في
$G$.
(c) برهن على أن كل زمرة جزئية من $S_5$ تحتوي على مبادلة وعلى
دورة خماسية هي $S_5$ \emph{(اقرن المبادلة بقوى الدورة)}. واخلص
إلى $G \cong S_5$، ومع \cref{thm:b3:galois:solvable}، إلى أن $P$ غير \omterm{def:b3:galois:radical}{قابل للحل
بالجذور}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:galois:12}
(المعادلة الرباعية ثنائية السطوح) ليكن $\alpha = \sqrt[4]2$ وليكن $L =
\Q(\alpha, \iu)$،
وهو \omterm{thm:b3:galois:splitting}{حقل الانشطار} لكثير الحدود $X^4 - 2$ على $\Q$.
(a) برهن على أن $[L : \Q] = 8$ وأن $G =
\operatorname{Gal}(L/\Q)$ مولَّدة بالعنصر $\sigma\colon
\alpha \mapsto \iu\alpha,\ \iu \mapsto \iu$
وبالاقتران العقدي $\tau$، مع $\sigma^4 = \tau^2 = e$ و $\tau\sigma\tau = \sigma^{-1}$: أي
$G \cong D_4$.
(b) اسرد شبكة الزمر الجزئية من $D_4$ (وهي عشر زمر) وطابق كلًّا
منها بحقلها الصامد؛ وتحقق على وجه الخصوص من أن $\Q(\sqrt2)$
و $\Q(\iu)$ و $\Q(\iu\sqrt2)$ هي الحقول الجزئية التربيعية
الثلاثة، وحدّد موضع $\Q(\alpha)$ و $\Q(\iu\alpha)$ و $\Q(\sqrt2, \iu)$.
(c) أي الحقول الوسيطة غالوايّ على $\Q$؟ طابق جوابك مع الزمر
الجزئية الناظمية من $D_4$، وفسّر لماذا يخفق $\Q(\alpha)/\Q$
بينما ينجح $\Q(\sqrt2)/\Q$.
\end{exercise}

\section{مسألة: غاوس والمضلع المنتظم ذو 17 ضلعًا}

\begin{problem}[{مسألة نهاية الأسبوع --- قابلية إنشاء المضلع ذي
17 ضلعًا}]\label{pb:b3:galois:1}
في 30 مارس 1796، برهن غاوس ذو التاسعة عشرة على أن المضلع المنتظم
ذا $17$ ضلعًا قابل للإنشاء --- وهو أول تقدّم في المسألة منذ
العصور القديمة. وسنعيد بناء حسابه بأدوات هذا الفصل. نضع $\zeta = \eu^{2\iu\pi/17}$،
$L =
\Q(\zeta)$، $G = \operatorname{Gal}(L/\Q)$.

\medskip
\noindent\textbf{الجزء الأول --- الزمرة وترشيحها.}
\begin{enumerate}
  \item برّر: $[L:\Q] = 16$، $G \cong (\Z/17\Z)^\times$، وهي دائرية رتبتها $16$.
        وتحقق من أن $3$ مولّد للزمرة $(\Z/17\Z)^\times$ \emph{(احسب قوى $3$
        بترديد $17$: $3, 9, 10, 13, 5, 15, 11, 16, \dots$)}.
  \item ليكن $\sigma \in G$ يحقق $\sigma(\zeta) = \zeta^3$، ولتكن $H_k = \langle \sigma^{2^k}\rangle$ من أجل
        $k = 0, \dots,
        4$. برهن على أن $G = H_0 \supset H_1 \supset H_2 \supset
        H_3 \supset H_4 = \{e\}$ حيث كل دليل $[H_k :
        H_{k+1}] = 2$، وعلى
        أن الحقول الصامدة $\Q = L_0
        \subset L_1 \subset L_2 \subset L_3 \subset L_4 = L$ تشكّل برجًا من الامتدادات
        التربيعية.
  \item اخلص \emph{قبليًّا}، باستعمال \cref{thm:b3:galois:wantzel}، إلى أن
        $\zeta$ --- ومنه المضلع ذا $17$ ضلعًا --- قابل للإنشاء.
        وبقية المسألة تجعل البرج صريحًا.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الدورات ذات الطول 8.}
نعرّف \emph{دورات غاوس}
\[
\eta_0 = \sum_{k \text{ زوجي}} \zeta^{3^k \bmod 17}
= \zeta^{1} + \zeta^{9} + \zeta^{13} + \zeta^{15} + \zeta^{16} +
\zeta^{8} + \zeta^{4} + \zeta^{2},
\qquad
\eta_1 = \sum_{k \text{ فردي}} \zeta^{3^k \bmod 17}.
\]
\begin{enumerate}[resume]
  \item برهن على أن $H_1$ يُثبّت $\eta_0, \eta_1$ وأن $\sigma$
        يبادل بينهما؛ واستنتج $\eta_0, \eta_1 \in L_1$ وأنهما جذرا معادلة من
        \omterm{def:b3:galois:extension}{الدرجة} الثانية على $\Q$.
  \item احسب $\eta_0 + \eta_1 = -1$. وبرهن على $\eta_0\eta_1 = -4$ \emph{(فكل جداء
        $\zeta^a\zeta^b$ هو $\zeta^c$ ما مع $c \neq 0$؛ أحصِ كم
        مرة يظهر كل $c$، أو حاجج بأن الجداء عدد صحيح ناطق
        يُثبّته $G$، ويساوي المجموع على جميع الجداءات $64$،
        واستعمل أن كل باقٍ غير معدوم يظهر بالتواتر نفسه)}.
  \item استنتج $\eta_0 = \frac{-1 + \sqrt{17}}2$، $\eta_1 =
        \frac{-1-\sqrt{17}}2$ (وعيّن أيهما أيّ عدديًا:
        $\eta_0 \approx 1.56$)، و $L_1 =
        \Q(\sqrt{17})$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الدورات ذات الطول 4 و 2.}
نعرّف
\[
\beta_0 = \zeta + \zeta^{13} + \zeta^{16} + \zeta^{4},\quad
\beta_1 = \zeta^3 + \zeta^5 + \zeta^{14} + \zeta^{12},\quad
\beta_2 = \zeta^9 + \zeta^{15} + \zeta^{8} + \zeta^{2},\quad
\beta_3 = \zeta^{10} + \zeta^{11} + \zeta^{7} + \zeta^{6}.
\]
\begin{enumerate}[resume]
  \item برهن على $\beta_0 + \beta_2 = \eta_0$ و $\beta_1 + \beta_3 =
        \eta_1$، وعلى أن $H_2$ يُثبّت $\beta_0, \beta_2$
        وأن $\sigma^2$ يبادل بينهما.
  \item احسب $\beta_0\beta_2 = -1$ و $\beta_1\beta_3 = -1$ \emph{(انشر: فالأُسس الستة عشر
        الناتجة تغطّي $1, \dots, 16$ مرة واحدة بالضبط)}.
  \item استنتج $\beta_0 = \frac{\eta_0 + \sqrt{\eta_0^2 + 4}}2$ (وتحقق من الإشارة عدديًا: $\beta_0 \approx 2.05$)
        والصيغة المماثلة من أجل $\beta_1$؛ ومنه $L_2 =
        \Q(\beta_0)$، وهو
        تربيعي على $L_1$.
  \item ليكن $\gamma_0 = \zeta + \zeta^{16} =
        2\cos\frac{2\pi}{17}$ وليكن $\gamma_1 = \zeta^{13} +
        \zeta^4$. برهن على $\gamma_0 + \gamma_1 = \beta_0$ و $\gamma_0\gamma_1 = \beta_1$،
        بحيث يكون $\gamma_0 =
        \frac{\beta_0 + \sqrt{\beta_0^2 - 4\beta_1}}2$.
  \item اجمع سلسلة الصيغ التي تعبّر عن $\cos\frac{2\pi}{17}$ بجذور تربيعية
        متداخلة، وأعطِ تحققًا عشريًا ($\cos\frac{2\pi}{17} \approx 0.93247$).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- خاتمة.}
\begin{enumerate}[resume]
  \item أين استعملت الحجّةُ بالضبط كونَ $17$ \emph{عددًا أوليًا
        لفيرما} ($17 = 2^{2^2} + 1$)؟ برهن على أنه من أجل عدد أولي $p$،
        يكون المضلع المنتظم ذو $p$ ضلعًا قابلًا للإنشاء إذا وفقط
        إذا كان $p
        = 2^{2^t} + 1$ من أجل $t$ ما \emph{(إذا كان
        $p - 1 = 2^m$، فبرهن على أن $m$ يجب أن يكون هو نفسه قوة
        للعدد 2)}.
  \item استنتج القائمة الكاملة للمضلعات المنتظمة ذات $n$ ضلعًا
        \omterm{def:b3:galois:constructible}{القابلة للإنشاء} من أجل $n \leq 20$، باستعمال محك
        غاوس--فانتزل الوارد في \cref{cor:b3:galois:impossible}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- مجاميع غاوس والتقابل
التربيعي.} تُخفي دورات الجزء الثاني كنزًا. من أجل عدد أولي فردي
$p$، يكون \emph{رمز لوجندر} $\bigl(\frac
ap\bigr)$ مساويًا $+1$ إذا كان $a$
مربّعًا غير معدوم بترديد $p$، و $-1$ إذا لم يكن كذلك، و $0$ إذا
كان $p \mid a$؛ ويعطي \cref{exo:b3:galois:4}(b) (محك أويلر) أن $\bigl(\frac ap\bigr) \equiv
a^{(p-1)/2} \pmod p$،
ومن هنا تأتي الضربية. نكتب $\zeta =
\eu^{2\iu\pi/p}$، $p^* = (-1)^{(p-1)/2}p$، ونعرّف \emph{مجموع
غاوس}
\[
g \;=\; \sum_{a=1}^{p-1}\Bigl(\frac ap\Bigr)\zeta^a .
\]
\begin{enumerate}[resume]
  \item برهن على $\sum_{a=1}^{p-1}\bigl(\frac ap\bigr) = 0$ (لأن عدد المربّعات يساوي عدد غير
        المربّعات)، وبرهن على الصيغة البديلة $g = \sum_{a=0}^{p-1}\zeta^{a^2}$ \emph{(إذ
        يُصاب كل مربّع غير معدوم مرتين، و $\sum_{a}\zeta^a = 0$)}. ومن أجل
        $p = 17$: اربط $g$ بدورات الجزء الثاني --- برهن على
        $g = \eta_0 - \eta_1$ \emph{(فالمربّعات بترديد $17$ هي بالضبط القوى
        الزوجية للمولّد $3$)}.
  \item برهن على $g^2 = p^*$: انشر
        \[
        g^2 = \sum_{a,b\neq0}\Bigl(\frac{ab}p\Bigr)
        \zeta^{a+b}
        = \sum_{c}\ \sum_{a \neq 0}\Bigl(\frac{a(c -
        a)}p\Bigr)\zeta^{c}
        \]
        (بوضع $b = c - a$)، وعوّض $c - a = at$ لتقييم المجموع
        الداخلي بالمقدار $\bigl(\frac{-1}p\bigr)(p - 1)$ من أجل $c = 0$ وبالمقدار
        $-\bigl(\frac{-1}p\bigr)$ فيما عدا ذلك، واخلص بالسؤال 14. وتحقق عدديًا:
        من أجل $p =
        17$، $(\eta_0 - \eta_1)^2 = 17$ (الجزء الثاني).
  \item استنتج $\sqrt{p^*} \in \Q(\zeta_p)$، واخلص إلى أن الحقل الجزئي التربيعي
        \emph{الوحيد} من $\Q(\zeta_p)$ هو $\Q(\sqrt{p^*})$ ---
        وهو وحيد لأن $\operatorname{Gal}(\Q(\zeta_p)/\Q)$ دائرية (\cref{thm:b3:galois:cyclotomicirred}) ولأن للزمرة
        الدائرية زمرةً جزئية واحدة بالضبط ذات الدليل $2$. (وكل
        حقل تربيعي ينغمر في حقل دائري ما --- وهذه أول حالة من
        مبرهنة كرونيكر--فيبر، التي تقع صيغتها العامة بعيدًا
        أمامنا.)
  \item وليكن الآن $q \neq p$ عددًا أوليًا فرديًا آخر. بالعمل في
        الحلقة $\Z[\zeta]$ بترديد $q$، برهن على
        \[
        g^q \equiv \Bigl(\frac qp\Bigr)\,g \pmod{q\Z[\zeta]}
        \]
        \emph{(بحلم المبتدئ: $(x + y)^q \equiv x^q + y^q$ بترديد $q$ في أي حلقة
        تبديلية؛ ثم $g^q \equiv
        \sum_a\bigl(\frac ap\bigr)^q\zeta^{aq}$، أعد ترقيم $b =
        aq$ وأخرج $\bigl(\frac{q^{-1}}p\bigr) =
        \bigl(\frac qp\bigr)$)}.
  \item ومن جهة أخرى، $g^q = g\,(g^2)^{(q-1)/2} =
        g\,(p^*)^{(q-1)/2}$؛ وباستعمال محك أويلر بترديد
        $q$، استنتج $g^q \equiv \bigl(\frac{p^*}q\bigr)g
        \pmod{q\Z[\zeta]}$، ثم --- بضرب التعبيرين عن $g^q$
        في $g$ واستعمال $g^2 = p^*$ القابل للقلب بترديد $q$ ---
        اخلص إلى
        \[
        \Bigl(\frac qp\Bigr) = \Bigl(\frac{p^*}q\Bigr) .
        \]
        \emph{(ولماذا تستلزم مطابقةٌ بين العددين الصحيحين
        $\pm p^*$ بترديد $q\Z[\zeta]$ تساويَهما؟ تقاطع مع
        $\Z$.)}
  \item افتح $\bigl(\frac{p^*}q\bigr) =
        \bigl(\frac{-1}q\bigr)^{(p-1)/2}\bigl(\frac pq\bigr)$ و $\bigl(\frac{-1}q\bigr) = (-1)^{(q-1)/2}$ للحصول على \emph{قانون التقابل
        التربيعي}:
        \[
        \Bigl(\frac pq\Bigr)\Bigl(\frac qp\Bigr)
        = (-1)^{\frac{p-1}2\cdot\frac{q-1}2} .
        \]
        وتحقق منه على $(p, q) = (17, 3)$ بسرد المربّعات بترديد $17$ وبترديد
        $3$، واستعمله لتقرّر في ثلاثة أسطر ما إذا كانت $x^2 \equiv 219 \pmod{383}$
        \omterm{def:b3:groups:derived}{قابلة للحل} ($383$ أولي، و $219 = 3\cdot73$).
\end{enumerate}

\medskip
\noindent\textbf{الجزء السادس --- إحصاء كثيرات الحدود غير
القابلة للاختزال: مبرهنة الأعداد الأولية للحلقة
$\mathbb F_q[X]$.} لنثبّت قوة عدد أولي $q$ وليكن $N_q(n)$ عدد
كثيرات الحدود \emph{الواحدية غير القابلة للاختزال} من \omterm{def:b3:galois:extension}{الدرجة}
$n$ على $\mathbb F_q$؛ ونذكّر من \cref{exo:b3:galois:6} بتفكيك $X^{q^n} - X$
وبالمتطابقة $q^n = \sum_{d\mid n}d\,N_q(d)$، وهي ما سنقلبه الآن ونعيد تفسيره ونستثمره.
\begin{enumerate}[resume]
  \item (الكلمات) نقول عن كلمة $w \in \mathbb F_q^n$ إنها \emph{بدائية} إذا لم
        تكن قوةً $u^{n/d} =
        u\cdots u$ لكلمة $u$ أقصر تمامًا، وليكن $A(d)$ عدد
        الكلمات البدائية ذات الطول $d$. برهن على أن كل كلمة طولها
        $n$ هي بطريقة وحيدة قوةٌ لكلمة بدائية طولها $d \mid n$
        ما، ومنه $q^n = \sum_{d \mid n}A(d)$؛ وبالمقارنة مع \cref{exo:b3:galois:6}، اخلص إلى
        $A(d) = d\,N_q(d)$ من أجل كل $d$، وفسّر هذا التوافق بتقابل صريح:
        فلعنصر $\alpha \in
        \mathbb F_{q^d}$ من \omterm{def:b3:galois:extension}{الدرجة} $d$ مدارُ فروبينيوس $(\alpha, \alpha^q, \dots, \alpha^{q^{d-1}})$
        المؤلَّف من $d$ عنصرًا متمايزًا بالضبط، وتوافق العناصر
        ذات \omterm{def:b3:galois:extension}{الدرجة} $d$ كثيراتِ الحدود غير القابلة للاختزال من
        \omterm{def:b3:galois:extension}{الدرجة} $d$ توافقًا $d$ إلى واحد.
  \item برهن على \emph{صيغة قلب موبيوس}: إذا كان $f(n) =
        \sum_{d\mid n}g(d)$ من أجل
        كل $n$، فإن $g(n) =
        \sum_{d\mid n}\mu(d)\,f(n/d)$، حيث $\mu$ دالة موبيوس (أي $\mu(m) = (-1)^{\#\text{عوامل أولية}}$
        إذا كان $m$ خاليًا من العوامل المربّعة، و $0$ فيما عدا
        ذلك) \emph{(والمبرهنة المساعدة المفتاحية: $\sum_{d \mid m}\mu(d) = 0$ من
        أجل $m >
        1$ --- زاوج القواسم التي تحتوي على عامل أولي
        مثبَّت مع تلك التي لا تحتوي عليه)}. واستنتج
        \[
        N_q(n) = \frac1n\sum_{d \mid n}\mu(d)\,q^{n/d} .
        \]
  \item برهن على $N_q(n) \geq \frac1n\bigl(q^n - 2q^{n/2}\bigr) > 0$ من أجل كل $n \geq 1$: وهو برهان جديد
        على وجود $\mathbb F_{q^n}$ من أجل كل $n$. وفسّر الحد المهيمن:
        فكثير حدود واحدي عشوائي من \omterm{def:b3:galois:extension}{الدرجة} $n$ يكون غير قابل
        للاختزال باحتمال $\sim \frac1n$ --- وهو النظير التام
        لمبرهنة الأعداد الأولية، مع استبدال $n$ بالمقدار $\log x$؛
        وتحقق عدديًا من أجل $q = 2$ و $n \leq 4$ (يسرد
        \cref{exo:b3:galois:6} الأعداد).
  \item برهن على النظير الضربي للسؤال 21:
        \[
        \prod_{\substack{\pi \text{ واحدي غير قابل للاختزال}\\ \deg\pi =
        n}}\pi \;=\; \prod_{d \mid n}
        \bigl(X^{q^d} - X\bigr)^{\mu(n/d)}
        \]
        \emph{(بقلب موبيوس في الزمرة الأبيلية للدوال الكسرية غير
        المعدومة)}؛ وتحقق منه يدويًا من أجل $q
        = 2$ و $n = 2$:
        $(X^4 - X)/(X^2 - X) = X^2 + X + 1$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء السابع --- خاتمتان.}
\begin{enumerate}[resume]
  \item (التكملة الثانية) تحسب طريقةُ الجزء الخامس أيضًا $\bigl(\frac2q\bigr)$.
        ليكن $\omega = \eu^{2\iu\pi/8}$ وليكن $g = \omega + \omega^{-1}$. برهن على $g^2 = 2$
        \emph{(لأن $\omega^2 = \iu$)}؛ ثم، من أجل عدد أولي فردي
        $q$، برهن في $\Z[\omega]$ بترديد $q$ على
        \[
        g^q \equiv \omega^q + \omega^{-q} \pmod{q\Z[\omega]},
        \]
        وعلى أن الطرف الأيمن يساوي $g$ إذا كان $q \equiv \pm1
        \pmod 8$ ويساوي
        $-g$ إذا كان $q \equiv \pm3 \pmod 8$. وبالمقارنة مع $g^q = g\,(g^2)^{(q-1)/2} \equiv
        \bigl(\frac2q\bigr)g$ كما في
        السؤال 18، اخلص إلى
        \[
        \Bigl(\frac2q\Bigr) = (-1)^{(q^2-1)/8},
        \]
        مع التحقق من أن $(q^2 - 1)/8$ زوجي بالضبط حين يكون
        $q
        \equiv \pm1 \pmod 8$. تحقق: العدد $2$ مربّع بترديد $7$ وبترديد $17$
        ($3^2$ و $6^2$)، وليس مربّعًا بترديد $3$ ولا بترديد $5$.
  \item (دالة زيتا للحلقة $\mathbb F_q[X]$) برهن على متطابقة
        المتسلسلات الصورية بالمتغيّر $t$:
        \[
        \prod_{n \geq 1}\bigl(1 - t^n\bigr)^{-N_q(n)}
        = \frac1{1 - qt}
        \]
        \emph{(بوحدانية التفكيك إلى كثيرات حدود واحدية غير قابلة
        للاختزال: انشر كل عامل على صورة متسلسلة هندسية وأحصِ
        كثيرات الحدود الواحدية من \omterm{def:b3:galois:extension}{الدرجة} $n$)}. واستعد المتطابقة
        $q^m = \sum_{d \mid m}d\,N_q(d)$ الواردة في \cref{exo:b3:galois:6} بأخذ اللوغاريتمات. وتحقق
        من معامل $t^2$ يدويًا من أجل $q = 2$، واستعمل صيغة
        السؤال 21 لحساب $N_2(6) = 9$، متحققًا من $2^6 = 1\cdot2 + 2\cdot1 + 3\cdot2 +
        6\cdot9$.
\end{enumerate}
\end{problem}
