\chapter{التطبيقات المطابقة ومبرهنة ريمان في التطبيق}
\label{ch:b3:conformal}

ينقل تقابلٌ هولومورفي بين مجالين كلَّ التحليل العقدي من أحدهما
إلى الآخر: فهذه التطبيقات --- \emph{المطابِقة}، لأنها تحفظ
الزوايا --- هي تماثلات العالم الهولومورفي. ويصنّف هذا الفصل هذه
التطبيقات حيث يمكن التصنيف (القرص، والمستوي: ومبرهنة شوارتز
المساعدة هي المفتاح، وهي متراجحة مذهلة القوة)، ويبني نظرية
التراص للعائلات \omterm{def:b3:holomorphic:holo}{الهولومورفية} (مونتيل)، ويبرهن على أعمق مبرهنات
الوجود في هذا الموضوع: أن \emph{كل} مجال جزئي فعلي بسيط الترابط
من $\C$، مهما كانت حافته متعرّجة، يكافئ القرصَ الواحدي مطابقةً.
ونختم بالدوال التوافقية \omterm{thm:b3:conformal:poisson}{ونواة بواسون}، فنحل مسألة ديريكليه على
القرص --- وهي العائد التحليلي للهندسة المطابقة. وفي كل ما يلي،
$\mathbb D = D(0,1)$ و $\mathbb H = \{\operatorname{Im}z >
0\}$.

\section{التطبيقات المطابقة؛ تحويلات موبيوس}

\begin{definition}\label{def:b3:conformal:conformal}
\emph{التطبيق المطابق}\index{تطبيق مطابق} (أو التقابل
الهولومورفي) بين مجموعتين مفتوحتين هو تقابل هولومورفي؛ ومقلوبه
هولومورفي تلقائيًّا (\cref{cor:b3:residues:openmapping}). ويقال إن مجالين \emph{متكافئان
مطابقةً} إذا وُجد تطبيق كهذا؛ ويرمز $\operatorname{Aut}(\Omega)$ إلى زمرة التطبيقات
الذاتية المطابقة. وحيث $f' \neq 0$ --- أي في كل مكان من أجل $f$
متباينة (ببرهان \cref{cor:b3:residues:openmapping}) --- يكون التفاضل هو الضرب في
$f'(z) \neq 0$: أي تشابهٌ، ومنه تحفظ التطبيقات المطابقة الزوايا
بين المنحنيات، بما فيها التوجيه.
\end{definition}

\begin{example}[تحويلات موبيوس]
\label{ex:b3:conformal:mobius}
من أجل $\bigl(\begin{smallmatrix}a & b\\ c &
d\end{smallmatrix}\bigr) \in GL_2(\C)$، يكون \emph{تحويل موبيوس}\index{تحويل موبيوس}
$z \mapsto
\frac{az + b}{cz + d}$ مطابقًا من $\C\setminus\{-d/c\}$ على $\C\setminus\{a/c\}$ (ومقلوبه من النمط نفسه،
انطلاقًا من المصفوفة المقلوبة؛ ويوافق التركيبُ جداءَ
المصفوفات). و\emph{تحويل كايلي}\index{تحويل كايلي}
\[
\varphi(z) = \frac{z - \iu}{z + \iu}
\]
يرسل $\mathbb H$ مطابقةً على $\mathbb D$: إذ يكون $\abs{z
- \iu} < \abs{z + \iu}$ بالضبط
حين يكون $z$ أقرب إلى $\iu$ منه إلى $-\iu$، أي $\operatorname{Im}z > 0$؛ ومقلوبه
$w \mapsto \iu\frac{1 + w}{1 - w}$. وترسل تحويلات موبيوس عائلةَ الدوائر والمستقيمات إلى
نفسها (\cref{exo:b3:conformal:1}).
\end{example}

\begin{example}[تطبيق جوكوفسكي]\label{ex:b3:conformal:joukowski}
وراء موبيوس، يكون أنفع \omterm{def:b3:conformal:conformal}{تطبيق مطابق} في الرياضيات التطبيقية
الكلاسيكية هو
\[
J(z) = \frac12\Bigl(z + \frac1z\Bigr) .
\]
وعلى خارج القرص الواحدي $\Omega = \{\abs z > 1\}$، يكون $J$ متباينًا: إذ يعطي
$J(z) = J(w)$ أن $(z - w)(1 -
\frac1{zw}) = 0$ مع $\abs{zw} > 1$. ولا تنعدم مشتقته
$J'(z) =
\frac12(1 - z^{-2})$ إلا عند $z = \pm1$ على الحافة: ومنه فإن $J$ تكافؤ مطابق
من $\Omega$ على صورته، وهي $\C\setminus\intcc{-1}1$ --- إذ تُطوى الدائرة الواحدية
نفسها اثنين إلى واحد على القطعة ($J(\eu^{\iu\theta}) = \cos\theta$). ومن ثَمّ فإن خارج
\emph{قطعة}، وهو مستوٍ مشقوق بلا حافة ناعمة، هو مطابقةً خارجُ
قرص: فالزوايا ليست عائقًا أمام التكافؤ المطابق، بل أمام نعومة
الحافة فحسب. وصور الدوائر المارّة بالنقطتين $\pm1$ لكن غير
المتمركزة منحنياتٌ على شكل مقطع جناح، وقد نقل تركيبُ $J$ مع
تحويلات موبيوس الجريانَ حول أسطوانة --- وهو قابل للحساب يدويًّا
--- إلى الجريان حول جناح: فكان هذا المثال، في النصف الأول من
القرن العشرين، \emph{هو} علم الديناميكا الهوائية. وهو أيضًا
الباب إلى تشيبيشيف: إذ يقارن $J$ بين $z \mapsto z^n$ وكثير حدود
تشيبيشيف $T_n$ (\cref{pb:b3:hilbert:1}، الجزء الخامس)، لأن $J(z^n) = \cos(n\theta)$ حين
$z = \eu^{\iu\theta}$.
\end{example}

\section{مبرهنة شوارتز المساعدة وزمر التشاكلات الذاتية}

\begin{theorem}[مبرهنة شوارتز المساعدة]\label{thm:b3:conformal:schwarz}
لتكن $f \colon \mathbb D \to \mathbb D$ \omterm{def:b3:holomorphic:holo}{هولومورفية} مع $f(0) = 0$.
عندئذٍ\index{مبرهنة شوارتز المساعدة}
\[
\abs{f(z)} \leq \abs z \ \ (z \in \mathbb D)
\qquad\text{و}\qquad \abs{f'(0)} \leq 1 ,
\]
وإذا كان $\abs{f(z_0)} = \abs{z_0}$ من أجل $z_0 \neq 0$ واحدة، أو كان $\abs{f'(0)} = 1$، فإن
$f(z) = \eu^{\iu\theta}z$ يكون دورانًا.
\end{theorem}

\begin{proof}
تمتدّ $g(z) = f(z)/z$ هولومورفيًّا إلى $\mathbb D$ (إذ الشذوذ
عند $0$ قابل للإزالة: لأن $g$ محدودة بجوار $0$، \cref{thm:b3:residues:riemanncw}؛
وقيمتها عند $0$ هي $f'(0)$). وعلى $\abs z = r < 1$: $\abs g \leq \frac1r$،
ومنه، بمبدأ الأعظمية (\cref{thm:b3:holomorphic:maximum})، $\abs g \leq \frac1r$ على $\bar D(0,r)$؛
ونجعل $r \to 1$: $\abs
g \leq 1$ على $\mathbb D$، وهما المتراجحتان معًا.
والتساوي عند نقطة داخلية يجعل $\abs g$ تبلغ قيمةً عظمى داخلية:
فتكون $g$ ثابتة طويلتها $1$.
\end{proof}

\begin{theorem}[التشاكلات الذاتية للقرص]
\label{thm:b3:conformal:autdisc}
من أجل $a \in \mathbb D$، يكون \emph{عامل بلاشكه}\index{عامل بلاشكه}
\[
\varphi_a(z) = \frac{z - a}{1 - \bar a z}
\]
تشاكلًا ذاتيًّا للقرص $\mathbb D$ يبادل بين $a$ و $0$، مع
$\varphi_a^{-1} = \varphi_{-a}$. وكل تشاكل ذاتي للقرص $\mathbb D$ هو $\eu^{\iu\theta}\varphi_a$ من أجل
$\theta
\in \R/2\pi\Z$ و $a \in \mathbb D$ وحيدين.
\end{theorem}

\begin{proof}
على $\abs z = 1$: $\abs{1 - \bar az} = \abs{\bar z}\abs{1 -
\bar az} = \abs{\bar z - \bar a\abs z^2} = \abs{\bar z - \bar
a} = \abs{z - a}$، ومنه $\abs{\varphi_a} = 1$ هناك؛ وبمبدأ الأعظمية
$\varphi_a(\mathbb D) \subseteq \bar{\mathbb
D}$، ويضع الانفتاح الصورةَ في $\mathbb D$. وتبيّن المتطابقة
الجبرية $\varphi_{-a}\circ\varphi_a = \mathrm{id}$ (بالحساب المباشر) أنه تقابل. والآن لتكن $f \in
\operatorname{Aut}(\mathbb D)$
وليكن $a = f^{-1}(0)$: فإن $g =
f\circ\varphi_{-a}$ تشاكل ذاتي يثبّت $0$. وبتطبيق
شوارتز على $g$ وعلى $g^{-1}$: $\abs{g(z)} \leq \abs z$ و $\abs{g^{-1}(w)} \leq \abs w$، ومنه $\abs{g(z)} = \abs z$: أي
دوران، $g = \eu^{\iu\theta}\,\mathrm{id}$، أي $f =
\eu^{\iu\theta}\varphi_a$. وأما الوحدانية: فمن $a = f^{-1}(0)$
ومن $\theta$ بتقييم من نمط $f'$ (أو من $f(0) =
-\eu^{\iu\theta}a$ وقيمة أخرى).
\end{proof}

\begin{theorem}[التشاكلات الذاتية للمستوي]
\label{thm:b3:conformal:autplane}
$\operatorname{Aut}(\C) = \{z \mapsto az + b : a \in \C^*,\ b
\in \C\}$. ومن ثَمّ فإن $\C$ و $\mathbb D$ غير متكافئين مطابقةً.
\end{theorem}

\begin{proof}
لتكن $f \in \operatorname{Aut}(\C)$ ولننظر في $g(z) =
f(1/z)$ على $\C^*$: وهي \omterm{def:b3:holomorphic:holo}{دالة هولومورفية}
لها \omterm{def:b3:residues:singularities}{شذوذ معزول} عند $0$. فلو كان جوهريًّا، لجعلت
كازوراتي--فايرشتراس (\cref{thm:b3:residues:riemanncw}) المجموعةَ $g\bigl(D(0,\varepsilon)\setminus\{0\}\bigr)$ كثيفة، بينما
$f(D(0, 1))$ مفتوحة ومنفصلة عنها (إذ $f$ متباينة: فالمجموعتان
صورتان لمجموعتين منفصلتين) --- وهو مستحيل من أجل \omterm{def:b3:topology:interior}{مجموعة كثيفة}
\omterm{def:b3:topology:topology}{ومجموعة مفتوحة} غير خالية. ومنه يكون $0$ قطبًا أو قابلًا
للإزالة من أجل $g$، أي إن $\abs{f(z)}$ ذو نمو حدودي على
الأكثر: فتكون $f$ كثير حدود (\cref{exo:b3:holomorphic:4}(a)). ويفرض التباين
الدرجةَ $1$: فلكثير حدود من درجة أعلى إما جذرٌ مضاعف للمقدار
$f - c$ في موضع ما (إذ تنعدم $f'$) وإما عدة صور عكسية متمايزة
(دالمبير--غاوس، \cref{pb:b3:holomorphic:1})؛ وفي الحالتين يخفق التباين. وأخيرًا،
فتطبيقٌ مطابق $\C \to
\mathbb D$ كان سيكون دالةً صحيحة محدودة: أي ثابتة
(بليوفيل) --- فلا تكافؤ.
\end{proof}

\section{مبرهنة مونتيل}

\begin{theorem}[مونتيل]\label{thm:b3:conformal:montel}
لتكن $\mathcal F \subseteq \mathcal H(\Omega)$ \emph{محدودة محليًّا}: أي لكل نقطة جوارٌ يكون عليه
$\sup_{f\in
\mathcal F}\sup\abs f < \infty$. عندئذٍ لكل متتالية من $\mathcal F$ متتاليةٌ جزئية
تتقارب بانتظام على جميع الأجزاء المتراصة من $\Omega$ (إلى نهاية
\omterm{def:b3:holomorphic:holo}{هولومورفية}).\index{مبرهنة مونتيل}
\end{theorem}

\begin{proof}
\omterm{def:b3:complete:equicontinuous}{تساوي الاتصال} المحلي: إذا كان $\abs f \leq M$ على $D(a, 2r)
\subseteq \Omega$ من أجل
كل $f \in \mathcal F$، أعطت صيغة كوشي، من أجل $z, z' \in D(a, r)$،
\[
\abs{f(z) - f(z')} = \frac{\abs{z - z'}}{2\pi}
\Bigl|\int_{C_{2r}}\frac{f(w)\,\dd w}{(w-z)(w-z')}\Bigr|
\leq \frac{\abs{z - z'}\,2\pi\cdot2r\,M}{2\pi\,r^2}
= \frac{2M}{r}\,\abs{z - z'} :
\]
أي حدًّا ليبشيتزيًّا منتظمًا. ونستنفد $\Omega$ بمتراصات $K_m$؛
وتغطّي كلَّ $K_m$ عددٌ منتهٍ من هذه الأقراص، ومنه تكون
$\mathcal F$ محدودة بانتظام \omterm{def:b3:complete:equicontinuous}{ومتساوية الاتصال} على $K_m$: فتستخرج
أرزيلا--أسكولي (\cref{thm:b3:complete:ascoli}) متتاليةً جزئية متقاربة بانتظام على
$K_m$؛ ثم نقطّر على $m$. والنهاية \omterm{def:b3:holomorphic:holo}{هولومورفية} حسب \cref{thm:b3:holomorphic:weierstrassconv}.
\end{proof}

\section{مبرهنة ريمان في التطبيق}

\begin{definition}\label{def:b3:conformal:simplyconnected}
تكون \omterm{def:b3:topology:topology}{المجموعة المفتوحة} المترابطة $\Omega \subseteq \C$ \emph{بسيطة
الترابط}\index{مجال بسيط الترابط} (بالمعنى الهومولوجي، وهو كافٍ
لجميع أغراضنا) إذا كان $\operatorname{Ind}_\gamma(w) = 0$ من أجل كل دورة $\gamma$ في
$\Omega$ ومن أجل كل $w \notin \Omega$ --- أي «لا تحيط أي دورة من $\Omega$
بثقب». وحسب مبرهنة كوشي الشاملة (\cref{thm:b3:residues:globalcauchy}) وحسب \cref{prop:b3:holomorphic:primitive}،
يكون على $\Omega$ كهذه \emph{لكل \omterm{def:b3:holomorphic:holo}{دالة هولومورفية} دالةٌ أصلية}؛
ومن ثَمّ لكل $f \in \mathcal H(\Omega)$ عديمة الجذور لوغاريتمٌ هولومورفي
($\exp\circ$(الدالة الأصلية للمقدار $f'/f$)، معدَّلًا بثابت،
بحيث $(f\eu^{-L})' = 0$) وجذور \omterm{def:b3:holomorphic:holo}{هولومورفية} من الرتبة $n$ هي
$\eu^{L/n}$.\index{لوغاريتم هولومورفي}
\end{definition}

\begin{theorem}[مبرهنة ريمان في التطبيق]
\label{thm:b3:conformal:rmt}
كل \omterm{def:b3:topology:topology}{مجموعة مفتوحة} \omterm{def:b3:conformal:simplyconnected}{بسيطة الترابط} $\Omega \subsetneq \C$ مع $\Omega
\neq \varnothing$ تكافئ
$\mathbb D$ مطابقةً؛ وبإعطاء $z_0 \in \Omega$، يوجد \omterm{def:b3:conformal:conformal}{تطبيق مطابق}
وحيد $f \colon
\Omega \to \mathbb D$ يحقق $f(z_0) = 0$ و $f'(z_0) >
0$.\index{مبرهنة ريمان في
التطبيق}
\end{theorem}

\begin{proof}
\emph{الخطوة 0: العائلة غير خالية.} نختار $b \notin \Omega$: فالمقدار
$z - b$ عديم الجذور على $\Omega$، ومنه له جذر تربيعي هولومورفي
$h$ (حيث $h^2 = z - b$). و $h$ متباين (إذ $h(z) =
h(z')$ يعطي بالتربيع
$z = z'$)، وإذا كان $w \in h(\Omega)$ فإن $-w
\notin h(\Omega)$ (لأن $h(z) = -h(z')$
يعطي أيضًا بالتربيع $z = z'$، فيعطي $w = -w = 0$، وهو مستحيل
لأن $h$ عديم الجذور). وبما أن $h(\Omega)$ مفتوحة، فهي تحتوي
قرصًا $D(h(z_0), \rho)$؛ عندئذٍ $D(-h(z_0), \rho) \cap h(\Omega) = \varnothing$، أي $\abs{h(z) + h(z_0)} \geq \rho$ من أجل كل $z \in \Omega$.
ومنه فإن
\[
g(z) = \frac{\rho}{2\,\bigl(h(z) + h(z_0)\bigr)}
\]
\omterm{def:b3:holomorphic:holo}{هولومورفية} ومتباينة (بوصفها \omterm{ex:b3:conformal:mobius}{تحويل موبيوس} مركَّبًا مع $h$
المتباين)، مع $\abs g \leq \frac12 < 1$. وبتركيبها مع \omterm{thm:b3:conformal:autdisc}{عامل بلاشكه} (\cref{thm:b3:conformal:autdisc})
لنقل $g(z_0)$ إلى $0$، تكون العائلة
\[
\mathcal F = \{f \colon \Omega \to \mathbb D \text{ هولومورفية ومتباينة،} f(z_0) = 0\}
\]
غير خالية.

\emph{الخطوة 1: عنصر أقصوي.} لتكن $s =
\sup_{\mathcal F}\abs{f'(z_0)} \in \intoc0{+\infty}$ ($> 0$: إذ الأعضاء
متباينون، ومنه $f'(z_0) \neq 0$). ونأخذ $f_n \in
\mathcal F$ تحقق $\abs{f_n'(z_0)} \to s$:
فالعائلة محدودة بالعدد $1$، ومنه يستخرج مونتيل (\cref{thm:b3:conformal:montel})
متتاليةً $f_n \to f$ بانتظام على المتراصات؛ و $f$ \omterm{def:b3:holomorphic:holo}{هولومورفية}،
و $f(z_0) = 0$، و $\abs{f'(z_0)} = s$ (\cref{thm:b3:holomorphic:weierstrassconv} من أجل المشتقات)، وعلى
وجه الخصوص $f$ غير ثابتة؛ و $f$ متباينة بهورفيتز
(\cref{exo:b3:residues:8}(b))، و $f(\Omega) \subseteq \bar{\mathbb D}$، ومنه $\subseteq
\mathbb D$ (بالتطبيق المفتوح).
ومنه تبلغ $f \in \mathcal F$ السوپريموم: أي $s < \infty$.

\emph{الخطوة 2: التطبيق الأقصوي غامر.} لنفترض أن $a \in
\mathbb D\setminus f(\Omega)$.
فالنقل ببلاشكه $\varphi_a\circ f$ عديم الجذور على $\Omega$ البسيط
الترابط: ومنه له جذر تربيعي هولومورفي $F$ (مع $F(\Omega) \subseteq \mathbb D$، إذ
$\abs F^2 =
\abs{\varphi_a\circ f} < 1$)، ومتباين (لأن المربّعات تميّز). ونعيِّر: $G = \varphi_{F(z_0)}\circ F \in
\mathcal F$.
وبالرجوع: $f = \varphi_{-a}\circ s_2 \circ
\varphi_{-F(z_0)}\circ G$ حيث $s_2(w) = w^2$؛ والتطبيق $\Psi
= \varphi_{-a}\circ s_2\circ\varphi_{-F(z_0)} \colon \mathbb D
\to \mathbb D$
هولومورفي مع $\Psi(0) = f(z_0) = 0$ و\emph{ليس} دورانًا (إذ ليس متباينًا: لأن
$s_2$ ليست كذلك). وبمبرهنة شوارتز المساعدة (الحالة الأكيدة):
$\abs{\Psi'(0)} < 1$، وتعطي قاعدة السلسلة $f = \Psi\circ G$ أن $\abs{f'(z_0)} =
\abs{\Psi'(0)}\,\abs{G'(z_0)} < \abs{G'(z_0)}$ --- وهو يناقض
الأقصوية (ولاحظ $G \in \mathcal F$). ومنه تكون $f$ غامرة: أي تكافؤًا
مطابقًا.

\emph{الخطوة 3: التعيير والوحدانية.} نضرب $f$ في $\eu^{-\iu\arg f'(z_0)}$ لنجعل
$f'(z_0) > 0$ (وهذا يبقى في $\mathcal F$). فإذا صلحت $f_1, f_2$
كلتاهما، ثبّت $\psi =
f_2\circ f_1^{-1} \in \operatorname{Aut}(\mathbb D)$ النقطةَ $0$ مع $\psi'(0) = f_2'(z_0)/f_1'(z_0) > 0$؛ وحسب \cref{thm:b3:conformal:autdisc}
يكون $\psi$ دورانًا $\eu^{\iu\theta}$ مع $\eu^{\iu\theta} > 0$: أي $\psi =
\mathrm{id}$.
\end{proof}

\begin{remark}\label{rem:b3:conformal:rmtscope}
المبرهنة عبارةُ وجود محضة ذات نطاق مذهل: فمربّع، ونصف مستوٍ،
ومتممة شقّ، والمنطقة بين دائرتين متماستين، ومجال ذو حافة
كسيرية --- كلها مطابقةً مطابقةٌ للقرص $\mathbb D$. وأما ما لا
تعطيه: فأي صيغة (إذ التطبيقات الصريحة هي الاستثناء:
\cref{exo:b3:conformal:5})، ولا السلوك على الحافة (وتتكفّل به نظرية أعمق ---
مبرهنة كاراتيودوري)، ولا وحدانية التمديد إلى $\C$ ولا المجالات
متعددة الترابط: فالتاج $\{1 < \abs z < 2\}$ \emph{ليس} مطابقةً قرصًا
مثقوبًا، والتيجان ذات نسب أنصاف أقطار مختلفة غير متكافئة (وهي
حقيقة أصعب فعلًا).
\end{remark}

\section{الدوال التوافقية ونواة بواسون}

\begin{proposition}\label{prop:b3:conformal:harmonicholo}
لتكن $\Omega$ \omterm{def:b3:conformal:simplyconnected}{بسيطة الترابط} ولتكن $u \colon \Omega \to \R$
توافقية\index{دالة توافقية} (أي من الصنف $\mathcal C^2$ مع
$\Delta u = u_{xx} + u_{yy} = 0$). عندئذٍ $u =
\operatorname{Re}F$ من أجل $F$ \omterm{def:b3:holomorphic:holo}{هولومورفية}، وحيدة إلى غاية
ثابت تخيّلي. ومن ثَمّ تكون $u$ من الصنف $\mathcal C^\infty$، وتحقق خاصية
القيمة المتوسطة، وتخضع لمبدأ الأعظمية (فلا قيمة قصوى داخلية
أكيدة إلا إذا كانت ثابتة).
\end{proposition}

\begin{proof}
تحقق $g = u_x - \iu u_y$ معادلتَي كوشي--ريمان ($P = u_x$ و $Q = -u_y$:
$P_x = u_{xx} = -u_{yy} = Q_y$ و $P_y = u_{xy} = u_{yx} = -Q_x$) بمشتقات جزئية متصلة: ومنه تكون $g$
\omterm{def:b3:holomorphic:holo}{هولومورفية} (\cref{prop:b3:holomorphic:cauchyriemann}؛ وتنتج القابلية للاشتقاق بالمعنى $\R$
من $\mathcal C^1$). وليكن $F_0$ دالةً أصلية (ببساطة الترابط،
\cref{def:b3:conformal:simplyconnected})؛ عندئذٍ يكون للمقدار $\operatorname{Re}F_0$ التدرّج $(u_x, u_y)$
(إذ يُفكّ $F_0' = g$ إلى ذلك بالضبط عبر كوشي--ريمان من أجل
$F_0$)، ومنه يكون $u - \operatorname{Re}F_0$ ثابتًا ($\Omega$ مترابطة): فنضبط
$F = F_0 + c$. وتنتقل الخصائص من \cref{thm:b3:holomorphic:maximum} ومن \cref{exo:b3:holomorphic:10}
(ومن أجل مبدأ الأعظمية مطبَّقًا على $u$ نفسها، استعمل $\eu^{F}$
كما هناك).
\end{proof}

\begin{theorem}[صيغة بواسون؛ مسألة ديريكليه على القرص]
\label{thm:b3:conformal:poisson}
من أجل $0 \leq r < 1$ نعرّف \emph{نواة بواسون}\index{نواة
بواسون}
\[
P_r(\theta) = \sum_{n\in\Z}r^{\abs n}\eu^{\iu n\theta}
= \frac{1 - r^2}{1 - 2r\cos\theta + r^2} \;>\; 0 .
\]
ولتكن $g \colon \partial\mathbb D \to \R$ متصلة، ونضع، من أجل $z = r\eu^{\iu\varphi} \in \mathbb D$،
\[
u(z) = \frac1{2\pi}\int_0^{2\pi}
P_r(\varphi - t)\,g(\eu^{\iu t})\,\dd t .
\]
عندئذٍ تكون $u$ توافقية على $\mathbb D$ وتمتدّ اتصاليًّا إلى
$\bar{\mathbb D}$ بقيم حدّية $g$: وهي \omterm{prop:b3:conformal:harmonicholo}{الدالة التوافقية} الوحيدة
كذلك.\index{مسألة ديريكليه}
\end{theorem}

\begin{proof}
\emph{متطابقات النواة}: بجمع متسلسلتين هندسيتين،
\[
\sum_{n\in\Z}r^{\abs n}\eu^{\iu n\theta}
= \operatorname{Re}\frac{1 + r\eu^{\iu\theta}}{1 -
r\eu^{\iu\theta}}
= \frac{1 - r^2}{\abs{1 - r\eu^{\iu\theta}}^2},
\]
وهو خارج القسمة المعروض؛ والإيجابية واضحة، و $\frac1{2\pi}\int_0^{2\pi}P_r = 1$ (إذ لا
ينجو سوى $n = 0$).

\emph{التوافقية}: مع $z = r\eu^{\iu\varphi}$،
\[
u(z) = \operatorname{Re}\biggl[\frac1{2\pi}\int_0^{2\pi}
\frac{\eu^{\iu t} + z}{\eu^{\iu t} - z}\,
g(\eu^{\iu t})\,\dd t\biggr],
\]
(وللنواة بين القوسين الجزءُ الحقيقي $P_r(\varphi - t)$: بالحساب)،
والمقدار بين القوسين هولومورفي في $z$ على $\mathbb D$
(\cref{exo:b3:holomorphic:7}): ومنه تكون $u$ الجزءَ الحقيقي \omterm{def:b3:holomorphic:holo}{لدالة هولومورفية}،
أي توافقية.

\emph{القيم الحدّية}: تكون $P_r(\cdot)$ وحدةً تقريبية حين
$r \to 1^-$: بكتلة $1$، ومن أجل $\delta \leq
\abs\theta \leq \pi$، $P_r(\theta) \leq \frac{1 - r^2}{1 -
2r\cos\delta + r^2} \to 0$ بانتظام.
ويعطي الشطر المعياري (باتصال $g$ بجوار $\eu^{\iu\varphi_0}$، وبالحدّ في
غيرها) أن $u(r\eu^{\iu\varphi}) \to
g(\eu^{\iu\varphi_0})$ حين $r\eu^{\iu\varphi} \to
\eu^{\iu\varphi_0}$، بانتظام في النقطة الحدّية:
فيكون التمديد \omterm{def:b3:topology:continuity}{متصلًا}. \emph{والوحدانية}: فرق حلّين توافقي على
$\mathbb D$ ومتصل على \omterm{def:b3:topology:interior}{الغلق} ومعدوم على الحافة: ومنه ينعدم
بمبدأ الأعظمية (مطبَّقًا على $\pm$ الفرق).
\end{proof}

\begin{exercise}[$\star$]\label{exo:b3:conformal:1}

\begin{exercise}[$\star\star\star$]\label{exo:b3:conformal:9}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:10}
باستعمال صمود التوافقية بالتطبيقات المطابقة (إذ تكون $u\circ f$

u(x + \iu y) = \frac1\pi\,\arg(x + \iu y)
\qquad (\arg \in \intoo0\pi \text{ على} \mathbb H)

توافقية على $\mathbb H$ (بوصفها الجزء التخيّلي \omterm{def:b3:conformal:simplyconnected}{للوغاريتم

إلى القرص بكايلي لتستنبط \cref{exo:b3:conformal:8}(c) من جديد.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:11}
(النقاط الصامدة والتكرار في القرص) لتكن $f\colon\mathbb
D \to \mathbb D$ \omterm{def:b3:holomorphic:holo}{هولومورفية}.
(a) برهن على أنه إذا كانت للدالة $f$ نقطتان صامدتان
\emph{متمايزتان}، فإن $f = \mathrm{id}$ \emph{(بنقل إحداهما إلى $0$

(b) لنفترض أن $f(0) = 0$ وأن $f$ ليست دورانًا. برهن على أن
المكرَّرات $f^{\circ n} \to 0$ بانتظام على كل متراصة $\bar D(0, r)$ حيث
$r < 1$ \emph{(إذ تعطي شوارتز أن $\abs{f(z)}
\leq c_r\abs z$ على $\bar D(0,r)$ مع
$c_r < 1$ --- وبرّر هذا الثابت الأكيد بمبدأ الأعظمية مطبَّقًا

(c) وضّح ذلك على $f(z) = \frac{z^2 + z}2$: النقاط الصامدة، وسرعة تقارب مدار
$z_0 =
\frac12$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:conformal:12}

(b) برهن على أنه على \omterm{def:b3:topology:topology}{مجموعة مفتوحة} \emph{نجمية الشكل}، تقبل كل

(c) أعطِ المثال المضاد المعياري على $\C^*$: أن $u = \ln\abs z$

(\cref{ch:b3:forms}).
\end{exercise}

\section{مسألة: مبرهنة المساحة ومبرهنة كوبه للربع}

\begin{problem}[{مسألة نهاية الأسبوع --- كم يجب أن يغطي تطبيق
أحادي التكافؤ؟}]\label{pb:b3:conformal:1}
الدالة \emph{أحادية التكافؤ} هي غمرٌ هولومورفي. والدوال
الأحادية التكافؤ المعيَّرة على القرص،
\[
\mathcal S = \bigl\{f \in \mathcal H(\mathbb D) \text{ متباينة},\ f(z) = z + a_2z^2 + a_3z^3 + \cdots\bigr\},
\]
مقيَّدة تقييدًا صارمًا: فنبرهن على متراجحة بيبرباخ $\abs{a_2} \leq 2$
ونستنتج \emph{مبرهنة كوبه للربع}: أن صورة أي $f \in \mathcal S$ تحتوي
القرص $D(0, \frac14)$ --- أي الثابت الشامل المضبوط للهندسة
المطابقة.\index{مبرهنة كوبه للربع}\index{مبرهنة المساحة}

\medskip
\noindent\textbf{الجزء الأول --- مبرهنة المساحة.} لتكن $g(w) = w
+ b_0 + \frac{b_1}w + \frac{b_2}{w^2} + \cdots$
\omterm{def:b3:holomorphic:holo}{هولومورفية} و\emph{متباينة} على $\{\abs w > 1\}$.
\begin{enumerate}
  \item من أجل $\rho > 1$، لتكن $A_\rho$ مساحةَ (\omterm{def:b3:measure:lebesgueouter}{قياس لوبيغ})
        المجموعة المتراصة $K_\rho = \C\setminus
        g(\{\abs w > \rho\})$، أي المنطقة المحاطة
        بمنحني جوردان الناعم $g(C_\rho)$. وباستعمال صيغة
        المساحة في مبرهنة غرين--ريمان من مجلد السنة الجامعية
        2 --- أي إن المساحة المحاطة هي $\frac1{2\iu}
        \oint\bar\zeta\,\dd\zeta$ على امتداد
        الحافة الموجَّهة إيجابيًّا --- برهن على أن
        \[
        A_\rho = \frac{1}{2\iu}\int_{C_\rho}
        \overline{g(w)}\,g'(w)\,\dd w
        = \pi\Bigl(\rho^2 -
        \sum_{n\geq1}n\,\abs{b_n}^2\rho^{-2n}\Bigr) :
        \]
        عوّض متسلسلتَي لوران للمقدارين $\bar g$ و $g'$ على
        $C_\rho$ وكامل حدًّا حدًّا (بالتقارب الناظمي؛ ولا تنجو
        سوى الجداءات ذات التردد المعدوم).
  \item اجعل $\rho \to 1^+$ واخلص إلى \emph{مبرهنة المساحة}:
        \[
        \sum_{n\geq1}n\,\abs{b_n}^2 \;\leq\; 1 .
        \]
        وعلى وجه الخصوص $\abs{b_1} \leq 1$. ومتى يكون $\abs{b_1}
        = 1$؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- متراجحة بيبرباخ $\abs{a_2} \leq
2$.}
لتكن $f = z + a_2z^2 + \cdots \in \mathcal S$.
\begin{enumerate}[resume]
  \item برهن على أن $f(z^2)/z^2$ \omterm{def:b3:holomorphic:holo}{هولومورفية} وعديمة الجذور على
        $\mathbb D$، وتقبل جذرًا تربيعيًّا هولومورفيًّا
        $\varphi$ يحقق $\varphi(0) = 1$؛ وضع $h(z) =
        z\varphi(z^2)$، بحيث
        $h(z)^2 = f(z^2)$. وبرهن على أن $h$ دالةٌ \emph{فردية} أحادية
        التكافؤ على $\mathbb
        D$ بنشر $h(z) = z + \frac{a_2}2z^3 +
        \cdots$. \emph{(التباين:
        $h(z)^2 = h(z')^2$ يفرض $z^2 = z'^2$؛ ثم أنهِ بالفردية.)}
  \item طبّق مبرهنة المساحة على $g(w) = 1/h(1/w) = w -
        \frac{a_2}{2w} + \cdots$ على $\{\abs w > 1\}$
        (وتحقق من أحادية التكافؤ ومن النشر)، واخلص إلى $\abs{a_2} \leq 2$.
  \item برهن على أن \emph{دالة كوبه}
        \[
        k(z) = \frac{z}{(1 - z)^2} = \sum_{n\geq1}n\,z^n
        \]
        تنتمي إلى $\mathcal S$، وأن $a_2 = 2$، وأنها ترسل
        $\mathbb D$ على $\C\setminus\intoc{-\infty}{-\frac14}$ \emph{(اكتب $k = \frac14\bigl[\bigl(\frac{1+z}{1 -
        z}\bigr)^2 - 1\bigr]$ وتتبّع
        الصور)}: فتكون جميع المتراجحات الآتية مضبوطة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- مبرهنة الربع.}
\begin{enumerate}[resume]
  \item لتكن $f \in \mathcal S$ وليكن $c \notin f(\mathbb D)$. برهن على أن
        \[
        F(z) = \frac{c\,f(z)}{c - f(z)}
        \]
        تنتمي إلى $\mathcal S$، واحسب معاملها الثاني: $A_2 = a_2 + \frac1c$.
  \item طبّق بيبرباخ على كلٍّ من $f$ و $F$: واخلص إلى $\abs{\frac1c} \leq \abs{A_2} + \abs{a_2} \leq 4$،
        أي $\abs c \geq \frac14$. \emph{فكل قيمة متجاهَلة طويلتها
        $\geq \frac14$}: أي $f(\mathbb D) \supseteq
        D(0, \frac14)$ --- وهي مبرهنة كوبه
        للربع. وتحقق من الضبط على دالة كوبه.
  \item استنتج تقديرًا كمّيًّا لتطبيق ريمان: أنه إذا كان
        $\varphi \colon \Omega \to \mathbb D$ تطبيقَ ريمان في \cref{thm:b3:conformal:rmt} عند $z_0$، فإن
        \[
        \frac{d\bigl(z_0, \partial\Omega\bigr)}{4}
        \;\leq\; \frac1{\varphi'(z_0)} \;\leq\;
        4\,d\bigl(z_0, \partial\Omega\bigr)
        \]
        --- وبرهن على الأقل على المتراجحة اليسرى بتطبيق كوبه
        على $\varphi^{-1}$ معيَّرًا بكيفية مناسبة، وعلى اليمنى
        بشوارتز مطبَّقة على $\varphi$ على القرص $D(z_0, d) \subseteq \Omega$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- المنظور.}
\begin{enumerate}[resume]
  \item حدس بيبرباخ (1916) أن $\abs{a_n} \leq n$ من أجل كل $n$، مع
        التساوي من أجل دورانات دالة كوبه فقط؛ وبرهن دي برانج
        على ذلك عام 1985. تحقق من الحدس يدويًّا من أجل دالة
        كوبه ودوراناتها $\eu^{-\iu\theta}k(\eu^{\iu\theta}z)$. ثم ادفع نشر السؤال 4 حدًّا
        إضافيًّا: بكتابة $h(z) = z + \frac{a_2}2z^3 + c_5z^5 + \cdots$، برهن على $c_5 = \frac{a_3}2 - \frac{a_2^2}8$ وعلى
        \[
        g(w) = w - \frac{a_2}{2}\,w^{-1} +
        \Bigl(\frac{3a_2^2}8 - \frac{a_3}2\Bigr)w^{-3} +
        \cdots ,
        \]
        بحيث تعطي مبرهنة المساحة المتراجحةَ المحسَّنة $\bigl|\frac{a_2}2\bigr|^2 + 3\bigl|\frac{3a_2^2}8 -
        \frac{a_3}2\bigr|^2 \leq 1$.
        وتحقق منها على دالة كوبه ($a_2 = 2$، $a_3 = 3$).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- مبرهنة التشويه.} متراجحة
بيبرباخ، منقولةً حول القرص بالتشاكلات الذاتية، تتحكم في $f'$
في كل مكان. ونثبّت $f \in \mathcal
S$.
\begin{enumerate}[resume]
  \item (تحويل كوبه) من أجل $z_0 \in \mathbb D$ لتكن $\varphi(z) = \frac{z + z_0}{1 + \bar z_0z}$، وهي تشاكل
        ذاتي للقرص (\cref{thm:b3:conformal:autdisc}) مع $\varphi(0) = z_0$. برهن على أن
        \[
        F(z) = \frac{f(\varphi(z)) - f(z_0)}
        {f'(z_0)\,\bigl(1 - \abs{z_0}^2\bigr)}
        \]
        تنتمي إلى $\mathcal S$ \emph{(إذ تُورَث أحادية
        التكافؤ؛ واحسب $\varphi'(0) = 1 - \abs{z_0}^2$ وتحقق من التعيير؛ وتذكّر أن
        $f' \neq 0$ من أجل $f$ متباينة، \cref{def:b3:conformal:conformal})}.
  \item احسب المعامل الثاني $A_2 = \frac12F''(0)$ للدالة $F$:
        \[
        A_2 = \frac12\Bigl[\bigl(1 - \abs{z_0}^2\bigr)
        \frac{f''(z_0)}{f'(z_0)} - 2\bar z_0\Bigr] ,
        \]
        واستنتج من بيبرباخ (السؤال 4)، من أجل $z =
        r\eu^{\iu\theta}$،
        \emph{المتراجحة الأساسية}:
        \[
        \Bigl|\,z\,\frac{f''(z)}{f'(z)} -
        \frac{2r^2}{1 - r^2}\Bigr|
        \;\leq\; \frac{4r}{1 - r^2} .
        \]
  \item استخرج الجزء الحقيقي:
        \[
        \frac{2r^2 - 4r}{1 - r^2} \;\leq\;
        \operatorname{Re}\Bigl(z\,\frac{f''(z)}{f'(z)}\Bigr)
        \;\leq\; \frac{2r^2 + 4r}{1 - r^2} .
        \]
  \item برهن على أن $\frac{\dd}{\dd t}\log\bigl|
        f'(t\eu^{\iu\theta})\bigr| = \frac1t
        \operatorname{Re}\bigl(z\frac{f''(z)}{f'(z)}\bigr)$ عند $z = t\eu^{\iu\theta}$ \emph{(إذ من أجل دالة
        $g$ غير منعدمة من الصنف $\mathcal C^1$ لمتغيّر حقيقي،
        $\frac{\dd}{\dd t}\log\abs g =
        \operatorname{Re}(g'/g)$)}، وكامل حدود السؤال 12 على امتداد نصف
        المستقيم لتحصل على \emph{مبرهنة التشويه}:
        \[
        \frac{1 - r}{(1 + r)^3} \;\leq\; \abs{f'(z)}
        \;\leq\; \frac{1 + r}{(1 - r)^3},
        \qquad \abs z = r .
        \]
  \item استنتج \emph{مبرهنة النمو}:
        \[
        \frac{r}{(1 + r)^2} \;\leq\; \abs{f(z)} \;\leq\;
        \frac{r}{(1 - r)^2},
        \qquad \abs z = r
        \]
        \emph{(الحدّ الأعلى: بمكاملة $f'$ على القطعة
        $[0, z]$؛ والحدّ الأدنى: إذا كان $\abs{f(z)} < \frac14$، فإن القطعة
        $[0, f(z)]$ تقع في $f(\mathbb D)$ حسب السؤال 7؛
        فأرجعها بواسطة $f^{-1}$ --- وهو هولومورفي حسب
        \cref{cor:b3:residues:openmapping} --- وحُدّ $\abs{f(z)} = \int_\gamma\abs{f'(\zeta)}
        \,\abs{\dd\zeta} \geq \int_0^r\frac{1 - t}{(1 +
        t)^3}\,\dd t$ باستعمال $\abs{\dd\zeta} \geq
        \dd\abs\zeta$)}.
  \item تحقق من أن دالة كوبه تحقق التساوي في الحدود الأربعة
        كلها، عند $z = r$ من أجل العلوية وعند $z = -r$ من أجل
        السفلية: فتكون $k$ في آن واحد أكثر أعضاء
        $\mathcal S$ توسيعًا، وعند النقطة المقابلة، أكثرها
        تقليصًا.
\end{enumerate}

\medskip
\noindent\textbf{الجزء السادس --- الصرامة الأقصوية.} في كل
متراجحة حتى الآن، يعيّن التساوي دالةَ كوبه إلى غاية دوران.
ونبرهن على ذلك، ثم نحصد.
\begin{enumerate}[resume]
  \item لنفترض أن $f \in \mathcal S$ تحقق $\abs{a_2} = 2$. تتبّع التساوي
        عبر الأسئلة 2--4: فتفرض مبرهنة المساحة أن $g(w) = w + b_0 + \eu^{\iu\alpha}/w$؛
        وتجعل فردية $h$ الدالةَ $g$ فردية، ومنه $b_0 = 0$؛
        ثم اقلب لتجد $h$، ثم $f$، واخلص إلى أن
        \[
        f(z) = \eu^{-\iu\theta}k\bigl(\eu^{\iu\theta}z\bigr)
        \quad\text{حيث} \eu^{\iu\theta} =
        -\eu^{\iu\alpha} :
        \]
        فتكون دورانات دالة كوبه هي أعضاء $\mathcal S$ الوحيدة
        التي تحقق $\abs{a_2} = 2$.
  \item برهن على أنه إذا تجاهلت $f \in \mathcal S$ قيمةً $c$ تحقق $\abs c = \frac14$
        بالضبط، فإن $f$ دورانٌ لدالة كوبه، وعيّن $c =
        -\eu^{-\iu\theta}/4$
        \emph{(بتتبّع التساوي عبر سلسلة السؤال 7 $4 = \abs{1/c} = \abs{A_2 - a_2}
        \leq \abs{A_2} + \abs{a_2} \leq 4$)}: فلا
        يبلغ ثابتَ مبرهنة الربع إلا العائلة الأقصوية.
  \item (المعاملات بثمن بخس) اجمع مبرهنة النمو مع \omterm{thm:b3:holomorphic:analytic}{تقديرات كوشي}
        (\cref{thm:b3:holomorphic:analytic}) على الدائرة $\abs z = 1 - \frac1n$ لتبرهن على
        \[
        \abs{a_n} \;\leq\; \eu\,n^2 \qquad (n \geq 2) .
        \]
        (ودي برانج، 1985: $\abs{a_n} \leq n$؛ والعامل $\eu n$ هو ثمن
        الأدوات الابتدائية.)
  \item (تغطية الأقراص الجزئية) برهن على أنه من أجل كل $0 < r <
        1$،
        \[
        f\bigl(D(0, r)\bigr) \supseteq
        D\Bigl(0, \frac{r}{(1 + r)^2}\Bigr),
        \]
        وهي مضبوطة من أجل دالة كوبه، واستعد مبرهنة الربع حين
        $r \to 1^-$. \emph{(نقاط حافة الصورة المفتوحة
        $f(D(0,r))$ تقع على $f(\partial D(0,r))$، ومنه طويلتها $\geq
        \frac{r}{(1+r)^2}$ حسب
        السؤال 14؛ وقطعةٌ من $0$ إلى نقطة متجاهَلة أصغر طويلةً
        كانت ستضطر إلى قطع تلك الحافة.)}
  \item (كوبه عند كل نقطة) لتكن $f$ أحادية التكافؤ على
        $\mathbb D$، وليست بالضرورة معيَّرة، ولتكن $z_0
        \in \mathbb D$.
        برهن على
        \[
        \tfrac14\bigl(1 - \abs{z_0}^2\bigr)\abs{f'(z_0)}
        \;\leq\;
        d\bigl(f(z_0), \partial f(\mathbb D)\bigr)
        \;\leq\;
        \bigl(1 - \abs{z_0}^2\bigr)\abs{f'(z_0)}
        \]
        \emph{(اليسرى: بمبرهنة الربع مطبَّقة على تحويل كوبه في
        السؤال 10؛ واليمنى: بشوارتز (\cref{thm:b3:conformal:schwarz}) مطبَّقة على
        $\psi^{-1}\circ\hat g$، حيث $\hat g(w) =
        f^{-1}\bigl(f(z_0) + dw\bigr)$ و $d$ المسافة و $\psi$ تشاكل ذاتي
        للقرص يرسل $0$ إلى $z_0$)}. ولماذا تكون $\partial f(\mathbb D)$ غير
        خالية؟
  \item تحقق من السؤال 20 على $f = k$ عند $z_0 = r \in
        \intoo01$: احسب $d\bigl(k(r), \partial k(\mathbb
        D)\bigr) = \frac{(1+r)^2}{4(1-r)^2}$
        وتحقق من أن المتراجحة اليسرى \emph{تساوٍ}: فتُشبع دالة
        كوبه مبرهنتها الخاصة عند كل نقطة من $\intoo01$.
  \item (العبرة) في عشرة أسطر: ما المبدأ الواحد الذي تقوم عليه
        مبرهنة المساحة، وكيف تتدفق منه بيبرباخ، ومبرهنة الربع،
        والتشويه، والنمو، والتغطية جميعًا؟ وقارن ذلك بعالم
        شوارتز--بيك في \cref{thm:b3:conformal:schwarz}: ففي كليهما، تصلّب متراجحةٌ
        داخلية واحدة الهندسةَ كلها، وتكون الأقصويات وحيدة إلى
        غاية دوران.

\end{enumerate}
\medskip
\noindent\textbf{الجزء السابع --- التراص والمقلوبات وفحص
واقعي.}
\begin{enumerate}[resume]
  \item برهن على أن الصنف $\mathcal S$ \emph{متراص} بالنسبة
        إلى التقارب المنتظم محليًّا: فهو محدود محليًّا بمبرهنة
        النمو، ومنه ناظمي (\cref{thm:b3:conformal:montel})؛ وكل نهاية منتظمة
        محليًّا لأعضاء $\mathcal S$ تنتمي بدورها إلى
        $\mathcal S$ \emph{(إذ تمرّ التعييرات إلى النهاية
        بتقارب فايرشتراس للمشتقات؛ ويبقى التباين بهورفيتز،
        \cref{exo:b3:residues:8}، لأن النهاية غير ثابتة)}. ولماذا يهمّ هذا
        في مسائل أقصوية مثل مسألة بيبرباخ؟
  \item من أجل $f \in \mathcal S$، ليكن $g = f^{-1}$ معرَّفًا بجوار $0$.
        برهن على $g(w) = w - a_2w^2 + O(w^3)$، ومنه يخضع المعامل الثاني للمقلوب
        للحدّ المضبوط نفسه $\abs{A_2} = \abs{a_2} \leq 2$، مع التساوي من أجل دوال
        كوبه المدارة بالضبط.
  \item عيّن من أجل أي $a \in \C$ ينتمي كثير الحدود $f(z)
        = z + az^2$
        إلى $\mathcal S$: برهن على أن $f$ متباينة على
        $\mathbb D$ إذا وفقط إذا كان $\abs a \leq \frac12$ \emph{(بتحليل
        $f(z_1) - f(z_2)$)}. واخلص: أن حدّ المعاملات الحقيقي من أجل
        كثيرات الحدود من الدرجة الثانية هو $\frac12$، أي أصغر
        بأربع مرات من حدّ بيبرباخ $2$ --- فأقصويات
        $\mathcal S$ كائنات متسامية فعلًا، ولا يقاربها أي كثير
        حدود.
\end{enumerate}
\end{problem}
