\chapter{المعادلات التفاضلية العادية}\label{ch:b3:ode}

حلّت السنة الجامعية 2 المعادلات التفاضلية الخطية وصاغت مبرهنة
كوشي--ليبشيتز؛ ويبرهن هذا الفصل عليها --- مرتين: الوجود
والوحدانية بنقطة باناخ الصامدة، والبنية الشاملة بنظرية الحلول
الأعظمية ومبرهنة \omterm{thm:b3:ode:maximal}{الهروب من المتراصات}. ثم يُعاد بناء النظرية
الخطية على أسس أمينة (\omterm{thm:b3:ode:linearstructure}{المُحلِّل}، \omterm{thm:b3:ode:linearstructure}{والفرونسكياني}، \omterm{thm:b3:ode:matrixexp}{والأُسّي
المصفوفي}، ودوهاميل)، ويفتح النصف الثاني من الفصل النظرية
\emph{النوعية} --- الانسيابات، ومواضع التوازن، ودوال ليابونوف،
\omterm{thm:b3:ode:linearization}{والاستقرار بالخطّية}: أي كيف نفهم حلولًا لن نحسبها أبدًا.
والنواس، في مسألة نهاية الأسبوع، هو دراسة الحالة الأبدية. وفي
كل ما يلي، تكون $U \subseteq \R\times\R^d$ مفتوحة وتكون $f \colon U \to \R^d$ متصلة؛ و\emph{الحل}
للمعادلة $x' = f(t, x)$ هو تطبيق $x \colon I \to \R^d$ من الصنف $\mathcal C^1$
(حيث $I$ فترة) منحنيه البياني في $U$ ويحقق المعادلة.

\section{كوشي--ليبشيتز}

\begin{definition}\label{def:b3:ode:lipschitz}
تكون $f$ \emph{ليبشيتزية محليًّا في $x$}\index{ليبشيتزية
محليًّا} إذا كان لكل نقطة من $U$ جوارٌ $V$ وثابتٌ $L$ يحققان
$\norm{f(t, x_1) - f(t, x_2)} \leq L\norm{x_1 -
x_2}$ من أجل $(t, x_1), (t, x_2) \in V$. وإذا كانت $f$ من الصنف $\mathcal
C^1$ (أو مجرّد وجود
$\partial_xf$ واتصالها)، كانت ليبشيتزية محليًّا في $x$: أي على
\omterm{def:b3:topology:topology}{جوار} محدّب متراص، بمتراجحة القيم المتوسطة مع $L =
\sup\vertiii{\partial_xf}$.
\end{definition}

\begin{theorem}[كوشي--ليبشيتز، الصيغة المحلية]
\label{thm:b3:ode:cauchylipschitz}
لتكن $f$ متصلة \omterm{def:b3:ode:lipschitz}{وليبشيتزية محليًّا} في $x$، ولتكن $(t_0,
x_0) \in U$. يوجد
$T > 0$ بحيث يكون لمسألة كوشي
\[
x' = f(t, x), \qquad x(t_0) = x_0
\]
حلٌّ واحد بالضبط على $[t_0 - T, t_0 +
T]$.\index{مبرهنة كوشي--ليبشيتز}
\end{theorem}

\begin{proof}
نختار $a, b > 0$ بحيث $Q = [t_0 - a, t_0 + a]\times\bar
B(x_0, b) \subseteq U$، ويكون عليه $\norm f \leq M$
وتكون $f$ ليبشيتزية بالثابت $L$ في $x$. والدالة من الصنف
$\mathcal C^1$ حلٌّ إذا وفقط إذا حققت المعادلة التكاملية
\[
x(t) = x_0 + \int_{t_0}^{t}f\bigl(s, x(s)\bigr)\,\dd s
\]
(بالمبرهنة الأساسية في التفاضل والتكامل، بالاتجاهين). ولتكن
$T =
\min\bigl(a, \frac bM, \frac1{2L}\bigr)$ و $I = [t_0 - T, t_0
+ T]$ و
\[
\mathcal E = \{x \in \mathcal C(I, \R^d) : \norm{x(t) - x_0}
\leq b\ \text{على} I\},
\]
وهي جزء مغلق من فضاء باناخ $(\mathcal C(I, \R^d),
\norm\cdot_\infty)$: أي تام (\cref{def:b3:complete:complete}). ونعرّف
$\Phi(x)(t) = x_0 +
\int_{t_0}^tf(s, x(s))\dd s$: فمن أجل $x \in \mathcal E$،
$\norm{\Phi(x)(t) - x_0} \leq M\abs{t - t_0} \leq MT \leq b$
--- ومنه يرسل $\Phi$ المجموعةَ $\mathcal E$ في نفسها --- ومن
أجل $x, y \in
\mathcal E$:
\[
\norm{\Phi(x)(t) - \Phi(y)(t)} \leq \Bigl|\int_{t_0}^t
L\,\norm{x(s) - y(s)}\,\dd s\Bigr| \leq LT\,\norm{x -
y}_\infty \leq \tfrac12\norm{x - y}_\infty :
\]
أي تقلّص. وتعطي نقطة باناخ الصامدة (\cref{thm:b3:complete:banach}) نقطةً صامدة
وحيدة في $\mathcal E$: أي الوجود، والوحدانية بين الحلول التي
تبقى في $\bar B(x_0, b)$ --- لكن كل حل على $I$ يبقى هناك
($\norm{x(t) - x_0} \leq M\abs{t - t_0} \leq b$ ما دام المنحني البياني في $Q$، بحجة اتصال): أي
الوحدانية على $I$.
\end{proof}

\begin{lemma}[غرونوال]\label{lem:b3:ode:gronwall}
لتكن $u \colon I \to \intco0\infty$ متصلة، ولتكن $t_0 \in
I$، ولنفترض أن\index{المبرهنة
المساعدة لغرونوال}
\[
u(t) \leq \alpha + \beta\,\Bigl|\int_{t_0}^{t}u(s)\,\dd
s\Bigr| \qquad (t \in I)
\]
مع $\alpha \geq 0$ و $\beta > 0$. عندئذٍ $u(t) \leq
\alpha\,\eu^{\beta\abs{t - t_0}}$ على $I$.
\end{lemma}

\begin{proof}
من أجل $t \geq t_0$: لتكن $v(t) = \alpha +
\beta\int_{t_0}^tu(s)\dd s$، ومنه $u \leq v$ و $v' = \beta u
\leq \beta v$
و $(v\eu^{-\beta(t - t_0)})' \leq 0$: أي $v(t)
\leq v(t_0)\eu^{\beta(t-t_0)} = \alpha\eu^{\beta(t - t_0)}$. ومن أجل $t \leq t_0$ نطبّق الشيء نفسه على
$\tilde u(t) = u(2t_0 -
t)$.
\end{proof}

\begin{corollary}[الوحدانية والتعلّق الاتصالي]
\label{cor:b3:ode:uniqueness}
تحت فرضيات \cref{thm:b3:ode:cauchylipschitz}، يتوافق حلّان للمعادلة $x' = f(t,x)$
يتوافقان عند نقطة واحدة على فترة تعريفهما المشتركة. وكمّيًّا،
إذا كان $x, y$ حلّين منحنياهما البيانيان في منطقة تكون فيها
$f$ ليبشيتزية بالثابت $L$ في $x$، فإن
\[
\norm{x(t) - y(t)} \leq \norm{x(t_0) -
y(t_0)}\,\eu^{L\abs{t - t_0}} .
\]
\end{corollary}

\begin{proof}
التقدير: يحقق $u = \norm{x - y}$ أن $u(t) \leq u(t_0)
+ L\abs{\int_{t_0}^tu}$ (بطرح المعادلتين التكامليتين)؛
ثم غرونوال. وأما الوحدانية الشاملة: فمجموعة التوافق $\{t : x(t)
= y(t)\}$
مغلقة في الفترة المشتركة، وغير خالية، ومفتوحة --- إذ حول أي نقطة
توافق، نغطّي قطعةً متراصة من المنحني البياني المشترك بعدد منتهٍ
من صناديق ليبشيتز ونطبّق التقدير مع $u(t_1) = 0$ على كلٍّ منها:
فيكون $x \equiv y$ محليًّا. وجزءٌ غير خالٍ مفتوح ومغلق من فترة
هو الفترة كلها.
\end{proof}

\section{الحلول الأعظمية}

\begin{theorem}[الحلول الأعظمية؛ الهروب من المتراصات]
\label{thm:b3:ode:maximal}
لنفترض أن $f$ متصلة \omterm{def:b3:ode:lipschitz}{وليبشيتزية محليًّا} في $x$.
\begin{enumerate}
  \item لكل مسألة كوشي حلٌّ \emph{أعظمي} وحيد $x \colon \intoo{T_-}{T_+} \to \R^d$: وكل حل
        آخر يمرّ بالنقطة $(t_0, x_0)$ هو تضييقٌ له. والفترة
        مفتوحة.
  \item (\emph{الهروب}) من أجل كل متراصة $K \subseteq U$ يوجد
        $\varepsilon > 0$ بحيث $(t, x(t)) \notin
        K$ من أجل كل $t \in \intoo{T_+ - \varepsilon}{T_+}$ (وبالتناظر عند
        $T_-$): أي إن \emph{المنحني البياني \omterm{thm:b3:ode:maximal}{لحل أعظمي} يغادر
        في النهاية كل جزء متراص من $U$}. وعلى وجه الخصوص، من
        أجل $U = \R\times\R^d$ و $T_+ < +\infty$: $\norm{x(t)} \to +\infty$ حين
        $t \to T_+^-$ (أي \emph{الانفجار}).\index{حل
        أعظمي}\index{الهروب من المتراصات}\index{انفجار}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) لتكن $\mathcal S$ مجموعةَ جميع الحلول المارّة بالنقطة
$(t_0, x_0)$؛ فحسب \cref{cor:b3:ode:uniqueness} يتوافق أي اثنين منها على تقاطع
فترتيهما، ومنه تلتحم: فعلى $J =
\bigcup_{y \in \mathcal S}I_y$، نعرّف $x(t) = y(t)$ من أجل
أي $y$ معرَّف عند $t$: وهو حلٌّ معرَّف جيدًا، وأعظمي ووحيد
بداهةً. و $J$ مفتوح: إذ حلٌّ معرَّف عند طرف كان سيُمدَّد
بواسطة \cref{thm:b3:ode:cauchylipschitz} عند ذلك الطرف.

(2) لنفترض أن الادعاء يخفق عند $T_+$: فتوجد $t_n \to
T_+$ تحقق $(t_n, x(t_n)) \in K$؛
ولاحظ أن هذا يفرض $T_+ <
\infty$، أو، إذا كان $T_+ = \infty$، فلا شيء
نبرهن عليه ($K$ محدودة زمنيًّا). فليكن $T_+ < \infty$. وأما
التراص: فالثوابت المنتظمة $M, L, a,
b$ تصلح من أجل جميع معطيات كوشي
في جوار للمتراصة $K$ --- وعمليًّا، نغطي $K$ بعدد منتهٍ من
الصناديق $Q_i$ كما في برهان المبرهنة المحلية وليكن $T^* > 0$
أصغرَ أزمنة الوجود الموافقة: فكل معطى كوشي في $K$ يطلق حلًّا
يعيش $T^*$ على الأقل بعد زمنه الابتدائي. وبتطبيق ذلك عند
$(t_n, x(t_n))$ مع $t_n > T_+ -
T^*/2$ يمتدّ $x$ وراء $T_+$ (ويتوافق التمديد
مع $x$ بالوحدانية، ثم يمدّده): وهو تناقض مع الأعظمية. ومنه
يغادر المنحني البياني $K$ نهائيًّا قبل $T_+$. ومن أجل $U = \R\times\R^d$:
إذا كان $\norm{x(t)}\not\to\infty$، أبقت متتالية $t_n \to T_+$ النقاطَ
$(t_n, x(t_n))$ في المتراصة $[t_0, T_+]\times\bar B(0, R)$: وهو مستبعد.
\end{proof}

\begin{corollary}[الوجود الشامل تحت النمو الخطي]
\label{cor:b3:ode:globallinear}
إذا كان $U = I\times\R^d$ ($I$ فترة مفتوحة) و $\norm{f(t, x)}
\leq \alpha(t)\norm x + \beta(t)$ مع $\alpha, \beta$
متصلتين، فإن كل \omterm{thm:b3:ode:maximal}{حل أعظمي} معرَّف على $I$ كله.
\end{corollary}

\begin{proof}
على متراصة $[t_0, T] \subseteq I$: $\norm{x(t)} \leq
\norm{x_0} + \int_{t_0}^t(\alpha\norm x + \beta)$، ومنه بغرونوال (مع $\alpha, \beta$
محدودتين بالعددين $A, B$ هناك) $\norm{x(t)} \leq (\norm{x_0} + B(T - t_0))\eu^{A(T - t_0)}$: أي محدود. وإذا كان
$T_+ < \sup I$، بقي المنحني البياني في متراصة من $I\times\R^d$
بجوار $T_+$: وهو يناقض الهروب (\cref{thm:b3:ode:maximal}).
\end{proof}

\section{الجمل الخطية}

في هذا المقطع تكون $A \colon I \to M_d(\R)$ و $b
\colon I \to \R^d$ متصلتين؛ والجملة هي $x' = A(t)x
+ b(t)$
--- أي نموّ خطي: فجميع الحلول الأعظمية تعيش على $I$ كله
(\cref{cor:b3:ode:globallinear}).

\begin{theorem}[البنية]\label{thm:b3:ode:linearstructure}
تشكّل حلول الجملة المتجانسة $x' = A(t)x$ فضاءً متجهيًّا $S_H$
بُعده $d$؛ ومن أجل كل $t_0$، يكون التقييم $x \mapsto x(t_0)$ تماثلًا $S_H \to
\R^d$.
و\emph{المُحلِّل}\index{محلِّل جملة خطية} $R(t, s) \in GL_d(\R)$، المعرَّف
بالشرط: أن $t \mapsto R(t,
s)v$ هو الحل الذي قيمته $v$ عند $s$، يحقق
\[
R(s,s) = I,\quad R(t, u)R(u, s) = R(t, s),\quad
\partial_tR(t,s) = A(t)R(t,s),
\]
وتُحلّ المسألة غير المتجانسة بواسطة \emph{صيغة
دوهاميل}\index{صيغة دوهاميل}:
\[
x(t) = R(t, t_0)\,x_0 + \int_{t_0}^{t}R(t, s)\,b(s)\,\dd s .
\]
وأخيرًا يخضع \emph{الفرونسكياني}\index{فرونسكياني} $w(t) = \det
R(t, s)$ لصيغة
ليوفيل $w' =
\operatorname{tr}A(t)\,w$، ومنه $w(t) =
\exp\bigl(\int_s^t\operatorname{tr}A\bigr) > 0$.
\end{theorem}

\begin{proof}
تجعل خطية المعادلة الحلولَ فضاءً متجهيًّا؛ والتقييم خطي،
ومتباين (بالوحدانية: إذ حلٌّ ينعدم عند $t_0$ هو $\equiv 0$)،
وغامر (بالوجود): فالبُعد $d$. وخصائص \omterm{thm:b3:ode:linearstructure}{المُحلِّل} تعيد صياغة
الوحدانية (إذ يحل طرفا كل متطابقة مسألةَ كوشي نفسها)؛ والقابلية
للقلب من $R(s,t)R(t,s) = I$. وأما دوهاميل: فنشتقّ الصيغة --- $x'(t) = A(t)R(t,t_0)x_0 + R(t,t)b(t) +
\int_{t_0}^tA(t)R(t,s)b(s)\dd s = A(t)x(t) + b(t)$ (والاشتقاق
تحت علامة التكامل مشروع: إذ المقدار المكامَل من الصنف
$\mathcal C^1$ في $t$ بمشتقة متصلة في $(t,s)$؛ أو نتحقق
بالمعادلة التكاملية). وأما ليوفيل: $w(t + h) = \det\bigl(R(t+h, t)\bigr)w(t)$ و $R(t + h, t) =
I + hA(t) + o(h)$ (من المعادلة
التكاملية)، ومنه $\det = 1
+ h\operatorname{tr}A(t) + o(h)$ (بنشر $\det$ عند $I$): أي $w'(t) = \operatorname{tr}A(t)\,w(t)$؛ ثم
نكامل المعادلة التفاضلية الخطية السلّمية.
\end{proof}

\begin{theorem}[الأُسّي المصفوفي]\label{thm:b3:ode:matrixexp}
من أجل $A \in M_d(\C)$، تتقارب المتسلسلة $\eu^{A} =
\sum_{n\geq0}\frac{A^n}{n!}$ (تقاربًا
مطلقًا، بأي معيار شبه ضربي)، ويكون $\eu^{A+B} = \eu^A\eu^B$ كلما كان $AB = BA$،
ويكون $t \mapsto \eu^{tA}$ مُحلِّل الجملة الثابتة: $R(t,s) = \eu^{(t-s)A}$؛ وهو من الصنف
$\mathcal
C^\infty$ مع $\frac{\dd}{\dd t}\eu^{tA} = A\eu^{tA}$. وفوق ذلك: إذا كان $\operatorname{Re}\lambda < -\alpha < 0$ من أجل كل قيمة ذاتية
$\lambda$ للمصفوفة $A$، فإن $\vertiii{\eu^{tA}}
\leq C\eu^{-\alpha t}$ من أجل $t \geq 0$.\index{أُسّي
مصفوفي}
\end{theorem}

\begin{proof}
التقارب: $\vertiii{A^n/n!} \leq \vertiii A^n/n!$، وهي قابلة للجمع (\cref{exo:b3:complete:1}(b) في جبر باناخ
$M_d$). ومن أجل $A, B$ متبادلتين: يُعاد ترتيب جداء كوشي
للمتسلسلتين المتقاربتين تقاربًا مطلقًا، عبر مبرهنة ذات الحدين
(الصحيحة حين $AB = BA$)، إلى $\sum_n\frac{(A+B)^n}{n!}$. وأما القابلية للاشتقاق
فمباشرةً: $\frac{\eu^{(t+h)A} - \eu^{tA}}h = \eu^{tA}\frac{\eu^{hA} -
I}{h} \to \eu^{tA}A$ لأن $\norm{\frac{\eu^{hA} - I}h - A}
\leq \sum_{n\geq2}\frac{\abs h^{n-1}\vertiii A^n}{n!} =
O(h)$. ومنه يحل $t \mapsto \eu^{(t - s)A}v$ مسألةَ كوشي
المعرِّفة للمقدار $R(t,s)v$. وأما الحدّ الطيفي: فبصيغة جوردان
(\cref{thm:b3:modules:jordan})، $A = P(D + N)P^{-1}$ حيث $D$ قطرية تحمل القيم الذاتية،
و $N$ معدومة القوة، و $DN =
ND$. عندئذٍ $\eu^{tA} = P\,\eu^{tD}\eu^{tN}P^{-1}$ مع $\vertiii{\eu^{tD}} \leq \eu^{-(\alpha + \delta)t}$ من أجل
$\delta > 0$ ما (حيث $t \geq 0$)، ويكون $\eu^{tN}$ كثير حدود
في $t$ (بانعدام القوة): فيكون الجداء $\leq C\eu^{-\alpha t}$ (إذ يغلب $\eu^{-\delta t}$
كثيرَ الحدود).
\end{proof}

\begin{example}[المستوي، مصنَّفًا]
\label{ex:b3:ode:planeclassification}
من أجل $x' = Ax$ مع $A \in M_2(\R)$ قابلة للقلب، تتحدّد صورة
الطور بجوار $0$ بالعددين $\tau = \operatorname{tr}A$ و $\delta = \det A$، عبر القيمتين
الذاتيتين $\lambda_\pm
= \frac{\tau \pm \sqrt{\tau^2 - 4\delta}}2$:
\begin{itemize}
  \item $\delta < 0$: قيمتان ذاتيتان حقيقيتان مختلفتا الإشارة
        --- أي \emph{سرج}؛ فيدخل مساران ويخرج مساران، وتمرّ
        البقية مرورًا. وهو غير مستقر دائمًا.
  \item $\delta > 0$ و $\tau^2 \geq 4\delta$: قيمتان ذاتيتان حقيقيتان
        بالإشارة نفسها ($= \operatorname{sign}\tau$) --- أي \emph{عقدة}، مستقرة
        إذا وفقط إذا كان $\tau < 0$؛ والمسارات مماسّة للاتجاه
        الذاتي البطيء.
  \item $\delta > 0$ و $\tau^2 < 4\delta$ و $\tau \neq 0$: قيمتان ذاتيتان
        عقديتان مرافقتان $\frac\tau2 \pm
        \iu\omega$ --- أي \emph{حلزون} (بؤرة)،
        مستقر إذا وفقط إذا كان $\tau < 0$؛ والحلول دورانات
        $\eu^{\tau t/2}\times$ بالدور $\frac{2\pi}\omega$.
  \item $\tau = 0$ و $\delta > 0$: قيمتان ذاتيتان تخيّليتان
        محضتان --- أي \emph{مركز}: بمدارات مغلقة (قطوع
        ناقصة)، واستقرارٌ دون استقرار مقارب، وهو بالضبط الحدّ
        الفاصل الذي لا تستطيع \cref{thm:b3:ode:linearization} البتّ فيه من أجل
        الجمل غير الخطية (وهنا يجلس موضع توازن النواس السفلي،
        \cref{pb:b3:ode:1}).
\end{itemize}
ويحمل القطع المكافئ الحدّي $\tau^2 = 4\delta$ العقدَ المنحلّة (أي كتل
جوردان: بمسارات ذات اتجاه مماس واحد). وكل شيء يُقرأ من عددين
--- ولهذا يكون أول ردّ فعل قبل أي صورة طور مستوية هو حساب
$\operatorname{tr}$ و $\det$؛ فمثلًا $A = \bigl(\begin{smallmatrix}0 & 1\\ -1 &
-c\end{smallmatrix}\bigr)$ (المذبذب المخمَّد): $\delta = 1 >
0$
و $\tau = -c$: أي حلزون مستقر من أجل $0 < c < 2$، وعقدة مستقرة
من أجل $c \geq 2$ --- أي التخميد الناقص مقابل التخميد الزائد،
في لمحة واحدة.
\end{example}

\section{الانسيابات ومواضع التوازن والاستقرار}

ننظر الآن في المعادلة \emph{الذاتية} $x' = F(x)$، حيث $F
\colon \Omega \to \R^d$
\omterm{def:b3:ode:lipschitz}{ليبشيتزية محليًّا} على المفتوحة $\Omega
\subseteq \R^d$. ونكتب $\varphi_t(x_0) = x(t)$ \omterm{thm:b3:ode:maximal}{للحل
الأعظمي} الذي يحقق $x(0) = x_0$ (أي \emph{انسياب} الحقل
المتجهي)\index{انسياب حقل متجهي}؛ وتعطي الذاتية خاصية الزمرة
$\varphi_{t+s} =
\varphi_t\circ\varphi_s$ حيثما عُرِّفت (إذ يحل الطرفان المسألة نفسها عند الزمن
$s$).

\begin{definition}\label{def:b3:ode:stability}
\emph{موضع التوازن} نقطةٌ $\bar x$ تحقق $F(\bar x) =
0$ (ومنه $\varphi_t(\bar x) = \bar x$).
ويكون \emph{مستقرًّا}\index{استقرار (ليابونوف)} إذا وُجد من أجل
كل $\varepsilon > 0$ عددٌ $\delta > 0$ بحيث يستلزم $\norm{x_0 - \bar x} < \delta$ وجود الحل من
أجل كل $t \geq 0$ مع $\norm{\varphi_t(x_0) - \bar
x} < \varepsilon$؛ ويكون \emph{مستقرًّا مقاربًا} إذا
كان فوق ذلك $\varphi_t(x_0) \to \bar x$ من أجل كل $x_0$ بجوار $\bar x$.
\end{definition}

\begin{theorem}[دوال ليابونوف]\label{thm:b3:ode:lyapunov}
ليكن $\bar x$ موضع توازن ولتكن $V \colon \mathcal V \to
\R$ دالةً من الصنف
$\mathcal C^1$ على \omterm{def:b3:topology:topology}{جوار} للنقطة $\bar x$
تحقق:\index{دالة ليابونوف}
\[
V(\bar x) = 0,\qquad V(x) > 0 \text{ من أجل} x \neq \bar x,
\qquad \dot V(x) = \nabla V(x)\cdot F(x) \leq 0 .
\]
عندئذٍ يكون $\bar x$ \omterm{def:b3:ode:stability}{مستقرًّا}. وإذا كان فوق ذلك $\dot V < 0$
خارج $\bar
x$، كان $\bar x$ \omterm{def:b3:ode:stability}{مستقرًّا} مقاربًا.
\end{theorem}

\begin{proof}
على امتداد حل، $\frac{\dd}{\dd t}V(x(t)) = \dot V(x(t))
\leq 0$: أي إن $V$ متناقصة. وبإعطاء $\varepsilon$
(صغير بما يكفي بحيث $\bar B(\bar x, \varepsilon) \subseteq \mathcal V$)، لتكن
$m = \min\{V(x) : \norm{x - \bar x} = \varepsilon\} > 0$
(بالتراص والإيجابية) ونختار $\delta < \varepsilon$ بحيث $V < m$ على $B(\bar x, \delta)$
(بالاتصال). فحلٌّ ينطلق من $B(\bar x, \delta)$ يحقق $V(x(t)) < m$ من أجل كل
الأزمنة اللاحقة، ومنه لا يمكنه أن يبلغ الكرة $\norm{x -
\bar x} = \varepsilon$ (حيث
$V \geq m$): فيبقى في الكرة --- وعندئذٍ يوجد من أجل كل
$t \geq 0$: إذ يبقى الحل في المتراصة $\bar B$، ومنه تفرض
\cref{thm:b3:ode:maximal}(2) (\omterm{thm:b3:ode:maximal}{الهروب من المتراصات}) أن $T_+ = +\infty$. وهذا
هو الاستقرار.

وأما الحالة المقاربة: فليكن $x(t)$ منطلقًا من $B(\bar x, \delta)$؛ تتناقص
$V(x(t))$ إلى $c \geq 0$ ما. فإذا كان $c > 0$: بقي المسار في
$K = \{x \in \bar B(\bar x,
\varepsilon): V(x) \geq c\}$، وهي مجموعة متراصة تستبعد جوارًا للنقطة $\bar x$ (إذ
$V$ متصلة مع $V(\bar x)
= 0 < c$). وعلى $K$ تكون الدالة $\dot V$ متصلة
وسالبة تمامًا، ومنه $\mu = \max_K\dot V < 0$ (بالتراص)؛ عندئذٍ $V(x(t)) \leq V(x(0)) + \mu t \to
-\infty$: وهو محال
لأن $V \geq 0$. ومنه $c = 0$، و $x(t) \to \bar x$ (إذ للنقاط التي تبعد
$\geq
\rho$ عن $\bar x$ داخل الكرة $V \geq m_\rho >
0$).
\end{proof}

\begin{theorem}[الاستقرار بالخطّية]
\label{thm:b3:ode:linearization}
لتكن $F$ من الصنف $\mathcal C^1$ مع $F(\bar x) = 0$ و
$A = DF(\bar x)$. إذا كان لكل قيمة ذاتية للمصفوفة $A$ الشرطُ
$\operatorname{Re}\lambda <
0$، فإن $\bar x$ مستقر مقاربًا.\index{الاستقرار بالخطّية}
\end{theorem}

\begin{proof}
ننقل $\bar x$ إلى $0$ ونكتب $F(x) = Ax + g(x)$ مع $g(x) = o(\norm x)$ (بالقابلية
للاشتقاق من الصنف $\mathcal C^1$). ونختار $\alpha > 0$ يحقق
$\vertiii{\eu^{tA}} \leq C\eu^{-\alpha t}$ (حيث $t \geq 0$؛ \cref{thm:b3:ode:matrixexp}) و $r > 0$ يحقق $\norm{g(x)} \leq \frac{\alpha}{2C}\norm x$
من أجل $\norm x \leq
r$. وبدوهاميل مع $b(s) = g(x(s))$:
\[
x(t) = \eu^{tA}x_0 + \int_0^t\eu^{(t-s)A}g(x(s))\,\dd s,
\]
وهي صحيحة ما دام $\norm{x(s)} \leq r$. عندئذٍ يحقق $u(t) =
\eu^{\alpha t}\norm{x(t)}$
\[
u(t) \leq C\norm{x_0} +
\int_0^{t}C\,\frac{\alpha}{2C}\,u(s)\,\dd s ,
\]
ومنه تعطي غرونوال أن $u(t) \leq C\norm{x_0}\eu^{\alpha t/2}$، أي $\norm{x(t)} \leq C\norm{x_0}\eu^{-\alpha t/2}$. وإذا كان $\norm{x_0} < r/C$،
أبقى الحدّ القبلي $\norm{x(t)} <
r$ من أجل كل $t$ (بحجة اتصال أو ترقية:
إذ مجموعة الأزمنة التي يكون فيها $\norm x \leq r$ مفتوحة
ومغلقة في $\intco0{T_+}$ بحكم التقدير الأكيد)، ويكون الحل
شاملًا، ويتقارب إلى $0$ أُسّيًّا: أي استقرار مقارب.
\end{proof}

\begin{method}\label{met:b3:ode:strategy}
حين تواجه معادلة تفاضلية: (1) الوجود والوحدانية --- تحقق من
الليبشيتزية المحلية (وهي عادةً $\mathcal C^1$)؛ (2) الشمول ---
بالنمو الخطي، أو الحدّية، أو متراصة صامدة عبر \omterm{thm:b3:ode:lyapunov}{دالة ليابونوف} أو
تكامل أول؛ وإن لم يكن، فاشتبه بالانفجار واختبر على الكاريكاتير
السلّمي $x' = x^2$؛ (3) الجمل الخطية --- \omterm{thm:b3:ode:linearstructure}{المُحلِّل}، ودوهاميل،
وبنيةُ القيم الذاتية للمصفوفة $A$ من أجل المعاملات الثابتة؛
(4) الأسئلة النوعية --- مواضع التوازن، والخطّية، وصيدُ \omterm{thm:b3:ode:lyapunov}{دالة
ليابونوف} (أي الطاقة حين تكون الجملة ميكانيكية) أو تكاملٍ أول
تحبس مجموعاتُ مستوياته المسارات. وتمشي مسألة نهاية الأسبوع
بالطريقة كلها عبر النواس.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:ode:1}
حُلّ صراحةً وعيّن الفترة الأعظمية:
(a) $x' = x^2$ مع $x(0) = 1$؛ (b) $x' = 1 + x^2$ مع
$x(0) = 0$؛ (c) $x' = x(1-x)$ مع $x(0) = \frac12$. ووفّق بين كل
جواب و\cref{thm:b3:ode:maximal}(2) و\cref{cor:b3:ode:globallinear}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:ode:2}
ليكن $x, y$ حلّين للمعادلة $x' = f(t,x)$ حيث $f$ ليبشيتزية
شاملًا بالثابت $L$ في $x$ على $\R\times\R^d$.
(a) برهن على $\norm{x(t) - y(t)} \leq \norm{x(0) -
y(0)}\eu^{L\abs t}$، وبيّن بمثال (خطي!) أن العامل
$\eu^{L\abs t}$ مبلوغ.
(b) استنتج أن تطبيق الانسياب $x_0 \mapsto x(t; x_0)$ متصل، بل ليبشيتزي على
المجموعات المحدودة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:3}
برهن على أن لكلٍّ مما يلي حلولًا أعظمية شاملة على $\R$، مستشهدًا
بالمبرهنة الصحيحة:
(a) $x' = \sin(tx)$؛
(b) $x' = \frac{t\,x}{1 + x^2}$؛
(c) $x'' + q(t)x = 0$ مع $q$ متصلة (بالتحويل إلى جملة من الرتبة الأولى)؛
(d) $x' = A(t)x$ مع $A$ متصلة ومحدودة --- وأعطِ حدّ غرونوال
للمقدار $\norm{x(t)}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:4}
(a) احسب $\eu^{tA}$ من أجل
$A = \bigl(\begin{smallmatrix}0 & -1\\ 1 &
0\end{smallmatrix}\bigr)$،
و $\bigl(\begin{smallmatrix}\lambda & 1\\ 0 &
\lambda\end{smallmatrix}\bigr)$،
و $\bigl(\begin{smallmatrix}0 & 1\\ 1 &
0\end{smallmatrix}\bigr)$.
(b) حُلّ المذبذب المُقحَم $x'' + x = \cos(\omega t)$ بدوهاميل (بصيغة الجملة)، من
أجل $\omega \neq 1$ ومن أجل $\omega =
1$: فيظهر الرنين بوصفه الحدّ
الدنيوي $t\sin t$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:5}
من أجل المعادلة السلّمية $x'' + p(t)x' + q(t)x = 0$:
(a) برهن على أن \omterm{thm:b3:ode:linearstructure}{الفرونسكياني} $w = x_1x_2' - x_1'x_2$ لحلّين يحقق $w' = -p\,w$
(أبيل)، واستنتج أن حلّين يكون $w \neq 0$ من أجلهما في موضع ما
يشكّلان أساسًا.
(b) وبإعطاء حل واحد غير منعدم $x_1$، جد الحل العام
\emph{بتخفيض الرتبة}: ضع $x_2 = x_1\int
\frac{\eu^{-\int p}}{x_1^2}$ وتحقق. وطبّق ذلك على $t^2x'' -
2x = 0$
على $\intoo0{+\infty}$ مع $x_1(t) = t^2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:6}
(اللوجستية) من أجل $x' = x(1 - x)$: عيّن جميع مواضع التوازن
واستقرارها (بواسطة \cref{thm:b3:ode:linearization} ومباشرةً)؛ وبرهن على أن كل حل
يحقق $x(0) \in \intoo01$ متزايدٌ وشامل، بنهايتين $0$ و $1$ عند $\mp\infty$؛
وحُلّ صراحةً للتأكيد. وبرهن عمومًا على أن الحلول الذاتية
السلّمية رتيبة، واخلص: أنه لا حلول دورية غير ثابتة في البُعد
$1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:7}
(التكاملات الأولى) لتكن $H \colon \Omega \to \R$ من الصنف $\mathcal
C^1$ ولننظر في
الجملة الهاملتونية المستوية $x' =
\partial_yH$، $y' = -\partial_xH$.
(a) برهن على أن $H$ ثابتة على امتداد الحلول.
(b) ومن أجل $H = \frac{y^2}2 + \frac{x^4}4$: برهن على أن جميع الحلول شاملة ومحدودة،
وعلى أن المبدأ مستقر (بليابونوف: $H$) رغم أن الخطّية ($\bigl(\begin{smallmatrix}0&1\\0&0\end{smallmatrix}\bigr)$)
\emph{ليست} مستقرة مقاربًا: فالخطّية قد تكون غير حاسمة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:8}
(النواس المخمَّد) $x'' + cx' + \sin x = 0$ مع $c > 0$؛ وبصيغة الجملة: $x' = y$
و $y' = -\sin x - cy$.
(a) برهن على أن $V(x, y) = \frac{y^2}2 + 1 - \cos x$ يحقق $\dot
V = -cy^2 \leq 0$: أي إن المبدأ مستقر.
(b) وينعدم $\dot V$ على المحور $y = 0$ كله: فيخفق محك
ليابونوف الأكيد. برهن على الاستقرار المقارب رغم ذلك، بالخطّية
(\cref{thm:b3:ode:linearization}): احسب القيم الذاتية للمصفوفة المخطَّطة عند
$(0,0)$ وتحقق من $\operatorname{Re} < 0$ من أجل كل $c > 0$.
(c) وماذا يحدث عند موضع التوازن $(\pi, 0)$؟ احسب الخطّية واخلص
(إذ قيمة ذاتية واحدة موجبة: أي عدم استقرار --- ولك أن تستعمل
عبارة عدم الاستقرار بصورة غير رسمية أو تنتج حلًّا هاربًا
صريحًا للجملة الخطية).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:9}
(حدود الوحدانية) من أجل $\alpha \in \intoo01$، برهن على أن للمسألة $x' = \abs x^{\alpha}$ مع
$x(0) = 0$ عددًا لانهائيًّا من الحلول (بتكييف \cref{pb:b3:complete:1}، الجزء
الثالث). وبرهن بالعكس على أنه من أجل $\alpha = 1$ (أي $x' = \abs
x$)
يكون الحل المارّ بالنقطة $0$ وحيدًا، وعيّن بالضبط أي فرضية من
\cref{thm:b3:ode:cauchylipschitz} تميّز بين الحالتين.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:10}
(الحدود القبلية تحبس الحلول) لتكن $F \colon \R^d \to \R^d$ \omterm{def:b3:ode:lipschitz}{ليبشيتزية محليًّا} مع
$\langle F(x), x\rangle \leq 0$ كلما كان $\norm x \geq R$.
(a) برهن على أن الكرة المغلقة $\bar B(0, R)$ صامدة إيجابيًّا:
أي إن الحلول التي تنطلق من الداخل تبقى فيه من أجل $t \geq
0$.
\emph{(إذا كان $\norm{x(t_2)} > R$، فانظر في آخر زمن $t_1 < t_2$ يحقق
$\norm{x(t_1)} = R$ وادرس $\frac{\dd}{\dd t}\norm{x(t)}^2$ على $\intcc{t_1}{t_2}$.)}
(b) استنتج الوجود الأمامي الشامل من أجل المعطيات في الكرة. ثم
عالج الجملة التدرّجية $x' = -\nabla G(x)$، $G \in
\mathcal C^2$ مع $G(x) \to +\infty$ حين $\norm x \to
\infty$:
برهن على أن $G$ متناقصة على امتداد الحلول، وعلى أن كل حل يبقى
في مجموعة المستوى الأدنى (المحدودة) $\{G \leq
G(x_0)\}$، واخلص إلى الوجود
الأمامي الشامل.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:ode:11}
(الانفجار بالمقارنة) لننظر في $x' = x^2 + t^2$ مع $x(0) =
1$.
(a) برهن على أن \omterm{thm:b3:ode:maximal}{الحل الأعظمي} موجود على $\intco0{T_+}$ ما مع
$T_+ < \infty$: بالمقارنة مع $y' = y^2$ و $y(0) = 1$ \emph{(وبرهن
على المبرهنة المساعدة للمقارنة التي تحتاجها: أنه إذا كان $x' \geq
F(x)$
وكان $y' = F(y)$ مع $x(0) \geq y(0)$، فإن $x \geq
y$ حيثما يعيش
كلاهما)}، واستنتج $T_+ \leq 1$.
(b) وحُدّ $T_+$ من الأسفل: فعلى $\intcc01$، $x' \leq x^2 +
1$؛ وقارن مع
الحل الأعلى $z' = z^2 + 1$ مع $z(0) =
1$، المحلول بواسطة $z(t) = \tan\bigl(t + \frac\pi4\bigr)$،
واخلص إلى $T_+ \geq \frac\pi4$.
(c) اجمع $\frac\pi4 \leq T_+ \leq 1$ وصُغ العبرة: أن النمو الفوق خطي للطرف الأيمن
هو ما يقتل الوجود الشامل (\cref{exo:b3:ode:3} بوصفه النقيض)، والحدّ
الفاصل هو تقارب $\int^{\infty}\frac{\dd s}{F(s)}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:ode:12}
(مبرهنة شتورم في المقارنة) ليكن $q_1 \leq q_2$ متصلتين على
فترة $I$، ولتكن $u \neq 0$ حلًّا للمعادلة $u''
+ q_1u = 0$، ولتكن
$v \neq 0$ حلًّا للمعادلة $v'' + q_2v = 0$.
(a) أثبت متطابقة \omterm{thm:b3:ode:linearstructure}{الفرونسكياني}: أنه مع $W = uv' - u'v$، $W' = (q_1 - q_2)\,uv$.
(b) (شتورم) برهن على أنه بين جذرين متتاليين $a < b$ للدالة $u$،
إما أن تنعدم $v$ في موضع ما من $\intoo ab$، وإما أن يكون
$q_1 = q_2$ و $v \propto u$ هناك \emph{(افترض $u > 0$ على
$\intoo ab$ و $v > 0$ أيضًا؛ وكامل (a) من $a$ إلى $b$ وافحص
إشارتَي الحدّين الحدّيين $W(a),
W(b)$)}.
(c) استنتج: أن حلول $u'' + qu = 0$ مع $q \geq m > 0$ تنعدم مرة
واحدة على الأقل في كل فترة طولها $\pi/\sqrt m$ (بالمقارنة مع
$v'' + mv = 0$)؛ وأن الحلول مع $q \leq 0$ تنعدم مرة واحدة على
الأكثر على $\R$. واختبر كليهما على $q \equiv \pm1$.
\end{exercise}

\section{مسألة: النواس، محلولًا تمامًا}

\begin{problem}[{مسألة نهاية الأسبوع --- التذبذبات والدورانات
والفاصلة والدور}]\label{pb:b3:ode:1}
معادلة النواس $x'' = -\sin x$ --- وبصيغة الجملة: $x' =
y$،
$y' = -\sin x$ على $\R^2$ --- هي ذبابة الفاكهة في الديناميكا:
سهلة الكتابة، مستحيلة الحل بالصيغ الابتدائية، ومع ذلك مفهومة
تمامًا بالطريقة النوعية. ولتكن $E(x, y) = \frac{y^2}2 - \cos x$ (أي \emph{الطاقة}).

\medskip
\noindent\textbf{الجزء الأول --- البنية الشاملة.}
\begin{enumerate}
  \item برهن على أن جميع الحلول الأعظمية شاملة (معرَّفة على
        $\R$): باستعمال $\dot E = 0$ و\cref{thm:b3:ode:maximal}. ومواضع
        التوازن: $(k\pi, 0)$؛ صنّف خطّياتها (من نمط المركز من
        أجل $k$ زوجي، ومن نمط السرج من أجل $k$ فردي).
  \item برهن على أن المسارات محتواة في مجموعات المستويات
        $\{E = E_0\}$، وارسم أو صف هذه المسارات حسب قيمة
        $E_0 \in \intco{-1}{+\infty}$: $E_0 = -1$ (مواضع التوازن)، و $-1 < E_0 < 1$
        (منحنيات مغلقة حول $(2k\pi, 0)$)، و $E_0 = 1$ (الفاصلة
        المارّة بالنقطتين $(\pm\pi, 0)$)، و $E_0 > 1$ (منحنيات بيانية
        فوق $x$: أي دورانات).
  \item برهن على أن موضع التوازن السفلي $(0,0)$ مستقر لكنه
        \emph{ليس} \omterm{def:b3:ode:stability}{مستقرًّا} مقاربًا. \emph{(بليابونوف مع
        $V = E + 1$؛ وعدم الاستقرار المقارب: إذ يحبس حفظُ
        الطاقة المداراتِ على منحنيات مستوية بعيدة عن المبدأ.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- التذبذبات ودورها.} نثبّت
$-1 < E_0 < 1$ ونكتب $E_0 = -\cos a$ مع $a \in
\intoo0\pi$ (أي السعة).
\begin{enumerate}[resume]
  \item برهن على أن الحل الذي يحقق $x(0) = a$ و $y(0) = 0$
        يتذبذب: $x(t) \in \intcc{-a}a$، ويكون المدار المنحني المغلق $y^2 = 2(\cos x - \cos a)$.
        وبرّر أن الحل دوري: فالمدار منحنٍ متراص بلا مواضع
        توازن، ويُجرى بسرعة محدودة من الأسفل --- اجعل هذا
        حجةً (إذ يعود الحل إلى نقطته الابتدائية في زمن منتهٍ،
        ثم تفرض الوحدانية الدورية).
  \item أثبت صيغة الدور
        \[
        T(a) = 4\int_0^{a}\frac{\dd x}{\sqrt{2(\cos x -
        \cos a)}}
        \]
        \emph{(على ربع مدار، $y = \frac{\dd x}{\dd t} >
        0$ ثم افصل المتغيّرات؛ وبرّر
        التقارب المعتلّ عند $x = a$)}.
  \item (التذبذبات الصغيرة) عوّض $\sin\frac x2 =
        \sin\frac a2\,\sin\varphi$ وبرهن على
        \[
        T(a) = 4\int_0^{\pi/2}\frac{\dd\varphi}{\sqrt{1 -
        k^2\sin^2\varphi}},
        \qquad k = \sin\frac a2
        \]
        (وهو تكامل إهليلجي تام). واستنتج بالتقارب المهيمن أن
        $T(a) \to 2\pi$ حين $a \to 0^+$: أي النهاية التوافقية،
        المستقلة عن السعة --- أي تساوي الأزمنة التقريبي عند
        غاليليو، مع تصحيحه المضبوط $T(a) = 2\pi\bigl(1 + \frac{k^2}4 +
        O(k^4)\bigr)$ (بنشر المقدار
        المكامَل والمكاملة حدًّا حدًّا، مبرّرًا ذلك بالتقارب
        الناظمي).
  \item برهن على $T(a) \to +\infty$ حين $a \to \pi^-$ \emph{(بحدّ المقدار
        المكامَل من الأسفل بجوار $\varphi =
        \frac\pi2$ حين $k \to 1$، أو
        بتطبيق التقارب الرتيب)}: فباقتراب النواس من الفاصلة،
        يتباطأ دون حدّ.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الفاصلة.}
\begin{enumerate}[resume]
  \item من أجل $E_0 = 1$، $y = 2\cos\frac x2$ على الفرع العلوي: افصل
        المتغيّرات وكامل لتجد الحل الصريح
        \[
        x(t) = 4\arctan\bigl(\eu^{t}\bigr) - \pi
        \]
        (مع $x(0) = 0$ و $y(0) = 2$). وتحقق مباشرةً من أنه
        يحل معادلة النواس، واحسب نهاياته ونهايات $y(t)$ حين
        $t \to
        \pm\infty$.
  \item اخلص: أن مدار الفاصلة يصل بين السرج $(-\pi, 0)$ (حين
        $t \to -\infty$) والسرج $(\pi, 0)$ (حين
        $t \to +\infty$) لكنه لا يبلغ أيًّا منهما في زمن منتهٍ
        --- وهو متسق مع الوحدانية (ولماذا كان بلوغُ سرجٍ في
        زمن منتهٍ سيناقض \cref{cor:b3:ode:uniqueness}؟).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الدورانات، والصورة
الكاملة.}
\begin{enumerate}[resume]
  \item من أجل $E_0 > 1$: برهن على أن $y$ لا ينعدم أبدًا، وأن
        $x$ رتيبة تمامًا وشاملة مع $x(t) \to
        \pm\infty$، وأن
        $t \mapsto y(t)$ دورية بالدور
        \[
        \tau(E_0) = \int_{-\pi}^{\pi}
        \frac{\dd x}{\sqrt{2(E_0 + \cos x)}} .
        \]
  \item ركّب صورة الطور الكاملة (أي شكل الفصل) مع تبرير كامل
        لكل سمة، واكتب ملخصًا من عشرة أسطر للطريقة: الطاقة،
        ومجموعات المستويات، والتراص، والوحدانية --- وكيف دخلت
        كل مبرهنة من مبرهنات الفصل. وأين احتجنا يومًا إلى صيغة
        للحل العام؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- دالة الدور تحت المجهر.}
\begin{enumerate}[resume]
  \item برهن على عزوم واليس
        \[
        W_n = \int_0^{\pi/2}\sin^{2n}\varphi\,\dd\varphi
        = \frac\pi2\cdot\frac{(2n)!}{4^n\,(n!)^2}
        \]
        بالتراجع (بالمكاملة بالتجزئة)، وانشر المقدار المكامَل
        في السؤال 6 بمتسلسلة ذات الحدين، وبرّر المكاملة حدًّا
        حدًّا لتحصل على المتسلسلة الكاملة
        \[
        T(a) = 2\pi\sum_{n\geq0}
        \Bigl(\frac{(2n)!}{4^n(n!)^2}\Bigr)^{\!2}k^{2n}
        = 2\pi\Bigl(1 + \frac{k^2}4 + \frac{9k^4}{64}
        + O(k^6)\Bigr),
        \qquad k = \sin\frac a2 .
        \]
  \item حوّل إلى السعة:
        \[
        T(a) = 2\pi\Bigl(1 + \frac{a^2}{16} +
        \frac{11\,a^4}{3072} + O(a^6)\Bigr)
        \]
        \emph{(بتعويض نشر $\sin\frac a2$ والتجميع)}. فيخفق
        تساوي الأزمنة من الرتبة $a^2$، وقد صار الإخفاق الآن
        مكمَّمًا حتى الرتبة $a^4$.
  \item برهن على أن $a \mapsto T(a)$ متصلة ومتزايدة تمامًا على
        $\intoo0\pi$، واخلص مع السؤالين 6 و7 إلى أن $T$ تقابل
        من $\intoo0\pi$ على $\intoo{2\pi}{+\infty}$: أي إن كل دور فوق حرج
        تحققه سعةٌ واحدة بالضبط.
  \item (حساب صانع الساعات) نواسٌ مضبوط عند سعة متلاشية يحفظ
        الزمن \omterm{def:b3:rings:ideal}{المثالي}؛ برهن على أنه إذا شُغّل عند السعة $a$
        تأخّر بالكسر $\frac{a^2}{16} + O(a^4)$ من الزمن \omterm{def:b3:rings:ideal}{المثالي}، واحسب الانحراف
        من أجل $a = 0.2$ راديان: نحو $216$ ثانية في اليوم.
        (وخدّا هويغنس الدويريان والسعات الثابتة الصغيرة في
        الميزانات جوابان على هذا العدد.)
  \item عُد إلى دور الدوران $\tau$ في السؤال 10: برهن على أن
        $\tau$ متناقص تمامًا على $\intoo1{+\infty}$، وأن $\tau(E_0) \to +\infty$ حين
        $E_0 \to 1^+$ (بالتقارب الرتيب)، وأن $\sqrt{2E_0}\,\tau(E_0) \to 2\pi$ حين
        $E_0 \to
        +\infty$ (بالتقارب المهيمن): أي إن الدوران السريع هو
        مقاربًا دورانٌ حر بالسرعة الزاوية $\sqrt{2E_0}$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء السادس --- الطريقة مصدَّرةً:
لوتكا--فولتيرا.} وصفة النواس --- تكاملٌ أول، ومنحنيات مستوية
متراصة، ووحدانية --- تحل نظامًا بيئيًّا. نثبّت $\alpha, \beta, \gamma, \delta > 0$ وننظر،
على الربع المفتوح $Q = \intoo0{+\infty}\times\intoo0{+\infty}$، في
\[
x' = x\,(\alpha - \beta y), \qquad
y' = y\,(\delta x - \gamma)
\]
(حيث $x$ الفريسة و $y$ المفترسات).
\begin{enumerate}[resume]
  \item برهن على أن $Q$ صامدة --- إذ المحوران اتحادان لمدارات
        قابلة للحساب صراحةً، لا يمكن لأي حل أن يعبرها
        (\cref{cor:b3:ode:uniqueness}) --- وعلى أن موضع التوازن الوحيد في $Q$
        هو $(x_*, y_*) =
        \bigl(\frac\gamma\delta, \frac\alpha\beta\bigr)$.
  \item برهن على أن
        \[
        H(x, y) = \delta x - \gamma\ln x + \beta y -
        \alpha\ln y
        \]
        تكاملٌ أول، وعلى أن $H = f(x) + g(y)$ مع $f, g$ محدّبتين تمامًا
        وسويّتين على $\intoo0{+\infty}$ بأصغريتين عند $x_*$ و $y_*$،
        واستنتج أن جميع الحلول الأعظمية في $Q$ شاملة.
  \item برهن على أنه من أجل $h > h_* = H(x_*, y_*)$ تكون مجموعة المستوى $\{H = h\}\cap Q$
        منحنيًا مغلقًا حول موضع التوازن: أي فرعان متصلان
        $y_\pm(x)$ فوق فترة متراصة $\intcc{x_-}{x_+} \ni x_*$، ملتحمان عند
        الطرفين --- وهو نظير بيضاويات النواس.
  \item برهن على أن كل مدار غير توازني في $Q$ دوري: أثبت
        الدوران عكس عقارب الساعة عبر المناطق الأربع التي
        يقطعها المستقيمان $x = x_*$ و $y = y_*$، وحُدّ زمن
        عبور كل قوس بتكامل ذي شذوذ جذري تربيعي متقارب (كما في
        السؤال 5)، ثم أغلق بالوحدانية (كما في السؤال 4).
  \item (قانون فولتيرا في المتوسطات) إذا كان $T$ دورَ مدار
        كهذا، برهن على أن
        \[
        \frac1T\int_0^Tx(t)\,\dd t = \frac\gamma\delta,
        \qquad
        \frac1T\int_0^Ty(t)\,\dd t = \frac\alpha\beta :
        \]
        أي إن المتوسطات الزمنية تساوي قيم التوازن، مهما كانت
        السعة \emph{(بمكاملة $(\ln x)' =
        \alpha - \beta y$ على دور واحد)}.
  \item (مفارقة الصيد) اصطد النوعين بالمعدل $\varepsilon \in \intoo0\alpha$: فتحتفظ
        الجملة بشكلها مع $\alpha - \varepsilon$ و $\gamma +
        \varepsilon$ بدل $\alpha$
        و $\gamma$. فماذا يحدث لمتوسطَي التعداد؟ واشرح ملاحظة
        دانكونا (1914--1918): أنه لما تناقص الصيد في الأدرياتيك
        أثناء الحرب، \emph{ازدادت} نسبة المفترسات (أسماك
        القرش) في المصيد --- ولماذا يفضّل الصيدُ المعتدل
        الفريسةَ.
  \item اكتب العبرة في عشرة أسطر: أي مبرهنات الفصل تشغّل كل
        خطوة، وما الذي يحلّ محل طاقة النواس، ولماذا لم تحتج
        أي من الجملتين --- ولا تقبل --- حلًّا ابتدائيًّا
        بصيغة مغلقة.

  \item (تضمين السرعة) في نظام الدوران $E_0 > 1$، برهن على أن
        $y = x'$ يتذبذب بين $\sqrt{2(E_0 - 1)}$ (عند $x \equiv \pi \bmod 2\pi$) و $\sqrt{2(E_0 + 1)}$
        (عند $x \equiv 0$)، وعلى أن المتوسط الزمني للمقدار $y$
        على دور واحد يساوي $\frac{2\pi}{\tau(E_0)}$ بالضبط، وعلى أن نسبة
        التضمين $\sqrt{\frac{E_0 + 1}{E_0 - 1}} \to 1$ حين $E_0
        \to \infty$: أي إن الدوران السريع
        منتظمٌ مقاربًا.
  \item (رتابة دور الدوران) برهن على أن $\tau(E_0)$ من الصنف
        $\mathcal C^1$ ومتناقص تمامًا على $\intoo1{+\infty}$ \emph{(بالاشتقاق
        تحت علامة التكامل، مع الهيمنة على كل $\intco{1 + \delta}\infty$)}، مع
        $\tau \to \infty$ حين $E_0 \to 1^+$ و $\tau
        \to 0$ حين
        $E_0 \to \infty$. وركّب صورة التشعّب الكاملة للنواس على
        امتداد محور الطاقة: مواضع التوازن عند $E_0 = -1$،
        والتأرجحات بدور يتزايد من $2\pi$ إلى $\infty$ على
        $-1 < E_0 < 1$، والفاصلة عند $E_0 = 1$، والدورانات
        بدور يتناقص من $\infty$ إلى $0$ بعدها.
\end{enumerate}
\end{problem}
