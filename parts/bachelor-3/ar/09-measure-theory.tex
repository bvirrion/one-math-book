\chapter{نظرية القياس}\label{ch:b3:measure}

كم يبلغ طول جزء من $\R$؟ الجواب الساذج --- أن نُسنِد إلى كل
مجموعة طولًا صامدًا بالانسحاب يمدّد طول الفترات --- \emph{مستحيل}:
إذ يُنتج إنشاء فيتالي، في آخر هذا الفصل، مجموعةً بلا طول متسق.
ونظرية \omterm{def:b3:measure:measure}{القياس} هي التراجع المنضبط: أن نقصر الاهتمام على صنف غني من
المجموعات \emph{القابلة \omterm{def:b3:measure:measure}{للقياس}}، يوجد عليها طولٌ جمعي قابلًا للعدّ
ويكون وحيدًا. والمكاسب هائلة --- فتكامل لوبيغ (\cref{ch:b3:lebesgue})، وفضاءات
$L^p$ في التحليل الدالي، والاحتمالات الحديثة بأكملها (\cref{ch:b3:probability})،
كلها مبنيّة على المبرهنات الثلاث المبرهَن عليها هنا: مبرهنة
دينكين المساعدة في الوحدانية، ومبرهنة كاراتيودوري في التمديد،
ووجود \omterm{def:b3:measure:lebesgueouter}{قياس لوبيغ}.

\section{الجبور من النمط $\sigma$}

\begin{definition}\label{def:b3:measure:sigmaalgebra}
\emph{الجبر من النمط $\sigma$}\index{جبر سيغما@$\sigma$-جبر} على
مجموعة $X$ هو عائلة $\mathcal A$ من الأجزاء تحتوي على
$\varnothing$، وصامدة بالمتمم وبالاتحادات \emph{القابلة للعدّ} (ومنه
بالتقاطعات القابلة للعدّ وبفروق المجموعات، وتحتوي على $X$). ويُسمّى
الزوج $(X, \mathcal A)$ \emph{فضاءً قابلًا \omterm{def:b3:measure:measure}{للقياس}}؛ وتُسمّى عناصر
$\mathcal A$ \emph{مجموعات قابلة \omterm{def:b3:measure:measure}{للقياس}}. ومن أجل أي عائلة
$\mathcal E$ من الأجزاء، يرمز $\sigma(\mathcal E)$ إلى أصغر جبر من النمط
$\sigma$ يحتوي على $\mathcal E$ (وهو تقاطع جميعها --- إذ إن تقاطع
جبور من النمط $\sigma$ هو جبر كذلك).
\end{definition}

\begin{definition}\label{def:b3:measure:borel}
\emph{جبر بوريل من النمط $\sigma$}\index{جبر بوريل
سيغما@$\sigma$-جبر بوريل} لفضاء طوبولوجي هو $\mathcal B(X) = \sigma(\{\text{المجموعات المفتوحة}\})$. وعلى $\R$:
يتولّد $\mathcal B(\R)$ أيضًا بالفترات المفتوحة، وبالفترات
المغلقة، وبالأشعة $(-\infty, a]$، وبالأشعة ذات الأطراف الناطقة
(\cref{exo:b3:measure:1}) --- إذ تولّد كل عائلة المفتوحاتِ بعمليات قابلة للعدّ،
فمثلًا كل \omterm{def:b3:topology:topology}{مجموعة مفتوحة} من $\R$ اتحادٌ قابل للعدّ لفترات مفتوحة
بمعطيات ناطقة.
\end{definition}

\begin{definition}\label{def:b3:measure:dynkin}
\emph{النظام من النمط $\pi$}\index{نظام باي@$\pi$-نظام} هو عائلة
صامدة بالتقاطعات المنتهية. و\emph{النظام من النمط
$\lambda$}\index{نظام لامبدا@$\lambda$-نظام} (أو صنف دينكين) هو
عائلة $\mathcal D$ تحقق: $X \in \mathcal
D$؛ و $A, B \in \mathcal D$ مع $A \subseteq B$
$\Rightarrow$ $B
\setminus A \in \mathcal D$؛ و $A_n \uparrow A$ مع $A_n \in
\mathcal D$ $\Rightarrow$
$A \in \mathcal D$.
\end{definition}

\begin{theorem}[مبرهنة دينكين المساعدة $\pi$--$\lambda$]
\label{thm:b3:measure:dynkin}
إذا احتوى نظام $\mathcal D$ من النمط $\lambda$ على نظام
$\mathcal P$ من النمط $\pi$، فإن $\mathcal D \supseteq
\sigma(\mathcal P)$.\index{مبرهنة دينكين
المساعدة}
\end{theorem}

\begin{proof}
ليكن $\mathcal D_0$ أصغر نظام من النمط $\lambda$ يحتوي على
$\mathcal P$ (وهو تقاطع جميعها)؛ ويكفي أن نبيّن أن
$\mathcal D_0$ جبر من النمط $\sigma$، لأن عندئذٍ $\sigma(\mathcal P) \subseteq \mathcal D_0 \subseteq \mathcal
D$. وكل
نظام من النمط $\lambda$ صامد بالتقاطعات المنتهية \emph{هو} جبر
من النمط $\sigma$: فالمتممات ($X \setminus A = X
\setminus A$ مع $A \subseteq X$)،
والاتحادات المنتهية ($A \cup B =
X \setminus ((X\setminus A)\cap(X\setminus B))$)، والاتحادات القابلة للعدّ عبر $\bigcup_{k \leq n}A_k \uparrow \bigcup_kA_k$.
ومنه نبرهن على أن $\mathcal D_0$ نظام من النمط $\pi$، في
خطوتين. لتكن
\[
\mathcal D_1 = \{A \in \mathcal D_0 : A \cap P \in \mathcal
D_0 \ \forall P \in \mathcal P\}.
\]
فإن $\mathcal D_1$ نظام من النمط $\lambda$ (إذ تُتحقّق البديهيات
الثلاث بالتقاطع مع $P$: فمثلًا $(B\setminus A)\cap P
= (B \cap P)\setminus(A \cap P)$، وهو فرق فعلي داخل
$\mathcal D_0$) ويحتوي على $\mathcal P$ (وهي نظام من النمط
$\pi$): ومنه $\mathcal D_1 = \mathcal D_0$. ولتكن الآن
\[
\mathcal D_2 = \{A \in \mathcal D_0 : A \cap D \in \mathcal
D_0\ \forall D \in \mathcal D_0\}.
\]
فحسب الخطوة السابقة $\mathcal D_2 \supseteq \mathcal P$؛ و $\mathcal D_2$ نظام من النمط
$\lambda$ بالتحقق نفسه: ومنه $\mathcal D_2 = \mathcal D_0$، وهذا يقول بالضبط إن
$\mathcal D_0$ صامد بالتقاطعات.
\end{proof}

\section{القياسات}

\begin{definition}\label{def:b3:measure:measure}
\emph{القياس}\index{قياس} على $(X, \mathcal A)$ هو تطبيق $\mu \colon \mathcal A \to [0, +\infty]$ يحقق $\mu(\varnothing)
= 0$
ويكون \emph{جمعيًا من النمط $\sigma$}: أي إنه من أجل
$(A_n)_{n\in\N}$ منفصلة مثنى مثنى،
\[
\mu\Bigl(\bigsqcup_n A_n\Bigr) = \sum_n \mu(A_n).
\]
ويُسمّى $(X, \mathcal A, \mu)$ \emph{فضاء قياس}؛ ويكون $\mu$ \emph{منتهيًا} إذا
كان $\mu(X) < \infty$، و\emph{قياس احتمال} إذا كان $\mu(X) = 1$،
و\emph{منتهيًا من النمط $\sigma$} إذا كان $X$ اتحادًا قابلًا للعدّ
لمجموعات ذات قياس منتهٍ. أمثلة: قياس العدّ على $(\N, \mathcal P(\N))$؛ وكتلة
ديراك $\delta_a(A)
= \mathbf 1_{a \in A}$؛ وموضوع هذا الفصل، أي \omterm{def:b3:measure:lebesgueouter}{قياس لوبيغ}.
\end{definition}

\begin{proposition}\label{prop:b3:measure:basics}
ليكن $\mu$ قياسًا. (a) الرتابة: $A \subseteq B
\Rightarrow \mu(A) \leq \mu(B)$. (b) التحت جمعية
القابلة للعدّ: $\mu(\bigcup A_n) \leq \sum\mu(A_n)$. (c) الاتصال من الأسفل: $A_n \uparrow A \Rightarrow \mu(A_n) \to \mu(A)$. (d) الاتصال من
الأعلى: $A_n \downarrow A$ \emph{مع $\mu(A_1) < \infty$} $\Rightarrow \mu(A_n) \to \mu(A)$.
\end{proposition}

\begin{proof}
(a) $B = A \sqcup (B\setminus A)$. (b) بجعلها منفصلة: $B_n = A_n
\setminus \bigcup_{k<n}A_k$ منفصلة ولها الاتحاد نفسه،
و $\mu(B_n) \leq \mu(A_n)$. (c) $A = \bigsqcup_n (A_n
\setminus A_{n-1})$ ($A_0 = \varnothing$): فالمجاميع الجزئية للمتسلسلة
$\sum\mu(A_n\setminus A_{n-1})$ هي $\mu(A_n)$. (d) نطبّق (c) على $A_1 \setminus A_n \uparrow A_1 \setminus A$ ونطرح من
$\mu(A_1)$ --- والانتهاء يجعل الطرح مشروعًا. ومثال مضادّ دونه:
$A_n = [n, \infty)$ من أجل \omterm{def:b3:measure:lebesgueouter}{قياس لوبيغ}: $A_n \downarrow \varnothing$ لكن $\mu(A_n) =
\infty$.
\end{proof}

\begin{theorem}[الوحدانية]\label{thm:b3:measure:uniqueness}
ليكن $\mu, \nu$ قياسين على $\sigma(\mathcal P)$، ولتكن $\mathcal
P$ نظامًا من النمط
$\pi$، مع $\mu = \nu$ على $\mathcal P$. إذا وُجدت مجموعات $P_k \in \mathcal P$
تحقق $P_k \uparrow X$ و $\mu(P_k) < \infty$، فإن $\mu = \nu$ على $\sigma(\mathcal P)$ كلها.
\end{theorem}

\begin{proof}
لنثبّت $k$ ولننظر في القياسين المنتهيين $\mu_k(A) = \mu(A \cap
P_k)$ و $\nu_k(A) = \nu(A \cap P_k)$ على
$\sigma(\mathcal
P)$: فهما يتوافقان على $\mathcal P$، لأن $P \cap P_k \in \mathcal
P$ (نظام من
النمط $\pi$)، ويُسنِدان إلى $X$ القيمة المنتهية نفسها
$\mu(P_k)$. والصنف $\mathcal D = \{A : \mu_k(A) =
\nu_k(A)\}$ نظامٌ من النمط $\lambda$: إذ $X \in \mathcal D$؛
والفروق الفعلية بالطرح (لأن القيم منتهية)؛ والنهايات المتزايدة
بالاتصال من الأسفل (\cref{prop:b3:measure:basics}(c)). وهو يحتوي على النظام
$\mathcal P$ من النمط $\pi$، ومنه تعطي مبرهنة دينكين
(\cref{thm:b3:measure:dynkin}) أن $\mathcal D \supseteq
\sigma(\mathcal P)$: أي $\mu_k = \nu_k$ في كل مكان. وأخيرًا،
من أجل أي $A \in \sigma(\mathcal P)$، يعطي الاتصال من الأسفل على $A
\cap P_k \uparrow A$ أن $\mu(A) = \lim_k\mu_k(A) =
\lim_k\nu_k(A) = \nu(A)$.
\end{proof}

\section{القياسات الخارجية ومبرهنة كاراتيودوري}

\begin{definition}\label{def:b3:measure:outer}
\emph{القياس الخارجي}\index{قياس خارجي} على $X$ هو تطبيق $\mu^* \colon \mathcal P(X) \to [0, \infty]$
يحقق $\mu^*(\varnothing) = 0$، ويكون رتيبًا وتحت جمعي قابلًا للعدّ. وتكون مجموعة $A$
\emph{قابلة \omterm{def:b3:measure:measure}{للقياس} بالمعنى $\mu^*$} (كاراتيودوري) إذا شطرت كل
مجموعة شطرًا جمعيًا:
\[
\mu^*(E) = \mu^*(E \cap A) + \mu^*(E \setminus A)
\qquad \text{من أجل كل } E \subseteq X
\]
(فالمتراجحة $\leq$ صحيحة دائمًا بالتحت جمعية؛ والمضمون هو
$\geq$).
\end{definition}

\begin{theorem}[كاراتيودوري]\label{thm:b3:measure:caratheodory}
تشكّل المجموعات القابلة \omterm{def:b3:measure:measure}{للقياس} بالمعنى $\mu^*$ جبرًا من النمط
$\sigma$ هو $\mathcal
M$، ويكون $\mu^*\restriction_{\mathcal M}$ قياسًا. وعلاوة على ذلك، تنتمي
كل مجموعة تحقق $\mu^*(N) = 0$ إلى $\mathcal M$ (أي إن \omterm{def:b3:measure:measure}{القياس}
\emph{تام}).\index{مبرهنة كاراتيودوري}
\end{theorem}

\begin{proof}
تحتوي $\mathcal M$ على $\varnothing$ وهي صامدة بالمتمم (لأن
الشرط المعرِّف متناظر في $A$ و $X\setminus A$). \emph{الاتحادات
المنتهية}: ليكن $A, B \in \mathcal
M$ ولتكن $E$ كيفية؛ بشطر $E$ بواسطة $A$، ثم
كل قطعة بواسطة $B$:
\[
\mu^*(E) = \mu^*(E\cap A\cap B) + \mu^*(E\cap A\setminus B) +
\mu^*(E\cap B\setminus A) + \mu^*(E\setminus(A\cup B)).
\]
وتغطّي القطع الثلاث الأولى المجموعةَ $E \cap (A \cup B)$، ومنه تعطي التحت
جمعية أن $\mu^*(E) \geq \mu^*(E\cap(A\cup B)) +
\mu^*(E\setminus(A\cup B))$: أي $A \cup B \in \mathcal M$. وبالتراجع نحصل على الاتحادات
المنتهية؛ ومع المتممات تتاح معالجات الانفصال المنتهي.

\emph{الجمعية على $\mathcal M$}: من أجل $A, B \in
\mathcal M$ منفصلة ومن أجل
أي $E$: $\mu^*(E\cap(A\sqcup B)) = \mu^*(E\cap
A) + \mu^*(E \cap B)$ (بالشطر بواسطة $A$)؛ وبالتراجع،
\[
\mu^*\Bigl(E \cap \bigsqcup_{k\leq n}A_k\Bigr)
= \sum_{k\leq n}\mu^*(E\cap A_k).
\tag{$*$}
\]

\emph{الاتحادات القابلة للعدّ}: لتكن $(A_k) \subseteq \mathcal M$ منفصلة (وهذا يكفي، بجعلها
منفصلة داخل الجبر $\mathcal M$)، ولتكن $A = \bigsqcup A_k$، ولتكن $E$ كيفية.
باستعمال $\bigsqcup_{k\leq n}A_k \in \mathcal M$ والرتابة:
\[
\mu^*(E) = \mu^*\Bigl(E\cap\bigsqcup_{k\leq n}A_k\Bigr) +
\mu^*\Bigl(E\setminus\bigsqcup_{k\leq n}A_k\Bigr)
\geq \sum_{k \leq n}\mu^*(E\cap A_k) + \mu^*(E\setminus A)
\]
حسب ($*$). ونجعل $n \to \infty$ ونستعمل التحت جمعية القابلة للعدّ في
الاتجاه العكسي:
\[
\mu^*(E) \geq \sum_{k}\mu^*(E\cap A_k) + \mu^*(E\setminus A)
\geq \mu^*(E \cap A) + \mu^*(E\setminus A) \geq \mu^*(E):
\]
فتكون جميع المتراجحات تساويات. وهذا يبرهن على $A \in
\mathcal M$ وعلى
الجمعية القابلة للعدّ \omterm{def:b3:measure:measure}{للقياس} $\mu^*$ على $\mathcal M$، بأخذ $E = A$.

\emph{المجموعات المعدومة \omterm{def:b3:measure:measure}{القياس}}: إذا كان $\mu^*(N) = 0$، فإن من
أجل أي $E$: $\mu^*(E \cap N) + \mu^*(E\setminus N) \leq 0 + \mu^*(E)$: ومنه $N \in \mathcal M$.
\end{proof}

\section{قياس لوبيغ على \texorpdfstring{$\R$}{R}}

\begin{definition}\label{def:b3:measure:lebesgueouter}
\emph{قياس لوبيغ الخارجي}\index{قياس لوبيغ} للمجموعة $A
\subseteq \R$ هو
\[
\lambda^*(A) = \inf\Bigl\{\sum_{n} (b_n - a_n) :
A \subseteq \bigcup_n \intoo{a_n}{b_n}\Bigr\}
\]
(بالتغطيات القابلة للعدّ بفترات مفتوحة).
\end{definition}

\begin{lemma}\label{lem:b3:measure:outerbasics}
التطبيق $\lambda^*$ قياسٌ خارجي، صامد بالانسحابات، ويحقق $\lambda^*(I) = \ell(I)$
(أي الطول) من أجل كل فترة $I$.
\end{lemma}

\begin{proof}
\omterm{def:b3:measure:outer}{قياس خارجي}: تغطّي المجموعةَ $\varnothing$ فتراتٌ صغيرة كما نشاء؛
والرتابة واضحة؛ وأما التحت جمعية: فإذا أُعطيت تغطيات لكل $A_n$
في حدود $\varepsilon 2^{-n}$ من الحد الأدنى، غطّى اتحادها $\bigcup A_n$ بطول
كلي $\leq \sum
\lambda^*(A_n) + \varepsilon$. وأما الصمود بالانسحاب: فبانسحاب التغطيات.

الطول: يكفي أن نعالج $I = \intcc ab$ (فالأنماط الأخرى تختلف
بالأطراف، وقياسها الخارجي $0$: بالتغطية بفترات دقيقة؛ ثم نحصر
بمقارنات من نمط $\intcc{a+\varepsilon}{b -
\varepsilon} \subseteq \intoo ab$). أما $\lambda^*(\intcc ab) \leq b - a$: فبالتغطية بالفترات $\intoo{a-\varepsilon}{b+\varepsilon}$.
وبالعكس، ليكن $\intcc
ab \subseteq \bigcup_n\intoo{a_n}{b_n}$: فبواسطة \emph{التراص} (بوريل--لوبيغ،
\cref{thm:b3:topology:metriccompact})، يكفي عدد منتهٍ من الفترات، وليكن $I_1, \dots, I_N$. ونبيّن
$\sum_{k\leq N}(b_k -
a_k) \geq b - a$ بالتراجع على $N$: نختار $I_{k_1} \ni a$؛ فإذا كان
$b_{k_1} > b$ فقد انتهينا ($b_{k_1} - a_{k_1} > b - a$)؛ وإلا فإن القطعة $\intcc{b_{k_1}}b$
مغطّاة بالفترات $N - 1$ الباقية، ويعطي التراجع $\sum_{k \neq k_1}(b_k - a_k)
\geq b - b_{k_1}$، بينما
$b_{k_1} - a_{k_1} > b_{k_1} - a$: ثم نجمع.
\end{proof}

\begin{theorem}[قياس لوبيغ]\label{thm:b3:measure:lebesgue}
كل مجموعة بوريلية من $\R$ قابلة \omterm{def:b3:measure:measure}{للقياس} بالمعنى $\lambda^*$.
ويكون قصر $\lambda$ للتطبيق $\lambda^*$ على الجبر من النمط
$\sigma$
$\mathcal L = \mathcal M_{\lambda^*} \supseteq \mathcal B(\R)$
(ويُسمّى \emph{جبر لوبيغ من النمط $\sigma$}) هو القياسَ الوحيد
على $\mathcal B(\R)$ الذي يُسنِد إلى كل فترة طولَها؛ وهو صامد
بالانسحاب ومنتهٍ من النمط $\sigma$.
\end{theorem}

\begin{proof}
حسب \cref{thm:b3:measure:caratheodory} يكفي أن نبيّن أن كل شعاع $A = \intoo{-\infty}c$ قابل \omterm{def:b3:measure:measure}{للقياس}
بالمعنى $\lambda^*$ (لأن الأشعة تولّد $\mathcal B$،
\cref{def:b3:measure:borel}). لتكن $E \subseteq \R$ مع $\lambda^*(E) < \infty$ ولتكن $\bigcup I_n \supseteq E$
تغطية تحقق $\sum\ell(I_n) \leq \lambda^*(E) + \varepsilon$. وينشطر كل $I_n$ إلى الفترتين $I_n' = I_n \cap A$
و $I_n'' = I_n\setminus A$ (لأن فترةً منقوصةً منها شعاع فترةٌ) مع $\ell(I_n') + \ell(I_n'') = \ell(I_n)$؛ وتغطّي
الفترات $I_n'$ المجموعةَ $E \cap A$ وتغطّي $I_n''$ المجموعةَ
$E \setminus A$ (بتوسيع كلٍّ منها إلى فترة مفتوحة طولها $\ell +
\varepsilon2^{-n}$
للبقاء داخل التعريف)، ومنه
\[
\lambda^*(E\cap A) + \lambda^*(E\setminus A)
\leq \sum_n\bigl(\ell(I_n') + \ell(I_n'')\bigr) + 2\varepsilon
\leq \lambda^*(E) + 3\varepsilon .
\]
وأما الوحدانية: فقياسان يتوافقان مع الطول على النظام من النمط
$\pi$ المؤلَّف من الفترات $\intoc ab$ (وهما منتهيان عليها)
يتوافقان على $\sigma(\text{الفترات}) = \mathcal B$ حسب \cref{thm:b3:measure:uniqueness} مع $P_k = \intoc{-k}k$. وأما الانتهاء
من النمط $\sigma$: $\R = \bigcup(-k, k]$.
\end{proof}

\begin{theorem}[الانتظام]\label{thm:b3:measure:regularity}
من أجل كل $A \in \mathcal L$:\index{انتظام قياس لوبيغ}
\[
\lambda(A) = \inf\{\lambda(U) : U \supseteq A \text{ مفتوحة}\}
= \sup\{\lambda(K) : K \subseteq A \text{ متراصة}\}.
\]
\end{theorem}

\begin{proof}
\emph{الخارجي}: تغطية $\bigcup I_n$ تحقق $\sum\ell(I_n) \leq
\lambda(A) + \varepsilon$ هي \omterm{def:b3:topology:topology}{مجموعة
مفتوحة} $U \supseteq A$ تحقق $\lambda(U) \leq \lambda(A) + \varepsilon$ (بالتحت جمعية)؛ وإذا كان
$\lambda(A) = \infty$ فالعبارة تافهة. \emph{الداخلي}: لتكن $A$ أولًا محدودة،
$A \subseteq [-M, M]$. نختار مفتوحة $U \supseteq ([-M,M]\setminus A)$ تحقق $\lambda(U) \leq
\lambda([-M,M]\setminus A) + \varepsilon$؛ عندئذٍ تكون $K = [-M,
M]\setminus U$
متراصة مع $K \subseteq A$، و
\[
\lambda(K) = \lambda([-M,M]) - \lambda([-M,M]\cap U)
\geq \lambda([-M,M]) - \bigl(\lambda([-M,M]) - \lambda(A) +
\varepsilon\bigr) = \lambda(A) - \varepsilon .
\]
ومن أجل $A$ عامة: $\lambda(A) = \lim_M\lambda(A\cap[-M,M])$ (بالاتصال من الأسفل) ثم نطبّق الحالة
المحدودة في الداخل.
\end{proof}

\begin{example}\label{ex:b3:measure:cantor}
مجموعة كانتور (\cref{exo:b3:topology:10}) تحقق $\lambda(C) = 0$: إذ $C \subseteq C_n$،
وهو اتحاد $2^n$ فترةً طول كلٍّ منها $3^{-n}$، ومنه $\lambda(C) \leq (2/3)^n \to 0$. أي
مجموعة غير قابلة للعدّ ومعدومة \omterm{def:b3:measure:measure}{القياس} --- فعدد العناصر لا يرى \omterm{def:b3:measure:measure}{القياس}.
وبالعكس، فإن \emph{مجموعات كانتور السمينة} (\cref{exo:b3:measure:5}) غير
كثيفة في أي مكان وذات \omterm{def:b3:measure:measure}{قياس} موجب: \omterm{def:b3:topology:topology}{فالطوبولوجيا} لا ترى \omterm{def:b3:measure:measure}{القياس}
أيضًا. وتدفع مسألة نهاية الأسبوع هذا التفاعل إلى خلاصته
اللافتة: أي إن هناك مجموعات قابلة \omterm{def:b3:measure:measure}{للقياس} بمعنى لوبيغ وليست
بوريلية.
\end{example}

\begin{theorem}[فيتالي]\label{thm:b3:measure:vitali}
لا يوجد على \emph{جميع} أجزاء $\R$ قياسٌ صامد بالانسحاب يُسنِد
إلى كل فترة طولَها. وعلى وجه الخصوص $\mathcal L \neq \mathcal P(\R)$: أي إن المجموعات غير
القابلة \omterm{def:b3:measure:measure}{للقياس} موجودة.\index{مجموعة فيتالي غير القابلة للقياس}
\end{theorem}

\begin{proof}
لنفترض وجود مثل هذا $\mu$. على $\intcc01$، لننظر في علاقة
التكافؤ $x \sim y \iff x - y \in \Q$؛ ونختار، بحكم \emph{بديهية الاختيار}، ممثّلًا واحدًا
في $\intcc01$ لكل صنف: فنحصل على مجموعة $V$. ومن أجل $q \in \Q\cap\intcc{-1}1$،
تكون الانسحابات $V + q$ منفصلة مثنى مثنى (إذ إن نقطتين من $V$
يختلفان بعدد ناطق ستكونان متكافئتين ومع ذلك ممثّلين متمايزين)،
و
\[
\intcc01 \subseteq \bigsqcup_{q \in \Q\cap\intcc{-1}1}(V + q)
\subseteq \intcc{-1}2 :
\]
والاحتواء الأول لأن كل $x \in \intcc01$ يختلف عن ممثّله $v$
بعدد ناطق $q = x - v \in
\intcc{-1}1$. وتعطي الرتابة والجمعية من النمط $\sigma$ أن
\[
1 \leq \sum_{q}\mu(V + q) \leq 3,
\qquad\text{مع } \mu(V + q) = \mu(V) \text{ من أجل كل } q .
\]
ومجموع لامنتهٍ للثابت $\mu(V)$ يساوي $0$ أو $\infty$: فلا يمكن
أن يتحقق الحدّان معًا. إذن لا يوجد مثل هذا $\mu$ --- ومنه
$V
\notin \mathcal L$، لأن $\lambda$ على $\mathcal L$ يملك جميع الخصائص
المستعملة.
\end{proof}

\begin{remark}\label{rem:b3:measure:banachtarski}
في $\R^3$ يكون الإخفاق أشدّ إثارة: فمفارقة باناخ--تارسكي تفكّك
كرة إلى خمس قطع تُعاد تركيبًا، بالدورانات والانسحابات، لتعطي
\emph{كرتين} بنصف القطر نفسه --- ومنه فلا يوجد حتى حجم جمعي
منتهيًا وصامد بالدوران على جميع أجزاء $\R^3$. والقطع، بطبيعة
الحال، غير قابلة \omterm{def:b3:measure:measure}{للقياس}. فالقابلية \omterm{def:b3:measure:measure}{للقياس} ليست حذرًا بيروقراطيًا؛
إنها حدود الاتساق.
\end{remark}

\begin{method}\label{met:b3:measure:goodsets}
\emph{مبدأ المجموعات الجيدة}: لإثبات أن جميع مجموعات $\sigma(\mathcal E)$
تحقق خاصية ما، بيّن أن المجموعات الجيدة تشكّل جبرًا من النمط
$\sigma$ (أو نظامًا من النمط $\lambda$، إذا كانت الخاصية قياسية
وكانت $\mathcal E$ نظامًا من النمط $\pi$ --- فدينكين عندئذٍ)
يحتوي على $\mathcal E$. وكل برهان تقريبًا في هذا الفصل والذي
يليه حالةٌ من ذلك. ولإثبات تساوي قياسين: تحقق منهما على نظام
مولِّد من النمط $\pi$ مع الانتهاء من النمط $\sigma$
(\cref{thm:b3:measure:uniqueness}). ولبناء \omterm{def:b3:measure:measure}{قياس}: ابنِ قياسًا خارجيًا بالتغطيات
واستشهد بكاراتيودوري.
\end{method}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:measure:2}
(a) برهن على مبدأ الاحتواء والاستبعاد من أجل \omterm{def:b3:measure:measure}{قياس} منتهٍ: $\mu(A\cup

وعلى الصيغة ذات المجموعات الثلاث.
(b) أعطِ مثالًا يبيّن أن الاتصال من الأعلى (\cref{prop:b3:measure:basics}(d))

(c) برهن على أن \omterm{def:b3:measure:lebesgueouter}{قياس لوبيغ} لمجموعة قابلة للعدّ معدومٌ. واستنتج $\lambda(\Q) = 0$
و $\lambda(\intcc01\setminus\Q) =
1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:3}
ليكن $\mu, \nu$ قياسَي احتمال على $\mathcal B(\R)$ يحققان
$\mu(\intoc{-\infty}t) = \nu(\intoc{-\infty}t)$ من أجل كل $t \in \R$. برهن على $\mu = \nu$. (وهذا يجعل
\emph{دالة التوزيع} $F(t) = \mu(\intoc{-\infty}t)$ صامدًا تامًّا --- وهو أساس
\cref{ch:b3:probability}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:4}
(بوريل--كانتيلي، صيغة \omterm{def:b3:measure:measure}{القياس}) لتكن $(A_n)$ قابلة \omterm{def:b3:measure:measure}{للقياس} مع
$\sum_n\mu(A_n) < \infty$، ولتكن $\limsup A_n =
\bigcap_N\bigcup_{n\geq N}A_n$ (أي النقط المنتمية إلى عدد لامنتهٍ من
المجموعات $A_n$). برهن على $\mu(\limsup A_n) = 0$. تطبيق: من أجل كل
$x \in \intcc01$ تقريبًا، لا يحقق سوى عدد منتهٍ من الأدلّة $n$
الشرطَ $\abs{x - p/q_n} \leq 4^{-n}$ من أجل العدد الناطق النوني $p/q_n$ من ترقيم
للمجموعة $\Q\cap\intcc01$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:5}
(مجموعة كانتور السمينة) أعد إنشاء كانتور على $\intcc01$، لكن
انزع عند الخطوة $n$ من كل فترة من الفترات $2^{n-1}$ فترةً
مفتوحةً \emph{متمركزةً} طولها $4^{-n}$ لا غير. برهن على أن
$K = \bigcap K_n$ الناتجة متراصة وداخلها خالٍ (فلا يبقى أي فترة)، وأن
\[
\lambda(K) = 1 - \sum_{n\geq1}2^{n-1}4^{-n} = \tfrac12 :
\]
أي مجموعة غير كثيفة في أي مكان وقياسها $\frac12$. واستنتج جزءًا
\emph{هزيلًا} من $\intcc01$ قياسه الكامل $1$، وجزءًا مفتوحًا
كثيفًا قياسه $< \varepsilon$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:6}
ليكن $\mu$ قياسًا على $\mathcal B(\R)$ صامدًا بالانسحابات مع
$c = \mu(\intoc01) < \infty$. برهن على $\mu =
c\,\lambda$ على $\mathcal B(\R)$. \emph{(احسب
$\mu$ على الفترات الثنائية بقسمة $\intoc01$ إلى $2^n$ انسحابًا،
ثم استشهد بما ورد في \cref{thm:b3:measure:uniqueness}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:7}
(التقريب) لتكن $A \in \mathcal L$ مع $\lambda(A) <
\infty$ و $\varepsilon > 0$. برهن على وجود
اتحاد \emph{منتهٍ} لفترات $B$ يحقق $\lambda(A\,\triangle\,B) <
\varepsilon$ (حيث $\triangle$ =
الفرق المتناظر). \emph{(بالانتظام: احصر $K \subseteq A \subseteq U$ واستعمل بنية
المفتوحة $U$ بوصفها اتحادًا قابلًا للعدّ لفترات، مع تراص $K$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:measure:8}
(شتاينهاوس) لتكن $A \in \mathcal L$ مع $\lambda(A) > 0$. برهن على أن $A - A = \{x - y : x, y \in A\}$
تحتوي على فترة حول $0$. \emph{(اختزل إلى $\lambda(A) < \infty$؛ وبانتظام من
نمط \cref{exo:b3:measure:7}، جد فترةً $I$ تحقق $\lambda(A \cap I) > \frac34\ell(I)$؛ ثم من أجل $\abs t <
\frac12\ell(I)$،
تقع المجموعتان $A\cap I$ و $(A\cap I) + t$ كلتاهما في فترة طولها
$\frac32\ell(I)$ وقياسهما الكلي $> \frac32\ell(I)$: ومنه يجب أن تتقاطعا.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:measure:9}
برهن على أن كل $A \in \mathcal L$ مع $\lambda(A) > 0$ تحتوي على جزء غير
قابل \omterm{def:b3:measure:measure}{للقياس}. \emph{(قاطع $A$ مع انسحابات $V + q$ لمجموعة
فيتالي: فلو كانت جميع المجموعات $A \cap (V+q)$ قابلة \omterm{def:b3:measure:measure}{للقياس}،
لكانت كلٌّ منها معدومة \omterm{def:b3:measure:measure}{القياس} بحجّة \cref{thm:b3:measure:vitali} --- وتساعد
مبرهنة شتاينهاوس (\cref{exo:b3:measure:8}): إذ إن مجموعة قابلة \omterm{def:b3:measure:measure}{للقياس} ذات
\omterm{def:b3:measure:measure}{قياس} موجب داخل $V + q$ ستعطي $(V+q) - (V+q) \supseteq$ فترةً، وهذا يناقض كون
مجموعة الفروق هذه لا تلاقي $\Q$ إلا عند $0$؛ ثم اخلص بالتحت
جمعية.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:10}
برهن على أن $A \subseteq \R$ مع $\lambda^*(A) < \infty$ قابلةٌ \omterm{def:b3:measure:measure}{للقياس} بمعنى
لوبيغ إذا وفقط إذا وُجدت، من أجل كل $\varepsilon > 0$، مفتوحةٌ
$U \supseteq A$ تحقق $\lambda^*(U \setminus A) <
\varepsilon$، وإذا وفقط إذا وُجدت مجموعة
$G_\delta$ هي $G \supseteq A$ تحقق $\lambda^*(G\setminus A) = 0$. (ومنه فإن مجموعات
لوبيغ هي مجموعات بوريل بترديد المجموعات المعدومة \omterm{def:b3:measure:measure}{القياس}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:measure:11}
(الاتصال على النهايات الرتيبة، ودقّته)
(a) برهن على أنه من أجل المجموعات القابلة \omterm{def:b3:measure:measure}{للقياس}، $\mu(\liminf A_n) \leq
\liminf\mu(A_n)$ (فاتو
للمجموعات)، وعلى أنه إذا كان $\mu\bigl(\bigcup A_n\bigr) < \infty$ فإن $\limsup\mu(A_n) \leq \mu(\limsup A_n)$ أيضًا.
(b) أعطِ، من أجل \omterm{def:b3:measure:lebesgueouter}{قياس لوبيغ} على $\R$، متتاليةً تحقق
$\mu(A_n) = 1$ من أجل كل $n$ ومع ذلك $\mu(\limsup A_n) = 0$: أي إن فرضية
الانتهاء في المتراجحة الثانية ليست زخرفية.
(c) استنتج: أنه إذا كان $\sum\mu(A_n) < \infty$ فإن $\mu(\limsup A_n) = 0$ (بوريل--كانتيلي مرة
أخرى)، وأنه إذا كانت المجموعات $A_n$ متزايدة أو متناقصة (مع
$\mu(A_1) < \infty$ في الحالة المتناقصة)، فإن $\mu(\lim A_n) = \lim\mu(A_n)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:measure:12}
(مبرهنة إيغوروف) لتكن $\mu(X) < \infty$ وليكن $f_n \to f$ نقطةً نقطة، مع
كون الجميع قابلًا \omterm{def:b3:measure:measure}{للقياس} (وحقيقي القيم). ومن أجل $k, N \geq 1$
نضع
\[
E_{k,N} = \bigcap_{n \geq N}\Bigl\{x : \abs{f_n(x) - f(x)}
\leq \tfrac1k\Bigr\} .
\]
(a) برهن على أنه من أجل $k$ مثبَّت يكون $E_{k,N} \nearrow X$ حين $N \to
\infty$،
واستنتج وجود $N_k$ يحقق $\mu(X \setminus E_{k,N_k})
\leq \varepsilon2^{-k}$.
(b) اخلص إلى \emph{مبرهنة إيغوروف}: أنه من أجل كل $\varepsilon
> 0$ توجد
مجموعة قابلة \omterm{def:b3:measure:measure}{للقياس} $A$ تحقق $\mu(X\setminus A) \leq
\varepsilon$ بحيث يكون $f_n \to f$
\emph{بانتظام على $A$} --- أي إن التقارب نقطةً نقطة هو تقارب
منتظم خارج مجموعة صغيرة كما نشاء.
(c) برهن على أن المبرهنة تخفق على $(\R, \lambda)$: فالنتوءات
المتحرّكة $f_n = \mathbf 1_{\intcc n{n+1}}$ تتقارب نقطةً نقطة إلى $0$ لكن ليس بانتظام على
متمم أي مجموعة ذات \omterm{def:b3:measure:measure}{قياس} منتهٍ. وأين استُعمل $\mu(X) < \infty$ في (a)؟
\end{exercise}

\begin{problem}[{مسألة نهاية الأسبوع --- سلّم الشيطان،
و $\mathcal B(\R) \subsetneq \mathcal L$}]
\label{pb:b3:measure:1}
نبني \emph{دالة كانتور--فيتالي} (سلّم الشيطان)، ونستعملها لنقل
\omterm{def:b3:measure:measure}{القياس} نقلًا شاذًّا، ونخلص إلى مبرهنة لا تعطيها أي حجّة لطيفة:
أي إن هناك مجموعات قابلة \omterm{def:b3:measure:measure}{للقياس} بمعنى لوبيغ وليست بوريلية.
والترميز: $C$ مجموعة كانتور، و $C_n$ مرحلتها النونية ($2^n$
فترةً طول كلٍّ منها $3^{-n}$)، ولكل $x \in C$ أرقام ثلاثية
$x = \sum 2b_n3^{-n}$ مع $b_n \in \{0,1\}$ (\cref{exo:b3:topology:10}).

\medskip
\noindent\textbf{الجزء الأول --- السلّم.} نعرّف $c_0(x) = x$
و $c_{n+1}$ انطلاقًا من $c_n$ بالعلاقة
\[
c_{n+1}(x) = \begin{cases}
\tfrac12\,c_n(3x) & 0 \leq x \leq \tfrac13,\\[2pt]
\tfrac12 & \tfrac13 \leq x \leq \tfrac23,\\[2pt]
\tfrac12 + \tfrac12\,c_n(3x - 2) & \tfrac23 \leq x \leq 1.
\end{cases}
\]
\begin{enumerate}
  \item برهن على أن كل $c_n$ متصلة وغير متناقصة مع $c_n(0) = 0$
        و $c_n(1) = 1$، وعلى أن $\norm{c_{n+1} - c_n}_\infty \leq
        \tfrac12\norm{c_n - c_{n-1}}_\infty$.
  \item استنتج أن $(c_n)$ تتقارب بانتظام إلى دالة متصلة غير
        متناقصة $c$ تحقق $c(0) = 0$ و $c(1) = 1$ (وهي
        \emph{دالة} كانتور--فيتالي)، وتحقق العلاقات الذاتية
        التشابه نفسها التي تحققها $c_{n+1}$ أعلاه.
  \item برهن على أن $c$ ثابتة على كل مركّبة مترابطة من $\intcc01\setminus C$،
        وعلى أنه من أجل $x = \sum_n
        2b_n3^{-n} \in C$ يكون $c(x) = \sum_n b_n2^{-n}$: أي إن السلّم يقرأ
        أرقام كانتور بالنظام الثنائي (وهي دالة $g$ الواردة في
        \cref{pb:b3:topology:1}، مجعولةً رتيبة وشاملة).
  \item استنتج أن $c$ قابلة للاشتقاق مع $c' = 0$ عند كل نقطة من
        $\intcc01\setminus C$: أي $c' = 0$ \emph{في كل مكان تقريبًا بالمعنى
        $\lambda$} (\cref{ex:b3:measure:cantor}). واخلص إلى أن مبرهنة التحليل
        الأساسية تخفق من أجل $c$:
        \[
        c(1) - c(0) = 1 \neq 0 = \int_0^1 c'(t)\,\dd t
        \]
        (والتكامل على المجموعة ذات \omterm{def:b3:measure:measure}{القياس} الكامل التي ينعدم
        عليها $c'$؛ واستباقًا لما في \cref{ch:b3:lebesgue}، لا تؤثر المجموعات
        المعدومة \omterm{def:b3:measure:measure}{القياس} في التكاملات). وأي فرضية من فرضيات
        المبرهنة الأساسية من الصنف $\mathcal C^1$ خُولفت؟
  \item برهن على $c(C) = \intcc01$: أي إن المجموعة المعدومة \omterm{def:b3:measure:measure}{القياس} $C$
        تُرسَل \emph{على} مجموعة ذات \omterm{def:b3:measure:measure}{قياس} كامل.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- \omterm{def:b3:topology:continuity}{التماثل الطوبولوجي} الملتوي.}
لتكن $h(x) = \frac{c(x) + x}{2}$.
\begin{enumerate}[resume]
  \item برهن على أن $h \colon \intcc01 \to \intcc01$ \omterm{def:b3:topology:continuity}{تماثل طوبولوجي} (فهي متزايدة تمامًا
        ومتصلة وشاملة).
  \item برهن على $\lambda\bigl(h(\intcc01\setminus C)\bigr) =
        \tfrac12$: إذ يفعل $h$ على كل فجوة طولها $\ell$
        بوصفه تطبيقًا تآلفيًا ميله $\tfrac12$، ويكون الطول الكلي
        للفجوات $1$.
  \item استنتج $\lambda\bigl(h(C)\bigr) = \tfrac12$: أي إن الصورة المماثلة طوبولوجيًا
        لمجموعة معدومة \omterm{def:b3:measure:measure}{القياس} يمكن أن يكون قياسها موجبًا. (وأين
        يناقض هذا الحدس الساذج حول «الحجم»؟)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- مجموعة قابلة \omterm{def:b3:measure:measure}{للقياس} وليست
بوريلية.}
\begin{enumerate}[resume]
  \item حسب \cref{exo:b3:measure:9}، اختر مجموعة غير قابلة \omterm{def:b3:measure:measure}{للقياس} $W
        \subseteq h(C)$.
        برهن على أن $Z = h^{-1}(W) \subseteq C$ قابلة \omterm{def:b3:measure:measure}{للقياس} بمعنى لوبيغ. \emph{(فهي
        جزء من مجموعة معدومة \omterm{def:b3:measure:measure}{القياس}؛ بالتمام، \cref{thm:b3:measure:caratheodory}.)}
  \item برهن على أن سابق مجموعة بوريلية \omterm{def:b3:topology:continuity}{بتطبيق متصل} بوريليٌّ.
        \emph{(بمبدأ المجموعات الجيدة: فإن $\{B : h^{-1}(B) \in \mathcal B\}$ جبر من النمط
        $\sigma$ يحتوي على \omterm{def:b3:topology:topology}{المجموعات المفتوحة} --- وانتبه إلى
        اتجاه التطبيق.)}
  \item اخلص إلى أن $Z$ \emph{ليست} بوريلية: إذ لو كانت كذلك
        لكانت $W = (h^{-1})^{-1}(Z)$ بوريلية (بتطبيق السؤال 10 على $h^{-1}$
        المتصل)، ومنه قابلة \omterm{def:b3:measure:measure}{للقياس} --- وهو تناقض. ومن ثَمّ
        \[
        \boxed{\ \mathcal B(\R) \subsetneq \mathcal L\ }
        \]
        ويوسّع التمامُ عالمَ بوريل فعلًا.
  \item أعطِ دالة $g$ قابلة \omterm{def:b3:measure:measure}{للقياس} بمعنى لوبيغ ودالة متصلة
        $\varphi$ بحيث لا تكون $g \circ
        \varphi$ قابلة \omterm{def:b3:measure:measure}{للقياس} بمعنى لوبيغ:
        فالقابلية \omterm{def:b3:measure:measure}{للقياس}، بخلاف الاتصال، لا تتركّب. \emph{(خذ
        $g =
        \mathbf 1_Z$ و $\varphi = h^{-1}$، استباقًا لتعريف الدوال القابلة \omterm{def:b3:measure:measure}{للقياس}
        من \cref{ch:b3:lebesgue}: فسوابق المجموعات البوريلية مجموعات لوبيغ.
        وأين يجب الحذر بشأن أي جبر من النمط $\sigma$ يُستعمل على
        المستقر؟)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- خاتمة.}
\begin{enumerate}[resume]
  \item رتّب الأصناف التالية بالاحتواء التام وبرّر كل تمامية
        بمثال من هذا الفصل ومسألته: المجموعات القابلة للعدّ؛
        والمجموعات البوريلية المعدومة \omterm{def:b3:measure:measure}{القياس}؛ والمجموعات
        المعدومة \omterm{def:b3:measure:measure}{القياس} بمعنى لوبيغ؛ ومجموعات بوريل؛ ومجموعات
        لوبيغ؛ والمجموعات الكيفية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- \omterm{def:b3:measure:measure}{قياس} كانتور: كتلة على مجموعة
معدومة \omterm{def:b3:measure:measure}{القياس}.} السلّم هو دالة توزيع \omterm{def:b3:measure:measure}{لقياس} لافت، وسنبنيه الآن
بأدوات هذا الفصل نفسه.
\begin{enumerate}[resume]
  \item (لوبيغ--ستيلتيس، الوجود) لتكن $F\colon\R\to\R$ غير
        متناقصة ومتصلة ومحدودة. نعرّف على الفترات نصف المفتوحة
        $\rho\bigl(\intoc ab\bigr) = F(b) -
        F(a)$، ومن أجل $A \subseteq \R$،
        \[
        \mu_F^*(A) = \inf\Bigl\{\sum_k\bigl(F(b_k) -
        F(a_k)\bigr) : A \subseteq
        \bigcup_k\intoc{a_k}{b_k}\Bigr\} .
        \]
        برهن على أن $\mu_F^*$ \omterm{def:b3:measure:outer}{قياس خارجي} وعلى أن
        $\mu_F^*\bigl(\intoc ab\bigr) = F(b) - F(a)$
        \emph{(قلّد حجّة التراص في \cref{thm:b3:measure:lebesgue}، بتوسيع كل
        $\intoc{a_k}{b_k}$ إلى فترة مفتوحة بكلفة $F$ قدرها $\leq \varepsilon2^{-k}$ --- وأين
        يُستعمل اتصال $F$؟)}.
  \item برهن على أن كل مجموعة بوريلية قابلة \omterm{def:b3:measure:measure}{للقياس} بالمعنى
        $\mu_F^*$ بمفهوم كاراتيودوري \emph{(كما في حالة لوبيغ،
        يكفي الاختبار على أنصاف المستقيمات؛ اتّبع برهان تطبيق
        \cref{thm:b3:measure:caratheodory})}، ومنه فإن $\mu_F = \mu_F^*$ مقصورًا على
        $\mathcal B(\R)$ قياسٌ يحقق $\mu_F(\intoc ab) = F(b) - F(a)$: وهو \emph{\omterm{def:b3:measure:measure}{قياس}
        لوبيغ--ستيلتيس} للدالة $F$.
  \item طبّق ذلك على السلّم ($F = c$ ممدَّدةً بالقيمة $0$ على
        $\R_-$ وبالقيمة $1$ على $\intco1\infty$): فنحصل على
        \emph{\omterm{def:b3:measure:measure}{قياس} كانتور} $\mu$. برهن على $\mu(\R) = 1$، وعلى
        أن كل فجوة من فجوات مجموعة كانتور معدومة \omterm{def:b3:measure:measure}{القياس} بالمعنى
        $\mu$ (لأن $c$ ثابتة هناك)، واخلص إلى
        \[
        \mu(C) = 1, \qquad \lambda(C) = 0 :
        \]
        أي إن $\mu$ و $\lambda$ يعيشان على حاملين منفصلين ($C$
        ومتممها). ويُقال عن قياسين في هذا الوضع إنهما
        \emph{مفردان تبادليًا}، ويُكتب $\mu \perp
        \lambda$.
  \item برهن على أنه ليست \omterm{def:b3:measure:measure}{للقياس} $\mu$ ذرّات: أي $\mu(\{x\}) = 0$ من
        أجل كل $x$ \emph{(باتصال $c$)}. أي \omterm{def:b3:measure:measure}{قياس} احتمال بلا ذرّات
        محمول على متراص معدوم \omterm{def:b3:measure:measure}{القياس} بمعنى لوبيغ: قارن ذلك
        بالقياسات الوحيدة التي رأيناها حتى الآن.
  \item (رمي القطعة النقدية متنكّرًا) من أجل كلمة $(\varepsilon_1,
        \dots, \varepsilon_m) \in \{0,1\}^m$، لتكن
        $C_{\varepsilon}$ مجموعة العناصر $x \in C$ التي تحقق أرقامها
        الثلاثية $b_i(x) = \varepsilon_i$ من أجل $i \leq m$ (أي إحدى قطع كانتور
        $2^m$ من العمق $m$). برهن على $\mu(C_\varepsilon) = 2^{-m}$ \emph{(فالسلّم
        يصعد $2^{-m}$ عبر تلك القطعة: استعمل الجزء الأول،
        السؤال 3)}. \omterm{def:b3:measure:measure}{فقياس} كانتور هو قانون متتالية لامنتهية من
        رميات قطعة نقدية عادلة مقروءةً بالنظام الثلاثي ---
        وسيجعل \cref{ch:b3:probability} ذلك دقيقًا.
  \item برهن على التشابه الذاتي: من أجل كل مجموعة بوريلية $A$،
        \[
        \mu(A) = \tfrac12\,\mu(3A) +
        \tfrac12\,\mu(3A - 2) ,
        \]
        حيث $3A - 2 = \{3x - 2 : x \in A\}$ \emph{(تحقق منه على الفترات المولِّدة
        $\intoc ab$ بالعلاقات الذاتية التشابه للدالة $c$، ثم
        استشهد بالوحدانية، \cref{thm:b3:measure:uniqueness})}.
  \item برهن على أن الانعكاس $s(x) = 1 - x$ يحفظ $\mu$: أي
        $\mu(s(A)) = \mu(A)$ \emph{(عبر $c(1 - x) = 1 -
        c(x)$، وهو ينتج من تناظر البناء ---
        برهن عليه)}.
  \item احسب أول عزمين \omterm{def:b3:measure:measure}{للقياس} $\mu$، أي لنقطة عشوائية $X$
        قانونها $\mu$ (ويمكن معالجة التكاملات بوصفها نهايات
        لمجاميع على قطع العمق $m$، استباقًا لما في \cref{ch:b3:lebesgue}):
        فيعطي التناظر $\int x\,\dd\mu = \frac12$، ويعطي التشابه الذاتي
        \[
        \int x^2\,\dd\mu = \frac38,
        \qquad\text{ومنه}\qquad
        \operatorname{Var}(X) = \frac18 .
        \]
        قارن ذلك بالقانون المنتظم على $\intcc01$ (وتباينه
        $\frac1{12}$): فكتلة كانتور، مدفوعةً إلى الحواف، تنتشر
        \emph{أكثر}.
  \item برهن على أن \emph{الحامل} (الطوبولوجي) \omterm{def:b3:measure:measure}{للقياس} $\mu$ ---
        أي أصغر مجموعة مغلقة ذات \omterm{def:b3:measure:measure}{قياس} كامل --- هو $C$ بالضبط.
  \item (تركيب) السلّم $c$ متصل وغير متناقص ومع ذلك يخالف
        مبرهنة التحليل الأساسية (الجزء الأول)؛ \omterm{def:b3:measure:measure}{والقياس} $\mu_c$
        \omterm{def:b3:measure:measure}{قياس} احتمال بلا ذرّات ومفرد بالنسبة إلى $\lambda$. اشرح
        في فقرة قصيرة كيف يكون هذان وجهين لظاهرة واحدة، واذكر
        العبرة العامة: أن الدوال غير المتناقصة توافق قياسات
        ($F \leftrightarrow \mu_F$)، وأن القابلية للاشتقاق في كل مكان تقريبًا
        توافق «الجزء المتصل اتصالًا مطلقًا»، وأن $c$ هي الشاهد
        المعياري على أن دالة متصلة $F$ يمكن ألّا تحمل \emph{أي}
        جزء متصل اتصالًا مطلقًا البتة.
  \item (معامل الاتصال الدقيق) لتكن $s =
        \frac{\ln 2}{\ln 3}$. برهن على أن $c$
        هولدرية متصلة من الأُسّ $s$:
        \[
        \abs{c(x) - c(y)} \leq 4\,\abs{x - y}^{s}
        \qquad (x, y \in \intcc01),
        \]
        وعلى أنه لا يصلح أي أُسّ $t > s$، ولا حتى محليًا. واستنتج
        الصيغة القياسية: أنه من أجل كل $x$ وكل $r \in \intoc01$،
        \[
        \mu\bigl(\intcc{x - r}{x + r}\bigr) \leq 8\,r^{s} .
        \]
        \emph{(قارن شبكة ثلاثية من العمق $m$ بمقياس
        $\abs{x - y}$؛ ويعطي السؤال 18 الصعودَ عبر كل قطعة.
        والأُسّ $s$ هو بُعد هاوسدورف للمجموعة $C$، كما ستقول
        مقرّرات لاحقة.)}
  \item (التشابه الذاتي يميّز $\mu$) برهن على عكس السؤال 19: أنه
        إذا كان $\nu$ \omterm{def:b3:measure:measure}{قياس} احتمال على $\mathcal B(\R)$ محمولًا
        على $\intcc01$ ويحقق
        \[
        \nu(A) = \tfrac12\,\nu(3A) + \tfrac12\,\nu(3A - 2)
        \qquad (A \in \mathcal B(\R)),
        \]
        فإن $\nu = \mu$. \emph{(كرّر العلاقة $m$ مرة لنشر $\nu$
        على قطع كانتور $2^m$ من العمق $m$، وقدّر $\nu(\intoc ab)$
        مقابل عدد القطع الواقعة داخل $\intoc ab$، ثم اجعل
        $m \to \infty$؛ وأنهِ بما ورد في \cref{thm:b3:measure:uniqueness}.)}
\end{enumerate}
\end{problem}
