\chapter{الدوال الهولومورفية}\label{ch:b3:holomorphic}

تبدو القابلية للاشتقاق العقدي تنويعًا صغيرًا على النظرية
الحقيقية --- نهايةٌ واحدة، ونسبةٌ واحدة. لكنها في الحقيقة كونٌ
آخر. فدالةٌ قابلة للاشتقاق \emph{مرة واحدة} على جزء مفتوح من
$\C$ تكون تلقائيًّا قابلة للاشتقاق عددًا لانهائيًّا من المرات،
وتحليلية، ومعيَّنة على مجال بأكمله بقيمها بجوار نقطة واحدة،
ومقيَّدة بمبادئ شاملة صارمة (ليوفيل، والطويلة العظمى). وكل هذا
يتدفق من معجزة واحدة، هي مبرهنة كوشي: أن تكامل \omterm{def:b3:holomorphic:holo}{دالة هولومورفية}
على امتداد مسار مغلق في مجال نجمي الشكل منعدم. ويبرهن هذا الفصل
على المعجزة (بحجة غورسا، دون افتراض أي انتظام وراء القابلية
للاشتقاق)، ويجني نتائجها الكلاسيكية، ويسدّد مبرهنةً استعملها
هذا الكتاب على الحساب منذ \cref{ch:b3:galois}: أن لكل كثير حدود عقدي غير
ثابت جذرًا.

وفي كل ما يلي، تكون $\Omega \subseteq \C$ مفتوحة، ويرمز $D(a, r)$ إلى القرص
المفتوح.

\section{القابلية للاشتقاق العقدي}

\begin{definition}\label{def:b3:holomorphic:holo}
تكون $f \colon \Omega \to \C$ \emph{هولومورفية}\index{دالة هولومورفية} على
$\Omega$ إذا كانت، من أجل كل $z_0 \in \Omega$،
\[
f'(z_0) = \lim_{h\to0}\frac{f(z_0 + h) - f(z_0)}{h}
\]
موجودةً (حيث $h \in \C^*$). ومجاميع الدوال الهولومورفية
وجداءاتها وخارج قسمتها (بمقامات غير معدومة) ومركّباتها
هولومورفية، بالصيغ المعتادة (وبراهين السنتين الجامعيتين 1 و2
حرفيًّا: إذ لا تستعمل سوى عمليات الحقل والنهايات). ويرمز $\mathcal H(\Omega)$
إلى مجموعة الدوال الهولومورفية على $\Omega$.
\end{definition}

\begin{proposition}[كوشي--ريمان]
\label{prop:b3:holomorphic:cauchyriemann}
نكتب $f(x + \iu y) = P(x,y) + \iu Q(x,y)$. عندئذٍ تكون $f$ \omterm{def:b3:holomorphic:holo}{هولومورفية} عند $z_0$ إذا وفقط إذا
كانت $f$ قابلة للاشتقاق بالمعنى $\R$ عند $z_0$ (بوصفها تطبيقًا
لمتغيّرين حقيقيين) و\index{معادلتا كوشي--ريمان}
\[
\frac{\partial P}{\partial x} = \frac{\partial Q}{\partial y},
\qquad
\frac{\partial P}{\partial y} = -\frac{\partial Q}{\partial x}
\qquad \text{عند} z_0 ;
\]
أي، بالتكافؤ، إذا كان التفاضل الحقيقي هو الضرب في العدد العقدي
$f'(z_0)$.
\end{proposition}

\begin{proof}
تقول القابلية للاشتقاق بالمعنى $\C$ إن $f(z_0 + h) = f(z_0) + ch +
o(\abs h)$ مع $c = f'(z_0)$:
أي تفاضلٌ خطي بالمعنى $\R$ هو الضرب في $c = a + \iu b$، أي
مصفوفته $\bigl(\begin{smallmatrix} a & -b\\ b & a
\end{smallmatrix}\bigr)$ في الأساس $(1, \iu)$ --- وهي بالضبط العلاقات
المعروضة بين المشتقات الجزئية. وبالعكس يكون تفاضلٌ كهذا خطيًّا
بالمعنى $\C$، ويتوافق تعريفا $o(\abs h)$.
\end{proof}

\begin{example}\label{ex:b3:holomorphic:examples}
كثيرات الحدود في $z$، والدوال الكسرية خارج أقطابها، و --- حسب
مبرهنة الاشتقاق حدًّا حدًّا لمتسلسلات القوى في السنة الجامعية
2، وبرهانها يشتغل بالكيفية نفسها على $\C$ --- كل مجموع لمتسلسلة
قوى $\sum a_n(z - a)^n$ داخل قرص تقاربها: \omterm{def:b3:holomorphic:holo}{هولومورفية}، ومشتقتها $\sum na_n(z -
a)^{n-1}$
(بنصف القطر نفسه). وعلى وجه الخصوص تكون $\exp z = \sum z^n/n!$ صحيحة
(\omterm{def:b3:holomorphic:holo}{هولومورفية} على $\C$) مع $\exp' = \exp$. ومن جهة أخرى فإن
$z \mapsto \bar z$ و $\abs z$ و $\operatorname{Re}z$ ليست \omterm{def:b3:holomorphic:holo}{هولومورفية} في أي مكان (إذ تخفق
كوشي--ريمان في كل مكان): \omterm{def:b3:holomorphic:holo}{فالهولومورفية} صرامةٌ تحفظ التوجيه
والزوايا، لا نعومة.
\end{example}

\section{تكاملات المسار}

\begin{definition}\label{def:b3:holomorphic:contour}
\emph{المسار} تطبيق $\gamma \colon
\intcc ab \to \C$ من الصنف $\mathcal C^1$ بالقطع؛
ويكون \emph{مغلقًا} إذا كان $\gamma(a) =
\gamma(b)$. ومن أجل $f$ متصلة على صورة
$\gamma$:\index{تكامل مسار}
\[
\int_\gamma f(z)\,\dd z =
\int_a^b f(\gamma(t))\,\gamma'(t)\,\dd t,
\qquad
\Bigl|\int_\gamma f\,\dd z\Bigr| \leq
\sup_{\gamma}\abs f\cdot\operatorname{length}(\gamma)
\]
(وهي \emph{متراجحة $ML$}؛ والطول $= \int_a^b\abs{\gamma'}$). والتكامل صامد بإعادة
الوسطنة المتزايدة من الصنف $\mathcal C^1$ ويغيّر إشارته بعكس
التوجيه.
\end{definition}

\begin{proposition}[الدوال الأصلية]\label{prop:b3:holomorphic:primitive}
من أجل $f$ متصلة على $\Omega$، تتكافأ العبارات التالية:
(أ) تقبل $f$ \emph{دالة أصلية} $F \in \mathcal H(\Omega)$ ($F'
= f$)؛ (ب) $\int_\gamma f\,\dd z = 0$ من أجل
كل مسار مغلق $\gamma$ في $\Omega$. وفي تلك الحالة يكون $\int_\gamma f\,\dd z =
F(\gamma(b)) - F(\gamma(a))$
من أجل كل مسار.
\end{proposition}

\begin{proof}
(أ)$\Rightarrow$: $\frac{\dd}{\dd t}F(\gamma(t)) =
F'(\gamma(t))\gamma'(t)$ (بقاعدة السلسلة، وهي صحيحة بالقطع)،
ومنه يتلسكب التكامل إلى فرق الطرفين؛ وتعطي المسارات المغلقة
$0$. (ب)$\Rightarrow$(أ): نثبّت $z_*$ في مركّبة مترابطة، ونعرّف
$F(z) = \int_{\gamma_z}f\,\dd z$ على امتداد أي مسار من $z_*$ إلى $z$ (وهو معرَّف جيدًا:
إذ يختلف مساران بمسار مغلق)؛ ومن أجل $h$ صغير، بأخذ القطعة من
$z$ إلى $z + h$،
\[
\frac{F(z + h) - F(z)}{h} - f(z)
= \frac1h\int_{[z, z+h]}\bigl(f(w) - f(z)\bigr)\dd w
\longrightarrow 0
\]
بمتراجحة $ML$ وباتصال $f$ عند $z$.
\end{proof}

\begin{definition}[عدد اللف]\label{def:b3:holomorphic:index}
من أجل مسار مغلق $\gamma$ ومن أجل $z \notin
\operatorname{im}\gamma$، يكون
\emph{الدليل}\index{عدد اللف}\index{دليل مسار}
\[
\operatorname{Ind}_\gamma(z) = \frac1{2\iu\pi}
\int_\gamma\frac{\dd w}{w - z} .
\]
وهو \emph{عدد صحيح}: فبوضع $\varphi(t) =
\int_a^t\frac{\gamma'(s)}{\gamma(s) - z}\dd s$، يكون للدالة $(\gamma(t) - z)\eu^{-\varphi(t)}$ مشتقة
معدومة (بالقطع)، ومنه فهي ثابتة؛ وعند $t = b$ يكون $\eu^{\varphi(b)} =
\frac{\gamma(b) - z}{\gamma(a) - z} = 1$، ومنه
$\varphi(b) \in
2\iu\pi\Z$. وبوصفه دالةً في $z$، يكون الدليل \omterm{def:b3:topology:continuity}{متصلًا} على $\C\setminus\operatorname{im}\gamma$
(بالتقارب المهيمن)، ومنه ثابتًا على كل مركّبة مترابطة، ومعدومًا
على المركّبة غير المحدودة (بمتراجحة $ML$: إذ يؤول التكامل إلى
$0$ حين $z \to
\infty$). ومن أجل الدائرة $\gamma(t) = a + r\eu^{\iu t}$، $t
\in \intcc0{2\pi}$: أي $\operatorname{Ind}_\gamma(z) = 1$ من
أجل $z
\in D(a,r)$ (بالحساب عند $z = a$: $\frac1{2\iu\pi}
\int_0^{2\pi}\frac{r\iu\eu^{\iu t}}{r\eu^{\iu t}}\dd t = 1$؛ ويتكفّل الثبات
بالباقي).
\end{definition}

\section{مبرهنة كوشي}

\begin{theorem}[غورسا]\label{thm:b3:holomorphic:goursat}
لتكن $f \in \mathcal H(\Omega)$ ولتكن $T \subseteq \Omega$ مثلثًا مصمتًا مغلقًا. عندئذٍ $\int_{\partial T}f\,\dd z = 0$
(بجريان الحافة مرة واحدة، بأي توجيه).\index{مبرهنة غورسا}
\end{theorem}

\begin{proof}
لتكن $I(T) = \int_{\partial T}f\,\dd z$. يشطر وصلُ منتصفات الأضلاع المثلثَ $T$ إلى أربعة
مثلثات نصفية الحجم $T^{(1)},
\dots, T^{(4)}$، وتتلاشى الحواف الداخلية مثنى مثنى:
$I(T) =
\sum_iI(T^{(i)})$. ونختار من بينها $T_1$ الذي يحقق $\abs{I(T_1)} \geq \frac14\abs{I(T)}$، ونكرّر: فنجد
متتالية متداخلة $T \supseteq T_1 \supseteq T_2 \supseteq\cdots$ تحقق
\[
\abs{I(T_n)} \geq 4^{-n}\abs{I(T)},
\qquad
\operatorname{diam}T_n = 2^{-n}\operatorname{diam}T,
\quad
\operatorname{length}(\partial T_n) =
2^{-n}\operatorname{length}(\partial T).
\]
والتقاطع $\bigcap T_n$ نقطةٌ واحدة $z_0$ (بالمتراصات المتداخلة
ذات الأقطار المتلاشية، \cref{thm:b3:topology:compactprops}(3)). وأما القابلية للاشتقاق
عند $z_0$: فبإعطاء $\varepsilon$، ومن أجل $n$ كبير، على $T_n$،
\[
f(z) = f(z_0) + f'(z_0)(z - z_0) + R(z),
\qquad \abs{R(z)} \leq \varepsilon\abs{z - z_0} \leq
\varepsilon\operatorname{diam}T_n .
\]
وللجزء التآلفي دالةٌ أصلية: ومنه ينعدم تكامله على الحافة
المغلقة $\partial T_n$ (\cref{prop:b3:holomorphic:primitive})، فيبقى
\[
\abs{I(T_n)} = \Bigl|\int_{\partial T_n}R\Bigr|
\leq \varepsilon\operatorname{diam}(T_n)\,
\operatorname{length}(\partial T_n)
= \varepsilon\,4^{-n}\operatorname{diam}(T)
\operatorname{length}(\partial T) .
\]
وبالمقارنة مع $\abs{I(T_n)} \geq 4^{-n}\abs{I(T)}$: $\abs{I(T)} \leq \varepsilon\cdot\text{ثابت}$ من أجل كل $\varepsilon$: ومنه
$I(T) = 0$.
\end{proof}

\begin{theorem}[مبرهنة كوشي، الصيغة النجمية]
\label{thm:b3:holomorphic:cauchy}
لتكن $\Omega$ \emph{نجمية الشكل} بالنسبة إلى $c$ (أي إن كل
قطعة $[c, z]$ حيث $z \in \Omega$ تقع في $\Omega$) --- مثل
المحدّبة. عندئذٍ لكل $f \in \mathcal H(\Omega)$ دالةٌ أصلية على $\Omega$؛ ومن ثَمّ
$\int_\gamma f\,\dd z = 0$ من أجل \emph{كل} مسار مغلق $\gamma$ في
$\Omega$.\index{مبرهنة كوشي}
\end{theorem}

\begin{proof}
نعرّف $F(z) = \int_{[c,z]}f\,\dd w$. ومن أجل $z, z + h \in
\Omega$ مع $[z, z+h] \subseteq \Omega$ (وهو صحيح من أجل $h$
صغير)، يقع المثلث ذو الرؤوس $c, z, z+h$ في $\Omega$ (بالنجمية:
إذ تقع كل نقطة منه على قطعة $[c,
w]$ مع $w \in [z, z+h] \subseteq \Omega$): فتعطي غورسا
\[
F(z + h) - F(z) = \int_{[z, z+h]}f\,\dd w,
\]
ويعطي حساب نسبة الفرق في \cref{prop:b3:holomorphic:primitive} أن $F' = f$. وينتج انعدام
تكاملات المسارات المغلقة من القضية نفسها.
\end{proof}

\begin{theorem}[صيغة كوشي التكاملية]
\label{thm:b3:holomorphic:formula}
لتكن $f \in \mathcal H(\Omega)$ و $\bar D(a, r) \subseteq
\Omega$، وليكن $C_r$ الدائرةَ $\partial D(a,r)$ مجراةً مرة
واحدة عكس عقارب الساعة. عندئذٍ يكون من أجل كل
$z \in D(a, r)$:\index{صيغة كوشي التكاملية}
\[
f(z) = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - z}\,\dd w .
\]
\end{theorem}

\begin{proof}
نثبّت $z$ ونعرّف على $\Omega$
\[
g(w) = \begin{cases}
\dfrac{f(w) - f(z)}{w - z} & w \neq z,\\[4pt]
f'(z) & w = z :
\end{cases}
\]
فتكون $g$ متصلة على $\Omega$ \omterm{def:b3:holomorphic:holo}{وهولومورفية} خارج $z$. وتصح غورسا
من أجل $g$ على كل مثلث $T \subseteq \Omega'$، حيث $\Omega'$ قرصٌ أكبر قليلًا
من $\bar D(a,r)$ داخل $\Omega$: فإذا كان $z \notin T$، فمباشرةً؛
وإذا كان $z \in T$، شطرنا $T$ إلى مثلثات صغيرة رأسُها $z$ مع
مثلثات تتفادى $z$؛ وعلى مثلث رأسه $z$ يعطي حدّ $ML$ أن $\abs{\int_{\partial T'}g} \leq \sup_{T'}\abs
g\cdot\operatorname{length} \to 0$
حين يتقلّص المثلث، وتنعدم القطع الباقية بغورسا --- ومنه $\int_{\partial T}g = 0$
في جميع الحالات. ولم يستعمل برهان \cref{thm:b3:holomorphic:cauchy} سوى خاصية المثلث
هذه: فتقبل $g$ دالة أصلية على المحدّبة $\Omega'$، ومنه $\int_{C_r}g = 0$،
أي
\[
\frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{w - z}\dd w
= f(z)\,\frac{1}{2\iu\pi}\int_{C_r}\frac{\dd w}{w - z}
= f(z)\operatorname{Ind}_{C_r}(z) = f(z) .
\]
\end{proof}

\section{التحليلية وسلسلتها المتتابعة}

\begin{theorem}[الهولومورفية $=$ التحليلية]
\label{thm:b3:holomorphic:analytic}
لتكن $f \in \mathcal H(\Omega)$ و $D(a, R) \subseteq \Omega$. عندئذٍ\index{تحليلية الدوال الهولومورفية}
\[
f(z) = \sum_{n\geq0}c_n\,(z - a)^n \quad \text{على} D(a, R),
\qquad
c_n = \frac1{2\iu\pi}\int_{C_r}\frac{f(w)}{(w -
a)^{n+1}}\,\dd w \ \ (0 < r < R),
\]
والمعاملات لا تتعلق بنصف القطر $r$. ومن ثَمّ تكون $f$ قابلة للاشتقاق
بالمعنى $\C$ عددًا لانهائيًّا من المرات، و $c_n = f^{(n)}(a)/n!$، وتصح
\emph{تقديرات كوشي}\index{تقديرات كوشي}:
\[
\abs{c_n} \;\leq\; \frac{\sup_{\abs{w - a} =
r}\abs{f(w)}}{r^{n}} .
\]
\end{theorem}

\begin{proof}
من أجل $\abs{z - a} < r$: ننشر نواة كوشي في المتسلسلة الهندسية
\[
\frac1{w - z} = \frac1{(w - a)\bigl(1 - \frac{z - a}{w -
a}\bigr)} = \sum_{n\geq0}\frac{(z - a)^n}{(w - a)^{n+1}},
\]
المتقاربة تقاربًا ناظميًّا في $w$ على $C_r$ ($\abs{\frac{z-a}{w-a}} =
\frac{\abs{z-a}}r < 1$): ثم نكامل
حدًّا حدًّا مقابل $\frac{f(w)}{2\iu\pi}$ (ويبرّر التقارب المنتظم التبديلَ)
ونطبّق \cref{thm:b3:holomorphic:formula}. ومتسلسلة القوى قابلة للاشتقاق عددًا
لانهائيًّا من المرات مع $c_n =
f^{(n)}(a)/n!$ (السنة الجامعية 2)، وهو يبيّن
أيضًا أن $c_n$ لا تتعلق بنصف القطر $r$. وأما التقديرات: فبحدّ تكامل
المعامل بمتراجحة $ML$.
\end{proof}

\begin{example}[الشذوذات تملي أنصاف الأقطار]
\label{ex:b3:holomorphic:radius}
لماذا يكون لمتسلسلة تايلور للدالة الحقيقية البريئة $\frac1{1 + x^2}$ عند
$x = 3$ تقاربٌ من أجل $\abs{x - 3} <
\sqrt{10}$ فقط، مع أن شيئًا لا يعتلّ على
المستقيم الحقيقي؟ لأن المبرهنة أعلاه تجعل نصف قطر التقارب عند
$a$ مساويًا للمسافة من $a$ إلى أقرب نقطة تخفق عندها
\omterm{def:b3:holomorphic:holo}{الهولومورفية}. وهنا تكون $f(z) = \frac1{1 + z^2}$ \omterm{def:b3:holomorphic:holo}{هولومورفية} على $\C\setminus\{\pm\iu\}$ بالضبط،
ومنه يتقارب النشر عند $a = 3$ على أكبر قرص يتفادى $\pm\iu$،
ونصف قطره $\abs{3 - \iu} = \sqrt{10}$ --- ولا يمكن أن يتقارب على قرص أكبر، إذ كان
المجموع سيمدّد $f$ هولومورفيًّا إلى \omterm{def:b3:topology:topology}{جوار} للنقطتين $\pm\iu$ حيث
$\abs f \to \infty$. فالنظرية الحقيقية ترى نصف القطر الغامض $\sqrt{10}$؛
والمستوي العقدي يرى قطبين. وهذه هي القاعدة العملية:
\emph{لتجد نصف قطر تقارب، عيّن الشذوذات} --- فمثلًا لمتسلسلة
تايلور للدالة $\tan$ عند $0$ نصفُ القطر $\frac\pi2$ (أي أقرب
جذور $\cos$)، وللدالة المولّدة لأعداد برنولي $\frac z{\eu^z - 1}$
(\cref{pb:b3:holomorphic:1}، الجزء السادس) نصفُ القطر $2\pi$ (أي أقرب جذور غير
معدومة للمقدار $\eu^z - 1$: $\pm2\iu\pi$).
\end{example}

\begin{corollary}[ليوفيل؛ دالمبير--غاوس]
\label{cor:b3:holomorphic:liouville}
كل دالة صحيحة محدودة ثابتة. ومن ثَمّ فلكل كثير حدود غير ثابت
على $\C$ جذرٌ: أي إن \emph{$\C$ مغلق
جبريًّا}.\index{مبرهنة ليوفيل}\index{المبرهنة الأساسية في الجبر}
\end{corollary}

\begin{proof}
إذا كان $\abs f \leq M$ على $\C$: فمن أجل كل $a$ و $r$، $\abs{c_1(a)}
= \abs{f'(a)} \leq M/r \to 0$:
ومنه $f' \equiv 0$، وتكون $f$ ثابتة (على $\C$ المترابط: إذ
المشتقة المعدومة تستلزم الثبات محليًّا --- بالمكاملة على
امتداد القطع). وإذا لم يقبل $P$ أي جذر، لكانت $1/P$ صحيحة
ومحدودة ($\abs{P(z)} \to
\infty$ حين $\abs z \to \infty$: إذ يهيمن الحدّ الرئيسي، ومنه
يكون $\abs{1/P}$ صغيرًا خارج قرص كبير \omterm{def:b3:topology:continuity}{ومتصلًا} على القرص
المتراص): فتكون ثابتة --- وهو محال من أجل $P$ غير ثابت. (وتعطي
مسألة نهاية الأسبوع برهانًا ثانيًا ابتدائيًّا والنتائج
الجبرية.)
\end{proof}

\begin{theorem}[الجذور معزولة؛ مبرهنة التطابق]
\label{thm:b3:holomorphic:identity}
لتكن $\Omega$ \emph{مترابطة} ولتكن $f \in \mathcal
H(\Omega)$ مع
$f \not\equiv 0$. عندئذٍ يكون لكل جذر $a$ للدالة $f$
\emph{رتبةٌ} منتهية: $f(z) = (z - a)^m\,g(z)$ مع $g \in
\mathcal H(\Omega)$ و $g(a) \neq 0$، وليس
لجذور $f$ أي نقطة تراكم في $\Omega$. ومن ثَمّ، إذا توافقت
دالتان هولومورفيتان على $\Omega$ على مجموعة لها نقطة تراكم في
$\Omega$، توافقتا في كل مكان.\index{مبرهنة التطابق}\index{جذور
معزولة}
\end{theorem}

\begin{proof}
لتكن $Z$ مجموعة النقاط التي تنعدم عندها جميع مشتقات $f$. تكون
$Z$ مغلقة (لأنها تقاطع مجموعات مغلقة) ومفتوحة: إذ إذا كانت
جميع $c_n = 0$ عند $a$، جعل نشرُ متسلسلة القوى $f
\equiv 0$ على قرص
حول $a$. وبالترابط: $Z =
\varnothing$ أو $Z = \Omega$؛ والثانية مستبعدة
بحكم $f
\not\equiv 0$. ومنه، عند جذر $a$، يكون معاملٌ ما غير معدوم:
وليكن $m$ الأصغرَ الذي يحقق $c_m \neq 0$؛ عندئذٍ $f(z) = (z -
a)^m\sum_{n\geq m}c_n(z-a)^{n-m}$ على
قرص، ويعرّف المجموع دالةً $g$ \omterm{def:b3:holomorphic:holo}{هولومورفية} بجوار $a$ مع $g(a) = c_m \neq 0$؛
ونمدّد $g = f/(z-a)^m$ خارج $a$ (وهي \omterm{def:b3:holomorphic:holo}{هولومورفية} هناك). وبما أن
$g(a) \neq 0$ و $g$ متصلة، لا تقبل $f$ أي جذر آخر في جوار
للنقطة $a$: \omterm{thm:b3:holomorphic:identity}{فالجذور معزولة}، وليس لمجموعة نقاط معزولة أي نقطة
تراكم في $\Omega$ (إذ نقطة تراكم للجذور جذرٌ --- بالاتصال ---
ولن تكون معزولة). وأما التطابق: فبالتطبيق على الفرق، الذي
لمجموعة جذوره نقطةُ تراكم، فيُدفع إلى الفرع $Z = \Omega$.
\end{proof}

\begin{theorem}[القيمة المتوسطة والطويلة العظمى]
\label{thm:b3:holomorphic:maximum}
لتكن $f \in \mathcal H(\Omega)$.
\begin{enumerate}
  \item (القيمة المتوسطة) من أجل $\bar D(a, r) \subseteq \Omega$: $f(a) = \frac1{2\pi}\int_0^{2\pi}f(a +
        r\eu^{\iu t})\,\dd t$.
  \item (مبدأ الأعظمية) إذا كانت $\Omega$ مترابطة وبلغت
        $\abs f$ قيمةً عظمى محلية عند نقطة ما من $\Omega$،
        فإن $f$ ثابتة. ومن ثَمّ، من أجل $\Omega$ محدودة و $f$
        متصلة على $\bar\Omega$: $\sup_{\bar\Omega}\abs f =
        \sup_{\partial\Omega}\abs f$.\index{مبدأ الأعظمية}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) هي صيغة كوشي عند المركز: بوسطنة $C_r$.
(2) لنقل إن $\abs f \leq \abs{f(a)}$ على $\bar D(a, \rho)
\subseteq \Omega$. إذا كان $f(a) = 0$، فإن
$f \equiv 0$ بجوار $a$. وإلا، فمن أجل $0 < r \leq \rho$، تعطي القيمة
المتوسطة
\[
\abs{f(a)} \leq \frac1{2\pi}\int_0^{2\pi}\abs{f(a +
r\eu^{\iu t})}\,\dd t \leq \abs{f(a)} :
\]
وللمقدار المكامَل المتصل غير السالب $\abs{f(a)} - \abs{f(a +
r\eu^{\iu t})}$ متوسطٌ معدوم، ومنه
ينعدم: أي إن $\abs f$ \emph{ثابتة} $= \abs{f(a)} \ne 0$ على القرص. ودالةٌ
\omterm{def:b3:holomorphic:holo}{هولومورفية} ذات طويلة ثابتة غير معدومة على قرص تكون ثابتة:
فباشتقاق $P^2 + Q^2 = \text{ثابت}$ نجد $PP_x + QQ_x
= 0$ و $PP_y + QQ_y = 0$؛ وبإدخال علاقتَي
كوشي--ريمان $P_y = -Q_x$ و $Q_y = P_x$ في المعادلة الثانية نجد
الجملة الخطية
\[
P\,P_x + Q\,Q_x = 0, \qquad -P\,Q_x + Q\,P_x = 0,
\]
ومحددها $P^2 + Q^2 \neq 0$: ومنه $P_x = Q_x = 0$، أي $f' = P_x + \iu Q_x = 0$ على القرص:
فتكون $f$ ثابتة هناك. وتنشر مبرهنة التطابق الثباتَ على $\Omega$
كله. وأما الصيغة الحدّية: فتبلغ $\abs f$ سوپريمومها على
المتراصة $\bar\Omega$؛ وقيمةٌ عظمى داخلية تجعل $f$ ثابتة،
ويكون السوپريموم مبلوغًا على الحافة في كل حالة.
\end{proof}

\begin{theorem}[مبرهنة فايرشتراس في التقارب]
\label{thm:b3:holomorphic:weierstrassconv}
إذا تقاربت $f_n \in \mathcal H(\Omega)$ إلى $f$ بانتظام على كل جزء متراص من
$\Omega$، فإن $f \in \mathcal
H(\Omega)$ و $f_n^{(k)} \to f^{(k)}$ بانتظام على المتراصات، من أجل
كل $k$.\index{مبرهنة فايرشتراس في التقارب}
\end{theorem}

\begin{proof}
تكون $f$ متصلة. ومن أجل أي مثلث مغلق $T \subseteq
\Omega$: $\int_{\partial T}f = \lim\int_{\partial T}f_n = 0$ (بالتقارب
المنتظم على المتراصة $\partial T$؛ وبغورسا من أجل $f_n$). وبحجة
\cref{thm:b3:holomorphic:cauchy}، تقبل $f$ دوالًّا أصلية محلية (إذ الأقراص محدّبة؛
ولم تُستعمل سوى خاصية المثلث)، أي $f = F'$ محليًّا مع $F$
\omterm{def:b3:holomorphic:holo}{هولومورفية}؛ و $F$ تحليلية (\cref{thm:b3:holomorphic:analytic})، ومنه كذلك $f = F'$:
أي \omterm{def:b3:holomorphic:holo}{هولومورفية}. (وهذه هي \emph{مبرهنة موريرا}\index{مبرهنة
موريرا}: أن الاتصال مع انعدام تكاملات المثلثات يستلزم
\omterm{def:b3:holomorphic:holo}{الهولومورفية}.) وأما المشتقات: فمن أجل $\bar D(a, 2r) \subseteq \Omega$ و $z \in \bar D(a, r)$، تعطي
صيغة كوشي للمشتقات (بالاشتقاق تحت علامة التكامل في \cref{thm:b3:holomorphic:formula}،
أو باستعمال صيغة المعاملات)
\[
\abs{f_n'(z) - f'(z)} =
\Bigl|\frac{1}{2\iu\pi}\int_{C_{2r}}\frac{f_n(w) - f(w)}{(w -
z)^2}\,\dd w\Bigr|
\leq \frac{2r\,\sup_{C_{2r}}\abs{f_n - f}}{r^2} \to 0
\]
بانتظام على $\bar D(a,r)$؛ ثم نغطي متراصةً بعدد منتهٍ من
الأقراص كهذه، ونكرّر من أجل $k$ الأعلى.
\end{proof}

\begin{method}\label{met:b3:holomorphic:toolkit}
العدّة اليومية. لتبرهن على أن دالةً \omterm{def:b3:holomorphic:holo}{هولومورفية}: أبرزها متسلسلةَ
قوى، أو مركّبةً، أو نهايةً منتظمة محليًّا (\cref{thm:b3:holomorphic:weierstrassconv})، أو
تكاملًا بوسيط هولومورفي (\cref{exo:b3:holomorphic:7} --- بالاشتقاق تحت علامة
$\int$ أو بتطبيق موريرا--فوبيني). ولتبرهن على متطابقات: أثبتها
على قطعة أو مجال جزئي واستدعِ مبرهنة التطابق. ولتحدّ: \omterm{thm:b3:holomorphic:analytic}{تقديرات
كوشي} على أكبر دائرة متاحة. ولتبرهن على الثبات أو عدم الوجود:
ليوفيل أو مبدأ الأعظمية. واعرف دائمًا \emph{أين} تكون دالتك
\omterm{def:b3:holomorphic:holo}{هولومورفية} و\emph{أي} الأقراص يسع $\Omega$.
\end{method}

(a) عند أي النقاط تكون $z \mapsto \bar z$ و $\abs z^2$ و $\operatorname{Re}z$ قابلةً

(b) برهن على أن $P(x, y) = x^2 - y^2$ هي الجزء الحقيقي \omterm{def:b3:holomorphic:holo}{لدالة هولومورفية} على

\begin{exercise}[$\star$]\label{exo:b3:holomorphic:2}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:3}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:4}

(b) لتكن $f$ صحيحة مع $\operatorname{Re}f$ محدودًا من الأعلى. برهن على أن

(c) استنتج «بيكار الصغرى من أجل التطبيقات التآلفية»: أن دالةً

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:5}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:6}
لتكن $f$ \omterm{def:b3:holomorphic:holo}{هولومورفية} على القرص الواحدي المفتوح $\mathbb D$،

(a) إذا لم تقبل $f$ أي جذر في $\mathbb D$، برهن على أن $f$

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:7}

قابلة \omterm{def:b3:measure:measure}{للقياس} في $x$، و $\abs g \leq h(x)$ مع $h$ \omterm{def:b3:lebesgue:l1}{قابلة للمكاملة}، بانتظام
محليًّا في $z$. برهن على أن $G(z) = \int_Xg(x, z)\dd\mu(x)$ \omterm{def:b3:holomorphic:holo}{هولومورفية} على $\Omega$.
\emph{(بموريرا: إذ تنعدم تكاملات المثلثات بفوبيني وغورسا؛
والاتصال بالتقارب المهيمن. ثم طبّق ذلك على $\Gamma(z) =
\int_0^\infty t^{z-1}\eu^{-t}\dd t$ على
$\{\operatorname{Re}z >
0\}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:8}
(غاوس--لوكاس) ليكن $P \in \C[X]$ غير ثابت. برهن على أن كل جذر
للمقدار $P'$ يقع في الغلاف المحدّب لجذور $P$. \emph{(اكتب
$\frac{P'}{P} = \sum_k\frac{m_k}{z - a_k}$ عند جذر $z$ للمقدار $P'$ ليس جذرًا للمقدار $P$، وخذ
المرافقات، واقرأ تركيبةً محدّبة.)} ووضّح ذلك على $P = z^3 - 1$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:9}
لتكن $f$ صحيحة ومزدوجة الدورية: $f(z + 1) = f(z + \iu)
= f(z)$ من أجل كل $z$. برهن على
أن $f$ ثابتة. \emph{(حُدّ $f$ على المربّع الأساسي المتراص، ثم

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:10}
(a) برهن على أن $P = \operatorname{Re}f$ \omterm{def:b3:holomorphic:holo}{لدالة هولومورفية} $f$ يحقق خاصية القيمة
المتوسطة $P(a) =
\frac1{2\pi}\int_0^{2\pi}P(a + r\eu^{\iu t})\dd t$ وأنه \emph{توافقي}: $\partial^2_{xx}P + \partial^2_{yy}P = 0$ (باشتقاق

(b) استنتج مبدأ الأعظمية من أجل الأجزاء الحقيقية للدوال

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:holomorphic:11}
(انعكاس شوارتز) لتكن $\Omega^+ = \{z : \abs z < 1,\
\operatorname{Im}z > 0\}$ و $I = \intoo{-1}1$، ولتكن $f$ \omterm{def:b3:holomorphic:holo}{هولومورفية}
على $\Omega^+$، ومتصلة على $\Omega^+\cup I$، و\emph{ذات قيم
حقيقية على $I$}. نعرّف
\[
F(z) = \begin{cases} f(z) & z \in \Omega^+\cup I,\\
\overline{f(\bar z)} & \bar z \in \Omega^+ . \end{cases}
\]
(a) برهن على أن $F$ معرَّفة جيدًا ومتصلة على $\Omega = \Omega^+\cup I\cup\Omega^-$،
\omterm{def:b3:holomorphic:holo}{وهولومورفية} على $\Omega^\pm$ \emph{(من أجل $\Omega^-$: تحقق من
كوشي--ريمان من أجل $\overline{f(\bar z)}$، أو انشر $f$ في متسلسلات قوى محلية
وارفق المعاملات)}.
(b) برهن على أن $F$ \omterm{def:b3:holomorphic:holo}{هولومورفية} على $\Omega$ كله بمحك موريرا:
أي $\int_{\partial T}F = 0$ من أجل كل مثلث $T \subseteq \Omega$ \emph{(اشطر المثلثات عند $I$
وادفع أضلاعها الأفقية عن المحور بمقدار $\varepsilon$، باستعمال
الاتصال المنتظم)}.
(c) استنتج: أن دالةً \omterm{def:b3:holomorphic:holo}{هولومورفية} على القرص، حقيقيةً على قطر،
تحقق $f(\bar z) = \overline{f(z)}$؛ وأن دالةً \omterm{def:b3:holomorphic:holo}{هولومورفية} غير ثابتة لا يمكن أن تكون
ذات قيم حقيقية على أي جزء مفتوح غير خالٍ من مجالها (المترابط).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:holomorphic:12}
(معادلة فيثاغورس العقدية) جد جميع أزواج الدوال \emph{الصحيحة}
التي تحقق $f^2 + g^2 = 1$.
(a) برهن على أن $h = f + \iu g$ صحيحة وعديمة الجذور، وعلى أن
كل دالة صحيحة عديمة الجذور هي $\eu^{\varphi}$ من أجل $\varphi$
صحيحة ما \emph{(إذ $h'/h$ صحيحة، ومنه لها دالة أصلية على $\C$
النجمي؛ اضبط الثابت وبرهن على أن $h\eu^{-\varphi}$ ثابتة)}.
(b) اخلص إلى $f = \cos\varphi$ و $g = \sin\varphi$ مع $\varphi$ صحيحة، وتحقق من
العكس. وما الحلول الصحيحة للمعادلة $f^2 + g^2 = 0$؟
\end{exercise}

\section{مسألة: المبرهنة الأساسية في الجبر، مرتين}

\begin{problem}[{مسألة نهاية الأسبوع --- $\C$ مغلق جبريًّا:
برهان دالمبير، وبرهان ليوفيل، والحصاد}]\label{pb:b3:holomorphic:1}
ليكن $P(z) = z^n + a_{n-1}z^{n-1} + \dots + a_0$ مع $n \geq 1$. نبرهن مرتين على أن $P$ يقبل جذرًا، ثم
نجمع ما ينتظره الجبر منذ \cref{ch:b3:galois}.

\medskip
\noindent\textbf{الجزء الأول --- القسرية والأصغرية.}
\begin{enumerate}
  \item برهن على أن $\abs{P(z)} \to +\infty$ حين $\abs z \to
        \infty$: وبالضبط، $\abs{P(z)} \geq \frac12\abs z^n$ من
        أجل $\abs z \geq R_0$ مناسبين.
  \item استنتج أن $\abs P$ تبلغ قيمةً صغرى شاملة على $\C$: أي
        يوجد $z_0$ يحقق $\abs{P(z_0)} = \inf_\C\abs P$ \emph{(بتراص قرص مغلق كبير،
        \cref{cor:b3:topology:heineborel})}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- هبوط دالمبير.} لنفترض،
للتناقض، أن $P(z_0) \neq 0$.
\begin{enumerate}[resume]
  \item انشر $Q(h) = P(z_0 + h)/P(z_0)$ بوصفه كثير حدود في $h$: $Q(h) = 1 + c_kh^k + h^{k+1}S(h)$ مع $c_k \neq
        0$
        حيث $k \geq 1$ و $S$ كثير حدود.
  \item اختر اتجاه الهبوط: من أجل $t > 0$ صغير، ضع
        $h = t\,\omega$ حيث $\omega^k = -1/c_k$ (ولماذا يوجد $\omega$ كهذا؟
        --- \emph{برهن} على وجود الجذور من الرتبة $k$ لأي عدد
        عقدي عبر الصيغة القطبية، مستقلًّا عن المبرهنة
        المبرهَن عليها). برهن على
        \[
        \abs{Q(t\omega)} \leq 1 - t^k + C\,t^{k+1}
        \]
        من أجل $t$ صغير، بثابت $C$ صريح.
  \item اخلص إلى $\abs{Q(t\omega)} < 1$ من أجل $t$ صغير --- وهو يناقض أصغرية
        $\abs{P(z_0)}$. ومنه $P(z_0) = 0$: أي إن \emph{لكل
        كثير حدود عقدي غير ثابت جذرًا} (دالمبير--أرغان).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- سطر ليوفيل الواحد، كاملًا.}
\begin{enumerate}[resume]
  \item اكتب بعناية برهان \cref{cor:b3:holomorphic:liouville}: إذا لم يقبل $P$ أي
        جذر، تحقق من أن $1/P$ صحيحة ومحدودة (وكمِّم ذلك،
        باستعمال السؤال 1)، ومنه ثابتة، واخلص. وقارن
        البرهانين: أي المكوّنات يستعمل كلٌّ منهما؟ (فالتراص
        يظهر في كليهما --- وأين؟)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الحصاد.}
\begin{enumerate}[resume]
  \item برهن على أن كل $P \in \C[X]$ من الدرجة $n$ ينشطر:
        $P = c\prod_{i}(X - \alpha_i)^{m_i}$ مع $\sum m_i =
        n$ \emph{(بالتراجع، وبالقسمة
        الإقليدية على $(X -
        \alpha)$)}.
  \item برهن على أن كثيرات الحدود غير القابلة للاختزال في
        $\R[X]$ هي الخطية والتربيعية ذات المميِّز السالب
        \emph{(بمزاوجة الجذور المرافقة)}؛ واستنتج أن لكل كثير
        حدود حقيقي من درجة فردية جذرًا حقيقيًّا، وأعطِ برهانًا
        ثانيًا ترتيبيًّا على تلك الحقيقة الأخيرة (بمبرهنة
        القيم الوسطى) --- متحققًا من توافقهما على
        $X^3 - X - 1$.
  \item استنتج الديون التي يمكن لهذا الكتاب أن يسدّدها الآن:
        (أ) أن لكل تشاكل داخلي لفضاء متجهي غير معدوم منتهي
        البُعد على $\C$ قيمةً ذاتية، ومنه لكل مصفوفة عقدية
        صيغةُ جوردان (\cref{thm:b3:modules:jordan})؛ (ب) وأن الحقل $\bar\Q$
        للأعداد الجبرية المستعمَل في \cref{rem:b3:galois:closureexamples} هو فعلًا \omterm{def:b3:galois:closure}{غلق
        جبري} للحقل $\Q$.
  \item (الخاتمة) عيّن بالضبط أين كان كل برهان سينكسر على حقل
        مثل $\Q(\iu)$: أي الخطوات تستعمل وجود الجذور من الرتبة
        $k$ (السؤال 4)، وأيها تستعمل التراص أو التمام
        (السؤالان 2 و6)؟ واخلص في خمسة أسطر: أن المبرهنة
        \emph{تحليلية} فعلًا --- إذ يستدعي كل برهان في موضع ما
        تمامَ $\R$ أو ترابطه --- مع أن عبارتها جبرية محضة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- سلّم صرامة الدوال الصحيحة.}
ليوفيل أول درجة في سلّم؛ ونصعده.
\begin{enumerate}[resume]
  \item (\omterm{thm:b3:holomorphic:analytic}{تقديرات كوشي}) انطلاقًا من صيغة كوشي على دائرة نصف
        قطرها $r$ حول $a$، برهن على
        \[
        \bigl|f^{(n)}(a)\bigr| \leq
        \frac{n!\,\sup_{\abs{z-a}=r}\abs f}{r^n} ,
        \]
        واستعد ليوفيل بوصفها الحالة $n = 1$ مع $r \to
        \infty$.
  \item (النمو الحدودي) برهن على أن دالةً صحيحة $f$ تحقق
        $\abs{f(z)} \leq A + B\abs z^m$ من أجل كل $z$ هي كثير حدود درجته $\leq m$
        \emph{(بإلغاء معاملات تايلور وراء $m$ بالسؤال 11)}.
  \item (الجزء الحقيقي المحدود) برهن على أن دالةً صحيحة $f$
        يكون $\operatorname{Re}f$ محدودًا من الأعلى تكون ثابتة \emph{(بتطبيق
        ليوفيل على $\eu^{f}$)}.
  \item (ازدواج الدورية) لتكن $f$ صحيحة مع $f(z + 1)
        = f(z)$ و $f(z + \iu) = f(z)$ من
        أجل كل $z$. برهن على أن $f$ ثابتة. واخلص: أن دالةً
        «إهليلجية» غير ثابتة يجب أن تكون لها شذوذات --- وهو
        السبب التاريخي لدخول الأقطاب في التحليل العقدي.
  \item (المجال الكثيف) برهن على أن مجال دالة صحيحة غير ثابتة
        كثيفٌ في $\C$: فإذا تجاهلت $f(\C)$ قرصًا $D(a, r)$،
        كانت $\frac1{f - a}$ صحيحة ومحدودة. (وقد برهن بيكار
        على أن المجال لا يتجاهل أكثر من \emph{نقطة واحدة}؛
        والكثافة هي المستوى الذي تبلغه أدواتنا.)
  \item (السويّة $\Rightarrow$ كثير حدود) لنفترض أن $f$ صحيحة
        وأن $\abs{f(z)} \to \infty$ حين $\abs z \to
        \infty$. برهن على: أن جذور $f$ منتهية
        العدد ($z_1, \dots, z_p$، بترافقات $m_i$)؛ وأن خارج القسمة
        $g = f/\prod(z - z_i)^{m_i}$ صحيح وعديم الجذور؛ وأن $1/g$ ذو نمو حدودي،
        ومنه (بالسؤال 12) كثير حدود، وهو بالضرورة ثابت (لأنه
        عديم الجذور)؛ واخلص إلى أن $f$ كثير حدود. ومنه فإن
        كثيرات الحدود هي بالضبط الدوال السويّة بين الدوال
        الصحيحة --- إذ تخفق $\eu^z$ في السويّة على امتداد
        $\R_-$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء السادس --- الظلال التوافقية وقيمة
غاوس المتوسطة.}
\begin{enumerate}[resume]
  \item لتكن $f = u + \iu v$ \omterm{def:b3:holomorphic:holo}{هولومورفية} على \omterm{def:b3:topology:topology}{مجموعة مفتوحة}.
        تحقق من أن $u = \operatorname{Re}f$ يحقق خاصية القيمة المتوسطة
        \[
        u(a) = \frac1{2\pi}\int_0^{2\pi}
        u\bigl(a + r\eu^{\iu\theta}\bigr)\,\dd\theta
        \]
        (بأخذ الجزء الحقيقي من صيغة كوشي)، واستنتج مبدأ
        الأعظمية من أجل $u$ على مجال محدود، ببرهان الترابط
        نفسه كما من أجل $\abs f$.
  \item (قيمة غاوس المتوسطة) من أجل $a \in \C$ و $r > 0$ مع
        $\abs a \neq r$، برهن على
        \[
        \frac1{2\pi}\int_0^{2\pi}
        \log\bigl|a - r\eu^{\iu\theta}\bigr|\,\dd\theta
        = \log\max\bigl(\abs a, r\bigr)
        \]
        \emph{(إذا كان $\abs a > r$: فإن $z \mapsto \log\abs{a - z}$ هو الجزء
        الحقيقي للوغاريتم هولومورفي على \omterm{def:b3:topology:topology}{جوار} للقرص المغلق ---
        ولماذا يوجد واحد؟ --- ومنه ينطبق السؤال 17؛ وإذا كان
        $\abs a < r$: فحلّل $\abs{a - r\eu^{\iu\theta}} = r\,\abs{1 -
        \frac ar\eu^{-\iu\theta}}$ واستعمل الحالة الأولى من
        جديد)}.
  \item (\omterm{def:b3:measure:measure}{قياس} ماهلر) من أجل $P = c\prod_{i=1}^n(X -
        \alpha_i) \in \C[X]$، استنتج \emph{صيغة ينسن
        من أجل كثيرات الحدود}:
        \[
        \frac1{2\pi}\int_0^{2\pi}\log\bigl|P(\eu^{\iu\theta})
        \bigr|\,\dd\theta
        = \log\Bigl(\abs c\prod_{i}\max(1,
        \abs{\alpha_i})\Bigr) :
        \]
        أي إن الوسط الهندسي للمقدار $\abs P$ على الدائرة
        الواحدية يقرأ الجذور الواقعة خارج القرص. وتحقق من ذلك
        على $P =
        X^2 - X$ وعلى $P = 2X - 1$.
  \item (أعداد برنولي) نعرّف المعاملات $B_n$ بالعلاقة $\frac{z}{\eu^z - 1} = \sum_{n\geq0}\frac{B_n}{n!}
        z^n$
        بجوار $0$ (ولماذا يكون الطرف الأيسر تحليليًّا عند
        $0$؟). استنبط العلاقة التراجعية $\sum_{k=0}^{n}\binom{n+1}kB_k = 0$ (حيث
        $n \geq 1$) من $(\eu^z - 1)\cdot\frac z{\eu^z-1} = z$، واحسب $B_0,
        \dots, B_6$، وبرهن على أن
        $B_{2k+1} = 0$ من أجل $k \geq 1$ \emph{(إذ الدالة
        $\frac z{\eu^z-1} + \frac z2$ زوجية)}. وستسعّر هذه الأعداد كل $\zeta(2k)$ في
        \cref{ch:b3:residues}.
  \item (الحقيقية) برهن على أن دالةً صحيحة تأخذ قيمًا حقيقية
        على $\R$ تحقق $f(\bar z) =
        \overline{f(z)}$ في كل مكان \emph{(بمقارنة معاملات
        تايلور عند $0$، أو بتطبيق مبرهنة التطابق على $z \mapsto \overline{f(\bar z)}$)}؛
        واستنتج من جديد أن الجذور غير الحقيقية لكثيرات الحدود
        الحقيقية تأتي أزواجًا مرافقة (أي مزاوجة السؤال 8،
        مبرهَنًا عليها تحليليًّا).
  \item (العبرة) ركّب سلّم الصرامة: المحدودة $\Rightarrow$
        ثابتة؛ والمحدودة حدوديًّا $\Rightarrow$ كثير حدود؛
        والسويّة $\Rightarrow$ كثير حدود؛ والتي تتجاهل قرصًا
        $\Rightarrow$ ثابتة؛ ومزدوجة الدورية $\Rightarrow$
        ثابتة. وقابل ذلك في فقرة قصيرة بالدوال الحقيقية من
        الصنف $\mathcal C^\infty$ (دوال النتوء، \cref{thm:b3:lp:regularization}): لماذا تفرض
        \omterm{def:b3:holomorphic:holo}{الهولومورفية}، وهي شرط محلي محض، قانونًا ونظامًا
        شاملَين؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء السابع --- الحصاد الأخير.}
\begin{enumerate}[resume]
  \item (متراجحة لانداو) من أجل $P = \sum_{k=0}^na_kX^k$، برهن على القيمة
        المتوسطة $\frac1{2\pi}\int_0^{2\pi}
        \abs{P(\eu^{\iu\theta})}^2\dd\theta =
        \sum_k\abs{a_k}^2$ (بتعامد المقادير $\eu^{\iu k\theta}$)، ثم،
        باستعمال الحدّ النقطي $\log t \leq t - 1$ لمقارنة متوسطَي
        $\log\abs P^2$ و $\abs P^2$، استنتج من السؤال 19 أن
        \[
        \abs c\prod_{i}\max\bigl(1, \abs{\alpha_i}\bigr)
        \;\leq\;
        \Bigl(\sum_{k=0}^{n}\abs{a_k}^2\Bigr)^{1/2} :
        \]
        أي إن جداء الجذور الواقعة خارج القرص الواحدي محكومٌ
        بحجم المعاملات بالمعنى $\ell^2$ (وعالج الجذور الواقعة
        \emph{على} الدائرة بتطبيق المتراجحة على $P(rX)$ ثم
        بجعل $r \to 1$). وتحقق من ذلك على $X^2 - X$.
  \item (أعداد برنولي تنمو مضروبيًّا) برهن على أن نصف قطر تقارب
        $\sum\frac{B_n}{n!}z^n$ هو \emph{بالضبط} $2\pi$: فهو لا يقل عن $2\pi$ لأن
        $z/(\eu^z - 1)$ تمتدّ هولومورفيًّا إلى $D(0, 2\pi)$،
        ولا يزيد على $2\pi$ لأن المجموع كان سيبقى وإلا محدودًا
        بجوار $2\pi\iu$ حيث $\abs{z/(\eu^z-1)} \to \infty$. واستنتج
        \[
        \limsup_{k\to\infty}
        \Bigl(\frac{\abs{B_{2k}}}{(2k)!}\Bigr)^{1/2k}
        = \frac1{2\pi} :
        \]
        أي إن أعداد برنولي تنمو مضروبيًّا. ومسلِّمًا بالمقدار
        $B_{12}
        = -\frac{691}{2730}$ (أي العلاقة التراجعية في السؤال 20، مجراةً
        أبعد)، قارن $\abs{B_{12}} \approx 0.25311$ بالتنبؤ الأدق $2\,(2k)!/(2\pi)^{2k}
        \approx 0.25305$ عند $k = 6$
        --- أي أربعة أرقام متطابقة لقانون مقارب سيبرهن عليه
        \cref{ch:b3:residues} بالضبط، عبر $\zeta(2k)$.
  \item (الجذور تتحرك اتصاليًّا) لتكن $(P_j)$ موحَّدة من الدرجة
        $n$ بمعاملات تتقارب إلى معاملات $P$ (الموحَّد). برهن
        على \emph{حدّ كوشي}: أن كل جذر لكثير حدود موحَّد
        $Q = X^n + \sum_{k<n}q_kX^k$ يحقق $\abs\alpha \leq 1 + \max_k\abs{q_k}$؛ واستنتج أن جذور $P_j$ تبقى في
        مجموعة متراصة ثابتة، وأنه، باستخراج متتاليات جزئية
        متقاربة من متجهات الجذور والمرور إلى النهاية في تحليل
        السؤال 7، تتقارب متعددات مجموعات جذور $P_j$ إلى مجموعة
        جذور $P$. وبرهن أخيرًا على أن الاتصال أقصى ما يمكن
        قوله: فمن أجل $P_\varepsilon = X^2 - 2X + 1 +
        \varepsilon$، يحرّك اضطرابٌ حجمه $\varepsilon =
        10^{-4}$
        الجذرَ المضاعف $1$ بمقدار $10^{-2}$ --- أي أُسّ هولدر
        $\frac1m$ عند جذر من الترافق $m$، ولا ليبشيتزية أبدًا:
        فعدديًّا، تكلّف الجذور المضاعفة نصف الأرقام.
\end{enumerate}
\end{problem}
