\chapter{الحلقات والحساب}\label{ch:b3:rings}

تتفكّك الأعداد الصحيحة العادية إلى أعداد أولية بطريقة وحيدة؛ وكذلك
تفعل كثيرات الحدود على حقل. فهل هاتان الحقيقتان مبرهنة واحدة؟
يجيب هذا الفصل بنعم، ويعيّن الفرضيات الدقيقة التي تجعل «الحساب»
ممكنًا في حلقة تبديلية: أي السلسلة
\[
\text{إقليدية} \;\Longrightarrow\; \text{رئيسية}
\;\Longrightarrow\; \text{تفكيكية},
\]
مع البرهان على كل الاستلزامات ونقض كل العكوس. ثم تُختبر النظرية
حيث تثبت جدواها: أعداد غاوس الصحيحة $\Z[\iu]$ (التي ستفكّ لغز
مبرهنة فيرما لمجموع مربعين في مسألة نهاية الأسبوع)، وحلقات
كثيرات الحدود بعدة متغيرات (مبرهنة غاوس المساعدة، ومحك
\omterm{thm:b3:rings:criteria}{أيزنشتاين})، والحلقات \omterm{def:b3:rings:noetherian}{النويثرية}، وصولًا إلى مبرهنة الأساس لهيلبرت.
ونعني بكلمة \emph{حلقة}، في كل ما يلي، حلقةً تبديلية ذات عنصر
واحدي $1 \neq 0$؛ ومثالا $\Z$ و $K[X]$ اللذان وردا في مجلد السنة
الجامعية 2 هما مثالانا الموجّهان.

\section{المثاليات وحلقات القسمة ومبرهنة التماثل}

\begin{definition}\label{def:b3:rings:ideal}
\emph{المثالي}\index{مثالي} $I$ في حلقة $A$ هو زمرة جزئية جمعية
تحقق $AI \subseteq I$. و\emph{حلقة القسمة}\index{حلقة القسمة}
$A/I$ هي \omterm{thm:b3:groups:quotient}{زمرة القسمة} $(A, +)/I$ مزوّدةً بالضرب $(a + I)(b + I) = ab + I$: وهو معرَّف
تعريفًا سليمًا، لأن تبديل $a$ بالمقدار $a + x$ و $b$ بالمقدار $b + y$ (حيث
$x, y \in I$) يغيّر $ab$ بمقدار $ay + xb + xy \in I$. والإسقاط $\pi \colon A
\to A/I$ تشاكل
حلقات شامل نواته $I$، ونوى تشاكلات الحلقات هي بالضبط المثاليات.
\end{definition}

\begin{theorem}[مبرهنة التماثل الأولى]\label{thm:b3:rings:firstiso}
إذا كان $f \colon A \to B$ تشاكل حلقات، فإن $\bar f\colon
A/\ker f \to \operatorname{im} f$، $a + \ker f \mapsto f(a)$، تماثل حلقات.
وبصورة أعمّ، يتحلّل $f$ عبر $A/I$ من أجل أي \omterm{def:b3:rings:ideal}{مثالي} $I \subseteq \ker f$. ومثاليات
$A/I$ هي المثاليات $J/I$ حيث $J \supseteq I$ \omterm{def:b3:rings:ideal}{مثالي} في $A$ (مبرهنة
التقابل).
\end{theorem}

\begin{proof}
كما في حالة الزمر (\cref{thm:b3:groups:firstiso,%
thm:b3:groups:correspondence})، مع ملاحظة أن جميع التطبيقات
المعنية تحفظ الجداءات أيضًا: فالتشاكل $\bar f$ معرَّف تعريفًا
سليمًا، وتقابلي على صورته، وضربي؛ ويحفظ التقابل $J \mapsto J/I$
و $\bar J \mapsto \pi^{-1}(\bar J)$ المثاليات في الاتجاهين لأن $\pi$ تشاكل حلقات شامل.
\end{proof}

\begin{definition}\label{def:b3:rings:primemaximal}
ليكن $I \subsetneq A$ مثاليًا فعليًا. يكون $I$ \emph{أوليًا}
\index{مثالي أولي} إذا كان $ab \in I \Rightarrow a \in I$ أو $b \in I$؛ ويكون $I$
\emph{أعظميًا}\index{مثالي أعظمي} إذا لم يقع أي \omterm{def:b3:rings:ideal}{مثالي} بينه وبين
$A$ وقوعًا تامًّا.
\end{definition}

\begin{proposition}\label{prop:b3:rings:primemaximal}
يكون $I$ \omterm{def:b3:rings:primemaximal}{أوليًا} $\iff$ تكون $A/I$ حلقة تامة؛ ويكون $I$ \omterm{def:b3:rings:primemaximal}{أعظميًا}
$\iff$ تكون $A/I$ حقلًا. وعلى وجه الخصوص، المثاليات الأعظمية
أولية.
\end{proposition}

\begin{proof}
لنكتب $\bar a$ للأصناف في $A/I$. تُترجَم عبارة «$I$ \omterm{def:b3:rings:divisibility}{أولي}» حرفيًا
إلى «$\bar a\bar b = 0 \Rightarrow \bar a = 0$ أو $\bar b
= 0$»، وتُترجَم $A/I \neq 0$ إلى $I \neq A$:
وهذا هو تعريف الحلقة التامة. أما الأعظمية، فنستعمل مبرهنة
التقابل: عدم وجود \omterm{def:b3:rings:ideal}{مثالي} واقع تمامًا بين $I$ و $A$ $\iff$ ليس
في $A/I$ \omterm{def:b3:rings:ideal}{مثالي} غير $0$ ونفسها $\iff$ تكون $A/I$ حقلًا --- ومن أجل
الخطوة الأخيرة: في حقلٍ ليس هناك مثاليات سوى $0$ والحقل كله (لأن
مثاليًا يحتوي على $x \ne 0$ يحتوي على $x^{-1}x = 1$)؛ وبالعكس،
إذا ولّد كل $x$ غير معدوم المثاليَ الواحدي، فإن $xy = 1$ من أجل
$y$ ما. والحقول حلقات تامة، ومنه فالمثاليات الأعظمية أولية.
\end{proof}

\begin{example}\label{ex:b3:rings:primemaximal}
في $\Z$: المثاليات الأولية هي $(0)$ والمثاليات $(p)$ حيث $p$
\omterm{def:b3:rings:divisibility}{أولي}؛ والأعظمية منها هي $(p)$ (لأن $\Z/p\Z = \mathbb F_p$ حقل، بينما
$\Z/(0) = \Z$ ليس كذلك). وفي $K[X, Y]$: المثاليان $(X) \subsetneq (X, Y)$ أوليان
كلاهما (لأن $K[X,Y]/(X) \cong K[Y]$ حلقة تامة، و $K[X,Y]/(X,Y) \cong K$ حقل)، ومنه فإن $(X)$
\omterm{def:b3:rings:divisibility}{أولي} وليس \omterm{def:b3:rings:primemaximal}{أعظميًا}.
\end{example}

ولكي نضمن \emph{وجود} مثاليات أعظمية في العموم التام، نحتاج إلى
مبدأ من مبادئ نظرية المجموعات. نقول عن مجموعة مرتّبة ترتيبًا
جزئيًا إنها \emph{استقرائية} إذا كان لكل جزء مرتّب ترتيبًا كليًا
(\emph{سلسلة}) حدٌّ أعلى.

\begin{theorem}[مبرهنة زورن المساعدة]\label{thm:b3:rings:zorn}
لكل مجموعة مرتّبة جزئيًا غير خالية واستقرائية عنصر
أعظمي.\index{مبرهنة زورن المساعدة}
\admitted
\end{theorem}

\begin{remark}\label{rem:b3:rings:zorn}
هذه ليست مبرهنة من مبرهنات الرياضيات العادية بل \emph{بديهية}:
فهي تكافئ، فوق بديهيات تسيرملو--فرنكل الأساسية لنظرية المجموعات،
بديهيةَ الاختيار («كل جداء لمجموعات غير خالية غير خالٍ»)، وهي
بديهية نقبلها في هذا الكتاب كله. وسننبّه إلى كل استعمال لها.
وسيستدعيها التحليل مرة أخرى (هان--باناخ، \cref{ch:b3:banach}).
\end{remark}

\begin{theorem}[كرول]\label{thm:b3:rings:krull}
كل \omterm{def:b3:rings:ideal}{مثالي} فعلي $I \subsetneq A$ محتوًى في \omterm{def:b3:rings:primemaximal}{مثالي
أعظمي}.\index{مبرهنة كرول}
\end{theorem}

\begin{proof}
لنرتّب بالاحتواء مجموعة $\mathcal E$ المؤلَّفة من المثاليات
الفعلية الحاوية على $I$؛ فهي غير خالية ($I \in \mathcal E$). ولكل سلسلة
$(J_\lambda)$ في $\mathcal E$ حدٌّ أعلى هو $J = \bigcup
J_\lambda$: وهو \omterm{def:b3:rings:ideal}{مثالي}
(لأن أي $a, b \in J$ يقعان في $J_\lambda$ واحد بحكم الترتيب
الكلي)، وفعلي (لأن $1 \notin J_\lambda$ من أجل كل $\lambda$)، وحاوٍ على $I$.
وتعطي مبرهنة زورن المساعدة عنصرًا \omterm{def:b3:rings:primemaximal}{أعظميًا} في $\mathcal
E$، وهو \omterm{def:b3:rings:primemaximal}{مثالي
أعظمي} يحتوي على $I$ (إذ إن مثاليًا فعليًا يقع فوقه تمامًا سيقع
في $\mathcal E$).
\end{proof}

\begin{theorem}[مبرهنة الباقي الصيني]\label{thm:b3:rings:crt}
لتكن $I_1, \dots, I_n$ مثاليات في $A$ \emph{متكاملة}\index{مثاليات متكاملة}
مثنى مثنى ($I_k + I_l = A$ من أجل $k \neq l$). عندئذٍ
\[
A\Big/\bigcap_{k=1}^n I_k \;\xrightarrow{\;\sim\;}\;
\prod_{k=1}^n A/I_k,
\qquad
a \longmapsto (a + I_1, \dots, a + I_n),
\]
وعلاوة على ذلك $\bigcap_k I_k = I_1 I_2 \cdots I_n$ (وهو \omterm{def:b3:rings:ideal}{المثالي} المولَّد بالجداءات).
\index{مبرهنة الباقي الصيني}
\end{theorem}

\begin{proof}
التطبيق $f(a) = (a + I_k)_k$ تشاكل حلقات نواته $\bigcap I_k$؛ ويكفي حسب
\cref{thm:b3:rings:firstiso} أن نبرهن على شموليته. لنثبّت $k$؛ ومن أجل كل
$l \neq k$ لنكتب $1 = u_l +
v_l$ حيث $u_l \in I_k$ و $v_l \in I_l$
(بالتكامل). عندئذٍ
\[
e_k = \prod_{l \neq k} v_l = \prod_{l\neq k}(1 - u_l)
\equiv 1 \pmod{I_k},
\qquad e_k \in I_l \ (l \neq k),
\]
ومنه $f(e_k) = (0, \dots, 1, \dots, 0)$؛ فإذا أُعطي هدفٌ $(a_k +
I_k)_k$، فإن العنصر
$\sum_k a_k e_k$ يُرسَل إليه.

الجداءات في مقابل التقاطع: لدينا $I_1\cdots I_n \subseteq \bigcap I_k$ دائمًا. وبالعكس، يكفي
بالتراجع أن نعالج الحالة $n = 2$ (إذ نتحقق من أن $I_1$
و $I_2\cdots I_n$ متكاملان: فبضرب $1 = u_l + v_l$ على $l \geq 2$
نجد $1 \in I_1 + I_2\cdots
I_n$). ومن أجل $n = 2$: لنكتب $1 = u + v$ حيث $u \in I_1$
و $v \in I_2$؛ فمن أجل $x \in I_1 \cap I_2$ لدينا $x = xu + xv \in I_2I_1 + I_1I_2 =
I_1I_2$.
\end{proof}

\begin{example}\label{ex:b3:rings:crtz}
في $\Z$ مع $I_k = (m_k)$ حيث $m_k$ أولية فيما بينها مثنى مثنى:
$\Z/(m_1\cdots
m_n)\Z \cong \prod \Z/m_k\Z$ --- وهي مبرهنة الباقي الصيني الواردة في مجلد السنة
الجامعية 2. وبالاقتصار على العناصر القابلة للقلب: $(\Z/mn\Z)^\times \cong
(\Z/m\Z)^\times \times (\Z/n\Z)^\times$ من أجل
$\gcd(m,n)=1$، ومن هنا تأتي ضربية دالة أويلر $\varphi$
(\cref{exo:b3:rings:8}).
\end{example}

\section{القابلية للقسمة: إقليدية ورئيسية وتفكيكية}

\begin{definition}\label{def:b3:rings:divisibility}
لتكن $A$ حلقة تامة وليكن $a, b \in A$. نقول إن $a$ \emph{يقسم}
$b$ (ونكتب $a \mid b$) إذا كان $b \in (a) = aA$. ويكون العنصران
$a,
b$ \emph{متصاحبين}\index{عنصران متصاحبان} إذا كان $a = ub$
حيث $u \in A^\times$ (وهذا يكافئ $(a) = (b)$). وليكن $p$ عنصرًا
غير معدوم وغير قابل للقلب؛ نقول إنه:
\begin{itemize}
  \item \emph{غير قابل للاختزال}\index{عنصر غير قابل للاختزال}
        إذا كان $p = ab$ يفرض $a \in A^\times$ أو
        $b \in A^\times$؛
  \item \emph{أولي}\index{عنصر أولي} إذا كان $p \mid ab$ يفرض
        $p
        \mid a$ أو $p \mid b$ (أي إن \omterm{def:b3:rings:ideal}{المثالي} $(p)$ أولي).
\end{itemize}
\end{definition}

\begin{proposition}\label{prop:b3:rings:primeirred}
في أي حلقة تامة: \omterm{def:b3:rings:divisibility}{أولي} $\Rightarrow$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال}. والعكس
خاطئ في العموم: ففي $\Z[\iu\sqrt 5] = \{a + \iu b\sqrt5 : a, b
\in \Z\}$، يكون العنصر $2$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال}
وليس \omterm{def:b3:rings:primemaximal}{أوليًا}.
\end{proposition}

\begin{proof}
ليكن $p$ \omterm{def:b3:rings:primemaximal}{أوليًا} وليكن $p = ab$. عندئذٍ $p \mid ab$، وليكن مثلًا
$p \mid a$: أي $a = pc$، ومنه $p = pcb$، وباختصار $p$ (لأن
الحلقة تامة) نجد $cb =
1$: أي $b \in A^\times$.

وفي $\Z[\iu\sqrt5]$، نستعمل المعيار $N(x + \iu y\sqrt 5) = x^2 +
5y^2$، وهو ضربي (لأنه
$\abs z^2$). فإذا كان $2 = ab$ حيث $a, b$ غير قابلين للقلب، فإن
$4 = N(a)N(b)$ مع $N(a), N(b) \neq
1$ (لأن عناصر المعيار $1$ هي $\pm1$، وهي
العناصر القابلة للقلب)، ومنه $N(a) = 2$: وهذا مستحيل، لأن
$x^2 + 5y^2 = 2$ ليس له حل صحيح. إذن $2$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال}. لكن
$2 \mid 6 = (1 + \iu\sqrt5)(1 - \iu\sqrt5)$ بينما $2$ لا يقسم أيًّا من العاملين ($\frac12 \pm \frac{\iu\sqrt5}2
\notin \Z[\iu\sqrt5]$): فهو ليس
\omterm{def:b3:rings:primemaximal}{أوليًا}.
\end{proof}

\begin{definition}\label{def:b3:rings:pidufd}
تكون الحلقة التامة $A$:
\begin{itemize}
  \item \emph{إقليدية}\index{حلقة إقليدية} إذا وُجد تطبيق $\nu \colon A \setminus \{0\} \to \N$
        (يُسمّى \emph{دالة إقليدية}) بحيث يوجد، من أجل كل
        $a, b$ مع $b \ne 0$، عنصران $q, r$ يحققان $a = bq + r$
        و ($r = 0$ أو $\nu(r) <
        \nu(b)$)؛
  \item \emph{رئيسية} (\emph{حلقة مثاليات رئيسية}\index{حلقة
        مثاليات رئيسية}) إذا كان كل \omterm{def:b3:rings:ideal}{مثالي} من الشكل $(a)$؛
  \item \emph{تفكيكية} (\emph{حلقة تفكيك وحيد}\index{حلقة تفكيك
        وحيد}) إذا كان كل عنصر غير معدوم وغير قابل للقلب جداءَ
        عناصر غير قابلة للاختزال، بطريقة وحيدة إلى حدود الترتيب
        والتصاحب.
\end{itemize}
\end{definition}

\begin{theorem}\label{thm:b3:rings:euclideanpid}
\omterm{def:b3:rings:pidufd}{إقليدية} $\Rightarrow$ رئيسية.
\end{theorem}

\begin{proof}
ليكن $I \neq (0)$ مثاليًا وليكن $b \in I \setminus\{0\}$ بحيث يكون $\nu(b)$
أصغريًا. من أجل $a \in I$ نقسم: $a = bq + r$؛ عندئذٍ $r = a
- bq \in I$،
ولو كان $\nu(r) < \nu(b)$ لناقض ذلك الأصغرية، ومنه $r = 0$ و $a \in (b)$:
أي $I = (b)$.
\end{proof}

\begin{example}\label{ex:b3:rings:euclidean}
الحلقة $\Z$ (مع $\nu = \abs\cdot$) والحلقة $K[X]$ (مع $\nu = \deg$)
إقليديتان --- وقد برهن مجلد السنة الجامعية 2 على القسمتين. وكذلك
$\Z[\iu]$ مع $\nu = N$ المعيارَ المربّع (\cref{exo:b3:rings:4})؛ وهندسة
البرهان مبيَّنة في الشكل أدناه. وتوجد \omterm{def:b3:rings:pidufd}{حلقة مثاليات رئيسية} غير
\omterm{def:b3:rings:pidufd}{إقليدية}، لكن إثبات ذلك دقيق (والمثال المعياري هو $\Z\bigl[\frac{1+\iu\sqrt{19}}2\bigr]$)؛ أما
\omterm{def:b3:rings:pidufd}{حلقة تفكيك وحيد} ليست رئيسية فمثالها سهل: $K[X, Y]$
(\cref{exo:b3:rings:6})، أو $\Z[X]$.
\end{example}

\begin{lemma}[السلاسل المتزايدة من المثاليات الرئيسية]
\label{lem:b3:rings:acc}
في \omterm{def:b3:rings:pidufd}{حلقة مثاليات رئيسية}، كل متتالية متزايدة من المثاليات $I_1 \subseteq I_2
\subseteq \cdots$
ثابتة ابتداءً من رتبة ما.
\end{lemma}

\begin{proof}
إن $I = \bigcup_n I_n$ \omterm{def:b3:rings:ideal}{مثالي} (لأن الاتحاد متزايد)، ومنه $I =
(a)$؛ ويقع العنصر
$a$ في $I_N$ ما، وعندئذٍ $I = (a)
\subseteq I_N \subseteq I_n \subseteq I$ من أجل $n \geq N$.
\end{proof}

\begin{lemma}[بيزو؛ مبرهنة إقليدس المساعدة]\label{lem:b3:rings:bezout}
لتكن $A$ \omterm{def:b3:rings:pidufd}{حلقة مثاليات رئيسية} وليكن $a, b \in A$. عندئذٍ $(a) + (b) = (d)$
من أجل $d$ ما، وهو \emph{قاسم مشترك أكبر}\index{قاسم مشترك أكبر}:
أي $d \mid a$ و $d \mid b$، وكل قاسم مشترك للعنصرين $a,
b$ يقسم
$d$؛ وعلاوة على ذلك $d = au + bv$ من أجل $u, v$ ما (بيزو). ومن
نتائج ذلك أن كل \omterm{def:b3:rings:divisibility}{عنصر غير قابل للاختزال} في \omterm{def:b3:rings:pidufd}{حلقة مثاليات رئيسية}
عنصرٌ \omterm{def:b3:rings:divisibility}{أولي}.
\end{lemma}

\begin{proof}
إن $(a) + (b)$ \omterm{def:b3:rings:ideal}{مثالي}، ومنه فهو $(d)$؛ ويعطي $a, b \in (d)$ أن
$d \mid
a, b$؛ و $d = au + bv \in (a) + (b)$. وكل قاسم مشترك $c$ للعنصرين $a, b$ يقسم
$au + bv = d$.

إقليدس: ليكن $p$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال} مع $p \mid ab$ و $p \nmid a$.
كل \omterm{lem:b3:rings:bezout}{قاسم مشترك أكبر} $d$ للعنصرين $p$ و $a$ يقسم $p$، ومنه فإن $d$
إما قابل للقلب وإما مصاحب للعنصر $p$ (لعدم قابلية الاختزال)؛
والتصاحب مستبعَد بسبب $p \nmid a$. إذن $1 = pu + av$، ومنه
$b = pub + abv$، ويقسم $p$ كلا الحدين: أي $p \mid b$.
\end{proof}

\begin{theorem}\label{thm:b3:rings:pidufd}
رئيسية $\Rightarrow$ تفكيكية.
\end{theorem}

\begin{proof}
\emph{الوجود.} لنفترض أن عنصرًا غير معدوم وغير قابل للقلب $a$ لا
يقبل تفكيكًا إلى عناصر غير قابلة للاختزال. عندئذٍ لا يكون $a$
\omterm{def:b3:rings:divisibility}{غير قابل للاختزال}: أي $a =
a_1b_1$ حيث العاملان غير قابلين للقلب؛
وأحدهما على الأقل، وليكن $a_1$، لا يقبل بدوره تفكيكًا (لأن جداء
عنصرين قابلين للتفكيك قابل للتفكيك). وبالتكرار نحصل على $a = a_0, a_1, a_2,
\dots$،
كلٌّ منها قاسم فعلي للذي قبله ولا يقبل تفكيكًا، ومنه $(a_0) \subsetneq (a_1) \subsetneq (a_2) \subsetneq \cdots$
--- والاحتواءات تامة لأن $a_n = a_{n+1}c$ مع $c$ غير قابل للقلب
يعني أن $(a_n) = (a_{n+1})$ سيفرض $c \in A^\times$ (بالاختصار في حلقة تامة).
وهذا يناقض \cref{lem:b3:rings:acc}.

\emph{الوحدانية.} ليكن $p_1 \cdots p_r = q_1 \cdots q_s$ حيث جميع العوامل غير قابلة
للاختزال و $r \leq s$، وذلك بالتراجع على $r$. يقسم \omterm{def:b3:rings:divisibility}{العنصر الأولي}
(\cref{lem:b3:rings:bezout}) $p_1$ الطرفَ الأيمن، ومنه فهو يقسم أحد العوامل
$q_j$؛ ولنعد الترقيم بحيث $j = 1$. وبما أن $q_1$ \omterm{def:b3:rings:divisibility}{غير قابل
للاختزال} و $p_1$ غير قابل للقلب، فإن $q_1 = u p_1$ حيث
$u \in A^\times$: أي إن $p_1,
q_1$ متصاحبان. نختصر $p_1$: $p_2 \cdots p_r = (u q_2)
q_3\cdots q_s$
ونستنتج بالتراجع (وحالة $r = 1$ تفرض $s = 1$: إذ لا يمكن لجداء
عنصر قابل للقلب بعناصر غير قابلة للاختزال أن يساوي $1$).
\end{proof}

\begin{remark}\label{rem:b3:rings:ufdgcd}
في \omterm{def:b3:rings:pidufd}{حلقة تفكيك وحيد}، توجد القواسم المشتركة الكبرى (بأخذ الأُسس
الأصغرية في التفكيكات) وتصح مبرهنة إقليدس المساعدة --- أي إن
\omterm{def:b3:rings:divisibility}{غير القابل للاختزال} $=$ \omterm{def:b3:rings:divisibility}{الأولي} (\cref{exo:b3:rings:2}) --- لكن متطابقة
بيزو قد تخفق: ففي $\Z[X]$ لدينا $\gcd(2, X) = 1$ ومع ذلك
$1 \neq 2U + XV$ (بالتقييم عند $X = 0$: نجد $1 = 2U(0)$، وهذا
مستحيل). فمتطابقات بيزو خاصية حصرية لحلقات المثاليات الرئيسية.
\end{remark}

\begin{example}[حلقة بلا تفكيك وحيد]
\label{ex:b3:rings:nonufd}
ليس أيٌّ من الاستلزامين \omterm{def:b3:rings:pidufd}{إقليدية} $\Rightarrow$ رئيسية
$\Rightarrow$ تفكيكية تكافؤًا، ويستحق إخفاق الأخير أن نراه مرة
واحدة بكامل تفاصيله. في
\[
A = \Z[\iu\sqrt5] = \{a + \iu b\sqrt5 : a, b \in \Z\},
\qquad N(a + \iu b\sqrt5) = a^2 + 5b^2,
\]
يكون المعيار ضربيًا ويتحقق $N(z) = 1$ إذا وفقط إذا كان $z \in
A^\times = \{\pm1\}$.
لننظر في
\[
6 = 2 \cdot 3 = (1 + \iu\sqrt5)(1 - \iu\sqrt5).
\]
جميع العوامل الأربعة غير قابلة للاختزال: فمعاييرها هي $4, 9, 6,
6$،
ولو وُجد تفكيك فعلي $z = z_1z_2$ لفُرض $N(z_1) \in \{2, 3\}$ --- لكن
$a^2 + 5b^2$ لا يساوي $2$ ولا $3$ أبدًا (فحالة $b = 0$ تترك
العددين غير المربّعين $2, 3$؛ وحالة $\abs b \geq
1$ تعطي $\geq 5$).
ومع ذلك فإن $2$ ليس مصاحبًا لأيٍّ من $1 \pm
\iu\sqrt5$ (لأن المعيارين
$4 \neq 6$): أي إن لدينا تفكيكين مختلفين فعلًا للعدد $6$ إلى
عناصر غير قابلة للاختزال. وبصيغة مكافئة، \omterm{def:b3:rings:divisibility}{غير القابل للاختزال}
$\neq$ \omterm{def:b3:rings:divisibility}{الأولي} هنا: إذ يقسم $2$ الجداءَ $(1 +
\iu\sqrt5)(1 - \iu\sqrt5) = 6$ ولا يقسم أيًّا
من العاملين (بالمعايير مرة أخرى). أما الإصلاح \omterm{def:b3:rings:ideal}{المثالي} لهذا
الإخفاق --- أي تفكيك \emph{المثاليات} بدل تفكيك العناصر ---
فهو ميلاد نظرية الأعداد الجبرية؛ وفي مستوانا، يُظهر هذا المثال
مدى خصوصية الحلقات \omterm{def:b3:rings:pidufd}{الإقليدية} $\Z$ و $K[X]$ و $\Z[\iu]$ الواردة
في هذا الفصل.
\end{example}

\begin{method}\label{met:b3:rings:quotient}
لتعيين حلقة قسمة $A/I$، ابحث عن تشاكل شامل $f \colon A \to B$ نواته $I$
واستعمل \cref{thm:b3:rings:firstiso}؛ وحين تكون $A = C[X]$ حلقةَ كثيرات حدود،
يكون $f$ عادةً تقييمًا. وهكذا نجد $\Z[X]/(X^2+1) \cong
\Z[\iu]$ (بالتقييم عند $\iu$)،
و $K[X,Y]/(Y - X^2) \cong K[X]$ (بتقييم $Y$ عند $X^2$)، و $\R[X]/(X^2+1) \cong \C$. ولإثبات أن $I$
\omterm{def:b3:rings:divisibility}{أولي} أو أعظمي، بيّن أن \omterm{def:b3:rings:ideal}{حلقة القسمة} تامة أو حقل (\cref{prop:b3:rings:primemaximal}).
\end{method}

\section{كثيرات الحدود على حلقة تفكيك وحيد: غاوس وأيزنشتاين}

في كل هذا القسم، $A$ \omterm{def:b3:rings:pidufd}{حلقة تفكيك وحيد} حقلُ كسورها $K$ (وهو مبنيّ
على أنه حقل الخوارج الصورية $a/b$ حيث $b \neq 0$، تمامًا كما
يُبنى $\Q$ انطلاقًا من $\Z$؛ وقد أنجز مجلد السنة الجامعية 2 هذا
البناء من أجل $\Q$، وهو ينتقل حرفيًا). وهدفنا هو: أن التفكيكية
تنتقل من $A$ إلى $A[X]$، وأن عدم قابلية الاختزال على $A$ هي في
جوهرها عدم قابلية الاختزال على الحقل الأوسع $K$.

\begin{definition}\label{def:b3:rings:content}
\emph{محتوى}\index{محتوى كثير حدود} كثير حدود غير معدوم
$P \in A[X]$، ويُرمَز إليه $c(P)$، هو \omterm{lem:b3:rings:bezout}{قاسم مشترك أكبر} لمعاملاته
(معرَّف إلى حدود عنصر قابل للقلب)؛ ويكون $P$ \emph{بدائيًا} إذا
كان $c(P) \in A^\times$. وكل $P
\in A[X]$ يُكتب $P = c(P)\,P_1$ حيث $P_1$
بدائي، وكل $P \in K[X]\setminus\{0\}$ يُكتب $P = \lambda P_1$ حيث $\lambda
\in K^\times$ و $P_1 \in A[X]$
بدائي (بتوحيد المقامات ثم إخراج المحتوى).
\end{definition}

\begin{lemma}[غاوس]\label{lem:b3:rings:gauss}
جداء كثيري حدود بدائيين من $A[X]$ بدائيٌّ؛ ومن نتائج ذلك $c(PQ) = c(P)c(Q)$
إلى حدود العناصر القابلة للقلب.\index{مبرهنة غاوس المساعدة}
\end{lemma}

\begin{proof}
ليكن $P, Q$ بدائيين ولنفترض أن عنصرًا \omterm{def:b3:rings:divisibility}{غير قابل للاختزال} (= \omterm{def:b3:rings:primemaximal}{أوليًا}،
لأن الحلقة تفكيكية) $p$ يقسم جميع معاملات $PQ$. لنختزل بترديد
$p$: في $(A/(p))[X]$ نجد $\bar P \bar Q = 0$. لكن $A/(p)$ حلقة تامة (لأن
$(p)$ \omterm{def:b3:rings:divisibility}{أولي})، ومنه فإن $(A/(p))[X]$ حلقة تامة (لأن المعاملات
المهيمنة تتضارب)، وهذا يفرض $\bar P = 0$ أو $\bar Q = 0$: أي إن
$p$ يقسم جميع معاملات $P$ أو جميع معاملات $Q$، وهذا يناقض
البدائية. أما النتيجة، فلنكتب $P = c(P)P_1$ و $Q = c(Q)Q_1$:
عندئذٍ $PQ =
c(P)c(Q) P_1Q_1$ مع $P_1Q_1$ بدائي.
\end{proof}

\begin{theorem}\label{thm:b3:rings:gaussufd}
لتكن $A$ \omterm{def:b3:rings:pidufd}{حلقة تفكيك وحيد} حقلُ كسورها $K$.
\begin{enumerate}
  \item يكون كثير الحدود البدائي $P \in A[X]$ من درجة $\geq 1$
        \omterm{def:b3:rings:divisibility}{غير قابل للاختزال} في $A[X]$ إذا وفقط إذا كان \omterm{def:b3:rings:divisibility}{غير قابل
        للاختزال} في $K[X]$.
  \item الحلقة $A[X]$ \omterm{def:b3:rings:pidufd}{حلقة تفكيك وحيد}؛ وعناصرها غير القابلة
        للاختزال هي عناصر $A$ غير القابلة للاختزال وكثيرات
        الحدود البدائية غير القابلة للاختزال على $K$. وعلى وجه
        الخصوص، فإن $\Z[X]$، ومنه بالتراجع $K[X_1, \dots, X_n]$ و $\Z[X_1, \dots, X_n]$،
        حلقات تفكيك وحيد.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) ($\Leftarrow$) إذا كان $P = QR$ في $A[X]$ حيث $Q, R$ غير
قابلين للقلب، فلا يكون أيٌّ من العاملين ثابتًا (لأن عاملًا ثابتًا
لكثير حدود بدائي يكون قابلًا للقلب)، ومنه فالتفكيك فعلي في
$K[X]$. ($\Rightarrow$) لنفترض $P = QR$ حيث $Q, R \in K[X]$ من
درجتين $\geq 1$. لنكتب $Q = \lambda Q_1$ و $R = \mu R_1$ حيث $Q_1, R_1
\in A[X]$
بدائيان: عندئذٍ $P = \lambda\mu\, Q_1R_1$، ويكون $Q_1R_1$ بدائيًا حسب غاوس.
وبأخذ المحتويات نجد $\lambda\mu \in A^\times$ (\omterm{def:b3:rings:content}{فمحتوى} الطرفين قابل للقلب؛ وبصورة
رسمية، $\lambda\mu = c(P) \in
A^\times$ إلى حدود العناصر القابلة للقلب، وعلى وجه الخصوص
$\lambda \mu \in A$): أي إن $P
= (\lambda\mu Q_1) R_1$ تفكيك فعلي في $A[X]$.

(2) الوجود: ليكن $P \neq 0$ غير قابل للقلب؛ نفكّك $P = c(P)P_1$،
ثم نفكّك $c(P)$ إلى عناصر $A$ غير قابلة للاختزال، ونفكّك $P_1$
في \omterm{def:b3:rings:pidufd}{حلقة التفكيك الوحيد} $K[X]$ على الصورة $\prod Q_i$ حيث
$Q_i \in K[X]$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال}؛ وبكتابة $Q_i = \lambda_i R_i$ حيث
$R_i \in A[X]$ بدائي (ومنه \omterm{def:b3:rings:divisibility}{غير قابل للاختزال} على $K$، ومنه في
$A[X]$ حسب (1))، يكون الجداء $\prod \lambda_i$ عنصرًا قابلًا للقلب من $A$
كما سبق، و $P_1 = u\prod
R_i$. الوحدانية: نقارن الجزء الثابت والجزء
كثير الحدود من التفكيك؛ فالثوابت تتضارب لتعطي $c(P)$ (غاوس)،
وهي وحيدة بتفكيكية $A$؛ وأما الأجزاء كثيرة الحدود فتعطي تفكيكين
في $K[X]$ لكثير الحدود نفسه، ومنه فهما متطابقان إلى حدود ثوابت
من $K^\times$ (بتفكيكية $K[X]$، \cref{thm:b3:rings:pidufd})، وكثيرا الحدود
البدائيان المتصاحبان في $K[X]$ متصاحبان في $A[X]$: إذ لو كان
$R = \lambda R'$ حيث $R, R'$ بدائيان و $\lambda \in K^\times$، لفرض أخذ
المحتويات أن $\lambda \in A^\times$.
\end{proof}

\begin{theorem}[محكات عدم قابلية الاختزال]\label{thm:b3:rings:criteria}
لتكن $A$ \omterm{def:b3:rings:pidufd}{حلقة تفكيك وحيد}، وليكن $K$ حقل كسورها، وليكن $P = a_nX^n + \dots
+ a_0 \in A[X]$
بدائيًا من الدرجة $n \geq 1$.
\begin{enumerate}
  \item (\emph{الاختزال}) إذا كان $p \in A$ \omterm{def:b3:rings:primemaximal}{أوليًا} و
        $p \nmid a_n$، وكان المختزَل $\bar P$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال}
        في $(A/(p))[X]$، فإن $P$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال} في $K[X]$
        (ومنه في $A[X]$).
  \item (\emph{أيزنشتاين}\index{محك أيزنشتاين}) إذا وُجد عدد
        \omterm{def:b3:rings:divisibility}{أولي} $p$ يحقق $p \nmid a_n$ و $p \mid a_i$ من أجل
        $0
        \leq i < n$ و $p^2 \nmid a_0$، فإن $P$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال} في
        $K[X]$ (ومنه في $A[X]$).
\end{enumerate}
\end{theorem}

\begin{proof}
حسب \cref{thm:b3:rings:gaussufd}(1)، يعطي تفكيك فعلي على $K$ أن $P = QR$ حيث
$Q, R \in A[X]$ و $\deg Q, \deg R \geq 1$ (والثوابت مستبعَدة: لأنها إما قابلة
للقلب وإما تُفسد البدائية).

(1) نختزل بترديد $p$: $\bar P = \bar Q\bar R$ في $(A/(p))[X]$. وبما أن
$p \nmid a_n$ وأن $\deg$ لا يمكن إلا أن تنخفض بالاختزال، فإن
$\deg \bar
Q = \deg Q \geq 1$ و $\deg\bar R = \deg R \geq 1$ (إذ تتضارب معاملاتهما المهيمنة لتعطي $\bar a_n \neq 0$،
فلا تنخفض أيٌّ منهما): أي إن $\bar P$ يتفكّك تفكيكًا فعليًا ---
وهو تناقض.

(2) نختزل بترديد $p$: $\bar Q \bar R = \bar P = \bar a_n X^n$ (إذ تنعدم جميع المعاملات الأدنى).
وفي الحلقة التامة $(A/(p))[X]$، تكون تفكيكات $cX^n$ (حيث
$c \ne 0$) إلى ثوابت وقوى صرفة $c'X^k$: فعلًا، إذا كان $\bar Q\bar R = \bar a_nX^n$،
ولو كان في $\bar Q$ معامل غير معدوم في درجة $< \deg\bar Q$،
لأخذنا الحدود الدنيا غير المعدومة: فنجد $\operatorname{val}(\bar Q\bar R) =
\operatorname{val}\bar Q + \operatorname{val}\bar R$ (لأن الحلقة
تامة)، وهذا يجب أن يساوي $n = \deg\bar Q + \deg\bar R$، وهو يفرض $\operatorname{val} = \deg$ من أجل
كليهما: ومنه فكلاهما وحيد الحد. وكما سبق، لا تنخفض الدرجات،
ومنه يكون الحدّان الثابتان $Q(0)$ و $R(0)$ قابلين للقسمة على
$p$ --- كلاهما، لأن المختزَلين وحيدا الحد ودرجتاهما $\geq 1$.
وعندئذٍ $p^2 \mid
Q(0)R(0) = a_0$: وهو تناقض.
\end{proof}

\begin{example}\label{ex:b3:rings:eisenstein}
كثير الحدود $X^n - p$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال} على $\Q$ من أجل كل عدد
\omterm{def:b3:rings:divisibility}{أولي} $p$ وكل $n
\geq 1$ (\omterm{thm:b3:rings:criteria}{أيزنشتاين} عند $p$): فتوجد كثيرات حدود غير
قابلة للاختزال من كل درجة على $\Q$ --- على النقيض التام من $\C$
(الدرجة $1$، دالمبير--غاوس، المبرهَن عليها في \cref{ch:b3:holomorphic}) ومن
$\R$ (الدرجتان $1, 2$). وتوسّع حيلةُ \emph{الإزاحة} مدى
\omterm{thm:b3:rings:criteria}{أيزنشتاين}: فكثير الحدود \emph{الدائري}\index{كثير حدود دائري}
النوني \omterm{def:b3:rings:divisibility}{الأولي} $\Phi_p = X^{p-1} +
\dots + X + 1 = \frac{X^p - 1}{X - 1}$ يحقق
\[
\Phi_p(X + 1) = \frac{(X+1)^p - 1}{X}
= X^{p-1} + \binom{p}{1}X^{p-2} + \dots + \binom{p}{p-1},
\]
وهو يحقق شرط \omterm{thm:b3:rings:criteria}{أيزنشتاين} عند $p$ (لأن $p \mid \binom pk$ من أجل $0 < k < p$،
و $\binom{p}{p-1} = p \not\equiv 0 \bmod p^2$): ومنه فإن $\Phi_p(X+1)$، ومنه $\Phi_p$، \omterm{def:b3:rings:divisibility}{غير قابل
للاختزال} على $\Q$. وهذا هو القلب الجبري لقصة المضلّع ذي $17$
ضلعًا المرويّة في \cref{ch:b3:galois}.
\end{example}

\begin{method}\label{met:b3:rings:irreducibility}
لإثبات أن $P \in \Z[X]$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال} على $\Q$: (أ) اجعل
$P$ بدائيًا؛ (ب) جرّب \omterm{thm:b3:rings:criteria}{أيزنشتاين}، على $P(X)$ وعلى الإزاحات
$P(X \pm
1)$؛ (ج) جرّب الاختزال بترديد أعداد أولية صغيرة لا تقسم
المعامل المهيمن --- فعدم قابلية الاختزال بترديد عدد \omterm{def:b3:rings:divisibility}{أولي}
\emph{واحد} يكفي، وعلى $\mathbb F_p$ يكون فحص عدم قابلية
الاختزال منتهيًا (فانعدام الجذور يستبعد العوامل من الدرجة $1$؛
ثم نختبر العوامل المنتهية العدد من كل درجة $\leq \deg P/2$)؛
(د) وإذا أخفق كل ذلك، فبالمعاملات غير المحدَّدة. واحذر: قابلية
الاختزال بترديد كل $p$ \emph{لا} تستلزم قابلية الاختزال على $\Q$
(\cref{exo:b3:rings:11}).
\end{method}

\section{الحلقات النويثرية}

\begin{definition}\label{def:b3:rings:noetherian}
تكون الحلقة $A$ \emph{نويثرية}\index{حلقة نويثرية} إذا كان كل
\omterm{def:b3:rings:ideal}{مثالي} فيها مولَّدًا بعدد منتهٍ من العناصر.
\end{definition}

\begin{proposition}\label{prop:b3:rings:noethacc}
تكون $A$ \omterm{def:b3:rings:noetherian}{نويثرية} إذا وفقط إذا كانت كل متتالية متزايدة من
المثاليات ثابتة ابتداءً من رتبة ما (\emph{شرط السلاسل
المتزايدة})، وإذا وفقط إذا كان لكل عائلة غير خالية من المثاليات
عنصرٌ أعظمي (بالنسبة إلى الاحتواء).
\end{proposition}

\begin{proof}
\emph{(التوليد المنتهي $\Rightarrow$ شرط السلاسل)}: من أجل
سلسلة $I_1 \subseteq I_2
\subseteq \cdots$، يكون الاتحاد $I$ مثاليًا مولَّدًا بالعناصر $x_1,
\dots, x_r$؛
وتقع جميع العناصر $x_i$ في $I_N$ ما، ومنه $I = I_N = I_n$ من
أجل $n \geq N$. \emph{(شرط السلاسل $\Rightarrow$ العناصر
الأعظمية)}: لو لم يكن لعائلة غير خالية $\mathcal F$ عنصر أعظمي،
لاخترنا $I_1 \in
\mathcal F$، ثم بالتتابع $I_{n+1} \supsetneq I_n$ في $\mathcal F$ (وهذا ممكن
لأن $I_n$ ليس \omterm{def:b3:rings:primemaximal}{أعظميًا}): فنحصل على سلسلة لامنتهية متزايدة تمامًا.
(ويستعمل هذا بديهيةَ الاختيارات المتعلقة، وهي صورة ضعيفة من
بديهية الاختيار لا نتوقف عندها.) \emph{(العناصر الأعظمية
$\Rightarrow$ التوليد المنتهي)}: ليكن $I$ مثاليًا؛ فعائلة
المثاليات المولَّدة بعدد منتهٍ والمحتواة في $I$ غير خالية (لأن
$(0)$ فيها)؛ وكل عنصر أعظمي $J = (x_1, \dots, x_r)$ يجب أن يساوي $I$: وإلا
لأنتج ضمُّ $x \in I \setminus J$ إلى المولِّدات عنصرًا أكبر تمامًا في العائلة.
\end{proof}

\begin{theorem}[مبرهنة الأساس لهيلبرت]\label{thm:b3:rings:hilbert}
إذا كانت $A$ \omterm{def:b3:rings:noetherian}{نويثرية} فإن $A[X]$ \omterm{def:b3:rings:noetherian}{نويثرية}. ومنه فإن $A[X_1, \dots,
X_n]$
\omterm{def:b3:rings:noetherian}{نويثرية}، وكذلك كل حلقة قسمة لها.\index{مبرهنة الأساس لهيلبرت}
\end{theorem}

\begin{proof}
ليكن $I$ مثاليًا في $A[X]$، ولنفترض أن $I$ غير مولَّد بعدد منتهٍ
من العناصر. لنبنِ متتالية: $f_1 \in I \setminus \{0\}$ من درجة أصغرية، ثم بالتتابع
$f_{k+1} \in I \setminus (f_1,
\dots, f_k)$ من درجة أصغرية (والمجموعة غير خالية بحكم الفرض). تكون
الدرجات $d_k = \deg f_k$ غير متناقصة (بأصغرية كل اختيار: فقد كان
$f_{k+1}$ متاحًا في الخطوة $k+1$\dots\ وبدقة، $f_{k+1} \notin (f_1,\dots,f_k) \supseteq
(f_1, \dots, f_{k-1})$، ومنه فقد
نافس $f_{k+1}$ في الخطوة $k$ فخسر أو تعادل: أي $d_{k+1} \geq d_k$). وليكن
$a_k \in A$ المعامل المهيمن لكثير الحدود $f_k$. تستقر سلسلة المثاليات
$(a_1) \subseteq (a_1,
a_2) \subseteq \cdots$: أي $a_{n+1} \in (a_1, \dots,
a_n)$ من أجل $n$ ما، وليكن $a_{n+1} = \sum_{k\leq n} u_k a_k$. لننظر في
\[
g = f_{n+1} - \sum_{k=1}^{n} u_k X^{\,d_{n+1} - d_k} f_k .
\]
عندئذٍ $g \in I \setminus (f_1, \dots, f_n)$ (لأن المجموع يقع في \omterm{def:b3:rings:ideal}{المثالي} بينما $f_{n+1}$ لا
يقع فيه)، ومع ذلك ينعدم معامل الدرجة $d_{n+1}$: أي $\deg g < d_{n+1}$،
وهذا يناقض أصغرية $d_{n+1} = \deg f_{n+1}$.

وبالتكرار تكون $A[X_1, \dots, X_n] = (A[X_1, \dots, X_{n-1}])[X_n]$ \omterm{def:b3:rings:noetherian}{نويثرية}؛ وتكون \omterm{def:b3:rings:ideal}{حلقة القسمة} $A/I$ \omterm{def:b3:rings:noetherian}{نويثرية}
لأن مثالياتها $J/I$ ترتفع إلى مثاليات في $A$ (بمبرهنة التقابل)،
حيث يُسقَط عددٌ منتهٍ من المولِّدات على مولِّدات.
\end{proof}

\begin{remark}\label{rem:b3:rings:noethmeaning}
\omterm{def:b3:rings:noetherian}{النويثرية} هي بديهية المنتهي في الهندسة الجبرية: فأي جملة من
المعادلات كثيرة الحدود ذات $n$ متغيرًا، مهما كانت لامنتهية، تكافئ
عددًا منتهيًا منها --- ومجموعة حلولها مقتطعة بعدد منتهٍ من
كثيرات الحدود. وحلقات المثاليات الرئيسية \omterm{def:b3:rings:noetherian}{نويثرية} (بداهةً)؛ أما
$\Z[X_1, X_2, \dots]$ بعدد لامنتهٍ من المتغيرات فليست كذلك ($(X_1) \subsetneq (X_1, X_2) \subsetneq \cdots$). وتظهر
الحلقات غير \omterm{def:b3:rings:noetherian}{النويثرية} أيضًا في التحليل بصورة طبيعية: فالدوال
المتصلة على $\intcc01$ تشكّل واحدة منها (\cref{exo:b3:rings:10}).
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:rings:1}
عيّن حلقات القسمة: (a) $\Z[X]/(X^2 + 1) \cong \Z[\iu]$؛ (b) $\R[X]/(X^2+1) \cong \C$؛ (c) برهن على أن
$\mathbb F_2[X]/(X^2 + X + 1)$ حقلٌ من $4$ عناصر --- واكتب جدول ضربه.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:rings:2}
(a) برهن على أن كل \omterm{def:b3:rings:divisibility}{عنصر غير قابل للاختزال} في \omterm{def:b3:rings:pidufd}{حلقة تفكيك وحيد}
عنصرٌ \omterm{def:b3:rings:divisibility}{أولي}.
(b) برهن على أن كل حلقة تامة منتهية حقلٌ.
(c) استنتج أن كل \omterm{def:b3:rings:primemaximal}{مثالي أولي} في حلقة منتهية أعظميٌّ.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b3:rings:3}
في $\Z[\iu\sqrt5]$: تحقق من أن $3$ و $1 + \iu\sqrt5$ و $1 -
\iu\sqrt5$
غير قابلة للاختزال، ومن أن $9 = 3\cdot 3 = (2 +
\iu\sqrt5)(2 - \iu\sqrt5)$، ثم اخلص مرة أخرى (بعد
\cref{prop:b3:rings:primeirred}) إلى أن $\Z[\iu\sqrt5]$ ليست \omterm{def:b3:rings:pidufd}{حلقة تفكيك وحيد}. وأين
تخفق الوحدانية بالضبط؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:4}
(a) برهن على أن $\Z[\iu]$ \omterm{def:b3:rings:pidufd}{إقليدية} من أجل المعيار $N(x + \iu y) =
x^2 + y^2$: إذا
أُعطي $a, b \neq 0$، اختر $q \in \Z[\iu]$ الأقرب إلى
$a/b \in \C$.
(b) عيّن $\Z[\iu]^\times$.
(c) الأسئلة نفسها من أجل $\Z[\iu\sqrt2]$ و $N(x + \iu y\sqrt2) =
x^2 + 2y^2$. ولماذا
تخفق الحجّة نفسها من أجل $\Z[\iu\sqrt5]$؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:5}
لتكن $A$ حلقة. (a) برهن على أنه إذا كان $x$ معدوم القوة (أي
$x^n = 0$ من أجل $n$ ما) فإن $1 + x \in A^\times$. (b) برهن على أنه إذا كانت
$A$ تامة فإن $A[X]^\times = A^\times$؛ وأعطِ مثالًا مضادًّا على $\Z/4\Z$.
(c) برهن على أنه ليس في حلقة تامة عناصر ذاتية التربيع
($e^2 = e$) غير $0, 1$، ولا عناصر معدومة القوة غير $0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:6}
في $A = K[X, Y]$: (a) برهن على أن \omterm{def:b3:rings:ideal}{المثالي} $(X, Y)$ أعظمي وليس
رئيسيًا --- ومنه فإن $K[X,Y]$ \omterm{def:b3:rings:pidufd}{حلقة تفكيك وحيد} (\cref{thm:b3:rings:gaussufd})
وليست \omterm{def:b3:rings:pidufd}{حلقة مثاليات رئيسية}؛ (b) عيّن $K[X, Y]/(Y - X^2)$ و $K[X,Y]/(XY - 1)$ على أنهما
حلقتان جزئيتان من الدوال الكسرية؛ (c) هل $(Y - X^2)$ \omterm{def:b3:rings:divisibility}{أولي}؟ وهل
هو أعظمي؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:7}
قابل للاختزال أو \omterm{def:b3:rings:divisibility}{غير قابل للاختزال} على $\Q$: $X^5 - 12X^3 + 36X - 12$؛\quad
$X^4 + X + 1$ \emph{(اختزل بترديد $2$)}؛\quad $X^4 + 4$؛\quad
$\Phi_8 = X^4 + 1$ \emph{(أزح بمقدار $1$)}؛\quad $X^3 - X - 1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:8}
(a) انطلاقًا من \cref{thm:b3:rings:crt}، برهن على أن دالة أويلر ضربية على
الوسائط الأولية فيما بينها وأن $\varphi(p^k) =
p^{k-1}(p-1)$؛ واستعد $\varphi(n) = n\prod_{p \mid n}(1 -
\frac1p)$.
(b) حل ما يلي: $x \equiv 2 \pmod 7$، $x \equiv 5 \pmod{11}$، $x
\equiv 1 \pmod{13}$، مُظهِرًا العناصر
ذاتية التربيع $e_k$ الواردة في برهان \cref{thm:b3:rings:crt}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:rings:9}
(الجذر الصفري) لتكن $\operatorname{Nil}(A)$ مجموعة العناصر معدومة القوة.
(a) برهن على أن $\operatorname{Nil}(A)$ \omterm{def:b3:rings:ideal}{مثالي} محتوًى في كل \omterm{def:b3:rings:primemaximal}{مثالي أولي}.
(b) وبالعكس، ليكن $a$ غير معدوم القوة؛ باستعمال مبرهنة زورن
المساعدة على المثاليات التي تتجنّب $S =
\{a^n : n \in \N\}$، أنتج مثاليًا \omterm{def:b3:rings:primemaximal}{أوليًا}
لا يحتوي على $a$. واخلص إلى:
\[
\operatorname{Nil}(A) = \bigcap_{\mathfrak p \text{ أولي}}
\mathfrak p .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:rings:10}
(a) لتكن $A$ \omterm{def:b3:rings:noetherian}{نويثرية} وليكن $f \colon A \to A$ تشاكل حلقات شاملًا. برهن على
أن $f$ متباين. \emph{(انظر في $\ker
f \subseteq \ker f^2 \subseteq \cdots$.)}
(b) برهن على أن حلقة الدوال المتصلة $\mathcal C(\intcc01, \R)$ ليست \omterm{def:b3:rings:noetherian}{نويثرية}.
\emph{(انظر في $I_n = \{f : f = 0
\text{ على } \intcc0{1/n}\}$.)}
(c) برهن على أنه في \emph{حلقة تامة} \omterm{def:b3:rings:noetherian}{نويثرية}، يكون كل عنصر غير
معدوم وغير قابل للقلب جداءً (منتهيًا) لعناصر غير قابلة للاختزال
--- ومنه فإن عدم تفكيكية $\Z[\iu\sqrt 5]$ إخفاقٌ في
\emph{الوحدانية} لا غير.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:rings:11}
ليكن $P = X^4 + 1$.
(a) برهن على أن $P$ \omterm{def:b3:rings:divisibility}{غير قابل للاختزال} على $\Q$ (\cref{exo:b3:rings:7}).
(b) برهن على أن $P$ قابل للاختزال بترديد \emph{كل} عدد \omterm{def:b3:rings:divisibility}{أولي} $p$:
عالج الحالة $p = 2$؛ ثم، من أجل $p$ فردي، برهن على أن
$8 \mid p^2 - 1$، واقبل مؤقتًا (وهو مُبرهَن عليه في \cref{ch:b3:galois})
أن الزمرة الضربية للحقل ذي $p^2$ عنصرًا دائرية، لتخلص إلى أن $P$
ينشطر إلى عاملين من الدرجة الثانية بترديد $p$؛ واكتبهما صراحةً
حين يكون أحد الأعداد $-1$، $2$، $-2$ مربّعًا بترديد $p$، وبرهن
على أن أحدها يكون كذلك دائمًا.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:rings:12}
(العناصر ذاتية التربيع تشطر الحلقات) نقول عن عنصر $e$ من حلقة
تبديلية $A$ إنه \emph{ذاتي التربيع} إذا كان $e^2 = e$.
(a) برهن على أنه إذا كان $e$ ذاتي التربيع فكذلك $1 - e$، وأن
التطبيق $x \mapsto (ex, (1-e)x)$ تماثل حلقات $A
\cong Ae \times A(1-e)$، حيث $Ae$ حلقة عنصرها
الواحدي $e$.
(b) عيّن جميع العناصر ذاتية التربيع في حلقة تامة، وفي
$\Z/12\Z$؛ وأظهر التماثل $\Z/12\Z \cong \Z/4\Z \times \Z/3\Z$ بتسمية عنصريه غير التافهين
ذاتيَي التربيع.
(c) برهن على أن تفكيك $\Z/n\Z$ حسب مبرهنة الباقي الصيني
(\cref{ex:b3:rings:crtz}) يوافق تمامًا العناصر ذاتية التربيع $e_i \equiv 1 \bmod p_i^{a_i}$، حيث
$e_i \equiv 0$ بترديد قوى الأعداد الأولية الأخرى: فالحلقات
تتفكّك على امتداد عناصرها ذاتية التربيع كما تتفكّك الفضاءات على
امتداد إسقاطاتها.
\end{exercise}

\section{مسألة: مبرهنة فيرما لمجموع مربعين}

\begin{problem}[{مسألة نهاية الأسبوع --- مجاميع مربعين، عبر
$\Z[\iu]$}]\label{pb:b3:rings:1}
أي الأعداد الصحيحة مجاميعُ مربعين؟ جواب فيرما (1640) من جواهر
الحساب؛ وتحوّل أعداد غاوس الصحيحة برهانَه إلى نظرية حلقات. في كل
ما يلي، يرمز $N(x + \iu y) = x^2 + y^2$ إلى المعيار، وتكون $\Z[\iu]$ \omterm{def:b3:rings:pidufd}{إقليدية}
(\cref{exo:b3:rings:4})، ومنه \omterm{def:b3:rings:pidufd}{حلقة مثاليات رئيسية} \omterm{def:b3:rings:pidufd}{وحلقة تفكيك وحيد}، ونعني
بعبارة \emph{عدد غاوس \omterm{def:b3:rings:divisibility}{أولي}} عنصرًا \omterm{def:b3:rings:primemaximal}{أوليًا} (= \omterm{def:b3:rings:divisibility}{غير قابل للاختزال})
من $\Z[\iu]$.

\medskip
\noindent\textbf{الجزء الأول --- المعايير وأعداد غاوس الأولية.}
\begin{enumerate}
  \item تحقق من $N(zw) = N(z)N(w)$، واستنتج مرة أخرى $\Z[\iu]^\times =
        \{\pm1, \pm\iu\}$، وبرهن على
        \emph{متطابقة براهماغوبتا}: جداء مجموعَي مربعين مجموعُ
        مربعين.
  \item برهن على أنه إذا كان $N(z)$ عددًا \omterm{def:b3:rings:primemaximal}{أوليًا} فإن $z$ عدد
        غاوس \omterm{def:b3:rings:divisibility}{أولي}.
  \item برهن على أن كل عدد غاوس \omterm{def:b3:rings:divisibility}{أولي} $\pi$ يقسم عددًا \omterm{def:b3:rings:primemaximal}{أوليًا}
        واحدًا بالضبط $p$ \emph{(انظر في $N(\pi) = \pi\bar\pi$)}، وأن عندئذٍ
        $N(\pi) \in \{p, p^2\}$.
  \item استنتج الانقسام الثنائي: من أجل كل عدد \omterm{def:b3:rings:divisibility}{أولي} $p$، إما أن
        يبقى $p$ \omterm{def:b3:rings:primemaximal}{أوليًا} في $\Z[\iu]$ (فلا يوجد عدد غاوس \omterm{def:b3:rings:divisibility}{أولي}
        معياره $p$)، وإما أن يكون $p = \pi\bar\pi$ حيث $\pi$ عدد
        غاوس \omterm{def:b3:rings:divisibility}{أولي} معياره $p$ --- وعندئذٍ $p = a^2 + b^2$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- مبرهنة ويلسون والعدد $-1$
بترديد $p$.}
\begin{enumerate}[resume]
  \item برهن على \emph{مبرهنة ويلسون}: من أجل $p$ \omterm{def:b3:rings:divisibility}{أولي}، $(p-1)!
        \equiv -1 \pmod p$.
        \emph{(زاوج كل بقية مع مقلوبها؛ وأيُّها مزاوَجة مع
        نفسها؟)}
  \item ليكن $p$ عددًا \omterm{def:b3:rings:primemaximal}{أوليًا} فرديًا وليكن $m = \frac{p-1}2$. برهن على أن
        $(m!)^2 \equiv (-1)^{m+1} \pmod p$ \emph{(في $(p-1)!$، استبدل بكل عامل $k > m$
        العنصرَ $-(p - k)$)}.
  \item اخلص إلى: يكون $-1$ مربّعًا بترديد $p$ إذا وفقط إذا كان
        $p = 2$ أو $p
        \equiv 1 \pmod 4$. \emph{(من أجل «فقط إذا»: إذا كان
        $x^2
        \equiv -1$، فما رتبة $x$ في $(\Z/p\Z)^\times$؟ وماذا تقول مبرهنة
        لاغرانج؟)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- قانون الانشطار.}
\begin{enumerate}[resume]
  \item ليكن $p \equiv 1 \pmod 4$، وليكن $x$ يحقق $p \mid x^2 + 1 =
        (x + \iu)(x - \iu)$. برهن على أن $p$
        \emph{ليس} عددًا غاوسيًا \omterm{def:b3:rings:primemaximal}{أوليًا}، واخلص مع الجزء الأول
        إلى: $p = a^2 + b^2$.
  \item ليكن $p \equiv 3 \pmod 4$. برهن مباشرةً على أن $p$ ليس مجموع مربعين
        \emph{(المربّعات بترديد $4$)}، واستنتج أن $p$ يبقى عددًا
        غاوسيًا \omterm{def:b3:rings:primemaximal}{أوليًا}.
  \item احسم الحالة $p = 2$: أظهر التفكيك $2 =
        -\iu(1+\iu)^2$ وتحقق من أن
        $1 + \iu$ عدد غاوس \omterm{def:b3:rings:divisibility}{أولي}. (والعدد $2$ هو العدد \omterm{def:b3:rings:divisibility}{الأولي}
        \emph{المتفرّع} الوحيد: أي القابل للقسمة على مربّع عدد
        غاوس \omterm{def:b3:rings:divisibility}{أولي} إلى حدود عنصر قابل للقلب.)
  \item كوِّن تصنيف أعداد غاوس الأولية، إلى حدود العناصر القابلة
        للقلب: $1 + \iu$؛ والأعداد الصحيحة $p \equiv 3 \pmod 4$؛ وأزواج
        المرافقات $\pi, \bar\pi$ ذات المعيار $p \equiv 1 \pmod
        4$. وتحقق منه
        على $5 = (2+\iu)(2-\iu)$ وعلى $3$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- مبرهنة مجموع مربعين.}
\begin{enumerate}[resume]
  \item برهن على النصف المباشر: إذا ظهر في التفكيك $n =
        \prod p_i^{\alpha_i}$ كل
        عدد \omterm{def:b3:rings:divisibility}{أولي} $\equiv 3 \pmod 4$ بأُسّ زوجي، فإن $n$ مجموع مربعين.
        \emph{(براهماغوبتا + الجزآن الثاني والثالث.)}
  \item برهن على العكس: إذا كان $n = a^2 + b^2 = N(a + \iu b)$ وكان $q \equiv 3 \pmod 4$ يقسم $n$،
        فبرهن على أن $q$، وهو عدد غاوس \omterm{def:b3:rings:divisibility}{أولي}، يقسم $a + \iu b$ أو
        $a - \iu b$، وأنه يقسم في الواقع كلًّا من $a$ و $b$، ثم
        اخلص بالتراجع على $n$ إلى أن أُسّ $q$ في $n$ زوجي.
  \item اذكر المبرهنة النهائية. وأيُّ الأعداد $2025$ و $2026$
        و $2027$ مجاميعُ مربعين؟ \emph{($2025 = 81 \cdot 25$؛ $2026 = 2 \cdot 1013$، حيث
        $1013$ \omterm{def:b3:rings:divisibility}{أولي}؛ و $2027$ \omterm{def:b3:rings:divisibility}{أولي}.)}
  \item (خاتمة) برهن على أن كل عدد \omterm{def:b3:rings:divisibility}{أولي} $p \equiv 1 \pmod 4$ مجموعُ مربعين
        بطريقة وحيدة جوهريًا: أي إذا كان $p =
        a^2 + b^2 = c^2 + d^2$ (أعداد صحيحة
        موجبة)، فإن $\{a, b\}
        = \{c, d\}$. \emph{(بوحدانية التفكيك في
        $\Z[\iu]$.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- إحصاء التمثيلات: صيغة ياكوبي
ومتسلسلة لايبنتز.} نكتب $r_2(n) = \#\{(a, b) \in
\Z^2 : a^2 + b^2 = n\}$ (الأزواج المرتّبة، مع الإشارات
والأصفار)، وليكن $\chi$ الطابع غير التافه بترديد $4$: أي
$\chi(d) = +1$ إذا كان $d \equiv 1$، و $-1$ إذا كان $d \equiv 3 \pmod4$،
و $0$ إذا كان $d$ زوجيًا.
\begin{enumerate}[resume]
  \item (تمهيد، على سبيل المقارنة) أي الأعداد الصحيحة \emph{فروق}
        مربعين؟ برهن على أن $n = a^2 - b^2$ حيث $a, b \in \Z$
        إذا وفقط إذا كان $n \not\equiv 2 \pmod 4$ --- دون حاجة إلى نظرية الحلقات،
        ودون بنية تُقارن بما سيأتي.
  \item برهن على أن $r_2(n)$ هو عدد العناصر $z \in \Z[\iu]$ التي
        تحقق $N(z) = n$. وبكتابة $n = 2^{a}\prod_jp_j^{b_j}
        \prod_kq_k^{c_k}$ حيث $p_j \equiv 1$
        و $q_k \equiv 3
        \pmod4$، استعمل تصنيف السؤال 11 ووحدانية التفكيك
        لتبرهن على أن مثل هذه العناصر $z$ توجد إذا وفقط إذا كانت
        جميع الأُسس $c_k$ زوجية، وأنه في تلك الحالة
        \[
        r_2(n) = 4\prod_j\,(b_j + 1) .
        \]
        \emph{(أحصِ: $z = u\,(1+\iu)^{a}\prod_j\pi_j^{s_j}
        \bar\pi_j^{\,b_j - s_j}\prod_kq_k^{c_k/2}$ حيث $u$ عنصر قابل للقلب و $0 \leq s_j \leq b_j$؛
        ولماذا تكون هذه القائمة شاملةً وخاليةً من التكرار؟)}
  \item برهن على أن $d \mapsto \chi(d)$ ضربي تمامًا، واستنتج أن $n \mapsto
        \sum_{d \mid n}\chi(d)$
        ضربي، ثم احسبه على قوى الأعداد الأولية: فهو يساوي $1$ على
        $2^a$؛ و $b + 1$ على $p^b$ (حيث $p \equiv 1$)؛ و $1$ أو
        $0$ على $q^c$ (حيث $q \equiv 3$) بحسب زوجية $c$ أو
        فرديته.
  \item اخلص إلى \emph{مبرهنة ياكوبي}:
        \[
        r_2(n) = 4\sum_{d \mid n}\chi(d) =
        4\bigl(d_1(n) - d_3(n)\bigr),
        \]
        حيث يحصي $d_i(n)$ القواسم $\equiv i \pmod 4$. وتحقق منها على
        $n = 3, 5, 9, 25$، واسرد تمثيلات $65$ الستة عشر.
  \item (الدائرة) برهن على أن $\sum_{n \leq x}r_2(n)$ هو عدد نقط الشبكة $\Z^2$
        الواقعة في القرص المغلق ذي نصف القطر $\sqrt x$، ثم برهن
        على
        \[
        \sum_{n\leq x}r_2(n) = \pi x + O(\sqrt x)
        \]
        \emph{(كل نقطة شبكية تملك مربّعًا واحديًا؛ قارن
        المساحات، والخطأ يقع في حلقة عرضها $O(1)$)}.
  \item (لايبنتز، مقروءةً حسابيًا) اجمع بين السؤالين 19 و 20:
        \[
        \sum_{d \leq x}\chi(d)\Bigl\lfloor\frac
        xd\Bigr\rfloor = \frac{\pi x}4 + O(\sqrt x),
        \]
        واستنتج --- بإزالة الجزأين الصحيحين بعناية --- متسلسلة
        لايبنتز
        \[
        1 - \frac13 + \frac15 - \frac17 + \dots = \frac\pi4 .
        \]
        فالمتسلسلة المتناوبة لمقلوبات الأعداد الفردية \emph{هي}
        متوسط زيادة القواسم $\equiv 1$ على القواسم $\equiv 3$:
        تحليلٌ يحسبه الحساب.
  \item (كم هي نادرة مجاميع المربعين؟) برهن على أنه ليس هناك عدد
        صحيح $\equiv 3 \pmod 4$ مجموعَ مربعين (بطريقتين: المربّعات بترديد
        $4$، أو محك الزوجية في السؤال 17)، ومنه فإن ربع الأعداد
        الصحيحة على الأقل مستبعَد؛ وبرهن على أن المتوسط $\frac1x\sum_{n\leq
        x}r_2(n) \to \pi$
        في السؤال 20 يتوافق مع كون الأعداد القابلة للتمثيل ذات
        كثافة $0$ --- أظهر أعدادًا لها عدد تمثيلات كبير على نحو
        شاذ (خذ جداءات لعدد كبير من الأعداد الأولية $\equiv 1 \bmod 4$)
        لتفسّر كيف يمكن لنسبة متلاشية أن تحمل مع ذلك متوسطًا
        موجبًا. (وقد برهن لانداو على أن الكثافة الحقيقية تتناقص
        مثل $1/\sqrt{\log x}$؛ وهذا يتجاوز أدواتنا، لكن الآلية صارت الآن
        مرئية.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء السادس --- تكملات: التمثيلات البدائية
وفيثاغورس.}
\begin{enumerate}[resume]
  \item نقول عن تمثيل $n = a^2 + b^2$ إنه \emph{بدائي} إذا كان
        $\gcd(a, b) = 1$. برهن على أن $n \geq 1$ يقبل تمثيلًا
        بدائيًا إذا وفقط إذا كان $4 \nmid n$ ولم يقسمه أي عدد
        \omterm{def:b3:rings:divisibility}{أولي} $q \equiv 3 \pmod 4$. \emph{(من أجل الشرط اللازم، أعد استعمال
        النزول في السؤال 13 والمربّعات بترديد $4$؛ ومن أجل الشرط
        الكافي، ابنِ $z$ من $1 + \iu$ ومن العناصر $\pi_j$ لا
        غير --- دون مرافقات --- وفسّر لماذا يفرض عاملٌ \omterm{def:b3:rings:divisibility}{أولي} مشترك
        بين $a$ و $b$ دخولَ كلٍّ من $\pi_j$ و $\bar\pi_j$، أو
        دخولَ $(1+\iu)^2$، في $z$.)}
  \item (ثلاثيات فيثاغورس) ليكن $a^2 + b^2 = c^2$ حيث $a, b,
        c$ موجبة و
        $\gcd(a, b) = 1$ و $b$ زوجي. برهن على أن $a + \iu b$
        و $a - \iu b$ أوليان فيما بينهما في $\Z[\iu]$ \emph{(إذ
        إن قاسمًا غاوسيًا \omterm{def:b3:rings:primemaximal}{أوليًا} مشتركًا سيقسم $2a$ و $2b$،
        و $c$ فردي)}، واستنتج من وحدانية التفكيك أن $a + \iu b = u(m + \iu n)^2$ من
        أجل عنصر قابل للقلب $u$، ثم اخلص إلى التوسيط الكلاسيكي:
        إلى حدود تبديل $a$ و $b$،
        \[
        a = m^2 - n^2, \qquad b = 2mn, \qquad c = m^2 + n^2,
        \]
        حيث $m > n \geq 1$ أوليان فيما بينهما ومختلفا الزوجية.
        واستعد $(3, 4, 5)$ و $(21, 20, 29)$ من $(m, n) =
        (2, 1)$ ومن
        $(5, 2)$.
  \item (تحقق عددي) خذ $x = 25$. احسب $r_2(n)$ من أجل $1 \leq n \leq 25$
        بصيغة ياكوبي، وتحقق من أن القيم غير المعدومة تقع بالضبط
        عند $n = 1,
        2, 4, 5, 8, 9, 10, 13, 16, 17, 18, 20, 25$، ومن أن
        \[
        \sum_{n \leq 25} r_2(n) = 80
        = 4\sum_{d \leq 25}\chi(d)
        \Bigl\lfloor\frac{25}d\Bigr\rfloor .
        \]
        وتحقق من أن القرص المغلق ذا نصف القطر $5$ يحتوي على $81$
        نقطة شبكية، وقارن ذلك بالمقدار $\pi x \approx 78.5$: فالخطأ يقع
        بأريحية داخل $O(\sqrt x)$ من السؤال 20.
\end{enumerate}
\end{problem}
