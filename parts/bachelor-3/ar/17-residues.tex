\chapter{متسلسلات لوران ومبرهنة البواقي}
\label{ch:b3:residues}

ماذا يحدث \omterm{def:b3:holomorphic:holo}{لدالة هولومورفية} بجوار نقطة \emph{لا} تكون معرَّفة
عندها؟ الجواب ثلاثيةٌ تامة --- نقطة قابلة للإزالة، أو قطب، أو
شذوذ جوهري --- تُقرأ من متسلسلة قوى ذات طرفين، هي نشر لوران.
ويتحكم معاملٌ واحد من ذلك النشر، هو \emph{\omterm{def:b3:residues:singularities}{الباقي}}، في كل \omterm{def:b3:holomorphic:contour}{تكامل
مسار} حول الشذوذ: فتحوّل مبرهنة البواقي التكاملاتِ المحدودة
الصعبة إلى جبر منتهٍ، وتعدّ جذور الدوال (مبدأ العمدة، وروشيه)،
وتبرهن على مبرهنة التطبيق المفتوح. ونرقّي أولًا مبرهنة كوشي من
المجالات النجمية الشكل إلى صيغتها النهائية الخالية من
الهومولوجيا --- بحجة ديكسون الأنيقة --- بحيث تصير متاحةً جميع
المسارات ذات عدد اللف المعدوم حول المتممة.

\section{مبرهنة كوشي الشاملة}

\emph{الدورة}\index{دورة} $\Gamma$ مجموعٌ شكلي منتهٍ لمسارات
مغلقة $\gamma_1, \dots, \gamma_m$؛ والتكاملات والأدلّة على امتداد $\Gamma$ هي
المجاميع الموافقة، و $\operatorname{im}\Gamma = \bigcup\operatorname{im}\gamma_j$.

\begin{theorem}[كوشي، الصيغة الشاملة]
\label{thm:b3:residues:globalcauchy}
لتكن $\Omega \subseteq \C$ مفتوحة، ولتكن $f \in \mathcal H(\Omega)$، ولتكن $\Gamma$ دورةً في
$\Omega$ تحقق
\[
\operatorname{Ind}_\Gamma(w) = 0
\qquad\text{من أجل كل} w \notin \Omega .
\]
عندئذٍ يكون من أجل كل $z \in \Omega\setminus\operatorname{im}\Gamma$،
\[
\frac1{2\iu\pi}\int_\Gamma\frac{f(w)}{w - z}\,\dd w =
\operatorname{Ind}_\Gamma(z)\,f(z),
\qquad\text{و}\qquad
\int_\Gamma f(w)\,\dd w = 0 .
\]
\index{مبرهنة كوشي (الشاملة)}
\end{theorem}

\begin{proof}[البرهان (ديكسون)]
نعرّف $g \colon \Omega\times\Omega \to \C$ بالعلاقة
\[
g(z, w) = \begin{cases}
\dfrac{f(w) - f(z)}{w - z} & w \neq z,\\[4pt]
f'(z) & w = z .
\end{cases}
\]
\emph{الدالة $g$ متصلة}: فخارج القطر، ذلك واضح. وبجوار نقطة
قطرية $(a, a)$، ننشر $f$ في متسلسلة قوى عند $a$ (\cref{thm:b3:holomorphic:analytic}):
$f(w) - f(z) =
\sum_{n\geq1}c_n\bigl((w-a)^n - (z-a)^n\bigr)$، وبقسمة كل حدّ على $w - z$ (بتحليل $u^n - v^n$) نجد، من
أجل $z, w \in D(a, r)$،
\[
g(z, w) =
\sum_{n\geq1}c_n\sum_{j=0}^{n-1}(w-a)^{\,j}(z-a)^{\,n-1-j},
\]
وهي صحيحة على القطر أيضًا (إذ يصير كل مجموع داخلي
$n(z-a)^{n-1}$، ويكون مجموعها $f'(z)$). ومن أجل $r$ صغير
تتقارب المتسلسلة بانتظام على $D(a,r)^2$ ($\abs{\text{الحدّ}} \leq n\abs{c_n}r^{n-1}$، وهي قابلة
للجمع داخل نصف القطر): فيكون المجموع \omterm{def:b3:topology:continuity}{متصلًا}.

نضع $h(z) = \frac1{2\iu\pi}\int_\Gamma g(z, w)\,\dd w$ على $\Omega$: فهي متصلة (بالاتصال المنتظم للدالة
$g$ على المتراصات)، \omterm{def:b3:holomorphic:holo}{وهولومورفية} --- بموريرا (أي محك
\cref{thm:b3:holomorphic:weierstrassconv}): إذ من أجل مثلث $T \subseteq \Omega$، تعطي فوبيني أن $\int_{\partial
T}h = \frac1{2\iu\pi}\int_\Gamma\bigl(\int_{\partial T}g(z,
w)\dd z\bigr)\dd w = 0$، إذ
ينعدم التكامل الداخلي لأن $z \mapsto g(z, w)$ \omterm{def:b3:holomorphic:holo}{هولومورفية} على $\Omega$ (وعند
$z = w$ يكون الشذوذ قابلًا للإزالة: إذ $g$ متصلة هناك
\omterm{def:b3:holomorphic:holo}{وهولومورفية} في غيرها --- بحجة التمديد في برهان \cref{thm:b3:holomorphic:formula}).

وعلى \omterm{def:b3:topology:topology}{المجموعة المفتوحة} $\Omega' = \{z \notin \operatorname{im}\Gamma
: \operatorname{Ind}_\Gamma(z) = 0\}$، نعرّف $h_1(z) =
\frac1{2\iu\pi}\int_\Gamma\frac{f(w)}{w - z}\dd w$: وهي \omterm{def:b3:holomorphic:holo}{هولومورفية} على
$\Omega'$ (بموريرا أو بالاشتقاق تحت علامة التكامل). ومن أجل
$z \in \Omega\cap\Omega'$:
\[
h(z) = \frac1{2\iu\pi}\int_\Gamma\frac{f(w)}{w-z}\dd w
- f(z)\operatorname{Ind}_\Gamma(z) = h_1(z) .
\]
وبالفرضية $\Omega\cup\Omega' = \C$ ($w \notin \Omega
\Rightarrow \operatorname{Ind}_\Gamma(w) = 0$)، ومنه تلتحم $h$ و $h_1$ في دالة صحيحة
$H$. وبما أن المركّبة غير المحدودة لمتممة $\operatorname{im}\Gamma$ تقع في
$\Omega'$ وأن $h_1(z) \to 0$ حين $\abs z \to \infty$ (بحدّ $ML$)، تكون $H$
محدودة وتؤول إلى $0$: فيعطي ليوفيل (\cref{cor:b3:holomorphic:liouville}) أن
$H \equiv 0$. ومنه $h \equiv 0$ على $\Omega$، وهي الصيغة
التكاملية. وبتطبيقها، من أجل $a \in
\Omega\setminus\operatorname{im}\Gamma$ مثبَّتة، على $\tilde f(w) =
(w - a)f(w)$ عند
$z = a$:
\[
\frac1{2\iu\pi}\int_\Gamma f(w)\dd w =
\frac1{2\iu\pi}\int_\Gamma \frac{\tilde f(w)}{w - a}\dd w =
\operatorname{Ind}_\Gamma(a)\,\tilde f(a) = 0 .
\]
\end{proof}

\section{متسلسلات لوران والشذوذات المعزولة}

\begin{theorem}[نشر لوران]\label{thm:b3:residues:laurent}
لتكن $f$ \omterm{def:b3:holomorphic:holo}{هولومورفية} على التاج $A = \{r < \abs{z - a} <
R\}$ ($0 \leq r < R \leq \infty$).
عندئذٍ\index{متسلسلة لوران}
\[
f(z) = \sum_{n\in\Z}c_n\,(z - a)^n
\qquad\text{على} A,
\qquad
c_n = \frac1{2\iu\pi}\int_{C_\rho}\frac{f(w)}{(w -
a)^{n+1}}\,\dd w
\]
من أجل أي $r < \rho < R$ (ولا يتعلق بنصف القطر $\rho$)، وتتقارب نصفا
المتسلسلة تقاربًا ناظميًّا على التيجان الجزئية المتراصة. والنشر
وحيد.
\end{theorem}

\begin{proof}
نثبّت $r < \rho_1 < \abs{z - a} < \rho_2 < R$ ولتكن $\Gamma =
C_{\rho_2} - C_{\rho_1}$ (الخارجية عكس عقارب الساعة والداخلية
مع عقارب الساعة): وهي دورة في $A$ تحقق $\operatorname{Ind}_\Gamma(w) = 0$ من أجل كل
$w \notin A$ (فالنقاط داخل القرص الصغير: $1 - 1 = 0$؛ وخارج
الكبير: $0 - 0$). وحسب \cref{thm:b3:residues:globalcauchy}، يعطي $\operatorname{Ind}_\Gamma(z) = 1 - 0 = 1$ أن
\[
f(z) = \frac1{2\iu\pi}\int_{C_{\rho_2}}\frac{f(w)}{w - z}\dd
w - \frac1{2\iu\pi}\int_{C_{\rho_1}}\frac{f(w)}{w - z}\dd w .
\]
وننشر النواة الأولى كما في \cref{thm:b3:holomorphic:analytic} (بقوى $\frac{z -
a}{w - a}$، وطويلتها
$< 1$): فيكون الجزء غير السالب $\sum_{n\geq0}c_n(z-a)^n$. وفي الثانية ننشر
بالاتجاه الآخر: $\frac{-1}{w - z} = \frac1{(z-a)(1 - \frac{w - a}{z -
a})} = \sum_{m\geq0}\frac{(w-a)^m}{(z - a)^{m+1}}$، وهي متقاربة تقاربًا ناظميًّا على
$C_{\rho_1}$: فيكون الجزء السالب $\sum_{n\leq-1}c_n(z-a)^n$ بالمعاملات المذكورة
(بالدليل $n = -m-1$). وأما استقلاله عن $\rho$: فتكاملات
المعاملات على $C_{\rho}$ و $C_{\rho'}$ تختلف بالمقدار
$\int_\Gamma$ على دورة معدومة الدليل \omterm{def:b3:holomorphic:holo}{للدالة الهولومورفية}
$\frac{f(w)}{(w-a)^{n+1}}$ في $A$: أي بصفر، حسب \cref{thm:b3:residues:globalcauchy} من جديد. وأما الوحدانية:
فنكامل $\sum c_n(z-a)^n$ مقابل $(z - a)^{-m-1}$ على $C_\rho$ حدًّا حدًّا
(بالتقارب الناظمي): فلا ينجو سوى $n = m$.
\end{proof}

\begin{definition}\label{def:b3:residues:singularities}
إذا كانت $f$ \omterm{def:b3:holomorphic:holo}{هولومورفية} على قرص مثقوب $D(a, R)\setminus
\{a\}$، نشرناها بلوران
(مع $r = 0$). وثمة ثلاث حالات متنافية:\index{شذوذ معزول}
\begin{itemize}
  \item أن تكون جميع $c_n = 0$ من أجل $n < 0$: أي \emph{\omterm{thm:b3:residues:riemanncw}{شذوذ
        قابل للإزالة}} (إذ تمدّد المتسلسلة غير السالبة $f$
        هولومورفيًّا إلى $a$)؛
  \item أن يكون $c_n \neq 0$ من أجل عدد منتهٍ، وواحد على
        الأقل، من $n < 0$: أي \emph{قطب} من الرتبة $m = -\min\{n : c_n \neq 0\}$؛ أي،
        بالتكافؤ، $f = g/(z-a)^m$ مع $g$ \omterm{def:b3:holomorphic:holo}{هولومورفية} و $g(a)
        \neq 0$؛
        أي، بالتكافؤ، $\abs{f(z)} \to \infty$ حين $z\to
        a$؛
  \item أن يكون $c_n \neq 0$ من أجل عدد لانهائي من $n$ السالبة: أي
        \emph{شذوذ جوهري}.
\end{itemize}
و\emph{الباقي}\index{باقٍ} هو $\operatorname{Res}(f, a) = c_{-1}$. \omterm{def:b3:holomorphic:holo}{والدالة الهولومورفية} على
$\Omega$ منزوعًا منها مجموعة أقطاب تسمّى
\emph{مرومورفية}\index{دالة مرومورفية} على $\Omega$.
\end{definition}

\begin{theorem}[ريمان؛ كازوراتي--فايرشتراس]
\label{thm:b3:residues:riemanncw}
لتكن $f$ \omterm{def:b3:holomorphic:holo}{هولومورفية} على $D(a,R)\setminus\{a\}$.
\begin{enumerate}
  \item (ريمان) إذا كانت $f$ \emph{محدودة} بجوار $a$، كان
        الشذوذ قابلًا للإزالة.\index{شذوذ قابل للإزالة}
  \item (كازوراتي--فايرشتراس) إذا كان $a$ جوهريًّا، فإن $f\bigl(D(a,\varepsilon)\setminus\{a\}\bigr)$
        كثيفة في $\C$ من أجل كل $\varepsilon$.
        \index{مبرهنة كازوراتي--فايرشتراس}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) من أجل $n < 0$ وحين $\rho \to 0$: $\abs{c_n} \leq
M\rho^{-n-1}\cdot\rho\cdot\rho^{-\,n}\dots$ بمتراجحة $ML$ على
$C_\rho$: أي $\abs{c_n} \leq \frac{1}{2\pi}\,2\pi\rho\cdot
M\rho^{-(n+1)} = M\rho^{-n} \to 0$ (لأن $-n > 0$): فتنعدم جميع المعاملات
السالبة.
(2) لو لم تُقارَب قيمةٌ ما $b$: لكان $\abs{f - b} \geq
\delta$ بجوار $a$، ومنه
تكون $g = 1/(f - b)$ \omterm{def:b3:holomorphic:holo}{هولومورفية} ومحدودة بجوار $a$: أي قابلة
للإزالة (1)، فتمتدّ $g$ بالقيمة $c$. فإذا كان $c \neq 0$، كانت
$f = b + 1/g$ محدودة بجوار $a$: أي قابلة للإزالة --- وهو
مستبعد. وإذا كان $c = 0$، كان للدالة $g$ جذر من رتبة منتهية $m$ عند
$a$ (\cref{thm:b3:holomorphic:identity}؛ إذ $g \not\equiv 0$)، وكان للدالة $f = b + 1/g$
قطبٌ من الرتبة $m$: وهو مستبعد أيضًا.
\end{proof}

\section{مبرهنة البواقي}

\begin{theorem}[مبرهنة البواقي]\label{thm:b3:residues:residue}
لتكن $\Omega$ مفتوحة، ولتكن $S \subseteq \Omega$ منتهية، ولتكن $f \in
\mathcal H(\Omega\setminus S)$، ولتكن
$\Gamma$ دورةً في $\Omega\setminus S$ تحقق $\operatorname{Ind}_\Gamma(w) = 0$ من أجل كل $w \notin \Omega$.
عندئذٍ\index{مبرهنة البواقي}
\[
\frac{1}{2\iu\pi}\int_\Gamma f(z)\,\dd z
= \sum_{a\in S}\operatorname{Ind}_\Gamma(a)\,
\operatorname{Res}(f, a) .
\]
\end{theorem}

\begin{proof}
من أجل كل $a \in S$، ليكن $P_a(z) = \sum_{n\leq-1}c_n^{(a)}(z -
a)^n$ الجزءَ الرئيسي للدالة $f$ عند
$a$: أي متسلسلة تتقارب على $\C\setminus\{a\}$ (إذ نصف قطرها في $1/(z-a)$
لانهائي: فذيل لوران يتقارب من أجل كل $\abs{z-a}$ صغير، ومنه،
بكونه متسلسلة قوى في $(z-a)^{-1}$، يتقارب في كل مكان)،
\omterm{def:b3:holomorphic:holo}{وهولومورفية} هناك. عندئذٍ يكون للدالة $g = f - \sum_{a\in S}P_a$ شذوذات قابلة
للإزالة عند كل نقطة من $S$ (إذ ليس لنشر لوران لها عند $a$ جزء
سالب: لأن بقية $P_{a'}$ \omterm{def:b3:holomorphic:holo}{هولومورفية} عند $a$)، ومنه تمتدّ $g$
هولومورفيًّا إلى $\Omega$، وتعطي \cref{thm:b3:residues:globalcauchy} أن $\int_\Gamma g = 0$. ويبقى أن
نكامل كل $P_a$: حدًّا حدًّا (بالتقارب الناظمي على المتراصة
$\operatorname{im}\Gamma$، وهي تتفادى $a$)،
\[
\frac1{2\iu\pi}\int_\Gamma(z - a)^n\,\dd z = 0 \ (n \leq -2:
\text{الدالة الأصلية} \tfrac{(z-a)^{n+1}}{n+1}),
\qquad
\frac1{2\iu\pi}\int_\Gamma\frac{\dd z}{z - a} =
\operatorname{Ind}_\Gamma(a),
\]
ومنه $\frac1{2\iu\pi}\int_\Gamma P_a =
c_{-1}^{(a)}\operatorname{Ind}_\Gamma(a)$. ثم نجمع على $a$.
\end{proof}

\begin{method}[حساب البواقي]\label{met:b3:residues:compute}
القطب البسيط: $\operatorname{Res}(f, a) = \lim_{z\to a}(z -
a)f(z)$؛ ومن أجل $f = g/h$ مع $g(a) \neq 0$
و $h(a) = 0$ و $h'(a)
\neq 0$: $\operatorname{Res} = g(a)/h'(a)$. والقطب من الرتبة $m$: $\operatorname{Res}(f, a) = \frac1{(m-1)!}\lim_{z\to
a}\bigl((z-a)^mf(z)\bigr)^{(m-1)}$.
وأما الشذوذات الجوهرية: فانشر واقرأ $c_{-1}$ (مثلًا من متسلسلات
معروفة). وتحقق دائمًا من الأقطاب التي يحيط بها المسار فعلًا،
وبأي دليل.
\end{method}

\begin{example}[الأنماط التكاملية الكلاسيكية الأربعة]
\label{ex:b3:residues:integrals}
(a) \emph{الكسرية على $\R$}: من أجل $\int_\R\frac{\dd x}{1 +
x^4}$، نغلق بنصف دائرة
كبيرة $S_R$ في نصف المستوي العلوي: فيكون المقدار المكامَل من
رتبة $O(R^{-4})$ هناك، ومنه $\int_{S_R} \to 0$ (بمتراجحة $ML$)، وتعطي
مبرهنة البواقي مع الأقطاب $\eu^{\iu\pi/4}, \eu^{3\iu\pi/4}$ (وهي بسيطة، وبواقيها $\frac1{4z^3} = \frac{z}{4z^4} = -\frac z4$
عند قطب)
\[
\int_\R\frac{\dd x}{1 + x^4}
= 2\iu\pi\Bigl(-\frac{\eu^{\iu\pi/4}}4 -
\frac{\eu^{3\iu\pi/4}}4\Bigr)
= \frac{\pi}{\sqrt2} .
\]
(b) \emph{النمط الفورييهي}: من أجل $t \geq 0$، $\int_\R\frac{\eu^{\iu tx}}{1 + x^2}\dd x =
2\iu\pi\operatorname{Res}\bigl(\tfrac{\eu^{\iu tz}}{1+z^2},
\iu\bigr) = 2\iu\pi\frac{\eu^{-t}}{2\iu} = \pi\eu^{-t}$ --- إذ
يصلح نصف الدائرة العلوي لأن $\abs{\eu^{\iu tz}} =
\eu^{-t\operatorname{Im}z} \leq 1$ هناك؛ وبأخذ الأجزاء
الحقيقية: $\int_\R\frac{\cos(tx)}{1+x^2}\dd x = \pi\eu^{-\abs t}$، فتُسدَّد الصيغة المسلَّم بها في \cref{exo:b3:lebesgue:10}.
(c) \emph{المثلثية على دور}: نعوّض $z =
\eu^{\iu t}$ و $\cos t = \frac{z + z^{-1}}2$ و $\dd t =
\frac{\dd z}{\iu z}$:
فيصير $\int_0^{2\pi}\frac{\dd t}{a + \cos t}$ (حيث $a > 1$) عدَّ بواقٍ داخل الدائرة الواحدية
(\cref{exo:b3:residues:2}).
(d) \emph{المتسلسلات}: نزاوج $f$ مع $\pi\cot(\pi z)$، وأقطابها
هي الأعداد الصحيحة \omterm{def:b3:residues:singularities}{بباقٍ} $1$: وتجمع مسألة نهاية الأسبوع
$\sum n^{-2}$ و $\sum n^{-4}$ بهذه الطريقة.
\end{example}

\section{مبدأ العمدة ومبرهنة روشيه}

\begin{theorem}[مبدأ العمدة]
\label{thm:b3:residues:argument}
لتكن $f$ \omterm{def:b3:residues:singularities}{مرومورفية} على $\Omega$، بجذور $z_j$ (من الرتب $m_j$)
وأقطاب $p_k$ (من الرتب $\mu_k$)، وليكن $\gamma$ مسارًا مغلقًا
في $\Omega$ يتفاداها جميعًا، مع $\operatorname{Ind}_\gamma = 0$ خارج $\Omega$.
عندئذٍ\index{مبدأ العمدة}
\[
\frac1{2\iu\pi}\int_\gamma\frac{f'(z)}{f(z)}\,\dd z
= \sum_j m_j\operatorname{Ind}_\gamma(z_j)
- \sum_k \mu_k\operatorname{Ind}_\gamma(p_k)
\]
(وعدد الحدود غير المعدومة منتهٍ). ومن أجل مسار بسيط عكس عقارب
الساعة، يعدّ التكامل الجذورَ ناقصَ الأقطاب في الداخل، بالترافق
--- ويساوي عدد لف المسار الصورة $f\circ\gamma$ حول $0$.
\end{theorem}

\begin{proof}
بجوار جذر من الرتبة $m$: $f = (z-a)^mg$ مع $g(a) \neq 0$، ومنه
$\frac{f'}f = \frac m{z - a} + \frac{g'}g$ والحدّ الثاني هولومورفي بجوار $a$: أي قطب بسيط \omterm{def:b3:residues:singularities}{بباقٍ}
$m$. وبجوار قطب من الرتبة $\mu$: يعطي $f = (z-a)^{-\mu}g$ الباقيَ $-\mu$.
وفي غير ذلك تكون $\frac{f'}f$ \omterm{def:b3:holomorphic:holo}{هولومورفية}. (وتقع الجذور
والأقطاب ذات الدليل غير المعدوم في منطقة متراصة يحيط بها
$\gamma$؛ وبمبرهنة التطابق تكون منتهية العدد هناك، إذ
$f \not\equiv 0$.) ثم نطبّق \cref{thm:b3:residues:residue}. وأما الملاحظة الأخيرة:
$\frac1{2\iu\pi}\int_\gamma\frac{f'}f =
\frac1{2\iu\pi}\int_{f\circ\gamma}\frac{\dd w}w =
\operatorname{Ind}_{f\circ\gamma}(0)$ (بالتعويض $w =
f(\gamma(t))$).
\end{proof}

\begin{theorem}[روشيه]\label{thm:b3:residues:rouche}
لتكن $f, g$ هولومورفيتين على $\Omega$، وليكن $\gamma$ مسارًا
مغلقًا مع $\operatorname{Ind}_\gamma \in \{0,1\}$، ومعدومَ الدليل خارج $\Omega$ (أي مسار بسيط).
إذا كان\index{مبرهنة روشيه}
\[
\abs{g(z)} < \abs{f(z)} \qquad \text{على}
\operatorname{im}\gamma,
\]
فإن للدالتين $f$ و $f + g$ العددَ نفسه من الجذور (بالترافق) في
المنطقة $\{\operatorname{Ind}_\gamma =
1\}$.
\end{theorem}

\begin{proof}
من أجل $t \in \intcc01$، لا يقبل $f_t = f + tg$ أي جذر على
$\operatorname{im}\gamma$ ($\abs{f_t} \geq \abs f - \abs g >
0$)، ومنه يكون
\[
N(t) = \frac1{2\iu\pi}\int_\gamma
\frac{f_t'(z)}{f_t(z)}\,\dd z
\]
معرَّفًا جيدًا؛ وهو يعدّ الجذور في المنطقة المحاطة (\cref{thm:b3:residues:argument}؛
إذ لا أقطاب). و $N$ متصل في $t$ (إذ المقدار المكامَل متصل
معًا، والمقامات محدودة من الأسفل بانتظام --- بالتقارب المهيمن)
وذو قيم صحيحة: أي ثابت. ومنه $N(0) = N(1)$.
\end{proof}

\begin{corollary}[مبرهنة التطبيق المفتوح]
\label{cor:b3:residues:openmapping}
كل \omterm{def:b3:holomorphic:holo}{دالة هولومورفية} غير ثابتة على \omterm{def:b3:topology:topology}{مجموعة مفتوحة} مترابطة تطبيقٌ
مفتوح. وعلى وجه الخصوص يصح (من جديد) مبدأ الأعظمية، ولتقابل
هولومورفي مقلوبٌ هولومورفي.\index{مبرهنة التطبيق المفتوح
(الهولومورفي)}
\end{corollary}

\begin{proof}
ليكن $f(a) = b$؛ فللدالة $f - b$ جذر من رتبة منتهية ما $m
\geq 1$
عند $a$ (بمبرهنة التطابق: إذ $f \not\equiv b$). ونختار $r$ بحيث
تكون $f - b$ عديمة الجذور على $\bar D(a, r)\setminus\{a\}$ (بعزل الجذور) ولنضع
$\delta = \min_{\abs{z - a} =
r}\abs{f(z) - b} > 0$. ومن أجل $\abs{w - b} < \delta$: على الدائرة، $\abs{(b - w)} < \delta \leq \abs{f - b}$، ومنه تقول روشيه
($f - b$ مقابل الثابت $b - w$) إن للدالة $f - w$ عدد $m$ من الجذور
بالضبط في $D(a, r)$: أي إن كل $w$ كهذه مبلوغة --- $f(D(a,r)) \supseteq D(b, \delta)$: أي
مفتوح. وأما مبدأ الأعظمية: فقيمة عظمى داخلية للمقدار $\abs f$
مستحيلة من أجل $f$ غير ثابتة، إذ تحتوي صورتها حول $f(a)$ نقاطًا
أكبر طويلةً. وأما المقلوب: فتقابلٌ هولومورفي $f$ يكون مفتوحًا،
ومنه يكون $f^{-1}$ \omterm{def:b3:topology:continuity}{متصلًا}؛ وجذر $f - f(a)$ عند $a$ بسيط (إذ
$m \geq 2$ كان سيعطي $m$ من الصور العكسية لقيم مجاورة ---
متمايزةً، لأن $f'$ لا تنعدم إلا عند نقاط معزولة، ومنه تكون
جذور $f - w$ البالغة $m$ بجوار $a$ بسيطةً ومتمايزة من أجل $w$ صغيرة
عامة: وهو تناقض مع التباين)؛ عندئذٍ $f'(a) \neq 0$ وتتقارب
نسبة الفرق للدالة $f^{-1}$: أي $\bigl(f^{-1}\bigr)'(b) =
1/f'(a)$.
\end{proof}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:5}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:6}
(a) برهن على أن للمقدار $\eu^{1/z}$ شذوذًا جوهريًّا عند $0$

(b) برهن على أن $\abs{\eu^{1/z}}$ غير محدود على كل \omterm{def:b3:topology:topology}{جوار} مثقوب للنقطة $0$
ومع ذلك ليس للمقدار $\eu^{1/z}$ أي قطب: فأي نهاية تخفق؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:7}
عُدَّ بروشيه:
(a) جذور $z^7 - 4z^3 + z - 1$ في $\abs z < 1$؛
(b) جذور $z^4 + 5z + 1$ في $\abs z < 1$ وفي $1 <

(c) وأعد البرهان على دالمبير--غاوس: أن لكثير حدود موحَّد من

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:residues:8}
(هورفيتز) لتكن $f_n \to f$ بانتظام على المتراصات، مع $f_n \in
\mathcal H(\Omega)$،
حيث $\Omega$ مترابطة و $f \not\equiv 0$.
(a) برهن على أنه إذا كانت جميع $f_n$ عديمة الجذور، كانت $f$
كذلك. \emph{(إذا كان $f(a) = 0$: فبمبدأ العمدة على دائرة صغيرة

$\int f_n'/f_n$.)}
(b) برهن على أنه إذا كانت جميع $f_n$ متباينة، كانت $f$ متباينة
أو ثابتة. \emph{(بتطبيق (a) على $z \mapsto f_n(z) - f_n(w)$ على $\Omega\setminus\{w\}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:residues:9}
من أجل $n \geq 2$، كامل $\frac1{1 + z^n}$ على حافة القطاع $\{0 \leq \arg z \leq \frac{2\pi}n,\ \abs z
\leq R\}$
واستنتج
\[
\int_0^{+\infty}\frac{\dd x}{1 + x^n} =
\frac{\pi}{n\,\sin(\pi/n)} .
\]
وتحقق من $n = 2$ مقابل $\arctan$، ومن النهاية $n \to
\infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:10}
لتكن $f$ دالةً كسرية مع $\deg(\text{المقام})
\geq \deg(\text{البسط}) + 2$. برهن على أن مجموع
\emph{جميع} بواقي $f$ معدوم \emph{(بالمكاملة على دوائر أكبر
فأكبر)}. واستعمل ذلك لإعادة حساب تفكيك $\frac1{z(z-1)(z-2)}$ إلى كسور جزئية
دون أي جبر خطي.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:residues:11}
(ثقب المفتاح: تكامل الانعكاس لأولير) من أجل $0 < a < 1$، احسب
\[
I(a) = \int_0^{\infty}\frac{x^{a-1}}{1 + x}\,\dd x =
\frac{\pi}{\sin(\pi a)}
\]
بمكاملة $f(z) = \frac{z^{a-1}}{1+z} =
\frac{\eu^{(a-1)\log z}}{1 + z}$ (باللوغاريتم المشقوق على امتداد $\R_+$،
$\arg z \in \intoo0{2\pi}$) على مسار ثقب المفتاح: خارجًا على امتداد أعلى الشق من
$\varepsilon$ إلى $R$، وحول $C_R$، وراجعًا تحت الشق، وحول
$C_\varepsilon$. وبرّر: أن الامتدادين المستقيمين يختلفان
بالعامل $\eu^{2\iu\pi(a-1)}$، وأن إسهامَي الدائرتين ينعدمان ($R^{a-1}\cdot R \to 0$
و $\varepsilon^{a-1}\cdot\varepsilon \to 0$)، وأن القطب الوحيد $z = -1$ باقيه $\eu^{\iu\pi(a - 1)}$. واستنتج
أيضًا $\Gamma(a)\Gamma(1 - a) = \frac\pi{\sin\pi a}$ \emph{(اكتب $\Gamma(a)\Gamma(1-a) = B(a, 1-a)$ حسب \cref{pb:b3:lebesgue:1} وعوّض $t =
\frac{x}{1+x}$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:residues:12}
(عدّ الجذور بمبدأ العمدة، عدديًّا) ليكن $P(z) = z^4 + 8z + 1$.
(a) كم جذرًا في القرص الواحدي؟ \emph{(بروشيه مقابل $8z + 1$.)}
(b) وكم في التاج $1 < \abs z < 3$؟ \emph{(بروشيه مقابل $z^4$
على $\abs z = 3$.)} وحدّد أكثر: برهن على أن طويلة كل جذر
$< 2.1$.
(c) وكم في نصف المستوي الأيمن؟ \emph{(عُدَّ على $\abs z = 2$
أولًا؛ ثم تتبّع صورة المحور التخيّلي: فللمقدار $P(\iu t) = t^4 + 1 + 8\iu t$ جزء
حقيقي موجب في كل مكان، ومنه لا جذور على المحور، ويمكن حساب
تغيّر العمدة على امتداده --- واخلص بنصف قرص كبير.)}
\end{exercise}

\section{مسألة: \texorpdfstring{$\zeta(2k)$}{زيتا(2k)} بواسطة
ظل التمام}

\begin{problem}[{مسألة نهاية الأسبوع --- جمع $\sum n^{-2k}$
بالبواقي}]\label{pb:b3:residues:1}
تجمع مبرهنة البواقي المتسلسلات: فمزاوجة دالة كسرية مع
$\pi\cot(\pi z)$، وأقطابها تجلس عند الأعداد الصحيحة، تحوّل
$\sum_{n}f(n)$ إلى عدّ بواقٍ. ونبرهن على الطريقة ونحسب $\zeta(2) = \frac{\pi^2}{6}$
و $\zeta(4) =
\frac{\pi^4}{90}$ --- أي القيمتين اللتين وُجدتا بمتسلسلات فورييه في
السنة الجامعية 2 وبآثار المؤثرات في \cref{ch:b3:spectral}، والآن بالمكاملة
على المسارات.

\medskip
\noindent\textbf{الجزء الأول --- نواة ظل التمام.}
\begin{enumerate}
  \item برهن على أن $\pi\cot(\pi z)$ \omterm{def:b3:residues:singularities}{مرومورفية} على $\C$
        بأقطاب بسيطة عند $z = n \in \Z$ بالضبط، وباقي كلٍّ
        منها $1$ \emph{(باحتساب $\lim_{z\to n}(z-n)\pi\cot\pi z$)}.
  \item احسب بداية نشر لوران عند $0$:
        \[
        \pi\cot(\pi z) = \frac1z - \frac{\pi^2}{3}\,z -
        \frac{\pi^4}{45}\,z^3 + O(z^5) ,
        \]
        بقسمة متسلسلة القوى للدالة $\cos$ على متسلسلة
        $\sin$ (وبرّر القسمة: إذ $\frac{\sin\pi z}{\pi z}$ \omterm{def:b3:holomorphic:holo}{هولومورفية} وغير
        معدومة بجوار $0$، ومنه يكون مقلوبها هولومورفيًّا؛
        وعيّن المعاملات حتى الرتبة $3$).
  \item ليكن $C_N$ حافةَ المربّع ذي الرؤوس $(\pm1\pm\iu)(N + \frac12)$. برهن على
        أن $\abs{\cot(\pi z)} \leq 2$ على $C_N$ من أجل كل $N \geq
        1$. \emph{(على
        الأضلاع الشاقولية، $\cot(\pi(\pm(N +
        \frac12) + \iu y)) = \mp\tan(\iu\pi y)$، وطويلته $\abs{\tanh(\pi y)} \leq 1$؛ وعلى
        الأضلاع الأفقية $\abs y = N + \frac12$، حُدّ $\abs{\cot(\pi(x\pm\iu y))} \leq \coth(\pi y) \leq
        \coth(\pi/2) < 1.1$.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- مبرهنة الجمع.}
\begin{enumerate}[resume]
  \item لتكن $f$ كسرية \omterm{def:b3:holomorphic:holo}{وهولومورفية} عند الأعداد الصحيحة، مع
        $\deg(\text{المقام}) \geq \deg(\text{البسط}) + 2$. وباستعمال مبرهنة البواقي على $C_N$ وحدّ
        السؤال 3، برهن على:
        \[
        \lim_{N\to\infty}\ \sum_{n = -N}^{N} f(n)
        = -\sum_{p\ \text{قطب للدالة}\ f}
        \operatorname{Res}\bigl(\pi\cot(\pi z)f(z),\,p\bigr).
        \]
  \item وأين تحتاج الحجة إلى شرط الدرجة؟ برهن بمثال (خذ
        $f(z) = 1/(z + \frac12)$) على أنه من أجل تناقص أبطأ قد توجد النهاية
        المتناظرة رغم تباعد المتسلسلة ذات الطرفين --- وأن
        الصيغة تحسب عندئذٍ \emph{القيمة الرئيسية}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- القيم.}
\begin{enumerate}[resume]
  \item طبّق الطريقة على $f(z) = 1/z^2$: وهنا يكون قطب $f$
        \emph{عند} عدد صحيح، فأجرِ الحجة مباشرةً --- كامل
        $g(z) = \frac{\pi\cot(\pi
        z)}{z^2}$ على $C_N$، وبرهن على أن التكامل $\to 0$،
        واحسب $\operatorname{Res}(g, 0)$ من السؤال 2. واخلص:
        \[
        2\sum_{n\geq1}\frac1{n^2} = \frac{\pi^2}{3},
        \qquad \zeta(2) = \frac{\pi^2}6 .
        \]
  \item والأمر نفسه مع $g(z) = \frac{\pi\cot(\pi z)}{z^4}$: احسب $\operatorname{Res}(g, 0)$ واستنتج $\zeta(4) = \frac{\pi^4}{90}$.
  \item اشرح النسق العام: أنه من أجل كل $k \geq 1$، يكون
        $\zeta(2k)$ مساويًا $-\frac12$ مضروبًا في معامل
        $z^{2k-1}$ في نشر لوران للمقدار $\pi\cot(\pi
        z)$ عند $0$ ---
        أي مضاعفًا ناطقًا للمقدار $\pi^{2k}$. واحسب $\zeta(6)$
        بدفع قسمة السؤال 2 خطوةً أبعد. وماذا تقول الطريقة عن
        $\zeta(3)$ --- ولماذا لا تقول شيئًا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- نشر ظل التمام إلى كسور
جزئية.}
\begin{enumerate}[resume]
  \item ثبّت $w \in \C\setminus\Z$ وطبّق طريقة الجزء الثاني على $f(z) = \dfrac{1}{(z - w)(z + w)}$ ---
        ملاحظًا أن $\pi\cot(\pi z)f(z)$ صارت لها أقطاب بسيطة إضافية عند
        $\pm w$، ويجب أن تنضم بواقيها إلى العدّ. واستنتج نشر
        الكسور الجزئية
        \[
        \pi\cot(\pi w) = \frac1w +
        \sum_{n\geq1}\frac{2w}{w^2 - n^2},
        \]
        والمتسلسلة متقاربة تقاربًا ناظميًّا على الأجزاء
        المتراصة من $\C\setminus\Z$.
  \item استعد من هذا النشر، بنشر كل حدّ في قوى $w$ (وبرّر
        التبديل)، معاملاتِ لوران \emph{نفسها} كما في السؤال 2
        --- فتُغلق الدائرة: إذ صيغة أولير $\sum\frac1{n^2} = \frac{\pi^2}6$ هي معامل $w$
        في وجهَي ظل التمام. وقارن ذلك ببرهان متسلسلة فورييه
        (السنة الجامعية 2) وببرهان الأثر (\cref{pb:b3:spectral:1}): ثلاث
        نظريات، وعدد واحد.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- جداء أولير للجيب.} نشر
السؤال 9 هو المشتقة اللوغاريتمية لجداء لانهائي؛ ونبرهن الآن
على تحليل أولير لعام 1734 بأمانة.
\begin{enumerate}[resume]
  \item من أجل $N \geq 1$ نضع $P_N(z) =
        z\prod_{n=1}^{N}\bigl(1 - \frac{z^2}{n^2}\bigr)$. برهن على أن $P_N$
        تتقارب، بانتظام على كل قرص $\bar D(0, R)$، إلى دالة
        صحيحة $P$ جذورها هي الأعداد الصحيحة بالضبط، وكلها
        بسيطة. \emph{(من أجل $n
        \geq 2R$ اكتب العامل على الصورة
        $\exp\log(1 -
        z^2/n^2)$ باللوغاريتم الرئيسي في \cref{exo:b3:holomorphic:3}، وحُدّ
        $\abs{\log(1+u)}
        \leq 2\abs u$ من أجل $\abs u \leq \frac12$ عبر المتسلسلة، وخذ الأُسّي
        لمجموع اللوغاريتمات المتقارب ناظميًّا؛ والعوامل
        المنتهية الباقية كثيرُ حدود. واخلص باستعمال \cref{thm:b3:holomorphic:weierstrassconv}.)}
  \item برهن على أنه على $\C\setminus\Z$،
        \[
        \frac{P'(z)}{P(z)} = \frac1z +
        \sum_{n\geq1}\frac{2z}{z^2 - n^2} = \pi\cot(\pi z)
        \]
        \emph{(باشتقاق الجداءات المنتهية، والمرور إلى النهاية
        باستعمال \cref{thm:b3:holomorphic:weierstrassconv} وانعدامِ جذور $P$ خارج $\Z$،
        وباستشهاد السؤال 9)}.
  \item برهن على أن $Q = \sin(\pi z)/P(z)$ يمتدّ إلى دالة صحيحة عديمة الجذور
        تحقق $Q' = 0$، واخلص إلى \emph{جداء أولير}:
        \[
        \sin(\pi z) = \pi z\prod_{n\geq1}
        \Bigl(1 - \frac{z^2}{n^2}\Bigr)
        \qquad (z \in \C) .
        \]
  \item (واليس، 1655) قيّم عند $z = \frac12$:
        \[
        \frac\pi2 = \prod_{n\geq1}\frac{4n^2}{4n^2 - 1}
        = \lim_{N\to\infty}
        \frac{2\cdot2\cdot4\cdot4\cdots(2N)(2N)}
        {1\cdot3\cdot3\cdot5\cdots(2N-1)(2N+1)} .
        \]
  \item من أجل $\abs z < 1$، انشر لوغاريتم الجداء في متسلسلة
        مزدوجة (وبرّر إعادة الترتيب) واستعد $\zeta(2) =
        \frac{\pi^2}6$ بمطابقة
        معامل $z^3$ في $\sin(\pi z) = \pi z - \frac{\pi^3}6z^3 + \cdots$ --- أي وجه الجداء لعدد أولير.
\end{enumerate}

\medskip
\noindent\textbf{الجزء السادس --- النوى الشقيقة.} لظل التمام
أشقاء؛ ويسعّر كلٌّ منهم عائلته من المتسلسلات.
\begin{enumerate}[resume]
  \item اشتقّ نشر السؤال 9 حدًّا حدًّا (وهو مبرَّر
        باستعمال \cref{thm:b3:holomorphic:weierstrassconv}) لتحصل، ناظميًّا على متراصات
        $\C\setminus\Z$، على
        \[
        \frac{\pi^2}{\sin^2(\pi z)} =
        \sum_{n\in\Z}\frac1{(z - n)^2} .
        \]
  \item قيّم عند $z = \frac12$: $\sum_{m\geq0}\frac1{(2m+1)^2} = \frac{\pi^2}8$؛ واستعد $\zeta(2)$ مرة
        أخرى بشطر الأعداد الصحيحة حسب التماثل.
  \item تحقق من متطابقة المضاعفة $\tan\theta =
        \cot\theta - 2\cot(2\theta)$ واستنتج
        \[
        \pi\tan(\pi z) =
        \sum_{m\geq0}\frac{8z}{(2m+1)^2 - 4z^2} ,
        \]
        ناظميًّا على المتراصات التي تتفادى $\frac12 + \Z$.
  \item انشر حول $0$ ($\abs z < \frac12$؛ وبفوبيني من جديد): فمع $\lambda(s) =
        \sum_{m\geq0}(2m+1)^{-s}$،
        \[
        \pi\tan(\pi z) =
        \sum_{k\geq0}8\cdot4^k\,\lambda(2k+2)\,z^{2k+1} ;
        \]
        وقارن مع $\tan u = u + \frac{u^3}3 + O(u^5)$ لتستعيد $\lambda(2) = \frac{\pi^2}8$ ولتحصل على $\lambda(4) = \frac{\pi^4}{96}$، ثم
        تحقق تقاطعيًّا من $\zeta(4) = \frac{\pi^4}{90}$ عبر $\lambda(4) = (1 -
        2^{-4})\,\zeta(4)$.
  \item تحقق من $\frac1{\sin\theta} = \cot\frac\theta2 -
        \cot\theta$ واستنتج
        \[
        \frac{\pi}{\sin(\pi z)} = \frac1z +
        \sum_{n\geq1}(-1)^n\,\frac{2z}{z^2 - n^2} .
        \]
        وافحص الإشارات مقابل بواقي $\pi/\sin(\pi z)$ عند الأعداد
        الصحيحة.
  \item اقرأ معامل $z$: $\eta(2) =
        \sum_{n\geq1}\frac{(-1)^{n-1}}{n^2} =
        \frac{\pi^2}{12}$، وأكّد الاتساق $\eta(2) = (1 - 2^{1-2})\,\zeta(2)$.
  \item (الخاتمة) قيّم نشر السؤال 20 عند $z =
        \frac12$ واستنتج
        \emph{صيغة لايبنتز}
        \[
        \frac\pi4 = 1 - \frac13 + \frac15 - \frac17 +
        \cdots
        \]
        واختم بفقرة قصيرة: نواةٌ لكل حساب --- أي نواة تسعّر أي
        عائلة من المتسلسلات، ولماذا تكون كلها عمياء بنيويًّا
        عن $\zeta(3)$.

\end{enumerate}
\medskip
\noindent\textbf{الجزء السابع --- قائمة الأسعار الكاملة:
أعداد برنولي.}
\begin{enumerate}[resume]
  \item اجمع نشر الكسور الجزئية للمقدار $\pi
        z\cot(\pi z)$ مع الدالة
        المولّدة لأعداد برنولي ($\frac{w}{\eu^w - 1} =
        \sum_n\frac{B_n}{n!}w^n$، \cref{pb:b3:holomorphic:1}، الجزء
        السادس): فمن
        \[
        \pi z\cot(\pi z) = \iu\pi z +
        \frac{2\iu\pi z}{\eu^{2\iu\pi z} - 1}
        \]
        (وبرهن على هذه المتطابقة أولًا)، استنتج الصيغة المغلقة
        \[
        \zeta(2k) = (-1)^{k+1}\,
        \frac{(2\pi)^{2k}\,B_{2k}}{2\,(2k)!}
        \qquad (k \geq 1).
        \]
  \item تحقق من الصيغة مقابل $B_2 = \frac16$ و $B_4 =
        -\frac1{30}$
        و $B_6 = \frac1{42}$: فاستعد $\zeta(2) =
        \frac{\pi^2}6$ و $\zeta(4) = \frac{\pi^4}{90}$، واحسب $\zeta(6) = \frac{\pi^6}{945}$.
  \item (تراجع أولير) انشر طرفَي $\bigl(z\cot z\bigr)' = \cot z - z(1 + \cot^2z)$ --- أو ربّع متسلسلة
        ظل التمام مباشرةً --- لتبرهن على
        \[
        \Bigl(k + \frac12\Bigr)\zeta(2k) =
        \sum_{j=1}^{k-1}\zeta(2j)\,\zeta(2k - 2j)
        \qquad (k \geq 2),
        \]
        وتحقق من أنها تحسب $\zeta(4)$ من $\zeta(2)$ و
        $\zeta(6)$ من $\zeta(2), \zeta(4)$ --- أي جميع قيم زيتا الزوجية من
        البذرة الواحدة $\frac{\pi^2}6$، دون أي مكاملة جديدة.
\end{enumerate}
\end{problem}
