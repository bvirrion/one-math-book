\chapter{الدوال المميِّزة ومبرهنة النهاية
المركزية}\label{ch:b3:clt}

يقول قانون الأعداد الكبيرة إن المتوسطات تتقارب؛ وتقول مبرهنة
النهاية المركزية \emph{كيف تتذبذب}: أي إن الخطأ، مكبَّرًا
بالعامل $\sqrt n$، غاوسيٌّ مقاربًا --- أيًّا كان القانون الذي
انطلقنا منه. وهذا الشمول أعمق حقائق الاحتمالات الابتدائية،
وبرهانه الطبيعي فورييهي: إذ تحوّل \emph{\omterm{def:b3:clt:cf}{الدالة المميِّزة}} (أي
تحويل فورييه للقانون) المجاميعَ \omterm{def:b3:probability:independence}{المستقلة} إلى جداءات، وتحوّل
آلة \cref{ch:b3:fouriertransform} --- التباين، والنقاط الصامدة \omterm{def:b3:clt:gaussianvector}{الغاوسية} --- التقاربَ
النقطي لهذه الجداءات إلى تقارب في القوانين (مبرهنة ليفي،
مبرهَنًا عليها كاملةً). وينتهي الفصل بالمتجهات \omterm{def:b3:clt:gaussianvector}{الغاوسية}
والاستنباط الأمين لفترات الثقة المستعملة في كل مكان من
الإحصاء؛ وتعطي مسألة نهاية الأسبوع برهان ليندبرغ الثاني على
مبرهنة النهاية المركزية، بمعدل خطأ صريح.

\section{الدوال المميِّزة}

\begin{definition}\label{def:b3:clt:cf}
\emph{الدالة المميِّزة}\index{دالة مميِّزة} \omterm{def:b3:probability:space}{لمتغيّر عشوائي}
حقيقي $X$ هي
\[
\varphi_X(\xi) = \E\bigl[\eu^{\iu\xi X}\bigr]
= \int_\R \eu^{\iu\xi x}\,\dd\P_X(x)
\qquad (\xi \in \R)
\]
(وتحسبها مبرهنة النقل من القانون؛ ومن أجل كثافة $f$، $\varphi_X(\xi) = \hat f(-\xi)$
باصطلاح \cref{ch:b3:fouriertransform}).
\end{definition}

\begin{proposition}\label{prop:b3:clt:cfbasics}
(a) $\varphi_X(0) = 1$، $\abs{\varphi_X} \leq 1$، و $\varphi_X$ متصلة بانتظام؛ و $\varphi_{aX + b}(\xi) = \eu^{\iu b\xi}\varphi_X(a\xi)$.
(b) وإذا كان $X, Y$ \emph{مستقلين}: $\varphi_{X+Y} = \varphi_X\,\varphi_Y$.
(c) وإذا كان $\E\abs X^k < \infty$، فإن $\varphi_X \in \mathcal
C^k$ مع $\varphi_X^{(j)}(0) = \iu^j\,\E[X^j]$ من أجل $j \leq
k$؛ وعلى
وجه الخصوص، من أجل $X \in L^2$ موسَّط بتباين $\sigma^2$:
\[
\varphi_X(\xi) = 1 - \frac{\sigma^2\xi^2}{2} +
o(\xi^2) \qquad (\xi \to 0).
\]
(d) الغاوسي: للمقدار $X \sim \mathcal N(m, \sigma^2)$ الدالةُ $\varphi_X(\xi) = \eu^{\iu m\xi - \sigma^2\xi^2/2}$.
\end{proposition}

\begin{proof}
(a) الحدود فورية؛ وأما الاتصال: $\abs{\varphi(\xi + h)
- \varphi(\xi)} \leq \E\abs{\eu^{\iu hX} - 1} \to 0$ حين $h
\to 0$ بالتقارب
المهيمن، بانتظام في $\xi$. وقاعدة التآلف تعويض. (b) $\eu^{\iu\xi(X+Y)} =
\eu^{\iu\xi X}\eu^{\iu\xi Y}$،
وتتحلّل آمال جداءات المتغيّرات \omterm{def:b3:probability:independence}{المستقلة} (\cref{thm:b3:probability:independence}، مطبَّقةً
على الجزأين الحقيقي والتخيّلي). (c) بالاشتقاق تحت علامة \omterm{def:b3:probability:space}{الأمل}،
مهيمَنًا عليه بالمقدار $\E\abs X^j$ (\cref{thm:b3:lebesgue:paramdiff})؛ ونشر تايلور
عند $0$ هو عندئذٍ تايلور--يونغ للدالة $\varphi$ من الصنف
$\mathcal C^2$. (d) من أجل $\mathcal N(0,1)$: يعطي التحويل الغاوسي
(\cref{ex:b3:fouriertransform:gaussian} مع $a = \frac12$) أن $\int\eu^{\iu\xi x}\frac{\eu^{-x^2/2}}{\sqrt{2\pi}}\dd
x = \eu^{-\xi^2/2}$؛ والحالة العامة بقاعدة
التآلف.
\end{proof}

\begin{theorem}[التباين]\label{thm:b3:clt:injectivity}
إذا كان $\varphi_X = \varphi_Y$، فإن للمتغيّرين $X$ و $Y$ القانون نفسه. وبمزيد من
الدقة، من أجل $N \sim \mathcal N(0,1)$ مستقل عن $X$ ومن أجل $\varepsilon > 0$، يكون
للمتغيّر المنعَّم $X +
\varepsilon N$ الكثافةُ
\[
p_\varepsilon(x) = \frac1{2\pi}\int_\R
\varphi_X(-\xi)\,\eu^{-\varepsilon^2\xi^2/2}\,
\eu^{\iu\xi x}\,\dd\xi ,
\]
معيَّنةً بواسطة $\varphi_X$ وحدها؛ وبجعل $\varepsilon \to 0$ نستعيد قانون
$X$.\index{الدالة المميِّزة، التباين}
\end{theorem}

\begin{proof}
للمقدار $X + \varepsilon N$ الكثافةُ $p_\varepsilon(x) =
\E\bigl[g_\varepsilon(x - X)\bigr]$، حيث $g_\varepsilon$ كثافة
$\mathcal N(0, \varepsilon^2)$: إذ من أجل $B$ بوريلية، يعطي \omterm{def:b3:probability:independence}{الاستقلال} وتونيلي أن $\P(X + \varepsilon N \in
B) = \int\!\!\int\mathbf 1_B(x + \varepsilon
n)g_1(n)\,\dd n\,\dd\P_X(x) = \int_B\E[g_\varepsilon(t -
X)]\dd t$
(بالتعويض، ثم بتونيلي من جديد). وبكتابة $g_\varepsilon$ بقلب
فورييه لتحويلها (\cref{exo:b3:fouriertransform:4}، معادةَ التحجيم): $g_\varepsilon(u) = \frac1{2\pi}\int
\eu^{-\varepsilon^2\xi^2/2}\eu^{\iu\xi u}\dd\xi$، وبفوبيني
(إذ يُهيمَن على كل شيء بالعامل الغاوسي):
\[
p_\varepsilon(x) = \frac1{2\pi}\int_\R
\varphi_X(-\xi)\,\eu^{-\varepsilon^2\xi^2/2}\,
\eu^{\iu\xi x}\,\dd\xi ,
\]
وهي مؤثّرة على $\varphi_X$ وحدها. فإذا كان $\varphi_X =
\varphi_Y$: كان للمقدارين $X + \varepsilon N$
و $Y + \varepsilon N$ القوانين نفسها من أجل كل $\varepsilon$؛ ومن أجل $f$
محدودة متصلة، $\E f(X + \varepsilon N) \to \E f(X)$ حين $\varepsilon
\to 0$ (بالتقارب المهيمن، إذ $X + \varepsilon N \to X$
نقطةً نقطة على الفضاء الجدائي)، ومنه $\E f(X) = \E f(Y)$ من أجل كل $f$ كهذه
--- وهذا يعيّن القانون: إذ من أجل كل $t$، نحصر $\mathbf 1_{\intoc{-\infty}t}$ بين
المنحدرين المحدودين المتصلين $f_k^\pm$ (اللذين يساويان $1$ على
$\intoc{-\infty}{t \mp \frac1k}$ و $0$ وراء $t \pm
\frac1k$، وتآلفيان بينهما)؛ وبالمرور إلى النهاية
في $\E
f_k^-(X) \leq F_X(t) \leq \E f_k^+(X)$ نجد $F_X(t) =
F_Y(t)$ عند كل $t$ يكون عندها كلاهما \omterm{def:b3:topology:continuity}{متصلًا}، ومن
ثَمّ في كل مكان بالاتصال من اليمين وبكثافة نقاط الاتصال
المشتركة (إذ لكلتا الدالتين $F$ عددٌ قابل للعدّ من القفزات)؛ ودالتا
توزيع متساويتان تفرضان قانونين متساويين (\cref{exo:b3:measure:3}، قائمًا على
\cref{thm:b3:measure:uniqueness}).
\end{proof}

\section{التقارب في التوزيع}

\begin{definition}\label{def:b3:clt:cid}
يقال إن $X_n$ \emph{تتقارب في التوزيع} (أو في القانون) إلى $X$،
ويُكتب $X_n \Rightarrow X$، إذا كان\index{تقارب في التوزيع}
\[
\E\bigl[f(X_n)\bigr] \longrightarrow \E\bigl[f(X)\bigr]
\qquad\text{من أجل كل دالة محدودة متصلة} f\colon\R\to\R .
\]
أي، بالتكافؤ (\cref{exo:b3:clt:4}): $F_{X_n}(t) \to F_X(t)$ عند كل نقطة اتصال $t$ للدالة
$F_X$. ولا يلزم أن تعيش $X_n$ على \omterm{def:b3:probability:space}{فضاء احتمالي} مشترك: فالقوانين
وحدها هي المهمة.
\end{definition}

\begin{theorem}[مبرهنة هيلي في الانتقاء]
\label{thm:b3:clt:helly}
لكل متتالية $(F_n)$ من دوال التوزيع متتاليةٌ جزئية تتقارب نقطةً
نقطة، عند كل نقطة اتصال للنهاية، إلى دالة $G \colon
\R \to \intcc01$ متزايدة
بالمعنى الواسع ومتصلة من اليمين --- وقد يكون $G(+\infty) - G(-\infty) <
1$ (إذ قد
تهرب الكتلة إلى اللانهاية).\index{مبرهنة هيلي}
\end{theorem}

\begin{proof}
يعطي الاستخراج القطري $F_{n_k}(q) \to \ell(q)$ من أجل كل $q$ ناطق (بقيم في
المتراصة $\intcc01$). ونعرّف $G(t) = \inf\{\ell(q) : q \in \Q, q > t\}$: فهي متزايدة بالمعنى الواسع؛
ومتصلة من اليمين (بوصفها إنفيمومًا على جوارات ناطقة متقلّصة من
اليمين). وعند نقطة اتصال $t$ للدالة $G$: من أجل ناطقين
$q_1 < t < q_2$،
\[
\ell(q_1) \leq \liminf F_{n_k}(t) \leq \limsup F_{n_k}(t)
\leq \ell(q_2),
\]
برتابة كل $F_{n_k}$. ومن تعريف $G$ بوصفها إنفيمومًا ومن رتابة
$\ell$ على الأعداد الناطقة: $G(s) \leq \ell(q) \leq G(q)$ كلما كان $s < q$. وبأخذ
$s < q_1 < t$ نجد $\ell(q_1) \geq G(s)$، و $\ell(q_2) \leq G(q_2)$؛ وبجعل $s \uparrow t$
و $q_2
\downarrow t$، يحصر اتصالُ $G$ عند $t$ كلًّا من $\liminf$ و
$\limsup$ عند $G(t)$.
\end{proof}

\begin{lemma}[الإحكام انطلاقًا من الدالة المميِّزة]
\label{lem:b3:clt:tightness}
من أجل أي \omterm{def:b3:probability:space}{متغيّر عشوائي} $X$ ومن أجل $u > 0$:
\[
\P\Bigl(\abs X \geq \frac2u\Bigr) \;\leq\;
\frac1u\int_{-u}^{u}\bigl(1 -
\operatorname{Re}\varphi_X(\xi)\bigr)\,\dd\xi .
\]
\end{lemma}

\begin{proof}
بتونيلي--فوبيني (إذ المقدار المكامَل محدود والمنطقة منتهية في
$\xi$):
\[
\frac1u\int_{-u}^u\bigl(1 -
\operatorname{Re}\varphi_X(\xi)\bigr)\dd\xi
= \E\Bigl[\frac1u\int_{-u}^u(1 - \cos(\xi X))\,\dd\xi\Bigr]
= 2\,\E\Bigl[1 - \frac{\sin(uX)}{uX}\Bigr]
\]
(ونفسّر المقدار بين القوسين بنهايته $0$ عند $X = 0$). والمقدار
المكامَل غير سالب ($\abs{\sin t} \leq \abs t$)، ومن أجل $\abs{uX} \geq 2$: $1 - \frac{\sin(uX)}{uX} \geq 1 -
\frac1{\abs{uX}} \geq \frac12$. ومنه فإن
الاحتفاظ بالحادثة $\{\abs{uX} \geq 2\}$ وحدها داخل \omterm{def:b3:probability:space}{الأمل} يترك على الأقل
$2 \cdot \frac12\,\P(\abs X \geq \frac2u)$، وهو الادعاء.
\end{proof}

\begin{theorem}[مبرهنة ليفي في الاتصال]
\label{thm:b3:clt:levy}
لتكن $(X_n)$ متغيّرات عشوائية تتقارب دوالها المميِّزة نقطةً
نقطة: $\varphi_{X_n}(\xi) \to
\varphi(\xi)$ من أجل كل $\xi$، حيث $\varphi =
\varphi_X$ هي \omterm{def:b3:clt:cf}{الدالة المميِّزة}
\omterm{def:b3:probability:space}{لمتغيّر عشوائي} $X$ ما. عندئذٍ $X_n \Rightarrow X$.\index{مبرهنة ليفي في
الاتصال}
\end{theorem}

\begin{proof}
\emph{الإحكام.} نثبّت $\varepsilon > 0$. وبما أن $\varphi$ متصلة عند $0$
مع $\varphi(0) = 1$، نختار $u > 0$ يحقق $\frac1u\int_{-u}^u(1 - \operatorname{Re}\varphi) <
\varepsilon$؛ وبالتقارب
المهيمن (إذ يُهيمَن على المقدار المكامَل بالعدد $2$ على الفترة
الثابتة $[-u,u]$)، يكون التكامل نفسه من أجل $\varphi_{X_n}$
$< 2\varepsilon$ من أجل $n$ كبير: فتعطي \cref{lem:b3:clt:tightness} أن $\P(\abs{X_n} \geq \frac2u)
\leq 2\varepsilon$ من
أجل $n$ كبير، ويعالج توسيعُ الثابت العددَ المنتهي الباقي:
فتكون القوانين \emph{محكمة} --- إذ لا تهرب أي كتلة.

\emph{المتتاليات الجزئية.} لتكن $(F_{n_k})$ أي متتالية جزئية؛
فبهيلي (\cref{thm:b3:clt:helly}) نستخرج $F_{n_{k_j}} \to G$ عند نقاط الاتصال. ويفرض
الإحكام أن $G(-\infty) = 0$ و $G(+\infty) = 1$ ($G(\frac2u) - G(-\frac2u) \geq 1 -
2\varepsilon$ عند نقاط
الاتصال): فتكون $G$ دالةَ توزيع فعلية \omterm{def:b3:probability:space}{لمتغيّر عشوائي} $Y$ ما.
عندئذٍ $X_{n_{k_j}} \Rightarrow Y$ (\cref{exo:b3:clt:4}، بالتقارب التوزيعي انطلاقًا من دوال
$F$)، ومنه $\varphi_{X_{n_{k_j}}} \to \varphi_Y$ \emph{نقطةً نقطة} (إذ $x
\mapsto \eu^{\iu\xi x}$ محدودة متصلة،
بالجزأين الحقيقي والتخيّلي على حدة)؛ وبالمقارنة مع الفرضية:
$\varphi_Y = \varphi = \varphi_X$، ويعطي التباين (\cref{thm:b3:clt:injectivity}) أن $Y \sim X$، أي $G =
F_X$.

\emph{الخاتمة.} لكل متتالية جزئية من $(F_n)$ متتاليةٌ جزئية
أخرى تتقارب إلى $F_X$ \emph{نفسها} (عند نقاط اتصالها)؛ ومن ثَمّ
$F_n(t) \to F_X(t)$ عند كل نقطة اتصال $t$ (إذ متتالية حقيقية لكل متتالياتها
الجزئية متتالياتٌ جزئية أخرى بالنهاية نفسها تتقارب): أي $X_n \Rightarrow X$.
\end{proof}

\section{مبرهنة النهاية المركزية}

\begin{theorem}[مبرهنة النهاية المركزية]\label{thm:b3:clt:clt}
لتكن $(X_n)$ \omterm{def:b3:probability:independence}{مستقلة} متماثلة مع $\E X_1 = m$ و $\V(X_1) =
\sigma^2 \in \intoo0\infty$.
عندئذٍ\index{مبرهنة النهاية المركزية}
\[
\frac{S_n - nm}{\sigma\sqrt n} \;\Longrightarrow\; \mathcal
N(0, 1) :
\qquad
\P\Bigl(a \leq \frac{S_n - nm}{\sigma\sqrt n} \leq
b\Bigr) \longrightarrow
\frac{1}{\sqrt{2\pi}}\int_a^b\eu^{-x^2/2}\,\dd x
\]
من أجل كل $a < b$.
\end{theorem}

\begin{proof}
نوسّط ونعيّر: $Z_i = \frac{X_i - m}{\sigma}$ (\omterm{def:b3:probability:independence}{مستقلة} متماثلة، بمتوسط $0$ وتباين $1$)
و $T_n =
\frac1{\sqrt n}\sum_{i\leq n}Z_i$. وبالاستقلال وقاعدة التآلف (\cref{prop:b3:clt:cfbasics}):
\[
\varphi_{T_n}(\xi) =
\varphi_{Z}\Bigl(\frac{\xi}{\sqrt n}\Bigr)^{n},
\qquad
\varphi_Z(\eta) = 1 - \frac{\eta^2}2 + \eta^2\rho(\eta),\quad
\rho(\eta)\to0 .
\]
نثبّت $\xi$ ونضع $a_n = \varphi_Z(\xi/\sqrt n)$، $b_n = 1 -
\frac{\xi^2}{2n}$: فلكليهما طويلة $\leq 1$ من
أجل $n$ كبير (إذ $\abs{b_n} \leq 1$ بمجرد أن يكون $\xi^2 \leq 4n$؛ و
$\abs{a_n} \leq 1$ دائمًا). وتعطي المتراجحة الابتدائية $\abs{a^n - b^n} \leq
n\abs{a - b}$ من أجل
$\abs a, \abs b \leq 1$ (بتلسكب $a^n - b^n = \sum a^k(a - b)b^{n-1-k}$) أن
\[
\Bigl|\varphi_{T_n}(\xi) - \Bigl(1 -
\frac{\xi^2}{2n}\Bigr)^{n}\Bigr|
\leq n\,\Bigl|\varphi_Z\Bigl(\frac\xi{\sqrt n}\Bigr) - 1 +
\frac{\xi^2}{2n}\Bigr|
= \xi^2\,\Bigl|\rho\Bigl(\frac{\xi}{\sqrt n}\Bigr)\Bigr|
\longrightarrow 0,
\]
بينما $\bigl(1 - \frac{\xi^2}{2n}\bigr)^n \to
\eu^{-\xi^2/2}$ (باللوغاريتم الحقيقي). ومنه $\varphi_{T_n}(\xi) \to
\eu^{-\xi^2/2} = \varphi_{\mathcal N(0,1)}(\xi)$
(\cref{prop:b3:clt:cfbasics}(d)) من أجل كل $\xi$: فتخلص ليفي (\cref{thm:b3:clt:levy}) إلى
$T_n \Rightarrow \mathcal
N(0,1)$. وتنتج احتمالات الفترات لأن $F_{\mathcal
N}$ متصلة في كل مكان.
\end{proof}

\begin{example}[فترات الثقة، مستنبطةً بأمانة]
\label{ex:b3:clt:confidence}
نستطلع $n$ من الناخبين المستقلين؛ ويقدّر $\hat p_n = S_n/n$ النسبةَ
الحقيقية $p$، مع $\sigma^2 = p(1-p) \leq \frac14$. وتعطي مبرهنة النهاية المركزية، من أجل
$n$ كبير،
\[
\P\Bigl(\abs{\hat p_n - p} \leq
\frac{z}{2\sqrt n}\Bigr)
\;\geq\; \P\Bigl(\Bigl|\frac{S_n - np}{\sigma\sqrt n}\Bigr|
\leq z\Bigr)
\longrightarrow \Phi(z) - \Phi(-z),
\]
حيث $\Phi$ دالة التوزيع \omterm{def:b3:clt:gaussianvector}{الغاوسية} المعيارية. ومع $z = 1.96$:
ثقةٌ مقاربة $95\%$، ويتطلب الهامش $\frac{1.96}{2\sqrt n} \leq 3\%$ أن يكون $n \geq
\bigl(\frac{1.96}{0.06}\bigr)^2 \approx 1068$ ---
أي العدد الكامن وراء كل «$\pm3$ نقاط، $95\%$» يقرأها المرء؛
وقارن ذلك بعدد تشيبيشيف $5556$ (\cref{exo:b3:probability:7}). والعامل $\sqrt n$
شامل: فلتنصيف الخطأ، ضاعف العيّنة أربع مرات --- وهو القانون
نفسه الذي يحدّد كلفة \omterm{ex:b3:probability:sllnapps}{مونتي كارلو} (\cref{exo:b3:clt:7}).
\end{example}

\section{المتجهات الغاوسية}

\begin{definition}\label{def:b3:clt:gaussianvector}
تكون المتجهة العشوائية $X = (X_1, \dots, X_d)$ \emph{غاوسية}\index{متجهة
غاوسية} إذا كانت كل تركيبة خطية $\langle t, X\rangle = \sum t_iX_i$ متغيّرًا غاوسيًّا
حقيقيًّا (وقد يكون منحلًّا). ويتعيّن قانونها بمتجهة المتوسط
$m = (\E X_i)$ و\emph{مصفوفة التغاير} $\Sigma = \bigl(\operatorname{Cov}
(X_i, X_j)\bigr)$: إذ إن \omterm{def:b3:clt:cf}{الدالة
المميِّزة} للمتجهة، $\varphi_X(t) = \E\eu^{\iu\langle t, X\rangle}$، هي قيمة \omterm{def:b3:clt:cf}{الدالة المميِّزة} للمقدار
$\langle t, X\rangle$ عند $1$:
\[
\varphi_X(t) = \exp\Bigl(\iu\langle t, m\rangle -
\tfrac12\,t^{\mathsf T}\Sigma\,t\Bigr),
\]
والدوال المميِّزة ذات البُعد $d$ متباينة (ببرهان التنعيم نفسه
كما في \cref{thm:b3:clt:injectivity}، بغاوسيات إحداثية).
\end{definition}

\begin{theorem}\label{thm:b3:clt:gaussianvector}
لتكن $X$ متجهةً \omterm{def:b3:clt:gaussianvector}{غاوسية}.
\begin{enumerate}
  \item كل صورة تآلفية $AX + b$ متجهةٌ \omterm{def:b3:clt:gaussianvector}{غاوسية}.
  \item والمركّبات $X_i$ \emph{\omterm{def:b3:probability:independence}{مستقلة}} إذا وفقط إذا كانت
        $\Sigma$ قطرية: أي إنه من أجل متغيّرات \omterm{def:b3:clt:gaussianvector}{غاوسية} معًا،
        عدمُ الارتباط $=$ \omterm{def:b3:probability:independence}{الاستقلال}.
  \item وإذا كانت $\Sigma$ قابلة للقلب، كانت للمتجهة $X$ الكثافةُ
        $\frac{1}{(2\pi)^{d/2}\sqrt{\det\Sigma}}
        \exp\bigl(-\frac12(x - m)^{\mathsf T}\Sigma^{-1}(x -
        m)\bigr)$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) التركيبات الخطية لمركّبات $AX + b$ دوالٌّ تآلفية لتركيبات
خطية من $X$: أي \omterm{def:b3:clt:gaussianvector}{غاوسية} (إذ الصورة التآلفية لمتغيّر غاوسي
\omterm{def:b3:clt:gaussianvector}{غاوسية}).
(2) إذا كانت $\Sigma$ قطرية، تحلّلت \omterm{def:b3:clt:cf}{الدالة المميِّزة}: $\varphi_X(t) = \prod_i\exp(\iu t_im_i -
\frac12\Sigma_{ii}t_i^2) = \prod\varphi_{X_i}(t_i)$،
وهي \omterm{def:b3:clt:cf}{الدالة المميِّزة} للقانون الجدائي (\cref{thm:b3:probability:independence} مقروءةً عبر
التباين ذي البُعد $d$): فتكون المركّبات \omterm{def:b3:probability:independence}{مستقلة}. والعكس هو انعدام
تغايرات المتغيّرات \omterm{def:b3:probability:independence}{المستقلة} من $L^2$.
(3) نقطرن $\Sigma = P D P^{\mathsf T}$ (حيث $P$ متعامدة و $D > 0$ قطرية ---
\cref{exo:b3:submanifolds:8})؛ فتكون المتجهة $Y = P^{\mathsf T}(X - m)$ \omterm{def:b3:clt:gaussianvector}{غاوسية} بتغاير $D$: وحسب (2)
تكون مركّباتها \omterm{def:b3:probability:independence}{مستقلة} $\mathcal N(0, d_i)$، ومنه تكون للمتجهة $Y$ الكثافة
الجدائية؛ ثم ندفعها بالتطبيق الحافظ للحجم $x = m + PY$
(\cref{thm:b3:product:linearchange}، $\abs{\det P} = 1$) ونعيد كتابة الأُسّ بكيفية صامدة.
\end{proof}

\begin{theorem}[مبرهنة النهاية المركزية المتعددة الأبعاد]
\label{thm:b3:clt:multiclt}
لتكن $(X_n)$ \emph{متجهات} عشوائية \omterm{def:b3:probability:independence}{مستقلة} متماثلة \omterm{def:b3:lebesgue:l1}{قابلة
للمكاملة} بالمربّع من $\R^d$ بمتوسط $m$ ومصفوفة تغاير $\Sigma$.
عندئذٍ تتقارب $\frac{S_n - nm}{\sqrt n}$ في التوزيع إلى \omterm{def:b3:clt:gaussianvector}{المتجهة الغاوسية}
$\mathcal N(0,
\Sigma)$.\index{مبرهنة النهاية المركزية!المتعددة الأبعاد}
\admitted
\end{theorem}

\begin{remark}\label{rem:b3:clt:multiclt}
كل شيء تقريبًا بين أيدينا سلفًا. فمن أجل كل اتجاه $t \in \R^d$،
يكون المتغيّر الحقيقي $\langle t,
\frac{S_n - nm}{\sqrt n}\rangle$ مجموعًا معيَّرًا لمتغيّرات حقيقية
\omterm{def:b3:probability:independence}{مستقلة} متماثلة تباينها $t^{\mathsf T}\Sigma t$، ومنه يعطي حساب \cref{thm:b3:clt:clt}
التقاربَ النقطي للدوال المميِّزة ذات البُعد $d$ إلى $\eu^{-t^{\mathsf T}\Sigma t/2}$، وهي
\omterm{def:b3:clt:cf}{الدالة المميِّزة} للمقدار $\mathcal N(0, \Sigma)$ (\cref{def:b3:clt:gaussianvector}). وما لم نعد
البرهان عليه هو مبرهنة ليفي في الاتصال \emph{في $\R^d$}: إذ
يُعمَّم انتقاء هيلي وتقدير الإحكام روتينيًّا (إحداثيًّا)،
ويُنجَز إرجاع كرامر--وولد هذا بأمانة في أي مقرر احتمالات عليا؛
ولا يلزم شيء وراء طرق هذا الفصل.
\end{remark}

\begin{method}\label{met:b3:clt:method}
لتعيين قانون نهائي: احسب الدوال المميِّزة، وخذ النهاية النقطية،
وتعرّف عليها (غاوسي $\eu^{-\sigma^2\xi^2/2}$، بواسون $\eu^{\lambda(\eu^{\iu\xi}-1)}$، أُسّي $\frac{\lambda}
{\lambda - \iu\xi}$،
\dots) واستدعِ ليفي. والطقس ذو الخطوات الثلاث (\omterm{def:b3:probability:independence}{الاستقلال} $\to$
الجداء؛ وتايلور عند $0$ $\to$ النهاية الأُسّية؛ وليفي $\to$
التقارب في القانون) يبرهن على مبرهنة النهاية المركزية، وعلى
قانون بواسون للحوادث النادرة (\cref{exo:b3:clt:5})، وعلى كل مبرهنة
نهاية كلاسيكية في هذا المقرر. وأما العبارات شبه الأكيدة، فارجع
إلى عدّة \cref{ch:b3:probability}: فالفصلان يجيبان عن سؤالين مختلفين عن
$S_n$ نفسه.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b3:clt:1}
احسب الدوال المميِّزة: للمنتظم على $\intcc{-1}1$؛ وللأُسّي
$\mathcal E(\lambda)$؛ ولبواسون $\mathcal P(\lambda)$؛ وللثنائي $\mathcal B(n, p)$. واستنتج عبر
\cref{thm:b3:clt:injectivity} أن مجموع متغيّرين بواسونيين مستقلين ($\lambda, \mu$)
بواسونيٌّ $(\lambda +
\mu)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:clt:2}
(a) برهن على أن $\varphi_X$ ذات قيم حقيقية إذا وفقط إذا كان
للمتغيّرين $X$ و $-X$ القانون نفسه (أي متغيّر \emph{متناظر}).
(b) لنفترض أن $\abs{\varphi_X(\xi_0)} = 1$ من أجل $\xi_0 \neq
0$ ما. برهن على أن حامل $X$
شبه أكيد على متتالية حسابية $a + \frac{2\pi}{\xi_0}\Z$ \emph{(اكتب $\varphi_X(\xi_0) = \eu^{\iu\theta}$ واحسب
$\E[1 -
\cos(\xi_0X - \theta)]$)}. واستنتج أنه إذا قبل $X$ كثافة، فإن $\abs{\varphi_X(\xi)} < 1$ من
أجل كل $\xi \neq 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:clt:3}
لتكن $X \sim \mathcal N(m_1, \sigma_1^2)$ و $Y \sim
\mathcal N(m_2, \sigma_2^2)$ مستقلتين. برهن على $X + Y
\sim \mathcal N(m_1 + m_2, \sigma_1^2 + \sigma_2^2)$، وعمومًا على
أن العائلة \omterm{def:b3:clt:gaussianvector}{الغاوسية} مستقرة بالمجاميع \omterm{def:b3:probability:independence}{المستقلة} وبالتطبيقات
التآلفية. وقابل ذلك: هل مجموع غاوسيين \emph{تابعين} غاوسيٌّ
دائمًا؟ (\cref{exo:b3:clt:9}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:clt:4}
(a) برهن على التكافؤ في \cref{def:b3:clt:cid}: أنه إذا كان $\E f(X_n) \to \E f(X)$ من أجل
كل $f$ محدودة متصلة، فإن $F_{X_n}(t) \to F_X(t)$ عند نقاط الاتصال \emph{(بحصر
$\mathbf 1_{\intoc{-\infty}t}$ بين منحدرين سلّميين متصلين)}؛ وبالعكس \emph{(بتقريب
دالة محدودة متصلة $f$ بمجاميع دوال منحدرة، أو بالشرطنة على
شبكة دقيقة من نقاط الاتصال)} --- ويمكن معالجة العكس من أجل $f$
متصلة بانتظام أولًا، ثم في العموم.
(b) برهن على أن $X_n \Rightarrow c$ (وهو ثابت) يستلزم $X_n
\to c$ احتماليًّا.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:clt:5}
(قانون الحوادث النادرة) لتكن $X_n \sim \mathcal B(n, p_n)$ مع $np_n \to \lambda > 0$. برهن، عبر
الدوال المميِّزة و\cref{thm:b3:clt:levy}، على أن $X_n \Rightarrow \mathcal
P(\lambda)$. وفحص سلامة عددي:
قارن $\P(X = 0)$ من أجل $\mathcal B(100, 0.02)$ و $\mathcal P(2)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:clt:6}
(a) يُرمى حجر نرد عادل $n = 1000$ مرة؛ قرّب احتمال أن يتجاوز
المجموع $3600$ (بمتوسط $3500$ وتباين $\frac{35}{12}$ لكل
رمية).
(b) ومن أجل $S \sim \mathcal B(100, \frac12)$، قرّب $\P(45 \leq S \leq 55)$ بمبرهنة النهاية المركزية مع
تصحيح الاتصال ($\pm\frac12$)، وعلّق على أثر التصحيح.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:clt:7}
(خطأ \omterm{ex:b3:probability:sllnapps}{مونتي كارلو}) في إطار \cref{pb:b3:probability:1}، السؤال 11، مع $g \in
L^2(\intcc01^d)$،
لتكن $\sigma^2 = \V(g(U_1))$ ولتكن $I = \int
g$. برهن على
\[
\sqrt n\,\Bigl(\frac1n\sum_{k\leq n}g(U_k) - I\Bigr)
\Longrightarrow \mathcal N(0, \sigma^2),
\]
واستنتج شريط الخطأ المقارب $95\%$ $\pm
1.96\,\sigma/\sqrt n$ --- وهو مستقل عن
البُعد $d$. وقارن ذلك بقاعدة النقطة الوسطى الحتمية في البُعد
$d$ (بخطأ $\sim n^{-2/d}$ من أجل مقادير مكامَلة من الصنف
$\mathcal C^2$): فمن أي بُعد فصاعدًا تفوز المعاينة العشوائية؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:clt:8}
(سلوتسكي) لنفترض أن $X_n \Rightarrow X$ وأن $Y_n \to c$ احتماليًّا (حيث $c$
ثابت). برهن على $X_n + Y_n \Rightarrow X +
c$ وعلى $Y_nX_n \Rightarrow cX$. \emph{(اعمل بالدوال
المميِّزة وبالحدّ $\abs{\E\eu^{\iu\xi
(X_n+Y_n)} - \eu^{\iu\xi c}\E\eu^{\iu\xi X_n}} \leq
\E\abs{\eu^{\iu\xi(Y_n - c)} - 1}$، بالشطر على $\abs{Y_n - c}
\leq \delta$.)} وتطبيقٌ: في
\cref{ex:b3:clt:confidence}، برّر استبدال $\sigma = \sqrt{p(1-p)}$ المجهول بالمقدار $\sqrt{\hat p_n(1 - \hat
p_n)}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b3:clt:9}
لتكن $X \sim \mathcal N(0,1)$ و $\varepsilon$ مستقلين مع $\P(\varepsilon = \pm1) = \frac12$؛ ونضع $Y =
\varepsilon X$.
(a) برهن على $Y \sim \mathcal N(0,1)$ وعلى $\operatorname{Cov}(X, Y) = 0$.
(b) برهن على أن $X$ و $Y$ \emph{ليسا} مستقلين، وعلى أن
$(X, Y)$ ليست متجهةً \omterm{def:b3:clt:gaussianvector}{غاوسية} \emph{(احسب $\P(X + Y =
0)$)}.
(c) العبرة: أن \cref{thm:b3:clt:gaussianvector}(2) تتطلب \omterm{def:b3:clt:gaussianvector}{الغاوسية} المشتركة ---
أما «غاوسيان غير مرتبطان» وحدهما لا يبرهنان على شيء.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:clt:10}
لقانون كوشي الكثافةُ $\frac1{\pi(1 + x^2)}$.
(a) برهن على أن دالته المميِّزة $\eu^{-\abs\xi}$ (\cref{exo:b3:fouriertransform:1}
والقلب).
(b) برهن على أنه إذا كانت $X_1, \dots, X_n$ \omterm{def:b3:probability:independence}{مستقلة} متماثلة كوشية، فإن
$\frac{S_n}n$ كوشيةٌ من جديد --- بالقانون \emph{نفسه}: فالمتوسط
لا يتركّز أبدًا.
(c) ووفّق ذلك مع قانونَي الأعداد الكبيرة ومع مبرهنة النهاية
المركزية: فأي الفرضيات تخفق؟ (احسب $\E\abs{X_1}$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:clt:11}
(القوانين المستقرة في المهد) لتكن $(X_n)$ \omterm{def:b3:probability:independence}{مستقلة} متماثلة كوشية
معيارية (\cref{exo:b3:clt:10}).
(a) برهن على أنه من أجل أي $a, b > 0$، يكون للمقدار $aX_1 + bX_2$
قانونُ $(a + b)X_1$: أي إن عائلة كوشي \emph{مستقرة تمامًا}
بالدليل $1$.
(b) برهن على أن العائلة \omterm{def:b3:clt:gaussianvector}{الغاوسية} مستقرة تمامًا بالدليل $2$:
$aX_1 + bX_2 \sim \sqrt{a^2 + b^2}\,X_1$ من أجل $X_i$ \omterm{def:b3:probability:independence}{مستقلة} متماثلة $\mathcal N(0,1)$.
(c) اشرح، عبر الدوال المميِّزة من الشكل $\eu^{-c\abs\xi^\alpha}$، لماذا يفرض
الاستقرار بالدليل $\alpha$ التعييرَ $n^{1/\alpha}$ من أجل
المجاميع، وماذا يقول ذلك عن أحواض جذب مبرهنة النهاية المركزية:
أي المجاميع \omterm{def:b3:probability:independence}{المستقلة} المتماثلة يمكنها أن تتقارب، بعد تعيير
تآلفي، إلى قانون كوشي بدل الغاوسي؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b3:clt:12}
(دالة التوزيع التجريبية) لتكن $(X_n)$ \omterm{def:b3:probability:independence}{مستقلة} متماثلة بدالة
توزيع $F$، ولتكن $F_n(t) =
\frac1n\#\{k \leq n : X_k \leq t\}$.
(a) نثبّت $t$. برهن على أن $n F_n(t) \sim \mathcal B(n, F(t))$، وعلى أن $F_n(t) \to F(t)$ شبه أكيد
(\cref{thm:b3:probability:slln})، وعلى أن
\[
\sqrt n\,\bigl(F_n(t) - F(t)\bigr) \Longrightarrow
\mathcal N\bigl(0,\ F(t)(1 - F(t))\bigr) .
\]
(b) وعند أي $t$ يكون التباين المقارب أعظميًّا؟ فسّر: أن الوسيط
هو الموضع الذي يصعب فيه تثبيت توزيع تجريبي أكثر ما يكون.
(c) ومن أجل $F$ متصلة، برهن على أن قانون $\sup_t\abs{F_n(t) - F(t)}$ لا يتعلق
بالدالة $F$ \emph{(بالإرجاع إلى متغيّرات منتظمة عبر \cref{exo:b3:probability:1})}
--- أي المعجزة الخالية من التوزيع وراء اختبار
كولموغوروف--سميرنوف؛ ولا يُطلب حساب ذلك القانون.
\end{exercise}

\section{مسألة: برهان ليندبرغ على مبرهنة النهاية المركزية،
بمعدل}

\begin{problem}[{مسألة نهاية الأسبوع --- طريقة الاستبدال}]
\label{pb:b3:clt:1}
برهن ليندبرغ (1922) على مبرهنة النهاية المركزية بفكرة مخلوعة
البساطة: \emph{استبدل الحدود واحدًا تلو الآخر بغاوسيات} وتحكم
في كل استبدال بنشر تايلور. ولا تحتاج الطريقة إلى أي تحليل
فورييهي، وتنتج معدل خطأ صريحًا، وهي تشغّل اليوم براهين الشمول
في أنحاء نظرية الاحتمالات. لتكن $(X_i)$ \omterm{def:b3:probability:independence}{مستقلة} متماثلة موسَّطة
مع $\V(X_1) = 1$ و $\beta = \E\abs{X_1}^3 < \infty$؛ ولتكن $(N_i)$ \omterm{def:b3:probability:independence}{مستقلة} متماثلة
$\mathcal N(0,1)$، \omterm{def:b3:probability:independence}{مستقلة} عن $X_i$ (والوجود: \cref{thm:b3:probability:existence}). ونضع
\[
T_n = \frac{X_1 + \dots + X_n}{\sqrt n},
\qquad
G_n = \frac{N_1 + \dots + N_n}{\sqrt n} \sim \mathcal N(0,1).
\]

\medskip
\noindent\textbf{الجزء الأول --- متطابقة الاستبدال.} نثبّت
$f
\in \mathcal C^3_b(\R)$ (بثلاث مشتقات محدودة متصلة؛ $M_3 = \sup\abs{f'''}$). ومن أجل $0 \leq i \leq n$
نعرّف المجاميع الهجينة
\[
H_i = \frac{X_1 + \dots + X_i + N_{i+1} + \dots +
N_n}{\sqrt n},
\]
بحيث $H_n = T_n$ و $H_0 = G_n$.
\begin{enumerate}
  \item اكتب $H_i = W_i + \frac{X_i}{\sqrt n}$ و $H_{i-1}
        = W_i + \frac{N_i}{\sqrt n}$ مع $W_i =
        \frac{1}{\sqrt n}\bigl(\sum_{j<i}X_j +
        \sum_{j>i}N_j\bigr)$، ولاحظ أن $W_i$ مستقل
        عن الزوج $(X_i, N_i)$. برّر ذلك.
  \item تايلور \omterm{def:b3:residues:singularities}{بباقٍ} تكاملي أو باقي لاغرانج: من أجل أي
        $w, h$ حقيقيين:
        \[
        \Bigl|f(w + h) - f(w) - f'(w)h -
        \tfrac12f''(w)h^2\Bigr| \leq
        \frac{M_3\,\abs h^3}{6} .
        \]
  \item طبّق السؤال 2 مرتين ($h = \frac{X_i}{\sqrt n}$ و $h = \frac{N_i}{\sqrt n}$ عند $w = W_i$)،
        وخذ الآمال، واستعمل \omterm{def:b3:probability:independence}{الاستقلال} مع تطابق العزمين
        الأولين للمتغيّرين $X_i$ و $N_i$ لتبرهن على
        \[
        \bigl|\E f(H_i) - \E f(H_{i-1})\bigr|
        \leq \frac{M_3}{6}\cdot
        \frac{\beta + \gamma}{n^{3/2}},
        \qquad \gamma = \E\abs{N_1}^3 =
        \frac{2\sqrt2}{\sqrt\pi} .
        \]
  \item تلسكب على $i$ واخلص إلى \emph{حدّ ليندبرغ}:
        \[
        \bigl|\E f(T_n) - \E f(G_n)\bigr| \leq
        \frac{M_3\,(\beta + \gamma)}{6\,\sqrt n} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- من $f$ الناعمة إلى مبرهنة
النهاية المركزية.}
\begin{enumerate}[resume]
  \item برهن على $\E f(T_n) \to \E f(N)$ من أجل كل $f \in
        \mathcal C_b^3$، وارقَ إلى جميع
        الدوال المحدودة المتصلة $f$: بإعطاء $f$ كهذه ومع
        $\varepsilon$، ابنِ $f_\varepsilon \in \mathcal C^3_b$ تحقق
        $\norm{f - f_\varepsilon}_\infty \leq \varepsilon$
        على فترة كبيرة --- مثلًا بالتفاف $f$ مع نتوء $\mathcal C^\infty$
        (\cref{thm:b3:lp:regularization}) --- وعالج الذيول بالإحكام ($\V(T_n) = 1$
        وتشيبيشيف). واخلص إلى $T_n \Rightarrow \mathcal N(0, 1)$: أي مبرهنة النهاية
        المركزية، معادًا البرهان عليها.
  \item وأين استعمل البرهان كون $X_i$ \emph{متماثلة} التوزيع؟
        برهن على أنه بالكاد فعل: صُغ وبرهن على الصيغة من أجل
        $X_i$ \omterm{def:b3:probability:independence}{مستقلة} موسَّطة غير متماثلة مع $\sum_i\V(X_i) = s_n^2$ وبعزوم
        ثالثة، فتحصل على الخطأ $\frac{M_3}{6s_n^3}\sum_i\bigl(\E\abs{X_i}^3
        + \V(X_i)^{3/2}\gamma\bigr)$ --- أي مبرهنة ليندبرغ
        الحقيقية في صيغتها لليابونوف.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- عوائد كمّية.}
\begin{enumerate}[resume]
  \item (دوال التوزيع) ليكن $t \in \R$ وقرّب $\mathbf 1_{\intoc{-\infty}t}$ من الأعلى
        ومن الأسفل بمنحدرات من الصنف $\mathcal C^3_b$ عرضها
        $\delta$ (ابنِها، مع $M_3 = O(\delta^{-3})$). وبضمّ ذلك إلى الجزء
        الأول، استنبط الحدَّ ذا الحدّين
        \[
        \sup_{t\in\R}\,\bigl|\P(T_n \leq t) -
        \Phi(t)\bigr| \;\leq\;
        \frac{C_1(\beta + \gamma)}{\delta^3\sqrt n} +
        C_2\,\delta
        \qquad (\text{من أجل كل} \delta > 0),
        \]
        بثوابت صريحة (ويستعمل الحدّ $C_2\delta$ أن للدالة $\Phi$
        كثافةً محدودة بالمقدار $\frac1{\sqrt{2\pi}}$)، وأمثِل $\delta \sim
        n^{-1/8}$ لتحصل
        على معدل منتظم من رتبة $n^{-1/8}$.
        (والمعدل الأمثل $n^{-1/2}$ --- أي بيري--إسين ---
        يتطلب أدوات أدق؛ والمقصود هو معدلٌ \emph{صريح}
        بالاستبدال الابتدائي.)
  \item (دي موافر--لابلاس، مكمَّمة) خصّص إلى $X_i = 2B_i - 1$
        (أي إشارات قطع نقدية عادلة): وقارن النتيجة بالتقدير
        المحلي في \cref{pb:b3:product:1}، السؤال 7 --- فماذا تعطي كل طريقة
        ولا تعطيه الأخرى؟
  \item (الشمول) اشرح في فقرة لماذا تبيّن طريقة الاستبدال
        أكثر من مبرهنة النهاية المركزية: أن أي إحصاءة من الشكل
        $\E f(\text{مجموع})$ مع $f$ ناعمة غير حساسة، عند الرتبة $n^{-1/2}$،
        لكامل \emph{قانون} الحدود وراء عزميه الأولين ---
        أي «مبدأ الصمود» الذي تقوم عليه نتائج الشمول الحديثة
        (المصفوفات العشوائية، وكثيرات الحدود العشوائية)،
        ومبرهنة النهاية المركزية أول حالاته.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- التنعيم، مدفوعًا: معدلات
أفضل.} جاءت الخسارة من $n^{-1/2}$ (من أجل $f$ ناعمة) إلى
$n^{-1/8}$ (من أجل دوال التوزيع) من تحميل $f'''$ بمعيار
السوپريموم. ويمكن للهجينات أن تصلح جزءًا منها: فهي تحتوي حدودًا
\omterm{def:b3:clt:gaussianvector}{غاوسية}، والغاوسيات \emph{تنعّم}.
\begin{enumerate}[resume]
  \item (غاوسي مختبئ) من أجل $1 \leq i \leq n - 1$، ومن أجل $h =
        \frac{X_i}{\sqrt n}$ أو
        $\frac{N_i}{\sqrt n}$، ومن أجل $\theta \in \intcc01$، اكتب $W_i + \theta h = A +
        Z$ مع $Z = \frac{N_{i+1} + \dots + N_n}{\sqrt
        n}$. برهن
        على أن $Z \sim \mathcal N\bigl(0,
        \frac{n-i}n\bigr)$ مستقل عن الزوج $(A,
        h)$، واستنتج، من أجل
        كل $g \in
        L^1(\R)$ متصلة،
        \[
        \E\bigl[\abs h^3\,\abs{g(W_i + \theta h)}\bigr]
        \;\leq\; \sqrt{\frac{n}{2\pi(n - i)}}\;
        \norm{g}_{L^1}\;\E\abs h^3 .
        \]
  \item اجمع السؤال 10 مع الصيغة التكاملية لباقي تايلور،
        \[
        f(w + h) = f(w) + f'(w)h + \tfrac12f''(w)h^2 +
        \int_0^1\frac{(1 - \theta)^2}2\,f'''(w + \theta
        h)\,h^3\,\dd\theta,
        \]
        لإعادة إجراء السؤالين 3 و4: فمن أجل $f \in \mathcal C^3_b$ مع فوق ذلك
        $f''' \in L^1(\R)$،
        \[
        \bigl|\E f(T_n) - \E f(G_n)\bigr| \leq
        \frac{\beta + \gamma}{3\sqrt{2\pi}}\cdot
        \frac{\norm{f'''}_{L^1}}{\sqrt n} +
        \frac{M_3(\beta + \gamma)}{6\,n^{3/2}}
        \]
        \emph{(ويعالج السؤال 10 الاستبدالات $i \leq n - 1$ ---
        باستعمال $\sum_{m=1}^{n-1}m^{-1/2} \leq 2\sqrt n$ --- ويعالج الحدّ الخشن في السؤال 3
        الاستبدالَ الأخير)}. وتحقق من أن منحدرات السؤال 7 تحقق
        $\norm{\psi_\delta'''}_{L^1} = K_1\delta^{-2}$
        بينما $M_3 = K\delta^{-3}$، وأدخلها، وأمثِل $\delta$: فيتحسّن معدل
        دالة التوزيع المنتظم إلى $O(n^{-1/6})$.
  \item (مطابقة عزم إضافي) لنفترض فوق ذلك $\E
        X_1^3 = 0$ و $\beta_4 = \E X_1^4 < \infty$.
        احسب $\E N_1^3$ و $\E N_1^4$، وانشر حتى الرتبة الرابعة،
        وبرهن على المنوال نفسه على أن معدل دالة التوزيع يصير
        $O(n^{-1/4})$ \emph{(فالآن $\norm{\psi_\delta^{(4)}}_{L^1} =
        K_2\delta^{-3}$ و $M_4 = K'\delta^{-4}$؛ واختر
        $\delta = n^{-1/4}$)}.
  \item (العائق) لنفترض أن العزوم $k$ الأولى للمتغيّر $X_1$
        توافق العزوم \omterm{def:b3:clt:gaussianvector}{الغاوسية} ($k = 2$ دائمًا؛ و $k = 3$ إذا
        وفقط إذا كان $\E X_1^3 = 0$؛ و $k \geq 4$ في لا حالة
        تقريبًا، إذ $\E N_1^4 = 3$). تحقق من أن مخطط الأسئلة
        10--12 يسلّم معدلَ دالة التوزيع $n^{-(k-1)/(2k+2)}$، بموازنة
        $\delta^{-k}n^{-(k-1)/2}$ مقابل $\delta$، ولاحظ أن الأُسّ لا يقارب قيمة
        بيري--إسين $\frac12$ إلا حين $k \to
        \infty$. واشرح في بضع
        جمل لماذا تتشبّع طريقة الاستبدال: إذ يُحمَّل كل استبدال
        بالقيمة المطلقة، بينما يستثمر الطريق الفورييهي
        (متراجحة إسين في التنعيم) تذبذبَ فرق الدوال المميِّزة
        ويبلغ $C\beta n^{-1/2}$ بثلاثة عزوم فقط.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- بُعدان: مبرهنة النهاية
المركزية المتعددة الأبعاد، بالاستبدال.} ولتكن الآن $X_i$
\emph{متجهات} عشوائية \omterm{def:b3:probability:independence}{مستقلة} متماثلة موسَّطة من $\R^2$ بمصفوفة
تغاير $\Sigma$ ومع $\beta' = \E\norm{X_1}^3 <
\infty$ (بالمعيار الإقليدي).
\begin{enumerate}[resume]
  \item (المتجهات \omterm{def:b3:clt:gaussianvector}{الغاوسية}، إلى الرتبة) قطرِن $\Sigma =
        PDP^{\mathsf T}$
        (\cref{exo:b3:submanifolds:8}) وضع $C = P\sqrt DP^{\mathsf T}$. ومن أجل $Z = (Z^1,
        Z^2)$ زوجًا من
        الغاوسيات المعيارية \omterm{def:b3:probability:independence}{المستقلة} (\cref{thm:b3:probability:existence})، برهن على
        أن $N
        = CZ$ متجهةٌ \omterm{def:b3:clt:gaussianvector}{غاوسية} (\cref{def:b3:clt:gaussianvector}) بمتوسط $0$
        وتغاير $\Sigma$، مع $\gamma' = \E\norm N^3 <
        \infty$؛ وعلى أن للمقدار $G_n = \frac{N_1 + \dots +
        N_n}{\sqrt n}$
        القانونَ $\mathcal N(0, \Sigma)$ \emph{بالضبط} من أجل نسخ \omterm{def:b3:probability:independence}{مستقلة}
        متماثلة $N_i$.
  \item (تايلور بمتغيّرين) من أجل $f \colon \R^2 \to
        \R$ من الصنف
        $\mathcal C^3$ مع $M_3 =
        \max_{\abs\alpha = 3}\sup\abs{\partial^\alpha f} <
        \infty$، برهن على
        \[
        \Bigl|f(w + h) - f(w) - \langle\nabla f(w),
        h\rangle - \tfrac12\langle h, D^2f(w)\,h\rangle
        \Bigr| \leq \frac{M_3}6\,\bigl(\abs{h_1} +
        \abs{h_2}\bigr)^3 \leq \frac{\sqrt2\,M_3}3\,
        \norm h^3
        \]
        \emph{(بدراسة $t \mapsto f(w + th)$ على $\intcc01$)}.
  \item (مبرهنة النهاية المركزية في $\R^2$) أجرِ مخطط
        الاستبدال على الهجينات المتجهية $H_i$: برهن على أن
        حدود الرتبتين الأولى والثانية تتلاشى (إذ تتطابق
        المتوسطات والتغايرات)، وتلسكب، وارقَ كما في السؤال 5
        (بالإحكام من $\E\norm{T_n}^2 =
        \operatorname{tr}\Sigma$؛ وبالتمليف الآن في $\R^2$،
        \cref{thm:b3:lp:regularization}) لتخلص إلى: أنه من أجل كل $f \colon
        \R^2 \to \R$ محدودة
        متصلة،
        \[
        \E\,f\Bigl(\frac{X_1 + \dots + X_n}{\sqrt n}\Bigr)
        \longrightarrow \E\,f(N), \qquad N \sim \mathcal
        N(0, \Sigma) :
        \]
        أي \cref{thm:b3:clt:multiclt} في البُعد $2$، بمعدل من أجل $f$ الناعمة
        ودون أي تحليل فورييهي.
  \item (كرامر--وولد، وتذبذب مشترك) استنتج أن $\langle t, \frac{S_n}{\sqrt n}\rangle
        \Rightarrow \mathcal N(0, t^{\mathsf T}\Sigma t)$ من أجل
        كل $t \in \R^2$ مثبَّت. وتطبيقٌ: من أجل $(\xi_i)$
        حقيقية \omterm{def:b3:probability:independence}{مستقلة} متماثلة موسَّطة مع $\E\xi_1^2 = 1$
        و $\E\xi_1^6 < \infty$ (بحيث ينطبق الجزء الخامس على $V_i = (\xi_i, \xi_i^2 - 1)$)، برهن
        على
        \[
        \frac1{\sqrt n}\Bigl(\sum_{i\leq n}\xi_i,\
        \sum_{i\leq n}(\xi_i^2 - 1)\Bigr) \Longrightarrow
        \mathcal N\Bigl(0, \begin{pmatrix} 1 & \E\xi_1^3\\
        \E\xi_1^3 & \E\xi_1^4 - 1\end{pmatrix}\Bigr) :
        \]
        أي إن المتوسط التجريبي والعزم الثاني التجريبي يتذبذبان
        غاوسيًّا معًا --- ومستقلَّين في النهاية إذا وفقط إذا
        كان $\E\xi_1^3 = 0$ (\cref{thm:b3:clt:gaussianvector}).
\end{enumerate}

\medskip
\noindent\textbf{الجزء السادس --- طريقة دلتا.}
\begin{enumerate}[resume]
  \item لتكن $(\hat\theta_n)$ متغيّرات عشوائية تحقق $\sqrt n(\hat\theta_n - \theta) \Rightarrow
        \mathcal N(0, \sigma^2)$ من
        أجل وسيط حقيقي $\theta$، ولتكن $g$ قابلة للاشتقاق عند
        $\theta$. برهن على \emph{طريقة دلتا}:
        \[
        \sqrt n\bigl(g(\hat\theta_n) - g(\theta)\bigr)
        \Longrightarrow \mathcal N\bigl(0,
        g'(\theta)^2\sigma^2\bigr)
        \]
        \emph{(اكتب $g(x) - g(\theta) = (g'(\theta) +
        \eta(x))(x - \theta)$ مع $\eta \to 0$ عند $\theta$؛
        وبرهن على $\hat\theta_n \to \theta$، ثم على $\eta(\hat\theta_n) \to 0$، احتماليًّا؛ وأنهِ
        بسلوتسكي، \cref{exo:b3:clt:8}، و\cref{exo:b3:clt:4}(b))}.
  \item التطبيقات. (a) من أجل $(\xi_i)$ حقيقية \omterm{def:b3:probability:independence}{مستقلة} متماثلة
        بمتوسط $\mu$ وتباين $\sigma^2$، ومن أجل $\bar X_n =
        \frac1n\sum_{i\leq n}\xi_i$: برهن
        على $\sqrt n(\bar
        X_n^2 - \mu^2) \Rightarrow \mathcal N(0,
        4\mu^2\sigma^2)$ حين $\mu \neq 0$، وعلى أن العبارة الصحيحة
        من أجل $\mu = 0$ تعيش عند سلّم آخر: $n\bar X_n^2 \Rightarrow \sigma^2N^2$ مع $N \sim \mathcal N(0,1)$
        (وعيّن دالة توزيع النهاية). (b) (تثبيت التباين) من أجل
        $\hat p_n$ تواترَ النجاح لعيّنة $\mathcal B(1, p)$، حيث
        $p \in \intoo01$: برهن على أن $g(p) = \arcsin\sqrt p$ يحقق
        \[
        \sqrt n\,\bigl(g(\hat p_n) - g(p)\bigr)
        \Longrightarrow \mathcal N\Bigl(0, \frac14\Bigr)
        \]
        \emph{أيًّا كان} $p$ --- أي شريط خطأ مقارب خالٍ من
        الوسيط المجهول؛ وقارن ذلك مع \cref{ex:b3:clt:confidence}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء السابع --- بواسون، بالطريقة نفسها:
مبرهنة لوكام.} يعرف الاستبدال صنف شمول ثانيًا: مجاميع عدد كبير
من الحوادث \emph{النادرة} \omterm{def:b3:probability:independence}{المستقلة}. ومن أجل القوانين على $\N$
تكون المسافة الصحيحة \emph{التغيّر الكلي}،
\[
d_{\mathrm{TV}}(\mu, \nu) = \sup_{A\subseteq\N}\,
\abs{\mu(A) - \nu(A)} .
\]
\begin{enumerate}[resume]
  \item برهن على $d_{\mathrm{TV}}(\mu, \nu) =
        \frac12\sum_{k\geq0}\abs{\mu(\{k\}) -
        \nu(\{k\})}$، وبرهن على حدّ الاقتران: أنه من أجل
        \emph{أي} زوج $(X, Y)$ من المتغيّرات العشوائية
        بالقانونين $\mu$ و $\nu$ على الفضاء نفسه، $d_{\mathrm{TV}}(\mu, \nu) \leq \P(X \neq Y)$.
  \item احسب بالضبط، من أجل $p \in \intoo01$:
        \[
        d_{\mathrm{TV}}\bigl(\mathcal B(1, p), \mathcal
        P(p)\bigr) = p\bigl(1 - \eu^{-p}\bigr) \leq p^2 .
        \]
  \item (لوكام، بالاستبدال) لتكن $X_i \sim \mathcal B(1,
        p_i)$ ولتكن $Y_i \sim \mathcal P(p_i)$،
        والمتغيّرات $2n$ \omterm{def:b3:probability:independence}{مستقلة}؛ و $S = X_1 + \dots + X_n$، وتذكّر $Y_1 + \dots + Y_n \sim \mathcal P(\lambda)$ مع
        $\lambda = \sum_ip_i$ (\cref{exo:b3:clt:1}). واستبدل إحداثيًّا واحدًا في كل
        مرة في الهجينات الصحيحة $H_i = Y_1 + \dots + Y_i + X_{i+1} + \dots
        + X_n$: برهن، من أجل كل
        $A \subseteq \N$، على
        \[
        \abs{\P(H_{i-1} \in A) - \P(H_i \in A)} \leq
        d_{\mathrm{TV}}\bigl(\mathcal B(1, p_i), \mathcal
        P(p_i)\bigr),
        \]
        واخلص إلى \emph{متراجحة لوكام}:
        \[
        d_{\mathrm{TV}}\bigl(\text{قانون} S,\ \mathcal
        P(\lambda)\bigr) \leq \sum_{i=1}^np_i^2 .
        \]
  \item العوائد. (a) من أجل $p_i = \frac\lambda n$: يكون الحدّ $\frac{\lambda^2}n$ --- أي
        قانون الحوادث النادرة (\cref{exo:b3:clt:5}) مرقًّى إلى معدل
        صريح، منتظم على جميع الحوادث، وصحيح من أجل $p_i$ غير
        متساوية أيضًا. (b) تُسلَّم $500$ رسالة، وتضلّ كلٌّ منها
        طريقها \omterm{def:b3:probability:independence}{باستقلال} باحتمال $\frac1{500}$: حُدّ خطأ نموذج
        بواسون بالوسيط $1$، وقدّر احتمال ألا تضلّ أي رسالة.
        (c) واختم المسألة: قارن صنفَي الشمول الملتقيَين هنا ---
        الغاوسي (إسهامات كثيرة صغيرة منتشرة؛ بعزمين متطابقين؛
        وتايلور) والبواسوني (إسهامات كثيرة نادرة؛ بمتوسط واحد
        متطابق؛ واقتران مضبوط بالتغيّر الكلي) --- وطريقة
        الاستبدال الواحدة وراءهما معًا.

  \item (الخطأ النسبي وتحويل اللوغاريتم) لتكن $(X_n)$ \omterm{def:b3:probability:independence}{مستقلة}
        متماثلة موجبة بمتوسط $\mu > 0$ وتباين $\sigma^2$،
        ولتكن $\bar X_n$ المتوسطَ التجريبي. برهن بطريقة دلتا
        على
        \[
        \sqrt n\,\bigl(\ln\bar X_n - \ln\mu\bigr)
        \Longrightarrow
        \mathcal N\Bigl(0,\ \frac{\sigma^2}{\mu^2}\Bigr) :
        \]
        فيكون الوسيط المقارب للمقدار $\ln\bar X_n$ هو
        \emph{معامل التغيّر} $\sigma/\mu$ --- أي خطأ نسبي خالٍ
        من السلّم. واستنتج فترةَ ثقة $95\%$ للمتوسط $\mu$ من
        الشكل الضربي $\bar X_n\cdot\eu^{\pm1.96\,\sigma/(\mu\sqrt
        n)}$، واشرح متى يكون مفضَّلًا على
        الشكل الجمعي.
  \item (العزم الثالث يوجّه الخطأ) من أجل $X \sim$ برنولي($p$)
        موسَّطًا، احسب $\E\bigl[(X -
        p)^3\bigr] = p(1-p)(1-2p)$. وباستعمال تحليل الجزء الرابع
        (إذ تقود العزوم الثالثة خطأ الاستبدال)، اشرح لماذا
        يكون التقريب الغاوسي للمقدار $\mathcal B(n, p)$ لامتناظرًا من أجل
        $p \neq \frac12$ --- متجاوزًا من جهة ومقصّرًا من الأخرى
        --- ولماذا يتمتع $p = \frac12$ بالمعدل الأسرع الموافق
        للعزوم المتطابقة. وتحقق من إشارة الالتواء عدديًّا على
        $\mathcal
        B(20, 0.1)$ مقابل $\mathcal N(2, 1.8)$: قارن $\P(S = 0) = 0.9^{20}$ بالكتلة \omterm{def:b3:clt:gaussianvector}{الغاوسية}
        للمقدار $\intoo{-\infty}{0.5}$.
\end{enumerate}
\end{problem}
