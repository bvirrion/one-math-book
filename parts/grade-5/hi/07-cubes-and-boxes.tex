\chapter{घन और घनाभ}\label{ch:g5:solids}

चपटे पन्ने से बाहर: \omterm{def:g5:solids:def}{ठोस} असली दुनिया के आकार हैं --- पासे, अनाज
के डिब्बे, टिन, गेंदें। यह अध्याय उन्हें नाम देना, उनके \omterm{def:g5:solids:def}{फलक} और
\omterm{def:g5:solids:def}{किनारे} गिनना, उन्हें खोलकर \omterm{def:g5:solids:net}{जाल} बनाना, और छोटे \omterm{def:g5:solids:def}{घन} गिनकर उनका
\omterm{def:g5:solids:volume}{आयतन} नापना सिखाता है। (\omterm{def:g5:solids:volume}{आयतन} के सूत्र
\cref{ch:g6:measure} में आते हैं।)

\section{ठोसों का परिवार}

\begin{definition}[ठोस और उनकी शब्दावली]\label{def:g5:solids:def}
\emph{ठोस}\index{ठोस} जगह घेरता है। उसकी चपटी सतहें
\emph{फलक}\index{फलक} हैं; फलक \emph{किनारों}\index{किनारा} पर मिलते हैं;
किनारे \emph{शीर्षों}\index{शीर्ष} पर मिलते हैं। परिवार के
चित्र:
\begin{itemize}
  \item \emph{घन}\index{घन}: $6$ वर्गाकार फलक, $12$ किनारे,
        $8$ शीर्ष;
  \item \emph{घनाभ}\index{घनाभ} (आयताकार ठोस): $6$
        आयताकार फलक, $12$ किनारे, $8$ शीर्ष;
  \item \emph{बेलन}\index{बेलन}: दो वृत्ताकार फलक और एक लिपटा हुआ फलक
        (टिन का डिब्बा);
  \item \emph{पिरामिड}\index{पिरामिड}: एक \omterm{def:g4:geometry:polygon}{बहुभुज} आधार और शिखर पर मिलने वाले
        त्रिभुज;
  \item \emph{गोला}: पूरी तरह गोल एक ही सतह, न फलक, न
        किनारा।
\end{itemize}
\end{definition}

\begin{example}[घन पर गिनती]\label{ex:g5:solids:count}
किसी पासे पर जाँचो: $6$ \omterm{def:g5:solids:def}{फलक} (संख्याएँ $1$ से $6$), $8$ \omterm{def:g5:solids:def}{शीर्ष}
(कोने), $12$ \omterm{def:g5:solids:def}{किनारे} (उन पर उँगली फिराओ: $4$ ऊपर, $4$
नीचे, $4$ खड़े)। और
$6 + 8 = 12 + 2$ --- यही अनोखा प्रतिरूप \omterm{def:g5:solids:def}{घनाभ} और \omterm{def:g5:solids:def}{पिरामिड} पर भी
चलता है (\cref{ex:g8:solids:count} उस पर लौटता है)।
\end{example}

\section{जाल}

\begin{definition}[जाल]\label{def:g5:solids:net}
किसी \omterm{def:g5:solids:def}{ठोस} का \emph{जाल}\index{जाल} उसके सारे फलकों का ऐसा चपटा
चित्र है, जिसमें \omterm{def:g5:solids:def}{फलक} \omterm{def:g5:solids:def}{किनारों} पर जुड़े हों और जिसे मोड़ने पर वही
\omterm{def:g5:solids:def}{ठोस} बन जाए --- मानो \omterm{def:g5:solids:def}{ठोस} को खोल दिया गया हो, जैसे गत्ते का डिब्बा चपटा कर दिया जाता है।
\end{definition}

तीसरी नहीं --- मोड़ने पर दो वर्ग एक ही \omterm{def:g5:solids:def}{फलक} पर आ जाते हैं और \omterm{def:g5:solids:def}{घन}
खुला रह जाता है। इन्हें काटकर आज़माओ!}

\begin{example}[जाल पढ़ना]\label{ex:g5:solids:read}
पासे पर आमने-सामने के फलकों का योग $7$ होता है। जाल पर
आमने-सामने के \omterm{def:g5:solids:def}{फलक} वे होते हैं जिनके बीच पंक्ति में ठीक एक वर्ग
हो (या जो कोने से घूमकर मिलें) --- पड़ोसी नहीं! मोड़ने से पहले
जाल पर जोड़े अंकित करना अंतरिक्ष को चपटा देखने का बढ़िया अभ्यास है।
\end{example}

\section{आयतन: घन गिनना}

\begin{definition}[गिनकर आयतन]\label{def:g5:solids:volume}
इकाई घनों से बने किसी \omterm{def:g5:solids:def}{ठोस} का \emph{आयतन}\index{आयतन} उसमें समाए
घनों की संख्या है। मात्रक सेंटीमीटर \omterm{def:g5:solids:def}{घन} (cm$^3$) या मीटर \omterm{def:g5:solids:def}{घन}
(m$^3$) हो सकता है।
\end{definition}

\begin{example}[घनाभ को परत-दर-परत गिनना]\label{ex:g5:solids:layers}
एक \omterm{def:g5:solids:def}{घनाभ} $4$ \omterm{def:g5:solids:def}{घन} लंबा, $3$ चौड़ा, $2$ ऊँचा: सबसे नीचे की परत में
$4 \times 3 = 12$ \omterm{def:g5:solids:def}{घन} आते हैं, और परतें $2$ हैं:
\[
12 \times 2 = 24 \text{ घन} .
\]
परतों से गिनना हमेशा चलता है --- और यह चुपचाप
\cref{ch:g6:measure} के सूत्र $V = L \times w \times h$ को सिद्ध कर देता है।
\end{example}

{\small इकाई घनों का $4 \times 3 \times 2$ \omterm{def:g5:solids:def}{घनाभ}, परत-दर-परत
गिना हुआ।}
\end{omfigure}

\begin{example}[वही घन, अलग आकार]\label{ex:g5:solids:rearrange}
आठ इकाई घनों से $2 \times 2 \times 2$ का \omterm{def:g5:solids:def}{घन}, $8 \times 1 \times 1$
की छड़ या $4 \times 2 \times 1$ का पटरा बनाया जा सकता है: तीन
अलग \omterm{def:g5:solids:def}{ठोस}, एक ही \omterm{def:g5:solids:volume}{आयतन} ($8$ \omterm{def:g5:solids:def}{घन})। \omterm{def:g5:solids:volume}{आयतन} सामान गिनता है, आकार नहीं।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g5:solids:1}
इन आकारों की कोई असली चीज़ बताओ: \omterm{def:g5:solids:def}{घन}; \omterm{def:g5:solids:def}{घनाभ}; \omterm{def:g5:solids:def}{बेलन}; गोला;
\omterm{def:g5:solids:def}{पिरामिड}।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:2}
\omterm{def:g5:solids:def}{घनाभ} के कितने \omterm{def:g5:solids:def}{फलक}, \omterm{def:g5:solids:def}{किनारे} और \omterm{def:g5:solids:def}{शीर्ष} होते हैं? प्रतिरूप
\omterm{def:g5:solids:def}{फलक} $+$ \omterm{def:g5:solids:def}{शीर्ष} $=$ \omterm{def:g5:solids:def}{किनारे} $+ 2$ जाँचो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:3}
\omterm{def:g5:solids:def}{घनाभ} के कौन-से \omterm{def:g5:solids:def}{फलक} एक जैसे होते हैं? (छहों फलकों को जोड़ों में
बाँटो।) \omterm{def:g5:solids:def}{घन} के फलकों में ख़ास क्या है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:4}
$2$ cm भुजा वाले \omterm{def:g5:solids:def}{घन} का \omterm{def:g5:solids:net}{जाल} बनाओ (इस अध्याय वाला क्रॉस-नमूना
लो), उसे काटो और मोड़ो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:5}
पासे के क्रॉस-जाल पर $1$ से $6$ तक संख्याएँ ऐसे रखो कि
आमने-सामने के फलकों का योग $7$ हो। (पहले
\cref{ex:g5:solids:read} से आमने-सामने के जोड़े ढूँढ़ो।)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:6}
इकाई \omterm{def:g5:solids:def}{घन} गिनो: $5$ लंबा, $2$ चौड़ा, $3$ ऊँचा \omterm{def:g5:solids:def}{घनाभ}; $3$ भुजा
वाला \omterm{def:g5:solids:def}{घन}; $3 \times 2 \times 1$ के पटरे पर $1 \times 2 \times 1$
का पटरा रखकर बनी L-आकार की ढेरी।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:7}
एक \omterm{def:g5:solids:def}{घनाभ} की एक परत में ठीक $12$ इकाई \omterm{def:g5:solids:def}{घन} आते हैं। परत के दो संभव
आकार ($L \times w$) बताओ, और हर एक के लिए \omterm{def:g5:solids:def}{घनाभ} का \omterm{def:g5:solids:volume}{आयतन} बताओ अगर
उसमें ऐसी $3$ परतें हों।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:8}
$27$ इकाई घनों से कौन-सा \omterm{def:g5:solids:def}{घन} बना सकते हो? $64$ से? (परतों की
गिनती उलटी करके सोचो।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:solids:9}
$1$ cm भुजा वाले चीनी के \omterm{def:g5:solids:def}{घन} ऐसे \omterm{def:g5:solids:def}{घनाभ} में आते हैं जो भीतर से
$10$ cm लंबा, $5$ cm चौड़ा और $4$ cm ऊँचा है। कितने चीनी के \omterm{def:g5:solids:def}{घन}
समाएँगे? कोई परिवार रोज़ $6$ \omterm{def:g5:solids:def}{घन} इस्तेमाल करे, तो एक डिब्बा कितने
दिन चलेगा (समझदारी से गोल करो)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:solids:10}
$3 \times 3 \times 3$ के एक \omterm{def:g5:solids:def}{घन} को बाहर से लाल रँगकर $27$ इकाई
घनों में काटा जाता है। कितने छोटे घनों पर ठीक $3$ फलकों पर रंग
है? $2$ पर? $1$ पर? किसी पर नहीं? (जाँच: चारों गिनतियाँ मिलकर
$27$ होनी चाहिए।)
\end{exercise}
