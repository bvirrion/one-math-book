\chapter{दशमलव संख्याओं की गणना}\label{ch:g5:decimalops}

दशमलव संख्याएँ लगभग पूर्ण संख्याओं की तरह ही जुड़ती, घटती और गुणा
होती हैं --- सारा हुनर यह जानने में है कि बिंदु कहाँ जाएगा। यह
अध्याय खानों वाले तरीक़े और $10$, $100$, $1000$ से गुणा या भाग
का जादू बताता है। (दो दशमलव संख्याओं का आपस में गुणा
\cref{ch:g6:decimals} में आएगा।)

\section{जोड़ और घटाव}

\begin{method}[बिंदु के साथ खाने]\label{met:g5:decimalops:addsub}
\begin{enumerate}
  \item संख्याएँ इस तरह लिखो कि \emph{\omterm{def:g4:decimals:point}{दशमलव बिंदु} एक सीध में}
        हों (तब इकाई के नीचे इकाई, दसवें भाग के नीचे दसवाँ भाग);
  \item छोटे दशमलव भाग में \omterm{def:g1:counting:zero}{शून्य} भर दो;
  \item पूर्ण संख्याओं की तरह जोड़ो या घटाओ;
  \item नतीजे में बिंदु सीधे नीचे उतार दो।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:decimalops:addsub}
$27.5 + 3.86$ ($27.5 = 27.50$ भरकर):
\[
\begin{array}{r}
2\,7.5\,0 \\
+\ \ \ 3.8\,6 \\
\hline
3\,1.3\,6
\end{array}
\qquad\qquad
\begin{array}{r}
1\,2.4\,0 \\
-\ \ \ 5.6\,5 \\
\hline
\ \ \,6.7\,5
\end{array}
\]
घटाव के लिए ($12.4 - 5.65$): भरो, आम तरीक़े से उधार लो, और वापस
जोड़कर जाँचो: $6.75 + 5.65 = 12.4$।
\end{example}

\begin{example}[मन-ही-मन पूरक]\label{ex:g5:decimalops:complement}
$10$ तक पहुँचने के लिए $7.6$ में क्या जोड़ना होगा? चढ़ो:
$7.6 \to 8$ पर $0.4$, फिर $8 \to 10$ पर $2$: पूरक $2.4$ है।
बाक़ी पैसों में काम आता है: $7.60$ की चीज़ के लिए $10$ दो, $2.40$ वापस मिलेंगे।
\end{example}

\section{दस गुना बड़ा, दस गुना छोटा}

\begin{proposition}[10, 100, 1000 से गुणा और भाग]\label{prop:g5:decimalops:shift}
$10$ से गुणा करने पर हर अंक दस गुना क़ीमती हो जाता है: अंक एक
स्थान बाईं ओर सरक जाते हैं --- जो देखने में ऐसा लगता है मानो
\emph{बिंदु एक स्थान दाईं ओर} खिसक गया हो। भाग देने पर उलटा होता है:
\[
3.47 \times 10 = 34.7, \qquad
3.47 \times 100 = 347, \qquad
52.1 \div 10 = 5.21, \qquad
6 \div 100 = 0.06 .
\]
\end{proposition}
\admitted

\begin{example}[फिर से मात्रक बदलना]\label{ex:g5:decimalops:convert}
माप के मात्रक दस की घातों के फ़ासले पर हैं, इसलिए मात्रक बदलना
इन्हीं सरकनों का खेल है: $2.35$ m $= 235$ cm ($\times 100$);
$470$ g $= 0.47$ kg ($\div 1000$); $1.5$ L $= 150$ cL।
\end{example}

\section{दशमलव संख्या का पूर्ण संख्या से गुणा}

\begin{method}[दशमलव गुना पूर्ण]\label{met:g5:decimalops:mult}
ऐसे निकालो मानो बिंदु है ही नहीं, फिर नतीजे में उतने ही दशमलव
अंक रखो जितने दशमलव गुणक में थे:
\[
4.35 \times 6: \quad 435 \times 6 = 2\,610
\ \to\ 4.35 \times 6 = 26.10 = 26.1 .
\]
बिंदु भरोसे से रखने के लिए अंदाज़ा लगाओ: $4.35$, $4$ के पास है,
और $4 \times 6 = 24$ --- इसलिए $26.1$, न $2.61$, न $261$।
\end{method}

\begin{example}[बार-बार जोड़ना अब भी चलता है]\label{ex:g5:decimalops:mult}
$0.75 \times 4 = 0.75 + 0.75 + 0.75 + 0.75 = 3$: चार \omterm{def:g3:division:half}{चौथाई} मिलकर
एक पूरा बनाते हैं, दशमलव के कपड़ों में। पूर्ण संख्या से गुणा अब
भी ``इतनी बार'' ही है।
\end{example}

\begin{example}[ख़रीदारी]\label{ex:g5:decimalops:shopping}
$2.45$-$2.45$ की तीन कॉपियाँ और $1.20$ की एक क़लम:
\begin{enumerate}
  \item कॉपियाँ: $2.45 \times 3 = 7.35$;
  \item कुल: $7.35 + 1.20 = 8.55$;
  \item $10$ में से बाक़ी: $10 - 8.55 = 1.45$।
\end{enumerate}
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:1}
खानों में निकालो: $45.7 + 8.93$; \quad $16.25 + 3.75$; \quad
$0.86 + 12.4$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:2}
खानों में निकालो और जाँचो: $9.4 - 2.72$; \quad $20 - 13.45$;
\quad $6.03 - 5.9$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:3}
$10$ तक के मन-ही-मन पूरक (जैसे
\cref{ex:g5:decimalops:complement} में): $6.2$; \quad $3.75$; \quad
$9.99$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:4}
खानों के बिना निकालो:
\[
7.24 \times 10, \qquad
0.58 \times 100, \qquad
34.5 \div 10, \qquad
7 \div 100, \qquad
0.3 \times 1000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:5}
बदलो: $4.05$ m को cm में; \quad $325$ cm को m में; \quad $0.6$ kg को g में;
\quad $85$ cL को L में।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:6}
पहले अंदाज़ा लगाओ, फिर निकालो: $6.2 \times 4$; \quad
$3.45 \times 8$; \quad $12.5 \times 6$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:7}
एक ट्रैक का एक चक्कर $0.4$ km का है। $7$ चक्कर कितने लंबे हैं?
और $25$ चक्कर?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:8}
लेआ $1.15$-$1.15$ की $2$ रोटियाँ और $12.60$ का एक केक ख़रीदती
है। वह $20$ का नोट देती है। उसका कुल ख़र्च और बाक़ी पैसे निकालो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:9}
एक बोतल में $1.5$ L आता है। $6$ बोतलों के पैकेट में कितने लीटर?
एक परिवार रोज़ $0.75$ L पीता है: एक पैकेट कितने दिन चलेगा?
($9$ में $0.75$ कितनी बार समाता है?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:decimalops:10}
मिलो $5.6 \times 3 = 15.18$ निकालता है, यह सोचकर कि
``$5 \times 3 = 15$ और $6 \times 3 = 18$''। अंदाज़े से दिखाओ कि
उत्तर ग़लत ही होगा, फिर सही निकालो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:decimalops:11}
एक धावक के चार चक्करों में $65.4$ s, $64.9$ s, $65.0$ s और
$66.1$ s लगे।
\begin{enumerate}
  \item चारों चक्करों का कुल समय निकालो।
  \item चार चक्करों का कीर्तिमान $260$ s है। क्या उसने उसे
        तोड़ा? कितने से?
\end{enumerate}
\end{exercise}
