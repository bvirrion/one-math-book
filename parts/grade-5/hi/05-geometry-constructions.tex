\chapter{ज्यामितीय रचनाएँ}\label{ch:g5:geometry}

मापपट्टी, \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका, परकार: इन तीन औज़ारों और थोड़े-से
तरीक़े से शब्दों में बताई गई कोई भी आकृति ठीक-ठीक बनाई जा सकती
है। यह अध्याय \omterm{def:g4:geometry:def}{लंब}, \omterm{def:g4:geometry:def}{समांतर रेखाएँ}, त्रिभुज और ख़ास \omterm{def:g4:geometry:polygon}{चतुर्भुज}
बनाता है --- पहले हुनर, सिद्धांत बाद में
(\cref{ch:g6:lines,ch:g6:shapes})।

\section{किसी बिंदु से होकर लंब और समांतर}

\begin{method}[दो बुनियादी रचनाएँ]\label{met:g5:geometry:basic}
\emph{$A$ से होकर $d$ पर \omterm{def:g4:geometry:def}{लंब}}: \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका को $d$ के साथ तब तक
सरकाओ जब तक उसका दूसरा समकोण-किनारा $A$ तक न पहुँचे; खींच दो;
\omterm{def:g3:shapes:rightangle}{समकोण} अंकित करो।

\emph{$A$ से होकर $d$ के \omterm{def:g4:geometry:def}{समांतर}}: $d$ पर रखी \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका से
मापपट्टी को लीक की तरह सटा दो; पट्टिका को $A$ तक सरकाओ;
खींच दो। (जाँच: दोनों रेखाओं के बीच की दूरी हर जगह एक-सी
है।)
\end{method}

\begin{example}[रचनाओं की कड़ी]\label{ex:g5:geometry:chain}
कई आकृतियाँ इन्हीं दो चालों की कड़ियाँ हैं। $AB = 5$ cm और
$BC = 3$ cm वाला आयत $ABCD$: $[AB]$ खींचो; $A$ पर \omterm{def:g4:geometry:def}{लंब};
$B$ पर \omterm{def:g4:geometry:def}{लंब}; एक ही ओर $3$ cm पर $D$ और $C$ अंकित करो;
जोड़ दो। हर कदम एक बुनियादी रचना है।
\end{example}

\section{परकार से त्रिभुज}

\begin{method}[तीन भुजाओं से त्रिभुज]\label{met:g5:geometry:triangle}
$6$ cm, $5$ cm, $4$ cm भुजाओं वाला त्रिभुज बनाने के लिए:
\begin{enumerate}
  \item सबसे लंबी भुजा खींचो, $[AB]$ जिसमें $AB = 6$ cm;
  \item केंद्र $A$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $5$ cm का चाप;
  \item केंद्र $B$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $4$ cm का चाप;
  \item उनका कटान-बिंदु तीसरा \omterm{def:g4:geometry:polygon}{शीर्ष} $C$ है; जोड़ दो।
\end{enumerate}
परकार लंबाइयों की गारंटी देता है: पहले चाप का हर बिंदु $A$ से
ठीक $5$ cm दूर है।
\end{method}

\begin{example}[समद्विबाहु और समबाहु मुफ़्त में]\label{ex:g5:geometry:special}
दोनों चापों के लिए परकार का एक ही खुलाव $\Rightarrow$ समद्विबाहु
त्रिभुज। खुलाव आधार के बराबर $\Rightarrow$ समबाहु। परकार नापता
नहीं: वह लंबाइयाँ \emph{उतारता} है, जो और भी अच्छा है।
\end{example}

\section{ख़ास चतुर्भुज, रचना से}

\begin{method}[परिवार बनाना]\label{met:g5:geometry:quadrilaterals}
\begin{enumerate}
  \item \emph{आयत}: \omterm{def:g4:geometry:def}{समांतर} रेखाओं के दो जोड़े जो \omterm{def:g3:shapes:rightangle}{समकोण} पर
        काटें --- या \cref{ex:g5:geometry:chain} वाली कड़ी;
  \item \emph{वर्ग}: वह आयत जिसकी चारों भुजाएँ एक ही लंबाई की
        हों --- उसे परकार से उतार लो;
  \item \emph{समचतुर्भुज}: चारों भुजाएँ बराबर --- एक भुजा खींचो,
        फिर उसके दोनों सिरों से एक ही खुलाव के दो चाप बाक़ी दोनों
        \omterm{def:g4:geometry:polygon}{शीर्ष} तय कर देते हैं (समचतुर्भुज एक विकर्ण पर जुड़े दो
        समद्विबाहु त्रिभुज ही है)।
\end{enumerate}
बनाने के बाद हमेशा औज़ारों से \emph{जाँचो}: कोनों पर \omterm{def:g3:shapes:rightangle}{समकोण}
पट्टिका, भुजाओं पर परकार।
\end{method}

\begin{example}[वर्णन से पहचानना]\label{ex:g5:geometry:recognize}
``चार-चार cm की चारों भुजाओं और एक \omterm{def:g3:shapes:rightangle}{समकोण} वाला \omterm{def:g4:geometry:polygon}{चतुर्भुज}'' ---
इसे बनाओ: चारों बराबर भुजाएँ आकार को समचतुर्भुज बना देती हैं, और
एक कोने को \omterm{def:g3:shapes:rightangle}{समकोण} बनाते ही बाक़ी सब सीधे हो जाते हैं: नतीजा एक
वर्ग है। कुछ वर्णन चुपके से उससे भी ख़ास आकार बता देते हैं जितना
वे बोलते हैं। (\omterm{def:g3:shapes:rightangle}{समकोण} चारों कोनों तक क्यों फैल जाता है, यह
\cref{ch:g7:central} में समझाया गया है।)
\end{example}

\section{रचनाओं में वृत्त}

\begin{example}[औज़ार के रूप में वृत्त]\label{ex:g5:geometry:circle}
``$A$ से $3$ cm दूर के सारे बिंदु ढूँढ़ो'': केंद्र $A$ और
\omterm{def:g3:shapes:shapes}{त्रिज्या} $3$ cm का वृत्त। ``$A$ से $3$ cm \emph{और} $B$ से
$4$ cm दूर के सारे बिंदु ढूँढ़ो'': दोनों वृत्त बनाओ; उनके कटान
(दो, एक या \omterm{def:g1:counting:zero}{शून्य} बिंदु) ही उत्तर हैं। ख़ज़ाने के नक़्शे और
त्रिभुजों की रचनाएँ ठीक इसी विचार पर चलती हैं।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g5:geometry:1}
एक रेखा $d$ और उससे लगभग $3$ cm दूर एक बिंदु $A$ बनाओ। $A$ से
होकर $d$ पर \omterm{def:g4:geometry:def}{लंब} बनाओ, फिर $A$ से होकर $d$ के \omterm{def:g4:geometry:def}{समांतर} रेखा।
\omterm{def:g3:shapes:rightangle}{समकोण} अंकित करो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:2}
$AB = 7$ cm और $BC = 4$ cm वाला आयत $ABCD$ बनाओ,
\cref{ex:g5:geometry:chain} के अनुसार। चौथे कोण को \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका
से जाँचो: रचना ठीक हो तो वह अपने-आप \omterm{def:g3:shapes:rightangle}{समकोण} निकलेगा।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:3}
$7$ cm, $6$ cm और $3$ cm भुजाओं वाला त्रिभुज बनाओ
(\cref{met:g5:geometry:triangle})।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:4}
$5$ cm आधार और $7$-$7$ cm बराबर भुजाओं वाला समद्विबाहु त्रिभुज
बनाओ; फिर $6$ cm भुजा वाला समबाहु त्रिभुज।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:5}
$5$ cm भुजा वाला वर्ग बनाओ। कौन-से औज़ार समकोणों की गारंटी देते
हैं? कौन-से बराबर भुजाओं की?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:6}
$4$ cm भुजा वाला ऐसा समचतुर्भुज बनाओ जो वर्ग न हो (पहले कोने के
लिए अपनी पसंद का खुलाव चुनो)। परकार से चारों भुजाएँ जाँचो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:7}
$AB = 5$ cm वाले दो बिंदु $A$ और $B$ बनाओ। वे सारे बिंदु बनाओ जो
$A$ से $4$ cm और $B$ से $3$ cm दूर हों। वे कितने
हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:8}
एक रचना का कार्यक्रम लिखो (क्रम से कदम, नाम वाले बिंदु): $6$ cm
$\times$ $2$ cm के आयत के ऊपर $4$-$4$ cm बराबर भुजाओं वाला
समद्विबाहु त्रिभुज (पेंसिल जैसी आकृति)। किसी सहपाठी से उसे
करवाओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:geometry:9}
$8$ cm, $3$ cm और $4$ cm भुजाओं वाला त्रिभुज बनाने की कोशिश करो।
चापों के साथ क्या होता है? एक वाक्य में समझाओ कि यह त्रिभुज
क्यों नहीं हो सकता।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:geometry:10}
ऐसा समचतुर्भुज बनाओ जिसके विकर्ण $6$ cm और $4$ cm के हों, यह
जानते हुए कि समचतुर्भुज के विकर्ण एक-दूसरे को उनके बीचों-बीच
\omterm{def:g3:shapes:rightangle}{समकोण} पर काटते हैं: \emph{पहले} विकर्ण खींचो, फिर उनके चारों
सिरे जोड़ दो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:geometry:11}
एक बकरी $4$ m की रस्सी से खूँटे $A$ से बँधी है, और समतल मैदान
में $A$ से $3$ m दूर एक सीधी बाड़ है। नक़्शा बनाओ (हर मीटर के
लिए 1 cm): बकरी किस इलाक़े तक पहुँच सकती है? बाड़ उसे कहाँ रोकती
है?
\end{exercise}
