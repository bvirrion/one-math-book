\chapter{भाग और गुणज}\label{ch:g5:division}

\cref{ch:g4:division} में सीखे लंबे भाग में दो सुधार आते हैं:
दो अंकों के \omterm{def:g5:division:multiple}{भाजक}, और ऐसे \omterm{def:g3:division:remainder}{भागफल} जो बिंदु के आगे भी चलते रहते
हैं --- $3 \div 4$ को आख़िरकार उत्तर मिल जाता है, $0.75$। अध्याय
के अंत में \omterm{def:g5:division:multiple}{गुणज}, \omterm{def:g5:division:multiple}{भाजक} और विभाज्यता की पहली तरकीबें हैं।

\section{दो अंकों की संख्या से भाग}

\begin{method}[दो अंकों के भाजक]\label{met:g5:division:twodigits}
तरीक़ा वही पुराना, बस एक हुनर और: यह ताड़ना कि \omterm{def:g5:division:multiple}{भाजक} कितनी बार
समाता है। पहले \omterm{def:g5:division:multiple}{भाजक} के गुणजों की एक छोटी सूची लिख लो।
$986 \div 23$ के लिए ($23$ के \omterm{def:g5:division:multiple}{गुणज}: 23, 46, 69, 92, 115,
138, 161, 184, 207):
\begin{enumerate}
  \item $98$ दहाई: $23 \times 4 = 92$ समा जाता है,
        $23 \times 5 = 115$ नहीं: अंक $4$, \omterm{def:g3:division:remainder}{शेषफल} $6$;
  \item $6$ नीचे उतारो: $66$: $23 \times 2 = 46$ समा जाता है:
        अंक $2$, \omterm{def:g3:division:remainder}{शेषफल} $20$;
  \item \omterm{def:g3:division:remainder}{भागफल} $42$, \omterm{def:g3:division:remainder}{शेषफल} $20$। जाँच:
        $23 \times 42 + 20 = 966 + 20 = 986$।
\end{enumerate}
\end{method}

\section{दशमलव भागफल}

\begin{example}[वह भाग जो शेषफल पर रुकने से मना कर दे]\label{ex:g5:division:decimal}
$3$ पिज़्ज़े $4$ बच्चों में बाँटो: $3 \div 4$ का \omterm{def:g3:division:remainder}{भागफल} $0$ है और
\omterm{def:g3:division:remainder}{शेषफल} $3$ --- किसी काम का नहीं! भाग को बिंदु के आगे बढ़ाओ:
\begin{enumerate}
  \item $3$ इकाई $= 30$ दसवें भाग; $30 \div 4$: $7$ दसवें भाग,
        \omterm{def:g3:division:remainder}{शेषफल} $2$ दसवें भाग;
  \item $2$ दसवें भाग $= 20$ सौवें भाग; $20 \div 4 = 5$ सौवें
        भाग, \omterm{def:g3:division:remainder}{शेषफल} $0$।
\end{enumerate}
तो $3 \div 4 = 0.75$: हर बच्चे को $0.75$ पिज़्ज़ा मिलता है ---
तीन \omterm{def:g3:division:half}{चौथाई}, जो भिन्नों की भाषा पहले से जानती थी
(\cref{prop:g5:fractions:decimal})।
\end{example}

\begin{method}[भाग को बिंदु के आगे बढ़ाना]\label{met:g5:division:continue}
जब इकाइयाँ ख़त्म हो जाएँ और \omterm{def:g3:division:remainder}{शेषफल} बचा हो:
\begin{enumerate}
  \item \omterm{def:g3:division:remainder}{भागफल} में \omterm{def:g4:decimals:point}{दशमलव बिंदु} लगाओ;
  \item \omterm{def:g3:division:remainder}{शेषफल} में एक \omterm{def:g1:counting:zero}{शून्य} जोड़ो (इकाई दसवें भाग बन जाती है,
        दसवाँ भाग सौवाँ\,\dots) और भाग देते रहो;
  \item तब रुको जब \omterm{def:g3:division:remainder}{शेषफल} $0$ हो जाए --- या जब अंक हमेशा के लिए
        दोहराने लगें, जैसे $1 \div 3 = 0.333\dots$: तब गोल किया
        हुआ मान दे दो।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:division:continue}
$27 \div 6$: \omterm{def:g3:division:remainder}{भागफल} $4$, \omterm{def:g3:division:remainder}{शेषफल} $3$; बिंदु; $30$ दसवें भाग
$\div\ 6 = 5$: तो $27 \div 6 = 4.5$। \quad
$22 \div 8$: $2$, \omterm{def:g3:division:remainder}{शेषफल} $6$; फिर $60 \div 8 = 7$, \omterm{def:g3:division:remainder}{शेषफल} $4$;
फिर $40 \div 8 = 5$: तो $22 \div 8 = 2.75$।
\end{example}

\section{गुणज और भाजक}

\begin{definition}[गुणज, भाजक]\label{def:g5:division:multiple}
$6$ के \emph{गुणज}\index{गुणज} उसके पहाड़े को हमेशा के लिए आगे
बढ़ाने पर मिलते हैं: $0, 6, 12, 18, 24, \dots$ जब $24$, $6$ का
गुणज है, तब यह भी कहते हैं कि $6$, $24$ का \emph{भाजक}\index{भाजक} है,
या कि $24$, $6$ से \emph{विभाज्य} है: भाग $24 \div 6$ कोई
\omterm{def:g3:division:remainder}{शेषफल} नहीं छोड़ता।
\end{definition}

\begin{proposition}[विभाज्यता की तरकीबें]\label{prop:g5:division:criteria}
बिना भाग दिए, कोई संख्या विभाज्य होती है:
\begin{itemize}
  \item $2$ से, जब उसका आख़िरी अंक $0$, $2$, $4$, $6$ या $8$ हो
        (\omterm{def:g3:numbers:evenodd}{सम संख्याएँ});
  \item $5$ से, जब उसका आख़िरी अंक $0$ या $5$ हो;
  \item $10$ से, जब उसका आख़िरी अंक $0$ हो;
  \item $3$ से, जब उसके \emph{अंकों का \omterm{def:g1:addition:def}{योगफल}} $3$ से विभाज्य हो
        (जैसे $741$: $7 + 4 + 1 = 12$, विभाज्य --- और सचमुच
        $741 = 3 \times 247$)।
\end{itemize}
\end{proposition}
\admitted

\begin{example}\label{ex:g5:division:criteria}
क्या $534$, $2$ से विभाज्य है? आख़िर में $4$ है: हाँ। $5$ से?
नहीं। $3$ से? $5 + 3 + 4 = 12$: हाँ। तो $534$, $6$ से भी
विभाज्य है ($2$ \emph{और} $3$ दोनों से) --- जाँच: $534 = 6 \times 89$।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g5:division:1}
लंबे भाग करो (पहले गुणजों की सूची बनाओ):
$851 \div 23$; \quad $704 \div 32$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:2}
ठीक-ठीक दशमलव \omterm{def:g3:division:remainder}{भागफल} निकालो: $9 \div 2$; \quad $27 \div 4$;
\quad $33 \div 5$; \quad $21 \div 8$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:3}
चार दोस्त $54$ का भोजनालय-बिल बराबर बाँटते हैं। हर एक कितना
देता है? (बिंदु के आगे बढ़ो।)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:4}
$10 \div 3$ निकालो, बिंदु के आगे तीन अंक तक। क्या दिखता है?
सौवें भाग तक गोल किया हुआ मान बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:5}
सूची बनाओ: $70$ तक $7$ के \omterm{def:g5:division:multiple}{गुणज}; $18$ के \omterm{def:g5:division:multiple}{भाजक};
$24$ के \omterm{def:g5:division:multiple}{भाजक}।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:6}
$2$ से विभाज्य? $5$ से? $10$ से? $3$ से? हर तरकीब इन पर आज़माओ:
$470$; \quad $735$; \quad $8\,001$; \quad $1\,314$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:7}
$60$ और $80$ के बीच की वे सारी संख्याएँ ढूँढ़ो जो $3$
\emph{और} $5$ दोनों से विभाज्य हैं। (यह किस एक विभाज्यता के
बराबर है?)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:8}
$7.2$ m का रिबन $6$ बराबर टुकड़ों में काटा जाता है। हर टुकड़ा
कितना लंबा है? (दशमलव में पूर्ण संख्या का भाग: पहले मीटर बाँटो,
फिर दसवें भाग।)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:9}
$1\,000$ कंचे $24$-$24$ के थैलों में रखे जाते हैं। कितने थैले
पूरे भरते हैं, और कितने कंचे बचते हैं? (दो अंकों का \omterm{def:g5:division:multiple}{भाजक}।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:division:10}
क्या $5$ दोस्त $7$ चॉकलेट बराबर बाँट सकते हैं, कुछ बचाए बिना?
हर हिस्सा दशमलव में बताओ। वही सवाल $7$ दोस्तों के $5$ चॉकलेट
बाँटने का --- वहाँ क्या फ़र्क़ पड़ता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:division:11}
अंकों-के-योग वाली तरकीब से वह सबसे छोटा अंक $d$ ढूँढ़ो जिससे
$52d$ (अंत में $d$ वाली तीन अंकों की संख्या) $3$ से विभाज्य हो।
फिर ऐसे सारे अंक $d$ ढूँढ़ो।
\end{exercise}
