\chapter{Perímetro e área}\label{ch:g5:areas}

O Ano 4 contou quadradinhos da malha (\cref{ch:g4:measure}); este
capítulo conquista a primeira \emph{fórmula} de \omterm{def:g4:measure:area}{área} --- comprimento
vezes largura, para o retângulo --- e aprende as unidades cm$^2$ e
m$^2$. As fórmulas dos triângulos e de outras formas amadurecem no
\cref{ch:g6:measure} e no \cref{ch:g7:areas}.

\section{Duas medidas diferentes}

\begin{example}[Contorno contra superfície]\label{ex:g5:areas:contrast}
Um jardineiro precisa do \emph{\omterm{def:g4:measure:perimeter}{perímetro}} para comprar a cerca e da
\emph{\omterm{def:g4:measure:area}{área}} para comprar a semente de grama. As duas coisas não
andam juntas: esticar um retângulo, deixando-o mais fino e mais
comprido, pode manter a \omterm{def:g4:measure:area}{área} enquanto o \omterm{def:g4:measure:perimeter}{perímetro} cresce --- compare
um retângulo $6 \times 2$ com um $4 \times 3$ (mesma \omterm{def:g4:measure:area}{área} $12$,
\omterm{def:g4:measure:perimeter}{perímetros} $16$ e $14$).
\end{example}

\section{A fórmula do retângulo}

\begin{proposition}[Área de um retângulo]\label{prop:g5:areas:rectangle}
Um retângulo de comprimento $L$ e largura $w$ (na mesma unidade) tem
\omterm{def:g4:measure:area}{área}
\[
A = L \times w ,
\]
em unidades quadradas: cm$^2$ (quadrados de lado $1$ cm), m$^2$
(quadrados de lado $1$ m), \dots
\end{proposition}

\begin{proof}[Por quê, contando]
Cubra o retângulo com quadrados unitários: $w$ fileiras de $L$
quadrados cada uma, logo $L \times w$ quadrados ao todo ---
exatamente o retângulo da multiplicação do \cref{ch:g3:mult}.
\end{proof}

\begin{omfigure}
\begin{tikzpicture}[scale=0.62]
  \draw[gray!50, thin] (0,0) grid (6,3);
  \draw[very thick, omDef] (0,0) rectangle (6,3);
  \fill[omThm!30] (0,2) rectangle (1,3);
  \draw[omThm, thick] (0,2) rectangle (1,3);
  \node[right, font=\small, omThm] at (1.15,2.5) {$1$ cm$^2$};
  \node[below, font=\small] at (3,-0.15) {$6$ cm};
  \node[left, font=\small] at (0,1.5) {$3$ cm};
  \node[font=\small] at (3.9,1.2) {$6 \times 3 = 18$ cm$^2$};
\end{tikzpicture}

{\small Três fileiras de seis centímetros quadrados: a fórmula
$L \times w$ é a contagem, escrita de uma vez por todas.}
\end{omfigure}

\begin{example}\label{ex:g5:areas:rectangle}
Um tapete mede $2.5$ m por $2$ m: \omterm{def:g4:measure:area}{área} $2.5 \times 2 = 5$ m$^2$. Um
selo mede $3$ cm por $2.4$ cm: \omterm{def:g4:measure:area}{área} $3 \times 2.4 = 7.2$ cm$^2$. A
mesma fórmula, em qualquer unidade --- desde que os dois lados usem a
\emph{mesma}.
\end{example}

\begin{example}[As unidades de área convertem de cem em cem]\label{ex:g5:areas:units}
$1$ m $= 100$ cm, mas $1$ m$^2 = 100 \times 100 = 10\,000$ cm$^2$:
um metro quadrado é uma malha de $100$ por $100$ centímetros
quadrados. As unidades de \omterm{def:g4:measure:area}{área} saltam de cem em cem, não de dez em
dez.
\end{example}

\section{Figuras compostas}

\begin{method}[Cortar, somar, subtrair]\label{met:g5:areas:composite}
Para uma figura feita de retângulos (um L, um T, uma moldura):
\begin{enumerate}
  \item corte-a em retângulos, ou complete-a até um retângulo
        grande;
  \item calcule cada \omterm{def:g4:measure:area}{área} retangular com a fórmula;
  \item some os pedaços --- ou subtraia o buraco do retângulo
        grande;
  \item confira com uma contagem aproximada dos quadradinhos da
        malha.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:areas:composite}
Um cômodo em forma de L: um retângulo de $6$ m $\times$ $4$ m com um
canto de $2$ m $\times$ $2$ m faltando.
\[
\text{Área} = 6 \times 4 - 2 \times 2 = 24 - 4 = 20 \text{ m}^2 .
\]
Ou corte o L numa faixa $6 \times 2$ e numa faixa $4 \times 2$:
$12 + 8 = 20$ m$^2$ --- dois caminhos, uma resposta.
\end{example}

\begin{example}[Metade de um retângulo]\label{ex:g5:areas:halfrect}
Cortar um retângulo pela diagonal dá dois triângulos de mesma \omterm{def:g4:measure:area}{área}:
cada um é \emph{\omterm{def:g3:division:half}{metade}} do retângulo. Um triângulo retângulo de
catetos $6$ cm e $4$ cm tem, portanto, \omterm{def:g4:measure:area}{área}
$\frac{6 \times 4}{2} = 12$ cm$^2$ --- uma imagem que vale a pena
guardar para o \cref{ch:g6:measure}, onde ela vira fórmula.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g5:areas:1}
Calcule a \omterm{def:g4:measure:area}{área} e o \omterm{def:g4:measure:perimeter}{perímetro} de um retângulo de $8$ cm $\times$
$5$ cm. Qual resposta vem em cm e qual em cm$^2$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:2}
Calcule as \omterm{def:g4:measure:area}{áreas}: um quadrado de lado $9$ cm; um retângulo
$12$ m $\times$ $7$ m; um retângulo $4.5$ cm $\times$ $6$ cm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:3}
Um retângulo tem \omterm{def:g4:measure:area}{área} $63$ cm$^2$ e comprimento $9$ cm. Encontre a
largura dele e depois o \omterm{def:g4:measure:perimeter}{perímetro}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:4}
Trace dois retângulos diferentes de \omterm{def:g4:measure:area}{área} $24$ quadradinhos no papel
quadriculado e calcule os dois \omterm{def:g4:measure:perimeter}{perímetros}. Mesma \omterm{def:g4:measure:area}{área} --- mesmo
\omterm{def:g4:measure:perimeter}{perímetro}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:5}
Converta: $3$ m$^2$ em cm$^2$; \quad $50\,000$ cm$^2$ em m$^2$.
(Lembre do \cref{ex:g5:areas:units}: de cem em cem!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:6}
Uma figura em forma de T é feita de uma barra horizontal $8 \times 2$
em cima de uma barra vertical $2 \times 5$ (em cm). Calcule a \omterm{def:g4:measure:area}{área}
dela somando dois retângulos.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:7}
Uma moldura de quadro: um retângulo $30$ cm $\times$ $20$ cm com uma
janela retangular de $24$ cm $\times$ $14$ cm recortada. Que \omterm{def:g4:measure:area}{área} de
madeira a moldura usa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:8}
Calcule a \omterm{def:g4:measure:area}{área} de um triângulo retângulo de catetos $8$ cm e $6$ cm
(\omterm{def:g3:division:half}{metade} de um retângulo, \cref{ex:g5:areas:halfrect}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:9}
Uma horta retangular mede $7$ m por $4$ m. A semente custa $2$ por
metro quadrado e a cerca, $3$ por metro. Calcule o custo da semente e
depois o custo da cerca.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:areas:10}
O piso de um corredor tem $12$ m de comprimento e $2$ m de largura, e
precisa ser coberto com ladrilhos quadrados de lado $50$ cm. Quantos
ladrilhos são necessários? (Converta primeiro, ou conte os ladrilhos
em cada direção.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:areas:11}
Dobre os lados de um retângulo de $4$ cm $\times$ $3$ cm. O que
acontece com o \omterm{def:g4:measure:perimeter}{perímetro}? E com a \omterm{def:g4:measure:area}{área}? (Calcule os dois antes e
depois --- as duas respostas são diferentes!)
\end{exercise}
