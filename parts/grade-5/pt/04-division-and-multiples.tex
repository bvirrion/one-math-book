\chapter{Divisão e múltiplos}\label{ch:g5:division}

A divisão armada aprendida no \cref{ch:g4:division} ganha dois
aperfeiçoamentos: \omterm{def:g5:division:multiple}{divisores} de dois algarismos e \omterm{def:g3:division:remainder}{quocientes} que
continuam depois do ponto --- $3 \div 4$ finalmente ganha uma
resposta, $0.75$. O capítulo termina com os \omterm{def:g5:division:multiple}{múltiplos}, os \omterm{def:g5:division:multiple}{divisores} e
os primeiros critérios de divisibilidade.

\section{Dividir por um número de dois algarismos}

\begin{method}[Divisores de dois algarismos]\label{met:g5:division:twodigits}
O mesmo método de antes, com uma habilidade a mais: adivinhar quantas
vezes o \omterm{def:g5:division:multiple}{divisor} cabe. Escreva antes uma pequena tabela de \omterm{def:g5:division:multiple}{múltiplos}
do \omterm{def:g5:division:multiple}{divisor}. Para $986 \div 23$ (\omterm{def:g5:division:multiple}{múltiplos} de 23: 23, 46, 69, 92, 115,
138, 161, 184, 207):
\begin{enumerate}
  \item $98$ dezenas: $23 \times 4 = 92$ cabe, $23 \times 5 = 115$
        não: algarismo $4$, \omterm{def:g3:division:remainder}{resto} $6$;
  \item baixe o $6$: $66$: $23 \times 2 = 46$ cabe: algarismo $2$,
        \omterm{def:g3:division:remainder}{resto} $20$;
  \item \omterm{def:g3:division:remainder}{quociente} $42$, \omterm{def:g3:division:remainder}{resto} $20$. Conferência:
        $23 \times 42 + 20 = 966 + 20 = 986$.
\end{enumerate}
\end{method}

\section{Quocientes decimais}

\begin{example}[A divisão que se recusa a parar no resto]\label{ex:g5:division:decimal}
Repartir $3$ pizzas entre $4$ crianças: $3 \div 4$ tem \omterm{def:g3:division:remainder}{quociente} $0$
e \omterm{def:g3:division:remainder}{resto} $3$ --- não serve de nada! Continue a divisão depois do
ponto:
\begin{enumerate}
  \item $3$ unidades $= 30$ décimos; $30 \div 4$: $7$ décimos, \omterm{def:g3:division:remainder}{resto}
        $2$ décimos;
  \item $2$ décimos $= 20$ centésimos; $20 \div 4 = 5$ centésimos,
        \omterm{def:g3:division:remainder}{resto} $0$.
\end{enumerate}
Logo $3 \div 4 = 0.75$: cada criança fica com $0.75$ pizza --- três
quartos, como a linguagem das frações já sabia
(\cref{prop:g5:fractions:decimal}).
\end{example}

\begin{method}[Continuar uma divisão depois do ponto]\label{met:g5:division:continue}
Quando as unidades acabam e ainda há \omterm{def:g3:division:remainder}{resto}:
\begin{enumerate}
  \item escreva o \omterm{def:g4:decimals:point}{ponto decimal} no \omterm{def:g3:division:remainder}{quociente};
  \item acrescente um \omterm{def:g1:counting:zero}{zero} ao \omterm{def:g3:division:remainder}{resto} (as unidades viram décimos, os
        décimos viram centésimos\,\dots) e continue dividindo;
  \item pare quando o \omterm{def:g3:division:remainder}{resto} for $0$ --- ou quando os algarismos
        começarem a se repetir para sempre, como em
        $1 \div 3 = 0.333\dots$: aí dê um valor arredondado.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:division:continue}
$27 \div 6$: \omterm{def:g3:division:remainder}{quociente} $4$, \omterm{def:g3:division:remainder}{resto} $3$; ponto; $30$ décimos
$\div\ 6 = 5$: logo $27 \div 6 = 4.5$. \quad
$22 \div 8$: $2$, \omterm{def:g3:division:remainder}{resto} $6$; depois $60 \div 8 = 7$ e \omterm{def:g3:division:remainder}{resto} $4$;
depois $40 \div 8 = 5$: logo $22 \div 8 = 2.75$.
\end{example}

\section{Múltiplos e divisores}

\begin{definition}[Múltiplo, divisor]\label{def:g5:division:multiple}
Os \emph{múltiplos}\index{múltiplo} de $6$ são a tabuada dele
continuada para sempre: $0, 6, 12, 18, 24, \dots$ Quando $24$ é
múltiplo de $6$, também se diz que $6$ é \emph{divisor}\index{divisor}
de $24$, ou que $24$ é \emph{divisível} por $6$: a divisão
$24 \div 6$ não deixa \omterm{def:g3:division:remainder}{resto}.
\end{definition}

\begin{proposition}[Critérios de divisibilidade]\label{prop:g5:division:criteria}
Sem dividir, um número é divisível:
\begin{itemize}
  \item por $2$ quando o último algarismo é $0$, $2$, $4$, $6$ ou $8$
        (os \omterm{def:g3:numbers:evenodd}{números pares});
  \item por $5$ quando o último algarismo é $0$ ou $5$;
  \item por $10$ quando o último algarismo é $0$;
  \item por $3$ quando a \emph{\omterm{def:g1:addition:def}{soma} dos algarismos} é divisível por
        $3$ (por exemplo $741$: $7 + 4 + 1 = 12$, divisível --- e de
        fato $741 = 3 \times 247$).
\end{itemize}
\end{proposition}
\admitted

\begin{example}\label{ex:g5:division:criteria}
$534$ é divisível por $2$? Termina em $4$: sim. Por $5$? Não. Por
$3$? $5 + 3 + 4 = 12$: sim. Então $534$ também é divisível por $6$
(por $2$ \emph{e} por $3$) --- confira: $534 = 6 \times 89$.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g5:division:1}
Faça as divisões armadas (tabela de \omterm{def:g5:division:multiple}{múltiplos} primeiro):
$851 \div 23$; \quad $704 \div 32$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:2}
Calcule os \omterm{def:g3:division:remainder}{quocientes} decimais exatos: $9 \div 2$; \quad
$27 \div 4$; \quad $33 \div 5$; \quad $21 \div 8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:3}
Quatro amigos dividem igualmente uma conta de restaurante de $54$.
Quanto cada um paga? (Continue depois do ponto.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:4}
Calcule $10 \div 3$, continuando três algarismos depois do ponto. O
que você observa? Dê o valor arredondado ao centésimo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:5}
Liste: os \omterm{def:g5:division:multiple}{múltiplos} de $7$ até $70$; os \omterm{def:g5:division:multiple}{divisores} de $18$; os
\omterm{def:g5:division:multiple}{divisores} de $24$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:6}
Divisível por $2$? por $5$? por $10$? por $3$? Teste cada critério
em: $470$; \quad $735$; \quad $8\,001$; \quad $1\,314$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:7}
Encontre todos os números entre $60$ e $80$ que são divisíveis por
$3$ \emph{e} por $5$. (Que divisibilidade única isso combina?)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:8}
Uma fita de $7.2$ m é cortada em $6$ pedaços iguais. Qual o
comprimento de cada pedaço? (Divida um decimal por um inteiro:
reparta os metros e depois os décimos.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:9}
$1\,000$ bolinhas são embaladas em saquinhos de $24$. Quantos
saquinhos cheios, e quantas bolinhas sobram? (\omterm{def:g5:division:multiple}{Divisor} de dois
algarismos.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:division:10}
$5$ amigos conseguem repartir $7$ barras de chocolate igualmente, sem
sobra? Dê a parte de cada um como decimal. A mesma pergunta para $7$
amigos repartindo $5$ barras --- o que muda?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:division:11}
Usando o critério da \omterm{def:g1:addition:def}{soma} dos algarismos, encontre o menor algarismo
$d$ que torna $52d$ (um número de três algarismos terminado em $d$)
divisível por $3$. Depois encontre todos esses algarismos $d$.
\end{exercise}
