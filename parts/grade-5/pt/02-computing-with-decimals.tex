\chapter{Calculando com decimais}\label{ch:g5:decimalops}

Os números decimais se somam, se subtraem e se multiplicam quase
como os inteiros --- toda a arte está em saber onde fica o ponto.
Este capítulo cobre as contas armadas e a mágica de multiplicar ou
dividir por $10$, $100$, $1000$. (Multiplicar dois decimais entre si
espera o \cref{ch:g6:decimals}.)

\section{Somar e subtrair}

\begin{method}[Contas armadas com ponto]\label{met:g5:decimalops:addsub}
\begin{enumerate}
  \item Escreva os números com os \emph{\omterm{def:g4:decimals:point}{pontos decimais} alinhados}
        (aí as unidades ficam sob as unidades e os décimos sob os
        décimos);
  \item complete com \omterm{def:g1:counting:zero}{zeros} a parte decimal mais curta;
  \item some ou subtraia como com números inteiros;
  \item desça o ponto do resultado em linha reta.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:decimalops:addsub}
$27.5 + 3.86$ (complete: $27.5 = 27.50$):
\[
\begin{array}{r}
2\,7.5\,0 \\
+\ \ \ 3.8\,6 \\
\hline
3\,1.3\,6
\end{array}
\qquad\qquad
\begin{array}{r}
1\,2.4\,0 \\
-\ \ \ 5.6\,5 \\
\hline
\ \ \,6.7\,5
\end{array}
\]
Na subtração ($12.4 - 5.65$): complete, peça emprestado como sempre e
confira somando de volta: $6.75 + 5.65 = 12.4$.
\end{example}

\begin{example}[Complementos de cabeça]\label{ex:g5:decimalops:complement}
Quanto falta somar a $7.6$ para chegar a $10$? Suba: de $7.6 \to 8$
falta $0.4$, e de $8 \to 10$ faltam $2$: o complemento é $2.4$. Útil
para o troco: pague $10$ por um artigo de $7.60$ e receba $2.40$.
\end{example}

\section{Dez vezes maior, dez vezes menor}

\begin{proposition}[Vezes e dividido por 10, 100, 1000]\label{prop:g5:decimalops:shift}
Multiplicar por $10$ faz cada algarismo valer dez vezes mais: os
algarismos deslizam uma casa para a esquerda --- o que parece ser o
\emph{ponto andando uma casa para a direita}. Dividir faz o
contrário:
\[
3.47 \times 10 = 34.7, \qquad
3.47 \times 100 = 347, \qquad
52.1 \div 10 = 5.21, \qquad
6 \div 100 = 0.06 .
\]
\end{proposition}
\admitted

\begin{example}[Conversões, de novo]\label{ex:g5:decimalops:convert}
As unidades de medida estão a potências de dez umas das outras, então
as conversões são exatamente esses deslocamentos: $2.35$ m $= 235$ cm
($\times 100$); $470$ g $= 0.47$ kg ($\div 1000$); $1.5$ L $= 150$
cL.
\end{example}

\section{Multiplicar um decimal por um número inteiro}

\begin{method}[Decimal vezes inteiro]\label{met:g5:decimalops:mult}
Calcule como se não houvesse ponto e depois dê ao resultado tantos
algarismos decimais quantos tem o fator decimal:
\[
4.35 \times 6: \quad 435 \times 6 = 2\,610
\ \to\ 4.35 \times 6 = 26.10 = 26.1 .
\]
Estime para pôr o ponto com segurança: $4.35$ está perto de $4$, e
$4 \times 6 = 24$ --- logo $26.1$, e não $2.61$ nem $261$.
\end{method}

\begin{example}[A adição repetida continua valendo]\label{ex:g5:decimalops:mult}
$0.75 \times 4 = 0.75 + 0.75 + 0.75 + 0.75 = 3$: quatro quartos fazem
um inteiro, com roupa decimal. Multiplicar por um número inteiro
continua sendo ``tantas vezes''.
\end{example}

\begin{example}[Compras]\label{ex:g5:decimalops:shopping}
Três cadernos a $2.45$ cada um e uma caneta a $1.20$:
\begin{enumerate}
  \item cadernos: $2.45 \times 3 = 7.35$;
  \item total: $7.35 + 1.20 = 8.55$;
  \item troco de $10$: $10 - 8.55 = 1.45$.
\end{enumerate}
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:1}
Arme a conta e calcule: $45.7 + 8.93$; \quad $16.25 + 3.75$; \quad
$0.86 + 12.4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:2}
Arme a conta e confira: $9.4 - 2.72$; \quad $20 - 13.45$;
\quad $6.03 - 5.9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:3}
Complementos de $10$, de cabeça (como no
\cref{ex:g5:decimalops:complement}): $6.2$; \quad $3.75$; \quad
$9.99$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:4}
Calcule sem armar a conta:
\[
7.24 \times 10, \qquad
0.58 \times 100, \qquad
34.5 \div 10, \qquad
7 \div 100, \qquad
0.3 \times 1000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:5}
Converta: $4.05$ m em cm; \quad $325$ cm em m; \quad $0.6$ kg em g;
\quad $85$ cL em L.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:6}
Estime primeiro e depois calcule: $6.2 \times 4$; \quad
$3.45 \times 8$; \quad $12.5 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:7}
Uma volta na pista mede $0.4$ km. Quanto são $7$ voltas? E $25$
voltas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:8}
Lia compra $2$ pães franceses a $1.15$ cada um e um bolo a $12.60$.
Ela paga com uma nota de $20$. Calcule o total e o troco dela.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:9}
Uma garrafa leva $1.5$ L. Quantos litros há num pacote de $6$
garrafas? Uma família bebe $0.75$ L por dia: para quantos dias dura
um pacote? (Quantas vezes $0.75$ cabe em $9$?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:decimalops:10}
Milo calcula $5.6 \times 3 = 15.18$, raciocinando ``$5 \times 3 = 15$
e $6 \times 3 = 18$''. Use uma estimativa para mostrar que a resposta
tem de estar errada e depois calcule corretamente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:decimalops:11}
As quatro voltas de uma corredora levaram $65.4$ s, $64.9$ s,
$65.0$ s e $66.1$ s.
\begin{enumerate}
  \item Calcule o tempo total das quatro voltas.
  \item O recorde das quatro voltas é $260$ s. Ela o bateu? Por
        quanto?
\end{enumerate}
\end{exercise}
