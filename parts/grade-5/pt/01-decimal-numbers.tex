\chapter{Números decimais}\label{ch:g5:decimals}

Os décimos e os centésimos apareceram no \cref{ch:g4:decimals};
acrescentar mais um nível, os milésimos, completa o sistema decimal.
Este capítulo consolida a leitura, a decomposição e --- acima de tudo
--- a comparação de números decimais sem cair nas armadilhas famosas
deles.

\section{Até os milésimos}

\begin{definition}[Casas decimais]\label{def:g5:decimals:places}
Depois do \omterm{def:g4:decimals:point}{ponto decimal} vêm os décimos, os centésimos, os milésimos
--- cada um valendo dez vezes menos que o anterior:
\[
4.362 = 4 + \frac{3}{10} + \frac{6}{100} + \frac{2}{1000}
= \frac{4\,362}{1000}.
\]
\end{definition}

\begin{example}[Várias leituras, um número]\label{ex:g5:decimals:readings}
$4.362$ pode ser lido ``quatro ponto três seis dois'', ou ``quatro
inteiros e $362$ milésimos'', ou decomposto como
$4 + 0.3 + 0.06 + 0.002$. E completar com \omterm{def:g1:counting:zero}{zeros} no fim não muda nada:
$4.362 = 4.3620$. Mas $4.362 \neq 4.0362$: um \omterm{def:g1:counting:zero}{zero} \emph{no meio}
empurra cada algarismo para uma casa menor.
\end{example}

\begin{example}[Onde ele fica na reta?]\label{ex:g5:decimals:line}
Para marcar $4.362$: entre $4$ e $5$; ampliando, entre $4.3$ e $4.4$;
ampliando de novo, entre $4.36$ e $4.37$, mais perto de $4.36$. Cada
algarismo decimal é mais um nível de ampliação.
\end{example}

\section{Comparar decimais}

\begin{method}[Comparar dois decimais]\label{met:g5:decimals:compare}
\begin{enumerate}
  \item Compare as partes inteiras: $6.1 > 5.987$.
  \item Partes inteiras iguais: compare as partes decimais algarismo
        por algarismo a partir do ponto --- décimos, depois
        centésimos, depois milésimos;
  \item completar com \omterm{def:g1:counting:zero}{zeros} no fim deixa a comparação justa: $2.5$
        contra $2.48$ vira $2.50$ contra $2.48$, e $50 > 48$
        centésimos.
\end{enumerate}
A armadilha, mais uma vez: \emph{uma parte decimal mais comprida não
significa um número maior} --- $2.48$ tem mais algarismos que $2.5$ e
é menor.
\end{method}

\begin{example}\label{ex:g5:decimals:order}
Ordene $7.3$; $7.09$; $7.31$; $7.299$. Complete até três casas
decimais: $7.300$; $7.090$; $7.310$; $7.299$. Então
\[
7.09 < 7.299 < 7.3 < 7.31 .
\]
Repare que $7.299 < 7.3$ mesmo com o $299$ parecendo grande: são
$299$ milésimos contra $300$ milésimos.
\end{example}

\begin{example}[Espremer entre dois decimais]\label{ex:g5:decimals:between}
Existe algum número entre $5.7$ e $5.8$? Sim, muitos: $5.75$,
$5.71$, $5.799$\,\dots\ E entre $5.79$ e $5.8$? Também muitos:
$5.795$, para começar. Entre dois decimais diferentes \emph{sempre}
cabe outro --- não existe um ``próximo'' número decimal.
\end{example}

\section{Arredondar decimais}

\begin{definition}[Arredondamento]\label{def:g5:decimals:round}
Para arredondar à unidade (ou ao décimo, ou ao centésimo), olhe o
algarismo seguinte: $5$ ou mais sobe, caso contrário desce. Assim,
$4.362$ arredonda para $4$ (unidade), para $4.4$ (décimo) e para
$4.36$ (centésimo).
\end{definition}

\begin{example}[Dinheiro arredonda ao centavo]\label{ex:g5:decimals:money}
Um preço calculado como $7.4983$ aparece como $7.50$: as quantias da
vida real são arredondadas ao centésimo. Repare na cascata:
$7.4983 \to 7.50$, em que o $49$ virou $50$ --- o \omterm{def:g4:numbers:round}{arredondamento} pode
se propagar por vários algarismos.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g5:decimals:1}
Escreva como número decimal: $5 + \frac{2}{10} + \frac{7}{1000}$;
\quad $\frac{85}{100}$; \quad $\frac{4\,507}{1000}$; \quad ``doze
inteiros e nove milésimos''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimals:2}
Em $27.418$: o que conta o $4$? E o $8$? Decomponha o número como na
\cref{def:g5:decimals:places}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimals:3}
Verdadeiro ou falso? Explique pelas casas.
$3.60 = 3.6$; \quad $0.5 = 0.05$; \quad $2.070 = 2.07$; \quad
$1.3 = 1.03$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimals:4}
Copie e complete com $<$, $>$ ou $=$:
\[
4.8 \;?\; 4.79, \qquad
0.302 \;?\; 0.32, \qquad
6.5 \;?\; 6.500, \qquad
9.09 \;?\; 9.1 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimals:5}
Ordene do menor para o maior: $3.14$; \ $3.4$; \ $3.041$; \
$3.104$; \ $3.41$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimals:6}
Entre quais dois números inteiros fica cada número? E entre quais
dois números com uma casa decimal? \quad $6.83$; \quad $0.492$; \quad
$12.06$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimals:7}
Arredonde $8.276$: à unidade; ao décimo; ao centésimo. Arredonde
$3.95$ ao décimo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimals:8}
Dê um número estritamente entre: $2.6$ e $2.7$; \quad $0.41$ e
$0.42$; \quad $5.99$ e $6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimals:9}
As alturas de quatro plantas são $0.85$ m, $1.2$ m, $0.9$ m e
$1.02$ m. Ordene-as da mais baixa para a mais alta.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:decimals:10}
Lúcia diz: ``$7.12 > 7.9$ porque $12 > 9$''. Corrija o erro dela,
comparando primeiro os décimos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:decimals:11}
Encontre todos os números com exatamente dois algarismos decimais que
arredondam para $6.4$ ao serem arredondados ao décimo. Quais são o
menor e o maior?
\end{exercise}
