\chapter{Frações como números}\label{ch:g5:fractions}

Uma \omterm{def:g4:fractions:def}{fração} não é só o desenho de uma torta: é um \emph{número} de
verdade, com um lugar na reta numérica --- às vezes exatamente o
mesmo lugar de um número decimal. Este capítulo liga os dois mundos.
(As regras completas da aritmética das frações vêm no
\cref{ch:g6:fractions} e mais adiante.)

\section{Frações na reta, de novo}

\begin{example}[Marcar e ler]\label{ex:g5:fractions:place}
Numa reta cortada em quintos, $\frac{7}{5}$ fica a $7$ passos do $0$:
depois do $1$ (que é $\frac55$), em $1 + \frac25$. Toda \omterm{def:g4:fractions:def}{fração} mora
em algum lugar da reta; \omterm{def:g4:fractions:def}{numerador} maior (com o mesmo \omterm{def:g4:fractions:def}{denominador})
significa mais à direita.
\end{example}

\begin{proposition}[O mesmo ponto, vários nomes]\label{prop:g5:fractions:equal}
Cortar cada parte em duas (ou em três\,\dots) não move o ponto:
\[
\frac{1}{2} = \frac{2}{4} = \frac{3}{6} = \frac{5}{10},
\qquad
\frac{2}{3} = \frac{4}{6} = \frac{20}{30}.
\]
Multiplicar (ou dividir) o \omterm{def:g4:fractions:def}{numerador} e o \omterm{def:g4:fractions:def}{denominador} pelo mesmo
número dá outro nome da mesma \omterm{def:g4:fractions:def}{fração}.
\end{proposition}
\admitted

\section{Frações e decimais}

\begin{proposition}[Frações com nome decimal]\label{prop:g5:fractions:decimal}
As frações de décimos, centésimos e milésimos \emph{são} números
decimais:
\[
\frac{3}{10} = 0.3, \qquad \frac{27}{100} = 0.27 .
\]
Algumas outras frações também têm nome decimal, encontrado passando
para décimos ou centésimos:
\[
\frac{1}{2} = \frac{5}{10} = 0.5, \qquad
\frac{1}{4} = \frac{25}{100} = 0.25, \qquad
\frac{3}{4} = 0.75, \qquad
\frac{1}{5} = \frac{2}{10} = 0.2 .
\]
Mas não todas: $\frac13 = 0.333\dots$ nunca termina --- a \omterm{def:g4:fractions:def}{fração} é o
único nome exato dele.
\end{proposition}
\admitted

\begin{example}[Escolher o nome mais prático]\label{ex:g5:fractions:choose}
Para pagar três quartos de $12$, o nome de \omterm{def:g4:fractions:def}{fração} é mais prático:
$\frac34$ de $12$ é $9$. Para comparar $\frac34$ e $0.8$, o nome
decimal é mais prático: $0.75 < 0.8$. Ser bilíngue compensa.
\end{example}

\section{Comparar frações}

\begin{method}[Três ferramentas de comparação]\label{met:g5:fractions:compare}
\begin{enumerate}
  \item \emph{Mesmo \omterm{def:g4:fractions:def}{denominador}}: ganha quem tem mais partes,
        $\frac57 > \frac37$;
  \item \emph{mesmo \omterm{def:g4:fractions:def}{numerador}}: ganham as partes maiores, ou seja,
        ganha o \omterm{def:g4:fractions:def}{denominador} \emph{menor}: $\frac35 > \frac38$ (os
        quintos são maiores que os oitavos);
  \item \emph{pontos de referência}: compare cada uma com $\frac12$
        ou com $1$: $\frac38 < \frac12 < \frac59$, logo $\frac38 <
        \frac59$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:fractions:compare}
Quem comeu mais pizza: Ali com $\frac58$ ou Bia com $\frac54$? A
\omterm{def:g4:fractions:def}{fração} de Bia é maior que $1$ (cinco quartos é mais que uma pizza
inteira) e a de Ali é menor que $1$: Bia comeu mais --- mais que uma
pizza inteira, aliás.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g5:fractions:1}
Trace uma reta numérica cortada em quintos de $0$ a $2$ e marque:
$\frac25$; \ $\frac55$; \ $\frac85$; \ $\frac{10}{5}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:2}
Complete os nomes equivalentes:
$\frac12 = \frac{?}{8}$; \quad
$\frac34 = \frac{?}{8}$; \quad
$\frac{2}{5} = \frac{4}{?}$; \quad
$\frac{6}{9} = \frac{2}{?}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:3}
Que frações nomeiam números inteiros? Dê o número inteiro quando for
o caso: $\frac82$; \quad $\frac{9}{4}$; \quad $\frac{15}{3}$; \quad
$\frac{20}{5}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:4}
Escreva como decimais: $\frac{7}{10}$; \quad $\frac{41}{100}$; \quad
$\frac12$; \quad $\frac34$; \quad $\frac15$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:5}
Escreva como frações (décimos ou centésimos): $0.9$; \quad $0.13$;
\quad $2.5$; \quad $0.75$ (e depois dê o nome em quartos).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:6}
Compare, dizendo qual ferramenta usou
(\cref{met:g5:fractions:compare}):
\[
\frac47 \;?\; \frac67, \qquad
\frac35 \;?\; \frac37, \qquad
\frac25 \;?\; \frac{7}{12}, \qquad
\frac98 \;?\; \frac{11}{12} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:7}
Ordene usando os nomes decimais: $\frac12$; \ $0.4$; \ $\frac34$; \
$0.6$; \ $\frac15$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:8}
Um quarto dos alunos de uma turma de $28$ toca um instrumento, e
\omterm{def:g3:division:half}{metade} pratica um esporte. Quantos alunos são em cada caso? Qual
grupo é maior?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:9}
Ligue cada \omterm{def:g4:fractions:def}{fração} a um tempo: $\frac14$ h, $\frac12$ h,
$\frac34$ h, $\frac{1}{60}$ h --- a $30$ min, $45$ min, $1$ min,
$15$ min.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:fractions:10}
Tomás bebeu $\frac35$ da garrafa dele, de $50$ cL; Ana bebeu
$\frac12$ da garrafa dela, de $60$ cL. Quem bebeu mais? (Calcule as
duas quantidades em cL --- comparar as frações puras não basta aqui,
e diga por quê.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:fractions:11}
Dê uma \omterm{def:g4:fractions:def}{fração} estritamente entre $\frac12$ e $1$; depois uma
estritamente entre $\frac12$ e $\frac34$. (Os nomes decimais podem
ajudar.)
\end{exercise}
