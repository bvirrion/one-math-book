\chapter{Fracciones como números}\label{ch:g5:fractions}

A \omterm{def:g4:fractions:def}{fracción} no es sólo la imagen de un pastel: es una auténtica
\emph{número}, con un lugar en la recta numérica --- a veces el mismo
mismo lugar que un número decimal. Este capítulo conecta los dos
mundos. (Las reglas completas de \omterm{def:g4:fractions:def}{fracción} entra la aritmética
\cref{ch:g6:fractions} y más allá.)

\section{Fracciones en la recta, otra vez}

\begin{example}[Colocar y leer]\label{ex:g5:fractions:place}
En una línea cortada en quintas, $\frac{7}{5}$ es $7$ pasos de $0$: pasado
$1$ (que es $\frac55$), en $1 + \frac25$. Cada \omterm{def:g4:fractions:def}{fracción} vive
en algún lugar de la línea; más grande \omterm{def:g4:fractions:def}{numerador} (mismo \omterm{def:g4:fractions:def}{denominador}) significa
más a la derecha.
\end{example}

\begin{proposition}[Mismo punto, muchos nombres.]\label{prop:g5:fractions:equal}
Cortando cada parte en dos (o tres\,\dots) no mueve el
punto:
\[
\frac{1}{2} = \frac{2}{4} = \frac{3}{6} = \frac{5}{10},
\qquad
\frac{2}{3} = \frac{4}{6} = \frac{20}{30}.
\]
Multiplicar (o dividir) el \omterm{def:g4:fractions:def}{numerador} y el \omterm{def:g4:fractions:def}{denominador} por el
mismo número da otro nombre del mismo \omterm{def:g4:fractions:def}{fracción}.
\end{proposition}
\admitted

\section{fracciones y decimales}

\begin{proposition}[Fracciones con nombres decimales]\label{prop:g5:fractions:decimal}
fracciones de décimas, centésimas, milésimas \emph{son} decimales
números:
\[
\frac{3}{10} = 0.3, \qquad \frac{27}{100} = 0.27 .
\]
Algún otro fracciones también tienen nombres decimales, que se encuentran cambiando a
décimas o centésimas:
\[
\frac{1}{2} = \frac{5}{10} = 0.5, \qquad
\frac{1}{4} = \frac{25}{100} = 0.25, \qquad
\frac{3}{4} = 0.75, \qquad
\frac{1}{5} = \frac{2}{10} = 0.2 .
\]
Pero no todos: $\frac13 = 0.333\dots $ nunca termina --- el \omterm{def:g4:fractions:def}{fracción} es
su único nombre exacto.
\end{proposition}
\admitted

\begin{example}[Elegir el nombre más práctico]\label{ex:g5:fractions:choose}
para pagar tres cuarteles de $12$: el \omterm{def:g4:fractions:def}{fracción} el nombre es más práctico,
$\frac34$ de $12$ es $9$. para comparar $\frac34$ y $0.8$: el
el nombre decimal es más útil, $0.75 < 0.8$. Ser bilingüe vale la pena.
\end{example}

\section{Comparar fracciones}

\begin{method}[Tres herramientas de comparación]\label{met:g5:fractions:compare}
\begin{enumerate}
  \item \emph{Mismo \omterm{def:g4:fractions:def}{denominador}}: más partes ganan,
        $\frac57 > \frac37$;
  \item \emph{mismo \omterm{def:g4:fractions:def}{numerador}}: las partes más grandes ganan, entonces \emph{menor}
        \omterm{def:g4:fractions:def}{denominador} gana: $\frac35 > \frac38$ (las quintas son más grandes
        de octavos);
  \item \emph{puntos de referencia}: comparar cada uno con $\frac12$ o a $1$:
        $\frac38 < \frac12 < \frac59$, entonces $\frac38 < \frac59$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:fractions:compare}
Quien comió más pizza: Ali con $\frac58$ o Bea con $\frac54$? bea
\omterm{def:g4:fractions:def}{fracción} es más grande que $1$ (cinco cuarteles es más de un todo
pizza), Ali's es más pequeño que $1$: Bea comió más --- más de un
pizza entera, de hecho.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g5:fractions:1}
Dibuja una recta numérica cortada en quintas desde $0$ a $2$ y lugar:
$\frac25$; \ $\frac55$; \ $\frac85$; \ $\frac{10}{5}$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:fractions:2}
Completa los nombres iguales:
$\frac12 = \frac{?}{8}$; \quad
$\frac34 = \frac{?}{8}$; \quad
$\frac{2}{5} = \frac{4}{?}$; \quad
$\frac{6}{9} = \frac{2}{?}$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:fractions:3}
Cual fracciones nombrar números enteros? Da el número entero cuando sea así:
$\frac82$; \quad $\frac{9}{4}$; \quad $\frac{15}{3}$; \quad
$\frac{20}{5}$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:fractions:4}
Escribe como decimales: $\frac{7}{10}$; \quad $\frac{41}{100}$; \quad
$\frac12$; \quad $\frac34$; \quad $\frac15$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:fractions:5}
escribir como fracciones (décimas o centésimas): $0.9$; \quad $0.13$;
\quad $2.5$; \quad $0.75$ (luego da su \omterm{def:g3:division:half}{cuarto} nombre).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:fractions:6}
Comparar, nombrando la herramienta utilizada
(\cref{met:g5:fractions:compare}):
\[
\frac47 \;?\; \frac67, \qquad
\frac35 \;?\; \frac37, \qquad
\frac25 \;?\; \frac{7}{12}, \qquad
\frac98 \;?\; \frac{11}{12} .
\]
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:fractions:7}
Ordenar con los nombres decimales: $\frac12$; \ $0.4$; \ $\frac34$; \
$0.6$; \ $\frac15$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:fractions:8}
A \omterm{def:g3:division:half}{cuarto} de los estudiantes de una clase de $28$ tocar un instrumento y
\omterm{def:g3:division:half}{medio} practicar un deporte. ¿Cuántos estudiantes tiene cada uno? ¿Qué grupo es más grande?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:fractions:9}
Combina cada uno \omterm{def:g4:fractions:def}{fracción} con un tiempo: $\frac14$ h, $\frac12$ h,
$\frac34$ h, $\frac{1}{60}$ h --- a $30$ minuto, $45$ minuto, $1$ minuto,
$15$ mín.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:fractions:10}
tom bebió $\frac35$ de su $50$ botella de cl; Ana bebió $\frac12$ de
ella $60$ botella de cl. ¿Quién bebió más? (Calcule ambas cantidades en cL ---
comparando lo desnudo fracciones no es suficiente aquí y diga por qué).
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:fractions:11}
dar un \omterm{def:g4:fractions:def}{fracción} estrictamente entre $\frac12$ y $1$; entonces uno
estrictamente entre $\frac12$ y $\frac34$. (Los nombres decimales pueden
ayuda.)
\end{exercise}
