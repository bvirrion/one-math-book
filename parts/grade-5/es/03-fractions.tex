\chapter{Las fracciones como números}\label{ch:g5:fractions}

Una \omterm{def:g4:fractions:def}{fracción} no es solo el dibujo de una tarta: es un auténtico
\emph{número}, con su sitio en la recta numérica --- a veces el mismo
sitio que un número decimal. Este capítulo conecta los dos
mundos. (Las reglas completas de la aritmética de fracciones llegan en el
\cref{ch:g6:fractions} y más adelante.)

\section{Otra vez las fracciones en la recta}

\begin{example}[Situar y leer]\label{ex:g5:fractions:place}
En una recta cortada en quintos, $\frac{7}{5}$ está a $7$ pasos del $0$: pasado
el $1$ (que es $\frac55$), en $1 + \frac25$. Toda \omterm{def:g4:fractions:def}{fracción} vive
en algún punto de la recta; a mayor \omterm{def:g4:fractions:def}{numerador} (con el mismo \omterm{def:g4:fractions:def}{denominador}),
más a la derecha.
\end{example}

\begin{proposition}[Un mismo punto, muchos nombres]\label{prop:g5:fractions:equal}
Cortar cada parte en dos (o en tres\,\dots) no mueve el
punto:
\[
\frac{1}{2} = \frac{2}{4} = \frac{3}{6} = \frac{5}{10},
\qquad
\frac{2}{3} = \frac{4}{6} = \frac{20}{30}.
\]
Multiplicar (o dividir) el \omterm{def:g4:fractions:def}{numerador} y el \omterm{def:g4:fractions:def}{denominador} por un
mismo número da otro nombre de la misma \omterm{def:g4:fractions:def}{fracción}.
\end{proposition}
\admitted

\section{Fracciones y decimales}

\begin{proposition}[Fracciones con nombre decimal]\label{prop:g5:fractions:decimal}
Las fracciones de décimas, centésimas y milésimas \emph{son} números
decimales:
\[
\frac{3}{10} = 0.3, \qquad \frac{27}{100} = 0.27 .
\]
Algunas otras fracciones también tienen nombre decimal, que se halla pasando a
décimas o a centésimas:
\[
\frac{1}{2} = \frac{5}{10} = 0.5, \qquad
\frac{1}{4} = \frac{25}{100} = 0.25, \qquad
\frac{3}{4} = 0.75, \qquad
\frac{1}{5} = \frac{2}{10} = 0.2 .
\]
Pero no todas: $\frac13 = 0.333\dots$ no se acaba nunca --- la \omterm{def:g4:fractions:def}{fracción} es
su único nombre exacto.
\end{proposition}
\admitted

\begin{example}[Elegir el nombre más cómodo]\label{ex:g5:fractions:choose}
Para pagar tres \omterm{def:g3:division:half}{cuartos} de $12$, el nombre de \omterm{def:g4:fractions:def}{fracción} es más cómodo:
$\frac34$ de $12$ es $9$. Para comparar $\frac34$ y $0.8$, el
nombre decimal es más cómodo: $0.75 < 0.8$. Ser bilingüe compensa.
\end{example}

\section{Comparar fracciones}

\begin{method}[Tres herramientas para comparar]\label{met:g5:fractions:compare}
\begin{enumerate}
  \item \emph{Mismo \omterm{def:g4:fractions:def}{denominador}}: gana quien tiene más partes,
        $\frac57 > \frac37$;
  \item \emph{mismo \omterm{def:g4:fractions:def}{numerador}}: ganan las partes más grandes, así que gana el
        \omterm{def:g4:fractions:def}{denominador} \emph{menor}: $\frac35 > \frac38$ (los quintos son mayores
        que los octavos);
  \item \emph{puntos de referencia}: compara cada una con $\frac12$ o con $1$:
        $\frac38 < \frac12 < \frac59$, así que $\frac38 < \frac59$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:fractions:compare}
¿Quién comió más pizza: Ali con $\frac58$ o Bea con $\frac54$? La
\omterm{def:g4:fractions:def}{fracción} de Bea es mayor que $1$ (cinco \omterm{def:g3:division:half}{cuartos} es más que una pizza
entera) y la de Ali es menor que $1$: comió más Bea --- de hecho, más de una
pizza entera.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:fractions:1}
Dibuja una recta numérica cortada en quintos del $0$ al $2$ y sitúa:
$\frac25$; \ $\frac55$; \ $\frac85$; \ $\frac{10}{5}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:2}
Completa los nombres iguales:
$\frac12 = \frac{?}{8}$; \quad
$\frac34 = \frac{?}{8}$; \quad
$\frac{2}{5} = \frac{4}{?}$; \quad
$\frac{6}{9} = \frac{2}{?}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:3}
¿Qué fracciones nombran números enteros? Di el número entero cuando así sea:
$\frac82$; \quad $\frac{9}{4}$; \quad $\frac{15}{3}$; \quad
$\frac{20}{5}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:4}
Escribe como decimales: $\frac{7}{10}$; \quad $\frac{41}{100}$; \quad
$\frac12$; \quad $\frac34$; \quad $\frac15$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:5}
Escribe como fracciones (décimas o centésimas): $0.9$; \quad $0.13$;
\quad $2.5$; \quad $0.75$ (y da después su nombre en \omterm{def:g3:division:half}{cuartos}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:6}
Compara y di qué herramienta has usado
(\cref{met:g5:fractions:compare}):
\[
\frac47 \;?\; \frac67, \qquad
\frac35 \;?\; \frac37, \qquad
\frac25 \;?\; \frac{7}{12}, \qquad
\frac98 \;?\; \frac{11}{12} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:7}
Ordena con los nombres decimales: $\frac12$; \ $0.4$; \ $\frac34$; \
$0.6$; \ $\frac15$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:8}
Un \omterm{def:g3:division:half}{cuarto} de los alumnos de una clase de $28$ tocan un instrumento y
la \omterm{def:g3:division:half}{mitad} practican un deporte. ¿Cuántos alumnos son en cada caso? ¿Qué grupo es mayor?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:9}
Empareja cada \omterm{def:g4:fractions:def}{fracción} con un tiempo: $\frac14$ h, $\frac12$ h,
$\frac34$ h, $\frac{1}{60}$ h --- con $30$ min, $45$ min, $1$ min
y $15$ min.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:fractions:10}
Tomás se bebió $\frac35$ de su botella de $50$ cL; Ana, $\frac12$ de
la suya de $60$ cL. ¿Quién bebió más? (Calcula las dos cantidades en cL ---
aquí no basta con comparar las fracciones a secas; explica por qué.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:fractions:11}
Da una \omterm{def:g4:fractions:def}{fracción} estrictamente comprendida entre $\frac12$ y $1$; después una
estrictamente comprendida entre $\frac12$ y $\frac34$. (Los nombres decimales pueden
ayudar.)
\end{exercise}
