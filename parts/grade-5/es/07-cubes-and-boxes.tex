\chapter{Cubos y ortoedros}\label{ch:g5:solids}

Salimos de la página plana: los \omterm{def:g5:solids:def}{cuerpos geométricos} son las formas del mundo real ---
dados, cajas de cereales, latas, balones. Este capítulo aprende a nombrarlos,
a contar sus \omterm{def:g5:solids:def}{caras} y \omterm{def:g5:solids:def}{aristas}, a desplegarlos en \omterm{def:g5:solids:net}{desarrollos} y a medir
su \omterm{def:g5:solids:volume}{volumen} contando \omterm{def:g5:solids:def}{cubos} pequeños. (Las fórmulas del \omterm{def:g5:solids:volume}{volumen} llegan en el
\cref{ch:g6:measure}.)

\section{La familia de los cuerpos}

\begin{definition}[Los cuerpos y su vocabulario]\label{def:g5:solids:def}
Un \emph{cuerpo geométrico}\index{cuerpo geométrico} ocupa espacio. Sus lados planos son las
\emph{caras}\index{cara}; las caras se encuentran a lo largo de las \emph{aristas}\index{arista};
las aristas se encuentran en los \emph{vértices}\index{vértice}. El retrato de
familia:
\begin{itemize}
  \item el \emph{cubo}\index{cubo}: $6$ caras cuadradas, $12$ aristas,
        $8$ vértices;
  \item el \emph{ortoedro}\index{ortoedro} (o caja rectangular): $6$
        caras rectangulares, $12$ aristas, $8$ vértices;
  \item el \emph{cilindro}\index{cilindro}: dos caras circulares y una cara enrollada
        (una lata);
  \item la \emph{pirámide}\index{pirámide}: una base poligonal y triángulos que se encuentran en
        un vértice;
  \item la \emph{bola} (esfera): una superficie perfectamente redonda, sin
        caras ni aristas.
\end{itemize}
\end{definition}

\begin{example}[Contar en el cubo]\label{ex:g5:solids:count}
Compruébalo en un dado: $6$ \omterm{def:g5:solids:def}{caras} (los números del $1$ al $6$), $8$ \omterm{def:g5:solids:def}{vértices}
(las esquinas) y $12$ \omterm{def:g5:solids:def}{aristas} (recórrelas con el dedo: $4$ arriba, $4$
abajo y $4$ verticales). Y
$6 + 8 = 12 + 2$ --- el mismo patrón curioso vale para el \omterm{def:g5:solids:def}{ortoedro} y
la \omterm{def:g5:solids:def}{pirámide} (el \cref{ex:g8:solids:count} vuelve sobre ello).
\end{example}

\section{Desarrollos}

\begin{definition}[Desarrollo]\label{def:g5:solids:net}
Un \emph{desarrollo}\index{desarrollo} de un cuerpo es un dibujo plano de todas sus
\omterm{def:g5:solids:def}{caras}, unidas por las \omterm{def:g5:solids:def}{aristas}, que se pliega hasta formar el cuerpo --- el
cuerpo desenvuelto, como una caja de cartón aplastada para reciclar.
\end{definition}

\begin{example}[Leer un desarrollo]\label{ex:g5:solids:read}
En un dado, las \omterm{def:g5:solids:def}{caras} opuestas suman $7$. En un \omterm{def:g5:solids:net}{desarrollo}, las \omterm{def:g5:solids:def}{caras} opuestas son
las que están separadas por exactamente un cuadrado en una fila (o al doblar una
esquina) --- ¡no las vecinas! Marcar las parejas en el \omterm{def:g5:solids:net}{desarrollo} antes de
plegar es un ejercicio estupendo para ver el espacio en plano.
\end{example}

\section{Volumen: contar cubos}

\begin{definition}[Volumen contando cubos]\label{def:g5:solids:volume}
El \emph{volumen}\index{volumen} de un cuerpo construido con \omterm{def:g5:solids:def}{cubos} unidad es
el número de \omterm{def:g5:solids:def}{cubos} que contiene. La unidad puede ser el centímetro cúbico
(cm$^3$) o el metro cúbico (m$^3$).
\end{definition}

\begin{example}[Contar un ortoedro capa a capa]\label{ex:g5:solids:layers}
Un \omterm{def:g5:solids:def}{ortoedro} de $4$ \omterm{def:g5:solids:def}{cubos} de largo, $3$ de ancho y $2$ de alto: la capa de abajo tiene
$4 \times 3 = 12$ \omterm{def:g5:solids:def}{cubos} y hay $2$ capas:
\[
12 \times 2 = 24 \text{ cubos} .
\]
Contar por capas funciona siempre --- y demuestra en silencio la
fórmula $V = L \times a \times h$ del \cref{ch:g6:measure}.
\end{example}

\begin{example}[Los mismos cubos, formas distintas]\label{ex:g5:solids:rearrange}
Con ocho \omterm{def:g5:solids:def}{cubos} unidad se puede construir un \omterm{def:g5:solids:def}{cubo} de $2 \times 2 \times 2$, una
barra de $8 \times 1 \times 1$ o una losa de $4 \times 2 \times 1$: tres
cuerpos distintos, un mismo \omterm{def:g5:solids:volume}{volumen} ($8$ \omterm{def:g5:solids:def}{cubos}). El \omterm{def:g5:solids:volume}{volumen} cuenta el
material, no la forma.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:solids:1}
Nombra un objeto real con forma de: \omterm{def:g5:solids:def}{cubo}; \omterm{def:g5:solids:def}{ortoedro}; \omterm{def:g5:solids:def}{cilindro}; bola;
\omterm{def:g5:solids:def}{pirámide}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:2}
¿Cuántas \omterm{def:g5:solids:def}{caras}, \omterm{def:g5:solids:def}{aristas} y \omterm{def:g5:solids:def}{vértices} tiene un \omterm{def:g5:solids:def}{ortoedro}? Comprueba el patrón
\omterm{def:g5:solids:def}{caras} $+$ \omterm{def:g5:solids:def}{vértices} $=$ \omterm{def:g5:solids:def}{aristas} $+ 2$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:3}
¿Qué \omterm{def:g5:solids:def}{caras} de un \omterm{def:g5:solids:def}{ortoedro} son iguales? (Agrupa las seis \omterm{def:g5:solids:def}{caras} en parejas.)
¿Qué tienen de especial las \omterm{def:g5:solids:def}{caras} del \omterm{def:g5:solids:def}{cubo}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:4}
Dibuja un \omterm{def:g5:solids:net}{desarrollo} de un \omterm{def:g5:solids:def}{cubo} de lado $2$ cm (usa el modelo en forma de cruz
del capítulo), recórtalo y pliégalo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:5}
En el \omterm{def:g5:solids:net}{desarrollo} en cruz de un dado, coloca los números del $1$ al $6$ de modo
que las \omterm{def:g5:solids:def}{caras} opuestas sumen $7$. (Usa el \cref{ex:g5:solids:read} para
hallar antes las parejas opuestas.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:6}
Cuenta los \omterm{def:g5:solids:def}{cubos} unidad: un \omterm{def:g5:solids:def}{ortoedro} de $5$ de largo, $2$ de ancho y $3$ de alto; un \omterm{def:g5:solids:def}{cubo} de
lado $3$; una pila en forma de L hecha de una losa de $3 \times 2 \times 1$
con otra de $1 \times 2 \times 1$ encima.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:7}
Un \omterm{def:g5:solids:def}{ortoedro} contiene exactamente $12$ \omterm{def:g5:solids:def}{cubos} unidad en una capa. Da dos formas posibles
de capa ($L \times a$) y, para cada una, el \omterm{def:g5:solids:volume}{volumen} del \omterm{def:g5:solids:def}{ortoedro} si tiene
$3$ capas iguales.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:8}
Con $27$ \omterm{def:g5:solids:def}{cubos} unidad, ¿qué \omterm{def:g5:solids:def}{cubo} puedes construir? ¿Y con $64$? (Piensa en
el recuento por capas al revés.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:solids:9}
Los terrones de azúcar de $1$ cm de lado vienen en una caja de $10$ cm de largo, $5$ cm de ancho
y $4$ cm de alto (medidas interiores). ¿Cuántos terrones caben? Si una
familia gasta $6$ terrones al día, ¿para cuántos días le dura una caja
(redondea con sentido común)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:solids:10}
Un \omterm{def:g5:solids:def}{cubo} de $3 \times 3 \times 3$ se pinta de rojo por fuera y luego se corta
en $27$ \omterm{def:g5:solids:def}{cubos} unidad. ¿Cuántos \omterm{def:g5:solids:def}{cubos} pequeños tienen pintura en exactamente
$3$ \omterm{def:g5:solids:def}{caras}? ¿Y en $2$? ¿Y en $1$? ¿Y en ninguna? (Comprobación: los cuatro recuentos deben sumar
$27$.)
\end{exercise}
