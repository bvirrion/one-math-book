\chapter{Cubos y ortoedros}\label{ch:g5:solids}

Saliendo de la página plana: \omterm{def:g5:solids:def}{sólidos} son las formas del mundo real ---
dados, \omterm{def:g5:solids:def}{cajas} de cereales, latas, pelotas. Este capítulo aprende a nombrarlos,
contar sus \omterm{def:g5:solids:def}{caras} y \omterm{def:g5:solids:def}{bordes}, desdóblalos en redesy medir
su \omterm{def:g5:solids:volume}{volumen} contando poco \omterm{def:g5:solids:def}{cubos}. (\omterm{def:g5:solids:volume}{volumen} las fórmulas entran
\cref{ch:g6:measure}.)

\section{La familia solida}

\begin{definition}[Sólidos y su vocabulario.]\label{def:g5:solids:def}
A \emph{sólido}\index{sólido} ocupa espacio. Los lados planos son
\emph{caras}\index{caras}; caras se encuentran \emph{bordes}\index{bordes};
los bordes se encuentran en \emph{vértices}\index{vértices}. la familia
retratos:
\begin{itemize}
  \item el \emph{cubo}\index{cubo}: $6$ caras cuadradas, $12$ bordes,
        $8$ vértices;
  \item el \emph{caja}\index{caja} (prisma rectangular): $6$
        caras rectangulares, $12$ bordes, $8$ vértices;
  \item el \emph{cilindro}\index{cilindro}: dos caras de disco y una cara laminada
        (una lata);
  \item el \emph{pirámide}\index{pirámide}: a \omterm{def:g4:geometry:polygon}{polígono} base y triángulos se encuentran en
        un ápice;
  \item el \emph{pelota} (esfera): una superficie perfectamente redonda, no
        cara, sin filo.
\end{itemize}
\end{definition}

\begin{example}[contando con el cubo]\label{ex:g5:solids:count}
Comprueba un dado: $6$ \omterm{def:g5:solids:def}{caras} (los números $1$ a $6$), $8$ \omterm{def:g5:solids:def}{vértices}
(las esquinas), $12$ \omterm{def:g5:solids:def}{bordes} (pasa un dedo por ellos: $4$ encima, $4$
en la parte inferior, $4$ vertical). Y
$6 + 8 = 12 + 2$ --- el mismo patrón curioso se aplica a la \omterm{def:g5:solids:def}{caja} y
el \omterm{def:g5:solids:def}{pirámide} (\cref{ex:g8:solids:count} vuelve a él).
\end{example}

\section{Redes}

\begin{definition}[Neto]\label{def:g5:solids:net}
A \emph{neto}\index{neto} de un \omterm{def:g5:solids:def}{sólido} es un dibujo plano de todas sus
\omterm{def:g5:solids:def}{caras}, adjunto a lo largo \omterm{def:g5:solids:def}{bordes}, que se pliega en el \omterm{def:g5:solids:def}{sólido} --- el
\omterm{def:g5:solids:def}{sólido} desenvuelto como un cartón \omterm{def:g5:solids:def}{caja} aplanado para su reciclaje.
\end{definition}

\section{Volumen: contar cubos}

\begin{definition}[Volumen contando]\label{def:g5:solids:volume}
El \emph{volumen}\index{volumen} de un \omterm{def:g5:solids:def}{sólido} construido a partir de la unidad \omterm{def:g5:solids:def}{cubos} es
el número de \omterm{def:g5:solids:def}{cubos} contiene. La unidad puede ser el centímetro. \omterm{def:g5:solids:def}{cubo}
(centímetro $^3$) o el medidor \omterm{def:g5:solids:def}{cubo} (metro $^3$).
\end{definition}

\begin{example}[Contando una caja capa por capa]\label{ex:g5:solids:layers}
A \omterm{def:g5:solids:def}{caja} $4$ \omterm{def:g5:solids:def}{cubos} largo, $3$ ancho, $2$ alto: la capa inferior sostiene
$4 \times 3 = 12$ \omterm{def:g5:solids:def}{cubos}, y hay $2$ capas:
\[
12 \times 2 = 24 \text{ cubes} .
\]
Contar por capas siempre funciona, y demuestra silenciosamente la
fórmula $ V = L \times w \times h $ de \cref{ch:g6:measure}.
\end{example}

\begin{example}[Mismos cubos, diferentes formas.]\label{ex:g5:solids:rearrange}
Ocho unidades \omterm{def:g5:solids:def}{cubos} puede construir un $2 \times 2 \times 2$ \omterm{def:g5:solids:def}{cubo}, a
$8 \times 1 \times 1$ palo, o un $4 \times 2 \times 1$ losa: tres
diferente \omterm{def:g5:solids:def}{sólidos}, uno \omterm{def:g5:solids:volume}{volumen} ($8$ \omterm{def:g5:solids:def}{cubos}). \omterm{def:g5:solids:volume}{volumen} cuenta el
material, no la forma.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g5:solids:1}
Nombra un objeto real con forma de: \omterm{def:g5:solids:def}{cubo}; a \omterm{def:g5:solids:def}{caja}; a \omterm{def:g5:solids:def}{cilindro}; una pelota;
a \omterm{def:g5:solids:def}{pirámide}.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:solids:2}
Cuántos \omterm{def:g5:solids:def}{caras}, \omterm{def:g5:solids:def}{bordes} y \omterm{def:g5:solids:def}{vértices} tiene un \omterm{def:g5:solids:def}{caja}? Comprueba el patrón
\omterm{def:g5:solids:def}{caras} $+$ \omterm{def:g5:solids:def}{vértices} $=$ \omterm{def:g5:solids:def}{bordes} $+ 2$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:solids:3}
Cual \omterm{def:g5:solids:def}{caras} de un \omterm{def:g5:solids:def}{caja} son identicos? (Agrupa los seis \omterm{def:g5:solids:def}{caras} en parejas.)
¿Qué tiene de especial el \omterm{def:g5:solids:def}{cubo}'s \omterm{def:g5:solids:def}{caras}?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:solids:4}
dibuja un \omterm{def:g5:solids:net}{neto} de un \omterm{def:g5:solids:def}{cubo} de lado $2$ cm (use el modelo en forma de cruz de
el capítulo), recórtalo y dóblalo.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:solids:5}
En forma de cruz \omterm{def:g5:solids:net}{neto} de un dado, coloca los números $1$ a $6$ entonces
ese opuesto \omterm{def:g5:solids:def}{caras} \omterm{def:g1:addition:def}{suma} a $7$. (Usar \cref{ex:g5:solids:read} a
encuentre primero los \omterm{def:g3:numbers:evenodd}{pares} opuestos.)
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:solids:6}
contar la unidad \omterm{def:g5:solids:def}{cubos}: a \omterm{def:g5:solids:def}{caja} $5$ largo, $2$ ancho, $3$ alto; a \omterm{def:g5:solids:def}{cubo} de lado $3$; una pila en forma de L hecha de $3 \times 2 \times 1$ losa
con un $1 \times 2 \times 1$ losa encima.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:solids:7}
A \omterm{def:g5:solids:def}{caja} se mantiene exactamente $12$ unidad \omterm{def:g5:solids:def}{cubos} en una sola capa. Da dos posibles
formas de capa ($ L \times w $), y para cada uno, el \omterm{def:g5:solids:def}{caja}'s \omterm{def:g5:solids:volume}{volumen} si es
tiene $3$ tales capas.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:solids:8}
Con $27$ unidad \omterm{def:g5:solids:def}{cubos}, qué \omterm{def:g5:solids:def}{cubo} ¿puedes construir? Con $64$? (Piensa en
la capa cuenta hacia atrás.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:solids:9}
Azúcar \omterm{def:g5:solids:def}{cubos} de lado $1$ cm vienen en un \omterm{def:g5:solids:def}{caja} $10$ centímetros de largo, $5$ centímetros de ancho,
$4$ cm de alto (medidas interiores). cuantos azucar \omterm{def:g5:solids:def}{cubos} ¿adaptar? si un
usos familiares $6$ \omterm{def:g5:solids:def}{cubos} un día, ¿por cuántos días uno \omterm{def:g5:solids:def}{caja} último
(\omterm{def:g4:numbers:round}{redondo} sesudamente)?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:solids:10}
A $3 \times 3 \times 3$ \omterm{def:g5:solids:def}{cubo} Se pinta de rojo por fuera y luego se corta.
en $27$ unidad \omterm{def:g5:solids:def}{cubos}. cuantos pequeños \omterm{def:g5:solids:def}{cubos} tener pintura exactamente
$3$ \omterm{def:g5:solids:def}{caras}? En $2$? En $1$? ¿En ninguno? (Compruebe: las cuatro cuentas deben sumar
hasta $27$.)
\end{exercise}
