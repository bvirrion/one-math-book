\chapter{División y múltiplos}\label{ch:g5:division}

División larga aprendida en \cref{ch:g4:division} obtiene dos actualizaciones:
dos dígitos \omterm{def:g5:division:multiple}{divisores}, y \omterm{def:g3:division:remainder}{conscientes} que continúan más allá del punto ---
$3 \div 4$ finalmente obtiene una respuesta, $0.75$. El capítulo termina con
múltiplos, \omterm{def:g5:division:multiple}{divisores} y los primeros atajos de divisibilidad.

\section{Dividiendo por un número de dos dígitos}

\begin{method}[Divisores de dos dígitos]\label{met:g5:division:twodigits}
Mismo método que antes, con una habilidad extra: adivinar cuántas veces
el \omterm{def:g5:division:multiple}{divisor} encaja. Escribe una pequeña tabla de múltiplos del \omterm{def:g5:division:multiple}{divisor}
primero. Para $986 \div 23$ (múltiplos de 23: 23, 46, 69, 92, 115,
138, 161, 184, 207):
\begin{enumerate}
  \item $98$ decenas: $23 \times 4 = 92$ encaja, $23 \times 5 = 115$ hace
        no: dígito $4$, \omterm{def:g3:division:remainder}{resto} $6$;
  \item derribar el $6$: $66$: $23 \times 2 = 46$ encaja: dígito
        $2$, \omterm{def:g3:division:remainder}{resto} $20$;
  \item cociente $42$, \omterm{def:g3:division:remainder}{resto} $20$. Controlar:
        $23 \times 42 + 20 = 966 + 20 = 986$.
\end{enumerate}
\end{method}

\section{Cocientes decimales}

\begin{example}[La división que se niega a quedarse en el resto]\label{ex:g5:division:decimal}
Compartir $3$ pizzas entre $4$ niños: $3 \div 4$ tiene cociente $0$ y
\omterm{def:g3:division:remainder}{resto} $3$ --- ¡inútil! Continuar la división más allá del punto:
\begin{enumerate}
  \item $3$ unidades $= 30$ décimos; $30 \div 4$: $7$ décimos, \omterm{def:g3:division:remainder}{resto}
        $2$ décimos;
  \item $2$ décimas $= 20$ centésimas; $20 \div 4 = 5$ centésimas,
        \omterm{def:g3:division:remainder}{resto} $0$.
\end{enumerate}
Entonces $3 \div 4 = 0.75$: cada niño recibe $0.75$ pizza --- tres
cuarteles, como el \omterm{def:g4:fractions:def}{fracción} el idioma ya lo sabia
(\cref{prop:g5:fractions:decimal}).
\end{example}

\begin{method}[Continuar una división más allá del punto]\label{met:g5:division:continue}
Cuando las unidades están agotadas y un \omterm{def:g3:division:remainder}{resto} queda:
\begin{enumerate}
  \item escribe el \omterm{def:g4:decimals:point}{coma decimal} en el cociente;
  \item agregar un \omterm{def:g1:counting:zero}{cero} hacia \omterm{def:g3:division:remainder}{resto} (las unidades se convierten en décimas, décimas
        convertirse en centésimas\,\dots) y seguir dividiendo;
  \item detenerse cuando el \omterm{def:g3:division:remainder}{resto} es $0$ --- o cuando comienzan los dígitos
        repitiendo por siempre, como $1 \div 3 = 0.333\dots $: entonces dale un
        valor redondeado.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:division:continue}
$27 \div 6$: cociente $4$, \omterm{def:g3:division:remainder}{resto} $3$; punto; $30$ décimas
$\div\ 6 = 5$: entonces $27 \div 6 = 4.5$. \quad
$22 \div 8$: $2$, \omterm{def:g3:division:remainder}{resto} $6$; entonces $60 \div 8 = 7$ r $4$;
entonces $40 \div 8 = 5$: entonces $22 \div 8 = 2.75$.
\end{example}

\section{Múltiplos y divisores}

\begin{definition}[múltiplo, divisor]\label{def:g5:division:multiple}
El \emph{múltiples}\index{múltiples} de $6$ ¿Continúa su tabla?
para siempre: $0, 6, 12, 18, 24, \dots $ Cuando $24$ es un múltiplo de $6$,
uno tambien dice que $6$ es un \emph{divisor}\index{divisor} de $24$,
o eso $24$ es \emph{divisible} por $6$: la división
$24 \div 6$ no deja \omterm{def:g3:division:remainder}{resto}.
\end{definition}

\begin{proposition}[Atajos de divisibilidad]\label{prop:g5:division:criteria}
Sin dividir, un número es divisible:
\begin{itemize}
  \item por $2$ cuando su último dígito es $0$, $2$, $4$, $6$ o $8$
        (el números \omterm{def:g3:numbers:evenodd}{pares});
  \item por $5$ cuando su último dígito es $0$ o $5$;
  \item por $10$ cuando su último dígito es $0$;
  \item por $3$ cuando el \emph{\omterm{def:g1:addition:def}{suma} de sus dígitos} es divisible por
        $3$ (p.ej.\ $741$: $7 + 4 + 1 = 12$, divisible --- y de hecho
        $741 = 3 \times 247$).
\end{itemize}
\end{proposition}
\admitted

\begin{example}\label{ex:g5:division:criteria}
Es $534$ divisible por $2$? termina en $4$: Sí. Por $5$? No. Por $3$?
$5 + 3 + 4 = 12$: Sí. Entonces $534$ es divisible por $6$ también (por $2$
\emph{y} por $3$) --- controlar: $534 = 6 \times 89$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g5:division:1}
Calcular las divisiones largas (tabla de múltiplos primero):
$851 \div 23$; \quad $704 \div 32$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:division:2}
Calcular el decimal exacto \omterm{def:g3:division:remainder}{conscientes}: $9 \div 2$; \quad $27 \div 4$;
\quad $33 \div 5$; \quad $21 \div 8$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:division:3}
Cuatro amigos comparten un $54$ factura del restaurante por igual. cuanto cuesta
cada uno paga? (Continúe más allá del punto.)
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:division:4}
Calcular $10 \div 3$, continuando tres dígitos más allá del punto. ¿Qué hacer?
¿observas? Da el valor redondeado a la centésima.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:division:5}
Lista: el múltiplos de $7$ arriba a $70$; el \omterm{def:g5:division:multiple}{divisores} de $18$; el
\omterm{def:g5:division:multiple}{divisores} de $24$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:division:6}
divisible por $2$? por $5$? por $10$? por $3$? Pruebe cada atajo en:
$470$; \quad $735$; \quad $8\,001$; \quad $1\,314$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:division:7}
Encuentra todos los números entre $60$ y $80$ que son divisibles por
$3$ \emph{y} por $5$. (¿Qué divisibilidad única hace eso?
combinar?)
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:division:8}
una cinta de $7.2$ m se corta en $6$ piezas iguales. cuanto mide cada uno
pieza? (Dividir un decimal por un entero: compartir los metros, luego el
décimas.)
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:division:9}
$1\,000$ Las canicas se empaquetan en bolsas de $24$. ¿Cuántas bolsas llenas y
¿Cuántas canicas quedan? (Dos dígitos \omterm{def:g5:division:multiple}{divisor}.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:division:10}
Poder $5$ amigos comparten $7$ barras de chocolate por igual, sin quedar nada?
Da cada acción como un decimal. La misma pregunta para $7$ amigos compartiendo
$5$ barras --- ¿qué es diferente?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:division:11}
Usando el \omterm{def:g1:addition:def}{suma} de dígitos atajo, encuentra el dígito más pequeño $ d $ eso hace
$52d $ (un número de tres dígitos que termina en $ d $) divisible por $3$. entonces
encontrar todos esos dígitos $ d$.
\end{exercise}
