\chapter{División y múltiplos}\label{ch:g5:division}

La división larga aprendida en el \cref{ch:g4:division} recibe dos mejoras:
\omterm{def:g5:division:multiple}{divisores} de dos cifras y \omterm{def:g3:division:remainder}{cocientes} que continúan pasado el punto ---
$3 \div 4$ tiene por fin respuesta, $0.75$. El capítulo termina con los
\omterm{def:g5:division:multiple}{múltiplos}, los \omterm{def:g5:division:multiple}{divisores} y los primeros criterios de divisibilidad.

\section{Dividir por un número de dos cifras}

\begin{method}[Divisores de dos cifras]\label{met:g5:division:twodigits}
El mismo método de antes, con una habilidad más: adivinar cuántas veces
cabe el \omterm{def:g5:division:multiple}{divisor}. Escribe primero una tablita de \omterm{def:g5:division:multiple}{múltiplos} del \omterm{def:g5:division:multiple}{divisor}.
Para $986 \div 23$ (\omterm{def:g5:division:multiple}{múltiplos} de 23: 23, 46, 69, 92, 115,
138, 161, 184, 207):
\begin{enumerate}
  \item $98$ decenas: $23 \times 4 = 92$ cabe y $23 \times 5 = 115$ no:
        cifra $4$, \omterm{def:g3:division:remainder}{resto} $6$;
  \item baja el $6$: $66$: $23 \times 2 = 46$ cabe: cifra
        $2$, \omterm{def:g3:division:remainder}{resto} $20$;
  \item \omterm{def:g3:division:remainder}{cociente} $42$, \omterm{def:g3:division:remainder}{resto} $20$. Comprobación:
        $23 \times 42 + 20 = 966 + 20 = 986$.
\end{enumerate}
\end{method}

\section{Cocientes decimales}

\begin{example}[La división que se niega a parar en el resto]\label{ex:g5:division:decimal}
Reparte $3$ pizzas entre $4$ niños: $3 \div 4$ tiene \omterm{def:g3:division:remainder}{cociente} $0$ y
\omterm{def:g3:division:remainder}{resto} $3$ --- ¡inútil! Sigue la división pasado el punto:
\begin{enumerate}
  \item $3$ unidades $= 30$ décimas; $30 \div 4$: $7$ décimas, \omterm{def:g3:division:remainder}{resto}
        $2$ décimas;
  \item $2$ décimas $= 20$ centésimas; $20 \div 4 = 5$ centésimas,
        \omterm{def:g3:division:remainder}{resto} $0$.
\end{enumerate}
Así, $3 \div 4 = 0.75$: cada niño recibe $0.75$ de pizza --- tres
\omterm{def:g3:division:half}{cuartos}, como ya sabía el lenguaje de las fracciones
(\cref{prop:g5:fractions:decimal}).
\end{example}

\begin{method}[Continuar una división pasado el punto]\label{met:g5:division:continue}
Cuando se acaban las unidades y queda \omterm{def:g3:division:remainder}{resto}:
\begin{enumerate}
  \item escribe el \omterm{def:g4:decimals:point}{punto decimal} en el \omterm{def:g3:division:remainder}{cociente};
  \item añade un \omterm{def:g1:counting:zero}{cero} al \omterm{def:g3:division:remainder}{resto} (las unidades pasan a décimas, las décimas
        a centésimas\,\dots) y sigue dividiendo;
  \item para cuando el \omterm{def:g3:division:remainder}{resto} sea $0$ --- o cuando las cifras empiecen
        a repetirse indefinidamente, como en $1 \div 3 = 0.333\dots$: entonces da un
        valor redondeado.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:division:continue}
$27 \div 6$: \omterm{def:g3:division:remainder}{cociente} $4$, \omterm{def:g3:division:remainder}{resto} $3$; punto; $30$ décimas
$\div\ 6 = 5$: así, $27 \div 6 = 4.5$. \quad
$22 \div 8$: $2$, \omterm{def:g3:division:remainder}{resto} $6$; luego $60 \div 8 = 7$ y \omterm{def:g3:division:remainder}{resto} $4$;
luego $40 \div 8 = 5$: así, $22 \div 8 = 2.75$.
\end{example}

\section{Múltiplos y divisores}

\begin{definition}[Múltiplo, divisor]\label{def:g5:division:multiple}
Los \emph{múltiplos}\index{múltiplo} de $6$ son su tabla continuada
indefinidamente: $0, 6, 12, 18, 24, \dots$ Cuando $24$ es múltiplo de $6$,
también se dice que $6$ es un \emph{divisor}\index{divisor} de $24$,
o que $24$ es \emph{divisible} entre $6$: la división
$24 \div 6$ no deja \omterm{def:g3:division:remainder}{resto}.
\end{definition}

\begin{proposition}[Criterios de divisibilidad]\label{prop:g5:division:criteria}
Sin dividir, un número es divisible:
\begin{itemize}
  \item entre $2$ cuando su última cifra es $0$, $2$, $4$, $6$ u $8$
        (los \omterm{def:g3:numbers:evenodd}{números pares});
  \item entre $5$ cuando su última cifra es $0$ o $5$;
  \item entre $10$ cuando su última cifra es $0$;
  \item entre $3$ cuando la \emph{\omterm{def:g1:addition:def}{suma} de sus cifras} es divisible entre
        $3$ (por ejemplo, $741$: $7 + 4 + 1 = 12$, divisible --- y, en efecto,
        $741 = 3 \times 247$).
\end{itemize}
\end{proposition}
\admitted

\begin{example}\label{ex:g5:division:criteria}
¿Es $534$ divisible entre $2$? Termina en $4$: sí. ¿Entre $5$? No. ¿Entre $3$?
$5 + 3 + 4 = 12$: sí. Por tanto, $534$ también es divisible entre $6$ (entre $2$
\emph{y} entre $3$) --- comprobación: $534 = 6 \times 89$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:division:1}
Calcula las divisiones largas (haz antes la tabla de \omterm{def:g5:division:multiple}{múltiplos}):
$851 \div 23$; \quad $704 \div 32$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:2}
Calcula los \omterm{def:g3:division:remainder}{cocientes} decimales exactos: $9 \div 2$; \quad $27 \div 4$;
\quad $33 \div 5$; \quad $21 \div 8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:3}
Cuatro amigos comparten a partes iguales una cuenta de restaurante de $54$. ¿Cuánto paga
cada uno? (Continúa pasado el punto.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:4}
Calcula $10 \div 3$ continuando tres cifras pasado el punto. ¿Qué
observas? Da el valor redondeado a la centésima.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:5}
Enumera: los \omterm{def:g5:division:multiple}{múltiplos} de $7$ hasta $70$; los \omterm{def:g5:division:multiple}{divisores} de $18$; los
\omterm{def:g5:division:multiple}{divisores} de $24$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:6}
¿Divisible entre $2$? ¿Entre $5$? ¿Entre $10$? ¿Entre $3$? Prueba cada criterio con:
$470$; \quad $735$; \quad $8\,001$; \quad $1\,314$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:7}
Halla todos los números entre $60$ y $80$ que son divisibles entre
$3$ \emph{y} entre $5$. (¿Qué única divisibilidad resume esa
combinación?)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:8}
Una cinta de $7.2$ m se corta en $6$ trozos iguales. ¿Cuánto mide cada
trozo? (Divide un decimal entre un entero: reparte los metros y después las
décimas.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:division:9}
Se empaquetan $1\,000$ canicas en bolsas de $24$. ¿Cuántas bolsas se llenan y
cuántas canicas sobran? (\omterm{def:g5:division:multiple}{Divisor} de dos cifras.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:division:10}
¿Pueden $5$ amigos repartirse $7$ tabletas de chocolate a partes iguales sin que sobre nada?
Da la parte de cada uno como decimal. La misma pregunta para $7$ amigos que se reparten
$5$ tabletas --- ¿qué ocurre de distinto?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:division:11}
Usando el criterio de la \omterm{def:g1:addition:def}{suma} de cifras, halla la menor cifra $d$ que hace
$52d$ (un número de tres cifras terminado en $d$) divisible entre $3$. Después
halla todas esas cifras $d$.
\end{exercise}
