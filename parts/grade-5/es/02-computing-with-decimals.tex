\chapter{Calcular con decimales}\label{ch:g5:decimalops}

Los números decimales se suman, se restan y se multiplican casi como los enteros
--- todo el arte está en saber dónde va el punto. Este capítulo
cubre los métodos en columna y la magia de multiplicar o dividir por
$10$, $100$ y $1000$. (Multiplicar dos decimales entre sí espera al
\cref{ch:g6:decimals}.)

\section{Sumar y restar}

\begin{method}[Columnas con punto]\label{met:g5:decimalops:addsub}
\begin{enumerate}
  \item escribe los números con los \emph{\omterm{def:g4:decimals:point}{puntos decimales} alineados}
        (así, las unidades quedan bajo las unidades y las décimas bajo las décimas);
  \item completa con \omterm{def:g1:counting:zero}{ceros} la parte decimal más corta;
  \item \omterm{def:g1:addition:def}{suma} o resta como con los números enteros;
  \item baja el punto del resultado en línea recta.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:decimalops:addsub}
$27.5 + 3.86$ (completa $27.5 = 27.50$):
\[
\begin{array}{r}
2\,7.5\,0 \\
+\ \ \ 3.8\,6 \\
\hline
3\,1.3\,6
\end{array}
\qquad\qquad
\begin{array}{r}
1\,2.4\,0 \\
-\ \ \ 5.6\,5 \\
\hline
\ \ \,6.7\,5
\end{array}
\]
Para la resta ($12.4 - 5.65$): completa, pide prestado como siempre y comprueba
sumando de vuelta: $6.75 + 5.65 = 12.4$.
\end{example}

\begin{example}[Complementos mentales]\label{ex:g5:decimalops:complement}
¿Qué hay que sumar a $7.6$ para llegar a $10$? Sube: de $7.6$ a $8$ hay
$0.4$ y de $8$ a $10$ hay $2$: el complemento es $2.4$. Muy práctico para el
cambio: pagas $10$ por un artículo de $7.60$ y te devuelven $2.40$.
\end{example}

\section{Diez veces mayor, diez veces menor}

\begin{proposition}[Por y entre 10, 100, 1000]\label{prop:g5:decimalops:shift}
Multiplicar por $10$ hace que cada cifra valga diez veces más: las
cifras se desplazan un lugar a la izquierda --- lo que parece que \emph{el punto
se mueve un lugar a la derecha}. Dividir hace lo contrario:
\[
3.47 \times 10 = 34.7, \qquad
3.47 \times 100 = 347, \qquad
52.1 \div 10 = 5.21, \qquad
6 \div 100 = 0.06 .
\]
\end{proposition}
\admitted

\begin{example}[Otra vez las conversiones]\label{ex:g5:decimalops:convert}
Las unidades de medida se diferencian en potencias de diez, así que las conversiones son solo
estos desplazamientos: $2.35$ m $= 235$ cm ($\times 100$); $470$ g
$= 0.47$ kg ($\div 1000$); $1.5$ L $= 150$ cL.
\end{example}

\section{Multiplicar un decimal por un número entero}

\begin{method}[Decimal por entero]\label{met:g5:decimalops:mult}
Calcula como si no hubiera punto y después da al resultado tantas
cifras decimales como tenga el factor decimal:
\[
4.35 \times 6: \quad 435 \times 6 = 2\,610
\ \to\ 4.35 \times 6 = 26.10 = 26.1 .
\]
Estima para colocar el punto con seguridad: $4.35$ está cerca de
$4$ y $4 \times 6 = 24$ --- así que $26.1$, ni $2.61$ ni $261$.
\end{method}

\begin{example}[La suma repetida sigue funcionando]\label{ex:g5:decimalops:mult}
$0.75 \times 4 = 0.75 + 0.75 + 0.75 + 0.75 = 3$: cuatro \omterm{def:g3:division:half}{cuartos} hacen
un entero, vestidos de decimal. Multiplicar por un número entero sigue siendo
«tantas veces».
\end{example}

\begin{example}[De compras]\label{ex:g5:decimalops:shopping}
Tres cuadernos a $2.45$ cada uno y un bolígrafo a $1.20$:
\begin{enumerate}
  \item cuadernos: $2.45 \times 3 = 7.35$;
  \item total: $7.35 + 1.20 = 8.55$;
  \item cambio de $10$: $10 - 8.55 = 1.45$.
\end{enumerate}
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:1}
Calcula en columna: $45.7 + 8.93$; \quad $16.25 + 3.75$; \quad
$0.86 + 12.4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:2}
Calcula en columna y comprueba: $9.4 - 2.72$; \quad $20 - 13.45$;
\quad $6.03 - 5.9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:3}
Complementos mentales a $10$ (como en el
\cref{ex:g5:decimalops:complement}): $6.2$; \quad $3.75$; \quad
$9.99$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:4}
Calcula sin columnas:
\[
7.24 \times 10, \qquad
0.58 \times 100, \qquad
34.5 \div 10, \qquad
7 \div 100, \qquad
0.3 \times 1000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:5}
Convierte: $4.05$ m en cm; \quad $325$ cm en m; \quad $0.6$ kg en g;
\quad $85$ cL en L.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:6}
Estima primero y después calcula: $6.2 \times 4$; \quad
$3.45 \times 8$; \quad $12.5 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:7}
Una vuelta a una pista mide $0.4$ km. ¿Cuánto son $7$ vueltas? ¿Y
$25$ vueltas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:8}
Lea compra $2$ barras de pan a $1.15$ cada una y un pastel de $12.60$.
Paga con un billete de $20$. Calcula su total y su cambio.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:9}
Una botella contiene $1.5$ L. ¿Cuántos litros hay en un pack de $6$ botellas?
Una familia bebe $0.75$ L al día: ¿para cuántos días le dura un pack?
(¿Cuántas veces cabe $0.75$ en $9$?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:decimalops:10}
Milo calcula $5.6 \times 3 = 15.18$ razonando «$5 \times 3 = 15$
y $6 \times 3 = 18$». Usa una estimación para mostrar que la respuesta tiene que ser
errónea y después calcula correctamente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:decimalops:11}
Las cuatro vueltas de una corredora duraron $65.4$ s, $64.9$ s, $65.0$ s y
$66.1$ s.
\begin{enumerate}
  \item calcula el tiempo total de las cuatro vueltas;
  \item el récord de las cuatro vueltas es de $260$ s. ¿Lo ha batido? ¿Por
        cuánto?
\end{enumerate}
\end{exercise}
