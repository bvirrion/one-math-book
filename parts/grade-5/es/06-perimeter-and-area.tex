\chapter{Perímetro y área}\label{ch:g5:areas}

El año pasado contamos casillas de la cuadrícula (\cref{ch:g4:measure}); este capítulo
gana la primera \emph{fórmula} de \omterm{def:g4:measure:area}{área} --- largo por ancho para el
rectángulo --- y aprende las unidades cm$^2$ y m$^2$. Las fórmulas de los
triángulos y de otras figuras maduran en los \cref{ch:g6:measure} y
\cref{ch:g7:areas}.

\section{Dos medidas distintas}

\begin{example}[Borde frente a superficie]\label{ex:g5:areas:contrast}
Un jardinero necesita el \emph{\omterm{def:g4:measure:perimeter}{perímetro}} para comprar la valla y el \emph{\omterm{def:g4:measure:area}{área}}
para comprar la semilla de césped. Los dos no van de la mano: estirar
un rectángulo hasta hacerlo más largo y estrecho puede conservar su \omterm{def:g4:measure:area}{área} mientras su \omterm{def:g4:measure:perimeter}{perímetro}
crece --- compara un rectángulo de $6 \times 2$ con uno de $4 \times 3$
(misma \omterm{def:g4:measure:area}{área} $12$ y \omterm{def:g4:measure:perimeter}{perímetros} $16$ y $14$).
\end{example}

\section{La fórmula del rectángulo}

\begin{proposition}[Área de un rectángulo]\label{prop:g5:areas:rectangle}
Un rectángulo de largo $L$ y ancho $a$ (en la misma unidad) tiene \omterm{def:g4:measure:area}{área}
\[
A = L \times w ,
\]
en unidades cuadradas: cm$^2$ (cuadrados de lado $1$ cm), m$^2$ (cuadrados de
lado $1$ m), \dots
\end{proposition}

\begin{proof}[Por qué, contando]
Cubre el rectángulo con cuadrados unidad: $a$ filas de $L$ cuadrados cada una,
o sea, $L \times a$ cuadrados en total --- exactamente el rectángulo
de la multiplicación del \cref{ch:g3:mult}.
\end{proof}

\begin{omfigure}
\begin{tikzpicture}[scale=0.62]
  \draw[gray!50, thin] (0,0) grid (6,3);
  \draw[very thick, omDef] (0,0) rectangle (6,3);
  \fill[omThm!30] (0,2) rectangle (1,3);
  \draw[omThm, thick] (0,2) rectangle (1,3);
  \node[right, font=\small, omThm] at (1.15,2.5) {$1$ cm$^2$};
  \node[below, font=\small] at (3,-0.15) {$6$ cm};
  \node[left, font=\small] at (0,1.5) {$3$ cm};
  \node[font=\small] at (3.9,1.2) {$6 \times 3 = 18$ cm$^2$};
\end{tikzpicture}

{\small Tres filas de seis cuadrados de un centímetro: la fórmula
$L \times a$ es ese recuento, escrito de una vez por todas.}
\end{omfigure}

\begin{example}\label{ex:g5:areas:rectangle}
Una alfombra mide $2.5$ m por $2$ m: \omterm{def:g4:measure:area}{área} $2.5 \times 2 = 5$ m$^2$.
Un sello mide $3$ cm por $2.4$ cm: \omterm{def:g4:measure:area}{área} $3 \times 2.4 = 7.2$
cm$^2$. La misma fórmula con cualquier unidad --- siempre que los dos lados usen la
\emph{misma}.
\end{example}

\begin{example}[Las unidades cuadradas se convierten de cien en cien]\label{ex:g5:areas:units}
$1$ m $= 100$ cm, pero $1$ m$^2 = 100 \times 100 = 10\,000$ cm$^2$:
un metro cuadrado es una cuadrícula de $100$ por $100$ centímetros cuadrados. Las unidades
de \omterm{def:g4:measure:area}{área} saltan de cien en cien, no de diez en diez.
\end{example}

\section{Figuras compuestas}

\begin{method}[Cortar, sumar, restar]\label{met:g5:areas:composite}
Para una figura hecha de rectángulos (una L, una T, un marco):
\begin{enumerate}
  \item córtala en rectángulos o complétala hasta un rectángulo grande;
  \item calcula el \omterm{def:g4:measure:area}{área} de cada rectángulo con la fórmula;
  \item \omterm{def:g1:addition:def}{suma} los trozos --- o resta el hueco del rectángulo
        grande;
  \item comprueba con un recuento aproximado de casillas.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:areas:composite}
Una habitación en forma de L: un rectángulo de $6$ m $\times$ $4$ m al que le falta una
esquina de $2$ m $\times$ $2$ m.
\[
\text{Área} = 6 \times 4 - 2 \times 2 = 24 - 4 = 20 \text{ m}^2 .
\]
O corta la L en una franja de $6 \times 2$ y otra de $4 \times 2$:
$12 + 8 = 20$ m$^2$ --- dos caminos, una respuesta.
\end{example}

\begin{example}[La mitad de un rectángulo]\label{ex:g5:areas:halfrect}
Cortar un rectángulo por una diagonal da dos triángulos de la misma
\omterm{def:g4:measure:area}{área}: cada uno es la \emph{\omterm{def:g3:division:half}{mitad}} del rectángulo. Un triángulo rectángulo de catetos
$6$ cm y $4$ cm tiene, por tanto, \omterm{def:g4:measure:area}{área} $\frac{6 \times 4}{2} = 12$
cm$^2$ --- una imagen que conviene recordar para el
\cref{ch:g6:measure}, donde se convierte en fórmula.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:areas:1}
Calcula el \omterm{def:g4:measure:area}{área} y el \omterm{def:g4:measure:perimeter}{perímetro} de un rectángulo de $8$ cm $\times$
$5$ cm. ¿Qué respuesta va en cm y cuál en cm$^2$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:2}
Calcula las \omterm{def:g4:measure:area}{áreas}: un cuadrado de lado $9$ cm; un rectángulo de
$12$ m $\times$ $7$ m; un rectángulo de $4.5$ cm $\times$ $6$ cm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:3}
Un rectángulo tiene \omterm{def:g4:measure:area}{área} $63$ cm$^2$ y largo $9$ cm. Halla su ancho
y después su \omterm{def:g4:measure:perimeter}{perímetro}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:4}
Dibuja en papel cuadriculado dos rectángulos distintos de \omterm{def:g4:measure:area}{área} $24$ casillas
y calcula los dos \omterm{def:g4:measure:perimeter}{perímetros}. Misma \omterm{def:g4:measure:area}{área}, ¿mismo \omterm{def:g4:measure:perimeter}{perímetro}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:5}
Convierte: $3$ m$^2$ en cm$^2$; \quad $50\,000$ cm$^2$ en m$^2$.
(Recuerda el \cref{ex:g5:areas:units}: ¡de cien en cien!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:6}
Una figura en forma de T está hecha de una barra horizontal de $8 \times 2$ encima de
una barra vertical de $2 \times 5$ (en cm). Calcula su \omterm{def:g4:measure:area}{área} sumando dos
rectángulos.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:7}
Un marco de fotos: un rectángulo de $30$ cm $\times$ $20$ cm con una
ventana rectangular de $24$ cm $\times$ $14$ cm recortada. ¿Qué \omterm{def:g4:measure:area}{área} de
madera ocupa el marco?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:8}
Calcula el \omterm{def:g4:measure:area}{área} de un triángulo rectángulo de catetos $8$ cm y $6$ cm
(la \omterm{def:g3:division:half}{mitad} de un rectángulo, \cref{ex:g5:areas:halfrect}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:9}
Un huerto rectangular mide $7$ m por $4$ m. La semilla cuesta
$2$ por metro cuadrado y la valla, $3$ por metro. Calcula el coste de
la semilla y después el de la valla.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:areas:10}
El suelo de un pasillo mide $12$ m de largo y $2$ m de ancho, y hay que cubrirlo
con baldosas cuadradas de $50$ cm de lado. ¿Cuántas baldosas hacen falta?
(Convierte primero o cuenta las baldosas en cada dirección.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:areas:11}
Duplica los lados de un rectángulo de $4$ cm $\times$ $3$ cm. ¿Qué le pasa
a su \omterm{def:g4:measure:perimeter}{perímetro}? ¿Y a su \omterm{def:g4:measure:area}{área}? (Calcula las dos cosas antes y después ---
¡las dos respuestas son distintas!)
\end{exercise}
