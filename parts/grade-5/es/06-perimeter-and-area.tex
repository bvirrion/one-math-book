\chapter{Perímetro y área}\label{ch:g5:areas}

Cuadrículas contadas de grado 4 (\cref{ch:g4:measure}); este capítulo
gana el primero \omterm{def:g4:measure:area}{área} \emph{fórmula} --- largo por ancho para el
rectángulo --- y aprende las unidades cm $^2$ y m $^2$. Las fórmulas para
triángulos y otras formas maduran en \cref{ch:g6:measure} y
\cref{ch:g7:areas}.

\section{Dos medidas diferentes}

\begin{example}[Frontera versus superficie]\label{ex:g5:areas:contrast}
Un jardinero necesita \emph{\omterm{def:g4:measure:perimeter}{perímetro}} comprar la valla y \emph{\omterm{def:g4:measure:area}{área}}
para comprar la semilla de pasto. Los dos no se suceden: estiramientos
un rectángulo más delgado y más largo puede mantener su \omterm{def:g4:measure:area}{área} mientras que su \omterm{def:g4:measure:perimeter}{perímetro}
crece --- comparar un $6 \times 2$ y un $4 \times 3$ rectángulo
(lo mismo \omterm{def:g4:measure:area}{área} $12$, \omterm{def:g4:measure:perimeter}{perímetros} $16$ y $14$).
\end{example}

\section{La fórmula del rectángulo}

\begin{proposition}[Área de un rectángulo]\label{prop:g5:areas:rectangle}
Un rectángulo de longitud $ L $ y ancho $ w $ (en la misma unidad) tiene \omterm{def:g4:measure:area}{área}
\[
A = L \times w ,
\]
en unidades cuadradas: cm$^2$ (\omterm{def:g3:shapes:shapes}{cuadrados} de lado $1$ centímetros), metros $^2$ (\omterm{def:g3:shapes:shapes}{cuadrados} de lado $1$ metro), \dots
\end{proposition}

\begin{proof}[Por qué, contando]
Cubre el rectángulo con \omterm{def:g3:shapes:shapes}{cuadrados} unitarios: $ w $ filas de $ L $ \omterm{def:g3:shapes:shapes}{cuadrados} cada uno,
entonces $ L \times w $ \omterm{def:g3:shapes:shapes}{cuadrados} en total --- exactamente la multiplicación
rectángulo de \cref{ch:g3:mult}.
\end{proof}

\begin{tikzpicture}[scale=0.62]
  \draw[gray!50, thin] (0,0) grid (6,3);
  \draw[very thick, omDef] (0,0) rectangle (6,3);
  \fill[omThm!30] (0,2) rectangle (1,3);
  \draw[omThm, thick] (0,2) rectangle (1,3);
  \node[right, font=\small, omThm] at (1.15,2.5) {$1$ cm $^2$};
  \node[below, font=\small] at (3,-0.15) {$6$ cm};
  \node[left, font=\small] at (0,1.5) {$3$ cm};
  \node[font=\small] at (3.9,1.2) {$6 \times 3 = 18$ cm $^2$};
\end{tikzpicture}

{\small Tres filas de \omterm{def:g3:shapes:shapes}{cuadrados} de seis centímetros: la fórmula
$ L \times w $ es el conteo, escrito de una vez por todas.}
\end{omfigure}

\begin{example}\label{ex:g5:areas:rectangle}
Una alfombra mide $2.5$ m por $2$ metro: \omterm{def:g4:measure:area}{área} $2.5 \times 2 = 5$ metro $^2$.
Un sello mide $3$ centímetros por $2.4$ centímetro: \omterm{def:g4:measure:area}{área} $3 \times 2.4 = 7.2$
centímetro $^2$. Misma fórmula, cualquier unidad --- siempre y cuando ambos lados usen la
\emph{mismo} uno.
\end{example}

\begin{example}[Conversión de unidades cuadradas por 100]\label{ex:g5:areas:units}
$1$ metro $= 100$ centímetros, pero $1$ metro $^2 = 100 \times 100 = 10\,000$ centímetro $^2$:
un metro cuadrado es un $100$-por-$100$ cuadrícula de centímetros \omterm{def:g3:shapes:shapes}{cuadrados}. \omterm{def:g4:measure:area}{área}
las unidades saltan en centenas, no en decenas.
\end{example}

\section{Figuras compuestas}

\begin{method}[Cortar, sumar, restar]\label{met:g5:areas:composite}
Para una figura hecha de rectángulos (una L, una T, un marco):
\begin{enumerate}
  \item córtelo en rectángulos o complételo en un rectángulo grande;
  \item calcular cada rectangular \omterm{def:g4:measure:area}{área} con la fórmula;
  \item \omterm{def:g1:addition:def}{suma} las piezas --- o resta el agujero del grande
        rectángulo;
  \item compruébelo con un recuento aproximado de los \omterm{def:g3:shapes:shapes}{cuadrados} de la cuadrícula.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:areas:composite}
Una habitación en forma de L: una $6$ metro $\times $ $4$ m rectángulo con un
$2$ metro $\times $ $2$ Falta m esquina.
\[
\text{Area} = 6 \times 4 - 2 \times 2 = 24 - 4 = 20 \text{ m}^2 .
\]
O cortar la L en una $6 \times 2$ tira y un $4 \times 2$ banda:
$12 + 8 = 20$ metro $^2$ --- dos caminos, una respuesta.
\end{example}

\begin{example}[Medio rectángulo]\label{ex:g5:areas:halfrect}
Al cortar un rectángulo a lo largo de una diagonal se obtienen dos triángulos iguales.
\omterm{def:g4:measure:area}{área}: cada uno es \emph{\omterm{def:g3:division:half}{medio}} el rectángulo. Un triángulo rectángulo con catetos.
$6$ centímetros y $4$ cm por lo tanto tiene \omterm{def:g4:measure:area}{área} $\frac{6 \times 4}{2} = 12$
centímetro $^2$ --- una imagen que vale la pena recordar
\cref{ch:g6:measure}, donde se convierte en una fórmula.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g5:areas:1}
Calcular el \omterm{def:g4:measure:area}{área} y el \omterm{def:g4:measure:perimeter}{perímetro} de un rectángulo $8$ centímetro $\times $
$5$ centímetro. ¿Cuál respuesta está en cm, cuál en cm?$^2$?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:areas:2}
Calcular el \omterm{def:g4:measure:area}{áreas}: un cuadrado de lado $9$ centímetro; un rectángulo
$12$ metro $\times $ $7$ metro; un rectángulo $4.5$ centímetro $\times $ $6$ centímetro.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:areas:3}
Un rectángulo tiene \omterm{def:g4:measure:area}{área} $63$ centímetro $^2$ y longitud $9$ centímetro. Encuentra su ancho,
entonces es \omterm{def:g4:measure:perimeter}{perímetro}.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:areas:4}
Dibuja dos rectángulos diferentes con \omterm{def:g4:measure:area}{área} $24$ \omterm{def:g3:shapes:shapes}{cuadrados} en papel cuadriculado,
y calcular ambos \omterm{def:g4:measure:perimeter}{perímetros}. Mismo \omterm{def:g4:measure:area}{área} --- mismo \omterm{def:g4:measure:perimeter}{perímetro}?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:areas:5}
Convertir: $3$ metro $^2$ en centímetros $^2$; \quad $50\,000$ centímetro $^2$ en m $^2$.
(Recordar \cref{ex:g5:areas:units}: ¡por cientos!)
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:areas:6}
Una figura en forma de T está hecha de un $8 \times 2$ barra horizontal encima de
un $2 \times 5$ barra vertical (en cm). Calcular su \omterm{def:g4:measure:area}{área} añadiendo dos
rectángulos.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:areas:7}
Un marco de imagen: un $30$ centímetro $\times $ $20$ rectángulo de cm con un
$24$ centímetro $\times $ $14$ Ventana rectangular de cm recortada. Qué \omterm{def:g4:measure:area}{área} de
¿Qué madera utiliza el marco?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:areas:8}
Calcular el \omterm{def:g4:measure:area}{área} de un triángulo rectángulo con catetos $8$ centímetros y $6$ centímetros (\omterm{def:g3:division:half}{medio} un rectángulo, \cref{ex:g5:areas:halfrect}).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:areas:9}
Un huerto rectangular mide $7$ m por $4$ metro. Costos de semillas
$2$ por metro cuadrado y vallado $3$ por metro. Calcular el costo de
la semilla, luego el costo de la cerca.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:areas:10}
El suelo de un pasillo es $12$ m de largo y $2$ m de ancho y debe cubrirse
con azulejos \omterm{def:g3:shapes:shapes}{cuadrados} de lado $50$ centímetro. ¿Cuántas fichas se necesitan?
(Convierta primero o cuente las fichas en cada dirección).
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:areas:11}
Duplicar los lados de un $4$ centímetro $\times $ $3$ cm rectángulo. que pasa
a su \omterm{def:g4:measure:perimeter}{perímetro}? a su \omterm{def:g4:measure:area}{área}? (Calcule tanto antes como después ---
¡las dos respuestas difieren!)
\end{exercise}
