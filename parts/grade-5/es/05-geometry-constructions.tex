\chapter{Construcciones geométricas}\label{ch:g5:geometry}

Regla, escuadra, compás: con estas tres herramientas y un poco de
método, cualquier figura descrita con palabras se puede construir exactamente. esto
construcciones de capítulos \omterm{def:g4:geometry:def}{perpendiculares}, paralelas, triángulos y el
especial \omterm{def:g4:geometry:polygon}{cuadriláteros} --- artesanía primero, teoría después
(\cref{ch:g6:lines,ch:g6:shapes}).

\section{Perpendiculares y paralelas que pasan por un punto.}

\begin{method}[Las dos construcciones básicas.]\label{met:g5:geometry:basic}
\emph{\omterm{def:g4:geometry:def}{perpendiculares} a $d $ a través de $ A $}: desliza la escuadra a lo largo
$ d $ hasta que su otro borde en \omterm{def:g3:shapes:rightangle}{ángulo recto} alcance $ A $; dibujar; marcar el
\omterm{def:g3:shapes:rightangle}{ángulo recto}.

\emph{paralelas a $ d $ a través de $ A $}: presione una regla contra el conjunto
cuadrado como un riel colocado a lo largo $ d $; deslice la escuadra hacia arriba $ A $;
dibujar. (Compruebe: el espacio entre las dos líneas es el mismo
en todas partes.)
\end{method}

\begin{example}[Encadenamiento de construcciones]\label{ex:g5:geometry:chain}
Muchas figuras son cadenas de estos dos movimientos. un rectángulo $ ABCD $ con
$ AB = 5$ centímetros y $ a.\,C. = 3$ cm: dibujar $[AB]$; \omterm{def:g4:geometry:def}{perpendiculares} en $ A $;
\omterm{def:g4:geometry:def}{perpendiculares} en $ B $; marca $ D $ y $ C $ en $3$ cm del mismo lado;
unirse. Cada paso es una construcción básica.
\end{example}

\section{Triángulos con el compás}

\begin{method}[Triángulo de tres lados]\label{met:g5:geometry:triangle}
Para construir un triángulo con lados. $6$ centímetro, $5$ centímetro, $4$ centímetro:
\begin{enumerate}
  \item dibuja el lado más largo, $[AB]$ con $ AB = 6$ centímetro;
  \item arco de centro $ A $ y \omterm{def:g3:shapes:shapes}{radio} $5$ centímetro;
  \item arco de centro $ B $ y \omterm{def:g3:shapes:shapes}{radio} $4$ centímetro;
  \item su cruce es el tercero vértice $ C $; unirse.
\end{enumerate}
La brújula garantiza las longitudes: cada punto del primer arco es
exactamente en $5$ centímetros de $ A $.
\end{method}

\begin{example}[Isósceles y equiláteros gratis]\label{ex:g5:geometry:special}
Misma apertura de compás para ambos arcos. $\Rightarrow $ triángulo isósceles.
Apertura igual a la base. $\Rightarrow $ equilátero. la brújula
no mide: \emph{copias} longitudes, lo cual es aún mejor.
\end{example}

\section{Los cuadriláteros especiales, por construcción.}

\begin{method}[Construyendo la familia]\label{met:g5:geometry:quadrilaterals}
\begin{enumerate}
  \item \emph{Rectángulo}: dos \omterm{def:g3:numbers:evenodd}{pares} de paralelas cruzando en correcto
        ángulos --- o la cadena de \cref{ex:g5:geometry:chain};
  \item \emph{cuadrado}: un rectángulo cuyos cuatro lados usan lo mismo
        longitud --- llévalo con la brújula;
  \item \emph{rombo}: cuatro lados iguales --- dibuja un lado, luego dos
        arcos de compás desde sus extremos con la misma abertura fijan el
        otros dos \omterm{def:g4:geometry:polygon}{vértices} (un rombo son dos triángulos isósceles
        pegado a lo largo de una diagonal).
\end{enumerate}
Después de construir, siempre \emph{controlar} con las herramientas: colocar escuadra
las esquinas, compás en los lados.
\end{method}

\begin{example}[Reconocer por una descripción]\label{ex:g5:geometry:recognize}
``Un \omterm{def:g4:geometry:polygon}{cuadrilátero} con cuatro $4$ cm de lados y uno \omterm{def:g3:shapes:rightangle}{ángulo recto}'' ---
construirlo: los cuatro lados iguales fuerzan una forma de rombo, y presionando
una esquina a una \omterm{def:g3:shapes:rightangle}{ángulo recto} endereza todos los demás: el resultado
es un cuadrado. Algunas descripciones describen en secreto algo más especial.
forma de la que anuncian. (¿Por qué el \omterm{def:g3:shapes:rightangle}{ángulo recto} se extiende a los cuatro
esquinas se explica en \cref{ch:g7:central}.)
\end{example}

\section{Círculos en construcciones.}

\begin{example}[El círculo como herramienta]\label{ex:g5:geometry:circle}
``Encontrar todos los puntos en $3$ centímetros de $ A $'': el círculo del centro $ A $,
\omterm{def:g3:shapes:shapes}{radio} $3$ centímetro. ``Encontrar todos los puntos en $3$ centímetros de $ A $ \emph{y}
$4$ centímetros de $ B $'': dibuja ambos círculos; sus cruces (dos, uno o
\omterm{def:g1:counting:zero}{cero} puntos) son las respuestas. Mapas del tesoro y triángulo.
Las construcciones utilizan exactamente esta idea.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g5:geometry:1}
dibuja una linea $ d $ y un punto $ A $ aproximadamente $3$ cm de él. construir
el \omterm{def:g4:geometry:def}{perpendiculares} a $ d $ a través de $ A $, entonces el paralelas a $ d $
a través de $ A $. Marque el correcto
        ángulos.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:geometry:2}
construir un rectángulo $ ABCD $ con $ AB = 7$ centímetros y $ a.\,C. = 4$ centímetros,
siguiente \cref{ex:g5:geometry:chain}. Comprueba el \omterm{def:g3:division:half}{cuarto} ángulo con
La escuadra: si la construcción es exacta, es correcta
automáticamente.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:geometry:3}
construir un \omterm{def:g3:shapes:shapes}{triángulo} con lados $7$ centímetro, $6$ centímetros y $3$ centímetros (\cref{met:g5:geometry:triangle}).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:geometry:4}
Construye un triángulo isósceles con base. $5$ cm y lados iguales de
$7$ centímetro; entonces un triángulo equilátero de lado $6$ centímetro.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:geometry:5}
construir un cuadrado de lado $5$ centímetro. ¿Qué herramientas garantizan la correcto
        ángulos? ¿Cuáles garantizan los lados iguales?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:geometry:6}
Construir un rombo de lado. $4$ cm que no sea un cuadrado (elige tu
propia apertura para la primera curva). Verifique los cuatro lados con el
brújula.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:geometry:7}
dibujar dos puntos $ A $ y $ B $ con $ AB = 5$ centímetro. Construir todos los puntos
que están en $4$ centímetros de $ A $ y en $3$ centímetros de $ B $. cuantos son
ahí?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g5:geometry:8}
Escriba un programa de construcción (pasos numerados, puntos con nombre) para un
rectángulo $6$ centímetro $\times $ $2$ cm rematado por un triángulo isósceles
con lados iguales $4$ cm (una forma de lápiz). Haz que un compañero ejecute
eso.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:geometry:9}
Intenta construir un triángulo con lados. $8$ centímetro, $3$ centímetros y $4$ centímetro.
¿Qué pasa con los arcos? Explica en una frase por qué esto
El triángulo no puede existir.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:geometry:10}
Construye un rombo cuyas diagonales midan $6$ centímetros y $4$ centímetros,
sabiendo que las diagonales de un rombo se cruzan en sus centros en un
\omterm{def:g3:shapes:rightangle}{ángulo recto}: dibuja las diagonales \emph{primero}, luego únete a sus cuatro
termina.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g5:geometry:11}
Una cabra está atada a una estaca. $ A $ por un $4$ m de cuerda, en un terreno llano con
una valla recta pasando $3$ soy de $ A$. Dibujar un plano (1 cm por
metro): ¿a qué región puede llegar la cabra? ¿Dónde limita la valla?
eso?
\end{exercise}
