\chapter{Construcciones geométricas}\label{ch:g5:geometry}

Regla, escuadra y compás: con estas tres herramientas y un poco de
método, cualquier figura descrita con palabras se puede construir con exactitud. Este
capítulo construye \omterm{def:g4:geometry:def}{perpendiculares}, \omterm{def:g4:geometry:def}{paralelas}, triángulos y los
\omterm{def:g4:geometry:polygon}{cuadriláteros} especiales --- primero el oficio y después la teoría
(\cref{ch:g6:lines,ch:g6:shapes}).

\section{Perpendiculares y paralelas por un punto}

\begin{method}[Las dos construcciones básicas]\label{met:g5:geometry:basic}
\emph{Perpendicular a $d$ por $A$}: desliza la escuadra a lo largo de
$d$ hasta que su otro borde del \omterm{def:g3:shapes:rightangle}{ángulo recto} alcance $A$; traza y marca el
\omterm{def:g3:shapes:rightangle}{ángulo recto}.

\emph{Paralela a $d$ por $A$}: aprieta una regla contra la escuadra
a modo de raíl, apoyada sobre $d$; desliza la escuadra hasta $A$ y
traza. (Comprobación: la separación entre las dos rectas es la misma
en todas partes.)
\end{method}

\begin{example}[Encadenar construcciones]\label{ex:g5:geometry:chain}
Muchas figuras son cadenas de estos dos movimientos. Un rectángulo $ABCD$ con
$AB = 5$ cm y $BC = 3$ cm: traza $[AB]$; perpendicular en $A$;
perpendicular en $B$; marca $D$ y $C$ a $3$ cm del mismo lado;
une. Cada paso es una construcción básica.
\end{example}

\section{Triángulos con el compás}

\begin{method}[Triángulo a partir de tres lados]\label{met:g5:geometry:triangle}
Para construir un triángulo de lados $6$ cm, $5$ cm y $4$ cm:
\begin{enumerate}
  \item traza el lado más largo, $[AB]$ con $AB = 6$ cm;
  \item arco de centro $A$ y \omterm{def:g3:shapes:shapes}{radio} $5$ cm;
  \item arco de centro $B$ y \omterm{def:g3:shapes:shapes}{radio} $4$ cm;
  \item su cruce es el tercer \omterm{def:g4:geometry:polygon}{vértice} $C$; une.
\end{enumerate}
El compás garantiza las longitudes: todos los puntos del primer arco están
exactamente a $5$ cm de $A$.
\end{method}

\begin{example}[Isósceles y equilátero, de regalo]\label{ex:g5:geometry:special}
Con la misma abertura del compás para los dos arcos $\Rightarrow$ triángulo isósceles.
Con la abertura igual a la base $\Rightarrow$ equilátero. El compás
no mide: \emph{copia} longitudes, que es todavía mejor.
\end{example}

\section{Los cuadriláteros especiales, por construcción}

\begin{method}[Construir la familia]\label{met:g5:geometry:quadrilaterals}
\begin{enumerate}
  \item \emph{Rectángulo}: dos pares de \omterm{def:g4:geometry:def}{paralelas} que se cortan en \omterm{def:g3:shapes:rightangle}{ángulos
        rectos} --- o la cadena del \cref{ex:g5:geometry:chain};
  \item \emph{cuadrado}: un rectángulo cuyos cuatro lados usan la misma
        longitud --- llévala con el compás;
  \item \emph{rombo}: cuatro lados iguales --- traza un lado y, con dos
        arcos de compás desde sus extremos con la misma abertura, quedan fijados
        los otros dos \omterm{def:g4:geometry:polygon}{vértices} (un rombo son dos triángulos isósceles
        pegados por una diagonal).
\end{enumerate}
Después de construir, \emph{comprueba} siempre con las herramientas: la escuadra en
las esquinas y el compás en los lados.
\end{method}

\begin{example}[Reconocer a partir de una descripción]\label{ex:g5:geometry:recognize}
«Un \omterm{def:g4:geometry:polygon}{cuadrilátero} con cuatro lados de $4$ cm y un \omterm{def:g3:shapes:rightangle}{ángulo recto}» ---
constrúyelo: los cuatro lados iguales obligan a que sea un rombo, y forzar
una esquina a \omterm{def:g3:shapes:rightangle}{ángulo recto} endereza todas las demás: el resultado
es un cuadrado. Algunas descripciones describen en secreto una figura más especial
de lo que anuncian. (Por qué el \omterm{def:g3:shapes:rightangle}{ángulo recto} se contagia a las cuatro
esquinas se explica en el \cref{ch:g7:central}.)
\end{example}

\section{Circunferencias en las construcciones}

\begin{example}[La circunferencia como herramienta]\label{ex:g5:geometry:circle}
«Halla todos los puntos que están a $3$ cm de $A$»: la circunferencia de centro $A$ y
\omterm{def:g3:shapes:shapes}{radio} $3$ cm. «Halla todos los puntos que están a $3$ cm de $A$ \emph{y} a
$4$ cm de $B$»: traza las dos circunferencias; sus cruces (dos, uno o
\omterm{def:g1:counting:zero}{cero} puntos) son las respuestas. Los mapas del tesoro y las construcciones de
triángulos usan exactamente esta idea.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:geometry:1}
Traza una recta $d$ y un punto $A$ a unos $3$ cm de ella. Construye
la perpendicular a $d$ por $A$ y después la paralela a $d$
por $A$. Marca los \omterm{def:g3:shapes:rightangle}{ángulos rectos}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:2}
Construye un rectángulo $ABCD$ con $AB = 7$ cm y $BC = 4$ cm,
siguiendo el \cref{ex:g5:geometry:chain}. Comprueba el \omterm{def:g3:division:half}{cuarto} ángulo con
la escuadra: si la construcción es exacta, sale recto
automáticamente.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:3}
Construye un triángulo de lados $7$ cm, $6$ cm y $3$ cm
(\cref{met:g5:geometry:triangle}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:4}
Construye un triángulo isósceles de base $5$ cm y lados iguales de
$7$ cm; después un triángulo equilátero de lado $6$ cm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:5}
Construye un cuadrado de lado $5$ cm. ¿Qué herramientas garantizan los \omterm{def:g3:shapes:rightangle}{ángulos
rectos}? ¿Cuáles garantizan los lados iguales?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:6}
Construye un rombo de lado $4$ cm que no sea un cuadrado (elige tú
la abertura de la primera esquina). Comprueba los cuatro lados con el
compás.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:7}
Traza dos puntos $A$ y $B$ con $AB = 5$ cm. Construye todos los puntos
que están a $4$ cm de $A$ y a $3$ cm de $B$. ¿Cuántos
hay?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:8}
Escribe un programa de construcción (pasos numerados, puntos con nombre) para un
rectángulo de $6$ cm $\times$ $2$ cm rematado por un triángulo isósceles
de lados iguales de $4$ cm (la forma de un lápiz). Haz que un compañero lo
ejecute.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:geometry:9}
Intenta construir un triángulo de lados $8$ cm, $3$ cm y $4$ cm.
¿Qué pasa con los arcos? Explica en una frase por qué ese
triángulo no puede existir.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:geometry:10}
Construye un rombo cuyas diagonales midan $6$ cm y $4$ cm,
sabiendo que las diagonales de un rombo se cortan en sus puntos medios formando un
\omterm{def:g3:shapes:rightangle}{ángulo recto}: traza \emph{primero} las diagonales y une después sus cuatro
extremos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:geometry:11}
Una cabra está atada a una estaca $A$ con una cuerda de $4$ m, en un prado llano con
una valla recta que pasa a $3$ m de $A$. Dibuja un plano (1 cm por
metro): ¿qué región puede alcanzar la cabra? ¿Dónde la limita la valla?
\end{exercise}
