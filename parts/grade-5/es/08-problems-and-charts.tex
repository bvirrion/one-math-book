\chapter{Problemas y gráficos}\label{ch:g5:data}

El último capítulo del año pone todo a trabajar: problemas de varios
pasos con dinero, cantidades y duraciones, y los gráficos que
presentan la información de un vistazo. La idea estrella es «por unidad» ---
hallar el precio de un solo artículo ---, que crecerá hasta la
proporcionalidad en el \cref{ch:g6:propdata}.

\section{Problemas de varios pasos}

\begin{method}[Domar un problema largo]\label{met:g5:data:steps}
\begin{enumerate}
  \item lee hasta el final y di cuál es la pregunta final;
  \item planifica hacia atrás: para responderla, ¿qué necesito antes?
  \item resuelve los pasos en orden, escribiendo una operación y una
        frase corta por paso;
  \item comprueba la respuesta final con el sentido común y con una
        estimación.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:data:multistep}
«Un colegio encarga $4$ cajas de $25$ libros a $6$ cada uno y paga además
$180$ por el envío. ¿Cuál es el coste total?»
\begin{enumerate}
  \item libros: $4 \times 25 = 100$ libros;
  \item coste de los libros: $100 \times 6 = 600$;
  \item total: $600 + 180 = 780$.
\end{enumerate}
Comprobación estimada: unos cien libros a $6$ son unos $600$, más
unos $200$: unos $800$ --- la respuesta $780$ es verosímil.
\end{example}

\begin{example}[Por unidad]\label{ex:g5:data:perone}
«$5$ tazas iguales cuestan $12.50$; ¿cuánto cuestan $8$ tazas?»
Halla primero el precio de \emph{una} taza:
\[
12.50 \div 5 = 2.50,
\qquad\text{y después}\qquad
8 \times 2.50 = 20 .
\]
Pasando por la unidad, todo queda al alcance --- este esquema de dos pasos
resuelve toda una familia de problemas (recetas, velocidades, precios) y es
la semilla del \cref{ch:g6:propdata}.
\end{example}

\begin{example}[Cambio y presupuesto]\label{ex:g5:data:budget}
Zoe tiene $30$. Compra un libro de $12.40$ y dos bolígrafos de $2.30$
cada uno. ¿Le llega también para un juego de mesa de $13$?
\begin{enumerate}
  \item bolígrafos: $2 \times 2.30 = 4.60$;
  \item gastado hasta ahora: $12.40 + 4.60 = 17$;
  \item le queda: $30 - 17 = 13$ --- justo lo necesario para el juego,
        sin que sobre nada.
\end{enumerate}
\end{example}

\section{Leer gráficos}

\begin{method}[Leer cualquier gráfico]\label{met:g5:data:read}
\begin{enumerate}
  \item lee el título: ¿qué se está contando?
  \item lee los ejes o la clave: ¿qué representa una división de la escala, una barra
        o un símbolo?
  \item solo entonces responde a las preguntas --- señalar el gráfico con una
        regla ayuda a leer las alturas con exactitud.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:data:chart}
Precipitaciones de un pueblo, mes a mes:

\begin{omfigure}
\begin{tikzpicture}
\begin{axis}[
  width=8.8cm, height=5.0cm,
  ybar, bar width=13pt,
  symbolic x coords={Ene,Feb,Mar,Abr,May,Jun},
  xtick=data,
  ymin=0, ymax=90,
  ytick={20,40,60,80},
  ylabel={lluvia (mm)},
  tick label style={font=\small},
  label style={font=\small}]
  \addplot[fill=omDef!30, draw=omDef] coordinates
    {(Ene,60) (Feb,45) (Mar,55) (Abr,70) (May,80) (Jun,30)};
\end{axis}
\end{tikzpicture}

{\small Un diagrama de barras de las precipitaciones. Una división $= 20$ mm.}
\end{omfigure}

Lecturas: el mes más lluvioso es mayo ($80$ mm); ¿llovió en junio la \omterm{def:g3:division:half}{mitad} que
en abril ($30$ frente a $70$)? No --- la \omterm{def:g3:division:half}{mitad} de $70$ es $35$, así que
\emph{no llega} a la \omterm{def:g3:division:half}{mitad}: ¡leer con precisión importa! Total de los
seis meses: $60 + 45 + 55 + 70 + 80 + 30 = 340$ mm.
\end{example}

\begin{example}[Una magnitud que cambia]\label{ex:g5:data:line}
La altura de una planta, medida cada semana:

\begin{omfigure}
\begin{tikzpicture}
\begin{axis}[omaxis,
  width=8.4cm, height=5.0cm,
  xmin=0, xmax=6.5, ymin=0, ymax=24,
  xlabel={semana}, ylabel={altura (cm)},
  xtick={1,2,3,4,5,6}, ytick={5,10,15,20}]
  \addplot[omDef, thick, mark=*, mark size=1.6pt] coordinates
    {(1,3) (2,6) (3,10) (4,15) (5,19) (6,21)};
\end{axis}
\end{tikzpicture}

{\small Al unir los puntos de medida se ve el \emph{crecimiento}: cuanto más
inclinado es el segmento, más deprisa creció la planta esa semana.}
\end{omfigure}

La planta creció más deprisa entre las semanas $3$ y $4$ ($+5$ cm) y
se frenó al final ($+2$ cm en la semana $6$).
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g5:data:1}
Resuelve por pasos: «Un club alquila un autobús por $240$ y compra $32$ entradas
de museo a $7$ cada una. ¿Cuál es el coste total de la excursión?»
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:2}
Resuelve con «por unidad»: «$3$ kg de manzanas cuestan $7.50$. ¿Cuánto cuestan
$5$ kg?»
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:3}
Resuelve: «Seis amigos comparten a partes iguales los $87$ que cuesta un picnic. ¿Cuánto
paga cada uno?»
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:4}
Samuel tiene $25$. Compra una camiseta de $13.90$ y una gorra de $8.50$. ¿Cuánto
dinero le queda?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:5}
Con el gráfico de precipitaciones del \cref{ex:g5:data:chart}: ¿qué meses
superaron los $50$ mm? ¿Cuánta lluvia más cayó en mayo que en
junio?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:6}
Con el gráfico de la planta del \cref{ex:g5:data:line}: ¿cuánto medía la
planta en la semana $2$? ¿Entre qué semanas creció exactamente
$4$ cm?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:7}
Dibuja un diagrama de barras con los goles marcados por un equipo:
septiembre $4$, octubre $7$, noviembre $3$, diciembre $6$. Elige la
escala y da el total.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:8}
Un pack de $8$ yogures cuesta $3.20$; sueltos, un yogur cuesta
$0.50$. Calcula el precio «por unidad» dentro del pack y cuánto se
ahorra por yogur comprando el pack.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:data:9}
«Un cine vendió $148$ entradas a $9.50$ el sábado y $95$ entradas
al mismo precio el domingo. ¿Cuánto dinero más recaudó el sábado
que el domingo?» Resuélvelo de dos maneras: calculando la recaudación de cada día o
calculando con la \emph{\omterm{ex:g1:subtraction:difference}{diferencia}} de entradas --- y comprueba que
las respuestas coinciden.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:data:10}
Una receta para $4$ personas necesita $300$ g de arroz. Alina cocina para
$10$ personas. ¿Cuánto arroz necesita? (Primero por persona ---
se permiten decimales.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:data:11}
Inventa un problema de tres pasos con dinero cuya respuesta final sea
$5.50$, escribe su solución completa al estilo del
\cref{met:g5:data:steps} y pruébalo con un compañero.
\end{exercise}
