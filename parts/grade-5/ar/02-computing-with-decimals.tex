\chapter{الحساب بالأعداد العشرية}\label{ch:g5:decimalops}

الأعداد العشرية تُجمع وتُطرح وتُضرب كالأعداد الصحيحة تقريبًا
--- والفنّ كله معرفة أين تقع النقطة. يغطي هذا الفصل طرائق
الأعمدة وسحر الضرب أو القسمة على $10$ و $100$ و $1000$. (أما ضرب
عددين عشريين فينتظر \cref{ch:g6:decimals}.)

\section{الجمع والطرح}

\begin{method}[الأعمدة مع نقطة]\label{met:g5:decimalops:addsub}
\begin{enumerate}
  \item اُكتب الأعداد مع \emph{محاذاة النقاط العشرية}
        (فتصير الآحاد تحت الآحاد، والأعشار تحت الأعشار)؛
  \item واحشُ الجزء العشري الأقصر بالأصفار؛
  \item واجمع أو اطرح كما في الأعداد الصحيحة؛
  \item وأنزِل نقطة النتيجة مستقيمةً إلى الأسفل.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:decimalops:addsub}
$27.5 + 3.86$ (باحتشاء $27.5 = 27.50$):
\[
\begin{array}{r}
2\,7.5\,0 \\
+\ \ \ 3.8\,6 \\
\hline
3\,1.3\,6
\end{array}
\qquad\qquad
\begin{array}{r}
1\,2.4\,0 \\
-\ \ \ 5.6\,5 \\
\hline
\ \ \,6.7\,5
\end{array}
\]
وفي الطرح ($12.4 - 5.65$): اِحشُ، واستلف كالعادة، وتحقّق
بالجمع العكسي: $6.75 + 5.65 = 12.4$.
\end{example}

\begin{example}[المتمّمات الذهنية]\label{ex:g5:decimalops:complement}
ماذا يجب أن نضيف إلى $7.6$ لنبلغ $10$؟ اِصعد: من $7.6 \to 8$
مقداره $0.4$، ثم من $8 \to 10$ مقداره $2$: فالمتمّم هو $2.4$. وهو
مفيد في \omterm{def:g3:division:remainder}{الباقي}: ادفع $10$ ثمنًا لشيء قيمته $7.60$، تسترد $2.40$.
\end{example}

\section{عشر مرات أكبر، وعشر مرات أصغر}

\begin{proposition}[الضرب والقسمة على 10 و100 و1000]\label{prop:g5:decimalops:shift}
الضرب في $10$ يجعل كل رقم يساوي عشرة أمثال ما كان: فتنزلق
الأرقام مرتبة واحدة إلى اليسار --- وهو ما يبدو كأنّ \emph{النقطة
تنتقل مرتبة واحدة إلى اليمين}. والقسمة تفعل العكس:
\[
3.47 \times 10 = 34.7, \qquad
3.47 \times 100 = 347, \qquad
52.1 \div 10 = 5.21, \qquad
6 \div 100 = 0.06 .
\]
\end{proposition}
\admitted

\begin{example}[التحويلات من جديد]\label{ex:g5:decimalops:convert}
وحدات القياس تفصل بينها قوى العشرة، فالتحويلات ليست إلا
هذه الانزلاقات: $2.35$ م $= 235$ سم ($\times 100$)؛ و $470$ غ
$= 0.47$ كغ ($\div 1000$)؛ و $1.5$ ل $= 150$ سل.
\end{example}

\section{ضرب عدد عشري في عدد صحيح}

\begin{method}[عشري في صحيح]\label{met:g5:decimalops:mult}
احسب كأن لا نقطة هناك، ثم أعطِ النتيجة من الأرقام العشرية
بعدد ما في العامل العشري:
\[
4.35 \times 6: \quad 435 \times 6 = 2\,610
\ \to\ 4.35 \times 6 = 26.10 = 26.1 .
\]
وقدِّر لتضع النقطة بثقة: فالعدد $4.35$ قريب من
$4$، و $4 \times 6 = 24$ --- إذن $26.1$، لا $2.61$ ولا $261$.
\end{method}

\begin{example}[الجمع المتكرر ما زال يعمل]\label{ex:g5:decimalops:mult}
$0.75 \times 4 = 0.75 + 0.75 + 0.75 + 0.75 = 3$: فأربعة أرباع تصنع
وحدة كاملة، بلباس عشري. والضرب في عدد صحيح ما زال
يعني «كذا مرة».
\end{example}

\begin{example}[التسوّق]\label{ex:g5:decimalops:shopping}
ثلاثة دفاتر ثمن الواحد $2.45$ وقلم ثمنه $1.20$:
\begin{enumerate}
  \item الدفاتر: $2.45 \times 3 = 7.35$؛
  \item \omterm{def:g1:addition:def}{والمجموع}: $7.35 + 1.20 = 8.55$؛
  \item \omterm{def:g3:division:remainder}{والباقي} من $10$: $10 - 8.55 = 1.45$.
\end{enumerate}
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:1}
احسب بالأعمدة: $45.7 + 8.93$؛ \quad $16.25 + 3.75$؛ \quad
$0.86 + 12.4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:2}
احسب بالأعمدة وتحقّق: $9.4 - 2.72$؛ \quad $20 - 13.45$؛
\quad $6.03 - 5.9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:3}
المتمّمات الذهنية إلى $10$ (كما في
\cref{ex:g5:decimalops:complement}): $6.2$؛ \quad $3.75$؛ \quad
$9.99$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:4}
احسب بلا أعمدة:
\[
7.24 \times 10, \qquad
0.58 \times 100, \qquad
34.5 \div 10, \qquad
7 \div 100, \qquad
0.3 \times 1000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:5}
حوِّل: $4.05$ م بالسنتيمتر؛ \quad $325$ سم بالمتر؛ \quad $0.6$ كغ
بالغرام؛ \quad $85$ سل باللتر.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:6}
قدِّر أولًا، ثم احسب: $6.2 \times 4$؛ \quad
$3.45 \times 8$؛ \quad $12.5 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:7}
طول لفّة واحدة من مضمار $0.4$ كم. فكم يبلغ طول $7$ لفّات؟ و
$25$ لفّة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:8}
تشتري لينا $2$ من الأرغفة ثمن الواحد $1.15$ وكعكة ثمنها $12.60$.
وتدفع بورقة $20$. احسب \omterm{def:g1:addition:def}{مجموع} ما دفعت \omterm{def:g3:division:remainder}{والباقي} لها.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:decimalops:9}
تسع قنينة $1.5$ ل. فكم لترًا في رزمة من $6$ قنينات؟
وتشرب أسرة $0.75$ ل في اليوم: فكم يومًا تكفيها الرزمة
الواحدة؟ (كم مرة يدخل $0.75$ في $9$؟)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:decimalops:10}
يحسب مالك $5.6 \times 3 = 15.18$، محتجًّا بأن «$5 \times 3 = 15$
و $6 \times 3 = 18$». اِستعمل تقديرًا لتبيّن أن الجواب لا بد أن
يكون خاطئًا، ثم احسب بالصواب.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:decimalops:11}
دارت أربع لفّات لعدّاءة في $65.4$ ثا و $64.9$ ثا و $65.0$ ثا و
$66.1$ ثا.
\begin{enumerate}
  \item احسب الزمن الكلي للفّات الأربع.
  \item والرقم القياسي لأربع لفّات هو $260$ ثا. فهل حطّمته؟
        وبكم؟
\end{enumerate}
\end{exercise}
