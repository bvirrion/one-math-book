\chapter{المحيط والمساحة}\label{ch:g5:areas}

في السنة الماضية عددنا مربعات الشبكة (\cref{ch:g4:measure})؛ وهذا
الفصل ينال أول \emph{صيغة} \omterm{def:g4:measure:area}{للمساحة} --- الطول في العرض
للمستطيل --- ويتعلم الوحدتين سم$^2$ و م$^2$. أما صيغ
المثلثات وغيرها من الأشكال فتنضج في \cref{ch:g6:measure} و
\cref{ch:g7:areas}.

\section{قياسان مختلفان}

\begin{example}[الحدّ مقابل السطح]\label{ex:g5:areas:contrast}
يحتاج البستانيّ إلى \emph{\omterm{def:g4:measure:perimeter}{المحيط}} ليشتري السياج، وإلى \emph{\omterm{def:g4:measure:area}{المساحة}}
ليشتري بذور العشب. والقياسان لا يتبع أحدهما الآخر: فمدّ
مستطيل ليصير أرفع وأطول قد يحفظ مساحته بينما ينمو
محيطه --- قارن مستطيل $6 \times 2$ ومستطيل $4 \times 3$
(\omterm{def:g4:measure:area}{المساحة} نفسها $12$، والمحيطان $16$ و $14$).
\end{example}

\section{صيغة المستطيل}

\begin{proposition}[مساحة المستطيل]\label{prop:g5:areas:rectangle}
مستطيل طوله $L$ وعرضه $w$ (بالوحدة نفسها) مساحته
\[
A = L \times w ,
\]
بوحدات مربعة: سم$^2$ (مربعات طول ضلعها $1$ سم)، و م$^2$ (مربعات
طول ضلعها $1$ م)، \dots
\end{proposition}

\begin{proof}[بيان بالعدّ]
غطِّ المستطيل بمربعات واحدية: $w$ صفًّا في كل منها $L$ مربعًا،
أي $L \times w$ مربعًا في المجموع --- وهو بالضبط مستطيل
الضرب في \cref{ch:g3:mult}.
\end{proof}

\end{tikzpicture}

{\small ثلاثة صفوف من ستة مربعات سنتيمترية: فالصيغة
$L \times w$ هي العدّ نفسه، مكتوبًا مرة واحدة وإلى الأبد.}
\end{omfigure}

\begin{example}\label{ex:g5:areas:rectangle}
سجادة قياسها $2.5$ م في $2$ م: مساحتها $2.5 \times 2 = 5$ م$^2$.
وطابع بريد قياسه $3$ سم في $2.4$ سم: مساحته $3 \times 2.4 = 7.2$
سم$^2$. الصيغة نفسها بأي وحدة --- ما دام الضلعان يستعملان
الوحدة \emph{نفسها}.
\end{example}

\begin{example}[وحدات المساحة تتحول بالمئات]\label{ex:g5:areas:units}
$1$ م $= 100$ سم، لكن $1$ م$^2 = 100 \times 100 = 10\,000$ سم$^2$:
فالمتر المربع شبكة $100$ في $100$ من السنتيمترات المربعة. ووحدات
\omterm{def:g4:measure:area}{المساحة} تقفز بالمئات، لا بالعشرات.
\end{example}

\section{الأشكال المركّبة}

\begin{method}[اقطع، واجمع، واطرح]\label{met:g5:areas:composite}
في الشكل المصنوع من مستطيلات (مثل L أو T أو إطار):
\begin{enumerate}
  \item اقطعه إلى مستطيلات، أو أكمله إلى مستطيل كبير؛
  \item واحسب \omterm{def:g4:measure:area}{مساحة} كل مستطيل بالصيغة؛
  \item ثم اجمع القطع --- أو اطرح الفجوة من المستطيل
        الكبير؛
  \item وتحقّق بعدّ تقريبي لمربعات الشبكة.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:areas:composite}
غرفة على هيئة L: مستطيل $6$ م $\times$ $4$ م ينقصه ركن
$2$ م $\times$ $2$ م.
\[
\text{المساحة} = 6 \times 4 - 2 \times 2 = 24 - 4 = 20 \text{ م}^2 .
\]
أو اقطع الشكل L إلى شريط $6 \times 2$ وشريط $4 \times 2$:
$12 + 8 = 20$ م$^2$ --- طريقان وجواب واحد.
\end{example}

\begin{example}[نصف مستطيل]\label{ex:g5:areas:halfrect}
قطع مستطيل على قطره يعطي مثلثين متساويي
\omterm{def:g4:measure:area}{المساحة}: كل منهما \emph{نصف} المستطيل. فمثلث قائم ضلعا قائمته
$6$ سم و $4$ سم مساحته إذن $\frac{6 \times 4}{2} = 12$
سم$^2$ --- وهي صورة جديرة بالحفظ من أجل
\cref{ch:g6:measure}، حيث تصير صيغة.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g5:areas:1}
احسب \omterm{def:g4:measure:area}{مساحة} \omterm{def:g4:measure:perimeter}{ومحيط} مستطيل $8$ سم $\times$
$5$ سم. فأي الجوابين بوحدة سم وأيّهما بوحدة سم$^2$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:2}
احسب المساحات: مربع طول ضلعه $9$ سم؛ ومستطيل
$12$ م $\times$ $7$ م؛ ومستطيل $4.5$ سم $\times$ $6$ سم.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:3}
مستطيل مساحته $63$ سم$^2$ وطوله $9$ سم. جِد عرضه،
ثم محيطه.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:4}
اُرسم مستطيلين مختلفين \omterm{def:g4:measure:area}{مساحة} كل منهما $24$ مربعًا على الورق
المربّعات، واحسب المحيطين. أفالمساحة نفسها تعني \omterm{def:g4:measure:perimeter}{المحيط} نفسه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:5}
حوِّل: $3$ م$^2$ إلى سم$^2$؛ \quad $50\,000$ سم$^2$ إلى م$^2$.
(وتذكّر \cref{ex:g5:areas:units}: بالمئات!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:6}
شكل على هيئة T مصنوع من شريط أفقي $8 \times 2$ فوق
شريط رأسي $2 \times 5$ (بالسنتيمتر). احسب مساحته بجمع
مستطيلين.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:7}
إطار صورة: مستطيل $30$ سم $\times$ $20$ سم قُطعت منه
نافذة مستطيلة $24$ سم $\times$ $14$ سم. فما \omterm{def:g4:measure:area}{مساحة}
الخشب الذي يستعمله الإطار؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:8}
احسب \omterm{def:g4:measure:area}{مساحة} مثلث قائم ضلعا قائمته $8$ سم و $6$ سم
(نصف مستطيل، \cref{ex:g5:areas:halfrect}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:areas:9}
حوض خضار مستطيل قياسه $7$ م في $4$ م. وثمن البذور
$2$ للمتر المربع، والسياج $3$ للمتر. احسب ثمن
البذور، ثم ثمن السياج.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:areas:10}
أرضية ممرّ طولها $12$ م وعرضها $2$ م، ويجب تغطيتها
ببلاطات مربعة طول ضلعها $50$ سم. فكم بلاطة تلزم؟
(حوّل أولًا، أو عُدّ البلاطات في كل اتجاه.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:areas:11}
ضاعِف ضلعَي مستطيل $4$ سم $\times$ $3$ سم. فماذا يحدث
لمحيطه؟ ولمساحته؟ (احسبهما قبل وبعد ---
فالجوابان مختلفان!)
\end{exercise}
