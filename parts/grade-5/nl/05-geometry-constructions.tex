\chapter{Meetkundige constructies}\label{ch:g5:geometry}

Lat, geodriehoek, passer: met deze drie stukken gereedschap en een beetje
methode kun je elke figuur die in woorden beschreven is, precies bouwen. Dit
hoofdstuk construeert loodlijnen, parallellen, driehoeken en de bijzondere
\omterm{def:g4:geometry:polygon}{vierhoeken} --- eerst het ambacht, later de theorie
(\cref{ch:g6:lines,ch:g6:shapes}).

\section{Loodlijnen en parallellen door een punt}

\begin{method}[De twee basisconstructies]\label{met:g5:geometry:basic}
\emph{De loodlijn op $d$ door $A$}: schuif de geodriehoek langs $d$ tot haar
andere rand van de \omterm{def:g3:shapes:rightangle}{rechte hoek} bij $A$ komt; teken; zet het teken van de
\omterm{def:g3:shapes:rightangle}{rechte hoek} erbij.

\emph{De \omterm{def:g4:geometry:def}{parallel} aan $d$ door $A$}: druk een lat als rail tegen de
geodriehoek die langs $d$ ligt; schuif de geodriehoek op tot bij $A$; teken.
(Controle: de afstand tussen de twee lijnen is overal gelijk.)
\end{method}

\begin{example}[Constructies aan elkaar rijgen]\label{ex:g5:geometry:chain}
Veel figuren zijn een reeks van deze twee bewegingen. Een rechthoek $ABCD$
met $AB = 5$ cm en $BC = 3$ cm: teken $[AB]$; de loodlijn in $A$; de loodlijn
in $B$; zet $D$ en $C$ op $3$ cm aan dezelfde kant; verbind. Elke stap is één
basisconstructie.
\end{example}

\section{Driehoeken met de passer}

\begin{method}[Een driehoek uit drie zijden]\label{met:g5:geometry:triangle}
Om een driehoek met zijden $6$ cm, $5$ cm en $4$ cm te bouwen:
\begin{enumerate}
  \item teken de langste zijde, $[AB]$ met $AB = 6$ cm;
  \item een boog met middelpunt $A$ en \omterm{def:g3:shapes:shapes}{radius} $5$ cm;
  \item een boog met middelpunt $B$ en \omterm{def:g3:shapes:shapes}{radius} $4$ cm;
  \item hun snijpunt is het derde \omterm{def:g4:geometry:polygon}{hoekpunt} $C$; verbind.
\end{enumerate}
De passer garandeert de lengtes: elk punt van de eerste boog ligt precies $5$
cm van $A$.
\end{method}

\begin{example}[Gelijkbenig en gelijkzijdig krijg je erbij]\label{ex:g5:geometry:special}
Dezelfde passeropening voor beide bogen $\Rightarrow$ een gelijkbenige
driehoek. Een opening gelijk aan de basis $\Rightarrow$ gelijkzijdig. De
passer meet niet: hij \emph{kopieert} lengtes, en dat is nog beter.
\end{example}

\section{De bijzondere vierhoeken, al bouwend}

\begin{method}[De familie bouwen]\label{met:g5:geometry:quadrilaterals}
\begin{enumerate}
  \item \emph{Rechthoek}: twee paar parallellen die elkaar onder \omterm{def:g3:shapes:rightangle}{rechte
        hoeken} snijden --- of de reeks van \cref{ex:g5:geometry:chain};
  \item \emph{vierkant}: een rechthoek waarvan de vier zijden dezelfde
        lengte gebruiken --- breng die met de passer over;
  \item \emph{ruit}: vier gelijke zijden --- teken één zijde, en twee bogen
        met dezelfde opening vanuit haar uiteinden leggen de andere twee
        \omterm{def:g4:geometry:polygon}{hoekpunten} vast (een ruit is twee gelijkbenige driehoeken die langs
        een diagonaal aan elkaar geplakt zijn).
\end{enumerate}
\emph{Controleer} na het bouwen altijd met het gereedschap: de geodriehoek op
de hoeken, de passer op de zijden.
\end{method}

\begin{example}[Herkennen uit een beschrijving]\label{ex:g5:geometry:recognize}
``Een \omterm{def:g4:geometry:polygon}{vierhoek} met vier zijden van $4$ cm en één \omterm{def:g3:shapes:rightangle}{rechte hoek}'' --- bouw hem:
de vier gelijke zijden dwingen de vorm van een ruit af, en één hoek recht
maken trekt alle andere ook recht: het resultaat is een vierkant. Sommige
beschrijvingen beschrijven in stilte een bijzonderder figuur dan ze
aankondigen. (Waarom die \omterm{def:g3:shapes:rightangle}{rechte hoek} zich over alle vier de hoeken verspreidt,
wordt uitgelegd in \cref{ch:g7:central}.)
\end{example}

\section{Cirkels in constructies}

\begin{example}[De cirkel als gereedschap]\label{ex:g5:geometry:circle}
``Zoek alle punten op $3$ cm van $A$'': dat is de cirkel met middelpunt $A$
en \omterm{def:g3:shapes:shapes}{radius} $3$ cm. ``Zoek alle punten op $3$ cm van $A$ \emph{en} $4$ cm van
$B$'': teken beide cirkels; hun snijpunten (twee, één of \omterm{def:g1:counting:zero}{nul}) zijn de
antwoorden. Schatkaarten en driehoeksconstructies gebruiken precies dit idee.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:geometry:1}
Teken een lijn $d$ en een punt $A$ op ongeveer $3$ cm ervan. Construeer de
loodlijn op $d$ door $A$, en daarna de \omterm{def:g4:geometry:def}{parallel} aan $d$ door $A$. Zet de
tekens voor de \omterm{def:g3:shapes:rightangle}{rechte hoeken} erbij.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:2}
Construeer een rechthoek $ABCD$ met $AB = 7$ cm en $BC = 4$ cm, volgens
\cref{ex:g5:geometry:chain}. Controleer de vierde hoek met de geodriehoek: is
de constructie precies, dan is hij automatisch recht.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:3}
Construeer een driehoek met zijden $7$ cm, $6$ cm en $3$ cm
(\cref{met:g5:geometry:triangle}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:4}
Construeer een gelijkbenige driehoek met basis $5$ cm en gelijke zijden van
$7$ cm; en daarna een gelijkzijdige driehoek met zijde $6$ cm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:5}
Construeer een vierkant met zijde $5$ cm. Welk gereedschap garandeert de
\omterm{def:g3:shapes:rightangle}{rechte hoeken}? En welk de gelijke zijden?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:6}
Construeer een ruit met zijde $4$ cm die geen vierkant is (kies zelf de
opening van de eerste hoek). Controleer de vier zijden met de passer.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:7}
Zet twee punten $A$ en $B$ met $AB = 5$ cm. Construeer alle punten die op $4$
cm van $A$ en op $3$ cm van $B$ liggen. Hoeveel zijn er?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:geometry:8}
Schrijf een tekenprogramma (genummerde stappen, punten met namen) voor een
rechthoek van $6$ cm $\times$ $2$ cm met daarop een gelijkbenige driehoek met
gelijke zijden van $4$ cm (de vorm van een potlood). Laat een klasgenoot het
uitvoeren.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:geometry:9}
Probeer een driehoek met zijden $8$ cm, $3$ cm en $4$ cm te construeren. Wat
gebeurt er met de bogen? Leg in één zin uit waarom deze driehoek niet kan
bestaan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:geometry:10}
Construeer een ruit waarvan de diagonalen $6$ cm en $4$ cm lang zijn, gegeven
dat de diagonalen van een ruit elkaar in hun middens onder een \omterm{def:g3:shapes:rightangle}{rechte hoek}
snijden: teken \emph{eerst} de diagonalen en verbind daarna hun vier
uiteinden.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:geometry:11}
Een geit staat met een touw van $4$ m vast aan een paal $A$, in een vlak veld
met een recht hek dat op $3$ m van $A$ langsloopt. Teken een plattegrond (1 cm
per meter): welk gebied kan de geit bereiken? Waar beperkt het hek haar?
\end{exercise}
