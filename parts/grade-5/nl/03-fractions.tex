\chapter{Breuken als getallen}\label{ch:g5:fractions}

Een \omterm{def:g4:fractions:def}{breuk} is niet alleen een plaatje van een taart: het is een echt
\emph{getal}, met een plaats op de getallenlijn --- \omterm{def:g1:addition:def}{soms} precies dezelfde
plaats als een decimaal getal. Dit hoofdstuk verbindt de twee werelden. (De
volledige rekenregels voor \omterm{def:g4:fractions:def}{breuken} komen in \cref{ch:g6:fractions} en
daarna.)

\section{Breuken op de lijn, nog eens}

\begin{example}[Plaatsen en lezen]\label{ex:g5:fractions:place}
Op een lijn die in vijfden verdeeld is, ligt $\frac{7}{5}$ $7$ stapjes van
$0$: voorbij $1$ (dat is $\frac55$), op $1 + \frac25$. Elke \omterm{def:g4:fractions:def}{breuk} woont
ergens op de lijn; bij dezelfde \omterm{def:g4:fractions:def}{noemer} betekent een grotere \omterm{def:g4:fractions:def}{teller}: verder
naar rechts.
\end{example}

\begin{proposition}[Hetzelfde punt, veel namen]\label{prop:g5:fractions:equal}
Elk deel in twee (of drie\,\dots) knippen verschuift het punt niet:
\[
\frac{1}{2} = \frac{2}{4} = \frac{3}{6} = \frac{5}{10},
\qquad
\frac{2}{3} = \frac{4}{6} = \frac{20}{30}.
\]
De \omterm{def:g4:fractions:def}{teller} en de \omterm{def:g4:fractions:def}{noemer} met hetzelfde getal vermenigvuldigen (of erdoor
delen) geeft een andere naam van dezelfde \omterm{def:g4:fractions:def}{breuk}.
\end{proposition}
\admitted

\section{Breuken en decimale getallen}

\begin{proposition}[Breuken met een decimale naam]\label{prop:g5:fractions:decimal}
\omterm{def:g4:fractions:def}{Breuken} van tienden, honderdsten en duizendsten \emph{zijn} decimale
getallen:
\[
\frac{3}{10} = 0.3, \qquad \frac{27}{100} = 0.27 .
\]
Ook een aantal andere \omterm{def:g4:fractions:def}{breuken} heeft een decimale naam; je vindt die door
naar tienden of honderdsten over te gaan:
\[
\frac{1}{2} = \frac{5}{10} = 0.5, \qquad
\frac{1}{4} = \frac{25}{100} = 0.25, \qquad
\frac{3}{4} = 0.75, \qquad
\frac{1}{5} = \frac{2}{10} = 0.2 .
\]
Maar niet alle: $\frac13 = 0.333\dots$ houdt nooit op --- voor dat getal is
de \omterm{def:g4:fractions:def}{breuk} de enige precieze naam.
\end{proposition}
\admitted

\begin{example}[De handigste naam kiezen]\label{ex:g5:fractions:choose}
Om driekwart van $12$ te betalen is de breuknaam handiger: $\frac34$ van
$12$ is $9$. Om $\frac34$ met $0.8$ te vergelijken is de decimale naam
handiger: $0.75 < 0.8$. Tweetalig zijn betaalt zich uit.
\end{example}

\section{Breuken vergelijken}

\begin{method}[Drie gereedschappen om te vergelijken]\label{met:g5:fractions:compare}
\begin{enumerate}
  \item \emph{Dezelfde \omterm{def:g4:fractions:def}{noemer}}: meer delen wint, $\frac57 > \frac37$;
  \item \emph{dezelfde \omterm{def:g4:fractions:def}{teller}}: grotere delen winnen, dus wint de
        \emph{kleinere} \omterm{def:g4:fractions:def}{noemer}: $\frac35 > \frac38$ (vijfden zijn groter
        dan achtsten);
  \item \emph{ijkpunten}: vergelijk elke \omterm{def:g4:fractions:def}{breuk} met $\frac12$ of met $1$:
        $\frac38 < \frac12 < \frac59$, dus $\frac38 < \frac59$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:fractions:compare}
Wie heeft er meer pizza gegeten: Ali met $\frac58$ of Bea met $\frac54$? De
\omterm{def:g4:fractions:def}{breuk} van Bea is groter dan $1$ (vijf \omterm{def:g3:division:half}{kwarten} is meer dan één hele pizza),
die van Ali is kleiner dan $1$: Bea heeft er meer gegeten --- zelfs meer dan
een hele pizza.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:fractions:1}
Teken een getallenlijn van $0$ tot $2$, verdeeld in vijfden, en zet erop:
$\frac25$; \ $\frac55$; \ $\frac85$; \ $\frac{10}{5}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:2}
Vul de gelijke namen aan:
$\frac12 = \frac{?}{8}$; \quad
$\frac34 = \frac{?}{8}$; \quad
$\frac{2}{5} = \frac{4}{?}$; \quad
$\frac{6}{9} = \frac{2}{?}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:3}
Welke \omterm{def:g4:fractions:def}{breuken} zijn de naam van een heel getal? Geef dat hele getal:
$\frac82$; \quad $\frac{9}{4}$; \quad $\frac{15}{3}$; \quad
$\frac{20}{5}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:4}
Schrijf als decimaal getal: $\frac{7}{10}$; \quad $\frac{41}{100}$; \quad
$\frac12$; \quad $\frac34$; \quad $\frac15$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:5}
Schrijf als \omterm{def:g4:fractions:def}{breuk} (tienden of honderdsten): $0.9$; \quad $0.13$; \quad
$2.5$; \quad $0.75$ (en geef daarna zijn naam in \omterm{def:g3:division:half}{kwarten}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:6}
Vergelijk, en noem het gereedschap dat je gebruikt
(\cref{met:g5:fractions:compare}):
\[
\frac47 \;?\; \frac67, \qquad
\frac35 \;?\; \frac37, \qquad
\frac25 \;?\; \frac{7}{12}, \qquad
\frac98 \;?\; \frac{11}{12} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:7}
Zet op volgorde met de decimale namen: $\frac12$; \ $0.4$; \ $\frac34$; \
$0.6$; \ $\frac15$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:8}
Een \omterm{def:g3:division:half}{kwart} van de leerlingen van een klas van $28$ speelt een instrument, en
de \omterm{def:g3:division:half}{helft} doet een sport. Hoeveel leerlingen zijn dat elk? Welke groep is
groter?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:fractions:9}
Koppel elke \omterm{def:g4:fractions:def}{breuk} aan een tijd: $\frac14$ u, $\frac12$ u, $\frac34$ u,
$\frac{1}{60}$ u --- bij $30$ min, $45$ min, $1$ min, $15$ min.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:fractions:10}
Tom heeft $\frac35$ van zijn fles van $50$ cL leeggedronken; Ana $\frac12$
van haar fles van $60$ cL. Wie heeft er meer gedronken? (Reken beide
hoeveelheden in cL uit --- de kale \omterm{def:g4:fractions:def}{breuken} vergelijken is hier niet genoeg,
en leg uit waarom.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:fractions:11}
Geef een \omterm{def:g4:fractions:def}{breuk} die strikt tussen $\frac12$ en $1$ ligt; en daarna een die
strikt tussen $\frac12$ en $\frac34$ ligt. (Decimale namen kunnen helpen.)
\end{exercise}
