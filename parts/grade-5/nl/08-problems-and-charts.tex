\chapter{Vraagstukken en diagrammen}\label{ch:g5:data}

Het laatste hoofdstuk van het jaar zet alles aan het werk: vraagstukken in
meerdere stappen met geld, hoeveelheden en tijdsduren, en de diagrammen die
informatie in één oogopslag laten zien. Het sterkste idee is ``per stuk'' ---
de prijs van één ding vinden --- en dat groeit in \cref{ch:g6:propdata} uit
tot evenredigheid.

\section{Vraagstukken in meerdere stappen}

\begin{method}[Een lang vraagstuk aanpakken]\label{met:g5:data:steps}
\begin{enumerate}
  \item Lees tot het einde; zeg wat de laatste vraag is;
  \item plan achterstevoren: wat heb ik eerst nodig om die te
        beantwoorden?
  \item los de stappen op volgorde op, met bij elke stap één bewerking en
        één korte zin;
  \item controleer het eindantwoord met je gezond verstand en met een
        schatting.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:data:multistep}
``Een school bestelt $4$ dozen met $25$ boeken van $6$ per stuk, en betaalt
daarbovenop $180$ voor de levering. Wat zijn de totale kosten?''
\begin{enumerate}
  \item Boeken: $4 \times 25 = 100$ boeken.
  \item Kosten van de boeken: $100 \times 6 = 600$.
  \item Totaal: $600 + 180 = 780$.
\end{enumerate}
Controle met een schatting: ongeveer honderd boeken van $6$ is ongeveer
$600$, plus ongeveer $200$: ongeveer $800$ --- het antwoord $780$ is
aannemelijk.
\end{example}

\begin{example}[Per stuk]\label{ex:g5:data:perone}
``$5$ dezelfde mokken kosten $12.50$; wat kosten $8$ mokken?'' Zoek eerst de
prijs van \emph{één} mok:
\[
12.50 \div 5 = 2.50,
\qquad\text{en dan}\qquad
8 \times 2.50 = 20 .
\]
Via één stuk wordt alles bereikbaar --- dit patroon van twee stappen lost een
hele familie vraagstukken op (recepten, snelheden, prijzen) en is het zaadje
van \cref{ch:g6:propdata}.
\end{example}

\begin{example}[Wisselgeld en budget]\label{ex:g5:data:budget}
Zoë heeft $30$. Ze koopt een boek van $12.40$ en twee pennen van $2.30$ per
stuk. Kan ze ook nog een bordspel van $13$ betalen?
\begin{enumerate}
  \item Pennen: $2 \times 2.30 = 4.60$.
  \item Tot nu toe uitgegeven: $12.40 + 4.60 = 17$.
  \item Over: $30 - 17 = 13$ --- precies genoeg voor het spel, en er blijft
        niets over.
\end{enumerate}
\end{example}

\section{Diagrammen lezen}

\begin{method}[Elk diagram lezen]\label{met:g5:data:read}
\begin{enumerate}
  \item Lees de titel: wat wordt er geteld?
  \item Lees de assen of de legenda: waar staat één streepje, één staaf of
        één symbool voor?
  \item Antwoord pas daarna op de vragen --- met een lat op het diagram lees
        je hoogtes precies af.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g5:data:chart}
De neerslag in een dorp, maand na maand:

\begin{omfigure}
\begin{tikzpicture}
\begin{axis}[
  width=8.8cm, height=5.0cm,
  ybar, bar width=13pt,
  symbolic x coords={jan,feb,mrt,apr,mei,jun},
  xtick=data,
  ymin=0, ymax=90,
  ytick={20,40,60,80},
  ylabel={regen (mm)},
  tick label style={font=\small},
  label style={font=\small}]
  \addplot[fill=omDef!30, draw=omDef] coordinates
    {(jan,60) (feb,45) (mrt,55) (apr,70) (mei,80) (jun,30)};
\end{axis}
\end{tikzpicture}

{\small Een staafdiagram van de neerslag. Eén streepje $= 20$ mm.}
\end{omfigure}

Aflezen: de natste maand is mei ($80$ mm); in juni viel de \omterm{def:g3:division:half}{helft} van de regen
van april ($30$ tegen $70$? nee --- de \omterm{def:g3:division:half}{helft} van $70$ is $35$, dus
\emph{net niet} de \omterm{def:g3:division:half}{helft}: precies lezen doet ertoe!); het totaal over de zes
maanden: $60 + 45 + 55 + 70 + 80 + 30 = 340$ mm.
\end{example}

\begin{example}[Een veranderende grootheid]\label{ex:g5:data:line}
De hoogte van een plantje, elke week gemeten:

\begin{omfigure}
\begin{tikzpicture}
\begin{axis}[omaxis,
  width=8.4cm, height=5.0cm,
  xmin=0, xmax=6.5, ymin=0, ymax=24,
  xlabel={week}, ylabel={hoogte (cm)},
  xtick={1,2,3,4,5,6}, ytick={5,10,15,20}]
  \addplot[omDef, thick, mark=*, mark size=1.6pt] coordinates
    {(1,3) (2,6) (3,10) (4,15) (5,19) (6,21)};
\end{axis}
\end{tikzpicture}

{\small De meetpunten verbinden laat de \emph{groei} zien: hoe steiler het
stuk, hoe sneller het plantje die week groeide.}
\end{omfigure}

Het plantje groeide het snelst tussen week $3$ en week $4$ ($+5$ cm) en
vertraagde aan het einde ($+2$ cm in week $6$).
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:data:1}
Los in stappen op: ``Een club huurt een bus voor $240$ en koopt $32$
museumkaartjes van $7$ per stuk. Wat zijn de totale kosten van het uitje?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:2}
Los op met ``per stuk'': ``$3$ kg appels kost $7.50$. Wat kost $5$ kg?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:3}
Los op: ``Zes vrienden verdelen de kosten van een picknick van $87$ gelijk.
Hoeveel betaalt ieder?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:4}
Sam heeft $25$. Hij koopt een T-shirt van $13.90$ en een pet van $8.50$.
Hoeveel geld houdt hij over?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:5}
Gebruik het neerslagdiagram van \cref{ex:g5:data:chart}: in welke maanden viel
er meer dan $50$ mm? Hoeveel regen viel er in mei meer dan in juni?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:6}
Gebruik het plantendiagram van \cref{ex:g5:data:line}: hoe hoog was het
plantje in week $2$? Tussen welke weken groeide het precies $4$ cm?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:7}
Teken een staafdiagram van het aantal doelpunten van een team: september $4$,
oktober $7$, november $3$, december $6$. Kies zelf de streepjes, en geef het
totaal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:data:8}
Een pak van $8$ yoghurtjes kost $3.20$; los verkocht kost één yoghurtje
$0.50$. Reken de prijs per stuk in het pak uit, en hoeveel je per yoghurtje
bespaart door het pak te kopen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:data:9}
``Een bioscoop verkocht op zaterdag $148$ kaartjes van $9.50$ en op zondag
$95$ kaartjes voor dezelfde prijs. Hoeveel geld bracht zaterdag meer op dan
zondag?'' Los het op twee manieren op: door de opbrengst van beide dagen uit
te rekenen, of door met het \emph{\omterm{ex:g1:subtraction:difference}{verschil}} in kaartjes te rekenen --- en
controleer dat de antwoorden overeenkomen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:data:10}
Een recept voor $4$ personen vraagt $300$ g rijst. Aline kookt voor $10$
personen. Hoeveel rijst heeft ze nodig? (Eerst per persoon --- decimalen
mogen.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:data:11}
Verzin een vraagstuk in drie stappen met geld waarvan het eindantwoord $5.50$
is, schrijf de volledige oplossing op in de stijl van
\cref{met:g5:data:steps}, en test het bij een klasgenoot.
\end{exercise}
