\chapter{Kubussen en balken}\label{ch:g5:solids}

We verlaten het vlakke papier: lichamen zijn de vormen van de echte wereld
--- dobbelstenen, pakken ontbijtgranen, blikjes, ballen. In dit hoofdstuk
leer je ze benoemen, hun \omterm{def:g5:solids:def}{zijvlakken} en \omterm{def:g5:solids:def}{ribben} tellen, ze tot een \omterm{def:g5:solids:net}{uitslag}
openvouwen, en hun \omterm{def:g5:solids:volume}{volume} meten door kleine kubusjes te tellen.
(Volumeformules komen in \cref{ch:g6:measure}.)

\section{De familie van de lichamen}

\begin{definition}[Lichamen en hun woorden]\label{def:g5:solids:def}
Een \emph{lichaam}\index{lichaam} neemt ruimte in. Zijn vlakke kanten zijn
\emph{zijvlakken}\index{zijvlak}; zijvlakken ontmoeten elkaar langs
\emph{ribben}\index{ribbe}; ribben ontmoeten elkaar in
\emph{hoekpunten}\index{hoekpunt}. De familieportretten:
\begin{itemize}
  \item de \emph{kubus}\index{kubus}: $6$ vierkante zijvlakken, $12$ ribben,
        $8$ hoekpunten;
  \item de \emph{balk}\index{balk}: $6$ rechthoekige zijvlakken, $12$
        ribben, $8$ hoekpunten;
  \item de \emph{cilinder}\index{cilinder}: twee ronde zijvlakken en één
        opgerold vlak (een blikje);
  \item de \emph{piramide}\index{piramide}: een \omterm{def:g4:geometry:polygon}{veelhoek} als grondvlak, met
        driehoeken die in een top samenkomen;
  \item de \emph{bol}: één volmaakt rond oppervlak, geen zijvlak, geen ribbe.
\end{itemize}
\end{definition}

\begin{example}[Tellen op de kubus]\label{ex:g5:solids:count}
Controleer het op een dobbelsteen: $6$ \omterm{def:g5:solids:def}{zijvlakken} (de getallen $1$ tot $6$),
$8$ \omterm{def:g5:solids:def}{hoekpunten} (de punten), $12$ \omterm{def:g5:solids:def}{ribben} (ga er met je vinger langs: $4$
bovenaan, $4$ onderaan, $4$ verticaal). En $6 + 8 = 12 + 2$ --- hetzelfde
merkwaardige patroon geldt voor de \omterm{def:g5:solids:def}{balk} en de \omterm{def:g5:solids:def}{piramide}
(\cref{ex:g8:solids:count} komt erop terug).
\end{example}

\section{Uitslagen}

\begin{definition}[Uitslag]\label{def:g5:solids:net}
Een \emph{uitslag}\index{uitslag} van een \omterm{def:g5:solids:def}{lichaam} is een vlakke tekening van
al zijn \omterm{def:g5:solids:def}{zijvlakken}, aan elkaar vast langs de \omterm{def:g5:solids:def}{ribben}, die je tot het \omterm{def:g5:solids:def}{lichaam}
kunt opvouwen --- het \omterm{def:g5:solids:def}{lichaam} uitgevouwen, zoals een kartonnen doos die je
plat maakt voor het oud papier.
\end{definition}

\begin{example}[Een uitslag lezen]\label{ex:g5:solids:read}
Op een dobbelsteen maken de \omterm{def:g5:solids:def}{zijvlakken} die tegenover elkaar liggen samen $7$.
Op een \omterm{def:g5:solids:net}{uitslag} zijn dat de \omterm{def:g5:solids:def}{zijvlakken} die precies één vierkant van elkaar
liggen in een rij (of om een hoek) --- niet de buren! De paren op de \omterm{def:g5:solids:net}{uitslag}
markeren voordat je vouwt, is een prachtige oefening in plat naar ruimte
kijken.
\end{example}

\section{Volume: kubusjes tellen}

\begin{definition}[Volume door te tellen]\label{def:g5:solids:volume}
Het \emph{volume}\index{volume} van een \omterm{def:g5:solids:def}{lichaam} dat uit eenheidskubusjes is
opgebouwd, is het aantal kubusjes dat het bevat. De eenheid kan de
kubieke centimeter (cm$^3$) of de kubieke meter (m$^3$) zijn.
\end{definition}

\begin{example}[Een balk laag voor laag tellen]\label{ex:g5:solids:layers}
Een \omterm{def:g5:solids:def}{balk} van $4$ kubusjes lang, $3$ breed en $2$ hoog: de onderste laag bevat
$4 \times 3 = 12$ kubusjes, en er zijn $2$ lagen:
\[
12 \times 2 = 24 \text{ kubusjes} .
\]
Per laag tellen werkt altijd --- en het bewijst in stilte de formule
$V = l \times b \times h$ van \cref{ch:g6:measure}.
\end{example}

\begin{example}[Dezelfde kubusjes, andere vormen]\label{ex:g5:solids:rearrange}
Met acht eenheidskubusjes kun je een \omterm{def:g5:solids:def}{kubus} van $2 \times 2 \times 2$ bouwen,
een staaf van $8 \times 1 \times 1$ of een plaat van $4 \times 2 \times 1$:
drie verschillende lichamen, één \omterm{def:g5:solids:volume}{volume} ($8$ kubusjes). Het \omterm{def:g5:solids:volume}{volume} telt het
materiaal, niet de vorm.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g5:solids:1}
Noem een echt voorwerp met de vorm van: een \omterm{def:g5:solids:def}{kubus}; een \omterm{def:g5:solids:def}{balk}; een \omterm{def:g5:solids:def}{cilinder}; een
bol; een \omterm{def:g5:solids:def}{piramide}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:2}
Hoeveel \omterm{def:g5:solids:def}{zijvlakken}, \omterm{def:g5:solids:def}{ribben} en \omterm{def:g5:solids:def}{hoekpunten} heeft een \omterm{def:g5:solids:def}{balk}? Controleer het
patroon \omterm{def:g5:solids:def}{zijvlakken} $+$ \omterm{def:g5:solids:def}{hoekpunten} $=$ \omterm{def:g5:solids:def}{ribben} $+ 2$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:3}
Welke \omterm{def:g5:solids:def}{zijvlakken} van een \omterm{def:g5:solids:def}{balk} zijn identiek? (Zet de zes \omterm{def:g5:solids:def}{zijvlakken} in
paren.) Wat is er bijzonder aan de \omterm{def:g5:solids:def}{zijvlakken} van de \omterm{def:g5:solids:def}{kubus}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:4}
Teken een \omterm{def:g5:solids:net}{uitslag} van een \omterm{def:g5:solids:def}{kubus} met zijde $2$ cm (gebruik het kruismodel uit
dit hoofdstuk), knip hem uit en vouw hem op.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:5}
Zet op de kruisvormige \omterm{def:g5:solids:net}{uitslag} van een dobbelsteen de getallen $1$ tot $6$ zo
neer dat \omterm{def:g5:solids:def}{zijvlakken} die tegenover elkaar liggen samen $7$ maken. (Zoek met
\cref{ex:g5:solids:read} eerst de paren.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:6}
Tel de eenheidskubusjes: een \omterm{def:g5:solids:def}{balk} van $5$ lang, $2$ breed en $3$ hoog; een
\omterm{def:g5:solids:def}{kubus} met zijde $3$; een L-vormige stapel van een plaat
$3 \times 2 \times 1$ met daarop een plaat $1 \times 2 \times 1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:7}
In één laag van een \omterm{def:g5:solids:def}{balk} passen precies $12$ eenheidskubusjes. Geef twee
mogelijke vormen voor die laag ($l \times b$), en bij elk het \omterm{def:g5:solids:volume}{volume} van de
\omterm{def:g5:solids:def}{balk} als hij $3$ zulke lagen heeft.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g5:solids:8}
Welke \omterm{def:g5:solids:def}{kubus} kun je met $27$ eenheidskubusjes bouwen? En met $64$? (Denk aan het
tellen per laag, achterstevoren.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:solids:9}
Suikerklontjes met zijde $1$ cm zitten in een doos van $10$ cm lang, $5$ cm
breed en $4$ cm hoog (binnenmaten). Hoeveel klontjes passen erin? Als een
gezin er $6$ per dag gebruikt, hoeveel dagen doet één doos dan (rond zinnig
af)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g5:solids:10}
Een \omterm{def:g5:solids:def}{kubus} van $3 \times 3 \times 3$ wordt aan de buitenkant rood geverfd en
daarna in $27$ eenheidskubusjes gezaagd. Hoeveel kleine kubusjes hebben verf
op precies $3$ \omterm{def:g5:solids:def}{zijvlakken}? Op $2$? Op $1$? Op geen enkel? (Controle: de vier
aantallen moeten samen $27$ zijn.)
\end{exercise}
