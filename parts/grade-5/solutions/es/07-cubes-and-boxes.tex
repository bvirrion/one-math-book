\section*{Capítulo \ref{ch:g5:solids} --- Cubos y ortoedros}

\begin{solution}{exo:g5:solids:1}
Por ejemplo: un dado (\omterm{def:g5:solids:def}{cubo}); un cereal \omterm{def:g5:solids:def}{caja} (\omterm{def:g5:solids:def}{caja}); una lata (\omterm{def:g5:solids:def}{cilindro}); un
fútbol (pelota); ciertos tejados o los monumentos de Egipto
(\omterm{def:g5:solids:def}{pirámide}).
\end{solution}

\begin{solution}{exo:g5:solids:2}
A \omterm{def:g5:solids:def}{caja} tiene $6$ \omterm{def:g5:solids:def}{caras}, $12$ \omterm{def:g5:solids:def}{bordes}, $8$ \omterm{def:g5:solids:def}{vértices} --- y
$6 + 8 = 12 + 2$. \checkmark
\end{solution}

\begin{solution}{exo:g5:solids:3}
los seis \omterm{def:g5:solids:def}{caras} vienen en tres \omterm{def:g3:numbers:evenodd}{pares} de opuestos idénticos \omterm{def:g5:solids:def}{caras}
(arriba--abajo, adelante--atrás, izquierda-derecha). en un \omterm{def:g5:solids:def}{cubo}, los seis \omterm{def:g5:solids:def}{caras}
son \omterm{def:g3:shapes:shapes}{cuadrados} idénticos.
\end{solution}

\begin{solution}{exo:g5:solids:4}
En forma de cruz \omterm{def:g5:solids:net}{neto}: cuatro $2$ \omterm{def:g3:shapes:shapes}{cuadrados} de cm seguidos, uno arriba y otro
debajo del segundo cuadrado. Doblar el anillo de cuatro forma los lados;
los dos \omterm{def:g3:shapes:shapes}{cuadrados} adicionales cierran la parte superior e inferior.
\end{solution}

\begin{solution}{exo:g5:solids:5}
en la cruz \omterm{def:g5:solids:net}{neto} (cuatro \omterm{def:g3:shapes:shapes}{cuadrados} seguidos, uno arriba, uno debajo del
segundo), los \omterm{def:g3:numbers:evenodd}{pares} opuestos son: \omterm{def:g3:shapes:shapes}{cuadrados} $1$ y $3$ de la fila;
cuadrícula $2$ y $4$ de la fila; los \omterm{def:g3:shapes:shapes}{cuadrados} superior e inferior. uno
etiquetado correcto: fila $= 1, 2, 6, 5$; arriba $= 3$; abajo $= 4$
--- entonces $1{+}6$, $2{+}5$, $3{+}4$ todos hacen $7$.
\end{solution}

\begin{solution}{exo:g5:solids:6}
\omterm{def:g5:solids:def}{caja}: $5 \times 2 = 10$ por capa, $3$ capas: $30$ \omterm{def:g5:solids:def}{cubos}. \omterm{def:g5:solids:def}{Cubo}:
$3 \times 3 \times 3 = 27$. Pila en L: $3 \times 2 \times 1 = 6$ más
$1 \times 2 \times 1 = 2$: $8$ \omterm{def:g5:solids:def}{cubos}.
\end{solution}

\begin{solution}{exo:g5:solids:7}
capas de $12$: por ejemplo $4 \times 3$ o $6 \times 2$ (o
$12 \times 1$). Con $3$ capas, el \omterm{def:g5:solids:volume}{volumen} es
$12 \times 3 = 36$ \omterm{def:g5:solids:def}{cubos} en todos los casos.
\end{solution}

\begin{solution}{exo:g5:solids:8}
$27 = 3 \times 3 \times 3$: a \omterm{def:g5:solids:def}{cubo} de lado $3$.
$64 = 4 \times 4 \times 4$: a \omterm{def:g5:solids:def}{cubo} de lado $4$.
\end{solution}

\begin{solution}{exo:g5:solids:9}
El \omterm{def:g5:solids:def}{caja} sostiene $10 \times 5 \times 4 = 200$ azúcar \omterm{def:g5:solids:def}{cubos}. En $6$ por
día: $200 = 6 \times 33 + 2$, entonces el \omterm{def:g5:solids:def}{caja} dura $33$ días completos (y
$2$ \omterm{def:g5:solids:def}{cubos} permanecer para la mañana del día $34$).
\end{solution}

\begin{solution}{exo:g5:solids:10}
pintar sobre $3$ \omterm{def:g5:solids:def}{caras}: el $8$ esquina \omterm{def:g5:solids:def}{cubos}. En $2$ \omterm{def:g5:solids:def}{caras}: los \omterm{def:g3:division:half}{medios}
de la $12$ \omterm{def:g5:solids:def}{bordes}: $12$ \omterm{def:g5:solids:def}{cubos}. En $1$ rostro: los centros de la $6$
\omterm{def:g5:solids:def}{caras}: $6$ \omterm{def:g5:solids:def}{cubos}. Sobre ninguno: el único centro oculto \omterm{def:g5:solids:def}{cubo}: $1$.
Controlar: $8 + 12 + 6 + 1 = 27$. \checkmark
\end{solution}
