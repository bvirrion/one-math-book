\section*{Capítulo \ref{ch:g5:solids} --- Cubos y ortoedros}

\begin{solution}{exo:g5:solids:1}
Por ejemplo: un dado (\omterm{def:g5:solids:def}{cubo}); una caja de cereales (\omterm{def:g5:solids:def}{ortoedro}); una lata (\omterm{def:g5:solids:def}{cilindro}); un
balón (bola); ciertos tejados o los monumentos de Egipto
(\omterm{def:g5:solids:def}{pirámide}).
\end{solution}

\begin{solution}{exo:g5:solids:2}
Un \omterm{def:g5:solids:def}{ortoedro} tiene $6$ \omterm{def:g5:solids:def}{caras}, $12$ \omterm{def:g5:solids:def}{aristas} y $8$ \omterm{def:g5:solids:def}{vértices} --- y
$6 + 8 = 12 + 2$. \checkmark
\end{solution}

\begin{solution}{exo:g5:solids:3}
Las seis \omterm{def:g5:solids:def}{caras} van en tres parejas de \omterm{def:g5:solids:def}{caras} opuestas iguales
(arriba--abajo, delante--detrás, izquierda--derecha). En un \omterm{def:g5:solids:def}{cubo}, las seis \omterm{def:g5:solids:def}{caras}
son cuadrados idénticos.
\end{solution}

\begin{solution}{exo:g5:solids:4}
\omterm{def:g5:solids:net}{Desarrollo} en cruz: cuatro cuadrados de $2$ cm en fila, uno encima y otro
debajo del segundo. Al plegar el anillo de cuatro se forman los costados;
los dos cuadrados sueltos cierran la tapa y el fondo.
\end{solution}

\begin{solution}{exo:g5:solids:5}
En el \omterm{def:g5:solids:net}{desarrollo} en cruz (cuatro cuadrados en fila, uno encima y otro debajo del
segundo), las parejas opuestas son: los cuadrados $1$ y $3$ de la fila;
los cuadrados $2$ y $4$ de la fila; el cuadrado de arriba y el de abajo. Una
numeración correcta: fila $= 1, 2, 6, 5$; arriba $= 3$; abajo $= 4$
--- así, $1{+}6$, $2{+}5$ y $3{+}4$ suman $7$.
\end{solution}

\begin{solution}{exo:g5:solids:6}
\omterm{def:g5:solids:def}{Ortoedro}: $5 \times 2 = 10$ por capa y $3$ capas: $30$ \omterm{def:g5:solids:def}{cubos}. \omterm{def:g5:solids:def}{Cubo}:
$3 \times 3 \times 3 = 27$. Pila en L: $3 \times 2 \times 1 = 6$ más
$1 \times 2 \times 1 = 2$: $8$ \omterm{def:g5:solids:def}{cubos}.
\end{solution}

\begin{solution}{exo:g5:solids:7}
Capas de $12$: por ejemplo, $4 \times 3$ o $6 \times 2$ (o
$12 \times 1$). Con $3$ capas, el \omterm{def:g5:solids:volume}{volumen} es
$12 \times 3 = 36$ \omterm{def:g5:solids:def}{cubos} en todos los casos.
\end{solution}

\begin{solution}{exo:g5:solids:8}
$27 = 3 \times 3 \times 3$: un \omterm{def:g5:solids:def}{cubo} de lado $3$.
$64 = 4 \times 4 \times 4$: un \omterm{def:g5:solids:def}{cubo} de lado $4$.
\end{solution}

\begin{solution}{exo:g5:solids:9}
En la caja caben $10 \times 5 \times 4 = 200$ terrones de azúcar. A $6$ al
día: $200 = 6 \times 33 + 2$, así que la caja dura $33$ días completos (y
quedan $2$ terrones para la mañana del día $34$).
\end{solution}

\begin{solution}{exo:g5:solids:10}
Con pintura en $3$ \omterm{def:g5:solids:def}{caras}: los $8$ \omterm{def:g5:solids:def}{cubos} de las esquinas. En $2$ \omterm{def:g5:solids:def}{caras}: los del medio
de las $12$ \omterm{def:g5:solids:def}{aristas}, $12$ \omterm{def:g5:solids:def}{cubos}. En $1$ \omterm{def:g5:solids:def}{cara}: los centros de las $6$
\omterm{def:g5:solids:def}{caras}, $6$ \omterm{def:g5:solids:def}{cubos}. En ninguna: el único \omterm{def:g5:solids:def}{cubo} central escondido, $1$.
Comprobación: $8 + 12 + 6 + 1 = 27$. \checkmark
\end{solution}
