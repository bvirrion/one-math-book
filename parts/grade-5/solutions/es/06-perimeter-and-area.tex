\section*{Capítulo \ref{ch:g5:areas} --- Perímetro y área}

\begin{solution}{exo:g5:areas:1}
\omterm{def:g4:measure:area}{área}: $8 \times 5 = 40$ centímetro $^2$. \omterm{def:g4:measure:perimeter}{perímetro}:
$2 \times (8 + 5) = 26$ centímetro.
\end{solution}

\begin{solution}{exo:g5:areas:2}
$9 \times 9 = 81$ centímetro $^2$; \quad $12 \times 7 = 84$ metro $^2$; \quad
$4.5 \times 6 = 27$ centímetro $^2$.
\end{solution}

\begin{solution}{exo:g5:areas:3}
Ancho: $63 \div 9 = 7$ centímetro. \omterm{def:g4:measure:perimeter}{perímetro}: $2 \times (9 + 7) = 32$ centímetro.
\end{solution}

\begin{solution}{exo:g5:areas:4}
Por ejemplo $6 \times 4$ (\omterm{def:g4:measure:perimeter}{perímetro} $20$) y $8 \times 3$
(\omterm{def:g4:measure:perimeter}{perímetro} $22$): mismo \omterm{def:g4:measure:area}{área} $24$,
diferente \omterm{def:g4:measure:perimeter}{perímetros}.
\end{solution}

\begin{solution}{exo:g5:areas:5}
$3$ metro $^2 = 30\,000$ centímetro $^2$; \quad $50\,000$ centímetro $^2 = 5$ metro $^2$.
\end{solution}

\begin{solution}{exo:g5:areas:6}
Bar: $8 \times 2 = 16$ centímetro $^2$; provenir: $2 \times 5 = 10$ centímetro $^2$; total $26$ centímetro $^2$.
\end{solution}

\begin{solution}{exo:g5:areas:7}
$30 \times 20 - 24 \times 14 = 600 - 336 = 264$ centímetro $^2$ de madera.
\end{solution}

\begin{solution}{exo:g5:areas:8}
\omterm{def:g3:division:half}{Medio} el $8 \times 6$ rectángulo: $48 \div 2 = 24$ centímetro $^2$.
\end{solution}

\begin{solution}{exo:g5:areas:9}
Semilla: \omterm{def:g4:measure:area}{área} $7 \times 4 = 28$ metro $^2$, costo $28 \times 2 = 56$.
Cerca: \omterm{def:g4:measure:perimeter}{perímetro} $2 \times (7 + 4) = 22$ m, costo
$22 \times 3 = 66$.
\end{solution}

\begin{solution}{exo:g5:areas:10}
dos azulejos de $50$ cm hacer un metro: a lo largo del $12$ longitud m,
$24$ azulejos; a lo largo del $2$ m de ancho, $4$ azulejos. Total:
$24 \times 4 = 96$ azulejos.
\end{solution}

\begin{solution}{exo:g5:areas:11}
Antes: \omterm{def:g4:measure:perimeter}{perímetro} $2 \times (4 + 3) = 14$ centímetro, \omterm{def:g4:measure:area}{área} $12$ centímetro $^2$.
Después (lados $8$ y $6$): \omterm{def:g4:measure:perimeter}{perímetro} $28$ centímetro, \omterm{def:g4:measure:area}{área} $48$ centímetro $^2$.
El \omterm{def:g4:measure:perimeter}{perímetro} \emph{duplicado}; el \omterm{def:g4:measure:area}{área} fue multiplicado por
\emph{cuatro} ($2 \times 2$): longitudes y \omterm{def:g4:measure:area}{áreas} no escalar igual
manera.
\end{solution}
