\begin{solution}{exo:g5:decimals:1}

\begin{solution}{exo:g5:decimals:2}
$4$ 表示十分位，$8$ 表示千分位：
\[
27.418 = 27 + \frac{4}{10} + \frac{1}{100} + \frac{8}{1000}.
\]
\end{solution}

\begin{solution}{exo:g5:decimals:3}
$3.60 = 3.6$：正确（末尾添零，数值不变）。

$0.5 = 0.05$：错误——$5$ 个十分之一与 $5$ 个百分之一不同。

$2.070 = 2.07$：正确。

$1.3 = 1.03$：错误——$3$ 个十分之一与 $3$ 个百分之一不同。
\end{solution}

\begin{solution}{exo:g5:decimals:4}
$4.8 > 4.79$；\quad $0.302 < 0.32$（十分位相同，百分位
$0 < 2$）；\quad $6.5 = 6.500$；\quad $9.09 < 9.1$。
\end{solution}

\begin{solution}{exo:g5:decimals:5}
$3.041 < 3.104 < 3.14 < 3.4 < 3.41$.
\end{solution}

\begin{solution}{exo:g5:decimals:6}
$6.83$：介于 $6$ 与 $7$ 之间；介于 $6.8$ 与 $6.9$ 之间。

$0.492$：介于 $0$ 与 $1$ 之间；介于 $0.4$ 与 $0.5$ 之间。

$12.06$：介于 $12$ 与 $13$ 之间；介于 $12.0$ 与 $12.1$ 之间。
\end{solution}

\begin{solution}{exo:g5:decimals:7}
$8.276$：精确到个位得 $8$；精确到十分位得 $8.3$；精确到百分位
得 $8.28$。对 $3.95$ 精确到十分位时，百分位上的数字是 $5$，按\omterm{def:g4:numbers:round}{四舍
五入}，进一位得 $4.0$。
\end{solution}

\begin{solution}{exo:g5:decimals:9}
$0.85 < 0.9 < 1.02 < 1.2$（单位为米）。
\end{solution}

\begin{solution}{exo:g5:decimals:10}
先比较十分位：$7.12$ 的十分位是 $1$，$7.9$ 的十分位是 $9$，
所以 $7.12 < 7.9$。露西把小数部分当作整数来比较，
即 $12$ 与 $9$，忘了 $12$ 个百分之一远小于
$90$ 个百分之一。
\end{solution}

\begin{solution}{exo:g5:decimals:11}
精确到十分位得 $6.4$ 的两位小数，是从
$6.35$（恰为\omterm{def:g3:division:half}{一半}时进一）到 $6.44$ 之间的数：$6.35$、$6.36$、
\dots，$6.44$。最小为 $6.35$，最大为 $6.44$。
\end{solution}
