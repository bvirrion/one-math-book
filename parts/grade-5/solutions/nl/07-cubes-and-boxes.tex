\section*{Hoofdstuk \ref{ch:g5:solids} --- Kubussen en balken}

\begin{solution}{exo:g5:solids:1}
Bijvoorbeeld: een dobbelsteen (\omterm{def:g5:solids:def}{kubus}); een pak ontbijtgranen (\omterm{def:g5:solids:def}{balk}); een
blikje (\omterm{def:g5:solids:def}{cilinder}); een voetbal (bol); bepaalde daken of de monumenten van
Egypte (\omterm{def:g5:solids:def}{piramide}).
\end{solution}

\begin{solution}{exo:g5:solids:2}
Een \omterm{def:g5:solids:def}{balk} heeft $6$ \omterm{def:g5:solids:def}{zijvlakken}, $12$ \omterm{def:g5:solids:def}{ribben} en $8$ \omterm{def:g5:solids:def}{hoekpunten} --- en
$6 + 8 = 12 + 2$. \checkmark
\end{solution}

\begin{solution}{exo:g5:solids:3}
De zes \omterm{def:g5:solids:def}{zijvlakken} vormen drie paren van identieke \omterm{def:g5:solids:def}{zijvlakken} die tegenover
elkaar liggen (boven--onder, voor--achter, links--rechts). Bij een \omterm{def:g5:solids:def}{kubus} zijn
alle zes de \omterm{def:g5:solids:def}{zijvlakken} identieke vierkanten.
\end{solution}

\begin{solution}{exo:g5:solids:4}
Kruisvormige \omterm{def:g5:solids:net}{uitslag}: vier vierkanten van $2$ cm op een rij, met één vierkant
boven en één onder het tweede. Als je de ring van vier omvouwt, krijg je de
zijkanten; de twee extra vierkanten sluiten boven- en onderkant.
\end{solution}

\begin{solution}{exo:g5:solids:5}
Op de kruisuitslag (vier vierkanten op een rij, één boven en één onder het
tweede) liggen tegenover elkaar: het $1$ste en het $3$de vierkant van de rij;
het $2$de en het $4$de vierkant van de rij; en het bovenste met het onderste
vierkant. Eén goede belettering: de rij $= 1, 2, 6, 5$; boven $= 3$; onder
$= 4$ --- dan maken $1{+}6$, $2{+}5$ en $3{+}4$ allemaal $7$.
\end{solution}

\begin{solution}{exo:g5:solids:6}
\omterm{def:g5:solids:def}{Balk}: $5 \times 2 = 10$ per laag, $3$ lagen: $30$ kubusjes. \omterm{def:g5:solids:def}{Kubus}:
$3 \times 3 \times 3 = 27$. L-stapel: $3 \times 2 \times 1 = 6$ plus
$1 \times 2 \times 1 = 2$: $8$ kubusjes.
\end{solution}

\begin{solution}{exo:g5:solids:7}
Lagen van $12$: bijvoorbeeld $4 \times 3$ of $6 \times 2$ (of
$12 \times 1$). Met $3$ lagen is het \omterm{def:g5:solids:volume}{volume} in elk geval
$12 \times 3 = 36$ kubusjes.
\end{solution}

\begin{solution}{exo:g5:solids:8}
$27 = 3 \times 3 \times 3$: een \omterm{def:g5:solids:def}{kubus} met zijde $3$.
$64 = 4 \times 4 \times 4$: een \omterm{def:g5:solids:def}{kubus} met zijde $4$.
\end{solution}

\begin{solution}{exo:g5:solids:9}
In de doos passen $10 \times 5 \times 4 = 200$ suikerklontjes. Bij $6$ per
dag: $200 = 6 \times 33 + 2$, dus doet de doos $33$ volle dagen (en blijven er
$2$ klontjes over voor de ochtend van dag $34$).
\end{solution}

\begin{solution}{exo:g5:solids:10}
Verf op $3$ \omterm{def:g5:solids:def}{zijvlakken}: de $8$ hoekkubusjes. Op $2$ \omterm{def:g5:solids:def}{zijvlakken}: de middens van
de $12$ \omterm{def:g5:solids:def}{ribben}, dus $12$ kubusjes. Op $1$ \omterm{def:g5:solids:def}{zijvlak}: de middens van de $6$
\omterm{def:g5:solids:def}{zijvlakken}, dus $6$ kubusjes. Op geen enkel: het ene verborgen kubusje in het
midden, dus $1$. Controle: $8 + 12 + 6 + 1 = 27$. \checkmark
\end{solution}
