\section*{الفصل \ref{ch:g5:areas} --- المحيط والمساحة}

\begin{solution}{exo:g5:areas:1}
\omterm{def:g4:measure:area}{المساحة}: $8 \times 5 = 40$ سم$^2$. \omterm{def:g4:measure:perimeter}{والمحيط}:
$2 \times (8 + 5) = 26$ سم.
\end{solution}

\begin{solution}{exo:g5:areas:2}
$9 \times 9 = 81$ سم$^2$؛ \quad $12 \times 7 = 84$ م$^2$؛ \quad
$4.5 \times 6 = 27$ سم$^2$.
\end{solution}

\begin{solution}{exo:g5:areas:3}
العرض: $63 \div 9 = 7$ سم. \omterm{def:g4:measure:perimeter}{والمحيط}: $2 \times (9 + 7) = 32$ سم.
\end{solution}

\begin{solution}{exo:g5:areas:4}
مثلًا $6 \times 4$ (\omterm{def:g4:measure:perimeter}{المحيط} $20$) و $8 \times 3$
(\omterm{def:g4:measure:perimeter}{المحيط} $22$): \omterm{def:g4:measure:area}{المساحة} نفسها $24$، والمحيطان مختلفان.
\end{solution}

\begin{solution}{exo:g5:areas:5}
$3$ م$^2 = 30\,000$ سم$^2$؛ \quad $50\,000$ سم$^2 = 5$ م$^2$.
\end{solution}

\begin{solution}{exo:g5:areas:6}
الشريط: $8 \times 2 = 16$ سم$^2$؛ والساق: $2 \times 5 = 10$ سم$^2$؛
\omterm{def:g1:addition:def}{والمجموع} $26$ سم$^2$.
\end{solution}

\begin{solution}{exo:g5:areas:7}
$30 \times 20 - 24 \times 14 = 600 - 336 = 264$ سم$^2$ من الخشب.
\end{solution}

\begin{solution}{exo:g5:areas:8}
نصف المستطيل $8 \times 6$: $48 \div 2 = 24$ سم$^2$.
\end{solution}

\begin{solution}{exo:g5:areas:9}
البذور: \omterm{def:g4:measure:area}{المساحة} $7 \times 4 = 28$ م$^2$، والثمن $28 \times 2 = 56$.
والسياج: \omterm{def:g4:measure:perimeter}{المحيط} $2 \times (7 + 4) = 22$ م، والثمن
$22 \times 3 = 66$.
\end{solution}

\begin{solution}{exo:g5:areas:10}
بلاطتان من $50$ سم تصنعان مترًا: فعلى امتداد الطول $12$ م،
$24$ بلاطة؛ وعلى امتداد العرض $2$ م، $4$ بلاطات. \omterm{def:g1:addition:def}{والمجموع}:
$24 \times 4 = 96$ بلاطة.
\end{solution}

\begin{solution}{exo:g5:areas:11}
قبل: \omterm{def:g4:measure:perimeter}{المحيط} $2 \times (4 + 3) = 14$ سم، \omterm{def:g4:measure:area}{والمساحة} $12$ سم$^2$.
وبعد (والضلعان $8$ و $6$): \omterm{def:g4:measure:perimeter}{المحيط} $28$ سم، \omterm{def:g4:measure:area}{والمساحة} $48$ سم$^2$.
\omterm{def:g4:measure:perimeter}{فالمحيط} \emph{تضاعف}؛ أما \omterm{def:g4:measure:area}{المساحة} فضُربت في
\emph{أربعة} ($2 \times 2$): فالأطوال والمساحات لا تتغير
بالمقياس نفسه.
\end{solution}
