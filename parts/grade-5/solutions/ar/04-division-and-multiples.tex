\section*{الفصل \ref{ch:g5:division} --- القسمة والمضاعفات}

\begin{solution}{exo:g5:division:1}
$851 \div 23$: \omterm{def:g5:division:multiple}{مضاعفات} $23$: $23, 46, 69, 92, 115, 138, 161,
184, 207$. ثم $85 \div 23$: $3$ مرات ($69$)، \omterm{def:g3:division:remainder}{والباقي} $16$؛
وأنزِل $1$: فيصير $161 \div 23 = 7$ بالضبط. \omterm{def:g3:division:remainder}{خارج القسمة} $37$،
\omterm{def:g3:division:remainder}{والباقي} $0$ ($23 \times 37 = 851$).

$704 \div 32$: و $70 \div 32$: $2$ مرتان ($64$)، \omterm{def:g3:division:remainder}{والباقي} $6$؛ وأنزِل
$4$: فيصير $64 \div 32 = 2$. \omterm{def:g3:division:remainder}{خارج القسمة} $22$، \omterm{def:g3:division:remainder}{والباقي} $0$.
\end{solution}

\begin{solution}{exo:g5:division:2}
$9 \div 2 = 4.5$؛ \quad $27 \div 4 = 6.75$؛ \quad
$33 \div 5 = 6.6$؛ \quad $21 \div 8 = 2.625$.
\end{solution}

\begin{solution}{exo:g5:division:3}
$54 \div 4 = 13.5$: يدفع كل واحد $13.50$.
\end{solution}

\begin{solution}{exo:g5:division:4}
$10 \div 3 = 3.333\dots$: فالرقم $3$ يتكرر بلا نهاية ---
والقسمة لا تتوقف أبدًا. ومقرَّبًا إلى الجزء من مئة: $3.33$.
\end{solution}

\begin{solution}{exo:g5:division:5}
\omterm{def:g5:division:multiple}{مضاعفات} $7$ حتى $70$: $7$، $14$، $21$، $28$، $35$، $42$، $49$،
$56$، $63$، $70$. وقواسم $18$: $1$، $2$، $3$، $6$، $9$، $18$.
وقواسم $24$: $1$، $2$، $3$، $4$، $6$، $8$، $12$، $24$.
\end{solution}

\begin{solution}{exo:g5:division:6}
$470$: على $2$ (ينتهي بالرقم $0$)، وعلى $5$، وعلى $10$؛ ومجموع أرقامه $11$:
فليس على $3$.

$735$: ينتهي بالرقم $5$: فعلى $5$، لا على $2$ ولا على $10$؛ \omterm{def:g1:addition:def}{ومجموع}
أرقامه $15$: فعلى $3$.

$8\,001$: فردي؛ ومجموع أرقامه $9$: فعلى $3$ وحده.

$1\,314$: زوجي؛ ومجموع أرقامه $9$: فعلى $2$ وعلى $3$ (ومن ثَمّ على $6$)،
لا على $5$.
\end{solution}

\begin{solution}{exo:g5:division:7}
القابلية للقسمة على $3$ وعلى $5$ تعني القابلية للقسمة على $15$: وبين $60$ و
$80$، العددان $60$ و $75$.
\end{solution}

\begin{solution}{exo:g5:division:8}
$7.2 \div 6 = 1.2$: فطول كل قطعة $1.2$ م (والتحقّق:
$1.2 \times 6 = 7.2$).
\end{solution}

\begin{solution}{exo:g5:division:9}
$1\,000 = 24 \times 41 + 16$: واحد وأربعون كيسًا ممتلئًا، وتبقى $16$
كرة.
\end{solution}

\begin{solution}{exo:g5:division:10}
$7 \div 5 = 1.4$: ينال كل واحد من الأصدقاء $5$ مقدار $1.4$ لوح ---
وذلك ممكن بقطع الألواح أعشارًا. أما $5 \div 7$: فالقسمة
تعطي $0.714285\dots$، وهي لا تنتهي أبدًا --- فلا وجود لنصيب عشري
دقيق؛ ولا يسمّيه بالضبط إلا \omterm{def:g4:fractions:def}{الكسر} $\frac57$.
\end{solution}

\begin{solution}{exo:g5:division:11}
مجموع الأرقام: $5 + 2 + d = 7 + d$. ويكون قابلًا للقسمة على $3$ حين يكون $7 + d$
هو $9$ أو $12$ أو $15$: أي $d = 2$ أو $5$ أو $8$. وأصغرها $d = 2$
(فيعطي $522 = 3 \times 174$).
\end{solution}
