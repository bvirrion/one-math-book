\section*{الفصل \ref{ch:g5:fractions} --- الكسور أعدادًا}

\begin{solution}{exo:g5:fractions:1}
$\frac25$: خطوتان من $0$؛ و $\frac55$: على $1$؛ و $\frac85$: ثلاث
خطوات بعد $1$؛ و $\frac{10}{5}$: على $2$.
\end{solution}

\begin{solution}{exo:g5:fractions:2}
$\frac12 = \frac48$؛ \quad $\frac34 = \frac68$؛ \quad
$\frac25 = \frac{4}{10}$؛ \quad $\frac69 = \frac23$.
\end{solution}

\begin{solution}{exo:g5:fractions:3}
$\frac82 = 4$؛ \quad $\frac94$ ليس صحيحًا (فالعدد $9$ ليس مضاعفًا
للعدد $4$)؛ \quad $\frac{15}{3} = 5$؛ \quad $\frac{20}{5} = 4$.
\end{solution}

\begin{solution}{exo:g5:fractions:4}
$0.7$؛ \quad $0.41$؛ \quad $0.5$؛ \quad $0.75$؛ \quad $0.2$.
\end{solution}

\begin{solution}{exo:g5:fractions:5}
$0.9 = \frac{9}{10}$؛ \quad $0.13 = \frac{13}{100}$؛ \quad
$2.5 = \frac{25}{10}$؛ \quad $0.75 = \frac{75}{100} = \frac34$.
\end{solution}

\begin{solution}{exo:g5:fractions:6}
$\frac47 < \frac67$ (\omterm{def:g4:fractions:def}{المقام} نفسه). \quad
$\frac35 > \frac37$ (\omterm{def:g4:fractions:def}{البسط} نفسه: فالأخماس أكبر). \quad
$\frac25 < \frac12 < \frac{7}{12}$، إذن $\frac25 < \frac{7}{12}$
(بمَعلَم $\frac12$). \quad
$\frac98 > 1 > \frac{11}{12}$ (بمَعلَم $1$).
\end{solution}

\begin{solution}{exo:g5:fractions:7}
بالأعداد العشرية: $0.5$؛ $0.4$؛ $0.75$؛ $0.6$؛ $0.2$. والترتيب:
\[
\frac15 < 0.4 < \frac12 < 0.6 < \frac34 .
\]
\end{solution}

\begin{solution}{exo:g5:fractions:8}
الآلة: $\frac14$ من $28 = 7$. والرياضة: $\frac12$ من $28 = 14$.
فمجموعة الرياضة أكبر.
\end{solution}

\begin{solution}{exo:g5:fractions:9}
$\frac14$ سا $= 15$ د؛ و $\frac12$ سا $= 30$ د؛ و $\frac34$ سا $= 45$
د؛ و $\frac{1}{60}$ سا $= 1$ د.
\end{solution}

\begin{solution}{exo:g5:fractions:10}
عمر: $\frac35$ من $50 = 30$ سل. وآية: $\frac12$ من $60 = 30$ سل.
فقد شربا المقدار نفسه! والكسور المجرّدة ($\frac35 > \frac12$)
تقارن حصصًا من قنينتين \emph{مختلفتين} --- ولا يمكن أن تُقارن إلا
المقادير الفعلية.
\end{solution}

\begin{solution}{exo:g5:fractions:11}
بين $\frac12$ و $1$: مثلًا $\frac34$ (أي $0.75$).
وبين $\frac12 = 0.5$ و $\frac34 = 0.75$: مثلًا $0.6 =
\frac35$.
\end{solution}
