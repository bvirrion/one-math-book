\section*{Bab \ref{ch:g5:fractions} --- Pecahan sebagai Bilangan}

\begin{solution}{exo:g5:fractions:1}
$\frac25$: dua langkah dari $0$; $\frac55$: tepat di $1$;
$\frac85$: tiga langkah melewati $1$; $\frac{10}{5}$: tepat di $2$.
\end{solution}

\begin{solution}{exo:g5:fractions:2}
$\frac12 = \frac48$; \quad $\frac34 = \frac68$; \quad
$\frac25 = \frac{4}{10}$; \quad $\frac69 = \frac23$.
\end{solution}

\begin{solution}{exo:g5:fractions:3}
$\frac82 = 4$; \quad $\frac94$ bukan bilangan bulat ($9$ bukan
kelipatan $4$); \quad $\frac{15}{3} = 5$; \quad $\frac{20}{5} = 4$.
\end{solution}

\begin{solution}{exo:g5:fractions:4}
$0.7$; \quad $0.41$; \quad $0.5$; \quad $0.75$; \quad $0.2$.
\end{solution}

\begin{solution}{exo:g5:fractions:5}
$0.9 = \frac{9}{10}$; \quad $0.13 = \frac{13}{100}$; \quad
$2.5 = \frac{25}{10}$; \quad $0.75 = \frac{75}{100} = \frac34$.
\end{solution}

\begin{solution}{exo:g5:fractions:6}
$\frac47 < \frac67$ (\omterm{def:g4:fractions:def}{penyebutnya} sama). \quad
$\frac35 > \frac37$ (\omterm{def:g4:fractions:def}{pembilangnya} sama: perlimaan lebih besar). \quad
$\frac25 < \frac12 < \frac{7}{12}$, jadi $\frac25 < \frac{7}{12}$
(patokan $\frac12$). \quad
$\frac98 > 1 > \frac{11}{12}$ (patokan $1$).
\end{solution}

\begin{solution}{exo:g5:fractions:7}
Nama desimalnya: $0.5$; $0.4$; $0.75$; $0.6$; $0.2$. Urutannya:
\[
\frac15 < 0.4 < \frac12 < 0.6 < \frac34 .
\]
\end{solution}

\begin{solution}{exo:g5:fractions:8}
Alat musik: $\frac14$ dari $28 = 7$. Olahraga: $\frac12$ dari
$28 = 14$. Kelompok olahraganya lebih besar.
\end{solution}

\begin{solution}{exo:g5:fractions:9}
$\frac14$ jam $= 15$ menit; $\frac12$ jam $= 30$ menit; $\frac34$ jam
$= 45$ menit; $\frac{1}{60}$ jam $= 1$ menit.
\end{solution}

\begin{solution}{exo:g5:fractions:10}
Tono: $\frac35$ dari $50 = 30$ cL. Ani: $\frac12$ dari $60 = 30$ cL.
Keduanya minum sama banyak! \omterm{def:g4:fractions:def}{Pecahannya} sendiri
($\frac35 > \frac12$) membandingkan bagian dari botol yang
\emph{berbeda} --- hanya takaran sesungguhnya yang dapat dibandingkan.
\end{solution}

\begin{solution}{exo:g5:fractions:11}
Antara $\frac12$ dan $1$: misalnya $\frac34$ (yaitu $0.75$). Antara
$\frac12 = 0.5$ dan $\frac34 = 0.75$: misalnya $0.6 =
\frac35$.

\end{solution}
