\section*{Bab \ref{ch:g5:decimals} --- Bilangan Desimal}

\begin{solution}{exo:g5:decimals:1}
$5.207$; \quad $0.85$; \quad $4.507$; \quad $12.009$.
\end{solution}

\begin{solution}{exo:g5:decimals:2}
Angka $4$ membilang persepuluhan, angka $8$ membilang perseribuan:
\[
27.418 = 27 + \frac{4}{10} + \frac{1}{100} + \frac{8}{1000}.
\]
\end{solution}

\begin{solution}{exo:g5:decimals:3}
$3.60 = 3.6$: benar (\omterm{def:g1:counting:zero}{nol} di ujung tidak menambah apa pun).

$0.5 = 0.05$: salah --- $5$ persepuluhan melawan $5$ perseratusan.

$2.070 = 2.07$: benar.

$1.3 = 1.03$: salah --- $3$ persepuluhan melawan $3$ perseratusan.
\end{solution}

\begin{solution}{exo:g5:decimals:4}
$4.8 > 4.79$; \quad $0.302 < 0.32$ (persepuluhannya sama,
perseratusannya $0 < 2$); \quad $6.5 = 6.500$; \quad $9.09 < 9.1$.
\end{solution}

\begin{solution}{exo:g5:decimals:5}
$3.041 < 3.104 < 3.14 < 3.4 < 3.41$.
\end{solution}

\begin{solution}{exo:g5:decimals:6}
$6.83$: antara $6$ dan $7$; antara $6.8$ dan $6.9$.

$0.492$: antara $0$ dan $1$; antara $0.4$ dan $0.5$.

$12.06$: antara $12$ dan $13$; antara $12.0$ dan $12.1$.
\end{solution}

\begin{solution}{exo:g5:decimals:7}
$8.276$: ke satuan $8$; ke persepuluhan $8.3$; ke perseratusan
$8.28$. Dan $3.95$ ke persepuluhan: angka perseratusannya $5$, jadi
\omterm{def:g4:numbers:round}{bulatkan} ke atas: $4.0$.
\end{solution}

\begin{solution}{exo:g5:decimals:8}
Misalnya $2.65$; \quad $0.415$; \quad $5.995$. (Ada tak hingga
banyak jawaban setiap kali.)
\end{solution}

\begin{solution}{exo:g5:decimals:9}
$0.85 < 0.9 < 1.02 < 1.2$ (dalam meter).
\end{solution}

\begin{solution}{exo:g5:decimals:10}
Bandingkan persepuluhannya dulu: $7.12$ punya $1$ persepuluhan,
$7.9$ punya $9$ persepuluhan, jadi $7.12 < 7.9$. Lusi membandingkan
bagian desimalnya sebagai bilangan bulat ($12$ lawan $9$), lupa
bahwa $12$ perseratusan jauh lebih sedikit daripada $90$ perseratusan.
\end{solution}

\begin{solution}{exo:g5:decimals:11}
\omterm{def:g4:numbers:round}{Pembulatan} ke persepuluhan memberi $6.4$ untuk bilangan dua angka
desimal dari $6.35$ (tepat di tengah \omterm{def:g4:numbers:round}{dibulatkan} naik) sampai $6.44$:
yaitu $6.35$, $6.36$, \dots, $6.44$. Terkecil: $6.35$; terbesar: $6.44$.

\end{solution}
