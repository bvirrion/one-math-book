\section*{Capítulo \ref{ch:g5:division} --- Divisão e múltiplos}

\begin{solution}{exo:g5:division:1}
$851 \div 23$: \omterm{def:g5:division:multiple}{múltiplos} de $23$: $23, 46, 69, 92, 115, 138, 161,
184, 207$. Então $85 \div 23$: $3$ vezes ($69$), \omterm{def:g3:division:remainder}{resto} $16$; baixe o
$1$: $161 \div 23 = 7$ exatamente. \omterm{def:g3:division:remainder}{Quociente} $37$, \omterm{def:g3:division:remainder}{resto} $0$
($23 \times 37 = 851$).

$704 \div 32$: $70 \div 32$: $2$ vezes ($64$), \omterm{def:g3:division:remainder}{resto} $6$; baixe o
$4$: $64 \div 32 = 2$. \omterm{def:g3:division:remainder}{Quociente} $22$, \omterm{def:g3:division:remainder}{resto} $0$.
\end{solution}

\begin{solution}{exo:g5:division:2}
$9 \div 2 = 4.5$; \quad $27 \div 4 = 6.75$; \quad
$33 \div 5 = 6.6$; \quad $21 \div 8 = 2.625$.
\end{solution}

\begin{solution}{exo:g5:division:3}
$54 \div 4 = 13.5$: cada um paga $13.50$.
\end{solution}

\begin{solution}{exo:g5:division:4}
$10 \div 3 = 3.333\dots$: o algarismo $3$ se repete para sempre --- a
divisão nunca acaba. Arredondado ao centésimo: $3.33$.
\end{solution}

\begin{solution}{exo:g5:division:5}
\omterm{def:g5:division:multiple}{Múltiplos} de $7$ até $70$: $7, 14, 21, 28, 35, 42, 49, 56, 63, 70$.
\omterm{def:g5:division:multiple}{Divisores} de $18$: $1, 2, 3, 6, 9, 18$. \omterm{def:g5:division:multiple}{Divisores} de $24$:
$1, 2, 3, 4, 6, 8, 12, 24$.
\end{solution}

\begin{solution}{exo:g5:division:6}
$470$: por $2$ (termina em $0$), por $5$ e por $10$; \omterm{def:g1:addition:def}{soma} dos
algarismos $11$: não por $3$.

$735$: termina em $5$: por $5$, nem por $2$ nem por $10$; \omterm{def:g1:addition:def}{soma} dos
algarismos $15$: por $3$.

$8\,001$: \omterm{def:g3:numbers:evenodd}{ímpar}; \omterm{def:g1:addition:def}{soma} dos algarismos $9$: só por $3$.

$1\,314$: par; \omterm{def:g1:addition:def}{soma} dos algarismos $9$: por $2$ e por $3$ (logo por
$6$), não por $5$.
\end{solution}

\begin{solution}{exo:g5:division:7}
Divisível por $3$ e por $5$ significa divisível por $15$: entre $60$
e $80$, os números $60$ e $75$.
\end{solution}

\begin{solution}{exo:g5:division:8}
$7.2 \div 6 = 1.2$: cada pedaço mede $1.2$ m (confere:
$1.2 \times 6 = 7.2$).
\end{solution}

\begin{solution}{exo:g5:division:9}
$1\,000 = 24 \times 41 + 16$: quarenta e um saquinhos cheios e
sobram $16$ bolinhas.
\end{solution}

\begin{solution}{exo:g5:division:10}
$7 \div 5 = 1.4$: cada um dos $5$ amigos fica com $1.4$ barra ---
possível cortando as barras em décimos. Para $5 \div 7$: a divisão dá
$0.714285\dots$, que nunca acaba --- não existe partilha decimal
exata; só a \omterm{def:g4:fractions:def}{fração} $\frac57$ a nomeia com exatidão.
\end{solution}

\begin{solution}{exo:g5:division:11}
\omterm{def:g1:addition:def}{Soma} dos algarismos: $5 + 2 + d = 7 + d$. É divisível por $3$ quando
$7 + d$ vale $9$, $12$ ou $15$: $d = 2$, $5$ ou $8$. O menor é
$d = 2$ (dando $522 = 3 \times 174$).
\end{solution}
