\section*{Capítulo \ref{ch:g5:solids} --- Cubos e blocos retangulares}

\begin{solution}{exo:g5:solids:1}
Por exemplo: um dado (\omterm{def:g5:solids:def}{cubo}); uma caixa de cereal (\omterm{def:g5:solids:def}{bloco retangular});
uma lata (\omterm{def:g5:solids:def}{cilindro}); uma bola de futebol (bola); certos telhados ou
os monumentos do Egito (\omterm{def:g5:solids:def}{pirâmide}).
\end{solution}

\begin{solution}{exo:g5:solids:2}
Um \omterm{def:g5:solids:def}{bloco retangular} tem $6$ \omterm{def:g5:solids:def}{faces}, $12$ \omterm{def:g5:solids:def}{arestas} e $8$ \omterm{def:g5:solids:def}{vértices} --- e
$6 + 8 = 12 + 2$. \checkmark
\end{solution}

\begin{solution}{exo:g5:solids:3}
As seis \omterm{def:g5:solids:def}{faces} vêm em três pares de \omterm{def:g5:solids:def}{faces} opostas idênticas
(cima--baixo, frente--fundo, esquerda--direita). Num \omterm{def:g5:solids:def}{cubo}, as seis
\omterm{def:g5:solids:def}{faces} são quadrados idênticos.
\end{solution}

\begin{solution}{exo:g5:solids:4}
\omterm{def:g5:solids:net}{Planificação} em cruz: quatro quadrados de $2$ cm numa fileira, um
acima e um abaixo do segundo quadrado. Dobrar o anel de quatro forma
as laterais; os dois quadrados extras fecham o topo e o fundo.
\end{solution}

\begin{solution}{exo:g5:solids:5}
Na \omterm{def:g5:solids:net}{planificação} em cruz (quatro quadrados numa fileira, um acima e um
abaixo do segundo), os pares opostos são: os quadrados $1$ e $3$ da
fileira; os quadrados $2$ e $4$ da fileira; o quadrado de cima e o de
baixo. Uma numeração correta: fileira $= 1, 2, 6, 5$; cima $= 3$;
baixo $= 4$ --- aí $1{+}6$, $2{+}5$ e $3{+}4$ dão todos $7$.
\end{solution}

\begin{solution}{exo:g5:solids:6}
Bloco: $5 \times 2 = 10$ por camada, $3$ camadas: $30$ cubinhos.
\omterm{def:g5:solids:def}{Cubo}: $3 \times 3 \times 3 = 27$. Pilha em L:
$3 \times 2 \times 1 = 6$ mais $1 \times 2 \times 1 = 2$: $8$
cubinhos.
\end{solution}

\begin{solution}{exo:g5:solids:7}
Camadas de $12$: por exemplo $4 \times 3$ ou $6 \times 2$ (ou
$12 \times 1$). Com $3$ camadas, o \omterm{def:g5:solids:volume}{volume} é $12 \times 3 = 36$
cubinhos em qualquer caso.
\end{solution}

\begin{solution}{exo:g5:solids:8}
$27 = 3 \times 3 \times 3$: um \omterm{def:g5:solids:def}{cubo} de lado $3$.
$64 = 4 \times 4 \times 4$: um \omterm{def:g5:solids:def}{cubo} de lado $4$.
\end{solution}

\begin{solution}{exo:g5:solids:9}
A caixa leva $10 \times 5 \times 4 = 200$ torrões de açúcar. A $6$
por dia: $200 = 6 \times 33 + 2$, então a caixa dura $33$ dias
completos (e sobram $2$ torrões para a manhã do dia $34$).
\end{solution}

\begin{solution}{exo:g5:solids:10}
Tinta em $3$ \omterm{def:g5:solids:def}{faces}: os $8$ cubinhos dos cantos. Em $2$ \omterm{def:g5:solids:def}{faces}: os do
meio das $12$ \omterm{def:g5:solids:def}{arestas}: $12$ cubinhos. Em $1$ \omterm{def:g5:solids:def}{face}: os do centro das
$6$ \omterm{def:g5:solids:def}{faces}: $6$ cubinhos. Em nenhuma: o único cubinho escondido no
centro: $1$. Confere: $8 + 12 + 6 + 1 = 27$. \checkmark
\end{solution}
