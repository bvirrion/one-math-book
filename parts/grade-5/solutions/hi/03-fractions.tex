\section*{अध्याय \ref{ch:g5:fractions} --- संख्या के रूप में भिन्नें}

\begin{solution}{exo:g5:fractions:1}
$\frac25$: $0$ से दो कदम; $\frac55$: $1$ पर; $\frac85$: $1$ से
तीन कदम आगे; $\frac{10}{5}$: $2$ पर।
\end{solution}

\begin{solution}{exo:g5:fractions:2}
$\frac12 = \frac48$; \quad $\frac34 = \frac68$; \quad
$\frac25 = \frac{4}{10}$; \quad $\frac69 = \frac23$।
\end{solution}

\begin{solution}{exo:g5:fractions:3}
$\frac82 = 4$; \quad $\frac94$ पूर्ण नहीं है ($9$, $4$ का गुणज
नहीं); \quad $\frac{15}{3} = 5$; \quad $\frac{20}{5} = 4$।
\end{solution}

\begin{solution}{exo:g5:fractions:4}
$0.7$; \quad $0.41$; \quad $0.5$; \quad $0.75$; \quad $0.2$।
\end{solution}

\begin{solution}{exo:g5:fractions:5}
$0.9 = \frac{9}{10}$; \quad $0.13 = \frac{13}{100}$; \quad
$2.5 = \frac{25}{10}$; \quad $0.75 = \frac{75}{100} = \frac34$।
\end{solution}

\begin{solution}{exo:g5:fractions:6}
$\frac47 < \frac67$ (हर एक जैसा)। \quad
$\frac35 > \frac37$ (\omterm{def:g4:fractions:def}{अंश} एक जैसा: पाँचवें भाग बड़े होते हैं)। \quad
$\frac25 < \frac12 < \frac{7}{12}$, इसलिए $\frac25 < \frac{7}{12}$
(ठिकाना $\frac12$)। \quad
$\frac98 > 1 > \frac{11}{12}$ (ठिकाना $1$)।
\end{solution}

\begin{solution}{exo:g5:fractions:7}
दशमलव: $0.5$; $0.4$; $0.75$; $0.6$; $0.2$। क्रम:
\[
\frac15 < 0.4 < \frac12 < 0.6 < \frac34 .
\]
\end{solution}

\begin{solution}{exo:g5:fractions:8}
वाद्य: $28$ का $\frac14 = 7$। खेल: $28$ का $\frac12 = 14$।
खेल वाला समूह बड़ा है।
\end{solution}

\begin{solution}{exo:g5:fractions:9}
$\frac14$ घंटा $= 15$ मिनट; $\frac12$ घंटा $= 30$ मिनट;
$\frac34$ घंटा $= 45$ मिनट; $\frac{1}{60}$ घंटा $= 1$ मिनट।
\end{solution}

\begin{solution}{exo:g5:fractions:10}
टॉम: $50$ का $\frac35 = 30$ cL। अना: $60$ का $\frac12 = 30$ cL।
दोनों ने बराबर पिया! सिर्फ़ भिन्नें ($\frac35 > \frac12$)
\emph{अलग-अलग} बोतलों के हिस्सों की तुलना करती हैं --- तुलना तो
असली मात्राओं की ही हो सकती है।
\end{solution}

\begin{solution}{exo:g5:fractions:11}
$\frac12$ और $1$ के बीच: जैसे $\frac34$ (यानी $0.75$)।
$\frac12 = 0.5$ और $\frac34 = 0.75$ के बीच: जैसे $0.6 =
\frac35$।

\end{solution}
