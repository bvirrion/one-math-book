\section*{अध्याय \ref{ch:g5:decimalops} --- दशमलव संख्याओं की गणना}

\begin{solution}{exo:g5:decimalops:1}
$45.7 + 8.93 = 54.63$; \quad
$16.25 + 3.75 = 20$; \quad
$0.86 + 12.4 = 13.26$।
\end{solution}

\begin{solution}{exo:g5:decimalops:2}
$9.4 - 2.72 = 6.68$ (जाँच: $6.68 + 2.72 = 9.4$)।

$20 - 13.45 = 6.55$ (जाँच: $6.55 + 13.45 = 20$)।

$6.03 - 5.9 = 0.13$।
\end{solution}

\begin{solution}{exo:g5:decimalops:3}
$6.2 \to 10$: $3.8$। \quad $3.75 \to 10$: $6.25$। \quad
$9.99 \to 10$: $0.01$।
\end{solution}

\begin{solution}{exo:g5:decimalops:4}
$72.4$; \quad $58$; \quad $3.45$; \quad $0.07$; \quad $300$।
\end{solution}

\begin{solution}{exo:g5:decimalops:5}
$4.05$ m $= 405$ cm; \quad $325$ cm $= 3.25$ m; \quad
$0.6$ kg $= 600$ g; \quad $85$ cL $= 0.85$ L।
\end{solution}

\begin{solution}{exo:g5:decimalops:6}
$6.2 \times 4$: अंदाज़ा $24$; ठीक-ठीक $24.8$।

$3.45 \times 8$: अंदाज़ा $28$ ($3.5 \times 8$); ठीक-ठीक $27.6$।

$12.5 \times 6$: अंदाज़ा $75$; ठीक-ठीक $75$।
\end{solution}

\begin{solution}{exo:g5:decimalops:7}
$7$ चक्कर: $0.4 \times 7 = 2.8$ km। $25$ चक्कर:
$0.4 \times 25 = 10$ km।
\end{solution}

\begin{solution}{exo:g5:decimalops:8}
रोटियाँ: $1.15 \times 2 = 2.30$। कुल: $2.30 + 12.60 = 14.90$।
बाक़ी: $20 - 14.90 = 5.10$।
\end{solution}

\begin{solution}{exo:g5:decimalops:9}
पैकेट: $1.5 \times 6 = 9$ L। दिन: $0.75 + 0.75 = 1.5$ L हर दो
दिन में, इसलिए $9$ L, $12$ दिन चलते हैं ($0.75$ की बारह ख़ुराकें:
$12 \times 0.75 = 9$)।
\end{solution}

\begin{solution}{exo:g5:decimalops:10}
अंदाज़ा: $5.6$, $5.5$ से ज़्यादा है, और $5.5 \times 3 = 16.5$ ---
इसलिए \omterm{def:g2:mult:def}{गुणनफल} $16.5$ से \emph{ज़्यादा} होना चाहिए, और $15.18$ बहुत
कम है। सही ढंग से: $56 \times 3 = 168$, एक दशमलव अंक, इसलिए
$5.6 \times 3 = 16.8$। मिलो ने अपने दोनों आंशिक \omterm{def:g2:mult:def}{गुणनफल} ($15$ और
$18$) उनके स्थानों के साथ जोड़ने के बजाय अगल-बगल चिपका दिए:
$15 + 1.8 = 16.8$।
\end{solution}

\begin{solution}{exo:g5:decimalops:11}

\emph{2.} $261.4 > 260$: उसने कीर्तिमान नहीं तोड़ा; वह उससे
$1.4$ s पीछे रही।
\end{solution}
