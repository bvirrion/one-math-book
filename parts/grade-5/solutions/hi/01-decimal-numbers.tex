\section*{अध्याय \ref{ch:g5:decimals} --- दशमलव संख्याएँ}

\begin{solution}{exo:g5:decimals:1}
$5.207$; \quad $0.85$; \quad $4.507$; \quad $12.009$।
\end{solution}

\begin{solution}{exo:g5:decimals:2}
$4$ दसवें भाग गिनता है, $8$ हज़ारवें भाग:
\[
27.418 = 27 + \frac{4}{10} + \frac{1}{100} + \frac{8}{1000}.
\]
\end{solution}

\begin{solution}{exo:g5:decimals:3}
$3.60 = 3.6$: सही (अंत का \omterm{def:g1:counting:zero}{शून्य} कुछ नहीं जोड़ता)।

$0.5 = 0.05$: ग़लत --- $5$ दसवें भाग बनाम $5$ सौवें भाग।

$2.070 = 2.07$: सही।

$1.3 = 1.03$: ग़लत --- $3$ दसवें भाग बनाम $3$ सौवें भाग।
\end{solution}

\begin{solution}{exo:g5:decimals:4}
$4.8 > 4.79$; \quad $0.302 < 0.32$ (दसवें भाग बराबर, सौवें भाग
$0 < 2$); \quad $6.5 = 6.500$; \quad $9.09 < 9.1$।
\end{solution}

\begin{solution}{exo:g5:decimals:5}

\end{solution}

\begin{solution}{exo:g5:decimals:6}
$6.83$: $6$ और $7$ के बीच; $6.8$ और $6.9$ के बीच।

$0.492$: $0$ और $1$ के बीच; $0.4$ और $0.5$ के बीच।

$12.06$: $12$ और $13$ के बीच; $12.0$ और $12.1$ के बीच।
\end{solution}

\begin{solution}{exo:g5:decimals:7}
$8.276$: इकाई तक $8$; दसवें भाग तक $8.3$; सौवें भाग तक
$8.28$। और $3.95$ दसवें भाग तक: सौवें भाग का अंक $5$ है, ऊपर
गोल करो: $4.0$।
\end{solution}

\begin{solution}{exo:g5:decimals:10}
पहले दसवें भाग की तुलना करो: $7.12$ में $1$ दसवाँ भाग है, $7.9$
में $9$, इसलिए $7.12 < 7.9$। लूसी ने दशमलव भागों की तुलना पूर्ण
संख्याओं की तरह की ($12$ बनाम $9$), यह भूलकर कि $12$ सौवें भाग
$90$ सौवें भागों से कहीं कम हैं।
\end{solution}

\begin{solution}{exo:g5:decimals:11}
दसवें भाग तक गोल करने पर $6.35$ (ठीक बीच वाली ऊपर जाती है) से
$6.44$ तक की दो दशमलव अंकों वाली सारी संख्याएँ $6.4$ देती हैं:
$6.35$, $6.36$, \dots, $6.44$। सबसे छोटी: $6.35$; सबसे बड़ी: $6.44$।
\end{solution}
