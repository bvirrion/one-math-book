\section{En expansión}

\begin{definition}[En expansión]\label{def:g9:algebra:expand}
\emph{En expansión}\index{En expansión} significa transformar un \omterm{def:g2:mult:def}{producto} en un
\omterm{def:g1:addition:def}{suma}, usando distributividad:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g9:algebra:expand}
Cada término del primer paréntesis multiplica cada término del segundo:
\begin{align*}
(x + 3)(2x + 5)
&= x \times 2x + x \times 5 + 3 \times 2x + 3 \times 5 \\
&= 2x^2 + 5x + 6x + 15 \\
&= 2x^2 + 11x + 15 .
\end{align*}
Los signos viajan con los términos:
\begin{align*}
(3x - 2)(x - 4)
&= 3x^2 - 12x - 2x + 8
= 3x^2 - 14x + 8 .
\end{align*}
\end{example}

\begin{theorem}[Las tres identidades]\label{thm:g9:algebra:identities}
Para todos los números $a $ y $ b $:
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
Expande cada lado izquierdo. Para el primero:
$(a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2$. Para el segundo,
reemplazar $ b $ por $-b $ en el primero. Para el tercero:
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$ (los términos cruzados se cancelan).
\end{proof}

\[

\]

$98 \times 102 = (100-2)(100+2) = 10000 - 4 = 9996$.

\section{Factorización}

\begin{method}[Factorización]\label{met:g9:algebra:factor}
Para factorizar una expresión, intente en orden:
\begin{enumerate}
  \item \emph{factor común}: $ ka + kb = k(a + b)$, dónde $ k $ puede en sí
        ser un soporte;
  \item \emph{\omterm{ex:g1:subtraction:difference}{diferencia} de cuadrados}: $ a^2 - b^2 = (a+b)(a-b)$;
  \item \emph{cuadrado perfecto}: $ a^2 + 2ab + b^2 = (a+b)^2$ (y con
        $-2ab $, $(a-b)^2$).
\end{enumerate}
Siempre consulte por \omterm{def:g9:algebra:expand}{en expansión} el resultado.
\end{method}

\begin{example}\label{ex:g9:algebra:factor}
\emph{Factor común:}
$15x^2 + 10x = 5x \times 3x + 5x \times 2 = 5x(3x + 2)$.

\emph{Soporte común:}
\[
(x+2)(x-1) + (x+2)(3x+4) = (x+2)\bigl[(x-1) + (3x+4)\bigr]
= (x+2)(4x+3).
\]

\emph{\omterm{ex:g1:subtraction:difference}{Diferencia} de cuadrados:}
$ x^2 - 81 = (x+9)(x-9)$, y
\[
(2x+1)^2 - 9 = (2x+1)^2 - 3^2 = (2x+1+3)(2x+1-3) = (2x+4)(2x-2).
\]
Cada factor todavía tiene un factor común: $(2x+4)(2x-2) = 2(x+2) \times
2(x-1) = 4(x+2)(x-1)$.
\end{example}

\section{Ecuaciones}

\begin{method}[Ecuaciones de primer grado]\label{met:g9:algebra:firstdegree}
para resolver un \omterm{def:g8:equations:def}{ecuación} como $5x - 3 = 2x + 9$:
\begin{enumerate}
  \item reunir el $ x $-los términos por un lado, los números por el otro, por
        añadiendo la misma cantidad a ambos lados;
  \item reducir cada lado;
  \item dividir ambos lados por el coeficiente de $ x $;
  \item comprobar la solución en el original \omterm{def:g8:equations:def}{ecuación}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:algebra:firstdegree}
\begin{align*}
5x - 3 &= 2x + 9 \\
5x - 2x &= 9 + 3 && \text{(add $3 - 2x $ to both sides)}\\
3x &= 12 \\
x &= 4 && \text{(divide by 3).}
\end{align*}
Controlar: $5 \times 4 - 3 = 17$ y $2 \times 4 + 9 = 17$. La solución es
$4$.

Con corchetes, expanda primero: $3(x - 2) = x + 8$ se convierte $3x - 6 = x + 8$,
entonces $2x = 14$ y $ x = 7$.
\end{example}

\begin{theorem}[Regla del producto cero]\label{thm:g9:algebra:zeroproduct}
A \omterm{def:g2:mult:def}{producto} es \omterm{def:g1:counting:zero}{cero} si y sólo si al menos un factor es \omterm{def:g1:counting:zero}{cero}:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ or } \ B = 0 .
\]
\end{theorem}

\begin{proof}
Si un factor es \omterm{def:g1:counting:zero}{cero} el \omterm{def:g2:mult:def}{producto} claramente lo es. Por el contrario si $ A B = 0$ con
$ A \neq 0$, dividiendo ambos lados por $ A $ da $ B = 0$.
\end{proof}

\begin{example}\label{ex:g9:algebra:zeroproduct}
Resolver $(x - 3)(2x + 10) = 0$: cualquiera $ x - 3 = 0$ o $2x + 10 = 0$, entonces las soluciones son $3$ y $-5$.

Resolver $ x^2 - 49 = 0$: factor primero, $(x-7)(x+7) = 0$, soluciones $7$ y
$-7$.

Resolver $ x^2 = 3x $: nunca dividir por $ x $! Mover y factorizar:
$ x^2 - 3x = x(x - 3) = 0$, soluciones $0$ y $3$.
\end{example}

\section{Desigualdades}

\begin{proposition}[Reglas para las desigualdades]\label{prop:g9:algebra:ineq}
Se conserva una desigualdad al sumar el mismo número a ambos lados, y
al multiplicar ambos lados por lo mismo \emph{positivo} número.
Multiplicando ambos lados por un \emph{negativo} El número invierte el
signo de desigualdad.
\end{proposition}

\begin{proof}
Si $ a \leq b $ entonces $ b - a \geq 0$. Añadiendo $ c $: $(b+c) - (a+c) = b - a
\geq 0$, entonces $ a + c \leq b + c $. Multiplicando por $ c > 0$:
$ bc - ac = (b-a)c \geq 0$, entonces $ ac \leq bc $. Multiplicando por $ c < 0$:
$(b-a)c \leq 0$, entonces $ ac \geq bc $: el orden se invierte.
\end{proof}

\begin{example}\label{ex:g9:algebra:ineq}
Resolver $7 - 2x < 15$:
\begin{align*}
7 - 2x &< 15 \\
-2x &< 8 && \text{(subtract 7)} \\
x &> -4 && \text{(divide by $-2$: se invierte el signo).}
\end{align*}
Las soluciones son todos los números mayores que $-4$: en una recta numérica, un
punto hueco en $-4$ y sombreado a la derecha.
\end{example}

\section{Problema: El álgebra de dígitos}

\begin{problem}[{Problema de fin de semana --- Por qué funcionan las reglas de divisibilidad, el truco 1089 desenmascarado y una aritmética mental que parece brujería}]
\label{pb:g9:algebra:1}
La escuela primaria te enseñó \emph{normas} (un número es \omterm{def:g6:wholes:divisible}{divisible} por
$9$ cuando su dígito \omterm{def:g1:addition:def}{suma} es); los magos realizan \emph{engaños}
(todos terminan en $1089$); los mercados saben \emph{atajos}
($35^2 = 1225$, al instante). Todo ello es el mismo secreto: un
número con dígitos $ a $, $ b $, $ c $ es la expresión
$100a + 10b + c $ y el álgebra de este capítulo
(\cref{thm:g9:algebra:identities}, \omterm{def:g9:algebra:expand}{en expansión}, factoring) puede tomar
aparte. Al final de este problema habrás demostrado la
reglas, desenmascaró el truco y aprendió la hechicería.

\medskip
\noindent\textbf{Parte I --- Digits under the microscope.}
\begin{enumerate}
  \item Un número de dos cifras con decenas $a $ y dígitos de unidades
        $ b $ es $10a + b $. Expande y factoriza \omterm{ex:g1:subtraction:difference}{diferencia}
        entre el número y su inversión,
        $(10a + b) - (10b + a)$ y declarar el descubrimiento. comprobar
        esta encendido $72 - 27$.
  \item Demuestre la regla del nueve para números de tres dígitos: verifique
        la identidad
        \[
        100a + 10b + c = 9\,(11a + b) + (a + b + c),
        \]
        y explica por qué muestra que un número y su dígito \omterm{def:g1:addition:def}{suma}
        dejar el \emph{mismo \omterm{def:g3:division:remainder}{resto}} en la división por $9$
        --- entonces uno es \omterm{def:g6:wholes:divisible}{divisible} por $9$ exactamente cuando el otro esta
        (\cref{prop:g9:arith:rules}, ahora demostrado).
  \item Deducir la regla de tres a partir de una misma identidad, en uno
        frase.
  \item ``Echando nueves'', el cheque de los contables: para probar
        $47 \times 86 = 4\,042$, reemplaza cada número por el
        dígito \omterm{def:g1:addition:def}{suma} de su dígito \omterm{def:g1:addition:def}{suma}: $47 \to 11 \to 2$,
        $86 \to 14 \to 5$, y $2 \times 5 = 10 \to 1$; entonces
        $4\,042 \to 10 \to 1$: coherente. Explique por qué una correcta
        \omterm{def:g2:mult:def}{producto} debe pasar la prueba (escribir $47 = 9k + 2$,
        $86 = 9l + 5$ y expandir), y encontrar un resultado incorrecto que
        la prueba no logra detectar (¿por qué se intercambian dígitos?
        ¿Invisible para él?).
  \item Demuestre la regla de once para números de cuatro dígitos: verifique
        \[
        1000a + 100b + 10c + d
        = 11\,(91a + 9b + c) + (- a + b - c + d),
        \]
        y usarlo para decidir si $2\,926$ es \omterm{def:g6:wholes:divisible}{divisible} por
        $11$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- The 1089 trick.}
El truco: piensa en un número de tres dígitos cuyo primer y último
los dígitos difieren por al menos $2$; invertirlo; restar el menor
del más grande; revertir el resultado; suma los dos últimos números.
El mago anuncia: \emph{1089}.
\begin{enumerate}[resume]
  \item Realiza el truco $742$ y en un número de tu
        elección.
  \item Deja que el número tenga dígitos. $ a $, $ b $, $ c $ con $ a > c $.
        Factoriza la primera resta,
        $(100a + 10b + c) - (100c + 10b + a)$ y enumere todos los
        valores que puede tomar cuando $a - c $ corre de $2$ a $9$.
  \item cada uno de esos \omterm{def:g5:division:multiple}{múltiplos} de $99$ tiene un dígito llamativo
        patrón. Escribiendo $ m = a - c $, verificar la identidad
        \[
        99m = 100\,(m - 1) + 90 + (10 - m),
        \]
        y lea los tres dígitos de $99m $.
  \item Agregar $99m $ a su propia reversión, usando las preguntas 8
        dígitos y demostrar que el total es $1089$ --- lo que
        $ m $ es. El truco está muerto; ¡Viva el álgebra!
  \item ¿Por qué el mago requirió el primer y el último dígito?
        diferir al menos en $2$? examinar los casos $ a = c $ y
        $ a - c = 1$ (donde la resta da $99$, que
        debe estar escrito $099$ para que la reversión funcione), y
        Indique la letra pequeña que mantiene el truco honesto.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Sorcery for daily use.}
\begin{enumerate}[resume]
  \item El truco del cumpleaños: "multiplica tu mes de nacimiento por $5$, agregar $7$, duplica, suma tu día de nacimiento, resta $14$''
        Demuestre que el resultado es siempre $10m + d $ --- mes y
        día, legible de un vistazo. (Pruébelo con alguien).
  \item Rediseñas el truco como: "multiplica el mes por $5$, agregar $9$, duplicar, sumar el día''. ¿Qué número debe
        espectador ahora resta al final por lo mismo
        lectura mes-día?
  \item El truco de los dígitos faltantes: el espectador elige cualquier número,
        resta su dígito \omterm{def:g1:addition:def}{suma} (la pregunta 2 dice que el resultado es
        siempre un \omterm{def:g9:arith:divisor}{múltiple} de $9$ --- explica por qué para un
        número de tres dígitos), luego tacha uno \emph{distinto de cero}
        dígito y le lee los dígitos restantes. Explica como
        recuperas el dígito tachado, y por qué $0$ tenía que ser
        excluido.
  \item Brujería de mercado: probar la identidad
        $(10t + 5)^2 = 100\,t(t+1) + 25$, establece la regla para
        elevar al cuadrado cualquier número que termine en $5$ y calcular $45^2$,
        $85^2$ y $105^2$ en tu cabeza.
  \item Final, extendiéndose \cref{exo:g9:algebra:10}: demostrar que
        el \omterm{def:g1:addition:def}{suma} de cinco números enteros consecutivos es siempre un
        \omterm{def:g9:arith:divisor}{múltiple} de $5$, pero que el \omterm{def:g1:addition:def}{suma} de \emph{cuatro}
        enteros consecutivos es \emph{nunca} a \omterm{def:g9:arith:divisor}{múltiple} de $4$.
        (Dos cálculos breves --- y una moraleja sobre lo que
        el álgebra hace con las palabras "siempre" y "nunca".)
\end{enumerate}
\end{problem}
