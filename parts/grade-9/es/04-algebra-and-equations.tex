\chapter{Álgebra y ecuaciones}\label{ch:g9:algebra}

Las letras representan números, y calcular con letras demuestra hechos sobre
\emph{todos} los números a la vez. Este capítulo practica el desarrollo y la
factorización --- incluidas las tres identidades que te acompañarán por
todas las matemáticas --- y resuelve ecuaciones de primer grado, ecuaciones
\omterm{def:g2:mult:def}{producto} e inecuaciones sencillas, siempre paso a paso y con justificación.

\section{Desarrollar}

\begin{definition}[Desarrollar]\label{def:g9:algebra:expand}
\emph{Desarrollar}\index{desarrollar} es transformar un \omterm{def:g2:mult:def}{producto} en una
\omterm{def:g1:addition:def}{suma}, usando la distributividad:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g9:algebra:expand}
Cada término del primer paréntesis multiplica a cada término del segundo:
\begin{align*}
(x + 3)(2x + 5)
&= x \times 2x + x \times 5 + 3 \times 2x + 3 \times 5 \\
&= 2x^2 + 5x + 6x + 15 \\
&= 2x^2 + 11x + 15 .
\end{align*}
Los signos viajan con los términos:
\begin{align*}
(3x - 2)(x - 4)
&= 3x^2 - 12x - 2x + 8
= 3x^2 - 14x + 8 .
\end{align*}
\end{example}

\begin{theorem}[Las tres identidades]\label{thm:g9:algebra:identities}
Para todos los números $a$ y $b$:
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
Desarrolla cada miembro izquierdo. Para la primera:
$(a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2$. Para la segunda,
sustituye $b$ por $-b$ en la primera. Para la tercera:
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$ (los términos cruzados se cancelan).
\end{proof}

\[
(x + 5)^2 = x^2 + 10x + 25, \qquad
(3x - 1)^2 = 9x^2 - 6x + 1, \qquad
(2x + 7)(2x - 7) = 4x^2 - 49 .
\]
Cálculo mental: $102^2 = (100 + 2)^2 = 10000 + 400 + 4 = 10404$, y
$98 \times 102 = (100-2)(100+2) = 10000 - 4 = 9996$.
\end{example}

\section{Factorizar}

\begin{method}[Factorizar]\label{met:g9:algebra:factor}
Para factorizar una expresión, prueba en este orden:
\begin{enumerate}
  \item \emph{factor común}: $ka + kb = k(a + b)$, donde $k$ puede ser
        él mismo un paréntesis;
  \item \emph{\omterm{ex:g1:subtraction:difference}{diferencia} de cuadrados}: $a^2 - b^2 = (a+b)(a-b)$;
  \item \emph{cuadrado perfecto}: $a^2 + 2ab + b^2 = (a+b)^2$ (y, con
        $-2ab$, $(a-b)^2$).
\end{enumerate}
Comprueba siempre desarrollando el resultado.
\end{method}

\begin{example}\label{ex:g9:algebra:factor}
\emph{Factor común:}
$15x^2 + 10x = 5x \times 3x + 5x \times 2 = 5x(3x + 2)$.

\emph{Paréntesis común:}
\[
(x+2)(x-1) + (x+2)(3x+4) = (x+2)\bigl[(x-1) + (3x+4)\bigr]
= (x+2)(4x+3).
\]

\emph{\omterm{ex:g1:subtraction:difference}{Diferencia} de cuadrados:}
$x^2 - 81 = (x+9)(x-9)$, y
\[
(2x+1)^2 - 9 = (2x+1)^2 - 3^2 = (2x+1+3)(2x+1-3) = (2x+4)(2x-2).
\]
Cada factor sigue teniendo un factor común: $(2x+4)(2x-2) = 2(x+2) \times
2(x-1) = 4(x+2)(x-1)$.
\end{example}

\section{Ecuaciones}

\begin{method}[Ecuaciones de primer grado]\label{met:g9:algebra:firstdegree}
Para resolver una \omterm{def:g8:equations:def}{ecuación} como $5x - 3 = 2x + 9$:
\begin{enumerate}
  \item reúne los términos en $x$ en un miembro y los números en el otro,
        sumando la misma cantidad a los dos miembros;
  \item reduce cada miembro;
  \item divide los dos miembros entre el coeficiente de $x$;
  \item comprueba la solución en la \omterm{def:g8:equations:def}{ecuación} de partida.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:algebra:firstdegree}
\begin{align*}
5x - 3 &= 2x + 9 \\
5x - 2x &= 9 + 3 && \text{(suma $3 - 2x$ a los dos miembros)}\\
3x &= 12 \\
x &= 4 && \text{(divide entre 3).}
\end{align*}
Comprobación: $5 \times 4 - 3 = 17$ y $2 \times 4 + 9 = 17$. La solución es
$4$.

Con paréntesis, desarrolla primero: $3(x - 2) = x + 8$ se convierte en $3x - 6 = x + 8$,
luego $2x = 14$ y $x = 7$.
\end{example}

\begin{theorem}[Regla del producto nulo]\label{thm:g9:algebra:zeroproduct}
Un \omterm{def:g2:mult:def}{producto} es \omterm{def:g1:counting:zero}{cero} si y solo si al menos uno de los factores es \omterm{def:g1:counting:zero}{cero}:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ o } \ B = 0 .
\]
\end{theorem}

\begin{proof}
Si un factor es \omterm{def:g1:counting:zero}{cero}, el \omterm{def:g2:mult:def}{producto} lo es claramente. Recíprocamente, si $A B = 0$ con
$A \neq 0$, dividir los dos miembros entre $A$ da $B = 0$.
\end{proof}

\begin{example}\label{ex:g9:algebra:zeroproduct}
Resuelve $(x - 3)(2x + 10) = 0$: o bien $x - 3 = 0$, o bien $2x + 10 = 0$, así que las
soluciones son $3$ y $-5$.

Resuelve $x^2 - 49 = 0$: factoriza primero, $(x-7)(x+7) = 0$, soluciones $7$ y
$-7$.

Resuelve $x^2 = 3x$: ¡nunca dividas entre $x$! Pasa todo a un lado y factoriza:
$x^2 - 3x = x(x - 3) = 0$, soluciones $0$ y $3$.
\end{example}

\section{Inecuaciones}

\begin{proposition}[Reglas de las desigualdades]\label{prop:g9:algebra:ineq}
Una desigualdad se conserva al sumar el mismo número a los dos miembros y
al multiplicar los dos miembros por el mismo número \emph{positivo}.
Multiplicar los dos miembros por un número \emph{negativo} invierte el
signo de desigualdad.
\end{proposition}

\begin{proof}
Si $a \leq b$, entonces $b - a \geq 0$. Sumando $c$: $(b+c) - (a+c) = b - a
\geq 0$, luego $a + c \leq b + c$. Multiplicando por $c > 0$:
$bc - ac = (b-a)c \geq 0$, luego $ac \leq bc$. Multiplicando por $c < 0$:
$(b-a)c \leq 0$, luego $ac \geq bc$: el orden se invierte.
\end{proof}

\begin{example}\label{ex:g9:algebra:ineq}
Resuelve $7 - 2x < 15$:
\begin{align*}
7 - 2x &< 15 \\
-2x &< 8 && \text{(resta 7)} \\
x &> -4 && \text{(divide entre $-2$: invierte el signo).}
\end{align*}
Las soluciones son todos los números mayores que $-4$: en una recta numérica, un
punto hueco en $-4$ y sombreado hacia la derecha.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g9:algebra:1}
Desarrolla y reduce:
\[
4(3x - 2), \qquad
(x + 6)(x + 2), \qquad
(2x - 3)(x + 5), \qquad
(x - 1)(x - 9).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:algebra:2}
Desarrolla usando las identidades:
\[
(x + 8)^2, \qquad
(3x - 4)^2, \qquad
(x + 10)(x - 10), \qquad
(5x + 2)(5x - 2).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:algebra:3}
Calcula mentalmente, usando una identidad: $101^2$; \quad $59^2$
(escribe $59 = 60 - 1$); \quad $53 \times 47$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:algebra:4}
Factoriza:
\[
9x + 12, \qquad
x^2 - 64, \qquad
x^2 + 14x + 49, \qquad
18x^2 - 6x .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:algebra:5}
Resuelve, escribiendo todos los pasos:
\[
4x + 7 = 19, \qquad
6x - 5 = 2x + 11, \qquad
5(x - 3) = 3x + 1 .

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:algebra:6}
Resuelve usando la regla del \omterm{def:g2:mult:def}{producto} nulo:

(x + 4)(3x - 12) = 0, \qquad
x^2 - 121 = 0, \qquad

\begin{exercise}[$\star\star$]\label{exo:g9:algebra:7}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:algebra:8}
Resuelve las inecuaciones y representa las soluciones en una recta numérica:

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:algebra:9}
Pienso un número, lo multiplico por $3$, le sumo $7$ y obtengo el mismo resultado
que si lo hubiera multiplicado por $5$ y le hubiera restado $9$. ¿Cuál es mi número?
(Plantea una \omterm{def:g8:equations:def}{ecuación} y resuélvela.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:algebra:10}
Sea $n$ un número entero.
\begin{enumerate}
  \item desarrolla $(n + 1)(n - 1)$ y deduce que el \omterm{def:g2:mult:def}{producto} de los dos
        vecinos de $n$ siempre es $n^2 - 1$ (por ejemplo, $24 \times 26 =
        25^2 - 1 = 624$);
  \item demuestra que la \omterm{def:g1:addition:def}{suma} de tres enteros consecutivos siempre es
        \omterm{def:g9:arith:divisor}{múltiplo} de $3$.
\end{enumerate}
\end{exercise}

\section{Problema: El álgebra de las cifras}

\begin{problem}[{Problema de fin de semana --- por qué funcionan los criterios de
divisibilidad, el truco del 1089 desenmascarado y un cálculo mental que parece
brujería}]
\label{pb:g9:algebra:1}
En primaria te enseñaron \emph{reglas} (un número es \omterm{def:g6:wholes:divisible}{divisible} entre
$9$ cuando lo es su \omterm{def:g1:addition:def}{suma} de cifras); los magos hacen \emph{trucos}
(todo el mundo acaba en $1089$); los mercados conocen \emph{atajos}
($35^2 = 1225$, al instante). Todo ello es el mismo secreto: un
número de cifras $a$, $b$, $c$ es la expresión
$100a + 10b + c$, y el álgebra de este capítulo
(\cref{thm:g9:algebra:identities}, \omterm{def:g9:algebra:expand}{desarrollar}, factorizar) sabe
desmontarlo. Al final de este problema habrás demostrado las
reglas, desenmascarado el truco y aprendido la brujería.

\medskip
\noindent\textbf{Parte I --- Las cifras al microscopio.}
\begin{enumerate}
  \item un número de dos cifras con cifra de las decenas $a$ y cifra de las unidades
        $b$ es $10a + b$. Desarrolla y factoriza la \omterm{ex:g1:subtraction:difference}{diferencia}
        entre el número y su volteado,
        $(10a + b) - (10b + a)$, y enuncia el descubrimiento. Compruébalo
        con $72 - 27$;
  \item demuestra el criterio del nueve para los números de tres cifras: verifica
        la identidad
        \[
        100a + 10b + c = 9\,(11a + b) + (a + b + c),
        \]
        y explica por qué eso muestra que un número y su \omterm{def:g1:addition:def}{suma} de cifras
        dejan el \emph{mismo \omterm{def:g3:division:remainder}{resto}} en la división entre $9$
        --- así que uno es \omterm{def:g6:wholes:divisible}{divisible} entre $9$ exactamente cuando lo es el otro
        (\cref{prop:g9:arith:rules}, ahora demostrado);
  \item deduce el criterio del tres de la misma identidad, en una
        frase;
  \item «la prueba del nueve», la comprobación de los contables: para verificar
        $47 \times 86 = 4\,042$, sustituye cada número por la
        \omterm{def:g1:addition:def}{suma} de cifras de su \omterm{def:g1:addition:def}{suma} de cifras: $47 \to 11 \to 2$,
        $86 \to 14 \to 5$, y $2 \times 5 = 10 \to 1$; después
        $4\,042 \to 10 \to 1$: coherente. Explica por qué un
        \omterm{def:g2:mult:def}{producto} correcto tiene que pasar la prueba (escribe $47 = 9k + 2$,
        $86 = 9l + 5$ y desarrolla) y encuentra un resultado erróneo que
        la prueba no detecte (¿por qué son invisibles para ella dos cifras
        intercambiadas?);
  \item demuestra el criterio del once para los números de cuatro cifras: verifica
        \[
        1000a + 100b + 10c + d
        = 11\,(91a + 9b + c) + (- a + b - c + d),
        \]
        y úsalo para decidir si $2\,926$ es \omterm{def:g6:wholes:divisible}{divisible} entre
        $11$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El truco del 1089.}
El truco: piensa un número de tres cifras cuya primera y última
cifras difieran al menos en $2$; voltéalo; resta el menor
del mayor; voltea el resultado; \omterm{def:g1:addition:def}{suma} los dos últimos números.
El mago anuncia: \emph{1089}.
\begin{enumerate}[resume]
  \item haz el truco con $742$ y con un número que elijas
        tú;
  \item sea el número de cifras $a$, $b$, $c$ con $a > c$.
        Factoriza la primera resta,
        $(100a + 10b + c) - (100c + 10b + a)$, y enumera todos los
        valores que puede tomar cuando $a - c$ va de $2$ a $9$;
  \item cada uno de esos \omterm{def:g5:division:multiple}{múltiplos} de $99$ tiene un patrón de cifras
        llamativo. Escribiendo $m = a - c$, verifica la identidad
        \[
        99m = 100\,(m - 1) + 90 + (10 - m),
        \]
        y lee las tres cifras de $99m$;
  \item \omterm{def:g1:addition:def}{suma} $99m$ con su propio volteado, usando las cifras de
        la pregunta 8, y demuestra que el total es $1089$ --- sea cual sea
        $m$. El truco ha muerto; viva el álgebra;
  \item ¿por qué exigía el mago que la primera y la última cifra
        difirieran al menos en $2$? Examina los casos $a = c$ y
        $a - c = 1$ (en el que la resta da $99$, que
        hay que escribir $099$ para que el volteo funcione) y
        enuncia la letra pequeña que mantiene honesto el truco.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Brujería de uso diario.}
\begin{enumerate}[resume]
  \item el truco del cumpleaños: «multiplica tu mes de nacimiento por $5$,
        súmale $7$, dóblalo, súmale tu día de nacimiento, réstale $14$».
        Demuestra que el resultado siempre es $10m + d$ --- mes y
        día, legibles de un vistazo. (Pruébalo con alguien.);
  \item rediseñas el truco así: «multiplica el mes por $5$,
        súmale $9$, dóblalo, súmale el día». ¿Qué número tiene que
        restar ahora el espectador al final para obtener la misma
        lectura mes--día?;
  \item el truco de la cifra que falta: el espectador elige un número cualquiera,
        le resta su \omterm{def:g1:addition:def}{suma} de cifras (la pregunta 2 dice que el resultado es
        siempre \omterm{def:g9:arith:divisor}{múltiplo} de $9$ --- explica por qué para un
        número de tres cifras), después tacha una cifra \emph{no nula}
        y te lee las cifras restantes. Explica cómo
        recuperas la cifra tachada y por qué había que
        excluir el $0$;
  \item brujería de mercado: demuestra la identidad
        $(10t + 5)^2 = 100\,t(t+1) + 25$, enuncia la regla para
        elevar al cuadrado cualquier número terminado en $5$ y calcula $45^2$,
        $85^2$ y $105^2$ de cabeza;
  \item final, ampliando el \cref{exo:g9:algebra:10}: demuestra que
        la \omterm{def:g1:addition:def}{suma} de cinco enteros consecutivos siempre es
        \omterm{def:g9:arith:divisor}{múltiplo} de $5$, pero que la \omterm{def:g1:addition:def}{suma} de \emph{cuatro}
        enteros consecutivos \emph{nunca} es \omterm{def:g9:arith:divisor}{múltiplo} de $4$.
        (Dos cálculos cortos --- y una moraleja sobre lo que
        el álgebra hace con las palabras «siempre» y «nunca».)
\end{enumerate}
\end{problem}
