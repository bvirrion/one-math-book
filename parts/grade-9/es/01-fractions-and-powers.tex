\chapter{Fracciones y potencias}\label{ch:g9:fractions}

Las fracciones y las \omterm{def:g9:fractions:power}{potencias} son las herramientas de cálculo de todos los días: repartir
cantidades, comparar proporciones, escribir números muy grandes o muy
pequeños. Este capítulo consolida las reglas para calcular con ellas ---
con todos los pasos intermedios escritos --- e introduce la \omterm{def:g9:fractions:scientific}{notación
científica}.

\section{Calcular con fracciones}

\begin{definition}[Fracción]\label{def:g9:fractions:fraction}
Una \emph{fracción}\index{fracción} $\dfrac ab$ (con $b \neq 0$) es el
\omterm{def:g3:division:remainder}{cociente} de $a$ entre $b$. Dos fracciones son iguales cuando una se obtiene de
la otra multiplicando (o dividiendo) \omterm{def:g4:fractions:def}{numerador} y \omterm{def:g4:fractions:def}{denominador} por el
mismo número no nulo:
\[
\frac ab = \frac{a \times k}{b \times k} \qquad (k \neq 0).
\]
\end{definition}

\begin{example}\label{ex:g9:fractions:simplify}
Simplifica $\dfrac{42}{56}$ paso a paso:
\[
\frac{42}{56} = \frac{42 \div 2}{56 \div 2} = \frac{21}{28}
= \frac{21 \div 7}{28 \div 7} = \frac34 .
\]
(En el \cref{ch:g9:arith} el máximo común \omterm{def:g5:division:multiple}{divisor} hará esto en un solo
paso.)
\end{example}

\begin{method}[Sumar y restar fracciones]\label{met:g9:fractions:add}
\begin{enumerate}
  \item pon las dos fracciones con un \emph{\omterm{def:g4:fractions:def}{denominador} común} (un
        \omterm{def:g5:division:multiple}{múltiplo} común de los dos \omterm{def:g4:fractions:def}{denominadores});
  \item \omterm{def:g1:addition:def}{suma} o resta los \omterm{def:g4:fractions:def}{numeradores}, conservando el \omterm{def:g4:fractions:def}{denominador};
  \item simplifica el resultado si se puede.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:fractions:add}
Calcula $\dfrac56 + \dfrac34$. Un \omterm{def:g4:fractions:def}{denominador} común de $6$ y $4$ es
$12$:
\[
\frac56 + \frac34
= \frac{5 \times 2}{6 \times 2} + \frac{3 \times 3}{4 \times 3}
= \frac{10}{12} + \frac{9}{12}
= \frac{19}{12}.
\]
Calcula $2 - \dfrac37$. Escribe $2$ como \omterm{def:g9:fractions:fraction}{fracción} con \omterm{def:g4:fractions:def}{denominador} $7$:
\[
2 - \frac37 = \frac{14}{7} - \frac37 = \frac{11}{7}.
\]
\end{example}

\begin{method}[Multiplicar y dividir fracciones]\label{met:g9:fractions:mult}
\begin{enumerate}
  \item para multiplicar, multiplica los \omterm{def:g4:fractions:def}{numeradores} entre sí y los \omterm{def:g4:fractions:def}{denominadores}
        entre sí: $\dfrac ab \times \dfrac cd = \dfrac{ac}{bd}$
        (simplifica \emph{antes} de multiplicar siempre que puedas);
  \item para dividir, multiplica por el \omterm{def:g8:fractions:inverse}{inverso} del \omterm{def:g5:division:multiple}{divisor}:
        $\dfrac ab \div \dfrac cd = \dfrac ab \times \dfrac dc$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:fractions:mult}
\[
\frac{7}{15} \times \frac{25}{14}
= \frac{7 \times 25}{15 \times 14}
= \frac{7 \times 25}{15 \times 2 \times 7}
= \frac{25}{30} = \frac56,
\]
donde hemos simplificado por el factor común $7$ y después por $5$. Y una
división:
\[
\frac{3}{4} \div \frac{9}{8}
= \frac34 \times \frac89
= \frac{3 \times 8}{4 \times 9}
= \frac{24}{36} = \frac23 .
\]
\end{example}

\section{Las potencias}

\begin{definition}[Potencia]\label{def:g9:fractions:power}
Para un número real $a$ y un número entero positivo $n$, la
\emph{potencia}\index{potencia} $a^n$ es el \omterm{def:g2:mult:def}{producto} de $n$ factores iguales a
$a$:
\[
a^n = \underbrace{a \times a \times \dots \times a}_{n \text{ factores}} .
\]
Por convenio $a^0 = 1$ (para $a \neq 0$), y los \omterm{def:g8:powers:def}{exponentes} negativos indican
\omterm{def:g8:fractions:inverse}{inversos}:
\[
a^{-n} = \frac{1}{a^n}.
\]
\end{definition}

\begin{example}
$2^4 = 2 \times 2 \times 2 \times 2 = 16$; \quad
$(-3)^2 = 9$, pero $-3^2 = -(3^2) = -9$ (el \omterm{def:g8:powers:def}{exponente} actúa antes que el
signo menos); \quad $10^{-3} = \frac{1}{1000} = 0.001$; \quad
$5^1 = 5$ y $5^0 = 1$.
\end{example}

\begin{theorem}[Reglas de los exponentes]\label{thm:g9:fractions:rules}
Para todos los $a$, $b$ no nulos y todos los números enteros $m$, $n$:
\[
a^m \times a^n = a^{m+n},
\qquad
\frac{a^m}{a^n} = a^{m-n},
\qquad
\left(a^m\right)^n = a^{mn},
\qquad
(ab)^n = a^n b^n .
\]
\end{theorem}

\begin{proof}
Para \omterm{def:g8:powers:def}{exponentes} positivos, cuenta los factores. Para la primera regla:
$a^m \times a^n$ son $m$ factores $a$ seguidos de $n$ factores $a$, en total
$m + n$ factores. Para la tercera:
$\left(a^m\right)^n$ repite $n$ veces un bloque de $m$ factores, lo que da
$mn$ factores. Las otras dos reglas son parecidas; la extensión al \omterm{def:g1:counting:zero}{cero} y a los
\omterm{def:g8:powers:def}{exponentes} negativos se comprueba a partir de $a^{-n} = \frac{1}{a^n}$ (y hace
del convenio $a^0 = 1$ la única elección coherente, ya que
$a^n \times a^0$ tiene que ser $a^{n+0} = a^n$).
\end{proof}

\begin{example}\label{ex:g9:fractions:rules}
Simplifica paso a paso:
\[
\frac{3^7 \times 3^{-2}}{3^4}
= \frac{3^{7 + (-2)}}{3^4}
= \frac{3^5}{3^4}
= 3^{5-4} = 3^1 = 3 .
\]
Y con dos letras: $(2a^3)^2 \times a = 4 a^6 \times a = 4a^7$.
\end{example}

\section{Las potencias de diez y la notación científica}

\begin{proposition}[Potencias de diez]\label{prop:g9:fractions:ten}
Para un número entero positivo $n$: $10^n$ se escribe «$1$ seguido de $n$
\omterm{def:g1:counting:zero}{ceros}», y $10^{-n} = 0.0\dots01$ con el $1$ en el $n$-ésimo lugar
decimal. Multiplicar un \omterm{def:g6:decimals:places}{número decimal} por $10^n$ (resp.\ $10^{-n}$) desplaza
su \omterm{def:g4:decimals:point}{punto decimal} $n$ lugares a la derecha (resp.\ a la izquierda).
\end{proposition}

\begin{definition}[Notación científica]\label{def:g9:fractions:scientific}
La \emph{notación científica}\index{notación científica} de un
\omterm{def:g6:decimals:places}{número decimal} positivo es la única manera de escribirlo como
\[
a \times 10^n,
\qquad\text{con } 1 \leq a < 10 \text{ y } n \text{ entero.}
\]
\end{definition}

\begin{example}\label{ex:g9:fractions:scientific}
La distancia Tierra--Sol es de unos $149\,600\,000$ km $= 1.496 \times
10^8$ km. El tamaño de una molécula de agua es de unos $0.000\,000\,000\,28$ m
$= 2.8 \times 10^{-10}$ m. Para calcular con estos números, agrupa las
partes decimales y las \omterm{def:g9:fractions:power}{potencias} de diez:
\[
(3 \times 10^5) \times (2.5 \times 10^{-2})
= (3 \times 2.5) \times 10^{5 + (-2)}
= 7.5 \times 10^{3}.
\]
\end{example}

\begin{method}[Poner un número en notación científica]\label{met:g9:fractions:scientific}
\begin{enumerate}
  \item desplaza el \omterm{def:g4:decimals:point}{punto decimal} justo detrás de la primera cifra no nula; eso
        da el factor $a$ con $1 \leq a < 10$;
  \item cuenta cuántos lugares se ha movido el punto: esa cuenta es $n$,
        positiva si el número de partida era $\geq 10$, negativa si era
        $< 1$;
  \item comprueba: $a \times 10^n$ tiene que reproducir el número de partida.
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g9:fractions:1}
Calcula y da el resultado como \omterm{def:g9:fractions:fraction}{fracción} totalmente simplificada:
\[
\frac13 + \frac15, \qquad
\frac72 - \frac53, \qquad
\frac{5}{12} + \frac38, \qquad
3 - \frac45 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:fractions:2}
Calcula y simplifica:
\[
\frac59 \times \frac{27}{35}, \qquad
\frac{8}{15} \div \frac{4}{5}, \qquad
\frac23 \times \frac34 \times \frac45 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:fractions:3}
Calcula:
\[
2^5, \qquad (-2)^4, \qquad -2^4, \qquad 4^{-2}, \qquad
\left(\tfrac23\right)^3, \qquad 7^0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:fractions:4}
Escribe como una sola \omterm{def:g9:fractions:power}{potencia}:
\[
5^3 \times 5^6, \qquad
\frac{7^9}{7^5}, \qquad
\left(2^3\right)^4, \qquad
\frac{3^2 \times 3^7}{3^5}, \qquad
2^5 \times 4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:fractions:5}
Escribe en \omterm{def:g9:fractions:scientific}{notación científica}: $45\,000$; $0.0072$; $302.5$;
$0.000\,000\,9$; doce millones.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:fractions:6}
Calcula y da el resultado en \omterm{def:g9:fractions:scientific}{notación científica}:
\[
(2 \times 10^7) \times (4.5 \times 10^{-3}),
\qquad
\frac{9 \times 10^5}{3 \times 10^{-2}},
\qquad
(5 \times 10^3)^2 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:fractions:7}
Un depósito está lleno hasta sus $\frac35$. Tras añadir $30$ litros está lleno hasta sus $\frac34$.
¿Cuál es la \omterm{def:g2:measure:units}{capacidad} del depósito? (Expresa primero la \omterm{def:g9:fractions:fraction}{fracción} del depósito
que se ha añadido.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:fractions:8}
Alicia se come $\frac14$ de un pastel, después Ben se come $\frac13$ de lo que queda
y después Carla se come $\frac12$ de lo que queda todavía. ¿Qué \omterm{def:g9:fractions:fraction}{fracción} del
pastel queda al final? (Calcula el \omterm{def:g3:division:remainder}{resto} después de cada paso.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:fractions:9}
La luz recorre $3 \times 10^8$ metros por segundo. El Sol está a unos
$1.5 \times 10^{11}$ m. ¿Cuánto tarda la luz del Sol en llegar hasta nosotros?
Da la respuesta en segundos (no hace falta \omterm{def:g9:fractions:scientific}{notación científica}) y después en
minutos y segundos.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:fractions:10}
¿Cuál es mayor, $2^{30}$ o $3^{20}$? (Pista: escribe los dos como \omterm{def:g9:fractions:power}{potencias} de
\omterm{def:g8:powers:def}{exponente} $10$ usando $\left(a^m\right)^n = a^{mn}$, y compara $2^3$ con
$3^2$.)
\end{exercise}

\section{Problema: El código secreto de los decimales periódicos}

\begin{problem}[{Problema de fin de semana --- toda fracción tiene una
escritura decimal que se para o se repite, todo decimal periódico es una
fracción, y $0.999\ldots$ vale exactamente $1$}]
\label{pb:g9:fractions:1}
Divide $3$ entre $11$ y las cifras caen en una cantinela:
$0.272727\ldots$ para siempre. Divide $1$ entre $4$ y las cifras se paran
en seco: $0.25$. Este problema demuestra que esos son los dos únicos
comportamientos posibles de una \omterm{def:g9:fractions:fraction}{fracción} --- y después invierte la
máquina: dado cualquier decimal periódico, reconstruye la \omterm{def:g9:fractions:fraction}{fracción}
que se esconde detrás. Por el camino zanja, de una vez por todas, la
igualdad más discutida de las matemáticas escolares.

\medskip
\noindent\textbf{Parte I --- De la \omterm{def:g9:fractions:fraction}{fracción} al decimal.}
\begin{enumerate}
  \item calcula por división larga las escrituras decimales de
        $\frac14$, $\frac13$, $\frac56$ y $\frac{3}{11}$.
        ¿Cuáles se paran, cuáles se repiten y con qué periodo?
  \item explica por qué la escritura decimal de \emph{cualquier} \omterm{def:g9:fractions:fraction}{fracción}
        $\frac ab$ tiene que pararse o acabar repitiéndose. (En la
        división larga entre $b$, ¿qué valores pueden tomar los
        \omterm{def:g3:division:remainder}{restos} sucesivos? ¿Qué pasa en cuanto un \omterm{def:g3:division:remainder}{resto}
        aparece por segunda vez?)
  \item lleva la división de $1$ entre $7$ lo bastante lejos como para hallar el
        periodo. ¿Qué longitud tiene y cómo se compara esa longitud
        con la cota prometida por la pregunta 2?
  \item algunas fracciones se paran porque su \omterm{def:g4:fractions:def}{denominador} se puede
        inflar hasta una \omterm{def:g9:fractions:power}{potencia} de diez: $\frac14 =
        \frac{25}{100}$, $\frac{3}{8} = \frac{375}{1000}$.
        explica el criterio y por qué ningún multiplicador entero
        puede convertir nunca $3$, $6$ u $11$ en $10$, $100$,
        $1000, \dots$ (¿en qué cifra terminaría el \omterm{def:g2:mult:def}{producto}?);
  \item sin dividir, clasifica $\frac{9}{40}$, $\frac{5}{12}$ y
        $\frac{33}{50}$ en «se para» y «se repite», y da después
        la escritura decimal de las dos que se paran.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Del decimal periódico de vuelta a la
\omterm{def:g9:fractions:fraction}{fracción}.}
\begin{enumerate}[resume]
  \item el truco del desplazamiento. Sea $x = 0.7777\ldots$ Calcula $10x$,
        después $10x - x$, y deduce $x$ como \omterm{def:g9:fractions:fraction}{fracción}. Comprueba
        dividiendo;
  \item sea $x = 0.727272\ldots$ ¿Qué \omterm{def:g9:fractions:power}{potencia} de diez desplaza $x$
        exactamente un periodo? Úsala para escribir $x$ como
        \omterm{def:g9:fractions:fraction}{fracción} irreducible;
  \item un caso mixto: $x = 0.58333\ldots$ (solo se repite el
        $3$). Calcula $1000x - 100x$ y deduce $x$ como
        \omterm{def:g9:fractions:fraction}{fracción} irreducible;
  \item el famoso: aplica el truco del desplazamiento a
        $x = 0.9999\ldots$ ¿Qué \omterm{def:g9:fractions:fraction}{fracción} --- qué \emph{número}
        --- encuentras? Concilia el resultado con tu
        intuición, recordando el vecino fantasma del
        \cref{pb:g6:decimals:1} y el \omterm{def:g3:division:remainder}{resto} que encoge del
        \cref{pb:g6:fractions:1}: ¿$0.999\ldots$ está «justo por debajo» de
        $1$, o es otro nombre de $1$?;
  \item convierte $2.454545\ldots$ y $0.123123123\ldots$ en
        fracciones irreducibles.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El diccionario completo.}
\begin{enumerate}[resume]
  \item las preguntas 6--10 sugieren un diccionario: un bloque de $k$
        cifras que se repite desde el \omterm{def:g4:decimals:point}{punto decimal} es igual a ese bloque
        sobre $k$ nueves ($0.727272\ldots = \frac{72}{99}$). Usa
        el diccionario para predecir $0.037037\ldots$ como \omterm{def:g9:fractions:fraction}{fracción},
        simplifica --- y admírate: ¿qué \omterm{def:g9:fractions:fraction}{fracción} sorprendentemente
        sencilla canturrea «$037$»? Compruébalo dividiendo;
  \item combina el diccionario con un desplazamiento: escribe
        $0.0272727\ldots$ como \omterm{def:g9:fractions:fraction}{fracción} irreducible;
  \item el número $0.101001000100001\ldots$ (un $1$, después un
        $0$, después un $1$, después dos $0$, después un $1$, después tres
        $0$, y así sucesivamente) no se para nunca. ¿Es un decimal periódico?
        ¿Qué dice entonces la Parte I sobre él: puede ser una
        \omterm{def:g9:fractions:fraction}{fracción}? (Acabas de encontrarte con tu primer número
        \emph{irracional} demostrable; el más famoso se atrapa
        en el \cref{ch:g9:sqrt});
  \item ¿aproximadamente cuántas cifras tiene $2^{30}$? Usa
        $2^{10} = 1\,024 \approx 1.024 \times 10^3$ y las
        reglas de los \omterm{def:g8:powers:def}{exponentes} (\cref{thm:g9:fractions:rules}) para escribir
        $2^{30}$ en \omterm{def:g9:fractions:scientific}{notación científica} (aproximada), y
        concluye. (Compara con el \cref{exo:g9:fractions:10}.);
  \item final: escribe $0.20262026\ldots$ como \omterm{def:g9:fractions:fraction}{fracción} y
        enuncia la moraleja de doble sentido de todo el problema: qué
        escrituras decimales son fracciones y qué fracciones tienen
        qué escrituras decimales.
\end{enumerate}
\end{problem}
