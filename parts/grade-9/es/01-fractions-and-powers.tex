\chapter{Fracciones y potencias}\label{ch:g9:fractions}

\omterm{def:g9:fractions:fraction}{fracciones} y \omterm{def:g9:fractions:power}{potestades} son las herramientas cotidianas de la computación: compartir
cantidades, comparar proporciones, escribir muy grande o muy pequeño
números. Este capítulo consolida las reglas para calcular con ellos ---
con cada paso intermedio escrito --- y presenta \omterm{def:g9:fractions:scientific}{notación científica}.

\section{Calcular con fracciones}

\begin{definition}[Fracción]\label{def:g9:fractions:fraction}
A \emph{fracción}\index{fracción} $\dfrac ab $ (con $ b \neq 0$) es el
\omterm{def:g3:division:remainder}{cociente} de $ a $ por $ b $. Dos fracciones son iguales cuando una se obtiene de
el otro multiplicando (o dividiendo) \omterm{def:g4:fractions:def}{numerador} y \omterm{def:g4:fractions:def}{denominador} por el
mismo número distinto de cero:
\[
\frac ab = \frac{a \times k}{b \times k} \qquad (k \neq 0).
\]
\end{definition}

\begin{example}\label{ex:g9:fractions:simplify}
Simplificar $\dfrac{42}{56}$ paso a paso:
\[
\frac{42}{56} = \frac{42 \div 2}{56 \div 2} = \frac{21}{28}
= \frac{21 \div 7}{28 \div 7} = \frac34 .
\]
(En \cref{ch:g9:arith} el mayor común \omterm{def:g5:division:multiple}{divisor} Haré esto en uno
paso.)
\end{example}

\begin{method}[Sumar y restar fracciones]\label{met:g9:fractions:add}
\begin{enumerate}
  \item poner ambos \omterm{def:g9:fractions:fraction}{fracciones} sobre un \emph{común \omterm{def:g4:fractions:def}{denominador}} (un común
        \omterm{def:g5:division:multiple}{múltiple} de los dos \omterm{def:g4:fractions:def}{denominadores});
  \item sumar o restar el \omterm{def:g4:fractions:def}{numeradores}, manteniendo el \omterm{def:g4:fractions:def}{denominador};
  \item Simplifique el resultado si es posible.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:fractions:add}
Calcular $\dfrac56 + \dfrac34$. un común \omterm{def:g4:fractions:def}{denominador} de $6$ y $4$ es
$12$:
\[
\frac56 + \frac34
= \frac{5 \times 2}{6 \times 2} + \frac{3 \times 3}{4 \times 3}
= \frac{10}{12} + \frac{9}{12}
= \frac{19}{12}.
\]
Calcular $2 - \dfrac37$. Escribir $2$ como un \omterm{def:g9:fractions:fraction}{fracción} encima $7$:
\[
2 - \frac37 = \frac{14}{7} - \frac37 = \frac{11}{7}.
\]
\end{example}

\begin{method}[Multiplicar y dividir fracciones]\label{met:g9:fractions:mult}
\begin{enumerate}
  \item Para multiplicar, multiplicar \omterm{def:g4:fractions:def}{numeradores} juntos y \omterm{def:g4:fractions:def}{denominadores}
        juntos: $\dfrac ab \times \dfrac cd = \dfrac{ac}{bd}$
        (simplificar \emph{antes} multiplicando siempre que sea posible);
  \item dividir, multiplicar por \omterm{def:g8:fractions:inverse}{inverso} del \omterm{def:g5:division:multiple}{divisor}:
        $\dfrac ab \div \dfrac cd = \dfrac ab \times \dfrac dc $.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:fractions:mult}
\[
\frac{7}{15} \times \frac{25}{14}
= \frac{7 \times 25}{15 \times 14}
= \frac{7 \times 25}{15 \times 2 \times 7}
= \frac{25}{30} = \frac56,
\]
donde simplificamos por el factor común $7$, entonces por $5$. y un
división:
\[
\frac{3}{4} \div \frac{9}{8}
= \frac34 \times \frac89
= \frac{3 \times 8}{4 \times 9}
= \frac{24}{36} = \frac23 .
\]
\end{example}

\section{Potestades}

\begin{definition}[Fuerza]\label{def:g9:fractions:power}
Para un número real $ a $ y un entero positivo $ n $, el
\emph{fuerza}\index{fuerza} $ a^n $ es el \omterm{def:g2:mult:def}{producto} de $ n $ factores iguales a
$ a $:
\[
a^n = \underbrace{a \times a \times \dots \times a}_{n \text{ factors}} .
\]
Por convención $ a^0 = 1$ (para $ a \neq 0$), y negativo \omterm{def:g8:powers:def}{exponentes} denotar
\omterm{def:g8:fractions:inverse}{inversas}:
\[
a^{-n} = \frac{1}{a^n}.
\]
\end{definition}

\begin{example}
$2^4 = 2 \times 2 \times 2 \times 2 = 16$; \quad
$(-3)^2 = 9$ pero $-3^2 = -(3^2) = -9$ (el \omterm{def:g8:powers:def}{exponente} se une ante el
signo menos); \quad $10^{-3} = \frac{1}{1000} = 0.001$; \quad
$5^1 = 5$ y $5^0 = 1$.
\end{example}

\begin{theorem}[Reglas de exponentes]\label{thm:g9:fractions:rules}
Para todos los distintos de cero $ a $, $ b $ y todos los números enteros $ m $, $ n $:
\[
a^m \times a^n = a^{m+n},
\qquad
\frac{a^m}{a^n} = a^{m-n},
\qquad
\left(a^m\right)^n = a^{mn},
\qquad
(ab)^n = a^n b^n .
\]
\end{theorem}

\begin{proof}
por positivo \omterm{def:g8:powers:def}{exponentes}, cuenta los factores. Para la primera regla:
$ a^m \times a^n $ es $ m $ factores $ a $ seguido por $ n $ factores $ a $, en total
$ m + n $ factores. Para el tercero:
$\left(a^m\right)^n $ repite $ n $ veces un bloque de $ m $ factores, dando
$ mn $ factores. Las otras dos reglas son similares; la extensión a \omterm{def:g1:counting:zero}{cero} y
negativo \omterm{def:g8:powers:def}{exponentes} se controla desde $ a^{-n} = \frac{1}{a^n}$ (y hace
la convención $ a^0 = 1$ la única opción consistente, ya que
$ a^n \times a^0$ debe ser $ a^{n+0} = a^n $).
\end{proof}

\begin{example}\label{ex:g9:fractions:rules}
Simplifica paso a paso:
\[
\frac{3^7 \times 3^{-2}}{3^4}
= \frac{3^{7 + (-2)}}{3^4}
= \frac{3^5}{3^4}
= 3^{5-4} = 3^1 = 3 .
\]
Y con dos letras: $(2a^3)^2 \times a = 4 a^6 \times a = 4a^7$.
\end{example}

\section{Potencias de diez y notación científica.}

\begin{proposition}[potencias de diez]\label{prop:g9:fractions:ten}
Para un entero positivo $ n $: $10^n $ se escribe ``$1$ seguido por $ n $
\omterm{def:g1:counting:zero}{ceros}'', y $10^{-n} = 0.0\dots01$ con el $1$ en el $ n $-ésimo decimal
lugar. multiplicando un \omterm{def:g6:decimals:places}{número decimal} por $10^n $ (resp.\ $10^{-n}$) turnos
es \omterm{def:g4:decimals:point}{coma decimal} $ n $ lugares a la derecha (resp.\ izquierda).
\end{proposition}

\begin{definition}[Notación científica]\label{def:g9:fractions:scientific}
El \emph{notación científica}\index{notación científica} de un positivo
\omterm{def:g6:decimals:places}{número decimal} es la forma única de escribirlo como
\[
a \times 10^n,
\qquad\text{con } 1 \leq a < 10 \text{ y } n \text{ an integer.}
\]
\end{definition}

\begin{example}\label{ex:g9:fractions:scientific}
La distancia Tierra-Sol es de aproximadamente $149\,600\,000$ kilómetros $= 1.496 \times
10^8$ km. El tamaño de una molécula de agua es aproximadamente $0.000\,000\,000\,28$ metro
$= 2.8 \times 10^{-10}$ metro. Para calcular con tales números, agrupe los
partes decimales y \omterm{def:g9:fractions:power}{potestades} a menudo:
\[
(3 \times 10^5) \times (2.5 \times 10^{-2})
= (3 \times 2.5) \times 10^{5 + (-2)}
= 7.5 \times 10^{3}.
\]
\end{example}

\begin{method}[Poner un número en notación científica]\label{met:g9:fractions:scientific}
\begin{enumerate}
  \item Mueva el \omterm{def:g4:decimals:point}{coma decimal} justo después del primer dígito distinto de cero; esto
        da el factor $ a $ con $1 \leq a < 10$;
  \item contar el número de lugares que el punto se movió: ese recuento es $ n $,
        positivo si el número original era $\geq 10$, negativo si fuera
        $< 1$;
  \item controlar: $ a \times 10^n $ debe reproducir el número original.
\end{enumerate}
\end{method}

\begin{exercise}[$\star\star $]\label{exo:g9:fractions:7}

\begin{exercise}[$\star\star $]\label{exo:g9:fractions:8}

luego carla come $\frac12$ de lo que aún queda. Qué \omterm{def:g9:fractions:fraction}{fracción} de la

\begin{exercise}[$\star\star $]\label{exo:g9:fractions:9}

\begin{exercise}[$\star\star\star $]\label{exo:g9:fractions:10}

$3^2$.)
\end{exercise}

\section{Problema: el código secreto de los decimales periódicos}

\begin{problem}[{Problema de fin de semana --- cada fracción tiene una escritura decimal que se detiene o se repite, cada decimal periódico es una fracción y $0,999\ldots$ equivale exactamente a $1$}]
\label{pb:g9:fractions:1}
Dividir $3$ por $11$ y los dígitos caen en un canto:
$0.272727\ldots $ para siempre. Dividir $1$ por $4$ y los dígitos se detienen
muerto: $0.25$. Este problema demuestra que estos son los únicos dos
posibles comportamientos para un \omterm{def:g9:fractions:fraction}{fracción} --- y luego invierte el
máquina: dado cualquier decimal periódico, reconstruye el \omterm{def:g9:fractions:fraction}{fracción}
escondiéndose detrás de él. En el camino se asienta, de una vez por todas, la
El tema más discutido es el de la igualdad en las matemáticas escolares.

\medskip
\noindent\textbf{Parte I --- From \omterm{def:g9:fractions:fraction}{fracción} to decimal.}
\begin{enumerate}
  \item Calcular por división larga las escrituras decimales de
        $\frac14$, $\frac13$, $\frac56$ y $\frac{3}{11}$.
        ¿Qué parada, qué repetición y con qué bloque repetitivo?
  \item Explique por qué la escritura decimal de \emph{cualquier} \omterm{def:g9:fractions:fraction}{fracción}
        $\frac ab $ debe detenerse o eventualmente repetirse. (En el
        división larga por $ b $, qué valores pueden los sucesivos
        \omterm{def:g3:division:remainder}{restos} ¿llevar? ¿Qué sucede en el momento en que \omterm{def:g3:division:remainder}{resto}
        aparece por segunda vez?)
  \item Impulsar la división de $1$ por $7$ lo suficientemente lejos como para encontrar el
        bloque repetitivo. ¿Cuánto mide y cómo mide esa longitud?
        comparar con el límite prometido en la pregunta 2?
  \item Alguno \omterm{def:g9:fractions:fraction}{fracciones} detenerse porque su \omterm{def:g4:fractions:def}{denominador} puede ser
        volado a un \omterm{def:g9:fractions:power}{fuerza} a menudo: $\frac14 =
        \frac{25}{100}$, $\frac{3}{8} = \frac{375}{1000}$.
        Explique el criterio y por qué no hay multiplicador de números enteros.
        alguna vez puede girar $3$, $6$ o $11$ en $10$, $100$,
        $1000, \dots $ (¿qué dígito sería el \omterm{def:g2:mult:def}{producto} terminar en?).
  \item Sin dividir, ordenar $\frac{9}{40}$, $\frac{5}{12}$ y
        $\frac{33}{50}$ en ``paradas'' y ``repeticiones'', luego dé
        la escritura decimal de los dos que paran.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- From repeating decimal back to the
\omterm{def:g9:fractions:fraction}{fracción}.}
\begin{enumerate}[resume]
  \item El truco del cambio. Sea $ x = 0.7777\ldots $ Calcular $10x $,
        entonces $10x - x $, y deducir $ x $ como un \omterm{def:g9:fractions:fraction}{fracción}. Consultar por
        dividiendo.
  \item Sea $ x = 0.727272\ldots $ Cual \omterm{def:g9:fractions:power}{fuerza} de diez turnos $ x $ por
        ¿Exactamente un bloque repetido? Úsalo para escribir $ x $ como un
        \omterm{def:g9:fractions:fraction}{fracción} en términos más bajos.
  \item Un caso mixto: $ x = 0.58333\ldots $ (el $3$ solo
        se repite). Calcular $1000x - 100x $ y deducir $ x $ como un
        \omterm{def:g9:fractions:fraction}{fracción} en términos más bajos.
  \item El famoso: aplica el truco del cambio a
        $ x = 0.9999\ldots $ Qué \omterm{def:g9:fractions:fraction}{fracción} --- qué \emph{número}
        --- ¿lo encuentras? Concilia el resultado con tu
        intuición, recordando al vecino fantasma de
        \cref{pb:g6:decimals:1} y el encogimiento \omterm{def:g3:division:remainder}{resto} de
        \cref{pb:g6:fractions:1}: es $0.999\ldots $ ``justo debajo''
        $1$, u otro nombre para $1$?
  \item Convertir $2.454545\ldots $ y $0.123123123\ldots $ en
        \omterm{def:g9:fractions:fraction}{fracciones} en términos más bajos.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- The dictionary completed.}
\begin{enumerate}[resume]
  \item Las preguntas 6 a 10 sugieren un diccionario: un bloque de $ k $
        dígitos que se repiten desde el \omterm{def:g4:decimals:point}{coma decimal} es igual a ese bloque
        encima $ k $ nueves ($0.727272\ldots = \frac{72}{99}$). uso
        el diccionario para predecir $0.037037\ldots $ como un \omterm{def:g9:fractions:fraction}{fracción},
        simplificar --- y maravillarse: qué sorprendentemente simple
        \omterm{def:g9:fractions:fraction}{fracción} canta ``$037$''? Compruébalo por división.
  \item Combina el diccionario con un turno: escribe
        $0.0272727\ldots $ como un \omterm{def:g9:fractions:fraction}{fracción} en términos más bajos.
  \item el número $0.101001000100001\ldots $ (a $1$, entonces uno
        $0$, entonces un $1$, entonces dos $0$ s, entonces un $1$, luego tres
        $0$ s, etc.) nunca se detiene. ¿Es un decimal periódico?
        ¿Qué dice entonces la Parte I al respecto? ¿Puede ser una
        \omterm{def:g9:fractions:fraction}{fracción}? (Acabas de conocer a tu primera persona demostrablemente
        \emph{irracional} número; el más famoso está atrapado
        en \cref{ch:g9:sqrt}.)
  \item ¿Cuántos dígitos aproximadamente tiene? $2^{30}$ ¿tener? Usar
        $2^{10} = 1\,024 \approx 1.024 \times 10^3$ y el
        \omterm{def:g8:powers:def}{exponente} normas (\cref{thm:g9:fractions:rules}) escribir
        $2^{30}$ en (aproximado) \omterm{def:g9:fractions:scientific}{notación científica}, y
        concluir. (Comparar \cref{exo:g9:fractions:10}.)
  \item final: escribir $0.20262026\ldots$ como un \omterm{def:g9:fractions:fraction}{fracción}, y
        enunciar la moraleja bidireccional de todo el problema: ¿cuál
        las escrituras decimales son \omterm{def:g9:fractions:fraction}{fracciones}, y cual \omterm{def:g9:fractions:fraction}{fracciones} tener
        ¿Qué escrituras decimales?
\end{enumerate}
\end{problem}
