\chapter{Raíces cuadradas}\label{ch:g9:sqrt}

¿Cuánto mide la diagonal de un cuadrado de lado $1$? El teorema de Pitágoras
responde $\sqrt 2$, un número cuyo cuadrado es $2$ --- y que resulta
no ser una \omterm{def:g9:fractions:fraction}{fracción}. Este capítulo define las \omterm{def:g9:sqrt:def}{raíces cuadradas}, establece las
reglas para calcular con ellas y enseña a simplificar expresiones
como $\sqrt{75}$.

\section{Definición y primeras propiedades}

\begin{definition}[Raíz cuadrada]\label{def:g9:sqrt:def}
Sea $a \geq 0$. La \emph{raíz cuadrada}\index{raíz cuadrada} de $a$, escrita
$\sqrt a$, es el único número \emph{no negativo} cuyo cuadrado es $a$:
\[
\sqrt a \geq 0
\qquad\text{y}\qquad
\left(\sqrt a\right)^2 = a .
\]
Los números negativos no tienen raíz cuadrada, porque todo cuadrado es no negativo.
\end{definition}

\begin{example}
$\sqrt{49} = 7$, $\sqrt 0 = 0$, $\sqrt 1 = 1$,
$\sqrt{2.25} = 1.5$. Las primeras \omterm{def:g9:sqrt:def}{raíces cuadradas} que hay que saberse de memoria son las de
los \emph{cuadrados perfectos}: $1, 4, 9, 16, 25, 36, 49, 64, 81, 100, 121,
144$.
\end{example}

\begin{proposition}[Raíz cuadrada de un cuadrado]\label{prop:g9:sqrt:square}
Para todo número real $x$ (positivo o no):
\[
\sqrt{x^2} = \abs{x} =
\begin{cases}
x & \text{si } x \geq 0,\\
-x & \text{si } x < 0 .
\end{cases}
\]
\end{proposition}

\begin{proof}
El número $\abs{x}$ es no negativo y su cuadrado es $x^2$ (un número y
su opuesto tienen el mismo cuadrado). Al ser el único número no negativo
de cuadrado $x^2$, es $\sqrt{x^2}$.
\end{proof}

\begin{example}
$\sqrt{(-5)^2} = \sqrt{25} = 5 = \abs{-5}$: la \omterm{def:g9:sqrt:def}{raíz cuadrada} «olvida»
el signo, no lo restituye. En particular, $\sqrt{x^2} = x$ es
\emph{falso} para $x$ negativo.
\end{example}

\section{Productos y cocientes de raíces cuadradas}

\begin{theorem}[Reglas de multiplicación y división]\label{thm:g9:sqrt:rules}
Para todos los $a \geq 0$ y $b \geq 0$:
\[
\sqrt{a b} = \sqrt a \times \sqrt b,
\qquad\text{y, para } b > 0:\quad
\sqrt{\frac ab} = \frac{\sqrt a}{\sqrt b} .
\]
\end{theorem}

\begin{proof}
El número $\sqrt a \times \sqrt b$ es no negativo (\omterm{def:g2:mult:def}{producto} de
números no negativos), y su cuadrado es
\[
\left(\sqrt a \times \sqrt b\right)^2
= \left(\sqrt a\right)^2 \times \left(\sqrt b\right)^2
= a b .
\]
Por unicidad del número no negativo cuyo cuadrado es $ab$, vale
$\sqrt{ab}$. La regla del \omterm{def:g3:division:remainder}{cociente} se demuestra igual.
\end{proof}

\begin{remark}[¡Nada de eso para las sumas!]
$\sqrt{a + b}$ \emph{no} es $\sqrt a + \sqrt b$ en general:
\[
\sqrt{9 + 16} = \sqrt{25} = 5,
\qquad\text{pero}\qquad
\sqrt 9 + \sqrt{16} = 3 + 4 = 7 .
\]
\end{remark}

\begin{method}[Simplificar $\sqrt n$]\label{met:g9:sqrt:simplify}
Para simplificar la \omterm{def:g9:sqrt:def}{raíz cuadrada} de un número entero:
\begin{enumerate}
  \item busca el mayor cuadrado perfecto que divida a $n$ (descompón $n$ si
        hace falta);
  \item separa con la regla del \omterm{def:g2:mult:def}{producto} y saca el cuadrado:
        $\sqrt{k^2 m} = k\sqrt m$;
  \item comprueba que ningún cuadrado perfecto \omterm{def:g9:arith:divisor}{divide} a lo que queda bajo la
        raíz.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:sqrt:simplify}
Simplifica $\sqrt{75}$: como $75 = 25 \times 3$,
\[
\sqrt{75} = \sqrt{25 \times 3} = \sqrt{25} \times \sqrt 3 = 5\sqrt3 .
\]
Simplifica $\sqrt{72}$: como $72 = 36 \times 2$,
$\sqrt{72} = 6\sqrt2$. Y un \omterm{def:g3:division:remainder}{cociente}:
$\sqrt{\dfrac{49}{16}} = \dfrac{\sqrt{49}}{\sqrt{16}} = \dfrac74$.
\end{example}

\begin{example}[Sumar raíces cuadradas]\label{ex:g9:sqrt:add}
Las \omterm{def:g1:addition:def}{sumas} de \omterm{def:g9:sqrt:def}{raíces cuadradas} solo se simplifican cuando las raíces son \emph{semejantes}.
Calcula $\sqrt{12} + \sqrt{27} - \sqrt{48}$, simplificando primero cada término:
\begin{align*}
\sqrt{12} &= \sqrt{4 \times 3} = 2\sqrt3, \\
\sqrt{27} &= \sqrt{9 \times 3} = 3\sqrt3, \\
\sqrt{48} &= \sqrt{16 \times 3} = 4\sqrt3,
\end{align*}
así que la \omterm{def:g1:addition:def}{suma} es $2\sqrt3 + 3\sqrt3 - 4\sqrt3 = (2 + 3 - 4)\sqrt3 = \sqrt3$.
\end{example}

\begin{example}[Desarrollar con raíces cuadradas]\label{ex:g9:sqrt:expand}
Las identidades del álgebra valen para las \omterm{def:g9:sqrt:def}{raíces cuadradas}. Desarrolla
$\left(\sqrt5 + 2\right)^2$ con $(a+b)^2 = a^2 + 2ab + b^2$:
\[
\left(\sqrt5 + 2\right)^2 = 5 + 2 \times 2\sqrt5 + 4 = 9 + 4\sqrt5 .
\]
Y con la tercera identidad, $(a+b)(a-b) = a^2 - b^2$:
\[
\left(\sqrt7 + \sqrt3\right)\left(\sqrt7 - \sqrt3\right) = 7 - 3 = 4 :
\]
el \omterm{def:g2:mult:def}{producto} de dos números irracionales puede ser un número entero.
\end{example}

\section{La ecuación \texorpdfstring{$x^2 = a$}{x2 = a}}

\begin{theorem}[Resolver $x^2 = a$]\label{thm:g9:sqrt:equation}
\begin{enumerate}
  \item si $a > 0$, la \omterm{def:g8:equations:def}{ecuación} $x^2 = a$ tiene exactamente dos soluciones:
        $\sqrt a$ y $-\sqrt a$;
  \item si $a = 0$, la única solución es $0$;
  \item si $a < 0$, no hay solución.
\end{enumerate}
\end{theorem}

\begin{proof}
Reescribe $x^2 = a$ como $x^2 - \left(\sqrt a\right)^2 = 0$ cuando $a \geq 0$,
y factoriza como \omterm{ex:g1:subtraction:difference}{diferencia} de cuadrados:
\[
\left(x - \sqrt a\right)\left(x + \sqrt a\right) = 0 ,
\]
luego $x = \sqrt a$ o $x = -\sqrt a$ (que coinciden cuando $a = 0$). Cuando
$a < 0$ no hay solución, porque $x^2 \geq 0 > a$ para todo $x$.
\end{proof}

\end{omfigure}

\begin{example}
$x^2 = 16$: soluciones $4$ y $-4$. \quad
$x^2 = 5$: soluciones $\sqrt5$ y $-\sqrt5$ (valores exactos; $\sqrt5
\approx 2.236$). \quad
$x^2 = -9$: no hay solución. \quad
$3x^2 = 21$: divide antes entre 3, $x^2 = 7$, soluciones $\pm\sqrt7$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g9:sqrt:1}
Calcula sin calculadora:
\[
\sqrt{64}, \qquad \sqrt{100}, \qquad \sqrt{0.09}, \qquad
\sqrt{\tfrac{1}{25}}, \qquad \left(\sqrt{13}\right)^2, \qquad
\sqrt{(-4)^2} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:sqrt:2}
Simplifica:
\[
\sqrt{50}, \qquad \sqrt{45}, \qquad \sqrt{98}, \qquad \sqrt{300} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:sqrt:3}
Calcula y simplifica:
\[
\sqrt2 \times \sqrt{18}, \qquad
\frac{\sqrt{75}}{\sqrt3}, \qquad
\sqrt5 \times \sqrt{20}, \qquad
\frac{\sqrt{8}}{\sqrt{50}} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:sqrt:4}
Resuelve: $x^2 = 36$; \quad $x^2 = 11$; \quad $x^2 + 4 = 0$; \quad
$2x^2 = 50$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:sqrt:5}
Reduce a un solo término: $\sqrt{20} + \sqrt{45}$, y después
$3\sqrt8 - \sqrt{18} + \sqrt2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:sqrt:6}
Desarrolla y simplifica:
\[
\left(\sqrt3 + 1\right)^2, \qquad
\left(2\sqrt5 - 3\right)^2, \qquad
\left(\sqrt6 + \sqrt2\right)\left(\sqrt6 - \sqrt2\right).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:sqrt:7}
Un campo cuadrado tiene \omterm{def:g6:measure:area}{área} $6400$ m$^2$. ¿Cuánto mide su lado? ¿Y
su diagonal (valor exacto y después redondeado al metro)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:sqrt:8}
Demuestra que $\dfrac{1}{\sqrt2} = \dfrac{\sqrt2}{2}$ (multiplica \omterm{def:g4:fractions:def}{numerador} y
\omterm{def:g4:fractions:def}{denominador} por $\sqrt2$). Usa el mismo truco para escribir $\dfrac{6}{\sqrt3}$
y $\dfrac{10}{\sqrt5}$ sin \omterm{def:g9:sqrt:def}{raíz cuadrada} en el \omterm{def:g4:fractions:def}{denominador}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:sqrt:9}
¿Verdadero o falso? Justifícalo con una demostración o un contraejemplo.
\begin{enumerate}
  \item para todos los $a, b \geq 0$: $\sqrt{ab} = \sqrt a \sqrt b$;
  \item para todos los $a, b \geq 0$: $\sqrt{a+b} = \sqrt a + \sqrt b$;
  \item para todo $x$: $\sqrt{x^2} = x$;
  \item $\left(3\sqrt2\right)^2 = 18$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:sqrt:10}
Sea $x = \sqrt{7 + 4\sqrt3}$. Calcula
$\left(2 + \sqrt3\right)^2$ y deduce una expresión más sencilla de $x$.
La misma pregunta para $\sqrt{9 - 4\sqrt5}$ (apunta a un cuadrado de la forma
$\left(\sqrt5 - 2\right)^2$, y cuidado con el signo).
\end{exercise}

\section{Problema: El número que no es una fracción}

\begin{problem}[{Problema de fin de semana --- la legendaria demostración de que
$\sqrt2$ es irracional, sus muchos primos hermanos y la receta
asombrosamente buena de Herón para aproximarlo}]
\label{pb:g9:sqrt:1}
La diagonal de un cuadrado unidad es una longitud perfectamente real --- la
dibujaste en el \cref{ex:g8:pythagoras:diagonal} --- y, sin embargo, como
los pitagóricos descubrieron con horror hace veinticinco
siglos, \emph{ninguna \omterm{def:g9:fractions:fraction}{fracción} en absoluto} la mide. La leyenda
cuenta que el descubrimiento se castigó con la muerte por ahogamiento. Este problema te lleva
de la mano por la demostración inmortal (cuatro preguntas y es tuya
de por vida), multiplica las víctimas y termina con la receta
que los ingenieros usaron durante dos mil años para domar el número
indomable.

\medskip
\noindent\textbf{Parte I --- La demostración.}
\begin{enumerate}
  \item primero acorrálalo: comprueba que
        $1.4 < \sqrt2 < 1.5$ y después que $1.41 < \sqrt2 < 1.42$,
        elevando al cuadrado las cotas. Una cifra más: ¿entre qué
        números de tres decimales está $\sqrt2$?;
  \item supón ahora --- para llegar a una contradicción --- que
        $\sqrt2 = \frac ab$ para cierta \omterm{def:g9:fractions:fraction}{fracción} \emph{irreducible}
        (\cref{met:g9:arith:simplify}). Eleva al cuadrado los dos miembros
        y demuestra que $a^2 = 2 b^2$. ¿Cuál es la paridad de
        $a^2$?;
  \item los hechos sobre paridad del \cref{pb:g8:pythagoras:1} dicen: los \omterm{def:g3:numbers:evenodd}{números
        impares} tienen cuadrados \omterm{def:g3:numbers:evenodd}{impares}. Deduce que $a$ es \omterm{def:g3:numbers:evenodd}{par}, escribe
        $a = 2k$, sustituye --- y demuestra que $b$ también tiene que ser
        \omterm{def:g3:numbers:evenodd}{par};
  \item ¿dónde está la contradicción? Concluye y enuncia el
        teorema completo: \emph{$\sqrt2$ es irracional --- no es
        \omterm{def:g3:division:remainder}{cociente} de números enteros};
  \item la demostración se apoyó una vez, discretamente, en la palabra
        «irreducible». Señala el paso exacto que se
        vendría abajo sin ella y explica por qué suponer la \omterm{def:g9:fractions:fraction}{fracción}
        irreducible era legítimo de entrada.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Las víctimas se multiplican.}
\begin{enumerate}[resume]
  \item un segundo estilo de demostración, vía descomposiciones en factores primos
        (\cref{thm:g9:arith:factorization}): en $a^2 = 3b^2$,
        compara la paridad del \emph{\omterm{def:g8:powers:def}{exponente} de $3$} en cada
        miembro (\cref{pb:g9:arith:1}: los cuadrados llevan \omterm{def:g8:powers:def}{exponentes}
        \omterm{def:g3:numbers:evenodd}{pares}). Concluye que $\sqrt3$ es irracional;
  \item aplica el mismo argumento de \omterm{def:g8:powers:def}{exponentes} a $a^2 = n b^2$ para un
        número entero $n$ cualquiera: ¿para qué $n$ produce
        una contradicción y para cuáles falla? Enuncia el
        resultado completo: $\sqrt n$ es irracional exactamente
        cuando\,\dots;
  \item demuestra el lema escudo: un racional no nulo por un
        irracional es irracional. (Supón $r x = q$ con $r, q$
        racionales, $r \neq 0$, y despeja $x$.) Deduce que
        $2\sqrt2$ y $\frac{\sqrt2}{2}$ son irracionales;
  \item demuestra que $1 + \sqrt2$ es irracional. Y después la bonita:
        suponiendo que $s = \sqrt2 + \sqrt3$ fuera racional,
        calcula $s^2$, despeja $\sqrt6$ y encuentra la
        contradicción;
  \item modera el entusiasmo: da dos números irracionales cuya
        \emph{\omterm{def:g1:addition:def}{suma}} sea racional, y dos cuyo \emph{\omterm{def:g2:mult:def}{producto}} sea
        racional. (La irracionalidad no se conserva por la aritmética
        --- cada caso necesita su propia demostración.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La receta de Herón.}
Dos mil años antes de las calculadoras, Herón de Alejandría
aproximaba $\sqrt2$ así: \emph{adivina $x$; sustituye la
estimación por la media de $x$ y $\frac2x$; repite.}
\begin{enumerate}[resume]
  \item parte de la estimación $x = 1$ y calcula las dos estimaciones
        siguientes como fracciones exactas;
  \item calcula la tercera estimación, otra vez como \omterm{def:g9:fractions:fraction}{fracción} exacta, y
        su valor decimal. Compara con la pregunta 1: ¿cuántos
        decimales de $\sqrt2$ son ya correctos?;
  \item la idea que hay detrás de la receta: un rectángulo de \omterm{def:g6:measure:area}{área} $2$ con
        un lado $x$ tiene el otro lado $\frac2x$. Demuestra que
        $\sqrt2$ siempre está \emph{entre} $x$ y $\frac2x$
        (considera los dos casos $x^2 > 2$ y $x^2 < 2$), así que
        promediar los dos lados aprieta el rectángulo hacia
        el cuadrado;
  \item las estimaciones $\frac32$, $\frac{17}{12}$,
        $\frac{577}{408}$ esconden una joya: calcula $3^2 - 2 \times
        2^2$, después $17^2 - 2 \times 12^2$ y después
        $577^2 - 2 \times 408^2$. La Parte I demostró que
        $a^2 - 2b^2 = 0$ es imposible --- ¿cuánto se acercan las
        fracciones de Herón a lo imposible?;
  \item final, en dos frases: la diagonal del cuadrado unidad
        existe sobre el papel, ninguna \omterm{def:g9:fractions:fraction}{fracción} la mide y su
        escritura decimal no se puede repetir nunca
        (\cref{pb:g9:fractions:1}). ¿Qué clase de «números nuevos»
        tiene que contener entonces la recta numérica --- y en qué punto de
        esta serie se cuenta su historia completa?
\end{enumerate}
\end{problem}
