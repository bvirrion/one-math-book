\chapter{Trigonometría en el triángulo rectángulo}\label{ch:g9:trig}

En un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, en cuanto conoces un ángulo agudo, la \emph{forma}
del triángulo queda fijada --- solo puede variar su tamaño. Las razones entre sus lados
dependen, por tanto, solo del ángulo: esas razones son el \omterm{def:g9:trig:ratios}{coseno}, el \omterm{def:g9:trig:ratios}{seno} y la
\omterm{def:g9:trig:ratios}{tangente}. Junto con el teorema de Pitágoras, permiten calcular
todos los lados y todos los ángulos de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} a partir de muy poca
información.

\section{El teorema de Pitágoras, otra vez}

\begin{theorem}[Pitágoras]\label{thm:g9:trig:pythagoras}
En un triángulo $ABC$ rectángulo en $A$, el cuadrado de la hipotenusa es la
\omterm{def:g1:addition:def}{suma} de los cuadrados de los dos catetos:
\[
BC^2 = AB^2 + AC^2 .
\]
Recíprocamente, si en un triángulo se cumple $BC^2 = AB^2 + AC^2$, entonces el triángulo es
rectángulo en $A$.
\end{theorem}
\admitted

\begin{example}\label{ex:g9:trig:pythagoras}
Un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} tiene catetos $AB = 5$ y $AC = 12$. Entonces
\[
BC^2 = 5^2 + 12^2 = 25 + 144 = 169,
\qquad\text{luego}\quad BC = \sqrt{169} = 13 .
\]
Para hallar un \emph{cateto}: si $BC = 10$ y $AB = 6$, entonces
$AC^2 = BC^2 - AB^2 = 100 - 36 = 64$, luego $AC = 8$.
\end{example}

\section{Coseno, seno, tangente}

En un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, respecto de un ángulo agudo $\theta$, los tres lados
tienen nombre: la \emph{hipotenusa} (opuesta al \omterm{def:g3:shapes:rightangle}{ángulo recto}, el lado más
largo), el lado \emph{opuesto} a $\theta$ y el lado
\emph{adyacente} a $\theta$ (el cateto que toca a $\theta$).

\begin{definition}[Razones trigonométricas]\label{def:g9:trig:ratios}
Para un ángulo agudo $\theta$ de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}:
\[
\cos\theta = \frac{\text{adyacente}}{\text{hipotenusa}},
\qquad
\sin\theta = \frac{\text{opuesto}}{\text{hipotenusa}},
\qquad
\tan\theta = \frac{\text{opuesto}}{\text{adyacente}} .
\]
Estas razones\index{coseno}\index{seno}\index{tangente} solo dependen de
$\theta$, no del tamaño del triángulo (todos los \omterm{def:g6:shapes:triangles}{triángulos rectángulos} con el
mismo ángulo agudo son escalados unos de otros, por Tales).
\end{definition}

\begin{proposition}[Primeras propiedades]\label{prop:g9:trig:properties}
Para todo ángulo agudo $\theta$:
\begin{enumerate}
  \item $0 < \cos\theta < 1$ y $0 < \sin\theta < 1$ (un cateto es más corto
        que la hipotenusa);
  \item $\cos^2\theta + \sin^2\theta = 1$;
  \item $\tan\theta = \dfrac{\sin\theta}{\cos\theta}$.
\end{enumerate}
\end{proposition}

\begin{proof}
Escribe $a$ para el lado adyacente, $o$ para el lado opuesto y $h$ para la
hipotenusa de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} de ángulo $\theta$.

1. $0 < a < h$ y $0 < o < h$, así que las dos razones están estrictamente entre
$0$ y $1$.

2. Por Pitágoras, $a^2 + o^2 = h^2$. Divide todo entre $h^2$:
\[
\cos^2\theta + \sin^2\theta
= \frac{a^2}{h^2} + \frac{o^2}{h^2}
= \frac{a^2 + o^2}{h^2} = 1 .
\]

3. $\dfrac{\sin\theta}{\cos\theta} = \dfrac{o/h}{a/h} = \dfrac oa
= \tan\theta$.
\end{proof}

\begin{method}[Hallar un lado]\label{met:g9:trig:side}
Para calcular un lado desconocido cuando se conocen un lado y un ángulo agudo:
\begin{enumerate}
  \item marca el ángulo conocido y nombra los tres lados (hipotenusa,
        opuesto, adyacente) \emph{desde ese ángulo};
  \item elige la razón (cos, sen o tg) en la que intervengan el lado conocido
        y el lado buscado;
  \item escribe la \omterm{def:g8:equations:def}{ecuación} y resuélvela;
  \item comprueba que tiene sentido: la hipotenusa tiene que salir la más larga.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:trig:side}
En un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, la hipotenusa mide $8$ y uno de los ángulos es de
$35^\circ$; calcula el lado opuesto a ese ángulo. La razón en la que intervienen el
lado opuesto y la hipotenusa es el \omterm{def:g9:trig:ratios}{seno}:
\[
\sin 35^\circ = \frac{\text{opuesto}}{8}
\quad\Longrightarrow\quad
\text{opuesto} = 8 \sin 35^\circ \approx 8 \times 0.574 \approx 4.6 .
\]
Para el lado adyacente, usa el \omterm{def:g9:trig:ratios}{coseno}:
$8\cos 35^\circ \approx 8 \times 0.819 \approx 6.6$. Comprobación con
Pitágoras: $4.6^2 + 6.6^2 \approx 21 + 43 \approx 64 = 8^2$.
\end{example}

\begin{method}[Hallar un ángulo]\label{met:g9:trig:angle}
Cuando se conocen dos lados, calcula la razón que forman, identifícala como
\omterm{def:g9:trig:ratios}{coseno}, \omterm{def:g9:trig:ratios}{seno} o \omterm{def:g9:trig:ratios}{tangente} del ángulo desconocido y aplica a esa razón la función
inversa de la calculadora ($\cos^{-1}$, $\sin^{-1}$ o $\tan^{-1}$).
\end{method}

\begin{example}\label{ex:g9:trig:angle}
Una escalera de $5$ m se apoya en una pared, con el pie a $1.4$ m de la
pared. El ángulo $\theta$ entre la escalera y el suelo cumple
\[
\cos\theta = \frac{1.4}{5} = 0.28,
\qquad\text{luego}\quad
\theta = \cos^{-1}(0.28) \approx 73.7^\circ .
\]
\end{example}

\begin{example}[Ángulos especiales]\label{ex:g9:trig:special}
Dos triángulos dan valores exactos. La \omterm{def:g3:division:half}{mitad} de un cuadrado de lado 1 (un
\omterm{def:g6:shapes:triangles}{triángulo isósceles} rectángulo) tiene hipotenusa $\sqrt2$ y ángulos de $45^\circ$:
\[
\cos 45^\circ = \sin 45^\circ = \frac{1}{\sqrt2} = \frac{\sqrt2}{2},
\qquad \tan 45^\circ = 1 .
\]
La \omterm{def:g3:division:half}{mitad} de un \omterm{def:g6:shapes:triangles}{triángulo equilátero} de lado 1 da los ángulos $30^\circ$ y
$60^\circ$:
\[
\cos 60^\circ = \sin 30^\circ = \frac12,
\qquad
\cos 30^\circ = \sin 60^\circ = \frac{\sqrt3}{2}.
\]
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g9:trig:1}
Un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} tiene catetos $9$ y $12$. Calcula su hipotenusa. Otro
tiene hipotenusa $17$ y un cateto $8$: calcula el otro cateto.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:2}
Un triángulo tiene lados $7$, $24$ y $25$. ¿Es rectángulo? La misma
pregunta para los lados $5$, $6$ y $8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:3}
En un triángulo $DEF$ rectángulo en $D$, con el ángulo en $E$ llamado
$\theta$: ¿qué lado es la hipotenusa? ¿Qué lado es el opuesto a
$\theta$? Escribe $\cos\theta$, $\sin\theta$ y $\tan\theta$ como razones de
los lados $DE$, $DF$, $EF$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:4}
En un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, la hipotenusa mide $10$ y uno de los ángulos agudos es de
$28^\circ$. Calcula los dos catetos (redondea a la décima;
$\cos 28^\circ \approx 0.883$, $\sin 28^\circ \approx 0.469$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:5}
En un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, el cateto adyacente al ángulo $\theta$ mide $6$
y el cateto opuesto mide $4.5$. Calcula $\tan\theta$ y después $\theta$
(redondea al grado; $\tan^{-1}(0.75) \approx 36.9^\circ$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:6}
Un dron vuela a una altitud de $120$ m. Desde un observador situado en el suelo,
se ve con un ángulo de elevación de $32^\circ$. ¿A qué distancia horizontal
del observador está el dron
($\tan 32^\circ \approx 0.625$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:7}
Una rampa de acceso tiene que subir $1.2$ m con un ángulo de como mucho $5^\circ$ respecto
de la horizontal. ¿Qué longitud mínima a lo largo de la \omterm{prop:g9:linfunc:affinegraph}{pendiente} tiene que tener la rampa
($\sin 5^\circ \approx 0.0872$)? Redondea hacia arriba a la décima de
metro.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:8}
Usando la \cref{prop:g9:trig:properties}: un ángulo agudo $\theta$ cumple
$\cos\theta = 0.6$. Calcula $\sin\theta$ sin hallar $\theta$ y después
$\tan\theta$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:trig:9}
Justifica los valores exactos del \cref{ex:g9:trig:special}: en el
\omterm{def:g6:shapes:triangles}{triángulo isósceles} rectángulo de catetos $1$, calcula la hipotenusa; en el
\omterm{def:g6:shapes:triangles}{triángulo equilátero} de lado $1$, calcula la altura y deduce el
\omterm{def:g9:trig:ratios}{coseno} y el \omterm{def:g9:trig:ratios}{seno} de $30^\circ$ y de $60^\circ$.
\end{exercise}

\section{Problema: Medir lo que no se puede alcanzar}

\begin{problem}[{Problema de fin de semana --- el método topográfico de las dos
estaciones para alturas inaccesibles, las pendientes de las carreteras, la distancia al
horizonte y la carrera de la tangente hacia el infinito}]
\label{pb:g9:trig:1}
Ninguna cinta métrica llega a la cima de una montaña, a lo alto de una
torre al otro lado de un río o al horizonte en el mar --- pero un
medidor de ángulos y las tres razones de este capítulo sí. La pieza central de este
problema es el clásico \emph{método de las dos estaciones} de los topógrafos,
que calcula una altura sin acercarse jamás a su base;
a su alrededor: señales de tráfico, escaleras de mano, escalones, la curvatura de la
Tierra y un primer vistazo a una función que explota hacia el infinito.

\medskip
\noindent\textbf{Parte I --- Una estación.}
(Valores: $\tan 35^\circ \approx 0.700$,
$\sin 35^\circ \approx 0.574$, $\cos 35^\circ \approx 0.819$,
$\tan 1^\circ \approx 0.0175$.)
\begin{enumerate}
  \item desde un punto situado a $60$ m del pie de un faro, se ve
        lo alto con una elevación de $35^\circ$ (medida desde
        la horizontal, a la altura de los ojos). ¿Cuánto sobresale lo alto por encima
        del nivel de los ojos?;
  \item una señal de tráfico anuncia una \omterm{prop:g9:linfunc:affinegraph}{pendiente} del $12\,\%$: la carretera sube
        $12$ m por cada $100$ m de avance horizontal. ¿Qué ángulo forma
        la carretera con la horizontal? ¿Y qué porcentaje de \omterm{prop:g9:linfunc:affinegraph}{pendiente}
        tendría una carretera de $45^\circ$?;
  \item los dos ángulos agudos de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} son
        complementarios, y el lado opuesto de cada uno es el lado
        adyacente del otro. Deduce las identidades «co»:
        \[
        \sin(90^\circ - \theta) = \cos\theta,
        \qquad
        \cos(90^\circ - \theta) = \sin\theta .
        \]
  \item una escalera de mano de $8$ m se apoya formando $60^\circ$ con el suelo.
        Usando los valores exactos del \cref{ex:g9:trig:special},
        da la altura alcanzada y la distancia del pie
        a la pared --- exactamente y después al centímetro;
  \item verifica $\cos^2\theta + \sin^2\theta = 1$ numéricamente para
        $\theta = 35^\circ$ y exactamente para $\theta = 60^\circ$.
        ¿Por qué es esta identidad
        (\cref{prop:g9:trig:properties}) una buena costumbre para
        comprobar lo que hace la calculadora?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Dos estaciones.}
Se ve la cima de una montaña, pero su base es inalcanzable. Desde un
punto $A$, la elevación de la cima es de $30^\circ$; caminando $200$ m
en línea recta hacia la montaña, hasta $B$, la elevación pasa a ser de
$40^\circ$. Escribe $h$ para la altura de la cima sobre el nivel de los ojos
y $x$ para la distancia horizontal de $B$ a la vertical de
la cima. ($\tan 30^\circ \approx 0.577$,
$\tan 40^\circ \approx 0.839$.)
\begin{enumerate}[resume]
  \item expresa $\tan 40^\circ$ y $\tan 30^\circ$ usando $h$,
        $x$ y $x + 200$: dos ecuaciones para dos incógnitas;
  \item a partir de cada \omterm{def:g8:equations:def}{ecuación}, expresa la distancia horizontal en
        función de $h$, resta y deduce la fórmula de las dos
        estaciones:
        \[
        h = \frac{200}
        {\ \dfrac{1}{\tan 30^\circ} - \dfrac{1}{\tan 40^\circ}\ } .
        \]
        calcula $h$ redondeando a la decena de metros;
  \item calcula también $x$ y comprueba que tus dos respuestas
        tienen sentido frente a la visual de $30^\circ$ desde $A$;
  \item en una frase: ¿por qué es indispensable el método de las dos
        estaciones? ¿Qué única medida, aquí imposible,
        habría exigido el método de una estación de la pregunta 1?;
  \item se ve una torre con $30^\circ$ y, tras caminar $200$ m
        hacia ella, con $60^\circ$. Usando
        $\tan 30^\circ = \frac{1}{\sqrt3}$ y
        $\tan 60^\circ = \sqrt3$, calcula la altura
        \emph{exactamente} --- la respuesta es un
        \omterm{def:g9:arith:divisor}{múltiplo} de $\sqrt3$ --- y después
        al metro.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Hasta el horizonte y pared arriba.}
\begin{enumerate}[resume]
  \item ¿a qué distancia está el horizonte? De pie, con los ojos a $h$ metros
        sobre una Tierra esférica de \omterm{def:g3:shapes:shapes}{radio} $R$, tu línea de
        visión roza la esfera: la visual, el \omterm{def:g3:shapes:shapes}{radio} hasta
        el punto de roce y el \omterm{def:g3:shapes:shapes}{radio} hasta tus pies forman un
        \omterm{def:g6:shapes:triangles}{triángulo rectángulo}. Demuestra con Pitágoras
        (\cref{thm:g9:trig:pythagoras}) que la distancia $d$ al
        horizonte cumple $d^2 = 2Rh + h^2 \approx 2Rh$,
        y calcula $d$ para unos ojos a $1.80$ m
        ($R = 6.37 \times 10^6$ m);
  \item la misma pregunta desde lo alto de una torre de $324$ m. (Regla práctica
        de los marinos: el horizonte en kilómetros es de unos
        $3.6\sqrt{h}$ con $h$ en metros --- contrasta con ella tus dos
        respuestas.);
  \item tu pulgar, de unos $2$ cm de ancho, con el brazo estirado,
        a unos $60$ cm del ojo: ¿qué ángulo abarca?
        La Luna llena abarca unos $0.5^\circ$: ¿qué parte de tu
        pulgar tapa la Luna entera?;
  \item una escalera cómoda tiene contrahuellas de $17$ cm y huellas de
        $29$ cm. ¿Con qué ángulo sube? Compáralo con los
        $30^\circ$ del \cref{ex:g9:trig:special} y con una
        escalera de desván empinada de $60^\circ$: ¿qué contrahuella
        necesitaría \emph{esa} con una huella de $29$ cm?;
  \item la carrera de la \omterm{def:g9:trig:ratios}{tangente}: calcula (con calculadora)
        $\tan 10^\circ$, $\tan 45^\circ$, $\tan 80^\circ$,
        $\tan 89^\circ$. ¿Qué pasa cuando el ángulo se acerca a
        $90^\circ$, y por qué (piensa en el lado adyacente)? La pared es
        vertical, la razón no tiene valor y la gráfica se dispara
        hacia el infinito --- tu primera \emph{asíntota}, una criatura
        que se estudia de cerca en el \omterm{def:g6:measure:volume}{volumen} de secundaria superior.
\end{enumerate}
\end{problem}
