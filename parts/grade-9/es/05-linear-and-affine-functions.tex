\chapter{Funciones lineales y afines}\label{ch:g9:linfunc}

"Tres kilogramos cuestan tres veces más que uno": es decir
\omterm{def:g9:linfunc:linear}{proporcionalidad}, la forma más sencilla de relacionar dos cantidades. \omterm{def:g9:linfunc:linear}{lineal
funciones} son \omterm{def:g9:linfunc:linear}{proporcionalidad} en el lenguaje de funciones; \omterm{def:g9:linfunc:affine}{funciones afines} agregue una cantidad inicial fija. Sus gráficas son líneas rectas,

\section{Proporcionalidad y funciones lineales.}

\begin{definition}[función lineal]\label{def:g9:linfunc:linear}
A \emph{función lineal}\index{función lineal} multiplica cada número
por un número fijo $a $:
\[
f(x) = a x .
\]
el número $ a $ es el \emph{coeficiente} de $ f $. Dos cantidades son
\emph{proporcionales}\index{proporcionales} exactamente cuando uno es lineal
función del otro.
\end{definition}

\begin{example}\label{ex:g9:linfunc:linear}
si las manzanas cuestan $3$ por kilogramo, el precio de $ x $ kilogramos es
$ f(x) = 3x $: comprar el doble cuesta el doble y el precio de
$2.5$ kg es $ f(2.5) = 7.5$.

Por el contrario, si un \omterm{def:g9:linfunc:linear}{función lineal} satisface $ f(4) = 10$, entonces
$ a = \frac{10}{4} = 2.5$ y $ f(x) = 2.5x $: \emph{un valor determina un
\omterm{def:g9:linfunc:linear}{función lineal} completamente.}
\end{example}

\begin{proposition}[Gráfica de una función lineal]\label{prop:g9:linfunc:lineargraph}
La gráfica de $ f(x) = ax $ es una linea recta \emph{a través del
origen}.
Por el contrario, cada línea no vertical que pasa por el origen es la gráfica de un
\omterm{def:g9:linfunc:linear}{función lineal}.
\end{proposition}
\admitted

\section{Funciones afines}

\begin{definition}[función afín]\label{def:g9:linfunc:affine}
Un \emph{función afín}\index{función afín} tiene la forma
\[
f(x) = a x + b ,
\]
dónde $ a $ y $ b $ son números fijos. \omterm{def:g9:linfunc:linear}{lineal
funciones} son los especiales
caso $ b = 0$; funciones constantes el caso $ a = 0$.
\end{definition}

\begin{example}
Un gimnasio cobra una tarifa de inscripción de $20$, entonces $5$ por visita: el total
costo de $ x $ visitas es $ f(x) = 5x + 20$. El costo es \emph{no}
\omterm{def:g9:linfunc:linear}{proporcional} al número de visitas (10 visitas no cuestan el doble
como 5), pero cada visita extra suma la misma cantidad, $5$.
\end{example}

\begin{proposition}[Gráfica de una función afín]\label{prop:g9:linfunc:affinegraph}
La gráfica de $ f(x) = ax + b $ es una recta:
\begin{itemize}
  \item $ b = f(0)$ es el \emph{$ y $-interceptar}: la línea cruza el
        eje vertical en $(0, b)$;
  \item $ a $ es el \emph{pendiente}\index{pendiente}: mover una unidad a la
        movimientos correctos $ a $ unidades arriba (abajo si $ a < 0$); la función es
        aumentando cuando $ a > 0$, disminuyendo cuando $ a < 0$;
  \item para dos entradas cualesquiera $ u \neq v $,
        $ a = \dfrac{f(v) - f(u)}{v - u}$.
\end{itemize}
\end{proposition}

\begin{method}[Encontrar una función afín a partir de dos valores]\label{met:g9:linfunc:findaffine}
Dado $ f(u)$ y $ f(v)$ con $ u \neq v $:
\begin{enumerate}
  \item calcular el \omterm{prop:g9:linfunc:affinegraph}{pendiente} $ a = \dfrac{f(v) - f(u)}{v - u}$;
  \item sustituir uno de los dos valores conocidos en $ f(x) = ax + b $ a
        encontrar $ b $;
  \item consultar con el otro valor.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:linfunc:findaffine}
Encuentra el \omterm{def:g9:linfunc:affine}{función afín} con $ f(2) = 7$ y $ f(5) = 16$.
\[
a = \frac{16 - 7}{5 - 2} = \frac93 = 3 ,
\]
entonces $ f(2) = 3 \times 2 + b = 7$ da $ b = 1$: $ f(x) = 3x + 1$. Controlar:
$ f(5) = 15 + 1 = 16$.
\end{example}

\section{Porcentajes}

\begin{proposition}[Cambio porcentual]\label{prop:g9:linfunc:percent}
Aumentar una cantidad en $ t\,\%$ lo multiplica por
$1 + \dfrac{t}{100}$; disminuyéndolo por $ t\,\%$ lo multiplica por
$1 - \dfrac{t}{100}$. Los cambios porcentuales son \omterm{def:g9:linfunc:linear}{funciones lineales}.
\end{proposition}

\begin{proof}
Creciente $ x $ por $ t\,\%$ significa agregar $\frac{t}{100}\,x $:
\[
x + \frac{t}{100}\,x = \left(1 + \frac{t}{100}\right) x .
\]
Mismo cálculo con un signo menos para una disminución.
\end{proof}

\begin{example}\label{ex:g9:linfunc:percent}
un precio de $80$ aumenta en $15\%$: nuevo precio
$80 \times 1.15 = 92$. un precio de $60$ disminuye en $30\%$:
$60 \times 0.7 = 42$.

\emph{Encadenar cambios multiplica los factores.} A $20\%$ aumentar
seguido por un $20\%$ disminución da
$1.2 \times 0.8 = 0.96$: a $4\%$ \emph{disminuir} en general, no es un regreso a
¡el comienzo!
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g9:linfunc:1}
Entre estas funciones, ¿cuáles son lineales? afín? ¿ni?
\[
f(x) = 4x, \qquad
g(x) = 3x - 5, \qquad
h(x) = x^2, \qquad
k(x) = -\tfrac x2, \qquad
m(x) = 7 .
\]
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g9:linfunc:2}
Sea $ f(x) = 2x - 3$. Calcular $ f(0)$, $ f(4)$, $ f(-1)$y encuentra el número
$ x $ tal que $ f(x) = 9$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g9:linfunc:3}
A \omterm{def:g9:linfunc:linear}{función lineal} satisface $ f(6) = 15$. Encuentre su coeficiente, entonces
calcular $ f(10)$ y $ f(-2)$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g9:linfunc:4}
Dibuje en el mismo sistema de coordenadas las gráficas de $ f(x) = 2x $,
$ g(x) = 2x + 3$ y $ h(x) = -x + 4$. ¿Qué se puede decir sobre las gráficas de
$ f $ y $ g $?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g9:linfunc:5}
Encuentra el \omterm{def:g9:linfunc:affine}{función afín} tal que $ f(1) = 5$ y $ f(3) = 11$; entonces el
uno tal que $ g(0) = 4$ y $ g(2) = 0$.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g9:linfunc:6}
Calcular: el precio después de un $25\%$ aumentar en $120$; el precio después de un
$40\%$ descuento en $65$; el precio original si un artículo cuesta $69$
después de un $25\%$ descuento.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g9:linfunc:7}
Un plan telefónico A cuesta $10$ por mes más $0.05$ por minuto de llamadas; planear
costos B $25$ por mes con llamadas ilimitadas.
\begin{enumerate}
  \item Exprese el costo mensual del plan A en función del número
        $ x $ de minutos.
  \item ¿A partir de cuántos minutos al mes sale más barato el plan B?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g9:linfunc:8}
una población de $2000$ las bacterias aumentan en $10\%$ cada hora.
\begin{enumerate}
  \item ¿Cuántas bacterias hay después de una hora? ¿Después de dos horas?
  \item Explica por qué la respuesta después de dos horas no es $2400$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g9:linfunc:9}
La gráfica de un \omterm{def:g9:linfunc:affine}{función afín} pasa por los puntos $(2, 3)$ y
$(6, 1)$.
\begin{enumerate}
  \item Calcular su \omterm{prop:g9:linfunc:affinegraph}{pendiente}. ¿La función es creciente o decreciente?
  \item Dé su expresión y calcule la entrada para la cual la salida
        es $0$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g9:linfunc:10}
Una tienda primero aumenta el precio en $ t\,\%$, luego reduce el nuevo precio en
$ t\,\%$.
\begin{enumerate}
  \item Demuestre que el precio final es igual al original multiplicado por
        $1 - \dfrac{t^2}{10000}$.
  \item Deduzca que el precio final es siempre \emph{más bajo} que el
        originales (para $0 < t \leq 100$), y encuentre el porcentaje general
        cambiar por $ t = 10$.
\end{enumerate}
\end{exercise}

\section{Problema: Los dos termómetros}

\begin{problem}[{Problema de fin de semana --- Celsius, Fahrenheit y las funciones afines de la vida diaria: una temperatura se lee igual en ambas escalas y la pendiente constante es la firma de la rectitud}]

por un \omterm{def:g9:linfunc:affine}{función afín} --- y construirlo a partir de dos hechos es
