\chapter{Funciones lineales y afines}\label{ch:g9:linfunc}

«Tres kilogramos cuestan el triple que uno»: eso es la
\omterm{def:g9:linfunc:linear}{proporcionalidad}, la manera más sencilla en que se pueden relacionar dos magnitudes. Las \omterm{def:g9:linfunc:linear}{funciones
lineales} son la \omterm{def:g9:linfunc:linear}{proporcionalidad} en el lenguaje de las funciones; las \omterm{def:g9:linfunc:affine}{funciones
afines} añaden una cantidad inicial fija. Sus gráficas son rectas,
y leer esas rectas es la destreza que construye este capítulo.

\section{Proporcionalidad y funciones lineales}

\begin{definition}[Función lineal]\label{def:g9:linfunc:linear}
Una \emph{función lineal}\index{función lineal} multiplica todos los números
por un número fijo $a$:
\[
f(x) = a x .
\]
El número $a$ es el \emph{coeficiente} de $f$. Dos magnitudes son
\emph{proporcionales}\index{proporcionalidad} exactamente cuando una es función
lineal de la otra.
\end{definition}

\begin{example}\label{ex:g9:linfunc:linear}
Si las manzanas cuestan $3$ por kilogramo, el precio de $x$ kilogramos es
$f(x) = 3x$: comprar el doble cuesta el doble, y el precio de
$2.5$ kg es $f(2.5) = 7.5$.

Recíprocamente, si una \omterm{def:g9:linfunc:linear}{función lineal} cumple $f(4) = 10$, entonces
$a = \frac{10}{4} = 2.5$ y $f(x) = 2.5x$: \emph{un solo valor determina por completo
una \omterm{def:g9:linfunc:linear}{función lineal}.}
\end{example}

\begin{proposition}[Gráfica de una función lineal]\label{prop:g9:linfunc:lineargraph}
La gráfica de $f(x) = ax$ es una recta \emph{que pasa por el origen}.
Recíprocamente, toda recta no vertical que pasa por el origen es la gráfica de una
\omterm{def:g9:linfunc:linear}{función lineal}.
\end{proposition}
\admitted

\section{Las funciones afines}

\begin{definition}[Función afín]\label{def:g9:linfunc:affine}
Una \emph{función afín}\index{función afín} tiene la forma
\[
f(x) = a x + b ,
\]
donde $a$ y $b$ son números fijos. Las \omterm{def:g9:linfunc:linear}{funciones lineales} son el caso
particular $b = 0$; las funciones constantes, el caso $a = 0$.
\end{definition}

\begin{example}
Un gimnasio cobra una cuota de inscripción de $20$ y después $5$ por visita: el coste
total de $x$ visitas es $f(x) = 5x + 20$. El coste \emph{no} es
proporcional al número de visitas (10 visitas no cuestan el doble
que 5), pero cada visita de más añade la misma cantidad, $5$.
\end{example}

\begin{proposition}[Gráfica de una función afín]\label{prop:g9:linfunc:affinegraph}
La gráfica de $f(x) = ax + b$ es una recta:
\begin{itemize}
  \item $b = f(0)$ es la \emph{ordenada en el origen}: la recta corta el
        eje vertical en $(0, b)$;
  \item $a$ es la \emph{pendiente}\index{pendiente}: avanzar una unidad hacia la
        derecha sube $a$ unidades (baja si $a < 0$); la función es
        creciente cuando $a > 0$ y decreciente cuando $a < 0$;
  \item para dos entradas cualesquiera $u \neq v$,
        $a = \dfrac{f(v) - f(u)}{v - u}$.
\end{itemize}
\end{proposition}

\begin{proof}[Demostración del último punto]
$f(v) - f(u) = (av + b) - (au + b) = a(v - u)$; divide entre $v - u$. El
\omterm{def:g3:division:remainder}{resto} se admite a este nivel (el \omterm{def:g6:measure:volume}{volumen} de secundaria superior da un tratamiento
completo de las rectas y sus ecuaciones).
\end{proof}

\begin{method}[Hallar una función afín a partir de dos valores]\label{met:g9:linfunc:findaffine}
Dados $f(u)$ y $f(v)$ con $u \neq v$:
\begin{enumerate}
  \item calcula la \omterm{prop:g9:linfunc:affinegraph}{pendiente} $a = \dfrac{f(v) - f(u)}{v - u}$;
  \item sustituye uno de los dos valores conocidos en $f(x) = ax + b$ para
        hallar $b$;
  \item comprueba con el otro valor.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:linfunc:findaffine}
Halla la \omterm{def:g9:linfunc:affine}{función afín} con $f(2) = 7$ y $f(5) = 16$.
\[
a = \frac{16 - 7}{5 - 2} = \frac93 = 3 ,
\]
y después $f(2) = 3 \times 2 + b = 7$ da $b = 1$: $f(x) = 3x + 1$. Comprobación:
$f(5) = 15 + 1 = 16$.
\end{example}

\section{Los porcentajes}

\begin{proposition}[Variación porcentual]\label{prop:g9:linfunc:percent}
Aumentar una magnitud un $t\,\%$ la multiplica por
$1 + \dfrac{t}{100}$; disminuirla un $t\,\%$ la multiplica por
$1 - \dfrac{t}{100}$. Las variaciones porcentuales son \omterm{def:g9:linfunc:linear}{funciones lineales}.
\end{proposition}

\begin{proof}
Aumentar $x$ un $t\,\%$ es sumarle $\frac{t}{100}\,x$:
\[
x + \frac{t}{100}\,x = \left(1 + \frac{t}{100}\right) x .
\]
El mismo cálculo con un signo menos para una disminución.
\end{proof}

\begin{example}\label{ex:g9:linfunc:percent}
Un precio de $80$ aumenta un $15\%$: precio nuevo
$80 \times 1.15 = 92$. Un precio de $60$ disminuye un $30\%$:
$60 \times 0.7 = 42$.

\emph{Encadenar variaciones multiplica los factores.} Un aumento del $20\%$
seguido de una bajada del $20\%$ da
$1.2 \times 0.8 = 0.96$: ¡una \emph{bajada} del $4\%$ en total, y no una vuelta
al punto de partida!
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:1}
Entre estas funciones, ¿cuáles son lineales? ¿Cuáles afines? ¿Cuáles ninguna de las dos?
\[
f(x) = 4x, \qquad
g(x) = 3x - 5, \qquad
h(x) = x^2, \qquad
k(x) = -\tfrac x2, \qquad
m(x) = 7 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:2}
Sea $f(x) = 2x - 3$. Calcula $f(0)$, $f(4)$, $f(-1)$ y halla el número
$x$ tal que $f(x) = 9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:3}
Una \omterm{def:g9:linfunc:linear}{función lineal} cumple $f(6) = 15$. Halla su coeficiente y calcula después
$f(10)$ y $f(-2)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:4}
Dibuja en el mismo sistema de coordenadas las gráficas de $f(x) = 2x$,
$g(x) = 2x + 3$ y $h(x) = -x + 4$. ¿Qué se puede decir de las gráficas de
$f$ y $g$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:5}
Halla la \omterm{def:g9:linfunc:affine}{función afín} tal que $f(1) = 5$ y $f(3) = 11$; y después la
que cumple $g(0) = 4$ y $g(2) = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:6}
Calcula: el precio tras un aumento del $25\%$ sobre $120$; el precio tras un
descuento del $40\%$ sobre $65$; el precio inicial si un artículo cuesta $69$
tras un descuento del $25\%$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:7}
Una tarifa de teléfono A cuesta $10$ al mes más $0.05$ por minuto de llamadas; la tarifa
B cuesta $25$ al mes con llamadas ilimitadas.
\begin{enumerate}
  \item expresa el coste mensual de la tarifa A en función del número
        $x$ de minutos;
  \item ¿a partir de cuántos minutos al mes sale más barata la tarifa B?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:8}
Una población de $2000$ bacterias aumenta un $10\%$ cada hora.
\begin{enumerate}
  \item ¿cuántas bacterias hay al cabo de una hora? ¿Y de dos horas?;
  \item explica por qué la respuesta al cabo de dos horas no es $2400$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:9}
La gráfica de una \omterm{def:g9:linfunc:affine}{función afín} pasa por los puntos $(2, 3)$ y
$(6, 1)$.
\begin{enumerate}
  \item calcula su \omterm{prop:g9:linfunc:affinegraph}{pendiente}. ¿La función es creciente o decreciente?;
  \item da su expresión y calcula la entrada para la que la salida
        es $0$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:linfunc:10}
Una tienda sube primero un precio un $t\,\%$ y después baja el precio nuevo un
$t\,\%$.
\begin{enumerate}
  \item demuestra que el precio final es igual al inicial multiplicado por
        $1 - \dfrac{t^2}{10000}$;
  \item deduce que el precio final siempre es \emph{menor} que el
        inicial (para $0 < t \leq 100$), y halla la variación porcentual
        total para $t = 10$.
\end{enumerate}
\end{exercise}

\section{Problema: Los dos termómetros}

\begin{problem}[{Problema de fin de semana --- Celsius, Fahrenheit y las
funciones afines de la vida diaria: una temperatura se lee igual en
las dos escalas, y la pendiente constante es la firma de la rectitud}]
\label{pb:g9:linfunc:1}
Los estadounidenses oyen «$68$ grados» y se visten ligeros; los europeos lo oyen
y agarran un abrigo. Los dos termómetros del mundo están enlazados
por una \omterm{def:g9:linfunc:affine}{función afín} --- y construirla a partir de dos datos es
exactamente la destreza del \cref{met:g9:linfunc:findaffine}. Este
problema construye la conversión, halla la inquietante temperatura en la
que las dos escalas coinciden, después lee taxis, grillos y
velas con las mismas gafas, y termina con la prueba
que reconoce la rectitud misma.

\medskip
\noindent\textbf{Parte I --- Construir la conversión.}
Dos datos fijan la escala Fahrenheit: el agua se congela a
$32\,^\circ$F y hierve a $212\,^\circ$F (a $0\,^\circ$C y
$100\,^\circ$C).
\begin{enumerate}
  \item halla la \omterm{def:g9:linfunc:affine}{función afín} $F = f(C)$ que convierte los grados Celsius
        en Fahrenheit (\cref{met:g9:linfunc:findaffine} con
        los valores $f(0) = 32$ y $f(100) = 212$);
  \item convierte a Fahrenheit: un día templado de $20\,^\circ$C; los
        $37\,^\circ$C del cuerpo; unos gélidos $-10\,^\circ$C;
  \item despeja $C$ para construir la conversión de vuelta y úsala
        con $50\,^\circ$F y con $98.6\,^\circ$F;
  \item ¿es $f$ una función \emph{lineal}
        (\cref{def:g9:linfunc:linear})? Prueba las señales delatoras:
        ¿cuánto vale $f(0)$ y doblar la lectura Celsius
        dobla la Fahrenheit? Concluye con la palabra correcta
        (\cref{def:g9:linfunc:affine});
  \item el acertijo clásico: ¿hay una temperatura que se lea con
        \emph{el mismo número} en las dos escalas? Resuelve
        $f(x) = x$ y describe qué significa tu solución en la
        gráfica de $f$ (¿qué recta corta ahí la gráfica?).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Las \omterm{def:g9:linfunc:affine}{funciones afines} en libertad.}
\begin{enumerate}[resume]
  \item el taxi A cobra $3$ euros más $1.50$ por km; el taxi B
        cobra $2$ euros por km a secas. Escribe las dos funciones
        de precio y compáralas para un trayecto de $4$ km y otro de
        $8$ km;
  \item halla la distancia de equilibrio en la que los dos taxis cuestan
        lo mismo y describe la situación en una gráfica (dos
        rectas --- ¿qué pasa en el punto de corte, antes de él
        y después?);
  \item regla práctica de los naturalistas: un grillo canta unas
        $N = 7T - 30$ veces por minuto a la temperatura
        $T\,^\circ$C. ¿Cuántos cantos en una tarde de $20\,^\circ$C?
        Cuentas $82$ cantos en un minuto: ¿qué
        temperatura anuncia el grillo?;
  \item un teléfono comprado por $600$ euros pierde $15$ euros de valor al
        mes: escribe la función de valor $v(t)$, halla cuándo el
        teléfono vale $240$ euros y cuándo el modelo dice que no
        vale nada. ¿La fórmula sigue significando algo
        después de esa fecha?;
  \item las variaciones porcentuales son \omterm{def:g9:linfunc:linear}{funciones lineales}: $+25\,\%$ es
        multiplicar por $1.25$, y $-20\,\%$ por $0.8$
        (\cref{prop:g9:linfunc:percent}). Compónlas: ¿qué le
        hace a un precio una subida del $+25\,\%$ seguida de una bajada del $-20\,\%$?
        Explica el pequeño milagro y compara con el
        \cref{exo:g9:linfunc:10}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La \omterm{prop:g9:linfunc:affinegraph}{pendiente} lo es todo.}
\begin{enumerate}[resume]
  \item una vela mide $20$ cm al encenderla y $17$ cm media
        hora después. Suponiendo que se derrite de forma afín, halla $h(t)$ (altura
        en cm, $t$ en minutos) y predice cuándo se apaga la vela;
  \item dos ciclistas recorren la misma carretera: la distancia del primero es
        $d_1(t) = 24t$ (km al cabo de $t$ horas) y la del segundo,
        $d_2(t) = 10 + 18t$. ¿Quién salió con ventaja, quién
        va más rápido y en qué momento y kilómetro alcanza el
        primero al segundo?;
  \item sean $f(x) = 2x + 3$ y $g(x) = -0.5x + 7$. Calcula
        $f(x) + g(x)$ y comprueba que vuelve a ser afín. ¿Cuáles son la
        \omterm{prop:g9:linfunc:affinegraph}{pendiente} y la ordenada en el origen de una \omterm{def:g1:addition:def}{suma}, en general?;
  \item discute por completo la \omterm{def:g8:equations:def}{ecuación} $mx + p = 0$: ¿cuántas
        soluciones hay cuando $m \neq 0$ (y cuál es)? ¿Y cuando $m = 0$
        y $p \neq 0$? ¿Y cuando $m = 0$ y $p = 0$? (Cada caso
        tiene su historia gráfica: ¿dónde corta al eje horizontal una recta
        de \omterm{prop:g9:linfunc:affinegraph}{pendiente} $m$?);
  \item la prueba de alineación: ¿están los puntos $(1, 5)$, $(3, 9)$,
        $(7, 17)$ sobre una misma recta? ¿Y $(1, 5)$, $(3, 9)$,
        $(6, 16)$? Calcula las tasas de variación entre parejas y
        decide. Enuncia la moraleja: una \emph{tasa de variación
        constante} es la firma de las \omterm{def:g9:linfunc:affine}{funciones afines} --- las gráficas
        curvas cambian de tasa, y medir \emph{eso} es
        historia para el \omterm{def:g6:measure:volume}{volumen} de secundaria superior.
\end{enumerate}
\end{problem}
