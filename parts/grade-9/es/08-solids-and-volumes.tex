\chapter{Cuerpos y volúmenes}\label{ch:g9:solids}

La geometría espacial comienza con una pequeña familia de \omterm{def:g5:solids:def}{sólidos} --- \omterm{def:g7:areas:prism}{prismas},
\omterm{def:g7:areas:prism}{cilindros}, \omterm{def:g8:solids:def}{pirámides}, \omterm{def:g8:solids:def}{conos}, esferas --- y dos preguntas: ¿cuánto
ellos sostienen (\omterm{def:g6:measure:volume}{volumen}), y ¿qué ves cuando las cortas (secciones)?
El capítulo cierra con un hecho de gran importancia práctica: ampliar una
\omterm{def:g5:solids:def}{sólido} por $k $ multiplica su \omterm{def:g6:measure:volume}{volumen} por $ k^3$.

\section{Los sólidos clásicos y sus volúmenes.}

\begin{theorem}[Fórmulas de volumen]\label{thm:g9:solids:volumes}
Escribir $ B $ para el \omterm{def:g6:measure:area}{área} de la base y $ h $ para la altura (la distancia
entre la base y el opuesto \omterm{def:g5:solids:def}{rostro} o ápice).
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{l|l}
\omterm{def:g5:solids:def}{sólido} & \omterm{def:g6:measure:volume}{volumen} \\
\hline
\omterm{def:g7:areas:prism}{prisma}, \omterm{def:g7:areas:prism}{cilindro} & $ V = B \times h $ \\[5pt]
\omterm{def:g8:solids:def}{pirámide}, \omterm{def:g8:solids:def}{cono} & $ V = \dfrac13\, B \times h $ \\[5pt]
sphere of \omterm{def:g3:shapes:shapes}{radio} $ r $ & $ V = \dfrac43\,\pi r^3$ \\
\end{tabular}
\end{center}
por un \omterm{def:g7:areas:prism}{cilindro} y un \omterm{def:g8:solids:def}{cono} de \omterm{def:g3:shapes:shapes}{radio} $ r $, la base \omterm{def:g6:measure:area}{área} es $ B = \pi r^2$;
la superficie de la esfera \omterm{def:g6:measure:area}{área} es $4\pi r^2$.
\end{theorem}
\admitted

\begin{example}\label{ex:g9:solids:volumes}
Una lata cilíndrica tiene \omterm{def:g3:shapes:shapes}{radio} $4$ cm y altura $10$ centímetro:
\[
V = \pi r^2 h = \pi \times 16 \times 10 = 160\pi \approx 503 \text{ cm}^3
\approx 0.5 \text{ L}.
\]
A \omterm{def:g8:solids:def}{cono} con la misma base y altura contiene un tercio de eso:
$\frac{160\pi}{3} \approx 168$ centímetro $^3$. una esfera de \omterm{def:g3:shapes:shapes}{radio} $4$ centímetro:
$ V = \frac43 \pi \times 64 = \frac{256\pi}{3} \approx 268$ centímetro $^3$.
\end{example}

\begin{example}[Una pirámide paso a paso]\label{ex:g9:solids:pyramid}
A \omterm{def:g8:solids:def}{pirámide} tiene una base cuadrada de lado $6$ m y altura $5$ metro.
\begin{enumerate}
  \item Base \omterm{def:g6:measure:area}{área}: $ B = 6^2 = 36$ metro $^2$.
  \item \omterm{def:g6:measure:volume}{Volumen}: $ V = \frac13 \times 36 \times 5 = 60$ metro $^3$.
\end{enumerate}
Cuidado con las unidades: si el lado estuviera en cm y la altura en m, uno de
primero habría que convertirlos.
\end{example}

\section{Secciones por planos}

\begin{proposition}[Secciones de los sólidos clásicos.]\label{prop:g9:solids:sections}
Cortar un \omterm{def:g5:solids:def}{sólido} por un plano produce una figura plana, su
\emph{sección transversal}\index{sección transversal}:
\begin{itemize}
  \item a \omterm{def:g7:areas:prism}{prisma} o \omterm{def:g7:areas:prism}{cilindro} cortar \emph{\omterm{def:g6:lines:perp}{paralelo} a su base} da una copia
        de la base, en cada altura;
  \item a \omterm{def:g8:solids:def}{pirámide} o \omterm{def:g8:solids:def}{cono} cortar \omterm{def:g6:lines:perp}{paralelo} a su base le da un
        \emph{reducción} de la base: a distancia $ d $ desde el ápice,
        el \omterm{def:g9:thales:scaling}{factor de escala} es $ k = \frac{d}{h}$;
  \item una esfera de \omterm{def:g3:shapes:shapes}{radio} $ r $ cortado por un avión a distancia $ d < r $ de
        el centro da un círculo de \omterm{def:g3:shapes:shapes}{radio} $\sqrt{r^2 - d^2}$ (por el
        Teorema de Pitágoras).
\end{itemize}
\end{proposition}
\admitted

\begin{example}\label{ex:g9:solids:section}
A \omterm{def:g8:solids:def}{cono} tiene base \omterm{def:g3:shapes:shapes}{radio} $6$ y altura $9$. La sección por un avión.
\omterm{def:g6:lines:perp}{paralelo} a la base, a distancia $3$ desde el ápice, es un disco escalado por
$ k = \frac39 = \frac13$: es \omterm{def:g3:shapes:shapes}{radio} es $6 \times \frac13 = 2$.

una esfera de \omterm{def:g3:shapes:shapes}{radio} $5$ es cortado por un avión a una distancia $3$ de su
centro: la sección es un círculo de \omterm{def:g3:shapes:shapes}{radio} $\sqrt{25 - 9} = 4$.
\end{example}

\section{Sólidos escalados}

\begin{theorem}[Efecto de una escala sobre longitudes, áreas, volúmenes.]\label{thm:g9:solids:scaling}
cuando un \omterm{def:g5:solids:def}{sólido} se amplía o reduce mediante el \omterm{def:g9:thales:scaling}{factor de escala} $ k > 0$:
\begin{itemize}
  \item todas las longitudes se multiplican por $ k $;
  \item todo \omterm{def:g6:measure:area}{áreas} se multiplican por $ k^2$;
  \item todo \omterm{def:g6:measure:volume}{volúmenes} se multiplican por $ k^3$.
\end{itemize}
\end{theorem}
\admitted

\begin{example}\label{ex:g9:solids:scaling}
Un modelo de auto en \omterm{def:g7:prop:scale}{escala} $\frac{1}{18}$: las longitudes son las del coche real
multiplicado por $ k = \frac{1}{18}$, superficie pintada multiplicada por
$ k^2 = \frac{1}{324}$, y \omterm{def:g6:measure:volume}{volumen} por $ k^3 = \frac{1}{5832}$.

Duplicando el \omterm{def:g3:shapes:shapes}{radio} de una esfera ($ k = 2$) multiplica su \omterm{def:g6:measure:volume}{volumen} por
$2^3 = 8$: verifique la fórmula,
$\frac43\pi (2r)^3 = \frac43 \pi \times 8r^3$.
\end{example}

\begin{example}[Cono truncado]\label{ex:g9:solids:truncated}
A \omterm{def:g8:solids:def}{cono} de altura $9$ y base \omterm{def:g3:shapes:shapes}{radio} $6$ (\omterm{def:g6:measure:volume}{volumen}
$\frac13 \pi \times 36 \times 9 = 108\pi $) se corta a distancia $3$ de
el ápice y el pequeño \omterm{def:g8:solids:def}{cono} encima se retira el corte. el pequeño \omterm{def:g8:solids:def}{cono} es
la reducción por $ k = \frac13$, entonces es \omterm{def:g6:measure:volume}{volumen} es
$108\pi \times \left(\frac13\right)^3 = 4\pi $. El restante \omterm{def:g5:solids:def}{sólido} (a
\emph{truncado \omterm{def:g8:solids:def}{cono}}) tiene \omterm{def:g6:measure:volume}{volumen} $108\pi - 4\pi = 104\pi \approx 327$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g9:solids:1}
Calcular el \omterm{def:g6:measure:volume}{volúmenes}: a \omterm{def:g5:solids:def}{caja} (rectangular \omterm{def:g7:areas:prism}{prisma}) de dimensiones $4 \times 5
\times 12$; a \omterm{def:g7:areas:prism}{cilindro} de \omterm{def:g3:shapes:shapes}{radio} $3$ y altura $7$ (valor exacto con
$\pi $, luego redondeado a la unidad).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g9:solids:2}
Calcular el \omterm{def:g6:measure:volume}{volumen} de un \omterm{def:g8:solids:def}{cono} de \omterm{def:g3:shapes:shapes}{radio} $5$ y altura $12$, y de un
\omterm{def:g8:solids:def}{pirámide} con base rectangular $8 \times 3$ y altura $10$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g9:solids:3}
Calcular el \omterm{def:g6:measure:volume}{volumen} y la superficie \omterm{def:g6:measure:area}{área} de una esfera de \omterm{def:g3:shapes:shapes}{radio} $6$ (exacto
valores con $\pi $).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g9:solids:4}
una esfera de \omterm{def:g3:shapes:shapes}{radio} $10$ es cortado por un avión a una distancia $8$ de su
centro. cual es el \omterm{def:g3:shapes:shapes}{radio} de la sección circular?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g9:solids:5}
A \omterm{def:g8:solids:def}{cono} tiene base \omterm{def:g3:shapes:shapes}{radio} $8$ y altura $12$. un avion \omterm{def:g6:lines:perp}{paralelo} a la base
lo corta a distancia $9$ desde el ápice. Calcular el \omterm{def:g3:shapes:shapes}{radio} de la
sección, entonces el \omterm{def:g6:measure:volume}{volumen} de los pequeños \omterm{def:g8:solids:def}{cono} encima del corte.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g9:solids:6}
Un vaso cilíndrico de interior. \omterm{def:g3:shapes:shapes}{radio} $3$ cm contiene agua hasta una altura de
$10$ centímetro. una bola de \omterm{def:g3:shapes:shapes}{radio} $2$ cm está completamente sumergido en él. por cuanto
¿Sube el nivel del agua? (El agregado \omterm{def:g6:measure:volume}{volumen} se extiende sobre el vaso
base \omterm{def:g6:measure:area}{área}; dar el aumento exacto, entonces \omterm{def:g4:numbers:round}{redondo} al milímetro.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g9:solids:7}
Una receta llena un molde esférico de \omterm{def:g3:shapes:shapes}{radio} $6$ centímetro. solo tienes un
molde esférico de \omterm{def:g3:shapes:shapes}{radio} $3$ centímetro. ¿Cuántos moldes pequeños puedes llenar?
(Responde sin computar tampoco \omterm{def:g6:measure:volume}{volumen} explícitamente.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g9:solids:8}
el grande \omterm{def:g8:solids:def}{Pirámide} de Giza tiene una base cuadrada de lado aproximadamente $230$ m y un
altura de aproximadamente $147$ metro. Estimar su \omterm{def:g6:measure:volume}{volumen}, y expresarlo en millones
de metros cúbicos.
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g9:solids:9}
Un embudo en forma de cono de \omterm{def:g3:shapes:shapes}{radio} $6$ cm y altura $12$ cm, con el ápice hacia abajo, es
lleno de agua hasta \omterm{def:g3:division:half}{medio} de su \emph{altura} (medido desde el
ápice).
\begin{enumerate}
  \item Qué \omterm{def:g9:fractions:fraction}{fracción} del embudo \omterm{def:g6:measure:volume}{volumen} esta lleno?
  \item Si en cambio está lleno de \omterm{def:g3:division:half}{medio} de su \emph{\omterm{def:g6:measure:volume}{volumen}} de agua,
        demuestre que la altura del agua $ d $ satisface
        $\left(\frac{d}{12}\right)^3 = \frac12$ y dar $d $ al
        milímetro ($\sqrt[3]{0.5} \approx 0.7937$).
\end{enumerate}
\end{exercise}

\section{Problema: la lápida de Arquímedes}

\begin{problem}[{Problema de fin de semana --- la esfera es dos tercios de su cilindro (el doble), la pila 1:2:3 y la ley del cuadrado-cubo que prohíbe a los gigantes}]
\label{pb:g9:solids:1}
Arquímedes demostró muchos teoremas, pero uno lo enorgulleció tanto que
él pidió su \emph{cifra} para ser tallado en su tumba: una esfera
anidado en su forma más estrecha \omterm{def:g7:areas:prism}{cilindro}. Un siglo después el escritor romano
Cicerón, mientras buscaba entre las zarzas cerca de Siracusa, reconoció el
tumba ``por la esfera y el \omterm{def:g7:areas:prism}{cilindro}''. Este problema recupera
lo que codifica la talla --- un doble milagro de dos tercios --- entonces
sigue \omterm{def:g6:measure:volume}{volúmenes} y superficies (\cref{thm:g9:solids:volumes},
\cref{thm:g9:solids:scaling}) a una ley que gobierna a los gigantes, las hormigas
y planetas enfriados.

\medskip
\noindent\textbf{Parte I --- The carving decoded.}
una esfera de \omterm{def:g3:shapes:shapes}{radio} $ r $ se encuentra exactamente dentro de un \omterm{def:g7:areas:prism}{cilindro}: mismo
\omterm{def:g3:shapes:shapes}{radio}, altura $2r $.
\begin{enumerate}
  \item Calcular el \omterm{def:g7:areas:prism}{cilindro}'s \omterm{def:g6:measure:volume}{volumen} en términos de $ r $.
  \item Calcular la razón de la esfera \omterm{def:g6:measure:volume}{volumen} hacia
        \omterm{def:g7:areas:prism}{cilindro}'s. ¿Por qué le habría gustado a Arquímedes que el
        la respuesta no contiene $\pi $ y no $ r $?
  \item Ahora las superficies: compara la superficie de la esfera. \omterm{def:g6:measure:area}{área} con el \omterm{def:g7:areas:prism}{cilindro}'s \emph{total} superficie (parte lateral más
        las dos tapas). ¿Qué proporción aparece --- otra vez?
  \item El espacio vacío entre la esfera y \omterm{def:g7:areas:prism}{cilindro} tiene \omterm{def:g6:measure:volume}{volumen}
        $2\pi r^3 - \frac43\pi r^3$. Demuestra que este sobrante
        es exactamente igual a \omterm{def:g6:measure:volume}{volumen} de \emph{dos \omterm{def:g8:solids:def}{conos}} de \omterm{def:g3:shapes:shapes}{radio}
        $ r $ y altura $ r $.
  \item una pelota de baloncesto de \omterm{def:g3:shapes:shapes}{radio} $12$ cm se vende en el más ajustado
        cilíndrico \omterm{def:g5:solids:def}{caja}. Calcular ambos \omterm{def:g6:measure:volume}{volúmenes} (al más cercano
        $100$ centímetro $^3$) y el porcentaje de \omterm{def:g5:solids:def}{caja} eso es
        vacío.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Bowls, molds and moons.}
\begin{enumerate}[resume]
  \item Un cuenco semiesférico tiene \omterm{def:g3:shapes:shapes}{radio} $10$ centímetro. Calcular su
        \omterm{def:g2:measure:units}{capacidad}, en centímetros $^3$ y en litros (al decilitro).
  \item La pila 1:2:3: una \omterm{def:g8:solids:def}{cono}, un hemisferio y un \omterm{def:g7:areas:prism}{cilindro},
        todos \omterm{def:g3:shapes:shapes}{radio} $10$ cm y altura $10$ centímetro. Calcular el
        tres \omterm{def:g6:measure:volume}{volúmenes} y verificar la proporción legendaria
        $1 : 2 : 3$. (Arquímedes habría apreciado esto
        uno también.)
  \item Una esfera de chocolate de \omterm{def:g3:shapes:shapes}{radio} $3$ cm se funde en un
        molde cilíndrico de \omterm{def:g3:shapes:shapes}{radio} $3$ centímetro. ¿Qué altura tiene el
        alcance del chocolate? (Exacto \omterm{def:g9:fractions:fraction}{fracción}, luego a la
        milímetro.)
  \item la tierra \omterm{def:g3:shapes:shapes}{radio} se trata de $6\,371$ km, la Luna
        acerca de $1\,737$ km. Calcule la razón de los radios, luego
        --- con \cref{thm:g9:solids:scaling} --- la proporción de
        el \omterm{def:g6:measure:volume}{volúmenes}: cuantas lunas caben en un hueco
        Tierra, por \omterm{def:g6:measure:volume}{volumen}?
  \item Del mismo teorema: ¿cuál es la relación Tierra-Luna de
        \emph{superficie \omterm{def:g6:measure:area}{áreas}}? Y en general, cuando un globo
        \omterm{def:g3:shapes:shapes}{radio} se duplica, ¿qué pasa con su \omterm{def:g6:measure:volume}{volumen} y a su
        superficie? Indique el \emph{ley del cuadrado-cubo}: como una forma
        \omterm{def:g7:prop:scale}{balanza} arriba, \omterm{def:g6:measure:volume}{volúmenes} superar las superficies.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- The square--cube law rules the
world.}
\begin{enumerate}[resume]
  \item El argumento de Galileo contra los gigantes: imagina un humano
        ampliado por un factor $10$, mismas proporciones. por que
        factor aumenta el peso (el peso sigue \omterm{def:g6:measure:volume}{volumen})? Por
        ¿Qué factor tiene el \omterm{prop:g9:solids:sections}{sección transversal} de los huesos crecen
        (un \omterm{def:g6:measure:area}{área})? Entonces, ¿por qué factor \emph{presión}
        en cada centímetro cuadrado de hueso crece --- y qué
        ¿Le pasa al gigante?
  \item La misma ley a la inversa explica la heroicidad de las hormigas: fuerza
        sigue el músculo \omterm{prop:g9:solids:sections}{sección transversal} (un \omterm{def:g6:measure:area}{área}), el peso sigue
        \omterm{def:g6:measure:volume}{volumen}. para un animal $100$ veces menor en cada
        dirección, ¿por qué factores influyen el peso y la fuerza?
        se encoge, y ¿por qué factor se reduce la relación resistencia-peso?
        ¿Mejora la relación?
  \item El calor es producido por el cuerpo. \omterm{def:g6:measure:volume}{volumen} y perdido a través
        su superficie. Demuestre que para una esfera la
        la relación superficie-volumen es $\frac{3}{r}$ y calcularlo
        para $r = 1$ y $ r = 2$. ¿Qué enfría más rápido, un ratón o
        un oso --- ¿y por qué los bebés necesitan gorros en invierno?
  \item Dos naranjas esféricas de radios. $4$ centímetros y $5$ cm son
        vendido al mismo precio. Calcule la razón de sus
        \omterm{def:g6:measure:volume}{volúmenes}: ¿cuánta naranja más da la grande?
        el mismo dinero?
  \item Final: describe con precisión lo que representa la figura de la lápida.
        codifica --- los dos ``dos tercios'' de las preguntas 2 y 3
        --- y responde a Cicerón en una frase: ¿por qué un
        matemático elige, sobre cada conquista de su
        genio de la ingeniería, una esfera en un \omterm{def:g7:areas:prism}{cilindro}?
\end{enumerate}
\end{problem}
