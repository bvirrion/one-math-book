\chapter{El teorema de Tales}\label{ch:g9:thales}

¿Cómo se mide la altura de una \omterm{def:g8:solids:def}{pirámide} sin subirse a ella? Tales
de Mileto comparó su sombra con la sombra de un palo. Su teorema
--- las \omterm{def:g6:lines:perp}{rectas paralelas} cortan \omterm{def:g6:lines:objects}{segmentos} en trozos \omterm{def:g9:linfunc:linear}{proporcionales} --- es el
corazón matemático de las maquetas a escala, los mapas y las ampliaciones, y uno de
los teoremas con nombre más antiguos.

\section{El teorema}

\begin{theorem}[Tales]\label{thm:g9:thales:direct}
Sean dos rectas que se cortan en un punto $A$. Toma $M$ sobre la primera recta y $N$
sobre la segunda, con $B$ más lejos sobre la primera y $C$ más lejos sobre
la segunda. Si las rectas $(MN)$ y $(BC)$ son \emph{\omterm{def:g6:lines:perp}{paralelas}}, entonces los
triángulos $AMN$ y $ABC$ tienen los lados \omterm{def:g9:linfunc:linear}{proporcionales}:
\[
\frac{AM}{AB} = \frac{AN}{AC} = \frac{MN}{BC}.
\]
\end{theorem}
\admitted

\begin{method}[Calcular una longitud con Tales]\label{met:g9:thales:compute}
\begin{enumerate}
  \item identifica el punto $A$ donde se cortan las dos rectas y las dos
        \omterm{def:g6:lines:perp}{rectas paralelas}; nombra los dos triángulos;
  \item escribe las tres razones iguales, siempre «triángulo pequeño entre
        triángulo grande»;
  \item quédate con la igualdad de las dos razones en las que intervienen las tres
        longitudes conocidas y la desconocida;
  \item resuelve la proporción resultante multiplicando en cruz.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:thales:compute}
En la configuración de la izquierda de arriba, supón $AM = 3$, $AB = 5$, $AN = 2.4$
y $MN = 3.3$, y calculemos $AC$ y $BC$. Tales da
\[
\frac{AM}{AB} = \frac{AN}{AC} = \frac{MN}{BC},
\qquad\text{es decir}\qquad
\frac35 = \frac{2.4}{AC} = \frac{3.3}{BC}.
\]
De $\frac35 = \frac{2.4}{AC}$, multiplicando en cruz:
$3 \times AC = 5 \times 2.4 = 12$, luego $AC = 4$. De
$\frac35 = \frac{3.3}{BC}$: $3 \times BC = 16.5$, luego $BC = 5.5$.
\end{example}

\begin{example}[Tales en la vida real]\label{ex:g9:thales:stick}
Un palo vertical de $1$ m de altura proyecta una sombra de $1.5$ m, mientras que un árbol
proyecta una sombra de $12$ m en el mismo momento. Los rayos del Sol son paralelos, así que
los triángulos palo-sombra y árbol-sombra están en configuración de
Tales:
\[
\frac{\text{altura del árbol}}{1} = \frac{12}{1.5},
\qquad\text{así que el árbol mide } 8 \text{ m.}
\]
\end{example}

\section{El recíproco}

\begin{theorem}[Recíproco de Tales]\label{thm:g9:thales:converse}
Con los puntos colocados como en el \cref{thm:g9:thales:direct} ($M$ sobre $(AB)$,
$N$ sobre $(AC)$, en el mismo orden en las dos rectas):
si
\[
\frac{AM}{AB} = \frac{AN}{AC},
\]
entonces las rectas $(MN)$ y $(BC)$ son \omterm{def:g6:lines:perp}{paralelas}.
\end{theorem}
\admitted

\begin{method}[Demostrar o refutar el paralelismo]\label{met:g9:thales:converse}
\begin{enumerate}
  \item calcula por separado las dos razones $\dfrac{AM}{AB}$ y
        $\dfrac{AN}{AC}$ (como fracciones, no como \omterm{def:g6:decimals:rounding}{redondeos});
  \item si son iguales \emph{y} los puntos están en el mismo orden en
        las dos rectas, las rectas son \omterm{def:g6:lines:perp}{paralelas} (recíproco de Tales);
  \item si difieren, las rectas \emph{no} son \omterm{def:g6:lines:perp}{paralelas} --- porque, si
        lo fueran, el teorema de Tales obligaría a que las razones fueran iguales.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:thales:converse}
En la recta 1, $AM = 4$ y $AB = 10$; en la recta 2, $AN = 6$ y $AC = 15$.
Entonces
\[
\frac{AM}{AB} = \frac{4}{10} = \frac25,
\qquad
\frac{AN}{AC} = \frac{6}{15} = \frac25 :
\]
razones iguales, mismo orden: $(MN) \parallel (BC)$.

Con $AC = 14$ en su lugar: $\frac{6}{14} = \frac37 \neq \frac25$, así que las
rectas no son \omterm{def:g6:lines:perp}{paralelas}.
\end{example}

\section{Ampliación y reducción}

\begin{definition}[Escalar una figura]\label{def:g9:thales:scaling}
Ampliar o reducir una figura con el \emph{factor de
escala}\index{factor de escala} $k > 0$ es multiplicar \emph{todas} sus
longitudes por $k$: una \emph{ampliación} cuando $k > 1$ y una \emph{reducción}
cuando $k < 1$. En la configuración de Tales, el triángulo $AMN$ es la
reducción de $ABC$ con el factor $k = \frac{AM}{AB}$.
\end{definition}

\begin{proposition}[Efecto sobre los ángulos y las áreas]\label{prop:g9:thales:scaling}
Escalar con el factor $k$ conserva los ángulos y la forma de las figuras,
multiplica todas las longitudes por $k$ y multiplica todas las \emph{\omterm{def:g6:measure:area}{áreas}} por
$k^2$.
\end{proposition}
\admitted

\begin{omfigure}
\begin{tikzpicture}[scale=0.72]
  \draw[thick, omDef, fill=omDef!8] (0,0) rectangle (2,1.2);
  \draw[thick, omThm, fill=omThm!8] (3.4,0) rectangle (7.4,2.4);
  \node[font=\small, omDef] at (1,-0.45) {$2 \times 1.2$};
  \node[font=\small, omThm] at (5.4,-0.45) {$4 \times 2.4$};
  \node[font=\small] at (1,0.6) {área $2.4$};
  \node[font=\small] at (5.4,1.2) {área $9.6$};
\end{tikzpicture}

{\small Duplicar las longitudes ($k = 2$) multiplica el \omterm{def:g6:measure:area}{área} por
$k^2 = 4$: cuatro copias del rectángulo pequeño recubren el grande.}
\end{omfigure}

\begin{example}
Una foto de $10 \times 15$ cm se amplía con \omterm{def:g9:thales:scaling}{factor de escala} $k = 3$: la
copia mide $30 \times 45$ cm. Su \omterm{def:g6:measure:area}{área} pasa de $150$ cm$^2$ a
$150 \times 9 = 1350$ cm$^2$ --- $9$ veces mayor, no $3$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g9:thales:1}
Las rectas $(MN)$ y $(BC)$ son \omterm{def:g6:lines:perp}{paralelas}, con $M$ sobre $[AB]$ y $N$ sobre
$[AC]$; $AM = 2$, $AB = 6$, $AN = 3$, $BC = 9$. Calcula $AC$ y $MN$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:2}
La misma configuración: $AM = 5$, $MB = 3$ (¡cuidado: $MB$, no $AB$!),
$AN = 4$. Calcula $AC$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:3}
En una configuración de mariposa, $A$ está entre $M$ y $B$ y entre $N$
y $C$, con $(MN) \parallel (BC)$; $AM = 3$, $AB = 7.5$, $AN = 2$,
$MN = 2.6$. Calcula $AC$ y $BC$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:4}
$M$ está sobre $[AB]$ con $AM = 6$, $AB = 8$; $N$ está sobre $[AC]$ con $AN = 9$,
$AC = 12$. ¿Son \omterm{def:g6:lines:perp}{paralelas} las rectas $(MN)$ y $(BC)$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:5}
Lo mismo que antes con $AM = 4$, $AB = 6$, $AN = 5$, $AC = 8$. ¿Son $(MN)$
y $(BC)$ \omterm{def:g6:lines:perp}{paralelas}? Justifícalo con cuidado.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:6}
Una persona de $1.8$ m de altura está a $2$ m de la base de una farola;
su sombra mide $3$ m. Dibuja la configuración de Tales que forman la
farola, la persona y la punta de la sombra, y calcula la altura de
la farola.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:7}
Un mapa tiene escala $1 : 25\,000$ (las longitudes del mapa son las longitudes reales
multiplicadas por $k = \frac{1}{25000}$).
\begin{enumerate}
  \item dos pueblos distan $6.8$ cm en el mapa. ¿Cuál es la distancia
        real, en km?;
  \item un bosque tiene un \omterm{def:g6:measure:area}{área} de $8$ cm$^2$ en el mapa. ¿Cuál es su
        \omterm{def:g6:measure:area}{área} real, en km$^2$?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:8}
El triángulo $ABC$ tiene $AB = 12$, $AC = 15$, $BC = 18$. El punto $M$ de
$[AB]$ cumple $AM = 8$, y la recta que pasa por $M$ y es paralela a $(BC)$
corta $[AC]$ en $N$.
\begin{enumerate}
  \item calcula $AN$ y $MN$;
  \item ¿cuál es el \omterm{def:g9:thales:scaling}{factor de escala} de $ABC$ a $AMN$? Compara los
        \omterm{def:g6:measure:perimeter}{perímetros} de los dos triángulos.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:thales:9}
Sea $ABCD$ un trapecio con $(AB) \parallel (CD)$, $AB = 4$ y
$CD = 6$, cuyas diagonales $[AC]$ y $[BD]$ se cortan en $O$.
Usando Tales en la configuración de mariposa alrededor de $O$, calcula la
razón $\dfrac{OA}{OC}$ y demuestra que $O$ corta las dos diagonales en la
misma razón.
\end{exercise}

\section{Problema: La regla sin marcas, la cámara estenopeica y las
tres medias de un trapecio}

\begin{problem}[{Problema de fin de semana --- Tales en acción: dividir cualquier
segmento en partes iguales, medir el Sol con una caja de zapatos y un
trapecio donde se dan cita las tres medias famosas}]
\label{pb:g9:thales:1}
El teorema de Tales es la matemática de los \emph{rayos}: rayos de sol,
rayos de luz por un agujerito, rayos de lápiz desde un \omterm{def:g5:solids:def}{vértice}. Este
problema lo usa de tres maneras --- como herramienta de construcción (dividir un
\omterm{def:g6:lines:objects}{segmento} de longitud \emph{cualquiera}, incluso $\sqrt2$, en partes perfectamente
iguales), como instrumento de medida (el \omterm{def:g4:geometry:circle}{diámetro} del Sol, con
una caja de zapatos y una moneda) y como lupa sobre el trapecio
del \cref{exo:g9:thales:9}, donde las medias aritmética, geométrica y
armónica de esta serie aparecen todas en una sola figura.

\medskip
\noindent\textbf{Parte I --- Dividir con una regla sin marcas.}
\begin{enumerate}
  \item dibuja un \omterm{def:g6:lines:objects}{segmento} $[AB]$ de $7$ cm. Para cortarlo en tres
        partes iguales con compás y regla sin marcas: traza un rayo cualquiera
        desde $A$ (que no pase por $B$); lleva sobre él tres longitudes
        iguales de compás, lo que da los puntos $P_1$, $P_2$, $P_3$; une
        $P_3$ con $B$; traza las \omterm{def:g6:lines:perp}{paralelas} a $(P_3 B)$ por
        $P_1$ y $P_2$. Haz la construcción y marca dónde
        cortan las \omterm{def:g6:lines:perp}{paralelas} a $[AB]$;
  \item justifica con el \cref{thm:g9:thales:direct} que los dos
        puntos marcados cortan $[AB]$ exactamente en sus puntos de trisección;
  \item adapta el método para dividir un \omterm{def:g6:lines:objects}{segmento} en $5$ partes
        iguales y después para construir $\frac35$ de un \omterm{def:g6:lines:objects}{segmento} dado;
  \item construye el punto $M$ de $[AB]$ con
        $\frac{AM}{MB} = \frac23$ (es decir, $M$ a $\frac25$ del
        camino desde $A$). ¿Cuántos pasos iguales sobre el rayo y
        qué punto se une con $B$?;
  \item ¿por qué es mejor esta construcción que medir con una
        regla graduada? Piensa en un \omterm{def:g6:lines:objects}{segmento} de longitud $\sqrt2$
        (la diagonal de un cuadrado unidad, irracional por el
        \cref{pb:g9:sqrt:1}): ¿qué daría medir y
        qué da Tales?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La cámara estenopeica.}
Pincha con un alfiler una \omterm{def:g5:solids:def}{cara} de una caja cerrada: en la \omterm{def:g5:solids:def}{cara}
opuesta aparece una imagen del mundo del revés. Un punto de un
objeto, el agujerito y el punto imagen están alineados --- la luz
viaja en línea recta --- así que el objeto (altura $H$, a distancia $D$
delante del agujero) y su imagen (altura $h$, sobre la pared
del fondo a profundidad $d$ detrás del agujero) están en la configuración
de mariposa de Tales.
\begin{enumerate}[resume]
  \item deduce la fórmula del estenopo
        $h = H \times \frac dD$ a partir de Tales en la mariposa
        alrededor del agujero;
  \item un árbol de $12$ m está a $30$ m de una caja de zapatos de profundidad
        $20$ cm. ¿Cuánto mide su imagen y en qué posición?;
  \item apunta la caja al Sol: a $d = 1$ m detrás del
        agujerito, la imagen del Sol es un disco de \omterm{def:g4:geometry:circle}{diámetro} de unos
        $9$ mm. Dada la distancia del Sol
        $D = 1.5 \times 10^{11}$ m, calcula su \omterm{def:g4:geometry:circle}{diámetro} en
        \omterm{def:g9:fractions:scientific}{notación científica} (la notación del
        \cref{pb:g9:fractions:1} en acción). Compáralo con el valor verdadero,
        $1.39 \times 10^9$ m;
  \item la Luna: \omterm{def:g4:geometry:circle}{diámetro} $3.5 \times 10^6$ m, distancia
        $3.8 \times 10^8$ m. Calcula la razón
        $\frac{\text{diámetro}}{\text{distancia}}$ para la Luna
        y para el Sol. ¿Qué afortunada coincidencia revelan los dos
        números --- y qué espectacular fenómeno la
        hace posible?;
  \item en una o dos frases: ¿por qué la imagen del estenopo está
        del revés y por qué agrandar el agujerito hace la
        imagen más luminosa pero más borrosa?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El trapecio de las tres
medias.}
$ABCD$ es el trapecio del \cref{exo:g9:thales:9}:
$(AB) \parallel (CD)$, $AB = 4$, $CD = 6$, con las diagonales cortándose en
$O$ y $\frac{OA}{OC} = \frac{OB}{OD} = \frac{4}{6} =
\frac23$.
\begin{enumerate}[resume]
  \item práctica de mariposa: dos rectas se cortan en $O$ con
        $(MN) \parallel (PQ)$ en posición de mariposa; $OM = 3$,
        $OP = 7.5$, $MN = 4$ y $OQ = 6$. Calcula $PQ$ y
        $ON$;
  \item explica cómo Tales, aplicado con razón $\frac12$,
        contiene el teorema de los puntos medios del
        \cref{thm:g8:midpoints:direct} como caso particular;
  \item traza la paralela a las bases por $O$; corta
        $[AD]$ en $E$. Trabajando en el triángulo $ACD$ (observa que
        $\frac{AO}{AC} = \frac25$), calcula $EO$;
  \item por el cálculo \omterm{def:g7:central:def}{simétrico} en el triángulo $BDC$, el
        trozo $OF$ (con $F$ sobre $[BC]$) tiene la misma longitud.
        Concluye que
        \[
        EF = 2 \times \frac{4 \times 6}{4 + 6} = 4.8 :
        \]
        la paralela por el cruce de las diagonales mide la
        \emph{\omterm{pb:g8:speed:1}{media armónica}} de las bases --- la media
        de la ida y vuelta del \cref{pb:g8:speed:1}, reapareciendo en geometría
        pura;
  \item gran final: calcula las tres medias clásicas de
        las bases $4$ y $6$ --- la aritmética
        $\frac{4+6}{2}$, la geométrica $\sqrt{4 \times 6}$ y la armónica
        $\frac{2 \times 4 \times 6}{4 + 6}$ --- y ordénalas.
        Cada una vive en el trapecio: la media aritmética es la
        paralela media que une los puntos medios de los lados oblicuos
        (\cref{pb:g7:areas:1}), la \omterm{pb:g8:speed:1}{media armónica} es tu
        \omterm{def:g6:lines:objects}{segmento} $EF$ y la \omterm{pb:g8:cosine:1}{media geométrica} es la paralela
        que corta el trapecio en dos trapecios \emph{semejantes}
        (comprueba su valor con una calculadora; la
        demostración es un bonito reto extra). ¿Dónde se ha demostrado en este libro
        la desigualdad entre las tres
        (\cref{pb:g8:cosine:1}, \cref{pb:g8:speed:1})?
\end{enumerate}
\end{problem}
