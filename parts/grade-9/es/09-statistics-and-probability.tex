\chapter{Estadística y probabilidad}\label{ch:g9:statproba}

Los datos y el azar están por todas partes: resultados deportivos, predicciones del tiempo,
juegos. Este capítulo enseña a resumir una serie de números con su
\omterm{def:g9:statproba:indicators}{media}, su \omterm{def:g9:statproba:indicators}{mediana} y su \omterm{def:g9:statproba:indicators}{recorrido}, y a calcular la \omterm{def:g9:statproba:probability}{probabilidad} de experimentos
aleatorios sencillos --- incluidos los experimentos en dos pasos, que se tratan con diagramas
de árbol. Los dos temas se desarrollan mucho más en el \omterm{def:g6:measure:volume}{volumen} de
secundaria superior de esta serie.

\section{Resumir datos}

\begin{definition}[Media, mediana, recorrido]\label{def:g9:statproba:indicators}
Para una serie de $N$ números:
\begin{itemize}
  \item la \emph{media}\index{media} es la \omterm{def:g1:addition:def}{suma} de todos los valores dividida
        entre $N$;
  \item la \emph{mediana}\index{mediana} es un valor que parte la serie ordenada
        en dos \omterm{def:g3:division:half}{mitades} del mismo tamaño: para $N$ \omterm{def:g3:numbers:evenodd}{impar}, el valor
        central; para $N$ \omterm{def:g3:numbers:evenodd}{par}, el punto medio de los dos valores centrales;
  \item el \emph{recorrido}\index{recorrido} es el valor mayor menos el
        menor.
\end{itemize}
\end{definition}

\begin{example}\label{ex:g9:statproba:indicators}
Notas de un alumno: $8,\ 12,\ 9,\ 15,\ 11$.
\begin{enumerate}
  \item \omterm{def:g9:statproba:indicators}{media}: $\dfrac{8 + 12 + 9 + 15 + 11}{5} = \dfrac{55}{5} = 11$;
  \item \omterm{def:g9:statproba:indicators}{mediana}: ordena la serie: $8, 9, 11, 12, 15$; el valor central
        (el 3.º) es $11$;
  \item \omterm{def:g9:statproba:indicators}{recorrido}: $15 - 8 = 7$.
\end{enumerate}
Con una sexta nota de $17$: ordenadas $8, 9, 11, 12, 15, 17$, la \omterm{def:g9:statproba:indicators}{mediana}
pasa a ser el punto medio de $11$ y $12$, es decir $11.5$, y la \omterm{def:g9:statproba:indicators}{media}
pasa a ser $\frac{72}{6} = 12$.
\end{example}

\begin{example}[Media ponderada]\label{ex:g9:statproba:weighted}
Las tallas de zapato vendidas en un día:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccc}
talla & $38$ & $39$ & $40$ & $41$ & $42$ \\
\hline
cantidad & $3$ & $7$ & $6$ & $3$ & $1$ \\
\end{tabular}
\end{center}
La talla \omterm{def:g9:statproba:indicators}{media} es
\[
\frac{3 \times 38 + 7 \times 39 + 6 \times 40 + 3 \times 41 + 1 \times 42}
{3 + 7 + 6 + 3 + 1}
= \frac{114 + 273 + 240 + 123 + 42}{20}
= \frac{792}{20} = 39.6 .
\]
\end{example}

\begin{remark}
La \omterm{def:g9:statproba:indicators}{media} y la \omterm{def:g9:statproba:indicators}{mediana} pueden diferir mucho. En la serie
$10, 10, 10, 10, 60$ la \omterm{def:g9:statproba:indicators}{media} es $20$ (arrastrada hacia arriba por el valor $60$), mientras que
la \omterm{def:g9:statproba:indicators}{mediana} es $10$. Pregúntate siempre qué indicador representa mejor
la situación.
\end{remark}

\section{La probabilidad}

\begin{definition}[Probabilidad]\label{def:g9:statproba:probability}
Un experimento aleatorio tiene varios \emph{resultados} posibles; un
\emph{suceso} es un conjunto de resultados. Cada resultado recibe una
\emph{probabilidad}\index{probabilidad}: un número entre $0$ y $1$,
de manera que todas las probabilidades de los resultados sumen $1$; la
probabilidad $\P(A)$ de un suceso $A$ es la \omterm{def:g1:addition:def}{suma} de las probabilidades de
sus resultados. Cuando los $n$ resultados son todos igual de probables,
\[
\P(A) = \frac{\text{número de casos favorables}}{n}.
\]
\end{definition}

\begin{example}\label{ex:g9:statproba:spinner}
Una ruleta está dividida en $8$ sectores iguales: $3$ rojos, $3$ azules, $2$ verdes.
Cada sector tiene \omterm{def:g9:statproba:probability}{probabilidad} $\frac18$, así que
\[
\P(\text{rojo}) = \frac38, \qquad
\P(\text{verde}) = \frac28 = \frac14, \qquad
\P(\text{no verde}) = 1 - \frac14 = \frac34 .
\]
\end{example}

\begin{proposition}[Suceso contrario]\label{prop:g9:statproba:complement}
Para todo suceso $A$, el \emph{suceso contrario} $\bar A$ («$A$ no
ocurre») cumple
\[
\P(\bar A) = 1 - \P(A).
\]
\end{proposition}

\begin{proof}
Todo resultado está exactamente en uno de los dos, $A$ o $\bar A$, así que
$\P(A) + \P(\bar A)$ \omterm{def:g1:addition:def}{suma} las \omterm{def:g9:statproba:probability}{probabilidades} de todos los resultados, lo que da
$1$.
\end{proof}

\section{Experimentos en dos pasos}

\begin{method}[Diagramas de árbol]\label{met:g9:statproba:tree}
Para un experimento en dos pasos:
\begin{enumerate}
  \item dibuja una rama por cada resultado del primer paso y continúa después
        cada rama con los resultados del segundo paso, escribiendo cada
        \omterm{def:g9:statproba:probability}{probabilidad} sobre su rama;
  \item multiplica las \omterm{def:g9:statproba:probability}{probabilidades} a lo largo de un camino para obtener la \omterm{def:g9:statproba:probability}{probabilidad}
        de ese camino;
  \item \omterm{def:g1:addition:def}{suma} las \omterm{def:g9:statproba:probability}{probabilidades} de todos los caminos que forman un suceso.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:statproba:tree}
Una bolsa contiene $2$ fichas rojas y $3$ negras. Saca una ficha, anota su
color, \emph{devuélvela} y saca otra vez. Cada extracción da roja con
\omterm{def:g9:statproba:probability}{probabilidad} $\frac25$ y negra con \omterm{def:g9:statproba:probability}{probabilidad} $\frac35$.

\begin{omfigure}
\begin{tikzpicture}[scale=0.95]
  \coordinate (root) at (0,0);
  \coordinate (R) at (2.2,1.0);
  \coordinate (B) at (2.2,-1.0);
  \coordinate (RR) at (4.7,1.55);
  \coordinate (RB) at (4.7,0.45);
  \coordinate (BR) at (4.7,-0.45);
  \coordinate (BB) at (4.7,-1.55);
  \draw (root) -- (R) node[midway, above, font=\small, sloped] {$\frac25$};
  \draw (root) -- (B) node[midway, below, font=\small, sloped] {$\frac35$};
  \draw (R) -- (RR) node[midway, above, font=\small, sloped] {$\frac25$};
  \draw (R) -- (RB) node[midway, below, font=\small, sloped] {$\frac35$};
  \draw (B) -- (BR) node[midway, above, font=\small, sloped] {$\frac25$};
  \draw (B) -- (BB) node[midway, below, font=\small, sloped] {$\frac35$};
  \node[omThm, font=\small, above left=-2pt] at (R) {R};
  \node[font=\small, below left=-2pt] at (B) {N};
  \node[right, font=\small] at (RR)
    {RR:\ $\frac25 \times \frac25 = \frac{4}{25}$};
  \node[right, font=\small] at (RB)
    {RN:\ $\frac25 \times \frac35 = \frac{6}{25}$};
  \node[right, font=\small] at (BR)
    {NR:\ $\frac35 \times \frac25 = \frac{6}{25}$};
  \node[right, font=\small] at (BB)
    {NN:\ $\frac35 \times \frac35 = \frac{9}{25}$};
\end{tikzpicture}

{\small El árbol de las dos extracciones con devolución: multiplica a lo largo de las
ramas y comprueba que los cuatro caminos suman $1$.}
\end{omfigure}

\omterm{def:g9:statproba:probability}{Probabilidad} de dos fichas del mismo color:
$\frac{4}{25} + \frac{9}{25} = \frac{13}{25}$. \omterm{def:g9:statproba:probability}{Probabilidad} de al menos
una roja: $1 - \P(\text{NN}) = 1 - \frac{9}{25} = \frac{16}{25}$.
\end{example}

\begin{remark}[Las frecuencias se acercan a las probabilidades]
Lanzar un dado equilibrado $6000$ veces da \emph{alrededor de} $1000$ seises --- no
exactamente. A medida que crece el número de repeticiones, las frecuencias observadas se acercan
cada vez más a las \omterm{def:g9:statproba:probability}{probabilidades}: eso es lo que hace de la \omterm{def:g9:statproba:probability}{probabilidad}
la herramienta adecuada para modelizar experimentos repetidos (una historia que se desarrolla en
el \omterm{def:g6:measure:volume}{volumen} de secundaria superior de esta serie).
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g9:statproba:1}
Calcula la \omterm{def:g9:statproba:indicators}{media}, la \omterm{def:g9:statproba:indicators}{mediana} y el \omterm{def:g9:statproba:indicators}{recorrido} de la serie
$14,\ 9,\ 17,\ 9,\ 12,\ 11,\ 12$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:statproba:2}
Las temperaturas al mediodía a lo largo de una semana fueron $19,\ 21,\ 24,\ 18,\ 22,\ 25,\
19$ (grados). Calcula la \omterm{def:g9:statproba:indicators}{media} (a la décima) y la \omterm{def:g9:statproba:indicators}{mediana}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:statproba:3}
El número de goles marcados por un equipo en $10$ partidos:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|cccc}
goles & $0$ & $1$ & $2$ & $3$ \\
\hline
partidos & $2$ & $4$ & $3$ & $1$ \\
\end{tabular}
\end{center}
Calcula el número medio de goles por partido y la \omterm{def:g9:statproba:indicators}{mediana}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:statproba:4}
Se lanza una vez un dado equilibrado. Calcula la \omterm{def:g9:statproba:probability}{probabilidad} de: «sacar un
$3$»; «sacar un \omterm{def:g3:numbers:evenodd}{número par}»; «sacar al menos un $3$»; «no
sacar un $6$».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:statproba:5}
Una caja contiene $12$ bolas: $5$ blancas, $4$ rojas y $3$ verdes; se saca una bola
al azar. Calcula $\P(\text{blanca})$, $\P(\text{roja o verde})$
y $\P(\text{no blanca})$. ¿Qué observas en las dos últimas?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:statproba:6}
Se lanza una moneda y después un dado equilibrado. Dibuja el árbol (o cuenta los
resultados) y calcula la \omterm{def:g9:statproba:probability}{probabilidad} de sacar cara \emph{y} un
$6$; y después la de sacar cara \emph{o} un $6$ (o las dos cosas).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:statproba:7}
En la bolsa del \cref{ex:g9:statproba:tree} ($2$ rojas, $3$ negras), las dos
extracciones se hacen ahora \emph{sin} devolución. Dibuja el árbol nuevo
(cuidado: las \omterm{def:g9:statproba:probability}{probabilidades} del segundo nivel cambian) y calcula la
\omterm{def:g9:statproba:probability}{probabilidad} de sacar dos fichas negras y después la de sacar dos fichas de colores
distintos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:statproba:8}
Tras $9$ partidos, una jugadora de baloncesto promedia $14$ puntos por partido.
\begin{enumerate}
  \item ¿cuántos puntos ha marcado en total?;
  \item ¿cuántos puntos tiene que marcar en el $10$.º partido para promediar
        $15$?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:statproba:9}
Un juego: lanza dos dados equilibrados; ganas si los dos resultados son iguales
(«un doble»).
\begin{enumerate}
  \item calcula la \omterm{def:g9:statproba:probability}{probabilidad} de ganar;
  \item juegas dos veces (partidas independientes). Con un árbol, calcula la
        \omterm{def:g9:statproba:probability}{probabilidad} de ganar al menos una vez.
\end{enumerate}
\end{exercise}

\section{Problema: Las apuestas ruinosas del Caballero}

\begin{problem}[{Problema de fin de semana --- la disputa de juego de 1654
que creó la teoría de la probabilidad, y la coincidencia de cumpleaños
que engaña a todas las clases}]
\label{pb:g9:statproba:1}
En 1654, un noble francés jugador empedernido, el Caballero de
Méré, se hizo rico con una apuesta de dados y empezó a perder con otra
que él creía equivalente. Desconcertado, preguntó al matemático
Blaise Pascal; Pascal escribió a Pierre de Fermat; y su
intercambio de cartas fundó la teoría de la \omterm{def:g9:statproba:probability}{probabilidad}. Este
problema repite todo el asunto con las herramientas de este capítulo
--- resultados igual de probables, árboles y la regla del suceso contrario
(\cref{prop:g9:statproba:complement}) --- y termina con la coincidencia más
contraintuitiva de todas.

\medskip
\noindent\textbf{Parte I --- Los dados, contados con honestidad.}
\begin{enumerate}
  \item lanza dos veces un dado equilibrado. Con un árbol (o una
        tabla $6 \times 6$), calcula la \omterm{def:g9:statproba:probability}{probabilidad} de que \emph{no} salga ningún seis en los
        dos lanzamientos, y deduce la \omterm{def:g9:statproba:probability}{probabilidad} de al menos un
        seis;
  \item un compañero argumenta: «una posibilidad entre seis por lanzamiento, dos veces,
        así que $\frac16 + \frac16 = \frac13$». Compara con la pregunta
        1 y señala los resultados exactos que su \omterm{def:g1:addition:def}{suma} cuenta
        dos veces;
  \item lanzando dos dados y sumando: calcula la \omterm{def:g9:statproba:probability}{probabilidad} de
        una \omterm{def:g1:addition:def}{suma} de $7$ y la de una \omterm{def:g1:addition:def}{suma} de $2$. ¿Por qué es el $7$ el
        favorito de los jugadores?;
  \item vieja disputa: con dos dados, ¿es más probable una \omterm{def:g1:addition:def}{suma} de $9$ o una \omterm{def:g1:addition:def}{suma} de
        $10$? Cuenta los resultados y zanja la cuestión;
  \item generaliza la pregunta 1: explica por qué la \omterm{def:g9:statproba:probability}{probabilidad} de al
        menos un seis en $n$ lanzamientos es
        \[
        1 - \left(\frac56\right)^{\!n} ,
        \]
        porque las ramas «sin seis» del árbol se multiplican de nivel
        en nivel.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Las dos apuestas del Caballero.}
\begin{enumerate}[resume]
  \item primera apuesta, a partes iguales: \emph{al menos un seis en cuatro
        lanzamientos de un dado}. Calcula
        $\left(\frac56\right)^4$ como \omterm{def:g9:fractions:fraction}{fracción} y como decimal,
        y la \omterm{def:g9:statproba:probability}{probabilidad} de ganar del Caballero. ¿Era buena la
        apuesta?;
  \item el propio razonamiento del Caballero era: «cuatro lanzamientos a
        $\frac16$ cada uno: $\frac46$ de posibilidad». Aquí daba casi
        la respuesta correcta --- pero llévalo hasta $7$ lanzamientos: ¿qué
        absurdo produce y qué error de la pregunta 2
        repite?;
  \item segunda apuesta, que él creía equivalente por \omterm{def:g9:linfunc:linear}{proporcionalidad}
        («$24$ lanzamientos a $\frac1{36}$ cada uno: otra vez
        $\frac{24}{36} = \frac46$»): \emph{al menos un
        seis doble en $24$ lanzamientos de dos dados}. Calcula la
        \omterm{def:g9:statproba:probability}{probabilidad} verdadera de ganar $1 - \left(\frac{35}{36}\right)^{24}$
        (con calculadora). ¿Por qué se arruinó poco a poco el Caballero?;
  \item ¿cuántos lanzamientos de dos dados le habrían devuelto la
        ventaja? Prueba $n = 25$ con la calculadora y
        concluye;
  \item enuncia la moraleja en dos frases: ¿qué falla, en
        general, al multiplicar una \omterm{def:g9:statproba:probability}{probabilidad} por el número de
        intentos --- y qué herramienta correcta sustituye a esa
        tentación?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- \omterm{def:g9:statproba:indicators}{Medias}, \omterm{def:g9:statproba:indicators}{medianas} y cumpleaños.}
\begin{enumerate}[resume]
  \item una empresa pequeña paga $2\,000$ euros al mes a nueve empleados
        y $20\,000$ al director. Calcula el salario medio y el
        mediano (\cref{def:g9:statproba:indicators}).
        ¿Qué número debería dar honestamente el anuncio de
        empleo?;
  \item construye una lista de cinco notas de examen con \omterm{def:g9:statproba:indicators}{media} $12$ y
        \omterm{def:g9:statproba:indicators}{mediana} $15$. ¿Qué dice esa pareja (\omterm{def:g9:statproba:indicators}{media} por debajo de la \omterm{def:g9:statproba:indicators}{mediana})
        sobre la forma de las notas?;
  \item el problema de los cumpleaños: en una clase de $23$ alumnos, ¿cuál
        es la \omterm{def:g9:statproba:probability}{probabilidad} de que al menos dos compartan cumpleaños?
        Plantea el suceso contrario (los $23$ cumpleaños distintos):
        \[
        \frac{365}{365} \times \frac{364}{365} \times
        \frac{363}{365} \times \dots \times \frac{343}{365} .
        \]
        Calcula el \omterm{def:g2:mult:def}{producto} de los tres primeros factores, describe
        cómo evolucionan los factores y --- sabiendo que el
        \omterm{def:g2:mult:def}{producto} completo vale unos $0.493$ --- responde a la pregunta. La mayoría de la
        gente espera «muy improbable»: ¿qué dicen las
        matemáticas?;
  \item la intuición reparada: ¿cuántas \emph{parejas} de
        alumnos pueden compartir cumpleaños en una clase de $23$
        (la cuenta de apretones de manos del \cref{pb:g6:wholes:1})? Explica en una
        frase por qué es ese número, y no el propio $23$, el que provoca
        la sorpresa;
  \item final: describe un experimento para comprobar cualquiera de los dos resultados
        --- la apuesta de de Méré con un dado real o el problema
        de los cumpleaños en varias clases --- y enuncia con cuidado
        qué esperas de las frecuencias observadas a medida que crece el número
        de repeticiones. (Esa expectativa tiene nombre: la
        ley de los grandes números; su demostración espera en el \omterm{def:g6:measure:volume}{volumen} de
        secundaria superior.)
\end{enumerate}
\end{problem}
