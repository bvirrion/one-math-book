\chapter{الدوال الخطية والتآلفية}\label{ch:g9:linfunc}

«ثلاثة كيلوغرامات تكلّف ثلاثة أمثال ما يكلّفه واحد»: وهذا هو
\omterm{def:g9:linfunc:linear}{التناسب}، أبسط علاقة تربط مقدارين. \omterm{def:g9:linfunc:linear}{والدوال الخطية} هي
\omterm{def:g9:linfunc:linear}{التناسب} بلغة الدوال؛ \omterm{def:g9:linfunc:affine}{والدوال التآلفية} تضيف مقدارًا
اِبتدائيًّا ثابتًا. وتمثيلاتها البيانية \omterm{def:g6:lines:objects}{مستقيمات}، وقراءة هذه
\omterm{def:g6:lines:objects}{المستقيمات} هي المهارة التي يبنيها هذا الفصل.

\section{التناسب والدوال الخطية}

\begin{definition}[الدالة الخطية]\label{def:g9:linfunc:linear}
\emph{الدالة الخطية}\index{دالة خطية} تضرب كل عدد في عدد ثابت
$a$:
\[
f(x) = a x .
\]
والعدد $a$ هو \emph{معامل} الدالة $f$. ويكون مقداران
\emph{متناسبين}\index{تناسب} بالضبط حين يكون أحدهما دالة خطية
للآخر.
\end{definition}

\begin{example}\label{ex:g9:linfunc:linear}
إذا كان ثمن التفاح $3$ للكيلوغرام، فثمن $x$ كيلوغرامًا هو
$f(x) = 3x$: فشراء الضِّعف يكلّف الضِّعف، وثمن $2.5$ كغ هو
$f(2.5) = 7.5$.

وبالعكس، إذا حقّقت \omterm{def:g9:linfunc:linear}{دالة خطية} $f(4) = 10$، كان
$a = \frac{10}{4} = 2.5$ و $f(x) = 2.5x$: \emph{فقيمة واحدة تعيّن
\omterm{def:g9:linfunc:linear}{الدالة الخطية} تعيينًا تامًّا.}
\end{example}

\begin{proposition}[التمثيل البياني لدالة خطية]\label{prop:g9:linfunc:lineargraph}
التمثيل البياني للدالة $f(x) = ax$ \omterm{def:g6:lines:objects}{مستقيم} \emph{يمرّ بالمبدأ}.
وبالعكس، كل \omterm{def:g6:lines:objects}{مستقيم} غير عمودي يمرّ بالمبدأ هو التمثيل البياني
\omterm{def:g9:linfunc:linear}{لدالة خطية}.
\end{proposition}
\admitted

\section{الدوال التآلفية}

\begin{definition}[الدالة التآلفية]\label{def:g9:linfunc:affine}
\emph{الدالة التآلفية}\index{دالة تآلفية} على الصورة
\[
f(x) = a x + b ,
\]
حيث $a$ و $b$ عددان ثابتان. \omterm{def:g9:linfunc:linear}{والدوال الخطية} هي الحالة الخاصّة
$b = 0$؛ والدوال الثابتة هي الحالة $a = 0$.
\end{definition}

\begin{example}
تأخذ قاعة رياضية رسم اِنخراط قدره $20$، ثم $5$ لكل زيارة: فتكلفة
$x$ زيارة كلّها هي $f(x) = 5x + 20$. والتكلفة \emph{ليست}
متناسبة مع عدد الزيارات (فعشر زيارات لا تكلّف ضِعف ما تكلّفه
خمس)، لكن كل زيارة إضافية تضيف المقدار نفسه، $5$.
\end{example}

\begin{proposition}[التمثيل البياني لدالة تآلفية]\label{prop:g9:linfunc:affinegraph}
التمثيل البياني للدالة $f(x) = ax + b$ \omterm{def:g6:lines:objects}{مستقيم}:
\begin{itemize}
  \item $b = f(0)$ هو \emph{الترتيب عند مبدأ محور $y$}: \omterm{def:g6:lines:objects}{فالمستقيم}
        يقطع المحور العمودي عند $(0, b)$؛
  \item و $a$ هو \emph{الميل}\index{ميل}: فالتقدّم وحدةً واحدة
        نحو اليمين يرفع $a$ وحدة (ويخفض إذا كان $a < 0$)؛
        والدالة متزايدة حين $a > 0$، ومتناقصة حين $a < 0$؛
  \item ومن أجل أيّ مدخلين $u \neq v$،
        $a = \dfrac{f(v) - f(u)}{v - u}$.
\end{itemize}
\end{proposition}

\begin{proof}[برهان النقطة الأخيرة]
$f(v) - f(u) = (av + b) - (au + b) = a(v - u)$؛ فاِقسم على $v - u$.
وما تبقّى مقبول في هذا المستوى (ويعالج مجلّد الثانوية \omterm{def:g6:lines:objects}{المستقيمات}
\omterm{def:g8:equations:def}{ومعادلاتها} معالجةً كاملة).
\end{proof}

\begin{method}[إيجاد دالة تآلفية من قيمتين]\label{met:g9:linfunc:findaffine}
بمعرفة $f(u)$ و $f(v)$ حيث $u \neq v$:
\begin{enumerate}
  \item اِحسب \omterm{prop:g9:linfunc:affinegraph}{الميل} $a = \dfrac{f(v) - f(u)}{v - u}$؛
  \item وعوّض بإحدى القيمتين المعلومتين في $f(x) = ax + b$
        لإيجاد $b$؛
  \item وتحقّق بالقيمة الأخرى.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:linfunc:findaffine}
أوجِد \omterm{def:g9:linfunc:affine}{الدالة التآلفية} التي $f(2) = 7$ و $f(5) = 16$.
\[
a = \frac{16 - 7}{5 - 2} = \frac93 = 3 ,
\]
ثم $f(2) = 3 \times 2 + b = 7$ يعطي $b = 1$: أي $f(x) = 3x + 1$.
والتحقّق: $f(5) = 15 + 1 = 16$.
\end{example}

\section{النسب المئوية}

\begin{proposition}[التغيّر بنسبة مئوية]\label{prop:g9:linfunc:percent}
زيادة مقدار بنسبة $t\,\%$ تضربه في $1 + \dfrac{t}{100}$؛ وإنقاصه
بنسبة $t\,\%$ يضربه في $1 - \dfrac{t}{100}$. والتغيّرات بالنسب
المئوية \omterm{def:g9:linfunc:linear}{دوال خطية}.
\end{proposition}

\begin{proof}
زيادة $x$ بنسبة $t\,\%$ تعني إضافة $\frac{t}{100}\,x$:
\[
x + \frac{t}{100}\,x = \left(1 + \frac{t}{100}\right) x .
\]
والحساب نفسه بإشارة الطرح من أجل الإنقاص.
\end{proof}

\begin{example}\label{ex:g9:linfunc:percent}
ثمن قدره $80$ يزيد بنسبة $15\%$: فالثمن الجديد
$80 \times 1.15 = 92$. وثمن قدره $60$ ينقص بنسبة $30\%$:
$60 \times 0.7 = 42$.

\emph{وتتابع التغيّرات يضرب العوامل بعضها في بعض.} فزيادة $20\%$
يتبعها نقصان $20\%$ تعطي
$1.2 \times 0.8 = 0.96$: أي \emph{نقصانًا} إجماليًّا قدره $4\%$، لا
عودةً إلى البداية!
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:1}
أيّ هذه \omterm{def:g9:linfunc:linear}{الدوال خطية}؟ وأيّها تآلفية؟ وأيّها ليست هذه ولا تلك؟
\[
f(x) = 4x, \qquad
g(x) = 3x - 5, \qquad
h(x) = x^2, \qquad
k(x) = -\tfrac x2, \qquad
m(x) = 7 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:2}
لتكن $f(x) = 2x - 3$. اِحسب $f(0)$ و $f(4)$ و $f(-1)$، وجِد العدد
$x$ الذي يحقّق $f(x) = 9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:3}
\omterm{def:g9:linfunc:linear}{دالة خطية} تحقّق $f(6) = 15$. أوجِد معاملها، ثم اِحسب $f(10)$
و $f(-2)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:4}
اُرسم في المعلم نفسه التمثيلات البيانية للدوال $f(x) = 2x$
و $g(x) = 2x + 3$ و $h(x) = -x + 4$. فماذا يمكن أن يقال عن تمثيلَي
$f$ و $g$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:5}
أوجِد \omterm{def:g9:linfunc:affine}{الدالة التآلفية} التي $f(1) = 5$ و $f(3) = 11$؛ ثم تلك التي
$g(0) = 4$ و $g(2) = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:6}
اِحسب: الثمن بعد زيادة $25\%$ على $120$؛ والثمن بعد تخفيض
$40\%$ على $65$؛ والثمن الأصلي إذا صار ثمن سلعة $69$ بعد تخفيض
$25\%$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:7}
اِشتراك هاتف A يكلّف $10$ في الشهر مضافًا إليها $0.05$ لكل دقيقة
مكالمة؛ والاِشتراك B يكلّف $25$ في الشهر بمكالمات غير محدودة.
\begin{enumerate}
  \item عبّر عن التكلفة الشهرية للاِشتراك A بدلالة عدد الدقائق
        $x$.
  \item واِبتداءً من كم دقيقة في الشهر يصير الاِشتراك B أرخص؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:8}
مستعمرة من $2000$ بكتيريا تزيد بنسبة $10\%$ كل ساعة.
\begin{enumerate}
  \item كم بكتيريا فيها بعد ساعة؟ وبعد ساعتين؟
  \item وفسّر لماذا ليس الجواب بعد ساعتين $2400$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:9}
التمثيل البياني \omterm{def:g9:linfunc:affine}{لدالة تآلفية} يمرّ بالنقطتين $(2, 3)$
و $(6, 1)$.
\begin{enumerate}
  \item اِحسب ميلها. فهل الدالة متزايدة أم متناقصة؟
  \item وأعطِ عبارتها، واِحسب المدخل الذي يكون المخرج عنده
        $0$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:linfunc:10}
يرفع متجر ثمنًا بنسبة $t\,\%$ أولًا، ثم يخفض الثمن الجديد بنسبة
$t\,\%$.
\begin{enumerate}
  \item بيّن أنّ الثمن النهائي يساوي الثمن الأصلي مضروبًا في
        $1 - \dfrac{t^2}{10000}$.
  \item واِستنتج أنّ الثمن النهائي \emph{أصغر} دائمًا من الأصلي
        (من أجل $0 < t \leq 100$)، وجِد التغيّر الإجمالي بالنسبة
        المئوية من أجل $t = 10$.
\end{enumerate}
\end{exercise}

\section{مسألة: الميزانان الحراريان}

\begin{problem}[{مسألة نهاية الأسبوع --- سيلسيوس وفهرنهايت والدوال
التآلفية في الحياة اليومية: درجة حرارة واحدة تُقرأ العدد نفسه على
السلّمين، وثبات الميل هو توقيع الاِستقامة}]
\label{pb:g9:linfunc:1}
يسمع الأمريكيون «$68$ درجة» فيلبسون خفيفًا؛ ويسمعها الأوروبيون
فيمسكون معاطفهم. وميزانا الحرارة في العالم تربطهما
\omterm{def:g9:linfunc:affine}{دالة تآلفية} --- وبناؤها من واقعتين هو بالضبط مهارة
\cref{met:g9:linfunc:findaffine}. وتبني هذه المسألة التحويل، وتجد
درجة الحرارة الغريبة التي يتّفق عندها السلّمان، ثم تقرأ سيارات
الأجرة والصراصير والشموع بالنظّارتين نفسيهما، وتنتهي بالاِختبار
الذي يتعرّف على الاِستقامة نفسها.

\medskip
\noindent\textbf{الجزء الأول --- بناء التحويل.}
واقعتان تثبّتان \omterm{def:g7:prop:scale}{سلّم} فهرنهايت: يتجمّد الماء عند $32\,^\circ$F
ويغلي عند $212\,^\circ$F (أي عند $0\,^\circ$C
و $100\,^\circ$C).
\begin{enumerate}
  \item أوجِد \omterm{def:g9:linfunc:affine}{الدالة التآلفية} $F = f(C)$ التي تحوّل سيلسيوس
        إلى فهرنهايت (\cref{met:g9:linfunc:findaffine}
        بالقيمتين $f(0) = 32$ و $f(100) = 212$).
  \item وحوّل إلى فهرنهايت: يومًا معتدلًا عند $20\,^\circ$C؛
        وحرارة الجسم $37\,^\circ$C؛ وصقيعًا عند $-10\,^\circ$C.
  \item وحُلّ من أجل $C$ لتبني التحويل العكسي، واِستعمله على
        $50\,^\circ$F وعلى $98.6\,^\circ$F.
  \item فهل $f$ دالة \emph{خطية} (\cref{def:g9:linfunc:linear})؟
        اِختبر العلامات الفارقة: ما $f(0)$، وهل مضاعفة قراءة
        سيلسيوس تضاعف قراءة فهرنهايت؟ اِستنتج بالكلمة الصحيحة
        (\cref{def:g9:linfunc:affine}).
  \item واللغز الكلاسيكي: هل توجد درجة حرارة تُقرأ \emph{العدد
        نفسه} على السلّمين؟ حُلّ $f(x) = x$، وصِف ماذا يعني
        حلّك على التمثيل البياني للدالة $f$ (أيّ \omterm{def:g6:lines:objects}{مستقيم} يقطعه
        التمثيل هناك؟).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- \omterm{def:g9:linfunc:affine}{الدوال التآلفية} في البرّية.}
\begin{enumerate}[resume]
  \item سيارة أجرة A تأخذ $3$ يورو مضافًا إليها $1.50$ لكل كلم؛
        وسيارة B تأخذ $2$ يورو لكل كلم من غير رسم ثابت. اُكتب
        دالّتَي الثمن وقارن بينهما من أجل رحلة $4$ كلم ورحلة
        $8$ كلم.
  \item وجِد مسافة التعادل التي تتساوى عندها تكلفة
        السيارتين، وصِف الوضع على تمثيل بياني (\omterm{def:g6:lines:objects}{مستقيمان} ---
        ماذا يحدث عند نقطة التقاطع، وقبلها، وبعدها؟).
  \item وقاعدة الطبيعيين التقريبية: يصرّ الصرصور نحو
        $N = 7T - 30$ مرّة في الدقيقة عند درجة الحرارة
        $T\,^\circ$C. فكم صريرًا في مساء عند $20\,^\circ$C؟
        وإذا عددت $82$ صريرًا في دقيقة: فأيّ درجة حرارة يعلنها
        الصرصور؟
  \item وهاتف ثمنه $600$ يورو تنقص قيمته $15$ يورو كل شهر:
        اُكتب دالة القيمة $v(t)$، وجِد متى تصير قيمة الهاتف
        $240$ يورو، ومتى يقول النموذج إنّها معدومة. فهل تبقى
        للصيغة دلالة بعد ذلك التاريخ؟
  \item والتغيّرات بالنسب المئوية \omterm{def:g9:linfunc:linear}{دوال خطية}: فالزيادة
        $+25\,\%$ هي الضرب في $1.25$، والنقصان $-20\,\%$ هو
        الضرب في $0.8$ (\cref{prop:g9:linfunc:percent}).
        ركّبهما: فماذا تفعل زيادة $+25\,\%$ يتبعها خفض
        $-20\,\%$ بثمنٍ ما؟ فسّر المعجزة الصغيرة، وقارن مع
        \cref{exo:g9:linfunc:10}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- \omterm{prop:g9:linfunc:affinegraph}{الميل} هو كل شيء.}
\begin{enumerate}[resume]
  \item شمعة طولها $20$ سم حين تُشعل و $17$ سم بعد نصف
        ساعة. بافتراض ذوبان تآلفي، أوجِد $h(t)$ (الاِرتفاع
        بالسنتيمتر، و $t$ بالدقائق)، وتنبّأ متى تموت الشمعة.
  \item ودرّاجان يسيران على الطريق نفسه: مسافة الأول
        $d_1(t) = 24t$ (بالكيلومترات بعد $t$ ساعة)، ومسافة
        الثاني $d_2(t) = 10 + 18t$. فمن اِنطلق بتقدّم، ومن يسير
        أسرع، وفي أيّ لحظة وعند أيّ كيلومتر يلحق الأول الثاني؟
  \item ولتكن $f(x) = 2x + 3$ و $g(x) = -0.5x + 7$. اِحسب
        $f(x) + g(x)$ وتحقّق أنّها تآلفية من جديد. فما
        ميل \omterm{def:g1:addition:def}{المجموع} وترتيبه عند المبدأ، في العموم؟
  \item وناقش \omterm{def:g8:equations:def}{المعادلة} $mx + p = 0$ مناقشة كاملة: كم حلًّا حين
        $m \neq 0$ (وما هو)؟ وحين $m = 0$ و $p \neq 0$؟ وحين
        $m = 0$ و $p = 0$؟ (ولكل حالة حكاية على التمثيل
        البياني: أين يقطع \omterm{def:g6:lines:objects}{مستقيم} ميله $m$ المحور الأفقي؟)
  \item واِختبار الاِستقامة: هل النقاط $(1, 5)$ و $(3, 9)$
        و $(7, 17)$ على \omterm{def:g6:lines:objects}{مستقيم} واحد؟ وهل النقاط $(1, 5)$
        و $(3, 9)$ و $(6, 16)$ كذلك؟ اِحسب معدّلات التغيّر بين
        الأزواج وقرّر. واُذكر العبرة: \emph{ثبات معدّل التغيّر}
        هو توقيع \omterm{def:g9:linfunc:affine}{الدوال التآلفية} --- أمّا التمثيلات المنحنية
        فتغيّر معدّلها، وقياس \emph{ذلك} حكاية لمجلّد الثانوية.
\end{enumerate}
\end{problem}
