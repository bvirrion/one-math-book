\chapter{الجبر والمعادلات}\label{ch:g9:algebra}

الحروف تنوب عن الأعداد، والحساب بالحروف يبرهن على وقائع تخصّ
\emph{كل} الأعداد دفعةً واحدة. ويتمرّن هذا الفصل على \omterm{def:g9:algebra:expand}{النشر}
والتفكيك --- بما فيه المتطابقات الثلاث التي سترافقك في الرياضيات
كلّها --- ويحلّ \omterm{def:g8:equations:def}{معادلات} الدرجة الأولى، \omterm{def:g8:equations:def}{ومعادلات}
\omterm{def:g2:mult:def}{الجداء}، والمتباينات البسيطة، خطوةً مبرَّرة في كل مرّة.

\section{النشر}

\begin{definition}[النشر]\label{def:g9:algebra:expand}
\emph{النشر}\index{نشر} هو تحويل \omterm{def:g2:mult:def}{جداء} إلى \omterm{def:g1:addition:def}{مجموع}،
باِستعمال التوزيعية:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g9:algebra:expand}
كل حدّ من القوس الأول يضرب كل حدّ من الثاني:
\begin{align*}
(x + 3)(2x + 5)
&= x \times 2x + x \times 5 + 3 \times 2x + 3 \times 5 \\
&= 2x^2 + 5x + 6x + 15 \\
&= 2x^2 + 11x + 15 .
\end{align*}
والإشارات ترافق الحدود:
\begin{align*}
(3x - 2)(x - 4)
&= 3x^2 - 12x - 2x + 8
= 3x^2 - 14x + 8 .
\end{align*}
\end{example}

\begin{theorem}[المتطابقات الثلاث]\label{thm:g9:algebra:identities}
من أجل كل عددين $a$ و $b$:
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
اُنشر الطرف الأول من كل متطابقة. ففي الأولى:
$(a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2$. وفي الثانية،
يكفي أن يصير $b$ هو $-b$ في الأولى. وفي الثالثة:
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$ (فالحدّان المتقاطعان
يتلاشيان).
\end{proof}

\begin{example}\label{ex:g9:algebra:identities}
\[
(x + 5)^2 = x^2 + 10x + 25, \qquad
(3x - 1)^2 = 9x^2 - 6x + 1, \qquad
(2x + 7)(2x - 7) = 4x^2 - 49 .
\]
وحساب ذهني: $102^2 = (100 + 2)^2 = 10000 + 400 + 4 = 10404$،
و $98 \times 102 = (100-2)(100+2) = 10000 - 4 = 9996$.
\end{example}

\section{التفكيك}

\begin{method}[التفكيك]\label{met:g9:algebra:factor}
لتفكيك عبارة، جرّب بالترتيب:
\begin{enumerate}
  \item \emph{العامل المشترك}: $ka + kb = k(a + b)$، حيث قد يكون
        $k$ نفسه قوسًا؛
  \item \emph{\omterm{ex:g1:subtraction:difference}{فرق} مربّعين}: $a^2 - b^2 = (a+b)(a-b)$؛
  \item \emph{المربّع التامّ}: $a^2 + 2ab + b^2 = (a+b)^2$ (ومع
        $-2ab$، يكون $(a-b)^2$).
\end{enumerate}
وتحقّق دائمًا \omterm{def:g9:algebra:expand}{بنشر} النتيجة.
\end{method}

\begin{example}\label{ex:g9:algebra:factor}
\emph{العامل المشترك:}
$15x^2 + 10x = 5x \times 3x + 5x \times 2 = 5x(3x + 2)$.

\emph{القوس المشترك:}
\[
(x+2)(x-1) + (x+2)(3x+4) = (x+2)\bigl[(x-1) + (3x+4)\bigr]
= (x+2)(4x+3).
\]

\emph{\omterm{ex:g1:subtraction:difference}{فرق} مربّعين:}
$x^2 - 81 = (x+9)(x-9)$، و
\[
(2x+1)^2 - 9 = (2x+1)^2 - 3^2 = (2x+1+3)(2x+1-3) = (2x+4)(2x-2).
\]
ولكل عامل عامل مشترك بعدُ: $(2x+4)(2x-2) = 2(x+2) \times
2(x-1) = 4(x+2)(x-1)$.
\end{example}

\section{المعادلات}

\begin{method}[معادلات الدرجة الأولى]\label{met:g9:algebra:firstdegree}
لحلّ \omterm{def:g8:equations:def}{معادلة} مثل $5x - 3 = 2x + 9$:
\begin{enumerate}
  \item اِجمع حدود $x$ في طرف، والأعداد في الطرف الآخر، بإضافة
        المقدار نفسه إلى الطرفين؛
  \item واِختزل كل طرف؛
  \item واِقسم الطرفين على معامل $x$؛
  \item وتحقّق من الحل في \omterm{def:g8:equations:def}{المعادلة} الأصلية.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:algebra:firstdegree}
\begin{align*}
5x - 3 &= 2x + 9 \\
5x - 2x &= 9 + 3 && \text{(بإضافة $3 - 2x$ إلى الطرفين)}\\
3x &= 12 \\
x &= 4 && \text{(بالقسمة على 3).}
\end{align*}
التحقّق: $5 \times 4 - 3 = 17$ و $2 \times 4 + 9 = 17$. فالحل هو
$4$.

ومع الأقواس، اُنشر أولًا: $3(x - 2) = x + 8$ تصير $3x - 6 = x + 8$،
إذن $2x = 14$ و $x = 7$.
\end{example}

\begin{theorem}[قاعدة الجداء المعدوم]\label{thm:g9:algebra:zeroproduct}
\omterm{def:g2:mult:def}{الجداء} معدوم إذا وفقط إذا كان أحد عوامله على الأقل معدومًا:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ أو } \ B = 0 .
\]
\end{theorem}

\begin{proof}
إذا كان أحد العوامل معدومًا فمن الواضح أنّ \omterm{def:g2:mult:def}{الجداء} معدوم.
وبالعكس، إذا كان $A B = 0$ مع $A \neq 0$، فبقسمة الطرفين على $A$
نجد $B = 0$.
\end{proof}

\begin{example}\label{ex:g9:algebra:zeroproduct}
حُلّ $(x - 3)(2x + 10) = 0$: إمّا $x - 3 = 0$ أو $2x + 10 = 0$،
إذن الحلّان $3$ و $-5$.

وحُلّ $x^2 - 49 = 0$: فكّك أولًا، $(x-7)(x+7) = 0$، والحلّان $7$
و $-7$.

وحُلّ $x^2 = 3x$: لا تقسم على $x$ أبدًا! اِنقل وفكّك:
$x^2 - 3x = x(x - 3) = 0$، والحلّان $0$ و $3$.
\end{example}

\section{المتباينات}

\begin{proposition}[قواعد المتباينات]\label{prop:g9:algebra:ineq}
تُحفظ المتباينة عند إضافة العدد نفسه إلى الطرفين، وعند ضرب
الطرفين في العدد \emph{الموجب} نفسه. أمّا الضرب في عدد
\emph{سالب} فيعكس إشارة المتباينة.
\end{proposition}

\begin{proof}
إذا كان $a \leq b$ فإنّ $b - a \geq 0$. وبإضافة $c$: $(b+c) - (a+c) = b - a
\geq 0$، إذن $a + c \leq b + c$. وبالضرب في $c > 0$:
$bc - ac = (b-a)c \geq 0$، إذن $ac \leq bc$. وبالضرب في $c < 0$:
$(b-a)c \leq 0$، إذن $ac \geq bc$: فينعكس الترتيب.
\end{proof}

\begin{example}\label{ex:g9:algebra:ineq}
حُلّ $7 - 2x < 15$:
\begin{align*}
7 - 2x &< 15 \\
-2x &< 8 && \text{(بطرح 7)} \\
x &> -4 && \text{(بالقسمة على $-2$: اِعكس الإشارة).}
\end{align*}
الحلول هي كل الأعداد الأكبر من $-4$: وعلى \omterm{def:g6:lines:objects}{مستقيم} الأعداد، نقطة
مجوّفة عند $-4$ وتظليل نحو اليمين.
\end{example}

\section{مسألة: جبر الأرقام}

\begin{problem}[{مسألة نهاية الأسبوع --- لماذا تشتغل قواعد قابلية
القسمة، وكشف حيلة 1089، وحساب ذهني يشبه السحر}]
\label{pb:g9:algebra:1}
علّمتك المدرسة \emph{قواعد} (فالعدد \omterm{def:g6:wholes:divisible}{قابل للقسمة} على $9$ حين
يكون مجموع أرقامه كذلك)؛ ويؤدّي السحرة \emph{حيلًا} (فينتهي
الجميع عند $1089$)؛ وتعرف الأسواق \emph{اِختصارات}
($35^2 = 1225$، في الحال). وكل ذلك سرّ واحد: فالعدد ذو الأرقام
$a$ و $b$ و $c$ هو العبارة $100a + 10b + c$، وجبر هذا الفصل
(\cref{thm:g9:algebra:identities}، \omterm{def:g9:algebra:expand}{والنشر}، والتفكيك) قادر
على تفكيكه. وفي نهاية هذه المسألة تكون قد برهنت على القواعد،
وكشفت الحيلة، وتعلّمت السحر.

\medskip
\noindent\textbf{الجزء الأول --- الأرقام تحت المجهر.}
\begin{enumerate}
  \item العدد ذو الرقمين الذي رقم عشراته $a$ ورقم آحاده $b$ هو
        $10a + b$. اُنشر وفكّك \omterm{ex:g1:subtraction:difference}{الفرق} بين العدد ومقلوبه،
        $(10a + b) - (10b + a)$، واُذكر الاِكتشاف. وتحقّق منه على
        $72 - 27$.
  \item وبرهن على قاعدة التسعة من أجل الأعداد ذات الثلاثة أرقام:
        تحقّق من المتطابقة
        \[
        100a + 10b + c = 9\,(11a + b) + (a + b + c),
        \]
        وفسّر لماذا تبيّن أنّ العدد ومجموع أرقامه يتركان
        \emph{\omterm{def:g3:division:remainder}{الباقي} نفسه} في القسمة على $9$ --- فيكون
        أحدهما \omterm{def:g6:wholes:divisible}{قابلًا للقسمة} على $9$ بالضبط حين يكون الآخر
        كذلك (\cref{prop:g9:arith:rules}، وقد بُرهن عليها الآن).
  \item واِستنتج قاعدة الثلاثة من المتطابقة نفسها، في جملة واحدة.
  \item و«طرح التساعات»، تحقّق المحاسبين: لاِختبار
        $47 \times 86 = 4\,042$، اِستبدل بكل عدد \omterm{def:g1:addition:def}{مجموع}
        مجموع أرقامه: $47 \to 11 \to 2$، و $86 \to 14 \to 5$،
        و $2 \times 5 = 10 \to 1$؛ ثم $4\,042 \to 10 \to 1$:
        فمتّسق. فسّر لماذا يجب أن يجتاز \omterm{def:g2:mult:def}{الجداء} الصحيح
        الاِختبار (اُكتب $47 = 9k + 2$، و $86 = 9l + 5$ واُنشر)،
        وجِد نتيجة خاطئة يعجز الاِختبار عن اِلتقاطها (لماذا تكون
        الأرقام المتبادلة غير مرئية له؟).
  \item وبرهن على قاعدة الأحد عشر من أجل الأعداد ذات الأربعة
        أرقام: تحقّق من
        \[
        1000a + 100b + 10c + d
        = 11\,(91a + 9b + c) + (- a + b - c + d),
        \]
        واِستعملها لتقرّر ما إذا كان $2\,926$ \omterm{def:g6:wholes:divisible}{قابلًا للقسمة}
        على $11$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- حيلة 1089.}
الحيلة: فكّر في عدد ذي ثلاثة أرقام يختلف رقمه الأول وآخره بمقدار
$2$ على الأقل؛ واِقلبه؛ واِطرح الأصغر من الأكبر؛ واِقلب النتيجة؛
واِجمع العددين الأخيرين. فيعلن الساحر: \emph{1089}.
\begin{enumerate}[resume]
  \item أدِّ الحيلة على $742$ وعلى عدد من اِختيارك.
  \item وليكن العدد ذا الأرقام $a$ و $b$ و $c$ مع $a > c$. فكّك
        الطرح الأول، $(100a + 10b + c) - (100c + 10b + a)$،
        واُذكر كل القيم التي يمكن أن يأخذها حين يمشي $a - c$ من
        $2$ إلى $9$.
  \item ولكل \omterm{def:g9:arith:divisor}{مضاعف} من \omterm{def:g5:division:multiple}{مضاعفات} $99$ هذه نمط لافت في أرقامه.
        وبكتابة $m = a - c$، تحقّق من المتطابقة
        \[
        99m = 100\,(m - 1) + 90 + (10 - m),
        \]
        واِقرأ منها الأرقام الثلاثة للعدد $99m$.
  \item واِجمع $99m$ مع مقلوبه، مستعملًا أرقام السؤال 8، وبرهن
        أنّ \omterm{def:g1:addition:def}{المجموع} هو $1089$ --- مهما كان $m$. ماتت الحيلة؛
        فليحيَ الجبر.
  \item ولماذا اِشترط الساحر أن يختلف الرقمان الأول والأخير
        بمقدار $2$ على الأقل؟ اُنظر في الحالتين $a = c$
        و $a - c = 1$ (حيث يعطي الطرح $99$، الذي يجب أن يُكتب
        $099$ ليشتغل القلب)، واُذكر الشرط الصغير الذي يُبقي
        الحيلة أمينة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- سحر للاِستعمال اليومي.}
\begin{enumerate}[resume]
  \item حيلة عيد الميلاد: «اِضرب شهر ميلادك في $5$، وأضِف $7$،
        وضاعِف، وأضِف يوم ميلادك، واِطرح $14$». بيّن أنّ النتيجة
        هي دائمًا $10m + d$ --- الشهر واليوم، مقروءين بنظرة
        واحدة. (جرّبها على أحدهم.)
  \item وتعيد تصميم الحيلة هكذا: «اِضرب الشهر في $5$، وأضِف $9$،
        وضاعِف، وأضِف اليوم». فما العدد الذي يجب أن يطرحه
        المتفرّج في النهاية ليقرأ الشهر واليوم نفسيهما؟
  \item وحيلة الرقم الناقص: يختار المتفرّج عددًا ما، ويطرح منه
        مجموع أرقامه (ويقول السؤال 2 إنّ النتيجة \omterm{def:g9:arith:divisor}{مضاعف}
        للعدد $9$ دائمًا --- فسّر ذلك من أجل عدد ذي ثلاثة
        أرقام)، ثم يشطب رقمًا \emph{غير معدوم} ويقرأ عليك
        الأرقام الباقية. فسّر كيف تستعيد الرقم المشطوب، ولماذا
        وجب اِستبعاد $0$.
  \item وسحر السوق: برهن على المتطابقة
        $(10t + 5)^2 = 100\,t(t+1) + 25$، واُذكر قاعدة تربيع أيّ
        عدد ينتهي بالرقم $5$، واِحسب $45^2$ و $85^2$ و $105^2$
        في رأسك.
  \item والختام، بتوسيع \cref{exo:g9:algebra:10}: برهن أنّ
        \omterm{def:g1:addition:def}{مجموع} خمسة أعداد صحيحة متتالية هو دائمًا \omterm{def:g9:arith:divisor}{مضاعف}
        للعدد $5$، لكنّ \omterm{def:g1:addition:def}{مجموع} \emph{أربعة} أعداد صحيحة
        متتالية \emph{ليس أبدًا} مضاعفًا للعدد $4$. (حسابان
        قصيران --- وعبرة واحدة عمّا يصنعه الجبر بكلمتَي «دائمًا»
        و«أبدًا».)
\end{enumerate}
\end{problem}
