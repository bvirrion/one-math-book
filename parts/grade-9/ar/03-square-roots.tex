\chapter{الجذور التربيعية}\label{ch:g9:sqrt}

كم يبلغ طول \omterm{def:g4:geometry:circle}{قطر} مربّع طول ضلعه $1$؟ تجيب مبرهنة فيثاغورس بالعدد
$\sqrt 2$، وهو عدد مربّعه $2$ --- ويتبيّن أنّه ليس كسرًا. يعرّف
هذا الفصل \omterm{def:g9:sqrt:def}{الجذور التربيعية}، ويرسي قواعد الحساب بها، ويعلّم كيف
تُختزل عبارات مثل $\sqrt{75}$.

\section{التعريف والخواص الأولى}

\begin{definition}[الجذر التربيعي]\label{def:g9:sqrt:def}
ليكن $a \geq 0$. \emph{الجذر التربيعي}\index{جذر تربيعي} للعدد $a$،
ويُكتب $\sqrt a$، هو العدد \emph{غير السالب} الوحيد الذي مربّعه $a$:
\[
\sqrt a \geq 0
\qquad\text{و}\qquad
\left(\sqrt a\right)^2 = a .
\]
وليس للأعداد السالبة جذر تربيعي، لأنّ كل مربّع غير سالب.
\end{definition}

\begin{example}
$\sqrt{49} = 7$، و $\sqrt 0 = 0$، و $\sqrt 1 = 1$،
و $\sqrt{2.25} = 1.5$. وأول \omterm{def:g9:sqrt:def}{الجذور التربيعية} التي تُحفظ عن ظهر
قلب هي جذور \emph{المربّعات التامّة}: $1, 4, 9, 16, 25, 36, 49, 64, 81, 100, 121,
144$.
\end{example}

\begin{proposition}[الجذر التربيعي لمربّع]\label{prop:g9:sqrt:square}
من أجل كل عدد حقيقي $x$ (موجبًا كان أو غير موجب):
\[
\sqrt{x^2} = \abs{x} =
\begin{cases}
x & \text{إذا كان } x \geq 0,\\
-x & \text{إذا كان } x < 0 .
\end{cases}
\]
\end{proposition}

\begin{proof}
العدد $\abs{x}$ غير سالب ومربّعه $x^2$ (فالعدد ونظيره لهما المربّع
نفسه). وبما أنّه العدد غير السالب الوحيد الذي مربّعه $x^2$، فهو
$\sqrt{x^2}$.
\end{proof}

\begin{example}
$\sqrt{(-5)^2} = \sqrt{25} = 5 = \abs{-5}$: \omterm{def:g9:sqrt:def}{فالجذر التربيعي} «ينسى»
الإشارة ولا يعيدها. وبخاصّة، فإنّ $\sqrt{x^2} = x$ \emph{خطأ} من
أجل $x$ سالب.
\end{example}

\section{جداءات الجذور التربيعية وخوارج قسمتها}

\begin{theorem}[قاعدتا الضرب والقسمة]\label{thm:g9:sqrt:rules}
من أجل كل $a \geq 0$ و $b \geq 0$:
\[
\sqrt{a b} = \sqrt a \times \sqrt b,
\qquad\text{ومن أجل } b > 0:\quad
\sqrt{\frac ab} = \frac{\sqrt a}{\sqrt b} .
\]
\end{theorem}

\begin{proof}
العدد $\sqrt a \times \sqrt b$ غير سالب (\omterm{def:g2:mult:def}{جداء} عددين غير
سالبين)، ومربّعه
\[
\left(\sqrt a \times \sqrt b\right)^2
= \left(\sqrt a\right)^2 \times \left(\sqrt b\right)^2
= a b .
\]
وبوحدانية العدد غير السالب الذي مربّعه $ab$، يكون مساويًا
$\sqrt{ab}$. وتُبرهن قاعدة \omterm{def:g3:division:remainder}{خارج القسمة} بالطريقة نفسها.
\end{proof}

\begin{remark}[لا قاعدة كهذه للمجاميع!]
$\sqrt{a + b}$ \emph{ليس} $\sqrt a + \sqrt b$ في العموم:
\[
\sqrt{9 + 16} = \sqrt{25} = 5,
\qquad\text{لكن}\qquad
\sqrt 9 + \sqrt{16} = 3 + 4 = 7 .
\]
\end{remark}

\begin{method}[اِختزال $\sqrt n$]\label{met:g9:sqrt:simplify}
لِاختزال \omterm{def:g9:sqrt:def}{الجذر التربيعي} لعدد صحيح:
\begin{enumerate}
  \item جِد أكبر مربّع تامّ \omterm{def:g9:arith:divisor}{يقسم} $n$ (وفكّك $n$ إن لزم الأمر)؛
  \item واِفصل بقاعدة \omterm{def:g2:mult:def}{الجداء} واِستخرج المربّع:
        $\sqrt{k^2 m} = k\sqrt m$؛
  \item وتحقّق أنّ لا مربّع تامّ \omterm{def:g9:arith:divisor}{يقسم} ما تبقّى تحت الجذر.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:sqrt:simplify}
اِختزل $\sqrt{75}$: بما أنّ $75 = 25 \times 3$، فإنّ
\[
\sqrt{75} = \sqrt{25 \times 3} = \sqrt{25} \times \sqrt 3 = 5\sqrt3 .
\]
واِختزل $\sqrt{72}$: بما أنّ $72 = 36 \times 2$، يكون
$\sqrt{72} = 6\sqrt2$. \omterm{def:g3:division:remainder}{وخارج قسمة}:
$\sqrt{\dfrac{49}{16}} = \dfrac{\sqrt{49}}{\sqrt{16}} = \dfrac74$.
\end{example}

\begin{example}[جمع الجذور التربيعية]\label{ex:g9:sqrt:add}
مجاميع \omterm{def:g9:sqrt:def}{الجذور التربيعية} تُختزل حين تكون الجذور
\emph{متشابهة} فقط. اِحسب $\sqrt{12} + \sqrt{27} - \sqrt{48}$،
مختزلًا كل حدّ أولًا:
\begin{align*}
\sqrt{12} &= \sqrt{4 \times 3} = 2\sqrt3, \\
\sqrt{27} &= \sqrt{9 \times 3} = 3\sqrt3, \\
\sqrt{48} &= \sqrt{16 \times 3} = 4\sqrt3,
\end{align*}
فيكون \omterm{def:g1:addition:def}{المجموع} $2\sqrt3 + 3\sqrt3 - 4\sqrt3 = (2 + 3 - 4)\sqrt3 = \sqrt3$.
\end{example}

\begin{example}[النشر مع الجذور التربيعية]\label{ex:g9:sqrt:expand}
تنطبق متطابقات الجبر على \omterm{def:g9:sqrt:def}{الجذور التربيعية}. اِنشر
$\left(\sqrt5 + 2\right)^2$ باِستعمال $(a+b)^2 = a^2 + 2ab + b^2$:
\[
\left(\sqrt5 + 2\right)^2 = 5 + 2 \times 2\sqrt5 + 4 = 9 + 4\sqrt5 .
\]
وبالمتطابقة الثالثة، $(a+b)(a-b) = a^2 - b^2$:
\[
\left(\sqrt7 + \sqrt3\right)\left(\sqrt7 - \sqrt3\right) = 7 - 3 = 4 :
\]
\omterm{def:g2:mult:def}{فجداء} عددين أصمّين قد يكون عددًا صحيحًا.
\end{example}

\section{المعادلة \texorpdfstring{$x^2 = a$}{x2 = a}}

\begin{theorem}[حل $x^2 = a$]\label{thm:g9:sqrt:equation}
\begin{enumerate}
  \item إذا كان $a > 0$، كان \omterm{def:g8:equations:def}{للمعادلة} $x^2 = a$ حلّان اِثنان
        بالضبط: $\sqrt a$ و $-\sqrt a$؛
  \item وإذا كان $a = 0$، كان الحل الوحيد هو $0$؛
  \item وإذا كان $a < 0$، فلا حلّ.
\end{enumerate}
\end{theorem}

\begin{proof}
أعِد كتابة $x^2 = a$ على صورة $x^2 - \left(\sqrt a\right)^2 = 0$ حين
$a \geq 0$، وفكّكها \omterm{ex:g1:subtraction:difference}{كفرق} مربّعين:
\[
\left(x - \sqrt a\right)\left(x + \sqrt a\right) = 0 ,
\]
إذن $x = \sqrt a$ أو $x = -\sqrt a$ (ويتطابق الحلّان حين $a = 0$).
وحين $a < 0$ فلا حلّ، لأنّ $x^2 \geq 0 > a$ من أجل كل $x$.
\end{proof}

الأفقي $y = 3$ عند الفاصلتين $\pm\sqrt3$.}
\end{omfigure}

\begin{example}
$x^2 = 16$: الحلّان $4$ و $-4$. \quad
$x^2 = 5$: الحلّان $\sqrt5$ و $-\sqrt5$ (قيمتان مضبوطتان؛ و $\sqrt5
\approx 2.236$). \quad
$x^2 = -9$: لا حلّ. \quad
$3x^2 = 21$: اِقسم على 3 أولًا، فيكون $x^2 = 7$، والحلّان $\pm\sqrt7$.
\end{example}

\begin{exercise}[$\star\star$]\label{exo:g9:sqrt:9}

  \item من أجل كل $a, b \geq 0$: $\sqrt{ab} = \sqrt a \sqrt b$.
  \item من أجل كل $a, b \geq 0$: $\sqrt{a+b} = \sqrt a + \sqrt b$.
  \item من أجل كل $x$: $\sqrt{x^2} = x$.
  \item $\left(3\sqrt2\right)^2 = 18$.
\end{enumerate}
\end{exercise}

\section{مسألة: العدد الذي ليس كسرًا}

\begin{problem}[{مسألة نهاية الأسبوع --- البرهان الأسطوري على أنّ
$\sqrt2$ عدد أصمّ، وأبناء عمومته الكثيرون، ووصفة هيرون الجيدة
جودةً مذهلة لتقريبه}]
\label{pb:g9:sqrt:1}
\omterm{def:g4:geometry:circle}{قطر} المربّع الواحدي طول حقيقي تمامًا --- وقد رسمته في
\cref{ex:g8:pythagoras:diagonal} --- ومع ذلك، كما اِكتشف
الفيثاغوريون مذعورين قبل نحو خمسة وعشرين قرنًا، \emph{لا يقيسه
أيّ \omterm{def:g9:fractions:fraction}{كسر} كان}. وتزعم الأسطورة أنّ الاِكتشاف عوقب بالإغراق.
وتقودك هذه المسألة عبر البرهان الخالد (أربعة أسئلة ويصير لك مدى
الحياة)، وتضاعف الضحايا، وتنتهي بالوصفة التي اِستعملها المهندسون
ألفي سنة لترويض العدد العصيّ على الترويض.

\medskip
\noindent\textbf{الجزء الأول --- البرهان.}
\begin{enumerate}
  \item تعقّبه أولًا: تحقّق أنّ
        $1.4 < \sqrt2 < 1.5$، ثم أنّ $1.41 < \sqrt2 < 1.42$،
        بتربيع الحدّين. ورقم إضافي: بين أيّ عددين بثلاثة أرقام
        عشرية يقع $\sqrt2$؟
  \item واِفترض الآن --- على سبيل الخُلف --- أنّ
        $\sqrt2 = \frac ab$ من أجل \omterm{def:g9:fractions:fraction}{كسر} \emph{مختزل إلى أبسط
        صورة} (\cref{met:g9:arith:simplify}). ربّع الطرفين وبيّن
        أنّ $a^2 = 2 b^2$. فما زوجية $a^2$؟
  \item وتقول وقائع الزوجية في \cref{pb:g8:pythagoras:1}:
        \omterm{def:g3:numbers:evenodd}{الأعداد الفردية} مربّعاتها فردية. اِستنتج أنّ $a$
        زوجي، واِكتب $a = 2k$، وعوّض --- وبيّن أنّ $b$ يجب
        أن يكون زوجيًا أيضًا.
  \item فأين التناقض؟ اِستنتج، واُذكر المبرهنة كاملة:
        \emph{العدد $\sqrt2$ أصمّ --- فهو ليس \omterm{def:g3:division:remainder}{خارج قسمة}
        عددين صحيحين}.
  \item وقد اِتّكأ البرهان مرّة واحدة، في خفاء، على عبارة «مختزل
        إلى أبسط صورة». أشِر إلى الخطوة الدقيقة التي تنهار من
        دونها، وفسّر لماذا كان اِفتراض الاِختزال مشروعًا أصلًا.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الضحايا تتكاثر.}
\begin{enumerate}[resume]
  \item وأسلوب برهان ثانٍ، عبر التفكيك إلى عوامل أولية
        (\cref{thm:g9:arith:factorization}): في $a^2 = 3b^2$،
        قارن زوجية \emph{\omterm{def:g8:powers:def}{أُسّ} العدد $3$} في كل طرف
        (\cref{pb:g9:arith:1}: المربّعات زوجية
        \omterm{def:g8:powers:def}{الأُسُس}). اِستنتج أنّ $\sqrt3$ أصمّ.
  \item وأجرِ حجّة \omterm{def:g8:powers:def}{الأُسّ} نفسها على $a^2 = n b^2$ من أجل عدد
        صحيح عام $n$: من أجل أيّ $n$ تعطي تناقضًا، ومن أجل أيّها
        تفشل؟ اُذكر النتيجة الكاملة: $\sqrt n$ أصمّ بالضبط
        حين\,\dots
  \item وبرهن على مبرهنة الدرع: \omterm{def:g2:mult:def}{جداء} عدد ناطق غير معدوم في عدد
        أصمّ هو عدد أصمّ. (اِفترض $r x = q$ حيث $r, q$ ناطقان،
        و $r \neq 0$، وحُلّ من أجل $x$.) واِستنتج أنّ $2\sqrt2$
        و $\frac{\sqrt2}{2}$ أصمّان.
  \item وبرهن أنّ $1 + \sqrt2$ أصمّ. ثم الجميل: بافتراض أنّ
        $s = \sqrt2 + \sqrt3$ ناطق، اِحسب $s^2$، واِعزل $\sqrt6$،
        وجِد التناقض.
  \item وخفّف الحماس: أعطِ عددين أصمّين \emph{مجموعهما} ناطق،
        وعددين \emph{جداؤهما} ناطق. (فالصمم لا يُحفظ في الحساب
        --- وكل حالة تحتاج برهانها الخاصّ.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- وصفة هيرون.}
قبل الآلات الحاسبة بألفي سنة، كان هيرون السكندري يقرّب $\sqrt2$
هكذا: \emph{خمّن $x$؛ ثم اِستبدل بالتخمين متوسّط $x$ و $\frac2x$؛
وكرّر.}
\begin{enumerate}[resume]
  \item اِنطلق من التخمين $x = 1$ واِحسب التخمينين التاليين
        كسرين مضبوطين.
  \item واِحسب التخمين الثالث، كسرًا مضبوطًا أيضًا، وقيمته
        العشرية. وقارن مع السؤال 1: كم رقمًا عشريًا من $\sqrt2$
        صار صحيحًا؟
  \item والفكرة وراء الوصفة: مستطيل مساحته $2$ وأحد ضلعيه $x$
        يكون ضلعه الآخر $\frac2x$. بيّن أنّ $\sqrt2$ يقع دائمًا
        \emph{بين} $x$ و $\frac2x$ (اُنظر في الحالتين $x^2 > 2$
        و $x^2 < 2$)، فيضغط متوسّط الضلعين المستطيل نحو المربّع.
  \item والتخمينات $\frac32$ و $\frac{17}{12}$ و
        $\frac{577}{408}$ تخفي جوهرة: اِحسب $3^2 - 2 \times
        2^2$، ثم $17^2 - 2 \times 12^2$، ثم
        $577^2 - 2 \times 408^2$. وقد برهن الجزء الأول على
        اِستحالة $a^2 - 2b^2 = 0$ --- فكم تقترب كسور هيرون
        من المستحيل؟
  \item والختام، في جملتين: \omterm{def:g4:geometry:circle}{قطر} المربّع الواحدي موجود على الورق،
        ولا يقيسه أيّ \omterm{def:g9:fractions:fraction}{كسر}، وكتابته العشرية لا يمكن أن تصير
        دورية أبدًا (\cref{pb:g9:fractions:1}). فأيّ «أعداد
        جديدة» يجب إذن أن يحويها \omterm{def:g6:lines:objects}{مستقيم} الأعداد --- وأين في هذه
        السلسلة تُروى حكايتها كاملة؟
\end{enumerate}
\end{problem}
