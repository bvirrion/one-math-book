\chapter{مبرهنة طاليس}\label{ch:g9:thales}

كيف تقيس اِرتفاع \omterm{def:g8:solids:def}{هرم} من غير أن تتسلّقه؟ قارن طاليس الميليتي
ظلّه بظلّ عصًا. ومبرهنته --- \omterm{def:g6:lines:objects}{المستقيمات} المتوازية تقطع \omterm{def:g6:lines:objects}{القطع}
قِطعًا متناسبة --- هي القلب الرياضي لنماذج \omterm{def:g7:prop:scale}{السلّم} والخرائط
والتكبيرات، وإحدى أقدم المبرهنات التي تحمل اِسمًا.

\section{المبرهنة}

\begin{theorem}[طاليس]\label{thm:g9:thales:direct}
ليتقاطع \omterm{def:g6:lines:objects}{مستقيمان} في نقطة $A$. وخذ $M$ على الأول و $N$ على الثاني،
مع $B$ أبعد على الأول و $C$ أبعد على الثاني. فإذا كان \omterm{def:g6:lines:objects}{المستقيمان}
$(MN)$ و $(BC)$ \emph{\omterm{def:g6:lines:perp}{متوازيين}}، كان للمثلثين $AMN$ و $ABC$
أضلاع متناسبة:
\[
\frac{AM}{AB} = \frac{AN}{AC} = \frac{MN}{BC}.
\]
\end{theorem}
\admitted

\begin{method}[حساب طول بمبرهنة طاليس]\label{met:g9:thales:compute}
\begin{enumerate}
  \item عيّن النقطة $A$ حيث يتقاطع \omterm{def:g6:lines:objects}{المستقيمان}،
        \omterm{def:g6:lines:objects}{والمستقيمين} \omterm{def:g6:lines:perp}{المتوازيين}؛ وسمِّ المثلثين؛
  \item واُكتب النسب الثلاث المتساوية، دائمًا «المثلث الصغير على
        المثلث الكبير»؛
  \item واِحتفظ بتساوي النسبتين اللتين تضمّان الأطوال الثلاثة
        المعلومة والطول المجهول؛
  \item وحُلّ \omterm{def:g9:linfunc:linear}{التناسب} الناتج بالضرب المتقاطع.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:thales:compute}
في الوضعية اليسرى أعلاه، ليكن $AM = 3$ و $AB = 5$ و $AN = 2.4$
و $MN = 3.3$، ولنحسب $AC$ و $BC$. تعطي مبرهنة طاليس
\[
\frac{AM}{AB} = \frac{AN}{AC} = \frac{MN}{BC},
\qquad\text{أي}\qquad
\frac35 = \frac{2.4}{AC} = \frac{3.3}{BC}.
\]
ومن $\frac35 = \frac{2.4}{AC}$، بالضرب المتقاطع:
$3 \times AC = 5 \times 2.4 = 12$، إذن $AC = 4$. ومن
$\frac35 = \frac{3.3}{BC}$: $3 \times BC = 16.5$، إذن $BC = 5.5$.
\end{example}

\begin{example}[طاليس في الحياة الواقعية]\label{ex:g9:thales:stick}
عصًا عمودية اِرتفاعها $1$ م تلقي ظلًّا طوله $1.5$ م، بينما تلقي
شجرة ظلًّا طوله $12$ م في اللحظة نفسها. وأشعّة الشمس
متوازية، فيكون مثلث العصا وظلّها ومثلث الشجرة وظلّها في وضعية
طاليس:
\[
\frac{\text{اِرتفاع الشجرة}}{1} = \frac{12}{1.5},
\qquad\text{فاِرتفاع الشجرة } 8 \text{ م.}
\]
\end{example}

\section{المبرهنة العكسية}

\begin{theorem}[عكس مبرهنة طاليس]\label{thm:g9:thales:converse}
بوضع النقط كما في \cref{thm:g9:thales:direct} (فتكون $M$ على $(AB)$،
و $N$ على $(AC)$، بالترتيب نفسه على \omterm{def:g6:lines:objects}{المستقيمين}):
إذا كان
\[
\frac{AM}{AB} = \frac{AN}{AC},
\]
كان \omterm{def:g6:lines:objects}{المستقيمان} $(MN)$ و $(BC)$ \omterm{def:g6:lines:perp}{متوازيين}.
\end{theorem}
\admitted

\begin{method}[البرهنة على التوازي أو نفيه]\label{met:g9:thales:converse}
\begin{enumerate}
  \item اِحسب على حدة النسبتين $\dfrac{AM}{AB}$
        و $\dfrac{AN}{AC}$ (كسورًا، لا تقريبات)؛
  \item فإذا تساوتا \emph{وكانت} النقط بالترتيب نفسه على
        \omterm{def:g6:lines:objects}{المستقيمين}، كان \omterm{def:g6:lines:objects}{المستقيمان} \omterm{def:g6:lines:perp}{متوازيين} (عكس مبرهنة طاليس)؛
  \item وإذا اِختلفتا، \omterm{def:g6:lines:objects}{فالمستقيمان} \emph{ليسا} \omterm{def:g6:lines:perp}{متوازيين} ---
        لأنّهما لو كانا كذلك، لفرضت مبرهنة طاليس تساوي النسبتين.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:thales:converse}
على \omterm{def:g6:lines:objects}{المستقيم} الأول، $AM = 4$ و $AB = 10$؛ وعلى الثاني، $AN = 6$
و $AC = 15$. إذن
\[
\frac{AM}{AB} = \frac{4}{10} = \frac25,
\qquad
\frac{AN}{AC} = \frac{6}{15} = \frac25 :
\]
نسبتان متساويتان، والترتيب نفسه: إذن $(MN) \parallel (BC)$.

وبأخذ $AC = 14$ بدلًا من ذلك: $\frac{6}{14} = \frac37 \neq \frac25$،
\omterm{def:g6:lines:objects}{فالمستقيمان} ليسا \omterm{def:g6:lines:perp}{متوازيين}.
\end{example}

\section{التكبير والتصغير}

\begin{definition}[تحجيم شكل]\label{def:g9:thales:scaling}
تكبير شكل أو تصغيره \emph{بمعامل التحجيم}\index{معامل تحجيم}
$k > 0$ يعني ضرب \emph{كل} أطواله في $k$: فيكون \emph{تكبيرًا} حين
$k > 1$، و\emph{تصغيرًا} حين $k < 1$. وفي وضعية طاليس، يكون المثلث
$AMN$ تصغيرًا للمثلث $ABC$ بالمعامل $k = \frac{AM}{AB}$.
\end{definition}

\begin{proposition}[الأثر في الزوايا والمساحات]\label{prop:g9:thales:scaling}
التحجيم بالمعامل $k$ يحفظ الزوايا وشكل الأشكال، ويضرب كل طول في
$k$، ويضرب كل \emph{\omterm{def:g6:measure:area}{مساحة}} في $k^2$.
\end{proposition}
\admitted

\begin{omfigure}
\begin{tikzpicture}[scale=0.72]
  \draw[thick, omDef, fill=omDef!8] (0,0) rectangle (2,1.2);
  \draw[thick, omThm, fill=omThm!8] (3.4,0) rectangle (7.4,2.4);
  \node[font=\small, omDef] at (1,-0.45) {$2 \times 1.2$};
  \node[font=\small, omThm] at (5.4,-0.45) {$4 \times 2.4$};
  \node[font=\small] at (1,0.6) {مساحة $2.4$};
  \node[font=\small] at (5.4,1.2) {مساحة $9.6$};
\end{tikzpicture}

{\small مضاعفة الأطوال ($k = 2$) تضرب \omterm{def:g6:measure:area}{المساحة} في
$k^2 = 4$: فأربع نسخ من المستطيل الصغير تبلّط الكبير.}
\end{omfigure}

\begin{example}
صورة قياسها $10 \times 15$ سم تُكبَّر \omterm{def:g9:thales:scaling}{بمعامل تحجيم} $k = 3$:
فتقيس المطبوعة $30 \times 45$ سم. وتنتقل مساحتها من $150$ سم$^2$
إلى $150 \times 9 = 1350$ سم$^2$ --- أي $9$ أمثال، لا $3$.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g9:thales:1}
\omterm{def:g6:lines:objects}{المستقيمان} $(MN)$ و $(BC)$ متوازيان، مع $M$ على $[AB]$ و $N$
على $[AC]$؛ و $AM = 2$ و $AB = 6$ و $AN = 3$ و $BC = 9$. اِحسب $AC$ و $MN$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:2}
الوضعية نفسها: $AM = 5$ و $MB = 3$ (اِنتبه: $MB$ لا $AB$!)
و $AN = 4$. اِحسب $AC$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:3}
في وضعية فراشة، تقع $A$ بين $M$ و $B$ وبين $N$ و $C$، مع
$(MN) \parallel (BC)$؛ و $AM = 3$ و $AB = 7.5$ و $AN = 2$
و $MN = 2.6$. اِحسب $AC$ و $BC$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:4}
$M$ على $[AB]$ حيث $AM = 6$ و $AB = 8$؛ و $N$ على $[AC]$ حيث $AN = 9$
و $AC = 12$. فهل \omterm{def:g6:lines:objects}{المستقيمان} $(MN)$ و $(BC)$ متوازيان؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:5}
كما سبق مع $AM = 4$ و $AB = 6$ و $AN = 5$ و $AC = 8$. فهل $(MN)$
و $(BC)$ متوازيان؟ برّر بعناية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:6}
شخص طوله $1.8$ م يقف على بعد $2$ م من قاعدة مصباح شارع؛ وظلّه
يقيس $3$ م. اُرسم وضعية طاليس التي يشكّلها المصباح والشخص وطرف
الظلّ، واِحسب اِرتفاع المصباح.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:7}
خريطة سلّمها $1 : 25\,000$ (فالأطوال على الخريطة هي الأطوال
الحقيقية مضروبة في $k = \frac{1}{25000}$).
\begin{enumerate}
  \item تبعد قريتان $6.8$ سم إحداهما عن الأخرى على الخريطة. فما
        المسافة الحقيقية، بالكيلومترات؟
  \item وغابة مساحتها $8$ سم$^2$ على الخريطة. فما مساحتها
        الحقيقية، بالكلم$^2$؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:8}
في المثلث $ABC$: $AB = 12$ و $AC = 15$ و $BC = 18$. والنقطة $M$
على $[AB]$ تحقّق $AM = 8$، \omterm{def:g6:lines:objects}{والمستقيم} المارّ بالنقطة $M$
\omterm{def:g6:lines:perp}{والموازي} \omterm{def:g6:lines:objects}{للمستقيم} $(BC)$ يقطع $[AC]$ في $N$.
\begin{enumerate}
  \item اِحسب $AN$ و $MN$.
  \item وما \omterm{def:g9:thales:scaling}{معامل التحجيم} من $ABC$ إلى $AMN$؟ وقارن
        محيطَي المثلثين.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:thales:9}
ليكن $ABCD$ شبه منحرف حيث $(AB) \parallel (CD)$ و $AB = 4$
و $CD = 6$، ويتقاطع قطراه $[AC]$ و $[BD]$ في $O$.
وباِستعمال طاليس في وضعية الفراشة حول $O$، اِحسب النسبة
$\dfrac{OA}{OC}$، وبيّن أنّ $O$ يقطع القطرين بالنسبة نفسها.
\end{exercise}

\section{مسألة: المسطرة غير المدرّجة، وآلة التصوير ذات الثقب،
والمتوسطات الثلاثة لشبه المنحرف}

\begin{problem}[{مسألة نهاية الأسبوع --- طاليس في العمل: تقسيم أيّ
قطعة إلى أجزاء متساوية، وقياس الشمس بعلبة أحذية، وشبه منحرف
تلتقي فيه المتوسطات الشهيرة الثلاثة}]
\label{pb:g9:thales:1}
مبرهنة طاليس هي رياضيات \emph{الأشعّة}: أشعّة الشمس، وأشعّة الضوء
عبر ثقب، وحزم الأشعّة من رأس. وتستعملها هذه المسألة على ثلاثة
وجوه --- أداةَ إنشاء (لتقسيم \omterm{def:g6:lines:objects}{قطعة} أيًّا كان طولها، حتى
$\sqrt2$، إلى أجزاء متساوية تمامًا)، وآلةَ قياس (\omterm{def:g4:geometry:circle}{قطر} الشمس،
بعلبة أحذية \omterm{def:g6:lines:objects}{وقطعة} نقود)، وعدسةً مكبّرة على شبه المنحرف في
\cref{exo:g9:thales:9}، حيث تظهر المتوسطات الحسابي والهندسي
والتوافقي في هذه السلسلة كلّها في شكل واحد.

\medskip
\noindent\textbf{الجزء الأول --- التقسيم بمسطرة غير مدرّجة.}
\begin{enumerate}
  \item اُرسم \omterm{def:g6:lines:objects}{قطعة} $[AB]$ طولها $7$ سم. ولتقسيمها إلى ثلاثة
        أجزاء متساوية بالفرجار والمسطرة غير المدرّجة: اُرسم نصف
        \omterm{def:g6:lines:objects}{مستقيم} أيًّا كان من $A$ (لا يمرّ بالنقطة $B$)؛ واِنقل
        عليه ثلاثة أطوال متساوية بالفرجار، فتحصل على النقط
        $P_1$ و $P_2$ و $P_3$؛ وصِل $P_3$ بالنقطة $B$؛ واُرسم
        \omterm{def:g6:lines:perp}{الموازيَين} \omterm{def:g6:lines:objects}{للمستقيم} $(P_3 B)$ المارّين بالنقطتين $P_1$
        و $P_2$. أنجِز الإنشاء وعلّم أين يقطع \omterm{def:g6:lines:perp}{الموازيان}
        \omterm{def:g6:lines:objects}{القطعة} $[AB]$.
  \item وبرّر باِستعمال \cref{thm:g9:thales:direct} أنّ النقطتين
        المعلَّمتين تقطعان $[AB]$ عند ثلثيها بالضبط.
  \item وكيّف الطريقة لتقسيم \omterm{def:g6:lines:objects}{قطعة} إلى $5$ أجزاء متساوية، ثم
        لإنشاء $\frac35$ من \omterm{def:g6:lines:objects}{قطعة} معطاة.
  \item وأنشِئ النقطة $M$ من $[AB]$ حيث
        $\frac{AM}{MB} = \frac23$ (أي $M$ عند $\frac25$ من
        الطريق اِنطلاقًا من $A$). فكم خطوة متساوية على نصف
        \omterm{def:g6:lines:objects}{المستقيم}، وأيّ نقطة تُوصَل بالنقطة $B$؟
  \item ولماذا يكون هذا الإنشاء أفضل من القياس بمسطرة مدرّجة؟
        اُنظر في \omterm{def:g6:lines:objects}{قطعة} طولها $\sqrt2$ (\omterm{def:g4:geometry:circle}{قطر} مربّع واحدي، وهو أصمّ
        حسب \cref{pb:g9:sqrt:1}): فماذا يعطي القياس، وماذا
        تعطي مبرهنة طاليس؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- آلة التصوير ذات الثقب.}
اِثقب بدبّوس أحد \omterm{def:g5:solids:def}{أوجه} علبة مغلقة: فتظهر على الوجه المقابل
صورة مقلوبة للعالم. فنقطة من الجسم، والثقب، ونقطة الصورة على
اِستقامة واحدة --- إذ يسير الضوء مستقيمًا --- فيقع الجسم
(اِرتفاعه $H$، وعلى بعد $D$ أمام الثقب) وصورته (اِرتفاعها $h$،
على الجدار الخلفي على عمق $d$ خلف الثقب) في وضعية الفراشة عند
طاليس.
\begin{enumerate}[resume]
  \item اِستنتج صيغة الثقب
        $h = H \times \frac dD$ من طاليس في الفراشة حول الثقب.
  \item وشجرة اِرتفاعها $12$ م تقف على بعد $30$ م من علبة أحذية
        عمقها $20$ سم. فما اِرتفاع صورتها، وفي أيّ اِتّجاه؟
  \item ووجّه العلبة نحو الشمس: على بعد $d = 1$ م خلف الثقب،
        تكون صورة الشمس قرصًا قطره نحو $9$ مم. وبمعرفة بعد
        الشمس $D = 1.5 \times 10^{11}$ م، اِحسب قطرها
        \omterm{def:g9:fractions:scientific}{بالترميز العلمي} (وترميز \cref{pb:g9:fractions:1} في
        العمل). وقارن بالقيمة الحقيقية،
        $1.39 \times 10^9$ م.
  \item والقمر: قطره $3.5 \times 10^6$ م، وبعده
        $3.8 \times 10^8$ م. اِحسب النسبة
        $\frac{\text{القطر}}{\text{البعد}}$ للقمر وللشمس. فأيّ
        مصادفة سعيدة يكشفها العددان --- وأيّ حدث مذهل يجعلانه
        ممكنًا؟
  \item وفي جملة أو جملتين: لماذا تكون صورة الثقب مقلوبة، ولماذا
        يجعل توسيع الثقب الصورة أسطع لكن أكثر ضبابية؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- شبه المنحرف ذو المتوسطات
الثلاثة.}
و $ABCD$ هو شبه المنحرف في \cref{exo:g9:thales:9}:
$(AB) \parallel (CD)$، و $AB = 4$، و $CD = 6$، ويتقاطع القطران في
$O$ حيث $\frac{OA}{OC} = \frac{OB}{OD} = \frac{4}{6} =
\frac23$.
\begin{enumerate}[resume]
  \item تمرين على الفراشة: يتقاطع \omterm{def:g6:lines:objects}{مستقيمان} في $O$ مع
        $(MN) \parallel (PQ)$ في وضعية فراشة؛ و $OM = 3$،
        و $OP = 7.5$، و $MN = 4$، و $OQ = 6$. اِحسب $PQ$
        و $ON$.
  \item وفسّر كيف تحوي مبرهنة طاليس، مطبَّقةً بالنسبة
        $\frac12$، مبرهنة المنتصفات في
        \cref{thm:g8:midpoints:direct} حالةً خاصّة منها.
  \item واُرسم \omterm{def:g6:lines:perp}{الموازي} للقاعدتين المارّ بالنقطة $O$؛ فيلتقي
        $[AD]$ في $E$. وبالعمل في المثلث $ACD$ (ولاحظ أنّ
        $\frac{AO}{AC} = \frac25$)، اِحسب $EO$.
  \item وبالحساب المرآتي في المثلث $BDC$، تكون \omterm{def:g6:lines:objects}{القطعة} $OF$
        (حيث $F$ على $[BC]$) بالطول نفسه. اِستنتج أنّ
        \[
        EF = 2 \times \frac{4 \times 6}{4 + 6} = 4.8 :
        \]
        أي أنّ \omterm{def:g6:lines:perp}{الموازي} المارّ بتقاطع القطرين يقيس
        \emph{\omterm{pb:g8:speed:1}{المتوسط التوافقي}} للقاعدتين --- متوسط الذهاب
        والإياب في \cref{pb:g8:speed:1}، عائدًا في هندسة خالصة.
  \item والختام الكبير: اِحسب المتوسطات الكلاسيكية الثلاثة
        للقاعدتين $4$ و $6$ --- الحسابي $\frac{4+6}{2}$،
        والهندسي $\sqrt{4 \times 6}$، والتوافقي
        $\frac{2 \times 4 \times 6}{4 + 6}$ --- ورتّبها. ولكلٍّ
        منها موضع في شبه المنحرف: فالمتوسط الحسابي هو الخط
        المتوسّط الواصل بين منتصفَي الساقين
        (\cref{pb:g7:areas:1})، \omterm{pb:g8:speed:1}{والمتوسط التوافقي} هو قطعتك
        $EF$، \omterm{pb:g8:cosine:1}{والمتوسط الهندسي} هو \omterm{def:g6:lines:perp}{الموازي} الذي يقطع شبه
        المنحرف إلى شبهَي منحرف \emph{متشابهين} (تحقّق من قيمته
        بالآلة الحاسبة؛ والبرهان تحدٍّ إضافي جميل). وأين بُرهن
        على المتباينة بين الثلاثة في هذا الكتاب
        (\cref{pb:g8:cosine:1}، و\cref{pb:g8:speed:1})؟
\end{enumerate}
\end{problem}
