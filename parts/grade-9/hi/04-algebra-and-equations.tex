\chapter{बीजगणित और समीकरण}\label{ch:g9:algebra}

अक्षर संख्याओं की जगह खड़े होते हैं, और अक्षरों के साथ गणना करने से
\emph{सारी} संख्याओं के बारे में एक ही बार में बात सिद्ध हो जाती है।
यह अध्याय \omterm{def:g9:algebra:expand}{फैलाना} और गुणनखंडन करना पक्का करता है --- उन तीन
सर्वसमिकाओं समेत जो पूरे गणित में तुम्हारे साथ चलेंगी --- और पहली \omterm{def:g9:fractions:power}{घात}
के \omterm{def:g8:equations:def}{समीकरण}, \omterm{def:g2:mult:def}{गुणनफल} वाले \omterm{def:g8:equations:def}{समीकरण} तथा आसान असमिकाएँ हल करता है, हमेशा एक
बार में एक कारण सहित क़दम।

\section{फैलाना}

\begin{definition}[फैलाना]\label{def:g9:algebra:expand}
\emph{फैलाना}\index{फैलाना} यानी वितरण नियम की मदद से किसी \omterm{def:g2:mult:def}{गुणनफल} को
\omterm{def:g1:addition:def}{योगफल} में बदल देना:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g9:algebra:expand}
पहले कोष्ठक का हर पद दूसरे कोष्ठक के हर पद से गुणा होता है:
\begin{align*}
(x + 3)(2x + 5)
&= x \times 2x + x \times 5 + 3 \times 2x + 3 \times 5 \\
&= 2x^2 + 5x + 6x + 15 \\
&= 2x^2 + 11x + 15 .
\end{align*}
चिह्न पदों के साथ-साथ चलते हैं:
\begin{align*}
(3x - 2)(x - 4)
&= 3x^2 - 12x - 2x + 8
= 3x^2 - 14x + 8 .
\end{align*}
\end{example}

\begin{theorem}[तीन उल्लेखनीय सर्वसमिकाएँ]\label{thm:g9:algebra:identities}
सभी संख्याओं $a$ और $b$ के लिए:
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
हर बाएँ पक्ष को फैलाओ। पहले के लिए:
$(a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2$। दूसरे के लिए,
पहले में $b$ की जगह $-b$ रख दो। तीसरे के लिए:
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$ (आड़े पद कट जाते हैं)।
\end{proof}

\begin{example}\label{ex:g9:algebra:identities}
\[
(x + 5)^2 = x^2 + 10x + 25, \qquad
(3x - 1)^2 = 9x^2 - 6x + 1, \qquad
(2x + 7)(2x - 7) = 4x^2 - 49 .
\]
मन में गणना: $102^2 = (100 + 2)^2 = 10000 + 400 + 4 = 10404$, और
$98 \times 102 = (100-2)(100+2) = 10000 - 4 = 9996$।
\end{example}

\section{गुणनखंडन}

\begin{method}[गुणनखंडन]\label{met:g9:algebra:factor}
किसी व्यंजक का गुणनखंडन करने के लिए क्रम से ये आज़माओ:
\begin{enumerate}
  \item \emph{साझा गुणनखंड}: $ka + kb = k(a + b)$, जहाँ $k$ ख़ुद कोई
        कोष्ठक भी हो सकता है;
  \item \emph{वर्गों का \omterm{ex:g1:subtraction:difference}{अंतर}}: $a^2 - b^2 = (a+b)(a-b)$;
  \item \emph{पूर्ण वर्ग}: $a^2 + 2ab + b^2 = (a+b)^2$ (और $-2ab$ के
        साथ, $(a-b)^2$)।
\end{enumerate}
नतीजे को फैलाकर हमेशा जाँच लो।
\end{method}

\begin{example}\label{ex:g9:algebra:factor}
\emph{साझा गुणनखंड:}
$15x^2 + 10x = 5x \times 3x + 5x \times 2 = 5x(3x + 2)$।

\emph{साझा कोष्ठक:}
\[
(x+2)(x-1) + (x+2)(3x+4) = (x+2)\bigl[(x-1) + (3x+4)\bigr]
= (x+2)(4x+3).
\]

\emph{वर्गों का \omterm{ex:g1:subtraction:difference}{अंतर}:}
$x^2 - 81 = (x+9)(x-9)$, और
\[
(2x+1)^2 - 9 = (2x+1)^2 - 3^2 = (2x+1+3)(2x+1-3) = (2x+4)(2x-2).
\]
हर गुणनखंड में अब भी एक साझा गुणनखंड बचा है: $(2x+4)(2x-2) = 2(x+2)
\times 2(x-1) = 4(x+2)(x-1)$।
\end{example}

\section{समीकरण}

\begin{method}[पहली घात के समीकरण]\label{met:g9:algebra:firstdegree}
$5x - 3 = 2x + 9$ जैसा \omterm{def:g8:equations:def}{समीकरण} हल करने के लिए:
\begin{enumerate}
  \item दोनों तरफ़ एक ही राशि जोड़कर $x$ वाले पद एक तरफ़ और संख्याएँ
        दूसरी तरफ़ इकट्ठी करो;
  \item हर तरफ़ को सरल करो;
  \item दोनों तरफ़ में $x$ के गुणांक का भाग दो;
  \item हल को असली \omterm{def:g8:equations:def}{समीकरण} में रखकर जाँचो।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:algebra:firstdegree}
\begin{align*}
5x - 3 &= 2x + 9 \\
5x - 2x &= 9 + 3 && \text{(दोनों तरफ़ $3 - 2x$ जोड़ो)}\\
3x &= 12 \\
x &= 4 && \text{($3$ से भाग दो)।}
\end{align*}
जाँच: $5 \times 4 - 3 = 17$ और $2 \times 4 + 9 = 17$। हल $4$ है।

कोष्ठक हों तो पहले फैलाओ: $3(x - 2) = x + 8$ बन जाता है
$3x - 6 = x + 8$, इसलिए $2x = 14$ और $x = 7$।
\end{example}

\begin{theorem}[शून्य-गुणनफल नियम]\label{thm:g9:algebra:zeroproduct}
कोई \omterm{def:g2:mult:def}{गुणनफल} \omterm{def:g1:counting:zero}{शून्य} तभी होता है जब उसका कम से कम एक गुणनखंड \omterm{def:g1:counting:zero}{शून्य} हो:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ या } \ B = 0 .
\]
\end{theorem}

\begin{proof}
अगर कोई गुणनखंड \omterm{def:g1:counting:zero}{शून्य} हो तो \omterm{def:g2:mult:def}{गुणनफल} साफ़ तौर पर \omterm{def:g1:counting:zero}{शून्य} है। और उलटा: अगर
$A B = 0$ हो और $A \neq 0$ हो, तो दोनों तरफ़ $A$ का भाग देने पर
$B = 0$ मिलता है।
\end{proof}

\begin{example}\label{ex:g9:algebra:zeroproduct}
$(x - 3)(2x + 10) = 0$ हल करो: या तो $x - 3 = 0$ या $2x + 10 = 0$,
इसलिए हल $3$ और $-5$ हैं।

$x^2 - 49 = 0$ हल करो: पहले गुणनखंड करो, $(x-7)(x+7) = 0$, हल $7$ और
$-7$।

$x^2 = 3x$ हल करो: $x$ का भाग कभी मत देना! पद इधर लाओ और गुणनखंड करो:
$x^2 - 3x = x(x - 3) = 0$, हल $0$ और $3$।
\end{example}

\section{असमिकाएँ}

\begin{proposition}[असमिकाओं के नियम]\label{prop:g9:algebra:ineq}
दोनों तरफ़ एक ही संख्या जोड़ने पर असमिका बनी रहती है, और दोनों तरफ़ एक
ही \emph{धन} संख्या से गुणा करने पर भी। दोनों तरफ़ किसी \emph{ऋण}
संख्या से गुणा करने पर असमिका का चिह्न उलट जाता है।
\end{proposition}

\begin{proof}
अगर $a \leq b$ हो तो $b - a \geq 0$। $c$ जोड़ने पर:
$(b+c) - (a+c) = b - a \geq 0$, इसलिए $a + c \leq b + c$। $c > 0$ से
गुणा करने पर: $bc - ac = (b-a)c \geq 0$, इसलिए $ac \leq bc$। और
$c < 0$ से गुणा करने पर: $(b-a)c \leq 0$, इसलिए $ac \geq bc$: क्रम
उलट गया।
\end{proof}

\begin{example}\label{ex:g9:algebra:ineq}
$7 - 2x < 15$ हल करो:
\begin{align*}
7 - 2x &< 15 \\
-2x &< 8 && \text{($7$ घटाओ)} \\
x &> -4 && \text{($-2$ से भाग दो: चिह्न उलट दो)।}
\end{align*}
हल $-4$ से बड़ी सारी संख्याएँ हैं: संख्या रेखा पर $-4$ पर एक खोखला
बिंदु, और उसके दाईं ओर रंगाई।
\end{example}

मेरी संख्या क्या है? (\omterm{def:g8:equations:def}{समीकरण} बनाओ और हल करो।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:algebra:10}
मान लो $n$ कोई पूर्ण संख्या है।
\begin{enumerate}
  \item $(n + 1)(n - 1)$ फैलाओ और इससे नतीजा निकालो कि $n$ के दोनों

        $24 \times 26 = 25^2 - 1 = 624$)।
  \item दिखाओ कि लगातार आने वाले तीन पूर्णांकों का \omterm{def:g1:addition:def}{योगफल} हमेशा $3$
        का \omterm{def:g9:arith:divisor}{गुणज} होता है।
\end{enumerate}
\end{exercise}

\section{समस्या: अंकों का बीजगणित}

\begin{problem}[{सप्ताहांत समस्या --- विभाज्यता की तरकीबें काम क्यों
करती हैं, 1089 वाले खेल का पर्दाफ़ाश, और जादू जैसी लगने वाली मन की
गणना}]
\label{pb:g9:algebra:1}
शुरुआती कक्षाओं ने तुम्हें \emph{नियम} सिखाए (कोई संख्या $9$ से
\omterm{def:g6:wholes:divisible}{विभाज्य} होती है जब उसके अंकों का \omterm{def:g1:addition:def}{योगफल} \omterm{def:g6:wholes:divisible}{विभाज्य} हो); जादूगर
\emph{खेल} दिखाते हैं (सबका आख़िर $1089$ पर होता है); बाज़ार
\emph{छोटे रास्ते} जानता है ($35^2 = 1225$, पलक झपकते)। इन सबके पीछे
एक ही राज़ है: अंक $a$, $b$, $c$ वाली संख्या दरअसल व्यंजक
$100a + 10b + c$ है, और इस अध्याय का बीजगणित
(\cref{thm:g9:algebra:identities}, \omterm{def:g9:algebra:expand}{फैलाना}, गुणनखंडन) उसे खोलकर रख
सकता है। इस समस्या के आख़िर तक तुम नियम सिद्ध कर चुके होगे, खेल का
पर्दा उठा चुके होगे, और जादू सीख चुके होगे।

\medskip
\noindent\textbf{भाग I --- अंक सूक्ष्मदर्शी के नीचे।}
\begin{enumerate}
  \item दहाई का अंक $a$ और इकाई का अंक $b$ वाली दो अंकों की संख्या
        $10a + b$ होती है। संख्या और उसके उलटे रूप का \omterm{ex:g1:subtraction:difference}{अंतर}
        $(10a + b) - (10b + a)$ फैलाओ और उसका गुणनखंड करो, फिर जो
        मिला वह बताओ। $72 - 27$ पर जाँचो।
  \item तीन अंकों की संख्याओं के लिए नौ का नियम सिद्ध करो:
        सर्वसमिका
        \[
        100a + 10b + c = 9\,(11a + b) + (a + b + c)
        \]
        जाँचो, और समझाओ कि इससे क्यों निकलता है कि $9$ का भाग देने
        पर संख्या और उसके अंकों का \omterm{def:g1:addition:def}{योगफल} \emph{एक ही \omterm{def:g3:division:remainder}{शेषफल}} छोड़ते
        हैं --- यानी एक $9$ से \omterm{def:g6:wholes:divisible}{विभाज्य} ठीक तभी है जब दूसरा हो
        (\cref{prop:g9:arith:rules}, अब सिद्ध)।
  \item उसी सर्वसमिका से तीन का नियम एक वाक्य में निकालो।
  \item ``नौ फेंकना'', हिसाब-किताब वालों की जाँच: $47 \times 86 =
        4\,042$ जाँचने के लिए हर संख्या की जगह उसके अंकों के \omterm{def:g1:addition:def}{योगफल}
        का अंकों वाला \omterm{def:g1:addition:def}{योगफल} रखो: $47 \to 11 \to 2$,
        $86 \to 14 \to 5$, और $2 \times 5 = 10 \to 1$; फिर
        $4\,042 \to 10 \to 1$: मेल खा गया। समझाओ कि सही \omterm{def:g2:mult:def}{गुणनफल} को
        यह जाँच क्यों पार करनी ही पड़ती है ($47 = 9k + 2$,
        $86 = 9l + 5$ लिखकर फैलाओ), और कोई ऐसा ग़लत नतीजा ढूँढ़ो
        जिसे यह जाँच पकड़ नहीं पाती (आगे-पीछे हुए अंक उसे क्यों
        नहीं दिखते?)।
  \item चार अंकों की संख्याओं के लिए ग्यारह का नियम सिद्ध करो:
        \[
        1000a + 100b + 10c + d
        = 11\,(91a + 9b + c) + (- a + b - c + d),
        \]
        जाँचो, और उससे तय करो कि $2\,926$, $11$ से \omterm{def:g6:wholes:divisible}{विभाज्य} है या
        नहीं।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- 1089 वाला खेल।}
खेल यह है: तीन अंकों की कोई संख्या सोचो जिसके पहले और आख़िरी अंक में
कम से कम $2$ का फ़र्क़ हो; उसे उलटा लिखो; बड़ी में से छोटी घटाओ; नतीजे
को उलटा लिखो; और आख़िरी दोनों संख्याएँ जोड़ो। जादूगर ऐलान करता है:
\emph{1089}।
\begin{enumerate}[resume]
  \item यह खेल $742$ पर और अपनी पसंद की एक संख्या पर करके देखो।
  \item मान लो संख्या के अंक $a$, $b$, $c$ हैं और $a > c$ है। पहले
        घटाव $(100a + 10b + c) - (100c + 10b + a)$ का गुणनखंड करो,
        और $a - c$ के $2$ से $9$ तक चलने पर उसके सारे संभव मान
        लिखो।
  \item $99$ के इन गुणजों में से हर एक के अंकों का एक चौंकाने वाला
        ढर्रा है। $m = a - c$ लिखकर सर्वसमिका
        \[
        99m = 100\,(m - 1) + 90 + (10 - m)
        \]
        जाँचो, और $99m$ के तीनों अंक पढ़ लो।
  \item सवाल 8 वाले अंकों से $99m$ को उसके अपने उलटे रूप में जोड़ो,
        और सिद्ध करो कि जोड़ $1089$ ही आता है --- $m$ चाहे कुछ भी
        हो। खेल ख़त्म; बीजगणित ज़िंदाबाद।
  \item जादूगर ने पहले और आख़िरी अंक में कम से कम $2$ का फ़र्क़ क्यों
        माँगा? $a = c$ और $a - c = 1$ वाली हालतें देखो (जहाँ घटाने
        पर $99$ आता है, और उलटा करने के लिए उसे $099$ लिखना पड़ता
        है), और वह बारीक शर्त लिखो जो खेल को ईमानदार रखती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- रोज़ के काम का जादू।}
\begin{enumerate}[resume]
  \item जन्मदिन वाला खेल: ``अपने जन्म के महीने को $5$ से गुणा करो,
        $7$ जोड़ो, दुगना करो, जन्म की तारीख़ जोड़ो, $14$ घटाओ।''
        दिखाओ कि नतीजा हमेशा $10m + d$ होता है --- यानी महीना और
        तारीख़, एक नज़र में पढ़ने लायक़। (किसी पर आज़माकर देखो।)
  \item अब तुम खेल को यूँ बदलते हो: ``महीने को $5$ से गुणा करो, $9$
        जोड़ो, दुगना करो, तारीख़ जोड़ो।'' वही महीना--तारीख़ पढ़ने के
        लिए दर्शक को आख़िर में अब कौन-सी संख्या घटानी पड़ेगी?
  \item ग़ायब अंक वाला खेल: दर्शक कोई भी संख्या चुनता है, उसमें से
        उसके अंकों का \omterm{def:g1:addition:def}{योगफल} घटाता है (सवाल 2 कहता है कि नतीजा हमेशा
        $9$ का \omterm{def:g9:arith:divisor}{गुणज} होता है --- तीन अंकों की संख्या के लिए इसका
        कारण बताओ), फिर एक \emph{शून्येतर} अंक काट देता है और बाक़ी
        अंक तुम्हें पढ़कर सुना देता है। समझाओ कि तुम काटा हुआ अंक
        कैसे वापस निकाल लेते हो, और $0$ को बाहर क्यों रखना पड़ा।
  \item बाज़ार का जादू: सर्वसमिका
        $(10t + 5)^2 = 100\,t(t+1) + 25$ सिद्ध करो, $5$ पर ख़त्म
        होने वाली किसी भी संख्या का वर्ग निकालने का नियम लिखो, और
        $45^2$, $85^2$ तथा $105^2$ मन में निकालो।
  \item समापन, \cref{exo:g9:algebra:10} को आगे बढ़ाते हुए: सिद्ध
        करो कि लगातार आने वाले पाँच पूर्णांकों का \omterm{def:g1:addition:def}{योगफल} हमेशा $5$
        का \omterm{def:g9:arith:divisor}{गुणज} होता है, पर लगातार आने वाले \emph{चार} पूर्णांकों
        का \omterm{def:g1:addition:def}{योगफल} $4$ का \omterm{def:g9:arith:divisor}{गुणज} \emph{कभी नहीं} होता। (दो छोटी गणनाएँ
        --- और एक सीख कि बीजगणित ``हमेशा'' तथा ``कभी नहीं'' शब्दों
        के साथ क्या कर देता है।)

\end{enumerate}
\end{problem}
