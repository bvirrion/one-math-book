\chapter{ठोस और आयतन}\label{ch:g9:solids}

त्रिविमीय ज्यामिति \omterm{def:g5:solids:def}{ठोसों} के एक छोटे-से परिवार से शुरू होती है ---
\omterm{def:g7:areas:prism}{प्रिज़्म}, \omterm{def:g7:areas:prism}{बेलन}, \omterm{def:g8:solids:def}{पिरामिड}, \omterm{def:g8:solids:def}{शंकु}, गोले --- और दो सवालों से: उनमें कितना
समाता है (\omterm{def:g6:measure:volume}{आयतन}), और उन्हें किसी तल से काटने पर क्या दिखता है (काटें)?
अध्याय बड़े काम की एक बात पर ख़त्म होता है: किसी \omterm{def:g5:solids:def}{ठोस} को $k$ से
घटाने-बढ़ाने पर उसका \omterm{def:g6:measure:volume}{आयतन} $k^3$ गुना हो जाता है।

\section{चिरपरिचित ठोस और उनके आयतन}

\begin{theorem}[आयतन के सूत्र]\label{thm:g9:solids:volumes}
आधार का \omterm{def:g6:measure:area}{क्षेत्रफल} $B$ लिखो और ऊँचाई $h$ (यानी आधार तथा सामने वाले \omterm{def:g5:solids:def}{फलक}
या शिखर के बीच की दूरी)।
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{l|l}
\omterm{def:g5:solids:def}{ठोस} & \omterm{def:g6:measure:volume}{आयतन} \\
\hline
\omterm{def:g7:areas:prism}{प्रिज़्म}, \omterm{def:g7:areas:prism}{बेलन} & $V = B \times h$ \\[5pt]
\omterm{def:g8:solids:def}{पिरामिड}, \omterm{def:g8:solids:def}{शंकु} & $V = \dfrac13\, B \times h$ \\[5pt]
$r$ \omterm{def:g3:shapes:shapes}{त्रिज्या} वाला गोला & $V = \dfrac43\,\pi r^3$ \\
\end{tabular}
\end{center}
$r$ \omterm{def:g3:shapes:shapes}{त्रिज्या} वाले \omterm{def:g7:areas:prism}{बेलन} और \omterm{def:g8:solids:def}{शंकु} के लिए आधार का \omterm{def:g6:measure:area}{क्षेत्रफल}
$B = \pi r^2$ है; और गोले की सतह का \omterm{def:g6:measure:area}{क्षेत्रफल} $4\pi r^2$।
\end{theorem}
\admitted

\begin{example}\label{ex:g9:solids:volumes}
\omterm{def:g7:areas:prism}{बेलन} के आकार वाले किसी डिब्बे की \omterm{def:g3:shapes:shapes}{त्रिज्या} $4$ cm और ऊँचाई $10$ cm है:
\[
V = \pi r^2 h = \pi \times 16 \times 10 = 160\pi \approx 503 \text{ cm}^3
\approx 0.5 \text{ L}.
\]
उसी आधार और उसी ऊँचाई वाला \omterm{def:g8:solids:def}{शंकु} इसका एक तिहाई समाता है:
$\frac{160\pi}{3} \approx 168$ cm$^3$। और $4$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} वाला गोला:
$V = \frac43 \pi \times 64 = \frac{256\pi}{3} \approx 268$ cm$^3$।
\end{example}

\begin{example}[क़दम दर क़दम एक पिरामिड]\label{ex:g9:solids:pyramid}
किसी \omterm{def:g8:solids:def}{पिरामिड} का आधार $6$ m भुजा वाला वर्ग है और ऊँचाई $5$ m।
\begin{enumerate}
  \item आधार का \omterm{def:g6:measure:area}{क्षेत्रफल}: $B = 6^2 = 36$ m$^2$।
  \item \omterm{def:g6:measure:volume}{आयतन}: $V = \frac13 \times 36 \times 5 = 60$ m$^3$।
\end{enumerate}
मात्रकों का ध्यान रखो: अगर भुजा cm में और ऊँचाई m में दी हो, तो पहले
किसी एक को बदलना पड़ेगा।
\end{example}

\section{तलों से बनी काटें}

\begin{proposition}[चिरपरिचित ठोसों की काटें]\label{prop:g9:solids:sections}
किसी \omterm{def:g5:solids:def}{ठोस} को किसी तल से काटने पर एक चपटी आकृति मिलती है, उसकी
\emph{अनुप्रस्थ काट}\index{अनुप्रस्थ काट}:
\begin{itemize}
  \item \omterm{def:g7:areas:prism}{प्रिज़्म} या \omterm{def:g7:areas:prism}{बेलन} को \emph{उसके आधार के \omterm{def:g6:lines:perp}{समांतर}} काटने पर हर
        ऊँचाई पर आधार की ही एक नक़ल मिलती है;
  \item \omterm{def:g8:solids:def}{पिरामिड} या \omterm{def:g8:solids:def}{शंकु} को उसके आधार के \omterm{def:g6:lines:perp}{समांतर} काटने पर आधार का एक
        \emph{घटा हुआ रूप} मिलता है: शिखर से $d$ दूरी पर \omterm{def:g9:thales:scaling}{मापक गुणक}
        $k = \frac{d}{h}$ होता है;
  \item $r$ \omterm{def:g3:shapes:shapes}{त्रिज्या} वाले गोले को उसके केंद्र से $d < r$ दूरी पर
        किसी तल से काटने पर $\sqrt{r^2 - d^2}$ \omterm{def:g3:shapes:shapes}{त्रिज्या} वाला वृत्त
        मिलता है (पाइथागोरस प्रमेय से)।
\end{itemize}
\end{proposition}
\admitted

शिखर से $3$ दूरी पर लिए गए तल की काट एक चक्र है जिसका \omterm{def:g9:thales:scaling}{मापक गुणक}
$k = \frac39 = \frac13$ है: यानी उसकी \omterm{def:g3:shapes:shapes}{त्रिज्या} $6 \times \frac13 = 2$।

$5$ \omterm{def:g3:shapes:shapes}{त्रिज्या} वाले गोले को उसके केंद्र से $3$ दूरी पर किसी तल से काटा
जाता है: काट $\sqrt{25 - 9} = 4$ \omterm{def:g3:shapes:shapes}{त्रिज्या} वाला वृत्त है।
\end{example}

\section{ठोसों को घटाना-बढ़ाना}

\begin{theorem}[लंबाई, क्षेत्रफल और आयतन पर मापक गुणक का असर]\label{thm:g9:solids:scaling}
किसी \omterm{def:g5:solids:def}{ठोस} को \omterm{def:g9:thales:scaling}{मापक गुणक} $k > 0$ से बढ़ाने या घटाने पर:
\begin{itemize}
  \item हर लंबाई $k$ से गुणा हो जाती है;
  \item हर \omterm{def:g6:measure:area}{क्षेत्रफल} $k^2$ से;
  \item हर \omterm{def:g6:measure:volume}{आयतन} $k^3$ से।
\end{itemize}
\end{theorem}
\admitted

\begin{example}\label{ex:g9:solids:scaling}
$\frac{1}{18}$ \omterm{def:g7:prop:scale}{मापनी} वाली कोई नमूना कार: उसकी लंबाइयाँ असली कार की
लंबाइयों को $k = \frac{1}{18}$ से गुणा करके मिलती हैं, रंगी जाने वाली
सतह $k^2 = \frac{1}{324}$ से, और \omterm{def:g6:measure:volume}{आयतन} $k^3 = \frac{1}{5832}$ से।

किसी गोले की \omterm{def:g3:shapes:shapes}{त्रिज्या} दुगनी करने ($k = 2$) पर उसका \omterm{def:g6:measure:volume}{आयतन} $2^3 = 8$
गुना हो जाता है: सूत्र पर जाँच लो,
$\frac43\pi (2r)^3 = \frac43 \pi \times 8r^3$।
\end{example}

\begin{example}[छिन्नक शंकु]\label{ex:g9:solids:truncated}
ऊँचाई $9$ और आधार-त्रिज्या $6$ वाले किसी \omterm{def:g8:solids:def}{शंकु} (\omterm{def:g6:measure:volume}{आयतन}
$\frac13 \pi \times 36 \times 9 = 108\pi$) को शिखर से $3$ दूरी पर
काटकर ऊपर वाला छोटा \omterm{def:g8:solids:def}{शंकु} हटा दिया जाता है। वह छोटा \omterm{def:g8:solids:def}{शंकु}
$k = \frac13$ गुणक वाला घटा हुआ रूप है, इसलिए उसका \omterm{def:g6:measure:volume}{आयतन}
$108\pi \times \left(\frac13\right)^3 = 4\pi$ है। बचे हुए \omterm{def:g5:solids:def}{ठोस} (यानी
एक \emph{छिन्नक \omterm{def:g8:solids:def}{शंकु}}) का \omterm{def:g6:measure:volume}{आयतन} $108\pi - 4\pi = 104\pi \approx 327$।
\end{example}

\begin{exercise}[$\star\star\star$]\label{exo:g9:solids:9}
$6$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} और $12$ cm ऊँचाई वाली \omterm{def:g8:solids:def}{शंकु} के आकार की कोई क़ीप, शिखर
नीचे की ओर, अपनी \emph{ऊँचाई} के आधे तक पानी से भरी है (ऊँचाई शिखर
से नापी गई है)।
\begin{enumerate}
  \item क़ीप के \omterm{def:g6:measure:volume}{आयतन} का कितना \omterm{def:g9:fractions:fraction}{भिन्न} भरा हुआ है?
  \item और अगर उसमें उसके \emph{\omterm{def:g6:measure:volume}{आयतन}} का \omterm{def:g3:division:half}{आधा} पानी भरा जाए, तो
        दिखाओ कि पानी की ऊँचाई $d$ शर्त
        $\left(\frac{d}{12}\right)^3 = \frac12$ मानती है, और $d$
        मिलीमीटर तक बताओ ($\sqrt[3]{0.5} \approx 0.7937$)।
\end{enumerate}
\end{exercise}

\section{समस्या: आर्किमिडीज़ की समाधि-शिला}

\begin{problem}[{सप्ताहांत समस्या --- गोला अपने बेलन का दो तिहाई है
(दो-दो बार), $1:2:3$ की तिकड़ी, और वह वर्ग--घन नियम जो दानवों को
मना कर देता है}]
\label{pb:g9:solids:1}
आर्किमिडीज़ ने बहुत-सी प्रमेय सिद्ध कीं, पर एक पर उन्हें इतना गर्व
था कि उन्होंने उसका \emph{चित्र} अपनी समाधि पर खुदवाने को कहा: अपने
सबसे कसे हुए \omterm{def:g7:areas:prism}{बेलन} के भीतर बैठा एक गोला। सौ साल बाद रोमन लेखक सिसरो
ने सिरैक्यूज़ के पास झाड़ियों में ढूँढ़ते हुए उस समाधि को ``गोले और
\omterm{def:g7:areas:prism}{बेलन} से'' पहचान लिया। यह समस्या उस खुदाई में छिपी बात --- दो तिहाई
का दोहरा चमत्कार --- फिर से निकालती है, और फिर आयतनों तथा सतहों
(\cref{thm:g9:solids:volumes}, \cref{thm:g9:solids:scaling}) के
सहारे उस नियम तक पहुँचती है जो दानवों, चींटियों और ठंडे पड़ते
ग्रहों, सब पर राज करता है।

\medskip
\noindent\textbf{भाग I --- खुदाई का मतलब।}
$r$ \omterm{def:g3:shapes:shapes}{त्रिज्या} वाला कोई गोला किसी \omterm{def:g7:areas:prism}{बेलन} के भीतर ठीक-ठीक बैठा है: वही
\omterm{def:g3:shapes:shapes}{त्रिज्या}, और ऊँचाई $2r$।
\begin{enumerate}
  \item \omterm{def:g7:areas:prism}{बेलन} का \omterm{def:g6:measure:volume}{आयतन} $r$ की मदद से निकालो।
  \item गोले के \omterm{def:g6:measure:volume}{आयतन} और \omterm{def:g7:areas:prism}{बेलन} के \omterm{def:g6:measure:volume}{आयतन} का अनुपात निकालो। आर्किमिडीज़
        को यह बात क्यों भाई होगी कि उत्तर में न $\pi$ है, न $r$?
  \item अब सतहें: गोले की सतह के \omterm{def:g6:measure:area}{क्षेत्रफल} की तुलना \omterm{def:g7:areas:prism}{बेलन} की
        \emph{पूरी} सतह से करो (बग़ली हिस्सा और दोनों ढक्कन)। कौन-सा
        अनुपात फिर से सामने आता है?
  \item गोले और \omterm{def:g7:areas:prism}{बेलन} के बीच की ख़ाली जगह का \omterm{def:g6:measure:volume}{आयतन}
        $2\pi r^3 - \frac43\pi r^3$ है। दिखाओ कि यह बचत ठीक $r$
        \omterm{def:g3:shapes:shapes}{त्रिज्या} और $r$ ऊँचाई वाले \emph{दो शंकुओं} के \omterm{def:g6:measure:volume}{आयतन} के
        बराबर है।
  \item $12$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} वाली कोई बास्केटबॉल सबसे कसे हुए बेलनाकार
        डिब्बे में बिकती है। दोनों \omterm{def:g6:measure:volume}{आयतन} निकालो (नज़दीकी
        $100$ cm$^3$ तक) और बताओ कि डिब्बे का कितने प्रतिशत हिस्सा
        ख़ाली रहता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- कटोरे, साँचे और चाँद।}
\begin{enumerate}[resume]
  \item किसी अर्धगोलाकार कटोरे की \omterm{def:g3:shapes:shapes}{त्रिज्या} $10$ cm है। उसकी \omterm{def:g2:measure:units}{धारिता}
        निकालो, cm$^3$ में और लीटर में (डेसीलीटर तक)।
  \item $1:2:3$ की तिकड़ी: एक \omterm{def:g8:solids:def}{शंकु}, एक अर्धगोला और एक \omterm{def:g7:areas:prism}{बेलन}, तीनों
        की \omterm{def:g3:shapes:shapes}{त्रिज्या} $10$ cm और ऊँचाई $10$ cm। तीनों \omterm{def:g6:measure:volume}{आयतन} निकालो और
        उस मशहूर \omterm{def:g7:prop:table}{समानुपात} $1 : 2 : 3$ की पुष्टि करो। (यह भी
        आर्किमिडीज़ को पसंद आता।)
  \item $3$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} वाला चॉकलेट का गोला पिघलाकर $3$ cm
        \omterm{def:g3:shapes:shapes}{त्रिज्या} वाले बेलनाकार साँचे में डाला जाता है। चॉकलेट किस
        ऊँचाई तक पहुँचेगी? (पहले बिलकुल सही \omterm{def:g9:fractions:fraction}{भिन्न}, फिर मिलीमीटर
        तक।)
  \item धरती की \omterm{def:g3:shapes:shapes}{त्रिज्या} क़रीब $6\,371$ km है और चाँद की क़रीब
        $1\,737$ km। \omterm{def:g3:shapes:shapes}{त्रिज्याओं} का अनुपात निकालो, फिर
        --- \cref{thm:g9:solids:scaling} की मदद से --- आयतनों का:
        \omterm{def:g6:measure:volume}{आयतन} के हिसाब से कितने चाँद एक खोखली धरती में समा जाएँगे?
  \item उसी प्रमेय से: धरती और चाँद की \emph{सतहों के \omterm{def:g6:measure:area}{क्षेत्रफल}}
        का अनुपात क्या है? और आम तौर पर, किसी गुब्बारे की \omterm{def:g3:shapes:shapes}{त्रिज्या}
        दुगनी होने पर उसके \omterm{def:g6:measure:volume}{आयतन} तथा उसकी सतह का क्या होता है?
        \emph{वर्ग--घन नियम} बताओ: कोई आकार जैसे-जैसे बड़ा होता
        जाता है, \omterm{def:g6:measure:volume}{आयतन} सतहों से आगे निकल जाते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- वर्ग--घन नियम दुनिया चलाता है।}
\begin{enumerate}[resume]
  \item दानवों के ख़िलाफ़ गैलीलियो की दलील: किसी आदमी को उन्हीं
        अनुपातों में $10$ गुना बड़ा कर दिया गया, ऐसा सोचो। उसका
        भार कितने गुना बढ़ेगा (भार \omterm{def:g6:measure:volume}{आयतन} के साथ चलता है)? हड्डियों
        की \omterm{prop:g9:solids:sections}{अनुप्रस्थ काट} कितने गुना बढ़ेगी (वह एक \omterm{def:g6:measure:area}{क्षेत्रफल} है)?
        और तब हड्डी के हर वर्ग सेंटीमीटर पर पड़ने वाला \emph{दाब}
        कितने गुना बढ़ेगा --- और उस दानव का क्या होगा?
  \item यही नियम उलटा चलकर चींटियों के करतब समझा देता है: ताक़त
        माँसपेशी की \omterm{prop:g9:solids:sections}{अनुप्रस्थ काट} (एक \omterm{def:g6:measure:area}{क्षेत्रफल}) के साथ चलती है,
        और भार \omterm{def:g6:measure:volume}{आयतन} के साथ। हर दिशा में $100$ गुना छोटे किसी जीव
        के लिए भार और ताक़त कितने-कितने गुना घटते हैं, और
        ताक़त-और-भार का अनुपात कितने गुना सुधर जाता है?
  \item ऊष्मा शरीर के \omterm{def:g6:measure:volume}{आयतन} से बनती है और उसकी सतह से बाहर जाती है।
        दिखाओ कि किसी गोले के लिए सतह-और-आयतन का अनुपात
        $\frac{3}{r}$ होता है, और उसे $r = 1$ तथा $r = 2$ पर
        निकालो। चूहा जल्दी ठंडा पड़ता है या भालू --- और सर्दियों
        में नन्हे बच्चों को टोपी क्यों चाहिए?
  \item $4$ cm और $5$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} वाले दो गोल संतरे एक ही दाम पर
        बिक रहे हैं। उनके आयतनों का अनुपात निकालो: उतने ही पैसों
        में बड़ा संतरा कितना ज़्यादा संतरा देता है?
  \item समापन: ठीक-ठीक बताओ कि समाधि वाला चित्र किस बात को अपने
        भीतर समेटे है --- सवाल 2 और 3 वाले दोनों ``दो तिहाई'' ---
        और सिसरो को एक वाक्य में जवाब दो: कोई गणितज्ञ अपनी
        इंजीनियरी प्रतिभा की हर विजय को छोड़कर \omterm{def:g7:areas:prism}{बेलन} में बैठा एक
        गोला ही क्यों चुनेगा?

\end{enumerate}
\end{problem}
