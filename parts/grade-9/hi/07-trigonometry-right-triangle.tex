\chapter{समकोण त्रिभुज में त्रिकोणमिति}\label{ch:g9:trig}

किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} में एक न्यून कोण जान लेने भर से उसकी \emph{शक्ल}
तय हो जाती है --- सिर्फ़ नाप बदल सकती है। इसलिए उसकी भुजाओं के अनुपात
भी सिर्फ़ कोण पर टिके होते हैं: यही अनुपात \omterm{def:g9:trig:ratios}{कोज्या}, \omterm{def:g9:trig:ratios}{ज्या} और \omterm{def:g9:trig:ratios}{स्पर्शज्या}
हैं। पाइथागोरस प्रमेय के साथ मिलकर ये बहुत थोड़ी-सी जानकारी से \omterm{def:g6:shapes:triangles}{समकोण
त्रिभुज} की हर भुजा और हर कोण निकाल देते हैं।

\section{पाइथागोरस प्रमेय, एक बार फिर}

\begin{theorem}[पाइथागोरस]\label{thm:g9:trig:pythagoras}
अगर त्रिभुज $ABC$ का $A$ पर \omterm{def:g6:shapes:triangles}{समकोण} हो, तो कर्ण का वर्ग दोनों \omterm{def:g6:shapes:triangles}{समकोण}
भुजाओं के वर्गों के \omterm{def:g1:addition:def}{योगफल} के बराबर होता है:
\[
BC^2 = AB^2 + AC^2 .
\]
और उलटे, अगर किसी त्रिभुज में $BC^2 = AB^2 + AC^2$ हो, तो उस त्रिभुज
का $A$ पर \omterm{def:g6:shapes:triangles}{समकोण} होता है।
\end{theorem}
\admitted

\begin{example}\label{ex:g9:trig:pythagoras}
किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} की \omterm{def:g6:shapes:triangles}{समकोण} भुजाएँ $AB = 5$ और $AC = 12$ हैं। तब
\[
BC^2 = 5^2 + 12^2 = 25 + 144 = 169,
\qquad\text{यानी}\quad BC = \sqrt{169} = 13 .
\]
कोई \emph{\omterm{def:g6:shapes:triangles}{समकोण} भुजा} निकालनी हो: अगर $BC = 10$ और $AB = 6$ हों, तो
$AC^2 = BC^2 - AB^2 = 100 - 36 = 64$, यानी $AC = 8$।
\end{example}

\section{कोज्या, ज्या, स्पर्शज्या}

किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} में, किसी न्यून कोण $\theta$ के सापेक्ष तीनों
भुजाओं के अपने-अपने नाम हैं: \emph{कर्ण} (\omterm{def:g6:shapes:triangles}{समकोण} के सामने वाली, सबसे
लंबी भुजा), $\theta$ की \emph{सम्मुख} भुजा, और $\theta$ की
\emph{संलग्न} भुजा (वह \omterm{def:g6:shapes:triangles}{समकोण} भुजा जो $\theta$ को छूती है)।

\begin{definition}[त्रिकोणमितीय अनुपात]\label{def:g9:trig:ratios}
किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} के न्यून कोण $\theta$ की \emph{कोज्या}\index{कोज्या},
\emph{ज्या}\index{ज्या} और \emph{स्पर्शज्या}\index{स्पर्शज्या} इस तरह
परिभाषित होती हैं:
\[
\cos\theta = \frac{\text{संलग्न भुजा}}{\text{कर्ण}},
\qquad
\sin\theta = \frac{\text{सम्मुख भुजा}}{\text{कर्ण}},
\qquad
\tan\theta = \frac{\text{सम्मुख भुजा}}{\text{संलग्न भुजा}} .
\]
ये अनुपात सिर्फ़ $\theta$ पर टिके हैं, त्रिभुज की नाप पर नहीं: एक ही
न्यून कोण वाले सारे \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज}, थेल्स के अनुसार, एक-दूसरे के
घटे-बढ़े रूप हैं।
\end{definition}

\begin{proposition}[पहले गुण]\label{prop:g9:trig:properties}
हर न्यून कोण $\theta$ के लिए:
\begin{enumerate}
  \item $0 < \cos\theta < 1$ और $0 < \sin\theta < 1$ (कोई भी \omterm{def:g6:shapes:triangles}{समकोण}
        भुजा कर्ण से छोटी होती है);
  \item $\cos^2\theta + \sin^2\theta = 1$;
  \item $\tan\theta = \dfrac{\sin\theta}{\cos\theta}$.
\end{enumerate}
\end{proposition}

\begin{proof}
$\theta$ कोण वाले किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} की संलग्न भुजा को $a$, सम्मुख
भुजा को $o$ और कर्ण को $h$ लिखो।

1. $0 < a < h$ और $0 < o < h$ हैं, इसलिए दोनों अनुपात $0$ और $1$ के
बीच सख़्ती से पड़ते हैं।

2. पाइथागोरस से $a^2 + o^2 = h^2$। सब कुछ $h^2$ से भाग दो:
\[
\cos^2\theta + \sin^2\theta
= \frac{a^2}{h^2} + \frac{o^2}{h^2}
= \frac{a^2 + o^2}{h^2} = 1 .
\]

3. $\dfrac{\sin\theta}{\cos\theta} = \dfrac{o/h}{a/h} = \dfrac oa
= \tan\theta$।
\end{proof}

\begin{method}[कोई भुजा निकालना]\label{met:g9:trig:side}
एक भुजा और एक न्यून कोण ज्ञात हों, तब कोई अज्ञात भुजा निकालने के लिए:
\begin{enumerate}
  \item ज्ञात कोण पर निशान लगाओ और तीनों भुजाओं के नाम \emph{उसी कोण
        से} लिखो (कर्ण, सम्मुख, संलग्न);
  \item वह अनुपात चुनो (\omterm{def:g9:trig:ratios}{कोज्या}, \omterm{def:g9:trig:ratios}{ज्या} या \omterm{def:g9:trig:ratios}{स्पर्शज्या}) जिसमें ज्ञात
        भुजा और चाही गई भुजा, दोनों आती हों;
  \item \omterm{def:g8:equations:def}{समीकरण} लिखकर उसे हल करो;
  \item समझदारी की जाँच: कर्ण सबसे लंबा निकलना चाहिए।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:trig:side}
किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} का कर्ण $8$ नापता है और एक कोण $35^\circ$ का है;
उसकी सम्मुख भुजा निकालो। सम्मुख भुजा और कर्ण को जोड़ने वाला अनुपात
\omterm{def:g9:trig:ratios}{ज्या} है:
\[
\sin 35^\circ = \frac{\text{सम्मुख भुजा}}{8}
\quad\Longrightarrow\quad
\text{सम्मुख भुजा} = 8 \sin 35^\circ \approx 8 \times 0.574 \approx 4.6 .
\]
संलग्न भुजा के लिए \omterm{def:g9:trig:ratios}{कोज्या} लो:
$8\cos 35^\circ \approx 8 \times 0.819 \approx 6.6$। पाइथागोरस से
जाँच: $4.6^2 + 6.6^2 \approx 21 + 43 \approx 64 = 8^2$।
\end{example}

\begin{method}[कोई कोण निकालना]\label{met:g9:trig:angle}
जब दो भुजाएँ ज्ञात हों, तो उनका बना अनुपात निकालो, पहचानो कि वह अज्ञात
कोण की \omterm{def:g9:trig:ratios}{कोज्या} है, \omterm{def:g9:trig:ratios}{ज्या} है या \omterm{def:g9:trig:ratios}{स्पर्शज्या}, और उस अनुपात पर कैलकुलेटर का
प्रतिलोम फलन ($\cos^{-1}$, $\sin^{-1}$ या $\tan^{-1}$) लगाओ।
\end{method}

\begin{example}\label{ex:g9:trig:angle}
$5$ m लंबी कोई सीढ़ी दीवार से टिकी है, और उसका पैर दीवार से $1.4$ m
दूर है। सीढ़ी तथा ज़मीन के बीच का कोण $\theta$ इस शर्त को मानता है
\[
\cos\theta = \frac{1.4}{5} = 0.28,
\qquad\text{यानी}\quad
\theta = \cos^{-1}(0.28) \approx 73.7^\circ .
\]
\end{example}

\begin{example}[ख़ास कोण]\label{ex:g9:trig:special}
दो त्रिभुज बिलकुल सही मान देते हैं। भुजा $1$ वाले वर्ग का \omterm{def:g3:division:half}{आधा} (एक
\omterm{def:g6:shapes:triangles}{समद्विबाहु} \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज}) कर्ण $\sqrt2$ और $45^\circ$ के कोण देता है:
\[
\cos 45^\circ = \sin 45^\circ = \frac{1}{\sqrt2} = \frac{\sqrt2}{2},
\qquad \tan 45^\circ = 1 .
\]
भुजा $1$ वाले \omterm{def:g6:shapes:triangles}{समबाहु त्रिभुज} का \omterm{def:g3:division:half}{आधा} $30^\circ$ और $60^\circ$ के कोण
देता है:
\[
\cos 60^\circ = \sin 30^\circ = \frac12,
\qquad
\cos 30^\circ = \sin 60^\circ = \frac{\sqrt3}{2}.
\]
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g9:trig:1}
किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} की \omterm{def:g6:shapes:triangles}{समकोण} भुजाएँ $9$ और $12$ हैं। उसका कर्ण निकालो।
एक दूसरे त्रिभुज का कर्ण $17$ है और एक \omterm{def:g6:shapes:triangles}{समकोण} भुजा $8$: दूसरी \omterm{def:g6:shapes:triangles}{समकोण}
भुजा निकालो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:2}
किसी त्रिभुज की भुजाएँ $7$, $24$ और $25$ हैं। क्या वह \omterm{def:g6:shapes:triangles}{समकोण} है?
भुजाओं $5$, $6$ और $8$ के लिए भी यही सवाल।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:3}
त्रिभुज $DEF$ का $D$ पर \omterm{def:g6:shapes:triangles}{समकोण} है, और $E$ पर बने कोण को $\theta$ कहा
गया है: कर्ण कौन-सी भुजा है? $\theta$ की सम्मुख भुजा कौन-सी है?
$\cos\theta$, $\sin\theta$ और $\tan\theta$ को भुजाओं $DE$, $DF$, $EF$
के अनुपातों के रूप में लिखो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:4}
किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} का कर्ण $10$ नापता है और एक न्यून कोण $28^\circ$
का है। दोनों \omterm{def:g6:shapes:triangles}{समकोण} भुजाएँ निकालो (दसवें भाग तक गोल करो;
$\cos 28^\circ \approx 0.883$, $\sin 28^\circ \approx 0.469$)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:5}
किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} में कोण $\theta$ की संलग्न भुजा $6$ नापती है और
सम्मुख भुजा $4.5$। $\tan\theta$ निकालो, फिर $\theta$ (डिग्री तक गोल
करो; $\tan^{-1}(0.75) \approx 36.9^\circ$)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:6}
कोई ड्रोन $120$ m की ऊँचाई पर उड़ रहा है। ज़मीन पर खड़े किसी देखने
वाले को वह $32^\circ$ के उन्नयन कोण पर दिखता है। ड्रोन उससे कितनी

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:7}
किसी ढालू रास्ते को $1.2$ m चढ़ना है, और क्षैतिज के साथ उसका कोण
ज़्यादा से ज़्यादा $5^\circ$ हो सकता है। \omterm{prop:g9:linfunc:affinegraph}{ढलान} के साथ-साथ उसकी कम से कम
लंबाई कितनी होनी चाहिए ($\sin 5^\circ \approx 0.0872$)? मीटर के दसवें
भाग तक ऊपर की ओर गोल करो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:8}
\cref{prop:g9:trig:properties} की मदद से: किसी न्यून कोण $\theta$ के
लिए $\cos\theta = 0.6$ है। $\theta$ निकाले बिना $\sin\theta$ निकालो,

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:trig:9}
\cref{ex:g9:trig:special} के बिलकुल सही मानों का कारण दो: \omterm{def:g6:shapes:triangles}{समकोण}

वाले \omterm{def:g6:shapes:triangles}{समबाहु त्रिभुज} में ऊँचाई निकालो, और उससे $30^\circ$ तथा

\end{exercise}

\section{समस्या: जहाँ तक पहुँचा ही नहीं जा सकता, उसे नापना}

\begin{problem}[{सप्ताहांत समस्या --- जहाँ पहुँचा नहीं जा सकता ऐसी
ऊँचाइयों के लिए भूमापकों की दो-ठिकाना विधि, सड़कों की चढ़ाई, क्षितिज
की दूरी, और स्पर्शज्या की अनंत तक दौड़}]
\label{pb:g9:trig:1}
किसी पहाड़ की चोटी तक, नदी के उस पार खड़ी मीनार के शिखर तक, या समंदर
के क्षितिज तक कोई फ़ीता नहीं पहुँचता --- पर एक कोणमापी और इस अध्याय के
तीन अनुपात पहुँच जाते हैं। इस समस्या का केंद्र भूमापकों की मशहूर
\emph{दो-ठिकाना विधि} है, जो किसी ऊँचाई को उसकी जड़ के पास गए बिना ही
निकाल देती है; उसके चारों ओर: सड़क के बोर्ड, सीढ़ियाँ, ज़ीने, धरती का
गोलपन, और किसी फलन को अनंत की ओर फटते हुए देखने की पहली झलक।

\medskip
\noindent\textbf{भाग I --- एक ठिकाना।}
(मान: $\tan 35^\circ \approx 0.700$,
$\sin 35^\circ \approx 0.574$, $\cos 35^\circ \approx 0.819$,
$\tan 1^\circ \approx 0.0175$।)
\begin{enumerate}
  \item किसी प्रकाशस्तंभ की जड़ से $60$ m दूर खड़े होने पर उसका
        शिखर $35^\circ$ के उन्नयन पर दिखता है (क्षैतिज से नापा गया,
        आँख की ऊँचाई पर)। शिखर आँख की ऊँचाई से कितना ऊपर है?
  \item सड़क का कोई बोर्ड $12\,\%$ की चढ़ाई बताता है: हर $100$ m
        क्षैतिज \omterm{def:g8:speed:def}{चाल} पर सड़क $12$ m चढ़ती है। सड़क क्षैतिज के साथ
        कितना कोण बनाती है? और $45^\circ$ वाली सड़क की चढ़ाई कितने
        प्रतिशत की होगी?
  \item किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} के दोनों न्यून कोण पूरक होते हैं, और एक
        की सम्मुख भुजा दूसरे की संलग्न भुजा होती है। इससे ``को''
        वाली सर्वसमिकाएँ निकालो:
        \[
        \sin(90^\circ - \theta) = \cos\theta,
        \qquad
        \cos(90^\circ - \theta) = \sin\theta .
        \]
  \item $8$ m की कोई सीढ़ी ज़मीन से $60^\circ$ पर टिकी है।
        \cref{ex:g9:trig:special} के बिलकुल सही मानों से बताओ कि वह
        कितनी ऊँचाई तक पहुँचती है और उसका पैर दीवार से कितनी दूर है
        --- पहले बिलकुल सही, फिर सेंटीमीटर तक।
  \item $\theta = 35^\circ$ पर $\cos^2\theta + \sin^2\theta = 1$ को
        संख्याओं से जाँचो, और $\theta = 60^\circ$ पर बिलकुल सही-सही।
        यह सर्वसमिका (\cref{prop:g9:trig:properties}) कैलकुलेटर के
        काम की जाँच के लिए अच्छी आदत क्यों है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दो ठिकाने।}
किसी पहाड़ की चोटी दिख रही है, पर उसकी जड़ तक पहुँचा नहीं जा सकता।
बिंदु $A$ से चोटी का उन्नयन $30^\circ$ है; पहाड़ की ओर सीधे $200$ m
चलकर $B$ पर पहुँचने पर उन्नयन $40^\circ$ हो जाता है। आँख की ऊँचाई से
ऊपर चोटी की ऊँचाई को $h$ लिखो, और $B$ से चोटी की खड़ी रेखा तक की
क्षैतिज दूरी को $x$। ($\tan 30^\circ \approx 0.577$,
$\tan 40^\circ \approx 0.839$।)
\begin{enumerate}[resume]
  \item $\tan 40^\circ$ और $\tan 30^\circ$ को $h$, $x$ तथा
        $x + 200$ की मदद से लिखो: दो अज्ञातों के लिए दो \omterm{def:g8:equations:def}{समीकरण}।
  \item हर \omterm{def:g8:equations:def}{समीकरण} से क्षैतिज दूरी को $h$ के रूप में निकालो, दोनों को
        घटाओ, और दो-ठिकाना सूत्र तक पहुँचो:
        \[
        h = \frac{200}
        {\ \dfrac{1}{\tan 30^\circ} - \dfrac{1}{\tan 40^\circ}\ } .
        \]
        $h$ को नज़दीकी दहाई मीटर तक निकालो।
  \item $x$ भी निकालो, और $A$ से लिए गए $30^\circ$ के दर्शन पर अपने
        दोनों उत्तरों की समझदारी जाँचो।
  \item एक वाक्य में: यहाँ दो-ठिकाना विधि के बिना काम क्यों नहीं
        चलता --- सवाल 1 वाली एक-ठिकाना विधि के लिए कौन-सी अकेली नाप
        चाहिए होती, जो यहाँ मुमकिन ही नहीं?
  \item कोई मीनार $30^\circ$ पर दिखती है, और उसकी ओर $200$ m चलने
        पर $60^\circ$ पर। $\tan 30^\circ = \frac{1}{\sqrt3}$ तथा
        $\tan 60^\circ = \sqrt3$ लेकर ऊँचाई \emph{बिलकुल सही-सही}
        निकालो --- उत्तर $\sqrt3$ का एक \omterm{def:g9:arith:divisor}{गुणज} है --- और फिर मीटर तक।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- क्षितिज तक, और दीवार पर ऊपर।}
\begin{enumerate}[resume]
  \item क्षितिज कितनी दूर है? $R$ \omterm{def:g3:shapes:shapes}{त्रिज्या} वाली गोल धरती पर आँखें
        $h$ मीटर ऊँची रखकर खड़े हो: तुम्हारी दृष्टि-रेखा गोले को
        छूती हुई निकल जाती है, और वह रेखा, छूने वाले बिंदु तक की
        \omterm{def:g3:shapes:shapes}{त्रिज्या} तथा तुम्हारे पैरों तक की \omterm{def:g3:shapes:shapes}{त्रिज्या} मिलकर एक \omterm{def:g6:shapes:triangles}{समकोण
        त्रिभुज} बनाती हैं। पाइथागोरस
        (\cref{thm:g9:trig:pythagoras}) से दिखाओ कि क्षितिज तक की
        दूरी $d$ शर्त $d^2 = 2Rh + h^2 \approx 2Rh$ मानती है, और
        $1.80$ m ऊँची आँखों के लिए $d$ निकालो
        ($R = 6.37 \times 10^6$ m)।
  \item $324$ m ऊँची मीनार के शिखर से वही सवाल। (नाविकों का मोटा
        नियम: किलोमीटर में क्षितिज क़रीब $3.6\sqrt{h}$ होता है, जहाँ
        $h$ मीटर में है --- अपने दोनों उत्तर इसी पर परखो।)
  \item तुम्हारा अंगूठा, क़रीब $2$ cm चौड़ा, बाँह भर दूर यानी आँख से
        क़रीब $60$ cm पर: वह कितना कोण ढकता है? पूरा चाँद क़रीब
        $0.5^\circ$ ढकता है: तुम्हारे अंगूठे का कितना हिस्सा पूरे
        चाँद को छिपा देता है?
  \item आरामदेह सीढ़ियों का हर क़दम $17$ cm ऊँचा और $29$ cm चौड़ा
        होता है। वे कितने कोण पर चढ़ती हैं? \cref{ex:g9:trig:special}
        वाले $30^\circ$ से तुलना करो, और $60^\circ$ पर खड़ी अटारी की
        सीढ़ी से भी: $29$ cm चौड़ाई पर \emph{उसे} कितनी ऊँचाई चाहिए
        होती?
  \item \omterm{def:g9:trig:ratios}{स्पर्शज्या} की दौड़: कैलकुलेटर से $\tan 10^\circ$,
        $\tan 45^\circ$, $\tan 80^\circ$, $\tan 89^\circ$ निकालो।
        कोण $90^\circ$ के पास पहुँचते ही क्या होता है, और क्यों
        (संलग्न भुजा के बारे में सोचो)? दीवार खड़ी है, अनुपात का कोई
        मान बचता ही नहीं, और आलेख अनंत की ओर छूट जाता है --- यह
        तुम्हारा पहला \emph{अनंतस्पर्शी} है, जिसे हाई-स्कूल खंड पास
        से देखता है।

\end{enumerate}
\end{problem}
