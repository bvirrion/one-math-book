\chapter{रैखिक और एकघाती फलन}\label{ch:g9:linfunc}

``तीन किलोग्राम की क़ीमत एक किलोग्राम से तीन गुनी होती है'': यही
\omterm{def:g7:prop:table}{समानुपात} है, दो राशियों के बीच का सबसे सादा रिश्ता। \omterm{def:g9:linfunc:linear}{रैखिक फलन}
\omterm{def:g7:prop:table}{समानुपात} को ही फलनों की भाषा में कहते हैं; \omterm{def:g9:linfunc:affine}{एकघाती फलन} उसमें एक तयशुदा
शुरुआती राशि और जोड़ देते हैं। इनके आलेख सीधी रेखाएँ होते हैं, और उन
रेखाओं को पढ़ना ही वह हुनर है जो यह अध्याय बनाता है।

\section{समानुपात और रैखिक फलन}

\begin{definition}[रैखिक फलन]\label{def:g9:linfunc:linear}
\emph{रैखिक फलन}\index{रैखिक फलन} हर संख्या को एक तयशुदा संख्या $a$
से गुणा कर देता है:
\[
f(x) = a x .
\]
संख्या $a$, $f$ का \emph{गुणांक} कहलाती है। दो राशियाँ
\emph{समानुपाती}\index{समानुपाती} ठीक तभी होती हैं जब एक दूसरी का
रैखिक फलन हो।
\end{definition}

\begin{example}\label{ex:g9:linfunc:linear}
अगर सेब $3$ प्रति किलोग्राम मिलते हैं, तो $x$ किलोग्राम की क़ीमत
$f(x) = 3x$ है: दुगना ख़रीदने पर दाम दुगना, और $2.5$ kg की क़ीमत
$f(2.5) = 7.5$।

और उलटे, अगर किसी \omterm{def:g9:linfunc:linear}{रैखिक फलन} के लिए $f(4) = 10$ हो, तो
$a = \frac{10}{4} = 2.5$ और $f(x) = 2.5x$: \emph{एक ही मान किसी \omterm{def:g9:linfunc:linear}{रैखिक
फलन} को पूरी तरह तय कर देता है।}
\end{example}

\begin{proposition}[रैखिक फलन का आलेख]\label{prop:g9:linfunc:lineargraph}
$f(x) = ax$ का आलेख \emph{मूल बिंदु से होकर जाती} एक सीधी रेखा है। और
उलटे, मूल बिंदु से होकर जाती हर वह रेखा जो खड़ी नहीं है, किसी \omterm{def:g9:linfunc:linear}{रैखिक
फलन} का आलेख होती है।
\end{proposition}
\admitted

\section{एकघाती फलन}

\begin{definition}[एकघाती फलन]\label{def:g9:linfunc:affine}
\emph{एकघाती फलन}\index{एकघाती फलन} इस रूप का होता है
\[
f(x) = a x + b ,
\]
जहाँ $a$ और $b$ तयशुदा संख्याएँ हैं। \omterm{def:g9:linfunc:linear}{रैखिक फलन} इसकी ख़ास हालत $b = 0$
हैं, और अचर फलन हालत $a = 0$।
\end{definition}

\begin{example}
कोई व्यायामशाला $20$ का पंजीकरण शुल्क लेती है, फिर हर बार आने के $5$:
$x$ बार आने का कुल ख़र्च $f(x) = 5x + 20$ है। ख़र्च आने की \omterm{def:g7:stats:frequency}{गिनती} के
\emph{\omterm{def:g9:linfunc:linear}{समानुपाती} नहीं} है ($10$ बार आना $5$ बार आने से दुगना महँगा नहीं
पड़ता), पर हर अतिरिक्त बार उतना ही, यानी $5$, जोड़ता है।
\end{example}

\begin{proposition}[एकघाती फलन का आलेख]\label{prop:g9:linfunc:affinegraph}
$f(x) = ax + b$ का आलेख एक सीधी रेखा है:
\begin{itemize}
  \item $b = f(0)$ उसका \emph{$y$-अंतःखंड} है: रेखा खड़ी अक्ष को
        $(0, b)$ पर काटती है;
  \item $a$ उसकी \emph{ढलान}\index{ढलान} है: एक इकाई दाईं ओर बढ़ने पर
        $a$ इकाई ऊपर चढ़ना पड़ता है ($a < 0$ हो तो उतरना); $a > 0$ पर
        फलन बढ़ता है और $a < 0$ पर घटता है;
  \item किन्हीं दो संख्याओं $u \neq v$ के लिए
        $a = \dfrac{f(v) - f(u)}{v - u}$।
\end{itemize}
\end{proposition}

\begin{proof}[आख़िरी बिंदु का प्रमाण]
$f(v) - f(u) = (av + b) - (au + b) = a(v - u)$; अब $v - u$ का भाग दो।
बाक़ी इस स्तर पर बिना प्रमाण के मान लिया जाता है (हाई-स्कूल खंड
रेखाओं और उनके समीकरणों को पूरी तरह देखता है)।
\end{proof}

\begin{method}[दो मानों से एकघाती फलन निकालना]\label{met:g9:linfunc:findaffine}
$u \neq v$ के साथ $f(u)$ और $f(v)$ दिए हों, तो:
\begin{enumerate}
  \item \omterm{prop:g9:linfunc:affinegraph}{ढलान} $a = \dfrac{f(v) - f(u)}{v - u}$ निकालो;
  \item दोनों ज्ञात मानों में से एक को $f(x) = ax + b$ में रखकर $b$
        निकालो;
  \item दूसरे मान से जाँचो।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:linfunc:findaffine}
वह \omterm{def:g9:linfunc:affine}{एकघाती फलन} निकालो जिसके लिए $f(2) = 7$ और $f(5) = 16$ हो।
\[
a = \frac{16 - 7}{5 - 2} = \frac93 = 3 ,
\]
फिर $f(2) = 3 \times 2 + b = 7$ से $b = 1$ मिलता है: $f(x) = 3x + 1$।
जाँच: $f(5) = 15 + 1 = 16$।
\end{example}

\section{प्रतिशत}

\begin{proposition}[प्रतिशत बदलाव]\label{prop:g9:linfunc:percent}
किसी राशि को $t\,\%$ बढ़ाना उसे $1 + \dfrac{t}{100}$ से गुणा करना है,
और $t\,\%$ घटाना उसे $1 - \dfrac{t}{100}$ से गुणा करना। प्रतिशत बदलाव
\omterm{def:g9:linfunc:linear}{रैखिक फलन} होते हैं।
\end{proposition}

\begin{proof}
$x$ को $t\,\%$ बढ़ाने का मतलब है उसमें $\frac{t}{100}\,x$ जोड़ना:
\[
x + \frac{t}{100}\,x = \left(1 + \frac{t}{100}\right) x .
\]
घटाने के लिए वही गणना, बस ऋण चिह्न के साथ।
\end{proof}

\begin{example}\label{ex:g9:linfunc:percent}
$80$ की क़ीमत $15\%$ बढ़ती है: नई क़ीमत
$80 \times 1.15 = 92$। $60$ की क़ीमत $30\%$ घटती है:
$60 \times 0.7 = 42$।

\emph{एक के बाद एक लगे बदलाव अपने गुणकों को गुणा कर देते हैं।}
$20\%$ की बढ़त के बाद $20\%$ की कमी लगाने पर
$1.2 \times 0.8 = 0.96$ मिलता है: कुल मिलाकर $4\%$ की \emph{कमी}, न
कि शुरुआत पर वापसी!
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:1}
इन फलनों में से कौन-से रैखिक हैं? कौन-से एकघाती? और कौन-से न वे, न ये?
\[
f(x) = 4x, \qquad
g(x) = 3x - 5, \qquad
h(x) = x^2, \qquad
k(x) = -\tfrac x2, \qquad
m(x) = 7 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:2}
मान लो $f(x) = 2x - 3$ है। $f(0)$, $f(4)$, $f(-1)$ निकालो, और वह
संख्या $x$ ढूँढ़ो जिसके लिए $f(x) = 9$ हो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:3}
किसी \omterm{def:g9:linfunc:linear}{रैखिक फलन} के लिए $f(6) = 15$ है। उसका गुणांक निकालो, फिर $f(10)$
और $f(-2)$ निकालो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:4}
एक ही निर्देशांक जाल पर $f(x) = 2x$, $g(x) = 2x + 3$ और
$h(x) = -x + 4$ के आलेख खींचो। $f$ और $g$ के आलेखों के बारे में क्या
कहा जा सकता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:5}
वह \omterm{def:g9:linfunc:affine}{एकघाती फलन} निकालो जिसके लिए $f(1) = 5$ और $f(3) = 11$ हो; फिर वह
जिसके लिए $g(0) = 4$ और $g(2) = 0$ हो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:6}
निकालो: $120$ पर $25\%$ की बढ़त के बाद की क़ीमत; $65$ पर $40\%$ की छूट
के बाद की क़ीमत; और वह असली क़ीमत जिस पर $25\%$ छूट के बाद कोई चीज़
$69$ की पड़ती है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:7}
फ़ोन की योजना A का दाम $10$ प्रति महीना और उसके ऊपर $0.05$ प्रति मिनट
बात है; योजना B का दाम $25$ प्रति महीना है, बात जितनी मर्ज़ी।
\begin{enumerate}
  \item योजना A का महीने का ख़र्च मिनटों की \omterm{def:g7:stats:frequency}{गिनती} $x$ के फलन के रूप
        में लिखो।
  \item महीने में कितने मिनट से आगे योजना B सस्ती पड़ती है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:8}
$2000$ जीवाणुओं की आबादी हर घंटे $10\%$ बढ़ती है।
\begin{enumerate}
  \item एक घंटे बाद कितने जीवाणु होंगे? और दो घंटे बाद?
  \item समझाओ कि दो घंटे बाद का उत्तर $2400$ क्यों नहीं है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:9}
किसी \omterm{def:g9:linfunc:affine}{एकघाती फलन} का आलेख बिंदुओं $(2, 3)$ और $(6, 1)$ से होकर जाता है।
\begin{enumerate}
  \item उसकी \omterm{prop:g9:linfunc:affinegraph}{ढलान} निकालो। फलन बढ़ता है या घटता है?
  \item उसका व्यंजक बताओ, और वह संख्या निकालो जिस पर फलन का मान $0$
        होता है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:linfunc:10}
कोई दुकान पहले क़ीमत $t\,\%$ बढ़ाती है, फिर नई क़ीमत $t\,\%$ घटा देती
है।
\begin{enumerate}
  \item दिखाओ कि आख़िरी क़ीमत असली क़ीमत को $1 - \dfrac{t^2}{10000}$ से
        गुणा करने पर मिलती है।
  \item इससे नतीजा निकालो कि आख़िरी क़ीमत हमेशा असली से \emph{कम} रहती
        है ($0 < t \leq 100$ के लिए), और $t = 10$ पर कुल प्रतिशत
        बदलाव निकालो।
\end{enumerate}
\end{exercise}

\section{समस्या: दो थर्मामीटर}

\begin{problem}[{सप्ताहांत समस्या --- सेल्सियस, फ़ारेनहाइट और रोज़मर्रा
के एकघाती फलन: एक तापमान दोनों मापनियों पर एक ही अंक दिखाता है, और
अचर ढलान ही सीधेपन की पहचान है}]
\label{pb:g9:linfunc:1}
कहीं लोग ``$68$ डिग्री'' सुनकर हल्के कपड़े पहन लेते हैं, कहीं वही
सुनकर कोट उठा लेते हैं। दुनिया के दोनों थर्मामीटर एक \omterm{def:g9:linfunc:affine}{एकघाती फलन} से
जुड़े हैं --- और दो तथ्यों से उसे बना लेना ठीक वही हुनर है जो
\cref{met:g9:linfunc:findaffine} सिखाती है। यह समस्या वह रूपांतरण
बनाती है, वह अजीब तापमान ढूँढ़ती है जिस पर दोनों मापनियाँ एक ही बात
कहती हैं, फिर उसी चश्मे से टैक्सियाँ, झींगुर और मोमबत्तियाँ पढ़ती है,
और उस जाँच पर ख़त्म होती है जो सीधेपन को ही पहचान लेती है।

\medskip
\noindent\textbf{भाग I --- रूपांतरण बनाना।}
दो तथ्य फ़ारेनहाइट \omterm{def:g7:prop:scale}{मापनी} को बाँध देते हैं: पानी $32\,^\circ$F पर जमता
है और $212\,^\circ$F पर उबलता है (यानी $0\,^\circ$C और
$100\,^\circ$C पर)।
\begin{enumerate}
  \item वह \omterm{def:g9:linfunc:affine}{एकघाती फलन} $F = f(C)$ निकालो जो सेल्सियस को फ़ारेनहाइट में
        बदलता है ($f(0) = 32$ और $f(100) = 212$ मानों के साथ
        \cref{met:g9:linfunc:findaffine})।
  \item फ़ारेनहाइट में बदलो: $20\,^\circ$C का सुहाना दिन; शरीर का
        $37\,^\circ$C; और जमा देने वाला $-10\,^\circ$C।
  \item $C$ के लिए हल करके उलटा रूपांतरण बनाओ, और उसे $50\,^\circ$F
        तथा $98.6\,^\circ$F पर लगाओ।
  \item क्या $f$ \emph{रैखिक} फलन है
        (\cref{def:g9:linfunc:linear})? पहचान की निशानियाँ जाँचो:
        $f(0)$ क्या है, और सेल्सियस का पाठ दुगना करने पर फ़ारेनहाइट
        का पाठ दुगना होता है क्या? सही शब्द के साथ नतीजा निकालो
        (\cref{def:g9:linfunc:affine})।
  \item मशहूर पहेली: क्या कोई ऐसा तापमान है जो दोनों मापनियों पर
        \emph{एक ही अंक} दिखाता हो? $f(x) = x$ हल करो, और बताओ कि
        तुम्हारे हल का $f$ के आलेख पर क्या मतलब है (आलेख वहाँ किस
        रेखा को काटता है?)।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- खुले में \omterm{def:g9:linfunc:affine}{एकघाती फलन}।}
\begin{enumerate}[resume]
  \item टैक्सी A $3$ यूरो और उसके ऊपर $1.50$ प्रति km लेती है;
        टैक्सी B सीधे $2$ यूरो प्रति km। दोनों दाम-फलन लिखो और $4$
        km तथा $8$ km की सवारी के लिए उनकी तुलना करो।
  \item वह दूरी निकालो जिस पर दोनों टैक्सियाँ बराबर पड़ती हैं, और
        आलेख पर पूरी स्थिति बताओ (दो रेखाएँ --- कटान-बिंदु पर, उससे
        पहले और उसके बाद क्या होता है?)।
  \item प्रकृति देखने वालों का मोटा नियम: तापमान $T\,^\circ$C पर
        झींगुर एक मिनट में क़रीब $N = 7T - 30$ बार बोलता है।
        $20\,^\circ$C की शाम को कितनी बार? तुम एक मिनट में $82$ बार
        गिनते हो: झींगुर कौन-सा तापमान बता रहा है?
  \item $600$ यूरो में ख़रीदा गया फ़ोन हर महीने $15$ यूरो क़ीमत खोता
        है: क़ीमत-फलन $v(t)$ लिखो, बताओ कि फ़ोन कब $240$ यूरो का रह
        जाएगा, और नमूने के हिसाब से कब उसकी कोई क़ीमत ही नहीं
        बचेगी। क्या उस तारीख़ के बाद भी सूत्र का कोई मतलब रह जाता
        है?
  \item प्रतिशत वाली चालें \omterm{def:g9:linfunc:linear}{रैखिक फलन} हैं: $+25\,\%$ यानी $1.25$ से
        गुणा, और $-20\,\%$ यानी $0.8$ से
        (\cref{prop:g9:linfunc:percent})। इन्हें एक के बाद एक लगाओ:
        $+25\,\%$ की बढ़त के बाद $-20\,\%$ की कटौती क़ीमत के साथ क्या
        करती है? इस छोटे चमत्कार को समझाओ, और
        \cref{exo:g9:linfunc:10} से तुलना करो।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- सब कुछ \omterm{prop:g9:linfunc:affinegraph}{ढलान} है।}
\begin{enumerate}[resume]
  \item जलाते समय कोई मोमबत्ती $20$ cm की है और आधे घंटे बाद $17$
        cm की। पिघलना एकघाती मानते हुए $h(t)$ निकालो (ऊँचाई cm में,
        $t$ मिनट में), और बताओ कि मोमबत्ती कब बुझ जाएगी।
  \item दो साइकिल-सवार एक ही सड़क पर चलते हैं: पहले की दूरी
        $d_1(t) = 24t$ है ($t$ घंटे बाद km में), दूसरे की
        $d_2(t) = 10 + 18t$। किसने आगे से शुरुआत की, कौन तेज़ चलता
        है, और पहला दूसरे को किस समय तथा किस किलोमीटर पर पकड़ लेता
        है?
  \item मान लो $f(x) = 2x + 3$ और $g(x) = -0.5x + 7$ हैं।
        $f(x) + g(x)$ निकालो और जाँचो कि वह फिर से एकघाती है। आम
        तौर पर किसी \omterm{def:g1:addition:def}{योगफल} की \omterm{prop:g9:linfunc:affinegraph}{ढलान} और अंतःखंड क्या होते हैं?
  \item \omterm{def:g8:equations:def}{समीकरण} $mx + p = 0$ पर पूरी चर्चा करो: $m \neq 0$ पर कितने
        हल होते हैं (और कौन-सा)? $m = 0$ तथा $p \neq 0$ पर? और
        $m = 0$ तथा $p = 0$ पर? (हर हालत की एक आलेख वाली कहानी है:
        \omterm{prop:g9:linfunc:affinegraph}{ढलान} $m$ वाली रेखा आड़ी अक्ष को कहाँ काटती है?)
  \item संरेखता की जाँच: क्या बिंदु $(1, 5)$, $(3, 9)$, $(7, 17)$
        एक ही सीधी रेखा पर हैं? और $(1, 5)$, $(3, 9)$, $(6, 16)$?
        जोड़ों के बीच बदलाव की दर निकालो और फ़ैसला करो। सीख लिखो:
        \emph{बदलाव की अचर दर} ही एकघाती फलनों की पहचान है ---
        टेढ़े आलेख अपनी दर बदलते रहते हैं, और \emph{उसे} नापना
        हाई-स्कूल खंड की कहानी है।

\end{enumerate}
\end{problem}
