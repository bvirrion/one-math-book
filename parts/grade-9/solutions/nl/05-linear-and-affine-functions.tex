\section*{Hoofdstuk \ref{ch:g9:linfunc} --- Lineaire en affiene functies}

\begin{solution}{exo:g9:linfunc:1}
Lineair: $f$ en $k$ (vorm $ax$). Affien maar niet lineair: $g$
(\omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} $3$, snijpunt $-5$) en $m$ (constant,
\omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} $0$). Geen van beide: $h$, door het kwadraat.
\end{solution}

\begin{solution}{exo:g9:linfunc:2}
$f(0) = -3$; $f(4) = 5$; $f(-1) = -5$. En $f(x) = 9$ betekent $2x - 3 = 9$, dus
$2x = 12$ en $x = 6$.
\end{solution}

\begin{solution}{exo:g9:linfunc:3}
$a = \frac{15}{6} = 2.5$, dus is $f(x) = 2.5x$. Dan is $f(10) = 25$ en
$f(-2) = -5$.
\end{solution}

\begin{solution}{exo:g9:linfunc:4}
$f$ gaat door de oorsprong met \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} $2$; $g$ heeft dezelfde
\omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} maar snijdt de $y$-as in $3$; $h$ daalt vanaf $(0, 4)$ met
\omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} $-1$. De grafieken van $f$ en $g$ zijn \emph{\omterm{def:g6:lines:perp}{parallelle
lijnen}} (dezelfde \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt}, verschillende snijpunten).
\end{solution}

\begin{solution}{exo:g9:linfunc:5}
Voor $f$: \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} $\frac{11 - 5}{3 - 1} = 3$, en dan geeft
$5 = 3 \times 1 + b$ dat $b = 2$: $f(x) = 3x + 2$.

Voor $g$: \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} $\frac{0 - 4}{2 - 0} = -2$, en $g(0) = 4$ is al het
snijpunt: $g(x) = -2x + 4$.
\end{solution}

\begin{solution}{exo:g9:linfunc:6}
Stijging van $25\%$: $120 \times 1.25 = 150$.

Korting van $40\%$: $65 \times 0.6 = 39$.

Oorspronkelijke prijs $p$ met $p \times 0.75 = 69$: $p = \frac{69}{0.75} = 92$.
\end{solution}

\begin{solution}{exo:g9:linfunc:7}
\emph{1.} $f(x) = 0.05x + 10$.

\emph{2.} Abonnement B is goedkoper wanneer $25 < 0.05x + 10$, dat wil zeggen
$15 < 0.05x$, dus $x > 300$. Vanaf $301$ minuten ($5$ uur) per maand wint
abonnement B.
\end{solution}

\begin{solution}{exo:g9:linfunc:8}
\emph{1.} Na één uur: $2000 \times 1.1 = 2200$. Na twee uur:
$2200 \times 1.1 = 2420$.

\emph{2.} De tweede stijging werkt op $2200$, niet op $2000$: twee keer met $10\%$
groeien vermenigvuldigt met $1.1^2 = 1.21$, een stijging van $21\%$ en niet van
$20\%$.
\end{solution}

\begin{solution}{exo:g9:linfunc:9}
\emph{1.} \omterm{prop:g9:linfunc:affinegraph}{Richtingscoëfficiënt}: $\frac{1 - 3}{6 - 2} = \frac{-2}{4} = -0.5$: de
functie is dalend.

\emph{2.} $f(x) = -0.5x + b$ met $f(2) = 3$: $-1 + b = 3$, dus $b = 4$ en
$f(x) = -0.5x + 4$. Output $0$: $-0.5x + 4 = 0$ geeft $x = 8$.
\end{solution}

\begin{solution}{exo:g9:linfunc:10}
\emph{1.} De twee veranderingen vermenigvuldigen de prijs met
\[
\left(1 + \frac{t}{100}\right)\left(1 - \frac{t}{100}\right)
= 1 - \frac{t^2}{10000}
\]
(het derde merkwaardige \omterm{def:g2:mult:def}{product} met $a = 1$, $b = \frac{t}{100}$).

\emph{2.} Voor $0 < t \leq 100$ is $\frac{t^2}{10000} > 0$, dus is de factor
kleiner dan $1$: de eindprijs is lager dan de oorspronkelijke. Voor $t = 10$:
factor $1 - \frac{100}{10000} = 0.99$, dat wil zeggen een totale daling van
$1\%$.
\end{solution}

\begin{solution}{pb:g9:linfunc:1}
\textbf{1.} \omterm{prop:g9:linfunc:affinegraph}{Richtingscoëfficiënt}: $\frac{f(100) - f(0)}{100 - 0} =
\frac{212 - 32}{100} = 1.8$; snijpunt $f(0) = 32$. Bijgevolg is
\[
f(C) = 1.8\,C + 32 .
\]

\textbf{2.} $f(20) = 68\,^\circ$F (de beroemde milde dag);
$f(37) = 98.6\,^\circ$F; $f(-10) = 14\,^\circ$F.

\textbf{3.} Uit $F = 1.8C + 32$ volgt $C = \frac{F - 32}{1.8}$. Dan wordt
$50\,^\circ$F $\to \frac{18}{1.8} = 10\,^\circ$C, en $98.6\,^\circ$F
$\to \frac{66.6}{1.8} = 37\,^\circ$C.

\textbf{4.} $f(0) = 32 \neq 0$, en $f(20) = 68$ is niet het dubbel van
$f(10) = 50$: niet lineair. De grafiek is een rechte \omterm{def:g6:lines:objects}{lijn} die de oorsprong mist:
$f$ is \emph{affien}, niet lineair.

\textbf{5.} $1.8x + 32 = x$ geeft $0.8x = -32$, dus $x = -40$: bij veertig onder
\omterm{def:g1:counting:zero}{nul} wijzen beide thermometers $-40$ aan. Op de grafiek snijdt de \omterm{def:g6:lines:objects}{lijn} van $f$ de
diagonale \omterm{def:g6:lines:objects}{lijn} $y = x$ precies daar.

\textbf{6.} $p_A(k) = 1.5k + 3$ en $p_B(k) = 2k$. Voor $4$ km: $9$ euro tegenover
$8$: B goedkoper. Voor $8$ km: $15$ tegenover $16$: A goedkoper.

\textbf{7.} $1.5k + 3 = 2k$ geeft $k = 6$: bij zes kilometer kosten beide $12$
euro. Op de grafiek begint de \omterm{def:g6:lines:objects}{lijn} van B (door de oorsprong, steiler) onder die van
A en snijdt haar in $(6, 12)$: B wint de korte ritten, A de lange.

\textbf{8.} $N = 7 \times 20 - 30 = 110$ tjirpen per minuut. Uit $82 = 7T - 30$:
$T = \frac{112}{7} = 16\,^\circ$C.

\textbf{9.} $v(t) = 600 - 15t$. Waard $240$: $600 - 15t = 240$, dus $t = 24$
maanden. Waard $0$: $t = 40$ maanden. Voorbij $40$ maanden wordt de formule
negatief --- maar de waarde van een telefoon stopt bij \omterm{def:g1:counting:zero}{nul}: elk affien model heeft
een gebied waarin het zin heeft.

\textbf{10.} Achter elkaar zetten: $\times 1.25$ en daarna $\times 0.8$
vermenigvuldigt met $1.25 \times 0.8 = 1$: de prijs komt precies op zijn begin
terug. Geen wonder: de percenten werken op verschillende referenties (de daling
werkt op de verhoogde prijs), en hier vallen de vermenigvuldigers toevallig tegen
elkaar weg --- het algemene verhaal, met zijn ingebouwde verlies
$1 - \frac{t^2}{10\,000}$, is \cref{exo:g9:linfunc:10}.

\textbf{11.} \omterm{prop:g9:linfunc:affinegraph}{Richtingscoëfficiënt}: $\frac{17 - 20}{30} = -0.1$ cm per minuut;
$h(t) = 20 - 0.1t$. Ze is opgebrand bij $h(t) = 0$: $t = 200$ minuten --- drie uur
en twintig minuten licht.

\textbf{12.} De tweede begon $10$ km voorop (het snijpunt met de as); de eerste is
sneller ($24 > 18$ km/h). Inhalen: $24t = 10 + 18t$ geeft $6t = 10$, dus
$t = \frac53$ h $= 1$ h $40$ min, op kilometer $24 \times \frac53 = 40$.

\textbf{13.} $f(x) + g(x) = (2 - 0.5)x + (3 + 7) = 1.5x + 10$: weer affien, met
als \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} de \emph{\omterm{def:g1:addition:def}{som} van de \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënten}} en als
snijpunt de \omterm{def:g1:addition:def}{som} van de snijpunten --- wat duidelijk wordt door in
$(mx + p) + (m'x + p')$ de gelijksoortige termen samen te nemen.

\textbf{14.} Is $m \neq 0$: precies één oplossing, $x = -\frac pm$ --- een
niet-horizontale \omterm{def:g6:lines:objects}{lijn} snijdt de as één keer. Is $m = 0$ en $p \neq 0$: geen
oplossing --- een horizontale \omterm{def:g6:lines:objects}{lijn} boven of onder de as raakt haar nooit. Is
$m = 0$ en $p = 0$: elke $x$ is een oplossing --- de \omterm{def:g6:lines:objects}{lijn} \emph{is} de as.

\textbf{15.} Eerste drietal: de snelheden $\frac{9 - 5}{3 - 1} = 2$ en
$\frac{17 - 9}{7 - 3} = 2$ zijn gelijk, dus liggen de punten op één \omterm{def:g6:lines:objects}{lijn} (op
$y = 2x + 3$). Tweede drietal: $\frac{9-5}{2} = 2$, maar
$\frac{16 - 9}{3} = \frac73 \neq 2$: niet op één \omterm{def:g6:lines:objects}{lijn}. De \omterm{def:g9:linfunc:affine}{affiene functies} zijn
precies die met een constante veranderingssnelheid --- één getal, de
\omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt}, vertelt het hele verhaal; bij krommen verandert dat tempo
zelf van punt tot punt, en haar opjagen leidt tot de afgeleide, in het
bovenbouwvolume.
\end{solution}
