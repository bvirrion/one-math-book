\section*{Hoofdstuk \ref{ch:g9:thales} --- De stelling van Thales}

\begin{solution}{exo:g9:thales:1}
Thales: $\dfrac{AM}{AB} = \dfrac{AN}{AC} = \dfrac{MN}{BC}$, dat wil zeggen
$\dfrac26 = \dfrac13$. Dus geeft $\dfrac{3}{AC} = \dfrac13$ dat $AC = 9$, en
$\dfrac{MN}{9} = \dfrac13$ dat $MN = 3$.
\end{solution}

\begin{solution}{exo:g9:thales:2}
Zoek eerst $AB = AM + MB = 5 + 3 = 8$. Dan geeft
$\dfrac{AM}{AB} = \dfrac{AN}{AC}$ dat $\dfrac58 = \dfrac{4}{AC}$, dus
$5 \times AC = 32$ en $AC = 6.4$.
\end{solution}

\begin{solution}{exo:g9:thales:3}
De vlinderstand werkt precies zoals de geneste stand:
$\dfrac{AM}{AB} = \dfrac{AN}{AC} = \dfrac{MN}{BC}$, dat wil zeggen
$\dfrac{3}{7.5} = \dfrac25$. Dus geeft $\dfrac{2}{AC} = \dfrac25$ dat $AC = 5$, en
$\dfrac{2.6}{BC} = \dfrac25$ dat $BC = \dfrac{2.6 \times 5}{2} = 6.5$.
\end{solution}

\begin{solution}{exo:g9:thales:4}
$\dfrac{AM}{AB} = \dfrac68 = \dfrac34$ en
$\dfrac{AN}{AC} = \dfrac{9}{12} = \dfrac34$: gelijke verhoudingen, punten in
dezelfde orde, dus is $(MN) \parallel (BC)$ volgens de omgekeerde stelling van
Thales.
\end{solution}

\begin{solution}{exo:g9:thales:5}
$\dfrac{AM}{AB} = \dfrac46 = \dfrac23$ en $\dfrac{AN}{AC} = \dfrac58$. Op
gemeenschappelijke \omterm{def:g4:fractions:def}{noemer}: $\dfrac23 = \dfrac{16}{24}$ en
$\dfrac58 = \dfrac{15}{24}$: de verhoudingen verschillen. Waren de \omterm{def:g6:lines:objects}{lijnen}
\omterm{def:g6:lines:perp}{parallel}, dan zou de stelling van Thales gelijke verhoudingen opleggen --- dus
zijn $(MN)$ en $(BC)$ \emph{niet} \omterm{def:g6:lines:perp}{parallel}.
\end{solution}

\begin{solution}{exo:g9:thales:6}
Zij $S$ het uiteinde van de schaduw, $P$ de bovenkant van het hoofd van de persoon
en $L$ de top van de lantaarn. De persoon ($1.8$ m op afstand $3$ m van $S$) en de
lantaarn (hoogte $h$ op afstand $3 + 2 = 5$ m van $S$) zijn twee parallelle
verticale lijnstukken die door de lichtstraal $(SL)$ worden gesneden: Thales vanuit
het punt $S$ geeft
\[
\frac{1.8}{h} = \frac{3}{5},
\qquad\text{dus}\quad
h = \frac{1.8 \times 5}{3} = 3 \text{ m}.
\]
\end{solution}

\begin{solution}{exo:g9:thales:7}
\emph{1.} Echte afstand: $6.8 \times 25\,000 = 170\,000$ cm $= 1.7$ km.

\emph{2.} \omterm{def:g6:measure:area}{Oppervlakten} schalen met $k^2$: echte \omterm{def:g6:measure:area}{oppervlakte}
$= 8 \times 25\,000^2$ cm$^2$ $= 8 \times 6.25 \times 10^8$ cm$^2$
$= 5 \times 10^9$ cm$^2$. Omdat $1$ km$^2 = 10^{10}$ cm$^2$, is dat $0.5$ km$^2$.
\end{solution}

\begin{solution}{exo:g9:thales:8}
\emph{1.} De verhouding is $\dfrac{AM}{AB} = \dfrac{8}{12} = \dfrac23$. Thales:
$AN = \frac23 \times 15 = 10$ en $MN = \frac23 \times 18 = 12$.

\emph{2.} De \omterm{def:g9:thales:scaling}{schaalfactor} is $k = \frac23$. Omtrekken: $ABC$ heeft
$12 + 15 + 18 = 45$, en $AMN$ heeft $8 + 10 + 12 = 30 = \frac23 \times 45$: ook de
\omterm{def:g6:measure:perimeter}{omtrek} schaalt met $k$, zoals elke lengte.
\end{solution}

\begin{solution}{exo:g9:thales:9}
Rond $O$ snijden de \omterm{def:g6:lines:objects}{lijnen} $(AC)$ en $(BD)$ elkaar, en $(AB) \parallel (CD)$:
vlinderstand met de driehoeken $OAB$ en $OCD$. Thales geeft
\[
\frac{OA}{OC} = \frac{OB}{OD} = \frac{AB}{CD} = \frac46 = \frac23 .
\]
Dus is $\dfrac{OA}{OC} = \dfrac23$, en de gelijkheid
$\dfrac{OA}{OC} = \dfrac{OB}{OD}$ zegt precies dat $O$ de twee diagonalen in
dezelfde verhouding verdeelt.
\end{solution}

\begin{solution}{pb:g9:thales:1}
\textbf{1.} De constructie brengt twee punten op $[AB]$ voort; noem ze $M_1$ (uit
$P_1$) en $M_2$ (uit $P_2$).

\textbf{2.} In de driehoek $A P_3 B$ verdelen de parallellen met $(P_3 B)$ door
$P_1$ en $P_2$ de zijden \omterm{def:g9:linfunc:linear}{evenredig} (\cref{thm:g9:thales:direct}):
\[
\frac{A M_1}{AB} = \frac{A P_1}{A P_3} = \frac13,
\qquad
\frac{A M_2}{AB} = \frac{A P_2}{A P_3} = \frac23 .
\]
De passer maakte $A P_1$, $P_1 P_2$ en $P_2 P_3$ gelijk, dus zijn de verhoudingen
precies derden --- wat de hoek van de straal en de gekozen passeropening ook zijn.

\textbf{3.} Voor vijfden: zet vijf gelijke lengten uit, verbind het vijfde punt met
$B$, en teken vier parallellen. Voor $\frac35$ van een \omterm{def:g6:lines:objects}{lijnstuk}: dezelfde figuur, en
neem het punt dat de \omterm{def:g6:lines:perp}{parallel} door $P_3$ afsnijdt: het ligt op $\frac{3}{5}$ van
$[AB]$ vanaf $A$.

\textbf{4.} $\frac{AM}{MB} = \frac23$ betekent $AM = \frac25 AB$: vijf gelijke
stappen op de straal, verbind $P_5$ met $B$, en de \omterm{def:g6:lines:perp}{parallel} door $P_2$ markeert
$M$.

\textbf{5.} $\sqrt2 = 1.41421\ldots$ cm meten gebruikt altijd maar eindig veel
decimalen: elk gemeten ``derde'' is een benadering. De constructie van Thales leest
nooit een getal af: ze levert het exacte punt op $\frac{\sqrt2}{3}$ --- gelijke
delen van een \omterm{def:g6:lines:objects}{lijnstuk} dat geen enkele lat zelfs maar kan uitdrukken
(\cref{pb:g9:sqrt:1}).

\textbf{6.} De bovenkant van het voorwerp, het gaatje en de onderkant van het beeld
liggen op één \omterm{def:g6:lines:objects}{lijn}, en zo ook de onderkant van het voorwerp en de bovenkant van het
beeld: twee \omterm{def:g6:lines:objects}{lijnen} die elkaar in het gaatje snijden, met het voorwerp en de
beeldwand \omterm{def:g6:lines:perp}{parallel}. Thales in de vlinderstand:
\[
\frac{h}{H} = \frac{d}{D},
\qquad\text{dus}\qquad
h = H \times \frac dD .
\]

\textbf{7.} $h = 12 \times \frac{0.20}{30} = 0.08$ m $= 8$ cm --- op zijn kop (de
stralen kruisen elkaar in het gaatje).

\textbf{8.} $H = h \times \frac Dd = 9 \times 10^{-3} \times
\frac{1.5 \times 10^{11}}{1} = 1.35 \times 10^{9}$ m. Werkelijke waarde
$1.39 \times 10^9$ m: een schoenendoos meet de Zon op een paar procent
nauwkeurig.

\textbf{9.} Maan: $\frac{3.5 \times 10^6}{3.8 \times 10^8} \approx 0.0092$. Zon:
$\frac{1.39 \times 10^9}{1.5 \times 10^{11}} \approx 0.0093$. De twee verhoudingen
--- de schijnbare grootten aan de hemel --- zijn bijna gelijk: de Maan kan de Zon
\emph{precies} bedekken, rand tegen rand. Dat kosmische toeval is de totale
zonsverduistering.

\textbf{10.} Elke straal moet door het ene gaatje, dus gaan stralen van de
bovenkant van het voorwerp \emph{naar beneden} verder en stralen van de onderkant
\emph{naar boven}: het beeld staat op zijn kop. Het gaatje vergroten laat een hele
bundel lichtjes verschoven exemplaren van het beeld door, die over elkaar liggen:
helderder, maar uitgesmeerd.

\textbf{11.} $\frac{PQ}{MN} = \frac{OP}{OM} = \frac{7.5}{3} = 2.5$, dus is
$PQ = 10$; en $\frac{ON}{OQ} = \frac{OM}{OP} = \frac{1}{2.5}$, dus is
$ON = \frac{6}{2.5} = 2.4$.

\textbf{12.} Gaat de \omterm{def:g6:lines:perp}{parallel} door het midden van één zijde, dan is de verhouding
van Thales $\frac12$, dus snijdt ze de tweede zijde in \emph{haar} midden, en meet
het parallelle \omterm{def:g6:lines:objects}{lijnstuk} de \omterm{def:g3:division:half}{helft} van de basis: precies
\cref{thm:g8:midpoints:direct}. Thales is de middenparallelstelling, bevrijd van
de verhouding $\frac12$.

\textbf{13.} In de driehoek $ACD$ is de \omterm{def:g6:lines:objects}{lijn} $(EO)$ \omterm{def:g6:lines:perp}{parallel} met de basis $(CD)$,
met $\frac{AO}{AC} = \frac{OA}{OA + OC} = \frac{4}{4 + 6} = \frac25$. Thales:
$\frac{EO}{DC} = \frac{AO}{AC} = \frac25$, dus is $EO = \frac25 \times 6 = 2.4$.

\textbf{14.} In de driehoek $BDC$ is net zo $\frac{BO}{BD} = \frac25$ en
$OF = \frac25 \times 6 = 2.4$. Bijgevolg is
\[
EF = EO + OF = 4.8
= \frac{2 \times 4 \times 6}{4 + 6} :
\]
het \omterm{pb:g8:speed:1}{harmonisch gemiddelde} van de basissen --- het gemiddelde van de
heen-en-terugritten (\cref{pb:g8:speed:1}), getekend in een trapezium.

\textbf{15.} Rekenkundig: $5$; meetkundig: $\sqrt{24} \approx 4.90$; harmonisch:
$4.8$. Op volgorde: $4.8 < 4.90\ldots < 5$, dat wil zeggen harmonisch $<$
meetkundig $<$ rekenkundig --- met gelijkheid alleen bij gelijke basissen. In de
figuur: de middenparallel meet $5$, de \omterm{def:g6:lines:perp}{parallel} door het snijpunt van de
diagonalen $4.8$, en de snede van de gelijkvormigheid $\sqrt{24}$, alle drie
tussen de basissen gestapeld in die orde. De ongelijkheden werden bewezen in
\cref{pb:g8:cosine:1} (meetkundig $\leq$ rekenkundig, met de cirkel en met een
merkwaardig \omterm{def:g2:mult:def}{product}) en \cref{pb:g8:speed:1} (harmonisch $\leq$ rekenkundig);
harmonisch $\leq$ meetkundig volgt door die te combineren ($H \times A = G^2$,
daar ook bewezen).
\end{solution}
