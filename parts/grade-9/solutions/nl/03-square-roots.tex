\section*{Hoofdstuk \ref{ch:g9:sqrt} --- Vierkantswortels}

\begin{solution}{exo:g9:sqrt:1}
$\sqrt{64} = 8$; \quad $\sqrt{100} = 10$; \quad
$\sqrt{0.09} = 0.3$ (want $0.3^2 = 0.09$); \quad
$\sqrt{\frac{1}{25}} = \frac15$; \quad
$\left(\sqrt{13}\right)^2 = 13$; \quad
$\sqrt{(-4)^2} = \sqrt{16} = 4 = \abs{-4}$.
\end{solution}

\begin{solution}{exo:g9:sqrt:2}
$\sqrt{50} = \sqrt{25 \times 2} = 5\sqrt2$; \quad
$\sqrt{45} = \sqrt{9 \times 5} = 3\sqrt5$; \quad
$\sqrt{98} = \sqrt{49 \times 2} = 7\sqrt2$; \quad
$\sqrt{300} = \sqrt{100 \times 3} = 10\sqrt3$.
\end{solution}

\begin{solution}{exo:g9:sqrt:3}
$\sqrt2 \times \sqrt{18} = \sqrt{36} = 6$.

$\dfrac{\sqrt{75}}{\sqrt3} = \sqrt{\dfrac{75}{3}} = \sqrt{25} = 5$.

$\sqrt5 \times \sqrt{20} = \sqrt{100} = 10$.

$\dfrac{\sqrt8}{\sqrt{50}} = \sqrt{\dfrac{8}{50}} = \sqrt{\dfrac{4}{25}}
= \dfrac25$.
\end{solution}

\begin{solution}{exo:g9:sqrt:4}
$x^2 = 36$: $x = 6$ of $x = -6$.

$x^2 = 11$: $x = \sqrt{11}$ of $x = -\sqrt{11}$.

$x^2 + 4 = 0$ betekent $x^2 = -4 < 0$: geen oplossing.

$2x^2 = 50$: $x^2 = 25$, dus $x = 5$ of $x = -5$.
\end{solution}

\begin{solution}{exo:g9:sqrt:5}
$\sqrt{20} + \sqrt{45} = 2\sqrt5 + 3\sqrt5 = 5\sqrt5$.

$3\sqrt8 - \sqrt{18} + \sqrt2 = 3 \times 2\sqrt2 - 3\sqrt2 + \sqrt2
= (6 - 3 + 1)\sqrt2 = 4\sqrt2$.
\end{solution}

\begin{solution}{exo:g9:sqrt:6}
$\left(\sqrt3 + 1\right)^2 = 3 + 2\sqrt3 + 1 = 4 + 2\sqrt3$.

$\left(2\sqrt5 - 3\right)^2 = 4 \times 5 - 2 \times 3 \times 2\sqrt5 + 9
= 29 - 12\sqrt5$.

$\left(\sqrt6 + \sqrt2\right)\left(\sqrt6 - \sqrt2\right) = 6 - 2 = 4$.
\end{solution}

\begin{solution}{exo:g9:sqrt:7}
Zijde: $\sqrt{6400} = 80$ m. Diagonaal (Pythagoras):
$\sqrt{80^2 + 80^2} = 80\sqrt2 \approx 113$ m.
\end{solution}

\begin{solution}{exo:g9:sqrt:8}
$\dfrac{1}{\sqrt2} = \dfrac{1 \times \sqrt2}{\sqrt2 \times \sqrt2}
= \dfrac{\sqrt2}{2}$.

$\dfrac{6}{\sqrt3} = \dfrac{6\sqrt3}{3} = 2\sqrt3$.
\quad
$\dfrac{10}{\sqrt5} = \dfrac{10\sqrt5}{5} = 2\sqrt5$.
\end{solution}

\begin{solution}{exo:g9:sqrt:9}
\emph{1. Waar}: dat is de productregel (\cref{thm:g9:sqrt:rules}).

\emph{2. Niet waar}: $\sqrt{9 + 16} = 5$, maar $\sqrt9 + \sqrt{16} = 7$.

\emph{3. Niet waar} voor negatieve $x$: $\sqrt{(-4)^2} = 4 \neq -4$. De juiste
uitspraak is $\sqrt{x^2} = \abs{x}$.

\emph{4. Waar}: $\left(3\sqrt2\right)^2 = 9 \times 2 = 18$.
\end{solution}

\begin{solution}{exo:g9:sqrt:10}
$\left(2 + \sqrt3\right)^2 = 4 + 4\sqrt3 + 3 = 7 + 4\sqrt3$. Dus is
$x = \sqrt{7 + 4\sqrt3} = \sqrt{\left(2+\sqrt3\right)^2} = 2 + \sqrt3$ (een
niet-negatief getal, zodat de wortel enkel het kwadraat weghaalt).

Net zo is $\left(\sqrt5 - 2\right)^2 = 5 - 4\sqrt5 + 4 = 9 - 4\sqrt5$, en
$\sqrt5 - 2 > 0$, dus is $\sqrt{9 - 4\sqrt5} = \sqrt5 - 2$ (niet
$2 - \sqrt5$, want dat is negatief!).
\end{solution}

\begin{solution}{pb:g9:sqrt:1}
\textbf{1.} $1.4^2 = 1.96 < 2 < 2.25 = 1.5^2$;
$1.41^2 = 1.9881 < 2 < 2.0164 = 1.42^2$. Met drie decimalen:
$1.414^2 = 1.999396 < 2 < 2.002225 = 1.415^2$, dus is
$1.414 < \sqrt2 < 1.415$.

\textbf{2.} $\sqrt2 = \frac ab$ kwadrateren geeft $2 = \frac{a^2}{b^2}$, en dus
$a^2 = 2b^2$: het getal $a^2$ is twee keer een heel getal --- \emph{\omterm{def:g3:numbers:evenodd}{even}}.

\textbf{3.} Was $a$ \omterm{def:g3:numbers:evenodd}{oneven}, dan zou $a^2$ \omterm{def:g3:numbers:evenodd}{oneven} zijn
(\cref{pb:g8:pythagoras:1}); omdat $a^2$ \omterm{def:g3:numbers:evenodd}{even} is, is $a$ \omterm{def:g3:numbers:evenodd}{even}: $a = 2k$.
Invullen geeft $4k^2 = 2b^2$, dus is $b^2 = 2k^2$ \omterm{def:g3:numbers:evenodd}{even}, en volgens hetzelfde feit
over pariteit is $b$ \omterm{def:g3:numbers:evenodd}{even}.

\textbf{4.} $a$ en $b$ beide \omterm{def:g3:numbers:evenodd}{even} betekent: beide \omterm{def:g6:wholes:divisible}{deelbaar} door $2$ --- maar
$\frac ab$ stond in eenvoudigste vorm en had geen gemeenschappelijke \omterm{def:g9:arith:divisor}{deler}.
Tegenspraak: de veronderstelde \omterm{def:g9:fractions:fraction}{breuk} kan niet bestaan. \emph{Stelling: $\sqrt2$ is
irrationaal.}

\textbf{5.} De tegenspraak zit volledig in ``in eenvoudigste vorm'': zonder die
woorden spreekt ``$a$ en $b$ beide \omterm{def:g3:numbers:evenodd}{even}'' niets tegen. De eenvoudigste vorm
aannemen mag, doordat elke \omterm{def:g9:fractions:fraction}{breuk} zo'n vorm \emph{heeft}
(\cref{met:g9:arith:simplify}: deel de ggd eruit) --- was $\sqrt2$ hoe dan ook een
\omterm{def:g9:fractions:fraction}{breuk}, dan zou het een \omterm{def:g9:fractions:fraction}{breuk} in eenvoudigste vorm zijn, en die is onmogelijk.

\textbf{6.} In priemfactorontbindingen dragen kwadraten \omterm{def:g3:numbers:evenodd}{even} \omterm{def:g8:powers:def}{exponenten}
(\cref{pb:g9:arith:1}). In $a^2 = 3b^2$ is de \omterm{def:g8:powers:def}{exponent} van $3$ \omterm{def:g3:numbers:evenodd}{even} in het
linkerlid, maar \omterm{def:g3:numbers:evenodd}{oneven} in het rechterlid (\omterm{def:g3:numbers:evenodd}{even} uit $b^2$, plus één). Geen enkel
getal heeft twee ontbindingen (\cref{thm:g9:arith:factorization}): tegenspraak, en
$\sqrt3$ is irrationaal.

\textbf{7.} In $a^2 = n b^2$ vindt het argument precies dan een \omterm{def:g9:arith:prime}{priemgetal} met
tegenstrijdige pariteit van de \omterm{def:g8:powers:def}{exponent}, wanneer een \omterm{def:g9:arith:prime}{priemgetal} in $n$ met een
\emph{\omterm{def:g3:numbers:evenodd}{oneven}} \omterm{def:g8:powers:def}{exponent} voorkomt. Zijn alle \omterm{def:g8:powers:def}{exponenten} van $n$ \omterm{def:g3:numbers:evenodd}{even}, dan is $n$ een
volkomen kwadraat ($n = m^2$, en $\sqrt n = m$ is heel). Volledig resultaat:
$\sqrt n$ is irrationaal voor elk heel getal $n$ dat geen volkomen kwadraat is ---
$\sqrt2, \sqrt3, \sqrt5, \sqrt6, \sqrt7, \sqrt8, \sqrt{10}, \dots$

\textbf{8.} Uit $r x = q$ met $r \neq 0$ volgt $x = \frac qr$, een \omterm{def:g3:division:remainder}{quotiënt} van
rationale getallen, en dus rationaal. Is $x$ dus irrationaal, dan kan $r x$ niet
rationaal zijn: $2\sqrt2$ en $\frac{\sqrt2}{2}$ zijn irrationaal (met
vermenigvuldigers $2$ en $\frac12$).

\textbf{9.} Was $1 + \sqrt2 = q$ rationaal, dan zou $\sqrt2 = q - 1$ rationaal
zijn: tegenspraak. Was $s = \sqrt2 + \sqrt3$ rationaal, dan zou
$s^2 = 2 + 2\sqrt6 + 3 = 5 + 2\sqrt6$ dat ook zijn, wat
$\sqrt6 = \frac{s^2 - 5}{2}$ rationaal maakt --- maar $\sqrt6$ is irrationaal
(vraag 7). Dus is $\sqrt2 + \sqrt3$ irrationaal.

\textbf{10.} Sommen: $\sqrt2$ en $-\sqrt2$ (of $\sqrt2$ en $1 - \sqrt2$) zijn elk
irrationaal, met de rationale sommen $0$ en $1$. \omterm{def:g2:mult:def}{Producten}:
$\sqrt2 \times \sqrt2 = 2$. Irrationaliteit kan in het rekenen verdampen ---
vandaar de bewijzen geval per geval.

\textbf{11.} Vanuit $x = 1$: het gemiddelde van $1$ en $\frac21 = 2$ is
$x_1 = \frac32$. Daarna het gemiddelde van $\frac32$ en
$\frac{2}{3/2} = \frac43$:
$x_2 = \frac12\left(\frac32 + \frac43\right)
= \frac12 \cdot \frac{17}{6} = \frac{17}{12} \approx 1.4167$.

\textbf{12.} $x_3 = \frac12\left(\frac{17}{12} + \frac{24}{17}\right)
= \frac12 \cdot \frac{289 + 288}{204} = \frac{577}{408} = 1.4142156\ldots$
Tegenover $\sqrt2 = 1.4142135\ldots$: vijf juiste decimalen na drie stappen ---
het recept verdubbelt het aantal juiste decimalen ruwweg bij elke ronde.

\textbf{13.} Is $x^2 > 2$ (schatting te groot), dan is $x > \sqrt2$ en
$\frac2x < \frac{2}{\sqrt2} = \sqrt2$: de bijhorende zijde is te klein. Is
$x^2 < 2$, dan keren de ongelijkheden om. In beide gevallen ligt $\sqrt2$ tussen
$x$ en $\frac2x$, zodat hun gemiddelde --- de volgende schatting --- dichterbij
ligt dan de slechtste van de twee zijden: de rechthoek met \omterm{def:g6:measure:area}{oppervlakte} $2$ wordt
bij elke stap vierkanter.

\textbf{14.} $9 - 8 = 1$; \ $289 - 288 = 1$; \ $332\,929 - 332\,928 = 1$. Elke
schatting $\frac ab$ van Heron voldoet aan $a^2 - 2b^2 = 1$: deel I bewees dat
$0$ treffen onmogelijk is, en deze \omterm{def:g9:fractions:fraction}{breuken} missen het onmogelijke met precies één
eenheid --- dichterbij kunnen hele getallen nooit komen.

\textbf{15.} De getallenlijn moet getallen bevatten die verder gaan dan de \omterm{def:g9:fractions:fraction}{breuken}
--- lengten als $\sqrt2$, waarvan de decimale schrijfwijze eindeloos doorloopt
zonder te repeteren: de \emph{reële getallen}. Hun eerlijke opbouw is een verhaal
voor het bovenbouwvolume en, in volle strengheid, voor de universitaire \omterm{def:g6:measure:volume}{volumes}.
\end{solution}
