\section*{Hoofdstuk \ref{ch:g9:arith} --- Getaltheorie: delers en priemgetallen}

\begin{solution}{exo:g9:arith:1}
\omterm{def:g9:arith:divisor}{Delers} van $36$: $1, 2, 3, 4, 6, 9, 12, 18, 36$.
\omterm{def:g9:arith:divisor}{Delers} van $45$: $1, 3, 5, 9, 15, 45$.
\omterm{def:g9:arith:divisor}{Delers} van $17$: alleen $1$ en $17$ ($17$ is priem).
\end{solution}

\begin{solution}{exo:g9:arith:2}
$2\,346$ eindigt op $6$: \omterm{def:g6:wholes:divisible}{deelbaar} door $2$, niet door $5$. Cijfersom
$2 + 3 + 4 + 6 = 15$: \omterm{def:g6:wholes:divisible}{deelbaar} door $3$, niet door $9$. Laatste twee cijfers
$46$, en $46 = 4 \times 11 + 2$ is niet \omterm{def:g6:wholes:divisible}{deelbaar} door $4$: $2\,346$ is niet
\omterm{def:g6:wholes:divisible}{deelbaar} door $4$.
\end{solution}

\begin{solution}{exo:g9:arith:3}
$72 = 2^3 \times 3^2$; \quad
$150 = 2 \times 3 \times 5^2$; \quad
$210 = 2 \times 3 \times 5 \times 7$; \quad

\end{solution}

\begin{solution}{exo:g9:arith:4}
$101$: toets de priemgetallen $p$ met $p^2 \leq 101$, dat wil zeggen
$2, 3, 5, 7$. Geen van hen \omterm{def:g9:arith:divisor}{deelt} $101$ (\omterm{def:g3:numbers:evenodd}{oneven}, cijfersom $2$, eindigt niet op
$0$ of $5$, en $101 = 7 \times 14 + 3$): $101$ is priem.

$91 = 7 \times 13$: niet priem.

$143 = 11 \times 13$: niet priem.
\end{solution}

\begin{solution}{exo:g9:arith:5}
Gemeenschappelijke \omterm{def:g9:arith:divisor}{delers} van $48$ en $60$: de \omterm{def:g9:arith:divisor}{delers} van $48$ zijn
$1, 2, 3, 4, 6, 8, 12, 16, 24, 48$; de \omterm{def:g9:arith:divisor}{delers} van $60$ zijn
$1, 2, 3, 4, 5, 6, 10, 12, 15, 20, 30, 60$; de gemeenschappelijke zijn
$1, 2, 3, 4, 6, 12$, dus is $\gcd(48,60) = 12$.

Met de ontbinding: $48 = 2^4 \times 3$ en $60 = 2^2 \times 3 \times 5$;
gemeenschappelijke priemgetallen met de kleinste \omterm{def:g8:powers:def}{exponenten}:
$2^2 \times 3 = 12$.
\end{solution}

\begin{solution}{exo:g9:arith:6}
$\gcd(255, 154)$:
\begin{align*}
255 &= 154 \times 1 + 101, \\
154 &= 101 \times 1 + 53, \\
101 &= 53 \times 1 + 48, \\
53 &= 48 \times 1 + 5, \\
48 &= 5 \times 9 + 3, \\
5 &= 3 \times 1 + 2, \\
3 &= 2 \times 1 + 1, \\
2 &= 1 \times 2 + 0 .
\end{align*}
Laatste \omterm{def:g3:division:remainder}{rest} verschillend van \omterm{def:g1:counting:zero}{nul}: $\gcd(255, 154) = 1$ (ze zijn \omterm{def:g9:arith:gcd}{relatief priem}).

$\gcd(1053, 325)$:
\begin{align*}
1053 &= 325 \times 3 + 78, \\
325 &= 78 \times 4 + 13, \\
78 &= 13 \times 6 + 0 .
\end{align*}
$\gcd(1053, 325) = 13$.
\end{solution}

\begin{solution}{exo:g9:arith:7}
Algoritme van Euclides: $588 = 504 \times 1 + 84$; $504 = 84 \times 6 + 0$:
$\gcd(588, 504) = 84$. Dan is
\[
\frac{588}{504} = \frac{588 \div 84}{504 \div 84} = \frac{7}{6},
\]
en dat is onvereenvoudigbaar.
\end{solution}

\begin{solution}{exo:g9:arith:8}
Het aantal boeketten moet zowel $84$ als $126$ delen; het grootst mogelijke is
$\gcd(84, 126)$. Ontbindingen: $84 = 2^2 \times 3 \times 7$,
$126 = 2 \times 3^2 \times 7$, dus is de ggd $2 \times 3 \times 7 = 42$. Ze kan
$42$ boeketten maken, met in elk $\frac{84}{42} = 2$ rozen en
$\frac{126}{42} = 3$ tulpen.
\end{solution}

\begin{solution}{exo:g9:arith:9}
We hebben het kleinste gemeenschappelijke \omterm{def:g9:arith:divisor}{veelvoud} nodig. $24 = 2^3 \times 3$ en
$36 = 2^2 \times 3^2$; neem je elk \omterm{def:g9:arith:prime}{priemgetal} met de \emph{grootste} \omterm{def:g8:powers:def}{exponent},
dan is $\lcm = 2^3 \times 3^2 = 72$. De veerboten vertrekken de volgende keer
samen $72$ minuten na 8:00, om 9:12.
\end{solution}

\begin{solution}{exo:g9:arith:10}
\emph{1.} Elke gemeenschappelijke \omterm{def:g9:arith:divisor}{deler} $d$ van $n$ en $n+1$ \omterm{def:g9:arith:divisor}{deelt} ook hun
\omterm{ex:g1:subtraction:difference}{verschil} $(n+1) - n = 1$, dus is $d = 1$: $\gcd(n, n+1) = 1$.

\emph{2.} Een \omterm{def:g9:fractions:fraction}{breuk} is onvereenvoudigbaar precies wanneer haar \omterm{def:g4:fractions:def}{teller} en \omterm{def:g4:fractions:def}{noemer}
\omterm{def:g9:arith:gcd}{relatief priem} zijn, en dat is volgens deel 1 het geval voor $n$ en $n + 1$.
\end{solution}

\begin{solution}{pb:g9:arith:1}
\textbf{1.} Vul de kan van $3$ en giet die in de kan van $5$ (\omterm{def:g2:measure:units}{inhoud}
$0/3 \to 3$ in de grote). Vul de kan van $3$ opnieuw en giet in de kan van $5$
tot die vol is: de grote kan neemt er nog maar $2$ op, zodat er
\[
3 - 2 = 1 \text{ L in de kleine kan blijft.}
\]
Zetten: vul $3$; giet $3 \to 5$; vul $3$; giet $3 \to 5$.

\textbf{2.} Vul de kan van $5$; giet in die van $3$ (er blijft $2$ in de grote);
maak die van $3$ leeg; giet de $2$ in die van $3$; vul die van $5$; giet in die
van $3$ tot ze vol is --- er gaat $1$ in, zodat er $\mathbf{4}$ L in de grote kan
blijft. Zes zetten.

\textbf{3.} Alle: $1$ (vraag 1), $2$ (na twee zetten van vraag 2), $3$ en $5$
(één vulling), $4$ (vraag 2), $6 = 3 + 3$ (een volle kleine kan plus $3$ in de
grote gegoten), $7 = 5 + 2$, $8 = 5 + 3$ (beide vol). Elke hele hoeveelheid van
$1$ tot $8$ L is met de kan van $5$ en die van $3$ af te meten.

\textbf{4.} Bij het begin bevatten beide kannen $0$, een \omterm{def:g9:arith:divisor}{veelvoud} van $2$. Vullen
zet een \omterm{def:g2:measure:units}{inhoud} op $6$ of $4$: \omterm{def:g3:numbers:evenodd}{even}. Leegmaken zet ze op $0$: \omterm{def:g3:numbers:evenodd}{even}. Gieten
verplaatst wat water tussen kannen waarvan de \omterm{def:g2:measure:units}{inhoud} \omterm{def:g3:numbers:evenodd}{even} was, en de gegoten
hoeveelheid is een \omterm{ex:g1:subtraction:difference}{verschil} van \omterm{def:g3:numbers:evenodd}{even} getallen (de overblijvende ruimte, of de
beschikbare hoeveelheid): alle \omterm{def:g2:measure:units}{inhouden} blijven dus eeuwig \omterm{def:g3:numbers:evenodd}{even}. Een \omterm{def:g3:numbers:evenodd}{oneven} doel
zoals $1$ L is onbereikbaar.

\textbf{5.} $\gcd(6, 4) = 2$: alleen \omterm{def:g3:numbers:evenodd}{even} hoeveelheden --- bevestigd door vraag
4. $\gcd(5, 3) = 1$: de wet laat elke hele hoeveelheid toe, en vraag 3 heeft ze
alle verwezenlijkt. De ggd is precies de maateenheid van de kannen.

\textbf{6.} $91 = 1 \times 65 + 26$; $65 = 2 \times 26 + 13$;
$26 = 2 \times 13 + 0$: $\gcd(91, 65) = 13$. En $2\,026 = 44 \times 46 + 2$;
$46 = 23 \times 2 + 0$: $\gcd(2\,026, 46) = 2$.

\textbf{7.} Herhaaldelijk $5$ in de kan van $13$ gieten: na twee vullingen bevat
de grote kan $10$; van de derde vulling gaat er maar $3$ in, zodat er
$5 - 3 = 2$ in de kleine kan blijft --- de \omterm{def:g3:division:remainder}{rest} van $13$ bij deling door $5$ was
$3$, en de hoeveelheden $3$ (de ruimte) en $2$ (wat overblijft) zijn precies de
getallen van Euclides ($13 = 2 \times 5 + 3$, $5 = 1 \times 3 + 2$). Ga je door,
dan duikt $3 - 2 = 1$ op: de volgende \omterm{def:g3:division:remainder}{rest} van het algoritme. De fontein voert de
delingen van Euclides met water uit.

\textbf{8.} $\gcd(13, 5) = 1$, dus laat de wet van vraag 5 elke hele hoeveelheid
toe --- en de cascade van vraag 7 bracht $1$ L werkelijk voort. Ja.

\textbf{9.} Een gemeenschappelijke \omterm{def:g9:arith:divisor}{deler} van $n$ en $2n + 1$ \omterm{def:g9:arith:divisor}{deelt}
$2n + 1 - 2 \times n = 1$: hij moet $1$ zijn. Bijgevolg is $\gcd(n, 2n+1) = 1$,
altijd.

\textbf{10.} Bus A: 7:12, 7:24, 7:36, 7:48, 8:00\,\dots; bus B: 7:18, 7:36,
7:54\,\dots\ Eerste gemeenschappelijke vertrek: 7:36, na $36$ minuten --- het
eerste gemeenschappelijke \omterm{def:g9:arith:divisor}{veelvoud} van $12$ en $18$. De wet:
$36 \times \gcd(12, 18) = 36 \times 6 = 216 = 12 \times 18$. Voor $5$ en $3$:
eerste gemeenschappelijke \omterm{def:g9:arith:divisor}{veelvoud} $15$, en
$15 \times \gcd(5,3) = 15 \times 1 = 15 = 5 \times 3$.

\textbf{11.} Met een cyclus van $17$ jaar: de volgende samenloop is het eerste
gemeenschappelijke \omterm{def:g9:arith:divisor}{veelvoud} van $17$ en $4$; omdat $\gcd(17, 4) = 1$, is dat
$17 \times 4 = 68$ jaar --- de cicaden ontmoeten de piek één keer op vier
opkomsten. Met een cyclus van $16$ jaar: $16$ is een \omterm{def:g9:arith:divisor}{veelvoud} van $4$, dus treft
\emph{elke} opkomst een piek. Een priemlengte van de cyclus heeft met geen enkele
kortere cyclus van een roofdier een factor gemeen, en drijft de samenlopen zo ver
mogelijk uit elkaar: getaltheorie als camouflage.

\textbf{12.} Vier keuzen van \omterm{def:g8:powers:def}{exponent} voor $2$, drie voor $3$, twee voor $5$:
$4 \times 3 \times 2 = 24$ \omterm{def:g9:arith:divisor}{delers}.

\textbf{13.} Is $m = 2^{a} \times 3^{b} \times \cdots$, dan is
$m^2 = 2^{2a} \times 3^{2b} \times \cdots$: elke \omterm{def:g8:powers:def}{exponent} is verdubbeld, en dus
\omterm{def:g3:numbers:evenodd}{even}. In $360 = 2^3 \times 3^2 \times 5$ zijn de \omterm{def:g8:powers:def}{exponenten} van $2$ en van $5$
\omterm{def:g3:numbers:evenodd}{oneven}: $360$ is geen volkomen kwadraat.

\textbf{14.} Een \omterm{def:g9:arith:divisor}{deler} is zijn eigen partner precies wanneer $d = \frac nd$, dat
wil zeggen $n = d^2$: alleen kwadraten hebben zo'n middelste \omterm{def:g9:arith:divisor}{deler}. Voor alle
andere $n$ vallen de \omterm{def:g9:arith:divisor}{delers} in paren uiteen, een \omterm{def:g3:numbers:evenodd}{even} aantal. Dus: \omterm{def:g3:numbers:evenodd}{oneven} aantal
\omterm{def:g9:arith:divisor}{delers} $\Leftrightarrow$ volkomen kwadraat. Controle: $36$ heeft de \omterm{def:g9:arith:divisor}{delers}
$1, 2, 3, 4, 6, 9, 12, 18, 36$ --- negen ervan, \omterm{def:g3:numbers:evenodd}{oneven}, en $36 = 6^2$; terwijl
$360$ er $24$ heeft (vraag 12), \omterm{def:g3:numbers:evenodd}{even}, en geen kwadraat is (vraag 13).

\textbf{15.} De stand van kluisje $n$ wordt één keer veranderd door elke leerling
$k$ wiens nummer $n$ \omterm{def:g9:arith:divisor}{deelt}: in totaal dus even veel keer als $n$ \omterm{def:g9:arith:divisor}{delers} heeft.
Een kluisje eindigt \emph{open} wanneer zijn stand een \omterm{def:g3:numbers:evenodd}{oneven} aantal keer is
veranderd --- volgens vraag 14 precies wanneer $n$ een volkomen kwadraat is. De
open kluisjes: $1, 4, 9, 16, 25, 36, 49, 64, 81, 100$ --- tien ervan.
\end{solution}
