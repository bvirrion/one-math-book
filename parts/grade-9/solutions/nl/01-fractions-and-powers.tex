\section*{Hoofdstuk \ref{ch:g9:fractions} --- Breuken en machten}

\begin{solution}{exo:g9:fractions:1}
$\dfrac13 + \dfrac15 = \dfrac{5}{15} + \dfrac{3}{15} = \dfrac{8}{15}$.

$\dfrac72 - \dfrac53 = \dfrac{21}{6} - \dfrac{10}{6} = \dfrac{11}{6}$.

$\dfrac{5}{12} + \dfrac38 = \dfrac{10}{24} + \dfrac{9}{24} =
\dfrac{19}{24}$.

$3 - \dfrac45 = \dfrac{15}{5} - \dfrac45 = \dfrac{11}{5}$.
\end{solution}

\begin{solution}{exo:g9:fractions:2}
$\dfrac59 \times \dfrac{27}{35} = \dfrac{5 \times 27}{9 \times 35}
= \dfrac{27}{9} \times \dfrac{5}{35} = 3 \times \dfrac17 = \dfrac37$
(vereenvoudigd door $9$ en door $5$).

$\dfrac{8}{15} \div \dfrac45 = \dfrac{8}{15} \times \dfrac54
= \dfrac{8 \times 5}{15 \times 4} = \dfrac{40}{60} = \dfrac23$.

$\dfrac23 \times \dfrac34 \times \dfrac45 = \dfrac{2 \times 3 \times 4}
{3 \times 4 \times 5} = \dfrac25$ (de $3$'s en de $4$'s vallen weg).
\end{solution}

\begin{solution}{exo:g9:fractions:3}
$2^5 = 32$; \quad $(-2)^4 = 16$; \quad $-2^4 = -16$; \quad
$4^{-2} = \frac{1}{16}$; \quad
$\left(\frac23\right)^3 = \frac{8}{27}$; \quad $7^0 = 1$.
\end{solution}

\begin{solution}{exo:g9:fractions:4}
$5^3 \times 5^6 = 5^9$; \quad $\dfrac{7^9}{7^5} = 7^4$; \quad
$\left(2^3\right)^4 = 2^{12}$; \quad
$\dfrac{3^2 \times 3^7}{3^5} = 3^{9-5} = 3^4$; \quad
$2^5 \times 4 = 2^5 \times 2^2 = 2^7$.
\end{solution}

\begin{solution}{exo:g9:fractions:5}
$45\,000 = 4.5 \times 10^4$; \quad
$0.0072 = 7.2 \times 10^{-3}$; \quad
$302.5 = 3.025 \times 10^2$; \quad
$0.000\,000\,9 = 9 \times 10^{-7}$; \quad
twaalf miljoen $= 1.2 \times 10^7$.
\end{solution}

\begin{solution}{exo:g9:fractions:6}
$(2 \times 10^7) \times (4.5 \times 10^{-3}) = 9 \times 10^{4}$.

$\dfrac{9 \times 10^5}{3 \times 10^{-2}} = 3 \times 10^{5 - (-2)}
= 3 \times 10^7$.

$(5 \times 10^3)^2 = 25 \times 10^6 = 2.5 \times 10^7$.
\end{solution}

\begin{solution}{exo:g9:fractions:7}
Het toegevoegde water is
$\dfrac34 - \dfrac35 = \dfrac{15}{20} - \dfrac{12}{20} = \dfrac{3}{20}$
van de tank. Dus is $\frac{3}{20}$ van de \omterm{def:g2:measure:units}{inhoud} $30$ liter: $\frac{1}{20}$ is
$10$ liter, en de \omterm{def:g2:measure:units}{inhoud} is $200$ liter.
\end{solution}

\begin{solution}{exo:g9:fractions:8}
Na Alice blijft $1 - \frac14 = \frac34$ van de cake over. Ben eet
$\frac13 \times \frac34 = \frac14$ op, waardoor $\frac34 - \frac14 = \frac12$
overblijft. Carla eet daarvan de \omterm{def:g3:division:half}{helft} op, $\frac14$, waardoor $\frac14$ van de
cake overblijft.
\end{solution}

\begin{solution}{exo:g9:fractions:9}
Tijd $= \dfrac{\text{afstand}}{\text{snelheid}} =
\dfrac{1.5 \times 10^{11}}{3 \times 10^8} = 0.5 \times 10^3 = 500$ seconden, dat
wil zeggen $8$ minuten en $20$ seconden.
\end{solution}

\begin{solution}{exo:g9:fractions:10}
Schrijf beide getallen als \omterm{def:g9:fractions:power}{machten} met \omterm{def:g8:powers:def}{exponent} $10$:
\[
2^{30} = \left(2^3\right)^{10} = 8^{10},
\qquad
3^{20} = \left(3^2\right)^{10} = 9^{10}.
\]
Omdat $8 < 9$, geldt na het vermenigvuldigen van tien exemplaren van elk:
$8^{10} < 9^{10}$, dus $2^{30} < 3^{20}$.
\end{solution}

\begin{solution}{pb:g9:fractions:1}
\textbf{1.} $\frac14 = 0.25$ (stopt); $\frac13 = 0.333\ldots$ (blok $3$);
$\frac56 = 0.8333\ldots$ (blok $3$ na de $8$); $\frac{3}{11} = 0.272727\ldots$
(blok $27$).

\textbf{2.} Bij elke stap van de deling door $b$ is de \omterm{def:g3:division:remainder}{rest} een van de $b$
waarden $0, 1, \dots, b - 1$ (\cref{thm:g6:wholes:division}). Duikt er een \omterm{def:g3:division:remainder}{rest}
$0$ op, dan stopt de deling. Zo niet, dan moet na ten hoogste $b$ stappen een
\omterm{def:g3:division:remainder}{rest} voor de tweede keer opduiken --- er zijn meer stappen dan mogelijke \omterm{def:g3:division:remainder}{resten}
--- en vanaf dat ogenblik herhaalt de deling precies de rij cijfers die ze na de
eerste keer voortbracht: de decimale schrijfwijze draait eeuwig rond. Stoppen of
repeteren: een derde gedrag bestaat niet.

\textbf{3.} $\frac17 = 0.142857\,142857\ldots$: het blok $142857$ heeft lengte
$6$. Vraag 2 beloofde een herhaling binnen ten hoogste $7$ stappen --- hier duiken
de \omterm{def:g3:division:remainder}{resten} $1, 3, 2, 6, 4, 5$ alle op vóór de $1$ terugkeert: de langst mogelijke
dreun voor een deling door $7$.

\textbf{4.} $\frac ab$ stopt precies wanneer een zekere
$\frac{a \times k}{b \times k}$ als \omterm{def:g4:fractions:def}{noemer} $10$, $100$, $1000, \dots$ heeft ---
dat wil zeggen wanneer de \omterm{def:g4:fractions:def}{noemer} tot een \omterm{def:g9:fractions:power}{macht} van tien kan worden
vermenigvuldigd, zoals $4 \times 25 = 100$ of $8 \times 125 = 1\,000$. Voor $3$ en
$6$ is dat hopeloos: een \omterm{def:g9:fractions:power}{macht} van tien heeft cijfersom $1$, en elk \omterm{def:g5:division:multiple}{veelvoud} van
$3$ heeft een cijfersom die een \omterm{def:g5:division:multiple}{veelvoud} van $3$ is --- dus is geen enkel \omterm{def:g5:division:multiple}{veelvoud}
van $3$ (en dus ook niet van $6$) ooit $10$, $100$, $1\,000, \dots$ Voor $11$:
$10$, $100$, $1\,000$ en $10\,000$ door $11$ delen laat altijd een \omterm{def:g3:division:remainder}{rest} ($10$,
$1$, $10$, $1, \dots$), nooit $0$. \omterm{def:g4:fractions:def}{Noemers} die een \omterm{def:g9:fractions:power}{macht} van tien bereiken,
stoppen; alle andere repeteren.

\textbf{5.} $\frac{9}{40}$: $40 \times 25 = 1\,000$, stopt:
$\frac{225}{1000} = 0.225$. $\frac{33}{50}$: $50 \times 2 = 100$, stopt:
$\frac{66}{100} = 0.66$. $\frac{5}{12}$: $12$ is een \omterm{def:g5:division:multiple}{veelvoud} van $3$, repeteert
($0.41666\ldots$).

\textbf{6.} $10x = 7.777\ldots$, dus is $10x - x = 7$ (de staarten vallen precies
tegen elkaar weg), $9x = 7$ en $x = \frac79$. Controle door te delen:
$7 \div 9 = 0.777\ldots$

\textbf{7.} Het blok heeft twee cijfers, dus verschuif je met $100$:
$100x - x = 72.7272\ldots - 0.7272\ldots = 72$, en dus is $99x = 72$ en
$x = \frac{72}{99} = \frac{8}{11}$.

\textbf{8.} $1000x = 583.333\ldots$ en $100x = 58.333\ldots$; aftrekken geeft
$900x = 525$, dus $x = \frac{525}{900} = \frac{7}{12}$ (deel door $75$). En
inderdaad is $\frac{7}{12} = 0.58333\ldots$

\textbf{9.} $10x - x = 9.999\ldots - 0.999\ldots = 9$, dus is $9x = 9$ en
$x = 1$. Niet ``net onder'' $1$: $0.999\ldots$ \emph{is} het getal $1$, in een
tweede kostuum geschreven. Elk getal strikt onder $1$ laat een gaatje open, en
\cref{pb:g6:decimals:1} liet zien dat in een gaatje altijd nog getallen zitten
--- terwijl er tussen $0.999\ldots$ en $1$ niets past. De eindeloze negens zijn de
chocoladerest van \cref{pb:g6:fractions:1}, gekrompen onder elke positieve
hoeveelheid: er blijft niets over.

\textbf{10.} $2.4545\ldots = 2 + \frac{45}{99} = 2 + \frac{5}{11} =
\frac{27}{11}$. En $0.123123\ldots = \frac{123}{999} = \frac{41}{333}$ (deel door
$3$).

\textbf{11.} Volgens het woordenboek: $0.037037\ldots = \frac{37}{999}$. Omdat
$999 = 27 \times 37$, vereenvoudigt dat tot $\frac{1}{27}$: de \omterm{def:g9:fractions:fraction}{breuk}
$\frac{1}{27}$ dreunt ``$037$''. Controle door te delen:
$1 \div 27 = 0.037037\ldots$

\textbf{12.} $0.0272727\ldots = \frac{1}{10} \times 0.272727\ldots
= \frac{1}{10} \times \frac{27}{99} = \frac{27}{990} = \frac{3}{110}$ (deel door
$9$).

\textbf{13.} De blokken nullen groeien eindeloos, dus kan geen vast blok eeuwig
repeteren: het is \emph{geen} repeterende decimale schrijfwijze (en ze stopt ook
nooit). Volgens deel I stopt of repeteert elke \omterm{def:g9:fractions:fraction}{breuk} --- dus is dit getal
helemaal geen \omterm{def:g9:fractions:fraction}{breuk}: het is irrationaal, op maat gemaakt. De superstar van de
irrationaliteit, $\sqrt2$, wordt in \cref{ch:g9:sqrt} ontmaskerd.

\textbf{14.} $2^{30} = \left(2^{10}\right)^3 \approx (1.024)^3 \times 10^9
\approx 1.07 \times 10^9$: een getal net boven $10^9$, en dus met $10$ cijfers.
(Precies: $2^{30} = 1\,073\,741\,824$.)

\textbf{15.} $0.20262026\ldots = \frac{2026}{9999}$, al in eenvoudigste vorm
($9999 = 3 \times 3 \times 11 \times 101$ heeft geen factor gemeen met
$2026 = 2 \times 1013$). Het afgemaakte woordenboek: \emph{de \omterm{def:g9:fractions:fraction}{breuken} zijn precies
de decimale schrijfwijzen die stoppen of repeteren} --- \omterm{def:g4:fractions:def}{noemers} die vriendjes van
tien zijn stoppen, alle andere dreunen een blok; en omgekeerd is elke dreun, zelfs
je afstudeerjaar, een vermomde \omterm{def:g9:fractions:fraction}{breuk}.
\end{solution}
