\section*{Hoofdstuk \ref{ch:g9:algebra} --- Algebra en vergelijkingen}

\begin{solution}{exo:g9:algebra:1}
$4(3x - 2) = 12x - 8$.

$(x+6)(x+2) = x^2 + 2x + 6x + 12 = x^2 + 8x + 12$.

$(2x-3)(x+5) = 2x^2 + 10x - 3x - 15 = 2x^2 + 7x - 15$.

$(x-1)(x-9) = x^2 - 9x - x + 9 = x^2 - 10x + 9$.
\end{solution}

\begin{solution}{exo:g9:algebra:2}
$(x+8)^2 = x^2 + 16x + 64$.

$(3x-4)^2 = 9x^2 - 24x + 16$.

$(x+10)(x-10) = x^2 - 100$.

$(5x+2)(5x-2) = 25x^2 - 4$.
\end{solution}

\begin{solution}{exo:g9:algebra:3}
$101^2 = (100+1)^2 = 10000 + 200 + 1 = 10201$.

$59^2 = (60-1)^2 = 3600 - 120 + 1 = 3481$.

$53 \times 47 = (50+3)(50-3) = 2500 - 9 = 2491$.
\end{solution}

\begin{solution}{exo:g9:algebra:4}
$9x + 12 = 3(3x + 4)$.

$x^2 - 64 = (x+8)(x-8)$.

$x^2 + 14x + 49 = (x+7)^2$.

$18x^2 - 6x = 6x(3x - 1)$.
\end{solution}

\begin{solution}{exo:g9:algebra:5}
$4x + 7 = 19$: $4x = 12$, dus $x = 3$.

$6x - 5 = 2x + 11$: $4x = 16$, dus $x = 4$.

$5(x-3) = 3x + 1$: $5x - 15 = 3x + 1$, dus $2x = 16$ en $x = 8$.
\end{solution}

\begin{solution}{exo:g9:algebra:6}
$(x+4)(3x-12) = 0$: $x = -4$ of $x = 4$.

$x^2 - 121 = (x-11)(x+11) = 0$: $x = 11$ of $x = -11$.

$(2x-1)^2 = 0$: de enige oplossing $x = \frac12$.

$x^2 = 8x$: $x(x - 8) = 0$, dus $x = 0$ of $x = 8$.
\end{solution}

\begin{solution}{exo:g9:algebra:7}
\omterm{ex:g1:subtraction:difference}{Verschil} van kwadraten met $a = 3x + 2$ en $b = 5$:
\[
E = (3x + 2)^2 - 5^2 = (3x + 2 + 5)(3x + 2 - 5) = (3x + 7)(3x - 3).
\]
Nulproductregel: $3x + 7 = 0$ geeft $x = -\frac73$, en $3x - 3 = 0$ geeft
$x = 1$.
\end{solution}

\begin{solution}{exo:g9:algebra:8}
$3x - 4 \geq 8$: $3x \geq 12$, dus $x \geq 4$ (gevuld bolletje bij $4$, arcering
naar rechts).

$5 - 2x > 1$: $-2x > -4$; delen door $-2$ keert het teken om: $x < 2$ (open
bolletje bij $2$, arcering naar links).

$4(x+1) \leq x - 5$: $4x + 4 \leq x - 5$, dus $3x \leq -9$ en $x \leq -3$ (gevuld
bolletje bij $-3$, arcering naar links).
\end{solution}

\begin{solution}{exo:g9:algebra:9}
Zij $x$ het getal: $3x + 7 = 5x - 9$. Dan is $16 = 2x$, dus $x = 8$. Controle:
$3 \times 8 + 7 = 31$ en $5 \times 8 - 9 = 31$.
\end{solution}

\begin{solution}{exo:g9:algebra:10}
\emph{1.} $(n+1)(n-1) = n^2 - 1$ (het derde merkwaardige \omterm{def:g2:mult:def}{product}). De buren van
$n$ zijn $n - 1$ en $n + 1$, dus is hun \omterm{def:g2:mult:def}{product} $n^2 - 1$: bijvoorbeeld
$24 \times 26 = 625 - 1 = 624$.

\emph{2.} Drie opeenvolgende gehele getallen kun je schrijven als $n - 1$, $n$,
$n + 1$. Hun \omterm{def:g1:addition:def}{som} is
\[
(n-1) + n + (n+1) = 3n ,
\]
een \omterm{def:g9:arith:divisor}{veelvoud} van $3$ (drie keer het middelste).
\end{solution}

\begin{solution}{pb:g9:algebra:1}
\textbf{1.} $(10a + b) - (10b + a) = 9a - 9b = 9(a - b)$: het \omterm{ex:g1:subtraction:difference}{verschil} tussen een
getal van twee cijfers en zijn omkering is altijd een \omterm{def:g9:arith:divisor}{veelvoud} van $9$. Controle:
$72 - 27 = 45 = 9 \times 5$, en $a - b = 7 - 2 = 5$.

\textbf{2.} Het rechterlid \omterm{def:g9:algebra:expand}{uitwerken} geeft
$9(11a + b) + a + b + c = 99a + 9b + a + b + c = 100a + 10b + c$. Het getal is dus
een \omterm{def:g9:arith:divisor}{veelvoud} van $9$ plus zijn cijfersom: bij deling door $9$ laten beide dezelfde
\omterm{def:g3:division:remainder}{rest}. In het bijzonder betekent \omterm{def:g3:division:remainder}{rest} $0$ voor het ene \omterm{def:g3:division:remainder}{rest} $0$ voor het andere: de
deelbaarheidsregel voor $9$, bewezen.

\textbf{3.} $9(11a + b)$ is ook een \omterm{def:g9:arith:divisor}{veelvoud} van $3$, dus laat dezelfde gelijkheid
zien dat het getal en zijn cijfersom bij deling door $3$ dezelfde \omterm{def:g3:division:remainder}{rest} laten: de
regel voor $3$.

\textbf{4.} Schrijf $47 = 9k + 2$ en $86 = 9l + 5$. Dan is
\[
47 \times 86 = (9k + 2)(9l + 5)
= 9\,(9kl + 5k + 2l) + 10 :
\]
het \omterm{def:g2:mult:def}{product} laat dezelfde \omterm{def:g3:division:remainder}{rest} als $2 \times 5 = 10$, namelijk $1$. Een juist
resultaat moet dus ook \omterm{def:g3:division:remainder}{rest} $1$ laten --- en dat doet $4\,042$. Maar de proef ziet
alleen \omterm{def:g3:division:remainder}{resten}: $4\,042$ en $4\,402$ (met verwisselde cijfers) doorstaan haar even
goed, dus bewijst doorstaan niets --- alleen zakken is bezwarend.

\textbf{5.} \omterm{def:g9:algebra:expand}{Uitwerken}: $11(91a + 9b + c) - a + b - c + d
= 1001a + 99b + 11c - a + b - c + d = 1000a + 100b + 10c + d$. Het getal is dus
precies dan \omterm{def:g6:wholes:divisible}{deelbaar} door $11$ wanneer zijn alternerende \omterm{def:g1:addition:def}{som} $-a + b - c + d$ dat
is. Voor $2\,926$: $-2 + 9 - 2 + 6 = 11$, \omterm{def:g6:wholes:divisible}{deelbaar} door $11$: ja,
$2\,926 = 11 \times 266$.

\textbf{6.} $742 - 247 = 495$; $495 + 594 = 1\,089$. Elk geldig voorbeeld landt
ook op $1\,089$.

\textbf{7.} $(100a + 10b + c) - (100c + 10b + a) = 99(a - c)$. Voor
$a - c = 2, 3, \dots, 9$:
\[
198,\ 297,\ 396,\ 495,\ 594,\ 693,\ 792,\ 891 .
\]

\textbf{8.} $100(m - 1) + 90 + (10 - m) = 100m - 100 + 90 + 10 - m = 99m$. De drie
cijfers van $99m$ zijn dus $m - 1$, dan $9$, dan $10 - m$ (controle op $495$:
$m = 5$ geeft $4$, $9$, $5$).

\textbf{9.} De omkering van $99m$ heeft de cijfers $10 - m$, $9$, $m - 1$, en is
dus gelijk aan $100(10 - m) + 90 + (m - 1)$. Optellen:
\[
100\,(m - 1 + 10 - m) + 180 + (10 - m + m - 1)
= 100 \times 9 + 180 + 9 = 1\,089 ,
\]
onafhankelijk van $m$. Wat het beginnetal ook is, de goochelaar zit veilig.

\textbf{10.} Is $a = c$, dan geeft de aftrekking $0$ en sterft de truc
onmiddellijk. Is $a - c = 1$, dan geeft ze $99$, dat als de rij van drie cijfers
$099$ behandeld moet worden (omkering $990$, en $99 + 990 = 1\,089$: gered!). De
kleine lettertjes: verschillen van $1$ werken alleen als de kijker het resultaat
van de aftrekking met drie cijfers opschrijft, voorloopnul inbegrepen --- een
\omterm{ex:g1:subtraction:difference}{verschil} van minstens $2$ eisen omzeilt de discussie.

\textbf{11.} $2(5m + 7) + d - 14 = 10m + 14 + d - 14 = 10m + d$: de tientallen (en
honderdtallen, voor oktober--december) geven de maand, de eenheden (en tientallen)
de dag.

\textbf{12.} $2(5m + 9) + d = 10m + 18 + d$: trek $18$ af.

\textbf{13.} Volgens vraag 2 is $n - (\text{cijfersom van } n) = 9(11a + b)$ voor
een $n$ van drie cijfers: altijd een \omterm{def:g9:arith:divisor}{veelvoud} van $9$ (voor elke lengte werken
soortgelijke gelijkheden). Een \omterm{def:g9:arith:divisor}{veelvoud} van $9$ heeft een cijfersom die een
\omterm{def:g9:arith:divisor}{veelvoud} van $9$ is (weer vraag 2), dus moeten de cijfers die de kijker voorleest
aangevuld worden tot het volgende \omterm{def:g9:arith:divisor}{veelvoud} van $9$: het ontbrekende cijfer is die
aanvulling. Dubbelzinnigheid: is de \omterm{def:g1:addition:def}{som} van de voorgelezen cijfers al een \omterm{def:g9:arith:divisor}{veelvoud}
van $9$, dan kan het doorgestreepte cijfer $0$ of $9$ zijn --- $0$ uitsluiten haalt
de twijfel weg.

\textbf{14.} $(10t + 5)^2 = 100t^2 + 100t + 25 = 100\,t(t + 1) + 25$: schrijf
$t(t+1)$, en plak er $25$ achter. $45^2$: $4 \times 5 = 20$, dus $2\,025$.
$85^2$: $8 \times 9 = 72$, dus $7\,225$. $105^2$: $10 \times 11 = 110$, dus
$11\,025$.

\textbf{15.} Vijf opeenvolgende: $n + (n+1) + (n+2) + (n+3) + (n+4) = 5n + 10
= 5(n + 2)$: altijd een \omterm{def:g9:arith:divisor}{veelvoud} van $5$. Vier opeenvolgende:
$n + (n+1) + (n+2) + (n+3) = 4n + 6 = 4(n + 1) + 2$: \omterm{def:g3:division:remainder}{rest} $2$ bij de deling door
$4$, dus nooit een \omterm{def:g9:arith:divisor}{veelvoud}. De algebra zet ``het lijkt te werken'' om in
\emph{altijd}, en ``ik vond geen voorbeeld'' in \emph{nooit} --- twee woorden die
geen enkele hoeveelheid proberen kan bereiken (\cref{met:g7:literal:test}).
\end{solution}
