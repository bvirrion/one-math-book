\section*{Hoofdstuk \ref{ch:g9:statproba} --- Statistiek en kansrekening}

\begin{solution}{exo:g9:statproba:1}
\omterm{def:g9:statproba:indicators}{Gemiddelde}: $\dfrac{14 + 9 + 17 + 9 + 12 + 11 + 12}{7} = \dfrac{84}{7} = 12$.

\omterm{def:g9:statproba:indicators}{Mediaan}: gesorteerd $9, 9, 11, 12, 12, 14, 17$; de middelste ($4$de) waarde is
$12$.

\omterm{def:g9:statproba:indicators}{Variatiebreedte}: $17 - 9 = 8$.
\end{solution}

\begin{solution}{exo:g9:statproba:2}
\omterm{def:g9:statproba:indicators}{Gemiddelde}: $\dfrac{19 + 21 + 24 + 18 + 22 + 25 + 19}{7} = \dfrac{148}{7}
\approx 21.1$ graden.

\omterm{def:g9:statproba:indicators}{Mediaan}: gesorteerd $18, 19, 19, 21, 22, 24, 25$; de $4$de waarde is $21$
graden.
\end{solution}

\begin{solution}{exo:g9:statproba:3}
\omterm{def:g9:statproba:indicators}{Gemiddelde}:
\[
\frac{2 \times 0 + 4 \times 1 + 3 \times 2 + 1 \times 3}{10}
= \frac{0 + 4 + 6 + 3}{10} = \frac{13}{10} = 1.3
\text{ doelpunten per wedstrijd.}
\]
\omterm{def:g9:statproba:indicators}{Mediaan}: de gesorteerde reeks is $0,0,1,1,1,1,2,2,2,3$; de $5$de en de $6$de
waarde zijn beide $1$: \omterm{def:g9:statproba:indicators}{mediaan} $1$.
\end{solution}

\begin{solution}{exo:g9:statproba:4}
$\P(3) = \frac16$. \quad
$\P(\text{even}) = \frac36 = \frac12$. \quad
$\P(\text{minstens } 3) = \P(\{3,4,5,6\}) = \frac46 = \frac23$. \quad
$\P(\text{geen } 6) = 1 - \frac16 = \frac56$.
\end{solution}

\begin{solution}{exo:g9:statproba:5}
$\P(\text{wit}) = \frac{5}{12}$.

$\P(\text{rood of groen}) = \frac{4 + 3}{12} = \frac{7}{12}$.

$\P(\text{niet wit}) = 1 - \frac{5}{12} = \frac{7}{12}$.

De laatste twee zijn gelijk: ``niet wit'' \emph{is} de gebeurtenis ``rood of
groen''.
\end{solution}

\begin{solution}{exo:g9:statproba:6}
De $12$ uitkomsten (zijde van de munt, waarde van de dobbelsteen) zijn even
waarschijnlijk.

Kop en een $6$: één uitkomst op $12$: $\frac{1}{12}$.

Kop of een $6$: de uitkomsten met kop ($6$ ervan) plus de uitkomst (munt, $6$):
$7$ uitkomsten, dus $\frac{7}{12}$. Elke gebeurtenis afzonderlijk tellen en de
overlap aftrekken geeft hetzelfde:
$\frac{6}{12} + \frac{2}{12} - \frac{1}{12} = \frac{7}{12}$.
\end{solution}

\begin{solution}{exo:g9:statproba:7}
Eerste trekking: rood $\frac25$, zwart $\frac35$. Tweede trekking zonder
teruglegging: na een rood blijven $1$ rood en $3$ zwarte over op $4$; na een zwart
$2$ rode en $2$ zwarte op $4$.

Twee zwarte: $\frac35 \times \frac24 = \frac{6}{20} = \frac{3}{10}$.

Verschillende kleuren (RZ of ZR):
$\frac25 \times \frac34 + \frac35 \times \frac24
= \frac{6}{20} + \frac{6}{20} = \frac{12}{20} = \frac35$.
\end{solution}

\begin{solution}{exo:g9:statproba:8}
\emph{1.} $9 \times 14 = 126$ punten.

\emph{2.} Een \omterm{def:g9:statproba:indicators}{gemiddelde} van $15$ over $10$ wedstrijden betekent $150$ punten in
totaal: ze moet $150 - 126 = 24$ punten maken.
\end{solution}

\begin{solution}{exo:g9:statproba:9}
\emph{1.} Van de $36$ even waarschijnlijke paren zijn de dubbels
$(1,1), (2,2), \dots, (6,6)$: zes ervan, dus
$\P(\text{dubbel}) = \frac{6}{36} = \frac16$.

\emph{2.} Twee onafhankelijke spelletjes: een boom met winst ($\frac16$) en verlies
($\frac56$) op elk niveau. \omterm{def:g9:statproba:probability}{Kans} om nooit te winnen:
$\frac56 \times \frac56 = \frac{25}{36}$. Dus is
\[
\P(\text{minstens één keer winst}) = 1 - \frac{25}{36} = \frac{11}{36}
\approx 0.31 .
\]
\end{solution}

\begin{solution}{pb:g9:statproba:1}
\textbf{1.} Elke worp heeft $5$ uitkomsten zonder zes op $6$; de tabel van
$6 \times 6$ laat $5 \times 5 = 25$ uitkomsten zonder zes zien op $36$:
$P(\text{geen zes}) = \frac{25}{36}$, dus
$P(\text{minstens één}) = 1 - \frac{25}{36} = \frac{11}{36} \approx 0.31$.

\textbf{2.} $\frac13 = \frac{12}{36} \neq \frac{11}{36}$. De optelling telt de
uitkomst ``zes en dan zes'' in elke $\frac16$ één keer mee: één keer te veel.
Kansen van gebeurtenissen die \emph{samen} kunnen optreden, tel je niet zomaar
op.

\textbf{3.} \omterm{def:g1:addition:def}{Som} $7$: de zes uitkomsten $(1,6), (2,5), (3,4), (4,3), (5,2),
(6,1)$: $P = \frac{6}{36} = \frac16$. \omterm{def:g1:addition:def}{Som} $2$: alleen $(1,1)$: $\frac{1}{36}$.
Zeven heeft de meeste manieren om op te treden --- het dikke midden van de tabel
van de sommen.

\textbf{4.} \omterm{def:g1:addition:def}{Som} $9$: $(3,6), (4,5), (5,4), (6,3)$ --- vier uitkomsten,
$\frac{4}{36}$. \omterm{def:g1:addition:def}{Som} $10$: $(4,6), (5,5), (6,4)$ --- drie, $\frac{3}{36}$. Negen is
waarschijnlijker: het aantal uitkomsten beslist, niet het aantal ``manieren om de
\omterm{def:g1:addition:def}{som} te schrijven'' met ongeordende zijden.

\textbf{5.} Elk niveau van de boom vermenigvuldigt de \omterm{def:g9:statproba:probability}{kans} op geen zes met
$\frac56$, wat er ook eerder gebeurde: na $n$ worpen is
$P(\text{helemaal geen zes}) = \left(\frac56\right)^n$, en de complementregel geeft
de formule.

\textbf{6.} $\left(\frac56\right)^4 = \frac{625}{1296} \approx 0.482$, dus won de
Chevalier met \omterm{def:g9:statproba:probability}{kans} $1 - 0.482 = 0.518$ --- een stille, gestage voorsprong van bijna
$2\,\%$ per spel: een uitstekende weddenschap, en ze betaalde zijn koetsen.

\textbf{7.} Bij $7$ worpen geeft zijn regel $\frac76$ --- een \omterm{def:g9:statproba:probability}{kans} groter dan $1$,
en dus onzin. Het is de dubbele telling van vraag 2, opgehoopt: de successen van de
worpen overlappen elkaar, en hun kansen optellen telt de overlappingen keer op keer
mee.

\textbf{8.} $1 - \left(\frac{35}{36}\right)^{24} \approx 1 - 0.5086 = 0.491$: onder
de \omterm{def:g3:division:half}{helft}. Door tegen gelijke inzet te wedden op een gebeurtenis van $49.1\,\%$
verloor de Chevalier gemiddeld ongeveer $2$ spelletjes op elke $100$ --- langzaam,
mysterieus (voor hem) en onvermijdelijk.

\textbf{9.} $1 - \left(\frac{35}{36}\right)^{25} \approx 1 - 0.494 = 0.506$: met
$25$ worpen wordt de weddenschap gunstig. Eén worp scheidde de Chevalier van
winstgevendheid.

\textbf{10.} Een \omterm{def:g9:statproba:probability}{kans} met het aantal pogingen vermenigvuldigen telt overlappende
successen meerdere keren mee, en levert uiteindelijk onmogelijke antwoorden boven
$1$ op. De juiste weg loopt altijd via het complement: knoop de kansen op mislukking
aan elkaar door te vermenigvuldigen, en trek daarna af van $1$.

\textbf{11.} \omterm{def:g9:statproba:indicators}{Gemiddelde}: $\frac{9 \times 2\,000 + 20\,000}{10} =
\frac{38\,000}{10} = 3\,800$ euro. \omterm{def:g9:statproba:indicators}{Mediaan}: de middelste lonen zijn beide
$2\,000$: \omterm{def:g9:statproba:indicators}{mediaan} $2\,000$ euro. De vacature zou de \omterm{def:g9:statproba:indicators}{mediaan} moeten vermelden ---
negen van de tien werknemers zien nooit iets als $3\,800$; het \omterm{def:g9:statproba:indicators}{gemiddelde} wordt
omhooggetrokken door één uitschieter (\cref{ex:g9:statproba:weighted} waarschuwde
al voor gewichten).

\textbf{12.} Bijvoorbeeld $2, 8, 15, 17, 18$: \omterm{def:g9:statproba:indicators}{mediaan} $15$ (de middelste waarde),
\omterm{def:g9:statproba:indicators}{gemiddelde} $\frac{60}{5} = 12$. Een \omterm{def:g9:statproba:indicators}{gemiddelde} onder de \omterm{def:g9:statproba:indicators}{mediaan} verraadt een staart
van \emph{lage} cijfers die het \omterm{def:g9:statproba:indicators}{gemiddelde} omlaag trekt, terwijl de bovenste \omterm{def:g3:division:half}{helft}
hoog zit.

\textbf{13.} De eerste drie factoren:
$1 \times \frac{364}{365} \times \frac{363}{365} \approx 0.992$. Elke nieuwe
leerling moet één ingenomen datum meer ontwijken, dus krimpen de factoren:
$\frac{343}{365} \approx 0.94$ voor de $23$ste. Het volledige \omterm{def:g2:mult:def}{product} zakt tot
ongeveer $0.493$ --- dus is
\[
P(\text{minstens één gedeelde geboortedag}) \approx 0.507 :
\]
meer dan de \omterm{def:g3:division:half}{helft}, in een klas van slechts $23$. De intuïtie zegt ``$23$ mensen,
$365$ dagen, geen \omterm{def:g9:statproba:probability}{kans}''; de wiskunde zegt ``gooi een munt op''.

\textbf{14.} $\frac{23 \times 22}{2} = 253$ paren. Elk paar is een nieuwe \omterm{def:g9:statproba:probability}{kans} op
een samenloop, en $253$ kansen van elk $\frac{1}{365}$ is geen kleinigheid meer: de
verrassing lost op zodra je paren telt in plaats van mensen.

\textbf{15.} Voorbeeld van een werkwijze: gooi een dobbelsteen in blokken van vier
en noteer voor $100$ blokken of er in elk blok een zes valt; of verzamel de lijsten
met geboortedagen van veel klassen van ongeveer $23$ leerlingen en noteer het deel
met een samenloop. De waargenomen \omterm{def:g7:stats:frequency}{frequentie} zal schommelen, maar naarmate het
aantal herhalingen groeit, zou ze zich steeds dichter bij de berekende kansen moeten
nestelen ($0.518$; $0.507$) --- \omterm{def:g7:stats:frequency}{frequenties} naderen kansen: de wet van de grote
aantallen, hier als verwachting geformuleerd en in het bovenbouwvolume bewezen.
\end{solution}
