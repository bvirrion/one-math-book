\section*{Hoofdstuk \ref{ch:g9:trig} --- Goniometrie in de rechthoekige driehoek}

\begin{solution}{exo:g9:trig:1}
Schuine zijde: $\sqrt{9^2 + 12^2} = \sqrt{81 + 144} = \sqrt{225} = 15$.

Andere rechthoekszijde: $\sqrt{17^2 - 8^2} = \sqrt{289 - 64} = \sqrt{225} = 15$.
\end{solution}

\begin{solution}{exo:g9:trig:2}
$7^2 + 24^2 = 49 + 576 = 625 = 25^2$: ja, \omterm{def:g6:shapes:triangles}{rechthoekig} (omgekeerde stelling van
Pythagoras), met de \omterm{def:g3:shapes:rightangle}{rechte hoek} tegenover de zijde $25$.

$5^2 + 6^2 = 61 \neq 64 = 8^2$: niet \omterm{def:g6:shapes:triangles}{rechthoekig}.
\end{solution}

\begin{solution}{exo:g9:trig:3}
De schuine zijde is $[EF]$ (tegenover de \omterm{def:g3:shapes:rightangle}{rechte hoek} in $D$). De zijde tegenover
$\theta = \widehat E$ is $[DF]$; de aanliggende zijde is $[DE]$. Bijgevolg is
\[
\cos\theta = \frac{DE}{EF}, \qquad
\sin\theta = \frac{DF}{EF}, \qquad
\tan\theta = \frac{DF}{DE}.
\]
\end{solution}

\begin{solution}{exo:g9:trig:4}
Overstaande rechthoekszijde: $10 \sin 28^\circ \approx 10 \times 0.469 = 4.7$.
Aanliggende rechthoekszijde: $10 \cos 28^\circ \approx 10 \times 0.883 = 8.8$.
(Controle: $4.7^2 + 8.8^2 \approx 22 + 77 \approx 100$.)
\end{solution}

\begin{solution}{exo:g9:trig:5}
$\tan\theta = \dfrac{4.5}{6} = 0.75$, dus is
$\theta = \tan^{-1}(0.75) \approx 37^\circ$.
\end{solution}

\begin{solution}{exo:g9:trig:6}
In de \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} die de waarnemer, het punt op de grond onder de drone
en de drone vormen: de hoogte van $120$ m is de zijde tegenover de hoogtehoek, en
de horizontale afstand $d$ ligt eraan. Dus is
\[
\tan 32^\circ = \frac{120}{d},
\qquad
d = \frac{120}{\tan 32^\circ} \approx \frac{120}{0.625} = 192 \text{ m}.
\]
\end{solution}

\begin{solution}{exo:g9:trig:7}
De stijging ($1.2$ m) ligt tegenover de hoek; de lengte $L$ van de oprit is de
schuine zijde:
\[
\sin 5^\circ = \frac{1.2}{L},
\qquad
L = \frac{1.2}{\sin 5^\circ} \approx \frac{1.2}{0.0872} \approx 13.8
\text{ m}.
\]
De oprit moet minstens $13.8$ m lang zijn.
\end{solution}

\begin{solution}{exo:g9:trig:8}
Uit $\cos^2\theta + \sin^2\theta = 1$ volgt $\sin^2\theta = 1 - 0.36 = 0.64$, en
$\sin\theta > 0$ voor een scherpe hoek, dus is $\sin\theta = 0.8$. Dan is
$\tan\theta = \dfrac{\sin\theta}{\cos\theta} = \dfrac{0.8}{0.6} = \dfrac43$.
\end{solution}

\begin{solution}{exo:g9:trig:9}
\emph{Rechthoekige \omterm{def:g6:shapes:triangles}{gelijkbenige driehoek}} met rechthoekszijden $1$: schuine zijde
$\sqrt{1 + 1} = \sqrt2$; elke scherpe hoek is $45^\circ$, en
\[
\cos 45^\circ = \frac{1}{\sqrt2} = \frac{\sqrt2}{2} = \sin 45^\circ,
\qquad
\tan 45^\circ = \frac11 = 1 .
\]

\emph{\omterm{def:g6:shapes:triangles}{Gelijkzijdige driehoek}} met zijde $1$: zijn hoogte verdeelt hem in twee
\omterm{def:g6:shapes:triangles}{rechthoekige driehoeken} met schuine zijde $1$, basis $\frac12$ en hoogte
$h = \sqrt{1 - \frac14} = \frac{\sqrt3}{2}$ (Pythagoras). De hoeken zijn
$60^\circ$ (op de basis) en $30^\circ$ (aan de top). De verhoudingen aflezen geeft
\[
\cos 60^\circ = \frac{1/2}{1} = \frac12, \quad
\sin 60^\circ = \frac{\sqrt3/2}{1} = \frac{\sqrt3}{2}, \quad
\cos 30^\circ = \frac{\sqrt3}{2}, \quad
\sin 30^\circ = \frac12 .
\]
\end{solution}

\begin{solution}{pb:g9:trig:1}
\textbf{1.} De hoogte boven ooghoogte is de zijde tegenover $35^\circ$, met
aanliggende zijde $60$ m: $h = 60 \tan 35^\circ \approx 60 \times 0.700 = 42$ m.

\textbf{2.} $\tan\theta = 0.12$ geeft $\theta = \tan^{-1}(0.12) \approx
6.8^\circ$ --- steil voor een weg, en toch een bescheiden hoek. Een weg van
$45^\circ$ heeft $\tan 45^\circ = 1$: een stijging van $100\,\%$. (``Honderd
procent'' is nog lang niet verticaal --- een geliefd misverstand bij skiliften.)

\textbf{3.} In een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} met scherpe hoeken $\theta$ en
$90^\circ - \theta$ is de zijde die tegenover de ene ligt, de aanliggende zijde van
de andere, en is de schuine zijde gemeenschappelijk. Dus is
$\sin(90^\circ - \theta) = \frac{\text{haar overstaande zijde}}{h} =
\frac{\text{aanliggende zijde van }\theta}{h} = \cos\theta$, en symmetrisch
$\cos(90^\circ - \theta) = \sin\theta$: de \emph{co}\omterm{def:g9:trig:ratios}{sinus} is de \omterm{def:g9:trig:ratios}{sinus} van het
\emph{co}mplement.

\textbf{4.} Hoogte: $8 \sin 60^\circ = 8 \times \frac{\sqrt3}{2} = 4\sqrt3
\approx 6.93$ m. Voet: $8 \cos 60^\circ = 4$ m, precies.

\textbf{5.} $0.574^2 + 0.819^2 = 0.329 + 0.671 = 1.000$ (op de \omterm{def:g6:decimals:rounding}{afronding} na); en
$\left(\frac12\right)^2 + \left(\frac{\sqrt3}{2}\right)^2 = \frac14 + \frac34 =
1$. Elk paar waarden dat de gelijkheid niet doorstaat, verraadt een verkeerde
aflezing of een rekenmachine in de verkeerde stand --- een gratis foutendetector.

\textbf{6.} $\tan 40^\circ = \dfrac hx$ en $\tan 30^\circ = \dfrac{h}{x + 200}$.

\textbf{7.} $x = \dfrac{h}{\tan 40^\circ}$ en
$x + 200 = \dfrac{h}{\tan 30^\circ}$; aftrekken geeft
$200 = h\left(\frac{1}{\tan 30^\circ} - \frac{1}{\tan 40^\circ}\right)$, en
vandaar de formule. Met getallen: $\frac{1}{0.577} \approx 1.732$,
$\frac{1}{0.839} \approx 1.192$, \omterm{ex:g1:subtraction:difference}{verschil} $0.540$:
$h \approx \frac{200}{0.540} \approx 370$ m.

\textbf{8.} $x = \frac{h}{\tan 40^\circ} \approx \frac{370}{0.839} \approx 441$ m.
Controle vanuit $A$: $\frac{h}{x + 200} \approx \frac{370}{641} \approx 0.577 =
\tan 30^\circ$: samenhangend.

\textbf{9.} De methode met één standplaats heeft de horizontale afstand nodig tot
het punt \emph{recht onder de top} --- niet te meten over een ravijn of binnen in
een berg. De methode met twee standplaatsen vervangt die door een afstand die je
zelf kunt afstappen: de $200$ m tussen je twee eigen standplaatsen.

\textbf{10.} $h = \dfrac{200}{\sqrt3 - \frac{1}{\sqrt3}}
= \dfrac{200}{\frac{2}{\sqrt3}} = 100\sqrt3 \approx 173$ m --- de tweede
verdieping van de Eiffeltoren, gemeten met twee peilingen en een wandeling.

\textbf{11.} Bliklijn $d$, radiussen $R$ (naar het scherende punt, \omterm{def:g6:lines:perp}{loodrecht} op de
bliklijn) en $R + h$ (naar het oog): Pythagoras geeft
$d^2 + R^2 = (R + h)^2 = R^2 + 2Rh + h^2$, dus
$d^2 = 2Rh + h^2 \approx 2Rh$ ($h$ is minuscuul tegenover $R$). Ogen op $1.80$ m:
$d \approx \sqrt{2 \times 6.37 \times 10^6 \times 1.8} \approx 4\,800$ m --- de
horizon op zee ligt nauwelijks vijf kilometer ver.

\textbf{12.} $d \approx \sqrt{2 \times 6.37 \times 10^6 \times 324} \approx 64$
km. Vuistregel van de zeelieden: $3.6\sqrt{1.8} \approx 4.8$ km en
$3.6\sqrt{324} = 64.8$ km --- beide kloppen.

\textbf{13.} $\tan\theta = \frac{2}{60}$ geeft $\theta \approx 1.9^\circ$. De
$0.5^\circ$ van de Maan is ongeveer een \emph{\omterm{def:g3:division:half}{kwart}} duim: de Maan lijkt enorm en
wordt door een vingernagel verborgen --- het oog is een slechte gradenboog.

\textbf{14.} Trap: $\tan^{-1}\frac{17}{29} \approx 30^\circ$ --- een comfortabele
trap klimt onder bijna precies de $30^\circ$ van de bijzondere waarden. Onder
$60^\circ$ bij een aantrede van $29$ cm zou de optrede
$29\tan 60^\circ = 29\sqrt3 \approx 50$ cm zijn: een ladder, geen trap.

\textbf{15.} $\tan 10^\circ \approx 0.18$; $\tan 45^\circ = 1$;
$\tan 80^\circ \approx 5.7$; $\tan 89^\circ \approx 57.3$. Nadert $\theta$ de
$90^\circ$, dan krimpt de aanliggende zijde tot niets terwijl de overstaande zijde
blijft: het \omterm{def:g3:division:remainder}{quotiënt} groeit boven elke grens uit. Bij precies $90^\circ$ is er geen
driehoek en geen waarde --- de grafiek van de \omterm{def:g9:trig:ratios}{tangens} klimt langs een verticale
asymptoot, de eerste van vele in het bovenbouwvolume.
\end{solution}
