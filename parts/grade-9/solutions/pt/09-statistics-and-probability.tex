\end{solution}

\begin{solution}{exo:g9:statproba:3}

\[
\frac{2 \times 0 + 4 \times 1 + 3 \times 2 + 1 \times 3}{10}

\begin{solution}{exo:g9:statproba:4}
$\P(3) = \frac16$. \quad
$\P(\text{par}) = \frac36 = \frac12$. \quad
$\P(\text{pelo menos } 3) = \P(\{3,4,5,6\}) = \frac46 = \frac23$. \quad
$\P(\text{não } 6) = 1 - \frac16 = \frac56$.
\end{solution}

\begin{solution}{exo:g9:statproba:5}
$\P(\text{branca}) = \frac{5}{12}$.

$\P(\text{vermelha ou verde}) = \frac{4 + 3}{12} = \frac{7}{12}$.

$\P(\text{não branca}) = 1 - \frac{5}{12} = \frac{7}{12}$.

As duas últimas são iguais: ``não branca'' \emph{é} o evento
``vermelha ou verde''.
\end{solution}

\begin{solution}{exo:g9:statproba:6}
Os $12$ resultados (\omterm{def:g5:solids:def}{face} da moeda, valor do dado) são igualmente
prováveis.

Cara e um $6$: um resultado em $12$: $\frac{1}{12}$.

Cara ou um $6$: os resultados com cara ($6$ deles) mais o resultado
(coroa, $6$): $7$ resultados, então $\frac{7}{12}$. Contando cada
evento e subtraindo a sobreposição dá o mesmo:
$\frac{6}{12} + \frac{2}{12} - \frac{1}{12} = \frac{7}{12}$.
\end{solution}

\begin{solution}{exo:g9:statproba:7}
Primeira retirada: vermelha $\frac25$, preta $\frac35$. Segunda
retirada, sem reposição: depois de uma vermelha, restam $1$ vermelha e
$3$ pretas entre $4$; depois de uma preta, restam $2$ vermelhas e $2$
pretas entre $4$.

Duas pretas: $\frac35 \times \frac24 = \frac{6}{20} = \frac{3}{10}$.

Cores diferentes (VP ou PV):
$\frac25 \times \frac34 + \frac35 \times \frac24
= \frac{6}{20} + \frac{6}{20} = \frac{12}{20} = \frac35$.
\end{solution}

\begin{solution}{exo:g9:statproba:8}
\emph{1.} $9 \times 14 = 126$ pontos.

\emph{2.} Uma \omterm{def:g9:statproba:indicators}{média} de $15$ em $10$ partidas significa $150$ pontos no
total: ela precisa marcar $150 - 126 = 24$ pontos.
\end{solution}

\begin{solution}{exo:g9:statproba:9}
\emph{1.} Entre os $36$ pares igualmente prováveis, as duplas são
$(1,1), (2,2), \dots, (6,6)$: seis delas, então
$\P(\text{dupla}) = \frac{6}{36} = \frac16$.

\emph{2.} Duas partidas independentes: árvore com ganhar ($\frac16$) /
perder ($\frac56$) em cada nível. \omterm{def:g9:statproba:probability}{Probabilidade} de nunca ganhar:
$\frac56 \times \frac56 = \frac{25}{36}$. Então
\[
\P(\text{pelo menos uma vitória}) = 1 - \frac{25}{36} = \frac{11}{36}
\approx 0.31 .
\]
\end{solution}

\begin{solution}{pb:g9:statproba:1}
\textbf{1.} Cada lançamento tem $5$ resultados sem seis entre $6$; a
tabela $6 \times 6$ mostra $5 \times 5 = 25$ resultados sem seis entre
$36$: $P(\text{nenhum seis}) = \frac{25}{36}$, então
$P(\text{pelo menos um}) = 1 - \frac{25}{36} = \frac{11}{36} \approx
0.31$.

\textbf{2.} $\frac13 = \frac{12}{36} \neq \frac{11}{36}$. A \omterm{def:g1:addition:def}{soma} conta
o resultado ``seis e depois seis'' uma vez em cada $\frac16$: uma vez a
mais. \omterm{def:g9:statproba:probability}{Probabilidades} de eventos que podem acontecer \emph{juntos} não
se somam simplesmente.

\textbf{3.} \omterm{def:g1:addition:def}{Soma} $7$: os seis resultados $(1,6), (2,5), (3,4), (4,3),
(5,2), (6,1)$: $P = \frac{6}{36} = \frac16$. \omterm{def:g1:addition:def}{Soma} $2$: apenas
$(1,1)$: $\frac{1}{36}$. O sete tem o maior número de maneiras de
acontecer --- o meio gordo da tabela das \omterm{def:g1:addition:def}{somas}.

\textbf{4.} \omterm{def:g1:addition:def}{Soma} $9$: $(3,6), (4,5), (5,4), (6,3)$ --- quatro
resultados, $\frac{4}{36}$. \omterm{def:g1:addition:def}{Soma} $10$: $(4,6), (5,5), (6,4)$ --- três,
$\frac{3}{36}$. O nove é mais provável: quem decide é a contagem dos
resultados, e não o número de ``maneiras de escrever'' a \omterm{def:g1:addition:def}{soma} com \omterm{def:g5:solids:def}{faces}
sem ordem.

\textbf{5.} Cada nível da árvore multiplica a \omterm{def:g9:statproba:probability}{probabilidade} de não sair
seis por $\frac56$, independentemente do que veio antes: depois de $n$
lançamentos, $P(\text{nenhum seis}) = \left(\frac56\right)^n$, e a
regra do complementar dá a fórmula.

\textbf{6.} $\left(\frac56\right)^4 = \frac{625}{1296} \approx 0.482$,
então o Cavaleiro ganhava com \omterm{def:g9:statproba:probability}{probabilidade} $1 - 0.482 = 0.518$ --- uma
vantagem discreta e constante de quase $2\,\%$ por partida: uma aposta
excelente, que pagou as carruagens dele.

\textbf{7.} Com $7$ lançamentos, a regra dele dá $\frac76$ --- uma
\omterm{def:g9:statproba:probability}{probabilidade} maior que $1$, um disparate. É a contagem dobrada da
questão 2, acumulada: os sucessos dos lançamentos se sobrepõem, e somar
as chances deles conta as sobreposições várias e várias vezes.

\textbf{8.} $1 - \left(\frac{35}{36}\right)^{24} \approx 1 - 0.5086 =
0.491$: abaixo da \omterm{def:g3:division:half}{metade}. Apostando um para um num evento de
$49.1\,\%$, o Cavaleiro perdia cerca de $2$ partidas em cada $100$, em
\omterm{def:g9:statproba:indicators}{média} --- lentamente, misteriosamente (para ele) e sem escapatória.

\textbf{9.} $1 - \left(\frac{35}{36}\right)^{25} \approx 1 - 0.494 =
0.506$: com $25$ lançamentos, a aposta fica favorável. Um único
lançamento separava o Cavaleiro do lucro.

\textbf{10.} Multiplicar uma \omterm{def:g9:statproba:probability}{probabilidade} pelo número de tentativas
conta várias vezes os sucessos que se sobrepõem e acaba produzindo
respostas impossíveis, acima de $1$. O caminho correto passa sempre
pelo complementar: encadeie as \omterm{def:g9:statproba:probability}{probabilidades} de fracasso por
multiplicação e depois subtraia de $1$.

\textbf{11.} \omterm{def:g9:statproba:indicators}{Média}: $\frac{9 \times 2\,000 + 20\,000}{10} =
\frac{38\,000}{10} = 3\,800$ euros. \omterm{def:g9:statproba:indicators}{Mediana}: os salários centrais são
os dois $2\,000$: \omterm{def:g9:statproba:indicators}{mediana} $2\,000$ euros. O anúncio deveria citar a
\omterm{def:g9:statproba:indicators}{mediana} --- nove dos dez trabalhadores nunca veem nada parecido com
$3\,800$; a \omterm{def:g9:statproba:indicators}{média} é puxada para cima por um único valor extremo (o
\cref{ex:g9:statproba:weighted} já avisava sobre os pesos).

\textbf{12.} Por exemplo, $2, 8, 15, 17, 18$: \omterm{def:g9:statproba:indicators}{mediana} $15$ (valor
central), \omterm{def:g9:statproba:indicators}{média} $\frac{60}{5} = 12$. Uma \omterm{def:g9:statproba:indicators}{média} abaixo da \omterm{def:g9:statproba:indicators}{mediana}
denuncia uma cauda de notas \emph{baixas} puxando a \omterm{def:g9:statproba:indicators}{média} para baixo,
enquanto a \omterm{def:g3:division:half}{metade} de cima se mantém alta.

\textbf{13.} Três primeiros fatores:
$1 \times \frac{364}{365} \times \frac{363}{365} \approx 0.992$. Cada
aluno novo precisa desviar de mais uma data ocupada, então os fatores
encolhem: $\frac{343}{365} \approx 0.94$ para o $23$º. O \omterm{def:g2:mult:def}{produto}
completo cai para cerca de $0.493$ --- então
\[
P(\text{pelo menos um aniversário em comum}) \approx 0.507 :
\]
mais que meio a meio, numa turma de apenas $23$. A intuição diz ``$23$
pessoas, $365$ dias, sem chance''; a matemática diz ``jogue uma
moeda''.

\textbf{14.} $\frac{23 \times 22}{2} = 253$ pares. Cada par é uma nova
oportunidade de coincidência, e $253$ oportunidades a $\frac{1}{365}$
cada já não são pouca coisa: a surpresa se dissolve assim que se contam
pares, e não pessoas.

\textbf{15.} Protocolo de exemplo: lançar um dado em blocos de quatro,
registrando se cada bloco mostra um seis, ao longo de $100$ blocos; ou
recolher as listas de aniversários de muitas turmas de cerca de $23$
alunos e registrar a proporção com coincidência. A frequência observada
vai oscilar, mas, à medida que o número de repetições cresce, ela deve
se acomodar cada vez mais perto das \omterm{def:g9:statproba:probability}{probabilidades} calculadas
($0.518$; $0.507$) --- as frequências convergem para as
\omterm{def:g9:statproba:probability}{probabilidades}: a lei dos grandes números, enunciada aqui como
expectativa e demonstrada no volume do ensino médio.
\end{solution}
