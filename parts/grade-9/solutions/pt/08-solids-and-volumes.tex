\section*{Capítulo \ref{ch:g9:solids} --- Sólidos e volumes}

\begin{solution}{exo:g9:solids:1}
Caixa: $V = 4 \times 5 \times 12 = 240$.

\omterm{def:g7:areas:prism}{Cilindro}: $V = \pi r^2 h = \pi \times 9 \times 7 = 63\pi \approx 198$.
\end{solution}

\begin{solution}{exo:g9:solids:2}
\omterm{def:g8:solids:def}{Cone}: $V = \frac13 \pi r^2 h = \frac13 \pi \times 25 \times 12 =
100\pi \approx 314$.

\omterm{def:g8:solids:def}{Pirâmide}: \omterm{def:g6:measure:area}{área} da base $B = 8 \times 3 = 24$, então
$V = \frac13 \times 24 \times 10 = 80$.
\end{solution}

\begin{solution}{exo:g9:solids:3}
\omterm{def:g6:measure:volume}{Volume}: $V = \frac43 \pi \times 6^3 = \frac43 \pi \times 216 = 288\pi$.

\omterm{def:g6:measure:area}{Área} da superfície: $4\pi \times 6^2 = 144\pi$.
\end{solution}

\begin{solution}{exo:g9:solids:4}
\omterm{def:g3:shapes:shapes}{Raio} da seção: $\sqrt{r^2 - d^2} = \sqrt{100 - 64} = \sqrt{36} = 6$.
\end{solution}

\begin{solution}{exo:g9:solids:5}
\omterm{def:g9:thales:scaling}{Fator de escala} do \omterm{def:g8:solids:def}{cone} inteiro para o pequeno:
$k = \frac{9}{12} = \frac34$. \omterm{def:g3:shapes:shapes}{Raio} da seção:
$8 \times \frac34 = 6$.

\omterm{def:g6:measure:volume}{Volume} do \omterm{def:g8:solids:def}{cone} inteiro: $\frac13\pi \times 64 \times 12 = 256\pi$.
\omterm{def:g6:measure:volume}{Volume} do \omterm{def:g8:solids:def}{cone} pequeno: $256\pi \times \left(\frac34\right)^3 =
256\pi \times \frac{27}{64} = 108\pi \approx 339$.
\end{solution}

\begin{solution}{exo:g9:solids:6}
\omterm{def:g6:measure:volume}{Volume} da bola: $\frac43\pi \times 2^3 = \frac{32\pi}{3}$ cm$^3$. Esse
\omterm{def:g6:measure:volume}{volume} se espalha pela \omterm{def:g6:measure:area}{área} da base $\pi \times 3^2 = 9\pi$ cm$^2$,
então o nível sobe
\[
\frac{32\pi/3}{9\pi} = \frac{32}{27} \approx 1.2 \text{ cm}
\]
(uns $12$ mm).
\end{solution}

\begin{solution}{exo:g9:solids:7}
A fôrma pequena é a redução da grande pelo fator
$k = \frac36 = \frac12$, então o \omterm{def:g6:measure:volume}{volume} dela é
$\left(\frac12\right)^3 = \frac18$ do da grande: a receita enche $8$
fôrmas pequenas.
\end{solution}

\begin{solution}{exo:g9:solids:8}
$V = \frac13 \times 230^2 \times 147 = \frac13 \times 52\,900 \times
147 = 52\,900 \times 49 = 2\,592\,100$ m$^3$ --- cerca de $2.6$ milhões
de metros cúbicos.
\end{solution}

\begin{solution}{exo:g9:solids:9}
\emph{1.} A água forma um \omterm{def:g8:solids:def}{cone} reduzido pelo fator $k = \frac12$
(\omterm{def:g5:solids:def}{vértice} para baixo, \omterm{def:g3:division:half}{metade} da altura), então o \omterm{def:g6:measure:volume}{volume} dela é
$\left(\frac12\right)^3 = \frac18$ do funil: um oitavo, bem menos que a
\omterm{def:g3:division:half}{metade}!

\emph{2.} A água até a altura $d$ forma um \omterm{def:g8:solids:def}{cone} reduzido pelo fator
$k = \frac{d}{12}$, de \omterm{def:g6:measure:volume}{volume} $k^3$ vezes o do funil. \omterm{def:g3:division:half}{Metade} do \omterm{def:g6:measure:volume}{volume}
significa $k^3 = \frac12$, isto é,
$\left(\frac{d}{12}\right)^3 = \frac12$. Então
$\frac{d}{12} = \sqrt[3]{0.5} \approx 0.7937$, logo
$d \approx 9.5$ cm: a segunda \omterm{def:g3:division:half}{metade} do \omterm{def:g6:measure:volume}{volume} ocupa apenas os $2.5$ cm
de cima --- a parte larga do \omterm{def:g8:solids:def}{cone} guarda quase toda a água.
\end{solution}

\begin{solution}{pb:g9:solids:1}
\textbf{1.} $V_{\text{cil}} = \pi r^2 \times 2r = 2\pi r^3$.

\textbf{2.} $\dfrac{V_{\text{esfera}}}{V_{\text{cil}}}
= \dfrac{\frac43 \pi r^3}{2 \pi r^3} = \dfrac23$. O $\pi$ e o $r^3$ se
cancelam: a proporção é \emph{universal} --- vale para uma bolinha de
gude e para um planeta. Uma relação entre formas, e não entre números:
exatamente o tipo de verdade que vale gravar na pedra.

\textbf{3.} Esfera: $4\pi r^2$. \omterm{def:g7:areas:prism}{Cilindro}: lateral
$2\pi r \times 2r = 4\pi r^2$, mais duas tampas
$2 \times \pi r^2$: total $6\pi r^2$. Razão:
$\frac{4\pi r^2}{6\pi r^2} = \frac23$ --- os mesmos dois terços, agora
para superfícies. (E um bônus: a \omterm{def:g6:measure:area}{área} da esfera é exatamente igual à
\omterm{def:g6:measure:area}{área} lateral do \omterm{def:g7:areas:prism}{cilindro}.)

\textbf{4.} Sobra: $2\pi r^3 - \frac43\pi r^3 = \frac23 \pi r^3$. Dois
\omterm{def:g8:solids:def}{cones} de \omterm{def:g3:shapes:shapes}{raio} $r$ e altura $r$:
$2 \times \frac13 \pi r^2 \times r = \frac23 \pi r^3$. Iguais.

\textbf{5.} Bola: $\frac43 \pi \times 12^3 \approx 7\,200$ cm$^3$;
caixa: $2\pi \times 12^3 \approx 10\,900$ cm$^3$. Vazio: um terço da
caixa, cerca de $33\,\%$ --- garantido pela questão 2, seja qual for o
tamanho da bola.

\textbf{6.} Meia esfera:
$\frac12 \times \frac43 \pi \times 10^3 = \frac23 \pi \times 1\,000
\approx 2\,094$ cm$^3 \approx 2.1$ L.

\textbf{7.} \omterm{def:g8:solids:def}{Cone}: $\frac13 \pi \times 10^2 \times 10 =
\frac{1\,000\pi}{3} \approx 1\,047$ cm$^3$. Meia esfera:
$\frac{2\,000\pi}{3} \approx 2\,094$ cm$^3$. \omterm{def:g7:areas:prism}{Cilindro}:
$1\,000\pi \approx 3\,142$ cm$^3$. Razões:
$\frac{1000\pi}{3} : \frac{2000\pi}{3} : \frac{3000\pi}{3}
= 1 : 2 : 3$, exatamente.

\textbf{8.} $\frac43 \pi \times 27 = \pi \times 9 \times h$ dá
$h = \frac43 \times 3 = 4$ cm, exatamente.

\textbf{9.} \omterm{def:g3:shapes:shapes}{Raios}: $\frac{6\,371}{1\,737} \approx 3.67$. Os \omterm{def:g6:measure:volume}{volumes}
mudam com o cubo (\cref{thm:g9:solids:scaling}):
$3.67^3 \approx 49$. Cabem umas cinquenta Luas na Terra.

\textbf{10.} As superfícies mudam com o quadrado:
$3.67^2 \approx 13.5$. Dobrar o \omterm{def:g3:shapes:shapes}{raio} de um balão multiplica o \omterm{def:g6:measure:volume}{volume}
dele por $2^3 = 8$, mas a superfície por apenas $2^2 = 4$: à medida que
as coisas crescem, o \omterm{def:g6:measure:volume}{volume} (peso, conteúdo, calor produzido) deixa a
superfície para trás (pele, material, calor perdido) --- a lei do
quadrado--cubo.

\textbf{11.} Peso: $\times 10^3 = 1\,000$. \omterm{prop:g9:solids:sections}{Seção transversal} dos ossos:
$\times 10^2 = 100$. Pressão --- peso por \omterm{def:g6:measure:area}{área} de osso:
$\times \frac{1000}{100} = 10$. Ossos construídos para a pressão humana
recebem dez vezes mais: o esqueleto do gigante estala sob o próprio
peso. Os gigantes são geometricamente impossíveis; os animais grandes
precisam de ossos desproporcionalmente grossos (compare as pernas de um
elefante com as de uma gazela).

\textbf{12.} Peso: $\div 100^3 = 10^6$. Força (\omterm{prop:g9:solids:sections}{seção transversal} do
músculo): $\div 100^2 = 10^4$. Força por peso:
$\times \frac{10^6}{10^4} = 100$. A formiga que carrega cinquenta vezes
o próprio peso não é uma superatleta --- ela é apenas pequena; um ser
humano encolhido ao tamanho de formiga faria o mesmo.

\textbf{13.} $\dfrac{S}{V} = \dfrac{4\pi r^2}{\frac43\pi r^3}
= \dfrac{3}{r}$: para $r = 1$ a razão é $3$, e para $r = 2$ ela é
$1.5$ --- reduzir o tamanho à \omterm{def:g3:division:half}{metade} dobra a superfície disponível por
unidade de \omterm{def:g6:measure:volume}{volume} produtor de calor. Os corpos pequenos perdem calor
depressa: o camundongo precisa comer o tempo todo, o urso pode
hibernar, e o bebê --- esfera pequena, $\frac Sr$ grande --- precisa da
touca.

\textbf{14.} $\left(\frac54\right)^3 = \frac{125}{64} \approx 1.95$: a
laranja grande entrega quase o \emph{dobro} de fruta pelo mesmo preço.
Compre \omterm{def:g3:shapes:shapes}{raio}.

\textbf{15.} A gravura diz: a esfera é $\frac23$ do \omterm{def:g7:areas:prism}{cilindro} em \omterm{def:g6:measure:volume}{volume}
\emph{e} $\frac23$ dele em superfície total --- duas proporções
exatas e universais ligando o \omterm{def:g5:solids:def}{sólido} mais redondo ao mais simples. A
resposta a Cícero: porque um teorema é o único monumento que nem
exércitos nem séculos corroem --- as máquinas de Arquimedes arderam com
Siracusa, mas os dois terços continuam sendo exatamente dois terços.
\end{solution}
