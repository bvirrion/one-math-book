\section*{Capítulo \ref{ch:g9:trig} --- Trigonometría en el triángulo rectángulo}

\begin{solution}{exo:g9:trig:1}
Hipotenusa: $\sqrt{9^2 + 12^2} = \sqrt{81 + 144} = \sqrt{225} = 15$.

Otro cateto: $\sqrt{17^2 - 8^2} = \sqrt{289 - 64} = \sqrt{225} = 15$.
\end{solution}

\begin{solution}{exo:g9:trig:2}
$7^2 + 24^2 = 49 + 576 = 625 = 25^2$: sí, rectángulo (recíproco de
Pitágoras), con el \omterm{def:g3:shapes:rightangle}{ángulo recto} opuesto al lado $25$.

$5^2 + 6^2 = 61 \neq 64 = 8^2$: no es rectángulo.
\end{solution}

\begin{solution}{exo:g9:trig:3}
La hipotenusa es $[EF]$ (opuesta al \omterm{def:g3:shapes:rightangle}{ángulo recto} en $D$). El lado
opuesto a $\theta = \widehat E$ es $[DF]$; el lado adyacente es
$[DE]$. Por tanto
\[
\cos\theta = \frac{DE}{EF}, \qquad
\sin\theta = \frac{DF}{EF}, \qquad
\tan\theta = \frac{DF}{DE}.
\]
\end{solution}

\begin{solution}{exo:g9:trig:4}
Cateto opuesto: $10 \sin 28^\circ \approx 10 \times 0.469 = 4.7$.
Cateto adyacente: $10 \cos 28^\circ \approx 10 \times 0.883 = 8.8$.
(Comprobación: $4.7^2 + 8.8^2 \approx 22 + 77 \approx 100$.)
\end{solution}

\begin{solution}{exo:g9:trig:5}
$\tan\theta = \dfrac{4.5}{6} = 0.75$, luego
$\theta = \tan^{-1}(0.75) \approx 37^\circ$.
\end{solution}

\begin{solution}{exo:g9:trig:6}
En el \omterm{def:g6:shapes:triangles}{triángulo rectángulo} que forman el observador, el punto del suelo
bajo el dron y el dron: la altitud de $120$ m es el lado
opuesto al ángulo de elevación, y la distancia horizontal $d$ es el adyacente.
Así que
\[
\tan 32^\circ = \frac{120}{d},
\qquad
d = \frac{120}{\tan 32^\circ} \approx \frac{120}{0.625} = 192 \text{ m}.
\]
\end{solution}

\begin{solution}{exo:g9:trig:7}
La subida ($1.2$ m) es el lado opuesto al ángulo; la longitud $L$ de la rampa es la
hipotenusa:
\[
\sin 5^\circ = \frac{1.2}{L},
\qquad
L = \frac{1.2}{\sin 5^\circ} \approx \frac{1.2}{0.0872} \approx 13.8
\text{ m}.
\]
La rampa tiene que medir al menos $13.8$ m.
\end{solution}

\begin{solution}{exo:g9:trig:8}
De $\cos^2\theta + \sin^2\theta = 1$:
$\sin^2\theta = 1 - 0.36 = 0.64$, y $\sin\theta > 0$ para un ángulo
agudo, luego $\sin\theta = 0.8$. Entonces
$\tan\theta = \dfrac{\sin\theta}{\cos\theta} = \dfrac{0.8}{0.6} =
\dfrac43$.
\end{solution}

\begin{solution}{exo:g9:trig:9}
\emph{\omterm{def:g6:shapes:triangles}{Triángulo isósceles} rectángulo} de catetos $1$: hipotenusa
$\sqrt{1 + 1} = \sqrt2$; cada ángulo agudo mide $45^\circ$, y
\[
\cos 45^\circ = \frac{1}{\sqrt2} = \frac{\sqrt2}{2} = \sin 45^\circ,
\qquad
\tan 45^\circ = \frac11 = 1 .
\]

\emph{\omterm{def:g6:shapes:triangles}{Triángulo equilátero}} de lado $1$: su altura lo parte en dos
\omterm{def:g6:shapes:triangles}{triángulos rectángulos} de hipotenusa $1$, base $\frac12$ y altura
$h = \sqrt{1 - \frac14} = \frac{\sqrt3}{2}$ (Pitágoras). Los ángulos son
$60^\circ$ (en la base) y $30^\circ$ (arriba). Leyendo las
razones:
\[
\cos 60^\circ = \frac{1/2}{1} = \frac12, \quad
\sin 60^\circ = \frac{\sqrt3/2}{1} = \frac{\sqrt3}{2}, \quad
\cos 30^\circ = \frac{\sqrt3}{2}, \quad
\sin 30^\circ = \frac12 .
\]
\end{solution}

\begin{solution}{pb:g9:trig:1}
\textbf{1.} La altura por encima del nivel de los ojos es el lado opuesto a
$35^\circ$, con lado adyacente $60$ m:
$h = 60 \tan 35^\circ \approx 60 \times 0.700 = 42$ m.

\textbf{2.} $\tan\theta = 0.12$ da
$\theta = \tan^{-1}(0.12) \approx 6.8^\circ$ --- empinado para una
carretera y, aun así, un ángulo modesto. Una carretera de $45^\circ$ tiene
$\tan 45^\circ = 1$: una \omterm{prop:g9:linfunc:affinegraph}{pendiente} del $100\,\%$. («Cien
por cien» está lejos de ser vertical --- un malentendido clásico
en los telesillas.)

\textbf{3.} En un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} de ángulos agudos $\theta$ y
$90^\circ - \theta$, el lado opuesto a uno es el adyacente del
otro y la hipotenusa es común. Así que
$\sin(90^\circ - \theta) = \frac{\text{su opuesto}}{h} =
\frac{\text{adyacente de }\theta}{h} = \cos\theta$, y
simétricamente $\cos(90^\circ - \theta) = \sin\theta$: el
\emph{co}\omterm{def:g9:trig:ratios}{seno} es el \omterm{def:g9:trig:ratios}{seno} del \emph{co}mplementario.

\textbf{4.} Altura: $8 \sin 60^\circ = 8 \times
\frac{\sqrt3}{2} = 4\sqrt3 \approx 6.93$ m. Pie:
$8 \cos 60^\circ = 4$ m exactamente.

\textbf{5.} $0.574^2 + 0.819^2 = 0.329 + 0.671 = 1.000$ (salvo
\omterm{def:g6:decimals:rounding}{redondeo}); y $\left(\frac12\right)^2 +
\left(\frac{\sqrt3}{2}\right)^2 = \frac14 + \frac34 = 1$. Cualquier
pareja de valores que no cumpla la identidad delata una lectura equivocada o
una calculadora en el modo equivocado --- un detector de errores gratis.

\textbf{6.} $\tan 40^\circ = \dfrac hx$ y
$\tan 30^\circ = \dfrac{h}{x + 200}$.

\textbf{7.} $x = \dfrac{h}{\tan 40^\circ}$ y
$x + 200 = \dfrac{h}{\tan 30^\circ}$; restando,
$200 = h\left(\frac{1}{\tan 30^\circ} -
\frac{1}{\tan 40^\circ}\right)$, de donde la fórmula. Números:
$\frac{1}{0.577} \approx 1.732$, $\frac{1}{0.839} \approx
1.192$, \omterm{ex:g1:subtraction:difference}{diferencia} $0.540$:
$h \approx \frac{200}{0.540} \approx 370$ m.

\textbf{8.} $x = \frac{h}{\tan 40^\circ} \approx
\frac{370}{0.839} \approx 441$ m. Comprobación desde $A$:
$\frac{h}{x + 200} \approx \frac{370}{641} \approx 0.577 =
\tan 30^\circ$: coherente.

\textbf{9.} El método de una estación necesita la distancia horizontal
al punto situado \emph{justo debajo de la cima} ---
imposible de medir a través de un barranco o dentro de una montaña. El
método de las dos estaciones la sustituye por una distancia que puedes recorrer
tú mismo: los $200$ m entre tus dos puntos de observación.

\textbf{10.} $h = \dfrac{200}{\sqrt3 - \frac{1}{\sqrt3}}
= \dfrac{200}{\frac{2}{\sqrt3}} = 100\sqrt3 \approx 173$ m ---
el segundo piso de la torre Eiffel, medido con dos visuales
y un paseo.

\textbf{11.} Visual $d$, \omterm{def:g3:shapes:shapes}{radios} $R$ (hasta el punto de roce,
perpendicular a la visual) y $R + h$ (hasta el ojo):
Pitágoras da $d^2 + R^2 = (R + h)^2 = R^2 + 2Rh + h^2$, luego
$d^2 = 2Rh + h^2 \approx 2Rh$ ($h$ es minúsculo frente a $R$).
Con los ojos a $1.80$ m:
$d \approx \sqrt{2 \times 6.37 \times 10^6 \times 1.8}
\approx 4\,800$ m --- el horizonte marino está a apenas cinco
kilómetros.

\textbf{12.} $d \approx \sqrt{2 \times 6.37 \times 10^6 \times
324} \approx 64$ km. Regla de los marinos: $3.6\sqrt{1.8} \approx
4.8$ km y $3.6\sqrt{324} = 64.8$ km --- las dos coinciden.

\textbf{13.} $\tan\theta = \frac{2}{60}$ da $\theta \approx
1.9^\circ$. Los $0.5^\circ$ de la Luna son aproximadamente un \emph{\omterm{def:g3:division:half}{cuarto}} de
pulgar: la Luna parece enorme y la tapa una uña
--- el ojo es mal transportador.

\textbf{14.} Escalera: $\tan^{-1}\frac{17}{29} \approx 30^\circ$
--- una escalera cómoda sube casi exactamente con los
$30^\circ$ de los valores especiales. Con $60^\circ$ y una huella de $29$ cm,
la contrahuella tendría que ser $29\tan 60^\circ = 29\sqrt3 \approx
50$ cm: una escalera de mano, no una escalera de casa.

\textbf{15.} $\tan 10^\circ \approx 0.18$;
$\tan 45^\circ = 1$; $\tan 80^\circ \approx 5.7$;
$\tan 89^\circ \approx 57.3$. Cuando $\theta \to 90^\circ$, el
lado adyacente se encoge hasta nada mientras el opuesto aguanta:
el \omterm{def:g3:division:remainder}{cociente} crece por encima de cualquier cota. En $90^\circ$ exactamente
no hay triángulo ni valor --- la gráfica de la \omterm{def:g9:trig:ratios}{tangente}
trepa por una asíntota vertical, la primera de muchas en el \omterm{def:g6:measure:volume}{volumen}
de secundaria superior.
\end{solution}
