\section*{Capítulo \ref{ch:g9:algebra} --- Álgebra y ecuaciones}

\begin{solution}{exo:g9:algebra:1}
$4(3x - 2) = 12x - 8$.

$(x+6)(x+2) = x^2 + 2x + 6x + 12 = x^2 + 8x + 12$.

$(2x-3)(x+5) = 2x^2 + 10x - 3x - 15 = 2x^2 + 7x - 15$.

$(x-1)(x-9) = x^2 - 9x - x + 9 = x^2 - 10x + 9$.
\end{solution}

\begin{solution}{exo:g9:algebra:2}
$(x+8)^2 = x^2 + 16x + 64$.

$(3x-4)^2 = 9x^2 - 24x + 16$.

$(x+10)(x-10) = x^2 - 100$.

$(5x+2)(5x-2) = 25x^2 - 4$.
\end{solution}

\begin{solution}{exo:g9:algebra:3}
$101^2 = (100+1)^2 = 10000 + 200 + 1 = 10201$.

$59^2 = (60-1)^2 = 3600 - 120 + 1 = 3481$.

$53 \times 47 = (50+3)(50-3) = 2500 - 9 = 2491$.
\end{solution}

\begin{solution}{exo:g9:algebra:4}
$9x + 12 = 3(3x + 4)$.

$x^2 - 64 = (x+8)(x-8)$.

$x^2 + 14x + 49 = (x+7)^2$.

$18x^2 - 6x = 6x(3x - 1)$.
\end{solution}

\begin{solution}{exo:g9:algebra:5}
$4x + 7 = 19$: $4x = 12$, luego $x = 3$.

$6x - 5 = 2x + 11$: $4x = 16$, luego $x = 4$.

$5(x-3) = 3x + 1$: $5x - 15 = 3x + 1$, luego $2x = 16$ y $x = 8$.
\end{solution}

\begin{solution}{exo:g9:algebra:6}
$(x+4)(3x-12) = 0$: $x = -4$ o $x = 4$.

$x^2 - 121 = (x-11)(x+11) = 0$: $x = 11$ o $x = -11$.

$(2x-1)^2 = 0$: la única solución $x = \frac12$.

$x^2 = 8x$: $x(x - 8) = 0$, luego $x = 0$ o $x = 8$.
\end{solution}

\begin{solution}{exo:g9:algebra:7}
\omterm{ex:g1:subtraction:difference}{Diferencia} de cuadrados con $a = 3x + 2$ y $b = 5$:
\[
E = (3x + 2)^2 - 5^2 = (3x + 2 + 5)(3x + 2 - 5) = (3x + 7)(3x - 3).
\]
Regla del \omterm{def:g2:mult:def}{producto} nulo: $3x + 7 = 0$ da $x = -\frac73$, y $3x - 3 = 0$
da $x = 1$.
\end{solution}

\begin{solution}{exo:g9:algebra:8}
$3x - 4 \geq 8$: $3x \geq 12$, luego $x \geq 4$ (punto relleno en $4$, sombreado
a la derecha).

$5 - 2x > 1$: $-2x > -4$; al dividir entre $-2$ se invierte el signo: $x < 2$
(punto hueco en $2$, sombreado a la izquierda).

$4(x+1) \leq x - 5$: $4x + 4 \leq x - 5$, luego $3x \leq -9$, luego
$x \leq -3$ (punto relleno en $-3$, sombreado a la izquierda).
\end{solution}

\begin{solution}{exo:g9:algebra:9}
Sea $x$ el número: $3x + 7 = 5x - 9$. Entonces $16 = 2x$, luego $x = 8$.
Comprobación: $3 \times 8 + 7 = 31$ y $5 \times 8 - 9 = 31$.
\end{solution}

\begin{solution}{exo:g9:algebra:10}
\emph{1.} $(n+1)(n-1) = n^2 - 1$ (tercera identidad). Los vecinos de $n$
son $n - 1$ y $n + 1$, así que su \omterm{def:g2:mult:def}{producto} es $n^2 - 1$: por ejemplo,
$24 \times 26 = 625 - 1 = 624$.

\emph{2.} Tres enteros consecutivos se pueden escribir $n - 1$, $n$,
$n + 1$. Su \omterm{def:g1:addition:def}{suma} es
\[
(n-1) + n + (n+1) = 3n ,
\]
\omterm{def:g9:arith:divisor}{múltiplo} de $3$ (tres veces el del medio).
\end{solution}

\begin{solution}{pb:g9:algebra:1}
\textbf{1.} $(10a + b) - (10b + a) = 9a - 9b = 9(a - b)$: la
\omterm{ex:g1:subtraction:difference}{diferencia} entre un número de dos cifras y su volteado siempre es
\omterm{def:g9:arith:divisor}{múltiplo} de $9$. Comprobación: $72 - 27 = 45 = 9 \times 5$, y
$a - b = 7 - 2 = 5$.

\textbf{2.} Desarrollando el miembro derecho:
$9(11a + b) + a + b + c = 99a + 9b + a + b + c
= 100a + 10b + c$. Así que el número es igual a un \omterm{def:g9:arith:divisor}{múltiplo} de $9$ más
su \omterm{def:g1:addition:def}{suma} de cifras: al dividir entre $9$, los dos dejan el mismo \omterm{def:g3:division:remainder}{resto}.
En particular, \omterm{def:g3:division:remainder}{resto} $0$ para uno significa \omterm{def:g3:division:remainder}{resto} $0$ para el
otro: el criterio del nueve, demostrado.

\textbf{3.} $9(11a + b)$ también es \omterm{def:g9:arith:divisor}{múltiplo} de $3$, así que la misma
identidad muestra que el número y su \omterm{def:g1:addition:def}{suma} de cifras dejan el mismo \omterm{def:g3:division:remainder}{resto} en
la división entre $3$: el criterio del tres.

\textbf{4.} Escribe $47 = 9k + 2$ y $86 = 9l + 5$. Entonces
\[
47 \times 86 = (9k + 2)(9l + 5)
= 9\,(9kl + 5k + 2l) + 10 :
\]
el \omterm{def:g2:mult:def}{producto} deja el mismo \omterm{def:g3:division:remainder}{resto} que $2 \times 5 = 10$,
es decir $1$. Por tanto, un resultado correcto también tiene que dejar \omterm{def:g3:division:remainder}{resto}
$1$ --- y $4\,042$ lo deja. Pero la prueba solo ve \omterm{def:g3:division:remainder}{restos}:
$4\,042$ y $4\,402$ (cifras intercambiadas) pasan igual, así que
pasarla no demuestra nada --- lo que condena es fallarla.

\textbf{5.} Desarrollando: $11(91a + 9b + c) - a + b - c + d
= 1001a + 99b + 11c - a + b - c + d
= 1000a + 100b + 10c + d$. Así que el número es \omterm{def:g6:wholes:divisible}{divisible} entre $11$
exactamente cuando lo es su \omterm{def:g1:addition:def}{suma} alterna $-a + b - c + d$. Para
$2\,926$: $-2 + 9 - 2 + 6 = 11$, \omterm{def:g6:wholes:divisible}{divisible} entre $11$: sí,
$2\,926 = 11 \times 266$.

\textbf{6.} $742 - 247 = 495$; $495 + 594 = 1\,089$. Cualquier
ejemplo válido aterriza también en $1\,089$.

\textbf{7.} $(100a + 10b + c) - (100c + 10b + a) = 99(a - c)$.
Para $a - c = 2, 3, \dots, 9$:
\[
198,\ 297,\ 396,\ 495,\ 594,\ 693,\ 792,\ 891 .
\]

\textbf{8.} $100(m - 1) + 90 + (10 - m) = 100m - 100 + 90 + 10
- m = 99m$. Así que las tres cifras de $99m$ son $m - 1$, después $9$
y después $10 - m$ (comprueba con $495$: $m = 5$ da $4$, $9$, $5$).

\textbf{9.} El volteado de $99m$ tiene cifras $10 - m$, $9$,
$m - 1$, así que vale $100(10 - m) + 90 + (m - 1)$. Sumando:
\[
100\,(m - 1 + 10 - m) + 180 + (10 - m + m - 1)
= 100 \times 9 + 180 + 9 = 1\,089 ,
\]
independientemente de $m$. Sea cual sea el número de partida, el mago
está a salvo.

\textbf{10.} Si $a = c$, la resta da $0$ y el truco
muere de inmediato. Si $a - c = 1$, da $99$, que hay que
tratar como la cadena de tres cifras $099$ (volteado $990$,
$99 + 990 = 1\,089$: ¡salvado!). La letra pequeña: las \omterm{ex:g1:subtraction:difference}{diferencias} de
$1$ solo funcionan si el espectador escribe el resultado de la resta
con tres cifras, \omterm{def:g1:counting:zero}{cero} inicial incluido --- exigir una separación de
al menos $2$ evita la discusión.

\textbf{11.} $2(5m + 7) + d - 14 = 10m + 14 + d - 14 =
10m + d$: las decenas (y las centenas, para octubre--diciembre) dan
el mes, y las unidades (y las decenas) el día.

\textbf{12.} $2(5m + 9) + d = 10m + 18 + d$: hay que restar $18$.

\textbf{13.} Por la pregunta 2, $n - (\text{suma de cifras de } n) =
9(11a + b)$ para un $n$ de tres cifras: siempre \omterm{def:g9:arith:divisor}{múltiplo} de $9$
(identidades parecidas valen para cualquier longitud). Un \omterm{def:g9:arith:divisor}{múltiplo} de $9$ tiene
\omterm{def:g1:addition:def}{suma} de cifras \omterm{def:g9:arith:divisor}{múltiplo} de $9$ (otra vez la pregunta 2), así que las cifras
que el espectador lee hay que completarlas hasta el siguiente \omterm{def:g9:arith:divisor}{múltiplo}
de $9$: la cifra que falta es ese complemento. Ambigüedad: si las
cifras leídas ya suman un \omterm{def:g9:arith:divisor}{múltiplo} de $9$, la cifra tachada
podría ser $0$ o $9$ --- excluir el $0$ quita la duda.

\textbf{14.} $(10t + 5)^2 = 100t^2 + 100t + 25 =
100\,t(t + 1) + 25$: escribe $t(t+1)$ y pega $25$ detrás.
$45^2$: $4 \times 5 = 20$, así que $2\,025$. $85^2$:
$8 \times 9 = 72$, así que $7\,225$. $105^2$: $10 \times 11 = 110$,
así que $11\,025$.

\textbf{15.} Cinco consecutivos: $n + (n+1) + (n+2) + (n+3) +
(n+4) = 5n + 10 = 5(n + 2)$: \omterm{def:g9:arith:divisor}{múltiplo} de $5$, siempre. Cuatro
consecutivos: $n + (n+1) + (n+2) + (n+3) = 4n + 6 =
4(n + 1) + 2$: \omterm{def:g3:division:remainder}{resto} $2$ en la división entre $4$, nunca
\omterm{def:g9:arith:divisor}{múltiplo}. El álgebra convierte «parece funcionar» en
\emph{siempre}, y «no he encontrado ningún ejemplo» en \emph{nunca} ---
dos palabras a las que ninguna cantidad de pruebas puede llegar
(\cref{met:g7:literal:test}).
\end{solution}
