\section*{Capítulo \ref{ch:g9:sqrt} --- Raíces cuadradas}

\begin{solution}{exo:g9:sqrt:1}
$\sqrt{64} = 8$; \quad $\sqrt{100} = 10$; \quad
$\sqrt{0.09} = 0.3$ (porque $0.3^2 = 0.09$); \quad
$\sqrt{\frac{1}{25}} = \frac15$; \quad
$\left(\sqrt{13}\right)^2 = 13$; \quad
$\sqrt{(-4)^2} = \sqrt{16} = 4 = \abs{-4}$.
\end{solution}

\begin{solution}{exo:g9:sqrt:2}
$\sqrt{50} = \sqrt{25 \times 2} = 5\sqrt2$; \quad
$\sqrt{45} = \sqrt{9 \times 5} = 3\sqrt5$; \quad
$\sqrt{98} = \sqrt{49 \times 2} = 7\sqrt2$; \quad
$\sqrt{300} = \sqrt{100 \times 3} = 10\sqrt3$.
\end{solution}

\begin{solution}{exo:g9:sqrt:3}
$\sqrt2 \times \sqrt{18} = \sqrt{36} = 6$.

$\dfrac{\sqrt{75}}{\sqrt3} = \sqrt{\dfrac{75}{3}} = \sqrt{25} = 5$.

$\sqrt5 \times \sqrt{20} = \sqrt{100} = 10$.

$\dfrac{\sqrt8}{\sqrt{50}} = \sqrt{\dfrac{8}{50}} = \sqrt{\dfrac{4}{25}}
= \dfrac25$.
\end{solution}

\begin{solution}{exo:g9:sqrt:4}
$x^2 = 36$: $x = 6$ o $x = -6$.

$x^2 = 11$: $x = \sqrt{11}$ o $x = -\sqrt{11}$.

$x^2 + 4 = 0$ significa $x^2 = -4 < 0$: no hay solución.

$2x^2 = 50$: $x^2 = 25$, luego $x = 5$ o $x = -5$.
\end{solution}

\begin{solution}{exo:g9:sqrt:5}
$\sqrt{20} + \sqrt{45} = 2\sqrt5 + 3\sqrt5 = 5\sqrt5$.

$3\sqrt8 - \sqrt{18} + \sqrt2 = 3 \times 2\sqrt2 - 3\sqrt2 + \sqrt2
= (6 - 3 + 1)\sqrt2 = 4\sqrt2$.
\end{solution}

\begin{solution}{exo:g9:sqrt:6}
$\left(\sqrt3 + 1\right)^2 = 3 + 2\sqrt3 + 1 = 4 + 2\sqrt3$.

$\left(2\sqrt5 - 3\right)^2 = 4 \times 5 - 2 \times 3 \times 2\sqrt5 + 9
= 29 - 12\sqrt5$.

$\left(\sqrt6 + \sqrt2\right)\left(\sqrt6 - \sqrt2\right) = 6 - 2 = 4$.
\end{solution}

\begin{solution}{exo:g9:sqrt:7}
Lado: $\sqrt{6400} = 80$ m. Diagonal (Pitágoras):
$\sqrt{80^2 + 80^2} = 80\sqrt2 \approx 113$ m.
\end{solution}

\begin{solution}{exo:g9:sqrt:8}
$\dfrac{1}{\sqrt2} = \dfrac{1 \times \sqrt2}{\sqrt2 \times \sqrt2}
= \dfrac{\sqrt2}{2}$.

$\dfrac{6}{\sqrt3} = \dfrac{6\sqrt3}{3} = 2\sqrt3$.
\quad
$\dfrac{10}{\sqrt5} = \dfrac{10\sqrt5}{5} = 2\sqrt5$.
\end{solution}

\begin{solution}{exo:g9:sqrt:9}
\emph{1. Verdadero}: es la regla del \omterm{def:g2:mult:def}{producto} (\cref{thm:g9:sqrt:rules}).

\emph{2. Falso}: $\sqrt{9 + 16} = 5$, pero $\sqrt9 + \sqrt{16} = 7$.

\emph{3. Falso} para $x$ negativo: $\sqrt{(-4)^2} = 4 \neq -4$. El
enunciado correcto es $\sqrt{x^2} = \abs{x}$.

\emph{4. Verdadero}: $\left(3\sqrt2\right)^2 = 9 \times 2 = 18$.
\end{solution}

\begin{solution}{exo:g9:sqrt:10}
$\left(2 + \sqrt3\right)^2 = 4 + 4\sqrt3 + 3 = 7 + 4\sqrt3$. Así que
$x = \sqrt{7 + 4\sqrt3} = \sqrt{\left(2+\sqrt3\right)^2} = 2 + \sqrt3$
(es un número no negativo, así que la raíz solo quita el cuadrado).

Igualmente $\left(\sqrt5 - 2\right)^2 = 5 - 4\sqrt5 + 4 = 9 - 4\sqrt5$,
y $\sqrt5 - 2 > 0$, luego
$\sqrt{9 - 4\sqrt5} = \sqrt5 - 2$ (¡y no $2 - \sqrt5$, que es
negativo!).
\end{solution}

\begin{solution}{pb:g9:sqrt:1}
\textbf{1.} $1.4^2 = 1.96 < 2 < 2.25 = 1.5^2$;
$1.41^2 = 1.9881 < 2 < 2.0164 = 1.42^2$. Con tres decimales:
$1.414^2 = 1.999396 < 2 < 2.002225 = 1.415^2$, luego
$1.414 < \sqrt2 < 1.415$.

\textbf{2.} Elevar al cuadrado $\sqrt2 = \frac ab$ da
$2 = \frac{a^2}{b^2}$, de donde $a^2 = 2b^2$: el número $a^2$ es
el doble de un número entero --- \emph{\omterm{def:g3:numbers:evenodd}{par}}.

\textbf{3.} Si $a$ fuera \omterm{def:g3:numbers:evenodd}{impar}, $a^2$ sería \omterm{def:g3:numbers:evenodd}{impar}
(\cref{pb:g8:pythagoras:1}); como $a^2$ es \omterm{def:g3:numbers:evenodd}{par}, $a$ es \omterm{def:g3:numbers:evenodd}{par}:
$a = 2k$. Sustituyendo: $4k^2 = 2b^2$, luego $b^2 = 2k^2$ es \omterm{def:g3:numbers:evenodd}{par},
y por el mismo hecho de paridad $b$ es \omterm{def:g3:numbers:evenodd}{par}.

\textbf{4.} Que $a$ y $b$ sean los dos \omterm{def:g3:numbers:evenodd}{pares} significa que los dos son \omterm{def:g6:wholes:divisible}{divisibles} entre $2$
--- pero $\frac ab$ era irreducible, sin ningún \omterm{def:g9:arith:divisor}{divisor}
común. Contradicción: la \omterm{def:g9:fractions:fraction}{fracción} supuesta no puede existir.
\emph{Teorema: $\sqrt2$ es irracional.}

\textbf{5.} La contradicción vive entera en la palabra
«irreducible»: sin ella, «$a$ y $b$ los dos \omterm{def:g3:numbers:evenodd}{pares}» no contradice
nada. Suponer la \omterm{def:g9:fractions:fraction}{fracción} irreducible es legítimo porque toda
\omterm{def:g9:fractions:fraction}{fracción} \emph{tiene} una forma irreducible
(\cref{met:g9:arith:simplify}: divide entre el MCD) --- si
$\sqrt2$ fuera una \omterm{def:g9:fractions:fraction}{fracción} cualquiera, sería una \omterm{def:g9:fractions:fraction}{fracción} irreducible,
y esa es imposible.

\textbf{6.} En las descomposiciones en factores primos, los cuadrados llevan \omterm{def:g8:powers:def}{exponentes}
\omterm{def:g3:numbers:evenodd}{pares} (\cref{pb:g9:arith:1}). En $a^2 = 3b^2$, el \omterm{def:g8:powers:def}{exponente}
de $3$ es \omterm{def:g3:numbers:evenodd}{par} en el miembro izquierdo, pero \omterm{def:g3:numbers:evenodd}{impar} en el derecho (\omterm{def:g3:numbers:evenodd}{par} por
$b^2$, más uno). Ningún número tiene dos descomposiciones
(\cref{thm:g9:arith:factorization}): contradicción, y
$\sqrt3$ es irracional.

\textbf{7.} En $a^2 = n b^2$, el argumento encuentra un primo con
paridad de \omterm{def:g8:powers:def}{exponente} contradictoria exactamente cuando algún primo aparece en
$n$ con \omterm{def:g8:powers:def}{exponente} \emph{\omterm{def:g3:numbers:evenodd}{impar}}. Si todos los \omterm{def:g8:powers:def}{exponentes} de $n$ son
\omterm{def:g3:numbers:evenodd}{pares}, $n$ es un cuadrado perfecto ($n = m^2$, y
$\sqrt n = m$ es entero). Resultado completo: $\sqrt n$ es
irracional para todo entero $n$ que no sea un cuadrado perfecto ---
$\sqrt2, \sqrt3, \sqrt5, \sqrt6, \sqrt7, \sqrt8, \sqrt{10},
\dots$

\textbf{8.} De $r x = q$ con $r \neq 0$:
$x = \frac qr$, un \omterm{def:g3:division:remainder}{cociente} de racionales, luego racional. Así que, si
$x$ es irracional, $r x$ no puede ser racional: $2\sqrt2$ y
$\frac{\sqrt2}{2}$ son irracionales (multiplicadores $2$ y
$\frac12$).

\textbf{9.} Si $1 + \sqrt2 = q$ fuera racional, entonces
$\sqrt2 = q - 1$ sería racional: contradicción. Si
$s = \sqrt2 + \sqrt3$ fuera racional, también lo sería
$s^2 = 2 + 2\sqrt6 + 3 = 5 + 2\sqrt6$, lo que daría
$\sqrt6 = \frac{s^2 - 5}{2}$ racional --- pero $\sqrt6$ es
irracional (pregunta 7). Así que $\sqrt2 + \sqrt3$ es irracional.

\textbf{10.} \omterm{def:g1:addition:def}{Sumas}: $\sqrt2$ y $-\sqrt2$ (o $\sqrt2$ y
$1 - \sqrt2$) son cada uno irracionales, con \omterm{def:g1:addition:def}{sumas} racionales $0$ y
$1$. \omterm{def:g2:mult:def}{Productos}: $\sqrt2 \times \sqrt2 = 2$. La irracionalidad puede
evaporarse en la aritmética --- de ahí las demostraciones caso por caso.

\textbf{11.} Desde $x = 1$: media de $1$ y $\frac21 = 2$:
$x_1 = \frac32$. Después, media de $\frac32$ y
$\frac{2}{3/2} = \frac43$:
$x_2 = \frac12\left(\frac32 + \frac43\right)
= \frac12 \cdot \frac{17}{6} = \frac{17}{12} \approx 1.4167$.

\textbf{12.} $x_3 = \frac12\left(\frac{17}{12} +
\frac{24}{17}\right) = \frac12 \cdot
\frac{289 + 288}{204} = \frac{577}{408} = 1.4142156\ldots$
Frente a $\sqrt2 = 1.4142135\ldots$: cinco decimales correctos tras
tres pasos --- la receta duplica más o menos los decimales correctos
en cada vuelta.

\textbf{13.} Si $x^2 > 2$ (estimación demasiado grande), entonces $x > \sqrt2$,
y $\frac2x < \frac{2}{\sqrt2} = \sqrt2$: el lado compañero es
demasiado pequeño. Si $x^2 < 2$, las desigualdades se invierten. En cualquier caso,
$\sqrt2$ está entre $x$ y $\frac2x$, así que su media ---
la estimación siguiente --- está más cerca que el peor de los dos lados:
el rectángulo de \omterm{def:g6:measure:area}{área} $2$ se vuelve más cuadrado en cada paso.

\textbf{14.} $9 - 8 = 1$; \ $289 - 288 = 1$; \
$332\,929 - 332\,928 = 1$. Toda estimación $\frac ab$ de Herón
cumple $a^2 - 2b^2 = 1$: la Parte I demostró que llegar a $0$ es
imposible, y estas fracciones fallan lo imposible por exactamente
una unidad --- lo más cerca que pueden llegar los números enteros.

\textbf{15.} La recta numérica tiene que contener números más allá de las
fracciones --- longitudes como $\sqrt2$, cuyas escrituras decimales corren
eternamente sin repetirse: los \emph{números reales}. Su
construcción honesta es historia para el \omterm{def:g6:measure:volume}{volumen} de secundaria superior y,
con todo rigor, para los universitarios.
\end{solution}
