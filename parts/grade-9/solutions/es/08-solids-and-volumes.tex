\section*{Capítulo \ref{ch:g9:solids} --- Cuerpos y volúmenes}

\begin{solution}{exo:g9:solids:1}
\omterm{def:g5:solids:def}{Ortoedro}: $V = 4 \times 5 \times 12 = 240$.

\omterm{def:g7:areas:prism}{Cilindro}: $V = \pi r^2 h = \pi \times 9 \times 7 = 63\pi \approx 198$.
\end{solution}

\begin{solution}{exo:g9:solids:2}
\omterm{def:g8:solids:def}{Cono}: $V = \frac13 \pi r^2 h = \frac13 \pi \times 25 \times 12 =
100\pi \approx 314$.

\omterm{def:g8:solids:def}{Pirámide}: \omterm{def:g6:measure:area}{área} de la base $B = 8 \times 3 = 24$, luego
$V = \frac13 \times 24 \times 10 = 80$.
\end{solution}

\begin{solution}{exo:g9:solids:3}
\omterm{def:g6:measure:volume}{Volumen}: $V = \frac43 \pi \times 6^3 = \frac43 \pi \times 216 = 288\pi$.

\omterm{def:g6:measure:area}{Área} de la superficie: $4\pi \times 6^2 = 144\pi$.
\end{solution}

\begin{solution}{exo:g9:solids:4}
\omterm{def:g3:shapes:shapes}{Radio} de la \omterm{prop:g9:solids:sections}{sección}: $\sqrt{r^2 - d^2} = \sqrt{100 - 64} = \sqrt{36}
= 6$.
\end{solution}

\begin{solution}{exo:g9:solids:5}
\omterm{def:g9:thales:scaling}{Factor de escala} del \omterm{def:g8:solids:def}{cono} entero al pequeño:
$k = \frac{9}{12} = \frac34$. \omterm{def:g3:shapes:shapes}{Radio} de la \omterm{prop:g9:solids:sections}{sección}: $8 \times \frac34 = 6$.

\omterm{def:g6:measure:volume}{Volumen} del \omterm{def:g8:solids:def}{cono} entero: $\frac13\pi \times 64 \times 12 = 256\pi$.
\omterm{def:g6:measure:volume}{Volumen} del \omterm{def:g8:solids:def}{cono} pequeño: $256\pi \times \left(\frac34\right)^3 =
256\pi \times \frac{27}{64} = 108\pi \approx 339$.
\end{solution}

\begin{solution}{exo:g9:solids:6}
\omterm{def:g6:measure:volume}{Volumen} de la bola: $\frac43\pi \times 2^3 = \frac{32\pi}{3}$ cm$^3$.
Ese \omterm{def:g6:measure:volume}{volumen} se reparte sobre el \omterm{def:g6:measure:area}{área} de la base $\pi \times 3^2 = 9\pi$ cm$^2$,
así que el nivel sube
\[
\frac{32\pi/3}{9\pi} = \frac{32}{27} \approx 1.2 \text{ cm}
\]
(unos $12$ mm).
\end{solution}

\begin{solution}{exo:g9:solids:7}
El molde pequeño es la reducción del grande con $k = \frac36 =
\frac12$, así que su \omterm{def:g6:measure:volume}{volumen} es $\left(\frac12\right)^3 = \frac18$ del
grande: la receta llena $8$ moldes pequeños.
\end{solution}

\begin{solution}{exo:g9:solids:8}
$V = \frac13 \times 230^2 \times 147 = \frac13 \times 52\,900 \times 147
= 52\,900 \times 49 = 2\,592\,100$ m$^3$ --- unos $2.6$ millones de metros
cúbicos.
\end{solution}

\begin{solution}{exo:g9:solids:9}
\emph{1.} El agua forma un \omterm{def:g8:solids:def}{cono} escalado con $k = \frac12$ (\omterm{def:g5:solids:def}{vértice} hacia abajo,
la \omterm{def:g3:division:half}{mitad} de la altura), así que su \omterm{def:g6:measure:volume}{volumen} es $\left(\frac12\right)^3 = \frac18$
del embudo: ¡un octavo, mucho menos que la \omterm{def:g3:division:half}{mitad}!

\emph{2.} El agua hasta la altura $d$ forma un \omterm{def:g8:solids:def}{cono} escalado con
$k = \frac{d}{12}$, de \omterm{def:g6:measure:volume}{volumen} $k^3$ veces el del embudo. La \omterm{def:g3:division:half}{mitad} del \omterm{def:g6:measure:volume}{volumen}
significa $k^3 = \frac12$, es decir
$\left(\frac{d}{12}\right)^3 = \frac12$. Entonces
$\frac{d}{12} = \sqrt[3]{0.5} \approx 0.7937$, luego
$d \approx 9.5$ cm: la segunda \omterm{def:g3:division:half}{mitad} del \omterm{def:g6:measure:volume}{volumen} ocupa solo los
$2.5$ cm de arriba --- la parte ancha del \omterm{def:g8:solids:def}{cono} contiene casi toda el agua.
\end{solution}

\begin{solution}{pb:g9:solids:1}
\textbf{1.} $V_{\text{cil}} = \pi r^2 \times 2r = 2\pi r^3$.

\textbf{2.} $\dfrac{V_{\text{esfera}}}{V_{\text{cil}}}
= \dfrac{\frac43 \pi r^3}{2 \pi r^3} = \dfrac23$. El $\pi$ y
el $r^3$ se cancelan: la proporción es \emph{universal} --- vale
para una canica y para un planeta. Una relación entre formas, no
entre números: exactamente el tipo de verdad que merece grabarse en
piedra.

\textbf{3.} Esfera: $4\pi r^2$. \omterm{def:g7:areas:prism}{Cilindro}: lateral
$2\pi r \times 2r = 4\pi r^2$, más dos tapas
$2 \times \pi r^2$: en total $6\pi r^2$. Razón:
$\frac{4\pi r^2}{6\pi r^2} = \frac23$ --- los mismos dos tercios,
ahora para superficies. (Y una propina: el \omterm{def:g6:measure:area}{área} de la esfera es exactamente igual
al \omterm{def:g6:measure:area}{área} lateral del \omterm{def:g7:areas:prism}{cilindro}.)

\textbf{4.} Sobrante: $2\pi r^3 - \frac43\pi r^3 =
\frac23 \pi r^3$. Dos \omterm{def:g8:solids:def}{conos} de \omterm{def:g3:shapes:shapes}{radio} $r$ y altura $r$:
$2 \times \frac13 \pi r^2 \times r = \frac23 \pi r^3$. Iguales.

\textbf{5.} Balón: $\frac43 \pi \times 12^3 \approx 7\,200$
cm$^3$; caja: $2\pi \times 12^3 \approx 10\,900$ cm$^3$. Vacío:
un tercio de la caja, alrededor del $33\,\%$ --- lo garantiza la
pregunta 2, sea cual sea el tamaño del balón.

\textbf{6.} Media esfera:
$\frac12 \times \frac43 \pi \times 10^3 = \frac23 \pi \times
1\,000 \approx 2\,094$ cm$^3 \approx 2.1$ L.

\textbf{7.} \omterm{def:g8:solids:def}{Cono}: $\frac13 \pi \times 10^2 \times 10 =
\frac{1\,000\pi}{3} \approx 1\,047$ cm$^3$. Semiesfera:
$\frac{2\,000\pi}{3} \approx 2\,094$ cm$^3$. \omterm{def:g7:areas:prism}{Cilindro}:
$1\,000\pi \approx 3\,142$ cm$^3$. Razones:
$\frac{1000\pi}{3} : \frac{2000\pi}{3} : \frac{3000\pi}{3}
= 1 : 2 : 3$ exactamente.

\textbf{8.} $\frac43 \pi \times 27 = \pi \times 9 \times h$
da $h = \frac43 \times 3 = 4$ cm exactamente.

\textbf{9.} \omterm{def:g3:shapes:shapes}{Radios}: $\frac{6\,371}{1\,737} \approx 3.67$.
Los volúmenes escalan como el \omterm{def:g5:solids:def}{cubo} (\cref{thm:g9:solids:scaling}):
$3.67^3 \approx 49$. En la Tierra caben unas cincuenta Lunas.

\textbf{10.} Las superficies escalan como el cuadrado: $3.67^2 \approx
13.5$. Duplicar el \omterm{def:g3:shapes:shapes}{radio} de un globo multiplica su \omterm{def:g6:measure:volume}{volumen} por
$2^3 = 8$, pero su superficie solo por $2^2 = 4$: al crecer las cosas,
el \omterm{def:g6:measure:volume}{volumen} (peso, contenido, calor producido) adelanta a la superficie (piel,
material, calor perdido) --- la ley del cuadrado-cubo.

\textbf{11.} Peso: $\times 10^3 = 1\,000$. \omterm{prop:g9:solids:sections}{Sección} del
hueso: $\times 10^2 = 100$. Presión --- peso por
\omterm{def:g6:measure:area}{área} de hueso: $\times \frac{1000}{100} = 10$. Unos huesos hechos para
la presión humana reciben diez veces más: el esqueleto del gigante
se parte bajo su propio peso. Los gigantes son geométricamente
imposibles; los animales grandes necesitan huesos desproporcionadamente gruesos
(compara las patas de un elefante con las de una gacela).

\textbf{12.} Peso: $\div 100^3 = 10^6$. Fuerza (\omterm{prop:g9:solids:sections}{sección}
del músculo): $\div 100^2 = 10^4$. Fuerza entre peso:
$\times \frac{10^6}{10^4} = 100$. La hormiga que levanta cincuenta veces
su peso no es una superatleta --- simplemente es pequeña; un
humano encogido al tamaño de una hormiga haría lo mismo.

\textbf{13.} $\dfrac{S}{V} = \dfrac{4\pi r^2}{\frac43\pi r^3}
= \dfrac{3}{r}$: para $r = 1$ la razón es $3$ y para $r = 2$ es
$1.5$ --- reducir el tamaño a la \omterm{def:g3:division:half}{mitad} duplica la superficie disponible
por unidad de \omterm{def:g6:measure:volume}{volumen} productor de calor. Los cuerpos pequeños pierden calor:
el ratón tiene que comer sin parar, el oso puede hibernar y el
bebé --- esfera pequeña, $\frac Sr$ grande --- necesita el gorro.

\textbf{14.} $\left(\frac54\right)^3 = \frac{125}{64} \approx
1.95$: la naranja grande contiene casi \emph{el doble} de fruta por
el mismo precio. Compra \omterm{def:g3:shapes:shapes}{radio}.

\textbf{15.} El grabado dice: la esfera es $\frac23$ del
\omterm{def:g7:areas:prism}{cilindro} en \omterm{def:g6:measure:volume}{volumen} \emph{y} $\frac23$ de él en superficie
total --- dos proporciones exactas y universales que enlazan el
cuerpo más redondo con el más sencillo. La respuesta a Cicerón: porque un
teorema es el único monumento que no erosionan ni los ejércitos ni los siglos
--- las máquinas de Arquímedes ardieron con Siracusa, pero los
dos tercios siguen siendo exactamente dos tercios.
\end{solution}
