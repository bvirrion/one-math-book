\section*{الفصل \ref{ch:g9:fractions} --- الكسور والقوى}

\begin{solution}{exo:g9:fractions:1}
$\dfrac13 + \dfrac15 = \dfrac{5}{15} + \dfrac{3}{15} = \dfrac{8}{15}$.

$\dfrac72 - \dfrac53 = \dfrac{21}{6} - \dfrac{10}{6} = \dfrac{11}{6}$.

$\dfrac{5}{12} + \dfrac38 = \dfrac{10}{24} + \dfrac{9}{24} =
\dfrac{19}{24}$.

$3 - \dfrac45 = \dfrac{15}{5} - \dfrac45 = \dfrac{11}{5}$.
\end{solution}

\begin{solution}{exo:g9:fractions:2}
$\dfrac59 \times \dfrac{27}{35} = \dfrac{5 \times 27}{9 \times 35}
= \dfrac{27}{9} \times \dfrac{5}{35} = 3 \times \dfrac17 = \dfrac37$
(بالاختزال على $9$ وعلى $5$).

$\dfrac{8}{15} \div \dfrac45 = \dfrac{8}{15} \times \dfrac54
= \dfrac{8 \times 5}{15 \times 4} = \dfrac{40}{60} = \dfrac23$.

$\dfrac23 \times \dfrac34 \times \dfrac45 = \dfrac{2 \times 3 \times 4}
{3 \times 4 \times 5} = \dfrac25$ (فالعددان $3$ و $4$ يتلاشيان).
\end{solution}

\begin{solution}{exo:g9:fractions:3}
$2^5 = 32$; \quad $(-2)^4 = 16$; \quad $-2^4 = -16$; \quad
$4^{-2} = \frac{1}{16}$; \quad
$\left(\frac23\right)^3 = \frac{8}{27}$; \quad $7^0 = 1$.
\end{solution}

\begin{solution}{exo:g9:fractions:4}
$5^3 \times 5^6 = 5^9$; \quad $\dfrac{7^9}{7^5} = 7^4$; \quad
$\left(2^3\right)^4 = 2^{12}$; \quad
$\dfrac{3^2 \times 3^7}{3^5} = 3^{9-5} = 3^4$; \quad
$2^5 \times 4 = 2^5 \times 2^2 = 2^7$.
\end{solution}

و $302.5 = 3.025 \times 10^2$؛ \quad

\begin{solution}{exo:g9:fractions:6}
$(2 \times 10^7) \times (4.5 \times 10^{-3}) = 9 \times 10^{4}$.

$\dfrac{9 \times 10^5}{3 \times 10^{-2}} = 3 \times 10^{5 - (-2)}
= 3 \times 10^7$.

$(5 \times 10^3)^2 = 25 \times 10^6 = 2.5 \times 10^7$.
\end{solution}

\begin{solution}{exo:g9:fractions:7}
الماء المضاف يمثّل
$\dfrac34 - \dfrac35 = \dfrac{15}{20} - \dfrac{12}{20} = \dfrac{3}{20}$
من الخزّان. إذن $\frac{3}{20}$ من \omterm{def:g2:measure:units}{السعة} هي $30$ لترًا:
ومنه $\frac{1}{20}$ هي $10$ لترات، \omterm{def:g2:measure:units}{والسعة} $200$ لتر.
\end{solution}

\begin{solution}{exo:g9:fractions:8}
بعد عالية: يبقى $1 - \frac14 = \frac34$ من الكعكة. ويأكل بدر
$\frac13 \times \frac34 = \frac14$، فيترك
$\frac34 - \frac14 = \frac12$. وتأكل كريمة نصف ذلك، أي $\frac14$،
فيبقى $\frac14$ من الكعكة.
\end{solution}

\begin{solution}{exo:g9:fractions:9}
الزمن $= \dfrac{\text{المسافة}}{\text{السرعة}} =
\dfrac{1.5 \times 10^{11}}{3 \times 10^8} = 0.5 \times 10^3 = 500$
ثانية، أي\ $8$ دقائق و $20$ ثانية.
\end{solution}

\begin{solution}{exo:g9:fractions:10}
اُكتب العددين قوّتين أُسّهما $10$:
\[
2^{30} = \left(2^3\right)^{10} = 8^{10},
\qquad
3^{20} = \left(3^2\right)^{10} = 9^{10}.
\]
وبما أن $8 < 9$، فبضرب عشر نسخ من كل منهما: $8^{10} < 9^{10}$، إذن
$2^{30} < 3^{20}$.
\end{solution}

\begin{solution}{pb:g9:fractions:1}
\textbf{1.} $\frac14 = 0.25$ (يتوقف)؛ و $\frac13 = 0.333\ldots$
(الدورة $3$)؛ و $\frac56 = 0.8333\ldots$ (الدورة $3$ بعد
الرقم $8$)؛ و $\frac{3}{11} = 0.272727\ldots$ (الدورة $27$).

\textbf{2.} في كل خطوة من القسمة على $b$، يكون \omterm{def:g3:division:remainder}{الباقي}
إحدى القيم $b$ التالية: $0, 1, \dots, b - 1$
(\cref{thm:g6:wholes:division}). فإن ظهر \omterm{def:g3:division:remainder}{الباقي} $0$، توقّفت
القسمة. وإلا، فبعد $b$ خطوة على الأكثر لا بدّ أن يظهر
باقٍ ما مرة ثانية --- فالخطوات أكثر
من البواقي الممكنة --- ومن تلك اللحظة تكرّر القسمة
بالضبط سلسلة الأرقام التي أنتجتها بعد
الظهور الأول: فتدور الكتابة العشرية إلى الأبد. توقّف أو
تكرار: ولا سلوك ثالث.

\textbf{3.} $\frac17 = 0.142857\,142857\ldots$: فالدورة
$142857$ طولها $6$. وقد وعد السؤال 2 بتكرار في
$7$ خطوات على الأكثر --- وهنا تظهر البواقي
$1, 3, 2, 6, 4, 5$ كلها قبل عودة $1$: أي
أطول ترنيمة ممكنة لقسمة على $7$.

\textbf{4.} يتوقف $\frac ab$ بالضبط حين يكون \omterm{def:g9:fractions:fraction}{لكسر} ما
$\frac{a \times k}{b \times k}$ \omterm{def:g4:fractions:def}{مقام} $10$ أو $100$ أو
$1000, \dots$ --- أي حين يمكن ضرب \omterm{def:g4:fractions:def}{المقام}
حتى يبلغ \omterm{def:g9:fractions:power}{قوة} للعشرة، مثل $4 \times 25 = 100$ أو
$8 \times 125 = 1\,000$. ومن أجل $3$ و $6$ هذا ميؤوس منه: \omterm{def:g9:fractions:power}{فقوة}
العشرة مجموع أرقامها $1$، وكل \omterm{def:g5:division:multiple}{مضاعف} للعدد $3$ \omterm{def:g1:addition:def}{مجموع}
أرقامه \omterm{def:g5:division:multiple}{مضاعف} للعدد $3$ --- إذن لا \omterm{def:g5:division:multiple}{مضاعف} للعدد $3$
(ومنه ولا للعدد $6$) يكون $10$ أو $100$ أو $1\,000, \dots$ ومن أجل
$11$: قسمة $10$ أو $100$ أو $1\,000$ أو $10\,000$ على $11$ تترك دائمًا
باقيًا ($10$ أو $1$ أو $10$ أو $1, \dots$)، ولا تعطي $0$ أبدًا.
فالمقامات التي تبلغ \omterm{def:g9:fractions:power}{قوة} للعشرة تتوقف؛ وكل ما عداها يتكرّر.

\textbf{5.} $\frac{9}{40}$: بما أن $40 \times 25 = 1\,000$، يتوقف:
$\frac{225}{1000} = 0.225$. و $\frac{33}{50}$:
بما أن $50 \times 2 = 100$، يتوقف: $\frac{66}{100} = 0.66$.
و $\frac{5}{12}$: العدد $12$ \omterm{def:g5:division:multiple}{مضاعف} للعدد $3$، فيتكرّر
($0.41666\ldots$).

\textbf{6.} $10x = 7.777\ldots$، إذن $10x - x = 7$ (فالذيلان
يتلاشيان تمامًا)، و $9x = 7$ و $x = \frac79$. والتحقّق بالقسمة:
$7 \div 9 = 0.777\ldots$

\textbf{7.} الدورة من رقمين، إذن أزِح بالعدد $100$:
$100x - x = 72.7272\ldots - 0.7272\ldots = 72$، ومنه
$99x = 72$ و $x = \frac{72}{99} = \frac{8}{11}$.

\textbf{8.} $1000x = 583.333\ldots$ و $100x = 58.333\ldots$؛
وبالطرح، $900x = 525$، إذن
$x = \frac{525}{900} = \frac{7}{12}$ (بالقسمة على $75$). وفعلًا
$\frac{7}{12} = 0.58333\ldots$

\textbf{9.} $10x - x = 9.999\ldots - 0.999\ldots = 9$، إذن
$9x = 9$ و $x = 1$. فليس «تحت $1$ بقليل»:
بل $0.999\ldots$ \emph{هو} العدد $1$، مكتوبًا في زيّ
ثانٍ. فأيّ عدد أصغر تمامًا من $1$ يترك فجوة، وقد
بيّن \cref{pb:g6:decimals:1} أن الفجوة تحوي دائمًا أعدادًا
أخرى --- بينما لا شيء يسع بين $0.999\ldots$ و $1$.
فالتسعات التي لا تنتهي هي \omterm{def:g3:division:remainder}{باقي} الشوكولاتة في
\cref{pb:g6:fractions:1} متقلّصًا تحت كل مقدار موجب:
فلا يبقى شيء.

\textbf{10.} $2.4545\ldots = 2 + \frac{45}{99} = 2 +
\frac{5}{11} = \frac{27}{11}$. و
$0.123123\ldots = \frac{123}{999} = \frac{41}{333}$ (بالقسمة على
$3$).

\textbf{11.} بالقاموس: $0.037037\ldots = \frac{37}{999}$.
وبما أن $999 = 27 \times 37$، يُختزل هذا إلى $\frac{1}{27}$:
\omterm{def:g9:fractions:fraction}{فالكسر} $\frac{1}{27}$ يرنّم «$037$». والتحقّق بالقسمة:
$1 \div 27 = 0.037037\ldots$

\textbf{12.} $0.0272727\ldots = \frac{1}{10} \times
0.272727\ldots = \frac{1}{10} \times \frac{27}{99}
= \frac{27}{990} = \frac{3}{110}$ (بالقسمة على $9$).

\textbf{13.} دورات الأصفار تنمو بلا نهاية، إذن لا دورة
ثابتة تتكرّر إلى الأبد: فهو \emph{ليس} عددًا عشريًّا دوريًّا
(ولا يتوقف أبدًا). وحسب الجزء الأول، كل \omterm{def:g9:fractions:fraction}{كسر} يتوقف أو
يتكرّر --- إذن هذا العدد ليس كسرًا البتة: فهو
أصمّ، مبنيّ عمدًا. وأما نجم الصمم الأكبر،
$\sqrt2$، فيُفضح في \cref{ch:g9:sqrt}.

\textbf{14.} $2^{30} = \left(2^{10}\right)^3 \approx
(1.024)^3 \times 10^9 \approx 1.07 \times 10^9$: أي عدد
أعلى قليلًا من $10^9$، ومنه فله $10$ أرقام. (وبالضبط:
$2^{30} = 1\,073\,741\,824$.)

\textbf{15.} $0.20262026\ldots = \frac{2026}{9999}$، وهو أصلًا
في أبسط صورة (فالعدد $9999 = 3 \times 3 \times 11 \times 101$ لا يشترك
في أيّ عامل مع $2026 = 2 \times 1013$). والقاموس
المكتمل: \emph{الكسور هي بالضبط \omterm{def:g6:decimals:places}{الأعداد العشرية} التي
تتوقف أو تتكرّر} --- فالمقامات صديقة العشرات تتوقف، وكل
ما عداها يرنّم دورة؛ وبالعكس كل ترنيمة، حتى سنة
تخرّجك، \omterm{def:g9:fractions:fraction}{كسر} متنكّر.
\end{solution}
