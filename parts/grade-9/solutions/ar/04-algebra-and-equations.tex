\section*{الفصل \ref{ch:g9:algebra} --- الجبر والمعادلات}

\begin{solution}{exo:g9:algebra:1}
$4(3x - 2) = 12x - 8$.

$(x+6)(x+2) = x^2 + 2x + 6x + 12 = x^2 + 8x + 12$.

$(2x-3)(x+5) = 2x^2 + 10x - 3x - 15 = 2x^2 + 7x - 15$.

$(x-1)(x-9) = x^2 - 9x - x + 9 = x^2 - 10x + 9$.
\end{solution}

\begin{solution}{exo:g9:algebra:2}
$(x+8)^2 = x^2 + 16x + 64$.

$(3x-4)^2 = 9x^2 - 24x + 16$.

$(x+10)(x-10) = x^2 - 100$.

$(5x+2)(5x-2) = 25x^2 - 4$.
\end{solution}

\begin{solution}{exo:g9:algebra:3}
$101^2 = (100+1)^2 = 10000 + 200 + 1 = 10201$.

$59^2 = (60-1)^2 = 3600 - 120 + 1 = 3481$.

$53 \times 47 = (50+3)(50-3) = 2500 - 9 = 2491$.
\end{solution}

\begin{solution}{exo:g9:algebra:4}
$9x + 12 = 3(3x + 4)$.

$x^2 - 64 = (x+8)(x-8)$.

$x^2 + 14x + 49 = (x+7)^2$.

$18x^2 - 6x = 6x(3x - 1)$.
\end{solution}

\begin{solution}{exo:g9:algebra:5}
$4x + 7 = 19$: $4x = 12$، إذن $x = 3$.

$6x - 5 = 2x + 11$: $4x = 16$، إذن $x = 4$.

$5(x-3) = 3x + 1$: $5x - 15 = 3x + 1$، إذن $2x = 16$ و $x = 8$.
\end{solution}

\begin{solution}{exo:g9:algebra:6}
$(x+4)(3x-12) = 0$: $x = -4$ أو $x = 4$.

$x^2 - 121 = (x-11)(x+11) = 0$: $x = 11$ أو $x = -11$.

$(2x-1)^2 = 0$: الحل الوحيد $x = \frac12$.

$x^2 = 8x$: $x(x - 8) = 0$، إذن $x = 0$ أو $x = 8$.
\end{solution}

\begin{solution}{exo:g9:algebra:7}
\omterm{ex:g1:subtraction:difference}{فرق} مربّعين مع $a = 3x + 2$ و $b = 5$:
\[
E = (3x + 2)^2 - 5^2 = (3x + 2 + 5)(3x + 2 - 5) = (3x + 7)(3x - 3).
\]
وقاعدة \omterm{def:g2:mult:def}{الجداء} المعدوم: $3x + 7 = 0$ يعطي $x = -\frac73$،
و $3x - 3 = 0$ يعطي $x = 1$.
\end{solution}

\begin{solution}{exo:g9:algebra:8}
$3x - 4 \geq 8$: $3x \geq 12$، إذن $x \geq 4$ (نقطة ممتلئة عند $4$،
وتظليل نحو اليمين).

$5 - 2x > 1$: $-2x > -4$، والقسمة على $-2$ تعكس الإشارة: $x < 2$
(نقطة مجوّفة عند $2$، وتظليل نحو اليسار).

$4(x+1) \leq x - 5$: $4x + 4 \leq x - 5$، إذن $3x \leq -9$، إذن
$x \leq -3$ (نقطة ممتلئة عند $-3$، وتظليل نحو اليسار).
\end{solution}

\begin{solution}{exo:g9:algebra:9}
ليكن $x$ العدد: $3x + 7 = 5x - 9$. إذن $16 = 2x$، ومنه $x = 8$.
والتحقّق: $3 \times 8 + 7 = 31$ و $5 \times 8 - 9 = 31$.
\end{solution}

\begin{solution}{exo:g9:algebra:10}
\emph{1.} $(n+1)(n-1) = n^2 - 1$ (المتطابقة الثالثة). وجارا العدد
$n$ هما $n - 1$ و $n + 1$، فيكون جداؤهما $n^2 - 1$: مثلًا
$24 \times 26 = 625 - 1 = 624$.

\emph{2.} تُكتب ثلاثة أعداد صحيحة متتالية $n - 1$ و $n$
و $n + 1$. ومجموعها
\[
(n-1) + n + (n+1) = 3n ,
\]
وهو \omterm{def:g9:arith:divisor}{مضاعف} للعدد $3$ (ثلاثة أمثال الأوسط).
\end{solution}

\begin{solution}{pb:g9:algebra:1}
\textbf{1.} $(10a + b) - (10b + a) = 9a - 9b = 9(a - b)$:
\omterm{ex:g1:subtraction:difference}{فالفرق} بين عدد ذي رقمين ومقلوبه هو دائمًا \omterm{def:g9:arith:divisor}{مضاعف} للعدد
$9$. والتحقّق: $72 - 27 = 45 = 9 \times 5$،
و $a - b = 7 - 2 = 5$.

\textbf{2.} \omterm{def:g9:algebra:expand}{بنشر} الطرف الثاني:
$9(11a + b) + a + b + c = 99a + 9b + a + b + c
= 100a + 10b + c$. إذن العدد يساوي مضاعفًا للعدد $9$ مضافًا
إليه مجموع أرقامه: فبالقسمة على $9$ يتركان \omterm{def:g3:division:remainder}{الباقي} نفسه.
وبخاصّة، \omterm{def:g3:division:remainder}{الباقي} $0$ لأحدهما يعني \omterm{def:g3:division:remainder}{الباقي} $0$ للآخر: وهي
قاعدة التسعة، مبرهنًا عليها.

\textbf{3.} والعدد $9(11a + b)$ \omterm{def:g9:arith:divisor}{مضاعف} للعدد $3$ أيضًا،
فالمتطابقة نفسها تبيّن أنّ العدد ومجموع أرقامه يتركان
\omterm{def:g3:division:remainder}{الباقي} نفسه في القسمة على $3$: وهي قاعدة الثلاثة.

\textbf{4.} اُكتب $47 = 9k + 2$ و $86 = 9l + 5$. إذن
\[
47 \times 86 = (9k + 2)(9l + 5)
= 9\,(9kl + 5k + 2l) + 10 :
\]
فيترك \omterm{def:g2:mult:def}{الجداء} \omterm{def:g3:division:remainder}{الباقي} نفسه الذي يتركه $2 \times 5 = 10$،
أي $1$. ومن ثمّ يجب أن تترك النتيجة الصحيحة \omterm{def:g3:division:remainder}{الباقي} $1$
أيضًا --- وهذا ما يفعله $4\,042$. لكنّ الاِختبار لا يرى سوى
البواقي: فالعددان $4\,042$ و $4\,402$ (بتبادل الرقمين)
يجتازانه سواءً بسواء، فالنجاح لا يبرهن على شيء --- والرسوب هو
الذي يُدين.

\textbf{5.} \omterm{def:g9:algebra:expand}{بالنشر}: $11(91a + 9b + c) - a + b - c + d
= 1001a + 99b + 11c - a + b - c + d
= 1000a + 100b + 10c + d$. إذن العدد \omterm{def:g6:wholes:divisible}{قابل للقسمة} على $11$
بالضبط حين يكون مجموعه المتناوب $-a + b - c + d$ كذلك. ومن أجل
$2\,926$: $-2 + 9 - 2 + 6 = 11$، وهو \omterm{def:g6:wholes:divisible}{قابل للقسمة} على $11$: نعم،
$2\,926 = 11 \times 266$.

\textbf{6.} $742 - 247 = 495$؛ و $495 + 594 = 1\,089$. وكل مثال
صالح يقع على $1\,089$ أيضًا.

\textbf{7.} $(100a + 10b + c) - (100c + 10b + a) = 99(a - c)$.
ومن أجل $a - c = 2, 3, \dots, 9$:
\[
198,\ 297,\ 396,\ 495,\ 594,\ 693,\ 792,\ 891 .
\]

\textbf{8.} $100(m - 1) + 90 + (10 - m) = 100m - 100 + 90 + 10
- m = 99m$. إذن أرقام العدد $99m$ الثلاثة هي $m - 1$، ثم $9$،
ثم $10 - m$ (تحقّق على $495$: فالقيمة $m = 5$ تعطي $4$ و $9$ و $5$).

\textbf{9.} \omterm{def:g8:fractions:inverse}{ومقلوب} $99m$ أرقامه $10 - m$ و $9$
و $m - 1$، فهو يساوي $100(10 - m) + 90 + (m - 1)$. وبالجمع:
\[
100\,(m - 1 + 10 - m) + 180 + (10 - m + m - 1)
= 100 \times 9 + 180 + 9 = 1\,089 ,
\]
من غير تعلّق بالعدد $m$. فمهما كان العدد المنطلق منه، الساحر في
أمان.

\textbf{10.} إذا كان $a = c$، أعطى الطرح $0$ وماتت الحيلة في
الحال. وإذا كان $a - c = 1$، أعطى $99$، الذي يجب أن يُعامل
كسلسلة من ثلاثة أرقام $099$ (مقلوبها $990$،
و $99 + 990 = 1\,089$: نجت!). والشرط الصغير: فروق
$1$ لا تشتغل إلا إذا كتب المتفرّج نتيجة الطرح بثلاثة أرقام، بما
فيها \omterm{def:g1:counting:zero}{الصفر} الأول --- واِشتراط فجوة $2$ على الأقل يتجنّب
الجدال.

\textbf{11.} $2(5m + 7) + d - 14 = 10m + 14 + d - 14 =
10m + d$: فالعشرات (والمئات، من أجل الشهور من العاشر إلى الثاني
عشر) تعطي الشهر، والآحاد (والعشرات) تعطي اليوم.

\textbf{12.} $2(5m + 9) + d = 10m + 18 + d$: فليطرح $18$.

\textbf{13.} حسب السؤال 2، $n - (\text{مجموع أرقام } n) =
9(11a + b)$ من أجل $n$ ذي ثلاثة أرقام: أي \omterm{def:g9:arith:divisor}{مضاعف} للعدد $9$
دائمًا (وتشتغل متطابقات مشابهة مهما كان الطول). \omterm{def:g9:arith:divisor}{ومضاعف}
العدد $9$ مجموع أرقامه \omterm{def:g9:arith:divisor}{مضاعف} للعدد $9$ (السؤال 2 مرّة
أخرى)، فالأرقام التي يقرؤها المتفرّج يجب أن تُكمَّل إلى
\omterm{def:g9:arith:divisor}{المضاعف} التالي للعدد $9$: والرقم الناقص هو ذلك المكمّل.
والاِلتباس: إذا كان مجموع الأرقام المقروءة مضاعفًا للعدد $9$
أصلًا، فقد يكون الرقم المشطوب $0$ أو $9$ --- واِستبعاد $0$ يزيل
الشكّ.

\textbf{14.} $(10t + 5)^2 = 100t^2 + 100t + 25 =
100\,t(t + 1) + 25$: اُكتب $t(t+1)$، ثم أَلصِق $25$ خلفه.
ففي $45^2$: $4 \times 5 = 20$، إذن $2\,025$. وفي $85^2$:
$8 \times 9 = 72$، إذن $7\,225$. وفي $105^2$: $10 \times 11 = 110$،
إذن $11\,025$.

\textbf{15.} خمسة متتالية: $n + (n+1) + (n+2) + (n+3) +
(n+4) = 5n + 10 = 5(n + 2)$: أي \omterm{def:g9:arith:divisor}{مضاعف} للعدد $5$، دائمًا.
وأربعة متتالية: $n + (n+1) + (n+2) + (n+3) = 4n + 6 =
4(n + 1) + 2$: أي \omterm{def:g3:division:remainder}{الباقي} $2$ في القسمة على $4$، فلا يكون
مضاعفًا أبدًا. والجبر يحوّل «يبدو أنّه يشتغل» إلى
\emph{دائمًا}، و«لم أجد مثالًا» إلى \emph{أبدًا} --- وهما كلمتان
لا يبلغهما أيّ قدر من الاِختبار
(\cref{met:g7:literal:test}).
\end{solution}
