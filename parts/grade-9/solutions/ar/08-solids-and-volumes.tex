\section*{الفصل \ref{ch:g9:solids} --- المجسّمات والحجوم}

\begin{solution}{exo:g9:solids:1}
العلبة: $V = 4 \times 5 \times 12 = 240$.

\omterm{def:g7:areas:prism}{والأسطوانة}: $V = \pi r^2 h = \pi \times 9 \times 7 = 63\pi \approx 198$.
\end{solution}

\begin{solution}{exo:g9:solids:2}
\omterm{def:g8:solids:def}{المخروط}: $V = \frac13 \pi r^2 h = \frac13 \pi \times 25 \times 12 =
100\pi \approx 314$.

\omterm{def:g8:solids:def}{والهرم}: \omterm{def:g6:measure:area}{مساحة} القاعدة $B = 8 \times 3 = 24$، إذن
$V = \frac13 \times 24 \times 10 = 80$.
\end{solution}

\begin{solution}{exo:g9:solids:3}
\omterm{def:g6:measure:volume}{الحجم}: $V = \frac43 \pi \times 6^3 = \frac43 \pi \times 216 = 288\pi$.

\omterm{def:g6:measure:area}{ومساحة} السطح: $4\pi \times 6^2 = 144\pi$.
\end{solution}

\begin{solution}{exo:g9:solids:4}
\omterm{def:g3:shapes:shapes}{نصف قطر} \omterm{prop:g9:solids:sections}{المقطع}: $\sqrt{r^2 - d^2} = \sqrt{100 - 64} = \sqrt{36}
= 6$.
\end{solution}

\begin{solution}{exo:g9:solids:5}
\omterm{def:g9:thales:scaling}{معامل التحجيم} من \omterm{def:g8:solids:def}{المخروط} كلّه إلى الصغير:
$k = \frac{9}{12} = \frac34$. \omterm{def:g3:shapes:shapes}{ونصف قطر} \omterm{prop:g9:solids:sections}{المقطع}: $8 \times \frac34 = 6$.

\omterm{def:g6:measure:volume}{وحجم} \omterm{def:g8:solids:def}{المخروط} كلّه: $\frac13\pi \times 64 \times 12 = 256\pi$.
\omterm{def:g6:measure:volume}{وحجم} \omterm{def:g8:solids:def}{المخروط} الصغير: $256\pi \times \left(\frac34\right)^3 =
256\pi \times \frac{27}{64} = 108\pi \approx 339$.
\end{solution}

\begin{solution}{exo:g9:solids:6}
\omterm{def:g6:measure:volume}{حجم} الكرة: $\frac43\pi \times 2^3 = \frac{32\pi}{3}$ سم$^3$.
وينتشر هذا \omterm{def:g6:measure:volume}{الحجم} على \omterm{def:g6:measure:area}{مساحة} القاعدة $\pi \times 3^2 = 9\pi$ سم$^2$،
فيرتفع المستوى بمقدار
\[
\frac{32\pi/3}{9\pi} = \frac{32}{27} \approx 1.2 \text{ سم}
\]
(أي نحو $12$ مم).
\end{solution}

\begin{solution}{exo:g9:solids:7}
القالب الصغير هو تصغير الكبير بالمعامل $k = \frac36 =
\frac12$، فحجمه $\left(\frac12\right)^3 = \frac18$ من
\omterm{def:g6:measure:volume}{حجم} الكبير: فتملأ الوصفة $8$ قوالب صغيرة.
\end{solution}

\begin{solution}{exo:g9:solids:8}
$V = \frac13 \times 230^2 \times 147 = \frac13 \times 52\,900 \times 147
= 52\,900 \times 49 = 2\,592\,100$ م$^3$ --- أي نحو $2.6$ مليون متر
مكعّب.
\end{solution}

\begin{solution}{exo:g9:solids:9}
\emph{1.} يشكّل الماء مخروطًا محجَّمًا بالمعامل $k = \frac12$
(فالرأس إلى الأسفل، ونصف الاِرتفاع)، فحجمه $\left(\frac12\right)^3 = \frac18$
من \omterm{def:g6:measure:volume}{حجم} القمع: أي الثمن، وهو أقلّ بكثير من النصف!

\emph{2.} والماء إلى الاِرتفاع $d$ يشكّل مخروطًا محجَّمًا بالمعامل
$k = \frac{d}{12}$، وحجمه $k^3$ مرّة \omterm{def:g6:measure:volume}{حجم} القمع. ونصف \omterm{def:g6:measure:volume}{الحجم}
يعني $k^3 = \frac12$، أي
$\left(\frac{d}{12}\right)^3 = \frac12$. ومنه
$\frac{d}{12} = \sqrt[3]{0.5} \approx 0.7937$، إذن
$d \approx 9.5$ سم: فالنصف الثاني من \omterm{def:g6:measure:volume}{الحجم} لا يشغل سوى أعلى
$2.5$ سم --- فالجزء العريض من \omterm{def:g8:solids:def}{المخروط} يحمل معظم الماء.
\end{solution}

\begin{solution}{pb:g9:solids:1}
\textbf{1.} $V_{\text{الأسطوانة}} = \pi r^2 \times 2r = 2\pi r^3$.

\textbf{2.} $\dfrac{V_{\text{الكرة}}}{V_{\text{الأسطوانة}}}
= \dfrac{\frac43 \pi r^3}{2 \pi r^3} = \dfrac23$. فيتلاشى $\pi$
و $r^3$: والنسبة \emph{كونية} --- صحيحة من أجل كرية زجاج ومن
أجل كوكب. وهي علاقة بين أشكال، لا بين أعداد: أي بالضبط نوع
الحقيقة التي تستحقّ أن تُنقش في الحجر.

\textbf{3.} الكرة: $4\pi r^2$. \omterm{def:g7:areas:prism}{والأسطوانة}: الجانب
$2\pi r \times 2r = 4\pi r^2$، مضافًا إليه الغطاءان
$2 \times \pi r^2$: أي \omterm{def:g1:addition:def}{المجموع} $6\pi r^2$. والنسبة:
$\frac{4\pi r^2}{6\pi r^2} = \frac23$ --- الثلثان أنفسهما،
من أجل السطوح. (وزيادة: \omterm{def:g6:measure:area}{مساحة} الكرة تساوي بالضبط
\omterm{def:g6:measure:area}{المساحة} الجانبية \omterm{def:g7:areas:prism}{للأسطوانة}.)

\textbf{4.} \omterm{def:g3:division:remainder}{الباقي}: $2\pi r^3 - \frac43\pi r^3 =
\frac23 \pi r^3$. \omterm{def:g8:solids:def}{ومخروطان} نصف قطرهما $r$ واِرتفاعهما $r$:
$2 \times \frac13 \pi r^2 \times r = \frac23 \pi r^3$. متساويان.

\textbf{5.} الكرة: $\frac43 \pi \times 12^3 \approx 7\,200$
سم$^3$؛ والعلبة: $2\pi \times 12^3 \approx 10\,900$ سم$^3$.
والفارغ: ثلث العلبة، أي نحو $33\,\%$ --- يضمنه
السؤال 2، مهما كان \omterm{def:g6:measure:volume}{حجم} الكرة.

\textbf{6.} نصف كرة:
$\frac12 \times \frac43 \pi \times 10^3 = \frac23 \pi \times
1\,000 \approx 2\,094$ سم$^3 \approx 2.1$ لتر.

\textbf{7.} \omterm{def:g8:solids:def}{المخروط}: $\frac13 \pi \times 10^2 \times 10 =
\frac{1\,000\pi}{3} \approx 1\,047$ سم$^3$. ونصف الكرة:
$\frac{2\,000\pi}{3} \approx 2\,094$ سم$^3$. \omterm{def:g7:areas:prism}{والأسطوانة}:
$1\,000\pi \approx 3\,142$ سم$^3$. والنسب:
$\frac{1000\pi}{3} : \frac{2000\pi}{3} : \frac{3000\pi}{3}
= 1 : 2 : 3$ بالضبط.

\textbf{8.} $\frac43 \pi \times 27 = \pi \times 9 \times h$
يعطي $h = \frac43 \times 3 = 4$ سم بالضبط.

\textbf{9.} نصفا \omterm{def:g4:geometry:circle}{القطر}: $\frac{6\,371}{1\,737} \approx 3.67$.
والحجوم تتحجّم كالمكعّب (\cref{thm:g9:solids:scaling}):
$3.67^3 \approx 49$. أي نحو خمسين قمرًا تسع في الأرض.

\textbf{10.} والسطوح تتحجّم كالمربّع: $3.67^2 \approx
13.5$. ومضاعفة \omterm{def:g3:shapes:shapes}{نصف قطر} بالون تضرب حجمه في
$2^3 = 8$ لكن سطحه في $2^2 = 4$ فقط: فحين تكبر الأشياء، يسبق
\omterm{def:g6:measure:volume}{الحجم} (الوزن، والمحتوى، والحرارة المنتَجة) السطح (الجلد،
والمادّة، والحرارة المفقودة) --- وهو قانون المربّع والمكعّب.

\textbf{11.} الوزن: $\times 10^3 = 1\,000$. ومقطع العظم:
$\times 10^2 = 100$. والضغط --- أي الوزن على \omterm{def:g6:measure:area}{مساحة} العظم:
$\times \frac{1000}{100} = 10$. فالعظام المبنية لضغط إنسان
تتلقّى عشرة أمثاله: فينكسر هيكل العملاق تحت وزنه هو. والعمالقة
مستحيلون هندسيًّا؛ والحيوانات الكبيرة تحتاج عظامًا سميكة سمكًا
غير متناسب (قارن قوائم الفيل بقوائم الغزال).

\textbf{12.} الوزن: $\div 100^3 = 10^6$. والقوّة (مقطع
العضلة): $\div 100^2 = 10^4$. والقوّة إلى الوزن:
$\times \frac{10^6}{10^4} = 100$. فالنملة التي ترفع خمسين مثل
وزنها ليست بطلة خارقة --- بل هي صغيرة فحسب؛ وإنسان مصغَّر إلى
\omterm{def:g6:measure:volume}{حجم} نملة كان ليفعل مثلها.

\textbf{13.} $\dfrac{S}{V} = \dfrac{4\pi r^2}{\frac43\pi r^3}
= \dfrac{3}{r}$: فمن أجل $r = 1$ تكون النسبة $3$، ومن أجل
$r = 2$ تكون $1.5$ --- فتنصيف \omterm{def:g6:measure:volume}{الحجم} يضاعف السطح المتاح لكل
وحدة من \omterm{def:g6:measure:volume}{الحجم} المنتِج للحرارة. والأجسام الصغيرة تنزف حرارة:
فالفأر مضطرّ إلى الأكل دائمًا، والدبّ يستطيع أن يشتو، والرضيع
--- كرة صغيرة، بنسبة $\frac Sr$ كبيرة --- يحتاج القبّعة.

\textbf{14.} $\left(\frac54\right)^3 = \frac{125}{64} \approx
1.95$: فالبرتقالة الكبيرة تحمل \emph{ضِعف} الفاكهة تقريبًا
بالثمن نفسه. فاِشترِ \omterm{def:g3:shapes:shapes}{نصف القطر}.

\textbf{15.} يقول النقش: الكرة $= \frac23$ من
\omterm{def:g7:areas:prism}{الأسطوانة} حجمًا، \emph{و} $\frac23$ منها في السطح الكلّي
--- أي نسبتان مضبوطتان كونيّتان تربطان \omterm{def:g5:solids:def}{المجسّم} الأكثر
اِستدارة بالأبسط. وجواب شيشرون: لأنّ المبرهنة هي النصب الوحيد
الذي لا تنال منه الجيوش ولا القرون --- فقد اِحترقت آلات
أرخميدس مع سيراكوزا، لكنّ الثلثين ما زالا ثلثين بالضبط.
\end{solution}
