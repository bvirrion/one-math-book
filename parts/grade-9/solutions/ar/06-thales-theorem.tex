\section*{الفصل \ref{ch:g9:thales} --- مبرهنة طاليس}

\begin{solution}{exo:g9:thales:1}
طاليس: $\dfrac{AM}{AB} = \dfrac{AN}{AC} = \dfrac{MN}{BC}$، أي
$\dfrac26 = \dfrac13$. إذن $\dfrac{3}{AC} = \dfrac13$ يعطي $AC = 9$،
و $\dfrac{MN}{9} = \dfrac13$ يعطي $MN = 3$.
\end{solution}

\begin{solution}{exo:g9:thales:2}
جِد أولًا $AB = AM + MB = 5 + 3 = 8$. ثم
$\dfrac{AM}{AB} = \dfrac{AN}{AC}$ يعطي $\dfrac58 = \dfrac{4}{AC}$،
إذن $5 \times AC = 32$ و $AC = 6.4$.
\end{solution}

\begin{solution}{exo:g9:thales:3}
تشتغل الفراشة تمامًا كالوضعية المتداخلة:
$\dfrac{AM}{AB} = \dfrac{AN}{AC} = \dfrac{MN}{BC}$، أي
$\dfrac{3}{7.5} = \dfrac25$. إذن $\dfrac{2}{AC} = \dfrac25$ يعطي
$AC = 5$، و $\dfrac{2.6}{BC} = \dfrac25$ يعطي
$BC = \dfrac{2.6 \times 5}{2} = 6.5$.
\end{solution}

\begin{solution}{exo:g9:thales:4}
$\dfrac{AM}{AB} = \dfrac68 = \dfrac34$
و $\dfrac{AN}{AC} = \dfrac{9}{12} = \dfrac34$: نسبتان متساويتان،
والنقط بالترتيب نفسه، إذن $(MN) \parallel (BC)$ بعكس مبرهنة طاليس.
\end{solution}

\begin{solution}{exo:g9:thales:5}
$\dfrac{AM}{AB} = \dfrac46 = \dfrac23$
و $\dfrac{AN}{AC} = \dfrac58$. وللمقارنة: $\dfrac23 = \dfrac{16}{24}$
و $\dfrac58 = \dfrac{15}{24}$: فالنسبتان مختلفتان. ولو كان
\omterm{def:g6:lines:objects}{المستقيمان} \omterm{def:g6:lines:perp}{متوازيين}، لفرضت مبرهنة طاليس تساويهما --- إذن
$(MN)$ و $(BC)$ \emph{ليسا} \omterm{def:g6:lines:perp}{متوازيين}.
\end{solution}

\begin{solution}{exo:g9:thales:6}
لتكن $S$ طرف الظلّ، و $P$ قمّة رأس الشخص، و $L$ قمّة المصباح.
فالشخص ($1.8$ م على بعد $3$ م من $S$) والمصباح (اِرتفاعه $h$ على
بعد $3 + 2 = 5$ م من $S$) متوازيان وهما \omterm{def:g6:lines:objects}{قطعتان} عموديتان
يقطعهما شعاع الضوء $(SL)$: فطاليس اِنطلاقًا من النقطة $S$ يعطي
\[
\frac{1.8}{h} = \frac{3}{5},
\qquad\text{إذن}\quad
h = \frac{1.8 \times 5}{3} = 3 \text{ م}.
\]
\end{solution}

\begin{solution}{exo:g9:thales:7}
\emph{1.} المسافة الحقيقية: $6.8 \times 25\,000 = 170\,000$ سم $=
1.7$ كلم.

\emph{2.} تتحجّم المساحات بالمعامل $k^2$: \omterm{def:g6:measure:area}{فالمساحة} الحقيقية
$= 8 \times 25\,000^2$ سم$^2$ $= 8 \times 6.25 \times 10^8$ سم$^2$
$= 5 \times 10^9$ سم$^2$. وبما أنّ $1$ كلم$^2 = 10^{10}$ سم$^2$، فهي $0.5$ كلم$^2$.
\end{solution}

\begin{solution}{exo:g9:thales:8}
\emph{1.} النسبة هي $\dfrac{AM}{AB} = \dfrac{8}{12} = \dfrac23$.
وطاليس: $AN = \frac23 \times 15 = 10$ و $MN = \frac23 \times 18 = 12$.

\emph{2.} \omterm{def:g9:thales:scaling}{معامل التحجيم} هو $k = \frac23$. والمحيطان: للمثلث $ABC$
\omterm{def:g6:measure:perimeter}{محيط} $12 + 15 + 18 = 45$، وللمثلث $AMN$ \omterm{def:g6:measure:perimeter}{محيط} $8 + 10 + 12 = 30 = \frac23 \times
45$: \omterm{def:g6:measure:perimeter}{فالمحيط} يتحجّم بالمعامل $k$ أيضًا، كأيّ طول.
\end{solution}

\begin{solution}{exo:g9:thales:9}
حول $O$، يتقاطع \omterm{def:g6:lines:objects}{المستقيمان} $(AC)$ و $(BD)$،
و $(AB) \parallel (CD)$: أي وضعية فراشة بالمثلثين $OAB$
و $OCD$. وتعطي مبرهنة طاليس
\[
\frac{OA}{OC} = \frac{OB}{OD} = \frac{AB}{CD} = \frac46 = \frac23 .
\]
إذن $\dfrac{OA}{OC} = \dfrac23$، والتساوي
$\dfrac{OA}{OC} = \dfrac{OB}{OD}$ يقول بالضبط إنّ $O$ يقطع
القطرين بالنسبة نفسها.
\end{solution}

\begin{solution}{pb:g9:thales:1}
\textbf{1.} يعطي الإنشاء نقطتين على $[AB]$؛
ولنسمِّهما $M_1$ (من $P_1$) و $M_2$ (من $P_2$).

\textbf{2.} في المثلث $A P_3 B$، يقطع \omterm{def:g6:lines:perp}{الموازيان} \omterm{def:g6:lines:objects}{للمستقيم}
$(P_3 B)$ المارّان بالنقطتين $P_1$ و $P_2$ الضلعين تقاطعًا
متناسبًا (\cref{thm:g9:thales:direct}):
\[
\frac{A M_1}{AB} = \frac{A P_1}{A P_3} = \frac13,
\qquad
\frac{A M_2}{AB} = \frac{A P_2}{A P_3} = \frac23 .
\]
وقد جعل الفرجار $A P_1$ و $P_1 P_2$ و $P_2 P_3$ متساوية، فالنسب
أثلاث بالضبط --- مهما كانت زاوية نصف \omterm{def:g6:lines:objects}{المستقيم} وفتحة الفرجار
المختارتان.

\textbf{3.} من أجل الأخماس: اِنقل خمسة أطوال متساوية، وصِل النقطة
الخامسة بالنقطة $B$، واُرسم أربعة \omterm{def:g6:lines:objects}{مستقيمات} موازية. ومن أجل
$\frac35$ من \omterm{def:g6:lines:objects}{قطعة}: الشكل نفسه، وخذ النقطة التي يقطعها \omterm{def:g6:lines:perp}{الموازي}
المارّ بالنقطة $P_3$: فهي عند $\frac{3}{5}$ من $[AB]$ اِنطلاقًا من $A$.

\textbf{4.} $\frac{AM}{MB} = \frac23$ تعني
$AM = \frac25 AB$: خمس خطوات متساوية على نصف \omterm{def:g6:lines:objects}{المستقيم}، وصِل $P_5$
بالنقطة $B$، فيعلّم \omterm{def:g6:lines:perp}{الموازي} المارّ بالنقطة $P_2$ النقطة $M$.

\textbf{5.} قياس $\sqrt2 = 1.41421\ldots$ سم لا يستعمل أبدًا سوى
عدد منتهٍ من الأرقام العشرية: فكل «ثلث» مقيس \omterm{def:g6:decimals:rounding}{تقريب}. أمّا إنشاء
طاليس فلا يقرأ عددًا قطّ: بل يعطي النقطة المضبوطة عند
$\frac{\sqrt2}{3}$ --- أجزاءً متساوية من \omterm{def:g6:lines:objects}{قطعة} لا تستطيع أيّ
مسطرة حتى التعبير عنها
(\cref{pb:g9:sqrt:1}).

\textbf{6.} قمّة الجسم والثقب وأسفل الصورة على اِستقامة واحدة،
وكذلك أسفل الجسم وقمّة الصورة: \omterm{def:g6:lines:objects}{مستقيمان} يتقاطعان عند الثقب،
والجسم وجدار الصورة متوازيان. وطاليس في الفراشة:
\[
\frac{h}{H} = \frac{d}{D},
\qquad\text{إذن}\qquad
h = H \times \frac dD .
\]

\textbf{7.} $h = 12 \times \frac{0.20}{30} = 0.08$ م $= 8$ سم
--- ومقلوبة (فالأشعّة تتقاطع عند الثقب).

\textbf{8.} $H = h \times \frac Dd = 9 \times 10^{-3} \times
\frac{1.5 \times 10^{11}}{1} = 1.35 \times 10^{9}$ م. والقيمة
الحقيقية $1.39 \times 10^9$ م: فعلبة أحذية تقيس الشمس في حدود
نسبة مئوية قليلة.

\textbf{9.} القمر: $\frac{3.5 \times 10^6}{3.8 \times 10^8}
\approx 0.0092$. والشمس: $\frac{1.39 \times 10^9}{1.5 \times
10^{11}} \approx 0.0093$. فالنسبتان --- وهما الحجمان الظاهريان
في السماء --- تكادان تتطابقان: فيستطيع القمر أن يغطّي الشمس
\emph{بالضبط}، حافّةً بحافّة. وتلك المصادفة الكونية هي الكسوف
الكلّي للشمس.

\textbf{10.} كل شعاع مضطرّ إلى المرور بالثقب الوحيد، فأشعّة قمّة
الجسم تمضي \emph{إلى الأسفل} وأشعّة أسفله تمضي \emph{إلى
الأعلى}: فتنقلب الصورة. وتوسيع الثقب يمرّر حزمة كاملة من نسخ
الصورة المزاحة قليلًا، فتتراكب: أسطع، لكن ملطّخة.

\textbf{11.} $\frac{PQ}{MN} = \frac{OP}{OM} = \frac{7.5}{3}
= 2.5$، إذن $PQ = 10$؛ و $\frac{ON}{OQ} = \frac{OM}{OP} =
\frac{1}{2.5}$، إذن $ON = \frac{6}{2.5} = 2.4$.

\textbf{12.} إذا مرّ \omterm{def:g6:lines:perp}{الموازي} بمنتصف أحد الضلعين، كانت نسبة طاليس
$\frac12$، فيقطع الضلع الثاني عند منتصفه \emph{هو}، وتقيس
\omterm{def:g6:lines:objects}{القطعة} الموازية نصف القاعدة: وهي بالضبط
\cref{thm:g8:midpoints:direct}. فطاليس هي مبرهنة المنتصفات
محرَّرةً من النسبة $\frac12$.

\textbf{13.} في المثلث $ACD$، \omterm{def:g6:lines:objects}{المستقيم} $(EO)$ \omterm{def:g6:lines:perp}{موازٍ} للقاعدة
$(CD)$، مع $\frac{AO}{AC} = \frac{OA}{OA + OC} =
\frac{4}{4 + 6} = \frac25$. وطاليس:
$\frac{EO}{DC} = \frac{AO}{AC} = \frac25$، إذن
$EO = \frac25 \times 6 = 2.4$.

\textbf{14.} وفي المثلث $BDC$، كذلك
$\frac{BO}{BD} = \frac25$ و $OF = \frac25 \times 6 = 2.4$.
ومنه
\[
EF = EO + OF = 4.8
= \frac{2 \times 4 \times 6}{4 + 6} :
\]
أي \omterm{pb:g8:speed:1}{المتوسط التوافقي} للقاعدتين --- متوسط رحلات الذهاب والإياب
(\cref{pb:g8:speed:1})، مرسومًا في شبه منحرف.

\textbf{15.} الحسابي: $5$؛ والهندسي:
$\sqrt{24} \approx 4.90$؛ والتوافقي: $4.8$. والترتيب:
$4.8 < 4.90\ldots < 5$، أي التوافقي $<$ الهندسي $<$
الحسابي --- ولا تساوي إلا عند تساوي القاعدتين. وفي الشكل: يقيس
الخط المتوسّط $5$، \omterm{def:g6:lines:perp}{والموازي} المارّ بتقاطع القطرين $4.8$، وقطع
التشابه $\sqrt{24}$، وكلّها مرصوفة بين القاعدتين بذلك الترتيب.
وقد بُرهن على المتباينات في \cref{pb:g8:cosine:1} (الهندسي
$\leq$ الحسابي، بالدائرة وبالمتطابقة) و\cref{pb:g8:speed:1}
(التوافقي $\leq$ الحسابي)؛ ويتبع التوافقي $\leq$ الهندسي
بتركيبهما ($H \times A = G^2$، وقد بُرهن عليه هناك أيضًا).
\end{solution}
