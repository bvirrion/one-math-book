\section*{الفصل \ref{ch:g9:statproba} --- الإحصاء والاحتمال}

\begin{solution}{exo:g9:statproba:1}
\omterm{def:g9:statproba:indicators}{المتوسّط}: $\dfrac{14 + 9 + 17 + 9 + 12 + 11 + 12}{7} = \dfrac{84}{7} = 12$.

\omterm{def:g9:statproba:indicators}{والوسيط}: بعد الترتيب، $9, 9, 11, 12, 12, 14, 17$؛ فالقيمة رقم $4$ هي
$12$.

\omterm{def:g9:statproba:indicators}{والمدى}: $17 - 9 = 8$.
\end{solution}

\begin{solution}{exo:g9:statproba:2}
\omterm{def:g9:statproba:indicators}{المتوسّط}: $\dfrac{19 + 21 + 24 + 18 + 22 + 25 + 19}{7} = \dfrac{148}{7}
\approx 21.1$ درجة.

\omterm{def:g9:statproba:indicators}{والوسيط}: بعد الترتيب، $18, 19, 19, 21, 22, 24, 25$؛ فالقيمة رقم $4$ هي $21$
درجة.
\end{solution}

\begin{solution}{exo:g9:statproba:3}
\omterm{def:g9:statproba:indicators}{المتوسّط}:
\[
\frac{2 \times 0 + 4 \times 1 + 3 \times 2 + 1 \times 3}{10}
= \frac{0 + 4 + 6 + 3}{10} = \frac{13}{10} = 1.3
\text{ هدفًا في المباراة.}
\]
\omterm{def:g9:statproba:indicators}{والوسيط}: السلسلة المرتّبة هي $0,0,1,1,1,1,2,2,2,3$؛ والقيمتان رقم $5$
ورقم $6$ كلتاهما $1$: أي \omterm{def:g9:statproba:indicators}{الوسيط} $1$.
\end{solution}

\begin{solution}{exo:g9:statproba:4}
$\P(3) = \frac16$. \quad
$\P(\text{زوجي}) = \frac36 = \frac12$. \quad
$\P(\text{على الأقل } 3) = \P(\{3,4,5,6\}) = \frac46 = \frac23$. \quad
$\P(\text{ليس } 6) = 1 - \frac16 = \frac56$.
\end{solution}

\begin{solution}{exo:g9:statproba:5}
$\P(\text{أبيض}) = \frac{5}{12}$.

$\P(\text{أحمر أو أخضر}) = \frac{4 + 3}{12} = \frac{7}{12}$.

$\P(\text{ليس أبيض}) = 1 - \frac{5}{12} = \frac{7}{12}$.

والأخيران متساويان: فحدث «ليس أبيض» \emph{هو} حدث «أحمر أو
أخضر».
\end{solution}

\begin{solution}{exo:g9:statproba:6}
المخارج $12$ (وجه \omterm{def:g6:lines:objects}{القطعة}، وقيمة النرد) متساوية
الإمكان.

صورة و $6$: مخرج واحد من $12$: أي $\frac{1}{12}$.

وصورة أو $6$: المخارج ذات الصورة (وعددها $6$) مضافًا إليها المخرج
(كتابة، $6$): أي $7$ مخارج، ومنه $\frac{7}{12}$. وعدّ كل حدث وطرح
التداخل يعطي النتيجة نفسها:
$\frac{6}{12} + \frac{2}{12} - \frac{1}{12} = \frac{7}{12}$.
\end{solution}

\begin{solution}{exo:g9:statproba:7}
السحب الأول: أحمر $\frac25$، وأسود $\frac35$. والسحب الثاني من
غير إعادة: فبعد حمراء، تبقى $1$ حمراء و $3$ سوداء من $4$؛ وبعد
سوداء، تبقى $2$ حمراء و $2$ سوداء من $4$.

\omterm{def:g6:lines:objects}{قطعتان} سوداوان: $\frac35 \times \frac24 = \frac{6}{20} = \frac{3}{10}$.

ولونان مختلفان (RB أو BR):
$\frac25 \times \frac34 + \frac35 \times \frac24
= \frac{6}{20} + \frac{6}{20} = \frac{12}{20} = \frac35$.
\end{solution}

\begin{solution}{exo:g9:statproba:8}
\emph{1.} $9 \times 14 = 126$ نقطة.

\emph{2.} \omterm{def:g9:statproba:indicators}{متوسّط} $15$ على $10$ مباريات يعني $150$ نقطة في
\omterm{def:g1:addition:def}{المجموع}: فيجب أن تسجّل $150 - 126 = 24$ نقطة.
\end{solution}

\begin{solution}{exo:g9:statproba:9}
\emph{1.} من بين الأزواج $36$ المتساوية الإمكان، تكون الأزواج
المتماثلة $(1,1), (2,2), \dots, (6,6)$: وعددها ستّة، إذن
$\P(\text{زوج}) = \frac{6}{36} = \frac16$.

\emph{2.} لعبتان مستقلّتان: شجرة بفوز ($\frac16$) وخسارة
($\frac56$) في كل مستوى. \omterm{def:g9:statproba:probability}{واِحتمال} عدم الفوز أبدًا:
$\frac56 \times \frac56 = \frac{25}{36}$. إذن
\[
\P(\text{فوز واحد على الأقل}) = 1 - \frac{25}{36} = \frac{11}{36}
\approx 0.31 .
\]
\end{solution}

\begin{solution}{pb:g9:statproba:1}
\textbf{1.} لكل رمية $5$ مخارج غير الستّة من $6$؛ ويبيّن الجدول
$6 \times 6$ أنّ $5 \times 5 = 25$ مخرجًا خاليًا من الستّة من
$36$: أي $P(\text{لا ستّة}) = \frac{25}{36}$، إذن
$P(\text{واحدة على الأقل}) = 1 - \frac{25}{36} = \frac{11}{36}
\approx 0.31$.

\textbf{2.} $\frac13 = \frac{12}{36} \neq \frac{11}{36}$. فالجمع
يعدّ المخرج «ستّة ثم ستّة» مرّة في كل $\frac16$: أي مرّة زائدة.
\omterm{def:g9:statproba:probability}{واِحتمالات} الأحداث التي يمكن أن تقع \emph{معًا} لا تُجمع
ببساطة.

\textbf{3.} \omterm{def:g1:addition:def}{المجموع} $7$: المخارج الستّة $(1,6), (2,5), (3,4),
(4,3), (5,2), (6,1)$: أي $P = \frac{6}{36} = \frac16$. \omterm{def:g1:addition:def}{والمجموع} $2$:
$(1,1)$ وحده: أي $\frac{1}{36}$. فللسبعة أكثر الطرق لتقع ---
وهي الوسط السمين في جدول المجاميع.

\textbf{4.} \omterm{def:g1:addition:def}{المجموع} $9$: $(3,6), (4,5), (5,4), (6,3)$ --- أربعة
مخارج، أي $\frac{4}{36}$. \omterm{def:g1:addition:def}{والمجموع} $10$: $(4,6), (5,5), (6,4)$ ---
ثلاثة، أي $\frac{3}{36}$. فالتسعة أرجح: إذ يحسم العدّ الأمر، لا
عدد «طرق كتابة» \omterm{def:g1:addition:def}{المجموع} \omterm{def:g5:solids:def}{بأوجه} غير مرتّبة.

\textbf{5.} يضرب كل مستوى من الشجرة \omterm{def:g9:statproba:probability}{اِحتمال} «لا ستّة» في
$\frac56$، مهما وقع من قبل: فبعد $n$ رمية، يكون
$P(\text{لا ستّة إطلاقًا}) = \left(\frac56\right)^n$، وتعطي
قاعدة الحدث المضادّ الصيغة.

\textbf{6.} $\left(\frac56\right)^4 = \frac{625}{1296} \approx
0.482$، إذن كان الفارس يفوز \omterm{def:g9:statproba:probability}{باِحتمال}
$1 - 0.482 = 0.518$ --- أي أفضلية هادئة ثابتة تقارب
$2\,\%$ في اللعبة: رهان ممتاز، وقد دفع ثمن عرباته.

\textbf{7.} وعند $7$ رميات تعطي قاعدته $\frac76$ --- أي
\omterm{def:g9:statproba:probability}{اِحتمالًا} أكبر من $1$، وهو عبث. وهو العدّ المزدوج في السؤال 2،
مركَّبًا: فنجاحات الرميات تتداخل، وجمع فرصها يعدّ التداخلات
مرّة بعد مرّة.

\textbf{8.} $1 - \left(\frac{35}{36}\right)^{24} \approx
1 - 0.5086 = 0.491$: أي دون النصف. وبالرهان بمال متكافئ على
حدث \omterm{def:g9:statproba:probability}{اِحتماله} $49.1\,\%$، كان الفارس يخسر نحو $2$ لعبة من كل
$100$ في \omterm{def:g9:statproba:indicators}{المتوسّط} --- ببطء، وبغموض (بالنسبة إليه)، ولا محالة.

\textbf{9.} $1 - \left(\frac{35}{36}\right)^{25} \approx
1 - 0.494 = 0.506$: فمع $25$ رمية يصير الرهان مواتيًا. ورمية
واحدة كانت تفصل الفارس عن الربح.

\textbf{10.} ضرب \omterm{def:g9:statproba:probability}{اِحتمال} في عدد المحاولات يعدّ النجاحات
المتداخلة عدّة مرّات، وينتج في النهاية أجوبة مستحيلة فوق $1$.
والطريق الصحيح يمرّ دائمًا بالحدث المضادّ: اِضرب \omterm{def:g9:statproba:probability}{اِحتمالات}
الفشل بعضها في بعض، ثم اِطرح من $1$.

\textbf{11.} \omterm{def:g9:statproba:indicators}{المتوسّط}: $\frac{9 \times 2\,000 + 20\,000}{10} =
\frac{38\,000}{10} = 3\,800$ يورو. \omterm{def:g9:statproba:indicators}{والوسيط}: الأجران الأوسطان
كلاهما $2\,000$: أي \omterm{def:g9:statproba:indicators}{الوسيط} $2\,000$ يورو. وينبغي أن يذكر
الإعلان \omterm{def:g9:statproba:indicators}{الوسيط} --- فتسعة من العشرة لا يرون شيئًا يشبه
$3\,800$؛ \omterm{def:g9:statproba:indicators}{والمتوسّط} تجرّه قيمة شاذّة واحدة إلى أعلى
(وقد حذّر \cref{ex:g9:statproba:weighted} من الأوزان).

\textbf{12.} مثلًا $2, 8, 15, 17, 18$: \omterm{def:g9:statproba:indicators}{فالوسيط} $15$
(القيمة الوسطى)، \omterm{def:g9:statproba:indicators}{والمتوسّط} $\frac{60}{5} = 12$. \omterm{def:g9:statproba:indicators}{ومتوسّط} دون
\omterm{def:g9:statproba:indicators}{الوسيط} يفضح ذيلًا من العلامات \emph{المنخفضة} يجرّ المعدّل إلى
أسفل بينما النصف الأعلى مرتفع.

\textbf{13.} العوامل الثلاثة الأولى:
$1 \times \frac{364}{365} \times \frac{363}{365} \approx
0.992$. وكل تلميذ جديد مضطرّ إلى تفادي تاريخ محجوز إضافي،
فتتقلّص العوامل: $\frac{343}{365} \approx 0.94$ للتلميذ
$23$. ويهبط \omterm{def:g2:mult:def}{الجداء} الكامل إلى نحو $0.493$ --- إذن
\[
P(\text{عيد ميلاد مشترك واحد على الأقل}) \approx 0.507 :
\]
أي أفضل من التكافؤ، في قسم فيه $23$ تلميذًا فقط. ويقول الحدس
«$23$ شخصًا، و $365$ يومًا، لا فرصة»؛ وتقول الرياضيات «اِرمِ
\omterm{def:g6:lines:objects}{قطعة} نقود».

\textbf{14.} $\frac{23 \times 22}{2} = 253$ زوجًا. وكل زوج فرصة
جديدة لمصادفة، و $253$ فرصة، كل واحدة $\frac{1}{365}$، لم تعد
شأنًا صغيرًا: فتتبدّد المفاجأة متى عُدّت الأزواج، لا الأشخاص.

\textbf{15.} بروتوكول مثال: اِرمِ نردًا في كتل من أربع رميات،
مسجّلًا ما إذا أظهرت كل \omterm{def:g2:measure:units}{كتلة} ستّة، من أجل $100$ \omterm{def:g2:measure:units}{كتلة}؛ أو اِجمع
قوائم أعياد الميلاد لعدّة أقسام فيها نحو $23$ تلميذًا وسجّل
نسبة الأقسام التي فيها مصادفة. \omterm{def:g7:stats:frequency}{والتواتر} الملاحَظ سيتذبذب،
لكن كلّما زاد عدد التكرارات وجب أن يستقرّ أقرب فأقرب من
\omterm{def:g9:statproba:probability}{الاِحتمالات} المحسوبة ($0.518$؛ و $0.507$) --- فالتواترات
تتقارب إلى \omterm{def:g9:statproba:probability}{الاِحتمالات}: وهو قانون الأعداد الكبيرة، مذكورًا هنا
توقّعًا ومبرهنًا عليه في مجلّد الثانوية.
\end{solution}
