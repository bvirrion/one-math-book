\begin{solution}{exo:g9:statproba:1}
\omterm{def:g9:statproba:indicators}{माध्य}: $\dfrac{14 + 9 + 17 + 9 + 12 + 11 + 12}{7} = \dfrac{84}{7} = 12$।

\omterm{def:g9:statproba:indicators}{माध्यिका}: क्रम से $9, 9, 11, 12, 12, 14, 17$; बीच वाला (चौथा) मान
$12$ है।

\omterm{def:g9:statproba:indicators}{परास}: $17 - 9 = 8$।
\end{solution}

\begin{solution}{exo:g9:statproba:2}

\approx 21.1$ डिग्री।

\omterm{def:g9:statproba:indicators}{माध्यिका}: क्रम से $18, 19, 19, 21, 22, 24, 25$; चौथा मान $21$ डिग्री

\end{solution}

\begin{solution}{exo:g9:statproba:3}
\omterm{def:g9:statproba:indicators}{माध्य}:
\[
\frac{2 \times 0 + 4 \times 1 + 3 \times 2 + 1 \times 3}{10}
= \frac{0 + 4 + 6 + 3}{10} = \frac{13}{10} = 1.3
\text{ गोल प्रति मुक़ाबला।}
\]
\omterm{def:g9:statproba:indicators}{माध्यिका}: क्रम से लगी श्रेणी $0,0,1,1,1,1,2,2,2,3$ है; पाँचवाँ और
छठा, दोनों मान $1$ हैं: यानी \omterm{def:g9:statproba:indicators}{माध्यिका} $1$।
\end{solution}

\begin{solution}{exo:g9:statproba:4}
$\P(3) = \frac16$। \quad
$\P(\text{सम}) = \frac36 = \frac12$। \quad
$\P(\text{कम से कम } 3) = \P(\{3,4,5,6\}) = \frac46 = \frac23$। \quad
$\P(6 \text{ नहीं}) = 1 - \frac16 = \frac56$।
\end{solution}

\begin{solution}{exo:g9:statproba:5}
$\P(\text{सफ़ेद}) = \frac{5}{12}$।

$\P(\text{लाल या हरी}) = \frac{4 + 3}{12} = \frac{7}{12}$।

$\P(\text{सफ़ेद नहीं}) = 1 - \frac{5}{12} = \frac{7}{12}$।

आख़िरी दोनों बराबर हैं: ``सफ़ेद नहीं'' दरअसल ``लाल या हरी'' घटना
\emph{ही} है।
\end{solution}

\begin{solution}{exo:g9:statproba:6}
बारहों परिणाम (सिक्के का \omterm{def:g5:solids:def}{फलक}, पासे का मान) समान रूप से संभावित हैं।

चित और $6$: बारह में से एक परिणाम: $\frac{1}{12}$।

चित या $6$: चित वाले परिणाम ($6$ हैं) और उनके साथ (पट, $6$) वाला परिणाम:
कुल $7$ परिणाम, यानी $\frac{7}{12}$। हर घटना को अलग गिनकर साझा
हिस्सा घटा देने पर भी वही मिलता है:
$\frac{6}{12} + \frac{2}{12} - \frac{1}{12} = \frac{7}{12}$।
\end{solution}

\begin{solution}{exo:g9:statproba:7}
पहला निकाल: लाल $\frac25$, काली $\frac35$। बिना वापस डाले दूसरा
निकाल: लाल आने के बाद $4$ में से $1$ लाल और $3$ काली बचती हैं; काली
आने के बाद $4$ में से $2$ लाल और $2$ काली।

दो काली: $\frac35 \times \frac24 = \frac{6}{20} = \frac{3}{10}$।

अलग-अलग रंग (लक या कल):
$\frac25 \times \frac34 + \frac35 \times \frac24
= \frac{6}{20} + \frac{6}{20} = \frac{12}{20} = \frac35$।
\end{solution}

\begin{solution}{exo:g9:statproba:8}
\emph{1.} $9 \times 14 = 126$ अंक।

\emph{2.} $10$ मुक़ाबलों में \omterm{def:g9:statproba:indicators}{माध्य} $15$ होने का मतलब है कुल $150$
अंक: यानी उसे $150 - 126 = 24$ अंक बनाने होंगे।
\end{solution}

\begin{solution}{exo:g9:statproba:9}
\emph{1.} समान रूप से संभावित $36$ जोड़ों में से जोड़े वाले परिणाम
$(1,1), (2,2), \dots, (6,6)$ हैं: कुल छह, इसलिए
$\P(\text{जोड़ा}) = \frac{6}{36} = \frac16$।

\emph{2.} दो स्वतंत्र खेल: हर स्तर पर जीत ($\frac16$) / हार
($\frac56$) वाला वृक्ष। एक बार भी न जीतने की \omterm{def:g9:statproba:probability}{प्रायिकता}:
$\frac56 \times \frac56 = \frac{25}{36}$। इसलिए
\[
\P(\text{कम से कम एक जीत}) = 1 - \frac{25}{36} = \frac{11}{36}
\approx 0.31 .
\]
\end{solution}

\begin{solution}{pb:g9:statproba:1}
\textbf{1.} हर फेंक में $6$ में से $5$ परिणाम छक्के के अलावा हैं;
$6 \times 6$ की सारणी में $36$ में से $5 \times 5 = 25$ परिणाम
छक्के से ख़ाली हैं: $\P(\text{कोई छक्का नहीं}) = \frac{25}{36}$,
इसलिए $\P(\text{कम से कम एक}) = 1 - \frac{25}{36} = \frac{11}{36}
\approx 0.31$।

\textbf{2.} $\frac13 = \frac{12}{36} \neq \frac{11}{36}$। वह जोड़
परिणाम ``छक्का फिर छक्का'' को दोनों $\frac16$ में एक-एक बार गिन
जाता है: यानी एक बार ज़्यादा। जो घटनाएँ \emph{एक साथ} हो सकती हैं,
उनकी प्रायिकताएँ यूँ ही नहीं जुड़तीं।

\textbf{3.} \omterm{def:g1:addition:def}{योगफल} $7$: छह परिणाम $(1,6), (2,5), (3,4),
(4,3), (5,2), (6,1)$: $\P = \frac{6}{36} = \frac16$। \omterm{def:g1:addition:def}{योगफल} $2$:
सिर्फ़ $(1,1)$: $\frac{1}{36}$। सात के होने के सबसे ज़्यादा रास्ते
हैं --- योगफलों की सारणी का मोटा बीच।

\textbf{4.} \omterm{def:g1:addition:def}{योगफल} $9$: $(3,6), (4,5), (5,4), (6,3)$ --- चार
परिणाम, $\frac{4}{36}$। \omterm{def:g1:addition:def}{योगफल} $10$: $(4,6), (5,5), (6,4)$ ---
तीन, $\frac{3}{36}$। नौ ज़्यादा संभव है: फ़ैसला परिणामों की \omterm{def:g7:stats:frequency}{गिनती}
करती है, यह नहीं कि बिना क्रम वाले फलकों से वह \omterm{def:g1:addition:def}{योगफल} कितने ढंग से
``लिखा'' जा सकता है।

\textbf{5.} वृक्ष का हर स्तर ``छक्का नहीं'' की \omterm{def:g9:statproba:probability}{प्रायिकता} को
$\frac56$ से गुणा कर देता है, पहले चाहे जो हुआ हो: यानी $n$ फेंकों
के बाद $\P(\text{एक भी छक्का नहीं}) = \left(\frac56\right)^n$, और
पूरक का नियम सूत्र दे देता है।

\textbf{6.} $\left(\frac56\right)^4 = \frac{625}{1296} \approx
0.482$, इसलिए शेवालिए $1 - 0.482 = 0.518$ \omterm{def:g9:statproba:probability}{प्रायिकता} से जीतते थे
--- हर खेल पर क़रीब $2\,\%$ की चुपचाप, टिकाऊ बढ़त: शानदार बाज़ी,
और उसी ने उनकी बग्घियों का ख़र्च उठाया।

\textbf{7.} $7$ फेंकों पर उनका नियम $\frac76$ देता है --- यानी
$1$ से बड़ी \omterm{def:g9:statproba:probability}{प्रायिकता}, जो बेतुकी है। यह सवाल 2 वाली दोहरी \omterm{def:g7:stats:frequency}{गिनती}
ही है, और भी बढ़ी हुई: अलग-अलग फेंकों की सफलताएँ आपस में ऊपर-नीचे
पड़ती हैं, और उनके मौक़े जोड़ते जाने से वे साझे हिस्से बार-बार गिने
जाते हैं।

\textbf{8.} $1 - \left(\frac{35}{36}\right)^{24} \approx
1 - 0.5086 = 0.491$: यानी आधे से नीचे। $49.1\,\%$ वाली घटना पर
बराबर दाँव लगाकर शेवालिए औसतन हर $100$ खेलों में क़रीब $2$ खेल
हारते थे --- धीरे-धीरे, (उनके लिए) रहस्यमय ढंग से, और लाज़मी तौर
पर।

\textbf{9.} $1 - \left(\frac{35}{36}\right)^{25} \approx
1 - 0.494 = 0.506$: $25$ फेंकों पर बाज़ी उनके हक़ में हो जाती।
सिर्फ़ एक फेंक शेवालिए को मुनाफ़े से अलग कर रही थी।

\textbf{10.} किसी \omterm{def:g9:statproba:probability}{प्रायिकता} को कोशिशों की \omterm{def:g7:stats:frequency}{गिनती} से गुणा कर देने पर
ऊपर-नीचे पड़ती सफलताएँ कई बार गिनी जाती हैं, और आख़िरकार $1$ से
बड़े नामुमकिन जवाब निकल आते हैं। सही रास्ता हमेशा पूरक से होकर जाता
है: नाकामी की प्रायिकताओं को गुणा करके एक लड़ी में बाँधो, फिर $1$
में से घटा दो।

\textbf{11.} \omterm{def:g9:statproba:indicators}{माध्य}: $\frac{9 \times 2\,000 + 20\,000}{10} =
\frac{38\,000}{10} = 3\,800$ यूरो। \omterm{def:g9:statproba:indicators}{माध्यिका}: बीच के दोनों वेतन
$2\,000$ हैं: यानी \omterm{def:g9:statproba:indicators}{माध्यिका} $2\,000$ यूरो। विज्ञापन में \omterm{def:g9:statproba:indicators}{माध्यिका} ही
छपनी चाहिए --- दसों में से नौ काम करने वालों को $3\,800$ जैसा कुछ
कभी दिखता ही नहीं; \omterm{def:g9:statproba:indicators}{माध्य} को अकेले एक बाहरी मान ने ऊपर खींच रखा है
(\cref{ex:g9:statproba:weighted} भार के बारे में चेता चुका है)।

\textbf{12.} जैसे $2, 8, 15, 17, 18$: \omterm{def:g9:statproba:indicators}{माध्यिका} $15$ (बीच वाला
मान), \omterm{def:g9:statproba:indicators}{माध्य} $\frac{60}{5} = 12$। \omterm{def:g9:statproba:indicators}{माध्यिका} से नीचे पड़ा \omterm{def:g9:statproba:indicators}{माध्य} बताता
है कि \emph{नीचे} के कुछ अंकों की पूँछ औसत को नीचे खींच रही है,
जबकि ऊपर वाला \omterm{def:g3:division:half}{आधा} हिस्सा ऊँचा बैठा है।

\textbf{13.} पहले तीन गुणक:
$1 \times \frac{364}{365} \times \frac{363}{365} \approx
0.992$। हर नए विद्यार्थी को एक और घिरी हुई तारीख़ से बचना पड़ता है,
इसलिए गुणक सिकुड़ते जाते हैं: तेईसवें के लिए
$\frac{343}{365} \approx 0.94$। पूरा \omterm{def:g2:mult:def}{गुणनफल} गिरकर क़रीब $0.493$ रह
जाता है --- इसलिए
\[
\P(\text{कम से कम एक साझा जन्मदिन}) \approx 0.507 :
\]
यानी आधे से भी ज़्यादा, और वह भी सिर्फ़ $23$ की कक्षा में। अटकल
कहती है ``$23$ लोग, $365$ दिन, कोई मौक़ा नहीं''; गणित कहता है
``सिक्का उछाल लो''।

\textbf{14.} $\frac{23 \times 22}{2} = 253$ जोड़े। हर जोड़ा संयोग
का एक नया मौक़ा है, और $\frac{1}{365}$ वाले $253$ मौक़े कोई छोटी
बात नहीं रह जाते: लोगों के बजाय जोड़े गिनते ही हैरानी घुल जाती है।

\textbf{15.} एक नमूना तरीक़ा: पासे को चार-चार फेंकों के गुच्छों में
फेंको और हर गुच्छे के लिए नोट करो कि उसमें छक्का आया या नहीं, ऐसे
$100$ गुच्छे; या फिर क़रीब $23$-$23$ विद्यार्थियों वाली कई कक्षाओं
की जन्मदिन-सूचियाँ जुटाओ और जिनमें संयोग निकला उनका अनुपात नोट
करो। देखी गई \omterm{def:g7:stats:frequency}{आवृत्ति} डगमगाएगी, पर दोहराव बढ़ने के साथ उसे निकाली
हुई प्रायिकताओं ($0.518$; $0.507$) के और भी पास जा बैठना चाहिए ---
आवृत्तियाँ प्रायिकताओं की ओर अभिसरित होती हैं: यही बृहत् संख्याओं
का नियम है, जो यहाँ एक उम्मीद के तौर पर कहा गया है और हाई-स्कूल
खंड में सिद्ध होता है।

\end{solution}
