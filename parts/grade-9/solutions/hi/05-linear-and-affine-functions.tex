\section*{अध्याय \ref{ch:g9:linfunc} --- रैखिक और एकघाती फलन}

\begin{solution}{exo:g9:linfunc:1}
रैखिक: $f$ और $k$ ($ax$ रूप वाले)। एकघाती पर रैखिक नहीं: $g$ (\omterm{prop:g9:linfunc:affinegraph}{ढलान}
$3$, अंतःखंड $-5$) और $m$ (अचर, \omterm{prop:g9:linfunc:affinegraph}{ढलान} $0$)। न वह, न यह: $h$, वर्ग की
वजह से।
\end{solution}

\begin{solution}{exo:g9:linfunc:2}
$f(0) = -3$; $f(4) = 5$; $f(-1) = -5$। और $f(x) = 9$ का मतलब
$2x - 3 = 9$, इसलिए $2x = 12$ और $x = 6$।
\end{solution}

\begin{solution}{exo:g9:linfunc:3}
$a = \frac{15}{6} = 2.5$, इसलिए $f(x) = 2.5x$। फिर $f(10) = 25$ और
$f(-2) = -5$।
\end{solution}

\begin{solution}{exo:g9:linfunc:4}
$f$ मूल बिंदु से होकर \omterm{prop:g9:linfunc:affinegraph}{ढलान} $2$ के साथ जाता है; $g$ की \omterm{prop:g9:linfunc:affinegraph}{ढलान} वही है पर
वह $y$-अक्ष को $3$ पर काटता है; $h$ घटता हुआ $(0, 4)$ से \omterm{prop:g9:linfunc:affinegraph}{ढलान} $-1$ के
साथ चलता है। $f$ और $g$ के आलेख \emph{\omterm{def:g6:lines:perp}{समांतर रेखाएँ}} हैं (एक ही \omterm{prop:g9:linfunc:affinegraph}{ढलान},
अलग-अलग अंतःखंड)।
\end{solution}

\begin{solution}{exo:g9:linfunc:5}
$f$ के लिए: \omterm{prop:g9:linfunc:affinegraph}{ढलान} $\frac{11 - 5}{3 - 1} = 3$, फिर $5 = 3 \times 1 + b$
से $b = 2$: यानी $f(x) = 3x + 2$।

$g$ के लिए: \omterm{prop:g9:linfunc:affinegraph}{ढलान} $\frac{0 - 4}{2 - 0} = -2$, और $g(0) = 4$ पहले से ही
अंतःखंड है: यानी $g(x) = -2x + 4$।
\end{solution}

\begin{solution}{exo:g9:linfunc:6}
$25\%$ की बढ़त: $120 \times 1.25 = 150$।

$40\%$ की छूट: $65 \times 0.6 = 39$।

असली क़ीमत $p$, जिसके लिए $p \times 0.75 = 69$:
$p = \frac{69}{0.75} = 92$।
\end{solution}

\begin{solution}{exo:g9:linfunc:7}
\emph{1.} $f(x) = 0.05x + 10$।

\emph{2.} योजना B तब सस्ती पड़ती है जब $25 < 0.05x + 10$ हो, यानी
$15 < 0.05x$, यानी $x > 300$। महीने के $301$ मिनट ($5$ घंटे) से आगे
योजना B जीत जाती है।
\end{solution}

\begin{solution}{exo:g9:linfunc:8}
\emph{1.} एक घंटे बाद: $2000 \times 1.1 = 2200$। दो घंटे बाद:
$2200 \times 1.1 = 2420$।

\emph{2.} दूसरी बढ़त $2200$ पर लगती है, $2000$ पर नहीं: दो बार $10\%$
बढ़ना $1.1^2 = 1.21$ से गुणा करना है, यानी $21\%$ की बढ़त, $20\%$ की
नहीं।
\end{solution}

\begin{solution}{exo:g9:linfunc:9}
\emph{1.} \omterm{prop:g9:linfunc:affinegraph}{ढलान}: $\frac{1 - 3}{6 - 2} = \frac{-2}{4} = -0.5$: यानी फलन
घटता है।

\emph{2.} $f(2) = 3$ के साथ $f(x) = -0.5x + b$: $-1 + b = 3$, इसलिए
$b = 4$ और $f(x) = -0.5x + 4$। मान $0$: $-0.5x + 4 = 0$ से $x = 8$।
\end{solution}

\begin{solution}{exo:g9:linfunc:10}
\emph{1.} दोनों बदलाव मिलकर क़ीमत को इससे गुणा करते हैं:
\[
\left(1 + \frac{t}{100}\right)\left(1 - \frac{t}{100}\right)
= 1 - \frac{t^2}{10000}
\]
(तीसरी सर्वसमिका, $a = 1$ और $b = \frac{t}{100}$ के साथ)।

\emph{2.} $0 < t \leq 100$ के लिए $\frac{t^2}{10000} > 0$ है, इसलिए
गुणक $1$ से छोटा है: आख़िरी क़ीमत असली से कम रहती है। और $t = 10$ के
लिए: गुणक $1 - \frac{100}{10000} = 0.99$, यानी कुल मिलाकर $1\%$ की
कमी।
\end{solution}

\begin{solution}{pb:g9:linfunc:1}
\textbf{1.} \omterm{prop:g9:linfunc:affinegraph}{ढलान}: $\frac{f(100) - f(0)}{100 - 0} =
\frac{212 - 32}{100} = 1.8$; अंतःखंड $f(0) = 32$। इसलिए
\[
f(C) = 1.8\,C + 32 .
\]

\textbf{2.} $f(20) = 68\,^\circ$F (वही मशहूर सुहाना दिन);
$f(37) = 98.6\,^\circ$F; $f(-10) = 14\,^\circ$F।

\textbf{3.} $F = 1.8C + 32$ से:
$C = \frac{F - 32}{1.8}$। तब $50\,^\circ$F
$\to \frac{18}{1.8} = 10\,^\circ$C, और $98.6\,^\circ$F
$\to \frac{66.6}{1.8} = 37\,^\circ$C।

\textbf{4.} $f(0) = 32 \neq 0$ है, और $f(20) = 68$, $f(10) = 50$ का
दुगना नहीं है: यानी रैखिक नहीं। आलेख एक सीधी रेखा है जो मूल बिंदु से
चूक जाती है: $f$ \emph{एकघाती} है, रैखिक नहीं।

\textbf{5.} $1.8x + 32 = x$ से $0.8x = -32$, यानी $x = -40$: चालीस
नीचे पर दोनों थर्मामीटर $-40$ ही दिखाते हैं। आलेख पर $f$ की रेखा
विकर्ण रेखा $y = x$ को ठीक वहीं काटती है।

\textbf{6.} $p_A(k) = 1.5k + 3$ और $p_B(k) = 2k$। $4$ km के लिए:
$9$ यूरो बनाम $8$: B सस्ती। $8$ km के लिए: $15$ बनाम $16$: A सस्ती।

\textbf{7.} $1.5k + 3 = 2k$ से $k = 6$: छह किलोमीटर पर दोनों $12$
यूरो पड़ती हैं। आलेख पर B की रेखा (मूल बिंदु से होकर, ज़्यादा तीखी)
A की रेखा से नीचे शुरू होती है और उसे $(6, 12)$ पर काटती है: छोटी
सवारियों में B जीतती है, लंबी में A।

\textbf{8.} $N = 7 \times 20 - 30 = 110$ बार प्रति मिनट। और
$82 = 7T - 30$ से: $T = \frac{112}{7} = 16\,^\circ$C।

\textbf{9.} $v(t) = 600 - 15t$। क़ीमत $240$: $600 - 15t = 240$ से
$t = 24$ महीने। क़ीमत $0$: $t = 40$ महीने। $40$ महीने के बाद सूत्र ऋण
हो जाता है --- पर फ़ोन की क़ीमत \omterm{def:g1:counting:zero}{शून्य} पर रुक जाती है: हर एकघाती नमूने
का एक दायरा होता है जिसके भीतर ही वह अर्थ रखता है।

\textbf{10.} एक के बाद एक: $\times 1.25$ फिर $\times 0.8$ का मतलब है
$1.25 \times 0.8 = 1$ से गुणा: क़ीमत ठीक अपनी शुरुआत पर लौट आती है।
इसमें चमत्कार कुछ नहीं: दोनों प्रतिशत अलग-अलग आधारों पर लगते हैं
(कटौती बढ़ी हुई क़ीमत पर लगती है), और यहाँ दोनों गुणक संयोग से कट
जाते हैं --- आम कहानी, अपनी अंदरूनी कमी
$1 - \frac{t^2}{10\,000}$ समेत, \cref{exo:g9:linfunc:10} है।

\textbf{11.} \omterm{prop:g9:linfunc:affinegraph}{ढलान}: $\frac{17 - 20}{30} = -0.1$ cm प्रति मिनट; यानी
$h(t) = 20 - 0.1t$। वह $h(t) = 0$ पर बुझती है: $t = 200$ मिनट ---
तीन घंटे बीस मिनट की रोशनी।

\textbf{12.} दूसरा $10$ km आगे से चला (अंतःखंड); पहला ज़्यादा तेज़ है
($24 > 18$ km/h)। पकड़ने का पल: $24t = 10 + 18t$ से $6t = 10$, यानी
$t = \frac53$ h $= 1$ h $40$ min, और वह किलोमीटर
$24 \times \frac53 = 40$ पर।

\textbf{13.} $f(x) + g(x) = (2 - 0.5)x + (3 + 7) = 1.5x + 10$: फिर
से एकघाती, जिसकी \omterm{prop:g9:linfunc:affinegraph}{ढलान} \emph{ढलानों का \omterm{def:g1:addition:def}{योगफल}} है और अंतःखंड अंतःखंडों
का \omterm{def:g1:addition:def}{योगफल} --- $(mx + p) + (m'x + p')$ में एक जैसे पद इकट्ठे करने से यह
साफ़ दिख जाता है।

\textbf{14.} अगर $m \neq 0$ हो: ठीक एक हल, $x = -\frac pm$ --- आड़ी
न होने वाली रेखा अक्ष को एक ही बार काटती है। अगर $m = 0$ और
$p \neq 0$ हो: कोई हल नहीं --- अक्ष के ऊपर या नीचे चलती आड़ी रेखा उसे
कभी छूती ही नहीं। और अगर $m = 0$ तथा $p = 0$ हो: हर $x$ हल है ---
रेखा ख़ुद अक्ष \emph{ही} है।

\textbf{15.} पहली तिकड़ी: दरें $\frac{9 - 5}{3 - 1} = 2$ और
$\frac{17 - 9}{7 - 3} = 2$: बराबर, यानी बिंदु संरेख हैं
($y = 2x + 3$ पर)। दूसरी तिकड़ी: $\frac{9-5}{2} = 2$ पर
$\frac{16 - 9}{3} = \frac73 \neq 2$: संरेख नहीं। \omterm{def:g9:linfunc:affine}{एकघाती फलन} ठीक वही
हैं जिनके बदलाव की दर अचर होती है --- एक ही संख्या, यानी \omterm{prop:g9:linfunc:affinegraph}{ढलान}, पूरी
कहानी कह देती है; टेढ़ी रेखाओं में दर ख़ुद बिंदु दर बिंदु बदलती है, और
उसका पीछा करते-करते हाई-स्कूल खंड में अवकलज तक पहुँचा जाता है।

\end{solution}
