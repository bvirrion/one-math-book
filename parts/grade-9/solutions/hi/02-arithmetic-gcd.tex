\section*{अध्याय \ref{ch:g9:arith} --- अंकगणित: भाजक और अभाज्य संख्याएँ}

\begin{solution}{exo:g9:arith:1}
$36$ के \omterm{def:g9:arith:divisor}{भाजक}: $1, 2, 3, 4, 6, 9, 12, 18, 36$।
$45$ के \omterm{def:g9:arith:divisor}{भाजक}: $1, 3, 5, 9, 15, 45$।
$17$ के \omterm{def:g9:arith:divisor}{भाजक}: सिर्फ़ $1$ और $17$ ($17$ अभाज्य है)।
\end{solution}

\begin{solution}{exo:g9:arith:2}
$2\,346$ के आख़िर में $6$ है: यह $2$ से \omterm{def:g6:wholes:divisible}{विभाज्य} है, $5$ से नहीं। अंकों
का \omterm{def:g1:addition:def}{योगफल} $2 + 3 + 4 + 6 = 15$: $3$ से \omterm{def:g6:wholes:divisible}{विभाज्य}, $9$ से नहीं। आख़िरी दो
अंक $46$ हैं, और $46 = 4 \times 11 + 2$, $4$ से \omterm{def:g6:wholes:divisible}{विभाज्य} नहीं है:
इसलिए $2\,346$ भी $4$ से \omterm{def:g6:wholes:divisible}{विभाज्य} नहीं है।
\end{solution}

\begin{solution}{exo:g9:arith:3}

$150 = 2 \times 3 \times 5^2$; \quad
$210 = 2 \times 3 \times 5 \times 7$; \quad

\end{solution}

\begin{solution}{exo:g9:arith:4}
$101$: उन अभाज्य संख्याओं $p$ को आज़माओ जिनके लिए $p^2 \leq 101$,
यानी $2, 3, 5, 7$। इनमें से कोई भी $101$ का \omterm{def:g9:arith:divisor}{भाजक} नहीं है (यह विषम है,
अंकों का \omterm{def:g1:addition:def}{योगफल} $2$ है, आख़िर में $0$ या $5$ नहीं है,
$101 = 7 \times 14 + 3$): इसलिए $101$ अभाज्य है।

$91 = 7 \times 13$: अभाज्य नहीं।

$143 = 11 \times 13$: अभाज्य नहीं।
\end{solution}

\begin{solution}{exo:g9:arith:5}
$48$ और $60$ के साझा \omterm{def:g9:arith:divisor}{भाजक}: $48$ के \omterm{def:g9:arith:divisor}{भाजक} $1$, $2$, $3$, $4$, $6$,
$8$, $12$, $16$, $24$, $48$ हैं; $60$ के \omterm{def:g9:arith:divisor}{भाजक} $1$, $2$, $3$, $4$,
$5$, $6$, $10$, $12$, $15$, $20$, $30$, $60$ हैं; साझा \omterm{def:g9:arith:divisor}{भाजक} $1$,
$2$, $3$, $4$, $6$, $12$ हैं, इसलिए $\gcd(48,60) = 12$।

गुणनखंडन से: $48 = 2^4 \times 3$ और $60 = 2^2 \times 3 \times 5$;
साझा अभाज्य संख्याएँ छोटे घातांकों के साथ: $2^2 \times 3 = 12$।
\end{solution}

\begin{solution}{exo:g9:arith:6}
$\gcd(255, 154)$:
\begin{align*}
255 &= 154 \times 1 + 101, \\
154 &= 101 \times 1 + 53, \\
101 &= 53 \times 1 + 48, \\
53 &= 48 \times 1 + 5, \\
48 &= 5 \times 9 + 3, \\
5 &= 3 \times 1 + 2, \\
3 &= 2 \times 1 + 1, \\
2 &= 1 \times 2 + 0 .
\end{align*}
आख़िरी शून्येतर \omterm{def:g3:division:remainder}{शेषफल}: $\gcd(255, 154) = 1$ (ये \omterm{def:g9:arith:gcd}{सह-अभाज्य} हैं)।

$\gcd(1053, 325)$:
\begin{align*}
1053 &= 325 \times 3 + 78, \\
325 &= 78 \times 4 + 13, \\
78 &= 13 \times 6 + 0 .
\end{align*}
$\gcd(1053, 325) = 13$.
\end{solution}

\begin{solution}{exo:g9:arith:7}
यूक्लिड की विधि: $588 = 504 \times 1 + 84$; $504 = 84 \times 6 + 0$:
इसलिए $\gcd(588, 504) = 84$। फिर
\[
\frac{588}{504} = \frac{588 \div 84}{504 \div 84} = \frac{7}{6},
\]
जो अलघुकरणीय है।
\end{solution}

\begin{solution}{exo:g9:arith:8}
गुलदस्तों की \omterm{def:g7:stats:frequency}{गिनती} $84$ और $126$ दोनों का \omterm{def:g9:arith:divisor}{भाजक} होनी चाहिए; सबसे बड़ी
संभव \omterm{def:g7:stats:frequency}{गिनती} $\gcd(84, 126)$ है। गुणनखंडन: $84 = 2^2 \times 3 \times 7$
और $126 = 2 \times 3^2 \times 7$, इसलिए \omterm{def:g9:arith:gcd}{महत्तम समापवर्तक}
$2 \times 3 \times 7 = 42$ है। वह $42$ गुलदस्ते बना सकती है, और हर एक
में $\frac{84}{42} = 2$ गुलाब तथा $\frac{126}{42} = 3$ ट्यूलिप होंगे।
\end{solution}

\begin{solution}{exo:g9:arith:9}
हमें सबसे छोटा साझा \omterm{def:g9:arith:divisor}{गुणज} चाहिए। $24 = 2^3 \times 3$ और
$36 = 2^2 \times 3^2$; हर \omterm{def:g9:arith:prime}{अभाज्य संख्या} को \emph{बड़े} \omterm{def:g8:powers:def}{घातांक} के साथ
लेने पर $\lcm = 2^3 \times 3^2 = 72$। नावें अगली बार 8:00 के $72$
मिनट बाद, यानी 9:12 पर एक साथ छूटेंगी।
\end{solution}

\begin{solution}{exo:g9:arith:10}
\emph{1.} $n$ और $n+1$ का जो भी साझा \omterm{def:g9:arith:divisor}{भाजक} $d$ हो, वह उनके \omterm{ex:g1:subtraction:difference}{अंतर}
$(n+1) - n = 1$ का भी \omterm{def:g9:arith:divisor}{भाजक} है, इसलिए $d = 1$: $\gcd(n, n+1) = 1$।

\emph{2.} कोई \omterm{def:g9:fractions:fraction}{भिन्न} अलघुकरणीय ठीक तभी होती है जब उसका \omterm{def:g4:fractions:def}{अंश} और हर
\omterm{def:g9:arith:gcd}{सह-अभाज्य} हों, और भाग 1 के अनुसार $n$ तथा $n + 1$ के साथ यही हाल है।
\end{solution}

\begin{solution}{pb:g9:arith:1}
\textbf{1.} $3$ वाला घड़ा भरो और उसे $5$ वाले में उड़ेल दो (बड़े में
$0$ से $3$)। $3$ वाला फिर भरो और $5$ वाले में तब तक उड़ेलो जब तक वह
भर न जाए: बड़े घड़े में सिर्फ़ $2$ और समाते हैं, इसलिए बचते हैं
\[
3 - 2 = 1 \text{ L छोटे घड़े में।}
\]
चालें: $3$ भरो; $3 \to 5$ उड़ेलो; $3$ फिर भरो; $3 \to 5$ उड़ेलो।

\textbf{2.} $5$ वाला भरो; उसे $3$ वाले में उड़ेलो (बड़े में $2$ बचते
हैं); $3$ वाला ख़ाली करो; वे $2$ लीटर $3$ वाले में डाल दो; $5$ वाला
भरो; उसे $3$ वाले में तब तक उड़ेलो जब तक वह भर न जाए --- उसमें $1$
जाता है, और बड़े घड़े में $\mathbf{4}$ L बच जाते हैं। छह चालें।

\textbf{3.} सब के सब: $1$ (सवाल 1), $2$ (सवाल 2 की दो चालों के बाद),
$3$ और $5$ (एक-एक घड़ा भरकर), $4$ (सवाल 2), $6 = 3 + 3$ (भरा हुआ छोटा
घड़ा और बड़े में उड़ेले हुए $3$), $7 = 5 + 2$, $8 = 5 + 3$ (दोनों भरे
हुए)। $1$ से $8$ L तक हर पूरी मात्रा $5$ और $3$ वाले घड़ों से नापी जा
सकती है।

\textbf{4.} शुरू में दोनों घड़ों में $0$ है, जो $2$ का \omterm{def:g9:arith:divisor}{गुणज} है। भरने
पर किसी घड़े में $6$ या $4$ आता है: सम। ख़ाली करने पर $0$: सम। उड़ेलने
पर पानी उन घड़ों के बीच जाता है जिनकी मात्राएँ सम थीं, और जितना
उड़ेला जाता है वह \omterm{def:g3:numbers:evenodd}{सम संख्याओं} का \omterm{ex:g1:subtraction:difference}{अंतर} है (बची हुई जगह, या मौजूद
पानी): इसलिए सारी मात्राएँ हमेशा सम ही रहती हैं। $1$ L जैसा विषम
लक्ष्य पहुँच से बाहर है।

\textbf{5.} $\gcd(6, 4) = 2$: सिर्फ़ सम मात्राएँ --- सवाल 4 ने यही
पक्का किया। $\gcd(5, 3) = 1$: नियम हर पूरी मात्रा की इजाज़त देता है,
और सवाल 3 ने वे सब बना भी दीं। \omterm{def:g9:arith:gcd}{महत्तम समापवर्तक} ही घड़ों की नाप की
इकाई है।

\textbf{6.} $91 = 1 \times 65 + 26$; $65 = 2 \times 26 + 13$;
$26 = 2 \times 13 + 0$: इसलिए $\gcd(91, 65) = 13$। और
$2\,026 = 44 \times 46 + 2$; $46 = 23 \times 2 + 0$:
$\gcd(2\,026, 46) = 2$।

\textbf{7.} $5$ वाले घड़े को $13$ वाले में बार-बार उड़ेलो: दो बार भरने
के बाद बड़े में $10$ हैं; तीसरी बार भरने पर उसमें सिर्फ़ $3$ समाते हैं,
और छोटे में $5 - 3 = 2$ बच जाते हैं --- $13$ में $5$ का \omterm{def:g3:division:remainder}{शेषफल} $3$
था, और मात्राएँ $3$ (बची जगह) तथा $2$ (बचा पानी) ठीक यूक्लिड की
संख्याएँ हैं ($13 = 2 \times 5 + 3$, $5 = 1 \times 3 + 2$)। आगे चलने
पर $3 - 2 = 1$ उभरता है: विधि का अगला \omterm{def:g3:division:remainder}{शेषफल}। फ़व्वारा यूक्लिड के भाग
पानी से कर रहा है।

\textbf{8.} $\gcd(13, 5) = 1$, इसलिए सवाल 5 का नियम हर पूरी मात्रा की
इजाज़त देता है --- और सवाल 7 की लड़ी ने सचमुच $1$ L बना भी दिया। हाँ।

\textbf{9.} $n$ और $2n + 1$ का कोई भी साझा \omterm{def:g9:arith:divisor}{भाजक}
$2n + 1 - 2 \times n = 1$ का भी \omterm{def:g9:arith:divisor}{भाजक} है: इसलिए वह $1$ ही हो सकता है।
यानी हमेशा $\gcd(n, 2n+1) = 1$।

\textbf{10.} पहली बस: 7:12, 7:24, 7:36, 7:48, 8:00\,\dots; दूसरी बस:
7:18, 7:36, 7:54\,\dots\ पहली साझा रवानगी: 7:36, यानी $36$ मिनट बाद
--- $12$ और $18$ का पहला साझा \omterm{def:g9:arith:divisor}{गुणज}। नियम:
$36 \times \gcd(12, 18) = 36 \times 6 = 216 = 12 \times
18$। $5$ और $3$ के लिए: पहला साझा \omterm{def:g9:arith:divisor}{गुणज} $15$, और
$15 \times \gcd(5,3) = 15 \times 1 = 15 = 5 \times 3$।

\textbf{11.} $17$ साल के चक्र के साथ: अगली मुलाक़ात $17$ और $4$ के
पहले साझा \omterm{def:g9:arith:divisor}{गुणज} पर होगी; और चूँकि $\gcd(17, 4) = 1$ है, वह
$17 \times 4 = 68$ साल बाद है --- यानी चार बार बाहर आने में एक ही बार
सिकाडा चरम से टकराते हैं। $16$ साल के चक्र के साथ: $16$, $4$ का \omterm{def:g9:arith:divisor}{गुणज}
है, इसलिए \emph{हर} बार बाहर आना किसी चरम से टकराएगा। अभाज्य लंबाई
वाला चक्र किसी भी छोटे शिकारी-चक्र के साथ कोई गुणनखंड साझा नहीं करता,
इसलिए मुलाक़ातें जितनी दूर-दूर हो सकती हैं उतनी हो जाती हैं: अंकगणित
ही छलावरण बन जाता है।

\textbf{12.} $2$ के लिए \omterm{def:g8:powers:def}{घातांक} के चार विकल्प, $3$ के लिए तीन और $5$
के लिए दो: कुल $4 \times 3 \times 2 = 24$ \omterm{def:g9:arith:divisor}{भाजक}।

\textbf{13.} अगर $m = 2^{a} \times 3^{b} \times \cdots$ हो, तो
$m^2 = 2^{2a} \times 3^{2b} \times \cdots$: हर \omterm{def:g8:powers:def}{घातांक} दुगना हो जाता
है, यानी सम। पर $360 = 2^3 \times 3^2 \times 5$ में $2$ और $5$ के
\omterm{def:g8:powers:def}{घातांक} विषम हैं: इसलिए $360$ पूर्ण वर्ग नहीं है।

\textbf{14.} कोई \omterm{def:g9:arith:divisor}{भाजक} अपना ही साथी तभी होता है जब $d = \frac nd$ हो,
यानी $n = d^2$: ऐसा बीच वाला \omterm{def:g9:arith:divisor}{भाजक} सिर्फ़ वर्गों के पास होता है। बाक़ी
हर $n$ के \omterm{def:g9:arith:divisor}{भाजक} जोड़ों में बँट जाते हैं, यानी उनकी \omterm{def:g7:stats:frequency}{गिनती} सम होती है।
इसलिए: भाजकों की विषम \omterm{def:g7:stats:frequency}{गिनती} $\Leftrightarrow$ पूर्ण वर्ग। जाँच: $36$
के \omterm{def:g9:arith:divisor}{भाजक} $1, 2, 3, 4, 6, 9, 12, 18, 36$ हैं --- नौ, यानी विषम, और
$36 = 6^2$; जबकि $360$ के $24$ \omterm{def:g9:arith:divisor}{भाजक} हैं (सवाल 12), यानी सम, और वह
कोई वर्ग नहीं है (सवाल 13)।

\textbf{15.} लॉकर $n$ को हर वह विद्यार्थी $k$ एक बार पलटता है जिसकी
संख्या $n$ का \omterm{def:g9:arith:divisor}{भाजक} है: यानी कुल उतनी बार जितने $n$ के \omterm{def:g9:arith:divisor}{भाजक} हैं। कोई
लॉकर आख़िर में \emph{खुला} तब रहता है जब वह विषम बार पलटा गया हो ---
सवाल 14 के अनुसार ठीक तभी जब $n$ पूर्ण वर्ग हो। खुले लॉकर:
$1, 4, 9, 16, 25, 36, 49, 64, 81, 100$ --- कुल दस।

\end{solution}
