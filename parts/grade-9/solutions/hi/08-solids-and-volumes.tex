\section*{अध्याय \ref{ch:g9:solids} --- ठोस और आयतन}

\begin{solution}{exo:g9:solids:1}
\omterm{def:g5:solids:def}{घनाभ}: $V = 4 \times 5 \times 12 = 240$।

\omterm{def:g7:areas:prism}{बेलन}: $V = \pi r^2 h = \pi \times 9 \times 7 = 63\pi \approx 198$।
\end{solution}

\begin{solution}{exo:g9:solids:2}
\omterm{def:g8:solids:def}{शंकु}: $V = \frac13 \pi r^2 h = \frac13 \pi \times 25 \times 12 =
100\pi \approx 314$।

\omterm{def:g8:solids:def}{पिरामिड}: आधार का \omterm{def:g6:measure:area}{क्षेत्रफल} $B = 8 \times 3 = 24$, इसलिए
$V = \frac13 \times 24 \times 10 = 80$।
\end{solution}

\begin{solution}{exo:g9:solids:3}
\omterm{def:g6:measure:volume}{आयतन}: $V = \frac43 \pi \times 6^3 = \frac43 \pi \times 216 = 288\pi$।

सतह का \omterm{def:g6:measure:area}{क्षेत्रफल}: $4\pi \times 6^2 = 144\pi$।
\end{solution}

\begin{solution}{exo:g9:solids:4}
काट की \omterm{def:g3:shapes:shapes}{त्रिज्या}: $\sqrt{r^2 - d^2} = \sqrt{100 - 64} = \sqrt{36}
= 6$।
\end{solution}

\begin{solution}{exo:g9:solids:5}
पूरे \omterm{def:g8:solids:def}{शंकु} से छोटे \omterm{def:g8:solids:def}{शंकु} तक का \omterm{def:g9:thales:scaling}{मापक गुणक}:
$k = \frac{9}{12} = \frac34$। काट की \omterm{def:g3:shapes:shapes}{त्रिज्या}: $8 \times \frac34 = 6$।

पूरे \omterm{def:g8:solids:def}{शंकु} का \omterm{def:g6:measure:volume}{आयतन}: $\frac13\pi \times 64 \times 12 = 256\pi$।
छोटे \omterm{def:g8:solids:def}{शंकु} का \omterm{def:g6:measure:volume}{आयतन}: $256\pi \times \left(\frac34\right)^3 =
256\pi \times \frac{27}{64} = 108\pi \approx 339$।
\end{solution}

\begin{solution}{exo:g9:solids:6}
गेंद का \omterm{def:g6:measure:volume}{आयतन}: $\frac43\pi \times 2^3 = \frac{32\pi}{3}$ cm$^3$। यह
\omterm{def:g6:measure:volume}{आयतन} आधार के \omterm{def:g6:measure:area}{क्षेत्रफल} $\pi \times 3^2 = 9\pi$ cm$^2$ पर फैल जाता
है, इसलिए सतह इतनी ऊपर उठती है:
\[
\frac{32\pi/3}{9\pi} = \frac{32}{27} \approx 1.2 \text{ cm}
\]
(यानी क़रीब $12$ mm)।
\end{solution}

\begin{solution}{exo:g9:solids:7}
छोटा साँचा बड़े का $k = \frac36 = \frac12$ गुणक वाला घटा हुआ रूप है,
इसलिए उसका \omterm{def:g6:measure:volume}{आयतन} बड़े के $\left(\frac12\right)^3 = \frac18$ के बराबर
है: यानी वह विधि $8$ छोटे साँचे भर देगी।
\end{solution}

\begin{solution}{exo:g9:solids:8}
$V = \frac13 \times 230^2 \times 147 = \frac13 \times 52\,900 \times 147
= 52\,900 \times 49 = 2\,592\,100$ m$^3$ --- यानी क़रीब $2.6$ मिलियन
\omterm{def:g5:solids:def}{घन} मीटर।
\end{solution}

\begin{solution}{exo:g9:solids:9}
\emph{1.} पानी $k = \frac12$ गुणक वाला \omterm{def:g8:solids:def}{शंकु} बनाता है (शिखर नीचे,
ऊँचाई आधी), इसलिए उसका \omterm{def:g6:measure:volume}{आयतन} क़ीप के \omterm{def:g6:measure:volume}{आयतन} का
$\left(\frac12\right)^3 = \frac18$ है: एक बटा आठ, आधे से कहीं कम!

\emph{2.} ऊँचाई $d$ तक भरा पानी $k = \frac{d}{12}$ गुणक वाला \omterm{def:g8:solids:def}{शंकु}
बनाता है, जिसका \omterm{def:g6:measure:volume}{आयतन} क़ीप के \omterm{def:g6:measure:volume}{आयतन} का $k^3$ गुना है। \omterm{def:g3:division:half}{आधा} \omterm{def:g6:measure:volume}{आयतन} होने का
मतलब है $k^3 = \frac12$, यानी
$\left(\frac{d}{12}\right)^3 = \frac12$। तब
$\frac{d}{12} = \sqrt[3]{0.5} \approx 0.7937$, यानी
$d \approx 9.5$ cm: \omterm{def:g6:measure:volume}{आयतन} का दूसरा \omterm{def:g3:division:half}{आधा} हिस्सा सिर्फ़ ऊपर वाले
$2.5$ cm में समा जाता है --- \omterm{def:g8:solids:def}{शंकु} का चौड़ा हिस्सा ही ज़्यादातर पानी
रखता है।
\end{solution}

\begin{solution}{pb:g9:solids:1}
\textbf{1.} $V_{\text{बेलन}} = \pi r^2 \times 2r = 2\pi r^3$।

\textbf{2.} $\dfrac{V_{\text{गोला}}}{V_{\text{बेलन}}}
= \dfrac{\frac43 \pi r^3}{2 \pi r^3} = \dfrac23$। $\pi$ और $r^3$
दोनों कट जाते हैं: यह \omterm{def:g7:prop:table}{समानुपात} \emph{सार्वभौम} है --- कंचे पर भी
उतना ही सच, ग्रह पर भी। यह संख्याओं का नहीं, आकारों का रिश्ता है:
ठीक वैसा सच जिसे पत्थर पर खुदवाया जाए।

\textbf{3.} गोला: $4\pi r^2$। \omterm{def:g7:areas:prism}{बेलन}: बग़ली सतह
$2\pi r \times 2r = 4\pi r^2$, और उस पर दो ढक्कन
$2 \times \pi r^2$: कुल $6\pi r^2$। अनुपात:
$\frac{4\pi r^2}{6\pi r^2} = \frac23$ --- वही दो तिहाई, इस बार
सतहों के लिए। (और एक मुफ़्त की बात: गोले का \omterm{def:g6:measure:area}{क्षेत्रफल} ठीक \omterm{def:g7:areas:prism}{बेलन} की
बग़ली सतह के बराबर निकलता है।)

\textbf{4.} बची जगह: $2\pi r^3 - \frac43\pi r^3 =
\frac23 \pi r^3$। और $r$ \omterm{def:g3:shapes:shapes}{त्रिज्या} तथा $r$ ऊँचाई वाले दो \omterm{def:g8:solids:def}{शंकु}:
$2 \times \frac13 \pi r^2 \times r = \frac23 \pi r^3$। बराबर।

\textbf{5.} गेंद: $\frac43 \pi \times 12^3 \approx 7\,200$
cm$^3$; डिब्बा: $2\pi \times 12^3 \approx 10\,900$ cm$^3$। ख़ाली:
डिब्बे का एक तिहाई, यानी क़रीब $33\,\%$ --- और यह सवाल 2 की
गारंटी है, गेंद चाहे जितनी बड़ी हो।

\textbf{6.} गोले का \omterm{def:g3:division:half}{आधा}:
$\frac12 \times \frac43 \pi \times 10^3 = \frac23 \pi \times
1\,000 \approx 2\,094$ cm$^3 \approx 2.1$ L।

\textbf{7.} \omterm{def:g8:solids:def}{शंकु}: $\frac13 \pi \times 10^2 \times 10 =
\frac{1\,000\pi}{3} \approx 1\,047$ cm$^3$। अर्धगोला:
$\frac{2\,000\pi}{3} \approx 2\,094$ cm$^3$। \omterm{def:g7:areas:prism}{बेलन}:
$1\,000\pi \approx 3\,142$ cm$^3$। अनुपात:
$\frac{1000\pi}{3} : \frac{2000\pi}{3} : \frac{3000\pi}{3}
= 1 : 2 : 3$, बिलकुल ठीक।

\textbf{8.} $\frac43 \pi \times 27 = \pi \times 9 \times h$ से
$h = \frac43 \times 3 = 4$ cm, बिलकुल सही।

\textbf{9.} \omterm{def:g3:shapes:shapes}{त्रिज्याएँ}: $\frac{6\,371}{1\,737} \approx 3.67$।
\omterm{def:g6:measure:volume}{आयतन} \omterm{def:g5:solids:def}{घन} के हिसाब से बदलते हैं (\cref{thm:g9:solids:scaling}):
$3.67^3 \approx 49$। यानी क़रीब पचास चाँद धरती में समा जाएँगे।

\textbf{10.} सतहें वर्ग के हिसाब से बदलती हैं:
$3.67^2 \approx 13.5$। किसी गुब्बारे की \omterm{def:g3:shapes:shapes}{त्रिज्या} दुगनी करने पर उसका
\omterm{def:g6:measure:volume}{आयतन} $2^3 = 8$ गुना होता है, पर सतह सिर्फ़ $2^2 = 4$ गुना: चीज़ें
जैसे-जैसे बड़ी होती हैं, \omterm{def:g6:measure:volume}{आयतन} (भार, समाई, बनी ऊष्मा) सतह (खाल,
सामग्री, खोई ऊष्मा) से आगे निकल जाता है --- यही वर्ग--घन नियम है।

\textbf{11.} भार: $\times 10^3 = 1\,000$ गुना। हड्डी की \omterm{prop:g9:solids:sections}{अनुप्रस्थ
काट}: $\times 10^2 = 100$ गुना। दाब --- यानी हड्डी के प्रति
\omterm{def:g6:measure:area}{क्षेत्रफल} भार: $\times \frac{1000}{100} = 10$ गुना। आदमी भर के दाब
के लिए बनी हड्डियों पर दस गुना दाब पड़ेगा: दानव का ढाँचा अपने ही
भार से चटक जाएगा। दानव ज्यामिति के हिसाब से नामुमकिन हैं; बड़े
जानवरों को अनुपात से कहीं ज़्यादा मोटी हड्डियाँ चाहिए होती हैं
(हाथी की टाँगें हिरन की टाँगों से मिलाकर देखो)।

\textbf{12.} भार: $\div 100^3 = 10^6$ गुना कम। ताक़त (माँसपेशी की
\omterm{prop:g9:solids:sections}{अनुप्रस्थ काट}): $\div 100^2 = 10^4$ गुना कम। ताक़त-और-भार:
$\times \frac{10^6}{10^4} = 100$ गुना बेहतर। अपने भार से पचास गुना
उठा लेने वाली चींटी कोई महाबली नहीं है --- वह बस छोटी है; चींटी के
नाप तक सिकोड़ दिया गया आदमी भी यही कर लेता।

\textbf{13.} $\dfrac{S}{V} = \dfrac{4\pi r^2}{\frac43\pi r^3}
= \dfrac{3}{r}$: $r = 1$ पर अनुपात $3$ है और $r = 2$ पर $1.5$ ---
यानी नाप आधी करने पर ऊष्मा बनाने वाले हर इकाई \omterm{def:g6:measure:volume}{आयतन} के हिस्से में
दुगनी सतह आ जाती है। छोटे शरीर ऊष्मा बहा देते हैं: चूहे को लगातार
खाना पड़ता है, भालू शीतनिद्रा ले सकता है, और नन्हा बच्चा --- छोटा
गोला, बड़ा $\frac SV$ --- टोपी माँगता है।

\textbf{14.} $\left(\frac54\right)^3 = \frac{125}{64} \approx
1.95$: उतने ही दाम में बड़ा संतरा क़रीब \emph{दुगना} फल देता है।
पैसा \omterm{def:g3:shapes:shapes}{त्रिज्या} पर लगाओ।

\textbf{15.} खुदाई यह कहती है: गोला \omterm{def:g6:measure:volume}{आयतन} में \omterm{def:g7:areas:prism}{बेलन} का $\frac23$ है,
\emph{और} कुल सतह में भी उसका $\frac23$ --- यानी दो बिलकुल सही,
सार्वभौम \omterm{def:g7:prop:table}{समानुपात}, जो सबसे गोल \omterm{def:g5:solids:def}{ठोस} को सबसे सादे \omterm{def:g5:solids:def}{ठोस} से बाँध देते
हैं। सिसरो को जवाब: क्योंकि प्रमेय ही वह अकेली यादगार है जिसे न
सेनाएँ मिटा पाती हैं, न सदियाँ --- आर्किमिडीज़ की मशीनें
सिरैक्यूज़ के साथ जल गईं, पर दो तिहाई आज भी ठीक दो तिहाई है।

\end{solution}
