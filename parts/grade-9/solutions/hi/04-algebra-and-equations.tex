\section*{अध्याय \ref{ch:g9:algebra} --- बीजगणित और समीकरण}

\begin{solution}{exo:g9:algebra:1}
$4(3x - 2) = 12x - 8$.

$(x+6)(x+2) = x^2 + 2x + 6x + 12 = x^2 + 8x + 12$.

$(2x-3)(x+5) = 2x^2 + 10x - 3x - 15 = 2x^2 + 7x - 15$.

$(x-1)(x-9) = x^2 - 9x - x + 9 = x^2 - 10x + 9$.
\end{solution}

\begin{solution}{exo:g9:algebra:2}
$(x+8)^2 = x^2 + 16x + 64$.

$(3x-4)^2 = 9x^2 - 24x + 16$.

$(x+10)(x-10) = x^2 - 100$.

$(5x+2)(5x-2) = 25x^2 - 4$.
\end{solution}

\begin{solution}{exo:g9:algebra:3}
$101^2 = (100+1)^2 = 10000 + 200 + 1 = 10201$.

$59^2 = (60-1)^2 = 3600 - 120 + 1 = 3481$.

$53 \times 47 = (50+3)(50-3) = 2500 - 9 = 2491$.
\end{solution}

\begin{solution}{exo:g9:algebra:4}
$9x + 12 = 3(3x + 4)$.

$x^2 - 64 = (x+8)(x-8)$.

$x^2 + 14x + 49 = (x+7)^2$.

$18x^2 - 6x = 6x(3x - 1)$.
\end{solution}

\begin{solution}{exo:g9:algebra:5}
$4x + 7 = 19$: $4x = 12$, इसलिए $x = 3$।

$6x - 5 = 2x + 11$: $4x = 16$, इसलिए $x = 4$।

$5(x-3) = 3x + 1$: $5x - 15 = 3x + 1$, इसलिए $2x = 16$ और $x = 8$।
\end{solution}

\begin{solution}{exo:g9:algebra:6}
$(x+4)(3x-12) = 0$: $x = -4$ या $x = 4$।

$x^2 - 121 = (x-11)(x+11) = 0$: $x = 11$ या $x = -11$।

$(2x-1)^2 = 0$: अकेला हल $x = \frac12$।

$x^2 = 8x$: $x(x - 8) = 0$, इसलिए $x = 0$ या $x = 8$।
\end{solution}

\begin{solution}{exo:g9:algebra:7}
$a = 3x + 2$ और $b = 5$ के साथ वर्गों का \omterm{ex:g1:subtraction:difference}{अंतर}:
\[
E = (3x + 2)^2 - 5^2 = (3x + 2 + 5)(3x + 2 - 5) = (3x + 7)(3x - 3).
\]
शून्य-गुणनफल नियम: $3x + 7 = 0$ से $x = -\frac73$, और $3x - 3 = 0$ से
$x = 1$।
\end{solution}

\begin{solution}{exo:g9:algebra:8}
$3x - 4 \geq 8$: $3x \geq 12$, इसलिए $x \geq 4$ ($4$ पर भरा बिंदु,
दाईं ओर रंगाई)।

$5 - 2x > 1$: $-2x > -4$, और $-2$ का भाग देने पर चिह्न उलट जाता है:
$x < 2$ ($2$ पर खोखला बिंदु, बाईं ओर रंगाई)।

$4(x+1) \leq x - 5$: $4x + 4 \leq x - 5$, इसलिए $3x \leq -9$, यानी
$x \leq -3$ ($-3$ पर भरा बिंदु, बाईं ओर रंगाई)।
\end{solution}

\begin{solution}{exo:g9:algebra:9}
मान लो संख्या $x$ है: $3x + 7 = 5x - 9$। तब $16 = 2x$, इसलिए $x = 8$।
जाँच: $3 \times 8 + 7 = 31$ और $5 \times 8 - 9 = 31$।
\end{solution}

\begin{solution}{exo:g9:algebra:10}
\emph{1.} $(n+1)(n-1) = n^2 - 1$ (तीसरी सर्वसमिका)। $n$ के पड़ोसी
$n - 1$ और $n + 1$ हैं, इसलिए उनका \omterm{def:g2:mult:def}{गुणनफल} $n^2 - 1$ है: जैसे
$24 \times 26 = 625 - 1 = 624$।

\emph{2.} लगातार आने वाले तीन पूर्णांक $n - 1$, $n$, $n + 1$ लिखे जा
सकते हैं। उनका \omterm{def:g1:addition:def}{योगफल}
\[
(n-1) + n + (n+1) = 3n ,
\]
$3$ का \omterm{def:g9:arith:divisor}{गुणज} है (बीच वाले का तिगुना)।
\end{solution}

\begin{solution}{pb:g9:algebra:1}
\textbf{1.} $(10a + b) - (10b + a) = 9a - 9b = 9(a - b)$: दो अंकों
की संख्या और उसके उलटे रूप का \omterm{ex:g1:subtraction:difference}{अंतर} हमेशा $9$ का \omterm{def:g9:arith:divisor}{गुणज} होता है। जाँच:
$72 - 27 = 45 = 9 \times 5$, और $a - b = 7 - 2 = 5$।

\textbf{2.} दायाँ पक्ष फैलाने पर:
$9(11a + b) + a + b + c = 99a + 9b + a + b + c
= 100a + 10b + c$। यानी संख्या, $9$ के किसी \omterm{def:g9:arith:divisor}{गुणज} और अपने अंकों के
\omterm{def:g1:addition:def}{योगफल} का जोड़ है: $9$ का भाग देने पर दोनों एक ही \omterm{def:g3:division:remainder}{शेषफल} छोड़ते हैं।
ख़ास तौर पर एक का \omterm{def:g3:division:remainder}{शेषफल} $0$ होने का मतलब दूसरे का भी \omterm{def:g3:division:remainder}{शेषफल} $0$: नौ
का नियम सिद्ध हुआ।

\textbf{3.} $9(11a + b)$, $3$ का भी \omterm{def:g9:arith:divisor}{गुणज} है, इसलिए वही सर्वसमिका
दिखाती है कि $3$ का भाग देने पर संख्या और उसके अंकों का \omterm{def:g1:addition:def}{योगफल} एक ही
\omterm{def:g3:division:remainder}{शेषफल} छोड़ते हैं: तीन का नियम।

\textbf{4.} $47 = 9k + 2$ और $86 = 9l + 5$ लिखो। तब
\[
47 \times 86 = (9k + 2)(9l + 5)
= 9\,(9kl + 5k + 2l) + 10 :
\]
\omterm{def:g2:mult:def}{गुणनफल} वही \omterm{def:g3:division:remainder}{शेषफल} छोड़ता है जो $2 \times 5 = 10$ छोड़ता है, यानी $1$।
इसलिए सही नतीजे को भी \omterm{def:g3:division:remainder}{शेषफल} $1$ छोड़ना ही चाहिए --- और $4\,042$
छोड़ता भी है। पर यह जाँच सिर्फ़ \omterm{def:g3:division:remainder}{शेषफल} देखती है: $4\,042$ और $4\,402$
(अंक आगे-पीछे) दोनों एक जैसे पार हो जाते हैं, इसलिए पार होना कुछ
सिद्ध नहीं करता --- फ़ैसला तो नाकाम होने से होता है।

\textbf{5.} फैलाने पर: $11(91a + 9b + c) - a + b - c + d
= 1001a + 99b + 11c - a + b - c + d
= 1000a + 100b + 10c + d$। यानी संख्या $11$ से \omterm{def:g6:wholes:divisible}{विभाज्य} ठीक तभी है जब
उसका बदल-बदलकर लिया गया \omterm{def:g1:addition:def}{योगफल} $-a + b - c + d$ \omterm{def:g6:wholes:divisible}{विभाज्य} हो। $2\,926$
के लिए: $-2 + 9 - 2 + 6 = 11$, जो $11$ से \omterm{def:g6:wholes:divisible}{विभाज्य} है: हाँ,
$2\,926 = 11 \times 266$।

\textbf{6.} $742 - 247 = 495$; $495 + 594 = 1\,089$। कोई भी सही
उदाहरण भी $1\,089$ पर ही उतरता है।

\textbf{7.} $(100a + 10b + c) - (100c + 10b + a) = 99(a - c)$।
$a - c = 2, 3, \dots, 9$ के लिए:
\[
198,\ 297,\ 396,\ 495,\ 594,\ 693,\ 792,\ 891 .
\]

\textbf{8.} $100(m - 1) + 90 + (10 - m) = 100m - 100 + 90 + 10
- m = 99m$। यानी $99m$ के तीनों अंक $m - 1$, फिर $9$, फिर $10 - m$
हैं ($495$ पर जाँचो: $m = 5$ से $4$, $9$, $5$)।

\textbf{9.} $99m$ के उलटे रूप के अंक $10 - m$, $9$, $m - 1$ हैं,
इसलिए वह $100(10 - m) + 90 + (m - 1)$ के बराबर है। जोड़ने पर:
\[
100\,(m - 1 + 10 - m) + 180 + (10 - m + m - 1)
= 100 \times 9 + 180 + 9 = 1\,089 ,
\]
और यह $m$ पर निर्भर ही नहीं करता। शुरुआती संख्या कोई भी हो, जादूगर
सुरक्षित है।

\textbf{10.} अगर $a = c$ हो, तो घटाने पर $0$ आता है और खेल वहीं दम
तोड़ देता है। अगर $a - c = 1$ हो, तो $99$ आता है, जिसे तीन अंकों की
लड़ी $099$ मानना पड़ता है (उलटा $990$, और $99 + 990 = 1\,089$: बच
गया!)। बारीक शर्त यह है: $1$ का \omterm{ex:g1:subtraction:difference}{अंतर} तभी चलता है जब दर्शक घटाने का
नतीजा तीन अंकों में लिखे, आगे लगे \omterm{def:g1:counting:zero}{शून्य} समेत --- और कम से कम $2$ का
फ़र्क़ माँग लेने से यह बहस ही नहीं उठती।

\textbf{11.} $2(5m + 7) + d - 14 = 10m + 14 + d - 14 =
10m + d$: दहाई (और अक्टूबर--दिसंबर के लिए सैकड़ा) महीना बताती है, और
इकाई (तथा दहाई) तारीख़।

\textbf{12.} $2(5m + 9) + d = 10m + 18 + d$: $18$ घटाओ।

\textbf{13.} सवाल 2 के अनुसार तीन अंकों की $n$ के लिए
$n - (\text{अंकों का योगफल}) = 9(11a + b)$: यानी हमेशा $9$ का \omterm{def:g9:arith:divisor}{गुणज}
(किसी भी लंबाई के लिए वैसी ही सर्वसमिकाएँ चलती हैं)। और $9$ के \omterm{def:g9:arith:divisor}{गुणज}
के अंकों का \omterm{def:g1:addition:def}{योगफल} भी $9$ का \omterm{def:g9:arith:divisor}{गुणज} होता है (फिर सवाल 2), इसलिए दर्शक
जो अंक पढ़कर सुनाता है उन्हें अगले $9$ के \omterm{def:g9:arith:divisor}{गुणज} तक पूरा करना होता है:
ग़ायब अंक वही पूरक है। दुविधा: अगर पढ़े हुए अंकों का \omterm{def:g1:addition:def}{योगफल} पहले से ही
$9$ का \omterm{def:g9:arith:divisor}{गुणज} हो, तो काटा हुआ अंक $0$ भी हो सकता है और $9$ भी --- $0$
को बाहर रख देने से शक मिट जाता है।

\textbf{14.} $(10t + 5)^2 = 100t^2 + 100t + 25 =
100\,t(t + 1) + 25$: पहले $t(t+1)$ लिखो, फिर उसके पीछे $25$ चिपका
दो। $45^2$: $4 \times 5 = 20$, इसलिए $2\,025$। $85^2$:
$8 \times 9 = 72$, इसलिए $7\,225$। $105^2$: $10 \times 11 = 110$,
इसलिए $11\,025$।

\textbf{15.} लगातार पाँच: $n + (n+1) + (n+2) + (n+3) +
(n+4) = 5n + 10 = 5(n + 2)$: हमेशा $5$ का \omterm{def:g9:arith:divisor}{गुणज}। लगातार चार:
$n + (n+1) + (n+2) + (n+3) = 4n + 6 =
4(n + 1) + 2$: $4$ का भाग देने पर \omterm{def:g3:division:remainder}{शेषफल} $2$, यानी कभी \omterm{def:g9:arith:divisor}{गुणज} नहीं।
बीजगणित ``ऐसा तो लगता है'' को \emph{हमेशा} में बदल देता है, और
``मुझे कोई उदाहरण नहीं मिला'' को \emph{कभी नहीं} में --- ये दो शब्द
कितनी भी जाँच से नहीं मिलते
(\cref{met:g7:literal:test})।

\end{solution}
