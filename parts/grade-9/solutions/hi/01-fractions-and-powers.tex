\section*{अध्याय \ref{ch:g9:fractions} --- भिन्नें और घातें}

\begin{solution}{exo:g9:fractions:1}
$\dfrac13 + \dfrac15 = \dfrac{5}{15} + \dfrac{3}{15} = \dfrac{8}{15}$.

$\dfrac72 - \dfrac53 = \dfrac{21}{6} - \dfrac{10}{6} = \dfrac{11}{6}$.

$\dfrac{5}{12} + \dfrac38 = \dfrac{10}{24} + \dfrac{9}{24} =
\dfrac{19}{24}$.

$3 - \dfrac45 = \dfrac{15}{5} - \dfrac45 = \dfrac{11}{5}$.
\end{solution}

\begin{solution}{exo:g9:fractions:2}
$\dfrac59 \times \dfrac{27}{35} = \dfrac{5 \times 27}{9 \times 35}
= \dfrac{27}{9} \times \dfrac{5}{35} = 3 \times \dfrac17 = \dfrac37$
($9$ से और $5$ से सरल किया)।

$\dfrac{8}{15} \div \dfrac45 = \dfrac{8}{15} \times \dfrac54
= \dfrac{8 \times 5}{15 \times 4} = \dfrac{40}{60} = \dfrac23$.

$\dfrac23 \times \dfrac34 \times \dfrac45 = \dfrac{2 \times 3 \times 4}
{3 \times 4 \times 5} = \dfrac25$ ($3$ और $4$ कट जाते हैं)।
\end{solution}

\begin{solution}{exo:g9:fractions:3}
$2^5 = 32$; \quad $(-2)^4 = 16$; \quad $-2^4 = -16$; \quad
$4^{-2} = \frac{1}{16}$; \quad
$\left(\frac23\right)^3 = \frac{8}{27}$; \quad $7^0 = 1$.
\end{solution}

\begin{solution}{exo:g9:fractions:4}
$5^3 \times 5^6 = 5^9$; \quad $\dfrac{7^9}{7^5} = 7^4$; \quad
$\left(2^3\right)^4 = 2^{12}$; \quad
$\dfrac{3^2 \times 3^7}{3^5} = 3^{9-5} = 3^4$; \quad
$2^5 \times 4 = 2^5 \times 2^2 = 2^7$.
\end{solution}

\begin{solution}{exo:g9:fractions:5}
$45\,000 = 4.5 \times 10^4$; \quad
$0.0072 = 7.2 \times 10^{-3}$; \quad
$302.5 = 3.025 \times 10^2$; \quad
$0.000\,000\,9 = 9 \times 10^{-7}$; \quad
एक करोड़ बीस लाख $= 1.2 \times 10^7$।
\end{solution}

\begin{solution}{exo:g9:fractions:6}
$(2 \times 10^7) \times (4.5 \times 10^{-3}) = 9 \times 10^{4}$.

$\dfrac{9 \times 10^5}{3 \times 10^{-2}} = 3 \times 10^{5 - (-2)}
= 3 \times 10^7$.

$(5 \times 10^3)^2 = 25 \times 10^6 = 2.5 \times 10^7$.
\end{solution}

\begin{solution}{exo:g9:fractions:7}
डाला गया पानी टंकी का
$\dfrac34 - \dfrac35 = \dfrac{15}{20} - \dfrac{12}{20} = \dfrac{3}{20}$
भाग है। यानी \omterm{def:g2:measure:units}{धारिता} का $\frac{3}{20}$ भाग $30$ लीटर है: तब
$\frac{1}{20}$ भाग $10$ लीटर हुआ, और \omterm{def:g2:measure:units}{धारिता} $200$ लीटर।
\end{solution}

\begin{solution}{exo:g9:fractions:8}
ऐलिस के बाद: केक का $1 - \frac14 = \frac34$ भाग बचता है। बेन
$\frac13 \times \frac34 = \frac14$ खाता है, और
$\frac34 - \frac14 = \frac12$ बचता है। कार्ला उसका \omterm{def:g3:division:half}{आधा}, यानी
$\frac14$, खाती है, और केक का $\frac14$ भाग बच जाता है।
\end{solution}

\begin{solution}{exo:g9:fractions:9}
समय $= \dfrac{\text{दूरी}}{\text{चाल}} =
\dfrac{1.5 \times 10^{11}}{3 \times 10^8} = 0.5 \times 10^3 = 500$
सेकंड, यानी $8$ मिनट $20$ सेकंड।
\end{solution}

\begin{solution}{exo:g9:fractions:10}
दोनों संख्याओं को \omterm{def:g8:powers:def}{घातांक} $10$ वाली घातों के रूप में लिखो:
\[
2^{30} = \left(2^3\right)^{10} = 8^{10},
\qquad
3^{20} = \left(3^2\right)^{10} = 9^{10}.
\]
और $8 < 9$ है, इसलिए दोनों की दस-दस नक़लें गुणा करने पर
$8^{10} < 9^{10}$, यानी $2^{30} < 3^{20}$।
\end{solution}

\begin{solution}{pb:g9:fractions:1}
\textbf{1.} $\frac14 = 0.25$ (रुक जाती है);
$\frac13 = 0.333\ldots$ (गुच्छा $3$); $\frac56 = 0.8333\ldots$
($8$ के बाद गुच्छा $3$); $\frac{3}{11} = 0.272727\ldots$ (गुच्छा
$27$)।

\textbf{2.} $b$ से भाग के हर क़दम पर \omterm{def:g3:division:remainder}{शेषफल} $0, 1, \dots, b - 1$
इन $b$ मानों में से कोई एक होता है
(\cref{thm:g6:wholes:division})। अगर कभी \omterm{def:g3:division:remainder}{शेषफल} $0$ आ गया तो भाग
रुक जाता है। नहीं तो ज़्यादा से ज़्यादा $b$ क़दमों में कोई \omterm{def:g3:division:remainder}{शेषफल}
दूसरी बार आना ही पड़ेगा --- क़दम संभव शेषफलों से ज़्यादा हैं --- और
उसी पल से भाग ठीक वही अंक दोहराने लगता है जो उसने पहली बार के बाद
बनाए थे: दशमलव लेखन हमेशा के लिए चक्र में पड़ जाता है। रुकना या
दोहराना: तीसरा बरताव है ही नहीं।

\textbf{3.} $\frac17 = 0.142857\,142857\ldots$: गुच्छा $142857$
छह अंकों का है। सवाल 2 ने ज़्यादा से ज़्यादा $7$ क़दमों में दोहराव
का वादा किया था --- और यहाँ \omterm{def:g3:division:remainder}{शेषफल} $1, 3, 2, 6, 4, 5$ सब के सब आ
जाते हैं, तब कहीं $1$ लौटता है: यानी $7$ से भाग का सबसे लंबा संभव
जाप।

\textbf{4.} $\frac ab$ ठीक तब रुकती है जब किसी
$\frac{a \times k}{b \times k}$ का हर $10$, $100$,
$1000, \dots$ हो --- यानी जब हर को गुणा करके दस की \omterm{def:g9:fractions:power}{घात} बनाया जा
सके, जैसे $4 \times 25 = 100$ या $8 \times 125 = 1\,000$। $3$ और
$6$ के लिए यह नामुमकिन है: दस की किसी भी \omterm{def:g9:fractions:power}{घात} के अंकों का \omterm{def:g1:addition:def}{योगफल} $1$
होता है, जबकि $3$ के हर \omterm{def:g5:division:multiple}{गुणज} के अंकों का \omterm{def:g1:addition:def}{योगफल} $3$ का \omterm{def:g5:division:multiple}{गुणज} होता
है --- इसलिए $3$ का (और तब $6$ का भी) कोई \omterm{def:g5:division:multiple}{गुणज} कभी $10$, $100$,
$1\,000, \dots$ नहीं बनता। और $11$ के लिए: $10$, $100$, $1\,000$,
$10\,000$ में $11$ का भाग हमेशा कोई \omterm{def:g3:division:remainder}{शेषफल} छोड़ता है ($10$, $1$,
$10$, $1, \dots$), कभी $0$ नहीं।
दस की \omterm{def:g9:fractions:power}{घात} तक पहुँचने वाले हर रुक जाते हैं; बाक़ी सब दोहराते हैं।

\textbf{5.} $\frac{9}{40}$: $40 \times 25 = 1\,000$, रुकती है:
$\frac{225}{1000} = 0.225$। $\frac{33}{50}$:
$50 \times 2 = 100$, रुकती है: $\frac{66}{100} = 0.66$।
$\frac{5}{12}$: $12$, $3$ का \omterm{def:g5:division:multiple}{गुणज} है, इसलिए दोहराती है
($0.41666\ldots$)।

\textbf{6.} $10x = 7.777\ldots$, इसलिए $10x - x = 7$ (पूँछें ठीक-ठीक
कट जाती हैं), $9x = 7$ और $x = \frac79$। भाग से जाँच:
$7 \div 9 = 0.777\ldots$

\textbf{7.} गुच्छा दो अंकों का है, इसलिए $100$ से सरकाओ:
$100x - x = 72.7272\ldots - 0.7272\ldots = 72$, यानी
$99x = 72$ और $x = \frac{72}{99} = \frac{8}{11}$।

\textbf{8.} $1000x = 583.333\ldots$ और $100x = 58.333\ldots$;
घटाने पर $900x = 525$, इसलिए
$x = \frac{525}{900} = \frac{7}{12}$ ($75$ से भाग देकर)। और सचमुच
$\frac{7}{12} = 0.58333\ldots$

\textbf{9.} $10x - x = 9.999\ldots - 0.999\ldots = 9$, इसलिए
$9x = 9$ और $x = 1$। यह $1$ से ``ज़रा-सा नीचे'' नहीं है:
$0.999\ldots$ वही संख्या $1$ \emph{है}, बस दूसरे भेस में। $1$ से
पक्के तौर पर नीचे कोई भी संख्या एक झिरी छोड़ देती है, और
\cref{pb:g6:decimals:1} ने दिखाया था कि किसी भी झिरी में और
संख्याएँ पड़ी होती हैं --- जबकि $0.999\ldots$ और $1$ के बीच कुछ
समाता ही नहीं। कभी न ख़त्म होने वाले नौ
\cref{pb:g6:fractions:1} वाली चॉकलेट का बचा हुआ हिस्सा हैं, जो हर
धन मात्रा से भी नीचे सिकुड़ चुका है: कुछ बचा ही नहीं।

\textbf{10.} $2.4545\ldots = 2 + \frac{45}{99} = 2 +
\frac{5}{11} = \frac{27}{11}$। और
$0.123123\ldots = \frac{123}{999} = \frac{41}{333}$ ($3$ से भाग
देकर)।

\textbf{11.} शब्दकोश से: $0.037037\ldots = \frac{37}{999}$। और
$999 = 27 \times 37$ है, इसलिए यह सरल होकर $\frac{1}{27}$ बन जाती
है: यानी \omterm{def:g9:fractions:fraction}{भिन्न} $\frac{1}{27}$ ही ``$037$'' का जाप करती है। भाग से
जाँच: $1 \div 27 = 0.037037\ldots$

\textbf{12.} $0.0272727\ldots = \frac{1}{10} \times
0.272727\ldots = \frac{1}{10} \times \frac{27}{99}
= \frac{27}{990} = \frac{3}{110}$ ($9$ से भाग देकर)।

\textbf{13.} शून्यों के गुच्छे बिना रुके बढ़ते जाते हैं, इसलिए कोई
तयशुदा गुच्छा हमेशा नहीं दोहरा सकता: यह आवर्ती दशमलव \emph{नहीं}
है (और रुकती भी नहीं)। भाग I से हर \omterm{def:g9:fractions:fraction}{भिन्न} या तो रुकती है या दोहराती
है --- इसलिए यह संख्या कोई \omterm{def:g9:fractions:fraction}{भिन्न} है ही नहीं: यह अपरिमेय है, और
जान-बूझकर ऐसी बनाई गई है। अपरिमेयता का महानायक, $\sqrt2$,
\cref{ch:g9:sqrt} में बेनक़ाब होता है।

\textbf{14.} $2^{30} = \left(2^{10}\right)^3 \approx
(1.024)^3 \times 10^9 \approx 1.07 \times 10^9$: यानी $10^9$ से
ज़रा-सी बड़ी संख्या, इसलिए उसमें $10$ अंक हैं। (ठीक-ठीक:
$2^{30} = 1\,073\,741\,824$।)

\textbf{15.} $0.20262026\ldots = \frac{2026}{9999}$, और यह पहले से
ही सबसे सरल रूप में है ($9999 = 3 \times 3 \times 11 \times 101$
का $2026 = 2 \times 1013$ के साथ कोई गुणनखंड साझा नहीं है)। पूरा
शब्दकोश: \emph{भिन्नें ठीक वही दशमलव हैं जो या तो रुक जाते हैं या
दोहराते हैं} --- दस के दोस्त बने हर रुक जाते हैं, बाक़ी सब कोई न कोई
गुच्छा जपते हैं; और उलटे तरफ़ से हर जाप, चाहे वह तुम्हारा पास होने
का साल ही क्यों न हो, भेस बदली हुई \omterm{def:g9:fractions:fraction}{भिन्न} है।

\end{solution}
