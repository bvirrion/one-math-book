\section*{Chapitre \ref{ch:g9:trig} --- Trigonométrie dans le triangle rectangle}

\begin{solution}{exo:g9:trig:1}
Hypoténuse~: $\sqrt{9^2 + 12^2} = \sqrt{81 + 144} = \sqrt{225} = 15$.

Autre côté~: $\sqrt{17^2 - 8^2} = \sqrt{289 - 64} = \sqrt{225} = 15$.
\end{solution}

\begin{solution}{exo:g9:trig:2}
$7^2 + 24^2 = 49 + 576 = 625 = 25^2$~: oui, rectangle (réciproque de
Pythagore), avec l'\omterm{def:g3:shapes:rightangle}{angle droit} opposé au côté $25$.

$5^2 + 6^2 = 61 \neq 64 = 8^2$~: non rectangle.
\end{solution}

\begin{solution}{exo:g9:trig:3}
L'hypoténuse est $[EF]$ (opposée à l'\omterm{def:g3:shapes:rightangle}{angle droit} en $D$). Le côté
opposé à $\theta = \widehat E$ est $[DF]$~; le côté adjacent est
$[DE]$. D'où
\[
\cos\theta = \frac{DE}{EF}, \qquad
\sin\theta = \frac{DF}{EF}, \qquad
\tan\theta = \frac{DF}{DE}.
\]
\end{solution}

\begin{solution}{exo:g9:trig:4}
Côté opposé~: $10 \sin 28^\circ \approx 10 \times 0.469 = 4.7$.
Côté adjacent~: $10 \cos 28^\circ \approx 10 \times 0.883 = 8.8$.
(Vérification~: $4.7^2 + 8.8^2 \approx 22 + 77 \approx 100$.)
\end{solution}

\begin{solution}{exo:g9:trig:5}
$\tan\theta = \dfrac{4.5}{6} = 0.75$, donc
$\theta = \tan^{-1}(0.75) \approx 37^\circ$.
\end{solution}

\begin{solution}{exo:g9:trig:6}
Dans le \omterm{def:g6:shapes:triangles}{triangle rectangle} formé par l'observateur, le point au sol sous
le drone, et le drone~: l'altitude $120$ m est le côté opposé à l'angle
d'élévation, la distance horizontale $d$ est adjacente. Donc
\[
\tan 32^\circ = \frac{120}{d},
\qquad
d = \frac{120}{\tan 32^\circ} \approx \frac{120}{0.625} = 192 \text{ m}.
\]
\end{solution}

\begin{solution}{exo:g9:trig:7}
La montée ($1.2$ m) est opposée à l'angle~; la longueur $L$ de la rampe
est l'hypoténuse~:
\[
\sin 5^\circ = \frac{1.2}{L},
\qquad
L = \frac{1.2}{\sin 5^\circ} \approx \frac{1.2}{0.0872} \approx 13.8
\text{ m}.
\]
La rampe doit mesurer au moins $13.8$ m.
\end{solution}

\begin{solution}{exo:g9:trig:8}
De $\cos^2\theta + \sin^2\theta = 1$~:
$\sin^2\theta = 1 - 0.36 = 0.64$, et $\sin\theta > 0$ pour un angle
aigu, donc $\sin\theta = 0.8$. Puis
$\tan\theta = \dfrac{\sin\theta}{\cos\theta} = \dfrac{0.8}{0.6} =
\dfrac43$.
\end{solution}

\begin{solution}{exo:g9:trig:9}
\emph{\omterm{def:g6:shapes:triangles}{Triangle rectangle} \omterm{def:g6:shapes:triangles}{isocèle}} de côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit} $1$~:
hypoténuse $\sqrt{1 + 1} = \sqrt2$~; chaque angle aigu vaut $45^\circ$,
et
\[
\cos 45^\circ = \frac{1}{\sqrt2} = \frac{\sqrt2}{2} = \sin 45^\circ,
\qquad
\tan 45^\circ = \frac11 = 1 .
\]

\emph{\omterm{def:g6:shapes:triangles}{Triangle équilatéral}} de côté $1$~: sa hauteur le partage en deux
\omterm{def:g6:shapes:triangles}{triangles rectangles} d'hypoténuse $1$, de base $\frac12$, et de hauteur
$h = \sqrt{1 - \frac14} = \frac{\sqrt3}{2}$ (Pythagore). Les angles
sont $60^\circ$ (à la base) et $30^\circ$ (au \omterm{def:g5:solids:def}{sommet}). En lisant les
rapports~:
\[
\cos 60^\circ = \frac{1/2}{1} = \frac12, \quad
\sin 60^\circ = \frac{\sqrt3/2}{1} = \frac{\sqrt3}{2}, \quad
\cos 30^\circ = \frac{\sqrt3}{2}, \quad
\sin 30^\circ = \frac12 .
\]
\end{solution}

\begin{solution}{pb:g9:trig:1}
\textbf{1.} La hauteur au-dessus du niveau de l'œil est le côté
opposé à $35^\circ$, le côté adjacent valant $60$ m~:
$h = 60 \tan 35^\circ \approx 60 \times 0.700 = 42$ m.

\textbf{2.} $\tan\theta = 0.12$ donne
$\theta = \tan^{-1}(0.12) \approx 6.8^\circ$ --- raide pour une
route, et pourtant un angle modeste. Une route à $45^\circ$ a
$\tan 45^\circ = 1$~: une pente de $100\,\%$. («~Cent pour cent~»
est bien loin de la verticale --- un malentendu classique au pied
des remontées mécaniques.)

\textbf{3.} Dans un \omterm{def:g6:shapes:triangles}{triangle rectangle} d'angles aigus $\theta$ et
$90^\circ - \theta$, le côté opposé à l'un est adjacent à l'autre,
et l'hypoténuse est commune. Donc
$\sin(90^\circ - \theta) = \frac{\text{son côté opposé}}{h} =
\frac{\text{côté adjacent à }\theta}{h} = \cos\theta$, et
symétriquement $\cos(90^\circ - \theta) = \sin\theta$~: le
\emph{co}\omterm{def:g9:trig:ratios}{sinus} est le \omterm{def:g9:trig:ratios}{sinus} du \emph{co}mplémentaire.

\textbf{4.} Hauteur~: $8 \sin 60^\circ = 8 \times
\frac{\sqrt3}{2} = 4\sqrt3 \approx 6.93$ m. Pied~:
$8 \cos 60^\circ = 4$ m exactement.

\textbf{5.} $0.574^2 + 0.819^2 = 0.329 + 0.671 = 1.000$ (aux
\omterm{def:g6:decimals:rounding}{arrondis} près)~; et $\left(\frac12\right)^2 +
\left(\frac{\sqrt3}{2}\right)^2 = \frac14 + \frac34 = 1$. Tout
couple de valeurs qui échoue à l'identité trahit une lecture
erronée ou une calculatrice mal réglée --- un détecteur d'erreurs
gratuit.

\textbf{6.} $\tan 40^\circ = \dfrac hx$ et
$\tan 30^\circ = \dfrac{h}{x + 200}$.

\textbf{7.} $x = \dfrac{h}{\tan 40^\circ}$ et
$x + 200 = \dfrac{h}{\tan 30^\circ}$~; en soustrayant,
$200 = h\left(\frac{1}{\tan 30^\circ} -
\frac{1}{\tan 40^\circ}\right)$, d'où la formule. Numériquement~:
$\frac{1}{0.577} \approx 1.732$, $\frac{1}{0.839} \approx 1.192$,
\omterm{ex:g1:subtraction:difference}{différence} $0.540$~: $h \approx \frac{200}{0.540} \approx 370$ m.

\textbf{8.} $x = \frac{h}{\tan 40^\circ} \approx
\frac{370}{0.839} \approx 441$ m. Contrôle depuis $A$~:
$\frac{h}{x + 200} \approx \frac{370}{641} \approx 0.577 =
\tan 30^\circ$~: cohérent.

\textbf{9.} La méthode à une station réclame la distance
horizontale jusqu'au point situé \emph{à la verticale du \omterm{def:g5:solids:def}{sommet}}
--- impossible à mesurer par-dessus une gorge ou à travers une
montagne. La méthode des deux stations la remplace par une distance
que l'on peut arpenter soi-même~: les $200$ m qui séparent nos deux
points de visée.

\textbf{10.} $h = \dfrac{200}{\sqrt3 - \frac{1}{\sqrt3}}
= \dfrac{200}{\frac{2}{\sqrt3}} = 100\sqrt3 \approx 173$ m --- le
deuxième étage de la tour Eiffel, mesuré avec deux visées et une
promenade.

\textbf{11.} Ligne de visée $d$, \omterm{def:g3:shapes:shapes}{rayons} $R$ (jusqu'au point de
contact, \omterm{def:g6:lines:perp}{perpendiculaire} à la ligne de visée) et $R + h$ (jusqu'à
l'œil)~: Pythagore donne $d^2 + R^2 = (R + h)^2 = R^2 + 2Rh + h^2$,
donc $d^2 = 2Rh + h^2 \approx 2Rh$ ($h$ est minuscule devant $R$).
Pour un œil à $1.80$ m~:
$d \approx \sqrt{2 \times 6.37 \times 10^6 \times 1.8}
\approx 4\,800$ m --- l'horizon marin est à peine à cinq
kilomètres.

\textbf{12.} $d \approx \sqrt{2 \times 6.37 \times 10^6 \times
324} \approx 64$ km. Règle des marins~: $3.6\sqrt{1.8} \approx
4.8$ km et $3.6\sqrt{324} = 64.8$ km --- les deux concordent.

\textbf{13.} $\tan\theta = \frac{2}{60}$ donne
$\theta \approx 1.9^\circ$. Les $0.5^\circ$ de la Lune font environ
un \emph{\omterm{def:g3:division:half}{quart}} de pouce~: la Lune paraît énorme et se cache
derrière un ongle --- l'œil est un piètre rapporteur.

\textbf{14.} Escalier~: $\tan^{-1}\frac{17}{29} \approx 30^\circ$
--- un escalier confortable monte presque exactement sous les
$30^\circ$ des valeurs particulières. À $60^\circ$ sur un giron de
$29$ cm, la contremarche vaudrait
$29\tan 60^\circ = 29\sqrt3 \approx 50$ cm~: une échelle, pas un
escalier.

\textbf{15.} $\tan 10^\circ \approx 0.18$~;
$\tan 45^\circ = 1$~; $\tan 80^\circ \approx 5.7$~;
$\tan 89^\circ \approx 57.3$. Lorsque $\theta$ tend vers
$90^\circ$, le côté adjacent se réduit à rien tandis que le côté
opposé tient bon~: le \omterm{def:g3:division:remainder}{quotient} croît au-delà de toute borne. À
$90^\circ$ exactement, il n'y a plus de triangle ni de valeur --- le
graphe de la \omterm{def:g9:trig:ratios}{tangente} escalade une asymptote verticale, la première
d'une longue série dans le \omterm{def:g6:measure:volume}{volume} suivant.
\end{solution}
