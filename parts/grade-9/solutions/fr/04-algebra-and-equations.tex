\section*{Chapitre \ref{ch:g9:algebra} --- Algèbre et équations}

\begin{solution}{exo:g9:algebra:1}
$4(3x - 2) = 12x - 8$.

$(x+6)(x+2) = x^2 + 2x + 6x + 12 = x^2 + 8x + 12$.

$(2x-3)(x+5) = 2x^2 + 10x - 3x - 15 = 2x^2 + 7x - 15$.

$(x-1)(x-9) = x^2 - 9x - x + 9 = x^2 - 10x + 9$.
\end{solution}

\begin{solution}{exo:g9:algebra:2}
$(x+8)^2 = x^2 + 16x + 64$.

$(3x-4)^2 = 9x^2 - 24x + 16$.

$(x+10)(x-10) = x^2 - 100$.

$(5x+2)(5x-2) = 25x^2 - 4$.
\end{solution}

\begin{solution}{exo:g9:algebra:3}
$101^2 = (100+1)^2 = 10000 + 200 + 1 = 10201$.

$59^2 = (60-1)^2 = 3600 - 120 + 1 = 3481$.

$53 \times 47 = (50+3)(50-3) = 2500 - 9 = 2491$.
\end{solution}

\begin{solution}{exo:g9:algebra:4}
$9x + 12 = 3(3x + 4)$.

$x^2 - 64 = (x+8)(x-8)$.

$x^2 + 14x + 49 = (x+7)^2$.

$18x^2 - 6x = 6x(3x - 1)$.
\end{solution}

\begin{solution}{exo:g9:algebra:5}
$4x + 7 = 19$~: $4x = 12$, donc $x = 3$.

$6x - 5 = 2x + 11$~: $4x = 16$, donc $x = 4$.

$5(x-3) = 3x + 1$~: $5x - 15 = 3x + 1$, donc $2x = 16$ et $x = 8$.
\end{solution}

\begin{solution}{exo:g9:algebra:6}
$(x+4)(3x-12) = 0$~: $x = -4$ ou $x = 4$.

$x^2 - 121 = (x-11)(x+11) = 0$~: $x = 11$ ou $x = -11$.

$(2x-1)^2 = 0$~: la solution unique $x = \frac12$.

$x^2 = 8x$~: $x(x - 8) = 0$, donc $x = 0$ ou $x = 8$.
\end{solution}

\begin{solution}{exo:g9:algebra:7}
\omterm{ex:g1:subtraction:difference}{Différence} de carrés avec $a = 3x + 2$ et $b = 5$~:
\[
E = (3x + 2)^2 - 5^2 = (3x + 2 + 5)(3x + 2 - 5) = (3x + 7)(3x - 3).
\]
Règle du \omterm{def:g2:mult:def}{produit} nul~: $3x + 7 = 0$ donne $x = -\frac73$, et
$3x - 3 = 0$ donne $x = 1$.
\end{solution}

\begin{solution}{exo:g9:algebra:8}
$3x - 4 \geq 8$~: $3x \geq 12$, donc $x \geq 4$ (point plein en $4$,
hachurer à droite).

$5 - 2x > 1$~: $-2x > -4$, diviser par $-2$ renverse le signe~: $x < 2$
(point creux en $2$, hachurer à gauche).

$4(x+1) \leq x - 5$~: $4x + 4 \leq x - 5$, donc $3x \leq -9$, donc
$x \leq -3$ (point plein en $-3$, hachurer à gauche).
\end{solution}

\begin{solution}{exo:g9:algebra:9}
Soit $x$ le nombre~: $3x + 7 = 5x - 9$. Alors $16 = 2x$, donc $x = 8$.
Vérification~: $3 \times 8 + 7 = 31$ et $5 \times 8 - 9 = 31$.
\end{solution}

\begin{solution}{exo:g9:algebra:10}
\emph{1.} $(n+1)(n-1) = n^2 - 1$ (troisième identité). Les voisins de
$n$ sont $n - 1$ et $n + 1$, donc leur \omterm{def:g2:mult:def}{produit} est $n^2 - 1$~: par ex.\
$24 \times 26 = 625 - 1 = 624$.

\emph{2.} Trois entiers consécutifs s'écrivent $n - 1$, $n$, $n + 1$.
Leur \omterm{def:g1:addition:def}{somme} est
\[
(n-1) + n + (n+1) = 3n ,
\]
un \omterm{def:g9:arith:divisor}{multiple} de $3$ (trois fois le milieu).
\end{solution}

\begin{solution}{pb:g9:algebra:1}
\textbf{1.} $(10a + b) - (10b + a) = 9a - 9b = 9(a - b)$~: la
\omterm{ex:g1:subtraction:difference}{différence} entre un nombre à deux chiffres et son retourné est
toujours un \omterm{def:g9:arith:divisor}{multiple} de $9$. Vérification~:
$72 - 27 = 45 = 9 \times 5$, et $a - b = 7 - 2 = 5$.

\textbf{2.} En développant le membre de droite~:
$9(11a + b) + a + b + c = 99a + 9b + a + b + c
= 100a + 10b + c$. Le nombre est donc égal à un \omterm{def:g9:arith:divisor}{multiple} de $9$
augmenté de la \omterm{def:g1:addition:def}{somme} de ses chiffres~: dans la division par $9$, les
deux laissent le même \omterm{def:g3:division:remainder}{reste}. En particulier, un \omterm{def:g3:division:remainder}{reste} $0$ pour l'un
signifie un \omterm{def:g3:division:remainder}{reste} $0$ pour l'autre~: le critère de divisibilité par
$9$ est démontré.

\textbf{3.} $9(11a + b)$ est aussi un \omterm{def:g9:arith:divisor}{multiple} de $3$~: la même
identité montre donc que le nombre et la \omterm{def:g1:addition:def}{somme} de ses chiffres
laissent le même reste dans la division par $3$, d'où le critère de
divisibilité par $3$.

\textbf{4.} Écrivons $47 = 9k + 2$ et $86 = 9l + 5$. Alors
\[
47 \times 86 = (9k + 2)(9l + 5)
= 9\,(9kl + 5k + 2l) + 10 :
\]
le \omterm{def:g2:mult:def}{produit} laisse le même \omterm{def:g3:division:remainder}{reste} que $2 \times 5 = 10$, à savoir
$1$. Un résultat correct doit donc lui aussi laisser le \omterm{def:g3:division:remainder}{reste} $1$
--- ce que fait $4\,042$. Mais le test ne voit que les \omterm{def:g3:division:remainder}{restes}~:
$4\,042$ et $4\,402$ (chiffres intervertis) le passent de façon
identique~; réussir ne prouve donc rien --- c'est l'échec qui
condamne.

\textbf{5.} En développant~: $11(91a + 9b + c) - a + b - c + d
= 1001a + 99b + 11c - a + b - c + d
= 1000a + 100b + 10c + d$. Le nombre est donc \omterm{def:g6:wholes:divisible}{divisible} par $11$
exactement lorsque sa \omterm{def:g1:addition:def}{somme} alternée $-a + b - c + d$ l'est. Pour
$2\,926$~: $-2 + 9 - 2 + 6 = 11$, \omterm{def:g6:wholes:divisible}{divisible} par $11$~: oui,
$2\,926 = 11 \times 266$.

\textbf{6.} $742 - 247 = 495$~; $495 + 594 = 1\,089$. Tout autre
exemple valable aboutit aussi à $1\,089$.

\textbf{7.} $(100a + 10b + c) - (100c + 10b + a) = 99(a - c)$. Pour
$a - c = 2, 3, \dots, 9$~:
\[
198,\ 297,\ 396,\ 495,\ 594,\ 693,\ 792,\ 891 .
\]

\textbf{8.} $100(m - 1) + 90 + (10 - m) = 100m - 100 + 90 + 10
- m = 99m$. Les trois chiffres de $99m$ sont donc $m - 1$, puis $9$,
puis $10 - m$ (vérification sur $495$~: $m = 5$ donne $4$, $9$,
$5$).

\textbf{9.} Le retourné de $99m$ a pour chiffres $10 - m$, $9$,
$m - 1$~: il vaut donc $100(10 - m) + 90 + (m - 1)$. En
additionnant~:
\[
100\,(m - 1 + 10 - m) + 180 + (10 - m + m - 1)
= 100 \times 9 + 180 + 9 = 1\,089 ,
\]
indépendamment de $m$. Quel que soit le nombre de départ, le
magicien ne risque rien.

\textbf{10.} Si $a = c$, la soustraction donne $0$ et le tour meurt
aussitôt. Si $a - c = 1$, elle donne $99$, qu'il faut traiter comme
la chaîne de trois chiffres $099$ (retourné $990$,
$99 + 990 = 1\,089$~: sauvé~!). Les petites lignes~: un écart de $1$
ne fonctionne que si le spectateur écrit le résultat de la
soustraction sur trois chiffres, \omterm{def:g1:counting:zero}{zéro} initial compris --- exiger un
écart d'au moins $2$ évite la discussion.

\textbf{11.} $2(5m + 7) + d - 14 = 10m + 14 + d - 14 = 10m + d$~:
les dizaines (et les centaines, d'octobre à décembre) donnent le
mois, les unités (et les dizaines) le jour.

\textbf{12.} $2(5m + 9) + d = 10m + 18 + d$~: il faut retrancher
$18$.

\textbf{13.} D'après la question 2,
$n - (\text{somme des chiffres de } n) = 9(11a + b)$ pour un $n$ à
trois chiffres~: c'est toujours un \omterm{def:g9:arith:divisor}{multiple} de $9$ (des identités
analogues valent pour n'importe quelle longueur). Or un \omterm{def:g9:arith:divisor}{multiple} de
$9$ a une \omterm{def:g1:addition:def}{somme} de chiffres \omterm{def:g9:arith:divisor}{multiple} de $9$ (question 2 encore)~:
les chiffres énoncés par le spectateur doivent donc être complétés
jusqu'au \omterm{def:g9:arith:divisor}{multiple} de $9$ suivant, et le chiffre manquant est ce
complément. Ambiguïté~: si les chiffres lus totalisent déjà un
\omterm{def:g9:arith:divisor}{multiple} de $9$, le chiffre barré pourrait être $0$ ou $9$ ---
exclure le $0$ lève le doute.

\textbf{14.} $(10t + 5)^2 = 100t^2 + 100t + 25 =
100\,t(t + 1) + 25$~: écrire $t(t+1)$, puis coller $25$ derrière.
$45^2$~: $4 \times 5 = 20$, donc $2\,025$. $85^2$~:
$8 \times 9 = 72$, donc $7\,225$. $105^2$~: $10 \times 11 = 110$,
donc $11\,025$.

\textbf{15.} Cinq entiers consécutifs~: $n + (n+1) + (n+2) + (n+3) +
(n+4) = 5n + 10 = 5(n + 2)$~: un \omterm{def:g9:arith:divisor}{multiple} de $5$, toujours. Quatre
entiers consécutifs~: $n + (n+1) + (n+2) + (n+3) = 4n + 6 =
4(n + 1) + 2$~: \omterm{def:g3:division:remainder}{reste} $2$ dans la division par $4$, donc jamais un
\omterm{def:g9:arith:divisor}{multiple}. L'algèbre transforme «~ça a l'air de marcher~» en
\emph{toujours}, et «~je n'ai pas trouvé d'exemple~» en
\emph{jamais} --- deux mots qu'aucune quantité de tests ne peut
atteindre (\cref{met:g7:literal:test}).
\end{solution}
