\begin{solution}{exo:g9:statproba:1}
\omterm{def:g9:statproba:indicators}{Moyenne}~: $\dfrac{14 + 9 + 17 + 9 + 12 + 11 + 12}{7} = \dfrac{84}{7} = 12$.

\omterm{def:g9:statproba:indicators}{Médiane}~: triée, $9, 9, 11, 12, 12, 14, 17$~; la valeur du milieu
($4$\textsuperscript{e}) est $12$.

\omterm{def:g9:statproba:indicators}{Étendue}~: $17 - 9 = 8$.
\end{solution}

\omterm{def:g9:statproba:indicators}{Médiane}~: triée, $18, 19, 19, 21, 22, 24, 25$~; la $4$\textsuperscript{e} valeur est

\begin{solution}{exo:g9:statproba:3}
\omterm{def:g9:statproba:indicators}{Moyenne}~:
\[
\frac{2 \times 0 + 4 \times 1 + 3 \times 2 + 1 \times 3}{10}
= \frac{0 + 4 + 6 + 3}{10} = \frac{13}{10} = 1.3
\text{ buts par match.}
\]
\omterm{def:g9:statproba:indicators}{Médiane}~: la série triée est $0,0,1,1,1,1,2,2,2,3$~; les $5$\textsuperscript{e} et
$6$\textsuperscript{e} valeurs sont toutes deux $1$~: \omterm{def:g9:statproba:indicators}{médiane} $1$.
\end{solution}

\begin{solution}{exo:g9:statproba:4}
$\P(3) = \frac16$. \quad
$\P(\text{pair}) = \frac36 = \frac12$. \quad
$\P(\text{au moins } 3) = \P(\{3,4,5,6\}) = \frac46 = \frac23$. \quad
$\P(\text{pas } 6) = 1 - \frac16 = \frac56$.
\end{solution}

\begin{solution}{exo:g9:statproba:5}
$\P(\text{blanche}) = \frac{5}{12}$.

$\P(\text{rouge ou verte}) = \frac{4 + 3}{12} = \frac{7}{12}$.

$\P(\text{pas blanche}) = 1 - \frac{5}{12} = \frac{7}{12}$.

Les deux dernières sont égales~: «~pas blanche~» \emph{est} l'événement
«~rouge ou verte~».
\end{solution}

\begin{solution}{exo:g9:statproba:6}
Les $12$ issues (\omterm{def:g5:solids:def}{face} de la pièce, valeur du dé) sont équiprobables.

Pile et un $6$~: une issue sur $12$~: $\frac{1}{12}$.

Pile ou un $6$~: les issues avec pile ($6$ d'entre elles) plus l'issue
(\omterm{def:g5:solids:def}{face}, $6$)~: $7$ issues, donc $\frac{7}{12}$. Compter chaque événement
et soustraire le chevauchement donne la même chose~:
$\frac{6}{12} + \frac{2}{12} - \frac{1}{12} = \frac{7}{12}$.
\end{solution}

\begin{solution}{exo:g9:statproba:7}
Premier tirage~: rouge $\frac25$, noir $\frac35$. Second tirage sans
remise~: après un rouge, $1$ rouge et $3$ noirs restent parmi $4$~;
après un noir, $2$ rouges et $2$ noirs parmi $4$.

Deux noirs~: $\frac35 \times \frac24 = \frac{6}{20} = \frac{3}{10}$.

Couleurs différentes (RN ou NR)~:
$\frac25 \times \frac34 + \frac35 \times \frac24
= \frac{6}{20} + \frac{6}{20} = \frac{12}{20} = \frac35$.
\end{solution}

\begin{solution}{exo:g9:statproba:8}
\emph{1.} $9 \times 14 = 126$ points.

\emph{2.} Une \omterm{def:g9:statproba:indicators}{moyenne} de $15$ sur $10$ matchs signifie $150$ points au
total~: elle doit marquer $150 - 126 = 24$ points.
\end{solution}

\begin{solution}{exo:g9:statproba:9}
\emph{1.} Parmi les $36$ paires équiprobables, les doubles sont
$(1,1), (2,2), \dots, (6,6)$~: six d'entre eux, donc
$\P(\text{double}) = \frac{6}{36} = \frac16$.

\emph{2.} Deux parties indépendantes~: arbre avec gain ($\frac16$) /
perte ($\frac56$) à chaque niveau. \omterm{def:g9:statproba:probability}{Probabilité} de ne jamais gagner~:
$\frac56 \times \frac56 = \frac{25}{36}$. Donc
\[
\P(\text{au moins un gain}) = 1 - \frac{25}{36} = \frac{11}{36}
\approx 0.31 .
\]
\end{solution}

\begin{solution}{pb:g9:statproba:1}
\textbf{1.} Chaque lancer offre $5$ issues sans six sur $6$~; le
tableau $6 \times 6$ montre $5 \times 5 = 25$ issues sans six sur
$36$~: $\P(\text{aucun six}) = \frac{25}{36}$, donc
$\P(\text{au moins un}) = 1 - \frac{25}{36} = \frac{11}{36}
\approx 0.31$.

\textbf{2.} $\frac13 = \frac{12}{36} \neq \frac{11}{36}$.
L'addition compte l'issue «~six puis six~» une fois dans chaque
$\frac16$~: une fois de trop. Les \omterm{def:g9:statproba:probability}{probabilités} d'événements pouvant
survenir \emph{ensemble} ne s'additionnent pas simplement.

\textbf{3.} \omterm{def:g1:addition:def}{Somme} $7$~: les six issues $(1,6), (2,5), (3,4),
(4,3), (5,2), (6,1)$~: $\P = \frac{6}{36} = \frac16$. \omterm{def:g1:addition:def}{Somme} $2$~:
seulement $(1,1)$~: $\frac{1}{36}$. Le sept a le plus de façons de
se produire --- c'est le ventre du tableau des \omterm{def:g1:addition:def}{sommes}.

\textbf{4.} \omterm{def:g1:addition:def}{Somme} $9$~: $(3,6), (4,5), (5,4), (6,3)$ --- quatre
issues, $\frac{4}{36}$. \omterm{def:g1:addition:def}{Somme} $10$~: $(4,6), (5,5), (6,4)$ ---
trois, $\frac{3}{36}$. Le neuf est plus probable~: c'est le compte
des issues qui tranche, et non le nombre de «~façons d'écrire~» la
\omterm{def:g1:addition:def}{somme} avec des \omterm{def:g5:solids:def}{faces} non ordonnées.

\textbf{5.} Chaque étage de l'arbre multiplie la \omterm{def:g9:statproba:probability}{probabilité} de
n'avoir pas de six par $\frac56$, quoi qu'il se soit passé avant~:
après $n$ lancers,
$\P(\text{aucun six}) = \left(\frac56\right)^n$, et la règle de
l'événement contraire donne la formule.

\textbf{6.} $\left(\frac56\right)^4 = \frac{625}{1296} \approx
0.482$, donc le Chevalier gagnait avec la \omterm{def:g9:statproba:probability}{probabilité}
$1 - 0.482 = 0.518$ --- un avantage discret et régulier de près de
$2\,\%$ par partie~: un excellent pari, qui a payé ses carrosses.

\textbf{7.} À $7$ lancers, sa règle donne $\frac76$ --- une
\omterm{def:g9:statproba:probability}{probabilité} supérieure à $1$, ce qui n'a aucun sens. C'est le double
comptage de la question 2, aggravé~: les réussites des lancers se
recouvrent, et additionner leurs chances recompte les recouvrements
encore et encore.

\textbf{8.} $1 - \left(\frac{35}{36}\right)^{24} \approx
1 - 0.5086 = 0.491$~: en dessous d'une chance sur deux. En pariant à
\omterm{def:g1:addition:def}{somme} égale sur un événement de \omterm{def:g9:statproba:probability}{probabilité} $49.1\,\%$, le Chevalier
perdait environ $2$ parties sur $100$ en \omterm{def:g9:statproba:indicators}{moyenne} --- lentement,
mystérieusement (pour lui), et inéluctablement.

\textbf{9.} $1 - \left(\frac{35}{36}\right)^{25} \approx
1 - 0.494 = 0.506$~: avec $25$ lancers, le pari devient favorable.
Un seul lancer séparait le Chevalier de la rentabilité.

\textbf{10.} Multiplier une \omterm{def:g9:statproba:probability}{probabilité} par le nombre d'essais
compte plusieurs fois les réussites qui se recouvrent, et finit par
produire des réponses impossibles, supérieures à $1$. Le chemin
correct passe toujours par l'événement contraire~: enchaîner les
\omterm{def:g9:statproba:probability}{probabilités} d'échec par multiplication, puis retrancher de $1$.

\textbf{11.} \omterm{def:g9:statproba:indicators}{Moyenne}~: $\frac{9 \times 2\,000 + 20\,000}{10} =
\frac{38\,000}{10} = 3\,800$ euros. \omterm{def:g9:statproba:indicators}{Médiane}~: les salaires du
milieu valent tous deux $2\,000$~: la \omterm{def:g9:statproba:indicators}{médiane} vaut $2\,000$ euros.
L'annonce devrait citer la \omterm{def:g9:statproba:indicators}{médiane} --- neuf des dix salariés ne
verront jamais quoi que ce soit qui ressemble à $3\,800$~; la
\omterm{def:g9:statproba:indicators}{moyenne} est tirée vers le haut par une seule valeur extrême
(\cref{ex:g9:statproba:weighted} mettait déjà en garde contre les
poids).

\textbf{12.} Par exemple $2, 8, 15, 17, 18$~: \omterm{def:g9:statproba:indicators}{médiane} $15$ (la
valeur centrale), \omterm{def:g9:statproba:indicators}{moyenne} $\frac{60}{5} = 12$. Une \omterm{def:g9:statproba:indicators}{moyenne}
inférieure à la \omterm{def:g9:statproba:indicators}{médiane} trahit une traîne de notes \emph{basses} qui
tire la \omterm{def:g9:statproba:indicators}{moyenne} vers le bas, tandis que la \omterm{def:g3:division:half}{moitié} supérieure \omterm{def:g3:division:remainder}{reste}
haute.

\textbf{13.} Les trois premiers facteurs~:
$1 \times \frac{364}{365} \times \frac{363}{365} \approx 0.992$.
Chaque nouvel élève doit éviter une date occupée de plus~: les
facteurs diminuent donc, jusqu'à
$\frac{343}{365} \approx 0.94$ pour le $23^\text{e}$. Le \omterm{def:g2:mult:def}{produit}
complet tombe à environ $0.493$ --- d'où
\[
\P(\text{au moins un anniversaire partagé}) \approx 0.507 :
\]
mieux qu'une chance sur deux, dans une classe de $23$ élèves
seulement. L'intuition dit «~$23$ personnes, $365$ jours, aucune
chance~»~; les mathématiques répondent «~à pile ou \omterm{def:g5:solids:def}{face}~».

\textbf{14.} $\frac{23 \times 22}{2} = 253$ couples. Chaque couple
est une occasion nouvelle de coïncidence, et $253$ occasions à
$\frac{1}{365}$ chacune, ce n'est plus une petite affaire~: la
surprise se dissipe dès qu'on compte les couples, et non les
personnes.

\textbf{15.} Protocole possible~: lancer un dé par blocs de quatre
en notant si chaque bloc contient un six, pour $100$ blocs~; ou
recueillir les listes d'anniversaires de nombreuses classes
d'environ $23$ élèves et noter la proportion de celles qui
présentent une coïncidence. La \omterm{def:g7:stats:frequency}{fréquence} observée oscillera, mais à
mesure que le nombre de répétitions croît, elle devrait se
rapprocher de plus en plus des \omterm{def:g9:statproba:probability}{probabilités} calculées ($0.518$~;
$0.507$) --- les \omterm{def:g7:stats:frequency}{fréquences} convergent vers les \omterm{def:g9:statproba:probability}{probabilités}~: c'est
la loi des grands nombres, énoncée ici comme une attente et
démontrée dans le \omterm{def:g6:measure:volume}{volume} suivant.
\end{solution}
