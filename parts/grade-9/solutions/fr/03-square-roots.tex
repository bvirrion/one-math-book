\section*{Chapitre \ref{ch:g9:sqrt} --- Racines carrées}

\begin{solution}{exo:g9:sqrt:1}
$\sqrt{64} = 8$~; \quad $\sqrt{100} = 10$~; \quad
$\sqrt{0.09} = 0.3$ (car $0.3^2 = 0.09$)~; \quad
$\sqrt{\frac{1}{25}} = \frac15$~; \quad
$\left(\sqrt{13}\right)^2 = 13$~; \quad
$\sqrt{(-4)^2} = \sqrt{16} = 4 = \abs{-4}$.
\end{solution}

\begin{solution}{exo:g9:sqrt:2}
$\sqrt{50} = \sqrt{25 \times 2} = 5\sqrt2$~; \quad
$\sqrt{45} = \sqrt{9 \times 5} = 3\sqrt5$~; \quad
$\sqrt{98} = \sqrt{49 \times 2} = 7\sqrt2$~; \quad
$\sqrt{300} = \sqrt{100 \times 3} = 10\sqrt3$.
\end{solution}

\begin{solution}{exo:g9:sqrt:3}
$\sqrt2 \times \sqrt{18} = \sqrt{36} = 6$.

$\dfrac{\sqrt{75}}{\sqrt3} = \sqrt{\dfrac{75}{3}} = \sqrt{25} = 5$.

$\sqrt5 \times \sqrt{20} = \sqrt{100} = 10$.

$\dfrac{\sqrt8}{\sqrt{50}} = \sqrt{\dfrac{8}{50}} = \sqrt{\dfrac{4}{25}}
= \dfrac25$.
\end{solution}

\begin{solution}{exo:g9:sqrt:4}
$x^2 = 36$~: $x = 6$ ou $x = -6$.

$x^2 = 11$~: $x = \sqrt{11}$ ou $x = -\sqrt{11}$.

$x^2 + 4 = 0$ signifie $x^2 = -4 < 0$~: pas de solution.

$2x^2 = 50$~: $x^2 = 25$, donc $x = 5$ ou $x = -5$.
\end{solution}

\begin{solution}{exo:g9:sqrt:5}
$\sqrt{20} + \sqrt{45} = 2\sqrt5 + 3\sqrt5 = 5\sqrt5$.

$3\sqrt8 - \sqrt{18} + \sqrt2 = 3 \times 2\sqrt2 - 3\sqrt2 + \sqrt2
= (6 - 3 + 1)\sqrt2 = 4\sqrt2$.
\end{solution}

\begin{solution}{exo:g9:sqrt:6}
$\left(\sqrt3 + 1\right)^2 = 3 + 2\sqrt3 + 1 = 4 + 2\sqrt3$.

$\left(2\sqrt5 - 3\right)^2 = 4 \times 5 - 2 \times 3 \times 2\sqrt5 + 9
= 29 - 12\sqrt5$.

$\left(\sqrt6 + \sqrt2\right)\left(\sqrt6 - \sqrt2\right) = 6 - 2 = 4$.
\end{solution}

\begin{solution}{exo:g9:sqrt:7}
Côté~: $\sqrt{6400} = 80$ m. Diagonale (Pythagore)~:
$\sqrt{80^2 + 80^2} = 80\sqrt2 \approx 113$ m.
\end{solution}

\begin{solution}{exo:g9:sqrt:8}
$\dfrac{1}{\sqrt2} = \dfrac{1 \times \sqrt2}{\sqrt2 \times \sqrt2}
= \dfrac{\sqrt2}{2}$.

$\dfrac{6}{\sqrt3} = \dfrac{6\sqrt3}{3} = 2\sqrt3$.
\quad
$\dfrac{10}{\sqrt5} = \dfrac{10\sqrt5}{5} = 2\sqrt5$.
\end{solution}

\begin{solution}{exo:g9:sqrt:9}
\emph{1. Vrai}~: c'est la règle du \omterm{def:g2:mult:def}{produit} (\cref{thm:g9:sqrt:rules}).

\emph{2. Faux}~: $\sqrt{9 + 16} = 5$ mais $\sqrt9 + \sqrt{16} = 7$.

\emph{3. Faux} pour $x$ négatif~: $\sqrt{(-4)^2} = 4 \neq -4$. L'énoncé
correct est $\sqrt{x^2} = \abs{x}$.

\emph{4. Vrai}~: $\left(3\sqrt2\right)^2 = 9 \times 2 = 18$.
\end{solution}

\begin{solution}{exo:g9:sqrt:10}
$\left(2 + \sqrt3\right)^2 = 4 + 4\sqrt3 + 3 = 7 + 4\sqrt3$. Donc
$x = \sqrt{7 + 4\sqrt3} = \sqrt{\left(2+\sqrt3\right)^2} = 2 + \sqrt3$
(un nombre positif ou nul, donc la racine enlève juste le carré).

De même $\left(\sqrt5 - 2\right)^2 = 5 - 4\sqrt5 + 4 = 9 - 4\sqrt5$,
et $\sqrt5 - 2 > 0$, donc
$\sqrt{9 - 4\sqrt5} = \sqrt5 - 2$ (pas $2 - \sqrt5$, qui est
négatif~!).
\end{solution}

\begin{solution}{pb:g9:sqrt:1}
\textbf{1.} $1.4^2 = 1.96 < 2 < 2.25 = 1.5^2$~;
$1.41^2 = 1.9881 < 2 < 2.0164 = 1.42^2$. Trois décimales~:
$1.414^2 = 1.999396 < 2 < 2.002225 = 1.415^2$, donc
$1.414 < \sqrt2 < 1.415$.

\textbf{2.} En élevant $\sqrt2 = \frac ab$ au carré, on obtient
$2 = \frac{a^2}{b^2}$, d'où $a^2 = 2b^2$~: le nombre $a^2$ est le
double d'un entier --- il est \emph{\omterm{def:g3:numbers:evenodd}{pair}}.

\textbf{3.} Si $a$ était \omterm{def:g3:numbers:evenodd}{impair}, $a^2$ serait \omterm{def:g3:numbers:evenodd}{impair}
(\cref{pb:g8:pythagoras:1})~; comme $a^2$ est \omterm{def:g3:numbers:evenodd}{pair}, $a$ est \omterm{def:g3:numbers:evenodd}{pair}~:
$a = 2k$. En substituant~: $4k^2 = 2b^2$, donc $b^2 = 2k^2$ est
\omterm{def:g3:numbers:evenodd}{pair}, et par le même fait de parité, $b$ est \omterm{def:g3:numbers:evenodd}{pair}.

\textbf{4.} $a$ et $b$ tous deux \omterm{def:g3:numbers:evenodd}{pairs} signifie qu'ils sont tous
deux \omterm{def:g6:wholes:divisible}{divisibles} par $2$ --- mais $\frac ab$ était irréductible, sans
aucun \omterm{def:g9:arith:divisor}{diviseur} commun. Contradiction~: la \omterm{def:g9:fractions:fraction}{fraction} supposée ne peut
pas exister. \emph{Théorème~: $\sqrt2$ est irrationnel.}

\textbf{5.} La contradiction tient tout entière au mot
«~irréductible~»~: sans lui, «~$a$ et $b$ tous deux \omterm{def:g3:numbers:evenodd}{pairs}~» ne
contredit rien. Supposer la \omterm{def:g9:fractions:fraction}{fraction} irréductible est légitime parce
que toute \omterm{def:g9:fractions:fraction}{fraction} \emph{possède} une forme irréductible
(\cref{met:g9:arith:simplify}~: diviser par le \omterm{def:g9:arith:gcd}{PGCD}) --- si $\sqrt2$
était une \omterm{def:g9:fractions:fraction}{fraction} quelconque, il serait aussi une \omterm{def:g9:fractions:fraction}{fraction}
irréductible, et celle-là est impossible.

\textbf{6.} Dans les décompositions en facteurs premiers, les carrés
portent des \omterm{def:g8:powers:def}{exposants} \omterm{def:g3:numbers:evenodd}{pairs} (\cref{pb:g9:arith:1}). Dans
$a^2 = 3b^2$, l'\omterm{def:g8:powers:def}{exposant} de $3$ est \omterm{def:g3:numbers:evenodd}{pair} à gauche, mais \omterm{def:g3:numbers:evenodd}{impair} à
droite (\omterm{def:g3:numbers:evenodd}{pair} venant de $b^2$, plus un). Or aucun nombre n'a deux
décompositions (\cref{thm:g9:arith:factorization})~: contradiction,
et $\sqrt3$ est irrationnel.

\textbf{7.} Dans $a^2 = n b^2$, l'argument trouve un \omterm{def:g9:arith:prime}{nombre premier}
dont la parité de l'\omterm{def:g8:powers:def}{exposant} se contredit exactement lorsqu'un
\omterm{def:g9:arith:prime}{nombre premier} apparaît dans $n$ avec un \omterm{def:g8:powers:def}{exposant} \emph{\omterm{def:g3:numbers:evenodd}{impair}}. Si
tous les \omterm{def:g8:powers:def}{exposants} de $n$ sont \omterm{def:g3:numbers:evenodd}{pairs}, $n$ est un carré parfait
($n = m^2$, et $\sqrt n = m$ est entier). Résultat complet~:
$\sqrt n$ est irrationnel pour tout entier $n$ qui n'est pas un
carré parfait --- $\sqrt2, \sqrt3, \sqrt5, \sqrt6, \sqrt7, \sqrt8,
\sqrt{10}, \dots$

\textbf{8.} De $r x = q$ avec $r \neq 0$ on tire
$x = \frac qr$, un \omterm{def:g3:division:remainder}{quotient} de rationnels, donc rationnel. Ainsi, si
$x$ est irrationnel, $r x$ ne peut pas être rationnel~: $2\sqrt2$ et
$\frac{\sqrt2}{2}$ sont irrationnels (multiplicateurs $2$ et
$\frac12$).

\textbf{9.} Si $1 + \sqrt2 = q$ était rationnel, alors
$\sqrt2 = q - 1$ le serait aussi~: contradiction. Si
$s = \sqrt2 + \sqrt3$ était rationnel, alors
$s^2 = 2 + 2\sqrt6 + 3 = 5 + 2\sqrt6$ le serait aussi, ce qui
rendrait $\sqrt6 = \frac{s^2 - 5}{2}$ rationnel --- or $\sqrt6$ est
irrationnel (question 7). Donc $\sqrt2 + \sqrt3$ est irrationnel.

\textbf{10.} \omterm{def:g1:addition:def}{Sommes}~: $\sqrt2$ et $-\sqrt2$ (ou $\sqrt2$ et
$1 - \sqrt2$) sont chacun irrationnels, de \omterm{def:g1:addition:def}{sommes} rationnelles $0$
et $1$. \omterm{def:g2:mult:def}{Produits}~: $\sqrt2 \times \sqrt2 = 2$. L'irrationalité peut
s'évaporer au fil des opérations --- d'où les démonstrations au cas
par cas.

\textbf{11.} À partir de $x = 1$~: moyenne de $1$ et de
$\frac21 = 2$~: $x_1 = \frac32$. Puis moyenne de $\frac32$ et de
$\frac{2}{3/2} = \frac43$~:
$x_2 = \frac12\left(\frac32 + \frac43\right)
= \frac12 \cdot \frac{17}{6} = \frac{17}{12} \approx 1.4167$.

\textbf{12.} $x_3 = \frac12\left(\frac{17}{12} +
\frac{24}{17}\right) = \frac12 \cdot
\frac{289 + 288}{204} = \frac{577}{408} = 1.4142156\ldots$ \omterm{def:g5:solids:def}{Face} à
$\sqrt2 = 1.4142135\ldots$~: cinq décimales exactes après trois
étapes --- la recette double à peu près le nombre de décimales
exactes à chaque tour.

\textbf{13.} Si $x^2 > 2$ (devinette trop grande), alors
$x > \sqrt2$, et $\frac2x < \frac{2}{\sqrt2} = \sqrt2$~: le côté
compagnon est trop petit. Si $x^2 < 2$, les inégalités s'inversent.
Dans les deux cas, $\sqrt2$ se situe entre $x$ et $\frac2x$, donc
leur moyenne --- la devinette suivante --- est plus proche que le
pire des deux côtés~: le rectangle d'\omterm{def:g6:measure:area}{aire} $2$ devient plus carré à
chaque étape.

\textbf{14.} $9 - 8 = 1$~; \ $289 - 288 = 1$~; \
$332\,929 - 332\,928 = 1$. Chaque devinette $\frac ab$ de Héron
vérifie $a^2 - 2b^2 = 1$~: la partie I a démontré qu'atteindre $0$
est impossible, et ces \omterm{def:g9:fractions:fraction}{fractions} manquent l'impossible d'exactement
une unité --- au plus près que des entiers puissent jamais aller.

\textbf{15.} La droite graduée doit contenir des nombres au-delà des
\omterm{def:g9:fractions:fraction}{fractions} --- des longueurs comme $\sqrt2$, dont les écritures
décimales courent indéfiniment sans jamais se répéter~: les
\emph{nombres réels}. Leur construction honnête est une histoire
pour le \omterm{def:g6:measure:volume}{volume} suivant et, en toute rigueur, pour les \omterm{def:g6:measure:volume}{volumes}
universitaires.
\end{solution}
