\section*{Chapitre \ref{ch:g9:linfunc} --- Fonctions linéaires et affines}

\begin{solution}{exo:g9:linfunc:1}
Linéaires~: $f$ et $k$ (forme $ax$). Affines mais non linéaires~: $g$
(coefficient $3$, ordonnée à l'origine $-5$) et $m$ (constante,
coefficient $0$). Ni l'un ni l'autre~: $h$, à cause du carré.
\end{solution}

\begin{solution}{exo:g9:linfunc:2}
$f(0) = -3$~; $f(4) = 5$~; $f(-1) = -5$. Et $f(x) = 9$ signifie
$2x - 3 = 9$, donc $2x = 12$ et $x = 6$.
\end{solution}

\begin{solution}{exo:g9:linfunc:3}
$a = \frac{15}{6} = 2.5$, donc $f(x) = 2.5x$. Puis $f(10) = 25$ et
$f(-2) = -5$.
\end{solution}

\begin{solution}{exo:g9:linfunc:4}
$f$ passe par l'origine avec coefficient $2$~; $g$ a le même coefficient
mais coupe l'axe des $y$ en $3$~; $h$ décroît depuis $(0, 4)$ avec
coefficient $-1$. Les graphiques de $f$ et $g$ sont des \emph{\omterm{def:g6:lines:perp}{droites
parallèles}} (même coefficient, ordonnées à l'origine différentes).
\end{solution}

\begin{solution}{exo:g9:linfunc:5}
Pour $f$~: coefficient $\frac{11 - 5}{3 - 1} = 3$, puis
$5 = 3 \times 1 + b$ donne $b = 2$~: $f(x) = 3x + 2$.

Pour $g$~: coefficient $\frac{0 - 4}{2 - 0} = -2$, et $g(0) = 4$ est
déjà l'ordonnée à l'origine~: $g(x) = -2x + 4$.
\end{solution}

\begin{solution}{exo:g9:linfunc:6}
Hausse de $25\,\%$~: $120 \times 1.25 = 150$.

Remise de $40\,\%$~: $65 \times 0.6 = 39$.

Prix d'origine $p$ avec $p \times 0.75 = 69$~:
$p = \frac{69}{0.75} = 92$.
\end{solution}

\begin{solution}{exo:g9:linfunc:7}
\emph{1.} $f(x) = 0.05x + 10$.

\emph{2.} Le forfait B est moins cher lorsque $25 < 0.05x + 10$,
c'est-à-dire $15 < 0.05x$, c'est-à-dire $x > 300$. À partir de $301$
minutes ($5$ heures) par mois, le forfait B l'emporte.
\end{solution}

\begin{solution}{exo:g9:linfunc:8}
\emph{1.} Après une heure~: $2000 \times 1.1 = 2200$. Après deux
heures~: $2200 \times 1.1 = 2420$.

\emph{2.} La seconde hausse s'applique à $2200$, et non à $2000$~:
croître de $10\,\%$ deux fois multiplie par $1.1^2 = 1.21$, une hausse
de $21\,\%$, pas de $20\,\%$.
\end{solution}

\begin{solution}{exo:g9:linfunc:9}
\emph{1.} Coefficient~: $\frac{1 - 3}{6 - 2} = \frac{-2}{4} = -0.5$~: la
fonction est décroissante.

\emph{2.} $f(x) = -0.5x + b$ avec $f(2) = 3$~: $-1 + b = 3$, donc
$b = 4$ et $f(x) = -0.5x + 4$. Sortie $0$~: $-0.5x + 4 = 0$ donne
$x = 8$.
\end{solution}

\begin{solution}{exo:g9:linfunc:10}
\emph{1.} Les deux variations multiplient le prix par
\[
\left(1 + \frac{t}{100}\right)\left(1 - \frac{t}{100}\right)
= 1 - \frac{t^2}{10000}
\]
(troisième identité avec $a = 1$, $b = \frac{t}{100}$).

\emph{2.} Pour $0 < t \leq 100$, $\frac{t^2}{10000} > 0$, donc le
facteur est inférieur à $1$~: le prix final est plus bas que l'original.
Pour $t = 10$~: facteur $1 - \frac{100}{10000} = 0.99$, soit une baisse
globale de $1\,\%$.
\end{solution}

\begin{solution}{pb:g9:linfunc:1}
\textbf{1.} \omterm{prop:g9:linfunc:affinegraph}{Coefficient directeur}~:
$\frac{f(100) - f(0)}{100 - 0} = \frac{212 - 32}{100} = 1.8$~;
ordonnée à l'origine $f(0) = 32$. D'où
\[
f(C) = 1.8\,C + 32 .
\]

\textbf{2.} $f(20) = 68\,^\circ$F (la fameuse journée douce)~;
$f(37) = 98.6\,^\circ$F~; $f(-10) = 14\,^\circ$F.

\textbf{3.} De $F = 1.8C + 32$ on tire $C = \frac{F - 32}{1.8}$.
Alors $50\,^\circ$F $\to \frac{18}{1.8} = 10\,^\circ$C, et
$98.6\,^\circ$F $\to \frac{66.6}{1.8} = 37\,^\circ$C.

\textbf{4.} $f(0) = 32 \neq 0$, et $f(20) = 68$ n'est pas le double
de $f(10) = 50$~: la fonction n'est pas linéaire. Son graphe est une
droite qui ne passe pas par l'origine~: $f$ est \emph{affine}, et
non linéaire.

\textbf{5.} $1.8x + 32 = x$ donne $0.8x = -32$, donc $x = -40$~: à
quarante degrés sous \omterm{def:g1:counting:zero}{zéro}, les deux thermomètres affichent $-40$.
Sur le graphe, la droite de $f$ coupe exactement là la droite
d'\omterm{def:g8:equations:def}{équation} $y = x$.

\textbf{6.} $p_A(k) = 1.5k + 3$ et $p_B(k) = 2k$. Pour $4$ km~:
$9$ euros contre $8$~: B est moins cher. Pour $8$ km~: $15$ contre
$16$~: A est moins cher.

\textbf{7.} $1.5k + 3 = 2k$ donne $k = 6$~: à six kilomètres, les
deux coûtent $12$ euros. Graphiquement, la droite de B (passant par
l'origine, plus pentue) part en dessous de celle de A et la coupe en
$(6, 12)$~: B gagne sur les courtes distances, A sur les longues.

\textbf{8.} $N = 7 \times 20 - 30 = 110$ stridulations par minute.
De $82 = 7T - 30$~: $T = \frac{112}{7} = 16\,^\circ$C.

\textbf{9.} $v(t) = 600 - 15t$. Il vaut $240$~:
$600 - 15t = 240$, $t = 24$ mois. Il ne vaut plus rien~: $t = 40$
mois. Au-delà de $40$ mois, la formule devient négative --- or la
valeur d'un téléphone s'arrête à \omterm{def:g1:counting:zero}{zéro}~: tout modèle affine a un
domaine où il a du sens.

\textbf{10.} En composant~: $\times 1.25$ puis $\times 0.8$ revient
à multiplier par $1.25 \times 0.8 = 1$~: le prix revient exactement
à son point de départ. Aucun miracle~: les pourcentages agissent sur
des références différentes (la baisse s'applique au prix déjà
augmenté), et ici les multiplicateurs se compensent par hasard ---
l'histoire générale, avec sa perte incompressible
$1 - \frac{t^2}{10\,000}$, est celle de
l'\cref{exo:g9:linfunc:10}.

\textbf{11.} \omterm{prop:g9:linfunc:affinegraph}{Coefficient directeur}~: $\frac{17 - 20}{30} = -0.1$ cm
par minute~; $h(t) = 20 - 0.1t$. Elle s'éteint lorsque
$h(t) = 0$~: $t = 200$ minutes --- trois heures et vingt minutes de
lumière.

\textbf{12.} Le second est parti avec $10$ km d'avance (l'ordonnée à
l'origine)~; le premier va plus vite ($24 > 18$ km/h). Rattrapage~:
$24t = 10 + 18t$ donne $6t = 10$, $t = \frac53$ h $= 1$ h $40$ min,
au kilomètre $24 \times \frac53 = 40$.

\textbf{13.} $f(x) + g(x) = (2 - 0.5)x + (3 + 7) = 1.5x + 10$~:
c'est encore une \omterm{def:g9:linfunc:affine}{fonction affine}, dont le \omterm{prop:g9:linfunc:affinegraph}{coefficient directeur} est
la \emph{\omterm{def:g1:addition:def}{somme} des \omterm{prop:g9:linfunc:affinegraph}{coefficients directeurs}} et l'ordonnée à
l'origine la \omterm{def:g1:addition:def}{somme} des ordonnées à l'origine --- cela se voit en
regroupant les termes semblables dans $(mx + p) + (m'x + p')$.

\textbf{14.} Si $m \neq 0$~: exactement une solution,
$x = -\frac pm$ --- une droite non horizontale coupe l'axe une fois.
Si $m = 0$ et $p \neq 0$~: aucune solution --- une droite
horizontale située au-dessus ou en dessous de l'axe ne le touche
jamais. Si $m = 0$ et $p = 0$~: tout $x$ est solution --- la droite
\emph{est} l'axe.

\textbf{15.} Premier triplet~: taux $\frac{9 - 5}{3 - 1} = 2$ et
$\frac{17 - 9}{7 - 3} = 2$~: égaux, les points sont alignés (sur
$y = 2x + 3$). Second triplet~: $\frac{9-5}{2} = 2$ mais
$\frac{16 - 9}{3} = \frac73 \neq 2$~: ils ne sont pas alignés. Les
\omterm{def:g9:linfunc:affine}{fonctions affines} sont exactement celles dont le taux
d'accroissement est constant --- un seul nombre, le \omterm{prop:g9:linfunc:affinegraph}{coefficient
directeur}, raconte toute l'histoire~; pour les courbes, le taux
lui-même change d'un point à l'autre, et le poursuivre conduit à la
dérivée, dans le \omterm{def:g6:measure:volume}{volume} suivant.
\end{solution}
