\chapter{Vierkantswortels}\label{ch:g9:sqrt}

Hoe lang is de diagonaal van een vierkant met zijde $1$? De stelling van
Pythagoras antwoordt $\sqrt 2$, een getal met kwadraat $2$ --- en dat, zo blijkt,
geen \omterm{def:g9:fractions:fraction}{breuk} is. Dit hoofdstuk definieert \omterm{def:g9:sqrt:def}{vierkantswortels}, stelt de rekenregels
ervoor vast, en leert hoe je uitdrukkingen als $\sqrt{75}$ vereenvoudigt.

\section{Definitie en eerste eigenschappen}

\begin{definition}[Vierkantswortel]\label{def:g9:sqrt:def}
Zij $a \geq 0$. De \emph{vierkantswortel}\index{vierkantswortel} van $a$,
geschreven $\sqrt a$, is het enige \emph{niet-negatieve} getal met kwadraat $a$:
\[
\sqrt a \geq 0
\qquad\text{en}\qquad
\left(\sqrt a\right)^2 = a .
\]
Negatieve getallen hebben geen vierkantswortel, want elk kwadraat is
niet-negatief.
\end{definition}

\begin{example}
$\sqrt{49} = 7$, $\sqrt 0 = 0$, $\sqrt 1 = 1$, $\sqrt{2.25} = 1.5$. De eerste
\omterm{def:g9:sqrt:def}{vierkantswortels} om uit het hoofd te kennen zijn die van de \emph{volkomen
kwadraten}: $1, 4, 9, 16, 25, 36, 49, 64, 81, 100, 121, 144$.
\end{example}

$\sqrt{1^2 + 1^2} = \sqrt2 \approx 1.414$, volgens de stelling van Pythagoras
(\cref{ch:g9:trig} haalt die weer op). Dit getal is irrationaal: zijn decimalen
repeteren nooit.}
\end{omfigure}

\begin{proposition}[Vierkantswortel van een kwadraat]\label{prop:g9:sqrt:square}
Voor elk reëel getal $x$ (positief of niet):
\[
\sqrt{x^2} = \abs{x} =
\begin{cases}
x & \text{als } x \geq 0,\\
-x & \text{als } x < 0 .
\end{cases}
\]
\end{proposition}

\begin{proof}
Het getal $\abs{x}$ is niet-negatief en zijn kwadraat is $x^2$ (een getal en zijn
\omterm{def:g7:negatives:def}{tegengestelde} hebben hetzelfde kwadraat). Omdat het het enige niet-negatieve getal
met kwadraat $x^2$ is, is het $\sqrt{x^2}$.
\end{proof}

\begin{example}
$\sqrt{(-5)^2} = \sqrt{25} = 5 = \abs{-5}$: de \omterm{def:g9:sqrt:def}{vierkantswortel} ``vergeet'' het
teken, hij herstelt het niet. In het bijzonder is $\sqrt{x^2} = x$ \emph{verkeerd}
voor negatieve $x$.
\end{example}

\section{Producten en quotiënten van vierkantswortels}

\begin{theorem}[Regels voor vermenigvuldigen en delen]\label{thm:g9:sqrt:rules}
Voor alle $a \geq 0$ en $b \geq 0$:
\[
\sqrt{a b} = \sqrt a \times \sqrt b,
\qquad\text{en voor } b > 0:\quad
\sqrt{\frac ab} = \frac{\sqrt a}{\sqrt b} .
\]
\end{theorem}

\begin{proof}
Het getal $\sqrt a \times \sqrt b$ is niet-negatief (een \omterm{def:g2:mult:def}{product} van
niet-negatieve getallen), en zijn kwadraat is
\[
\left(\sqrt a \times \sqrt b\right)^2
= \left(\sqrt a\right)^2 \times \left(\sqrt b\right)^2
= a b .
\]
Omdat het niet-negatieve getal met kwadraat $ab$ uniek is, is het gelijk aan
$\sqrt{ab}$. De regel voor het \omterm{def:g3:division:remainder}{quotiënt} bewijs je op dezelfde manier.
\end{proof}

\begin{remark}[Zo'n regel bestaat niet voor sommen!]
$\sqrt{a + b}$ is in het algemeen \emph{niet} $\sqrt a + \sqrt b$:
\[
\sqrt{9 + 16} = \sqrt{25} = 5,
\qquad\text{maar}\qquad
\sqrt 9 + \sqrt{16} = 3 + 4 = 7 .
\]
\end{remark}

\begin{method}[$\sqrt n$ vereenvoudigen]\label{met:g9:sqrt:simplify}
Om de \omterm{def:g9:sqrt:def}{vierkantswortel} van een geheel getal te vereenvoudigen:
\begin{enumerate}
  \item zoek het grootste volkomen kwadraat dat $n$ \omterm{def:g9:arith:divisor}{deelt} (ontbind $n$ als dat
        nodig is);
  \item splits met de productregel en haal het kwadraat eruit:
        $\sqrt{k^2 m} = k\sqrt m$;
  \item ga na dat geen volkomen kwadraat meer \omterm{def:g9:arith:divisor}{deelt} wat onder de wortel is
        overgebleven.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:sqrt:simplify}
Vereenvoudig $\sqrt{75}$: omdat $75 = 25 \times 3$, is
\[
\sqrt{75} = \sqrt{25 \times 3} = \sqrt{25} \times \sqrt 3 = 5\sqrt3 .
\]
Vereenvoudig $\sqrt{72}$: omdat $72 = 36 \times 2$, is $\sqrt{72} = 6\sqrt2$. En
een \omterm{def:g3:division:remainder}{quotiënt}: $\sqrt{\dfrac{49}{16}} = \dfrac{\sqrt{49}}{\sqrt{16}} =
\dfrac74$.
\end{example}

\begin{example}[Vierkantswortels optellen]\label{ex:g9:sqrt:add}
Sommen van \omterm{def:g9:sqrt:def}{vierkantswortels} vereenvoudigen alleen wanneer de wortels
\emph{gelijksoortig} zijn. Bereken $\sqrt{12} + \sqrt{27} - \sqrt{48}$, en
vereenvoudig eerst elke term:
\begin{align*}
\sqrt{12} &= \sqrt{4 \times 3} = 2\sqrt3, \\
\sqrt{27} &= \sqrt{9 \times 3} = 3\sqrt3, \\
\sqrt{48} &= \sqrt{16 \times 3} = 4\sqrt3,
\end{align*}
zodat de \omterm{def:g1:addition:def}{som} $2\sqrt3 + 3\sqrt3 - 4\sqrt3 = (2 + 3 - 4)\sqrt3 = \sqrt3$ is.
\end{example}

\begin{example}[Uitwerken met vierkantswortels]\label{ex:g9:sqrt:expand}
De merkwaardige \omterm{def:g2:mult:def}{producten} van de algebra gelden ook voor \omterm{def:g9:sqrt:def}{vierkantswortels}. Werk
$\left(\sqrt5 + 2\right)^2$ uit met $(a+b)^2 = a^2 + 2ab + b^2$:
\[
\left(\sqrt5 + 2\right)^2 = 5 + 2 \times 2\sqrt5 + 4 = 9 + 4\sqrt5 .
\]
En met het derde \omterm{def:g2:mult:def}{product}, $(a+b)(a-b) = a^2 - b^2$:
\[
\left(\sqrt7 + \sqrt3\right)\left(\sqrt7 - \sqrt3\right) = 7 - 3 = 4 :
\]
het \omterm{def:g2:mult:def}{product} van twee irrationale getallen kan een geheel getal zijn.
\end{example}

\section{De vergelijking \texorpdfstring{$x^2 = a$}{x2 = a}}

\begin{theorem}[$x^2 = a$ oplossen]\label{thm:g9:sqrt:equation}
\begin{enumerate}
  \item Is $a > 0$, dan heeft de \omterm{def:g8:equations:def}{vergelijking} $x^2 = a$ precies twee oplossingen:
        $\sqrt a$ en $-\sqrt a$;
  \item is $a = 0$, dan is $0$ de enige oplossing;
  \item is $a < 0$, dan is er geen oplossing.
\end{enumerate}
\end{theorem}

\begin{proof}
Herschrijf $x^2 = a$ als $x^2 - \left(\sqrt a\right)^2 = 0$ wanneer $a \geq 0$, en
ontbind als een \omterm{ex:g1:subtraction:difference}{verschil} van kwadraten:
\[
\left(x - \sqrt a\right)\left(x + \sqrt a\right) = 0 ,
\]
dus is $x = \sqrt a$ of $x = -\sqrt a$ (die vallen samen als $a = 0$). Is
$a < 0$, dan is er geen oplossing, want $x^2 \geq 0 > a$ voor elke $x$.
\end{proof}

\begin{axis}[omaxis,

  ytick={3}, yticklabels={$3$}]
  \addplot[omDef, thick, domain=-2.25:2.25, samples=90] {x^2};

\end{axis}

{\small $x^2 = 3$ oplossen op de parabool $y = x^2$: de horizontale \omterm{def:g6:lines:objects}{lijn} $y = 3$
snijdt haar in de twee abscissen $\pm\sqrt3$.}
\end{omfigure}

\begin{example}
$x^2 = 16$: oplossingen $4$ en $-4$. \quad
$x^2 = 5$: oplossingen $\sqrt5$ en $-\sqrt5$ (exacte waarden;
$\sqrt5 \approx 2.236$). \quad
$x^2 = -9$: geen oplossing. \quad
$3x^2 = 21$: deel eerst door 3, dan is $x^2 = 7$, met oplossingen $\pm\sqrt7$.
\end{example}

Vereenvoudig:

\sqrt{50}, \qquad \sqrt{45}, \qquad \sqrt{98}, \qquad \sqrt{300} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:sqrt:3}
Bereken en vereenvoudig:
\[
\sqrt2 \times \sqrt{18}, \qquad
\frac{\sqrt{75}}{\sqrt3}, \qquad
\sqrt5 \times \sqrt{20}, \qquad
\frac{\sqrt{8}}{\sqrt{50}} .

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:sqrt:4}
Los op: $x^2 = 36$; \quad $x^2 = 11$; \quad $x^2 + 4 = 0$; \quad
$2x^2 = 50$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:sqrt:5}
Breng samen tot één term: $\sqrt{20} + \sqrt{45}$, en daarna
$3\sqrt8 - \sqrt{18} + \sqrt2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:sqrt:6}
Werk uit en vereenvoudig:
\[
\left(\sqrt3 + 1\right)^2, \qquad
\left(2\sqrt5 - 3\right)^2, \qquad
\left(\sqrt6 + \sqrt2\right)\left(\sqrt6 - \sqrt2\right).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:sqrt:7}
Een vierkant veld heeft \omterm{def:g6:measure:area}{oppervlakte} $6400$ m$^2$. Hoe lang is zijn zijde? En zijn
diagonaal (exacte waarde, daarna afgerond op de meter)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:sqrt:8}
Toon aan dat $\dfrac{1}{\sqrt2} = \dfrac{\sqrt2}{2}$ (vermenigvuldig \omterm{def:g4:fractions:def}{teller} en
\omterm{def:g4:fractions:def}{noemer} met $\sqrt2$). Schrijf met dezelfde truc $\dfrac{6}{\sqrt3}$ en
$\dfrac{10}{\sqrt5}$ zonder \omterm{def:g9:sqrt:def}{vierkantswortel} in de \omterm{def:g4:fractions:def}{noemer}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:sqrt:9}
Waar of niet waar? Verantwoord met een bewijs of een tegenvoorbeeld.
\begin{enumerate}
  \item Voor alle $a, b \geq 0$: $\sqrt{ab} = \sqrt a \sqrt b$.
  \item Voor alle $a, b \geq 0$: $\sqrt{a+b} = \sqrt a + \sqrt b$.
  \item Voor alle $x$: $\sqrt{x^2} = x$.
  \item $\left(3\sqrt2\right)^2 = 18$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:sqrt:10}
Zij $x = \sqrt{7 + 4\sqrt3}$. Bereken $\left(2 + \sqrt3\right)^2$ en leid daaruit
een eenvoudiger uitdrukking voor $x$ af. Dezelfde vraag voor
$\sqrt{9 - 4\sqrt5}$ (mik op een kwadraat van de vorm
$\left(\sqrt5 - 2\right)^2$, en let op het teken).
\end{exercise}

\section{Opgave: het getal dat geen breuk is}

\begin{problem}[{Weekendopgave --- het legendarische bewijs dat $\sqrt2$
irrationaal is, zijn vele neefjes, en het griezelig goede recept van Heron om het
te benaderen}]
\label{pb:g9:sqrt:1}
De diagonaal van een eenheidsvierkant is een volkomen echte lengte --- je hebt ze
in \cref{ex:g8:pythagoras:diagonal} getekend --- en toch meet, zoals de
pythagoreërs zo'n vijfentwintig eeuwen geleden tot hun ontzetting ontdekten,
\emph{geen enkele \omterm{def:g9:fractions:fraction}{breuk}} haar. De legende beweert dat de ontdekking met
verdrinking werd bestraft. Deze opgave loodst je door het onsterfelijke bewijs
(vier vragen en het is voor het leven van jou), vermenigvuldigt de slachtoffers,
en eindigt met het recept waarmee ingenieurs tweeduizend jaar lang het
onhandelbare getal handelbaar maakten.

\medskip
\noindent\textbf{Deel I --- Het bewijs.}
\begin{enumerate}
  \item Zit het eerst op de hielen: ga na dat $1.4 < \sqrt2 < 1.5$, en daarna dat
        $1.41 < \sqrt2 < 1.42$, door de grenzen te kwadrateren. Nog één cijfer:
        tussen welke getallen met drie decimalen ligt $\sqrt2$?
  \item Stel nu --- om tot een tegenspraak te komen --- dat $\sqrt2 = \frac ab$
        voor een zekere \omterm{def:g9:fractions:fraction}{breuk} \emph{in eenvoudigste vorm}
        (\cref{met:g9:arith:simplify}). Kwadrateer beide leden en toon aan dat
        $a^2 = 2 b^2$. Wat is de pariteit van $a^2$?
  \item De feiten over pariteit uit \cref{pb:g8:pythagoras:1} zeggen: \omterm{def:g3:numbers:evenodd}{oneven}
        getallen hebben \omterm{def:g3:numbers:evenodd}{oneven} kwadraten. Leid af dat $a$ \omterm{def:g3:numbers:evenodd}{even} is, schrijf
        $a = 2k$, vul in --- en toon aan dat ook $b$ \omterm{def:g3:numbers:evenodd}{even} moet zijn.
  \item Waar zit de tegenspraak? Besluit, en formuleer de stelling helemaal:
        \emph{$\sqrt2$ is irrationaal --- het is geen \omterm{def:g3:division:remainder}{quotiënt} van hele getallen}.
  \item Het bewijs leunde één keer, onopvallend, op de woorden ``in eenvoudigste
        vorm''. Wijs de precieze stap aan die zonder die woorden zou instorten, en
        leg uit waarom je die eenvoudigste vorm van meet af aan mocht aannemen.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De slachtoffers vermenigvuldigen zich.}
\begin{enumerate}[resume]
  \item Een tweede bewijsstijl, via priemfactorontbindingen
        (\cref{thm:g9:arith:factorization}): vergelijk in $a^2 = 3b^2$ de pariteit
        van de \emph{\omterm{def:g8:powers:def}{exponent} van $3$} in beide leden (\cref{pb:g9:arith:1}:
        kwadraten dragen \omterm{def:g3:numbers:evenodd}{even} \omterm{def:g8:powers:def}{exponenten}). Besluit dat $\sqrt3$ irrationaal is.
  \item Voer hetzelfde argument met \omterm{def:g8:powers:def}{exponenten} uit op $a^2 = n b^2$ voor een
        willekeurig heel getal $n$: voor welke $n$ levert het een tegenspraak, en
        voor welke mislukt het? Formuleer het volledige resultaat: $\sqrt n$ is
        irrationaal precies wanneer\,\dots
  \item Bewijs het schildlemma: een rationaal getal verschillend van \omterm{def:g1:counting:zero}{nul} maal een
        irrationaal getal is irrationaal. (Stel $r x = q$ met $r$, $q$ rationaal en
        $r \neq 0$, en los op naar $x$.) Leid af dat $2\sqrt2$ en
        $\frac{\sqrt2}{2}$ irrationaal zijn.
  \item Bewijs dat $1 + \sqrt2$ irrationaal is. En dan de mooie: stel dat
        $s = \sqrt2 + \sqrt3$ rationaal zou zijn, bereken $s^2$, zonder $\sqrt6$
        af, en vind de tegenspraak.
  \item Tem het enthousiasme: geef twee irrationale getallen waarvan de
        \emph{\omterm{def:g1:addition:def}{som}} rationaal is, en twee waarvan het \emph{\omterm{def:g2:mult:def}{product}} rationaal is.
        (Irrationaliteit blijft niet bewaard onder rekenkundige bewerkingen ---
        elk geval vraagt zijn eigen bewijs.)
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Het recept van Heron.}
Tweeduizend jaar vóór de rekenmachines benaderde Heron van Alexandrië $\sqrt2$
zo: \emph{neem een schatting $x$; vervang de schatting door het gemiddelde van $x$
en $\frac2x$; herhaal.}
\begin{enumerate}[resume]
  \item Begin bij de schatting $x = 1$ en bereken de volgende twee schattingen als
        exacte \omterm{def:g9:fractions:fraction}{breuken}.
  \item Bereken de derde schatting, opnieuw als een exacte \omterm{def:g9:fractions:fraction}{breuk}, en haar decimale
        waarde. Vergelijk met vraag 1: hoeveel decimalen van $\sqrt2$ zijn al
        juist?
  \item Het idee achter het recept: een rechthoek met \omterm{def:g6:measure:area}{oppervlakte} $2$ en één zijde
        $x$ heeft als andere zijde $\frac2x$. Toon aan dat $\sqrt2$ altijd
        \emph{tussen} $x$ en $\frac2x$ ligt (bekijk de twee gevallen $x^2 > 2$ en
        $x^2 < 2$), zodat het gemiddelde nemen van de twee zijden de rechthoek naar
        het vierkant toe knijpt.
  \item In de schattingen $\frac32$, $\frac{17}{12}$, $\frac{577}{408}$ zit een
        juweel verborgen: bereken $3^2 - 2 \times 2^2$, daarna
        $17^2 - 2 \times 12^2$, en daarna $577^2 - 2 \times 408^2$. Deel I bewees
        dat $a^2 - 2b^2 = 0$ onmogelijk is --- hoe dicht komen de \omterm{def:g9:fractions:fraction}{breuken} van Heron
        bij het onmogelijke?
  \item Slot, in twee zinnen: de diagonaal van het eenheidsvierkant bestaat op
        papier, geen enkele \omterm{def:g9:fractions:fraction}{breuk} meet haar, en haar decimale schrijfwijze kan nooit
        repeteren (\cref{pb:g9:fractions:1}). Welk soort ``nieuwe getallen'' moet de
        getallenlijn dus bevatten --- en waar in deze reeks wordt hun volledige
        verhaal verteld?
\end{enumerate}
\end{problem}
