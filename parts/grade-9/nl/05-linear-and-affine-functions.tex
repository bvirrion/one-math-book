\chapter{Lineaire en affiene functies}\label{ch:g9:linfunc}

``Drie kilogram kost drie keer zoveel als één'': dat is \omterm{def:g9:linfunc:linear}{evenredigheid}, de
eenvoudigste manier waarop twee grootheden met elkaar kunnen samenhangen. \omterm{def:g9:linfunc:linear}{Lineaire
functies} zijn \omterm{def:g9:linfunc:linear}{evenredigheid} in de taal van de functies; \omterm{def:g9:linfunc:affine}{affiene functies} voegen er
een vast startbedrag aan toe. Hun grafieken zijn rechte \omterm{def:g6:lines:objects}{lijnen}, en die \omterm{def:g6:lines:objects}{lijnen} lezen
is de vaardigheid die dit hoofdstuk opbouwt.

\section{Evenredigheid en lineaire functies}

\begin{definition}[Lineaire functie]\label{def:g9:linfunc:linear}
Een \emph{lineaire functie}\index{lineaire functie} vermenigvuldigt elk getal met
een vast getal $a$:
\[
f(x) = a x .
\]
Het getal $a$ is de \emph{coëfficiënt} van $f$. Twee grootheden zijn
\emph{evenredig}\index{evenredigheid} precies wanneer de ene een lineaire functie
van de andere is.
\end{definition}

\begin{example}\label{ex:g9:linfunc:linear}
Kosten appels $3$ per kilogram, dan is de prijs van $x$ kilogram $f(x) = 3x$:
twee keer zoveel kopen kost twee keer zoveel, en de prijs van $2.5$ kg is
$f(2.5) = 7.5$.

Omgekeerd: voldoet een \omterm{def:g9:linfunc:linear}{lineaire functie} aan $f(4) = 10$, dan is
$a = \frac{10}{4} = 2.5$ en $f(x) = 2.5x$: \emph{één waarde bepaalt een \omterm{def:g9:linfunc:linear}{lineaire
functie} volledig.}
\end{example}

\begin{proposition}[Grafiek van een lineaire functie]\label{prop:g9:linfunc:lineargraph}
De grafiek van $f(x) = ax$ is een rechte \omterm{def:g6:lines:objects}{lijn} \emph{door de oorsprong}. Omgekeerd
is elke niet-verticale \omterm{def:g6:lines:objects}{lijn} door de oorsprong de grafiek van een \omterm{def:g9:linfunc:linear}{lineaire functie}.
\end{proposition}
\admitted

\section{Affiene functies}

\begin{definition}[Affiene functie]\label{def:g9:linfunc:affine}
Een \emph{affiene functie}\index{affiene functie} heeft de vorm
\[
f(x) = a x + b ,
\]
waar $a$ en $b$ vaste getallen zijn. \omterm{def:g9:linfunc:linear}{Lineaire functies} zijn het bijzondere geval
$b = 0$; constante functies het geval $a = 0$.
\end{definition}

\begin{example}
Een sportclub rekent $20$ inschrijvingsgeld en daarna $5$ per bezoek: de totale
kosten van $x$ bezoeken zijn $f(x) = 5x + 20$. De kosten zijn \emph{niet}
\omterm{def:g9:linfunc:linear}{evenredig} met het aantal bezoeken (10 bezoeken kosten niet twee keer zoveel als
5), maar elk bezoek extra voegt hetzelfde bedrag toe, $5$.
\end{example}

\begin{proposition}[Grafiek van een affiene functie]\label{prop:g9:linfunc:affinegraph}
De grafiek van $f(x) = ax + b$ is een rechte \omterm{def:g6:lines:objects}{lijn}:
\begin{itemize}
  \item $b = f(0)$ is het \emph{snijpunt met de $y$-as}: de \omterm{def:g6:lines:objects}{lijn} snijdt de
        verticale as in $(0, b)$;
  \item $a$ is de \emph{richtingscoëfficiënt}\index{richtingscoëfficiënt}: één
        eenheid naar rechts gaan betekent $a$ eenheden omhoog (omlaag als
        $a < 0$); de functie is stijgend wanneer $a > 0$ en dalend wanneer
        $a < 0$;
  \item voor twee willekeurige inputs $u \neq v$ geldt
        $a = \dfrac{f(v) - f(u)}{v - u}$.
\end{itemize}
\end{proposition}

\begin{proof}[Bewijs van het laatste punt]
$f(v) - f(u) = (av + b) - (au + b) = a(v - u)$; deel door $v - u$. De \omterm{def:g3:division:remainder}{rest} wordt
op dit niveau aangenomen (het bovenbouwvolume behandelt \omterm{def:g6:lines:objects}{lijnen} en hun
\omterm{def:g8:equations:def}{vergelijkingen} volledig).
\end{proof}

\begin{method}[Een affiene functie uit twee waarden vinden]\label{met:g9:linfunc:findaffine}
Gegeven $f(u)$ en $f(v)$ met $u \neq v$:
\begin{enumerate}
  \item bereken de \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} $a = \dfrac{f(v) - f(u)}{v - u}$;
  \item vul een van de twee bekende waarden in $f(x) = ax + b$ in om $b$ te
        vinden;
  \item ga het na met de andere waarde.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:linfunc:findaffine}
Zoek de \omterm{def:g9:linfunc:affine}{affiene functie} met $f(2) = 7$ en $f(5) = 16$.
\[
a = \frac{16 - 7}{5 - 2} = \frac93 = 3 ,
\]
en dan geeft $f(2) = 3 \times 2 + b = 7$ dat $b = 1$: $f(x) = 3x + 1$. Controle:
$f(5) = 15 + 1 = 16$.
\end{example}

\section{Percenten}

\begin{proposition}[Verandering in percent]\label{prop:g9:linfunc:percent}
Een grootheid met $t\,\%$ verhogen vermenigvuldigt ze met
$1 + \dfrac{t}{100}$; ze met $t\,\%$ verlagen vermenigvuldigt ze met
$1 - \dfrac{t}{100}$. Veranderingen in percent zijn \omterm{def:g9:linfunc:linear}{lineaire functies}.
\end{proposition}

\begin{proof}
$x$ met $t\,\%$ verhogen betekent $\frac{t}{100}\,x$ optellen:
\[
x + \frac{t}{100}\,x = \left(1 + \frac{t}{100}\right) x .
\]
Dezelfde berekening met een minteken voor een verlaging.
\end{proof}

\begin{example}\label{ex:g9:linfunc:percent}
Een prijs van $80$ stijgt met $15\%$: nieuwe prijs $80 \times 1.15 = 92$. Een
prijs van $60$ daalt met $30\%$: $60 \times 0.7 = 42$.

\emph{Veranderingen achter elkaar zetten vermenigvuldigt de factoren.} Een
stijging van $20\%$ gevolgd door een daling van $20\%$ geeft
$1.2 \times 0.8 = 0.96$: een \emph{daling} van $4\%$ in totaal, en geen terugkeer
naar het begin!
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:1}
Welke van deze functies zijn lineair? Welke affien? Welke geen van beide?
\[
f(x) = 4x, \qquad
g(x) = 3x - 5, \qquad
h(x) = x^2, \qquad
k(x) = -\tfrac x2, \qquad
m(x) = 7 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:2}
Zij $f(x) = 2x - 3$. Bereken $f(0)$, $f(4)$ en $f(-1)$, en zoek het getal $x$
waarvoor $f(x) = 9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:3}
Een \omterm{def:g9:linfunc:linear}{lineaire functie} voldoet aan $f(6) = 15$. Zoek haar coëfficiënt, en bereken
daarna $f(10)$ en $f(-2)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:4}
Teken in hetzelfde assenstelsel de grafieken van $f(x) = 2x$, $g(x) = 2x + 3$ en
$h(x) = -x + 4$. Wat kun je zeggen over de grafieken van $f$ en $g$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:5}
Zoek de \omterm{def:g9:linfunc:affine}{affiene functie} waarvoor $f(1) = 5$ en $f(3) = 11$; en daarna die
waarvoor $g(0) = 4$ en $g(2) = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:6}
Bereken: de prijs na een stijging van $25\%$ op $120$; de prijs na een korting van
$40\%$ op $65$; de oorspronkelijke prijs als een artikel na een korting van $25\%$
$69$ kost.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:7}
Telefoonabonnement A kost $10$ per maand plus $0.05$ per minuut bellen;
abonnement B kost $25$ per maand met onbeperkt bellen.
\begin{enumerate}
  \item Druk de maandkosten van abonnement A uit als functie van het aantal
        minuten $x$.
  \item Vanaf hoeveel minuten per maand is abonnement B goedkoper?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:8}
Een populatie van $2000$ bacteriën groeit elk uur met $10\%$.
\begin{enumerate}
  \item Hoeveel bacteriën zijn er na één uur? En na twee uur?
  \item Leg uit waarom het antwoord na twee uur niet $2400$ is.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:9}
De grafiek van een \omterm{def:g9:linfunc:affine}{affiene functie} gaat door de punten $(2, 3)$ en $(6, 1)$.
\begin{enumerate}
  \item Bereken haar \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt}. Is de functie stijgend of dalend?
  \item Geef haar uitdrukking, en bereken de input waarvoor de output $0$ is.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:linfunc:10}
Een winkel verhoogt een prijs eerst met $t\,\%$ en verlaagt daarna de nieuwe prijs
met $t\,\%$.
\begin{enumerate}
  \item Toon aan dat de eindprijs gelijk is aan de oorspronkelijke, vermenigvuldigd
        met $1 - \dfrac{t^2}{10000}$.
  \item Leid af dat de eindprijs altijd \emph{lager} is dan de oorspronkelijke
        (voor $0 < t \leq 100$), en zoek de totale verandering in percent voor
        $t = 10$.
\end{enumerate}
\end{exercise}

\section{Opgave: de twee thermometers}

\begin{problem}[{Weekendopgave --- Celsius, Fahrenheit en de affiene functies van
het dagelijks leven: één temperatuur leest hetzelfde op beide schalen, en een
constante richtingscoëfficiënt is het handschrift van rechtheid}]
\label{pb:g9:linfunc:1}
Amerikanen horen ``$68$ graden'' en kleden zich licht; Europeanen horen het en
grijpen naar een jas. De twee thermometers van de wereld hangen samen via een
\omterm{def:g9:linfunc:affine}{affiene functie} --- en die uit twee feiten opbouwen is precies de vaardigheid van
\cref{met:g9:linfunc:findaffine}. Deze opgave construeert de omzetting, vindt de
onheilspellende temperatuur waarop de twee schalen het eens zijn, leest daarna
taxi's, krekels en kaarsen met dezelfde bril, en eindigt met de toets die rechtheid
zelf herkent.

\medskip
\noindent\textbf{Deel I --- De omzetting opbouwen.}
Twee feiten leggen de fahrenheitschaal vast: water bevriest bij $32\,^\circ$F en
kookt bij $212\,^\circ$F (bij $0\,^\circ$C en $100\,^\circ$C).
\begin{enumerate}
  \item Zoek de \omterm{def:g9:linfunc:affine}{affiene functie} $F = f(C)$ die graden Celsius omzet in graden
        Fahrenheit (\cref{met:g9:linfunc:findaffine} met de waarden $f(0) = 32$ en
        $f(100) = 212$).
  \item Zet om in graden Fahrenheit: een milde dag van $20\,^\circ$C; de
        $37\,^\circ$C van het \omterm{def:g5:solids:def}{lichaam}; een ijzige $-10\,^\circ$C.
  \item Los op naar $C$ om de omgekeerde omzetting op te bouwen, en gebruik die op
        $50\,^\circ$F en op $98.6\,^\circ$F.
  \item Is $f$ een \emph{lineaire} functie (\cref{def:g9:linfunc:linear})? Toets de
        verklikkers: wat is $f(0)$, en verdubbelt de waarde in Fahrenheit als je de
        waarde in Celsius verdubbelt? Besluit met het juiste woord
        (\cref{def:g9:linfunc:affine}).
  \item Het klassieke raadsel: bestaat er een temperatuur die \emph{hetzelfde
        getal} aanwijst op beide schalen? Los $f(x) = x$ op, en beschrijf wat je
        oplossing op de grafiek van $f$ betekent (welke \omterm{def:g6:lines:objects}{lijn} snijdt de grafiek
        daar?).
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- \omterm{def:g9:linfunc:affine}{Affiene functies} in het wild.}
\begin{enumerate}[resume]
  \item Taxi A rekent $3$ euro plus $1.50$ per km; taxi B rekent een vast bedrag
        van $2$ euro per km. Schrijf de twee prijsfuncties op en vergelijk ze voor
        een rit van $4$ km en een rit van $8$ km.
  \item Zoek de afstand waarbij de twee taxi's evenveel kosten, en beschrijf de
        situatie op een grafiek (twee \omterm{def:g6:lines:objects}{lijnen} --- wat gebeurt er in het snijpunt,
        ervoor en erna?).
  \item Vuistregel van natuurkenners: een krekel tjirpt bij een temperatuur van
        $T\,^\circ$C ongeveer $N = 7T - 30$ keer per minuut. Hoeveel tjirpen op een
        avond van $20\,^\circ$C? Je telt $82$ tjirpen in een minuut: welke
        temperatuur kondigt de krekel aan?
  \item Een telefoon die $600$ euro kostte, verliest $15$ euro waarde per maand:
        schrijf de waardefunctie $v(t)$ op, zoek wanneer de telefoon $240$ euro
        waard is, en wanneer het model zegt dat hij niets meer waard is. Betekent
        de formule na die datum nog iets?
  \item Veranderingen in percent zijn \omterm{def:g9:linfunc:linear}{lineaire functies}: $+25\,\%$ is
        vermenigvuldigen met $1.25$, en $-20\,\%$ met $0.8$
        (\cref{prop:g9:linfunc:percent}). Zet ze achter elkaar: wat doet een
        stijging van $+25\,\%$ gevolgd door een daling van $-20\,\%$ met een prijs?
        Leg het kleine wonder uit, en vergelijk met \cref{exo:g9:linfunc:10}.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} is alles.}
\begin{enumerate}[resume]
  \item Een kaars meet $20$ cm wanneer ze wordt aangestoken en $17$ cm een half uur
        later. Neem aan dat het smelten affien verloopt, zoek $h(t)$ (hoogte in cm,
        $t$ in minuten), en voorspel wanneer de kaars opgebrand is.
  \item Twee wielrenners rijden dezelfde weg: de afstand van de eerste is
        $d_1(t) = 24t$ (km na $t$ uur), die van de tweede $d_2(t) = 10 + 18t$. Wie
        begon met een voorsprong, wie rijdt sneller, en op welk tijdstip en op welke
        kilometer haalt de eerste de tweede in?
  \item Zij $f(x) = 2x + 3$ en $g(x) = -0.5x + 7$. Bereken $f(x) + g(x)$ en ga na
        dat het weer affien is. Wat zijn in het algemeen de \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} en
        het snijpunt met de $y$-as van een \omterm{def:g1:addition:def}{som}?
  \item Bespreek de \omterm{def:g8:equations:def}{vergelijking} $mx + p = 0$ volledig: hoeveel oplossingen zijn er
        als $m \neq 0$ (en welke)? Als $m = 0$ en $p \neq 0$? Als $m = 0$ en
        $p = 0$? (Elk geval heeft zijn verhaal op de grafiek: waar snijdt een \omterm{def:g6:lines:objects}{lijn}
        met \omterm{prop:g9:linfunc:affinegraph}{richtingscoëfficiënt} $m$ de horizontale as?)
  \item De toets op één \omterm{def:g6:lines:objects}{lijn} liggen: liggen de punten $(1, 5)$, $(3, 9)$,
        $(7, 17)$ op één rechte \omterm{def:g6:lines:objects}{lijn}? En $(1, 5)$, $(3, 9)$, $(6, 16)$? Bereken de
        veranderingssnelheden tussen paren en beslis. Formuleer de les: een
        \emph{constante veranderingssnelheid} is het handschrift van \omterm{def:g9:linfunc:affine}{affiene
        functies} --- gekromde grafieken veranderen dat tempo van punt tot punt, en
        \emph{dat} meten
        is een verhaal voor het bovenbouwvolume.
\end{enumerate}
\end{problem}
