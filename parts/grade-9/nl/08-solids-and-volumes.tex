\chapter{Lichamen en inhouden}\label{ch:g9:solids}

De ruimtemeetkunde begint met een kleine familie van lichamen --- \omterm{def:g7:areas:prism}{prisma}'s,
\omterm{def:g7:areas:prism}{cilinders}, \omterm{def:g8:solids:def}{piramides}, \omterm{def:g8:solids:def}{kegels}, bollen --- en met twee vragen: hoeveel bevatten ze
(\omterm{def:g2:measure:units}{inhoud}), en wat zie je wanneer je ze doorsnijdt (\omterm{prop:g9:solids:sections}{doorsneden})? Het hoofdstuk sluit
af met een feit van groot praktisch belang: een \omterm{def:g5:solids:def}{lichaam} met $k$ schalen
vermenigvuldigt zijn \omterm{def:g2:measure:units}{inhoud} met $k^3$.

\section{De klassieke lichamen en hun inhoud}

\begin{theorem}[Formules voor de inhoud]\label{thm:g9:solids:volumes}
Schrijf $B$ voor de \omterm{def:g6:measure:area}{oppervlakte} van het grondvlak en $h$ voor de hoogte (de afstand
tussen het grondvlak en het overstaande \omterm{def:g5:solids:def}{zijvlak} of de top).
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{l|l}
\omterm{def:g5:solids:def}{lichaam} & \omterm{def:g2:measure:units}{inhoud} \\
\hline
\omterm{def:g7:areas:prism}{prisma}, \omterm{def:g7:areas:prism}{cilinder} & $V = B \times h$ \\[5pt]
\omterm{def:g8:solids:def}{piramide}, \omterm{def:g8:solids:def}{kegel} & $V = \dfrac13\, B \times h$ \\[5pt]
bol met \omterm{def:g3:shapes:shapes}{radius} $r$ & $V = \dfrac43\,\pi r^3$ \\
\end{tabular}
\end{center}
Voor een \omterm{def:g7:areas:prism}{cilinder} en een \omterm{def:g8:solids:def}{kegel} met \omterm{def:g3:shapes:shapes}{radius} $r$ is de \omterm{def:g6:measure:area}{oppervlakte} van het grondvlak
$B = \pi r^2$; de \omterm{def:g6:measure:area}{oppervlakte} van het boloppervlak is $4\pi r^2$.
\end{theorem}
\admitted

\begin{example}\label{ex:g9:solids:volumes}
Een cilindrisch blikje heeft \omterm{def:g3:shapes:shapes}{radius} $4$ cm en hoogte $10$ cm:
\[
V = \pi r^2 h = \pi \times 16 \times 10 = 160\pi \approx 503 \text{ cm}^3
\approx 0.5 \text{ L}.
\]
Een \omterm{def:g8:solids:def}{kegel} met hetzelfde grondvlak en dezelfde hoogte bevat daarvan een derde:
$\frac{160\pi}{3} \approx 168$ cm$^3$. Een bol met \omterm{def:g3:shapes:shapes}{radius} $4$ cm:
$V = \frac43 \pi \times 64 = \frac{256\pi}{3} \approx 268$ cm$^3$.
\end{example}

\begin{example}[Een piramide stap voor stap]\label{ex:g9:solids:pyramid}
Een \omterm{def:g8:solids:def}{piramide} heeft een vierkant grondvlak met zijde $6$ m en hoogte $5$ m.
\begin{enumerate}
  \item \omterm{def:g6:measure:area}{Oppervlakte} van het grondvlak: $B = 6^2 = 36$ m$^2$.
  \item \omterm{def:g2:measure:units}{Inhoud}: $V = \frac13 \times 36 \times 5 = 60$ m$^3$.
\end{enumerate}
Let op de eenheden: was de zijde in cm en de hoogte in m gegeven, dan zou eerst een
van beide omgezet moeten worden.
\end{example}

\section{Doorsneden met vlakken}

\begin{proposition}[Doorsneden van de klassieke lichamen]\label{prop:g9:solids:sections}
Een \omterm{def:g5:solids:def}{lichaam} met een vlak doorsnijden geeft een vlakke figuur, de
\emph{doorsnede}\index{doorsnede}:
\begin{itemize}
  \item een \omterm{def:g7:areas:prism}{prisma} of \omterm{def:g7:areas:prism}{cilinder}, \emph{\omterm{def:g6:lines:perp}{parallel} met zijn grondvlak} \omterm{prop:g9:solids:sections}{doorsneden},
        geeft op elke hoogte een kopie van het grondvlak;
  \item een \omterm{def:g8:solids:def}{piramide} of \omterm{def:g8:solids:def}{kegel}, \omterm{def:g6:lines:perp}{parallel} met zijn grondvlak \omterm{prop:g9:solids:sections}{doorsneden}, geeft een
        \emph{verkleining} van het grondvlak: op afstand $d$ van de top is de
        \omterm{def:g9:thales:scaling}{schaalfactor} $k = \frac{d}{h}$;
  \item een bol met \omterm{def:g3:shapes:shapes}{radius} $r$, \omterm{prop:g9:solids:sections}{doorsneden} met een vlak op afstand $d < r$ van het
        middelpunt, geeft een cirkel met \omterm{def:g3:shapes:shapes}{radius} $\sqrt{r^2 - d^2}$ (volgens de
        stelling van Pythagoras).
\end{itemize}
\end{proposition}
\admitted

\begin{example}\label{ex:g9:solids:section}
Een \omterm{def:g8:solids:def}{kegel} heeft grondradius $6$ en hoogte $9$. De \omterm{prop:g9:solids:sections}{doorsnede} met een vlak \omterm{def:g6:lines:perp}{parallel}
met het grondvlak, op afstand $3$ van de top, is een schijf geschaald met
$k = \frac39 = \frac13$: haar \omterm{def:g3:shapes:shapes}{radius} is $6 \times \frac13 = 2$.

Een bol met \omterm{def:g3:shapes:shapes}{radius} $5$ wordt \omterm{prop:g9:solids:sections}{doorsneden} met een vlak op afstand $3$ van haar
middelpunt: de \omterm{prop:g9:solids:sections}{doorsnede} is een cirkel met \omterm{def:g3:shapes:shapes}{radius} $\sqrt{25 - 9} = 4$.
\end{example}

\section{Lichamen schalen}

\begin{theorem}[Gevolg van een schaling voor lengten, oppervlakten en inhouden]\label{thm:g9:solids:scaling}
Wordt een \omterm{def:g5:solids:def}{lichaam} vergroot of verkleind met de \omterm{def:g9:thales:scaling}{schaalfactor} $k > 0$, dan:
\begin{itemize}
  \item worden alle lengten met $k$ vermenigvuldigd;
  \item worden alle \omterm{def:g6:measure:area}{oppervlakten} met $k^2$ vermenigvuldigd;
  \item worden alle \omterm{def:g2:measure:units}{inhouden} met $k^3$ vermenigvuldigd.
\end{itemize}
\end{theorem}
\admitted

\begin{example}\label{ex:g9:solids:scaling}
Een modelauto op schaal $\frac{1}{18}$: de lengten zijn die van de echte auto
vermenigvuldigd met $k = \frac{1}{18}$, het geverfde oppervlak vermenigvuldigd met
$k^2 = \frac{1}{324}$, en de \omterm{def:g2:measure:units}{inhoud} met $k^3 = \frac{1}{5832}$.

De \omterm{def:g3:shapes:shapes}{radius} van een bol verdubbelen ($k = 2$) vermenigvuldigt haar \omterm{def:g2:measure:units}{inhoud} met
$2^3 = 8$: ga het na op de formule,
$\frac43\pi (2r)^3 = \frac43 \pi \times 8r^3$.
\end{example}

\begin{example}[Afgeknotte kegel]\label{ex:g9:solids:truncated}
Een \omterm{def:g8:solids:def}{kegel} met hoogte $9$ en grondradius $6$ (\omterm{def:g2:measure:units}{inhoud}
$\frac13 \pi \times 36 \times 9 = 108\pi$) wordt op afstand $3$ van de top
\omterm{prop:g9:solids:sections}{doorsneden}, en de kleine \omterm{def:g8:solids:def}{kegel} boven de \omterm{prop:g9:solids:sections}{doorsnede} wordt weggehaald. De kleine \omterm{def:g8:solids:def}{kegel}
is de verkleining met $k = \frac13$, dus is zijn \omterm{def:g2:measure:units}{inhoud}
$108\pi \times \left(\frac13\right)^3 = 4\pi$. Het \omterm{def:g5:solids:def}{lichaam} dat overblijft (een
\emph{afgeknotte \omterm{def:g8:solids:def}{kegel}}) heeft \omterm{def:g2:measure:units}{inhoud} $108\pi - 4\pi = 104\pi \approx 327$.
\end{example}

\end{enumerate}
\end{exercise}

\section{Opgave: de grafsteen van Archimedes}

\begin{problem}[{Weekendopgave --- de bol is twee derde van haar cilinder (twee
keer), de stapel 1:2:3, en de kwadraat-kubuswet die reuzen verbiedt}]
\label{pb:g9:solids:1}
Archimedes bewees vele stellingen, maar op één was hij zo trots dat hij vroeg om
haar \emph{figuur} in zijn grafsteen te beitelen: een bol genesteld in haar nauwste
\omterm{def:g7:areas:prism}{cilinder}. Een eeuw later herkende de Romeinse schrijver Cicero, zoekend in de
braamstruiken bij Syracuse, het graf ``aan de bol en de \omterm{def:g7:areas:prism}{cilinder}''. Deze opgave
haalt terug wat dat beeldhouwwerk in zich draagt --- een dubbel wonder van twee
derde --- en
volgt daarna \omterm{def:g2:measure:units}{inhouden} en \omterm{def:g6:measure:area}{oppervlakten} (\cref{thm:g9:solids:volumes},
\cref{thm:g9:solids:scaling}) naar een wet die reuzen, mieren en afkoelende planeten
regeert.

\medskip
\noindent\textbf{Deel I --- Het beeldhouwwerk ontcijferd.}
Een bol met \omterm{def:g3:shapes:shapes}{radius} $r$ zit precies in een \omterm{def:g7:areas:prism}{cilinder}: dezelfde \omterm{def:g3:shapes:shapes}{radius}, hoogte $2r$.
\begin{enumerate}
  \item Bereken de \omterm{def:g2:measure:units}{inhoud} van de \omterm{def:g7:areas:prism}{cilinder} in $r$.
  \item Bereken de verhouding van de \omterm{def:g2:measure:units}{inhoud} van de bol tot die van de \omterm{def:g7:areas:prism}{cilinder}.
        Waarom zou Archimedes het mooi gevonden hebben dat het antwoord geen $\pi$
        en geen $r$ bevat?
  \item Nu de \omterm{def:g6:measure:area}{oppervlakten}: vergelijk de \omterm{def:g6:measure:area}{oppervlakte} van het boloppervlak met de
        \emph{totale} \omterm{def:g6:measure:area}{oppervlakte} van de \omterm{def:g7:areas:prism}{cilinder} (het \omterm{def:g5:solids:def}{zijvlak} plus de twee
        deksels). Welke verhouding duikt op --- opnieuw?
  \item De lege ruimte tussen bol en \omterm{def:g7:areas:prism}{cilinder} heeft \omterm{def:g2:measure:units}{inhoud}
        $2\pi r^3 - \frac43\pi r^3$. Toon aan dat wat overblijft precies gelijk is
        aan de \omterm{def:g2:measure:units}{inhoud} van \emph{twee \omterm{def:g8:solids:def}{kegels}} met \omterm{def:g3:shapes:shapes}{radius} $r$ en hoogte $r$.
  \item Een basketbal met \omterm{def:g3:shapes:shapes}{radius} $12$ cm wordt in de nauwste cilindrische doos
        verkocht. Bereken beide \omterm{def:g2:measure:units}{inhouden} (op $100$ cm$^3$ nauwkeurig) en het
        percentage van de doos dat leeg is.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Schalen, vormen en manen.}
\begin{enumerate}[resume]
  \item Een halfbolvormige schaal heeft \omterm{def:g3:shapes:shapes}{radius} $10$ cm. Bereken haar \omterm{def:g2:measure:units}{inhoud}, in
        cm$^3$ en in liter (op de deciliter).
  \item De stapel 1:2:3: een \omterm{def:g8:solids:def}{kegel}, een halve bol en een \omterm{def:g7:areas:prism}{cilinder}, alle met \omterm{def:g3:shapes:shapes}{radius}
        $10$ cm en hoogte $10$ cm. Bereken de drie \omterm{def:g2:measure:units}{inhouden} en ga de legendarische
        verhouding $1 : 2 : 3$ na. (Ook deze zou Archimedes gesmaakt hebben.)
  \item Een chocoladebol met \omterm{def:g3:shapes:shapes}{radius} $3$ cm wordt gesmolten in een cilindrische vorm
        met \omterm{def:g3:shapes:shapes}{radius} $3$ cm. Welke hoogte bereikt de chocolade? (Exacte \omterm{def:g9:fractions:fraction}{breuk}, en
        daarna op de millimeter.)
  \item De \omterm{def:g3:shapes:shapes}{radius} van de Aarde is ongeveer $6\,371$ km, die van de Maan ongeveer
        $1\,737$ km. Bereken de verhouding van de radiussen, en daarna --- met
        \cref{thm:g9:solids:scaling} --- de verhouding van de \omterm{def:g2:measure:units}{inhouden}: hoeveel
        Manen zouden er, naar \omterm{def:g2:measure:units}{inhoud}, in een holle Aarde passen?
  \item Uit dezelfde stelling: wat is de verhouding van de \emph{\omterm{def:g6:measure:area}{oppervlakten}} van
        Aarde en Maan? En in het algemeen: wat gebeurt er met de \omterm{def:g2:measure:units}{inhoud} en met het
        oppervlak van een ballon wanneer zijn \omterm{def:g3:shapes:shapes}{radius} verdubbelt? Formuleer de
        \emph{kwadraat-kubuswet}: schaalt een vorm op, dan lopen de \omterm{def:g2:measure:units}{inhouden} de
        \omterm{def:g6:measure:area}{oppervlakten} voorbij.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De kwadraat-kubuswet regeert de wereld.}
\begin{enumerate}[resume]
  \item Het argument van Galilei tegen reuzen: stel je een mens voor die met een
        factor $10$ is opgeschaald, met dezelfde verhoudingen. Met welke factor
        groeit het gewicht (gewicht volgt de \omterm{def:g2:measure:units}{inhoud})? Met welke factor groeit de
        \omterm{prop:g9:solids:sections}{doorsnede} van de beenderen (een \omterm{def:g6:measure:area}{oppervlakte})? Met welke factor groeit dan de
        \emph{druk} op elke vierkante centimeter been --- en wat gebeurt er met de
        reus?
  \item Dezelfde wet in omgekeerde richting verklaart het heldendom van mieren:
        kracht volgt de \omterm{prop:g9:solids:sections}{doorsnede} van de spieren (een \omterm{def:g6:measure:area}{oppervlakte}), gewicht volgt de
        \omterm{def:g2:measure:units}{inhoud}. Met welke factoren krimpen gewicht en kracht voor een dier dat in
        elke richting $100$ keer kleiner is, en met welke factor verbetert de
        verhouding van kracht tot gewicht?
  \item Warmte wordt geproduceerd door de \omterm{def:g2:measure:units}{inhoud} van het \omterm{def:g5:solids:def}{lichaam} en verloren via

        twee
        keer ``twee derde'' van vragen 2 en 3 --- en antwoord Cicero in één zin:
