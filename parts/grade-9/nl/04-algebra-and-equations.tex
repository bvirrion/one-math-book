\chapter{Algebra en vergelijkingen}\label{ch:g9:algebra}

Letters staan voor getallen, en rekenen met letters bewijst feiten over
\emph{alle} getallen tegelijk. Dit hoofdstuk oefent het \omterm{def:g9:algebra:expand}{uitwerken} en het ontbinden
in factoren --- met de drie merkwaardige \omterm{def:g2:mult:def}{producten} die je door de hele wiskunde
zullen volgen --- en lost \omterm{def:g8:equations:def}{vergelijkingen} van de eerste graad,
productvergelijkingen en eenvoudige ongelijkheden op, telkens één verantwoorde
stap per keer.

\section{Uitwerken}

\begin{definition}[Uitwerken]\label{def:g9:algebra:expand}
\emph{Uitwerken}\index{uitwerken} betekent een \omterm{def:g2:mult:def}{product} omzetten in een \omterm{def:g1:addition:def}{som}, met
behulp van de distributiviteit:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g9:algebra:expand}
Elke term van het eerste haakje vermenigvuldigt elke term van het tweede:
\begin{align*}
(x + 3)(2x + 5)
&= x \times 2x + x \times 5 + 3 \times 2x + 3 \times 5 \\
&= 2x^2 + 5x + 6x + 15 \\
&= 2x^2 + 11x + 15 .
\end{align*}
De tekens reizen met de termen mee:
\begin{align*}
(3x - 2)(x - 4)
&= 3x^2 - 12x - 2x + 8
= 3x^2 - 14x + 8 .
\end{align*}
\end{example}

\begin{theorem}[De drie merkwaardige producten]\label{thm:g9:algebra:identities}
Voor alle getallen $a$ en $b$:
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
Werk elk linkerlid uit. Voor het eerste:
$(a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2$. Voor het tweede: vervang in
het eerste $b$ door $-b$. Voor het derde:
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$ (de kruistermen vallen weg).
\end{proof}

\begin{example}\label{ex:g9:algebra:identities}
\[
(x + 5)^2 = x^2 + 10x + 25, \qquad
(3x - 1)^2 = 9x^2 - 6x + 1, \qquad
(2x + 7)(2x - 7) = 4x^2 - 49 .
\]
Hoofdrekenen: $102^2 = (100 + 2)^2 = 10000 + 400 + 4 = 10404$, en
$98 \times 102 = (100-2)(100+2) = 10000 - 4 = 9996$.
\end{example}

\section{Ontbinden in factoren}

\begin{method}[Ontbinden in factoren]\label{met:g9:algebra:factor}
Om een uitdrukking te ontbinden, probeer je op deze volgorde:
\begin{enumerate}
  \item \emph{gemeenschappelijke factor}: $ka + kb = k(a + b)$, waarbij $k$ zelf
        een haakje mag zijn;
  \item \emph{\omterm{ex:g1:subtraction:difference}{verschil} van kwadraten}: $a^2 - b^2 = (a+b)(a-b)$;
  \item \emph{volkomen kwadraat}: $a^2 + 2ab + b^2 = (a+b)^2$ (en met $-2ab$:
        $(a-b)^2$).
\end{enumerate}
Ga het resultaat altijd na door het uit te werken.
\end{method}

\begin{example}\label{ex:g9:algebra:factor}
\emph{Gemeenschappelijke factor:}
$15x^2 + 10x = 5x \times 3x + 5x \times 2 = 5x(3x + 2)$.

\emph{Gemeenschappelijk haakje:}
\[
(x+2)(x-1) + (x+2)(3x+4) = (x+2)\bigl[(x-1) + (3x+4)\bigr]
= (x+2)(4x+3).
\]

\emph{\omterm{ex:g1:subtraction:difference}{Verschil} van kwadraten:} $x^2 - 81 = (x+9)(x-9)$, en
\[
(2x+1)^2 - 9 = (2x+1)^2 - 3^2 = (2x+1+3)(2x+1-3) = (2x+4)(2x-2).
\]
Elke factor heeft nog een gemeenschappelijke factor:
$(2x+4)(2x-2) = 2(x+2) \times 2(x-1) = 4(x+2)(x-1)$.
\end{example}

\section{Vergelijkingen}

\begin{method}[Vergelijkingen van de eerste graad]\label{met:g9:algebra:firstdegree}
Om een \omterm{def:g8:equations:def}{vergelijking} als $5x - 3 = 2x + 9$ op te lossen:
\begin{enumerate}
  \item breng de $x$-termen in het ene lid en de getallen in het andere, door bij
        beide leden dezelfde grootheid op te tellen;
  \item herleid elk lid;
  \item deel beide leden door de coëfficiënt van $x$;
  \item ga de oplossing na in de oorspronkelijke \omterm{def:g8:equations:def}{vergelijking}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:algebra:firstdegree}
\begin{align*}
5x - 3 &= 2x + 9 \\
5x - 2x &= 9 + 3 && \text{(tel $3 - 2x$ bij beide leden op)}\\
3x &= 12 \\
x &= 4 && \text{(deel door 3).}
\end{align*}
Controle: $5 \times 4 - 3 = 17$ en $2 \times 4 + 9 = 17$. De oplossing is $4$.

Met haakjes werk je eerst uit: $3(x - 2) = x + 8$ wordt $3x - 6 = x + 8$, dus
$2x = 14$ en $x = 7$.
\end{example}

\begin{theorem}[Nulproductregel]\label{thm:g9:algebra:zeroproduct}
Een \omterm{def:g2:mult:def}{product} is \omterm{def:g1:counting:zero}{nul} als en slechts als minstens één factor \omterm{def:g1:counting:zero}{nul} is:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ of } \ B = 0 .
\]
\end{theorem}

\begin{proof}
Is een factor \omterm{def:g1:counting:zero}{nul}, dan is het \omterm{def:g2:mult:def}{product} dat duidelijk ook. Omgekeerd: is $A B = 0$
met $A \neq 0$, dan geeft beide leden door $A$ delen $B = 0$.
\end{proof}

\begin{example}\label{ex:g9:algebra:zeroproduct}
Los $(x - 3)(2x + 10) = 0$ op: ofwel $x - 3 = 0$, ofwel $2x + 10 = 0$, dus zijn de
oplossingen $3$ en $-5$.

Los $x^2 - 49 = 0$ op: ontbind eerst, $(x-7)(x+7) = 0$, met oplossingen $7$ en
$-7$.

Los $x^2 = 3x$ op: deel nooit door $x$! Breng over en ontbind:
$x^2 - 3x = x(x - 3) = 0$, met oplossingen $0$ en $3$.
\end{example}

\section{Ongelijkheden}

\begin{proposition}[Regels voor ongelijkheden]\label{prop:g9:algebra:ineq}
Een ongelijkheid blijft bewaard wanneer je bij beide leden hetzelfde getal
optelt, en wanneer je beide leden met hetzelfde \emph{positieve} getal
vermenigvuldigt. Beide leden met een \emph{negatief} getal vermenigvuldigen keert
het ongelijkheidsteken om.
\end{proposition}

\begin{proof}
Is $a \leq b$, dan is $b - a \geq 0$. Tel $c$ op:
$(b+c) - (a+c) = b - a \geq 0$, dus is $a + c \leq b + c$. Vermenigvuldig met
$c > 0$: $bc - ac = (b-a)c \geq 0$, dus is $ac \leq bc$. Vermenigvuldig met
$c < 0$: $(b-a)c \leq 0$, dus is $ac \geq bc$: de orde is omgekeerd.
\end{proof}

\begin{example}\label{ex:g9:algebra:ineq}
Los $7 - 2x < 15$ op:
\begin{align*}
7 - 2x &< 15 \\
-2x &< 8 && \text{(trek 7 af)} \\
x &> -4 && \text{(deel door $-2$: keer het teken om).}
\end{align*}
De oplossingen zijn alle getallen groter dan $-4$: op een getallenlijn een open
bolletje bij $-4$ en arcering naar rechts.
\end{example}

\section{Opgave: de algebra van de cijfers}

\begin{problem}[{Weekendopgave --- waarom de deelbaarheidsregels werken, de truc
met 1089 ontmaskerd, en hoofdrekenen dat op toverkunst lijkt}]
\label{pb:g9:algebra:1}
Op de basisschool leerde je \emph{regels} (een getal is \omterm{def:g6:wholes:divisible}{deelbaar} door $9$ wanneer
zijn cijfersom dat is); goochelaars voeren \emph{trucs} op (iedereen eindigt op
$1089$); op de markt kennen ze \emph{kortere wegen} ($35^2 = 1225$, in een
oogwenk). Het is allemaal hetzelfde geheim: een getal met cijfers $a$, $b$, $c$ is
de uitdrukking $100a + 10b + c$, en de algebra van dit hoofdstuk
(\cref{thm:g9:algebra:identities}, \omterm{def:g9:algebra:expand}{uitwerken}, ontbinden) kan die uit elkaar halen.
Aan het einde van deze opgave heb je de regels bewezen, de truc ontmaskerd en de
toverkunst geleerd.

\medskip
\noindent\textbf{Deel I --- Cijfers onder de microscoop.}
\begin{enumerate}
  \item Een getal van twee cijfers met tiental $a$ en eenheid $b$ is $10a + b$.
        Werk het \omterm{ex:g1:subtraction:difference}{verschil} tussen het getal en zijn omkering,
        $(10a + b) - (10b + a)$, uit en ontbind het, en formuleer de ontdekking. Ga
        het na op $72 - 27$.
  \item Bewijs de deelbaarheidsregel voor $9$ voor getallen van drie cijfers: ga
        de gelijkheid
        \[
        100a + 10b + c = 9\,(11a + b) + (a + b + c)
        \]
        na, en leg uit waarom die laat zien dat een getal en zijn cijfersom
        \emph{dezelfde \omterm{def:g3:division:remainder}{rest}} laten bij de deling door $9$ --- zodat het ene precies
        dan \omterm{def:g6:wholes:divisible}{deelbaar} is door $9$ als het andere dat is
        (\cref{prop:g9:arith:rules}, nu bewezen).
  \item Leid de regel voor $3$ in één zin uit dezelfde gelijkheid af.
  \item De negenproef, de controle van de boekhouders: om
        $47 \times 86 = 4\,042$ te toetsen, vervang je elk getal door de cijfersom
        van zijn cijfersom: $47 \to 11 \to 2$, $86 \to 14 \to 5$, en
        $2 \times 5 = 10 \to 1$; en dan $4\,042 \to 10 \to 1$: samenhangend. Leg
        uit waarom een juist \omterm{def:g2:mult:def}{product} de proef moet doorstaan (schrijf
        $47 = 9k + 2$, $86 = 9l + 5$ en werk uit), en zoek een verkeerd resultaat
        dat de proef niet opmerkt (waarom zijn verwisselde cijfers voor haar
        onzichtbaar?).
  \item Bewijs de deelbaarheidsregel voor $11$ voor getallen van vier cijfers: ga
        \[
        1000a + 100b + 10c + d
        = 11\,(91a + 9b + c) + (- a + b - c + d)
        \]
        na, en gebruik die om te beslissen of $2\,926$ \omterm{def:g6:wholes:divisible}{deelbaar} is door $11$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De truc met 1089.}
De truc: denk aan een getal van drie cijfers waarvan het eerste en het laatste
cijfer minstens $2$ verschillen; keer het om; trek het kleinste van het grootste
af; keer het resultaat om; tel de laatste twee getallen op. De goochelaar kondigt
aan: \emph{1089}.
\begin{enumerate}[resume]
  \item Voer de truc uit op $742$ en op een getal van je keuze.
  \item Zij het getal met cijfers $a$, $b$, $c$ en $a > c$. Ontbind de eerste
        aftrekking, $(100a + 10b + c) - (100c + 10b + a)$, en noem alle waarden die
        ze kan aannemen wanneer $a - c$ van $2$ tot $9$ loopt.
  \item Elk van die \omterm{def:g5:division:multiple}{veelvouden} van $99$ heeft een opvallend cijferpatroon. Ga, met
        $m = a - c$, de gelijkheid
        \[
        99m = 100\,(m - 1) + 90 + (10 - m)
        \]
        na, en lees de drie cijfers van $99m$ af.
  \item Tel $99m$ bij zijn eigen omkering op, met de cijfers van vraag 8, en bewijs
        dat het totaal $1089$ is --- wat $m$ ook is. De truc is dood; lang leve de
        algebra.
  \item Waarom eiste de goochelaar dat het eerste en het laatste cijfer minstens
        $2$ verschillen? Bekijk de gevallen $a = c$ en $a - c = 1$ (waar de
        aftrekking $99$ geeft, dat als $099$ geschreven moet worden om de omkering
        te laten werken), en formuleer de kleine lettertjes die de truc eerlijk
        houden.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Toverkunst voor dagelijks gebruik.}
\begin{enumerate}[resume]
  \item De verjaardagstruc: ``vermenigvuldig je geboortemaand met $5$, tel er $7$
        bij op, verdubbel, tel je geboortedag erbij op, trek er $14$ van af.'' Toon
        aan dat het resultaat altijd $10m + d$ is --- maand en dag, in één oogopslag
        leesbaar. (Probeer het op iemand uit.)
  \item Je herontwerpt de truc als: ``vermenigvuldig de maand met $5$, tel er $9$
        bij op, verdubbel, tel de dag erbij op.'' Welk getal moet de kijker nu aan
        het einde aftrekken om dezelfde uitkomst met maand en dag te lezen?
  \item De truc van het ontbrekende cijfer: de kijker kiest een willekeurig getal,
        trekt er zijn cijfersom van af (vraag 2 zegt dat het resultaat altijd een
        \omterm{def:g9:arith:divisor}{veelvoud} van $9$ is --- leg dat uit voor een getal van drie cijfers), streept
        daarna één cijfer \emph{verschillend van \omterm{def:g1:counting:zero}{nul}} door en leest je de
        overblijvende cijfers voor. Leg uit hoe je het doorgestreepte cijfer
        terugvindt, en waarom $0$ uitgesloten moest worden.
  \item Toverkunst van de markt: bewijs de gelijkheid
        $(10t + 5)^2 = 100\,t(t+1) + 25$, formuleer de regel om elk getal dat op
        $5$ eindigt te kwadrateren, en bereken $45^2$, $85^2$ en $105^2$ uit het
        hoofd.
  \item Slot, in het verlengde van \cref{exo:g9:algebra:10}: bewijs dat de \omterm{def:g1:addition:def}{som} van
        vijf opeenvolgende gehele getallen altijd een \omterm{def:g9:arith:divisor}{veelvoud} van $5$ is, maar dat
        de \omterm{def:g1:addition:def}{som} van \emph{vier} opeenvolgende gehele getallen \emph{nooit} een
        \omterm{def:g9:arith:divisor}{veelvoud} van $4$ is. (Twee korte berekeningen --- en één les over wat de
        algebra met de woorden ``altijd'' en ``nooit'' doet.)
\end{enumerate}
\end{problem}
