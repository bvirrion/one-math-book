\chapter{Trigonometria no triângulo retângulo}\label{ch:g9:trig}

Num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, assim que se conhece um ângulo agudo, a
\emph{forma} do triângulo fica fixa --- só o tamanho dele pode variar.
As razões entre os lados dependem, portanto, apenas do ângulo: essas
razões são o \omterm{def:g9:trig:ratios}{cosseno}, o \omterm{def:g9:trig:ratios}{seno} e a \omterm{def:g9:trig:ratios}{tangente}. Junto com o teorema de
Pitágoras, elas permitem calcular todos os lados e todos os ângulos de
um \omterm{def:g6:shapes:triangles}{triângulo retângulo} a partir de pouquíssima informação.

\section{O teorema de Pitágoras, de novo}

\begin{theorem}[Pitágoras]\label{thm:g9:trig:pythagoras}
Num triângulo $ABC$ retângulo em $A$, o quadrado da hipotenusa é a \omterm{def:g1:addition:def}{soma}
dos quadrados dos dois catetos:
\[
BC^2 = AB^2 + AC^2 .
\]
Reciprocamente, se $BC^2 = AB^2 + AC^2$ num triângulo, então o
triângulo é retângulo em $A$.
\end{theorem}
\admitted

\begin{example}\label{ex:g9:trig:pythagoras}
Um \omterm{def:g6:shapes:triangles}{triângulo retângulo} tem catetos $AB = 5$ e $AC = 12$. Então
\[
BC^2 = 5^2 + 12^2 = 25 + 144 = 169,
\qquad\text{então}\quad BC = \sqrt{169} = 13 .
\]
Para achar um \emph{cateto}: se $BC = 10$ e $AB = 6$, então
$AC^2 = BC^2 - AB^2 = 100 - 36 = 64$, então $AC = 8$.
\end{example}

\section{Cosseno, seno, tangente}

Num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, em relação a um ângulo agudo $\theta$, os três
lados têm nomes: a \emph{hipotenusa} (oposta ao \omterm{def:g3:shapes:rightangle}{ângulo reto}, o lado
mais longo), o cateto \emph{oposto} a $\theta$ e o cateto
\emph{adjacente} a $\theta$ (o cateto que toca $\theta$).

\begin{definition}[Razões trigonométricas]\label{def:g9:trig:ratios}
Para um ângulo agudo $\theta$ de um \omterm{def:g6:shapes:triangles}{triângulo retângulo}:
\[
\cos\theta = \frac{\text{adjacente}}{\text{hipotenusa}},
\qquad
\sin\theta = \frac{\text{oposto}}{\text{hipotenusa}},
\qquad
\tan\theta = \frac{\text{oposto}}{\text{adjacente}} .
\]
Essas razões\index{cosseno}\index{seno}\index{tangente} dependem apenas
de $\theta$, e não do tamanho do triângulo (todos os \omterm{def:g6:shapes:triangles}{triângulos
retângulos} com o mesmo ângulo agudo são mudanças de escala uns dos
outros, por Tales).
\end{definition}

\begin{proposition}[Primeiras propriedades]\label{prop:g9:trig:properties}
Para todo ângulo agudo $\theta$:
\begin{enumerate}
  \item $0 < \cos\theta < 1$ e $0 < \sin\theta < 1$ (um cateto é mais
        curto que a hipotenusa);
  \item $\cos^2\theta + \sin^2\theta = 1$;
  \item $\tan\theta = \dfrac{\sin\theta}{\cos\theta}$.
\end{enumerate}
\end{proposition}

\begin{proof}
Escreva $a$ para o cateto adjacente, $o$ para o cateto oposto e $h$
para a hipotenusa de um \omterm{def:g6:shapes:triangles}{triângulo retângulo} com ângulo $\theta$.

1. $0 < a < h$ e $0 < o < h$, então as duas razões ficam estritamente
entre $0$ e $1$.

2. Por Pitágoras, $a^2 + o^2 = h^2$. Divida tudo por $h^2$:
\[
\cos^2\theta + \sin^2\theta
= \frac{a^2}{h^2} + \frac{o^2}{h^2}
= \frac{a^2 + o^2}{h^2} = 1 .
\]

3. $\dfrac{\sin\theta}{\cos\theta} = \dfrac{o/h}{a/h} = \dfrac oa
= \tan\theta$.
\end{proof}

\begin{method}[Achar um lado]\label{met:g9:trig:side}
Para calcular um lado desconhecido quando se conhecem um lado e um
ângulo agudo:
\begin{enumerate}
  \item marque o ângulo conhecido e nomeie os três lados (hipotenusa,
        oposto, adjacente) \emph{a partir desse ângulo};
  \item escolha a razão (\omterm{def:g9:trig:ratios}{cosseno}, \omterm{def:g9:trig:ratios}{seno} ou \omterm{def:g9:trig:ratios}{tangente}) que envolve o lado
        conhecido e o lado procurado;
  \item escreva a \omterm{def:g8:equations:def}{equação} e resolva-a;
  \item confira o bom senso: a hipotenusa tem de sair como o lado mais
        longo.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:trig:side}
Num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, a hipotenusa mede $8$ e um ângulo é
$35^\circ$; calcule o cateto oposto a ele. A razão que envolve o cateto
oposto e a hipotenusa é o \omterm{def:g9:trig:ratios}{seno}:
\[
\sin 35^\circ = \frac{\text{oposto}}{8}
\quad\Longrightarrow\quad
\text{oposto} = 8 \sin 35^\circ \approx 8 \times 0.574 \approx 4.6 .
\]
Para o cateto adjacente, use o \omterm{def:g9:trig:ratios}{cosseno}:
$8\cos 35^\circ \approx 8 \times 0.819 \approx 6.6$. Confira com
Pitágoras: $4.6^2 + 6.6^2 \approx 21 + 43 \approx 64 = 8^2$.
\end{example}

\begin{method}[Achar um ângulo]\label{met:g9:trig:angle}
Quando se conhecem dois lados, calcule a razão que eles formam,
identifique-a como \omterm{def:g9:trig:ratios}{cosseno}, \omterm{def:g9:trig:ratios}{seno} ou \omterm{def:g9:trig:ratios}{tangente} do ângulo desconhecido e
aplique a função inversa da calculadora ($\cos^{-1}$, $\sin^{-1}$ ou
$\tan^{-1}$) a essa razão.
\end{method}

\begin{example}\label{ex:g9:trig:angle}
Uma escada de $5$ m de comprimento se apoia numa parede, com o pé a
$1.4$ m da parede. O ângulo $\theta$ entre a escada e o chão satisfaz
\[
\cos\theta = \frac{1.4}{5} = 0.28,
\qquad\text{então}\quad
\theta = \cos^{-1}(0.28) \approx 73.7^\circ .
\]
\end{example}

\begin{example}[Ângulos especiais]\label{ex:g9:trig:special}
Dois triângulos dão valores exatos. A \omterm{def:g3:division:half}{metade} de um quadrado de lado 1
(um \omterm{def:g6:shapes:triangles}{triângulo retângulo} \omterm{def:g6:shapes:triangles}{isósceles}) tem hipotenusa $\sqrt2$ e ângulos de
$45^\circ$:
\[
\cos 45^\circ = \sin 45^\circ = \frac{1}{\sqrt2} = \frac{\sqrt2}{2},
\qquad \tan 45^\circ = 1 .
\]
A \omterm{def:g3:division:half}{metade} de um \omterm{def:g6:shapes:triangles}{triângulo equilátero} de lado 1 dá os ângulos $30^\circ$
e $60^\circ$:
\[
\cos 60^\circ = \sin 30^\circ = \frac12,
\qquad
\cos 30^\circ = \sin 60^\circ = \frac{\sqrt3}{2}.
\]
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g9:trig:1}
Um \omterm{def:g6:shapes:triangles}{triângulo retângulo} tem catetos $9$ e $12$. Calcule a hipotenusa
dele. Outro tem hipotenusa $17$ e um cateto $8$: calcule o outro
cateto.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:2}
Um triângulo tem lados $7$, $24$ e $25$. Ele é retângulo? Mesma
pergunta para os lados $5$, $6$ e $8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:3}
Num triângulo $DEF$ retângulo em $D$, com o ângulo em $E$ chamado
$\theta$: que lado é a hipotenusa? Que lado é o oposto a $\theta$?
Escreva $\cos\theta$, $\sin\theta$ e $\tan\theta$ como razões entre os
lados $DE$, $DF$, $EF$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:4}
Num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, a hipotenusa mede $10$ e um ângulo agudo é
$28^\circ$. Calcule os dois catetos (arredonde ao décimo;
$\cos 28^\circ \approx 0.883$, $\sin 28^\circ \approx 0.469$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:5}
Num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, o cateto adjacente ao ângulo $\theta$ mede $6$
e o cateto oposto mede $4.5$. Calcule $\tan\theta$ e depois $\theta$
(arredonde ao grau; $\tan^{-1}(0.75) \approx 36.9^\circ$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:6}
Um drone voa a uma altitude de $120$ m. De um observador no chão, ele é
visto sob um ângulo de elevação de $32^\circ$. A que distância
horizontal do observador o drone está
($\tan 32^\circ \approx 0.625$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:7}
Uma rampa de acesso precisa subir $1.2$ m com um ângulo de no máximo
$5^\circ$ em relação à horizontal. Que comprimento mínimo, ao longo da
inclinação, a rampa precisa ter ($\sin 5^\circ \approx 0.0872$)?
Arredonde para cima, ao décimo de metro.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:8}
Usando a \cref{prop:g9:trig:properties}: um ângulo agudo $\theta$
satisfaz $\cos\theta = 0.6$. Calcule $\sin\theta$ sem achar $\theta$ e
depois $\tan\theta$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:trig:9}
Justifique os valores exatos do \cref{ex:g9:trig:special}: no \omterm{def:g6:shapes:triangles}{triângulo
retângulo} \omterm{def:g6:shapes:triangles}{isósceles} de catetos $1$, calcule a hipotenusa; no \omterm{def:g6:shapes:triangles}{triângulo
equilátero} de lado $1$, calcule a altura e deduza o \omterm{def:g9:trig:ratios}{cosseno} e o \omterm{def:g9:trig:ratios}{seno} de
$30^\circ$ e de $60^\circ$.
\end{exercise}

\section{Problema: medir o que não se alcança}

\begin{problem}[{Problema de fim de semana --- o método topográfico de
duas estações para alturas inacessíveis, a inclinação das estradas, a
distância até o horizonte e a corrida da tangente ao infinito}]
\label{pb:g9:trig:1}
Trena nenhuma alcança o topo de uma montanha, o cume de uma torre do
outro lado de um rio ou o horizonte no mar --- mas um medidor de
ângulos e as três razões deste capítulo alcançam. A peça central deste
problema é o clássico \emph{método de duas estações} dos topógrafos,
que calcula uma altura sem jamais se aproximar da base dela; em volta
dele: placas de estrada, escadas, degraus, a curvatura da Terra e um
primeiro vislumbre de uma função que explode ao infinito.

\medskip
\noindent\textbf{Parte I --- Uma estação.}
(Valores: $\tan 35^\circ \approx 0.700$,
$\sin 35^\circ \approx 0.574$, $\cos 35^\circ \approx 0.819$,
$\tan 1^\circ \approx 0.0175$.)
\begin{enumerate}
  \item De um ponto a $60$ m do pé de um farol, o topo é visto sob uma
        elevação de $35^\circ$ (medida a partir da horizontal, na
        altura dos olhos). Que altura o topo tem acima da altura dos
        olhos?
  \item Uma placa de estrada anuncia uma inclinação de $12\,\%$: a
        estrada sobe $12$ m a cada $100$ m percorridos na horizontal.
        Que ângulo a estrada faz com a horizontal? E que porcentagem de
        inclinação teria uma estrada de $45^\circ$?
  \item Os dois ângulos agudos de um \omterm{def:g6:shapes:triangles}{triângulo retângulo} são
        complementares, e o cateto oposto a cada um é o cateto
        adjacente ao outro. Deduza as identidades do ``co'':
        \[
        \sin(90^\circ - \theta) = \cos\theta,
        \qquad
        \cos(90^\circ - \theta) = \sin\theta .
        \]
  \item Uma escada de $8$ m se apoia a $60^\circ$ do chão. Usando os
        valores exatos do \cref{ex:g9:trig:special}, dê a altura
        alcançada e a distância do pé até a parede --- de forma exata e
        depois ao centímetro.
  \item Verifique $\cos^2\theta + \sin^2\theta = 1$ numericamente para
        $\theta = 35^\circ$ e de forma exata para $\theta = 60^\circ$.
        Por que essa identidade (\cref{prop:g9:trig:properties}) é um
        bom hábito para conferir o trabalho na calculadora?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Duas estações.}
O cume de uma montanha está visível, e a base dela é inalcançável. De
um ponto $A$, a elevação do cume é $30^\circ$; andando $200$ m em linha
reta em direção à montanha, até $B$, a elevação passa a $40^\circ$.
Escreva $h$ para a altura do cume acima da altura dos olhos e $x$ para
a distância horizontal de $B$ até a vertical do cume.
($\tan 30^\circ \approx 0.577$, $\tan 40^\circ \approx 0.839$.)
\begin{enumerate}[resume]
  \item Expresse $\tan 40^\circ$ e $\tan 30^\circ$ usando $h$, $x$ e
        $x + 200$: duas \omterm{def:g8:equations:def}{equações} para duas incógnitas.
  \item De cada \omterm{def:g8:equations:def}{equação}, expresse a distância horizontal em função de
        $h$, subtraia e obtenha a fórmula das duas estações:
        \[
        h = \frac{200}
        {\ \dfrac{1}{\tan 30^\circ} - \dfrac{1}{\tan 40^\circ}\ } .
        \]
        Calcule $h$ arredondando à dezena de metros.
  \item Calcule também $x$ e confira as suas duas respostas contra a
        visada de $30^\circ$ a partir de $A$.
  \item Em uma frase: por que o método das duas estações é
        indispensável --- que única medida, impossível aqui, o método
        de uma estação da questão 1 teria exigido?
  \item Uma torre é avistada a $30^\circ$ e a $60^\circ$ depois de
        andar $200$ m em direção a ela. Usando
        $\tan 30^\circ = \frac{1}{\sqrt3}$ e
        $\tan 60^\circ = \sqrt3$, calcule a altura \emph{exatamente}
        --- a resposta é um \omterm{def:g9:arith:divisor}{múltiplo} de $\sqrt3$ --- e depois ao metro.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Até o horizonte e parede acima.}
\begin{enumerate}[resume]
  \item A que distância fica o horizonte? De pé, com os olhos a $h$
        metros acima de uma Terra esférica de \omterm{def:g3:shapes:shapes}{raio} $R$, a sua linha de
        visão roça a esfera: a linha de visão, o \omterm{def:g3:shapes:shapes}{raio} até o ponto de
        contato e o \omterm{def:g3:shapes:shapes}{raio} até os seus pés formam um \omterm{def:g6:shapes:triangles}{triângulo retângulo}.
        Mostre, com Pitágoras (\cref{thm:g9:trig:pythagoras}), que a
        distância $d$ até o horizonte satisfaz
        $d^2 = 2Rh + h^2 \approx 2Rh$, e calcule $d$ para olhos a
        $1.80$ m ($R = 6.37 \times 10^6$ m).
  \item Mesma pergunta do alto de uma torre de $324$ m. (Regra prática
        dos marinheiros: o horizonte em quilômetros é cerca de
        $3.6\sqrt{h}$, com $h$ em metros --- confira as suas duas
        respostas com ela.)
  \item O seu polegar, de uns $2$ cm de largura, esticado à distância
        do braço, a uns $60$ cm do olho: que ângulo ele cobre? A Lua
        cheia cobre cerca de $0.5^\circ$: que \omterm{def:g9:fractions:fraction}{fração} do seu polegar
        esconde a Lua inteira?
  \item Escadas confortáveis têm degraus de $17$ cm de altura e $29$ cm
        de profundidade. Que ângulo elas sobem? Compare com os
        $30^\circ$ do \cref{ex:g9:trig:special} e com uma escada de
        sótão bem íngreme, a $60^\circ$: que altura de degrau
        \emph{ela} precisaria numa profundidade de $29$ cm?
  \item A corrida da \omterm{def:g9:trig:ratios}{tangente}: calcule (com a calculadora)
        $\tan 10^\circ$, $\tan 45^\circ$, $\tan 80^\circ$,
        $\tan 89^\circ$. O que acontece quando o ângulo se aproxima de
        $90^\circ$, e por quê (pense no cateto adjacente)? A parede é
        vertical, a razão não tem valor, e o gráfico dispara para o
        infinito --- a sua primeira \emph{assíntota}, uma criatura
        estudada de perto no volume do ensino médio.
\end{enumerate}
\end{problem}
