\chapter{Frações e potências}\label{ch:g9:fractions}

Frações e \omterm{def:g9:fractions:power}{potências} são as ferramentas do dia a dia do cálculo:
repartir quantidades, comparar proporções, escrever números muito
grandes ou muito pequenos. Este capítulo consolida as regras de cálculo
com elas --- com cada passo intermediário escrito --- e apresenta a
\omterm{def:g9:fractions:scientific}{notação científica}.

\section{Calcular com frações}

\begin{definition}[Fração]\label{def:g9:fractions:fraction}
Uma \emph{fração}\index{fração} $\dfrac ab$ (com $b \neq 0$) é o
\omterm{def:g3:division:remainder}{quociente} de $a$ por $b$. Duas frações são iguais quando uma é obtida
da outra multiplicando (ou dividindo) \omterm{def:g4:fractions:def}{numerador} e \omterm{def:g4:fractions:def}{denominador} pelo
mesmo número não nulo:
\[
\frac ab = \frac{a \times k}{b \times k} \qquad (k \neq 0).
\]
\end{definition}

\begin{example}\label{ex:g9:fractions:simplify}
Simplifique $\dfrac{42}{56}$ passo a passo:
\[
\frac{42}{56} = \frac{42 \div 2}{56 \div 2} = \frac{21}{28}
= \frac{21 \div 7}{28 \div 7} = \frac34 .
\]
(No \cref{ch:g9:arith}, o máximo \omterm{def:g5:division:multiple}{divisor} comum fará isso num passo
só.)
\end{example}

\begin{method}[Somar e subtrair frações]\label{met:g9:fractions:add}
\begin{enumerate}
  \item Coloque as duas frações sobre um \emph{\omterm{def:g4:fractions:def}{denominador} comum} (um
        \omterm{def:g5:division:multiple}{múltiplo} comum dos dois \omterm{def:g4:fractions:def}{denominadores});
  \item some ou subtraia os \omterm{def:g4:fractions:def}{numeradores}, mantendo o \omterm{def:g4:fractions:def}{denominador};
  \item simplifique o resultado, se der.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:fractions:add}
Calcule $\dfrac56 + \dfrac34$. Um \omterm{def:g4:fractions:def}{denominador} comum de $6$ e $4$ é
$12$:
\[
\frac56 + \frac34
= \frac{5 \times 2}{6 \times 2} + \frac{3 \times 3}{4 \times 3}
= \frac{10}{12} + \frac{9}{12}
= \frac{19}{12}.
\]
Calcule $2 - \dfrac37$. Escreva $2$ como \omterm{def:g9:fractions:fraction}{fração} de \omterm{def:g4:fractions:def}{denominador} $7$:
\[
2 - \frac37 = \frac{14}{7} - \frac37 = \frac{11}{7}.
\]
\end{example}

\begin{method}[Multiplicar e dividir frações]\label{met:g9:fractions:mult}
\begin{enumerate}
  \item Para multiplicar, multiplique os \omterm{def:g4:fractions:def}{numeradores} entre si e os
        \omterm{def:g4:fractions:def}{denominadores} entre si:
        $\dfrac ab \times \dfrac cd = \dfrac{ac}{bd}$ (simplifique
        \emph{antes} de multiplicar sempre que puder);
  \item para dividir, multiplique pelo \omterm{def:g8:fractions:inverse}{inverso} do \omterm{def:g5:division:multiple}{divisor}:
        $\dfrac ab \div \dfrac cd = \dfrac ab \times \dfrac dc$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:fractions:mult}
\[
\frac{7}{15} \times \frac{25}{14}
= \frac{7 \times 25}{15 \times 14}
= \frac{7 \times 25}{15 \times 2 \times 7}
= \frac{25}{30} = \frac56,
\]
onde simplificamos pelo fator comum $7$ e depois por $5$. E uma
divisão:
\[
\frac{3}{4} \div \frac{9}{8}
= \frac34 \times \frac89
= \frac{3 \times 8}{4 \times 9}
= \frac{24}{36} = \frac23 .
\]
\end{example}

\section{Potências}

\begin{definition}[Potência]\label{def:g9:fractions:power}
Para um número real $a$ e um número inteiro positivo $n$, a
\emph{potência}\index{potência} $a^n$ é o \omterm{def:g2:mult:def}{produto} de $n$ fatores
iguais a $a$:
\[
a^n = \underbrace{a \times a \times \dots \times a}_{n \text{ fatores}} .
\]
Por convenção, $a^0 = 1$ (para $a \neq 0$), e os \omterm{def:g8:powers:def}{expoentes} negativos
indicam \omterm{def:g8:fractions:inverse}{inversos}:
\[
a^{-n} = \frac{1}{a^n}.
\]
\end{definition}

\begin{example}
$2^4 = 2 \times 2 \times 2 \times 2 = 16$; \quad
$(-3)^2 = 9$, mas $-3^2 = -(3^2) = -9$ (o \omterm{def:g8:powers:def}{expoente} age antes do sinal
de menos); \quad $10^{-3} = \frac{1}{1000} = 0.001$; \quad
$5^1 = 5$ e $5^0 = 1$.
\end{example}

\begin{theorem}[Regras dos expoentes]\label{thm:g9:fractions:rules}
Para todos os $a$, $b$ não nulos e todos os inteiros $m$, $n$:
\[
a^m \times a^n = a^{m+n},
\qquad
\frac{a^m}{a^n} = a^{m-n},
\qquad
\left(a^m\right)^n = a^{mn},
\qquad
(ab)^n = a^n b^n .
\]
\end{theorem}

\begin{proof}
Para \omterm{def:g8:powers:def}{expoentes} positivos, conte os fatores. Na primeira regra:
$a^m \times a^n$ são $m$ fatores $a$ seguidos de $n$ fatores $a$, no
total $m + n$ fatores. Na terceira: $\left(a^m\right)^n$ repete $n$
vezes um bloco de $m$ fatores, dando $mn$ fatores. As outras duas
regras são parecidas; a extensão aos \omterm{def:g8:powers:def}{expoentes} \omterm{def:g1:counting:zero}{zero} e negativos se
confere a partir de $a^{-n} = \frac{1}{a^n}$ (e faz da convenção
$a^0 = 1$ a única escolha coerente, já que $a^n \times a^0$ tem de ser
$a^{n+0} = a^n$).
\end{proof}

\begin{example}\label{ex:g9:fractions:rules}
Simplifique passo a passo:
\[
\frac{3^7 \times 3^{-2}}{3^4}
= \frac{3^{7 + (-2)}}{3^4}
= \frac{3^5}{3^4}
= 3^{5-4} = 3^1 = 3 .
\]
E com duas letras: $(2a^3)^2 \times a = 4 a^6 \times a = 4a^7$.
\end{example}

\section{Potências de dez e notação científica}

\begin{proposition}[Potências de dez]\label{prop:g9:fractions:ten}
Para um número inteiro positivo $n$: $10^n$ se escreve ``$1$ seguido
de $n$ \omterm{def:g1:counting:zero}{zeros}'', e $10^{-n} = 0.0\dots01$, com o $1$ na $n$-ésima casa
decimal. Multiplicar um \omterm{def:g6:decimals:places}{número decimal} por $10^n$ (resp.\ $10^{-n}$)
desloca o \omterm{def:g4:decimals:point}{ponto decimal} dele $n$ casas para a direita (resp.\ para a
esquerda).
\end{proposition}

\begin{definition}[Notação científica]\label{def:g9:fractions:scientific}
A \emph{notação científica}\index{notação científica} de um \omterm{def:g6:decimals:places}{número
decimal} positivo é o único jeito de escrevê-lo como
\[
a \times 10^n,
\qquad\text{com } 1 \leq a < 10 \text{ e } n \text{ inteiro.}
\]
\end{definition}

\begin{example}\label{ex:g9:fractions:scientific}
A distância da Terra ao Sol é de cerca de $149\,600\,000$ km
$= 1.496 \times 10^8$ km. O tamanho de uma molécula de água é de cerca
de $0.000\,000\,000\,28$ m $= 2.8 \times 10^{-10}$ m. Para calcular com
números assim, agrupe as partes decimais e as \omterm{def:g9:fractions:power}{potências} de dez:
\[
(3 \times 10^5) \times (2.5 \times 10^{-2})
= (3 \times 2.5) \times 10^{5 + (-2)}
= 7.5 \times 10^{3}.
\]
\end{example}

\begin{method}[Colocar um número em notação científica]\label{met:g9:fractions:scientific}
\begin{enumerate}
  \item Mova o \omterm{def:g4:decimals:point}{ponto decimal} para logo depois do primeiro algarismo não
        nulo; isso dá o fator $a$, com $1 \leq a < 10$;
  \item conte quantas casas o ponto andou: essa contagem é $n$,
        positivo se o número original era $\geq 10$, negativo se era
        $< 1$;
  \item confira: $a \times 10^n$ tem de reproduzir o número original.
\end{enumerate}
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g9:fractions:1}
Calcule e dê o resultado como \omterm{def:g9:fractions:fraction}{fração} totalmente simplificada:
\[
\frac13 + \frac15, \qquad
\frac72 - \frac53, \qquad
\frac{5}{12} + \frac38, \qquad
3 - \frac45 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:fractions:2}
Calcule e simplifique:
\[
\frac59 \times \frac{27}{35}, \qquad
\frac{8}{15} \div \frac{4}{5}, \qquad
\frac23 \times \frac34 \times \frac45 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:fractions:3}
Calcule:
\[
2^5, \qquad (-2)^4, \qquad -2^4, \qquad 4^{-2}, \qquad
\left(\tfrac23\right)^3, \qquad 7^0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:fractions:4}
Escreva como uma única \omterm{def:g9:fractions:power}{potência}:
\[
5^3 \times 5^6, \qquad
\frac{7^9}{7^5}, \qquad
\left(2^3\right)^4, \qquad
\frac{3^2 \times 3^7}{3^5}, \qquad
2^5 \times 4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:fractions:5}
Escreva em \omterm{def:g9:fractions:scientific}{notação científica}: $45\,000$; $0.0072$; $302.5$;
$0.000\,000\,9$; doze milhões.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:fractions:6}
Calcule, dando o resultado em \omterm{def:g9:fractions:scientific}{notação científica}:
\[
(2 \times 10^7) \times (4.5 \times 10^{-3}),
\qquad
\frac{9 \times 10^5}{3 \times 10^{-2}},
\qquad
(5 \times 10^3)^2 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:fractions:7}
Um reservatório está com $\frac35$ da \omterm{def:g2:measure:units}{capacidade}. Depois de receber
$30$ litros, ele fica com $\frac34$. Qual é a \omterm{def:g2:measure:units}{capacidade} do
reservatório? (Expresse primeiro a \omterm{def:g9:fractions:fraction}{fração} acrescentada.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:fractions:8}
Alice come $\frac14$ de um bolo, depois Bento come $\frac13$ do que
sobrou, e depois Carla come $\frac12$ do que ainda sobrou. Que \omterm{def:g9:fractions:fraction}{fração}
do bolo resta no fim? (Calcule o que sobra depois de cada etapa.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:fractions:9}
A luz percorre $3 \times 10^8$ metros por segundo. O Sol fica a cerca
de $1.5 \times 10^{11}$ m de distância. Quanto tempo a luz do Sol leva
para chegar até nós? Dê a resposta em segundos (sem precisar de \omterm{def:g9:fractions:scientific}{notação
científica}) e depois em minutos e segundos.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:fractions:10}
Qual é maior, $2^{30}$ ou $3^{20}$? (Dica: escreva os dois como
\omterm{def:g9:fractions:power}{potências} de \omterm{def:g8:powers:def}{expoente} $10$ usando
$\left(a^m\right)^n = a^{mn}$, e compare $2^3$ com $3^2$.)
\end{exercise}

\section{Problema: o código secreto das dízimas periódicas}

\begin{problem}[{Problema de fim de semana --- toda fração tem uma
escrita decimal que para ou se repete, toda dízima periódica é uma
fração, e $0.999\ldots$ é exatamente $1$}]
\label{pb:g9:fractions:1}
Divida $3$ por $11$ e os algarismos entram numa cantoria:
$0.272727\ldots$ para sempre. Divida $1$ por $4$ e os algarismos param
secos: $0.25$. Este problema demonstra que esses são os dois únicos
comportamentos possíveis de uma \omterm{def:g9:fractions:fraction}{fração} --- e depois inverte a máquina:
dada uma dízima periódica qualquer, ele reconstrói a \omterm{def:g9:fractions:fraction}{fração} escondida
atrás dela. Pelo caminho, resolve de uma vez por todas a igualdade mais
discutida da matemática escolar.

\medskip
\noindent\textbf{Parte I --- Da \omterm{def:g9:fractions:fraction}{fração} ao decimal.}
\begin{enumerate}
  \item Calcule, pela divisão longa, as escritas decimais de
        $\frac14$, $\frac13$, $\frac56$ e $\frac{3}{11}$. Quais param,
        quais se repetem, e com que bloco de repetição?
  \item Explique por que a escrita decimal de \emph{qualquer} \omterm{def:g9:fractions:fraction}{fração}
        $\frac ab$ tem de parar ou acabar se repetindo. (Na divisão
        longa por $b$, que valores os \omterm{def:g3:division:remainder}{restos} sucessivos podem tomar? O
        que acontece no instante em que um \omterm{def:g3:division:remainder}{resto} aparece pela segunda
        vez?)
  \item Leve a divisão de $1$ por $7$ longe o bastante para achar o
        bloco que se repete. Que comprimento ele tem, e como esse
        comprimento se compara com o limite prometido pela questão 2?
  \item Algumas frações param porque o \omterm{def:g4:fractions:def}{denominador} delas pode ser
        inflado até uma \omterm{def:g9:fractions:power}{potência} de dez:
        $\frac14 = \frac{25}{100}$,
        $\frac{3}{8} = \frac{375}{1000}$. Explique o critério e por que
        nenhum multiplicador inteiro consegue transformar $3$, $6$ ou
        $11$ em $10$, $100$, $1000$\,\dots\ (em que algarismo o \omterm{def:g2:mult:def}{produto}
        terminaria?).
  \item Sem dividir, separe $\frac{9}{40}$, $\frac{5}{12}$ e
        $\frac{33}{50}$ entre ``para'' e ``se repete'' e depois dê a
        escrita decimal das duas que param.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Da dízima periódica de volta à \omterm{def:g9:fractions:fraction}{fração}.}
\begin{enumerate}[resume]
  \item O truque do deslocamento. Seja $x = 0.7777\ldots$ Calcule
        $10x$, depois $10x - x$, e deduza $x$ como \omterm{def:g9:fractions:fraction}{fração}. Confira
        dividindo.
  \item Seja $x = 0.727272\ldots$ Que \omterm{def:g9:fractions:power}{potência} de dez desloca $x$ em
        exatamente um bloco de repetição? Use-a para escrever $x$ como
        \omterm{def:g9:fractions:fraction}{fração} irredutível.
  \item Um caso misto: $x = 0.58333\ldots$ (só o $3$ se repete).
        Calcule $1000x - 100x$ e deduza $x$ como \omterm{def:g9:fractions:fraction}{fração} irredutível.
  \item O famoso: aplique o truque do deslocamento a
        $x = 0.9999\ldots$ Que \omterm{def:g9:fractions:fraction}{fração} --- que \emph{número} --- você
        encontra? Concilie o resultado com a sua intuição, lembrando o
        vizinho fantasma do \cref{pb:g6:decimals:1} e o \omterm{def:g3:division:remainder}{resto} que
        encolhe do \cref{pb:g6:fractions:1}: $0.999\ldots$ fica
        ``logo abaixo'' de $1$ ou é outro nome para $1$?
  \item Converta $2.454545\ldots$ e $0.123123123\ldots$ em frações
        irredutíveis.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O dicionário completo.}
\begin{enumerate}[resume]
  \item As questões 6 a 10 sugerem um dicionário: um bloco de $k$
        algarismos que se repete a partir do \omterm{def:g4:decimals:point}{ponto decimal} é igual a
        esse bloco sobre $k$ noves ($0.727272\ldots = \frac{72}{99}$).
        Use o dicionário para prever $0.037037\ldots$ como \omterm{def:g9:fractions:fraction}{fração},
        simplifique --- e admire-se: que \omterm{def:g9:fractions:fraction}{fração} surpreendentemente
        simples canta ``$037$''? Confira dividindo.
  \item Combine o dicionário com um deslocamento: escreva
        $0.0272727\ldots$ como \omterm{def:g9:fractions:fraction}{fração} irredutível.
  \item O número $0.101001000100001\ldots$ (um $1$, depois um $0$,
        depois um $1$, depois dois $0$, depois um $1$, depois três $0$,
        e assim por diante) nunca para. Ele é uma dízima periódica? O
        que a Parte I diz então sobre ele --- pode ser uma \omterm{def:g9:fractions:fraction}{fração}?
        (Você acaba de conhecer o seu primeiro número comprovadamente
        \emph{irracional}; o mais famoso deles é capturado no
        \cref{ch:g9:sqrt}.)
  \item Aproximadamente quantos algarismos tem $2^{30}$? Use
        $2^{10} = 1\,024 \approx 1.024 \times 10^3$ e as regras dos
        \omterm{def:g8:powers:def}{expoentes} (\cref{thm:g9:fractions:rules}) para escrever
        $2^{30}$ em \omterm{def:g9:fractions:scientific}{notação científica} (aproximada) e conclua. (Compare
        com o \cref{exo:g9:fractions:10}.)
  \item Final: escreva $0.20262026\ldots$ como \omterm{def:g9:fractions:fraction}{fração} e enuncie a moral
        de mão dupla do problema inteiro: que escritas decimais são
        frações, e que frações têm que escritas decimais?
\end{enumerate}
\end{problem}
