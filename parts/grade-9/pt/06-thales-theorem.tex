\chapter{O teorema de Tales}\label{ch:g9:thales}

Como se mede a altura de uma \omterm{def:g8:solids:def}{pirâmide} sem escalá-la? Tales de Mileto
comparou a sombra dela com a sombra de uma vara. O teorema dele --- as
\omterm{def:g6:lines:perp}{retas paralelas} cortam \omterm{def:g6:lines:objects}{segmentos} em pedaços \omterm{def:g9:linfunc:linear}{proporcionais} --- é o
coração matemático das maquetes, dos mapas e das ampliações, e um dos
teoremas mais antigos a ter nome próprio.

\section{O teorema}

\begin{theorem}[Tales]\label{thm:g9:thales:direct}
Sejam duas retas que se encontram num ponto $A$. Tome $M$ sobre a
primeira reta e $N$ sobre a segunda, com $B$ mais adiante na primeira e
$C$ mais adiante na segunda. Se as retas $(MN)$ e $(BC)$ são
\emph{\omterm{def:g6:lines:perp}{paralelas}}, então os triângulos $AMN$ e $ABC$ têm lados
\omterm{def:g9:linfunc:linear}{proporcionais}:
\[
\frac{AM}{AB} = \frac{AN}{AC} = \frac{MN}{BC}.
\]
\end{theorem}
\admitted

\begin{method}[Calcular um comprimento com Tales]\label{met:g9:thales:compute}
\begin{enumerate}
  \item Identifique o ponto $A$ em que as duas retas se cruzam e as
        duas \omterm{def:g6:lines:perp}{retas paralelas}; nomeie os dois triângulos;
  \item escreva as três razões iguais, sempre ``triângulo pequeno sobre
        triângulo grande'';
  \item guarde a igualdade das duas razões que envolvem os três
        comprimentos conhecidos e o desconhecido;
  \item resolva a proporção obtida multiplicando em cruz.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:thales:compute}
Na configuração da esquerda, acima, suponha $AM = 3$, $AB = 5$,
$AN = 2.4$ e $MN = 3.3$, e vamos calcular $AC$ e $BC$. Tales dá
\[
\frac{AM}{AB} = \frac{AN}{AC} = \frac{MN}{BC},
\qquad\text{isto é,}\qquad
\frac35 = \frac{2.4}{AC} = \frac{3.3}{BC}.
\]
De $\frac35 = \frac{2.4}{AC}$, multiplicando em cruz:
$3 \times AC = 5 \times 2.4 = 12$, então $AC = 4$. De
$\frac35 = \frac{3.3}{BC}$: $3 \times BC = 16.5$, então $BC = 5.5$.
\end{example}

\begin{example}[Tales na vida real]\label{ex:g9:thales:stick}
Uma vara vertical de $1$ m de altura projeta uma sombra de $1.5$ m,
enquanto uma árvore projeta uma sombra de $12$ m no mesmo instante. Os
raios de sol são paralelos, então o triângulo da vara com a sombra dela
e o da árvore com a sombra dela estão em configuração de Tales:
\[
\frac{\text{altura da árvore}}{1} = \frac{12}{1.5},
\qquad\text{então a árvore tem } 8 \text{ m de altura.}
\]
\end{example}

\section{A recíproca}

\begin{theorem}[Recíproca de Tales]\label{thm:g9:thales:converse}
Com os pontos posicionados como no \cref{thm:g9:thales:direct} ($M$
sobre $(AB)$, $N$ sobre $(AC)$, na mesma ordem nas duas retas): se
\[
\frac{AM}{AB} = \frac{AN}{AC},
\]
então as retas $(MN)$ e $(BC)$ são \omterm{def:g6:lines:perp}{paralelas}.
\end{theorem}
\admitted

\begin{method}[Demonstrar ou refutar o paralelismo]\label{met:g9:thales:converse}
\begin{enumerate}
  \item Calcule separadamente as duas razões $\dfrac{AM}{AB}$ e
        $\dfrac{AN}{AC}$ (como frações, não como \omterm{def:g6:decimals:rounding}{arredondamentos});
  \item se elas forem iguais \emph{e} os pontos estiverem na mesma ordem
        nas duas retas, as retas são \omterm{def:g6:lines:perp}{paralelas} (recíproca de Tales);
  \item se forem diferentes, as retas \emph{não} são \omterm{def:g6:lines:perp}{paralelas} ---
        porque, se fossem, o teorema de Tales obrigaria as razões a
        serem iguais.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:thales:converse}
Na reta 1, $AM = 4$ e $AB = 10$; na reta 2, $AN = 6$ e $AC = 15$.
Então
\[
\frac{AM}{AB} = \frac{4}{10} = \frac25,
\qquad
\frac{AN}{AC} = \frac{6}{15} = \frac25 :
\]
razões iguais, mesma ordem: $(MN) \parallel (BC)$.

Com $AC = 14$ em vez disso: $\frac{6}{14} = \frac37 \neq \frac25$,
então as retas não são \omterm{def:g6:lines:perp}{paralelas}.
\end{example}

\section{Ampliação e redução}

\begin{definition}[Mudar a escala de uma figura]\label{def:g9:thales:scaling}
Ampliar ou reduzir uma figura pelo \emph{fator de escala}\index{fator
de escala} $k > 0$ é multiplicar \emph{todos} os comprimentos dela por
$k$: uma \emph{ampliação} quando $k > 1$, uma \emph{redução} quando
$k < 1$. Na configuração de Tales, o triângulo $AMN$ é a redução de
$ABC$ pelo fator $k = \frac{AM}{AB}$.
\end{definition}

\begin{proposition}[Efeito sobre ângulos e áreas]\label{prop:g9:thales:scaling}
Mudar a escala pelo fator $k$ preserva os ângulos e a forma das
figuras, multiplica todo comprimento por $k$ e multiplica toda
\emph{\omterm{def:g6:measure:area}{área}} por $k^2$.
\end{proposition}
\admitted

\begin{omfigure}
\begin{tikzpicture}[scale=0.72]
  \draw[thick, omDef, fill=omDef!8] (0,0) rectangle (2,1.2);
  \draw[thick, omThm, fill=omThm!8] (3.4,0) rectangle (7.4,2.4);
  \node[font=\small, omDef] at (1,-0.45) {$2 \times 1.2$};
  \node[font=\small, omThm] at (5.4,-0.45) {$4 \times 2.4$};
  \node[font=\small] at (1,0.6) {área $2.4$};
  \node[font=\small] at (5.4,1.2) {área $9.6$};
\end{tikzpicture}

{\small Dobrar os comprimentos ($k = 2$) multiplica a \omterm{def:g6:measure:area}{área} por
$k^2 = 4$: quatro cópias do retângulo pequeno ladrilham o grande.}
\end{omfigure}

\begin{example}
Uma foto de $10 \times 15$ cm é ampliada com fator de escala $k = 3$: a
cópia mede $30 \times 45$ cm. A \omterm{def:g6:measure:area}{área} dela passa de $150$ cm$^2$ para
$150 \times 9 = 1350$ cm$^2$ --- $9$ vezes maior, e não $3$.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g9:thales:1}
As retas $(MN)$ e $(BC)$ são \omterm{def:g6:lines:perp}{paralelas}, com $M$ sobre $[AB]$ e $N$
sobre $[AC]$; $AM = 2$, $AB = 6$, $AN = 3$, $BC = 9$. Calcule $AC$ e
$MN$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:2}
Mesma configuração: $AM = 5$, $MB = 3$ (atenção: $MB$, e não $AB$!),
$AN = 4$. Calcule $AC$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:3}
Numa configuração borboleta, $A$ está entre $M$ e $B$ e entre $N$ e
$C$, com $(MN) \parallel (BC)$; $AM = 3$, $AB = 7.5$, $AN = 2$,
$MN = 2.6$. Calcule $AC$ e $BC$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:4}
$M$ está sobre $[AB]$ com $AM = 6$, $AB = 8$; $N$ está sobre $[AC]$ com
$AN = 9$, $AC = 12$. As retas $(MN)$ e $(BC)$ são \omterm{def:g6:lines:perp}{paralelas}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:5}
Como acima, com $AM = 4$, $AB = 6$, $AN = 5$, $AC = 8$. As retas $(MN)$
e $(BC)$ são \omterm{def:g6:lines:perp}{paralelas}? Justifique com cuidado.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:6}
Uma pessoa de $1.8$ m de altura está a $2$ m da base de um poste de
iluminação; a sombra dela mede $3$ m. Trace a configuração de Tales
formada pelo poste, pela pessoa e pela ponta da sombra, e calcule a
altura do poste.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:7}
Um mapa tem escala $1 : 25\,000$ (os comprimentos no mapa são os
comprimentos reais multiplicados por $k = \frac{1}{25000}$).
\begin{enumerate}
  \item Dois povoados distam $6.8$ cm um do outro no mapa. Qual é a
        distância real, em km?
  \item Uma floresta tem \omterm{def:g6:measure:area}{área} de $8$ cm$^2$ no mapa. Qual é a \omterm{def:g6:measure:area}{área} real
        dela, em km$^2$?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:8}
O triângulo $ABC$ tem $AB = 12$, $AC = 15$, $BC = 18$. O ponto $M$
sobre $[AB]$ satisfaz $AM = 8$, e a reta que passa por $M$
paralelamente a $(BC)$ corta $[AC]$ em $N$.
\begin{enumerate}
  \item Calcule $AN$ e $MN$.
  \item Qual é o \omterm{def:g9:thales:scaling}{fator de escala} de $ABC$ para $AMN$? Compare os
        \omterm{def:g6:measure:perimeter}{perímetros} dos dois triângulos.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:thales:9}
Seja $ABCD$ um trapézio com $(AB) \parallel (CD)$, $AB = 4$ e
$CD = 6$, cujas diagonais $[AC]$ e $[BD]$ se encontram em $O$. Usando
Tales na configuração borboleta em torno de $O$, calcule a razão
$\dfrac{OA}{OC}$ e mostre que $O$ corta as duas diagonais na mesma
razão.
\end{exercise}

\section{Problema: a régua sem marcação, a câmara escura e as três
médias de um trapézio}

\begin{problem}[{Problema de fim de semana --- Tales em ação: dividir
um segmento qualquer em partes iguais, medir o Sol com uma caixa de
sapatos e um trapézio em que as três médias famosas se encontram}]
\label{pb:g9:thales:1}
O teorema de Tales é a matemática dos \emph{raios}: raios de sol, raios
de luz por um furinho, feixes de retas que saem de um \omterm{def:g5:solids:def}{vértice}. Este
problema o usa de três maneiras --- como ferramenta de construção
(dividir um \omterm{def:g6:lines:objects}{segmento} de comprimento \emph{qualquer}, até mesmo
$\sqrt2$, em partes perfeitamente iguais), como instrumento de medida
(o \omterm{def:g4:geometry:circle}{diâmetro} do Sol, com uma caixa de sapatos e uma moeda) e como lupa
sobre o trapézio do \cref{exo:g9:thales:9}, em que as médias
aritmética, geométrica e harmônica desta série aparecem todas numa
figura só.

\medskip
\noindent\textbf{Parte I --- Dividir com uma régua sem marcação.}
\begin{enumerate}
  \item Trace um \omterm{def:g6:lines:objects}{segmento} $[AB]$ de $7$ cm. Para cortá-lo em três
        partes iguais com compasso e régua sem marcação: trace uma
        semirreta qualquer a partir de $A$ (que não passe por $B$);
        marque nela três aberturas iguais de compasso, obtendo os
        pontos $P_1$, $P_2$, $P_3$; ligue $P_3$ a $B$; trace as
        \omterm{def:g6:lines:perp}{paralelas} a $(P_3 B)$ que passam por $P_1$ e por $P_2$. Execute
        a construção e marque onde as \omterm{def:g6:lines:perp}{paralelas} cortam $[AB]$.
  \item Justifique com o \cref{thm:g9:thales:direct} que os dois pontos
        marcados cortam $[AB]$ exatamente nos terços dele.
  \item Adapte o método para dividir um \omterm{def:g6:lines:objects}{segmento} em $5$ partes iguais e
        depois para construir $\frac35$ de um \omterm{def:g6:lines:objects}{segmento} dado.
  \item Construa o ponto $M$ de $[AB]$ com $\frac{AM}{MB} = \frac23$
        (isto é, $M$ a $\frac25$ do caminho a partir de $A$). Quantos
        passos iguais na semirreta, e que ponto se liga a $B$?
  \item Por que essa construção é melhor do que medir com uma régua
        graduada? Considere um \omterm{def:g6:lines:objects}{segmento} de comprimento $\sqrt2$ (a
        diagonal de um quadrado de lado $1$, irracional pelo
        \cref{pb:g9:sqrt:1}): o que a medição daria, e o que Tales dá?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A câmara escura de furinho.}
Fure com um alfinete uma \omterm{def:g5:solids:def}{face} de uma caixa fechada: na \omterm{def:g5:solids:def}{face} oposta,
aparece uma imagem do mundo de cabeça para baixo. Um ponto do objeto, o
furinho e o ponto da imagem ficam alinhados --- a luz viaja em linha
reta --- então o objeto (altura $H$, à distância $D$ na frente do
furo) e a imagem dele (altura $h$, na parede do fundo, à profundidade
$d$ atrás do furo) ficam na configuração borboleta de Tales.
\begin{enumerate}[resume]
  \item Deduza a fórmula do furinho, $h = H \times \frac dD$, de Tales
        na borboleta em torno do furo.
  \item Uma árvore de $12$ m fica a $30$ m de uma caixa de sapatos de
        $20$ cm de profundidade. Que altura tem a imagem dela, e em que
        posição?
  \item Aponte a caixa para o Sol: a $d = 1$ m atrás do furinho, a
        imagem do Sol é um disco de cerca de $9$ mm de \omterm{def:g4:geometry:circle}{diâmetro}. Dada a
        distância do Sol, $D = 1.5 \times 10^{11}$ m, calcule o
        \omterm{def:g4:geometry:circle}{diâmetro} dele em \omterm{def:g9:fractions:scientific}{notação científica} (a notação do
        \cref{pb:g9:fractions:1} em ação). Compare com o valor
        verdadeiro, $1.39 \times 10^9$ m.
  \item A Lua: \omterm{def:g4:geometry:circle}{diâmetro} $3.5 \times 10^6$ m, distância
        $3.8 \times 10^8$ m. Calcule a razão
        $\frac{\text{diâmetro}}{\text{distância}}$ para a Lua e para o
        Sol. Que coincidência de sorte os dois números revelam --- e
        que acontecimento espetacular ela torna possível?
  \item Em uma ou duas frases: por que a imagem do furinho fica de
        cabeça para baixo, e por que alargar o furo deixa a imagem mais
        clara, porém mais borrada?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O trapézio das três médias.}
$ABCD$ é o trapézio do \cref{exo:g9:thales:9}:
$(AB) \parallel (CD)$, $AB = 4$, $CD = 6$, com as diagonais se cruzando
em $O$ e $\frac{OA}{OC} = \frac{OB}{OD} = \frac{4}{6} = \frac23$.
\begin{enumerate}[resume]
  \item Treino de borboleta: duas retas se cruzam em $O$ com
        $(MN) \parallel (PQ)$ em posição de borboleta; $OM = 3$,
        $OP = 7.5$, $MN = 4$ e $OQ = 6$. Calcule $PQ$ e $ON$.
  \item Explique como Tales, aplicado com razão $\frac12$, contém o
        teorema dos pontos médios do \cref{thm:g8:midpoints:direct}
        como caso particular.
  \item Trace a paralela às bases que passa por $O$; ela encontra
        $[AD]$ em $E$. Trabalhando no triângulo $ACD$ (note que
        $\frac{AO}{AC} = \frac25$), calcule $EO$.
  \item Pela conta espelhada no triângulo $BDC$, o pedaço $OF$ (com $F$
        sobre $[BC]$) tem o mesmo comprimento. Conclua que
        \[
        EF = 2 \times \frac{4 \times 6}{4 + 6} = 4.8 :
        \]
        a paralela que passa pelo cruzamento das diagonais mede a
        \emph{\omterm{pb:g8:speed:1}{média harmônica}} das bases --- a média da ida e volta do
        \cref{pb:g8:speed:1}, reaparecendo na geometria pura.
  \item O grande final: calcule as três médias clássicas das bases $4$
        e $6$ --- aritmética $\frac{4+6}{2}$, geométrica
        $\sqrt{4 \times 6}$, harmônica
        $\frac{2 \times 4 \times 6}{4 + 6}$ --- e ordene-as. Cada uma
        mora no trapézio: a média aritmética é a base média, que liga
        os pontos médios dos lados oblíquos (\cref{pb:g7:areas:1}); a
        \omterm{pb:g8:speed:1}{média harmônica} é o seu \omterm{def:g6:lines:objects}{segmento} $EF$; e a \omterm{pb:g8:cosine:1}{média geométrica} é a
        paralela que corta o trapézio em dois trapézios
        \emph{semelhantes} (confira o valor dela com a calculadora; a
        demonstração é um belo desafio extra). Onde a desigualdade
        entre as três foi demonstrada neste livro
        (\cref{pb:g8:cosine:1}, \cref{pb:g8:speed:1})?
\end{enumerate}
\end{problem}
