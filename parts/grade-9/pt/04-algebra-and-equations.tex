\chapter{Álgebra e equações}\label{ch:g9:algebra}

As letras representam números, e calcular com letras demonstra fatos
sobre \emph{todos} os números de uma vez. Este capítulo treina
\omterm{def:g9:algebra:expand}{desenvolver} e fatorar --- inclusive os três \omterm{def:g2:mult:def}{produtos} notáveis que vão
acompanhar você por toda a matemática --- e resolve \omterm{def:g8:equations:def}{equações} do primeiro
grau, \omterm{def:g2:mult:def}{equações-produto} e desigualdades simples, sempre um passo
justificado de cada vez.

\section{Desenvolver}

\begin{definition}[Desenvolvimento]\label{def:g9:algebra:expand}
\emph{Desenvolver}\index{desenvolver} é transformar um \omterm{def:g2:mult:def}{produto} numa
\omterm{def:g1:addition:def}{soma}, usando a distributividade:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g9:algebra:expand}
Cada termo do primeiro parêntese multiplica cada termo do segundo:
\begin{align*}
(x + 3)(2x + 5)
&= x \times 2x + x \times 5 + 3 \times 2x + 3 \times 5 \\
&= 2x^2 + 5x + 6x + 15 \\
&= 2x^2 + 11x + 15 .
\end{align*}
Os sinais viajam junto com os termos:
\begin{align*}
(3x - 2)(x - 4)
&= 3x^2 - 12x - 2x + 8
= 3x^2 - 14x + 8 .
\end{align*}
\end{example}

\begin{theorem}[Os três produtos notáveis]\label{thm:g9:algebra:identities}
Para todos os números $a$ e $b$:
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
Desenvolva cada lado esquerdo. No primeiro:
$(a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2$. No segundo,
troque $b$ por $-b$ no primeiro. No terceiro:
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$ (os termos cruzados se
cancelam).
\end{proof}

\begin{example}\label{ex:g9:algebra:identities}
\[
(x + 5)^2 = x^2 + 10x + 25, \qquad
(3x - 1)^2 = 9x^2 - 6x + 1, \qquad
(2x + 7)(2x - 7) = 4x^2 - 49 .
\]
Cálculo mental: $102^2 = (100 + 2)^2 = 10000 + 400 + 4 = 10404$, e
$98 \times 102 = (100-2)(100+2) = 10000 - 4 = 9996$.
\end{example}

\section{Fatorar}

\begin{method}[Fatorar]\label{met:g9:algebra:factor}
Para fatorar uma expressão, tente, nesta ordem:
\begin{enumerate}
  \item \emph{fator comum}: $ka + kb = k(a + b)$, em que o próprio $k$
        pode ser um parêntese;
  \item \emph{\omterm{ex:g1:subtraction:difference}{diferença} de quadrados}: $a^2 - b^2 = (a+b)(a-b)$;
  \item \emph{quadrado perfeito}: $a^2 + 2ab + b^2 = (a+b)^2$ (e, com
        $-2ab$, $(a-b)^2$).
\end{enumerate}
Confira sempre desenvolvendo o resultado.
\end{method}

\begin{example}\label{ex:g9:algebra:factor}
\emph{Fator comum:}
$15x^2 + 10x = 5x \times 3x + 5x \times 2 = 5x(3x + 2)$.

\emph{Parêntese comum:}
\[
(x+2)(x-1) + (x+2)(3x+4) = (x+2)\bigl[(x-1) + (3x+4)\bigr]
= (x+2)(4x+3).
\]

\emph{\omterm{ex:g1:subtraction:difference}{Diferença} de quadrados:}
$x^2 - 81 = (x+9)(x-9)$, e
\[
(2x+1)^2 - 9 = (2x+1)^2 - 3^2 = (2x+1+3)(2x+1-3) = (2x+4)(2x-2).
\]
Cada fator ainda tem um fator comum:
$(2x+4)(2x-2) = 2(x+2) \times 2(x-1) = 4(x+2)(x-1)$.
\end{example}

\section{Equações}

\begin{method}[Equações do primeiro grau]\label{met:g9:algebra:firstdegree}
Para resolver uma \omterm{def:g8:equations:def}{equação} como $5x - 3 = 2x + 9$:
\begin{enumerate}
  \item junte os termos em $x$ de um lado e os números do outro,
        somando a mesma quantidade aos dois lados;
  \item reduza cada lado;
  \item divida os dois lados pelo coeficiente de $x$;
  \item confira a solução na \omterm{def:g8:equations:def}{equação} original.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:algebra:firstdegree}
\begin{align*}
5x - 3 &= 2x + 9 \\
5x - 2x &= 9 + 3 && \text{(some $3 - 2x$ aos dois lados)}\\
3x &= 12 \\
x &= 4 && \text{(divida por 3).}
\end{align*}
Conferindo: $5 \times 4 - 3 = 17$ e $2 \times 4 + 9 = 17$. A solução é
$4$.

Com parênteses, desenvolva primeiro: $3(x - 2) = x + 8$ vira
$3x - 6 = x + 8$, então $2x = 14$ e $x = 7$.
\end{example}

\begin{theorem}[Regra do produto nulo]\label{thm:g9:algebra:zeroproduct}
Um \omterm{def:g2:mult:def}{produto} é nulo se, e somente se, pelo menos um fator é nulo:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ ou } \ B = 0 .
\]
\end{theorem}

\begin{proof}
Se um fator é nulo, o \omterm{def:g2:mult:def}{produto} claramente é. Reciprocamente, se
$A B = 0$ com $A \neq 0$, dividir os dois lados por $A$ dá $B = 0$.
\end{proof}

\begin{example}\label{ex:g9:algebra:zeroproduct}
Resolva $(x - 3)(2x + 10) = 0$: ou $x - 3 = 0$, ou $2x + 10 = 0$, então
as soluções são $3$ e $-5$.

Resolva $x^2 - 49 = 0$: fatore primeiro, $(x-7)(x+7) = 0$, soluções $7$
e $-7$.

Resolva $x^2 = 3x$: nunca divida por $x$! Passe tudo para um lado e
fatore: $x^2 - 3x = x(x - 3) = 0$, soluções $0$ e $3$.
\end{example}

\section{Desigualdades}

\begin{proposition}[Regras das desigualdades]\label{prop:g9:algebra:ineq}
Uma desigualdade se conserva quando se \omterm{def:g1:addition:def}{soma} o mesmo número aos dois
lados e quando se multiplicam os dois lados pelo mesmo número
\emph{positivo}. Multiplicar os dois lados por um número
\emph{negativo} inverte o sinal da desigualdade.
\end{proposition}

\begin{proof}
Se $a \leq b$, então $b - a \geq 0$. Somando $c$:
$(b+c) - (a+c) = b - a \geq 0$, então $a + c \leq b + c$. Multiplicando
por $c > 0$: $bc - ac = (b-a)c \geq 0$, então $ac \leq bc$.
Multiplicando por $c < 0$: $(b-a)c \leq 0$, então $ac \geq bc$: a ordem
se inverte.
\end{proof}

\begin{example}\label{ex:g9:algebra:ineq}
Resolva $7 - 2x < 15$:
\begin{align*}
7 - 2x &< 15 \\
-2x &< 8 && \text{(subtraia 7)} \\
x &> -4 && \text{(divida por $-2$: inverta o sinal).}
\end{align*}
As soluções são todos os números maiores que $-4$: numa reta numérica,
uma bolinha vazada em $-4$ e o traço para a direita.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g9:algebra:1}
Desenvolva e reduza:
\[
4(3x - 2), \qquad
(x + 6)(x + 2), \qquad
(2x - 3)(x + 5), \qquad
(x - 1)(x - 9).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:algebra:2}
Desenvolva usando os \omterm{def:g2:mult:def}{produtos} notáveis:
\[
(x + 8)^2, \qquad
(3x - 4)^2, \qquad
(x + 10)(x - 10), \qquad
(5x + 2)(5x - 2).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:algebra:3}
Calcule de cabeça, usando um \omterm{def:g2:mult:def}{produto} notável: $101^2$; \quad $59^2$
(escreva $59 = 60 - 1$); \quad $53 \times 47$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:algebra:4}
Fatore:
\[
9x + 12, \qquad
x^2 - 64, \qquad
x^2 + 14x + 49, \qquad
18x^2 - 6x .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:algebra:5}
Resolva, escrevendo cada passo:
\[
4x + 7 = 19, \qquad
6x - 5 = 2x + 11, \qquad
5(x - 3) = 3x + 1 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:algebra:6}
Resolva usando a regra do \omterm{def:g2:mult:def}{produto} nulo:
\[
(x + 4)(3x - 12) = 0, \qquad
x^2 - 121 = 0, \qquad
(2x - 1)^2 = 0, \qquad
x^2 = 8x .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:algebra:7}
Fatore $E = (3x + 2)^2 - 25$ e depois resolva $E = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:algebra:8}
Resolva as desigualdades e represente as soluções numa reta numérica:
\[
3x - 4 \geq 8, \qquad
5 - 2x > 1, \qquad
4(x + 1) \leq x - 5 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:algebra:9}
Penso num número, multiplico por $3$, somo $7$ e obtenho o mesmo
resultado que se tivesse multiplicado por $5$ e subtraído $9$. Qual é o
meu número? (Monte uma \omterm{def:g8:equations:def}{equação} e resolva.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:algebra:10}
Seja $n$ um número inteiro.
\begin{enumerate}
  \item Desenvolva $(n + 1)(n - 1)$ e deduza que o \omterm{def:g2:mult:def}{produto} dos dois
        vizinhos de $n$ é sempre $n^2 - 1$ (por exemplo,
        $24 \times 26 = 25^2 - 1 = 624$).
  \item Mostre que a \omterm{def:g1:addition:def}{soma} de três inteiros consecutivos é sempre
        múltipla de $3$.
\end{enumerate}
\end{exercise}

\section{Problema: a álgebra dos algarismos}

\begin{problem}[{Problema de fim de semana --- por que os critérios de
divisibilidade funcionam, o truque do 1089 desmascarado e um cálculo
mental que parece feitiçaria}]
\label{pb:g9:algebra:1}
A escola ensinou \emph{regras} a você (um número é \omterm{def:g6:wholes:divisible}{divisível} por $9$
quando a \omterm{def:g1:addition:def}{soma} dos algarismos é); os mágicos fazem \emph{truques} (todo
mundo termina em $1089$); as feiras conhecem \emph{atalhos}
($35^2 = 1225$, na hora). Tudo isso é o mesmo segredo: um número de
algarismos $a$, $b$, $c$ é a expressão $100a + 10b + c$, e a álgebra
deste capítulo (\cref{thm:g9:algebra:identities}, \omterm{def:g9:algebra:expand}{desenvolver}, fatorar)
consegue desmontá-la. No fim deste problema, você terá demonstrado as
regras, desmascarado o truque e aprendido a feitiçaria.

\medskip
\noindent\textbf{Parte I --- Os algarismos no microscópio.}
\begin{enumerate}
  \item Um número de dois algarismos com dezena $a$ e unidade $b$ é
        $10a + b$. Desenvolva e fatore a \omterm{ex:g1:subtraction:difference}{diferença} entre o número e o
        invertido dele, $(10a + b) - (10b + a)$, e enuncie a
        descoberta. Confira em $72 - 27$.
  \item Demonstre o critério do nove para números de três algarismos:
        verifique a identidade
        \[
        100a + 10b + c = 9\,(11a + b) + (a + b + c),
        \]
        e explique por que ela mostra que um número e a \omterm{def:g1:addition:def}{soma} dos
        algarismos dele deixam o \emph{mesmo \omterm{def:g3:division:remainder}{resto}} na divisão por $9$
        --- então um é \omterm{def:g6:wholes:divisible}{divisível} por $9$ exatamente quando o outro é
        (\cref{prop:g9:arith:rules}, agora demonstrada).
  \item Deduza o critério do três da mesma identidade, em uma frase.
  \item A ``prova dos nove'', a conferência dos contadores: para testar
        $47 \times 86 = 4\,042$, troque cada número pela \omterm{def:g1:addition:def}{soma} dos
        algarismos da \omterm{def:g1:addition:def}{soma} dos algarismos dele: $47 \to 11 \to 2$,
        $86 \to 14 \to 5$, e $2 \times 5 = 10 \to 1$; depois
        $4\,042 \to 10 \to 1$: coerente. Explique por que um \omterm{def:g2:mult:def}{produto}
        correto tem de passar no teste (escreva $47 = 9k + 2$,
        $86 = 9l + 5$ e desenvolva) e encontre um resultado errado que
        o teste não pega (por que algarismos trocados de lugar são
        invisíveis para ele?).
  \item Demonstre o critério do onze para números de quatro
        algarismos: verifique
        \[
        1000a + 100b + 10c + d
        = 11\,(91a + 9b + c) + (- a + b - c + d),
        \]
        e use-o para decidir se $2\,926$ é \omterm{def:g6:wholes:divisible}{divisível} por $11$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O truque do 1089.}
O truque: pense num número de três algarismos cujo primeiro e último
algarismos difiram em pelo menos $2$; inverta-o; subtraia o menor do
maior; inverta o resultado; some os dois últimos números. O mágico
anuncia: \emph{1089}.
\begin{enumerate}[resume]
  \item Execute o truque com $742$ e com um número da sua escolha.
  \item Seja o número de algarismos $a$, $b$, $c$ com $a > c$. Fatore a
        primeira subtração, $(100a + 10b + c) - (100c + 10b + a)$, e
        liste todos os valores que ela pode tomar quando $a - c$ vai de
        $2$ a $9$.
  \item Cada um desses \omterm{def:g5:division:multiple}{múltiplos} de $99$ tem um padrão de algarismos
        marcante. Escrevendo $m = a - c$, verifique a identidade
        \[
        99m = 100\,(m - 1) + 90 + (10 - m),
        \]
        e leia nela os três algarismos de $99m$.
  \item Some $99m$ ao invertido dele, usando os algarismos da questão
        8, e demonstre que o total é $1089$ --- seja qual for $m$. O
        truque morreu; viva a álgebra.
  \item Por que o mágico exigiu que o primeiro e o último algarismos
        diferissem em pelo menos $2$? Examine os casos $a = c$ e
        $a - c = 1$ (em que a subtração dá $99$, que precisa ser
        escrito $099$ para a inversão funcionar) e enuncie as letras
        miúdas que mantêm o truque honesto.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Feitiçaria de uso diário.}
\begin{enumerate}[resume]
  \item O truque do aniversário: ``multiplique o mês de nascimento por
        $5$, some $7$, dobre, some o dia de nascimento, subtraia
        $14$''. Mostre que o resultado é sempre $10m + d$ --- mês e
        dia, legíveis de relance. (Teste em alguém.)
  \item Você redesenha o truque assim: ``multiplique o mês por $5$,
        some $9$, dobre, some o dia''. Que número o espectador tem de
        subtrair no fim, agora, para a mesma leitura mês--dia?
  \item O truque do algarismo escondido: o espectador escolhe um número
        qualquer, subtrai dele a \omterm{def:g1:addition:def}{soma} dos algarismos (a questão 2 diz
        que o resultado é sempre \omterm{def:g9:arith:divisor}{múltiplo} de $9$ --- explique por quê,
        para um número de três algarismos), depois risca um algarismo
        \emph{não nulo} e lê para você os algarismos restantes.
        Explique como você recupera o algarismo riscado e por que o $0$
        teve de ser excluído.
  \item Feitiçaria de feira: demonstre a identidade
        $(10t + 5)^2 = 100\,t(t+1) + 25$, enuncie a regra para elevar
        ao quadrado qualquer número terminado em $5$ e calcule $45^2$,
        $85^2$ e $105^2$ de cabeça.
  \item Final, estendendo o \cref{exo:g9:algebra:10}: demonstre que a
        \omterm{def:g1:addition:def}{soma} de cinco inteiros consecutivos é sempre múltipla de $5$,
        mas que a \omterm{def:g1:addition:def}{soma} de \emph{quatro} inteiros consecutivos
        \emph{nunca} é múltipla de $4$. (Duas continhas --- e uma moral
        sobre o que a álgebra faz com as palavras ``sempre'' e
        ``nunca''.)
\end{enumerate}
\end{problem}
