\chapter{Fonctions linéaires et affines}\label{ch:g9:linfunc}

«~Trois kilogrammes coûtent trois fois plus cher qu'un~»~: c'est la
\omterm{def:g9:linfunc:linear}{proportionnalité}, la façon la plus simple dont deux quantités peuvent
être liées. Les \omterm{def:g9:linfunc:linear}{fonctions linéaires} sont la \omterm{def:g9:linfunc:linear}{proportionnalité} dans le
langage des fonctions~; les \omterm{def:g9:linfunc:affine}{fonctions affines} ajoutent un montant de
départ fixe. Leurs graphiques sont des droites, et lire ces droites est
la compétence que ce chapitre construit.

\section{Proportionnalité et fonctions linéaires}

\begin{definition}[Fonction linéaire]\label{def:g9:linfunc:linear}
Une \emph{fonction linéaire}\index{fonction linéaire} multiplie tout
nombre par un nombre fixe $a$~:
\[
f(x) = a x .
\]
Le nombre $a$ est le \emph{coefficient} de $f$. Deux quantités sont
\emph{proportionnelles}\index{proportionnalité} exactement lorsque l'une
est une fonction linéaire de l'autre.
\end{definition}

\begin{example}\label{ex:g9:linfunc:linear}
Si les pommes coûtent $3$ l'unité de kilogramme, le prix de $x$
kilogrammes est $f(x) = 3x$~: en acheter deux fois plus coûte deux fois
plus, et le prix de $2.5$ kg est $f(2.5) = 7.5$.

Réciproquement, si une \omterm{def:g9:linfunc:linear}{fonction linéaire} satisfait $f(4) = 10$, alors
$a = \frac{10}{4} = 2.5$ et $f(x) = 2.5x$~: \emph{une seule valeur
détermine complètement une \omterm{def:g9:linfunc:linear}{fonction linéaire}.}
\end{example}

\begin{proposition}[Graphique d'une fonction linéaire]\label{prop:g9:linfunc:lineargraph}
Le graphique de $f(x) = ax$ est une droite \emph{passant par l'origine}.
Réciproquement, toute droite non verticale passant par l'origine est le
graphique d'une \omterm{def:g9:linfunc:linear}{fonction linéaire}.
\end{proposition}
\admitted

\section{Fonctions affines}

\begin{definition}[Fonction affine]\label{def:g9:linfunc:affine}
Une \emph{fonction affine}\index{fonction affine} a la forme
\[
f(x) = a x + b ,
\]
où $a$ et $b$ sont des nombres fixes. Les \omterm{def:g9:linfunc:linear}{fonctions linéaires} sont le cas
particulier $b = 0$~; les fonctions constantes le cas $a = 0$.
\end{definition}

\begin{example}
Une salle de sport facture des frais d'inscription de $20$, puis $5$ par
visite~: le coût total de $x$ visites est $f(x) = 5x + 20$. Le coût
n'est \emph{pas} proportionnel au nombre de visites (10 visites ne
coûtent pas deux fois plus que 5), mais chaque visite supplémentaire
ajoute le même montant, $5$.
\end{example}

\begin{proposition}[Graphique d'une fonction affine]\label{prop:g9:linfunc:affinegraph}
Le graphique de $f(x) = ax + b$ est une droite~:
\begin{itemize}
  \item $b = f(0)$ est l'\emph{ordonnée à l'origine}~: la droite coupe
        l'axe vertical en $(0, b)$~;
  \item $a$ est le \emph{coefficient directeur}\index{coefficient
        directeur}~: avancer d'une unité vers la droite fait monter de
        $a$ unités (descendre si $a < 0$)~; la fonction est croissante
        lorsque $a > 0$, décroissante lorsque $a < 0$~;
  \item pour deux entrées $u \neq v$,
        $a = \dfrac{f(v) - f(u)}{v - u}$.
\end{itemize}
\end{proposition}

\begin{proof}[Preuve du dernier point]
$f(v) - f(u) = (av + b) - (au + b) = a(v - u)$~; diviser par $v - u$. Le
\omterm{def:g3:division:remainder}{reste} est admis à ce niveau (le \omterm{def:g6:measure:volume}{volume} suivant donne un traitement
complet des droites et de leurs \omterm{def:g8:equations:def}{équations}).
\end{proof}

\begin{method}[Trouver une fonction affine à partir de deux valeurs]\label{met:g9:linfunc:findaffine}
Étant donnés $f(u)$ et $f(v)$ avec $u \neq v$~:
\begin{enumerate}
  \item calculer le \omterm{prop:g9:linfunc:affinegraph}{coefficient directeur}
        $a = \dfrac{f(v) - f(u)}{v - u}$~;
  \item substituer l'une des deux valeurs connues dans $f(x) = ax + b$
        pour trouver $b$~;
  \item vérifier avec l'autre valeur.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:linfunc:findaffine}
Trouver la \omterm{def:g9:linfunc:affine}{fonction affine} avec $f(2) = 7$ et $f(5) = 16$.
\[
a = \frac{16 - 7}{5 - 2} = \frac93 = 3 ,
\]
puis $f(2) = 3 \times 2 + b = 7$ donne $b = 1$~: $f(x) = 3x + 1$.
Vérification~: $f(5) = 15 + 1 = 16$.
\end{example}

\section{Pourcentages}

\begin{proposition}[Variation en pourcentage]\label{prop:g9:linfunc:percent}
Augmenter une quantité de $t\,\%$ la multiplie par
$1 + \dfrac{t}{100}$~; la diminuer de $t\,\%$ la multiplie par
$1 - \dfrac{t}{100}$. Les variations en pourcentage sont des \omterm{def:g9:linfunc:linear}{fonctions
linéaires}.
\end{proposition}

\begin{proof}
Augmenter $x$ de $t\,\%$ signifie ajouter $\frac{t}{100}\,x$~:
\[
x + \frac{t}{100}\,x = \left(1 + \frac{t}{100}\right) x .
\]
Même calcul avec un signe moins pour une diminution.
\end{proof}

\begin{example}\label{ex:g9:linfunc:percent}
Un prix de $80$ augmente de $15\,\%$~: nouveau prix
$80 \times 1.15 = 92$. Un prix de $60$ diminue de $30\,\%$~:
$60 \times 0.7 = 42$.

\emph{Enchaîner des variations multiplie les facteurs.} Une hausse de
$20\,\%$ suivie d'une baisse de $20\,\%$ donne
$1.2 \times 0.8 = 0.96$~: une \emph{baisse} de $4\,\%$ au total, pas un
retour au départ~!
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:1}
Parmi ces fonctions, lesquelles sont linéaires~? affines~? ni l'un ni
l'autre~?
\[
f(x) = 4x, \qquad
g(x) = 3x - 5, \qquad
h(x) = x^2, \qquad
k(x) = -\tfrac x2, \qquad
m(x) = 7 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:2}
Soit $f(x) = 2x - 3$. Calculer $f(0)$, $f(4)$, $f(-1)$, et trouver le
nombre $x$ tel que $f(x) = 9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:3}
Une \omterm{def:g9:linfunc:linear}{fonction linéaire} satisfait $f(6) = 15$. Trouver son coefficient,
puis calculer $f(10)$ et $f(-2)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:4}
Tracer sur le même repère les graphiques de $f(x) = 2x$,
$g(x) = 2x + 3$ et $h(x) = -x + 4$. Que peut-on dire des graphiques de
$f$ et $g$~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:linfunc:5}
Trouver la \omterm{def:g9:linfunc:affine}{fonction affine} telle que $f(1) = 5$ et $f(3) = 11$~; puis
celle telle que $g(0) = 4$ et $g(2) = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:6}
Calculer~: le prix après une hausse de $25\,\%$ sur $120$~; le prix après
une remise de $40\,\%$ sur $65$~; le prix d'origine si un article coûte
$69$ après une remise de $25\,\%$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:7}
Un forfait téléphonique A coûte $10$ par mois plus $0.05$ par minute
d'appels~; le forfait B coûte $25$ par mois avec appels illimités.
\begin{enumerate}
  \item Exprimer le coût mensuel du forfait A en fonction du nombre $x$
        de minutes.
  \item À partir de combien de minutes par mois le forfait B est-il
        moins cher~?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:8}
Une population de $2000$ bactéries augmente de $10\,\%$ chaque heure.
\begin{enumerate}
  \item Combien de bactéries y a-t-il après une heure~? Après deux
        heures~?
  \item Expliquer pourquoi la réponse après deux heures n'est pas
        $2400$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:linfunc:9}
Le graphique d'une \omterm{def:g9:linfunc:affine}{fonction affine} passe par les points $(2, 3)$ et
$(6, 1)$.
\begin{enumerate}
  \item Calculer son \omterm{prop:g9:linfunc:affinegraph}{coefficient directeur}. La fonction est-elle
        croissante ou décroissante~?
  \item Donner son expression, et calculer l'entrée pour laquelle la
        sortie est $0$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:linfunc:10}
Un magasin augmente d'abord un prix de $t\,\%$, puis baisse le nouveau
prix de $t\,\%$.
\begin{enumerate}
  \item Montrer que le prix final égale l'original multiplié par
        $1 - \dfrac{t^2}{10000}$.
  \item En déduire que le prix final est toujours \emph{plus bas} que
        l'original (pour $0 < t \leq 100$), et trouver la variation en
        pourcentage globale pour $t = 10$.
\end{enumerate}
\end{exercise}

\section{Problème~: les deux thermomètres}

\begin{problem}[{Devoir maison --- Celsius, Fahrenheit et les
fonctions affines de la vie courante~: une température se lit de la
même façon sur les deux échelles, et un coefficient directeur
constant est la signature de la droite}]
\label{pb:g9:linfunc:1}
Les Américains entendent «~$68$ degrés~» et s'habillent légèrement~;
les Européens l'entendent et attrapent un manteau. Les deux
thermomètres du monde sont reliés par une \omterm{def:g9:linfunc:affine}{fonction affine} --- et la
construire à partir de deux données est exactement le savoir-faire
de la \cref{met:g9:linfunc:findaffine}. Ce problème construit la
conversion, trouve l'étrange température où les deux échelles
s'accordent, lit ensuite les taxis, les grillons et les bougies avec
les mêmes lunettes, et s'achève sur le test qui reconnaît la
rectitude elle-même.

\medskip
\noindent\textbf{Partie I --- Construire la conversion.}
Deux données fixent l'échelle Fahrenheit~: l'eau gèle à
$32\,^\circ$F et bout à $212\,^\circ$F (soit à $0\,^\circ$C et
$100\,^\circ$C).
\begin{enumerate}
  \item Trouver la \omterm{def:g9:linfunc:affine}{fonction affine} $F = f(C)$ qui convertit les
        degrés Celsius en degrés Fahrenheit
        (\cref{met:g9:linfunc:findaffine} avec les valeurs
        $f(0) = 32$ et $f(100) = 212$).
  \item Convertir en degrés Fahrenheit~: une douce journée à
        $20\,^\circ$C~; les $37\,^\circ$C du corps humain~; un
        glacial $-10\,^\circ$C.
  \item Résoudre en $C$ pour construire la conversion \omterm{def:g8:fractions:inverse}{inverse}, et
        l'appliquer à $50\,^\circ$F puis à $98.6\,^\circ$F.
  \item $f$ est-elle une fonction \emph{linéaire}
        (\cref{def:g9:linfunc:linear})~? Vérifier les signes qui ne
        trompent pas~: que vaut $f(0)$, et doubler la lecture
        Celsius double-t-il la lecture Fahrenheit~? Conclure avec le
        mot juste (\cref{def:g9:linfunc:affine}).
  \item L'énigme classique~: existe-t-il une température qui se lit
        \emph{avec le même nombre} sur les deux échelles~? Résoudre
        $f(x) = x$, et décrire ce que la solution signifie sur la
        représentation graphique de $f$ (quelle droite le graphe
        coupe-t-il en ce point~?).
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Des \omterm{def:g9:linfunc:affine}{fonctions affines} en liberté.}
\begin{enumerate}[resume]
  \item Le taxi A facture $3$ euros plus $1.50$ par km~; le taxi B
        facture $2$ euros par km, sans prise en charge. Écrire les
        deux fonctions de prix et les comparer pour une course de
        $4$ km puis de $8$ km.
  \item Trouver la distance d'équilibre à laquelle les deux taxis
        coûtent la même chose, et décrire la situation sur un graphe
        (deux droites --- que se passe-t-il au point
        d'intersection, avant, après~?).
  \item Règle empirique des naturalistes~: un grillon stridule
        environ $N = 7T - 30$ fois par minute à la température
        $T\,^\circ$C. Combien de stridulations par une soirée à
        $20\,^\circ$C~? On compte $82$ stridulations en une minute~:
        quelle température le grillon annonce-t-il~?
  \item Un téléphone acheté $600$ euros perd $15$ euros de valeur
        par mois~: écrire la fonction de valeur $v(t)$, trouver
        quand le téléphone vaut $240$ euros, puis quand le modèle
        affirme qu'il ne vaut plus rien. La formule a-t-elle encore
        un sens au-delà de cette date~?
  \item Les variations en pourcentage sont des \omterm{def:g9:linfunc:linear}{fonctions
        linéaires}~: $+25\,\%$ est la multiplication par $1.25$, et
        $-20\,\%$ la multiplication par $0.8$
        (\cref{prop:g9:linfunc:percent}). Les composer~: que fait à
        un prix une hausse de $25\,\%$ suivie d'une baisse de
        $20\,\%$~? Expliquer le petit miracle, et comparer avec
        \cref{exo:g9:linfunc:10}.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Tout est dans le \omterm{prop:g9:linfunc:affinegraph}{coefficient
directeur}.}
\begin{enumerate}[resume]
  \item Une bougie mesure $20$ cm au moment où on l'allume et
        $17$ cm une demi-heure plus tard. En supposant une fonte
        affine, trouver $h(t)$ (hauteur en cm, $t$ en minutes), et
        prévoir l'instant où la bougie s'éteint.
  \item Deux cyclistes roulent sur la même route~: la distance du
        premier est $d_1(t) = 24t$ (en km après $t$ heures), celle
        du second $d_2(t) = 10 + 18t$. Lequel est parti avec de
        l'avance, lequel roule le plus vite, et à quel instant et à
        quel kilomètre le premier rattrape-t-il le second~?
  \item Soient $f(x) = 2x + 3$ et $g(x) = -0.5x + 7$. Calculer
        $f(x) + g(x)$ et vérifier que c'est encore une \omterm{def:g9:linfunc:affine}{fonction
        affine}. Quels sont, en général, le \omterm{prop:g9:linfunc:affinegraph}{coefficient directeur} et
        l'ordonnée à l'origine d'une \omterm{def:g1:addition:def}{somme}~?
  \item Discuter complètement l'\omterm{def:g8:equations:def}{équation} $mx + p = 0$~: combien de
        solutions lorsque $m \neq 0$ (et laquelle)~? Lorsque
        $m = 0$ et $p \neq 0$~? Lorsque $m = 0$ et $p = 0$~? (Chaque
        cas a son histoire graphique~: où une droite de \omterm{prop:g9:linfunc:affinegraph}{coefficient
        directeur} $m$ coupe-t-elle l'axe horizontal~?)
  \item Le test d'alignement~: les points $(1, 5)$, $(3, 9)$,
        $(7, 17)$ sont-ils sur une même droite~? Et $(1, 5)$,
        $(3, 9)$, $(6, 16)$~? Calculer les taux d'accroissement
        entre les couples et trancher. Énoncer la morale~: un
        \emph{taux d'accroissement constant} est la signature des
        \omterm{def:g9:linfunc:affine}{fonctions affines} --- les courbes, elles, changent de taux,
        et mesurer \emph{cela} est une histoire pour le \omterm{def:g6:measure:volume}{volume}
        suivant.
\end{enumerate}
\end{problem}
