\chapter{Le théorème de Thalès}\label{ch:g9:thales}

Comment mesurer la hauteur d'une \omterm{def:g8:solids:def}{pyramide} sans y grimper~? Thalès de
Milet a comparé son ombre avec l'ombre d'un bâton. Son théorème --- des
\omterm{def:g6:lines:perp}{droites parallèles} découpent des \omterm{def:g6:lines:objects}{segments} en parts \omterm{def:g9:linfunc:linear}{proportionnelles} ---
est le cœur mathématique des maquettes, des cartes et des agrandissements,
et l'un des plus anciens théorèmes portant un nom.

\section{Le théorème}

\begin{theorem}[Thalès]\label{thm:g9:thales:direct}
Soient deux droites se rencontrant en un point $A$. Prendre $M$ sur la
première droite et $N$ sur la seconde, avec $B$ plus loin sur la
première et $C$ plus loin sur la seconde. Si les droites $(MN)$ et
$(BC)$ sont \emph{\omterm{def:g6:lines:perp}{parallèles}}, alors les triangles $AMN$ et $ABC$ ont
des côtés proportionnels~:
\[
\frac{AM}{AB} = \frac{AN}{AC} = \frac{MN}{BC}.
\]
\end{theorem}
\admitted

\begin{method}[Calculer une longueur avec Thalès]\label{met:g9:thales:compute}
\begin{enumerate}
  \item Identifier le point $A$ où les deux droites se croisent et les
        deux \omterm{def:g6:lines:perp}{droites parallèles}~; nommer les deux triangles~;
  \item écrire les trois rapports égaux, toujours «~petit triangle sur
        grand triangle~»~;
  \item garder l'égalité des deux rapports impliquant les trois
        longueurs connues et l'inconnue~;
  \item résoudre la proportion résultante par \omterm{def:g2:mult:def}{produits} en croix.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:thales:compute}
Dans la configuration de gauche ci-dessus, supposons $AM = 3$, $AB = 5$,
$AN = 2.4$ et $MN = 3.3$, et calculons $AC$ et $BC$. Thalès donne
\[
\frac{AM}{AB} = \frac{AN}{AC} = \frac{MN}{BC},
\qquad\text{c'est-à-dire}\qquad
\frac35 = \frac{2.4}{AC} = \frac{3.3}{BC}.
\]
De $\frac35 = \frac{2.4}{AC}$, en croisant les \omterm{def:g2:mult:def}{produits}~:
$3 \times AC = 5 \times 2.4 = 12$, donc $AC = 4$. De
$\frac35 = \frac{3.3}{BC}$~: $3 \times BC = 16.5$, donc $BC = 5.5$.
\end{example}

\begin{example}[Thalès dans la vie réelle]\label{ex:g9:thales:stick}
Un bâton vertical de hauteur $1$ m projette une ombre de $1.5$ m,
tandis qu'un arbre projette une ombre de $12$ m au même moment. Les
\omterm{def:g3:shapes:shapes}{rayons} du Soleil sont \omterm{def:g6:lines:perp}{parallèles}, donc les triangles bâton-ombre et
arbre-ombre sont en configuration de Thalès~:
\[
\frac{\text{hauteur de l'arbre}}{1} = \frac{12}{1.5},
\qquad\text{donc l'arbre mesure } 8 \text{ m de haut.}
\]
\end{example}

\section{La réciproque}

\begin{theorem}[Réciproque de Thalès]\label{thm:g9:thales:converse}
Avec les points placés comme en \cref{thm:g9:thales:direct} ($M$ sur
$(AB)$, $N$ sur $(AC)$, dans le même ordre sur les deux droites)~: si
\[
\frac{AM}{AB} = \frac{AN}{AC},
\]
alors les droites $(MN)$ et $(BC)$ sont \omterm{def:g6:lines:perp}{parallèles}.
\end{theorem}
\admitted

\begin{method}[Prouver ou infirmer le parallélisme]\label{met:g9:thales:converse}
\begin{enumerate}
  \item Calculer séparément les deux rapports $\dfrac{AM}{AB}$ et
        $\dfrac{AN}{AC}$ (en \omterm{def:g9:fractions:fraction}{fractions}, pas en \omterm{def:g6:decimals:rounding}{arrondis})~;
  \item s'ils sont égaux \emph{et} que les points sont dans le même
        ordre sur les deux droites, les droites sont \omterm{def:g6:lines:perp}{parallèles}
        (réciproque de Thalès)~;
  \item s'ils diffèrent, les droites ne sont \emph{pas} \omterm{def:g6:lines:perp}{parallèles} ---
        car si elles l'étaient, le théorème de Thalès forcerait les
        rapports à être égaux.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:thales:converse}
Sur la droite 1, $AM = 4$ et $AB = 10$~; sur la droite 2, $AN = 6$ et
$AC = 15$. Alors
\[
\frac{AM}{AB} = \frac{4}{10} = \frac25,
\qquad
\frac{AN}{AC} = \frac{6}{15} = \frac25 :
\]
rapports égaux, même ordre~: $(MN) \parallel (BC)$.

Avec $AC = 14$ à la place~: $\frac{6}{14} = \frac37 \neq \frac25$,
donc les droites ne sont pas \omterm{def:g6:lines:perp}{parallèles}.
\end{example}

\section{Agrandissement et réduction}

\begin{definition}[Homothétie d'une figure]\label{def:g9:thales:scaling}
Agrandir ou réduire une figure par le \emph{facteur
d'échelle}\index{facteur d'échelle} $k > 0$ signifie multiplier
\emph{toutes} ses longueurs par $k$~: un \emph{agrandissement} lorsque
$k > 1$, une \emph{réduction} lorsque $k < 1$. Dans la configuration de
Thalès, le triangle $AMN$ est la réduction de $ABC$ par le facteur
$k = \frac{AM}{AB}$.
\end{definition}

\begin{proposition}[Effet sur les angles et les aires]\label{prop:g9:thales:scaling}
L'homothétie de facteur $k$ conserve les angles et la forme des figures,
multiplie chaque longueur par $k$, et multiplie chaque \emph{\omterm{def:g6:measure:area}{aire}} par
$k^2$.
\end{proposition}
\admitted

\begin{omfigure}
\begin{tikzpicture}[scale=0.72]
  \draw[thick, omDef, fill=omDef!8] (0,0) rectangle (2,1.2);
  \draw[thick, omThm, fill=omThm!8] (3.4,0) rectangle (7.4,2.4);
  \node[font=\small, omDef] at (1,-0.45) {$2 \times 1.2$};
  \node[font=\small, omThm] at (5.4,-0.45) {$4 \times 2.4$};
  \node[font=\small] at (1,0.6) {aire $2.4$};
  \node[font=\small] at (5.4,1.2) {aire $9.6$};
\end{tikzpicture}

{\small Doubler les longueurs ($k = 2$) multiplie l'\omterm{def:g6:measure:area}{aire} par
$k^2 = 4$~: quatre copies du petit rectangle pavent le grand.}
\end{omfigure}

\begin{example}
Une photo de $10 \times 15$ cm est agrandie avec le \omterm{def:g9:thales:scaling}{facteur d'échelle}
$k = 3$~: le tirage mesure $30 \times 45$ cm. Son \omterm{def:g6:measure:area}{aire} passe de
$150$ cm$^2$ à $150 \times 9 = 1350$ cm$^2$ --- $9$ fois plus grande,
pas $3$.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g9:thales:1}
Les droites $(MN)$ et $(BC)$ sont \omterm{def:g6:lines:perp}{parallèles}, avec $M$ sur $[AB]$ et $N$
sur $[AC]$~; $AM = 2$, $AB = 6$, $AN = 3$, $BC = 9$. Calculer $AC$ et
$MN$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:2}
Même configuration~: $AM = 5$, $MB = 3$ (attention~: $MB$, pas $AB$~!),
$AN = 4$. Calculer $AC$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:3}
En configuration papillon, $A$ est entre $M$ et $B$ et entre $N$ et $C$,
avec $(MN) \parallel (BC)$~; $AM = 3$, $AB = 7.5$, $AN = 2$, $MN = 2.6$.
Calculer $AC$ et $BC$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:thales:4}
$M$ est sur $[AB]$ avec $AM = 6$, $AB = 8$~; $N$ est sur $[AC]$ avec
$AN = 9$, $AC = 12$. Les droites $(MN)$ et $(BC)$ sont-elles
\omterm{def:g6:lines:perp}{parallèles}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:5}
Comme ci-dessus avec $AM = 4$, $AB = 6$, $AN = 5$, $AC = 8$. Les droites
$(MN)$ et $(BC)$ sont-elles \omterm{def:g6:lines:perp}{parallèles}~? Justifier soigneusement.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:6}
Une personne de $1.8$ m de haut se tient à $2$ m de la base d'un lampadaire~;
son ombre mesure $3$ m. Dessiner la configuration de Thalès formée par le
lampadaire, la personne et le bout de l'ombre, et calculer la hauteur du
lampadaire.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:7}
Une carte est à l'échelle $1 : 25\,000$ (les longueurs sur la carte sont
les longueurs réelles multipliées par $k = \frac{1}{25000}$).
\begin{enumerate}
  \item Deux villages sont distants de $6.8$ cm sur la carte. Quelle est
        la distance réelle, en km~?
  \item Une forêt a une \omterm{def:g6:measure:area}{aire} de $8$ cm$^2$ sur la carte. Quelle est son
        \omterm{def:g6:measure:area}{aire} réelle, en km$^2$~?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:thales:8}
Le triangle $ABC$ a $AB = 12$, $AC = 15$, $BC = 18$. Le point $M$ sur
$[AB]$ satisfait $AM = 8$, et la droite passant par $M$ \omterm{def:g6:lines:perp}{parallèle} à
$(BC)$ coupe $[AC]$ en $N$.
\begin{enumerate}
  \item Calculer $AN$ et $MN$.
  \item Quel est le \omterm{def:g9:thales:scaling}{facteur d'échelle} de $ABC$ vers $AMN$~? Comparer les
        \omterm{def:g6:measure:perimeter}{périmètres} des deux triangles.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:thales:9}
Soit $ABCD$ un trapèze avec $(AB) \parallel (CD)$, $AB = 4$ et $CD = 6$,
dont les diagonales $[AC]$ et $[BD]$ se rencontrent en $O$. En utilisant
Thalès dans la configuration papillon autour de $O$, calculer le rapport
$\dfrac{OA}{OC}$, et montrer que $O$ coupe les deux diagonales dans le
même rapport.
\end{exercise}

\section{Problème~: la règle non graduée, le sténopé et les trois
moyennes d'un trapèze}

\begin{problem}[{Devoir maison --- Thalès à l'œuvre~: partager un
segment quelconque en parts égales, mesurer le Soleil avec une boîte
à chaussures, et un trapèze où les trois moyennes célèbres se
rencontrent}]
\label{pb:g9:thales:1}
Le théorème de Thalès est la mathématique des \emph{\omterm{def:g3:shapes:shapes}{rayons}}~: \omterm{def:g3:shapes:shapes}{rayons}
du Soleil, \omterm{def:g3:shapes:shapes}{rayons} lumineux traversant un trou d'épingle, \omterm{def:g3:shapes:shapes}{rayons} de
crayon issus d'un \omterm{def:g5:solids:def}{sommet}. Ce problème l'utilise de trois façons ---
comme outil de construction (partager un \omterm{def:g6:lines:objects}{segment} de longueur
\emph{quelconque}, même $\sqrt2$, en parts parfaitement égales),
comme instrument de mesure (le \omterm{def:g4:geometry:circle}{diamètre} du Soleil, avec une boîte à
chaussures et une pièce de monnaie), et comme loupe braquée sur le
trapèze de l'\cref{exo:g9:thales:9}, où les moyennes arithmétique,
géométrique et harmonique de cette série se retrouvent toutes dans
une même figure.

\medskip
\noindent\textbf{Partie I --- Partager à la règle non graduée.}
\begin{enumerate}
  \item Tracer un \omterm{def:g6:lines:objects}{segment} $[AB]$ de $7$ cm. Pour le partager en
        trois parts égales au compas et à la règle non graduée~:
        tracer une demi-droite quelconque issue de $A$ (ne passant
        pas par $B$)~; y reporter au compas trois longueurs égales,
        ce qui donne les points $P_1$, $P_2$, $P_3$~; joindre $P_3$
        à $B$~; tracer les \omterm{def:g6:lines:perp}{parallèles} à $(P_3 B)$ passant par $P_1$
        et $P_2$. Effectuer la construction et marquer les points où
        les \omterm{def:g6:lines:perp}{parallèles} coupent $[AB]$.
  \item Justifier par \cref{thm:g9:thales:direct} que les deux
        points marqués partagent $[AB]$ exactement en ses tiers.
  \item Adapter la méthode pour partager un \omterm{def:g6:lines:objects}{segment} en $5$ parts
        égales, puis pour construire les $\frac35$ d'un \omterm{def:g6:lines:objects}{segment}
        donné.
  \item Construire le point $M$ de $[AB]$ tel que
        $\frac{AM}{MB} = \frac23$ (c'est-à-dire $M$ aux $\frac25$
        du trajet depuis $A$). Combien de reports égaux sur la
        demi-droite, et quel point faut-il joindre à $B$~?
  \item Pourquoi cette construction vaut-elle mieux qu'une mesure à
        la règle graduée~? Considérer un \omterm{def:g6:lines:objects}{segment} de longueur
        $\sqrt2$ (la diagonale d'un carré unité, irrationnelle
        d'après \cref{pb:g9:sqrt:1})~: que donnerait la mesure, et
        que donne Thalès~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Le sténopé.}
Percer une \omterm{def:g5:solids:def}{face} d'une boîte fermée avec une épingle~: sur la \omterm{def:g5:solids:def}{face}
opposée apparaît une image du monde, à l'envers. Un point de l'objet,
le trou et le point image sont alignés --- la lumière va en ligne
droite --- de sorte que l'objet (de hauteur $H$, à la distance $D$
devant le trou) et son image (de hauteur $h$, sur la paroi du fond, à
la profondeur $d$ derrière le trou) forment la configuration papillon
de Thalès.
\begin{enumerate}[resume]
  \item Établir la formule du sténopé $h = H \times \frac dD$ à
        partir de Thalès dans le papillon autour du trou.
  \item Un arbre de $12$ m se dresse à $30$ m d'une boîte à
        chaussures de $20$ cm de profondeur. Quelle est la hauteur
        de son image, et dans quel sens~?
  \item Pointons la boîte vers le Soleil~: à $d = 1$ m derrière le
        trou, l'image du Soleil est un disque d'environ $9$ mm de
        \omterm{def:g4:geometry:circle}{diamètre}. Sachant que la distance du Soleil vaut
        $D = 1.5 \times 10^{11}$ m, calculer son \omterm{def:g4:geometry:circle}{diamètre} en
        \omterm{def:g9:fractions:scientific}{notation scientifique} (la notation du
        \cref{pb:g9:fractions:1} à l'œuvre). Comparer avec la valeur
        réelle, $1.39 \times 10^9$ m.
  \item La Lune~: \omterm{def:g4:geometry:circle}{diamètre} $3.5 \times 10^6$ m, distance
        $3.8 \times 10^8$ m. Calculer le rapport
        $\frac{\text{diamètre}}{\text{distance}}$ pour la Lune puis
        pour le Soleil. Quelle heureuse coïncidence les deux nombres
        révèlent-ils --- et quel spectacle rendent-ils possible~?
  \item En une ou deux phrases~: pourquoi l'image du sténopé est-elle
        à l'envers, et pourquoi agrandir le trou rend-il l'image plus
        lumineuse mais plus floue~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Le trapèze des trois moyennes.}
$ABCD$ est le trapèze de l'\cref{exo:g9:thales:9}~:
$(AB) \parallel (CD)$, $AB = 4$, $CD = 6$, les diagonales se coupant
en $O$ avec $\frac{OA}{OC} = \frac{OB}{OD} = \frac{4}{6} =
\frac23$.
\begin{enumerate}[resume]
  \item Entraînement au papillon~: deux droites se coupent en $O$
        avec $(MN) \parallel (PQ)$ en position papillon~; $OM = 3$,
        $OP = 7.5$, $MN = 4$ et $OQ = 6$. Calculer $PQ$ et $ON$.
  \item Expliquer comment Thalès, appliqué avec le rapport
        $\frac12$, contient le théorème des milieux du
        \cref{thm:g8:midpoints:direct} comme cas particulier.
  \item Tracer la \omterm{def:g6:lines:perp}{parallèle} aux bases passant par $O$~; elle coupe
        $[AD]$ en $E$. En travaillant dans le triangle $ACD$
        (remarquer que $\frac{AO}{AC} = \frac25$), calculer $EO$.
  \item Par le calcul \omterm{def:g7:central:def}{symétrique} dans le triangle $BDC$, le morceau
        $OF$ (avec $F$ sur $[BC]$) a la même longueur. En conclure
        que
        \[
        EF = 2 \times \frac{4 \times 6}{4 + 6} = 4.8 :
        \]
        la \omterm{def:g6:lines:perp}{parallèle} passant par le point de croisement des
        diagonales mesure la \emph{\omterm{pb:g8:speed:1}{moyenne harmonique}} des bases ---
        la moyenne des aller-retours du~\cref{pb:g8:speed:1}, qui
        réapparaît en géométrie pure.
  \item Le grand final~: calculer les trois moyennes classiques des
        bases $4$ et $6$ --- arithmétique $\frac{4+6}{2}$,
        géométrique $\sqrt{4 \times 6}$, harmonique
        $\frac{2 \times 4 \times 6}{4 + 6}$ --- et les ranger.
        Chacune habite le trapèze~: la moyenne arithmétique est la
        ligne des milieux joignant les milieux des côtés obliques
        (\cref{pb:g7:areas:1}), la \omterm{pb:g8:speed:1}{moyenne harmonique} est notre
        \omterm{def:g6:lines:objects}{segment} $EF$, et la \omterm{pb:g8:cosine:1}{moyenne géométrique} est la \omterm{def:g6:lines:perp}{parallèle} qui
        découpe le trapèze en deux trapèzes \emph{semblables}
        (vérifier sa valeur à la calculatrice~; la démonstration est
        un joli défi supplémentaire). Où l'inégalité entre les trois
        a-t-elle été démontrée dans ce livre
        (\cref{pb:g8:cosine:1}, \cref{pb:g8:speed:1})~?
\end{enumerate}
\end{problem}
