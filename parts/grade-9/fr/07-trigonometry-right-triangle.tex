\chapter{Trigonométrie dans le triangle rectangle}\label{ch:g9:trig}

Dans un \omterm{def:g6:shapes:triangles}{triangle rectangle}, une fois qu'on connaît un angle aigu, la
\emph{forme} du triangle est fixée --- seule sa taille peut varier. Les
rapports de ses côtés ne dépendent donc que de l'angle~: ces rapports
sont le \omterm{def:g9:trig:ratios}{cosinus}, le \omterm{def:g9:trig:ratios}{sinus} et la \omterm{def:g9:trig:ratios}{tangente}. Avec le théorème de Pythagore,
ils permettent de calculer chaque côté et chaque angle d'un \omterm{def:g6:shapes:triangles}{triangle
rectangle} à partir de très peu d'informations.

\section{Le théorème de Pythagore, encore}

\begin{theorem}[Pythagore]\label{thm:g9:trig:pythagoras}
Dans un triangle $ABC$ rectangle en $A$, le carré de l'hypoténuse est la
\omterm{def:g1:addition:def}{somme} des carrés des deux côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit}~:
\[
BC^2 = AB^2 + AC^2 .
\]
Réciproquement, si $BC^2 = AB^2 + AC^2$ dans un triangle, alors le
triangle est rectangle en $A$.
\end{theorem}
\admitted

\begin{example}\label{ex:g9:trig:pythagoras}
Un \omterm{def:g6:shapes:triangles}{triangle rectangle} a pour côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit} $AB = 5$ et
$AC = 12$. Alors
\[
BC^2 = 5^2 + 12^2 = 25 + 144 = 169,
\qquad\text{donc}\quad BC = \sqrt{169} = 13 .
\]
Pour trouver un \emph{côté de l'\omterm{def:g3:shapes:rightangle}{angle droit}}~: si $BC = 10$ et
$AB = 6$, alors $AC^2 = BC^2 - AB^2 = 100 - 36 = 64$, donc $AC = 8$.
\end{example}

\section{Cosinus, sinus, tangente}

Dans un \omterm{def:g6:shapes:triangles}{triangle rectangle}, par rapport à un angle aigu $\theta$, les
trois côtés ont des noms~: l'\emph{hypoténuse} (opposée à l'\omterm{def:g3:shapes:rightangle}{angle droit},
le plus long côté), le côté \emph{opposé} à $\theta$, et le côté
\emph{adjacent} à $\theta$ (le côté de l'\omterm{def:g3:shapes:rightangle}{angle droit} touchant $\theta$).

\begin{definition}[Rapports trigonométriques]\label{def:g9:trig:ratios}
Pour un angle aigu $\theta$ d'un \omterm{def:g6:shapes:triangles}{triangle rectangle}~:
\[
\cos\theta = \frac{\text{adjacent}}{\text{hypoténuse}},
\qquad
\sin\theta = \frac{\text{opposé}}{\text{hypoténuse}},
\qquad
\tan\theta = \frac{\text{opposé}}{\text{adjacent}} .
\]
Ces rapports\index{cosinus}\index{sinus}\index{tangente} ne dépendent
que de $\theta$, pas de la taille du triangle (tous les \omterm{def:g6:shapes:triangles}{triangles
rectangles} de même angle aigu sont des homothéties les uns des autres,
par Thalès).
\end{definition}

\begin{proposition}[Premières propriétés]\label{prop:g9:trig:properties}
Pour tout angle aigu $\theta$~:
\begin{enumerate}
  \item $0 < \cos\theta < 1$ et $0 < \sin\theta < 1$ (un côté de l'\omterm{def:g3:shapes:rightangle}{angle
        droit} est plus court que l'hypoténuse)~;
  \item $\cos^2\theta + \sin^2\theta = 1$~;
  \item $\tan\theta = \dfrac{\sin\theta}{\cos\theta}$.
\end{enumerate}
\end{proposition}

\begin{proof}
Noter $a$ le côté adjacent, $o$ le côté opposé, $h$ l'hypoténuse d'un
\omterm{def:g6:shapes:triangles}{triangle rectangle} d'angle $\theta$.

1. $0 < a < h$ et $0 < o < h$, donc les deux rapports sont strictement
entre $0$ et $1$.

2. Par Pythagore, $a^2 + o^2 = h^2$. Diviser tout par $h^2$~:
\[
\cos^2\theta + \sin^2\theta
= \frac{a^2}{h^2} + \frac{o^2}{h^2}
= \frac{a^2 + o^2}{h^2} = 1 .
\]

3. $\dfrac{\sin\theta}{\cos\theta} = \dfrac{o/h}{a/h} = \dfrac oa
= \tan\theta$.
\end{proof}

\begin{method}[Trouver un côté]\label{met:g9:trig:side}
Pour calculer un côté inconnu lorsqu'un côté et un angle aigu sont
connus~:
\begin{enumerate}
  \item marquer l'angle connu et étiqueter les trois côtés (hypoténuse,
        opposé, adjacent) \emph{depuis cet angle}~;
  \item choisir le rapport (cos, sin ou tan) qui implique le côté connu
        et le côté cherché~;
  \item écrire l'\omterm{def:g8:equations:def}{équation} et la résoudre~;
  \item contrôle de bon sens~: l'hypoténuse doit sortir la plus longue.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g9:trig:side}
Dans un \omterm{def:g6:shapes:triangles}{triangle rectangle}, l'hypoténuse mesure $8$ et un angle est
$35^\circ$~; calculer le côté qui lui est opposé. Le rapport impliquant
le côté opposé et l'hypoténuse est le \omterm{def:g9:trig:ratios}{sinus}~:
\[
\sin 35^\circ = \frac{\text{opposé}}{8}
\quad\Longrightarrow\quad
\text{opposé} = 8 \sin 35^\circ \approx 8 \times 0.574 \approx 4.6 .
\]
Pour le côté adjacent, utiliser le \omterm{def:g9:trig:ratios}{cosinus}~:
$8\cos 35^\circ \approx 8 \times 0.819 \approx 6.6$. Vérifier avec
Pythagore~: $4.6^2 + 6.6^2 \approx 21 + 43 \approx 64 = 8^2$.
\end{example}

\begin{method}[Trouver un angle]\label{met:g9:trig:angle}
Lorsque deux côtés sont connus, calculer le rapport qu'ils forment,
l'identifier comme \omterm{def:g9:trig:ratios}{cosinus}, \omterm{def:g9:trig:ratios}{sinus} ou \omterm{def:g9:trig:ratios}{tangente} de l'angle inconnu, et
appliquer la fonction \omterm{def:g8:fractions:inverse}{inverse} de la calculatrice ($\cos^{-1}$,
$\sin^{-1}$ ou $\tan^{-1}$) au rapport.
\end{method}

\begin{example}\label{ex:g9:trig:angle}
Une échelle de $5$ m s'appuie contre un mur, son pied à $1.4$ m du mur.
L'angle $\theta$ entre l'échelle et le sol satisfait
\[
\cos\theta = \frac{1.4}{5} = 0.28,
\qquad\text{donc}\quad
\theta = \cos^{-1}(0.28) \approx 73.7^\circ .
\]
\end{example}

\begin{example}[Angles remarquables]\label{ex:g9:trig:special}
Deux triangles donnent des valeurs exactes. La \omterm{def:g3:division:half}{moitié} d'un carré de côté
$1$ (un \omterm{def:g6:shapes:triangles}{triangle rectangle} \omterm{def:g6:shapes:triangles}{isocèle}) a pour hypoténuse $\sqrt2$ et des
angles de $45^\circ$~:
\[
\cos 45^\circ = \sin 45^\circ = \frac{1}{\sqrt2} = \frac{\sqrt2}{2},
\qquad \tan 45^\circ = 1 .
\]
La \omterm{def:g3:division:half}{moitié} d'un \omterm{def:g6:shapes:triangles}{triangle équilatéral} de côté $1$ donne les angles
$30^\circ$ et $60^\circ$~:
\[
\cos 60^\circ = \sin 30^\circ = \frac12,
\qquad
\cos 30^\circ = \sin 60^\circ = \frac{\sqrt3}{2}.
\]
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g9:trig:1}
Un \omterm{def:g6:shapes:triangles}{triangle rectangle} a pour côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit} $9$ et $12$.
Calculer son hypoténuse. Un autre a pour hypoténuse $17$ et un côté de
l'\omterm{def:g3:shapes:rightangle}{angle droit} $8$~: calculer l'autre côté.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:2}
Un triangle a pour côtés $7$, $24$ et $25$. Est-il rectangle~? Même
question pour les côtés $5$, $6$ et $8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:3}
Dans un triangle $DEF$ rectangle en $D$, avec l'angle en $E$ noté
$\theta$~: quel côté est l'hypoténuse~? Quel côté est opposé à
$\theta$~? Écrire $\cos\theta$, $\sin\theta$ et $\tan\theta$ comme
rapports des côtés $DE$, $DF$, $EF$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:4}
Dans un \omterm{def:g6:shapes:triangles}{triangle rectangle}, l'hypoténuse mesure $10$ et un angle aigu est
$28^\circ$. Calculer les deux côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit} (\omterm{def:g4:numbers:round}{arrondir} au
dixième~; $\cos 28^\circ \approx 0.883$, $\sin 28^\circ \approx 0.469$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g9:trig:5}
Dans un \omterm{def:g6:shapes:triangles}{triangle rectangle}, le côté adjacent à l'angle $\theta$ mesure
$6$ et le côté opposé mesure $4.5$. Calculer $\tan\theta$, puis $\theta$
(\omterm{def:g4:numbers:round}{arrondir} au degré~; $\tan^{-1}(0.75) \approx 36.9^\circ$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:6}
Un drone vole à une altitude de $120$ m. Depuis un observateur au sol, on
le voit sous un angle d'élévation de $32^\circ$. À quelle distance
horizontale de l'observateur se trouve le drone
($\tan 32^\circ \approx 0.625$)~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:7}
Une rampe d'accès doit monter de $1.2$ m avec un angle d'au plus
$5^\circ$ par rapport à l'horizontale. Quelle longueur minimale le long
de la pente la rampe doit-elle avoir ($\sin 5^\circ \approx 0.0872$)~?
Arrondir au dixième de mètre supérieur.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g9:trig:8}
En utilisant \cref{prop:g9:trig:properties}~: un angle aigu $\theta$
satisfait $\cos\theta = 0.6$. Calculer $\sin\theta$ sans trouver
$\theta$, puis $\tan\theta$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g9:trig:9}
Justifier les valeurs exactes de \cref{ex:g9:trig:special}~: dans le
\omterm{def:g6:shapes:triangles}{triangle rectangle} \omterm{def:g6:shapes:triangles}{isocèle} de côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit} $1$, calculer
l'hypoténuse~; dans le \omterm{def:g6:shapes:triangles}{triangle équilatéral} de côté $1$, calculer la
hauteur, et en déduire le \omterm{def:g9:trig:ratios}{cosinus} et le \omterm{def:g9:trig:ratios}{sinus} de $30^\circ$ et
$60^\circ$.
\end{exercise}

\section{Problème~: mesurer ce qu'on ne peut pas atteindre}

\begin{problem}[{Devoir maison --- la méthode des deux stations des
géomètres pour les hauteurs inaccessibles, les pentes de route, la
distance de l'horizon, et la course de la tangente vers l'infini}]
\label{pb:g9:trig:1}
Aucun mètre ruban n'atteint le \omterm{def:g5:solids:def}{sommet} d'une montagne, le faîte d'une
tour de l'autre côté d'une rivière, ni l'horizon en mer --- mais un
instrument de visée et les trois rapports de ce chapitre, si. La
pièce maîtresse de ce problème est la classique \emph{méthode des
deux stations} des géomètres, qui calcule une hauteur sans jamais
approcher son pied~; autour d'elle~: des panneaux routiers, des
échelles, des escaliers, la courbure de la Terre, et un premier
aperçu d'une fonction qui explose vers l'infini.

\medskip
\noindent\textbf{Partie I --- Une seule station.}
(Valeurs~: $\tan 35^\circ \approx 0.700$,
$\sin 35^\circ \approx 0.574$, $\cos 35^\circ \approx 0.819$,
$\tan 1^\circ \approx 0.0175$.)
\begin{enumerate}
  \item D'un point situé à $60$ m du pied d'un phare, on voit le
        \omterm{def:g5:solids:def}{sommet} sous un angle d'élévation de $35^\circ$ (mesuré
        depuis l'horizontale, à hauteur d'œil). À quelle hauteur
        au-dessus du niveau de l'œil se trouve le \omterm{def:g5:solids:def}{sommet}~?
  \item Un panneau routier annonce une pente de $12\,\%$~: la route
        s'élève de $12$ m par $100$ m parcourus à l'horizontale.
        Quel angle la route fait-elle avec l'horizontale~? Et quel
        pourcentage de pente aurait une route inclinée à
        $45^\circ$~?
  \item Les deux angles aigus d'un \omterm{def:g6:shapes:triangles}{triangle rectangle} sont
        complémentaires, et le côté opposé à l'un est le côté
        adjacent à l'autre. En déduire les identités du «~co~»~:
        \[
        \sin(90^\circ - \theta) = \cos\theta,
        \qquad
        \cos(90^\circ - \theta) = \sin\theta .
        \]
  \item Une échelle de $8$ m est appuyée à $60^\circ$ du sol. À
        l'aide des valeurs exactes de l'\cref{ex:g9:trig:special},
        donner la hauteur atteinte et la distance du pied au mur ---
        exactement, puis au centimètre près.
  \item Vérifier $\cos^2\theta + \sin^2\theta = 1$ numériquement
        pour $\theta = 35^\circ$ et exactement pour
        $\theta = 60^\circ$. Pourquoi cette identité
        (\cref{prop:g9:trig:properties}) est-elle une bonne habitude
        pour contrôler un travail à la calculatrice~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Deux stations.}
Le \omterm{def:g5:solids:def}{sommet} d'une montagne est visible, son pied inaccessible. Depuis
un point $A$, l'angle d'élévation du \omterm{def:g5:solids:def}{sommet} vaut $30^\circ$~; après
$200$ m de marche en ligne droite vers la montagne, jusqu'en $B$,
l'élévation devient $40^\circ$. Notons $h$ la hauteur du \omterm{def:g5:solids:def}{sommet}
au-dessus du niveau de l'œil et $x$ la distance horizontale de $B$ à
la verticale du \omterm{def:g5:solids:def}{sommet}. ($\tan 30^\circ \approx 0.577$,
$\tan 40^\circ \approx 0.839$.)
\begin{enumerate}[resume]
  \item Exprimer $\tan 40^\circ$ et $\tan 30^\circ$ à l'aide de $h$,
        $x$ et $x + 200$~: deux \omterm{def:g8:equations:def}{équations} à deux inconnues.
  \item Tirer de chaque \omterm{def:g8:equations:def}{équation} la distance horizontale en fonction
        de $h$, soustraire, et établir la formule des deux
        stations~:
        \[
        h = \frac{200}
        {\ \dfrac{1}{\tan 30^\circ} - \dfrac{1}{\tan 40^\circ}\ } .
        \]
        Calculer $h$ à la dizaine de mètres près.
  \item Calculer aussi $x$, et contrôler la vraisemblance des deux
        réponses avec la visée à $30^\circ$ depuis $A$.
  \item En une phrase~: pourquoi la méthode des deux stations
        est-elle indispensable --- quelle mesure unique, impossible
        ici, la méthode à une station de la question 1 aurait-elle
        exigée~?
  \item Une tour est visée sous $30^\circ$, puis sous $60^\circ$
        après $200$ m de marche vers elle. À l'aide de
        $\tan 30^\circ = \frac{1}{\sqrt3}$ et de
        $\tan 60^\circ = \sqrt3$, calculer la hauteur
        \emph{exactement} --- la réponse est un \omterm{def:g9:arith:divisor}{multiple} de
        $\sqrt3$ --- puis au mètre près.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Jusqu'à l'horizon et le long du
mur.}
\begin{enumerate}[resume]
  \item À quelle distance est l'horizon~? Debout, l'œil à $h$ mètres
        au-dessus d'une Terre sphérique de \omterm{def:g3:shapes:shapes}{rayon} $R$, la ligne de
        visée effleure la sphère~: la ligne de visée, le \omterm{def:g3:shapes:shapes}{rayon}
        aboutissant au point de contact et le \omterm{def:g3:shapes:shapes}{rayon} aboutissant à
        nos pieds forment un \omterm{def:g6:shapes:triangles}{triangle rectangle}. Montrer par
        Pythagore (\cref{thm:g9:trig:pythagoras}) que la distance
        $d$ jusqu'à l'horizon vérifie $d^2 = 2Rh + h^2 \approx 2Rh$,
        et calculer $d$ pour un œil à $1.80$ m
        ($R = 6.37 \times 10^6$ m).
  \item Même question depuis le \omterm{def:g5:solids:def}{sommet} d'une tour de $324$ m.
        (Règle empirique des marins~: l'horizon en kilomètres vaut
        environ $3.6\sqrt{h}$ avec $h$ en mètres --- contrôler les
        deux réponses avec elle.)
  \item Notre pouce, large d'environ $2$ cm, tenu à bout de bras, à
        environ $60$ cm de l'œil~: quel angle couvre-t-il~? La
        pleine Lune couvre environ $0.5^\circ$~: quelle portion du
        pouce cache la Lune tout entière~?
  \item Un escalier confortable a des contremarches de $17$ cm et
        des girons de $29$ cm. Sous quel angle monte-t-il~? Comparer
        avec les $30^\circ$ de l'\cref{ex:g9:trig:special}, puis avec
        une échelle de grenier inclinée à $60^\circ$~: quelle
        contremarche \emph{celle-ci} exigerait-elle sur un giron de
        $29$ cm~?
  \item La course de la \omterm{def:g9:trig:ratios}{tangente}~: calculer à la calculatrice
        $\tan 10^\circ$, $\tan 45^\circ$, $\tan 80^\circ$,
        $\tan 89^\circ$. Que se passe-t-il lorsque l'angle approche
        $90^\circ$, et pourquoi (penser au côté adjacent)~? Le mur
        est vertical, le rapport n'a plus de valeur, et le graphe
        file vers l'infini --- voilà notre première
        \emph{asymptote}, une créature étudiée de près dans le
        \omterm{def:g6:measure:volume}{volume} suivant.
\end{enumerate}
\end{problem}
