\chapter{Eerste breuken}\label{ch:g4:fractions}

De \omterm{def:g3:division:half}{helft} van een appel, een kwartier, driekwart van de klas: \omterm{def:g4:fractions:def}{breuken} geven
de stukken van dingen een naam. In dit hoofdstuk komt de schrijfwijze
$\frac{a}{b}$ aan de orde, lezen we \omterm{def:g4:fractions:def}{breuken} op plaatjes en op de
getallenlijn, en vergelijken we de eenvoudigste. (\omterm{def:g4:fractions:def}{Breuken} worden groot in
\cref{ch:g6:fractions}.)

\section{De stukken een naam geven}

\begin{definition}[Breuk]\label{def:g4:fractions:def}
Verdeel een geheel in gelijke delen en neem er een paar --- dat is een
\emph{breuk}\index{breuk}:
\[
\frac{3}{4}
\quad
\begin{array}{l}
3 \text{: hoeveel delen je neemt (de \emph{teller}\index{teller});}\\
4 \text{: in hoeveel gelijke delen het geheel is verdeeld (de
\emph{noemer}\index{noemer}).}
\end{array}
\]
$\frac12$ is een \emph{\omterm{def:g3:division:half}{helft}}, $\frac13$ een \emph{derde}, $\frac14$ een
\emph{\omterm{def:g3:division:half}{kwart}}, $\frac{1}{10}$ een \emph{tiende}.
\end{definition}

\begin{example}[Breuken van een groep]\label{ex:g4:fractions:collection}
$\frac13$ van $12$ knikkers: verdeel de $12$ knikkers over $3$ gelijke
groepjes van $4$; één groepje is $\frac13$, dus $\frac13$ van $12$ is $4$
--- en $\frac23$ van $12$ zijn twee groepjes, dus $8$.
\end{example}

\section{Breuken op de getallenlijn}

\begin{method}[$\frac{a}{b}$ plaatsen]\label{met:g4:fractions:line}
Verdeel elke eenheid van de lijn in $b$ gelijke stapjes; loop vanaf $0$
$a$ stapjes. Is $a$ groter dan $b$, dan kom je voorbij $1$: \omterm{def:g4:fractions:def}{breuken} kunnen
groter zijn dan één geheel!
\end{method}

$2$ en $\frac34$.}

\begin{example}[Hele getallen verstopt in breuken]\label{ex:g4:fractions:whole}
$\frac44 = 1$ (vier \omterm{def:g3:division:half}{kwarten} maken het geheel); $\frac82 = 4$;
$\frac{12}{3} = 4$. En $\frac74$ is $1 + \frac34$: één geheel en drie
\omterm{def:g3:division:half}{kwarten} erbij.
\end{example}

\section{Eenvoudige breuken vergelijken en optellen}

\begin{proposition}[Dezelfde noemer]\label{prop:g4:fractions:compare}
Bij dezelfde \omterm{def:g4:fractions:def}{noemer} zijn de stukken even groot --- dan hoef je ze alleen
te tellen:
\[
\frac{2}{8} < \frac{5}{8},
\qquad
\frac{3}{8} + \frac{4}{8} = \frac{7}{8}.
\]
Vergelijken met één geheel gaat ook makkelijk:
$\frac{5}{8} < 1 < \frac{9}{8}$ (minder, en dan meer, dan $8$ achtsten).
\end{proposition}

\begin{proof}[Waarom, op een plaatje]
Als je $3$ achtsten van een strook inkleurt en daarna nog $4$ achtsten van
dezelfde strook, is er in totaal $7$ achtsten ingekleurd.
\end{proof}

\begin{example}[Verschillende noemers, hetzelfde punt]\label{ex:g4:fractions:equal}
Op de getallenlijn komen $\frac12$, $\frac24$ en $\frac{5}{10}$ allemaal
op hetzelfde punt uit: het zijn drie namen voor hetzelfde getal. Elke \omterm{def:g3:division:half}{helft}
in twee stukken knippen geeft \omterm{def:g3:division:half}{kwarten}; in vijf stukken, tienden. (De
algemene regel staat in \cref{ch:g6:fractions}.)
\end{example}

\begin{example}[Kwartieren]\label{ex:g4:fractions:hour}
Een uur heeft $60$ minuten, dus een \omterm{def:g3:division:half}{kwart} van een uur is
$60 \div 4 = 15$ minuten, en driekwart uur is $45$ minuten. ``Half drie''
is $\frac12$ uur na twee uur.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:fractions:1}
Schrijf de \omterm{def:g4:fractions:def}{breuk} op die je ziet: een pizza in $6$ stukken gesneden waarvan
er $5$ over zijn; een chocoladereep van $8$ blokjes waarvan er $3$ zijn
opgegeten (de opgegeten \omterm{def:g4:fractions:def}{breuk}); een vlag in $3$ gelijke verticale banen
waarvan $1$ gekleurd is.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:2}
Teken een strook in $5$ gelijke delen en kleur $\frac35$ in. Kleur daarna
$\frac{2}{5}$ van een even grote strook in. Welke gekleurde stukken zijn
groter?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:3}
Reken uit: $\frac12$ van $18$; \quad $\frac14$ van $20$; \quad
$\frac13$ van $21$; \quad $\frac34$ van $20$ (drie groepjes van een
\omterm{def:g3:division:half}{kwart}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:4}
Zet op een getallenlijn die in derden verdeeld is: $\frac13$; \
$\frac33$; \ $\frac53$; \ $2$. Welke \omterm{def:g4:fractions:def}{breuk} komt precies op $2$ uit?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:5}
Neem over en vul in met $<$, $>$ of $=$:
\[
\frac38 \;?\; \frac68, \qquad
\frac55 \;?\; 1, \qquad
\frac74 \;?\; 1, \qquad
\frac12 \;?\; \frac24 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:6}
Reken uit: $\frac26 + \frac36$; \quad $\frac58 - \frac28$; \quad
$\frac14 + \frac24$; \quad $1 - \frac13$ (hoeveel derden maken $1$?).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:7}
Schrijf als een heel getal plus een \omterm{def:g4:fractions:def}{breuk} kleiner dan $1$:
$\frac54$; \quad $\frac73$; \quad $\frac{9}{2}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:8}
Hoeveel minuten zijn: een half uur; een \omterm{def:g3:division:half}{kwart} van een uur; driekwart uur;
$\frac13$ van een uur?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:9}
In een klas van $24$ leerlingen draagt $\frac14$ een bril. Hoeveel
leerlingen dragen een bril? En hoeveel niet?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:fractions:10}
Nina heeft $\frac38$ van een pizza opgegeten en Sam $\frac48$ van dezelfde
pizza. Welke \omterm{def:g4:fractions:def}{breuk} hebben ze samen opgegeten? Welke \omterm{def:g4:fractions:def}{breuk} blijft over? Wie
heeft er meer gegeten?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:fractions:11}
Waar of niet waar, met een plaatje of een voorbeeld: ``het kan dat
$\frac12$ van een kleine pizza minder is dan $\frac14$ van een heel grote
pizza''. Wat moet je altijd weten voordat je twee \omterm{def:g4:fractions:def}{breuken} van iets
vergelijkt?
\end{exercise}
