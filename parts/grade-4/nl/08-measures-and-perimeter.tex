\chapter{Maten en omtrek}\label{ch:g4:measure}

In jaar 3 maten we met latten, weegschalen en klokken
(\cref{ch:g3:measure}); in dit hoofdstuk komt de maattabel erbij --- de
machine die alles omzet --- en twee manieren om een vlakke figuur te meten:
de lengte van haar rand (de \omterm{def:g4:measure:perimeter}{omtrek}) en de ruimte erbinnen (de \omterm{def:g4:measure:area}{oppervlakte},
voorlopig geteld in vakjes).

\section{De maattabel}

\begin{method}[Omzetten met de tabel]\label{met:g4:measure:table}
Bij lengtes is elke kolom tien keer de volgende waard:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|c|c|c|c|c|c}
km & hm & dam & m & dm & cm & mm \\
\hline
 & & & $3$ & $4$ & $0$ & \\
\end{tabular}
\end{center}
\begin{enumerate}
  \item Schrijf het getal met één cijfer per kolom, en zet het cijfer van
        de eenheden van de maat in de kolom van zijn eenheid (hier $3.4$ m:
        de $3$ onder m);
  \item lees het getal in de nieuwe eenheid: $3.4$ m $= 340$ cm (vul lege
        kolommen met nullen);
  \item \omterm{def:g2:measure:units}{massa}'s (kg\,\dots\,g) en \omterm{def:g2:measure:units}{inhouden} (L\,\dots\,cL) werken op dezelfde
        manier.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g4:measure:convert}
$5$ km $= 5\,000$ m; \quad $270$ cm $= 2.7$ m (de $2$ komt onder m);
\quad $3$ kg $250$ g $= 3\,250$ g; \quad $2.5$ L $= 250$ cL.
\end{example}

\section{Omtrek}

\begin{definition}[Omtrek]\label{def:g4:measure:perimeter}
De \emph{omtrek}\index{omtrek} van een \omterm{def:g4:geometry:polygon}{veelhoek} is de totale lengte van zijn
rand: tel de lengtes van alle zijden op (in dezelfde eenheid!). Voor de
bekende figuren zijn er kortere formules:
\[
\text{rechthoek: } P = 2 \times (l + b),
\qquad
\text{vierkant: } P = 4 \times z .
\]
\end{definition}

\begin{example}\label{ex:g4:measure:perimeter}
Een rechthoekige tuin van $12$ m bij $7$ m:
\[
P = 2 \times (12 + 7) = 2 \times 19 = 38 \text{ m}
\]
hek. Een vierkante fotolijst met zijde $18$ cm:
$P = 4 \times 18 = 72$ cm.
\end{example}

\section{Oppervlakte: vakjes tellen}

\begin{definition}[Oppervlakte door te tellen]\label{def:g4:measure:area}
De \emph{oppervlakte}\index{oppervlakte} van een figuur op een raster is het
aantal rastervakjes dat ze bedekt. Twee halve vakjes tellen als één. De
oppervlakte antwoordt op ``hoeveel oppervlak'', de \omterm{def:g4:measure:perimeter}{omtrek} op ``hoe lang is de
rand'' --- twee verschillende vragen!
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=0.55]
  \draw[gray!40, thin] (0,0) grid (13,4);
  \draw[very thick, omDef, fill=omDef!15] (1,1) rectangle (5,3);
  \node[font=\small, omDef] at (3,2) {$8$ vakjes};
  \draw[very thick, omThm, fill=omThm!15]
    (7,1) -- (11,1) -- (11,2) -- (9,2) -- (9,3) -- (7,3) -- cycle;
  \node[font=\small, omThm] at (8.4,1.5) {$6$ vakjes};
\end{tikzpicture}

{\small Twee figuren op het raster: \omterm{def:g4:measure:area}{oppervlaktes} $8$ en $6$ vakjes. Loop nu
de randen af en tel de eenheden: $12$ voor beide! Dezelfde \omterm{def:g4:measure:perimeter}{omtrek},
verschillende \omterm{def:g4:measure:area}{oppervlaktes}.}
\end{omfigure}

\begin{example}[Halve vakjes]\label{ex:g4:measure:half}
Een rechthoekige driehoek op het raster met rechthoekszijden $4$ en $2$
bedekt $3$ hele vakjes en $2$ halve\,\dots\ tel zorgvuldig: haar \omterm{def:g4:measure:area}{oppervlakte}
is de \omterm{def:g3:division:half}{helft} van de rechthoek $4 \times 2$, dus $4$ vakjes. Een rechthoek
langs zijn diagonaal knippen geeft altijd twee driehoeken met dezelfde
\omterm{def:g4:measure:area}{oppervlakte}.
\end{example}

\begin{example}[Zelfde oppervlakte, andere omtrek]\label{ex:g4:measure:samearea}
Een vierkant van $4 \times 4$ en een rechthoek van $8 \times 2$ bedekken
beide $16$ vakjes --- dezelfde \omterm{def:g4:measure:area}{oppervlakte}. Randen: $16$ eenheden voor het
vierkant, $20$ voor de rechthoek. Geen van de twee maten bepaalt de andere:
een lange dunne vorm heeft veel rand bij weinig oppervlak.
\end{example}

\section{Tijdsduren}

\begin{method}[Tijden optellen over het hele uur]\label{met:g4:measure:time}
Om tijdsduren op te tellen of af te trekken, werk je met sprongen via de
hele uren (denk aan $1$ u $= 60$ min):
\[
9\text{ u }40 + 35 \text{ min}:
\quad 9\text{ u }40 \xrightarrow{+20} 10\text{ u }00
\xrightarrow{+15} 10\text{ u }15 .
\]
Tel nooit minuten voorbij $60$ op zonder om te zetten: $40 + 35 = 75$ min
$= 1$ u $15$ min.
\end{method}

\begin{example}\label{ex:g4:measure:timetable}
De trein vertrekt om $8{:}47$ en komt aan om $11{:}05$. Duur:
$8{:}47 \to 9{:}00$ is $13$ min; $9{:}00 \to 11{:}00$ is $2$ u;
$11{:}00 \to 11{:}05$ is $5$ min. Samen: $2$ u $18$ min.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:measure:1}
Zet om met de maattabel: $7$ m in cm; $3.2$ km in m; $85$ mm in cm;
$4\,500$ m in km.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:2}
Zet om: $2$ kg in g; $1\,800$ g in kg en g; $3.5$ L in cL; $25$ cL in L.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:3}
Reken de \omterm{def:g4:measure:perimeter}{omtrek} uit: een rechthoek van $9$ cm $\times$ $5$ cm; een vierkant
met zijde $7.5$ cm; een driehoek met zijden $6$ cm, $8$ cm en $10$ cm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:4}
Een rechthoek heeft \omterm{def:g4:measure:perimeter}{omtrek} $26$ cm en lengte $8$ cm. Zoek zijn breedte (zoek
eerst lengte $+$ breedte).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:5}
Teken op ruitjespapier een L-vormige figuur die precies $10$ vakjes bedekt,
en tel de eenheden van haar \omterm{def:g4:measure:perimeter}{omtrek}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:6}
Teken twee verschillende rechthoeken met een \omterm{def:g4:measure:area}{oppervlakte} van $12$ vakjes.
Reken van elk de \omterm{def:g4:measure:perimeter}{omtrek} uit. Welke is ``ronder'', welke is ``dunner''?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:7}
Op het raster is een rechthoekige driehoek getekend met rechthoekszijden $6$
en $3$. Wat is zijn \omterm{def:g4:measure:area}{oppervlakte} in vakjes? (Gebruik
\cref{ex:g4:measure:half}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:8}
Reken de tijdsduren uit: van $10{:}20$ tot $12{:}45$; van $7{:}35$ tot
$9{:}10$. Een film duurt $1$ u $50$ en begint om $20{:}30$: hoe laat is hij
uit?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:9}
Kim loopt $3$ rondjes om een rechthoekig veld van $60$ m bij $45$ m. Hoeveel
meter loopt ze?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:measure:10}
Een boer wil een vierkant veld met zijde $85$ m omheinen en laat een poort
van $4$ m zonder hek. Hoeveel meter hek moet hij kopen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:measure:11}
Waar of niet waar, met voorbeelden op het raster: ``hebben twee figuren
dezelfde \omterm{def:g4:measure:perimeter}{omtrek}, dan hebben ze dezelfde \omterm{def:g4:measure:area}{oppervlakte}''; ``hebben twee figuren
dezelfde \omterm{def:g4:measure:area}{oppervlakte}, dan hebben ze dezelfde \omterm{def:g4:measure:perimeter}{omtrek}''.
(\Cref{ex:g4:measure:samearea} en de figuur van dit hoofdstuk helpen je.)
\end{exercise}
