\chapter{Symmetrie}\label{ch:g4:symmetry}

Vlinders, gezichten, sneeuwvlokken: de natuur houdt van figuren waarvan de
twee \omterm{def:g3:division:half}{helften} bij het vouwen precies op elkaar passen. In dit hoofdstuk
verkennen we symmetrie met vouwen en ruitjespapier --- constructies met de
geodriehoek wachten tot \cref{ch:g6:symmetry}.

\section{Vouwen en op elkaar leggen}

\begin{definition}[Symmetrieas]\label{def:g4:symmetry:def}
Een lijn is een \emph{symmetrieas}\index{symmetrieas} van een figuur als de
twee \omterm{def:g3:division:half}{helften} \emph{precies} op elkaar passen wanneer je de figuur langs die
lijn vouwt. Een figuur kan geen as, één as of meerdere assen hebben.
\end{definition}

  \draw[thick, fill=omDef!12] (0,0) -- (2.4,0) -- (1.2,2.6) -- cycle;
  \draw[omThm, dashed, thick] (1.2,-0.5) -- (1.2,3.0);
  \node[below, font=\small] at (1.2,-0.7) {één as};
  % rechthoek: twee assen
  \begin{scope}[xshift=5.4cm]
  \draw[thick, fill=omDef!12] (0,0.3) rectangle (3.2,2.3);
  \draw[omThm, dashed, thick] (1.6,-0.4) -- (1.6,3.0);
  \draw[omThm, dashed, thick] (-0.5,1.3) -- (3.7,1.3);
  \node[below, font=\small] at (1.6,-0.7) {twee assen};
  \end{scope}
  % ongelijkzijdig: geen as
  \begin{scope}[xshift=11.2cm]
  \draw[thick, fill=omDef!12] (0,0) -- (3.0,0.4) -- (0.8,2.2) -- cycle;

  \end{scope}
\end{tikzpicture}

{\small Vouw langs de stippellijn: de \omterm{def:g3:division:half}{helften} passen op elkaar. Bij de
laatste driehoek werkt geen enkele vouw --- geen \omterm{def:g4:symmetry:def}{symmetrieas}.}

\begin{example}[Testen met overtrekpapier]\label{ex:g4:symmetry:trace}
Trek de figuur over, klap het overtrekpapier om langs de lijn die je wilt
testen, en leg het terug: bedekt de overtrek de figuur precies, dan is de as
echt. Je ogen laten zich vaak bedriegen --- de diagonaal van een rechthoek
\emph{lijkt} een as, maar de vouwtest zegt van niet (probeer het!).
\end{example}

\section{Figuren afmaken op een raster}

\begin{method}[Het spiegelbeeld van een figuur in een rasterlijn]\label{met:g4:symmetry:grid}
De as is een verticale (of horizontale) lijn van het raster.
\begin{enumerate}
  \item Neem elk \omterm{def:g4:geometry:polygon}{hoekpunt} van de figuur één voor één;
  \item tel in vakjes hoe ver het van de as ligt;
  \item zet het beeldhoekpunt op \emph{dezelfde} afstand aan de
        \emph{andere} kant, op dezelfde rij (of in dezelfde kolom);
  \item verbind de beeldhoekpunten in dezelfde volgorde en controleer: het
        beeld is de spiegeltweeling van de figuur.
\end{enumerate}
\end{method}

{\small Afmaken over de rode as: elk \omterm{def:g4:geometry:polygon}{hoekpunt} springt naar dezelfde afstand
aan de andere kant. \omterm{def:g4:geometry:polygon}{Hoekpunten} die op de as liggen, blijven staan.}
\end{omfigure}

\begin{example}[Zelfde vorm, zelfde maat, omgeklapt]\label{ex:g4:symmetry:properties}
Het beeld heeft precies dezelfde lengtes en dezelfde hoeken als het
origineel: spiegelen kopieert de figuur --- maar klapt haar om, zoals een
linkerhand en een rechterhand. Is de letter F het origineel, dan is haar
spiegelbeeld een omgekeerde F, geen tweede F.
\end{example}

\section{De assen van de bekende figuren}

\begin{example}[Assen tellen]\label{ex:g4:symmetry:shapes}
Door te vouwen:
\begin{itemize}
  \item een vierkant: $4$ assen (twee door de middens van de zijden die
        tegenover elkaar liggen, twee langs de diagonalen);
  \item een rechthoek: $2$ assen (alleen door de middens van de zijden!);
  \item een gelijkbenige driehoek: $1$; een gelijkzijdige driehoek: $3$;
  \item een cirkel: elke lijn door zijn middelpunt --- meer assen dan je
        kunt tellen.
\end{itemize}
\end{example}

\begin{example}[Symmetrie in het alfabet]\label{ex:g4:symmetry:letters}
A, M, T, U, V hebben een verticale as; B, C, D, E een horizontale; H, I, O,
X hebben er twee. Woorden van de juiste letters kunnen ook symmetrisch zijn:
MAM heeft een verticale as; OXO werkt in beide richtingen.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:1}
Trek over en vouw: welke van deze figuren hebben een \omterm{def:g4:symmetry:def}{symmetrieas} --- een
vierkant; een rechthoek die geen vierkant is; een ongelijkzijdige driehoek
(alle zijden verschillend); een cirkel?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:2}
Teken de symmetrieassen van: een vierkant; een rechthoek; een gelijkzijdige
driehoek. Hoeveel zijn het per figuur?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:3}
Laat met de overtrekpapiertest van \cref{ex:g4:symmetry:trace} zien dat de
diagonaal van een rechthoek van $6 \times 4$ \emph{geen} \omterm{def:g4:symmetry:def}{symmetrieas} is.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:4}
Teken op ruitjespapier een verticale as en de letter L (drie vakjes hoog,
twee breed) op $2$ vakjes van de as. Construeer haar spiegelbeeld.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:5}
Teken op ruitjespapier een horizontale as en daarboven een vlaggetje (een
driehoekje van $1 \times 1$ op een mast van $3$ vakjes). Maak de figuur
onder de as af met symmetrie.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:6}
Een figuur raakt haar \omterm{def:g4:symmetry:def}{symmetrieas} in een punt $P$. Waar ligt het beeld van
$P$? Leg het uit met het vouwplaatje.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:7}
Sorteer de hoofdletters A, B, E, F, H, N, O, T: verticale as? horizontale
as? beide? geen van beide?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:8}
Een driehoek heeft zijden van $5$ cm, $5$ cm en $3$ cm. Heeft hij een
\omterm{def:g4:symmetry:def}{symmetrieas}? Welke zijden wisselt de vouw om?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:symmetry:9}
Teken op ruitjespapier een eigen figuur met \emph{precies twee}
symmetrieassen, en teken beide assen erbij. (Tip: denk aan een rechthoek of
aan een kruis.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:symmetry:10}
De \omterm{def:g3:division:half}{helft} van een symmetrische tekening is gegeven: de linkerhelft van een
huis (een vierkante muur en de halve driehoek van het dak), met een
verticale as langs de rechterrand. Beschrijf of teken het hele huis. Hoe
verhouden de lengtes van de afgemaakte rechterhelft zich tot die van de
linkerhelft?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:symmetry:11}
Aya beweert: ``elke driehoek met twee gelijke zijden heeft een \omterm{def:g4:symmetry:def}{symmetrieas},
en elke driehoek met een \omterm{def:g4:symmetry:def}{symmetrieas} heeft twee gelijke zijden.'' Test haar
dubbele bewering op tekeningen: een gelijkbenige driehoek, een gelijkzijdige
en een ongelijkzijdige. Steunt het vouwen haar?
\end{exercise}
