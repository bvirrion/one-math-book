\chapter{Grote getallen}\label{ch:g4:numbers}

Duizendtallen waren het plafond van \cref{ch:g3:numbers}; nu gaat het
plafond omhoog naar \emph{miljoenen}. De truc die enorme getallen
leesbaar maakt, is hun cijfers in groepen van drie knippen.

\section{Groepen van drie cijfers}

\begin{definition}[Groepen van drie]\label{def:g4:numbers:classes}
Vanaf rechts worden de cijfers van een getal in \emph{groepen} van drie
verdeeld: de groep van de \emph{eenheden}, de groep van de
\emph{duizenden}, de groep van de \emph{miljoenen}. Elke groep heeft haar
eigen eenheden, tientallen en honderdtallen. Zo is
\[
6\,309\,452 = 6 \text{ miljoen} + 309 \text{ duizend}
+ 452 \text{ eenheden},
\]
gelezen als ``zes miljoen driehonderdnegenduizend
vierhonderdtweeënvijftig''.
\end{definition}

\begin{example}[Schrijven met de groepen]\label{ex:g4:numbers:write}
``Twaalf miljoen veertigduizend vijfhonderd'' heeft $12$ in de groep van
de miljoenen, $040$ in die van de duizenden en $500$ in die van de
eenheden:
\[
12\,040\,500 .
\]
De nullen van $040$ zijn onmisbaar: elke groep (behalve de meest linkse)
moet haar drie cijfers laten zien.
\end{example}

\begin{example}[Hoeveel duizenden?]\label{ex:g4:numbers:contained}
Het getal $6\,309\,452$ \emph{bevat} $6\,309$ duizenden (alles links van
de groep van de eenheden): $6\,309\,452 = 6\,309 \times 1\,000 + 452$.
``Hoeveel duizenden?'' vraagt dus niet naar één cijfer, maar naar een heel
stuk vooraan het getal.
\end{example}

\section{Vergelijken en afronden}

\begin{method}[Grote getallen vergelijken]\label{met:g4:numbers:compare}
Precies zoals in \cref{met:g3:numbers:compare}, met de groepen als
kortere weg: meer cijfers wint; bij even veel cijfers vergelijk je vanaf
links --- groep na groep, en dan cijfer na cijfer.
\end{method}

\begin{example}\label{ex:g4:numbers:compare}
$2\,145\,890$ en $2\,148\,540$: de miljoenen zijn gelijk ($2$), en in de
groep van de duizenden is $145 < 148$, dus
\[
2\,145\,890 < 2\,148\,540 .
\]
\end{example}

\begin{definition}[Afronden]\label{def:g4:numbers:round}
Een getal \emph{afronden}\index{afronden} op het naaste duizendtal (of
tiental, honderdtal, miljoen\,\dots) is het vervangen door het
dichtstbijzijnde veelvoud van $1\,000$. Ligt het getal precies
halverwege, dan rond je naar boven af. In de praktijk: kijk naar het
cijfer net onder de plaats waarop je afrondt --- bij $5$ of meer rond je
naar boven af, bij $4$ of minder naar beneden.
\end{definition}

\begin{example}\label{ex:g4:numbers:round}
$37\,614$ afgerond op het naaste duizendtal: het cijfer van de
honderdtallen is $6$, dus naar boven: $38\,000$. Afgerond op het naaste
honderdtal: $37\,600$. Afgerond op het naaste tienduizendtal: $40\,000$.
\end{example}

\begin{example}[Grootteordes in de wereld]\label{ex:g4:numbers:world}
Zo worden grote getallen in het echt gebruikt: een land met
$67\,842\,591$ inwoners heeft ``ongeveer $68$ miljoen'' inwoners; een
stad met $214\,376$ inwoners heeft er ``ongeveer $200\,000$''. Het
precieze aantal verandert toch elke dag --- het afgeronde getal draagt de
nuttige informatie.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:numbers:1}
Schrijf in cijfers, met de groepen gescheiden: ``drie miljoen
vijfhonderdtwintigduizend''; ``zeven miljoen acht'';
``tweehonderdveertigduizend zestig''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:2}
Schrijf in woorden: $5\,204\,300$; \quad $10\,050\,000$; \quad
$999\,999$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:3}
In $8\,472\,156$: welk cijfer staat bij de tienduizenden? En welk bij de
honderdtallen van de groep van de miljoenen? Hoeveel \emph{duizenden}
bevat het getal (zie \cref{ex:g4:numbers:contained})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:4}
Neem over en vul in met $<$ of $>$:
\[
980\,000 \;?\; 1\,020\,000, \qquad
3\,456\,000 \;?\; 3\,465\,000, \qquad
7\,090\,050 \;?\; 7\,090\,500 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:5}
Zet op volgorde, van klein naar groot: $2\,300\,000$; \ $980\,500$; \
$2\,030\,000$; \ $2\,299\,999$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:6}
Rond $63\,782$ af: op het naaste tiental; op het naaste honderdtal; op
het naaste duizendtal; op het naaste tienduizendtal.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:7}
Rond af op het naaste miljoen: $4\,721\,000$; \quad $6\,499\,999$;
\quad $9\,500\,000$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:8}
Tel met stappen: drie stappen van $1\,000$ vanaf $998\,500$; twee stappen
van $100\,000$ vanaf $950\,000$. Wat gebeurt er met de groepen als je een
miljoen voorbijgaat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:9}
In een stadion passen $81\,338$ bezoekers. Een krant schrijft ``meer dan
$80\,000$ fans''; de radio zegt ``ongeveer $81\,000$''. Hebben ze beide
gelijk? Op welke plaats heeft ieder afgerond?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:numbers:10}
Gebruik elk van de cijfers $1$ tot $7$ precies één keer en schrijf het
grootste getal van zeven cijfers, het kleinste, en het getal dat het
dichtst bij $4\,000\,000$ ligt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:numbers:11}
Wat is het grootste hele getal dat, afgerond op het naaste duizendtal,
$50\,000$ geeft? En het kleinste?
\end{exercise}
