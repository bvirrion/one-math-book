\chapter{Lijnen en veelhoeken}\label{ch:g4:geometry}

Met een lat, een geodriehoek en een passer wordt meetkunde een ambacht. In
dit hoofdstuk leer je de twee bijzondere standen van lijnen ---
\omterm{def:g4:geometry:def}{loodrecht} en \omterm{def:g4:geometry:def}{parallel} --- met het gereedschap in de hand, en maken we een
rondje langs de familie van de \omterm{def:g4:geometry:polygon}{veelhoeken}. (De theorie van deze begrippen
komt in \cref{ch:g6:lines}.)

\section{Loodrecht en parallel}

\begin{definition}[Loodrecht, parallel]\label{def:g4:geometry:def}
Twee lijnen staan \emph{loodrecht}\index{loodrechte lijnen} op elkaar als
ze elkaar onder een \omterm{def:g3:shapes:rightangle}{rechte hoek} snijden (te controleren met de
geodriehoek, aangegeven met een vierkantje). Twee lijnen zijn
\emph{parallel}\index{parallelle lijnen} als ze dezelfde richting hebben en
elkaar nooit snijden --- zoals de twee rails van een spoor, altijd op
dezelfde afstand van elkaar.
\end{definition}

\begin{method}[Tekenen met de geodriehoek]\label{met:g4:geometry:draw}
\emph{De loodlijn op een lijn $d$ door een punt $P$}: leg één rand van de
\omterm{def:g3:shapes:rightangle}{rechte hoek} langs $d$, schuif tot de andere rand bij $P$ komt, en teken.

\emph{De \omterm{def:g4:geometry:def}{parallel} aan $d$ door $P$}: leg de rand van de geodriehoek langs
$d$, druk de lat als rail tegen een andere rand, schuif de geodriehoek
langs de lat tot bij $P$, en teken.
\end{method}

\begin{example}[Ze in figuren herkennen]\label{ex:g4:geometry:spot}
In een rechthoek staat elke zijde \omterm{def:g4:geometry:def}{loodrecht} op haar twee buren en is ze
\omterm{def:g4:geometry:def}{parallel} aan de zijde die ertegenover ligt. Ruitjespapier is een
kant-en-klaar netwerk van parallelle en \omterm{def:g4:geometry:def}{loodrechte lijnen} --- daarom is het
zo prettig om erop te tekenen.
\end{example}

\section{Veelhoeken}

\begin{definition}[Veelhoek]\label{def:g4:geometry:polygon}
Een \emph{veelhoek}\index{veelhoek} is een gesloten figuur waarvan de rand
uit rechte lijnstukken bestaat, de \emph{zijden}; de hoekpunten zijn de
\emph{hoekpunten}\index{hoekpunt}. Veelhoeken worden genoemd naar hun
aantal zijden: \emph{driehoek} ($3$), \emph{vierhoek}\index{vierhoek}
($4$), \emph{vijfhoek}\index{vijfhoek} ($5$),
\emph{zeshoek}\index{zeshoek} ($6$), \emph{achthoek}\index{achthoek}
($8$).
\end{definition}

\begin{example}[Een veelhoek beschrijven]\label{ex:g4:geometry:describe}
Een nuttig spel: beschrijf een figuur zo precies dat een klasgenoot hem kan
natekenen zonder hem te zien. ``Een \omterm{def:g4:geometry:polygon}{vierhoek} $ABCD$ met $AB = 5$ cm, een
\omterm{def:g3:shapes:rightangle}{rechte hoek} in $A$, $AD = 3$ cm\,\dots'' --- de juiste woorden zijn wat de
boodschap laat werken.
\end{example}

\section{De cirkel en de passer}

\begin{definition}[Cirkel]\label{def:g4:geometry:circle}
De \emph{cirkel} met middelpunt $O$ en \omterm{def:g3:shapes:shapes}{radius} $3$ cm bestaat uit alle
punten die precies $3$ cm van $O$ liggen. De passer tekent hem: punt op
$O$, opening $3$ cm, ronddraaien. Een \emph{diameter}\index{diameter} is
een lijnstuk door het middelpunt dat twee punten van de cirkel verbindt;
hij is twee keer zo lang als de \omterm{def:g3:shapes:shapes}{radius}.
\end{definition}

\begin{example}[De passer als afstandsbewaarder]\label{ex:g4:geometry:compass}
De passer doet meer dan cirkels tekenen: hij \emph{verplaatst afstanden}.
Om alle punten op $4$ cm van een punt $A$ te markeren, teken je de cirkel
met middelpunt $A$ en \omterm{def:g3:shapes:shapes}{radius} $4$ cm; om twee lijnstukken zonder lat te
vergelijken, zet je de passer op het ene en probeer je hem op het andere.
\end{example}

\begin{method}[Een figuur op een raster natekenen]\label{met:g4:geometry:reproduce}
\begin{enumerate}
  \item Kies een beginhoekpunt en zijn plaats op het lege raster;
  \item loop de figuur \omterm{def:g4:geometry:polygon}{hoekpunt} na \omterm{def:g4:geometry:polygon}{hoekpunt} af en tel de stappen op het
        raster (``$3$ naar rechts, $2$ naar boven'');
  \item verbind de \omterm{def:g4:geometry:polygon}{hoekpunten} op volgorde en vergelijk met het voorbeeld:
        dezelfde lengtes, dezelfde richtingen.
\end{enumerate}
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:geometry:1}
Teken een lijn $d$ en een punt $P$ dat er niet op ligt. Construeer de
loodlijn op $d$ door $P$, en daarna de \omterm{def:g4:geometry:def}{parallel} aan $d$ door $P$ (met de
lat als rail). Zet het teken voor de \omterm{def:g3:shapes:rightangle}{rechte hoek} erbij.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:2}
Bekijk de hoofdletters E, H, N, T, Z. Zoek in elke letter een paar
parallelle strepen en een paar loodrechte strepen, als die er zijn.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:3}
Noem elke \omterm{def:g4:geometry:polygon}{veelhoek}: $6$ zijden; $4$ zijden; $3$ zijden; $8$ zijden; $5$
zijden. Hoeveel \emph{\omterm{def:g4:geometry:polygon}{hoekpunten}} heeft elk?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:4}
Teken een willekeurige \omterm{def:g4:geometry:polygon}{vijfhoek} (niet noodzakelijk regelmatig) en noem de
\omterm{def:g4:geometry:polygon}{hoekpunten} $A$, $B$, $C$, $D$, $E$. Hoeveel zijden heeft hij? Teken de
diagonalen vanuit $A$ (lijnstukken van $A$ naar \omterm{def:g4:geometry:polygon}{hoekpunten} die geen buren
zijn): hoeveel zijn dat?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:5}
Teken een cirkel met middelpunt $O$ en \omterm{def:g3:shapes:shapes}{radius} $4$ cm; een \omterm{def:g4:geometry:circle}{diameter} $[AB]$;
en een punt $M$ van de cirkel dat niet op $[AB]$ ligt. Geef de lengtes
$OA$, $OM$ en $AB$ zonder te meten.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:6}
Vergelijk met de passer als afstandsbewaarder
(\cref{ex:g4:geometry:compass}) de zijden van een driehoek die een
klasgenoot heeft getekend, en beslis of hij twee gelijke zijden heeft.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:7}
Teken dit pad op ruitjespapier na en sluit het tot een \omterm{def:g4:geometry:polygon}{veelhoek}: begin in
een punt $A$; ga $4$ naar rechts; $2$ naar boven; $3$ naar links; $2$ naar
boven; en dan in een rechte lijn terug naar $A$. Hoe heet de \omterm{def:g4:geometry:polygon}{veelhoek} die
je krijgt (tel de zijden)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:8}
Waar of niet waar? Leg uit met een tekening.
``Twee \omterm{def:g4:geometry:def}{loodrechte lijnen} snijden elkaar altijd.''
``Twee \omterm{def:g4:geometry:def}{parallelle lijnen} snijden elkaar nooit.''
``Een lijn kan \omterm{def:g4:geometry:def}{parallel} zijn aan de ene lijn en \omterm{def:g4:geometry:def}{loodrecht} staan op een
andere.''
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:9}
Teken een lijn $d$, dan een lijn $e$ die \omterm{def:g4:geometry:def}{loodrecht} op $d$ staat, en dan een
lijn $f$ die \omterm{def:g4:geometry:def}{loodrecht} op $e$ staat. Bekijk $d$ en $f$: wat valt je op?
(Deze observatie wordt een stelling in \cref{ch:g6:lines}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:10}
Beschrijf de volgende figuur in woorden voor een klasgenoot, zonder hem te
laten zien: een vierkant met zijde $4$ cm, met een cirkel van \omterm{def:g3:shapes:shapes}{radius} $2$ cm
met het midden van het vierkant als middelpunt (die de vier zijden raakt).
Schrijf de beschrijving als genummerde instructies.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:11}
Hoeveel diagonalen heeft een \omterm{def:g4:geometry:polygon}{vierhoek}? En een \omterm{def:g4:geometry:polygon}{vijfhoek}? En een \omterm{def:g4:geometry:polygon}{zeshoek}?
Teken de drie gevallen, tel zorgvuldig, en beschrijf het patroon dat je
ziet.
\end{exercise}
