\chapter{Delen}\label{ch:g4:division}

In jaar 3 deelden we met de tafels en met kleine \omterm{def:g3:division:remainder}{resten}
(\cref{ch:g3:division}); in dit hoofdstuk komt de volledige schriftelijke
manier --- de staartdeling --- die elk getal aankan, cijfergroep na
cijfergroep.

\section{De staartdeling}

\begin{method}[Staartdeling door een getal van één cijfer]\label{met:g4:division:method}
Om $749$ door $6$ te delen:
\begin{enumerate}
  \item neem het meest linkse cijfer, $7$ (honderdtallen): $6$ gaat $1$
        keer in $7$; schrijf $1$ in het \omterm{def:g3:division:remainder}{quotiënt}, trek $6$ af, \omterm{def:g3:division:remainder}{rest} $1$;
  \item \emph{haal} het volgende cijfer naar beneden, de $4$: deel nu
        $14$ (tientallen): $6 \times 2 = 12$; schrijf $2$, \omterm{def:g3:division:remainder}{rest} $2$;
  \item haal de $9$ naar beneden: deel $29$: $6 \times 4 = 24$; schrijf
        $4$, \omterm{def:g3:division:remainder}{rest} $5$;
  \item er is geen cijfer meer over: het \omterm{def:g3:division:remainder}{quotiënt} is $124$, de \omterm{def:g3:division:remainder}{rest} is
        $5$.
\end{enumerate}
Controle: $6 \times 124 + 5 = 744 + 5 = 749$, en $5 < 6$.
\end{method}

\begin{example}[Een nul in het quotiënt]\label{ex:g4:division:zero}
Deel $618$ door $3$: honderdtallen, $6 \div 3 = 2$; tientallen, haal de
$1$ naar beneden: $3$ gaat $0$ keer in $1$ --- \emph{schrijf die} $0$! ---
\omterm{def:g3:division:remainder}{rest} $1$; eenheden, haal de $8$ naar beneden: $18 \div 3 = 6$. \omterm{def:g3:division:remainder}{Quotiënt}
$206$, \omterm{def:g3:division:remainder}{rest} $0$. De \omterm{def:g1:counting:zero}{nul} in het midden vergeten (en dus $26$ schrijven) is
de klassieke fout; de controle $3 \times 26 = 78 \neq 618$ ontmaskert hem
onmiddellijk.
\end{example}

\begin{example}[De grootte van het quotiënt schatten]\label{ex:g4:division:size}
Zet voordat je $749$ door $6$ deelt de uitkomst tussen grenzen:
$6 \times 100 = 600$ en $6 \times 200 = 1\,200$, dus ligt het \omterm{def:g3:division:remainder}{quotiënt}
tussen $100$ en $200$ --- het heeft drie cijfers. Die schatting van één
regel voorkomt bijna alle fouten met verschoven cijfers.
\end{example}

\section{Kiezen wat de vraag vraagt}

\begin{method}[Quotiënt, rest, of quotiënt plus één]\label{met:g4:division:interpret}
Zoals in \cref{met:g3:division:interpret} beslist het verhaal:
\begin{enumerate}
  \item ``hoeveel volle \dots'' of ``hoeveel elk'': het \omterm{def:g3:division:remainder}{quotiënt};
  \item ``hoeveel blijft er over'': de \omterm{def:g3:division:remainder}{rest};
  \item ``hoeveel dozen om alles in te doen'': het \omterm{def:g3:division:remainder}{quotiënt}, plus één als
        de \omterm{def:g3:division:remainder}{rest} niet \omterm{def:g1:counting:zero}{nul} is.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g4:division:problem}
$530$ boeken moeten in dozen van $8$ worden verpakt.
\[
530 = 8 \times 66 + 2 .
\]
Hoeveel \emph{volle} dozen? $66$. Hoeveel boeken zitten niet in een volle
doos? $2$. Hoeveel dozen om \emph{alle} boeken te verpakken? $67$. Drie
vragen, één deling.
\end{example}

\begin{example}[Geld eerlijk verdelen]\label{ex:g4:division:money}
Vier vrienden delen de kosten van een cadeau van $92$ gelijk:
$92 \div 4 = 23$ precies (\omterm{def:g3:division:remainder}{rest} $0$). Ieder betaalt $23$. Als de \omterm{def:g3:division:remainder}{rest} \omterm{def:g1:counting:zero}{nul}
is, ``komt de deling precies uit'' en is de verdeling volmaakt eerlijk.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g4:division:1}
Zet elk \omterm{def:g3:division:remainder}{quotiënt} tussen grenzen zoals in \cref{ex:g4:division:size}
(tussen welke twee ``ronde'' getallen?) en maak daarna de staartdeling:
$96 \div 4$; \quad $87 \div 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:2}
Maak de staartdelingen en controleer elke uitkomst:
$672 \div 4$; \quad $925 \div 7$; \quad $804 \div 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:3}
Pas op met de \omterm{def:g1:counting:zero}{nul} (zie \cref{ex:g4:division:zero}):
$816 \div 4$; \quad $420 \div 7$; \quad $2\,512 \div 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:4}
Schrijf de controlegelijkheid ($a = b \times q + r$) op voor:
$85 \div 9$; \quad $1\,000 \div 3$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:5}
$156$ eieren worden in dozen van $6$ gedaan. Hoeveel dozen worden er
gevuld?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:6}
Een lint van $250$ cm wordt in stukken van $8$ cm geknipt. Hoeveel stukken
krijg je, en welke lengte blijft over?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:7}
$375$ leerlingen gaan naar een voorstelling in bussen met $52$ plaatsen.
Hoeveel bussen zijn er nodig? (Welk geval van
\cref{met:g4:division:interpret} is dit? Deel door $52$ met behulp van
veelvouden: $52 \times 7 = 364$.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:8}
Drie vrienden verdelen $234$ knikkers gelijk. Hoeveel knikkers krijgt
ieder? Controleer met een vermenigvuldiging.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:9}
Zoek het deeltal: de deling door $7$ geeft \omterm{def:g3:division:remainder}{quotiënt} $58$ en \omterm{def:g3:division:remainder}{rest} $3$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:division:10}
Een bibliothecaris zet $438$ boeken in de kast, $9$ per plank. Planken
komen in kasten van $6$ planken. Hoeveel planken zijn er nodig? Hoeveel
kasten moeten er gekocht worden? (Twee delingen, beide met een vraag van
het type ``plus één''.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:division:11}
Bij een deling door $8$ is het \omterm{def:g3:division:remainder}{quotiënt} gelijk aan de \omterm{def:g3:division:remainder}{rest}. Welke deeltallen
zijn mogelijk? Noem ze allemaal. (De \omterm{def:g3:division:remainder}{rest} moet onder $8$ blijven.)
\end{exercise}
