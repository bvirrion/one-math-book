\chapter{सममिति}\label{ch:g4:symmetry}

तितलियाँ, चेहरे, बर्फ़ के फाहे: प्रकृति को ऐसी आकृतियाँ पसंद हैं
जिनके दोनों आधे मोड़ने पर आपस में मिल जाएँ। यह अध्याय मोड़ों और
जाल वाले काग़ज़ से सममिति खोजता है --- \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका से रचनाएँ
\cref{ch:g6:symmetry} में आएँगी।

\section{मोड़ना और मिलाना}

\begin{definition}[सममिति अक्ष]\label{def:g4:symmetry:def}
कोई रेखा किसी आकृति की \emph{सममिति अक्ष}\index{सममिति अक्ष} तब
होती है जब उस रेखा पर आकृति को मोड़ने से दोनों आधे
\emph{बिलकुल} मिल जाएँ। किसी आकृति की कोई अक्ष न हो, एक हो, या
कई हो सकती हैं।
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=0.6]
  % सममित आकृति: समद्विबाहु त्रिभुज
  \draw[thick, fill=omDef!12] (0,0) -- (2.4,0) -- (1.2,2.6) -- cycle;
  \draw[omThm, dashed, thick] (1.2,-0.5) -- (1.2,3.0);
  \node[below, font=\small] at (1.2,-0.7) {एक अक्ष};
  % आयत: दो अक्ष
  \begin{scope}[xshift=5.4cm]
  \draw[thick, fill=omDef!12] (0,0.3) rectangle (3.2,2.3);
  \draw[omThm, dashed, thick] (1.6,-0.4) -- (1.6,3.0);
  \draw[omThm, dashed, thick] (-0.5,1.3) -- (3.7,1.3);
  \node[below, font=\small] at (1.6,-0.7) {दो अक्ष};
  \end{scope}
  % विषमबाहु: कोई नहीं
  \begin{scope}[xshift=11.2cm]
  \draw[thick, fill=omDef!12] (0,0) -- (3.0,0.4) -- (0.8,2.2) -- cycle;
  \node[below, font=\small] at (1.4,-0.7) {कोई अक्ष नहीं};
  \end{scope}
\end{tikzpicture}

{\small बिंदुदार रेखा पर मोड़ो: दोनों आधे मिल जाते हैं। आख़िरी
त्रिभुज में कोई भी मोड़ काम नहीं करता --- कोई \omterm{def:g4:symmetry:def}{सममिति अक्ष} नहीं।}
\end{omfigure}

\begin{example}[अनुरेखण काग़ज़ से जाँच]\label{ex:g4:symmetry:trace}
आकृति को उतारो, अनुरेखण काग़ज़ को मानी हुई अक्ष पर पलटो, और वापस
रख दो: अगर उतारी हुई आकृति मूल आकृति को ठीक ढक ले, तो अक्ष सचमुच
की है। अकेली आँख अक्सर धोखा खाती है --- आयत का विकर्ण अक्ष जैसा
\emph{दिखता} है, पर मोड़ की जाँच मना कर देती है (आज़माओ!)।
\end{example}

\section{जाल पर आकृतियाँ पूरी करना}

\begin{method}[जाल की रेखा पर सममित आकृति]\label{met:g4:symmetry:grid}
अक्ष जाल की कोई खड़ी (या आड़ी) रेखा है।
\begin{enumerate}
  \item आकृति के हर \omterm{def:g4:geometry:polygon}{शीर्ष} को बारी-बारी लो;
  \item अक्ष से उसकी दूरी खानों में गिनो;
  \item प्रतिबिंब \omterm{def:g4:geometry:polygon}{शीर्ष} को दूसरी ओर \emph{उतनी ही} दूरी पर,
        उसी पंक्ति (या स्तंभ) में रखो;
  \item प्रतिबिंब शीर्षों को उसी क्रम में जोड़ो, और जाँचो:
        प्रतिबिंब आकृति का दर्पण-जुड़वाँ है।
\end{enumerate}
\end{method}

\begin{omfigure}
\begin{tikzpicture}[scale=0.55]
  \draw[gray!40, thin] (0,0) grid (12,5);
  \draw[omThm, very thick] (6,-0.4) -- (6,5.4);
  \draw[thick, omDef, fill=omDef!15]
    (1,1) -- (5,1) -- (5,2) -- (3,2) -- (3,4) -- (1,4) -- cycle;
  \draw[thick, omMeth, fill=omMeth!15]
    (11,1) -- (7,1) -- (7,2) -- (9,2) -- (9,4) -- (11,4) -- cycle;
  \draw[dashed] (5,2) -- (7,2);
  \draw[dashed] (3,4) -- (9,4);
\end{tikzpicture}

{\small लाल अक्ष के दूसरी ओर पूरा करना: हर \omterm{def:g4:geometry:polygon}{शीर्ष} उतनी ही दूरी पर
कूद जाता है। अक्ष को छूने वाले \omterm{def:g4:geometry:polygon}{शीर्ष} अपनी जगह ही रहते।}
\end{omfigure}

\begin{example}[वही आकार, वही नाप, पर पलटा हुआ]\label{ex:g4:symmetry:properties}
प्रतिबिंब की लंबाइयाँ और कोण बिलकुल वही होते हैं जो मूल आकृति
के: सममिति आकृति की नक़ल बनाती है --- पर उसे पलट देती है, जैसे
बायाँ हाथ और दायाँ हाथ। मूल अक्षर F हो, तो उसका
दर्पण-प्रतिबिंब उलटा F होता है, दूसरा F नहीं।
\end{example}

\section{आम आकारों की अक्ष}

\begin{example}[अक्ष गिनना]\label{ex:g4:symmetry:shapes}
मोड़कर देखने पर:
\begin{itemize}
  \item वर्ग: $4$ अक्ष (दो आमने-सामने की भुजाओं के बीचों से
        होकर, दो विकर्णों पर);
  \item आयत: $2$ अक्ष (केवल आमने-सामने की भुजाओं के बीचों से!);
  \item समद्विबाहु त्रिभुज: $1$; समबाहु त्रिभुज: $3$;
  \item वृत्त: उसके केंद्र से होकर जाने वाली हर रेखा --- गिनी न
        जा सकें उतनी अक्ष।
\end{itemize}
\end{example}

\begin{example}[वर्णमाला में सममिति]\label{ex:g4:symmetry:letters}
A, M, T, U, V में खड़ी अक्ष है; B, C, D, E में आड़ी;
H, I, O, X में दोनों। सही अक्षरों से बने शब्द भी सममित हो सकते
हैं: MUM में खड़ी अक्ष है; OXO दोनों तरह चलता है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:1}
उतारो और मोड़ो: इनमें से किनमें \omterm{def:g4:symmetry:def}{सममिति अक्ष} है --- एक वर्ग;
एक (वर्ग न होने वाला) आयत; एक विषमबाहु त्रिभुज (सारी भुजाएँ
अलग); एक वृत्त?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:2}
इनकी \omterm{def:g4:symmetry:def}{सममिति अक्ष} खींचो: एक वर्ग; एक आयत; एक समबाहु त्रिभुज।
हर एक की कितनी हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:3}
\cref{ex:g4:symmetry:trace} की अनुरेखण-काग़ज़ वाली जाँच से दिखाओ
कि $6 \times 4$ के आयत का विकर्ण \omterm{def:g4:symmetry:def}{सममिति अक्ष} \emph{नहीं}
है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:4}
जाल वाले काग़ज़ पर एक खड़ी अक्ष बनाओ और अक्ष से $2$ खाने दूर
अक्षर L (तीन खाने ऊँचा, दो चौड़ा) बनाओ। उसका दर्पण-प्रतिबिंब
बनाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:5}
जाल वाले काग़ज़ पर एक आड़ी अक्ष बनाओ और उसके ऊपर एक छोटा झंडा
($3$ खानों के डंडे पर $1 \times 1$ का त्रिभुज)। अक्ष के नीचे
सममिति से आकृति पूरी करो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:6}
कोई आकृति अपनी \omterm{def:g4:symmetry:def}{सममिति अक्ष} को बिंदु $P$ पर छूती है। $P$ का
प्रतिबिंब कहाँ है? मोड़ वाले चित्र से समझाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:7}
बड़े अक्षर A, B, E, F, H, N, O, T को छाँटो: खड़ी अक्ष?
आड़ी अक्ष? दोनों? कोई नहीं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:8}
एक त्रिभुज की भुजाएँ $5$ cm, $5$ cm और $3$ cm हैं। क्या उसकी
\omterm{def:g4:symmetry:def}{सममिति अक्ष} है? मोड़ किन भुजाओं की अदला-बदली करता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:symmetry:9}
जाल वाले काग़ज़ पर अपनी एक ऐसी आकृति बनाओ जिसकी \emph{ठीक दो}
\omterm{def:g4:symmetry:def}{सममिति अक्ष} हों, और दोनों अक्ष भी खींचो। (संकेत: आयत सोचो,
या एक क्रॉस।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:symmetry:10}
एक सममित चित्र का \omterm{def:g3:division:half}{आधा} दिया है: घर का बायाँ \omterm{def:g3:division:half}{आधा} (एक वर्गाकार
दीवार और त्रिभुजाकार छत का \omterm{def:g3:division:half}{आधा}), जिसके दाएँ किनारे पर खड़ी अक्ष
है। पूरे घर का वर्णन करो या उसे बनाओ। पूरा किया हुआ दायाँ \omterm{def:g3:division:half}{आधा}
लंबाइयों में बाएँ आधे से कैसा है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:symmetry:11}
आया का दावा है: ``दो बराबर भुजाओं वाले हर त्रिभुज की \omterm{def:g4:symmetry:def}{सममिति अक्ष}
होती है, और \omterm{def:g4:symmetry:def}{सममिति अक्ष} वाले हर त्रिभुज की दो भुजाएँ बराबर
होती हैं।'' उसके दोहरे दावे को चित्रों पर आज़माओ: एक समद्विबाहु
त्रिभुज, एक समबाहु, एक विषमबाहु। क्या मोड़ना उसका साथ देता है?
\end{exercise}
