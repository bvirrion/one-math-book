\chapter{पहली भिन्नें}\label{ch:g4:fractions}

\omterm{def:g3:division:half}{आधा} सेब, घंटे की \omterm{def:g3:division:half}{चौथाई}, कक्षा के तीन \omterm{def:g3:division:half}{चौथाई}: भिन्नें चीज़ों के
टुकड़ों को नाम देती हैं। यह अध्याय $\frac{a}{b}$ लिखने का ढंग
बताता है, चित्रों और संख्या रेखा पर भिन्नें पढ़ता है, और सबसे
आसान भिन्नों की तुलना करता है। (भिन्नें
\cref{ch:g6:fractions} में बड़ी होती हैं।)

\section{टुकड़ों को नाम देना}

\begin{definition}[भिन्न]\label{def:g4:fractions:def}
किसी पूरे को बराबर हिस्सों में काटो और कुछ हिस्से लो --- वही एक
\emph{भिन्न}\index{भिन्न} है:
\[
\frac{3}{4}
\quad
\begin{array}{l}
3 \text{: कितने हिस्से लिए गए (\emph{अंश}\index{अंश});}\\
4 \text{: पूरे को कितने बराबर हिस्सों में काटा गया
(\emph{हर}\index{हर})।}
\end{array}
\]
$\frac12$ \emph{\omterm{def:g3:division:half}{आधा}} है, $\frac13$ \emph{तिहाई}, $\frac14$
\emph{\omterm{def:g3:division:half}{चौथाई}}, $\frac{1}{10}$ \emph{दसवाँ भाग}।
\end{definition}

\begin{example}[संग्रह की भिन्न]\label{ex:g4:fractions:collection}
$12$ कंचों का $\frac13$: $12$ कंचों को $4$-$4$ के $3$ बराबर
समूहों में बाँटो; एक समूह $\frac13$ है, इसलिए $12$ का $\frac13$,
$4$ है --- और $12$ का $\frac23$ दो समूह, यानी $8$।
\end{example}

\section{संख्या रेखा पर भिन्नें}

\begin{method}[$\frac{a}{b}$ को रखना]\label{met:g4:fractions:line}
रेखा की हर इकाई को $b$ बराबर कदमों में काटो; $0$ से $a$ कदम
चलो। $a$, $b$ से बड़ा हो तो चाल $1$ के पार निकल जाती है: भिन्नें
एक पूरे से बड़ी भी हो सकती हैं!
\end{method}

\begin{example}[भिन्नों में छिपी पूर्ण संख्याएँ]\label{ex:g4:fractions:whole}
$\frac44 = 1$ (चार \omterm{def:g3:division:half}{चौथाई} मिलकर पूरा बनाते हैं); $\frac82 = 4$;
$\frac{12}{3} = 4$। और $\frac74$, $1 + \frac34$ है: एक पूरा और
तीन \omterm{def:g3:division:half}{चौथाई}।
\end{example}

\section{सरल भिन्नों की तुलना और जोड़}

\begin{proposition}[एक ही हर]\label{prop:g4:fractions:compare}
हर एक जैसा हो तो टुकड़े भी एक ही नाप के होते हैं --- बस उन्हें
गिन लो:
\[
\frac{2}{8} < \frac{5}{8},
\qquad
\frac{3}{8} + \frac{4}{8} = \frac{7}{8}.
\]
पूरे से तुलना भी आसान है: $\frac{5}{8} < 1 < \frac{9}{8}$
($8$ आठवें भागों से कम, फिर ज़्यादा)।
\end{proposition}

\begin{proof}[चित्र पर, क्यों]
एक ही पट्टी के $3$ आठवें भाग रँगकर फिर $4$ आठवें भाग और रँगने
पर कुल $7$ आठवें भाग रँग जाते हैं।
\end{proof}

\begin{example}[अलग-अलग हर, वही बिंदु]\label{ex:g4:fractions:equal}
संख्या रेखा पर $\frac12$, $\frac24$ और $\frac{5}{10}$ सब एक ही
बिंदु पर आते हैं: ये एक ही संख्या के तीन नाम हैं। हर आधे को दो
में काटने पर चौथाइयाँ बनती हैं; पाँच में काटने पर दसवें भाग।
(सामान्य नियम \cref{ch:g6:fractions} में है।)
\end{example}

\begin{example}[घंटे की चौथाइयाँ]\label{ex:g4:fractions:hour}
एक घंटे में $60$ मिनट होते हैं, इसलिए घंटे की \omterm{def:g3:division:half}{चौथाई}
$60 \div 4 = 15$ मिनट है, और घंटे के तीन \omterm{def:g3:division:half}{चौथाई}
$45$ मिनट। ``साढ़े दो'' का मतलब है दो बजे के $\frac12$ घंटे
बाद।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:fractions:1}
दिखाई गई \omterm{def:g4:fractions:def}{भिन्न} लिखो: $6$ में कटा पिज़्ज़ा जिसके $5$ टुकड़े बचे
हैं; $8$ ख़ानों की चॉकलेट जिसके $3$ खा लिए गए (खाई गई \omterm{def:g4:fractions:def}{भिन्न});
तीन बराबर खड़ी पट्टियों में बँटा झंडा, जिसकी $1$ रंगी है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:2}
$5$ बराबर हिस्सों में कटी एक पट्टी बनाओ और $\frac35$ रँगो। फिर
उतनी ही एक और पट्टी का $\frac{2}{5}$ रँगो। कौन-सी रँगाई बड़ी है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:3}
निकालो: $18$ का $\frac12$; \quad $20$ का $\frac14$; \quad
$21$ का $\frac13$; \quad $20$ का $\frac34$ (\omterm{def:g3:division:half}{चौथाई} के तीन
समूह)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:4}
तिहाइयों में कटी संख्या रेखा पर रखो: $\frac13$; \ $\frac33$; \
$\frac53$; \ $2$। कौन-सी \omterm{def:g4:fractions:def}{भिन्न} ठीक $2$ पर आती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:5}
उतार कर $<$, $>$ या $=$ से पूरा करो:
\[
\frac38 \;?\; \frac68, \qquad
\frac55 \;?\; 1, \qquad
\frac74 \;?\; 1, \qquad
\frac12 \;?\; \frac24 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:6}
निकालो: $\frac26 + \frac36$; \quad $\frac58 - \frac28$; \quad
$\frac14 + \frac24$; \quad $1 - \frac13$ ($1$ में कितनी तिहाइयाँ
होती हैं?)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:7}
एक पूर्ण संख्या और $1$ से छोटी \omterm{def:g4:fractions:def}{भिन्न} के जोड़ के रूप में लिखो:
$\frac54$; \quad $\frac73$; \quad $\frac{9}{2}$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:8}
कितने मिनट होते हैं: आधा घंटा; घंटे की \omterm{def:g3:division:half}{चौथाई}; घंटे के तीन
\omterm{def:g3:division:half}{चौथाई}; घंटे का $\frac13$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:9}
$24$ विद्यार्थियों की कक्षा में $\frac14$ चश्मा पहनते हैं। कितने
विद्यार्थी चश्मा पहनते हैं? कितने नहीं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:fractions:10}
नीना ने एक पिज़्ज़े का $\frac38$ खाया और सैम ने उसी पिज़्ज़े का
$\frac48$। दोनों ने मिलकर कितनी \omterm{def:g4:fractions:def}{भिन्न} खाई? कितनी बची?
किसने ज़्यादा खाया?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:fractions:11}
सही या ग़लत, चित्र या उदाहरण के साथ: ``छोटे पिज़्ज़े का
$\frac12$, बहुत बड़े पिज़्ज़े के $\frac14$ से कम हो सकता है''।
किसी चीज़ की दो भिन्नों की तुलना करने से पहले हमेशा क्या पता
होना चाहिए?
\end{exercise}
