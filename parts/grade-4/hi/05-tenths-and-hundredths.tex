\chapter{दसवें और सौवें भाग}\label{ch:g4:decimals}

$3$ और $4$ के बीच पूर्ण संख्याएँ चुप हैं --- पर कोई धावक $3$
मीटर और थोड़ा-सा आगे हो सकता है। इकाई को दस में, फिर सौ में
काटने पर \emph{दशमलव भिन्नें} मिलती हैं, और उनके लिए एक शानदार
छोटा रूप: \omterm{def:g4:decimals:point}{दशमलव बिंदु}। (दशमलव संख्याओं का पूरा अध्याय
\cref{ch:g5:decimals} में है।)

\section{दशमलव भिन्नें}

\begin{definition}[दसवें और सौवें भाग]\label{def:g4:decimals:def}
किसी इकाई को $10$ बराबर हिस्सों में काटने पर \emph{दसवें भाग}
मिलते हैं, $\frac{1}{10}$; हर दसवें भाग को फिर $10$ में काटने पर
\emph{सौवें भाग}, $\frac{1}{100}$। तो
\[
1 = \frac{10}{10} = \frac{100}{100},
\qquad
\frac{1}{10} = \frac{10}{100}.
\]
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=0.42]
  \draw[gray!50, thin] (0,0) grid (10,10);
  \foreach \i in {0,...,9} \fill[omDef!35] (3,\i+0.06)
    rectangle (3.94,\i+0.94);
  \fill[omThm!70] (6.05,4.05) rectangle (6.95,4.95);
  \draw[thick] (0,0) rectangle (10,10);
  \node[below, font=\small] at (5,-0.4)
    {पूरा वर्ग $= 1$;\ \ एक स्तंभ (नीला) $= \frac{1}{10}$;\ \
    एक छोटा ख़ाना (लाल) $= \frac{1}{100}$};
\end{tikzpicture}

{\small सौ का वर्ग: दस-दस ख़ानों के दस स्तंभ। एक स्तंभ दसवाँ
भाग है, एक ख़ाना सौवाँ भाग।}
\end{omfigure}

\begin{example}\label{ex:g4:decimals:count}
सौ के वर्ग के $3$ स्तंभ और $7$ अतिरिक्त ख़ाने रँगने पर वर्ग का
\[
\frac{3}{10} + \frac{7}{100} = \frac{30}{100} + \frac{7}{100}
= \frac{37}{100}
\]
हिस्सा रँग जाता है --- सैंतीस सौवें भाग।
\end{example}

\section{दशमलव बिंदु}

\begin{definition}[दशमलव लेखन]\label{def:g4:decimals:point}
संख्या $2 + \frac{3}{10} + \frac{7}{100}$ को ऐसे लिखते हैं
\[
2.37
\]
--- \emph{दशमलव बिंदु}\index{दशमलव बिंदु} पूर्ण भाग ($2$) को दशमलव भाग से
अलग करता है; बिंदु के बाद पहला अंक दसवें भाग गिनता है, दूसरा
सौवें भाग। पढ़ना: ``दो दशमलव तीन सात'', या ``दो और सैंतीस सौवें
भाग''।
\end{definition}

\begin{example}\label{ex:g4:decimals:convert}
\[
\frac{7}{10} = 0.7, \qquad
\frac{43}{100} = 0.43, \qquad
\frac{5}{100} = 0.05, \qquad
3 + \frac{4}{10} = 3.4 .
\]
$\frac{5}{100}$ से सावधान: पाँच \emph{सौवें} भागों के लिए दसवें
स्थान पर $0$ चाहिए --- $0.05$, $0.5$ नहीं।
\end{example}

\begin{tikzpicture}[scale=1.35]
  \draw[-Stealth] (-0.2,0) -- (8.4,0);
  \foreach \i in {0,10,20,30,40,50,60,70,80}

  \foreach \x/\l in {0/2, 1/{2.1}, 2/{2.2}, 3/{2.3}, 4/{2.4},
                     5/{2.5}, 6/{2.6}, 7/{2.7}, 8/{2.8}}
    \node[below, font=\footnotesize] at (\x,-0.1) {$\l$};

\end{tikzpicture}

{\small $2$ और $2.8$ के बीच ज़ूम: हर दसवें भाग पर बड़ा निशान, हर
सौवें भाग पर छोटा। संख्या $2.37$, $2.3$ और $2.4$ के बीच,
सातवें छोटे निशान पर बैठती है।}
\end{omfigure}

\begin{example}[पैसा सौवें भागों में है]\label{ex:g4:decimals:money}
एक सेंट एक यूरो (या डॉलर, या पाउंड) का सौवाँ भाग है: $4.85$ की
क़ीमत का मतलब है $4$ पूरे सिक्के और $85$ सौवें भाग। \omterm{def:g4:decimals:point}{दशमलव बिंदु}
तुम रोज़ पैसों में इस्तेमाल करते ही हो: $10$ सेंट $= 0.10$; ढाई
यूरो $= 2.50$; और $0.05$ पाँच सेंट का सिक्का है, पचास का नहीं!
\end{example}

\section{दसवें भागों से तुलना}

\begin{method}[दशमलव संख्याओं की तुलना (पहली मुलाक़ात)]\label{met:g4:decimals:compare}
\begin{enumerate}
  \item पहले पूर्ण भाग की तुलना करो: $3.2 > 2.9$।
  \item पूर्ण भाग बराबर हों: दसवें भाग, फिर सौवें भाग देखो:
        $2.37 < 2.5$, क्योंकि $3$ दसवें भाग $< 5$ दसवें भाग।
  \item दोनों को दो दशमलव अंकों में लिखने से मदद मिलती है:
        $2.5 = 2.50$, और $37 < 50$ सौवें भाग।
\end{enumerate}
जाल यह है: $2.37$ में $2.5$ से ज़्यादा अंक हैं, फिर भी वह
\emph{छोटी} है --- ज़्यादा अंक होने का मतलब बड़ा होना नहीं!
\end{method}

\begin{example}\label{ex:g4:decimals:compare}
$0.4$, $0.35$ और $0.09$ को क्रम में लगाओ: सौवें भागों में
$0.40$, $0.35$, $0.09$; इसलिए
\[
0.09 < 0.35 < 0.4 .
\]
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:decimals:1}
सौ के वर्ग पर कितना रँगा जाएगा: $5$ स्तंभ; $2$ स्तंभ और $3$
ख़ाने; $45$ ख़ाने? उत्तर भिन्नों में दो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:2}
\omterm{def:g4:decimals:point}{दशमलव बिंदु} से लिखो:
$\frac{9}{10}$; \quad $\frac{27}{100}$; \quad $\frac{3}{100}$; \quad
$5 + \frac{6}{10}$; \quad $\frac{60}{100}$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:3}
\omterm{def:g4:fractions:def}{भिन्न} के रूप में लिखो (दसवें या सौवें भाग):
$0.3$; \quad $0.51$; \quad $0.07$; \quad $1.9$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:4}
$6.58$ में: अंक $5$ क्या गिनता है? अंक $8$?
\cref{def:g4:decimals:point} की तरह संख्या को हिस्सों में बाँटो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:5}
$4$ से $5$ तक दसवें भागों में अंकित संख्या रेखा पर रखो:
$4.2$; \ $4.75$ (किन दो निशानों के बीच?); \ $4.9$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:6}
उतार कर $<$, $>$ या $=$ से पूरा करो:
\[
0.6 \;?\; 0.60, \qquad
2.8 \;?\; 2.75, \qquad
0.09 \;?\; 0.1, \qquad
5.3 \;?\; 5.13 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:7}
सबसे छोटी से सबसे बड़ी के क्रम में लिखो: $1.05$; \ $1.5$; \ $0.95$; \ $1.45$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:8}
दशमलव लेखन में लिखो: तीन यूरो पाँच सेंट; बारह यूरो पचास सेंट;
निन्यानवे सेंट। फिर तीनों क़ीमतों को क्रम में लगाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:9}
एक दशमलव अंक वाली कौन-सी दशमलव संख्याएँ ठीक $3$ और $4$ के बीच
हैं? सब गिनाओ। वे कितनी हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:decimals:10}
टॉम कहता है: ``$0.25$, $0.5$ से बड़ी है, क्योंकि $25$, $5$ से
बड़ा है''। सौ के वर्ग से उसकी ग़लती दिखाओ (हर एक कितने ख़ाने
रँगती है?)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:decimals:11}
एक धावक की तीन कूदें $3.6$ m, $3.58$ m और $3.65$ m नापी गईं।
कूदों को सबसे छोटी से सबसे बड़ी के क्रम में लगाओ। किन दो में
ठीक एक मीटर के $\frac{5}{100}$ का \omterm{ex:g1:subtraction:difference}{अंतर} है?
\end{exercise}
