\chapter{रेखाएँ और बहुभुज}\label{ch:g4:geometry}

मापपट्टी, \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका और परकार के साथ ज्यामिति एक हुनर बन जाती
है। यह अध्याय रेखाओं की दो ख़ास स्थितियाँ --- \omterm{def:g4:geometry:def}{लंब} और \omterm{def:g4:geometry:def}{समांतर} ---
औज़ारों के स्तर पर सिखाता है, और बहुभुजों के परिवार की सैर कराता
है। (इन धारणाओं का सिद्धांत \cref{ch:g6:lines} में आता
है।)

\section{लंब और समांतर}

\begin{definition}[लंब, समांतर]\label{def:g4:geometry:def}
दो रेखाएँ \emph{लंब}\index{लंब रेखाएँ} तब होती हैं जब वे
\omterm{def:g3:shapes:rightangle}{समकोण} पर काटती हैं (\omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका से जाँची जाती हैं, छोटे वर्ग
से अंकित)। दो रेखाएँ \emph{समांतर}\index{समांतर रेखाएँ}
तब होती हैं जब वे एक ही दिशा में चलती हैं और कभी नहीं काटतीं ---
पटरी की दो लीकों की तरह, जो हमेशा एक-दूसरे से एक ही दूरी पर हों।
\end{definition}

\begin{method}[समकोण पट्टिका से बनाना]\label{met:g4:geometry:draw}
\emph{रेखा $d$ पर बिंदु $P$ से होकर \omterm{def:g4:geometry:def}{लंब}}: \omterm{def:g3:shapes:rightangle}{समकोण} का एक किनारा
$d$ पर रखो, तब तक सरकाओ जब तक दूसरा किनारा $P$ तक न पहुँचे,
फिर खींच दो।

\emph{$P$ से होकर $d$ के \omterm{def:g4:geometry:def}{समांतर}}: पट्टिका का किनारा $d$ पर रखो,
दूसरे किनारे से मापपट्टी को लीक की तरह सटा दो, पट्टिका को
मापपट्टी के साथ $P$ तक सरकाओ, फिर खींच दो।
\end{method}

\begin{example}[आकृतियों में इन्हें पहचानना]\label{ex:g4:geometry:spot}
आयत में हर भुजा अपनी दोनों पड़ोसी भुजाओं पर \omterm{def:g4:geometry:def}{लंब} है और सामने वाली
भुजा के \omterm{def:g4:geometry:def}{समांतर}। जाल वाला काग़ज़ \omterm{def:g4:geometry:def}{समांतर} और \omterm{def:g4:geometry:def}{लंब} रेखाओं का बना-बनाया
जाल है --- इसीलिए उस पर बनाना इतना आरामदेह लगता
है।
\end{example}

\section{बहुभुज}

\begin{definition}[बहुभुज]\label{def:g4:geometry:polygon}
\emph{बहुभुज}\index{बहुभुज} एक बंद आकृति है जिसकी सीमा सीधे
रेखाखंडों से बनी होती है, ये उसकी \emph{भुजाएँ} हैं; कोनों के
बिंदु \emph{शीर्ष}\index{शीर्ष} कहलाते हैं। बहुभुजों के नाम उनकी भुजाओं की
संख्या से रखे जाते हैं: \emph{त्रिभुज} ($3$),
\emph{चतुर्भुज}\index{चतुर्भुज} ($4$), \emph{पंचभुज}\index{पंचभुज} ($5$),
\emph{षट्भुज}\index{षट्भुज} ($6$), \emph{अष्टभुज}\index{अष्टभुज}
($8$)।
\end{definition}

\emph{सम} हैं: बराबर भुजाएँ और बराबर कोने)। वृत्त \omterm{def:g4:geometry:polygon}{बहुभुज} नहीं

\end{omfigure}

\begin{example}[बहुभुज का वर्णन]\label{ex:g4:geometry:describe}
एक काम का खेल: किसी आकृति का इतना ठीक-ठीक वर्णन करो कि सहपाठी
उसे देखे बिना दोबारा बना सके। ``एक \omterm{def:g4:geometry:polygon}{चतुर्भुज} $ABCD$ जिसमें
$AB = 5$ cm, $A$ पर \omterm{def:g3:shapes:rightangle}{समकोण}, $AD = 3$ cm\,\dots'' ---
संदेश शब्दावली के भरोसे ही चलता है।
\end{example}

\section{वृत्त और परकार}

\begin{definition}[वृत्त]\label{def:g4:geometry:circle}
केंद्र $O$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $3$ cm वाला \emph{वृत्त} उन सारे बिंदुओं
से बनता है जो $O$ से ठीक $3$ cm दूर हैं। परकार उसे खींचता है:
नोक $O$ पर, खुलाव $3$ cm, और घुमा दो। \emph{व्यास}\index{व्यास} वह रेखाखंड है
जो केंद्र से होकर वृत्त के दो बिंदुओं को जोड़ता है; वह \omterm{def:g3:shapes:shapes}{त्रिज्या}
से दुगुना नापता है।
\end{definition}

\begin{example}[परकार, दूरी सँभालने वाला औज़ार]\label{ex:g4:geometry:compass}
परकार वृत्त खींचने से कहीं ज़्यादा करता है: वह \emph{दूरियाँ
ढोता} है। किसी बिंदु $A$ से $4$ cm दूर के सारे बिंदु अंकित करने
हों तो केंद्र $A$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $4$ cm का वृत्त खींच दो; बिना
मापपट्टी के दो रेखाखंडों की तुलना करनी हो तो परकार को एक पर खोलो और दूसरे पर आज़मा लो।
\end{example}

\begin{method}[जाल पर आकृति दोबारा बनाना]\label{met:g4:geometry:reproduce}
\begin{enumerate}
  \item शुरू का एक \omterm{def:g4:geometry:polygon}{शीर्ष} चुनो और ख़ाली जाल पर उसकी जगह तय करो;
  \item आकृति पर शीर्ष-दर-शीर्ष चलो, जाल के कदम गिनते हुए
        (``$3$ दाईं ओर, $2$ ऊपर'');
  \item शीर्षों को उसी क्रम में जोड़ो, और नमूने से मिलाओ:
        वही लंबाइयाँ, वही दिशाएँ।
\end{enumerate}
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g4:geometry:1}
एक रेखा $d$ और उससे बाहर एक बिंदु $P$ बनाओ। $P$ से होकर $d$ पर
\omterm{def:g4:geometry:def}{लंब} बनाओ, फिर $P$ से होकर $d$ के \omterm{def:g4:geometry:def}{समांतर} रेखा (मापपट्टी-लीक वाले
तरीक़े से)। \omterm{def:g3:shapes:rightangle}{समकोण} अंकित करो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:2}
बड़े अक्षर E, H, N, T, Z देखो। हर एक में \omterm{def:g4:geometry:def}{समांतर} लकीरों का एक
जोड़ा और \omterm{def:g4:geometry:def}{लंब} लकीरों का एक जोड़ा ढूँढ़ो, जहाँ वे हों।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:3}
हर \omterm{def:g4:geometry:polygon}{बहुभुज} का नाम बताओ: $6$ भुजाएँ; $4$ भुजाएँ; $3$ भुजाएँ;
$8$ भुजाएँ; $5$ भुजाएँ। हर एक के कितने \emph{\omterm{def:g4:geometry:polygon}{शीर्ष}} होते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:4}
कोई भी \omterm{def:g4:geometry:polygon}{पंचभुज} बनाओ (ज़रूरी नहीं कि सम हो) और उसके शीर्षों के नाम
$A$, $B$, $C$, $D$, $E$ रखो। उसकी कितनी भुजाएँ हैं? $A$ से उसके
विकर्ण खींचो ($A$ को उन शीर्षों से जोड़ने वाले रेखाखंड जो पड़ोसी
नहीं हैं): वे कितने हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:5}
केंद्र $O$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $4$ cm का वृत्त बनाओ; एक \omterm{def:g4:geometry:circle}{व्यास} $[AB]$;
और वृत्त का एक बिंदु $M$ जो $[AB]$ पर न हो। बिना नापे $OA$,
$OM$ और $AB$ की लंबाइयाँ बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:6}
परकार को दूरी सँभालने वाले औज़ार की तरह इस्तेमाल करके
(\cref{ex:g4:geometry:compass}) किसी सहपाठी के बनाए त्रिभुज की
भुजाओं की तुलना करो और तय करो कि उसकी दो भुजाएँ बराबर हैं या नहीं।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:7}
जाल वाले काग़ज़ पर यह रास्ता दोबारा बनाओ और उसे बंद करके \omterm{def:g4:geometry:polygon}{बहुभुज}
बनाओ: बिंदु $A$ से शुरू करो; $4$ दाईं ओर; $2$ ऊपर; $3$ बाईं ओर;
$2$ ऊपर; फिर सीधी रेखा में $A$ पर वापस। जो \omterm{def:g4:geometry:polygon}{बहुभुज} बनता है उसका
नाम क्या है (भुजाएँ गिनो)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:8}
सही या ग़लत? चित्र से समझाओ।
``दो \omterm{def:g4:geometry:def}{लंब रेखाएँ} हमेशा काटती हैं।''
``दो \omterm{def:g4:geometry:def}{समांतर रेखाएँ} कभी नहीं काटतीं।''
``कोई रेखा एक रेखा के \omterm{def:g4:geometry:def}{समांतर} और दूसरी पर \omterm{def:g4:geometry:def}{लंब} हो
सकती है।''
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:9}
एक रेखा $d$ बनाओ, फिर $d$ पर \omterm{def:g4:geometry:def}{लंब} रेखा $e$, फिर $e$ पर \omterm{def:g4:geometry:def}{लंब} रेखा
$f$। $d$ और $f$ को देखो: क्या दिखाई देता है?
(यह अवलोकन \cref{ch:g6:lines} में प्रमेय बन जाता है।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:10}
नीचे बताई आकृति को दिखाए बिना सहपाठी के लिए शब्दों में बताओ:
$4$ cm भुजा वाला वर्ग, और उसके बीच में केंद्रित $2$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या}
का वृत्त (जो चारों भुजाओं को छूता है)। वर्णन को क्रम से लिखे
निर्देशों के रूप में लिखो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:11}
\omterm{def:g4:geometry:polygon}{चतुर्भुज} के कितने विकर्ण होते हैं? \omterm{def:g4:geometry:polygon}{पंचभुज} के? \omterm{def:g4:geometry:polygon}{षट्भुज} के?
तीनों बनाओ, ध्यान से गिनो, और जो प्रतिरूप दिखे उसे बताओ।
\end{exercise}
