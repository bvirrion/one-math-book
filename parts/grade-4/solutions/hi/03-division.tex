\section*{अध्याय \ref{ch:g4:division} --- भाग}

\begin{solution}{exo:g4:division:1}
$96 \div 4$: $4 \times 20 = 80$ और $4 \times 30 = 120$ के बीच,
इसलिए \omterm{def:g3:division:remainder}{भागफल} दो अंकों का है; लंबा भाग $24$ देता है, \omterm{def:g3:division:remainder}{शेषफल}
$0$।

$87 \div 5$: $5 \times 10 = 50$ और $5 \times 20 = 100$ के बीच;
\omterm{def:g3:division:remainder}{भागफल} $17$, \omterm{def:g3:division:remainder}{शेषफल} $2$ ($87 = 5 \times 17 + 2$)।
\end{solution}

\begin{solution}{exo:g4:division:2}
$672 \div 4 = 168$, \omterm{def:g3:division:remainder}{शेषफल} $0$ (जाँच: $4 \times 168 = 672$)।

$925 \div 7 = 132$, \omterm{def:g3:division:remainder}{शेषफल} $1$ (जाँच:
$7 \times 132 + 1 = 925$)।

$804 \div 6 = 134$, \omterm{def:g3:division:remainder}{शेषफल} $0$ (जाँच: $6 \times 134 = 804$)।
\end{solution}

\begin{solution}{exo:g4:division:3}
$816 \div 4 = 204$ (दहाई वाला कदम $1 \div 4$ है: \omterm{def:g1:counting:zero}{शून्य} बार ---
वह $0$ लिखो)।

$420 \div 7 = 60$।

$2\,512 \div 5 = 502$, \omterm{def:g3:division:remainder}{शेषफल} $2$ (जाँच:
$5 \times 502 + 2 = 2\,512$)।
\end{solution}

\begin{solution}{exo:g4:division:4}
$85 = 9 \times 9 + 4$ (\omterm{def:g3:division:remainder}{भागफल} $9$, \omterm{def:g3:division:remainder}{शेषफल} $4$)।

$1\,000 = 3 \times 333 + 1$।
\end{solution}

\begin{solution}{exo:g4:division:5}
$156 \div 6 = 26$: छब्बीस डिब्बे भरते हैं, कुछ नहीं बचता।
\end{solution}

\begin{solution}{exo:g4:division:6}
$250 = 8 \times 31 + 2$: इकतीस टुकड़े, और $2$ cm रिबन बच
जाता है।
\end{solution}

\begin{solution}{exo:g4:division:7}
$375 = 52 \times 7 + 11$: सात बसें $364$ विद्यार्थियों को ले
जाती हैं, $11$ बचते हैं --- ``और एक'' वाला मामला: $8$ बसें चाहिए।
\end{solution}

\begin{solution}{exo:g4:division:8}
$234 \div 3 = 78$: हर एक को $78$ कंचे मिलते हैं। जाँच:
$3 \times 78 = 234$।
\end{solution}

\begin{solution}{exo:g4:division:9}
भाज्य $= 7 \times 58 + 3 = 406 + 3 = 409$।
\end{solution}

\begin{solution}{exo:g4:division:10}
ख़ाने: $438 = 9 \times 48 + 6$: $48$ भरे ख़ाने और $6$ किताबें
और --- $49$ ख़ाने चाहिए। अलमारियाँ:
$49 = 6 \times 8 + 1$: आठ भरी अलमारियाँ और एक ख़ाना अतिरिक्त ---
$9$ अलमारियाँ ख़रीदनी होंगी।
\end{solution}

\begin{solution}{exo:g4:division:11}
\omterm{def:g3:division:remainder}{भागफल} $=$ \omterm{def:g3:division:remainder}{शेषफल} $= r$, और $r < 8$, इसलिए $r$ का मान
$0, 1, \dots, 7$ है और भाज्य $8r + r = 9r$ है: संभव भाज्य हैं
$0$, $9$, $18$, $27$, $36$, $45$, $54$, $63$।

\end{solution}
