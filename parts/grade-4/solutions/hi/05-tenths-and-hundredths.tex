\section*{अध्याय \ref{ch:g4:decimals} --- दसवें और सौवें भाग}

\begin{solution}{exo:g4:decimals:1}
$5$ स्तंभ: $\frac{5}{10}$ (यानी $\frac{50}{100}$)। $2$ स्तंभ
और $3$ ख़ाने: $\frac{23}{100}$। $45$ ख़ाने: $\frac{45}{100}$।
\end{solution}

\begin{solution}{exo:g4:decimals:2}
$0.9$; \quad $0.27$; \quad $0.03$; \quad $5.6$; \quad
$0.60 = 0.6$।
\end{solution}

\begin{solution}{exo:g4:decimals:3}
$0.3 = \frac{3}{10}$; \quad $0.51 = \frac{51}{100}$; \quad
$0.07 = \frac{7}{100}$; \quad $1.9 = 1 + \frac{9}{10} =
\frac{19}{10}$।
\end{solution}

\begin{solution}{exo:g4:decimals:4}
$5$ दसवें भाग गिनता है, $8$ सौवें भाग:
\[
6.58 = 6 + \frac{5}{10} + \frac{8}{100}.
\]
\end{solution}

\begin{solution}{exo:g4:decimals:5}
$4.2$: $4$ के बाद दूसरे निशान पर। $4.75$: निशानों
$4.7$ और $4.8$ के ठीक बीच। $4.9$: $5$ से एक निशान पहले।
\end{solution}

\begin{solution}{exo:g4:decimals:6}
$0.6 = 0.60$; \quad $2.8 > 2.75$ ($2.80$ से तुलना करो); \quad
$0.09 < 0.1$ ($9$ सौवें भाग बनाम $10$); \quad
$5.3 > 5.13$।
\end{solution}

\begin{solution}{exo:g4:decimals:7}
$0.95 < 1.05 < 1.45 < 1.5$।
\end{solution}

\begin{solution}{exo:g4:decimals:8}
$3.05$; \quad $12.50$; \quad $0.99$। क्रम:
$0.99 < 3.05 < 12.50$।
\end{solution}

\begin{solution}{exo:g4:decimals:9}
$3.1$, $3.2$, $3.3$, $3.4$, $3.5$, $3.6$, $3.7$, $3.8$, $3.9$ ---
नौ संख्याएँ।
\end{solution}

\begin{solution}{exo:g4:decimals:10}
सौ के वर्ग पर $0.25$, $25$ ख़ाने रँगती है पर $0.5 = 0.50$,
$50$ ख़ाने: $0.5$ ही बड़ी है। टॉम ने $25$ और $5$ की तुलना पूर्ण
संख्याओं की तरह की; उसे $25$ और $50$ सौवें भागों की तुलना करनी चाहिए।
\end{solution}

\begin{solution}{exo:g4:decimals:11}
क्रम: $3.58 < 3.6 < 3.65$। कूदों $3.6$ और $3.65$ में एक मीटर के
$0.05 = \frac{5}{100}$ का \omterm{ex:g1:subtraction:difference}{अंतर} है (पाँच सेंटीमीटर)।

\end{solution}
