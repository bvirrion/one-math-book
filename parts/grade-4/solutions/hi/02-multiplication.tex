\section*{अध्याय \ref{ch:g4:mult} --- बड़ी संख्याओं का गुणा}

\begin{solution}{exo:g4:mult:1}
$470$; \quad $2\,300$; \quad $720$; \quad $4\,500$; \quad
$2\,100$।
\end{solution}

\begin{solution}{exo:g4:mult:2}
$34 \times 21 = 34 \times 20 + 34 = 680 + 34 = 714$; खाने भी
वही दो पंक्तियाँ देते हैं: $34$ और $680$।
\end{solution}

\begin{solution}{exo:g4:mult:3}
$63 \times 24 = 1\,512$; \quad
$85 \times 47 = 3\,995$; \quad
$126 \times 53 = 6\,678$।
\end{solution}

\begin{solution}{exo:g4:mult:4}
$208 \times 34$: अंदाज़ा $200 \times 34 = 6\,800$; ठीक-ठीक
$7\,072$।

$517 \times 62$: अंदाज़ा $500 \times 60 = 30\,000$; ठीक-ठीक
$32\,054$।
\end{solution}

\begin{solution}{exo:g4:mult:5}
$45 \times 11 = 450 + 45 = 495$।

$32 \times 99 = 3\,200 - 32 = 3\,168$।

$24 \times 45 = 12 \times 90 = 1\,080$ (या एक बार और:
$6 \times 180 = 1\,080$)।
\end{solution}

\begin{solution}{exo:g4:mult:6}
$26 \times 18 = 468$ सीटें।
\end{solution}

\begin{solution}{exo:g4:mult:7}
अंदाज़ा: $30 \times 50 = 1\,500$। ठीक-ठीक:
$32 \times 48 = 1\,536$ पेटियाँ।
\end{solution}

\begin{solution}{exo:g4:mult:8}
वह भूल गई कि $23$ का $2$ \emph{दहाई} गिनता है: दूसरी पंक्ति
$57 \times 20 = 1\,140$ होनी चाहिए, $114$ नहीं। सही कुल:
$171 + 1\,140 = 1\,311$।
\end{solution}

\begin{solution}{exo:g4:mult:9}
ट्यूलिप: $14 \times 35 = 490$। नरगिस: $12 \times 28 = 336$।
कुल: $490 + 336 = 826$ फूल।
\end{solution}

\begin{solution}{exo:g4:mult:10}
अंदाज़ा: $70 \times 30 = 2\,100$ kg। ठीक-ठीक:
$67 \times 30 = 2\,010$ kg।
\end{solution}

\begin{solution}{exo:g4:mult:11}
इकाई वर्गों के $15 \times 12$ आयत को एक खड़ी रेखा से
$15 \times 10$ और $15 \times 2$ के हिस्सों में काटा जा सकता है:
दोनों हिस्से गिनने पर $150 + 30 = 180$ मिलता है। दूसरी कटाई भी
उतनी ही अच्छी चलती है, जैसे $15 = 10 + 5$:
$10 \times 12 + 5 \times 12 = 120 + 60 = 180$। आयत को कैसे भी
काटो, वर्गों की कुल संख्या नहीं बदलती।
\end{solution}
