\section*{अध्याय \ref{ch:g4:measure} --- माप और परिमाप}

\begin{solution}{exo:g4:measure:1}
$7$ m $= 700$ cm; \quad $3.2$ km $= 3\,200$ m; \quad
$85$ mm $= 8.5$ cm; \quad $4\,500$ m $= 4.5$ km।
\end{solution}

\begin{solution}{exo:g4:measure:2}
$2$ kg $= 2\,000$ g; \quad $1\,800$ g $= 1$ kg $800$ g; \quad
$3.5$ L $= 350$ cL; \quad $25$ cL $= 0.25$ L।
\end{solution}

\begin{solution}{exo:g4:measure:3}
आयत: $2 \times (9 + 5) = 28$ cm। वर्ग:
$4 \times 7.5 = 30$ cm। त्रिभुज: $6 + 8 + 10 = 24$ cm।
\end{solution}

\begin{solution}{exo:g4:measure:4}
लंबाई $+$ चौड़ाई $= 26 \div 2 = 13$ cm, इसलिए चौड़ाई
$13 - 8 = 5$ cm है।
\end{solution}

\begin{solution}{exo:g4:measure:5}
कई L-आकार चलते हैं; $4 \times 2$ के आयत और $2 \times 1$ के पैर
से बने $10$-खाने वाले L की सीमा पर चलने पर $16$ इकाइयाँ मिलती
हैं। (कोई भी सही आकृति और ध्यान से की गई गिनती सही है; \omterm{def:g4:measure:perimeter}{परिमाप}
चुने हुए L पर निर्भर करता है।)
\end{solution}

\begin{solution}{exo:g4:measure:6}
$3 \times 4$: \omterm{def:g4:measure:perimeter}{परिमाप} $14$। $2 \times 6$: \omterm{def:g4:measure:perimeter}{परिमाप} $16$।
($1 \times 12$: \omterm{def:g4:measure:perimeter}{परिमाप} $26$।) आकृति वर्ग के जितनी पास होगी,
उसी \omterm{def:g4:measure:area}{क्षेत्रफल} के लिए \omterm{def:g4:measure:perimeter}{परिमाप} उतना ही छोटा होगा।
\end{solution}

\begin{solution}{exo:g4:measure:7}
$6 \times 3$ के आयत का \omterm{def:g3:division:half}{आधा}: $18 \div 2 = 9$ खाने।
\end{solution}

\begin{solution}{exo:g4:measure:8}
$10{:}20 \to 12{:}45$: $2$ घंटे $25$ मिनट। $7{:}35 \to 9{:}10$:
$1$ घंटा $35$ मिनट। फ़िल्म: $20{:}30 + 1$ घंटा $50$ मिनट $= 22{:}20$।
\end{solution}

\begin{solution}{exo:g4:measure:9}
एक चक्कर: $2 \times (60 + 45) = 210$ m। तीन चक्कर:
$3 \times 210 = 630$ m।
\end{solution}

\begin{solution}{exo:g4:measure:10}
\omterm{def:g4:measure:perimeter}{परिमाप}: $4 \times 85 = 340$ m; फाटक घटाकर:
$340 - 4 = 336$ m बाड़।
\end{solution}

\begin{solution}{exo:g4:measure:11}
दोनों दावे \emph{ग़लत} हैं। \omterm{def:g4:measure:perimeter}{परिमाप} वही, \omterm{def:g4:measure:area}{क्षेत्रफल} अलग:
$4 \times 2$ का आयत और इस अध्याय की आकृति वाला L-आकार (दोनों का
\omterm{def:g4:measure:perimeter}{परिमाप} $12$; \omterm{def:g4:measure:area}{क्षेत्रफल} $8$ और $6$)। \omterm{def:g4:measure:area}{क्षेत्रफल} वही, \omterm{def:g4:measure:perimeter}{परिमाप}
अलग: $4 \times 4$ का वर्ग और $8 \times 2$ का आयत
(दोनों का \omterm{def:g4:measure:area}{क्षेत्रफल} $16$; \omterm{def:g4:measure:perimeter}{परिमाप} $16$ और $20$)।

\end{solution}
