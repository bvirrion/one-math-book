\section*{अध्याय \ref{ch:g4:numbers} --- बड़ी संख्याएँ}

\begin{solution}{exo:g4:numbers:1}
$3\,520\,000$; \quad $7\,000\,008$; \quad $240\,060$।
\end{solution}

\begin{solution}{exo:g4:numbers:2}
$5\,204\,300$: पाँच मिलियन दो सौ चार हज़ार तीन सौ।

$10\,050\,000$: दस मिलियन पचास हज़ार।

$999\,999$: नौ सौ निन्यानवे हज़ार नौ सौ
निन्यानवे।
\end{solution}

\begin{solution}{exo:g4:numbers:3}
दस हज़ार का अंक: $7$। मिलियन समूह का सैकड़ा: सात अंकों की संख्या
में सौ-मिलियन का अंक होता ही नहीं --- $8\,472\,156$ के मिलियन
समूह में केवल अंक $8$ है (मिलियन की इकाई)। समाए हुए हज़ार:
$8\,472$ (क्योंकि
$8\,472\,156 = 8\,472 \times 1\,000 + 156$)।
\end{solution}

\begin{solution}{exo:g4:numbers:4}
$980\,000 < 1\,020\,000$; \quad
$3\,456\,000 < 3\,465\,000$ (हज़ार समूह $456 < 465$); \quad
$7\,090\,050 < 7\,090\,500$।
\end{solution}

\begin{solution}{exo:g4:numbers:5}
$980\,500 < 2\,030\,000 < 2\,299\,999 < 2\,300\,000$।
\end{solution}

\begin{solution}{exo:g4:numbers:6}
दहाई तक: $63\,780$। सैकड़े तक: $63\,800$। हज़ार तक:
$64\,000$। दस हज़ार तक: $60\,000$।
\end{solution}

\begin{solution}{exo:g4:numbers:7}
$5\,000\,000$; \quad $6\,000\,000$ ($499\,999$ आधे मिलियन से
कम पड़ता है); \quad $10\,000\,000$ (ठीक बीच वाला ऊपर गोल होता है)।
\end{solution}

\begin{solution}{exo:g4:numbers:8}
$998\,500 \to 999\,500 \to 1\,000\,500 \to 1\,001\,500$: दूसरा
कदम मिलियन पार कर जाता है, और बाईं ओर एक नया समूह जग जाता है।

$950\,000 \to 1\,050\,000 \to 1\,150\,000$: यहाँ पहला ही कदम पार
कर जाता है।
\end{solution}

\begin{solution}{exo:g4:numbers:9}
दोनों सही हैं। ``$80\,000$ से ज़्यादा'' दस हज़ार तक नीचे गोल
करता है ($81\,338 \to 80\,000$); ``लगभग $81\,000$'' नज़दीकी
हज़ार तक गोल करता है ($81\,338 \to 81\,000$)।
\end{solution}

\begin{solution}{exo:g4:numbers:10}
सबसे बड़ी: $7\,654\,321$। सबसे छोटी: $1\,234\,567$।
$4\,000\,000$ के सबसे नज़दीक: नीचे की ओर सबसे अच्छी $3\,765\,421$
है, जो $234\,579$ दूर है, जबकि ऊपर की ओर $4\,123\,567$ केवल
$123\,567$ दूर है: सबसे नज़दीक $4\,123\,567$ है।
\end{solution}

\begin{solution}{exo:g4:numbers:11}
नज़दीकी हज़ार तक गोल करने पर $49\,500$ (ठीक बीच वाली ऊपर जाती
है) से $50\,499$ तक की सारी संख्याएँ $50\,000$ देती हैं। सबसे
बड़ी: $50\,499$; सबसे छोटी: $49\,500$।
\end{solution}
