\section*{第 \ref{ch:g4:decimals} 章——十分位与百分位}

\begin{solution}{exo:g4:decimals:1}
$5$ 列：$\frac{5}{10}$（即 $\frac{50}{100}$）。$2$ 列
和 $3$ 格：$\frac{23}{100}$。$45$ 格：$\frac{45}{100}$。
\end{solution}

\begin{solution}{exo:g4:decimals:2}
$0.9$；\quad $0.27$；\quad $0.03$；\quad $5.6$；\quad
$0.60 = 0.6$。
\end{solution}

\begin{solution}{exo:g4:decimals:3}
$0.3 = \frac{3}{10}$；\quad $0.51 = \frac{51}{100}$；\quad
$0.07 = \frac{7}{100}$；\quad $1.9 = 1 + \frac{9}{10} =
\frac{19}{10}$。
\end{solution}

\begin{solution}{exo:g4:decimals:4}
$5$ 表示十分位，$8$ 表示百分位：
\[
6.58 = 6 + \frac{5}{10} + \frac{8}{100}.
\]
\end{solution}

\begin{solution}{exo:g4:decimals:5}
$4.2$：在 $4$ 之后的第二个刻度上。$4.75$：介于
$4.7$ 与 $4.8$ 之间，恰好居中。$4.9$：在 $5$ 前面一个刻度上。
\end{solution}

\begin{solution}{exo:g4:decimals:6}
$0.6 = 0.60$；\quad $2.8 > 2.75$（比较 $2.80$）；\quad
$0.09 < 0.1$（$9$ 个百分之一与 $10$ 个百分之一比较）；\quad
$5.3 > 5.13$。
\end{solution}

\begin{solution}{exo:g4:decimals:7}
$0.95 < 1.05 < 1.45 < 1.5$。
\end{solution}

\begin{solution}{exo:g4:decimals:8}
$3.05$；\quad $12.50$；\quad $0.99$。顺序为：
$0.99 < 3.05 < 12.50$。
\end{solution}

\begin{solution}{exo:g4:decimals:9}
$3.1$、$3.2$、$3.3$、$3.4$、$3.5$、$3.6$、$3.7$、$3.8$、$3.9$——
共 9 个数。
\end{solution}

\begin{solution}{exo:g4:decimals:10}
在百格图上，$0.25$ 涂了 $25$ 格，而 $0.5 = 0.50$
涂了 $50$ 格：$0.5$ 较大。汤姆把 $25$ 与
$5$ 当作整数来比较；他应比较 $25$ 个百分之一和 $50$ 个百分之一。
\end{solution}

\begin{solution}{exo:g4:decimals:11}
顺序为：$3.58 < 3.6 < 3.65$。$3.6$ 与 $3.65$ 相差
$0.05 = \frac{5}{100}$ 米，即 5 厘米。
\end{solution}
