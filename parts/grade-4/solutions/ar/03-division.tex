\section*{الفصل \ref{ch:g4:division} --- القسمة}

\begin{solution}{exo:g4:division:1}
$96 \div 4$: بين $4 \times 20 = 80$ و $4 \times 30 = 120$، إذن
\omterm{def:g3:division:remainder}{خارج القسمة} من رقمين؛ والقسمة المطوّلة تعطي $24$، \omterm{def:g3:division:remainder}{والباقي}
$0$.

$87 \div 5$: بين $5 \times 10 = 50$ و $5 \times 20 = 100$؛
\omterm{def:g3:division:remainder}{وخارج القسمة} $17$، \omterm{def:g3:division:remainder}{والباقي} $2$ ($87 = 5 \times 17 + 2$).
\end{solution}

\begin{solution}{exo:g4:division:2}
$672 \div 4 = 168$، \omterm{def:g3:division:remainder}{والباقي} $0$ (التحقّق: $4 \times 168 = 672$).

$925 \div 7 = 132$، \omterm{def:g3:division:remainder}{والباقي} $1$ (التحقّق:
$7 \times 132 + 1 = 925$).

$804 \div 6 = 134$، \omterm{def:g3:division:remainder}{والباقي} $0$ (التحقّق: $6 \times 134 = 804$).
\end{solution}

\begin{solution}{exo:g4:division:3}
$816 \div 4 = 204$ (وخطوة العشرات هي $1 \div 4$: \omterm{def:g1:counting:zero}{صفر} مرة ---
فاكتب $0$).

$420 \div 7 = 60$.

$2\,512 \div 5 = 502$، \omterm{def:g3:division:remainder}{والباقي} $2$ (التحقّق:
$5 \times 502 + 2 = 2\,512$).
\end{solution}

\begin{solution}{exo:g4:division:4}
$85 = 9 \times 9 + 4$ (\omterm{def:g3:division:remainder}{خارج القسمة} $9$، \omterm{def:g3:division:remainder}{والباقي} $4$).

$1\,000 = 3 \times 333 + 1$.
\end{solution}

\begin{solution}{exo:g4:division:5}
$156 \div 6 = 26$: تمتلئ ستّ وعشرون علبة، ولا يبقى شيء.
\end{solution}

\begin{solution}{exo:g4:division:6}
$250 = 8 \times 31 + 2$: إحدى وثلاثون قطعة، ويفضل $2$ سم من
الشريط.
\end{solution}

\begin{solution}{exo:g4:division:7}
$375 = 52 \times 7 + 11$: سبع حافلات تحمل $364$ تلميذًا، ويبقى
$11$ --- وهي حالة «زائد واحد»: فتلزم $8$ حافلات.
\end{solution}

\begin{solution}{exo:g4:division:8}
$234 \div 3 = 78$: ينال كل واحد $78$ كرة. والتحقّق:
$3 \times 78 = 234$.
\end{solution}

\begin{solution}{exo:g4:division:9}
المقسوم $= 7 \times 58 + 3 = 406 + 3 = 409$.
\end{solution}

\begin{solution}{exo:g4:division:10}
الرفوف: $438 = 9 \times 48 + 6$: $48$ رفًّا ممتلئًا و $6$ كتب
زيادة --- فيلزم $49$ رفًّا. والخزائن:
$49 = 6 \times 8 + 1$: ثماني خزائن ممتلئة ورفّ واحد زيادة ---
فيجب شراء $9$ خزائن.
\end{solution}

\begin{solution}{exo:g4:division:11}
\omterm{def:g3:division:remainder}{خارج القسمة} $=$ \omterm{def:g3:division:remainder}{الباقي} $= r$ مع $r < 8$، إذن $r$ هو $0, 1, \dots,
7$ والمقسوم هو $8r + r = 9r$: فالمقسومات الممكنة هي
$0$ و $9$ و $18$ و $27$ و $36$ و $45$ و $54$ و $63$.
\end{solution}
