\section*{الفصل \ref{ch:g4:fractions} --- الكسور الأولى}

\begin{solution}{exo:g4:fractions:1}
البيتزا: بقي $\frac56$. والشوكولاتة: أُكل $\frac38$. والعلم:
لوّن $\frac13$.
\end{solution}

\begin{solution}{exo:g4:fractions:2}
$\frac35$ يلوّن ثلاثة أجزاء، و $\frac25$ لا يلوّن إلا جزأين:
فالتلوين الأكبر هو $\frac35$.
\end{solution}

\begin{solution}{exo:g4:fractions:3}
نصف $18$: $9$. وربع $20$: $5$. وثلث $21$: $7$.
و $\frac34$ من $20$: ثلاثة أرباع، أي $3 \times 5 = 15$.
\end{solution}

\begin{solution}{exo:g4:fractions:4}
$\frac13$: خطوة واحدة؛ و $\frac33$: على $1$؛ و $\frac53$: خطوتان بعد
$1$. \omterm{def:g4:fractions:def}{والكسر} الذي يقع على $2$ بالضبط هو $\frac63$.
\end{solution}

\begin{solution}{exo:g4:fractions:5}
$\frac38 < \frac68$؛ \quad $\frac55 = 1$؛ \quad $\frac74 > 1$؛
\quad $\frac12 = \frac24$.
\end{solution}

\begin{solution}{exo:g4:fractions:6}
$\frac26 + \frac36 = \frac56$؛ \quad
$\frac58 - \frac28 = \frac38$؛ \quad
$\frac14 + \frac24 = \frac34$؛ \quad
$1 = \frac33$، إذن $1 - \frac13 = \frac23$.
\end{solution}

\begin{solution}{exo:g4:fractions:7}
$\frac54 = 1 + \frac14$؛ \quad
$\frac73 = 2 + \frac13$؛ \quad
$\frac92 = 4 + \frac12$.
\end{solution}

\begin{solution}{exo:g4:fractions:8}
نصف ساعة: $30$ دقيقة. والربع: $15$ دقيقة. وثلاثة الأرباع: $45$
دقيقة. والثلث: $20$ دقيقة.
\end{solution}

\begin{solution}{exo:g4:fractions:9}
$\frac14$ من $24$ هو $6$ تلاميذ يلبسون نظارات؛ و $24 - 6 = 18$
لا يلبسونها.
\end{solution}

\begin{solution}{exo:g4:fractions:10}
معًا: $\frac38 + \frac48 = \frac78$. \omterm{def:g3:division:remainder}{والباقي}: $\frac18$. وسامي أكل
أكثر ($4$ أثمان مقابل $3$).
\end{solution}

\begin{solution}{exo:g4:fractions:11}
ممكن فعلًا: فنصف بيتزا صغيرة قد يكون طعامًا أقلّ من
ربع بيتزا ضخمة. والكسور تقارن أجزاءً \emph{من الوحدة
نفسها}؛ فقبل مقارنة $\frac12$ و $\frac14$ «من شيء ما»،
يجب أن نعرف أن الشيئين متساويان.
\end{solution}
