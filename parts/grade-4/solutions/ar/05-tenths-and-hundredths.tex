\section*{الفصل \ref{ch:g4:decimals} --- الأعشار والأجزاء من مئة}

\begin{solution}{exo:g4:decimals:1}
$5$ أعمدة: $\frac{5}{10}$ (أي $\frac{50}{100}$). و $2$ من الأعمدة
و $3$ خانات: $\frac{23}{100}$. و $45$ خانة: $\frac{45}{100}$.
\end{solution}

\begin{solution}{exo:g4:decimals:2}
$0.9$؛ \quad $0.27$؛ \quad $0.03$؛ \quad $5.6$؛ \quad
$0.60 = 0.6$.
\end{solution}

\begin{solution}{exo:g4:decimals:3}
$0.3 = \frac{3}{10}$؛ \quad $0.51 = \frac{51}{100}$؛ \quad
$0.07 = \frac{7}{100}$؛ \quad $1.9 = 1 + \frac{9}{10} =
\frac{19}{10}$.
\end{solution}

\begin{solution}{exo:g4:decimals:4}
الرقم $5$ يعدّ الأعشار، والرقم $8$ يعدّ الأجزاء من مئة:
\[
6.58 = 6 + \frac{5}{10} + \frac{8}{100}.
\]
\end{solution}

\begin{solution}{exo:g4:decimals:5}
$4.2$: عند العلامة الثانية بعد $4$. و $4.75$: بين العلامتين
$4.7$ و $4.8$، في المنتصف. و $4.9$: علامة واحدة قبل $5$.
\end{solution}

\begin{solution}{exo:g4:decimals:6}
$0.6 = 0.60$؛ \quad $2.8 > 2.75$ (قارن $2.80$)؛ \quad
$0.09 < 0.1$ ($9$ أجزاء من مئة مقابل $10$)؛ \quad
$5.3 > 5.13$.
\end{solution}

\begin{solution}{exo:g4:decimals:7}
$0.95 < 1.05 < 1.45 < 1.5$.
\end{solution}

\begin{solution}{exo:g4:decimals:8}
$3.05$؛ \quad $12.50$؛ \quad $0.99$. والترتيب:
$0.99 < 3.05 < 12.50$.
\end{solution}

\begin{solution}{exo:g4:decimals:9}
$3.1$، $3.2$، $3.3$، $3.4$، $3.5$، $3.6$، $3.7$، $3.8$، $3.9$ ---
تسعة أعداد.
\end{solution}

\begin{solution}{exo:g4:decimals:10}
على مربع المئة، يلوّن $0.25$ عدد $25$ خانة، أما $0.5 = 0.50$
فيلوّن $50$ خانة: فالعدد $0.5$ هو الأكبر. وقد قارن عمر $25$ و
$5$ كعددين صحيحين؛ وكان عليه أن يقارن $25$ و $50$ جزءًا من مئة.
\end{solution}

\begin{solution}{exo:g4:decimals:11}
الترتيب: $3.58 < 3.6 < 3.65$. والقفزتان $3.6$ و $3.65$ يفرق
بينهما $0.05 = \frac{5}{100}$ من المتر (أي خمسة سنتيمترات).
\end{solution}
