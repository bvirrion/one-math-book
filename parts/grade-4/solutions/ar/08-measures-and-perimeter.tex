\section*{الفصل \ref{ch:g4:measure} --- القياسات والمحيط}

\begin{solution}{exo:g4:measure:1}
$7$ م $= 700$ سم؛ \quad $3.2$ كم $= 3\,200$ م؛ \quad
$85$ مم $= 8.5$ سم؛ \quad $4\,500$ م $= 4.5$ كم.
\end{solution}

\begin{solution}{exo:g4:measure:2}
$2$ كغ $= 2\,000$ غ؛ \quad $1\,800$ غ $= 1$ كغ و $800$ غ؛ \quad
$3.5$ ل $= 350$ سل؛ \quad $25$ سل $= 0.25$ ل.
\end{solution}

\begin{solution}{exo:g4:measure:3}
المستطيل: $2 \times (9 + 5) = 28$ سم. والمربع:
$4 \times 7.5 = 30$ سم. والمثلث: $6 + 8 + 10 = 24$ سم.
\end{solution}

\begin{solution}{exo:g4:measure:4}
الطول $+$ العرض $= 26 \div 2 = 13$ سم، إذن العرض هو
$13 - 8 = 5$ سم.
\end{solution}

\begin{solution}{exo:g4:measure:5}
أشكال L كثيرة تصلح؛ فمثلًا شكل L من $10$ مربعات مكوَّن من
مستطيل $4 \times 2$ زائد قدم $2 \times 1$، والسير على حدّه يعطي $16$
وحدة. (وكل شكل صحيح وعدّ دقيق صواب؛ \omterm{def:g4:measure:perimeter}{فالمحيط}
يتوقف على شكل L المختار.)
\end{solution}

\begin{solution}{exo:g4:measure:6}
$3 \times 4$: \omterm{def:g4:measure:perimeter}{المحيط} $14$. و $2 \times 6$: \omterm{def:g4:measure:perimeter}{المحيط} $16$.
(و $1 \times 12$: \omterm{def:g4:measure:perimeter}{المحيط} $26$.) وكلما اقترب الشكل من المربع
صغر محيطه عند \omterm{def:g4:measure:area}{المساحة} نفسها.
\end{solution}

\begin{solution}{exo:g4:measure:7}
نصف \omterm{def:g4:measure:area}{مساحة} المستطيل $6 \times 3$: $18 \div 2 = 9$ مربعات.
\end{solution}

\begin{solution}{exo:g4:measure:8}
$10{:}20 \to 12{:}45$: $2$ ساعة و $25$ دقيقة. و $7{:}35 \to 9{:}10$:
$1$ ساعة و $35$ دقيقة. والفيلم: $20{:}30 + 1$ سا $50 = 22{:}20$.
\end{solution}

\begin{solution}{exo:g4:measure:9}
اللفّة الواحدة: $2 \times (60 + 45) = 210$ م. وثلاث لفّات:
$3 \times 210 = 630$ م.
\end{solution}

\begin{solution}{exo:g4:measure:10}
\omterm{def:g4:measure:perimeter}{المحيط}: $4 \times 85 = 340$ م؛ ناقص البوابة:
$340 - 4 = 336$ م من السياج.
\end{solution}

\begin{solution}{exo:g4:measure:11}
الادّعاءان \emph{خطأ} كلاهما. \omterm{def:g4:measure:perimeter}{فالمحيط} نفسه بمساحتين مختلفتين:
المستطيل $4 \times 2$ وشكل L في صورة الفصل (\omterm{def:g4:measure:perimeter}{محيط} كل
منهما $12$؛ ومساحتاهما $8$ و $6$). \omterm{def:g4:measure:area}{والمساحة} نفسها بمحيطين
مختلفين: المربع $4 \times 4$ والمستطيل $8 \times 2$
(\omterm{def:g4:measure:area}{مساحة} كل منهما $16$؛ ومحيطاهما $16$ و $20$).
\end{solution}
