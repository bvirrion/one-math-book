\section*{Capítulo \ref{ch:g4:division} --- La división}

\begin{solution}{exo:g4:division:1}
$96 \div 4$: entre $4 \times 20 = 80$ y $4 \times 30 = 120$, así que
el \omterm{def:g3:division:remainder}{cociente} tiene dos cifras; la división larga da $24$, \omterm{def:g3:division:remainder}{resto}
$0$.

$87 \div 5$: entre $5 \times 10 = 50$ y $5 \times 20 = 100$;
\omterm{def:g3:division:remainder}{cociente} $17$, \omterm{def:g3:division:remainder}{resto} $2$ ($87 = 5 \times 17 + 2$).
\end{solution}

\begin{solution}{exo:g4:division:2}
$672 \div 4 = 168$, \omterm{def:g3:division:remainder}{resto} $0$ (comprobación: $4 \times 168 = 672$).

$925 \div 7 = 132$, \omterm{def:g3:division:remainder}{resto} $1$ (comprobación:
$7 \times 132 + 1 = 925$).

$804 \div 6 = 134$, \omterm{def:g3:division:remainder}{resto} $0$ (comprobación: $6 \times 134 = 804$).
\end{solution}

\begin{solution}{exo:g4:division:3}
$816 \div 4 = 204$ (el paso de las decenas es $1 \div 4$: \omterm{def:g1:counting:zero}{cero} veces ---
hay que escribir el $0$).

$420 \div 7 = 60$.

$2\,512 \div 5 = 502$, \omterm{def:g3:division:remainder}{resto} $2$ (comprobación:
$5 \times 502 + 2 = 2\,512$).
\end{solution}

\begin{solution}{exo:g4:division:4}
$85 = 9 \times 9 + 4$ (\omterm{def:g3:division:remainder}{cociente} $9$, \omterm{def:g3:division:remainder}{resto} $4$).

$1\,000 = 3 \times 333 + 1$.
\end{solution}

\begin{solution}{exo:g4:division:5}
$156 \div 6 = 26$: se llenan veintiséis hueveras y no sobra ninguno.
\end{solution}

\begin{solution}{exo:g4:division:6}
$250 = 8 \times 31 + 2$: treinta y un trozos y sobran $2$ cm de
cinta.
\end{solution}

\begin{solution}{exo:g4:division:7}
$375 = 52 \times 7 + 11$: siete autobuses llevan a $364$ alumnos y quedan $11$
--- es el caso del «más uno»: hacen falta $8$ autobuses.
\end{solution}

\begin{solution}{exo:g4:division:8}
$234 \div 3 = 78$: cada uno recibe $78$ canicas. Comprobación:
$3 \times 78 = 234$.
\end{solution}

\begin{solution}{exo:g4:division:9}
Dividendo $= 7 \times 58 + 3 = 406 + 3 = 409$.
\end{solution}

\begin{solution}{exo:g4:division:10}
Estantes: $438 = 9 \times 48 + 6$: $48$ estantes llenos y $6$ libros
más --- hacen falta $49$ estantes. Estanterías:
$49 = 6 \times 8 + 1$: ocho estanterías completas y un estante suelto ---
hay que comprar $9$ estanterías.
\end{solution}

\begin{solution}{exo:g4:division:11}
\omterm{def:g3:division:remainder}{Cociente} $=$ \omterm{def:g3:division:remainder}{resto} $= r$ con $r < 8$, así que $r$ vale $0, 1, \dots,
7$ y el dividendo es $8r + r = 9r$: los dividendos posibles son
$0$, $9$, $18$, $27$, $36$, $45$, $54$, $63$.
\end{solution}
