\section*{Capítulo \ref{ch:g4:geometry} --- Rectas y polígonos}

\begin{solution}{exo:g4:geometry:1}
Construcciones con \cref{met:g4:geometry:draw}; la pequeña plaza
marca el \omterm{def:g3:shapes:rightangle}{ángulo recto} entre $d $ y el \omterm{def:g4:geometry:def}{perpendiculares}.
\end{solution}

\begin{solution}{exo:g4:geometry:2}
E: los tres trazos horizontales son paralelas; cada uno es \omterm{def:g4:geometry:def}{perpendiculares}
al vertical. H: los dos trazos verticales son paralelas, el
barra transversal es \omterm{def:g4:geometry:def}{perpendiculares} a ambos. N: los dos trazos verticales son
paralelas; la diagonal no es ninguna de las dos. T: los dos trazos son
\omterm{def:g4:geometry:def}{perpendiculares}. Z: los trazos superior e inferior son paralelas; el
la diagonal no es ninguna de las dos cosas.
\end{solution}

\begin{solution}{exo:g4:geometry:3}
\omterm{def:g4:geometry:polygon}{Hexágono}; \omterm{def:g4:geometry:polygon}{cuadrilátero}; triángulo; \omterm{def:g4:geometry:polygon}{octágono}; \omterm{def:g4:geometry:polygon}{pentágono}. A \omterm{def:g4:geometry:polygon}{polígono} tiene
tantos \omterm{def:g4:geometry:polygon}{vértices} como lados: $6$, $4$, $3$, $8$, $5$.
\end{solution}

\begin{solution}{exo:g4:geometry:4}
A \omterm{def:g4:geometry:polygon}{pentágono} tiene $5$ lados. De $ A $, los no vecinos son $ C $ y
$ D $: dos diagonales, $[AC]$ y $[d.\,C.]$.
\end{solution}

\begin{solution}{exo:g4:geometry:5}
$ OA = 4$ centímetros (\omterm{def:g3:shapes:shapes}{radio}); $ OM = 4$ cm (cada punto del círculo);
$ AB = 8$ centímetros (un diámetro es el doble de \omterm{def:g3:shapes:shapes}{radio}).
\end{solution}

\begin{solution}{exo:g4:geometry:6}
Abra la brújula a lo largo de un lado, luego lleve esa abertura a los dos
otros lados: si el tramo de púa y lápiz cabe en un segundo lado
exactamente, esos dos lados son iguales.
\end{solution}

\begin{solution}{exo:g4:geometry:7}
El camino se cierra en un \omterm{def:g4:geometry:polygon}{polígono} con $5$ lados --- un \omterm{def:g4:geometry:polygon}{pentágono} (no es un
uno normal!).
\end{solution}

\begin{solution}{exo:g4:geometry:8}
``rectas \omterm{def:g4:geometry:def}{perpendiculares} siempre se cruzan'': verdadero --- se cruzan en el
\omterm{def:g3:shapes:rightangle}{ángulo recto} sí mismo.

``Líneas paralelas nunca cruzar '': verdadero --- esa es su definición.

``Una línea puede ser paralelas a una línea y \omterm{def:g4:geometry:def}{perpendiculares} a
another'': true --- en papel cuadriculado, se dibuja una línea horizontal paralelas a
cada línea horizontal y \omterm{def:g4:geometry:def}{perpendiculares} a cada vertical.
\end{solution}

\begin{solution}{exo:g4:geometry:9}
las lineas $ d $ y $ f $ mirar paralelas --- y siempre lo son: dos
lineas \omterm{def:g4:geometry:def}{perpendiculares} a la misma línea están paralelas
(\cref{prop:g6:lines:facts} hace esto oficial).
\end{solution}

\begin{solution}{exo:g4:geometry:10}
Un programa correcto: ``1. dibujar un cuadrado $ ABCD $ de lado $4$ centímetro.
2. Marca el punto $ O $ donde se cruzan sus diagonales. 3. Dibuja el círculo
de centro $ O $ y \omterm{def:g3:shapes:shapes}{radio} $2$ cm.'' El círculo toca cada lado en
su \omterm{def:g3:division:half}{medio}.
\end{solution}

\begin{solution}{exo:g4:geometry:11}
\omterm{def:g4:geometry:polygon}{Cuadrilátero}: $2$ diagonales. \omterm{def:g4:geometry:polygon}{Pentágono}: $5$. \omterm{def:g4:geometry:polygon}{Hexágono}: $9$. Cada nuevo
vértice agrega más y más diagonales --- los conteos crecen más rápido y
más rápido ($+3$, entonces $+4$, \dots).
\end{solution}
