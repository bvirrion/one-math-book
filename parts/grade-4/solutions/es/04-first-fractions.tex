\section*{Capítulo \ref{ch:g4:fractions} --- Las primeras fracciones}

\begin{solution}{exo:g4:fractions:1}
Pizza: quedan $\frac56$. Chocolate: se han comido $\frac38$. Bandera: $\frac13$
coloreada.
\end{solution}

\begin{solution}{exo:g4:fractions:2}
$\frac35$ sombrea tres partes y $\frac25$ solo dos: el sombreado mayor
es el de $\frac35$.
\end{solution}

\begin{solution}{exo:g4:fractions:3}
La \omterm{def:g3:division:half}{mitad} de $18$: $9$. El \omterm{def:g3:division:half}{cuarto} de $20$: $5$. El tercio de $21$: $7$.
$\frac34$ de $20$: tres \omterm{def:g3:division:half}{cuartos}, $3 \times 5 = 15$.
\end{solution}

\begin{solution}{exo:g4:fractions:4}
$\frac13$: un paso; $\frac33$: en el $1$; $\frac53$: dos pasos pasado
el $1$. La \omterm{def:g4:fractions:def}{fracción} que cae exactamente en $2$ es $\frac63$.
\end{solution}

\begin{solution}{exo:g4:fractions:5}
$\frac38 < \frac68$; \quad $\frac55 = 1$; \quad $\frac74 > 1$;
\quad $\frac12 = \frac24$.
\end{solution}

\begin{solution}{exo:g4:fractions:6}
$\frac26 + \frac36 = \frac56$; \quad
$\frac58 - \frac28 = \frac38$; \quad
$\frac14 + \frac24 = \frac34$; \quad
$1 = \frac33$, así que $1 - \frac13 = \frac23$.
\end{solution}

\begin{solution}{exo:g4:fractions:7}
$\frac54 = 1 + \frac14$; \quad
$\frac73 = 2 + \frac13$; \quad
$\frac92 = 4 + \frac12$.
\end{solution}

\begin{solution}{exo:g4:fractions:8}
Media hora: $30$ min. Un \omterm{def:g3:division:half}{cuarto}: $15$ min. Tres \omterm{def:g3:division:half}{cuartos}: $45$
min. Un tercio: $20$ min.
\end{solution}

\begin{solution}{exo:g4:fractions:9}
$\frac14$ de $24$ son $6$ alumnos con gafas; $24 - 6 = 18$
sin ellas.
\end{solution}

\begin{solution}{exo:g4:fractions:10}
Entre los dos: $\frac38 + \frac48 = \frac78$. Queda: $\frac18$. Comió más
Samuel ($4$ octavos frente a $3$).
\end{solution}

\begin{solution}{exo:g4:fractions:11}
Es posible, en efecto: la \omterm{def:g3:division:half}{mitad} de una pizza pequeña puede ser menos comida que un
\omterm{def:g3:division:half}{cuarto} de una gigante. Las fracciones comparan partes \emph{de un mismo
todo}; antes de comparar $\frac12$ y $\frac14$ «de algo»,
hay que saber que esos dos algos son iguales.
\end{solution}
