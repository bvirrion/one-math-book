\section*{Capítulo \ref{ch:g4:geometry} --- Retas e polígonos}

\begin{solution}{exo:g4:geometry:1}
Construções com o \cref{met:g4:geometry:draw}; o quadradinho marca o
\omterm{def:g3:shapes:rightangle}{ângulo reto} entre $d$ e a perpendicular.
\end{solution}

\begin{solution}{exo:g4:geometry:2}
E: os três traços horizontais são paralelos; cada um é perpendicular
ao traço vertical. H: os dois traços verticais são paralelos, e a
barra do meio é perpendicular aos dois. N: os dois traços verticais
são paralelos; a diagonal não é nem uma coisa nem outra. T: os dois
traços são \omterm{def:g4:geometry:def}{perpendiculares}. Z: o traço de cima e o de baixo são
paralelos; a diagonal não é nem uma coisa nem outra.
\end{solution}

\begin{solution}{exo:g4:geometry:3}
\omterm{def:g4:geometry:polygon}{Hexágono}; \omterm{def:g4:geometry:polygon}{quadrilátero}; triângulo; \omterm{def:g4:geometry:polygon}{octógono}; \omterm{def:g4:geometry:polygon}{pentágono}. Um \omterm{def:g4:geometry:polygon}{polígono}
tem tantos \omterm{def:g4:geometry:polygon}{vértices} quantos lados: $6$, $4$, $3$, $8$, $5$.
\end{solution}

\begin{solution}{exo:g4:geometry:4}
Um \omterm{def:g4:geometry:polygon}{pentágono} tem $5$ lados. A partir de $A$, os \omterm{def:g4:geometry:polygon}{vértices} não vizinhos
são $C$ e $D$: duas diagonais, $[AC]$ e $[AD]$.
\end{solution}

\begin{solution}{exo:g4:geometry:5}
$OA = 4$ cm (\omterm{def:g3:shapes:shapes}{raio}); $OM = 4$ cm (todo ponto do círculo);
$AB = 8$ cm (um \omterm{def:g4:geometry:circle}{diâmetro} é o dobro do \omterm{def:g3:shapes:shapes}{raio}).
\end{solution}

\begin{solution}{exo:g4:geometry:6}
Abra o compasso sobre um dos lados e leve essa abertura aos outros
dois: se o vão entre ponta seca e grafite couber exatamente num
segundo lado, esses dois lados são iguais.
\end{solution}

\begin{solution}{exo:g4:geometry:7}
O caminho se fecha num \omterm{def:g4:geometry:polygon}{polígono} de $5$ lados --- um \omterm{def:g4:geometry:polygon}{pentágono} (não
regular!).
\end{solution}

\begin{solution}{exo:g4:geometry:8}
``\omterm{def:g4:geometry:def}{Retas perpendiculares} sempre se cruzam'': verdadeiro --- elas se
cruzam no próprio \omterm{def:g3:shapes:rightangle}{ângulo reto}.

``\omterm{def:g4:geometry:def}{Retas paralelas} nunca se cruzam'': verdadeiro --- é a definição
delas.

``Uma reta pode ser paralela a uma reta e perpendicular a outra'':
verdadeiro --- no papel quadriculado, uma linha horizontal é paralela
a toda linha horizontal e perpendicular a toda linha vertical.
\end{solution}

\begin{solution}{exo:g4:geometry:9}
As retas $d$ e $f$ parecem \omterm{def:g4:geometry:def}{paralelas} --- e sempre são: duas \omterm{def:g4:geometry:def}{retas
perpendiculares} a uma mesma reta são \omterm{def:g4:geometry:def}{paralelas}
(\cref{prop:g6:lines:facts} torna isso oficial).
\end{solution}

\begin{solution}{exo:g4:geometry:10}
Um programa correto: ``1. Trace um quadrado $ABCD$ de lado $4$ cm.
2. Marque o ponto $O$ onde as diagonais dele se cruzam. 3. Trace o
círculo de centro $O$ e \omterm{def:g3:shapes:shapes}{raio} $2$ cm.'' O círculo toca cada lado no
meio dele.
\end{solution}

\begin{solution}{exo:g4:geometry:11}
\omterm{def:g4:geometry:polygon}{Quadrilátero}: $2$ diagonais. \omterm{def:g4:geometry:polygon}{Pentágono}: $5$. \omterm{def:g4:geometry:polygon}{Hexágono}: $9$. Cada
\omterm{def:g4:geometry:polygon}{vértice} novo acrescenta mais e mais diagonais --- as contagens crescem
cada vez mais depressa ($+3$, depois $+4$, \dots).
\end{solution}
