\section*{Bab \ref{ch:g4:numbers} --- Bilangan Besar}

\begin{solution}{exo:g4:numbers:1}
$3\,520\,000$; \quad $7\,000\,008$; \quad $240\,060$.
\end{solution}

\begin{solution}{exo:g4:numbers:2}
$5\,204\,300$: lima juta dua ratus empat ribu
tiga ratus.

$10\,050\,000$: sepuluh juta lima puluh ribu.

$999\,999$: sembilan ratus sembilan puluh sembilan ribu sembilan
ratus sembilan puluh sembilan.
\end{solution}

\begin{solution}{exo:g4:numbers:3}
Puluhan ribu: $7$. Ratusan pada \omterm{def:g4:numbers:classes}{kelas} jutaan: bilangan tujuh angka
tidak punya angka ratusan juta --- \omterm{def:g4:numbers:classes}{kelas} jutaan pada $8\,472\,156$
hanya memuat angka $8$ (satuan jutaan). Ribuan yang dimuat: $8\,472$
(karena $8\,472\,156 = 8\,472 \times 1\,000 + 156$).
\end{solution}

\begin{solution}{exo:g4:numbers:4}
$980\,000 < 1\,020\,000$; \quad
$3\,456\,000 < 3\,465\,000$ (\omterm{def:g4:numbers:classes}{kelas} ribuan $456 < 465$); \quad
$7\,090\,050 < 7\,090\,500$.
\end{solution}

\begin{solution}{exo:g4:numbers:5}
$980\,500 < 2\,030\,000 < 2\,299\,999 < 2\,300\,000$.
\end{solution}

\begin{solution}{exo:g4:numbers:6}
Ke puluhan: $63\,780$. Ke ratusan: $63\,800$. Ke ribuan:
$64\,000$. Ke puluhan ribu: $60\,000$.
\end{solution}

\begin{solution}{exo:g4:numbers:7}
$5\,000\,000$; \quad $6\,000\,000$ ($499\,999$ belum mencapai
setengah juta); \quad $10\,000\,000$ (tepat di tengah \omterm{def:g4:numbers:round}{dibulatkan} naik).
\end{solution}

\begin{solution}{exo:g4:numbers:8}
$998\,500 \to 999\,500 \to 1\,000\,500 \to 1\,001\,500$: langkah
kedua menyeberangi satu juta, dan sebuah \omterm{def:g4:numbers:classes}{kelas} baru menyala di kiri.

$950\,000 \to 1\,050\,000 \to 1\,150\,000$: penyeberangan yang sama
terjadi pada langkah pertama.
\end{solution}

\begin{solution}{exo:g4:numbers:9}
Keduanya benar. ``Lebih dari $80\,000$'' \omterm{def:g4:numbers:round}{membulatkan} ke bawah pada
puluhan ribu ($81\,338 \to 80\,000$); ``sekitar $81\,000$''
\omterm{def:g4:numbers:round}{membulatkan} ke ribuan terdekat ($81\,338 \to 81\,000$).
\end{solution}

\begin{solution}{exo:g4:numbers:10}
Terbesar: $7\,654\,321$. Terkecil: $1\,234\,567$. Paling dekat
dengan $4\,000\,000$: yang tepat di bawah paling baik ---
$3\,765\,421$ berjarak $234\,579$, sedangkan yang tepat di atas,
$4\,123\,567$, hanya berjarak $123\,567$: jadi yang paling dekat adalah $4\,123\,567$.
\end{solution}

\begin{solution}{exo:g4:numbers:11}
\omterm{def:g4:numbers:round}{Pembulatan} ke ribuan terdekat memberi $50\,000$ untuk semua bilangan
dari $49\,500$ (tepat di tengah \omterm{def:g4:numbers:round}{dibulatkan} naik) sampai $50\,499$.
Terbesar: $50\,499$; terkecil: $49\,500$.

\end{solution}
