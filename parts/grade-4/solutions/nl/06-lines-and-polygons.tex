\section*{Hoofdstuk \ref{ch:g4:geometry} --- Lijnen en veelhoeken}

\begin{solution}{exo:g4:geometry:1}
Constructies met \cref{met:g4:geometry:draw}; het vierkantje geeft de
\omterm{def:g3:shapes:rightangle}{rechte hoek} tussen $d$ en de loodlijn aan.
\end{solution}

\begin{solution}{exo:g4:geometry:2}
E: de drie horizontale strepen zijn \omterm{def:g4:geometry:def}{parallel}; elk staat \omterm{def:g4:geometry:def}{loodrecht} op de
verticale streep. H: de twee verticale strepen zijn \omterm{def:g4:geometry:def}{parallel}, de dwarsstreep
staat \omterm{def:g4:geometry:def}{loodrecht} op beide. N: de twee verticale strepen zijn \omterm{def:g4:geometry:def}{parallel}; de
diagonaal is geen van beide. T: de twee strepen staan \omterm{def:g4:geometry:def}{loodrecht} op elkaar.
Z: de bovenste en de onderste streep zijn \omterm{def:g4:geometry:def}{parallel}; de diagonaal is geen van
beide.
\end{solution}

\begin{solution}{exo:g4:geometry:3}
\omterm{def:g4:geometry:polygon}{Zeshoek}; \omterm{def:g4:geometry:polygon}{vierhoek}; driehoek; \omterm{def:g4:geometry:polygon}{achthoek}; \omterm{def:g4:geometry:polygon}{vijfhoek}. Een \omterm{def:g4:geometry:polygon}{veelhoek} heeft even
veel \omterm{def:g4:geometry:polygon}{hoekpunten} als zijden: $6$, $4$, $3$, $8$, $5$.
\end{solution}

\begin{solution}{exo:g4:geometry:4}
Een \omterm{def:g4:geometry:polygon}{vijfhoek} heeft $5$ zijden. Vanuit $A$ zijn $C$ en $D$ geen buren: twee
diagonalen, $[AC]$ en $[AD]$.
\end{solution}

\begin{solution}{exo:g4:geometry:5}
$OA = 4$ cm (de \omterm{def:g3:shapes:shapes}{radius}); $OM = 4$ cm (elk punt van de cirkel); $AB = 8$ cm
(een \omterm{def:g4:geometry:circle}{diameter} is twee keer de \omterm{def:g3:shapes:shapes}{radius}).
\end{solution}

\begin{solution}{exo:g4:geometry:6}
Zet de passer op één zijde en breng die opening naar de twee andere zijden:
past de opening precies op een tweede zijde, dan zijn die twee zijden even
lang.
\end{solution}

\begin{solution}{exo:g4:geometry:7}
Het pad sluit tot een \omterm{def:g4:geometry:polygon}{veelhoek} met $5$ zijden --- een \omterm{def:g4:geometry:polygon}{vijfhoek} (en geen
regelmatige!).
\end{solution}

\begin{solution}{exo:g4:geometry:8}
``Twee \omterm{def:g4:geometry:def}{loodrechte lijnen} snijden elkaar altijd'': waar --- ze snijden elkaar
juist in de \omterm{def:g3:shapes:rightangle}{rechte hoek}.

``Twee \omterm{def:g4:geometry:def}{parallelle lijnen} snijden elkaar nooit'': waar --- dat is hun
definitie.

``Een lijn kan \omterm{def:g4:geometry:def}{parallel} zijn aan de ene lijn en \omterm{def:g4:geometry:def}{loodrecht} staan op een
andere'': waar --- op ruitjespapier is een horizontale lijn \omterm{def:g4:geometry:def}{parallel} aan elke
horizontale lijn en \omterm{def:g4:geometry:def}{loodrecht} op elke verticale.
\end{solution}

\begin{solution}{exo:g4:geometry:9}
De lijnen $d$ en $f$ lijken \omterm{def:g4:geometry:def}{parallel} --- en dat zijn ze altijd: twee lijnen
die \omterm{def:g4:geometry:def}{loodrecht} op dezelfde lijn staan, zijn \omterm{def:g4:geometry:def}{parallel}
(\cref{prop:g6:lines:facts} maakt het officieel).
\end{solution}

\begin{solution}{exo:g4:geometry:10}
Een goed programma is bijvoorbeeld: ``1. Teken een vierkant $ABCD$ met zijde
$4$ cm. 2. Zet het punt $O$ waar zijn diagonalen elkaar snijden. 3. Teken de
cirkel met middelpunt $O$ en \omterm{def:g3:shapes:shapes}{radius} $2$ cm.'' De cirkel raakt elke zijde in
haar midden.
\end{solution}

\begin{solution}{exo:g4:geometry:11}
\omterm{def:g4:geometry:polygon}{Vierhoek}: $2$ diagonalen. \omterm{def:g4:geometry:polygon}{Vijfhoek}: $5$. \omterm{def:g4:geometry:polygon}{Zeshoek}: $9$. Elk nieuw \omterm{def:g4:geometry:polygon}{hoekpunt}
voegt er meer diagonalen bij --- de aantallen groeien steeds sneller ($+3$,
dan $+4$, \dots).
\end{solution}
