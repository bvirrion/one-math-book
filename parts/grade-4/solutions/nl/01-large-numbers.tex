\section*{Hoofdstuk \ref{ch:g4:numbers} --- Grote getallen}

\begin{solution}{exo:g4:numbers:1}
$3\,520\,000$; \quad $7\,000\,008$; \quad $240\,060$.
\end{solution}

\begin{solution}{exo:g4:numbers:2}
$5\,204\,300$: vijf miljoen tweehonderdvierduizend driehonderd.

$10\,050\,000$: tien miljoen vijftigduizend.

$999\,999$: negenhonderdnegenennegentigduizend
negenhonderdnegenennegentig.
\end{solution}

\begin{solution}{exo:g4:numbers:3}
Tienduizenden: $7$. Honderdtallen van de groep van de miljoenen: die zijn
er niet in een getal van zeven cijfers --- de groep van de miljoenen van
$8\,472\,156$ bevat alleen het cijfer $8$ (de eenheden van de miljoenen).
Aantal duizenden: $8\,472$ (want
$8\,472\,156 = 8\,472 \times 1\,000 + 156$).
\end{solution}

\begin{solution}{exo:g4:numbers:4}
$980\,000 < 1\,020\,000$; \quad
$3\,456\,000 < 3\,465\,000$ (groep van de duizenden: $456 < 465$); \quad
$7\,090\,050 < 7\,090\,500$.
\end{solution}

\begin{solution}{exo:g4:numbers:5}
$980\,500 < 2\,030\,000 < 2\,299\,999 < 2\,300\,000$.
\end{solution}

\begin{solution}{exo:g4:numbers:6}
Op het tiental: $63\,780$. Op het honderdtal: $63\,800$. Op het
duizendtal: $64\,000$. Op het tienduizendtal: $60\,000$.
\end{solution}

\begin{solution}{exo:g4:numbers:7}
$5\,000\,000$; \quad $6\,000\,000$ (die $499\,999$ komt net niet aan een
half miljoen); \quad $10\,000\,000$ (precies halverwege rondt naar boven
af).
\end{solution}

\begin{solution}{exo:g4:numbers:8}
$998\,500 \to 999\,500 \to 1\,000\,500 \to 1\,001\,500$: bij de tweede
stap ga je het miljoen voorbij, en links licht een nieuwe groep op.

$950\,000 \to 1\,050\,000 \to 1\,150\,000$: hier gebeurt dat al bij de
eerste stap.
\end{solution}

\begin{solution}{exo:g4:numbers:9}
Ze hebben beide gelijk. ``Meer dan $80\,000$'' is naar beneden afgerond op
het tienduizendtal ($81\,338 \to 80\,000$); ``ongeveer $81\,000$'' is
afgerond op het naaste duizendtal ($81\,338 \to 81\,000$).
\end{solution}

\begin{solution}{exo:g4:numbers:10}
Grootste: $7\,654\,321$. Kleinste: $1\,234\,567$. Het dichtst bij
$4\,000\,000$: net eronder lukt het minder goed --- $3\,765\,421$ ligt er
$234\,579$ vandaan, terwijl net erboven $4\,123\,567$ er maar $123\,567$
vandaan ligt: het dichtstbijzijnde is $4\,123\,567$.
\end{solution}

\begin{solution}{exo:g4:numbers:11}
Afronden op het naaste duizendtal geeft $50\,000$ voor alle getallen van
$49\,500$ (precies halverwege rondt naar boven af) tot $50\,499$.
Grootste: $50\,499$; kleinste: $49\,500$.
\end{solution}
