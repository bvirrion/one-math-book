\chapter{الأعداد الكبيرة}\label{ch:g4:numbers}

كانت الآلاف سقف \cref{ch:g3:numbers}؛ والآن يرتفع السقف إلى
\emph{الملايين}. والحيلة التي تجعل الأعداد الضخمة مقروءة هي
تقطيع أرقامها إلى مجموعات من ثلاثة --- وهي \omterm{def:g4:numbers:classes}{الفئات}.

\section{فئات من ثلاثة أرقام}

\begin{definition}[الفئات]\label{def:g4:numbers:classes}
اِبتداءً من اليمين، تُجمَّع أرقام العدد في \emph{فئات} من
ثلاثة: فئة \emph{الآحاد}، وفئة \emph{الألوف}، وفئة
\emph{الملايين}. ولكل فئة آحادها وعشراتها ومئاتها.
فمثلًا
\[
6\,309\,452 = 6 \text{ ملايين} + 309 \text{ ألفًا}
+ 452 \text{ واحدًا},
\]
ويُقرأ «ستة ملايين وثلاثمئة وتسعة آلاف وأربعمئة
واثنان وخمسون».
\end{definition}

\begin{example}[الكتابة بالفئات]\label{ex:g4:numbers:write}
«اثنا عشر مليونًا وأربعون ألفًا وخمسمئة» فيه $12$ في
فئة الملايين، و $040$ في فئة الألوف، و $500$ في فئة
الآحاد:
\[
12\,040\,500 .
\]
والأصفار في $040$ ضرورية: فكل فئة (إلا اليسرى) يجب أن تُظهر
أرقامها الثلاثة.
\end{example}

\begin{example}[كم ألفًا؟]\label{ex:g4:numbers:contained}
العدد $6\,309\,452$ \emph{يحتوي} $6\,309$ ألفًا
(كل ما على يسار فئة الآحاد): $6\,309\,452 = 6\,309 \times
1\,000 + 452$. والسؤال «كم ألفًا» ليس سؤالًا عن رقم واحد،
بل عن بادئة كاملة من العدد.
\end{example}

\section{المقارنة والتقريب}

\begin{method}[مقارنة الأعداد الكبيرة]\label{met:g4:numbers:compare}
تمامًا كما في \cref{met:g3:numbers:compare}، مع \omterm{def:g4:numbers:classes}{الفئات}
مختصَرًا: الأرقام الأكثر تفوز؛ وعند تساوي الطول، قارن من اليسار
--- فئةً فئةً، ثم رقمًا رقمًا.
\end{method}

\begin{example}\label{ex:g4:numbers:compare}
$2\,145\,890$ و $2\,148\,540$: الملايين متساوية ($2$)، وفئة
الألوف $145 < 148$، إذن
\[
2\,145\,890 < 2\,148\,540 .
\]
\end{example}

\begin{definition}[التقريب]\label{def:g4:numbers:round}
\emph{تقريب}\index{تقريب} عدد إلى أقرب ألف (أو عشرة أو مئة أو
مليون\,\dots) يعني استبداله بأقرب مضاعف للعدد $1\,000$. وإن وقع
العدد في المنتصف تمامًا، قرِّب إلى الأعلى. وعمليًّا: اُنظر إلى
الرقم الذي يلي مرتبة التقريب مباشرة --- فإن كان $5$ فما فوق
قرّبنا إلى الأعلى، وإن كان $4$ فما دون قرّبنا إلى الأسفل.
\end{definition}

\begin{example}\label{ex:g4:numbers:round}
\omterm{def:g4:numbers:round}{تقريب} $37\,614$ إلى أقرب ألف: رقم المئات هو
$6$، إذن نقرّب إلى الأعلى: $38\,000$. وإلى أقرب مئة:
$37\,600$. وإلى أقرب عشرة آلاف: $40\,000$.
\end{example}

\begin{example}[رتب القدر في العالم]\label{ex:g4:numbers:world}
\omterm{def:g4:numbers:round}{التقريب} هو الطريقة التي تُستعمل بها الأعداد الكبيرة فعلًا: فبلد
فيه $67\,842\,591$ نسمة هو «نحو $68$ مليونًا»؛ ومدينة فيها
$214\,376$ نسمة هي «نحو $200\,000$». والرقم الدقيق يتغير كل
يوم على أي حال --- أما المقرَّب فيحمل المعلومة
المفيدة.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:numbers:1}
اُكتب بالأرقام، مع فصل \omterm{def:g4:numbers:classes}{الفئات}: «ثلاثة ملايين وخمسمئة
وعشرون ألفًا»؛ «سبعة ملايين وثمانية»؛ «مئتان وأربعون
ألفًا وستون».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:2}
اُكتب بالحروف: $5\,204\,300$؛ \quad $10\,050\,000$؛ \quad
$999\,999$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:3}
في $8\,472\,156$: أي رقم هو عشرات الألوف؟ وأيّها مئات
فئة الملايين؟ وكم \emph{ألفًا} يحتوي العدد (اُنظر
\cref{ex:g4:numbers:contained})؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:4}
اُنقل وأكمل بالإشارة $<$ أو $>$:
\[
980\,000 \;?\; 1\,020\,000, \qquad
3\,456\,000 \;?\; 3\,465\,000, \qquad
7\,090\,050 \;?\; 7\,090\,500 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:5}
رتّب من الأصغر إلى الأكبر: $2\,300\,000$؛ \ $980\,500$؛ \
$2\,030\,000$؛ \ $2\,299\,999$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:6}
قرِّب $63\,782$: إلى أقرب عشرة؛ وإلى أقرب مئة؛ وإلى أقرب
ألف؛ وإلى أقرب عشرة آلاف.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:7}
قرِّب إلى أقرب مليون: $4\,721\,000$؛ \quad $6\,499\,999$؛
\quad $9\,500\,000$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:8}
عُدّ بالخطوات: ثلاث خطوات من $1\,000$ ابتداءً من $998\,500$؛
وخطوتان من $100\,000$ ابتداءً من $950\,000$. فماذا يحدث \omterm{def:g4:numbers:classes}{للفئات}
حين تعبر المليون؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:9}
يسع ملعب $81\,338$ متفرجًا. وتكتب صحيفة «أكثر من
$80\,000$ مشجّع»؛ وتقول إذاعة «نحو $81\,000$». أكلاهما
محقّ؟ وإلى أي مرتبة قرّب كل منهما؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:numbers:10}
مستعملًا كلًّا من الأرقام $1$ إلى $7$ مرة واحدة بالضبط، اُكتب
أكبر عدد ممكن من سبعة أرقام، وأصغره، وأقربها إلى
$4\,000\,000$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:numbers:11}
ما أكبر عدد صحيح يقرَّب إلى $50\,000$ عند \omterm{def:g4:numbers:round}{التقريب} إلى
أقرب ألف؟ وما أصغره؟
\end{exercise}
