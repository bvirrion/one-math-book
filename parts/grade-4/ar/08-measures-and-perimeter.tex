\chapter{القياسات والمحيط}\label{ch:g4:measure}

في السنة الماضية قِسنا بالمساطر والموازين والساعات
(\cref{ch:g3:measure})؛ ويضيف هذا الفصل جدول الوحدات --- وهو
الآلة التي تحوّل كل شيء --- وطريقتين لقياس شكل مستوٍ:
طول حدّه (\omterm{def:g4:measure:perimeter}{المحيط}) والحيّز داخله
(\omterm{def:g4:measure:area}{المساحة}، معدودة بالمربعات في الوقت الحالي).

\section{جدول الوحدات}

\begin{method}[التحويل بالجدول]\label{met:g4:measure:table}
في الأطوال، كل عمود يساوي عشرة أمثال العمود التالي:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|c|c|c|c|c|c}
كم & هم & دام & م & دم & سم & مم \\
\hline
 & & & $3$ & $4$ & $0$ & \\
\end{tabular}
\end{center}
\begin{enumerate}
  \item اُكتب العدد برقم واحد في كل عمود، ورقم آحاد
        القياس في عمود وحدته (هنا $3.4$ م: الرقم $3$
        تحت م)؛
  \item ثم اقرأ العدد بالوحدة الجديدة: $3.4$ م $= 340$ سم (واملأ
        الأعمدة الفارغة بالأصفار)؛
  \item والكتل (كغ\,\dots\,غ) والسعات (ل\,\dots\,سل) تعمل
        بالطريقة نفسها.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g4:measure:convert}
$5$ كم $= 5\,000$ م؛ \quad $270$ سم $= 2.7$ م (فالرقم $2$ يقع
تحت م)؛ \quad $3$ كغ و $250$ غ $= 3\,250$ غ؛ \quad
$2.5$ ل $= 250$ سل.
\end{example}

\section{المحيط}

\begin{definition}[المحيط]\label{def:g4:measure:perimeter}
\emph{محيط}\index{محيط} \omterm{def:g4:geometry:polygon}{مضلع} هو الطول الكلي
لحدّه: اِجمع أطوال أضلاعه كلها (بالوحدة نفسها!). ومختصرات
الأشكال المعتادة:
\[
\text{المستطيل: } P = 2 \times (L + w),
\qquad
\text{المربع: } P = 4 \times c .
\]
\end{definition}

\begin{example}\label{ex:g4:measure:perimeter}
حديقة مستطيلة $12$ م في $7$ م:
\[
P = 2 \times (12 + 7) = 2 \times 19 = 38 \text{ م}
\]
من السياج. وإطار صورة مربع طول ضلعه $18$ سم:
$P = 4 \times 18 = 72$ سم.
\end{example}

\section{المساحة: عدّ المربعات}

\begin{definition}[المساحة بالعدّ]\label{def:g4:measure:area}
\emph{مساحة}\index{مساحة} شكل مرسوم على شبكة هي عدد
مربعات الشبكة التي يغطيها. ونصفا مربع يُعدّان مربعًا واحدًا.
والمساحة تجيب عن «كم من السطح»، \omterm{def:g4:measure:perimeter}{والمحيط} يجيب عن «كم يبلغ طول
الحدّ» --- وهما سؤالان مختلفان!
\end{definition}

\begin{tikzpicture}[scale=0.55]
  \draw[gray!40, thin] (0,0) grid (13,4);
  \draw[very thick, omDef, fill=omDef!15] (1,1) rectangle (5,3);
  \node[font=\small, omDef] at (3,2) {$8$ مربعات};
  \draw[very thick, omThm, fill=omThm!15]
    (7,1) -- (11,1) -- (11,2) -- (9,2) -- (9,3) -- (7,3) -- cycle;
  \node[font=\small, omThm] at (8.4,1.5) {$6$ مربعات};
\end{tikzpicture}

{\small شكلان على الشبكة: مساحتاهما $8$ و $6$ مربعات. والآن سِر
على الحدّين وعُدّ الوحدات: $12$ لكليهما! \omterm{def:g4:measure:perimeter}{فالمحيط} نفسه،
والمساحتان مختلفتان.}
\end{omfigure}

\begin{example}[أنصاف المربعات]\label{ex:g4:measure:half}
مثلث قائم مرسوم على الشبكة ضلعا قائمته $4$ و $2$ يغطي $3$
مربعات كاملة و $2$ من أنصاف المربعات\,\dots\ عُدّ بعناية: فمساحته
نصف \omterm{def:g4:measure:area}{مساحة} المستطيل $4 \times 2$، أي $4$ مربعات. وقطع
مستطيل على قطره يعطي دائمًا مثلثين متساويي
\omterm{def:g4:measure:area}{المساحة}.
\end{example}

\begin{example}[المساحة نفسها، ومحيطان مختلفان]\label{ex:g4:measure:samearea}
مربع $4 \times 4$ ومستطيل $8 \times 2$ يغطيان كلاهما $16$
مربعًا --- \omterm{def:g4:measure:area}{فالمساحة} نفسها. أما الحدود: $16$ وحدة للمربع، و $20$
للمستطيل. ولا يحدد أحد القياسين الآخر: فالشكل الطويل الرفيع
له حدّ كثير مقابل سطح قليل.
\end{example}

\section{المدد}

\begin{method}[جمع الأوقات عبر الساعة]\label{met:g4:measure:time}
لجمع المدد أو طرحها، اعمل بالقفزات عبر الساعات الكاملة
(وتذكّر أن $1$ ساعة $= 60$ دقيقة):
\[
9\text{ سا }40 + 35 \text{ د}:
\quad 9\text{ سا }40 \xrightarrow{+20} 10\text{ سا }00
\xrightarrow{+15} 10\text{ سا }15 .
\]
ولا تجمع الدقائق بعد $60$ دون تحويل: $40 + 35 = 75$ دقيقة
$= 1$ ساعة و $15$ دقيقة.
\end{method}

\begin{example}\label{ex:g4:measure:timetable}
يغادر القطار في $8{:}47$ ويصل في $11{:}05$. والمدة:
من $8{:}47 \to 9{:}00$ هي $13$ دقيقة؛ ومن $9{:}00 \to 11{:}00$
هي $2$ ساعة؛ ومن $11{:}00 \to 11{:}05$ هي $5$ دقائق. \omterm{def:g1:addition:def}{والمجموع}: $2$ ساعة و $18$ دقيقة.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:measure:1}
حوِّل بجدول الوحدات: $7$ م بالسنتيمتر؛ و $3.2$ كم بالمتر؛
و $85$ مم بالسنتيمتر؛ و $4\,500$ م بالكيلومتر.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:2}
حوِّل: $2$ كغ بالغرام؛ و $1\,800$ غ بالكيلوغرام والغرام؛
و $3.5$ ل بالسنتيلتر؛ و $25$ سل باللتر.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:3}
احسب \omterm{def:g4:measure:perimeter}{المحيط}: مستطيل $9$ سم $\times$ $5$ سم؛ ومربع
طول ضلعه $7.5$ سم؛ ومثلث أضلاعه $6$ سم و $8$ سم و $10$ سم.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:4}
مستطيل محيطه $26$ سم وطوله $8$ سم. جِد عرضه
(وابدأ بإيجاد الطول $+$ العرض).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:5}
على الورق المربّعات، اُرسم شكلًا على هيئة الحرف L يغطي $10$
مربعات بالضبط، وعُدّ وحدات محيطه.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:6}
اُرسم مستطيلين مختلفين \omterm{def:g4:measure:area}{مساحة} كل منهما $12$ مربعًا. واحسب
\omterm{def:g4:measure:perimeter}{محيط} كل منهما. فأيّهما «أقرب إلى المربع»، وأيّهما «أرفع»؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:7}
مثلث قائم مرسوم على الشبكة ضلعا قائمته $6$ و $3$. فما
مساحته بالمربعات؟ (اِستعمل \cref{ex:g4:measure:half}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:8}
احسب المدد: من $10{:}20$ إلى $12{:}45$؛ ومن $7{:}35$ إلى
$9{:}10$. وفيلم يدوم $1$ ساعة و $50$ دقيقة ويبدأ في $20{:}30$:
فمتى ينتهي؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:9}
تجري كريمة $3$ لفّات حول ملعب مستطيل $60$ م في $45$ م. فكم مترًا
تجري؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:measure:10}
يريد مزارع أن يسيّج حقلًا مربعًا طول ضلعه $85$ م، تاركًا
بوابة من $4$ م بلا سياج. فكم مترًا من السياج عليه أن يشتري؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:measure:11}
صواب أم خطأ، مع أمثلة على الشبكة: «إذا تساوى محيطا شكلين
تساوت مساحتاهما»؛ «وإذا تساوت مساحتا شكلين تساوى
محيطاهما».
(و\Cref{ex:g4:measure:samearea} وشكل الفصل يساعدان.)
\end{exercise}
