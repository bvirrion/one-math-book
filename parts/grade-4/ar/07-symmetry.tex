\chapter{التماثل}\label{ch:g4:symmetry}

الفراشات، والوجوه، ورقاقات الثلج: تحب الطبيعة الأشكال التي
ينطبق نصفاها عند الطيّ. يستكشف هذا الفصل التماثل بالطيّ وبالورق
المربّعات --- أما الإنشاءات بالكوس فتنتظر
\cref{ch:g6:symmetry}.

\section{الطيّ والانطباق}

\begin{definition}[محور التماثل]\label{def:g4:symmetry:def}
يكون المستقيم \emph{محور تماثل}\index{محور تماثل} لشكل
حين يجعل طيُّ الشكل على ذلك المستقيم نصفيه ينطبقان
\emph{تمامًا}. وقد يكون للشكل محور واحد، أو محاور عدة، أو لا
محور له.
\end{definition}

\begin{example}[الاختبار بورق الشفّ]\label{ex:g4:symmetry:trace}
اِنقل الشكل على ورق الشفّ، واقلب الورقة على المحور المرشَّح،
ثم أعِدها: فإن غطّى النقلُ الشكلَ تمامًا، فالمحور
صحيح. والعين وحدها تُخدع كثيرًا --- \omterm{def:g4:geometry:circle}{فقطر} المستطيل
\emph{يبدو} محورًا، لكن اختبار الطيّ يقول لا (جرّبه!).
\end{example}

\section{إكمال الأشكال على الشبكة}

\begin{method}[مماثل شكل بالنسبة إلى خط من الشبكة]\label{met:g4:symmetry:grid}
المحور خط رأسي (أو أفقي) من خطوط الشبكة.
\begin{enumerate}
  \item خذ كل رأس من رؤوس الشكل بالتناوب؛
  \item وعُدّ بُعده عن المحور بالمربعات؛
  \item وضع رأس الصورة على البُعد \emph{نفسه} من الجهة
        \emph{الأخرى}، في السطر نفسه (أو العمود نفسه)؛
  \item وصِل رؤوس الصورة بالترتيب نفسه، وتحقّق: فالصورة
        هي التوأم المرآتي للشكل.
\end{enumerate}
\end{method}

\begin{example}[الشكل نفسه، والحجم نفسه، مقلوبًا]\label{ex:g4:symmetry:properties}
للصورة الأطوال نفسها والزوايا نفسها تمامًا كالأصل:
فالتماثل ينسخ الشكل --- لكنه يقلبه، كاليد اليسرى واليد اليمنى.
فإن كان الحرف الأصلي F، فصورته المرآتية F معكوس، لا F
آخر.
\end{example}

\section{محاور الأشكال المعتادة}

\begin{example}[عدّ المحاور]\label{ex:g4:symmetry:shapes}
بالطيّ:
\begin{itemize}
  \item للمربع $4$ محاور (اثنان يمرّان بمنتصفَي ضلعين
        متقابلين، واثنان على القطرين)؛
  \item وللمستطيل $2$ من المحاور (منتصفا الضلعين المتقابلين لا غير!)؛
  \item وللمثلث المتساوي الساقين $1$؛ وللمتساوي الأضلاع $3$؛
  \item وللدائرة كل مستقيم يمرّ بمركزها --- محاور أكثر من أن
        تُعدّ.
\end{itemize}
\end{example}

\begin{example}[التماثل في الأبجدية]\label{ex:g4:symmetry:letters}
الحروف A و M و T و U و V لها محور رأسي؛ والحروف B و C و D و E لها
محور أفقي؛ والحروف H و I و O و X لها المحوران معًا. وكلمات مصنوعة
من الحروف المناسبة قد تكون متماثلة أيضًا: فكلمة MUM لها محور
رأسي؛ وكلمة OXO تنجح في الاتجاهين.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:1}
اِنقل واطوِ: أيّها له \omterm{def:g4:symmetry:def}{محور تماثل} --- مربع؛
أم مستطيل (غير مربع)؛ أم مثلث مختلف الأضلاع (كل أضلاعه
مختلفة)؛ أم دائرة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:2}
اُرسم محاور التماثل: للمربع؛ وللمستطيل؛ وللمثلث المتساوي
الأضلاع. وكم محورًا لكل منها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:3}
اِستعمل اختبار ورق الشفّ في \cref{ex:g4:symmetry:trace} لتبيّن
أن \omterm{def:g4:geometry:circle}{قطر} مستطيل $6 \times 4$ \emph{ليس} \omterm{def:g4:symmetry:def}{محور
تماثل}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:4}
على الورق المربّعات، اُرسم محورًا رأسيًّا وحرف L (بارتفاع ثلاثة
مربعات وعرض مربعين) على بُعد $2$ من المربعات عن المحور. وأنشئ
صورته المرآتية.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:5}
على الورق المربّعات، اُرسم محورًا أفقيًّا وعلمًا صغيرًا (مثلث $1 \times
1$ على سارية من $3$ مربعات) فوقه. وأكمل الشكل
بالتماثل تحت المحور.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:6}
يمسّ شكل محورَ تماثله في نقطة $P$. فأين صورة
$P$؟ اِشرح بصورة الطيّ.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:7}
صنّف الحروف الكبيرة A و B و E و F و H و N و O و T: محور رأسي؟
أم محور أفقي؟ أم كلاهما؟ أم لا محور؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:symmetry:8}
مثلث أضلاعه $5$ سم و $5$ سم و $3$ سم. فهل له \omterm{def:g4:symmetry:def}{محور
تماثل}؟ وأي ضلعين يبدّل الطيّ بينهما؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:symmetry:9}
على الورق المربّعات، اُرسم شكلًا من عندك له \emph{محورا تماثل
بالضبط}، واُرسم المحورين. (تلميح: فكّر في مستطيل،
أو في شكل علامة الزائد.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:symmetry:10}
أُعطي نصف رسم متماثل: النصف الأيسر لبيت
(جدار مربع ونصف سقف مثلث)، ومحور رأسي
على حافته اليمنى. صِف البيت كاملًا أو ارسمه. وكيف تقارن
أطوال النصف الأيمن المكتمل بأطوال النصف الأيسر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:symmetry:11}
تدّعي آية: «كل مثلث فيه ضلعان متساويان له \omterm{def:g4:symmetry:def}{محور
تماثل}، وكل مثلث له \omterm{def:g4:symmetry:def}{محور تماثل} ففيه ضلعان
متساويان». اِختبر ادّعاءها المزدوج على رسوم: مثلث متساوي الساقين،
ومثلث متساوي الأضلاع، ومثلث مختلف الأضلاع. فهل يؤيدها الطيّ؟
\end{exercise}
