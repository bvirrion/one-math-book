\chapter{القسمة}\label{ch:g4:division}

في السنة الماضية قسمنا بالجداول وببواقٍ صغيرة
(\cref{ch:g3:division})؛ وهذا الفصل يركّب الطريقة المكتوبة
الكاملة --- القسمة المطوّلة --- التي تعالج أي عدد، مجموعةَ أرقام
بعد مجموعة.

\section{طريقة القسمة المطوّلة}

\begin{method}[القسمة المطوّلة على عدد من رقم واحد]\label{met:g4:division:method}
لقسمة $749$ على $6$:
\begin{enumerate}
  \item خذ الرقم الأيسر، $7$ (مئات): يدخل $6$ مرة $1$ في $7$؛
        اُكتب $1$ في \omterm{def:g3:division:remainder}{خارج القسمة}، واطرح
        $6$، \omterm{def:g3:division:remainder}{فالباقي} $1$؛
  \item \emph{أنزل} الرقم التالي، $4$: والآن اِقسم $14$
        (عشرات): $6 \times 2 = 12$؛ اُكتب $2$، \omterm{def:g3:division:remainder}{والباقي} $2$؛
  \item وأنزل $9$: اِقسم $29$: $6 \times 4 = 24$؛ اُكتب
        $4$، \omterm{def:g3:division:remainder}{والباقي} $5$؛
  \item ولم يبق رقم: \omterm{def:g3:division:remainder}{فخارج القسمة} هو $124$، \omterm{def:g3:division:remainder}{والباقي} $5$.
\end{enumerate}
والتحقّق: $6 \times 124 + 5 = 744 + 5 = 749$، و $5 < 6$.
\end{method}

\begin{example}[صفر في خارج القسمة]\label{ex:g4:division:zero}
اِقسم $618$ على $3$: المئات، $6 \div 3 = 2$؛ والعشرات، أنزِل
$1$: يدخل $3$ عددَ مرات $0$ في $1$ --- \emph{فاكتب} \omterm{def:g1:counting:zero}{الصفر} $0$! ---
\omterm{def:g3:division:remainder}{والباقي} $1$؛ والآحاد، أنزِل $8$: $18 \div 3 = 6$. \omterm{def:g3:division:remainder}{فخارج القسمة}
$206$، \omterm{def:g3:division:remainder}{والباقي} $0$. ونسيان \omterm{def:g1:counting:zero}{الصفر} الأوسط (بكتابة $26$)
هو الخطأ الكلاسيكي؛ والتحقّق
$3 \times 26 = 78 \neq 618$ يكشفه فورًا.
\end{example}

\begin{example}[تقدير حجم خارج القسمة]\label{ex:g4:division:size}
قبل قسمة $749$ على $6$، أطِّر الجواب:
$6 \times 100 = 600$ و $6 \times 200 = 1\,200$، إذن \omterm{def:g3:division:remainder}{خارج القسمة}
بين $100$ و $200$ --- فسيكون من ثلاثة أرقام. وهذا التقدير من سطر
واحد يمنع أكثر أخطاء وضع الأرقام في غير مواضعها.
\end{example}

\section{اختيار ما يطلبه السؤال}

\begin{method}[خارج القسمة، أم الباقي، أم خارج القسمة زائد واحد]\label{met:g4:division:interpret}
كما في \cref{met:g3:division:interpret}، القصة هي التي تقرر:
\begin{enumerate}
  \item «كم عددًا كاملًا \dots» أو «كم لكل واحد»: \omterm{def:g3:division:remainder}{خارج القسمة}؛
  \item «وكم بقي»: \omterm{def:g3:division:remainder}{الباقي}؛
  \item «وكم وعاءً يلزم للجميع»: \omterm{def:g3:division:remainder}{خارج القسمة}، زائد
        واحد إن لم يكن \omterm{def:g3:division:remainder}{الباقي} صفرًا.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g4:division:problem}
يجب رصّ $530$ كتابًا في علب من $8$.
\[
530 = 8 \times 66 + 2 .
\]
فكم علبة \emph{ممتلئة}؟ $66$. وكم كتابًا خارج العلب
الممتلئة؟ $2$. وكم علبة لرصّ \emph{كل} الكتب؟ $67$.
ثلاثة أسئلة، وقسمة واحدة.
\end{example}

\begin{example}[تقاسم عادل للمال]\label{ex:g4:division:money}
يتقاسم أربعة أصدقاء ثمن هدية قدره $92$ بالتساوي:
$92 \div 4 = 23$ بالضبط (\omterm{def:g3:division:remainder}{والباقي} $0$). فيدفع كل واحد $23$. وحين
يكون \omterm{def:g3:division:remainder}{الباقي} صفرًا، تكون القسمة «تامّة» ويكون التقاسم عادلًا
تمامًا.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:division:1}
أطِّر كل \omterm{def:g3:division:remainder}{خارج قسمة} كما في \cref{ex:g4:division:size} (بين أي
عددين «مستديرين»؟)، ثم احسب القسمة المطوّلة:
$96 \div 4$؛ \quad $87 \div 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:2}
احسب القسمات المطوّلة وتحقّق من كل واحدة:
$672 \div 4$؛ \quad $925 \div 7$؛ \quad $804 \div 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:3}
اِنتبه إلى \omterm{def:g1:counting:zero}{الصفر} (اُنظر \cref{ex:g4:division:zero}):
$816 \div 4$؛ \quad $420 \div 7$؛ \quad $2\,512 \div 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:4}
اُكتب مساواة التحقّق ($a = b \times q + r$) في: $85 \div 9$؛
\quad $1\,000 \div 3$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:5}
تُوضع $156$ بيضة في علب من $6$. فكم علبة تمتلئ؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:6}
يُقطع شريط طوله $250$ سم إلى قطع من $8$ سم. فكم قطعة،
وما الطول الذي يفضل منه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:7}
يذهب $375$ تلميذًا إلى عرض في حافلات فيها $52$ مقعدًا. فكم
حافلة تلزم؟ (وأي حالة من حالات
\cref{met:g4:division:interpret} هذه؟ اِقسم على $52$ مستعملًا
المضاعفات: $52 \times 7 = 364$.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:8}
يتقاسم ثلاثة أصدقاء $234$ كرة بالتساوي. فكم كرة لكل واحد؟
تحقّق بعملية ضرب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:9}
جِد المقسوم: القسمة على $7$ تعطي \omterm{def:g3:division:remainder}{خارج قسمة} $58$
وباقيًا $3$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:division:10}
يرصّ أمين مكتبة $438$ كتابًا، $9$ في كل رفّ. والرفوف تأتي في
خزائن من $6$ رفوف. فكم رفًّا يلزم؟ وكم خزانة يجب أن
يشتري؟ (قسمتان، وفي كلتيهما سؤال
«زائد واحد».)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:division:11}
في قسمة على $8$، يساوي \omterm{def:g3:division:remainder}{خارج القسمة} الباقيَ. فما المقسومات
الممكنة؟ اُذكرها كلها. (\omterm{def:g3:division:remainder}{فالباقي} يجب أن يبقى
دون $8$.)
\end{exercise}
