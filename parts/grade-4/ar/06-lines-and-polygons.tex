\chapter{المستقيمات والمضلعات}\label{ch:g4:geometry}

بالمسطرة والكوس والفرجار تصير الهندسة حرفة. يعلّم هذا الفصل
الوضعين الخاصّين للمستقيمات --- التعامد والتوازي --- على مستوى
الأدوات، ويطوف بعائلة المضلعات. (ونظرية هذه المفاهيم تأتي في
\cref{ch:g6:lines}.)

\section{التعامد والتوازي}

\begin{definition}[المتعامدان والمتوازيان]\label{def:g4:geometry:def}
يكون المستقيمان \emph{متعامدين}\index{مستقيمان متعامدان} حين
يتقاطعان في \omterm{def:g3:shapes:rightangle}{زاوية قائمة} (يُتحقَّق منها بالكوس، وتُعلَّم بمربع
صغير). ويكونان \emph{متوازيين}\index{مستقيمان متوازيان} حين
يسيران في الاتجاه نفسه ولا يتقاطعان أبدًا --- كسكّتَي قطار،
على المسافة نفسها من بعضهما دائمًا.
\end{definition}

\begin{method}[الرسم بالكوس]\label{met:g4:geometry:draw}
\emph{العمودي على المستقيم $d$ المارّ بالنقطة $P$}: ضع أحد
حرفَي \omterm{def:g3:shapes:rightangle}{الزاوية القائمة} على $d$، وأزلق حتى يبلغ الحرف الآخر $P$،
ثم اُرسم.

\emph{الموازي للمستقيم $d$ المارّ بالنقطة $P$}: ضع حرف الكوس
على $d$، وثبّت المسطرة على حرف آخر كسكّة، وأزلق الكوس على
المسطرة حتى $P$، ثم اُرسم.
\end{method}

\begin{example}[التعرّف عليهما في الأشكال]\label{ex:g4:geometry:spot}
في المستطيل، كل ضلع عمودي على جاريه وموازٍ للضلع
المقابل. والورق المربّعات شبكة جاهزة من المتوازيات
والمتعامدات --- وهذا ما يجعل الرسم عليه مريحًا
جدًّا.
\end{example}

\section{المضلعات}

\begin{definition}[المضلع]\label{def:g4:geometry:polygon}
\emph{المضلع}\index{مضلع} شكل مغلق حدّه مصنوع من قطع
مستقيمة، وهي \emph{أضلاعه}؛ ونقاط الأركان هي
\emph{رؤوسه}\index{رأس}. وتُسمَّى المضلعات بعدد أضلاعها:
\emph{مثلث} ($3$)، و\emph{رباعي}\index{رباعي} ($4$)،
و\emph{خماسي}\index{خماسي} ($5$)، و\emph{سداسي}\index{سداسي} ($6$)،
و\emph{مثمّن}\index{مثمّن}
($8$).
\end{definition}

\begin{example}[وصف مضلع]\label{ex:g4:geometry:describe}
لعبة مفيدة: صِف شكلًا وصفًا دقيقًا حتى يستطيع زميل أن
يعيد رسمه دون أن يراه. «\omterm{def:g4:geometry:polygon}{رباعي} $ABCD$ فيه
$AB = 5$ سم، والزاوية عند $A$ قائمة، و $AD = 3$ سم\,\dots» ---
فالمفردات هي التي تجعل الرسالة تصل.
\end{example}

\section{الدائرة والفرجار}

\begin{definition}[الدائرة]\label{def:g4:geometry:circle}
\emph{الدائرة} التي مركزها $O$ ونصف قطرها $3$ سم هي مجموعة كل
النقط التي تبعد $3$ سم بالضبط عن $O$. والفرجار يرسمها:
السنّ على $O$، والفتحة $3$ سم، ثم أدِر. و\emph{القطر}\index{قطر}
قطعة تمرّ بالمركز وتصل بين نقطتين من الدائرة؛ وطوله ضعف
\omterm{def:g3:shapes:shapes}{نصف القطر}.
\end{definition}

\begin{example}[الفرجار حافظًا للمسافات]\label{ex:g4:geometry:compass}
الفرجار يفعل أكثر من رسم الدوائر: فهو \emph{ينقل
المسافات}. فلتعليم كل النقط التي تبعد $4$ سم عن نقطة $A$، اُرسم
الدائرة التي مركزها $A$ ونصف قطرها $4$ سم؛ ولمقارنة قطعتين بلا
مسطرة، اِفتح الفرجار على إحداهما واختبره على الأخرى.
\end{example}

\begin{method}[نقل شكل على شبكة]\label{met:g4:geometry:reproduce}
\begin{enumerate}
  \item اختر رأسًا للبداية وموضعه على الشبكة الفارغة؛
  \item وسِر على الشكل رأسًا بعد \omterm{def:g4:geometry:polygon}{رأس}، عادًّا خطوات
        الشبكة («$3$ يمينًا و $2$ فوق»)؛
  \item وصِل الرؤوس بالترتيب، ثم قارن بالنموذج:
        الأطوال نفسها، والاتجاهات نفسها.
\end{enumerate}
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g4:geometry:1}
اُرسم مستقيمًا $d$ ونقطة $P$ خارجه. وأنشئ
العمودي على $d$ المارّ بالنقطة $P$، ثم الموازي للمستقيم $d$
المارّ بالنقطة $P$ (بطريقة المسطرة السكّة). وعلّم \omterm{def:g3:shapes:rightangle}{الزاوية القائمة}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:2}
اُنظر إلى الحروف الكبيرة E و H و N و T و Z. وجِد في كل
منها زوجًا من الخطوط المتوازية وزوجًا من الخطوط المتعامدة إن
وُجدا.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:3}
سمِّ كل \omterm{def:g4:geometry:polygon}{مضلع}: $6$ أضلاع؛ و $4$ أضلاع؛ و $3$ أضلاع؛ و $8$ أضلاع؛
و $5$ أضلاع. وكم \emph{رأسًا} لكل واحد منها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:4}
اُرسم خماسيًّا أيًّا كان (وليس بالضرورة منتظمًا) وسمِّ \omterm{def:g4:geometry:polygon}{رؤوسه}
$A$ و $B$ و $C$ و $D$ و $E$. فكم ضلعًا له؟ واُرسم أقطاره
الخارجة من $A$ (قطع تصل $A$ برؤوس غير مجاورة):
فكم عددها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:5}
اُرسم دائرة مركزها $O$ ونصف قطرها $4$ سم؛ وقطرًا $[AB]$؛
ونقطة $M$ من الدائرة ليست على $[AB]$. وأعطِ الأطوال $OA$
و $OM$ و $AB$ بلا قياس.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:6}
بالفرجار حافظًا للمسافات
(\cref{ex:g4:geometry:compass})، قارن أضلاع مثلث رسمه زميل
لك، وقرّر هل فيه ضلعان متساويان.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:7}
على الورق المربّعات، اِنقل هذا المسار وأغلقه فيصير مضلعًا:
اِبدأ من نقطة $A$؛ وسِر $4$ يمينًا؛ ثم $2$ فوق؛ ثم $3$ يسارًا؛ ثم $2$ فوق؛
ثم عُد إلى $A$ في خط مستقيم. فما اسم \omterm{def:g4:geometry:polygon}{المضلع}
الذي تحصل عليه (عُدّ أضلاعه)؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:8}
صواب أم خطأ؟ اِشرح برسم.
«\omterm{def:g4:geometry:def}{المستقيمان المتعامدان} يتقاطعان دائمًا».
«\omterm{def:g4:geometry:def}{المستقيمان المتوازيان} لا يتقاطعان أبدًا».
«يمكن أن يكون مستقيم موازيًا لمستقيم وعموديًّا على
آخر».
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:9}
اُرسم مستقيمًا $d$، ثم مستقيمًا $e$ عموديًّا على $d$، ثم مستقيمًا
$f$ عموديًّا على $e$. واُنظر إلى $d$ و $f$: فماذا تلاحظ؟
(وتصير هذه الملاحظة مبرهنة في \cref{ch:g6:lines}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:10}
صِف الشكل التالي بالكلمات لزميل، دون أن تريه
إياه: مربع طول ضلعه $4$ سم، وفيه دائرة نصف قطرها $2$ سم مركزها
منتصف المربع (تمسّ الأضلاع الأربعة). واُكتب الوصف
تعليمات مرقّمة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:11}
كم قطرًا \omterm{def:g4:geometry:polygon}{للرباعي}؟ وللخماسي؟ وللسداسي؟
اُرسم الحالات الثلاث، وعُدّ بعناية، وصِف النمط الذي
تراه.
\end{exercise}
