\chapter{La división}\label{ch:g4:division}

El año pasado dividimos con las tablas y \omterm{def:g3:division:remainder}{restos} pequeños
(\cref{ch:g3:division}); este capítulo instala el método escrito
completo --- la división larga --- que sirve para cualquier número, grupo de cifras
a grupo de cifras.

\section{El método de la división larga}

\begin{method}[Dividir por un número de una cifra]\label{met:g4:division:method}
Para dividir $749$ entre $6$:
\begin{enumerate}
  \item toma la cifra de más a la izquierda, $7$ (centenas): el $6$ cabe $1$ vez
        en $7$; escribe $1$ en el \omterm{def:g3:division:remainder}{cociente}, resta
        $6$ y queda \omterm{def:g3:division:remainder}{resto} $1$;
  \item \emph{baja} la cifra siguiente, el $4$: ahora divide $14$
        (decenas): $6 \times 2 = 12$; escribe $2$, \omterm{def:g3:division:remainder}{resto} $2$;
  \item baja el $9$: divide $29$: $6 \times 4 = 24$; escribe
        $4$, \omterm{def:g3:division:remainder}{resto} $5$;
  \item ya no quedan cifras: el \omterm{def:g3:division:remainder}{cociente} es $124$ y el \omterm{def:g3:division:remainder}{resto}, $5$.
\end{enumerate}
Comprobación: $6 \times 124 + 5 = 744 + 5 = 749$, y $5 < 6$.
\end{method}

\begin{example}[Un cero en el cociente]\label{ex:g4:division:zero}
Divide $618$ entre $3$: centenas, $6 \div 3 = 2$; decenas, baja el
$1$: el $3$ cabe $0$ veces en $1$ --- ¡\emph{escribe el} $0$! ---
y queda \omterm{def:g3:division:remainder}{resto} $1$; unidades, baja el $8$: $18 \div 3 = 6$. \omterm{def:g3:division:remainder}{Cociente}
$206$, \omterm{def:g3:division:remainder}{resto} $0$. Olvidar el \omterm{def:g1:counting:zero}{cero} del medio (y escribir $26$)
es el error clásico; la comprobación
$3 \times 26 = 78 \neq 618$ lo detecta al instante.
\end{example}

\begin{example}[Estimar el tamaño del cociente]\label{ex:g4:division:size}
Antes de dividir $749$ entre $6$, acota la respuesta:
$6 \times 100 = 600$ y $6 \times 200 = 1\,200$, así que el \omterm{def:g3:division:remainder}{cociente} está
entre $100$ y $200$ --- tendrá tres cifras. Esta estimación de una línea
evita casi todos los errores de cifra mal colocada.
\end{example}

\section{Elegir lo que pide la pregunta}

\begin{method}[Cociente, resto o cociente más uno]\label{met:g4:division:interpret}
Como en el \cref{met:g3:division:interpret}, la historia decide:
\begin{enumerate}
  \item «cuántos \dots\ completos» o «cuántos para cada uno»: el \omterm{def:g3:division:remainder}{cociente};
  \item «cuántos sobran»: el \omterm{def:g3:division:remainder}{resto};
  \item «cuántos recipientes hacen falta para todo»: el \omterm{def:g3:division:remainder}{cociente} más
        uno si el \omterm{def:g3:division:remainder}{resto} no es \omterm{def:g1:counting:zero}{cero}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g4:division:problem}
Hay que empaquetar $530$ libros en cajas de $8$.
\[
530 = 8 \times 66 + 2 .
\]
¿Cuántas cajas \emph{completas}? $66$. ¿Cuántos libros quedan fuera de una caja
completa? $2$. ¿Cuántas cajas para empaquetar \emph{todos} los libros? $67$.
Tres preguntas, una sola división.
\end{example}

\begin{example}[Repartir un gasto]\label{ex:g4:division:money}
Cuatro amigos comparten a partes iguales el coste de un regalo de $92$:
$92 \div 4 = 23$ exactamente (\omterm{def:g3:division:remainder}{resto} $0$). Cada uno paga $23$. Cuando el
\omterm{def:g3:division:remainder}{resto} es \omterm{def:g1:counting:zero}{cero}, la división «es exacta» y el reparto
es perfectamente justo.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:division:1}
Acota cada \omterm{def:g3:division:remainder}{cociente} como en el \cref{ex:g4:division:size} (¿entre qué
dos números «redondos»?) y calcula después la división larga:
$96 \div 4$; \quad $87 \div 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:2}
Calcula las divisiones largas y comprueba cada una:
$672 \div 4$; \quad $925 \div 7$; \quad $804 \div 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:3}
Cuidado con el \omterm{def:g1:counting:zero}{cero} (véase el \cref{ex:g4:division:zero}):
$816 \div 4$; \quad $420 \div 7$; \quad $2\,512 \div 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:4}
Escribe la igualdad de comprobación ($a = b \times q + r$) para: $85 \div 9$;
\quad $1\,000 \div 3$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:5}
Se meten $156$ huevos en hueveras de $6$. ¿Cuántas hueveras se llenan?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:6}
Una cinta de $250$ cm se corta en trozos de $8$ cm. ¿Cuántos trozos salen
y qué longitud sobra?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:7}
$375$ alumnos van a un espectáculo en autobuses de $52$ plazas. ¿Cuántos autobuses
hacen falta? (¿Qué caso del
\cref{met:g4:division:interpret} es este? Divide entre $52$ usando
múltiplos: $52 \times 7 = 364$.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:8}
Tres amigos se reparten $234$ canicas a partes iguales. ¿Cuántas canicas para cada uno?
Compruébalo con una multiplicación.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:9}
Halla el dividendo: la división entre $7$ da \omterm{def:g3:division:remainder}{cociente} $58$ y
\omterm{def:g3:division:remainder}{resto} $3$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:division:10}
Un bibliotecario coloca $438$ libros, $9$ por estante. Los estantes vienen en
estanterías de $6$ estantes. ¿Cuántos estantes hacen falta? ¿Cuántas
estanterías hay que comprar? (Dos divisiones, las dos con una
pregunta de «más uno».)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:division:11}
En una división entre $8$, el \omterm{def:g3:division:remainder}{cociente} es igual al \omterm{def:g3:division:remainder}{resto}. ¿Cuáles son
los dividendos posibles? Enuméralos todos. (El \omterm{def:g3:division:remainder}{resto} debe quedarse
por debajo de $8$.)
\end{exercise}
