\section{El método de la división larga}

\begin{method}[División larga por un número de un dígito]\label{met:g4:division:method}
dividir $749$ por $6$:
\begin{enumerate}
  \item toma el dígito más a la izquierda, $7$ (cientos): $6$ va $1$ tiempo
        en $7$; escribir $1$ en el cociente, restar
        $6$, \omterm{def:g3:division:remainder}{resto} $1$;
  \item \emph{reducir} el siguiente dígito, $4$: ahora divide $14$
        (decenas): $6 \times 2 = 12$; escribir $2$, \omterm{def:g3:division:remainder}{resto} $2$;
  \item derribar el $9$: dividir $29$: $6 \times 4 = 24$; escribir
        $4$, \omterm{def:g3:division:remainder}{resto} $5$;
  \item no queda ningún dígito: el cociente es $124$, el \omterm{def:g3:division:remainder}{resto} $5$.
\end{enumerate}
Controlar: $6 \times 124 + 5 = 744 + 5 = 749$, y $5 < 6$.
\end{method}

\begin{example}[Un cero en el cociente.]\label{ex:g4:division:zero}
Dividir $618$ por $3$: cientos, $6 \div 3 = 2$; decenas, baja el
$1$: $3$ va $0$ veces en $1$ --- \emph{escribe el} $0$! ---
\omterm{def:g3:division:remainder}{resto} $1$; unidades, bajar el $8$: $18 \div 3 = 6$. cociente
$206$, \omterm{def:g3:division:remainder}{resto} $0$. olvidando el \omterm{def:g3:division:half}{medio} \omterm{def:g1:counting:zero}{cero} (escribiendo $26$)
es el error clásico; el cheque
$3 \times 26 = 78 \neq 618$ lo atrapa inmediatamente.
\end{example}

\begin{example}[Estimando el tamaño del cociente]\label{ex:g4:division:size}
antes de dividir $749$ por $6$, enmarca la respuesta:
$6 \times 100 = 600$ y $6 \times 200 = 1\,200$, entonces el cociente es
entre $100$ y $200$ --- tendrá tres dígitos. Esta línea
La estimación evita la mayoría de errores de dígitos extraviados.
\end{example}

\section{Elegir lo que pregunta la pregunta}

\begin{method}[Cociente, resto o cociente más uno]\label{met:g4:division:interpret}
Como en \cref{met:g3:division:interpret}, la historia decide:
\begin{enumerate}
  \item ``cuantos llenos \dots'' o ``cuántos cada uno'': el cociente;
  \item ``cuantos quedan'': el \omterm{def:g3:division:remainder}{resto};
  \item ``cuántos contenedores se necesitan para todos'': el cociente, más
        uno si el \omterm{def:g3:division:remainder}{resto} no es \omterm{def:g1:counting:zero}{cero}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g4:division:problem}
$530$ Los libros deben empacarse en cajas de $8$.
\[
530 = 8 \times 66 + 2 .
\]
Cuántos \emph{lleno} cajas? $66$. ¿Cuántos libros no están completos?
caja? $2$. cuantas cajas empacar \emph{todo} los libros? $67$.
Tres preguntas, una división.
\end{example}

\begin{example}[reparto justo del dinero]\label{ex:g4:division:money}
Cuatro amigos comparten el costo de una $92$ regalo por igual:
$92 \div 4 = 23$ exactamente (\omterm{def:g3:division:remainder}{resto} $0$). cada uno paga $23$. cuando el
\omterm{def:g3:division:remainder}{resto} es \omterm{def:g1:counting:zero}{cero}, la división ``sale igualada'' y el reparto
es perfectamente justo.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g4:division:1}
Enmarcar cada uno cociente como en \cref{ex:g4:division:size} (entre los cuales
dos ``\omterm{def:g4:numbers:round}{redondo}'' ¿números?), luego calcula la división larga:
$96 \div 4$; \quad $87 \div 5$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:division:2}
Calcula las divisiones largas y verifica cada una:
$672 \div 4$; \quad $925 \div 7$; \quad $804 \div 6$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:division:3}
Cuidado con el \omterm{def:g1:counting:zero}{cero} (ver \cref{ex:g4:division:zero}):
$816 \div 4$; \quad $420 \div 7$; \quad $2\,512 \div 5$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:division:4}
Escribe la igualdad de control ($ a = b \times q + r $) para: $85 \div 9$;
\quad $1\,000 \div 3$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:division:5}
$156$ Los huevos se ponen en cajas de $6$. ¿Cuántas cajas están llenas?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:division:6}
una cinta de $250$ cm se corta en trozos de $8$ centímetro. cuantas piezas
y que largo sobra?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:division:7}
$375$ Los estudiantes van a un espectáculo en autobuses de $52$ asientos. cuantos autobuses
son necesarios? (¿Qué caso de
\cref{met:g4:division:interpret} ¿Es esto? dividir por $52$ usando
múltiplos: $52 \times 7 = 364$.)
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:division:8}
tres amigos comparten $234$ canicas por igual. ¿Cuántas canicas cada una?
Compruébalo con una multiplicación.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:division:9}
Encuentra el dividendo: la división por $7$ da cociente $58$ y
\omterm{def:g3:division:remainder}{resto} $3$.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g4:division:10}
Los estantes de un bibliotecario $438$ libros, $9$ por estante. Los estantes entran
libreros de $6$ estantes. ¿Cuántos estantes se necesitan? cuantos
¿Hay que comprar librerías? (Dos divisiones, ambas con un
pregunta "más uno".)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g4:division:11}
en una división por $8$, el cociente es igual al \omterm{def:g3:division:remainder}{resto}. que son
¿Los posibles dividendos? Enumérelos todos. (El \omterm{def:g3:division:remainder}{resto} debe quedarse
abajo $8$.)
\end{exercise}
