\end{example}

\section{Perímetro}

\begin{definition}[Perímetro]\label{def:g4:measure:perimeter}
El \emph{perímetro}\index{perímetro} de un \omterm{def:g4:geometry:polygon}{polígono} es el total
longitud de su borde: \omterm{def:g1:addition:def}{suma} las longitudes de todos sus lados (en la misma
unidad!). Atajos para los clásicos:
\[
\text{rectangle: } P = 2 \times (L + w),
\qquad
\text{square: } P = 4 \times c .
\]
\end{definition}

\begin{example}\label{ex:g4:measure:perimeter}
Un jardín rectangular $12$ m por $7$ metro:
\[
P = 2 \times (12 + 7) = 2 \times 19 = 38 \text{ m}
\]
de valla. Un marco de fotos cuadrado de lado. $18$ centímetro:
$ P = 4 \times 18 = 72$ centímetro.
\end{example}

\section{Área: contar cuadrados}

\begin{definition}[Área contando]\label{def:g4:measure:area}
La \emph{área}\index{área} de una figura dibujada en una cuadrícula es el número
de \omterm{def:g3:shapes:shapes}{cuadrados} de la cuadrícula que cubre. Dos \omterm{def:g3:division:half}{medios} \omterm{def:g3:shapes:shapes}{cuadrados} cuentan como uno. Área
responde ``cuánta superficie'', \omterm{def:g4:measure:perimeter}{perímetro} responde "¿cuánto mide el
frontera '' --- ¡dos preguntas diferentes!
\end{definition}

\section{Duraciones}

\begin{method}[Agregar tiempos a lo largo de la hora]\label{met:g4:measure:time}
Para sumar o restar duraciones, trabaje en saltos durante las horas completas
(recordar $1$ h $= 60$ mín.):
\[
9\text{ h }40 + 35 \text{ min}:
\quad 9\text{ h }40 \xrightarrow{+20} 10\text{ h }00
\xrightarrow{+15} 10\text{ h }15 .
\]
Nunca agregue minutos pasados $60$ sin convertir: $40 + 35 = 75$ mín.
$= 1$ h $15$ mín.
\end{method}

\begin{example}\label{ex:g4:measure:timetable}
El tren sale a las $8{:}47$ y llega a $11{:}05$. Duración:
$8{:}47 \to 9{:}00$ es $13$ mín; $9{:}00 \to 11{:}00$ es $2$ h;
$11{:}00 \to 11{:}05$ es $5$ mín. Total: $2$ h $18$ mín.
\end{example}

\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:measure:2}
Convertir: $2$ kg en g; $1\,800$ g en kg y g; $3.5$ L en cl;
$25$ cL en L.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:measure:3}
Calcular el \omterm{def:g4:measure:perimeter}{perímetro}: un rectángulo $9$ centímetro $\times $ $5$ centímetro; un cuadrado
de lado $7.5$ centímetro; un \omterm{def:g3:shapes:shapes}{triángulo} con lados $6$ centímetro, $8$ centímetros y $10$ centímetro.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:measure:4}
Un rectángulo tiene \omterm{def:g4:measure:perimeter}{perímetro} $26$ cm y longitud $8$ centímetro. Encuentra su ancho

\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:measure:5}
