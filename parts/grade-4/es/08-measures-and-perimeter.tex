\chapter{Medidas y perímetro}\label{ch:g4:measure}

El año pasado medimos con reglas, balanzas y relojes
(\cref{ch:g3:measure}); este capítulo añade la tabla de unidades --- la
máquina que lo convierte todo --- y dos maneras de medir una figura
plana: la longitud de su borde (el \omterm{def:g4:measure:perimeter}{perímetro}) y el sitio que hay dentro
(el \omterm{def:g4:measure:area}{área}, contada en casillas por ahora).

\section{La tabla de unidades}

\begin{method}[Convertir con la tabla]\label{met:g4:measure:table}
Para las longitudes, cada columna vale diez veces la siguiente:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|c|c|c|c|c|c}
km & hm & dam & m & dm & cm & mm \\
\hline
 & & & $3$ & $4$ & $0$ & \\
\end{tabular}
\end{center}
\begin{enumerate}
  \item escribe el número con una cifra por columna, y la cifra de las unidades
        de la medida en la columna de su unidad (aquí, $3.4$ m: el $3$
        debajo de m);
  \item lee el número en la unidad nueva: $3.4$ m $= 340$ cm (rellena
        con \omterm{def:g1:counting:zero}{ceros} las columnas vacías);
  \item las \omterm{def:g2:measure:units}{masas} (kg\,\dots\,g) y las \omterm{def:g2:measure:units}{capacidades} (L\,\dots\,cL) funcionan
        igual.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g4:measure:convert}
$5$ km $= 5\,000$ m; \quad $270$ cm $= 2.7$ m (el $2$ cae debajo de
m); \quad $3$ kg $250$ g $= 3\,250$ g; \quad
$2.5$ L $= 250$ cL.
\end{example}

\section{El perímetro}

\begin{definition}[Perímetro]\label{def:g4:measure:perimeter}
El \emph{perímetro}\index{perímetro} de un \omterm{def:g4:geometry:polygon}{polígono} es la longitud
total de su borde: \omterm{def:g1:addition:def}{suma} las longitudes de todos sus lados (¡en la misma
unidad!). Atajos para los casos clásicos:
\[
\text{rectángulo: } P = 2 \times (L + a),
\qquad
\text{cuadrado: } P = 4 \times c .
\]
\end{definition}

\begin{example}\label{ex:g4:measure:perimeter}
Un jardín rectangular de $12$ m por $7$ m:
\[
P = 2 \times (12 + 7) = 2 \times 19 = 38 \text{ m}
\]
de valla. Un marco de fotos cuadrado de lado $18$ cm:
$P = 4 \times 18 = 72$ cm.
\end{example}

\section{El área: contar casillas}

\begin{definition}[Área contando casillas]\label{def:g4:measure:area}
El \emph{área}\index{área} de una figura dibujada en una cuadrícula es el número
de casillas que cubre. Dos medias casillas cuentan como una. El área
responde a «cuánta superficie» y el \omterm{def:g4:measure:perimeter}{perímetro}, a «cuánto mide el
borde» --- ¡dos preguntas distintas!
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=0.55]
  \draw[gray!40, thin] (0,0) grid (13,4);
  \draw[very thick, omDef, fill=omDef!15] (1,1) rectangle (5,3);
  \node[font=\small, omDef] at (3,2) {$8$ casillas};
  \draw[very thick, omThm, fill=omThm!15]
    (7,1) -- (11,1) -- (11,2) -- (9,2) -- (9,3) -- (7,3) -- cycle;
  \node[font=\small, omThm] at (8.4,1.5) {$6$ casillas};
\end{tikzpicture}

{\small Dos figuras en la cuadrícula: \omterm{def:g4:measure:area}{áreas} de $8$ y $6$ casillas. Ahora recorre
los bordes y cuenta las unidades: ¡$12$ en las dos! Mismo \omterm{def:g4:measure:perimeter}{perímetro},
\omterm{def:g4:measure:area}{áreas} distintas.}
\end{omfigure}

\begin{example}[Medias casillas]\label{ex:g4:measure:half}
Un triángulo rectángulo dibujado en la cuadrícula con catetos $4$ y $2$ cubre $3$
casillas enteras y $2$ medias casillas\,\dots\ cuenta con cuidado: su \omterm{def:g4:measure:area}{área}
es la \omterm{def:g3:division:half}{mitad} del rectángulo de $4 \times 2$, es decir, $4$ casillas. Cortar un
rectángulo por su diagonal da siempre dos triángulos de igual
\omterm{def:g4:measure:area}{área}.
\end{example}

\begin{example}[Misma área, perímetros distintos]\label{ex:g4:measure:samearea}
Un cuadrado de $4 \times 4$ y un rectángulo de $8 \times 2$ cubren los dos $16$
casillas --- la misma \omterm{def:g4:measure:area}{área}. Los bordes: $16$ unidades el cuadrado y $20$ el
rectángulo. Ninguna de las dos medidas determina la otra: una figura larga y estrecha
tiene mucho borde para poca superficie.
\end{example}

\section{Duraciones}

\begin{method}[Sumar horas cruzando la hora en punto]\label{met:g4:measure:time}
Para sumar o restar duraciones, avanza a saltos por las horas completas
(recuerda que $1$ h $= 60$ min):
\[
9\text{ h }40 + 35 \text{ min}:
\quad 9\text{ h }40 \xrightarrow{+20} 10\text{ h }00
\xrightarrow{+15} 10\text{ h }15 .
\]
Nunca sumes minutos por encima de $60$ sin convertir: $40 + 35 = 75$ min
$= 1$ h $15$ min.
\end{method}

\begin{example}\label{ex:g4:measure:timetable}
El tren sale a las $8{:}47$ y llega a las $11{:}05$. Duración:
de $8{:}47$ a $9{:}00$ hay $13$ min; de $9{:}00$ a $11{:}00$, $2$ h;
de $11{:}00$ a $11{:}05$, $5$ min. Total: $2$ h $18$ min.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:measure:1}
Convierte con la tabla de unidades: $7$ m en cm; $3.2$ km en m; $85$ mm en
cm; $4\,500$ m en km.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:2}
Convierte: $2$ kg en g; $1\,800$ g en kg y g; $3.5$ L en cL;
$25$ cL en L.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:3}
Calcula el \omterm{def:g4:measure:perimeter}{perímetro}: un rectángulo de $9$ cm $\times$ $5$ cm; un cuadrado
de lado $7.5$ cm; un triángulo de lados $6$ cm, $8$ cm y $10$ cm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:4}
Un rectángulo tiene \omterm{def:g4:measure:perimeter}{perímetro} $26$ cm y largo $8$ cm. Halla su ancho
(halla primero largo $+$ ancho).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:5}
En papel cuadriculado, dibuja una figura en forma de L que cubra exactamente $10$
casillas y cuenta las unidades de su \omterm{def:g4:measure:perimeter}{perímetro}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:6}
Dibuja dos rectángulos distintos de \omterm{def:g4:measure:area}{área} $12$ casillas. Calcula el
\omterm{def:g4:measure:perimeter}{perímetro} de cada uno. ¿Cuál es más «redondo» y cuál más «estrecho»?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:7}
Se dibuja en la cuadrícula un triángulo rectángulo de catetos $6$ y $3$. ¿Cuál es
su \omterm{def:g4:measure:area}{área} en casillas? (Usa el \cref{ex:g4:measure:half}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:8}
Calcula las duraciones: de $10{:}20$ a $12{:}45$; de $7{:}35$ a
$9{:}10$. Una película dura $1$ h $50$ y empieza a las $20{:}30$: ¿a qué hora
termina?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:measure:9}
Kim da $3$ vueltas a un campo rectangular de $60$ m por $45$ m. ¿Cuántos
metros corre?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:measure:10}
Un granjero quiere vallar un campo cuadrado de lado $85$ m dejando
una cancela de $4$ m sin vallar. ¿Cuántos metros de valla tiene que comprar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:measure:11}
¿Verdadero o falso? Justifícalo con ejemplos en la cuadrícula: «si dos figuras tienen el mismo
\omterm{def:g4:measure:perimeter}{perímetro}, tienen la misma \omterm{def:g4:measure:area}{área}»; «si dos figuras tienen la misma
\omterm{def:g4:measure:area}{área}, tienen el mismo \omterm{def:g4:measure:perimeter}{perímetro}».
(\Cref{ex:g4:measure:samearea} y la figura del capítulo te ayudarán.)
\end{exercise}
