\chapter{Décimas y centésimas}\label{ch:g4:decimals}

Entre $3$ y $4$ los números enteros callan --- pero un corredor puede
ir $3$ metros y pico por delante. Cortar la unidad en diez y luego en
cien da las \emph{fracciones decimales} y una abreviatura estupenda
para ellas: el \omterm{def:g4:decimals:point}{punto decimal}. (Los números decimales tienen su
capítulo entero en el \cref{ch:g5:decimals}.)

\section{Fracciones decimales}

\begin{definition}[Décimas y centésimas]\label{def:g4:decimals:def}
Cortar una unidad en $10$ partes iguales da las \emph{décimas},
$\frac{1}{10}$; cortar cada décima otra vez en $10$ da las
\emph{centésimas}, $\frac{1}{100}$. Así,
\[
1 = \frac{10}{10} = \frac{100}{100},
\qquad
\frac{1}{10} = \frac{10}{100}.
\]
\end{definition}

\begin{example}\label{ex:g4:decimals:count}
Sombrear $3$ columnas y $7$ casillas sueltas del cuadrado de cien sombrea
\[
\frac{3}{10} + \frac{7}{100} = \frac{30}{100} + \frac{7}{100}
= \frac{37}{100}
\]
del cuadrado --- treinta y siete centésimas.
\end{example}

\section{El punto decimal}

\begin{definition}[La escritura decimal]\label{def:g4:decimals:point}
El número $2 + \frac{3}{10} + \frac{7}{100}$ se escribe
\[
2.37
\]
--- el \emph{punto decimal}\index{punto decimal} separa la parte entera ($2$) de la
parte decimal; la primera cifra después del punto cuenta las décimas y la
segunda, las centésimas. Se lee: «dos punto tres siete» o «dos
y treinta y siete centésimas».
\end{definition}

\begin{example}\label{ex:g4:decimals:convert}
\[
\frac{7}{10} = 0.7, \qquad
\frac{43}{100} = 0.43, \qquad
\frac{5}{100} = 0.05, \qquad
3 + \frac{4}{10} = 3.4 .
\]
Cuidado con $\frac{5}{100}$: cinco \emph{centésimas} necesitan un $0$ en
el lugar de las décimas --- $0.05$, no $0.5$.
\end{example}

\begin{example}[El dinero son centésimas]\label{ex:g4:decimals:money}
Un céntimo es una centésima de euro (o de dólar, o de libra): un precio
de $4.85$ significa $4$ monedas enteras y $85$ centésimas. El dinero es el
\omterm{def:g4:decimals:point}{punto decimal} que ya usas a diario: $10$ céntimos $= 0.10$; dos
euros con cincuenta $= 2.50$; y $0.05$ es una moneda de cinco céntimos, ¡no de cincuenta!
\end{example}

\section{Comparar con las décimas}

\begin{method}[Comparar decimales (primer contacto)]\label{met:g4:decimals:compare}
\begin{enumerate}
  \item compara primero las partes enteras: $3.2 > 2.9$;
  \item con la misma parte entera: compara las décimas y luego las centésimas:
        $2.37 < 2.5$ porque $3$ décimas $< 5$ décimas;
  \item escribir los dos con dos cifras decimales ayuda: $2.5 = 2.50$ y
        $37 < 50$ centésimas.
\end{enumerate}
La trampa: $2.37$ tiene más cifras que $2.5$, pero es
\emph{menor} --- ¡más cifras no significa mayor!
\end{method}

\begin{example}\label{ex:g4:decimals:compare}
Ordena $0.4$, $0.35$ y $0.09$: con centésimas, $0.40$, $0.35$,
$0.09$; por tanto,
\[
0.09 < 0.35 < 0.4 .
\]
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:decimals:1}
En un cuadrado de cien, ¿cuánto queda sombreado si se sombrean: $5$ columnas;
$2$ columnas y $3$ casillas; $45$ casillas? Responde con fracciones.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:2}
Escribe con \omterm{def:g4:decimals:point}{punto decimal}:
$\frac{9}{10}$; \quad $\frac{27}{100}$; \quad $\frac{3}{100}$; \quad
$5 + \frac{6}{10}$; \quad $\frac{60}{100}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:3}
Escribe como \omterm{def:g4:fractions:def}{fracción} (décimas o centésimas):
$0.3$; \quad $0.51$; \quad $0.07$; \quad $1.9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:4}
En $6.58$: ¿qué cuenta la cifra $5$? ¿Y la cifra $8$? Descompón
el número como en la \cref{def:g4:decimals:point}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:5}
Sitúa en una recta numérica graduada en décimas del $4$ al $5$:
$4.2$; \ $4.75$ (¿entre qué dos marcas?); \ $4.9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:6}
Copia y completa con $<$, $>$ o $=$:
\[
0.6 \;?\; 0.60, \qquad
2.8 \;?\; 2.75, \qquad
0.09 \;?\; 0.1, \qquad
5.3 \;?\; 5.13 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:7}
Ordena de menor a mayor: $1.05$; \ $1.5$; \ $0.95$; \ $1.45$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:8}
Escribe en escritura decimal: tres euros y cinco céntimos; doce euros
y cincuenta céntimos; noventa y nueve céntimos. Después ordena los tres precios.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:9}
¿Qué números decimales con una sola cifra decimal están estrictamente entre
$3$ y $4$? Enuméralos todos. ¿Cuántos son?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:decimals:10}
Tomás dice: «$0.25$ es mayor que $0.5$ porque $25$ es mayor que
$5$». Enséñale su error con el cuadrado de cien (¿cuántas casillas
sombrea cada uno?).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:decimals:11}
Los tres saltos de un atleta midieron $3.6$ m, $3.58$ m y $3.65$ m.
Ordena los saltos de menor a mayor. ¿Qué dos se diferencian en
exactamente $\frac{5}{100}$ de metro?
\end{exercise}
