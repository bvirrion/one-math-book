\section{Plegado y combinación}

\begin{definition}[Eje de simetría]\label{def:g4:symmetry:def}
Una línea es una \emph{eje de simetria}\index{eje de simetria} de un
figura al doblar la figura a lo largo de esa línea se forman las dos mitades
partido \emph{exactamente}. Una figura puede no tener eje, tener un solo eje o
varios.
\end{definition}

\section{Completar figuras en una cuadrícula}

\begin{method}[Simétrica de una figura a través de una línea de cuadrícula]\label{met:g4:symmetry:grid}
El eje es una línea vertical (u horizontal) de la cuadrícula.
\begin{enumerate}
  \item Toma cada vértice de la figura a su vez;
  \item cuente su distancia al eje en \omterm{def:g3:shapes:shapes}{cuadrados};
  \item colocar la imagen vértice al \emph{mismo} distancia en el
        \emph{otro} lado, en la misma fila (o columna);
  \item únete a la imagen \omterm{def:g4:geometry:polygon}{vértices} en el mismo orden y verifique: el
        La imagen es el espejo gemelo de la figura.
\end{enumerate}
\end{method}

La imagen es una F al revés, no otra F.
\end{example}

\section{Ejes de las formas habituales.}

\begin{example}[contando ejes]\label{ex:g4:symmetry:shapes}
Al doblar:
\begin{itemize}
  \item un cuadrado: $4$ ejes (dos por el centro de ejes opuestos)
        lados, dos a lo largo de las diagonales);
  \item un rectángulo: $2$ ejes (¡solo los centros de los lados opuestos!);
  \item un triángulo isósceles: $1$; un triángulo equilátero: $3$;
  \item un círculo: cada línea que pasa por su centro --- más ejes que
        uno puede contar.
\end{itemize}
\end{example}

\begin{example}[Simetría en el alfabeto]\label{ex:g4:symmetry:letters}
A, M, T, U, V tienen un eje vertical; B, C, D, E tienen una dirección horizontal.
uno; H, I, O, X tienen ambos. Las palabras formadas con las letras correctas pueden ser
simétrico también: MUM tiene un eje vertical; OXO funciona en ambos sentidos.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g4:symmetry:1}
Trazar y doblar: ¿cuáles de estos tienen un eje de simetría --- un
cuadrado; un rectángulo (no cuadrado); un triángulo escaleno (todos los lados
diferente); un \omterm{def:g3:shapes:shapes}{circulo}?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:symmetry:2}
Dibuja los ejes de simetría de: un cuadrado; un rectángulo; un equilátero
triángulo. ¿Cuantos para cada uno?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:symmetry:3}
Utilice la prueba del papel de calco de \cref{ex:g4:symmetry:trace} mostrar
que la diagonal de un $6 \times 4$ el rectángulo es \emph{no} un eje de simetría.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:symmetry:4}
En papel cuadriculado, dibuja un eje vertical y la letra L (tres \omterm{def:g3:shapes:shapes}{cuadrados}
alto, dos de ancho) en $2$ \omterm{def:g3:shapes:shapes}{cuadrados} del eje. Construye su espejo
imagen.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:symmetry:5}
En papel cuadriculado, dibuja un eje horizontal y una pequeña bandera (una $1 \times
1$ triángulo en un $3$-mástil cuadrado) encima de él. Completa la figura con
simetría debajo del eje.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:symmetry:6}
Una figura toca su eje de simetría en un punto $ P $. donde esta el
imagen de $ P $? Explica con el dibujo plegable.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:symmetry:7}
Ordenar las letras mayúsculas A, B, E, F, H, N, O, T: ¿eje vertical?
eje horizontal? ¿ambos? ¿ni?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:symmetry:8}
Un \omterm{def:g3:shapes:shapes}{triángulo} tiene lados $5$ centímetro, $5$ centímetros y $3$ centímetro. ¿Tiene un
eje de simetría? ¿Qué lados intercambia el pliegue?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g4:symmetry:9}
En papel cuadriculado, dibuja una figura propia que tenga \emph{exactamente dos}
ejes de simetría y dibuja ambos ejes. (Pista: piensa en un rectángulo,
o de una cruz.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g4:symmetry:10}
\omterm{def:g3:division:half}{Medio} de un dibujo simétrico se da: la izquierda \omterm{def:g3:division:half}{medio} de una casa
(una pared cuadrada y \omterm{def:g3:division:half}{medio} un techo triangular), con un eje vertical
a lo largo de su borde derecho. Describe o dibuja la casa completa. ¿Cómo
las longitudes del derecho completado \omterm{def:g3:division:half}{medio} comparar con la izquierda \omterm{def:g3:division:half}{medio}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:symmetry:11}
Aya afirma: "Todo triángulo con dos lados iguales tiene un eje de simetría, y cada triángulo con un eje de simetría tiene dos iguales
lados.'' Pruebe su doble afirmación en dibujos: un triángulo isósceles,
uno equilátero, uno escaleno. ¿El plegado la sostiene?
\end{exercise}
