\chapter{Los números grandes}\label{ch:g4:numbers}

Los millares eran el techo del \cref{ch:g3:numbers}; ahora el techo
sube hasta los \emph{millones}. El truco que hace legibles los números enormes
es partir sus cifras en grupos de tres --- las \omterm{def:g4:numbers:classes}{clases}.

\section{Clases de tres cifras}

\begin{definition}[Clases]\label{def:g4:numbers:classes}
Empezando por la derecha, las cifras de un número se agrupan en
\emph{clases} de tres: la clase de las \emph{unidades}, la clase de los
\emph{millares} y la clase de los \emph{millones}. Cada clase tiene sus
propias unidades, decenas y centenas. Por ejemplo,
\[
6\,309\,452 = 6 \text{ millones} + 309 \text{ millares}
+ 452 \text{ unidades},
\]
que se lee «seis millones trescientos nueve mil cuatrocientos
cincuenta y dos».
\end{definition}

\begin{example}[Escribir con las clases]\label{ex:g4:numbers:write}
«Doce millones cuarenta mil quinientos» lleva $12$ en la
clase de los millones, $040$ en la de los millares y $500$ en la de las
unidades:
\[
12\,040\,500 .
\]
Los \omterm{def:g1:counting:zero}{ceros} del $040$ son imprescindibles: cada clase (salvo la de más a la izquierda)
debe mostrar sus tres cifras.
\end{example}

\begin{example}[¿Cuántos millares?]\label{ex:g4:numbers:contained}
El número $6\,309\,452$ \emph{contiene} $6\,309$ millares
(todo lo que hay a la izquierda de la clase de las unidades): $6\,309\,452 = 6\,309 \times
1\,000 + 452$. Preguntar «cuántos millares» no es preguntar por una
cifra, sino por todo un principio del número.
\end{example}

\section{Comparar y redondear}

\begin{method}[Comparar números grandes]\label{met:g4:numbers:compare}
Exactamente como en el \cref{met:g3:numbers:compare}, con las \omterm{def:g4:numbers:classes}{clases} como
atajo: gana quien tiene más cifras; con la misma cantidad, compara desde la izquierda ---
clase a clase y luego cifra a cifra.
\end{method}

\begin{example}\label{ex:g4:numbers:compare}
$2\,145\,890$ y $2\,148\,540$: millones iguales ($2$) y, en la clase de los millares,
$145 < 148$, así que
\[
2\,145\,890 < 2\,148\,540 .
\]
\end{example}

\begin{definition}[Redondear]\label{def:g4:numbers:round}
\emph{Redondear}\index{redondeo} un número al millar más cercano
(o a la decena, la centena, el millón\,\dots) es sustituirlo por el múltiplo
de $1\,000$ más próximo. Si el número queda justo a \omterm{def:g3:division:half}{mitad} de camino, se redondea hacia arriba. En
la práctica: mira la cifra situada justo por debajo del lugar de redondeo --- $5$ o
más sube, $4$ o menos baja.
\end{definition}

\begin{example}\label{ex:g4:numbers:round}
$37\,614$ redondeado al millar más cercano: la cifra de las centenas es
$6$, así que redondeamos hacia arriba: $38\,000$. Redondeado a la centena más cercana:
$37\,600$. Redondeado a la decena de millar más cercana: $40\,000$.
\end{example}

\begin{example}[Órdenes de magnitud en el mundo]\label{ex:g4:numbers:world}
Así es como se usan de verdad los números grandes: un país de
$67\,842\,591$ habitantes tiene «unos $68$ millones»; una ciudad de
$214\,376$ personas tiene «unos $200\,000$». De todos modos, la cifra exacta cambia
cada día --- la redondeada es la que lleva la información
útil.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:numbers:1}
Escribe con cifras, separando las \omterm{def:g4:numbers:classes}{clases}: «tres millones quinientos
veinte mil»; «siete millones ocho»; «doscientos
cuarenta mil sesenta».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:2}
Escribe con letras: $5\,204\,300$; \quad $10\,050\,000$; \quad
$999\,999$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:3}
En $8\,472\,156$: ¿qué cifra ocupa las decenas de millar? ¿Y las centenas
de la clase de los millones? ¿Cuántos \emph{millares} contiene el número
(véase el \cref{ex:g4:numbers:contained})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:4}
Copia y completa con $<$ o $>$:
\[
980\,000 \;?\; 1\,020\,000, \qquad
3\,456\,000 \;?\; 3\,465\,000, \qquad
7\,090\,050 \;?\; 7\,090\,500 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:5}
Ordena de menor a mayor: $2\,300\,000$; \ $980\,500$; \
$2\,030\,000$; \ $2\,299\,999$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:6}
Redondea $63\,782$: a la decena más cercana; a la centena; al
millar; a la decena de millar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:7}
Redondea al millón más cercano: $4\,721\,000$; \quad $6\,499\,999$;
\quad $9\,500\,000$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:8}
Cuenta a pasos: tres pasos de $1\,000$ desde $998\,500$; dos pasos
de $100\,000$ desde $950\,000$. ¿Qué les ocurre a las \omterm{def:g4:numbers:classes}{clases} al
cruzar el millón?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:numbers:9}
Un estadio tiene $81\,338$ localidades. Un periódico escribe «más de
$80\,000$ aficionados»; una \omterm{def:g3:shapes:shapes}{radio} dice «unos $81\,000$». ¿Tienen razón
los dos? ¿A qué lugar ha redondeado cada uno?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:numbers:10}
Usando cada una de las cifras del $1$ al $7$ exactamente una vez, escribe el mayor
número posible de siete cifras, el menor y el más cercano a
$4\,000\,000$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:numbers:11}
¿Cuál es el mayor número entero que redondea a $50\,000$ al
redondear al millar más cercano? ¿Y el menor?
\end{exercise}
