\chapter{Rectas y polígonos}\label{ch:g4:geometry}

Con una regla, una escuadra y un compás, la geometría se convierte en un oficio.
Este capítulo enseña las dos posiciones especiales de las líneas ---
\omterm{def:g4:geometry:def}{perpendiculares} y paralelas --- a nivel de herramienta, y hace un recorrido por
el \omterm{def:g4:geometry:polygon}{polígono} familia. (La teoría de estas nociones entra en juego
\cref{ch:g6:lines}.)

\section{Perpendicular y paralela}

\begin{definition}[Perpendicular, paralela]\label{def:g4:geometry:def}
Dos líneas son \emph{perpendicular}\index{perpendicular} cuando
they cross at a \omterm{def:g3:shapes:rightangle}{ángulo recto} (comprobado con la escuadra, marcada con
un cuadrito). Dos líneas son \emph{paralelo}\index{paralelo}
cuando corren en la misma dirección y nunca se cruzan --- como los dos
carriles de una vía, siempre a la misma distancia entre sí.
\end{definition}

\begin{method}[Dibujar con la escuadra]\label{met:g4:geometry:draw}
\emph{\omterm{def:g4:geometry:def}{perpendiculares} a una línea $d $ a través de un punto $ P $}: poner un borde
de la \omterm{def:g3:shapes:rightangle}{ángulo recto} a lo largo de $ d $, deslice hasta que el otro borde llegue $ P $,
dibujar.

\emph{paralelas a $ d $ a través de $ P $}: coloque el borde de la escuadra a lo largo
$ d $, presione la regla contra otro borde a modo de riel, deslice el conjunto
cuadrado a lo largo de la regla hasta $ P $,
dibujar.
\end{method}

\begin{example}[Detectándolos en cifras]\label{ex:g4:geometry:spot}
En un rectángulo, cada lado es \omterm{def:g4:geometry:def}{perpendiculares} a sus dos vecinos y
paralelas al lado opuesto. El papel cuadriculado es una red ya preparada de
paralelas y \omterm{def:g4:geometry:def}{perpendiculares} --- eso es lo que lo hace así
cómodo para dibujar.
\end{example}

\section{Polígonos}

\begin{definition}[Polígono]\label{def:g4:geometry:polygon}
A \emph{polígono}\index{polígono} es una figura cerrada cuyo borde es
hecho de segmentos rectos, su \emph{lados}; los puntos de las esquinas son
el \emph{vértices}\index{vértices}. Los polígonos se nombran por su número de lados:
\emph{triángulo} ($3$), \emph{cuadrilátero}\index{cuadrilátero} ($4$),
\emph{pentágono}\index{pentágono} ($5$), \emph{hexágono}\index{hexágono} ($6$),
\emph{octágono}\index{octágono}
($8$).
\end{definition}

\begin{example}[Describiendo un polígono]\label{ex:g4:geometry:describe}
Un juego útil: describe una figura con tanta precisión que un compañero pueda
Vuelve a dibujarlo sin verlo. ``Un \omterm{def:g4:geometry:polygon}{cuadrilátero} $ ABCD $ con
$ AB = 5$ cm, el ángulo en $ A $ bien, $ d.\,C. = 3$ centímetro\,\dots'' ---
El vocabulario es lo que hace que el mensaje funcione.
\end{example}

\section{El círculo y el compás}

\begin{definition}[Círculo]\label{def:g4:geometry:circle}
El \emph{círculo} de centro $ O $ y \omterm{def:g3:shapes:shapes}{radio} $3$ cm es el conjunto de todos
apunta exactamente $3$ centímetros de $ O $. La brújula lo dibuja: punta en
$ O $, apertura $3$ cm, girar. A \emph{diametro}\index{diametro} es un segmento a través de
el centro que une dos puntos del círculo; mide el doble
\omterm{def:g3:shapes:shapes}{radio}.
\end{definition}

\begin{example}[La brújula como indicador de distancia]\label{ex:g4:geometry:compass}
La brújula hace más que dibujar círculos: \emph{transporta
distancias}. Para marcar todos los puntos en $4$ cm desde un punto $ A $, dibuja el
círculo de centro $ A $, \omterm{def:g3:shapes:shapes}{radio} $4$ centímetro; comparar dos segmentos sin
una regla, abre el compás de uno y pruébalo en el otro.
\end{example}

\begin{method}[Reproducir una figura en una cuadrícula.]\label{met:g4:geometry:reproduce}
\begin{enumerate}
  \item Elige un inicio vértice y su posición en la cuadrícula en blanco;
  \item viajar a lo largo de la figura vértice por vértice, contando la cuadrícula
        pasos (``$3$ bien, $2$ arriba'');
  \item únete al \omterm{def:g4:geometry:polygon}{vértices} en orden y comparar con el modelo:
        mismas longitudes, mismas direcciones.
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g4:geometry:1}
dibuja una linea $ d $ y un punto $ P $ no en eso. Construir el
\omterm{def:g4:geometry:def}{perpendiculares} a $ d $ a través de $ P $, entonces el paralelas a $ d $ a través de
$ P $ (con el método regla-carril). Marque el \omterm{def:g3:shapes:rightangle}{ángulo recto}.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:geometry:2}
Observa las letras mayúsculas E, H, N, T, Z. En cada una, encuentra un \omterm{def:g3:numbers:evenodd}{par} de
paralelas trazos y un \omterm{def:g3:numbers:evenodd}{par} de \omterm{def:g4:geometry:def}{perpendiculares} golpes cuando ellos
existir.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:geometry:3}
Nombra cada uno \omterm{def:g4:geometry:polygon}{polígono}: $6$ lados; $4$ lados; $3$ lados; $8$ lados;
$5$ lados. Cuántos \emph{\omterm{def:g4:geometry:polygon}{vértices}} tiene cada uno?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:geometry:4}
Dibuja cualquier \omterm{def:g4:geometry:polygon}{pentágono} (no necesariamente regular) y etiquetar su \omterm{def:g4:geometry:polygon}{vértices}
$ A $, $ B $, $ C $, $ D $, $ E $. ¿Cuantos lados tiene? Dibuja su
diagonales de $ A $ (segmentos que se unen $ A $ a no vecino \omterm{def:g4:geometry:polygon}{vértices}):
cuantos hay?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:geometry:5}
Dibuja un círculo de centro. $ O $ y \omterm{def:g3:shapes:shapes}{radio} $4$ centímetro; a diámetro $[AB]$;
y un punto $ M $ del círculo no en $[AB]$. Dar las longitudes $ OA $,
$ OM $ y $ AB $ sin medir.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:geometry:6}
Con la brújula como distanciador
(\cref{ex:g4:geometry:compass}), compara los lados de un triángulo
dibujado por un compañero y decide si tiene dos lados iguales.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:geometry:7}
En papel cuadriculado, reproduzca este camino y ciérrelo en un \omterm{def:g4:geometry:polygon}{polígono}:
comenzar en un punto $ A $; ir $4$ bien; $2$ arriba; $3$ izquierda; $2$ arriba;
luego de vuelta a $ A $ en línea recta. ¿Cuál es el nombre del \omterm{def:g4:geometry:polygon}{polígono}
obtienes (cuenta sus lados)?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g4:geometry:8}
¿Verdadero o falso? Explica con un dibujo.
``Dos lineas \omterm{def:g4:geometry:def}{perpendiculares} siempre cruzado."
``Dos lineas paralelas Nunca cruces."
``Una línea puede ser paralelas a una línea y \omterm{def:g4:geometry:def}{perpendiculares} a
otro."
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g4:geometry:9}
dibuja una linea $ d $, luego una línea $ e $ \omterm{def:g4:geometry:def}{perpendiculares} a $ d $, luego una línea
$ f $ \omterm{def:g4:geometry:def}{perpendiculares} a $ e $. Mira a $ d $ y $ f $: ¿Qué notas?
(Esta observación se convierte en un teorema en \cref{ch:g6:lines}.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g4:geometry:10}
Describe la siguiente figura con palabras para un compañero de clase, sin mostrar
eso: un cuadrado de lado $4$ cm, con un círculo de \omterm{def:g3:shapes:shapes}{radio} $2$ cm centrado
en el \omterm{def:g3:division:half}{medio} del cuadrado (tocando los cuatro lados). Escribe el
Descripción como instrucciones numeradas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:11}
¿Cuántas diagonales tiene un \omterm{def:g4:geometry:polygon}{cuadrilátero} ¿tener? A \omterm{def:g4:geometry:polygon}{pentágono}? A \omterm{def:g4:geometry:polygon}{hexágono}?
Dibuja los tres casos, cuenta con cuidado y describe el patrón que
ver.
\end{exercise}
