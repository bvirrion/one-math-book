\chapter{Rectas y polígonos}\label{ch:g4:geometry}

Con una regla, una escuadra y un compás, la geometría se convierte en un oficio.
Este capítulo enseña las dos posiciones especiales de las rectas ---
\omterm{def:g4:geometry:def}{perpendiculares} y \omterm{def:g4:geometry:def}{paralelas} --- desde las herramientas, y recorre
la familia de los \omterm{def:g4:geometry:polygon}{polígonos}. (La teoría de estas nociones llega en el
\cref{ch:g6:lines}.)

\section{Perpendiculares y paralelas}

\begin{definition}[Perpendiculares, paralelas]\label{def:g4:geometry:def}
Dos rectas son \emph{perpendiculares}\index{rectas perpendiculares} cuando
se cortan formando un \omterm{def:g3:shapes:rightangle}{ángulo recto} (se comprueba con la escuadra y se marca con
un cuadradito). Dos rectas son \emph{paralelas}\index{rectas paralelas}
cuando llevan la misma dirección y no se cortan nunca --- como los dos
raíles de una vía, siempre a la misma distancia una de otra.
\end{definition}

\begin{method}[Dibujar con la escuadra]\label{met:g4:geometry:draw}
\emph{Perpendicular a una recta $d$ que pase por un punto $P$}: apoya un borde
del \omterm{def:g3:shapes:rightangle}{ángulo recto} sobre $d$, desliza hasta que el otro borde alcance $P$ y
traza.

\emph{Paralela a $d$ que pase por $P$}: apoya un borde de la escuadra sobre
$d$, aprieta la regla contra otro borde a modo de raíl, desliza la escuadra
a lo largo de la regla hasta $P$ y traza.
\end{method}

\begin{example}[Reconocerlas en las figuras]\label{ex:g4:geometry:spot}
En un rectángulo, cada lado es perpendicular a sus dos vecinos y
paralelo al lado opuesto. El papel cuadriculado es una red ya hecha de
\omterm{def:g4:geometry:def}{paralelas} y \omterm{def:g4:geometry:def}{perpendiculares} --- por eso resulta tan
cómodo dibujar en él.
\end{example}

\section{Los polígonos}

\begin{definition}[Polígono]\label{def:g4:geometry:polygon}
Un \emph{polígono}\index{polígono} es una figura cerrada cuyo borde está
hecho de segmentos rectos, sus \emph{lados}; los puntos de las esquinas son
los \emph{vértices}\index{vértice}. Los polígonos se nombran por su número de lados:
\emph{triángulo} ($3$), \emph{cuadrilátero}\index{cuadrilátero} ($4$),
\emph{pentágono}\index{pentágono} ($5$), \emph{hexágono}\index{hexágono} ($6$),
\emph{octógono}\index{octógono}
($8$).
\end{definition}

\begin{example}[Describir un polígono]\label{ex:g4:geometry:describe}
Un juego muy útil: describe una figura con tanta precisión que un compañero pueda
volver a dibujarla sin verla. «Un \omterm{def:g4:geometry:polygon}{cuadrilátero} $ABCD$ con
$AB = 5$ cm, el ángulo en $A$ recto, $AD = 3$ cm\,\dots» ---
el vocabulario es lo que hace que el mensaje funcione.
\end{example}

\section{La circunferencia y el compás}

\begin{definition}[Circunferencia]\label{def:g4:geometry:circle}
La \emph{circunferencia} de centro $O$ y \omterm{def:g3:shapes:shapes}{radio} $3$ cm es el conjunto de todos
los puntos que están exactamente a $3$ cm de $O$. El compás la traza: punta en
$O$, abertura $3$ cm y giro. Un \emph{diámetro}\index{diámetro} es un segmento que pasa por
el centro y une dos puntos de la circunferencia; mide el doble que el
\omterm{def:g3:shapes:shapes}{radio}.
\end{definition}

\begin{example}[El compás como guardián de distancias]\label{ex:g4:geometry:compass}
El compás hace algo más que trazar circunferencias: \emph{transporta
distancias}. Para marcar todos los puntos que están a $4$ cm de un punto $A$, traza la
circunferencia de centro $A$ y \omterm{def:g3:shapes:shapes}{radio} $4$ cm; para comparar dos segmentos sin
regla, abre el compás sobre uno y pruébalo en el otro.
\end{example}

\begin{method}[Reproducir una figura en la cuadrícula]\label{met:g4:geometry:reproduce}
\begin{enumerate}
  \item elige un \omterm{def:g4:geometry:polygon}{vértice} de partida y su posición en la cuadrícula vacía;
  \item recorre la figura de \omterm{def:g4:geometry:polygon}{vértice} en \omterm{def:g4:geometry:polygon}{vértice} contando los pasos
        de la cuadrícula («$3$ a la derecha, $2$ arriba»);
  \item une los \omterm{def:g4:geometry:polygon}{vértices} en orden y compara con el modelo:
        mismas longitudes, mismas direcciones.
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:geometry:1}
Traza una recta $d$ y un punto $P$ que no esté en ella. Construye la
perpendicular a $d$ por $P$ y después la paralela a $d$ por
$P$ (con el método de la regla-raíl). Marca el \omterm{def:g3:shapes:rightangle}{ángulo recto}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:2}
Fíjate en las letras mayúsculas E, H, N, T y Z. En cada una, busca un \omterm{def:g3:numbers:evenodd}{par} de
trazos paralelos y un \omterm{def:g3:numbers:evenodd}{par} de trazos \omterm{def:g4:geometry:def}{perpendiculares} cuando los
haya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:3}
Nombra cada \omterm{def:g4:geometry:polygon}{polígono}: $6$ lados; $4$ lados; $3$ lados; $8$ lados;
$5$ lados. ¿Cuántos \emph{\omterm{def:g4:geometry:polygon}{vértices}} tiene cada uno?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:4}
Dibuja un \omterm{def:g4:geometry:polygon}{pentágono} cualquiera (no necesariamente regular) y nombra sus \omterm{def:g4:geometry:polygon}{vértices}
$A$, $B$, $C$, $D$ y $E$. ¿Cuántos lados tiene? Traza sus
diagonales desde $A$ (segmentos que unen $A$ con los \omterm{def:g4:geometry:polygon}{vértices} no vecinos):
¿cuántas hay?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:5}
Traza una circunferencia de centro $O$ y \omterm{def:g3:shapes:shapes}{radio} $4$ cm; un \omterm{def:g4:geometry:circle}{diámetro} $[AB]$;
y un punto $M$ de la circunferencia que no esté en $[AB]$. Di las longitudes $OA$,
$OM$ y $AB$ sin medir.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:6}
Con el compás como guardián de distancias
(\cref{ex:g4:geometry:compass}), compara los lados de un triángulo
dibujado por un compañero y decide si tiene dos lados iguales.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:7}
En papel cuadriculado, reproduce este recorrido y ciérralo en un \omterm{def:g4:geometry:polygon}{polígono}:
sal de un punto $A$; $4$ a la derecha; $2$ arriba; $3$ a la izquierda; $2$ arriba;
y vuelve a $A$ en línea recta. ¿Cómo se llama el \omterm{def:g4:geometry:polygon}{polígono}
que obtienes (cuenta sus lados)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:geometry:8}
¿Verdadero o falso? Explícalo con un dibujo.
«Dos \omterm{def:g4:geometry:def}{rectas perpendiculares} siempre se cortan.»
«Dos \omterm{def:g4:geometry:def}{rectas paralelas} no se cortan nunca.»
«Una recta puede ser paralela a una recta y perpendicular a
otra.»
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:9}
Traza una recta $d$, después una recta $e$ perpendicular a $d$ y después una recta
$f$ perpendicular a $e$. Mira $d$ y $f$: ¿qué observas?
(Esta observación se convierte en un teorema en el \cref{ch:g6:lines}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:10}
Describe con palabras la figura siguiente a un compañero, sin enseñársela:
un cuadrado de lado $4$ cm con una circunferencia de \omterm{def:g3:shapes:shapes}{radio} $2$ cm centrada
en el centro del cuadrado (tangente a los cuatro lados). Escribe la
descripción como instrucciones numeradas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:geometry:11}
¿Cuántas diagonales tiene un \omterm{def:g4:geometry:polygon}{cuadrilátero}? ¿Y un \omterm{def:g4:geometry:polygon}{pentágono}? ¿Y un \omterm{def:g4:geometry:polygon}{hexágono}?
Dibuja los tres casos, cuenta con cuidado y describe el patrón que
observas.
\end{exercise}
