\chapter{Multiplicar números mayores}\label{ch:g4:mult}

El año pasado multiplicamos por una cifra (\cref{ch:g3:mult}); este capítulo
multiplica por números de dos cifras --- y explica la misteriosa
«línea desplazada» del método en columna, que no es más que una
descomposición en decenas y unidades.

\section{Multiplicar por decenas}

\begin{proposition}[Por 10, por 20, por 300\,\dots]\label{prop:g4:mult:tens}
Multiplicar por $10$ añade un \omterm{def:g1:counting:zero}{cero} y por $100$, dos \omterm{def:g1:counting:zero}{ceros}. Para multiplicar
por $20$, multiplica por $2$ y después por $10$:
\[
36 \times 20 = 36 \times 2 \times 10 = 72 \times 10 = 720 .
\]
Del mismo modo, $14 \times 300 = 14 \times 3 \times 100 = 4\,200$.
\end{proposition}
\admitted

\section{El método en columna con dos cifras}

\begin{example}[¿Por qué la línea desplazada?]\label{ex:g4:mult:why}
Para calcular $23 \times 12$, descompón $12$ en $10 + 2$:
\[
23 \times 12 = 23 \times 2 + 23 \times 10 = 46 + 230 = 276 .
\]
El método en columna escribe exactamente estos dos \omterm{def:g2:mult:def}{productos}, uno debajo del
otro --- el segundo desplazado a la izquierda porque es un número de
\emph{decenas}:
\[
\begin{array}{r}
2\,3 \\
\times\ 1\,2 \\
\hline
4\,6 \\
2\,3\,\cdot \\
\hline
2\,7\,6
\end{array}
\]
(el punto marca el lugar de las unidades de la línea desplazada, que a menudo se deja
en blanco).
\end{example}

\begin{method}[Multiplicar en columna por un número de dos cifras]\label{met:g4:mult:column}
Para calcular $457 \times 36$:
\begin{enumerate}
  \item primera línea: $457 \times 6 = 2\,742$;
  \item segunda línea: $457 \times 3$ \emph{decenas} $= 1\,371$ decenas ---
        escribe $1\,371$ desplazado un lugar a la izquierda;
  \item \omterm{def:g1:addition:def}{suma} las dos líneas: $2\,742 + 13\,710 = 16\,452$;
  \item estima para comprobar: $457 \times 36 \approx 500 \times 36
        = 18\,000$? Better: $450 \times 36$ is about $16\,000$ ---
        verosímil.
\end{enumerate}
\[
\begin{array}{r}
4\,5\,7 \\
\times\ \ \,3\,6 \\
\hline
2\,7\,4\,2 \\
1\,3\,7\,1\,\cdot \\
\hline
1\,6\,4\,5\,2
\end{array}
\]
\end{method}

\begin{example}[Trucos de cálculo mental]\label{ex:g4:mult:tricks}
\begin{itemize}
  \item \emph{Descomponer de forma amable}: $18 \times 11 = 18 \times 10
        + 18 = 198$.
  \item \emph{Usar un número cercano a cien}: $25 \times 99 = 25 \times 100 -
        25 = 2\,475$.
  \item \emph{Doblar y partir por la \omterm{def:g3:division:half}{mitad}}: $16 \times 35 = 8 \times 70 =
        560$ (partir un factor por la \omterm{def:g3:division:half}{mitad} y doblar el otro conserva el
        \omterm{def:g2:mult:def}{producto}).
\end{itemize}
\end{example}

\section{La multiplicación en los problemas}

\begin{example}\label{ex:g4:mult:problem}
Un colegio encarga $28$ cajas de $145$ hojas de papel.
\begin{enumerate}
  \item operación: $145 \times 28$;
  \item estimación: $150 \times 30 = 4\,500$, un poco por encima en los dos
        factores;
  \item columnas: $145 \times 8 = 1\,160$; $145 \times 2$ decenas
        $= 2\,900$; total, $4\,060$;
  \item frase: el colegio recibe $4\,060$ hojas.
\end{enumerate}
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g4:mult:1}
Calcula de cabeza: $47 \times 10$; \quad $23 \times 100$; \quad
$36 \times 20$; \quad $15 \times 300$; \quad $42 \times 50$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:2}
Calcula $34 \times 21$ descomponiendo como en el \cref{ex:g4:mult:why}
($34 \times 21 = 34 \times 20 + 34$) y comprueba después en columna.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:3}
Calcula en columna: $63 \times 24$; \quad $85 \times 47$; \quad
$126 \times 53$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:4}
Calcula en columna, con una estimación previa: $208 \times 34$; \quad
$517 \times 62$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:5}
Calcula con un truco del \cref{ex:g4:mult:tricks}:
$45 \times 11$; \quad $32 \times 99$; \quad $24 \times 45$ (dobla
y parte por la \omterm{def:g3:division:half}{mitad} --- dos veces si quieres).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:6}
Un cine tiene $26$ filas de $18$ butacas. ¿Cuántas butacas hay en total?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:7}
Un camión lleva $32$ palés de $48$ cajas. ¿Cuántas cajas son? Da
primero una estimación y luego la respuesta exacta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:8}
Busca el error: Ana calcula $57 \times 23$ así:
$57 \times 3 = 171$ y $57 \times 2 = 114$, y después \omterm{def:g1:addition:def}{suma}
$171 + 114 = 285$. La respuesta correcta es $1\,311$. ¿Qué ha
olvidado?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:mult:9}
Un jardinero planta $14$ filas de $35$ tulipanes y $12$ filas de $28$
narcisos. ¿Cuántas flores hay en total? (Dos \omterm{def:g2:mult:def}{productos} y luego una \omterm{def:g1:addition:def}{suma}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:mult:10}
Cada día, una panadería gasta $67$ kg de harina. ¿Cuánta harina es
eso en un mes de $30$ días? Estima con números redondeados y después
calcula con exactitud.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:mult:11}
Explica, con un dibujo rectangular como el de este capítulo, por qué
$15 \times 12$ se puede calcular como $15 \times 10 + 15 \times 2$.
Después inventa otra descomposición de $15 \times 12$ que también funcione.
\end{exercise}
