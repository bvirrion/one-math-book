\chapter{Décimos e centésimos}\label{ch:g4:decimals}

Entre $3$ e $4$ os números inteiros ficam calados --- mas um corredor
pode estar $3$ metros e um pouco na frente. Cortar a unidade em dez
e depois em cem partes dá as \emph{frações decimais}, e uma escrita
abreviada formidável para elas: o \omterm{def:g4:decimals:point}{ponto decimal}. (Os números decimais
ganham o seu capítulo inteiro no \cref{ch:g5:decimals}.)

\section{Frações decimais}

\begin{definition}[Décimos e centésimos]\label{def:g4:decimals:def}
Cortar uma unidade em $10$ partes iguais dá os \emph{décimos},
$\frac{1}{10}$; cortar cada décimo de novo em $10$ dá os
\emph{centésimos}, $\frac{1}{100}$. Assim,
\[
1 = \frac{10}{10} = \frac{100}{100},
\qquad
\frac{1}{10} = \frac{10}{100}.
\]
\end{definition}

\begin{example}\label{ex:g4:decimals:count}
Pintar $3$ colunas e mais $7$ casinhas do quadrado de cem pinta
\[
\frac{3}{10} + \frac{7}{100} = \frac{30}{100} + \frac{7}{100}
= \frac{37}{100}
\]
do quadrado --- trinta e sete centésimos.
\end{example}

\section{O ponto decimal}

\begin{definition}[Escrita decimal]\label{def:g4:decimals:point}
O número $2 + \frac{3}{10} + \frac{7}{100}$ escreve-se
\[
2.37
\]
--- o \emph{ponto decimal}\index{ponto decimal} separa a parte inteira ($2$) da parte
decimal; o primeiro algarismo depois do ponto conta os décimos, e o
segundo, os centésimos. Leitura: ``dois ponto três sete'', ou ``dois
inteiros e trinta e sete centésimos''.
\end{definition}

\begin{example}\label{ex:g4:decimals:convert}
\[
\frac{7}{10} = 0.7, \qquad
\frac{43}{100} = 0.43, \qquad
\frac{5}{100} = 0.05, \qquad
3 + \frac{4}{10} = 3.4 .
\]
Cuidado com $\frac{5}{100}$: cinco \emph{centésimos} precisam de um
$0$ na casa dos décimos --- $0.05$, e não $0.5$.
\end{example}

\begin{example}[Dinheiro é centésimo]\label{ex:g4:decimals:money}
Um centavo é um centésimo de um real (ou dólar, ou euro): um preço
de $4.85$ quer dizer $4$ moedas inteiras e $85$ centésimos. O
dinheiro é o \omterm{def:g4:decimals:point}{ponto decimal} que você já usa todo dia: $10$ centavos
$= 0.10$; dois reais e cinquenta $= 2.50$; e $0.05$ é uma moeda de
cinco centavos, não de cinquenta!
\end{example}

\section{Comparar com os décimos}

\begin{method}[Comparar decimais (primeiro contato)]\label{met:g4:decimals:compare}
\begin{enumerate}
  \item Compare primeiro as partes inteiras: $3.2 > 2.9$.
  \item Partes inteiras iguais: compare os décimos e depois os
        centésimos: $2.37 < 2.5$ porque $3$ décimos $< 5$ décimos.
  \item Escrever os dois com duas casas decimais ajuda:
        $2.5 = 2.50$, e $37 < 50$ centésimos.
\end{enumerate}
A armadilha: $2.37$ tem mais algarismos que $2.5$, mas é
\emph{menor} --- mais algarismos não quer dizer maior!
\end{method}

\begin{example}\label{ex:g4:decimals:compare}
Ordene $0.4$, $0.35$ e $0.09$: em centésimos, $0.40$, $0.35$,
$0.09$; logo
\[
0.09 < 0.35 < 0.4 .
\]
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g4:decimals:1}
Num quadrado de cem, quanto fica pintado se pintarmos: $5$ colunas;
$2$ colunas e $3$ casinhas; $45$ casinhas? Responda com frações.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:2}
Escreva com \omterm{def:g4:decimals:point}{ponto decimal}:
$\frac{9}{10}$; \quad $\frac{27}{100}$; \quad $\frac{3}{100}$; \quad
$5 + \frac{6}{10}$; \quad $\frac{60}{100}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:3}
Escreva como \omterm{def:g4:fractions:def}{fração} (décimos ou centésimos):
$0.3$; \quad $0.51$; \quad $0.07$; \quad $1.9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:4}
Em $6.58$: o que conta o algarismo $5$? E o algarismo $8$?
Decomponha o número como na \cref{def:g4:decimals:point}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:5}
Marque numa reta numérica graduada em décimos de $4$ a $5$:
$4.2$; \ $4.75$ (entre quais duas marcas?); \ $4.9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:6}
Copie e complete com $<$, $>$ ou $=$:
\[
0.6 \;?\; 0.60, \qquad
2.8 \;?\; 2.75, \qquad
0.09 \;?\; 0.1, \qquad
5.3 \;?\; 5.13 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:7}
Ordene do menor para o maior: $1.05$; \ $1.5$; \ $0.95$; \ $1.45$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:8}
Escreva na escrita decimal: três reais e cinco centavos; doze reais
e cinquenta centavos; noventa e nove centavos. Depois ordene os três
preços.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:decimals:9}
Quais números decimais com uma casa decimal ficam estritamente entre
$3$ e $4$? Liste todos. Quantos são?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:decimals:10}
Tomás diz: ``$0.25$ é maior que $0.5$ porque $25$ é maior que $5$''.
Mostre a ele o erro com o quadrado de cem (quantas casinhas cada um
pinta?).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:decimals:11}
Os três saltos de um atleta mediram $3.6$ m, $3.58$ m e $3.65$ m.
Ordene os saltos do mais curto ao mais longo. Quais dois diferem por
exatamente $\frac{5}{100}$ de metro?
\end{exercise}
