\chapter{Multiplicando números maiores}\label{ch:g4:mult}

O Ano 3 multiplicou por um algarismo (\cref{ch:g3:mult}); este
capítulo multiplica por números de dois algarismos --- e explica a
misteriosa ``linha deslocada'' da conta armada, que nada mais é do
que uma decomposição em dezenas e unidades.

\section{Multiplicar por dezenas}

\begin{proposition}[Vezes 10, 20, 300\,\dots]\label{prop:g4:mult:tens}
Multiplicar por $10$ acrescenta um \omterm{def:g1:counting:zero}{zero}; por $100$, dois \omterm{def:g1:counting:zero}{zeros}. Para
multiplicar por $20$, multiplique por $2$ e depois por $10$:
\[
36 \times 20 = 36 \times 2 \times 10 = 72 \times 10 = 720 .
\]
Do mesmo modo, $14 \times 300 = 14 \times 3 \times 100 = 4\,200$.
\end{proposition}
\admitted

\section{A conta armada com dois algarismos}

\begin{example}[Por que a linha deslocada?]\label{ex:g4:mult:why}
Para calcular $23 \times 12$, decomponha $12$ em $10 + 2$:
\[
23 \times 12 = 23 \times 2 + 23 \times 10 = 46 + 230 = 276 .
\]
A conta armada escreve exatamente esses dois \omterm{def:g2:mult:def}{produtos}, um embaixo do
outro --- o segundo deslocado para a esquerda porque é um número de
\emph{dezenas}:
\[
\begin{array}{r}
2\,3 \\
\times\ 1\,2 \\
\hline
4\,6 \\
2\,3\,\cdot \\
\hline
2\,7\,6
\end{array}
\]
(o ponto marca a casa das unidades da linha deslocada, muitas vezes
deixada em branco).
\end{example}

\begin{method}[Multiplicação armada por um número de dois algarismos]\label{met:g4:mult:column}
Para calcular $457 \times 36$:
\begin{enumerate}
  \item primeira linha: $457 \times 6 = 2\,742$;
  \item segunda linha: $457 \times 3$ \emph{dezenas} $= 1\,371$
        dezenas --- escreva $1\,371$ deslocado uma casa para a
        esquerda;
  \item some as duas linhas: $2\,742 + 13\,710 = 16\,452$;
  \item estime para conferir: $457 \times 36 \approx 500 \times 36
        = 18\,000$? Melhor: $450 \times 36$ dá cerca de $16\,000$ ---
        plausível.
\end{enumerate}
\[
\begin{array}{r}
4\,5\,7 \\
\times\ \ \,3\,6 \\
\hline
2\,7\,4\,2 \\
1\,3\,7\,1\,\cdot \\
\hline
1\,6\,4\,5\,2
\end{array}
\]
\end{method}

\begin{example}[Truques de cálculo mental]\label{ex:g4:mult:tricks}
\begin{itemize}
  \item \emph{Decompor do jeito amigável}: $18 \times 11 = 18 \times
        10 + 18 = 198$.
  \item \emph{Usar um número quase redondo}: $25 \times 99 = 25
        \times 100 - 25 = 2\,475$.
  \item \emph{Dobrar e reduzir à \omterm{def:g3:division:half}{metade}}: $16 \times 35 = 8 \times 70
        = 560$ (reduzir um fator à \omterm{def:g3:division:half}{metade} e dobrar o outro mantém o
        \omterm{def:g2:mult:def}{produto}).
\end{itemize}
\end{example}

\section{A multiplicação em problemas}

\begin{example}\label{ex:g4:mult:problem}
Uma escola encomenda $28$ caixas de $145$ folhas de papel.
\begin{enumerate}
  \item Operação: $145 \times 28$.
  \item Estimativa: $150 \times 30 = 4\,500$, um pouco por cima nas
        duas contas.
  \item Conta armada: $145 \times 8 = 1\,160$; $145 \times 2$
        dezenas $= 2\,900$; total $4\,060$.
  \item Frase: a escola recebe $4\,060$ folhas.
\end{enumerate}
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g4:mult:1}
Calcule de cabeça: $47 \times 10$; \quad $23 \times 100$; \quad
$36 \times 20$; \quad $15 \times 300$; \quad $42 \times 50$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:2}
Calcule $34 \times 21$ decompondo como no \cref{ex:g4:mult:why}
($34 \times 21 = 34 \times 20 + 34$) e depois confira armando a
conta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:3}
Arme a conta e calcule: $63 \times 24$; \quad $85 \times 47$; \quad
$126 \times 53$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:4}
Arme a conta, com uma estimativa antes: $208 \times 34$; \quad
$517 \times 62$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:5}
Calcule com um truque do \cref{ex:g4:mult:tricks}:
$45 \times 11$; \quad $32 \times 99$; \quad $24 \times 45$ (dobre e
reduza à \omterm{def:g3:division:half}{metade} --- duas vezes, se quiser).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:6}
Um cinema tem $26$ fileiras de $18$ poltronas. Quantas poltronas ao
todo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:7}
Um caminhão leva $32$ paletes de $48$ engradados. Quantos
engradados? Dê primeiro uma estimativa e depois a resposta exata.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:mult:8}
Encontre o erro: Ana calcula $57 \times 23$ fazendo
$57 \times 3 = 171$ e $57 \times 2 = 114$, e depois \omterm{def:g1:addition:def}{soma}
$171 + 114 = 285$. A resposta correta é $1\,311$. O que ela
esqueceu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:mult:9}
Um jardineiro planta $14$ fileiras de $35$ tulipas e $12$ fileiras de
$28$ narcisos. Quantas flores ao todo? (Dois \omterm{def:g2:mult:def}{produtos} e depois uma
\omterm{def:g1:addition:def}{soma}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:mult:10}
Todo dia, uma padaria usa $67$ kg de farinha. Quanta farinha é isso,
mais ou menos, num mês de $30$ dias --- estime com números
arredondados e depois calcule exatamente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:mult:11}
Explique, com um desenho retangular como o deste capítulo, por que
$15 \times 12$ pode ser calculado como
$15 \times 10 + 15 \times 2$. Depois invente outra decomposição de
$15 \times 12$ que também funcione.
\end{exercise}
