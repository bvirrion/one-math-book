\chapter{Primeiras frações}\label{ch:g4:fractions}

Meia maçã, um quarto de hora, três quartos da turma: as frações dão
nome aos pedaços das coisas. Este capítulo apresenta a escrita
$\frac{a}{b}$, lê frações em desenhos e na reta numérica, e compara
as mais simples. (As frações crescem no \cref{ch:g6:fractions}.)

\section{Dar nome aos pedaços}

\begin{definition}[Fração]\label{def:g4:fractions:def}
Corte um inteiro em partes iguais e tome algumas --- isso é uma
\emph{fração}\index{fração}:
\[
\frac{3}{4}
\quad
\begin{array}{l}
3 \text{: quantas partes são tomadas (o \emph{numerador}\index{numerador});}\\
4 \text{: em quantas partes iguais o inteiro é cortado (o
\emph{denominador}\index{denominador}).}
\end{array}
\]
$\frac12$ é \emph{um meio}, $\frac13$ \emph{um terço}, $\frac14$
\emph{um quarto}, $\frac{1}{10}$ \emph{um décimo}.
\end{definition}

\begin{example}[Frações de uma coleção]\label{ex:g4:fractions:collection}
$\frac13$ de $12$ bolinhas: reparta as $12$ bolinhas em $3$ grupos
iguais de $4$; um grupo é $\frac13$, então $\frac13$ de $12$ é $4$
--- e $\frac23$ de $12$ são dois grupos, $8$.
\end{example}

\section{Frações na reta numérica}

\begin{method}[Marcar $\frac{a}{b}$]\label{met:g4:fractions:line}
Corte cada unidade da reta em $b$ passos iguais; a partir do $0$,
ande $a$ passos. Se $a$ for maior que $b$, a caminhada passa do $1$:
uma \omterm{def:g4:fractions:def}{fração} pode ser maior que um inteiro!
\end{method}

$\frac12$; $\frac54$ está um quarto depois do $1$; $\frac{11}{4}$ é

\begin{example}[Números inteiros escondidos em frações]\label{ex:g4:fractions:whole}
$\frac44 = 1$ (quatro quartos fazem o inteiro); $\frac82 = 4$;
$\frac{12}{3} = 4$. E $\frac74$ é $1 + \frac34$: um inteiro e mais
três quartos.
\end{example}

\section{Comparar e somar frações simples}

\begin{proposition}[Mesmo denominador]\label{prop:g4:fractions:compare}
Com o mesmo \omterm{def:g4:fractions:def}{denominador}, os pedaços têm o mesmo tamanho --- então
basta contá-los:
\[
\frac{2}{8} < \frac{5}{8},
\qquad
\frac{3}{8} + \frac{4}{8} = \frac{7}{8}.
\]
Comparar com um inteiro também é fácil: $\frac{5}{8} < 1 <
\frac{9}{8}$ (menos e depois mais que $8$ oitavos).
\end{proposition}

\begin{proof}[Por quê, no desenho]
Pintar $3$ oitavos e depois mais $4$ oitavos da mesma barra deixa $7$
oitavos pintados ao todo.
\end{proof}

\begin{example}[Denominadores diferentes, o mesmo ponto]\label{ex:g4:fractions:equal}
Na reta numérica, $\frac12$, $\frac24$ e $\frac{5}{10}$ caem todos no
mesmo ponto: são três nomes do mesmo número. Cortar cada \omterm{def:g3:division:half}{metade} em
duas dá quartos; em cinco, décimos. (A regra geral está no
\cref{ch:g6:fractions}.)
\end{example}

\begin{example}[Quartos de hora]\label{ex:g4:fractions:hour}
Uma hora tem $60$ minutos, então um quarto de hora é
$60 \div 4 = 15$ minutos, e três quartos de hora são $45$ minutos.
``Duas e meia'' é $\frac12$ hora depois das duas.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g4:fractions:1}
Escreva a \omterm{def:g4:fractions:def}{fração} mostrada: uma pizza cortada em $6$ com $5$ fatias
sobrando; uma barra de chocolate de $8$ quadradinhos com $3$ comidos
(\omterm{def:g4:fractions:def}{fração} comida); uma bandeira dividida em $3$ faixas verticais
iguais, $1$ colorida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:2}
Desenhe uma barra cortada em $5$ partes iguais e pinte $\frac35$.
Depois pinte $\frac{2}{5}$ de outra barra igual. Qual pintura é
maior?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:3}
Calcule: $\frac12$ de $18$; \quad $\frac14$ de $20$; \quad
$\frac13$ de $21$; \quad $\frac34$ de $20$ (três grupos de um
quarto).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:4}
Marque numa reta numérica cortada em terços: $\frac13$; \
$\frac33$; \ $\frac53$; \ $2$. Que \omterm{def:g4:fractions:def}{fração} cai exatamente sobre o
$2$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:5}
Copie e complete com $<$, $>$ ou $=$:
\[
\frac38 \;?\; \frac68, \qquad
\frac55 \;?\; 1, \qquad
\frac74 \;?\; 1, \qquad
\frac12 \;?\; \frac24 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:6}
Calcule: $\frac26 + \frac36$; \quad $\frac58 - \frac28$; \quad
$\frac14 + \frac24$; \quad $1 - \frac13$ (quantos terços fazem
$1$?).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:7}
Escreva como um número inteiro mais uma \omterm{def:g4:fractions:def}{fração} menor que $1$:
$\frac54$; \quad $\frac73$; \quad $\frac{9}{2}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:8}
Quantos minutos são: meia hora; um quarto de hora; três quartos de
hora; $\frac13$ de hora?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:fractions:9}
Numa turma de $24$ alunos, $\frac14$ usa óculos. Quantos alunos usam
óculos? Quantos não usam?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:fractions:10}
Nina comeu $\frac38$ de uma pizza e Samuel comeu $\frac48$ da mesma
pizza. Que \omterm{def:g4:fractions:def}{fração} eles comeram juntos? Que \omterm{def:g4:fractions:def}{fração} sobrou? Quem comeu
mais?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:fractions:11}
Verdadeiro ou falso, com um desenho ou um exemplo: ``é possível que
$\frac12$ de uma pizza pequena seja menos que $\frac14$ de uma pizza
enorme''. O que é preciso saber sempre antes de comparar duas frações
de alguma coisa?
\end{exercise}
