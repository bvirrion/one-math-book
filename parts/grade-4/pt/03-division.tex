\chapter{Divisão}\label{ch:g4:division}

O Ano 3 dividiu com as tabuadas e \omterm{def:g3:division:remainder}{restos} pequenos
(\cref{ch:g3:division}); este capítulo instala a conta completa --- a
divisão armada --- que dá conta de qualquer número, grupo de
algarismos por grupo de algarismos.

\section{O algoritmo da divisão armada}

\begin{method}[Divisão armada por um número de um algarismo]\label{met:g4:division:method}
Para dividir $749$ por $6$:
\begin{enumerate}
  \item tome o algarismo mais à esquerda, $7$ (centenas): o $6$ cabe
        $1$ vez em $7$; escreva $1$ no \omterm{def:g3:division:remainder}{quociente}, subtraia $6$, resta
        $1$;
  \item \emph{baixe} o algarismo seguinte, $4$: agora divida $14$
        (dezenas): $6 \times 2 = 12$; escreva $2$, resta $2$;
  \item baixe o $9$: divida $29$: $6 \times 4 = 24$; escreva $4$,
        resta $5$;
  \item não sobrou algarismo: o \omterm{def:g3:division:remainder}{quociente} é $124$ e o \omterm{def:g3:division:remainder}{resto} é $5$.
\end{enumerate}
Conferência: $6 \times 124 + 5 = 744 + 5 = 749$, e $5 < 6$.
\end{method}

\begin{example}[Um zero no quociente]\label{ex:g4:division:zero}
Divida $618$ por $3$: centenas, $6 \div 3 = 2$; dezenas, baixe o
$1$: o $3$ cabe $0$ vez em $1$ --- \emph{escreva o} $0$! --- resta
$1$; unidades, baixe o $8$: $18 \div 3 = 6$. \omterm{def:g3:division:remainder}{Quociente} $206$, \omterm{def:g3:division:remainder}{resto}
$0$. Esquecer o \omterm{def:g1:counting:zero}{zero} do meio (escrever $26$) é o erro clássico; a
conferência $3 \times 26 = 78 \neq 618$ o denuncia na hora.
\end{example}

\begin{example}[Estimar o tamanho do quociente]\label{ex:g4:division:size}
Antes de dividir $749$ por $6$, enquadre a resposta:
$6 \times 100 = 600$ e $6 \times 200 = 1\,200$, então o \omterm{def:g3:division:remainder}{quociente}
está entre $100$ e $200$ --- vai ter três algarismos. Essa
estimativa de uma linha evita quase todos os erros de algarismo fora
de lugar.
\end{example}

\section{Escolher o que a pergunta pede}

\begin{method}[Quociente, resto ou quociente mais um]\label{met:g4:division:interpret}
Como no \cref{met:g3:division:interpret}, a história decide:
\begin{enumerate}
  \item ``quantos \dots\ completos'' ou ``quantos para cada um'': o
        \omterm{def:g3:division:remainder}{quociente};
  \item ``quantos sobram'': o \omterm{def:g3:division:remainder}{resto};
  \item ``quantos recipientes para tudo'': o \omterm{def:g3:division:remainder}{quociente}, mais um se o
        \omterm{def:g3:division:remainder}{resto} não for \omterm{def:g1:counting:zero}{zero}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g4:division:problem}
$530$ livros têm de ser embalados em caixas de $8$.
\[
530 = 8 \times 66 + 2 .
\]
Quantas caixas \emph{cheias}? $66$. Quantos livros não estão numa
caixa cheia? $2$. Quantas caixas para embalar \emph{todos} os
livros? $67$. Três perguntas, uma divisão.
\end{example}

\begin{example}[Divisão justa de uma despesa]\label{ex:g4:division:money}
Quatro amigos dividem igualmente o custo de um presente de $92$:
$92 \div 4 = 23$ exatamente (\omterm{def:g3:division:remainder}{resto} $0$). Cada um paga $23$. Quando o
\omterm{def:g3:division:remainder}{resto} é \omterm{def:g1:counting:zero}{zero}, a divisão é ``exata'' e a partilha fica perfeitamente
justa.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g4:division:1}
Enquadre cada \omterm{def:g3:division:remainder}{quociente} como no \cref{ex:g4:division:size} (entre
quais dois números ``redondos''?) e depois faça a divisão armada:
$96 \div 4$; \quad $87 \div 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:2}
Faça as divisões armadas e confira cada uma:
$672 \div 4$; \quad $925 \div 7$; \quad $804 \div 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:3}
Cuidado com o \omterm{def:g1:counting:zero}{zero} (veja o \cref{ex:g4:division:zero}):
$816 \div 4$; \quad $420 \div 7$; \quad $2\,512 \div 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:4}
Escreva a igualdade de conferência ($a = b \times q + r$) para:
$85 \div 9$; \quad $1\,000 \div 3$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:5}
$156$ ovos são postos em caixas de $6$. Quantas caixas ficam cheias?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:6}
Uma fita de $250$ cm é cortada em pedaços de $8$ cm. Quantos
pedaços, e que comprimento sobra?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:7}
$375$ alunos vão a um espetáculo em ônibus de $52$ lugares. Quantos
ônibus são necessários? (Qual caso do
\cref{met:g4:division:interpret} é este? Divida por $52$ usando
múltiplos: $52 \times 7 = 364$.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:8}
Três amigos repartem $234$ bolinhas igualmente. Quantas bolinhas para
cada um? Confira com uma multiplicação.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g4:division:9}
Descubra o dividendo: a divisão por $7$ dá \omterm{def:g3:division:remainder}{quociente} $58$ e \omterm{def:g3:division:remainder}{resto}
$3$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:division:10}
Um bibliotecário arruma $438$ livros, $9$ por prateleira. As
prateleiras vêm em estantes de $6$ prateleiras. Quantas prateleiras
são necessárias? Quantas estantes é preciso comprar? (Duas divisões,
as duas com uma pergunta do tipo ``mais um''.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g4:division:11}
Numa divisão por $8$, o \omterm{def:g3:division:remainder}{quociente} é igual ao \omterm{def:g3:division:remainder}{resto}. Quais são os
dividendos possíveis? Liste todos. (O \omterm{def:g3:division:remainder}{resto} tem de ficar abaixo de
$8$.)
\end{exercise}
