\chapter{Proporcionalidade, velocidade e médias}\label{ch:g8:speed}

A \omterm{def:g7:prop:table}{proporcionalidade} (\cref{ch:g7:prop}) agora encontra o mundo físico:
velocidade, vazão, conversões de unidades --- grandezas que são
\emph{\omterm{def:g3:division:remainder}{quocientes}} de duas outras. O capítulo se fecha com as \omterm{def:g8:speed:average}{médias
ponderadas}, em que o raciocínio proporcional evita um erro clássico.

\section{Velocidade média}

\begin{definition}[Velocidade média]\label{def:g8:speed:def}
A \emph{velocidade média}\index{velocidade média} de um percurso é o
\omterm{def:g3:division:remainder}{quociente}
\[
v = \frac{d}{t}
\qquad
\left(\text{distância dividida pela duração}\right),
\]
em quilômetros por hora (km/h), metros por segundo (m/s)\,\dots\ A
fórmula se vira do avesso: $d = v \times t$ e $t = \frac{d}{v}$.
\end{definition}

\begin{example}\label{ex:g8:speed:basic}
Um trem percorre $270$ km em $1$ h $30$ min. Primeiro converta o
tempo: $1$ h $30$ min $= 1.5$ h. Então
\[
v = \frac{270}{1.5} = 180 \text{ km/h}.
\]
Distância a $90$ km/h durante $2$ h $20$ min $= \frac73$ h:
$d = 90 \times \frac73 = 210$ km. Tempo para $35$ km a $14$ km/h:
$t = \frac{35}{14} = 2.5$ h $= 2$ h $30$ min.
\end{example}

\begin{remark}[Minutos não são decimais]
$1$ h $30$ min é $1.5$ h, mas $1$ h $20$ min \emph{não} é $1.2$ h: é
$1 + \frac{20}{60} = \frac43 \approx 1.33$ h. Converta sempre os
minutos numa \omterm{def:g6:fractions:def}{fração} de hora ($60$ min $= 1$ h) antes de dividir.
\end{remark}

\begin{method}[Converter velocidades]\label{met:g8:speed:convert}
Para converter km/h em m/s: $1$ km $= 1000$ m e $1$ h $= 3600$ s,
então
\[
v \text{ km/h} = \frac{v \times 1000}{3600} \text{ m/s}
= \frac{v}{3.6} \text{ m/s}.
\]
Para converter m/s em km/h, multiplique por $3.6$.
\end{method}

\begin{example}\label{ex:g8:speed:convert}
$90$ km/h $= \frac{90}{3.6} = 25$ m/s. Um velocista que corre $100$ m
em $10$ s se move a $10$ m/s $= 36$ km/h.
\end{example}

\section{Grandezas quociente}

\begin{example}[Vazão, densidade, preço por quilo]\label{ex:g8:speed:quotients}
A velocidade tem muitas primas, todas tratadas do mesmo jeito:
\begin{itemize}
  \item uma torneira enche $48$ L em $4$ min: vazão
        $\frac{48}{4} = 12$ L/min, então encher uma banheira de
        $150$ L leva $\frac{150}{12} = 12.5$ min;
  \item $0.6$ kg de queijo custa $9$ euros: preço
        $\frac{9}{0.6} = 15$ euros por kg;
  \item um carro gasta $6.3$ L em $90$ km: consumo
        $\frac{6.3}{90} \times 100 = 7$ L por $100$ km.
\end{itemize}
Identifique o \omterm{def:g3:division:remainder}{quociente}, e a fórmula de três pontas
($q = \frac ab$, $a = qb$, $b = \frac aq$) faz o \omterm{def:g3:division:remainder}{resto}.
\end{example}

\section{Médias ponderadas}

\begin{definition}[Média ponderada]\label{def:g8:speed:average}
Quando os valores $v_1, v_2, \dots$ vêm com quantidades (ou pesos)
$n_1, n_2, \dots$, a \emph{média ponderada}\index{média ponderada}
deles é
\[
\bar v = \frac{n_1 v_1 + n_2 v_2 + \dots}{n_1 + n_2 + \dots} :
\]
total dos valores, dividido pelo total dos pesos.
\end{definition}

\begin{example}\label{ex:g8:speed:average}
Uma prova foi feita por dois grupos: grupo 1, $12$ alunos, média $11$;
grupo 2, $18$ alunos, média $14$. Média da turma inteira:
\[
\bar v = \frac{12 \times 11 + 18 \times 14}{12 + 18}
= \frac{132 + 252}{30} = \frac{384}{30} = 12.8 .
\]
\emph{Não} $\frac{11 + 14}{2} = 12.5$: o grupo maior puxa a média para
a média dele próprio --- os pesos contam.
\end{example}

\begin{example}[Velocidade média em dois trechos]\label{ex:g8:speed:twolegs}
Um ciclista pedala $30$ km a $30$ km/h e depois $30$ km a $15$ km/h. A
\omterm{def:g8:speed:def}{velocidade média} do trajeto inteiro \emph{não} é
$\frac{30 + 15}{2} = 22.5$ km/h. Calcule pela definição:
\begin{enumerate}
  \item tempos: $\frac{30}{30} = 1$ h e depois $\frac{30}{15} = 2$ h;
        total $3$ h;
  \item distância total $60$ km, então $v = \frac{60}{3} = 20$ km/h.
\end{enumerate}
O trecho lento dura mais, então pesa mais --- uma média de velocidades
tem de ser ponderada pelo \emph{tempo}.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g8:speed:1}
Calcule a \omterm{def:g8:speed:def}{velocidade média}: $150$ km em $2$ h; $27$ km em $45$ min;
$100$ m em $12.5$ s.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:2}
Calcule a distância: $2$ h a $85$ km/h; $40$ min a $90$ km/h. Calcule
o tempo: $315$ km a $90$ km/h (em h e min).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:3}
Converta: $72$ km/h em m/s; $5$ m/s em km/h; $108$ km/h em m/s.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:4}
Uma torneira entrega $15$ L/min. Quanto tempo leva para encher um
reservatório de $600$ L? Quantos litros ela entrega em $2$ h $30$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:5}
Qual é a melhor compra: $1.2$ kg de maçãs por $3.30$ euros ou $0.8$ kg
por $2.16$ euros? Compare os preços por quilograma.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:6}
Uma turma de $25$ alunos tem média $12.4$ numa prova; outra turma, de
$35$ alunos, tem média $10$. Calcule a média das duas turmas juntas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:7}
O som viaja a cerca de $340$ m/s. Você vê um relâmpago e ouve o trovão
$6$ segundos depois. A que distância o \omterm{def:g3:shapes:shapes}{raio} caiu (arredondando à
centena de metros)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:speed:8}
Um caminhante anda $2$ h a $5$ km/h, descansa $30$ min e depois anda
$1$ h $30$ a $4$ km/h. Calcule a distância total e a \omterm{def:g8:speed:def}{velocidade média}
\emph{incluindo o descanso} (distância total dividida pelo tempo total
decorrido).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:speed:9}
Notas com pesos: uma aluna tirou $15$ (peso $3$), $9$ (peso $2$) e
$12$ (peso $1$). Calcule a \omterm{def:g8:speed:average}{média ponderada}. Que média simples (sem
pesos) ela teria, e por que as duas diferem?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:speed:10}
Um carro roda $120$ km a $80$ km/h e depois $120$ km a $120$ km/h.
\begin{enumerate}
  \item Calcule a duração de cada trecho e depois a \omterm{def:g8:speed:def}{velocidade média}
        do trajeto inteiro.
  \item Explique por que a resposta é menor que $100$ km/h, o ponto
        médio das duas velocidades.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:speed:11}
Dois povoados distam $18$ km um do outro. Ana sai do primeiro às
$10{:}00$ andando a $5$ km/h em direção ao segundo; Bruno sai do
segundo na mesma hora andando a $4$ km/h em direção ao primeiro. A que
horas eles se encontram, e a que distância do povoado de Ana? (Juntos,
eles fecham a distância a $5 + 4$ km/h.)
\end{exercise}

\section{Problema: a média harmônica, ou por que a viagem de volta
estraga a média}

\begin{problem}[{Problema de fim de semana --- a \omterm{def:g8:speed:def}{velocidade média} em
distâncias iguais é $\frac{2 v_1 v_2}{v_1 + v_2}$, e ela nunca supera o
ponto médio das velocidades}]
\label{pb:g8:speed:1}
O \cref{ex:g8:speed:twolegs} e o \cref{exo:g8:speed:10} terminaram os
dois com a mesma surpresa: vá a uma velocidade, volte a outra, e a
\omterm{def:g8:speed:def}{velocidade média} cai \emph{abaixo} do ponto médio das duas. Este
problema encontra a fórmula exata por trás da surpresa --- um novo tipo
de média, a \emph{média harmônica}\index{média harmônica} ---,
demonstra que a surpresa é um teorema e a leva ao extremo lógico: uma
recuperação que nenhuma velocidade do mundo consegue.

\medskip
\noindent\textbf{Parte I --- A fórmula da ida e volta.}
Um percurso cobre uma distância $d$ à velocidade $v_1$ e depois a mesma
distância $d$ de novo à velocidade $v_2$ (todos números positivos).
\begin{enumerate}
  \item Expresse as duas durações e depois a duração total como
        frações que envolvam $d$, $v_1$, $v_2$.
  \item A \omterm{def:g8:speed:def}{velocidade média} do percurso inteiro é
        $\bar v = \dfrac{2d}{\ \dfrac{d}{v_1} + \dfrac{d}{v_2}\ }$ ---
        uma \omterm{def:g6:fractions:def}{fração} de dois andares
        (\cref{ex:g8:fractions:complex}). Simplifique-a e demonstre:
        \[
        \bar v = \frac{2\, v_1 v_2}{v_1 + v_2} .
        \]
  \item Confira a fórmula nas duas contas citadas acima: $v_1 = 30$,
        $v_2 = 15$ km/h (\cref{ex:g8:speed:twolegs}), e $v_1 = 80$,
        $v_2 = 120$ km/h (\cref{exo:g8:speed:10}).
  \item A distância $d$ desapareceu da fórmula. O que isso significa,
        concretamente, para o ciclista?
  \item Agora suponha que o percurso passe, em vez disso, o mesmo
        \emph{tempo} $T$ a cada velocidade. Mostre que a \omterm{def:g8:speed:def}{velocidade
        média} é então $\frac{v_1 + v_2}{2}$, o ponto médio simples. Na
        linguagem da \cref{def:g8:speed:average}: por que grandeza uma
        média de velocidades tem sempre de ser ponderada, e o que muda
        entre os dois cenários?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Harmônica contra aritmética.}
Para dois números positivos $v_1$, $v_2$, ponha
\[
H = \frac{2\, v_1 v_2}{v_1 + v_2}
\quad\text{(a \emph{média harmônica} deles),}
\qquad
A = \frac{v_1 + v_2}{2}
\quad\text{(a \emph{média aritmética} deles).}
\]
\begin{enumerate}[resume]
  \item Calcule $H$ e $A$ para os pares $(30, 15)$ e $(80, 120)$. Qual
        das duas médias ganha em cada caso?
  \item Coloque $A - H$ sobre o \omterm{def:g4:fractions:def}{denominador} comum $2(v_1 + v_2)$,
        desenvolva o \omterm{def:g4:fractions:def}{numerador} com os \omterm{def:g2:mult:def}{produtos} notáveis
        (\cref{pb:g8:equations:1}) e demonstre:
        \[
        A - H = \frac{(v_1 - v_2)^2}{2\,(v_1 + v_2)} .
        \]
  \item Deduza o teorema por trás de todas as surpresas: para
        velocidades positivas, $H \leq A$ sempre, com igualdade
        exatamente quando $v_1 = v_2$. Por que um quadrado no \omterm{def:g4:fractions:def}{numerador}
        resolve a questão?
  \item Demonstre a identidade elegante $H \times A = v_1 \times v_2$:
        as duas médias multiplicadas devolvem o \omterm{def:g2:mult:def}{produto} original.
  \item Um carro percorre $60$ km a $40$ km/h e $60$ km a $60$ km/h.
        Preveja a \omterm{def:g8:speed:def}{velocidade média} com a fórmula e depois confirme pelo
        caminho longo, com os tempos e a distância total.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A recuperação impossível.}
\begin{enumerate}[resume]
  \item Uma ciclista entra numa prova de $30$ km esperando fazer uma
        média de $30$ km/h. Que tempo total a prova pode levar, no
        máximo? Ela pedala os primeiros $15$ km a $15$ km/h: quanto
        sobra desse orçamento de tempo?
  \item Mostre que a média esperada ficou literalmente impossível:
        calcule a \omterm{def:g8:speed:def}{velocidade média} dela se pedalar a segunda \omterm{def:g3:division:half}{metade} a
        $90$ km/h, e explique por que mesmo uma velocidade
        arbitrariamente enorme a deixa aquém dos $30$ km/h.
  \item Ela baixa a meta para $20$ km/h. Que velocidade nos $15$ km
        seguintes alcança isso? Confira a resposta com a fórmula da
        \omterm{pb:g8:speed:1}{média harmônica}.
  \item Sozinha, a torneira de água fria enche uma banheira em $20$
        min; sozinha, a de água quente a enche em $30$ min. Que \omterm{def:g6:fractions:def}{fração}
        da banheira as duas torneiras juntas enchem por minuto, e
        quanto tempo elas levam? Compare a resposta com $H(20, 30)$ ---
        o que você percebe?
  \item Um avião voa a mesma rota de ida e de volta: $900$ km/h a favor
        do vento, $700$ km/h contra ele. Calcule a \omterm{def:g8:speed:def}{velocidade média} da
        ida e volta e explique numa frase por que um vento, sopre como
        soprar, sempre \emph{alonga} uma ida e volta.
\end{enumerate}
\end{problem}
