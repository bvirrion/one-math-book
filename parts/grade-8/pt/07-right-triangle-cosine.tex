\chapter{O triângulo retângulo e o cosseno}\label{ch:g8:cosine}

Dois fatos bonitos ligam o \omterm{def:g6:shapes:triangles}{triângulo retângulo} ao círculo: um \omterm{def:g6:shapes:triangles}{triângulo
retângulo} cabe exatamente num semicírculo, com a hipotenusa como
\omterm{def:g4:geometry:circle}{diâmetro}. E um único número, o \emph{cosseno} de um ângulo, guarda a
forma de todo \omterm{def:g6:shapes:triangles}{triângulo retângulo} que tenha esse ângulo --- a primeira
razão trigonométrica, antes das irmãs dela, o seno e a tangente
(\cref{ch:g9:trig}).

\section{O triângulo retângulo e o círculo dele}

\begin{theorem}[Teorema do círculo]\label{thm:g8:cosine:circle}
\begin{enumerate}
  \item Se um triângulo $ABC$ é retângulo em $A$, então $A$ está sobre
        o círculo cujo \omterm{def:g4:geometry:circle}{diâmetro} é a hipotenusa $[BC]$.
  \item Reciprocamente, se $A$ está sobre um círculo de \omterm{def:g4:geometry:circle}{diâmetro} $[BC]$
        (com $A \neq B, C$), então o triângulo $ABC$ é retângulo em
        $A$.
\end{enumerate}
De modo equivalente: num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, o ponto médio da
hipotenusa fica à mesma distância dos três \omterm{def:g5:solids:def}{vértices} --- a mediana que
sai do \omterm{def:g3:shapes:rightangle}{ângulo reto} mede a \omterm{def:g3:division:half}{metade} da hipotenusa.
\end{theorem}

\begin{proof}[Demonstração de 1]
Seja $O$ o ponto médio de $[BC]$ e $D$ o \omterm{def:g7:central:def}{simétrico} de $A$ em relação a
$O$. As diagonais de $ABDC$ se cruzam no ponto médio comum $O$, então
$ABDC$ é um \omterm{def:g7:central:parallelogram}{paralelogramo}
(\cref{def:g7:central:parallelogram}) --- e tem um \omterm{def:g3:shapes:rightangle}{ângulo reto} em $A$:
é um retângulo. As diagonais de um retângulo são iguais, então
$OA = \frac{AD}{2} = \frac{BC}{2}$: o ponto $A$ fica à distância
$\frac{BC}{2}$ de $O$, isto é, sobre o círculo de \omterm{def:g4:geometry:circle}{diâmetro} $[BC]$. (O
ponto 2 se demonstra percorrendo o argumento de trás para a frente.)
\end{proof}

Um triângulo tem hipotenusa de $10$ cm. Sem saber mais nada, a mediana

\end{example}

\section{O cosseno de um ângulo agudo}

\begin{definition}[Cosseno]\label{def:g8:cosine:def}
Num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, para um ângulo agudo $\theta$:
\[
\cos\theta
= \frac{\text{cateto adjacente a } \theta}{\text{hipotenusa}}
\]
--- o cateto que toca $\theta$, dividido pela hipotenusa. Essa razão
depende apenas do ângulo, não do tamanho do triângulo: todos os
\omterm{def:g6:shapes:triangles}{triângulos retângulos} com o mesmo ângulo agudo são ampliações uns dos
outros, e as ampliações preservam as razões de comprimentos (o
\cref{ch:g8:midpoints} começou essa história; o \cref{ch:g9:thales} a
termina).
\end{definition}

$\frac{\text{adjacente}}{\text{hipotenusa}}$ é o mesmo para os dois ---
e esse valor comum é $\cos\theta$.}
\end{omfigure}

\begin{proposition}[Primeiros valores e limites]\label{prop:g8:cosine:bounds}
Para todo ângulo agudo $\theta$: $0 < \cos\theta < 1$, e o cosseno
\emph{diminui} à medida que o ângulo se abre: um ângulo maior tem
cosseno menor. Marcos: $\cos 0^\circ = 1$,
$\cos 60^\circ = \frac12$, $\cos 90^\circ = 0$ (os valores extremos
correspondem a triângulos achatados).
\end{proposition}
\admitted

\begin{method}[Usar o cosseno]\label{met:g8:cosine:use}
Num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, quando as quantidades conhecidas e procuradas
são um ângulo agudo, o cateto adjacente a ele e a hipotenusa:
\begin{enumerate}
  \item escreva a definição:
        $\cos\theta = \frac{\text{adjacente}}{\text{hipotenusa}}$, já
        com os valores conhecidos no lugar;
  \item resolva a equaçãozinha para a incógnita (multiplicando ou
        dividindo);
  \item se a incógnita é um \emph{ângulo}, aplique o $\cos^{-1}$ da
        calculadora à razão calculada;
  \item confira o bom senso: um comprimento tem de sair menor que a
        hipotenusa; um ângulo, estritamente entre $0^\circ$ e
        $90^\circ$.
\end{enumerate}
\end{method}

\begin{example}[Achar um lado]\label{ex:g8:cosine:side}
Uma rampa de $6$ m faz um ângulo de $20^\circ$ com o chão horizontal.
Distância horizontal vencida (adjacente a $20^\circ$, hipotenusa
$6$ m):
\[
\cos 20^\circ = \frac{d}{6}
\quad\Longrightarrow\quad
d = 6 \cos 20^\circ \approx 6 \times 0.940 \approx 5.6 \text{ m}.
\]
\end{example}

\begin{example}[Achar um ângulo]\label{ex:g8:cosine:angle}
Num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, o cateto adjacente ao ângulo $\theta$ mede
$3.5$ e a hipotenusa mede $5$:
\[
\cos\theta = \frac{3.5}{5} = 0.7,
\qquad
\theta = \cos^{-1}(0.7) \approx 45.6^\circ .
\]
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g8:cosine:1}
Um \omterm{def:g6:shapes:triangles}{triângulo retângulo} tem hipotenusa de $13$ cm. Qual é o comprimento
da mediana que sai do \omterm{def:g5:solids:def}{vértice} do \omterm{def:g3:shapes:rightangle}{ângulo reto}? Qual é o \omterm{def:g3:shapes:shapes}{raio} do círculo
que passa pelos três \omterm{def:g5:solids:def}{vértices}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:2}
Trace um círculo de \omterm{def:g4:geometry:circle}{diâmetro} $8$ cm com um \omterm{def:g4:geometry:circle}{diâmetro} $[BC]$, escolha um
ponto $A$ qualquer sobre o círculo e trace $ABC$. Qual ângulo é reto?
Cite o teorema. Meça $AB$ e $AC$ e confira Pitágoras.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:3}
Num triângulo $DEF$ retângulo em $D$, nomeie o cateto adjacente ao
ângulo $\widehat E$, o cateto oposto a ele e a hipotenusa. Escreva
$\cos \widehat E$ como uma razão.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:4}
Calcule a quantidade que falta ($\cos 35^\circ \approx 0.819$,
$\cos 50^\circ \approx 0.643$):
\begin{enumerate}
  \item hipotenusa $10$, ângulo $35^\circ$: cateto adjacente?
  \item cateto adjacente $6$, ângulo $50^\circ$: hipotenusa?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:5}
Num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, o cateto adjacente a $\theta$ mede $8$ e a
hipotenusa mede $10$. Calcule $\cos\theta$ e depois $\theta$
($\cos^{-1}(0.8) \approx 36.9^\circ$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:6}
Explique por que $\cos\theta$ nunca pode valer $1.2$ para um ângulo
agudo de um \omterm{def:g6:shapes:triangles}{triângulo retângulo}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:7}
Uma tirolesa de $25$ m desce de uma plataforma até o chão, fazendo um
ângulo de $12^\circ$ com a horizontal
($\cos 12^\circ \approx 0.978$). Que distância horizontal ela vence?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:8}
Ordene sem calculadora: $\cos 20^\circ$, $\cos 70^\circ$,
$\cos 45^\circ$ (use a \cref{prop:g8:cosine:bounds}). Depois confira
com a calculadora.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:9}
Uma escada de comprimento $L$ se apoia numa parede, com um ângulo
$\theta$ entre a escada e o \emph{chão}. O pé dela fica a $1.2$ m da
parede e $\theta = 68^\circ$ ($\cos 68^\circ \approx 0.375$). Calcule
$L$ e depois a altura alcançada (Pitágoras).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:10}
$[BC]$ é um \omterm{def:g4:geometry:circle}{diâmetro} de um círculo de centro $O$ e \omterm{def:g3:shapes:shapes}{raio} $4.5$ cm, e
$A$ é um ponto do círculo com $AB = 5.4$ cm.
\begin{enumerate}
  \item Por que $ABC$ é retângulo em $A$?
  \item Calcule $AC$ e depois $\cos \widehat B$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:cosine:11}
Usando a \omterm{def:g3:division:half}{metade} de um \omterm{def:g6:shapes:triangles}{triângulo equilátero} de lado $1$ (cortado ao
longo de uma altura), justifique o marco $\cos 60^\circ = \frac12$ da
\cref{prop:g8:cosine:bounds}. (Onde cai o pé da altura sobre a base? A
versão trigonométrica completa é o \cref{ex:g9:trig:special}.)
\end{exercise}

\section{Problema: as relações de Euclides, e Pitágoras mais uma vez}

\begin{problem}[{Problema de fim de semana --- a altura de um triângulo
retângulo: $AB^2 = BH \times BC$, $AH^2 = BH \times HC$, uma segunda
demonstração de Pitágoras e uma máquina de construir raízes
quadradas}]
\label{pb:g8:cosine:1}
Baixe, do \omterm{def:g3:shapes:rightangle}{ângulo reto} de um \omterm{def:g6:shapes:triangles}{triângulo retângulo}, a perpendicular à
hipotenusa: esse \omterm{def:g6:lines:objects}{segmento} curto --- a \emph{altura} --- satisfaz
relações tão úteis que Euclides as colocou no coração dos
\emph{Elementos} dele. Neste problema, a propriedade que define o
cosseno, ``a razão depende apenas do ângulo''
(\cref{def:g8:cosine:def}), demonstra todas elas; pelo caminho, você
vai redemonstrar o teorema de Pitágoras por uma via completamente
diferente e terminar com uma máquina de régua e compasso que constrói
$\sqrt2$, $\sqrt5$, $\sqrt{n}$ para todo número inteiro $n$.

Em todo o problema, $ABC$ é um \omterm{def:g6:shapes:triangles}{triângulo retângulo} em $A$, e $H$ é o pé
da perpendicular baixada de $A$ até a hipotenusa $[BC]$, de modo que
$H$ fica entre $B$ e $C$ e $AH \perp BC$.

\medskip
\noindent\textbf{Parte I --- Um ângulo, dois triângulos.}
\begin{enumerate}
  \item Usando a \omterm{def:g1:addition:def}{soma} dos ângulos de um triângulo
        (\cref{thm:g7:triangles:sum}) em $ABH$ e em $ABC$, mostre que
        $\widehat{BAH} = \widehat{ACB}$: a altura corta o \omterm{def:g6:shapes:triangles}{triângulo
        retângulo} em dois triângulos menores que carregam os
        \emph{mesmos ângulos} do original.
  \item Os triângulos $ABC$ (retângulo em $A$) e $ABH$ (retângulo em
        $H$) compartilham o ângulo $\widehat B$. Escreva
        $\cos\widehat B$ em cada um deles e deduza, da
        \cref{def:g8:cosine:def}, que
        \[
        AB^2 = BH \times BC .
        \]
        (Para passar de razões iguais a \omterm{def:g2:mult:def}{produtos} iguais, multiplique os
        dois lados pelos dois \omterm{def:g4:fractions:def}{denominadores},
        \cref{thm:g8:equations:balance}.)
  \item Enuncie e demonstre a relação gêmea no \omterm{def:g5:solids:def}{vértice} $C$:
        \[
        AC^2 = CH \times CB .
        \]
  \item Tome o triângulo $3$--$4$--$5$: $AB = 3$, $AC = 4$, $BC = 5$.
        Calcule $BH$ e $CH$ pelas questões 2 e 3, e confira que
        $BH + HC = BC$.
  \item Em geral, expresse $BH$ e $CH$ como frações que envolvam apenas
        os três lados. Em que sentido o pé $H$ reparte a hipotenusa
        ``proporcionalmente aos quadrados dos catetos''?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Pitágoras de novo, e a altura.}
\begin{enumerate}[resume]
  \item Some as relações das questões 2 e 3 e use $BH + HC = BC$ para
        obter
        \[
        AB^2 + AC^2 = BC^2 :
        \]
        o teorema de Pitágoras (\cref{thm:g8:pythagoras:direct}),
        redemonstrado --- sem nenhum quebra-cabeça de \omterm{def:g6:measure:area}{áreas} à vista
        (compare com o \cref{exo:g8:pythagoras:11}).
  \item Aplique Pitágoras no triângulo pequeno $ABH$ para escrever
        $AH^2 = AB^2 - BH^2$, substitua $AB^2$ por $BH \times BC$ e
        coloque $BH$ em evidência para demonstrar a \emph{relação da
        altura de Euclides}:
        \[
        AH^2 = BH \times HC .
        \]
  \item Calcule a \omterm{def:g6:measure:area}{área} de $ABC$ de duas maneiras --- com os catetos
        como base e altura, e depois com a hipotenusa como base
        (\cref{thm:g7:areas:triangle}) --- e deduza a terceira
        relação:
        \[
        AB \times AC = BC \times AH .
        \]
  \item De volta ao triângulo $3$--$4$--$5$: calcule $AH$ pela questão
        8 e depois verifique numericamente a relação da altura da
        questão 7, usando os valores de $BH$ e $CH$ achados na questão
        4.
  \item O pé da altura reparte a hipotenusa de certo \omterm{def:g6:shapes:triangles}{triângulo
        retângulo} em \omterm{def:g6:lines:objects}{segmentos} $BH = 2$ cm e $HC = 8$ cm. Calcule a
        altura $AH$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Uma máquina de raízes quadradas.}
Trace um \omterm{def:g6:lines:objects}{segmento} $[BC]$ feito de dois pedaços colocados um após o
outro: $BH = p$ e $HC = q$. Trace o semicírculo de \omterm{def:g4:geometry:circle}{diâmetro} $[BC]$, e
seja $A$ o ponto em que a perpendicular a $(BC)$ em $H$ o encontra.
\begin{enumerate}[resume]
  \item Usando o \cref{thm:g8:cosine:circle}, explique por que o
        triângulo $ABC$ é retângulo em $A$ --- de modo que a questão 7
        se aplica e
        \[
        AH^2 = p \times q .
        \]
        (O comprimento $AH$ se chama \emph{média
        geométrica}\index{média geométrica} de $p$ e $q$.)
  \item Tome $p = 1$ e $q = 2$. Quanto vale $AH^2$? Que comprimento
        famoso do \cref{ex:g8:pythagoras:diagonal} a sua figura de
        régua e compasso acaba de construir?
  \item Descreva a receita que constrói um \omterm{def:g6:lines:objects}{segmento} de comprimento
        $\sqrt n$ para qualquer número inteiro $n \geq 1$, e diga que
        escolha de $p$ e $q$ constrói $\sqrt 5$.
  \item Seja $O$ o centro do semicírculo. Explique por que $AH$ nunca
        pode ultrapassar o \omterm{def:g3:shapes:shapes}{raio} $\frac{p + q}{2}$, e para que posição
        de $H$ os dois são iguais.
  \item Deduza a desigualdade: para todos os números positivos $p$ e
        $q$,
        \[
        p \times q \leq \left(\frac{p + q}{2}\right)^2,
        \]
        com igualdade exatamente quando $p = q$. Depois redemonstre-a
        sem geometria nenhuma: desenvolva
        $\left(\frac{p+q}{2}\right)^2 - p q$ e reconheça um \omterm{def:g2:mult:def}{produto}
        notável (\cref{pb:g8:equations:1}).
\end{enumerate}
\end{problem}
