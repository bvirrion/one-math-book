\chapter{Pontos médios e paralelas}\label{ch:g8:midpoints}

Ligue os pontos médios de dois lados de um triângulo: o \omterm{def:g6:lines:objects}{segmento} obtido
é sempre paralelo ao terceiro lado e tem exatamente a \omterm{def:g3:division:half}{metade} do
comprimento dele. Esse ``teorema dos pontos médios'' e a recíproca dele
são a primeira amostra de uma grande ideia --- as \omterm{def:g6:lines:perp}{paralelas} cortam
comprimentos proporcionalmente --- que floresce no teorema de Tales no
\cref{ch:g9:thales}.

\section{O teorema dos pontos médios}

\begin{theorem}[Teorema dos pontos médios]\label{thm:g8:midpoints:direct}
Num triângulo $ABC$, seja $I$ o ponto médio de $[AB]$ e $J$ o ponto
médio de $[AC]$. Então a reta $(IJ)$ é paralela a $(BC)$ e
\[
IJ = \frac{BC}{2} .
\]
\end{theorem}

\begin{proof}[Ideia da demonstração]
Seja $K$ o \omterm{def:g7:central:def}{simétrico} de $I$ em relação a $J$ (uma meia volta,
\cref{ch:g7:central}). O \omterm{def:g4:geometry:polygon}{quadrilátero} $AICK$ tem diagonais $[IK]$ e
$[AC]$ que se cruzam no ponto médio comum $J$: ele é um \omterm{def:g7:central:parallelogram}{paralelogramo},
então $KC$ é paralelo a $AI$, isto é, a $(AB)$, e $KC = AI = IB$.
Depois, $IBCK$ tem dois lados opostos ($[IB]$ e $[KC]$) paralelos e
iguais: também é um \omterm{def:g7:central:parallelogram}{paralelogramo}
(\cref{thm:g7:central:recognize}), logo $(IK) \parallel (BC)$ --- ou
seja, $(IJ) \parallel (BC)$ --- e $BC = IK = 2\,IJ$.
\end{proof}

\begin{example}\label{ex:g8:midpoints:direct}
Num triângulo $ABC$ com $BC = 9$ cm, ligam-se os pontos médios $I$ de
$[AB]$ e $J$ de $[AC]$. Sem medição nenhuma:
$(IJ) \parallel (BC)$ e $IJ = \frac92 = 4.5$ cm. O triângulo pequeno
$AIJ$ é uma cópia de $ABC$ em tamanho \omterm{def:g3:division:half}{metade} --- o \omterm{def:g6:measure:perimeter}{perímetro} dele
também é a \omterm{def:g3:division:half}{metade} do \omterm{def:g6:measure:perimeter}{perímetro} de $ABC$.
\end{example}

\begin{theorem}[Recíproca]\label{thm:g8:midpoints:converse}
Num triângulo $ABC$, a reta que passa pelo ponto médio $I$ de $[AB]$ e
é \emph{paralela} a $(BC)$ cruza o lado $[AC]$ no ponto médio dele.
\end{theorem}
\admitted

\begin{example}\label{ex:g8:midpoints:converse}
Um caminho atravessa um campo triangular, partindo do meio de uma
borda e correndo paralelo à borda oposta. A recíproca garante que o
caminho sai exatamente no meio da outra borda --- sem precisar medir
nada do lado de lá.
\end{example}

\begin{method}[Escolher entre teorema e recíproca]\label{met:g8:midpoints:choose}
\begin{enumerate}
  \item \emph{Dois pontos médios conhecidos} $\Rightarrow$ use o
        teorema direto: conclua o paralelismo e o meio comprimento.
  \item \emph{Um ponto médio e uma paralela conhecidos} $\Rightarrow$
        use a recíproca: conclua que o ponto de cruzamento é um ponto
        médio.
  \item Escreva que triângulo, que pontos médios e que teorema você
        usa --- uma frase para cada.
\end{enumerate}
\end{method}

\section{Triângulos em tamanho metade}

\begin{proposition}[O triângulo medial]\label{prop:g8:midpoints:medial}
Ligar os três pontos médios dos lados de um triângulo o corta em quatro
triângulos de \omterm{def:g6:measure:area}{áreas} iguais, cada um uma cópia do original em tamanho
\omterm{def:g3:division:half}{metade}.
\end{proposition}

\begin{proof}
Cada lado do triângulo medial é paralelo a um lado de $ABC$ e tem
\omterm{def:g3:division:half}{metade} do comprimento dele (\cref{thm:g8:midpoints:direct}, aplicado
três vezes). Os quatro triângulos pequenos têm lados dos mesmos três
comprimentos, então são cópias idênticas; juntos, eles ladrilham
$ABC$, então cada um tem um quarto da \omterm{def:g6:measure:area}{área} dele.
\end{proof}

\begin{remark}[Rumo a Tales]
O teorema dos pontos médios é o caso ``\omterm{def:g3:division:half}{metade}'' de um fato mais geral:
uma reta paralela a um lado de um triângulo corta os outros dois lados
em \emph{razões iguais} --- um terço e um terço, dois quintos e dois
quintos\,\dots\ Esse enunciado geral é o teorema de Tales
(\cref{ch:g9:thales}); o caso dos pontos médios é o único necessário
este ano.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:1}
Num triângulo $RST$, $M$ é o ponto médio de $[RS]$ e $N$ é o ponto
médio de $[RT]$, com $ST = 7$ cm. O que se pode dizer da reta $(MN)$ e
do comprimento $MN$? Cite o teorema usado.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:2}
Num triângulo $ABC$, $I$ é o ponto médio de $[AB]$ e a reta que passa
por $I$ paralelamente a $(BC)$ corta $[AC]$ em $J$, com $AC = 11$ cm.
Calcule $AJ$, citando o teorema usado.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:3}
Trace um triângulo $ABC$ com $AB = 8$ cm, $AC = 6$ cm e $BC = 7$ cm,
marque os pontos médios $I$ de $[AB]$ e $J$ de $[AC]$, e meça $IJ$ para
conferir o teorema. Calcule o \omterm{def:g6:measure:perimeter}{perímetro} de $AIJ$ sem medir mais nada.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:4}
$IJK$ é o triângulo medial de $ABC$, cujos lados medem $10$, $12$ e
$16$ cm. Dê os três lados de $IJK$ e o \omterm{def:g6:measure:perimeter}{perímetro} dele. Como os dois
\omterm{def:g6:measure:perimeter}{perímetros} se comparam?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:5}
A \omterm{def:g6:measure:area}{área} de um triângulo é $36$ cm$^2$. Qual é a \omterm{def:g6:measure:area}{área} do triângulo medial
dele? E a de cada um dos quatro triângulos pequenos?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:midpoints:6}
Num triângulo $ABC$, $I$ é o ponto médio de $[AB]$, $J$ o de $[AC]$ e
$K$ o de $[BC]$. Mostre que $IJKB$ é um \omterm{def:g7:central:parallelogram}{paralelogramo}. (Compare $[IJ]$
e $[BK]$: paralelos? iguais?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:midpoints:7}
Uma vela triangular tem a borda de baixo com $6.4$ m de comprimento.
Uma costura de reforço liga os pontos médios das outras duas bordas. Que
comprimento de costura é necessário? E se, em vez disso, a costura
ligar os pontos situados a um quarto de cada borda, a partir do \omterm{def:g5:solids:def}{vértice}
de cima? (Conjecture com um desenho; a demonstração é a de Tales,
\cref{ch:g9:thales}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:midpoints:8}
No triângulo $ABC$, retângulo em $A$, $M$ é o ponto médio da hipotenusa
$[BC]$, e a paralela a $(AB)$ que passa por $M$ encontra $[AC]$ em
$N$.
\begin{enumerate}
  \item Mostre que $N$ é o ponto médio de $[AC]$.
  \item Explique por que $(MN)$ é perpendicular a $(AC)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:midpoints:9}
Seja $ABCD$ um \omterm{def:g4:geometry:polygon}{quadrilátero} \emph{qualquer} e $I$, $J$, $K$, $L$ os
pontos médios de $[AB]$, $[BC]$, $[CD]$, $[DA]$ (a conjectura do
\cref{exo:g7:central:11}).
\begin{enumerate}
  \item Aplique o teorema dos pontos médios no triângulo $ABC$ ao
        \omterm{def:g6:lines:objects}{segmento} $[IJ]$, e no triângulo $ACD$ ao \omterm{def:g6:lines:objects}{segmento} $[LK]$:
        compare cada um com a diagonal $[AC]$.
  \item Conclua que $IJKL$ é sempre um \omterm{def:g7:central:parallelogram}{paralelogramo}.
\end{enumerate}
\end{exercise}

\section{Problema: as três medianas e o centro de gravidade}

\begin{problem}[{Problema de fim de semana --- as medianas de um
triângulo se encontram num único ponto, a dois terços do caminho de
cada uma}]
\label{pb:g8:midpoints:1}
Recorte um triângulo de papelão duro e ele vai se equilibrar,
perfeitamente na horizontal, sobre a ponta de um lápis colocada num
ponto especial: o \emph{centro de gravidade}\index{centro de gravidade}
do triângulo. Este problema encontra esse ponto. Uma
\emph{mediana}\index{mediana} de um triângulo é o \omterm{def:g6:lines:objects}{segmento} que liga um
\omterm{def:g5:solids:def}{vértice} ao \emph{ponto médio} do lado oposto. Você vai demonstrar ---
com o teorema dos pontos médios usado duas vezes, de um jeito
inesperado --- que as três medianas passam todas por um ponto, e vai
localizá-lo com precisão.

\medskip
\noindent\textbf{Parte I --- Aquecimento.}
\begin{enumerate}
  \item Trace um triângulo $ABC$ grande (sem forma especial), construa
        os pontos médios $I$ de $[AB]$, $J$ de $[AC]$ e $K$ de $[BC]$,
        e trace as três medianas $[AK]$, $[BJ]$, $[CI]$. O que você
        observa?
  \item Demonstre que uma mediana corta o triângulo em dois triângulos
        de \omterm{def:g6:measure:area}{áreas} iguais. (Compare bases e alturas e use a fórmula da
        \omterm{def:g6:measure:area}{área}: base $\times$ altura $\div\ 2$.) Para um triângulo de
        \omterm{def:g6:measure:area}{área} $36$ cm$^2$, quais são as \omterm{def:g6:measure:area}{áreas} das duas peças?
  \item No triângulo $ABC$, o que o \cref{thm:g8:midpoints:direct} diz
        sobre o \omterm{def:g6:lines:objects}{segmento} $[IJ]$?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Duas medianas se encontram a dois
terços.}
As medianas $[BJ]$ e $[CI]$ se cruzam num ponto; chame-o de $G$. Seja
$M$ o ponto médio de $[GB]$ e $N$ o ponto médio de $[GC]$.
\begin{enumerate}[resume]
  \item Aplique o teorema dos pontos médios \emph{no triângulo $GBC$}:
        o que ele diz sobre o \omterm{def:g6:lines:objects}{segmento} $[MN]$?
  \item Deduza que $(IJ) \parallel (MN)$ e $IJ = MN$, e conclua, com a
        \cref{thm:g7:central:recognize}, que $IJNM$ é um
        \omterm{def:g7:central:parallelogram}{paralelogramo}.
  \item Explique por que $I$ e $N$ estão os dois sobre a mediana
        $(CI)$, e $J$ e $M$ os dois sobre a mediana $(BJ)$ --- de modo
        que as diagonais do \omterm{def:g7:central:parallelogram}{paralelogramo} $IJNM$ são $[IN]$ e $[JM]$,
        e elas se cruzam exatamente em $G$. O que a \emph{definição} de
        \omterm{def:g7:central:parallelogram}{paralelogramo} diz, então, sobre $G$?
  \item Deduza que $BM = MG = GJ$ e conclua:
        \[
        BG = 2 \times GJ
        \]
        --- o ponto $G$ fica sobre a mediana $[BJ]$ a dois terços do
        caminho a partir do \omterm{def:g5:solids:def}{vértice} $B$. Enuncie e justifique o
        resultado análogo na mediana $[CI]$.
  \item Agora esqueça $[CI]$ e refaça exatamente o mesmo argumento com
        o par de medianas $[BJ]$ e $[AK]$: o ponto de cruzamento delas
        também fica a dois terços de $[BJ]$ a partir de $B$. Explique
        por que isso o obriga a ser o \emph{mesmo} ponto $G$ --- e
        conclua que as três medianas passam por $G$.
  \item Num triângulo em que a mediana $[BJ]$ mede $9$ cm, a que
        distância $G$ fica de $B$ e de $J$?
  \item Justifique (em uma frase) que $AG = 2 \times GK$ também.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Seis fatias iguais e coordenadas.}
As três medianas cortam o triângulo em seis triângulos pequenos em
torno de $G$.
\begin{enumerate}[resume]
  \item No triângulo $GBC$, o \omterm{def:g6:lines:objects}{segmento} $[GK]$ é uma mediana. Deduza que
        os dois triângulos pequenos $GBK$ e $GKC$ têm \omterm{def:g6:measure:area}{áreas} iguais.
  \item Usando a questão 2 nos triângulos $ABK$ e $AKC$ (os dois
        cortados pela mediana $[AK]$ de $ABC$), mostre que os
        triângulos $ABG$ e $ACG$ têm \omterm{def:g6:measure:area}{áreas} iguais. Repetindo com outra
        mediana, conclua que os três triângulos $ABG$, $BCG$, $CAG$ têm
        todos \omterm{def:g6:measure:area}{área} igual a um terço de $ABC$.
  \item Cada um desses três triângulos é cortado ao meio por um pedaço
        de mediana (por exemplo, $[GI]$ é uma mediana do triângulo
        $ABG$ --- vista de que \omterm{def:g5:solids:def}{vértice}?). Conclua: as seis fatias
        pequenas em torno de $G$ têm todas a mesma \omterm{def:g6:measure:area}{área}, um sexto do
        triângulo.
  \item Num plano cartesiano, marque $A(0, 0)$, $B(6, 0)$ e $C(0, 6)$,
        com $K(3, 3)$ o ponto médio de $[BC]$. Use o teorema dos dois
        terços da Parte II para calcular as coordenadas de $G$ sobre a
        mediana $[AK]$; depois confira que o mesmo ponto fica a dois
        terços da mediana $[BJ]$, em que $J(0, 3)$. Compare as
        coordenadas de $G$ com as médias
        \[
        \frac{0 + 6 + 0}{3},
        \qquad
        \frac{0 + 0 + 6}{3} .
        \]
  \item O final do equilíbrio: usando as questões 2 e 8, explique por
        que o triângulo de papelão se equilibra sobre um fio de faca
        apoiado ao longo de qualquer mediana --- quantidades iguais de
        papelão de cada lado --- e por que a ponta do lápis tem, então,
        de ser colocada em $G$, o ponto que os físicos chamam de \omterm{pb:g8:midpoints:1}{centro
        de gravidade}. (Uma demonstração de equilíbrio totalmente
        rigorosa precisa do cálculo integral dos volumes de graduação;
        as \omterm{def:g6:measure:area}{áreas} iguais já a tornam crível hoje.)
\end{enumerate}
\end{problem}
