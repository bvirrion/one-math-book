\chapter{Potências}\label{ch:g8:powers}

Dobre uma folha de papel ao meio $10$ vezes (se conseguir!): a espessura
dobra a cada vez, e $10$ dobras a multiplicam por $2^{10} = 1024$. As
potências são a escrita abreviada da multiplicação repetida; este
capítulo monta a notação e as regras dela, com um papel especial para
as potências de $10$.

\section{Definição}

\begin{definition}[Potência]\label{def:g8:powers:def}
Para um número $a$ e um número inteiro $n \geq 1$:
\[
a^n = \underbrace{a \times a \times \dots \times a}_{n
\text{ fatores}},
\]
que se lê ``$a$ elevado a $n$''; $a$ é a \emph{base} e $n$ é o
\emph{expoente}\index{expoente}. Nomes especiais: $a^2$ é ``$a$ ao
quadrado'' e $a^3$, ``$a$ ao cubo''. Por convenção:
\[
a^1 = a,
\qquad
a^0 = 1 \ (a \neq 0),
\qquad
a^{-n} = \frac{1}{a^n}.
\]
\end{definition}

\begin{example}\label{ex:g8:powers:def}
$3^4 = 3 \times 3 \times 3 \times 3 = 81$; \quad
$10^3 = 1000$; \quad $5^{-2} = \frac{1}{25} = 0.04$; \quad
$(-2)^3 = -8$ e $(-2)^4 = +16$
(\cref{ex:g8:negprod:powers}). Atenção: $a^n$ \emph{não} é
$a \times n$: \ $2^5 = 32$, e não $10$.
\end{example}

\section{As regras dos expoentes}

\begin{theorem}[Regras dos expoentes]\label{thm:g8:powers:rules}
Para uma base $a$ não nula e \omterm{def:g8:powers:def}{expoentes} inteiros $m$, $n$:
\[
a^m \times a^n = a^{m+n},
\qquad
\frac{a^m}{a^n} = a^{m-n},
\qquad
\left(a^m\right)^n = a^{m \times n},
\qquad
(ab)^n = a^n b^n .
\]
\end{theorem}

\begin{proof}[Demonstração contando fatores]
$a^m \times a^n$ enfileira $m$ fatores $a$ seguidos de mais $n$:
$m + n$ ao todo. $\left(a^m\right)^n$ repete um bloco de $m$ fatores
$n$ vezes: $mn$ fatores. $(ab)^n$ contém $n$ letras $a$ e $n$ letras
$b$, que podem ser reagrupadas. A regra do \omterm{def:g3:division:remainder}{quociente} vem de cancelar
$n$ dos $m$ fatores; com as convenções $a^0 = 1$ e
$a^{-n} = \frac{1}{a^n}$, ela continua valendo mesmo quando
$n \geq m$.
\end{proof}

\begin{example}\label{ex:g8:powers:rules}
\[
2^3 \times 2^5 = 2^8 = 256,
\qquad
\frac{7^6}{7^4} = 7^2 = 49,
\qquad
\left(5^2\right)^3 = 5^6,
\qquad
2^4 \times 5^4 = 10^4 .
\]
Uma armadilha: as regras valem para uma \emph{base comum} (ou um
\omterm{def:g8:powers:def}{expoente} comum, no caso da última). Nenhuma regra simplifica
$2^3 \times 5^2$ --- é só calcular: $8 \times 25 = 200$.
\end{example}

\section{Potências de dez}

\begin{proposition}[Potências de dez]\label{prop:g8:powers:ten}
Para $n \geq 1$: $10^n = 1\underbrace{0\dots0}_{n}$ e
$10^{-n} = 0.\underbrace{0\dots0}_{n-1}1$. As regras dos \omterm{def:g8:powers:def}{expoentes} se
leem assim: multiplicar potências de dez \omterm{def:g1:addition:def}{soma} os \omterm{def:g8:powers:def}{expoentes}.
\end{proposition}

\begin{example}\label{ex:g8:powers:ten}
$10^4 \times 10^3 = 10^7$; \quad
$\dfrac{10^2}{10^5} = 10^{-3} = 0.001$. Quantidades enormes e
minúsculas ficam legíveis:
\[
\text{um bilhão} = 10^9,
\qquad
\text{um milionésimo} = 10^{-6}.
\]
Combinadas com decimais: $3.2 \times 10^5 = 320\,000$ e
$4.7 \times 10^{-3} = 0.0047$ (deslocando o \omterm{def:g4:decimals:point}{ponto decimal},
\cref{prop:g6:decimals:shift}). O uso sistemático dessa escrita --- a
\emph{notação científica} --- é desenvolvido no
\cref{ch:g9:fractions}.
\end{example}

\begin{example}[Ordens de grandeza]\label{ex:g8:powers:magnitude}
A luz percorre cerca de $3 \times 10^8$ m/s; um ano tem cerca de
$3.2 \times 10^7$ segundos. Um ano-luz é, portanto, cerca de
\[
3 \times 3.2 \times 10^{8+7} \approx 10 \times 10^{15} = 10^{16}
\text{ m}
\]
--- dez milhões de bilhões de metros. As potências de dez fazem as
contas astronômicas caberem numa linha.
\end{example}

\begin{method}[Simplificar uma expressão com potências]\label{met:g8:powers:simplify}
\begin{enumerate}
  \item Agrupe os fatores base por base;
  \item aplique as regras dos \omterm{def:g8:powers:def}{expoentes} dentro de cada base;
  \item calcule as potências pequenas que restarem, ou deixe a resposta
        em forma de potência se ela for grande.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:powers:simplify}
\[
\frac{3^5 \times 3^{-2} \times 4^2}{3^2}
= \frac{3^{5 + (-2)}}{3^2} \times 16
= 3^{3 - 2} \times 16
= 3 \times 16 = 48 .
\]
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g8:powers:1}
Calcule:
\[
2^6, \qquad 3^3, \qquad 10^5, \qquad 1^{100}, \qquad 0.1^2, \qquad
6^0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:2}
Calcule:
\[
(-3)^2, \qquad -3^2, \qquad (-1)^{15}, \qquad (-5)^3, \qquad
\left(\tfrac{2}{3}\right)^2 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:3}
Escreva como uma única potência:
\[
7^4 \times 7^5, \qquad
\frac{2^9}{2^3}, \qquad
\left(10^3\right)^4, \qquad
5^6 \times 5^{-2}, \qquad
3^4 \times 7^4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:4}
Escreva como \omterm{def:g6:decimals:places}{número decimal}: $10^{-2}$; \quad $4 \times 10^3$; \quad
$2.5 \times 10^{-4}$; \quad $10^0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:5}
Escreva com uma potência de dez: cem mil; um décimo; dez bilhões;
$0.000\,001$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:6}
Simplifique e depois calcule:
\[
\frac{10^7 \times 10^{-3}}{10^2},
\qquad
\frac{2^5 \times 2^4}{2^6},
\qquad
\left(2^2\right)^3 \times 2^{-4} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:7}
Verdadeiro ou falso? Corrija as falsas.
\[
2^3 \times 2^4 = 2^{12}; \qquad
5^2 + 5^3 = 5^5; \qquad
(3^2)^4 = 3^8; \qquad
10^3 \times 10^3 = 100^3 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:powers:8}
Um boato se espalha: no dia 1, três pessoas o conhecem; a cada dia,
cada pessoa que o conhece conta a três pessoas novas. Escreva com uma
potência o número de pessoas \emph{novas} informadas no dia $4$ e
calcule quantas pessoas conhecem o boato no fim do dia 4 (incluindo as
três iniciais).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:powers:9}
Uma folha de papel tem $0.1$ mm de espessura, isto é, $10^{-4}$ m.
Dobrá-la dobra a espessura a cada vez.
\begin{enumerate}
  \item Expresse a espessura depois de $10$ dobras como um \omterm{def:g2:mult:def}{produto} e
        calcule-a em centímetros ($2^{10} = 1024$).
  \item Depois de $42$ dobras, a espessura seria
        $2^{42} \times 10^{-4}$ m, com
        $2^{42} \approx 4.4 \times 10^{12}$. Mostre que isso ultrapassa
        a distância Terra--Lua, de cerca de $3.8 \times 10^8$ m.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:powers:10}
Ordene do menor para o maior, sem calculadora:
\[
2^{10}, \qquad 10^3, \qquad 3^6, \qquad 5^4 .
\]
(Calcule cada um; $2^{10}$ e $10^3$ são vizinhos famosos.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:powers:11}
O que é maior, $2^{100}$ ou $10^{30}$? Use $2^{10} = 1024 > 10^3$ para
comparar $2^{100} = \left(2^{10}\right)^{10}$ com
$\left(10^3\right)^{10}$.
\end{exercise}

\section{Problema: o tabuleiro de xadrez e as potências de dois}

\begin{problem}[{Problema de fim de semana --- a \omterm{def:g1:addition:def}{soma} geométrica
$1 + 2 + 4 + \dots + 2^{n-1} = 2^n - 1$, de uma lenda famosa aos
números binários}]
\label{pb:g8:powers:1}
A lenda: como recompensa por ter inventado o xadrez, o sábio Sissa
pediu ao rei um grão de trigo na primeira casa do tabuleiro, dois na
segunda, quatro na terceira --- dobrando de casa em casa, até a
sexagésima quarta. O rei riu de tanta modéstia. Este problema calcula o
que o rei prometeu, usando as regras dos \omterm{def:g8:powers:def}{expoentes} da
\cref{thm:g8:powers:rules}, e termina onde a história leva em segredo:
os números binários dentro de todo computador.

\medskip
\noindent\textbf{Parte I --- O truque da duplicação.}
Para $n \geq 1$, seja $S_n$ o número total de grãos nas $n$ primeiras
casas.
\begin{enumerate}
  \item Expresse o número de grãos da casa $k$ como potência de $2$.
        Que potência fica na casa $64$?
  \item Calcule $S_1$, $S_2$, $S_3$, $S_4$ e $S_5$ e compare cada um
        com uma potência de $2$ próxima. Conjecture uma fórmula para
        $S_n$.
  \item O \emph{truque da duplicação}: escreva as \omterm{def:g1:addition:def}{somas} $S_n$ e
        $2 \times S_n$ uma embaixo da outra, subtraia e demonstre a sua
        conjectura:
        \[
        S_n = 1 + 2 + 4 + \dots + 2^{n-1} = 2^n - 1 .
        \]
  \item Quantos grãos o rei prometeu ao todo? Expresse a resposta com
        uma potência de $2$ e complete a observação clássica: ``o
        tabuleiro inteiro carrega um grão a menos do que carregaria uma
        única sexagésima quinta casa''.
  \item Mostre que a segunda \omterm{def:g3:division:half}{metade} do tabuleiro (casas $33$ a $64$)
        carrega \emph{exatamente} $2^{32}$ vezes mais grãos que a
        primeira \omterm{def:g3:division:half}{metade}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Qual é o tamanho de $2^{64}$?}
A comparação $2^{10} = 1024 > 10^3$ do \cref{exo:g8:powers:11} é a
chave de todas as estimativas abaixo.
\begin{enumerate}[resume]
  \item Mostre que
        $2^{64} = 2^4 \times \left(2^{10}\right)^6 > 1.6 \times
        10^{19}$.
  \item Um grão de trigo pesa cerca de $0.05$ g, isto é,
        $5 \times 10^{-2}$ g. Mostre que o trigo prometido pesa mais
        que $8 \times 10^{17}$ g e converta isso em toneladas ($1$
        tonelada $= 10^6$ g).
  \item O mundo inteiro colhe hoje cerca de $8 \times 10^8$ toneladas
        de trigo por ano. Quantos anos de colheita mundial, no mínimo,
        o rei prometeu?
  \item Encontre o menor número inteiro $n$ tal que $2^n > 10^6$ ---
        isto é, de quantas duplicações se precisa para passar de um
        milhão. (Calcule $2^{19}$ e $2^{20}$ exatamente, usando
        $2^{10} = 1024$.)
  \item Um espertalhão oferece a você um salário de um mês: $1$ centavo
        no dia $1$ e depois o dobro do pagamento do dia anterior a cada
        dia. Em que dia o pagamento \emph{diário} sozinho passa pela
        primeira vez de um milhão de reais ($10^8$ centavos)? (Calcule
        $2^{26}$ e $2^{27}$ exatamente.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Pesos binários.}
Uma comerciante tem cinco pesos: $1$, $2$, $4$, $8$ e $16$ gramas, um
de cada. Ela põe alguns deles num prato da balança para pesar
mercadorias no outro prato.
\begin{enumerate}[resume]
  \item Que pesos ela põe para pesar $21$ g? E para pesar $27$ g?
  \item Explique por que qualquer alvo de $16$ g ou mais \emph{tem}
        de usar o peso de $16$ g, e por que qualquer alvo de $15$ g ou
        menos \emph{não} pode usá-lo. (A questão 3 diz até onde os
        pesos $1, 2, 4, 8$ chegam.) Explique por que o mesmo raciocínio
        se repete com o peso seguinte, em cada etapa.
  \item Deduza que todo alvo inteiro de $1$ a $31$ g pode ser pesado, e
        \emph{de um único jeito}: cada número entre $1$ e $31$ é \omterm{def:g1:addition:def}{soma}
        de potências de $2$ distintas de uma única maneira.
  \item A comerciante compra um sexto peso, de $32$ g. Até que alvo ela
        consegue pesar agora? Escreva $45$ g como \omterm{def:g1:addition:def}{soma} de potências de
        $2$ distintas.
  \item Num computador, um número de ``64 bits'' é guardado em $64$
        casas, cada uma com um $0$ ou um $1$ --- a casa $k$
        contribuindo com $2^{k-1}$ quando contém um $1$, como os grãos
        da lenda. Usando a Parte I, explique por que os números
        inteiros que uma máquina dessas consegue guardar vão exatamente
        de $0$ a $2^{64} - 1$.
\end{enumerate}
\end{problem}
