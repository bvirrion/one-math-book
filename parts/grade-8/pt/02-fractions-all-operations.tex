\chapter{Frações: as quatro operações}\label{ch:g8:fractions}

Este capítulo completa a aritmética das frações: adição e subtração com
\omterm{def:g4:fractions:def}{denominadores} \emph{quaisquer}, multiplicação e --- a novidade --- a
divisão, que se revela uma multiplicação disfarçada. Os sinais do
\cref{ch:g8:negprod} vêm junto na carona.

\section{Somar com denominadores quaisquer}

\begin{method}[Denominador comum]\label{met:g8:fractions:add}
Para somar ou subtrair $\frac ab$ e $\frac cd$:
\begin{enumerate}
  \item encontre um \omterm{def:g5:division:multiple}{múltiplo} comum de $b$ e $d$ (o \omterm{def:g2:mult:def}{produto}
        $b \times d$ sempre serve; um menor poupa trabalho);
  \item reescreva as duas frações com esse \omterm{def:g4:fractions:def}{denominador};
  \item some ou subtraia os \omterm{def:g4:fractions:def}{numeradores}; simplifique.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:fractions:add}
Calcule $\dfrac{5}{6} + \dfrac{3}{4}$. Um \omterm{def:g5:division:multiple}{múltiplo} comum de $6$ e $4$ é
$12$:
\[
\frac{5}{6} = \frac{10}{12},
\qquad
\frac{3}{4} = \frac{9}{12},
\qquad
\frac{5}{6} + \frac{3}{4} = \frac{10 + 9}{12} = \frac{19}{12}.
\]
Com sinais: $\dfrac{2}{3} - \dfrac{7}{5} = \dfrac{10}{15} -
\dfrac{21}{15} = -\dfrac{11}{15}$.
\end{example}

\section{Multiplicar frações}

\begin{theorem}[Produto de frações]\label{thm:g8:fractions:mult}
\[
\frac{a}{b} \times \frac{c}{d} = \frac{a \times c}{b \times d} .
\]
\end{theorem}

\begin{proof}[Por quê, num desenho]
Tome $\frac34$ de $\frac25$ de um quadrado: corte o quadrado em $5$
faixas verticais e fique com $2$; corte na horizontal em $4$ e fique
com $3$ fileiras do que sobrou. A parte guardada é uma malha de
$3 \times 2$ casinhas em $4 \times 5$: \omterm{def:g6:fractions:def}{fração} $\frac{6}{20}$.
\end{proof}

\begin{omfigure}
\begin{tikzpicture}[scale=0.68]
  \draw[gray!50, thin] (0,0) grid[xstep=1, ystep=1] (5,4);
  \fill[omDef!18] (0,0) rectangle (2,4);
  \fill[omThm!35] (0,0) rectangle (2,3);
  \draw[thick] (0,0) rectangle (5,4);
  \draw[omDef, thick] (2,0) -- (2,4);
  \draw[omThm, thick] (0,3) -- (5,3);
  \node[below, font=\small, omDef] at (1,0) {$\frac25$ guardados};
  \node[right, font=\small, omThm] at (5.05,1.5) {$\frac34$ disso};
\end{tikzpicture}

{\small $\frac34 \times \frac25$: a região pintada duas vezes tem
$3 \times 2 = 6$ casinhas em $4 \times 5 = 20$, isto é,
$\frac{6}{20} = \frac{3}{10}$ --- \omterm{def:g4:fractions:def}{numeradores} multiplicados,
\omterm{def:g4:fractions:def}{denominadores} multiplicados.}
\end{omfigure}

\begin{example}\label{ex:g8:fractions:mult}
Simplifique \emph{antes} de multiplicar, cortando os fatores comuns:
\[
\frac{7}{12} \times \frac{9}{14}
= \frac{7 \times 9}{12 \times 14}
= \frac{7 \times 3 \times 3}{3 \times 4 \times 2 \times 7}
= \frac{3}{8}
\]
(o fator $7$ e um fator $3$ se cancelam entre o de cima e o de baixo).
Com sinais, as regras do \cref{thm:g8:negprod:rules} valem primeiro:
$\left(-\frac23\right) \times \left(-\frac{5}{4}\right) =
+\frac{10}{12} = \frac56$.
\end{example}

\section{Dividir frações}

\begin{definition}[Inverso]\label{def:g8:fractions:inverse}
O \emph{inverso}\index{inverso} de um número não nulo $x$ é o número
que, multiplicado por $x$, dá $1$. O inverso de $\frac ab$ é
$\frac ba$, pois $\frac ab \times \frac ba = \frac{ab}{ab} = 1$; o
inverso de um número inteiro $n$ é $\frac1n$.
\end{definition}

\begin{theorem}[Dividir é multiplicar pelo inverso]\label{thm:g8:fractions:div}
Para $c \neq 0$:
\[
\frac{a}{b} \div \frac{c}{d} = \frac{a}{b} \times \frac{d}{c} .
\]
\end{theorem}

\begin{proof}
Por definição, o \omterm{def:g3:division:remainder}{quociente} $q$ de $\frac ab$ por $\frac cd$ é o número
com $q \times \frac cd = \frac ab$. Confira o candidato
$q = \frac ab \times \frac dc$:
\[
\left(\frac ab \times \frac dc\right) \times \frac cd
= \frac ab \times \left(\frac dc \times \frac cd\right)
= \frac ab \times 1 = \frac ab . \qedhere
\]
\end{proof}

\begin{example}\label{ex:g8:fractions:div}
\[
\frac{3}{5} \div \frac{7}{10}
= \frac{3}{5} \times \frac{10}{7}
= \frac{3 \times 10}{5 \times 7}
= \frac{30}{35} = \frac{6}{7}.
\]
Uma conferência de bom senso: dividir por $\frac{7}{10}$, um número
menor que $1$, tem de dar um resultado \emph{maior} que $\frac35$ ---
e $\frac67 > \frac35$ de fato ($\frac{30}{35} > \frac{21}{35}$).
\end{example}

\begin{example}[Barras de fração dentro de barras de fração]\label{ex:g8:fractions:complex}
Uma \omterm{def:g6:fractions:def}{fração} ``de dois andares'' é apenas uma divisão escrita na
vertical:
\[
\frac{\ \frac{2}{3}\ }{\ \frac{5}{6}\ }
= \frac{2}{3} \div \frac{5}{6}
= \frac{2}{3} \times \frac{6}{5}
= \frac{12}{15} = \frac{4}{5}.
\]
Localize primeiro a barra \emph{principal} (a mais comprida) e depois
aplique o \cref{thm:g8:fractions:div}.
\end{example}

\begin{method}[Cálculos mistos]\label{met:g8:fractions:mixed}
Numa expressão que mistura frações e as quatro operações:
\begin{enumerate}
  \item aplique as prioridades de sempre (\cref{ch:g7:priorities});
  \item converta toda divisão numa multiplicação pelo \omterm{def:g8:fractions:inverse}{inverso};
  \item decida os sinais primeiro (\cref{met:g8:negprod:compute}) e
        depois multiplique ou procure \omterm{def:g4:fractions:def}{denominadores} comuns;
  \item simplifique na primeira oportunidade.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:fractions:mixed}
\begin{align*}
\frac12 + \frac{2}{3} \times \frac{9}{4}
&= \frac12 + \frac{2 \times 9}{3 \times 4}
&& \text{(o produto primeiro!)} \\
&= \frac12 + \frac{18}{12} = \frac12 + \frac32
= \frac{4}{2} = 2 .
\end{align*}
\end{example}

\begin{exercise}[$\star$]\label{exo:g8:fractions:1}
Calcule e simplifique:
\[
\frac{3}{4} + \frac{2}{5}, \qquad
\frac{5}{6} - \frac{3}{8}, \qquad
\frac{7}{10} + \frac{8}{15} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:2}
Calcule (os sinais primeiro):
\[
-\frac{2}{7} + \frac{3}{14}, \qquad
\frac{1}{6} - \frac{5}{4}, \qquad
-\frac{3}{5} - \frac{1}{2} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:3}
Calcule, simplificando antes de multiplicar:
\[
\frac{5}{8} \times \frac{4}{15}, \qquad
\frac{9}{14} \times \frac{7}{3}, \qquad
\left(-\frac{6}{5}\right) \times \frac{10}{9} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:4}
Dê o \omterm{def:g8:fractions:inverse}{inverso} de: $\dfrac{3}{7}$; \quad $5$; \quad
$\dfrac{1}{4}$; \quad $-\dfrac{2}{9}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:5}
Calcule:
\[
\frac{4}{9} \div \frac{2}{3}, \qquad
\frac{7}{5} \div 14, \qquad
6 \div \frac{3}{4} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:6}
Calcule as frações de dois andares:
\[
\frac{\ \frac{3}{4}\ }{\ \frac{9}{8}\ },
\qquad
\frac{\ \frac{5}{6}\ }{\ 10\ },
\qquad
\frac{\ 2\ }{\ \frac{4}{7}\ } .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:7}
Calcule, respeitando as prioridades:
\[
\frac{1}{3} + \frac{1}{2} \times \frac{4}{5},
\qquad
\left(\frac{1}{3} + \frac{1}{2}\right) \times \frac{4}{5} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:fractions:8}
Um tanque está com $\frac{2}{5}$ da \omterm{def:g2:measure:units}{capacidade}. Acrescenta-se
$\frac{1}{3}$ \emph{da \omterm{def:g2:measure:units}{capacidade} do tanque}. Que \omterm{def:g6:fractions:def}{fração} do tanque
está cheia agora? E que \omterm{def:g6:fractions:def}{fração} continua vazia?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:fractions:9}
Três quartos dos alunos de uma turma passaram numa prova; entre eles,
dois terços tiraram mais de $14$ em $20$. Que \omterm{def:g6:fractions:def}{fração} da turma tirou
mais de $14$? Se isso representa $12$ alunos, qual é o tamanho da
turma?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:fractions:10}
Uma corda de $\frac{15}{2}$ m tem de ser cortada em pedaços de
$\frac{3}{4}$ m cada. Quantos pedaços se obtêm? (Uma divisão de
frações --- confira que a resposta é um número inteiro.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:fractions:11}
Calcule
\[
E = \frac{2}{3} - \frac{5}{6} \div \frac{10}{9},
\qquad
F = \left(-\frac{1}{2}\right)^2 \times \frac{8}{3} .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:fractions:12}

\]

\end{exercise}

\section{Problema: frações egípcias}

\begin{problem}[{Problema de fim de semana --- toda fração é uma soma
de frações unitárias distintas}]
\label{pb:g8:fractions:1}
Uma \emph{\omterm{def:g6:fractions:def}{fração} unitária} é uma \omterm{def:g6:fractions:def}{fração} de \omterm{def:g4:fractions:def}{numerador} $1$. Os escribas
do Egito antigo --- o papiro de Rhind foi copiado por volta de
$1550$~a.C. --- escreviam toda \omterm{def:g6:fractions:def}{fração} como \omterm{def:g1:addition:def}{soma} de frações unitárias
\emph{distintas}: nunca $\frac27 = \frac17 + \frac17$, e sim
$\frac27 = \frac14 + \frac{1}{28}$. Uma \omterm{def:g1:addition:def}{soma} dessas chama-se
\emph{escrita egípcia} da \omterm{def:g6:fractions:def}{fração}. Este problema desenvolve as técnicas
de adição e comparação deste capítulo (\cref{met:g8:fractions:add}) num
método --- publicado por Fibonacci em $1202$ --- que produz uma escrita
egípcia de qualquer \omterm{def:g6:fractions:def}{fração} entre $0$ e $1$, e termina com um célebre
quebra-cabeça de herança.

\medskip
\noindent\textbf{Parte I --- Frações unitárias e a identidade de
divisão.}
\begin{enumerate}
  \item Calcule $\frac12 + \frac13$ e $\frac12 + \frac14$ e deduza
        escritas egípcias de $\frac56$ e de $\frac34$.
  \item Verifique a entrada dos escribas para $\frac27$: mostre que
        $\frac14 + \frac{1}{28} = \frac27$.
  \item Confira que $\frac13 + \frac16 = \frac12$ e que
        $\frac14 + \frac{1}{12} = \frac13$. Depois demonstre a
        \emph{identidade de divisão}: para todo número inteiro
        $n \geq 1$,
        \[
        \frac{1}{n} = \frac{1}{n+1} + \frac{1}{n(n+1)} .
        \]
  \item Deduza uma escrita egípcia do próprio $1$, como \omterm{def:g1:addition:def}{soma} de três
        frações unitárias distintas.
  \item Divida o último termo de $\frac56 = \frac12 + \frac13$ para
        obter uma \emph{segunda} escrita egípcia de $\frac56$, com três
        frações unitárias. Explique por que nenhuma \omterm{def:g6:fractions:def}{fração} tem escrita
        egípcia única.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O método guloso de Fibonacci.}
O método é: subtrair a \emph{maior} \omterm{def:g6:fractions:def}{fração} unitária que couber e
repetir com o que sobrar.
\begin{enumerate}[resume]
  \item Usando \omterm{def:g4:fractions:def}{denominadores} comuns, mostre que
        $\frac14 < \frac27 < \frac13$. Por que isso demonstra que
        $\frac14$ é a maior \omterm{def:g6:fractions:def}{fração} unitária menor que $\frac27$?
  \item Calcule $\frac27 - \frac14$. Que escrita de $\frac27$ o método
        guloso produz?
  \item Agora tome $\frac57$. Mostre que a maior \omterm{def:g6:fractions:def}{fração} unitária menor
        que $\frac57$ é $\frac12$ e calcule $\frac57 - \frac12$.
  \item Mostre que a maior \omterm{def:g6:fractions:def}{fração} unitária menor que $\frac{3}{14}$ é
        $\frac15$ (compare $\frac{3}{14}$ com $\frac14$ e com
        $\frac15$), calcule $\frac{3}{14} - \frac15$ e conclua:
        \[
        \frac57 = \frac12 + \frac15 + \frac{1}{70} .
        \]
        Verifique essa escrita diretamente com o \omterm{def:g4:fractions:def}{denominador} comum
        $70$.
  \item Rode o método guloso em $\frac45$, conferindo a cada passo que
        a sua \omterm{def:g6:fractions:def}{fração} unitária é a maior que cabe, e verifique a escrita
        final.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Por que o método sempre para, e o
décimo oitavo camelo.}
\begin{enumerate}[resume]
  \item Nas questões 7 a 10, o método foi aplicado a $\frac27$,
        $\frac57$ e $\frac45$. Para cada uma, liste os \omterm{def:g4:fractions:def}{numeradores} dos
        \omterm{def:g3:division:remainder}{restos} sucessivos (antes de qualquer simplificação). O que você
        observa?
  \item Para uma \omterm{def:g6:fractions:def}{fração} $\frac ab$ e um número inteiro $n$, mostre que
        \[
        \frac{a}{b} - \frac{1}{n}
        = \frac{a \times n - b}{b \times n} .
        \]
  \item Suponha que o método guloso subtraia $\frac1n$ de $\frac ab$:
        então $\frac1n$ cabe, mas $\frac{1}{n-1}$ já é grande demais,
        $\frac{1}{n-1} > \frac ab$. Pondo os dois sobre o \omterm{def:g4:fractions:def}{denominador}
        comum $b \times (n - 1)$, deduza que $b > a \times n - a$ e,
        portanto, que o novo \omterm{def:g4:fractions:def}{numerador} satisfaz
        \[
        a \times n - b < a .
        \]
        Explique por que isso demonstra que o método sempre para.
  \item Uma velha história: um pai morre deixando $17$ camelos e lega
        $\frac12$ do rebanho ao filho mais velho, $\frac13$ ao segundo
        e $\frac19$ ao terceiro --- e nenhuma das partes é um número
        inteiro de camelos. Um vizinho empresta à família um décimo
        oitavo camelo; os filhos levam $9$, $6$ e $2$ camelos, e o
        camelo emprestado é devolvido. Calcule
        $\frac12 + \frac13 + \frac19$ e explique o truque: por que o
        rebanho pôde ser repartido em camelos inteiros, e por que
        nenhum filho recebeu menos do que o testamento prometia?
  \item Conserte o testamento: proponha três frações unitárias
        \emph{distintas} que somem exatamente $1$ e dê o menor tamanho
        de rebanho para o qual as três partes sejam números inteiros de
        camelos.
\end{enumerate}
\end{problem}
