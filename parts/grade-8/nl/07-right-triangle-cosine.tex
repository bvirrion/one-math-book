\chapter{De rechthoekige driehoek en de cosinus}\label{ch:g8:cosine}

Twee mooie feiten binden de \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} aan de cirkel: een
\omterm{def:g6:shapes:triangles}{rechthoekige driehoek} past precies in een halve cirkel, met zijn schuine zijde
als \omterm{def:g4:geometry:circle}{diameter}. En één getal, de \emph{cosinus} van een hoek, legt de vorm vast van
elke \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} met die hoek --- de eerste goniometrische verhouding,
nog vóór haar zusjes sinus en tangens (\cref{ch:g9:trig}).

\section{De rechthoekige driehoek en zijn cirkel}

\begin{theorem}[Cirkelstelling]\label{thm:g8:cosine:circle}
\begin{enumerate}
  \item Is een driehoek $ABC$ \omterm{def:g6:shapes:triangles}{rechthoekig} in $A$, dan ligt $A$ op de cirkel met
        de schuine zijde $[BC]$ als \omterm{def:g4:geometry:circle}{diameter}.
  \item Omgekeerd: ligt $A$ op een cirkel met \omterm{def:g4:geometry:circle}{diameter} $[BC]$ (met
        $A \neq B, C$), dan is de driehoek $ABC$ \omterm{def:g6:shapes:triangles}{rechthoekig} in $A$.
\end{enumerate}
\omterm{def:g3:numbers:evenodd}{Even} goed gezegd: in een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} ligt het midden van de schuine
zijde op gelijke afstand van de drie \omterm{def:g5:solids:def}{hoekpunten} --- de \omterm{pb:g8:midpoints:1}{zwaartelijn} uit de \omterm{def:g3:shapes:rightangle}{rechte
hoek} meet de \omterm{def:g3:division:half}{helft} van de schuine zijde.
\end{theorem}

\begin{proof}[Bewijs van 1]
Zij $O$ het midden van $[BC]$ en $D$ het beeld van $A$ door de \omterm{def:g7:central:def}{puntspiegeling} om
$O$. De diagonalen van $ABDC$ snijden elkaar in hun gemeenschappelijke midden
$O$, dus is $ABDC$ een \omterm{def:g7:central:parallelogram}{parallellogram} (\cref{def:g7:central:parallelogram}) ---
met een \omterm{def:g3:shapes:rightangle}{rechte hoek} in $A$: het is een rechthoek. De diagonalen van een rechthoek
zijn gelijk, dus is $OA = \frac{AD}{2} = \frac{BC}{2}$: het punt $A$ ligt op
afstand $\frac{BC}{2}$ van $O$, dat wil zeggen op de cirkel met \omterm{def:g4:geometry:circle}{diameter} $[BC]$.
(Punt 2 bewijs je door het betoog in omgekeerde richting te doorlopen.)
\end{proof}

\begin{example}\label{ex:g8:cosine:circle}
Een driehoek heeft een schuine zijde van $10$ cm. Zonder verder iets te weten:
de \omterm{pb:g8:midpoints:1}{zwaartelijn} uit de \omterm{def:g3:shapes:rightangle}{rechte hoek} meet $5$ cm, en de \omterm{pb:g6:symmetry:1}{omgeschreven cirkel} van de
driehoek heeft \omterm{def:g3:shapes:shapes}{radius} $5$ cm, met het midden van de schuine zijde als middelpunt.
\end{example}

\section{De cosinus van een scherpe hoek}

\begin{definition}[Cosinus]\label{def:g8:cosine:def}
In een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} geldt voor een scherpe hoek $\theta$:
\[
\cos\theta = \frac{\text{aanliggende zijde bij } \theta}{\text{schuine zijde}}
\]
--- de rechthoekszijde die aan $\theta$ raakt, gedeeld door de schuine zijde.
Deze verhouding hangt alleen van de hoek af, niet van de grootte van de driehoek:
alle \omterm{def:g6:shapes:triangles}{rechthoekige driehoeken} met dezelfde scherpe hoek zijn vergrotingen van
elkaar, en vergrotingen behouden verhoudingen van lengten
(\cref{ch:g8:midpoints} begon dit verhaal; \cref{ch:g9:thales} maakt het af).
\end{definition}

\begin{proposition}[Eerste waarden en grenzen]\label{prop:g8:cosine:bounds}
Voor elke scherpe hoek $\theta$ geldt $0 < \cos\theta < 1$, en de cosinus
\emph{daalt} als de hoek verder opengaat: een wijdere hoek heeft een kleinere
cosinus. Herkenningspunten: $\cos 0^\circ = 1$, $\cos 60^\circ = \frac12$,
$\cos 90^\circ = 0$ (de uiterste waarden horen bij platgedrukte driehoeken).
\end{proposition}
\admitted

\begin{method}[De cosinus gebruiken]\label{met:g8:cosine:use}
In een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek}, wanneer de bekende en de gevraagde grootheden een
scherpe hoek, zijn aanliggende zijde en de schuine zijde zijn:
\begin{enumerate}
  \item schrijf de definitie op:
        $\cos\theta = \frac{\text{aanliggend}}{\text{schuine zijde}}$, met de
        bekende waarden ingevuld;
  \item los de kleine \omterm{def:g8:equations:def}{vergelijking} op naar de onbekende (vermenigvuldig of deel);
  \item is de \emph{hoek} onbekend, pas dan de $\cos^{-1}$ van de rekenmachine
        toe op de berekende verhouding;
  \item controles op het gezond verstand: een lengte moet korter uitkomen dan de
        schuine zijde; een hoek strikt tussen $0^\circ$ en $90^\circ$.
\end{enumerate}
\end{method}

\begin{example}[Een zijde vinden]\label{ex:g8:cosine:side}
Een helling van $6$ m maakt een hoek van $20^\circ$ met de horizontale grond.
De afgelegde horizontale afstand (aanliggend bij $20^\circ$, schuine zijde
$6$ m):
\[
\cos 20^\circ = \frac{d}{6}
\quad\Longrightarrow\quad
d = 6 \cos 20^\circ \approx 6 \times 0.940 \approx 5.6 \text{ m}.
\]
\end{example}

\begin{example}[Een hoek vinden]\label{ex:g8:cosine:angle}
In een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} meet de zijde die aan de hoek $\theta$ ligt $3.5$ en
de schuine zijde $5$:
\[
\cos\theta = \frac{3.5}{5} = 0.7,
\qquad
\theta = \cos^{-1}(0.7) \approx 45.6^\circ .
\]
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g8:cosine:1}
Een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} heeft een schuine zijde van $13$ cm. Hoe lang is de
\omterm{pb:g8:midpoints:1}{zwaartelijn} uit het \omterm{def:g5:solids:def}{hoekpunt} met de \omterm{def:g3:shapes:rightangle}{rechte hoek}? Wat is de \omterm{def:g3:shapes:shapes}{radius} van de cirkel
door de drie \omterm{def:g5:solids:def}{hoekpunten}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:2}
Teken een cirkel met \omterm{def:g4:geometry:circle}{diameter} $8$ cm en daarin een \omterm{def:g4:geometry:circle}{diameter} $[BC]$, kies een
willekeurig punt $A$ op de cirkel en teken $ABC$. Welke hoek is recht? Noem de
stelling. Meet $AB$ en $AC$ en ga Pythagoras na.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:3}
Benoem in een driehoek $DEF$, \omterm{def:g6:shapes:triangles}{rechthoekig} in $D$, de zijde die aan de hoek
$\widehat E$ ligt, de zijde tegenover die hoek, en de schuine zijde. Schrijf
$\cos \widehat E$ als een verhouding.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:4}
Bereken de ontbrekende grootheid ($\cos 35^\circ \approx 0.819$,
$\cos 50^\circ \approx 0.643$):
\begin{enumerate}
  \item schuine zijde $10$, hoek $35^\circ$: aanliggende zijde?
  \item aanliggende zijde $6$, hoek $50^\circ$: schuine zijde?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:5}
In een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} meet de zijde die aan $\theta$ ligt $8$ en de
schuine zijde $10$. Bereken $\cos\theta$, en daarna $\theta$
($\cos^{-1}(0.8) \approx 36.9^\circ$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:6}
Leg uit waarom $\cos\theta$ nooit $1.2$ kan zijn voor een scherpe hoek van een
\omterm{def:g6:shapes:triangles}{rechthoekige driehoek}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:7}
Een tokkelbaan van $25$ m daalt van een platform naar de grond en maakt een hoek
van $12^\circ$ met de horizontale ($\cos 12^\circ \approx 0.978$). Welke
horizontale afstand overspant hij?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:8}
Zet op volgorde zonder rekenmachine: $\cos 20^\circ$, $\cos 70^\circ$,
$\cos 45^\circ$ (gebruik \cref{prop:g8:cosine:bounds}). Ga het daarna na met de
rekenmachine.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:9}
Een ladder met lengte $L$ staat tegen een muur, met een hoek $\theta$ tussen
ladder en \emph{grond}. Zijn voet staat $1.2$ m van de muur en
$\theta = 68^\circ$ ($\cos 68^\circ \approx 0.375$). Bereken $L$, en daarna de
bereikte hoogte (Pythagoras).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:10}
$[BC]$ is een \omterm{def:g4:geometry:circle}{diameter} van een cirkel met middelpunt $O$ en \omterm{def:g3:shapes:shapes}{radius} $4.5$ cm, en
$A$ is een punt van de cirkel met $AB = 5.4$ cm.
\begin{enumerate}
  \item Waarom is $ABC$ \omterm{def:g6:shapes:triangles}{rechthoekig} in $A$?
  \item Bereken $AC$, en daarna $\cos \widehat B$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:cosine:11}
Verantwoord met de \omterm{def:g3:division:half}{helft} van een \omterm{def:g6:shapes:triangles}{gelijkzijdige driehoek} met zijde $1$ (geknipt
langs een hoogtelijn) het herkenningspunt $\cos 60^\circ = \frac12$ van
\cref{prop:g8:cosine:bounds}. (Waar komt de voet van de hoogtelijn op de basis
terecht? De volledige goniometrische versie is \cref{ex:g9:trig:special}.)
\end{exercise}

\section{Opgave: de betrekkingen van Euclides, en Pythagoras nog een keer}

\begin{problem}[{Weekendopgave --- de hoogtelijn van een rechthoekige driehoek:
$AB^2 = BH \times BC$, $AH^2 = BH \times HC$, een tweede bewijs van Pythagoras,
en een machine om vierkantswortels te construeren}]
\label{pb:g8:cosine:1}
Laat uit de \omterm{def:g3:shapes:rightangle}{rechte hoek} van een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} de loodlijn op de schuine
zijde neer: dat korte \omterm{def:g6:lines:objects}{lijnstuk} --- de \emph{hoogtelijn} --- voldoet aan
betrekkingen die zo nuttig zijn dat Euclides ze in het hart van zijn
\emph{Elementen} plaatste. In deze opgave bewijst de bepalende eigenschap van de
cosinus, ``de verhouding hangt alleen van de hoek af''
(\cref{def:g8:cosine:def}), ze alle; onderweg bewijs je de stelling van
Pythagoras opnieuw, langs een volstrekt andere weg, en aan het einde staat een
machine met passer en lat die $\sqrt2$, $\sqrt5$ en $\sqrt{n}$ voor elk heel
getal $n$ construeert.

Overal is $ABC$ een driehoek die \omterm{def:g6:shapes:triangles}{rechthoekig} is in $A$, en is $H$ de voet van de
loodlijn uit $A$ op de schuine zijde $[BC]$, zodat $H$ tussen $B$ en $C$ ligt en
$AH \perp BC$.

\medskip
\noindent\textbf{Deel I --- Één hoek, twee driehoeken.}
\begin{enumerate}
  \item Toon met de hoekensom van een driehoek (\cref{thm:g7:triangles:sum}) in
        $ABH$ en in $ABC$ aan dat $\widehat{BAH} = \widehat{ACB}$: de hoogtelijn
        verdeelt de \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} in twee kleinere driehoeken met
        \emph{dezelfde hoeken} als de oorspronkelijke.
  \item De driehoeken $ABC$ (\omterm{def:g6:shapes:triangles}{rechthoekig} in $A$) en $ABH$ (\omterm{def:g6:shapes:triangles}{rechthoekig} in $H$)
        hebben de hoek $\widehat B$ gemeen. Schrijf $\cos\widehat B$ in elk van
        beide op en leid uit \cref{def:g8:cosine:def} af dat
        \[
        AB^2 = BH \times BC .
        \]
        (Om van gelijke verhoudingen naar gelijke \omterm{def:g2:mult:def}{producten} te gaan,
        vermenigvuldig je beide leden met beide \omterm{def:g4:fractions:def}{noemers},
        \cref{thm:g8:equations:balance}.)
  \item Formuleer en bewijs de tegenhanger in het \omterm{def:g5:solids:def}{hoekpunt} $C$:
        \[
        AC^2 = CH \times CB .
        \]
  \item Neem de $3$--$4$--$5$-driehoek: $AB = 3$, $AC = 4$, $BC = 5$. Bereken
        $BH$ en $CH$ met vragen 2 en 3, en ga na dat $BH + HC = BC$.
  \item Druk $BH$ en $CH$ in het algemeen uit als \omterm{def:g6:fractions:def}{breuken} waarin alleen de drie
        zijden voorkomen. In welke zin verdeelt de voet $H$ de schuine zijde
        ``\omterm{def:g6:propdata:def}{evenredig} met de kwadraten van de rechthoekszijden''?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Pythagoras opnieuw, en de hoogtelijn.}
\begin{enumerate}[resume]
  \item Tel de betrekkingen van vragen 2 en 3 op en gebruik $BH + HC = BC$ om te
        komen tot
        \[
        AB^2 + AC^2 = BC^2 :
        \]
        de stelling van Pythagoras (\cref{thm:g8:pythagoras:direct}), opnieuw
        bewezen --- zonder ook maar één oppervlakteraadsel in zicht (vergelijk
        \cref{exo:g8:pythagoras:11}).
  \item Pas Pythagoras toe in de kleine driehoek $ABH$ om $AH^2 = AB^2 - BH^2$ te
        schrijven, vervang $AB^2$ door $BH \times BC$, en breng $BH$ buiten
        haakjes om de \emph{hoogtelijnbetrekking van Euclides} te bewijzen:
        \[
        AH^2 = BH \times HC .
        \]
  \item Bereken de \omterm{def:g6:measure:area}{oppervlakte} van $ABC$ op twee manieren --- met de
        rechthoekszijden als basis en hoogte, en daarna met de schuine zijde als
        basis (\cref{thm:g7:areas:triangle}) --- en leid de derde betrekking af:
        \[
        AB \times AC = BC \times AH .
        \]
  \item Terug naar de $3$--$4$--$5$-driehoek: bereken $AH$ met vraag 8, en ga
        daarna de hoogtelijnbetrekking van vraag 7 met getallen na, met de in
        vraag 4 gevonden waarden van $BH$ en $CH$.
  \item De voet van de hoogtelijn verdeelt de schuine zijde van een zekere
        \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} in de stukken $BH = 2$ cm en $HC = 8$ cm. Bereken
        de hoogtelijn $AH$.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Een machine voor vierkantswortels.}
Teken een \omterm{def:g6:lines:objects}{lijnstuk} $[BC]$ dat uit twee stukken achter elkaar bestaat: $BH = p$ en
$HC = q$. Teken de halve cirkel met \omterm{def:g4:geometry:circle}{diameter} $[BC]$, en zij $A$ het punt waar de
loodlijn op $(BC)$ in $H$ die halve cirkel ontmoet.
\begin{enumerate}[resume]
  \item Leg met \cref{thm:g8:cosine:circle} uit waarom de driehoek $ABC$
        \omterm{def:g6:shapes:triangles}{rechthoekig} is in $A$ --- zodat vraag 7 van toepassing is en
        \[
        AH^2 = p \times q .
        \]
        (De lengte $AH$ heet het \emph{meetkundig
        gemiddelde}\index{meetkundig gemiddelde} van $p$ en $q$.)
  \item Neem $p = 1$ en $q = 2$. Wat is $AH^2$? Welke beroemde lengte uit
        \cref{ex:g8:pythagoras:diagonal} heeft je figuur met passer en lat net
        geconstrueerd?
  \item Beschrijf het recept dat een \omterm{def:g6:lines:objects}{lijnstuk} met lengte $\sqrt n$ construeert
        voor elk heel getal $n \geq 1$, en zeg welke keuze van $p$ en $q$
        $\sqrt 5$ construeert.
  \item Zij $O$ het middelpunt van de halve cirkel. Leg uit waarom $AH$ nooit
        groter kan zijn dan de \omterm{def:g3:shapes:shapes}{radius} $\frac{p + q}{2}$, en voor welke stand van
        $H$ de twee gelijk zijn.
  \item Leid de ongelijkheid af: voor alle positieve getallen $p$ en $q$ geldt
        \[
        p \times q \leq \left(\frac{p + q}{2}\right)^2,
        \]
        met gelijkheid precies als $p = q$. Bewijs die daarna opnieuw zonder een
        greintje meetkunde: werk $\left(\frac{p+q}{2}\right)^2 - p q$ uit en
        herken een merkwaardig \omterm{def:g2:mult:def}{product} (\cref{pb:g8:equations:1}).
\end{enumerate}
\end{problem}
