\chapter{Negatieve getallen vermenigvuldigen}\label{ch:g8:negprod}

In jaar 7 telden we getallen met een teken op en trokken we ze af
(\cref{ch:g7:negatives}); nu moeten we ze nog vermenigvuldigen en delen. Eén vraag
overheerst dit hoofdstuk: waarom zou ``min keer min'' plus zijn? We geven een reden,
niet alleen een regel.

\section{De tekenregels}

\begin{theorem}[Teken van een product]\label{thm:g8:negprod:rules}
Het \omterm{def:g2:mult:def}{product} van twee getallen met een teken heeft als afstand tot \omterm{def:g1:counting:zero}{nul} het \omterm{def:g2:mult:def}{product} van
de afstanden, en zijn teken staat in deze tabel:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|cc}
$\times$ & $+$ & $-$ \\
\hline
$+$ & $+$ & $-$ \\
$-$ & $-$ & $+$ \\
\end{tabular}
\end{center}
\emph{Dezelfde tekens: een positief \omterm{def:g2:mult:def}{product}. \omterm{def:g7:negatives:def}{Tegengestelde} tekens: een negatief
\omterm{def:g2:mult:def}{product}.} Dezelfde tabel geldt voor \omterm{def:g3:division:remainder}{quotiënten}.
\end{theorem}

\begin{proof}[Waarom $(-1) \times (-1) = +1$]
Ten eerste betekent $3 \times (-4)$ dat $(-4) + (-4) + (-4) = -12$: een positief
getal keer een negatief getal is negatief. Kijk nu naar het patroon:
\[
3 \times (-4) = -12, \quad
2 \times (-4) = -8, \quad
1 \times (-4) = -4, \quad
0 \times (-4) = 0 .
\]
Bij elke stap daalt de eerste factor met $1$ en \emph{stijgt} de uitkomst met $4$. Als
we het patroon één stap verder zetten:
\[
(-1) \times (-4) = 0 + 4 = +4 .
\]
Elke andere keuze zou de regelmaat van de rekenkunde (de distributiviteit) breken.
Min keer min moet dus plus zijn.
\end{proof}

\begin{example}\label{ex:g8:negprod:products}
\begin{align*}
(-7) \times (+6) &= -42 && \text{(tegengestelde tekens)} \\
(-5) \times (-8) &= +40 && \text{(dezelfde tekens)} \\
(+2.5) \times (-4) &= -10 \\
(-36) \div (-9) &= +4 \\
(+15) \div (-2) &= -7.5 .
\end{align*}
\end{example}

\section{Producten van meerdere factoren}

\begin{proposition}[Teken van een lang product]\label{prop:g8:negprod:many}
Het teken van een \omterm{def:g2:mult:def}{product} van meerdere factoren (geen \omterm{def:g1:counting:zero}{nul}) hangt alleen af van het
aantal negatieve factoren:
\begin{itemize}
  \item een \emph{\omterm{def:g3:numbers:evenodd}{even}} aantal negatieve factoren geeft een positief \omterm{def:g2:mult:def}{product};
  \item een \emph{\omterm{def:g3:numbers:evenodd}{oneven}} aantal geeft een negatief \omterm{def:g2:mult:def}{product}.
\end{itemize}
\end{proposition}

\begin{proof}
De negatieve factoren gaan twee aan twee samen, en elk paar levert een positief teken
op (\cref{thm:g8:negprod:rules}); bij een \omterm{def:g3:numbers:evenodd}{oneven} aantal blijft er één negatieve factor
zonder partner over, en die maakt het hele \omterm{def:g2:mult:def}{product} negatief.
\end{proof}

\begin{example}\label{ex:g8:negprod:many}
$(-2) \times (+3) \times (-5) \times (-1)$: drie negatieve factoren (\omterm{def:g3:numbers:evenodd}{oneven}), dus is
het \omterm{def:g2:mult:def}{product} negatief; de afstanden: $2 \times 3 \times 5 \times 1 = 30$. Uitkomst:
$-30$. Beslis \emph{eerst} het teken en vermenigvuldig daarna de afstanden --- twee
makkelijke klusjes in plaats van één foutgevoelige.
\end{example}

\begin{example}[Machten van negatieve getallen]\label{ex:g8:negprod:powers}
$(-2)^2 = (-2) \times (-2) = +4$, en $(-2)^3 = (-2)^2 \times (-2) = -8$: \omterm{def:g3:numbers:evenodd}{even}
exponenten geven positieve waarden, \omterm{def:g3:numbers:evenodd}{oneven} exponenten houden het teken. Pas op met de
notatie: $(-3)^2 = 9$ maar $-3^2 = -(3 \times 3) = -9$.
\end{example}

\section{Rekenen met alle vier de bewerkingen}

\begin{method}[Veilig rekenen met getallen met een teken]\label{met:g8:negprod:compute}
\begin{enumerate}
  \item Respecteer de voorrangsregels van \cref{ch:g7:priorities}: haakjes, dan
        $\times$ en $\div$, dan $+$ en $-$;
  \item zoek bij elke vermenigvuldiging of deling \emph{eerst het teken} en dan de
        afstand;
  \item schrijf één stap per regel op.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:negprod:compute}
\begin{align*}
5 - 3 \times (-4)
&= 5 - (-12) && \text{(eerst het product: tegengestelde tekens)}\\
&= 5 + 12 = 17 .
\end{align*}
\begin{align*}
\frac{(-6) \times (-10)}{-4}
&= \frac{+60}{-4} && \text{(de teller: dezelfde tekens)}\\
&= -15 && \text{(het quotiënt: tegengestelde tekens).}
\end{align*}
\end{example}

\begin{example}[Negatieve waarden invullen]\label{ex:g8:negprod:substitute}
Reken $E = 3x^2 - 2x$ uit voor $x = -5$, met haakjes om de waarde:
\[
E = 3 \times (-5)^2 - 2 \times (-5)
= 3 \times 25 - (-10)
= 75 + 10 = 85 .
\]
De haakjes in $(-5)^2$ zijn onmisbaar: zonder haakjes zou het kwadraat alleen op de
$5$ werken.
\end{example}

\begin{exercise}[$\star\star$]\label{exo:g8:negprod:9}

  \item Het \omterm{def:g2:mult:def}{product} van twee getallen met een teken is altijd minstens zo groot als

  \item Het kwadraat van een \omterm{def:g7:negatives:def}{getal met een teken} is nooit negatief.

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:negprod:11}
Werk met de distributiviteit (zoals in het bewijs van
\cref{thm:g8:negprod:rules}) elke stap van
\[
0 = (-1) \times 0 = (-1) \times \bigl(1 + (-1)\bigr)
= (-1) \times 1 + (-1) \times (-1)
\]
uit en verantwoord haar, en besluit dat $(-1) \times (-1)$ gelijk moet zijn aan $+1$.
\end{exercise}

\section{Opgave: waarom min keer min plus is}

\begin{problem}[{Weekendopgave --- de tekenregels afgeleid uit de distributiviteit,
en de alternerende som $1 - 2 + 3 - 4 + \dots$}]
\label{pb:g8:negprod:1}
Het bewijs van \cref{thm:g8:negprod:rules} zette een patroon voort en beweerde dat
``elke andere keuze de distributiviteit zou breken''. In deze opgave maak je die
bewering precies: vanuit de distributiviteit alleen
(\cref{thm:g7:priorities:distributivity}) ga je de tekenregels \emph{bewijzen} ---
geen plaatje, geen patroon, zoals de algebra het doet --- en zet je ze daarna aan het
werk op machten van $-1$ en op een beroemde \omterm{def:g1:addition:def}{som} met wisselende tekens. Overal zijn $a$
en $b$ getallen met een teken; we nemen de optelregels van \cref{ch:g7:negatives} aan,
dat $1 \times a = a$, en dat je een \omterm{def:g2:mult:def}{product} in elke volgorde mag uitrekenen (zodat de
distributiviteit aan beide kanten van een \omterm{def:g2:mult:def}{product} geldt).

\medskip
\noindent\textbf{Deel I --- De tekenregels zijn stellingen.}
\begin{enumerate}
  \item Bewijs dat $0 \times a = 0$ voor elk getal $a$ met een teken. (Schrijf
        $0 = 0 + 0$, werk $(0 + 0) \times a$ uit, en vraag je af welk getal, opgeteld
        bij $0 \times a$, weer $0 \times a$ teruggeeft.)
  \item Werk $\bigl(1 + (-1)\bigr) \times a$ uit om te laten zien dat
        $(-1) \times a + a = 0$, en besluit dat
        \[
        (-1) \times a = -a :
        \]
        met $-1$ vermenigvuldigen geeft het \omterm{def:g7:negatives:def}{tegengestelde}. Vergelijk met
        \cref{exo:g8:negprod:11}, het geval $a = -1$.
  \item Leid af dat $(-a) \times b = -(a \times b)$: één minteken komt onveranderd uit
        een \omterm{def:g2:mult:def}{product}.
  \item Leid af dat $(-a) \times (-b) = a \times b$: twee mintekens heffen elkaar op.
  \item Elk \omterm{def:g7:negatives:def}{getal met een teken} is zijn afstand tot \omterm{def:g1:counting:zero}{nul} met een teken ervoor:
        $a = +d$ of $a = -d$. Leg uit hoe de vragen 3 en~4 alle vier de gevallen van de
        tekentabel van \cref{thm:g8:negprod:rules} bewijzen, afstanden inbegrepen.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Machten van $-1$.}
\begin{enumerate}[resume]
  \item Reken $(-1)^2$, $(-1)^3$, $(-1)^4$ en $(-1)^5$ uit, en geef daarna, met een
        verantwoording vanuit \cref{prop:g8:negprod:many}, de waarde van $(-1)^n$ voor
        elk heel getal $n \geq 1$.
  \item Geef (zonder één afstand uit te rekenen) het teken van
        \[
        (-1) \times (-2) \times (-3) \times \dots \times (-10),
        \]
        en schrijf de afstand van dit \omterm{def:g2:mult:def}{product} als een \omterm{def:g2:mult:def}{product} van hele getallen, zonder
        het uit te werken.
  \item Laat zien dat $(-a)^2 = a^2$ en dat $(-a)^3 = -a^3$. Bij welke exponenten
        overleeft het minteken?
  \item Laat met de tekenregels zien dat het kwadraat van een \omterm{def:g7:negatives:def}{getal met een teken} nooit
        negatief is, en leid af dat geen enkel getal $x$ met een teken aan $x^2 = -4$
        voldoet.
  \item Zoë beweert: ``$(-1)^{2026} + (-1)^{2027} = 0$, en algemener heffen twee
        opeenvolgende machten van $-1$ elkaar altijd op.'' Heeft ze gelijk? Verantwoord
        het.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De alternerende \omterm{def:g1:addition:def}{som}.}
Met de tekenregels in de hand kunnen we sommen uitrekenen waarin beide tekens door
elkaar staan. Zij voor een heel getal $n \geq 1$
\[
A_n = 1 - 2 + 3 - 4 + \dots
\]
de \omterm{def:g1:addition:def}{som} van de hele getallen van $1$ tot $n$ met wisselende tekens: de laatste term is
$+n$ als $n$ \omterm{def:g3:numbers:evenodd}{oneven} is, en $-n$ als $n$ \omterm{def:g3:numbers:evenodd}{even} is. Dus $A_1 = 1$ en
$A_2 = 1 - 2 = -1$.
\begin{enumerate}[resume]
  \item Reken $A_3$, $A_4$, $A_5$, $A_6$ en $A_7$ uit. Wat vermoed je?
  \item Neem aan dat $n$ \omterm{def:g3:numbers:evenodd}{even} is. Groepeer de termen twee aan twee,
        $(1 - 2) + (3 - 4) + \dots$, tel de paren, en bewijs dat
        $A_n = -\frac{n}{2}$.
  \item Neem aan dat $n$ \omterm{def:g3:numbers:evenodd}{oneven} is. Leg uit waarom $A_n = A_{n-1} + n$, en leid met de
        vorige vraag af dat $A_n = \frac{n + 1}{2}$.
  \item Reken $A_{100}$, $A_{101}$ en $A_{2026}$ uit.
  \item Alma rekende $A_{101} - A_{100} = 51 - (-50) = 101$ uit en was verrast dat er
        precies $101$ uitkwam. Had ze verrast moeten zijn? Leg het in één zin uit, en
        geef de waarde van $A_{2027} - A_{2026}$ zonder verder te rekenen.
\end{enumerate}
\end{problem}
