\chapter{Piramides en kegels}\label{ch:g8:solids}

Na de \omterm{def:g7:areas:prism}{prisma}'s en \omterm{def:g7:areas:prism}{cilinders} van \cref{ch:g7:areas} --- lichamen met een constante
doorsnede --- komen de lichamen met een punt: \omterm{def:g8:solids:def}{piramides} en \omterm{def:g8:solids:def}{kegels}. Hun formule
voor de \omterm{def:g2:measure:units}{inhoud} draagt een beroemde factor $\frac13$, die dit hoofdstuk
geloofwaardig maakt en gebruikt; de studie van de lichamen gaat verder met bollen
en doorsneden in \cref{ch:g9:solids}.

\section{Piramides en kegels beschrijven}

\begin{definition}[Piramide, kegel]\label{def:g8:solids:def}
\begin{itemize}
  \item Een \emph{piramide}\index{piramide} heeft een \omterm{def:g4:geometry:polygon}{veelhoek} als
        \emph{grondvlak} en een punt, de \emph{top}, die met driehoekige
        \omterm{def:g5:solids:def}{zijvlakken} met elk \omterm{def:g5:solids:def}{hoekpunt} van het grondvlak verbonden is. Haar
        \emph{hoogte} is de afstand van de top tot het vlak van het grondvlak.
  \item Een \emph{kegel}\index{kegel} is dezelfde constructie boven een schijf:
        een grondschijf, een top, en een gekromd \omterm{def:g5:solids:def}{zijvlak}.
\end{itemize}
Een piramide is \emph{regelmatig} wanneer haar grondvlak een regelmatige \omterm{def:g4:geometry:polygon}{veelhoek}
is (vierkant, \omterm{def:g6:shapes:triangles}{gelijkzijdige driehoek}, \dots) en haar top verticaal boven het
middelpunt van het grondvlak ligt.
\end{definition}

\begin{example}[Zijvlakken, ribben en hoekpunten tellen]\label{ex:g8:solids:count}
Een \omterm{def:g8:solids:def}{piramide} met een vierkant grondvlak heeft $5$ \omterm{def:g5:solids:def}{zijvlakken} ($1$ vierkant $+ 4$
driehoeken), $8$ \omterm{def:g5:solids:def}{ribben} en $5$ \omterm{def:g5:solids:def}{hoekpunten}. Met een grondvlak met $n$ zijden:
$n + 1$ \omterm{def:g5:solids:def}{zijvlakken}, $2n$ \omterm{def:g5:solids:def}{ribben}, $n + 1$ \omterm{def:g5:solids:def}{hoekpunten}. Ga het kleine patroon van
Euler na: \omterm{def:g5:solids:def}{zijvlakken} $+$ \omterm{def:g5:solids:def}{hoekpunten} $=$ \omterm{def:g5:solids:def}{ribben} $+ 2$.
\end{example}

\section{Inhoud}

\begin{theorem}[Inhoud van een piramide of kegel]\label{thm:g8:solids:volume}
Voor een \omterm{def:g8:solids:def}{piramide} of een \omterm{def:g8:solids:def}{kegel} met grondvlak van \omterm{def:g6:measure:area}{oppervlakte} $B$ en hoogte $h$:
\[
V = \frac{1}{3}\, B \times h
\]
--- een derde van het \omterm{def:g7:areas:prism}{prisma} of de \omterm{def:g7:areas:prism}{cilinder} met hetzelfde grondvlak en dezelfde
hoogte. Voor een \omterm{def:g8:solids:def}{kegel} met grondradius $r$: $V = \frac13 \pi r^2 h$.
\end{theorem}
\admitted

\begin{remark}[Waarom een derde?]
Vul een holle \omterm{def:g8:solids:def}{piramide} met water en giet het in het \omterm{def:g7:areas:prism}{prisma} met hetzelfde
grondvlak en dezelfde hoogte: er gaan precies drie vullingen in. Nog mooier: een
\omterm{def:g5:solids:def}{kubus} kan in drie identieke \omterm{def:g8:solids:def}{piramides} worden verdeeld, die elk een \omterm{def:g5:solids:def}{zijvlak} van de
\omterm{def:g5:solids:def}{kubus} als grondvlak en één \omterm{def:g5:solids:def}{hoekpunt} van de \omterm{def:g5:solids:def}{kubus} als top hebben (probeer het je
voor te stellen!). Een algemeen bewijs gebruikt integralen, gereedschap uit het
bovenbouwvolume.
\end{remark}

\begin{example}\label{ex:g8:solids:volume}
Een \omterm{def:g8:solids:def}{piramide} heeft een vierkant grondvlak met zijde $5$ cm en hoogte $9$ cm:
\begin{enumerate}
  \item \omterm{def:g6:measure:area}{oppervlakte} van het grondvlak: $B = 5^2 = 25$ cm$^2$;
  \item \omterm{def:g2:measure:units}{inhoud}: $V = \frac13 \times 25 \times 9 = 25 \times 3 = 75$ cm$^3$.
\end{enumerate}
Een ijshoorntje met \omterm{def:g3:shapes:shapes}{radius} $3$ cm en hoogte $10$ cm:
\[
V = \frac13 \times \pi \times 3^2 \times 10 = 30\pi \approx 94
\text{ cm}^3 .
\]
\end{example}

\begin{example}[Pas op met de hoogte]\label{ex:g8:solids:slant}
Verwar bij een \omterm{def:g8:solids:def}{kegel} de hoogte $h$ (van de top tot het middelpunt van het
grondvlak, \omterm{def:g6:lines:perp}{loodrecht}) niet met de \emph{schuine hoogte} $s$ (van de top tot de
rand). Ze hangen samen via Pythagoras (\cref{thm:g8:pythagoras:direct}) in de
\omterm{def:g6:shapes:triangles}{rechthoekige driehoek} top--middelpunt--rand:
\[
s^2 = h^2 + r^2 .
\]
Een \omterm{def:g8:solids:def}{kegel} met $r = 3$ en $s = 5$ heeft dus hoogte $h = \sqrt{25 - 9} = 4$, en
\omterm{def:g2:measure:units}{inhoud} $\frac13 \pi \times 9 \times 4 = 12\pi$.
\end{example}

\begin{method}[Vraagstukken over inhoud]\label{met:g8:solids:method}
\begin{enumerate}
  \item Bepaal welk \omterm{def:g5:solids:def}{lichaam} het is (\omterm{def:g7:areas:prism}{prisma}/\omterm{def:g7:areas:prism}{cilinder}: $V = Bh$; \omterm{def:g8:solids:def}{piramide}/\omterm{def:g8:solids:def}{kegel}:
        $V = \frac13 Bh$);
  \item bereken eerst de \omterm{def:g6:measure:area}{oppervlakte} $B$ van het grondvlak, als een aparte stap;
  \item ga na dat de hoogte \omterm{def:g6:lines:perp}{loodrecht} op het grondvlak staat (gebruik Pythagoras
        als de schuine hoogte gegeven is);
  \item vermenigvuldig, houd de eenheden bij elkaar passend, en zet aan het einde
        om als het nodig is ($1$ L $= 1000$ cm$^3$).
\end{enumerate}
\end{method}

\section{Uitslagen}

\begin{example}[Uitslag van een piramide]\label{ex:g8:solids:net}
Een regelmatige \omterm{def:g8:solids:def}{piramide} met vierkant grondvlak uitvouwen maakt haar plat tot haar
\emph{\omterm{def:g5:solids:net}{uitslag}}: het vierkante grondvlak met vier identieke \omterm{def:g6:shapes:triangles}{gelijkbenige
driehoeken} aan zijn zijden. De gelijke zijden van de driehoeken zijn de
\emph{zijribben} van de \omterm{def:g8:solids:def}{piramide} --- hun lengte is niet $h$ en ook niet de
schuine hoogte van een \omterm{def:g5:solids:def}{zijvlak}, dus benoem alles zorgvuldig vóór je karton
knipt.
\end{example}

\section{Opgave: een kubus verdeeld in drie piramides}

\begin{problem}[{Weekendopgave --- waar de factor $\frac13$ vandaan komt, en wat
er met een piramide gebeurt die op halve hoogte wordt doorgesneden}]
\label{pb:g8:solids:1}
De opmerking na \cref{thm:g8:solids:volume} beweert dat een \omterm{def:g5:solids:def}{kubus} in drie
identieke \omterm{def:g8:solids:def}{piramides} kan worden verdeeld --- ``probeer het je voor te stellen!''.
Deze opgave stelt het precies voor, haalt de beroemde factor $\frac13$ er
eerlijk uit, verdeelt de \omterm{def:g5:solids:def}{kubus} daarna op een tweede manier in \emph{zes}
\omterm{def:g8:solids:def}{piramides}, en sluit af door een \omterm{def:g8:solids:def}{piramide} op halve hoogte door te snijden --- met
een uitkomst die weinig mensen bij de eerste poging goed raden.

Overal is $ABCDEFGH$ een \omterm{def:g5:solids:def}{kubus} met zijde $a$: grondvlak $ABCD$, bovenvlak
$EFGH$, met $E$ boven $A$, $F$ boven $B$, $G$ boven $C$ en $H$ boven $D$.

\medskip
\noindent\textbf{Deel I --- Drie \omterm{def:g8:solids:def}{piramides} met dezelfde top.}
\begin{enumerate}
  \item Het \omterm{def:g5:solids:def}{hoekpunt} $G$ hoort bij drie \omterm{def:g5:solids:def}{zijvlakken} van de \omterm{def:g5:solids:def}{kubus}. Noem de drie
        \omterm{def:g5:solids:def}{zijvlakken} die $G$ \emph{niet} bevatten, en beschrijf de drie \omterm{def:g8:solids:def}{piramides}
        die je krijgt door elk van die \omterm{def:g5:solids:def}{zijvlakken} als grondvlak te nemen, telkens
        met top $G$. Hoeveel \omterm{def:g5:solids:def}{zijvlakken}, \omterm{def:g5:solids:def}{ribben} en \omterm{def:g5:solids:def}{hoekpunten} heeft elke \omterm{def:g8:solids:def}{piramide}
        (\cref{ex:g8:solids:count})?
  \item Leg uit waarom een draai van de \omterm{def:g5:solids:def}{kubus} over een derde slag om zijn lange
        diagonaal $(AG)$ de \omterm{def:g5:solids:def}{kubus} onveranderd laat, maar de drie grondvlakken van
        vraag 1 cyclisch verwisselt --- en waarom de drie \omterm{def:g8:solids:def}{piramides} daarom
        identieke kopieën van elkaar zijn.
  \item Neem aan dat de drie \omterm{def:g8:solids:def}{piramides} de \omterm{def:g5:solids:def}{kubus} zonder overlapping vullen (een
        model van karton is behoorlijk overtuigend), en leid daaruit de \omterm{def:g2:measure:units}{inhoud} van
        elk van hen af.
  \item Ga dit na met de formule van \cref{thm:g8:solids:volume}: de \omterm{def:g8:solids:def}{piramide} met
        grondvlak $ABCD$ en top $G$ heeft haar top rechtstreeks boven een
        \emph{hoek} van haar grondvlak, precies zoals in
        \cref{exo:g8:solids:11}. Bepaal haar hoogte, pas de formule toe en
        vergelijk.
  \item Geef voor een \omterm{def:g5:solids:def}{kubus} met zijde $6$ cm de \omterm{def:g2:measure:units}{inhoud} van elk van de drie
        \omterm{def:g8:solids:def}{piramides} --- en leg uit wat deze verdeling te maken heeft met de
        gietproef uit het hoofdstuk (drie vullingen van de \omterm{def:g8:solids:def}{piramide} voor één
        \omterm{def:g7:areas:prism}{prisma}).
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Zes \omterm{def:g8:solids:def}{piramides} met hetzelfde middelpunt.}
Zij nu $O$ het middelpunt van de \omterm{def:g5:solids:def}{kubus}, en verbind $O$ met de vier \omterm{def:g5:solids:def}{hoekpunten} van
elk van de zes \omterm{def:g5:solids:def}{zijvlakken}.
\begin{enumerate}[resume]
  \item Beschrijf de zes \omterm{def:g8:solids:def}{piramides} die je zo krijgt, en leg uit waarom het
        identieke kopieën van elkaar zijn. Wat is de hoogte van elk van hen?
  \item Leid, door de \omterm{def:g2:measure:units}{inhoud} van de \omterm{def:g5:solids:def}{kubus} te verdelen, de \omterm{def:g2:measure:units}{inhoud} van elk van de
        zes \omterm{def:g8:solids:def}{piramides} af.
  \item Bereken die \omterm{def:g2:measure:units}{inhoud} opnieuw met de formule $V = \frac13 B h$, en ga na dat
        de twee antwoorden overeenstemmen.
  \item De twee verdelingen bevestigen de factor $\frac13$ voor enkele heel
        bijzondere \omterm{def:g8:solids:def}{piramides}. Leg uit waarom ze \cref{thm:g8:solids:volume} nog
        niet voor elke \omterm{def:g8:solids:def}{piramide} en \omterm{def:g8:solids:def}{kegel} \emph{bewijzen} --- en zeg hoe het
        hoofdstuk met dat gat omgaat: welke uitspraak wordt aangenomen, welke
        proef ondersteunt haar, en in welk \omterm{def:g6:measure:volume}{volume} van de reeks wordt ze eerlijk
        bewezen?
  \item Een \omterm{def:g5:solids:def}{kubus} met zijde $6$ cm wordt in zijn zes \omterm{def:g8:solids:def}{piramides} om het middelpunt
        verdeeld. Bereken de \omterm{def:g2:measure:units}{inhoud} van elk, op beide manieren.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Een \omterm{def:g8:solids:def}{piramide} op halve hoogte doorsnijden.}
$SABCD$ is een regelmatige \omterm{def:g8:solids:def}{piramide} met vierkant grondvlak: zijde van het
grondvlak $c$, hoogte $h$, top $S$ boven het middelpunt van het grondvlak. Snijd
haar door met het vlak door de \emph{middens} van de vier zijribben $[SA]$,
$[SB]$, $[SC]$, $[SD]$.
\begin{enumerate}[resume]
  \item Pas de middenparallelstelling (\cref{thm:g8:midpoints:direct}) toe in de
        driehoek $SAB$: wat zijn de richting en de lengte van het \omterm{def:g6:lines:objects}{lijnstuk} dat de
        middens van $[SA]$ en $[SB]$ verbindt? Leid daaruit af dat de doorsnede
        een vierkant met zijde $\frac{c}{2}$ is, \omterm{def:g6:lines:perp}{parallel} met het grondvlak, op
        hoogte $\frac{h}{2}$.
  \item Het stuk boven de doorsnede is zelf een \omterm{def:g8:solids:def}{piramide} met vierkant grondvlak.
        Geef de zijde van haar grondvlak en haar hoogte, en toon aan dat haar
        \omterm{def:g2:measure:units}{inhoud} precies \emph{een achtste} van de oorspronkelijke \omterm{def:g2:measure:units}{inhoud} $V$ is.
  \item Leid de \omterm{def:g2:measure:units}{inhoud} van het onderste stuk af (de \emph{afgeknotte \omterm{def:g8:solids:def}{piramide}},
        de vorm van een onvoltooide \omterm{def:g8:solids:def}{piramide}). In welke verhouding verdeelt de
        doorsnede op halve hoogte de \omterm{def:g2:measure:units}{inhoud}?
  \item De \omterm{def:g8:solids:def}{piramide} van het Louvre (\cref{exo:g8:solids:7}: zijde van het
        grondvlak $35$ m, hoogte $22$ m) wordt in twee campagnes gepoetst: eerst
        het glas boven halve hoogte, daarna het glas eronder. Welk deel van de
        \emph{\omterm{def:g2:measure:units}{inhoud}} zit onder halve hoogte? Schat het in kubieke meter, op
        honderd nauwkeurig.
  \item De algemene les: wat gebeurt er met de \omterm{def:g2:measure:units}{inhoud} van een \omterm{def:g8:solids:def}{piramide} als
