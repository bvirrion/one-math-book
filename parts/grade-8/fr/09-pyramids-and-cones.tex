\chapter{Pyramides et cônes}\label{ch:g8:solids}

Après les \omterm{def:g7:areas:prism}{prismes} et \omterm{def:g7:areas:prism}{cylindres} du \cref{ch:g7:areas} --- \omterm{def:g5:solids:def}{solides} à
section constante --- viennent les \omterm{def:g5:solids:def}{solides} à pointe~: \omterm{def:g8:solids:def}{pyramides} et
\omterm{def:g8:solids:def}{cônes}. Leur formule de \omterm{def:g6:measure:volume}{volume} porte un célèbre facteur $\frac13$, que
ce chapitre rend plausible et utilise~; l'étude des \omterm{def:g5:solids:def}{solides} se
poursuit avec les sphères et les sections au \cref{ch:g9:solids}.

\section{Décrire pyramides et cônes}

\begin{definition}[Pyramide, cône]\label{def:g8:solids:def}
\begin{itemize}
  \item Une \emph{pyramide}\index{pyramide} a un \omterm{def:g4:geometry:polygon}{polygone} pour
        \emph{base} et un point, le \emph{\omterm{def:g5:solids:def}{sommet}}, joint à chaque
        \omterm{def:g5:solids:def}{sommet} de la base par des \omterm{def:g5:solids:def}{faces} triangulaires. Sa
        \emph{hauteur} est la distance du \omterm{def:g5:solids:def}{sommet} au plan de la base.
  \item Un \emph{cône}\index{cône} est la même construction sur un
        disque~: un disque de base, un \omterm{def:g5:solids:def}{sommet}, et une surface latérale
        courbe.
\end{itemize}
Une pyramide est \emph{régulière} lorsque sa base est un \omterm{def:g4:geometry:polygon}{polygone}
régulier (carré, \omterm{def:g6:shapes:triangles}{triangle équilatéral}, \dots) et que son \omterm{def:g5:solids:def}{sommet} se
trouve à la verticale du centre de la base.
\end{definition}

\begin{example}[Compter faces, arêtes, sommets]\label{ex:g8:solids:count}
Une \omterm{def:g8:solids:def}{pyramide} à base carrée a $5$ \omterm{def:g5:solids:def}{faces} ($1$ carré $+ 4$ triangles),
$8$ \omterm{def:g5:solids:def}{arêtes} et $5$ \omterm{def:g5:solids:def}{sommets}. Avec une base à $n$ côtés~: $n + 1$
\omterm{def:g5:solids:def}{faces}, $2n$ \omterm{def:g5:solids:def}{arêtes}, $n + 1$ \omterm{def:g5:solids:def}{sommets}. Vérifier le petit motif
d'Euler~: \omterm{def:g5:solids:def}{faces} $+$ \omterm{def:g5:solids:def}{sommets} $=$ \omterm{def:g5:solids:def}{arêtes} $+ 2$.
\end{example}

\section{Volume}

\begin{theorem}[Volume d'une pyramide ou d'un cône]\label{thm:g8:solids:volume}
Pour une \omterm{def:g8:solids:def}{pyramide} ou un \omterm{def:g8:solids:def}{cône} d'\omterm{def:g6:measure:area}{aire} de base $B$ et de hauteur $h$~:
\[
V = \frac{1}{3}\, B \times h
\]
--- un tiers du \omterm{def:g7:areas:prism}{prisme} ou du \omterm{def:g7:areas:prism}{cylindre} de même base et même hauteur.
Pour un \omterm{def:g8:solids:def}{cône} de \omterm{def:g3:shapes:shapes}{rayon} de base $r$~: $V = \frac13 \pi r^2 h$.
\end{theorem}
\admitted

\begin{remark}[Pourquoi un tiers~?]
Remplir une \omterm{def:g8:solids:def}{pyramide} creuse d'eau et la verser dans le \omterm{def:g7:areas:prism}{prisme} de même
base et même hauteur~: il faut exactement trois remplissages. Mieux
encore~: un \omterm{def:g5:solids:def}{cube} peut se découper en trois \omterm{def:g8:solids:def}{pyramides} identiques,
chacune ayant une \omterm{def:g5:solids:def}{face} du \omterm{def:g5:solids:def}{cube} pour base et un \omterm{def:g5:solids:def}{sommet} du \omterm{def:g5:solids:def}{cube} pour
apex (essayer de se le représenter~!). Une preuve générale utilise
l'intégration, un outil du \omterm{def:g6:measure:volume}{volume} suivant.
\end{remark}

\begin{example}\label{ex:g8:solids:volume}
Une \omterm{def:g8:solids:def}{pyramide} a une base carrée de côté $5$ cm et une hauteur de
$9$ cm~:
\begin{enumerate}
  \item \omterm{def:g6:measure:area}{aire} de base~: $B = 5^2 = 25$ cm$^2$~;
  \item \omterm{def:g6:measure:volume}{volume}~: $V = \frac13 \times 25 \times 9 = 25 \times 3 = 75$
        cm$^3$.
\end{enumerate}
Un cornet de glace de \omterm{def:g3:shapes:shapes}{rayon} $3$ cm et de hauteur $10$ cm~:
\[
V = \frac13 \times \pi \times 3^2 \times 10 = 30\pi \approx 94
\text{ cm}^3 .
\]
\end{example}

\begin{example}[Attention à la hauteur]\label{ex:g8:solids:slant}
Pour un \omterm{def:g8:solids:def}{cône}, ne pas confondre la hauteur $h$ (\omterm{def:g5:solids:def}{sommet} au centre de la
base, \omterm{def:g6:lines:perp}{perpendiculaire}) avec la \emph{génératrice} $s$ (\omterm{def:g5:solids:def}{sommet} au
bord). Elles sont liées par Pythagore
(\cref{thm:g8:pythagoras:direct}) dans le \omterm{def:g6:shapes:triangles}{triangle rectangle}
sommet--centre--bord~:
\[
s^2 = h^2 + r^2 .
\]
Un \omterm{def:g8:solids:def}{cône} avec $r = 3$ et $s = 5$ a donc pour hauteur
$h = \sqrt{25 - 9} = 4$, et pour \omterm{def:g6:measure:volume}{volume}
$\frac13 \pi \times 9 \times 4 = 12\pi$.
\end{example}

\begin{method}[Problèmes de volume]\label{met:g8:solids:method}
\begin{enumerate}
  \item Identifier le \omterm{def:g5:solids:def}{solide} (\omterm{def:g7:areas:prism}{prisme}/\omterm{def:g7:areas:prism}{cylindre}~: $V = Bh$~;
        \omterm{def:g8:solids:def}{pyramide}/\omterm{def:g8:solids:def}{cône}~: $V = \frac13 Bh$)~;
  \item calculer d'abord l'\omterm{def:g6:measure:area}{aire} de base $B$, comme une étape
        séparée~;
  \item vérifier que la hauteur est \omterm{def:g6:lines:perp}{perpendiculaire} à la base
        (utiliser Pythagore si la génératrice est donnée)~;
  \item multiplier, garder des unités cohérentes, convertir à la fin
        si besoin ($1$ L $= 1000$ cm$^3$).
\end{enumerate}
\end{method}

\section{Patrons}

\begin{example}[Patron d'une pyramide]\label{ex:g8:solids:net}
Déplier une \omterm{def:g8:solids:def}{pyramide} régulière à base carrée la met à plat en son
\emph{\omterm{def:g5:solids:net}{patron}}~: la base carrée avec quatre \omterm{def:g6:shapes:triangles}{triangles isocèles}
identiques attachés à ses côtés. Les côtés égaux des triangles sont
les \emph{\omterm{def:g5:solids:def}{arêtes} latérales} de la \omterm{def:g8:solids:def}{pyramide} --- leur longueur n'est ni
$h$ ni la hauteur d'une \omterm{def:g5:solids:def}{face}, donc étiqueter soigneusement avant de
découper le carton.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g8:solids:2}

\begin{exercise}[$\star$]\label{exo:g8:solids:3}

\begin{exercise}[$\star$]\label{exo:g8:solids:6}

\omterm{def:g8:solids:def}{cône} est le \omterm{def:g3:shapes:shapes}{rayon} de ce secteur~: la hauteur $h$ ou la génératrice

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:solids:8}
Un \omterm{def:g8:solids:def}{cône} a pour génératrice $13$ cm et pour \omterm{def:g3:shapes:shapes}{rayon} de base $5$ cm.
Calculer sa hauteur, puis son \omterm{def:g6:measure:volume}{volume} (exact avec $\pi$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:solids:9}

Quelle hauteur le liquide atteint-il dans le \omterm{def:g7:areas:prism}{cylindre}~?

\begin{exercise}[$\star\star$]\label{exo:g8:solids:10}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:solids:11}
Une \omterm{def:g8:solids:def}{pyramide} à base carrée de côté $6$ cm a son \omterm{def:g5:solids:def}{sommet} à la verticale
d'un coin de la base, à la hauteur $8$ cm.
\begin{enumerate}
  \item La formule $V = \frac13 Bh$ s'applique-t-elle encore~?
        (Oui --- le \omterm{def:g5:solids:def}{sommet} n'a pas besoin d'être centré. Calculer
        $V$.)
  \item Calculer la longueur de la plus longue \omterm{def:g5:solids:def}{arête} latérale, du
        \omterm{def:g5:solids:def}{sommet} au coin opposé de la base (deux étapes de Pythagore~:
        la diagonale de la base d'abord).
\end{enumerate}
\end{exercise}

\section{Problème~: un cube découpé en trois pyramides}

\begin{problem}[{Devoir maison --- d'où vient le facteur $\frac13$,
et ce qu'il advient d'une pyramide coupée à mi-hauteur}]
\label{pb:g8:solids:1}
La remarque qui suit \cref{thm:g8:solids:volume} affirme qu'un \omterm{def:g5:solids:def}{cube}
peut se découper en trois \omterm{def:g8:solids:def}{pyramides} identiques --- «~essayer de se le
représenter~!~». Ce problème se le représente précisément, en tire
honnêtement le fameux facteur $\frac13$, découpe ensuite le \omterm{def:g5:solids:def}{cube}
d'une seconde façon en \emph{six} \omterm{def:g8:solids:def}{pyramides}, et s'achève en tranchant
une \omterm{def:g8:solids:def}{pyramide} à mi-hauteur --- avec un résultat que peu de gens
devinent correctement du premier coup.

Partout, $ABCDEFGH$ est un \omterm{def:g5:solids:def}{cube} d'\omterm{def:g5:solids:def}{arête} $a$~: base $ABCD$, \omterm{def:g5:solids:def}{face}
supérieure $EFGH$, avec $E$ au-dessus de $A$, $F$ au-dessus de $B$,
$G$ au-dessus de $C$ et $H$ au-dessus de $D$.

\medskip
\noindent\textbf{Partie I --- Trois \omterm{def:g8:solids:def}{pyramides} de même \omterm{def:g5:solids:def}{sommet}.}
\begin{enumerate}
  \item Le \omterm{def:g5:solids:def}{sommet} $G$ appartient à trois \omterm{def:g5:solids:def}{faces} du \omterm{def:g5:solids:def}{cube}. Énumérer les
        trois \omterm{def:g5:solids:def}{faces} qui ne contiennent \emph{pas} $G$, et décrire
        les trois \omterm{def:g8:solids:def}{pyramides} obtenues en prenant chacune d'elles pour
        base, avec $G$ pour \omterm{def:g5:solids:def}{sommet} à chaque fois. Combien chaque
        \omterm{def:g8:solids:def}{pyramide} a-t-elle de \omterm{def:g5:solids:def}{faces}, d'\omterm{def:g5:solids:def}{arêtes} et de \omterm{def:g5:solids:def}{sommets}
        (\cref{ex:g8:solids:count})~?
  \item Expliquer pourquoi faire tourner le \omterm{def:g5:solids:def}{cube} d'un tiers de tour
        autour de sa grande diagonale $(AG)$ laisse le \omterm{def:g5:solids:def}{cube}
        inchangé mais permute circulairement les trois bases de la
        question 1 --- et pourquoi les trois \omterm{def:g8:solids:def}{pyramides} sont donc des
        copies identiques les unes des autres.
  \item En admettant que les trois \omterm{def:g8:solids:def}{pyramides} remplissent le \omterm{def:g5:solids:def}{cube}
        sans se chevaucher (une maquette en carton est fort
        convaincante), en déduire le \omterm{def:g6:measure:volume}{volume} de chacune.
  \item Contrôler ce résultat avec la formule du
        \cref{thm:g8:solids:volume}~: la \omterm{def:g8:solids:def}{pyramide} de base $ABCD$ et
        de \omterm{def:g5:solids:def}{sommet} $G$ a son \omterm{def:g5:solids:def}{sommet} à la verticale d'un \emph{coin}
        de sa base, exactement comme dans
        \cref{exo:g8:solids:11}. Identifier sa hauteur, appliquer la
        formule, et comparer.
  \item Pour un \omterm{def:g5:solids:def}{cube} d'\omterm{def:g5:solids:def}{arête} $6$ cm, donner le \omterm{def:g6:measure:volume}{volume} de chacune des
        trois \omterm{def:g8:solids:def}{pyramides} --- et expliquer le rapport entre cette
        décomposition et l'expérience du transvasement du chapitre
        (trois remplissages de la \omterm{def:g8:solids:def}{pyramide} pour un \omterm{def:g7:areas:prism}{prisme}).
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Six \omterm{def:g8:solids:def}{pyramides} de même \omterm{def:g5:solids:def}{sommet}, le
centre.}
Soit maintenant $O$ le centre du \omterm{def:g5:solids:def}{cube}~; joindre $O$ aux quatre
\omterm{def:g5:solids:def}{sommets} de chacune des six \omterm{def:g5:solids:def}{faces}.
\begin{enumerate}[resume]
  \item Décrire les six \omterm{def:g8:solids:def}{pyramides} obtenues, et expliquer pourquoi
        elles sont des copies identiques les unes des autres. Quelle
        est la hauteur de chacune~?
  \item En déduire, en partageant le \omterm{def:g6:measure:volume}{volume} du \omterm{def:g5:solids:def}{cube}, le \omterm{def:g6:measure:volume}{volume} de
        chacune des six \omterm{def:g8:solids:def}{pyramides}.
  \item Recalculer ce \omterm{def:g6:measure:volume}{volume} avec la formule $V = \frac13 B h$, et
        vérifier que les deux réponses concordent.
  \item Les deux décompositions confirment le facteur $\frac13$ pour
        des \omterm{def:g8:solids:def}{pyramides} très particulières. Expliquer pourquoi elles
        ne \emph{démontrent} pas encore
        \cref{thm:g8:solids:volume} pour toute \omterm{def:g8:solids:def}{pyramide} et tout
        \omterm{def:g8:solids:def}{cône} --- et dire comment le chapitre traite cette lacune~:
        quel énoncé est admis, quelle expérience l'appuie, et dans
        quel \omterm{def:g6:measure:volume}{volume} de la série il est honnêtement démontré.
  \item Un \omterm{def:g5:solids:def}{cube} d'\omterm{def:g5:solids:def}{arête} $6$ cm est découpé en ses six \omterm{def:g8:solids:def}{pyramides} de
        centre. Calculer le \omterm{def:g6:measure:volume}{volume} de chacune, des deux façons.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Couper une \omterm{def:g8:solids:def}{pyramide} à mi-hauteur.}
$SABCD$ est une \omterm{def:g8:solids:def}{pyramide} régulière à base carrée~: côté de base $c$,
hauteur $h$, \omterm{def:g5:solids:def}{sommet} $S$ à la verticale du centre de la base. La
couper par le plan passant par les \emph{milieux} des quatre \omterm{def:g5:solids:def}{arêtes}
latérales $[SA]$, $[SB]$, $[SC]$, $[SD]$.
\begin{enumerate}[resume]
  \item Appliquer le théorème des milieux
        (\cref{thm:g8:midpoints:direct}) dans le triangle $SAB$~:
        quelles sont la direction et la longueur du \omterm{def:g6:lines:objects}{segment} joignant
        les milieux de $[SA]$ et de $[SB]$~? En déduire que la
        section est un carré de côté $\frac{c}{2}$, \omterm{def:g6:lines:perp}{parallèle} à la
        base, à la hauteur $\frac{h}{2}$.
  \item Le morceau situé au-dessus de la coupe est lui-même une
        \omterm{def:g8:solids:def}{pyramide} à base carrée. Donner son côté de base et sa
        hauteur, et montrer que son \omterm{def:g6:measure:volume}{volume} vaut exactement le
        \emph{huitième} du \omterm{def:g6:measure:volume}{volume} initial $V$.
  \item En déduire le \omterm{def:g6:measure:volume}{volume} du morceau inférieur (le \emph{tronc de
        \omterm{def:g8:solids:def}{pyramide}}, la forme d'une \omterm{def:g8:solids:def}{pyramide} inachevée). Dans quel
        rapport la coupe à mi-hauteur partage-t-elle le \omterm{def:g6:measure:volume}{volume}~?
  \item La \omterm{def:g8:solids:def}{pyramide} du Louvre (\cref{exo:g8:solids:7}~: côté de base
        $35$ m, hauteur $22$ m) est nettoyée en deux campagnes~: le
        verre situé au-dessus de la mi-hauteur, puis celui situé en
