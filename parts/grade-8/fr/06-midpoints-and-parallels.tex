\chapter{Milieux et parallèles}\label{ch:g8:midpoints}

Joindre les milieux de deux côtés d'un triangle~: le \omterm{def:g6:lines:objects}{segment} obtenu
est toujours \omterm{def:g6:lines:perp}{parallèle} au troisième côté, et exactement deux fois
plus court. Ce «~théorème des milieux~» et sa réciproque sont le
premier avant-goût d'une grande idée --- les \omterm{def:g6:lines:perp}{parallèles} découpent des
longueurs proportionnellement --- qui s'épanouit en le théorème de
Thalès au \cref{ch:g9:thales}.

\section{Le théorème des milieux}

\begin{theorem}[Théorème des milieux]\label{thm:g8:midpoints:direct}
Dans un triangle $ABC$, soit $I$ le milieu de $[AB]$ et $J$ le milieu
de $[AC]$. Alors la droite $(IJ)$ est \omterm{def:g6:lines:perp}{parallèle} à $(BC)$, et
\[
IJ = \frac{BC}{2} .
\]
\end{theorem}

\begin{proof}[Idée de preuve]
Soit $K$ le \omterm{def:g7:central:def}{symétrique} de $I$ par rapport à $J$ (un demi-tour,
\cref{ch:g7:central}). Le \omterm{def:g4:geometry:polygon}{quadrilatère} $AICK$ a des diagonales
$[IK]$ et $[AC]$ se croisant en leur milieu commun $J$~: c'est un
\omterm{def:g7:central:parallelogram}{parallélogramme}, donc $KC$ est \omterm{def:g6:lines:perp}{parallèle} à $AI$, c'est-à-dire à
$(AB)$, et $KC = AI = IB$. Alors $IBCK$ a deux côtés opposés
($[IB]$ et $[KC]$) \omterm{def:g6:lines:perp}{parallèles} et égaux~: c'est un \omterm{def:g7:central:parallelogram}{parallélogramme}
aussi (\cref{thm:g7:central:recognize}), donc
$(IK) \parallel (BC)$ --- c'est-à-dire $(IJ) \parallel (BC)$ --- et
$BC = IK = 2\,IJ$.
\end{proof}

\begin{example}\label{ex:g8:midpoints:direct}
Dans un triangle $ABC$ avec $BC = 9$ cm, les milieux $I$ de $[AB]$ et
$J$ de $[AC]$ sont joints. Sans aucune mesure~:
$(IJ) \parallel (BC)$ et $IJ = \frac92 = 4.5$ cm. Le petit triangle
$AIJ$ est une copie à demi-échelle de $ABC$ --- son \omterm{def:g6:measure:perimeter}{périmètre} est
aussi la \omterm{def:g3:division:half}{moitié} du \omterm{def:g6:measure:perimeter}{périmètre} de $ABC$.
\end{example}

\begin{theorem}[Réciproque]\label{thm:g8:midpoints:converse}
Dans un triangle $ABC$, la droite passant par le milieu $I$ de $[AB]$
et \emph{\omterm{def:g6:lines:perp}{parallèle}} à $(BC)$ coupe le côté $[AC]$ en son milieu.
\end{theorem}
\admitted

\begin{example}\label{ex:g8:midpoints:converse}
Un chemin traverse un champ triangulaire, partant du milieu d'un côté
et courant \omterm{def:g6:lines:perp}{parallèle} au côté opposé. La réciproque garantit que le
chemin sort exactement au milieu de l'autre côté --- sans mesure
nécessaire du côté lointain.
\end{example}

\begin{method}[Choisir entre théorème et réciproque]\label{met:g8:midpoints:choose}
\begin{enumerate}
  \item \emph{Deux milieux connus} $\Rightarrow$ utiliser le théorème
        direct~: conclure parallélisme et demi-longueur.
  \item \emph{Un milieu et une \omterm{def:g6:lines:perp}{parallèle} connus} $\Rightarrow$
        utiliser la réciproque~: conclure que le point d'intersection
        est un milieu.
  \item Écrire quel triangle, quels milieux et quel théorème on
        utilise --- une phrase chacun.
\end{enumerate}
\end{method}

\section{Triangles à demi-échelle}

\begin{proposition}[Le triangle des milieux]\label{prop:g8:midpoints:medial}
Joindre les trois milieux des côtés d'un triangle le découpe en
quatre triangles d'\omterm{def:g6:measure:area}{aires} égales, chacun une copie à demi-échelle de
l'original.
\end{proposition}

\begin{proof}
Chaque côté du triangle des milieux est \omterm{def:g6:lines:perp}{parallèle} à un côté de $ABC$
et deux fois plus court (\cref{thm:g8:midpoints:direct}, appliqué
trois fois). Les quatre petits triangles ont des côtés des trois
mêmes longueurs, donc ce sont des copies identiques~; ensemble ils
pavent $ABC$, donc chacun a le \omterm{def:g3:division:half}{quart} de son \omterm{def:g6:measure:area}{aire}.
\end{proof}

\begin{remark}[Vers Thalès]
Le théorème des milieux est le cas «~une \omterm{def:g3:division:half}{moitié}~» d'un fait plus
général~: une droite \omterm{def:g6:lines:perp}{parallèle} à un côté d'un triangle coupe les deux
autres côtés dans des \emph{rapports égaux} --- un tiers et un tiers,
deux cinquièmes et deux cinquièmes\,\dots\ Cet énoncé général est le
théorème de Thalès (\cref{ch:g9:thales})~; le cas des milieux est le
seul nécessaire cette année.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:1}
Dans un triangle $RST$, $M$ est le milieu de $[RS]$ et $N$ le milieu
de $[RT]$, avec $ST = 7$ cm. Que peut-on dire de la droite $(MN)$ et
de la longueur $MN$~? Citer le théorème utilisé.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:2}
Dans un triangle $ABC$, $I$ est le milieu de $[AB]$ et la droite
passant par $I$ \omterm{def:g6:lines:perp}{parallèle} à $(BC)$ coupe $[AC]$ en $J$, avec
$AC = 11$ cm. Calculer $AJ$, en citant le théorème utilisé.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:3}
Tracer un triangle $ABC$ avec $AB = 8$ cm, $AC = 6$ cm, $BC = 7$ cm,
placer les milieux $I$ de $[AB]$ et $J$ de $[AC]$, et mesurer $IJ$
pour vérifier le théorème. Calculer le \omterm{def:g6:measure:perimeter}{périmètre} de $AIJ$ sans rien
mesurer d'autre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:4}
$IJK$ est le triangle des milieux de $ABC$, dont les côtés mesurent
$10$, $12$ et $16$ cm. Donner les trois côtés de $IJK$ et son
\omterm{def:g6:measure:perimeter}{périmètre}. Comment se comparent les deux \omterm{def:g6:measure:perimeter}{périmètres}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:5}
L'\omterm{def:g6:measure:area}{aire} d'un triangle est $36$ cm$^2$. Quelle est l'\omterm{def:g6:measure:area}{aire} de son
triangle des milieux~? De chacun des quatre petits triangles~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:midpoints:6}
Dans un triangle $ABC$, $I$ est le milieu de $[AB]$, $J$ celui de
$[AC]$ et $K$ celui de $[BC]$. Montrer que $IJKB$ est un
\omterm{def:g7:central:parallelogram}{parallélogramme}. (Comparer $[IJ]$ et $[BK]$~: \omterm{def:g6:lines:perp}{parallèles}~? égaux~?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:midpoints:7}
Une voile triangulaire a son bord le plus bas long de $6.4$ m. Une
couture de renfort joint les milieux des deux autres bords. Quelle
longueur de couture faut-il~? Et si la couture joignait au contraire
les points situés au \omterm{def:g3:division:half}{quart} de chaque bord, en partant du \omterm{def:g5:solids:def}{sommet} du
haut~? (Conjecture avec un dessin~; la preuve est celle de Thalès,
\cref{ch:g9:thales}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:midpoints:8}
Dans le triangle $ABC$, rectangle en $A$, $M$ est le milieu de
l'hypoténuse $[BC]$, et la \omterm{def:g6:lines:perp}{parallèle} à $(AB)$ passant par $M$
rencontre $[AC]$ en $N$.
\begin{enumerate}
  \item Montrer que $N$ est le milieu de $[AC]$.
  \item Expliquer pourquoi $(MN)$ est \omterm{def:g6:lines:perp}{perpendiculaire} à $(AC)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:midpoints:9}
Soit $ABCD$ un \omterm{def:g4:geometry:polygon}{quadrilatère} \emph{quelconque}, et $I$, $J$, $K$, $L$
les milieux de $[AB]$, $[BC]$, $[CD]$, $[DA]$ (la conjecture de
\cref{exo:g7:central:11}).
\begin{enumerate}
  \item Appliquer le théorème des milieux dans le triangle $ABC$ au
        \omterm{def:g6:lines:objects}{segment} $[IJ]$, et dans le triangle $ACD$ au \omterm{def:g6:lines:objects}{segment} $[LK]$~:
        comparer chacun à la diagonale $[AC]$.
  \item Conclure que $IJKL$ est toujours un \omterm{def:g7:central:parallelogram}{parallélogramme}.
\end{enumerate}
\end{exercise}

\section{Problème~: les trois médianes et le centre de gravité}

\begin{problem}[{Devoir maison --- les médianes d'un triangle se
coupent en un même point, aux deux tiers de chacune d'elles}]
\label{pb:g8:midpoints:1}
Découper un triangle dans du carton rigide~: il tiendra en
équilibre, parfaitement à plat, sur une pointe de crayon placée en un
point bien particulier --- le \emph{centre de gravité}\index{centre
de gravité} du triangle. Ce problème trouve ce point. Une
\emph{médiane}\index{médiane} d'un triangle est le \omterm{def:g6:lines:objects}{segment} qui joint
un \omterm{def:g5:solids:def}{sommet} au \emph{milieu} du côté opposé. On démontrera --- avec le
théorème des milieux employé deux fois, d'une manière inattendue ---
que les trois médianes passent toutes par un même point, et on
situera ce point précisément.

\medskip
\noindent\textbf{Partie I --- Échauffement.}
\begin{enumerate}
  \item Tracer un grand triangle $ABC$ (sans forme particulière),
        construire les milieux $I$ de $[AB]$, $J$ de $[AC]$ et $K$
        de $[BC]$, et tracer les trois médianes $[AK]$, $[BJ]$,
        $[CI]$. Qu'observe-t-on~?
  \item Démontrer qu'une médiane découpe le triangle en deux
        triangles de même \omterm{def:g6:measure:area}{aire}. (Comparer les bases et les hauteurs,
        et utiliser la formule de l'\omterm{def:g6:measure:area}{aire}~: base $\times$ hauteur
        $\div\ 2$.) Pour un triangle d'\omterm{def:g6:measure:area}{aire} $36$ cm$^2$, quelles
        sont les \omterm{def:g6:measure:area}{aires} des deux morceaux~?
  \item Dans le triangle $ABC$, que dit
        \cref{thm:g8:midpoints:direct} du \omterm{def:g6:lines:objects}{segment} $[IJ]$~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Deux médianes se coupent aux deux
tiers.}
Les médianes $[BJ]$ et $[CI]$ se coupent en un point~; appelons-le
$G$. Soit $M$ le milieu de $[GB]$ et $N$ le milieu de $[GC]$.
\begin{enumerate}[resume]
  \item Appliquer le théorème des milieux \emph{dans le triangle
        $GBC$}~: que dit-il du \omterm{def:g6:lines:objects}{segment} $[MN]$~?
  \item En déduire que $(IJ) \parallel (MN)$ et $IJ = MN$, et
        conclure à l'aide du~\cref{thm:g7:central:recognize} que
        $IJNM$ est un \omterm{def:g7:central:parallelogram}{parallélogramme}.
  \item Expliquer pourquoi $I$ et $N$ sont tous deux sur la médiane
        $(CI)$, et $J$ et $M$ tous deux sur la médiane $(BJ)$ ---
        les diagonales du \omterm{def:g7:central:parallelogram}{parallélogramme} $IJNM$ sont donc $[IN]$
        et $[JM]$, et elles se coupent exactement en $G$. Que dit
        alors la \emph{définition} d'un \omterm{def:g7:central:parallelogram}{parallélogramme} au sujet de
        $G$~?
  \item En déduire que $BM = MG = GJ$, et conclure~:
        \[
        BG = 2 \times GJ
        \]
        --- le point $G$ se trouve sur la médiane $[BJ]$ aux deux
        tiers depuis le \omterm{def:g5:solids:def}{sommet} $B$. Énoncer et justifier le résultat
        analogue sur la médiane $[CI]$.
  \item Oublions maintenant $[CI]$ et reprenons exactement le même
        raisonnement sur le couple de médianes $[BJ]$ et $[AK]$~:
        leur point d'intersection se trouve lui aussi aux deux tiers
        de $[BJ]$ depuis $B$. Expliquer pourquoi cela oblige ce
        point à être le \emph{même} point $G$ --- et conclure que
        les trois médianes passent par $G$.
  \item Dans un triangle où la médiane $[BJ]$ mesure $9$ cm, à
        quelle distance $G$ se trouve-t-il de $B$, et de $J$~?
  \item Justifier en une phrase que $AG = 2 \times GK$ également.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Six parts égales, et des
coordonnées.}
Les trois médianes découpent le triangle en six petits triangles
autour de $G$.
\begin{enumerate}[resume]
  \item Dans le triangle $GBC$, le \omterm{def:g6:lines:objects}{segment} $[GK]$ est une médiane.
        En déduire que les deux petits triangles $GBK$ et $GKC$ ont
        la même \omterm{def:g6:measure:area}{aire}.
  \item À l'aide de la question 2 dans les triangles $ABK$ et $AKC$
        (tous deux coupés par la médiane $[AK]$ de $ABC$), montrer
        que les triangles $ABG$ et $ACG$ ont la même \omterm{def:g6:measure:area}{aire}. En
        recommençant avec une autre médiane, conclure que les trois
        triangles $ABG$, $BCG$, $CAG$ ont tous pour \omterm{def:g6:measure:area}{aire} le tiers de
        celle de $ABC$.
  \item Chacun de ces trois triangles est coupé en deux par un
        morceau de médiane (par exemple $[GI]$ est une médiane du
        triangle $ABG$ --- issue de quel \omterm{def:g5:solids:def}{sommet}~?). Conclure~: les
        six petites parts autour de $G$ ont toutes la même \omterm{def:g6:measure:area}{aire}, le
        sixième du triangle.
  \item Dans un repère quadrillé, placer $A(0, 0)$, $B(6, 0)$ et
        $C(0, 6)$, avec $K(3, 3)$ milieu de $[BC]$. Utiliser le
        théorème des deux tiers de la partie II pour calculer les
        coordonnées de $G$ sur la médiane $[AK]$~; vérifier ensuite
        que le même point se trouve aux deux tiers de la médiane
        $[BJ]$, où $J(0, 3)$. Comparer les coordonnées de $G$ aux
        moyennes
        \[
        \frac{0 + 6 + 0}{3},
        \qquad
        \frac{0 + 0 + 6}{3} .
        \]
  \item Le final de l'équilibre~: à l'aide des questions 2 et 8,
        expliquer pourquoi le triangle de carton tient en équilibre
        sur le tranchant d'un couteau posé le long d'une médiane
        quelconque --- autant de carton de chaque côté --- et
        pourquoi la pointe du crayon doit donc être placée en $G$,
        le point que les physiciens appellent le \omterm{pb:g8:midpoints:1}{centre de gravité}.
        (Une démonstration d'équilibre pleinement rigoureuse
        réclame le calcul intégral des \omterm{def:g6:measure:volume}{volumes} universitaires~;
        l'égalité des \omterm{def:g6:measure:area}{aires} la rend crédible dès aujourd'hui.)
\end{enumerate}
\end{problem}
