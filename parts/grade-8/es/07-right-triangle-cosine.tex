\chapter{El triángulo rectángulo y el coseno}\label{ch:g8:cosine}

Dos hechos preciosos atan el \omterm{def:g6:shapes:triangles}{triángulo rectángulo} a la circunferencia: un \omterm{def:g6:shapes:triangles}{triángulo
rectángulo} encaja exactamente en una semicircunferencia, con la hipotenusa como
\omterm{def:g4:geometry:circle}{diámetro}. Y un solo número, el \emph{coseno} de un ángulo, codifica la
forma de todos los \omterm{def:g6:shapes:triangles}{triángulos rectángulos} con ese ángulo --- la primera
razón trigonométrica, por delante de sus hermanos el seno y la tangente
(\cref{ch:g9:trig}).

\section{El triángulo rectángulo y su circunferencia}

\begin{theorem}[Teorema de la circunferencia]\label{thm:g8:cosine:circle}
\begin{enumerate}
  \item si un triángulo $ABC$ es rectángulo en $A$, entonces $A$ está sobre
        la circunferencia cuyo \omterm{def:g4:geometry:circle}{diámetro} es la hipotenusa $[BC]$;
  \item recíprocamente, si $A$ está sobre una circunferencia de \omterm{def:g4:geometry:circle}{diámetro} $[BC]$ (con
        $A \neq B, C$), entonces el triángulo $ABC$ es rectángulo en
        $A$.
\end{enumerate}
Equivalentemente: en un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, el punto medio de la hipotenusa está
a igual distancia de los tres \omterm{def:g5:solids:def}{vértices} --- la mediana desde el
\omterm{def:g3:shapes:rightangle}{ángulo recto} mide la \omterm{def:g3:division:half}{mitad} de la hipotenusa.
\end{theorem}

\begin{proof}[Demostración de 1]
Sea $O$ el punto medio de $[BC]$ y $D$ el \omterm{def:g7:central:def}{simétrico} de $A$ respecto de
$O$. Las diagonales de $ABDC$ se cortan en su punto medio común $O$, así que
$ABDC$ es un \omterm{def:g7:central:parallelogram}{paralelogramo} (\cref{def:g7:central:parallelogram}) ---
con un \omterm{def:g3:shapes:rightangle}{ángulo recto} en $A$: es un rectángulo. Las diagonales de un
rectángulo son iguales, así que $OA = \frac{AD}{2} = \frac{BC}{2}$: el
punto $A$ está a distancia $\frac{BC}{2}$ de $O$, es decir, sobre la circunferencia
de \omterm{def:g4:geometry:circle}{diámetro} $[BC]$. (El punto 2 se demuestra recorriendo el argumento
al revés.)
\end{proof}

\begin{example}\label{ex:g8:cosine:circle}
Un triángulo tiene una hipotenusa de $10$ cm. Sin saber nada
más, la mediana desde el \omterm{def:g3:shapes:rightangle}{ángulo recto} mide $5$ cm y la
\omterm{pb:g6:symmetry:1}{circunferencia circunscrita} del triángulo tiene \omterm{def:g3:shapes:shapes}{radio} $5$ cm, con centro en
el punto medio de la hipotenusa.
\end{example}

\section{El coseno de un ángulo agudo}

\begin{definition}[Coseno]\label{def:g8:cosine:def}
En un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, para un ángulo agudo $\theta$:
\[
\cos\theta = \frac{\text{cateto contiguo a } \theta}{\text{hipotenusa}}
\]
--- el cateto que toca a $\theta$, dividido entre la hipotenusa. Esta razón
depende solo del ángulo, no del tamaño del triángulo: todos los \omterm{def:g6:shapes:triangles}{triángulos
rectángulos} con el mismo ángulo agudo son ampliaciones unos de otros,
y las ampliaciones conservan las razones de longitudes
(el \cref{ch:g8:midpoints} empezó esta historia; el \cref{ch:g9:thales}
la termina).
\end{definition}

$\frac{\text{contiguo}}{\text{hipotenusa}}$ es el mismo para los dos ---
ese valor común es $\cos\theta$.}
\end{omfigure}

\begin{proposition}[Primeros valores y cotas]\label{prop:g8:cosine:bounds}
Para todo ángulo agudo $\theta$: $0 < \cos\theta < 1$, y el coseno
\emph{decrece} a medida que el ángulo se abre: un ángulo más abierto tiene menor
coseno. Referencias: $\cos 0^\circ = 1$, $\cos 60^\circ = \frac12$,
$\cos 90^\circ = 0$ (los valores extremos corresponden a triángulos
aplastados).
\end{proposition}
\admitted

\begin{method}[Usar el coseno]\label{met:g8:cosine:use}
En un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, cuando lo conocido y lo buscado son un
ángulo agudo, su cateto contiguo y la hipotenusa:
\begin{enumerate}
  \item escribe la definición:
        $\cos\theta = \frac{\text{contiguo}}{\text{hipotenusa}}$ con
        los valores conocidos en su sitio;
  \item resuelve la pequeña \omterm{def:g8:equations:def}{ecuación} para la incógnita (multiplica o divide);
  \item para un \emph{ángulo} desconocido, aplica el $\cos^{-1}$ de la
        calculadora a la razón calculada;
  \item comprobaciones de sentido común: una longitud tiene que salir más corta que la
        hipotenusa; un ángulo, estrictamente entre $0^\circ$ y
        $90^\circ$.
\end{enumerate}
\end{method}

\begin{example}[Hallar un lado]\label{ex:g8:cosine:side}
Una rampa de $6$ m forma un ángulo de $20^\circ$ con el suelo horizontal.
Distancia horizontal recorrida (contigua a $20^\circ$, con la hipotenusa
$6$ m):
\[
\cos 20^\circ = \frac{d}{6}
\quad\Longrightarrow\quad
d = 6 \cos 20^\circ \approx 6 \times 0.940 \approx 5.6 \text{ m}.
\]
\end{example}

\begin{example}[Hallar un ángulo]\label{ex:g8:cosine:angle}
En un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, el cateto contiguo al ángulo $\theta$ mide
$3.5$ y la hipotenusa, $5$:
\[
\cos\theta = \frac{3.5}{5} = 0.7,
\qquad
\theta = \cos^{-1}(0.7) \approx 45.6^\circ .
\]
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g8:cosine:1}
Un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} tiene una hipotenusa de $13$ cm. ¿Cuánto mide
la mediana desde el \omterm{def:g5:solids:def}{vértice} del \omterm{def:g3:shapes:rightangle}{ángulo recto}? ¿Cuál es el \omterm{def:g3:shapes:shapes}{radio} de la
circunferencia que pasa por los tres \omterm{def:g5:solids:def}{vértices}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:2}
Traza una circunferencia de \omterm{def:g4:geometry:circle}{diámetro} $8$ cm con un \omterm{def:g4:geometry:circle}{diámetro} $[BC]$, elige un
punto cualquiera $A$ de la circunferencia y dibuja $ABC$. ¿Qué ángulo es recto? Cita
el teorema. Mide $AB$ y $AC$ y comprueba Pitágoras.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:3}
En un triángulo $DEF$ rectángulo en $D$, nombra el cateto contiguo
al ángulo $\widehat E$, el cateto opuesto y la hipotenusa.
Escribe $\cos \widehat E$ como una razón.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:4}
Calcula la magnitud que falta ($\cos 35^\circ \approx 0.819$,
$\cos 50^\circ \approx 0.643$):
\begin{enumerate}
  \item hipotenusa $10$, ángulo $35^\circ$: ¿cateto contiguo?
  \item cateto contiguo $6$, ángulo $50^\circ$: ¿hipotenusa?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:5}
En un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, el cateto contiguo a $\theta$ mide $8$ y
la hipotenusa, $10$. Calcula $\cos\theta$ y después $\theta$
($\cos^{-1}(0.8) \approx 36.9^\circ$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:6}
Explica por qué $\cos\theta$ nunca puede valer $1.2$ para un ángulo agudo
de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:7}
Una tirolina de $25$ m baja desde una plataforma hasta el suelo formando un
ángulo de $12^\circ$ con la horizontal ($\cos 12^\circ \approx
0.978$). ¿Qué distancia horizontal cubre?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:8}
Ordena sin calculadora: $\cos 20^\circ$, $\cos 70^\circ$ y
$\cos 45^\circ$ (usa la \cref{prop:g8:cosine:bounds}). Compruébalo después con
calculadora.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:9}
Una escalera de longitud $L$ se apoya en una pared formando un ángulo $\theta$
entre la escalera y el \emph{suelo}. Su pie está a $1.2$ m de la pared
y $\theta = 68^\circ$ ($\cos 68^\circ \approx 0.375$). Calcula $L$
y después la altura alcanzada (Pitágoras).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:cosine:10}
$[BC]$ es un \omterm{def:g4:geometry:circle}{diámetro} de una circunferencia de centro $O$ y \omterm{def:g3:shapes:shapes}{radio} $4.5$ cm,
y $A$ es un punto de la circunferencia con $AB = 5.4$ cm.
\begin{enumerate}
  \item ¿por qué es $ABC$ rectángulo en $A$?
  \item calcula $AC$ y después $\cos \widehat B$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:cosine:11}
Usando la \omterm{def:g3:division:half}{mitad} de un \omterm{def:g6:shapes:triangles}{triángulo equilátero} de lado $1$ (cortado por una altura),
justifica la referencia $\cos 60^\circ = \frac12$ de la
\cref{prop:g8:cosine:bounds}. (¿Dónde cae el pie de la altura
sobre la base? La versión trigonométrica completa está en el
\cref{ex:g9:trig:special}.)
\end{exercise}

\section{Problema: Las relaciones de Euclides y Pitágoras otra
vez}

\begin{problem}[{Problema de fin de semana --- la altura de un triángulo
rectángulo: $AB^2 = BH \times BC$, $AH^2 = BH \times HC$, una segunda
demostración de Pitágoras y una máquina para construir raíces cuadradas}]
\label{pb:g8:cosine:1}
Traza, desde el \omterm{def:g3:shapes:rightangle}{ángulo recto} de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, la perpendicular
a la hipotenusa: ese \omterm{def:g6:lines:objects}{segmento} corto --- la \emph{altura} ---
cumple relaciones tan útiles que Euclides las puso en el corazón de
sus \emph{Elementos}. En este problema, la propiedad que define al coseno,
«la razón depende solo del ángulo»
(\cref{def:g8:cosine:def}), las demuestra todas; por el camino
volverás a demostrar el teorema de Pitágoras por una vía completamente distinta
y terminarás con una máquina de regla y compás que construye
$\sqrt2$, $\sqrt5$ y $\sqrt{n}$ para todo número entero $n$.

En todo el problema, $ABC$ es un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} en $A$ y $H$ es
el pie de la perpendicular trazada desde $A$ a la hipotenusa $[BC]$,
de modo que $H$ está entre $B$ y $C$ y $AH \perp BC$.

\medskip
\noindent\textbf{Parte I --- Un ángulo, dos triángulos.}
\begin{enumerate}
  \item usando la \omterm{def:g1:addition:def}{suma} de los ángulos de un triángulo
        (\cref{thm:g7:triangles:sum}) en $ABH$ y en $ABC$, demuestra
        que $\widehat{BAH} = \widehat{ACB}$: la altura corta
        el \omterm{def:g6:shapes:triangles}{triángulo rectángulo} en dos triángulos menores que llevan
        los \emph{mismos ángulos} que el original;
  \item los triángulos $ABC$ (rectángulo en $A$) y $ABH$
        (rectángulo en $H$) comparten el ángulo $\widehat B$. Escribe
        $\cos\widehat B$ en cada uno y deduce de la
        \cref{def:g8:cosine:def} que
        \[
        AB^2 = BH \times BC .
        \]
        (para pasar de razones iguales a \omterm{def:g2:mult:def}{productos} iguales, multiplica
        los dos miembros por los dos \omterm{def:g4:fractions:def}{denominadores},
        \cref{thm:g8:equations:balance}.)
  \item enuncia y demuestra la relación gemela en el \omterm{def:g5:solids:def}{vértice} $C$:
        \[
        AC^2 = CH \times CB .
        \]
  \item toma el triángulo $3$--$4$--$5$: $AB = 3$, $AC = 4$,
        $BC = 5$. Calcula $BH$ y $CH$ con las preguntas 2 y 3,
        y comprueba que $BH + HC = BC$;
  \item en general, expresa $BH$ y $CH$ como fracciones en las que solo
        aparezcan los tres lados. ¿En qué sentido reparte el pie $H$
        la hipotenusa «proporcionalmente a los cuadrados de
        los catetos»?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Pitágoras otra vez y la altura.}
\begin{enumerate}[resume]
  \item \omterm{def:g1:addition:def}{suma} las relaciones de las preguntas 2 y 3 y usa
        $BH + HC = BC$ para obtener
        \[
        AB^2 + AC^2 = BC^2 :
        \]
        el teorema de Pitágoras (\cref{thm:g8:pythagoras:direct}),
        demostrado de nuevo --- sin ningún rompecabezas de \omterm{def:g6:measure:area}{áreas} a la vista (compara el
        \cref{exo:g8:pythagoras:11}).
  \item aplica Pitágoras en el triángulo pequeño $ABH$ para escribir
        $AH^2 = AB^2 - BH^2$, sustituye $AB^2$ por
        $BH \times BC$ y saca factor común $BH$ para demostrar
        la \emph{relación de la altura de Euclides}:
        \[
        AH^2 = BH \times HC .
        \]
  \item calcula el \omterm{def:g6:measure:area}{área} de $ABC$ de dos maneras --- con los catetos como base
        y altura, y después con la hipotenusa como base
        (\cref{thm:g7:areas:triangle}) --- y deduce la tercera
        relación:
        \[
        AB \times AC = BC \times AH .
        \]
  \item de vuelta al triángulo $3$--$4$--$5$: calcula $AH$ con la
        pregunta 8 y verifica después numéricamente la relación de la altura de la pregunta
        7, usando los valores de $BH$ y $CH$ hallados en la
        pregunta 4;
  \item el pie de la altura parte la hipotenusa de cierto \omterm{def:g6:shapes:triangles}{triángulo
        rectángulo} en \omterm{def:g6:lines:objects}{segmentos} $BH = 2$ cm y $HC = 8$ cm.
        Calcula la altura $AH$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Una máquina de raíces cuadradas.}
Traza un \omterm{def:g6:lines:objects}{segmento} $[BC]$ formado por dos trozos puestos uno a continuación del otro:
$BH = p$ y $HC = q$. Traza la semicircunferencia de \omterm{def:g4:geometry:circle}{diámetro} $[BC]$
y sea $A$ el punto donde la perpendicular a $(BC)$ en $H$
la corta.
\begin{enumerate}[resume]
  \item usando el \cref{thm:g8:cosine:circle}, explica por qué el
        triángulo $ABC$ es rectángulo en $A$ --- así que se aplica la pregunta 7
        y
        \[
        AH^2 = p \times q .
        \]
        (la longitud $AH$ se llama la \emph{media
        geométrica}\index{media geométrica} de $p$ y $q$);
  \item toma $p = 1$ y $q = 2$. ¿Cuánto vale $AH^2$? ¿Qué longitud
        famosa del \cref{ex:g8:pythagoras:diagonal} acaba de construir tu
        figura de regla y compás?
  \item describe la receta que construye un \omterm{def:g6:lines:objects}{segmento} de longitud
        $\sqrt n$ para cualquier número entero $n \geq 1$, y di qué
        elección de $p$ y $q$ construye $\sqrt 5$;
  \item sea $O$ el centro de la semicircunferencia. Explica por qué
        $AH$ nunca puede superar el \omterm{def:g3:shapes:shapes}{radio} $\frac{p + q}{2}$ y
        para qué posición de $H$ los dos son iguales.
  \item deduce la desigualdad: para todos los números positivos $p$ y
        $q$,
        \[
        p \times q \leq \left(\frac{p + q}{2}\right)^2,
        \]
        con igualdad exactamente cuando $p = q$. Vuelve a demostrarla después sin
        nada de geometría: desarrolla
        $\left(\frac{p+q}{2}\right)^2 - p q$ y reconoce una
        identidad notable (\cref{pb:g8:equations:1}).
\end{enumerate}
\end{problem}
