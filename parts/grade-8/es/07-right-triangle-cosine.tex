\chapter{El triángulo rectángulo y el coseno}\label{ch:g8:cosine}

Dos hermosos hechos vinculan la \omterm{def:g6:shapes:triangles}{triángulo rectángulo} al círculo: un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} encaja exactamente en un semicírculo, siendo su hipotenusa una
\omterm{def:g4:geometry:circle}{diámetro}. Y un número, el \emph{coseno} de un ángulo, codifica la
forma de cada \omterm{def:g6:shapes:triangles}{triángulo rectángulo} con ese angulo --- el primero
razón trigonométrica, por delante de sus hermanos seno y tangente
(\cref{ch:g9:trig}).

\section{El triángulo rectángulo y su círculo.}

\begin{theorem}[Teorema del círculo]\label{thm:g8:cosine:circle}
\begin{enumerate}
  \item si un triángulo $ABC $ es \omterm{def:g6:shapes:triangles}{en ángulo recto} en $ A $, entonces $ A $ se encuentra en
        el círculo cuyo \omterm{def:g4:geometry:circle}{diámetro} es la hipotenusa $[a.\,C.]$.
  \item Por el contrario, si $ A $ se encuentra en un círculo de \omterm{def:g4:geometry:circle}{diámetro} $[a.\,C.]$ (con
        $ A \neq B, C $), entonces el triángulo $ ABC $ es \omterm{def:g6:shapes:triangles}{en ángulo recto} en
        $ A $.
\end{enumerate}
Equivalentemente: en un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, el punto medio de la hipotenusa es
a igual distancia de los tres \omterm{def:g5:solids:def}{vértices} --- la mediana de la
\omterm{def:g3:shapes:rightangle}{ángulo recto} medidas \omterm{def:g3:division:half}{medio} la hipotenusa.
\end{theorem}

\begin{proof}[Prueba de 1]
dejar $ O $ ser el punto medio de $[a.\,C.]$ y $ D $ el \omterm{def:g7:central:def}{simétrico} de $ A $ acerca de
$ O $. las diagonales de $ ABDC $ cortar en su punto medio común $ O $, entonces
$ ABDC $ es un \omterm{def:g7:central:parallelogram}{paralelogramo} (\cref{def:g7:central:parallelogram}) ---
con un \omterm{def:g3:shapes:rightangle}{ángulo recto} en $ A $: es un rectángulo. Las diagonales de un
rectángulos son iguales, entonces $ OA = \frac{d.\,C.}{2} = \frac{a.\,C.}{2}$: el punto $ A $ esta a distancia $\frac{a.\,C.}{2}$ de $ O $, es decir,\ en el círculo
de \omterm{def:g4:geometry:circle}{diámetro} $[a.\,C.]$. (El punto 2 se prueba ejecutando el argumento
hacia atrás.)
\end{proof}

\section{El coseno de un ángulo agudo.}

\begin{definition}[Coseno]\label{def:g8:cosine:def}
en un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, para un ángulo agudo $\theta $:
\[
\cos\theta = \frac{\text{side adjacent to } \theta}{\text{hypotenuse}}
\]
--- la pierna tocándose $\theta $, dividido por la hipotenusa. Esta relación
depende sólo del ángulo, no del tamaño del triángulo: todo \omterm{def:g6:shapes:triangles}{triangulos rectángulos} con el mismo ángulo agudo son ampliaciones entre sí,
y las ampliaciones preservan las proporciones de longitudes
(\cref{ch:g8:midpoints} comenzó esta historia; \cref{ch:g9:thales}
lo termina).
\end{definition}

ese valor común es $\cos\theta $.}
\end{omfigure}

\begin{proposition}[Primeros valores y límites.]\label{prop:g8:cosine:bounds}
Para cada ángulo agudo $\theta $: $0 < \cos\theta < 1$, y el coseno
\emph{disminuye} a medida que se abre el ángulo: un ángulo más amplio tiene un ángulo más pequeño
coseno. Puntos de referencia: $\cos 0^\circ = 1$, $\cos 60^\circ = \frac12$,
$\cos 90^\circ = 0$ (los valores extremos correspondientes a aplanados
triángulos).
\end{proposition}
\admitted

\begin{method}[Usando el coseno]\label{met:g8:cosine:use}
en un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, cuando las cantidades conocidas y deseadas son una
ángulo agudo, su lado adyacente y la hipotenusa:
\begin{enumerate}
  \item escribe la definición:
        $\cos\theta = \frac{\text{adjacent}}{\text{hypotenuse}}$ con
        los valores conocidos vigentes;
  \item resolver lo pequeño \omterm{def:g8:equations:def}{ecuación} por lo desconocido (multiplicar o dividir);
  \item por un desconocido \emph{ángulo}, aplica la calculadora
        $\cos^{-1}$ a la relación calculada;
  \item controles de cordura: una longitud debe ser más corta que la
        hipotenusa; un ángulo estrictamente entre $0^\circ $ y
        $90^\circ $.
\end{enumerate}
\end{method}

\begin{example}[Encontrar un lado]\label{ex:g8:cosine:side}
A $6$ La rampa m forma un ángulo de $20^\circ $ con el suelo horizontal.
Distancia horizontal recorrida (adyacente a $20^\circ $, hipotenusa
$6$ metro):
\[
\cos 20^\circ = \frac{d}{6}
\quad\Longrightarrow\quad
d = 6 \cos 20^\circ \approx 6 \times 0.940 \approx 5.6 \text{ m}.
\]
\end{example}

\begin{example}[Encontrar un ángulo]\label{ex:g8:cosine:angle}
en un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, el lado adyacente al ángulo $\theta $ medidas
$3.5$ y la hipotenusa $5$:
\[
\cos\theta = \frac{3.5}{5} = 0.7,
\qquad
\theta = \cos^{-1}(0.7) \approx 45.6^\circ .
\]
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g8:cosine:1}
A \omterm{def:g6:shapes:triangles}{triángulo rectángulo} tiene una hipotenusa de $13$ centímetro. ¿Cuál es la longitud de
la mediana desde el ángulo recto \omterm{def:g5:solids:def}{vértice}? cual es el \omterm{def:g3:shapes:shapes}{radio} de la
circula por los tres \omterm{def:g5:solids:def}{vértices}?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g8:cosine:2}
Dibuja un círculo de \omterm{def:g4:geometry:circle}{diámetro} $8$ centímetros con un \omterm{def:g4:geometry:circle}{diámetro} $[a.\,C.]$, elige cualquiera
punto $ A $ en el círculo y dibujar $ ABC $. ¿Qué ángulo es el correcto? citar
el teorema. Medida $ AB $ y $ AC $ y mira a Pitágoras.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g8:cosine:3}
en un triángulo $ DEF $ \omterm{def:g6:shapes:triangles}{en ángulo recto} en $ D $, nombra el lado adyacente a
el ángulo $\widehat E $, el lado opuesto y la hipotenusa.
Escribir $\cos \widehat E $ como una proporción.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g8:cosine:4}
Calcule la cantidad que falta ($\cos 35^\circ \approx 0.819$,
$\cos 50^\circ \approx 0.643$):
\begin{enumerate}
  \item hipotenusa $10$, ángulo $35^\circ $: lado adyacente?
  \item lado adyacente $6$, ángulo $50^\circ $: hipotenusa?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g8:cosine:5}
en un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, el lado adyacente a $\theta $ medidas $8$ y la hipotenusa $10$. Calcular $\cos\theta $, entonces $\theta $
($\cos^{-1}(0.8) \approx 36.9^\circ $).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g8:cosine:6}
Explica por qué $\cos\theta $ nunca puede igualar $1.2$ para un ángulo agudo de
un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g8:cosine:7}
A $25$ m desciende desde una plataforma hasta el suelo, haciendo un
ángulo de $12^\circ $ con la horizontal ($\cos 12^\circ \approx
0.978$). ¿Qué distancia horizontal abarca?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g8:cosine:8}
Ordene sin calculadora: $\cos 20^\circ $, $\cos 70^\circ $,
$\cos 45^\circ $ (usar \cref{prop:g8:cosine:bounds}). Luego consulte con un
calculadora.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g8:cosine:9}
Una escalera de longitud $ L $ se apoya contra una pared con un ángulo $\theta $
entre escalera y \emph{suelo}. Su pie es $1.2$ m de la pared
y $\theta = 68^\circ $ ($\cos 68^\circ \approx 0.375$). Calcular $ L $,
luego alcanzó la altura (Pitágoras).
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g8:cosine:10}
$[a.\,C.]$ es un \omterm{def:g4:geometry:circle}{diámetro} de un círculo de centro $ O $ y \omterm{def:g3:shapes:shapes}{radio} $4.5$ centímetros, y $ A $ es un punto del círculo con $ AB = 5.4$ centímetro.
\begin{enumerate}
  \item ¿Por qué es $ ABC $ \omterm{def:g6:shapes:triangles}{en ángulo recto} en $ A $?
  \item Calcular $ AC $, entonces $\cos \widehat B $.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g8:cosine:11}
Usando \omterm{def:g3:division:half}{medio} un \omterm{def:g6:shapes:triangles}{triángulo equilatero} de lado $1$ (cortado a lo largo de una altura),
justificar el hito $\cos 60^\circ = \frac12$ de
\cref{prop:g8:cosine:bounds}. (¿De dónde viene el pie de la altura?
caer sobre la base? La versión trigonométrica completa es
\cref{ex:g9:trig:special}.)
\end{exercise}

\section{Problema: las relaciones de Euclides y Pitágoras por todos lados
otra vez}

\begin{problem}[{Problema de fin de semana --- la altura de un triángulo rectángulo: $AB^2 = BH \times a.\,C.$, $AH^2 = BH \times HC$, una segunda prueba de Pitágoras y una máquina para construir raíces cuadradas}]
\label{pb:g8:cosine:1}
Caída, desde el \omterm{def:g3:shapes:rightangle}{ángulo recto} de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, el \omterm{def:g6:lines:perp}{perpendicular}
a la hipotenusa: esta corta \omterm{def:g6:lines:objects}{segmento} --- el \emph{altitud} ---
satisface relaciones tan útiles que Euclides las puso en el centro de
su \emph{Elementos}. En este problema el coseno define
propiedad, ``la relación depende sólo del ángulo''
(\cref{def:g8:cosine:def}), los prueba todos; en el camino tu
volverá a demostrar el teorema de Pitágoras por una ruta completamente diferente,
y terminar con una máquina de regla y compás que construye
$\sqrt2$, $\sqrt5$, $\sqrt{n}$ por cada número entero $ n $.

A lo largo de, $ ABC $ es un triángulo \omterm{def:g6:shapes:triangles}{en ángulo recto} en $ A $, y $ H $ es
el pie de la \omterm{def:g6:lines:perp}{perpendicular} de $ A $ a la hipotenusa $[a.\,C.]$, de modo que $ H $ se encuentra entre $ B $ y $ C $ y $ AH \perp a.\,C. $.

\medskip
\noindent\textbf{Parte I --- One angle, two triangles.}
\begin{enumerate}
  \item Usando el ángulo \omterm{def:g1:addition:def}{suma} de un triángulo
        (\cref{thm:g7:triangles:sum}) en $ ABH $ y en $ ABC $, muestra que $\widehat{BAH} = \widehat{ACB}$: la altitud corta
        el \omterm{def:g6:shapes:triangles}{triángulo rectángulo} en dos triángulos más pequeños que llevan
        el \emph{mismos ángulos} como el original.
  \item los triangulos $ ABC $ (\omterm{def:g6:shapes:triangles}{en ángulo recto} en $ A $) y $ ABH $
        (\omterm{def:g6:shapes:triangles}{en ángulo recto} en $ H $) comparte el ángulo $\widehat B $. Escribir
        $\cos\widehat B $ en cada uno de ellos, y deducir de
        \cref{def:g8:cosine:def} eso
        \[
        AB^2 = BH \times a.\,C. .
        \]
        (Para pasar de proporciones iguales a proporciones iguales \omterm{def:g2:mult:def}{productos}, multiplicar
        ambos lados por ambos \omterm{def:g4:fractions:def}{denominadores},
        \cref{thm:g8:equations:balance}.)
  \item Enunciar y demostrar la relación de gemela en el \omterm{def:g5:solids:def}{vértice} $ C $:
        \[
        AC^2 = CH \times CB .
        \]
  \item Toma el $3$--$4$--$5$ triángulo: $ AB = 3$, $ AC = 4$,
        $ a.\,C. = 5$. Calcular $ BH $ y $ CH $ de las preguntas 2 y 3,
        y comprueba eso $ BH + HC = a.\,C. $.
  \item En general, expresar $ BH $ y $ CH $ como \omterm{def:g6:fractions:def}{fracciones} involucrando
        sólo los tres lados. ¿En qué sentido el pie $ H $
        dividir la hipotenusa "proporcionalmente a los cuadrados de
        las piernas''?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Pythagoras again, and the altitude.}
\begin{enumerate}[resume]
  \item Suma las relaciones de las preguntas 2 y 3 y usa
        $ BH + HC = a.\,C. $ obtener
        \[
        AB^2 + AC^2 = a.\,C.^2 :
        \]
        Teorema de Pitágoras (\cref{thm:g8:pythagoras:direct}),
        reprobado --- sin \omterm{def:g6:measure:area}{área} rompecabezas a la vista (comparar
        \cref{exo:g8:pythagoras:11}).
  \item Aplicar Pitágoras en el triángulo pequeño. $ ABH $ escribir
        $ AH^2 = AB^2 - BH^2$, reemplazar $ AB^2$ por
        $ BH \times a.\,C. $y factorizar $ BH $ probar
        \emph{Relación de altitud de Euclides}:
        \[
        AH^2 = BH \times HC .
        \]
  \item Calcular el \omterm{def:g6:measure:area}{área} de $ ABC $ de dos maneras --- patas como base
        y altura, luego hipotenusa como base
        (\cref{thm:g7:areas:triangle}) --- y deducir el tercero
        relación:
        \[
        AB \times AC = a.\,C. \times AH .
        \]
  \item Volver a la $3$--$4$--$5$ triángulo: calcular $ AH $ con
        pregunta 8, luego verifique la relación de altitud de la pregunta
        7 numéricamente, usando los valores de $ BH $ y $ CH $ encontrado en
        pregunta 4.
  \item El pie de la altitud divide la hipotenusa de algunos. \omterm{def:g6:shapes:triangles}{triángulo rectángulo} en \omterm{def:g6:lines:objects}{segmentos} $ BH = 2$ centímetros y $ HC = 8$ centímetro.
        Calcular la altitud $ AH $.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A machine for square roots.}
dibuja un \omterm{def:g6:lines:objects}{segmento} $[a.\,C.]$ formado por dos piezas colocadas una al lado de la otra:
$ BH = p $ y $ HC = q $. Dibuja el semicírculo de \omterm{def:g4:geometry:circle}{diámetro} $[a.\,C.]$, y deja $ A $ ser el punto donde el \omterm{def:g6:lines:perp}{perpendicular} a $(a.\,C.)$ en $ H $
lo cumple.
\begin{enumerate}[resume]
  \item Usando \cref{thm:g8:cosine:circle}, explica por qué el
        triángulo $ ABC $ es \omterm{def:g6:shapes:triangles}{en ángulo recto} en $ A $ --- entonces pregunta 7
        se aplica y
        \[
        AH^2 = p \times q .
        \]
        (La longitud $ AH $ se llama el \emph{geométrico
        significar}\index{geométrico
        significar} de $ p $ y $ q $.)
  \item Llevar $ p = 1$ y $ q = 2$. Qué es $ AH^2$? cual famoso
        longitud de \cref{ex:g8:pythagoras:diagonal} tiene tu
        ¿Se acaba de construir la figura de regla y compás?
  \item Describe la receta que construye un \omterm{def:g6:lines:objects}{segmento} de longitud
        $\sqrt n $ para cualquier número entero $ n \geq 1$y decir cual
        elección de $ p $ y $ q $ construye $\sqrt 5$.
  \item Sea $ O $ el centro del semicírculo. Explica por qué
        $ AH $ nunca puede exceder el \omterm{def:g3:shapes:shapes}{radio} $\frac{p + q}{2}$, y
        para qué posición de $ H $ los dos son iguales.
  \item Deducir la desigualdad: para todos los números positivos $ p $ y
        $ q $,
        \[
        p \times q \leq \left(\frac{p + q}{2}\right)^2,
        \]
        con igualdad exactamente cuando $ p = q $. Luego repréndelo con
        sin geometría en absoluto: expandir
        $\left(\frac{p+q}{2}\right)^2 - p q$ y reconocer un
        identidad notable (\cref{pb:g8:equations:1}).
\end{enumerate}
\end{problem}
