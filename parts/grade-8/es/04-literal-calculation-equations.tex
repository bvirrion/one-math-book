\chapter{Cálculo literal y ecuaciones}\label{ch:g8:equations}

El año pasado presentamos las letras y redujimos expresiones sencillas
(\cref{ch:g7:literal}). Este capítulo añade las dos destrezas que hacen
poderosa al álgebra: \emph{desarrollar} \omterm{def:g2:mult:def}{productos} de paréntesis y
\emph{resolver ecuaciones} --- hallar el número escondido detrás de la
letra.

\section{Desarrollar}

\begin{theorem}[Distributividad simple y doble]\label{thm:g8:equations:expand}
Para todos los números:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{theorem}

\begin{proof}
La primera regla es el \cref{thm:g7:priorities:distributivity} (con los signos
incluidos: $k$ y los términos pueden ser negativos,
\cref{ch:g8:negprod}). Para la segunda, aplica dos veces la primera regla
tratando $(c + d)$ como un solo número:
\[
(a + b)(c + d) = a(c + d) + b(c + d) = ac + ad + bc + bd .
\qedhere
\]
\end{proof}

\begin{example}\label{ex:g8:equations:expand}
Cuidado con los signos en cada término:
\begin{align*}
3(2x - 5) &= 6x - 15, \\
-2(4x - 7) &= -8x + 14
&& \text{(un menos delante cambia los dos signos)}, \\
(x + 3)(x + 5) &= x^2 + 5x + 3x + 15 = x^2 + 8x + 15, \\
(2x - 1)(x - 4) &= 2x^2 - 8x - x + 4 = 2x^2 - 9x + 4 .
\end{align*}
\end{example}

\begin{method}[Desarrollar y reducir]\label{met:g8:equations:reduce}
\begin{enumerate}
  \item desarrolla cada \omterm{def:g2:mult:def}{producto} escribiendo cada término con su signo;
  \item agrupa los términos semejantes ($x^2$ con $x^2$, $x$ con $x$, números
        con números);
  \item comprueba con un valor: sustituye, por ejemplo, $x = 2$ en la expresión original
        y en el resultado --- si coinciden, quedan cazados casi todos los errores.
\end{enumerate}
\end{method}

\section{Resolver ecuaciones}

\begin{definition}[Ecuación]\label{def:g8:equations:def}
Una \emph{ecuación}\index{ecuación} es una igualdad con un número
desconocido, que se escribe $x$ (o cualquier letra). \emph{Resolverla} es hallar
todos los valores de $x$ que hacen cierta la igualdad --- las
\emph{soluciones}.
\end{definition}

\begin{theorem}[Reglas de la balanza]\label{thm:g8:equations:balance}
Una \omterm{def:g8:equations:def}{ecuación} conserva exactamente las mismas soluciones cuando:
\begin{enumerate}
  \item se \omterm{def:g1:addition:def}{suma} (o se resta) el mismo número a los dos miembros;
  \item se multiplican (o se dividen) los dos miembros por un mismo
        número \emph{no nulo}.
\end{enumerate}
\end{theorem}
\admitted

\begin{method}[Resolver una ecuación de primer grado]\label{met:g8:equations:solve}
\begin{enumerate}
  \item desarrolla y reduce cada miembro si hace falta;
  \item \omterm{def:g1:addition:def}{suma} o resta en los dos miembros para reunir los términos en $x$ en un
        lado y los números solos en el otro;
  \item reduce y divide después los dos miembros entre el coeficiente de $x$;
  \item \emph{comprueba} sustituyendo el valor hallado en la \omterm{def:g8:equations:def}{ecuación}
        original.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:equations:solve}
Resuelve $7x - 5 = 4x + 16$:
\begin{align*}
7x - 5 &= 4x + 16 \\
7x - 4x &= 16 + 5
&& \text{(suma $5$ y resta $4x$ en los dos miembros)} \\
3x &= 21 \\
x &= 7 && \text{(divide los dos miembros entre 3).}
\end{align*}
Comprobación: primer miembro $7 \times 7 - 5 = 44$; segundo miembro
$4 \times 7 + 16 = 44$. La solución es $7$.
\end{example}

\begin{example}[Con paréntesis y negativos]\label{ex:g8:equations:brackets}
Resuelve $3(x - 4) = 5 - 2(x + 1)$:
\begin{align*}
3x - 12 &= 5 - 2x - 2 && \text{(desarrolla los dos miembros)} \\
3x - 12 &= 3 - 2x && \text{(reduce)} \\
5x &= 15 && \text{(suma $2x + 12$ a los dos miembros)} \\
x &= 3 .
\end{align*}
Comprobación: $3(3 - 4) = -3$ y $5 - 2(3+1) = 5 - 8 = -3$. \quad Solución:
$3$.
\end{example}

\begin{method}[Resolver un problema con una ecuación]\label{met:g8:equations:problem}
\begin{enumerate}
  \item elige la incógnita: «sea $x$ \dots» (la cantidad que se pide
        o una relacionada más sencilla);
  \item traduce la historia a una \omterm{def:g8:equations:def}{ecuación};
  \item resuélvela;
  \item interpreta: responde a la pregunta original con una frase, con
        unidades, y contrasta la respuesta con la historia.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:equations:problem}
Lucas tiene $4$ años más que su hermana; entre los dos suman
$26$ años. Sea $x$ la edad de la hermana; entonces Lucas tiene $x + 4$ y
\[
x + (x + 4) = 26
\ \Longrightarrow\
2x + 4 = 26
\ \Longrightarrow\
2x = 22
\ \Longrightarrow\
x = 11 .
\]
La hermana tiene $11$ y Lucas, $15$. Comprobación: $11 + 15 = 26$ y
$15 - 11 = 4$.
\end{example}

\section{Orden y desigualdades}

\begin{proposition}[Operaciones y orden]\label{prop:g8:equations:order}
Sumar el mismo número a los dos miembros de una desigualdad la conserva;
multiplicar los dos miembros por un número \emph{positivo} la conserva;
multiplicar por un número \emph{negativo} la \emph{invierte}:
\[
2 < 5 \ \text{ pero }\ (-1) \times 2 > (-1) \times 5 .
\]
\end{proposition}
\admitted

\begin{example}\label{ex:g8:equations:inequality}
Resuelve $5 - 2x \leq 11$: resta $5$ en los dos miembros, $-2x \leq 6$;
divide entre $-2$ \emph{e invierte}: $x \geq -3$. Todos los números
mayores o iguales que $-3$ son soluciones --- una \omterm{def:g8:equations:def}{ecuación} suele
tener una solución y una inecuación tiene toda una semirrecta de ellas.
\end{example}

\section{Problema: Las identidades notables y la escalera de los números
impares}

\begin{problem}[{Problema de fin de semana --- $(a+b)^2$, $(a-b)^2$,
$(a+b)(a-b)$ y la suma $1 + 3 + 5 + \dots + (2n-1) = n^2$}]
\label{pb:g8:equations:1}
Tres desarrollos aparecen tan a menudo que el álgebra se los sabe de
memoria: se llaman \emph{identidades notables}. Este
problema las deduce de la distributividad doble
(\cref{thm:g8:equations:expand}), las convierte en un superpoder de cálculo mental,
las usa para demostrar un hecho precioso sobre los \omterm{def:g3:numbers:evenodd}{números impares}
y termina resolviendo ecuaciones de un tipo nuevo: ecuaciones con un
cuadrado.

\medskip
\noindent\textbf{Parte I --- Las tres identidades.}
\begin{enumerate}
  \item desarrolla $(a + b)^2 = (a + b)(a + b)$ y reduce, para demostrar
        \[
        (a + b)^2 = a^2 + 2ab + b^2 .
        \]
  \item demuestra igualmente las dos compañeras:
        \[
        (a - b)^2 = a^2 - 2ab + b^2,
        \qquad
        (a + b)(a - b) = a^2 - b^2 .
        \]
  \item dibuja un cuadrado de lado $a + b$, parte cada lado en $a$
        y $b$, y corta el cuadrado en cuatro piezas. ¿Qué parte
        de la primera identidad representa cada trozo del
        dibujo?
  \item cálculo mental: usa las identidades para calcular, sin
        plantear ninguna multiplicación,
        \[
        101^2, \qquad 99^2, \qquad 49 \times 51 .
        \]
  \item una trampa clásica: Zoe escribe «$(a + b)^2 = a^2 + b^2$».
        Prueba su fórmula con $a = 3$ y $b = 4$, y explica en el
        dibujo de la pregunta 3 exactamente qué trozos olvida
        su fórmula.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La escalera de los \omterm{def:g3:numbers:evenodd}{números impares}.}
\begin{enumerate}[resume]
  \item calcula $1$, después $1 + 3$, después $1 + 3 + 5$, después
        $1 + 3 + 5 + 7$ y después $1 + 3 + 5 + 7 + 9$. ¿Qué
        observas en los resultados?
  \item explica por qué el $n$-ésimo \omterm{def:g3:numbers:evenodd}{número impar} es $2n - 1$ (comprueba:
        $n = 1, 2, 3$ dan $1, 3, 5$);
  \item usa una identidad de la Parte I para desarrollar y reducir
        $(n + 1)^2 - n^2$. ¿Qué tiene que ver el resultado con
        la pregunta 7?
  \item la pregunta 8 dice: para pasar de un cuadrado de lado $n$ a un
        cuadrado de lado $n + 1$ se añaden exactamente el siguiente \omterm{def:g3:numbers:evenodd}{número
        impar} de casillas unidad --- una franja en forma de L a lo largo de dos
        lados y la esquina. Partiendo de $1 = 1^2$ y haciendo crecer
        el cuadrado capa a capa, explica por qué
        \[
        1 + 3 + 5 + \dots + (2n - 1) = n^2 :
        \]
        la \omterm{def:g1:addition:def}{suma} de los $n$ primeros \omterm{def:g3:numbers:evenodd}{números impares} es el $n$-ésimo cuadrado;
  \item deduce, sin sumarlos uno a uno: el valor de
        $1 + 3 + 5 + \dots + 99$; y cuántos \omterm{def:g3:numbers:evenodd}{números impares}, empezando
        por $1$, suman exactamente $400$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Ecuaciones con un cuadrado.}
\begin{enumerate}[resume]
  \item reconoce una identidad notable en $x^2 + 6x + 9$ y
        resuelve la \omterm{def:g8:equations:def}{ecuación} $x^2 + 6x + 9 = 0$;
  \item explica, usando la regla del signo y la distancia del
        \cref{thm:g8:negprod:rules}, por qué un \omterm{def:g2:mult:def}{producto} de dos números
        puede ser \omterm{def:g1:counting:zero}{cero} \emph{solo si} uno de los dos factores es \omterm{def:g1:counting:zero}{cero}.
        Úsalo para resolver $(x - 2)(x + 5) = 0$;
  \item resuelve $x^2 = 49$ pasando todo a un miembro y
        factorizando $x^2 - 49$ con la tercera identidad. ¿Cuántas
        soluciones tiene esta \omterm{def:g8:equations:def}{ecuación} y en qué se \omterm{ex:g1:subtraction:difference}{diferencia} eso
        de todas las ecuaciones resueltas hasta ahora en este capítulo?
  \item en el \cref{exo:g8:equations:3} comprobaste que $x = 3$ es
        solución de $x^2 = 6x - 9$. Demuestra que es la
        \emph{única}: pasa todo a un miembro, reconoce
        una identidad y concluye;
  \item final: calcula $2026^2 - 2025^2$ de cabeza y
        explica con una identidad por qué la \omterm{ex:g1:subtraction:difference}{diferencia} de dos
        cuadrados consecutivos es siempre la \omterm{def:g1:addition:def}{suma} de los dos números
        --- que es otra vez la pregunta 8.
\end{enumerate}
\end{problem}
