\chapter{Fracciones: las cuatro operaciones}\label{ch:g8:fractions}

Este capítulo completa la aritmética de las fracciones: la \omterm{def:g1:addition:def}{suma} y la
resta con \emph{cualesquiera} \omterm{def:g4:fractions:def}{denominadores}, la multiplicación y ---
la recién llegada --- la división, que resulta ser una multiplicación
disfrazada. Los signos del \cref{ch:g8:negprod} vienen de acompañantes.

\section{Sumar con denominadores cualesquiera}

\begin{method}[Denominador común]\label{met:g8:fractions:add}
Para sumar o restar $\frac ab$ y $\frac cd$:
\begin{enumerate}
  \item busca un \omterm{def:g5:division:multiple}{múltiplo} común de $b$ y $d$ (el \omterm{def:g2:mult:def}{producto}
        $b \times d$ vale siempre; uno más pequeño ahorra trabajo);
  \item reescribe las dos fracciones con ese \omterm{def:g4:fractions:def}{denominador};
  \item \omterm{def:g1:addition:def}{suma} o resta los \omterm{def:g4:fractions:def}{numeradores} y simplifica.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:fractions:add}
Calcula $\dfrac{5}{6} + \dfrac{3}{4}$. Un \omterm{def:g5:division:multiple}{múltiplo} común de $6$ y
$4$ es $12$:
\[
\frac{5}{6} = \frac{10}{12},
\qquad
\frac{3}{4} = \frac{9}{12},
\qquad
\frac{5}{6} + \frac{3}{4} = \frac{10 + 9}{12} = \frac{19}{12}.
\]
Con signos: $\dfrac{2}{3} - \dfrac{7}{5} = \dfrac{10}{15} -
\dfrac{21}{15} = -\dfrac{11}{15}$.
\end{example}

\section{Multiplicar fracciones}

\begin{theorem}[Producto de fracciones]\label{thm:g8:fractions:mult}
\[
\frac{a}{b} \times \frac{c}{d} = \frac{a \times c}{b \times d} .
\]
\end{theorem}

\begin{proof}[Por qué, con un dibujo]
Toma $\frac34$ de $\frac25$ de un cuadrado: corta el cuadrado en $5$
franjas verticales y quédate con $2$; corta en horizontal en $4$ y quédate con $3$
filas de lo que quedaba. La parte conservada es una rejilla de $3 \times 2$ casillas
pequeñas de $4 \times 5$: la \omterm{def:g6:fractions:def}{fracción} $\frac{6}{20}$.
\end{proof}

\begin{omfigure}
\begin{tikzpicture}[scale=0.68]
  \draw[gray!50, thin] (0,0) grid[xstep=1, ystep=1] (5,4);
  \fill[omDef!18] (0,0) rectangle (2,4);
  \fill[omThm!35] (0,0) rectangle (2,3);
  \draw[thick] (0,0) rectangle (5,4);
  \draw[omDef, thick] (2,0) -- (2,4);
  \draw[omThm, thick] (0,3) -- (5,3);
  \node[below, font=\small, omDef] at (1,0) {$\frac25$ conservado};
  \node[right, font=\small, omThm] at (5.05,1.5) {$\frac34$ de ello};
\end{tikzpicture}

{\small $\frac34 \times \frac25$: la región doblemente sombreada tiene
$3 \times 2 = 6$ casillas de $4 \times 5 = 20$, es decir,
$\frac{6}{20} = \frac{3}{10}$ --- \omterm{def:g4:fractions:def}{numeradores} multiplicados y \omterm{def:g4:fractions:def}{denominadores}
multiplicados.}
\end{omfigure}

\begin{example}\label{ex:g8:fractions:mult}
Simplifica \emph{antes} de multiplicar, tachando factores comunes:
\[
\frac{7}{12} \times \frac{9}{14}
= \frac{7 \times 9}{12 \times 14}
= \frac{7 \times 3 \times 3}{3 \times 4 \times 2 \times 7}
= \frac{3}{8}
\]
(el factor $7$ y un factor $3$ se cancelan entre arriba y abajo).
Con signos, se aplican primero las reglas del \cref{thm:g8:negprod:rules}:
$\left(-\frac23\right) \times \left(-\frac{5}{4}\right) =
+\frac{10}{12} = \frac56$.
\end{example}

\section{Dividir fracciones}

\begin{definition}[Inverso]\label{def:g8:fractions:inverse}
El \emph{inverso}\index{inverso} de un número $x$ no nulo es el
número que, multiplicado por $x$, da $1$. El inverso de
$\frac ab$ es $\frac ba$, ya que
$\frac ab \times \frac ba = \frac{ab}{ab} = 1$; el inverso de un
entero $n$ es $\frac1n$.
\end{definition}

\begin{theorem}[Dividir es multiplicar por el inverso]\label{thm:g8:fractions:div}
Para $c \neq 0$:
\[
\frac{a}{b} \div \frac{c}{d} = \frac{a}{b} \times \frac{d}{c} .
\]
\end{theorem}

\begin{proof}
Por definición, el \omterm{def:g3:division:remainder}{cociente} $q$ de $\frac ab$ entre $\frac cd$ es el
número tal que $q \times \frac cd = \frac ab$. Comprobemos el candidato
$q = \frac ab \times \frac dc$:
\[
\left(\frac ab \times \frac dc\right) \times \frac cd
= \frac ab \times \left(\frac dc \times \frac cd\right)
= \frac ab \times 1 = \frac ab . \qedhere
\]
\end{proof}

\begin{example}\label{ex:g8:fractions:div}
\[
\frac{3}{5} \div \frac{7}{10}
= \frac{3}{5} \times \frac{10}{7}
= \frac{3 \times 10}{5 \times 7}
= \frac{30}{35} = \frac{6}{7}.
\]
Una comprobación de sentido común: dividir entre $\frac{7}{10}$, un número menor que
$1$, tiene que dar un resultado \emph{mayor} que $\frac35$ --- y
$\frac67 > \frac35$, en efecto ($\frac{30}{35} > \frac{21}{35}$).
\end{example}

\begin{example}[Barras de fracción dentro de barras de fracción]\label{ex:g8:fractions:complex}
Un «piso doble» no es más que una división escrita en vertical:
\[
\frac{\ \frac{2}{3}\ }{\ \frac{5}{6}\ }
= \frac{2}{3} \div \frac{5}{6}
= \frac{2}{3} \times \frac{6}{5}
= \frac{12}{15} = \frac{4}{5}.
\]
Localiza primero la barra \emph{principal} (la más larga) y aplica después el
\cref{thm:g8:fractions:div}.
\end{example}

\begin{method}[Cálculos mixtos]\label{met:g8:fractions:mixed}
En una expresión que mezcla fracciones y las cuatro operaciones:
\begin{enumerate}
  \item aplica la jerarquía habitual (\cref{ch:g7:priorities});
  \item convierte cada división en una multiplicación por el \omterm{def:g8:fractions:inverse}{inverso};
  \item decide primero los signos (\cref{met:g8:negprod:compute}) y después
        multiplica o busca \omterm{def:g4:fractions:def}{denominadores} comunes;
  \item simplifica en cuanto puedas.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:fractions:mixed}
\begin{align*}
\frac12 + \frac{2}{3} \times \frac{9}{4}
&= \frac12 + \frac{2 \times 9}{3 \times 4}
&& \text{(¡primero el producto!)} \\
&= \frac12 + \frac{18}{12} = \frac12 + \frac32
= \frac{4}{2} = 2 .
\end{align*}
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g8:fractions:1}
Calcula y simplifica:
\[
\frac{3}{4} + \frac{2}{5}, \qquad
\frac{5}{6} - \frac{3}{8}, \qquad
\frac{7}{10} + \frac{8}{15} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:2}
Calcula (primero los signos):
\[
-\frac{2}{7} + \frac{3}{14}, \qquad
\frac{1}{6} - \frac{5}{4}, \qquad
-\frac{3}{5} - \frac{1}{2} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:3}
Calcula, simplificando antes de multiplicar:
\[
\frac{5}{8} \times \frac{4}{15}, \qquad
\frac{9}{14} \times \frac{7}{3}, \qquad
\left(-\frac{6}{5}\right) \times \frac{10}{9} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:4}
Da el \omterm{def:g8:fractions:inverse}{inverso} de: $\dfrac{3}{7}$; \quad $5$; \quad
$\dfrac{1}{4}$; \quad $-\dfrac{2}{9}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:5}
Calcula:
\[
\frac{4}{9} \div \frac{2}{3}, \qquad
\frac{7}{5} \div 14, \qquad
6 \div \frac{3}{4} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:6}
Calcula los pisos dobles:
\[
\frac{\ \frac{3}{4}\ }{\ \frac{9}{8}\ },
\qquad
\frac{\ \frac{5}{6}\ }{\ 10\ },
\qquad
\frac{\ 2\ }{\ \frac{4}{7}\ } .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:fractions:7}
Calcula respetando la jerarquía:
\[
\frac{1}{3} + \frac{1}{2} \times \frac{4}{5},
\qquad
\left(\frac{1}{3} + \frac{1}{2}\right) \times \frac{4}{5} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:fractions:8}
Un depósito está lleno hasta $\frac{2}{5}$. Se añade $\frac{1}{3}$ \emph{de la
\omterm{def:g2:measure:units}{capacidad} del depósito}. ¿Qué \omterm{def:g6:fractions:def}{fracción} del depósito está llena ahora? ¿Qué
\omterm{def:g6:fractions:def}{fracción} sigue vacía?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:fractions:9}
Tres \omterm{def:g3:division:half}{cuartos} de los alumnos de una clase aprobaron un examen; de ellos,
dos tercios sacaron más de $14$ sobre $20$. ¿Qué \omterm{def:g6:fractions:def}{fracción} de la clase
sacó más de $14$? Si eso son $12$ alumnos, ¿cuántos alumnos tiene
la clase?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:fractions:10}
Una cuerda de $\frac{15}{2}$ m se tiene que cortar en trozos de $\frac{3}{4}$ m
cada uno. ¿Cuántos trozos salen? (Una división de fracciones ---
comprueba que la respuesta es un número entero.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:fractions:11}
Calcula
\[
E = \frac{2}{3} - \frac{5}{6} \div \frac{10}{9},
\qquad
F = \left(-\frac{1}{2}\right)^2 \times \frac{8}{3} .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:fractions:12}

\]

\end{exercise}

\section{Problema: Las fracciones egipcias}

\begin{problem}[{Problema de fin de semana --- toda fracción es una suma de
fracciones unitarias distintas}]
\label{pb:g8:fractions:1}
Una \emph{\omterm{def:g6:fractions:def}{fracción} unitaria} es una \omterm{def:g6:fractions:def}{fracción} de \omterm{def:g4:fractions:def}{numerador} $1$. Los
escribas del antiguo Egipto --- el papiro Rhind se copió hacia el
$1550$ a.\,C. --- escribían toda \omterm{def:g6:fractions:def}{fracción} como \omterm{def:g1:addition:def}{suma} de fracciones
unitarias \emph{distintas}: nunca $\frac27 = \frac17 + \frac17$, sino
$\frac27 = \frac14 + \frac{1}{28}$. Una \omterm{def:g1:addition:def}{suma} así se llama una
\emph{escritura egipcia} de la \omterm{def:g6:fractions:def}{fracción}. Este problema convierte las
técnicas de \omterm{def:g1:addition:def}{suma} y comparación de este capítulo
(\cref{met:g8:fractions:add}) en un método --- publicado por
Fibonacci en $1202$ --- que produce una escritura egipcia de cualquier
\omterm{def:g6:fractions:def}{fracción} entre $0$ y $1$, y termina con un célebre
problema de herencia.

\medskip
\noindent\textbf{Parte I --- Fracciones unitarias y la identidad de
partición.}
\begin{enumerate}
  \item calcula $\frac12 + \frac13$ y $\frac12 + \frac14$, y
        deduce escrituras egipcias de $\frac56$ y $\frac34$;
  \item verifica la entrada de los escribas para $\frac27$: muestra que
        $\frac14 + \frac{1}{28} = \frac27$.
  \item comprueba que $\frac13 + \frac16 = \frac12$ y que
        $\frac14 + \frac{1}{12} = \frac13$. Demuestra después la
        \emph{identidad de partición}: para todo número entero
        $n \geq 1$,
        \[
        \frac{1}{n} = \frac{1}{n+1} + \frac{1}{n(n+1)} .
        \]
  \item deduce una escritura egipcia del propio $1$, como \omterm{def:g1:addition:def}{suma} de
        tres fracciones unitarias distintas;
  \item parte el último término de $\frac56 = \frac12 + \frac13$ para
        obtener una \emph{segunda} escritura egipcia de $\frac56$, con
        tres fracciones unitarias. Explica por qué ninguna \omterm{def:g6:fractions:def}{fracción} tiene una
        escritura egipcia única.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El método voraz de Fibonacci.}
El método es: resta la \emph{mayor} \omterm{def:g6:fractions:def}{fracción} unitaria que
quepa y repite con lo que quede.
\begin{enumerate}[resume]
  \item usando \omterm{def:g4:fractions:def}{denominadores} comunes, muestra que
        $\frac14 < \frac27 < \frac13$. ¿Por qué demuestra esto que
        $\frac14$ es la mayor \omterm{def:g6:fractions:def}{fracción} unitaria menor que
        $\frac27$?
  \item calcula $\frac27 - \frac14$. ¿Qué escritura de $\frac27$
        produce el método voraz?
  \item toma ahora $\frac57$. Muestra que la mayor \omterm{def:g6:fractions:def}{fracción} unitaria
        menor que $\frac57$ es $\frac12$, y calcula
        $\frac57 - \frac12$.
  \item muestra que la mayor \omterm{def:g6:fractions:def}{fracción} unitaria menor que
        $\frac{3}{14}$ es $\frac15$ (compara $\frac{3}{14}$ con
        $\frac14$ y con $\frac15$), calcula
        $\frac{3}{14} - \frac15$ y concluye:
        \[
        \frac57 = \frac12 + \frac15 + \frac{1}{70} .
        \]
        Verifica esta escritura directamente con el \omterm{def:g4:fractions:def}{denominador} común
        $70$.
  \item aplica el método voraz a $\frac45$, comprobando en cada paso
        que tu \omterm{def:g6:fractions:def}{fracción} unitaria es la mayor que cabe, y
        verifica la escritura final.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Por qué el método siempre se para y el
decimoctavo camello.}
\begin{enumerate}[resume]
  \item en las preguntas 7--10 el método se aplicó a $\frac27$,
        $\frac57$ y $\frac45$. Para cada una, enumera los \omterm{def:g4:fractions:def}{numeradores} de
        los \omterm{def:g3:division:remainder}{restos} sucesivos (antes de simplificar).
        ¿Qué observas?
  \item para una \omterm{def:g6:fractions:def}{fracción} $\frac ab$ y un número entero $n$, demuestra
        que
        \[
        \frac{a}{b} - \frac{1}{n}
        = \frac{a \times n - b}{b \times n} .
        \]
  \item supón que el método voraz resta $\frac1n$ de
        $\frac ab$: entonces $\frac1n$ cabe pero $\frac{1}{n-1}$ ya
        es demasiado grande, $\frac{1}{n-1} > \frac ab$. Poniendo los dos
        sobre el \omterm{def:g4:fractions:def}{denominador} común $b \times (n - 1)$, deduce
        que $b > a \times n - a$ y, por tanto, que el nuevo \omterm{def:g4:fractions:def}{numerador}
        cumple
        \[
        a \times n - b < a .
        \]
        Explica por qué esto demuestra que el método siempre se para.
  \item un cuento antiguo: un padre muere dejando $17$ camellos y lega
        $\frac12$ del rebaño a su hijo mayor, $\frac13$ al
        segundo y $\frac19$ al tercero --- y ninguna de las
        partes es un número entero de camellos. Un vecino presta a la
        familia un decimoctavo camello; los hijos se llevan $9$, $6$ y
        $2$ camellos y el camello prestado se devuelve. Calcula
        $\frac12 + \frac13 + \frac19$ y explica el truco: ¿por qué
        se pudo repartir el rebaño en camellos enteros y por qué ningún
        hijo recibió menos de lo que prometía el testamento?
  \item repara el testamento: propón tres fracciones unitarias
        \emph{distintas} que sumen exactamente $1$ y da el
        menor rebaño para el que las tres partes sean un número
        entero de camellos.
\end{enumerate}
\end{problem}
