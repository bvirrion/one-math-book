\chapter{Proporcionalidad, velocidad y medias}\label{ch:g8:speed}

La \omterm{def:g7:prop:table}{proporcionalidad} (\cref{ch:g7:prop}) se encuentra ahora con el mundo físico:
la \omterm{def:g8:speed:def}{velocidad}, los caudales, las conversiones de unidades --- magnitudes que son
\emph{\omterm{def:g3:division:remainder}{cocientes}} de otras dos. El capítulo cierra con las \omterm{def:g8:speed:average}{medias
ponderadas}, donde pensar en proporciones evita un error clásico.

\section{La velocidad media}

\begin{definition}[Velocidad media]\label{def:g8:speed:def}
La \emph{velocidad media}\index{velocidad} de un trayecto es el \omterm{def:g3:division:remainder}{cociente}
\[
v = \frac{d}{t}
\qquad
\left(\text{la distancia dividida entre la duración}\right),
\]
en kilómetros por hora (km/h), metros por segundo (m/s), \dots\ La
fórmula se da la vuelta: $d = v \times t$ y $t = \frac{d}{v}$.
\end{definition}

\begin{example}\label{ex:g8:speed:basic}
Un tren recorre $270$ km en $1$ h $30$ min. Convierte primero el tiempo:
$1$ h $30$ min $= 1.5$ h. Entonces,
\[
v = \frac{270}{1.5} = 180 \text{ km/h}.
\]
Distancia a $90$ km/h durante $2$ h $20$ min $= \frac73$ h:
$d = 90 \times \frac73 = 210$ km. Tiempo para $35$ km a $14$ km/h:
$t = \frac{35}{14} = 2.5$ h $= 2$ h $30$ min.
\end{example}

\begin{remark}[Los minutos no son decimales]
$1$ h $30$ min es $1.5$ h, pero $1$ h $20$ min \emph{no} es $1.2$ h:
es $1 + \frac{20}{60} = \frac43 \approx 1.33$ h. Convierte siempre los
minutos en \omterm{def:g6:fractions:def}{fracción} de hora ($60$ min $= 1$ h) antes de dividir.
\end{remark}

\begin{method}[Convertir velocidades]\label{met:g8:speed:convert}
Para convertir km/h en m/s: $1$ km $= 1000$ m y $1$ h $= 3600$ s, así que
\[
v \text{ km/h} = \frac{v \times 1000}{3600} \text{ m/s}
= \frac{v}{3.6} \text{ m/s}.
\]
Para convertir m/s en km/h, multiplica por $3.6$.
\end{method}

\begin{example}\label{ex:g8:speed:convert}
$90$ km/h $= \frac{90}{3.6} = 25$ m/s. Un velocista que corre $100$ m en
$10$ s va a $10$ m/s $= 36$ km/h.
\end{example}

\section{Magnitudes cociente}

\begin{example}[Caudal, densidad, precio por kilo]\label{ex:g8:speed:quotients}
La \omterm{def:g8:speed:def}{velocidad} tiene muchos primos, todos tratados igual:
\begin{itemize}
  \item un grifo llena $48$ L en $4$ min: caudal
        $\frac{48}{4} = 12$ L/min, así que llenar una bañera de $150$ L lleva
        $\frac{150}{12} = 12.5$ min;
  \item $0.6$ kg de queso cuestan $9$ euros: precio
        $\frac{9}{0.6} = 15$ euros por kg;
  \item un coche gasta $6.3$ L en $90$ km: consumo
        $\frac{6.3}{90} \times 100 = 7$ L cada $100$ km.
\end{itemize}
Identifica el \omterm{def:g3:division:remainder}{cociente} y la fórmula de tres vías
($q = \frac ab$, $a = qb$, $b = \frac aq$) hace el resto.
\end{example}

\section{Medias ponderadas}

\begin{definition}[Media ponderada]\label{def:g8:speed:average}
Cuando unos valores $v_1, v_2, \dots$ vienen con recuentos (o pesos)
$n_1, n_2, \dots$, su \emph{media ponderada}\index{media ponderada}
es
\[
\bar v = \frac{n_1 v_1 + n_2 v_2 + \dots}{n_1 + n_2 + \dots} :
\]
el total de los valores, dividido entre el total de los pesos.
\end{definition}

\begin{example}\label{ex:g8:speed:average}
Un examen lo hicieron dos grupos: el grupo 1, de $12$ alumnos, con media
$11$; el grupo 2, de $18$ alumnos, con media $14$. Media de toda la
clase:
\[
\bar v = \frac{12 \times 11 + 18 \times 14}{12 + 18}
= \frac{132 + 252}{30} = \frac{384}{30} = 12.8 .
\]
\emph{No} $\frac{11 + 14}{2} = 12.5$: el grupo mayor arrastra la
media hacia su lado --- los pesos importan.
\end{example}

\begin{example}[Velocidad media en dos tramos]\label{ex:g8:speed:twolegs}
Una ciclista recorre $30$ km a $30$ km/h y después $30$ km a $15$ km/h. La
\omterm{def:g8:speed:def}{velocidad media} de todo el trayecto \emph{no} es
$\frac{30 + 15}{2} = 22.5$ km/h. Calcúlala con la definición:
\begin{enumerate}
  \item tiempos: $\frac{30}{30} = 1$ h y después $\frac{30}{15} = 2$ h;
        en total, $3$ h;
  \item distancia total $60$ km, así que
        $v = \frac{60}{3} = 20$ km/h.
\end{enumerate}
El tramo lento dura más, así que pesa más --- una media de \omterm{def:g8:speed:def}{velocidades}
hay que ponderarla por el \emph{tiempo}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g8:speed:1}
Calcula la \omterm{def:g8:speed:def}{velocidad media}: $150$ km en $2$ h; $27$ km en $45$ min;
$100$ m en $12.5$ s.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:2}
Calcula la distancia: $2$ h a $85$ km/h; $40$ min a $90$ km/h.
Calcula el tiempo: $315$ km a $90$ km/h (en h y min).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:3}
Convierte: $72$ km/h en m/s; $5$ m/s en km/h; $108$ km/h en m/s.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:4}
Un grifo da $15$ L/min. ¿Cuánto tarda en llenar un depósito de $600$ L? ¿Cuántos
litros da en $2$ h $30$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:5}
¿Qué sale mejor: $1.2$ kg de manzanas por $3.30$ euros o
$0.8$ kg por $2.16$ euros? Compara los precios por kilogramo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:6}
Una clase de $25$ alumnos tiene una media de $12.4$ en un examen; otra
clase de $35$ alumnos tiene una media de $10$. Calcula la media de
las dos clases juntas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:speed:7}
El sonido viaja a unos $340$ m/s. Ves un relámpago y oyes
el trueno $6$ segundos después. ¿A qué distancia cayó el rayo
(redondeando a los $100$ m)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:speed:8}
Un senderista camina $2$ h a $5$ km/h, descansa $30$ min y después camina $1$ h $30$
a $4$ km/h. Calcula la distancia total y la \omterm{def:g8:speed:def}{velocidad media}
\emph{incluyendo el descanso} (distancia total entre tiempo total transcurrido).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:speed:9}
Notas con coeficientes: un alumno sacó $15$ (coeficiente $3$), $9$
(coeficiente $2$) y $12$ (coeficiente $1$). Calcula la \omterm{def:g8:speed:average}{media
ponderada}. ¿Qué media simple (sin ponderar) tendría y por qué
son distintas?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:speed:10}
Un coche recorre $120$ km a $80$ km/h y después $120$ km a $120$ km/h.
\begin{enumerate}
  \item calcula la duración de cada tramo y después la \omterm{def:g8:speed:def}{velocidad media} de
        todo el trayecto;
  \item explica por qué la respuesta es menor que $100$ km/h, el punto medio
        de las dos \omterm{def:g8:speed:def}{velocidades}.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:speed:11}
Dos pueblos distan $18$ km. Ana sale del primero a las $10{:}00$
andando a $5$ km/h hacia el segundo; Boris sale del segundo a
la misma hora andando a $4$ km/h hacia el primero. ¿A qué hora se
encuentran y a qué distancia del pueblo de Ana? (Entre los dos
reducen la separación a $5 + 4$ km/h.)
\end{exercise}

\section{Problema: La media armónica, o por qué el viaje de vuelta arruina
la media}

\begin{problem}[{Problema de fin de semana --- la \omterm{def:g8:speed:def}{velocidad media} en distancias
iguales es $\frac{2 v_1 v_2}{v_1 + v_2}$ y nunca supera el
punto medio de las \omterm{def:g8:speed:def}{velocidades}}]
\label{pb:g8:speed:1}
El \Cref{ex:g8:speed:twolegs} y el \cref{exo:g8:speed:10} terminaron los dos
con la misma sorpresa: ir a una \omterm{def:g8:speed:def}{velocidad}, volver a
otra, y la \omterm{def:g8:speed:def}{velocidad media} cae \emph{por debajo} del punto medio de
las dos. Este problema halla la fórmula exacta que hay detrás de la sorpresa
--- una clase nueva de media, la \emph{media
armónica}\index{media armónica} ---, demuestra que la sorpresa es un
teorema y la lleva a su extremo lógico: un alcance que ninguna
\omterm{def:g8:speed:def}{velocidad} del mundo puede lograr.

\medskip
\noindent\textbf{Parte I --- La fórmula del viaje de ida y vuelta.}
Un trayecto cubre una distancia $d$ a \omterm{def:g8:speed:def}{velocidad} $v_1$ y después la misma
distancia $d$ otra vez a \omterm{def:g8:speed:def}{velocidad} $v_2$ (todos números positivos).
\begin{enumerate}
  \item expresa las dos duraciones y después la duración total como
        fracciones en las que aparezcan $d$, $v_1$ y $v_2$;
  \item la \omterm{def:g8:speed:def}{velocidad media} de todo el trayecto es
        $\bar v = \dfrac{2d}{\ \dfrac{d}{v_1} + \dfrac{d}{v_2}\ }$
        --- una \omterm{def:g6:fractions:def}{fracción} de piso doble
        (\cref{ex:g8:fractions:complex}). Simplifícala y demuestra:
        \[
        \bar v = \frac{2\, v_1 v_2}{v_1 + v_2} .
        \]
  \item comprueba la fórmula con los dos cálculos citados arriba:
        $v_1 = 30$ y $v_2 = 15$ km/h
        (\cref{ex:g8:speed:twolegs}), y $v_1 = 80$ y
        $v_2 = 120$ km/h (\cref{exo:g8:speed:10});
  \item la distancia $d$ ha desaparecido de la fórmula. ¿Qué
        significa eso, en concreto, para la ciclista?
  \item supón ahora que el trayecto pasa en cambio el mismo
        \emph{tiempo} $T$ a cada \omterm{def:g8:speed:def}{velocidad}. Demuestra que la \omterm{def:g8:speed:def}{velocidad
        media} es entonces $\frac{v_1 + v_2}{2}$, el punto medio simple.
        En el lenguaje de la \cref{def:g8:speed:average}: ¿por qué
        magnitud hay que ponderar siempre una media de \omterm{def:g8:speed:def}{velocidades}
        y qué cambia entre los dos escenarios?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Armónica frente a aritmética.}
Para dos números positivos $v_1$ y $v_2$, sean
\[
H = \frac{2\, v_1 v_2}{v_1 + v_2}
\quad\text{(su \emph{media armónica}),}
\qquad
A = \frac{v_1 + v_2}{2}
\quad\text{(su \emph{media aritmética}).}
\]
\begin{enumerate}[resume]
  \item calcula $H$ y $A$ para las parejas $(30, 15)$ y
        $(80, 120)$. ¿Cuál de las dos medias gana en cada caso?
  \item pon $A - H$ sobre el \omterm{def:g4:fractions:def}{denominador} común $2(v_1 + v_2)$,
        desarrolla el \omterm{def:g4:fractions:def}{numerador} con las identidades notables
        (\cref{pb:g8:equations:1}) y demuestra:
        \[
        A - H = \frac{(v_1 - v_2)^2}{2\,(v_1 + v_2)} .
        \]
  \item deduce el teorema que hay detrás de todas las sorpresas: para \omterm{def:g8:speed:def}{velocidades}
        positivas, $H \leq A$ siempre, con igualdad exactamente cuando
        $v_1 = v_2$. ¿Por qué zanja la cuestión un cuadrado en el \omterm{def:g4:fractions:def}{numerador}?
  \item demuestra la elegante identidad $H \times A = v_1 \times v_2$:
        las dos medias multiplicadas devuelven el \omterm{def:g2:mult:def}{producto} original;
  \item un coche recorre $60$ km a $40$ km/h y $60$ km a
        $60$ km/h. Predice la \omterm{def:g8:speed:def}{velocidad media} con la fórmula
        y confírmala después por la vía larga, con tiempos y distancia
        total.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El alcance imposible.}
\begin{enumerate}[resume]
  \item una ciclista participa en una carrera de $30$ km esperando promediar
        $30$ km/h. ¿Cuánto tiempo puede durarle la carrera, como
        mucho? Recorre los primeros $15$ km a $15$ km/h: ¿cuánto
        le queda de ese presupuesto de tiempo?
  \item demuestra que la media esperada se ha vuelto literalmente
        imposible: calcula su \omterm{def:g8:speed:def}{velocidad media} si recorre la
        segunda \omterm{def:g3:division:half}{mitad} a $90$ km/h y explica por qué ni siquiera una
        \omterm{def:g8:speed:def}{velocidad} arbitrariamente enorme le permite alcanzar los
        $30$ km/h;
  \item baja su objetivo a $20$ km/h. ¿Qué \omterm{def:g8:speed:def}{velocidad} en los
        segundos $15$ km lo consigue? Comprueba tu respuesta con la
        fórmula de la \omterm{pb:g8:speed:1}{media armónica};
  \item un grifo de agua fría solo llena una bañera en $20$ min; el de agua caliente solo,
        en $30$ min. ¿Qué \omterm{def:g6:fractions:def}{fracción} de la bañera llenan los dos
        grifos juntos por minuto y cuánto tardan?
        Compara tu respuesta con $H(20, 30)$: ¿qué
        observas?
  \item un avión vuela la misma ruta de ida y de vuelta: $900$ km/h
        con el viento a favor y $700$ km/h en contra. Calcula la
        \omterm{def:g8:speed:def}{velocidad media} del viaje completo y explica en una frase por qué
        un viento, sople como sople, siempre \emph{alarga} un viaje
        de ida y vuelta.
\end{enumerate}
\end{problem}
