\chapter{Puntos medios y paralelas}\label{ch:g8:midpoints}

Unir los puntos medios de dos lados de un triángulo: el \omterm{def:g6:lines:objects}{segmento} lo que obtienes es
siempre \omterm{def:g6:lines:perp}{paralelo} al tercer lado, y exactamente \omterm{def:g3:division:half}{medio} tanto tiempo. esto
El "teorema del punto medio" y su recíproco son la primera muestra de un gran
idea--- \omterm{def:g6:lines:perp}{paralelas} cortar longitudes proporcionalmente --- que florece en
El teorema de Tales en \cref{ch:g9:thales}.

\section{El teorema del punto medio}

\begin{theorem}[Teorema del punto medio]\label{thm:g8:midpoints:direct}
en un triángulo $ABC $, dejar $ I $ ser el punto medio de $[AB]$ y $ J $ el punto medio de $[AC]$. Entonces la línea $(IJ)$ es \omterm{def:g6:lines:perp}{paralelo} a $(a.\,C.)$, y
\[
IJ = \frac{a.\,C.}{2} .
\]
\end{theorem}

\begin{proof}[Idea de prueba]
dejar $ K $ ser el \omterm{def:g7:central:def}{simétrico} de $ I $ acerca de $ J $ (media vuelta,
\cref{ch:g7:central}). El \omterm{def:g4:geometry:polygon}{cuadrilátero} $ AICK $ tiene diagonales $[IK]$
y $[AC]$ cruce en su punto medio común $ J $: es un
\omterm{def:g7:central:parallelogram}{paralelogramo}, entonces $ KC $ es \omterm{def:g6:lines:perp}{paralelo} a $ AI $, es decir, \ a $(AB)$, y
$ KC = AI = IB $. Entonces $ IBCK $ tiene dos lados opuestos ($[IB]$ y
$[KC]$) \omterm{def:g6:lines:perp}{paralelo} e igual: es un \omterm{def:g7:central:parallelogram}{paralelogramo} también
(\cref{thm:g7:central:recognize}), entonces $(IK) \parallel (a.\,C.)$ ---
eso es $(IJ) \parallel (a.\,C.)$ --- y $ a.\,C. = IK = 2\,IJ $.
\end{proof}

\begin{example}\label{ex:g8:midpoints:direct}
en un triángulo $ ABC $ con $ a.\,C. = 9$ cm, los puntos medios $ I $ de $[AB]$ y
$ J $ de $[AC]$ están unidos. Sin ninguna medida:
$(IJ) \parallel (a.\,C.)$ y $ IJ = \frac92 = 4.5$ centímetro. El pequeño triángulo
$ AIJ $ es una copia de tamaño medio de $ ABC $ --- es \omterm{def:g6:measure:perimeter}{perímetro} es \omterm{def:g3:division:half}{medio} el
\omterm{def:g6:measure:perimeter}{perímetro} de $ ABC $ también.
\end{example}

\begin{theorem}[Conversar]\label{thm:g8:midpoints:converse}
en un triángulo $ ABC $, la recta que pasa por el punto medio $ I $ de $[AB]$ y
\emph{\omterm{def:g6:lines:perp}{paralelo}} a $(a.\,C.)$ cruza el lado $[AC]$ en su punto medio.
\end{theorem}
\admitted

\begin{example}\label{ex:g8:midpoints:converse}
Un camino cruza un campo triangular, comenzando en el medio de un borde.
y corriendo \omterm{def:g6:lines:perp}{paralelo} al borde opuesto. Lo contrario garantiza
que el camino sale exactamente en el medio del otro borde --- no
Es necesario medir en el otro lado.
\end{example}

\begin{method}[Elegir entre teorema y recíproco]\label{met:g8:midpoints:choose}
\begin{enumerate}
  \item \emph{Dos puntos medios conocidos} $\Rightarrow $ utilizar el directo
        teorema: concluir paralelismo y media longitud.
  \item \emph{Un punto medio y un \omterm{def:g6:lines:perp}{paralelo} conocido} $\Rightarrow $ utilizar
        lo contrario: concluir que el punto de cruce es un punto medio.
  \item Escribe qué triángulo, qué puntos medios y qué teorema
        use --- una oración cada uno.
\end{enumerate}
\end{method}

\section{Triángulos de tamaño medio}

\begin{proposition}[El triángulo del punto medio]\label{prop:g8:midpoints:medial}
Unir los tres puntos medios de los lados de un triángulo lo corta en
cuatro triángulos de igual \omterm{def:g6:measure:area}{áreas}, cada uno una copia de tamaño medio del original.
\end{proposition}

\begin{proof}
Cada lado del punto medio del triángulo es \omterm{def:g6:lines:perp}{paralelo} a un lado de $ ABC $ y
\omterm{def:g3:division:half}{medio} siempre y cuando (\cref{thm:g8:midpoints:direct}, aplicado tres veces).
Los cuatro triángulos pequeños tienen lados de las mismas tres longitudes, por lo que
son copias idénticas; juntos tejan $ ABC $, por lo que cada uno tiene un
\omterm{def:g3:division:half}{cuarto} de su \omterm{def:g6:measure:area}{área}.
\end{proof}

\begin{remark}[Hacia Tales]
El teorema del punto medio es el caso "uno \omterm{def:g3:division:half}{medio}'' de un hecho más general:
una linea \omterm{def:g6:lines:perp}{paralelo} a un lado de un triángulo se cortan los otros dos lados en
\emph{proporciones iguales} --- un tercio y un tercio, dos quintos y dos
quintos\,\dots\ Ese enunciado general es el teorema de Tales.
(\cref{ch:g9:thales}); el caso del punto medio es el único necesario
año.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g8:midpoints:1}
en un triángulo $ RST $, $ M $ es el punto medio de $[RS]$ y $ N $ el punto medio de $[RT]$, con $ ST = 7$ centímetro. ¿Qué se puede decir de la línea?
$(MN)$ y la longitud $ MN $? Cite el teorema utilizado.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g8:midpoints:2}
en un triángulo $ ABC $, $ I $ es el punto medio de $[AB]$ y la linea
a través de $ I $ \omterm{def:g6:lines:perp}{paralelo} a $(a.\,C.)$ cortes $[AC]$ en $ J $, con
$ AC = 11$ centímetro. Calcular $ AJ $, citando el teorema utilizado.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g8:midpoints:3}
dibuja un triángulo $ ABC $ con $ AB = 8$ centímetro, $ AC = 6$ centímetro, $ a.\,C. = 7$ centímetros,
colocar los puntos medios $ I $ de $[AB]$ y $ J $ de $[AC]$ y medir $IJ $
para comprobar el teorema. Calcular el \omterm{def:g6:measure:perimeter}{perímetro} de $ AIJ $ sin
midiendo cualquier otra cosa.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g8:midpoints:4}
$ IJK $ es el triángulo del punto medio de $ ABC $, cuyos lados miden $10$,
$12$ y $16$ centímetro. Da los tres lados de $ IJK $ y su \omterm{def:g6:measure:perimeter}{perímetro}.
¿Cómo los dos? \omterm{def:g6:measure:perimeter}{perímetros} ¿comparar?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g8:midpoints:5}
El \omterm{def:g6:measure:area}{área} de un triángulo es $36$ centímetro $^2$. cual es el \omterm{def:g6:measure:area}{área} de su
triángulo del punto medio? ¿De cada uno de los cuatro triángulos pequeños?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g8:midpoints:6}
en un triángulo $ ABC $, $ I $ es el punto medio de $[AB]$, $ J $ el de
$[AC]$ y $ K $ el de $[a.\,C.]$. mostrar eso $ IJKB $ es un \omterm{def:g7:central:parallelogram}{paralelogramo}.
(Comparar $[IJ]$ y $[BK]$: \omterm{def:g6:lines:perp}{paralelo}? ¿igual?)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g8:midpoints:7}
Una vela triangular tiene su borde más bajo. $6.4$ m de largo. un refuerzo
La costura une los puntos medios de los otros dos bordes. ¿Qué longitud de costura?
es necesario? ¿Qué pasa si la costura une los puntos ubicados en uno?
\omterm{def:g3:division:half}{cuarto} de cada borde, comenzando desde la parte superior \omterm{def:g5:solids:def}{vértice}? (Conjetura con
un dibujo; la prueba es la de Tales, \cref{ch:g9:thales}.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g8:midpoints:8}
en triángulo $ ABC $, \omterm{def:g6:shapes:triangles}{en ángulo recto} en $ A $, $ M $ es el punto medio del
hipotenusa $[a.\,C.]$, y el \omterm{def:g6:lines:perp}{paralelo} a $(AB)$ a través de $ M $ cumple
$[AC]$ en $ N $.
\begin{enumerate}
  \item mostrar eso $ N $ es el punto medio de $[AC]$.
  \item Explica por qué $(MN)$ es \omterm{def:g6:lines:perp}{perpendicular} a $(AC)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g8:midpoints:9}
Sea $ ABCD $ \emph{cualquier} \omterm{def:g4:geometry:polygon}{cuadrilátero}, y $ I $, $ J $, $ K $, $ L $ el
puntos medios de $[AB]$, $[a.\,C.]$, $[CD]$, $[DA]$ (la conjetura de
\cref{exo:g7:central:11}).
\begin{enumerate}
  \item Aplicar el teorema del punto medio en el triángulo. $ ABC $ hacia
        \omterm{def:g6:lines:objects}{segmento} $[IJ]$, y en el triángulo $ ACD $ hacia \omterm{def:g6:lines:objects}{segmento}
        $[LK]$: compara cada uno con la diagonal $[AC]$.
  \item Concluir que $ IJKL $ es siempre un \omterm{def:g7:central:parallelogram}{paralelogramo}.
\end{enumerate}
\end{exercise}

\section{Problema: Las tres medianas y el centro de gravedad.}

\begin{problem}[{Problema de fin de semana --- Las medianas de un triángulo se encuentran en un solo punto, dos tercios del camino hacia abajo de cada uno de ellos.}]
\label{pb:g8:midpoints:1}
Corta un triángulo de cartón rígido y se equilibrará.
perfectamente plano, sobre la punta de un lápiz colocada en un punto especial: el
\emph{centro de
        gravedad}\index{centro de
        gravedad} del triángulo.
Este problema encuentra ese punto. A \emph{mediano}\index{mediano} de un
el triángulo es el \omterm{def:g6:lines:objects}{segmento} unirse a un \omterm{def:g5:solids:def}{vértice} hacia \emph{punto medio} de
el lado opuesto. Demostrarás --- con el teorema del punto medio
usado dos veces, de una manera inesperada --- que las tres medianas todas
pasar por un punto y localizarlo con precisión.

\medskip
\noindent\textbf{Parte I --- Warm-up.}
\begin{enumerate}
  \item Dibuja un triángulo grande $ ABC $ (sin forma especial), construir
        los puntos medios $ I $ de $[AB]$, $ J $ de $[AC]$ y $ K $ de
        $[a.\,C.]$ y dibuja las tres medianas $[AK]$, $[BJ]$, $[CI]$.
        ¿Qué observas?
  \item Demuestra que una mediana corta el triángulo en dos triángulos.
        de igual \omterm{def:g6:measure:area}{áreas}. (Compare bases y alturas, y use el
        \omterm{def:g6:measure:area}{área} fórmula: base $\times $ altura $\div\ 2$.) Para un
        triángulo de \omterm{def:g6:measure:area}{área} $36$ centímetro $^2$¿Cuáles son las dos piezas?
        \omterm{def:g6:measure:area}{áreas}?
  \item en el triángulo $ ABC $, ¿qué hace
        \cref{thm:g8:midpoints:direct} decir sobre el \omterm{def:g6:lines:objects}{segmento}
        $[IJ]$?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Two medians meet two thirds of the
way.}
las medianas $[BJ]$ y $[CI]$ cruzar en un punto; llámalo $ G $. Dejar
$ M $ ser el punto medio de $[GB]$ y $ N $ el punto medio de $[GC]$.
\begin{enumerate}[resume]
  \item Aplicar el teorema del punto medio \emph{en el triángulo $ GBC $}:
        ¿Qué dice sobre el \omterm{def:g6:lines:objects}{segmento} $[MN]$?
  \item deducir eso $(IJ) \parallel (MN)$ y $ IJ = MN $, y
        concluir con \cref{thm:g7:central:recognize} eso $ IJNM $
        es un \omterm{def:g7:central:parallelogram}{paralelogramo}.
  \item Explica por qué $ I $ y $ N $ ambos se encuentran en la mediana $(CI)$,
        y $ J $ y $ M $ ambos en la mediana $(BJ)$ --- entonces el
        diagonales de la \omterm{def:g7:central:parallelogram}{paralelogramo} $ IJNM $ son $[IN]$ y
        $[JM]$, y se cruzan exactamente en $ G $. ¿Qué hace el
        \emph{definición} de un \omterm{def:g7:central:parallelogram}{paralelogramo} entonces di sobre $ G $?
  \item deducir eso $ BM = MG = GJ $, y concluye:
        \[
        BG = 2 \times GJ
        \]
        --- el punto $ G $ se sienta en la mediana $[BJ]$ en dos tercios
        del camino desde el \omterm{def:g5:solids:def}{vértice} $ B $. Enuncie y justifique la
        resultado análogo en la mediana $[CI]$.
  \item Ahora olvídate $[CI]$ y ejecutar el mismo argumento en el
        par de medianas $[BJ]$ y $[AK]$: su punto de cruce
        también se sitúa a dos tercios de $[BJ]$ de $ B $. Explica por qué
        esto lo obliga a ser el \emph{mismo} punto $ G $ --- y
        Concluimos que las tres medianas pasan por $ G $.
  \item En un triángulo donde la mediana $[BJ]$ medidas $9$ centímetros,
        ¿Qué tan lejos está? $ G $ de $ B $, y de $ J $?
  \item Justifica (una frase) que $ AG = 2 \times GK $ también.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Six equal slices, and coordinates.}
Las tres medianas cortan el triángulo en seis triángulos pequeños.
alrededor $ G $.
\begin{enumerate}[resume]
  \item en el triángulo $ GBC $, el \omterm{def:g6:lines:objects}{segmento} $[GK]$ es una mediana.
        Deduzca que los dos triángulos pequeños $ GBK $ y $ GKC $ tener
        igual \omterm{def:g6:measure:area}{áreas}.
  \item Usando la pregunta 2 en los triángulos $ ABK $ y $ AKC $ (ambos
        cortado por la mediana $[AK]$ de $ ABC $), demuestran que el
        triangulos $ ABG $ y $ ACG $ tener
        igual \omterm{def:g6:measure:area}{áreas}. repitiendo
        con otra mediana, concluye que los tres triángulos
        $ ABG $, $ BCG $, $ CAG $ todos tienen \omterm{def:g6:measure:area}{área} un tercio de $ ABC $.
  \item Cada uno de estos tres triángulos está cortado en \omterm{def:g3:division:half}{medio} por una pieza
        de mediana (por ejemplo $[GI]$ es una mediana de la
        triángulo $ ABG $ --- visto desde donde \omterm{def:g5:solids:def}{vértice}?). Conclusión: los seis
        pequeñas rodajas alrededor $ G $ todos tienen lo mismo \omterm{def:g6:measure:area}{área}, una sexta parte
        del triángulo.
  \item En una cuadrícula de coordenadas, trace $ A(0, 0)$, $ B(6, 0)$ y
        $ C(0, 6)$, con $ K(3, 3)$ el punto medio de $[a.\,C.]$. Utilice el
        teorema de dos tercios de la Parte II para calcular las coordenadas
        de $ G $ en la mediana $[AK]$; luego comprueba que sea lo mismo
        El punto se sitúa a dos tercios de la mediana. $[BJ]$, dónde
        $ J(0, 3)$. Comparar $ G $las coordenadas de con los promedios
        \[
        \frac{0 + 6 + 0}{3},
        \qquad
        \frac{0 + 0 + 6}{3} .
        \]
  \item El balance final: utilizando las preguntas 2 y 8, explique por qué
        El triángulo de cartón se equilibra sobre el filo de un cuchillo colocado
        a lo largo de cualquier mediana --- cantidades iguales de cartón en cada
        lado --- y por qué la punta del lápiz debe colocarse
        en $ G$, el punto que los físicos llaman \omterm{pb:g8:midpoints:1}{centro de
        gravedad}. (Una prueba de balanza totalmente rigurosa necesita la
        calculo integral de la universidad \omterm{def:g6:measure:volume}{volúmenes}; el igual
        \omterm{def:g6:measure:area}{áreas} hazlo creíble hoy.)
\end{enumerate}
\end{problem}
