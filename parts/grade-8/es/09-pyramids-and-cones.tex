\chapter{Pirámides y conos}\label{ch:g8:solids}

Después de los \omterm{def:g7:areas:prism}{prismas} y los \omterm{def:g7:areas:prism}{cilindros} del \cref{ch:g7:areas} --- cuerpos de
sección constante --- llegan los cuerpos con punta: las \omterm{def:g8:solids:def}{pirámides} y
los \omterm{def:g8:solids:def}{conos}. Su fórmula del \omterm{def:g6:measure:volume}{volumen} lleva un factor famoso, $\frac13$, que
este capítulo hace verosímil y usa; el estudio de los cuerpos continúa
con las esferas y las secciones en el \cref{ch:g9:solids}.

\section{Describir pirámides y conos}

\begin{definition}[Pirámide, cono]\label{def:g8:solids:def}
\begin{itemize}
  \item una \emph{pirámide}\index{pirámide} tiene un \omterm{def:g4:geometry:polygon}{polígono} como
        \emph{base} y un punto, el \emph{\omterm{def:g5:solids:def}{vértice}}, unido a todos los
        \omterm{def:g5:solids:def}{vértices} de la base por \omterm{def:g5:solids:def}{caras} triangulares. Su \emph{altura} es
        la distancia del \omterm{def:g5:solids:def}{vértice} al plano de la base;
  \item un \emph{cono}\index{cono} es la misma construcción sobre un
        disco: un disco como base, un \omterm{def:g5:solids:def}{vértice} y una superficie lateral curva.
\end{itemize}
Una pirámide es \emph{regular} cuando su base es un \omterm{def:g4:geometry:polygon}{polígono} regular
(cuadrado, \omterm{def:g6:shapes:triangles}{triángulo equilátero}, \dots) y su \omterm{def:g5:solids:def}{vértice} queda en la vertical
del centro de la base.
\end{definition}

\begin{example}[Contar caras, aristas y vértices]\label{ex:g8:solids:count}
Una \omterm{def:g8:solids:def}{pirámide} de base cuadrada tiene $5$ \omterm{def:g5:solids:def}{caras} ($1$ cuadrado $+ 4$
triángulos), $8$ \omterm{def:g5:solids:def}{aristas} y $5$ \omterm{def:g5:solids:def}{vértices}. Con una base de $n$ lados:
$n + 1$ \omterm{def:g5:solids:def}{caras}, $2n$ \omterm{def:g5:solids:def}{aristas} y $n + 1$ \omterm{def:g5:solids:def}{vértices}. Comprueba el patroncito de
Euler: \omterm{def:g5:solids:def}{caras} $+$ \omterm{def:g5:solids:def}{vértices} $=$ \omterm{def:g5:solids:def}{aristas} $+ 2$.
\end{example}

\section{El volumen}

\begin{theorem}[Volumen de una pirámide o de un cono]\label{thm:g8:solids:volume}
Para una \omterm{def:g8:solids:def}{pirámide} o un \omterm{def:g8:solids:def}{cono} de \omterm{def:g6:measure:area}{área} de la base $B$ y altura $h$:
\[
V = \frac{1}{3}\, B \times h
\]
--- un tercio del \omterm{def:g7:areas:prism}{prisma} o del \omterm{def:g7:areas:prism}{cilindro} de la misma base y altura.
Para un \omterm{def:g8:solids:def}{cono} de \omterm{def:g3:shapes:shapes}{radio} de la base $r$: $V = \frac13 \pi r^2 h$.
\end{theorem}
\admitted

\begin{remark}[¿Por qué un tercio?]
Llena de agua una \omterm{def:g8:solids:def}{pirámide} hueca y viértela en el \omterm{def:g7:areas:prism}{prisma} de la misma
base y altura: hacen falta exactamente tres llenados. Mejor aún: un \omterm{def:g5:solids:def}{cubo}
se puede cortar en tres \omterm{def:g8:solids:def}{pirámides} idénticas, cada una con una \omterm{def:g5:solids:def}{cara} del
\omterm{def:g5:solids:def}{cubo} como base y un \omterm{def:g5:solids:def}{vértice} del \omterm{def:g5:solids:def}{cubo} como \omterm{def:g5:solids:def}{vértice} (¡intenta imaginarlo!).
Una demostración general usa la integración, herramienta del \omterm{def:g6:measure:volume}{volumen} de secundaria superior.
\end{remark}

\begin{example}\label{ex:g8:solids:volume}
Una \omterm{def:g8:solids:def}{pirámide} tiene una base cuadrada de lado $5$ cm y altura $9$ cm:
\begin{enumerate}
  \item \omterm{def:g6:measure:area}{área} de la base: $B = 5^2 = 25$ cm$^2$;
  \item \omterm{def:g6:measure:volume}{volumen}: $V = \frac13 \times 25 \times 9 = 25 \times 3 = 75$
        cm$^3$.
\end{enumerate}
Un \omterm{def:g8:solids:def}{cono} de helado de \omterm{def:g3:shapes:shapes}{radio} $3$ cm y altura $10$ cm:
\[
V = \frac13 \times \pi \times 3^2 \times 10 = 30\pi \approx 94
\text{ cm}^3 .
\]
\end{example}

\begin{example}[Cuidado con la altura]\label{ex:g8:solids:slant}
En un \omterm{def:g8:solids:def}{cono}, no confundas la altura $h$ (del \omterm{def:g5:solids:def}{vértice} al centro de la
base, perpendicular) con la \emph{generatriz} $s$ (del \omterm{def:g5:solids:def}{vértice} al
borde). Están relacionadas por Pitágoras
(\cref{thm:g8:pythagoras:direct}) en el \omterm{def:g6:shapes:triangles}{triángulo rectángulo}
vértice--centro--borde:
\[
s^2 = h^2 + r^2 .
\]
Un \omterm{def:g8:solids:def}{cono} con $r = 3$ y $s = 5$ tiene, por tanto, altura
$h = \sqrt{25 - 9} = 4$ y \omterm{def:g6:measure:volume}{volumen}
$\frac13 \pi \times 9 \times 4 = 12\pi$.
\end{example}

\begin{method}[Problemas de volumen]\label{met:g8:solids:method}
\begin{enumerate}
  \item identifica el cuerpo (\omterm{def:g7:areas:prism}{prisma}/\omterm{def:g7:areas:prism}{cilindro}: $V = Bh$; \omterm{def:g8:solids:def}{pirámide}/\omterm{def:g8:solids:def}{cono}:
        $V = \frac13 Bh$);
  \item calcula primero el \omterm{def:g6:measure:area}{área} de la base $B$, como paso aparte;
  \item comprueba que la altura es perpendicular a la base (usa
        Pitágoras si te dan la generatriz);
  \item multiplica, mantén las unidades coherentes y convierte al final si hace falta
        ($1$ L $= 1000$ cm$^3$).
\end{enumerate}
\end{method}

\section{Desarrollos}

\begin{example}[Desarrollo de una pirámide]\label{ex:g8:solids:net}
Desplegar una \omterm{def:g8:solids:def}{pirámide} regular de base cuadrada la aplana en su
\emph{\omterm{def:g5:solids:net}{desarrollo}}: la base cuadrada con cuatro \omterm{def:g6:shapes:triangles}{triángulos isósceles} idénticos
pegados a sus lados. Los lados iguales de los triángulos son las
\emph{\omterm{def:g5:solids:def}{aristas} laterales} de la \omterm{def:g8:solids:def}{pirámide} --- su longitud no es ni $h$
ni la altura de una \omterm{def:g5:solids:def}{cara}, así que etiquétalo bien antes de recortar
el cartón.
\end{example}

\section{Ejercicios}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:solids:2}

altura $10$ cm; y el de una \omterm{def:g8:solids:def}{pirámide} de base rectangular

\begin{exercise}[$\star$]\label{exo:g8:solids:3}
Calcula el \omterm{def:g6:measure:volume}{volumen} de un \omterm{def:g8:solids:def}{cono} de \omterm{def:g3:shapes:shapes}{radio} $5$ cm y altura $9$ cm
(exacto con $\pi$ y después redondeado al cm$^3$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:solids:4}

\begin{exercise}[$\star$]\label{exo:g8:solids:5}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:solids:6}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:solids:8}
Un \omterm{def:g8:solids:def}{cono} tiene generatriz $13$ cm y \omterm{def:g3:shapes:shapes}{radio} de la base $5$ cm. Calcula su

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:solids:9}

\begin{exercise}[$\star\star$]\label{exo:g8:solids:10}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:solids:11}
Una \omterm{def:g8:solids:def}{pirámide} de base cuadrada de lado $6$ cm tiene su \omterm{def:g5:solids:def}{vértice} justo
encima de una esquina de la base, a $8$ cm de altura.
\begin{enumerate}
  \item ¿sigue valiendo la fórmula $V = \frac13 Bh$? (Sí ---
        el \omterm{def:g5:solids:def}{vértice} no tiene por qué estar centrado. Calcula $V$.)
  \item calcula la longitud de la \omterm{def:g5:solids:def}{arista} lateral más larga, del \omterm{def:g5:solids:def}{vértice}
        a la esquina opuesta de la base (dos pasos de Pitágoras: primero la
        diagonal de la base).
\end{enumerate}
\end{exercise}

\section{Problema: Un cubo cortado en tres pirámides}

\begin{problem}[{Problema de fin de semana --- de dónde sale el factor $\frac13$
y qué le pasa a una pirámide cortada a media altura}]
\label{pb:g8:solids:1}
La observación posterior al \cref{thm:g8:solids:volume} afirma que un \omterm{def:g5:solids:def}{cubo}
se puede cortar en tres \omterm{def:g8:solids:def}{pirámides} idénticas --- «¡intenta imaginarlo!».
Este problema lo imagina con precisión, saca de ahí honradamente el famoso factor
$\frac13$, corta después el \omterm{def:g5:solids:def}{cubo} de una segunda manera en
\emph{seis} \omterm{def:g8:solids:def}{pirámides} y termina rebanando una \omterm{def:g8:solids:def}{pirámide} a media altura ---
con un resultado que poca gente acierta a la primera.

En todo el problema, $ABCDEFGH$ es un \omterm{def:g5:solids:def}{cubo} de lado $a$: base $ABCD$ y \omterm{def:g5:solids:def}{cara}
superior $EFGH$, con $E$ encima de $A$, $F$ encima de $B$, $G$ encima de $C$ y
$H$ encima de $D$.

\medskip
\noindent\textbf{Parte I --- Tres \omterm{def:g8:solids:def}{pirámides} con un \omterm{def:g5:solids:def}{vértice} común.}
\begin{enumerate}
  \item el \omterm{def:g5:solids:def}{vértice} $G$ pertenece a tres \omterm{def:g5:solids:def}{caras} del \omterm{def:g5:solids:def}{cubo}. Enumera
        las tres \omterm{def:g5:solids:def}{caras} que \emph{no} contienen a $G$ y
        describe las tres \omterm{def:g8:solids:def}{pirámides} que se obtienen tomando cada una de
        ellas como base, con \omterm{def:g5:solids:def}{vértice} $G$ siempre. ¿Cuántas \omterm{def:g5:solids:def}{caras},
        \omterm{def:g5:solids:def}{aristas} y \omterm{def:g5:solids:def}{vértices} tiene cada \omterm{def:g8:solids:def}{pirámide}
        (\cref{ex:g8:solids:count})?
  \item explica por qué girar el \omterm{def:g5:solids:def}{cubo} un tercio de vuelta alrededor de
        su diagonal larga $(AG)$ deja el \omterm{def:g5:solids:def}{cubo} igual pero
        permuta cíclicamente las tres bases de la pregunta 1 ---
        y por qué las tres \omterm{def:g8:solids:def}{pirámides} son, por tanto, copias
        idénticas unas de otras;
  \item admitiendo que las tres \omterm{def:g8:solids:def}{pirámides} llenan el \omterm{def:g5:solids:def}{cubo} sin
        solaparse (una maqueta de cartón resulta bastante convincente),
        deduce el \omterm{def:g6:measure:volume}{volumen} de cada una;
  \item compruébalo con la fórmula del
        \cref{thm:g8:solids:volume}: la \omterm{def:g8:solids:def}{pirámide} de base $ABCD$
        y \omterm{def:g5:solids:def}{vértice} $G$ tiene su \omterm{def:g5:solids:def}{vértice} justo encima de una \emph{esquina}
        de su base, exactamente como en el \cref{exo:g8:solids:11}.
        Identifica su altura, aplica la fórmula y compara;
  \item para un \omterm{def:g5:solids:def}{cubo} de lado $6$ cm, da el \omterm{def:g6:measure:volume}{volumen} de cada una de las
        tres \omterm{def:g8:solids:def}{pirámides} --- y explica qué tiene que ver esta descomposición
        con el experimento del agua del
        capítulo (tres llenados de la \omterm{def:g8:solids:def}{pirámide} para un \omterm{def:g7:areas:prism}{prisma}).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Seis \omterm{def:g8:solids:def}{pirámides} con el centro común.}
Sea ahora $O$ el centro del \omterm{def:g5:solids:def}{cubo}, y une $O$ con los cuatro
\omterm{def:g5:solids:def}{vértices} de cada una de las seis \omterm{def:g5:solids:def}{caras}.
\begin{enumerate}[resume]
  \item describe las seis \omterm{def:g8:solids:def}{pirámides} que se obtienen y explica por qué
        son copias idénticas unas de otras. ¿Cuál es la altura
        de cada una?
  \item deduce, repartiendo el \omterm{def:g6:measure:volume}{volumen} del \omterm{def:g5:solids:def}{cubo}, el \omterm{def:g6:measure:volume}{volumen} de cada
        una de las seis \omterm{def:g8:solids:def}{pirámides};
  \item vuelve a calcular ese \omterm{def:g6:measure:volume}{volumen} con la fórmula
        $V = \frac13 B h$ y comprueba que las dos respuestas coinciden;
  \item las dos descomposiciones confirman el factor $\frac13$ para
        unas \omterm{def:g8:solids:def}{pirámides} muy especiales. Explica por qué todavía no
        \emph{demuestran} el \cref{thm:g8:solids:volume} para toda
        \omterm{def:g8:solids:def}{pirámide} y todo \omterm{def:g8:solids:def}{cono} --- y di cómo trata el capítulo
        esa laguna: qué enunciado se admite, qué experimento
        lo apoya y en qué \omterm{def:g6:measure:volume}{volumen} de la serie se
        demuestra honradamente;
  \item un \omterm{def:g5:solids:def}{cubo} de lado $6$ cm se corta en sus seis \omterm{def:g8:solids:def}{pirámides}
        centrales. Calcula el \omterm{def:g6:measure:volume}{volumen} de cada una, de las dos maneras.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Cortar una \omterm{def:g8:solids:def}{pirámide} a media altura.}
$SABCD$ es una \omterm{def:g8:solids:def}{pirámide} regular de base cuadrada: lado de la base $c$, altura
$h$ y \omterm{def:g5:solids:def}{vértice} $S$ encima del centro de la base. Córtala por el plano
que pasa por los \emph{puntos medios} de las cuatro \omterm{def:g5:solids:def}{aristas} laterales $[SA]$,
$[SB]$, $[SC]$, $[SD]$.
\begin{enumerate}[resume]
  \item aplica el teorema del punto medio
        (\cref{thm:g8:midpoints:direct}) en el triángulo $SAB$:
        ¿cuáles son la dirección y la longitud del \omterm{def:g6:lines:objects}{segmento} que une
        los puntos medios de $[SA]$ y $[SB]$? Deduce que el corte
        es un cuadrado de lado $\frac{c}{2}$, paralelo a la base,
        a altura $\frac{h}{2}$;
  \item el trozo de encima del corte es a su vez una \omterm{def:g8:solids:def}{pirámide} de base cuadrada.
        Da su lado de la base y su altura, y demuestra que su \omterm{def:g6:measure:volume}{volumen}
        es exactamente \emph{una octava parte} del \omterm{def:g6:measure:volume}{volumen} original $V$;
  \item deduce el \omterm{def:g6:measure:volume}{volumen} del trozo de abajo (el \emph{tronco de \omterm{def:g8:solids:def}{pirámide}},
        la forma de una \omterm{def:g8:solids:def}{pirámide} sin terminar). ¿En qué razón reparte
        el \omterm{def:g6:measure:volume}{volumen} el corte a media altura?
  \item la \omterm{def:g8:solids:def}{pirámide} del Louvre (\cref{exo:g8:solids:7}: lado de la base
        $35$ m, altura $22$ m) se limpia en dos campañas: el
        cristal por encima de la media altura y después el de por debajo. ¿Qué
        \omterm{def:g6:fractions:def}{fracción} del \emph{\omterm{def:g6:measure:volume}{volumen}} queda por debajo de la media altura?
        Estímala en metros cúbicos, redondeando a la centena;
  \item la moraleja general: si \emph{todas} las dimensiones de una
        \omterm{def:g8:solids:def}{pirámide} se multiplican por $2$, ¿qué le pasa a su
        \omterm{def:g6:measure:volume}{volumen}? ¿Por qué factor crece el \omterm{def:g6:measure:volume}{volumen} cuando cada
        dimensión se multiplica por $k$ --- y cómo explica esto,
        en una línea, la «octava parte» de la pregunta 12?
\end{enumerate}
\end{problem}
