\section{Describiendo pirámides y conos.}

\begin{definition}[pirámide, cono]\label{def:g8:solids:def}
\begin{itemize}
  \item A \emph{pirámide}\index{pirámide} tiene un \omterm{def:g4:geometry:polygon}{polígono} como su
        \emph{base} y un punto, el \emph{ápex}, unido a cada
        \omterm{def:g5:solids:def}{vértice} de la base por triangular \omterm{def:g5:solids:def}{caras}. Es \emph{altura} es
        la distancia desde el vértice al plano de la base.
  \item A \emph{cono}\index{cono} es la misma construcción sobre un
        disco: un disco base, un ápice y una superficie lateral curva.
\end{itemize}
Una pirámide es \emph{regular} cuando su base es regular \omterm{def:g4:geometry:polygon}{polígono}
(cuadrado, \omterm{def:g6:shapes:triangles}{triángulo equilatero}, \dots) y su vértice se encuentra verticalmente
encima del centro de la base.
\end{definition}

\begin{example}[Contar caras, aristas, vértices.]\label{ex:g8:solids:count}
A \omterm{def:g8:solids:def}{pirámide} con base cuadrada tiene $5$ \omterm{def:g5:solids:def}{caras} ($1$ cuadrado $+ 4$
triángulos), $8$ bordes y $5$ \omterm{def:g5:solids:def}{vértices}. con un $ n $-base lateral:
$ n + 1$ \omterm{def:g5:solids:def}{caras}, $2n $ bordes, $ n + 1$ \omterm{def:g5:solids:def}{vértices}. Mira el pequeño de Euler
patrón: \omterm{def:g5:solids:def}{caras} $+$ \omterm{def:g5:solids:def}{vértices} $=$ bordes $+ 2$.
\end{example}

\section{Volumen}

\begin{theorem}[Volumen de una pirámide o cono.]\label{thm:g8:solids:volume}
por un \omterm{def:g8:solids:def}{pirámide} o un \omterm{def:g8:solids:def}{cono} con base \omterm{def:g6:measure:area}{área} $ B $ y altura $ h $:
\[
V = \frac{1}{3}\, B \times h
\]
--- un tercio de la \omterm{def:g7:areas:prism}{prisma} o \omterm{def:g7:areas:prism}{cilindro} con la misma base y altura.
por un \omterm{def:g8:solids:def}{cono} de base \omterm{def:g3:shapes:shapes}{radio} $ r $: $ V = \frac13 \pi r^2 h $.
\end{theorem}
\admitted

\begin{remark}[¿Por qué un tercio?]
llenar un hueco \omterm{def:g8:solids:def}{pirámide} con agua y verterlo en el \omterm{def:g7:areas:prism}{prisma} del mismo
base y altura: se necesitan exactamente tres rellenos. Aún mejor: un \omterm{def:g5:solids:def}{cubo}
se puede cortar en tres idénticos \omterm{def:g8:solids:def}{pirámides}, teniendo cada uno un \omterm{def:g5:solids:def}{rostro} del
\omterm{def:g5:solids:def}{cubo} como base y uno \omterm{def:g5:solids:def}{vértice} del \omterm{def:g5:solids:def}{cubo} como ápice (¡intenta imaginártelo!).
Una prueba general utiliza la integración, una herramienta del volumen de High School.
\end{remark}

\begin{example}\label{ex:g8:solids:volume}
A \omterm{def:g8:solids:def}{pirámide} tiene una base cuadrada de lado $5$ cm y altura $9$ centímetro:
\begin{enumerate}
  \item base \omterm{def:g6:measure:area}{área}: $ B = 5^2 = 25$ centímetro $^2$;
  \item \omterm{def:g6:measure:volume}{volumen}: $ V = \frac13 \times 25 \times 9 = 25 \times 3 = 75$
        centímetro $^3$.
\end{enumerate}
un helado \omterm{def:g8:solids:def}{cono} de \omterm{def:g3:shapes:shapes}{radio} $3$ cm y altura $10$ centímetro:
\[
V = \frac13 \times \pi \times 3^2 \times 10 = 30\pi \approx 94
\text{ cm}^3 .
\]
\end{example}

\begin{example}[Cuidado con la altura]\label{ex:g8:solids:slant}
por un \omterm{def:g8:solids:def}{cono}, no confundas la altura $ h $ (ápice al centro de la
base, \omterm{def:g6:lines:perp}{perpendicular}) con el \emph{altura inclinada} $ s $ (ápice de la
llanta). Están relacionados por Pitágoras.
(\cref{thm:g8:pythagoras:direct}) en el \omterm{def:g6:shapes:triangles}{triángulo rectángulo}
ápice--centro--borde:
\[
s^2 = h^2 + r^2 .
\]
A \omterm{def:g8:solids:def}{cono} con $ r = 3$ y $ s = 5$ por lo tanto tiene altura
$ h = \sqrt{25 - 9} = 4$, y \omterm{def:g6:measure:volume}{volumen}
$\frac13 \pi \times 9 \times 4 = 12\pi $.
\end{example}

\begin{method}[Problemas de volumen]\label{met:g8:solids:method}
\begin{enumerate}
  \item Identificar el \omterm{def:g5:solids:def}{sólido} (\omterm{def:g7:areas:prism}{prisma}/\omterm{def:g7:areas:prism}{cilindro}: $ V = Bh $; \omterm{def:g8:solids:def}{pirámide}/\omterm{def:g8:solids:def}{cono}:
        $ V = \frac13 Bh $);
  \item calcular la base \omterm{def:g6:measure:area}{área} $ B $ primero, como paso separado;
  \item comprobar que la altura es \omterm{def:g6:lines:perp}{perpendicular} a la base (usar
        Pitágoras si se da la altura inclinada);
  \item multiplicar, mantener las unidades consistentes, convertir al final si es necesario
        ($1$ l $= 1000$ centímetro $^3$).
\end{enumerate}
\end{method}

\section{Redes}

\begin{example}[Red de una pirámide]\label{ex:g8:solids:net}
Desplegando un cuadrado regular \omterm{def:g8:solids:def}{pirámide} lo aplana en su
\emph{\omterm{def:g5:solids:net}{neto}}: la base cuadrada con cuatro idénticos \omterm{def:g6:shapes:triangles}{triángulos isósceles}
pegado a sus costados. Los lados iguales de los triángulos son los
\emph{bordes laterales} del \omterm{def:g8:solids:def}{pirámide} --- su longitud no es ni $ h $
ni la altura inclinada de un \omterm{def:g5:solids:def}{rostro}, así que etiquételo cuidadosamente antes de cortar
cartón.
\end{example}

\section{Problema: un cubo cortado en tres pirámides}

\begin{problem}[{Problema de fin de semana --- de dónde viene el factor $\frac13$ y qué le sucede a una pirámide cortada a la mitad de su altura}]
\label{pb:g8:solids:1}
El comentario después \cref{thm:g8:solids:volume} afirma que un \omterm{def:g5:solids:def}{cubo}
se puede cortar en tres idénticos \omterm{def:g8:solids:def}{pirámides} --- ``intenta imaginar
¡Eso! ''. Este problema lo representa con precisión, obtiene el famoso factor
$\frac13$ salir de allí honestamente, luego corta el \omterm{def:g5:solids:def}{cubo} una segunda manera de entrar
\emph{seis} \omterm{def:g8:solids:def}{pirámides}, y termina cortando un \omterm{def:g8:solids:def}{pirámide} en \omterm{def:g3:division:half}{medio} es
altura --- con el resultado de que pocas personas aciertan en el primer intento.

A lo largo de, $ ABCDEFGH $ es un \omterm{def:g5:solids:def}{cubo} de lado $ a $: base $ ABCD $, arriba
\omterm{def:g5:solids:def}{rostro} $ EFGH $, con $ E $ arriba $ A $, $ F $ arriba $ B $, $ G $ arriba $ C $ y
$ H $ arriba $ D $.

\medskip
\noindent\textbf{Parte I --- Three \omterm{def:g8:solids:def}{pirámides} sharing an apex.}
\begin{enumerate}
  \item El \omterm{def:g5:solids:def}{vértice} $ G $ pertenece a tres \omterm{def:g5:solids:def}{caras} del \omterm{def:g5:solids:def}{cubo}. Lista
        los tres \omterm{def:g5:solids:def}{caras} que hacer \emph{no} contener $ G $, y
        describir los tres \omterm{def:g8:solids:def}{pirámides} obtenido tomando cada uno de
        ellos como base, con ápice $ G $ cada vez. Cuántos \omterm{def:g5:solids:def}{caras}, bordes y \omterm{def:g5:solids:def}{vértices} tiene cada uno \omterm{def:g8:solids:def}{pirámide}
        (\cref{ex:g8:solids:count})?
  \item Explique por qué girar el \omterm{def:g5:solids:def}{cubo} por un tercio de vuelta
        su larga diagonal $(AG)$ deja el \omterm{def:g5:solids:def}{cubo} sin cambios pero
        baraja las tres bases de la pregunta 1 en un ciclo ---
        y por que los tres \omterm{def:g8:solids:def}{pirámides} son por lo tanto identicos
        copias unas de otras.
  \item Admitiendo que los tres \omterm{def:g8:solids:def}{pirámides} llenar
el \omterm{def:g5:solids:def}{cubo} sin
        superposición (un modelo de cartón es bastante convincente),
        deducir el \omterm{def:g6:measure:volume}{volumen} de cada uno.
  \item Compara esto con la fórmula de
        \cref{thm:g8:solids:volume}: el \omterm{def:g8:solids:def}{pirámide} con base $ ABCD $
        y ápice $ G $ tiene su vértice directamente encima de un \emph{esquina}
        de su base, exactamente como en \cref{exo:g8:solids:11}.
        Identifica su altura, aplica la fórmula y compara.
  \item por un \omterm{def:g5:solids:def}{cubo} de lado $6$ cm, dale el \omterm{def:g6:measure:volume}{volumen} de cada uno de los
        tres \omterm{def:g8:solids:def}{pirámides} --- y explica en qué consiste esta descomposición
        tiene que ver con el experimento de verter agua del
        capítulo (tres rellenos del \omterm{def:g8:solids:def}{pirámide} por uno \omterm{def:g7:areas:prism}{prisma}).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Six \omterm{def:g8:solids:def}{pirámides} sharing the center.}
Ahora deja $ O $ ser el centro de la \omterm{def:g5:solids:def}{cubo}y únete $ O $ a los cuatro
\omterm{def:g5:solids:def}{vértices} de cada uno de los seis \omterm{def:g5:solids:def}{caras}.
\begin{enumerate}[resume]
  \item Describe los seis \omterm{def:g8:solids:def}{pirámides} obtenidos y explicar por qué
        son copias idénticas unas de otras. cual es la altura
        de cada uno?
  \item Deducir, compartiendo el \omterm{def:g5:solids:def}{cubo}'s \omterm{def:g6:measure:volume}{volumen}, el \omterm{def:g6:measure:volume}{volumen} de cada uno de los seis \omterm{def:g8:solids:def}{pirámides}.
  \item Recalcular eso \omterm{def:g6:measure:volume}{volumen} con la fórmula
        $ V = \frac13 B h $y comprueba que las dos respuestas concuerden.
  \item Las dos descomposiciones confirman el factor $\frac13$ para
        algunos muy especiales \omterm{def:g8:solids:def}{pirámides}. Explique por qué todavía no lo hacen.
        \emph{probar} \cref{thm:g8:solids:volume} por cada
        \omterm{def:g8:solids:def}{pirámide} y \omterm{def:g8:solids:def}{cono} --- y decir cómo trata el capítulo
        esa brecha: qué afirmación se admite, qué experimento
        lo apoya, y en el que \omterm{def:g6:measure:volume}{volumen} de la serie es
        honestamente demostrado.
  \item A \omterm{def:g5:solids:def}{cubo} de lado $6$ cm se corta en sus seis centros
        \omterm{def:g8:solids:def}{pirámides}. Calcular el \omterm{def:g6:measure:volume}{volumen} de cada uno, en ambos sentidos.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Cutting a \omterm{def:g8:solids:def}{pirámide} at \omterm{def:g3:division:half}{medio} height.}
$ SABCD $ es un cuadrado regular de base \omterm{def:g8:solids:def}{pirámide}: lado base $ c $, altura
$ h $, ápice $ S $ encima del centro de la base. Cortalo por el avión
a través del \emph{puntos medios} de los cuatro bordes laterales $[SA]$,
$[SB]$, $[SC]$, $[SD]$.
\begin{enumerate}[resume]
  \item Aplicar el teorema del punto medio
        (\cref{thm:g8:midpoints:direct}) en el triángulo $ SAB $:
        ¿cuáles son la dirección y la longitud del \omterm{def:g6:lines:objects}{segmento} unirse
        los puntos medios de $[SA]$ y $[SB]$? Deducir que el corte
        es un cuadrado de lado $\frac{c}{2}$, \omterm{def:g6:lines:perp}{paralelo} a la base,
        en altura $\frac{h}{2}$.
  \item La pieza encima del corte es en sí misma una base cuadrada. \omterm{def:g8:solids:def}{pirámide}.
        Indique el lado de su base y su altura, y demuestre que su \omterm{def:g6:measure:volume}{volumen}
        es exactamente \emph{un octavo} del original \omterm{def:g6:measure:volume}{volumen} $ V $.
  \item deducir el \omterm{def:g6:measure:volume}{volumen} de la pieza inferior (la \emph{tronco},
        la forma de un inacabado \omterm{def:g8:solids:def}{pirámide}). ¿En qué proporción
        el corte de media altura comparte el \omterm{def:g6:measure:volume}{volumen}?
  \item El Louvre \omterm{def:g8:solids:def}{pirámide} (\cref{exo:g8:solids:7}: lado base
        $35$ m, altura $22$ m) se limpia en dos campañas: la
        vidrio arriba \omterm{def:g3:division:half}{medio} altura, luego el vidrio de abajo. Qué
        \omterm{def:g6:fractions:def}{fracción} del \emph{\omterm{def:g6:measure:volume}{volumen}} se sienta debajo \omterm{def:g3:division:half}{medio} ¿altura?
        Estímalo en metros cúbicos, a la centena más cercana.
  \item La moraleja general: si \emph{todo} las dimensiones de un
        \omterm{def:g8:solids:def}{pirámide} se multiplican por $2$, ¿qué pasa con su
        \omterm{def:g6:measure:volume}{volumen}? ¿Por qué factor \omterm{def:g6:measure:volume}{volumen} crecer cuando cada
        dimensión se multiplica por $ k$ --- ¿Y cómo funciona esto?
        ¿Explicar, en una línea, el “un octavo” de la pregunta 12?
\end{enumerate}
\end{problem}
