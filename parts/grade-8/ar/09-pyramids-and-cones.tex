\section{وصف الأهرام والمخاريط}

\begin{definition}[الهرم والمخروط]\label{def:g8:solids:def}
\begin{itemize}
  \item \emph{الهرم}\index{هرم} له \omterm{def:g4:geometry:polygon}{مضلع} \emph{قاعدةً}
        ونقطة، هي \emph{القمّة}، موصولة بكل
        رأس من القاعدة \omterm{def:g5:solids:def}{بأوجه} مثلثية. و\emph{اِرتفاعه}
        هو بُعد القمّة عن مستوي القاعدة.
  \item و\emph{المخروط}\index{مخروط} هو البناء نفسه فوق
        قرص: قرص قاعدةً، وقمّة، وسطح جانبي منحنٍ.
\end{itemize}
ويكون الهرم \emph{منتظمًا} حين تكون قاعدته مضلعًا منتظمًا
(مربعًا، أو مثلثًا \omterm{def:g6:shapes:triangles}{متساوي الأضلاع}، \dots) وتقع قمّته عموديًّا
فوق مركز القاعدة.
\end{definition}

\begin{example}[عدّ الأوجه والأحرف والرؤوس]\label{ex:g8:solids:count}
\omterm{def:g8:solids:def}{الهرم} ذو القاعدة المربعة له $5$ \omterm{def:g5:solids:def}{أوجه} ($1$ مربع $+ 4$
مثلثات)، و $8$ \omterm{def:g5:solids:def}{أحرف}، و $5$ \omterm{def:g5:solids:def}{رؤوس}. ومع قاعدة ذات $n$ ضلعًا:
$n + 1$ وجهًا، و $2n$ حرفًا، و $n + 1$ رأسًا. وتحقّق من نمط أويلر
الصغير: \omterm{def:g5:solids:def}{الأوجه} $+$ \omterm{def:g5:solids:def}{الرؤوس} $=$ \omterm{def:g5:solids:def}{الأحرف} $+ 2$.
\end{example}

\section{الحجم}

\begin{theorem}[حجم الهرم أو المخروط]\label{thm:g8:solids:volume}
من أجل \omterm{def:g8:solids:def}{هرم} أو \omterm{def:g8:solids:def}{مخروط} \omterm{def:g6:measure:area}{مساحة} قاعدته $B$ واِرتفاعه $h$:
\[
V = \frac{1}{3}\, B \times h
\]
--- أي ثلث \omterm{def:g7:areas:prism}{الموشور} أو \omterm{def:g7:areas:prism}{الأسطوانة} بالقاعدة والارتفاع نفسيهما.
ومن أجل \omterm{def:g8:solids:def}{مخروط} \omterm{def:g3:shapes:shapes}{نصف قطر} قاعدته $r$: $V = \frac13 \pi r^2 h$.
\end{theorem}
\admitted

\begin{remark}[لماذا الثلث؟]
اِملأ هرمًا مجوّفًا بالماء واِسكبه في \omterm{def:g7:areas:prism}{الموشور} ذي القاعدة
والارتفاع نفسيهما: فيلزم ثلاث ملآت بالضبط. وأفضل من ذلك: يمكن قطع
\omterm{def:g5:solids:def}{مكعب} إلى ثلاثة أهرام متطابقة، لكل منها وجه من
\omterm{def:g5:solids:def}{المكعب} قاعدةً ورأس من \omterm{def:g5:solids:def}{المكعب} قمّةً (حاول تصوّر ذلك!).
والبرهان العامّ يستعمل التكامل، وهو أداة من مجلّد المرحلة الثانوية.
\end{remark}

\begin{example}\label{ex:g8:solids:volume}
\omterm{def:g8:solids:def}{هرم} قاعدته مربعة طول ضلعها $5$ سم واِرتفاعه $9$ سم:
\begin{enumerate}
  \item \omterm{def:g6:measure:area}{مساحة} القاعدة: $B = 5^2 = 25$ سم$^2$؛
  \item \omterm{def:g6:measure:volume}{والحجم}: $V = \frac13 \times 25 \times 9 = 25 \times 3 = 75$
        سم$^3$.
\end{enumerate}
\omterm{def:g8:solids:def}{ومخروط} مثلجات نصف قطره $3$ سم واِرتفاعه $10$ سم:
\[
V = \frac13 \times \pi \times 3^2 \times 10 = 30\pi \approx 94
\text{ سم}^3 .
\]
\end{example}

\begin{example}[اِنتبه للارتفاع]\label{ex:g8:solids:slant}
في \omterm{def:g8:solids:def}{المخروط}، لا تخلط بين الارتفاع $h$ (من القمّة إلى مركز
القاعدة، عموديًّا) و\emph{الطول المائل} $s$ (من القمّة إلى
الحافة). وترتبط بينهما مبرهنة فيثاغورس
(\cref{thm:g8:pythagoras:direct}) في \omterm{def:g6:shapes:triangles}{المثلث القائم الزاوية}
قمّة--مركز--حافة:
\[
s^2 = h^2 + r^2 .
\]
ومنه \omterm{def:g8:solids:def}{فالمخروط} الذي فيه $r = 3$ و $s = 5$ اِرتفاعه
$h = \sqrt{25 - 9} = 4$، وحجمه
$\frac13 \pi \times 9 \times 4 = 12\pi$.
\end{example}

\begin{method}[مسائل الحجوم]\label{met:g8:solids:method}
\begin{enumerate}
  \item عيّن \omterm{def:g5:solids:def}{المجسّم} (\omterm{def:g7:areas:prism}{موشور}/\omterm{def:g7:areas:prism}{أسطوانة}: $V = Bh$؛ \omterm{def:g8:solids:def}{وهرم}/\omterm{def:g8:solids:def}{مخروط}:
        $V = \frac13 Bh$)؛
  \item واِحسب \omterm{def:g6:measure:area}{مساحة} القاعدة $B$ أولًا، في خطوة مستقلة؛
  \item وتحقّق من أن الارتفاع عمودي على القاعدة (اِستعمل
        فيثاغورس إذا أُعطي الطول المائل)؛
  \item واِضرب، واِحفظ اِتّساق الوحدات، وحوّل في النهاية عند اللزوم
        ($1$ لتر $= 1000$ سم$^3$).
\end{enumerate}
\end{method}

\section{التصاميم}

\begin{example}[تصميم هرم]\label{ex:g8:solids:net}
نشر \omterm{def:g8:solids:def}{هرم} منتظم ذي قاعدة مربعة يسطّحه في
\emph{تصميمه}: أي القاعدة المربعة مع أربعة مثلثات متساوية الساقين متطابقة
ملتصقة بأضلاعها. وساقا المثلثات المتساويتان هما
\emph{\omterm{def:g5:solids:def}{الأحرف} الجانبية} \omterm{def:g8:solids:def}{للهرم} --- وطولها ليس $h$
ولا الطول المائل لوجه، فاِحرص على الترميز قبل قصّ
الورق المقوّى.
\end{example}

\section{مسألة: مكعب مقطوع إلى ثلاثة أهرام}

\begin{problem}[{مسألة نهاية الأسبوع --- من أين يأتي العامل $\frac13$،
وماذا يحدث لهرم يُقطع عند نصف اِرتفاعه}]
\label{pb:g8:solids:1}
تدّعي الملاحظة التي بعد \cref{thm:g8:solids:volume} أن \omterm{def:g5:solids:def}{المكعب}
يمكن قطعه إلى ثلاثة أهرام متطابقة --- «حاول تصوّر
ذلك!». وتصوّر هذه المسألة ذلك بدقّة، وتستخرج العامل المشهور
$\frac13$ منه اِستخراجًا أمينًا، ثم تقطع \omterm{def:g5:solids:def}{المكعب} بطريقة ثانية إلى
\emph{ستة} أهرام، وتنتهي بتشريح \omterm{def:g8:solids:def}{هرم} عند نصف
اِرتفاعه --- بنتيجة قلّ من يخمّنها صحيحة من أول محاولة.

وفي كل ما يلي، $ABCDEFGH$ \omterm{def:g5:solids:def}{مكعب} طول ضلعه $a$: قاعدته $ABCD$، ووجهه
الأعلى $EFGH$، مع $E$ فوق $A$، و $F$ فوق $B$، و $G$ فوق $C$، و
$H$ فوق $D$.

\medskip
\noindent\textbf{الجزء الأول --- ثلاثة أهرام تشترك في قمّة.}
\begin{enumerate}
  \item الرأس $G$ ينتمي إلى ثلاثة \omterm{def:g5:solids:def}{أوجه} من \omterm{def:g5:solids:def}{المكعب}. اُذكر
        \omterm{def:g5:solids:def}{الأوجه} الثلاثة التي \emph{لا} تحوي $G$، و
        صِف الأهرام الثلاثة المحصولة بأخذ كل واحد
        منها قاعدةً، مع القمّة $G$ في كل مرة. وكم وجهًا
        وحرفًا ورأسًا لكل \omterm{def:g8:solids:def}{هرم}
        (\cref{ex:g8:solids:count})؟
  \item اِشرح لماذا تترك إدارة \omterm{def:g5:solids:def}{المكعب} ثلث دورة حول
        قطره الطويل $(AG)$ المكعبَ دون تغيير لكنها
        تخلط قواعد السؤال 1 الثلاث في دورة ---
        ولماذا تكون الأهرام الثلاثة إذن نسخًا
        متطابقة بعضها لبعض.
  \item وبالتسليم بأن الأهرام الثلاثة تملأ \omterm{def:g5:solids:def}{المكعب} دون
        تراكب (ونموذج من الورق المقوّى مقنع جدًّا)،
        اِستنتج \omterm{def:g6:measure:volume}{حجم} كل واحد منها.
  \item وتحقّق من ذلك بمقابلته بصيغة
        \cref{thm:g8:solids:volume}: \omterm{def:g8:solids:def}{فالهرم} ذو القاعدة $ABCD$
        والقمّة $G$ قمّته فوق \emph{ركن}
        من قاعدته مباشرةً، تمامًا كما في \cref{exo:g8:solids:11}.
        عيّن اِرتفاعه، وطبّق الصيغة، وقارن.
  \item ومن أجل \omterm{def:g5:solids:def}{مكعب} طول ضلعه $6$ سم، أعطِ \omterm{def:g6:measure:volume}{حجم} كل واحد من
        الأهرام الثلاثة --- واِشرح ما علاقة هذا التفكيك
        بتجربة سكب الماء في الفصل
        (ثلاث ملآت من \omterm{def:g8:solids:def}{الهرم} مقابل \omterm{def:g7:areas:prism}{موشور} واحد).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- ستة أهرام تشترك في المركز.}
والآن ليكن $O$ مركز \omterm{def:g5:solids:def}{المكعب}، وصِل $O$ \omterm{def:g5:solids:def}{بالرؤوس} الأربعة
لكل وجه من \omterm{def:g5:solids:def}{الأوجه} الستة.
\begin{enumerate}[resume]
  \item صِف الأهرام الستة المحصولة، واِشرح لماذا تكون
        نسخًا متطابقة بعضها لبعض. وما اِرتفاع
        كل واحد منها؟
  \item واِستنتج، بتقاسم \omterm{def:g6:measure:volume}{حجم} \omterm{def:g5:solids:def}{المكعب}، \omterm{def:g6:measure:volume}{حجم} كل واحد
        من الأهرام الستة.
  \item وأعِد حساب ذلك \omterm{def:g6:measure:volume}{الحجم} بالصيغة
        $V = \frac13 B h$، وتحقّق من توافق الجوابين.
  \item يؤكد التفكيكان العامل $\frac13$ من أجل
        أهرام خاصّة جدًّا. اِشرح لماذا لا \emph{يبرهنان} بعدُ
        على \cref{thm:g8:solids:volume} من أجل كل
        \omterm{def:g8:solids:def}{هرم} \omterm{def:g8:solids:def}{ومخروط} --- وقل كيف يعالج الفصل
        تلك الفجوة: أيّ عبارة مقبولة، وأيّ تجربة
        تدعمها، وفي أيّ مجلّد من السلسلة يُبرهن
        عليها برهانًا أمينًا.
  \item \omterm{def:g5:solids:def}{مكعب} طول ضلعه $6$ سم يُقطع إلى أهرامه المركزية
        الستة. اِحسب \omterm{def:g6:measure:volume}{حجم} كل واحد، بالطريقتين.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- قطع \omterm{def:g8:solids:def}{هرم} عند نصف الارتفاع.}
$SABCD$ \omterm{def:g8:solids:def}{هرم} منتظم ذو قاعدة مربعة: طول ضلع القاعدة $c$، والارتفاع
$h$، والقمّة $S$ فوق مركز القاعدة. اِقطعه بالمستوي
المارّ \emph{بمنتصفات} \omterm{def:g5:solids:def}{الأحرف} الجانبية الأربعة $[SA]$
و $[SB]$ و $[SC]$ و $[SD]$.
\begin{enumerate}[resume]
  \item طبّق مبرهنة المنتصفين
        (\cref{thm:g8:midpoints:direct}) في المثلث $SAB$:
        فما اِتجاه وطول \omterm{def:g6:lines:objects}{القطعة} الواصلة بين
        منتصفَي $[SA]$ و $[SB]$؟ واِستنتج أن المقطع
        مربع طول ضلعه $\frac{c}{2}$، \omterm{def:g6:lines:perp}{موازٍ} للقاعدة،
        على اِرتفاع $\frac{h}{2}$.
  \item \omterm{def:g6:lines:objects}{والقطعة} التي فوق المقطع \omterm{def:g8:solids:def}{هرم} ذو قاعدة مربعة بنفسه.
        أعطِ طول ضلع قاعدته واِرتفاعه، وبيّن أن حجمه
        هو \emph{ثُمن} \omterm{def:g6:measure:volume}{الحجم} الأصلي $V$ بالضبط.
  \item واِستنتج \omterm{def:g6:measure:volume}{حجم} \omterm{def:g6:lines:objects}{القطعة} السفلى (وهي \emph{جذع \omterm{def:g8:solids:def}{الهرم}}،
        شكل \omterm{def:g8:solids:def}{الهرم} غير المكتمل). فبأيّ نسبة يتقاسم
        قطعُ نصف الارتفاع الحجمَ؟
  \item \omterm{def:g8:solids:def}{هرم} اللوفر (\cref{exo:g8:solids:7}: طول ضلع القاعدة
        $35$ م، والارتفاع $22$ م) يُنظَّف في حملتين: الزجاج
        فوق نصف الارتفاع، ثم الزجاج تحته. فأيّ
        \omterm{def:g6:fractions:def}{كسر} من \emph{\omterm{def:g6:measure:volume}{الحجم}} يقع تحت نصف الارتفاع؟
        قدّره بالأمتار المكعبة، إلى أقرب مئة.
  \item والدرس العامّ: إذا ضُربت \emph{كل} أبعاد
        \omterm{def:g8:solids:def}{هرم} في $2$، فماذا يحدث
        لحجمه؟ وبأيّ معامل ينمو \omterm{def:g6:measure:volume}{الحجم} حين يُضرب كل
        بُعد في $k$ --- وكيف يفسّر هذا، في سطر واحد،
        «الثُّمن» في السؤال 12؟
\end{enumerate}
\end{problem}
