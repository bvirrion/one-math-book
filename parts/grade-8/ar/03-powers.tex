\chapter{القوى}\label{ch:g8:powers}

اِطوِ ورقة إلى نصفين $10$ مرات (إن اِستطعت!): فيتضاعف سُمكها
في كل مرة، \omterm{def:g5:division:multiple}{والمضاعفات} $10$ تضربه في
$2^{10} = 1024$. والقوى اِختصار الضرب المتكرّر؛
ويقيم هذا الفصل الترميز وقواعده، مع دور خاصّ
لقوى العدد $10$.

\section{التعريف}

\begin{definition}[القوة]\label{def:g8:powers:def}
من أجل عدد $a$ وعدد صحيح $n \geq 1$:
\[
a^n = \underbrace{a \times a \times \dots \times a}_{n
\text{ عاملًا}},
\]
ويُقرأ «$a$ أُسّ $n$»؛ فالعدد $a$ هو \emph{الأساس}، و $n$ هو
\emph{الأُسّ}\index{أس}. وأسماء خاصّة: $a^2$ هو «مربع $a$»،
و $a^3$ هو «\omterm{def:g5:solids:def}{مكعب} $a$». وباصطلاح:
\[
a^1 = a,
\qquad
a^0 = 1 \ (a \neq 0),
\qquad
a^{-n} = \frac{1}{a^n}.
\]
\end{definition}

\begin{example}\label{ex:g8:powers:def}
$3^4 = 3 \times 3 \times 3 \times 3 = 81$؛ \quad
و $10^3 = 1000$؛ \quad و $5^{-2} = \frac{1}{25} = 0.04$؛ \quad
و $(-2)^3 = -8$ و $(-2)^4 = +16$
(\cref{ex:g8:negprod:powers}). واِنتبه: فالقوة $a^n$ \emph{ليست}
$a \times n$: \ فالقوة $2^5 = 32$، لا $10$.
\end{example}

\section{قواعد الأُسُس}

\begin{theorem}[قواعد الأُسُس]\label{thm:g8:powers:rules}
من أجل أساس $a$ غير معدوم وأُسّين صحيحين $m$ و $n$:
\[
a^m \times a^n = a^{m+n},
\qquad
\frac{a^m}{a^n} = a^{m-n},
\qquad
\left(a^m\right)^n = a^{m \times n},
\qquad
(ab)^n = a^n b^n .
\]
\end{theorem}

\begin{proof}[برهان بعدّ العوامل]
\omterm{def:g2:mult:def}{الجداء} $a^m \times a^n$ يصفّ $m$ عاملًا $a$ يتبعها $n$ أخرى:
أي $m + n$ في المجموع. والقوة $\left(a^m\right)^n$ تكرّر \omterm{def:g2:measure:units}{كتلة} من $m$ عاملًا
$n$ مرة: أي $mn$ عاملًا. والقوة $(ab)^n$ تحوي $n$ حرف $a$ و $n$
حرف $b$، ويمكن إعادة تجميعها. وتأتي قاعدة \omterm{def:g3:division:remainder}{خارج القسمة} من
تلاشي $n$ من العوامل $m$؛ ومع الاصطلاحين
$a^0 = 1$ و $a^{-n} = \frac{1}{a^n}$، تبقى صحيحة حتى حين
$n \geq m$.
\end{proof}

\begin{example}\label{ex:g8:powers:rules}
\[
2^3 \times 2^5 = 2^8 = 256,
\qquad
\frac{7^6}{7^4} = 7^2 = 49,
\qquad
\left(5^2\right)^3 = 5^6,
\qquad
2^4 \times 5^4 = 10^4 .
\]
وفخّ: القواعد تنطبق على \emph{أساس مشترك} (أو \omterm{def:g8:powers:def}{أُسّ}
مشترك، في الأخيرة). ولا قاعدة تبسّط $2^3 \times 5^2$ ---
بل اِحسب فحسب: $8 \times 25 = 200$.
\end{example}

\section{قوى العشرة}

\begin{proposition}[قوى العشرة]\label{prop:g8:powers:ten}
من أجل $n \geq 1$: $10^n = 1\underbrace{0\dots0}_{n}$، و
$10^{-n} = 0.\underbrace{0\dots0}_{n-1}1$. وتُقرأ قواعد \omterm{def:g8:powers:def}{الأُسُس}
هكذا: ضرب قوى العشرة يجمع \omterm{def:g8:powers:def}{الأُسُس}.
\end{proposition}

\begin{example}\label{ex:g8:powers:ten}
$10^4 \times 10^3 = 10^7$؛ \quad
و $\dfrac{10^2}{10^5} = 10^{-3} = 0.001$. فتصير المقادير الكبيرة والصغيرة
مقروءة:
\[
\text{مليار} = 10^9,
\qquad
\text{جزء من مليون} = 10^{-6}.
\]
ومع \omterm{def:g6:decimals:places}{الأعداد العشرية}: $3.2 \times 10^5 = 320\,000$ و
$4.7 \times 10^{-3} = 0.0047$ (بنقل \omterm{def:g4:decimals:point}{النقطة العشرية}،
\cref{prop:g6:decimals:shift}). والاستعمال المنهجي لهذه الكتابة ---
\emph{الترميز العلمي} --- يُطوَّر في
\cref{ch:g9:fractions}.
\end{example}

\begin{example}[رتب المقادير]\label{ex:g8:powers:magnitude}
يقطع الضوء نحو $3 \times 10^8$ م في الثانية؛ وفي السنة نحو
$3.2 \times 10^7$ ثانية. ومنه فالسنة الضوئية نحو
\[
3 \times 3.2 \times 10^{8+7} \approx 10 \times 10^{15} = 10^{16}
\text{ م}
\]
--- أي عشرة ملايين مليار متر. فقوى العشرة تجعل الحسابات
الفلكية تسع في سطر واحد.
\end{example}

\begin{method}[تبسيط عبارة فيها قوى]\label{met:g8:powers:simplify}
\begin{enumerate}
  \item جمّع العوامل أساسًا أساسًا؛
  \item وطبّق قواعد \omterm{def:g8:powers:def}{الأُسُس} داخل كل أساس؛
  \item واِحسب القوى الصغيرة الباقية، أو اِترك الجواب على صورة
        قوة إذا كان كبيرًا.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:powers:simplify}
\[
\frac{3^5 \times 3^{-2} \times 4^2}{3^2}
= \frac{3^{5 + (-2)}}{3^2} \times 16
= 3^{3 - 2} \times 16
= 3 \times 16 = 48 .
\]
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g8:powers:1}
اِحسب:
\[
2^6, \qquad 3^3, \qquad 10^5, \qquad 1^{100}, \qquad 0.1^2, \qquad
6^0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:2}
اِحسب:
\[
(-3)^2, \qquad -3^2, \qquad (-1)^{15}, \qquad (-5)^3, \qquad
\left(\tfrac{2}{3}\right)^2 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:3}
اُكتب على صورة قوة واحدة:
\[
7^4 \times 7^5, \qquad
\frac{2^9}{2^3}, \qquad
\left(10^3\right)^4, \qquad
5^6 \times 5^{-2}, \qquad
3^4 \times 7^4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:4}
اُكتب على صورة \omterm{def:g6:decimals:places}{عدد عشري}: $10^{-2}$؛ \quad و $4 \times 10^3$؛ \quad
و $2.5 \times 10^{-4}$؛ \quad و $10^0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:5}
اُكتب بقوة للعشرة: مئة ألف؛ وعُشر؛ وعشرة
مليارات؛ و $0.000\,001$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:6}
بسّط، ثم اِحسب:
\[
\frac{10^7 \times 10^{-3}}{10^2},
\qquad
\frac{2^5 \times 2^4}{2^6},
\qquad
\left(2^2\right)^3 \times 2^{-4} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:7}
صواب أم خطأ؟ صحّح الخاطئة.
\[
2^3 \times 2^4 = 2^{12}; \qquad
5^2 + 5^3 = 5^5; \qquad
(3^2)^4 = 3^8; \qquad
10^3 \times 10^3 = 100^3 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:powers:8}
تنتشر إشاعة: ففي اليوم 1 يعرفها ثلاثة أشخاص؛ وفي كل يوم، يخبر كل
شخص يعرفها ثلاثة أشخاص جدد. اُكتب بقوة عدد
الأشخاص \emph{الجدد} الذين أُخبروا في اليوم $4$، واِحسب كم شخصًا
يعرف الإشاعة في نهاية اليوم 4 (بمن فيهم الثلاثة الأوائل).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:powers:9}
سُمك ورقة $0.1$ مم، أي\ $10^{-4}$ م. وطيّها
يضاعف سُمكها في كل مرة.
\begin{enumerate}
  \item عبّر عن السُّمك بعد $10$ طيّات على صورة \omterm{def:g2:mult:def}{جداء}،
        واحسبه بالسنتيمتر ($2^{10} = 1024$).
  \item وبعد $42$ طيّة يكون السُّمك
        $2^{42} \times 10^{-4}$ م، مع $2^{42} \approx 4.4 \times
        10^{12}$. بيّن أن هذا يتجاوز مسافة الأرض--القمر،
        وهي نحو $3.8 \times 10^8$ م.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:powers:10}
رتّب من الأصغر إلى الأكبر، دون آلة حاسبة:
\[
2^{10}, \qquad 10^3, \qquad 3^6, \qquad 5^4 .
\]
(اِحسب كل واحد؛ فالقوّتان $2^{10}$ و $10^3$ جارتان مشهورتان.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:powers:11}
أيّهما أكبر، $2^{100}$ أم $10^{30}$؟ اِستعمل
$2^{10} = 1024 > 10^3$ لمقارنة
$2^{100} = \left(2^{10}\right)^{10}$ مع
$\left(10^3\right)^{10}$.
\end{exercise}

\section{مسألة: رقعة الشطرنج وقوى الاثنين}

\begin{problem}[{مسألة نهاية الأسبوع --- \omterm{def:g1:addition:def}{المجموع} الهندسي
$1 + 2 + 4 + \dots + 2^{n-1} = 2^n - 1$، من أسطورة مشهورة إلى
الأعداد الثنائية}]
\label{pb:g8:powers:1}
الأسطورة: مكافأةً على اِختراع الشطرنج، طلب الحكيم سيسّا
من ملكه حبة قمح واحدة على المربع الأول من الرقعة،
واِثنتين على الثاني، وأربعًا على الثالث --- مع المضاعفة من مربع إلى
مربع، حتى الرابع والستين. فضحك الملك من هذا التواضع.
وتحسب هذه المسألة ما وعد به الملك، مستعملةً قواعد \omterm{def:g8:powers:def}{الأُسُس}
في \cref{thm:g8:powers:rules}، وتنتهي حيث تقود القصة
سرًّا: إلى الأعداد الثنائية داخل كل حاسوب.

\medskip
\noindent\textbf{الجزء الأول --- حيلة المضاعفة.}
من أجل $n \geq 1$، ليكن $S_n$ العدد الكلي للحبّات على المربعات
$n$ الأولى.
\begin{enumerate}
  \item عبّر عن عدد الحبّات على المربع $k$ بقوة للعدد
        $2$. فأيّ قوة تقع على المربع $64$؟
  \item اِحسب $S_1$ و $S_2$ و $S_3$ و $S_4$ و $S_5$، وقارن
        كل واحد بقوة قريبة للعدد $2$. وخمّن صيغة
        \omterm{def:g1:addition:def}{للمجموع} $S_n$.
  \item \emph{حيلة المضاعفة}: اُكتب المجموعين $S_n$ و
        $2 \times S_n$ أحدهما تحت الآخر، واِطرح، وبَرهِن على
        تخمينك:
        \[
        S_n = 1 + 2 + 4 + \dots + 2^{n-1} = 2^n - 1 .
        \]
  \item فكم حبة وعد بها الملك في المجموع؟ عبّر عن
        الجواب بقوة للعدد $2$، وأكمل الملاحظة
        الكلاسيكية: «تحمل الرقعة كلها حبة واحدة أقلّ مما
        كان يحمله مربع خامس وستون وحده.»
  \item بيّن أن النصف الثاني من الرقعة (المربعات من $33$ إلى
        $64$) يحمل \emph{بالضبط} $2^{32}$ مثل ما يحمله
        النصف الأول.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- كم يبلغ $2^{64}$؟}
المقارنة $2^{10} = 1024 > 10^3$ في
\cref{exo:g8:powers:11} هي مفتاح كل التقديرات أدناه.
\begin{enumerate}[resume]
  \item بيّن أن
        $2^{64} = 2^4 \times \left(2^{10}\right)^6 > 1.6 \times
        10^{19}$.
  \item تزن حبة القمح نحو $0.05$ غ، أي\
        $5 \times 10^{-2}$ غ. بيّن أن القمح الموعود يزن
        أكثر من $8 \times 10^{17}$ غ، وحوّل هذا إلى
        أطنان ($1$ طن $= 10^6$ غ).
  \item يحصد العالم كله اليوم نحو $8 \times 10^8$
        طن من القمح في السنة. فكم سنة من حصاد العالم
        وعد بها الملك على الأقلّ؟
  \item جِد أصغر عدد صحيح $n$ يحقّق
        $2^n > 10^6$ --- أي كم مضاعفة يلزم
        لتجاوز المليون. (اِحسب $2^{19}$ و $2^{20}$ بالضبط،
        مستعملًا $2^{10} = 1024$.)
  \item يعرض عليك محتال راتب شهر: $1$ سنت في اليوم
        $1$، ثم ضعف أجر اليوم السابق في كل يوم. ففي أيّ
        يوم يتجاوز الأجر \emph{اليومي} وحده أول مرة مليون
        يورو ($10^8$ سنت)؟ (اِحسب $2^{26}$ و
        $2^{27}$ بالضبط.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الأوزان الثنائية.}
عند تاجرة خمسة صنجات: $1$ و $2$ و $4$ و $8$ و $16$ غرامًا،
واحد من كل. وتضع بعضها في كفّة ميزان
لتزن بضاعة في الكفّة الأخرى.
\begin{enumerate}[resume]
  \item فأيّ صنجات تضع لتزن $21$ غ؟ ولتزن
        $27$ غ؟
  \item اِشرح لماذا \emph{يجب} على أيّ هدف من $16$ غ فأكثر أن
        يستعمل الصنجة $16$ غ، ولماذا يجب على أيّ هدف من $15$ غ فأقلّ
        ألا يستعملها. (والسؤال 3 يخبرك بأقصى ما تبلغه
        الصنجات $1, 2, 4, 8$.) واِشرح لماذا يتكرّر
        الاستدلال نفسه مع الصنجة التالية في الكبر، في
        كل مرحلة.
  \item اِستنتج أن كل هدف صحيح من $1$ إلى $31$ غ يمكن
        وزنه، و\emph{بطريقة واحدة بالضبط}: فكل عدد
        بين $1$ و $31$ \omterm{def:g1:addition:def}{مجموع} قوى متمايزة للعدد $2$ بأسلوب
        واحد.
  \item تشتري التاجرة صنجة سادسة، وزنها $32$ غ. فإلى أيّ
        هدف تستطيع أن تزن الآن؟ واُكتب $45$ غ مجموعَ
        قوى متمايزة للعدد $2$.
  \item في الحاسوب، يُخزَّن العدد «ذو 64 خانة» على $64$
        مربعًا، يحمل كل منها $0$ أو $1$ --- فيسهم المربع $k$
        بالمقدار $2^{k-1}$ حين يحمل $1$، كحبّات
        الأسطورة. وباستعمال الجزء الأول، اِشرح لماذا تمتدّ الأعداد
        الصحيحة التي تخزّنها آلة كهذه بالضبط من $0$ إلى
        $2^{64} - 1$.
\end{enumerate}
\end{problem}
