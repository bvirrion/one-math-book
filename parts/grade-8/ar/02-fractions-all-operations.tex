\chapter{الكسور: العمليات الأربع}\label{ch:g8:fractions}

يُتمّ هذا الفصل حساب الكسور: الجمع والطرح
بمقامات \emph{أيًّا} كانت، والضرب، و---
الوافدة الجديدة --- القسمة، التي تتبيّن أنها ضرب
متنكّر. وتأتي معها الإشارات من \cref{ch:g8:negprod}.

\section{الجمع بأيّ مقامات}

\begin{method}[المقام المشترك]\label{met:g8:fractions:add}
لجمع $\frac ab$ و $\frac cd$ أو طرحهما:
\begin{enumerate}
  \item جِد مضاعفًا مشتركًا للعددين $b$ و $d$ (\omterm{def:g2:mult:def}{والجداء}
        $b \times d$ يصلح دائمًا؛ وأصغر منه يوفّر الجهد)؛
  \item وأعِد كتابة الكسرين بذلك \omterm{def:g4:fractions:def}{المقام}؛
  \item واجمع البسطين أو اِطرحهما؛ ثم اِختزل.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:fractions:add}
اِحسب $\dfrac{5}{6} + \dfrac{3}{4}$. \omterm{def:g5:division:multiple}{ومضاعف} مشترك للعددين $6$ و
$4$ هو $12$:
\[
\frac{5}{6} = \frac{10}{12},
\qquad
\frac{3}{4} = \frac{9}{12},
\qquad
\frac{5}{6} + \frac{3}{4} = \frac{10 + 9}{12} = \frac{19}{12}.
\]
ومع الإشارات: $\dfrac{2}{3} - \dfrac{7}{5} = \dfrac{10}{15} -
\dfrac{21}{15} = -\dfrac{11}{15}$.
\end{example}

\section{ضرب الكسور}

\begin{theorem}[جداء كسرين]\label{thm:g8:fractions:mult}
\[
\frac{a}{b} \times \frac{c}{d} = \frac{a \times c}{b \times d} .
\]
\end{theorem}

\begin{proof}[لماذا، على صورة]
خذ $\frac34$ من $\frac25$ من مربع: اِقسم المربع إلى $5$
شرائح عمودية واِحتفظ بعدد $2$ منها؛ ثم اِقسم أفقيًّا إلى $4$ واِحتفظ بعدد $3$
أسطر مما بقي. والجزء المحتفظ به شبكة من $3 \times 2$ خلية
صغيرة من $4 \times 5$: أي \omterm{def:g6:fractions:def}{الكسر} $\frac{6}{20}$.
\end{proof}

\begin{example}\label{ex:g8:fractions:mult}
اِختزل \emph{قبل} الضرب، بشطب العوامل المشتركة:
\[
\frac{7}{12} \times \frac{9}{14}
= \frac{7 \times 9}{12 \times 14}
= \frac{7 \times 3 \times 3}{3 \times 4 \times 2 \times 7}
= \frac{3}{8}
\]
(فالعامل $7$ وعامل $3$ واحد يتلاشيان بين الأعلى والأسفل).
ومع الإشارات، تنطبق قواعد \cref{thm:g8:negprod:rules} أولًا:
$\left(-\frac23\right) \times \left(-\frac{5}{4}\right) =
+\frac{10}{12} = \frac56$.
\end{example}

\section{قسمة الكسور}

\begin{definition}[المقلوب]\label{def:g8:fractions:inverse}
\emph{مقلوب}\index{مقلوب} عدد $x$ غير معدوم هو
العدد الذي إذا ضُرب في $x$ أعطى $1$. ومقلوب
$\frac ab$ هو $\frac ba$، لأن
$\frac ab \times \frac ba = \frac{ab}{ab} = 1$؛ ومقلوب العدد
الصحيح $n$ هو $\frac1n$.
\end{definition}

\begin{theorem}[القسمة ضرب في المقلوب]\label{thm:g8:fractions:div}
من أجل $c \neq 0$:
\[
\frac{a}{b} \div \frac{c}{d} = \frac{a}{b} \times \frac{d}{c} .
\]
\end{theorem}

\begin{proof}
بالتعريف، خارجُ القسمة $q$ \omterm{def:g6:fractions:def}{للكسر} $\frac ab$ على $\frac cd$ هو
العدد الذي يحقّق $q \times \frac cd = \frac ab$. وتحقّق من المرشّح
$q = \frac ab \times \frac dc$:
\[
\left(\frac ab \times \frac dc\right) \times \frac cd
= \frac ab \times \left(\frac dc \times \frac cd\right)
= \frac ab \times 1 = \frac ab . \qedhere
\]
\end{proof}

\begin{example}\label{ex:g8:fractions:div}
\[
\frac{3}{5} \div \frac{7}{10}
= \frac{3}{5} \times \frac{10}{7}
= \frac{3 \times 10}{5 \times 7}
= \frac{30}{35} = \frac{6}{7}.
\]
وتحقّق سريع: القسمة على $\frac{7}{10}$، وهو عدد أصغر من
$1$، يجب أن تعطي نتيجة \emph{أكبر} من $\frac35$ --- وفعلًا
$\frac67 > \frac35$ ($\frac{30}{35} > \frac{21}{35}$).
\end{example}

\begin{example}[شرائط كسر داخل شرائط كسر]\label{ex:g8:fractions:complex}
\omterm{def:g6:fractions:def}{الكسر} «ذو الطابقين» ليس إلا قسمة مكتوبة عموديًّا:
\[
\frac{\ \frac{2}{3}\ }{\ \frac{5}{6}\ }
= \frac{2}{3} \div \frac{5}{6}
= \frac{2}{3} \times \frac{6}{5}
= \frac{12}{15} = \frac{4}{5}.
\]
حدّد الشريط \emph{الرئيسي} أولًا (وهو الأطول)، ثم طبّق
\cref{thm:g8:fractions:div}.
\end{example}

\begin{method}[حسابات مختلطة]\label{met:g8:fractions:mixed}
في عبارة تخلط الكسور والعمليات الأربع:
\begin{enumerate}
  \item طبّق الأولويات المعتادة (\cref{ch:g7:priorities})؛
  \item وحوّل كل قسمة إلى ضرب في \omterm{def:g8:fractions:inverse}{المقلوب}؛
  \item وحدّد الإشارات أولًا (\cref{met:g8:negprod:compute})، ثم
        اِضرب أو جِد المقامات المشتركة؛
  \item واِختزل في أول فرصة.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:fractions:mixed}
\begin{align*}
\frac12 + \frac{2}{3} \times \frac{9}{4}
&= \frac12 + \frac{2 \times 9}{3 \times 4}
&& \text{(الجداء أولًا!)} \\
&= \frac12 + \frac{18}{12} = \frac12 + \frac32
= \frac{4}{2} = 2 .
\end{align*}
\end{example}

\section{مسألة: الكسور المصرية}

\begin{problem}[{مسألة نهاية الأسبوع --- كل كسر مجموع
كسور واحدية متمايزة}]
\label{pb:g8:fractions:1}
\emph{\omterm{def:g6:fractions:def}{الكسر} الواحدي} \omterm{def:g6:fractions:def}{كسر} بسطه $1$. وكان
كتبة مصر القديمة --- ونُسخت بردية ريند نحو سنة
$1550$~قبل الميلاد --- يكتبون كل \omterm{def:g6:fractions:def}{كسر} مجموعَ كسور واحدية
\emph{متمايزة}: فلا يكتبون $\frac27 = \frac17 + \frac17$ أبدًا، بل
$\frac27 = \frac14 + \frac{1}{28}$. ويُسمّى \omterm{def:g1:addition:def}{مجموع} كهذا
\emph{كتابة مصرية} \omterm{def:g6:fractions:def}{للكسر}. وتطوّر هذه المسألة
تقنيات الجمع والمقارنة في هذا الفصل
(\cref{met:g8:fractions:add}) إلى طريقة --- نشرها
فيبوناتشي سنة $1202$ --- تنتج كتابة مصرية لأيّ
\omterm{def:g6:fractions:def}{كسر} بين $0$ و $1$، وتنتهي بلغز ميراث
مشهور.

\medskip
\noindent\textbf{الجزء الأول --- الكسور الواحدية ومتطابقة
التشطير.}
\begin{enumerate}
  \item اِحسب $\frac12 + \frac13$ و $\frac12 + \frac14$، واِستنتج
        كتابتين مصريتين للكسرين $\frac56$ و $\frac34$.
  \item تحقّق من مدخل الكتبة \omterm{def:g6:fractions:def}{للكسر} $\frac27$: بيّن أن
        $\frac14 + \frac{1}{28} = \frac27$.
  \item تحقّق من أن $\frac13 + \frac16 = \frac12$ ومن أن
        $\frac14 + \frac{1}{12} = \frac13$. ثم بَرهِن على
        \emph{متطابقة التشطير}: من أجل كل عدد صحيح
        $n \geq 1$،
        \[
        \frac{1}{n} = \frac{1}{n+1} + \frac{1}{n(n+1)} .
        \]
  \item اِستنتج كتابة مصرية للعدد $1$ نفسه، مجموعَ
        ثلاثة كسور واحدية متمايزة.
  \item شطّر الحدّ الأخير من $\frac56 = \frac12 + \frac13$
        لتحصل على كتابة مصرية \emph{ثانية} \omterm{def:g6:fractions:def}{للكسر} $\frac56$، بثلاثة
        كسور واحدية. واِشرح لماذا لا يكون لأيّ \omterm{def:g6:fractions:def}{كسر} كتابة
        مصرية وحيدة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- طريقة فيبوناتشي الجشعة.}
والطريقة هي: اِطرح أكبر \omterm{def:g6:fractions:def}{كسر} واحدي
يدخل، وكرّر مع ما يبقى.
\begin{enumerate}[resume]
  \item باستعمال المقامات المشتركة، بيّن أن
        $\frac14 < \frac27 < \frac13$. ولماذا يبرهن هذا على أن
        $\frac14$ أكبر \omterm{def:g6:fractions:def}{كسر} واحدي أصغر من
        $\frac27$؟
  \item اِحسب $\frac27 - \frac14$. فأيّ كتابة \omterm{def:g6:fractions:def}{للكسر} $\frac27$
        تنتجها الطريقة الجشعة؟
  \item والآن خذ $\frac57$. بيّن أن أكبر \omterm{def:g6:fractions:def}{كسر} واحدي
        أصغر من $\frac57$ هو $\frac12$، واِحسب
        $\frac57 - \frac12$.
  \item بيّن أن أكبر \omterm{def:g6:fractions:def}{كسر} واحدي أصغر من
        $\frac{3}{14}$ هو $\frac15$ (قارن $\frac{3}{14}$ مع
        $\frac14$ ومع $\frac15$)، واِحسب
        $\frac{3}{14} - \frac15$، واِستنتج:
        \[
        \frac57 = \frac12 + \frac15 + \frac{1}{70} .
        \]
        وتحقّق من هذه الكتابة مباشرةً \omterm{def:g4:fractions:def}{بالمقام} المشترك
        $70$.
  \item شغّل الطريقة الجشعة على $\frac45$، متحقّقًا في كل خطوة
        من أن كسرك الواحدي هو الأكبر الذي يدخل،
        وتحقّق من الكتابة النهائية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- لماذا تتوقف الطريقة دائمًا، والجمل
الثامن عشر.}
\begin{enumerate}[resume]
  \item في الأسئلة 7--10 طُبّقت الطريقة على $\frac27$
        و $\frac57$ و $\frac45$. ومن أجل كل واحد، اُذكر بسوط
        البواقي المتعاقبة (قبل أيّ اِختزال).
        فماذا تلاحظ؟
  \item من أجل \omterm{def:g6:fractions:def}{كسر} $\frac ab$ وعدد صحيح $n$، بيّن
        أن
        \[
        \frac{a}{b} - \frac{1}{n}
        = \frac{a \times n - b}{b \times n} .
        \]
  \item اِفترض أن الطريقة الجشعة تطرح $\frac1n$ من
        $\frac ab$: فيدخل $\frac1n$ لكن $\frac{1}{n-1}$
        كبير أصلًا، أي $\frac{1}{n-1} > \frac ab$. وبوضع الاثنين
        على \omterm{def:g4:fractions:def}{المقام} المشترك $b \times (n - 1)$، اِستنتج
        أن $b > a \times n - a$، ومنه أن \omterm{def:g4:fractions:def}{البسط} الجديد
        يحقّق
        \[
        a \times n - b < a .
        \]
        واِشرح لماذا يبرهن هذا على أن الطريقة تتوقف دائمًا.
  \item حكاية قديمة: يموت أب تاركًا $17$ جملًا، موصيًا
        \omterm{def:g6:fractions:def}{بالكسر} $\frac12$ من القطيع لابنه الأكبر، و $\frac13$
        للثاني، و $\frac19$ للثالث --- ولا يكون أيّ من
        الأنصبة عددًا صحيحًا من الجمال. فيقرض جار
        الأسرةَ جملًا ثامن عشر؛ فيأخذ الأولاد $9$ و $6$
        و $2$ جمال، ويُردّ الجمل المُقترض. اِحسب
        $\frac12 + \frac13 + \frac19$ واِشرح الحيلة: لماذا
        أمكن تقاسم القطيع بجمال صحيحة، ولماذا لم ينل أيّ
        اِبن أقلّ مما وعدت به الوصية؟
  \item أصلح الوصية: اِقترح ثلاثة كسور واحدية
        \emph{متمايزة} مجموعها $1$ بالضبط، وأعطِ
        أصغر \omterm{def:g6:measure:volume}{حجم} قطيع تكون فيه الأنصبة الثلاثة كلها أعدادًا
        صحيحة من الجمال.
\end{enumerate}
\end{problem}
