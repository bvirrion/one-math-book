\chapter{الحساب الحرفي والمعادلات}\label{ch:g8:equations}

قدّمت السنة السابعة الحروف واِختزلت العبارات البسيطة
(\cref{ch:g7:literal}). ويضيف هذا الفصل المهارتين اللتين تجعلان
الجبر قويًّا: \emph{نشر} \omterm{def:g2:mult:def}{جداءات} الأقواس، و
\emph{حلّ \omterm{def:g8:equations:def}{المعادلات}} --- أي إيجاد العدد المختبئ وراء
الحرف.

\section{النشر}

\begin{theorem}[التوزيعية البسيطة والمضاعفة]\label{thm:g8:equations:expand}
من أجل كل الأعداد:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{theorem}

\begin{proof}
القاعدة الأولى هي \cref{thm:g7:priorities:distributivity} (بما فيها
الإشارات: فقد يكون $k$ والحدود سالبة،
\cref{ch:g8:negprod}). وأما الثانية، فطبّق القاعدة الأولى مرتين،
معاملًا $(c + d)$ عددًا واحدًا:
\[
(a + b)(c + d) = a(c + d) + b(c + d) = ac + ad + bc + bd .
\qedhere
\]
\end{proof}

\begin{example}\label{ex:g8:equations:expand}
واِنتبه للإشارات في كل حدّ:
\begin{align*}
3(2x - 5) &= 6x - 15, \\
-2(4x - 7) &= -8x + 14
&& \text{(الناقص في المقدمة يقلب الإشارتين)}, \\
(x + 3)(x + 5) &= x^2 + 5x + 3x + 15 = x^2 + 8x + 15, \\
(2x - 1)(x - 4) &= 2x^2 - 8x - x + 4 = 2x^2 - 9x + 4 .
\end{align*}
\end{example}

\begin{method}[اُنشر واِختزل]\label{met:g8:equations:reduce}
\begin{enumerate}
  \item اُنشر كل \omterm{def:g2:mult:def}{جداء}، كاتبًا كل حدّ بإشارته؛
  \item وجمّع الحدود المتشابهة ($x^2$ مع $x^2$، و $x$ مع $x$، والأعداد
        مع الأعداد)؛
  \item وتحقّق بقيمة: عوّض مثلًا $x = 2$ في الأصل
        وفي النتيجة --- فتساوي النتيجتين يمسك معظم الأخطاء.
\end{enumerate}
\end{method}

\section{حلّ المعادلات}

\begin{definition}[المعادلة]\label{def:g8:equations:def}
\emph{المعادلة}\index{معادلة} تساوٍ فيه عدد
مجهول، يُكتب $x$ (أو أيّ حرف). و\emph{حلّها} يعني إيجاد
كل قيم $x$ التي تجعل التساوي صحيحًا --- وهي
\emph{الحلول}.
\end{definition}

\begin{theorem}[قواعد التوازن]\label{thm:g8:equations:balance}
تحتفظ \omterm{def:g8:equations:def}{المعادلة} بالحلول نفسها بالضبط حين:
\begin{enumerate}
  \item يُجمع العدد نفسه إلى الطرفين (أو يُطرح منهما)؛
  \item ويُضرب الطرفان (أو يُقسمان) في العدد نفسه
        \emph{غير المعدوم}.
\end{enumerate}
\end{theorem}
\admitted

\begin{method}[حلّ معادلة من الدرجة الأولى]\label{met:g8:equations:solve}
\begin{enumerate}
  \item اُنشر واِختزل كل طرف عند اللزوم؛
  \item واِجمع أو اِطرح في الطرفين لتجمع حدود $x$ في طرف
        والأعداد الصرفة في الآخر؛
  \item واِختزل، ثم اِقسم الطرفين على معامل $x$؛
  \item و\emph{تحقّق} بتعويض القيمة الموجودة في \omterm{def:g8:equations:def}{المعادلة}
        الأصلية.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:equations:solve}
حُلّ $7x - 5 = 4x + 16$:
\begin{align*}
7x - 5 &= 4x + 16 \\
7x - 4x &= 16 + 5
&& \text{(بجمع $5$ وطرح $4x$ في الطرفين)} \\
3x &= 21 \\
x &= 7 && \text{(بقسمة الطرفين على 3).}
\end{align*}
والتحقّق: الطرف الأيسر $7 \times 7 - 5 = 44$؛ والطرف الأيمن
$4 \times 7 + 16 = 44$. فالحلّ هو $7$.
\end{example}

\begin{example}[بأقواس وأعداد سالبة]\label{ex:g8:equations:brackets}
حُلّ $3(x - 4) = 5 - 2(x + 1)$:
\begin{align*}
3x - 12 &= 5 - 2x - 2 && \text{(بنشر الطرفين)} \\
3x - 12 &= 3 - 2x && \text{(بالاختزال)} \\
5x &= 15 && \text{(بجمع $2x + 12$ إلى الطرفين)} \\
x &= 3 .
\end{align*}
والتحقّق: $3(3 - 4) = -3$ و $5 - 2(3+1) = 5 - 8 = -3$. \quad والحلّ:
$3$.
\end{example}

\begin{method}[حلّ مسألة بمعادلة]\label{met:g8:equations:problem}
\begin{enumerate}
  \item اِختر المجهول: «ليكن $x$ هو \dots» (المقدار المطلوب،
        أو مقدارًا أبسط مرتبطًا به)؛
  \item وترجم الحكاية إلى \omterm{def:g8:equations:def}{معادلة}؛
  \item وحُلّها؛
  \item وفسّر: أجب عن السؤال الأصلي في جملة، مع
        الوحدات، وتحقّق من الجواب بمقابلته بالحكاية.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:equations:problem}
لؤي أكبر من أخته بمقدار $4$ سنوات؛ \omterm{def:g1:addition:def}{ومجموع} سنّيهما معًا
$26$. ليكن $x$ سنّ الأخت؛ فيكون لؤي $x + 4$، ومنه
\[
x + (x + 4) = 26
\ \Longrightarrow\
2x + 4 = 26
\ \Longrightarrow\
2x = 22
\ \Longrightarrow\
x = 11 .
\]
فالأخت $11$، ولؤي $15$. والتحقّق: $11 + 15 = 26$ و
$15 - 11 = 4$.
\end{example}

\section{الترتيب والمتراجحات}

\begin{proposition}[العمليات والترتيب]\label{prop:g8:equations:order}
جمع العدد نفسه إلى طرفَي متراجحة يحفظها؛
وضرب الطرفين في عدد \emph{موجب} يحفظها؛
والضرب في عدد \emph{سالب} \emph{يعكسها}:
\[
2 < 5 \ \text{ لكن }\ (-1) \times 2 > (-1) \times 5 .
\]
\end{proposition}
\admitted

\begin{example}\label{ex:g8:equations:inequality}
حُلّ $5 - 2x \leq 11$: اِطرح $5$ من الطرفين، فتجد $-2x \leq 6$؛
واِقسم على $-2$ \emph{مع العكس}: فتجد $x \geq -3$. فكل الأعداد
الأكبر من $-3$ أو المساوية له حلول --- \omterm{def:g8:equations:def}{فالمعادلة} عادةً
لها حلّ واحد، وأما المتراجحة فلها نصف \omterm{def:g6:lines:objects}{مستقيم} كامل من الحلول.
\end{example}

\section{مسألة: المتطابقات الشهيرة وسُلَّم الأعداد
الفردية}

\begin{problem}[{مسألة نهاية الأسبوع --- $(a+b)^2$ و $(a-b)^2$
و $(a+b)(a-b)$، والمجموع $1 + 3 + 5 + \dots + (2n-1) = n^2$}]
\label{pb:g8:equations:1}
ثلاث عمليات نشر تتكرّر كثيرًا حتى صار الجبر يحفظها عن
ظهر قلب: وتُسمّى \emph{المتطابقات الشهيرة}. وتستنتجها هذه
المسألة من التوزيعية المضاعفة
(\cref{thm:g8:equations:expand})، وتحوّلها قوّةً خارقة في الحساب
الذهني، وتستعملها لتبرهن على واقعة جميلة عن \omterm{def:g3:numbers:evenodd}{الأعداد الفردية}،
وتنتهي بحلّ \omterm{def:g8:equations:def}{معادلات} من نوع جديد تمامًا: \omterm{def:g8:equations:def}{معادلات} فيها
مربع.

\medskip
\noindent\textbf{الجزء الأول --- المتطابقات الثلاث.}
\begin{enumerate}
  \item اُنشر $(a + b)^2 = (a + b)(a + b)$ واِختزل، لتبرهن على
        \[
        (a + b)^2 = a^2 + 2ab + b^2 .
        \]
  \item وبَرهِن كذلك على الرفيقتين:
        \[
        (a - b)^2 = a^2 - 2ab + b^2,
        \qquad
        (a + b)(a - b) = a^2 - b^2 .
        \]
  \item اُرسم مربعًا طول ضلعه $a + b$، واِقسم كل ضلع إلى $a$
        و $b$، واِقطع المربع إلى أربع قطع. فأيّ جزء
        من المتطابقة الأولى يمثّله كل جزء من
        الصورة؟
  \item الحساب الذهني: اِستعمل المتطابقات لتحسب، دون
        إجراء أيّ ضرب،
        \[
        101^2, \qquad 99^2, \qquad 49 \times 51 .
        \]
  \item وفخّ كلاسيكي: تكتب زينة «$(a + b)^2 = a^2 + b^2$».
        اِختبر صيغتها بالقيمتين $a = 3$ و $b = 4$، واِشرح على
        صورة السؤال 3 أيّ \omterm{def:g6:lines:objects}{القطع} بالضبط تنساها
        صيغتها.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- سُلَّم \omterm{def:g3:numbers:evenodd}{الأعداد الفردية}.}
\begin{enumerate}[resume]
  \item اِحسب $1$، ثم $1 + 3$، ثم $1 + 3 + 5$، ثم
        $1 + 3 + 5 + 7$، ثم $1 + 3 + 5 + 7 + 9$. فماذا
        تلاحظ على النتائج؟
  \item اِشرح لماذا يكون \omterm{def:g3:numbers:evenodd}{العدد الفردي} ذو الرتبة $n$ هو $2n - 1$ (والتحقّق:
        $n = 1, 2, 3$ تعطي $1, 3, 5$).
  \item اِستعمل متطابقة من الجزء الأول لتنشر وتختزل
        $(n + 1)^2 - n^2$. فما علاقة النتيجة
        بالسؤال 7؟
  \item يقول السؤال 8: للانتقال من مربع طول ضلعه $n$ إلى
        مربع طول ضلعه $n + 1$، يُضاف بالضبط \omterm{def:g3:numbers:evenodd}{العدد الفردي}
        التالي من خلايا الوحدة --- أي حاشية على شكل حرف L على طول ضلعين
        والركن. واِنطلاقًا من $1 = 1^2$ وتنمية
        المربع طبقةً طبقة، اِشرح لماذا
        \[
        1 + 3 + 5 + \dots + (2n - 1) = n^2 :
        \]
        أي إن \omterm{def:g1:addition:def}{مجموع} \omterm{def:g3:numbers:evenodd}{الأعداد الفردية} $n$ الأولى هو المربع ذو الرتبة $n$.
  \item اِستنتج، دون جمعها واحدًا واحدًا: قيمة
        $1 + 3 + 5 + \dots + 99$؛ وكم عددًا فرديًّا، اِبتداءً
        من $1$، مجموعها $400$ بالضبط.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- \omterm{def:g8:equations:def}{معادلات} فيها مربع.}
\begin{enumerate}[resume]
  \item تعرّف على متطابقة شهيرة في $x^2 + 6x + 9$، و
        حُلّ \omterm{def:g8:equations:def}{المعادلة} $x^2 + 6x + 9 = 0$.
  \item اِشرح، مستعملًا قاعدة الإشارة والمسافة في
        \cref{thm:g8:negprod:rules}، لماذا لا يمكن \omterm{def:g2:mult:def}{لجداء} عددين
        أن يكون معدومًا \emph{إلا إذا} كان أحد العاملين معدومًا.
        واِستعمل هذا لحلّ $(x - 2)(x + 5) = 0$.
  \item حُلّ $x^2 = 49$ بنقل كل شيء إلى طرف واحد
        وتعميل $x^2 - 49$ بالمتطابقة الثالثة. فكم حلًّا
        لهذه \omterm{def:g8:equations:def}{المعادلة} --- وبم يختلف ذلك
        عن كل \omterm{def:g8:equations:def}{معادلة} حُلّت في هذا الفصل حتى الآن؟
  \item في \cref{exo:g8:equations:3} تحقّقت من أن $x = 3$
        حلّ \omterm{def:g8:equations:def}{للمعادلة} $x^2 = 6x - 9$. بيّن أنه الحلّ
        \emph{الوحيد}: اُنقل كل شيء إلى طرف واحد، وتعرّف على
        متطابقة، واِستنتج.
  \item الخاتمة: اِحسب $2026^2 - 2025^2$ ذهنيًّا، واِشرح
        بمتطابقة لماذا يكون \omterm{ex:g1:subtraction:difference}{فرق} مربعين
        متتاليين دائمًا \omterm{def:g1:addition:def}{مجموع} العددين
        --- وهو السؤال 8 من جديد.
\end{enumerate}
\end{problem}
