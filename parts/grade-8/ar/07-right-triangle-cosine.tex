\chapter{المثلث القائم الزاوية وجيب التمام}\label{ch:g8:cosine}

واقعتان جميلتان تربطان \omterm{def:g6:shapes:triangles}{المثلث القائم الزاوية} بالدائرة: \omterm{def:g6:shapes:triangles}{فالمثلث
القائم الزاوية} يسع بالضبط في نصف دائرة، ووتره
\omterm{def:g4:geometry:circle}{قطر} لها. وعدد واحد، هو \emph{جيب تمام} الزاوية، يشفّر
شكل كل \omterm{def:g6:shapes:triangles}{مثلث قائم الزاوية} فيه تلك الزاوية --- وهو أول
نسبة مثلثية، قبل أخويه الجيب والظلّ
(\cref{ch:g9:trig}).

\section{المثلث القائم الزاوية ودائرته}

\begin{theorem}[مبرهنة الدائرة]\label{thm:g8:cosine:circle}
\begin{enumerate}
  \item إذا كان مثلث $ABC$ \omterm{def:g6:shapes:triangles}{قائم الزاوية} في $A$، فإن $A$ تقع على
        الدائرة التي قطرها الوتر $[BC]$.
  \item وبالعكس، إذا كانت $A$ على دائرة قطرها $[BC]$ (مع
        $A \neq B, C$)، فإن المثلث $ABC$ \omterm{def:g6:shapes:triangles}{قائم الزاوية} في
        $A$.
\end{enumerate}
وبعبارة مكافئة: في \omterm{def:g6:shapes:triangles}{المثلث القائم الزاوية}، يكون منتصف الوتر
على بُعد متساوٍ من \omterm{def:g5:solids:def}{الرؤوس} الثلاثة --- فالمتوسط الصادر من
\omterm{def:g3:shapes:rightangle}{الزاوية القائمة} يقيس نصف الوتر.
\end{theorem}

\begin{proof}[برهان 1]
ليكن $O$ منتصف $[BC]$ و $D$ \omterm{def:g7:central:def}{مماثلة} $A$ بالنسبة إلى
$O$. فقطرا $ABDC$ يتقاطعان في منتصفهما المشترك $O$، إذن
$ABDC$ \omterm{def:g7:central:parallelogram}{متوازي أضلاع} (\cref{def:g7:central:parallelogram}) ---
فيه \omterm{def:g3:shapes:rightangle}{زاوية قائمة} في $A$: فهو مستطيل. وقطرا
المستطيل متساويان، إذن $OA = \frac{AD}{2} = \frac{BC}{2}$: أي إن
النقطة $A$ على بُعد $\frac{BC}{2}$ من $O$، أي\ على الدائرة
التي قطرها $[BC]$. (والنقطة 2 يُبرهن عليها بتشغيل الاستدلال
\omterm{def:g8:fractions:inverse}{بالمقلوب}.)
\end{proof}

\section{جيب تمام زاوية حادّة}

\begin{definition}[جيب التمام]\label{def:g8:cosine:def}
في \omterm{def:g6:shapes:triangles}{مثلث قائم الزاوية}، من أجل زاوية حادّة $\theta$:
\[
\cos\theta = \frac{\text{الضلع المجاور للزاوية } \theta}{\text{الوتر}}
\]
--- أي ضلع القائمة الملامس للزاوية $\theta$، مقسومًا على الوتر. وهذه النسبة
تتوقف على الزاوية وحدها، لا على \omterm{def:g6:measure:volume}{حجم} المثلث: فكل المثلثات
القائمة الزاوية ذات الزاوية الحادّة نفسها تكبيرات بعضها لبعض،
والتكبيرات تحفظ نسب الأطوال
(\cref{ch:g8:midpoints} بدأ هذه القصة؛ و\cref{ch:g9:thales}
يتمّها).
\end{definition}

\begin{proposition}[أول القيم والحدود]\label{prop:g8:cosine:bounds}
من أجل كل زاوية حادّة $\theta$: $0 < \cos\theta < 1$، وجيب التمام
\emph{يتناقص} كلما اِنفتحت الزاوية: فالزاوية الأوسع جيب تمامها أصغر.
والمعالم: $\cos 0^\circ = 1$، و $\cos 60^\circ = \frac12$،
و $\cos 90^\circ = 0$ (وهي القيم القصوى الموافقة لمثلثات
مسطّحة).
\end{proposition}
\admitted

\begin{method}[اِستعمال جيب التمام]\label{met:g8:cosine:use}
في \omterm{def:g6:shapes:triangles}{مثلث قائم الزاوية}، حين تكون المقادير المعلومة والمطلوبة زاويةً
حادّة وضلعها المجاور والوتر:
\begin{enumerate}
  \item اُكتب التعريف:
        $\cos\theta = \frac{\text{المجاور}}{\text{الوتر}}$ مع
        وضع القيم المعلومة مكانها؛
  \item وحُلّ \omterm{def:g8:equations:def}{المعادلة} الصغيرة لإيجاد المجهول (بالضرب أو القسمة)؛
  \item ومن أجل \emph{زاوية} مجهولة، طبّق زرّ الآلة الحاسبة
        $\cos^{-1}$ على النسبة المحسوبة؛
  \item وتحقّقات بالنظر: فالطول يجب أن يخرج أقصر من
        الوتر؛ والزاوية بين $0^\circ$ و
        $90^\circ$ تمامًا.
\end{enumerate}
\end{method}

\begin{example}[إيجاد ضلع]\label{ex:g8:cosine:side}
منحدر طوله $6$ م يصنع زاوية $20^\circ$ مع الأرض الأفقية.
والمسافة الأفقية المقطوعة (المجاورة للزاوية $20^\circ$، والوتر
$6$ م):
\[
\cos 20^\circ = \frac{d}{6}
\quad\Longrightarrow\quad
d = 6 \cos 20^\circ \approx 6 \times 0.940 \approx 5.6 \text{ م}.
\]
\end{example}

\begin{example}[إيجاد زاوية]\label{ex:g8:cosine:angle}
في \omterm{def:g6:shapes:triangles}{مثلث قائم الزاوية}، يقيس الضلع المجاور للزاوية $\theta$
$3.5$ والوتر $5$:
\[
\cos\theta = \frac{3.5}{5} = 0.7,
\qquad
\theta = \cos^{-1}(0.7) \approx 45.6^\circ .
\]
\end{example}

\begin{exercise}[$\star$]\label{exo:g8:cosine:2}
اُرسم دائرة قطرها $8$ سم مع \omterm{def:g4:geometry:circle}{قطر} $[BC]$، واِختر نقطة

\end{exercise}

  \item والضلع المجاور $6$، والزاوية $50^\circ$: فما الوتر؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:cosine:5}

\begin{exercise}[$\star$]\label{exo:g8:cosine:6}

\section{مسألة: علاقات إقليدس، وفيثاغورس من
جديد}

\begin{problem}[{مسألة نهاية الأسبوع --- اِرتفاع المثلث القائم
الزاوية: $AB^2 = BH \times BC$ و $AH^2 = BH \times HC$، وبرهان ثانٍ
لفيثاغورس، وآلة لإنشاء الجذور التربيعية}]
\label{pb:g8:cosine:1}
أنزِل، من \omterm{def:g3:shapes:rightangle}{الزاوية القائمة} في \omterm{def:g6:shapes:triangles}{مثلث قائم الزاوية}، العمودَ
على الوتر: فهذه \omterm{def:g6:lines:objects}{القطعة} القصيرة --- وهي \emph{الارتفاع} ---
تحقّق علاقات من النفع بحيث وضعها إقليدس في قلب
\emph{أصوله}. وفي هذه المسألة تبرهن خاصّية جيب التمام
المميِّزة، وهي «أن النسبة تتوقف على الزاوية وحدها»
(\cref{def:g8:cosine:def})، على هذه العلاقات كلها؛ وستعيد في الطريق
البرهنة على مبرهنة فيثاغورس بطريق مختلف تمامًا،
وتنتهي بآلة بالمسطرة والفرجار تنشئ
$\sqrt2$ و $\sqrt5$ و $\sqrt{n}$ من أجل كل عدد صحيح $n$.

وفي كل ما يلي، $ABC$ \omterm{def:g6:shapes:triangles}{مثلث قائم الزاوية} في $A$، و $H$
قدم العمود النازل من $A$ على الوتر $[BC]$،
بحيث تقع $H$ بين $B$ و $C$ ويكون $AH \perp BC$.

\medskip
\noindent\textbf{الجزء الأول --- زاوية واحدة، ومثلثان.}
\begin{enumerate}
  \item باستعمال \omterm{def:g1:addition:def}{مجموع} زوايا المثلث
        (\cref{thm:g7:triangles:sum}) في $ABH$ وفي $ABC$، بيّن
        أن $\widehat{BAH} = \widehat{ACB}$: فالارتفاع يقطع
        \omterm{def:g6:shapes:triangles}{المثلث القائم الزاوية} إلى مثلثين أصغر يحملان
        \emph{الزوايا نفسها} التي للأصل.
  \item المثلثان $ABC$ (\omterm{def:g6:shapes:triangles}{القائم الزاوية} في $A$) و $ABH$
        (\omterm{def:g6:shapes:triangles}{القائم الزاوية} في $H$) يشتركان في الزاوية $\widehat B$. اُكتب
        $\cos\widehat B$ في كل منهما، واِستنتج من
        \cref{def:g8:cosine:def} أن
        \[
        AB^2 = BH \times BC .
        \]
        (وللانتقال من تساوي النسبتين إلى تساوي الجداءين، اِضرب
        الطرفين في المقامين معًا،
        \cref{thm:g8:equations:balance}.)
  \item اُذكر العلاقة التوأم عند الرأس $C$ وبَرهِن عليها:
        \[
        AC^2 = CH \times CB .
        \]
  \item خذ المثلث $3$--$4$--$5$: $AB = 3$، و $AC = 4$، و
        $BC = 5$. اِحسب $BH$ و $CH$ من السؤالين 2 و3،
        وتحقّق من أن $BH + HC = BC$.
  \item وعمومًا، عبّر عن $BH$ و $CH$ بكسور لا تتضمّن
        إلا الأضلاع الثلاثة. وبأيّ معنى تشقّ القدم $H$
        الوترَ «\omterm{def:g7:prop:table}{بتناسب} مع مربعَي
        ضلعَي القائمة»؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- فيثاغورس من جديد، والارتفاع.}
\begin{enumerate}[resume]
  \item اِجمع علاقتَي السؤالين 2 و3 واِستعمل
        $BH + HC = BC$ لتحصل على
        \[
        AB^2 + AC^2 = BC^2 :
        \]
        أي مبرهنة فيثاغورس (\cref{thm:g8:pythagoras:direct})،
        مبرهنًا عليها من جديد --- بلا أيّ أحجية مساحات (قارن
        \cref{exo:g8:pythagoras:11}).
  \item طبّق فيثاغورس في المثلث الصغير $ABH$ لتكتب
        $AH^2 = AB^2 - BH^2$، وعوّض $AB^2$ بالمقدار
        $BH \times BC$، وعمّل $BH$ لتبرهن على
        \emph{علاقة اِرتفاع إقليدس}:
        \[
        AH^2 = BH \times HC .
        \]
  \item اِحسب \omterm{def:g6:measure:area}{مساحة} $ABC$ بطريقتين --- بضلعَي القائمة قاعدةً
        واِرتفاعًا، ثم بالوتر قاعدةً
        (\cref{thm:g7:areas:triangle}) --- واِستنتج العلاقة
        الثالثة:
        \[
        AB \times AC = BC \times AH .
        \]
  \item عودًا إلى المثلث $3$--$4$--$5$: اِحسب $AH$
        بالسؤال 8، ثم تحقّق من علاقة اِرتفاع السؤال
        7 عدديًّا، مستعملًا قيمتَي $BH$ و $CH$ الموجودتين في
        السؤال 4.
  \item قدم اِرتفاع تشقّ وتر \omterm{def:g6:shapes:triangles}{مثلث قائم الزاوية} ما
        إلى \omterm{def:g6:lines:objects}{قطعتين} $BH = 2$ سم و $HC = 8$ سم.
        اِحسب الارتفاع $AH$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- آلة للجذور التربيعية.}
اُرسم \omterm{def:g6:lines:objects}{قطعة} $[BC]$ مكوّنة من \omterm{def:g6:lines:objects}{قطعتين} موضوعتين طرفًا لطرف:
$BH = p$ و $HC = q$. واُرسم نصف الدائرة التي قطرها $[BC]$،
ولتكن $A$ النقطة التي يلتقيها فيها العمود على $(BC)$ في $H$.
\begin{enumerate}[resume]
  \item باستعمال \cref{thm:g8:cosine:circle}، اِشرح لماذا يكون
        المثلث $ABC$ \omterm{def:g6:shapes:triangles}{قائم الزاوية} في $A$ --- فينطبق السؤال 7
        ويكون
        \[
        AH^2 = p \times q .
        \]
        (ويُسمّى الطول $AH$ \emph{المتوسط
        الهندسي}\index{متوسط هندسي} للعددين $p$ و $q$.)
  \item خذ $p = 1$ و $q = 2$. فما $AH^2$؟ وأيّ طول
        مشهور من \cref{ex:g8:pythagoras:diagonal} أنشأه
        شكلك بالمسطرة والفرجار للتوّ؟
  \item صِف الوصفة التي تنشئ \omterm{def:g6:lines:objects}{قطعة} طولها
        $\sqrt n$ من أجل أيّ عدد صحيح $n \geq 1$، وقل أيّ
        اِختيار للعددين $p$ و $q$ ينشئ $\sqrt 5$.
  \item ليكن $O$ مركز نصف الدائرة. اِشرح لماذا لا يمكن
        للطول $AH$ أن يتجاوز \omterm{def:g3:shapes:shapes}{نصف القطر} $\frac{p + q}{2}$، و
        من أجل أيّ موضع للنقطة $H$ يتساويان.
  \item اِستنتج المتراجحة: من أجل كل عددين موجبين $p$ و
        $q$،
        \[
        p \times q \leq \left(\frac{p + q}{2}\right)^2,
        \]
        مع التساوي بالضبط حين $p = q$. ثم أعِد البرهنة عليها
        بلا هندسة البتة: اُنشر
        $\left(\frac{p+q}{2}\right)^2 - p q$ وتعرّف على
        متطابقة شهيرة (\cref{pb:g8:equations:1}).
\end{enumerate}
\end{problem}
