\chapter{ऋण संख्याओं का गुणा}\label{ch:g8:negprod}

पिछले साल हमने \omterm{def:g7:negatives:def}{सापेक्ष संख्याएँ} जोड़ी और घटाई थीं
(\cref{ch:g7:negatives}); अब उन्हें गुणा और भाग करना बाक़ी है। पूरे
अध्याय पर एक ही सवाल छाया रहता है: ``ऋण गुना ऋण'' धन क्यों होना
चाहिए? हम सिर्फ़ नियम नहीं, कारण बताएँगे।

\section{चिह्न के नियम}

\begin{theorem}[गुणनफल का चिह्न]\label{thm:g8:negprod:rules}
दो \omterm{def:g7:negatives:def}{सापेक्ष संख्याओं} के \omterm{def:g2:mult:def}{गुणनफल} की \omterm{def:g1:counting:zero}{शून्य} से दूरी उनकी दूरियों के
\omterm{def:g2:mult:def}{गुणनफल} के बराबर होती है, और उसका चिह्न इस सारणी से मिलता है:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|cc}
$\times$ & $+$ & $-$ \\
\hline
$+$ & $+$ & $-$ \\
$-$ & $-$ & $+$ \\
\end{tabular}
\end{center}
\emph{एक जैसे चिह्न: धन \omterm{def:g2:mult:def}{गुणनफल}। उलटे चिह्न: ऋण \omterm{def:g2:mult:def}{गुणनफल}।}
यही सारणी भागफलों पर भी चलती है।
\end{theorem}

\begin{proof}[$(-1) \times (-1) = +1$ क्यों]
पहले, $3 \times (-4)$ का मतलब है $(-4) + (-4) + (-4) = -12$: धन गुना
ऋण, ऋण होता है। अब यह ढर्रा देखो:
\[
3 \times (-4) = -12, \quad
2 \times (-4) = -8, \quad
1 \times (-4) = -4, \quad
0 \times (-4) = 0 .
\]
हर क़दम पर पहला गुणक $1$ घटता है और उत्तर $4$ \emph{बढ़ता} है। इसी
ढर्रे को एक क़दम और आगे बढ़ाओ:
\[
(-1) \times (-4) = 0 + 4 = +4 .
\]
कोई भी दूसरा चुनाव अंकगणित की इस नियमितता (वितरण नियम) को तोड़ देगा।
इसलिए ऋण गुना ऋण धन ही होना चाहिए।
\end{proof}

\begin{tikzpicture}
\begin{axis}[omaxis,

  xmin=-2.6, xmax=3.6, ymin=-14, ymax=10,

  xtick={-2,-1,1,2,3}, ytick={-12,-8,-4,4,8}]
  \addplot[omDef, thick, domain=-2.3:3.3] {-4*x};
  \addplot[omDef, only marks, mark=*, mark size=1.6pt] coordinates
    {(3,-12) (2,-8) (1,-4) (0,0) (-1,4) (-2,8)};
\end{axis}
\end{tikzpicture}

{\small एक चित्र में ढर्रे वाला तर्क: बिंदु $(k,\ k \times (-4))$ एक
ही रेखा पर बैठते हैं, और उस रेखा को $k = 0$ के आगे बढ़ाने पर
$(-1)\times(-4) = +4$ तथा $(-2)\times(-4) = +8$ मानना ही पड़ता है।}
\end{omfigure}

\begin{example}\label{ex:g8:negprod:products}
\begin{align*}
(-7) \times (+6) &= -42 && \text{(उलटे चिह्न)} \\
(-5) \times (-8) &= +40 && \text{(एक जैसे चिह्न)} \\
(+2.5) \times (-4) &= -10 \\
(-36) \div (-9) &= +4 \\
(+15) \div (-2) &= -7.5 .
\end{align*}
\end{example}

\section{कई गुणकों के गुणनफल}

\begin{proposition}[लंबे गुणनफल का चिह्न]\label{prop:g8:negprod:many}
कई शून्येतर गुणकों के \omterm{def:g2:mult:def}{गुणनफल} का चिह्न सिर्फ़ इस पर टिका है कि ऋण
गुणक कितने हैं:
\begin{itemize}
  \item ऋण गुणकों की \emph{सम} संख्या धन \omterm{def:g2:mult:def}{गुणनफल} देती है;
  \item \emph{विषम} संख्या ऋण \omterm{def:g2:mult:def}{गुणनफल} देती है।
\end{itemize}
\end{proposition}

\begin{proof}
ऋण गुणक जोड़ों में बँट जाते हैं, और प्रत्येक जोड़ा धन चिह्न देता है
(\cref{thm:g8:negprod:rules}); विषम \omterm{def:g7:stats:frequency}{गिनती} में एक ऋण गुणक बिना
जोड़ीदार रह जाता है, जो पूरे \omterm{def:g2:mult:def}{गुणनफल} को ऋण बना देता है।
\end{proof}

\begin{example}\label{ex:g8:negprod:many}
$(-2) \times (+3) \times (-5) \times (-1)$: तीन ऋण गुणक (विषम),
इसलिए \omterm{def:g2:mult:def}{गुणनफल} ऋण है; दूरियाँ: $2 \times 3 \times 5 \times 1 = 30$।
उत्तर: $-30$। \emph{पहले} चिह्न तय करो, फिर दूरियाँ गुणा करो --- एक
ग़लती-भरे काम की जगह दो आसान काम।
\end{example}

\begin{example}[ऋण संख्याओं की घातें]\label{ex:g8:negprod:powers}
$(-2)^2 = (-2) \times (-2) = +4$, और
$(-2)^3 = (-2)^2 \times (-2) = -8$: सम घातांक धन मान देते हैं, और
विषम घातांक चिह्न बनाए रखते हैं। लिखावट से सावधान:
$(-3)^2 = 9$ है पर $-3^2 = -(3 \times 3) = -9$।
\end{example}

\section{चारों संक्रियाओं के साथ गणना}

\begin{method}[सापेक्ष संख्याओं के साथ पक्की गणना]\label{met:g8:negprod:compute}
\begin{enumerate}
  \item \cref{ch:g7:priorities} की प्राथमिकताएँ मानो: पहले कोष्ठक,
        फिर $\times$ और $\div$, फिर $+$ और $-$;
  \item प्रत्येक गुणा या भाग पर पहले \emph{चिह्न} निकालो, फिर दूरी;
  \item एक पंक्ति में एक ही क़दम लिखो।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:negprod:compute}
\begin{align*}
5 - 3 \times (-4)
&= 5 - (-12) && \text{(पहले गुणनफल: उलटे चिह्न)}\\
&= 5 + 12 = 17 .
\end{align*}
\begin{align*}
\frac{(-6) \times (-10)}{-4}
&= \frac{+60}{-4} && \text{(अंश: एक जैसे चिह्न)}\\
&= -15 && \text{(भागफल: उलटे चिह्न)।}
\end{align*}
\end{example}

\begin{example}[ऋण मान रखना]\label{ex:g8:negprod:substitute}
$x = -5$ पर $E = 3x^2 - 2x$ का मान निकालो, मान के चारों ओर कोष्ठक लगाकर:
\[
E = 3 \times (-5)^2 - 2 \times (-5)
= 3 \times 25 - (-10)
= 75 + 10 = 85 .
\]
$(-5)^2$ के कोष्ठक ज़रूरी हैं: उनके बिना वर्ग सिर्फ़ $5$ पर लगता।
\end{example}

\]
और नतीजा निकालो कि $(-1) \times (-1)$ को $+1$ ही होना पड़ेगा।
\end{exercise}

\section{समस्या: ऋण गुना ऋण धन क्यों होता है}

\begin{problem}[{सप्ताहांत समस्या --- वितरण नियम से निकाले गए चिह्न
के नियम, और बारी-बारी चिह्न वाला योग $1 - 2 + 3 - 4 + \dots$}]
\label{pb:g8:negprod:1}
\cref{thm:g8:negprod:rules} के प्रमाण ने एक ढर्रा आगे बढ़ाया था और
कहा था कि ``कोई भी दूसरा चुनाव वितरण नियम तोड़ देगा''। यह समस्या उस
दावे को ठीक-ठीक कर देती है: सिर्फ़ वितरण नियम से शुरू करके
(\cref{thm:g7:priorities:distributivity}) तुम चिह्न के नियम
\emph{सिद्ध} करोगे --- न कोई चित्र, न कोई ढर्रा, जैसे बीजगणित करता
है --- और फिर उन्हें $-1$ की घातों पर तथा बारी-बारी चिह्न वाले एक
मशहूर योग पर काम में लाओगे। पूरे समय $a$ और $b$ \omterm{def:g7:negatives:def}{सापेक्ष संख्याएँ}
हैं; हम \cref{ch:g7:negatives} के जोड़ने के नियम, $1 \times a = a$,
और यह मान लेते हैं कि \omterm{def:g2:mult:def}{गुणनफल} किसी भी क्रम में निकाला जा सकता है
(इसलिए वितरण नियम \omterm{def:g2:mult:def}{गुणनफल} की किसी भी तरफ़ लगता है)।

\medskip
\noindent\textbf{भाग I --- चिह्न के नियम प्रमेय हैं।}
\begin{enumerate}
  \item सिद्ध करो कि प्रत्येक \omterm{def:g7:negatives:def}{सापेक्ष संख्या} $a$ के लिए
        $0 \times a = 0$ होता है। ($0 = 0 + 0$ लिखो,
        $(0 + 0) \times a$ फैलाओ, और पूछो कि $0 \times a$ में
        कौन-सी संख्या जोड़ने पर $0 \times a$ ही वापस मिलता है।)
  \item $\bigl(1 + (-1)\bigr) \times a$ फैलाकर दिखाओ कि
        $(-1) \times a + a = 0$, और नतीजा निकालो कि
        \[
        (-1) \times a = -a :
        \]
        $-1$ से गुणा करने पर \omterm{def:g7:negatives:def}{विपरीत संख्या} मिलती है।
        \cref{exo:g8:negprod:11} से तुलना करो, जो $a = -1$ वाली
        स्थिति है।
  \item इससे निकालो कि $(-a) \times b = -(a \times b)$: एक ऋण चिह्न
        \omterm{def:g2:mult:def}{गुणनफल} में से जस का तस बाहर आ जाता है।
  \item इससे निकालो कि $(-a) \times (-b) = a \times b$: दो ऋण चिह्न
        आपस में कट जाते हैं।
  \item प्रत्येक \omterm{def:g7:negatives:def}{सापेक्ष संख्या} \omterm{def:g1:counting:zero}{शून्य} से अपनी दूरी है, आगे एक चिह्न
        लगा हुआ: $a = +d$ या $a = -d$। समझाओ कि सवाल 3 और~4 मिलकर
        \cref{thm:g8:negprod:rules} की चिह्न-सारणी के चारों मामले
        कैसे सिद्ध कर देते हैं, दूरियों समेत।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- $-1$ की घातें।}
\begin{enumerate}[resume]
  \item $(-1)^2$, $(-1)^3$, $(-1)^4$, $(-1)^5$ निकालो, फिर
        \cref{prop:g8:negprod:many} से कारण देते हुए प्रत्येक पूर्ण
        संख्या $n \geq 1$ के लिए $(-1)^n$ का मान बताओ।
  \item बिना कोई दूरी निकाले
        \[
        (-1) \times (-2) \times (-3) \times \dots \times (-10),
        \]
        का चिह्न बताओ, और इस \omterm{def:g2:mult:def}{गुणनफल} की दूरी को पूर्ण संख्याओं के
        \omterm{def:g2:mult:def}{गुणनफल} के रूप में लिखो, हल किए बिना।
  \item दिखाओ कि $(-a)^2 = a^2$ और $(-a)^3 = -a^3$। किन घातांकों पर
        ऋण चिह्न बचा रहता है?
  \item चिह्न के नियमों से दिखाओ कि किसी \omterm{def:g7:negatives:def}{सापेक्ष संख्या} का वर्ग कभी
        ऋण नहीं होता, और इससे निकालो कि कोई भी \omterm{def:g7:negatives:def}{सापेक्ष संख्या} $x$
        $x^2 = -4$ नहीं मानती।
  \item ज़ोए का दावा है: ``$(-1)^{2026} + (-1)^{2027} = 0$, और आम
        तौर पर $-1$ की दो लगातार घातें हमेशा एक-दूसरे को काट देती
        हैं।'' क्या वह सही है? कारण बताओ।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- बारी-बारी चिह्न वाला योग।}
चिह्न के नियम पक्के हो जाने पर हम ऐसे योग निकाल सकते हैं जिनमें
दोनों चिह्न मिले हों। किसी पूर्ण संख्या $n \geq 1$ के लिए मान लो
\[
A_n = 1 - 2 + 3 - 4 + \dots
\]
$1$ से $n$ तक की पूर्ण संख्याओं का, बारी-बारी चिह्न लगाकर बना, योग
है: आख़िरी पद $+n$ होता है जब $n$ विषम हो, और $-n$ जब $n$ सम हो।
तो $A_1 = 1$ और $A_2 = 1 - 2 = -1$।
\begin{enumerate}[resume]
  \item $A_3$, $A_4$, $A_5$, $A_6$ और $A_7$ निकालो। तुम्हारी अटकल
        क्या है?
  \item मान लो $n$ सम है। पदों को दो-दो करके जोड़ो,
        $(1 - 2) + (3 - 4) + \dots$, जोड़े गिनो, और सिद्ध करो कि
        $A_n = -\frac{n}{2}$।
  \item मान लो $n$ विषम है। समझाओ कि $A_n = A_{n-1} + n$ क्यों है,
        और पिछले सवाल से निकालो कि $A_n = \frac{n + 1}{2}$।
  \item $A_{100}$, $A_{101}$ और $A_{2026}$ निकालो।
  \item आल्मा ने $A_{101} - A_{100} = 51 - (-50) = 101$ निकाला और
        ठीक $101$ पाकर हैरान रह गई। क्या उसे हैरान होना चाहिए था?
        एक वाक्य में समझाओ, और बिना और कोई गणना किए
        $A_{2027} - A_{2026}$ का मान बताओ।
\end{enumerate}
\end{problem}
