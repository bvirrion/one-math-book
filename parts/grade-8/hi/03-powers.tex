\chapter{घातें}\label{ch:g8:powers}

काग़ज़ की एक शीट को $10$ बार आधा-आधा मोड़ो (अगर मोड़ सको!): उसकी
मोटाई हर बार दुगुनी होती है, और $10$ बार दुगुना करने से वह
$2^{10} = 1024$ गुनी हो जाती है। घातें बार-बार किए गए गुणा की
छोटी लिखावट हैं; यह अध्याय वह लिखावट और उसके नियम बनाता है, और $10$
की घातों को ख़ास जगह देता है।

\section{परिभाषा}

\begin{definition}[घात]\label{def:g8:powers:def}
किसी संख्या $a$ और किसी पूर्ण संख्या $n \geq 1$ के लिए:
\[
a^n = \underbrace{a \times a \times \dots \times a}_{n
\text{ गुणक}},
\]
जिसे ``$a$ की घात $n$'' पढ़ते हैं; $a$ \emph{आधार} है और $n$
\emph{घातांक}\index{घातांक}। ख़ास नाम: $a^2$ यानी ``$a$ का वर्ग'',
$a^3$ यानी ``$a$ का \omterm{def:g5:solids:def}{घन}''। और यह तय किया गया है:
\[
a^1 = a,
\qquad
a^0 = 1 \ (a \neq 0),
\qquad
a^{-n} = \frac{1}{a^n}.
\]
\end{definition}

\begin{example}\label{ex:g8:powers:def}
$3^4 = 3 \times 3 \times 3 \times 3 = 81$; \quad
$10^3 = 1000$; \quad $5^{-2} = \frac{1}{25} = 0.04$; \quad
$(-2)^3 = -8$ और $(-2)^4 = +16$
(\cref{ex:g8:negprod:powers})। सावधान: $a^n$, $a \times n$
\emph{नहीं} है: \ $2^5 = 32$ होता है, $10$ नहीं।
\end{example}

भागती है।}
\end{omfigure}

\section{घातांक के नियम}

\begin{theorem}[घातांक के नियम]\label{thm:g8:powers:rules}
किसी शून्येतर आधार $a$ और पूर्ण-संख्या घातांकों $m$, $n$ के लिए:
\[
a^m \times a^n = a^{m+n},
\qquad
\frac{a^m}{a^n} = a^{m-n},
\qquad
\left(a^m\right)^n = a^{m \times n},
\qquad
(ab)^n = a^n b^n .
\]
\end{theorem}

\begin{proof}[गुणक गिनकर प्रमाण]
$a^m \times a^n$ में $a$ के $m$ गुणकों के बाद $n$ गुणक और लग जाते
हैं: कुल $m + n$। $\left(a^m\right)^n$ में $m$ गुणकों का एक गुच्छा
$n$ बार दोहराया जाता है: $mn$ गुणक। $(ab)^n$ में $a$ के $n$ अक्षर
और $b$ के $n$ अक्षर होते हैं, जिन्हें नए सिरे से जमाया जा सकता है।
\omterm{def:g3:division:remainder}{भागफल} वाला नियम $m$ में से $n$ गुणक काट देने से आता है; और
$a^0 = 1$ तथा $a^{-n} = \frac{1}{a^n}$ तय कर लेने पर वह $n \geq m$
होने पर भी सही रहता है।
\end{proof}

\begin{example}\label{ex:g8:powers:rules}
\[
2^3 \times 2^5 = 2^8 = 256,
\qquad
\frac{7^6}{7^4} = 7^2 = 49,
\qquad
\left(5^2\right)^3 = 5^6,
\qquad
2^4 \times 5^4 = 10^4 .
\]
एक जाल: ये नियम \emph{एक ही आधार} पर लगते हैं (या आख़िरी नियम के लिए
एक ही \omterm{def:g8:powers:def}{घातांक} पर)। $2^3 \times 5^2$ को कोई नियम सरल नहीं करता ---
बस निकाल डालो: $8 \times 25 = 200$।
\end{example}

\section{दस की घातें}

\begin{proposition}[दस की घातें]\label{prop:g8:powers:ten}
$n \geq 1$ के लिए: $10^n = 1\underbrace{0\dots0}_{n}$, और
$10^{-n} = 0.\underbrace{0\dots0}_{n-1}1$। \omterm{def:g8:powers:def}{घातांक} के नियम यहाँ ऐसे
पढ़े जाते हैं: दस की घातें गुणा करने पर \omterm{def:g8:powers:def}{घातांक} जुड़ जाते हैं।
\end{proposition}

\begin{example}\label{ex:g8:powers:ten}
$10^4 \times 10^3 = 10^7$; \quad
$\dfrac{10^2}{10^5} = 10^{-3} = 0.001$। बड़ी और छोटी राशियाँ पढ़ने
लायक़ हो जाती हैं:
\[
\text{एक बिलियन} = 10^9,
\qquad
\text{एक मिलियनवाँ भाग} = 10^{-6}.
\]
दशमलव के साथ मिलाकर: $3.2 \times 10^5 = 320\,000$ और
$4.7 \times 10^{-3} = 0.0047$ (\omterm{def:g4:decimals:point}{दशमलव बिंदु} सरका दो,
\cref{prop:g6:decimals:shift})। इस लिखावट का बाक़ायदा इस्तेमाल ---
\emph{वैज्ञानिक लेखन} --- \cref{ch:g9:fractions} में आता है।
\end{example}

\begin{example}[कोटि-क्रम]\label{ex:g8:powers:magnitude}
प्रकाश क़रीब $3 \times 10^8$ m/s चलता है; और एक साल में क़रीब
$3.2 \times 10^7$ सेकंड होते हैं। इसलिए एक प्रकाश-वर्ष क़रीब
\[
3 \times 3.2 \times 10^{8+7} \approx 10 \times 10^{15} = 10^{16}
\text{ m}
\]
--- यानी दस मिलियन बिलियन मीटर। दस की घातें खगोल की गणनाओं को एक ही
पंक्ति में समेट देती हैं।
\end{example}

\begin{method}[घातों वाला व्यंजक सरल करना]\label{met:g8:powers:simplify}
\begin{enumerate}
  \item गुणकों को आधार के हिसाब से गुच्छों में बाँटो;
  \item प्रत्येक आधार के भीतर \omterm{def:g8:powers:def}{घातांक} के नियम लगाओ;
  \item बची हुई छोटी घातें निकाल डालो, या उत्तर बड़ा हो तो उसे घात
        के रूप में ही रहने दो।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:powers:simplify}
\[
\frac{3^5 \times 3^{-2} \times 4^2}{3^2}
= \frac{3^{5 + (-2)}}{3^2} \times 16
= 3^{3 - 2} \times 16
= 3 \times 16 = 48 .
\]
\end{example}

\begin{exercise}[$\star$]\label{exo:g8:powers:1}

2^6, \qquad 3^3, \qquad 10^5, \qquad 1^{100}, \qquad 0.1^2, \qquad
6^0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:2}
गणना करो:
\[
(-3)^2, \qquad -3^2, \qquad (-1)^{15}, \qquad (-5)^3, \qquad
\left(\tfrac{2}{3}\right)^2 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:3}
एक ही घात के रूप में लिखो:
\[
7^4 \times 7^5, \qquad
\frac{2^9}{2^3}, \qquad
\left(10^3\right)^4, \qquad
5^6 \times 5^{-2}, \qquad
3^4 \times 7^4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:4}
\omterm{def:g6:decimals:places}{दशमलव संख्या} के रूप में लिखो: $10^{-2}$; \quad $4 \times 10^3$;
\quad $2.5 \times 10^{-4}$; \quad $10^0$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:5}
दस की घात के साथ लिखो: एक लाख; एक दसवाँ भाग; दस बिलियन;
$0.000\,001$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:6}
सरल करो, फिर गणना करो:
\[
\frac{10^7 \times 10^{-3}}{10^2},
\qquad
\frac{2^5 \times 2^4}{2^6},
\qquad
\left(2^2\right)^3 \times 2^{-4} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:powers:7}
सही या ग़लत? जो ग़लत हों उन्हें ठीक करो।
\[
2^3 \times 2^4 = 2^{12}; \qquad
5^2 + 5^3 = 5^5; \qquad
(3^2)^4 = 3^8; \qquad
10^3 \times 10^3 = 100^3 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:powers:8}
एक अफ़वाह फैलती है: पहले दिन तीन लोग उसे जानते हैं; और हर दिन,
जानने वाला हर आदमी तीन नए लोगों को बताता है। चौथे दिन जिन्हें
\emph{नई} ख़बर मिली उनकी संख्या घात के साथ लिखो, और निकालो कि चौथे
दिन के अंत तक कितने लोग अफ़वाह जानते हैं (शुरुआती तीन समेत)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:powers:9}
काग़ज़ की एक शीट $0.1$ mm मोटी है, यानी $10^{-4}$ m। उसे मोड़ने पर
मोटाई हर बार दुगुनी हो जाती है।
\begin{enumerate}
  \item $10$ बार मोड़ने के बाद की मोटाई एक \omterm{def:g2:mult:def}{गुणनफल} के रूप में लिखो,
        और उसे सेंटीमीटर में निकालो ($2^{10} = 1024$)।
  \item $42$ बार मोड़ने के बाद मोटाई $2^{42} \times 10^{-4}$ m होगी,
        जहाँ $2^{42} \approx 4.4 \times 10^{12}$। दिखाओ कि यह
        पृथ्वी--चंद्रमा की दूरी, क़रीब $3.8 \times 10^8$ m, से भी
        ज़्यादा है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:powers:10}
बिना कैलकुलेटर के छोटे से बड़े के क्रम में लगाओ:
\[
2^{10}, \qquad 10^3, \qquad 3^6, \qquad 5^4 .
\]
(हर एक निकालो; $2^{10}$ और $10^3$ मशहूर पड़ोसी हैं।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:powers:11}
कौन बड़ा है, $2^{100}$ या $10^{30}$?
$2^{100} = \left(2^{10}\right)^{10}$ की तुलना
$\left(10^3\right)^{10}$ से करने के लिए $2^{10} = 1024 > 10^3$
इस्तेमाल करो।
\end{exercise}

\section{समस्या: शतरंज की बिसात और दो की घातें}

\begin{problem}[{सप्ताहांत समस्या --- गुणोत्तर योग
$1 + 2 + 4 + \dots + 2^{n-1} = 2^n - 1$, एक मशहूर किंवदंती से
द्विआधारी संख्याओं तक}]
\label{pb:g8:powers:1}
किंवदंती यह है: शतरंज बनाने के इनाम में सिस्सा नाम के विद्वान ने
अपने राजा से बिसात के पहले ख़ाने पर गेहूँ का एक दाना माँगा, दूसरे पर
दो, तीसरे पर चार --- ख़ाने-दर-ख़ाने दुगुना करते हुए, चौंसठवें तक।
राजा उसकी इतनी छोटी माँग पर हँस पड़ा। यह समस्या
\cref{thm:g8:powers:rules} के घातांक-नियमों से निकालती है कि राजा ने
वादा क्या किया था, और वहीं ख़त्म होती है जहाँ यह कहानी चुपके से ले
जाती है: हर कंप्यूटर के भीतर बैठी द्विआधारी संख्याएँ।

\medskip
\noindent\textbf{भाग I --- दुगुना करने वाली तरकीब।}
$n \geq 1$ के लिए मान लो $S_n$ पहले $n$ ख़ानों पर पड़े दानों की कुल
संख्या है।
\begin{enumerate}
  \item ख़ाने $k$ पर पड़े दानों की संख्या $2$ की घात के रूप में
        लिखो। ख़ाने $64$ पर कौन-सी घात बैठती है?
  \item $S_1$, $S_2$, $S_3$, $S_4$ और $S_5$ निकालो, और हर एक की
        तुलना $2$ की पास वाली घात से करो। $S_n$ का सूत्र अटकलो।
  \item \emph{दुगुना करने वाली तरकीब}: योग $S_n$ और
        $2 \times S_n$ एक के नीचे एक लिखो, घटाओ, और अपनी अटकल
        सिद्ध करो:
        \[
        S_n = 1 + 2 + 4 + \dots + 2^{n-1} = 2^n - 1 .
        \]
  \item राजा ने कुल कितने दानों का वादा किया? उत्तर $2$ की घात के
        साथ लिखो, और यह जाना-पहचाना वाक्य पूरा करो: ``पूरी बिसात पर
        उतने दाने हैं जितने अकेले एक पैंसठवें ख़ाने पर होते, बस एक
        कम।''
  \item दिखाओ कि बिसात के दूसरे आधे हिस्से (ख़ाने $33$ से $64$) पर
        पहले आधे हिस्से से \emph{ठीक} $2^{32}$ गुने दाने हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- $2^{64}$ कितना बड़ा है?}
\cref{exo:g8:powers:11} वाली तुलना $2^{10} = 1024 > 10^3$ नीचे के
सारे अंदाज़ों की चाबी है।
\begin{enumerate}[resume]
  \item दिखाओ कि
        $2^{64} = 2^4 \times \left(2^{10}\right)^6 > 1.6 \times
        10^{19}$।
  \item गेहूँ के एक दाने का वज़न क़रीब $0.05$ g होता है, यानी
        $5 \times 10^{-2}$ g। दिखाओ कि वादा किए गए गेहूँ का वज़न
        $8 \times 10^{17}$ g से ज़्यादा है, और उसे टन में बदलो
        ($1$ टन $= 10^6$ g)।
  \item आज पूरी दुनिया साल भर में क़रीब $8 \times 10^8$ टन गेहूँ
        उगाती है। राजा ने कम से कम कितने साल की दुनिया भर की फ़सल
        का वादा कर डाला?
  \item वह सबसे छोटी पूर्ण संख्या $n$ ढूँढ़ो जिसके लिए
        $2^n > 10^6$ हो --- यानी दस लाख पार करने में कितनी बार
        दुगुना करना पड़ता है। ($2^{10} = 1024$ से $2^{19}$ और
        $2^{20}$ ठीक-ठीक निकालो।)
  \item एक ठग तुम्हें एक महीने की तनख़्वाह का लालच देता है: पहले दिन
        $1$ सेंट, फिर हर दिन पिछले दिन की तनख़्वाह का दुगुना। किस
        दिन अकेली \emph{उस दिन की} तनख़्वाह पहली बार दस लाख यूरो
        ($10^8$ सेंट) से ऊपर निकलती है? ($2^{26}$ और $2^{27}$
        ठीक-ठीक निकालो।)
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- द्विआधारी बाट।}
एक व्यापारी के पास पाँच बाट हैं: $1$, $2$, $4$, $8$ और $16$ ग्राम
के, हर एक का एक। वह उनमें से कुछ तराज़ू के एक पलड़े पर रखकर दूसरे
पलड़े पर रखा सामान तोलती है।
\begin{enumerate}[resume]
  \item $21$ g तोलने के लिए वह कौन-से बाट रखेगी? और $27$ g तोलने
        के लिए?
  \item समझाओ कि $16$ g या उससे ज़्यादा के किसी भी सामान में $16$ g
        वाला बाट लगना \emph{ही} पड़ेगा, और $15$ g या उससे कम के
        किसी भी सामान में वह \emph{नहीं} लग सकता। (सवाल 3 बताता है
        कि बाट $1, 2, 4, 8$ ज़्यादा से ज़्यादा कहाँ तक पहुँचते
        हैं।) समझाओ कि हर चरण पर, उससे अगले बड़े बाट के साथ यही तर्क
        दोहरा जाता है।
  \item इससे निकालो कि $1$ से $31$ g तक का हर पूर्ण-संख्या सामान
        तोला जा सकता है, और \emph{ठीक एक ही तरीक़े से}: $1$ और $31$
        के बीच की हर संख्या अलग-अलग $2$ की घातों का योग एक ही ढंग
        से बनती है।
  \item व्यापारी $32$ g का छठा बाट ख़रीद लेती है। अब वह कहाँ तक तोल
        सकती है? $45$ g को अलग-अलग $2$ की घातों के योग के रूप में
        लिखो।
  \item किसी कंप्यूटर में ``64-बिट'' संख्या $64$ ख़ानों पर रखी जाती
        है, और हर ख़ाने में या तो $0$ होता है या $1$ --- ख़ाना $k$,
        $1$ होने पर $2^{k-1}$ का योगदान देता है, ठीक किंवदंती के
        दानों की तरह। भाग I से समझाओ कि ऐसी मशीन जो पूर्ण संख्याएँ
        रख सकती है वे ठीक $0$ से $2^{64} - 1$ तक ही क्यों हैं।
\end{enumerate}
\end{problem}
