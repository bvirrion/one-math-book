\chapter{अक्षर-गणना और समीकरण}\label{ch:g8:equations}

पिछले साल अक्षर आए थे और आसान व्यंजक सरल किए गए थे
(\cref{ch:g7:literal})। यह अध्याय वे दो हुनर जोड़ता है जिनसे
बीजगणित ताक़तवर बनता है: कोष्ठकों के \omterm{def:g2:mult:def}{गुणनफल} \emph{फैलाना}, और
\emph{\omterm{def:g8:equations:def}{समीकरण} हल करना} --- यानी अक्षर के पीछे छिपी संख्या ढूँढ़ना।

\section{फैलाना}

\begin{theorem}[सादा और दुहरा वितरण नियम]\label{thm:g8:equations:expand}
सभी संख्याओं के लिए:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{theorem}

\begin{proof}
पहला नियम \cref{thm:g7:priorities:distributivity} ही है (चिह्नों
समेत: $k$ और पद ऋण भी हो सकते हैं, \cref{ch:g8:negprod})। दूसरे के
लिए, $(c + d)$ को एक ही संख्या मानकर पहला नियम दो बार लगाओ:
\[
(a + b)(c + d) = a(c + d) + b(c + d) = ac + ad + bc + bd .
\qedhere
\]
\end{proof}

\begin{example}\label{ex:g8:equations:expand}
हर पद पर चिह्नों का ध्यान रखो:
\begin{align*}
3(2x - 5) &= 6x - 15, \\
-2(4x - 7) &= -8x + 14
&& \text{(आगे लगा ऋण दोनों चिह्न पलट देता है)}, \\
(x + 3)(x + 5) &= x^2 + 5x + 3x + 15 = x^2 + 8x + 15, \\
(2x - 1)(x - 4) &= 2x^2 - 8x - x + 4 = 2x^2 - 9x + 4 .
\end{align*}
\end{example}

\begin{method}[फैलाओ और सरल करो]\label{met:g8:equations:reduce}
\begin{enumerate}
  \item हर \omterm{def:g2:mult:def}{गुणनफल} फैलाओ, और हर पद उसके चिह्न के साथ लिखो;
  \item समान पद मिलाओ ($x^2$ के साथ $x^2$, $x$ के साथ $x$, संख्या
        के साथ संख्या);
  \item किसी मान से जाँचो: असली व्यंजक में और उत्तर में जैसे
        $x = 2$ रखो --- दोनों बराबर निकलें तो ज़्यादातर ग़लतियाँ
        पकड़ में आ जाती हैं।
\end{enumerate}
\end{method}

\section{समीकरण हल करना}

\begin{definition}[समीकरण]\label{def:g8:equations:def}
\emph{समीकरण}\index{समीकरण} एक ऐसी समानता है जिसमें एक अज्ञात
संख्या हो, जिसे $x$ (या कोई भी अक्षर) लिखा जाता है। उसे \emph{हल
करने} का मतलब है $x$ के वे सारे मान ढूँढ़ना जिन पर समानता सही हो
जाए --- यानी \emph{हल}।
\end{definition}

\begin{theorem}[तुला के नियम]\label{thm:g8:equations:balance}
इन कामों से \omterm{def:g8:equations:def}{समीकरण} के हल बिलकुल वही रहते हैं:
\begin{enumerate}
  \item दोनों तरफ़ एक ही संख्या जोड़ना (या घटाना);
  \item दोनों तरफ़ एक ही \emph{शून्येतर} संख्या से गुणा करना (या
        भाग देना)।
\end{enumerate}
\end{theorem}
\admitted

\begin{method}[प्रथम घात का समीकरण हल करना]\label{met:g8:equations:solve}
\begin{enumerate}
  \item ज़रूरत हो तो दोनों तरफ़ फैलाओ और सरल करो;
  \item दोनों तरफ़ जोड़-घटाकर $x$ वाले पद एक तरफ़ और सादी संख्याएँ
        दूसरी तरफ़ इकट्ठी करो;
  \item सरल करो, फिर दोनों तरफ़ $x$ के गुणांक से भाग दो;
  \item मिला हुआ मान असली \omterm{def:g8:equations:def}{समीकरण} में रखकर \emph{जाँचो}।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:equations:solve}
$7x - 5 = 4x + 16$ हल करो:
\begin{align*}
7x - 5 &= 4x + 16 \\
7x - 4x &= 16 + 5
&& \text{(दोनों तरफ़ $5$ जोड़ो और $4x$ घटाओ)} \\
3x &= 21 \\
x &= 7 && \text{(दोनों तरफ़ $3$ से भाग दो)।}
\end{align*}
जाँच: बाईं तरफ़ $7 \times 7 - 5 = 44$; दाईं तरफ़
$4 \times 7 + 16 = 44$। हल $7$ है।
\end{example}

\begin{example}[कोष्ठक और ऋण संख्याओं के साथ]\label{ex:g8:equations:brackets}
$3(x - 4) = 5 - 2(x + 1)$ हल करो:
\begin{align*}
3x - 12 &= 5 - 2x - 2 && \text{(दोनों तरफ़ फैलाओ)} \\
3x - 12 &= 3 - 2x && \text{(सरल करो)} \\
5x &= 15 && \text{(दोनों तरफ़ $2x + 12$ जोड़ो)} \\
x &= 3 .
\end{align*}
जाँच: $3(3 - 4) = -3$ और $5 - 2(3+1) = 5 - 8 = -3$। \quad हल:
$3$।
\end{example}

\begin{method}[समीकरण से कोई समस्या हल करना]\label{met:g8:equations:problem}
\begin{enumerate}
  \item अज्ञात चुनो: ``मान लो $x$ \dots\ है'' (या तो वही राशि जो
        पूछी गई है, या उससे जुड़ी कोई आसान राशि);
  \item कहानी को \omterm{def:g8:equations:def}{समीकरण} में बदलो;
  \item उसे हल करो;
  \item मतलब निकालो: मूल सवाल का जवाब एक वाक्य में, मात्रक के साथ
        दो, और उत्तर को कहानी से मिलाकर जाँचो।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g8:equations:problem}
लूकास अपनी बहन से $4$ साल बड़ा है; दोनों की उम्र मिलाकर $26$ है।
मान लो बहन की उम्र $x$ है; तब लूकास $x + 4$ का हुआ, और
\[
x + (x + 4) = 26
\ \Longrightarrow\
2x + 4 = 26
\ \Longrightarrow\
2x = 22
\ \Longrightarrow\
x = 11 .
\]
बहन $11$ की है और लूकास $15$ का। जाँच: $11 + 15 = 26$ और
$15 - 11 = 4$।
\end{example}

\section{क्रम और असमिकाएँ}

\begin{proposition}[संक्रियाएँ और क्रम]\label{prop:g8:equations:order}
किसी असमिका के दोनों तरफ़ एक ही संख्या जोड़ने पर वह बनी रहती है;
दोनों तरफ़ किसी \emph{धन} संख्या से गुणा करने पर भी बनी रहती है; पर
किसी \emph{ऋण} संख्या से गुणा करने पर वह \emph{उलट} जाती है:
\[
2 < 5 \ \text{ पर }\ (-1) \times 2 > (-1) \times 5 .
\]
\end{proposition}
\admitted

\begin{example}\label{ex:g8:equations:inequality}
$5 - 2x \leq 11$ हल करो: दोनों तरफ़ से $5$ घटाओ, $-2x \leq 6$; अब
$-2$ से भाग दो \emph{और असमिका उलट दो}: $x \geq -3$। $-3$ से बड़ी
या बराबर सारी संख्याएँ हल हैं --- \omterm{def:g8:equations:def}{समीकरण} का आम तौर पर एक हल होता
है, पर असमिका के हल पूरी आधी रेखा भर होते हैं।
\end{example}

\section{समस्या: उल्लेखनीय सर्वसमिकाएँ और विषम संख्याओं की
सीढ़ी}

\begin{problem}[{सप्ताहांत समस्या --- $(a+b)^2$, $(a-b)^2$,
$(a+b)(a-b)$, और योग $1 + 3 + 5 + \dots + (2n-1) = n^2$}]
\label{pb:g8:equations:1}
तीन फैलाव इतनी बार आते हैं कि बीजगणित उन्हें ज़बानी याद रखता है:
उन्हें \emph{उल्लेखनीय सर्वसमिकाएँ} कहते हैं। यह समस्या उन्हें दुहरे
वितरण नियम (\cref{thm:g8:equations:expand}) से निकालती है, उन्हें
मानसिक गणित की महाशक्ति में बदलती है, उनसे \omterm{def:g3:numbers:evenodd}{विषम संख्याओं} के बारे में
एक सुंदर बात सिद्ध करती है, और आख़िर में बिलकुल नई तरह के \omterm{def:g8:equations:def}{समीकरण} हल
करती है: वे \omterm{def:g8:equations:def}{समीकरण} जिनमें वर्ग आता है।

\medskip
\noindent\textbf{भाग I --- तीन सर्वसमिकाएँ।}
\begin{enumerate}
  \item $(a + b)^2 = (a + b)(a + b)$ को फैलाकर सरल करो, और सिद्ध
        करो कि
        \[
        (a + b)^2 = a^2 + 2ab + b^2 .
        \]
  \item इसी तरह उसकी दोनों साथी सर्वसमिकाएँ सिद्ध करो:
        \[
        (a - b)^2 = a^2 - 2ab + b^2,
        \qquad
        (a + b)(a - b) = a^2 - b^2 .
        \]
  \item भुजा $a + b$ का एक वर्ग बनाओ, हर भुजा को $a$ और $b$ में
        बाँटो, और वर्ग को चार टुकड़ों में काटो। चित्र का हर हिस्सा
        पहली सर्वसमिका के किस टुकड़े को दिखाता है?
  \item मानसिक गणित: सर्वसमिकाओं से, बिना कोई गुणा किए, निकालो
        \[
        101^2, \qquad 99^2, \qquad 49 \times 51 .
        \]
  \item एक जाना-पहचाना जाल: ज़ोए लिखती है
        ``$(a + b)^2 = a^2 + b^2$''। उसके सूत्र को $a = 3$,
        $b = 4$ पर जाँचो, और सवाल 3 के चित्र पर ठीक-ठीक बताओ कि
        उसका सूत्र कौन-से टुकड़े भूल रहा है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- \omterm{def:g3:numbers:evenodd}{विषम संख्याओं} की सीढ़ी।}
\begin{enumerate}[resume]
  \item $1$ निकालो, फिर $1 + 3$, फिर $1 + 3 + 5$, फिर
        $1 + 3 + 5 + 7$, फिर $1 + 3 + 5 + 7 + 9$। उत्तरों में
        तुम्हें क्या दिखता है?
  \item समझाओ कि $n$-वीं \omterm{def:g3:numbers:evenodd}{विषम संख्या} $2n - 1$ क्यों है (जाँच:
        $n = 1, 2, 3$ से $1, 3, 5$ मिलते हैं)।
  \item भाग I की किसी सर्वसमिका से $(n + 1)^2 - n^2$ फैलाकर सरल
        करो। उत्तर का सवाल 7 से क्या नाता है?
  \item सवाल 8 कहता है: भुजा $n$ के वर्ग से भुजा $n + 1$ के वर्ग
        तक जाने में ठीक अगली \omterm{def:g3:numbers:evenodd}{विषम संख्या} जितने इकाई-ख़ाने जोड़ने
        पड़ते हैं --- दो भुजाओं और कोने के साथ बनी एक L-आकार की
        किनार। $1 = 1^2$ से शुरू करके वर्ग को एक-एक परत बढ़ाते हुए
        समझाओ कि
        \[
        1 + 3 + 5 + \dots + (2n - 1) = n^2 :
        \]
        यानी पहली $n$ \omterm{def:g3:numbers:evenodd}{विषम संख्याओं} का \omterm{def:g1:addition:def}{योगफल} $n$-वाँ वर्ग होता है।
  \item बिना एक-एक करके जोड़े निकालो:
        $1 + 3 + 5 + \dots + 99$ का मान; और $1$ से शुरू करके कितनी
        \omterm{def:g3:numbers:evenodd}{विषम संख्याएँ} जुड़कर ठीक $400$ बनाती हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- वर्ग वाले \omterm{def:g8:equations:def}{समीकरण}।}
\begin{enumerate}[resume]
  \item $x^2 + 6x + 9$ में कोई उल्लेखनीय सर्वसमिका पहचानो, और
        \omterm{def:g8:equations:def}{समीकरण} $x^2 + 6x + 9 = 0$ हल करो।
  \item \cref{thm:g8:negprod:rules} के चिह्न-और-दूरी वाले नियम से
        समझाओ कि दो संख्याओं का \omterm{def:g2:mult:def}{गुणनफल} \omterm{def:g1:counting:zero}{शून्य} \emph{तभी} हो सकता है
        जब दोनों में से कोई एक गुणक \omterm{def:g1:counting:zero}{शून्य} हो। इससे
        $(x - 2)(x + 5) = 0$ हल करो।
  \item सब कुछ एक तरफ़ ले जाकर और तीसरी सर्वसमिका से $x^2 - 49$ के
        गुणनखंड बनाकर $x^2 = 49$ हल करो। इस \omterm{def:g8:equations:def}{समीकरण} के कितने हल हैं
        --- और यह इस अध्याय में अब तक हल किए गए हर \omterm{def:g8:equations:def}{समीकरण} से कैसे
        अलग है?
  \item \cref{exo:g8:equations:3} में तुमने जाँचा था कि $x = 3$,
        $x^2 = 6x - 9$ का हल है। दिखाओ कि वही \emph{अकेला} हल है:
        सब कुछ एक तरफ़ लाओ, सर्वसमिका पहचानो, और नतीजा निकालो।
  \item आख़िरी दाँव: $2026^2 - 2025^2$ मन ही मन निकालो, और किसी
        सर्वसमिका से समझाओ कि दो लगातार वर्गों का \omterm{ex:g1:subtraction:difference}{अंतर} हमेशा उन
        दोनों संख्याओं का \omterm{def:g1:addition:def}{योगफल} क्यों होता है --- यानी वही सवाल 8,
        फिर से।
\end{enumerate}
\end{problem}
