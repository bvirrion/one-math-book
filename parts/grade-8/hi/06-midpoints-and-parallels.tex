\chapter{मध्यबिंदु और समांतर रेखाएँ}\label{ch:g8:midpoints}

किसी त्रिभुज की दो भुजाओं के मध्यबिंदु मिला दो: जो \omterm{def:g6:lines:objects}{रेखाखंड} मिलता है
वह हमेशा तीसरी भुजा के \omterm{def:g6:lines:perp}{समांतर} होता है और ठीक उसका \omterm{def:g3:division:half}{आधा} लंबा। यह
``मध्यबिंदु प्रमेय'' और उसका विलोम एक बड़े विचार की पहली झलक हैं ---
\omterm{def:g6:lines:perp}{समांतर रेखाएँ} लंबाइयों को \omterm{def:g7:prop:table}{समानुपात} में काटती हैं --- जो
\cref{ch:g9:thales} में थेल्स प्रमेय बनकर खिल उठता है।

\section{मध्यबिंदु प्रमेय}

\begin{theorem}[मध्यबिंदु प्रमेय]\label{thm:g8:midpoints:direct}
किसी त्रिभुज $ABC$ में मान लो $I$, $[AB]$ का मध्यबिंदु है और $J$,
$[AC]$ का। तब रेखा $(IJ)$, $(BC)$ के \omterm{def:g6:lines:perp}{समांतर} होती है, और
\[
IJ = \frac{BC}{2} .
\]
\end{theorem}

\begin{proof}[प्रमाण का विचार]
मान लो $K$, $J$ के बारे में $I$ का \omterm{def:g7:central:def}{सममित} बिंदु है (यानी \omterm{def:g3:division:half}{आधा} चक्कर,
\cref{ch:g7:central})। \omterm{def:g4:geometry:polygon}{चतुर्भुज} $AICK$ के विकर्ण $[IK]$ और $[AC]$
अपने साझे मध्यबिंदु $J$ पर कटते हैं: इसलिए वह \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} है,
और तब $KC$, $AI$ यानी $(AB)$ के \omterm{def:g6:lines:perp}{समांतर} है, तथा $KC = AI = IB$। अब
$IBCK$ की आमने-सामने की दो भुजाएँ ($[IB]$ और $[KC]$) \omterm{def:g6:lines:perp}{समांतर} तथा
बराबर हैं: इसलिए वह भी \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} है
(\cref{thm:g7:central:recognize}), और तब
$(IK) \parallel (BC)$ --- यानी $(IJ) \parallel (BC)$ --- तथा
$BC = IK = 2\,IJ$।
\end{proof}

\begin{example}\label{ex:g8:midpoints:direct}
जिस त्रिभुज $ABC$ में $BC = 9$ cm है, उसमें $[AB]$ का मध्यबिंदु $I$
और $[AC]$ का मध्यबिंदु $J$ मिलाए गए। बिना कुछ नापे:
$(IJ) \parallel (BC)$ और $IJ = \frac92 = 4.5$ cm। छोटा त्रिभुज
$AIJ$, $ABC$ की आधी नाप वाली नक़ल है --- इसलिए उसका \omterm{def:g6:measure:perimeter}{परिमाप} भी $ABC$
के \omterm{def:g6:measure:perimeter}{परिमाप} का \omterm{def:g3:division:half}{आधा} है।
\end{example}

\begin{theorem}[विलोम]\label{thm:g8:midpoints:converse}
किसी त्रिभुज $ABC$ में, $[AB]$ के मध्यबिंदु $I$ से होकर $(BC)$ के
\emph{\omterm{def:g6:lines:perp}{समांतर}} जाने वाली रेखा भुजा $[AC]$ को उसके मध्यबिंदु पर काटती
है।
\end{theorem}
\admitted

\begin{example}\label{ex:g8:midpoints:converse}
एक पगडंडी किसी त्रिभुजाकार खेत को पार करती है, एक \omterm{def:g5:solids:def}{किनारे} के बीच से
शुरू होकर सामने वाले \omterm{def:g5:solids:def}{किनारे} के \omterm{def:g6:lines:perp}{समांतर} चलती हुई। विलोम की गारंटी है
कि वह पगडंडी दूसरे \omterm{def:g5:solids:def}{किनारे} के ठीक बीच से निकलेगी --- उस तरफ़ कुछ भी
नापने की ज़रूरत नहीं।
\end{example}

\begin{method}[प्रमेय और विलोम में से चुनना]\label{met:g8:midpoints:choose}
\begin{enumerate}
  \item \emph{दो मध्यबिंदु पता हों} $\Rightarrow$ सीधी प्रमेय लगाओ:
        समांतरपन और आधी लंबाई का नतीजा निकालो।
  \item \emph{एक मध्यबिंदु और एक \omterm{def:g6:lines:perp}{समांतर} रेखा पता हो} $\Rightarrow$
        विलोम लगाओ: नतीजा निकालो कि कटान-बिंदु एक मध्यबिंदु है।
  \item लिखो कि तुमने कौन-सा त्रिभुज, कौन-से मध्यबिंदु और कौन-सी
        प्रमेय इस्तेमाल की --- हर एक पर एक वाक्य।
\end{enumerate}
\end{method}

\section{आधी नाप वाले त्रिभुज}

\begin{proposition}[मध्यबिंदु त्रिभुज]\label{prop:g8:midpoints:medial}
किसी त्रिभुज की तीनों भुजाओं के मध्यबिंदु मिला देने पर वह बराबर
\omterm{def:g6:measure:area}{क्षेत्रफल} के चार त्रिभुजों में कट जाता है, और हर एक असली त्रिभुज की
आधी नाप वाली नक़ल होता है।
\end{proposition}

\begin{proof}
मध्यबिंदु त्रिभुज की हर भुजा $ABC$ की किसी भुजा के \omterm{def:g6:lines:perp}{समांतर} और उसकी
आधी लंबी है (\cref{thm:g8:midpoints:direct}, तीन बार लगाकर)। चारों
छोटे त्रिभुजों की भुजाएँ उन्हीं तीन लंबाइयों की हैं, इसलिए वे एक जैसी
नक़लें हैं; और वे मिलकर $ABC$ को ढक देते हैं, इसलिए हर एक का
\omterm{def:g6:measure:area}{क्षेत्रफल} उसका एक \omterm{def:g3:division:half}{चौथाई} है।
\end{proof}

\begin{remark}[थेल्स की ओर]
मध्यबिंदु प्रमेय एक और बड़ी बात की ``एक \omterm{def:g3:division:half}{आधा}'' वाली स्थिति है: किसी
त्रिभुज की एक भुजा के \omterm{def:g6:lines:perp}{समांतर} रेखा बाक़ी दोनों भुजाओं को \emph{बराबर
अनुपातों} में काटती है --- एक तिहाई और एक तिहाई, दो पाँचवें और दो
पाँचवें\,\dots\ वह आम कथन थेल्स प्रमेय है
(\cref{ch:g9:thales}); इस साल सिर्फ़ मध्यबिंदु वाली स्थिति चाहिए।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:1}
किसी त्रिभुज $RST$ में $M$, $[RS]$ का मध्यबिंदु है और $N$, $[RT]$
का, और $ST = 7$ cm है। रेखा $(MN)$ और लंबाई $MN$ के बारे में क्या
कहा जा सकता है? इस्तेमाल की गई प्रमेय का नाम लो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:2}
किसी त्रिभुज $ABC$ में $I$, $[AB]$ का मध्यबिंदु है और $I$ से होकर
$(BC)$ के \omterm{def:g6:lines:perp}{समांतर} जाने वाली रेखा $[AC]$ को $J$ पर काटती है, और
$AC = 11$ cm है। इस्तेमाल की गई प्रमेय का नाम लेते हुए $AJ$ निकालो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:3}
$AB = 8$ cm, $AC = 6$ cm, $BC = 7$ cm वाला त्रिभुज $ABC$ बनाओ,
$[AB]$ का मध्यबिंदु $I$ और $[AC]$ का $J$ रखो, और प्रमेय जाँचने के
लिए $IJ$ नापो। फिर और कुछ नापे बिना $AIJ$ का \omterm{def:g6:measure:perimeter}{परिमाप} निकालो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:4}
$IJK$, $ABC$ का मध्यबिंदु त्रिभुज है, और $ABC$ की भुजाएँ $10$, $12$
तथा $16$ cm नापती हैं। $IJK$ की तीनों भुजाएँ और उसका \omterm{def:g6:measure:perimeter}{परिमाप} बताओ।
दोनों परिमापों की आपस में क्या तुलना है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g8:midpoints:5}
किसी त्रिभुज का \omterm{def:g6:measure:area}{क्षेत्रफल} $36$ cm$^2$ है। उसके मध्यबिंदु त्रिभुज का
\omterm{def:g6:measure:area}{क्षेत्रफल} क्या है? और चारों छोटे त्रिभुजों में से हर एक का?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:midpoints:6}
किसी त्रिभुज $ABC$ में $I$, $[AB]$ का मध्यबिंदु है, $J$, $[AC]$ का,
और $K$, $[BC]$ का। दिखाओ कि $IJKB$ एक \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} है। ($[IJ]$
और $[BK]$ की तुलना करो: \omterm{def:g6:lines:perp}{समांतर}? बराबर?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:midpoints:7}
किसी त्रिभुजाकार पाल का सबसे नीचे वाला \omterm{def:g5:solids:def}{किनारा} $6.4$ m लंबा है। एक
मज़बूती वाली सिलाई बाक़ी दोनों \omterm{def:g5:solids:def}{किनारों} के मध्यबिंदुओं को जोड़ती है।
कितनी लंबी सिलाई चाहिए? और अगर वही सिलाई ऊपर वाले \omterm{def:g5:solids:def}{शीर्ष} से शुरू
करके हर \omterm{def:g5:solids:def}{किनारे} के एक \omterm{def:g3:division:half}{चौथाई} पर पड़े बिंदुओं को जोड़े तो? (चित्र बनाकर
अटकल लगाओ; प्रमाण थेल्स का है, \cref{ch:g9:thales}।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g8:midpoints:8}
त्रिभुज $ABC$ का $A$ पर \omterm{def:g6:shapes:triangles}{समकोण} है, $M$ कर्ण $[BC]$ का मध्यबिंदु है,
और $M$ से होकर $(AB)$ के \omterm{def:g6:lines:perp}{समांतर} जाने वाली रेखा $[AC]$ को $N$ पर
मिलती है।
\begin{enumerate}
  \item दिखाओ कि $N$, $[AC]$ का मध्यबिंदु है।
  \item समझाओ कि $(MN)$, $(AC)$ पर \omterm{def:g6:lines:perp}{लंब} क्यों है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g8:midpoints:9}
मान लो $ABCD$ \emph{कोई भी} \omterm{def:g4:geometry:polygon}{चतुर्भुज} है, और $I$, $J$, $K$, $L$
क्रमशः $[AB]$, $[BC]$, $[CD]$, $[DA]$ के मध्यबिंदु हैं (यही
\cref{exo:g7:central:11} वाली अटकल है)।
\begin{enumerate}
  \item त्रिभुज $ABC$ में \omterm{def:g6:lines:objects}{रेखाखंड} $[IJ]$ पर और त्रिभुज $ACD$ में
        \omterm{def:g6:lines:objects}{रेखाखंड} $[LK]$ पर मध्यबिंदु प्रमेय लगाओ: दोनों की तुलना
        विकर्ण $[AC]$ से करो।
  \item नतीजा निकालो कि $IJKL$ हमेशा एक \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} होता है।
\end{enumerate}
\end{exercise}

\section{समस्या: तीन माध्यिकाएँ और गुरुत्व केंद्र}

\begin{problem}[{सप्ताहांत समस्या --- किसी त्रिभुज की माध्यिकाएँ एक
ही बिंदु पर मिलती हैं, हर एक के दो-तिहाई पर}]
\label{pb:g8:midpoints:1}
किसी सख़्त गत्ते से त्रिभुज काटो और वह एक ख़ास बिंदु पर रखी पेंसिल की
नोक पर बिलकुल चपटा टिका रहेगा: यही उस त्रिभुज का \emph{गुरुत्व
केंद्र}\index{गुरुत्व केंद्र} है। यह समस्या वही बिंदु ढूँढ़ती है।
किसी त्रिभुज की \emph{माध्यिका}\index{माध्यिका} वह \omterm{def:g6:lines:objects}{रेखाखंड} है जो
किसी \omterm{def:g5:solids:def}{शीर्ष} को सामने वाली भुजा के \emph{मध्यबिंदु} से जोड़ता है। तुम
सिद्ध करोगे --- मध्यबिंदु प्रमेय को दो बार, कुछ अनोखे ढंग से,
इस्तेमाल करके --- कि तीनों माध्यिकाएँ एक ही बिंदु से गुज़रती हैं, और
उस बिंदु की ठीक जगह भी बता दोगे।

\medskip
\noindent\textbf{भाग I --- गरमाहट के लिए।}
\begin{enumerate}
  \item एक बड़ा त्रिभुज $ABC$ बनाओ (कोई ख़ास शक्ल नहीं), $[AB]$ का
        मध्यबिंदु $I$, $[AC]$ का $J$ और $[BC]$ का $K$ बनाओ, और
        तीनों माध्यिकाएँ $[AK]$, $[BJ]$, $[CI]$ खींचो। तुम्हें
        क्या दिखता है?
  \item सिद्ध करो कि कोई भी माध्यिका त्रिभुज को बराबर \omterm{def:g6:measure:area}{क्षेत्रफल} के
        दो त्रिभुजों में काट देती है। (आधार और ऊँचाइयाँ तुलो, और
        \omterm{def:g6:measure:area}{क्षेत्रफल} का सूत्र लगाओ: आधार $\times$ ऊँचाई $\div\ 2$।)
        $36$ cm$^2$ \omterm{def:g6:measure:area}{क्षेत्रफल} वाले त्रिभुज के दोनों टुकड़ों का
        \omterm{def:g6:measure:area}{क्षेत्रफल} क्या होगा?
  \item त्रिभुज $ABC$ में \cref{thm:g8:midpoints:direct} \omterm{def:g6:lines:objects}{रेखाखंड}
        $[IJ]$ के बारे में क्या कहती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दो माध्यिकाएँ दो-तिहाई पर मिलती
हैं।}
माध्यिकाएँ $[BJ]$ और $[CI]$ एक बिंदु पर कटती हैं; उसे $G$ कहो। मान
लो $M$, $[GB]$ का मध्यबिंदु है और $N$, $[GC]$ का।
\begin{enumerate}[resume]
  \item मध्यबिंदु प्रमेय \emph{त्रिभुज $GBC$ में} लगाओ: वह \omterm{def:g6:lines:objects}{रेखाखंड}
        $[MN]$ के बारे में क्या कहती है?
  \item इससे निकालो कि $(IJ) \parallel (MN)$ और $IJ = MN$, और
        \cref{thm:g7:central:recognize} से नतीजा निकालो कि $IJNM$
        एक \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} है।
  \item समझाओ कि $I$ और $N$ दोनों माध्यिका $(CI)$ पर क्यों पड़ते
        हैं, और $J$ तथा $M$ दोनों माध्यिका $(BJ)$ पर --- यानी
        \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} $IJNM$ के विकर्ण $[IN]$ और $[JM]$ हैं, और
        वे ठीक $G$ पर कटते हैं। तब \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} की
        \emph{परिभाषा} $G$ के बारे में क्या कहती है?
  \item इससे निकालो कि $BM = MG = GJ$, और नतीजा निकालो:
        \[
        BG = 2 \times GJ
        \]
        --- यानी बिंदु $G$ माध्यिका $[BJ]$ पर \omterm{def:g5:solids:def}{शीर्ष} $B$ से
        दो-तिहाई दूरी पर बैठता है। माध्यिका $[CI]$ पर वैसा ही
        नतीजा लिखो और उसका कारण दो।
  \item अब $[CI]$ को भूल जाओ और ठीक वही तर्क माध्यिकाओं $[BJ]$
        तथा $[AK]$ के जोड़े पर चलाओ: उनका कटान-बिंदु भी $[BJ]$ पर
        $B$ से दो-तिहाई पर बैठता है। समझाओ कि इससे वह \emph{वही}
        बिंदु $G$ होने पर क्यों मजबूर हो जाता है --- और नतीजा
        निकालो कि तीनों माध्यिकाएँ $G$ से गुज़रती हैं।
  \item जिस त्रिभुज की माध्यिका $[BJ]$, $9$ cm नापती है, उसमें
        $G$, $B$ से कितनी दूर है और $J$ से कितनी?
  \item एक वाक्य में कारण दो कि $AG = 2 \times GK$ भी होता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- छह बराबर फाँकें, और निर्देशांक।}
तीनों माध्यिकाएँ त्रिभुज को $G$ के चारों ओर छह छोटे त्रिभुजों में
काट देती हैं।
\begin{enumerate}[resume]
  \item त्रिभुज $GBC$ में \omterm{def:g6:lines:objects}{रेखाखंड} $[GK]$ एक माध्यिका है। इससे
        निकालो कि दोनों छोटे त्रिभुजों $GBK$ और $GKC$ के \omterm{def:g6:measure:area}{क्षेत्रफल}
        बराबर हैं।
  \item त्रिभुजों $ABK$ और $AKC$ में सवाल 2 लगाकर (दोनों को $ABC$
        की माध्यिका $[AK]$ ने काटा है) दिखाओ कि त्रिभुजों $ABG$
        और $ACG$ के \omterm{def:g6:measure:area}{क्षेत्रफल} बराबर हैं। किसी दूसरी माध्यिका के
        साथ यही दोहराकर नतीजा निकालो कि तीनों त्रिभुजों $ABG$,
        $BCG$, $CAG$ में से हर एक का \omterm{def:g6:measure:area}{क्षेत्रफल} $ABC$ का एक तिहाई
        है।
  \item इन तीनों में से हर त्रिभुज माध्यिका के एक टुकड़े से आधा-आधा
        कट जाता है (जैसे $[GI]$, त्रिभुज $ABG$ की माध्यिका है ---
        किस \omterm{def:g5:solids:def}{शीर्ष} से देखने पर?)। नतीजा निकालो: $G$ के चारों ओर की
        छहों छोटी फाँकों का \omterm{def:g6:measure:area}{क्षेत्रफल} एक ही है, यानी त्रिभुज का एक
        बटा छह।
  \item निर्देशांक जाल पर $A(0, 0)$, $B(6, 0)$ और $C(0, 6)$ लगाओ,
        जहाँ $K(3, 3)$, $[BC]$ का मध्यबिंदु है। भाग II की
        दो-तिहाई वाली प्रमेय से माध्यिका $[AK]$ पर $G$ के
        निर्देशांक निकालो; फिर जाँचो कि वही बिंदु माध्यिका $[BJ]$
        के भी दो-तिहाई पर बैठता है, जहाँ $J(0, 3)$ है। $G$ के
        निर्देशांकों की तुलना इन औसतों से करो
        \[
        \frac{0 + 6 + 0}{3},
        \qquad
        \frac{0 + 0 + 6}{3} .
        \]
  \item तराज़ू वाला अंतिम दाँव: सवाल 2 और 8 से समझाओ कि गत्ते का
        त्रिभुज किसी भी माध्यिका के साथ रखी छुरी की धार पर क्यों
        टिक जाता है --- दोनों तरफ़ बराबर गत्ता --- और इसलिए पेंसिल
        की नोक $G$ पर ही क्यों रखनी पड़ेगी, यानी उस बिंदु पर जिसे
        भौतिकी वाले \omterm{pb:g8:midpoints:1}{गुरुत्व केंद्र} कहते हैं। (पूरी तरह पक्का
        संतुलन-प्रमाण विश्वविद्यालय वाले आयतनों का समाकलन गणित
        माँगता है; आज के लिए बराबर \omterm{def:g6:measure:area}{क्षेत्रफल} ही उसे मानने लायक़ बना
        देते हैं।)
\end{enumerate}
\end{problem}
