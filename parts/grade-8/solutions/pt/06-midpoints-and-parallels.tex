\section*{Capítulo \ref{ch:g8:midpoints} --- Pontos médios e paralelas}

\begin{solution}{exo:g8:midpoints:1}
$M$ e $N$ são os pontos médios de dois lados do triângulo $RST$: pelo
teorema dos pontos médios (\cref{thm:g8:midpoints:direct}),
$(MN) \parallel (ST)$ e $MN = \frac{ST}{2} = 3.5$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:2}
$I$ é um ponto médio e $(IJ) \parallel (BC)$: pela \emph{recíproca}
(\cref{thm:g8:midpoints:converse}), $J$ é o ponto médio de $[AC]$,
então $AJ = \frac{11}{2} = 5.5$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:3}
A medição dá $IJ = 3.5$ cm $= \frac{BC}{2}$. \omterm{def:g6:measure:perimeter}{Perímetro} de $AIJ$:
$AI = 4$, $AJ = 3$, $IJ = 3.5$: total $10.5$ cm --- a \omterm{def:g3:division:half}{metade} do
\omterm{def:g6:measure:perimeter}{perímetro} de $ABC$ ($8 + 6 + 7 = 21$ cm).
\end{solution}

\begin{solution}{exo:g8:midpoints:4}
Cada lado de $IJK$ é a \omterm{def:g3:division:half}{metade} de um lado de $ABC$: $5$, $6$ e $8$ cm.
\omterm{def:g6:measure:perimeter}{Perímetro}: $19$ cm, a \omterm{def:g3:division:half}{metade} de $10 + 12 + 16 = 38$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:5}
O triângulo medial é um dos quatro triângulos iguais da
\cref{prop:g8:midpoints:medial}: \omterm{def:g6:measure:area}{área} $\frac{36}{4} = 9$ cm$^2$ --- e
cada um dos quatro triângulos pequenos tem \omterm{def:g6:measure:area}{área} $9$ cm$^2$.
\end{solution}

\begin{solution}{exo:g8:midpoints:6}
Pelo teorema dos pontos médios em $ABC$: $(IJ) \parallel (BC)$, isto
é, $(IJ) \parallel (BK)$, e $IJ = \frac{BC}{2} = BK$ (pois $K$ é o
ponto médio de $[BC]$). Dois lados opostos de $IJKB$ são paralelos e
iguais: ele é um \omterm{def:g7:central:parallelogram}{paralelogramo}
(\cref{thm:g7:central:recognize}, critério 3).
\end{solution}

\begin{solution}{exo:g8:midpoints:7}
A costura liga dois pontos médios: $\frac{6.4}{2} = 3.2$ m. A um
quarto a partir do \omterm{def:g5:solids:def}{vértice} de cima, o desenho sugere uma costura de
$\frac{6.4}{4} = 1.6$ m, paralela à base --- confirmado por Tales no
\cref{ch:g9:thales}.
\end{solution}

\begin{solution}{exo:g8:midpoints:8}
\emph{1.} $M$ é o ponto médio de $[BC]$ e $(MN) \parallel (AB)$: pela
recíproca do teorema dos pontos médios (no triângulo $ABC$, lado
$[AC]$), $N$ é o ponto médio de $[AC]$.

\emph{2.} $(MN) \parallel (AB)$ e $(AB) \perp (AC)$ (\omterm{def:g3:shapes:rightangle}{ângulo reto} em
$A$): uma reta paralela a $(AB)$ também é perpendicular a $(AC)$
(\cref{prop:g6:lines:facts}). Logo $(MN) \perp (AC)$ --- o \omterm{def:g6:lines:objects}{segmento}
$[MN]$ é o pedaço de \omterm{def:g6:symmetry:bisector}{mediatriz} que mostra de novo que $MA = MC$
(\cref{thm:g8:cosine:circle}).
\end{solution}

\begin{solution}{exo:g8:midpoints:9}
\emph{1.} No triângulo $ABC$, $I$ e $J$ são pontos médios de $[AB]$ e
$[BC]$: $(IJ) \parallel (AC)$ e $IJ = \frac{AC}{2}$. No triângulo
$ACD$, $L$ e $K$ são pontos médios de $[DA]$ e $[CD]$:
$(LK) \parallel (AC)$ e $LK = \frac{AC}{2}$.

\emph{2.} Então $[IJ]$ e $[LK]$ são paralelos (os dois paralelos à
diagonal $(AC)$) e iguais (os dois iguais à \omterm{def:g3:division:half}{metade} de $AC$): $IJKL$ é
um \omterm{def:g7:central:parallelogram}{paralelogramo} (\cref{thm:g7:central:recognize}, critério 3) ---
para \emph{todo} \omterm{def:g4:geometry:polygon}{quadrilátero} $ABCD$, como conjecturado no
\cref{exo:g7:central:11}.
\end{solution}

\begin{solution}{pb:g8:midpoints:1}
\textbf{1.} Qualquer que seja o triângulo traçado, as três medianas
parecem passar por um único ponto, situado dentro do triângulo,
visivelmente mais perto do ponto médio de cada lado do que do \omterm{def:g5:solids:def}{vértice}
oposto. O resto do problema demonstra isso.

\textbf{2.} Tome a mediana $[AK]$ do triângulo $ABC$: os triângulos
$ABK$ e $AKC$ têm bases $BK$ e $KC$ de mesmo comprimento ($K$ é o ponto
médio de $[BC]$) apoiadas na mesma reta $(BC)$, e a mesma altura: a
distância de $A$ até $(BC)$. Pela fórmula da \omterm{def:g6:measure:area}{área}, cada um tem \omterm{def:g6:measure:area}{área}
base $\times$ altura $\div\ 2$, com bases iguais e alturas iguais: as
\omterm{def:g6:measure:area}{áreas} são iguais. Para um triângulo de \omterm{def:g6:measure:area}{área} $36$ cm$^2$: duas peças de
$18$ cm$^2$ cada.

\textbf{3.} $I$ e $J$ são os pontos médios dos lados $[AB]$ e $[AC]$ do
triângulo $ABC$, então, pelo \cref{thm:g8:midpoints:direct}:
$(IJ) \parallel (BC)$ e $IJ = \frac{BC}{2}$.

\textbf{4.} $M$ e $N$ são os pontos médios dos lados $[GB]$ e $[GC]$ do
triângulo $GBC$, então, pelo mesmo teorema: $(MN) \parallel (BC)$ e
$MN = \frac{BC}{2}$.

\textbf{5.} Tanto $(IJ)$ quanto $(MN)$ são \omterm{def:g6:lines:perp}{paralelas} a $(BC)$, logo
\omterm{def:g6:lines:perp}{paralelas} entre si; e $IJ = \frac{BC}{2} = MN$. O \omterm{def:g4:geometry:polygon}{quadrilátero} $IJNM$
tem, assim, dois lados opostos, $[IJ]$ e $[NM]$, paralelos e de mesmo
comprimento: pelo critério 3 do \cref{thm:g7:central:recognize},
$IJNM$ é um \omterm{def:g7:central:parallelogram}{paralelogramo}.

\textbf{6.} $G$ está sobre a mediana $[CI]$, e $N$ é o ponto médio de
$[GC]$, um \omterm{def:g6:lines:objects}{segmento} apoiado nessa mesma reta mediana: então $I$, $N$ (e
$G$, $C$) estão todos sobre $(CI)$. Do mesmo modo, $J$, $M$ e $G$ estão
sobre a reta mediana $(BJ)$. No \omterm{def:g7:central:parallelogram}{paralelogramo} $IJNM$, as diagonais são
$[IN]$ e $[JM]$; elas estão apoiadas nas duas retas medianas, que se
cruzam em $G$ --- então as diagonais se cruzam em $G$. Pela
\emph{definição} de \omterm{def:g7:central:parallelogram}{paralelogramo} (critério 1 da
\cref{thm:g7:central:recognize}: as diagonais têm o mesmo ponto médio),
$G$ é o ponto médio de $[IN]$ \emph{e} de $[JM]$: $GI = GN$ e
$GJ = GM$.

\textbf{7.} $M$ é o ponto médio de $[GB]$, então $BM = MG$; e
$MG = GJ$ pela questão 6. Logo $BM = MG = GJ$: a mediana $[BJ]$ é
cortada em três partes iguais, das quais $BG$ toma duas:
\[
BG = 2 \times GJ .
\]
Simetricamente, $N$ é o ponto médio de $[GC]$ e $GN = GI$, então
$CN = NG = GI$ e $CG = 2 \times GI$: na mediana $[CI]$ também, o ponto
de cruzamento fica a dois terços do caminho a partir do \omterm{def:g5:solids:def}{vértice}.

\textbf{8.} O argumento não usou nada de especial do par $[BJ]$,
$[CI]$: aplicado ao par $[BJ]$, $[AK]$, ele mostra que o ponto de
cruzamento deles também fica sobre $[BJ]$ a dois terços do caminho a
partir de $B$. Mas existe apenas \emph{um} ponto do \omterm{def:g6:lines:objects}{segmento} $[BJ]$ à
distância $\frac23 BJ$ de $B$ --- então o ponto de cruzamento de $[BJ]$
com $[AK]$ é o próprio $G$. As três medianas, portanto, passam por
$G$: elas são \emph{concorrentes}.

\textbf{9.} $BG = \frac23 \times 9 = 6$ cm e
$GJ = \frac13 \times 9 = 3$ cm.

\textbf{10.} Pela questão 8, a mesma conta dos dois terços vale na
mediana $[AK]$ (repita o argumento das questões 4 a 7 com o par
$[AK]$, $[BJ]$): $AG = 2 \times GK$.

\textbf{11.} No triângulo $GBC$, o ponto $K$ é o ponto médio do lado
$[BC]$, então $[GK]$ é uma mediana de $GBC$; pela questão 2, os
triângulos $GBK$ e $GKC$ têm \omterm{def:g6:measure:area}{áreas} iguais.

\textbf{12.} A mediana $[AK]$ corta $ABC$ em $ABK$ e $AKC$, de \omterm{def:g6:measure:area}{áreas}
iguais (questão 2). Retirando de cada um o triângulo pequeno da questão
11:
\[
\text{área}(ABG) = \text{área}(ABK) - \text{área}(GBK),
\qquad
\text{área}(ACG) = \text{área}(AKC) - \text{área}(GKC),
\]
e as \omterm{def:g6:measure:area}{áreas} retiradas são iguais, então
$\text{área}(ABG) = \text{área}(ACG)$. A mesma conta ao longo da
mediana $[BJ]$ --- que divide $ABC$ ao meio em $ABJ$ e $CBJ$, enquanto
$[GJ]$, mediana do triângulo $GAC$, o divide ao meio em $GAJ$ e $GCJ$
--- dá $\text{área}(ABG) = \text{área}(CBG)$. Assim, os três triângulos
$ABG$, $BCG$, $CAG$ têm \omterm{def:g6:measure:area}{áreas} iguais e, como juntos ladrilham $ABC$,
cada um tem \omterm{def:g6:measure:area}{área} igual a um terço do todo.

\textbf{13.} No triângulo $ABG$, o ponto $I$ é o ponto médio do lado
$[AB]$, então $[GI]$ é uma mediana de $ABG$ \emph{a partir do \omterm{def:g5:solids:def}{vértice}
$G$}: ela corta $ABG$ em dois triângulos de \omterm{def:g6:measure:area}{áreas} iguais (questão 2),
a saber, $AIG$ e $IBG$. O mesmo acontece em $BCG$ (mediana $[GK]$) e em
$CAG$ (mediana $[GJ]$). Cada terço de $ABC$ é cortado ao meio: as seis
fatias em torno de $G$ têm cada uma \omterm{def:g6:measure:area}{área}
$\frac13 \times \frac12 = \frac16$ do triângulo.

\textbf{14.} Na mediana $[AK]$, de $A(0,0)$ até $K(3,3)$: $G$ fica a
dois terços do caminho, então as coordenadas dele são
$\left(\frac23 \times 3,\ \frac23 \times 3\right) = (2, 2)$. Na
mediana $[BJ]$, de $B(6, 0)$ até $J(0, 3)$: dois terços do caminho
significa somar dois terços do deslocamento ($-6$ na horizontal, $+3$
na vertical):
\[
\left(6 + \tfrac23 \times (-6),\ 0 + \tfrac23 \times 3\right)
= (2, 2) .
\]
O mesmo ponto --- como a questão 8 prometia. E
$\frac{0 + 6 + 0}{3} = 2$ e $\frac{0 + 0 + 6}{3} = 2$: as coordenadas
do \omterm{pb:g8:midpoints:1}{centro de gravidade} são as \emph{médias} das coordenadas dos três
\omterm{def:g5:solids:def}{vértices}.

\textbf{15.} Pela questão 2, um fio de faca apoiado ao longo de uma
mediana tem a mesma quantidade de papelão de cada lado (\omterm{def:g6:measure:area}{áreas} iguais),
então o triângulo se equilibra ao longo de cada uma das três retas
medianas. Um ponto de equilíbrio tem de estar sobre as três retas de
equilíbrio ao mesmo tempo, e, pela questão 8, as três medianas
compartilham exatamente um ponto: $G$. É ali que vai a ponta do lápis.
(\omterm{def:g6:measure:area}{Áreas} iguais dos dois lados é a versão crível do argumento; a versão
honesta, que pesa \emph{onde} o papelão está e não só quanto há dele,
precisa do cálculo integral dos volumes de graduação --- que confirma
$G$.)
\end{solution}
