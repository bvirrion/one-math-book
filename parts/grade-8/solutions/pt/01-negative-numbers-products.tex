\section*{Capítulo \ref{ch:g8:negprod} --- Multiplicando números negativos}

\begin{solution}{exo:g8:negprod:1}
$(-8) \times (+7) = -56$; \quad
$(-9) \times (-6) = +54$; \quad
$(+12) \times (-0.5) = -6$; \quad
$(-1) \times (-1) \times (-1) = -1$ (três fatores negativos: \omterm{def:g3:numbers:evenodd}{ímpar}).
\end{solution}

\begin{solution}{exo:g8:negprod:2}
$(-63) \div (+9) = -7$; \quad
$(-8) \div (-16) = +0.5$; \quad
$\dfrac{45}{-5} = -9$; \quad
$\dfrac{-7.2}{-8} = +0.9$.
\end{solution}

\begin{solution}{exo:g8:negprod:3}
Três fatores negativos: \emph{negativo}. \quad
$(-1)^{10}$: dez fatores negativos, par: \emph{positivo} (vale $1$).
\quad
$(-2)^7$: sete fatores, \omterm{def:g3:numbers:evenodd}{ímpar}: \emph{negativo}. \quad
$(-4) \times 0 \times (-6) = 0$: nem positivo nem negativo.
\end{solution}

\begin{solution}{exo:g8:negprod:4}
$(-2)^4 = +16$; \quad $-2^4 = -16$ (o menos não é elevado ao
quadrado); \quad $(-3)^3 = -27$; \quad $(-1)^{2026} = +1$ (expoente
par).
\end{solution}

\begin{solution}{exo:g8:negprod:5}
$7 + 2 \times (-6) = 7 + (-12) = -5$.

$(-4) \times (5 - 9) = (-4) \times (-4) = +16$.

$18 \div (-3) - (-2) = -6 + 2 = -4$.
\end{solution}

\begin{solution}{exo:g8:negprod:6}
$\dfrac{(-5) \times (+8)}{-4} = \dfrac{-40}{-4} = +10$.

$\dfrac{(-3) \times (-6)}{(-2) \times (+9)} = \dfrac{+18}{-18} = -1$.
\end{solution}

\begin{solution}{exo:g8:negprod:7}
Para $x = -3$: \ $5x = -15$; \quad $x^2 = (-3)^2 = 9$; \quad
$-x^2 = -9$; \quad $2x^2 + 4x = 18 - 12 = 6$; \quad
$(2x)^2 = (-6)^2 = 36$.
\end{solution}

\begin{solution}{exo:g8:negprod:8}
$(-6) \times (-7) = 42$; \quad
$10 \div (-4) = -2.5$; \quad
$(-5) \times \frac15 = -1$, então o número que falta é
$+\frac15 = 0.2$.
\end{solution}

\begin{solution}{exo:g8:negprod:9}
\emph{1. Falso}: $0.5 \times 0.5 = 0.25$ é menor que
$0.5 + 0.5 = 1$ (ou: $2 \times 3 = 6 > 5$, mas a afirmação teria de
valer sempre).

\emph{2. Verdadeiro}: um quadrado é um \omterm{def:g2:mult:def}{produto} de dois números de mesmo
sinal, logo positivo ou nulo.

\emph{3. Falso}: $(-2) \times (-3) \times 4 = +24$ é positivo com dois
fatores negativos.
\end{solution}

\begin{solution}{exo:g8:negprod:10}
Seja $r$ o número de respostas certas; então $20 - r$ são erradas, e
\[
5r - 3(20 - r) = 36
\ \Longrightarrow\
5r - 60 + 3r = 36
\ \Longrightarrow\
8r = 96
\ \Longrightarrow\
r = 12 .
\]
Doze respostas certas (e oito erradas: confere, $60 - 24 = 36$).
\end{solution}

\begin{solution}{exo:g8:negprod:11}
\omterm{def:g1:counting:zero}{Zero} vezes qualquer coisa é \omterm{def:g1:counting:zero}{zero}, então $(-1) \times 0 = 0$. Escrevendo
$0 = 1 + (-1)$ e distribuindo:
\[
0 = (-1) \times \bigl(1 + (-1)\bigr)
= (-1) \times 1 + (-1) \times (-1)
= -1 + (-1) \times (-1).
\]
O número $(-1) \times (-1)$, somado a $-1$, dá $0$: ele tem de ser o
oposto de $-1$, isto é, $+1$. A distributividade não deixa outra
escolha.
\end{solution}

\begin{solution}{pb:g8:negprod:1}
\textbf{1.} Como $0 = 0 + 0$, a distributividade
(\cref{thm:g7:priorities:distributivity}) dá
\[
0 \times a = (0 + 0) \times a = 0 \times a + 0 \times a .
\]
Então somar $0 \times a$ a si mesmo não muda nada --- e o único número
que se pode somar sem mudar nada é $0$. (Diretamente: subtraia
$0 \times a$ dos dois lados.) Logo $0 \times a = 0$.

\textbf{2.} Desenvolvendo e depois usando $1 + (-1) = 0$ e a questão
1:
\[
a + (-1) \times a
= 1 \times a + (-1) \times a
= \bigl(1 + (-1)\bigr) \times a
= 0 \times a = 0 .
\]
Então $(-1) \times a$ é o número que, somado a $a$, dá $0$: é
exatamente o oposto de $a$, e $(-1) \times a = -a$. O
\cref{exo:g8:negprod:11} é o caso particular $a = -1$: ali,
$(-1) \times (-1) = -(-1) = +1$.

\textbf{3.} Escreva $-a = (-1) \times a$ (questão 2) e reagrupe os
fatores:
\[
(-a) \times b
= \bigl((-1) \times a\bigr) \times b
= (-1) \times (a \times b)
= -(a \times b) .
\]

\textbf{4.} Aplique a questão 3 duas vezes (uma em cada fator):
\[
(-a) \times (-b)
= -\bigl(a \times (-b)\bigr)
= -\bigl(-(a \times b)\bigr)
= a \times b ,
\]
já que o oposto do oposto de um número é o próprio número.

\textbf{5.} Escreva $a = +d$ ou $a = -d$ e $b = +e$ ou $b = -e$, em que
$d$ e $e$ são as distâncias até o \omterm{def:g1:counting:zero}{zero}. Os quatro casos:
\[
(+d) \times (+e) = de, \quad
(+d) \times (-e) = -de, \quad
(-d) \times (+e) = -de, \quad
(-d) \times (-e) = +de,
\]
os dois do meio pela questão 3 e o último pela questão 4. Em todos os
casos, a distância do resultado é $d \times e$, o \omterm{def:g2:mult:def}{produto} das
distâncias, e o sinal é o anunciado pela tabela da
\cref{thm:g8:negprod:rules}: sinais iguais, positivo; sinais opostos,
negativo.

\textbf{6.} $(-1)^2 = +1$, $(-1)^3 = -1$, $(-1)^4 = +1$,
$(-1)^5 = -1$. Em geral, $(-1)^n$ é um \omterm{def:g2:mult:def}{produto} de $n$ fatores
negativos, cada um de distância $1$: pela
\cref{prop:g8:negprod:many}, ele vale $+1$ quando $n$ é par e $-1$
quando $n$ é \omterm{def:g3:numbers:evenodd}{ímpar}.

\textbf{7.} São dez fatores negativos, uma quantidade par, então o
\omterm{def:g2:mult:def}{produto} é \emph{positivo}
(\cref{prop:g8:negprod:many}). A distância dele é
$1 \times 2 \times 3 \times \dots \times 10$.

\textbf{8.} Pela questão 4,
$(-a)^2 = (-a) \times (-a) = a \times a = a^2$. Depois, pela questão
3,
\[
(-a)^3 = (-a)^2 \times (-a) = a^2 \times (-a) = -(a^2 \times a)
= -a^3 .
\]
O sinal de menos desaparece nos expoentes pares e sobrevive nos
\omterm{def:g3:numbers:evenodd}{ímpares}, de acordo com o \cref{ex:g8:negprod:powers}.

\textbf{9.} Se $a = 0$, então $a^2 = 0$. Caso contrário, $a$ e $a$ têm
o \emph{mesmo} sinal, então $a^2 = a \times a$ é positivo
(\cref{thm:g8:negprod:rules}). Um quadrado, portanto, nunca é
negativo; como $-4$ é negativo, nenhum \omterm{def:g7:negatives:def}{número relativo} $x$ satisfaz
$x^2 = -4$.

\textbf{10.} Zoe tem razão. Os expoentes $2026$ e $2027$ têm paridades
opostas, então $(-1)^{2026} = +1$ e $(-1)^{2027} = -1$, que somam $0$.
Em geral, de dois expoentes consecutivos, um é par e o outro é \omterm{def:g3:numbers:evenodd}{ímpar},
então as duas potências são $+1$ e $-1$ em alguma ordem: números
opostos, cuja \omterm{def:g1:addition:def}{soma} é sempre $0$.

\textbf{11.} $A_3 = 1 - 2 + 3 = 2$, depois $A_4 = A_3 - 4 = -2$,
$A_5 = A_4 + 5 = 3$, $A_6 = A_5 - 6 = -3$, $A_7 = A_6 + 7 = 4$.
Conjectura: $A_n = -\frac{n}{2}$ para $n$ par e
$A_n = \frac{n+1}{2}$ para $n$ \omterm{def:g3:numbers:evenodd}{ímpar}.

\textbf{12.} Para $n$ par, os termos se emparelham completamente:
\[
A_n = (1 - 2) + (3 - 4) + \dots + \bigl((n-1) - n\bigr),
\]
e cada par vale $-1$. Há $\frac{n}{2}$ pares (dois termos por par, $n$
termos ao todo), então $A_n = \frac{n}{2} \times (-1) = -\frac{n}{2}$.

\textbf{13.} Para $n$ \omterm{def:g3:numbers:evenodd}{ímpar}, a \omterm{def:g1:addition:def}{soma} $A_n$ é a \omterm{def:g1:addition:def}{soma} $A_{n-1}$ dos
primeiros $n - 1$ termos mais o último termo, que é $+n$ porque $n$ é
\omterm{def:g3:numbers:evenodd}{ímpar}. Como $n - 1$ é par, a questão 12 dá
\[
A_n = A_{n-1} + n = -\frac{n-1}{2} + n
= \frac{-(n-1) + 2n}{2} = \frac{n+1}{2} .
\]

\textbf{14.} $A_{100} = -\frac{100}{2} = -50$, \quad
$A_{101} = \frac{102}{2} = 51$, \quad
$A_{2026} = -\frac{2026}{2} = -1013$.

\textbf{15.} Nenhuma surpresa: passar de $A_{100}$ para $A_{101}$
acrescenta exatamente um termo, a saber, $+101$, então a \omterm{ex:g1:subtraction:difference}{diferença}
tinha de ser $101$. Pela mesma razão,
$A_{2027} - A_{2026} = +2027$ --- sem calcular $A_{2027}$ nenhuma
vez.
\end{solution}
