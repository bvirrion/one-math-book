\section*{Capítulo \ref{ch:g8:negprod} --- Multiplicando números negativos}

\begin{solution}{exo:g8:negprod:1}
$(-8) \times (+7) = -56$; \quad
$(-9) \times (-6) = +54$; \quad
$(+12) \times (-0.5) = -6$; \quad
$(-1) \times (-1) \times (-1) = -1$ (três fatores negativos: ímpar).
\end{solution}

\begin{solution}{exo:g8:negprod:2}

$(-8) \div (-16) = +0.5$; \quad
$\dfrac{45}{-5} = -9$; \quad
$\dfrac{-7.2}{-8} = +0.9$.
\end{solution}

\begin{solution}{exo:g8:negprod:3}
Três fatores negativos: \emph{negativo}. \quad
$(-1)^{10}$: dez fatores negativos, par: \emph{positivo} (é igual
$1$). \quad
$(-2)^7$: sete fatores, ímpar: \emph{negativo}. \quad
$(-4) \times 0 \times (-6) = 0$: nem positivo nem negativo.
\end{solution}

\begin{solution}{exo:g8:negprod:4}
$(-2)^4 = +16$; \quad $-2^4 = -16$ (o menos não é elevado ao quadrado); \quad
$(-3)^3 = -27$; \quad $(-1)^{2026} = +1$ (par expoente).
\end{solution}

\begin{solution}{exo:g8:negprod:5}
$7 + 2 \times (-6) = 7 + (-12) = -5$.

$(-4) \times (5 - 9) = (-4) \times (-4) = +16$.

$18 \div (-3) - (-2) = -6 + 2 = -4$.
\end{solution}

\begin{solution}{exo:g8:negprod:6}
$\dfrac{(-5) \times (+8)}{-4} = \dfrac{-40}{-4} = +10$.

$\dfrac{(-3) \times (-6)}{(-2) \times (+9)} = \dfrac{+18}{-18} = -1$.

\begin{solution}{exo:g8:negprod:7}
Para $ x = -3$: \ $5x = -15$; \quad $ x^2 = (-3)^2 = 9$; \quad
$-x^2 = -9$; \quad $2x^2 + 4x = 18 - 12 = 6$; \quad
$(2x)^2 = (-6)^2 = 36$.
\end{solution}

\begin{solution}{exo:g8:negprod:8}
$(-6) \times (-7) = 42$; \quad
$10 \div (-4) = -2.5$; \quad
$(-5) \times \frac15 = -1$, então o número que falta é
$+\frac15 = 0.2$.
\end{solution}

\begin{solution}{exo:g8:negprod:9}
\emph{1. Falso}: $0.5 \times 0.5 = 0.25$ é menor que
$0.5 + 0.5 = 1$ (ou: $2 \times 3 = 6 > 5$, mas a reivindicação deve ser válida
sempre).

\emph{2. Verdadeiro}: um quadrado é um \omterm{def:g2:mult:def}{produto} de dois números iguais
sinal, portanto positivo ou \omterm{def:g1:counting:zero}{zero}.

\emph{3. Falso}: $(-2) \times (-3) \times 4 = +24$ é positivo com

\end{solution}

\begin{solution}{exo:g8:negprod:10}
Deixar $ r $ seja o número de respostas certas; então $20 - r $ estão errados e
\[
5r - 3(20 - r) = 36
\ \Longrightarrow\
5r - 60 + 3r = 36
\ \Longrightarrow\
8r = 96
\ \Longrightarrow\
r = 12 .
\]
Doze respostas certas (e oito erradas: verifique,

\end{solution}

\begin{solution}{exo:g8:negprod:11}
\omterm{def:g1:counting:zero}{zero} vezes qualquer coisa é \omterm{def:g1:counting:zero}{zero}, então $(-1) \times 0 = 0$. Escrita
$0 = 1 + (-1)$ e distribuindo:
\[
0 = (-1) \times \bigl(1 + (-1)\bigr)
= (-1) \times 1 + (-1) \times (-1)
= -1 + (-1) \times (-1).
\]
O número $(-1) \times (-1)$ adicionado a $-1$ dá $0$: deve ser o
oposto de $-1$, aquilo é $+1$. A distributividade não deixa outra
escolha.
\end{solution}

\begin{solution}{pb:g8:negprod:1}
\textbf{1.} Desde $0 = 0 + 0$, distributividade
(\cref{thm:g7:priorities:distributivity}) dá
\[
0 \times a = (0 + 0) \times a = 0 \times a + 0 \times a .
\]
Então adicionando $0 \times a $ para si mesmo não muda nada --- e o único
o número um que pode adicionar sem alterar nada é $0$. (Diretamente:
subtrair $0 \times a $ de ambos os lados.) Portanto $0 \times a = 0$.

\textbf{2.} Expandindo e depois usando $1 + (-1) = 0$ e pergunta 1:
\[
a + (-1) \times a
= 1 \times a + (-1) \times a
= \bigl(1 + (-1)\bigr) \times a
= 0 \times a = 0 .
\]
Então $(-1) \times a $ é o número que, somado a $ a $, dá $0$:
isso é exatamente o oposto de $ a $, e $(-1) \times a = -a $.
\Cref{exo:g8:negprod:11} é o caso especial $ a = -1$: lá,
$(-1) \times (-1) = -(-1) = +1$.

\textbf{3.} Escrever $-a = (-1) \times a $ (questão 2) e reagrupar os
fatores:
\[
(-a) \times b
= \bigl((-1) \times a\bigr) \times b
= (-1) \times (a \times b)
= -(a \times b) .
\]

\textbf{4.} Aplique a questão 3 duas vezes (uma vez em cada fator):
\[
(-a) \times (-b)
= -\bigl(a \times (-b)\bigr)
= -\bigl(-(a \times b)\bigr)
= a \times b ,
\]
já que o oposto do oposto de um número é o próprio número.

\textbf{5.} Escrever $ a = +d $ ou $ a = -d $ e $ b = +e $ ou $ b = -e $,
onde $ d $ e $ e $ são as distâncias \omterm{def:g1:counting:zero}{zero}. Os quatro casos:
\[
(+d) \times (+e) = de, \quad
(+d) \times (-e) = -de, \quad
(-d) \times (+e) = -de, \quad
(-d) \times (-e) = +de,
\]
os dois do meio pela pergunta 3, o último pela pergunta 4. Em todos os casos
a distância do resultado é $ d \times e $, o \omterm{def:g2:mult:def}{produto} do
distâncias, e o sinal é o anunciado pela tabela de
\cref{thm:g8:negprod:rules}: mesmos sinais positivos, sinais opostos
negativo.

\textbf{6.} $(-1)^2 = +1$, $(-1)^3 = -1$, $(-1)^4 = +1$,
$(-1)^5 = -1$. Em geral $(-1)^n $ é um \omterm{def:g2:mult:def}{produto} de $ n $ negativo
fatores, cada um de distância $1$: por \cref{prop:g8:negprod:many} isso é
$+1$ quando $ n $ é par e $-1$ quando $ n $ é ímpar.

\textbf{7.} Existem dez fatores negativos --- um número par --- então
o \omterm{def:g2:mult:def}{produto} é \emph{positivo} (\cref{prop:g8:negprod:many}). É
a distância é $1 \times 2 \times 3 \times \dots \times 10$.

\textbf{8.} Pela pergunta 4, $(-a)^2 = (-a) \times (-a) = a \times a =
a^2$. Então, pela pergunta 3,
\[
(-a)^3 = (-a)^2 \times (-a) = a^2 \times (-a) = -(a^2 \times a)
= -a^3 .
\]
O sinal de menos desaparece para par expoentes e sobrevive por ímpar
aqueles, de acordo com \cref{ex:g8:negprod:powers}.

\textbf{9.} Se $ a = 0$ então $ a^2 = 0$. De outra forma $ a $ e $ a $ tem
o \emph{par} assinar, então $ a^2 = a \times a $ é positivo
(\cref{thm:g8:negprod:rules}). Um quadrado, portanto, nunca é
negativo; desde $-4$ é negativo, não \omterm{def:g7:negatives:def}{número relativo} $ x $ satisfaz
$ x^2 = -4$.

\textbf{10.} Zoe está certa. Os expoentes $2026$ e $2027$ tem
paridades opostas, então $(-1)^{2026} = +1$ e $(-1)^{2027} = -1$,
que adiciona a $0$. Em geral, de dois expoentes consecutivos um é
par e um é ímpar, então os dois poderes são $+1$ e $-1$ em alguns
ordem: números opostos, cujo \omterm{def:g1:addition:def}{soma} é sempre $0$.

\textbf{11.} $ A_3 = 1 - 2 + 3 = 2$, então
$ A_4 = A_3 - 4 = -2$, $ A_5 = A_4 + 5 = 3$, $ A_6 = A_5 - 6 = -3$,
$ A_7 = A_6 + 7 = 4$. Conjectura: $ A_n = -\frac{n}{2}$ para par $ n $,
e $ A_n = \frac{n+1}{2}$ para ímpar $ n $.

\textbf{12.} Para par $ n $ os termos combinam completamente:
\[
A_n = (1 - 2) + (3 - 4) + \dots + \bigl((n-1) - n\bigr),
\]
e cada par é igual $-1$. Há $\frac{n}{2}$ pares (dois termos
por par, $ n $ termos em tudo), então
$ A_n = \frac{n}{2} \times (-1) = -\frac{n}{2}$.

\textbf{13.} Para ímpar $ n $, o \omterm{def:g1:addition:def}{soma} $ A_n $ é o \omterm{def:g1:addition:def}{soma} $ A_{n-1}$ do
primeiro $ n - 1$ termos, mais o último termo, que é $+n $ desde $ n $ é
ímpar. Como $ n - 1$ é par, a questão 12 dá
\[
A_n = A_{n-1} + n = -\frac{n-1}{2} + n
= \frac{-(n-1) + 2n}{2} = \frac{n+1}{2} .
\]

\textbf{14.} $ A_{100} = -\frac{100}{2} = -50$, \quad
$ A_{101} = \frac{102}{2} = 51$, \quad
$ A_{2026} = -\frac{2026}{2} = -1013$.

\textbf{15.} Nenhuma surpresa: passando de $ A_{100}$ a $ A_{101}$ adiciona
exatamente um termo, a saber $+101$, então o \omterm{ex:g1:subtraction:difference}{diferença} tinha que ser $101$.
Pela mesma razão, $ A_{2027} - A_{2026} = +2027$ --- sem
computação $ A_{2027}$ de forma alguma.
\end{solution}
