\begin{solution}{exo:g8:pythagoras:4}

\begin{solution}{exo:g8:pythagoras:6}

\begin{solution}{exo:g8:pythagoras:9}

\begin{solution}{exo:g8:pythagoras:11}
\emph{1.} Diretamente: $(a + b)^2$. Em peças: quatro triângulos de
\omterm{def:g6:measure:area}{área} $\frac{ab}{2}$ cada um, mais o quadrado inclinado $c^2$:
\[
(a + b)^2 = 4 \times \frac{ab}{2} + c^2 = 2ab + c^2 .
\]

\emph{2.} Desenvolvendo o lado esquerdo: $(a+b)^2 = a^2 + 2ab + b^2$.
Então
\[
a^2 + 2ab + b^2 = 2ab + c^2
\quad\Longrightarrow\quad
a^2 + b^2 = c^2 . \qed
\]
\end{solution}

\begin{solution}{pb:g8:pythagoras:1}
\textbf{1.} $5^2 + 12^2 = 25 + 144 = 169 = 13^2$; \quad
$8^2 + 15^2 = 64 + 225 = 289 = 17^2$; \quad
$20^2 + 21^2 = 400 + 441 = 841 = 29^2$. Os três são ternos e, pela
\cref{thm:g8:pythagoras:converse}, um triângulo com esses comprimentos
de lado é retângulo, com o \omterm{def:g3:shapes:rightangle}{ângulo reto} de frente para o lado mais
longo.

\textbf{2.} A corda forma um triângulo de lados $3$, $4$, $5$ (em
\omterm{def:g6:lines:objects}{segmentos}), e $3^2 + 4^2 = 25 = 5^2$: pela recíproca
(\cref{thm:g8:pythagoras:converse}), o ângulo entre os lados de $3$ e
$4$ \omterm{def:g6:lines:objects}{segmentos} é exatamente reto --- sem transferidor nenhum, só com
nós.

\textbf{3.} Se $a^2 + b^2 = c^2$, então
\[
(ka)^2 + (kb)^2 = k^2 a^2 + k^2 b^2
= k^2\left(a^2 + b^2\right) = k^2 c^2 = (kc)^2 .
\]
A partir de $(3, 4, 5)$: por exemplo, $(6, 8, 10)$, $(9, 12, 15)$ e
$(30, 40, 50)$.

\textbf{4.} Um \omterm{def:g5:division:multiple}{múltiplo} $(3k, 4k, 5k)$ com $5$ como menor número
exigiria $3k = 5$, e nenhum número inteiro $k$ satisfaz isso ($k = 1$
dá $3$ e $k = 2$ dá $6$). Logo $(5, 12, 13)$ não é \omterm{def:g5:division:multiple}{múltiplo} de
$(3, 4, 5)$.

\textbf{5.} A questão 3 mostra que a ampliação produz apenas os ternos
$(3k, 4k, 5k)$; a questão 4 exibe um terno legítimo que não tem essa
forma. Então a família de todos os \omterm{pb:g8:pythagoras:1}{ternos pitagóricos} é estritamente
maior que a dos \omterm{def:g5:division:multiple}{múltiplos} de $(3, 4, 5)$: produzi-los exige algo novo.

\textbf{6.} Escreva $A = m^2$ e $B = n^2$. Pelos \omterm{def:g2:mult:def}{produtos} notáveis
(\cref{pb:g8:equations:1}):
\[
a^2 = (A - B)^2 = A^2 - 2AB + B^2,
\qquad
b^2 = (2mn)^2 = 4m^2n^2 = 4AB .
\]
Somando:
\[
a^2 + b^2 = A^2 - 2AB + B^2 + 4AB
= A^2 + 2AB + B^2 = (A + B)^2
= \left(m^2 + n^2\right)^2 = c^2 .
\]

\textbf{7.}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccc}
$(m, n)$ & $m^2 - n^2$ & $2mn$ & $m^2 + n^2$ \\
\hline
$(2, 1)$ & $3$ & $4$ & $5$ \\
$(3, 1)$ & $8$ & $6$ & $10$ \\
$(3, 2)$ & $5$ & $12$ & $13$ \\
$(4, 1)$ & $15$ & $8$ & $17$ \\
$(4, 3)$ & $7$ & $24$ & $25$ \\
\end{tabular}
\end{center}

\textbf{8.} $(8, 6, 10)$ é o dobro de $(4, 3, 5)$ --- ou seja, do
$(3, 4, 5)$ dos construtores. A máquina às vezes redescobre um terno
antigo disfarçado.

\textbf{9.} Precisamos de $m^2 - n^2 = 21$ e $2mn = 20$, ou seja,
$mn = 10$: tentando $m = 5$ e $n = 2$, dá $25 - 4 = 21$ e
$2 \times 10 = 20$, com hipotenusa $25 + 4 = 29$. A máquina entrega
$(21, 20, 29)$ para $(m, n) = (5, 2)$.

\textbf{10.} $m = 8$ e $n = 6$ dão $m^2 + n^2 = 64 + 36 = 100$. O
terno é
\[
\left(64 - 36,\ 2 \times 48,\ 100\right) = (28,\ 96,\ 100),
\]
e de fato $28^2 + 96^2 = 784 + 9216 = 10\,000 = 100^2$.

\textbf{11.} Com $m = n + 1$:
\[
m^2 - n^2 = (n + 1)^2 - n^2 = 2n + 1
\]
(\omterm{ex:g1:subtraction:difference}{diferença} de quadrados consecutivos, \cref{pb:g8:equations:1});
$2mn = 2n(n + 1) = 2n^2 + 2n$; e
$m^2 + n^2 = n^2 + n^2 + 2n + 1 = 2n^2 + 2n + 1$. A hipotenusa
ultrapassa o cateto par exatamente por $1$. Para $n = 1, 2, 3, 4$:
\[
(3, 4, 5), \quad (5, 12, 13), \quad (7, 24, 25), \quad
(9, 40, 41) .
\]

\textbf{12.} Todo \omterm{def:g3:numbers:evenodd}{número ímpar} de $3$ em diante é $2n + 1$ para algum
$n \geq 1$, e a questão 11 fornece um terno cujo cateto \omterm{def:g3:numbers:evenodd}{ímpar} é
exatamente $2n + 1$. Para $11 = 2 \times 5 + 1$, tome $n = 5$:
\[
(11,\ 60,\ 61),
\qquad
11^2 + 60^2 = 121 + 3600 = 3721 = 61^2 .
\]

\textbf{13.} Com $n = 1$, a máquina dá
$\left(m^2 - 1,\ 2m,\ m^2 + 1\right)$, cujo cateto par $2m$ pode ser
qualquer \omterm{def:g3:numbers:evenodd}{número par} a partir de $4$. Para $14 = 2 \times 7$, tome
$m = 7$: o terno $(48, 14, 50)$. O mesmo terno aparece dobrando
$(7, 24, 25)$ da tabela da questão 7: $(14, 48, 50)$ --- os mesmos três
números.

\textbf{14.} Um \omterm{def:g3:numbers:evenodd}{número par} é $2k$, e $(2k)^2 = 4k^2$ é par (\omterm{def:g5:division:multiple}{múltiplo} de
$4$, portanto par). Um \omterm{def:g3:numbers:evenodd}{número ímpar} é $2k + 1$, e o primeiro \omterm{def:g2:mult:def}{produto}
notável dá
\[
(2k + 1)^2 = 4k^2 + 4k + 1 = 2\left(2k^2 + 2k\right) + 1 ,
\]
um \omterm{def:g3:numbers:evenodd}{número par} mais $1$: \omterm{def:g3:numbers:evenodd}{ímpar}.

\textbf{15.} Suponha que $a$, $b$, $c$ fossem todos \omterm{def:g3:numbers:evenodd}{ímpares}. Pela
questão 14, $a^2$ e $b^2$ são \omterm{def:g3:numbers:evenodd}{ímpares}, então $a^2 + b^2$ é uma \omterm{def:g1:addition:def}{soma} de
dois \omterm{def:g3:numbers:evenodd}{números ímpares}: \emph{par}. Mas $c^2$ seria \emph{\omterm{def:g3:numbers:evenodd}{ímpar}}. A
igualdade $a^2 + b^2 = c^2$ fica, então, impossível. Logo, em todo
\omterm{pb:g8:pythagoras:1}{terno pitagórico}, pelo menos um dos três números é par --- como cada
linha da tabela da questão 7 confirma.
\end{solution}
