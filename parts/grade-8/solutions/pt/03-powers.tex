\section*{Capítulo \ref{ch:g8:powers} --- Potências}

\begin{solution}{exo:g8:powers:1}
$2^6 = 64$; \quad $3^3 = 27$; \quad $10^5 = 100\,000$; \quad
$1^{100} = 1$; \quad $0.1^2 = 0.01$; \quad $6^0 = 1$.
\end{solution}

\begin{solution}{exo:g8:powers:2}
$(-3)^2 = 9$; \quad $-3^2 = -9$; \quad $(-1)^{15} = -1$; \quad
$(-5)^3 = -125$; \quad $\left(\frac23\right)^2 = \frac49$.
\end{solution}

\begin{solution}{exo:g8:powers:3}
$7^4 \times 7^5 = 7^9$; \quad
$\dfrac{2^9}{2^3} = 2^6$; \quad
$\left(10^3\right)^4 = 10^{12}$; \quad
$5^6 \times 5^{-2} = 5^4$; \quad
$3^4 \times 7^4 = 21^4$ (mesmo \omterm{def:g8:powers:def}{expoente}: multiplicam-se as bases).
\end{solution}

\begin{solution}{exo:g8:powers:4}
$10^{-2} = 0.01$; \quad
$4 \times 10^3 = 4000$; \quad
$2.5 \times 10^{-4} = 0.00025$; \quad
$10^0 = 1$.
\end{solution}

\begin{solution}{exo:g8:powers:5}
$10^5$; \quad $10^{-1}$; \quad $10^{10}$; \quad $10^{-6}$.
\end{solution}

\begin{solution}{exo:g8:powers:6}
$\dfrac{10^7 \times 10^{-3}}{10^2} = \dfrac{10^4}{10^2} = 10^2 =
100$.

$\dfrac{2^5 \times 2^4}{2^6} = \dfrac{2^9}{2^6} = 2^3 = 8$.

$\left(2^2\right)^3 \times 2^{-4} = 2^6 \times 2^{-4} = 2^2 = 4$.
\end{solution}

\begin{solution}{exo:g8:powers:7}
$2^3 \times 2^4 = 2^{12}$: \emph{falso} --- os \omterm{def:g8:powers:def}{expoentes} se somam:
$2^7$.

$5^2 + 5^3 = 5^5$: \emph{falso} --- não há regra para \omterm{def:g1:addition:def}{somas}:
$25 + 125 = 150$, enquanto $5^5 = 3125$.

$(3^2)^4 = 3^8$: \emph{verdadeiro}.

$10^3 \times 10^3 = 100^3$: \emph{verdadeiro} --- os dois valem $10^6$
(à esquerda, $10^{3+3}$; à direita, $(10^2)^3$).
\end{solution}

\begin{solution}{exo:g8:powers:8}
Pessoas novas no dia $4$: cada uma das $3^3 = 27$ pessoas informadas no
dia 3 conta a três outras: $3^4 = 81$. Sabendo do boato no fim do dia
4: $3 + 9 + 27 + 81 = 120$ pessoas.
\end{solution}

\begin{solution}{exo:g8:powers:9}
\emph{1.} Espessura: $2^{10} \times 10^{-4}$ m
$= 1024 \times 10^{-4}$ m $\approx 0.1$ m $= 10$ cm.

\emph{2.} $2^{42} \times 10^{-4} \approx 4.4 \times 10^{12} \times
10^{-4} = 4.4 \times 10^8$ m, maior que $3.8 \times 10^8$ m: depois de
$42$ dobras (teóricas!), o maço de papel passaria da Lua.
\end{solution}

\begin{solution}{exo:g8:powers:10}
$2^{10} = 1024$; $10^3 = 1000$; $3^6 = 729$; $5^4 = 625$. Ordem:
\[
5^4 < 3^6 < 10^3 < 2^{10} .
\]
\end{solution}

\begin{solution}{exo:g8:powers:11}
$2^{100} = \left(2^{10}\right)^{10} = 1024^{10}$ e
$10^{30} = \left(10^3\right)^{10} = 1000^{10}$. Como $1024 > 1000$,
multiplicar dez cópias de cada mantém a desigualdade:
$2^{100} > 10^{30}$.
\end{solution}

\begin{solution}{pb:g8:powers:1}
\textbf{1.} Os grãos dobram de casa em casa a partir de $1 = 2^0$: a
casa $k$ leva $2^{k-1}$ grãos (\cref{def:g8:powers:def}). A casa $64$
leva $2^{63}$.

\textbf{2.} $S_1 = 1$, $S_2 = 1 + 2 = 3$, $S_3 = 3 + 4 = 7$,
$S_4 = 7 + 8 = 15$, $S_5 = 15 + 16 = 31$: sempre um a menos que a
potência de $2$ seguinte ($2$, $4$, $8$, $16$, $32$). Conjectura:
$S_n = 2^n - 1$.

\textbf{3.} Dobrar cada termo de $S_n$ sobe cada potência em um
($2 \times 2^{k} = 2^{k+1}$):
\begin{align*}
S_n &= 1 + 2 + 4 + \dots + 2^{n-1}, \\
2 \times S_n &= \phantom{1 + {}} 2 + 4 + \dots + 2^{n-1} + 2^n .
\end{align*}
Subtraindo a primeira linha da segunda, cada termo de $2$ a $2^{n-1}$
aparece nas duas e se cancela:
\[
2 \times S_n - S_n = 2^n - 1,
\qquad\text{isto é}\qquad
S_n = 2^n - 1 .
\]

\textbf{4.} O total é $S_{64} = 2^{64} - 1$ grãos. Uma sexagésima
quinta casa levaria $2^{64}$ grãos: o tabuleiro inteiro carrega
exatamente um grão a menos do que essa única casa carregaria.

\textbf{5.} A primeira \omterm{def:g3:division:half}{metade} leva $S_{32} = 2^{32} - 1$ grãos. O
tabuleiro inteiro leva $2^{64} - 1$, então a segunda \omterm{def:g3:division:half}{metade} leva
\[
\left(2^{64} - 1\right) - \left(2^{32} - 1\right)
= 2^{64} - 2^{32}
= 2^{32} \times \left(2^{32} - 1\right)
\]
(pondo $2^{32}$ em evidência e usando
$2^{32} \times 2^{32} = 2^{64}$, \cref{thm:g8:powers:rules}):
exatamente $2^{32}$ vezes a primeira \omterm{def:g3:division:half}{metade}.

\textbf{6.} Pelas regras dos \omterm{def:g8:powers:def}{expoentes},
$2^{64} = 2^{4 + 60} = 2^4 \times \left(2^{10}\right)^6$. Como
$2^{10} = 1024 > 10^3$,
\[
2^{64} > 16 \times \left(10^3\right)^6 = 16 \times 10^{18}
= 1.6 \times 10^{19} .
\]

\textbf{7.} Mais de $1.6 \times 10^{19}$ grãos a
$5 \times 10^{-2}$ g cada:
\[
1.6 \times 10^{19} \times 5 \times 10^{-2}
= 8 \times 10^{17} \text{ g} .
\]
Dividindo por $10^6$ g por tonelada: mais de $8 \times 10^{11}$
toneladas --- oitocentos bilhões de toneladas.

\textbf{8.} $\dfrac{8 \times 10^{11}}{8 \times 10^{8}} = 10^3$: o rei
prometeu pelo menos \emph{mil anos} da colheita mundial inteira de
hoje. (A lenda conta que os conselheiros dele lhe disseram isso.)

\textbf{9.} $2^{19} = 2^9 \times 2^{10} = 512 \times 1024 =
524\,288 < 10^6$, enquanto
$2^{20} = \left(2^{10}\right)^2 = 1024^2 = 1\,048\,576 > 10^6$. Então o
menor \omterm{def:g8:powers:def}{expoente} é $n = 20$: vinte duplicações passam do milhão.

\textbf{10.} O pagamento do dia $n$ é $2^{n-1}$ centavos (dia 1:
$2^0 = 1$). Ora,
\[
2^{26} = 2^6 \times \left(2^{10}\right)^2
= 64 \times 1\,048\,576 = 67\,108\,864 < 10^8,
\]
\[
2^{27} = 2 \times 2^{26} = 134\,217\,728 > 10^8 .
\]
Então o pagamento diário passa pela primeira vez de $10^8$ centavos
quando $n - 1 = 27$: no dia $28$.

\textbf{11.} $21 = 16 + 4 + 1$: pesos de $16$, $4$ e $1$ g.
$27 = 16 + 8 + 2 + 1$: pesos de $16$, $8$, $2$ e $1$ g.

\textbf{12.} Pela questão 3, os pesos $1, 2, 4, 8$ juntos pesam
$S_4 = 2^4 - 1 = 15$ g. Então, sem o peso de $16$ g, a comerciante não
consegue passar de $15$ g: qualquer alvo de $16$ g ou mais tem de
usá-lo. E um alvo de $15$ g ou menos não pode usá-lo, já que o peso de
$16$ g sozinho já ultrapassa o alvo. A escolha do maior peso é,
portanto, \emph{forçada}. O que resta é um alvo de no máximo $15$ g,
para ser formado com $1, 2, 4, 8$ --- e o mesmo argumento se repete: o
$8$ é forçado (usado se o alvo restante for $\geq 8$, não usado caso
contrário, porque $1 + 2 + 4 = 7$), depois o $4$ (porque
$1 + 2 = 3$), depois o $2$, depois o $1$.

\textbf{13.} Seguindo as escolhas forçadas, o alvo restante depois de
cada etapa é no máximo o total dos pesos que sobram, então o processo
termina com \omterm{def:g3:division:remainder}{resto} $0$: todo alvo de $1$ a $31$ é alcançado. E, como
toda escolha pelo caminho foi forçada, nenhuma outra seleção de pesos
alcança o mesmo alvo: a escrita de cada número de $1$ a $31$ como \omterm{def:g1:addition:def}{soma}
de potências de $2$ distintas existe e é \emph{única}.

\textbf{14.} Os seis pesos totalizam $S_6 = 2^6 - 1 = 63$ g, e o mesmo
argumento de escolha forçada cobre todo alvo de $1$ a $63$ g. Para
$45$: o alvo é $\geq 32$, então use o $32$; resta $13 < 16$, então pule
o $16$; use o $8$ (resta $5$), use o $4$ (resta $1$), pule o $2$, use o
$1$:
\[
45 = 32 + 8 + 4 + 1 = 2^5 + 2^3 + 2^2 + 2^0 .
\]

\textbf{15.} Escolher um $0$ ou um $1$ em cada uma das $64$ casas é
escolher que potências $2^0, 2^1, \dots, 2^{63}$ entram numa \omterm{def:g1:addition:def}{soma} ---
exatamente a pesagem da comerciante com $64$ pesos. O menor número
guardável é $0$ (todas as casas em $0$); o maior é a \omterm{def:g1:addition:def}{soma} de
\emph{todas} as potências, que é o total do rei:
$S_{64} = 2^{64} - 1$ (Parte I). Pelo argumento da escolha forçada,
todo número inteiro entre eles é alcançado exatamente uma vez: uma
máquina de $64$ bits guarda precisamente os números inteiros de $0$ a
$2^{64} - 1$.
\end{solution}
