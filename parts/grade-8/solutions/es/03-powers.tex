\section*{Capítulo \ref{ch:g8:powers} --- Las potencias}

\begin{solution}{exo:g8:powers:1}
$2^6 = 64$; \quad $3^3 = 27$; \quad $10^5 = 100\,000$; \quad
$1^{100} = 1$; \quad $0.1^2 = 0.01$; \quad $6^0 = 1$.
\end{solution}

\begin{solution}{exo:g8:powers:2}
$(-3)^2 = 9$; \quad $-3^2 = -9$; \quad $(-1)^{15} = -1$; \quad
$(-5)^3 = -125$; \quad $\left(\frac23\right)^2 = \frac49$.
\end{solution}

\begin{solution}{exo:g8:powers:3}
$7^4 \times 7^5 = 7^9$; \quad
$\dfrac{2^9}{2^3} = 2^6$; \quad
$\left(10^3\right)^4 = 10^{12}$; \quad
$5^6 \times 5^{-2} = 5^4$; \quad
$3^4 \times 7^4 = 21^4$ (mismo \omterm{def:g8:powers:def}{exponente}: se multiplican las bases).
\end{solution}

\begin{solution}{exo:g8:powers:4}
$10^{-2} = 0.01$; \quad
$4 \times 10^3 = 4000$; \quad
$2.5 \times 10^{-4} = 0.00025$; \quad
$10^0 = 1$.
\end{solution}

\begin{solution}{exo:g8:powers:5}
$10^5$; \quad $10^{-1}$; \quad $10^{10}$; \quad $10^{-6}$.
\end{solution}

\begin{solution}{exo:g8:powers:6}
$\dfrac{10^7 \times 10^{-3}}{10^2} = \dfrac{10^4}{10^2} = 10^2 =
100$.

$\dfrac{2^5 \times 2^4}{2^6} = \dfrac{2^9}{2^6} = 2^3 = 8$.

$\left(2^2\right)^3 \times 2^{-4} = 2^6 \times 2^{-4} = 2^2 = 4$.
\end{solution}

\begin{solution}{exo:g8:powers:7}
$2^3 \times 2^4 = 2^{12}$: \emph{falso} --- los \omterm{def:g8:powers:def}{exponentes} se suman:
$2^7$.

$5^2 + 5^3 = 5^5$: \emph{falso} --- no hay regla para las \omterm{def:g1:addition:def}{sumas}:
$25 + 125 = 150$, mientras que $5^5 = 3125$.

$(3^2)^4 = 3^8$: \emph{verdadero}.

$10^3 \times 10^3 = 100^3$: \emph{verdadero} --- los dos valen $10^6$
(izquierda: $10^{3+3}$; derecha: $(10^2)^3$).
\end{solution}

\begin{solution}{exo:g8:powers:8}
Personas nuevas el día $4$: cada una de las $3^3 = 27$ personas informadas el día
3 se lo cuenta a otras tres: $3^4 = 81$. Lo conocen al final del día 4:
$3 + 9 + 27 + 81 = 120$ personas.
\end{solution}

\begin{solution}{exo:g8:powers:9}
\emph{1.} Grosor: $2^{10} \times 10^{-4}$ m
$= 1024 \times 10^{-4}$ m $\approx 0.1$ m $= 10$ cm.

\emph{2.} $2^{42} \times 10^{-4} \approx 4.4 \times 10^{12} \times
10^{-4} = 4.4 \times 10^8$ m, mayor que $3.8 \times 10^8$ m: tras
$42$ dobleces (¡teóricos!), el fajo de papel pasaría de la Luna.
\end{solution}

\begin{solution}{exo:g8:powers:10}
$2^{10} = 1024$; $10^3 = 1000$; $3^6 = 729$; $5^4 = 625$. Orden:
\[
5^4 < 3^6 < 10^3 < 2^{10} .
\]
\end{solution}

\begin{solution}{exo:g8:powers:11}
$2^{100} = \left(2^{10}\right)^{10} = 1024^{10}$ y
$10^{30} = \left(10^3\right)^{10} = 1000^{10}$. Como $1024 > 1000$,
multiplicar diez copias de cada uno conserva la desigualdad:
$2^{100} > 10^{30}$.
\end{solution}

\begin{solution}{pb:g8:powers:1}
\textbf{1.} Los granos se duplican de casilla en casilla empezando en
$1 = 2^0$: la casilla $k$ tiene $2^{k-1}$ granos
(\cref{def:g8:powers:def}). La casilla $64$ tiene $2^{63}$.

\textbf{2.} $S_1 = 1$, $S_2 = 1 + 2 = 3$, $S_3 = 3 + 4 = 7$,
$S_4 = 7 + 8 = 15$, $S_5 = 15 + 16 = 31$: siempre uno menos que
la potencia de $2$ siguiente ($2$, $4$, $8$, $16$, $32$). Conjetura:
$S_n = 2^n - 1$.

\textbf{3.} Duplicar todos los términos de $S_n$ sube cada potencia en
uno ($2 \times 2^{k} = 2^{k+1}$):
\begin{align*}
S_n &= 1 + 2 + 4 + \dots + 2^{n-1}, \\
2 \times S_n &= \phantom{1 + {}} 2 + 4 + \dots + 2^{n-1} + 2^n .
\end{align*}
Restando la primera línea de la segunda, todos los términos de $2$
a $2^{n-1}$ aparecen en las dos y se cancelan:
\[
2 \times S_n - S_n = 2^n - 1,
\qquad\text{es decir}\qquad
S_n = 2^n - 1 .
\]

\textbf{4.} El total es $S_{64} = 2^{64} - 1$ granos. Una
casilla sexagésima quinta tendría $2^{64}$ granos: el tablero entero
lleva exactamente un grano menos que esa sola casilla.

\textbf{5.} La primera \omterm{def:g3:division:half}{mitad} tiene $S_{32} = 2^{32} - 1$ granos.
El tablero entero tiene $2^{64} - 1$, así que la segunda \omterm{def:g3:division:half}{mitad} tiene
\[
\left(2^{64} - 1\right) - \left(2^{32} - 1\right)
= 2^{64} - 2^{32}
= 2^{32} \times \left(2^{32} - 1\right)
\]
(saca factor común $2^{32}$, usando $2^{32} \times 2^{32} = 2^{64}$,
\cref{thm:g8:powers:rules}): exactamente $2^{32}$ veces la primera
\omterm{def:g3:division:half}{mitad}.

\textbf{6.} Por las reglas de los \omterm{def:g8:powers:def}{exponentes}, $2^{64} = 2^{4 + 60} =
2^4 \times \left(2^{10}\right)^6$. Since $2^{10} = 1024 > 10^3$,
\[
2^{64} > 16 \times \left(10^3\right)^6 = 16 \times 10^{18}
= 1.6 \times 10^{19} .
\]

\textbf{7.} Más de $1.6 \times 10^{19}$ granos a
$5 \times 10^{-2}$ g cada uno:
\[
1.6 \times 10^{19} \times 5 \times 10^{-2}
= 8 \times 10^{17} \text{ g} .
\]
Dividiendo entre $10^6$ g por tonelada: más de $8 \times 10^{11}$
toneladas --- ochocientos mil millones de toneladas.

\textbf{8.} $\dfrac{8 \times 10^{11}}{8 \times 10^{8}} = 10^3$:
el rey prometió al menos \emph{mil años} de la cosecha
mundial entera de hoy. (Cuenta la leyenda que sus consejeros se lo
dijeron.)

\textbf{9.} $2^{19} = 2^9 \times 2^{10} = 512 \times 1024 =
524\,288 < 10^6$, while $2^{20} = \left(2^{10}\right)^2 =
1024^2 = 1\,048\,576 > 10^6$. Así que el menor \omterm{def:g8:powers:def}{exponente} es
$n = 20$: veinte duplicaciones pasan del millón.

\textbf{10.} La paga del día $n$ es $2^{n-1}$ céntimos (día 1:
$2^0 = 1$). Ahora bien,
\[
2^{26} = 2^6 \times \left(2^{10}\right)^2
= 64 \times 1\,048\,576 = 67\,108\,864 < 10^8,
\]
\[
2^{27} = 2 \times 2^{26} = 134\,217\,728 > 10^8 .
\]
Así que la paga diaria supera por primera vez $10^8$ céntimos cuando $n - 1 = 27$:
el día $28$.

\textbf{11.} $21 = 16 + 4 + 1$: pesas de $16$, $4$ y $1$ g.
$27 = 16 + 8 + 2 + 1$: pesas de $16$, $8$, $2$ y $1$ g.

\textbf{12.} Por la pregunta 3, las pesas $1, 2, 4, 8$ juntas
pesan $S_4 = 2^4 - 1 = 15$ g. Así que, sin la pesa de $16$ g, la
comerciante no puede pasar de $15$ g: cualquier objetivo de $16$ g o más tiene que
usarla. Y un objetivo de $15$ g o menos no debe usarla, ya que la
pesa de $16$ g sola ya supera el objetivo. La elección de la
pesa mayor está, por tanto, \emph{forzada}. Lo que queda es un
objetivo de como mucho $15$ g que hay que formar con $1, 2, 4, 8$ --- y
el mismo argumento se repite: el $8$ está forzado (se usa si el objetivo
restante es $\geq 8$ y no se usa si no, porque $1 + 2 + 4 = 7$),
después el $4$ (porque $1 + 2 = 3$), después el $2$ y después el $1$.

\textbf{13.} Siguiendo las elecciones forzadas, el objetivo que queda
después de cada etapa es como mucho el total de las pesas restantes,
así que el proceso termina con \omterm{def:g3:division:remainder}{resto} $0$: se alcanzan todos los objetivos de $1$ a
$31$. Y, como todas las elecciones del camino estaban forzadas,
ninguna otra selección de pesas puede alcanzar el mismo objetivo: la
escritura de cada número de $1$ a $31$ como \omterm{def:g1:addition:def}{suma} de potencias de $2$
distintas existe y es \emph{única}.

\textbf{14.} Las seis pesas suman $S_6 = 2^6 - 1 = 63$ g, y
el mismo argumento de elección forzada cubre todos los objetivos de $1$ a
$63$ g. Para $45$: el objetivo es $\geq 32$, así que usa $32$; quedan
$13 < 16$, así que salta el $16$; usa $8$ (quedan $5$), usa $4$ (queda
$1$), salta el $2$ y usa $1$:
\[
45 = 32 + 8 + 4 + 1 = 2^5 + 2^3 + 2^2 + 2^0 .
\]

\textbf{15.} Elegir un $0$ o un $1$ en cada una de las $64$ casillas
equivale a elegir qué potencias $2^0, 2^1, \dots, 2^{63}$ se
incluyen en una \omterm{def:g1:addition:def}{suma} --- exactamente la pesada de la comerciante con $64$
pesas. El número almacenable menor es $0$ (todas las casillas a
$0$); el mayor es la \omterm{def:g1:addition:def}{suma} de \emph{todas} las potencias, que es
el total del rey: $S_{64} = 2^{64} - 1$ (Parte I). Por el
argumento de la elección forzada, todo número entero intermedio se alcanza
exactamente una vez: una máquina de $64$ bits guarda precisamente los números
enteros de $0$ a $2^{64} - 1$.
\end{solution}
