\section*{Capítulo \ref{ch:g8:midpoints} --- Puntos medios y paralelas}

\begin{solution}{exo:g8:midpoints:1}
$M$ y $N$ son los puntos medios de dos lados del triángulo $RST$: por
el teorema de los puntos medios (\cref{thm:g8:midpoints:direct}),
$(MN) \parallel (ST)$ y $MN = \frac{ST}{2} = 3.5$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:2}
$I$ es un punto medio y $(IJ) \parallel (BC)$: por el \emph{recíproco}
(\cref{thm:g8:midpoints:converse}), $J$ es el punto medio de $[AC]$,
luego $AJ = \frac{11}{2} = 5.5$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:3}
La medida da $IJ = 3.5$ cm $= \frac{BC}{2}$. \omterm{def:g6:measure:perimeter}{Perímetro} de
$AIJ$: $AI = 4$, $AJ = 3$, $IJ = 3.5$: en total $10.5$ cm --- la \omterm{def:g3:division:half}{mitad} del
\omterm{def:g6:measure:perimeter}{perímetro} de $ABC$ ($8 + 6 + 7 = 21$ cm).
\end{solution}

\begin{solution}{exo:g8:midpoints:4}
Cada lado de $IJK$ es la \omterm{def:g3:division:half}{mitad} de un lado de $ABC$: $5$, $6$ y $8$ cm.
\omterm{def:g6:measure:perimeter}{Perímetro}: $19$ cm, la \omterm{def:g3:division:half}{mitad} de $10 + 12 + 16 = 38$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:5}
El triángulo de los puntos medios es uno de los cuatro triángulos iguales de la
\cref{prop:g8:midpoints:medial}: \omterm{def:g6:measure:area}{área} $\frac{36}{4} = 9$ cm$^2$ ---
y cada uno de los cuatro triángulos pequeños tiene \omterm{def:g6:measure:area}{área} $9$ cm$^2$.
\end{solution}

\begin{solution}{exo:g8:midpoints:6}
Por el teorema de los puntos medios en $ABC$: $(IJ) \parallel (BC)$, es decir
$(IJ) \parallel (BK)$, y $IJ = \frac{BC}{2} = BK$ (ya que $K$ es el
punto medio de $[BC]$). Dos lados opuestos de $IJKB$ son paralelos e
iguales: es un \omterm{def:g7:central:parallelogram}{paralelogramo}
(\cref{thm:g7:central:recognize}, criterio 3).
\end{solution}

\begin{solution}{exo:g8:midpoints:7}
La costura une dos puntos medios: $\frac{6.4}{2} = 3.2$ m. A un
\omterm{def:g3:division:half}{cuarto} del \omterm{def:g5:solids:def}{vértice} superior, el dibujo sugiere una costura de
$\frac{6.4}{4} = 1.6$ m, paralela a la base --- lo confirmará Tales
en el \cref{ch:g9:thales}.
\end{solution}

\begin{solution}{exo:g8:midpoints:8}
\emph{1.} $M$ es el punto medio de $[BC]$ y $(MN) \parallel (AB)$:
por el recíproco del teorema de los puntos medios (en el triángulo $ABC$,
lado $[AC]$), $N$ es el punto medio de $[AC]$.

\emph{2.} $(MN) \parallel (AB)$ y $(AB) \perp (AC)$ (\omterm{def:g3:shapes:rightangle}{ángulo recto}
en $A$): una recta paralela a $(AB)$ es también perpendicular a $(AC)$
(\cref{prop:g6:lines:facts}). Así que $(MN) \perp (AC)$ --- el \omterm{def:g6:lines:objects}{segmento}
$[MN]$ es el trozo de \omterm{def:g6:symmetry:bisector}{mediatriz} que vuelve a mostrar que
$MA = MC$ (\cref{thm:g8:cosine:circle}).
\end{solution}

\begin{solution}{exo:g8:midpoints:9}
\emph{1.} En el triángulo $ABC$, $I$ y $J$ son puntos medios de $[AB]$ y
$[BC]$: $(IJ) \parallel (AC)$ y $IJ = \frac{AC}{2}$. En el triángulo
$ACD$, $L$ y $K$ son puntos medios de $[DA]$ y $[CD]$:
$(LK) \parallel (AC)$ y $LK = \frac{AC}{2}$.

\emph{2.} Así que $[IJ]$ y $[LK]$ son paralelos (los dos paralelos a la
diagonal $(AC)$) e iguales (los dos la \omterm{def:g3:division:half}{mitad} de $AC$): $IJKL$ es un
\omterm{def:g7:central:parallelogram}{paralelogramo} (\cref{thm:g7:central:recognize}, criterio 3) ---
para \emph{todo} \omterm{def:g4:geometry:polygon}{cuadrilátero} $ABCD$, como se conjeturaba en el
\cref{exo:g7:central:11}.
\end{solution}

\begin{solution}{pb:g8:midpoints:1}
\textbf{1.} Sea cual sea el triángulo dibujado, las tres medianas parecen
pasar por un único punto, situado dentro del triángulo,
visiblemente más cerca del punto medio de cada lado que del
\omterm{def:g5:solids:def}{vértice} opuesto. El resto del problema lo demuestra.

\textbf{2.} Toma la mediana $[AK]$ del triángulo $ABC$: los
triángulos $ABK$ y $AKC$ tienen bases $BK$ y $KC$ de la misma
longitud ($K$ es el punto medio de $[BC]$) llevadas por la misma recta
$(BC)$, y la misma altura: la distancia de $A$ a $(BC)$.
Por la fórmula del \omterm{def:g6:measure:area}{área}, cada uno tiene \omterm{def:g6:measure:area}{área}
base $\times$ altura $\div\ 2$, con bases iguales y alturas
iguales: las \omterm{def:g6:measure:area}{áreas} son iguales. Para un triángulo de \omterm{def:g6:measure:area}{área}
$36$ cm$^2$: dos trozos de $18$ cm$^2$ cada uno.

\textbf{3.} $I$ y $J$ son los puntos medios de los lados $[AB]$ y
$[AC]$ del triángulo $ABC$, así que, por el
\cref{thm:g8:midpoints:direct}: $(IJ) \parallel (BC)$ y
$IJ = \frac{BC}{2}$.

\textbf{4.} $M$ y $N$ son los puntos medios de los lados $[GB]$ y
$[GC]$ del triángulo $GBC$, así que, por el mismo teorema:
$(MN) \parallel (BC)$ y $MN = \frac{BC}{2}$.

\textbf{5.} Tanto $(IJ)$ como $(MN)$ son \omterm{def:g6:lines:perp}{paralelas} a $(BC)$, luego
\omterm{def:g6:lines:perp}{paralelas} entre sí; y $IJ = \frac{BC}{2} = MN$. El
\omterm{def:g4:geometry:polygon}{cuadrilátero} $IJNM$ tiene, pues, dos lados opuestos, $[IJ]$ y
$[NM]$, paralelos y de la misma longitud: por el criterio 3 del
\cref{thm:g7:central:recognize}, $IJNM$ es un \omterm{def:g7:central:parallelogram}{paralelogramo}.

\textbf{6.} $G$ está en la mediana $[CI]$, y $N$ es el
punto medio de $[GC]$, un \omterm{def:g6:lines:objects}{segmento} llevado por esa misma recta mediana:
así que $I$, $N$ (y $G$, $C$) están todos sobre $(CI)$. Igualmente $J$, $M$
y $G$ están sobre la recta mediana $(BJ)$. En el \omterm{def:g7:central:parallelogram}{paralelogramo}
$IJNM$, las diagonales son $[IN]$ y $[JM]$; están llevadas por
las dos rectas medianas, que se cortan en $G$ --- luego las diagonales
se cortan en $G$. Por la \emph{definición} de \omterm{def:g7:central:parallelogram}{paralelogramo}
(criterio 1 del \cref{thm:g7:central:recognize}: las diagonales
tienen el mismo punto medio), $G$ es el punto medio de $[IN]$ \emph{y}
de $[JM]$: $GI = GN$ y $GJ = GM$.

\textbf{7.} $M$ es el punto medio de $[GB]$, luego $BM = MG$; y
$MG = GJ$ por la pregunta 6. Por tanto $BM = MG = GJ$: el trozo de mediana
$[BJ]$ queda cortado en tres partes iguales, de las que $BG$ ocupa dos:
\[
BG = 2 \times GJ .
\]
Simétricamente, $N$ es el punto medio de $[GC]$ y $GN = GI$, luego
$CN = NG = GI$ y $CG = 2 \times GI$: también sobre la mediana $[CI]$,
el punto de corte se sitúa a dos tercios del \omterm{def:g5:solids:def}{vértice}.

\textbf{8.} El razonamiento no ha usado nada especial de la pareja
$[BJ]$, $[CI]$: aplicado a la pareja $[BJ]$, $[AK]$, muestra que su
punto de corte también está sobre $[BJ]$ a dos tercios de
$B$. Ahora bien, solo hay \emph{un} punto del \omterm{def:g6:lines:objects}{segmento} $[BJ]$ a
distancia $\frac23 BJ$ de $B$ --- luego el punto de corte de
$[BJ]$ y $[AK]$ es exactamente ese mismo $G$. Las tres medianas
pasan, pues, por $G$: son \emph{concurrentes}.

\textbf{9.} $BG = \frac23 \times 9 = 6$ cm y
$GJ = \frac13 \times 9 = 3$ cm.

\textbf{10.} Por la pregunta 8, el mismo cálculo de los dos tercios vale
en la mediana $[AK]$ (aplica el razonamiento de las preguntas 4--7 a
la pareja $[AK]$, $[BJ]$): $AG = 2 \times GK$.

\textbf{11.} En el triángulo $GBC$, el punto $K$ es el punto medio
del lado $[BC]$, luego $[GK]$ es una mediana de $GBC$; por la pregunta
2, los triángulos $GBK$ y $GKC$ tienen \omterm{def:g6:measure:area}{áreas} iguales.

\textbf{12.} La mediana $[AK]$ corta $ABC$ en $ABK$ y $AKC$, de
\omterm{def:g6:measure:area}{áreas} iguales (pregunta 2). Quitando a cada uno su triangulito
de la pregunta 11:
\[
\text{área}(ABG) = \text{área}(ABK) - \text{área}(GBK),
\qquad
\text{área}(ACG) = \text{área}(AKC) - \text{área}(GKC),
\]
y las \omterm{def:g6:measure:area}{áreas} quitadas son iguales, luego
$\text{área}(ABG) = \text{área}(ACG)$. El mismo cálculo
a lo largo de la mediana $[BJ]$ --- que parte $ABC$ en $ABJ$ y
$CBJ$, mientras que $[GJ]$, mediana del triángulo $GAC$, lo parte
en $GAJ$ y $GCJ$ --- da
$\text{área}(ABG) = \text{área}(CBG)$. Así que los tres triángulos
$ABG$, $BCG$, $CAG$ tienen \omterm{def:g6:measure:area}{áreas} iguales y, como juntos
recubren $ABC$, cada uno tiene por \omterm{def:g6:measure:area}{área} un tercio del total.

\textbf{13.} En el triángulo $ABG$, el punto $I$ es el punto medio
del lado $[AB]$, luego $[GI]$ es una mediana de $ABG$ \emph{desde el
\omterm{def:g5:solids:def}{vértice} $G$}: corta $ABG$ en dos triángulos de \omterm{def:g6:measure:area}{áreas} iguales
(pregunta 2), a saber $AIG$ e $IBG$. Lo mismo ocurre en $BCG$
(mediana $[GK]$) y en $CAG$ (mediana $[GJ]$). Cada tercio de
$ABC$ queda cortado por la \omterm{def:g3:division:half}{mitad}: las seis porciones alrededor de $G$ tienen cada una \omterm{def:g6:measure:area}{área}
$\frac13 \times \frac12 = \frac16$ del triángulo.

\textbf{14.} En la mediana $[AK]$, de $A(0,0)$ a $K(3,3)$: $G$
se sitúa a dos tercios del camino, así que sus coordenadas son
$\left(\frac23 \times 3,\ \frac23 \times 3\right) = (2, 2)$. En
la mediana $[BJ]$, de $B(6, 0)$ a $J(0, 3)$: dos tercios del
camino significa añadir dos tercios del desplazamiento ($-6$
en horizontal, $+3$ en vertical):
\[
\left(6 + \tfrac23 \times (-6),\ 0 + \tfrac23 \times 3\right)
= (2, 2) .
\]
El mismo punto --- como prometía la pregunta 8. Y
$\frac{0 + 6 + 0}{3} = 2$, $\frac{0 + 0 + 6}{3} = 2$: las coordenadas del \omterm{pb:g8:midpoints:1}{centro
de gravedad} son las \emph{medias} de las coordenadas de los tres
\omterm{def:g5:solids:def}{vértices}.

\textbf{15.} Por la pregunta 2, el filo de un cuchillo apoyado a lo largo de una mediana deja
la misma cantidad de cartón a cada lado (\omterm{def:g6:measure:area}{áreas} iguales), así que el
triángulo se equilibra sobre cada una de las tres rectas medianas. Un punto
de equilibrio tiene que estar sobre las tres rectas de equilibrio a la vez y, por
la pregunta 8, las tres medianas comparten exactamente un punto: $G$. Ahí
es donde va la punta del lápiz. (Las \omterm{def:g6:measure:area}{áreas} iguales a ambos lados son la
versión creíble del argumento; la honesta, que pesa
\emph{dónde} está el cartón y no solo cuánto hay,
necesita el cálculo integral de los volúmenes de la universidad --- que
confirma $G$.)
\end{solution}
