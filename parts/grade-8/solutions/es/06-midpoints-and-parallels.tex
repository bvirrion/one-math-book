\section*{Capítulo \ref{ch:g8:midpoints} --- Puntos medios y paralelas}

\begin{solution}{exo:g8:midpoints:1}
$M $ y $ N $ son los puntos medios de dos lados del triángulo $ RST $: por
el teorema del punto medio (\cref{thm:g8:midpoints:direct}),
$(MN) \parallel (ST)$ y $ MN = \frac{ST}{2} = 3.5$ centímetro.
\end{solution}

\begin{solution}{exo:g8:midpoints:2}
$ I $ es un punto medio y $(IJ) \parallel (a.\,C.)$: por el \emph{conversar}
(\cref{thm:g8:midpoints:converse}), $ J $ es el punto medio de $[AC]$,
entonces $ AJ = \frac{11}{2} = 5.5$ centímetro.
\end{solution}

\begin{solution}{exo:g8:midpoints:3}
La medida da $ IJ = 3.5$ centímetro $= \frac{a.\,C.}{2}$. \omterm{def:g6:measure:perimeter}{Perímetro} de
$ AIJ $: $ AI = 4$, $ AJ = 3$, $ IJ = 3.5$: total $10.5$ centímetros--- \omterm{def:g3:division:half}{medio} el
\omterm{def:g6:measure:perimeter}{perímetro} de $ ABC $ ($8 + 6 + 7 = 21$ centímetro).
\end{solution}

\begin{solution}{exo:g8:midpoints:4}

\omterm{def:g6:measure:perimeter}{Perímetro}: $19$ centímetro, \omterm{def:g3:division:half}{medio} de $10 + 12 + 16 = 38$ centímetro.
\end{solution}

\begin{solution}{exo:g8:midpoints:5}
El triángulo del punto medio es uno de los cuatro triángulos iguales de
\cref{prop:g8:midpoints:medial}: \omterm{def:g6:measure:area}{área} $\frac{36}{4} = 9$ centímetro $^2$ ---
y cada uno de los cuatro triángulos pequeños tiene \omterm{def:g6:measure:area}{área} $9$ centímetro $^2$.
\end{solution}

\begin{solution}{exo:g8:midpoints:6}
Por el teorema del punto medio en $ ABC $: $(IJ) \parallel (a.\,C.)$, es decir.\
$(IJ) \parallel (BK)$, y $ IJ = \frac{a.\,C.}{2} = BK $ (como $ K $ es el punto medio de $[a.\,C.]$). Dos lados opuestos de $ IJKB $ son \omterm{def:g6:lines:perp}{paralelo} e igual: es un \omterm{def:g7:central:parallelogram}{paralelogramo}
(\cref{thm:g7:central:recognize}, criterio 3).
\end{solution}

\begin{solution}{exo:g8:midpoints:7}
La costura une dos puntos medios: $\frac{6.4}{2} = 3.2$ metro. De acuerdo
\omterm{def:g3:division:half}{cuarto} desde arriba \omterm{def:g5:solids:def}{vértice}, el dibujo sugiere una costura de
$\frac{6.4}{4} = 1.6$ metro, \omterm{def:g6:lines:perp}{paralelo} a la base --- confirmado por Thales
en \cref{ch:g9:thales}.
\end{solution}

\begin{solution}{exo:g8:midpoints:8}
\emph{1.} $ M $ es el punto medio de $[a.\,C.]$ y $(MN) \parallel (AB)$:
por el inverso del teorema del punto medio (en el triángulo $ ABC $,
lado $[AC]$), $ N $ es el punto medio de $[AC]$.

\emph{2.} $(MN) \parallel (AB)$ y $(AB) \perp (AC)$ (\omterm{def:g3:shapes:rightangle}{ángulo recto}
en $ A $): una línea \omterm{def:g6:lines:perp}{paralelo} a $(AB)$ también es \omterm{def:g6:lines:perp}{perpendicular} a $(AC)$
(\cref{prop:g6:lines:facts}). Entonces $(MN) \perp (AC)$ --- el \omterm{def:g6:lines:objects}{segmento}
$[MN]$ es el \omterm{def:g6:symmetry:bisector}{bisectriz perpendicular} pieza que muestra nuevamente que
$ MA = MC $ (\cref{thm:g8:cosine:circle}).
\end{solution}

\begin{solution}{exo:g8:midpoints:9}
\emph{1.} en triángulo $ ABC $, $ I $ y $ J $ son puntos medios de $[AB]$ y
$[a.\,C.]$: $(IJ) \parallel (AC)$ y $ IJ = \frac{AC}{2}$. en triángulo
$ ACD $, $ L $ y $ K $ son puntos medios de $[DA]$ y $[CD]$:
$(LK) \parallel (AC)$ y $ LK = \frac{AC}{2}$.

\emph{2.} Entonces $[IJ]$ y $[LK]$ son \omterm{def:g6:lines:perp}{paralelo} (ambos \omterm{def:g6:lines:perp}{paralelo} al
diagonal $(AC)$) e igual (ambos \omterm{def:g3:division:half}{medio} de $ AC $): $ IJKL $ es un
\omterm{def:g7:central:parallelogram}{paralelogramo} (\cref{thm:g7:central:recognize}, criterio 3) ---
para \emph{cada} \omterm{def:g4:geometry:polygon}{cuadrilátero} $ ABCD $, como se conjetura en
\cref{exo:g7:central:11}.
\end{solution}

\begin{solution}{pb:g8:midpoints:1}
\textbf{1.} Cualquiera que sea el triángulo que se dibuje, las tres medianas aparecen
pasar por un solo punto, situado dentro del triángulo,
visiblemente más cerca del punto medio de cada lado que del opuesto
\omterm{def:g5:solids:def}{vértice}. El resto del problema lo demuestra.

\textbf{2.} toma la mediana $[AK]$ del triángulo $ ABC $: el
triangulos $ ABK $ y $ AKC $ tener bases $ BK $ y $ KC $ del mismo
longitud ($ K $ es el punto medio de $[a.\,C.]$) llevado por la misma línea
$(a.\,C.)$, y la misma altura: la distancia desde $ A $ a $(a.\,C.)$.
por el \omterm{def:g6:measure:area}{área} fórmula, cada uno tiene \omterm{def:g6:measure:area}{área}
base $\times $ altura $\div\ 2$, con bases iguales e iguales
alturas: la \omterm{def:g6:measure:area}{áreas} son iguales. Para un triángulo de \omterm{def:g6:measure:area}{área}
$36$ centímetro $^2$: dos piezas de $18$ centímetro $^2$ cada.

\textbf{3.} $ I $ y $ J $ son los puntos medios de los lados $[AB]$ y
$[AC]$ del triángulo $ ABC $, entonces por
\cref{thm:g8:midpoints:direct}: $(IJ) \parallel (a.\,C.)$ y
$ IJ = \frac{a.\,C.}{2}$.

\textbf{4.} $ M $ y $ N $ son los puntos medios de los lados $[GB]$ y
$[GC]$ del triángulo $ GBC $, entonces por el mismo teorema:
$(MN) \parallel (a.\,C.)$ y $ MN = \frac{a.\,C.}{2}$.

\textbf{5.} Ambos $(IJ)$ y $(MN)$ son \omterm{def:g6:lines:perp}{paralelo} a $(a.\,C.)$, por eso
\omterm{def:g6:lines:perp}{paralelo} el uno al otro; y $ IJ = \frac{a.\,C.}{2} = MN $. El
\omterm{def:g4:geometry:polygon}{cuadrilátero} $ IJNM $ por lo tanto tiene dos lados opuestos, $[IJ]$ y
$[NM]$, \omterm{def:g6:lines:perp}{paralelo} y de la misma longitud: según el criterio 3 de
\cref{thm:g7:central:recognize}, $ IJNM $ es un \omterm{def:g7:central:parallelogram}{paralelogramo}.

\textbf{6.} $ G $ se encuentra en la mediana $[CI]$, y $ N $ es el punto medio de $[GC]$, a \omterm{def:g6:lines:objects}{segmento} llevado por esa misma línea media:
entonces $ I $, $ N $ (y $ G $, $ C $) todos se acuestan $(CI)$. Asimismo $ J $, $ M $
y $ G $ situarse en la línea media $(BJ)$. En el \omterm{def:g7:central:parallelogram}{paralelogramo}
$ IJNM $, las diagonales son $[IN]$ y $[JM]$; son llevados por
las dos líneas medianas, que se cruzan en $ G $ --- entonces las diagonales
cruzar en $ G $.
por el \emph{definición} de un \omterm{def:g7:central:parallelogram}{paralelogramo}
(criterio 1 de \cref{thm:g7:central:recognize}: las diagonales
tienen el mismo punto medio), $ G $ es el punto medio de $[IN]$ \emph{y}
de $[JM]$: $ GI = GN $ y $ GJ = GM $.

\textbf{7.} $ M $ es el punto medio de $[GB]$, entonces $ BM = MG $; y
$ MG = GJ $ por la pregunta 6. Por lo tanto $ BM = MG = GJ $: la pieza mediana
$[BJ]$ se corta en tres partes iguales, de las cuales $ BG $ toma dos:
\[
BG = 2 \times GJ .
\]
Simétricamente, $ N $ es el punto medio de $[GC]$ y $ GN = GI $, entonces
$ CN = NG = GI $ y $ CG = 2 \times GI $: en la mediana $[CI]$ también,
El punto de cruce se encuentra a dos tercios del camino desde la \omterm{def:g5:solids:def}{vértice}.

\textbf{8.} El argumento no usó nada especial sobre la pareja.
$[BJ]$, $[CI]$: ejecutar en el par $[BJ]$, $[AK]$, muestra su
El punto de cruce también se encuentra en $[BJ]$ a dos tercios del camino desde
$ B $. Pero solo hay \emph{uno} punto de la \omterm{def:g6:lines:objects}{segmento} $[BJ]$ a distancia $\frac23 BJ $ de $ B $ --- entonces el punto de cruce de
$[BJ]$ y $[AK]$ es lo mismo $ G $. Las tres medianas
por lo tanto pasa por $ G $: ellos son \emph{concurrente}.

\textbf{9.} $ BG = \frac23 \times 9 = 6$ centímetros y
$ GJ = \frac13 \times 9 = 3$ centímetro.

\textbf{10.} En la pregunta 8 se cumple el mismo cálculo de dos tercios
en la mediana $[AK]$ (ejecute el argumento de las preguntas 4--7 con
la pareja $[AK]$, $[BJ]$): $ AG = 2 \times GK $.

\textbf{11.} en el triángulo $ GBC $, el punto $ K $ es el punto medio
del lado $[a.\,C.]$, entonces $[GK]$ es una mediana de $ GBC $; por pregunta
2, los triángulos $ GBK $ y $ GKC $ tener
        igual \omterm{def:g6:measure:area}{áreas}.

\textbf{12.} la mediana $[AK]$ cortes $ ABC $ en $ ABK $ y $ AKC $ de
igual \omterm{def:g6:measure:area}{áreas} (pregunta 2). Quitando de cada uno su pequeño triángulo.
de la pregunta 11:
\[
\text{area}(ABG) = \text{area}(ABK) - \text{area}(GBK),
\qquad
\text{area}(ACG) = \text{area}(AKC) - \text{area}(GKC),
\]
y el eliminado \omterm{def:g6:measure:area}{áreas} son iguales, entonces
$\text{area}(ABG) = \text{area}(ACG)$. El mismo cálculo
a lo largo de la mediana $[BJ]$ --- que mitades $ ABC $ en $ ABJ $ y
$ CBJ $, mientras $[GJ]$, una mediana del triángulo $ GAC $, la mitad
en $ GAJ $ y $ GCJ $ --- da
$\text{area}(ABG) = \text{area}(CBG)$. Entonces los tres triángulos
$ ABG $, $ BCG $, $ CAG $ tener
        igual \omterm{def:g6:measure:area}{áreas}, y ya que juntos
azulejo $ ABC $, cada uno tiene \omterm{def:g6:measure:area}{área} un tercio del total.

\textbf{13.} en el triángulo $ ABG $, el punto $ I $ es el punto medio
del lado $[AB]$, entonces $[GI]$ es una mediana de $ ABG $ \emph{desde
\omterm{def:g5:solids:def}{vértice} $ G $}: corta $ ABG $ en dos triángulos de igual \omterm{def:g6:measure:area}{áreas}
(pregunta 2), a saber $ AIG $ y $ IBG $. Lo mismo sucede en $ BCG $
(mediana $[GK]$) y en $ CAG $ (mediana $[GJ]$). Cada tercio de
$ ABC $ está cortado \omterm{def:g3:division:half}{medio}: las seis rebanadas alrededor $ G $ cada uno tiene \omterm{def:g6:measure:area}{área}
$\frac13 \times \frac12 = \frac16$ del triángulo.

\textbf{14.} en la mediana $[AK]$, de $ A(0,0)$ a $ K(3,3)$: $ G $
se encuentra a dos tercios del camino, por lo que sus coordenadas son
$\left(\frac23 \times 3,\ \frac23 \times 3\right) = (2, 2)$. encendido
la mediana $[BJ]$, de $ B(6, 0)$ a $ J(0, 3)$: dos tercios del
manera significa sumar dos tercios del desplazamiento ($-6$
horizontalmente, $+3$ verticalmente):
\[
\left(6 + \tfrac23 \times (-6),\ 0 + \tfrac23 \times 3\right)
= (2, 2) .
\]
El mismo punto, como prometía la pregunta 8. Y
$\frac{0 + 6 + 0}{3} = 2$, $\frac{0 + 0 + 6}{3} = 2$: el \omterm{pb:g8:midpoints:1}{centro de
        gravedad}Las coordenadas son las \emph{promedios} de los tres
\omterm{def:g5:solids:def}{vértices}'coordenadas.

\textbf{15.} En la pregunta 2, un filo de cuchillo colocado a lo largo de una mediana tiene
la misma cantidad de cartón en cada lado (igual \omterm{def:g6:measure:area}{áreas}), por lo que el
El triángulo se equilibra a lo largo de cada una de las tres líneas medianas. un punto
del equilibrio debe recaer en las tres líneas de equilibrio a la vez, y por
Pregunta 8 las tres medianas comparten exactamente un punto: $ G $. eso
es donde va la punta del lápiz. (Igual \omterm{def:g6:measure:area}{áreas} en ambos lados está el
versión creíble del argumento; el honesto, pesando
\emph{dónde} el cartón se asienta y no sólo lo que hay,
necesita el calculo integral de la universidad \omterm{def:g6:measure:volume}{volúmenes} --- cual
confirma $ G$.)
\end{solution}
