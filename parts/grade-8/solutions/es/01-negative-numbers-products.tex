\section*{Capítulo \ref{ch:g8:negprod} --- Multiplicar números relativos}

\begin{solution}{exo:g8:negprod:1}
$(-8) \times (+7) = -56$; \quad
$(-9) \times (-6) = +54$; \quad
$(+12) \times (-0.5) = -6$; \quad
$(-1) \times (-1) \times (-1) = -1$ (tres factores negativos: \omterm{def:g3:numbers:evenodd}{impar}).
\end{solution}

\begin{solution}{exo:g8:negprod:2}
$(-63) \div (+9) = -7$; \quad
$(-8) \div (-16) = +0.5$; \quad
$\dfrac{45}{-5} = -9$; \quad
$\dfrac{-7.2}{-8} = +0.9$.
\end{solution}

\begin{solution}{exo:g8:negprod:3}
Tres factores negativos: \emph{negativo}. \quad
$(-1)^{10}$: diez factores negativos, \omterm{def:g3:numbers:evenodd}{par}: \emph{positivo} (vale
$1$). \quad
$(-2)^7$: siete factores, \omterm{def:g3:numbers:evenodd}{impar}: \emph{negativo}. \quad
$(-4) \times 0 \times (-6) = 0$: ni positivo ni negativo.
\end{solution}

\begin{solution}{exo:g8:negprod:4}
$(-2)^4 = +16$; \quad $-2^4 = -16$ (el menos no se eleva al cuadrado); \quad
$(-3)^3 = -27$; \quad $(-1)^{2026} = +1$ (exponente \omterm{def:g3:numbers:evenodd}{par}).
\end{solution}

\begin{solution}{exo:g8:negprod:5}
$7 + 2 \times (-6) = 7 + (-12) = -5$.

$(-4) \times (5 - 9) = (-4) \times (-4) = +16$.

$18 \div (-3) - (-2) = -6 + 2 = -4$.
\end{solution}

\begin{solution}{exo:g8:negprod:6}
$\dfrac{(-5) \times (+8)}{-4} = \dfrac{-40}{-4} = +10$.

$\dfrac{(-3) \times (-6)}{(-2) \times (+9)} = \dfrac{+18}{-18} = -1$.
\end{solution}

\begin{solution}{exo:g8:negprod:7}
Para $x = -3$: \ $5x = -15$; \quad $x^2 = (-3)^2 = 9$; \quad
$-x^2 = -9$; \quad $2x^2 + 4x = 18 - 12 = 6$; \quad
$(2x)^2 = (-6)^2 = 36$.
\end{solution}

\begin{solution}{exo:g8:negprod:8}
$(-6) \times (-7) = 42$; \quad
$10 \div (-4) = -2.5$; \quad
$(-5) \times \frac15 = -1$, así que el número que falta es
$+\frac15 = 0.2$.
\end{solution}

\begin{solution}{exo:g8:negprod:9}
\emph{1. Falso}: $0.5 \times 0.5 = 0.25$ es menor que
$0.5 + 0.5 = 1$ (o bien: $2 \times 3 = 6 > 5$, pero la afirmación tendría que valer
siempre).

\emph{2. Verdadero}: un cuadrado es un \omterm{def:g2:mult:def}{producto} de dos números del mismo
signo y, por tanto, positivo o \omterm{def:g1:counting:zero}{cero}.

\emph{3. Falso}: $(-2) \times (-3) \times 4 = +24$ es positivo con
dos factores negativos.
\end{solution}

\begin{solution}{exo:g8:negprod:10}
Sea $r$ el número de aciertos; entonces $20 - r$ son fallos, y
\[
5r - 3(20 - r) = 36
\ \Longrightarrow\
5r - 60 + 3r = 36
\ \Longrightarrow\
8r = 96
\ \Longrightarrow\
r = 12 .
\]
Doce aciertos (y ocho fallos: comprobación,
$60 - 24 = 36$).
\end{solution}

\begin{solution}{exo:g8:negprod:11}
\omterm{def:g1:counting:zero}{Cero} por cualquier cosa es \omterm{def:g1:counting:zero}{cero}, así que $(-1) \times 0 = 0$. Escribiendo
$0 = 1 + (-1)$ y distribuyendo:
\[
0 = (-1) \times \bigl(1 + (-1)\bigr)
= (-1) \times 1 + (-1) \times (-1)
= -1 + (-1) \times (-1).
\]
El número $(-1) \times (-1)$ sumado a $-1$ da $0$: tiene que ser el
opuesto de $-1$, es decir, $+1$. La distributividad no deja otra
opción.
\end{solution}

\begin{solution}{pb:g8:negprod:1}
\textbf{1.} Como $0 = 0 + 0$, la distributividad
(\cref{thm:g7:priorities:distributivity}) da
\[
0 \times a = (0 + 0) \times a = 0 \times a + 0 \times a .
\]
Así que sumar $0 \times a$ a sí mismo no cambia nada --- y el único
número que se puede sumar sin cambiar nada es $0$. (Directamente:
resta $0 \times a$ en los dos miembros.) Luego $0 \times a = 0$.

\textbf{2.} Desarrollando y usando después $1 + (-1) = 0$ y la pregunta 1:
\[
a + (-1) \times a
= 1 \times a + (-1) \times a
= \bigl(1 + (-1)\bigr) \times a
= 0 \times a = 0 .
\]
Así que $(-1) \times a$ es el número que, sumado a $a$, da $0$:
es exactamente el opuesto de $a$, y $(-1) \times a = -a$.
El \Cref{exo:g8:negprod:11} es el caso particular $a = -1$: allí,
$(-1) \times (-1) = -(-1) = +1$.

\textbf{3.} Escribe $-a = (-1) \times a$ (pregunta 2) y reagrupa los
factores:
\[
(-a) \times b
= \bigl((-1) \times a\bigr) \times b
= (-1) \times (a \times b)
= -(a \times b) .
\]

\textbf{4.} Aplica dos veces la pregunta 3 (una por cada factor):
\[
(-a) \times (-b)
= -\bigl(a \times (-b)\bigr)
= -\bigl(-(a \times b)\bigr)
= a \times b ,
\]
ya que el opuesto del opuesto de un número es el propio número.

\textbf{5.} Escribe $a = +d$ o $a = -d$ y $b = +e$ o $b = -e$,
donde $d$ y $e$ son las distancias al \omterm{def:g1:counting:zero}{cero}. Los cuatro casos:
\[
(+d) \times (+e) = de, \quad
(+d) \times (-e) = -de, \quad
(-d) \times (+e) = -de, \quad
(-d) \times (-e) = +de,
\]
los dos del medio por la pregunta 3 y el último por la pregunta 4. En todos los casos,
la distancia del resultado es $d \times e$, el \omterm{def:g2:mult:def}{producto} de las
distancias, y el signo es el que anuncia la tabla del
\cref{thm:g8:negprod:rules}: signos iguales, positivo; signos opuestos,
negativo.

\textbf{6.} $(-1)^2 = +1$, $(-1)^3 = -1$, $(-1)^4 = +1$,
$(-1)^5 = -1$. En general, $(-1)^n$ es un \omterm{def:g2:mult:def}{producto} de $n$ factores
negativos, cada uno de distancia $1$: por la \cref{prop:g8:negprod:many} vale
$+1$ cuando $n$ es \omterm{def:g3:numbers:evenodd}{par} y $-1$ cuando $n$ es \omterm{def:g3:numbers:evenodd}{impar}.

\textbf{7.} Hay diez factores negativos --- un \omterm{def:g3:numbers:evenodd}{número par} ---, así que
el \omterm{def:g2:mult:def}{producto} es \emph{positivo} (\cref{prop:g8:negprod:many}). Su
distancia es $1 \times 2 \times 3 \times \dots \times 10$.

\textbf{8.} Por la pregunta 4, $(-a)^2 = (-a) \times (-a) = a \times a =
a^2$. Y después, por la pregunta 3,
\[
(-a)^3 = (-a)^2 \times (-a) = a^2 \times (-a) = -(a^2 \times a)
= -a^3 .
\]
El signo menos desaparece con los exponentes \omterm{def:g3:numbers:evenodd}{pares} y sobrevive con los \omterm{def:g3:numbers:evenodd}{impares},
de acuerdo con el \cref{ex:g8:negprod:powers}.

\textbf{9.} Si $a = 0$, entonces $a^2 = 0$. Si no, $a$ y $a$ tienen
el \emph{mismo} signo, así que $a^2 = a \times a$ es positivo
(\cref{thm:g8:negprod:rules}). Un cuadrado no es, pues, nunca
negativo; como $-4$ es negativo, ningún \omterm{def:g7:negatives:def}{número relativo} $x$ cumple
$x^2 = -4$.

\textbf{10.} Zoe tiene razón. Los exponentes $2026$ y $2027$ tienen
paridades opuestas, así que $(-1)^{2026} = +1$ y $(-1)^{2027} = -1$,
que suman $0$. En general, de dos exponentes consecutivos uno es
\omterm{def:g3:numbers:evenodd}{par} y el otro \omterm{def:g3:numbers:evenodd}{impar}, así que las dos potencias son $+1$ y $-1$ en algún
orden: números opuestos, cuya \omterm{def:g1:addition:def}{suma} es siempre $0$.

\textbf{11.} $A_3 = 1 - 2 + 3 = 2$, después
$A_4 = A_3 - 4 = -2$, $A_5 = A_4 + 5 = 3$, $A_6 = A_5 - 6 = -3$,
$A_7 = A_6 + 7 = 4$. Conjetura: $A_n = -\frac{n}{2}$ para $n$ \omterm{def:g3:numbers:evenodd}{par}
y $A_n = \frac{n+1}{2}$ para $n$ \omterm{def:g3:numbers:evenodd}{impar}.

\textbf{12.} Para $n$ \omterm{def:g3:numbers:evenodd}{par}, los términos se emparejan por completo:
\[
A_n = (1 - 2) + (3 - 4) + \dots + \bigl((n-1) - n\bigr),
\]
y cada pareja vale $-1$. Hay $\frac{n}{2}$ parejas (dos términos
por pareja y $n$ términos en total), así que
$A_n = \frac{n}{2} \times (-1) = -\frac{n}{2}$.

\textbf{13.} Para $n$ \omterm{def:g3:numbers:evenodd}{impar}, la \omterm{def:g1:addition:def}{suma} $A_n$ es la \omterm{def:g1:addition:def}{suma} $A_{n-1}$ de los
$n - 1$ primeros términos más el último, que es $+n$ porque $n$ es
\omterm{def:g3:numbers:evenodd}{impar}. Como $n - 1$ es \omterm{def:g3:numbers:evenodd}{par}, la pregunta 12 da
\[
A_n = A_{n-1} + n = -\frac{n-1}{2} + n
= \frac{-(n-1) + 2n}{2} = \frac{n+1}{2} .
\]

\textbf{14.} $A_{100} = -\frac{100}{2} = -50$, \quad
$A_{101} = \frac{102}{2} = 51$, \quad
$A_{2026} = -\frac{2026}{2} = -1013$.

\textbf{15.} Ninguna sorpresa: pasar de $A_{100}$ a $A_{101}$ añade
exactamente un término, $+101$, así que la \omterm{ex:g1:subtraction:difference}{diferencia} tenía que ser $101$.
Por la misma razón, $A_{2027} - A_{2026} = +2027$ --- sin
calcular $A_{2027}$ en absoluto.
\end{solution}
