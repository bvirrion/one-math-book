\section*{Capítulo \ref{ch:g8:negprod} --- Multiplicar números negativos}

\begin{solution}{exo:g8:negprod:1}
$(-8) \times (+7) = -56$; \quad
$(-9) \times (-6) = +54$; \quad
$(+12) \times (-0.5) = -6$; \quad
$(-1) \times (-1) \times (-1) = -1$ (tres factores negativos: \omterm{def:g3:numbers:evenodd}{extraño}).
\end{solution}

\begin{solution}{exo:g8:negprod:2}
$(-63) \div (+9) = -7$; \quad
$(-8) \div (-16) = +0.5$; \quad
$\dfrac{45}{-5} = -9$; \quad
$\dfrac{-7.2}{-8} = +0.9$.
\end{solution}

\begin{solution}{exo:g8:negprod:3}
Tres factores negativos: \emph{negativo}. \quad
$(-1)^{10}$: diez factores negativos, \omterm{def:g3:numbers:evenodd}{incluso}: \emph{positivo} (es igual a
$1$). \quad
$(-2)^7$: siete factores, \omterm{def:g3:numbers:evenodd}{extraño}: \emph{negativo}. \quad
$(-4) \times 0 \times (-6) = 0$: ni positivo ni negativo.
\end{solution}

\begin{solution}{exo:g8:negprod:4}
$(-2)^4 = +16$; \quad $-2^4 = -16$ (el menos no está al cuadrado); \quad
$(-3)^3 = -27$; \quad $(-1)^{2026} = +1$ (\omterm{def:g3:numbers:evenodd}{incluso} exponente).
\end{solution}

\begin{solution}{exo:g8:negprod:5}
$7 + 2 \times (-6) = 7 + (-12) = -5$.

$(-4) \times (5 - 9) = (-4) \times (-4) = +16$.

$18 \div (-3) - (-2) = -6 + 2 = -4$.
\end{solution}

\begin{solution}{exo:g8:negprod:6}
$\dfrac{(-5) \times (+8)}{-4} = \dfrac{-40}{-4} = +10$.

$\dfrac{(-3) \times (-6)}{(-2) \times (+9)} = \dfrac{+18}{-18} = -1$.
\end{solution}

\begin{solution}{exo:g8:negprod:7}
Para $ x = -3$: \ $5x = -15$; \quad $ x^2 = (-3)^2 = 9$; \quad
$-x^2 = -9$; \quad $2x^2 + 4x = 18 - 12 = 6$; \quad
$(2x)^2 = (-6)^2 = 36$.
\end{solution}

\begin{solution}{exo:g8:negprod:8}
$(-6) \times (-7) = 42$; \quad
$10 \div (-4) = -2.5$; \quad
$(-5) \times \frac15 = -1$, entonces el número que falta es
$+\frac15 = 0.2$.
\end{solution}

\begin{solution}{exo:g8:negprod:9}
\emph{1. Falso}: $0.5 \times 0.5 = 0.25$ es más pequeño que
$0.5 + 0.5 = 1$ (o: $2 \times 3 = 6 > 5$, pero el reclamo debe sostenerse
siempre).

\emph{2. Verdadero}: un cuadrado es un \omterm{def:g2:mult:def}{producto} de dos números iguales
signo, por lo tanto positivo o \omterm{def:g1:counting:zero}{cero}.

\emph{3. Falso}: $(-2) \times (-3) \times 4 = +24$ es positivo con
dos factores negativos.
\end{solution}

\begin{solution}{exo:g8:negprod:10}
Sea $ r $ el número de respuestas correctas; entonces $20 - r $ están equivocados y
\[
5r - 3(20 - r) = 36
\ \Longrightarrow\
5r - 60 + 3r = 36
\ \Longrightarrow\
8r = 96
\ \Longrightarrow\
r = 12 .
\]
Doce respuestas correctas (y ocho incorrectas: verifique,

\end{solution}

\begin{solution}{exo:g8:negprod:11}
\omterm{def:g1:counting:zero}{Cero} veces cualquier cosa es \omterm{def:g1:counting:zero}{cero}, entonces $(-1) \times 0 = 0$. Escribiendo
$0 = 1 + (-1)$ y distribuyendo:
\[
0 = (-1) \times \bigl(1 + (-1)\bigr)
= (-1) \times 1 + (-1) \times (-1)
= -1 + (-1) \times (-1).
\]
el número $(-1) \times (-1)$ añadido a $-1$ da $0$: debe ser el
opuesto de $-1$, eso es $+1$. La distributividad no deja otra
elección.
\end{solution}

\begin{solution}{pb:g8:negprod:1}
\textbf{1.} Desde $0 = 0 + 0$, distributividad
(\cref{thm:g7:priorities:distributivity}) da
\[
0 \times a = (0 + 0) \times a = 0 \times a + 0 \times a .
\]
Entonces agregando $0 \times a $ para sí mismo no cambia nada --- y el único
El número uno que se puede sumar sin cambiar nada es. $0$. (Directamente:
restar $0 \times a $ de ambos lados.) Por lo tanto $0 \times a = 0$.

\textbf{2.} Expandiendo y luego usando $1 + (-1) = 0$ y pregunta 1:
\[
a + (-1) \times a
= 1 \times a + (-1) \times a
= \bigl(1 + (-1)\bigr) \times a
= 0 \times a = 0 .
\]
Entonces $(-1) \times a $ es el número que, sumado a $ a $, da $0$:
eso es exactamente lo contrario de $ a $, y $(-1) \times a = -a $.
\Cref{exo:g8:negprod:11} es el caso especial $ a = -1$: allá,
$(-1) \times (-1) = -(-1) = +1$.

\textbf{3.} Escribir $-a = (-1) \times a $ (pregunta 2) y reagrupar los
factores:
\[
(-a) \times b
= \bigl((-1) \times a\bigr) \times b
= (-1) \times (a \times b)
= -(a \times b) .
\]

\textbf{4.} Aplique la pregunta 3 dos veces (una vez en cada factor):
\[
(-a) \times (-b)
= -\bigl(a \times (-b)\bigr)
= -\bigl(-(a \times b)\bigr)
= a \times b ,
\]
ya que el opuesto del opuesto de un número es el número mismo.

\textbf{5.} Escribir $ a = +d $ o $ a = -d $ y $ b = +e $ o $ b = -e $, dónde $ d $ y $ e $ son las distancias a \omterm{def:g1:counting:zero}{cero}. Los cuatro casos:
\[
(+d) \times (+e) = de, \quad
(+d) \times (-e) = -de, \quad
(-d) \times (+e) = -de, \quad
(-d) \times (-e) = +de,
\]
los dos del medio por la pregunta 3, el último por la pregunta 4. En todos los casos
la distancia del resultado es $ d \times e $, el \omterm{def:g2:mult:def}{producto} de la
distancias, y el signo es el anunciado por la tabla de
\cref{thm:g8:negprod:rules}: mismos signos positivos, signos opuestos
negativo.

\textbf{6.} $(-1)^2 = +1$, $(-1)^3 = -1$, $(-1)^4 = +1$,
$(-1)^5 = -1$. En general $(-1)^n $ es un \omterm{def:g2:mult:def}{producto} de $ n $ negativo
factores, cada uno de distancia $1$: por \cref{prop:g8:negprod:many} es
$+1$ cuando $ n $ es \omterm{def:g3:numbers:evenodd}{incluso} y $-1$ cuando $ n $ es \omterm{def:g3:numbers:evenodd}{extraño}.

\textbf{7.} Hay diez factores negativos --- un \omterm{def:g3:numbers:evenodd}{número par} --- entonces el \omterm{def:g2:mult:def}{producto} es \emph{positivo} (\cref{prop:g8:negprod:many}). su
la distancia es $1 \times 2 \times 3 \times \dots \times 10$.

\textbf{8.} Por la pregunta 4, $(-a)^2 = (-a) \times (-a) = a \times a =
a^2$. Luego por la pregunta 3,
\[
(-a)^3 = (-a)^2 \times (-a) = a^2 \times (-a) = -(a^2 \times a)
= -a^3 .
\]
El signo menos desaparece durante \omterm{def:g3:numbers:evenodd}{incluso} exponentes y sobrevive durante \omterm{def:g3:numbers:evenodd}{extraño}
unos, en consonancia con \cref{ex:g8:negprod:powers}.

\textbf{9.} Si $ a = 0$ entonces $ a^2 = 0$. De lo contrario $ a $ y $ a $ tener
el \emph{mismo} firmar, entonces $ a^2 = a \times a $ es positivo
(\cref{thm:g8:negprod:rules}). Por lo tanto, un cuadrado nunca es
negativo; desde $-4$ es negativo, no \omterm{def:g7:negatives:def}{número relativo} $ x $ satisface
$ x^2 = -4$.

\textbf{10.} Zoé tiene razón. los exponentes $2026$ y $2027$ tener
paridades opuestas, por lo que $(-1)^{2026} = +1$ y $(-1)^{2027} = -1$,
que se suman a $0$. En general, de dos exponentes consecutivos uno es
\omterm{def:g3:numbers:evenodd}{incluso} y uno es \omterm{def:g3:numbers:evenodd}{extraño}, entonces las dos potencias son $+1$ y $-1$ en algunos
orden: números opuestos, cuyo \omterm{def:g1:addition:def}{suma} es siempre $0$.

\textbf{11.} $ A_3 = 1 - 2 + 3 = 2$, entonces
$ A_4 = A_3 - 4 = -2$, $ A_5 = A_4 + 5 = 3$, $ A_6 = A_5 - 6 = -3$,
$ A_7 = A_6 + 7 = 4$. Conjetura: $ A_n = -\frac{n}{2}$ para \omterm{def:g3:numbers:evenodd}{incluso} $ n $,
y $ A_n = \frac{n+1}{2}$ para \omterm{def:g3:numbers:evenodd}{extraño} $ n $.

\textbf{12.} Para \omterm{def:g3:numbers:evenodd}{incluso} $ n $ los términos se emparejan completamente:
\[
A_n = (1 - 2) + (3 - 4) + \dots + \bigl((n-1) - n\bigr),
\]
y cada par es igual $-1$. Hay $\frac{n}{2}$ pares (dos términos
por par, $ n $ términos en total), por lo que
$ A_n = \frac{n}{2} \times (-1) = -\frac{n}{2}$.

\textbf{13.} Para \omterm{def:g3:numbers:evenodd}{extraño} $ n $, el \omterm{def:g1:addition:def}{suma} $ A_n $ es el \omterm{def:g1:addition:def}{suma} $ A_{n-1}$ del primero $ n - 1$ términos, más el último término, que es $+n $ desde $ n $ es
\omterm{def:g3:numbers:evenodd}{extraño}. Como $ n - 1$ es \omterm{def:g3:numbers:evenodd}{incluso}, la pregunta 12 da
\[
A_n = A_{n-1} + n = -\frac{n-1}{2} + n
= \frac{-(n-1) + 2n}{2} = \frac{n+1}{2} .
\]

\textbf{14.} $ A_{100} = -\frac{100}{2} = -50$, \quad
$ A_{101} = \frac{102}{2} = 51$, \quad
$ A_{2026} = -\frac{2026}{2} = -1013$.

\textbf{15.} No es de extrañar: pasando de $ A_{100}$ a $ A_{101}$ agrega
exactamente un término, es decir $+101$, entonces el \omterm{ex:g1:subtraction:difference}{diferencia} tenía que ser $101$.
Por la misma razón, $ A_{2027} - A_{2026} = +2027$ --- sin
informática $ A_{2027}$ en absoluto.
\end{solution}
