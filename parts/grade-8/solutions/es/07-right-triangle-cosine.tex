\section*{Capítulo \ref{ch:g8:cosine} --- El triángulo rectángulo y el coseno}

\begin{solution}{exo:g8:cosine:1}
Mediana desde el \omterm{def:g3:shapes:rightangle}{ángulo recto}: la \omterm{def:g3:division:half}{mitad} de la hipotenusa, $6.5$ cm
(\cref{thm:g8:cosine:circle}). La \omterm{pb:g6:symmetry:1}{circunferencia circunscrita} tiene su centro
en el punto medio de la hipotenusa y \omterm{def:g3:shapes:shapes}{radio} $6.5$ cm.
\end{solution}

\begin{solution}{exo:g8:cosine:2}
El ángulo en $A$ es recto ($[BC]$ es un \omterm{def:g4:geometry:circle}{diámetro}: punto 2 del
\cref{thm:g8:cosine:circle}). Las medidas cumplen
$AB^2 + AC^2 \approx 64 = BC^2$, salvo precisión de medida.
\end{solution}

\begin{solution}{exo:g8:cosine:3}
Hipotenusa: $[EF]$ (opuesta al \omterm{def:g3:shapes:rightangle}{ángulo recto} $D$). Adyacente a
$\widehat E$: $[ED]$. Opuesto: $[DF]$. Luego
$\cos\widehat E = \dfrac{ED}{EF}$.
\end{solution}

\begin{solution}{exo:g8:cosine:4}

\emph{2.} hipotenusa $= \dfrac{6}{\cos 50^\circ} \approx

\end{solution}

\begin{solution}{exo:g8:cosine:5}
$\cos\theta = \frac{8}{10} = 0.8$, luego
$\theta = \cos^{-1}(0.8) \approx 37^\circ$.
\end{solution}

\begin{solution}{exo:g8:cosine:6}
$\cos\theta$ es un cateto dividido entre la hipotenusa, y la hipotenusa
es el lado \emph{más largo} de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}: el \omterm{def:g3:division:remainder}{cociente} es
siempre menor que $1$. Un valor de $1.2$ significaría un cateto más largo
que la hipotenusa --- imposible.
\end{solution}

\begin{solution}{exo:g8:cosine:7}
Alcance horizontal $= 25 \cos 12^\circ \approx 25 \times 0.978 \approx
24.5$ m.
\end{solution}

\begin{solution}{exo:g8:cosine:8}
El coseno disminuye a medida que el ángulo crece
(\cref{prop:g8:cosine:bounds}):
\[
\cos 70^\circ < \cos 45^\circ < \cos 20^\circ .
\]
(Calculadora: $0.342 < 0.707 < 0.940$.)
\end{solution}

\begin{solution}{exo:g8:cosine:9}
La distancia en el suelo es adyacente a $\theta$:
$\cos 68^\circ = \frac{1.2}{L}$, luego
$L = \frac{1.2}{0.375} = 3.2$ m. Altura (Pitágoras):
$\sqrt{3.2^2 - 1.2^2} = \sqrt{10.24 - 1.44} = \sqrt{8.8} \approx 3.0$
m.
\end{solution}

\begin{solution}{exo:g8:cosine:10}
\emph{1.} $A$ está en una circunferencia de \omterm{def:g4:geometry:circle}{diámetro} $[BC]$: \omterm{def:g3:shapes:rightangle}{ángulo recto} en $A$
(\cref{thm:g8:cosine:circle}).

\emph{2.} $BC = 9$ cm (el doble del \omterm{def:g3:shapes:shapes}{radio}). Pitágoras:
$AC = \sqrt{9^2 - 5.4^2} = \sqrt{81 - 29.16} = \sqrt{51.84} = 7.2$
cm. Y $\cos\widehat B = \dfrac{AB}{BC} = \dfrac{5.4}{9} = 0.6$.
\end{solution}

\begin{solution}{exo:g8:cosine:11}
En el \omterm{def:g6:shapes:triangles}{triángulo equilátero} de lado $1$, la altura desde un \omterm{def:g5:solids:def}{vértice}
cae en el \emph{punto medio} del lado opuesto (\omterm{def:g6:symmetry:axis}{eje de
simetría}). La \omterm{def:g3:division:half}{mitad} del triángulo es rectángulo, con hipotenusa $1$
(un lado entero), un ángulo de $60^\circ$ en el \omterm{def:g5:solids:def}{vértice} de la base y
lado adyacente $\frac12$ (la \omterm{def:g3:division:half}{mitad} de la base). Por tanto
\[
\cos 60^\circ = \frac{1/2}{1} = \frac12 .
\]
\end{solution}

\begin{solution}{pb:g8:cosine:1}
\textbf{1.} En el triángulo $ABH$, rectángulo en $H$, la \omterm{def:g1:addition:def}{suma}
de los ángulos (\cref{thm:g7:triangles:sum}) da
$\widehat B + 90^\circ + \widehat{BAH} = 180^\circ$, luego
$\widehat{BAH} = 90^\circ - \widehat B$. En el triángulo $ABC$,
rectángulo en $A$:
$\widehat B + \widehat C + 90^\circ = 180^\circ$, luego
$\widehat C = 90^\circ - \widehat B$. Por tanto
$\widehat{BAH} = \widehat{ACB}$: los dos triángulos pequeños repiten los
ángulos $\widehat B$, $\widehat C$, $90^\circ$ del original.

\textbf{2.} En $ABC$ (rectángulo en $A$), el lado adyacente a
$\widehat B$ es $AB$ y la hipotenusa es $BC$:
$\cos\widehat B = \frac{AB}{BC}$. En $ABH$ (rectángulo en $H$),
el lado adyacente a $\widehat B$ es $BH$ y la hipotenusa es
$AB$: $\cos\widehat B = \frac{BH}{AB}$. La razón solo depende del
ángulo (\cref{def:g8:cosine:def}), luego
\[
\frac{AB}{BC} = \frac{BH}{AB} ;
\]
multiplicando los dos miembros por $BC \times AB$
(\cref{thm:g8:equations:balance}):
$AB^2 = BH \times BC$.

\textbf{3.} Los triángulos $ABC$ (rectángulo en $A$) y $ACH$
(rectángulo en $H$) comparten el ángulo $\widehat C$. Lado adyacente
entre hipotenusa en cada uno:
\[
\cos\widehat C = \frac{AC}{CB} = \frac{CH}{AC}
\qquad\Longrightarrow\qquad
AC^2 = CH \times CB .
\]

\textbf{4.} $AB^2 = BH \times BC$ se lee $9 = BH \times 5$, luego
$BH = 1.8$; y $16 = CH \times 5$ da $CH = 3.2$. Comprobación:
$BH + HC = 1.8 + 3.2 = 5 = BC$ --- como tenía que ser, ya que $H$ está sobre
la hipotenusa, entre $B$ y $C$.

\textbf{5.} Dividiendo cada relación entre $BC$:
\[
BH = \frac{AB^2}{BC},
\qquad
CH = \frac{AC^2}{BC} .
\]
Así que $BH$ y $CH$ son \omterm{def:g6:propdata:def}{proporcionales} a $AB^2$ y $AC^2$: el pie
de la altura parte la hipotenusa en la razón de los cuadrados
de los catetos ($1.8 : 3.2 = 9 : 16$ en la pregunta 4).

\textbf{6.} Sumando las dos relaciones y sacando factor común $BC$
(distributividad):
\[
AB^2 + AC^2 = BH \times BC + CH \times BC
= (BH + CH) \times BC = BC \times BC = BC^2 .
\]
Esto es el \cref{thm:g8:pythagoras:direct}, obtenido solo a partir del coseno
--- una demostración auténticamente distinta del rompecabezas de \omterm{def:g6:measure:area}{áreas}
de los cuatro triángulos del \cref{exo:g8:pythagoras:11}.

\textbf{7.} Pitágoras en $ABH$ (rectángulo en $H$, hipotenusa
$[AB]$): $AH^2 = AB^2 - BH^2$. Sustituyendo $AB^2$ por
$BH \times BC$ (pregunta 2) y sacando factor común $BH$:
\[
AH^2 = BH \times BC - BH^2
= BH \times (BC - BH)
= BH \times HC .
\]

\textbf{8.} Tomando los catetos como base y altura, el \omterm{def:g6:measure:area}{área} de $ABC$
es $\frac{AB \times AC}{2}$; tomando la hipotenusa $[BC]$ como base,
la altura es exactamente $AH$, así que el \omterm{def:g6:measure:area}{área} también es
$\frac{BC \times AH}{2}$ (\cref{thm:g7:areas:triangle}). Igualando
y multiplicando por dos: $AB \times AC = BC \times AH$.

\textbf{9.} $AH = \frac{AB \times AC}{BC} = \frac{3 \times 4}{5}
= 2.4$. Comprobación de la pregunta 7:
$AH^2 = 2.4^2 = 5.76$ y
$BH \times HC = 1.8 \times 3.2 = 5.76$. Iguales.

\textbf{10.} $AH^2 = BH \times HC = 2 \times 8 = 16$, luego
$AH = \sqrt{16} = 4$ cm.

\textbf{11.} El punto $A$ está en la circunferencia de \omterm{def:g4:geometry:circle}{diámetro} $[BC]$
(y es distinto de $B$ y de $C$), así que, por el punto 2 del
\cref{thm:g8:cosine:circle}, el triángulo $ABC$ es rectángulo en
$A$. Por construcción $(AH) \perp (BC)$ con $H$ entre $B$ y
$C$: el \omterm{def:g6:lines:objects}{segmento} $[AH]$ es precisamente la altura de la pregunta 7,
y $AH^2 = BH \times HC = p \times q$.

\textbf{12.} $AH^2 = 1 \times 2 = 2$: la figura construye, solo con
regla y compás, un \omterm{def:g6:lines:objects}{segmento} cuyo cuadrado es $2$ --- la
diagonal del cuadrado unidad, $\sqrt2 \approx 1.414$, del
\cref{ex:g8:pythagoras:diagonal}.

\textbf{13.} Receta: pon uno a continuación del otro dos \omterm{def:g6:lines:objects}{segmentos} de longitudes
$p = 1$ y $q = n$; traza la semicircunferencia cuyo \omterm{def:g4:geometry:circle}{diámetro} es el
\omterm{def:g6:lines:objects}{segmento} entero (de longitud $n + 1$); levanta la perpendicular en el
punto de unión; corta a la semicircunferencia en un punto cuya
distancia al punto de unión es $\sqrt{1 \times n} = \sqrt n$. Para
$\sqrt5$: toma $p = 1$ y $q = 5$ (una semicircunferencia de \omterm{def:g4:geometry:circle}{diámetro}
$6$).

\textbf{14.} $A$ está en la circunferencia de centro $O$, luego
$OA = \frac{p+q}{2}$, el \omterm{def:g3:shapes:shapes}{radio}. Si $H \neq O$, el triángulo
$AHO$ es rectángulo en $H$ con hipotenusa $[OA]$, y un cateto es
más corto que la hipotenusa: $AH < OA$. Si $H = O$, entonces
$AH = OA$ exactamente. En todos los casos $AH \leq \frac{p+q}{2}$, con
igualdad exactamente cuando $H$ es el centro --- es decir, cuando
$p = q$.

\textbf{15.} Elevando al cuadrado $AH \leq \frac{p+q}{2}$ (los dos miembros
son positivos) y usando $AH^2 = pq$:
\[
p \times q \leq \left(\frac{p+q}{2}\right)^2 ,
\]
con igualdad exactamente para $p = q$. Nueva demostración algebraica, con las
identidades notables (\cref{pb:g8:equations:1}):
\[
\left(\frac{p+q}{2}\right)^2 - pq
= \frac{p^2 + 2pq + q^2 - 4pq}{4}
= \frac{p^2 - 2pq + q^2}{4}
= \left(\frac{p-q}{2}\right)^2 ,
\]
un cuadrado, luego nunca negativo --- y \omterm{def:g1:counting:zero}{cero} exactamente cuando
$p = q$. La \omterm{pb:g8:cosine:1}{media geométrica} de dos números nunca supera su
semisuma.
\end{solution}
