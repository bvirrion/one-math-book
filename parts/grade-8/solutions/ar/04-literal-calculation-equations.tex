\begin{solution}{exo:g8:equations:1}
$5(2x + 3) = 10x + 15$.

$-3(4x - 1) = -12x + 3$.

$2(3x - 2) + 4(x + 5) = 6x - 4 + 4x + 20 = 10x + 16$.
\end{solution}

\begin{solution}{exo:g8:equations:2}
$(x+2)(x+7) = x^2 + 7x + 2x + 14 = x^2 + 9x + 14$.

$(3x+1)(2x-5) = 6x^2 - 15x + 2x - 5 = 6x^2 - 13x - 5$.

$(x-4)(x-6) = x^2 - 6x - 4x + 24 = x^2 - 10x + 24$.
\end{solution}

\begin{solution}{exo:g8:equations:3}
$4 \times 3 - 5 = 7$: نعم. \quad
و $2 \times 3 + 1 = 7$ و $3 + 4 = 7$: نعم. \quad
و $3^2 = 9$ و $6 \times 3 - 9 = 9$: نعم --- فالقيمة $x = 3$ تحلّ الثلاث
كلها.
\end{solution}

\begin{solution}{exo:g8:equations:4}
$x + 9 = 4$: $x = -5$ (والتحقّق: $-5 + 9 = 4$).

$5x = -35$: $x = -7$.

$\frac x3 = 8$: $x = 24$.

$2x - 7 = 13$: $2x = 20$، و $x = 10$ (والتحقّق: $20 - 7 = 13$).
\end{solution}

\begin{solution}{exo:g8:equations:5}
$6x + 5 = 2x + 25$: $4x = 20$، إذن $x = 5$.

$9 - 4x = x - 11$: $20 = 5x$، إذن $x = 4$.

$3(x + 2) = 2x + 11$: $3x + 6 = 2x + 11$، إذن $x = 5$.
\end{solution}

\begin{solution}{exo:g8:equations:6}

$4(28 - 3) = 100$ و $5(14 + 6) = 100$.
\end{solution}

\begin{solution}{exo:g8:equations:8}
العرض $x$، والطول $x + 5$؛ \omterm{def:g6:measure:perimeter}{والمحيط}
$2\bigl(x + (x + 5)\bigr) = 4x + 10 = 58$، إذن $4x = 48$ و $x = 12$:
فالعرض $12$ سم، والطول $17$ سم (والتحقّق:
$2 \times (12 + 17) = 58$).
\end{solution}

\begin{solution}{exo:g8:equations:9}

\begin{solution}{exo:g8:equations:10}
$3x + 4 > 19$: $3x > 15$، إذن $x > 5$.

$7 - 2x \geq 1$: $-2x \geq -6$، والقسمة على $-2$ تعكس:
$x \leq 3$.

$5(x-1) < 2x + 7$: $5x - 5 < 2x + 7$، إذن $3x < 12$ و $x < 4$.
\end{solution}

\begin{solution}{exo:g8:equations:11}
بعد $n$ شهرًا، تكون عند منى $140 + 15n$ وعند طارق $200 + 10n$. وتتقدّم منى
حين
\[
140 + 15n > 200 + 10n
\ \Longleftrightarrow\
5n > 60
\ \Longleftrightarrow\
n > 12 :
\]
أي اِبتداءً من الشهر $13$.
\end{solution}

\begin{solution}{exo:g8:equations:12}
اِضرب الطرفين في $20$:
\[
5(x + 3) = 4(2x - 1)
\ \Longrightarrow\
5x + 15 = 8x - 4
\ \Longrightarrow\
19 = 3x
\ \Longrightarrow\
x = \frac{19}{3}.
\]
والتحقّق: $\frac{19/3 + 3}{4} = \frac{28/3}{4} = \frac{28}{12} =
\frac73$، و $\frac{2 \times 19/3 - 1}{5} = \frac{35/3}{5} =
\frac{35}{15} = \frac73$. متساويان: فالحلّ هو $\frac{19}{3}$.
\end{solution}

\begin{solution}{pb:g8:equations:1}
\textbf{1.} التوزيعية المضاعفة
(\cref{thm:g8:equations:expand}) مع $c = a$ و $d = b$:
\[
(a + b)(a + b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2 ,
\]
لأن $ab$ و $ba$ هما \omterm{def:g2:mult:def}{الجداء} نفسه.

\textbf{2.} النشر نفسه، مع اِنتباه للإشارات:
\[
(a - b)(a - b) = a^2 - ab - ba + b^2 = a^2 - 2ab + b^2 ,
\]
\[
(a + b)(a - b) = a^2 - ab + ba - b^2 = a^2 - b^2 :
\]
وهذه المرة يتلاشى الحدّان المتقاطعان $-ab$ و $+ba$ بدل أن
يتضاعفا.

\textbf{3.} المربع الذي طول ضلعه $a + b$ ينقطع إلى أربع قطع:
مربع $a \times a$ (وهو $a^2$)، ومربع $b \times b$ (وهو
$b^2$)، و\emph{مستطيلان} $a \times b$ (وهما معًا
$2ab$). والمتطابقة $(a+b)^2 = a^2 + 2ab + b^2$ هي القول
بأن المساحات الأربع تملأ المربع الكبير --- وهي صورة
التوزيعية المضاعفة مع شقّ الطرفين بالطريقة نفسها.

\textbf{4.} بالمتطابقات:
\begin{align*}
101^2 &= (100 + 1)^2 = 10\,000 + 2 \times 100 + 1 = 10\,201, \\
99^2 &= (100 - 1)^2 = 10\,000 - 200 + 1 = 9\,801, \\
49 \times 51 &= (50 - 1)(50 + 1) = 50^2 - 1^2 = 2\,499 .
\end{align*}

\textbf{5.} من أجل $a = 3$ و $b = 4$: $(3 + 4)^2 = 49$ لكن
$3^2 + 4^2 = 9 + 16 = 25$. فصيغة زينة تنسى المستطيلين
$a \times b$ في الصورة --- أي المقدار الناقص
$2ab = 24$، وفعلًا $25 + 24 = 49$.

\textbf{6.} $1$؛ و $1 + 3 = 4$؛ و $1 + 3 + 5 = 9$؛
و $1 + 3 + 5 + 7 = 16$؛ و $1 + 3 + 5 + 7 + 9 = 25$. والنتائج هي
المربعات التامّة $1^2, 2^2, 3^2, 4^2, 5^2$.

\textbf{7.} تبدأ \omterm{def:g3:numbers:evenodd}{الأعداد الفردية} من $1$ وتصعد بخطوات مقدارها
$2$: فذو الرتبة $n$ يُبلَغ بعد $n - 1$ خطوة مقدارها $2$ من
$1$، إذن يساوي $1 + 2(n - 1) = 2n - 1$. والتحقّق: $n = 1, 2, 3$
تعطي $1, 3, 5$.

\textbf{8.} بالمتطابقة الأولى،
\[
(n + 1)^2 - n^2 = n^2 + 2n + 1 - n^2 = 2n + 1 :
\]
\omterm{ex:g1:subtraction:difference}{فالفرق} بين مربعين متتاليين هو بالضبط
\omterm{def:g3:numbers:evenodd}{العدد الفردي} ذو الرتبة $(n+1)$ (السؤال 7 مع $n + 1$ مكان $n$:
$2(n+1) - 1 = 2n + 1$).

\textbf{9.} اِبنِ المربعات واحدًا بعد الآخر. اِبدأ بخلية
واحدة: $1 = 1^2$. وللانتقال من المربع ذي الضلع $1$ إلى
المربع ذي الضلع $2$، أضِف $2 \times 1 + 1 = 3$ خلايا (السؤال
8): فتجد $1 + 3 = 2^2$. وللانتقال إلى الضلع $3$، أضِف $2 \times 2 + 1 = 5$
خلايا: فتجد $1 + 3 + 5 = 3^2$. وكل \omterm{def:g3:numbers:evenodd}{عدد فردي} جديد هو بالضبط
الطبقة على شكل حرف L التي تكمل المربع التالي، إذن بعد إضافة
\omterm{def:g3:numbers:evenodd}{الأعداد الفردية} $n$ الأولى يكتمل المربع ذو الضلع $n$:
$1 + 3 + \dots + (2n - 1) = n^2$.

\textbf{10.} \omterm{def:g3:numbers:evenodd}{الأعداد الفردية} من $1$ إلى $99$ هي الأولى $50$
منها (\omterm{def:g8:equations:def}{فالمعادلة} $2n - 1 = 99$ تعطي $n = 50$)، إذن
\[
1 + 3 + 5 + \dots + 99 = 50^2 = 2\,500 .
\]
\omterm{def:g1:addition:def}{والمجموع} $1 + 3 + \dots + (2n - 1) = 400 = 20^2$ يستلزم $n = 20$:
\omterm{def:g3:numbers:evenodd}{فالأعداد الفردية} \emph{العشرون} الأولى (من $1$ إلى $39$) مجموعها
$400$.

\textbf{11.} $x^2 + 6x + 9 = x^2 + 2 \times 3 \times x + 3^2 =
(x + 3)^2$: وهي المتطابقة الأولى مع $a = x$ و $b = 3$. إذن تُقرأ
\omterm{def:g8:equations:def}{المعادلة} $(x + 3)^2 = 0$. والمربع لا ينعدم إلا حين ينعدم
العدد نفسه (\cref{thm:g8:negprod:rules}: فالعدد غير المعدوم
مسافته غير معدومة، إذن مسافة مربعه غير معدومة
أيضًا). ومنه $x + 3 = 0$: فالحلّ الوحيد هو $x = -3$.

\textbf{12.} حسب \cref{thm:g8:negprod:rules}، تكون مسافة
\omterm{def:g2:mult:def}{الجداء} \omterm{def:g2:mult:def}{جداء} المسافتين. فإذا كان العاملان
غير معدومين، كانت المسافتان غير معدومتين، فتكون مسافة \omterm{def:g2:mult:def}{الجداء}
غير معدومة: فلا يمكن \omterm{def:g2:mult:def}{للجداء} أن يكون $0$. ومنه \omterm{def:g2:mult:def}{فالجداء} المعدوم
يفرض عاملًا معدومًا واحدًا على الأقلّ. ومن أجل $(x - 2)(x + 5) = 0$: إما
$x - 2 = 0$ وإما $x + 5 = 0$، فيعطيان الحلّين $x = 2$ و
$x = -5$.

\textbf{13.} تصير $x^2 = 49$ هي $x^2 - 49 = 0$، والمتطابقة
الثالثة تعمّلها:
\[
x^2 - 49 = x^2 - 7^2 = (x - 7)(x + 7) = 0 .
\]
وبالسؤال 12، $x = 7$ أو $x = -7$: أي \emph{حلّان}. وكل
\omterm{def:g8:equations:def}{معادلة} حُلّت حتى الآن في هذا الفصل كانت من الدرجة الأولى ولها
حلّ واحد بالضبط؛ أما المربع في \omterm{def:g8:equations:def}{المعادلة} فقد ينتج
حلّين.

\textbf{14.} اُنقل كل شيء إلى الطرف الأيسر:
$x^2 - 6x + 9 = 0$، وتعرّف على المتطابقة الثانية:
$x^2 - 2 \times 3 \times x + 3^2 = (x - 3)^2$. إذن \omterm{def:g8:equations:def}{المعادلة} هي
$(x - 3)^2 = 0$، وهي تفرض $x - 3 = 0$ (كما في السؤال 11):
فالقيمة $x = 3$ هي الحلّ الوحيد لا غير --- وقد وجده تحقّق
\cref{exo:g8:equations:3}، والتعميل يبرهن
على أنه لا حلول غيره.

\textbf{15.} بالمتطابقة الثالثة،
\[
2026^2 - 2025^2 = (2026 - 2025)(2026 + 2025)
= 1 \times 4051 = 4\,051 .
\]
وعمومًا $(n+1)^2 - n^2 = (n + 1 - n)(n + 1 + n) = 2n + 1$: أي إن
\omterm{ex:g1:subtraction:difference}{فرق} مربعين متتاليين هو \omterm{def:g1:addition:def}{مجموع} العددين
--- وهو $2n + 1$ نفسه الذي في السؤال 8، محصولًا عليه الآن من
المتطابقة الثالثة بدل الأولى.
\end{solution}
