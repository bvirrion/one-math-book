\section*{الفصل \ref{ch:g8:cosine} --- المثلث القائم الزاوية وجيب التمام}

\begin{solution}{exo:g8:cosine:1}
المتوسط الصادر من \omterm{def:g3:shapes:rightangle}{الزاوية القائمة}: نصف الوتر، أي $6.5$ سم
(\cref{thm:g8:cosine:circle}). \omterm{pb:g6:symmetry:1}{والدائرة المحيطة} مركزها
منتصف الوتر ونصف قطرها $6.5$ سم.
\end{solution}

\begin{solution}{exo:g8:cosine:2}
الزاوية عند $A$ قائمة (\omterm{def:g6:lines:objects}{فالقطعة} $[BC]$ \omterm{def:g4:geometry:circle}{قطر}: النقطة 2 من
\cref{thm:g8:cosine:circle}). والقياسات تحقّق
$AB^2 + AC^2 \approx 64 = BC^2$، في حدود دقّة القياس.
\end{solution}

\begin{solution}{exo:g8:cosine:3}
الوتر: $[EF]$ (المقابل \omterm{def:g3:shapes:rightangle}{للزاوية القائمة} $D$). والمجاور
للزاوية $\widehat E$: $[ED]$. والمقابل: $[DF]$. إذن
$\cos\widehat E = \dfrac{ED}{EF}$.
\end{solution}

\begin{solution}{exo:g8:cosine:4}
\emph{1.} المجاور $= 10 \cos 35^\circ \approx 8.2$.

\emph{2.} الوتر $= \dfrac{6}{\cos 50^\circ} \approx
\dfrac{6}{0.643} \approx 9.3$.
\end{solution}

\begin{solution}{exo:g8:cosine:5}
$\cos\theta = \frac{8}{10} = 0.8$، إذن
$\theta = \cos^{-1}(0.8) \approx 37^\circ$.
\end{solution}

\begin{solution}{exo:g8:cosine:6}
جيب التمام $\cos\theta$ ضلع قائمة مقسوم على الوتر، والوتر
هو \emph{أطول} أضلاع \omterm{def:g6:shapes:triangles}{المثلث القائم الزاوية}: \omterm{def:g3:division:remainder}{فخارج القسمة}
أصغر دائمًا من $1$. والقيمة $1.2$ كانت ستعني ضلع قائمة أطول
من الوتر --- وهذا مستحيل.
\end{solution}

\begin{solution}{exo:g8:cosine:7}
المسافة الأفقية $= 25 \cos 12^\circ \approx 25 \times 0.978 \approx
24.5$ م.
\end{solution}

\begin{solution}{exo:g8:cosine:8}
جيب التمام يتناقص كلما كبرت الزاوية
(\cref{prop:g8:cosine:bounds}):
\[
\cos 70^\circ < \cos 45^\circ < \cos 20^\circ .
\]
(وبالآلة الحاسبة: $0.342 < 0.707 < 0.940$.)
\end{solution}

\begin{solution}{exo:g8:cosine:9}
المسافة على الأرض مجاورة للزاوية $\theta$:
$\cos 68^\circ = \frac{1.2}{L}$، إذن
$L = \frac{1.2}{0.375} = 3.2$ م. والارتفاع (بفيثاغورس):
$\sqrt{3.2^2 - 1.2^2} = \sqrt{10.24 - 1.44} = \sqrt{8.8} \approx 3.0$
م.
\end{solution}

\begin{solution}{exo:g8:cosine:10}
\emph{1.} النقطة $A$ على دائرة قطرها $[BC]$: \omterm{def:g3:shapes:rightangle}{فالزاوية قائمة} في $A$
(\cref{thm:g8:cosine:circle}).

\emph{2.} $BC = 9$ سم (ضعف \omterm{def:g3:shapes:shapes}{نصف القطر}). وبفيثاغورس:
$AC = \sqrt{9^2 - 5.4^2} = \sqrt{81 - 29.16} = \sqrt{51.84} = 7.2$
سم. و $\cos\widehat B = \dfrac{AB}{BC} = \dfrac{5.4}{9} = 0.6$.
\end{solution}

\begin{solution}{exo:g8:cosine:11}
في \omterm{def:g6:shapes:triangles}{المثلث المتساوي الأضلاع} الذي طول ضلعه $1$، يقع الارتفاع الصادر من رأس
على \emph{منتصف} الضلع المقابل (وهو \omterm{def:g6:symmetry:axis}{محور
تماثل}). ونصف \omterm{def:g6:shapes:triangles}{المثلث قائم الزاوية} وتره $1$
(أي ضلع كامل)، وفيه زاوية $60^\circ$ عند رأس القاعدة، و
ضلع مجاور $\frac12$ (أي نصف القاعدة). ومنه
\[
\cos 60^\circ = \frac{1/2}{1} = \frac12 .
\]
\end{solution}

\begin{solution}{pb:g8:cosine:1}
\textbf{1.} في المثلث $ABH$، \omterm{def:g6:shapes:triangles}{القائم الزاوية} في $H$، يعطي \omterm{def:g1:addition:def}{مجموع}
الزوايا (\cref{thm:g7:triangles:sum})
$\widehat B + 90^\circ + \widehat{BAH} = 180^\circ$، إذن
$\widehat{BAH} = 90^\circ - \widehat B$. وفي المثلث $ABC$،
\omterm{def:g6:shapes:triangles}{القائم الزاوية} في $A$:
$\widehat B + \widehat C + 90^\circ = 180^\circ$، إذن
$\widehat C = 90^\circ - \widehat B$. ومنه
$\widehat{BAH} = \widehat{ACB}$: فكلا المثلثين الصغيرين يكرّر
الزوايا $\widehat B$ و $\widehat C$ و $90^\circ$ التي للأصل.

\textbf{2.} في $ABC$ (\omterm{def:g6:shapes:triangles}{القائم الزاوية} في $A$)، الضلع المجاور
للزاوية $\widehat B$ هو $AB$ والوتر هو $BC$:
إذن $\cos\widehat B = \frac{AB}{BC}$. وفي $ABH$ (\omterm{def:g6:shapes:triangles}{القائم الزاوية} في $H$)،
الضلع المجاور للزاوية $\widehat B$ هو $BH$ والوتر هو
$AB$: إذن $\cos\widehat B = \frac{BH}{AB}$. والنسبة تتوقف على
الزاوية وحدها (\cref{def:g8:cosine:def})، إذن
\[
\frac{AB}{BC} = \frac{BH}{AB} ;
\]
وبضرب الطرفين في $BC \times AB$
(\cref{thm:g8:equations:balance}):
$AB^2 = BH \times BC$.

\textbf{3.} المثلثان $ABC$ (\omterm{def:g6:shapes:triangles}{القائم الزاوية} في $A$) و $ACH$
(\omterm{def:g6:shapes:triangles}{القائم الزاوية} في $H$) يشتركان في الزاوية $\widehat C$. والمجاور
على الوتر في كل منهما:
\[
\cos\widehat C = \frac{AC}{CB} = \frac{CH}{AC}
\qquad\Longrightarrow\qquad
AC^2 = CH \times CB .
\]

\textbf{4.} تُقرأ $AB^2 = BH \times BC$ هكذا: $9 = BH \times 5$، إذن
$BH = 1.8$؛ و $16 = CH \times 5$ يعطي $CH = 3.2$. والتحقّق:
$BH + HC = 1.8 + 3.2 = 5 = BC$ --- كما يجب، لأن $H$ تقع على
الوتر بين $B$ و $C$.

\textbf{5.} بقسمة كل علاقة على $BC$:
\[
BH = \frac{AB^2}{BC},
\qquad
CH = \frac{AC^2}{BC} .
\]
إذن $BH$ و $CH$ متناسبان مع $AB^2$ و $AC^2$: فقدم
الارتفاع تشقّ الوتر بنسبة مربعَي
ضلعَي القائمة ($1.8 : 3.2 = 9 : 16$ في السؤال 4).

\textbf{6.} بجمع العلاقتين وتعميل $BC$
(بالتوزيعية):
\[
AB^2 + AC^2 = BH \times BC + CH \times BC
= (BH + CH) \times BC = BC \times BC = BC^2 .
\]
وهذه هي \cref{thm:g8:pythagoras:direct}، محصولًا عليها من جيب التمام
وحده --- وهو برهان مختلف حقًّا عن أحجية مساحات
المثلثات الأربعة في \cref{exo:g8:pythagoras:11}.

\textbf{7.} بفيثاغورس في $ABH$ (\omterm{def:g6:shapes:triangles}{القائم الزاوية} في $H$، والوتر
$[AB]$): $AH^2 = AB^2 - BH^2$. وبتعويض $AB^2$ بالمقدار
$BH \times BC$ (السؤال 2) وتعميل $BH$:
\[
AH^2 = BH \times BC - BH^2
= BH \times (BC - BH)
= BH \times HC .
\]

\textbf{8.} بضلعَي القائمة قاعدةً واِرتفاعًا، تكون \omterm{def:g6:measure:area}{مساحة} $ABC$
هي $\frac{AB \times AC}{2}$؛ وبالوتر $[BC]$ قاعدةً،
يكون الارتفاع هو $AH$ بالضبط، إذن \omterm{def:g6:measure:area}{المساحة} أيضًا
$\frac{BC \times AH}{2}$ (\cref{thm:g7:areas:triangle}). وبالمساواة
والمضاعفة: $AB \times AC = BC \times AH$.

\textbf{9.} $AH = \frac{AB \times AC}{BC} = \frac{3 \times 4}{5}
= 2.4$. والتحقّق من السؤال 7:
$AH^2 = 2.4^2 = 5.76$ و
$BH \times HC = 1.8 \times 3.2 = 5.76$. متساويان.

\textbf{10.} $AH^2 = BH \times HC = 2 \times 8 = 16$، إذن
$AH = \sqrt{16} = 4$ سم.

\textbf{11.} النقطة $A$ تقع على الدائرة التي قطرها $[BC]$
(وتختلف عن $B$ و $C$)، إذن بالنقطة 2 من
\cref{thm:g8:cosine:circle} يكون المثلث $ABC$ \omterm{def:g6:shapes:triangles}{قائم الزاوية} في
$A$. وبالإنشاء $(AH) \perp (BC)$ مع وقوع $H$ بين $B$ و
$C$: \omterm{def:g6:lines:objects}{فالقطعة} $[AH]$ هي بالضبط اِرتفاع السؤال 7،
و $AH^2 = BH \times HC = p \times q$.

\textbf{12.} $AH^2 = 1 \times 2 = 2$: فالشكل ينشئ، بالمسطرة
والفرجار وحدهما، \omterm{def:g6:lines:objects}{قطعة} مربعها $2$ --- أي
\omterm{def:g4:geometry:circle}{قطر} مربع الوحدة، $\sqrt2 \approx 1.414$، في
\cref{ex:g8:pythagoras:diagonal}.

\textbf{13.} الوصفة: ضع طرفًا لطرف \omterm{def:g6:lines:objects}{قطعتين} طولاهما
$p = 1$ و $q = n$؛ واُرسم نصف الدائرة التي قطرها
\omterm{def:g6:lines:objects}{القطعة} كلها (وطولها $n + 1$)؛ وأقِم العمود عند
نقطة الوصل؛ فيلتقي نصف الدائرة في نقطة
بُعدها عن نقطة الوصل $\sqrt{1 \times n} = \sqrt n$. ومن أجل
$\sqrt5$: خذ $p = 1$ و $q = 5$ (فنصف دائرة قطرها
$6$).

\textbf{14.} النقطة $A$ على الدائرة التي مركزها $O$، إذن
$OA = \frac{p+q}{2}$، وهو \omterm{def:g3:shapes:shapes}{نصف القطر}. فإذا كان $H \neq O$، كان المثلث
$AHO$ \omterm{def:g6:shapes:triangles}{قائم الزاوية} في $H$ ووتره $[OA]$، وضلع القائمة
أقصر من الوتر: أي $AH < OA$. وإذا كان $H = O$، فإن
$AH = OA$ بالضبط. وفي كل الحالات $AH \leq \frac{p+q}{2}$، مع
التساوي بالضبط حين تكون $H$ هي المركز --- أي حين
$p = q$.

\textbf{15.} بتربيع $AH \leq \frac{p+q}{2}$ (والطرفان
موجبان) واِستعمال $AH^2 = pq$:
\[
p \times q \leq \left(\frac{p+q}{2}\right)^2 ,
\]
مع التساوي بالضبط من أجل $p = q$. وإعادة البرهنة جبريًّا، بالمتطابقات
الشهيرة (\cref{pb:g8:equations:1}):
\[
\left(\frac{p+q}{2}\right)^2 - pq
= \frac{p^2 + 2pq + q^2 - 4pq}{4}
= \frac{p^2 - 2pq + q^2}{4}
= \left(\frac{p-q}{2}\right)^2 ,
\]
أي مربع، ومنه فهو ليس سالبًا أبدًا --- وينعدم بالضبط حين
$p = q$. \omterm{pb:g8:cosine:1}{فالمتوسط الهندسي} لعددين لا يتجاوز أبدًا
نصف مجموعهما.
\end{solution}
