\section*{الفصل \ref{ch:g8:negprod} --- ضرب الأعداد السالبة}

\begin{solution}{exo:g8:negprod:1}
$(-8) \times (+7) = -56$؛ \quad
$(-9) \times (-6) = +54$؛ \quad
$(+12) \times (-0.5) = -6$؛ \quad
$(-1) \times (-1) \times (-1) = -1$ (ثلاثة عوامل سالبة: \omterm{def:g3:numbers:evenodd}{عدد فردي}).
\end{solution}

\begin{solution}{exo:g8:negprod:2}

$(-8) \div (-16) = +0.5$؛ \quad
$\dfrac{45}{-5} = -9$؛ \quad
$\dfrac{-7.2}{-8} = +0.9$.
\end{solution}

\begin{solution}{exo:g8:negprod:3}
ثلاثة عوامل سالبة: \emph{سالب}. \quad
و $(-1)^{10}$: عشرة عوامل سالبة، \omterm{def:g3:numbers:evenodd}{عدد زوجي}: \emph{موجب} (فهو يساوي
$1$). \quad
و $(-2)^7$: سبعة عوامل، \omterm{def:g3:numbers:evenodd}{عدد فردي}: \emph{سالب}. \quad
و $(-4) \times 0 \times (-6) = 0$: لا موجب ولا سالب.
\end{solution}

\begin{solution}{exo:g8:negprod:4}
$(-2)^4 = +16$؛ \quad و $-2^4 = -16$ (فالناقص غير مربّع)؛ \quad
و $(-3)^3 = -27$؛ \quad و $(-1)^{2026} = +1$ (أُسّ زوجي).
\end{solution}

\begin{solution}{exo:g8:negprod:5}
$7 + 2 \times (-6) = 7 + (-12) = -5$.

$(-4) \times (5 - 9) = (-4) \times (-4) = +16$.

$18 \div (-3) - (-2) = -6 + 2 = -4$.
\end{solution}

\begin{solution}{exo:g8:negprod:6}
$\dfrac{(-5) \times (+8)}{-4} = \dfrac{-40}{-4} = +10$.

$\dfrac{(-3) \times (-6)}{(-2) \times (+9)} = \dfrac{+18}{-18} = -1$.
\end{solution}

\begin{solution}{exo:g8:negprod:7}
من أجل $x = -3$: \ $5x = -15$؛ \quad و $x^2 = (-3)^2 = 9$؛ \quad
و $-x^2 = -9$؛ \quad و $2x^2 + 4x = 18 - 12 = 6$؛ \quad
و $(2x)^2 = (-6)^2 = 36$.
\end{solution}

\begin{solution}{exo:g8:negprod:8}

$10 \div (-4) = -2.5$؛ \quad
$(-5) \times \frac15 = -1$، إذن العدد الناقص هو
$+\frac15 = 0.2$.
\end{solution}

\begin{solution}{exo:g8:negprod:9}
\emph{1. خطأ}: \omterm{def:g2:mult:def}{فالجداء} $0.5 \times 0.5 = 0.25$ أصغر من
$0.5 + 0.5 = 1$ (أو: $2 \times 3 = 6 > 5$، لكن الادّعاء يجب أن يصحّ
دائمًا).

\emph{2. صواب}: فالمربع \omterm{def:g2:mult:def}{جداء} عددين لهما الإشارة
نفسها، ومنه فهو موجب أو معدوم.

\emph{3. خطأ}: \omterm{def:g2:mult:def}{فالجداء} $(-2) \times (-3) \times 4 = +24$ موجب مع
عاملين سالبين.
\end{solution}

\begin{solution}{exo:g8:negprod:10}
ليكن $r$ عدد الأجوبة الصحيحة؛ فتكون $20 - r$ خاطئة، ومنه
\[
5r - 3(20 - r) = 36
\ \Longrightarrow\
5r - 60 + 3r = 36
\ \Longrightarrow\
8r = 96
\ \Longrightarrow\
r = 12 .
\]
أي اِثنا عشر جوابًا صحيحًا (وثمانية خاطئة: والتحقّق،
$60 - 24 = 36$).
\end{solution}

\begin{solution}{exo:g8:negprod:11}
\omterm{def:g1:counting:zero}{الصفر} في أيّ شيء \omterm{def:g1:counting:zero}{صفر}، إذن $(-1) \times 0 = 0$. وبكتابة
$0 = 1 + (-1)$ والتوزيع:
\[
0 = (-1) \times \bigl(1 + (-1)\bigr)
= (-1) \times 1 + (-1) \times (-1)
= -1 + (-1) \times (-1).
\]
العدد $(-1) \times (-1)$ مجموعًا إلى $-1$ يعطي $0$: فلا بدّ أن يكون
\omterm{def:g7:negatives:def}{معاكس} $-1$، أي $+1$. والتوزيعية لا تترك خيارًا
آخر.
\end{solution}

\begin{solution}{pb:g8:negprod:1}
\textbf{1.} بما أن $0 = 0 + 0$، تعطي التوزيعية
(\cref{thm:g7:priorities:distributivity})
\[
0 \times a = (0 + 0) \times a = 0 \times a + 0 \times a .
\]
إذن جمع $0 \times a$ إلى نفسه لا يغيّر شيئًا --- والعدد الوحيد
الذي يمكن جمعه دون تغيير شيء هو $0$. (ومباشرةً:
اِطرح $0 \times a$ من الطرفين.) ومنه $0 \times a = 0$.

\textbf{2.} بالنشر، ثم باستعمال $1 + (-1) = 0$ والسؤال 1:
\[
a + (-1) \times a
= 1 \times a + (-1) \times a
= \bigl(1 + (-1)\bigr) \times a
= 0 \times a = 0 .
\]
إذن $(-1) \times a$ هو العدد الذي، إذا جُمع إلى $a$، أعطى $0$:
وذلك بالضبط \omterm{def:g7:negatives:def}{معاكس} $a$، ومنه $(-1) \times a = -a$.
وتمثّل \cref{exo:g8:negprod:11} الحالة الخاصّة $a = -1$: فهناك
$(-1) \times (-1) = -(-1) = +1$.

\textbf{3.} اُكتب $-a = (-1) \times a$ (السؤال 2) وأعِد تجميع
العوامل:
\[
(-a) \times b
= \bigl((-1) \times a\bigr) \times b
= (-1) \times (a \times b)
= -(a \times b) .
\]

\textbf{4.} طبّق السؤال 3 مرتين (مرة على كل عامل):
\[
(-a) \times (-b)
= -\bigl(a \times (-b)\bigr)
= -\bigl(-(a \times b)\bigr)
= a \times b ,
\]
لأن \omterm{def:g7:negatives:def}{معاكس} \omterm{def:g7:negatives:def}{معاكس} عدد هو العدد نفسه.

\textbf{5.} اُكتب $a = +d$ أو $a = -d$ و $b = +e$ أو $b = -e$،
حيث $d$ و $e$ المسافتان إلى \omterm{def:g1:counting:zero}{الصفر}. والحالات الأربع:
\[
(+d) \times (+e) = de, \quad
(+d) \times (-e) = -de, \quad
(-d) \times (+e) = -de, \quad
(-d) \times (-e) = +de,
\]
والاثنتان الوسطيان بالسؤال 3، والأخيرة بالسؤال 4. وفي كل حالة
تكون مسافة النتيجة $d \times e$، أي \omterm{def:g2:mult:def}{جداء}
المسافتين، والإشارة هي المعلنة في جدول
\cref{thm:g8:negprod:rules}: الإشارتان نفسهما موجب، وإشارتان متعاكستان
سالب.

\textbf{6.} $(-1)^2 = +1$، و $(-1)^3 = -1$، و $(-1)^4 = +1$،
و $(-1)^5 = -1$. وعمومًا $(-1)^n$ \omterm{def:g2:mult:def}{جداء} $n$ عاملًا سالبًا،
مسافة كل منها $1$: وحسب \cref{prop:g8:negprod:many} يكون
$+1$ حين يكون $n$ زوجيًّا و $-1$ حين يكون $n$ فرديًّا.

\textbf{7.} هناك عشرة عوامل سالبة --- وهو \omterm{def:g3:numbers:evenodd}{عدد زوجي} --- إذن
\omterm{def:g2:mult:def}{الجداء} \emph{موجب} (\cref{prop:g8:negprod:many}). ومسافته
$1 \times 2 \times 3 \times \dots \times 10$.

\textbf{8.} بالسؤال 4، $(-a)^2 = (-a) \times (-a) = a \times a =
a^2$. ثم بالسؤال 3،
\[
(-a)^3 = (-a)^2 \times (-a) = a^2 \times (-a) = -(a^2 \times a)
= -a^3 .
\]
فتختفي إشارة الناقص من أجل الأُسُس الزوجية وتصمد من أجل
الفردية، بموافقة \cref{ex:g8:negprod:powers}.

\textbf{9.} إذا كان $a = 0$ فإن $a^2 = 0$. وإلا كان للعددين $a$ و $a$
الإشارة \emph{نفسها}، إذن $a^2 = a \times a$ موجب
(\cref{thm:g8:negprod:rules}). فالمربع إذن ليس سالبًا
أبدًا؛ وبما أن $-4$ سالب، فلا يوجد \omterm{def:g7:negatives:def}{عدد نسبي} $x$ يحقّق
$x^2 = -4$.

\textbf{10.} زينة على صواب. فالأُسّان $2026$ و $2027$ لهما
زوجيتان متعاكستان، إذن $(-1)^{2026} = +1$ و $(-1)^{2027} = -1$،
ومجموعهما $0$. وعمومًا، من بين أُسّين متتاليين واحد
زوجي وواحد فردي، إذن القوّتان $+1$ و $-1$ بترتيب
ما: عددان \omterm{def:g7:negatives:def}{متعاكسان} مجموعهما $0$ دائمًا.

\textbf{11.} $A_3 = 1 - 2 + 3 = 2$، ثم
$A_4 = A_3 - 4 = -2$، و $A_5 = A_4 + 5 = 3$، و $A_6 = A_5 - 6 = -3$،
و $A_7 = A_6 + 7 = 4$. والتخمين: $A_n = -\frac{n}{2}$ من أجل $n$ الزوجي،
و $A_n = \frac{n+1}{2}$ من أجل $n$ الفردي.

\textbf{12.} من أجل $n$ الزوجي تتزاوج الحدود تزاوجًا كاملًا:
\[
A_n = (1 - 2) + (3 - 4) + \dots + \bigl((n-1) - n\bigr),
\]
وكل زوج يساوي $-1$. وعدد الأزواج $\frac{n}{2}$ (حدّان
في كل زوج، و $n$ حدًّا في المجموع)، إذن
$A_n = \frac{n}{2} \times (-1) = -\frac{n}{2}$.

\textbf{13.} من أجل $n$ الفردي، يكون \omterm{def:g1:addition:def}{المجموع} $A_n$ هو \omterm{def:g1:addition:def}{المجموع} $A_{n-1}$ للحدود
$n - 1$ الأولى، زائد الحدّ الأخير، وهو $+n$ لأن $n$
فردي. وبما أن $n - 1$ زوجي، يعطي السؤال 12
\[
A_n = A_{n-1} + n = -\frac{n-1}{2} + n
= \frac{-(n-1) + 2n}{2} = \frac{n+1}{2} .
\]

\textbf{14.} $A_{100} = -\frac{100}{2} = -50$، \quad
و $A_{101} = \frac{102}{2} = 51$، \quad
و $A_{2026} = -\frac{2026}{2} = -1013$.

\textbf{15.} لا مفاجأة: فالانتقال من $A_{100}$ إلى $A_{101}$ يضيف
حدًّا واحدًا بالضبط، هو $+101$، إذن كان \omterm{ex:g1:subtraction:difference}{الفرق} لا بدّ أن يكون $101$.
وللسبب نفسه، $A_{2027} - A_{2026} = +2027$ --- دون
حساب $A_{2027}$ البتة.
\end{solution}
