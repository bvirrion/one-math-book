\begin{solution}{exo:g8:pythagoras:4}

\begin{solution}{exo:g8:pythagoras:10}

\begin{solution}{exo:g8:pythagoras:11}
\emph{1.} مباشرةً: $(a + b)^2$. وعلى صورة قطع: أربعة مثلثات \omterm{def:g6:measure:area}{مساحة} كل منها
$\frac{ab}{2}$، زائد المربع المائل $c^2$:
\[
(a + b)^2 = 4 \times \frac{ab}{2} + c^2 = 2ab + c^2 .
\]

\emph{2.} بنشر الطرف الأيسر:
$(a+b)^2 = a^2 + 2ab + b^2$. ومنه
\[
a^2 + 2ab + b^2 = 2ab + c^2
\quad\Longrightarrow\quad
a^2 + b^2 = c^2 . \qed
\]
\end{solution}

\begin{solution}{pb:g8:pythagoras:1}
\textbf{1.} $5^2 + 12^2 = 25 + 144 = 169 = 13^2$؛ \quad
و $8^2 + 15^2 = 64 + 225 = 289 = 17^2$؛ \quad
و $20^2 + 21^2 = 400 + 441 = 841 = 29^2$. فالثلاث كلها ثلاثيات،
وحسب \cref{thm:g8:pythagoras:converse} يكون المثلث ذو أطوال
الأضلاع هذه \omterm{def:g6:shapes:triangles}{قائم الزاوية}، \omterm{def:g3:shapes:rightangle}{والزاوية القائمة} تواجه الضلع
الأطول.

\textbf{2.} يكوّن الحبل مثلثًا أضلاعه $3$ و $4$ و $5$
(\omterm{def:g6:lines:objects}{بالقطع})، و $3^2 + 4^2 = 25 = 5^2$: فبالعكس
(\cref{thm:g8:pythagoras:converse})، تكون الزاوية بين ضلعَي
$3$ و $4$ قطع قائمة بالضبط --- بلا منقلة،
بالعقد وحدها.

\textbf{3.} إذا كان $a^2 + b^2 = c^2$، فإن
\[
(ka)^2 + (kb)^2 = k^2 a^2 + k^2 b^2
= k^2\left(a^2 + b^2\right) = k^2 c^2 = (kc)^2 .
\]
ومن $(3, 4, 5)$: مثلًا $(6, 8, 10)$ و $(9, 12, 15)$ و
$(30, 40, 50)$.

\textbf{4.} \omterm{def:g5:division:multiple}{المضاعف} $(3k, 4k, 5k)$ الذي يحوي $5$ عددًا
أصغر فيه كان يستلزم $3k = 5$، ولا عدد صحيح $k$
يحقّق ذلك ($k = 1$ يعطي $3$، و $k = 2$ يعطي $6$). إذن
$(5, 12, 13)$ ليست مضاعفًا للثلاثية $(3, 4, 5)$.

\textbf{5.} يبيّن السؤال 3 أن التكبير لا ينتج إلا الثلاثيات
$(3k, 4k, 5k)$؛ ويعرض السؤال 4 ثلاثية حقيقية ليست
من هذه الصورة. إذن عائلة كل الثلاثيات الفيثاغورية
أوسع تمامًا من \omterm{def:g5:division:multiple}{مضاعفات} $(3, 4, 5)$: فإنتاجها
يحتاج شيئًا جديدًا.

\textbf{6.} اُكتب $A = m^2$ و $B = n^2$. وبالمتطابقات
الشهيرة (\cref{pb:g8:equations:1}):
\[
a^2 = (A - B)^2 = A^2 - 2AB + B^2,
\qquad
b^2 = (2mn)^2 = 4m^2n^2 = 4AB .
\]
وبالجمع:
\[
a^2 + b^2 = A^2 - 2AB + B^2 + 4AB
= A^2 + 2AB + B^2 = (A + B)^2
= \left(m^2 + n^2\right)^2 = c^2 .
\]

\textbf{7.}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccc}
$(m, n)$ & $m^2 - n^2$ & $2mn$ & $m^2 + n^2$ \\
\hline
$(2, 1)$ & $3$ & $4$ & $5$ \\
$(3, 1)$ & $8$ & $6$ & $10$ \\
$(3, 2)$ & $5$ & $12$ & $13$ \\
$(4, 1)$ & $15$ & $8$ & $17$ \\
$(4, 3)$ & $7$ & $24$ & $25$ \\
\end{tabular}
\end{center}

\textbf{8.} الثلاثية $(8, 6, 10)$ هي ضعف $(4, 3, 5)$ --- أي
ضعف ثلاثية البنّائين $(3, 4, 5)$. فالآلة تعيد اِكتشاف
ثلاثية قديمة متنكّرة أحيانًا.

\textbf{9.} نحتاج $m^2 - n^2 = 21$ و $2mn = 20$، إذن
$mn = 10$: وبتجريب $m = 5$ و $n = 2$ نجد $25 - 4 = 21$ و
$2 \times 10 = 20$، والوتر $25 + 4 = 29$. فتُخرج الآلة
$(21, 20, 29)$ من أجل $(m, n) = (5, 2)$.

\textbf{10.} القيمتان $m = 8$ و $n = 6$ تعطيان $m^2 + n^2 = 64 + 36 = 100$.
والثلاثية هي
\[
\left(64 - 36,\ 2 \times 48,\ 100\right) = (28,\ 96,\ 100),
\]
وفعلًا $28^2 + 96^2 = 784 + 9216 = 10\,000 = 100^2$.

\textbf{11.} مع $m = n + 1$:
\[
m^2 - n^2 = (n + 1)^2 - n^2 = 2n + 1
\]
(\omterm{ex:g1:subtraction:difference}{فرق} مربعين متتاليين، \cref{pb:g8:equations:1})؛
و $2mn = 2n(n + 1) = 2n^2 + 2n$؛ و
$m^2 + n^2 = n^2 + n^2 + 2n + 1 = 2n^2 + 2n + 1$. فالوتر
يزيد على ضلع القائمة الزوجي بمقدار $1$ بالضبط. ومن أجل $n = 1, 2, 3, 4$:
\[
(3, 4, 5), \quad (5, 12, 13), \quad (7, 24, 25), \quad
(9, 40, 41) .
\]

\textbf{12.} كل \omterm{def:g3:numbers:evenodd}{عدد فردي} لا يقلّ عن $3$ هو $2n + 1$ من أجل
$n \geq 1$ ما، والسؤال 11 يقدّم ثلاثية ضلع قائمتها الفردي
هو $2n + 1$ بالضبط. ومن أجل $11 = 2 \times 5 + 1$، خذ $n = 5$:
\[
(11,\ 60,\ 61),
\qquad
11^2 + 60^2 = 121 + 3600 = 3721 = 61^2 .
\]

\textbf{13.} مع $n = 1$ تعطي الآلة
$\left(m^2 - 1,\ 2m,\ m^2 + 1\right)$، وضلع قائمتها الزوجي $2m$ يمكن أن يكون
أيّ \omterm{def:g3:numbers:evenodd}{عدد زوجي} من $4$ فصاعدًا. ومن أجل $14 = 2 \times 7$، خذ
$m = 7$: فتكون الثلاثية $(48, 14, 50)$. وتظهر الثلاثية نفسها
بمضاعفة $(7, 24, 25)$ من جدول السؤال 7:
$(14, 48, 50)$ --- أي الأعداد الثلاثة نفسها.

\textbf{14.} \omterm{def:g3:numbers:evenodd}{العدد الزوجي} هو $2k$، والمربع $(2k)^2 = 4k^2$ زوجي
(فهو \omterm{def:g5:division:multiple}{مضاعف} للعدد $4$، وهو زوجي). \omterm{def:g3:numbers:evenodd}{والعدد الفردي} هو $2k + 1$، و
المتطابقة الشهيرة الأولى تعطي
\[
(2k + 1)^2 = 4k^2 + 4k + 1 = 2\left(2k^2 + 2k\right) + 1 ,
\]
أي \omterm{def:g3:numbers:evenodd}{عدد زوجي} زائد $1$: فهو فردي.

\textbf{15.} اِفترض أن $a$ و $b$ و $c$ كلها فردية. فبالسؤال
14، يكون $a^2$ و $b^2$ فرديين، إذن $a^2 + b^2$ \omterm{def:g1:addition:def}{مجموع} عددين
فرديين: أي \emph{زوجي}. لكن $c^2$ كان سيكون \emph{فرديًّا}. فيصير التساوي
$a^2 + b^2 = c^2$ مستحيلًا. إذن في كل \omterm{pb:g8:pythagoras:1}{ثلاثية
فيثاغورية} واحد على الأقلّ من الأعداد الثلاثة زوجي --- كما يؤكد كل
سطر من جدول السؤال 7.
\end{solution}
