\section*{الفصل \ref{ch:g8:powers} --- القوى}

\begin{solution}{exo:g8:powers:1}
$2^6 = 64$; \quad $3^3 = 27$; \quad $10^5 = 100\,000$; \quad
$1^{100} = 1$; \quad $0.1^2 = 0.01$; \quad $6^0 = 1$.
\end{solution}

\begin{solution}{exo:g8:powers:2}
$(-3)^2 = 9$; \quad $-3^2 = -9$; \quad $(-1)^{15} = -1$; \quad
$(-5)^3 = -125$; \quad $\left(\frac23\right)^2 = \frac49$.
\end{solution}

\begin{solution}{exo:g8:powers:3}
$7^4 \times 7^5 = 7^9$; \quad
$\dfrac{2^9}{2^3} = 2^6$; \quad
$\left(10^3\right)^4 = 10^{12}$; \quad
$5^6 \times 5^{-2} = 5^4$; \quad
$3^4 \times 7^4 = 21^4$ (\omterm{def:g8:powers:def}{الأُسّ} نفسه: اِضرب الأساسين).
\end{solution}

\begin{solution}{exo:g8:powers:4}
$10^{-2} = 0.01$; \quad
$4 \times 10^3 = 4000$; \quad
$2.5 \times 10^{-4} = 0.00025$; \quad
$10^0 = 1$.
\end{solution}

\begin{solution}{exo:g8:powers:5}
$10^5$; \quad $10^{-1}$; \quad $10^{10}$; \quad $10^{-6}$.
\end{solution}

\begin{solution}{exo:g8:powers:6}
$\dfrac{10^7 \times 10^{-3}}{10^2} = \dfrac{10^4}{10^2} = 10^2 =
100$.

$\dfrac{2^5 \times 2^4}{2^6} = \dfrac{2^9}{2^6} = 2^3 = 8$.

$\left(2^2\right)^3 \times 2^{-4} = 2^6 \times 2^{-4} = 2^2 = 4$.
\end{solution}

\begin{solution}{exo:g8:powers:7}
$2^3 \times 2^4 = 2^{12}$: \emph{خطأ} --- \omterm{def:g8:powers:def}{فالأُسُس} تُجمع:
$2^7$.

$5^2 + 5^3 = 5^5$: \emph{خطأ} --- فلا قاعدة للمجاميع:
$25 + 125 = 150$، بينما $5^5 = 3125$.

$(3^2)^4 = 3^8$: \emph{صواب}.

$10^3 \times 10^3 = 100^3$: \emph{صواب} --- فكلاهما يساوي $10^6$
(اليسار: $10^{3+3}$؛ واليمين: $(10^2)^3$).
\end{solution}

\begin{solution}{exo:g8:powers:8}
الأشخاص الجدد في اليوم $4$: كل واحد من الأشخاص $3^3 = 27$ الذين أُخبروا في اليوم
3 يخبر ثلاثة آخرين: أي $3^4 = 81$. والذين يعرفون في نهاية اليوم 4:
$3 + 9 + 27 + 81 = 120$ شخصًا.
\end{solution}

\begin{solution}{exo:g8:powers:9}
\emph{1.} السُّمك: $2^{10} \times 10^{-4}$ م
$= 1024 \times 10^{-4}$ م $\approx 0.1$ م $= 10$ سم.

\emph{2.} $2^{42} \times 10^{-4} \approx 4.4 \times 10^{12} \times
10^{-4} = 4.4 \times 10^8$ م، وهو أكبر من $3.8 \times 10^8$ م: فبعد
$42$ طيّة (نظرية!)، تتجاوز رزمة الورق القمر.
\end{solution}

\begin{solution}{exo:g8:powers:10}
$2^{10} = 1024$؛ و $10^3 = 1000$؛ و $3^6 = 729$؛ و $5^4 = 625$. والترتيب:
\[
5^4 < 3^6 < 10^3 < 2^{10} .
\]
\end{solution}

\begin{solution}{exo:g8:powers:11}
$2^{100} = \left(2^{10}\right)^{10} = 1024^{10}$ و
$10^{30} = \left(10^3\right)^{10} = 1000^{10}$. وبما أن $1024 > 1000$،
فضرب عشر نسخ من كل منهما يحفظ المتراجحة:
$2^{100} > 10^{30}$.
\end{solution}

\begin{solution}{pb:g8:powers:1}
\textbf{1.} تتضاعف الحبّات من مربع إلى مربع اِنطلاقًا من
$1 = 2^0$: فالمربع $k$ يحمل $2^{k-1}$ حبة
(\cref{def:g8:powers:def}). والمربع $64$ يحمل $2^{63}$.

\textbf{2.} $S_1 = 1$، و $S_2 = 1 + 2 = 3$، و $S_3 = 3 + 4 = 7$،
و $S_4 = 7 + 8 = 15$، و $S_5 = 15 + 16 = 31$: أي دائمًا أقلّ بواحد من
القوة التالية للعدد $2$ ($2$ و $4$ و $8$ و $16$ و $32$). والتخمين:
$S_n = 2^n - 1$.

\textbf{3.} مضاعفة كل حدّ من $S_n$ ترفع كل قوة
بواحد ($2 \times 2^{k} = 2^{k+1}$):
\begin{align*}
S_n &= 1 + 2 + 4 + \dots + 2^{n-1}, \\
2 \times S_n &= \phantom{1 + {}} 2 + 4 + \dots + 2^{n-1} + 2^n .
\end{align*}
وبطرح السطر الأول من الثاني، يظهر كل حدّ من $2$
إلى $2^{n-1}$ في الاثنين فيتلاشى:
\[
2 \times S_n - S_n = 2^n - 1,
\qquad\text{أي}\qquad
S_n = 2^n - 1 .
\]

\textbf{4.} \omterm{def:g1:addition:def}{المجموع} هو $S_{64} = 2^{64} - 1$ حبة. والمربع
الخامس والستون كان سيحمل $2^{64}$ حبة: فالرقعة كلها
تحمل بالضبط حبة واحدة أقلّ مما كان يحمله ذلك المربع وحده.

\textbf{5.} النصف الأول يحمل $S_{32} = 2^{32} - 1$ حبة.
والرقعة كلها تحمل $2^{64} - 1$، إذن يحمل النصف الثاني
\[
\left(2^{64} - 1\right) - \left(2^{32} - 1\right)
= 2^{64} - 2^{32}
= 2^{32} \times \left(2^{32} - 1\right)
\]
(بتعميل $2^{32}$، مستعملين $2^{32} \times 2^{32} = 2^{64}$،
\cref{thm:g8:powers:rules}): أي بالضبط $2^{32}$ مثل النصف
الأول.

\textbf{6.} بقواعد \omterm{def:g8:powers:def}{الأُسُس}، $2^{64} = 2^{4 + 60} =
2^4 \times \left(2^{10}\right)^6$. وبما أن $2^{10} = 1024 > 10^3$،
\[
2^{64} > 16 \times \left(10^3\right)^6 = 16 \times 10^{18}
= 1.6 \times 10^{19} .
\]

\textbf{7.} أكثر من $1.6 \times 10^{19}$ حبة، وزن كل منها
$5 \times 10^{-2}$ غ:
\[
1.6 \times 10^{19} \times 5 \times 10^{-2}
= 8 \times 10^{17} \text{ غ} .
\]
وبالقسمة على $10^6$ غ للطن: أكثر من $8 \times 10^{11}$
طن --- أي ثمانمئة مليار طن.

\textbf{8.} $\dfrac{8 \times 10^{11}}{8 \times 10^{8}} = 10^3$:
فقد وعد الملك بما لا يقلّ عن \emph{ألف سنة} من حصاد
العالم كله اليوم. (وتقول الأسطورة إن مستشاريه أخبروه
بذلك.)

\textbf{9.} $2^{19} = 2^9 \times 2^{10} = 512 \times 1024 =
524\,288 < 10^6$، بينما $2^{20} = \left(2^{10}\right)^2 =
1024^2 = 1\,048\,576 > 10^6$. إذن أصغر \omterm{def:g8:powers:def}{أُسّ} هو
$n = 20$: أي عشرون مضاعفة تتجاوز المليون.

\textbf{10.} الأجر في اليوم $n$ هو $2^{n-1}$ سنتًا (فاليوم 1:
$2^0 = 1$). والآن
\[
2^{26} = 2^6 \times \left(2^{10}\right)^2
= 64 \times 1\,048\,576 = 67\,108\,864 < 10^8,
\]
\[
2^{27} = 2 \times 2^{26} = 134\,217\,728 > 10^8 .
\]
إذن يتجاوز الأجر اليومي $10^8$ سنت أول مرة حين $n - 1 = 27$:
أي في اليوم $28$.

\textbf{11.} $21 = 16 + 4 + 1$: أي الصنجات $16$ و $4$ و $1$ غ.
و $27 = 16 + 8 + 2 + 1$: أي الصنجات $16$ و $8$ و $2$ و $1$ غ.

\textbf{12.} بالسؤال 3، تزن الصنجات $1, 2, 4, 8$ معًا
$S_4 = 2^4 - 1 = 15$ غ. إذن دون الصنجة $16$ غ لا تستطيع
التاجرة تجاوز $15$ غ: فأيّ هدف من $16$ غ فأكثر يجب أن
يستعملها. وأيّ هدف من $15$ غ فأقلّ يجب ألا يستعملها، لأن
الصنجة $16$ غ وحدها تتجاوز الهدف أصلًا. ومنه يكون اِختيار
الصنجة الأكبر \emph{مفروضًا}. ويبقى
هدف لا يتجاوز $15$ غ يُكوَّن من $1, 2, 4, 8$ ---
ويتكرّر الاستدلال نفسه: فالصنجة $8$ مفروضة (تُستعمل إذا كان الهدف
\omterm{def:g3:division:remainder}{الباقي} $\geq 8$، وتُترك وإلا، لأن $1 + 2 + 4 = 7$)،
ثم $4$ (لأن $1 + 2 = 3$)، ثم $2$، ثم $1$.

\textbf{13.} باتّباع الاختيارات المفروضة، يكون الهدف \omterm{def:g3:division:remainder}{الباقي}
بعد كل مرحلة لا يتجاوز \omterm{def:g1:addition:def}{مجموع} الصنجات الباقية،
فتنتهي العملية بباقٍ $0$: أي إن كل هدف من $1$ إلى
$31$ يُبلَغ. وبما أن كل اِختيار في الطريق كان مفروضًا،
فلا يمكن لأيّ اِنتقاء آخر من الصنجات أن يبلغ الهدف نفسه: فكتابة
كل عدد من $1$ إلى $31$ مجموعَ قوى متمايزة
للعدد $2$ موجودة و\emph{وحيدة}.

\textbf{14.} تزن الصنجات الست $S_6 = 2^6 - 1 = 63$ غ،
ويغطّي اِستدلال الاختيار المفروض نفسه كل هدف من $1$ إلى
$63$ غ. ومن أجل $45$: الهدف $\geq 32$، إذن اِستعمل $32$؛ فيبقى
$13 < 16$، إذن اُترك $16$؛ واِستعمل $8$ (فيبقى $5$)، واِستعمل $4$ (فيبقى
$1$)، واُترك $2$، واِستعمل $1$:
\[
45 = 32 + 8 + 4 + 1 = 2^5 + 2^3 + 2^2 + 2^0 .
\]

\textbf{15.} اِختيار $0$ أو $1$ على كل مربع من المربعات $64$
يعادل اِختيار أيّ القوى $2^0, 2^1, \dots, 2^{63}$ تُدرَج
في \omterm{def:g1:addition:def}{مجموع} --- وهو بالضبط وزن التاجرة بالصنجات $64$.
وأصغر عدد يمكن تخزينه هو $0$ (فكل المربعات على
$0$)؛ وأكبره \omterm{def:g1:addition:def}{مجموع} \emph{كل} القوى، وهو
\omterm{def:g1:addition:def}{مجموع} الملك: $S_{64} = 2^{64} - 1$ (الجزء الأول). وباستدلال
الاختيار المفروض، يُبلَغ كل عدد صحيح بينهما
مرة واحدة بالضبط: فالآلة ذات $64$ خانة تخزّن بالضبط الأعداد
الصحيحة من $0$ إلى $2^{64} - 1$.
\end{solution}
