\section*{अध्याय \ref{ch:g8:cosine} --- समकोण त्रिभुज और कोज्या}

\begin{solution}{exo:g8:cosine:1}
\omterm{def:g6:shapes:triangles}{समकोण} से खींची माध्यिका: कर्ण की आधी, यानी $6.5$ cm
(\cref{thm:g8:cosine:circle})। \omterm{pb:g6:symmetry:1}{परिवृत्त} का केंद्र कर्ण का
मध्यबिंदु है और \omterm{def:g3:shapes:shapes}{त्रिज्या} $6.5$ cm।
\end{solution}

\begin{solution}{exo:g8:cosine:2}
$A$ पर कोण \omterm{def:g6:shapes:triangles}{समकोण} है ($[BC]$ एक \omterm{def:g4:geometry:circle}{व्यास} है: \cref{thm:g8:cosine:circle}
की बात 2)। नापों से $AB^2 + AC^2 \approx 64 = BC^2$ मिलता है, नापने
की सटीकता तक।
\end{solution}

\begin{solution}{exo:g8:cosine:3}
कर्ण: $[EF]$ (\omterm{def:g6:shapes:triangles}{समकोण} $D$ के सामने)। $\widehat E$ की संलग्न भुजा:
$[ED]$। सामने वाली: $[DF]$। इसलिए
$\cos\widehat E = \dfrac{ED}{EF}$।
\end{solution}

\begin{solution}{exo:g8:cosine:4}
\emph{1.} संलग्न भुजा $= 10 \cos 35^\circ \approx 8.2$।

\emph{2.} कर्ण $= \dfrac{6}{\cos 50^\circ} \approx
\dfrac{6}{0.643} \approx 9.3$।
\end{solution}

\begin{solution}{exo:g8:cosine:5}
$\cos\theta = \frac{8}{10} = 0.8$, इसलिए
$\theta = \cos^{-1}(0.8) \approx 37^\circ$।
\end{solution}

\begin{solution}{exo:g8:cosine:6}
$\cos\theta$ कोई \omterm{def:g6:shapes:triangles}{समकोण} भुजा कर्ण से भाग देकर मिलता है, और कर्ण
\omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} की \emph{सबसे लंबी} भुजा है: इसलिए यह \omterm{def:g3:division:remainder}{भागफल} हमेशा $1$
से छोटा होता है। $1.2$ का मतलब होता कोई \omterm{def:g6:shapes:triangles}{समकोण} भुजा कर्ण से भी लंबी
--- जो असंभव है।
\end{solution}

\begin{solution}{exo:g8:cosine:7}
आड़ी दूरी $= 25 \cos 12^\circ \approx 25 \times 0.978 \approx
24.5$ m।
\end{solution}

\begin{solution}{exo:g8:cosine:8}
कोण बढ़ने के साथ कोज्या घटती है
(\cref{prop:g8:cosine:bounds}):
\[
\cos 70^\circ < \cos 45^\circ < \cos 20^\circ .
\]
(कैलकुलेटर से: $0.342 < 0.707 < 0.940$।)
\end{solution}

\begin{solution}{exo:g8:cosine:9}
ज़मीन वाली दूरी $\theta$ की संलग्न है:
$\cos 68^\circ = \frac{1.2}{L}$, इसलिए
$L = \frac{1.2}{0.375} = 3.2$ m। ऊँचाई (पाइथागोरस से):
$\sqrt{3.2^2 - 1.2^2} = \sqrt{10.24 - 1.44} = \sqrt{8.8} \approx 3.0$
m।
\end{solution}

\begin{solution}{exo:g8:cosine:10}
\emph{1.} $A$ \omterm{def:g4:geometry:circle}{व्यास} $[BC]$ वाले वृत्त पर है: इसलिए $A$ पर \omterm{def:g6:shapes:triangles}{समकोण}
(\cref{thm:g8:cosine:circle})।

\emph{2.} $BC = 9$ cm (\omterm{def:g3:shapes:shapes}{त्रिज्या} का दुगुना)। पाइथागोरस से:
$AC = \sqrt{9^2 - 5.4^2} = \sqrt{81 - 29.16} = \sqrt{51.84} = 7.2$
cm। और $\cos\widehat B = \dfrac{AB}{BC} = \dfrac{5.4}{9} = 0.6$।
\end{solution}

\begin{solution}{exo:g8:cosine:11}
भुजा $1$ के \omterm{def:g6:shapes:triangles}{समबाहु त्रिभुज} में किसी \omterm{def:g5:solids:def}{शीर्ष} से खींची ऊँचाई सामने वाली
भुजा के \emph{मध्यबिंदु} पर गिरती है (\omterm{def:g6:symmetry:axis}{सममिति अक्ष})। त्रिभुज का \omterm{def:g3:division:half}{आधा}
हिस्सा \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} है, जिसका कर्ण $1$ है (यानी पूरी भुजा), आधार
वाले \omterm{def:g5:solids:def}{शीर्ष} पर कोण $60^\circ$ है, और संलग्न भुजा $\frac12$ (यानी
आधार की आधी)। इसलिए
\[
\cos 60^\circ = \frac{1/2}{1} = \frac12 .
\]
\end{solution}

\begin{solution}{pb:g8:cosine:1}
\textbf{1.} $H$ पर \omterm{def:g6:shapes:triangles}{समकोण} वाले त्रिभुज $ABH$ में कोणों का योग
(\cref{thm:g7:triangles:sum}) देता है
$\widehat B + 90^\circ + \widehat{BAH} = 180^\circ$, इसलिए
$\widehat{BAH} = 90^\circ - \widehat B$। और $A$ पर \omterm{def:g6:shapes:triangles}{समकोण} वाले
त्रिभुज $ABC$ में:
$\widehat B + \widehat C + 90^\circ = 180^\circ$, इसलिए
$\widehat C = 90^\circ - \widehat B$। यानी
$\widehat{BAH} = \widehat{ACB}$: दोनों छोटे त्रिभुज असली त्रिभुज के
कोण $\widehat B$, $\widehat C$, $90^\circ$ दोहरा देते हैं।

\textbf{2.} $ABC$ ($A$ पर \omterm{def:g6:shapes:triangles}{समकोण}) में $\widehat B$ की संलग्न भुजा
$AB$ है और कर्ण $BC$: इसलिए $\cos\widehat B = \frac{AB}{BC}$। और
$ABH$ ($H$ पर \omterm{def:g6:shapes:triangles}{समकोण}) में $\widehat B$ की संलग्न भुजा $BH$ है और
कर्ण $AB$: इसलिए $\cos\widehat B = \frac{BH}{AB}$। यह अनुपात सिर्फ़
कोण पर टिका है (\cref{def:g8:cosine:def}), इसलिए
\[
\frac{AB}{BC} = \frac{BH}{AB} ;
\]
दोनों तरफ़ $BC \times AB$ से गुणा करने पर
(\cref{thm:g8:equations:balance}):
$AB^2 = BH \times BC$।

\textbf{3.} त्रिभुज $ABC$ ($A$ पर \omterm{def:g6:shapes:triangles}{समकोण}) और $ACH$ ($H$ पर \omterm{def:g6:shapes:triangles}{समकोण})
में कोण $\widehat C$ साझा है। दोनों में संलग्न भुजा बटा कर्ण:
\[
\cos\widehat C = \frac{AC}{CB} = \frac{CH}{AC}
\qquad\Longrightarrow\qquad
AC^2 = CH \times CB .
\]

\textbf{4.} $AB^2 = BH \times BC$ यानी $9 = BH \times 5$, इसलिए
$BH = 1.8$; और $16 = CH \times 5$ से $CH = 3.2$। जाँच:
$BH + HC = 1.8 + 3.2 = 5 = BC$ --- और ऐसा होना ही था, क्योंकि $H$
कर्ण पर $B$ और $C$ के बीच पड़ता है।

\textbf{5.} दोनों संबंधों को $BC$ से भाग देने पर:
\[
BH = \frac{AB^2}{BC},
\qquad
CH = \frac{AC^2}{BC} .
\]
यानी $BH$ और $CH$, $AB^2$ तथा $AC^2$ के \omterm{def:g6:propdata:def}{समानुपाती} हैं: शीर्षलंब का
पैर कर्ण को \omterm{def:g6:shapes:triangles}{समकोण} भुजाओं के वर्गों के अनुपात में बाँटता है (सवाल 4
में $1.8 : 3.2 = 9 : 16$)।

\textbf{6.} दोनों संबंध जोड़कर और $BC$ बाहर निकालकर (वितरण नियम
से):
\[
AB^2 + AC^2 = BH \times BC + CH \times BC
= (BH + CH) \times BC = BC \times BC = BC^2 .
\]
यह \cref{thm:g8:pythagoras:direct} ही है, और वह सिर्फ़ कोज्या से आ
गया --- \cref{exo:g8:pythagoras:11} वाली चार-त्रिभुज \omterm{def:g6:measure:area}{क्षेत्रफल}
पहेली से सचमुच अलग प्रमाण।

\textbf{7.} $ABH$ में पाइथागोरस ($H$ पर \omterm{def:g6:shapes:triangles}{समकोण}, कर्ण $[AB]$):
$AH^2 = AB^2 - BH^2$। $AB^2$ की जगह $BH \times BC$ रखकर (सवाल 2)
और $BH$ बाहर निकालकर:
\[
AH^2 = BH \times BC - BH^2
= BH \times (BC - BH)
= BH \times HC .
\]

\textbf{8.} \omterm{def:g6:shapes:triangles}{समकोण} भुजाओं को आधार और ऊँचाई मानने पर $ABC$ का
\omterm{def:g6:measure:area}{क्षेत्रफल} $\frac{AB \times AC}{2}$ है; और कर्ण $[BC]$ को आधार मानने
पर ऊँचाई ठीक $AH$ है, इसलिए \omterm{def:g6:measure:area}{क्षेत्रफल}
$\frac{BC \times AH}{2}$ भी है (\cref{thm:g7:areas:triangle})।
दोनों बराबर रखकर और दुगुना करने पर: $AB \times AC = BC \times AH$।

\textbf{9.} $AH = \frac{AB \times AC}{BC} = \frac{3 \times 4}{5}
= 2.4$। सवाल 7 की जाँच: $AH^2 = 2.4^2 = 5.76$ और
$BH \times HC = 1.8 \times 3.2 = 5.76$। बराबर।

\textbf{10.} $AH^2 = BH \times HC = 2 \times 8 = 16$, इसलिए
$AH = \sqrt{16} = 4$ cm।

\textbf{11.} बिंदु $A$ \omterm{def:g4:geometry:circle}{व्यास} $[BC]$ वाले वृत्त पर पड़ता है (और $B$
तथा $C$ से अलग है), इसलिए \cref{thm:g8:cosine:circle} की बात 2 से
त्रिभुज $ABC$ का $A$ पर \omterm{def:g6:shapes:triangles}{समकोण} है। और रचना से $(AH) \perp (BC)$ है,
$H$ के $B$ तथा $C$ के बीच होते हुए: यानी \omterm{def:g6:lines:objects}{रेखाखंड} $[AH]$ ठीक सवाल 7
वाला शीर्षलंब है, और $AH^2 = BH \times HC = p \times q$।

\textbf{12.} $AH^2 = 1 \times 2 = 2$: यह आकृति सिर्फ़ पैमाने और
परकार से ऐसा \omterm{def:g6:lines:objects}{रेखाखंड} बना देती है जिसका वर्ग $2$ है --- यानी इकाई
वर्ग का विकर्ण, $\sqrt2 \approx 1.414$, जो
\cref{ex:g8:pythagoras:diagonal} में मिला था।

\textbf{13.} नुस्ख़ा: $p = 1$ और $q = n$ लंबाइयों के दो \omterm{def:g6:lines:objects}{रेखाखंड}
सिरे से सिरे रखो; पूरे \omterm{def:g6:lines:objects}{रेखाखंड} (लंबाई $n + 1$) को \omterm{def:g4:geometry:circle}{व्यास} मानकर \omterm{def:g3:division:half}{आधा}
वृत्त बनाओ; जोड़ वाले बिंदु पर \omterm{def:g6:lines:perp}{लंब} खड़ा करो; वह आधे वृत्त से जिस
बिंदु पर मिलता है उसकी जोड़ वाले बिंदु से दूरी
$\sqrt{1 \times n} = \sqrt n$ होती है। $\sqrt5$ के लिए: $p = 1$
और $q = 5$ लो (यानी \omterm{def:g4:geometry:circle}{व्यास} $6$ का \omterm{def:g3:division:half}{आधा} वृत्त)।

\textbf{14.} $A$ केंद्र $O$ वाले वृत्त पर है, इसलिए
$OA = \frac{p+q}{2}$, यानी \omterm{def:g3:shapes:shapes}{त्रिज्या}। अगर $H \neq O$ हो तो त्रिभुज
$AHO$ का $H$ पर \omterm{def:g6:shapes:triangles}{समकोण} है और उसका कर्ण $[OA]$ है, और कोई \omterm{def:g6:shapes:triangles}{समकोण} भुजा
कर्ण से छोटी होती है: इसलिए $AH < OA$। और अगर $H = O$ हो तो
$AH = OA$ ठीक-ठीक। हर हाल में $AH \leq \frac{p+q}{2}$, और बराबरी
ठीक तब जब $H$ केंद्र पर हो --- यानी जब $p = q$।

\textbf{15.} $AH \leq \frac{p+q}{2}$ का वर्ग करके (दोनों तरफ़ धन
हैं) और $AH^2 = pq$ इस्तेमाल करके:
\[
p \times q \leq \left(\frac{p+q}{2}\right)^2 ,
\]
और बराबरी ठीक $p = q$ पर। बीजगणित से दोबारा प्रमाण, उल्लेखनीय
सर्वसमिकाओं से (\cref{pb:g8:equations:1}):
\[
\left(\frac{p+q}{2}\right)^2 - pq
= \frac{p^2 + 2pq + q^2 - 4pq}{4}
= \frac{p^2 - 2pq + q^2}{4}
= \left(\frac{p-q}{2}\right)^2 ,
\]
जो एक वर्ग है, इसलिए कभी ऋण नहीं होता --- और ठीक $p = q$ पर \omterm{def:g1:counting:zero}{शून्य}
होता है। यानी दो संख्याओं का \omterm{pb:g8:cosine:1}{गुणोत्तर माध्य} उनके आधे योग से कभी
ज़्यादा नहीं होता।

\end{solution}
