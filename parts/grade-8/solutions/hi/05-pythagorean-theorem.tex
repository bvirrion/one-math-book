\begin{solution}{exo:g8:pythagoras:4}

\begin{solution}{exo:g8:pythagoras:6}
ऊँचाई आधार के मध्यबिंदु पर गिरती है (\omterm{def:g6:shapes:triangles}{समद्विबाहु त्रिभुज}), इसलिए हर
आधे त्रिभुज का कर्ण $10$ और एक \omterm{def:g6:shapes:triangles}{समकोण} भुजा $6$ है:
$h = \sqrt{100 - 36} = \sqrt{64} = 8$ cm। \omterm{def:g6:measure:area}{क्षेत्रफल}:
$\frac{12 \times 8}{2} = 48$ cm$^2$।
\end{solution}

\begin{solution}{exo:g8:pythagoras:7}

\begin{solution}{exo:g8:pythagoras:8}

$\sqrt{24^2 + 10^2} = \sqrt{576 + 100} = \sqrt{676} = 26$ km।

\begin{solution}{exo:g8:pythagoras:9}

पास ही है --- चल जाएगा।
\end{solution}

\begin{solution}{exo:g8:pythagoras:10}
\omterm{def:g6:shapes:triangles}{समकोण} भुजाएँ $5 - 1 = 4$ (आड़ी) और $5 - 2 = 3$ (खड़ी) नापती हैं,
इसलिए
\[
AB = \sqrt{4^2 + 3^2} = \sqrt{25} = 5 .
\]
\end{solution}

\begin{solution}{exo:g8:pythagoras:11}
\emph{1.} सीधे: $(a + b)^2$। टुकड़ों के रूप में: हर एक
$\frac{ab}{2}$ \omterm{def:g6:measure:area}{क्षेत्रफल} के चार त्रिभुज, और ऊपर से तिरछा वर्ग
$c^2$:
\[
(a + b)^2 = 4 \times \frac{ab}{2} + c^2 = 2ab + c^2 .
\]

\emph{2.} बाईं तरफ़ फैलाने पर: $(a+b)^2 = a^2 + 2ab + b^2$।
इसलिए
\[
a^2 + 2ab + b^2 = 2ab + c^2
\quad\Longrightarrow\quad
a^2 + b^2 = c^2 . \qed
\]
\end{solution}

\begin{solution}{pb:g8:pythagoras:1}
\textbf{1.} $5^2 + 12^2 = 25 + 144 = 169 = 13^2$; \quad
$8^2 + 15^2 = 64 + 225 = 289 = 17^2$; \quad
$20^2 + 21^2 = 400 + 441 = 841 = 29^2$। तीनों त्रिक हैं, और
\cref{thm:g8:pythagoras:converse} से इन भुजाओं वाला त्रिभुज \omterm{def:g6:shapes:triangles}{समकोण}
होता है, और \omterm{def:g6:shapes:triangles}{समकोण} सबसे लंबी भुजा के सामने पड़ता है।

\textbf{2.} रस्सी से $3$, $4$, $5$ (\omterm{def:g6:lines:objects}{रेखाखंडों} में) भुजाओं वाला
त्रिभुज बनता है, और $3^2 + 4^2 = 25 = 5^2$: इसलिए विलोम से
(\cref{thm:g8:pythagoras:converse}) $3$ और $4$ \omterm{def:g6:lines:objects}{रेखाखंडों} वाली
भुजाओं के बीच का कोण ठीक \omterm{def:g6:shapes:triangles}{समकोण} है --- चाँदे की ज़रूरत नहीं, बस
गाँठों की।

\textbf{3.} अगर $a^2 + b^2 = c^2$ हो, तो
\[
(ka)^2 + (kb)^2 = k^2 a^2 + k^2 b^2
= k^2\left(a^2 + b^2\right) = k^2 c^2 = (kc)^2 .
\]
$(3, 4, 5)$ से: जैसे $(6, 8, 10)$, $(9, 12, 15)$ और
$(30, 40, 50)$।

\textbf{4.} जिस \omterm{def:g5:division:multiple}{गुणज} $(3k, 4k, 5k)$ की सबसे छोटी संख्या $5$ हो
उसके लिए $3k = 5$ चाहिए, और कोई पूर्ण संख्या $k$ ऐसा नहीं करती
($k = 1$ से $3$, $k = 2$ से $6$)। इसलिए $(5, 12, 13)$,
$(3, 4, 5)$ का \omterm{def:g5:division:multiple}{गुणज} नहीं है।

\textbf{5.} सवाल 3 दिखाता है कि बढ़ाने से सिर्फ़ त्रिक
$(3k, 4k, 5k)$ ही बनते हैं; और सवाल 4 एक सच्चा त्रिक सामने रखता है
जो इस रूप का नहीं है। इसलिए सारे पाइथागोरसी त्रिकों का कुनबा
$(3, 4, 5)$ के गुणजों से पक्के तौर पर बड़ा है: उन्हें बनाने के लिए
कुछ नया चाहिए।

\textbf{6.} $A = m^2$ और $B = n^2$ लिखो। उल्लेखनीय सर्वसमिकाओं
(\cref{pb:g8:equations:1}) से:
\[
a^2 = (A - B)^2 = A^2 - 2AB + B^2,
\qquad
b^2 = (2mn)^2 = 4m^2n^2 = 4AB .
\]
जोड़ने पर:
\[
a^2 + b^2 = A^2 - 2AB + B^2 + 4AB
= A^2 + 2AB + B^2 = (A + B)^2
= \left(m^2 + n^2\right)^2 = c^2 .
\]

\textbf{7.}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccc}
$(m, n)$ & $m^2 - n^2$ & $2mn$ & $m^2 + n^2$ \\
\hline
$(2, 1)$ & $3$ & $4$ & $5$ \\
$(3, 1)$ & $8$ & $6$ & $10$ \\
$(3, 2)$ & $5$ & $12$ & $13$ \\
$(4, 1)$ & $15$ & $8$ & $17$ \\
$(4, 3)$ & $7$ & $24$ & $25$ \\
\end{tabular}
\end{center}

\textbf{8.} $(8, 6, 10)$, $(4, 3, 5)$ का दुगुना है --- यानी
राजमिस्त्रियों वाले $(3, 4, 5)$ का ही। मशीन कभी-कभी किसी पुराने
त्रिक को भेस बदलकर फिर से खोज लेती है।

\textbf{9.} हमें $m^2 - n^2 = 21$ और $2mn = 20$ चाहिए, यानी
$mn = 10$: $m = 5$, $n = 2$ आज़माने पर $25 - 4 = 21$ और
$2 \times 10 = 20$ मिलते हैं, और कर्ण $25 + 4 = 29$। इसलिए
$(m, n) = (5, 2)$ पर मशीन $(21, 20, 29)$ उगलती है।

\textbf{10.} $m = 8$, $n = 6$ से $m^2 + n^2 = 64 + 36 = 100$
मिलता है। त्रिक है
\[
\left(64 - 36,\ 2 \times 48,\ 100\right) = (28,\ 96,\ 100),
\]
और सचमुच $28^2 + 96^2 = 784 + 9216 = 10\,000 = 100^2$।

\textbf{11.} $m = n + 1$ के साथ:
\[
m^2 - n^2 = (n + 1)^2 - n^2 = 2n + 1
\]
(दो लगातार वर्गों का \omterm{ex:g1:subtraction:difference}{अंतर}, \cref{pb:g8:equations:1});
$2mn = 2n(n + 1) = 2n^2 + 2n$; और
$m^2 + n^2 = n^2 + n^2 + 2n + 1 = 2n^2 + 2n + 1$। कर्ण सम \omterm{def:g6:shapes:triangles}{समकोण}
भुजा से ठीक $1$ बड़ा है। $n = 1, 2, 3, 4$ के लिए:
\[
(3, 4, 5), \quad (5, 12, 13), \quad (7, 24, 25), \quad
(9, 40, 41) .
\]

\textbf{12.} $3$ से बड़ी या बराबर हर \omterm{def:g3:numbers:evenodd}{विषम संख्या} किसी $n \geq 1$
के लिए $2n + 1$ है, और सवाल 11 ऐसा त्रिक दे देता है जिसकी विषम
\omterm{def:g6:shapes:triangles}{समकोण} भुजा ठीक $2n + 1$ हो। $11 = 2 \times 5 + 1$ के लिए $n = 5$
लो:
\[
(11,\ 60,\ 61),
\qquad
11^2 + 60^2 = 121 + 3600 = 3721 = 61^2 .
\]

\textbf{13.} $n = 1$ पर मशीन
$\left(m^2 - 1,\ 2m,\ m^2 + 1\right)$ देती है, जिसकी सम \omterm{def:g6:shapes:triangles}{समकोण} भुजा
$2m$ है और वह $4$ से आगे की कोई भी \omterm{def:g3:numbers:evenodd}{सम संख्या} हो सकती है।
$14 = 2 \times 7$ के लिए $m = 7$ लो: त्रिक $(48, 14, 50)$। वही
त्रिक सवाल 7 की सारणी वाले $(7, 24, 25)$ को दुगुना करने पर भी आ
जाता है: $(14, 48, 50)$ --- वही तीन संख्याएँ।

\textbf{14.} कोई \omterm{def:g3:numbers:evenodd}{सम संख्या} $2k$ होती है, और $(2k)^2 = 4k^2$ सम
है ($4$ का \omterm{def:g5:division:multiple}{गुणज}, यानी सम)। कोई \omterm{def:g3:numbers:evenodd}{विषम संख्या} $2k + 1$ होती है, और
पहली उल्लेखनीय सर्वसमिका से
\[
(2k + 1)^2 = 4k^2 + 4k + 1 = 2\left(2k^2 + 2k\right) + 1 ,
\]
यानी एक \omterm{def:g3:numbers:evenodd}{सम संख्या} और $1$: इसलिए विषम।

\textbf{15.} मान लो $a$, $b$, $c$ तीनों विषम होते। सवाल 14 से
$a^2$ और $b^2$ विषम होते, इसलिए $a^2 + b^2$ दो \omterm{def:g3:numbers:evenodd}{विषम संख्याओं} का
\omterm{def:g1:addition:def}{योगफल} होता: \emph{सम}। पर $c^2$ \emph{विषम} होता। तब बराबरी
$a^2 + b^2 = c^2$ असंभव है। इसलिए हर \omterm{pb:g8:pythagoras:1}{पाइथागोरसी त्रिक} में तीनों में
से कम से कम एक संख्या सम होती है --- जैसा सवाल 7 की सारणी की हर
पंक्ति बता रही है।

\end{solution}
