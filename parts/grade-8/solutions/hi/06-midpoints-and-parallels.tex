\section*{अध्याय \ref{ch:g8:midpoints} --- मध्यबिंदु और समांतर रेखाएँ}

\begin{solution}{exo:g8:midpoints:1}
$M$ और $N$ त्रिभुज $RST$ की दो भुजाओं के मध्यबिंदु हैं: इसलिए
मध्यबिंदु प्रमेय (\cref{thm:g8:midpoints:direct}) से
$(MN) \parallel (ST)$ और $MN = \frac{ST}{2} = 3.5$ cm।
\end{solution}

\begin{solution}{exo:g8:midpoints:2}
$I$ एक मध्यबिंदु है और $(IJ) \parallel (BC)$: इसलिए \emph{विलोम}
(\cref{thm:g8:midpoints:converse}) से $J$, $[AC]$ का मध्यबिंदु है,
और $AJ = \frac{11}{2} = 5.5$ cm।
\end{solution}

\begin{solution}{exo:g8:midpoints:3}
नापने पर $IJ = 3.5$ cm $= \frac{BC}{2}$ मिलता है। $AIJ$ का \omterm{def:g6:measure:perimeter}{परिमाप}:
$AI = 4$, $AJ = 3$, $IJ = 3.5$: कुल $10.5$ cm --- यानी $ABC$ के
\omterm{def:g6:measure:perimeter}{परिमाप} ($8 + 6 + 7 = 21$ cm) का \omterm{def:g3:division:half}{आधा}।
\end{solution}

\begin{solution}{exo:g8:midpoints:4}
$IJK$ की हर भुजा $ABC$ की किसी भुजा की आधी है: $5$, $6$ और $8$ cm।
\omterm{def:g6:measure:perimeter}{परिमाप}: $19$ cm, यानी $10 + 12 + 16 = 38$ cm का \omterm{def:g3:division:half}{आधा}।
\end{solution}

\begin{solution}{exo:g8:midpoints:5}
मध्यबिंदु त्रिभुज \cref{prop:g8:midpoints:medial} वाले चार बराबर
त्रिभुजों में से एक है: \omterm{def:g6:measure:area}{क्षेत्रफल} $\frac{36}{4} = 9$ cm$^2$ ---
और चारों छोटे त्रिभुजों में से हर एक का \omterm{def:g6:measure:area}{क्षेत्रफल} $9$ cm$^2$ है।
\end{solution}

\begin{solution}{exo:g8:midpoints:6}
$ABC$ में मध्यबिंदु प्रमेय से: $(IJ) \parallel (BC)$, यानी
$(IJ) \parallel (BK)$, और $IJ = \frac{BC}{2} = BK$ (क्योंकि $K$,
$[BC]$ का मध्यबिंदु है)। $IJKB$ की आमने-सामने की दो भुजाएँ \omterm{def:g6:lines:perp}{समांतर}
और बराबर हैं: इसलिए वह \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} है
(\cref{thm:g7:central:recognize}, कसौटी 3)।
\end{solution}

\begin{solution}{exo:g8:midpoints:7}
सिलाई दो मध्यबिंदुओं को जोड़ती है: $\frac{6.4}{2} = 3.2$ m। ऊपर
वाले \omterm{def:g5:solids:def}{शीर्ष} से एक \omterm{def:g3:division:half}{चौथाई} पर, चित्र से लगता है कि सिलाई
$\frac{6.4}{4} = 1.6$ m की होगी, और आधार के \omterm{def:g6:lines:perp}{समांतर} --- जिसकी पुष्टि
\cref{ch:g9:thales} में थेल्स करते हैं।
\end{solution}

\begin{solution}{exo:g8:midpoints:8}
\emph{1.} $M$, $[BC]$ का मध्यबिंदु है और $(MN) \parallel (AB)$:
इसलिए मध्यबिंदु प्रमेय के विलोम से (त्रिभुज $ABC$ में, भुजा $[AC]$
पर) $N$, $[AC]$ का मध्यबिंदु है।

\emph{2.} $(MN) \parallel (AB)$ और $(AB) \perp (AC)$ ($A$ पर
\omterm{def:g6:shapes:triangles}{समकोण}): किसी $(AB)$ के \omterm{def:g6:lines:perp}{समांतर} रेखा भी $(AC)$ पर \omterm{def:g6:lines:perp}{लंब} होती है
(\cref{prop:g6:lines:facts})। इसलिए $(MN) \perp (AC)$ --- और
\omterm{def:g6:lines:objects}{रेखाखंड} $[MN]$ \omterm{def:g6:symmetry:bisector}{लंब समद्विभाजक} का वह टुकड़ा है जो फिर से दिखा देता
है कि $MA = MC$ (\cref{thm:g8:cosine:circle})।
\end{solution}

\begin{solution}{exo:g8:midpoints:9}
\emph{1.} त्रिभुज $ABC$ में $I$ और $J$, $[AB]$ तथा $[BC]$ के
मध्यबिंदु हैं: $(IJ) \parallel (AC)$ और $IJ = \frac{AC}{2}$।
त्रिभुज $ACD$ में $L$ और $K$, $[DA]$ तथा $[CD]$ के मध्यबिंदु हैं:
$(LK) \parallel (AC)$ और $LK = \frac{AC}{2}$।

\emph{2.} इसलिए $[IJ]$ और $[LK]$ \omterm{def:g6:lines:perp}{समांतर} हैं (दोनों विकर्ण $(AC)$
के \omterm{def:g6:lines:perp}{समांतर}) और बराबर भी (दोनों $AC$ के आधे): यानी $IJKL$ एक \omterm{def:g7:central:parallelogram}{समांतर
चतुर्भुज} है (\cref{thm:g7:central:recognize}, कसौटी 3) --- और यह
\emph{हर} \omterm{def:g4:geometry:polygon}{चतुर्भुज} $ABCD$ के लिए, जैसा \cref{exo:g7:central:11} में
अटकल लगाई गई थी।
\end{solution}

\begin{solution}{pb:g8:midpoints:1}
\textbf{1.} त्रिभुज चाहे जैसा बनाया जाए, तीनों माध्यिकाएँ एक ही
बिंदु से गुज़रती दिखती हैं, जो त्रिभुज के भीतर पड़ता है और
साफ़-साफ़ हर भुजा के मध्यबिंदु के, सामने वाले \omterm{def:g5:solids:def}{शीर्ष} से ज़्यादा,
पास होता है। बाक़ी समस्या यही सिद्ध करती है।

\textbf{2.} त्रिभुज $ABC$ की माध्यिका $[AK]$ लो: त्रिभुजों $ABK$
और $AKC$ के आधार $BK$ तथा $KC$ बराबर लंबाई के हैं ($K$, $[BC]$ का
मध्यबिंदु है) और एक ही रेखा $(BC)$ पर पड़े हैं, और दोनों की ऊँचाई
भी एक ही है: $A$ से $(BC)$ तक की दूरी। \omterm{def:g6:measure:area}{क्षेत्रफल} के सूत्र से हर एक
का \omterm{def:g6:measure:area}{क्षेत्रफल} आधार $\times$ ऊँचाई $\div\ 2$ है, और आधार तथा ऊँचाई
दोनों बराबर हैं: इसलिए \omterm{def:g6:measure:area}{क्षेत्रफल} भी बराबर हैं। $36$ cm$^2$
\omterm{def:g6:measure:area}{क्षेत्रफल} वाले त्रिभुज के लिए: $18$-$18$ cm$^2$ के दो टुकड़े।

\textbf{3.} $I$ और $J$, त्रिभुज $ABC$ की भुजाओं $[AB]$ तथा $[AC]$
के मध्यबिंदु हैं, इसलिए \cref{thm:g8:midpoints:direct} से:
$(IJ) \parallel (BC)$ और $IJ = \frac{BC}{2}$।

\textbf{4.} $M$ और $N$, त्रिभुज $GBC$ की भुजाओं $[GB]$ तथा $[GC]$
के मध्यबिंदु हैं, इसलिए उसी प्रमेय से: $(MN) \parallel (BC)$ और
$MN = \frac{BC}{2}$।

\textbf{5.} $(IJ)$ और $(MN)$ दोनों $(BC)$ के \omterm{def:g6:lines:perp}{समांतर} हैं, इसलिए
आपस में भी \omterm{def:g6:lines:perp}{समांतर}; और $IJ = \frac{BC}{2} = MN$। इस तरह \omterm{def:g4:geometry:polygon}{चतुर्भुज}
$IJNM$ की आमने-सामने की दो भुजाएँ, $[IJ]$ और $[NM]$, \omterm{def:g6:lines:perp}{समांतर} तथा
बराबर लंबाई की हैं: इसलिए \cref{thm:g7:central:recognize} की कसौटी
3 से $IJNM$ एक \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} है।

\textbf{6.} $G$ माध्यिका $[CI]$ पर पड़ता है, और $N$, $[GC]$ का
मध्यबिंदु है, जो उसी माध्यिका-रेखा पर पड़ा \omterm{def:g6:lines:objects}{रेखाखंड} है: इसलिए $I$,
$N$ (और $G$, $C$) सब $(CI)$ पर हैं। इसी तरह $J$, $M$ और $G$
माध्यिका-रेखा $(BJ)$ पर हैं। \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} $IJNM$ के विकर्ण
$[IN]$ और $[JM]$ हैं; वे उन्हीं दो माध्यिका-रेखाओं पर पड़े हैं, जो
$G$ पर कटती हैं --- इसलिए विकर्ण $G$ पर कटते हैं। और \omterm{def:g7:central:parallelogram}{समांतर
चतुर्भुज} की \emph{परिभाषा} से (\cref{thm:g7:central:recognize} की
कसौटी 1: विकर्णों का मध्यबिंदु एक ही होता है) $G$, $[IN]$ का
\emph{और} $[JM]$ का मध्यबिंदु है: यानी $GI = GN$ तथा $GJ = GM$।

\textbf{7.} $M$, $[GB]$ का मध्यबिंदु है, इसलिए $BM = MG$; और सवाल
6 से $MG = GJ$। यानी $BM = MG = GJ$: माध्यिका $[BJ]$ तीन बराबर
हिस्सों में कट जाती है, जिनमें से दो $BG$ के हिस्से आते हैं:
\[
BG = 2 \times GJ .
\]
इसी तरह $N$, $[GC]$ का मध्यबिंदु है और $GN = GI$, इसलिए
$CN = NG = GI$ और $CG = 2 \times GI$: माध्यिका $[CI]$ पर भी
कटान-बिंदु \omterm{def:g5:solids:def}{शीर्ष} से दो-तिहाई दूरी पर बैठता है।

\textbf{8.} इस तर्क में $[BJ]$, $[CI]$ के जोड़े की कोई ख़ास बात
इस्तेमाल नहीं हुई: $[BJ]$, $[AK]$ के जोड़े पर चलाने से वह दिखा देता
है कि उनका कटान-बिंदु भी $[BJ]$ पर $B$ से दो-तिहाई दूरी पर पड़ता
है। पर \omterm{def:g6:lines:objects}{रेखाखंड} $[BJ]$ पर $B$ से $\frac23 BJ$ दूरी वाला बिंदु
\emph{एक ही} होता है --- इसलिए $[BJ]$ और $[AK]$ का कटान-बिंदु वही
$G$ है। यानी तीनों माध्यिकाएँ $G$ से गुज़रती हैं: वे
\emph{संगामी} हैं।

\textbf{9.} $BG = \frac23 \times 9 = 6$ cm और
$GJ = \frac13 \times 9 = 3$ cm।

\textbf{10.} सवाल 8 से वही दो-तिहाई वाली गणना माध्यिका $[AK]$ पर
भी चलती है (सवाल 4--7 का तर्क $[AK]$, $[BJ]$ के जोड़े पर चलाओ):
$AG = 2 \times GK$।

\textbf{11.} त्रिभुज $GBC$ में बिंदु $K$ भुजा $[BC]$ का मध्यबिंदु
है, इसलिए $[GK]$, $GBC$ की एक माध्यिका है; और सवाल 2 से त्रिभुजों
$GBK$ तथा $GKC$ के \omterm{def:g6:measure:area}{क्षेत्रफल} बराबर हैं।

\textbf{12.} माध्यिका $[AK]$, $ABC$ को बराबर \omterm{def:g6:measure:area}{क्षेत्रफल} वाले $ABK$
और $AKC$ में काटती है (सवाल 2)। दोनों में से सवाल 11 वाला छोटा
त्रिभुज हटा दो:
\[
\text{क्षेत्रफल}(ABG) = \text{क्षेत्रफल}(ABK) - \text{क्षेत्रफल}(GBK),
\qquad
\text{क्षेत्रफल}(ACG) = \text{क्षेत्रफल}(AKC) - \text{क्षेत्रफल}(GKC),
\]
और हटाए गए \omterm{def:g6:measure:area}{क्षेत्रफल} बराबर हैं, इसलिए
$\text{क्षेत्रफल}(ABG) = \text{क्षेत्रफल}(ACG)$। वही गणना माध्यिका
$[BJ]$ के साथ --- जो $ABC$ को $ABJ$ और $CBJ$ में आधा-आधा करती है,
जबकि त्रिभुज $GAC$ की माध्यिका $[GJ]$ उसे $GAJ$ तथा $GCJ$ में
आधा-आधा करती है --- से मिलता है
$\text{क्षेत्रफल}(ABG) = \text{क्षेत्रफल}(CBG)$। इसलिए तीनों
त्रिभुजों $ABG$, $BCG$, $CAG$ के \omterm{def:g6:measure:area}{क्षेत्रफल} बराबर हैं, और चूँकि वे
मिलकर $ABC$ को ढकते हैं, हर एक का \omterm{def:g6:measure:area}{क्षेत्रफल} पूरे का एक तिहाई है।

\textbf{13.} त्रिभुज $ABG$ में बिंदु $I$ भुजा $[AB]$ का मध्यबिंदु
है, इसलिए $[GI]$, $ABG$ की \emph{\omterm{def:g5:solids:def}{शीर्ष} $G$ से खींची} माध्यिका है:
वह $ABG$ को बराबर \omterm{def:g6:measure:area}{क्षेत्रफल} वाले दो त्रिभुजों में काट देती है
(सवाल 2), यानी $AIG$ और $IBG$ में। यही $BCG$ में (माध्यिका
$[GK]$) और $CAG$ में (माध्यिका $[GJ]$) होता है। $ABC$ का हर तिहाई
आधा-आधा कट जाता है: इसलिए $G$ के चारों ओर की छहों फाँकों में से हर
एक का \omterm{def:g6:measure:area}{क्षेत्रफल} $\frac13 \times \frac12 = \frac16$ है।

\textbf{14.} माध्यिका $[AK]$ पर, $A(0,0)$ से $K(3,3)$ तक: $G$
दो-तिहाई दूरी पर बैठता है, इसलिए उसके निर्देशांक
$\left(\frac23 \times 3,\ \frac23 \times 3\right) = (2, 2)$ हैं।
माध्यिका $[BJ]$ पर, $B(6, 0)$ से $J(0, 3)$ तक: दो-तिहाई दूरी का
मतलब है खिसकाव का दो-तिहाई जोड़ना (आड़े $-6$, खड़े $+3$):
\[
\left(6 + \tfrac23 \times (-6),\ 0 + \tfrac23 \times 3\right)
= (2, 2) .
\]
वही बिंदु --- जैसा सवाल 8 ने वादा किया था। और
$\frac{0 + 6 + 0}{3} = 2$, $\frac{0 + 0 + 6}{3} = 2$: यानी \omterm{pb:g8:midpoints:1}{गुरुत्व
केंद्र} के निर्देशांक तीनों \omterm{def:g5:solids:def}{शीर्षों} के निर्देशांकों के \emph{औसत}
हैं।

\textbf{15.} सवाल 2 से किसी माध्यिका के साथ रखी छुरी की धार के
दोनों तरफ़ बराबर गत्ता होता है (बराबर \omterm{def:g6:measure:area}{क्षेत्रफल}), इसलिए त्रिभुज
तीनों माध्यिका-रेखाओं में से हर एक पर टिक जाता है। और संतुलन के
बिंदु को तीनों संतुलन-रेखाओं पर एक साथ पड़ना पड़ेगा, जबकि सवाल 8 से
तीनों माध्यिकाओं का साझा बिंदु ठीक एक ही है: $G$। पेंसिल की नोक
वहीं जाती है। (दोनों तरफ़ बराबर \omterm{def:g6:measure:area}{क्षेत्रफल} इस तर्क का मानने लायक़ रूप
है; ईमानदार रूप --- जिसमें यह भी तौला जाता है कि गत्ता \emph{कहाँ}
पड़ा है, सिर्फ़ कितना है यह नहीं --- विश्वविद्यालय वाले आयतनों का
समाकलन गणित माँगता है, और वह $G$ की ही पुष्टि करता है।)

\end{solution}
