\begin{solution}{exo:g8:speed:9}

$\frac{3 \times 15 + 2 \times 9 + 1 \times 12}{3 + 2 + 1}

बिना भार का: $\frac{15 + 9 + 12}{3} = 12$। दोनों इसलिए अलग हैं कि

\end{solution}

\begin{solution}{exo:g8:speed:10}

h। \omterm{def:g8:speed:def}{औसत चाल}: $\frac{240}{2.5} = 96$ km/h।

\emph{2.} गाड़ी $80$ km/h पर \emph{ज़्यादा समय} ($1.5$ h) बिताती है
और $120$ km/h पर कम ($1$ h): इसलिए समय के भार वाले औसत में धीमी \omterm{def:g8:speed:def}{चाल}
का भार ज़्यादा है, और वह औसत को बीच वाले $100$ से नीचे खींच लेती है।
\end{solution}

\begin{solution}{exo:g8:speed:11}

$\frac{18}{9} = 2$ h बाद, यानी $12{:}00$ बजे, मिलते हैं। तब तक
अन्ना $2 \times 5 = 10$ km चल चुकी है: इसलिए वे अन्ना के गाँव से
$10$ km दूर मिलते हैं (और बोरिस के गाँव से $8$ km --- जाँच:
$10 + 8 = 18$)।
\end{solution}

\begin{solution}{pb:g8:speed:1}
\textbf{1.} \cref{def:g8:speed:def} से $t_1 = \frac{d}{v_1}$ और
$t_2 = \frac{d}{v_2}$; और पूरा सफ़र $2d$ दूरी के लिए
$\frac{d}{v_1} + \frac{d}{v_2}$ चलता है।

\textbf{2.} हर में से $d$ बाहर निकालो और उसे काट दो:
\[
\bar v
= \frac{2d}{d\left(\frac{1}{v_1} + \frac{1}{v_2}\right)}
= \frac{2}{\ \frac{1}{v_1} + \frac{1}{v_2}\ } .
\]
भीतर उभयनिष्ठ हर लेने पर:
$\frac{1}{v_1} + \frac{1}{v_2} = \frac{v_2 + v_1}{v_1 v_2}$, और भाग
देना यानी \omterm{def:g8:fractions:inverse}{व्युत्क्रम} से गुणा करना
(\cref{thm:g8:fractions:div}):
\[
\bar v = 2 \times \frac{v_1 v_2}{v_1 + v_2}
= \frac{2\, v_1 v_2}{v_1 + v_2} .
\]

\textbf{3.} $\dfrac{2 \times 30 \times 15}{30 + 15}
= \dfrac{900}{45} = 20$ km/h, ठीक वैसे ही जैसे
\cref{ex:g8:speed:twolegs} में; और
$\dfrac{2 \times 80 \times 120}{80 + 120} = \dfrac{19\,200}{200}
= 96$ km/h, जैसा \cref{exo:g8:speed:10} में मिला था।

\textbf{4.} औसत सिर्फ़ दोनों चालों पर टिका है, $d$ पर नहीं: उन्हीं
दो चालों से $1$ km के और $100$ km के आने-जाने वाले सफ़र की \omterm{def:g8:speed:def}{औसत चाल}
बिलकुल एक ही रहती है। (इसीलिए सूत्र अलग-अलग लंबाई के सफ़रों पर जाँचा
जा सका।)

\textbf{5.} हर \omterm{def:g8:speed:def}{चाल} पर समय $T$ बीतने से दूरियाँ $v_1 T$ और $v_2 T$
होती हैं, इसलिए
\[
\bar v = \frac{v_1 T + v_2 T}{2T}
= \frac{(v_1 + v_2)\,T}{2T}
= \frac{v_1 + v_2}{2} .
\]
चालों का औसत हमेशा \emph{समय} के भार से निकलता है
(\cref{def:g8:speed:average})। बराबर समय पर दोनों चालों का भार
बराबर होता है: सादा बीच वाला मान। और बराबर दूरियों पर धीमी \omterm{def:g8:speed:def}{चाल}
\emph{ज़्यादा} समय घेरती है, इसलिए उसका भार भी ज़्यादा --- और औसत
उसी की तरफ़ सरक जाता है।

\textbf{6.} $(30, 15)$ के लिए: $H = 20$ और $A = 22.5$। $(80, 120)$
के लिए: $H = 96$ और $A = 100$। दोनों बार \omterm{def:g6:lines:perp}{समांतर} माध्य जीतता है।

\textbf{7.} उभयनिष्ठ हर $2(v_1 + v_2)$ पर:
\[
A - H
= \frac{(v_1 + v_2)^2 - 4\, v_1 v_2}{2\,(v_1 + v_2)} ,
\]
और \omterm{def:g4:fractions:def}{अंश} \cref{pb:g8:equations:1} की सर्वसमिकाओं से फैल जाता है:
\[
(v_1 + v_2)^2 - 4 v_1 v_2
= v_1^2 + 2 v_1 v_2 + v_2^2 - 4 v_1 v_2
= v_1^2 - 2 v_1 v_2 + v_2^2
= (v_1 - v_2)^2 .
\]

\textbf{8.} कोई वर्ग कभी ऋण नहीं होता, और $2(v_1 + v_2)$ धन है:
इसलिए हर धन चालों के जोड़े के लिए $A - H \geq 0$, यानी $H \leq A$।
बराबरी के लिए $(v_1 - v_2)^2 = 0$ चाहिए, यानी $v_1 = v_2$: जैसे ही
दोनों चालें अलग हुईं, आने-जाने के सफ़र का औसत बीच वाले मान से पक्के
तौर पर नीचे गिर जाता है --- यानी वह हैरानी दरअसल एक प्रमेय है।

\textbf{9.} गुणा करो:
\[
H \times A
= \frac{2\, v_1 v_2}{v_1 + v_2} \times \frac{v_1 + v_2}{2}
= v_1 v_2 ,
\]
$2$ और $v_1 + v_2$ के गुणनखंड कट जाने के बाद।

\textbf{10.} सूत्र से: $\bar v = \dfrac{2 \times 40 \times 60}
{40 + 60} = \dfrac{4800}{100} = 48$ km/h। लंबे रास्ते से:
$\frac{60}{40} = 1.5$ h और $\frac{60}{60} = 1$ h, यानी $2.5$ h में
$120$ km: $\frac{120}{2.5} = 48$ km/h। दोनों मिल जाते हैं।

\textbf{11.} $30$ km पर $30$ km/h का औसत रखने का मतलब है कुल समय
ज़्यादा से ज़्यादा $\frac{30}{30} = 1$ h। और पहले $15$ km,
$15$ km/h से चलने में ही $\frac{15}{15} = 1$ h लग जाता है: यानी
\emph{पूरा} समय-बजट ख़र्च, और आधी दौड़ अभी बाक़ी।

\textbf{12.} $90$ km/h पर दूसरा \omterm{def:g3:division:half}{आधा} हिस्सा
$\frac{15}{90} = \frac16$ h लेता है, इसलिए दौड़ $\frac76$ h की हुई
और
\[
\bar v = \frac{30}{\ 7/6\ } = \frac{180}{7} \approx 25.7
\text{ km/h} < 30 .
\]
दूसरे आधे हिस्से की \omterm{def:g8:speed:def}{चाल} चाहे जो हो, उसकी अवधि एक धन संख्या ही
होगी, इसलिए कुल समय $1$ h से ऊपर निकल जाता है और औसत $30$ km/h से
पक्के तौर पर नीचे ही रहता है। (सूत्र से: $\frac{2 \times 15
\times v_2}{15 + v_2} = 30$ रखने पर $30 v_2 = 450 + 30 v_2$, यानी
$0 = 450$ --- कोई भी \omterm{def:g8:speed:def}{चाल} $v_2$ काम नहीं करती।)

\textbf{13.} $20$ km/h का औसत $\frac{30}{20} = 1.5$ h देता है;
पहले एक घंटे के बाद $0.5$ h में $15$ km बाक़ी हैं: इसलिए उसे
$\frac{15}{0.5} = 30$ km/h चाहिए। जाँच:
$H(15, 30) = \frac{2 \times 15 \times 30}{45} = \frac{900}{45}
= 20$ km/h।

\textbf{14.} एक मिनट में नल टब का $\frac{1}{20}$ और
$\frac{1}{30}$ भाग भरते हैं; दोनों मिलकर
\[
\frac{1}{20} + \frac{1}{30}
= \frac{3}{60} + \frac{2}{60}
= \frac{5}{60} = \frac{1}{12}
\]
भाग प्रति मिनट: यानी टब $12$ min में भर जाता है। और
$H(20, 30) = \frac{2 \times 600}{50} = 24 = 2 \times 12$: दोनों नल
मिलकर अपने अकेले-अकेले समयों के \emph{\omterm{pb:g8:speed:1}{हरात्मक माध्य} का ठीक \omterm{def:g3:division:half}{आधा}}
लेते हैं --- दरें जुड़ती हैं, इसलिए यहाँ भी \omterm{pb:g8:speed:1}{हरात्मक माध्य} का ही राज
है।

\textbf{15.} दोनों तरफ़ दूरी एक ही है, इसलिए
\[
\bar v = \frac{2 \times 900 \times 700}{900 + 700}
= \frac{1\,260\,000}{1\,600} = 787.5 \text{ km/h},
\]
जो बिना हवा वाले $800$ km/h से कम है। एक वाक्य में: जहाज़ हवा के
ख़िलाफ़ उड़ने में हवा के साथ उड़ने से \emph{ज़्यादा} देर लगाता है,
इसलिए समय के भार वाले औसत में धीमे चरण का भार बड़ा हो जाता है ---
हवा आने-जाने के सफ़र को लंबा ही कर सकती है।

\end{solution}
