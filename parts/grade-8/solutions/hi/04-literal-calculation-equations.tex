\section*{अध्याय \ref{ch:g8:equations} --- अक्षर-गणना और समीकरण}

\begin{solution}{exo:g8:equations:1}
$5(2x + 3) = 10x + 15$.

$-3(4x - 1) = -12x + 3$.

$2(3x - 2) + 4(x + 5) = 6x - 4 + 4x + 20 = 10x + 16$.
\end{solution}

\begin{solution}{exo:g8:equations:2}
$(x+2)(x+7) = x^2 + 7x + 2x + 14 = x^2 + 9x + 14$.

$(3x+1)(2x-5) = 6x^2 - 15x + 2x - 5 = 6x^2 - 13x - 5$.

$(x-4)(x-6) = x^2 - 6x - 4x + 24 = x^2 - 10x + 24$.
\end{solution}

\begin{solution}{exo:g8:equations:3}
$4 \times 3 - 5 = 7$: हाँ। \quad
$2 \times 3 + 1 = 7$ और $3 + 4 = 7$: हाँ। \quad
$3^2 = 9$ और $6 \times 3 - 9 = 9$: हाँ --- $x = 3$ तीनों को हल कर
देता है।
\end{solution}

\begin{solution}{exo:g8:equations:4}
$x + 9 = 4$: $x = -5$ (जाँच: $-5 + 9 = 4$)।

$5x = -35$: $x = -7$.

$\frac x3 = 8$: $x = 24$.

$2x - 7 = 13$: $2x = 20$, $x = 10$ (जाँच: $20 - 7 = 13$)।
\end{solution}

\begin{solution}{exo:g8:equations:5}
$6x + 5 = 2x + 25$: $4x = 20$, इसलिए $x = 5$।

$9 - 4x = x - 11$: $20 = 5x$, इसलिए $x = 4$।

$3(x + 2) = 2x + 11$: $3x + 6 = 2x + 11$, इसलिए $x = 5$।
\end{solution}

\begin{solution}{exo:g8:equations:6}
$8x - 12 = 5x + 30$, इसलिए $3x = 42$ और $x = 14$। जाँच:
$4(28 - 3) = 100$ और $5(14 + 6) = 100$।
\end{solution}

\begin{solution}{exo:g8:equations:7}
``$3x - 8 = 19$'': $3x = 27$, $x = 9$.

``$2x + 5 = x + 12$'': $x = 7$.
\end{solution}

\begin{solution}{exo:g8:equations:8}
चौड़ाई $x$, लंबाई $x + 5$; \omterm{def:g6:measure:perimeter}{परिमाप}
$2\bigl(x + (x + 5)\bigr) = 4x + 10 = 58$, इसलिए $4x = 48$ और
$x = 12$: चौड़ाई $12$ cm, लंबाई $17$ cm (जाँच:
$2 \times (12 + 17) = 58$)।
\end{solution}

\begin{solution}{exo:g8:equations:9}
संख्याएँ $n - 1$, $n$, $n + 1$ हैं और उनका \omterm{def:g1:addition:def}{योगफल} $3n = 141$ है,
इसलिए $n = 47$: संख्याएँ $46$, $47$, $48$ हैं।
\end{solution}

\begin{solution}{exo:g8:equations:10}
$3x + 4 > 19$: $3x > 15$, इसलिए $x > 5$।

$7 - 2x \geq 1$: $-2x \geq -6$, और $-2$ से भाग देने पर असमिका उलट
जाती है: $x \leq 3$।

$5(x-1) < 2x + 7$: $5x - 5 < 2x + 7$, इसलिए $3x < 12$ और $x < 4$।
\end{solution}

\begin{solution}{exo:g8:equations:11}
$n$ महीनों बाद मीया के पास $140 + 15n$ होंगे और टॉम के पास
$200 + 10n$। मीया तब आगे निकलती है जब
\[
140 + 15n > 200 + 10n
\ \Longleftrightarrow\
5n > 60
\ \Longleftrightarrow\
n > 12 :
\]
यानी $13$-वें महीने से आगे।
\end{solution}

\begin{solution}{exo:g8:equations:12}
दोनों तरफ़ $20$ से गुणा करो:
\[
5(x + 3) = 4(2x - 1)
\ \Longrightarrow\
5x + 15 = 8x - 4
\ \Longrightarrow\
19 = 3x
\ \Longrightarrow\
x = \frac{19}{3}.
\]
जाँच: $\frac{19/3 + 3}{4} = \frac{28/3}{4} = \frac{28}{12} =
\frac73$, और $\frac{2 \times 19/3 - 1}{5} = \frac{35/3}{5} =
\frac{35}{15} = \frac73$। बराबर: इसलिए हल $\frac{19}{3}$ है।
\end{solution}

\begin{solution}{pb:g8:equations:1}
\textbf{1.} दुहरा वितरण नियम
(\cref{thm:g8:equations:expand}) $c = a$, $d = b$ के साथ:
\[
(a + b)(a + b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2 ,
\]
क्योंकि $ab$ और $ba$ एक ही \omterm{def:g2:mult:def}{गुणनफल} हैं।

\textbf{2.} वही फैलाव, चिह्नों का ध्यान रखते हुए:
\[
(a - b)(a - b) = a^2 - ab - ba + b^2 = a^2 - 2ab + b^2 ,
\]
\[
(a + b)(a - b) = a^2 - ab + ba - b^2 = a^2 - b^2 :
\]
इस बार आड़े पद $-ab$ और $+ba$ दुगुने होने के बजाय आपस में कट जाते
हैं।

\textbf{3.} भुजा $a + b$ वाला वर्ग चार टुकड़ों में कटता है: एक
$a \times a$ का वर्ग (यानी $a^2$), एक $b \times b$ का वर्ग (यानी
$b^2$), और $a \times b$ के \emph{दो} आयत (दोनों मिलकर $2ab$)।
सर्वसमिका $(a+b)^2 = a^2 + 2ab + b^2$ यही कहती है कि चारों
\omterm{def:g6:measure:area}{क्षेत्रफल} मिलकर बड़ा वर्ग भर देते हैं --- यह दुहरे वितरण नियम का
चित्र है, जिसमें दोनों तरफ़ एक ही तरह बाँटी गई हैं।

\textbf{4.} सर्वसमिकाओं से:
\begin{align*}
101^2 &= (100 + 1)^2 = 10\,000 + 2 \times 100 + 1 = 10\,201, \\
99^2 &= (100 - 1)^2 = 10\,000 - 200 + 1 = 9\,801, \\
49 \times 51 &= (50 - 1)(50 + 1) = 50^2 - 1^2 = 2\,499 .
\end{align*}

\textbf{5.} $a = 3$, $b = 4$ पर: $(3 + 4)^2 = 49$ है पर
$3^2 + 4^2 = 9 + 16 = 25$। ज़ोए का सूत्र चित्र वाले $a \times b$
के दोनों आयत भूल जाता है --- यानी छूटा हुआ $2ab = 24$, और सचमुच
$25 + 24 = 49$।

\textbf{6.} $1$; $1 + 3 = 4$; $1 + 3 + 5 = 9$;
$1 + 3 + 5 + 7 = 16$; $1 + 3 + 5 + 7 + 9 = 25$। उत्तर पूरे वर्ग
$1^2, 2^2, 3^2, 4^2, 5^2$ हैं।

\textbf{7.} \omterm{def:g3:numbers:evenodd}{विषम संख्याएँ} $1$ से शुरू होती हैं और $2$-$2$ के क़दमों
से बढ़ती हैं: $n$-वीं तक पहुँचने में $1$ से $2$ के $n - 1$ क़दम लगते
हैं, इसलिए वह $1 + 2(n - 1) = 2n - 1$ के बराबर है। जाँच:
$n = 1, 2, 3$ से $1, 3, 5$ मिलते हैं।

\textbf{8.} पहली सर्वसमिका से,
\[
(n + 1)^2 - n^2 = n^2 + 2n + 1 - n^2 = 2n + 1 :
\]
दो लगातार वर्गों का \omterm{ex:g1:subtraction:difference}{अंतर} ठीक $(n+1)$-वीं \omterm{def:g3:numbers:evenodd}{विषम संख्या} है (सवाल 7 में
$n$ की जगह $n + 1$ रखने पर: $2(n+1) - 1 = 2n + 1$)।

\textbf{9.} वर्गों को एक के बाद एक बनाओ। एक ही ख़ाने से शुरू करो:
$1 = 1^2$। भुजा $1$ के वर्ग से भुजा $2$ के वर्ग तक जाने के लिए
$2 \times 1 + 1 = 3$ ख़ाने जोड़ो (सवाल 8): $1 + 3 = 2^2$। भुजा $3$
तक जाने के लिए $2 \times 2 + 1 = 5$ ख़ाने जोड़ो:
$1 + 3 + 5 = 3^2$। हर नई \omterm{def:g3:numbers:evenodd}{विषम संख्या} ठीक वही L-आकार की परत है जो
अगला वर्ग पूरा करती है, इसलिए पहली $n$ \omterm{def:g3:numbers:evenodd}{विषम संख्याएँ} जोड़ने के बाद
भुजा $n$ का वर्ग पूरा हो जाता है:
$1 + 3 + \dots + (2n - 1) = n^2$।

\textbf{10.} $1$ से $99$ तक की \omterm{def:g3:numbers:evenodd}{विषम संख्याएँ} उनमें से पहली $50$
हैं ($2n - 1 = 99$ से $n = 50$), इसलिए
\[
1 + 3 + 5 + \dots + 99 = 50^2 = 2\,500 .
\]
और $1 + 3 + \dots + (2n - 1) = 400 = 20^2$ के लिए $n = 20$ चाहिए:
पहली \emph{बीस} \omterm{def:g3:numbers:evenodd}{विषम संख्याएँ} ($1$ से $39$ तक) जुड़कर $400$ बनाती
हैं।

\textbf{11.} $x^2 + 6x + 9 = x^2 + 2 \times 3 \times x + 3^2 =
(x + 3)^2$: यानी $a = x$, $b = 3$ के साथ पहली सर्वसमिका। इसलिए
\omterm{def:g8:equations:def}{समीकरण} $(x + 3)^2 = 0$ बन जाता है। कोई वर्ग \omterm{def:g1:counting:zero}{शून्य} तभी होता है जब
संख्या ख़ुद \omterm{def:g1:counting:zero}{शून्य} हो (\cref{thm:g8:negprod:rules}: शून्येतर संख्या
की दूरी शून्येतर होती है, इसलिए उसके वर्ग की दूरी भी शून्येतर
होगी)। इसलिए $x + 3 = 0$: अकेला हल $x = -3$ है।

\textbf{12.} \cref{thm:g8:negprod:rules} से \omterm{def:g2:mult:def}{गुणनफल} की दूरी दूरियों
का \omterm{def:g2:mult:def}{गुणनफल} होती है। अगर दोनों गुणक शून्येतर हों तो दोनों दूरियाँ
शून्येतर हैं, इसलिए \omterm{def:g2:mult:def}{गुणनफल} की दूरी भी शून्येतर है: तब \omterm{def:g2:mult:def}{गुणनफल} $0$ हो
ही नहीं सकता। इसलिए \omterm{def:g1:counting:zero}{शून्य} \omterm{def:g2:mult:def}{गुणनफल} कम से कम एक \omterm{def:g1:counting:zero}{शून्य} गुणक माँगता है।
$(x - 2)(x + 5) = 0$ के लिए: या तो $x - 2 = 0$ या $x + 5 = 0$,
जिससे दो हल मिलते हैं, $x = 2$ और $x = -5$।

\textbf{13.} $x^2 = 49$ से $x^2 - 49 = 0$ बनता है, और तीसरी
सर्वसमिका उसके गुणनखंड बना देती है:
\[
x^2 - 49 = x^2 - 7^2 = (x - 7)(x + 7) = 0 .
\]
सवाल 12 से $x = 7$ या $x = -7$: यानी \emph{दो} हल। इस अध्याय में
अब तक हल किया गया हर \omterm{def:g8:equations:def}{समीकरण} प्रथम घात का था और उसका ठीक एक हल था;
पर \omterm{def:g8:equations:def}{समीकरण} में वर्ग आते ही दो हल बन सकते हैं।

\textbf{14.} सब कुछ बाईं तरफ़ ले आओ: $x^2 - 6x + 9 = 0$, और दूसरी
सर्वसमिका पहचानो: $x^2 - 2 \times 3 \times x + 3^2 = (x - 3)^2$।
इसलिए \omterm{def:g8:equations:def}{समीकरण} $(x - 3)^2 = 0$ है, जो $x - 3 = 0$ पर मजबूर कर देता
है (सवाल 11 की तरह): $x = 3$ अकेला हल है ---
\cref{exo:g8:equations:3} की जाँच ने उसे ढूँढ़ा था, और गुणनखंड
बनाना सिद्ध कर देता है कि और कोई हल नहीं है।

\textbf{15.} तीसरी सर्वसमिका से,
\[
2026^2 - 2025^2 = (2026 - 2025)(2026 + 2025)
= 1 \times 4051 = 4\,051 .
\]
आम तौर पर $(n+1)^2 - n^2 = (n + 1 - n)(n + 1 + n) = 2n + 1$: दो
लगातार वर्गों का \omterm{ex:g1:subtraction:difference}{अंतर} उन दोनों संख्याओं का \omterm{def:g1:addition:def}{योगफल} होता है --- वही
$2n + 1$ जो सवाल 8 में था, बस अब पहली के बजाय तीसरी सर्वसमिका से
निकला हुआ।

\end{solution}
