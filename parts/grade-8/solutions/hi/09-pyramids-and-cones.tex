\end{solution}

\begin{solution}{exo:g8:solids:10}

\approx 18.8$ cm$^3$। घड़ी में ऐसे दो \omterm{def:g8:solids:def}{शंकु} समाते हैं, इसलिए बालू कुल

\end{solution}

\begin{solution}{exo:g8:solids:11}
\emph{1.} हाँ: सूत्र $V = \frac13 Bh$ शिखर की किसी भी जगह के लिए
चलता है, बशर्ते $h$ आधार-तल तक की \omterm{def:g6:lines:perp}{लंब} दूरी हो।
$V = \frac13 \times 36 \times 8 = 96$ cm$^3$।

\emph{2.} आधार का विकर्ण: $\sqrt{6^2 + 6^2} = 6\sqrt2$ cm। सबसे लंबा
बग़ली \omterm{def:g5:solids:def}{किनारा} उस \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} का कर्ण है जिसकी \omterm{def:g6:shapes:triangles}{समकोण} भुजाएँ
$6\sqrt2$ (आधार-तल में) और $8$ (खड़ी) हैं:
\[
\sqrt{(6\sqrt2)^2 + 8^2} = \sqrt{72 + 64} = \sqrt{136} \approx 11.7
\text{ cm}.
\]
\end{solution}

\begin{solution}{pb:g8:solids:1}
\textbf{1.} $G$ ऊपरी \omterm{def:g5:solids:def}{फलक} $EFGH$ में और दोनों बग़ली फलकों $BCGF$
तथा $DCGH$ में पड़ता है। जिन तीन फलकों में $G$ \emph{नहीं} है वे हैं
नीचे वाला $ABCD$, सामने वाला $ABFE$ और बाईं ओर वाला $ADHE$ (तीनों
में \omterm{def:g5:solids:def}{शीर्ष} $A$ साझा है, यानी $G$ के सामने वाला कोना)। तीनों \omterm{def:g8:solids:def}{पिरामिड}
हैं $ABCDG$, $ABFEG$ और $ADHEG$: हर एक का आधार \omterm{def:g5:solids:def}{घन} का कोई वर्गाकार
\omterm{def:g5:solids:def}{फलक} है और शिखर दूर वाला \omterm{def:g5:solids:def}{शीर्ष} $G$। हर एक वर्गाकार-आधार \omterm{def:g8:solids:def}{पिरामिड} है:
$5$ \omterm{def:g5:solids:def}{फलक}, $8$ \omterm{def:g5:solids:def}{किनारे}, $5$ \omterm{def:g5:solids:def}{शीर्ष} (\cref{ex:g8:solids:count})।

\textbf{2.} लंबा विकर्ण $[AG]$ उन्हीं दो \omterm{def:g5:solids:def}{शीर्षों} को जोड़ता है जो
तीनों आधारों में से किसी में नहीं हैं। रेखा $(AG)$ के चारों ओर एक
तिहाई चक्कर का घुमाव \omterm{def:g5:solids:def}{घन} को उसी पर भेज देता है ($A$ से निकलते तीनों
\omterm{def:g5:solids:def}{किनारे} --- $B$, $D$, $E$ की ओर --- आपस में चक्र में बदल जाते हैं,
और $G$ पर आते तीनों \omterm{def:g5:solids:def}{किनारे} भी)। वह आधार $ABCD$ को $ADHE$ पर, फिर
$ADHE$ को $ABFE$ पर, और फिर वापस ले जाता है: शिखर $G$ पर टिके रहते
हैं और आधार चक्र में घूमते हैं। इस तरह हर \omterm{def:g8:solids:def}{पिरामिड} अगले पर चला जाता
है: यानी तीनों एक जैसी नक़लें हैं।

\textbf{3.} \omterm{def:g5:solids:def}{घन} को भरने वाले तीन एक जैसे टुकड़े उसका \omterm{def:g6:measure:volume}{आयतन} बराबर-बराबर
बाँट लेते हैं:
\[
V_{\text{पिरामिड}} = \frac{a^3}{3} .
\]

\textbf{4.} \omterm{def:g8:solids:def}{पिरामिड} $ABCDG$ का आधार $ABCD$ है, जिसका \omterm{def:g6:measure:area}{क्षेत्रफल}
$a^2$ है, और उसका शिखर $G$ कोने $C$ के ठीक ऊपर, ऊँचाई $CG = a$ पर
बैठा है (यानी \omterm{def:g5:solids:def}{घन} का एक खड़ा \omterm{def:g5:solids:def}{किनारा})। \cref{exo:g8:solids:11} की तरह
सूत्र \omterm{def:g6:lines:perp}{लंब} ऊँचाई $h = a$ के साथ लागू होता है:
\[
V = \frac13 \times a^2 \times a = \frac{a^3}{3} ,
\]
जो सवाल 3 से पूरी तरह मेल खाता है --- यानी बँटवारा और सूत्र एक-दूसरे
की पुष्टि कर देते हैं।

\textbf{5.} $a = 6$ के लिए:
हर एक का $V = \frac{6^3}{3} = \frac{216}{3} = 72$ cm$^3$। पानी
उँडेलने वाला प्रयोग इसी बँटवारे का तरल रूप है: आधार $ABCD$ पर
ऊँचाई $a$ वाला \omterm{def:g7:areas:prism}{प्रिज़्म} ख़ुद \omterm{def:g5:solids:def}{घन} ही है, और उसमें ठीक तीन बार \omterm{def:g8:solids:def}{पिरामिड}
भरकर डालना पड़ता है --- यहाँ तो वे तीनों भराव ज्यामिति में जुड़कर \omterm{def:g5:solids:def}{घन}
भी बना देते हैं।

\textbf{6.} केंद्र $O$ को हर \omterm{def:g5:solids:def}{फलक} के चारों कोनों से जोड़ने पर छह
\omterm{def:g8:solids:def}{पिरामिड} बनते हैं, हर \omterm{def:g5:solids:def}{फलक} पर एक: वर्गाकार आधार, शिखर $O$। केंद्र
छहों फलकों के सापेक्ष एक जैसा बैठा है (\omterm{def:g5:solids:def}{घन} का कोई भी घुमाव या सममिति
$O$ को वहीं रखती है और फलकों को आपस में बदल देती है), इसलिए छहों
\omterm{def:g8:solids:def}{पिरामिड} एक जैसे हैं। हर एक की ऊँचाई $O$ से किसी \omterm{def:g5:solids:def}{फलक} तक की दूरी है:
यानी भुजा की आधी, $\frac{a}{2}$।

\textbf{7.} छह एक जैसे टुकड़े \omterm{def:g5:solids:def}{घन} को बाँट लेते हैं: हर एक का
$V = \frac{a^3}{6}$।

\textbf{8.} सूत्र से: आधार का \omterm{def:g6:measure:area}{क्षेत्रफल} $a^2$, ऊँचाई $\frac a2$:
\[
V = \frac13 \times a^2 \times \frac{a}{2} = \frac{a^3}{6} .
\]
दोनों उत्तर मिल जाते हैं।

\textbf{9.} दोनों बँटवारे \omterm{def:g5:solids:def}{घन} में से तराशे गए बहुत ख़ास अनुपातों वाले
पिरामिडों से जुड़े हैं; कोई पतला \omterm{def:g8:solids:def}{पिरामिड}, कोई टेढ़ा \omterm{def:g8:solids:def}{पिरामिड}, या कोई
\omterm{def:g8:solids:def}{शंकु} इस तरह नहीं जोड़ा जा सकता (तीन \omterm{def:g8:solids:def}{शंकु} किसी \omterm{def:g7:areas:prism}{बेलन} को नहीं भरते)।
इसीलिए इस स्तर पर \cref{thm:g8:solids:volume} \emph{मान ली} जाती
है: बँटवारे और पानी उँडेलने वाला प्रयोग $\frac13$ को पूरी तरह मानने
लायक़ बना देते हैं, और ईमानदार आम प्रमाण --- यानी समाकलन ---
हाई-स्कूल खंड में दिया गया है।

\textbf{10.} $V = \frac{6^3}{6} = 36$ cm$^3$; और सूत्र से,
$V = \frac13 \times 36 \times 3 = 36$ cm$^3$। दोनों मिलते हैं।

\textbf{11.} त्रिभुज $SAB$ में भुजाओं $[SA]$ और $[SB]$ के
मध्यबिंदुओं को जोड़ता \omterm{def:g6:lines:objects}{रेखाखंड} $(AB)$ के \omterm{def:g6:lines:perp}{समांतर} होता है और उसकी लंबाई
$\frac{AB}{2} = \frac{c}{2}$ होती है
(\cref{thm:g8:midpoints:direct})। यही हर बग़ली \omterm{def:g5:solids:def}{फलक} पर होता है,
इसलिए चारों मध्यबिंदु भुजा $\frac c2$ का एक वर्ग बनाते हैं, जिसकी
भुजाएँ आधार के \omterm{def:g6:lines:perp}{समांतर} हैं। और शिखर से गुज़रती खड़ी रेखा पर यह काट
ऊँचाई के मध्यबिंदु से गुज़रती है (फिर वही मध्यबिंदु प्रमेय, $S$,
आधार के केंद्र और आधार के किसी \omterm{def:g5:solids:def}{शीर्ष} वाले त्रिभुज में): यानी काटने
वाला तल ऊँचाई $\frac h2$ पर बैठता है।

\textbf{12.} ऊपर वाला टुकड़ा ख़ुद आधार-भुजा $\frac c2$ और ऊँचाई
$\frac h2$ वाला वर्गाकार-आधार सम \omterm{def:g8:solids:def}{पिरामिड} है (उसका आधार वही काट है
और शिखर $S$)। उसका \omterm{def:g6:measure:volume}{आयतन} है
\[
\frac13 \times \left(\frac c2\right)^2 \times \frac h2
= \frac13 \times \frac{c^2}{4} \times \frac h2
= \frac{1}{8} \times \frac{c^2 h}{3}
= \frac{V}{8} :
\]
एक बटा आठ --- यानी \omterm{def:g3:division:half}{आधा} नहीं।

\textbf{13.} छिन्नक के हिस्से बाक़ी सब आता है:
$V - \frac V8 = \frac{7V}{8}$। आधी ऊँचाई वाली काट \omterm{def:g6:measure:volume}{आयतन} को
$1 : 7$ के अनुपात में बाँटती है --- यानी नीचे वाले मामूली दिखते
हिस्से में ऊपर की नोक से सात गुना \omterm{def:g6:measure:volume}{आयतन} है।

\textbf{14.} आधी ऊँचाई से नीचे \omterm{def:g6:measure:volume}{आयतन} का $\frac78$ भाग पड़ता है। और
$V = \frac13 \times 35^2 \times 22 = \frac{26\,950}{3} \approx
8\,983$ m$^3$ (\cref{exo:g8:solids:7}) के साथ:
\[
\frac78 \times \frac{26\,950}{3} \approx 7\,860 \approx
7\,900 \text{ m}^3
\]
नज़दीकी सौ तक --- यानी ``नीचे वाला \omterm{def:g3:division:half}{आधा}'' चरण ऊपर वाले से क़रीब आठ
गुना काँच-आयतन साफ़ करता है।

\textbf{15.} हर नाप दुगुनी करने से आधार का \omterm{def:g6:measure:area}{क्षेत्रफल} $2^2 = 4$ गुना
और ऊँचाई $2$ गुनी हो जाती है, इसलिए \omterm{def:g6:measure:volume}{आयतन}
$4 \times 2 = 8 = 2^3$ गुना। $k$ से बढ़ाने पर \omterm{def:g6:measure:area}{क्षेत्रफल} $k^2$ गुने
होते हैं और एक और लंबाई $k$ गुनी: यानी \omterm{def:g6:measure:volume}{आयतन} $k^3$ के गुणक से बढ़ता
है। सवाल 12 वाला ऊपरी टुकड़ा पूरे \omterm{def:g8:solids:def}{पिरामिड} की $k = \frac12$ वाली
बढ़ाई-घटाई नक़ल है, इसलिए उसका \omterm{def:g6:measure:volume}{आयतन} पूरे का
$\left(\frac12\right)^3 = \frac18$ है --- यही एक पंक्ति वाली
व्याख्या है।

\end{solution}
