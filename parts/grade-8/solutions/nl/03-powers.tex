\section*{Hoofdstuk \ref{ch:g8:powers} --- Machten}

\begin{solution}{exo:g8:powers:1}
$2^6 = 64$; \quad $3^3 = 27$; \quad $10^5 = 100\,000$; \quad
$1^{100} = 1$; \quad $0.1^2 = 0.01$; \quad $6^0 = 1$.
\end{solution}

\begin{solution}{exo:g8:powers:2}
$(-3)^2 = 9$; \quad $-3^2 = -9$; \quad $(-1)^{15} = -1$; \quad
$(-5)^3 = -125$; \quad $\left(\frac23\right)^2 = \frac49$.
\end{solution}

\begin{solution}{exo:g8:powers:3}
$7^4 \times 7^5 = 7^9$; \quad
$\dfrac{2^9}{2^3} = 2^6$; \quad
$\left(10^3\right)^4 = 10^{12}$; \quad
$5^6 \times 5^{-2} = 5^4$; \quad
$3^4 \times 7^4 = 21^4$ (dezelfde \omterm{def:g8:powers:def}{exponent}: vermenigvuldig de grondtallen).
\end{solution}

\begin{solution}{exo:g8:powers:4}
$10^{-2} = 0.01$; \quad
$4 \times 10^3 = 4000$; \quad
$2.5 \times 10^{-4} = 0.00025$; \quad
$10^0 = 1$.
\end{solution}

\begin{solution}{exo:g8:powers:5}
$10^5$; \quad $10^{-1}$; \quad $10^{10}$; \quad $10^{-6}$.
\end{solution}

\begin{solution}{exo:g8:powers:6}
$\dfrac{10^7 \times 10^{-3}}{10^2} = \dfrac{10^4}{10^2} = 10^2 = 100$.

$\dfrac{2^5 \times 2^4}{2^6} = \dfrac{2^9}{2^6} = 2^3 = 8$.

$\left(2^2\right)^3 \times 2^{-4} = 2^6 \times 2^{-4} = 2^2 = 4$.
\end{solution}

\begin{solution}{exo:g8:powers:7}
$2^3 \times 2^4 = 2^{12}$: \emph{niet waar} --- \omterm{def:g8:powers:def}{exponenten} tel je op: $2^7$.

$5^2 + 5^3 = 5^5$: \emph{niet waar} --- er is geen regel voor sommen:
$25 + 125 = 150$, terwijl $5^5 = 3125$.

$(3^2)^4 = 3^8$: \emph{waar}.

$10^3 \times 10^3 = 100^3$: \emph{waar} --- beide zijn $10^6$ (links:
$10^{3+3}$; rechts: $(10^2)^3$).
\end{solution}

\begin{solution}{exo:g8:powers:8}
Nieuwe mensen op dag $4$: elk van de $3^3 = 27$ mensen die op dag 3 zijn ingelicht,
vertelt het aan drie anderen: $3^4 = 81$. Aan het einde van dag 4 kennen
$3 + 9 + 27 + 81 = 120$ mensen het gerucht.
\end{solution}

\begin{solution}{exo:g8:powers:9}
\emph{1.} Dikte: $2^{10} \times 10^{-4}$ m $= 1024 \times 10^{-4}$ m $\approx 0.1$ m
$= 10$ cm.

\emph{2.} $2^{42} \times 10^{-4} \approx 4.4 \times 10^{12} \times 10^{-4} =
4.4 \times 10^8$ m, groter dan $3.8 \times 10^8$ m: na $42$ (theoretische!) vouwen zou
de prop papier de maan voorbij zijn.
\end{solution}

\begin{solution}{exo:g8:powers:10}
$2^{10} = 1024$; $10^3 = 1000$; $3^6 = 729$; $5^4 = 625$. Op volgorde:
\[
5^4 < 3^6 < 10^3 < 2^{10} .
\]
\end{solution}

\begin{solution}{exo:g8:powers:11}
$2^{100} = \left(2^{10}\right)^{10} = 1024^{10}$ en
$10^{30} = \left(10^3\right)^{10} = 1000^{10}$. Omdat $1024 > 1000$, blijft de
ongelijkheid gelden als je van elk tien exemplaren met elkaar vermenigvuldigt:
$2^{100} > 10^{30}$.
\end{solution}

\begin{solution}{pb:g8:powers:1}
\textbf{1.} De korrels verdubbelen van veld tot veld vanaf $1 = 2^0$: veld $k$ bevat
$2^{k-1}$ korrels (\cref{def:g8:powers:def}). Veld $64$ bevat $2^{63}$.

\textbf{2.} $S_1 = 1$, $S_2 = 1 + 2 = 3$, $S_3 = 3 + 4 = 7$, $S_4 = 7 + 8 = 15$,
$S_5 = 15 + 16 = 31$: altijd één minder dan de volgende macht van $2$ ($2$, $4$, $8$,
$16$, $32$). Vermoeden: $S_n = 2^n - 1$.

\textbf{3.} Elke term van $S_n$ verdubbelen schuift elke macht één op
($2 \times 2^{k} = 2^{k+1}$):
\begin{align*}
S_n &= 1 + 2 + 4 + \dots + 2^{n-1}, \\
2 \times S_n &= \phantom{1 + {}} 2 + 4 + \dots + 2^{n-1} + 2^n .
\end{align*}
Trek de eerste regel van de tweede af: elke term van $2$ tot $2^{n-1}$ staat in beide en
valt weg:
\[
2 \times S_n - S_n = 2^n - 1,
\qquad\text{dus}\qquad
S_n = 2^n - 1 .
\]

\textbf{4.} Het totaal is $S_{64} = 2^{64} - 1$ korrels. Een vijfenzestigste veld zou
$2^{64}$ korrels bevatten: het hele bord draagt precies één korrel minder dan dat ene
veld.

\textbf{5.} De eerste \omterm{def:g3:division:half}{helft} bevat $S_{32} = 2^{32} - 1$ korrels. Het hele bord bevat
$2^{64} - 1$, dus bevat de tweede \omterm{def:g3:division:half}{helft}
\[
\left(2^{64} - 1\right) - \left(2^{32} - 1\right)
= 2^{64} - 2^{32}
= 2^{32} \times \left(2^{32} - 1\right)
\]
($2^{32}$ buiten haakjes gebracht, met $2^{32} \times 2^{32} = 2^{64}$,
\cref{thm:g8:powers:rules}): precies $2^{32}$ keer de eerste \omterm{def:g3:division:half}{helft}.

\textbf{6.} Met de rekenregels is $2^{64} = 2^{4 + 60} = 2^4 \times
\left(2^{10}\right)^6$. Omdat $2^{10} = 1024 > 10^3$, geldt
\[
2^{64} > 16 \times \left(10^3\right)^6 = 16 \times 10^{18}
= 1.6 \times 10^{19} .
\]

\textbf{7.} Meer dan $1.6 \times 10^{19}$ korrels van elk $5 \times 10^{-2}$ g:
\[
1.6 \times 10^{19} \times 5 \times 10^{-2}
= 8 \times 10^{17} \text{ g} .
\]
Delen door $10^6$ g per ton: meer dan $8 \times 10^{11}$ ton --- achthonderd miljard
ton.

\textbf{8.} $\dfrac{8 \times 10^{11}}{8 \times 10^{8}} = 10^3$: de koning beloofde
minstens \emph{duizend jaar} van de hele huidige wereldoogst. (De legende zegt dat zijn
raadgevers hem dat vertelden.)

\textbf{9.} $2^{19} = 2^9 \times 2^{10} = 512 \times 1024 = 524\,288 < 10^6$, terwijl
$2^{20} = \left(2^{10}\right)^2 = 1024^2 = 1\,048\,576 > 10^6$. De kleinste \omterm{def:g8:powers:def}{exponent} is
dus $n = 20$: twintig verdubbelingen gaan het miljoen voorbij.

\textbf{10.} Het loon op dag $n$ is $2^{n-1}$ cent (dag 1: $2^0 = 1$). Nu is
\[
2^{26} = 2^6 \times \left(2^{10}\right)^2
= 64 \times 1\,048\,576 = 67\,108\,864 < 10^8,
\]
\[
2^{27} = 2 \times 2^{26} = 134\,217\,728 > 10^8 .
\]
Het dagloon gaat dus voor het eerst boven $10^8$ cent als $n - 1 = 27$: op dag $28$.

\textbf{11.} $21 = 16 + 4 + 1$: de gewichten $16$, $4$ en $1$ g.
$27 = 16 + 8 + 2 + 1$: de gewichten $16$, $8$, $2$ en $1$ g.

\textbf{12.} Volgens vraag 3 wegen de gewichten $1, 2, 4, 8$ samen
$S_4 = 2^4 - 1 = 15$ g. Zonder het gewicht van $16$ g komt de koopvrouw dus niet
voorbij $15$ g: elk doelgewicht van $16$ g of meer moet het gebruiken. En een
doelgewicht van $15$ g of minder mag het niet gebruiken, want het gewicht van $16$ g
alleen is al te zwaar. De keuze van het grootste gewicht is dus \emph{gedwongen}. Wat
overblijft, is een doelgewicht van hoogstens $15$ g dat met $1, 2, 4, 8$ gevormd moet
worden --- en daar herhaalt hetzelfde argument zich: $8$ is gedwongen (gebruikt als het
resterende doel $\geq 8$ is, anders niet, want $1 + 2 + 4 = 7$), dan $4$ (want
$1 + 2 = 3$), dan $2$, dan $1$.

\textbf{13.} Als je de gedwongen keuzes volgt, is het resterende doel na elke stap
hoogstens het totaal van de resterende gewichten, dus eindigt het proces met \omterm{def:g3:division:remainder}{rest} $0$:
elk doel van $1$ tot $31$ wordt bereikt. En omdat elke keuze onderweg gedwongen was, kan
geen andere selectie gewichten hetzelfde doel bereiken: de schrijfwijze van elk getal van
$1$ tot $31$ als een \omterm{def:g1:addition:def}{som} van verschillende machten van $2$ bestaat en is \emph{uniek}.

\textbf{14.} De zes gewichten zijn samen $S_6 = 2^6 - 1 = 63$ g, en hetzelfde argument
met gedwongen keuzes dekt elk doel van $1$ tot $63$ g. Voor $45$: het doel is
$\geq 32$, dus gebruik $32$; er blijft $13 < 16$, dus $16$ overslaan; gebruik $8$ (blijft
$5$), gebruik $4$ (blijft $1$), sla $2$ over, gebruik $1$:
\[
45 = 32 + 8 + 4 + 1 = 2^5 + 2^3 + 2^2 + 2^0 .
\]

\textbf{15.} Op elk van de $64$ velden een $0$ of een $1$ kiezen komt neer op kiezen
welke machten $2^0, 2^1, \dots, 2^{63}$ in een \omterm{def:g1:addition:def}{som} meedoen --- precies het wegen van de
koopvrouw met $64$ gewichten. Het kleinste getal dat je kunt bewaren is $0$ (alle velden
op $0$); het grootste is de \omterm{def:g1:addition:def}{som} van \emph{alle} machten, en dat is het totaal van de
koning: $S_{64} = 2^{64} - 1$ (deel I). Volgens het argument met de gedwongen keuzes
wordt elk heel getal daartussen precies één keer bereikt: een $64$-bitsmachine bewaart
precies de hele getallen van $0$ tot $2^{64} - 1$.
\end{solution}
