\section*{Hoofdstuk \ref{ch:g8:equations} --- Letterrekenen en vergelijkingen}

\begin{solution}{exo:g8:equations:1}
$5(2x + 3) = 10x + 15$.

$-3(4x - 1) = -12x + 3$.

$2(3x - 2) + 4(x + 5) = 6x - 4 + 4x + 20 = 10x + 16$.
\end{solution}

\begin{solution}{exo:g8:equations:2}
$(x+2)(x+7) = x^2 + 7x + 2x + 14 = x^2 + 9x + 14$.

$(3x+1)(2x-5) = 6x^2 - 15x + 2x - 5 = 6x^2 - 13x - 5$.

$(x-4)(x-6) = x^2 - 6x - 4x + 24 = x^2 - 10x + 24$.
\end{solution}

\begin{solution}{exo:g8:equations:3}
$4 \times 3 - 5 = 7$: ja. \quad
$2 \times 3 + 1 = 7$ en $3 + 4 = 7$: ja. \quad
$3^2 = 9$ en $6 \times 3 - 9 = 9$: ja --- $x = 3$ is een oplossing van alle
drie.
\end{solution}

\begin{solution}{exo:g8:equations:4}
$x + 9 = 4$: $x = -5$ (controle: $-5 + 9 = 4$).

$5x = -35$: $x = -7$.

$\frac x3 = 8$: $x = 24$.

$2x - 7 = 13$: $2x = 20$, $x = 10$ (controle: $20 - 7 = 13$).
\end{solution}

\begin{solution}{exo:g8:equations:5}
$6x + 5 = 2x + 25$: $4x = 20$, dus $x = 5$.

$9 - 4x = x - 11$: $20 = 5x$, dus $x = 4$.

$3(x + 2) = 2x + 11$: $3x + 6 = 2x + 11$, dus $x = 5$.
\end{solution}

\begin{solution}{exo:g8:equations:6}
$8x - 12 = 5x + 30$, dus $3x = 42$ en $x = 14$. Controle:
$4(28 - 3) = 100$ en $5(14 + 6) = 100$.
\end{solution}

\begin{solution}{exo:g8:equations:7}
``$3x - 8 = 19$'': $3x = 27$, $x = 9$.

``$2x + 5 = x + 12$'': $x = 7$.

\begin{solution}{exo:g8:equations:8}
Breedte $x$, lengte $x + 5$; \omterm{def:g6:measure:perimeter}{omtrek}
$2\bigl(x + (x + 5)\bigr) = 4x + 10 = 58$, dus $4x = 48$ en $x = 12$: breedte
$12$ cm, lengte $17$ cm (controle: $2 \times (12 + 17) = 58$).
\end{solution}

\begin{solution}{exo:g8:equations:9}
De getallen zijn $n - 1$, $n$, $n + 1$ met \omterm{def:g1:addition:def}{som} $3n = 141$, dus is $n = 47$: de
getallen zijn $46$, $47$, $48$.
\end{solution}

\begin{solution}{exo:g8:equations:10}
$3x + 4 > 19$: $3x > 15$, dus $x > 5$.

$7 - 2x \geq 1$: $-2x \geq -6$; delen door $-2$ keert de ongelijkheid om:
$x \leq 3$.

$5(x-1) < 2x + 7$: $5x - 5 < 2x + 7$, dus $3x < 12$ en $x < 4$.
\end{solution}

\begin{solution}{exo:g8:equations:11}
Na $n$ maanden heeft Mia $140 + 15n$ en Tom $200 + 10n$. Mia loopt voorop
wanneer
\[
140 + 15n > 200 + 10n
\ \Longleftrightarrow\
5n > 60
\ \Longleftrightarrow\
n > 12 :
\]
vanaf de $13$de maand.
\end{solution}

\begin{solution}{exo:g8:equations:12}
Vermenigvuldig beide leden met $20$:
\[
5(x + 3) = 4(2x - 1)
\ \Longrightarrow\
5x + 15 = 8x - 4
\ \Longrightarrow\
19 = 3x
\ \Longrightarrow\
x = \frac{19}{3}.
\]
Controle: $\frac{19/3 + 3}{4} = \frac{28/3}{4} = \frac{28}{12} =
\frac73$, en $\frac{2 \times 19/3 - 1}{5} = \frac{35/3}{5} =
\frac{35}{15} = \frac73$. Gelijk: de oplossing is $\frac{19}{3}$.
\end{solution}

\begin{solution}{pb:g8:equations:1}
\textbf{1.} Dubbele distributiviteit
(\cref{thm:g8:equations:expand}) met $c = a$, $d = b$:
\[
(a + b)(a + b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2 ,
\]
want $ab$ en $ba$ zijn hetzelfde \omterm{def:g2:mult:def}{product}.

\textbf{2.} Dezelfde uitwerking, met aandacht voor de tekens:
\[
(a - b)(a - b) = a^2 - ab - ba + b^2 = a^2 - 2ab + b^2 ,
\]
\[
(a + b)(a - b) = a^2 - ab + ba - b^2 = a^2 - b^2 :
\]
deze keer heffen de kruistermen $-ab$ en $+ba$ elkaar op in plaats van te
verdubbelen.

\textbf{3.} Het vierkant met zijde $a + b$ valt uiteen in vier stukken: een
vierkant $a \times a$ (de $a^2$), een vierkant $b \times b$ (de $b^2$), en
\emph{twee} rechthoeken $a \times b$ (samen de $2ab$). Het merkwaardige \omterm{def:g2:mult:def}{product}
$(a+b)^2 = a^2 + 2ab + b^2$ zegt precies dat de vier \omterm{def:g6:measure:area}{oppervlakten} het grote
vierkant volledig vullen --- het is de tekening van de dubbele distributiviteit
met beide zijden op dezelfde manier verdeeld.

\textbf{4.} Met de merkwaardige \omterm{def:g2:mult:def}{producten}:
\begin{align*}
101^2 &= (100 + 1)^2 = 10\,000 + 2 \times 100 + 1 = 10\,201, \\
99^2 &= (100 - 1)^2 = 10\,000 - 200 + 1 = 9\,801, \\
49 \times 51 &= (50 - 1)(50 + 1) = 50^2 - 1^2 = 2\,499 .
\end{align*}

\textbf{5.} Voor $a = 3$, $b = 4$: $(3 + 4)^2 = 49$, maar
$3^2 + 4^2 = 9 + 16 = 25$. De formule van Zoë vergeet de twee rechthoeken
$a \times b$ van de tekening --- de ontbrekende $2ab = 24$, en inderdaad is
$25 + 24 = 49$.

\textbf{6.} $1$; $1 + 3 = 4$; $1 + 3 + 5 = 9$;
$1 + 3 + 5 + 7 = 16$; $1 + 3 + 5 + 7 + 9 = 25$. De uitkomsten zijn de kwadraten
$1^2, 2^2, 3^2, 4^2, 5^2$.

\textbf{7.} De \omterm{def:g3:numbers:evenodd}{oneven} getallen beginnen bij $1$ en gaan met stappen van $2$
omhoog: het $n$-de bereik je na $n - 1$ stappen van $2$ vanaf $1$, dus is het
$1 + 2(n - 1) = 2n - 1$. Controle: $n = 1, 2, 3$ geven $1, 3, 5$.

\textbf{8.} Volgens het eerste merkwaardige \omterm{def:g2:mult:def}{product} is
\[
(n + 1)^2 - n^2 = n^2 + 2n + 1 - n^2 = 2n + 1 :
\]
het \omterm{ex:g1:subtraction:difference}{verschil} tussen twee opeenvolgende kwadraten is precies het $(n+1)$-de
\omterm{def:g3:numbers:evenodd}{oneven getal} (vraag 7 met $n + 1$ in de plaats van $n$: $2(n+1) - 1 = 2n + 1$).

\textbf{9.} Bouw de vierkanten na elkaar op. Begin met één vakje: $1 = 1^2$. Om
van het vierkant met zijde $1$ naar het vierkant met zijde $2$ te gaan, voeg je
$2 \times 1 + 1 = 3$ vakjes toe (vraag 8): $1 + 3 = 2^2$. Om naar zijde $3$ te
gaan, voeg je $2 \times 2 + 1 = 5$ vakjes toe: $1 + 3 + 5 = 3^2$. Elk nieuw
\omterm{def:g3:numbers:evenodd}{oneven getal} is precies de L-vormige laag die het volgende vierkant afmaakt, dus
is na het toevoegen van de eerste $n$ \omterm{def:g3:numbers:evenodd}{oneven} getallen het vierkant met zijde $n$
volledig: $1 + 3 + \dots + (2n - 1) = n^2$.

\textbf{10.} De \omterm{def:g3:numbers:evenodd}{oneven} getallen van $1$ tot $99$ zijn de eerste $50$ ervan
($2n - 1 = 99$ geeft $n = 50$), dus is
\[
1 + 3 + 5 + \dots + 99 = 50^2 = 2\,500 .
\]
En $1 + 3 + \dots + (2n - 1) = 400 = 20^2$ vraagt $n = 20$: de eerste
\emph{twintig} \omterm{def:g3:numbers:evenodd}{oneven} getallen ($1$ tot en met $39$) zijn samen $400$.

\textbf{11.} $x^2 + 6x + 9 = x^2 + 2 \times 3 \times x + 3^2 = (x + 3)^2$: het
eerste merkwaardige \omterm{def:g2:mult:def}{product} met $a = x$, $b = 3$. De \omterm{def:g8:equations:def}{vergelijking} luidt dus
$(x + 3)^2 = 0$. Een kwadraat is alleen \omterm{def:g1:counting:zero}{nul} als het getal zelf \omterm{def:g1:counting:zero}{nul} is
(\cref{thm:g8:negprod:rules}: een getal verschillend van \omterm{def:g1:counting:zero}{nul} heeft een afstand
verschillend van \omterm{def:g1:counting:zero}{nul}, en dus ook zijn kwadraat). Bijgevolg is $x + 3 = 0$: de
enige oplossing is $x = -3$.

\textbf{12.} Volgens \cref{thm:g8:negprod:rules} is de afstand van een \omterm{def:g2:mult:def}{product}
het \omterm{def:g2:mult:def}{product} van de afstanden. Zijn beide factoren verschillend van \omterm{def:g1:counting:zero}{nul}, dan zijn
beide afstanden verschillend van \omterm{def:g1:counting:zero}{nul}, en dus ook de afstand van het \omterm{def:g2:mult:def}{product}: het
\omterm{def:g2:mult:def}{product} kan dan geen $0$ zijn. Een \omterm{def:g2:mult:def}{product} dat \omterm{def:g1:counting:zero}{nul} is, dwingt dus minstens één
factor \omterm{def:g1:counting:zero}{nul} te zijn. Voor $(x - 2)(x + 5) = 0$: ofwel $x - 2 = 0$, ofwel
$x + 5 = 0$, wat de twee oplossingen $x = 2$ en $x = -5$ geeft.

\textbf{13.} $x^2 = 49$ wordt $x^2 - 49 = 0$, en het derde merkwaardige \omterm{def:g2:mult:def}{product}
ontbindt het:
\[
x^2 - 49 = x^2 - 7^2 = (x - 7)(x + 7) = 0 .
\]
Volgens vraag 12 is $x = 7$ of $x = -7$: \emph{twee} oplossingen. Elke
\omterm{def:g8:equations:def}{vergelijking} die in dit hoofdstuk tot nu toe is opgelost, was van de eerste graad
en had precies één oplossing; een kwadraat in de \omterm{def:g8:equations:def}{vergelijking} kan er twee
opleveren.

\textbf{14.} Breng alles naar het linkerlid: $x^2 - 6x + 9 = 0$, en herken het
tweede merkwaardige \omterm{def:g2:mult:def}{product}: $x^2 - 2 \times 3 \times x + 3^2 = (x - 3)^2$. De
\omterm{def:g8:equations:def}{vergelijking} is dus $(x - 3)^2 = 0$, wat $x - 3 = 0$ dwingt (net als in vraag
11): $x = 3$ is de enige oplossing --- de controle van
\cref{exo:g8:equations:3} vond die oplossing, en de ontbinding bewijst dat er
geen andere zijn.

\textbf{15.} Volgens het derde merkwaardige \omterm{def:g2:mult:def}{product} is
\[
2026^2 - 2025^2 = (2026 - 2025)(2026 + 2025)
= 1 \times 4051 = 4\,051 .
\]
In het algemeen is $(n+1)^2 - n^2 = (n + 1 - n)(n + 1 + n) = 2n + 1$: het
\omterm{ex:g1:subtraction:difference}{verschil} van twee opeenvolgende kwadraten is de \omterm{def:g1:addition:def}{som} van de twee getallen --- weer
dezelfde $2n + 1$ als in vraag 8, nu verkregen uit het derde merkwaardige \omterm{def:g2:mult:def}{product}
in plaats van uit het eerste.
\end{solution}
