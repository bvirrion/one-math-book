\section*{Hoofdstuk \ref{ch:g8:negprod} --- Negatieve getallen vermenigvuldigen}

\begin{solution}{exo:g8:negprod:1}
$(-8) \times (+7) = -56$; \quad
$(-9) \times (-6) = +54$; \quad
$(+12) \times (-0.5) = -6$; \quad
$(-1) \times (-1) \times (-1) = -1$ (drie negatieve factoren: \omterm{def:g3:numbers:evenodd}{oneven}).
\end{solution}

\begin{solution}{exo:g8:negprod:2}
$(-63) \div (+9) = -7$; \quad
$(-8) \div (-16) = +0.5$; \quad
$\dfrac{45}{-5} = -9$; \quad
$\dfrac{-7.2}{-8} = +0.9$.
\end{solution}

\begin{solution}{exo:g8:negprod:3}
Drie negatieve factoren: \emph{negatief}. \quad
$(-1)^{10}$: tien negatieve factoren, \omterm{def:g3:numbers:evenodd}{even}: \emph{positief} (het is $1$). \quad
$(-2)^7$: zeven factoren, \omterm{def:g3:numbers:evenodd}{oneven}: \emph{negatief}. \quad
$(-4) \times 0 \times (-6) = 0$: niet positief en niet negatief.
\end{solution}

\begin{solution}{exo:g8:negprod:4}
$(-2)^4 = +16$; \quad $-2^4 = -16$ (het minteken wordt niet gekwadrateerd); \quad
$(-3)^3 = -27$; \quad $(-1)^{2026} = +1$ (\omterm{def:g3:numbers:evenodd}{even} exponent).
\end{solution}

\begin{solution}{exo:g8:negprod:5}
$7 + 2 \times (-6) = 7 + (-12) = -5$.

$(-4) \times (5 - 9) = (-4) \times (-4) = +16$.

$18 \div (-3) - (-2) = -6 + 2 = -4$.
\end{solution}

\begin{solution}{exo:g8:negprod:6}
$\dfrac{(-5) \times (+8)}{-4} = \dfrac{-40}{-4} = +10$.

$\dfrac{(-3) \times (-6)}{(-2) \times (+9)} = \dfrac{+18}{-18} = -1$.
\end{solution}

\begin{solution}{exo:g8:negprod:7}
Voor $x = -3$: \ $5x = -15$; \quad $x^2 = (-3)^2 = 9$; \quad $-x^2 = -9$; \quad
$2x^2 + 4x = 18 - 12 = 6$; \quad $(2x)^2 = (-6)^2 = 36$.
\end{solution}

\begin{solution}{exo:g8:negprod:8}
$(-6) \times (-7) = 42$; \quad
$10 \div (-4) = -2.5$; \quad
$(-5) \times \frac15 = -1$, dus is het ontbrekende getal $+\frac15 = 0.2$.
\end{solution}

\begin{solution}{exo:g8:negprod:9}
\emph{1. Niet waar}: $0.5 \times 0.5 = 0.25$ is kleiner dan
$0.5 + 0.5 = 1$ (of: $2 \times 3 = 6 > 5$, maar de bewering moet altijd gelden).

\emph{2. Waar}: een kwadraat is een \omterm{def:g2:mult:def}{product} van twee getallen met hetzelfde teken, en

\begin{solution}{exo:g8:negprod:10}
Zij $r$ het aantal goede antwoorden; dan zijn er $20 - r$ verkeerd, en
\[
5r - 3(20 - r) = 36
\ \Longrightarrow\
5r - 60 + 3r = 36
\ \Longrightarrow\
8r = 96
\ \Longrightarrow\
r = 12 .
\]
Twaalf goede antwoorden (en acht verkeerde: controle,
$60 - 24 = 36$).
\end{solution}

\begin{solution}{exo:g8:negprod:11}
\omterm{def:g1:counting:zero}{Nul} keer wat dan ook is \omterm{def:g1:counting:zero}{nul}, dus $(-1) \times 0 = 0$. Schrijf $0 = 1 + (-1)$ en werk
uit:
\[
0 = (-1) \times \bigl(1 + (-1)\bigr)
= (-1) \times 1 + (-1) \times (-1)
= -1 + (-1) \times (-1).
\]
Het getal $(-1) \times (-1)$, opgeteld bij $-1$, geeft $0$: het moet dus het
\omterm{def:g7:negatives:def}{tegengestelde} van $-1$ zijn, en dat is $+1$. De distributiviteit laat geen andere keuze.
\end{solution}

\begin{solution}{pb:g8:negprod:1}
\textbf{1.} Omdat $0 = 0 + 0$, geeft de distributiviteit
(\cref{thm:g7:priorities:distributivity})
\[
0 \times a = (0 + 0) \times a = 0 \times a + 0 \times a .
\]
$0 \times a$ bij zichzelf optellen verandert dus niets --- en het enige getal dat je
zonder gevolgen kunt optellen, is $0$. (Rechtstreeks: trek $0 \times a$ van beide
kanten af.) Bijgevolg is $0 \times a = 0$.

\textbf{2.} Uitwerken, en dan $1 + (-1) = 0$ en vraag 1 gebruiken:
\[
a + (-1) \times a
= 1 \times a + (-1) \times a
= \bigl(1 + (-1)\bigr) \times a
= 0 \times a = 0 .
\]
Dus is $(-1) \times a$ het getal dat, opgeteld bij $a$, $0$ geeft: precies het
\omterm{def:g7:negatives:def}{tegengestelde} van $a$, en dus $(-1) \times a = -a$.
\Cref{exo:g8:negprod:11} is het bijzondere geval $a = -1$: daar is
$(-1) \times (-1) = -(-1) = +1$.

\textbf{3.} Schrijf $-a = (-1) \times a$ (vraag 2) en groepeer de factoren opnieuw:
\[
(-a) \times b
= \bigl((-1) \times a\bigr) \times b
= (-1) \times (a \times b)
= -(a \times b) .
\]

\textbf{4.} Pas vraag 3 twee keer toe (één keer op elke factor):
\[
(-a) \times (-b)
= -\bigl(a \times (-b)\bigr)
= -\bigl(-(a \times b)\bigr)
= a \times b ,
\]
want het \omterm{def:g7:negatives:def}{tegengestelde} van het \omterm{def:g7:negatives:def}{tegengestelde} van een getal is het getal zelf.

\textbf{5.} Schrijf $a = +d$ of $a = -d$ en $b = +e$ of $b = -e$, waarbij $d$ en $e$ de
afstanden tot \omterm{def:g1:counting:zero}{nul} zijn. De vier gevallen:
\[
(+d) \times (+e) = de, \quad
(+d) \times (-e) = -de, \quad
(-d) \times (+e) = -de, \quad
(-d) \times (-e) = +de,
\]
de middelste twee volgens vraag 3, de laatste volgens vraag 4. In elk geval is de
afstand van de uitkomst $d \times e$, het \omterm{def:g2:mult:def}{product} van de afstanden, en is het teken dat
wat de tabel van \cref{thm:g8:negprod:rules} aankondigt: dezelfde tekens positief,
\omterm{def:g7:negatives:def}{tegengestelde} tekens negatief.

\textbf{6.} $(-1)^2 = +1$, $(-1)^3 = -1$, $(-1)^4 = +1$, $(-1)^5 = -1$. In het algemeen
is $(-1)^n$ een \omterm{def:g2:mult:def}{product} van $n$ negatieve factoren met elk afstand $1$: volgens
\cref{prop:g8:negprod:many} is het $+1$ als $n$ \omterm{def:g3:numbers:evenodd}{even} is en $-1$ als $n$ \omterm{def:g3:numbers:evenodd}{oneven} is.

\textbf{7.} Er zijn tien negatieve factoren --- een \omterm{def:g3:numbers:evenodd}{even} aantal --- dus is het \omterm{def:g2:mult:def}{product}
\emph{positief} (\cref{prop:g8:negprod:many}). Zijn afstand is
$1 \times 2 \times 3 \times \dots \times 10$.

\textbf{8.} Volgens vraag 4 is $(-a)^2 = (-a) \times (-a) = a \times a = a^2$. En dan
volgens vraag 3:
\[
(-a)^3 = (-a)^2 \times (-a) = a^2 \times (-a) = -(a^2 \times a)
= -a^3 .
\]
Het minteken verdwijnt bij \omterm{def:g3:numbers:evenodd}{even} exponenten en overleeft bij \omterm{def:g3:numbers:evenodd}{oneven} exponenten, in \omterm{def:g6:lines:objects}{lijn}
met \cref{ex:g8:negprod:powers}.

\textbf{9.} Is $a = 0$, dan is $a^2 = 0$. Anders hebben $a$ en $a$ \emph{hetzelfde}
teken, dus is $a^2 = a \times a$ positief
(\cref{thm:g8:negprod:rules}). Een kwadraat is dus nooit negatief; en omdat $-4$
negatief is, voldoet geen enkel getal $x$ met een teken aan $x^2 = -4$.

\textbf{10.} Zoë heeft gelijk. De exponenten $2026$ en $2027$ hebben verschillende
pariteit, dus is $(-1)^{2026} = +1$ en $(-1)^{2027} = -1$, en die zijn samen $0$. In het
algemeen is van twee opeenvolgende exponenten er één \omterm{def:g3:numbers:evenodd}{even} en één \omterm{def:g3:numbers:evenodd}{oneven}, dus zijn de
twee machten $+1$ en $-1$ in een of andere volgorde: \omterm{def:g7:negatives:def}{tegengestelde} getallen, met \omterm{def:g1:addition:def}{som}
altijd $0$.

\textbf{11.} $A_3 = 1 - 2 + 3 = 2$, en dan $A_4 = A_3 - 4 = -2$,
$A_5 = A_4 + 5 = 3$, $A_6 = A_5 - 6 = -3$, $A_7 = A_6 + 7 = 4$. Vermoeden:
$A_n = -\frac{n}{2}$ voor \omterm{def:g3:numbers:evenodd}{even} $n$, en $A_n = \frac{n+1}{2}$ voor \omterm{def:g3:numbers:evenodd}{oneven} $n$.

\textbf{12.} Voor \omterm{def:g3:numbers:evenodd}{even} $n$ gaan de termen volledig in paren:
\[
A_n = (1 - 2) + (3 - 4) + \dots + \bigl((n-1) - n\bigr),
\]
en elk paar is $-1$. Er zijn $\frac{n}{2}$ paren (twee termen per paar, $n$ termen in
totaal), dus is $A_n = \frac{n}{2} \times (-1) = -\frac{n}{2}$.

\textbf{13.} Voor \omterm{def:g3:numbers:evenodd}{oneven} $n$ is de \omterm{def:g1:addition:def}{som} $A_n$ de \omterm{def:g1:addition:def}{som} $A_{n-1}$ van de eerste $n - 1$
termen, plus de laatste term, die $+n$ is omdat $n$ \omterm{def:g3:numbers:evenodd}{oneven} is. Omdat $n - 1$ \omterm{def:g3:numbers:evenodd}{even} is,
geeft vraag 12
\[
A_n = A_{n-1} + n = -\frac{n-1}{2} + n
= \frac{-(n-1) + 2n}{2} = \frac{n+1}{2} .
\]

\textbf{14.} $A_{100} = -\frac{100}{2} = -50$, \quad
$A_{101} = \frac{102}{2} = 51$, \quad
$A_{2026} = -\frac{2026}{2} = -1013$.

\textbf{15.} Geen verrassing: van $A_{100}$ naar $A_{101}$ komt er precies één term bij,
namelijk $+101$, dus moest het \omterm{ex:g1:subtraction:difference}{verschil} $101$ zijn. Om dezelfde reden is
$A_{2027} - A_{2026} = +2027$ --- zonder $A_{2027}$ ook maar uit te rekenen.
\end{solution}
