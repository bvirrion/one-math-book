\section*{Hoofdstuk \ref{ch:g8:midpoints} --- Middens en parallellen}

\begin{solution}{exo:g8:midpoints:1}
$M$ en $N$ zijn de middens van twee zijden van de driehoek $RST$: volgens de
middenparallelstelling (\cref{thm:g8:midpoints:direct}) is
$(MN) \parallel (ST)$ en $MN = \frac{ST}{2} = 3.5$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:2}
$I$ is een midden en $(IJ) \parallel (BC)$: volgens de \emph{omgekeerde} stelling
(\cref{thm:g8:midpoints:converse}) is $J$ het midden van $[AC]$, dus
$AJ = \frac{11}{2} = 5.5$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:3}
De meting geeft $IJ = 3.5$ cm $= \frac{BC}{2}$. \omterm{def:g6:measure:perimeter}{Omtrek} van $AIJ$: $AI = 4$,
$AJ = 3$, $IJ = 3.5$, samen $10.5$ cm --- de \omterm{def:g3:division:half}{helft} van de \omterm{def:g6:measure:perimeter}{omtrek} van $ABC$
($8 + 6 + 7 = 21$ cm).
\end{solution}

\begin{solution}{exo:g8:midpoints:4}
Elke zijde van $IJK$ is de \omterm{def:g3:division:half}{helft} van een zijde van $ABC$: $5$, $6$ en $8$ cm.
\omterm{def:g6:measure:perimeter}{Omtrek}: $19$ cm, de \omterm{def:g3:division:half}{helft} van $10 + 12 + 16 = 38$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:5}
De middendriehoek is een van de vier gelijke driehoeken van
\cref{prop:g8:midpoints:medial}: \omterm{def:g6:measure:area}{oppervlakte} $\frac{36}{4} = 9$ cm$^2$ --- en elk
van de vier kleine driehoeken heeft \omterm{def:g6:measure:area}{oppervlakte} $9$ cm$^2$.
\end{solution}

\begin{solution}{exo:g8:midpoints:6}
Volgens de middenparallelstelling in $ABC$: $(IJ) \parallel (BC)$, dat wil zeggen
$(IJ) \parallel (BK)$, en $IJ = \frac{BC}{2} = BK$ (want $K$ is het midden van
$[BC]$). Twee overstaande zijden van $IJKB$ zijn \omterm{def:g6:lines:perp}{parallel} en gelijk: het is een
\omterm{def:g7:central:parallelogram}{parallellogram} (\cref{thm:g7:central:recognize}, kenmerk 3).
\end{solution}

\begin{solution}{exo:g8:midpoints:7}
De naad verbindt twee middens: $\frac{6.4}{2} = 3.2$ m. Op een \omterm{def:g3:division:half}{kwart} vanaf het
bovenste \omterm{def:g5:solids:def}{hoekpunt} suggereert de tekening een naad van $\frac{6.4}{4} = 1.6$ m,
\omterm{def:g6:lines:perp}{parallel} met de basis --- wat Thales in \cref{ch:g9:thales} bevestigt.
\end{solution}

\begin{solution}{exo:g8:midpoints:8}
\emph{1.} $M$ is het midden van $[BC]$ en $(MN) \parallel (AB)$: volgens de
omgekeerde middenparallelstelling (in de driehoek $ABC$, zijde $[AC]$) is $N$ het
midden van $[AC]$.

\emph{2.} $(MN) \parallel (AB)$ en $(AB) \perp (AC)$ (\omterm{def:g3:shapes:rightangle}{rechte hoek} in $A$): een
\omterm{def:g6:lines:objects}{lijn} \omterm{def:g6:lines:perp}{parallel} met $(AB)$ staat ook \omterm{def:g6:lines:perp}{loodrecht} op $(AC)$
(\cref{prop:g6:lines:facts}). Dus is $(MN) \perp (AC)$ --- het \omterm{def:g6:lines:objects}{lijnstuk} $[MN]$
ligt op de \omterm{def:g6:symmetry:bisector}{middelloodlijn} van $[AC]$, wat nog eens laat zien dat $MA = MC$
(\cref{thm:g8:cosine:circle}).
\end{solution}

\begin{solution}{exo:g8:midpoints:9}
\emph{1.} In de driehoek $ABC$ zijn $I$ en $J$ de middens van $[AB]$ en $[BC]$:
$(IJ) \parallel (AC)$ en $IJ = \frac{AC}{2}$. In de driehoek $ACD$ zijn $L$ en
$K$ de middens van $[DA]$ en $[CD]$: $(LK) \parallel (AC)$ en
$LK = \frac{AC}{2}$.

\emph{2.} Dus zijn $[IJ]$ en $[LK]$ \omterm{def:g6:lines:perp}{parallel} (beide \omterm{def:g6:lines:perp}{parallel} met de diagonaal
$(AC)$) en gelijk (beide de \omterm{def:g3:division:half}{helft} van $AC$): $IJKL$ is een \omterm{def:g7:central:parallelogram}{parallellogram}
(\cref{thm:g7:central:recognize}, kenmerk 3) --- voor \emph{elke} \omterm{def:g4:geometry:polygon}{vierhoek}
$ABCD$, zoals in \cref{exo:g7:central:11} werd vermoed.
\end{solution}

\begin{solution}{pb:g8:midpoints:1}
\textbf{1.} Welke driehoek je ook tekent, de drie \omterm{pb:g8:midpoints:1}{zwaartelijnen} lijken door één
punt te gaan, dat binnen de driehoek ligt en zichtbaar dichter bij het midden van
elke zijde dan bij het overstaande \omterm{def:g5:solids:def}{hoekpunt}. De \omterm{def:g3:division:remainder}{rest} van de opgave bewijst het.

\textbf{2.} Neem de \omterm{pb:g8:midpoints:1}{zwaartelijn} $[AK]$ van de driehoek $ABC$: de driehoeken $ABK$
en $AKC$ hebben basissen $BK$ en $KC$ van dezelfde lengte ($K$ is het midden van
$[BC]$) op dezelfde \omterm{def:g6:lines:objects}{lijn} $(BC)$, en dezelfde hoogte: de afstand van $A$ tot
$(BC)$. Volgens de formule voor de \omterm{def:g6:measure:area}{oppervlakte} heeft elk als \omterm{def:g6:measure:area}{oppervlakte}
basis $\times$ hoogte $\div\ 2$, met gelijke basissen en gelijke hoogten: de
\omterm{def:g6:measure:area}{oppervlakten} zijn gelijk. Voor een driehoek met \omterm{def:g6:measure:area}{oppervlakte} $36$ cm$^2$: twee
stukken van elk $18$ cm$^2$.

\textbf{3.} $I$ en $J$ zijn de middens van de zijden $[AB]$ en $[AC]$ van de
driehoek $ABC$, dus volgens \cref{thm:g8:midpoints:direct}:
$(IJ) \parallel (BC)$ en $IJ = \frac{BC}{2}$.

\textbf{4.} $M$ en $N$ zijn de middens van de zijden $[GB]$ en $[GC]$ van de
driehoek $GBC$, dus volgens dezelfde stelling: $(MN) \parallel (BC)$ en
$MN = \frac{BC}{2}$.

\textbf{5.} Zowel $(IJ)$ als $(MN)$ is \omterm{def:g6:lines:perp}{parallel} met $(BC)$, en dus zijn ze
\omterm{def:g6:lines:perp}{parallel} met elkaar; en $IJ = \frac{BC}{2} = MN$. De \omterm{def:g4:geometry:polygon}{vierhoek} $IJNM$ heeft dus
twee overstaande zijden, $[IJ]$ en $[NM]$, die \omterm{def:g6:lines:perp}{parallel} en even lang zijn:
volgens kenmerk 3 van \cref{thm:g7:central:recognize} is $IJNM$ een
\omterm{def:g7:central:parallelogram}{parallellogram}.

\textbf{6.} $G$ ligt op de \omterm{pb:g8:midpoints:1}{zwaartelijn} $[CI]$, en $N$ is het midden van $[GC]$,
een \omterm{def:g6:lines:objects}{lijnstuk} op diezelfde \omterm{pb:g8:midpoints:1}{zwaartelijn}: $I$, $N$ (en $G$, $C$) liggen dus alle op
$(CI)$. Net zo liggen $J$, $M$ en $G$ op de \omterm{pb:g8:midpoints:1}{zwaartelijn} $(BJ)$. In het
\omterm{def:g7:central:parallelogram}{parallellogram} $IJNM$ zijn de diagonalen $[IN]$ en $[JM]$; ze liggen op de twee
\omterm{pb:g8:midpoints:1}{zwaartelijnen}, die elkaar in $G$ snijden --- dus snijden de diagonalen elkaar in
$G$. Volgens de \emph{definitie} van een \omterm{def:g7:central:parallelogram}{parallellogram} (kenmerk 1 van
\cref{thm:g7:central:recognize}: de diagonalen hebben hetzelfde midden) is $G$
het midden van $[IN]$ \emph{en} van $[JM]$: $GI = GN$ en $GJ = GM$.

\textbf{7.} $M$ is het midden van $[GB]$, dus $BM = MG$; en $MG = GJ$ volgens
vraag 6. Bijgevolg is $BM = MG = GJ$: de \omterm{pb:g8:midpoints:1}{zwaartelijn} $[BJ]$ wordt in drie gelijke
stukken verdeeld, waarvan $BG$ er twee neemt:
\[
BG = 2 \times GJ .
\]
Symmetrisch is $N$ het midden van $[GC]$ en $GN = GI$, dus $CN = NG = GI$ en
$CG = 2 \times GI$: ook op de \omterm{pb:g8:midpoints:1}{zwaartelijn} $[CI]$ ligt het snijpunt op twee derde
van de weg vanaf het \omterm{def:g5:solids:def}{hoekpunt}.

\textbf{8.} Het betoog gebruikte niets bijzonders van het paar $[BJ]$, $[CI]$:
uitgevoerd op het paar $[BJ]$, $[AK]$ laat het zien dat ook hun snijpunt op
$[BJ]$ ligt, op twee derde van de weg vanaf $B$. Maar er is slechts \emph{één}
punt van het \omterm{def:g6:lines:objects}{lijnstuk} $[BJ]$ op afstand $\frac23 BJ$ van $B$ --- dus is het
snijpunt van $[BJ]$ en $[AK]$ precies diezelfde $G$. De drie \omterm{pb:g8:midpoints:1}{zwaartelijnen} gaan
dus alle door $G$: ze zijn \emph{concurrent}.

\textbf{9.} $BG = \frac23 \times 9 = 6$ cm en $GJ = \frac13 \times 9 = 3$ cm.

\textbf{10.} Volgens vraag 8 geldt dezelfde berekening met twee derde op de
\omterm{pb:g8:midpoints:1}{zwaartelijn} $[AK]$ (voer het betoog van vragen 4--7 uit met het paar $[AK]$,
$[BJ]$): $AG = 2 \times GK$.

\textbf{11.} In de driehoek $GBC$ is het punt $K$ het midden van de zijde $[BC]$,
dus is $[GK]$ een \omterm{pb:g8:midpoints:1}{zwaartelijn} van $GBC$; volgens vraag 2 hebben de driehoeken
$GBK$ en $GKC$ gelijke \omterm{def:g6:measure:area}{oppervlakte}.

\textbf{12.} De \omterm{pb:g8:midpoints:1}{zwaartelijn} $[AK]$ verdeelt $ABC$ in $ABK$ en $AKC$ met gelijke
\omterm{def:g6:measure:area}{oppervlakte} (vraag 2). Haal uit elk zijn kleine driehoek van vraag 11 weg:
\[
\text{opp}(ABG) = \text{opp}(ABK) - \text{opp}(GBK),
\qquad
\text{opp}(ACG) = \text{opp}(AKC) - \text{opp}(GKC),
\]
en de weggehaalde \omterm{def:g6:measure:area}{oppervlakten} zijn gelijk, dus is
$\text{opp}(ABG) = \text{opp}(ACG)$. Dezelfde berekening langs de \omterm{pb:g8:midpoints:1}{zwaartelijn}
$[BJ]$ --- die $ABC$ in twee gelijke \omterm{def:g3:division:half}{helften} $ABJ$ en $CBJ$ verdeelt, terwijl
$[GJ]$, een \omterm{pb:g8:midpoints:1}{zwaartelijn} van de driehoek $GAC$, die in $GAJ$ en $GCJ$ halveert ---
geeft $\text{opp}(ABG) = \text{opp}(CBG)$. De drie driehoeken $ABG$, $BCG$, $CAG$
hebben dus gelijke \omterm{def:g6:measure:area}{oppervlakte}, en omdat ze samen $ABC$ betegelen, heeft elk als
\omterm{def:g6:measure:area}{oppervlakte} een derde van het geheel.

\textbf{13.} In de driehoek $ABG$ is het punt $I$ het midden van de zijde $[AB]$,
dus is $[GI]$ een \omterm{pb:g8:midpoints:1}{zwaartelijn} van $ABG$ \emph{vanuit het \omterm{def:g5:solids:def}{hoekpunt} $G$}: die
verdeelt $ABG$ in twee driehoeken met gelijke \omterm{def:g6:measure:area}{oppervlakte} (vraag 2), namelijk
$AIG$ en $IBG$. Hetzelfde gebeurt in $BCG$ (\omterm{pb:g8:midpoints:1}{zwaartelijn} $[GK]$) en in $CAG$
(\omterm{pb:g8:midpoints:1}{zwaartelijn} $[GJ]$). Elk derde van $ABC$ wordt in twee gelijke \omterm{def:g3:division:half}{helften} verdeeld:
de zes stukken rond $G$ hebben elk als \omterm{def:g6:measure:area}{oppervlakte}
$\frac13 \times \frac12 = \frac16$ van de driehoek.

\textbf{14.} Op de \omterm{pb:g8:midpoints:1}{zwaartelijn} $[AK]$, van $A(0,0)$ naar $K(3,3)$: $G$ ligt op
twee derde van de weg, dus zijn zijn coördinaten
$\left(\frac23 \times 3,\ \frac23 \times 3\right) = (2, 2)$. Op de \omterm{pb:g8:midpoints:1}{zwaartelijn}
$[BJ]$, van $B(6, 0)$ naar $J(0, 3)$: twee derde van de weg betekent dat je twee
derde van de verplaatsing optelt ($-6$ horizontaal, $+3$ verticaal):
\[
\left(6 + \tfrac23 \times (-6),\ 0 + \tfrac23 \times 3\right)
= (2, 2) .
\]
Hetzelfde punt --- zoals vraag 8 beloofde. En $\frac{0 + 6 + 0}{3} = 2$,
$\frac{0 + 0 + 6}{3} = 2$: de coördinaten van het \omterm{pb:g8:midpoints:1}{zwaartepunt} zijn de
\emph{gemiddelden} van de coördinaten van de drie \omterm{def:g5:solids:def}{hoekpunten}.

\textbf{15.} Volgens vraag 2 heeft een mesrand die langs een \omterm{pb:g8:midpoints:1}{zwaartelijn} wordt
gelegd, aan elke kant evenveel karton (gelijke \omterm{def:g6:measure:area}{oppervlakten}), dus balanceert de
driehoek langs elk van de drie \omterm{pb:g8:midpoints:1}{zwaartelijnen}. Een punt van evenwicht moet op alle
drie die evenwichtslijnen tegelijk liggen, en volgens vraag 8 hebben de drie
\omterm{pb:g8:midpoints:1}{zwaartelijnen} precies één punt gemeen: $G$. Daar hoort de potloodpunt. (Gelijke
\omterm{def:g6:measure:area}{oppervlakten} aan beide kanten is de geloofwaardige versie van het argument; de
eerlijke versie, die weegt \emph{waar} het karton zit en niet alleen hoeveel er
is, vraagt de integraalrekening uit de universitaire \omterm{def:g6:measure:volume}{volumes} --- en die
bevestigt $G$.)
\end{solution}
