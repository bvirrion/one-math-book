\section*{Hoofdstuk \ref{ch:g8:cosine} --- De rechthoekige driehoek en de cosinus}

\begin{solution}{exo:g8:cosine:1}
\omterm{pb:g8:midpoints:1}{Zwaartelijn} uit de \omterm{def:g3:shapes:rightangle}{rechte hoek}: de \omterm{def:g3:division:half}{helft} van de schuine zijde, $6.5$ cm
(\cref{thm:g8:cosine:circle}). De \omterm{pb:g6:symmetry:1}{omgeschreven cirkel} heeft het midden van de
schuine zijde als middelpunt en \omterm{def:g3:shapes:shapes}{radius} $6.5$ cm.
\end{solution}

\begin{solution}{exo:g8:cosine:2}
De hoek in $A$ is recht ($[BC]$ is een \omterm{def:g4:geometry:circle}{diameter}: punt 2 van
\cref{thm:g8:cosine:circle}). De metingen geven $AB^2 + AC^2 \approx 64 = BC^2$,
op de meetnauwkeurigheid na.
\end{solution}

\begin{solution}{exo:g8:cosine:3}
Schuine zijde: $[EF]$ (tegenover de \omterm{def:g3:shapes:rightangle}{rechte hoek} $D$). Aanliggend bij
$\widehat E$: $[ED]$. Tegenover: $[DF]$. Dus is
$\cos\widehat E = \dfrac{ED}{EF}$.
\end{solution}

\begin{solution}{exo:g8:cosine:4}
\emph{1.} aanliggende zijde $= 10 \cos 35^\circ \approx 8.2$.

\emph{2.} schuine zijde $= \dfrac{6}{\cos 50^\circ} \approx
\dfrac{6}{0.643} \approx 9.3$.
\end{solution}

\begin{solution}{exo:g8:cosine:5}
$\cos\theta = \frac{8}{10} = 0.8$, dus is
$\theta = \cos^{-1}(0.8) \approx 37^\circ$.
\end{solution}

\begin{solution}{exo:g8:cosine:6}
$\cos\theta$ is een rechthoekszijde gedeeld door de schuine zijde, en de schuine
zijde is de \emph{langste} zijde van een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek}: het \omterm{def:g3:division:remainder}{quotiënt} is
altijd kleiner dan $1$. Een waarde van $1.2$ zou een rechthoekszijde betekenen
die langer is dan de schuine zijde --- onmogelijk.
\end{solution}

\begin{solution}{exo:g8:cosine:7}
Horizontale overspanning $= 25 \cos 12^\circ \approx 25 \times 0.978 \approx
24.5$ m.
\end{solution}

\begin{solution}{exo:g8:cosine:8}
De cosinus daalt als de hoek groter wordt (\cref{prop:g8:cosine:bounds}):
\[
\cos 70^\circ < \cos 45^\circ < \cos 20^\circ .
\]
(Rekenmachine: $0.342 < 0.707 < 0.940$.)
\end{solution}

\begin{solution}{exo:g8:cosine:9}
De afstand op de grond ligt aan $\theta$: $\cos 68^\circ = \frac{1.2}{L}$, dus is
$L = \frac{1.2}{0.375} = 3.2$ m. Hoogte (Pythagoras):
$\sqrt{3.2^2 - 1.2^2} = \sqrt{10.24 - 1.44} = \sqrt{8.8} \approx 3.0$ m.
\end{solution}

\begin{solution}{exo:g8:cosine:10}
\emph{1.} $A$ ligt op een cirkel met \omterm{def:g4:geometry:circle}{diameter} $[BC]$: \omterm{def:g3:shapes:rightangle}{rechte hoek} in $A$
(\cref{thm:g8:cosine:circle}).

\emph{2.} $BC = 9$ cm (twee keer de \omterm{def:g3:shapes:shapes}{radius}). Pythagoras:
$AC = \sqrt{9^2 - 5.4^2} = \sqrt{81 - 29.16} = \sqrt{51.84} = 7.2$ cm. En
$\cos\widehat B = \dfrac{AB}{BC} = \dfrac{5.4}{9} = 0.6$.
\end{solution}

\begin{solution}{exo:g8:cosine:11}
In de \omterm{def:g6:shapes:triangles}{gelijkzijdige driehoek} met zijde $1$ komt de hoogtelijn uit een \omterm{def:g5:solids:def}{hoekpunt} op
het \emph{midden} van de overstaande zijde terecht (\omterm{def:g6:symmetry:axis}{symmetrieas}). De \omterm{def:g3:division:half}{helft} van de
driehoek is \omterm{def:g6:shapes:triangles}{rechthoekig} met schuine zijde $1$ (een volledige zijde), een hoek van
$60^\circ$ in het \omterm{def:g5:solids:def}{hoekpunt} op de basis, en aanliggende zijde $\frac12$ (de \omterm{def:g3:division:half}{helft}
van de basis). Bijgevolg is
\[
\cos 60^\circ = \frac{1/2}{1} = \frac12 .
\]
\end{solution}

\begin{solution}{pb:g8:cosine:1}
\textbf{1.} In de driehoek $ABH$, \omterm{def:g6:shapes:triangles}{rechthoekig} in $H$, geeft de hoekensom
(\cref{thm:g7:triangles:sum}) dat
$\widehat B + 90^\circ + \widehat{BAH} = 180^\circ$, dus
$\widehat{BAH} = 90^\circ - \widehat B$. In de driehoek $ABC$, \omterm{def:g6:shapes:triangles}{rechthoekig} in
$A$: $\widehat B + \widehat C + 90^\circ = 180^\circ$, dus
$\widehat C = 90^\circ - \widehat B$. Bijgevolg is
$\widehat{BAH} = \widehat{ACB}$: beide kleine driehoeken herhalen de hoeken
$\widehat B$, $\widehat C$, $90^\circ$ van de oorspronkelijke.

\textbf{2.} In $ABC$ (\omterm{def:g6:shapes:triangles}{rechthoekig} in $A$) is de zijde die aan $\widehat B$ ligt
$AB$ en is de schuine zijde $BC$: $\cos\widehat B = \frac{AB}{BC}$. In $ABH$
(\omterm{def:g6:shapes:triangles}{rechthoekig} in $H$) is de zijde die aan $\widehat B$ ligt $BH$ en is de schuine
zijde $AB$: $\cos\widehat B = \frac{BH}{AB}$. De verhouding hangt alleen van de
hoek af (\cref{def:g8:cosine:def}), dus is
\[
\frac{AB}{BC} = \frac{BH}{AB} ;
\]
beide leden met $BC \times AB$ vermenigvuldigen
(\cref{thm:g8:equations:balance}) geeft $AB^2 = BH \times BC$.

\textbf{3.} De driehoeken $ABC$ (\omterm{def:g6:shapes:triangles}{rechthoekig} in $A$) en $ACH$ (\omterm{def:g6:shapes:triangles}{rechthoekig} in
$H$) hebben de hoek $\widehat C$ gemeen. Aanliggende zijde gedeeld door schuine
zijde, in elk van beide:
\[
\cos\widehat C = \frac{AC}{CB} = \frac{CH}{AC}
\qquad\Longrightarrow\qquad
AC^2 = CH \times CB .
\]

\textbf{4.} $AB^2 = BH \times BC$ luidt $9 = BH \times 5$, dus is $BH = 1.8$; en
$16 = CH \times 5$ geeft $CH = 3.2$. Controle:
$BH + HC = 1.8 + 3.2 = 5 = BC$ --- zoals het moet, want $H$ ligt op de schuine
zijde tussen $B$ en $C$.

\textbf{5.} Elke betrekking door $BC$ delen geeft
\[
BH = \frac{AB^2}{BC},
\qquad
CH = \frac{AC^2}{BC} .
\]
Dus zijn $BH$ en $CH$ \omterm{def:g6:propdata:def}{evenredig} met $AB^2$ en $AC^2$: de voet van de hoogtelijn
verdeelt de schuine zijde in de verhouding van de kwadraten van de
rechthoekszijden ($1.8 : 3.2 = 9 : 16$ in vraag 4).

\textbf{6.} De twee betrekkingen optellen en $BC$ buiten haakjes brengen
(distributiviteit):
\[
AB^2 + AC^2 = BH \times BC + CH \times BC
= (BH + CH) \times BC = BC \times BC = BC^2 .
\]
Dit is \cref{thm:g8:pythagoras:direct}, verkregen uit de cosinus alleen --- een
werkelijk ander bewijs dan het oppervlakteraadsel met de vier driehoeken uit
\cref{exo:g8:pythagoras:11}.

\textbf{7.} Pythagoras in $ABH$ (\omterm{def:g6:shapes:triangles}{rechthoekig} in $H$, schuine zijde $[AB]$):
$AH^2 = AB^2 - BH^2$. Vervang $AB^2$ door $BH \times BC$ (vraag 2) en breng $BH$
buiten haakjes:
\[
AH^2 = BH \times BC - BH^2
= BH \times (BC - BH)
= BH \times HC .
\]

\textbf{8.} Met de rechthoekszijden als basis en hoogte is de \omterm{def:g6:measure:area}{oppervlakte} van
$ABC$ gelijk aan $\frac{AB \times AC}{2}$; met de schuine zijde $[BC]$ als basis
is de hoogte precies $AH$, dus is de \omterm{def:g6:measure:area}{oppervlakte} ook
$\frac{BC \times AH}{2}$ (\cref{thm:g7:areas:triangle}). Gelijkstellen en
verdubbelen: $AB \times AC = BC \times AH$.

\textbf{9.} $AH = \frac{AB \times AC}{BC} = \frac{3 \times 4}{5} = 2.4$.
Nagaan van vraag 7: $AH^2 = 2.4^2 = 5.76$ en
$BH \times HC = 1.8 \times 3.2 = 5.76$. Gelijk.

\textbf{10.} $AH^2 = BH \times HC = 2 \times 8 = 16$, dus is
$AH = \sqrt{16} = 4$ cm.

\textbf{11.} Het punt $A$ ligt op de cirkel met \omterm{def:g4:geometry:circle}{diameter} $[BC]$ (en is
verschillend van $B$ en $C$), dus is de driehoek $ABC$ volgens punt 2 van
\cref{thm:g8:cosine:circle} \omterm{def:g6:shapes:triangles}{rechthoekig} in $A$. Door de constructie is
$(AH) \perp (BC)$ met $H$ tussen $B$ en $C$: het \omterm{def:g6:lines:objects}{lijnstuk} $[AH]$ is precies de
hoogtelijn van vraag 7, en $AH^2 = BH \times HC = p \times q$.

\textbf{12.} $AH^2 = 1 \times 2 = 2$: de figuur construeert, met passer en lat
alleen, een \omterm{def:g6:lines:objects}{lijnstuk} waarvan het kwadraat $2$ is --- de diagonaal van het
eenheidsvierkant, $\sqrt2 \approx 1.414$, uit
\cref{ex:g8:pythagoras:diagonal}.

\textbf{13.} Recept: leg twee lijnstukken met lengten $p = 1$ en $q = n$ achter
elkaar; teken de halve cirkel met het hele \omterm{def:g6:lines:objects}{lijnstuk} (lengte $n + 1$) als
\omterm{def:g4:geometry:circle}{diameter}; richt de loodlijn op in het verbindingspunt; die ontmoet de halve
cirkel in een punt waarvan de afstand tot het verbindingspunt
$\sqrt{1 \times n} = \sqrt n$ is. Voor $\sqrt5$: neem $p = 1$ en $q = 5$ (een
halve cirkel met \omterm{def:g4:geometry:circle}{diameter} $6$).

\textbf{14.} $A$ ligt op de cirkel met middelpunt $O$, dus is
$OA = \frac{p+q}{2}$, de \omterm{def:g3:shapes:shapes}{radius}. Is $H \neq O$, dan is de driehoek $AHO$
\omterm{def:g6:shapes:triangles}{rechthoekig} in $H$ met schuine zijde $[OA]$, en een rechthoekszijde is korter dan
de schuine zijde: $AH < OA$. Is $H = O$, dan is $AH = OA$ precies. In alle
gevallen is $AH \leq \frac{p+q}{2}$, met gelijkheid precies als $H$ het
middelpunt is --- dat wil zeggen als $p = q$.

\textbf{15.} $AH \leq \frac{p+q}{2}$ kwadrateren (beide leden positief) en
$AH^2 = pq$ gebruiken geeft
\[
p \times q \leq \left(\frac{p+q}{2}\right)^2 ,
\]
met gelijkheid precies voor $p = q$. Algebraïsch nogmaals bewezen, met de
merkwaardige \omterm{def:g2:mult:def}{producten} (\cref{pb:g8:equations:1}):
\[
\left(\frac{p+q}{2}\right)^2 - pq
= \frac{p^2 + 2pq + q^2 - 4pq}{4}
= \frac{p^2 - 2pq + q^2}{4}
= \left(\frac{p-q}{2}\right)^2 ,
\]
een kwadraat, en dus nooit negatief --- en \omterm{def:g1:counting:zero}{nul} precies als $p = q$. Het
\omterm{pb:g8:cosine:1}{meetkundig gemiddelde} van twee getallen is nooit groter dan hun halve \omterm{def:g1:addition:def}{som}.
\end{solution}
