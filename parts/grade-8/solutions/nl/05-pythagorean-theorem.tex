\section*{Hoofdstuk \ref{ch:g8:pythagoras} --- De stelling van Pythagoras}

\begin{solution}{exo:g8:pythagoras:1}
$\sqrt{81 + 144} = \sqrt{225} = 15$; \quad
$\sqrt{25 + 144} = \sqrt{169} = 13$; \quad
$\sqrt{64 + 225} = \sqrt{289} = 17$.
\end{solution}

\begin{solution}{exo:g8:pythagoras:2}
Rechthoekszijde$^2 = 25^2 - 7^2 = 625 - 49 = 576$, dus is de rechthoekszijde
$\sqrt{576} = 24$.
\end{solution}

\begin{solution}{exo:g8:pythagoras:3}
Rechthoek: $\sqrt{12^2 + 9^2} = \sqrt{144 + 81} = \sqrt{225} = 15$ cm.

Vierkant: $d = 5\sqrt2 \approx 7.1$ cm
(\cref{ex:g8:pythagoras:diagonal}).
\end{solution}

\begin{solution}{exo:g8:pythagoras:4}
$(20, 21, 29)$: $29^2 = 841$ en $20^2 + 21^2 = 400 + 441 = 841$: \omterm{def:g6:shapes:triangles}{rechthoekig}.

$(7, 8, 11)$: $121 \neq 49 + 64 = 113$: niet \omterm{def:g6:shapes:triangles}{rechthoekig}.

$(1.5, 2, 2.5)$: $6.25 = 2.25 + 4$: \omterm{def:g6:shapes:triangles}{rechthoekig} (het is $(3,4,5)$ gehalveerd).
\end{solution}

\begin{solution}{exo:g8:pythagoras:5}
Plank $= \sqrt{3^2 + 1.6^2} = \sqrt{9 + 2.56} = \sqrt{11.56} = 3.4$ m.
\end{solution}

\begin{solution}{exo:g8:pythagoras:6}
De hoogte komt neer op het midden van de basis (\omterm{def:g6:shapes:triangles}{gelijkbenige driehoek}), dus heeft
elke halve driehoek schuine zijde $10$ en één rechthoekszijde $6$:
$h = \sqrt{100 - 36} = \sqrt{64} = 8$ cm. \omterm{def:g6:measure:area}{Oppervlakte}:
$\frac{12 \times 8}{2} = 48$ cm$^2$.
\end{solution}

\begin{solution}{exo:g8:pythagoras:7}
Diagonaal: $\sqrt{88.5^2 + 49.8^2} = \sqrt{7832.25 + 2480.04}
= \sqrt{10312.29} \approx 101.5$ cm. In inch:
$101.5 \div 2.54 \approx 40$ inch.
\end{solution}

\begin{solution}{exo:g8:pythagoras:8}
De twee rechthoekszijden zijn $24$ km en $10$ km:
$\sqrt{24^2 + 10^2} = \sqrt{576 + 100} = \sqrt{676} = 26$ km.
\end{solution}

\begin{solution}{exo:g8:pythagoras:9}
Afstand van de voet: $\sqrt{2.5^2 - 2.4^2} = \sqrt{6.25 - 5.76} =
\sqrt{0.49} = 0.7$ m. Een \omterm{def:g3:division:half}{kwart} van de ladder is $2.5 \div 4 \approx 0.6$ m: de
gevonden $0.7$ m ligt dicht bij de aangeraden plaatsing --- aanvaardbaar.
\end{solution}

\begin{solution}{exo:g8:pythagoras:10}
De rechthoekszijden meten $5 - 1 = 4$ (horizontaal) en $5 - 2 = 3$ (verticaal),
dus is
\[
AB = \sqrt{4^2 + 3^2} = \sqrt{25} = 5 .
\]
\end{solution}

\begin{solution}{exo:g8:pythagoras:11}
\emph{1.} Rechtstreeks: $(a + b)^2$. Als stukken: vier driehoeken met elk
\omterm{def:g6:measure:area}{oppervlakte} $\frac{ab}{2}$, plus het gekantelde vierkant $c^2$:
\[
(a + b)^2 = 4 \times \frac{ab}{2} + c^2 = 2ab + c^2 .
\]

\emph{2.} Het linkerlid uitwerken geeft $(a+b)^2 = a^2 + 2ab + b^2$. Dus is
\[
a^2 + 2ab + b^2 = 2ab + c^2
\quad\Longrightarrow\quad
a^2 + b^2 = c^2 . \qed
\]
\end{solution}

\begin{solution}{pb:g8:pythagoras:1}
\textbf{1.} $5^2 + 12^2 = 25 + 144 = 169 = 13^2$; \quad
$8^2 + 15^2 = 64 + 225 = 289 = 17^2$; \quad
$20^2 + 21^2 = 400 + 441 = 841 = 29^2$. Alle drie zijn drietallen, en volgens
\cref{thm:g8:pythagoras:converse} is een driehoek met deze zijden \omterm{def:g6:shapes:triangles}{rechthoekig},
met de \omterm{def:g3:shapes:rightangle}{rechte hoek} tegenover de langste zijde.

\textbf{2.} Het koord vormt een driehoek met zijden $3$, $4$, $5$ (in stukken),
en $3^2 + 4^2 = 25 = 5^2$: volgens de omgekeerde stelling
(\cref{thm:g8:pythagoras:converse}) is de hoek tussen de zijden van $3$ en $4$
stukken precies recht --- geen gradenboog nodig, alleen knopen.

\textbf{3.} Is $a^2 + b^2 = c^2$, dan is
\[
(ka)^2 + (kb)^2 = k^2 a^2 + k^2 b^2
= k^2\left(a^2 + b^2\right) = k^2 c^2 = (kc)^2 .
\]
Uit $(3, 4, 5)$: bijvoorbeeld $(6, 8, 10)$, $(9, 12, 15)$ en $(30, 40, 50)$.

\textbf{4.} Een \omterm{def:g5:division:multiple}{veelvoud} $(3k, 4k, 5k)$ met $5$ als kleinste getal zou
$3k = 5$ vragen, en geen enkel heel getal $k$ voldoet daaraan ($k = 1$ geeft $3$,
$k = 2$ geeft $6$). Dus is $(5, 12, 13)$ geen \omterm{def:g5:division:multiple}{veelvoud} van $(3, 4, 5)$.

\textbf{5.} Vraag 3 laat zien dat vermenigvuldigen alleen de drietallen
$(3k, 4k, 5k)$ oplevert; vraag 4 toont een echt drietal dat niet van die vorm is.
De familie van alle pythagorese drietallen is dus strikt groter dan de \omterm{def:g5:division:multiple}{veelvouden}
van $(3, 4, 5)$: om ze voort te brengen is iets nieuws nodig.

\textbf{6.} Schrijf $A = m^2$ en $B = n^2$. Met de merkwaardige \omterm{def:g2:mult:def}{producten}
(\cref{pb:g8:equations:1}):
\[
a^2 = (A - B)^2 = A^2 - 2AB + B^2,
\qquad
b^2 = (2mn)^2 = 4m^2n^2 = 4AB .
\]
Optellen:
\[
a^2 + b^2 = A^2 - 2AB + B^2 + 4AB
= A^2 + 2AB + B^2 = (A + B)^2
= \left(m^2 + n^2\right)^2 = c^2 .
\]

\textbf{7.}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccc}
$(m, n)$ & $m^2 - n^2$ & $2mn$ & $m^2 + n^2$ \\
\hline
$(2, 1)$ & $3$ & $4$ & $5$ \\
$(3, 1)$ & $8$ & $6$ & $10$ \\
$(3, 2)$ & $5$ & $12$ & $13$ \\
$(4, 1)$ & $15$ & $8$ & $17$ \\
$(4, 3)$ & $7$ & $24$ & $25$ \\
\end{tabular}
\end{center}

\textbf{8.} $(8, 6, 10)$ is het dubbel van $(4, 3, 5)$ --- dat wil zeggen van de
$(3, 4, 5)$ van de bouwers. De machine vindt \omterm{def:g1:addition:def}{soms} een oud drietal in vermomming
terug.

\textbf{9.} We hebben $m^2 - n^2 = 21$ en $2mn = 20$ nodig, dus $mn = 10$: met
$m = 5$, $n = 2$ geeft dat $25 - 4 = 21$ en $2 \times 10 = 20$, met schuine zijde
$25 + 4 = 29$. Voor $(m, n) = (5, 2)$ levert de machine $(21, 20, 29)$.

\textbf{10.} $m = 8$, $n = 6$ geeft $m^2 + n^2 = 64 + 36 = 100$. Het drietal is
\[
\left(64 - 36,\ 2 \times 48,\ 100\right) = (28,\ 96,\ 100),
\]
en inderdaad is $28^2 + 96^2 = 784 + 9216 = 10\,000 = 100^2$.

\textbf{11.} Met $m = n + 1$:
\[
m^2 - n^2 = (n + 1)^2 - n^2 = 2n + 1
\]
(\omterm{ex:g1:subtraction:difference}{verschil} van opeenvolgende kwadraten, \cref{pb:g8:equations:1});
$2mn = 2n(n + 1) = 2n^2 + 2n$; en
$m^2 + n^2 = n^2 + n^2 + 2n + 1 = 2n^2 + 2n + 1$. De schuine zijde is precies
$1$ groter dan de \omterm{def:g3:numbers:evenodd}{even} rechthoekszijde. Voor $n = 1, 2, 3, 4$:
\[
(3, 4, 5), \quad (5, 12, 13), \quad (7, 24, 25), \quad
(9, 40, 41) .
\]

\textbf{12.} Elk \omterm{def:g3:numbers:evenodd}{oneven getal} vanaf $3$ is $2n + 1$ voor een zekere $n \geq 1$,
en vraag 11 geeft een drietal waarvan de \omterm{def:g3:numbers:evenodd}{oneven} rechthoekszijde precies $2n + 1$
is. Voor $11 = 2 \times 5 + 1$ neem je $n = 5$:
\[
(11,\ 60,\ 61),
\qquad
11^2 + 60^2 = 121 + 3600 = 3721 = 61^2 .
\]

\textbf{13.} Met $n = 1$ geeft de machine
$\left(m^2 - 1,\ 2m,\ m^2 + 1\right)$, waarvan de \omterm{def:g3:numbers:evenodd}{even} rechthoekszijde $2m$ elk
\omterm{def:g3:numbers:evenodd}{even getal} vanaf $4$ kan zijn. Voor $14 = 2 \times 7$ neem je $m = 7$: het
drietal $(48, 14, 50)$. Hetzelfde drietal komt er ook uit door $(7, 24, 25)$ uit
de tabel van vraag 7 te verdubbelen: $(14, 48, 50)$ --- dezelfde drie getallen.

\textbf{14.} Een \omterm{def:g3:numbers:evenodd}{even getal} is $2k$, en $(2k)^2 = 4k^2$ is \omterm{def:g3:numbers:evenodd}{even} (een \omterm{def:g5:division:multiple}{veelvoud} van
$4$, en dus \omterm{def:g3:numbers:evenodd}{even}). Een \omterm{def:g3:numbers:evenodd}{oneven getal} is $2k + 1$, en het eerste merkwaardige
\omterm{def:g2:mult:def}{product} geeft
\[
(2k + 1)^2 = 4k^2 + 4k + 1 = 2\left(2k^2 + 2k\right) + 1 ,
\]
een \omterm{def:g3:numbers:evenodd}{even getal} plus $1$: \omterm{def:g3:numbers:evenodd}{oneven}.

\textbf{15.} Stel dat $a$, $b$ en $c$ alle drie \omterm{def:g3:numbers:evenodd}{oneven} zouden zijn. Volgens vraag
14 zijn $a^2$ en $b^2$ dan \omterm{def:g3:numbers:evenodd}{oneven}, dus is $a^2 + b^2$ een \omterm{def:g1:addition:def}{som} van twee \omterm{def:g3:numbers:evenodd}{oneven}
getallen: \emph{\omterm{def:g3:numbers:evenodd}{even}}. Maar $c^2$ zou \emph{\omterm{def:g3:numbers:evenodd}{oneven}} zijn. De gelijkheid
$a^2 + b^2 = c^2$ is dan onmogelijk. In elk \omterm{pb:g8:pythagoras:1}{pythagorees drietal} is dus minstens
één van de drie getallen \omterm{def:g3:numbers:evenodd}{even} --- zoals elke regel van de tabel van vraag 7
bevestigt.
\end{solution}
