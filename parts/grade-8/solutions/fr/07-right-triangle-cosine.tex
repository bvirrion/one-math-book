\section*{Chapitre \ref{ch:g8:cosine} --- Le triangle rectangle et le cosinus}

\begin{solution}{exo:g8:cosine:1}
Médiane issue de l'\omterm{def:g3:shapes:rightangle}{angle droit}~: la \omterm{def:g3:division:half}{moitié} de l'hypoténuse, $6.5$ cm
(\cref{thm:g8:cosine:circle}). Le \omterm{pb:g6:symmetry:1}{cercle circonscrit} est centré au
milieu de l'hypoténuse avec un \omterm{def:g3:shapes:shapes}{rayon} de $6.5$ cm.
\end{solution}

\begin{solution}{exo:g8:cosine:2}
L'angle en $A$ est droit ($[BC]$ est un \omterm{def:g4:geometry:circle}{diamètre}~: point 2 de
\cref{thm:g8:cosine:circle}). Les mesures satisfont
$AB^2 + AC^2 \approx 64 = BC^2$, à la précision de mesure près.
\end{solution}

\begin{solution}{exo:g8:cosine:3}
Hypoténuse~: $[EF]$ (opposée à l'\omterm{def:g3:shapes:rightangle}{angle droit} $D$). Adjacent à
$\widehat E$~: $[ED]$. Opposé~: $[DF]$. Donc
$\cos\widehat E = \dfrac{ED}{EF}$.
\end{solution}

\begin{solution}{exo:g8:cosine:4}
\emph{1.} adjacent $= 10 \cos 35^\circ \approx 8.2$.

\emph{2.} hypoténuse $= \dfrac{6}{\cos 50^\circ} \approx
\dfrac{6}{0.643} \approx 9.3$.
\end{solution}

\begin{solution}{exo:g8:cosine:5}
$\cos\theta = \frac{8}{10} = 0.8$, donc
$\theta = \cos^{-1}(0.8) \approx 37^\circ$.
\end{solution}

\begin{solution}{exo:g8:cosine:6}
$\cos\theta$ est un côté de l'\omterm{def:g3:shapes:rightangle}{angle droit} divisé par l'hypoténuse, et
l'hypoténuse est le \emph{plus long} côté d'un \omterm{def:g6:shapes:triangles}{triangle rectangle}~: le
\omterm{def:g3:division:remainder}{quotient} est toujours plus petit que $1$. Une valeur de $1.2$
signifierait un côté plus long que l'hypoténuse --- impossible.
\end{solution}

\begin{solution}{exo:g8:cosine:7}
Portée horizontale $= 25 \cos 12^\circ \approx 25 \times 0.978 \approx
24.5$ m.
\end{solution}

\begin{solution}{exo:g8:cosine:8}
Le cosinus diminue lorsque l'angle croît
(\cref{prop:g8:cosine:bounds})~:
\[
\cos 70^\circ < \cos 45^\circ < \cos 20^\circ .
\]
(Calculatrice~: $0.342 < 0.707 < 0.940$.)
\end{solution}

\begin{solution}{exo:g8:cosine:9}
La distance au sol est adjacente à $\theta$~:
$\cos 68^\circ = \frac{1.2}{L}$, donc
$L = \frac{1.2}{0.375} = 3.2$ m. Hauteur (Pythagore)~:
$\sqrt{3.2^2 - 1.2^2} = \sqrt{10.24 - 1.44} = \sqrt{8.8} \approx 3.0$
m.
\end{solution}

\begin{solution}{exo:g8:cosine:10}
\emph{1.} $A$ est sur un cercle de \omterm{def:g4:geometry:circle}{diamètre} $[BC]$~: \omterm{def:g3:shapes:rightangle}{angle droit} en
$A$ (\cref{thm:g8:cosine:circle}).

\emph{2.} $BC = 9$ cm (deux fois le \omterm{def:g3:shapes:shapes}{rayon}). Pythagore~:
$AC = \sqrt{9^2 - 5.4^2} = \sqrt{81 - 29.16} = \sqrt{51.84} = 7.2$
cm. Et $\cos\widehat B = \dfrac{AB}{BC} = \dfrac{5.4}{9} = 0.6$.
\end{solution}

\begin{solution}{exo:g8:cosine:11}
Dans le \omterm{def:g6:shapes:triangles}{triangle équilatéral} de côté $1$, la hauteur issue d'un
\omterm{def:g5:solids:def}{sommet} tombe au \emph{milieu} du côté opposé (\omterm{def:g6:symmetry:axis}{axe de symétrie}). La
\omterm{def:g3:division:half}{moitié} du triangle est rectangle avec hypoténuse $1$ (un côté
entier), un angle de $60^\circ$ au \omterm{def:g5:solids:def}{sommet} de la base, et côté
adjacent $\frac12$ (la \omterm{def:g3:division:half}{moitié} de la base). D'où
\[
\cos 60^\circ = \frac{1/2}{1} = \frac12 .
\]
\end{solution}

\begin{solution}{pb:g8:cosine:1}
\textbf{1.} Dans le triangle $ABH$, rectangle en $H$, la \omterm{def:g1:addition:def}{somme} des
angles (\cref{thm:g7:triangles:sum}) donne
$\widehat B + 90^\circ + \widehat{BAH} = 180^\circ$, donc
$\widehat{BAH} = 90^\circ - \widehat B$. Dans le triangle $ABC$,
rectangle en $A$~: $\widehat B + \widehat C + 90^\circ = 180^\circ$,
donc $\widehat C = 90^\circ - \widehat B$. D'où
$\widehat{BAH} = \widehat{ACB}$~: les deux petits triangles répètent
bien les angles $\widehat B$, $\widehat C$, $90^\circ$ du triangle
de départ.

\textbf{2.} Dans $ABC$ (rectangle en $A$), le côté adjacent à
$\widehat B$ est $AB$ et l'hypoténuse est $BC$~:
$\cos\widehat B = \frac{AB}{BC}$. Dans $ABH$ (rectangle en $H$), le
côté adjacent à $\widehat B$ est $BH$ et l'hypoténuse est $AB$~:
$\cos\widehat B = \frac{BH}{AB}$. Le rapport ne dépend que de
l'angle (\cref{def:g8:cosine:def}), donc
\[
\frac{AB}{BC} = \frac{BH}{AB} ;
\]
en multipliant les deux membres par $BC \times AB$
(\cref{thm:g8:equations:balance})~: $AB^2 = BH \times BC$.

\textbf{3.} Les triangles $ABC$ (rectangle en $A$) et $ACH$
(rectangle en $H$) ont l'angle $\widehat C$ en commun. Côté adjacent
sur hypoténuse dans chacun~:
\[
\cos\widehat C = \frac{AC}{CB} = \frac{CH}{AC}
\qquad\Longrightarrow\qquad
AC^2 = CH \times CB .
\]

\textbf{4.} $AB^2 = BH \times BC$ s'écrit $9 = BH \times 5$, donc
$BH = 1.8$~; et $16 = CH \times 5$ donne $CH = 3.2$. Vérification~:
$BH + HC = 1.8 + 3.2 = 5 = BC$ --- comme il se doit, puisque $H$ est
sur l'hypoténuse entre $B$ et $C$.

\textbf{5.} En divisant chaque relation par $BC$~:
\[
BH = \frac{AB^2}{BC},
\qquad
CH = \frac{AC^2}{BC} .
\]
Ainsi $BH$ et $CH$ sont proportionnels à $AB^2$ et $AC^2$~: le pied
de la hauteur partage l'hypoténuse dans le rapport des carrés des
côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit} ($1.8 : 3.2 = 9 : 16$ à la question 4).

\textbf{6.} En additionnant les deux relations et en factorisant par
$BC$ (distributivité)~:
\[
AB^2 + AC^2 = BH \times BC + CH \times BC
= (BH + CH) \times BC = BC \times BC = BC^2 .
\]
C'est \cref{thm:g8:pythagoras:direct}, obtenu à partir du seul
cosinus --- une démonstration véritablement différente du puzzle
d'\omterm{def:g6:measure:area}{aires} à quatre triangles de l'\cref{exo:g8:pythagoras:11}.

\textbf{7.} Pythagore dans $ABH$ (rectangle en $H$, d'hypoténuse
$[AB]$)~: $AH^2 = AB^2 - BH^2$. En remplaçant $AB^2$ par
$BH \times BC$ (question 2) et en factorisant par $BH$~:
\[
AH^2 = BH \times BC - BH^2
= BH \times (BC - BH)
= BH \times HC .
\]

\textbf{8.} Avec les côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit} comme base et hauteur,
l'\omterm{def:g6:measure:area}{aire} de $ABC$ vaut $\frac{AB \times AC}{2}$~; avec l'hypoténuse
$[BC]$ pour base, la hauteur est précisément $AH$, et l'\omterm{def:g6:measure:area}{aire} vaut
aussi $\frac{BC \times AH}{2}$ (\cref{thm:g7:areas:triangle}). En
égalant et en doublant~: $AB \times AC = BC \times AH$.

\textbf{9.} $AH = \frac{AB \times AC}{BC} = \frac{3 \times 4}{5}
= 2.4$. Vérification de la question 7~:
$AH^2 = 2.4^2 = 5.76$ et $BH \times HC = 1.8 \times 3.2 = 5.76$.
Égalité.

\textbf{10.} $AH^2 = BH \times HC = 2 \times 8 = 16$, donc
$AH = \sqrt{16} = 4$ cm.

\textbf{11.} Le point $A$ est sur le cercle de \omterm{def:g4:geometry:circle}{diamètre} $[BC]$ (et
diffère de $B$ et de $C$), donc d'après le point 2 du
\cref{thm:g8:cosine:circle} le triangle $ABC$ est rectangle en $A$.
Par construction, $(AH) \perp (BC)$ avec $H$ entre $B$ et $C$~: le
\omterm{def:g6:lines:objects}{segment} $[AH]$ est précisément la hauteur de la question 7, et
$AH^2 = BH \times HC = p \times q$.

\textbf{12.} $AH^2 = 1 \times 2 = 2$~: la figure construit, à la
règle et au compas seuls, un \omterm{def:g6:lines:objects}{segment} dont le carré vaut $2$ --- la
diagonale du carré unité, $\sqrt2 \approx 1.414$, de
l'\cref{ex:g8:pythagoras:diagonal}.

\textbf{13.} Recette~: mettre bout à bout deux \omterm{def:g6:lines:objects}{segments} de longueurs
$p = 1$ et $q = n$~; tracer le demi-cercle dont le \omterm{def:g4:geometry:circle}{diamètre} est le
\omterm{def:g6:lines:objects}{segment} entier (de longueur $n + 1$)~; élever la \omterm{def:g6:lines:perp}{perpendiculaire} au
point de jonction~; elle rencontre le demi-cercle en un point dont
la distance à la jonction vaut $\sqrt{1 \times n} = \sqrt n$. Pour
$\sqrt5$~: prendre $p = 1$ et $q = 5$ (un demi-cercle de \omterm{def:g4:geometry:circle}{diamètre}
$6$).

\textbf{14.} $A$ est sur le cercle de centre $O$, donc
$OA = \frac{p+q}{2}$, le \omterm{def:g3:shapes:shapes}{rayon}. Si $H \neq O$, le triangle $AHO$ est
rectangle en $H$ d'hypoténuse $[OA]$, et un côté de l'\omterm{def:g3:shapes:rightangle}{angle droit}
est plus court que l'hypoténuse~: $AH < OA$. Si $H = O$, alors
$AH = OA$ exactement. Dans tous les cas
$AH \leq \frac{p+q}{2}$, avec égalité précisément lorsque $H$ est le
centre --- c'est-à-dire lorsque $p = q$.

\textbf{15.} En élevant au carré $AH \leq \frac{p+q}{2}$ (les deux
membres sont positifs) et en utilisant $AH^2 = pq$~:
\[
p \times q \leq \left(\frac{p+q}{2}\right)^2 ,
\]
avec égalité exactement pour $p = q$. Redémonstration algébrique,
avec les identités remarquables (\cref{pb:g8:equations:1})~:
\[
\left(\frac{p+q}{2}\right)^2 - pq
= \frac{p^2 + 2pq + q^2 - 4pq}{4}
= \frac{p^2 - 2pq + q^2}{4}
= \left(\frac{p-q}{2}\right)^2 ,
\]
un carré, donc jamais négatif --- et nul exactement lorsque
$p = q$. La \omterm{pb:g8:cosine:1}{moyenne géométrique} de deux nombres ne dépasse jamais
leur \omterm{def:g1:addition:def}{demi-somme}.
\end{solution}
