\section*{Chapitre \ref{ch:g8:midpoints} --- Milieux et parallèles}

\begin{solution}{exo:g8:midpoints:1}
$M$ et $N$ sont les milieux de deux côtés du triangle $RST$~: par le
théorème des milieux (\cref{thm:g8:midpoints:direct}),
$(MN) \parallel (ST)$ et $MN = \frac{ST}{2} = 3.5$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:2}
$I$ est un milieu et $(IJ) \parallel (BC)$~: par la
\emph{réciproque} (\cref{thm:g8:midpoints:converse}), $J$ est le
milieu de $[AC]$, donc $AJ = \frac{11}{2} = 5.5$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:3}
La mesure donne $IJ = 3.5$ cm $= \frac{BC}{2}$. \omterm{def:g6:measure:perimeter}{Périmètre} de $AIJ$~:
$AI = 4$, $AJ = 3$, $IJ = 3.5$~: total $10.5$ cm --- la \omterm{def:g3:division:half}{moitié} du
\omterm{def:g6:measure:perimeter}{périmètre} de $ABC$ ($8 + 6 + 7 = 21$ cm).
\end{solution}

\begin{solution}{exo:g8:midpoints:4}
Chaque côté de $IJK$ est la \omterm{def:g3:division:half}{moitié} d'un côté de $ABC$~: $5$, $6$ et
$8$ cm. \omterm{def:g6:measure:perimeter}{Périmètre}~: $19$ cm, la \omterm{def:g3:division:half}{moitié} de $10 + 12 + 16 = 38$ cm.
\end{solution}

\begin{solution}{exo:g8:midpoints:5}
Le triangle des milieux est l'un des quatre triangles égaux de
\cref{prop:g8:midpoints:medial}~: \omterm{def:g6:measure:area}{aire} $\frac{36}{4} = 9$ cm$^2$ ---
et chacun des quatre petits triangles a pour \omterm{def:g6:measure:area}{aire} $9$ cm$^2$.
\end{solution}

\begin{solution}{exo:g8:midpoints:6}
Par le théorème des milieux dans $ABC$~: $(IJ) \parallel (BC)$,
c'est-à-dire $(IJ) \parallel (BK)$, et $IJ = \frac{BC}{2} = BK$
(puisque $K$ est le milieu de $[BC]$). Deux côtés opposés de $IJKB$
sont \omterm{def:g6:lines:perp}{parallèles} et égaux~: c'est un \omterm{def:g7:central:parallelogram}{parallélogramme}
(\cref{thm:g7:central:recognize}, critère 3).
\end{solution}

\begin{solution}{exo:g8:midpoints:7}
La couture joint deux milieux~: $\frac{6.4}{2} = 3.2$ m. Au \omterm{def:g3:division:half}{quart}
depuis le \omterm{def:g5:solids:def}{sommet} du haut, le dessin suggère une couture de
$\frac{6.4}{4} = 1.6$ m, \omterm{def:g6:lines:perp}{parallèle} à la base --- confirmé par Thalès
au \cref{ch:g9:thales}.
\end{solution}

\begin{solution}{exo:g8:midpoints:8}
\emph{1.} $M$ est le milieu de $[BC]$ et $(MN) \parallel (AB)$~: par
la réciproque du théorème des milieux (dans le triangle $ABC$, côté
$[AC]$), $N$ est le milieu de $[AC]$.

\emph{2.} $(MN) \parallel (AB)$ et $(AB) \perp (AC)$ (\omterm{def:g3:shapes:rightangle}{angle droit} en
$A$)~: une droite \omterm{def:g6:lines:perp}{parallèle} à $(AB)$ est aussi \omterm{def:g6:lines:perp}{perpendiculaire} à
$(AC)$ (\cref{prop:g6:lines:facts}). Donc $(MN) \perp (AC)$ --- le
\omterm{def:g6:lines:objects}{segment} $[MN]$ est le morceau de \omterm{def:g6:symmetry:bisector}{médiatrice} montrant à nouveau que
$MA = MC$ (\cref{thm:g8:cosine:circle}).
\end{solution}

\begin{solution}{exo:g8:midpoints:9}
\emph{1.} Dans le triangle $ABC$, $I$ et $J$ sont milieux de $[AB]$
et $[BC]$~: $(IJ) \parallel (AC)$ et $IJ = \frac{AC}{2}$. Dans le
triangle $ACD$, $L$ et $K$ sont milieux de $[DA]$ et $[CD]$~:
$(LK) \parallel (AC)$ et $LK = \frac{AC}{2}$.

\emph{2.} Donc $[IJ]$ et $[LK]$ sont \omterm{def:g6:lines:perp}{parallèles} (tous deux \omterm{def:g6:lines:perp}{parallèles}
à la diagonale $(AC)$) et égaux (tous deux la \omterm{def:g3:division:half}{moitié} de $AC$)~:
$IJKL$ est un \omterm{def:g7:central:parallelogram}{parallélogramme} (\cref{thm:g7:central:recognize},
critère 3) --- pour \emph{tout} \omterm{def:g4:geometry:polygon}{quadrilatère} $ABCD$, comme conjecturé
en \cref{exo:g7:central:11}.
\end{solution}

\begin{solution}{pb:g8:midpoints:1}
\textbf{1.} Quel que soit le triangle tracé, les trois médianes
semblent passer par un même point, situé à l'intérieur du triangle,
visiblement plus proche du milieu de chaque côté que du \omterm{def:g5:solids:def}{sommet}
opposé. La suite du problème le démontre.

\textbf{2.} Prenons la médiane $[AK]$ du triangle $ABC$~: les
triangles $ABK$ et $AKC$ ont des bases $BK$ et $KC$ de même longueur
($K$ est le milieu de $[BC]$) portées par la même droite $(BC)$, et
la même hauteur~: la distance de $A$ à $(BC)$. Par la formule de
l'\omterm{def:g6:measure:area}{aire}, chacun a pour \omterm{def:g6:measure:area}{aire} base $\times$ hauteur $\div\ 2$, avec des
bases égales et des hauteurs égales~: les \omterm{def:g6:measure:area}{aires} sont donc égales.
Pour un triangle d'\omterm{def:g6:measure:area}{aire} $36$ cm$^2$~: deux morceaux de $18$ cm$^2$
chacun.

\textbf{3.} $I$ et $J$ sont les milieux des côtés $[AB]$ et $[AC]$
du triangle $ABC$, donc d'après \cref{thm:g8:midpoints:direct}~:
$(IJ) \parallel (BC)$ et $IJ = \frac{BC}{2}$.

\textbf{4.} $M$ et $N$ sont les milieux des côtés $[GB]$ et $[GC]$
du triangle $GBC$, donc par le même théorème~:
$(MN) \parallel (BC)$ et $MN = \frac{BC}{2}$.

\textbf{5.} $(IJ)$ et $(MN)$ sont toutes deux \omterm{def:g6:lines:perp}{parallèles} à $(BC)$,
donc \omterm{def:g6:lines:perp}{parallèles} entre elles~; et $IJ = \frac{BC}{2} = MN$. Le
\omterm{def:g4:geometry:polygon}{quadrilatère} $IJNM$ a ainsi deux côtés opposés, $[IJ]$ et $[NM]$,
\omterm{def:g6:lines:perp}{parallèles} et de même longueur~: d'après le critère 3 du
\cref{thm:g7:central:recognize}, $IJNM$ est un \omterm{def:g7:central:parallelogram}{parallélogramme}.

\textbf{6.} $G$ est sur la médiane $[CI]$, et $N$ est le milieu de
$[GC]$, \omterm{def:g6:lines:objects}{segment} porté par cette même droite médiane~: donc $I$, $N$
(ainsi que $G$ et $C$) sont tous sur $(CI)$. De même, $J$, $M$ et
$G$ sont sur la droite médiane $(BJ)$. Dans le \omterm{def:g7:central:parallelogram}{parallélogramme}
$IJNM$, les diagonales sont $[IN]$ et $[JM]$~; elles sont portées
par les deux droites médianes, qui se coupent en $G$ --- les
diagonales se coupent donc en $G$. D'après la \emph{définition} d'un
\omterm{def:g7:central:parallelogram}{parallélogramme} (critère 1 du~\cref{thm:g7:central:recognize}~: les
diagonales ont le même milieu), $G$ est le milieu de $[IN]$
\emph{et} de $[JM]$~: $GI = GN$ et $GJ = GM$.

\textbf{7.} $M$ est le milieu de $[GB]$, donc $BM = MG$~; et
$MG = GJ$ par la question 6. D'où $BM = MG = GJ$~: la médiane
$[BJ]$ est découpée en trois parts égales, dont $BG$ prend deux~:
\[
BG = 2 \times GJ .
\]
Symétriquement, $N$ est le milieu de $[GC]$ et $GN = GI$, donc
$CN = NG = GI$ et $CG = 2 \times GI$~: sur la médiane $[CI]$ aussi,
le point d'intersection se trouve aux deux tiers depuis le \omterm{def:g5:solids:def}{sommet}.

\textbf{8.} Le raisonnement n'a rien utilisé de particulier au
couple $[BJ]$, $[CI]$~: mené sur le couple $[BJ]$, $[AK]$, il montre
que leur point d'intersection est lui aussi sur $[BJ]$ aux deux
tiers depuis $B$. Or il n'y a qu'\emph{un seul} point du \omterm{def:g6:lines:objects}{segment}
$[BJ]$ situé à la distance $\frac23 BJ$ de $B$ --- le point
d'intersection de $[BJ]$ et $[AK]$ est donc ce même $G$. Les trois
médianes passent ainsi par $G$~: elles sont \emph{concourantes}.

\textbf{9.} $BG = \frac23 \times 9 = 6$ cm et
$GJ = \frac13 \times 9 = 3$ cm.

\textbf{10.} D'après la question 8, le même calcul des deux tiers
vaut sur la médiane $[AK]$ (reprendre le raisonnement des questions
4 à 7 avec le couple $[AK]$, $[BJ]$)~: $AG = 2 \times GK$.

\textbf{11.} Dans le triangle $GBC$, le point $K$ est le milieu du
côté $[BC]$, donc $[GK]$ est une médiane de $GBC$~; d'après la
question 2, les triangles $GBK$ et $GKC$ ont la même \omterm{def:g6:measure:area}{aire}.

\textbf{12.} La médiane $[AK]$ découpe $ABC$ en $ABK$ et $AKC$ de
même \omterm{def:g6:measure:area}{aire} (question 2). En retranchant de chacun son petit triangle
de la question 11~:
\[
\text{aire}(ABG) = \text{aire}(ABK) - \text{aire}(GBK),
\qquad
\text{aire}(ACG) = \text{aire}(AKC) - \text{aire}(GKC),
\]
et les \omterm{def:g6:measure:area}{aires} retranchées sont égales, donc
$\text{aire}(ABG) = \text{aire}(ACG)$. Le même calcul le long de la
médiane $[BJ]$ --- qui partage $ABC$ en $ABJ$ et $CBJ$ de même \omterm{def:g6:measure:area}{aire},
tandis que $[GJ]$, médiane du triangle $GAC$, le partage en $GAJ$ et
$GCJ$ --- donne $\text{aire}(ABG) = \text{aire}(CBG)$. Les trois
triangles $ABG$, $BCG$, $CAG$ ont donc la même \omterm{def:g6:measure:area}{aire}, et comme ils
pavent ensemble $ABC$, chacun a pour \omterm{def:g6:measure:area}{aire} le tiers du tout.

\textbf{13.} Dans le triangle $ABG$, le point $I$ est le milieu du
côté $[AB]$, donc $[GI]$ est une médiane de $ABG$ \emph{issue du
\omterm{def:g5:solids:def}{sommet} $G$}~: elle découpe $ABG$ en deux triangles de même \omterm{def:g6:measure:area}{aire}
(question 2), à savoir $AIG$ et $IBG$. Il en va de même dans $BCG$
(médiane $[GK]$) et dans $CAG$ (médiane $[GJ]$). Chaque tiers de
$ABC$ est coupé en deux~: les six parts autour de $G$ ont donc
chacune pour \omterm{def:g6:measure:area}{aire} $\frac13 \times \frac12 = \frac16$ du triangle.

\textbf{14.} Sur la médiane $[AK]$, de $A(0,0)$ à $K(3,3)$~: $G$ se
trouve aux deux tiers, ses coordonnées valent donc
$\left(\frac23 \times 3,\ \frac23 \times 3\right) = (2, 2)$. Sur la
médiane $[BJ]$, de $B(6, 0)$ à $J(0, 3)$~: être aux deux tiers
revient à ajouter les deux tiers du déplacement ($-6$
horizontalement, $+3$ verticalement)~:
\[
\left(6 + \tfrac23 \times (-6),\ 0 + \tfrac23 \times 3\right)
= (2, 2) .
\]
Le même point --- comme la question 8 le promettait. Et
$\frac{0 + 6 + 0}{3} = 2$, $\frac{0 + 0 + 6}{3} = 2$~: les
coordonnées du \omterm{pb:g8:midpoints:1}{centre de gravité} sont les \emph{moyennes} des
coordonnées des trois \omterm{def:g5:solids:def}{sommets}.

\textbf{15.} D'après la question 2, un tranchant de couteau posé le
long d'une médiane laisse la même quantité de carton de chaque côté
(\omterm{def:g6:measure:area}{aires} égales)~: le triangle tient donc en équilibre le long de
chacune des trois droites médianes. Un point d'équilibre doit se
trouver simultanément sur les trois droites d'équilibre, et d'après
la question 8 les trois médianes n'ont qu'un point commun~: $G$.
C'est là qu'il faut poser la pointe du crayon. (L'égalité des \omterm{def:g6:measure:area}{aires}
de part et d'autre est la version crédible de l'argument~; la
version honnête, qui pèse \emph{où} se trouve le carton et pas
seulement combien il y en a, réclame le calcul intégral des \omterm{def:g6:measure:volume}{volumes}
universitaires --- lequel confirme $G$.)
\end{solution}
