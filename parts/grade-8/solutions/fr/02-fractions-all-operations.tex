\section*{Chapitre \ref{ch:g8:fractions} --- Fractions~: les quatre opérations}

\begin{solution}{exo:g8:fractions:1}
$\dfrac34 + \dfrac25 = \dfrac{15}{20} + \dfrac{8}{20} =
\dfrac{23}{20}$.

$\dfrac56 - \dfrac38 = \dfrac{20}{24} - \dfrac{9}{24} =

$\dfrac{7}{10} + \dfrac{8}{15} = \dfrac{21}{30} + \dfrac{16}{30} =

\end{solution}

\begin{solution}{exo:g8:fractions:3}
$\dfrac58 \times \dfrac{4}{15} = \dfrac{5 \times 4}{8 \times 15}
= \dfrac{1}{2 \times 3} = \dfrac16$ (simplifier par $5$ et par $4$).

$\dfrac{9}{14} \times \dfrac73 = \dfrac{9 \times 7}{14 \times 3}
= \dfrac{3}{2}$ (simplifier par $7$ et par $3$).

$\left(-\dfrac65\right) \times \dfrac{10}{9} = -\dfrac{60}{45}
= -\dfrac43$ (signes contraires~: négatif).
\end{solution}

\begin{solution}{exo:g8:fractions:4}

\end{solution}

\begin{solution}{exo:g8:fractions:5}
$\dfrac49 \div \dfrac23 = \dfrac49 \times \dfrac32 = \dfrac{12}{18}
= \dfrac23$.

$\dfrac75 \div 14 = \dfrac75 \times \dfrac{1}{14} = \dfrac{7}{70}
= \dfrac{1}{10}$.

$6 \div \dfrac34 = 6 \times \dfrac43 = 8$.
\end{solution}

\begin{solution}{exo:g8:fractions:6}

$\dfrac{5/6}{10} = \dfrac56 \times \dfrac{1}{10} = \dfrac{5}{60} =

$\dfrac{2}{4/7} = 2 \times \dfrac74 = \dfrac72$.
\end{solution}

\begin{solution}{exo:g8:fractions:7}
\omterm{def:g2:mult:def}{Produit} d'abord~:
$\dfrac13 + \dfrac12 \times \dfrac45 = \dfrac13 + \dfrac{4}{10}
= \dfrac{10}{30} + \dfrac{12}{30} = \dfrac{22}{30} = \dfrac{11}{15}$.

Parenthèses d'abord~:
$\left(\dfrac13 + \dfrac12\right) \times \dfrac45
= \dfrac56 \times \dfrac45 = \dfrac{20}{30} = \dfrac23$.
\end{solution}

\begin{solution}{exo:g8:fractions:9}

\begin{solution}{exo:g8:fractions:10}
$\dfrac{15}{2} \div \dfrac34 = \dfrac{15}{2} \times \dfrac43
= \dfrac{60}{6} = 10$ morceaux --- un nombre entier, pas de corde
perdue.
\end{solution}

\begin{solution}{exo:g8:fractions:11}
$E$~: division d'abord~:
$\dfrac56 \div \dfrac{10}{9} = \dfrac56 \times \dfrac{9}{10}
= \dfrac{45}{60} = \dfrac34$~; puis
$E = \dfrac23 - \dfrac34 = \dfrac{8}{12} - \dfrac{9}{12} =
-\dfrac{1}{12}$.

$F = \left(-\dfrac12\right)^2 \times \dfrac83 = \dfrac14 \times
\dfrac83 = \dfrac{8}{12} = \dfrac23$ (le carré est positif).
\end{solution}

\begin{solution}{exo:g8:fractions:12}
De l'intérieur vers l'extérieur~: $1 + 1 = 2$~; puis
$1 + \dfrac12 = \dfrac32$~; puis
\[
1 + \frac{1}{3/2} = 1 + \frac23 = \frac53 .
\]
\end{solution}

\begin{solution}{pb:g8:fractions:1}
\textbf{1.} $\frac12 + \frac13 = \frac36 + \frac26 = \frac56$ et
$\frac12 + \frac14 = \frac24 + \frac14 = \frac34$. Donc
$\frac56 = \frac12 + \frac13$ et $\frac34 = \frac12 + \frac14$ sont
des écritures égyptiennes (les \omterm{def:g6:fractions:def}{fractions} unitaires y sont bien
distinctes).

\textbf{2.} Avec le \omterm{def:g4:fractions:def}{dénominateur} commun $28$~:
\[
\frac14 + \frac{1}{28} = \frac{7}{28} + \frac{1}{28}
= \frac{8}{28} = \frac{2}{7} .
\]

\textbf{3.} $\frac13 + \frac16 = \frac26 + \frac16 = \frac36 =
\frac12$, et $\frac14 + \frac{1}{12} = \frac{3}{12} +
\frac{1}{12} = \frac{4}{12} = \frac13$. En général, le \omterm{def:g4:fractions:def}{dénominateur}
commun de $\frac{1}{n+1}$ et de $\frac{1}{n(n+1)}$ est $n(n+1)$~:
\[
\frac{1}{n+1} + \frac{1}{n(n+1)}
= \frac{n}{n(n+1)} + \frac{1}{n(n+1)}
= \frac{n+1}{n(n+1)}
= \frac{1}{n} .
\]

\textbf{4.} Partir de $1 = \frac12 + \frac12$ et dédoubler la
seconde \omterm{def:g3:division:half}{moitié} par la question 3~:
\[
1 = \frac12 + \frac13 + \frac16 ,
\]
trois \omterm{def:g6:fractions:def}{fractions} unitaires distinctes.

\textbf{5.} En dédoublant le $\frac13$ de $\frac56 = \frac12 +
\frac13$, on obtient
\[
\frac56 = \frac12 + \frac14 + \frac{1}{12} .
\]
Aucune écriture égyptienne n'est jamais unique~: l'identité de
dédoublement peut toujours remplacer la \omterm{def:g6:fractions:def}{fraction} unitaire de plus
grand \omterm{def:g4:fractions:def}{dénominateur} par deux nouvelles \omterm{def:g6:fractions:def}{fractions} unitaires encore
plus petites --- de toute écriture on en fabrique donc une autre.

\textbf{6.} Avec le \omterm{def:g4:fractions:def}{dénominateur} $21$~: $\frac27 = \frac{6}{21} <
\frac{7}{21} = \frac13$. Avec le \omterm{def:g4:fractions:def}{dénominateur} $28$~: $\frac14 =
\frac{7}{28} < \frac{8}{28} = \frac27$. Les \omterm{def:g6:fractions:def}{fractions} unitaires
décroissent à mesure que leur \omterm{def:g4:fractions:def}{dénominateur} grandit,
$1 > \frac12 > \frac13 > \frac14 > \dots$~; comme $\frac13$ (et tout
ce qui la dépasse) est trop grande et que $\frac14$ tient, $\frac14$
est la plus grande \omterm{def:g6:fractions:def}{fraction} unitaire inférieure à $\frac27$.

\textbf{7.} $\frac27 - \frac14 = \frac{8}{28} - \frac{7}{28} =
\frac{1}{28}$~: la méthode gloutonne \omterm{def:g2:mult:def}{produit}
$\frac27 = \frac14 + \frac{1}{28}$, exactement l'entrée des scribes
de la question 2.

\textbf{8.} Avec le \omterm{def:g4:fractions:def}{dénominateur} $14$~: $\frac12 = \frac{7}{14} <
\frac{10}{14} = \frac57$, tandis que la seule \omterm{def:g6:fractions:def}{fraction} unitaire plus
grande, $1$, dépasse $\frac57$. Donc $\frac12$ est la plus grande
qui tienne, et
\[
\frac57 - \frac12 = \frac{10}{14} - \frac{7}{14} = \frac{3}{14} .
\]

\textbf{9.} Avec le \omterm{def:g4:fractions:def}{dénominateur} $28$~: $\frac{3}{14} = \frac{6}{28}
< \frac{7}{28} = \frac14$, donc $\frac14$ est trop grande. Avec le
\omterm{def:g4:fractions:def}{dénominateur} $70$~: $\frac15 = \frac{14}{70} < \frac{15}{70} =
\frac{3}{14}$, donc $\frac15$ tient~: c'est la plus grande \omterm{def:g6:fractions:def}{fraction}
unitaire inférieure à $\frac{3}{14}$. Alors
\[
\frac{3}{14} - \frac15 = \frac{15}{70} - \frac{14}{70}
= \frac{1}{70},
\qquad\text{d'où}\qquad
\frac57 = \frac12 + \frac15 + \frac{1}{70} .
\]
Vérification~: $\frac{35}{70} + \frac{14}{70} + \frac{1}{70} =
\frac{50}{70} = \frac57$.

\textbf{10.} Première étape~: $\frac12 = \frac{5}{10} <
\frac{8}{10} = \frac45$ et $1 > \frac45$, on retranche donc
$\frac12$~:
$\frac45 - \frac12 = \frac{8}{10} - \frac{5}{10} = \frac{3}{10}$.
Deuxième étape~: $\frac13 = \frac{10}{30} > \frac{9}{30} =
\frac{3}{10}$ est trop grande, et $\frac14 = \frac{5}{20} <
\frac{6}{20} = \frac{3}{10}$ tient~; on retranche donc $\frac14$~:
$\frac{3}{10} - \frac14 = \frac{6}{20} - \frac{5}{20} =
\frac{1}{20}$. Conclusion~:
\[
\frac45 = \frac12 + \frac14 + \frac{1}{20},
\qquad
\frac{10}{20} + \frac{5}{20} + \frac{1}{20} = \frac{16}{20}
= \frac45 . \checkmark
\]

\textbf{11.} Les \omterm{def:g4:fractions:def}{numérateurs} des \omterm{def:g3:division:remainder}{restes} successifs sont~: pour
$\frac27$~: $2$, puis $1$~; pour $\frac57$~: $5$, puis $3$, puis
$1$~; pour $\frac45$~: $4$, puis $3$, puis $1$. Chaque liste est
\emph{strictement décroissante}.

\textbf{12.} Le \omterm{def:g4:fractions:def}{dénominateur} commun de $\frac ab$ et de $\frac1n$
est $b \times n$~:
\[
\frac{a}{b} - \frac{1}{n}
= \frac{a \times n}{b \times n} - \frac{b}{b \times n}
= \frac{a \times n - b}{b \times n} .
\]

\textbf{13.} Au \omterm{def:g4:fractions:def}{dénominateur} commun $b \times (n-1)$, l'inégalité
$\frac{1}{n-1} > \frac ab$ compare les \omterm{def:g4:fractions:def}{numérateurs} $b$ et
$a \times (n - 1)$~:
\[
b > a \times (n - 1) = a \times n - a ,
\]
et en ajoutant $a - b$ aux deux membres on obtient
$a > a \times n - b$. Ainsi, après chaque étape gloutonne, le
\omterm{def:g4:fractions:def}{numérateur} du \omterm{def:g3:division:remainder}{reste} (question 12) est un entier strictement plus
petit que le précédent. Or des entiers ne peuvent pas décroître
strictement indéfiniment~: après au plus $a$ étapes, le \omterm{def:g3:division:remainder}{reste} a pour
\omterm{def:g4:fractions:def}{numérateur} $1$ --- c'est une \omterm{def:g6:fractions:def}{fraction} unitaire --- ou vaut $0$, et
la méthode s'arrête.

\textbf{14.} Avec le \omterm{def:g4:fractions:def}{dénominateur} commun $18$~:
\[
\frac12 + \frac13 + \frac19
= \frac{9}{18} + \frac{6}{18} + \frac{2}{18}
= \frac{17}{18} ,
\]
c'est-à-dire \emph{moins que} $1$~: le testament a oublié
$\frac{1}{18}$ du troupeau. Ce manque est tout le secret du tour.
Sur $18$ chameaux, les parts $\frac12$, $\frac13$, $\frac19$ sont
les entiers $9$, $6$ et $2$, dont la \omterm{def:g1:addition:def}{somme} vaut
$9 + 6 + 2 = 17$~: exactement le troupeau, et le dix-huitième
chameau rentre chez lui. Et personne ne peut se plaindre~: sur $17$
chameaux, le testament promettait $\frac{17}{2} = 8.5$, puis
$\frac{17}{3} \approx 5.67$, puis $\frac{17}{9} \approx 1.89$
chameaux, et chaque enfant a reçu strictement davantage ($9$, $6$ et
$2$). Le $\frac{1}{18}$ oublié est ce que le chameau du voisin a
comblé le temps du partage.

\textbf{15.} La question 4 fournit le testament réparé~:
\[
\frac12 + \frac13 + \frac16 = 1 .
\]
Les parts sont entières lorsque la taille du troupeau est \omterm{def:g6:wholes:divisible}{divisible}
par $2$, par $3$ et par $6$ --- c'est-à-dire \omterm{def:g6:wholes:divisible}{divisible} par $6$. Le
plus petit troupeau convenable compte $6$ chameaux~: parts $3$, $2$
et $1$, et rien ne reste.
\end{solution}
