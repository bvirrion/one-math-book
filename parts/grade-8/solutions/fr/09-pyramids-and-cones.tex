\end{solution}

\begin{solution}{exo:g8:solids:4}

\begin{solution}{exo:g8:solids:6}
La surface latérale se déroule en un secteur de disque dont le \omterm{def:g3:shapes:shapes}{rayon}
est la \emph{génératrice} $s$ (la distance du \omterm{def:g5:solids:def}{sommet} au bord --- ce

\end{solution}

\begin{solution}{exo:g8:solids:7}

\begin{solution}{exo:g8:solids:8}
Hauteur~: $h = \sqrt{13^2 - 5^2} = \sqrt{169 - 25} = \sqrt{144} = 12$
cm. \omterm{def:g6:measure:volume}{Volume}~: $V = \frac13 \pi \times 25 \times 12 = 100\pi$ cm$^3$.
\end{solution}

\begin{solution}{exo:g8:solids:9}
Même \omterm{def:g3:shapes:shapes}{rayon}, donc le \omterm{def:g6:measure:volume}{volume} du \omterm{def:g8:solids:def}{cône} est le tiers du \omterm{def:g6:measure:volume}{volume} du \omterm{def:g7:areas:prism}{cylindre}
\emph{pour la même hauteur}~: le liquide atteint $\frac{9}{3} = 3$ cm
dans le \omterm{def:g7:areas:prism}{cylindre}.
\end{solution}

\begin{solution}{exo:g8:solids:10}
Sable $=$ un \omterm{def:g8:solids:def}{cône}~: $V = \frac13 \pi \times 4 \times 4.5 = 6\pi
\approx 18.8$ cm$^3$. Le sablier contient deux tels \omterm{def:g8:solids:def}{cônes}, donc le
sable remplit exactement la \omterm{def:g3:division:half}{moitié} du \omterm{def:g6:measure:volume}{volume} intérieur total.
\end{solution}

\begin{solution}{exo:g8:solids:11}
\emph{1.} Oui~: la formule $V = \frac13 Bh$ vaut pour toute position
du \omterm{def:g5:solids:def}{sommet}, tant que $h$ est la distance \omterm{def:g6:lines:perp}{perpendiculaire} au plan de la
base. $V = \frac13 \times 36 \times 8 = 96$ cm$^3$.

\emph{2.} Diagonale de la base~: $\sqrt{6^2 + 6^2} = 6\sqrt2$ cm. La
plus longue \omterm{def:g5:solids:def}{arête} latérale est l'hypoténuse d'un \omterm{def:g6:shapes:triangles}{triangle rectangle} de
côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit} $6\sqrt2$ (dans le plan de la base) et $8$
(vertical)~:
\[
\sqrt{(6\sqrt2)^2 + 8^2} = \sqrt{72 + 64} = \sqrt{136} \approx 11.7
\text{ cm}.
\]
\end{solution}

\begin{solution}{pb:g8:solids:1}
\textbf{1.} $G$ appartient à la \omterm{def:g5:solids:def}{face} supérieure $EFGH$ et aux deux
\omterm{def:g5:solids:def}{faces} latérales $BCGF$ et $DCGH$. Les trois \omterm{def:g5:solids:def}{faces} qui ne contiennent
\emph{pas} $G$ sont la base $ABCD$, la \omterm{def:g5:solids:def}{face} avant $ABFE$ et la \omterm{def:g5:solids:def}{face}
gauche $ADHE$ (elles ont toutes le \omterm{def:g5:solids:def}{sommet} $A$ en commun, le coin
opposé à $G$). Les trois \omterm{def:g8:solids:def}{pyramides} sont $ABCDG$, $ABFEG$ et
$ADHEG$~: chacune a pour base une \omterm{def:g5:solids:def}{face} carrée du \omterm{def:g5:solids:def}{cube} et pour \omterm{def:g5:solids:def}{sommet}
le lointain $G$. Chacune est une \omterm{def:g8:solids:def}{pyramide} à base carrée~: $5$ \omterm{def:g5:solids:def}{faces},
$8$ \omterm{def:g5:solids:def}{arêtes}, $5$ \omterm{def:g5:solids:def}{sommets} (\cref{ex:g8:solids:count}).

\textbf{2.} La grande diagonale $[AG]$ joint les deux seuls \omterm{def:g5:solids:def}{sommets}
n'appartenant à aucune des trois bases. Une rotation d'un tiers de
tour autour de la droite $(AG)$ envoie le \omterm{def:g5:solids:def}{cube} sur lui-même (les
trois \omterm{def:g5:solids:def}{arêtes} issues de $A$ --- vers $B$, $D$, $E$ --- sont permutées
circulairement, de même que les trois \omterm{def:g5:solids:def}{arêtes} aboutissant à $G$).
Elle transporte la base $ABCD$ sur $ADHE$, puis $ADHE$ sur $ABFE$,
et ainsi de suite~: les \omterm{def:g5:solids:def}{sommets} restent en $G$, les bases tournent.
Chaque \omterm{def:g8:solids:def}{pyramide} est ainsi transportée sur la suivante~: les trois
sont des copies identiques.

\textbf{3.} Trois morceaux identiques remplissant le \omterm{def:g5:solids:def}{cube} s'en
partagent le \omterm{def:g6:measure:volume}{volume} à parts égales~:
\[
V_{\text{pyramide}} = \frac{a^3}{3} .
\]

\textbf{4.} La \omterm{def:g8:solids:def}{pyramide} $ABCDG$ a pour base $ABCD$, d'\omterm{def:g6:measure:area}{aire} $a^2$, et
son \omterm{def:g5:solids:def}{sommet} $G$ se trouve à la verticale du coin $C$, à la hauteur
$CG = a$ (une \omterm{def:g5:solids:def}{arête} verticale du \omterm{def:g5:solids:def}{cube}). Comme dans
\cref{exo:g8:solids:11}, la formule s'applique avec la hauteur
\omterm{def:g6:lines:perp}{perpendiculaire} $h = a$~:
\[
V = \frac13 \times a^2 \times a = \frac{a^3}{3} ,
\]
en parfait accord avec la question 3 --- la décomposition et la
formule se confirment l'une l'autre.

\textbf{5.} Pour $a = 6$~: $V = \frac{6^3}{3} = \frac{216}{3} = 72$
cm$^3$ chacune. L'expérience du transvasement est cette
décomposition rendue liquide~: le \omterm{def:g7:areas:prism}{prisme} de base $ABCD$ et de
hauteur $a$ est le \omterm{def:g5:solids:def}{cube} lui-même, et il contient exactement trois
remplissages de la \omterm{def:g8:solids:def}{pyramide} --- ici, les trois remplissages
s'assemblent même géométriquement en le \omterm{def:g5:solids:def}{cube}.

\textbf{6.} Joindre le centre $O$ aux quatre coins de chaque \omterm{def:g5:solids:def}{face}
\omterm{def:g2:mult:def}{produit} six \omterm{def:g8:solids:def}{pyramides}, une par \omterm{def:g5:solids:def}{face}~: base carrée, \omterm{def:g5:solids:def}{sommet} $O$. Le
centre est également placé par rapport aux six \omterm{def:g5:solids:def}{faces} (toute rotation
ou symétrie du \omterm{def:g5:solids:def}{cube} fixe $O$ et permute les \omterm{def:g5:solids:def}{faces})~: les six
\omterm{def:g8:solids:def}{pyramides} sont donc identiques. La hauteur de chacune est la
distance de $O$ à une \omterm{def:g5:solids:def}{face}~: la \omterm{def:g3:division:half}{moitié} de l'\omterm{def:g5:solids:def}{arête}, $\frac{a}{2}$.

\textbf{7.} Six morceaux identiques se partagent le \omterm{def:g5:solids:def}{cube}~:
$V = \frac{a^3}{6}$ chacun.

\textbf{8.} Par la formule~: \omterm{def:g6:measure:area}{aire} de base $a^2$, hauteur
$\frac a2$~:
\[
V = \frac13 \times a^2 \times \frac{a}{2} = \frac{a^3}{6} .
\]
Les deux réponses concordent.

\textbf{9.} Les deux décompositions ne concernent que des \omterm{def:g8:solids:def}{pyramides}
aux proportions très particulières, taillées dans un \omterm{def:g5:solids:def}{cube}~; une
\omterm{def:g8:solids:def}{pyramide} élancée, une \omterm{def:g8:solids:def}{pyramide} de guingois ou un \omterm{def:g8:solids:def}{cône} ne s'assemblent
pas ainsi (trois \omterm{def:g8:solids:def}{cônes} ne remplissent pas un \omterm{def:g7:areas:prism}{cylindre}). C'est
pourquoi \cref{thm:g8:solids:volume} est \emph{admis} à ce niveau~:
les décompositions et l'expérience du transvasement rendent le
$\frac13$ tout à fait crédible, et la démonstration générale honnête
--- l'intégration --- est donnée dans le \omterm{def:g6:measure:volume}{volume} suivant.

\textbf{10.} $V = \frac{6^3}{6} = 36$ cm$^3$~; et par la formule,
$V = \frac13 \times 36 \times 3 = 36$ cm$^3$. Les deux concordent.

\textbf{11.} Dans le triangle $SAB$, les milieux des côtés $[SA]$ et
$[SB]$ sont joints par un \omterm{def:g6:lines:objects}{segment} \omterm{def:g6:lines:perp}{parallèle} à $(AB)$ et de longueur
$\frac{AB}{2} = \frac{c}{2}$ (\cref{thm:g8:midpoints:direct}). Il en
va de même sur chaque \omterm{def:g5:solids:def}{face} latérale~: les quatre milieux forment
donc un carré de côté $\frac c2$, aux côtés \omterm{def:g6:lines:perp}{parallèles} à ceux de la
base. Sur la verticale issue du \omterm{def:g5:solids:def}{sommet}, la coupe passe au milieu de
la hauteur (à nouveau le théorème des milieux, dans un triangle
passant par $S$, le centre de la base et un \omterm{def:g5:solids:def}{sommet} de la base)~: le
plan de coupe se situe à la hauteur $\frac h2$.

\textbf{12.} Le morceau supérieur est une \omterm{def:g8:solids:def}{pyramide} régulière à base
carrée, de côté de base $\frac c2$ et de hauteur $\frac h2$ (sa base
est la section, son \omterm{def:g5:solids:def}{sommet} est $S$). Son \omterm{def:g6:measure:volume}{volume} vaut
\[
\frac13 \times \left(\frac c2\right)^2 \times \frac h2
= \frac13 \times \frac{c^2}{4} \times \frac h2
= \frac{1}{8} \times \frac{c^2 h}{3}
= \frac{V}{8} :
\]
le huitième du \omterm{def:g6:measure:volume}{volume} initial --- et non la \omterm{def:g3:division:half}{moitié}.

\textbf{13.} Le tronc de \omterm{def:g8:solids:def}{pyramide} garde le \omterm{def:g3:division:remainder}{reste}~:
$V - \frac V8 = \frac{7V}{8}$. La coupe à mi-hauteur partage donc le
\omterm{def:g6:measure:volume}{volume} dans le rapport $1 : 7$ --- la dalle du bas, d'apparence
modeste, contient sept fois le \omterm{def:g6:measure:volume}{volume} de la pointe.

\textbf{14.} Sous la mi-hauteur se trouvent les $\frac78$ du \omterm{def:g6:measure:volume}{volume}.
Avec $V = \frac13 \times 35^2 \times 22 = \frac{26\,950}{3} \approx
8\,983$ m$^3$ (\cref{exo:g8:solids:7})~:
\[
\frac78 \times \frac{26\,950}{3} \approx 7\,860 \approx
7\,900 \text{ m}^3
\]
à la centaine près --- la campagne de la «~\omterm{def:g3:division:half}{moitié} inférieure~»
nettoie près de huit fois le \omterm{def:g6:measure:volume}{volume} de verre de la campagne
supérieure.

\textbf{15.} Doubler toutes les dimensions multiplie l'\omterm{def:g6:measure:area}{aire} de base
par $2^2 = 4$ et la hauteur par $2$, donc le \omterm{def:g6:measure:volume}{volume} par
$4 \times 2 = 8 = 2^3$. Un agrandissement de rapport $k$ multiplie
les \omterm{def:g6:measure:area}{aires} par $k^2$ et une longueur de plus par $k$~: les \omterm{def:g6:measure:volume}{volumes}
sont donc multipliés par $k^3$. Le morceau supérieur de la question
12 est une réduction de la \omterm{def:g8:solids:def}{pyramide} entière de rapport
$k = \frac12$~: son \omterm{def:g6:measure:volume}{volume} vaut donc
$\left(\frac12\right)^3 = \frac18$ du tout --- voilà l'explication
en une ligne.
\end{solution}
