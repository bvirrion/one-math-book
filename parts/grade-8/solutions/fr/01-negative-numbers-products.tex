\section*{Chapitre \ref{ch:g8:negprod} --- Multiplier des nombres relatifs}

\begin{solution}{exo:g8:negprod:1}
$(-8) \times (+7) = -56$~; \quad
$(-9) \times (-6) = +54$~; \quad
$(+12) \times (-0.5) = -6$~; \quad
$(-1) \times (-1) \times (-1) = -1$ (trois facteurs négatifs~: \omterm{def:g3:numbers:evenodd}{impair}).
\end{solution}

\begin{solution}{exo:g8:negprod:2}
$(-63) \div (+9) = -7$~; \quad
$(-8) \div (-16) = +0.5$~; \quad
$\dfrac{45}{-5} = -9$~; \quad
$\dfrac{-7.2}{-8} = +0.9$.
\end{solution}

\begin{solution}{exo:g8:negprod:3}
Trois facteurs négatifs~: \emph{négatif}. \quad
$(-1)^{10}$~: dix facteurs négatifs, \omterm{def:g3:numbers:evenodd}{pair}~: \emph{positif} (cela
vaut $1$). \quad
$(-2)^7$~: sept facteurs, \omterm{def:g3:numbers:evenodd}{impair}~: \emph{négatif}. \quad
$(-4) \times 0 \times (-6) = 0$~: ni positif ni négatif.
\end{solution}

\begin{solution}{exo:g8:negprod:4}
$(-2)^4 = +16$~; \quad $-2^4 = -16$ (le moins n'est pas élevé au
carré)~; \quad
$(-3)^3 = -27$~; \quad $(-1)^{2026} = +1$ (exposant \omterm{def:g3:numbers:evenodd}{pair}).
\end{solution}

\begin{solution}{exo:g8:negprod:5}
$7 + 2 \times (-6) = 7 + (-12) = -5$.

$(-4) \times (5 - 9) = (-4) \times (-4) = +16$.

$18 \div (-3) - (-2) = -6 + 2 = -4$.
\end{solution}

\begin{solution}{exo:g8:negprod:6}
$\dfrac{(-5) \times (+8)}{-4} = \dfrac{-40}{-4} = +10$.

$\dfrac{(-3) \times (-6)}{(-2) \times (+9)} = \dfrac{+18}{-18} = -1$.
\end{solution}

\begin{solution}{exo:g8:negprod:7}
Pour $x = -3$~: \ $5x = -15$~; \quad $x^2 = (-3)^2 = 9$~; \quad
$-x^2 = -9$~; \quad $2x^2 + 4x = 18 - 12 = 6$~; \quad
$(2x)^2 = (-6)^2 = 36$.
\end{solution}

\begin{solution}{exo:g8:negprod:8}
$(-6) \times (-7) = 42$~; \quad
$10 \div (-4) = -2.5$~; \quad
$(-5) \times \frac15 = -1$, donc le nombre manquant est
$+\frac15 = 0.2$.
\end{solution}

\begin{solution}{exo:g8:negprod:9}
\emph{1. Faux}~: $0.5 \times 0.5 = 0.25$ est plus petit que
$0.5 + 0.5 = 1$ (ou~: $2 \times 3 = 6 > 5$, mais l'affirmation doit
valoir toujours).

\emph{2. Vrai}~: un carré est un \omterm{def:g2:mult:def}{produit} de deux nombres de même
signe, donc positif ou nul.

\emph{3. Faux}~: $(-2) \times (-3) \times 4 = +24$ est positif avec
deux facteurs négatifs.
\end{solution}

\begin{solution}{exo:g8:negprod:10}
Soit $r$ le nombre de bonnes réponses~; alors $20 - r$ sont fausses,
et
\[
5r - 3(20 - r) = 36
\ \Longrightarrow\
5r - 60 + 3r = 36
\ \Longrightarrow\
8r = 96
\ \Longrightarrow\
r = 12 .
\]
Douze bonnes réponses (et huit fausses~: vérification,
$60 - 24 = 36$).
\end{solution}

\begin{solution}{exo:g8:negprod:11}
\omterm{def:g1:counting:zero}{Zéro} multiplié par quoi que ce soit donne \omterm{def:g1:counting:zero}{zéro}, donc
$(-1) \times 0 = 0$. En écrivant $0 = 1 + (-1)$ et en distribuant~:
\[
0 = (-1) \times \bigl(1 + (-1)\bigr)
= (-1) \times 1 + (-1) \times (-1)
= -1 + (-1) \times (-1).
\]
Le nombre $(-1) \times (-1)$ ajouté à $-1$ donne $0$~: ce doit être
l'opposé de $-1$, c'est-à-dire $+1$. La distributivité ne laisse pas
d'autre choix.
\end{solution}

\begin{solution}{pb:g8:negprod:1}
\textbf{1.} Puisque $0 = 0 + 0$, la distributivité
(\cref{thm:g7:priorities:distributivity}) donne
\[
0 \times a = (0 + 0) \times a = 0 \times a + 0 \times a .
\]
Ajouter $0 \times a$ à lui-même ne change donc rien --- et le seul
nombre que l'on puisse ajouter sans rien changer est $0$.
(Directement~: retrancher $0 \times a$ des deux côtés.) D'où
$0 \times a = 0$.

\textbf{2.} En développant, puis en utilisant $1 + (-1) = 0$ et la
question 1~:
\[
a + (-1) \times a
= 1 \times a + (-1) \times a
= \bigl(1 + (-1)\bigr) \times a
= 0 \times a = 0 .
\]
Ainsi $(-1) \times a$ est le nombre qui, ajouté à $a$, donne $0$~:
c'est exactement l'opposé de $a$, et $(-1) \times a = -a$.
\Cref{exo:g8:negprod:11} est le cas particulier $a = -1$~: on y
trouve $(-1) \times (-1) = -(-1) = +1$.

\textbf{3.} Écrivons $-a = (-1) \times a$ (question 2) et
regroupons les facteurs~:
\[
(-a) \times b
= \bigl((-1) \times a\bigr) \times b
= (-1) \times (a \times b)
= -(a \times b) .
\]

\textbf{4.} Appliquons deux fois la question 3 (une fois sur chaque
facteur)~:
\[
(-a) \times (-b)
= -\bigl(a \times (-b)\bigr)
= -\bigl(-(a \times b)\bigr)
= a \times b ,
\]
puisque l'opposé de l'opposé d'un nombre est ce nombre lui-même.

\textbf{5.} Écrivons $a = +d$ ou $a = -d$ et $b = +e$ ou $b = -e$,
où $d$ et $e$ sont les distances à \omterm{def:g1:counting:zero}{zéro}. Les quatre cas~:
\[
(+d) \times (+e) = de, \quad
(+d) \times (-e) = -de, \quad
(-d) \times (+e) = -de, \quad
(-d) \times (-e) = +de,
\]
les deux du milieu par la question 3, le dernier par la question 4.
Dans tous les cas, la distance à \omterm{def:g1:counting:zero}{zéro} du résultat vaut
$d \times e$, le \omterm{def:g2:mult:def}{produit} des distances, et le signe est celui
qu'annonce le tableau du~\cref{thm:g8:negprod:rules}~: mêmes signes,
résultat positif~; signes contraires, résultat négatif.

\textbf{6.} $(-1)^2 = +1$, $(-1)^3 = -1$, $(-1)^4 = +1$,
$(-1)^5 = -1$. En général, $(-1)^n$ est un \omterm{def:g2:mult:def}{produit} de $n$ facteurs
négatifs, chacun de distance $1$~: d'après
\cref{prop:g8:negprod:many}, il vaut $+1$ lorsque $n$ est \omterm{def:g3:numbers:evenodd}{pair} et
$-1$ lorsque $n$ est \omterm{def:g3:numbers:evenodd}{impair}.

\textbf{7.} Il y a dix facteurs négatifs --- un \omterm{def:g3:numbers:evenodd}{nombre pair} --- donc
le \omterm{def:g2:mult:def}{produit} est \emph{positif} (\cref{prop:g8:negprod:many}). Sa
distance à \omterm{def:g1:counting:zero}{zéro} vaut $1 \times 2 \times 3 \times \dots \times 10$.

\textbf{8.} Par la question 4,
$(-a)^2 = (-a) \times (-a) = a \times a = a^2$. Puis, par la
question 3,
\[
(-a)^3 = (-a)^2 \times (-a) = a^2 \times (-a) = -(a^2 \times a)
= -a^3 .
\]
Le signe moins disparaît pour les exposants \omterm{def:g3:numbers:evenodd}{pairs} et survit pour les
exposants \omterm{def:g3:numbers:evenodd}{impairs}, conformément à l'\cref{ex:g8:negprod:powers}.

\textbf{9.} Si $a = 0$, alors $a^2 = 0$. Sinon, $a$ et $a$ ont le
\emph{même} signe, donc $a^2 = a \times a$ est positif
(\cref{thm:g8:negprod:rules}). Un carré n'est donc jamais négatif~;
comme $-4$ est négatif, aucun \omterm{def:g7:negatives:def}{nombre relatif} $x$ ne vérifie
$x^2 = -4$.

\textbf{10.} Zoé a raison. Les exposants $2026$ et $2027$ sont de
parités contraires, donc $(-1)^{2026} = +1$ et
$(-1)^{2027} = -1$, dont la \omterm{def:g1:addition:def}{somme} vaut $0$. En général, de deux
exposants consécutifs l'un est \omterm{def:g3:numbers:evenodd}{pair} et l'autre \omterm{def:g3:numbers:evenodd}{impair}~: les deux
puissances valent donc $+1$ et $-1$ dans un ordre ou dans l'autre
--- deux nombres opposés, dont la \omterm{def:g1:addition:def}{somme} est toujours $0$.

\textbf{11.} $A_3 = 1 - 2 + 3 = 2$, puis $A_4 = A_3 - 4 = -2$,
$A_5 = A_4 + 5 = 3$, $A_6 = A_5 - 6 = -3$, $A_7 = A_6 + 7 = 4$.
Conjecture~: $A_n = -\frac{n}{2}$ pour $n$ \omterm{def:g3:numbers:evenodd}{pair}, et
$A_n = \frac{n+1}{2}$ pour $n$ \omterm{def:g3:numbers:evenodd}{impair}.

\textbf{12.} Pour $n$ \omterm{def:g3:numbers:evenodd}{pair}, les termes s'apparient complètement~:
\[
A_n = (1 - 2) + (3 - 4) + \dots + \bigl((n-1) - n\bigr),
\]
et chaque paire vaut $-1$. Il y a $\frac{n}{2}$ paires (deux termes
par paire, $n$ termes en tout), donc
$A_n = \frac{n}{2} \times (-1) = -\frac{n}{2}$.

\textbf{13.} Pour $n$ \omterm{def:g3:numbers:evenodd}{impair}, la \omterm{def:g1:addition:def}{somme} $A_n$ est la \omterm{def:g1:addition:def}{somme} $A_{n-1}$
des $n - 1$ premiers termes, augmentée du dernier terme, qui vaut
$+n$ puisque $n$ est \omterm{def:g3:numbers:evenodd}{impair}. Comme $n - 1$ est \omterm{def:g3:numbers:evenodd}{pair}, la question 12
donne
\[
A_n = A_{n-1} + n = -\frac{n-1}{2} + n
= \frac{-(n-1) + 2n}{2} = \frac{n+1}{2} .
\]

\textbf{14.} $A_{100} = -\frac{100}{2} = -50$, \quad
$A_{101} = \frac{102}{2} = 51$, \quad
$A_{2026} = -\frac{2026}{2} = -1013$.

\textbf{15.} Aucune surprise~: passer de $A_{100}$ à $A_{101}$
ajoute exactement un terme, à savoir $+101$~; la \omterm{ex:g1:subtraction:difference}{différence} devait
donc valoir $101$. Pour la même raison,
$A_{2027} - A_{2026} = +2027$ --- sans calculer $A_{2027}$ du tout.
\end{solution}
