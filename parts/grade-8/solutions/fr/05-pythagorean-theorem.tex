\begin{solution}{exo:g8:pythagoras:1}
$\sqrt{81 + 144} = \sqrt{225} = 15$~; \quad
$\sqrt{25 + 144} = \sqrt{169} = 13$~; \quad
$\sqrt{64 + 225} = \sqrt{289} = 17$.
\end{solution}

\begin{solution}{exo:g8:pythagoras:2}
Côté$^2 = 25^2 - 7^2 = 625 - 49 = 576$, donc le côté mesure
$\sqrt{576} = 24$.
\end{solution}

\begin{solution}{exo:g8:pythagoras:3}
Rectangle~: $\sqrt{12^2 + 9^2} = \sqrt{144 + 81} = \sqrt{225} = 15$
cm.

Carré~: $d = 5\sqrt2 \approx 7.1$ cm
(\cref{ex:g8:pythagoras:diagonal}).
\end{solution}

\begin{solution}{exo:g8:pythagoras:4}
$(20, 21, 29)$~: $29^2 = 841$ et $20^2 + 21^2 = 400 + 441 = 841$~:
rectangle.

$(7, 8, 11)$~: $121 \neq 49 + 64 = 113$~: non rectangle.

$(1.5, 2, 2.5)$~: $6.25 = 2.25 + 4$~: rectangle (c'est
$(3,4,5)$ réduit de \omterm{def:g3:division:half}{moitié}).
\end{solution}

\begin{solution}{exo:g8:pythagoras:5}
Planche $= \sqrt{3^2 + 1.6^2} = \sqrt{9 + 2.56} = \sqrt{11.56} = 3.4$
m.
\end{solution}

\begin{solution}{exo:g8:pythagoras:6}
La hauteur tombe au milieu de la base (\omterm{def:g6:shapes:triangles}{triangle isocèle}), donc chaque
demi-triangle a pour hypoténuse $10$ et un côté de l'\omterm{def:g3:shapes:rightangle}{angle droit}
$6$~: $h = \sqrt{100 - 36} = \sqrt{64} = 8$ cm. \omterm{def:g6:measure:area}{Aire}~:
$\frac{12 \times 8}{2} = 48$ cm$^2$.
\end{solution}

\begin{solution}{exo:g8:pythagoras:7}
Diagonale~: $\sqrt{88.5^2 + 49.8^2} = \sqrt{7832.25 + 2480.04}
= \sqrt{10312.29} \approx 101.5$ cm. En pouces~:
$101.5 \div 2.54 \approx 40$ pouces.
\end{solution}

\begin{solution}{exo:g8:pythagoras:8}
Les deux côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit} mesurent $24$ km et $10$ km~:
$\sqrt{24^2 + 10^2} = \sqrt{576 + 100} = \sqrt{676} = 26$ km.
\end{solution}

\begin{solution}{exo:g8:pythagoras:9}
Distance du pied~: $\sqrt{2.5^2 - 2.4^2} = \sqrt{6.25 - 5.76} =
\sqrt{0.49} = 0.7$ m. Un \omterm{def:g3:division:half}{quart} de l'échelle vaut
$2.5 \div 4 \approx 0.6$ m~: les $0.7$ m trouvés sont proches du
placement conseillé --- acceptable.
\end{solution}

\begin{solution}{exo:g8:pythagoras:10}
Les côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit} mesurent $5 - 1 = 4$ (horizontal) et
$5 - 2 = 3$ (vertical), donc
\[
AB = \sqrt{4^2 + 3^2} = \sqrt{25} = 5 .
\]
\end{solution}

\begin{solution}{exo:g8:pythagoras:11}
\emph{1.} Directement~: $(a + b)^2$. En morceaux~: quatre triangles
d'\omterm{def:g6:measure:area}{aire} $\frac{ab}{2}$ chacun, plus le carré incliné $c^2$~:
\[
(a + b)^2 = 4 \times \frac{ab}{2} + c^2 = 2ab + c^2 .
\]

\emph{2.} En développant le membre de gauche~:
$(a+b)^2 = a^2 + 2ab + b^2$. Donc
\[
a^2 + 2ab + b^2 = 2ab + c^2
\quad\Longrightarrow\quad
a^2 + b^2 = c^2 . \qed
\]
\end{solution}

\begin{solution}{pb:g8:pythagoras:1}
\textbf{1.} $5^2 + 12^2 = 25 + 144 = 169 = 13^2$~; \quad
$8^2 + 15^2 = 64 + 225 = 289 = 17^2$~; \quad
$20^2 + 21^2 = 400 + 441 = 841 = 29^2$. Les trois sont des
triplets, et d'après \cref{thm:g8:pythagoras:converse} un triangle
ayant ces longueurs de côtés est rectangle, l'\omterm{def:g3:shapes:rightangle}{angle droit} faisant
face au côté le plus long.

\textbf{2.} La corde forme un triangle de côtés $3$, $4$, $5$ (en
\omterm{def:g6:lines:objects}{segments}), et $3^2 + 4^2 = 25 = 5^2$~: par la réciproque
(\cref{thm:g8:pythagoras:converse}), l'angle entre les côtés de $3$
et $4$ \omterm{def:g6:lines:objects}{segments} est exactement droit --- aucun rapporteur n'est
nécessaire, seulement des nœuds.

\textbf{3.} Si $a^2 + b^2 = c^2$, alors
\[
(ka)^2 + (kb)^2 = k^2 a^2 + k^2 b^2
= k^2\left(a^2 + b^2\right) = k^2 c^2 = (kc)^2 .
\]
À partir de $(3, 4, 5)$~: par exemple $(6, 8, 10)$, $(9, 12, 15)$
et $(30, 40, 50)$.

\textbf{4.} Un \omterm{def:g5:division:multiple}{multiple} $(3k, 4k, 5k)$ dont le plus petit nombre
serait $5$ exigerait $3k = 5$, et aucun entier $k$ ne le vérifie
($k = 1$ donne $3$, $k = 2$ donne $6$). Donc $(5, 12, 13)$ n'est pas
un \omterm{def:g5:division:multiple}{multiple} de $(3, 4, 5)$.

\textbf{5.} La question 3 montre que l'agrandissement ne \omterm{def:g2:mult:def}{produit} que
les triplets $(3k, 4k, 5k)$~; la question 4 exhibe un véritable
triplet qui n'est pas de cette forme. La famille de tous les
\omterm{pb:g8:pythagoras:1}{triplets pythagoriciens} est donc strictement plus vaste que les
\omterm{def:g5:division:multiple}{multiples} de $(3, 4, 5)$~: pour les produire, il faut du nouveau.

\textbf{6.} Posons $A = m^2$ et $B = n^2$. Par les identités
remarquables (\cref{pb:g8:equations:1})~:
\[
a^2 = (A - B)^2 = A^2 - 2AB + B^2,
\qquad
b^2 = (2mn)^2 = 4m^2n^2 = 4AB .
\]
En additionnant~:
\[
a^2 + b^2 = A^2 - 2AB + B^2 + 4AB
= A^2 + 2AB + B^2 = (A + B)^2
= \left(m^2 + n^2\right)^2 = c^2 .
\]

\textbf{7.}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccc}
$(m, n)$ & $m^2 - n^2$ & $2mn$ & $m^2 + n^2$ \\
\hline
$(2, 1)$ & $3$ & $4$ & $5$ \\
$(3, 1)$ & $8$ & $6$ & $10$ \\
$(3, 2)$ & $5$ & $12$ & $13$ \\
$(4, 1)$ & $15$ & $8$ & $17$ \\
$(4, 3)$ & $7$ & $24$ & $25$ \\
\end{tabular}
\end{center}

\textbf{8.} $(8, 6, 10)$ est le double de $(4, 3, 5)$ --- autrement
dit du $(3, 4, 5)$ des bâtisseurs. La machine redécouvre parfois un
vieux triplet déguisé.

\textbf{9.} Il faut $m^2 - n^2 = 21$ et $2mn = 20$, donc
$mn = 10$~: en essayant $m = 5$, $n = 2$, on obtient
$25 - 4 = 21$ et $2 \times 10 = 20$, avec l'hypoténuse
$25 + 4 = 29$. La machine \omterm{def:g2:mult:def}{produit} $(21, 20, 29)$ pour
$(m, n) = (5, 2)$.

\textbf{10.} $m = 8$, $n = 6$ donnent $m^2 + n^2 = 64 + 36 = 100$.
Le triplet est
\[
\left(64 - 36,\ 2 \times 48,\ 100\right) = (28,\ 96,\ 100),
\]
et l'on a bien $28^2 + 96^2 = 784 + 9216 = 10\,000 = 100^2$.

\textbf{11.} Avec $m = n + 1$~:
\[
m^2 - n^2 = (n + 1)^2 - n^2 = 2n + 1
\]
(\omterm{ex:g1:subtraction:difference}{différence} de carrés consécutifs, \cref{pb:g8:equations:1})~;
$2mn = 2n(n + 1) = 2n^2 + 2n$~; et
$m^2 + n^2 = n^2 + n^2 + 2n + 1 = 2n^2 + 2n + 1$. L'hypoténuse
dépasse le côté \omterm{def:g3:numbers:evenodd}{pair} d'exactement $1$. Pour $n = 1, 2, 3, 4$~:
\[
(3, 4, 5), \quad (5, 12, 13), \quad (7, 24, 25), \quad
(9, 40, 41) .
\]

\textbf{12.} Tout \omterm{def:g3:numbers:evenodd}{nombre impair} au moins égal à $3$ s'écrit
$2n + 1$ pour un certain $n \geq 1$, et la question 11 fournit un
triplet dont le côté \omterm{def:g3:numbers:evenodd}{impair} vaut exactement $2n + 1$. Pour
$11 = 2 \times 5 + 1$, prendre $n = 5$~:
\[
(11,\ 60,\ 61),
\qquad
11^2 + 60^2 = 121 + 3600 = 3721 = 61^2 .
\]

\textbf{13.} Avec $n = 1$, la machine donne
$\left(m^2 - 1,\ 2m,\ m^2 + 1\right)$, dont le côté \omterm{def:g3:numbers:evenodd}{pair} $2m$ peut
être n'importe quel \omterm{def:g3:numbers:evenodd}{nombre pair} à partir de $4$. Pour
$14 = 2 \times 7$, prendre $m = 7$~: le triplet $(48, 14, 50)$. Le
même triplet apparaît en doublant $(7, 24, 25)$ du tableau de la
question 7~: $(14, 48, 50)$ --- les trois mêmes nombres.

\textbf{14.} Un \omterm{def:g3:numbers:evenodd}{nombre pair} s'écrit $2k$, et $(2k)^2 = 4k^2$ est
\omterm{def:g3:numbers:evenodd}{pair} (c'est un \omterm{def:g5:division:multiple}{multiple} de $4$, donc \omterm{def:g3:numbers:evenodd}{pair}). Un \omterm{def:g3:numbers:evenodd}{nombre impair}
s'écrit $2k + 1$, et la première identité remarquable donne
\[
(2k + 1)^2 = 4k^2 + 4k + 1 = 2\left(2k^2 + 2k\right) + 1 ,
\]
un \omterm{def:g3:numbers:evenodd}{nombre pair} augmenté de $1$~: c'est \omterm{def:g3:numbers:evenodd}{impair}.

\textbf{15.} Supposons $a$, $b$, $c$ tous \omterm{def:g3:numbers:evenodd}{impairs}. D'après la
question 14, $a^2$ et $b^2$ sont \omterm{def:g3:numbers:evenodd}{impairs}, donc $a^2 + b^2$ est une
\omterm{def:g1:addition:def}{somme} de deux \omterm{def:g3:numbers:evenodd}{nombres impairs}~: elle est \emph{paire}. Or $c^2$
serait \emph{\omterm{def:g3:numbers:evenodd}{impair}}. L'égalité $a^2 + b^2 = c^2$ est alors
impossible. Dans tout \omterm{pb:g8:pythagoras:1}{triplet pythagoricien}, au moins l'un des trois
nombres est donc \omterm{def:g3:numbers:evenodd}{pair} --- comme le confirme chaque ligne du tableau
de la question 7.
\end{solution}
