\begin{solution}{exo:g8:speed:1}

$45$ min $= 0.75$ h~: $\frac{27}{0.75} = 36$ km/h. \quad
$\frac{100}{12.5} = 8$ m/s.
\end{solution}

\end{solution}

\begin{solution}{exo:g8:speed:8}
Distances~: $2 \times 5 = 10$ km, puis $1.5 \times 4 = 6$ km~: total
$16$ km. Temps écoulé~: $2 + 0.5 + 1.5 = 4$ h. \omterm{def:g8:speed:def}{Vitesse moyenne}~:
$\frac{16}{4} = 4$ km/h.
\end{solution}

\begin{solution}{exo:g8:speed:9}
Pondérée~:
$\frac{3 \times 15 + 2 \times 9 + 1 \times 12}{3 + 2 + 1}
= \frac{45 + 18 + 12}{6} = \frac{75}{6} = 12.5$.
Simple~: $\frac{15 + 9 + 12}{3} = 12$. Elles diffèrent car les
coefficients donnent à la note $15$ trois fois le poids de la note
$12$.
\end{solution}

\begin{solution}{exo:g8:speed:10}
\emph{1.} Temps~: $\frac{120}{80} = 1.5$ h et $\frac{120}{120} = 1$
h. \omterm{def:g8:speed:def}{Vitesse moyenne}~: $\frac{240}{2.5} = 96$ km/h.

\emph{2.} La voiture passe \emph{plus de temps} à $80$ km/h ($1.5$ h)
qu'à $120$ km/h ($1$ h)~: la \omterm{def:g8:speed:def}{vitesse} lente pèse plus dans la \omterm{def:g8:speed:average}{moyenne
pondérée} par le temps, la tirant sous le milieu $100$.
\end{solution}

\begin{solution}{exo:g8:speed:11}
L'écart de $18$ km se referme à $5 + 4 = 9$ km/h~: ils se rencontrent
après $\frac{18}{9} = 2$ h, à $12{:}00$. Anna a alors marché
$2 \times 5 = 10$ km~: ils se rencontrent à $10$ km du village
d'Anna (et $8$ km de celui de Boris --- vérification~: $10 + 8 =
18$).
\end{solution}

\begin{solution}{pb:g8:speed:1}
\textbf{1.} D'après \cref{def:g8:speed:def}, $t_1 = \frac{d}{v_1}$
et $t_2 = \frac{d}{v_2}$~; le trajet entier dure
$\frac{d}{v_1} + \frac{d}{v_2}$ pour une distance de $2d$.

\textbf{2.} Factorisons $d$ au \omterm{def:g4:fractions:def}{dénominateur} et simplifions-le~:
\[
\bar v
= \frac{2d}{d\left(\frac{1}{v_1} + \frac{1}{v_2}\right)}
= \frac{2}{\ \frac{1}{v_1} + \frac{1}{v_2}\ } .
\]
\omterm{def:g4:fractions:def}{Dénominateur} commun à l'intérieur~:
$\frac{1}{v_1} + \frac{1}{v_2} = \frac{v_2 + v_1}{v_1 v_2}$, et
diviser revient à multiplier par l'\omterm{def:g8:fractions:inverse}{inverse}
(\cref{thm:g8:fractions:div})~:
\[
\bar v = 2 \times \frac{v_1 v_2}{v_1 + v_2}
= \frac{2\, v_1 v_2}{v_1 + v_2} .
\]

\textbf{3.} $\dfrac{2 \times 30 \times 15}{30 + 15}
= \dfrac{900}{45} = 20$ km/h, comme dans
\cref{ex:g8:speed:twolegs}~; et
$\dfrac{2 \times 80 \times 120}{80 + 120} = \dfrac{19\,200}{200}
= 96$ km/h, comme trouvé dans \cref{exo:g8:speed:10}.

\textbf{4.} La moyenne ne dépend que des deux \omterm{def:g8:speed:def}{vitesses}, et pas de
$d$~: un aller-retour de $1$ km et un aller-retour de $100$ km aux
deux mêmes \omterm{def:g8:speed:def}{vitesses} ont exactement la même \omterm{def:g8:speed:def}{vitesse moyenne}. (C'est
pourquoi la formule a pu être contrôlée sur des trajets de longueurs
différentes.)

\textbf{5.} En roulant la durée $T$ à chaque \omterm{def:g8:speed:def}{vitesse}, les distances
valent $v_1 T$ et $v_2 T$, donc
\[
\bar v = \frac{v_1 T + v_2 T}{2T}
= \frac{(v_1 + v_2)\,T}{2T}
= \frac{v_1 + v_2}{2} .
\]
Une moyenne de \omterm{def:g8:speed:def}{vitesses} est toujours pondérée par le \emph{temps}
(\cref{def:g8:speed:average}). À durées égales, les deux \omterm{def:g8:speed:def}{vitesses}
ont le même poids~: c'est le simple milieu. À distances égales, la
\omterm{def:g8:speed:def}{vitesse} la plus lente occupe \emph{plus} de temps, donc davantage de
poids --- et la moyenne glisse vers elle.

\textbf{6.} Pour $(30, 15)$~: $H = 20$ et $A = 22.5$. Pour
$(80, 120)$~: $H = 96$ et $A = 100$. La moyenne arithmétique
l'emporte les deux fois.

\textbf{7.} Au \omterm{def:g4:fractions:def}{dénominateur} commun $2(v_1 + v_2)$~:
\[
A - H
= \frac{(v_1 + v_2)^2 - 4\, v_1 v_2}{2\,(v_1 + v_2)} ,
\]
et le \omterm{def:g4:fractions:def}{numérateur} se développe avec les identités du
\cref{pb:g8:equations:1}~:
\[
(v_1 + v_2)^2 - 4 v_1 v_2
= v_1^2 + 2 v_1 v_2 + v_2^2 - 4 v_1 v_2
= v_1^2 - 2 v_1 v_2 + v_2^2
= (v_1 - v_2)^2 .
\]

\textbf{8.} Un carré n'est jamais négatif, et $2(v_1 + v_2)$ est
positif~: donc $A - H \geq 0$, c'est-à-dire $H \leq A$, pour tout
couple de \omterm{def:g8:speed:def}{vitesses} positives. L'égalité exige
$(v_1 - v_2)^2 = 0$, c'est-à-dire $v_1 = v_2$~: dès que les deux
\omterm{def:g8:speed:def}{vitesses} diffèrent, la moyenne de l'aller-retour tombe strictement
en dessous du milieu --- la surprise est un théorème.

\textbf{9.} Multiplions~:
\[
H \times A
= \frac{2\, v_1 v_2}{v_1 + v_2} \times \frac{v_1 + v_2}{2}
= v_1 v_2 ,
\]
après simplification du facteur $2$ et du facteur $v_1 + v_2$.

\textbf{10.} Par la formule~: $\bar v = \dfrac{2 \times 40 \times
60}{40 + 60} = \dfrac{4800}{100} = 48$ km/h. Par le chemin long~:
$\frac{60}{40} = 1.5$ h et $\frac{60}{60} = 1$ h, soit $120$ km en
$2.5$ h~: $\frac{120}{2.5} = 48$ km/h. Les deux concordent.

\textbf{11.} Réaliser une moyenne de $30$ km/h sur $30$ km demande
une durée totale d'au plus $\frac{30}{30} = 1$ h. Or les $15$
premiers kilomètres à $15$ km/h prennent déjà
$\frac{15}{15} = 1$ h~: le budget-temps \emph{entier} est dépensé,
avec la \omterm{def:g3:division:half}{moitié} du parcours restant à faire.

\textbf{12.} À $90$ km/h, la seconde \omterm{def:g3:division:half}{moitié} prend
$\frac{15}{90} = \frac16$ h, donc la course dure $\frac76$ h et
\[
\bar v = \frac{30}{\ 7/6\ } = \frac{180}{7} \approx 25.7
\text{ km/h} < 30 .
\]
Quelle que soit la \omterm{def:g8:speed:def}{vitesse} de la seconde \omterm{def:g3:division:half}{moitié}, sa durée est un
nombre positif~: la durée totale dépasse donc $1$ h et la moyenne
reste strictement inférieure à $30$ km/h. (Par la formule~:
$\frac{2 \times 15 \times v_2}{15 + v_2} = 30$ imposerait
$30 v_2 = 450 + 30 v_2$, c'est-à-dire $0 = 450$ --- aucune \omterm{def:g8:speed:def}{vitesse}
$v_2$ ne convient.)

\textbf{13.} Une moyenne de $20$ km/h autorise
$\frac{30}{20} = 1.5$ h~; après la première heure, il reste $15$ km
à faire en $0.5$ h~: il lui faut $\frac{15}{0.5} = 30$ km/h.
Vérification~:
$H(15, 30) = \frac{2 \times 15 \times 30}{45} = \frac{900}{45}
= 20$ km/h.

\textbf{14.} Par minute, les robinets remplissent $\frac{1}{20}$ et
$\frac{1}{30}$ de la baignoire~; ensemble
\[
\frac{1}{20} + \frac{1}{30}
= \frac{3}{60} + \frac{2}{60}
= \frac{5}{60} = \frac{1}{12}
\]
de la baignoire par minute~: la baignoire se remplit en $12$ min.
Et $H(20, 30) = \frac{2 \times 600}{50} = 24 = 2 \times 12$~: les
deux robinets réunis mettent exactement la \emph{\omterm{def:g3:division:half}{moitié} de la
\omterm{pb:g8:speed:1}{moyenne harmonique}} de leurs durées individuelles --- les débits
s'ajoutent, on est donc encore en terrain de \omterm{pb:g8:speed:1}{moyenne harmonique}.

\textbf{15.} Même distance à l'aller et au retour, donc
\[
\bar v = \frac{2 \times 900 \times 700}{900 + 700}
= \frac{1\,260\,000}{1\,600} = 787.5 \text{ km/h},
\]
en dessous des $800$ km/h par air calme. En une phrase~: l'avion
passe \emph{plus} de temps à voler contre le vent qu'avec lui, donc
l'étape lente pèse davantage dans la \omterm{def:g8:speed:average}{moyenne pondérée} par le temps
--- un vent ne peut qu'allonger un aller-retour.
\end{solution}
