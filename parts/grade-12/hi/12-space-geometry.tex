\chapter{अंतरिक्ष में सदिश, रेखाएँ और समतल}\label{ch:g12:space}

\omterm{def:g11:vect:vector}{सदिश} और \omterm{def:g10:coordgeom:system}{निर्देशांक} उपलब्ध हो जाने पर त्रिविमीय ज्यामिति गणनीय हो जाती है:
रेखाओं को प्राचलिक निरूपण मिल जाते हैं, समतलों को कार्तीय \omterm{def:g10:algebra:equation}{समीकरण}, और \omterm{def:g12:space:dot}{अदिश
गुणनफल} कोण तथा दूरियाँ नाप लेता है। यह अध्याय यही औज़ार-बक्सा बनाता है और
उससे प्रतिच्छेद तथा दूरी की चिरपरिचित समस्याएँ हल करता है।

\section{अंतरिक्ष में सदिश}

अंतरिक्ष के \omterm{def:g11:vect:vector}{सदिश} वही नियम मानते हैं जो समतल में थे: वे जुड़ते हैं, गुणित होते
हैं, और $\vect{AB} = \vect{CD}$ ठीक तब है जब $ABDC$ \omterm{def:g12:seq:arith-geom}{समांतर} चतुर्भुज हो। \omterm{def:g10:coordgeom:system}{निर्देशांक-पद्धति} $(O; \vec\imath, \vec\jmath, \vec k)$ में हर
\omterm{def:g11:vect:vector}{सदिश} के \omterm{def:g10:coordgeom:system}{निर्देशांक} $(x, y, z)$ होते हैं।

\begin{definition}[संरेखता, समतलीयता]\label{def:g12:space:collinear}
दो \omterm{def:g11:vect:vector}{सदिश} \emph{संरेखी} हैं यदि एक दूसरे का \omterm{def:g12:arith:divides}{गुणज} हो। तीन \omterm{def:g11:vect:vector}{सदिश} $\vec u, \vec v, \vec w$
\emph{समतलीय}\index{समतलीय} हैं यदि उनमें से किसी एक को शेष दोनों के संयोजन
के रूप में लिखा जा सके, जैसे $\vec w = a\vec u + b\vec v$।
\end{definition}

\begin{proposition}\label{prop:g12:space:basis}
यदि $\vec u, \vec v, \vec w$ \omterm{def:g12:space:collinear}{समतलीय} न हों, तो अंतरिक्ष का हर \omterm{def:g11:vect:vector}{सदिश} $x\vec u + y\vec v + z\vec w$ के रूप में अद्वितीय ढंग से
बँट जाता है।
\end{proposition}

\begin{proof}[उपपत्ति का विचार]
जिस \omterm{def:g11:vect:vector}{सदिश} को बाँटना है उसके सिरे से $\vec w$ के \omterm{def:g12:seq:arith-geom}{समांतर} रेखा खींचिए; वह $\vec u, \vec v$ के
समतल से एक ही बिंदु पर मिलती है, जो \omterm{def:g11:vect:vector}{सदिश} को उस समतल में पड़ने वाले घटक
(समतल-ज्यामिति से अद्वितीय रूप से $x\vec u + y\vec
v$) और $\vec w$ के अनुदिश एक घटक में बाँट
देता है। अद्वितीयता: दो विभाजनों का अंतर किसी एक \omterm{def:g11:vect:vector}{सदिश} को शेष दो के पदों में
व्यक्त कर देता, जो असमतलीयता का खंडन है।
\end{proof}

\section{रेखाएँ और समतल}

\begin{definition}[रेखा का प्राचलिक निरूपण]\label{def:g12:space:line}
$A(x_A, y_A, z_A)$ से होकर \omterm{def:g11:vect:direction}{दिक् सदिश} $\vec u (a, b, c) \neq \vec 0$ वाली रेखा बिंदुओं का यह समुच्चय है
\[
M(x_A + ta,\; y_A + tb,\; z_A + tc), \qquad t \in \R .
\]
\end{definition}

\begin{definition}[समतल]\label{def:g12:space:plane}
$A$ से होकर दो असंरेखी \omterm{def:g10:vectors:vector}{सदिशों} $\vec u, \vec v$ द्वारा दिशित समतल उन बिंदुओं $M$ का
समुच्चय है जिनके लिए $\vect{AM} = s\vec u + t\vec v$, $(s, t) \in \R^2$ हों।
\end{definition}

\begin{proposition}[सापेक्ष स्थितियाँ]\label{prop:g12:space:positions}
दो भिन्न समतल या तो \omterm{def:g12:seq:arith-geom}{समांतर} होते हैं या किसी रेखा पर मिलते हैं। कोई रेखा और
समतल या तो \omterm{def:g12:seq:arith-geom}{समांतर} होते हैं (और रेखा समतल में समाई भी हो सकती है) या एक ही
बिंदु पर मिलते हैं। अंतरिक्ष की दो रेखाएँ प्रतिच्छेदी, कड़ाई से \omterm{def:g12:seq:arith-geom}{समांतर}, समरूप,
या \emph{विषमतलीय} (असमतलीय) हो सकती हैं।
\end{proposition}

\begin{proof}[रूपरेखा]
ये आपतन के कथन हैं; \omterm{def:g10:coordgeom:system}{निर्देशांकों} में हर स्थिति-विश्लेषण किसी \omterm{def:g10:lines:system}{रैखिक निकाय} को
हल करके उसके हल गिनने पर सिमट आता है --- \cref{met:g12:space:intersections} देखिए।
\end{proof}

\section{अदिश गुणनफल}

\begin{definition}[अदिश गुणनफल]\label{def:g12:space:dot}
$\vec u$ और $\vec v$ का \emph{अदिश गुणनफल}\index{अदिश गुणनफल} है
\[
\vec u \cdot \vec v = \tfrac12\left(\norm{\vec u + \vec v}^2
- \norm{\vec u}^2 - \norm{\vec v}^2\right),
\]
जहाँ $\norm{\vec u}$ $\vec u$ की लंबाई है। यदि दोनों \omterm{def:g11:vect:vector}{सदिश} शून्येतर हों, तो $\vec u \cdot \vec v = \norm{\vec u}\,\norm{\vec v}\cos\theta$, जहाँ $\theta$
उनके बीच का कोण है; और किसी \emph{\omterm{def:g10:coordgeom:system}{लंबकोणिक}} \omterm{def:g10:coordgeom:system}{निर्देशांक-पद्धति} में
\[
\vec u \cdot \vec v = xx' + yy' + zz' .
\]
दो \omterm{def:g11:vect:vector}{सदिश} \emph{\omterm{def:g11:scal:orthogonal}{लांबिक}} हैं यदि उनका अदिश गुणनफल $0$ हो।
\end{definition}

\begin{proposition}[गुण]\label{prop:g12:space:dotprops}
\omterm{def:g12:space:dot}{अदिश गुणनफल} सममित ($\vec u\cdot\vec v = \vec v\cdot\vec u$), द्विरैखिक ($(a\vec u + b\vec v)\cdot \vec w = a\,\vec u\cdot\vec w +
b\,\vec v\cdot \vec w$) है, और $\vec u \cdot \vec u = \norm{\vec u}^2 \geq 0$।
\end{proposition}

\begin{proof}
\omterm{def:g10:coordgeom:system}{लंबकोणिक पद्धति} में तीनों गुण सूत्र $xx' + yy' + zz'$ पर तत्काल दिख जाते हैं, और वह सूत्र
स्वयं परिभाषा तथा लंबाइयों की पाइथागोरस गणना से निकलता है: $\norm{\vec u}^2 = x^2 + y^2 + z^2$।
\end{proof}

\begin{definition}[अभिलंब सदिश]\label{def:g12:space:normal}
किसी समतल $\mathcal P$ का \emph{अभिलंब सदिश}\index{अभिलंब सदिश} वह शून्येतर \omterm{def:g11:vect:vector}{सदिश} है
जो $\mathcal P$ में पड़े हर \omterm{def:g11:vect:vector}{सदिश} के \omterm{def:g11:scal:orthogonal}{लांबिक} हो (इतना पर्याप्त है कि वह $\mathcal P$ की दो
असंरेखी दिशाओं के \omterm{def:g11:scal:orthogonal}{लांबिक} हो)।
\end{definition}

\begin{theorem}[समतल का कार्तीय समीकरण]\label{thm:g12:space:planeeq}
किसी \omterm{def:g10:coordgeom:system}{लंबकोणिक} \omterm{def:g10:coordgeom:system}{निर्देशांक-पद्धति} में $A$ से होकर \omterm{def:g12:space:normal}{अभिलंब सदिश} $\vec n(a, b, c)$ वाले समतल
का \omterm{def:g10:algebra:equation}{समीकरण}
\[
a x + b y + c z + d = 0,
\]
है, जहाँ $d = -(ax_A + by_A + cz_A)$ है। इसके उलट, $(a,b,c) \neq (0,0,0)$ वाला ऐसा हर \omterm{def:g10:algebra:equation}{समीकरण} \omterm{def:g12:space:normal}{अभिलंब सदिश} $(a, b, c)$ वाला
एक समतल बताता है।
\end{theorem}

\begin{proof}
$M \in \mathcal P \iff \vect{AM} \perp \vec n \iff
a(x - x_A) + b(y - y_A) + c(z - z_A) = 0$, जिसका प्रसार वही \omterm{def:g10:algebra:equation}{समीकरण} है। इसके उलट, \omterm{def:g10:algebra:equation}{समीकरण} का कोई भी हल-बिंदु $A$
लीजिए; वही गणना उल्टी चलाने पर दिखता है कि हल-समुच्चय $\{M : \vect{AM}\cdot\vec n = 0\}$ है, यानी $A$ से
होकर $\vec n$ के \omterm{def:g11:scal:orthogonal}{लांबिक} समतल।
\end{proof}

\begin{method}[व्यवहार में प्रतिच्छेद]\label{met:g12:space:intersections}
\begin{itemize}
  \item \emph{रेखा $\cap$ समतल}: रेखा के प्राचलिक \omterm{def:g10:coordgeom:system}{निर्देशांक} समतल के \omterm{def:g10:algebra:equation}{समीकरण}
        में रखिए; $t$ के लिए हल कीजिए (एक हल: एक बिंदु; $0 = c$ (शून्येतर $c$):
        \omterm{def:g12:seq:arith-geom}{समांतर}; $0 = 0$: रेखा समतल में समाई हुई)।
  \item \emph{समतल $\cap$ समतल}: दोनों \omterm{def:g10:algebra:equation}{समीकरणों} का निकाय हल कीजिए और किसी एक
        मुक्त \omterm{def:g10:coordgeom:system}{निर्देशांक} से प्राचलन कीजिए; हल एक रेखा है (या \omterm{def:g12:space:normal}{अभिलंब सदिश}
        \omterm{def:g12:space:collinear}{संरेखी} होने पर समतल \omterm{def:g12:seq:arith-geom}{समांतर} हैं)।
  \item \emph{लंबता की जाँच}: रेखा $\perp$ समतल तभी जब उसकी दिशा अभिलंब के
        \omterm{def:g12:space:collinear}{संरेखी} हो; और दो समतल परस्पर लंब तभी जब उनके अभिलंब \omterm{def:g11:scal:orthogonal}{लांबिक} हों।
\end{itemize}
\end{method}

\begin{proposition}[बिंदु से समतल तक की दूरी]\label{prop:g12:space:distance}
\index{दूरी!बिंदु से समतल तक}
किसी \omterm{def:g10:coordgeom:system}{लंबकोणिक पद्धति} में $M_0(x_0, y_0, z_0)$ की समतल $\mathcal P : ax + by + cz + d = 0$ से दूरी है
\[
\operatorname{dist}(M_0, \mathcal P)
= \frac{\abs{ax_0 + by_0 + cz_0 + d}}{\sqrt{a^2 + b^2 + c^2}}.
\]
\end{proposition}

\begin{proof}
मान लीजिए $H$ $\mathcal P$ पर $M_0$ का \omterm{def:g11:scal:orthogonal}{लांबिक} प्रक्षेप है: लंब का पाद, इसलिए $\vect{HM_0}$
इकाई अभिलंब $\frac{\vec n}{\norm{\vec n}}$ का \omterm{def:g12:space:collinear}{संरेखी} है, और दूरी $\abs{\vect{HM_0}\cdot \frac{\vec n}{\norm{\vec n}}}$ है। किसी भी $H \in \mathcal P$ के लिए $\vect{HM_0}\cdot\vec n = ax_0 + by_0 + cz_0 - (ax_H + by_H + cz_H)
= ax_0 + by_0 + cz_0 + d$
($H$ पर \omterm{def:g10:algebra:equation}{समीकरण} का उपयोग करके), जिससे $\norm{\vec n} = \sqrt{a^2+b^2+c^2}$ से भाग देने पर सूत्र निकल आता है।
\end{proof}

\begin{example}\label{ex:g12:space:distance}
मूल बिंदु की समतल $x + 2y + 2z - 6 = 0$ से दूरी: $\frac{\abs{-6}}{\sqrt{1 + 4 + 4}} = \frac63 = 2$।
\end{example}

\section{अभ्यास}

सभी अभ्यासों में \omterm{def:g10:coordgeom:system}{निर्देशांक-पद्धति} \omterm{def:g10:coordgeom:system}{लंबकोणिक} है।

\begin{exercise}[$\star$]\label{exo:g12:space:1}
मान लीजिए $A(1, 0, 2)$, $B(3, 1, 1)$ और $C(2, -1, 3)$ हैं। क्या बिंदु $A$, $B$, $C$ संरेख
हैं? रेखा $(AB)$ का प्राचलिक निरूपण दीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:space:2}
$A(1, 1, 0)$ से होकर \omterm{def:g12:space:normal}{अभिलंब सदिश} $\vec n(2, -1, 3)$ वाले समतल का कार्तीय \omterm{def:g10:algebra:equation}{समीकरण} ज्ञात कीजिए। क्या
बिंदु $B(0, 2, 1)$ उसका है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:space:3}
शीर्षों $A(0,0,0)$, $B(1, 1, 0)$ और $C(1, 0, 1)$ वाले त्रिभुज का $A$ पर कोण (निकटतम अंश तक)
निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:space:4}
$A(1, 2, 0)$ से होकर दिशा $\vec u(1, -1, 2)$ वाली रेखा $\mathcal D$, और समतल $\mathcal P : 2x + y - z + 1 = 0$ पर विचार कीजिए।
$\mathcal D \cap \mathcal P$ ज्ञात कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:space:5}
मान लीजिए $\mathcal P : x - y + 2z = 1$ और $\mathcal Q : 2x + y + z = 4$ हैं। दिखाइए कि $\mathcal P$ और $\mathcal Q$ किसी रेखा पर मिलते
हैं और उसका प्राचलिक निरूपण दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:space:6}
दिखाइए कि रेखाएँ
\[
\mathcal D_1 : (x, y, z) = (1 + t,\; 2t,\; 3 - t), \qquad
\mathcal D_2 : (x, y, z) = (2 + s,\; 1 + s,\; 1 + 2s)
\]
विषमतलीय हैं (\omterm{def:g12:space:collinear}{समतलीय} नहीं)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:space:7}
मान लीजिए $\mathcal P : 2x - y + 2z - 3 = 0$ है।
\begin{enumerate}
  \item $M_0(3, 1, 1)$ की $\mathcal P$ से दूरी निकालिए।
  \item $\mathcal P$ पर $M_0$ के \omterm{def:g11:scal:orthogonal}{लांबिक} प्रक्षेप $H$ के \omterm{def:g10:coordgeom:system}{निर्देशांक} ज्ञात कीजिए,
        और दूरी $M_0H$ जाँचिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:space:8}
घन $ABCDEFGH$ की भुजा $1$ है; \omterm{def:g10:coordgeom:system}{निर्देशांक} ऐसे रखिए कि $A(0,0,0)$, $B(1,0,0)$, $D(0,1,0)$, $E(0,0,1)$
हों ($A$ से चलने वाले \omterm{def:g10:vectors:vector}{सदिशों} $C = B + D$, $F = B + E$, $G = B + D + E$, $H = D + E$ के साथ)।
\begin{enumerate}
  \item दिखाइए कि विकर्ण $(AG)$ समतल $(BDE)$ के \omterm{def:g11:scal:orthogonal}{लांबिक} है।
  \item $A$ की समतल $(BDE)$ से दूरी और वह बिंदु निकालिए जहाँ $(AG)$ उसे भेदता
        है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:space:9}
\emph{(चतुष्फलक का आयतन.)} मान लीजिए $A(1,1,1)$, $B(2,1,0)$, $C(0,2,1)$ और $D(2,2,2)$ हैं।
\begin{enumerate}
  \item जाँचिए कि $\vect{AB}$, $\vect{AC}$, $\vect{AD}$ \omterm{def:g12:space:collinear}{समतलीय} नहीं हैं (बिंदु सचमुच का
        चतुष्फलक बनाते हैं)।
  \item ऐसा \omterm{def:g11:vect:vector}{सदिश} $\vec n$ ज्ञात कीजिए जो $\vect{BC}$ और $\vect{BD}$ दोनों के \omterm{def:g11:scal:orthogonal}{लांबिक} हो, और
        समतल $(BCD)$ का कार्तीय \omterm{def:g10:algebra:equation}{समीकरण} निकालिए।
  \item $A$ की समतल $(BCD)$ से दूरी और त्रिभुज $BCD$ का क्षेत्रफल निकालिए,
        और इसके लिए $\vec u, \vec v$ द्वारा रचे गए त्रिभुज के लिए सूत्र $\text{क्षेत्रफल} = \frac12\sqrt{\norm{\vec u}^2\norm{\vec v}^2 -
        (\vec u \cdot \vec v)^2}$ काम में
        लाइए।
  \item चतुष्फलक का आयतन निकालिए ($V = \frac13 \times \text{आधार का क्षेत्रफल} \times \text{ऊँचाई}$)।
\end{enumerate}
\end{exercise}

\section{समस्या: घन को काटना}

\begin{problem}[{सप्ताहांत समस्या --- एक समतल घन को काटकर पूर्ण षट्भुज बना
देता है, कोनों में एक सम चतुष्फलक छिपा है, और विकर्ण का कोण हर हीरे की आकृति
गढ़ता है}]
\label{pb:g12:space:1}
घन को काटिए तो आप वर्ग और आयत की अपेक्षा करते हैं --- फिर भी एक प्रसिद्ध कटान
\emph{सम षट्भुज} बना देता है, और कोनों के चार चतुर कटान पीछे एक \emph{सम
चतुष्फलक} छोड़ जाते हैं। ये दोनों अचरज, और वह $109.5^\circ$ का कोण जिसकी पूजा कार्बन
परमाणु करते हैं, इसी अध्याय के औज़ार-बक्से के आगे झुक जाते हैं: प्राचलिक
रेखाएँ, समतल-समीकरण और \omterm{def:g12:space:dot}{अदिश गुणनफल} (\cref{thm:g12:space:planeeq}, \cref{prop:g12:space:distance})। पूरे समय \cref{exo:g12:space:8} का घन काम
में लाइए: भुजा $1$, $A(0,0,0)$, $B(1,0,0)$, $D(0,1,0)$, $E(0,0,1)$, $C(1,1,0)$, $F(1,0,1)$, $H(0,1,1)$, $G(1,1,1)$।

\medskip
\noindent\textbf{भाग I --- प्रवाह।}
\begin{enumerate}
  \item $B(1,0,0)$ से होकर \omterm{def:g12:space:normal}{अभिलंब सदिश} $(1,1,1)$ वाले समतल का \omterm{def:g10:algebra:equation}{समीकरण}, $A$ से होकर
        $(1,1,1)$ द्वारा दिशित रेखा का प्राचलिक निरूपण, और उनका प्रतिच्छेद बिंदु
        लिखिए।
  \item $A$ की उस समतल से दूरी निकालिए (\cref{prop:g12:space:distance})।
  \item क्या \omterm{def:g11:vect:vector}{सदिश} $(1,-1,0)$ और $(1,1,-2)$ \omterm{def:g11:scal:orthogonal}{लांबिक} हैं?
  \item समतलों $x + y + z = 1$ और $2x + 2y + 2z = 5$ की स्थिति क्या है? और $A$ से होकर $(1,1,0)$
        द्वारा दिशित रेखा की पहले समतल के सापेक्ष स्थिति?
  \item घन की छह कोरों $[BC], [CD], [DH], [HE], [EF], [FB]$ के \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} गिनाइए, और जाँचिए कि छहों \omterm{def:g10:algebra:equation}{समीकरण}
        $x + y + z = \frac32$ को संतुष्ट करते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- षट्भुजीय काट।}
मान लीजिए $P$ समतल $x + y + z = \frac32$ है।
\begin{enumerate}[resume]
  \item दिखाइए कि $P$ बड़े विकर्ण $(AG)$ पर लंब है और घन के केंद्र से होकर
        जाता है।
  \item काट के चारों ओर प्रश्न 5 के क्रमागत मध्यबिंदुओं के बीच की दूरियाँ
        निकालिए: आपको क्या मिलता है?
  \item काट के बहुभुज का किसी एक शीर्ष पर कोण निकालिए (दो क्रमागत कोर-सदिश
        और एक बिंदु-गुणनफल)। निष्कर्ष निकालिए: काट एक \emph{सम षट्भुज} है।
  \item षट्भुज का परिमाप और क्षेत्रफल निकालिए, और उसके क्षेत्रफल की तुलना
        घन के किसी फलक से कीजिए। (जानकारी के लिए: इकाई घन की सबसे बड़ी समतल
        काट $1 \times \sqrt2$ वाला विकर्ण आयत है, जिसका क्षेत्रफल $\sqrt2$ है --- सम षट्भुज
        को रजत पदक मिलता है।)
  \item विकर्ण $(AG)$ $P$ को कहाँ भेदता है? जाँचिए कि भेदन-बिंदु $[AG]$ का
        \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} है।
  \item कटान को सरकाइए: $c$ के $0$ से $3$ तक बढ़ने पर काटों $x + y + z = c$ का
        वर्णन कीजिए --- $c = \frac12$ के लिए काट निकालिए (कौन-सा बहुभुज, किस आकार
        का?), और उस रूपांतर की कहानी कहिए जो $c = \frac32$ पर आपके षट्भुज से होकर
        गुज़रती है।
  \item कोई समतल घन को \emph{सात} भुजाओं वाले बहुभुज में क्यों नहीं काट सकता?
        (काट की हर भुजा को कहाँ पड़ना ही चाहिए?)
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- छिपा हुआ चतुष्फलक।}
\begin{enumerate}[resume]
  \item कोने वाले चतुष्फलक $ABDE$ का आयतन निकालिए (आधार, ऊँचाई, और वह
        एक-तिहाई वाला सूत्र जिसे माध्यमिक विद्यालय खंड ने स्वीकार किया था
        --- \cref{exo:g12:space:9} ऐसे आयतन सामान्य रूप से निकाल देता है)।
  \item चारों शीर्ष $B$, $D$, $E$, $G$: जोड़े-जोड़े छहों दूरियाँ
        निकालिए और निष्कर्ष निकालिए कि $BDEG$ \emph{सम} चतुष्फलक है। फिर घन
        में से कोने वाले चतुष्फलक घटाकर उसका आयतन निकालिए।
  \item घन के केंद्र की समतल $(BDE)$ से दूरी निकालिए।
  \item हीरे का कोण: घन के विकर्णों $\vect{AG}$ और $\vect{BH}$ के बीच का कोण निकालिए।
        उसका संपूरक, लगभग $109.5^\circ$, हीरे और मेथेन में बंधों के बीच का कोण है
        --- किसी कार्बन परमाणु के चारों ओर के चार इलेक्ट्रॉन-युग्म प्रश्न
        14 के चतुष्फलक के कोने ही क्यों चुनते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- सचमुच त्रिविमीय।}
\begin{enumerate}[resume]
  \item $B(1,0,0)$ से होकर $(0,1,1)$ द्वारा दिशित रेखा षट्भुज के समतल $P$ को कहाँ
        भेदती है, यह ज्ञात कीजिए।
  \item दिखाइए कि रेखाएँ $(AB)$ और $(EG)$ न मिलती हैं न \omterm{def:g12:seq:arith-geom}{समांतर} हैं:
        \emph{विषमतलीय} रेखाएँ, यानी वह परिघटना जो किसी समतल में हो ही नहीं
        सकती।
  \item $P$ पर $B$ का \omterm{def:g11:scal:orthogonal}{लांबिक} प्रक्षेप निकालिए ($B$ से अभिलंब के अनुदिश
        समतल तक चलिए), और जाँचिए कि आपका बिंदु $P$ के \omterm{def:g10:algebra:equation}{समीकरण} को संतुष्ट
        करता है।
  \item समापन --- अंतरिक्ष के भूमापक का बक्सा: भेदन के लिए प्राचलिक रेखाएँ,
        कोणों और दूरियों के लिए \omterm{def:g12:space:normal}{अभिलंब सदिश}, काटों के लिए समतल-समीकरण; और
        घन वह खेल का मैदान जहाँ उन्होंने एक षट्भुज, एक सम चतुष्फलक, हीरे का
        कोण और विषमतलीय रेखाओं की एक जोड़ी पैदा कर दी। हर एक पर एक वाक्य।
\end{enumerate}
\end{problem}
