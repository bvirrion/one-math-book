\chapter{यादृच्छिक चर और द्विपद बंटन}\label{ch:g12:randvar}

\omterm{def:g12:randvar:rv}{यादृच्छिक चर} किसी प्रयोग के हर परिणाम के साथ एक संख्या जोड़ देता है: कोई लाभ,
कोई गिनती, कोई अवधि। उसकी \omterm{def:g12:randvar:exp}{प्रत्याशा} उन मानों का दीर्घकालिक औसत है और उसका
\omterm{def:g12:randvar:exp}{प्रसरण} उनका प्रसार नापता है। इस अध्याय का सितारा \omterm{def:g12:randvar:binomial}{द्विपद बंटन} है, जो दोहराए गए
\omterm{def:g12:condprob:indep}{स्वतंत्र} परीक्षणों में सफलताएँ गिनता है। \omterm{def:g12:randvar:rv}{यादृच्छिक चर} और \omterm{def:g12:randvar:binomial}{द्विपद बंटन} से पहली
भेंट \cref{ch:g11:prob,ch:g11:binom} में हुई थी; यह अध्याय उन्हें दोहराता और गहरा करता है, और अब \cref{ch:g12:comb} के
क्रमचय-संचय वाले औज़ार भी उपलब्ध हैं।

\section{विविक्त यादृच्छिक चर}

\begin{definition}[यादृच्छिक चर, बंटन]\label{def:g12:randvar:rv}
किसी परिमित \omterm{def:g11:prob:model}{प्रतिदर्श समष्टि} $\Omega$ पर \emph{यादृच्छिक चर}\index{यादृच्छिक चर}
एक \omterm{def:g11:func:function}{फलन} $X \colon \Omega \to \R$ है। उसका \emph{बंटन}\index{बंटन} (या नियम) उसके संभव मानों $x_1, \dots, x_k$ और
\omterm{def:g10:proba:distribution}{प्रायिकताओं}
\[
p_i = \P(X = x_i), \qquad \sum_{i=1}^{k} p_i = 1 .
\]
का ब्योरा है।
\end{definition}

\begin{definition}[प्रत्याशा, प्रसरण, मानक विचलन]\label{def:g12:randvar:exp}
$X$ की \emph{प्रत्याशा}\index{प्रत्याशा} है
\[
\E(X) = \sum_{i=1}^{k} p_i\, x_i ,
\]
और उसका \emph{प्रसरण}\index{प्रसरण} तथा \emph{\omterm{def:g11:stat:variance}{मानक विचलन}} हैं
\[
\V(X) = \E\bigl((X - \E(X))^2\bigr) = \sum_{i=1}^k p_i\bigl(x_i - \E(X)\bigr)^2,
\qquad
\sigma(X) = \sqrt{\V(X)} .
\]
\end{definition}

\begin{proposition}[कोनिग--होयगेंस सूत्र]\label{prop:g12:randvar:konig}
\index{कोनिग--होयगेंस सूत्र}
$\V(X) = \E(X^2) - \E(X)^2$।
\end{proposition}

\begin{proof}
$m = \E(X)$ लिखिए और प्रसार कीजिए:
\[
\V(X) = \sum_i p_i (x_i^2 - 2m x_i + m^2)
= \E(X^2) - 2m\sum_i p_i x_i + m^2\sum_i p_i
= \E(X^2) - 2m^2 + m^2 . \qedhere
\]
\end{proof}

\begin{proposition}[एकघात रूपांतरण]\label{prop:g12:randvar:affine}
$a, b \in \R$ के लिए:
\[
\E(aX + b) = a\,\E(X) + b, \qquad
\V(aX + b) = a^2\,\V(X) .
\]
\end{proposition}

\begin{proof}
पहला परिभाषा वाले योग की ही पुनर्व्यवस्था है। दूसरे के लिए: $aX + b$ अपने \omterm{def:g11:stat:mean}{माध्य}
से $a(X - \E(X))$ जितना हटता है, और वर्ग करने पर वह $a^2$ गुना हो जाता है।
\end{proof}

\begin{example}[न्यायसंगत खेल]\label{ex:g12:randvar:game}
किसी खेल का शुल्क $2$ यूरो है; एक पासा फेंका जाता है, और यदि आया मान कम से
कम $5$ हो तो खिलाड़ी वही मान पाता है, अन्यथा कुछ नहीं। लाभ $G$ मान $-2$
(\omterm{def:g10:proba:distribution}{प्रायिकता} $\frac46$), $3$ ($\frac16$), $4$ ($\frac16$) लेता है:
\[
\E(G) = \frac{-8 + 3 + 4}{6} = -\frac{1}{6} < 0 .
\]
औसतन खिलाड़ी प्रति खेल $17$ सेंट गँवाता है: खेल प्रतिकूल है (जैसे अधिकांश
असली खेल होते हैं)।
\end{example}

\section{बर्नूली परीक्षण और द्विपद बंटन}

\begin{definition}[बर्नूली बंटन]\label{def:g12:randvar:bernoulli}
\emph{बर्नूली परीक्षण}\index{बर्नूली परीक्षण} दो परिणामों वाला प्रयोग है:
\emph{सफलता} (\omterm{def:g10:proba:distribution}{प्रायिकता} $p$) और \emph{विफलता} ($q = 1 - p$)। सफलता का सूचक $X$
(सफलता पर $X = 1$, विफलता पर $0$) \emph{बर्नूली \omterm{def:g12:randvar:rv}{बंटन}} $\mathcal B(p)$ का पालन करता है:
\[
\E(X) = p, \qquad \V(X) = p(1-p) .
\]
(सचमुच $\E(X) = p$, $\E(X^2) = p$, और कोनिग--होयगेंस $\V(X) = p - p^2$ दे देता है।)
\end{definition}

\begin{definition}[द्विपद बंटन]\label{def:g12:randvar:binomial}
किसी \omterm{def:g12:randvar:bernoulli}{बर्नूली परीक्षण} को $n$ बार \omterm{def:g12:condprob:indep}{स्वतंत्र} रूप से दोहराइए और मान लीजिए $X$
सफलताओं की कुल संख्या है। $X$ का \omterm{def:g12:randvar:rv}{बंटन} \emph{द्विपद बंटन}\index{द्विपद बंटन}
$\mathcal B(n, p)$ है।
\end{definition}

\begin{theorem}\label{thm:g12:randvar:binomial}
यदि $X \sim \mathcal B(n, p)$ हो, तो $0 \leq k \leq n$ के लिए
\[
\P(X = k) = \binom{n}{k} p^k (1-p)^{n-k},
\]
और
\[
\E(X) = np, \qquad \V(X) = np(1-p) .
\]
\end{theorem}

\begin{proof}
$k$ सफलताओं और $n-k$ विफलताओं वाले किसी नियत परिणाम-क्रम की \omterm{def:g10:proba:distribution}{प्रायिकता}
\omterm{def:g12:condprob:indep}{स्वतंत्रता} से $p^k(1-p)^{n-k}$ है; ऐसे क्रमों की संख्या $n$ परीक्षणों में $k$ सफलताएँ
रखने के तरीक़ों की संख्या है, यानी $\binom nk$ (\cref{ch:g12:comb})। सभी क्रमों पर जोड़ने से सूत्र
मिल जाता है --- और द्विपद प्रमेय $\sum_k \P(X = k) = (p + q)^n = 1$ की पुष्टि कर देती है।

\omterm{def:g12:randvar:exp}{प्रत्याशा} के लिए $k\binom nk = n\binom{n-1}{k-1}$ (\cref{exo:g12:comb:7}) का उपयोग करने पर:
\[
\E(X) = \sum_{k=1}^{n} k\binom nk p^k q^{n-k}
= np\sum_{k=1}^{n} \binom{n-1}{k-1} p^{k-1} q^{(n-1)-(k-1)}
= np\,(p + q)^{n-1} = np .
\]
\omterm{def:g12:randvar:exp}{प्रसरण} का सूत्र इसी तरह सर्वसमिका $k(k-1)\binom nk = n(n-1)\binom{n-2}{k-2}$ से सिद्ध होता है, जो $\E(X(X-1)) = n(n-1)p^2$ देती है,
जिससे $\V(X) = n(n-1)p^2 + np - (np)^2 = np(1-p)$ निकलता है। (संरचनात्मक उपपत्ति --- \omterm{def:g12:condprob:indep}{स्वतंत्र} चरों के योग का \omterm{def:g12:randvar:exp}{प्रसरण}
--- \cref{thm:g12:sums:variance} के साथ आती है।)
\end{proof}

\begin{method}[द्विपद स्थिति पहचानना]\label{met:g12:randvar:recognize}
$X \sim \mathcal B(n,p)$ लिखने से पहले तीनों सामग्रियाँ जाँचिए: परीक्षणों की \emph{नियत संख्या}
$n$; हर परीक्षण के \emph{दो परिणाम} और \emph{एक ही} सफलता-प्रायिकता $p$;
और परीक्षणों की \emph{\omterm{def:g12:condprob:indep}{स्वतंत्रता}} (वापसी \emph{सहित} प्रतिदर्शन, या किसी बड़ी
समष्टि से)। फिर ``कम से कम एक सफलता'' के लिए
\[
\P(X \geq 1) = 1 - (1-p)^n
\]
काम में लाइए, और सामान्य $\P(X \leq k)$ के लिए कैलकुलेटर या संचयी सारणियाँ।
\end{method}

\begin{example}\label{ex:g12:randvar:atleastone}
छक्का आने की कम से कम $99\%$ संभावना पाने के लिए पासा कितनी बार फेंकना पड़ेगा?
$X \sim \mathcal B\left(n, \frac16\right)$ के साथ: $\P(X \geq 1) = 1 - \left(\frac56\right)^n \geq 0.99$ का अर्थ है $\left(\frac56\right)^n \leq 0.01$, \emph{अर्थात्} $n \geq \frac{\ln 0.01}{\ln(5/6)} \approx 25.3$: $n = 26$ फेंकों से आगे।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g12:randvar:1}
कोई \omterm{def:g12:randvar:rv}{यादृच्छिक चर} $X$ मान $-1, 0, 2, 5$ \omterm{def:g10:proba:distribution}{प्रायिकताओं} $0.3, 0.2, 0.4, 0.1$ के साथ लेता है। $\E(X)$, $\V(X)$
और $\sigma(X)$ निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:randvar:2}
किसी बहुविकल्पीय परीक्षा में $10$ प्रश्न हैं, और हर प्रश्न में $4$ विकल्प
हैं जिनमें एक सही है। कोई विद्यार्थी \omterm{def:g12:condprob:indep}{स्वतंत्र} रूप से एकसमान यादृच्छिक ढंग से
उत्तर देता है। मान लीजिए $X$ सही उत्तरों की संख्या है।
\begin{enumerate}
  \item $X$, $\E(X)$ और $\sigma(X)$ का \omterm{def:g12:randvar:rv}{बंटन} बताइए।
  \item $\P(X = 0)$, $\P(X = 5)$ और $\P(X \geq 1)$ निकालिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:randvar:3}
कोई बीमा कंपनी $n = 400$ ग्राहकों का बीमा करती है; हर एक वर्ष भर में \omterm{def:g12:condprob:indep}{स्वतंत्र} रूप
से \omterm{def:g10:proba:distribution}{प्रायिकता} $p = 0.05$ से दावा दायर करता है। मान लीजिए $X$ दावों की संख्या है।
$X$ का \omterm{def:g12:randvar:rv}{बंटन} पहचानिए और उसकी \omterm{def:g12:randvar:exp}{प्रत्याशा} तथा \omterm{def:g11:stat:variance}{मानक विचलन} निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:4}
\cref{ex:g12:randvar:game} के खेल में आयोजक प्रवेश-दाम $c$ बदलकर खेल को न्यायसंगत ($\E(G) = 0$) बनाना
चाहता है। $c$ ज्ञात कीजिए। इस न्यायसंगत रूप में लाभ का \omterm{def:g12:randvar:exp}{प्रसरण} निकालिए; क्या
``न्यायसंगत'' का अर्थ ``जोखिम-रहित'' भी है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:5}
कोई बास्केटबॉल खिलाड़ी \omterm{def:g10:proba:distribution}{प्रायिकता} $0.7$ से मुक्त फेंक में अंक बनाती है। वह
$8$ बार फेंकती है (\omterm{def:g12:condprob:indep}{स्वतंत्र} फेंक)। ये \omterm{def:g10:proba:distribution}{प्रायिकताएँ} निकालिए कि वह ठीक $6$
बार अंक बनाती है; कम से कम $6$ बार; कम से कम एक बार। अंकों की सबसे संभावित
संख्या क्या है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:6}
कोई विमान-कंपनी जानती है कि हर बुक किया हुआ यात्री \omterm{def:g12:condprob:indep}{स्वतंत्र} रूप से \omterm{def:g10:proba:distribution}{प्रायिकता}
$0.9$ से पहुँचता है। किसी उड़ान में $100$ सीटें हैं और कंपनी $104$ टिकट बेचती
है। \omterm{def:g12:randvar:binomial}{द्विपद बंटन} का उपयोग करके यह \omterm{def:g10:proba:distribution}{प्रायिकता} व्यक्त कीजिए कि सीटों से अधिक
यात्री पहुँच जाएँ, और कैलकुलेटर से उसका संख्यात्मक आकलन कीजिए (यथार्थ व्यंजक
दीजिए)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:7}
मान लीजिए $X \sim \mathcal B(n, p)$ है। दिखाइए कि
\[
\frac{\P(X = k+1)}{\P(X = k)} = \frac{n-k}{k+1}\cdot\frac{p}{1-p},
\]
और निष्कर्ष निकालिए कि \omterm{def:g12:randvar:rv}{बंटन} $k^* = \floor{(n+1)p}$ तक बढ़ता है और उसके बाद घटता है (द्विपद का
\emph{बहुलक})।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:randvar:8}
\emph{(सेंट पीटर्सबर्ग, सधा हुआ.)} कोई न्यायसंगत सिक्का चित आने तक उछाला जाता
है, पर अधिक से अधिक $10$ बार। मान लीजिए $N$ काम में आए उछालों की संख्या
है, और चित आ जाने पर खिलाड़ी $2^N$ यूरो पाता है, अन्यथा $0$।
\begin{enumerate}
  \item जीत वाले परिणामों तक सीमित $N$ का \omterm{def:g12:randvar:rv}{बंटन} बताइए: $1 \leq k \leq 10$ के लिए $\P(k \text{ पर पहली चित}) = 2^{-k}$,
        और जीतने की कुल \omterm{def:g10:proba:distribution}{प्रायिकता} जाँचिए।
  \item प्रत्याशित भुगतान निकालिए। $10$ उछालों की सीमा हटा दें तो वह क्या
        हो जाता?
\end{enumerate}
\end{exercise}

\section{समस्या: अधिविक्रय वाली उड़ान}

\begin{problem}[{सप्ताहांत समस्या --- विमान-कंपनियाँ सीटों से अधिक टिकट बेचती
हैं, कारख़ाने मुश्किल से जाँची गई खेप स्वीकार कर लेते हैं, और द्विपद बंटन
दोनों की पंचायत करता है}]
\label{pb:g12:randvar:1}
$100$ सीटों वाली कोई विमान-कंपनी ख़ुशी-ख़ुशी $105$ टिकट बेच देती है: लगभग
$10\,\%$ यात्री कभी पहुँचते ही नहीं, और ख़ाली सीटें कुछ नहीं कमातीं। कंपनी कितनी
दूर तक जा सकती है, इससे पहले कि वंचित यात्री उसका लाभ चट कर जाएँ? \omterm{def:g12:randvar:binomial}{द्विपद बंटन}
(\cref{thm:g12:randvar:binomial}) दशमलव तक उत्तर दे देता है --- और वही मशीनरी कारख़ाने की खेप जाँचती
है, पुरस्कार-लॉटरी का दाम लगाती है, और बीमाकर्ताओं को टिकाए रखती है। दोहराव
के अधीन निर्णय: यही द्विपद की रोज़ की नौकरी है।

\medskip
\noindent\textbf{भाग I --- प्रवाह।}
\begin{enumerate}
  \item पासे के तीन फेंकों में छक्कों की संख्या $X$ की पूरी बंटन-सारणी
        दीजिए।
  \item $\E(X)$ और $V(X)$ निकालिए (\cref{def:g12:randvar:exp})।
  \item कोई दुकान तीन फेंकों के $1$ यूरो लेती है और हर छक्के पर $2$ यूरो
        देती है। प्रत्याशित लाभ निकालिए: क्या खेल न्यायसंगत है?
  \item जाँच-सूची का अभ्यास (\cref{met:g12:randvar:recognize}): \emph{बिना} वापसी के निकाले गए $5$
        पत्तों में पान की संख्या क्या द्विपद है? और वापसी सहित? पुष्ट कीजिए।
  \item $X \sim \mathcal B(20,\ 0.3)$ के लिए: $\E(X)$ और $\sigma(X)$ बताइए, फिर अंक $S = 5X - 10$ के लिए $\E(S)$ तथा
        $\sigma(S)$ (\cref{prop:g12:randvar:affine})।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- अधिविक्रय वाली उड़ान।}
सीटें: $100$। बिके टिकट: $105$। हर टिकटधारी \omterm{def:g12:condprob:indep}{स्वतंत्र} रूप से \omterm{def:g10:proba:distribution}{प्रायिकता} $0.9$
से पहुँचता है; मान लीजिए $X$ पहुँचने वालों की संख्या है।
\begin{enumerate}[resume]
  \item प्रतिरूप: जाँच-सूची से $X \sim \mathcal B(105,\ 0.9)$ को पुष्ट कीजिए --- और प्रतिरूप की सबसे
        कमज़ोर कड़ी स्वीकार कीजिए (क्या न-पहुँचना सचमुच \omterm{def:g12:condprob:indep}{स्वतंत्र} है? समूहों
        और तूफ़ानों के बारे में सोचिए)।
  \item $\E(X)$ और $\sigma(X)$ निकालिए।
  \item वंचन तब होता है जब $X \geq 101$ हो: $\P(X \geq 101)$ को पाँच द्विपद पदों के स्पष्ट योग
        के रूप में लिखिए।
  \item योग का मान निकालिए (कैलकुलेटर): कितने अंश उड़ानों में कम से कम एक
        यात्री वंचित रह जाता है?
  \item पैसा: पाँच अतिरिक्त टिकट $5 \times 200 = 1\,000$ यूरो लाते हैं; और हर वंचित यात्री पर
        $800$ यूरो का मुआवज़ा लगता है। वंचित यात्रियों की प्रत्याशित संख्या
        $\sum_{k=101}^{105} (k - 100)\,\P(X = k)$, प्रत्याशित मुआवज़ा, और इस नीति पर फ़ैसला निकालिए।
  \item रचना: नियामक $\P(\text{वंचन}) < 5\,\%$ तक सहन करता है। $n = 105, 106, 107, \dots$ टिकटों को जाँचकर सबसे बड़ा
        अनुमत $n$ ज्ञात कीजिए।
  \item घंटी वाला छोटा रास्ता: $n = 105$ के लिए वंचन-सीमा $z = \frac{100.5 - \E(X)}{\sigma(X)}$ का \omterm{pb:g11:stat:1}{z-मान}
        निकालिए, और मोटे अनुमान ``लगभग $2\sigma$, इसलिए ऊपरी पूँछ में क़रीब
        $2$--$3\,\%$'' की तुलना अपने यथार्थ उत्तर से कीजिए। (इस छोटे रास्ते
        के पीछे का वक्र आगे के अध्यायों का सितारा है।)
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- कारख़ाने का फाटक।}
पुर्ज़ों की कोई खेप तभी स्वीकार होती है जब $20$ के \omterm{def:g10:proba:sample}{प्रतिदर्श} में अधिक से अधिक
एक दोषपूर्ण पुर्ज़ा हो।
\begin{enumerate}[resume]
  \item यदि सच्ची दोष-दर $2\,\%$ हो (ईमानदार खेप), तो स्वीकृति की \omterm{def:g10:proba:distribution}{प्रायिकता}
        निकालिए।
  \item यदि दर $10\,\%$ हो (ख़राब खेप), तो उसे फिर निकालिए।
  \item इस प्रक्रिया के दोनों जोखिमों के नाम बताइए (ईमानदार खेप का अस्वीकार
        होना; ख़राब खेप का स्वीकार हो जाना), उनके मान प्रश्न 13--14 से पढ़िए,
        और बताइए कि कौन-सा एक बदलाव दोनों को एक साथ सुधार देता है --- और किस
        क़ीमत पर।
  \item सुधार के पीछे: दिखाइए कि $n$ आकार के \omterm{def:g10:proba:sample}{प्रतिदर्श} में प्रेक्षित
        दोष-\emph{\omterm{def:g10:stats:series}{आपेक्षिक बारंबारता}} $\frac Xn$ का \omterm{def:g11:stat:variance}{मानक विचलन} $\sqrt{\frac{p(1-p)}{n}}$ है, और
        निष्कर्ष निकालिए कि यथार्थता प्रतिदर्श-आकार के साथ कैसे बढ़ती है (एक
        पुराना मित्र: कक्षा 10 का $\frac{1}{\sqrt n}$)।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- खेल, लॉटरी, कोष।}
\begin{enumerate}[resume]
  \item \cref{exo:g12:randvar:8} के सीमा-बद्ध सेंट पीटर्सबर्ग खेल का प्रत्याशित भुगतान $10$
        यूरो है। \cref{pb:g11:prob:1} में मिले असीमित विरोधाभास से एक अनुच्छेद में तुलना
        कीजिए: सीमा ठीक-ठीक किसे साध लेती है, और अब कौन-सा दाम न्यायसंगत
        होगा?
  \item कोई परोपकारी लॉटरी $2$ यूरो के $200$ टिकट बेचती है; पुरस्कार: एक
        $100$-यूरो और दो $50$-यूरो की टोकरियाँ। किसी टिकट का प्रत्याशित
        लाभ और लॉटरी का लाभांश निकालिए; और यूरोपीय रूले के $\frac{1}{37} \approx 2.7\,\%$ जुआघर-लाभांश
        से तुलना कीजिए। लॉटरी इससे बच कैसे निकलती है?
  \item किसी बीमाकर्ता के पास $1\,000$ \omterm{def:g12:condprob:indep}{स्वतंत्र} बीमा-पत्र हैं: हर एक पर दावे की
        \omterm{def:g10:proba:distribution}{प्रायिकता} $0.01$, दावे का आकार $10\,000$ यूरो, और प्रीमियम $130$ यूरो।
        प्रत्याशित वार्षिक लाभ \emph{और} कुल दावों का \omterm{def:g11:stat:variance}{मानक विचलन} निकालिए।
        दोनों संख्याओं की तुलना कीजिए: यह तुलना असली बीमाकर्ताओं को क्या
        रखने पर मजबूर कर देती है?
  \item समापन --- निर्णय के पंच के रूप में द्विपद: पहचान की जाँच-सूची;
        दिशा-सूई $\E \pm \sigma$; जोखिम की क़ीमत के रूप में पूँछ की \omterm{def:g10:proba:distribution}{प्रायिकताएँ}; और दो
        स्थायी चेतावनियाँ (\omterm{def:g12:condprob:indep}{स्वतंत्रता} प्रतिरूपण का दावा है, और विनाश \omterm{def:g11:stat:mean}{माध्य}
        से नहीं, विरली पूँछों से आता है)। हर एक पर एक वाक्य, और यह संकेत:
        बृहत् संख्याओं का नियम और घंटी-वक्र, जो अगले अध्यायों में हैं, इस
        पंच की नियम-पुस्तिका पूरी कर देते हैं।
\end{enumerate}
\end{problem}
