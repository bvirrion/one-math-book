\chapter{अनुक्रम}\label{ch:g12:seq}

\omterm{def:g12:seq:sequence}{अनुक्रम} \omterm{def:g10:numbers:sets}{प्राकृत संख्याओं} से सूचकित \omterm{def:g10:numbers:sets}{वास्तविक संख्याओं} की सूची है। \omterm{def:g12:seq:sequence}{अनुक्रम}
विविक्त विकास का प्रतिरूप बनाते हैं --- वर्ष-दर-वर्ष गिनी गई जनसंख्या, बैंक
खाते के शेष, किसी संख्या के क्रमिक \omterm{def:g10:numbers:approx}{सन्निकटन} --- और उनकी सीमाएँ अनंत से पहली
गंभीर भेंट हैं। यह अध्याय शब्दावली, आगमन का सिद्धांत, और अभिसरण की मूलभूत
प्रमेय तैयार करता है।

\section{आगमन द्वारा तर्क}

\begin{theorem}[आगमन का सिद्धांत]\label{thm:g12:seq:induction}
मान लीजिए $P(n)$ किसी \omterm{def:g10:numbers:sets}{पूर्णांक} $n$ पर निर्भर कथन है, और $n_0 \in \N$ है। यदि
\begin{enumerate}
  \item \emph{(आधार-स्थिति)} $P(n_0)$ सत्य हो, और
  \item \emph{(आगमन-चरण)} हर $n \geq n_0$ के लिए $P(n)$ से $P(n+1)$ निकलता हो,
\end{enumerate}
तो $P(n)$ हर $n \geq n_0$ के लिए सत्य है।
\end{theorem}

\begin{proof}
विरोधाभास के लिए मान लीजिए उन \omterm{def:g10:numbers:sets}{पूर्णांकों} $n \geq n_0$ का समुच्चय $A$, जिनके लिए
$P(n)$ असत्य है, रिक्त नहीं है। तब $A$ का एक सबसे छोटा अवयव $m$ है।\footnote{$\N$
के हर अरिक्त उपसमुच्चय का एक सबसे छोटा अवयव होता है; $\N$ के इस गुण को
अभिगृहीत मान लिया जाता है।} चूँकि $P(n_0)$ सत्य है, इसलिए $m > n_0$, अतः $m - 1 \geq n_0$ और
$m-1 \notin A$, अर्थात् $P(m-1)$ सत्य है। तब $n = m-1$ पर आगमन-चरण लगाने से $P(m)$ सत्य निकलता
है, जो $m \in A$ का खंडन करता है।
\end{proof}

\begin{example}\label{ex:g12:seq:bernoulli}
आइए \emph{बर्नूली असमिका} सिद्ध करें: हर वास्तविक $a > 0$ और हर $n \in \N$ के लिए
\[
(1+a)^n \geq 1 + na.
\]
\emph{आधार-स्थिति:} $n = 0$ के लिए दोनों पक्ष $1$ के बराबर हैं।
\emph{आगमन-चरण:} मान लीजिए किसी $n \in \N$ के लिए $(1+a)^n \geq 1+na$ है। चूँकि $1 + a > 0$ है, इसलिए
दोनों पक्षों को $1+a$ से गुणा करने पर असमिका बनी रहती है:
\[
(1+a)^{n+1} \geq (1+na)(1+a) = 1 + (n+1)a + na^2 \geq 1 + (n+1)a .
\]
आगमन द्वारा असमिका सभी $n \in \N$ के लिए सही है।
\end{example}

\begin{method}[आगमन की उपपत्ति लिखना]\label{met:g12:seq:induction}
शुरू करने से पहले कथन $P(n)$ सदा स्पष्ट रूप से लिख लीजिए। पूरी उपपत्ति के तीन
भाग दिखाई देने चाहिए: आधार-स्थिति, आगमन-चरण (``मान लीजिए $P(n)$; हम $P(n+1)$ सिद्ध
करते हैं''), और आगमन के सिद्धांत का हवाला देते हुए निष्कर्ष। सबसे आम भूल
आगमन-चरण को परिकल्पना $P(n)$ का उपयोग किए बिना सिद्ध कर देना है: ऐसा हो जाए तो
या तो उपपत्ति ग़लत है या आगमन की ज़रूरत ही नहीं थी।
\end{method}

\section{अनुक्रमों की शब्दावली}

\begin{definition}[अनुक्रम]\label{def:g12:seq:sequence}
\emph{अनुक्रम}\index{अनुक्रम} एक \omterm{def:g11:func:function}{फलन} $u \colon \N \to \R$ है (या $\{n \in \N : n \geq n_0\}$ से $\R$ तक)। $n$ का
\omterm{def:g10:functions:function}{प्रतिबिंब} $u_n$ लिखा जाता है, और अनुक्रम स्वयं $(u_n)_{n\in\N}$ या केवल $(u_n)$।
\end{definition}

\omterm{def:g12:seq:sequence}{अनुक्रम} \emph{स्पष्ट रूप से}, किसी सूत्र $u_n = f(n)$ से, या \emph{पुनरावृत्ति} से,
अपने पहले पद और किसी संबंध $u_{n+1} = f(u_n)$ से परिभाषित किया जा सकता है।

\begin{definition}[एकदिष्टता]\label{def:g12:seq:monotonic}
\omterm{def:g12:seq:sequence}{अनुक्रम} $(u_n)$ \emph{वर्धमान}\index{अनुक्रम!वर्धमान} है यदि सभी $n$ के लिए
$u_{n+1} \geq u_n$ हो, \emph{\omterm{def:g10:functions:variations}{ह्रासमान}} है यदि सभी $n$ के लिए $u_{n+1} \leq u_n$ हो, और
\emph{एकदिष्ट} है यदि वह वर्धमान या \omterm{def:g10:functions:variations}{ह्रासमान} हो। असमिकाएँ कड़ी हों तो वह
\emph{निरंतर} वर्धमान (क्रमशः \omterm{def:g10:functions:variations}{ह्रासमान}) है।
\end{definition}

\begin{method}[किसी अनुक्रम की एकदिष्टता का अध्ययन]\label{met:g12:seq:monotonicity}
तीन मानक तकनीकें:
\begin{enumerate}
  \item $u_{n+1} - u_n$ के चिह्न का अध्ययन कीजिए;
  \item यदि सभी पद धनात्मक हों, तो $\dfrac{u_{n+1}}{u_n}$ की $1$ से तुलना कीजिए;
  \item यदि $\intco{0}{+\infty}$ पर परिभाषित $f$ के साथ $u_n = f(n)$ हो, तो $f$ के परिवर्तन काम
        में लाइए।
\end{enumerate}
\end{method}

\begin{definition}[परिबद्ध अनुक्रम]\label{def:g12:seq:bounded}
\omterm{def:g12:seq:sequence}{अनुक्रम} $(u_n)$ \emph{ऊपर परिबद्ध}\index{अनुक्रम!परिबद्ध} है यदि ऐसा $M \in \R$ हो
जिसके लिए सभी $n$ के लिए $u_n \leq M$ हो; \emph{नीचे परिबद्ध} है यदि ऐसा $m \in \R$ हो
जिसके लिए सभी $n$ के लिए $u_n \geq m$ हो; और \emph{परिबद्ध} है यदि दोनों बातें सही
हों।
\end{definition}

\subsection{समांतर और गुणोत्तर अनुक्रम}

\begin{definition}[समांतर और गुणोत्तर अनुक्रम]\label{def:g12:seq:arith-geom}
\omterm{def:g12:seq:sequence}{अनुक्रम} $(u_n)$ \omterm{def:g11:seq:arithmetic}{सार्व अंतर} $r$ वाला \emph{समांतर}\index{अनुक्रम!समांतर} है यदि
सभी $n$ के लिए $u_{n+1} = u_n + r$ हो, और \omterm{def:g11:seq:geometric}{सार्व अनुपात} $q$ वाला
\emph{गुणोत्तर}\index{अनुक्रम!गुणोत्तर} है यदि सभी $n$ के लिए $u_{n+1} = q\,u_n$ हो।
\end{definition}

\begin{proposition}[स्पष्ट रूप और योग]\label{prop:g12:seq:arith-geom}
मान लीजिए $n \in \N$ है।
\begin{enumerate}
  \item यदि $(u_n)$ \omterm{def:g11:seq:arithmetic}{सार्व अंतर} $r$ वाला \omterm{def:g12:seq:arith-geom}{समांतर} हो, तो $u_n = u_0 + nr$ और
        \[
        u_0 + u_1 + \dots + u_n = (n+1)\,\frac{u_0 + u_n}{2}.
        \]
  \item यदि $(u_n)$ \omterm{def:g11:seq:geometric}{सार्व अनुपात} $q \neq 1$ वाला \omterm{def:g12:seq:arith-geom}{गुणोत्तर} हो, तो $u_n = u_0\, q^n$ और
        \[
        u_0 + u_1 + \dots + u_n = u_0\,\frac{1 - q^{n+1}}{1 - q}.
        \]
\end{enumerate}
\end{proposition}

\begin{proof}
स्पष्ट रूप तत्काल आगमन से निकल आते हैं। \omterm{def:g12:seq:arith-geom}{समांतर} योग के लिए $S = u_0 + \dots + u_n$ लिखिए और उसी
योग को उल्टे क्रम में लिखकर जोड़िए: $n+1$ स्तंभ-योगों में से हर एक $u_0 + u_n$ के
बराबर है, इसलिए $2S = (n+1)(u_0+u_n)$। \omterm{def:g12:seq:arith-geom}{गुणोत्तर} योग के लिए $S - qS$ निकालिए: पहले और अंतिम को
छोड़कर सभी पद जोड़ों में कट जाते हैं, इसलिए $(1-q)S = u_0(1 - q^{n+1})$।
\end{proof}

\section{अनुक्रम की सीमा}

\begin{definition}[अभिसारी अनुक्रम]\label{def:g12:seq:limit}
\omterm{def:g12:seq:sequence}{अनुक्रम} $(u_n)$ \omterm{def:g10:numbers:sets}{वास्तविक संख्या} $\ell$ पर \emph{अभिसरित}\index{सीमा!अनुक्रम की}
होता है यदि $\ell$ को समेटने वाले हर खुले \omterm{def:g10:numbers:interval}{अंतराल} में किसी सूचक से आगे के सभी
पद $u_n$ आ जाएँ। तब हम $\lim\limits_{n\to+\infty} u_n = \ell$ लिखते हैं।

समतुल्य रूप से: हर $\varepsilon > 0$ के लिए ऐसा $N \in \N$ है कि सभी $n \geq N$ के लिए $\abs{u_n - \ell} \leq \varepsilon$ हो।
\end{definition}

\begin{definition}[अनंत की ओर अपसरण]\label{def:g12:seq:infinity}
\omterm{def:g12:seq:sequence}{अनुक्रम} $(u_n)$ \emph{$+\infty$ की ओर जाता है} यदि हर $A \in \R$ के लिए ऐसा $N \in \N$ हो कि
सभी $n \geq N$ के लिए $u_n \geq A$ हो। हम $\lim\limits_{n\to+\infty} u_n = +\infty$ लिखते हैं; $\lim u_n = -\infty$ की परिभाषा इसी जैसी है।
जो \omterm{def:g12:seq:sequence}{अनुक्रम} \omterm{def:g12:seq:limit}{अभिसरित} नहीं होता वह \emph{अपसारी} कहलाता है।
\end{definition}

\begin{remark}
कोई \omterm{def:g12:seq:sequence}{अनुक्रम} $\pm\infty$ की ओर गए बिना भी अपसारी हो सकता है: \omterm{def:g12:seq:sequence}{अनुक्रम} $u_n = (-1)^n$ केवल मान
$1$ और $-1$ लेता है और उसकी कोई सीमा नहीं है।
\end{remark}

\begin{proposition}[सीमा की अद्वितीयता]\label{prop:g12:seq:uniqueness}
यदि $(u_n)$ \omterm{def:g12:seq:limit}{अभिसरित} हो, तो उसकी सीमा अद्वितीय है।
\end{proposition}

\begin{proof}
मान लीजिए $u_n \to \ell$ और $u_n \to \ell'$ हैं, जहाँ $\ell \neq \ell'$ है, मान लीजिए $\ell < \ell'$। $\varepsilon = \frac{\ell' - \ell}{3} > 0$ रखिए।
किसी सूचक से आगे $\abs{u_n - \ell} \leq \varepsilon$ और $\abs{u_n - \ell'} \leq \varepsilon$, अतः
\[
\ell' - \ell \leq \abs{\ell' - u_n} + \abs{u_n - \ell}
\leq 2\varepsilon = \tfrac{2}{3}(\ell' - \ell) < \ell' - \ell,
\]
जो विरोधाभास है।
\end{proof}

\begin{proposition}[सीमाओं पर संक्रियाएँ]\label{prop:g12:seq:operations}
मान लीजिए $(u_n)$ और $(v_n)$ ऐसे \omterm{def:g12:seq:sequence}{अनुक्रम} हैं जिनकी सीमाएँ $\ell$ और $\ell'$ हैं
(परिमित या अनंत)। तब, जब तक दाहिना पक्ष अनिर्धार्य रूप न हो,
\[
\lim (u_n + v_n) = \ell + \ell', \qquad
\lim (u_n v_n) = \ell\,\ell', \qquad
\lim \frac{u_n}{v_n} = \frac{\ell}{\ell'}.
\]
अनिर्धार्य रूप $(+\infty) + (-\infty)$, $0 \times \infty$, $\frac{\infty}{\infty}$ और $\frac{0}{0}$ हैं।
\end{proposition}

\begin{proof}
हम परिमित सीमाओं के लिए योग का नियम सिद्ध करते हैं; शेष स्थितियाँ इसी जैसी
हैं और अभ्यास के लिए छोड़ दी गई हैं। मान लीजिए $\varepsilon > 0$ है। ऐसे $N_1, N_2$ हैं कि
$n \geq N_1$ के लिए $\abs{u_n - \ell} \leq \varepsilon/2$ और $n \geq N_2$ के लिए $\abs{v_n - \ell'} \leq \varepsilon/2$ हो। $n \geq \max(N_1, N_2)$ के लिए त्रिभुज असमिका
देती है
\[
\abs{(u_n + v_n) - (\ell + \ell')}
\leq \abs{u_n - \ell} + \abs{v_n - \ell'} \leq \varepsilon. \qedhere
\]
\end{proof}

\begin{method}[अनिर्धार्य रूप हटाना]\label{met:g12:seq:indeterminate}
अनिर्धार्य रूप सामने आए तो हावी पद बाहर निकाल लीजिए। जैसे
\[
n^2 - n = n^2\left(1 - \tfrac{1}{n}\right) \xrightarrow[n\to+\infty]{} +\infty,
\qquad
\frac{2n^2+1}{n^2 - n} = \frac{2 + 1/n^2}{1 - 1/n}
\xrightarrow[n\to+\infty]{} 2 .
\]
\end{method}

\section{अभिसरण की प्रमेय}

\begin{theorem}[तुलना और संपीडन प्रमेय]\label{thm:g12:seq:squeeze}
मान लीजिए $(u_n)$, $(v_n)$, $(w_n)$ \omterm{def:g12:seq:sequence}{अनुक्रम} हैं।
\begin{enumerate}
  \item यदि किसी सूचक से आगे $u_n \leq v_n$ हो और $u_n \to +\infty$ हो, तो $v_n \to +\infty$।
  \item \emph{(संपीडन प्रमेय)}\index{संपीडन प्रमेय} यदि किसी सूचक से आगे
        $u_n \leq v_n \leq w_n$ हो और $(u_n)$ तथा $(w_n)$ दोनों एक ही सीमा $\ell$ पर \omterm{def:g12:seq:limit}{अभिसरित} हों,
        तो $(v_n)$ भी~$\ell$ पर \omterm{def:g12:seq:limit}{अभिसरित} होता है।
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} मान लीजिए $A \in \R$ है। चूँकि $u_n \to +\infty$, इसलिए ऐसा $N$ है कि $n \geq N$ के
लिए $u_n \geq A$ हो; आवश्यकता हो तो $N$ बड़ा करके $n \geq N$ के लिए $v_n \geq u_n \geq A$।

\emph{2.} मान लीजिए $\varepsilon > 0$ है। किसी सूचक से आगे $\ell - \varepsilon \leq u_n$ और $w_n \leq \ell + \varepsilon$ दोनों सही
हैं, अतः $\ell - \varepsilon \leq u_n \leq v_n \leq w_n \leq \ell + \varepsilon$, अर्थात् $\abs{v_n - \ell} \leq \varepsilon$।
\end{proof}

\begin{example}
सभी $n \geq 1$ के लिए $-\frac{1}{n} \leq \frac{(-1)^n}{n} \leq \frac{1}{n}$, और दोनों सीमाएँ $0$ की ओर जाती हैं; अतः $\frac{(-1)^n}{n} \to 0$।
\end{example}

\begin{theorem}[एकदिष्ट अभिसरण प्रमेय]\label{thm:g12:seq:monotone}
\omterm{def:g12:seq:bounded}{ऊपर परिबद्ध} \omterm{def:g12:seq:monotonic}{वर्धमान} \omterm{def:g12:seq:sequence}{अनुक्रम} \omterm{def:g12:seq:limit}{अभिसरित} होता है। \omterm{def:g12:seq:bounded}{नीचे परिबद्ध} \omterm{def:g10:functions:variations}{ह्रासमान} \omterm{def:g12:seq:sequence}{अनुक्रम}
\omterm{def:g12:seq:limit}{अभिसरित} होता है। जो \omterm{def:g12:seq:monotonic}{वर्धमान} \omterm{def:g12:seq:sequence}{अनुक्रम} \omterm{def:g12:seq:bounded}{ऊपर परिबद्ध} न हो वह $+\infty$ की ओर जाता है।
\end{theorem}

\begin{proof}[आंशिक उपपत्ति]
हम तीसरा कथन सिद्ध करते हैं। मान लीजिए $(u_n)$ \omterm{def:g12:seq:monotonic}{वर्धमान} है और \omterm{def:g12:seq:bounded}{ऊपर परिबद्ध} नहीं,
और $A \in \R$ है। चूँकि $A$ कोई ऊपरी परिबंध नहीं है, इसलिए ऐसा $N$ है कि
$u_N \geq A$ हो; एकदिष्टता से सभी $n \geq N$ के लिए $u_n \geq u_N \geq A$। अतः $u_n \to +\infty$।

दोनों अभिसरण-कथन $\R$ के लघुतम ऊपरी परिबंध वाले गुण पर टिके हैं; उन्हें
\emph{इस स्तर पर स्वीकार कर लिया जाता है} (और स्नातक के पहले वर्ष में सिद्ध
किया जाता है)।
\end{proof}

\begin{remark}
यह प्रमेय सीमा के \emph{अस्तित्व} की गारंटी देती है, पर उसका मान नहीं बताती।
$M$ से \omterm{def:g12:seq:bounded}{ऊपर परिबद्ध} कोई \omterm{def:g12:seq:monotonic}{वर्धमान} \omterm{def:g12:seq:sequence}{अनुक्रम} किसी $\ell \leq M$ पर \omterm{def:g12:seq:limit}{अभिसरित} होता है, और
आवश्यक नहीं कि वह $M$ ही हो।
\end{remark}

\begin{theorem}[गुणोत्तर अनुक्रमों की सीमा]\label{thm:g12:seq:geometric}
मान लीजिए $q \in \R$ है।
\begin{enumerate}
  \item यदि $q > 1$ हो, तो $q^n \to +\infty$।
  \item यदि $q = 1$ हो, तो $q^n \to 1$।
  \item यदि $\abs{q} < 1$ हो, तो $q^n \to 0$।
  \item यदि $q \leq -1$ हो, तो $(q^n)$ अपसारी है और उसकी कोई सीमा नहीं है।
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} $a > 0$ के साथ $q = 1 + a$ लिखिए। बर्नूली असमिका (\cref{ex:g12:seq:bernoulli}) से $q^n \geq 1 + na \to +\infty$ मिलता
है, और तुलना (\cref{thm:g12:seq:squeeze}) से निष्कर्ष निकल आता है।

\emph{2.} तत्काल।

\emph{3.} यदि $q = 0$ हो तो दावा स्पष्ट है। अन्यथा $\abs{q} < 1$ से $1/\abs{q} > 1$ मिलता है,
इसलिए बिंदु 1 से $(1/\abs{q})^n \to +\infty$, अतः $\abs{q}^n \to 0$, और $-\abs{q}^n \leq q^n \leq \abs{q}^n$ से संपीडन प्रमेय द्वारा निष्कर्ष
निकाला जा सकता है।

\emph{4.} $q \leq -1$ के लिए $(q^{2n})$ मान $\geq 1$ लेता है जबकि $(q^{2n+1})$ मान $\leq -1$ लेता है:
कोई एक सीमा दोनों उपअनुक्रमों को नहीं खींच सकती।
\end{proof}

\begin{method}[पुनरावर्ती अनुक्रम $u_{n+1} = f(u_n)$]\label{met:g12:seq:recurrent}
$u_{n+1} = f(u_n)$ से परिभाषित \omterm{def:g12:seq:sequence}{अनुक्रम} का अध्ययन करने के लिए:
\begin{enumerate}
  \item आगमन से सिद्ध कीजिए कि $(u_n)$ ऐसे \omterm{def:g10:numbers:interval}{अंतराल} $I$ में बना रहता है जिस
        पर $f$ का व्यवहार अच्छा है (और प्रायः यह भी कि $(u_n)$ \omterm{def:g12:seq:monotonic}{एकदिष्ट} है);
  \item \omterm{def:g12:seq:monotonic}{एकदिष्ट} अभिसरण प्रमेय से अभिसरण निकालिए;
  \item संबंध $u_{n+1} = f(u_n)$ में सीमा लीजिए: यदि $f$ संतत हो और $u_n \to \ell \in I$ हो, तो $\ell$
        $f(\ell) = \ell$ को संतुष्ट करता है (\cref{ch:g12:limcont} देखिए); इस \omterm{def:g10:algebra:equation}{समीकरण} को हल कीजिए और
        सही \omterm{def:g11:quad:discriminant}{मूल} चुनिए।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g12:seq:sqrt2}
मान लीजिए $u_0 = 2$ और $u_{n+1} = \frac{1}{2}\left(u_n + \frac{2}{u_n}\right)$ है। आगमन से जाँचा जा सकता है कि सभी $n$ के लिए
$u_n \geq \sqrt{2}$ है ($x>0$ के लिए असमिका $\frac{1}{2}(x + 2/x) \geq \sqrt{2}$ $(x - \sqrt2)^2 \geq 0$ के तुल्य है), और फिर यह कि $(u_n)$
\omterm{def:g10:functions:variations}{ह्रासमान} है, क्योंकि
\[
u_{n+1}-u_n=\frac{2-u_n^2}{2u_n}\leq 0 .
\]
\omterm{def:g10:functions:variations}{ह्रासमान} और \omterm{def:g12:seq:bounded}{नीचे परिबद्ध} होने के कारण $(u_n)$ किसी $\ell \geq \sqrt{2}$ पर \omterm{def:g12:seq:limit}{अभिसरित} होता है,
जिसे $\ell = \frac{1}{2}(\ell + 2/\ell)$ को संतुष्ट करना ही चाहिए, अर्थात् $\ell^2 = 2$। अतः $u_n \to \sqrt{2}$। यही हीरोन की
कलनविधि है, जिसे बेबीलोन वाले पहले से काम में ला रहे थे; उसका अभिसरण
अत्यंत तेज़ है ($u_3$ ही $\sqrt 2$ को आठ दशमलव स्थानों तक दे देता है)।
\end{example}

\begin{exercise}[$\star$]\label{exo:g12:seq:1}

\begin{exercise}[$\star$]\label{exo:g12:seq:2}

a_n = \frac{n+1}{n}, \qquad
b_n = \frac{2^n}{n}, \qquad
c_n = n^2 - 10n .

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:3}
इन व्यापक पदों वाले \omterm{def:g11:seq:sequence}{अनुक्रमों} की सीमाएँ निकालिए:
\[
u_n = \frac{3n^2 - n + 1}{2n^2 + 5}, \qquad
v_n = \sqrt{n+1} - \sqrt{n}, \qquad
w_n = \frac{2^n - 3^n}{3^n + 1}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:4}
मान लीजिए $(u_n)$ $u_0 = 5$ और \omterm{def:g11:seq:arithmetic}{सार्व अंतर} $r = 3$ वाला \omterm{def:g11:seq:arithmetic}{समांतर अनुक्रम} है, तथा $(v_n)$
$v_0 = 8$ और \omterm{def:g11:seq:geometric}{सार्व अनुपात} $q = \frac{1}{2}$ वाला \omterm{def:g11:seq:geometric}{गुणोत्तर अनुक्रम}। $u_n$, $v_n$, $\sum_{k=0}^{n} u_k$ और
$\sum_{k=0}^{n} v_k$ निकालिए, और $n \to +\infty$ पर चारों व्यंजकों की सीमाएँ भी।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:5}
संपीडन प्रमेय का उपयोग करके निकालिए:
\[
\lim_{n\to+\infty} \frac{n + \cos n}{n + 1}
\qquad\text{और}\qquad
\lim_{n\to+\infty} \frac{n!}{n^n},
\]
जहाँ $n! = 1 \times 2 \times \dots \times n$ है। दूसरी सीमा के लिए $\frac{n!}{n^n}$ को किसी \omterm{def:g11:seq:geometric}{गुणोत्तर अनुक्रम} के पद से
\omterm{def:g12:seq:bounded}{परिबद्ध} कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:6}
मान लीजिए $u_0 = 0$ है और सभी $n \in \N$ के लिए $u_{n+1} = \sqrt{u_n + 2}$ है।
\begin{enumerate}
  \item आगमन से सिद्ध कीजिए कि सभी $n$ के लिए $0 \leq u_n \leq 2$ है।
  \item दिखाइए कि $(u_n)$ \omterm{def:g12:seq:monotonic}{वर्धमान} है।
  \item निष्कर्ष निकालिए कि $(u_n)$ \omterm{def:g12:seq:limit}{अभिसरित} होता है और उसकी सीमा ज्ञात कीजिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:7}
एक रोगी हर सुबह किसी दवा की $1$ मात्रक ख़ुराक लेता है। हर 24-घंटे की अवधि
में शरीर मौजूद दवा का $40\%$ निकाल देता है। मान लीजिए $u_n$ दिन $n$ की
ख़ुराक के ठीक बाद शरीर में दवा की मात्रा है, जिससे $u_0 = 1$।
\begin{enumerate}
  \item पुष्ट कीजिए कि $u_{n+1} = 0.6\,u_n + 1$।
  \item मान लीजिए $v_n = u_n - 2.5$ है। दिखाइए कि $(v_n)$ \omterm{def:g12:seq:arith-geom}{गुणोत्तर} है और $u_n$ का स्पष्ट
        सूत्र निकालिए।
  \item दीर्घ काल में शरीर में दवा की मात्रा ज्ञात कीजिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:8}
मान लीजिए $(u_n)$ $u_0 = 3$ और $u_{n+1} = \frac{4u_n - 1}{u_n + 2}$ से परिभाषित है।
\begin{enumerate}
  \item आगमन से दिखाइए कि सभी $n \in \N$ के लिए $u_n > 1$ है।
  \item दिखाइए कि $v_n = \dfrac{1}{u_n - 1}$ एक \omterm{def:g11:seq:arithmetic}{समांतर अनुक्रम} परिभाषित करता है।
  \item $v_n$ और $u_n$ के स्पष्ट सूत्र तथा $(u_n)$ की सीमा निकालिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:seq:9}
$n \geq 1$ के लिए मान लीजिए $H_n = 1 + \frac{1}{2} + \frac{1}{3} + \dots + \frac{1}{n}$ है।
\begin{enumerate}
  \item दिखाइए कि सभी $n \geq 1$ के लिए $H_{2n} - H_n \geq \frac{1}{2}$ है।
  \item निष्कर्ष निकालिए कि सभी $k \in \N$ के लिए $H_{2^k} \geq 1 + \frac{k}{2}$ है, और फिर यह कि $H_n \to +\infty$।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:seq:10}
\emph{(संलग्न \omterm{def:g12:seq:sequence}{अनुक्रम}.)} दो \omterm{def:g12:seq:sequence}{अनुक्रम} $(a_n)$ और $(b_n)$ \emph{संलग्न} हैं यदि
$(a_n)$ \omterm{def:g12:seq:monotonic}{वर्धमान} हो, $(b_n)$ \omterm{def:g10:functions:variations}{ह्रासमान} हो, और $b_n - a_n \to 0$ हो।
\begin{enumerate}
  \item दिखाइए कि सभी $n$ के लिए $a_n \leq b_n$ है। (संकेत: $(b_n - a_n)$ की एकदिष्टता का
        अध्ययन कीजिए।)
  \item दिखाइए कि संलग्न \omterm{def:g12:seq:sequence}{अनुक्रम} दोनों \omterm{def:g12:seq:limit}{अभिसरित} होते हैं, और एक ही सीमा पर।
  \item अनुप्रयोग: दिखाइए कि \omterm{def:g12:seq:sequence}{अनुक्रम} $a_n = \sum_{k=0}^{n} \frac{1}{k!}$ और $b_n = a_n + \frac{1}{n \cdot n!}$ ($n \geq 1$) संलग्न हैं।
        (उनकी उभयनिष्ठ सीमा संख्या $\eu$ है, जिसका अध्ययन \cref{ch:g12:exp} में है।)
\end{enumerate}
\end{exercise}

\section{समस्या: हीरोन का अनुक्रम, अंततः कठघरे में}

\begin{problem}[{सप्ताहांत समस्या --- आगमन प्रमाणित करता है, एकदिष्ट अभिसरण
फ़ैसला सुनाता है, और $\sqrt2$ का दो हज़ार साल पुराना नुस्ख़ा आख़िरकार अपनी
उपपत्ति पा लेता है (मिठाई में गाउस का अद्भुत माध्य)}]
\label{pb:g12:seq:1}
इस शृंखला में तीन बार हीरोन का नुस्ख़ा मिल चुका है --- अनुमान का $2/\text{अनुमान}$ के साथ
औसत लीजिए --- और तीनों बार वह केवल यह \emph{देख} पाया कि नुस्ख़ा काम करता
है। इस अध्याय के पास आख़िरकार न्याय के यंत्र हैं: आगमन (\cref{thm:g12:seq:induction}), \omterm{def:g12:seq:monotonic}{एकदिष्ट}
अभिसरण प्रमेय (\cref{thm:g12:seq:monotone}), और पुनरावृत्तियों की सीमाएँ। फ़ैसला और प्रमाणित गति
इस समस्या के हृदय में हैं; उनके चारों ओर आगमन के चिरपरिचित जाल, गणित का
सबसे धीमा अपसरण, और गाउस को मिला सबसे तेज़ अभिसरण।

\medskip
\noindent\textbf{भाग I --- आगमन की तैयारी।}
\begin{enumerate}
  \item आगमन से सिद्ध कीजिए: $1 + 3 + 5 + \dots + (2n - 1) = n^2$ (विषम संख्याओं की वही सीढ़ी, जो माध्यमिक
        विद्यालय खंड में खींची गई थी, अब प्रमाणित)।
  \item आगमन से सिद्ध कीजिए कि हर $n \in \N$ के लिए $2^n > n$ है।
  \item बर्नूली असमिका आगमन से सिद्ध कीजिए: $x \geq 0$ और $n \in \N$ के लिए $(1 + x)^n \geq 1 + nx$।
  \item चिरपरिचित जाल: ``सारे कंचों का रंग एक ही होता है --- एक कंचे के लिए
        सत्य; और यदि कोई भी $n$ कंचे सदा एक ही रंग के हों, तो $n + 1$ कंचों
        में पहले $n$ का रंग एक है और अंतिम $n$ का रंग एक है, इसलिए
        सभी $n + 1$ का एक ही है।'' हर बच्चा जानता है कि निष्कर्ष बेतुका है:
        ठीक-ठीक वह चरण ढूँढ़िए जहाँ आगमन टूट जाता है।
  \item आगमन से सिद्ध कीजिए कि हर $n \in \N$ के लिए $4^n - 1$ $3$ से विभाज्य है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- हीरोन का मुक़दमा।}
मान लीजिए $x_0 = 2$ और $x_{n+1} = \dfrac12\left(x_n +
\dfrac{2}{x_n}\right)$ हैं।
\begin{enumerate}[resume]
  \item $x_1$, $x_2$, $x_3$ को यथार्थ भिन्नों के रूप में निकालिए (पुराने
        मित्र)।
  \item मुख्य सर्वसमिका
        \[
        x_{n+1}^2 - 2 = \left(\frac{x_n^2 - 2}{2x_n}\right)^{\!2}
        \geq 0,
        \]
        सिद्ध कीजिए और आगमन से निकालिए कि हर $n$ के लिए $x_n > 0$ और $x_n^2 > 2$ है।
  \item दिखाइए कि $(x_n)$ निरंतर \omterm{def:g10:functions:variations}{ह्रासमान} है ($x_{n+1} - x_n$ निकालिए और प्रश्न 7 काम
        में लाइए)।
  \item \omterm{def:g12:seq:monotonic}{एकदिष्ट} अभिसरण प्रमेय का हवाला दीजिए: $(x_n)$ किसी सीमा $L \geq 1$ पर
        \omterm{def:g12:seq:limit}{अभिसरित} क्यों होता है?
  \item सीमा पहचानिए: पुनरावृत्ति में सीमा लीजिए (\cref{prop:g12:seq:operations}) और $L = \sqrt2$ निष्कर्ष
        निकालिए। ऐतिहासिक फ़ैसला सुनाइए: दो हज़ार वर्षों की वफ़ादार सेवा के
        बाद हीरोन का नुस्ख़ा \omterm{def:g12:seq:limit}{अभिसरित} होता \emph{सिद्ध} हो गया।
  \item प्रमाणित गति: $e_n = x_n - \sqrt2$ के साथ सिद्ध कीजिए
        \[
        e_{n+1} = \frac{e_n^2}{2 x_n} ,
        \]
        और $e_{n+1} \leq \frac{e_n^2}{2\sqrt2}$ निष्कर्ष निकालिए: हर क़दम पर त्रुटि का वर्ग हो जाता है ---
        माध्यमिक विद्यालय खंड से देखा जाता रहा अंकों का दुगुना होना, अब एक
        प्रमेय।
  \item संख्याओं से पुष्टि कीजिए: $e_0, e_1, e_2, e_3$ निकालिए (प्रश्न 6 से) और जाँचिए कि
        हर $\frac{e_{n+1}}{e_n^2}$ $\frac{1}{2x_n}$ के पास है।
\end{enumerate}

\noindent\textbf{भाग III --- सबसे धीमा अपसरण।}
\begin{enumerate}[resume]
  \item \Cref{exo:g12:seq:9} ने हरात्मक योगों के लिए $H_{2^k} \geq 1 + \frac k2$ सिद्ध किया था। $H_n > 10$ की गारंटी
        के लिए कितने पद चाहिए? (2 की कोई घात काम आएगी; उसके आकार पर अचरज
        कीजिए।)
  \item इसके विपरीत \omterm{def:g12:seq:arith-geom}{गुणोत्तर} योग $1 + \frac12 + \frac14 + \dots + \frac{1}{2^n} =
        2 - \frac{1}{2^n}$ $2$ पर \omterm{def:g12:seq:limit}{अभिसरित} होते हैं (\cref{thm:g12:seq:geometric}):
        माध्यमिक विद्यालय खंड का चॉकलेट वाला अंतर्ज्ञान, अंततः सीमा का
        कथन। दो पंक्ति की उपपत्ति लिखिए।
  \item इन दोनों के बीच: दिखाइए कि योग $S_n = 1 + \frac{1}{4} + \frac{1}{9} + \dots +
        \frac{1}{n^2}$ \omterm{def:g12:seq:limit}{अभिसरित} होते हैं --- इसके लिए
        $\frac{1}{k^2} \leq \frac{1}{k(k-1)} =
        \frac{1}{k-1} - \frac{1}{k}$ ($k \geq 2$ के लिए) से \omterm{def:g12:seq:bounded}{परिबद्ध} कीजिए, क्रमिक निरसन कीजिए, और
        \omterm{def:g12:seq:monotonic}{एकदिष्ट} अभिसरण लगाइए। (सीमा $\frac{\pi^2}{6}$ ऑयलर के चमत्कारों में से एक है,
        जो स्नातक खंडों में सिद्ध होती है।)
  \item प्रश्न 13--15 की सीख दो वाक्यों में बताइए: ``पद $0$ की ओर जाते
        हैं'' से योगों के अभिसरण के बारे में क्या तय होता है --- और क्या
        नहीं?
\end{enumerate}

\noindent\textbf{भाग IV --- गाउस का समांतर--गुणोत्तर \omterm{def:g11:stat:mean}{माध्य}।}
मान लीजिए $a_0 = 1$, $b_0 = 2$, और
\[
a_{n+1} = \sqrt{a_n b_n},
\qquad
b_{n+1} = \frac{a_n + b_n}{2} .
\]
\begin{enumerate}[resume]
  \item $a_1, b_1, a_2, b_2$ निकालिए (पाँच दशमलव स्थान)। गति के बारे में आप क्या देखते
        हैं?
  \item दिखाइए कि हर $n$ के लिए $a_n \leq b_n$ है (समांतर--गुणोत्तर असमिका, जो इस
        शृंखला में जगह-जगह मिली है), कि $(a_n)$ बढ़ता है और $(b_n)$ घटता है।
  \item दिखाइए कि
        $b_{n+1} - a_{n+1} \leq \frac{b_n - a_n}{2}$
        ($b_{n+1} - a_{n+1} =
        \frac{(\sqrt{b_n} - \sqrt{a_n})^2}{2}$ का \omterm{def:g10:algebra:expand}{गुणनखंडन} करके तुलना कीजिए), और \cref{exo:g12:seq:10} के साथ निष्कर्ष
        निकालिए कि दोनों \omterm{def:g12:seq:sequence}{अनुक्रम} संलग्न हैं: उनकी एक उभयनिष्ठ सीमा $M(1, 2)$ है,
        जो \emph{समांतर--गुणोत्तर \omterm{def:g11:stat:mean}{माध्य}} कहलाती है।
  \item $M(1, 2)$ को छह दशमलव स्थानों तक निकालिए (कितनी पुनरावृत्तियाँ लगीं?)।
        30 मई 1799 को गाउस ने $M(1, \sqrt2)$ को ग्यारह दशमलव स्थानों तक निकाला, $\frac{\pi}{M(1,\sqrt2)}$
        को एक ज्ञात समाकल के रूप में पहचाना, और लिखा कि ``विश्लेषण का एक नया
        क्षेत्र'' खुल गया है --- और सचमुच खुल गया था: दीर्घवृत्तीय समाकल,
        जिनकी कहानी स्नातक खंडों में है। इस समस्या में देखी गई अभिसरण-गतियों
        को सबसे धीमी से सबसे तेज़ के क्रम में लिखकर समापन कीजिए।
\end{enumerate}
\end{problem}
