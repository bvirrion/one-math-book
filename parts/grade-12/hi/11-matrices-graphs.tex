\chapter{आव्यूह और ग्राफ़}\label{ch:g12:matrix}

\omterm{def:g12:matrix:matrix}{आव्यूह} संख्याओं की आयताकार सारणी है, जिसे जोड़ने और गुणा करने के नियम इस तरह
गढ़े गए हैं कि आव्यूह-बीजगणित रैखिक रूपांतरणों के संयोजन को दर्शा सके। \omterm{def:g12:matrix:matrix}{आव्यूह}
\omterm{def:g10:lines:system}{रैखिक निकाय} हल करते हैं, युग्मित पुनरावर्ती \omterm{def:g12:seq:sequence}{अनुक्रम} चलाते हैं, और जालों में
चालें गिनते हैं --- यही गणित खोज-इंजनों और सबसे छोटे रास्ते की कलनविधियों के
पीछे है।

\section{आव्यूह-बीजगणित}

\begin{definition}[आव्यूह]\label{def:g12:matrix:matrix}
\emph{$m \times n$ आव्यूह}\index{आव्यूह} \omterm{def:g10:numbers:sets}{वास्तविक संख्याओं} की ऐसी सारणी है जिसमें $m$
पंक्तियाँ और $n$ स्तंभ हों:
$A = (a_{ij})$, जहाँ $a_{ij}$ पंक्ति $i$, स्तंभ $j$ की प्रविष्टि है। एक ही आकार के दो आव्यूह
प्रविष्टि-दर-प्रविष्टि जोड़े जाते हैं, और $\lambda A = (\lambda a_{ij})$।
\end{definition}

\begin{definition}[आव्यूह गुणनफल]\label{def:g12:matrix:product}
मान लीजिए $A$ $m \times n$ का है और $B$ $n \times p$ का। गुणनफल $AB$ वह $m \times p$ \omterm{def:g12:matrix:matrix}{आव्यूह} है
जिसकी $(i,j)$ प्रविष्टि
\[
(AB)_{ij} = \sum_{k=1}^{n} a_{ik} b_{kj}
\]
है (``$A$ की पंक्ति $i$ गुणा $B$ का स्तंभ $j$'' वाला नियम)।
\end{definition}

\begin{example}
$\begin{pmatrix} 1 & 2\\ 3 & 4\end{pmatrix}
\begin{pmatrix} 0 & 1\\ 1 & 1\end{pmatrix}
= \begin{pmatrix} 2 & 3\\ 4 & 7\end{pmatrix}$, \quad जबकि \quad $\begin{pmatrix} 0 & 1\\ 1 & 1\end{pmatrix}
\begin{pmatrix} 1 & 2\\ 3 & 4\end{pmatrix}
= \begin{pmatrix} 3 & 4\\ 4 & 6\end{pmatrix}$: \emph{\omterm{def:g12:matrix:matrix}{आव्यूहों} का गुणन क्रमविनिमेय नहीं है}।
\end{example}

\begin{proposition}[आव्यूह-बीजगणित के नियम]\label{prop:g12:matrix:rules}
जब भी आकार गुणनफलों को अर्थपूर्ण बनाते हों:
\[
(AB)C = A(BC), \qquad A(B + C) = AB + AC, \qquad (A+B)C = AC + BC,
\]
और \emph{तत्समक \omterm{def:g12:matrix:matrix}{आव्यूह}} $I_n$ (विकर्ण पर एक, अन्यत्र शून्य) $m \times n$ आकार वाले
$A$ के लिए $I_m A = A I_n = A$ संतुष्ट करता है।
\end{proposition}

\begin{proof}
सभी \cref{def:g12:matrix:product} से प्रविष्टि-दर-प्रविष्टि सत्यापन हैं; साहचर्य, जो अकेला अ-तुच्छ है,
वस्तुतः दो परिमित योगों की अदला-बदली ही है: $\bigl((AB)C\bigr)_{ij} = \sum_l \left(\sum_k a_{ik}b_{kl}\right) c_{lj}
= \sum_k a_{ik} \left(\sum_l b_{kl} c_{lj}\right)
= \bigl(A(BC)\bigr)_{ij}$।
\end{proof}

\begin{definition}[प्रतिलोम]\label{def:g12:matrix:inverse}
$n$ आकार का कोई वर्ग \omterm{def:g12:matrix:matrix}{आव्यूह} $A$ \emph{प्रतिलोमनीय}\index{आव्यूह!प्रतिलोम}
है यदि ऐसा \omterm{def:g12:matrix:matrix}{आव्यूह} $B$ हो कि $AB = BA = I_n$ हो; तब $B$ अद्वितीय होता है और उसे $A^{-1}$
लिखते हैं।
\end{definition}

\begin{proposition}[$2\times2$ आव्यूह का प्रतिलोम]\label{prop:g12:matrix:inverse2x2}
मान लीजिए $A = \begin{pmatrix} a & b\\ c & d\end{pmatrix}$ और $\det A = ad - bc$ (\emph{सारणिक}\index{सारणिक}) हैं। तब $A$
\omterm{def:g12:matrix:inverse}{प्रतिलोमनीय} है यदि और केवल यदि $\det A \neq 0$ हो, और उस स्थिति में
\[
A^{-1} = \frac{1}{ad - bc}\begin{pmatrix} d & -b\\ -c & a\end{pmatrix}.
\]
\end{proposition}

\begin{proof}
एक गणना $A \begin{pmatrix} d & -b\\ -c & a\end{pmatrix}
= \begin{pmatrix} d & -b\\ -c & a\end{pmatrix} A = (ad - bc) I_2$ देती है; यदि $ad - bc \neq 0$ हो तो भाग दे दीजिए। इसके उलट, यदि $ad - bc = 0$ हो तो
$A$ के स्तंभ समानुपाती होते हैं, और किसी भी $B$ के लिए $AB$ के स्तंभ भी;
पर $I_2$ के स्तंभ समानुपाती नहीं हैं, इसलिए कोई $B$ $AB = I_2$ को संतुष्ट नहीं कर
सकता।
\end{proof}

\begin{method}[रैखिक निकाय]\label{met:g12:matrix:systems}
निकाय
$\begin{cases} ax + by = e\\ cx + dy = f \end{cases}$
$X = \begin{pmatrix} x \\ y\end{pmatrix}$, $Y = \begin{pmatrix} e \\ f\end{pmatrix}$ के साथ \omterm{def:g12:matrix:matrix}{आव्यूह} \omterm{def:g10:algebra:equation}{समीकरण} $AX = Y$ है। यदि $\det A \neq 0$ हो, तो उसका अद्वितीय हल
$X = A^{-1}Y$ है। यही औपचारिकता $n$ अज्ञातों में $n$ \omterm{def:g10:algebra:equation}{समीकरण} भी सँभाल लेती है।
\end{method}

\section{आव्यूह की घातें और पुनरावर्ती अनुक्रम}

\begin{definition}\label{def:g12:matrix:power}
किसी वर्ग \omterm{def:g12:matrix:matrix}{आव्यूह} $A$ और $k \in \N$ के लिए $A^k = A \times \dots \times A$ ($k$ गुणनखंड), और $A^0 = I$।
\end{definition}

\begin{method}[विकर्ण-जमा-शून्यघाती और विकर्णीयकरणीय स्थितियाँ]\label{met:g12:matrix:powers}
$A^k$ निकालने के दो मानक तरीक़े:
\begin{itemize}
  \item यदि $A = \lambda I + N$ हो, जहाँ $N^2 = 0$, तो द्विपद प्रमेय (यहाँ वैध, क्योंकि $I$ और
        $N$ क्रमविनिमेय हैं) सिमटकर दो पदों में आ जाती है: $A^k = \lambda^k I + k \lambda^{k-1} N$।
  \item यदि ऐसा \omterm{def:g12:matrix:inverse}{प्रतिलोमनीय} $P$ मिल जाए कि $A = PDP^{-1}$ हो और $D$ विकर्ण हो, तो
        $A^k = P D^k P^{-1}$, और $D^k$ प्रविष्टि-दर-प्रविष्टि निकल आता है। (ऐसा $P$
        व्यवस्थित ढंग से ढूँढ़ना \emph{विकर्णीकरण} का सिद्धांत है, जो स्नातक
        स्तर पर विकसित होता है; इस स्तर पर $P$ दे दिया जाता है।)
\end{itemize}
\end{method}

\begin{example}[युग्मित अनुक्रम]\label{ex:g12:matrix:coupled}
मान लीजिए $u_{n+1} = 3u_n + v_n$ और $v_{n+1} = u_n + 3v_n$ हैं। $X_n = \begin{pmatrix} u_n\\ v_n \end{pmatrix}$ और $A = \begin{pmatrix} 3 & 1\\ 1 & 3\end{pmatrix}$ रखने पर $X_{n+1} = AX_n$ मिलता है, इसलिए $X_n = A^n X_0$।
सहायक \omterm{def:g12:seq:sequence}{अनुक्रम} $s_n = u_n + v_n$ और $d_n = u_n - v_n$ $s_{n+1} = 4s_n$ तथा $d_{n+1} = 2d_n$ को संतुष्ट करते हैं, इसलिए $s_n = 4^n s_0$,
$d_n = 2^n d_0$ और
\[
u_n = \frac{4^n(u_0+v_0) + 2^n(u_0-v_0)}{2}, \qquad
v_n = \frac{4^n(u_0+v_0) - 2^n(u_0-v_0)}{2}.
\]
(पर्दे के पीछे: $(1,1)$ और $(1,-1)$ $A$ की अभिलक्षणिक सदिश-दिशाएँ हैं।)
\end{example}

\section{ग्राफ़ और चालें}

\begin{definition}[ग्राफ़, आसन्नता आव्यूह]\label{def:g12:matrix:graph}
\emph{ग्राफ़}\index{ग्राफ़} में शीर्ष $1, 2, \dots, n$ और शीर्षों के कुछ युग्मों को
जोड़ने वाली कोरें होती हैं (\emph{दिष्ट} ग्राफ़ के लिए क्रमित युग्म)। उसका
\emph{आसन्नता आव्यूह}\index{आसन्नता आव्यूह} वह $n \times n$ \omterm{def:g12:matrix:matrix}{आव्यूह} $M$ है जिसमें
$i$ से $j$ तक कोई कोर होने पर $m_{ij} = 1$ और अन्यथा $0$ होता है। $i$ से
$j$ तक लंबाई $k$ की \emph{चाल} $k$ क्रमागत कोरों का ऐसा क्रम है जो $i$
से $j$ तक ले जाए।
\end{definition}

\begin{tikzpicture}[baseline=(current bounding box.center)]
  \node[circle, draw, inner sep=2.5pt] (v1) at (0,0) {$1$};
  \node[circle, draw, inner sep=2.5pt] (v2) at (2.6,1.2) {$2$};
  \node[circle, draw, inner sep=2.5pt] (v3) at (2.6,-1.2) {$3$};
  \draw[-Stealth, omDef, thick] (v1) -- (v2);
  \draw[-Stealth, omDef, thick] (v2) -- (v3);
  \draw[-Stealth, omDef, thick] (v1) to[bend right=18] (v3);
  \draw[-Stealth, omDef, thick] (v3) to[bend right=18] (v1);
\end{tikzpicture}
\qquad\qquad
$M = \begin{pmatrix} 0 & 1 & 1\\ 0 & 0 & 1\\ 1 & 0 & 0 \end{pmatrix}$

{\small एक दिष्ट \omterm{def:g12:matrix:graph}{ग्राफ़} और उसका \omterm{def:g12:matrix:graph}{आसन्नता आव्यूह} (\cref{exo:g12:matrix:6}): $m_{ij} = 1$ ठीक तब जब $i$
से $j$ तक कोई कोर हो।}
\end{omfigure}

\begin{theorem}[चालें गिनना]\label{thm:g12:matrix:walks}
शीर्ष $i$ से शीर्ष $j$ तक लंबाई $k$ की चालों की संख्या $M^k$ की $(i,j)$
प्रविष्टि है।
\end{theorem}

\begin{proof}
$k$ पर आगमन। $k = 1$ के लिए यह $M$ की परिभाषा ही है। मान लीजिए दावा $k$
के लिए सही है। $i$ से $j$ तक लंबाई $k+1$ की चाल $i$ से किसी शीर्ष $l$
तक लंबाई $k$ की चाल है, और उसके बाद $l$ से $j$ तक एक कोर; योग और गुणन
के सिद्धांतों से उनकी संख्या है
\[
\sum_{l=1}^{n} \bigl(M^k\bigr)_{il}\, m_{lj} = \bigl(M^{k+1}\bigr)_{ij}.
\qedhere
\]
\end{proof}

\begin{example}
त्रिभुज \omterm{def:g12:matrix:graph}{ग्राफ़} ($3$ शीर्ष, सभी युग्म जुड़े हुए) के लिए $M = \begin{pmatrix} 0&1&1\\ 1&0&1\\ 1&1&0\end{pmatrix}$ और $M^2 = \begin{pmatrix} 2&1&1\\ 1&2&1\\ 1&1&2\end{pmatrix}$: हर
शीर्ष से अपने ही पास लौटने वाली लंबाई $2$ की $2$ चालें हैं (किसी भी
पड़ोसी से होकर) और हर दूसरे शीर्ष तक $1$।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g12:matrix:1}
मान लीजिए $A = \begin{pmatrix} 1 & 2\\ 0 & 1 \end{pmatrix}$ और $B = \begin{pmatrix} 2 & 0\\ 1 & 1 \end{pmatrix}$ हैं। $A + B$, $AB$, $BA$ और $A^2$ निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:matrix:2}
बताइए कि निम्नलिखित \omterm{def:g12:matrix:matrix}{आव्यूह} \omterm{def:g12:matrix:inverse}{प्रतिलोमनीय} हैं या नहीं, और जहाँ प्रतिलोम हों वहाँ
उन्हें निकालिए:
\[
A = \begin{pmatrix} 2 & 5\\ 1 & 3\end{pmatrix}, \qquad
B = \begin{pmatrix} 3 & 6\\ 2 & 4\end{pmatrix}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:matrix:3}
आव्यूह-प्रतिलोमन से निकाय
$\begin{cases} 2x + 5y = 1\\ x + 3y = 2 . \end{cases}$
हल कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:matrix:4}
मान लीजिए $N = \begin{pmatrix} 0 & 1\\ 0 & 0 \end{pmatrix}$ के साथ $A = \begin{pmatrix} 2 & 1\\ 0 & 2\end{pmatrix} = 2I + N$ है।
\begin{enumerate}
  \item जाँचिए कि $N^2 = 0$, और यह भी कि $I$ तथा $N$ क्रमविनिमेय हैं।
  \item सभी $k \in \N$ के लिए $A^k$ निकालिए और $k=2$ का सूत्र सीधी गणना से
        सत्यापित कीजिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:matrix:5}
मान लीजिए $A = \begin{pmatrix} 0 & 1\\ 1 & 1\end{pmatrix}$ है और $F_n$ फिबोनाच्ची \omterm{def:g12:seq:sequence}{अनुक्रम} ($F_0 = 0$, $F_1 = 1$, $F_{n+2} = F_{n+1} + F_n$)। आगमन
से दिखाइए कि $n \geq 1$ के लिए
\[
A^n = \begin{pmatrix} F_{n-1} & F_n\\ F_n & F_{n+1}\end{pmatrix},
\]
और सर्वसमिका $F_{n+1}F_{n-1} - F_n^2 = (-1)^n$ निकालिए। (संकेत: \omterm{def:g11:vect:det}{सारणिक} गुणा होते हैं: $\det(MN) = \det M \det N$, जिसे आप
$2\times2$ \omterm{def:g12:matrix:matrix}{आव्यूहों} के लिए जाँच सकते हैं।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:matrix:6}
शीर्षों $\{1, 2, 3\}$ वाले किसी दिष्ट \omterm{def:g12:matrix:graph}{ग्राफ़} में कोरें $1\to2$, $2\to3$, $3\to1$ और $1\to3$
हैं।
\begin{enumerate}
  \item \omterm{def:g12:matrix:graph}{आसन्नता आव्यूह} $M$ लिखिए और $M^2$ तथा $M^3$ निकालिए।
  \item $1$ से $1$ तक लंबाई $3$ की कितनी चालें हैं? उन्हें गिनाइए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:matrix:7}
कोई कार-साझा कंपनी दो नगरों $A$ और $B$ के बीच वाहन घुमाती है। हर सप्ताह
$A$ की $80\%$ कारें $A$ में ही रहती हैं और $20\%$ $B$ चली जाती हैं; $B$
की $30\%$ कारें $A$ चली जाती हैं और $70\%$ वहीं रहती हैं। मान लीजिए $a_n, b_n$ हर
नगर में बेड़े के अनुपात हैं।
\begin{omfigure}
\begin{tikzpicture}
  \node[circle, draw, inner sep=3pt] (A) at (0,0) {$A$};
  \node[circle, draw, inner sep=3pt] (B) at (3.2,0) {$B$};
  \draw[-Stealth, omDef, thick] (A) to[bend left=25]

  \draw[-Stealth, omDef, thick] (B) to[bend left=25]

\begin{enumerate}
  \item $X_n = \begin{pmatrix} a_n\\ b_n\end{pmatrix}$ के साथ $X_{n+1} = MX_n$ लिखिए और $M$ पहचानिए।
  \item साम्य अनुपात ज्ञात कीजिए ($a + b = 1$ के साथ $MX = X$ हल कीजिए)।
  \item दिखाइए कि $c_n = a_n - 0.6$ $c_{n+1} = 0.5\,c_n$ को संतुष्ट करता है, और निष्कर्ष निकालिए कि
        बेड़े का बँटवारा साम्य पर \omterm{def:g12:seq:limit}{अभिसरित} होता है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:matrix:8}
मान लीजिए $A = \begin{pmatrix} 3 & 1\\ 1 & 3 \end{pmatrix}$, $P = \begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix}$ हैं।
\begin{enumerate}
  \item $P^{-1}$ निकालिए, फिर $D = P^{-1}AP$, और जाँचिए कि $D$ विकर्ण है।
  \item $A^n$ का बंद सूत्र निकालिए और \cref{ex:g12:matrix:coupled} से तुलना कीजिए।
\end{enumerate}
\end{exercise}

\section{समस्या: वह आव्यूह जो फिबोनाच्ची (और मौसम) जानता है}

\begin{problem}[{सप्ताहांत समस्या --- एक $2 \times 2$ आव्यूह पूरा फिबोनाच्ची उठाए
रहता है, एक मार्कोव आव्यूह दीर्घ काल का मौसम बता देता है, और एक अभिलक्षणिक
सदिश अरबों का मोल रखता है}]
\label{pb:g12:matrix:1}
\omterm{def:g12:matrix:matrix}{आव्यूह} एक ऐसी मशीन है जो कोई अवस्था खाती है और अगली लौटा देती है --- और इसलिए
उसकी \emph{घातें} पूरे-पूरे भविष्य समेटे रहती हैं। यह समस्या उस चकित कर देने
वाले \omterm{def:g12:matrix:matrix}{आव्यूह} से शुरू होती है जिसकी घातें फिबोनाच्ची संख्याएँ गिना देती हैं
(और उनकी सर्वसमिकाएँ एक-एक पंक्ति में सिद्ध कर देती हैं), फिर मौसम को मार्कोव
शृंखला के रूप में उसकी स्थायी अवस्था तक चलाती है, और उस अभिलक्षणिक \omterm{def:g11:vect:vector}{सदिश} पर
समाप्त होती है जिस पर एक खोज-इंजन खड़ा किया गया था (\cref{thm:g12:matrix:walks}, \cref{met:g12:matrix:powers})।

\medskip
\noindent\textbf{भाग I --- प्रवाह।}
\begin{enumerate}
  \item $A = \begin{pmatrix} 1 & 2\\ 3 & 4\end{pmatrix}$ और $B = \begin{pmatrix} 0 & 1\\ 1 & 0\end{pmatrix}$ के साथ: $AB$ और $BA$ निकालिए। क्रमविनिमेयता पर क्या
        फ़ैसला है?
  \item $\begin{pmatrix} 2 & 1\\ 5 & 3\end{pmatrix}$ का प्रतिलोम (\cref{prop:g12:matrix:inverse2x2}) निकालिए और उससे $2x + y = 4$, $5x + 3y = 7$ हल कीजिए।
  \item मान लीजिए $N = \begin{pmatrix} 0 & 1\\ 0 & 0\end{pmatrix}$ है: $N^2$ निकालिए, और हर $n$ के लिए $(I + N)^n = I + nN$ निकालिए।
  \item त्रिभुज \omterm{def:g12:matrix:graph}{ग्राफ़} (तीन शीर्ष, सभी युग्म जुड़े): उसका \omterm{def:g12:matrix:graph}{आसन्नता आव्यूह}
        $A$ लिखिए, $A^3$ निकालिए, और विकर्ण की प्रविष्टियों की व्याख्या
        कीजिए (\cref{thm:g12:matrix:walks})।
  \item $D = \begin{pmatrix} 2 & 0\\ 0 & \frac12
        \end{pmatrix}$ के लिए: $D^n$ और $n \to \infty$ पर उसका व्यवहार बताइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- फिबोनाच्ची \omterm{def:g12:matrix:matrix}{आव्यूह}।}
मान लीजिए $F = \begin{pmatrix} 1 & 1\\ 1 & 0\end{pmatrix}$ है और $F_1 = F_2 = 1, F_3 = 2, \dots$ \cref{pb:g11:seq:1} की फिबोनाच्ची संख्याएँ हैं।
\begin{enumerate}[resume]
  \item $F^2$, $F^3$, $F^4$ निकालिए और फिबोनाच्ची संख्याओं के पदों में $F^n$
        के व्यापक रूप का अनुमान लगाइए।
  \item अनुमान $F^n = \begin{pmatrix} F_{n+1} & F_n\\ F_n & F_{n-1}
        \end{pmatrix}$ आगमन से सिद्ध कीजिए।
  \item दोनों पक्षों के \omterm{def:g11:vect:det}{सारणिक} लीजिए (गुणनफल का \omterm{def:g11:vect:det}{सारणिक} \omterm{def:g11:vect:det}{सारणिकों} का गुणनफल है
        --- यदि आपने यह पहले न देखा हो तो $2 \times 2$ \omterm{def:g12:matrix:matrix}{आव्यूहों} पर जाँच लीजिए): और
        \emph{कासीनी सर्वसमिका} $F_{n+1}F_{n-1} - F_n^2 = (-1)^n$ निकालिए --- वही ``ग़ायब वर्ग'' का
        इंजन, अब एक पंक्ति में सिद्ध।
  \item $F^{m+n} = F^m F^n$ से ऊपर-दाईं प्रविष्टियाँ पढ़िए और \emph{योग सूत्र}
        \[
        F_{m+n} = F_{m+1} F_n + F_m F_{n-1} .
        \]
        निकालिए। $m = n = 3$ के लिए उसकी जाँच कीजिए।
  \item योग सूत्र से ($k$ पर आगमन द्वारा) निकालिए कि $F_n$ $F_{kn}$ को
        \omterm{def:g12:arith:divides}{विभाजित} करता है, और $F_3 \mid F_6$ तथा $F_3 \mid F_9$ पर उसकी पुष्टि कीजिए।
  \item $F_{100}$ निकालने के लिए $100$ \omterm{def:g12:matrix:matrix}{आव्यूह} गुणा करने की आवश्यकता नहीं: बार-बार
        वर्ग कीजिए ($F^2, F^4, F^8, \dots$) और जोड़ते जाइए। कितने आव्यूह-गुणन पर्याप्त हैं,
        और माध्यमिक विद्यालय खंड की कौन-सी प्राचीन गुणन-तरकीब यही है, अब
        \omterm{def:g12:matrix:matrix}{आव्यूहों} तक पदोन्नत?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- मौसम की मशीन।}
किसी नगर में: धूप वाले दिन के बाद अगला दिन \omterm{def:g10:proba:distribution}{प्रायिकता} $0.8$ से धूप वाला होता
है; वर्षा वाले दिन के बाद \omterm{def:g10:proba:distribution}{प्रायिकता} $0.4$ से धूप वाला। दिन के \omterm{def:g11:prob:rv}{बंटन} को स्तंभ
$\binom{p_{\text{धूप}}}{p_{\text{वर्षा}}}$ के रूप में और विकास को
\[
M = \begin{pmatrix} 0.8 & 0.4\\ 0.2 & 0.6 \end{pmatrix}.
\]
से लिखिए।
\begin{enumerate}[resume]
  \item जाँचिए कि $M$ के हर स्तंभ का योग $1$ है, और बताइए कि किसी भी
        मौसम-मशीन में यह गुण क्यों होना ही चाहिए।
  \item आज धूप है। कल और परसों का पूर्वानुमान निकालिए।
  \item \emph{स्थायी अवस्था} ज्ञात कीजिए: वह \omterm{def:g11:prob:rv}{बंटन} $v$ जिसके लिए $Mv = v$ हो
        (और जिसकी प्रविष्टियों का योग $1$ हो)। दीर्घ काल में कितने अंश दिन
        धूप वाले होते हैं?
  \item वर्षा वाले दिन $\binom01$ से शुरू कीजिए और $M$ चार बार लगाइए, और हर
        क़दम पर स्थायी अवस्था से दूरी दर्ज कीजिए। प्रति क़दम अंतर कितने गुना
        सिकुड़ता है --- और यह किस प्रकार का अभिसरण है?
  \item लघुरूप में पेजरैंक: तीन पृष्ठ, जिनकी कड़ियाँ $A \to B$, $A \to C$, $B \to C$,
        $C \to A$ हैं। कोई यादृच्छिक पाठक बाहर जाने वाली किसी कड़ी को एकसमान
        यादृच्छिक ढंग से पकड़ता है। संक्रमण \omterm{def:g12:matrix:matrix}{आव्यूह} लिखिए, स्थायी अवस्था ज्ञात
        कीजिए, और पृष्ठों की श्रेणी बनाइए।
  \item श्रेणी की व्याख्या कीजिए: कम पृष्ठों से कड़ियाँ पाने के बावजूद $C$
        $A$ जितना ऊँचा अंक क्यों पाता है --- स्थायी अवस्था असल में क्या
        नापती है? (असली पेजरैंक बंद गलियों और छलाँगों के लिए एक अवमंदन गुणक
        जोड़ता है; अभिलक्षणिक \omterm{def:g11:vect:vector}{सदिश} वाला विचार ठीक यही है।)
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- विकर्ण का लाभांश।}
\begin{enumerate}[resume]
  \item दो युग्मित राशियाँ $u_{n+1} = 3u_n + v_n$, $v_{n+1} = u_n + 3v_n$ मानती हैं, अर्थात् \cref{exo:g12:matrix:8} का \omterm{def:g12:matrix:matrix}{आव्यूह}
        $A$। उसी अभ्यास के विकर्णीकरण ($D = \operatorname{diag}(4, 2)$) का उपयोग करके $u_0 = 1$, $v_0 = 0$
        होने पर $u_n$ का बंद सूत्र दीजिए, और $n = 1, 2, 3$ के लिए उसे सीधी गणना से
        जाँचिए।
  \item एक-दो वाक्यों में: विकर्णीकरण किसी युग्मित तंत्र के साथ \emph{करता}
        क्या है --- और प्रश्न 14 की मार्कोव स्थायी अवस्था भी किस अर्थ में
        अभिलक्षणिक \omterm{def:g11:vect:vector}{सदिश} की ही कहानी है?
  \item समापन --- इस सप्ताहांत \omterm{def:g12:matrix:matrix}{आव्यूह} के तीन चेहरे: बही-खाता (निकाय और
        प्रतिलोम), क्रमचय-संचय (घातों से गिनी गई चालें और कड़ियाँ), और विकास
        (फिबोनाच्ची, मौसम, जाल --- अभिलक्षणिक दिशाओं से पढ़े गए भविष्य)। हर
        एक पर एक वाक्य, और आगे का संकेत: स्नातक खंडों का रैखिक बीजगणित इनमें
        से हर चेहरे को एक पूरा सिद्धांत बना देता है।
\end{enumerate}
\end{problem}
