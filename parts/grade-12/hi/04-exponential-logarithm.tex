\chapter{चरघातांकी फलन और लघुगणक}\label{ch:g12:exp}

\omterm{thm:g12:exp:existence}{चरघातांकी फलन} वह अकेला \omterm{def:g11:func:function}{फलन} है जो अपने ही \omterm{def:g12:deriv:derivative}{अवकलज} के बराबर है और $0$ पर मान $1$
लेता है। वह योगों को गुणनफलों में बदल देता है; और उसका प्रतिलोम, \omterm{def:g12:exp:ln}{प्राकृत
लघुगणक}, गुणनफलों को योगों में। दोनों मिलकर हर उस परिघटना का वर्णन करते हैं
जिसके परिवर्तन की दर उसके आकार के समानुपाती हो: रेडियोधर्मी क्षय, जनसंख्या
वृद्धि, चक्रवृद्धि ब्याज।

\section{चरघातांकी फलन}

\begin{theorem}[अस्तित्व और अद्वितीयता]\label{thm:g12:exp:existence}
ऐसा अकेला \omterm{def:g12:deriv:derivative}{अवकलनीय} \omterm{def:g11:func:function}{फलन} $f \colon \R \to \R$ है कि
\[
f' = f \qquad\text{और}\qquad f(0) = 1 .
\]
उसे \emph{चरघातांकी फलन}\index{चरघातांकी फलन} कहते हैं और $\exp$ या $x \mapsto \eu^x$ लिखते
हैं।
\end{theorem}

\begin{proof}[अद्वितीयता की उपपत्ति]
पहले, ऐसा \omterm{def:g11:func:function}{फलन} कभी लुप्त नहीं होता। सचमुच, $g(x) = f(x)f(-x)$ लीजिए; तब
\[
g'(x) = f'(x)f(-x) - f(x)f'(-x) = f(x)f(-x) - f(x)f(-x) = 0,
\]
इसलिए $g$ अचर है और $g(0) = 1$ के बराबर: हर $x$ के लिए $f(x)f(-x) = 1$, और विशेष रूप से
$f(x) \neq 0$।

अब मान लीजिए $f_1, f_2$ दो हल हैं और $h = \dfrac{f_1}{f_2}$ है (यह वैध है क्योंकि $f_2$ कभी लुप्त
नहीं होता)। तब
\[
h' = \frac{f_1' f_2 - f_1 f_2'}{f_2^2} = \frac{f_1 f_2 - f_1 f_2}{f_2^2} = 0,
\]
इसलिए $h$ अचर है और $h(0) = 1$ के बराबर, \emph{अर्थात्} $f_1 = f_2$।

\emph{अस्तित्व} इस स्तर पर स्वीकार कर लिया जाता है (उसे \cref{exo:g12:seq:10} के \omterm{def:g12:seq:sequence}{अनुक्रम} से,
या \cref{ch:g12:integ} में समाकलन से बने \omterm{def:g12:exp:ln}{लघुगणक} के प्रतिलोम के रूप में प्राप्त किया जा सकता
है)।
\end{proof}

\begin{proposition}[फलनीय समीकरण]\label{prop:g12:exp:funceq}
सभी $x, y \in \R$ और $n \in \Z$ के लिए:
\[
\eu^{x+y} = \eu^x\,\eu^y, \qquad
\eu^{-x} = \frac{1}{\eu^x}, \qquad
\eu^{x-y} = \frac{\eu^x}{\eu^y}, \qquad
\bigl(\eu^{x}\bigr)^n = \eu^{nx}.
\]
इसके अतिरिक्त सभी $x$ के लिए $\eu^x > 0$।
\end{proposition}

\begin{proof}
$y$ स्थिर रखिए और $\varphi(x) = \dfrac{\exp(x+y)}{\exp(x)\exp(y)}$ पर विचार कीजिए। $x$ के \omterm{def:g10:functions:function}{फलनों} के रूप में उसका अंश और
हर दोनों अचर $\exp(y)$ तक $f' = f$ के हल हैं; $\varphi$ का सीधे अवकलन (भागफल के नियम से)
$\varphi' = 0$ देता है, इसलिए $\varphi \equiv \varphi(0) = 1$, जिससे पहली सर्वसमिका सिद्ध हो जाती है। $y = -x$ लेने पर
दूसरी मिलती है (मान $\eu^0 = 1$ के साथ), और तीसरी उससे निकल आती है। चौथी $n \geq 0$ के लिए
पहली से आगमन है, जिसे दूसरी की सहायता से $n < 0$ तक बढ़ा दिया जाता है।

धनात्मकता: $\eu^x = \left(\eu^{x/2}\right)^2 \geq 0$ और $\eu^x \neq 0$ (\cref{thm:g12:exp:existence} में दिखाया गया), इसलिए $\eu^x > 0$।
\end{proof}

\begin{proposition}[परिवर्तन और सीमाएँ]\label{prop:g12:exp:variations}
\omterm{thm:g12:exp:existence}{चरघातांकी फलन} निरंतर \omterm{def:g12:seq:monotonic}{वर्धमान} और \omterm{def:g12:deriv:convex}{उत्तल} है, तथा
\[
\lim_{x\to-\infty} \eu^x = 0, \qquad
\lim_{x\to+\infty} \eu^x = +\infty .
\]
\end{proposition}

\begin{proof}
$(\exp)' = \exp > 0$ से निरंतर वृद्धि मिलती है; $(\exp)'' = \exp > 0$ से \omterm{def:g12:deriv:convex}{उत्तलता}। \omterm{def:g12:deriv:convex}{उत्तलता} से $\eu^x \geq 1 + x$ ($0$ पर
\omterm{def:g12:deriv:tangent}{स्पर्श रेखा}, \cref{ex:g12:deriv:ineq} देखिए), इसलिए तुलना से $x \to +\infty$ पर $\eu^x \to +\infty$। फिर $x \to -\infty$ पर $\eu^{x} = 1/\eu^{-x} \to 0$।
\end{proof}

\begin{theorem}[वृद्धि की तुलना]\label{thm:g12:exp:growth}
\index{वृद्धि की तुलना}
हर \omterm{def:g10:numbers:sets}{पूर्णांक} $n \geq 1$ के लिए
\[
\lim_{x\to+\infty} \frac{\eu^x}{x^n} = +\infty,
\qquad
\lim_{x\to-\infty} x^n \eu^x = 0 .
\]
शब्दों में: \omterm{thm:g12:exp:existence}{चरघातांकी फलन} $x$ की हर घात को हरा देता है।
\end{theorem}

\begin{proof}
$n = 1$ के लिए: $t = x/2$ पर $\eu^t \geq 1 + t \geq t$ लगाने पर
\[
\frac{\eu^x}{x} = \frac{\left(\eu^{x/2}\right)^2}{x}
\geq \frac{(x/2)^2}{x} = \frac{x}{4} \xrightarrow[x\to+\infty]{} +\infty .
\]
सामान्य $n$ के लिए लिखिए
\[
\frac{\eu^x}{x^n}
= \frac{1}{n^n}\left(\frac{\eu^{x/n}}{x/n}\right)^{n}
\xrightarrow[x\to+\infty]{} +\infty
\]
जो $n=1$ वाली स्थिति और संयोजन से निकलता है। $-\infty$ पर सीमा प्रतिस्थापन $x \mapsto -x$ से
मिलती है: $\abs{x^n \eu^x} = \frac{\abs{x}^n}{\eu^{\abs x}} \to 0$।
\end{proof}

\section{प्राकृत लघुगणक}

\begin{definition}[प्राकृत लघुगणक]\label{def:g12:exp:ln}
\omterm{thm:g12:exp:existence}{चरघातांकी फलन} $\R$ से $\intoo{0}{+\infty}$ पर \omterm{def:g12:limcont:continuity}{संतत} और निरंतर \omterm{def:g12:seq:monotonic}{वर्धमान} है; एकैकी आच्छादन प्रमेय
(\cref{thm:g12:limcont:bijection}) से हर $y > 0$ के लिए \omterm{def:g10:algebra:equation}{समीकरण} $\eu^x = y$ का अद्वितीय हल है। यही हल $y$ का
\emph{प्राकृत लघुगणक}\index{लघुगणक} है, जिसे $\ln y$ लिखते हैं। अतः
\[
x \in \R,\ y > 0 \text{ के लिए}: \qquad y = \eu^x \iff x = \ln y .
\]
विशेष रूप से $\ln 1 = 0$, $\ln \eu = 1$, तथा $\eu^{\ln y} = y$, $\ln(\eu^x) = x$।
\end{definition}

\begin{proposition}[बीजगणितीय गुण]\label{prop:g12:exp:lnalg}
सभी $a, b > 0$ और $n \in \Z$ के लिए:
\[
\ln(ab) = \ln a + \ln b, \quad
\ln\frac{1}{a} = -\ln a, \quad
\ln\frac{a}{b} = \ln a - \ln b, \quad
\ln(a^n) = n \ln a, \quad
\ln\sqrt{a} = \tfrac12 \ln a .
\]
\end{proposition}

\begin{proof}
$\eu^{\ln a + \ln b} = \eu^{\ln a}\eu^{\ln b} = ab$, और दोनों पक्षों का $\ln$ लेने पर पहली सर्वसमिका मिल जाती है। शेष \cref{prop:g12:exp:funceq}
की संगत सर्वसमिकाओं से उसी तंत्र द्वारा निकलती हैं।
\end{proof}

\begin{proposition}[विश्लेषणात्मक गुण]\label{prop:g12:exp:lnanalytic}
\omterm{def:g11:func:function}{फलन} $\ln$ $\intoo{0}{+\infty}$ पर \omterm{def:g12:deriv:derivative}{अवकलनीय} है और
\[
(\ln x)' = \frac{1}{x},
\]
वह निरंतर \omterm{def:g12:seq:monotonic}{वर्धमान} और \omterm{def:g12:deriv:convex}{अवतल} है, तथा $\lim\limits_{x\to0^+} \ln x = -\infty$, $\lim\limits_{x\to+\infty} \ln x = +\infty$। इसके अतिरिक्त, हर \omterm{def:g10:numbers:sets}{पूर्णांक}
$n \geq 1$ के लिए
\[
\lim_{x\to+\infty} \frac{\ln x}{x^{1/n}} = 0, \qquad
\lim_{x\to0^+} x \ln x = 0 .
\]
\end{proposition}

\begin{proof}
प्रतिलोम \omterm{def:g11:func:function}{फलन} की अवकलनीयता इस स्तर पर स्वीकार कर ली जाती है; उसे मानकर
सर्वसमिका $\eu^{\ln x} = x$ का शृंखला नियम से अवकलन कीजिए: $(\ln x)'\,\eu^{\ln x} = 1$, इसलिए $(\ln x)' = \frac{1}{x} > 0$, जिससे
निरंतर वृद्धि मिलती है, और $(\ln x)'' = -\frac1{x^2} < 0$, जिससे अवतलता। $0^+$ और $+\infty$ पर सीमाएँ
एकैकी आच्छादन के आरपार $\exp$ की सीमाओं का दर्पण हैं। \omterm{thm:g12:exp:growth}{वृद्धि की तुलना} के लिए
$x = \eu^{t}$ प्रतिस्थापित कीजिए: \cref{thm:g12:exp:growth} से $t \to +\infty$ पर $\frac{\ln x}{x} = \frac{t}{\eu^t} \to 0$, और $x = \eu^{-t}$ के साथ उसी तरह
शेष सीमाओं के लिए, जिससे $x\ln x = -t\eu^{-t} \to 0$ मिलता है।
\end{proof}

\begin{method}[$\exp$ और $\ln$ वाले समीकरण हल करना]\label{met:g12:exp:equations}
दोनों \omterm{def:g11:func:function}{फलन} निरंतर \omterm{def:g12:seq:monotonic}{वर्धमान} हैं, इसलिए उन्हें किसी \omterm{def:g10:algebra:equation}{समीकरण} या असमिका के दोनों
पक्षों पर लगाया (या हटाया) जा सकता है और दिशा नहीं बदलती:
\[
\eu^{u} = \eu^{v} \iff u = v, \qquad
\eu^{u} \leq \eu^{v} \iff u \leq v,
\]
और धनात्मक राशियों पर $\ln$ के लिए भी वैसे ही। \emph{\omterm{def:g11:func:function}{प्रांत} सदा पहले जाँचिए}
($\ln$ धनात्मक तर्क माँगता है)। $a>0$, $a \neq 1$ वाले $a^x = b$ जैसे \omterm{def:g10:algebra:equation}{समीकरणों} के लिए
$a^x = \eu^{x\ln a}$ फिर से लिखिए और $x = \frac{\ln b}{\ln a}$ हल कीजिए।
\end{method}

\begin{example}\label{ex:g12:exp:halflife}
कोई रेडियोधर्मी \omterm{def:g10:proba:sample}{प्रतिदर्श} $N(t) = N_0\,\eu^{-\lambda t}$ के अनुसार क्षय होता है। उसकी \emph{अर्धायु}
$T$ $N(T) = N_0/2$ को संतुष्ट करती है, \emph{अर्थात्} $\eu^{-\lambda T} = \frac12$, इसलिए $T = \frac{\ln 2}{\lambda}$: अर्धायु
आरंभिक मात्रा पर निर्भर नहीं करती।
\end{example}

\begin{exercise}[$\star$]\label{exo:g12:exp:2}

\text{(a) } \eu^{2x} - 3\eu^{x} + 2 = 0;
\qquad

\begin{exercise}[$\star$]\label{exo:g12:exp:3}

\lim_{x\to+\infty} \frac{\eu^x - x^2}{\eu^x + 1}, \qquad
\lim_{x\to+\infty} \frac{\ln x}{\sqrt x}, \qquad
\lim_{x\to0^+} x^2 \ln x, \qquad
\lim_{x\to+\infty} \bigl(x - \ln x\bigr).

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:4}
$\R$ पर \omterm{def:g11:func:function}{फलन} $f(x) = x\,\eu^{-x}$ का अध्ययन कीजिए: परिवर्तन, सीमाएँ, \omterm{def:g12:deriv:extremum}{चरम मान}; और दिखाइए कि
उसके वक्र का एक \omterm{def:g12:deriv:inflection}{नति-परिवर्तन बिंदु} है तथा उसके \omterm{def:g10:coordgeom:system}{निर्देशांक} बताइए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:5}
दिखाइए कि सभी $x > -1$ के लिए $\ln(1 + x) \leq x$ है, और समता केवल $x = 0$ पर। इससे निकालिए कि
सभी $n \geq 1$ के लिए
\[
\left(1 + \frac{1}{n}\right)^n \leq \eu \leq \left(1 - \frac{1}{n+1}\right)^{-(n+1)} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:6}
कोई पूँजी $C_0$ वार्षिक दर $3\%$ पर लगाई जाती है, और ब्याज हर वर्ष जुड़ता है।
\begin{enumerate}
  \item $n$ वर्षों बाद की पूँजी $C_n$ व्यक्त कीजिए।
  \item कितने वर्षों में पूँजी दुगुनी हो जाती है? $\ln$ का उपयोग करके यथार्थ
        उत्तर, फिर संख्यात्मक मान बताइए।
  \item बैंकर जिस \omterm{def:g10:numbers:approx}{सन्निकटन} का उपयोग करते हैं --- ``$70$ भाग दर प्रतिशत में''
        --- उससे तुलना कीजिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:7}
असमिका $\eu^{2x} - \eu^{x+1} > 0$ हल कीजिए, फिर असमिका $\ln(x^2 - 1) \leq \ln(x + 5)$।
\end{exercise}

\begin{enumerate}
  \item $\intoo{0}{+\infty}$ पर $f$ के परिवर्तन का अध्ययन कीजिए और उसका \omterm{def:g10:functions:extrema}{न्यूनतम} बताइए।
  \item $k$ के किन मानों के लिए \omterm{def:g10:algebra:equation}{समीकरण} $\eu^x = kx$ के $\intoo{0}{+\infty}$ में $0$, $1$ या
        $2$ हल हैं?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:exp:9}
$n \geq 1$ के लिए मान लीजिए $u_n = \left(1 + \frac1n\right)^n$ है।
\begin{enumerate}
  \item \cref{exo:g12:exp:5} का उपयोग करके दिखाइए कि $(u_n)$ ऊपर से~$\eu$ द्वारा \omterm{def:g12:seq:bounded}{परिबद्ध} है।
  \item दिखाइए कि $\ln u_n = n \ln\left(1 + \frac1n\right) \to 1$, और इससे $u_n \to \eu$ निकालिए। (संकेत: $1$ पर $\ln$ का
        कोई अंतर-भागफल पहचानिए।)
\end{enumerate}
\end{exercise}

\section{समस्या: लघुगणक संसार को साध लेता है}

\begin{problem}[{सप्ताहांत समस्या --- दुगुना होने के समय और 72 का नियम, भूकंप,
अम्ल तथा पियानो के मापक्रम, और अचर $\eu$ जो हर जगह अपने निशान छोड़ जाता है}]
\label{pb:g12:exp:1}
जो कुछ भी नियत \emph{प्रतिशत} से बढ़ता है वह चरघातांकी रूप से बढ़ता है ---
बचत, जीवाणु, महामारियाँ --- और जो कुछ भी दस की इतनी घातों में फैला हो कि पकड़
में न आए --- भूकंप की ऊर्जा, अम्लता, ध्वनि की तीव्रता --- उसे \omterm{def:g12:exp:ln}{लघुगणक} साध लेता
है। यह समस्या दुगुना होने के समय निकालती है और बैंकरों के 72 वाले नियम का
पर्दाफ़ाश करती है, संसार के लघुगणकीय मापक्रम पढ़ती है, और वे निशान इकट्ठे करती
है जो संख्या $\eu$ हर वारदात की जगह छोड़ आती है (\cref{prop:g12:exp:funceq}, \cref{thm:g12:exp:growth}, \cref{met:g12:exp:equations})।

\medskip
\noindent\textbf{भाग I --- प्रवाह।}
\begin{enumerate}
  \item हल कीजिए: $\eu^{2x} = 5$; \quad $\ln(3x - 1) = 2$; \quad $\eu^{2x} - 3\eu^x + 2 = 0$ (भेस बदला हुआ द्विघात)।
  \item सरल कीजिए: $\dfrac{\ln 8}{\ln 2}$; \quad $\ln\!\left(\eu^3 \sqrt{\eu}\right)$; \quad $\eu^{\ln 5 - \ln 2}$.
  \item \omterm{def:g11:quad:discriminant}{मूल} असमिका सिद्ध कीजिए: सभी वास्तविक $x$ के लिए $\eu^x \geq 1 + x$ ($f(x) = \eu^x - 1 - x$ का
        अध्ययन कीजिए), और $u > -1$ के लिए $\ln(1 + u) \leq u$ निकालिए।
  \item $\lim_{x \to +\infty} x^2 \eu^{-x}$, $\lim_{x \to +\infty} \frac{\ln x}{x}$ (\cref{thm:g12:exp:growth}) और $\lim_{x \to 0} \frac{\eu^x - 1}{x}$ (कोई \omterm{def:g12:deriv:derivative}{अवकलज} पहचानिए) निकालिए।
  \item $\intoo{0}{+\infty}$ पर $f(x) = x \ln x$ का अध्ययन कीजिए: परिवर्तन, \omterm{def:g10:functions:extrema}{न्यूनतम}, और $0^+$ पर सीमा
        (स्वीकृत: $x \ln x \to 0$)। यह छोटा-सा \omterm{def:g11:func:function}{फलन} स्नातक खंडों में सूचना और
        एन्ट्रॉपी नापता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दुगुना होने के समय और 72 का नियम।}
\begin{enumerate}[resume]
  \item बचत प्रति वर्ष $3\,\%$ से बढ़ती है। $1.03^n = 2$ हल कीजिए: पूँजी दुगुनी होने
        में कितना समय लगता है?
  \item बैंकर दुगुना होने का समय $\frac{72}{\text{\% में दर}}$ से आँकते हैं। इस नियम की जाँच $3\,\%$,
        $6\,\%$ और $9\,\%$ पर यथार्थ $\frac{\ln 2}{\ln(1 + r)}$ के सामने कीजिए, फिर उसे समझाइए: छोटे
        $r$ के लिए $\ln(1 + r) \approx r$ (प्रश्न 3 की असमिका आधी कहानी है), इसलिए यथार्थ
        अचर $100 \ln 2 \approx 69.3$ है --- तो बैंकर $72$ क्यों पसंद करते हैं?
  \item कोई जीवाणु हर $20$ मिनट में विभाजित होता है: $p(t) = 2^{t/20}$ ($t$ मिनट में)।
        उसे $\eu^{\lambda t}$ के रूप में फिर से लिखिए, फिर $24$ घंटों बाद $p$
        निकालिए। उत्तर ($10^{21}$ से भी अधिक) प्रतिरूप के बारे में क्या सिद्ध
        करता है --- और असली उपनिवेशों को कौन रोकता है?
  \item कैफ़ीन शरीर से लगभग $5$ घंटे की अर्धायु के साथ निकलता है। 15:00
        बजे $100$ mg वाली कॉफ़ी पीने के बाद 23:00 बजे कितना बचता है? वह किस
        समय $10$ mg से नीचे गिर जाता है? (अनिद्रा का भी एक \omterm{def:g12:exp:ln}{लघुगणक} है।)
  \item $100\,\%$ वार्षिक ब्याज पर एक यूरो: वार्षिक, मासिक और दैनिक चक्रवृद्धि
        के साथ वर्ष के अंत की पूँजी निकालिए, और वह छत बताइए जिसे \cref{exo:g12:exp:9} ने
        अटूट सिद्ध किया था। शुद्ध लोभ की सीमा कौन-सा प्रसिद्ध अचर है?
  \item किसी महामारी के आरंभिक चरण में वैज्ञानिक समय के सामने $\ln(\text{स्थितियाँ})$ का \omterm{def:g10:functions:graph}{आलेख}
        क्यों खींचते हैं? उस \omterm{def:g10:functions:graph}{आलेख} पर सीधी रेखा क्या उजागर करती है, और उसकी
        \omterm{def:g10:reffunc:affine}{प्रवणता} क्या नापती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- संसार के लघुगणकीय मापक्रम।}
($\log_{10} x = \frac{\ln x}{\ln 10}$ लिखिए।)
\begin{enumerate}[resume]
  \item डेसिबल में ध्वनि-स्तर: $L = 10 \log_{10}(I/I_0)$। बातचीत $60$ dB और कोई रॉक-संगीत
        समारोह $120$ dB नापता है: दोनों की ध्वनि-तीव्रताओं में कितने गुने
        का अंतर है?
  \item भूकंपीय आयाम $10$ गुना होने पर भूकंप का परिमाण $1$ बढ़ जाता है,
        और मुक्त ऊर्जा आयाम$^{3/2}$ की तरह बढ़ती है। परिमाण-$5$ और
        परिमाण-$7$ के भूकंपों की तुलना कीजिए: पहले आयाम का अनुपात, फिर
        ऊर्जा का।
  \item रसायन: $\text{pH} = -\log_{10}[\mathrm{H^+}]$। नींबू के रस का pH $2$ है और दूध का pH $6.5$: उनकी
        अम्ल-सांद्रताओं का अनुपात क्या है?
  \item संगीत: हर सप्तक आवृत्ति को दुगुना कर देता है। पियानो के सबसे नीचे
        वाले A ($27.5$ Hz) से सबसे ऊपर वाले C ($4\,186$ Hz) तक कुंजी-पटल कितने
        सप्तक में फैला है? और हमारी इंद्रियाँ --- श्रवण, दृष्टि, भूकंप का
        अनुभव --- लघुगणकीय मापक्रम क्यों पसंद करती हैं? (एक वाक्य।)
  \item गणक-पट्टी, जो $350$ वर्षों तक अभियंताओं का कैलकुलेटर रही: दो
        पट्टियाँ ऐसे अंकित कि चिह्न $x$ तक की \emph{लंबाई} $\ln x$ के
        समानुपाती हो। समझाइए कि एक पट्टी को दूसरी पर सरकाने से संख्याएँ कैसे
        गुणा हो जाती हैं, और \cref{prop:g12:exp:lnalg} की वह सर्वसमिका बताइए जो यह काम करती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- $\eu$ के निशान।}
\begin{enumerate}[resume]
  \item \cref{exo:g12:exp:8} से: \omterm{def:g10:algebra:equation}{समीकरण} $\eu^x = kx$ ($x > 0$) के $0$, $1$ या $2$ हल होते हैं,
        जो $k$ की किसी सीमा के सापेक्ष स्थिति पर निर्भर करता है। परिणाम
        फिर से बताइए --- और उसके पीछे का ज्यामितीय तथ्य जाँचिए: रेखा $y = \eu x$
        ठीक वही है जो मूल बिंदु से होकर \omterm{thm:g12:exp:existence}{चरघातांकी फलन} को स्पर्श करती है।
  \item ``बस चूक गए'' वाला अचर: जीतने की \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac1n$ वाले $n$ लॉटरी
        टिकटों के साथ $\P(\text{कोई जीत नहीं}) = \left(1 - \frac1n\right)^n$। $n \ln\!\left(1 - \frac1n\right)$ से (मानक सीमा $\frac{\ln(1 + u)}{u} \to 1$ का उपयोग करके) उसकी
        सीमा निकालिए, और \cref{pb:g11:binom:1} के रहस्यमय $0.368$ से मिलाइए।
  \item $37\,\%$ का नियम: यादृच्छिक क्रम में साक्षात्कार किए जा रहे $n$
        उम्मीदवारों में से सर्वोत्तम चुनने के लिए (लौटना मना है) इष्टतम
        रणनीति पहले $\frac{n}{\eu} \approx 37\,\%$ को अस्वीकार कर देती है और फिर पहले ऐसे उम्मीदवार
        को ले लेती है जो अब तक के सबसे अच्छे से भी अच्छा हो --- और उसकी
        सफलता की \omterm{def:g10:proba:distribution}{प्रायिकता} लगभग $\frac1\eu$ होती है। प्रश्न 18 और 19 के दोनों
        $\frac1\eu$ कहाँ से आते हैं --- साझा तंत्र बताइए (अनेक स्वतंत्र, छोटी
        संभावना वाली \omterm{def:g10:proba:events}{घटनाएँ}), और $\frac1\eu$ तीन दशमलव स्थानों तक निकालिए।
  \item समापन --- $\eu$ का चित्र: चक्रवृद्धि की छत (प्रश्न 10); वह आधार
        जिसकी $0$ पर \omterm{def:g12:deriv:tangent}{स्पर्श रेखा} की \omterm{def:g10:reffunc:affine}{प्रवणता} ठीक $1$ है (प्रश्न 3 और 4);
        वह वृद्धि जिसे कोई बहुपद नहीं पकड़ पाता (प्रश्न 4); ``बस चूक गए''
        और इष्टतम ठहराव का अचर (प्रश्न 18--19); और उसका प्रतिलोम $\ln$, जो
        गुणनफलों को योगों में (प्रश्न 16) और दशकों को इंचों में (भाग III)
        बदल देता है। हर एक पर एक वाक्य।
\end{enumerate}
\end{problem}
