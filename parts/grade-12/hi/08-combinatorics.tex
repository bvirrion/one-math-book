\chapter{क्रमचय-संचय और गणना}\label{ch:g12:comb}

क्रमचय-संचय बिना सूची बनाए गिनने की कला है। उसके दो प्रारंभिक सिद्धांत ---
असंयुक्त विकल्पों के आकार जोड़ दीजिए, और स्वतंत्र चुनावों की संख्याएँ गुणा कर
दीजिए --- किसी परिमित समुच्चय के \omterm{def:g12:comb:tuples}{विन्यास}, \omterm{def:g12:comb:permutation}{क्रमचय} और उपसमुच्चय गिनने के लिए
पर्याप्त हैं, और द्विपद प्रमेय पर जाकर अपनी चरम सीमा छूते हैं।

\section{गणना के दो सिद्धांत}

किसी परिमित समुच्चय $E$ के अवयवों की संख्या (\emph{गणनांक}) को हम $\abs{E}$ लिखते
हैं।

\begin{proposition}[योग का सिद्धांत]\label{prop:g12:comb:addition}
यदि किसी परिमित समुच्चय $E$ को उपसमुच्चयों $A_1, \dots, A_k$ में विभाजित किया जाए (जो
जोड़े-जोड़े असंयुक्त हों और जिनका \omterm{def:g10:numbers:interunion}{सम्मिलन} $E$ हो), तो
\[
\abs{E} = \abs{A_1} + \abs{A_2} + \dots + \abs{A_k}.
\]
\end{proposition}

\begin{proposition}[गुणन का सिद्धांत]\label{prop:g12:comb:multiplication}
यदि कोई वस्तु $k$ क्रमिक चुनावों से बनती हो, जिनमें पहले चुनाव के लिए $n_1$
विकल्प हों और, \emph{पिछले चुनाव चाहे जो हों}, $i$-वें के लिए $n_i$ विकल्प
हों, तो बनी हुई वस्तुओं की संख्या $n_1 \times n_2 \times \dots \times n_k$ है।
\end{proposition}

\begin{proof}
दोनों कथन $k$ पर आगमन से सिद्ध होते हैं; दूसरे की स्थिति $k = 2$ वस्तुतः किसी
आयताकार सारणी को पंक्ति-दर-पंक्ति गिनना ही है।
\end{proof}

\begin{example}
किसी भोजनालय में 4 प्रारंभिक व्यंजन, 6 मुख्य व्यंजन और 3 मिष्ठान्न मिलते हैं:
इस तरह $4 \times 6 \times 3 =
72$ भिन्न तीन-भागी भोजन बनते हैं।
\end{example}

\section{सूचियाँ, क्रमचय, क्रमगुणित}

\begin{definition}[$k$-सूचियाँ]\label{def:g12:comb:tuples}
किसी समुच्चय $E$ की \emph{$k$-सूची} $E$ के अवयवों की क्रमित सूची $(x_1, \dots, x_k)$ है,
जिसमें पुनरावृत्ति की छूट है। \emph{भिन्न} अवयवों वाली $k$-सूची $E$ के $k$
अवयवों का \emph{विन्यास}\index{विन्यास} कहलाती है।
\end{definition}

\begin{proposition}\label{prop:g12:comb:tuples}
मान लीजिए $\abs E = n$ है। $E$ की $k$-सूचियों की संख्या $n^k$ है। $E$ के $k$
अवयवों के \omterm{def:g12:comb:tuples}{विन्यासों} की संख्या ($0 \leq k \leq n$) है
\[
n(n-1)(n-2)\cdots(n-k+1) = \frac{n!}{(n-k)!},
\]
जहाँ $n! = 1 \times 2 \times \dots \times n$ (और $0! = 1$) $n$ का \emph{क्रमगुणित}\index{क्रमगुणित} है।
\end{proposition}

\begin{proof}
गुणन का सिद्धांत: $k$-सूची के लिए $k$ चरणों में से हर एक पर $n$ विकल्प
हैं; और \omterm{def:g12:comb:tuples}{विन्यास} के लिए $x_1$ हेतु $n$ विकल्प, फिर $x_2$ हेतु $n - 1$ (एक अवयव
काम में आ चुका), \dots, $x_k$ हेतु $n - k + 1$।
\end{proof}

\begin{definition}[क्रमचय]\label{def:g12:comb:permutation}
$E$ का \emph{क्रमचय}\index{क्रमचय} $E$ के सभी $n$ अवयवों का \omterm{def:g12:comb:tuples}{विन्यास} है:
अर्थात् $E$ का एक क्रम। \cref{prop:g12:comb:tuples} (स्थिति $k = n$) से $n$-अवयवी समुच्चय के
क्रमचयों की संख्या $n!$ है।
\end{definition}

\begin{example}
पाँच धावक किसी दौड़ को $5! = 120$ भिन्न क्रमों में पूरा कर सकते हैं। संभव मंच
(पहले तीन स्थान) की संख्या $5 \times 4 \times 3 = 60$ है।
\end{example}

\section{संचय और द्विपद गुणांक}

\begin{definition}[संचय]\label{def:g12:comb:combination}
$E$ के $k$ अवयवों का \emph{संचय}\index{संचय} $E$ का वह उपसमुच्चय है जिसमें
$k$ अवयव हों (न क्रम, न पुनरावृत्ति)। उनकी संख्या $\dbinom{n}{k}$\index{द्विपद गुणांक}
लिखी जाती है और ``$k$ में से $n$ चुनना'' पढ़ी जाती है।
\end{definition}

\begin{theorem}\label{thm:g12:comb:binomformula}
$0 \leq k \leq n$ के लिए:
\[
\binom{n}{k} = \frac{n!}{k!\,(n-k)!} .
\]
\end{theorem}

\begin{proof}
$E$ के $k$ अवयवों के \omterm{def:g12:comb:tuples}{विन्यास} दो तरह से गिनिए। सीधे: $\frac{n!}{(n-k)!}$। वैकल्पिक रूप से,
पहले नीचे का उपसमुच्चय चुनिए ($\binom nk$ तरीक़े), फिर उसे क्रम में लगाइए ($k!$
तरीक़े); गुणन का सिद्धांत $\binom{n}{k}\,k!$ देता है। दोनों को बराबर रखने पर $\binom nk = \frac{n!}{k!(n-k)!}$।
\end{proof}

\begin{proposition}[बुनियादी सर्वसमिकाएँ]\label{prop:g12:comb:identities}
$0 \leq k \leq n$ के लिए:
\[
\binom{n}{0} = \binom{n}{n} = 1, \qquad
\binom{n}{1} = n, \qquad
\binom{n}{k} = \binom{n}{n-k},
\]
और \emph{पास्कल का नियम}\index{पास्कल त्रिभुज}: $1 \leq k \leq n-1$ के लिए
\[
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}.
\]
\end{proposition}

\begin{proof}
सममिति $\binom nk = \binom{n}{n-k}$ इसलिए सही है कि \omterm{def:g10:proba:operations}{पूरक} लेने पर $k$-अवयवी उपसमुच्चय $(n-k)$-अवयवी
उपसमुच्चयों से एक-एक करके मिल जाते हैं। पास्कल के नियम के लिए कोई अवयव $a \in E$
नियत कीजिए और $k$-अवयवी उपसमुच्चयों को दो भागों में बाँटिए: वे जिनमें $a$
है --- जो $E \setminus \{a\}$ के किसी $(k-1)$-अवयवी उपसमुच्चय में $a$ जोड़ने से मिलते हैं और
जिनकी संख्या $\binom{n-1}{k-1}$ है --- और वे जिनमें $a$ नहीं है, जो $E \setminus \{a\}$ के $k$-अवयवी
उपसमुच्चय हैं और जिनकी संख्या $\binom{n-1}{k}$ है। योग के सिद्धांत से निष्कर्ष निकल आता
है।
\end{proof}

\omterm{prop:g12:comb:identities}{पास्कल का नियम} गुणांकों को पंक्ति-दर-पंक्ति बना देता है --- \emph{\omterm{prop:g11:binom:pascal}{पास्कल
त्रिभुज}}: हर प्रविष्टि अपने ऊपर की दो प्रविष्टियों का योग है।

\begin{theorem}[द्विपद प्रमेय]\label{thm:g12:comb:binomial}
\index{द्विपद प्रमेय}
सभी $a, b \in \R$ (या $\C$) और $n \in \N$ के लिए:
\[
(a+b)^n = \sum_{k=0}^{n} \binom{n}{k}\, a^{k}\, b^{\,n-k} .
\]
\end{theorem}

\begin{proof}
गुणनफल $(a+b)(a+b)\cdots(a+b)$ ($n$ गुणनखंड) का प्रसार कीजिए: प्रसार का हर पद हर गुणनखंड में
से $a$ या $b$ उठाता है, जिससे $a^k b^{n-k}$ बनता है, जहाँ $k$ $a$ देने वाले
गुणनखंडों की संख्या है। $n$ में से ये $k$ गुणनखंड चुनने के तरीक़ों की
संख्या $\binom nk$ है, और इसलिए वही $a^k b^{n-k}$ का गुणांक है।
\end{proof}

\begin{corollary}\label{cor:g12:comb:sums}
$\displaystyle\sum_{k=0}^{n} \binom{n}{k} = 2^n$
\quad और \quad
$\displaystyle\sum_{k=0}^{n} (-1)^k\binom{n}{k} = 0$ \ ($n \geq 1$)।
\end{corollary}

\begin{proof}
द्विपद प्रमेय में $a = b = 1$, फिर $a = -1$, $b = 1$ लीजिए। पहली सर्वसमिका का सीधा अर्थ
भी है: $n$-अवयवी समुच्चय के $2^n$ उपसमुच्चय होते हैं (हर अवयव भीतर या
बाहर: गुणन का सिद्धांत), जिन्हें आकार के अनुसार छाँट दिया गया है।
\end{proof}

\begin{method}[सही प्रतिरूप चुनना]\label{met:g12:comb:model}
गिनने से पहले दो प्रश्नों का उत्तर दीजिए: \emph{क्या क्रम मायने रखता है?} और
\emph{क्या पुनरावृत्ति की छूट है?}
\begin{center}
\begin{tabular}{l|cc}
 & क्रम मायने रखता है & क्रम अप्रासंगिक\\
\hline
पुनरावृत्ति की छूट & $n^k$ (सूचियाँ) & (स्नातक स्तर)\\
पुनरावृत्ति नहीं & $\frac{n!}{(n-k)!}$ (\omterm{def:g12:comb:tuples}{विन्यास}) & $\binom nk$ (उपसमुच्चय)
\end{tabular}
\end{center}
कलश से गेंदें निकालना: \emph{वापसी सहित, क्रम में} $\to$ सूचियाँ; \emph{बिना
वापसी, क्रम में} $\to$ \omterm{def:g12:comb:tuples}{विन्यास}; \emph{एक ही बार में मुट्ठी भर} $\to$
उपसमुच्चय।
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g12:comb:1}
किसी वाहन-पट्टिका पर 2 अक्षर ($A$--$Z$), फिर 3 अंक, फिर 2 अक्षर होते हैं।
कितनी पट्टिकाएँ संभव हैं? कितनी ऐसी हैं जिनमें कोई भी चिह्न दोहराया न गया हो?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:2}
$\dbinom{8}{3}$, $\dbinom{10}{8}$ निकालिए, और $\dfrac{\binom{n}{2}}{\binom{n+1}{2}}$ को सरल कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:3}
30 विद्यार्थियों की कक्षा में 4 विद्यार्थियों की एक समिति चुननी है, और फिर
समिति के भीतर एक अध्यक्ष तथा एक कोषाध्यक्ष (एक ही व्यक्ति दोनों पद नहीं ले
सकता)। कितने परिणाम संभव हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:4}
द्विपद प्रमेय से $(x + 2)^5$ और $(1 - x)^6$ का प्रसार कीजिए। $(2x + 3)^7$ में $x^3$ का गुणांक
क्या है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:5}
ताश के 52 पत्तों की गड्डी से मानक पोकर का एक हाथ 5 पत्तों का होता है।
\begin{enumerate}
  \item कुल कितने हाथ बनते हैं?
  \item कितने हाथों में ठीक एक इक्का होता है? कम से कम एक इक्का?
  \item कितने हाथ ``फ़ुल हाउस'' होते हैं (एक कोटि के तीन पत्ते और दूसरी कोटि
        के दो)?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:6}
शब्द $\mathrm{MATH}$ के कितने अक्षर-विन्यास (अक्षरों की पुनर्व्यवस्थाएँ,
अर्थपूर्ण हों या न हों) बनते हैं? और शब्द $\mathrm{BANANA}$ के? ($\mathrm{BANANA}$
के लिए संकेत: पहले तीनों $A$ रखिए।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:7}
सर्वसमिका $k\dbinom{n}{k} = n\dbinom{n-1}{k-1}$ ($1 \leq k \leq n$) दो तरह से सिद्ध कीजिए: \omterm{prop:g12:comb:tuples}{क्रमगुणित} वाले सूत्र से, और
$n$ लोगों में से चुने गए युग्मों ($k$ लोगों की समिति, उसका अध्यक्ष) को दो
तरह से गिनकर।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:8}
समतल में कोई पथ $(0,0)$ से $(m, n)$ तक पूर्व या उत्तर दिशा में एक-एक मात्रक के
क़दमों से जाता है। दिखाइए कि ऐसे पथों की संख्या $\dbinom{m+n}{m}$ है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:comb:9}
\emph{वांडरमोंड की सर्वसमिका} सिद्ध कीजिए: $0 \leq k \leq m + n$ के लिए
\[
\binom{m+n}{k} = \sum_{j=0}^{k} \binom{m}{j}\binom{n}{k-j},
\]
और इसके लिए ऐसे समुच्चय के $k$-अवयवी उपसमुच्चय गिनिए जो $m$ के एक समूह
और $n$ के एक समूह में बँटा हो। इससे $\displaystyle\sum_{j=0}^{n}\binom{n}{j}^{\!2} = \binom{2n}{n}$ निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:comb:10}
द्विपद प्रमेय का उपयोग करके दिखाइए कि सभी $n \geq 1$ के लिए
\[
\sum_{k=1}^{n} k \binom{n}{k} = n\,2^{n-1}.
\]
(संकेत: या तो $(1+x)^n$ का अवकलन कीजिए, या \cref{exo:g12:comb:7} का उपयोग कीजिए।)
\end{exercise}

\section{समस्या: दो बार गिनने की कला}

\begin{problem}[{सप्ताहांत समस्या --- तारे और छड़ें, बिखरी हुई टोपियाँ, और
एक ही वस्तु को दो तरह से गिनकर सिद्ध की गई सर्वसमिकाएँ}]
\label{pb:g12:comb:1}
क्रमचय-संचय की सबसे गहरी तरकीब निहायत सरल है: एक ही संग्रह को दो अलग-अलग
तरीक़ों से गिनिए और दोनों उत्तरों को बराबर रख दीजिए। यह समस्या \cref{met:g12:comb:model} के
प्रतिरूपों का अभ्यास कराती है, फिर एक ऐसी तकनीक जोड़ती है जिसकी अध्याय के
पाठ्यक्रम को आवश्यकता नहीं पड़ी --- कुल्फ़ी गिनने वाले \emph{तारे और छड़ें}
--- और फिर प्रसिद्ध \emph{बिखरी हुई टोपियाँ} ठीक-ठीक गिनती है, और टोपियों के
ढेर की तह में प्रतीक्षा करती संख्या $\frac1\eu$ पा लेती है, जो इस पुस्तक में उसका
तीसरा दर्शन है।

\medskip
\noindent\textbf{भाग I --- प्रतिरूप का चुनाव।}
\begin{enumerate}
  \item $2$ अक्षरों और उसके बाद $3$ अंकों वाली वाहन-पट्टिकाएँ गिनिए; फिर
        $\mathrm{BANANA}$ के अक्षर-विन्यास।
  \item $32$ पत्तों की गड्डी से $5$ पत्तों के हाथ गिनिए; फिर वे हाथ जिनमें
        $4$ इक्कों में से ठीक $2$ हों।
  \item कोई रोबोट $(0,0)$ से $(4,3)$ तक केवल एक-एक मात्रक के दाएँ या ऊपर वाले
        क़दमों से चलता है: कितने पथ बनते हैं? (हर पथ को $R$ और $U$ वाले शब्द
        के रूप में लिखिए।)
  \item द्विपद प्रमेय (\cref{thm:g12:comb:binomial}) से $(1 + x)^4$ का प्रसार कीजिए; फिर $x = 1$ और $x = -1$
        पर मान निकालिए: संख्याओं $\binom nk$ के बारे में कौन-सी दो सर्वसमिकाएँ
        निकल आती हैं?
  \item दोहरी गिनती से सिद्ध कीजिए कि $k\binom nk = n\binom{n-1}{k-1}$ (अध्यक्ष-सहित-समितियाँ दो तरह
        से गिनिए), और $\sum_{k=0}^{n} k\binom nk = n\,2^{n-1}$ निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- तारे और छड़ें।}
\begin{enumerate}[resume]
  \item कुल्फ़ी की एक दुकान $4$ स्वाद बेचती है; आप $10$ गोले मँगाते हैं
        (स्वाद दोहराए जा सकते हैं, और प्याले में क्रम अप्रासंगिक है)। किसी
        ऑर्डर को $10$ तारों (गोलों) की पंक्ति के रूप में लिखिए जिन्हें $3$
        छड़ें (स्वाद बदलने के बिंदु) अलग करती हों, और ऑर्डर गिनिए।
  \item $x + y + z = 12$ वाले अऋणात्मक \omterm{def:g10:numbers:sets}{पूर्णांकों} के त्रिक गिनिए।
  \item $x + y + z = 12$ वाले \emph{धनात्मक} \omterm{def:g10:numbers:sets}{पूर्णांकों} के त्रिक गिनिए ($x = 1 + x'$ आदि
        प्रतिस्थापित कीजिए)।
  \item $(a + b + c)^5$ के प्रसार में कितने भिन्न एकपदी आते हैं?
  \item विधि की जाँच कीजिए: $2$ स्वादों से $3$ गोलों के ऑर्डर सूत्र से
        गिनिए, फिर उन सबकी सूची बनाकर तुलना कीजिए।
  \item ठीक-ठीक बताइए कि ``गोले एक जैसे हैं'' यह बात इस लेखन में कहाँ घुसी
        --- और \cref{met:g12:comb:model} की जाँच-सूची से गिनिए कि यदि गोले क्रम से खाए जाएँ
        (भिन्न स्थान) तो इसके बदले क्या होता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- बिखरी हुई टोपियाँ।}
\emph{विपर्यय} $n$ टोपियों को उनके $n$ मालिकों में इस तरह बाँटना है कि
\emph{किसी को भी} अपनी टोपी न मिले; मान लीजिए $D_n$ ऐसे बँटवारों की गिनती
करता है। (\Cref{pb:g11:prob:1} ने दिखाया था कि औसतन एक अतिथि को अपनी टोपी वापस मिल जाती है
--- अब हम पूरी तरह अभागी दावतें ठीक-ठीक गिनते हैं।)
\begin{enumerate}[resume]
  \item $D_1$, $D_2$, $D_3$ सूची बनाकर निकालिए, और $D_4$ धैर्य से (या
        चतुराई से)।
  \item पुनरावृत्ति $D_n = (n - 1)\left(D_{n-1} + D_{n-2}\right)$ को पुष्ट कीजिए: अतिथि 1 को कोई टोपी $k \neq 1$ मिलती है
        ($n - 1$ विकल्प); फिर इस आधार पर बाँटिए कि अतिथि $k$ को टोपी 1 मिलती
        है या नहीं। जाँचिए कि यह $D_4$ दोबारा दे देती है, और $D_5$ निकालिए।
  \item $n = 3$ के लिए समावेशन--अपवर्जन से सिद्ध कीजिए (कम से कम एक टोपी सही
        रखने वाले बँटवारे घटाइए, फिर अधिक गिनी गई बातें वापस जोड़िए) कि
        $D_3 = 3!\left(1 - \frac{1}{1!} + \frac{1}{2!} -
        \frac{1}{3!}\right)$, और सामान्य सूत्र बताइए।
  \item $\frac{D_5}{5!}$ निकालिए और $\frac1\eu \approx 0.3679$ से तुलना कीजिए: किसी बड़ी फेंटी हुई दावत के
        पूरी तरह बिखर जाने की \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac1\eu$ है --- लॉटरी और \cref{pb:g12:exp:1} के
        सचिव के बाद इस अचर का तीसरा दर्शन। (कारण: प्रश्न 14 का सूत्र $\eu^{-1}$
        की एक प्रसिद्ध श्रेणी का आरंभ है, जिसकी कहानी स्नातक खंडों में है।)
  \item $10$ मित्रों के बीच गुप्त उपहार: नाम एकसमान रूप से यादृच्छिक ढंग
        से निकाले जाते हैं। निकाले का वैध होने (किसी को अपना ही नाम न मिलने)
        की \omterm{def:g10:proba:distribution}{प्रायिकता} क्या है, और समूह को कितनी बार फिर से निकालना पड़ेगा?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- दो बार गिनिए, दो बार जीतिए।}
\begin{enumerate}[resume]
  \item हस्तमिलन प्रमेयिका: किसी भी दावत में हर अतिथि ने जितने हाथ मिलाए
        उनका योग हर हस्तमिलन को ठीक दो बार गिन लेता है। इससे निकालिए कि
        \emph{विषम संख्या में हाथ मिलाने वाले अतिथियों की संख्या सदा सम होती
        है} --- और जाँचिए कि तीन अतिथियों की दावत में यह दावा अर्थपूर्ण है।
  \item रत्न $1^3 + 2^3 + \dots + n^3 = (1 + 2 + \dots + n)^2$ आगमन से सिद्ध कीजिए, और $n = 3$ के लिए उसकी पुष्टि कीजिए।
        (नन्हे गाउस का योग, वर्ग करने पर, घनों की गिनती कर देता है।)
  \item पथों से वांडरमोंड की सर्वसमिका (\cref{exo:g12:comb:9}): $\binom{2n}{n}$ की व्याख्या प्रश्न 3
        जैसे जालक-पथों के रूप में $(0,0)$ से $(n,n)$ तक कीजिए, हर पथ को
        प्रति-विकर्ण पार करने वाली जगह पर काटिए, और समझाइए कि $\sum_j \binom nj^2$ कैसे
        प्रकट होता है।
  \item समापन --- गिनने वाले की चार चालें, हर एक पर एक पंक्ति और इस समस्या
        से एक उदाहरण: चरण गुणा कीजिए और स्थितियाँ जोड़िए; चतुराई से लिखिए
        (तारे और छड़ें, पथ-शब्द); एक ही वस्तु दो बार गिनिए
        (अध्यक्ष-सहित-समिति, हस्तमिलन); अनचाहे घटाइए और अधिक गिनी बातें
        सुधारिए (विपर्यय)। और यह भी लिखिए कि यह गिनती आगे कहाँ काम पर लगती
        है: \omterm{def:g10:proba:distribution}{प्रायिकता} में, और आव्यूह-तथा-ग्राफ़ वाले अध्याय के पथों में।
\end{enumerate}
\end{problem}
