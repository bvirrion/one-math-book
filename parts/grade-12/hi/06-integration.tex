\chapter{समाकलन}\label{ch:g12:integ}

समाकलन एक साथ दो प्रश्नों का उत्तर देता है: किसी वक्र के नीचे का क्षेत्रफल
क्या है, और किसी \omterm{def:g11:func:function}{फलन} को उसके परिवर्तन की दर से वापस कैसे पाया जाए? कलन की मूल
प्रमेय कहती है कि ये दोनों एक ही प्रश्न हैं --- सत्रहवीं सदी के गणित की सबसे
गहरी और सबसे उपयोगी खोज।

\section{संतत फलन का समाकल}

\begin{definition}[धनात्मक फलन का समाकल]\label{def:g12:integ:area}
मान लीजिए $f$ $\intcc{a}{b}$ पर \omterm{def:g12:limcont:continuity}{संतत} और धनात्मक है। \emph{समाकल}\index{समाकल}
\[
\int_a^b f(x)\,\dd x
\]
उस क्षेत्र का क्षेत्रफल है (क्षेत्रफल के मात्रकों में) जो $f$ के वक्र,
$x$-अक्ष, और ऊर्ध्वाधर रेखाओं $x = a$ तथा $x = b$ से घिरा है।
\end{definition}

\begin{proposition}[समाकल के गुण]\label{prop:g12:integ:properties}
मान लीजिए $f, g$ $a, b, c$ को समेटने वाले किसी \omterm{def:g10:numbers:interval}{अंतराल} पर \omterm{def:g12:limcont:continuity}{संतत} हैं, और $\lambda \in \R$ है।
\begin{enumerate}
  \item \emph{रैखिकता:} $\displaystyle\int_a^b (f + \lambda g) = \int_a^b f + \lambda \int_a^b g$।
  \item \emph{शाल का संबंध:} $\displaystyle\int_a^c f = \int_a^b f + \int_b^c f$।
  \item \emph{धनात्मकता:} यदि $\intcc{a}{b}$ पर $a \leq b$ और $f \geq 0$ हो, तो $\displaystyle\int_a^b f \geq 0$; और यदि $f \leq g$
        हो, तो $\displaystyle\int_a^b f \leq \int_a^b g$।
\end{enumerate}
\end{proposition}

\begin{proof}
शाल और धनात्मकता क्षेत्रफल वाली व्याख्या से तत्काल मिल जाते हैं (क्षेत्र
साथ-साथ रखने पर क्षेत्रफल जुड़ जाते हैं; धनात्मक ऊँचाई वाले क्षेत्र का
क्षेत्रफल धनात्मक होता है)। तुलना धनात्मकता को $g - f$ पर लगाने से निकलती है।
धनात्मक \omterm{def:g10:functions:function}{फलनों} के योग के लिए रैखिकता अंतर्ज्ञान से स्पष्ट है (क्षेत्रफल एक के
ऊपर एक रख दीजिए) और उसकी कठोर उपपत्ति स्नातक स्तर पर होती है; \emph{यहाँ उसे
स्वीकार कर लिया जाता है}।
\end{proof}

\section{कलन की मूल प्रमेय}

\begin{definition}[प्रतिअवकलज]\label{def:g12:integ:primitive}
किसी \omterm{def:g10:numbers:interval}{अंतराल} $I$ पर $f$ का \emph{प्रतिअवकलज}\index{प्रतिअवकलज} $I$ पर
\omterm{def:g12:deriv:derivative}{अवकलनीय} ऐसा \omterm{def:g11:func:function}{फलन} $F$ है जिसके लिए $F' = f$ हो।
\end{definition}

\begin{proposition}\label{prop:g12:integ:primunique}
यदि $F$ किसी \omterm{def:g10:numbers:interval}{अंतराल} $I$ पर $f$ का \omterm{def:g12:integ:primitive}{प्रतिअवकलज} हो, तो $I$ पर $f$ के
\omterm{def:g12:integ:primitive}{प्रतिअवकलज} ठीक \omterm{def:g11:func:function}{फलन} $F + c$, $c \in \R$ हैं। $x_0 \in I$ और $y_0 \in \R$ दिए होने पर $F(x_0) = y_0$ वाला
अद्वितीय \omterm{def:g12:integ:primitive}{प्रतिअवकलज} होता है।
\end{proposition}

\begin{proof}
यदि $G' = F' = f$ हो, तो \omterm{def:g10:numbers:interval}{अंतराल} $I$ पर $(G - F)' = 0$, इसलिए $G - F$ अचर है।\footnote{किसी \omterm{def:g10:numbers:interval}{अंतराल}
पर शून्य \omterm{def:g12:deriv:derivative}{अवकलज} वाला \omterm{def:g11:func:function}{फलन} अचर होता है, यह \cref{thm:g12:deriv:variations} से निकलता है: वह \omterm{def:g12:seq:monotonic}{वर्धमान} और
\omterm{def:g10:functions:variations}{ह्रासमान} दोनों है।} इसके उलट हर $F + c$ \omterm{def:g12:integ:primitive}{प्रतिअवकलज} है। शर्त $F(x_0) = y_0$ अचर को तय कर
देती है।
\end{proof}

\begin{theorem}[कलन की मूल प्रमेय]\label{thm:g12:integ:ftc}
\index{कलन की मूल प्रमेय}
मान लीजिए $f$ किसी \omterm{def:g10:numbers:interval}{अंतराल} $I$ पर \omterm{def:g12:limcont:continuity}{संतत} है और $a \in I$ है। \omterm{def:g11:func:function}{फलन}
\[
F \colon x \longmapsto \int_a^x f(t)\,\dd t
\]
$I$ पर $f$ का वह \omterm{def:g12:integ:primitive}{प्रतिअवकलज} है जो $a$ पर लुप्त होता है। परिणामस्वरूप, $f$
के किसी भी \omterm{def:g12:integ:primitive}{प्रतिअवकलज} $G$ और $a, b \in I$ के लिए:
\[
\int_a^b f(t)\,\dd t = \bigl[G(t)\bigr]_a^b = G(b) - G(a).
\]
\end{theorem}

जो लंबाई $h$ के आधार पर ऊँचाइयों $f(x)$ और $f(x+h)$ वाले आयतों के बीच दबी हुई है।}

\begin{proof}[$f$ के वर्धमान होने पर उपपत्ति]
$x \in I$ और $h > 0$ स्थिर रखिए, जहाँ $x + h \in I$ है। शाल से
\[
F(x+h) - F(x) = \int_x^{x+h} f(t)\,\dd t .
\]
चूँकि $f$ \omterm{def:g12:seq:monotonic}{वर्धमान} है, इसलिए $t \in \intcc{x}{x+h}$ के लिए $f(x) \leq f(t) \leq f(x+h)$, और \omterm{def:g12:integ:area}{समाकलों} की तुलना से
(लंबाई $h$ के \omterm{def:g10:numbers:interval}{अंतराल} पर अचर $c$ का \omterm{def:g12:integ:area}{समाकल} $ch$ होता है):
\[
h\,f(x) \leq F(x+h) - F(x) \leq h\,f(x+h),
\]
इसलिए
\[
f(x) \leq \frac{F(x+h) - F(x)}{h} \leq f(x+h).
\]
$h \to 0^+$ पर \omterm{def:g12:limcont:continuity}{सांतत्य} से $f(x+h) \to f(x)$, और संपीडन प्रमेय से अंतर-भागफल $f(x)$ की ओर जाता है;
स्थिति $h < 0$ सममित है। अतः $F' = f$, और $F(a) = 0$। सामान्य (\omterm{def:g12:seq:monotonic}{अ-एकदिष्ट}) \omterm{def:g12:limcont:continuity}{संतत} स्थिति
स्नातक स्तर पर सिद्ध होती है।

अंत में, यदि $G$ कोई भी \omterm{def:g12:integ:primitive}{प्रतिअवकलज} हो, तो $G = F + c$ (\cref{prop:g12:integ:primunique}), इसलिए $G(b) - G(a) = F(b) - F(a) = \int_a^b f$।
\end{proof}

\begin{example}
$\displaystyle\int_0^1 x^2\,\dd x = \left[\frac{x^3}{3}\right]_0^1 =
\frac13$: \omterm{def:g11:quad:parabola}{परवलय} के नीचे का क्षेत्रफल इकाई वर्ग का एक-तिहाई है, जैसा आर्किमिडीज़
जानते थे।
\end{example}

\omterm{def:g12:integ:primitive}{प्रतिअवकलजों} की सारणी \omterm{def:g12:deriv:derivative}{अवकलजों} की सारणी से पढ़ी जाती है ($u$ किसी \omterm{def:g12:deriv:derivative}{अवकलनीय} \omterm{def:g11:func:function}{फलन}
को और $c$ किसी मनमाने अचर को दर्शाता है):
\begin{center}
\renewcommand{\arraystretch}{1.7}
\begin{tabular}{c|c||c|c}
$f(x)$ & \omterm{def:g12:integ:primitive}{प्रतिअवकलज} & $f$ & \omterm{def:g12:integ:primitive}{प्रतिअवकलज} \\
\hline
$x^n \ (n \neq -1)$ & $\dfrac{x^{n+1}}{n+1} + c$ &
$u'u^n \ (n \neq -1)$ & $\dfrac{u^{n+1}}{n+1} + c$\\
$\dfrac1x \ (x > 0)$ & $\ln x + c$ &
$\dfrac{u'}{u} \ (u > 0)$ & $\ln u + c$\\
$\eu^x$ & $\eu^x + c$ &
$u'\eu^u$ & $\eu^u + c$\\
$\cos x$ & $\sin x + c$ &
$\dfrac{u'}{\sqrt u} \ (u>0)$ & $2\sqrt u + c$\\
$\sin x$ & $-\cos x + c$ & &
\end{tabular}
\end{center}

\begin{method}[$u' \times (u \text{ का कोई व्यंजक})$ का रूप पहचानना]\label{met:g12:integ:uprime}
किसी गुणनफल का समाकलन करने के लिए ऐसा गुणनखंड ढूँढ़िए जो किसी भीतरी \omterm{def:g11:func:function}{फलन} $u$
का \omterm{def:g12:deriv:derivative}{अवकलज} हो, किसी गुणक अचर तक। जैसे $\int_0^1 x\,\eu^{x^2}\dd x$ में गुणनखंड $x$ $\frac12 (x^2)'$ है:
\[
\int_0^1 x\,\eu^{x^2}\dd x = \frac12\left[\eu^{x^2}\right]_0^1
= \frac{\eu - 1}{2}.
\]
\end{method}

\begin{theorem}[खंडशः समाकलन]\label{thm:g12:integ:ibp}
\index{खंडशः समाकलन}
मान लीजिए $u, v$ $\intcc{a}{b}$ पर \omterm{def:g12:deriv:derivative}{अवकलनीय} हैं और उनके \omterm{def:g12:deriv:derivative}{अवकलज} \omterm{def:g12:limcont:continuity}{संतत} हैं। तब
\[
\int_a^b u'(t)\,v(t)\,\dd t
= \bigl[u(t)\,v(t)\bigr]_a^b - \int_a^b u(t)\,v'(t)\,\dd t .
\]
\end{theorem}

\begin{proof}
गुणनफल के नियम से $(uv)' = u'v + uv'$; दोनों पक्षों का $\intcc{a}{b}$ पर समाकलन करके बाएँ पक्ष के
लिए मूल प्रमेय का उपयोग करने पर $\bigl[uv\bigr]_a^b = \int_a^b u'v + \int_a^b uv'$ मिलता है।
\end{proof}

\begin{example}\label{ex:g12:integ:ibp}
$\displaystyle\int_0^1 t\,\eu^{t}\,\dd t$: $u' = \eu^t$, $v = t$ लीजिए, इसलिए $u = \eu^t$, $v' = 1$:
\[
\int_0^1 t\,\eu^t\,\dd t = \bigl[t\,\eu^t\bigr]_0^1 - \int_0^1 \eu^t\,\dd t
= \eu - (\eu - 1) = 1 .
\]
\end{example}

\section{अनुप्रयोग}

\begin{definition}[माध्य मान]\label{def:g12:integ:mean}
$\intcc{a}{b}$ ($a<b$) पर \omterm{def:g12:limcont:continuity}{संतत} \omterm{def:g11:func:function}{फलन} $f$ का \emph{माध्य मान}\index{माध्य मान} है
\[
\mu = \frac{1}{b-a}\int_a^b f(t)\,\dd t .
\]
\end{definition}

\begin{proposition}\label{prop:g12:integ:meanbounds}
यदि $\intcc{a}{b}$ पर $m \leq f \leq M$ हो, तो $m \leq \mu \leq M$।
\end{proposition}

\begin{proof}
असमिकाओं $m \leq f(t) \leq M$ का $\intcc{a}{b}$ पर समाकलन कीजिए और $b - a > 0$ से भाग दीजिए।
\end{proof}

\begin{method}[दो वक्रों के बीच का क्षेत्रफल]\label{met:g12:integ:areabetween}
यदि $\intcc{a}{b}$ पर $f \geq g$ हो, तो दोनों वक्रों के बीच का क्षेत्रफल $\int_a^b \bigl(f(x) - g(x)\bigr)\dd x$ है। यदि
वक्र एक-दूसरे को काटते हों, तो \omterm{def:g10:numbers:interval}{अंतराल} को कटान-बिंदुओं पर बाँट दीजिए और $\abs{f - g}$
का टुकड़ा-दर-टुकड़ा समाकलन कीजिए।
\end{method}

\begin{exercise}[$\star\star$]\label{exo:g12:integ:6}
मान लीजिए $I = \displaystyle\int_0^1 \frac{\dd t}{1 + t}$ है।
\begin{enumerate}
  \item $I$ निकालिए।
  \item $n \in \N$ के लिए मान लीजिए $I_n = \displaystyle\int_0^1 \frac{t^n}{1+t}\dd t$ है। दिखाइए कि $I_n + I_{n+1} = \dfrac{1}{n+1}$, और यह भी कि $0 \leq I_n \leq \dfrac{1}{n+1}$।
  \item निष्कर्ष निकालिए कि $1 - \frac12 + \frac13 - \dots + \frac{(-1)^{n-1}}{n}
        \xrightarrow[n \to +\infty]{} \ln 2$।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:integ:7}
प्रस्थान के बाद पहले $100$ सेकंड में किसी रेलगाड़ी की चाल (m/s में) $v(t) = 30\bigl(1 - \eu^{-t/50}\bigr)$ से
प्रतिरूपित होती है। इन $100$ सेकंडों में तय की गई दूरी और इस \omterm{def:g10:numbers:interval}{अंतराल} पर
रेलगाड़ी की औसत चाल निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:integ:8}
$n \geq 1$ के लिए मान लीजिए $S_n = \dfrac1n \displaystyle\sum_{k=1}^{n} \frac{1}{1 + k/n}$ है।
\begin{enumerate}
  \item $S_n$ की व्याख्या ऐसे आयतों के क्षेत्रफल के रूप में कीजिए जो $\intcc{0}{1}$
        पर $t \mapsto \frac{1}{1+t}$ के वक्र के नीचे के क्षेत्र का \omterm{def:g10:numbers:approx}{सन्निकटन} करते हैं।
  \item $t \mapsto \frac{1}{1+t}$ की एकदिष्टता का उपयोग करके दिखाइए कि
        \[
        S_n \leq \int_0^1 \frac{\dd t}{1+t} \leq S_n + \frac1n\left(1 - \frac12\right),
        \]
        और $\lim\limits_{n\to+\infty} S_n = \ln 2$ निकालिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:integ:9}
\emph{(वालिस \omterm{def:g12:integ:area}{समाकल}.)} $n \in \N$ के लिए मान लीजिए $W_n = \displaystyle\int_0^{\pi/2} \sin^n t\,\dd t$ है।
\begin{enumerate}
  \item $W_0$ और $W_1$ निकालिए।
  \item $\sin^{n+2}t = \sin t \cdot \sin^{n+1} t$ लिखकर और \omterm{thm:g12:integ:ibp}{खंडशः समाकलन} करके दिखाइए कि $W_{n+2} = \dfrac{n+1}{n+2}\,W_n$।
  \item $W_2$, $W_3$, $W_4$ निकालिए और दिखाइए कि $(W_n)$ \omterm{def:g10:functions:variations}{ह्रासमान} तथा धनात्मक
        है।
\end{enumerate}
\end{exercise}

\section{समस्या: आर्किमिडीज़ बनाम मशीन}

\begin{problem}[{सप्ताहांत समस्या --- परवलय के नीचे का क्षेत्रफल, बाईस सदियों
में तीन तरीक़ों से निकाला गया, और लघुगणक की गुप्त पहचान एक क्षेत्रफल के रूप
में}]
\label{pb:g12:integ:1}
लगभग 240 ईसा पूर्व आर्किमिडीज़ ने परवलयीय खंड का यथार्थ क्षेत्रफल निकाल लिया
था --- बिना \omterm{def:g10:coordgeom:system}{निर्देशांकों} के, बिना सीमाओं के, बिना बीजगणित के। उन्नीस सदियों
बाद रीमान के आयतों ने उसे शुद्ध शिकंजे से फिर निकाला; और अब कलन की मूल प्रमेय
(\cref{thm:g12:integ:ftc}) उसे एक ही पंक्ति में कर देती है। यह समस्या तीनों मुक़ाबले खेलती है,
और फिर उसी मशीन से यह उजागर करती है कि \omterm{def:g12:exp:ln}{लघुगणक} असल में है क्या: एक ऐसा
क्षेत्रफल जिसमें मापन की सममिति है।

\medskip
\noindent\textbf{भाग I --- प्रवाह।}
\begin{enumerate}
  \item $\displaystyle\int_0^1 (3x^2 - 2x + 1)\,\dd x$, \ $\displaystyle\int_1^{\eu} \frac{\dd x}{x}$, \ और $\displaystyle\int_0^{\pi/2} \cos x\,\dd x$ निकालिए।
  \item $u'u$-रूप पहचानिए (\cref{met:g12:integ:uprime}): $\displaystyle\int_0^1 x\,\eu^{x^2}\dd x$ और $\displaystyle\int_0^1 \frac{2x}{x^2 + 1}\,\dd x$।
  \item खंडशः (\cref{thm:g12:integ:ibp}): $\displaystyle\int_0^1 x\,\eu^x \dd x$ और $\displaystyle\int_1^{\eu} \ln x\,\dd x$।
  \item $\intcc{0}{\pi}$ पर $\sin$ का \omterm{def:g12:integ:mean}{माध्य मान} (\cref{def:g12:integ:mean}) निकालिए --- और ध्यान दीजिए कि वह
        $\frac12$ \emph{नहीं} है।
  \item $\intcc{0}{1}$ पर रेखा $y = x$ और \omterm{def:g11:quad:parabola}{परवलय} $y = x^2$ के बीच का क्षेत्रफल निकालिए
        (\cref{met:g12:integ:areabetween})।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- \omterm{def:g11:quad:parabola}{परवलय}, तीन तरीक़ों से।}
\begin{enumerate}[resume]
  \item $\intcc{0}{1}$ पर $y = x^2$ के नीचे के क्षेत्रफल के लिए रीमान की व्यवस्था: $n$
        बराबर आयतों पर अधोयोग $L_n$ और उपरियोग $U_n$ लिखिए (जैसा \cref{exo:g12:integ:8} में
        था)।
  \item आगमन (\cref{pb:g12:seq:1} की मशीन) से वर्गों के योग का सूत्र सिद्ध कीजिए:
        \[
        1^2 + 2^2 + \dots + n^2 = \frac{n(n+1)(2n+1)}{6} .
        \]
  \item $U_n$ और $L_n$ के बंद रूप निकालिए, उनकी उभयनिष्ठ सीमा निकालिए, और
        निष्कर्ष निकालिए: क्षेत्रफल $\frac13$ है।
  \item अब मशीन: मूल प्रमेय से $\int_0^1 x^2\,\dd x$ एक ही पंक्ति में निकालिए। दोनों मेहनतों
        की तुलना कीजिए।
  \item आर्किमिडीज़ ने इसे अलग तरह से कहा था: \emph{परवलयीय खंड अपने अंतर्लिखित
        त्रिभुज का $\frac43$ होता है}। $(-1, 1)$ को $(1, 1)$ से जोड़ने वाली जीवा द्वारा
        $y = x^2$ से काटे गए खंड के लिए: \omterm{def:g12:integ:area}{समाकल} से खंड का क्षेत्रफल, अंतर्लिखित
        त्रिभुज का क्षेत्रफल (शीर्ष $(0, 0)$ पर) निकालिए, और उस्ताद का अनुपात
        सत्यापित कीजिए।
  \item आर्किमिडीज़ की अपनी विधि: खंड को पहले बड़े त्रिभुज $T$ से भरिए,
        फिर कुल $\frac T4$ वाले दो त्रिभुजों से, फिर कुल $\frac{T}{16}$ वाले चार से, और
        इसी तरह आगे। \omterm{def:g12:seq:arith-geom}{गुणोत्तर} श्रेणी जोड़िए और $\frac43 T$ फिर से पाइए ---
        माध्यमिक विद्यालय खंड की अनंत बार कुतरी गई चॉकलेट, जिसे एक यूनानी
        ज्यामितिज्ञ खा गया।
  \item एक-एक वाक्य में: निःशेषण (आर्किमिडीज़), शिकंजा (रीमान), प्रतिअवकलन
        (न्यूटन--लाइबनिज़) --- हर एक को क्या चाहिए, और हर एक क्या देता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{def:g12:exp:ln}{लघुगणक} एक क्षेत्रफल है।}
$x > 0$ के लिए $A(x) = \displaystyle\int_1^x \frac{\dd t}{t}$ रखिए।
\begin{enumerate}[resume]
  \item $A(1)$ और $A'(x)$ (\cref{thm:g12:integ:ftc}) बताइए, और निष्कर्ष निकालिए कि $A$ ठीक \cref{def:g12:exp:ln}
        का \omterm{def:g12:exp:ln}{प्राकृत लघुगणक} है।
  \item मापन का चमत्कार: $a > 0$ स्थिर रखिए और $g(x) = A(ax) - A(x)$ का अध्ययन कीजिए। $g'$
        निकालिए (शृंखला नियम), निष्कर्ष निकालिए कि $g$ अचर है, अचर का मान
        निकालिए --- और शुद्ध कलन से सिद्ध फलनीय \omterm{def:g10:algebra:equation}{समीकरण}
        \[
        \ln(ab) = \ln a + \ln b ,
        \]
        प्राप्त कीजिए: $1$ से $ab$ तक का क्षेत्रफल मापित प्रतियों में
        बँट जाता है।
  \item प्रश्न 14 से निकालिए: $\ln(a^n) = n\ln a$ और $\ln\frac1a = -\ln a$।
  \item \Cref{exo:g12:integ:8} $\frac{1}{n+1} + \frac{1}{n+2} + \dots + \frac{1}{2n}$ को $\frac{1}{1 + t}$ के नीचे के आयतों के रूप में पढ़ता है: $n = 10$ के
        लिए यह योग निकालिए (तीन दशमलव स्थान) और $\ln 2$ से तुलना कीजिए।
        कौन-से हरात्मक स्वाद वाले योग, जो पद-दर-पद अपसारी हैं, यहाँ किसी
        क्षेत्रफल पर \omterm{def:g12:seq:limit}{अभिसरित} हो जाते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- संचयन।}
\begin{enumerate}[resume]
  \item कोई कार $t \in \intcc{0}{10}$ तक चाल $v(t) = 3t^2$ m/s से त्वरित होती है। तय की गई दूरी,
        \omterm{def:g11:stat:mean}{माध्य} चाल, और वह क्षण निकालिए जिस पर तात्क्षणिक चाल \omterm{def:g11:stat:mean}{माध्य} चाल के
        बराबर हो जाती है।
  \item $4$-मीटर की छड़ का रैखिक घनत्व $\rho(x) = 2 + x$ kg/m है। उसका कुल द्रव्यमान
        और उसका द्रव्यमान केंद्र $\dfrac{\int_0^4 x\,\rho(x)\,\dd x}{\int_0^4
        \rho(x)\,\dd x}$ निकालिए --- माध्यमिक विद्यालय खंड
        के गत्ते के त्रिभुज वाला संतुलन-बिंदु, अब बदलते भारों के साथ
        निकाला हुआ।
  \item वालिस के \omterm{def:g12:integ:area}{समाकल} (\cref{exo:g12:integ:9}): \cref{pb:g12:trigo:1} के रैखिकीकरण का उपयोग करके $W_0$,
        $W_1$ और $W_2 = \int_0^{\pi/2} \sin^2 t\,\dd t$ निकालिए। (उनकी अनंत सीढ़ी स्नातक खंडों में चढ़कर
        $\pi$ के एक गुणनफल-सूत्र तक और घंटी-वक्र के प्रसामान्यीकरण तक जाती
        है।)
  \item समापन --- समाकलन के तीन चेहरे: परिभाषा से एक \emph{क्षेत्रफल},
        संसार में एक \emph{संचयन} (दूरी, द्रव्यमान), और मूल प्रमेय से एक
        \emph{\omterm{def:g12:integ:primitive}{प्रतिअवकलज}}; और उसका इतिहास तीन नामों में। अध्याय के मोती के
        साथ समापन कीजिए: कैलकुलेटर का कौन-सा रोज़मर्रा का बटन चुपके से $\frac1t$
        के नीचे का क्षेत्रफल है --- और उस क्षेत्रफल ने कौन-सी सर्वसमिका
        सिद्ध कर दी?
\end{enumerate}
\end{problem}
