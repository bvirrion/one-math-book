\chapter{त्रिकोणमितीय फलन}\label{ch:g12:trigo}

\omterm{def:g11:func:function}{फलन} $\cos$ और $\sin$ वृत्त की ज्यामिति को विश्लेषण में बदल देते हैं। वास्तविक रेखा
को इकाई वृत्त पर लपेटकर परिभाषित ये \omterm{def:g11:func:function}{फलन} आवर्ती \omterm{def:g10:functions:function}{फलनों} के आदिरूप हैं, और उनके
\omterm{def:g12:deriv:derivative}{अवकलज} उन्हें दोलन के \omterm{def:g10:algebra:equation}{समीकरण} $y'' = -y$ का हल बना देते हैं।

\section{इकाई वृत्त}

\begin{definition}[कोज्या और ज्या]\label{def:g12:trigo:cossin}
मान लीजिए $\mathcal{C}$ किसी \omterm{def:g10:coordgeom:system}{लंबकोणिक} \omterm{def:g10:coordgeom:system}{निर्देशांक-पद्धति} के मूल बिंदु पर केंद्रित
त्रिज्या 1 का वृत्त है। हर वास्तविक $t$ के साथ वह बिंदु $M(t)$ जोड़िए जो बिंदु
$I(1,0)$ से $\mathcal{C}$ के अनुदिश लंबाई $\abs{t}$ लपेटने पर मिलता है --- $t \geq 0$ होने पर वामावर्त
और अन्यथा दक्षिणावर्त। तब \emph{$\cos t$ और $\sin t$ $M(t)$ के
निर्देशांक हैं}\index{कोज्या}\index{ज्या}:
\[
M(t) = (\cos t, \sin t).
\]
\end{definition}

चूँकि वृत्त की परिधि $2\pi$ है, इसलिए $t$ के $2\pi$ बढ़ने पर बिंदु $M(t)$ वही रहता
है; और चूँकि $M(t)$ इकाई वृत्त पर है:

\begin{proposition}[मूलभूत सर्वसमिकाएँ]\label{prop:g12:trigo:identities}
सभी $t \in \R$ और $k \in \Z$ के लिए:
\[
\cos^2 t + \sin^2 t = 1, \qquad
\cos(t + 2k\pi) = \cos t, \qquad
\sin(t + 2k\pi) = \sin t,
\]
\[
\cos(-t) = \cos t, \qquad \sin(-t) = -\sin t,
\]
\[
\cos(\pi - t) = -\cos t, \quad \sin(\pi - t) = \sin t, \quad
\cos\left(\tfrac{\pi}{2} - t\right) = \sin t, \quad
\sin\left(\tfrac{\pi}{2} - t\right) = \cos t .
\]
\end{proposition}

\begin{proof}
पहली सर्वसमिका इकाई वृत्त का \omterm{def:g10:algebra:equation}{समीकरण} है; शेष वृत्त की सममितियाँ व्यक्त करती
हैं: $M(-t)$ $x$-अक्ष में $M(t)$ का परावर्तन है, $M(\pi - t)$ $y$-अक्ष में, और $M(\frac\pi2 - t)$
रेखा $y = x$ में।
\end{proof}

कंठस्थ रखने योग्य मान:
\begin{center}
\begin{tabular}{c|ccccc}
$t$ & $0$ & $\dfrac{\pi}{6}$ & $\dfrac{\pi}{4}$ & $\dfrac{\pi}{3}$ & $\dfrac{\pi}{2}$\\[6pt]
\hline\rule{0pt}{14pt}%
$\cos t$ & $1$ & $\dfrac{\sqrt3}{2}$ & $\dfrac{\sqrt2}{2}$ & $\dfrac12$ & $0$\\[6pt]
$\sin t$ & $0$ & $\dfrac12$ & $\dfrac{\sqrt2}{2}$ & $\dfrac{\sqrt3}{2}$ & $1$
\end{tabular}
\end{center}

\begin{proposition}[योग सूत्र]\label{prop:g12:trigo:addition}
सभी $a, b \in \R$ के लिए:
\begin{align*}
\cos(a+b) &= \cos a \cos b - \sin a \sin b, &
\sin(a+b) &= \sin a \cos b + \cos a \sin b,\\
\cos(a-b) &= \cos a \cos b + \sin a \sin b, &
\sin(a-b) &= \sin a \cos b - \cos a \sin b.
\end{align*}
विशेष रूप से $\cos 2a = 2\cos^2 a - 1 = 1 - 2\sin^2 a$ और $\sin 2a = 2 \sin a \cos a$।
\end{proposition}

\begin{proof}
\omterm{def:g11:vect:vector}{सदिश} $\vect{u} = (\cos a, \sin a)$ और $\vect{v} = (\cos b, \sin b)$ इकाई \omterm{def:g11:vect:vector}{सदिश} हैं जो $a - b$ का कोण बनाते हैं; उनका \omterm{def:g11:scal:dot}{अदिश गुणनफल}
$\cos a\cos b + \sin a \sin b$ (\omterm{def:g10:coordgeom:system}{निर्देशांकों} से) और $\norm{\vect u}\,\norm{\vect v}\cos(a-b) = \cos(a - b)$ (ज्यामिति से) दोनों के बराबर है, और यही तीसरा
सूत्र है। $b$ के स्थान पर $-b$ रखने पर पहला मिलता है; \omterm{def:g12:trigo:cossin}{ज्या} वाले सूत्र $\sin x = \cos(\frac\pi2 - x)$
की सहायता से निकलते हैं। द्विकोण सूत्र $b = a$ वाली स्थिति को $\cos^2 + \sin^2 = 1$ के साथ मिलाने
से मिलते हैं।
\end{proof}

\section{विश्लेषणात्मक अध्ययन}

\begin{lemma}[मूलभूत सीमा]\label{lem:g12:trigo:sinxx}
\[
\lim_{t \to 0} \frac{\sin t}{t} = 1
\qquad\text{और}\qquad
\lim_{t \to 0} \frac{\cos t - 1}{t} = 0 .
\]
\end{lemma}

\begin{proof}
$0 < t < \frac{\pi}{2}$ के लिए इकाई वृत्त में तीन क्षेत्रफलों की तुलना कीजिए: त्रिभुज $OIM(t)$,
त्रिज्यखंड $OIM(t)$, और आधार $OI$ तथा ऊँचाई $\tan t$ वाला समकोण त्रिभुज:
\[
\frac{\sin t}{2} \leq \frac{t}{2} \leq \frac{\tan t}{2}.
\]

त्रिज्यखंड (लाल चाप से घिरा, क्षेत्रफल $\frac{t}{2}$), और ऊँचाई $\tan t$ वाला समकोण
त्रिभुज (नारंगी, क्षेत्रफल $\frac{\tan t}{2}$)।}
\end{omfigure}

पहली असमिका से $\frac{\sin t}{t} \leq 1$ मिलता है; दूसरी से $\cos t \leq \frac{\sin t}{t}$। $t \to 0^+$ पर $\cos t \to 1$ (\omterm{def:g12:limcont:continuity}{सांतत्य} से), और
संपीडन प्रमेय $\frac{\sin t}{t} \to 1$ दे देती है; सम-विषमता इसे $t \to 0$ तक बढ़ा देती है। दूसरी सीमा
के लिए
\[
\frac{\cos t - 1}{t} = \frac{\cos^2 t - 1}{t(\cos t + 1)}
= -\frac{\sin t}{t}\cdot\frac{\sin t}{\cos t + 1}
\xrightarrow[t\to0]{} -1 \cdot \frac{0}{2} = 0. \qedhere
\]
\end{proof}

\begin{theorem}[ज्या और कोज्या के अवकलज]\label{thm:g12:trigo:derivatives}
\omterm{def:g11:func:function}{फलन} $\sin$ और $\cos$ $\R$ पर \omterm{def:g12:deriv:derivative}{अवकलनीय} हैं, और
\[
(\sin)' = \cos, \qquad (\cos)' = -\sin .
\]
अधिक सामान्य रूप से, $(\sin(ax+b))' = a\cos(ax+b)$ और $(\cos(ax+b))' = -a\sin(ax+b)$।
\end{theorem}

\begin{proof}
योग सूत्रों से
\[
\frac{\sin(x+h) - \sin x}{h}
= \sin x\,\frac{\cos h - 1}{h} + \cos x\,\frac{\sin h}{h}
\xrightarrow[h\to0]{} \sin x \cdot 0 + \cos x \cdot 1 = \cos x,
\]
जहाँ \cref{lem:g12:trigo:sinxx} का उपयोग हुआ है। $\cos$ के लिए गणना ठीक वैसी ही है, और सामान्य रूप
शृंखला नियम से निकलता है।
\end{proof}

\begin{proposition}[परिवर्तन]\label{prop:g12:trigo:variations}
$\intcc{0}{\pi}$ पर $\cos$ $1$ से $-1$ तक घटता है। $\intcc{-\frac\pi2}{\frac\pi2}$ पर $\sin$ $-1$ से $1$ तक
बढ़ता है। दोनों \omterm{def:g11:func:function}{फलन} $2\pi$-आवर्ती हैं और उनका परिसर $\intcc{-1}{1}$ है; $\cos$ सम है और
$\sin$ विषम।
\end{proposition}

\begin{proof}
$\intoo{0}{\pi}$ पर $(\cos)' = -\sin < 0$, क्योंकि वहाँ बिंदु $M(t)$ की कोटि धनात्मक है; इसी तरह $\intoo{-\frac\pi2}{\frac\pi2}$
पर $(\sin)' = \cos > 0$। आवर्तिता और सम-विषमता \cref{prop:g12:trigo:identities} से आती हैं।
\end{proof}

\begin{method}[$\cos x = a$ और $\sin x = a$ हल करना]\label{met:g12:trigo:equations}
$a \in \intcc{-1}{1}$ के लिए कोई एक विशेष हल $\alpha$ ज्ञात कीजिए (मानों की सारणी या कैलकुलेटर
से)। तब सभी हल ये हैं:
\[
\cos x = \cos\alpha \iff x = \alpha + 2k\pi \text{ या } x = -\alpha + 2k\pi,
\quad k \in \Z;
\]
\[
\sin x = \sin\alpha \iff x = \alpha + 2k\pi \text{ या } x = \pi - \alpha + 2k\pi,
\quad k \in \Z.
\]
असमिकाओं के लिए इकाई वृत्त पर हल वाले चाप ढूँढ़िए और एक आवर्तकाल के भीतर के
\omterm{def:g10:numbers:interval}{अंतराल} पढ़ लीजिए।
\end{method}

\begin{example}\label{ex:g12:trigo:equation}
$\intoc{-\pi}{\pi}$ में $\cos x = \frac12$ हल कीजिए: एक विशेष हल $\frac\pi3$ है, इसलिए हल $x = \frac\pi3$ और $x = -\frac\pi3$ हैं।
पूरे $\R$ में: $x = \pm\frac\pi3 + 2k\pi$, $k \in \Z$।
\end{example}

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:4}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:5}
$\intcc{0}{2\pi}$ में असमिका $2\sin^2 x - \sin x - 1 \geq 0$ हल कीजिए। (संकेत: $\sin x$ में द्विघात का \omterm{def:g10:algebra:expand}{गुणनखंडन}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:6}

\lim_{x\to0} \frac{\sin 3x}{x}, \qquad
\lim_{x\to0} \frac{1 - \cos x}{x^2}, \qquad

\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:7}
$\intcc{0}{2\pi}$ पर \omterm{def:g11:func:function}{फलन} $f(x) = x + \cos x$ का अध्ययन कीजिए: परिवर्तन और \omterm{def:g12:deriv:extremum}{चरम मान}। दिखाइए कि $f$
$\R$ पर \omterm{def:g12:seq:monotonic}{वर्धमान} है, यद्यपि $f'$ अनंत बिंदुओं पर लुप्त होता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:trigo:8}
मान लीजिए $(A,B) \neq (0,0)$ के साथ $f(x) = A\cos x + B \sin x$ है।
\begin{enumerate}
  \item दिखाइए कि $f$ को $R = \sqrt{A^2 + B^2}$ और किसी उपयुक्त $\varphi$ के साथ $f(x) = R\cos(x - \varphi)$ लिखा जा
        सकता है।
  \item $f$ के \omterm{def:g10:functions:extrema}{अधिकतम} और \omterm{def:g10:functions:extrema}{न्यूनतम} निकालिए, और $\intoc{-\pi}{\pi}$ में $\cos x + \sin x = 1$ हल कीजिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:trigo:9}
दिखाइए कि $\sin$ और $\cos$ दोनों अवकल \omterm{def:g10:algebra:equation}{समीकरण} $y'' = -y$ को संतुष्ट करते हैं। इसके
उलट, मान लीजिए $y$ $y(0) = a$ और $y'(0) = b$ वाला $y'' = -y$ का कोई हल है; दिखाइए कि $y = a\cos + b\sin$।
(संकेत: $g = (y - a\cos - b\sin)$ और ``ऊर्जा'' $E = g^2 + (g')^2$ पर विचार कीजिए।)
\end{exercise}

\section{समस्या: लोलक का रहस्य}

\begin{problem}[{सप्ताहांत समस्या --- एक ज्यामितीय सीमा ज्या और कोज्या के
अवकलज चलाती है, और उन्हीं से सारा दोलन: स्प्रिंग, लोलक, विस्पंद, और वह मीटर
जो बनते-बनते रह गया}]
\label{pb:g12:trigo:1}
जितनी भी घड़ियाँ कभी टिक-टिक कर पाईं और जितने भी तार कभी बजे, सब एक ही गणित
मानते हैं: विस्थापन के समानुपाती प्रत्यानयन बल, अतः \omterm{def:g10:algebra:equation}{समीकरण} $y'' = -\omega^2 y$, अतः
कोज्याएँ। इस सबकी जड़ में एक भोली-सी दिखने वाली सीमा बैठी है, $\frac{\sin x}{x} \to 1$ --- जिसका
वादा कक्षा 9 में हुआ था, जिसका उपयोग \cref{thm:g12:trigo:derivatives} ने किया, और जिसकी उपपत्ति यहाँ एक
चित्र से मिलती है। यह समस्या उसी सीमा से चढ़कर एक मीटर के लोलक के झूले तक
जाती है, और बताती है कि मीटर बाल-बाल उससे परिभाषित होने से क्यों बच गया।

\medskip
\noindent\textbf{भाग I --- मूलभूत सीमा।}
\begin{enumerate}
  \item संख्याओं से तैयारी: $x = 0.1$ और $x = 0.01$ के लिए $\frac{\sin x}{x}$ निकालिए (\omterm{def:g11:trigo:radian}{रेडियन}
        में!)।
  \item $0 < x < \frac\pi2$ के साथ इकाई वृत्त पर सैंडविच: प्रारंभिक ज्यामिति से इन
        क्षेत्रफलों को व्यक्त कीजिए: (क) शीर्षों $O$, $A(1,0)$ और वृत्त-बिंदु
        $M(\cos x, \sin x)$ वाला त्रिभुज; (ख) त्रिज्यखंड $OAM$; (ग) समकोण त्रिभुज
        $OAT$, जहाँ $T(1, \tan x)$।
  \item क्षेत्रफलों के सैंडविच से $\sin x < x < \tan x$ निकालिए, फिर $\cos x < \frac{\sin x}{x} < 1$, और संपीडन प्रमेय
        से निष्कर्ष निकालिए कि $\lim_{x \to 0} \frac{\sin x}{x} = 1$ (सम-विषमता $x < 0$ सँभाल लेती है)।
  \item $\lim_{x \to 0} \frac{1 - \cos x}{x^2} = \frac12$ निकालिए (संयुग्मी से गुणा कीजिए)।
  \item समझाइए कि यही सीमा \cref{thm:g12:trigo:derivatives} का इंजन क्यों है ($\sin'(0)$ क्या है?), और यह
        भी कि वह एक पुराना क़र्ज़ कैसे चुका देती है: किस अर्थ में
        \emph{\omterm{def:g11:trigo:radian}{रेडियन}} ही कोण का वह अकेला मात्रक है जिसके लिए $\sin' = \cos$ बिना किसी
        अचर के सही रहता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- सरल आवर्त गति।}
\begin{enumerate}[resume]
  \item शृंखला नियम से जाँचिए कि $y(t) = R\cos(\omega t - \varphi)$ $y'' = -\omega^2 y$ को संतुष्ट करता है।
  \item स्प्रिंग पर लटका द्रव्यमान $y'' = -4y$ मानता है, जहाँ $y(0) = 3$ और $y'(0) = 8$ हैं।
        यह स्वीकार करते हुए (\cref{exo:g12:trigo:9} से) कि सभी हल $A\cos 2t + B\sin 2t$ के रूप के हैं, $A$
        और $B$ ज्ञात कीजिए।
  \item हल को $R\cos(2t - \varphi)$ (\cref{exo:g12:trigo:8}) के रूप में फिर से लिखिए: आयाम $R$, आवर्तकाल,
        और कला $\varphi$ बताइए (तीन दशमलव स्थान)।
  \item द्रव्यमान पहली बार साम्य स्थिति $y = 0$ से किस समय गुज़रता है?
  \item लोलक: लंबाई $L$ की डोरी पर लटका गोलक \emph{ठीक-ठीक} $\theta'' = -\frac gL \sin\theta$ मानता है
        --- जो प्रारंभिक \omterm{def:g10:functions:function}{फलनों} से हल नहीं होता। छोटे झूलों के लिए $\sin\theta$ के
        स्थान पर $\theta$ रखिए (भाग I उसे उचित ठहराता है!) और आवर्तकाल $T = 2\pi\sqrt{\frac Lg}$
        निकालिए। $L = 1$ m ($g = 9.81$) के लिए $T$ निकालिए: गैलीलियो की प्रसिद्ध
        खोज --- यह सूत्र किस पर निर्भर \emph{नहीं} करता?
  \item सेकंड-लोलक: कौन-सी लंबाई ठीक $T = 2$ s देती है? 1791 की विज्ञान अकादमी
        \emph{मीटर} को परिभाषित करने के लिए इस लंबाई और चतुर्थांश-याम्योत्तर
        के एक करोड़वें भाग के बीच झिझक रही थी --- दोनों दावेदारों में कितने
        मिलीमीटर का अंतर है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- योग सूत्र काम पर।}
\begin{enumerate}[resume]
  \item योग सूत्रों (\cref{prop:g12:trigo:addition}) से दिखाइए कि $\cos\left(x + \frac\pi2\right) = -\sin x$ और $\sin\left(x + \frac\pi2\right) = \cos x$: वृत्तीय अवकलन एक
        चौथाई चक्कर है --- \cref{thm:g12:trigo:derivatives} के साथ संगति जाँचिए।
  \item गुणनफल-से-योग वाली सर्वसमिका $\cos a \cos b = \frac12\left(\cos(a - b) +
        \cos(a + b)\right)$ और रैखिकीकरण $\cos^2 x = \frac{1 + \cos 2x}{2}$ निकालिए
        (समाकलन वाला अध्याय आपका आभारी होगा)।
  \item विस्पंद: $\sin p + \sin q = 2 \sin\frac{p + q}{2}
        \cos\frac{p - q}{2}$ निकालिए, और उसे $440$ तथा $444$ Hz के दो स्वरित्र
        द्विभुजों पर लगाइए: कान कौन-सा स्वर सुनता है, और प्रति सेकंड कितनी
        धड़कनें? (वाद्यवृंद इसी तरह अपने साज़ मिलाते हैं।)
  \item त्रिकोण-कोण सर्वसमिका $\cos 3x = 4\cos^3 x - 3\cos x$ निकालिए ($3x = 2x + x$ लिखिए)। $x = 20^\circ$ रखने पर
        $\cos 20^\circ$ को कौन-सा त्रिघात \omterm{def:g10:algebra:equation}{समीकरण} संतुष्ट करना पड़ता है? (यह त्रिघात
        अकेले वर्गमूलों से हल नहीं होता, और ठीक इसीलिए कोण का त्रिभाजन दो
        सहस्राब्दी तक पटरी और परकार को हराता रहा --- पूरी कहानी स्नातक
        खंडों में है।)
  \item कोई प्रत्यावर्ती धारा $I(t) = 3\cos(100\pi t) + 3\sqrt3 \sin(100\pi t)$ है। उसे $R\cos(100\pi t - \varphi)$ के रूप में फिर से लिखिए:
        शिखर धारा और कला क्या हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- दोलन करता संसार।}
\begin{enumerate}[resume]
  \item आवर्तकाल: $\sin^2 t$ का (सबसे छोटा) आवर्तकाल क्या है (प्रश्न 13 का उपयोग
        कीजिए)? और $\sin(2t) + \sin(3t)$ का?
  \item असली लोलक अवमंदित होता है: $y(t) = \eu^{-t/10}\cos(2\pi t)$। गति का वर्णन कीजिए, $t = 10$ पर
        अन्वालोप का मान निकालिए (एक पुराना मित्र प्रकट होता है, \cref{pb:g12:exp:1}), और
        उस अध्याय का नाम बताइए जहाँ ऐसे हलों वाले \omterm{def:g10:algebra:equation}{समीकरण} हल किए जाते हैं।
  \item हर कंपन पर \omterm{def:g12:trigo:cossin}{ज्या} और \omterm{def:g12:trigo:cossin}{कोज्या} का राज क्यों है? भौतिकी-से-गणित का शब्दकोश
        दो वाक्यों में दीजिए (विस्थापन के समानुपाती प्रत्यानयन बल $\rightarrow$
        कौन-सा \omterm{def:g10:algebra:equation}{समीकरण} $\rightarrow$ कौन-से हल), और बताइए कि \cref{exo:g12:trigo:9} इस दावे में क्या
        जोड़ता है।
  \item समापन: अध्याय की चाप चार क़दमों में --- एक ज्यामितीय सीमा, दो \omterm{def:g12:deriv:derivative}{अवकलज},
        एक अवकल \omterm{def:g10:algebra:equation}{समीकरण}, और सारा दोलन। और ईमानदार तारांकन: बड़े झूलों के लिए
        लोलक का सच्चा आवर्तकाल एक दीर्घवृत्तीय समाकल माँगता है, जिसे बिजली
        की गति से निकालता है\,\dots\ गाउस का समांतर--गुणोत्तर \omterm{def:g11:stat:mean}{माध्य}, यानी
        \cref{pb:g12:seq:1} की मिठाई। शृंखला के धागे आपस में बँध जाते हैं।
\end{enumerate}
\end{problem}
