\chapter{यादृच्छिक चरों के योग और बृहत् संख्याओं का नियम}\label{ch:g12:sums}

जुआघर अंत में हमेशा क्यों जीतते हैं, और सर्वेक्षण काम क्यों करते हैं? क्योंकि
अनेक \omterm{def:g12:sums:indep}{स्वतंत्र} यादृच्छिक राशियों के औसत कम से कम उतार-चढ़ाव करते हैं। यह अध्याय
इसे सिद्ध करता है: \omterm{def:g12:randvar:exp}{प्रत्याशा} की रैखिकता, \omterm{def:g12:sums:indep}{स्वतंत्र} चरों के लिए \omterm{def:g12:randvar:exp}{प्रसरण} की
योज्यता, ब्येनेमे--चेबिशेव असमिका, और बृहत् संख्याओं का नियम।

\section{यादृच्छिक चरों के योग}

एक ही परिमित \omterm{def:g11:prob:model}{प्रतिदर्श समष्टि} $\Omega$ पर दो \omterm{def:g12:randvar:rv}{यादृच्छिक चर} $X, Y$ दिए होने पर योग $X + Y$
वह \omterm{def:g12:randvar:rv}{यादृच्छिक चर} $\omega \mapsto X(\omega) + Y(\omega)$ है।

\begin{theorem}[प्रत्याशा की रैखिकता]\label{thm:g12:sums:linearity}
\index{प्रत्याशा!रैखिकता}
$\Omega$ पर सभी \omterm{def:g12:randvar:rv}{यादृच्छिक चरों} $X, Y$ और $a, b \in \R$ के लिए:
\[
\E(X + Y) = \E(X) + \E(Y), \qquad \E(aX + b) = a\E(X) + b .
\]
\emph{इसके लिए \omterm{def:g12:condprob:indep}{स्वतंत्रता} की कोई मान्यता आवश्यक नहीं है।}
\end{theorem}

\begin{proof}
\omterm{def:g12:randvar:exp}{प्रत्याशा} को परिणामों पर योग के रूप में लिखिए: चूँकि $\P(X = x) = \sum_{\omega : X(\omega) = x} \P(\{\omega\})$, इसलिए पद समूहित
करने पर $\E(X) = \sum_{\omega \in \Omega} \P(\{\omega\})\,X(\omega)$ मिलता है। तब
\[
\E(X + Y) = \sum_{\omega} \P(\{\omega\})\bigl(X(\omega) + Y(\omega)\bigr)
= \sum_{\omega} \P(\{\omega\})X(\omega)
+ \sum_{\omega} \P(\{\omega\})Y(\omega) = \E(X) + \E(Y). \qedhere
\]
\end{proof}

\begin{definition}[स्वतंत्र यादृच्छिक चर]\label{def:g12:sums:indep}
$X$ और $Y$ \emph{स्वतंत्र}\index{यादृच्छिक चर!स्वतंत्र} हैं यदि सभी मानों
$x, y$ के लिए
\[
\P(X = x \text{ और } Y = y) = \P(X = x)\,\P(Y = y).
\]
हो। कई चर $X_1, \dots, X_n$ स्वतंत्र हैं यदि यह गुणन-नियम हर उपपरिवार के हर मान-चयन के लिए
सही हो।
\end{definition}

\begin{proposition}\label{prop:g12:sums:prodexp}
यदि $X$ और $Y$ \omterm{def:g12:sums:indep}{स्वतंत्र} हों, तो $\E(XY) = \E(X)\,\E(Y)$।
\end{proposition}

\begin{proof}
\[
\begin{aligned}
\E(XY) &= \sum_{x, y} xy\;\P(X = x \text{ और } Y = y)
= \sum_{x, y} xy\,\P(X=x)\P(Y=y) \\
&= \Bigl(\sum_x x\P(X=x)\Bigr)\Bigl(\sum_y y\P(Y=y)\Bigr). \qedhere
\end{aligned}
\]
\end{proof}

\begin{theorem}[योग का प्रसरण]\label{thm:g12:sums:variance}
\index{प्रसरण!योग का}
यदि $X$ और $Y$ \emph{\omterm{def:g12:sums:indep}{स्वतंत्र}} हों, तो
\[
\V(X + Y) = \V(X) + \V(Y).
\]
और अधिक सामान्य रूप से, \omterm{def:g12:sums:indep}{स्वतंत्र} $X_1, \dots, X_n$ के लिए $\V(X_1 + \dots + X_n) = \V(X_1) + \dots + \V(X_n)$।
\end{theorem}

\begin{proof}
कोनिग--होयगेंस (\cref{prop:g12:randvar:konig}) और रैखिकता का उपयोग करने पर:
\begin{align*}
\V(X + Y) &= \E\bigl((X+Y)^2\bigr) - \bigl(\E X + \E Y\bigr)^2\\
&= \E(X^2) + 2\E(XY) + \E(Y^2)
- \E(X)^2 - 2\E(X)\E(Y) - \E(Y)^2\\
&= \V(X) + \V(Y) + 2\bigl(\E(XY) - \E(X)\E(Y)\bigr),
\end{align*}
और \omterm{def:g12:sums:indep}{स्वतंत्र} चरों के लिए अंतिम कोष्ठक लुप्त हो जाता है (\cref{prop:g12:sums:prodexp})। सामान्य स्थिति
आगमन से निकलती है।
\end{proof}

\begin{example}[द्विपद पर फिर एक नज़र]\label{ex:g12:sums:binomial}
कोई द्विपद चर $X \sim \mathcal B(n, p)$ $n$ \omterm{def:g12:sums:indep}{स्वतंत्र} बर्नूली चरों का योग $X = X_1 + \dots + X_n$ है। अतः संरचनात्मक
रूप से:
\[
\E(X) = np, \qquad \V(X) = np(1-p),
\]
जिससे बिना किसी गणना के \cref{thm:g12:randvar:binomial} फिर मिल जाता है।
\end{example}

\begin{proposition}[प्रतिदर्श माध्य]\label{prop:g12:sums:mean}
मान लीजिए $X_1, \dots, X_n$ \omterm{def:g12:sums:indep}{स्वतंत्र} हैं और सबका \omterm{def:g12:randvar:rv}{बंटन} $X$ जैसा है (\omterm{def:g12:randvar:exp}{प्रत्याशा} $\mu$,
\omterm{def:g12:randvar:exp}{प्रसरण} $\sigma^2$), और $M_n = \frac{X_1 + \dots + X_n}{n}$ उनका \emph{प्रतिदर्श माध्य}\index{प्रतिदर्श माध्य} है।
तब
\[
\E(M_n) = \mu, \qquad \V(M_n) = \frac{\sigma^2}{n}, \qquad
\sigma(M_n) = \frac{\sigma}{\sqrt n}.
\]
\end{proposition}

\begin{proof}
रैखिकता से $\E(M_n) = \frac{n\mu}{n} = \mu$; \omterm{def:g12:condprob:indep}{स्वतंत्रता} से $\V(X_1 + \dots + X_n) = n\sigma^2$, और $n$ से भाग देने पर \omterm{def:g12:randvar:exp}{प्रसरण} $\frac{1}{n^2}$ गुना
हो जाता है (\cref{prop:g12:randvar:affine})।
\end{proof}

$\frac{\sigma}{\sqrt n}$ ही मूलभूत \emph{वर्गमूल नियम} है: किसी औसत के उतार-चढ़ाव आधे करने के लिए
प्रतिदर्श-आकार चौगुना कीजिए।

\section{संकेंद्रण की असमिकाएँ}

\begin{theorem}[मार्कोव असमिका]\label{thm:g12:sums:markov}
\index{मार्कोव असमिका}
यदि $X \geq 0$ और $a > 0$ हों, तो
\[
\P(X \geq a) \leq \frac{\E(X)}{a}.
\]
\end{theorem}

\begin{proof}
$\E(X) = \sum_i p_i x_i$ में (सभी पद अऋणात्मक हैं) केवल वे पद रखिए जिनके लिए $x_i \geq a$ हो: हर एक कम से
कम $a\,p_i$ है, इसलिए $\E(X) \geq a \sum_{x_i \geq a} p_i = a\,\P(X \geq a)$।
\end{proof}

\begin{theorem}[ब्येनेमे--चेबिशेव असमिका]\label{thm:g12:sums:chebyshev}
\index{ब्येनेमे--चेबिशेव असमिका}
किसी भी \omterm{def:g12:randvar:rv}{यादृच्छिक चर} $X$ और किसी भी $\varepsilon > 0$ के लिए:
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
\leq \frac{\V(X)}{\varepsilon^2}.
\]
\end{theorem}

\begin{proof}
अऋणात्मक चर $Y = (X - \E(X))^2$ पर $a = \varepsilon^2$ के साथ मार्कोव असमिका लगाइए:
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
= \P(Y \geq \varepsilon^2)
\leq \frac{\E(Y)}{\varepsilon^2} = \frac{\V(X)}{\varepsilon^2}. \qedhere
\]
\end{proof}

है कि $X$ पूँछों (लाल) में, यानी अपनी \omterm{def:g12:randvar:exp}{प्रत्याशा} से कम से कम $\varepsilon$ दूरी पर,

\begin{theorem}[बृहत् संख्याओं का नियम]\label{thm:g12:sums:lln}
\index{बृहत् संख्याओं का नियम}
मान लीजिए $X_1, \dots, X_n$ \omterm{def:g12:sums:indep}{स्वतंत्र} हैं और सबका \omterm{def:g12:randvar:rv}{बंटन} एक ही है (\omterm{def:g12:randvar:exp}{प्रत्याशा} $\mu$, \omterm{def:g12:randvar:exp}{प्रसरण}
$\sigma^2$), और $M_n$ उनका \omterm{prop:g12:sums:mean}{प्रतिदर्श माध्य} है। तब हर $\varepsilon > 0$ के लिए:
\[
\P\bigl(\abs{M_n - \mu} \geq \varepsilon\bigr)
\leq \frac{\sigma^2}{n\,\varepsilon^2}
\xrightarrow[n \to +\infty]{} 0 .
\]
\omterm{prop:g12:sums:mean}{प्रतिदर्श माध्य} \omterm{def:g12:randvar:exp}{प्रत्याशा} के चारों ओर सिमट जाता है।
\end{theorem}

\begin{proof}
$M_n$ पर ब्येनेमे--चेबिशेव लगाइए, जिसकी \omterm{def:g12:randvar:exp}{प्रत्याशा} $\mu$ और \omterm{def:g12:randvar:exp}{प्रसरण} $\frac{\sigma^2}{n}$ है
(\cref{prop:g12:sums:mean})।
\end{proof}

\begin{remark}
यह प्रमेय प्रायिकता-सिद्धांत और सांख्यिकी के बीच का पुल है: अनेक \omterm{def:g12:sums:indep}{स्वतंत्र}
पुनरावृत्तियों पर किसी \omterm{def:g11:prob:model}{घटना} की \emph{\omterm{def:g10:stats:series}{आपेक्षिक बारंबारता}} उसकी \emph{\omterm{def:g10:proba:distribution}{प्रायिकता}}
के पास पहुँच जाती है ($X_i$ को \omterm{def:g11:prob:model}{घटना} का सूचक लीजिए, जिससे $\mu = p$)। यही अनुकरण से
\omterm{def:g10:proba:distribution}{प्रायिकता} आँकने (\emph{मोंते कार्लो विधि}) और सर्वेक्षण से किसी समष्टि का
अनुपात आँकने को उचित ठहराती है --- और मात्रात्मक रूप से भी: \cref{ch:g12:contdist} देखिए।
\end{remark}

\begin{method}[ब्येनेमे--चेबिशेव का उपयोग]\label{met:g12:sums:chebyshev}
$\P(\abs{M_n - \mu} \geq \varepsilon) \leq \alpha$ की गारंटी के लिए इतना पर्याप्त है कि $n \geq \frac{\sigma^2}{\alpha\,\varepsilon^2}$ हो। यह परिबंध मोटा है
(वास्तविक उतार-चढ़ाव प्रायः कहीं छोटे होते हैं) पर पूरी तरह सार्वभौमिक: वह
\omterm{def:g12:randvar:rv}{बंटन} के बारे में परिमित \omterm{def:g12:randvar:exp}{प्रसरण} के सिवा कुछ नहीं माँगता।
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g12:sums:1}
दो न्यायसंगत पासे फेंके जाते हैं; मान लीजिए $S$ उनका योग है। रैखिकता से ($S$
के \omterm{def:g12:randvar:rv}{बंटन} से नहीं!) $\E(S)$ निकालिए; फिर \omterm{def:g12:condprob:indep}{स्वतंत्रता} से $\V(S)$ निकालिए, यह दिया होने
पर कि एक न्यायसंगत पासे का \omterm{def:g12:randvar:exp}{प्रसरण} $\frac{35}{12}$ है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sums:2}
मान लीजिए $X \sim \mathcal B(100,\ 0.5)$ है। मार्कोव असमिका से $\P(X \geq 75)$ और ब्येनेमे--चेबिशेव से $\P(\abs{X - 50} \geq 25)$
\omterm{def:g12:seq:bounded}{परिबद्ध} कीजिए। तुलना कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sums:3}
कोई न्यायसंगत सिक्का $n$ बार उछाला जाता है और $F_n$ चितों की \omterm{def:g10:stats:series}{आपेक्षिक
बारंबारता} है। $n$ कितना बड़ा होना चाहिए कि ब्येनेमे--चेबिशेव परिबंध से $\P\bigl(\abs{F_n - 0.5} \geq 0.05\bigr) \leq 0.05$
हो जाए?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:4}
कोई निवेशक अपनी पूँजी $n$ \omterm{def:g12:sums:indep}{स्वतंत्र} परिसंपत्तियों में बराबर बाँट देती है, और
हर एक का प्रत्याशित प्रतिफल $\mu = 5\%$ तथा \omterm{def:g11:stat:variance}{मानक विचलन} $\sigma = 20\%$ है। निवेश-संचय के
प्रतिफल $M_n$ की \omterm{def:g12:randvar:exp}{प्रत्याशा} और \omterm{def:g11:stat:variance}{मानक विचलन} निकालिए, और \omterm{def:g11:stat:variance}{मानक विचलन} को $4\%$ से
नीचे लाने के लिए आवश्यक परिसंपत्तियों की संख्या भी। यह किस वित्तीय सिद्धांत
को दर्शाता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:5}
मान लीजिए $X$ और $Y$ \omterm{def:g12:sums:indep}{स्वतंत्र} हैं और दोनों $\{1, 2, 3\}$ पर एकसमान हैं।
\begin{enumerate}
  \item $S = X + Y$ का \omterm{def:g12:randvar:rv}{बंटन} बताइए और उसी से सीधे $\E(S)$, $\V(S)$ निकालिए।
  \item दोनों मान रैखिकता और \omterm{def:g12:randvar:exp}{प्रसरण} की योज्यता से फिर पाइए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:6}
दिखाइए कि \omterm{def:g12:condprob:indep}{स्वतंत्रता} के बिना \omterm{def:g12:randvar:exp}{प्रसरण} की योज्यता टूट सकती है: $Y = X$ के लिए $\V(X + Y)$
निकालिए और $\V(X) + \V(Y)$ से तुलना कीजिए। किन चरों $X$ के लिए $\V(2X) = 2\V(X)$ होता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:7}
किसी पासे पर भारित होने का संदेह है। उसे $1200$ बार फेंका जाता है और वह $260$
बार छक्का दिखाता है ($\frac16 \approx 0.1667$ के बजाय $f = 0.2167$)। पासे के न्यायसंगत होने की परिकल्पना
के अधीन ब्येनेमे--चेबिशेव से $\P\bigl(\abs{F_n - \frac16} \geq 0.05\bigr)$ \omterm{def:g12:seq:bounded}{परिबद्ध} कीजिए, और चर्चा कीजिए कि
न्यायसंगतता की परिकल्पना विश्वसनीय है या नहीं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:sums:8}
\emph{(सिक्के के लिए एक बेहतर असमिका.)} मान लीजिए $X \sim \mathcal B(n,\ p)$ और $F_n = \frac Xn$ हैं।
\begin{enumerate}
  \item दिखाइए कि $p \in \intcc{0}{1}$ के लिए $p(1-p) \leq \frac14$ है।
  \item बंटन-निरपेक्ष परिबंध $\P\bigl(\abs{F_n - p} \geq \varepsilon\bigr) \leq
        \frac{1}{4n\varepsilon^2}$ निकालिए।
  \item इस परिबंध का उपयोग करके बताइए कि कितने लोगों का सर्वेक्षण करना पड़ेगा
        ताकि प्रेक्षित \omterm{def:g10:stats:series}{आपेक्षिक बारंबारता} कम से कम $95\%$ \omterm{def:g10:proba:distribution}{प्रायिकता} से सच्चे
        अनुपात के $3$ अंक के भीतर रहे? (असली सर्वेक्षण अधिक पैने आकलन काम
        में लाते हैं, पर परिमाण की कोटि यही है।)
\end{enumerate}
\end{exercise}

\section{समस्या: जुआघर हमेशा जीतता है}

\begin{problem}[{सप्ताहांत समस्या --- बृहत् संख्याओं का नियम जुआघरों,
सर्वेक्षणों और बीमे की व्याख्या कर देता है, और रास्ते में जुआरी की भ्रांति
ध्वस्त कर देता है}]
\label{pb:g12:sums:1}
$100$ चक्करों के बाद कोई रूले खिलाड़ी लगभग आधी बार \emph{आगे} ही होता है। पर
दस लाख चक्कर चलाने वाला जुआघर इतनी निश्चितता से आगे होता है कि कोई अदालत सवाल
न उठाए। वही खेल, वही नन्ही बढ़त --- अंतर $\sqrt n$ का है, और यही इस अध्याय का
विषय है (\cref{prop:g12:sums:mean}, \cref{thm:g12:sums:chebyshev}, \cref{thm:g12:sums:lln})। यह समस्या जुआघर का बही-खाता चलाती है, किसी
चुनाव-सर्वेक्षण का आकार तय करती है, विविधीकरण का दाम लगाती है --- और जुए के
इतिहास की सबसे महँगी भ्रांति को ढहा देती है।

\medskip
\noindent\textbf{भाग I --- प्रवाह।}
\begin{enumerate}
  \item दो \omterm{def:g12:sums:indep}{स्वतंत्र} पासे: $V(X + Y)$ निकालिए (\cref{thm:g12:sums:variance})।
  \item $100$ \omterm{def:g12:sums:indep}{स्वतंत्र} पासे फेंकिए और परिणामों का औसत लीजिए: \omterm{prop:g12:sums:mean}{प्रतिदर्श माध्य}
        की \omterm{def:g12:randvar:exp}{प्रत्याशा} और \omterm{def:g11:stat:variance}{मानक विचलन} बताइए (\cref{prop:g12:sums:mean}; एक पासे के लिए $\sigma = \sqrt{35/12} \approx 1.71$)।
  \item किसी अऋणात्मक \omterm{def:g12:randvar:rv}{यादृच्छिक चर} की \omterm{def:g12:randvar:exp}{प्रत्याशा} $2$ है। मार्कोव असमिका
        (\cref{thm:g12:sums:markov}) $\P(X \geq 10)$ के बारे में क्या कहती है?
  \item प्रश्न 2 के पासों वाले औसत के लिए ब्येनेमे--चेबिशेव से $\P\left(\abs{\bar X_{100} - 3.5} \geq
        0.5\right)$ \omterm{def:g12:seq:bounded}{परिबद्ध}
        कीजिए।
  \item प्रत्याशाएँ सदा जुड़ती हैं, \omterm{def:g12:randvar:exp}{प्रसरण} केवल \omterm{def:g12:condprob:indep}{स्वतंत्रता} के अधीन: ऐसे
        परावलंबी $X, Y$ दीजिए (संकेत: $Y = -X$) जिनके लिए $V(X + Y) \neq V(X) + V(Y)$ हो।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- जुआघर का बही-खाता।}
यूरोपीय रूले, लाल पर $1$ यूरो का दाँव: \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac{18}{37}$ से $+1$ की जीत और
\omterm{def:g10:proba:distribution}{प्रायिकता} $\frac{19}{37}$ से $-1$ की हार।
\begin{enumerate}[resume]
  \item एक दाँव के लाभ की \omterm{def:g12:randvar:exp}{प्रत्याशा} और \omterm{def:g11:stat:variance}{मानक विचलन} निकालिए।
  \item कोई जुआरी $100$ \omterm{def:g12:sums:indep}{स्वतंत्र} दाँव लगाता है; मान लीजिए $G$ कुल लाभ है।
        $\E(G)$ और $\sigma(G)$ निकालिए, फिर अनुपात $\frac{\abs{\E(G)}}{\sigma(G)}$। एक चौथाई \omterm{def:g11:stat:variance}{मानक विचलन} का
        बहाव आज रात जुआरी के आगे रहने की संभावनाओं के लिए क्या अर्थ रखता है?
  \item जुआघर $1\,000\,000$ दाँव देखता है। उसकी कुल वसूली की \omterm{def:g12:randvar:exp}{प्रत्याशा} और \omterm{def:g11:stat:variance}{मानक
        विचलन} तथा वही अनुपात निकालिए। संख्या $27$ की व्याख्या कीजिए।
  \item चेबिशेव से प्रमाणित कीजिए: दस लाख दाँवों पर जुआघर के पैसे
        \emph{गँवाने} की \omterm{def:g10:proba:distribution}{प्रायिकता} \omterm{def:g12:seq:bounded}{परिबद्ध} कीजिए।
  \item दाँव लगाने की पद्धतियाँ: हारने पर दुगुना करना, आगे होते ही उठ
        जाना\,\dots\ (संभवतः यादृच्छिक) दाँव-क्रम पर \omterm{def:g12:randvar:exp}{प्रत्याशा} की रैखिकता
        (\cref{thm:g12:sums:linearity}) लगाकर समझाइए कि ऋणात्मक-बढ़त वाले खेल में \emph{हर} रणनीति
        का प्रत्याशित लाभ ऋणात्मक क्यों होता है --- कोई जिताऊ पद्धति किस
        बात का उल्लंघन करेगी?
  \item बताइए कि बृहत् संख्याओं का नियम (\cref{thm:g12:sums:lln}) लाल की \omterm{def:g10:stats:series}{आपेक्षिक बारंबारता} के
        बारे में क्या वादा करता है --- और लगातार दस लाल आने के बाद अगले
        चक्कर के बारे में वह क्या वादा \emph{नहीं} करता। भ्रांति का नाम
        बताइए।
  \item सबसे सूक्ष्म बात: नियम \emph{तनुकरण} से काम करता है, भरपाई से नहीं।
        यदि किसी शाम के बाद लाल \omterm{def:g12:randvar:exp}{प्रत्याशा} से $100$ आगे चल रहे हों, तो वह
        अधिशेष कभी ``लौटाया'' नहीं जाता --- इसके बदले $n = 10^6$ पर \emph{अंश}
        $\frac{100}{n}$ का क्या होता है, यह निकालिए, और जुआरी की भ्रांति की भूल एक
        वाक्य में फिर से लिखिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- सर्वेक्षण।}
\begin{enumerate}[resume]
  \item \cref{exo:g12:sums:8} से: बंटन-निरपेक्ष चेबिशेव परिबंध के अनुसार कितने लोगों का
        सर्वेक्षण करना पड़ेगा ताकि प्रेक्षित \omterm{def:g10:stats:series}{आपेक्षिक बारंबारता} कम से कम
        $95\,\%$ \omterm{def:g10:proba:distribution}{प्रायिकता} से सत्य के $3$ अंक के भीतर रहे?
  \item असली सर्वेक्षण संस्थान $95\,\%$ पर $\pm 3$-अंक की सीमा के लिए लगभग $1\,100$
        लोग लेते हैं: उनका सूत्र $p(1-p) \leq \frac14$ के साथ $n \approx \frac{1.96^2 \times p(1-p)}
        {\varepsilon^2}$ है। उसका मान निकालिए,
        और प्रश्न 13 से अंतर समझाइए (उतार-चढ़ाव की आकृति के बारे में चेबिशेव
        क्या नहीं जानता?)।
  \item त्रुटि-सीमा आधी करने के लिए \omterm{def:g10:proba:sample}{प्रतिदर्श} कितना बढ़ाना पड़ेगा? $1\,100$ पर
        $\pm 3$ अंक से चलकर कौन-सा \omterm{def:g10:proba:sample}{प्रतिदर्श} $\pm 1.5$ अंक देता है?
  \item चिरपरिचित उलटबाँसी: आवश्यक प्रतिदर्श-आकार ने कभी समष्टि का आकार काम
        में लिया ही नहीं --- किसी नगर के लिए भी और किसी महाद्वीप के लिए भी
        $1\,100$ लोग पर्याप्त हैं। प्रतिरूप में वह जगह बताइए जहाँ समष्टि का
        आकार अनुपस्थित है, और सर्वेक्षक जो रसोई वाली उपमा देते हैं वह भी।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- विविधीकरण।}
\begin{enumerate}[resume]
  \item \cref{pb:g12:randvar:1} का बीमाकर्ता $n$ बीमा-पत्रों पर \omterm{def:g12:randvar:exp}{प्रत्याशा} में $30n$ यूरो
        कमाता है, जबकि दावों का \omterm{def:g11:stat:variance}{मानक विचलन} लगभग $315\sqrt n$ है। किस $n$ के लिए
        प्रत्याशित लाभ अंततः दावों के एक \omterm{def:g11:stat:variance}{मानक विचलन} से आगे निकल जाता है?
        $\sqrt n$ बीमाकर्ता के लिए क्या कर रहा है?
  \item किसी एक शेयर का वार्षिक प्रतिफल $\sigma = 20\,\%$ के साथ उतार-चढ़ाव करता है।
        पैसा $25$ ऐसे \emph{\omterm{def:g12:sums:indep}{स्वतंत्र}} शेयरों में बराबर बाँट दीजिए:
        निवेश-संचय का $\sigma$ निकालिए। वित्त विविधीकरण को इकलौती मुफ़्त की
        दावत कहता है --- वह दावत ठीक-ठीक क्या है?
  \item अब मान लीजिए $25$ शेयर पूरी तरह सहसंबंधित हैं (सब एक साथ चलते
        हैं): तब निवेश-संचय का $\sigma$ क्या है? दोनों छोरों की तुलना कीजिए
        और बताइए कि विविधीकरण का हर तर्क चुपके से कौन-सी मान्यता उधार लेता
        है --- और 2008 में जब वह पूरे तंत्र में टूट गई तब क्या हुआ।
  \item समापन --- $\sqrt n$ की संगीत-रचना: योग $\sqrt n$ की तरह उतार-चढ़ाव करते हैं
        जबकि उनके \omterm{def:g11:stat:mean}{माध्य} $\frac{1}{\sqrt n}$ की तरह जमते जाते हैं; इस मुख्य स्वर को इस
        समस्या के चारों उद्योगों (जुआघर, सर्वेक्षण, बीमा, निवेश-संचय) में
        बजाइए, दोनों स्थायी दुरुपयोगकर्ताओं के नाम बताइए (जुआरी की भ्रांति
        और नक़ली \omterm{def:g12:condprob:indep}{स्वतंत्रता}), और आगे का संकेत दीजिए: उतार-चढ़ाव की
        \emph{आकृति} --- घंटी --- अगले अध्याय का \omterm{def:g12:limcont:continuity}{संतत} सितारा है, और उसकी
        पूरी प्रमेय स्नातक खंडों का ताज है।
\end{enumerate}
\end{problem}
