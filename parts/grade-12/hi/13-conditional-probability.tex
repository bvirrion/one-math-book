\chapter{सप्रतिबंध प्रायिकता और स्वतंत्रता}\label{ch:g12:condprob}

\omterm{def:g10:proba:distribution}{प्रायिकता} अनिश्चितता को संख्या में बताती है; \omterm{def:g12:condprob:cond}{सप्रतिबंध प्रायिकता} बताती है कि
सूचना उसे कितना बदल देती है। यह जान लेना कि \omterm{def:g11:prob:model}{घटना} $B$ घटित हुई, बाक़ी सभी
\omterm{def:g10:proba:events}{घटनाओं} की \omterm{def:g10:proba:distribution}{प्रायिकताएँ} नए सिरे से गढ़ देता है --- यही तंत्र बेज़ ने औपचारिक
किया और यही अदालतों तथा अख़बारों में रोज़ ग़लत काम में लाया जाता है। यह अध्याय
प्रतिबंधन के नियम और \omterm{def:g12:condprob:indep}{स्वतंत्रता} का ठीक-ठीक अर्थ तैयार करता है।

\section{प्रायिकता समष्टि (पुनरावलोकन)}

परिमित परिणामों वाले प्रयोग का प्रतिरूप एक \emph{\omterm{def:g11:prob:model}{प्रतिदर्श समष्टि}} $\Omega$
(परिणामों का समुच्चय) और एक \omterm{def:g10:proba:distribution}{प्रायिकता} $\P$ से बनता है, जो हर \omterm{def:g11:prob:model}{घटना} $A \subseteq \Omega$ को एक
संख्या $\P(A) \in \intcc{0}{1}$ देती है, असंयुक्त \omterm{def:g10:numbers:interunion}{सम्मिलनों} पर योज्य है, और जिसमें $\P(\Omega) = 1$ है। बुनियादी
नियम याद कीजिए:
\[
\P(\bar A) = 1 - \P(A),
\qquad
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B).
\]
जब सभी परिणाम समान रूप से संभावित हों, तब $\P(A) = \frac{\abs A}{\abs\Omega}$ --- और \omterm{def:g10:proba:distribution}{प्रायिकताएँ} निकालना \cref{ch:g12:comb}
की गणना-तकनीकों पर आ जाता है।

\section{सप्रतिबंध प्रायिकता}

\begin{definition}[सप्रतिबंध प्रायिकता]\label{def:g12:condprob:cond}
मान लीजिए $B$ ऐसी \omterm{def:g11:prob:model}{घटना} है जिसके लिए $\P(B) > 0$ है। \emph{$B$ दिए होने पर $A$ की
प्रायिकता}\index{सप्रतिबंध प्रायिकता} है
\[
\pcond{B}{A} = \frac{\P(A \cap B)}{\P(B)} .
\]
\end{definition}

प्रतिचित्रण $A \mapsto \pcond BA$ स्वयं एक \omterm{def:g10:proba:distribution}{प्रायिकता} है (सभी नियम लागू होते हैं); वह उस
व्यक्ति के ज्ञान की नई अवस्था दर्शाता है जिसे पता चल गया है कि $B$ घटित हुई।

\begin{proposition}[गुणन का नियम]\label{prop:g12:condprob:product}
शून्येतर \omterm{def:g10:proba:distribution}{प्रायिकताओं} वाली \omterm{def:g10:proba:events}{घटनाओं} के लिए:
\[
\P(A \cap B) = \P(B)\,\pcond{B}{A} = \P(A)\,\pcond{A}{B},
\]
और अधिक सामान्य रूप से $\P(A_1 \cap A_2 \cap A_3) = \P(A_1)\,\pcond{A_1}{A_2}\,
\pcond{A_1 \cap A_2}{A_3}$, इत्यादि।
\end{proposition}

\begin{proof}
परिभाषा को फिर से सजाइए; शृंखला-सूत्र दोहराने से निकल आता है।
\end{proof}

\begin{theorem}[कुल प्रायिकता का नियम]\label{thm:g12:condprob:total}
\index{कुल प्रायिकता का नियम}
मान लीजिए $B_1, \dots, B_n$ $\Omega$ का शून्येतर \omterm{def:g10:proba:distribution}{प्रायिकता} वाली \omterm{def:g10:proba:events}{घटनाओं} में विभाजन है। तब हर
\omterm{def:g11:prob:model}{घटना} $A$ के लिए:
\[
\P(A) = \sum_{i=1}^{n} \P(B_i)\,\pcond{B_i}{A}.
\]
\end{theorem}

\begin{proof}
समुच्चय $A \cap B_i$ जोड़े-जोड़े असंयुक्त हैं और उनका \omterm{def:g10:numbers:interunion}{सम्मिलन} $A$ है, इसलिए गुणन के
नियम से $\P(A) = \sum_i \P(A \cap B_i) = \sum_i \P(B_i)\pcond{B_i}{A}$।
\end{proof}

\begin{method}[प्रायिकता वृक्ष]\label{met:g12:condprob:tree}
\omterm{met:g10:proba:tree}{वृक्ष आरेख} सप्रतिबंध \omterm{def:g10:proba:distribution}{प्रायिकताओं} को व्यवस्थित कर देता है: हर शाखा अब तक के पथ
को \emph{दिए होने पर} अगली \omterm{def:g11:prob:model}{घटना} की \omterm{def:g10:proba:distribution}{प्रायिकता} उठाए रहती है।
\begin{itemize}
  \item किसी पत्ते (पूरे पथ) की \omterm{def:g10:proba:distribution}{प्रायिकता} उसकी शाखाओं की \omterm{def:g10:proba:distribution}{प्रायिकताओं} का
        \emph{गुणनफल} है (गुणन का नियम)।
  \item किसी \omterm{def:g11:prob:model}{घटना} की \omterm{def:g10:proba:distribution}{प्रायिकता} उसे साकार करने वाले पत्तों की \omterm{def:g10:proba:distribution}{प्रायिकताओं} का
        \emph{योग} है (कुल \omterm{def:g10:proba:distribution}{प्रायिकता})।
  \item किसी एक गाँठ से निकलने वाली शाखाओं की \omterm{def:g10:proba:distribution}{प्रायिकताओं} का योग $1$ होता
        है।
\end{itemize}
\end{method}

\begin{theorem}[बेज़ का सूत्र]\label{thm:g12:condprob:bayes}
\index{बेज़ का सूत्र}
मान लीजिए $B_1, \dots, B_n$ ऊपर की तरह $\Omega$ का विभाजन है और $A$ ऐसी \omterm{def:g11:prob:model}{घटना} है जिसके लिए
$\P(A) > 0$ है। तब
\[
\pcond{A}{B_j} =
\frac{\P(B_j)\,\pcond{B_j}{A}}{\sum_{i=1}^{n} \P(B_i)\,\pcond{B_i}{A}} .
\]
\end{theorem}

\begin{proof}
$\pcond{A}{B_j} = \frac{\P(A \cap B_j)}{\P(A)}$; अंश का प्रसार गुणन के नियम से और हर का कुल \omterm{def:g10:proba:distribution}{प्रायिकता} से कीजिए।
\end{proof}

\begin{example}[छानबीन परीक्षण]\label{ex:g12:condprob:test}
कोई रोग समष्टि के $1\%$ को प्रभावित करता है। परीक्षण उसे \omterm{def:g10:proba:distribution}{प्रायिकता} $0.99$ से
पकड़ लेता है (सुग्राहिता) और \omterm{def:g10:proba:distribution}{प्रायिकता} $0.05$ से मिथ्या धनात्मक देता है।
धनात्मक परीक्षण मिलने पर सचमुच रोग होने की \omterm{def:g10:proba:distribution}{प्रायिकता} है
\[
\frac{0.01 \times 0.99}{0.01\times0.99 + 0.99\times0.05}
= \frac{0.0099}{0.0099 + 0.0495} = \frac{1}{6} \approx 0.17 .
\]
इतने यथार्थ परीक्षण के बावजूद छह में से पाँच धनात्मक मिथ्या हैं --- क्योंकि
रोग विरला है। $\pcond{A}{B}$ को $\pcond{B}{A}$ समझ लेना ही \emph{अभियोजक की भ्रांति} है।
\end{example}

\section{स्वतंत्रता}

\begin{definition}[स्वतंत्र घटनाएँ]\label{def:g12:condprob:indep}
दो \omterm{def:g10:proba:events}{घटनाएँ} $A$ और $B$ \emph{स्वतंत्र}\index{स्वतंत्रता} हैं यदि
\[
\P(A \cap B) = \P(A)\,\P(B) .
\]
$\P(B) > 0$ होने पर यह $\pcond{B}{A} = \P(A)$ के तुल्य है: $B$ जान लेने से $A$ की \omterm{def:g10:proba:distribution}{प्रायिकता} नहीं
बदलती।
\end{definition}

\begin{proposition}\label{prop:g12:condprob:indepcompl}
यदि $A$ और $B$ \omterm{def:g12:condprob:indep}{स्वतंत्र} हों, तो $A$ और $\bar B$ भी \omterm{def:g12:condprob:indep}{स्वतंत्र} हैं (और $\bar A$ तथा
$\bar B$ भी)।
\end{proposition}

\begin{proof}
$\P(A \cap \bar B) = \P(A) - \P(A \cap B) = \P(A) - \P(A)\P(B)
= \P(A)\bigl(1 - \P(B)\bigr) = \P(A)\,\P(\bar B)$।
\end{proof}

\begin{remark}
\emph{\omterm{def:g12:condprob:indep}{स्वतंत्र}} ($\P(A\cap B) = \P(A)\P(B)$) को \emph{\omterm{def:g10:proba:operations}{असंगत}} ($A \cap B = \varnothing$) समझने की भूल मत कीजिए। शून्येतर
\omterm{def:g10:proba:distribution}{प्रायिकता} वाली दो \omterm{def:g10:proba:operations}{असंगत} \omterm{def:g10:proba:events}{घटनाएँ} कभी \omterm{def:g12:condprob:indep}{स्वतंत्र} नहीं होतीं: एक के घटित होने का
पता चलना दूसरी के न घटित होने की गारंटी दे देता है।
\end{remark}

\begin{definition}[स्वतंत्र पुनरावृत्तियाँ]\label{def:g12:condprob:repet}
जब कोई प्रयोग $n$ बार इस तरह दोहराया जाए कि हर परीक्षण का परिणाम दूसरों को
प्रभावित न करे, तब परिणामों के किसी नियत क्रम की \omterm{def:g10:proba:distribution}{प्रायिकता} अलग-अलग
\omterm{def:g10:proba:distribution}{प्रायिकताओं} का गुणनफल होती है। यही प्रतिरूप द्विपद \omterm{def:g11:prob:rv}{बंटन} (\cref{ch:g12:randvar}) के नीचे है।
\end{definition}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g12:condprob:1}
ताश के 52 पत्तों की मानक गड्डी से एक पत्ता निकाला जाता है। यह \omterm{def:g10:proba:distribution}{प्रायिकता}
निकालिए कि वह बादशाह है, यह दिए होने पर कि वह तस्वीर का पत्ता है (ग़ुलाम,
बेगम या बादशाह)। क्या \omterm{def:g10:proba:events}{घटनाएँ} ``बादशाह'' और ``पान'' \omterm{def:g12:condprob:indep}{स्वतंत्र} हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:condprob:2}
किसी कलश में 5 लाल और 3 नीली गेंदें हैं। दो गेंदें एक के बाद एक \emph{बिना}
वापसी के निकाली जाती हैं।
\begin{enumerate}
  \item \omterm{def:g10:proba:distribution}{प्रायिकता} \omterm{met:g10:proba:tree}{वृक्ष} खींचिए।
  \item यह \omterm{def:g10:proba:distribution}{प्रायिकता} निकालिए कि दोनों लाल हैं, और यह भी कि दूसरी लाल है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:condprob:3}
दो न्यायसंगत पासे फेंके जाते हैं। मान लीजिए $A$ = ``योग $7$ है'', $B$ =
``पहला पासा $3$ दिखाता है'', $C$ = ``योग $6$ है''। बताइए कि $A$ और $B$
\omterm{def:g12:condprob:indep}{स्वतंत्र} हैं या नहीं, फिर यह कि $B$ और $C$ \omterm{def:g12:condprob:indep}{स्वतंत्र} हैं या नहीं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:4}
किसी कारख़ाने में तीन मशीनें क्रमशः कुल उत्पादन का $50\%$, $30\%$ और $20\%$ बनाती
हैं, और उनकी दोष-दरें $1\%$, $2\%$ तथा $4\%$ हैं।
\begin{enumerate}
  \item उत्पादन का कितना अनुपात दोषपूर्ण है?
  \item यादृच्छिक रूप से चुनी गई कोई वस्तु दोषपूर्ण है। उसके तीसरी मशीन से
        आने की \omterm{def:g10:proba:distribution}{प्रायिकता} क्या है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:5}
\cref{ex:g12:condprob:test} में $1\%$ के बजाय रोग की किस व्यापकता $p$ पर धनात्मक परीक्षण का अर्थ
रोग की कम से कम $90\%$ संभावना होगा? असमिका हल कीजिए और टिप्पणी कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:6}
कोई पक्षपाती सिक्का \omterm{def:g10:proba:distribution}{प्रायिकता} $p \in \intoo{0}{1}$ से चित देता है। उसे तीन बार उछाला जाता
है, और उछाल \omterm{def:g12:condprob:indep}{स्वतंत्र} हैं।
\begin{enumerate}
  \item \omterm{def:g11:prob:model}{घटना} $E$ = ``ठीक दो चित'' की \omterm{def:g10:proba:distribution}{प्रायिकता} निकालिए।
  \item $p$ के किस मान के लिए $\P(E)$ \omterm{def:g10:functions:extrema}{अधिकतम} है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:condprob:7}
\emph{(मॉन्टी हॉल.)} तीन दरवाज़ों में से एक के पीछे पुरस्कार छिपा है। आप एक
दरवाज़ा चुनते हैं; संचालक, जो जानता है कि पुरस्कार कहाँ है, बचे हुए दो
दरवाज़ों में से एक खोलता है और हमेशा ख़ाली दरवाज़ा दिखाता है (यदि दोनों ख़ाली
हों तो यादृच्छिक रूप से चुनकर), और आपको दूसरे बंद दरवाज़े पर बदल जाने का
अवसर देता है। बेज़ के सूत्र से बदलने पर और न बदलने पर जीतने की \omterm{def:g10:proba:distribution}{प्रायिकता}
निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:condprob:8}
कोई सूचना-माध्यम बिट भेजता है। हर बिट \omterm{def:g12:condprob:indep}{स्वतंत्र} रूप से \omterm{def:g10:proba:distribution}{प्रायिकता} $\varepsilon = 0.1$ से शोर
के कारण पलट जाता है। किसी बिट की रक्षा के लिए उसे तीन बार भेजा जाता है और
बहुमत से गूढ़वाचन किया जाता है।
\begin{enumerate}
  \item गूढ़वाचित बिट के ग़लत होने की \omterm{def:g10:proba:distribution}{प्रायिकता} निकालिए।
  \item प्राप्त शब्द $101$ है। भेजे गए बिट के $1$ होने की \omterm{def:g10:proba:distribution}{प्रायिकता} क्या
        है? (मान लीजिए $0$ और $1$ बराबर \omterm{def:g10:proba:distribution}{प्रायिकता} से भेजे जाते हैं।)
\end{enumerate}
\end{exercise}

\section{समस्या: चिकित्सक, न्यायाधीश और स्पैम छन्नी}

\begin{problem}[{सप्ताहांत समस्या --- आधार-दर की भ्रांति के विरुद्ध बेज़ का
सूत्र: $99\,\%$ का परीक्षण $2\,\%$ का अर्थ क्यों रख सकता है, दस लाख में एक का मेल
किसी को दोषी क्यों नहीं ठहराता, और आपका डाकख़ाना कैसे सोचता है}]
\label{pb:g12:condprob:1}
$99\,\%$ स्थितियों को पकड़ लेने वाला कोई परीक्षण धनात्मक आता है --- और रोग की
सच्ची \omterm{def:g10:proba:distribution}{प्रायिकता} $2\,\%$ से भी नीचे रहती है। दस लाख में एक व्यक्ति से मेल खाने
वाला DNA चित्र मिलता है --- और संदिग्ध फिर भी संभवतः निर्दोष है। दोनों कथन
सही हैं, दोनों पर चिकित्सक और अदालतें ग़लत निर्णय दे चुकी हैं, और दोनों इस
अध्याय की एक ही पंक्ति से सुलझ जाते हैं: बेज़ का सूत्र (\cref{thm:g12:condprob:bayes})। यह समस्या
पहले कलशों पर उसका अभ्यास कराती है, फिर चिकित्सालय, अदालत और डाकख़ाने में।

\medskip
\noindent\textbf{भाग I --- प्रवाह।}
\begin{enumerate}
  \item एक न्यायसंगत सिक्का कलश I ($2$ लाल, $1$ नीली) या कलश II ($1$
        लाल, $3$ नीली) चुनता है, और फिर एक गेंद निकाली जाती है। कुल
        \omterm{def:g10:proba:distribution}{प्रायिकता} के नियम (\cref{thm:g12:condprob:total}) से $\P(\text{लाल})$ निकालिए।
  \item गेंद लाल है: $\P(\text{कलश I} \mid
        \text{लाल})$ निकालिए।
  \item पासे का एक फेंक: क्या \omterm{def:g10:proba:events}{घटनाएँ} ``सम'' और ``$\leq 4$'' \omterm{def:g12:condprob:indep}{स्वतंत्र} हैं
        (\cref{def:g12:condprob:indep})? और ``सम'' तथा ``$\leq 3$''?
  \item आग लगने पर दो \omterm{def:g12:condprob:indep}{स्वतंत्र} धुआँ-चेतावनी यंत्र प्रत्येक \omterm{def:g10:proba:distribution}{प्रायिकता} $0.9$
        से बजते हैं। \cref{prop:g12:condprob:indepcompl} का उपयोग करके कम से कम एक के बजने की \omterm{def:g10:proba:distribution}{प्रायिकता}
        निकालिए।
  \item बिना वापसी के दो पत्ते: गुणन के नियम (\cref{prop:g12:condprob:product}) से $\P(\text{दो बादशाह})$ निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- चिकित्सक।}
कोई रोग $1\,000$ में $1$ व्यक्ति को होता है। परीक्षण रोगियों का $99\,\%$ पकड़
लेता है (सुग्राहिता) पर स्वस्थ लोगों के $5\,\%$ पर भी बज जाता है (मिथ्या
धनात्मक)।
\begin{enumerate}[resume]
  \item स्वाभाविक बारंबारताएँ: $100\,000$ लोगों की कल्पना कीजिए। कितने रोगग्रस्त
        हैं, कितने स्वस्थ? हर समूह कितने धनात्मक देता है, और कुल कितने
        धनात्मक?
  \item गिनतियों से $\P(\text{बीमार} \mid \text{धनात्मक})$ निकालिए। धक्का पचा लीजिए।
  \item उसे बेज़ के सूत्र से फिर निकालिए और मेल जाँचिए।
  \item सर्वेक्षणों में अधिकांश चिकित्सक ``लगभग $99\,\%$'' उत्तर देते हैं। बताइए
        कि कौन-सी दो सप्रतिबंध \omterm{def:g10:proba:distribution}{प्रायिकताएँ} गड्डमड्ड की जा रही हैं, और समझाइए
        कि नन्ही-सी आधार-दर सच्चे उत्तर को कैसे चलाती है।
  \item एक दूसरा, \omterm{def:g12:condprob:indep}{स्वतंत्र} परीक्षण भी धनात्मक आता है। नवीनीकरण कीजिए: $1.9\,\%$
        को नई पूर्व \omterm{def:g10:proba:distribution}{प्रायिकता} मानकर $\P(\text{बीमार} \mid \text{दो धनात्मक})$ निकालिए। प्रमाणों का \emph{संचय}
        क्या कर रहा है?
  \item यदि वही परीक्षण केवल $5\,\%$ व्यापकता वाले उच्च-जोखिम समूह पर काम में
        लाया जाए, तो $\P(\text{बीमार} \mid \text{धनात्मक})$ निकालिए। विरले रोगों की व्यापक छानबीन के बारे में
        जन-स्वास्थ्य वाली सीख निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- न्यायाधीश।}
\begin{enumerate}[resume]
  \item अपराध-स्थल के किसी DNA चित्र से दस लाख में एक व्यक्ति का मेल खाता
        है। नगर के $10$ लाख निवासियों के आँकड़े-कोश में खोजकर मिला संदिग्ध
        मेल खा जाता है। नगर में निर्दोष मेलों की प्रत्याशित संख्या कितनी है?
        केवल मेल दिए होने पर $\P(\text{निर्दोष} \mid
        \text{मेल})$ मोटे तौर पर क्या है?
  \item भ्रांति का नाम बताइए --- अभियोजक ने कौन-सी \omterm{def:g10:proba:distribution}{प्रायिकता} उद्धृत की, और
        अदालत को किसकी आवश्यकता थी? --- और प्रश्न 9 से उसकी ठीक-ठीक
        समांतरता।
  \item \omterm{def:g12:condprob:indep}{स्वतंत्रता} का जाल: एक ही परिवार की दो विरली दुर्घटनाओं को कभी
        \omterm{def:g12:condprob:indep}{स्वतंत्र} मानकर उनका गुणा ``दस लाख में $73$'' बना दिया गया था, और
        एक निर्दोष माँ को सज़ा हो गई थी (सैली क्लार्क का मामला)। यदि दोनों
        \omterm{def:g10:proba:events}{घटनाओं} में पारिवारिक जोखिम-कारक साझे हों, जिससे पहली दिए होने पर
        दूसरी आधार-दर से $5$ गुना अधिक संभावित हो, तो परिमाण की कोटि फिर
        से निकालिए --- और \omterm{def:g10:proba:distribution}{प्रायिकताओं} को गुणा करने के बारे में सीख बताइए।
  \item बचाव पक्ष भी बेईमानी कर सकता है: ``दस निर्दोष मेल खाते हैं, इसलिए मेल
        कुछ भी सिद्ध नहीं करता'' बाक़ी सब कुछ अनदेखा कर देता है। बेज़ का सूत्र
        \emph{कई} प्रमाणों के साथ क्या करता है, और हर जूरी-सदस्य को कौन-सा
        एक वाक्य जानना चाहिए?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- स्पैम छन्नी।}
\begin{enumerate}[resume]
  \item आपकी डाक का $40\,\%$ स्पैम है। शब्द ``लॉटरी'' स्पैम के $15\,\%$ में और
        ईमानदार डाक के $0.5\,\%$ में आता है। किसी संदेश में वह आता है: $\P(\text{स्पैम} \mid \text{लॉटरी})$
        निकालिए।
  \item त्रुटि-सुधारक पुनरावृत्ति (\cref{exo:g12:condprob:8} आगे बढ़ाते हुए): पलटने की \omterm{def:g10:proba:distribution}{प्रायिकता}
        $\varepsilon = 0.1$ के साथ \emph{पाँच} प्रतियों का बहुमत-गूढ़वाचन किस \omterm{def:g10:proba:distribution}{प्रायिकता} से
        विफल होता है? तीन-प्रति वाले कूट से तुलना कीजिए।
  \item कोई बिट $4$ \omterm{def:g12:condprob:indep}{स्वतंत्र} शोरीले प्रसारकों से गुज़रता है, और हर प्रसारक
        उसे \omterm{def:g10:proba:distribution}{प्रायिकता} $0.1$ से पलट देता है: वह ठीक तभी सही पहुँचता है जब
        पलटावों की संख्या सम हो। वह \omterm{def:g10:proba:distribution}{प्रायिकता} दो बार निकालिए: द्विपद पदों
        को जोड़कर, और सुंदर सूत्र $\frac{1 + (1 - 2\varepsilon)^4}{2}$ से।
  \item असली स्पैम छन्नियाँ अनेक शब्दों के प्रमाण ऐसे गुणा कर देती हैं मानो
        वे \omterm{def:g12:condprob:indep}{स्वतंत्र} हों (``भोला'' बेज़)। ``जीत'' और ``लॉटरी'' जैसे शब्दों के
        लिए यह मान्यता असत्य क्यों है --- और फिर भी छन्नी अच्छा काम क्यों
        करती है? (एक-दो वाक्य।)
  \item समापन --- विश्वास के अंकगणित के रूप में बेज़: पूर्व $\times$ संभाव्यता,
        फिर से प्रसामान्यीकरण, पश्च। तीनों फ़ैसले (चिकित्सालय, अदालत,
        डाकख़ाना) एक-एक वाक्य में दोहराइए, फिर इस समस्या द्वारा उजागर तीनों
        जाल गिनाइए: भुला दी गई आधार-दरें, नक़ली \omterm{def:g12:condprob:indep}{स्वतंत्रता}, और प्रमाणों को
        एक-एक करके पढ़ना।
\end{enumerate}
\end{problem}
