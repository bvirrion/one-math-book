\chapter{Rijen}\label{ch:g12:seq}

Een \omterm{def:g12:seq:sequence}{rij} is een lijst \omterm{def:g10:numbers:sets}{reële getallen}, geïndexeerd door de \omterm{def:g10:numbers:sets}{natuurlijke getallen}.
\omterm{def:g12:seq:sequence}{Rijen} modelleren discrete evoluties --- populaties die jaar na jaar geteld
worden, saldi van een bankrekening, opeenvolgende \omterm{def:g10:numbers:approx}{benaderingen} van een getal
--- en hun limieten vormen de eerste ernstige ontmoeting met het oneindige.
Dit hoofdstuk zet de woordenschat op, samen met het principe van de inductie
en de fundamentele convergentiestellingen.

\section{Redeneren met inductie}

\begin{theorem}[Principe van de inductie]\label{thm:g12:seq:induction}
Zij $P(n)$ een uitspraak die van een \omterm{def:g10:numbers:sets}{geheel getal} $n$ afhangt, en zij
$n_0 \in \N$. Als
\begin{enumerate}
  \item \emph{(basisgeval)} $P(n_0)$ waar is, en
  \item \emph{(inductiestap)} voor elke $n \geq n_0$ uit $P(n)$ volgt dat
        $P(n+1)$,
\end{enumerate}
dan is $P(n)$ waar voor elke $n \geq n_0$.
\end{theorem}

\begin{proof}
Stel uit het ongerijmde dat de verzameling $A$ van de \omterm{def:g10:numbers:sets}{gehele getallen}
$n \geq n_0$ waarvoor $P(n)$ vals is, niet leeg is. Dan heeft $A$ een kleinste
element $m$.\footnote{Elke niet-lege deelverzameling van $\N$ heeft een
kleinste element; die eigenschap van $\N$ nemen we als axioma.} Omdat
$P(n_0)$ waar is, geldt $m > n_0$, dus $m - 1 \geq n_0$ en $m-1 \notin A$,
dus is $P(m-1)$ waar. De inductiestap toegepast op $n = m-1$ toont dan dat
$P(m)$ waar is, in tegenspraak met $m \in A$.
\end{proof}

\begin{example}\label{ex:g12:seq:bernoulli}
We bewijzen de \emph{ongelijkheid van Bernoulli}: voor elke reële $a > 0$ en
elke $n \in \N$ geldt
\[
(1+a)^n \geq 1 + na.
\]
\emph{Basisgeval.} Voor $n = 0$ zijn beide leden gelijk aan $1$.
\emph{Inductiestap.} Stel dat $(1+a)^n \geq 1+na$ voor een zekere $n \in \N$.
Omdat $1 + a > 0$, behoudt beide leden met $1+a$ vermenigvuldigen de
ongelijkheid:
\[
(1+a)^{n+1} \geq (1+na)(1+a) = 1 + (n+1)a + na^2 \geq 1 + (n+1)a .
\]
Met inductie geldt de ongelijkheid dus voor alle $n \in \N$.
\end{example}

\begin{method}[Een inductiebewijs schrijven]\label{met:g12:seq:induction}
Maak de uitspraak $P(n)$ altijd expliciet vóór je begint. Een volledig bewijs
heeft drie zichtbare delen: het basisgeval, de inductiestap (``neem $P(n)$
aan; we bewijzen $P(n+1)$'') en het besluit dat het principe van de inductie
inroept. De meest voorkomende fout is de inductiestap bewijzen zonder ooit de
hypothese $P(n)$ te gebruiken: gebeurt dat, dan is ofwel het bewijs verkeerd,
ofwel was inductie niet nodig.
\end{method}

\section{Woordenschat van de rijen}

\begin{definition}[Rij]\label{def:g12:seq:sequence}
Een \emph{rij}\index{rij} is een \omterm{def:g11:func:function}{functie} $u \colon \N \to \R$ (of van
$\{n \in \N : n \geq n_0\}$ naar $\R$). Het \omterm{def:g10:functions:function}{beeld} van $n$ wordt $u_n$
geschreven, en de rij zelf $(u_n)_{n\in\N}$ of gewoon $(u_n)$.
\end{definition}

Een \omterm{def:g12:seq:sequence}{rij} kan \emph{expliciet} gegeven zijn, door een formule $u_n = f(n)$, of
door \emph{recursie}, door haar eerste term en een betrekking
$u_{n+1} = f(u_n)$.

\begin{definition}[Monotonie]\label{def:g12:seq:monotonic}
Een \omterm{def:g12:seq:sequence}{rij} $(u_n)$ heet \emph{stijgend}\index{rij!stijgende} wanneer
$u_{n+1} \geq u_n$ voor alle $n$, \emph{\omterm{def:g10:functions:variations}{dalend}} wanneer $u_{n+1} \leq u_n$
voor alle $n$, en \emph{monotoon} wanneer ze stijgend of \omterm{def:g10:functions:variations}{dalend} is. Ze is
\emph{strikt} stijgend (respectievelijk \omterm{def:g10:functions:variations}{dalend}) wanneer de ongelijkheden
strikt zijn.
\end{definition}

\begin{method}[De monotonie van een rij bestuderen]\label{met:g12:seq:monotonicity}
Drie gebruikelijke technieken:
\begin{enumerate}
  \item bestudeer het teken van $u_{n+1} - u_n$;
  \item zijn alle termen positief, vergelijk dan $\dfrac{u_{n+1}}{u_n}$ met
        $1$;
  \item is $u_n = f(n)$ met $f$ gedefinieerd op $\intco{0}{+\infty}$, gebruik
        dan het verloop van $f$.
\end{enumerate}
\end{method}

\begin{definition}[Begrensde rij]\label{def:g12:seq:bounded}
Een \omterm{def:g12:seq:sequence}{rij} $(u_n)$ heet \emph{naar boven begrensd}\index{rij!begrensde} wanneer
er een $M \in \R$ bestaat met $u_n \leq M$ voor alle $n$; \emph{naar beneden
begrensd} wanneer er een $m \in \R$ bestaat met $u_n \geq m$ voor alle $n$; en
\emph{begrensd} wanneer beide gelden.
\end{definition}

\subsection{Rekenkundige en meetkundige rijen}

\begin{definition}[Rekenkundige en meetkundige rijen]\label{def:g12:seq:arith-geom}
Een \omterm{def:g12:seq:sequence}{rij} $(u_n)$ heet \emph{rekenkundig}\index{rij!rekenkundige} met \omterm{def:g11:seq:arithmetic}{verschil}
$r$ wanneer $u_{n+1} = u_n + r$ voor alle $n$, en
\emph{meetkundig}\index{rij!meetkundige} met \omterm{def:g11:seq:geometric}{reden} $q$ wanneer
$u_{n+1} = q\,u_n$ voor alle $n$.
\end{definition}

\begin{proposition}[Expliciete vorm en sommen]\label{prop:g12:seq:arith-geom}
Zij $n \in \N$.
\begin{enumerate}
  \item Is $(u_n)$ \omterm{def:g12:seq:arith-geom}{rekenkundig} met \omterm{def:g11:seq:arithmetic}{verschil} $r$, dan is $u_n = u_0 + nr$ en
        \[
        u_0 + u_1 + \dots + u_n = (n+1)\,\frac{u_0 + u_n}{2}.
        \]
  \item Is $(u_n)$ \omterm{def:g12:seq:arith-geom}{meetkundig} met \omterm{def:g11:seq:geometric}{reden} $q \neq 1$, dan is $u_n = u_0\, q^n$
        en
        \[
        u_0 + u_1 + \dots + u_n = u_0\,\frac{1 - q^{n+1}}{1 - q}.
        \]
\end{enumerate}
\end{proposition}

\begin{proof}
De expliciete vormen volgen met onmiddellijke inducties. Voor de
rekenkundige som schrijf je $S = u_0 + \dots + u_n$ en tel je dezelfde som in
omgekeerde volgorde erbij op: elk van de $n+1$ kolomsommen is gelijk aan
$u_0 + u_n$, dus $2S = (n+1)(u_0+u_n)$. Voor de meetkundige som bereken je
$S - qS$: alle termen vallen paarsgewijs weg behalve de eerste en de laatste,
zodat $(1-q)S = u_0(1 - q^{n+1})$.
\end{proof}

\section{Limiet van een rij}

\begin{definition}[Convergente rij]\label{def:g12:seq:limit}
Een \omterm{def:g12:seq:sequence}{rij} $(u_n)$ \emph{convergeert}\index{limiet!van een rij} naar het reële
getal $\ell$ wanneer elk open \omterm{def:g10:numbers:interval}{interval} dat $\ell$ bevat, vanaf een zekere
index alle termen $u_n$ bevat. We schrijven dan
$\lim\limits_{n\to+\infty} u_n = \ell$.

Gelijkwaardig: voor elke $\varepsilon > 0$ bestaat er een $N \in \N$ zodat
$\abs{u_n - \ell} \leq \varepsilon$ voor alle $n \geq N$.
\end{definition}

\begin{definition}[Divergentie naar oneindig]\label{def:g12:seq:infinity}
De \omterm{def:g12:seq:sequence}{rij} $(u_n)$ \emph{streeft naar $+\infty$} wanneer er voor elke $A \in \R$
een $N \in \N$ bestaat zodat $u_n \geq A$ voor alle $n \geq N$. We schrijven
$\lim\limits_{n\to+\infty} u_n = +\infty$; de definitie van
$\lim u_n = -\infty$ verloopt analoog. Van een \omterm{def:g12:seq:sequence}{rij} die niet \omterm{def:g12:seq:limit}{convergeert} zegt
men dat ze \emph{divergeert}.
\end{definition}

\begin{remark}
Een \omterm{def:g12:seq:sequence}{rij} kan divergeren zonder naar $\pm\infty$ te streven: de \omterm{def:g12:seq:sequence}{rij}
$u_n = (-1)^n$ neemt alleen de waarden $1$ en $-1$ aan en heeft geen limiet.
\end{remark}

\begin{proposition}[Uniciteit van de limiet]\label{prop:g12:seq:uniqueness}
\omterm{def:g12:seq:limit}{Convergeert} $(u_n)$, dan is haar limiet uniek.
\end{proposition}

\begin{proof}
Stel $u_n \to \ell$ en $u_n \to \ell'$ met $\ell \neq \ell'$, zeg
$\ell < \ell'$. Stel $\varepsilon = \frac{\ell' - \ell}{3} > 0$. Vanaf een
zekere index geldt $\abs{u_n - \ell} \leq \varepsilon$ en
$\abs{u_n - \ell'} \leq \varepsilon$, dus
\[
\ell' - \ell \leq \abs{\ell' - u_n} + \abs{u_n - \ell}
\leq 2\varepsilon = \tfrac{2}{3}(\ell' - \ell) < \ell' - \ell,
\]
een tegenspraak.
\end{proof}

\begin{proposition}[Bewerkingen met limieten]\label{prop:g12:seq:operations}
Zijn $(u_n)$ en $(v_n)$ \omterm{def:g12:seq:sequence}{rijen} met limieten $\ell$ en $\ell'$ (eindig of
oneindig), dan geldt, zolang het rechterlid geen onbepaalde vorm is,
\[
\lim (u_n + v_n) = \ell + \ell', \qquad
\lim (u_n v_n) = \ell\,\ell', \qquad
\lim \frac{u_n}{v_n} = \frac{\ell}{\ell'}.
\]
De onbepaalde vormen zijn $(+\infty) + (-\infty)$, $0 \times \infty$,
$\frac{\infty}{\infty}$ en $\frac{0}{0}$.
\end{proposition}

\begin{proof}
We bewijzen de somregel voor eindige limieten; de andere gevallen verlopen
analoog en blijven als oefening. Zij $\varepsilon > 0$. Er bestaan $N_1, N_2$
zodat $\abs{u_n - \ell} \leq \varepsilon/2$ voor $n \geq N_1$ en
$\abs{v_n - \ell'} \leq \varepsilon/2$ voor $n \geq N_2$. Voor
$n \geq \max(N_1, N_2)$ geeft de driehoeksongelijkheid
\[
\abs{(u_n + v_n) - (\ell + \ell')}
\leq \abs{u_n - \ell} + \abs{v_n - \ell'} \leq \varepsilon. \qedhere
\]
\end{proof}

\begin{method}[Een onbepaalde vorm opheffen]\label{met:g12:seq:indeterminate}
Sta je voor een onbepaalde vorm, haal dan de overheersende term buiten
haakjes. Bijvoorbeeld:
\[
n^2 - n = n^2\left(1 - \tfrac{1}{n}\right) \xrightarrow[n\to+\infty]{} +\infty,
\qquad
\frac{2n^2+1}{n^2 - n} = \frac{2 + 1/n^2}{1 - 1/n}
\xrightarrow[n\to+\infty]{} 2 .
\]
\end{method}

\section{Convergentiestellingen}

\begin{theorem}[Vergelijkings- en insluitstelling]\label{thm:g12:seq:squeeze}
Zij $(u_n)$, $(v_n)$, $(w_n)$ \omterm{def:g12:seq:sequence}{rijen}.
\begin{enumerate}
  \item Is $u_n \leq v_n$ vanaf een zekere index en $u_n \to +\infty$, dan is
        $v_n \to +\infty$.
  \item \emph{(Insluitstelling)}\index{insluitstelling} Is
        $u_n \leq v_n \leq w_n$ vanaf een zekere index en convergeren $(u_n)$
        en $(w_n)$ allebei naar dezelfde limiet $\ell$, dan \omterm{def:g12:seq:limit}{convergeert}
        $(v_n)$ naar~$\ell$.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} Zij $A \in \R$. Omdat $u_n \to +\infty$, bestaat er een $N$ met
$u_n \geq A$ voor $n \geq N$; door $N$ zo nodig te vergroten geldt
$v_n \geq u_n \geq A$ voor $n \geq N$.

\emph{2.} Zij $\varepsilon > 0$. Vanaf een zekere index gelden zowel
$\ell - \varepsilon \leq u_n$ als $w_n \leq \ell + \varepsilon$, dus
$\ell - \varepsilon \leq u_n \leq v_n \leq w_n \leq \ell + \varepsilon$, en
dat is $\abs{v_n - \ell} \leq \varepsilon$.
\end{proof}

\begin{example}
Voor alle $n \geq 1$ geldt $-\frac{1}{n} \leq \frac{(-1)^n}{n} \leq
\frac{1}{n}$, en beide grenzen streven naar $0$; bijgevolg is
$\frac{(-1)^n}{n} \to 0$.
\end{example}

\begin{theorem}[Stelling van de monotone convergentie]\label{thm:g12:seq:monotone}
Een stijgende \omterm{def:g12:seq:sequence}{rij} die \omterm{def:g12:seq:bounded}{naar boven begrensd} is, \omterm{def:g12:seq:limit}{convergeert}. Een dalende \omterm{def:g12:seq:sequence}{rij} die
naar beneden begrensd is, \omterm{def:g12:seq:limit}{convergeert}. Een stijgende \omterm{def:g12:seq:sequence}{rij} die niet \omterm{def:g12:seq:bounded}{naar boven
begrensd} is, streeft naar $+\infty$.
\end{theorem}

\begin{proof}[Gedeeltelijk bewijs]
We bewijzen de derde uitspraak. Zij $(u_n)$ \omterm{def:g12:seq:monotonic}{stijgend} en niet \omterm{def:g12:seq:bounded}{naar boven
begrensd}, en zij $A \in \R$. Omdat $A$ geen bovengrens is, bestaat er een $N$
met $u_N \geq A$; door de monotonie is $u_n \geq u_N \geq A$ voor alle
$n \geq N$. Bijgevolg is $u_n \to +\infty$.

De twee convergentie-uitspraken steunen op de eigenschap van de kleinste
bovengrens van $\R$; ze worden \emph{op dit niveau aangenomen} (en in het
eerste bachelorjaar bewezen).
\end{proof}

\begin{remark}
De stelling waarborgt het \emph{bestaan} van de limiet maar geeft haar waarde
niet. Een stijgende \omterm{def:g12:seq:sequence}{rij} die \omterm{def:g12:seq:bounded}{naar boven begrensd} is door $M$ \omterm{def:g12:seq:limit}{convergeert} naar
een zekere $\ell \leq M$, niet noodzakelijk naar $M$.
\end{remark}

\begin{theorem}[Limiet van meetkundige rijen]\label{thm:g12:seq:geometric}
Zij $q \in \R$.
\begin{enumerate}
  \item Is $q > 1$, dan is $q^n \to +\infty$.
  \item Is $q = 1$, dan is $q^n \to 1$.
  \item Is $\abs{q} < 1$, dan is $q^n \to 0$.
  \item Is $q \leq -1$, dan divergeert $(q^n)$ en heeft ze geen limiet.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} Schrijf $q = 1 + a$ met $a > 0$. De ongelijkheid van Bernoulli
(\cref{ex:g12:seq:bernoulli}) geeft $q^n \geq 1 + na \to +\infty$, en we
besluiten door \omterm{def:g10:algebra:equation}{vergelijking} (\cref{thm:g12:seq:squeeze}).

\emph{2.} Onmiddellijk.

\emph{3.} Is $q = 0$, dan is de bewering duidelijk. Anders geeft
$\abs{q} < 1$ dat $1/\abs{q} > 1$, dus $(1/\abs{q})^n \to +\infty$ volgens
punt 1, dus $\abs{q}^n \to 0$; en met
$-\abs{q}^n \leq q^n \leq \abs{q}^n$ besluiten we met de insluitstelling.

\emph{4.} Voor $q \leq -1$ neemt $(q^{2n})$ waarden $\geq 1$ aan terwijl
$(q^{2n+1})$ waarden $\leq -1$ aanneemt: geen enkele limiet kan beide
deelrijen aantrekken.
\end{proof}

\begin{method}[Recursieve rijen $u_{n+1} = f(u_n)$]\label{met:g12:seq:recurrent}
Om een \omterm{def:g12:seq:sequence}{rij} bepaald door $u_{n+1} = f(u_n)$ te bestuderen:
\begin{enumerate}
  \item bewijs met inductie dat $(u_n)$ in een \omterm{def:g10:numbers:interval}{interval} $I$ blijft waarop $f$
        zich netjes gedraagt (en, vaak, dat $(u_n)$ monotoon is);
  \item leid de convergentie af uit de stelling van de monotone
        convergentie;
  \item ga in de betrekking $u_{n+1} = f(u_n)$ over naar de limiet: is $f$
        continu en $u_n \to \ell \in I$, dan voldoet $\ell$ aan
        $f(\ell) = \ell$ (zie \cref{ch:g12:limcont}); los die \omterm{def:g10:algebra:equation}{vergelijking} op
        en kies de juiste \omterm{def:g11:quad:discriminant}{wortel}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g12:seq:sqrt2}
Zij $u_0 = 2$ en $u_{n+1} = \frac{1}{2}\left(u_n + \frac{2}{u_n}\right)$. Met
inductie ga je na dat $u_n \geq \sqrt{2}$ voor alle $n$ (de ongelijkheid
$\frac{1}{2}(x + 2/x) \geq \sqrt{2}$ voor $x>0$ is gelijkwaardig met
$(x - \sqrt2)^2 \geq 0$), en daarna dat $(u_n)$ \omterm{def:g10:functions:variations}{dalend} is, want
\[
u_{n+1}-u_n=\frac{2-u_n^2}{2u_n}\leq 0 .
\]
\omterm{def:g10:functions:variations}{Dalend} en naar beneden begrensd \omterm{def:g12:seq:limit}{convergeert} $(u_n)$ naar een zekere
$\ell \geq \sqrt{2}$, die moet voldoen aan
$\ell = \frac{1}{2}(\ell + 2/\ell)$, dus aan $\ell^2 = 2$. Bijgevolg is
$u_n \to \sqrt{2}$. Dat is het algoritme van Heron, al door de Babyloniërs
gebruikt; de convergentie is bijzonder snel ($u_3$ geeft $\sqrt 2$ al op acht
decimalen).
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:seq:1}
Bewijs met inductie dat voor alle $n \in \N$ geldt:
\[
1^2 + 2^2 + \dots + n^2 = \frac{n(n+1)(2n+1)}{6}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:2}
Bestudeer de monotonie van de \omterm{def:g12:seq:sequence}{rijen} die voor $n \geq 1$ bepaald zijn door
\[
a_n = \frac{n+1}{n}, \qquad
b_n = \frac{2^n}{n}, \qquad
c_n = n^2 - 10n .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:3}
Bereken de limieten van de \omterm{def:g12:seq:sequence}{rijen} met algemene termen
\[
u_n = \frac{3n^2 - n + 1}{2n^2 + 5}, \qquad
v_n = \sqrt{n+1} - \sqrt{n}, \qquad
w_n = \frac{2^n - 3^n}{3^n + 1}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:4}
Zij $(u_n)$ de \omterm{def:g11:seq:arithmetic}{rekenkundige rij} met $u_0 = 5$ en \omterm{def:g11:seq:arithmetic}{verschil} $r = 3$, en
$(v_n)$ de \omterm{def:g11:seq:geometric}{meetkundige rij} met $v_0 = 8$ en \omterm{def:g11:seq:geometric}{reden} $q = \frac{1}{2}$. Bereken
$u_n$, $v_n$, $\sum_{k=0}^{n} u_k$ en $\sum_{k=0}^{n} v_k$, en de limieten
van alle vier de uitdrukkingen voor $n \to +\infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:5}
Bereken met de insluitstelling
\[
\lim_{n\to+\infty} \frac{n + \cos n}{n + 1}
\qquad\text{en}\qquad
\lim_{n\to+\infty} \frac{n!}{n^n},
\]
waarbij $n! = 1 \times 2 \times \dots \times n$. Begrens voor de tweede
limiet $\frac{n!}{n^n}$ door een term van een \omterm{def:g11:seq:geometric}{meetkundige rij}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:6}
Zij $u_0 = 0$ en $u_{n+1} = \sqrt{u_n + 2}$ voor alle $n \in \N$.
\begin{enumerate}
  \item Bewijs met inductie dat $0 \leq u_n \leq 2$ voor alle $n$.
  \item Toon aan dat $(u_n)$ \omterm{def:g12:seq:monotonic}{stijgend} is.
  \item Leid af dat $(u_n)$ \omterm{def:g12:seq:limit}{convergeert} en bepaal haar limiet.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:7}
Een patiënt neemt elke ochtend een dosis van $1$ eenheid van een geneesmiddel.
Tijdens elke periode van 24 uur verwijdert het lichaam $40\%$ van het
aanwezige middel. Zij $u_n$ de hoeveelheid middel in het lichaam vlak na de
dosis op dag $n$, zodat $u_0 = 1$.
\begin{enumerate}
  \item Verantwoord dat $u_{n+1} = 0.6\,u_n + 1$.
  \item Zij $v_n = u_n - 2.5$. Toon aan dat $(v_n)$ \omterm{def:g12:seq:arith-geom}{meetkundig} is en leid een
        expliciete formule voor $u_n$ af.
  \item Bepaal de hoeveelheid middel in het lichaam op lange termijn.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:8}
Zij $(u_n)$ bepaald door $u_0 = 3$ en $u_{n+1} = \frac{4u_n - 1}{u_n + 2}$.
\begin{enumerate}
  \item Toon met inductie aan dat $u_n > 1$ voor alle $n \in \N$.
  \item Toon aan dat $v_n = \dfrac{1}{u_n - 1}$ een \omterm{def:g11:seq:arithmetic}{rekenkundige rij} bepaalt.
  \item Leid expliciete formules voor $v_n$ en $u_n$ af, en de limiet van
        $(u_n)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:seq:9}
Zij voor $n \geq 1$
$H_n = 1 + \frac{1}{2} + \frac{1}{3} + \dots + \frac{1}{n}$.
\begin{enumerate}
  \item Toon aan dat voor alle $n \geq 1$ geldt: $H_{2n} - H_n \geq
        \frac{1}{2}$.
  \item Leid af dat $H_{2^k} \geq 1 + \frac{k}{2}$ voor alle $k \in \N$, en
        besluit dat $H_n \to +\infty$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:seq:10}
\emph{(Aangrenzende \omterm{def:g12:seq:sequence}{rijen}.)} Twee \omterm{def:g12:seq:sequence}{rijen} $(a_n)$ en $(b_n)$ heten
\emph{aangrenzend} wanneer $(a_n)$ \omterm{def:g12:seq:monotonic}{stijgend} is, $(b_n)$ \omterm{def:g10:functions:variations}{dalend}, en
$b_n - a_n \to 0$.
\begin{enumerate}
  \item Toon aan dat $a_n \leq b_n$ voor alle $n$. (Tip: bestudeer de
        monotonie van $(b_n - a_n)$.)
  \item Toon aan dat aangrenzende \omterm{def:g12:seq:sequence}{rijen} allebei convergeren, naar dezelfde
        limiet.
  \item Toepassing: toon aan dat de \omterm{def:g12:seq:sequence}{rijen} $a_n = \sum_{k=0}^{n} \frac{1}{k!}$
        en $b_n = a_n + \frac{1}{n \cdot n!}$ ($n \geq 1$) aangrenzend zijn.
        (Hun gemeenschappelijke limiet is het getal $\eu$, bestudeerd in
        \cref{ch:g12:exp}.)
\end{enumerate}
\end{exercise}

\section{Opgave: de rij van Heron, eindelijk berecht}

\begin{problem}[{Weekendopgave --- inductie bekrachtigt, de monotone
convergentie velt het vonnis, en het tweeduizend jaar oude recept voor
$\sqrt2$ krijgt eindelijk zijn bewijs (met als toetje het wonderbaarlijke
gemiddelde van Gauss)}]
\label{pb:g12:seq:1}
Drie keer eerder kwam deze reeks het recept van Heron tegen --- middel de
schatting met $2/\text{schatting}$ --- en drie keer kon ze alleen maar
\emph{vaststellen} dat het werkt. Dit hoofdstuk bezit eindelijk de
instrumenten van het oordeel: de inductie (\cref{thm:g12:seq:induction}), de
stelling van de monotone convergentie (\cref{thm:g12:seq:monotone}) en
limieten van recursies. Het vonnis, en de bewezen snelheid, vormen het hart
van deze opgave; eromheen staan de klassieke valkuilen van de inductie, de
traagste divergentie van de wiskunde, en de snelste convergentie die Gauss
ooit vond.

\medskip
\noindent\textbf{Deel I --- Opwarmen met inductie.}
\begin{enumerate}
  \item Bewijs met inductie dat $1 + 3 + 5 + \dots + (2n - 1) = n^2$ (de trap
        van de oneven getallen, in het onderbouwvolume getekend en nu
        bekrachtigd).
  \item Bewijs met inductie dat $2^n > n$ voor elke $n \in \N$.
  \item Bewijs de ongelijkheid van Bernoulli met inductie: voor $x \geq 0$ en
        $n \in \N$ geldt $(1 + x)^n \geq 1 + nx$.
  \item De klassieke valkuil: ``alle knikkers hebben dezelfde kleur --- waar
        voor één knikker; en zijn elke $n$ knikkers altijd eenkleurig, dan
        delen bij $n + 1$ knikkers de eerste $n$ een kleur en de laatste $n$
        een kleur, dus alle $n + 1$.'' Elk kind weet dat het besluit
        onzinnig is: zoek de precieze stap waar de inductie breekt.
  \item Bewijs met inductie dat $4^n - 1$ deelbaar is door $3$ voor elke
        $n \in \N$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het proces tegen Heron.}
Zij $x_0 = 2$ en $x_{n+1} = \dfrac12\left(x_n + \dfrac{2}{x_n}\right)$.
\begin{enumerate}[resume]
  \item Bereken $x_1$, $x_2$, $x_3$ als exacte breuken (oude bekenden).
  \item Bewijs de sleutelidentiteit
        \[
        x_{n+1}^2 - 2 = \left(\frac{x_n^2 - 2}{2x_n}\right)^{\!2}
        \geq 0,
        \]
        en leid met inductie af dat $x_n > 0$ en $x_n^2 > 2$ voor elke $n$.
  \item Toon aan dat $(x_n)$ strikt \omterm{def:g10:functions:variations}{dalend} is (bereken $x_{n+1} - x_n$ en
        gebruik vraag 7).
  \item Roep de stelling van de monotone convergentie in: waarom \omterm{def:g12:seq:limit}{convergeert}
        $(x_n)$ naar een zekere limiet $L \geq 1$?
  \item Bepaal de limiet: ga in de recursie over naar de limiet
        (\cref{prop:g12:seq:operations}) en besluit dat $L = \sqrt2$.
        Formuleer het historische vonnis: na tweeduizend jaar trouwe dienst is
        \emph{bewezen} dat het recept van Heron \omterm{def:g12:seq:limit}{convergeert}.
  \item De bewezen snelheid: bewijs met $e_n = x_n - \sqrt2$ dat
        \[
        e_{n+1} = \frac{e_n^2}{2 x_n} ,
        \]
        en leid af dat $e_{n+1} \leq \frac{e_n^2}{2\sqrt2}$: de fout wordt bij
        elke stap gekwadrateerd --- het verdubbelen van de decimalen dat sinds
        het onderbouwvolume werd waargenomen, nu een stelling.
  \item Bevestig het met getallen: bereken $e_0, e_1, e_2, e_3$ (uit vraag 6)
        en ga na dat elke $\frac{e_{n+1}}{e_n^2}$ dicht bij $\frac{1}{2x_n}$
        ligt.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De traagste divergentie.}
\begin{enumerate}[resume]
  \item \Cref{exo:g12:seq:9} bewees $H_{2^k} \geq 1 + \frac k2$ voor de
        harmonische sommen. Hoeveel termen waarborgen $H_n > 10$? (Een macht
        van twee volstaat; verbaas je over haar omvang.)
  \item Daartegenover convergeren de meetkundige sommen
        $1 + \frac12 + \frac14 + \dots + \frac{1}{2^n} = 2 - \frac{1}{2^n}$
        naar $2$ (\cref{thm:g12:seq:geometric}): de intuïtie van de
        chocoladereep uit het onderbouwvolume, eindelijk een uitspraak over
        limieten. Schrijf het bewijs van twee regels uit.
  \item Ertussenin: toon aan dat de sommen
        $S_n = 1 + \frac{1}{4} + \frac{1}{9} + \dots + \frac{1}{n^2}$
        convergeren, door $\frac{1}{k^2} \leq \frac{1}{k(k-1)} =
        \frac{1}{k-1} - \frac{1}{k}$ (voor $k \geq 2$) te begrenzen, te
        telescoperen en de monotone convergentie toe te passen. (De limiet,
        $\frac{\pi^2}{6}$, is een van de wonderen van Euler, bewezen in de
        universitaire volumes.)
  \item Formuleer de moraal van de vragen 13--15 in twee zinnen: wat beslist
        ``de termen streven naar $0$'' over de convergentie van de sommen ---
        en wat niet?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Het \omterm{def:g12:seq:arith-geom}{rekenkundig-meetkundig} \omterm{def:g11:stat:mean}{gemiddelde} van
Gauss.}
Zij $a_0 = 1$, $b_0 = 2$, en
\[
a_{n+1} = \sqrt{a_n b_n},
\qquad
b_{n+1} = \frac{a_n + b_n}{2} .
\]
\begin{enumerate}[resume]
  \item Bereken $a_1, b_1, a_2, b_2$ (vijf decimalen). Wat merk je op over de
        snelheid?
  \item Toon aan dat $a_n \leq b_n$ voor elke $n$ (de ongelijkheid tussen het
        \omterm{def:g12:seq:arith-geom}{rekenkundig} en het \omterm{def:g12:seq:arith-geom}{meetkundig} \omterm{def:g11:stat:mean}{gemiddelde}, die deze reeks door en door
        kent), dat $(a_n)$ stijgt en dat $(b_n)$ daalt.
  \item Toon aan dat $b_{n+1} - a_{n+1} \leq \frac{b_n - a_n}{2}$ (ontbind
        $b_{n+1} - a_{n+1} = \frac{(\sqrt{b_n} - \sqrt{a_n})^2}{2}$ en
        vergelijk), en besluit met \cref{exo:g12:seq:10} dat de twee \omterm{def:g12:seq:sequence}{rijen}
        aangrenzend zijn: ze delen een gemeenschappelijke limiet $M(1, 2)$,
        het \emph{\omterm{def:g12:seq:arith-geom}{rekenkundig-meetkundig} \omterm{def:g11:stat:mean}{gemiddelde}}.
  \item Bereken $M(1, 2)$ op zes decimalen (hoeveel iteraties had je nodig?).
        Op 30 mei 1799 berekende Gauss $M(1, \sqrt2)$ op elf decimalen,
        herkende $\frac{\pi}{M(1,\sqrt2)}$ als een bekende integraal, en
        schreef dat er ``een nieuw veld van de analyse'' was opengegaan --- en
        dat was ook zo: de elliptische integralen, verteld in de universitaire
        volumes. Sluit af met de waargenomen convergentiesnelheden uit deze
        opgave, van de traagste naar de snelste.
\end{enumerate}
\end{problem}
