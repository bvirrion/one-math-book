\chapter{Complexe getallen}\label{ch:g12:complex}

De \omterm{def:g10:algebra:equation}{vergelijking} $x^2 = -1$ heeft geen reële oplossing. Breid je $\R$ uit met
één enkel nieuw getal $\iu$ waarvan het kwadraat $-1$ is, dan krijg je het
lichaam $\C$ van de \omterm{def:g12:complex:def}{complexe getallen}, waarin \emph{elke} veeltermvergelijking
oplossingen heeft --- en waarvan de rekenkunde de meetkunde van het vlak
codeert: verschuivingen, draaiingen en vermenigvuldigingen worden optellingen
en producten.

\section{Algebraïsche vorm}

\begin{definition}[Complexe getallen]\label{def:g12:complex:def}
De verzameling van de \emph{complexe getallen}\index{complex getal} is
\[
\C = \{a + \iu b : a, b \in \R\},
\]
waarbij $\iu$ een symbool is dat aan de enkele regel $\iu^2 = -1$ voldoet;
optellen en vermenigvuldigen gebeuren zoals bij reële veeltermen in $\iu$,
waarna $\iu^2$ tot $-1$ herleid wordt. De \omterm{def:g10:numbers:sets}{reële getallen} $a$ en $b$ zijn het
\emph{reële deel}\index{reëel deel} $\Rea(z)$ en het \emph{imaginaire
deel}\index{imaginair deel} $\Ima(z)$ van $z = a + \iu b$; deze uitdrukking
is de \emph{algebraïsche vorm}\index{algebraïsche vorm} van $z$, en ze is
uniek: $a + \iu b = a' + \iu b'$ als en slechts als $a = a'$ en $b = b'$.
\end{definition}

\begin{example}
$(2 + 3\iu)(1 - \iu) = 2 - 2\iu + 3\iu - 3\iu^2 = 5 + \iu$.
\end{example}

\begin{definition}[Complex toegevoegde]\label{def:g12:complex:conjugate}
De \emph{complex toegevoegde}\index{complex toegevoegde} van $z = a + \iu b$
is $\conj{z} = a - \iu b$.
\end{definition}

\begin{proposition}[Eigenschappen van de toevoeging]\label{prop:g12:complex:conjugate}
Voor alle $z, w \in \C$ geldt
\[
\conj{z + w} = \conj z + \conj w, \quad
\conj{zw} = \conj z\,\conj w, \quad
\conj{\conj z} = z, \quad
z + \conj z = 2\Rea(z), \quad
z - \conj z = 2\iu\,\Ima(z),
\]
en $z\conj z = a^2 + b^2 \geq 0$ voor $z = a + \iu b$. Bovendien is
$z \in \R$ als en slechts als $z = \conj z$.
\end{proposition}

\begin{proof}
Alles volgt rechtstreeks uit de \omterm{def:g12:complex:def}{algebraïsche vormen}; bijvoorbeeld
$z\conj z = (a + \iu b)(a - \iu b) = a^2 - (\iu b)^2 = a^2 + b^2$.
\end{proof}

\begin{method}[Complexe getallen delen]\label{met:g12:complex:division}
Om een quotiënt in \omterm{def:g12:complex:def}{algebraïsche vorm} te schrijven, vermenigvuldig je teller en
noemer met de \omterm{def:g12:complex:conjugate}{complex toegevoegde} van de noemer:
\[
\frac{1}{2 + \iu} = \frac{2 - \iu}{(2+\iu)(2-\iu)} = \frac{2 - \iu}{5}
= \frac25 - \frac15\iu .
\]
Elk \omterm{def:g12:complex:def}{complex getal} verschillend van nul heeft dus een inverse: $\C$ is een
\emph{lichaam}.
\end{method}

\section{Het complexe vlak, modulus en argument}

\begin{definition}[Beeldgetal, modulus]\label{def:g12:complex:modulus}
In het vlak met een \omterm{def:g10:coordgeom:system}{orthonormaal assenstelsel} is het punt $M(a, b)$ het
\emph{beeldpunt} van $z = a + \iu b$, en is $z$ het
\emph{beeldgetal}\index{beeldgetal} van $M$. De
\emph{modulus}\index{modulus} van $z$ is de afstand van $M$ tot de oorsprong:
\[
\abs{z} = \sqrt{a^2 + b^2} = \sqrt{z\conj z}.
\]
\end{definition}

\begin{proposition}[Eigenschappen van de modulus]\label{prop:g12:complex:modulus}
Voor $z, w \in \C$ geldt
\[
\abs{zw} = \abs z\,\abs w, \qquad
\abs{\tfrac zw} = \tfrac{\abs z}{\abs w}\ (w \neq 0), \qquad
\abs{\conj z} = \abs z,
\]
\[
\abs{z + w} \leq \abs{z} + \abs{w} \qquad \text{(driehoeksongelijkheid)}.
\]
Bovendien is $\abs{z_B - z_A}$ de afstand tussen de punten met beeldgetallen
$z_A$ en $z_B$.
\end{proposition}

\begin{proof}
$\abs{zw}^2 = zw\,\conj{zw} = z\conj z\, w \conj w = \abs z^2 \abs w^2$ geeft
de eerste; de tweede en de derde gaan net zo. Voor de
driehoeksongelijkheid:
\[
\abs{z+w}^2 = (z+w)(\conj z + \conj w)
= \abs z^2 + \abs w^2 + 2\Rea(z \conj w)
\leq \abs z^2 + \abs w^2 + 2\abs{z\conj w}
= \bigl(\abs z + \abs w\bigr)^2,
\]
waarbij $\Rea(u) \leq \abs u$ gebruikt wordt. De uitspraak over de afstand is
de formule van Pythagoras toegepast op de \omterm{def:g10:coordgeom:system}{coördinaten} van $z_B - z_A$.
\end{proof}

\begin{definition}[Argument, goniometrische en exponentiële vorm]\label{def:g12:complex:argument}
Zij $z \neq 0$. Een \emph{argument}\index{argument} van $z$, genoteerd
$\Arg(z)$, is elke maat $\theta$ van de hoek van de positieve reële as naar
de halfrechte $OM$; het ligt vast op $2k\pi$ na. Met $r = \abs z$ is
\[
z = r(\cos\theta + \iu\sin\theta)
\qquad\text{(goniometrische vorm)}.
\]
Voer je de notatie\index{exponentiële vorm}
\[
\eu^{\iu\theta} = \cos\theta + \iu\sin\theta
\]
in, dan wordt dit de \emph{exponentiële vorm} $z = r\,\eu^{\iu\theta}$.
\end{definition}

\begin{theorem}\label{thm:g12:complex:expform}
Voor alle $\theta, \theta' \in \R$ geldt
\[
\eu^{\iu\theta}\,\eu^{\iu\theta'} = \eu^{\iu(\theta + \theta')},
\qquad
\conj{\eu^{\iu\theta}} = \eu^{-\iu\theta},
\qquad
\abs{\eu^{\iu\theta}} = 1 .
\]
Bijgevolg geldt voor $z = r\eu^{\iu\theta}$ en $z' = r'\eu^{\iu\theta'}$
verschillend van nul:
\[
zz' = rr'\,\eu^{\iu(\theta+\theta')},
\qquad
\frac{z}{z'} = \frac{r}{r'}\,\eu^{\iu(\theta - \theta')} :
\]
\emph{moduli vermenigvuldigen zich, \omterm{def:g12:complex:argument}{argumenten} tellen op.}
\end{theorem}

\begin{proof}
De eerste identiteit \emph{is} het paar somformules
(\cref{prop:g12:trigo:addition}):
\[
(\cos\theta + \iu\sin\theta)(\cos\theta' + \iu\sin\theta')
= (\cos\theta\cos\theta' - \sin\theta\sin\theta')
+ \iu(\sin\theta\cos\theta' + \cos\theta\sin\theta').
\]
De rest volgt uit een rechtstreekse berekening.
\end{proof}

\begin{corollary}[Formule van De Moivre]\label{cor:g12:complex:demoivre}
\index{formule van De Moivre}
Voor $\theta \in \R$ en $n \in \Z$ geldt
$\bigl(\cos\theta + \iu\sin\theta\bigr)^n = \cos n\theta + \iu \sin n\theta$.
\end{corollary}

\begin{proof}
Inductie op $n \geq 0$ met \cref{thm:g12:complex:expform}; breid uit naar
$n < 0$ door inversen te nemen.
\end{proof}

\begin{method}[Van de ene vorm naar de andere]\label{met:g12:complex:forms}
Van algebraïsch naar exponentieel: bereken $r = \abs z$ en zoek daarna
$\theta$ met $\cos\theta = \frac ar$ en $\sin\theta = \frac br$ (bepaal eerst
het juiste kwadrant voordat je een inverse \omterm{def:g12:trigo:cossin}{cosinus} inroept). Van exponentieel
naar algebraïsch: werk $r\cos\theta + \iu\, r\sin\theta$ uit. Gebruik de
\omterm{def:g12:complex:def}{algebraïsche vorm} voor sommen en de \omterm{def:g12:complex:argument}{exponentiële vorm} voor producten, machten
en quotiënten.
\end{method}

\begin{example}
$z = 1 + \iu$: $\abs z = \sqrt2$ en $\cos\theta = \sin\theta =
\frac{1}{\sqrt2}$, dus $\theta = \frac\pi4$ en $z = \sqrt2\,\eu^{\iu\pi/4}$.
Bijgevolg is $z^{20} = 2^{10}\,\eu^{5\iu\pi} = -1024$.
\end{example}

\section{Vierkantsvergelijkingen en eenheidswortels}

\begin{theorem}[Vierkantsvergelijkingen met reële coëfficiënten]\label{thm:g12:complex:quadratic}
Zij $a, b, c \in \R$, $a \neq 0$, en $\Delta = b^2 - 4ac$. De \omterm{def:g10:algebra:equation}{vergelijking}
$az^2 + bz + c = 0$ heeft in $\C$ de oplossingen
\[
\begin{aligned}
&\Delta > 0:\ z = \frac{-b \pm \sqrt{\Delta}}{2a} \ (\text{twee reële}); \qquad
\Delta = 0:\ z = \frac{-b}{2a} \ (\text{dubbele}); \\
&\Delta < 0:\ z = \frac{-b \pm \iu\sqrt{-\Delta}}{2a} \ (\text{twee
toegevoegde}).
\end{aligned}
\]
\end{theorem}

\begin{proof}
Vervolledig het kwadraat:
$az^2 + bz + c = a\left(\left(z + \frac{b}{2a}\right)^2 -
\frac{\Delta}{4a^2}\right)$. Is $\Delta < 0$, dan is
$\frac{\Delta}{4a^2} = \left(\iu\frac{\sqrt{-\Delta}}{2a}\right)^2$, en het
\omterm{def:g11:seq:arithmetic}{verschil} van twee kwadraten ontbindt zoals gewoonlijk.
\end{proof}

\begin{definition}[Eenheidswortels]\label{def:g12:complex:rootsofunity}
Voor $n \geq 1$ zijn de \emph{$n$-de eenheidswortels}\index{eenheidswortels}
de oplossingen van $z^n = 1$.
\end{definition}

\begin{theorem}\label{thm:g12:complex:rootsofunity}
De \omterm{def:g10:algebra:equation}{vergelijking} $z^n = 1$ heeft precies $n$ oplossingen:
\[
\omega_k = \eu^{2\iu k\pi/n}, \qquad k = 0, 1, \dots, n-1 .
\]
Hun beeldpunten zijn de hoekpunten van een regelmatige $n$-hoek ingeschreven
in de eenheidscirkel, en voor $n \geq 2$ is hun som $0$.
\end{theorem}

\begin{proof}
Elke $\omega_k$ voldoet aan $\omega_k^n = \eu^{2\iu k\pi} = 1$, en ze zijn
paarsgewijs verschillend omdat hun \omterm{def:g12:complex:argument}{argumenten} in $\intco{0}{2\pi}$ liggen.
Omgekeerd: geldt $z^n = 1$, dan is $\abs z^n = 1$, dus $\abs z = 1$ (positief
reëel) en $z = \eu^{\iu\theta}$ met $n\theta \equiv 0 \pmod{2\pi}$, dat wil
zeggen $\theta = \frac{2k\pi}{n}$; door $k$ modulo $n$ te herleiden kom je bij
de $\omega_k$ uit. De hoekpunten liggen op gelijke hoekafstand
$\frac{2\pi}{n}$: een regelmatige $n$-hoek. Ten slotte geldt met
$\omega = \omega_1$, waarbij $\omega \neq 1$, volgens de meetkundige som
\[
\sum_{k=0}^{n-1}\omega_k = \sum_{k=0}^{n-1}\omega^k
= \frac{\omega^n - 1}{\omega - 1} = 0 . \qedhere
\]
\end{proof}

\end{example}

\section{Complexe getallen en vlakke meetkunde}

\begin{proposition}[Meetkundig woordenboek]\label{prop:g12:complex:geometry}
Zij $A, B, C, D$ punten met beeldgetallen $a, b, c, d$ ($a \neq b$,
$c \neq d$).
\begin{enumerate}
  \item De verschuiving over de \omterm{def:g11:vect:vector}{vector} met \omterm{def:g12:complex:modulus}{beeldgetal} $t$ is
        $z \mapsto z + t$.
  \item De draaiing met centrum $\omega$ (\omterm{def:g12:complex:modulus}{beeldgetal}) en hoek $\theta$ is
        $z \mapsto \omega + \eu^{\iu\theta}(z - \omega)$.
  \item De homothetie met centrum $\omega$ en factor $k \in \R^*$ is
        $z \mapsto \omega + k(z - \omega)$.
  \item $\dfrac{d - c}{b - a}$ heeft \omterm{def:g12:complex:modulus}{modulus} $\dfrac{CD}{AB}$ en als \omterm{def:g12:complex:argument}{argument}
        de hoek tussen de \omterm{def:g11:vect:vector}{vectoren} $\vect{AB}$ en $\vect{CD}$. In het
        bijzonder is $AB \perp CD$ als en slechts als $\frac{d-c}{b-a}$ zuiver
        imaginair is, en zijn $A$, $B$, $C$ \omterm{def:g11:vect:collinear}{collineair} ($C \ne A$, $C\ne B$)
        als en slechts als $\frac{c-a}{b-a} \in \R$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) en (3) herformuleren de formules in \omterm{def:g10:coordgeom:system}{coördinaten}. Voor (2):
$w \mapsto \eu^{\iu\theta} w$ vermenigvuldigt de moduli met $1$ en telt
$\theta$ bij de \omterm{def:g12:complex:argument}{argumenten} op: het is de draaiing over $\theta$ om de
oorsprong; toevoeging met de verschuiving die $\omega$ naar $0$ brengt geeft
het algemene centrum. Voor (4): bij een quotiënt worden de moduli gedeeld en
de \omterm{def:g12:complex:argument}{argumenten} afgetrokken (\cref{thm:g12:complex:expform}).
\end{proof}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:complex:1}
Schrijf in \omterm{def:g12:complex:def}{algebraïsche vorm}:
$(3 - 2\iu)(1 + 4\iu)$, \quad $\dfrac{2 + \iu}{1 - \iu}$, \quad

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:complex:2}
Los op in $\C$: (a) $z^2 - 4z + 13 = 0$; (b) $z^2 + z + 1 = 0$. Ga in elk
geval na dat de twee oplossingen toegevoegd zijn, en bereken hun product.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:complex:3}
Schrijf in \omterm{def:g12:complex:argument}{exponentiële vorm}: $z_1 = -1 + \iu$ en $z_2 = \sqrt3 - \iu$, en
bereken $\dfrac{z_1}{z_2}$ eerst in exponentiële, daarna in \omterm{def:g12:complex:def}{algebraïsche
vorm}. Leid de exacte waarde van $\cos\dfrac{11\pi}{12}$ af.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:4}
Beschrijf \omterm{def:g12:seq:arith-geom}{meetkundig} de verzameling van de punten $M$ met \omterm{def:g12:complex:modulus}{beeldgetal} $z$
waarvoor:
(a) $\abs{z - 2} = \abs{z + \iu}$;\quad
(b) $\abs{z - 1 - \iu} = 3$;\quad
(c) $\dfrac{z - 1}{z + 1}$ zuiver imaginair is.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:5}
Druk met de formule van De Moivre en het binomium van Newton $\cos 3\theta$
uit als een veelterm in $\cos\theta$, en $\sin 3\theta$ in \omterm{def:g11:func:function}{functie} van
$\sin\theta$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:6}
\emph{(Lineariseren.)} Lineariseer met
$\cos\theta = \frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}$ de uitdrukking
$\cos^3\theta$ (schrijf ze als een combinatie van $\cos k\theta$), en leid
$\displaystyle\int_0^{\pi/2}\cos^3\theta\,\dd\theta$ af.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:7}
Zij $A$ en $B$ de punten met beeldgetallen $a = 1 + \iu$ en $b = 3 + 2\iu$.
Bepaal de twee punten $C$ die de driehoek $ABC$ gelijkzijdig maken, als de
\omterm{def:g10:functions:function}{beelden} van $B$ onder de draaiingen met centrum $A$ en hoek

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:8}
Los $z^4 = -4$ op in $\C$ en teken de oplossingen. Ontbind $z^4 + 4$ in twee
tweedegraadsveeltermen met reële coëfficiënten.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:9}
Bereken voor $\theta \in \intoo{0}{2\pi}$ en $n \in \N$
\[
S = 1 + \cos\theta + \cos 2\theta + \dots + \cos n\theta
\]
door de meetkundige som $\sum_{k=0}^n \eu^{\iu k\theta}$ te berekenen en de
reële delen te nemen. (Antwoord:
$S = \dfrac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}}
\cos\frac{n\theta}{2}$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:10}
Zij $\omega = \eu^{2\iu\pi/5}$.
\begin{enumerate}
  \item Verantwoord dat $1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$.
  \item Zij $u = \omega + \omega^4$ en $v = \omega^2 + \omega^3$. Toon aan dat
        $u + v = -1$ en $uv = -1$, en leid af dat $u = \frac{-1+\sqrt5}{2}$.
  \item Besluit dat $\cos\dfrac{2\pi}{5} = \dfrac{\sqrt5 - 1}{4}$.
\end{enumerate}
\end{exercise}

\section{Opgave: de vijfhoek in de spiegel van de eenheid}

\begin{problem}[{Weekendopgave --- de vijfde eenheidswortels berekenen
$\cos 72^\circ$ exact, de gulden snede tekent het resultaat, en
vermenigvuldigen blijkt draaien te zijn}]
\label{pb:g12:complex:1}
Euclides kon de regelmatige vijfhoek construeren, maar de \omterm{def:g11:seq:geometric}{reden} \emph{waarom}
die met passer en liniaal lukt --- terwijl de nederige hoek van $20^\circ$
weerstand biedt --- bleef tweeduizend jaar verborgen. Ze schuilt in dit
hoofdstuk: de vijf vijfde \omterm{def:g12:complex:rootsofunity}{eenheidswortels}
(\cref{thm:g12:complex:rootsofunity}) hebben som nul, en die ene regel algebra
perst $\cos 72^\circ$ uit een vierkantsvergelijking --- alleen
vierkantswortels, dus construeerbaar, en uiteraard getekend door de gulden
snede. Rondom dit juweel: vlotheid, de kortere weg van De Moivre naar de
goniometrie, en vermenigvuldigen ontmaskerd als meetkunde.

\medskip
\noindent\textbf{Deel I --- Vlotheid.}
\begin{enumerate}
  \item Bereken $(2 + \iu)(3 - \iu)$, \ $\dfrac{1 + \iu}{1 - \iu}$
        (\cref{met:g12:complex:division}), en $\abs{3 + 4\iu}$.
  \item Schrijf in \omterm{def:g12:complex:argument}{exponentiële vorm}: $1 + \iu$; \ $-2$; \
        $1 - \iu\sqrt3$.
  \item Bereken $(1 + \iu)^8$ met de \omterm{def:g12:complex:argument}{exponentiële vorm}.
  \item Los $z^2 = \iu$ op, en daarna $z^2 - 2z + 5 = 0$
        (\cref{thm:g12:complex:quadratic}).
  \item \omterm{def:g12:seq:arith-geom}{Meetkundig} woordenboek (\cref{prop:g12:complex:geometry}): beschrijf
        de afbeelding $z \mapsto \iu z$, en de verzameling
        $\abs{z - (1 + \iu)} = 2$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Vijf \omterm{def:g11:quad:discriminant}{wortels}, één vijfhoek.}
Zij $\omega = \eu^{2\iu\pi/5}$ en beschouw
$1, \omega, \omega^2, \omega^3, \omega^4$.
\begin{enumerate}[resume]
  \item Ga na dat dit precies de oplossingen van $z^5 = 1$ zijn, en beschrijf
        de figuur die ze in het vlak tekenen.
  \item Bewijs dat hun som nul is. (Meetkundige som --- of vermenigvuldig de
        som met $1 - \omega$.)
  \item Neem de reële delen en leid af dat
        \[
        1 + 2\cos\frac{2\pi}{5} + 2\cos\frac{4\pi}{5} = 0 .
        \]
  \item Stel $c = \cos\frac{2\pi}{5}$ en zet vraag 8 met de formule voor de
        dubbele hoek toegepast op $\cos\frac{4\pi}{5}$ om in een
        vierkantsvergelijking in $c$; los ze op en besluit
        \[
        \cos 72^\circ = \frac{\sqrt5 - 1}{4} .
        \]
  \item Controleer numeriek, en laat daarna de gulden snede haar werk
        signeren: toon aan dat $c = \frac{1}{2\varphi}$ met
        $\varphi = \frac{1 + \sqrt5}{2}$ (\cref{pb:g10:algebra:1}) --- en
        formuleer de beroemde meetkundige echo: in een regelmatige vijfhoek is
        de diagonaal $\varphi$ keer de zijde.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De sluipwegen van De Moivre.}
\begin{enumerate}[resume]
  \item Het constructieverdict: $\cos 72^\circ$ heeft alleen
        vierkantswortels nodig (vraag 9), dus de vijfhoek is met passer en
        liniaal construeerbaar; $\cos 20^\circ$ voldoet aan een onbreekbare
        derdegraadsvergelijking (\cref{pb:g12:trigo:1}), dus de hoek van
        $60^\circ$ kan niet in drie gedeeld worden. Formuleer het criterium
        dat zich aftekent (vierkantswortels bouwbaar, derdemachtswortels
        niet); de volledige theorie ervan --- Gauss, negentien jaar, en de
        regelmatige zeventienhoek --- staat in de universitaire volumes.
  \item Los $z^4 = -4$ op met de \omterm{def:g12:complex:argument}{exponentiële vorm} en vind de ontbinding
        $z^4 + 4 = (z^2 - 2z + 2)(z^2 + 2z + 2)$ van
        \cref{exo:g12:complex:8} terug door toegevoegde \omterm{def:g11:quad:discriminant}{wortels} te koppelen.
  \item Werk $(\cos\theta + \iu\sin\theta)^3$ uit met het binomium van Newton
        en De Moivre, en leid $\cos 3\theta = 4\cos^3\theta - 3\cos\theta$ in
        drie regels opnieuw af --- vergelijk met de weg via de somformules in
        \cref{pb:g12:trigo:1}.
  \item Kruiscontrole met \cref{exo:g12:complex:9}: wat is voor
        $\theta = \frac{2\pi}{5}$ en $n = 4$ de som
        $1 + \cos\theta + \dots + \cos 4\theta$, en waarom strookt dat met
        vraag 8?
  \item Bereken de drie derdemachtswortels van $8\iu$ in \omterm{def:g12:complex:def}{algebraïsche vorm}.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Vermenigvuldigen is meetkunde.}
\begin{enumerate}[resume]
  \item Beschrijf de afbeelding $z \mapsto (1 + \iu)z$ volledig: over welke
        hoek draait ze, met welke factor vergroot ze, en wat doet ze met het
        eenheidsvierkant?
  \item Bewijs $\abs{z_1 z_2} = \abs{z_1}\,\abs{z_2}$ met toegevoegden
        (\cref{prop:g12:complex:modulus}), bereken daarna $(1+\iu)^n$ voor
        $n = 1, 2, 3, 4$ en beschrijf de spiraal die de machten tekenen.
  \item Het getal $\mathrm{j} = \eu^{2\iu\pi/3}$: toon aan dat
        $1 + \mathrm{j} + \mathrm{j}^2 = 0$, en ga op de driehoek
        $\left(1, \mathrm{j}, \mathrm{j}^2\right)$ het klassieke criterium na:
        $a + \mathrm{j} b + \mathrm{j}^2 c = 0$ geldt voor een gelijkzijdige
        driehoek $abc$ (in één van de twee oriëntaties).
  \item De hoofdstelling van de algebra (aangenomen:
        d'Alembert--Gauss): elke veelterm ontbindt volledig over $\C$. Leid
        het gevolg voor de reële wereld af --- elke reële veelterm ontbindt in
        reële factoren van graad $1$ en $2$ --- en verwijs naar vraag 12 als
        voorbeeld.
  \item Slotstuk --- het portret van $\C$, telkens één zin: getallen worden
        punten; vermenigvuldigen wordt draaien en vergroten; de
        \omterm{def:g12:complex:rootsofunity}{eenheidswortels} worden regelmatige veelhoeken; De Moivre maakt van
        machten goniometrische identiteiten; en de algebraïsche geslotenheid
        waarborgt dat geen enkele \omterm{def:g10:algebra:equation}{vergelijking} ooit een grotere wereld nodig
        heeft. Coda: welke twee beroemde getallen ontmoetten elkaar in
        vraag 10?
\end{enumerate}
\end{problem}
