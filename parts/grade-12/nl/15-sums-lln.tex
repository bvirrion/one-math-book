\chapter{Sommen van toevalsvariabelen en de wet van de grote aantallen}\label{ch:g12:sums}

Waarom wint het casino op het einde altijd, en waarom werken peilingen? Omdat
de \omterm{def:g11:stat:mean}{gemiddelden} van veel onafhankelijke toevalsgrootheden steeds minder
schommelen. Dit hoofdstuk bewijst het: de lineariteit van de
\omterm{def:g12:randvar:exp}{verwachtingswaarde}, de optelbaarheid van de \omterm{def:g12:randvar:exp}{variantie} voor onafhankelijke
variabelen, de ongelijkheid van Bienaymé--Chebyshev, en de wet van de grote
aantallen.

\section{Sommen van toevalsvariabelen}

Zijn $X$ en $Y$ twee \omterm{def:g12:randvar:rv}{toevalsvariabelen} op dezelfde eindige
\omterm{def:g11:prob:model}{uitkomstenverzameling} $\Omega$, dan is de som $X + Y$ de \omterm{def:g12:randvar:rv}{toevalsvariabele}
$\omega \mapsto X(\omega) + Y(\omega)$.

\begin{theorem}[Lineariteit van de verwachtingswaarde]\label{thm:g12:sums:linearity}
\index{verwachtingswaarde!lineariteit}
Voor alle \omterm{def:g12:randvar:rv}{toevalsvariabelen} $X$ en $Y$ op $\Omega$ en alle $a, b \in \R$
geldt
\[
\E(X + Y) = \E(X) + \E(Y), \qquad \E(aX + b) = a\E(X) + b .
\]
\emph{Er is geen enkele onderstelling van \omterm{def:g12:condprob:indep}{onafhankelijkheid} nodig.}
\end{theorem}

\begin{proof}
Schrijf de \omterm{def:g12:randvar:exp}{verwachtingswaarde} als een som over de uitkomsten: omdat
$\P(X = x) = \sum_{\omega : X(\omega) = x} \P(\{\omega\})$, geeft het
hergroeperen van de termen
$\E(X) = \sum_{\omega \in \Omega} \P(\{\omega\})\,X(\omega)$. Dan is
\[
\E(X + Y) = \sum_{\omega} \P(\{\omega\})\bigl(X(\omega) + Y(\omega)\bigr)
= \sum_{\omega} \P(\{\omega\})X(\omega)
+ \sum_{\omega} \P(\{\omega\})Y(\omega) = \E(X) + \E(Y). \qedhere
\]
\end{proof}

\begin{definition}[Onafhankelijke toevalsvariabelen]\label{def:g12:sums:indep}
$X$ en $Y$ heten \emph{onafhankelijk}\index{toevalsvariabele!onafhankelijke}
als voor alle waarden $x$ en $y$ geldt
\[
\P(X = x \text{ en } Y = y) = \P(X = x)\,\P(Y = y).
\]
Verscheidene variabelen $X_1, \dots, X_n$ heten onafhankelijk als die
productregel geldt voor elke keuze van waarden van elke deelfamilie.
\end{definition}

\begin{proposition}\label{prop:g12:sums:prodexp}
Zijn $X$ en $Y$ \omterm{def:g12:sums:indep}{onafhankelijk}, dan is $\E(XY) = \E(X)\,\E(Y)$.
\end{proposition}

\begin{proof}
\[
\begin{aligned}
\E(XY) &= \sum_{x, y} xy\;\P(X = x \text{ en } Y = y)
= \sum_{x, y} xy\,\P(X=x)\P(Y=y) \\
&= \Bigl(\sum_x x\P(X=x)\Bigr)\Bigl(\sum_y y\P(Y=y)\Bigr). \qedhere
\end{aligned}
\]
\end{proof}

\begin{theorem}[Variantie van een som]\label{thm:g12:sums:variance}
\index{variantie!van een som}
Zijn $X$ en $Y$ \emph{\omterm{def:g12:sums:indep}{onafhankelijk}}, dan geldt
\[
\V(X + Y) = \V(X) + \V(Y).
\]
Algemener geldt voor onafhankelijke $X_1, \dots, X_n$:
$\V(X_1 + \dots + X_n) = \V(X_1) + \dots + \V(X_n)$.
\end{theorem}

\begin{proof}
Met König--Huygens (\cref{prop:g12:randvar:konig}) en de lineariteit:
\begin{align*}
\V(X + Y) &= \E\bigl((X+Y)^2\bigr) - \bigl(\E X + \E Y\bigr)^2\\
&= \E(X^2) + 2\E(XY) + \E(Y^2)
- \E(X)^2 - 2\E(X)\E(Y) - \E(Y)^2\\
&= \V(X) + \V(Y) + 2\bigl(\E(XY) - \E(X)\E(Y)\bigr),
\end{align*}
en het laatste haakje verdwijnt voor onafhankelijke variabelen
(\cref{prop:g12:sums:prodexp}). Het algemene geval volgt met inductie.
\end{proof}

\begin{example}[De binomiale verdeling opnieuw bekeken]\label{ex:g12:sums:binomial}
Een binomiale variabele $X \sim \mathcal B(n, p)$ is een som
$X = X_1 + \dots + X_n$ van $n$ onafhankelijke variabelen van Bernoulli. Dus
geldt, structureel,
\[
\E(X) = np, \qquad \V(X) = np(1-p),
\]
waarmee \cref{thm:g12:randvar:binomial} zonder enige berekening teruggevonden
wordt.
\end{example}

\begin{proposition}[Steekproefgemiddelde]\label{prop:g12:sums:mean}
Zij $X_1, \dots, X_n$ \omterm{def:g12:sums:indep}{onafhankelijk} met dezelfde verdeling als $X$
(\omterm{def:g12:randvar:exp}{verwachtingswaarde} $\mu$, \omterm{def:g12:randvar:exp}{variantie} $\sigma^2$), en zij
$M_n = \frac{X_1 + \dots + X_n}{n}$ hun
\emph{steekproefgemiddelde}\index{steekproefgemiddelde}. Dan geldt
\[
\E(M_n) = \mu, \qquad \V(M_n) = \frac{\sigma^2}{n}, \qquad
\sigma(M_n) = \frac{\sigma}{\sqrt n}.
\]
\end{proposition}

\begin{proof}
De lineariteit geeft $\E(M_n) = \frac{n\mu}{n} = \mu$; de \omterm{def:g12:condprob:indep}{onafhankelijkheid}
geeft $\V(X_1 + \dots + X_n) = n\sigma^2$, en delen door $n$ schaalt de
\omterm{def:g12:randvar:exp}{variantie} met $\frac{1}{n^2}$ (\cref{prop:g12:randvar:affine}).
\end{proof}

De $\frac{\sigma}{\sqrt n}$ is de fundamentele \emph{vierkantswortelwet}: om
de schommelingen van een \omterm{def:g11:stat:mean}{gemiddelde} te halveren, moet je de \omterm{def:g10:proba:sample}{steekproef}
verviervoudigen.

\section{Concentratieongelijkheden}

\begin{theorem}[Ongelijkheid van Markov]\label{thm:g12:sums:markov}
\index{ongelijkheid van Markov}
Geldt $X \geq 0$ en $a > 0$, dan is
\[
\P(X \geq a) \leq \frac{\E(X)}{a}.
\]
\end{theorem}

\begin{proof}
Houd in $\E(X) = \sum_i p_i x_i$ (alle termen zijn niet-negatief) alleen de
termen met $x_i \geq a$ over: elk daarvan is minstens $a\,p_i$, dus
$\E(X) \geq a \sum_{x_i \geq a} p_i = a\,\P(X \geq a)$.
\end{proof}

\begin{theorem}[Ongelijkheid van Bienaymé--Chebyshev]\label{thm:g12:sums:chebyshev}
\index{ongelijkheid van Bienaymé--Chebyshev}
Voor elke \omterm{def:g12:randvar:rv}{toevalsvariabele} $X$ en elke $\varepsilon > 0$ geldt
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
\leq \frac{\V(X)}{\varepsilon^2}.
\]
\end{theorem}

\begin{proof}
Pas de ongelijkheid van Markov toe op de niet-negatieve variabele
$Y = (X - \E(X))^2$ met $a = \varepsilon^2$:
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
= \P(Y \geq \varepsilon^2)
\leq \frac{\E(Y)}{\varepsilon^2} = \frac{\V(X)}{\varepsilon^2}. \qedhere
\]
\end{proof}

\begin{theorem}[Wet van de grote aantallen]\label{thm:g12:sums:lln}
\index{wet van de grote aantallen}
Zij $X_1, \dots, X_n$ \omterm{def:g12:sums:indep}{onafhankelijk} met dezelfde verdeling
(\omterm{def:g12:randvar:exp}{verwachtingswaarde} $\mu$, \omterm{def:g12:randvar:exp}{variantie} $\sigma^2$) en zij $M_n$ hun
\omterm{prop:g12:sums:mean}{steekproefgemiddelde}. Voor elke $\varepsilon > 0$ geldt
\[
\P\bigl(\abs{M_n - \mu} \geq \varepsilon\bigr)
\leq \frac{\sigma^2}{n\,\varepsilon^2}
\xrightarrow[n \to +\infty]{} 0 .
\]
Het \omterm{prop:g12:sums:mean}{steekproefgemiddelde} concentreert zich rond de \omterm{def:g12:randvar:exp}{verwachtingswaarde}.
\end{theorem}

\begin{proof}
Bienaymé--Chebyshev toegepast op $M_n$, waarvan de \omterm{def:g12:randvar:exp}{verwachtingswaarde} $\mu$ is
en de \omterm{def:g12:randvar:exp}{variantie} $\frac{\sigma^2}{n}$ (\cref{prop:g12:sums:mean}).
\end{proof}

\begin{remark}
Deze stelling is de brug tussen de kansrekening en de statistiek: de
\emph{\omterm{def:g10:stats:series}{frequentie}} van een \omterm{def:g11:prob:model}{gebeurtenis} over vele onafhankelijke herhalingen
nadert haar \emph{\omterm{def:g10:proba:distribution}{kans}} (neem $X_i$ de indicator van de \omterm{def:g11:prob:model}{gebeurtenis}, zodat
$\mu = p$). Ze rechtvaardigt het schatten van een \omterm{def:g10:proba:distribution}{kans} door simulatie (de
\emph{methode van Monte Carlo}) en van een aandeel in een populatie door een
peiling --- kwantitatief: zie \cref{ch:g12:contdist}.
\end{remark}

\begin{method}[Bienaymé--Chebyshev gebruiken]\label{met:g12:sums:chebyshev}
Om $\P(\abs{M_n - \mu} \geq \varepsilon) \leq \alpha$ te waarborgen, volstaat
$n \geq \frac{\sigma^2}{\alpha\,\varepsilon^2}$. De grens is grof (echte
schommelingen zijn meestal veel kleiner), maar volkomen algemeen: ze vraagt
niets over de verdeling behalve een eindige \omterm{def:g12:randvar:exp}{variantie}.
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:sums:1}
Er wordt met twee zuivere dobbelstenen gegooid; zij $S$ de som. Bereken
$\E(S)$ met de lineariteit (niet met de verdeling van $S$!); bereken daarna
$\V(S)$ met de \omterm{def:g12:condprob:indep}{onafhankelijkheid}, gegeven dat één zuivere dobbelsteen
\omterm{def:g12:randvar:exp}{variantie} $\frac{35}{12}$ heeft.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sums:2}
Zij $X \sim \mathcal B(100,\ 0.5)$. Begrens $\P(X \geq 75)$ met de
ongelijkheid van Markov, en daarna $\P(\abs{X - 50} \geq 25)$ met
Bienaymé--Chebyshev. Vergelijk.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sums:3}
Een zuivere munt wordt $n$ keer opgeworpen en $F_n$ noteert de \omterm{def:g10:stats:series}{frequentie} van
kruis. Hoe groot moet $n$ zijn opdat volgens de grens van
Bienaymé--Chebyshev $\P\bigl(\abs{F_n - 0.5} \geq 0.05\bigr) \leq 0.05$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:4}
Een belegster verdeelt haar kapitaal gelijk over $n$ onafhankelijke
beleggingen, elk met verwacht rendement $\mu = 5\%$ en \omterm{def:g11:stat:variance}{standaardafwijking}
$\sigma = 20\%$. Bereken de \omterm{def:g12:randvar:exp}{verwachtingswaarde} en de \omterm{def:g11:stat:variance}{standaardafwijking} van
het rendement $M_n$ van de portefeuille, en het aantal beleggingen dat nodig
is om de \omterm{def:g11:stat:variance}{standaardafwijking} onder $4\%$ te brengen. Welk financieel principe
illustreert dit?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:5}
Zij $X$ en $Y$ \omterm{def:g12:sums:indep}{onafhankelijk} en allebei uniform verdeeld op $\{1, 2, 3\}$.
\begin{enumerate}
  \item Geef de verdeling van $S = X + Y$ en bereken $\E(S)$ en $\V(S)$
        rechtstreeks daaruit.
  \item Vind beide waarden terug met de lineariteit en de optelbaarheid van
        de \omterm{def:g12:randvar:exp}{variantie}.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:6}
Toon aan dat de optelbaarheid van de \omterm{def:g12:randvar:exp}{variantie} zonder \omterm{def:g12:condprob:indep}{onafhankelijkheid} kan
falen: bereken $\V(X + Y)$ voor $Y = X$ en vergelijk met $\V(X) + \V(Y)$. Voor
welke variabelen $X$ geldt $\V(2X) = 2\V(X)$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:7}
Er rijst een vermoeden dat een dobbelsteen verzwaard is. Hij wordt $1200$ keer
geworpen en toont $260$ keer een zes ($f = 0.2167$ in plaats van
$\frac16 \approx 0.1667$). Begrens onder de hypothese dat de dobbelsteen
zuiver is $\P\bigl(\abs{F_n - \frac16} \geq 0.05\bigr)$ met
Bienaymé--Chebyshev, en bespreek of die hypothese aannemelijk is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:sums:8}
\emph{(Een betere ongelijkheid voor de munt.)} Zij $X \sim \mathcal B(n,\ p)$
en $F_n = \frac Xn$.
\begin{enumerate}
  \item Toon aan dat $p(1-p) \leq \frac14$ voor $p \in \intcc{0}{1}$.
  \item Leid de grens af die van de verdeling \omterm{def:g12:sums:indep}{onafhankelijk} is:
        $\P\bigl(\abs{F_n - p} \geq \varepsilon\bigr) \leq
        \frac{1}{4n\varepsilon^2}$.
  \item Hoeveel mensen moeten er bevraagd worden opdat de waargenomen
        \omterm{def:g10:stats:series}{frequentie} met \omterm{def:g10:proba:distribution}{kans} minstens $95\%$ binnen $3$ punten van het echte
        aandeel ligt, volgens deze grens? (Echte peilingen gebruiken scherpere
        schattingen, maar de grootteorde klopt.)
\end{enumerate}
\end{exercise}

\section{Opgave: de bank wint altijd}

\begin{problem}[{Weekendopgave --- de wet van de grote aantallen verklaart
casino's, peilingen en verzekeringen, en sloopt onderweg de denkfout van de
speler}]
\label{pb:g12:sums:1}
Een roulettespeler staat na $100$ draaien bijna even vaak wel als niet
\emph{voor}. Het casino dat een miljoen draaien laat lopen, staat voor met een
zekerheid die geen enkele rechtbank in twijfel zou trekken. Hetzelfde spel,
hetzelfde piepkleine voordeel --- het \omterm{def:g11:seq:arithmetic}{verschil} is $\sqrt n$, en dat is het
onderwerp van dit hoofdstuk (\cref{prop:g12:sums:mean},
\cref{thm:g12:sums:chebyshev}, \cref{thm:g12:sums:lln}). Deze opgave doet de
boekhouding van het casino, bepaalt de omvang van een verkiezingspeiling,
prijst de spreiding van risico --- en ontmantelt de duurste denkfout uit de
geschiedenis van het gokken.

\medskip
\noindent\textbf{Deel I --- Vlotheid.}
\begin{enumerate}
  \item Twee onafhankelijke dobbelstenen: bereken $V(X + Y)$
        (\cref{thm:g12:sums:variance}).
  \item Gooi $100$ onafhankelijke dobbelstenen en middel de uitkomsten: geef
        de \omterm{def:g12:randvar:exp}{verwachtingswaarde} en de \omterm{def:g11:stat:variance}{standaardafwijking} van het
        \omterm{prop:g12:sums:mean}{steekproefgemiddelde} (\cref{prop:g12:sums:mean}; voor één dobbelsteen
        is $\sigma = \sqrt{35/12} \approx 1.71$).
  \item Een niet-negatieve \omterm{def:g12:randvar:rv}{toevalsvariabele} heeft \omterm{def:g12:randvar:exp}{verwachtingswaarde} $2$. Wat
        zegt de ongelijkheid van Markov (\cref{thm:g12:sums:markov}) over
        $\P(X \geq 10)$?
  \item Begrens $\P\left(\abs{\bar X_{100} - 3.5} \geq 0.5\right)$ voor het
        \omterm{def:g11:stat:mean}{gemiddelde} van de dobbelstenen uit vraag 2 met
        Bienaymé--Chebyshev.
  \item \omterm{def:g12:randvar:exp}{Verwachtingswaarden} tellen altijd op, \omterm{def:g12:randvar:exp}{varianties} alleen onder
        \omterm{def:g12:condprob:indep}{onafhankelijkheid}: geef afhankelijke $X$ en $Y$ (tip: $Y = -X$)
        waarvoor $V(X + Y) \neq V(X) + V(Y)$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De boeken van de bank.}
Europese roulette, inzet van $1$ euro op rood: winst $+1$ met \omterm{def:g10:proba:distribution}{kans}
$\frac{18}{37}$, verlies $-1$ met \omterm{def:g10:proba:distribution}{kans} $\frac{19}{37}$.
\begin{enumerate}[resume]
  \item Bereken de \omterm{def:g12:randvar:exp}{verwachtingswaarde} en de \omterm{def:g11:stat:variance}{standaardafwijking} van de winst op
        één inzet.
  \item Een speler zet $100$ keer \omterm{def:g12:sums:indep}{onafhankelijk} in; zij $G$ de totale winst.
        Bereken $\E(G)$ en $\sigma(G)$, en daarna de verhouding
        $\frac{\abs{\E(G)}}{\sigma(G)}$. Wat betekent een afdrijving van een
        kwart \omterm{def:g11:stat:variance}{standaardafwijking} voor de \omterm{def:g10:proba:distribution}{kans} dat de speler vanavond
        voorstaat?
  \item Het casino ziet $1\,000\,000$ inzetten. Bereken de \omterm{def:g12:randvar:exp}{verwachtingswaarde}
        en de \omterm{def:g11:stat:variance}{standaardafwijking} van zijn totale opbrengst, en dezelfde
        verhouding. Duid het getal $27$.
  \item Bevestig met Chebyshev: begrens de \omterm{def:g10:proba:distribution}{kans} dat het casino over die
        miljoen inzetten \emph{verlies} lijdt.
  \item Speelsystemen: verdubbelen na verlies, stoppen zodra je voorstaat\,\dots\
        Leg met de lineariteit van de \omterm{def:g12:randvar:exp}{verwachtingswaarde}
        (\cref{thm:g12:sums:linearity}) over de (eventueel toevallige)
        opeenvolging van inzetten uit waarom \emph{elke} strategie in een spel
        met negatief voordeel een negatieve verwachte winst heeft --- wat zou
        een winnend systeem schenden?
  \item Formuleer wat de wet van de grote aantallen
        (\cref{thm:g12:sums:lln}) belooft over de \omterm{def:g10:stats:series}{frequentie} van rood --- en
        wat ze \emph{niet} belooft over het volgende draaien na tien keer rood
        na elkaar. Noem de denkfout.
  \item Het subtielste punt: de wet werkt door \emph{verdunning}, niet door
        vereffening. Lopen de rode uitkomsten na een avond $100$ voor op de
        \omterm{def:g12:randvar:exp}{verwachtingswaarde}, dan wordt dat overschot nooit ``terugbetaald'' ---
        bereken wat er in de plaats met het \emph{aandeel} $\frac{100}{n}$
        gebeurt tegen $n = 10^6$, en herschrijf de vergissing van de speler in
        één zin.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Peilingen.}
\begin{enumerate}[resume]
  \item Uit \cref{exo:g12:sums:8}: hoeveel mensen moeten er bevraagd worden
        opdat de waargenomen \omterm{def:g10:stats:series}{frequentie} met \omterm{def:g10:proba:distribution}{kans} minstens $95\,\%$ binnen $3$
        punten van de waarheid valt, volgens de grens van Chebyshev die van de
        verdeling \omterm{def:g12:sums:indep}{onafhankelijk} is?
  \item Echte peilingsbureaus gebruiken ongeveer $1\,100$ mensen voor een
        marge van $\pm 3$ punten op $95\,\%$: hun formule is
        $n \approx \frac{1.96^2 \times p(1-p)}{\varepsilon^2}$ met
        $p(1-p) \leq \frac14$. Bereken ze, en verklaar de kloof met vraag 13
        (wat weet Chebyshev niet over de vorm van de schommelingen?).
  \item Hoe moet de \omterm{def:g10:proba:sample}{steekproef} groeien om de foutenmarge te halveren? Welke
        \omterm{def:g10:proba:sample}{steekproef} geeft, vertrekkend van $\pm 3$ punten bij $1\,100$, een
        marge van $\pm 1.5$ punten?
  \item De klassieke tegenintuïtie: de vereiste omvang van de \omterm{def:g10:proba:sample}{steekproef}
        gebruikte nooit de omvang van de populatie --- $1\,100$ mensen
        volstaan voor een stad of voor een werelddeel. Wijs de plaats in het
        model aan waar de omvang van de populatie ontbreekt, en geef de
        keukenvergelijking die peilers gebruiken.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Spreiding van risico.}
\begin{enumerate}[resume]
  \item De verzekeraar uit \cref{pb:g12:randvar:1} verdient in
        \omterm{def:g12:randvar:exp}{verwachtingswaarde} $30n$ euro op $n$ polissen, met een
        \omterm{def:g11:stat:variance}{standaardafwijking} van de schade van ongeveer $315\sqrt n$. Voor welke
        $n$ overtreft de verwachte winst eindelijk één \omterm{def:g11:stat:variance}{standaardafwijking} van
        de schade? Wat doet de $\sqrt n$ voor de verzekeraar?
  \item Het jaarrendement van één aandeel schommelt met $\sigma = 20\,\%$.
        Verdeel het geld gelijk over $25$ \emph{onafhankelijke} zulke
        aandelen: bereken de $\sigma$ van de portefeuille. De financiële
        wereld noemt spreiding het enige gratis maal --- wat is dat maal
        precies?
  \item Laat de $25$ aandelen nu volmaakt gecorreleerd zijn (ze bewegen
        allemaal samen): wat is dan de $\sigma$ van de portefeuille? Vergelijk
        de twee uitersten en zeg welke onderstelling elk pleidooi voor
        spreiding stiekem huurt --- en wat er gebeurde toen die in 2008
        systeembreed faalde.
  \item Slotstuk --- de symfonie van de $\sqrt n$: sommen schommelen als
        $\sqrt n$ terwijl hun \omterm{def:g11:stat:mean}{gemiddelden} tot rust komen als
        $\frac{1}{\sqrt n}$; speel het thema door de vier bedrijfstakken van
        deze opgave (casino, peiling, verzekering, portefeuille), noem de twee
        vaste misbruikers (de denkfout van de speler en de valse
        \omterm{def:g12:condprob:indep}{onafhankelijkheid}), en kijk vooruit: de \emph{vorm} van de
        schommelingen --- de klok --- is de continue ster van het volgende
        hoofdstuk, en haar volledige stelling bekroont de universitaire
        volumes.
\end{enumerate}
\end{problem}
