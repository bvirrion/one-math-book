\chapter{Goniometrische functies}\label{ch:g12:trigo}

De \omterm{def:g11:func:function}{functies} $\cos$ en $\sin$ zetten de meetkunde van de cirkel om in analyse.
Gedefinieerd door de reële lijn om de eenheidscirkel te rollen, zijn ze het
oerbeeld van de periodieke \omterm{def:g11:func:function}{functies}, en hun \omterm{def:g12:deriv:derivative}{afgeleiden} maken hen tot
oplossingen van de \omterm{def:g10:algebra:equation}{vergelijking} van de trillingen $y'' = -y$.

\section{De eenheidscirkel}

\begin{definition}[Cosinus en sinus]\label{def:g12:trigo:cossin}
Zij $\mathcal{C}$ de cirkel met straal 1 en middelpunt de oorsprong van een
\omterm{def:g10:coordgeom:system}{orthonormaal assenstelsel}. Koppel aan elke reële $t$ het punt $M(t)$ dat je
krijgt door vanaf het punt $I(1,0)$ een lengte $\abs{t}$ langs
$\mathcal{C}$ op te rollen, tegen de wijzers van de klok in wanneer
$t \geq 0$ en met de wijzers mee anders. Dan zijn \emph{$\cos t$ en $\sin t$
de coördinaten van $M(t)$}\index{cosinus}\index{sinus}:
\[
M(t) = (\cos t, \sin t).
\]
\end{definition}

Omdat de cirkel omtrek $2\pi$ heeft, blijft het punt $M(t)$ ongewijzigd
wanneer $t$ met $2\pi$ toeneemt; en omdat $M(t)$ op de eenheidscirkel ligt:

\begin{proposition}[Fundamentele identiteiten]\label{prop:g12:trigo:identities}
Voor alle $t \in \R$ en $k \in \Z$ geldt
\[
\cos^2 t + \sin^2 t = 1, \qquad
\cos(t + 2k\pi) = \cos t, \qquad
\sin(t + 2k\pi) = \sin t,
\]
\[
\cos(-t) = \cos t, \qquad \sin(-t) = -\sin t,
\]
\[
\cos(\pi - t) = -\cos t, \quad \sin(\pi - t) = \sin t, \quad
\cos\left(\tfrac{\pi}{2} - t\right) = \sin t, \quad
\sin\left(\tfrac{\pi}{2} - t\right) = \cos t .
\]
\end{proposition}

\begin{proof}
De eerste identiteit is de \omterm{def:g10:algebra:equation}{vergelijking} van de eenheidscirkel; de andere
drukken de symmetrieën van de cirkel uit: $M(-t)$ is het spiegelbeeld van
$M(t)$ om de $x$-as, $M(\pi - t)$ om de $y$-as, en $M(\frac\pi2 - t)$ om de
rechte $y = x$.
\end{proof}

De waarden die je uit het hoofd moet kennen:
\begin{center}
\begin{tabular}{c|ccccc}
$t$ & $0$ & $\dfrac{\pi}{6}$ & $\dfrac{\pi}{4}$ & $\dfrac{\pi}{3}$ & $\dfrac{\pi}{2}$\\[6pt]
\hline\rule{0pt}{14pt}%
$\cos t$ & $1$ & $\dfrac{\sqrt3}{2}$ & $\dfrac{\sqrt2}{2}$ & $\dfrac12$ & $0$\\[6pt]
$\sin t$ & $0$ & $\dfrac12$ & $\dfrac{\sqrt2}{2}$ & $\dfrac{\sqrt3}{2}$ & $1$
\end{tabular}
\end{center}

\begin{proposition}[Somformules]\label{prop:g12:trigo:addition}
Voor alle $a, b \in \R$ geldt
\begin{align*}
\cos(a+b) &= \cos a \cos b - \sin a \sin b, &
\sin(a+b) &= \sin a \cos b + \cos a \sin b,\\
\cos(a-b) &= \cos a \cos b + \sin a \sin b, &
\sin(a-b) &= \sin a \cos b - \cos a \sin b.
\end{align*}
In het bijzonder is $\cos 2a = 2\cos^2 a - 1 = 1 - 2\sin^2 a$ en
$\sin 2a = 2 \sin a \cos a$.
\end{proposition}

\begin{proof}
De \omterm{def:g11:vect:vector}{vectoren} $\vect{u} = (\cos a, \sin a)$ en $\vect{v} = (\cos b, \sin b)$
hebben lengte $1$ en maken een hoek $a - b$; hun \omterm{def:g11:scal:dot}{scalair product} is zowel
$\cos a\cos b + \sin a \sin b$ (\omterm{def:g10:coordgeom:system}{coördinaten}) als
$\norm{\vect u}\,\norm{\vect v}\cos(a-b) = \cos(a - b)$ (meetkunde), en dat
is de derde formule. $b$ door $-b$ vervangen geeft de eerste; de formules
voor de \omterm{def:g12:trigo:cossin}{sinus} volgen met $\sin x = \cos(\frac\pi2 - x)$. De formules voor de
dubbele hoek zijn het geval $b = a$, gecombineerd met
$\cos^2 + \sin^2 = 1$.
\end{proof}

\section{Analytische studie}

\begin{lemma}[Fundamentele limiet]\label{lem:g12:trigo:sinxx}
\[
\lim_{t \to 0} \frac{\sin t}{t} = 1
\qquad\text{en}\qquad
\lim_{t \to 0} \frac{\cos t - 1}{t} = 0 .
\]
\end{lemma}

\begin{proof}
Vergelijk voor $0 < t < \frac{\pi}{2}$ drie oppervlakten in de
eenheidscirkel: de driehoek $OIM(t)$, de cirkelsector $OIM(t)$, en de
rechthoekige driehoek met basis $OI$ en hoogte $\tan t$:
\[
\frac{\sin t}{2} \leq \frac{t}{2} \leq \frac{\tan t}{2}.
\]

\end{tikzpicture}

{\small De drie geneste oppervlakten: de driehoek $OIM$ (blauw, oppervlakte
$\frac{\sin t}{2}$), de cirkelsector (begrensd door de rode boog, oppervlakte
$\frac{t}{2}$) en de rechthoekige driehoek met hoogte $\tan t$ (oranje,
oppervlakte $\frac{\tan t}{2}$).}
\end{omfigure}

De eerste ongelijkheid geeft $\frac{\sin t}{t} \leq 1$; de tweede geeft
$\cos t \leq \frac{\sin t}{t}$. Voor $t \to 0^+$ is $\cos t \to 1$
(\omterm{def:g12:limcont:continuity}{continuïteit}), en de insluitstelling levert $\frac{\sin t}{t} \to 1$; de
pariteit breidt dat uit tot $t \to 0$. Voor de tweede limiet:
\[
\frac{\cos t - 1}{t} = \frac{\cos^2 t - 1}{t(\cos t + 1)}
= -\frac{\sin t}{t}\cdot\frac{\sin t}{\cos t + 1}
\xrightarrow[t\to0]{} -1 \cdot \frac{0}{2} = 0. \qedhere
\]
\end{proof}

\begin{theorem}[Afgeleiden van sinus en cosinus]\label{thm:g12:trigo:derivatives}
De \omterm{def:g11:func:function}{functies} $\sin$ en $\cos$ zijn \omterm{def:g12:deriv:derivative}{differentieerbaar} op $\R$, met
\[
(\sin)' = \cos, \qquad (\cos)' = -\sin .
\]
Algemener geldt $(\sin(ax+b))' = a\cos(ax+b)$ en
$(\cos(ax+b))' = -a\sin(ax+b)$.
\end{theorem}

\begin{proof}
Met de somformules is
\[
\frac{\sin(x+h) - \sin x}{h}
= \sin x\,\frac{\cos h - 1}{h} + \cos x\,\frac{\sin h}{h}
\xrightarrow[h\to0]{} \sin x \cdot 0 + \cos x \cdot 1 = \cos x,
\]
met \cref{lem:g12:trigo:sinxx}. De berekening voor $\cos$ verloopt identiek,
en de algemene vorm volgt uit de kettingregel.
\end{proof}

\begin{proposition}[Verloop]\label{prop:g12:trigo:variations}
Op $\intcc{0}{\pi}$ daalt $\cos$ van $1$ naar $-1$. Op
$\intcc{-\frac\pi2}{\frac\pi2}$ stijgt $\sin$ van $-1$ naar $1$. Beide
\omterm{def:g11:func:function}{functies} zijn $2\pi$-periodiek met \omterm{def:g10:functions:function}{beeld} $\intcc{-1}{1}$; $\cos$ is even,
$\sin$ is oneven.
\end{proposition}

\begin{proof}
Op $\intoo{0}{\pi}$ is $(\cos)' = -\sin < 0$, want het punt $M(t)$ heeft daar
een positieve \omterm{def:g10:coordgeom:system}{ordinaat}; net zo is $(\sin)' = \cos > 0$ op
$\intoo{-\frac\pi2}{\frac\pi2}$. De periodiciteit en de pariteit komen uit
\cref{prop:g12:trigo:identities}.
\end{proof}

\begin{method}[$\cos x = a$ en $\sin x = a$ oplossen]\label{met:g12:trigo:equations}
Zoek voor $a \in \intcc{-1}{1}$ één bijzondere oplossing $\alpha$ (uit de
tabel met waarden of met een rekenmachine). Dan zijn alle oplossingen:
\[
\cos x = \cos\alpha \iff x = \alpha + 2k\pi \text{ of } x = -\alpha + 2k\pi,
\quad k \in \Z;
\]
\[
\sin x = \sin\alpha \iff x = \alpha + 2k\pi \text{ of } x = \pi - \alpha + 2k\pi,
\quad k \in \Z.
\]
Voor ongelijkheden lokaliseer je de oplossingsbogen op de eenheidscirkel en
lees je de intervallen binnen één periode af.
\end{method}

\begin{example}\label{ex:g12:trigo:equation}
Los $\cos x = \frac12$ op in $\intoc{-\pi}{\pi}$: een bijzondere oplossing is
$\frac\pi3$, dus zijn de oplossingen $x = \frac\pi3$ en $x = -\frac\pi3$. Op
heel $\R$: $x = \pm\frac\pi3 + 2k\pi$, $k \in \Z$.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:trigo:1}
Bereken $\cos\frac{2\pi}{3}$, $\sin\frac{5\pi}{6}$, $\cos\frac{7\pi}{4}$ en
$\sin\left(-\frac{\pi}{3}\right)$ met de identiteiten van
\cref{prop:g12:trigo:identities}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:trigo:2}
Leid $f(x) = \sin^2 x$, $g(x) = \cos(3x) + x\sin x$ en
$h(x) = \dfrac{\sin x}{2 + \cos x}$ af.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:trigo:3}
Los op in $\intoc{-\pi}{\pi}$, en daarna in $\R$:
\[
\text{(a) } \sin x = \frac{\sqrt3}{2};
\qquad
\text{(b) } \cos\left(2x\right) = \frac{\sqrt2}{2} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:4}
Bereken met de somformules de exacte waarde van $\cos\frac{\pi}{12}$ en
$\sin\frac{\pi}{12}$. (Tip: $\frac{\pi}{12} = \frac{\pi}{3} -
\frac{\pi}{4}$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:5}
Los in $\intcc{0}{2\pi}$ de ongelijkheid $2\sin^2 x - \sin x - 1 \geq 0$ op.
(Tip: ontbind de kwadratische uitdrukking in $\sin x$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:6}
Bereken de limieten
\[
\lim_{x\to0} \frac{\sin 3x}{x}, \qquad
\lim_{x\to0} \frac{1 - \cos x}{x^2}, \qquad
\lim_{x\to+\infty} x \sin\frac{1}{x}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:7}
Bestudeer de \omterm{def:g11:func:function}{functie} $f(x) = x + \cos x$ op $\intcc{0}{2\pi}$: verloop en
extrema. Toon aan dat $f$ \omterm{def:g12:seq:monotonic}{stijgend} is op $\R$ hoewel $f'$ in oneindig veel
punten verdwijnt.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:trigo:8}
Zij $f(x) = A\cos x + B \sin x$ met $(A,B) \neq (0,0)$.
\begin{enumerate}
  \item Toon aan dat $f$ te schrijven is als $f(x) = R\cos(x - \varphi)$ met
        $R = \sqrt{A^2 + B^2}$ en een geschikte $\varphi$.
  \item Leid het \omterm{def:g10:functions:extrema}{maximum} en het \omterm{def:g10:functions:extrema}{minimum} van $f$ af, en los
        $\cos x + \sin x = 1$ op in $\intoc{-\pi}{\pi}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:trigo:9}
Toon aan dat $\sin$ en $\cos$ allebei aan de differentiaalvergelijking
$y'' = -y$ voldoen. Toon omgekeerd aan, voor een oplossing $y$ van
$y'' = -y$ met $y(0) = a$ en $y'(0) = b$, dat $y = a\cos + b\sin$. (Tip:
bekijk $g = (y - a\cos - b\sin)$ en de ``energie'' $E = g^2 + (g')^2$.)
\end{exercise}

\section{Opgave: het geheim van de slinger}

\begin{problem}[{Weekendopgave --- één meetkundige limiet drijft de
afgeleiden van sinus en cosinus aan, en van daaruit alle trillingen: veren,
slingers, zwevingen en de meter die het bijna werd}]
\label{pb:g12:trigo:1}
Elke klok die ooit tikte en elke snaar die ooit klonk, gehoorzaamt aan
dezelfde wiskunde: een terugdrijvende kracht evenredig met de uitwijking, dus
de \omterm{def:g10:algebra:equation}{vergelijking} $y'' = -\omega^2 y$, dus cosinussen. Aan de basis van dat
alles ligt één onschuldig ogende limiet, $\frac{\sin x}{x} \to 1$ --- in jaar
9 beloofd, door \cref{thm:g12:trigo:derivatives} gebruikt, en hier met een
tekening bewezen. Deze opgave klimt van die limiet naar de slag van een
slinger van één meter, en legt uit waarom de meter er op een haar na niet mee
gedefinieerd is.

\medskip
\noindent\textbf{Deel I --- De fundamentele limiet.}
\begin{enumerate}
  \item Warm op met getallen: bereken $\frac{\sin x}{x}$ voor $x = 0.1$ en
        $x = 0.01$ (in \omterm{def:g11:trigo:radian}{radialen}!).
  \item De insluiting, op de eenheidscirkel met $0 < x < \frac\pi2$: druk met
        elementaire meetkunde de oppervlakten uit van (i) de driehoek met
        hoekpunten $O$, $A(1,0)$ en het cirkelpunt $M(\cos x, \sin x)$;
        (ii) de cirkelsector $OAM$; (iii) de rechthoekige driehoek $OAT$ met
        $T(1, \tan x)$.
  \item Leid uit de insluiting van de oppervlakten af dat
        $\sin x < x < \tan x$, daarna dat $\cos x < \frac{\sin x}{x} < 1$, en
        besluit met de insluitstelling dat
        $\lim_{x \to 0} \frac{\sin x}{x} = 1$ (de pariteit regelt $x < 0$).
  \item Leid $\lim_{x \to 0} \frac{1 - \cos x}{x^2} = \frac12$ af
        (vermenigvuldig met de toegevoegde uitdrukking).
  \item Leg uit waarom die limiet de motor van
        \cref{thm:g12:trigo:derivatives} is (wat is $\sin'(0)$?), en waarom
        ze ook een oude schuld vereffent: in welke zin zijn \emph{\omterm{def:g11:trigo:radian}{radialen}}
        de enige hoekeenheid waarvoor $\sin' = \cos$ zonder constante geldt?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Harmonische beweging.}
\begin{enumerate}[resume]
  \item Ga met de kettingregel na dat $y(t) = R\cos(\omega t - \varphi)$
        voldoet aan $y'' = -\omega^2 y$.
  \item Een massa aan een veer voldoet aan $y'' = -4y$, met $y(0) = 3$ en
        $y'(0) = 8$. Neem (uit \cref{exo:g12:trigo:9}) aan dat alle
        oplossingen van de vorm $A\cos 2t + B\sin 2t$ zijn, en bepaal $A$ en
        $B$.
  \item Herschrijf de oplossing als $R\cos(2t - \varphi)$
        (\cref{exo:g12:trigo:8}): geef de amplitude $R$, de periode en de
        fase $\varphi$ (drie decimalen).
  \item Op welk tijdstip passeert de massa voor het eerst de evenwichtsstand
        $y = 0$?
  \item De slinger: een bol aan een koord met lengte $L$ voldoet
        \emph{exact} aan $\theta'' = -\frac gL \sin\theta$ --- niet oplosbaar
        met elementaire \omterm{def:g11:func:function}{functies}. Vervang voor kleine uitslagen $\sin\theta$
        door $\theta$ (deel I verantwoordt dat!) en leid de periode
        $T = 2\pi\sqrt{\frac Lg}$ af. Bereken $T$ voor $L = 1$ m
        ($g = 9.81$): de beroemde ontdekking van Galilei --- waarvan hangt de
        formule \emph{niet} af?
  \item De secondeslinger: welke lengte geeft precies $T = 2$ s? De Franse
        Academie aarzelde in 1791 tussen die lengte en het tienmiljoenste van
        de kwartmeridiaan om de \emph{meter} te definiëren --- hoeveel
        millimeter verschillen de twee kandidaten?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De somformules aan het werk.}
\begin{enumerate}[resume]
  \item Toon met de somformules (\cref{prop:g12:trigo:addition}) aan dat
        $\cos\left(x + \frac\pi2\right) = -\sin x$ en
        $\sin\left(x + \frac\pi2\right) = \cos x$: afleiden is op de cirkel
        een kwartdraai --- ga de overeenstemming met
        \cref{thm:g12:trigo:derivatives} na.
  \item Leid de identiteit
        $\cos a \cos b = \frac12\left(\cos(a - b) + \cos(a + b)\right)$ af,
        die producten in sommen omzet, en de linearisatie
        $\cos^2 x = \frac{1 + \cos 2x}{2}$ (het hoofdstuk over integreren zal
        je dankbaar zijn).
  \item Zwevingen: leid $\sin p + \sin q = 2 \sin\frac{p + q}{2}
        \cos\frac{p - q}{2}$ af, en pas het toe op twee stemvorken van $440$
        en $444$ Hz: welke toon hoort het oor, en hoeveel zwevingen per
        seconde? (Zo stemmen orkesten.)
  \item Leid de identiteit voor de drievoudige hoek
        $\cos 3x = 4\cos^3 x - 3\cos x$ af (schrijf $3x = 2x + x$). Aan welke
        derdegraadsvergelijking moet $\cos 20^\circ$ voldoen als je
        $x = 20^\circ$ stelt? (Dat die \omterm{def:g10:algebra:equation}{vergelijking} niet met vierkantswortels
        alleen op te lossen is, is precies de \omterm{def:g11:seq:geometric}{reden} waarom het driedelen van
        een hoek twee millennia lang aan passer en liniaal weerstond --- het
        volledige verhaal staat in de universitaire volumes.)
  \item Een wisselstroom luidt
        $I(t) = 3\cos(100\pi t) + 3\sqrt3 \sin(100\pi t)$. Herschrijf hem als
        $R\cos(100\pi t - \varphi)$: wat is de piekstroom en wat de fase?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De trillende wereld.}
\begin{enumerate}[resume]
  \item Perioden: wat is de (kleinste) periode van $\sin^2 t$ (gebruik vraag
        13)? En van $\sin(2t) + \sin(3t)$?
  \item Een echte slinger dempt uit: $y(t) = \eu^{-t/10}\cos(2\pi t)$.
        Beschrijf de beweging, bereken de waarde van de omhullende in
        $t = 10$ (er duikt een oude bekende op, \cref{pb:g12:exp:1}), en noem
        het hoofdstuk waarin \omterm{def:g10:algebra:equation}{vergelijkingen} met zulke oplossingen opgelost
        worden.
  \item Waarom beheersen sinussen en cosinussen elke trilling? Geef het
        woordenboek van natuurkunde naar wiskunde in twee zinnen
        (terugdrijvende kracht evenredig met de uitwijking $\rightarrow$
        welke \omterm{def:g10:algebra:equation}{vergelijking} $\rightarrow$ welke oplossingen), en zeg wat
        \cref{exo:g12:trigo:9} aan die bewering bijdraagt.
  \item Slotstuk: de boog van dit hoofdstuk in vier stappen --- één
        meetkundige limiet, twee \omterm{def:g12:deriv:derivative}{afgeleiden}, één differentiaalvergelijking,
        en alle trillingen. En het eerlijke sterretje: bij grote uitslagen
        bevat de echte periode van de slinger een elliptische integraal, die
        bliksemsnel berekend wordt door\,\dots\ het \omterm{def:g12:seq:arith-geom}{rekenkundig-meetkundig}
        \omterm{def:g11:stat:mean}{gemiddelde} van Gauss, het toetje van \cref{pb:g12:seq:1}. De draden
        van de reeks knopen zichzelf aan elkaar.
\end{enumerate}
\end{problem}
