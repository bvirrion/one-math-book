\chapter{Combinatoriek en tellen}\label{ch:g12:comb}

De combinatoriek is de kunst van het tellen zonder op te sommen. Haar twee
elementaire principes --- tel de omvang van disjuncte alternatieven op,
vermenigvuldig de aantallen onafhankelijke keuzen --- volstaan om de
\omterm{def:g12:comb:tuples}{variaties}, de \omterm{def:g12:comb:permutation}{permutaties} en de deelverzamelingen van een eindige verzameling
te tellen, en monden uit in het binomium van Newton.

\section{De twee telprincipes}

We schrijven $\abs{E}$ voor het aantal elementen (de
\emph{kardinaliteit}) van een eindige verzameling $E$.

\begin{proposition}[Somprincipe]\label{prop:g12:comb:addition}
Wordt een eindige verzameling $E$ verdeeld in deelverzamelingen
$A_1, \dots, A_k$ (paarsgewijs disjunct, met \omterm{def:g10:numbers:interunion}{vereniging} $E$), dan geldt
\[
\abs{E} = \abs{A_1} + \abs{A_2} + \dots + \abs{A_k}.
\]
\end{proposition}

\begin{proposition}[Productprincipe]\label{prop:g12:comb:multiplication}
Wordt een object opgebouwd met een opeenvolging van $k$ keuzen, met $n_1$
mogelijkheden voor de eerste keuze en, \emph{wat de vorige keuzen ook waren},
$n_i$ mogelijkheden voor de $i$-de, dan is het aantal gebouwde objecten
gelijk aan $n_1 \times n_2 \times \dots \times n_k$.
\end{proposition}

\begin{proof}
Beide uitspraken bewijs je met inductie op $k$; het geval $k = 2$ van de
tweede komt neer op een rechthoekige tabel \omterm{def:g12:seq:sequence}{rij} per rij tellen.
\end{proof}

\begin{example}
Een restaurant biedt 4 voorgerechten, 6 hoofdgerechten en 3 nagerechten:
$4 \times 6 \times 3 = 72$ verschillende driegangenmenu's.
\end{example}

\section{\texorpdfstring{$k$-tallen}{k-tallen}, permutaties, faculteiten}

\begin{definition}[$k$-tallen]\label{def:g12:comb:tuples}
Een \emph{$k$-tal} van een verzameling $E$ is een geordende lijst
$(x_1, \dots, x_k)$ van elementen van $E$, met herhalingen toegestaan. Een
$k$-tal van \emph{verschillende} elementen heet een
\emph{variatie}\index{variatie} van $k$ elementen van $E$.
\end{definition}

\begin{proposition}\label{prop:g12:comb:tuples}
Zij $\abs E = n$. Het aantal $k$-tallen van $E$ is $n^k$. Het aantal
\omterm{def:g12:comb:tuples}{variaties} van $k$ elementen van $E$ ($0 \leq k \leq n$) is
\[
n(n-1)(n-2)\cdots(n-k+1) = \frac{n!}{(n-k)!},
\]
waarbij $n! = 1 \times 2 \times \dots \times n$ (en $0! = 1$) de
\emph{faculteit}\index{faculteit} van $n$ is.
\end{proposition}

\begin{proof}
Productprincipe: voor een $k$-tal zijn er $n$ mogelijkheden bij elk van de
$k$ stappen; voor een \omterm{def:g12:comb:tuples}{variatie} zijn er $n$ mogelijkheden voor $x_1$, daarna
$n - 1$ voor $x_2$ (één element is opgebruikt), \dots, en $n - k + 1$ voor
$x_k$.
\end{proof}

\begin{definition}[Permutatie]\label{def:g12:comb:permutation}
Een \emph{permutatie}\index{permutatie} van $E$ is een \omterm{def:g12:comb:tuples}{variatie} van alle $n$
elementen van $E$: een rangschikking van $E$. Volgens
\cref{prop:g12:comb:tuples} (het geval $k = n$) is het aantal permutaties van
een verzameling met $n$ elementen gelijk aan $n!$.
\end{definition}

\begin{example}
Vijf lopers kunnen een wedstrijd in $5! = 120$ verschillende volgordes
beëindigen. Het aantal mogelijke podia (de eerste drie plaatsen) is
$5 \times 4 \times 3 = 60$.
\end{example}

\section{Combinaties en binomiaalcoëfficiënten}

\begin{definition}[Combinaties]\label{def:g12:comb:combination}
Een \emph{combinatie}\index{combinatie} van $k$ elementen van $E$ is een
deelverzameling van $E$ met $k$ elementen (zonder volgorde, zonder
herhaling). Hun aantal wordt $\dbinom{n}{k}$\index{binomiaalcoëfficiënt}
geschreven en gelezen als ``$n$ boven $k$''.
\end{definition}

\begin{theorem}\label{thm:g12:comb:binomformula}
Voor $0 \leq k \leq n$ geldt
\[
\binom{n}{k} = \frac{n!}{k!\,(n-k)!} .
\]
\end{theorem}

\begin{proof}
Tel de \omterm{def:g12:comb:tuples}{variaties} van $k$ elementen van $E$ op twee manieren. Rechtstreeks:
$\frac{n!}{(n-k)!}$. Anders: kies eerst de onderliggende deelverzameling
($\binom nk$ manieren) en rangschik ze daarna ($k!$ manieren); het
productprincipe geeft $\binom{n}{k}\,k!$. Gelijkstellen levert
$\binom nk = \frac{n!}{k!(n-k)!}$.
\end{proof}

\begin{proposition}[Basisidentiteiten]\label{prop:g12:comb:identities}
Voor $0 \leq k \leq n$ geldt
\[
\binom{n}{0} = \binom{n}{n} = 1, \qquad
\binom{n}{1} = n, \qquad
\binom{n}{k} = \binom{n}{n-k},
\]
en de \emph{regel van Pascal}\index{driehoek van Pascal}: voor
$1 \leq k \leq n-1$,
\[
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}.
\]
\end{proposition}

\begin{proof}
De symmetrie $\binom nk = \binom{n}{n-k}$ geldt omdat het \omterm{def:g10:proba:operations}{complement} nemen de
deelverzamelingen met $k$ elementen één op één koppelt aan die met $n-k$
elementen. Voor de \omterm{prop:g12:comb:identities}{regel van Pascal} kies je een element $a \in E$ vast en
sorteer je de deelverzamelingen met $k$ elementen in die welke $a$ bevatten
--- verkregen door $a$ toe te voegen aan een deelverzameling met $k-1$
elementen van $E \setminus \{a\}$, en daarvan zijn er $\binom{n-1}{k-1}$ ---
en die welke $a$ mijden, en dat zijn de deelverzamelingen met $k$ elementen
van $E \setminus \{a\}$, dus $\binom{n-1}{k}$ stuks. Besluit met het
somprincipe.
\end{proof}

De \omterm{prop:g12:comb:identities}{regel van Pascal} brengt de coëfficiënten rij per rij voort --- de
\emph{\omterm{prop:g11:binom:pascal}{driehoek van Pascal}}: elk getal is de som van de twee erboven.

\begin{theorem}[Binomium van Newton]\label{thm:g12:comb:binomial}
\index{binomium van Newton}
Voor alle $a, b \in \R$ (of $\C$) en $n \in \N$ geldt
\[
(a+b)^n = \sum_{k=0}^{n} \binom{n}{k}\, a^{k}\, b^{\,n-k} .
\]
\end{theorem}

\begin{proof}
Werk het product $(a+b)(a+b)\cdots(a+b)$ ($n$ factoren) uit: elke term van de
uitwerking kiest in elke factor $a$ of $b$, en levert $a^k b^{n-k}$ op,
waarbij $k$ het aantal factoren is dat $a$ bijdraagt. Het aantal manieren om
die $k$ factoren uit $n$ te kiezen is $\binom nk$, en dat is dus de
coëfficiënt van $a^k b^{n-k}$.
\end{proof}

\begin{corollary}\label{cor:g12:comb:sums}
$\displaystyle\sum_{k=0}^{n} \binom{n}{k} = 2^n$
\quad en \quad
$\displaystyle\sum_{k=0}^{n} (-1)^k\binom{n}{k} = 0$ \ ($n \geq 1$).
\end{corollary}

\begin{proof}
Neem $a = b = 1$, en daarna $a = -1$, $b = 1$ in het binomium van Newton. De
eerste identiteit heeft ook een rechtstreekse betekenis: een verzameling met
$n$ elementen heeft $2^n$ deelverzamelingen (elk element zit erin of niet:
productprincipe), gesorteerd naar omvang.
\end{proof}

\begin{method}[Het juiste model kiezen]\label{met:g12:comb:model}
Beantwoord vóór je telt twee vragen: \emph{doet de volgorde ertoe?} en
\emph{zijn herhalingen toegestaan?}
\begin{center}
\begin{tabular}{l|cc}
 & volgorde telt & volgorde telt niet\\
\hline
herhaling toegestaan & $n^k$ ($k$-tallen) & (universiteit)\\
geen herhaling & $\frac{n!}{(n-k)!}$ (\omterm{def:g12:comb:tuples}{variaties}) & $\binom nk$
(deelverzamelingen)
\end{tabular}
\end{center}
Ballen uit een urne trekken: \emph{met teruglegging, in volgorde} $\to$
$k$-tallen; \emph{zonder teruglegging, in volgorde} $\to$ \omterm{def:g12:comb:tuples}{variaties}; \emph{een
handvol in één greep} $\to$ deelverzamelingen.
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:comb:1}
Een nummerplaat bestaat uit 2 letters (A--Z), dan 3 cijfers, dan 2 letters.
Hoeveel platen zijn er mogelijk? Hoeveel ervan hebben geen enkel herhaald
teken?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:2}
Bereken $\dbinom{8}{3}$ en $\dbinom{10}{8}$, en vereenvoudig
$\dfrac{\binom{n}{2}}{\binom{n+1}{2}}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:3}
In een klas van 30 leerlingen moet een comité van 4 leerlingen verkozen
worden, en daarna binnen dat comité een voorzitter en een penningmeester
(één persoon kan niet beide ambten bekleden). Hoeveel uitkomsten zijn er
mogelijk?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:4}
Werk $(x + 2)^5$ en $(1 - x)^6$ uit met het binomium van Newton. Wat is de
coëfficiënt van $x^3$ in $(2x + 3)^7$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:5}
Een pokerhand bestaat uit 5 kaarten uit een spel van 52 kaarten.
\begin{enumerate}
  \item Hoeveel handen zijn er?
  \item Hoeveel handen bevatten precies één aas? En minstens één aas?
  \item Hoeveel handen zijn een ``full house'' (drie kaarten van één waarde
        en twee van een andere)?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:6}
Hoeveel anagrammen (herschikkingen van de letters, met of zonder betekenis)
heeft het woord GETAL? En het woord BANANA? (Tip voor BANANA: plaats eerst de
drie A's.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:7}
Bewijs de identiteit $k\dbinom{n}{k} = n\dbinom{n-1}{k-1}$
($1 \leq k \leq n$) op twee manieren: met de formule met \omterm{prop:g12:comb:tuples}{faculteiten}, en door
de paren (comité van $k$ personen, zijn voorzitter), gekozen uit $n$ mensen,
op twee manieren te tellen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:8}
Een pad in het vlak gaat van $(0,0)$ naar $(m, n)$ met eenheidsstappen naar
het oosten of naar het noorden. Toon aan dat het aantal zulke paden
$\dbinom{m+n}{m}$ is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:comb:9}
Bewijs de \emph{identiteit van Vandermonde}: voor $0 \leq k \leq m + n$ geldt
\[
\binom{m+n}{k} = \sum_{j=0}^{k} \binom{m}{j}\binom{n}{k-j},
\]
door de deelverzamelingen met $k$ elementen te tellen van een verzameling die
in een groep van $m$ en een groep van $n$ gesplitst is. Leid af dat
$\displaystyle\sum_{j=0}^{n}\binom{n}{j}^{\!2} = \binom{2n}{n}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:comb:10}
Toon met het binomium van Newton aan dat voor alle $n \geq 1$ geldt
\[
\sum_{k=1}^{n} k \binom{n}{k} = n\,2^{n-1}.
\]
(Tip: leid $(1+x)^n$ af, of gebruik \cref{exo:g12:comb:7}.)
\end{exercise}

\section{Opgave: de kunst van het dubbel tellen}

\begin{problem}[{Weekendopgave --- sterren en strepen, verwisselde hoeden, en
identiteiten bewezen door één ding op twee manieren te tellen}]
\label{pb:g12:comb:1}
De diepste truc van de combinatoriek is ontwapenend eenvoudig: tel dezelfde
verzameling twee keer, op twee verschillende manieren, en stel de antwoorden
gelijk. Deze opgave oefent de modellen van \cref{met:g12:comb:model}, voegt
er een techniek aan toe die de cursus van dit hoofdstuk niet nodig had --- de
\emph{sterren en strepen} van het ijsjes tellen --- telt daarna precies de
beroemde \emph{verwisselde hoeden}, en vindt onder in de hoedenstapel het
getal $\frac1\eu$, dat hier voor de derde keer in dit boek opduikt.

\medskip
\noindent\textbf{Deel I --- Het model kiezen.}
\begin{enumerate}
  \item Tel de nummerplaten van $2$ letters gevolgd door $3$ cijfers; en
        daarna de anagrammen van BANANA.
  \item Tel uit een spel van $32$ kaarten de handen van $5$ kaarten; en
        daarna de handen met precies $2$ van de $4$ azen.
  \item Een robot loopt van $(0,0)$ naar $(4,3)$ met alleen
        eenheidsstappen naar rechts of naar boven: hoeveel paden zijn er?
        (Codeer een pad als een woord in R en B.)
  \item Werk $(1 + x)^4$ uit met het binomium van Newton
        (\cref{thm:g12:comb:binomial}); vul daarna $x = 1$ en $x = -1$ in:
        welke twee identiteiten over de getallen $\binom nk$ vallen eruit?
  \item Bewijs met dubbel tellen dat $k\binom nk = n\binom{n-1}{k-1}$ (tel de
        comités met voorzitter op twee manieren), en leid
        $\sum_{k=0}^{n} k\binom nk = n\,2^{n-1}$ af.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Sterren en strepen.}
\begin{enumerate}[resume]
  \item Een ijssalon verkoopt $4$ smaken; je bestelt $10$ bollen (smaken
        mogen zich herhalen, en de volgorde in het hoorntje doet er niet toe).
        Codeer een bestelling als een \omterm{def:g12:seq:sequence}{rij} van $10$ sterren (de bollen)
        gescheiden door $3$ strepen (de overgangen tussen smaken), en tel de
        bestellingen.
  \item Tel de drietallen niet-negatieve \omterm{def:g10:numbers:sets}{gehele getallen} met
        $x + y + z = 12$.
  \item Tel de drietallen \emph{positieve} \omterm{def:g10:numbers:sets}{gehele getallen} met
        $x + y + z = 12$ (substitueer $x = 1 + x'$, enzovoort).
  \item Hoeveel verschillende monomen komen voor in de uitwerking van
        $(a + b + c)^5$?
  \item Toets de methode: tel met de formule de bestellingen van $3$ bollen
        uit $2$ smaken, som ze daarna allemaal op en vergelijk.
  \item Zeg precies waar ``de bollen zijn identiek'' in de codering
        binnenkwam --- en tel wat er in de plaats gebeurt wanneer de bollen in
        volgorde opgegeten worden (verschillende plaatsen), met de checklist
        van \cref{met:g12:comb:model}.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De verwisselde hoeden.}
Een \emph{derangement} is een herverdeling van $n$ hoeden onder hun $n$
eigenaars waarbij \emph{niemand} zijn eigen hoed krijgt; zij $D_n$ hun
aantal. (\Cref{pb:g11:prob:1} toonde dat gemiddeld één gast zijn eigen hoed
terugvindt --- nu tellen we de volledig ongelukkige feesten exact.)
\begin{enumerate}[resume]
  \item Bereken $D_1$, $D_2$ en $D_3$ door op te sommen, en $D_4$ geduldig
        (of slim).
  \item Verantwoord de recursie $D_n = (n - 1)\left(D_{n-1} + D_{n-2}\right)$:
        gast 1 krijgt een hoed $k \neq 1$ ($n - 1$ keuzen); splits daarna op
        naargelang gast $k$ hoed 1 krijgt of niet. Ga na dat ze $D_4$ oplevert
        en bereken $D_5$.
  \item Bewijs voor $n = 3$ met inclusie en exclusie (trek de toewijzingen af
        die minstens één hoed op zijn plaats laten, en tel de dubbeltellingen
        weer op) dat
        $D_3 = 3!\left(1 - \frac{1}{1!} + \frac{1}{2!} - \frac{1}{3!}\right)$,
        en formuleer de algemene formule.
  \item Bereken $\frac{D_5}{5!}$ en vergelijk met
        $\frac1\eu \approx 0.3679$: de \omterm{def:g10:proba:distribution}{kans} dat een groot geschud feest
        volledig verwisselt, is $\frac1\eu$ --- het derde optreden van die
        constante, na de loterij en de secretaresse van \cref{pb:g12:exp:1}.
        (Waarom: de formule van vraag 14 is het begin van een beroemde reeks
        voor $\eu^{-1}$, verteld in de universitaire volumes.)
  \item Secret Santa onder $10$ vrienden: de namen worden uniform willekeurig
        getrokken. Wat is de \omterm{def:g10:proba:distribution}{kans} dat de trekking geldig is (niemand trekt
        zichzelf), en hoeveel hertrekkingen mag de groep verwachten?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Twee keer tellen, twee keer winnen.}
\begin{enumerate}[resume]
  \item Het handdruklemma: op elk feest telt de som over de gasten van het
        aantal handen dat elk schudde, elke handdruk precies twee keer. Leid
        af dat \emph{het aantal gasten dat een oneven aantal handen schudde
        altijd even is} --- en ga na dat de bewering klopt op een feest met
        drie gasten.
  \item Bewijs het juweel $1^3 + 2^3 + \dots + n^3 = (1 + 2 + \dots + n)^2$
        met inductie, en ga het na voor $n = 3$. (De som van de kleine Gauss,
        gekwadrateerd, telt derde machten.)
  \item De identiteit van Vandermonde (\cref{exo:g12:comb:9}) via paden: duid
        $\binom{2n}{n}$ als het aantal roosterpaden uit vraag 3 van $(0,0)$
        naar $(n,n)$, knip elk pad door waar het de antidiagonaal kruist, en
        leg uit hoe $\sum_j \binom nj^2$ verschijnt.
  \item Slotstuk --- de vier zetten van de teller, telkens één regel met een
        voorbeeld uit deze opgave: vermenigvuldig fasen en tel gevallen op;
        codeer slim (sterren en strepen, padwoorden); tel hetzelfde twee keer
        (comité met voorzitter, handdrukken); trek het ongewenste af en
        corrigeer de dubbeltellingen (de derangementen). En noteer waar het
        tellen vervolgens aan de slag gaat: in de kansrekening, en bij de
        paden uit het hoofdstuk over matrices en grafen.
\end{enumerate}
\end{problem}
