\chapter{Afgeleiden en convexiteit}\label{ch:g12:deriv}

De \omterm{def:g12:deriv:derivative}{afgeleide} meet de ogenblikkelijke veranderingssnelheid van een \omterm{def:g11:func:function}{functie}; ze
is de \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} van de \omterm{def:g12:deriv:tangent}{raaklijn} aan haar \omterm{def:g10:functions:graph}{grafiek}. Dit hoofdstuk
herneemt en verruimt de rekenregels voor \omterm{def:g12:deriv:derivative}{afgeleiden}, verbindt het teken van
$f'$ met het verloop van $f$, en voert de \omterm{def:g12:deriv:convex}{convexiteit} in, die door de tweede
\omterm{def:g12:deriv:derivative}{afgeleide} bestuurd wordt.

\section{De afgeleide}

\begin{definition}[Afgeleide in een punt]\label{def:g12:deriv:derivative}
Zij $f$ gedefinieerd op een \omterm{def:g10:numbers:interval}{interval} $I$ en $a \in I$. De \omterm{def:g11:func:function}{functie} $f$ is
\emph{differentieerbaar in $a$}\index{afgeleide}\index{differentieerbaar}
wanneer het differentiequotiënt
\[
\frac{f(a+h) - f(a)}{h}
\]
een eindige limiet heeft voor $h \to 0$. Die limiet heet de \emph{afgeleide
van $f$ in $a$} en wordt $f'(a)$ geschreven. Is $f$ in elk punt van $I$
differentieerbaar, dan is de \omterm{def:g11:func:function}{functie} $f' \colon x \mapsto f'(x)$ de
\emph{afgeleide} van $f$.
\end{definition}

\begin{definition}[Raaklijn]\label{def:g12:deriv:tangent}
Is $f$ \omterm{def:g12:deriv:derivative}{differentieerbaar} in $a$, dan is de \emph{raaklijn}\index{raaklijn}
aan de kromme van $f$ in het punt $(a, f(a))$ de rechte met \omterm{def:g10:algebra:equation}{vergelijking}
\[
y = f(a) + f'(a)(x - a).
\]
\end{definition}

\begin{proposition}\label{prop:g12:deriv:diffcont}
Een \omterm{def:g11:func:function}{functie} die \omterm{def:g12:deriv:derivative}{differentieerbaar} is in $a$, is \omterm{def:g12:limcont:continuity}{continu} in $a$. Het
omgekeerde geldt niet.
\end{proposition}

\begin{proof}
Voor kleine $h \neq 0$ is $f(a+h) - f(a) = h \cdot \frac{f(a+h)-f(a)}{h}$.
Voor $h \to 0$ streeft het rechterlid naar $0 \times f'(a) = 0$, dus
$f(a+h) \to f(a)$. Voor het omgekeerde: $x \mapsto \abs{x}$ is \omterm{def:g12:limcont:continuity}{continu} in
$0$, maar haar differentiequotiënt in $0$ is $\frac{\abs h}{h} = \pm 1$
naargelang het teken van $h$, en heeft geen limiet.
\end{proof}

\subsection{Rekenregels voor afgeleiden}

\begin{proposition}[Bewerkingen]\label{prop:g12:deriv:operations}
Zijn $u, v$ \omterm{def:g12:deriv:derivative}{differentieerbaar} op $I$ en is $\lambda \in \R$, dan zijn
$u + v$, $\lambda u$ en $uv$ \omterm{def:g12:deriv:derivative}{differentieerbaar} op $I$, en $\frac{u}{v}$ waar
$v \neq 0$, met
\[
(u+v)' = u' + v', \quad (\lambda u)' = \lambda u', \quad
(uv)' = u'v + uv', \quad
\left(\frac{u}{v}\right)' = \frac{u'v - uv'}{v^2}.
\]
\end{proposition}

\begin{proof}[Bewijs van de productregel]
Schrijf het differentiequotiënt van $uv$ in $a$ als
\[
\frac{u(a+h)v(a+h) - u(a)v(a)}{h}
= \frac{u(a+h) - u(a)}{h}\,v(a+h) + u(a)\,\frac{v(a+h) - v(a)}{h}.
\]
Voor $h \to 0$ is $v(a+h) \to v(a)$ door de \omterm{def:g12:limcont:continuity}{continuïteit}
(\cref{prop:g12:deriv:diffcont}), zodat het rechterlid naar
$u'(a)v(a) + u(a)v'(a)$ streeft. De andere regels bewijs je op dezelfde
manier.
\end{proof}

\begin{theorem}[Kettingregel]\label{thm:g12:deriv:chain}
\index{kettingregel}
Zij $u$ \omterm{def:g12:deriv:derivative}{differentieerbaar} op $I$ met waarden in $J$, en $g$
\omterm{def:g12:deriv:derivative}{differentieerbaar} op $J$. Dan is $g \circ u$ \omterm{def:g12:deriv:derivative}{differentieerbaar} op $I$ en
geldt
\[
(g \circ u)' = (g' \circ u) \cdot u' .
\]
In het bijzonder geldt voor differentieerbare $u$:
\[
(u^n)' = n\,u'\,u^{n-1} \ (n \in \Z), \qquad
\left(\sqrt{u}\right)' = \frac{u'}{2\sqrt{u}} \ (u > 0), \qquad
\left(\eu^{u}\right)' = u'\,\eu^{u}.
\]
\end{theorem}

\begin{proof}[Schets van het bewijs]
Wanneer $u(a+h) \neq u(a)$ voor kleine $h$, schrijf je
\[
\frac{g(u(a+h)) - g(u(a))}{h}
= \frac{g(u(a+h)) - g(u(a))}{u(a+h) - u(a)} \cdot \frac{u(a+h) - u(a)}{h}.
\]
Voor $h \to 0$ is $u(a+h) \to u(a)$, zodat de eerste factor naar $g'(u(a))$
streeft en de tweede naar $u'(a)$. (Een volledig bewijs moet ook het geval
behandelen waarin $u(a+h) = u(a)$ voor willekeurig kleine $h$; dat technische
punt komt aan de universiteit aan bod.)
\end{proof}

\begin{example}
De gebruikelijke \omterm{def:g12:deriv:derivative}{afgeleiden}, geldig op de natuurlijke \omterm{def:g11:func:function}{domeinen}:
\[
(x^n)' = nx^{n-1}, \quad
\left(\frac{1}{x}\right)' = -\frac{1}{x^2}, \quad
(\sqrt{x})' = \frac{1}{2\sqrt{x}}, \quad
(\cos x)' = -\sin x, \quad
(\sin x)' = \cos x .
\]
De eerste bewijs je met inductie uit de productregel, en de laatste twee in
\cref{ch:g12:trigo}.
\end{example}

\section{Verloop en extrema}

\begin{theorem}[Teken van de afgeleide en verloop]\label{thm:g12:deriv:variations}
Zij $f$ \omterm{def:g12:deriv:derivative}{differentieerbaar} op een \omterm{def:g10:numbers:interval}{interval} $I$.
\begin{enumerate}
  \item $f$ is \omterm{def:g12:seq:monotonic}{stijgend} op $I$ dan en slechts dan als $f' \geq 0$ op $I$.
  \item $f$ is \omterm{def:g10:functions:variations}{dalend} op $I$ dan en slechts dan als $f' \leq 0$ op $I$.
  \item Is $f' > 0$ op $I$ behalve in eindig veel punten waar ze verdwijnt,
        dan is $f$ strikt \omterm{def:g12:seq:monotonic}{stijgend} op $I$.
\end{enumerate}
\end{theorem}

\begin{proof}[Gedeeltelijk bewijs]
Is $f$ \omterm{def:g12:seq:monotonic}{stijgend}, dan is elk differentiequotiënt
$\frac{f(a+h)-f(a)}{h}$ groter dan of gelijk aan $0$, en overgaan naar de
limiet geeft $f'(a) \geq 0$. De omgekeerde implicaties steunen op de
middelwaardestelling, die aan de universiteit bewezen wordt; ze worden
\emph{op dit niveau aangenomen}.
\end{proof}

\begin{definition}[Lokaal extremum]\label{def:g12:deriv:extremum}
$f$ heeft in $a$ een \emph{lokaal maximum}\index{extremum} wanneer
$f(x) \leq f(a)$ voor alle $x$ in de buurt van $a$; lokale minima worden
symmetrisch gedefinieerd.
\end{definition}

\begin{proposition}[Eerste-ordevoorwaarde]\label{prop:g12:deriv:critical}
Is $f$ \omterm{def:g12:deriv:derivative}{differentieerbaar} op een \emph{open} \omterm{def:g10:numbers:interval}{interval} $I$ en heeft ze in
$a \in I$ een lokaal \omterm{def:g12:deriv:extremum}{extremum}, dan is $f'(a) = 0$. Het omgekeerde geldt niet
($f(x) = x^3$ in $a = 0$).
\end{proposition}

\begin{proof}
Zeg dat $a$ een \omterm{def:g12:deriv:extremum}{lokaal maximum} is. Voor kleine $h > 0$ is
$\frac{f(a+h)-f(a)}{h} \leq 0$, dus geeft $h \to 0^+$ dat $f'(a) \leq 0$;
voor $h < 0$ is het quotiënt $\geq 0$, wat $f'(a) \geq 0$ geeft. Bijgevolg is
$f'(a) = 0$.
\end{proof}

\begin{method}[Verlooptabel]\label{met:g12:deriv:table}
Om een \omterm{def:g11:func:function}{functie} $f$ te bestuderen: bepaal haar \omterm{def:g11:func:function}{domein} en de limieten aan de
rand; bereken $f'$ en ontbind ze; bepaal het teken van $f'$ op elk
deelinterval; noteer alles in een \omterm{def:g10:functions:table}{verlooptabel}, met de extrema aangeduid;
leid het aantal oplossingen van $f(x) = k$ af met de bijectiestelling
(\cref{thm:g12:limcont:bijection}).
\end{method}

\section{Convexiteit}

\begin{definition}[Convexe functie]\label{def:g12:deriv:convex}
Een \omterm{def:g11:func:function}{functie} $f$ gedefinieerd op een \omterm{def:g10:numbers:interval}{interval} $I$ heet
\emph{convex}\index{convexiteit} op $I$ wanneer elke koorde van haar \omterm{def:g10:functions:graph}{grafiek}
boven de \omterm{def:g10:functions:graph}{grafiek} ligt: voor alle $a, b \in I$ en $t \in \intcc{0}{1}$ geldt
\[
f\bigl((1-t)a + t b\bigr) \leq (1-t) f(a) + t f(b).
\]
Ze heet \emph{concaaf}\index{concaaf} wanneer de omgekeerde ongelijkheid
geldt ($-f$ convex).
\end{definition}

\begin{theorem}[Convexiteit en afgeleiden]\label{thm:g12:deriv:convexity}
Zij $f$ \omterm{def:g12:deriv:derivative}{differentieerbaar} op een \omterm{def:g10:numbers:interval}{interval} $I$. De volgende uitspraken zijn
gelijkwaardig:
\begin{enumerate}
  \item $f$ is \omterm{def:g12:deriv:convex}{convex} op $I$;
  \item $f'$ is \omterm{def:g12:seq:monotonic}{stijgend} op $I$;
  \item de \omterm{def:g10:functions:graph}{grafiek} van $f$ ligt boven elk van haar \omterm{def:g12:deriv:tangent}{raaklijnen}.
\end{enumerate}
Is $f$ twee keer \omterm{def:g12:deriv:derivative}{differentieerbaar}, dan zijn die uitspraken ook gelijkwaardig
met $f'' \geq 0$ op $I$.
\end{theorem}

\begin{proof}[Bewijs van (2) $\Rightarrow$ (3)]
Kies $a \in I$ en stel $g(x) = f(x) - f(a) - f'(a)(x-a)$, de kloof tussen de
kromme en de \omterm{def:g12:deriv:tangent}{raaklijn} in $a$. Dan is $g$ \omterm{def:g12:deriv:derivative}{differentieerbaar} met
$g'(x) = f'(x) - f'(a)$. Is $f'$ \omterm{def:g12:seq:monotonic}{stijgend}, dan is $g' \leq 0$ op
$I \cap \intoc{-\infty}{a}$ en $g' \geq 0$ op $I \cap \intco{a}{+\infty}$:
$g$ daalt vóór $a$ en stijgt erna, zodat $g$ haar \omterm{def:g10:functions:extrema}{minimum} $g(a) = 0$
bereikt, en dus $g \geq 0$. De overige implicaties worden op dit niveau
aangenomen; de gelijkwaardigheid met $f'' \geq 0$ volgt uit
\cref{thm:g12:deriv:variations} toegepast op $f'$.
\end{proof}

\begin{definition}[Buigpunt]\label{def:g12:deriv:inflection}
Een punt waar de kromme van $f$ haar \omterm{def:g12:deriv:tangent}{raaklijn} kruist heet een
\emph{buigpunt}\index{buigpunt}. Is $f$ twee keer \omterm{def:g12:deriv:derivative}{differentieerbaar}, dan zijn
de buigpunten de punten waar $f''$ van teken verandert.
\end{definition}

\begin{example}\label{ex:g12:deriv:cube}
Voor $f(x) = x^3$ is $f''(x) = 6x$: $f$ is \omterm{def:g12:deriv:convex}{concaaf} op $\intoc{-\infty}{0}$ en
\omterm{def:g12:deriv:convex}{convex} op $\intco{0}{+\infty}$, met een \omterm{def:g12:deriv:inflection}{buigpunt} in de oorsprong, waar de
kromme haar \omterm{def:g12:deriv:tangent}{raaklijn} (de $x$-as) kruist.
\end{example}

\begin{example}[Ongelijkheden uit convexiteit]\label{ex:g12:deriv:ineq}
De exponentiële \omterm{def:g11:func:function}{functie} is \omterm{def:g12:deriv:convex}{convex} op $\R$ (haar tweede \omterm{def:g12:deriv:derivative}{afgeleide} is ze zelf,
en die is positief). Haar \omterm{def:g12:deriv:tangent}{raaklijn} in $0$ is $y = 1 + x$, dus
\[
\eu^x \geq 1 + x \quad \text{voor alle } x \in \R .
\]
Zulke \emph{raaklijnongelijkheden} zijn een standaardproduct van de
\omterm{def:g12:deriv:convex}{convexiteit}.
\end{example}

\begin{method}[Convexiteit gebruiken]\label{met:g12:deriv:convexity}
\begin{itemize}
  \item Om een ongelijkheid $f(x) \geq ax + b$ te bewijzen: herken de rechte
        als een \omterm{def:g12:deriv:tangent}{raaklijn} van een convexe $f$ en roep
        \cref{thm:g12:deriv:convexity}(3) in.
  \item Om \omterm{def:g12:deriv:inflection}{buigpunten} te lokaliseren: los $f''(x) = 0$ op \emph{en} ga na dat
        $f''$ daar van teken verandert.
  \item In een \omterm{def:g10:functions:table}{verlooptabel} verfijnt de \omterm{def:g12:deriv:convex}{convexiteit} de schets van de kromme
        (welke kant ze op buigt).
\end{itemize}
\end{method}

$\R$ is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:deriv:9}
Toon aan dat onder alle rechthoeken met vaste omtrek $p$ het vierkant de
grootste oppervlakte heeft. Toon daarna aan dat onder alle rechthoeken met
vaste oppervlakte $A$ het vierkant de kleinste omtrek heeft, en leg uit hoe
de twee uitspraken samenhangen.
\end{exercise}

\section{Opgave: de berekening van de redder}

\begin{problem}[{Weekendopgave --- de snelste reddingsweg gehoorzaamt aan de
wet van Snellius, de convexiteit bekrachtigt elk optimum, en het ideale
blikje is precies even hoog als breed}]
\label{pb:g12:deriv:1}
Een strandredder ziet een zwemmer in nood --- schuin verderop het strand
af, in het water. Lopen gaat sneller dan zwemmen: de rechte lijn is
\emph{niet} de snelste weg. De weg die de tijd minimaal maakt buigt aan de
waterlijn, en haar buigingswet is precies de wet waarmee licht breekt bij het
binnendringen van water: ook de natuur leidt af. Deze opgave oefent de
kettingregel (\cref{thm:g12:deriv:chain}), voert de redding uit, oogst de
ongelijkheden van de \omterm{def:g12:deriv:convex}{convexiteit} (\cref{thm:g12:deriv:convexity}), en ontwerpt
een blikje.

\medskip
\noindent\textbf{Deel I --- Vlot met de kettingregel.}
\begin{enumerate}
  \item Leid af: $\left(3x^2 + 1\right)^5$; \ $\sqrt{x^2 + 9}$; \
        $\dfrac{1}{x^2 + 1}$.
  \item Geef de \omterm{def:g12:deriv:tangent}{raaklijn} aan $y = \sqrt{x^2 + 9}$ in $x = 4$.
  \item Bouw de \omterm{def:g10:functions:table}{verlooptabel} van $f(x) = \dfrac{x}{x^2 + 1}$ op $\R$
        (extrema inbegrepen).
  \item Voor $g(x) = x^3 - 3x^2 + 4$: \omterm{def:g10:functions:table}{verlooptabel} \emph{en} convexiteitstabel
        ($g''$), met het \omterm{def:g12:deriv:inflection}{buigpunt} (\cref{def:g12:deriv:inflection}).
  \item Bereken de \omterm{def:g12:deriv:tangent}{raaklijn} aan $g$ in haar \omterm{def:g12:deriv:inflection}{buigpunt}, en toon aan dat
        $g(x) - (\text{raaklijn}) = (x - 1)^3$: wat zegt de tekenwissel over
        de manier waarop de \omterm{def:g12:deriv:tangent}{raaklijn} de kromme in een \omterm{def:g12:deriv:inflection}{buigpunt} ontmoet?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De redding.}
De waterlijn loopt langs de $x$-as; de redder staat op het zand in
$A(0, 30)$, de zwemmer wacht in $B(40, -20)$ (in meter). De redder loopt met
$5$ m/s op het zand, zwemt met $2$ m/s, en gaat het water in op een zelf
gekozen punt $(x, 0)$.
\begin{enumerate}[resume]
  \item Model: druk de totale reddingstijd $T(x)$ uit, en zeg waarom alleen
        $x \in \intcc{0}{40}$ de moeite van het bekijken waard is.
  \item Leid $T$ af (de kettingregel aan het werk --- twee keer).
  \item Zij $\theta_1$ de hoek van het loopstuk met de \emph{loodlijn} op de
        waterlijn, en $\theta_2$ die van het zwemstuk. Toon aan dat
        $\sin\theta_1 = \frac{x}{\sqrt{x^2 + 900}}$ en dat de voorwaarde
        $T'(x) = 0$ luidt
        \[
        \frac{\sin\theta_1}{5} = \frac{\sin\theta_2}{2}
        \]
        --- de \emph{wet van Snellius}, met de twee snelheden in de plaats van
        de lichtsnelheden in lucht en in water.
  \item Het rechte lijnstuk $[AB]$ kruist de waterlijn in $x = 24$. Bereken
        $T(24)$, $T(40)$ (langs het strand lopen en dan recht naar buiten
        zwemmen) en $T(0)$: is de meetkundige rechte lijn de snelste? En is
        een van de uiterste strategieën dat?
  \item Lokaliseer het optimale instappunt met \omterm{thm:g12:limcont:ivt}{bisectie} op $T'$ (het ligt
        rond $33.7$ m): geef $x^*$ op de meter en de recordtijd $T(x^*)$ op
        een tiende seconde. Hoeveel wint de optimale knik ten opzichte van de
        rechte lijn?
  \item Waarom is het kritieke punt een \emph{globaal} \omterm{def:g10:functions:extrema}{minimum}? (Elke term van
        $T$ is \omterm{def:g12:deriv:convex}{convex} --- neem aan dat een som van convexe \omterm{def:g11:func:function}{functies} \omterm{def:g12:deriv:convex}{convex} is
        --- en pas \cref{exo:g12:deriv:8} toe.)
  \item Het principe van Fermat zegt dat licht altijd de weg van de kortste
        \emph{tijd} volgt. Leid af waarom een lichtstraal precies aan het
        oppervlak buigt bij het binnendringen van water (waar licht trager
        is), en noem de alledaagse waarneming die dit verklaart over een
        potlood in een glas water.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De oogst van de \omterm{def:g12:deriv:convex}{convexiteit}.}
\begin{enumerate}[resume]
  \item Toon aan dat $x \mapsto x^2$ \omterm{def:g12:deriv:convex}{convex} is, en bewijs de
        middenongelijkheid
        $\left(\frac{a + b}{2}\right)^2 \leq \frac{a^2 + b^2}{2}$ twee keer:
        één keer uit de \omterm{def:g12:deriv:convex}{convexiteit}, één keer door $(a - b)^2 \geq 0$ uit te
        werken.
  \item Leid af dat het \emph{kwadratisch \omterm{def:g11:stat:mean}{gemiddelde}}
        $\sqrt{\frac{a^2 + b^2}{2}}$ het \omterm{def:g12:seq:arith-geom}{rekenkundig} \omterm{def:g11:stat:mean}{gemiddelde} overtreft, en
        stel de volledige ketting van deze reeks samen --- harmonisch $\leq$
        \omterm{def:g12:seq:arith-geom}{meetkundig} $\leq$ \omterm{def:g12:seq:arith-geom}{rekenkundig} $\leq$ kwadratisch --- en ga alle vier
        na op $a = 2$, $b = 8$.
  \item \omterm{def:g12:deriv:tangent}{Raaklijn} onder de kromme: $x \mapsto \frac1x$ is \omterm{def:g12:deriv:convex}{convex} op
        $\intoo{0}{+\infty}$; schrijf haar \omterm{def:g12:deriv:tangent}{raaklijn} in $1$ op en leid de
        ongelijkheid $\frac1x \geq 2 - x$ voor alle $x > 0$ af. Waar is het
        een gelijkheid?
  \item Een \omterm{def:g12:deriv:inflection}{buigpunt} in het wild: tijdens een epidemie volgt het cumulatieve
        aantal gevallen een S-vormige kromme. Wat gebeurt er in haar \omterm{def:g12:deriv:inflection}{buigpunt},
        en waarom is dat de datum waar epidemiologen naar uitkijken? Breng het
        in verband met het teken van de tweede \omterm{def:g12:deriv:derivative}{afgeleide} aan elke kant.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Het ideale blikje.}
Een cilindrisch blikje moet $V = 330$ cm$^3$ bevatten met zo weinig mogelijk
metaal: oppervlakte $S = 2\pi r^2 + 2\pi r h$, met de voorwaarde
$\pi r^2 h = V$.
\begin{enumerate}[resume]
  \item Druk $S(r) = 2\pi r^2 + \dfrac{2V}{r}$ uit, leid af, en bereken de
        optimale straal en hoogte voor $V = 330$ (op de millimeter
        nauwkeurig).
  \item Bewijs de sierlijke algemene wet: in het optimum is $h = 2r$ --- het
        ideale blikje is precies even hoog als breed.
  \item Ga $S''(r) > 0$ na en besluit (opnieuw via \cref{exo:g12:deriv:8}) dat
        het optimum globaal is. Echte frisdrankblikjes zijn zichtbaar hoger
        dan breed: noem een niet-wiskundige \omterm{def:g11:seq:geometric}{reden}.
  \item Slotstuk --- de pijplijn van het optimaliseren: modelleren, de
        \omterm{def:g12:deriv:derivative}{afgeleide} raadplegen, de kritieke punten oplossen, bekrachtigen met
        \omterm{def:g12:deriv:convex}{convexiteit} of een tabel, en duiden. Loop de lijst af voor de redder,
        het blikje en de sterkste balk uit jaar 11
        (\cref{pb:g11:deriv:1}) --- en zeg wat het principe van Fermat vertelt
        over wie er nog meer met deze pijplijn werkt.
\end{enumerate}
\end{problem}
