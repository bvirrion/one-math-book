\chapter{Differentiaalvergelijkingen}\label{ch:g12:diffeq}

Een \omterm{def:g12:diffeq:ode}{differentiaalvergelijking} verbindt een \omterm{def:g11:func:function}{functie} met haar \omterm{def:g12:deriv:derivative}{afgeleiden}. De
natuurkunde, de scheikunde, de biologie en de economie drukken hun wetten in
die vorm uit: de afkoelsnelheid van een lichaam, het verval van een
radioactieve kern en de groei van een populatie zijn alle uitspraken over
$y'$. Dit hoofdstuk lost de lineaire \omterm{def:g10:algebra:equation}{vergelijkingen} van de eerste orde met
constante coëfficiënten volledig op.

\section{De vergelijking \texorpdfstring{$y' = ay$}{y' = ay}}

\begin{definition}[Differentiaalvergelijking]\label{def:g12:diffeq:ode}
Een \emph{differentiaalvergelijking}\index{differentiaalvergelijking} is een
\omterm{def:g10:algebra:equation}{vergelijking} met een \omterm{def:g11:func:function}{functie} $y$ als onbekende, waarin $y$ en haar \omterm{def:g12:deriv:derivative}{afgeleiden}
voorkomen. Een \emph{oplossing} op een \omterm{def:g10:numbers:interval}{interval} $I$ is een differentieerbare
\omterm{def:g11:func:function}{functie} die in elk punt van $I$ aan de \omterm{def:g10:algebra:equation}{vergelijking} voldoet.
\end{definition}

\begin{theorem}[Oplossingen van $y' = ay$]\label{thm:g12:diffeq:homogeneous}
Zij $a \in \R$. De oplossingen op $\R$ van de \omterm{def:g10:algebra:equation}{vergelijking} $y' = ay$ zijn
precies de \omterm{def:g11:func:function}{functies}
\[
y(x) = C\,\eu^{ax}, \qquad C \in \R .
\]
Voor elk paar $(x_0, y_0)$ is er precies één oplossing met $y(x_0) = y_0$.
\end{theorem}

\begin{proof}
Elke $y = C\eu^{ax}$ voldoet aan $y' = Ca\,\eu^{ax} = ay$. Omgekeerd, zij $y$
een willekeurige oplossing en stel $z(x) = y(x)\,\eu^{-ax}$. Dan is
\[
z'(x) = y'(x)\,\eu^{-ax} - a\,y(x)\,\eu^{-ax}
= \bigl(y'(x) - a y(x)\bigr)\eu^{-ax} = 0,
\]
dus is $z$ constant, zeg $C$, en $y(x) = C\eu^{ax}$. De beginvoorwaarde
$y(x_0) = y_0$ legt $C = y_0 \eu^{-a x_0}$ ondubbelzinnig vast.
\end{proof}

\begin{remark}
De \omterm{thm:g12:exp:existence}{exponentiële functie} wordt dus \emph{gekenmerkt} door de eenvoudigste van
alle \omterm{def:g12:diffeq:ode}{differentiaalvergelijkingen}: groei evenredig met de omvang. Daarom duikt
ze overal in de natuur op.
\end{remark}

\begin{example}[Radioactief verval]\label{ex:g12:diffeq:decay}
Een radioactieve hoeveelheid voldoet aan $N' = -\lambda N$ met
$\lambda > 0$, dus $N(t) = N_0\,\eu^{-\lambda t}$; zie
\cref{ex:g12:exp:halflife} voor de halveringstijd.
\end{example}

\section{De vergelijking \texorpdfstring{$y' = ay + b$}{y' = ay + b}}

\begin{theorem}[Oplossingen van $y' = ay + b$]\label{thm:g12:diffeq:affine}
Zij $a \neq 0$ en $b \in \R$. De oplossingen op $\R$ van $y' = ay + b$ zijn
precies de \omterm{def:g11:func:function}{functies}
\[
y(x) = C\,\eu^{ax} - \frac{b}{a}, \qquad C \in \R .
\]
De constante \omterm{def:g11:func:function}{functie} $y_p = -\frac ba$ is de \emph{evenwichtsoplossing}. Voor
elke $(x_0, y_0)$ is er precies één oplossing met $y(x_0) = y_0$; en is
$a < 0$, dan streeft elke oplossing voor $x \to +\infty$ naar het evenwicht
$-\frac ba$.
\end{theorem}

\begin{proof}
De constante $y_p = -\frac{b}{a}$ voldoet aan $y_p' = 0 = a y_p + b$. Nu is
$y$ een oplossing dan en slechts dan als
\[
(y - y_p)' = y' = ay + b = a(y - y_p) + \underbrace{a y_p + b}_{=\,0}
= a (y - y_p),
\]
dus dan en slechts dan als $z = y - y_p$ voldoet aan $z' = az$. Volgens
\cref{thm:g12:diffeq:homogeneous} is $z = C\eu^{ax}$, waaruit de formule, het
bestaan en de uniciteit volgen. Is $a < 0$, dan is $\eu^{ax} \to 0$ voor
$x \to +\infty$, dus $y(x) \to -\frac ba$.
\end{proof}

\begin{method}[$y' = ay + b$ met een beginvoorwaarde oplossen]\label{met:g12:diffeq:solve}
\begin{enumerate}
  \item Zoek de evenwichtsoplossing $y_p = -\frac{b}{a}$ (los $y' = 0$ op).
  \item Schrijf de algemene oplossing $y = C\eu^{ax} + y_p$ op.
  \item Bepaal $C$ uit de beginvoorwaarde.
  \item Toets het gedrag op lange termijn aan de fysische intuïtie
        (\omterm{def:g12:seq:limit}{convergeert} de oplossing naar het evenwicht?).
\end{enumerate}
Dezelfde strategie --- \emph{een particuliere oplossing plus de algemene
oplossing van de homogene \omterm{def:g10:algebra:equation}{vergelijking}} --- gaat door voor $y' = ay + f(x)$:
zie \cref{prop:g12:diffeq:general}.
\end{method}

\begin{example}[Afkoelingswet van Newton]\label{ex:g12:diffeq:cooling}
Een kop koffie van $80\,^\circ$C staat in een kamer van $20\,^\circ$C. De wet
van Newton zegt dat de temperatuur $T$ voldoet aan $T' = -k(T - 20)$ voor een
zekere $k > 0$, dus $T' = -kT + 20k$. Het evenwicht is $20$, en
$T(t) = 20 + 60\,\eu^{-kt}$: de koffie koelt exponentieel snel af tot
kamertemperatuur.
\end{example}

\section{De vergelijking \texorpdfstring{$y' = ay + f(x)$}{y' = ay + f(x)}}

\begin{proposition}[Structuur van de oplossingsverzameling]\label{prop:g12:diffeq:general}
Zij $f$ \omterm{def:g12:limcont:continuity}{continu} op een \omterm{def:g10:numbers:interval}{interval} $I$ en zij $y_p$ één particuliere oplossing
van
\[
y' = ay + f(x)
\]
op $I$. Dan zijn de oplossingen op $I$ precies de \omterm{def:g11:func:function}{functies}
$y = C\eu^{ax} + y_p$ met $C \in \R$.
\end{proposition}

\begin{proof}
Zoals in \cref{thm:g12:diffeq:affine}: $y$ is een oplossing dan en slechts
dan als $z = y - y_p$ voldoet aan $z' = (ay + f) - (ay_p + f) = az$, dus dan
en slechts dan als $z = C\eu^{ax}$.
\end{proof}

\begin{method}[Een particuliere oplossing raden]\label{met:g12:diffeq:particular}
Zoek een particuliere oplossing \emph{van dezelfde gedaante} als $f$:
\begin{itemize}
  \item is $f$ een veelterm van graad $n$: probeer een veelterm van graad
        $n$;
  \item is $f(x) = \alpha\,\eu^{kx}$ met $k \neq a$: probeer
        $y_p = \beta\,\eu^{kx}$;
  \item is $f(x) = \alpha\,\eu^{ax}$ (het resonante geval): probeer
        $y_p = \beta x\,\eu^{ax}$.
\end{itemize}
Vul in de \omterm{def:g10:algebra:equation}{vergelijking} in en stel de coëfficiënten gelijk.
\end{method}

\begin{example}\label{ex:g12:diffeq:particular}
Los $y' = 2y + 4x$ op. Probeer $y_p = \alpha x + \beta$: uit
$\alpha = 2(\alpha x + \beta) + 4x$ voor alle $x$ volgt $2\alpha + 4 = 0$ en
$\alpha = 2\beta$, dus $\alpha = -2$ en $\beta = -1$. Algemene oplossing:
$y = C\eu^{2x} - 2x - 1$.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:diffeq:1}
Los op op $\R$:
(a) $y' = 3y$ met $y(0) = 2$;\quad
(b) $2y' + y = 0$ met $y(0) = -1$;\quad
(c) $y' = -y + 5$ met $y(0) = 0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:diffeq:2}
Een bacteriepopulatie groeit met een snelheid evenredig met haar omvang en
verdubbelt elke $3$ uur. Schrijf de \omterm{def:g12:diffeq:ode}{differentiaalvergelijking} op waaraan de
populatie $N(t)$ voldoet, en bepaal de evenredigheidsconstante.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:diffeq:3}
Ga na dat $y_p(x) = x\,\eu^{x}$ een oplossing is van $y' = y + \eu^x$, en
geef alle oplossingen op $\R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:4}
Los $y' = -2y + \eu^{x}$ op met $y(0) = 1$. (Tip: zoek een particuliere
oplossing van de vorm $\beta\,\eu^{x}$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:5}
Koolstof-14 vervalt met een halveringstijd van $5730$ jaar. Een
archeologisch monster bevat $60\%$ van de koolstof-14 van een levend
organisme. Schat zijn ouderdom.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:6}
Een tank bevat $100$ L zuiver water. Pekel met $0.2$ kg zout per liter
stroomt binnen met $5$ L/min, en het (voortdurend gemengde) mengsel stroomt
met dezelfde snelheid weg. Zij $m(t)$ de massa zout in de tank op tijdstip
$t$ (in minuten).
\begin{enumerate}
  \item Verantwoord dat $m' = 1 - \dfrac{m}{20}$.
  \item Los op, en bepaal de limiet van $m(t)$ voor $t \to +\infty$. Duid het
        resultaat.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:7}
Een parachutespringster van $80$ kg valt onder invloed van de zwaartekracht
($g = 9.8\ \text{m/s}^2$) en van luchtweerstand evenredig met de snelheid,
zodat haar snelheid voldoet aan $v' = g - \frac{k}{m}v$ met $k = 16$ kg/s.
\begin{enumerate}
  \item Los de \omterm{def:g10:algebra:equation}{vergelijking} op met $v(0) = 0$.
  \item Bereken de \emph{eindsnelheid} $\lim_{t\to+\infty} v(t)$, en de tijd
        die nodig is om $95\%$ ervan te bereiken.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:diffeq:8}
\emph{(Logistische \omterm{def:g10:algebra:equation}{vergelijking}.)} Een populatie $y(t) \in \intoo{0}{1}$
(uitgedrukt als fractie van de maximale populatie) voldoet aan
\[
y' = y(1 - y).
\]
\begin{enumerate}
  \item Zij $z = \dfrac{1}{y}$. Toon aan dat $z$ voldoet aan de lineaire
        \omterm{def:g10:algebra:equation}{vergelijking} $z' = -z + 1$.
  \item Los op naar $z$, en daarna naar $y$, met $y(0) = \frac{1}{10}$.
  \item Toon aan dat $y(t) \to 1$ voor $t \to +\infty$ en schets de vorm van
        de oplossingskromme.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:diffeq:9}
Zij $y$ een oplossing van $y' = ay + b$ en zij $u_n = y(n)$ voor $n \in \N$.
Toon aan dat $(u_n)$ voldoet aan een affiene recursie
$u_{n+1} = q u_n + r$, en druk $q$ en $r$ uit in $a$ en $b$. Waarmee stemt de
voorwaarde $\abs{q} < 1$ overeen voor de \omterm{def:g12:diffeq:ode}{differentiaalvergelijking}?
\end{exercise}

\section{Opgave: de klok in de dingen}

\begin{problem}[{Weekendopgave --- koolstof-14 dateert de grotten, de
afkoelingswet van Newton klokt een misdaad, en één kleine vergelijking draagt
vier kostuums}]
\label{pb:g12:diffeq:1}
Een \omterm{def:g12:diffeq:ode}{differentiaalvergelijking} is een wet van verandering; haar oplossen maakt
van die wet een \emph{klok}. Diezelfde kleine \omterm{def:g10:algebra:equation}{vergelijking} $y' = ay + b$
(\cref{thm:g12:diffeq:affine}) tikt in prehistorische houtskool, in
afkoelende koffie, in vallende regendruppels en in ziekenhuisinfusen --- en
die klokken aflezen is het werk van deze opgave: ze dateert de
schilderingen van Lascaux, legt een tijdstip van overlijden vast, en
controleert haar eigen aannames zoals een goede wetenschapper betaamt.

\medskip
\noindent\textbf{Deel I --- Vlot rekenen.}
\begin{enumerate}
  \item Los $y' = 3y$ met $y(0) = 2$ op; en daarna $y' = -2y + 6$ met
        $y(0) = 0$ (\cref{met:g12:diffeq:solve}).
  \item Leid de uniciteitstruc achter \cref{thm:g12:diffeq:homogeneous}
        opnieuw af: bereken voor $y' = ay$ de \omterm{def:g12:deriv:derivative}{afgeleide} van
        $y(t)\,\eu^{-at}$ en besluit dat \emph{elke} oplossing $C\eu^{at}$
        is.
  \item Zoek voor $y' = -2y + 6$ de evenwichtsoplossing, en beschrijf het lot
        van elke andere oplossing voor $t \to +\infty$. (De vaste punten van
        de recursies uit \cref{pb:g11:seq:1}, maar dan \omterm{def:g12:limcont:continuity}{continu}.)
  \item Toon voor de vervalwet $y' = -\frac{y}{\tau}$ aan dat de
        halveringstijd $t_{1/2} = \tau \ln 2$ is.
  \item Schets (of beschrijf) de familie oplossingen van $y' = -2y + 6$: wat
        doen de oplossingen die boven $3$ starten? En die eronder? En die op
        $3$?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Koolstof-14.}
Levend weefsel houdt een constante verhouding radioactieve koolstof-14 aan;
bij het overlijden stopt de opname en vervalt de voorraad:
$N' = -\lambda N$, met halveringstijd $5\,730$ jaar.
\begin{enumerate}[resume]
  \item Bereken $\lambda$ (per jaar).
  \item Een bot bevat nog $20\,\%$ van de verhouding van levend weefsel: hoe
        oud is het?
  \item Houtskool uit de beschilderde grotten van Lascaux bevat er nog
        ongeveer $15\,\%$: dateer de schilderingen.
  \item De ijsmummie Ötzi, gevonden in een Alpengletsjer, mat ongeveer
        $53\,\%$: dateer hem (de archeologen zeggen ongeveer $5\,300$ jaar
        --- hoe deed jij het?).
  \item Waarom kan koolstof-14 geen dinosauriërs dateren? Bereken de fractie
        die na tien halveringstijden overblijft, geef de fractie na $65$
        miljoen jaar als een macht van $2$, en besluit.
  \item Foutenmarges: is de $53\,\%$ van Ötzi slechts tot op $\pm 1\,\%$
        bekend, bereken dan het ouderdomsbereik. Welke nauwkeurigheid in het
        labo koopt welke nauwkeurigheid in de datering?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De afkoeling van Newton, en een misdaad.}
Een lichaam met temperatuur $T$ in een omgeving met constante $T_a$ koelt af
volgens $T' = -k\,(T - T_a)$.
\begin{enumerate}[resume]
  \item Los de \omterm{def:g10:algebra:equation}{vergelijking} op (substitueer $z = T - T_a$):
        \[
        T(t) = T_a + (T_0 - T_a)\,\eu^{-kt} .
        \]
  \item Koffie die op $90\,^\circ$C wordt uitgeschonken in een kamer van
        $20\,^\circ$C staat na $5$ minuten op $70\,^\circ$C. Bepaal $k$, en
        daarna de wachttijd tot de drinkbare $55\,^\circ$C.
  \item De melkvraag: om over tien minuten de warmst mogelijke koffie te
        drinken, moet de koude melk er nu in of pas op het laatste ogenblik?
        Antwoord met de \omterm{def:g10:algebra:equation}{vergelijking} (wat doet melk toevoegen met het \omterm{def:g11:seq:arithmetic}{verschil}
        $T - T_a$, en hoe stuurt dat \omterm{def:g11:seq:arithmetic}{verschil} het verlies aan?).
  \item Forensisch onderzoek: om middernacht wordt een lichaam gevonden van
        $30\,^\circ$C in een kamer van $20\,^\circ$C; een uur later meet het
        $28\,^\circ$C. Neem $37\,^\circ$C aan op het ogenblik van overlijden,
        haal $k$ uit de twee metingen, en bereken daarna het tijdstip van
        overlijden.
  \item Controleer de klok van de wetsdokter: noem drie manieren waarop de
        werkelijkheid de aannames van het model schendt, en de richting
        waarin elk het geschatte tijdstip van overlijden zou vertekenen.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Vier kostuums en een S-kromme.}
\begin{enumerate}[resume]
  \item De logistische \omterm{def:g10:algebra:equation}{vergelijking} van \cref{exo:g12:diffeq:8},
        $y' = y(1 - y)$ met $y(0) = 0.1$: leid met de substitutie van die
        oefening af dat $y(t) = \dfrac{1}{1 + 9\eu^{-t}}$, en zoek het
        tijdstip van het \omterm{def:g12:deriv:inflection}{buigpunt} $y = \frac12$ --- de datum van de
        epidemioloog uit \cref{pb:g12:deriv:1}, nu berekenbaar.
  \item Een regendruppel voldoet aan $v' = 10 - \frac v2$ (zwaartekracht min
        weerstand) met $v(0) = 0$. Zoek de eindsnelheid, los op naar $v(t)$,
        en bereken wanneer de druppel $95\,\%$ van de eindsnelheid haalt.
  \item Een ziekenhuisinfuus dient een geneesmiddel toe met een constant
        debiet terwijl het lichaam het evenredig verwijdert:
        $c' = 4 - 0.5c$ met $c(0) = 0$. Zoek de evenwichtsconcentratie en de
        tijd om de helft ervan te bereiken. Formuleer in één zin de
        overeenkomst met de leningrecursies uit \cref{exo:g12:diffeq:9}.
  \item Slotstuk --- één \omterm{def:g10:algebra:equation}{vergelijking}, vier kostuums: verval, afkoeling,
        vallen en doseren, alle $y' = ay + b$ met andere tekens en andere
        namen. Herhaal de modelleerlus die deze opgave vier keer doorliep
        (wet $\to$ \omterm{def:g10:algebra:equation}{vergelijking} $\to$ oplossing $\to$ ijking $\to$
        voorspelling $\to$ controle), en noem het verschijnsel dat $y''$
        nodig heeft --- en welke opgave het al ontmoette.
\end{enumerate}
\end{problem}
