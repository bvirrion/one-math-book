\chapter{Toevalsvariabelen en de binomiale verdeling}\label{ch:g12:randvar}

Een \omterm{def:g12:randvar:rv}{toevalsvariabele} hangt aan elke uitkomst van een experiment een getal:
een winst, een aantal, een duur. Haar \omterm{def:g12:randvar:exp}{verwachtingswaarde} is het \omterm{def:g11:stat:mean}{gemiddelde} op
lange termijn van de waarden die ze voortbrengt, haar \omterm{def:g12:randvar:exp}{variantie} meet hun
spreiding. De ster van dit hoofdstuk is de \omterm{def:g12:randvar:binomial}{binomiale verdeling}, die de
successen telt in herhaalde onafhankelijke proeven. \omterm{def:g12:randvar:rv}{Toevalsvariabelen} en de
\omterm{def:g12:randvar:binomial}{binomiale verdeling} kwamen voor het eerst aan bod in
\cref{ch:g11:prob,ch:g11:binom}; dit hoofdstuk herneemt en verdiept ze, nu de
combinatorische gereedschappen van \cref{ch:g12:comb} beschikbaar zijn.

\section{Discrete toevalsvariabelen}

\begin{definition}[Toevalsvariabele, kansverdeling]\label{def:g12:randvar:rv}
Een \emph{toevalsvariabele}\index{toevalsvariabele} op een eindige
\omterm{def:g11:prob:model}{uitkomstenverzameling} $\Omega$ is een \omterm{def:g11:func:function}{functie} $X \colon \Omega \to \R$. Haar
\emph{kansverdeling}\index{kansverdeling} bestaat uit haar mogelijke waarden
$x_1, \dots, x_k$ en de kansen
\[
p_i = \P(X = x_i), \qquad \sum_{i=1}^{k} p_i = 1 .
\]
\end{definition}

\begin{definition}[Verwachtingswaarde, variantie, standaardafwijking]\label{def:g12:randvar:exp}
De \emph{verwachtingswaarde}\index{verwachtingswaarde} van $X$ is
\[
\E(X) = \sum_{i=1}^{k} p_i\, x_i ,
\]
haar \emph{variantie}\index{variantie} en haar \emph{\omterm{def:g11:stat:variance}{standaardafwijking}} zijn
\[
\V(X) = \E\bigl((X - \E(X))^2\bigr) = \sum_{i=1}^k p_i\bigl(x_i - \E(X)\bigr)^2,
\qquad
\sigma(X) = \sqrt{\V(X)} .
\]
\end{definition}

\begin{proposition}[Formule van König--Huygens]\label{prop:g12:randvar:konig}
\index{formule van König--Huygens}
$\V(X) = \E(X^2) - \E(X)^2$.
\end{proposition}

\begin{proof}
Schrijf $m = \E(X)$ en werk uit:
\[
\V(X) = \sum_i p_i (x_i^2 - 2m x_i + m^2)
= \E(X^2) - 2m\sum_i p_i x_i + m^2\sum_i p_i
= \E(X^2) - 2m^2 + m^2 . \qedhere
\]
\end{proof}

\begin{proposition}[Affiene transformatie]\label{prop:g12:randvar:affine}
Voor $a, b \in \R$ geldt
\[
\E(aX + b) = a\,\E(X) + b, \qquad
\V(aX + b) = a^2\,\V(X) .
\]
\end{proposition}

\begin{proof}
De eerste is een herschikking van de definiërende som. Voor de tweede: $aX + b$
wijkt van zijn \omterm{def:g11:stat:mean}{gemiddelde} af over $a(X - \E(X))$, en kwadrateren
vermenigvuldigt met $a^2$.
\end{proof}

\begin{example}[Eerlijke spelen]\label{ex:g12:randvar:game}
Een spel kost $2$ euro; er wordt met een dobbelsteen gegooid, en de speler
ontvangt de getoonde waarde als die minstens $5$ is, en anders niets. De winst
$G$ neemt de waarden $-2$ (\omterm{def:g10:proba:distribution}{kans} $\frac46$), $3$ ($\frac16$) en $4$
($\frac16$) aan:
\[
\E(G) = \frac{-8 + 3 + 4}{6} = -\frac{1}{6} < 0 .
\]
Gemiddeld verliest de speler $17$ cent per spel: het spel is ongunstig (zoals
de meeste echte spelen).
\end{example}

\section{Bernoulli-experimenten en de binomiale verdeling}

\begin{definition}[Verdeling van Bernoulli]\label{def:g12:randvar:bernoulli}
Een \emph{bernoulli-experiment}\index{bernoulli-experiment} is een experiment
met twee uitkomsten, \emph{succes} (\omterm{def:g10:proba:distribution}{kans} $p$) en \emph{mislukking}
($q = 1 - p$). De indicator $X$ van succes ($X = 1$ bij succes, $0$ bij
mislukking) volgt de \emph{verdeling van Bernoulli} $\mathcal B(p)$:
\[
\E(X) = p, \qquad \V(X) = p(1-p) .
\]
(Inderdaad is $\E(X) = p$ en $\E(X^2) = p$, en König--Huygens geeft
$\V(X) = p - p^2$.)
\end{definition}

\begin{definition}[Binomiale verdeling]\label{def:g12:randvar:binomial}
Herhaal een \omterm{def:g12:randvar:bernoulli}{bernoulli-experiment} $n$ keer \omterm{def:g12:condprob:indep}{onafhankelijk}, en zij $X$ het totale
aantal successen. De \omterm{def:g12:randvar:rv}{kansverdeling} van $X$ is de \emph{binomiale
verdeling}\index{binomiale verdeling} $\mathcal B(n, p)$.
\end{definition}

\begin{theorem}\label{thm:g12:randvar:binomial}
Is $X \sim \mathcal B(n, p)$, dan geldt voor $0 \leq k \leq n$
\[
\P(X = k) = \binom{n}{k} p^k (1-p)^{n-k},
\]
en
\[
\E(X) = np, \qquad \V(X) = np(1-p) .
\]
\end{theorem}

\begin{proof}
Een opgegeven opeenvolging van uitkomsten met $k$ successen en $n-k$
mislukkingen heeft wegens de \omterm{def:g12:condprob:indep}{onafhankelijkheid} \omterm{def:g10:proba:distribution}{kans} $p^k(1-p)^{n-k}$; het
aantal zulke opeenvolgingen is het aantal manieren om de $k$ successen onder
de $n$ proeven te plaatsen, namelijk $\binom nk$ (\cref{ch:g12:comb}).
Optellen over de opeenvolgingen geeft de formule --- en het binomium van
Newton bevestigt dat $\sum_k \P(X = k) = (p + q)^n = 1$.

Voor de \omterm{def:g12:randvar:exp}{verwachtingswaarde}, met $k\binom nk = n\binom{n-1}{k-1}$
(\cref{exo:g12:comb:7}):
\[
\E(X) = \sum_{k=1}^{n} k\binom nk p^k q^{n-k}
= np\sum_{k=1}^{n} \binom{n-1}{k-1} p^{k-1} q^{(n-1)-(k-1)}
= np\,(p + q)^{n-1} = np .
\]
De formule voor de \omterm{def:g12:randvar:exp}{variantie} bewijs je net zo met de identiteit
$k(k-1)\binom nk = n(n-1)\binom{n-2}{k-2}$, die $\E(X(X-1)) = n(n-1)p^2$
geeft, waaruit $\V(X) = n(n-1)p^2 + np - (np)^2 = np(1-p)$. (Een structureel
bewijs --- de \omterm{def:g12:randvar:exp}{variantie} van een som van onafhankelijke variabelen --- komt met
\cref{thm:g12:sums:variance}.)
\end{proof}

\begin{method}[Een binomiale situatie herkennen]\label{met:g12:randvar:recognize}
Ga de drie bestanddelen na voordat je $X \sim \mathcal B(n,p)$ schrijft: een
\emph{vast aantal} $n$ proeven; \emph{twee uitkomsten} per proef met
\emph{dezelfde} \omterm{def:g10:proba:distribution}{kans} $p$ op succes; \emph{\omterm{def:g12:condprob:indep}{onafhankelijkheid}} van de proeven
(trekken \emph{met} teruglegging, of uit een grote populatie). Gebruik dan
\[
\P(X \geq 1) = 1 - (1-p)^n
\]
voor ``minstens één succes'', en een rekentoestel of cumulatieve tabellen voor
$\P(X \leq k)$ in het algemeen.
\end{method}

\begin{example}\label{ex:g12:randvar:atleastone}
Hoe vaak moet je met een dobbelsteen gooien om minstens $99\%$ \omterm{def:g10:proba:distribution}{kans} op een zes
te hebben? Met $X \sim \mathcal B\left(n, \frac16\right)$ betekent
$\P(X \geq 1) = 1 - \left(\frac56\right)^n \geq 0.99$ dat
$\left(\frac56\right)^n \leq 0.01$, dat wil zeggen
$n \geq \frac{\ln 0.01}{\ln(5/6)} \approx 25.3$: vanaf $n = 26$ worpen.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:randvar:1}
Een \omterm{def:g12:randvar:rv}{toevalsvariabele} $X$ neemt de waarden $-1, 0, 2, 5$ aan met de kansen
$0.3, 0.2, 0.4, 0.1$. Bereken $\E(X)$, $\V(X)$ en $\sigma(X)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:randvar:2}
Een meerkeuzetoets heeft $10$ vragen met elk $4$ mogelijkheden, waarvan er één
juist is. Een leerling antwoordt uniform willekeurig en \omterm{def:g12:condprob:indep}{onafhankelijk}. Zij $X$
het aantal juiste antwoorden.
\begin{enumerate}
  \item Geef de \omterm{def:g12:randvar:rv}{kansverdeling} van $X$, $\E(X)$ en $\sigma(X)$.
  \item Bereken $\P(X = 0)$, $\P(X = 5)$ en $\P(X \geq 1)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:randvar:3}
Een verzekeraar verzekert $n = 400$ klanten; elk dient in de loop van het jaar
een schadeclaim in met \omterm{def:g10:proba:distribution}{kans} $p = 0.05$, \omterm{def:g12:condprob:indep}{onafhankelijk} van de andere. Zij $X$
het aantal claims. Bepaal de \omterm{def:g12:randvar:rv}{kansverdeling} van $X$ en bereken haar
\omterm{def:g12:randvar:exp}{verwachtingswaarde} en \omterm{def:g11:stat:variance}{standaardafwijking}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:4}
De organisator van het spel uit \cref{ex:g12:randvar:game} wil er een eerlijk
spel van maken ($\E(G) = 0$) door de inzet $c$ aan te passen. Bepaal $c$.
Bereken de \omterm{def:g12:randvar:exp}{variantie} van de winst voor die eerlijke versie; is ``eerlijk''
hetzelfde als ``zonder risico''?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:5}
Een basketbalspeelster scoort vrijworpen met \omterm{def:g10:proba:distribution}{kans} $0.7$. Ze werpt $8$ keer
(onafhankelijke worpen). Bereken de \omterm{def:g10:proba:distribution}{kans} dat ze precies $6$ keer scoort, dat
ze minstens $6$ keer scoort, en dat ze minstens één keer scoort. Wat is het
waarschijnlijkste aantal treffers?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:6}
Een luchtvaartmaatschappij weet dat elke geboekte passagier met \omterm{def:g10:proba:distribution}{kans} $0.9$
komt opdagen, \omterm{def:g12:condprob:indep}{onafhankelijk} van de andere. Een vlucht heeft $100$
zitplaatsen en de maatschappij verkoopt $104$ tickets. Druk met een \omterm{def:g12:randvar:binomial}{binomiale
verdeling} de \omterm{def:g10:proba:distribution}{kans} uit dat er meer passagiers komen opdagen dan er plaatsen
zijn, en schat ze numeriek met een rekentoestel (geef de exacte uitdrukking).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:7}
Zij $X \sim \mathcal B(n, p)$. Toon aan dat
\[
\frac{\P(X = k+1)}{\P(X = k)} = \frac{n-k}{k+1}\cdot\frac{p}{1-p},
\]
en leid af dat de verdeling stijgt tot $k^* = \floor{(n+1)p}$ en daarna daalt
(de \emph{modus} van de \omterm{def:g12:randvar:binomial}{binomiale verdeling}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:randvar:8}
\emph{(Sint-Petersburg, getemd.)} Een zuivere munt wordt opgeworpen tot er
kruis valt, maar hoogstens $10$ keer. Zij $N$ het aantal gebruikte worpen; de
speler ontvangt $2^N$ euro als er kruis viel, en anders niets.
\begin{enumerate}
  \item Geef de verdeling van $N$ beperkt tot de winnende uitkomsten:
        $\P(\text{eerste kruis bij worp } k) = 2^{-k}$ voor
        $1 \leq k \leq 10$, en ga de totale \omterm{def:g10:proba:distribution}{kans} om te winnen na.
  \item Bereken de verwachte uitbetaling. Wat zou ze worden zonder de grens
        van $10$ worpen?
\end{enumerate}
\end{exercise}

\section{Opgave: de overboekte vlucht}

\begin{problem}[{Weekendopgave --- luchtvaartmaatschappijen verkopen meer
plaatsen dan ze hebben, fabrieken aanvaarden partijen die ze amper
gecontroleerd hebben, en de binomiale verdeling is in beide gevallen
scheidsrechter}]
\label{pb:g12:randvar:1}
Een luchtvaartmaatschappij met $100$ zitplaatsen verkoopt vrolijk $105$
tickets: ongeveer $10\,\%$ van de passagiers komt nooit opdagen, en lege
stoelen brengen niets op. Hoe ver kan de maatschappij gaan voordat de
geweigerde passagiers de winst opeten? De \omterm{def:g12:randvar:binomial}{binomiale verdeling}
(\cref{thm:g12:randvar:binomial}) antwoordt tot op de komma --- en dezelfde
machinerie keurt fabriekspartijen, prijst tombola's en houdt verzekeraars
solvabel. Beslissen onder herhaling: dat is het dagelijkse werk van de
\omterm{def:g12:randvar:binomial}{binomiale verdeling}.

\medskip
\noindent\textbf{Deel I --- Vlotheid.}
\begin{enumerate}
  \item Geef de volledige tabel van de \omterm{def:g12:randvar:rv}{kansverdeling} van het aantal zessen $X$
        in drie worpen met een dobbelsteen.
  \item Bereken $\E(X)$ en $V(X)$ (\cref{def:g12:randvar:exp}).
  \item Een kraam vraagt $1$ euro voor drie worpen en betaalt $2$ euro per
        behaalde zes. Bereken de verwachte winst: een eerlijk spel?
  \item Oefening met de checklist (\cref{met:g12:randvar:recognize}): is het
        aantal harten in $5$ kaarten die \emph{zonder} teruglegging getrokken
        worden binomiaal verdeeld? En met teruglegging? Verantwoord.
  \item Geef voor $X \sim \mathcal B(20,\ 0.3)$ de waarden $\E(X)$ en
        $\sigma(X)$, en daarna $\E(S)$ en $\sigma(S)$ voor de score
        $S = 5X - 10$ (\cref{prop:g12:randvar:affine}).
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De overboekte vlucht.}
Zitplaatsen: $100$. Verkochte tickets: $105$. Elke tickethouder komt met \omterm{def:g10:proba:distribution}{kans}
$0.9$ opdagen, \omterm{def:g12:condprob:indep}{onafhankelijk} van de andere; zij $X$ het aantal dat opdaagt.
\begin{enumerate}[resume]
  \item Model: verantwoord $X \sim \mathcal B(105,\ 0.9)$ met de checklist ---
        en beken het zwakste punt van het model (zijn de afwezigen werkelijk
        \omterm{def:g12:condprob:indep}{onafhankelijk}? denk aan groepen en aan stormen).
  \item Bereken $\E(X)$ en $\sigma(X)$.
  \item Er worden passagiers geweigerd zodra $X \geq 101$: schrijf
        $\P(X \geq 101)$ als een expliciete som van vijf binomiale termen.
  \item Bereken die som (met het rekentoestel): bij welk aandeel van de
        vluchten wordt er minstens één passagier geweigerd?
  \item Geld: de vijf extra tickets brengen $5 \times 200 = 1\,000$ euro op;
        elke geweigerde passagier kost $800$ euro aan vergoeding. Bereken het
        verwachte aantal geweigerde passagiers,
        $\sum_{k=101}^{105} (k - 100)\,\P(X = k)$, de verwachte vergoeding, en
        het oordeel over dit beleid.
  \item Ontwerp: de toezichthouder aanvaardt $\P(\text{weigering}) < 5\,\%$.
        Zoek door $n = 105, 106, 107, \dots$ tickets te toetsen de grootste
        toegelaten $n$.
  \item De sluipweg via de klok: bereken voor $n = 105$ de \omterm{pb:g11:stat:1}{z-score} van de
        weigeringsdrempel, $z = \frac{100.5 - \E(X)}{\sigma(X)}$, en vergelijk
        de ruwe schatting ``ongeveer $2\sigma$, dus zowat $2$ à $3\,\%$ in de
        bovenstaart'' met je exacte antwoord. (De kromme achter die sluipweg
        is de ster van de volgende hoofdstukken.)
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De fabriekspoort.}
Een partij onderdelen wordt aanvaard als een \omterm{def:g10:proba:sample}{steekproef} van $20$ hoogstens één
defect stuk bevat.
\begin{enumerate}[resume]
  \item Bereken de \omterm{def:g10:proba:distribution}{kans} op aanvaarding als het werkelijke uitvalpercentage
        $2\,\%$ is (een eerlijke partij).
  \item Bereken ze opnieuw als het percentage $10\,\%$ is (een slechte
        partij).
  \item Noem de twee risico's van deze procedure (de eerlijke partij wordt
        geweigerd; de slechte partij wordt aanvaard), lees hun waarden af uit
        de vragen 13 en 14, en zeg welke ene aanpassing beide tegelijk
        verbetert --- en tegen welke prijs.
  \item Achter die verbetering: toon aan dat de waargenomen uitvalfrequentie
        $\frac Xn$ in een \omterm{def:g10:proba:sample}{steekproef} van omvang $n$ \omterm{def:g11:stat:variance}{standaardafwijking}
        $\sqrt{\frac{p(1-p)}{n}}$ heeft, en besluit hoe de precisie met de
        omvang van de \omterm{def:g10:proba:sample}{steekproef} schaalt (een oude bekende: de
        $\frac{1}{\sqrt n}$ uit het eerste jaar van dit volume).
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Spelen, tombola's, reserves.}
\begin{enumerate}[resume]
  \item Het begrensde spel van Sint-Petersburg uit \cref{exo:g12:randvar:8}
        heeft $10$ euro als verwachte uitbetaling. Zet dat in één alinea af
        tegen de onbegrensde paradox uit \cref{pb:g11:prob:1}: wat temt de
        grens precies, en welke prijs zou nu eerlijk zijn?
  \item Een tombola voor het goede doel verkoopt $200$ tickets aan $2$ euro;
        de prijzen zijn één mand van $100$ euro en twee manden van $50$ euro.
        Bereken de verwachte winst per ticket en de marge van de tombola;
        vergelijk met de huismarge van $\frac{1}{37} \approx 2.7\,\%$ bij het
        Europese roulettespel. Waarom komt de tombola ermee weg?
  \item Een verzekeraar heeft $1\,000$ onafhankelijke polissen: \omterm{def:g10:proba:distribution}{kans} op een
        claim $0.01$ per polis, schadebedrag $10\,000$ euro, premie $130$
        euro. Bereken de verwachte jaarwinst \emph{en} de \omterm{def:g11:stat:variance}{standaardafwijking}
        van de totale schade. Vergelijk beide getallen: wat dwingt die
        \omterm{def:g10:algebra:equation}{vergelijking} echte verzekeraars aan te houden?
  \item Slotstuk --- de \omterm{def:g12:randvar:binomial}{binomiale verdeling} als scheidsrechter van
        beslissingen: de checklist om ze te herkennen; het kompas
        $\E \pm \sigma$; staartkansen als de prijs van het risico; en de twee
        blijvende voorbehouden (\omterm{def:g12:condprob:indep}{onafhankelijkheid} is een bewering over het
        model, en zeldzame staarten, niet \omterm{def:g11:stat:mean}{gemiddelden}, veroorzaken het
        bankroet). Telkens één zin, met de blik vooruit: de wet van de grote
        aantallen en de klokkromme, de volgende hoofdstukken, vervolledigen
        het reglement van de scheidsrechter.
\end{enumerate}
