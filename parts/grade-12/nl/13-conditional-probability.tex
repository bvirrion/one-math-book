\chapter{Voorwaardelijke kansen en onafhankelijkheid}\label{ch:g12:condprob}

De kansrekening meet onzekerheid; de \omterm{def:g12:condprob:cond}{voorwaardelijke kans} meet hoe informatie
die onzekerheid verandert. Vernemen dat een \omterm{def:g11:prob:model}{gebeurtenis} $B$ zich voorgedaan
heeft, hertekent de kansen van alle andere gebeurtenissen --- een mechanisme
dat door Bayes geformaliseerd werd en dagelijks misbruikt wordt in
rechtszalen en kranten. Dit hoofdstuk zet de regels van het conditioneren op
en geeft de precieze betekenis van \omterm{def:g12:condprob:indep}{onafhankelijkheid}.

\section{Kansruimten (herhaling)}

Een experiment met eindig veel uitkomsten wordt gemodelleerd door een
\emph{\omterm{def:g11:prob:model}{uitkomstenverzameling}} $\Omega$ (de verzameling van de uitkomsten) en
een \omterm{def:g10:proba:distribution}{kans} $\P$ die aan elke \omterm{def:g11:prob:model}{gebeurtenis} $A \subseteq \Omega$ een getal
$\P(A) \in \intcc{0}{1}$ toekent, additief over disjuncte \omterm{def:g10:numbers:interunion}{verenigingen} en met
$\P(\Omega) = 1$. Herinner je de basisregels:
\[
\P(\bar A) = 1 - \P(A),
\qquad
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B).
\]
Zijn alle uitkomsten even waarschijnlijk, dan is
$\P(A) = \frac{\abs A}{\abs\Omega}$ --- en herleidt het berekenen van kansen
zich tot de teltechnieken van \cref{ch:g12:comb}.

\section{Voorwaardelijke kans}

\begin{definition}[Voorwaardelijke kans]\label{def:g12:condprob:cond}
Zij $B$ een \omterm{def:g11:prob:model}{gebeurtenis} met $\P(B) > 0$. De \emph{voorwaardelijke
kans}\index{voorwaardelijke kans} op $A$ gegeven $B$ is
\[
\pcond{B}{A} = \frac{\P(A \cap B)}{\P(B)} .
\]
\end{definition}

De afbeelding $A \mapsto \pcond BA$ is zelf een \omterm{def:g10:proba:distribution}{kans} (alle regels gelden); ze
geeft de nieuwe kennistoestand weer van wie vernomen heeft dat $B$ zich
voorgedaan heeft.

\begin{proposition}[Productregel]\label{prop:g12:condprob:product}
Voor gebeurtenissen met kansen verschillend van nul geldt
\[
\P(A \cap B) = \P(B)\,\pcond{B}{A} = \P(A)\,\pcond{A}{B},
\]
en algemener
$\P(A_1 \cap A_2 \cap A_3) = \P(A_1)\,\pcond{A_1}{A_2}\,
\pcond{A_1 \cap A_2}{A_3}$, enzovoort.
\end{proposition}

\begin{proof}
Herschik de definitie; de kettingformule volgt door te herhalen.
\end{proof}

\begin{theorem}[Wet van de totale kans]\label{thm:g12:condprob:total}
\index{wet van de totale kans}
Zij $B_1, \dots, B_n$ een partitie van $\Omega$ in gebeurtenissen met \omterm{def:g10:proba:distribution}{kans}
verschillend van nul. Voor elke \omterm{def:g11:prob:model}{gebeurtenis} $A$ geldt
\[
\P(A) = \sum_{i=1}^{n} \P(B_i)\,\pcond{B_i}{A}.
\]
\end{theorem}

\begin{proof}
De verzamelingen $A \cap B_i$ zijn paarsgewijs disjunct met \omterm{def:g10:numbers:interunion}{vereniging} $A$,
dus is $\P(A) = \sum_i \P(A \cap B_i) = \sum_i \P(B_i)\pcond{B_i}{A}$ volgens
de productregel.
\end{proof}

\begin{method}[Boomdiagrammen voor kansen]\label{met:g12:condprob:tree}
Een \omterm{met:g10:proba:tree}{boomdiagram} ordent de voorwaardelijke kansen: elke tak draagt de \omterm{def:g10:proba:distribution}{kans} op
de volgende \omterm{def:g11:prob:model}{gebeurtenis} \emph{gegeven} het tot dan afgelegde pad.
\begin{itemize}
  \item De \omterm{def:g10:proba:distribution}{kans} op een blad (een volledig pad) is het \emph{product} van de
        kansen langs zijn takken (productregel).
  \item De \omterm{def:g10:proba:distribution}{kans} op een \omterm{def:g11:prob:model}{gebeurtenis} is de \emph{som} van de kansen op de
        bladeren die ze verwezenlijken (totale \omterm{def:g10:proba:distribution}{kans}).
  \item De kansen op de takken die uit één knoop vertrekken hebben som $1$.
\end{itemize}
\end{method}

\begin{theorem}[Formule van Bayes]\label{thm:g12:condprob:bayes}
\index{formule van Bayes}
Zij $B_1, \dots, B_n$ een partitie van $\Omega$ zoals hierboven en $A$ een
\omterm{def:g11:prob:model}{gebeurtenis} met $\P(A) > 0$. Dan geldt
\[
\pcond{A}{B_j} =
\frac{\P(B_j)\,\pcond{B_j}{A}}{\sum_{i=1}^{n} \P(B_i)\,\pcond{B_i}{A}} .
\]
\end{theorem}

\begin{proof}
$\pcond{A}{B_j} = \frac{\P(A \cap B_j)}{\P(A)}$; werk de teller uit met de
productregel en de noemer met de wet van de totale \omterm{def:g10:proba:distribution}{kans}.
\end{proof}

\begin{example}[Screeningstest]\label{ex:g12:condprob:test}
Een ziekte treft $1\%$ van een bevolking. Een test spoort ze op met \omterm{def:g10:proba:distribution}{kans}
$0.99$ (gevoeligheid) en geeft een vals positief resultaat met \omterm{def:g10:proba:distribution}{kans} $0.05$.
Gegeven een positieve test is de \omterm{def:g10:proba:distribution}{kans} dat je de ziekte werkelijk hebt
\[
\frac{0.01 \times 0.99}{0.01\times0.99 + 0.99\times0.05}
= \frac{0.0099}{0.0099 + 0.0495} = \frac{1}{6} \approx 0.17 .
\]
Ondanks de nauwkeurige test zijn vijf van de zes positieven vals --- omdat de
ziekte zeldzaam is. $\pcond{A}{B}$ met $\pcond{B}{A}$ verwarren is de
\emph{denkfout van de aanklager}.
\end{example}

\section{Onafhankelijkheid}

\begin{definition}[Onafhankelijke gebeurtenissen]\label{def:g12:condprob:indep}
Twee gebeurtenissen $A$ en $B$ heten
\emph{onafhankelijk}\index{onafhankelijkheid} als
\[
\P(A \cap B) = \P(A)\,\P(B) .
\]
Is $\P(B) > 0$, dan is dit gelijkwaardig met $\pcond{B}{A} = \P(A)$: $B$
kennen verandert de \omterm{def:g10:proba:distribution}{kans} op $A$ niet.
\end{definition}

\begin{proposition}\label{prop:g12:condprob:indepcompl}
Zijn $A$ en $B$ \omterm{def:g12:condprob:indep}{onafhankelijk}, dan zijn $A$ en $\bar B$ dat ook (en $\bar A$
en $\bar B$ eveneens).
\end{proposition}

\begin{proof}
$\P(A \cap \bar B) = \P(A) - \P(A \cap B) = \P(A) - \P(A)\P(B)
= \P(A)\bigl(1 - \P(B)\bigr) = \P(A)\,\P(\bar B)$.
\end{proof}

\begin{remark}
Verwar \emph{\omterm{def:g12:condprob:indep}{onafhankelijk}} ($\P(A\cap B) = \P(A)\P(B)$) niet met
\emph{\omterm{def:g10:proba:operations}{onverenigbaar}} ($A \cap B = \varnothing$). Twee onverenigbare
gebeurtenissen met \omterm{def:g10:proba:distribution}{kans} verschillend van nul zijn nooit \omterm{def:g12:condprob:indep}{onafhankelijk}: weten
dat de ene zich voordeed, waarborgt dat de andere zich niet voordeed.
\end{remark}

\begin{definition}[Onafhankelijke herhalingen]\label{def:g12:condprob:repet}
Wordt een experiment $n$ keer herhaald zodat de uitkomst van elke proef de
andere niet beïnvloedt, dan is de \omterm{def:g10:proba:distribution}{kans} op een opgegeven opeenvolging van
uitkomsten het product van de afzonderlijke kansen. Dat is het model achter
de binomiale verdeling (\cref{ch:g12:randvar}).
\end{definition}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g12:condprob:1}
Er wordt een kaart getrokken uit een spel van 52 kaarten. Bereken de \omterm{def:g10:proba:distribution}{kans} dat
het een heer is, gegeven dat het een pop is (boer, dame of heer). Zijn de
gebeurtenissen ``heer'' en ``harten'' \omterm{def:g12:condprob:indep}{onafhankelijk}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:condprob:2}
Een urne bevat 5 rode en 3 blauwe ballen. Er worden twee ballen na elkaar
getrokken, \emph{zonder} teruglegging.
\begin{enumerate}
  \item Teken het \omterm{met:g10:proba:tree}{boomdiagram} met de kansen.
  \item Bereken de \omterm{def:g10:proba:distribution}{kans} dat beide rood zijn, en de \omterm{def:g10:proba:distribution}{kans} dat de tweede rood is.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:condprob:3}
Er wordt met twee zuivere dobbelstenen gegooid. Zij $A$ = ``de som is $7$'',
$B$ = ``de eerste dobbelsteen toont $3$'' en $C$ = ``de som is $6$''. Bepaal
of $A$ en $B$ \omterm{def:g12:condprob:indep}{onafhankelijk} zijn, en daarna of $B$ en $C$ dat zijn.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:4}
Een fabriek heeft drie machines die respectievelijk $50\%$, $30\%$ en $20\%$
van de totale productie leveren, met uitvalpercentages $1\%$, $2\%$ en $4\%$.
\begin{enumerate}
  \item Welk aandeel van de productie is defect?
  \item Een willekeurig gekozen stuk is defect. Wat is de \omterm{def:g10:proba:distribution}{kans} dat het van de
        derde machine komt?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:5}
Bij welke prevalentie $p$ van de ziekte (in plaats van $1\%$) zou een
positieve test in \cref{ex:g12:condprob:test} minstens $90\%$ \omterm{def:g10:proba:distribution}{kans} op ziekte
betekenen? Los de ongelijkheid op en geef commentaar.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:6}
Een valse munt valt op kruis met \omterm{def:g10:proba:distribution}{kans} $p \in \intoo{0}{1}$. Ze wordt drie keer
opgeworpen, waarbij de worpen \omterm{def:g12:condprob:indep}{onafhankelijk} zijn.
\begin{enumerate}
  \item Bereken de \omterm{def:g10:proba:distribution}{kans} op de \omterm{def:g11:prob:model}{gebeurtenis} $E$ = ``precies twee keer kruis''.
  \item Voor welke $p$ is $\P(E)$ maximaal?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:condprob:7}
\emph{(Monty Hall.)} Achter een van drie deuren zit een prijs verstopt. Je
kiest een deur; de presentator, die weet waar de prijs zit, opent een van de
twee overige deuren en toont altijd een lege (en kiest willekeurig als beide
leeg zijn), waarna hij je aanbiedt om naar de andere gesloten deur over te
stappen. Bereken met de formule van Bayes de \omterm{def:g10:proba:distribution}{kans} om te winnen als je
overstapt, en als je dat niet doet.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:condprob:8}
Een informatiekanaal verstuurt bits. Elke bit wordt door ruis omgeklapt met
\omterm{def:g10:proba:distribution}{kans} $\varepsilon = 0.1$, \omterm{def:g12:condprob:indep}{onafhankelijk} van de andere. Om een bit te
beschermen wordt hij drie keer verstuurd en bij meerderheid gedecodeerd.
\begin{enumerate}
  \item Bereken de \omterm{def:g10:proba:distribution}{kans} dat de gedecodeerde bit fout is.
  \item Het ontvangen woord is $101$. Wat is de \omterm{def:g10:proba:distribution}{kans} dat de verstuurde bit
        $1$ was? (Onderstel dat $0$ en $1$ met gelijke \omterm{def:g10:proba:distribution}{kans} verstuurd worden.)
\end{enumerate}
\end{exercise}

\section{Opgave: de arts, de rechter en het spamfilter}

\begin{problem}[{Weekendopgave --- de formule van Bayes tegen de denkfout van
het basispercentage: waarom een test van $99\,\%$ $2\,\%$ kan betekenen,
waarom een treffer op een miljoen niemand veroordeelt, en hoe je mailbox
denkt}]
\label{pb:g12:condprob:1}
Een test die $99\,\%$ van de gevallen opspoort, komt positief terug --- en de
werkelijke \omterm{def:g10:proba:distribution}{kans} op ziekte ligt onder $2\,\%$. Er wordt een DNA-profiel
gevonden dat op één persoon op een miljoen past --- en de verdachte is
waarschijnlijk onschuldig. Beide uitspraken kloppen, beide zijn door artsen en
rechtbanken verkeerd beoordeeld, en beide worden beslecht door één regel uit
dit hoofdstuk: de formule van Bayes (\cref{thm:g12:condprob:bayes}). Deze
opgave traint ze op urnen, en daarna in de kliniek, de rechtszaal en de
postbus.

\medskip
\noindent\textbf{Deel I --- Vlotheid.}
\begin{enumerate}
  \item Een zuivere munt kiest urne I ($2$ rood, $1$ blauw) of urne II ($1$
        rood, $3$ blauw), waarna één bal getrokken wordt. Bereken
        $\P(\text{rood})$ met de wet van de totale \omterm{def:g10:proba:distribution}{kans}
        (\cref{thm:g12:condprob:total}).
  \item De bal is rood: bereken $\P(\text{urne I} \mid \text{rood})$.
  \item Eén worp met een dobbelsteen: zijn de gebeurtenissen ``even'' en
        ``$\leq 4$'' \omterm{def:g12:condprob:indep}{onafhankelijk} (\cref{def:g12:condprob:indep})? En ``even''
        en ``$\leq 3$''?
  \item Twee onafhankelijke rookmelders slaan elk bij brand met \omterm{def:g10:proba:distribution}{kans} $0.9$
        aan. Bereken met \cref{prop:g12:condprob:indepcompl} de \omterm{def:g10:proba:distribution}{kans} dat er
        minstens één aanslaat.
  \item Twee kaarten zonder teruglegging: bereken $\P(\text{twee heren})$ met
        de productregel (\cref{prop:g12:condprob:product}).
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De arts.}
Een ziekte treft $1$ persoon op $1\,000$. De test spoort $99\,\%$ van de
zieken op (gevoeligheid), maar slaat ook aan bij $5\,\%$ van de gezonden
(vals positieven).
\begin{enumerate}[resume]
  \item Natuurlijke \omterm{def:g10:stats:series}{frequenties}: stel je $100\,000$ mensen voor. Hoeveel zijn
        er ziek, hoeveel gezond? Hoeveel positieven levert elke groep op, en
        hoeveel positieven zijn er in totaal?
  \item Bereken uit die aantallen $\P(\text{ziek} \mid \text{positief})$.
        Verwerk de schok.
  \item Bereken het opnieuw met de formule van Bayes en ga na dat het klopt.
  \item In enquêtes antwoorden de meeste artsen ``ongeveer $99\,\%$''. Noem de
        twee voorwaardelijke kansen die verward worden, en leg uit waarom het
        piepkleine basispercentage het echte antwoord bepaalt.
  \item Een tweede, onafhankelijke test komt ook positief terug. Werk bij:
        bereken met $1.9\,\%$ als nieuwe voorkans
        $\P(\text{ziek} \mid \text{twee positieven})$. Wat doet het
        \emph{opstapelen} van bewijs?
  \item Bereken $\P(\text{ziek} \mid \text{positief})$ als dezelfde test enkel
        gebruikt wordt op een risicogroep met prevalentie $5\,\%$. Leid de
        moraal voor de volksgezondheid af over het massaal screenen op
        zeldzame aandoeningen.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De rechter.}
\begin{enumerate}[resume]
  \item Een DNA-profiel van een plaats delict past op één persoon op een
        miljoen. De verdachte, gevonden door een databank van de $10$ miljoen
        inwoners van de stad te doorzoeken, past erop. Verwacht aantal
        onschuldige treffers in de stad? Wat is, met alleen de treffer,
        $\P(\text{onschuldig} \mid \text{treffer})$ bij \omterm{def:g10:numbers:approx}{benadering}?
  \item Noem de denkfout --- welke \omterm{def:g10:proba:distribution}{kans} citeerde de aanklager, welke had de
        rechtbank nodig? --- en de exacte gelijkenis met vraag 9.
  \item De valstrik van de \omterm{def:g12:condprob:indep}{onafhankelijkheid}: twee zeldzame drama's in één
        gezin werden ooit tot ``één op $73$ miljoen'' gekwadrateerd door ze als
        \omterm{def:g12:condprob:indep}{onafhankelijk} te behandelen, wat een onschuldige moeder veroordeelde
        (de zaak Sally Clark). Herbereken de grootteorde als de gebeurtenissen
        gedeelde risicofactoren hebben zodat de tweede, gegeven de eerste, $5$
        keer waarschijnlijker is dan het basispercentage --- en formuleer de
        les over het vermenigvuldigen van kansen.
  \item Ook de verdediging kan vals spelen: ``tien onschuldigen passen erop,
        dus de treffer bewijst niets'' negeert al de rest. Wat doet de formule
        van Bayes met \emph{verscheidene} bewijsstukken, en welke ene zin zou
        elke jurylid moeten kennen?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Het spamfilter.}
\begin{enumerate}[resume]
  \item Je post bestaat voor $40\,\%$ uit spam. Het woord ``loterij'' komt
        voor in $15\,\%$ van de spam en in $0.5\,\%$ van de eerlijke post. Een
        bericht bevat het: bereken $\P(\text{spam} \mid \text{loterij})$.
  \item Foutcorrectie door herhaling (\cref{exo:g12:condprob:8} voortgezet):
        met omklapkans $\varepsilon = 0.1$, met welke \omterm{def:g10:proba:distribution}{kans} faalt het decoderen
        bij meerderheid van \emph{vijf} kopieën? Vergelijk met de code met
        drie kopieën.
  \item Een bit doorkruist $4$ onafhankelijke ruisende schakels, die hem elk
        met \omterm{def:g10:proba:distribution}{kans} $0.1$ omklappen: hij komt precies dan correct aan wanneer het
        aantal omklappingen even is. Bereken die \omterm{def:g10:proba:distribution}{kans} twee keer: door de
        binomiale termen op te tellen, en met de sierlijke formule
        $\frac{1 + (1 - 2\varepsilon)^4}{2}$.
  \item Echte spamfilters vermenigvuldigen het bewijs van vele woorden alsof
        die \omterm{def:g12:condprob:indep}{onafhankelijk} zijn (``naïeve'' Bayes). Waarom is die onderstelling
        vals voor woorden als ``winnen'' en ``loterij'' --- en waarom werkt het
        filter toch goed? (Eén of twee zinnen.)
  \item Slotstuk --- Bayes als de rekenkunde van het geloof: voorkans maal
        aannemelijkheid, herschalen, nakans. Herhaal de drie oordelen
        (kliniek, rechtszaal, postbus) in telkens één zin, en som daarna de
        drie valstrikken op die deze opgave blootlegde: vergeten
        basispercentages, valse \omterm{def:g12:condprob:indep}{onafhankelijkheid}, en bewijs dat stuk voor
        stuk gelezen wordt.
\end{enumerate}
\end{problem}
