\chapter{المعادلات التفاضلية}\label{ch:g12:diffeq}

تربط \omterm{def:g12:diffeq:ode}{المعادلة التفاضلية} \omterm{def:g11:func:function}{دالة} بمشتقاتها. وتعبّر الفيزياء
والكيمياء والبيولوجيا والاقتصاد عن قوانينها بهذه الصورة: فمعدل
تبرّد جسم، وتفكك نواة مشعّة، ونمو
مجموعة سكانية، كلها عبارات عن $y'$. ويحل هذا الفصل حلًا كاملًا
المعادلات الخطية من المرتبة الأولى ذات المعاملات الثابتة.

\section{المعادلة \texorpdfstring{$y' = ay$}{y' = ay}}

\begin{definition}[المعادلة التفاضلية]\label{def:g12:diffeq:ode}
\emph{المعادلة التفاضلية}\index{معادلة تفاضلية} هي \omterm{def:g10:algebra:equation}{معادلة}
مجهولها \omterm{def:g11:func:function}{دالة} $y$، وتتضمن $y$ ومشتقاتها. و
\emph{الحل} على \omterm{def:g10:numbers:interval}{فترة} $I$ هو \omterm{def:g11:func:function}{دالة} قابلة للاشتقاق تحقق
\omterm{def:g10:algebra:equation}{المعادلة} عند كل نقطة من $I$.
\end{definition}

\begin{theorem}[حلول $y' = ay$]\label{thm:g12:diffeq:homogeneous}
ليكن $a \in \R$. حلول \omterm{def:g10:algebra:equation}{المعادلة} $y' = ay$ على $\R$ هي بالضبط
الدوال
\[
y(x) = C\,\eu^{ax}, \qquad C \in \R .
\]
ومن أجل كل زوج $(x_0, y_0)$ يوجد حل وحيد حيث
$y(x_0) = y_0$.
\end{theorem}

\begin{proof}
كل \omterm{def:g11:func:function}{دالة} $y = C\eu^{ax}$ تحقق $y' = Ca\,\eu^{ax} = ay$. وبالعكس، لتكن $y$
أي حل ولنضع $z(x) = y(x)\,\eu^{-ax}$. عندئذٍ
\[
z'(x) = y'(x)\,\eu^{-ax} - a\,y(x)\,\eu^{-ax}
= \bigl(y'(x) - a y(x)\bigr)\eu^{-ax} = 0,
\]
إذن $z$ ثابتة، ولتكن $C$، ويكون $y(x) = C\eu^{ax}$. والشرط الابتدائي
$y(x_0) = y_0$ يفرض $C = y_0 \eu^{-a x_0}$، على نحو وحيد.
\end{proof}

\begin{remark}
وهكذا \emph{تتميز} \omterm{thm:g12:exp:existence}{الدالة الأسية} بأبسط المعادلات
التفاضلية على الإطلاق: نمو يتناسب مع الحجم. ولهذا تظهر
في كل مكان من الطبيعة.
\end{remark}

\begin{example}[التفكك الإشعاعي]\label{ex:g12:diffeq:decay}
تحقق كمية مشعّة العلاقة $N' = -\lambda N$ حيث $\lambda > 0$، إذن
$N(t) = N_0\,\eu^{-\lambda t}$؛ وانظر \cref{ex:g12:exp:halflife} من أجل
نصف العمر.
\end{example}

\section{المعادلة \texorpdfstring{$y' = ay + b$}{y' = ay + b}}

\begin{theorem}[حلول $y' = ay + b$]\label{thm:g12:diffeq:affine}
ليكن $a \neq 0$ و $b \in \R$. حلول \omterm{def:g10:algebra:equation}{المعادلة} $y' = ay + b$ على $\R$ هي
بالضبط الدوال
\[
y(x) = C\,\eu^{ax} - \frac{b}{a}, \qquad C \in \R .
\]
\omterm{def:g11:func:function}{والدالة} الثابتة $y_p = -\frac ba$ هي \emph{الحل التوازني}.
ومن أجل كل $(x_0, y_0)$ يوجد حل وحيد حيث $y(x_0) = y_0$؛ وإذا كان
$a < 0$، آل كل حل إلى التوازن $-\frac ba$ لما
$x \to +\infty$.
\end{theorem}

\begin{proof}
المقدار الثابت $y_p = -\frac{b}{a}$ يحقق $y_p' = 0 = a y_p + b$. والآن تكون $y$
حلًا إذا وفقط إذا كان
\[
(y - y_p)' = y' = ay + b = a(y - y_p) + \underbrace{a y_p + b}_{=\,0}
= a (y - y_p),
\]
أي إذا وفقط إذا كانت $z = y - y_p$ حلًا \omterm{def:g10:algebra:equation}{للمعادلة} $z' = az$. وحسب
\cref{thm:g12:diffeq:homogeneous}، $z = C\eu^{ax}$، ومنه الصيغة
والوجود والوحدانية. وإذا كان $a < 0$، فإن $\eu^{ax} \to 0$ لما
$x \to +\infty$، إذن $y(x) \to -\frac ba$.
\end{proof}

\begin{method}[حل $y' = ay + b$ مع شرط ابتدائي]\label{met:g12:diffeq:solve}
\begin{enumerate}
  \item أوجد الحل التوازني $y_p = -\frac{b}{a}$ (بحل $y' = 0$).
  \item اكتب الحل العام $y = C\eu^{ax} + y_p$.
  \item حدّد $C$ من الشرط الابتدائي.
  \item تحقق من السلوك بعيد \omterm{def:g10:stats:quartiles}{المدى} في مواجهة الحدس الفيزيائي (فهل يتقارب
        الحل إلى التوازن؟).
\end{enumerate}
والاستراتيجية نفسها --- \emph{حل خاص + الحل العام \omterm{def:g10:algebra:equation}{للمعادلة}
المتجانسة} --- تمتد إلى $y' = ay + f(x)$: انظر
\cref{prop:g12:diffeq:general}.
\end{method}

\begin{example}[قانون نيوتن في التبرّد]\label{ex:g12:diffeq:cooling}
فنجان قهوة عند $80\,^\circ$C يقف في غرفة عند $20\,^\circ$C. وينص قانون نيوتن
على أن درجة الحرارة $T$ تحقق $T' = -k(T - 20)$ من أجل $k > 0$ ما،
\emph{أي} $T' = -kT + 20k$. والتوازن هو $20$، و
$T(t) = 20 + 60\,\eu^{-kt}$: فتبرد القهوة بسرعة أسية إلى درجة حرارة
الغرفة.
\end{example}

\section{المعادلة \texorpdfstring{$y' = ay + f(x)$}{y' = ay + f(x)}}

\begin{proposition}[بنية مجموعة الحلول]\label{prop:g12:diffeq:general}
لتكن $f$ \omterm{def:g12:limcont:continuity}{متصلة} على \omterm{def:g10:numbers:interval}{فترة} $I$ وليكن $y_p$ حلًا خاصًا واحدًا
\omterm{def:g10:algebra:equation}{للمعادلة}
\[
y' = ay + f(x)
\]
على $I$. عندئذٍ تكون الحلول على $I$ هي بالضبط الدوال
$y = C\eu^{ax} + y_p$، حيث $C \in \R$.
\end{proposition}

\begin{proof}
كما في \cref{thm:g12:diffeq:affine}: تكون $y$ حلًا إذا وفقط إذا كانت
$z = y - y_p$ تحقق
$z' = (ay + f) - (ay_p + f) = az$، أي إذا وفقط إذا كانت $z = C\eu^{ax}$.
\end{proof}

\begin{method}[تخمين حل خاص]\label{met:g12:diffeq:particular}
ابحث عن حل خاص \emph{من شكل $f$ نفسه}:
\begin{itemize}
  \item إذا كانت $f$ كثير حدود درجته $n$: فجرّب كثير حدود درجته $n$؛
  \item $f(x) = \alpha\,\eu^{kx}$ حيث $k \neq a$: فجرّب $y_p = \beta\,\eu^{kx}$؛
  \item $f(x) = \alpha\,\eu^{ax}$ (حالة الرنين): فجرّب
        $y_p = \beta x\,\eu^{ax}$.
\end{itemize}
ثم عوّض في \omterm{def:g10:algebra:equation}{المعادلة} وطابق المعاملات.
\end{method}

\begin{example}\label{ex:g12:diffeq:particular}
حل $y' = 2y + 4x$. جرّب $y_p = \alpha x + \beta$:
فإن $\alpha = 2(\alpha x + \beta) + 4x$ من أجل كل $x$ يفرض
$2\alpha + 4 = 0$ و $\alpha = 2\beta$، إذن $\alpha = -2$ و $\beta = -1$.
والحل العام: $y = C\eu^{2x} - 2x - 1$.
\end{example}

(أ) $y' = 3y$ حيث $y(0) = 2$؛\quad

\begin{exercise}[$\star$]\label{exo:g12:diffeq:3}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:4}
حل $y' = -2y + \eu^{x}$ حيث $y(0) = 1$.

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:5}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:6}
خزان يحوي $100$ L من الماء النقي. ويدخله محلول ملحي فيه $0.2$ kg من الملح في
كل لتر بمعدل $5$ L/min، ويخرج المزيج (المحفوظ متجانسًا) بالمعدل
نفسه. وليكن $m(t)$ كتلة الملح في الخزان عند اللحظة $t$ (بالدقائق).
\begin{enumerate}
  \item برّر أن $m' = 1 - \dfrac{m}{20}$.
  \item حل، وحدّد نهاية $m(t)$ لما $t \to +\infty$.

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:7}
قافزة بالمظلة كتلتها $80$ kg تسقط تحت تأثير الثقالة
($g = 9.8\ \text{m/s}^2$) ومقاومة الهواء المتناسبة مع السرعة، بحيث
تحقق سرعتها $v' = g - \frac{k}{m}v$ حيث $k = 16$ kg/s.
\begin{enumerate}
  \item حل \omterm{def:g10:algebra:equation}{المعادلة} حيث $v(0) = 0$.
  \item احسب \emph{السرعة الحدية} $\lim_{t\to+\infty} v(t)$، ثم
        الزمن اللازم لبلوغ $95\%$ منها.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:diffeq:8}
\emph{(\omterm{def:g10:algebra:equation}{المعادلة} اللوجستية.)} تحقق مجموعة سكانية $y(t) \in \intoo{0}{1}$ (منسوبةً
إلى التعداد الأقصى) العلاقة
\[
y' = y(1 - y).
\]
\begin{enumerate}
  \item ليكن $z = \dfrac{1}{y}$. بيّن أن $z$ تحقق \omterm{def:g10:algebra:equation}{المعادلة} الخطية
        $z' = -z + 1$.
  \item حل بالنسبة إلى $z$، ثم بالنسبة إلى $y$، حيث $y(0) = \frac{1}{10}$.
  \item بيّن أن $y(t) \to 1$ لما $t \to +\infty$ وارسم تقريبيًا شكل
        \omterm{def:g10:functions:graph}{منحنى} الحل.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:diffeq:9}
لتكن $y$ حلًا \omterm{def:g10:algebra:equation}{للمعادلة} $y' = ay + b$ وليكن $u_n = y(n)$ من أجل $n \in \N$.
بيّن أن $(u_n)$ تحقق علاقة تراجعية \omterm{def:g12:seq:arith-geom}{حسابية} \omterm{def:g12:seq:arith-geom}{هندسية}
$u_{n+1} = q u_n + r$، وعبّر عن $q$ و $r$ بدلالة $a$ و $b$. وما الذي
يوافقه الشرط $\abs{q} < 1$ بالنسبة إلى \omterm{def:g12:diffeq:ode}{المعادلة
التفاضلية}؟
\end{exercise}

\section{مسألة: الساعة التي داخل الأشياء}

\begin{problem}[{مسألة نهاية الأسبوع --- الكربون 14 يؤرّخ الكهوف،
وقانون نيوتن في التبرّد يوقّت جريمة، ومعادلة صغيرة واحدة ترتدي
أربعة أثواب}]
\label{pb:g12:diffeq:1}
\omterm{def:g12:diffeq:ode}{المعادلة التفاضلية} قانون تغيّر؛ وحلّها يحوّل
القانون إلى \emph{ساعة}. \omterm{def:g10:algebra:equation}{والمعادلة} الصغيرة نفسها
$y' = ay + b$ (\cref{thm:g12:diffeq:affine}) تدقّ داخل
الفحم ما قبل التاريخي، والقهوة المتبرّدة، وقطرات المطر الساقطة، و
المحاليل الوريدية في المستشفى --- وقراءة تلك الساعات هي شغل هذه
المسألة: فهي تؤرّخ رسوم لاسكو، وتحدّد ساعة وفاة،
وتدقّق فرضياتها كما يفعل عالِم جيد.

\medskip
\noindent\textbf{الجزء الأول --- التمكّن.}
\begin{enumerate}
  \item حل $y' = 3y$ حيث $y(0) = 2$؛ ثم
        $y' = -2y + 6$ حيث $y(0) = 0$
        (\cref{met:g12:diffeq:solve}).
  \item أعد استخراج حيلة الوحدانية وراء
        \cref{thm:g12:diffeq:homogeneous}: إذا كان $y' = ay$، فاحسب
        \omterm{def:g12:deriv:derivative}{مشتقة} $y(t)\,\eu^{-at}$ واستنتج أن
        \emph{كل} حل هو $C\eu^{at}$.
  \item من أجل $y' = -2y + 6$: أوجد الحل التوازني، ثم
        صِف مصير كل حل آخر لما
        $t \to +\infty$. (وهي النقط الثابتة لعلاقات
        \cref{pb:g11:seq:1} التراجعية، وقد صارت \omterm{def:g12:limcont:continuity}{متصلة}.)
  \item من أجل قانون التفكك $y' = -\frac{y}{\tau}$: بيّن أن
        نصف العمر هو $t_{1/2} = \tau \ln 2$.
  \item ارسم تقريبيًا (أو صِف) عائلة حلول
        $y' = -2y + 6$: فماذا تفعل الحلول التي تبدأ فوق $3$؟
        وتحت $3$؟ وعند $3$؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الكربون 14.}
تحافظ الأنسجة الحية على نسبة ثابتة من الكربون 14
المشعّ؛ وعند الموت يتوقف الاستهلاك ويتفكك المخزون:
$N' = -\lambda N$، ونصف العمر $5\,730$ سنة.
\begin{enumerate}[resume]
  \item احسب $\lambda$ (في السنة).
  \item عظمة تحتفظ بنسبة $20\,\%$ من نسبة الحي: فكم عمرها؟
  \item يحتفظ فحم من كهوف لاسكو المرسومة
        بنحو $15\,\%$: فأرّخ الرسوم.
  \item رجل الجليد أوتزي، الذي وُجد في نهر جليدي في الألب، بلغ قياسه
        نحو $53\,\%$: فأرّخه (ويقول علماء الآثار نحو
        $5\,300$ سنة --- فكيف كان أداؤك؟).
  \item لماذا لا يستطيع الكربون 14 تأريخ الديناصورات؟ احسب
        الجزء الباقي بعد عشرة أنصاف أعمار، وصُغ
        الجزء بعد $65$ مليون سنة على صورة قوة للعدد $2$،
        ثم اختم.
  \item حدود الخطأ: إذا لم تكن نسبة أوتزي $53\,\%$ معلومة إلا بدقة
        $\pm 1\,\%$، فاحسب مجال العمر. فأي دقة في المخبر
        تشتري أي دقة في التأريخ؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- تبرّد نيوتن، وجريمة.}
جسم درجة حرارته $T$ في محيط درجة حرارته ثابتة $T_a$
يبرد وفق $T' = -k\,(T - T_a)$.
\begin{enumerate}[resume]
  \item حل \omterm{def:g10:algebra:equation}{المعادلة} (بالتعويض $z = T - T_a$):
        \[
        T(t) = T_a + (T_0 - T_a)\,\eu^{-kt} .
        \]
  \item قهوة تُصبّ عند $90\,^\circ$C في غرفة عند $20\,^\circ$C
        تقيس $70\,^\circ$C بعد $5$ دقائق. حدّد $k$،
        ثم زمن الانتظار حتى تصير قابلة للشرب عند
        $55\,^\circ$C.
  \item مسألة الحليب: لشرب أدفأ قهوة ممكنة
        خلال عشر دقائق، أينبغي وضع الحليب البارد الآن أم في
        اللحظة الأخيرة؟ أجب \omterm{def:g10:algebra:equation}{بالمعادلة} (فماذا تفعل
        إضافة الحليب بالفجوة $T - T_a$، وكيف تقود
        الفجوة الفقد؟).
  \item الطب الشرعي: يُعثر على جثة عند \omterm{prop:g10:coordgeom:midpoint}{منتصف} الليل عند
        $30\,^\circ$C في غرفة عند $20\,^\circ$C؛ وبعد ساعة
        تقيس $28\,^\circ$C. وبافتراض $37\,^\circ$C عند
        لحظة الوفاة، استعد $k$ من القياسين،
        ثم احسب ساعة الوفاة.
  \item دقّق ساعة الطبيب الشرعي: سمِّ ثلاثة انتهاكات واقعية
        لفرضيات النموذج، وسمِّ الجهة
        التي يحرف نحوها كل منها ساعة الوفاة المقدَّرة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- أربعة أثواب \omterm{def:g10:functions:graph}{ومنحنى} S.}
\begin{enumerate}[resume]
  \item \omterm{def:g10:algebra:equation}{المعادلة} اللوجستية في \cref{exo:g12:diffeq:8}،
        $y' = y(1 - y)$ حيث $y(0) = 0.1$: باستعمال تعويض
        التمرين، استخرج
        $y(t) = \dfrac{1}{1 + 9\eu^{-t}}$، ثم أوجد لحظة
        \omterm{def:g12:deriv:inflection}{نقطة الانعطاف} $y = \frac12$ --- وهو
        تاريخ عالِم الأوبئة في \cref{pb:g12:deriv:1}، وقد صار
        قابلًا للحساب.
  \item قطرة مطر تطيع $v' = 10 - \frac v2$ (الثقالة ناقص
        قوة السحب)، حيث $v(0) = 0$. أوجد السرعة الحدية، ثم حل
        لإيجاد $v(t)$، ثم احسب متى تبلغ القطرة
        $95\,\%$ من السرعة الحدية.
  \item محلول وريدي في مستشفى يسلّم دواءً بمعدل ثابت بينما
        يطرحه الجسم تناسبيًا:
        $c' = 4 - 0.5c$ حيث $c(0) = 0$. أوجد التركيز
        المستقر والزمن اللازم لبلوغ نصفه. وصُغ
        في جملة واحدة التوازي مع علاقات القرض التراجعية في
        \cref{exo:g12:diffeq:9}.
  \item الخاتمة --- \omterm{def:g10:algebra:equation}{معادلة} واحدة، وأربعة أثواب: التفكك، والتبرّد،
        والسقوط، والجرعات، وكلها $y' = ay + b$ بإشارات
        وأسماء مختلفة. اسرد حلقة النمذجة التي شغّلتها هذه المسألة
        أربع مرات (قانون $\to$ \omterm{def:g10:algebra:equation}{معادلة} $\to$ حل $\to$
        معايرة $\to$ تنبؤ $\to$ تدقيق)، وسمِّ
        الظاهرة التي تحتاج إلى $y''$ --- وأي مسألة
        لقيتها من قبل.
\end{enumerate}
\end{problem}
