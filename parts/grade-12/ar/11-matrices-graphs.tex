\chapter{المصفوفات والبيانات}\label{ch:g12:matrix}

\omterm{def:g12:matrix:matrix}{المصفوفة} جدول مستطيلي من الأعداد، يُجمع ويُضرب بقواعد
مصمَّمة بحيث يمثّل جبر المصفوفات تركيب التحويلات
الخطية. وتحل المصفوفات الجمل الخطية، وتقود المتتاليات التراجعية
المقترنة، وتعدّ المسالك في الشبكات --- وهي الرياضيات وراء
محركات البحث وخوارزميات أقصر مسار.

\section{جبر المصفوفات}

\begin{definition}[المصفوفة]\label{def:g12:matrix:matrix}
\emph{المصفوفة ذات القياس $m \times n$}\index{مصفوفة} جدول من \omterm{def:g10:numbers:sets}{الأعداد الحقيقية} فيه
$m$ سطرًا و $n$ عمودًا:
$A = (a_{ij})$، حيث $a_{ij}$ هو المدخل في السطر $i$ والعمود $j$.
وتُجمع مصفوفتان لهما القياس نفسه مدخلًا مدخلًا، و
$\lambda A = (\lambda a_{ij})$.
\end{definition}

\begin{definition}[جداء المصفوفات]\label{def:g12:matrix:product}
لتكن $A$ ذات القياس $m \times n$ ولتكن $B$ ذات القياس $n \times p$. الجداء $AB$ هو
\omterm{def:g12:matrix:matrix}{المصفوفة} ذات القياس $m \times p$ التي مدخلها $(i,j)$ هو
\[
(AB)_{ij} = \sum_{k=1}^{n} a_{ik} b_{kj}
\]
(وهي قاعدة ``السطر $i$ من $A$ مضروبًا في العمود $j$ من $B$'').
\end{definition}

\begin{example}
$\begin{pmatrix} 1 & 2\\ 3 & 4\end{pmatrix}
\begin{pmatrix} 0 & 1\\ 1 & 1\end{pmatrix}
= \begin{pmatrix} 2 & 3\\ 4 & 7\end{pmatrix}$،
\quad بينما \quad
$\begin{pmatrix} 0 & 1\\ 1 & 1\end{pmatrix}
\begin{pmatrix} 1 & 2\\ 3 & 4\end{pmatrix}
= \begin{pmatrix} 3 & 4\\ 4 & 6\end{pmatrix}$:
\emph{فضرب المصفوفات ليس تبديليًا}.
\end{example}

\begin{proposition}[قواعد جبر المصفوفات]\label{prop:g12:matrix:rules}
كلما جعلت القياسات الجداءات ذات معنى:
\[
(AB)C = A(BC), \qquad A(B + C) = AB + AC, \qquad (A+B)C = AC + BC,
\]
و\emph{\omterm{def:g12:matrix:matrix}{مصفوفة} الوحدة} $I_n$ (وفيها الواحدات على القطر والأصفار في ما عداه)
تحقق $I_m A = A I_n = A$ من أجل $A$ ذات القياس $m \times n$.
\end{proposition}

\begin{proof}
كلها تحققات مدخلًا مدخلًا انطلاقًا من
\cref{def:g12:matrix:product}؛ وأما التجميعية، وهي الوحيدة غير التافهة،
فترجع إلى تبادل مجموعين منتهيين:
$\bigl((AB)C\bigr)_{ij} = \sum_l \left(\sum_k a_{ik}b_{kl}\right) c_{lj}
= \sum_k a_{ik} \left(\sum_l b_{kl} c_{lj}\right)
= \bigl(A(BC)\bigr)_{ij}$.
\end{proof}

\begin{definition}[المقلوب]\label{def:g12:matrix:inverse}
تكون \omterm{def:g12:matrix:matrix}{مصفوفة} مربعة $A$ من القياس $n$ \emph{قابلة للقلب}\index{مصفوفة!مقلوب}
إذا وُجدت \omterm{def:g12:matrix:matrix}{مصفوفة} $B$ حيث $AB = BA = I_n$؛ وتكون $B$ عندئذٍ وحيدة، وتُكتب
$A^{-1}$.
\end{definition}

\begin{proposition}[مقلوب مصفوفة $2\times2$]\label{prop:g12:matrix:inverse2x2}
لتكن $A = \begin{pmatrix} a & b\\ c & d\end{pmatrix}$ وليكن
$\det A = ad - bc$ (وهو \emph{المحدد}\index{محدد}). عندئذٍ تكون $A$
\omterm{def:g12:matrix:inverse}{قابلة للقلب} إذا وفقط إذا كان $\det A \neq 0$، ويكون عندئذٍ
\[
A^{-1} = \frac{1}{ad - bc}\begin{pmatrix} d & -b\\ -c & a\end{pmatrix}.
\]
\end{proposition}

\begin{proof}
يعطي حساب مباشر
$A \begin{pmatrix} d & -b\\ -c & a\end{pmatrix}
= \begin{pmatrix} d & -b\\ -c & a\end{pmatrix} A = (ad - bc) I_2$؛ فإذا كان
$ad - bc \neq 0$، فاقسم. وبالعكس، إذا كان $ad - bc = 0$، كان عمودا $A$
متناسبين، وكذلك عمودا $AB$ من أجل أي $B$؛ لكن
عمودَي $I_2$ ليسا متناسبين، إذن لا يمكن لأي $B$ أن تحقق $AB = I_2$.
\end{proof}

\begin{method}[الجمل الخطية]\label{met:g12:matrix:systems}
الجملة
$\begin{cases} ax + by = e\\ cx + dy = f \end{cases}$
هي \omterm{def:g10:algebra:equation}{المعادلة} المصفوفية $AX = Y$ حيث
$X = \begin{pmatrix} x \\ y\end{pmatrix}$
و $Y = \begin{pmatrix} e \\ f\end{pmatrix}$. وإذا كان $\det A \neq 0$، فحلها
الوحيد هو $X = A^{-1}Y$. ويعالج الشكلانية نفسها $n$ \omterm{def:g10:algebra:equation}{معادلة} ذات $n$
مجهولًا.
\end{method}

\section{قوى المصفوفات والمتتاليات التراجعية}

\begin{definition}\label{def:g12:matrix:power}
من أجل \omterm{def:g12:matrix:matrix}{مصفوفة} مربعة $A$ و $k \in \N$، $A^k = A \times \dots \times A$
($k$ عاملًا)، مع $A^0 = I$.
\end{definition}

\begin{method}[الحالتان القطرية-زائد-عديمة القوة والقابلة للقطرنة]\label{met:g12:matrix:powers}
طريقتان معياريتان لحساب $A^k$:
\begin{itemize}
  \item إذا كان $A = \lambda I + N$ حيث $N^2 = 0$، فإن مبرهنة ثنائي الحد
        (وهي صالحة هنا لأن $I$ و $N$ متبادلتان) تنطوي إلى حدين:
        $A^k = \lambda^k I + k \lambda^{k-1} N$.
  \item وإذا وُجدت $P$ \omterm{def:g12:matrix:inverse}{قابلة للقلب} حيث $A = PDP^{-1}$ و $D$ قطرية،
        فإن $A^k = P D^k P^{-1}$، وتُحسب $D^k$ مدخلًا مدخلًا.
        (وإيجاد $P$ كهذه بصورة منهجية هو نظرية
        \emph{القطرنة}، المطوَّرة في الجامعة؛ وفي هذا المستوى تُعطى $P$.)
\end{itemize}
\end{method}

\begin{example}[متتاليتان مقترنتان]\label{ex:g12:matrix:coupled}
لتكن $u_{n+1} = 3u_n + v_n$ و $v_{n+1} = u_n + 3v_n$. وبوضع
$X_n = \begin{pmatrix} u_n\\ v_n \end{pmatrix}$ و
$A = \begin{pmatrix} 3 & 1\\ 1 & 3\end{pmatrix}$، نجد
$X_{n+1} = AX_n$، إذن $X_n = A^n X_0$. والمتتاليتان المساعدتان
$s_n = u_n + v_n$ و $d_n = u_n - v_n$ تحققان $s_{n+1} = 4s_n$ و
$d_{n+1} = 2d_n$، إذن $s_n = 4^n s_0$ و $d_n = 2^n d_0$ و
\[
u_n = \frac{4^n(u_0+v_0) + 2^n(u_0-v_0)}{2}, \qquad
v_n = \frac{4^n(u_0+v_0) - 2^n(u_0-v_0)}{2}.
\]
(ووراء الكواليس: $(1,1)$ و $(1,-1)$ اتجاها متجهتين ذاتيتين \omterm{def:g12:matrix:matrix}{للمصفوفة} $A$.)
\end{example}

\section{البيانات والمسالك}

\begin{definition}[البيان ومصفوفة الجوار]\label{def:g12:matrix:graph}
يتكوّن \emph{البيان}\index{بيان} من رؤوس $1, 2, \dots, n$ وحواف
تصل أزواجًا معينة من الرؤوس (أزواجًا مرتبة في حالة البيان
\emph{الموجَّه}). و\emph{مصفوفة جواره}\index{مصفوفة الجوار} هي
\omterm{def:g12:matrix:matrix}{المصفوفة} $M$ ذات القياس $n \times n$ حيث $m_{ij} = 1$ إذا وُجدت حافة من $i$ إلى
$j$، و $0$ فيما عدا ذلك. و\emph{المسلك} ذو الطول $k$ من $i$ إلى $j$ هو
\omterm{def:g12:seq:sequence}{متتالية} من $k$ حافة متعاقبة تقود من $i$ إلى $j$.
\end{definition}

\begin{theorem}[عدّ المسالك]\label{thm:g12:matrix:walks}
عدد المسالك ذات الطول $k$ من الرأس $i$ إلى الرأس $j$ هو
المدخل $(i,j)$ في $M^k$.
\end{theorem}

\begin{proof}
بالتراجع على $k$. ومن أجل $k = 1$ هذا هو تعريف $M$. ولنفترض
الادعاء من أجل $k$. فالمسلك ذو الطول $k+1$ من $i$ إلى $j$ هو مسلك طوله
$k$ من $i$ إلى رأس ما $l$، متبوعًا بحافة من $l$ إلى $j$؛ وبمبدأي
الجمع والضرب، يكون عددها
\[
\sum_{l=1}^{n} \bigl(M^k\bigr)_{il}\, m_{lj} = \bigl(M^{k+1}\bigr)_{ij}.
\qedhere
\]
\end{proof}

\begin{example}
من أجل \omterm{def:g12:matrix:graph}{بيان} المثلث ($3$ رؤوس، وكل الأزواج موصولة)،
$M = \begin{pmatrix} 0&1&1\\ 1&0&1\\ 1&1&0\end{pmatrix}$ و
$M^2 = \begin{pmatrix} 2&1&1\\ 1&2&1\\ 1&1&2\end{pmatrix}$: فمن كل رأس
يوجد $2$ من المسالك ذات الطول $2$ عائدة إليه (عبر أي من الجارين) و
$1$ إلى كل رأس آخر.
\end{example}

\begin{exercise}[$\star\star$]\label{exo:g12:matrix:4}
لتكن $A = \begin{pmatrix} 2 & 1\\ 0 & 2\end{pmatrix} = 2I + N$ حيث
$N = \begin{pmatrix} 0 & 1\\ 0 & 0 \end{pmatrix}$.
\begin{enumerate}
  \item تحقق من أن $N^2 = 0$ ومن أن $I$ و $N$ متبادلتان.
  \item استنتج $A^k$ من أجل كل $k \in \N$ وتحقق من الصيغة من أجل $k=2$
        بحساب مباشر.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:matrix:5}
لتكن $A = \begin{pmatrix} 0 & 1\\ 1 & 1\end{pmatrix}$ ولتكن $F_n$ \omterm{def:g12:seq:sequence}{متتالية}
فيبوناتشي ($F_0 = 0$ و $F_1 = 1$ و $F_{n+2} = F_{n+1} + F_n$). بيّن بالتراجع
أنه من أجل $n \geq 1$،
\[
A^n = \begin{pmatrix} F_{n-1} & F_n\\ F_n & F_{n+1}\end{pmatrix},
\]
واستنتج المتطابقة $F_{n+1}F_{n-1} - F_n^2 = (-1)^n$.
(إرشاد: المحددات تُضرب: $\det(MN) = \det M \det N$، ويمكنك
التحقق من ذلك من أجل المصفوفات $2\times2$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:matrix:6}
\omterm{def:g12:matrix:graph}{بيان} موجَّه على الرؤوس $\{1, 2, 3\}$ له الحواف

\begin{enumerate}
  \item اكتب \omterm{def:g12:matrix:graph}{مصفوفة الجوار} $M$ واحسب $M^2$ و $M^3$.
  \item كم مسلكًا طوله $3$ يذهب من $1$ إلى $1$؟ اسردها.
\end{enumerate}
\end{exercise}

\section{مسألة: المصفوفة التي تعرف فيبوناتشي (و
الطقس)}

\begin{problem}[{مسألة نهاية الأسبوع --- مصفوفة $2 \times 2$ واحدة
تحمل فيبوناتشي كلها، ومصفوفة ماركوف تتنبأ بطقس
المدى الطويل، ومتجهة ذاتية تساوي مليار دولار}]
\label{pb:g12:matrix:1}
\omterm{def:g12:matrix:matrix}{المصفوفة} آلة تأكل حالة وتعيد الحالة التالية
--- ومنه فإن \emph{قواها} تحمل مستقبلات كاملة. وتفتتح هذه
المسألة \omterm{def:g12:matrix:matrix}{بالمصفوفة} المدهشة التي تسرد قواها
أعداد فيبوناتشي (وتبرهن على متطابقاتها في سطر واحد لكل منها)،
ثم تشغّل الطقس سلسلةَ ماركوف حتى حالته المستقرة، و
تختم \omterm{def:g11:vect:vector}{بالمتجهة} الذاتية التي بُني عليها محرك بحث
(\cref{thm:g12:matrix:walks} و
\cref{met:g12:matrix:powers}).

\medskip
\noindent\textbf{الجزء الأول --- التمكّن.}
\begin{enumerate}
  \item مع $A = \begin{pmatrix} 1 & 2\\ 3 & 4\end{pmatrix}$
        و $B = \begin{pmatrix} 0 & 1\\ 1 & 0\end{pmatrix}$:
        احسب $AB$ و $BA$. فما الحكم على التبديلية؟
  \item اقلب $\begin{pmatrix} 2 & 1\\ 5 & 3\end{pmatrix}$
        (\cref{prop:g12:matrix:inverse2x2}) واستعمل المقلوب
        لحل $2x + y = 4$ و $5x + 3y = 7$.
  \item لتكن $N = \begin{pmatrix} 0 & 1\\ 0 & 0\end{pmatrix}$:
        احسب $N^2$، واستنتج
        $(I + N)^n = I + nN$ من أجل كل $n$.
  \item \omterm{def:g12:matrix:graph}{بيان} المثلث (ثلاثة رؤوس، وكل الأزواج موصولة):
        اكتب \omterm{def:g12:matrix:graph}{مصفوفة جواره} $A$، واحسب $A^3$، وفسّر
        مدخلات القطر
        (\cref{thm:g12:matrix:walks}).
  \item من أجل $D = \begin{pmatrix} 2 & 0\\ 0 & \frac12
        \end{pmatrix}$: أعطِ $D^n$ وسلوكها لما
        $n \to \infty$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- \omterm{def:g12:matrix:matrix}{مصفوفة} فيبوناتشي.}
لتكن $F = \begin{pmatrix} 1 & 1\\ 1 & 0\end{pmatrix}$ ولتكن
$F_1 = F_2 = 1, F_3 = 2, \dots$ أعداد فيبوناتشي في
\cref{pb:g11:seq:1}.
\begin{enumerate}[resume]
  \item احسب $F^2$ و $F^3$ و $F^4$ وخمّن الصورة العامة
        \omterm{def:g12:matrix:matrix}{للمصفوفة} $F^n$ بدلالة أعداد فيبوناتشي.
  \item برهن على التخمين
        $F^n = \begin{pmatrix} F_{n+1} & F_n\\ F_n & F_{n-1}
        \end{pmatrix}$ بالتراجع.
  \item خذ محددي الطرفين (\omterm{def:g11:vect:det}{فمحدد} الجداء هو
        جداء المحددين --- وتحقق من ذلك
        من أجل المصفوفات $2 \times 2$ إن لم تره من قبل):
        واستنتج \emph{متطابقة كاسيني}
        $F_{n+1}F_{n-1} - F_n^2 = (-1)^n$ ---
        وهي محرّك المربع المتلاشي، مبرهَنًا عليه في سطر واحد.
  \item انطلاقًا من $F^{m+n} = F^m F^n$، اقرأ المدخلين
        في أعلى اليمين واستخرج \emph{صيغة الجمع}
        \[
        F_{m+n} = F_{m+1} F_n + F_m F_{n-1} .
        \]
        وتحقق منها من أجل $m = n = 3$.
  \item استنتج من صيغة الجمع (بالتراجع على $k$)
        أن $F_n$ \omterm{def:g12:arith:divides}{يقسم} $F_{kn}$، وتحقق من ذلك على
        $F_3 \mid F_6$ و $F_3 \mid F_9$.
  \item لحساب $F_{100}$، لا يلزم ضرب $100$
        \omterm{def:g12:matrix:matrix}{مصفوفة}: فربّع تكرارًا ($F^2, F^4, F^8, \dots$)
        ثم ركّب. فكم ضربًا مصفوفيًا يكفي،
        وأي حيلة ضرب قديمة من الكتاب السابق
        هي هذه، وقد رُقّيت إلى المصفوفات؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- آلة الطقس.}
في مدينة ما: بعد يوم مشمس، يكون التالي مشمسًا \omterm{def:g10:proba:distribution}{باحتمال}
$0.8$؛ وبعد يوم ممطر، يكون مشمسًا \omterm{def:g10:proba:distribution}{باحتمال}
$0.4$. رمّز \omterm{def:g11:prob:rv}{توزيع} اليوم بعمود
$\binom{p_{\text{شمس}}}{p_{\text{مطر}}}$ والتطور بالعلاقة
\[
M = \begin{pmatrix} 0.8 & 0.4\\ 0.2 & 0.6 \end{pmatrix}.
\]
\begin{enumerate}[resume]
  \item تحقق من أن مجموع كل عمود من $M$ يساوي $1$، وقل لماذا
        يجب أن تملك أي آلة طقس هذه الخاصية.
  \item اليوم مشمس. احسب التنبؤ من أجل الغد و
        من أجل بعد الغد.
  \item أوجد \emph{الحالة المستقرة}: أي \omterm{def:g11:prob:rv}{التوزيع} $v$ الذي يحقق
        $Mv = v$ (ومجموع مدخلاته $1$). فما نسبة
        الأيام المشمسة على \omterm{def:g10:stats:quartiles}{المدى} الطويل؟
  \item انطلق من يوم ممطر، $\binom01$، وطبّق $M$ أربع
        مرات، متتبعًا المسافة إلى الحالة المستقرة عند
        كل خطوة. فبأي معامل تنكمش الفجوة في كل خطوة ---
        وأي نوع من التقارب هذا؟
  \item \omterm{def:g10:coordgeom:system}{ترتيب} الصفحات مصغَّرًا: ثلاث صفحات، بالوصلات
        $A \to B$ و $A \to C$ و $B \to C$ و $C \to A$. ويتبع متصفح
        عشوائي وصلة خارجة مختارة عشوائيًا بانتظام.
        اكتب \omterm{def:g12:matrix:matrix}{مصفوفة} الانتقال، وأوجد الحالة المستقرة، ثم
        رتّب الصفحات.
  \item فسّر \omterm{def:g10:coordgeom:system}{الترتيب}: لماذا تنال $C$ درجة عالية مثل
        $A$ مع أنها تتلقى وصلات من صفحات أقل --- فماذا
        تقيس الحالة المستقرة فعلًا؟ (ويضيف \omterm{def:g10:coordgeom:system}{ترتيب} الصفحات الحقيقي
        معامل تخميد من أجل الطرق المسدودة والقفزات؛ أما فكرة
        \omterm{def:g11:vect:vector}{المتجهة} الذاتية فهي هذه بالضبط.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- أرباح القطرنة.}
\begin{enumerate}[resume]
  \item مقداران مقترنان يطيعان
        $u_{n+1} = 3u_n + v_n$ و $v_{n+1} = u_n + 3v_n$، أي
        \omterm{def:g12:matrix:matrix}{المصفوفة} $A$ في \cref{exo:g12:matrix:8}. وباستعمال
        قطرنة ذلك التمرين
        ($D = \operatorname{diag}(4, 2)$)، أعطِ الصيغة
        المغلقة للمقدار $u_n$ عندما $u_0 = 1$ و $v_0 = 0$، وتحقق
        منها في مواجهة الحساب المباشر من أجل $n = 1, 2, 3$.
  \item في جملة أو جملتين: ماذا \emph{تفعل} القطرنة
        بجملة مقترنة --- وبأي معنى تكون الحالة المستقرة
        لسلسلة ماركوف في السؤال 14 حكاية \omterm{def:g11:vect:vector}{متجهة} ذاتية أيضًا؟
  \item الخاتمة --- أوجه \omterm{def:g12:matrix:matrix}{المصفوفة} الثلاثة في نهاية هذا الأسبوع:
        مسك الدفاتر (الجمل والمقلوبات)، والتوافيق
        (المسالك والوصلات معدودةً بالقوى)، والتطور
        (فيبوناتشي، والطقس، والشبكة --- مستقبلات تُقرأ
        من الاتجاهات الذاتية). جملة واحدة لكل وجه، مع
        الإشارة إلى الأمام: الجبر الخطي في الكتب الجامعية
        يجعل كل وجه من هذه الأوجه نظرية.
\end{enumerate}
\end{problem}
