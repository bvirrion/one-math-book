\chapter{المتتاليات}\label{ch:g12:seq}

\omterm{def:g12:seq:sequence}{المتتالية} قائمة من \omterm{def:g10:numbers:sets}{الأعداد الحقيقية} مدلَّلة \omterm{def:g10:numbers:sets}{بالأعداد الطبيعية}. وتنمذج المتتاليات
التطورات المتقطعة --- تعدادات السكان محسوبة سنة بعد سنة، وأرصدة
حساب مصرفي، والتقريبات المتعاقبة لعدد ما --- ونهاياتها هي
أول لقاء جدي مع اللانهاية. ويرسي هذا الفصل
المفردات، ومبدأ التراجع، ومبرهنات التقارب
الأساسية.

\section{الاستدلال بالتراجع}

\begin{theorem}[مبدأ التراجع]\label{thm:g12:seq:induction}
لتكن $P(n)$ عبارة تتعلق \omterm{def:g10:numbers:sets}{بعدد صحيح} $n$، وليكن $n_0 \in \N$.
إذا تحقق
\begin{enumerate}
  \item \emph{(الحالة الابتدائية)} $P(n_0)$ صحيحة، و
  \item \emph{(خطوة التراجع)} من أجل كل $n \geq n_0$، تستلزم $P(n)$ العبارة $P(n+1)$،
\end{enumerate}
فإن $P(n)$ صحيحة من أجل كل $n \geq n_0$.
\end{theorem}

\begin{proof}
لنفترض، بغرض الوصول إلى تناقض، أن مجموعة \omterm{def:g10:numbers:sets}{الأعداد الصحيحة} $A$ التي $n \geq n_0$ و
$P(n)$ خاطئة من أجلها غير خالية. عندئذٍ تملك $A$ عنصرًا أصغريًا
$m$.\footnote{كل جزء غير خالٍ من $\N$ له عنصر أصغري؛ وتُؤخذ هذه
الخاصية للمجموعة $\N$ بديهيةً.} وبما أن $P(n_0)$ صحيحة، فإن $m > n_0$، إذن
$m - 1 \geq n_0$ و $m-1 \notin A$، أي إن $P(m-1)$ صحيحة. وعندئذٍ تبيّن خطوة
التراجع مطبَّقة على $n = m-1$ أن $P(m)$ صحيحة، وهذا يناقض
$m \in A$.
\end{proof}

\begin{example}\label{ex:g12:seq:bernoulli}
لنبرهن على \emph{متراجحة برنولي}: من أجل كل \omterm{def:g10:numbers:sets}{عدد حقيقي} $a > 0$ وكل
$n \in \N$،
\[
(1+a)^n \geq 1 + na.
\]
\emph{الحالة الابتدائية.} من أجل $n = 0$، يساوي الطرفان $1$.
\emph{خطوة التراجع.} لنفترض أن $(1+a)^n \geq 1+na$ من أجل $n \in \N$ ما. وبما أن
$1 + a > 0$، فإن ضرب الطرفين في $1+a$ يحافظ على المتراجحة:
\[
(1+a)^{n+1} \geq (1+na)(1+a) = 1 + (n+1)a + na^2 \geq 1 + (n+1)a .
\]
وبالتراجع، تصح المتراجحة من أجل كل $n \in \N$.
\end{example}

\begin{method}[كتابة برهان بالتراجع]\label{met:g12:seq:induction}
اجعل العبارة $P(n)$ صريحة دائمًا قبل أن تبدأ. وللبرهان الكامل
ثلاثة أجزاء ظاهرة: الحالة الابتدائية، وخطوة التراجع (``لنفترض $P(n)$؛
ولنبرهن على $P(n+1)$'')، والخاتمة التي تستدعي مبدأ التراجع.
وأشيع الأخطاء هو أن تُبرهن خطوة التراجع دون أن تُستعمل
الفرضية $P(n)$ قط: فإن حدث ذلك، فإما أن البرهان خاطئ وإما أن التراجع
لم يكن ضروريًا.
\end{method}

\section{مفردات المتتاليات}

\begin{definition}[المتتالية]\label{def:g12:seq:sequence}
\emph{المتتالية}\index{متتالية} هي \omterm{def:g11:func:function}{دالة} $u \colon \N \to \R$ (أو من
$\{n \in \N : n \geq n_0\}$ إلى $\R$). وتُكتب صورة $n$ على الصورة $u_n$، وتُكتب
المتتالية نفسها $(u_n)_{n\in\N}$ أو ببساطة $(u_n)$.
\end{definition}

ويمكن تعريف \omterm{def:g12:seq:sequence}{متتالية} \emph{صراحةً}، بصيغة $u_n = f(n)$، أو
\emph{بالتراجع}، بحدها الأول وعلاقة $u_{n+1} = f(u_n)$.

\begin{definition}[الرتابة]\label{def:g12:seq:monotonic}
تكون \omterm{def:g12:seq:sequence}{المتتالية} $(u_n)$ \emph{متزايدة}\index{متتالية!متزايدة} إذا كان
$u_{n+1} \geq u_n$ من أجل كل $n$، و\emph{\omterm{def:g10:functions:variations}{متناقصة}} إذا كان $u_{n+1} \leq u_n$ من
أجل كل $n$، و\emph{رتيبة} إذا كانت متزايدة أو \omterm{def:g10:functions:variations}{متناقصة}. وتكون
متزايدة (أو \omterm{def:g10:functions:variations}{متناقصة}) \emph{تمامًا} عندما تكون المتراجحات
تامة.
\end{definition}

\begin{method}[دراسة رتابة متتالية]\label{met:g12:seq:monotonicity}
ثلاث تقنيات معيارية:
\begin{enumerate}
  \item ادرس إشارة $u_{n+1} - u_n$؛
  \item إذا كانت كل الحدود موجبة، فقارن $\dfrac{u_{n+1}}{u_n}$ بالعدد $1$؛
  \item إذا كان $u_n = f(n)$ مع $f$ معرَّفة على $\intco{0}{+\infty}$، فاستعمل
        تغيرات $f$.
\end{enumerate}
\end{method}

\begin{definition}[المتتالية المحدودة]\label{def:g12:seq:bounded}
تكون \omterm{def:g12:seq:sequence}{المتتالية} $(u_n)$ \emph{محدودة من الأعلى}\index{متتالية!محدودة} إذا وُجد
$M \in \R$ حيث $u_n \leq M$ من أجل كل $n$؛ و\emph{محدودة من الأسفل} إذا وُجد
$m \in \R$ حيث $u_n \geq m$ من أجل كل $n$؛ و\emph{محدودة} إذا تحقق
الشرطان معًا.
\end{definition}

\subsection{المتتاليات الحسابية والهندسية}

\begin{definition}[المتتاليات الحسابية والهندسية]\label{def:g12:seq:arith-geom}
تكون \omterm{def:g12:seq:sequence}{المتتالية} $(u_n)$ \emph{حسابية}\index{متتالية!حسابية} ذات \omterm{def:g11:seq:arithmetic}{أساس}
$r$ إذا كان $u_{n+1} = u_n + r$ من أجل كل $n$، و
\emph{هندسية}\index{متتالية!هندسية} ذات \omterm{def:g11:seq:geometric}{أساس هندسي} $q$ إذا كان
$u_{n+1} = q\,u_n$ من أجل كل $n$.
\end{definition}

\begin{proposition}[الصورة الصريحة والمجاميع]\label{prop:g12:seq:arith-geom}
ليكن $n \in \N$.
\begin{enumerate}
  \item إذا كانت $(u_n)$ \omterm{def:g12:seq:arith-geom}{حسابية} ذات \omterm{def:g11:seq:arithmetic}{أساس} $r$، فإن
        $u_n = u_0 + nr$ و
        \[
        u_0 + u_1 + \dots + u_n = (n+1)\,\frac{u_0 + u_n}{2}.
        \]
  \item إذا كانت $(u_n)$ \omterm{def:g12:seq:arith-geom}{هندسية} ذات \omterm{def:g11:seq:geometric}{أساس هندسي} $q \neq 1$، فإن
        $u_n = u_0\, q^n$ و
        \[
        u_0 + u_1 + \dots + u_n = u_0\,\frac{1 - q^{n+1}}{1 - q}.
        \]
\end{enumerate}
\end{proposition}

\begin{proof}
تنتج الصور الصريحة بتراجعات مباشرة. وأما المجموع الحسابي، فاكتب
$S = u_0 + \dots + u_n$ ثم اجمعه إلى المجموع نفسه مكتوبًا \omterm{def:g10:coordgeom:system}{بالترتيب} المعكوس:
فيساوي كل عمود من الأعمدة $n+1$ المقدار $u_0 + u_n$، إذن $2S = (n+1)(u_0+u_n)$.
وأما المجموع الهندسي، فاحسب $S - qS$: فتختصر كل الحدود أزواجًا عدا
الأول والأخير، إذن $(1-q)S = u_0(1 - q^{n+1})$.
\end{proof}

\section{نهاية متتالية}

\begin{definition}[المتتالية المتقاربة]\label{def:g12:seq:limit}
\emph{تتقارب}\index{نهاية!متتالية} \omterm{def:g12:seq:sequence}{المتتالية} $(u_n)$ إلى \omterm{def:g10:numbers:sets}{العدد الحقيقي}
$\ell$ إذا كانت كل \omterm{def:g10:numbers:interval}{فترة} مفتوحة تحوي $\ell$ تحوي كل الحدود
$u_n$ ابتداءً من دليل ما. ونكتب عندئذٍ $\lim\limits_{n\to+\infty} u_n = \ell$.

وبصيغة مكافئة: من أجل كل $\varepsilon > 0$، يوجد $N \in \N$ بحيث
يكون من أجل كل $n \geq N$، $\abs{u_n - \ell} \leq \varepsilon$.
\end{definition}

\begin{definition}[التباعد إلى ما لا نهاية]\label{def:g12:seq:infinity}
\emph{تؤول \omterm{def:g12:seq:sequence}{المتتالية} $(u_n)$ إلى $+\infty$} إذا وُجد، من أجل كل $A \in \R$،
عدد $N \in \N$ بحيث يكون $u_n \geq A$ من أجل كل $n \geq N$. ونكتب
$\lim\limits_{n\to+\infty} u_n = +\infty$؛ وتعريف
$\lim u_n = -\infty$ نظير له. ويقال عن \omterm{def:g12:seq:sequence}{متتالية} لا \omterm{def:g12:seq:limit}{تتقارب} إنها
\emph{تتباعد}.
\end{definition}

\begin{remark}
يمكن أن تتباعد \omterm{def:g12:seq:sequence}{متتالية} دون أن تؤول إلى $\pm\infty$: \omterm{def:g12:seq:sequence}{فالمتتالية}
$u_n = (-1)^n$ لا تأخذ إلا القيمتين $1$ و $-1$ وليست لها نهاية.
\end{remark}

\begin{proposition}[وحدانية النهاية]\label{prop:g12:seq:uniqueness}
إذا تقاربت $(u_n)$، فنهايتها وحيدة.
\end{proposition}

\begin{proof}
لنفترض أن $u_n \to \ell$ و $u_n \to \ell'$ مع $\ell \neq \ell'$، وليكن مثلًا
$\ell < \ell'$. ضع $\varepsilon = \frac{\ell' - \ell}{3} > 0$. فابتداءً من
دليل ما، يكون $\abs{u_n - \ell} \leq \varepsilon$ و
$\abs{u_n - \ell'} \leq \varepsilon$، ومنه
\[
\ell' - \ell \leq \abs{\ell' - u_n} + \abs{u_n - \ell}
\leq 2\varepsilon = \tfrac{2}{3}(\ell' - \ell) < \ell' - \ell,
\]
وهذا تناقض.
\end{proof}

\begin{proposition}[العمليات على النهايات]\label{prop:g12:seq:operations}
لتكن $(u_n)$ و $(v_n)$ متتاليتين نهايتاهما $\ell$ و $\ell'$ (منتهيتين أو
لانهائيتين). عندئذٍ، كلما لم يكن الطرف الأيمن صورة غير محددة،
\[
\lim (u_n + v_n) = \ell + \ell', \qquad
\lim (u_n v_n) = \ell\,\ell', \qquad
\lim \frac{u_n}{v_n} = \frac{\ell}{\ell'}.
\]
والصور غير المحددة هي
$(+\infty) + (-\infty)$ و $0 \times \infty$ و $\frac{\infty}{\infty}$ و
$\frac{0}{0}$.
\end{proposition}

\begin{proof}
نبرهن على قاعدة المجموع من أجل نهايتين منتهيتين؛ والحالات الأخرى مماثلة وتُترك
تمارين. ليكن $\varepsilon > 0$. يوجد $N_1, N_2$ بحيث
يكون $\abs{u_n - \ell} \leq \varepsilon/2$ من أجل $n \geq N_1$ و
$\abs{v_n - \ell'} \leq \varepsilon/2$ من أجل $n \geq N_2$. ومن أجل
$n \geq \max(N_1, N_2)$، تعطي متراجحة المثلث
\[
\abs{(u_n + v_n) - (\ell + \ell')}
\leq \abs{u_n - \ell} + \abs{v_n - \ell'} \leq \varepsilon. \qedhere
\]
\end{proof}

\begin{method}[رفع صورة غير محددة]\label{met:g12:seq:indeterminate}
أمام صورة غير محددة، أخرج الحد المهيمن عاملًا مشتركًا. فمثلًا
\[
n^2 - n = n^2\left(1 - \tfrac{1}{n}\right) \xrightarrow[n\to+\infty]{} +\infty,
\qquad
\frac{2n^2+1}{n^2 - n} = \frac{2 + 1/n^2}{1 - 1/n}
\xrightarrow[n\to+\infty]{} 2 .
\]
\end{method}

\section{مبرهنات التقارب}

\begin{theorem}[مبرهنتا المقارنة والحصر]\label{thm:g12:seq:squeeze}
لتكن $(u_n)$ و $(v_n)$ و $(w_n)$ متتاليات.
\begin{enumerate}
  \item إذا كان $u_n \leq v_n$ ابتداءً من دليل ما وكان $u_n \to +\infty$، فإن
        $v_n \to +\infty$.
  \item \emph{(مبرهنة الحصر)}\index{مبرهنة الحصر} إذا كان
        $u_n \leq v_n \leq w_n$ ابتداءً من دليل ما وتقاربت $(u_n)$ و $(w_n)$
        كلتاهما إلى النهاية نفسها $\ell$، فإن $(v_n)$ \omterm{def:g12:seq:limit}{تتقارب}
        إلى~$\ell$.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} ليكن $A \in \R$. بما أن $u_n \to +\infty$، يوجد $N$ حيث
$u_n \geq A$ من أجل $n \geq N$؛ وبتكبير $N$ عند اللزوم، يكون $v_n \geq u_n \geq A$
من أجل $n \geq N$.

\emph{2.} ليكن $\varepsilon > 0$. فابتداءً من دليل ما، يتحقق
$\ell - \varepsilon \leq u_n$ و $w_n \leq \ell + \varepsilon$ معًا، ومنه
$\ell - \varepsilon \leq u_n \leq v_n \leq w_n \leq \ell + \varepsilon$، أي
$\abs{v_n - \ell} \leq \varepsilon$.
\end{proof}

\begin{example}
من أجل كل $n \geq 1$، يكون $-\frac{1}{n} \leq \frac{(-1)^n}{n} \leq \frac{1}{n}$، وتؤول
الحاصرتان كلتاهما إلى $0$؛ ومنه $\frac{(-1)^n}{n} \to 0$.
\end{example}

\begin{theorem}[مبرهنة التقارب الرتيب]\label{thm:g12:seq:monotone}
كل \omterm{def:g12:seq:sequence}{متتالية} \omterm{def:g12:seq:monotonic}{متزايدة} \omterm{def:g12:seq:bounded}{ومحدودة من الأعلى} \omterm{def:g12:seq:limit}{تتقارب}. وكل \omterm{def:g12:seq:sequence}{متتالية} \omterm{def:g10:functions:variations}{متناقصة}
\omterm{def:g12:seq:bounded}{ومحدودة من الأسفل} \omterm{def:g12:seq:limit}{تتقارب}. وكل \omterm{def:g12:seq:sequence}{متتالية} \omterm{def:g12:seq:monotonic}{متزايدة} غير \omterm{def:g12:seq:bounded}{محدودة
من الأعلى} تؤول إلى $+\infty$.
\end{theorem}

\begin{proof}[برهان جزئي]
نبرهن على العبارة الثالثة. لتكن $(u_n)$ \omterm{def:g12:seq:monotonic}{متزايدة} وغير \omterm{def:g12:seq:bounded}{محدودة
من الأعلى}، وليكن $A \in \R$. بما أن $A$ ليس حدًا أعلى، يوجد $N$
حيث $u_N \geq A$؛ وبالرتابة، يكون $u_n \geq u_N \geq A$ من أجل كل $n \geq N$.
ومنه $u_n \to +\infty$.

أما عبارتا التقارب فتعتمدان على خاصية الحد الأعلى الأصغري في
$\R$؛ وهما \emph{مقبولتان في هذا المستوى} (ويُبرهن عليهما في السنة الأولى
الجامعية).
\end{proof}

\begin{remark}
تضمن المبرهنة \emph{وجود} النهاية لكنها لا تعطي
قيمتها. \omterm{def:g12:seq:sequence}{فمتتالية} \omterm{def:g12:seq:monotonic}{متزايدة} \omterm{def:g12:seq:bounded}{ومحدودة من الأعلى} بالعدد $M$ \omterm{def:g12:seq:limit}{تتقارب} إلى
$\ell \leq M$ ما، وليس بالضرورة إلى $M$.
\end{remark}

\begin{theorem}[نهاية المتتاليات الهندسية]\label{thm:g12:seq:geometric}
ليكن $q \in \R$.
\begin{enumerate}
  \item إذا كان $q > 1$، فإن $q^n \to +\infty$.
  \item إذا كان $q = 1$، فإن $q^n \to 1$.
  \item إذا كان $\abs{q} < 1$، فإن $q^n \to 0$.
  \item إذا كان $q \leq -1$، فإن $(q^n)$ تتباعد وليست لها نهاية.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} اكتب $q = 1 + a$ مع $a > 0$. فمتراجحة برنولي
(\cref{ex:g12:seq:bernoulli}) تعطي $q^n \geq 1 + na \to +\infty$، ونستنتج
بالمقارنة (\cref{thm:g12:seq:squeeze}).

\emph{2.} مباشرة.

\emph{3.} إذا كان $q = 0$ فالادعاء واضح. وإلا فإن $\abs{q} < 1$ يعطي
$1/\abs{q} > 1$، إذن $(1/\abs{q})^n \to +\infty$ حسب النقطة 1، ومنه
$\abs{q}^n \to 0$، ويسمح $-\abs{q}^n \leq q^n \leq \abs{q}^n$ بأن
نستنتج بمبرهنة الحصر.

\emph{4.} من أجل $q \leq -1$، تأخذ $(q^{2n})$ قيمًا $\geq 1$ بينما
تأخذ $(q^{2n+1})$ قيمًا $\leq -1$: فلا نهاية وحيدة تجذب
المتتاليتين الجزئيتين معًا.
\end{proof}

\begin{method}[المتتاليات التراجعية $u_{n+1} = f(u_n)$]\label{met:g12:seq:recurrent}
لدراسة \omterm{def:g12:seq:sequence}{متتالية} معرَّفة بالعلاقة $u_{n+1} = f(u_n)$:
\begin{enumerate}
  \item برهن بالتراجع على أن $(u_n)$ تبقى في \omterm{def:g10:numbers:interval}{فترة} $I$ تكون $f$
        حسنة السلوك عليها (وعلى أن $(u_n)$ \omterm{def:g12:seq:monotonic}{رتيبة}، في الغالب)؛
  \item استنتج التقارب من مبرهنة التقارب الرتيب؛
  \item انتقل إلى النهاية في العلاقة $u_{n+1} = f(u_n)$: فإذا كانت $f$
        متصلة وكان $u_n \to \ell \in I$، فإن $\ell$ يحقق
        $f(\ell) = \ell$ (انظر \cref{ch:g12:limcont})؛ فحل هذه \omterm{def:g10:algebra:equation}{المعادلة}
        واختر \omterm{def:g11:quad:discriminant}{الجذر} الصحيح.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g12:seq:sqrt2}
ليكن $u_0 = 2$ و $u_{n+1} = \frac{1}{2}\left(u_n + \frac{2}{u_n}\right)$.
ويُتحقق بالتراجع من أن $u_n \geq \sqrt{2}$ من أجل كل $n$ (فالمتراجحة
$\frac{1}{2}(x + 2/x) \geq \sqrt{2}$ من أجل $x>0$ تكافئ
$(x - \sqrt2)^2 \geq 0$)، ثم من أن $(u_n)$ \omterm{def:g10:functions:variations}{متناقصة}، إذ
\[
u_{n+1}-u_n=\frac{2-u_n^2}{2u_n}\leq 0 .
\]
وبما أنها \omterm{def:g10:functions:variations}{متناقصة} \omterm{def:g12:seq:bounded}{ومحدودة من الأسفل}، فإن $(u_n)$
\omterm{def:g12:seq:limit}{تتقارب} إلى $\ell \geq \sqrt{2}$ ما، ولا بد أن يحقق
$\ell = \frac{1}{2}(\ell + 2/\ell)$، أي $\ell^2 = 2$. ومنه
$u_n \to \sqrt{2}$. وهذه خوارزمية هيرون، وقد استعملها
البابليون من قبل؛ وتقاربها سريع جدًا ($u_3$ يعطي $\sqrt 2$
بثمانية أرقام عشرية أصلًا).
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:seq:1}
برهن بالتراجع على أنه من أجل كل $n \in \N$،
\[
1^2 + 2^2 + \dots + n^2 = \frac{n(n+1)(2n+1)}{6}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:2}
ادرس رتابة المتتاليتين المعرَّفتين من أجل $n \geq 1$ بالعلاقتين
\[
a_n = \frac{n+1}{n}, \qquad
b_n = \frac{2^n}{n}, \qquad
c_n = n^2 - 10n .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:3}
احسب نهايات المتتاليات ذات الحدود العامة
\[
u_n = \frac{3n^2 - n + 1}{2n^2 + 5}, \qquad
v_n = \sqrt{n+1} - \sqrt{n}, \qquad
w_n = \frac{2^n - 3^n}{3^n + 1}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:4}
لتكن $(u_n)$ \omterm{def:g11:seq:arithmetic}{المتتالية الحسابية} حيث $u_0 = 5$ وأساسها
$r = 3$، ولتكن $(v_n)$ \omterm{def:g11:seq:geometric}{المتتالية الهندسية} حيث $v_0 = 8$ وأساسها الهندسي
$q = \frac{1}{2}$. احسب $u_n$ و $v_n$ و $\sum_{k=0}^{n} u_k$ و
$\sum_{k=0}^{n} v_k$، ونهايات العبارات الأربع كلها لما
$n \to +\infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:5}
باستعمال مبرهنة الحصر، احسب
\[
\lim_{n\to+\infty} \frac{n + \cos n}{n + 1}
\qquad\text{و}\qquad
\lim_{n\to+\infty} \frac{n!}{n^n},
\]
حيث $n! = 1 \times 2 \times \dots \times n$. ومن أجل النهاية الثانية، احصر
$\frac{n!}{n^n}$ بحد من \omterm{def:g11:seq:geometric}{متتالية هندسية}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:6}
ليكن $u_0 = 0$ و $u_{n+1} = \sqrt{u_n + 2}$ من أجل كل $n \in \N$.
\begin{enumerate}
  \item برهن بالتراجع على أن $0 \leq u_n \leq 2$ من أجل كل $n$.
  \item بيّن أن $(u_n)$ \omterm{def:g12:seq:monotonic}{متزايدة}.
  \item استنتج أن $(u_n)$ \omterm{def:g12:seq:limit}{تتقارب} وحدّد نهايتها.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:7}
يأخذ مريض جرعة مقدارها $1$ وحدة من دواء كل صباح. وخلال كل
\omterm{def:g10:numbers:interval}{فترة} من 24 ساعة، يطرح الجسم $40\%$ من الدواء الموجود. وليكن $u_n$
كمية الدواء في الجسم بُعيد جرعة اليوم $n$، بحيث
$u_0 = 1$.
\begin{enumerate}
  \item برّر أن $u_{n+1} = 0.6\,u_n + 1$.
  \item ليكن $v_n = u_n - 2.5$. بيّن أن $(v_n)$ \omterm{def:g12:seq:arith-geom}{هندسية} واستنتج
        صيغة صريحة للمقدار $u_n$.
  \item حدّد كمية الدواء في الجسم على \omterm{def:g10:stats:quartiles}{المدى} الطويل.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:8}
لتكن $(u_n)$ معرَّفة بالعلاقتين $u_0 = 3$ و $u_{n+1} = \frac{4u_n - 1}{u_n + 2}$.
\begin{enumerate}
  \item بيّن بالتراجع أن $u_n > 1$ من أجل كل $n \in \N$.
  \item بيّن أن $v_n = \dfrac{1}{u_n - 1}$ يعرّف \omterm{def:g11:seq:arithmetic}{متتالية حسابية}.
  \item استنتج صيغتين صريحتين للمقدارين $v_n$ و $u_n$، ونهاية
        $(u_n)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:seq:9}
من أجل $n \geq 1$، ليكن
$H_n = 1 + \frac{1}{2} + \frac{1}{3} + \dots + \frac{1}{n}$.
\begin{enumerate}
  \item بيّن أنه من أجل كل $n \geq 1$، يكون $H_{2n} - H_n \geq \frac{1}{2}$.
  \item استنتج أن $H_{2^k} \geq 1 + \frac{k}{2}$ من أجل كل $k \in \N$، ثم
        استنتج أن $H_n \to +\infty$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:seq:10}
\emph{(المتتاليتان المتجاورتان.)} تكون متتاليتان $(a_n)$ و $(b_n)$
\emph{متجاورتين} إذا كانت $(a_n)$ \omterm{def:g12:seq:monotonic}{متزايدة} و $(b_n)$ \omterm{def:g10:functions:variations}{متناقصة} و
$b_n - a_n \to 0$.
\begin{enumerate}
  \item بيّن أنه من أجل كل $n$، يكون $a_n \leq b_n$. (إرشاد: ادرس رتابة
        \omterm{def:g12:seq:sequence}{المتتالية} $(b_n - a_n)$.)
  \item بيّن أن المتتاليتين المتجاورتين تتقاربان كلتاهما إلى النهاية نفسها.
  \item تطبيق: بيّن أن المتتاليتين
        $a_n = \sum_{k=0}^{n} \frac{1}{k!}$ و
        $b_n = a_n + \frac{1}{n \cdot n!}$ ($n \geq 1$) متجاورتان. (ونهايتهما
        المشتركة هي العدد $\eu$، المدروس في \cref{ch:g12:exp}.)
\end{enumerate}
\end{exercise}

\section{مسألة: متتالية هيرون، تُحاكم أخيرًا}

\begin{problem}[{مسألة نهاية الأسبوع --- التراجع يصادق، والتقارب
الرتيب يحكم، ووصفة $\sqrt2$ ذات الألفي سنة تنال
برهانها أخيرًا (ومتوسط غاوس البديع للتحلية)}]
\label{pb:g12:seq:1}
ثلاث مرات التقت هذه السلسلة بوصفة هيرون --- خذ \omterm{def:g10:stats:mean}{متوسط}
التخمين مع $2/\text{التخمين}$ --- وثلاث مرات لم تستطع إلا أن
\emph{تلاحظ} أن الوصفة تعمل. وهذا الفصل يملك أخيرًا
أدوات الحكم: التراجع
(\cref{thm:g12:seq:induction})، ومبرهنة التقارب الرتيب
(\cref{thm:g12:seq:monotone})، ونهايات العلاقات التراجعية. ويشغل
الحكمُ والسرعةُ المصادَق عليها قلب هذه
المسألة؛ وحولهما، مصائد التراجع الكلاسيكية، وأبطأ
تباعد في الرياضيات، وأسرع تقارب وجده غاوس
قط.

\medskip
\noindent\textbf{الجزء الأول --- تمارين إحماء على التراجع.}
\begin{enumerate}
  \item برهن بالتراجع على:
        $1 + 3 + 5 + \dots + (2n - 1) = n^2$ (درج
        الأعداد \omterm{def:g11:func:parity}{الفردية}، المرسوم في الكتاب السابق، وقد صار
        الآن مصادَقًا عليه).
  \item برهن بالتراجع على أن $2^n > n$ من أجل كل
        $n \in \N$.
  \item برهن بالتراجع على متراجحة برنولي: من أجل
        $x \geq 0$ و $n \in \N$،
        $(1 + x)^n \geq 1 + nx$.
  \item المصيدة الكلاسيكية: ``كل الكريات لها اللون نفسه ---
        فهذا صحيح من أجل كرية واحدة؛ وإذا كانت أي $n$ كرية أحادية
        اللون دائمًا، فإن الكريات $n$ الأولى من بين $n + 1$ كرية
        تشترك في لون، والكريات $n$ الأخيرة تشترك في لون، فتشترك
        الكريات $n + 1$ كلها.'' وكل طفل يعرف أن الخاتمة
        سخيفة: فأوجد الخطوة التي ينهار عندها التراجع بالضبط.
  \item برهن بالتراجع على أن $4^n - 1$ يقبل القسمة على $3$
        من أجل كل $n \in \N$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- محاكمة هيرون.}
ليكن $x_0 = 2$ و $x_{n+1} = \dfrac12\left(x_n +
\dfrac{2}{x_n}\right)$.
\begin{enumerate}[resume]
  \item احسب $x_1$ و $x_2$ و $x_3$ كسورًا مضبوطة (أصدقاء
        قدامى).
  \item برهن على المتطابقة المفتاحية
        \[
        x_{n+1}^2 - 2 = \left(\frac{x_n^2 - 2}{2x_n}\right)^{\!2}
        \geq 0,
        \]
        واستنتج بالتراجع أن $x_n > 0$ و
        $x_n^2 > 2$ من أجل كل $n$.
  \item بيّن أن $(x_n)$ \omterm{def:g10:functions:variations}{متناقصة} تمامًا (احسب
        $x_{n+1} - x_n$ واستعمل السؤال 7).
  \item استدعِ مبرهنة التقارب الرتيب: لماذا \omterm{def:g12:seq:limit}{تتقارب}
        $(x_n)$ إلى نهاية $L \geq 1$ ما؟
  \item حدّد النهاية: انتقل بالعلاقة التراجعية إلى النهاية
        (\cref{prop:g12:seq:operations}) واستنتج
        $L = \sqrt2$. وصُغ الحكم التاريخي: بعد ألفي
        سنة من الخدمة الأمينة، صار \emph{مبرهَنًا} أن وصفة هيرون
        \omterm{def:g12:seq:limit}{تتقارب}.
  \item السرعة المصادَق عليها: مع $e_n = x_n - \sqrt2$، برهن على
        \[
        e_{n+1} = \frac{e_n^2}{2 x_n} ,
        \]
        واستنتج $e_{n+1} \leq \frac{e_n^2}{2\sqrt2}$: أي أن
        الخطأ يُربَّع عند كل خطوة --- وهو تضاعف الأرقام
        الملاحَظ منذ الكتاب السابق، وقد صار مبرهنة.
  \item تأكد عدديًا: احسب $e_0, e_1, e_2, e_3$ (انطلاقًا من
        السؤال 6) وتحقق من أن كل $\frac{e_{n+1}}{e_n^2}$
        قريب من $\frac{1}{2x_n}$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- أبطأ تباعد.}
\begin{enumerate}[resume]
  \item برهن \Cref{exo:g12:seq:9} على
        $H_{2^k} \geq 1 + \frac k2$ من أجل المجاميع التوافقية. فكم
        حدًا يضمن $H_n > 10$؟ (وتكفي قوة للعدد اثنين؛
        فاعجب من حجمها.)
  \item وفي المقابل، \omterm{def:g12:seq:limit}{تتقارب} المجاميع \omterm{def:g12:seq:arith-geom}{الهندسية}
        $1 + \frac12 + \frac14 + \dots + \frac{1}{2^n} =
        2 - \frac{1}{2^n}$ إلى $2$
        (\cref{thm:g12:seq:geometric}): وهذا حدس لوح الشوكولاتة
        في الكتاب السابق، وقد صار أخيرًا عبارة عن
        نهاية. اكتب البرهان في سطرين.
  \item وبين الاثنين: بيّن أن المجاميع
        $S_n = 1 + \frac{1}{4} + \frac{1}{9} + \dots +
        \frac{1}{n^2}$ \omterm{def:g12:seq:limit}{تتقارب}، بحصر
        $\frac{1}{k^2} \leq \frac{1}{k(k-1)} =
        \frac{1}{k-1} - \frac{1}{k}$ (من أجل $k \geq 2$)،
        ثم التلسكوب، ثم تطبيق التقارب الرتيب. (وهذه
        النهاية، $\frac{\pi^2}{6}$، إحدى معجزات أويلر،
        المبرهَن عليها في الكتب الجامعية.)
  \item صُغ عبرة الأسئلة 13--15 في جملتين:
        ماذا تقرّر عبارة ``الحدود تؤول إلى $0$'' بخصوص
        تقارب المجاميع --- وماذا لا تقرّر؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- \omterm{def:g11:stat:mean}{المتوسط الحسابي} الهندسي عند
غاوس.}
ليكن $a_0 = 1$ و $b_0 = 2$ و
\[
a_{n+1} = \sqrt{a_n b_n},
\qquad
b_{n+1} = \frac{a_n + b_n}{2} .
\]
\begin{enumerate}[resume]
  \item احسب $a_1, b_1, a_2, b_2$ (بخمسة أرقام عشرية). فماذا
        تلاحظ بخصوص السرعة؟
  \item بيّن أن $a_n \leq b_n$ من أجل كل $n$ (وهي
        المتراجحة بين المتوسطين الحسابي والهندسي، التي لقيناها في هذه
        السلسلة كلها)، وأن $(a_n)$ \omterm{def:g12:seq:monotonic}{متزايدة} و $(b_n)$ \omterm{def:g10:functions:variations}{متناقصة}.
  \item بيّن أن
        $b_{n+1} - a_{n+1} \leq \frac{b_n - a_n}{2}$
        (بإخراج $b_{n+1} - a_{n+1} =
        \frac{(\sqrt{b_n} - \sqrt{a_n})^2}{2}$ عاملًا مشتركًا والمقارنة)،
        واستنتج حسب \cref{exo:g12:seq:10} أن المتتاليتين
        متجاورتان: فلهما نهاية مشتركة
        $M(1, 2)$، هي \emph{\omterm{def:g11:stat:mean}{المتوسط الحسابي} الهندسي}.
  \item احسب $M(1, 2)$ بستة أرقام عشرية (وكم تكرارًا
        لزمك؟). في 30 أيار 1799، حسب غاوس
        $M(1, \sqrt2)$ بأحد عشر رقمًا عشريًا، وتعرّف على
        $\frac{\pi}{M(1,\sqrt2)}$ بوصفه تكاملًا معروفًا،
        وكتب أن ``حقلًا جديدًا من التحليل'' قد انفتح ---
        وقد انفتح فعلًا: التكاملات الإهليلجية، المحكية في الكتب
        الجامعية. واختم بسرعات التقارب الملاحَظة في
        هذه المسألة، من الأبطأ إلى الأسرع.
\end{enumerate}
\end{problem}
