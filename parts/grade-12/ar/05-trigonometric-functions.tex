\chapter{الدوال المثلثية}\label{ch:g12:trigo}

تحوّل الدالتان $\cos$ و $\sin$ هندسة الدائرة إلى
تحليل. وإذ تُعرَّفان بلفّ المستقيم الحقيقي حول دائرة الوحدة، فهما
النموذج الأصلي للدوال الدورية، ومشتقتاهما تجعلانهما
حلَّين \omterm{def:g10:algebra:equation}{لمعادلة} التذبذبات $y'' = -y$.

\section{دائرة الوحدة}

\begin{definition}[جيب التمام والجيب]\label{def:g12:trigo:cossin}
لتكن $\mathcal{C}$ الدائرة التي نصف قطرها 1 ومركزها مبدأ
\omterm{def:g10:coordgeom:system}{نظام متعامد ومتجانس}. وأرفق بكل \omterm{def:g10:numbers:sets}{عدد حقيقي} $t$ النقطة $M(t)$
التي نبلغها بلفّ طول $\abs{t}$ على $\mathcal{C}$ انطلاقًا من النقطة
$I(1,0)$، في الاتجاه المباشر إذا كان $t \geq 0$ وفي الاتجاه المعاكس فيما عدا ذلك. عندئذٍ
\emph{يكون $\cos t$ و $\sin t$ إحداثيَي
$M(t)$}\index{جيب التمام}\index{جيب}:
\[
M(t) = (\cos t, \sin t).
\]
\end{definition}

وبما أن محيط الدائرة $2\pi$، لا تتغير النقطة $M(t)$ عندما
يزيد $t$ بمقدار $2\pi$؛ وبما أن $M(t)$ تقع على دائرة الوحدة:

\begin{proposition}[المتطابقات الأساسية]\label{prop:g12:trigo:identities}
من أجل كل $t \in \R$ وكل $k \in \Z$:
\[
\cos^2 t + \sin^2 t = 1, \qquad
\cos(t + 2k\pi) = \cos t, \qquad
\sin(t + 2k\pi) = \sin t,
\]
\[
\cos(-t) = \cos t, \qquad \sin(-t) = -\sin t,
\]
\[
\cos(\pi - t) = -\cos t, \quad \sin(\pi - t) = \sin t, \quad
\cos\left(\tfrac{\pi}{2} - t\right) = \sin t, \quad
\sin\left(\tfrac{\pi}{2} - t\right) = \cos t .
\]
\end{proposition}

\begin{proof}
المتطابقة الأولى هي \omterm{def:g10:algebra:equation}{معادلة} دائرة الوحدة؛ وأما الأخريات فتعبّر عن
تناظرات الدائرة: فالنقطة $M(-t)$ هي نظيرة $M(t)$ بالنسبة إلى
المحور $x$، و $M(\pi - t)$ بالنسبة إلى المحور $y$، و $M(\frac\pi2 - t)$ بالنسبة إلى المستقيم
$y = x$.
\end{proof}

والقيم الواجب حفظها عن ظهر قلب:
\begin{center}
\begin{tabular}{c|ccccc}
$t$ & $0$ & $\dfrac{\pi}{6}$ & $\dfrac{\pi}{4}$ & $\dfrac{\pi}{3}$ & $\dfrac{\pi}{2}$\\[6pt]
\hline\rule{0pt}{14pt}%
$\cos t$ & $1$ & $\dfrac{\sqrt3}{2}$ & $\dfrac{\sqrt2}{2}$ & $\dfrac12$ & $0$\\[6pt]
$\sin t$ & $0$ & $\dfrac12$ & $\dfrac{\sqrt2}{2}$ & $\dfrac{\sqrt3}{2}$ & $1$
\end{tabular}
\end{center}

\begin{proposition}[صيغ الجمع]\label{prop:g12:trigo:addition}
من أجل كل $a, b \in \R$:
\begin{align*}
\cos(a+b) &= \cos a \cos b - \sin a \sin b, &
\sin(a+b) &= \sin a \cos b + \cos a \sin b,\\
\cos(a-b) &= \cos a \cos b + \sin a \sin b, &
\sin(a-b) &= \sin a \cos b - \cos a \sin b.
\end{align*}
وخاصة $\cos 2a = 2\cos^2 a - 1 = 1 - 2\sin^2 a$ و
$\sin 2a = 2 \sin a \cos a$.
\end{proposition}

\begin{proof}
المتجهتان $\vect{u} = (\cos a, \sin a)$ و $\vect{v} = (\cos b, \sin b)$
متجهتان واحديتان تصنعان زاوية $a - b$؛ وجداؤهما السلّمي يساوي
$\cos a\cos b + \sin a \sin b$ (\omterm{def:g10:coordgeom:system}{بالإحداثيات}) و
$\norm{\vect u}\,\norm{\vect v}\cos(a-b) = \cos(a - b)$ (بالهندسة) معًا، وهذه هي
الصيغة الثالثة. وتعويض $-b$ محل $b$ يعطي الأولى؛ وتنتج صيغتا \omterm{def:g11:trigo:cossin}{الجيب}
باستعمال $\sin x = \cos(\frac\pi2 - x)$. أما صيغتا الضعف فهما
الحالة $b = a$ مقرونة بالمتطابقة $\cos^2 + \sin^2 = 1$.
\end{proof}

\section{الدراسة التحليلية}

\begin{lemma}[النهاية الأساسية]\label{lem:g12:trigo:sinxx}
\[
\lim_{t \to 0} \frac{\sin t}{t} = 1
\qquad\text{و}\qquad
\lim_{t \to 0} \frac{\cos t - 1}{t} = 0 .
\]
\end{lemma}

\begin{proof}
من أجل $0 < t < \frac{\pi}{2}$، قارن ثلاث مساحات في دائرة الوحدة:
المثلث $OIM(t)$، والقطاع الدائري $OIM(t)$، والمثلث القائم ذي
القاعدة $OI$ والارتفاع $\tan t$:
\[
\frac{\sin t}{2} \leq \frac{t}{2} \leq \frac{\tan t}{2}.
\]

المتراجحة الأولى تعطي $\frac{\sin t}{t} \leq 1$؛ والثانية تعطي
$\cos t \leq \frac{\sin t}{t}$. ولما $t \to 0^+$، يكون $\cos t \to 1$ (\omterm{def:g12:limcont:continuity}{بالاتصال})،
وتعطي مبرهنة الحصر $\frac{\sin t}{t} \to 1$؛ وتمدّد \omterm{def:g11:func:parity}{الزوجية} ذلك
إلى $t \to 0$. وأما النهاية الثانية،
\[
\frac{\cos t - 1}{t} = \frac{\cos^2 t - 1}{t(\cos t + 1)}
= -\frac{\sin t}{t}\cdot\frac{\sin t}{\cos t + 1}
\xrightarrow[t\to0]{} -1 \cdot \frac{0}{2} = 0. \qedhere
\]
\end{proof}

\begin{theorem}[مشتقتا الجيب وجيب التمام]\label{thm:g12:trigo:derivatives}
الدالتان $\sin$ و $\cos$ قابلتان للاشتقاق على $\R$، مع
\[
(\sin)' = \cos, \qquad (\cos)' = -\sin .
\]
وبوجه أعم، $(\sin(ax+b))' = a\cos(ax+b)$ و
$(\cos(ax+b))' = -a\sin(ax+b)$.
\end{theorem}

\begin{proof}
حسب صيغ الجمع،
\[
\frac{\sin(x+h) - \sin x}{h}
= \sin x\,\frac{\cos h - 1}{h} + \cos x\,\frac{\sin h}{h}
\xrightarrow[h\to0]{} \sin x \cdot 0 + \cos x \cdot 1 = \cos x,
\]
باستعمال \cref{lem:g12:trigo:sinxx}. والحساب من أجل $\cos$ مطابق،
وتنتج الصورة العامة من قاعدة السلسلة.
\end{proof}

\begin{proposition}[التغيرات]\label{prop:g12:trigo:variations}
على $\intcc{0}{\pi}$، تتناقص $\cos$ من $1$ إلى $-1$. وعلى
$\intcc{-\frac\pi2}{\frac\pi2}$، تتزايد $\sin$ من $-1$ إلى $1$. والدالتان
دوريتان دورهما $2\pi$ ومجموعة قيمهما $\intcc{-1}{1}$؛ و $\cos$
\omterm{def:g11:func:parity}{زوجية}، و $\sin$ \omterm{def:g11:func:parity}{فردية}.
\end{proposition}

\begin{proof}
على $\intoo{0}{\pi}$، $(\cos)' = -\sin < 0$ لأن \omterm{def:g10:coordgeom:system}{ترتيب} النقطة $M(t)$
موجب هناك؛ وبالمثل $(\sin)' = \cos > 0$ على
$\intoo{-\frac\pi2}{\frac\pi2}$. وتأتي الدورية \omterm{def:g11:func:parity}{والزوجية} من
\cref{prop:g12:trigo:identities}.
\end{proof}

\begin{method}[حل $\cos x = a$ و $\sin x = a$]\label{met:g12:trigo:equations}
من أجل $a \in \intcc{-1}{1}$، أوجد حلًا خاصًا $\alpha$ (من
جدول القيم أو بآلة حاسبة). عندئذٍ تكون كل الحلول:
\[
\cos x = \cos\alpha \iff x = \alpha + 2k\pi \text{ أو } x = -\alpha + 2k\pi,
\quad k \in \Z;
\]
\[
\sin x = \sin\alpha \iff x = \alpha + 2k\pi \text{ أو } x = \pi - \alpha + 2k\pi,
\quad k \in \Z.
\]
وأما المتراجحات، فحدّد أقواس الحلول على دائرة الوحدة واقرأ
الفترات داخل دور واحد.
\end{method}

\begin{example}\label{ex:g12:trigo:equation}
حل $\cos x = \frac12$ في $\intoc{-\pi}{\pi}$: حل خاص هو
$\frac\pi3$، إذن الحلول هي $x = \frac\pi3$ و $x = -\frac\pi3$.
وفي $\R$ كلها: $x = \pm\frac\pi3 + 2k\pi$، حيث $k \in \Z$.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:trigo:1}
احسب $\cos\frac{2\pi}{3}$ و $\sin\frac{5\pi}{6}$ و $\cos\frac{7\pi}{4}$ و
$\sin\left(-\frac{\pi}{3}\right)$ باستعمال متطابقات
\cref{prop:g12:trigo:identities}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:trigo:2}
اشتق $f(x) = \sin^2 x$ و $g(x) = \cos(3x) + x\sin x$ و
$h(x) = \dfrac{\sin x}{2 + \cos x}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:trigo:3}
حل في $\intoc{-\pi}{\pi}$، ثم في $\R$:
\[
\text{(أ) } \sin x = \frac{\sqrt3}{2};
\qquad
\text{(ب) } \cos\left(2x\right) = \frac{\sqrt2}{2} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:4}
باستعمال صيغ الجمع، احسب القيمة المضبوطة للمقدارين
$\cos\frac{\pi}{12}$ و $\sin\frac{\pi}{12}$.
(إرشاد: $\frac{\pi}{12} = \frac{\pi}{3} - \frac{\pi}{4}$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:5}
حل في $\intcc{0}{2\pi}$ المتراجحة $2\sin^2 x - \sin x - 1 \geq 0$.
(إرشاد: حلّل التربيعية في $\sin x$ إلى عوامل.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:6}
احسب النهايات
\[
\lim_{x\to0} \frac{\sin 3x}{x}, \qquad
\lim_{x\to0} \frac{1 - \cos x}{x^2}, \qquad
\lim_{x\to+\infty} x \sin\frac{1}{x}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:7}
ادرس \omterm{def:g11:func:function}{الدالة} $f(x) = x + \cos x$ على $\intcc{0}{2\pi}$: التغيرات و
القيم الحدية. وبيّن أن $f$ \omterm{def:g12:seq:monotonic}{متزايدة} على $\R$ مع أن $f'$ تنعدم عند
عدد لانهائي من النقط.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:trigo:8}
لتكن $f(x) = A\cos x + B \sin x$ مع $(A,B) \neq (0,0)$.
\begin{enumerate}
  \item بيّن أنه يمكن كتابة $f$ على الصورة $f(x) = R\cos(x - \varphi)$ حيث
        $R = \sqrt{A^2 + B^2}$ ومع $\varphi$ مناسب.
  \item استنتج القيمتين العظمى والصغرى \omterm{def:g11:func:function}{للدالة} $f$، ثم حل
        $\cos x + \sin x = 1$ في $\intoc{-\pi}{\pi}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:trigo:9}
بيّن أن $\sin$ و $\cos$ تحققان كلتاهما \omterm{def:g10:algebra:equation}{المعادلة} التفاضلية
$y'' = -y$. وبالعكس، لتكن $y$ حلًا \omterm{def:g10:algebra:equation}{للمعادلة} $y'' = -y$ حيث
$y(0) = a$ و $y'(0) = b$؛ بيّن أن $y = a\cos + b\sin$.
(إرشاد: اعتبر $g = (y - a\cos - b\sin)$ و``الطاقة''
$E = g^2 + (g')^2$.)
\end{exercise}

\section{مسألة: سرّ النواس}

\begin{problem}[{مسألة نهاية الأسبوع --- نهاية هندسية واحدة تشغّل
مشتقتَي الجيب وجيب التمام، ومنهما ينبع كل
التذبذب: النوابض، والنواسات، والضربات، والمتر الذي كاد
يكون}]
\label{pb:g12:trigo:1}
كل ساعة دقّت يومًا، وكل وتر رنّ يومًا،
يطيع الرياضيات نفسها: قوة إرجاع تتناسب مع
الإزاحة، ومنه \omterm{def:g10:algebra:equation}{المعادلة} $y'' = -\omega^2 y$، ومنها
جيوب التمام. وفي \omterm{def:g11:seq:arithmetic}{أساس} ذلك كله تجلس نهاية تبدو بريئة،
$\frac{\sin x}{x} \to 1$ --- وُعِد بها في الكتاب السابق، واستعملتها
\cref{thm:g12:trigo:derivatives}، ويُبرهن عليها هنا برسم.
وتتسلق هذه المسألة من تلك النهاية إلى تأرجح نواس طوله متر
واحد، وتفسّر لماذا لم يُعرَّف المتر بها إلا بشعرة.

\medskip
\noindent\textbf{الجزء الأول --- النهاية الأساسية.}
\begin{enumerate}
  \item أحمِ نفسك عدديًا: احسب $\frac{\sin x}{x}$ من أجل
        $x = 0.1$ و $x = 0.01$ (\omterm{def:g11:trigo:radian}{بالراديان}!).
  \item الحصر، على دائرة الوحدة مع
        $0 < x < \frac\pi2$: عبّر بالهندسة الابتدائية عن
        مساحات (أ) المثلث ذي الرؤوس $O$ و $A(1,0)$
        ونقطة الدائرة $M(\cos x, \sin x)$؛ (ب) القطاع
        الدائري $OAM$؛ (ج) المثلث القائم $OAT$
        حيث $T(1, \tan x)$.
  \item استنتج من حصر المساحات
        $\sin x < x < \tan x$، ثم
        $\cos x < \frac{\sin x}{x} < 1$، ثم اختم بمبرهنة
        الحصر أن
        $\lim_{x \to 0} \frac{\sin x}{x} = 1$ (\omterm{def:g11:func:parity}{والزوجية} تعالج
        $x < 0$).
  \item استنتج
        $\lim_{x \to 0} \frac{1 - \cos x}{x^2} = \frac12$
        (بالضرب في المرافق).
  \item فسّر لماذا تكون هذه النهاية محرّك
        \cref{thm:g12:trigo:derivatives} (ما هو
        $\sin'(0)$؟)، ولماذا تسدّ أيضًا دينًا قديمًا: بأي
        معنى تكون \emph{الراديانات} وحدة الزوايا الوحيدة
        التي تصح من أجلها $\sin' = \cos$ بلا ثابت؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الحركة التوافقية.}
\begin{enumerate}[resume]
  \item تحقق بقاعدة السلسلة من أن
        $y(t) = R\cos(\omega t - \varphi)$ تحقق
        $y'' = -\omega^2 y$.
  \item كتلة على نابض تطيع $y'' = -4y$، حيث
        $y(0) = 3$ و $y'(0) = 8$. وبقبول (انطلاقًا من
        \cref{exo:g12:trigo:9}) أن كل الحلول هي
        $A\cos 2t + B\sin 2t$، حدّد $A$ و $B$.
  \item أعد كتابة الحل على الصورة $R\cos(2t - \varphi)$
        (\cref{exo:g12:trigo:8}): وأعطِ السعة $R$، و
        الدور، والطور $\varphi$ (بثلاثة أرقام عشرية).
  \item في أي لحظة تمر الكتلة أول مرة بوضع
        التوازن $y = 0$؟
  \item النواس: كرة على خيط طوله $L$ تطيع
        \emph{بالضبط} $\theta'' = -\frac gL \sin\theta$ ---
        وهي غير قابلة للحل بالدوال الابتدائية. ومن أجل التأرجحات الصغيرة،
        عوّض $\theta$ محل $\sin\theta$ (والجزء الأول يبرّر ذلك!)
        واستنتج الدور
        $T = 2\pi\sqrt{\frac Lg}$. احسب $T$ من أجل
        $L = 1$ m (حيث $g = 9.81$): وهو اكتشاف غاليليو الشهير ---
        فعلى أي مقدار \emph{لا} تتعلق الصيغة؟
  \item نواس الثانية: أي طول يعطي
        $T = 2$ s بالضبط؟ لقد ترددت أكاديمية العلوم الفرنسية سنة 1791
        بين هذا الطول وبين جزء من عشرة ملايين من
        ربع خط الزوال لتعريف \emph{المتر} --- فبكم
        مليمترًا يختلف المرشحان؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- صيغ الجمع في العمل.}
\begin{enumerate}[resume]
  \item بيّن بصيغ الجمع
        (\cref{prop:g12:trigo:addition}) أن
        $\cos\left(x + \frac\pi2\right) = -\sin x$ و
        $\sin\left(x + \frac\pi2\right) = \cos x$: فالاشتقاق
        الدائري ربع دورة --- وتحقق من الاتساق
        مع \cref{thm:g12:trigo:derivatives}.
  \item استخرج متطابقة تحويل الجداء إلى مجموع
        $\cos a \cos b = \frac12\left(\cos(a - b) +
        \cos(a + b)\right)$ والتخطيط
        $\cos^2 x = \frac{1 + \cos 2x}{2}$ (وسيشكرك فصل
        التكامل).
  \item الضربات: استخرج
        $\sin p + \sin q = 2 \sin\frac{p + q}{2}
        \cos\frac{p - q}{2}$، وطبّقها على شوكتين رنانتين
        عند $440$ و $444$ Hz: فأي نغمة تسمعها الأذن، وكم
        نبضة في الثانية؟ (وهكذا تُدوزَن الأوركسترات.)
  \item استخرج متطابقة الزاوية الثلاثية
        $\cos 3x = 4\cos^3 x - 3\cos x$
        (بكتابة $3x = 2x + x$). وبوضع $x = 20^\circ$، أي
        \omterm{def:g10:algebra:equation}{معادلة} تكعيبية يجب أن يحققها $\cos 20^\circ$؟ (وكون هذه
        التكعيبية غير قابلة للحل بالجذور التربيعية وحدها هو
        بالضبط سبب هزيمة تثليث الزاوية للمسطرة
        والفرجار طوال ألفي سنة --- والحكاية الكاملة
        محكية في الكتب الجامعية.)
  \item تيار متناوب مقداره
        $I(t) = 3\cos(100\pi t) + 3\sqrt3 \sin(100\pi t)$.
        أعد كتابته على الصورة $R\cos(100\pi t - \varphi)$: فما
        التيار الأعظمي والطور؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- العالم المتذبذب.}
\begin{enumerate}[resume]
  \item الأدوار: ما (أصغر) دور \omterm{def:g11:func:function}{للدالة}
        $\sin^2 t$ (استعمل السؤال 13)؟ وما دور
        $\sin(2t) + \sin(3t)$؟
  \item النواس الحقيقي يُخمَّد:
        $y(t) = \eu^{-t/10}\cos(2\pi t)$. صِف الحركة،
        واحسب قيمة الغلاف عند $t = 10$ (فيظهر صديق قديم،
        \cref{pb:g12:exp:1})، وسمِّ الفصل الذي
        تُحل فيه المعادلات ذات الحلول كهذه.
  \item لماذا تحكم الجيوب وجيوب التمام كل اهتزاز؟ أعطِ
        قاموس الفيزياء إلى الرياضيات في جملتين
        (قوة إرجاع تتناسب مع الإزاحة
        $\rightarrow$ أي \omterm{def:g10:algebra:equation}{معادلة} $\rightarrow$ أي
        حلول)، وقل ماذا يسهم به \cref{exo:g12:trigo:9}
        في هذا الادعاء.
  \item الخاتمة: قوس الفصل في أربع خطوات --- نهاية
        \omterm{def:g12:seq:arith-geom}{هندسية} واحدة، ومشتقتان، \omterm{def:g10:algebra:equation}{ومعادلة} تفاضلية
        واحدة، وكل التذبذب. وأما النجمة الأمينة:
        فمن أجل التأرجحات الواسعة يتضمن دور النواس الحقيقي
        تكاملًا إهليلجيًا، يُحسب بسرعة البرق بواسطة\,\dots
        \omterm{def:g11:stat:mean}{المتوسط الحسابي} الهندسي عند غاوس، تحلية
        \cref{pb:g12:seq:1}. فخيوط السلسلة تعقد نفسها بنفسها.
\end{enumerate}
\end{problem}
