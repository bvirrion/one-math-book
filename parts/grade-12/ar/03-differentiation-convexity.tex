\chapter{الاشتقاق والتحدب}\label{ch:g12:deriv}

تقيس \omterm{def:g12:deriv:derivative}{المشتقة} معدل تغير \omterm{def:g11:func:function}{دالة} اللحظي؛ وهي
\omterm{def:g10:reffunc:affine}{ميل} \omterm{def:g12:deriv:tangent}{المماس} لمنحناها. ويراجع هذا الفصل ويمدّد
قواعد الاشتقاق، ويربط إشارة $f'$ بتغيرات $f$،
ويقدّم \omterm{def:g12:deriv:convex}{التحدب}، الذي تحكمه \omterm{def:g12:deriv:derivative}{المشتقة} الثانية.

\section{المشتقة}

\begin{definition}[المشتقة عند نقطة]\label{def:g12:deriv:derivative}
لتكن $f$ معرَّفة على \omterm{def:g10:numbers:interval}{فترة} $I$ وليكن $a \in I$. تكون \omterm{def:g11:func:function}{الدالة} $f$
\emph{قابلة للاشتقاق عند $a$}\index{مشتقة}\index{قابلية الاشتقاق} إذا كان لمعدل التغير
\[
\frac{f(a+h) - f(a)}{h}
\]
نهاية منتهية لما $h \to 0$. وتُسمّى هذه النهاية \emph{مشتقة
$f$ عند $a$} وتُكتب $f'(a)$. وإذا كانت $f$ قابلة للاشتقاق عند كل نقطة من
$I$، \omterm{def:g11:func:function}{فالدالة} $f' \colon x \mapsto f'(x)$ هي \emph{مشتقة}
$f$.
\end{definition}

\begin{definition}[المماس]\label{def:g12:deriv:tangent}
إذا كانت $f$ قابلة للاشتقاق عند $a$، فإن \emph{المماس}\index{مماس} \omterm{def:g10:functions:graph}{لمنحنى}
$f$ عند النقطة $(a, f(a))$ هو المستقيم ذو \omterm{def:g10:algebra:equation}{المعادلة}
\[
y = f(a) + f'(a)(x - a).
\]
\end{definition}

\begin{proposition}\label{prop:g12:deriv:diffcont}
كل \omterm{def:g11:func:function}{دالة} قابلة للاشتقاق عند $a$ \omterm{def:g12:limcont:continuity}{متصلة} عند $a$. والعكس خاطئ.
\end{proposition}

\begin{proof}
من أجل $h \neq 0$ صغير، $f(a+h) - f(a) = h \cdot \frac{f(a+h)-f(a)}{h}$. ولما
$h \to 0$، يؤول الطرف الأيمن إلى $0 \times f'(a) = 0$، إذن
$f(a+h) \to f(a)$. وأما العكس، \omterm{def:g11:func:function}{فالدالة} $x \mapsto \abs{x}$ \omterm{def:g12:limcont:continuity}{متصلة} عند
$0$، لكن معدل تغيرها عند $0$ يساوي $\frac{\abs h}{h} = \pm 1$
بحسب إشارة $h$، وليست له نهاية.
\end{proof}

\subsection{قواعد الاشتقاق}

\begin{proposition}[العمليات]\label{prop:g12:deriv:operations}
لتكن $u, v$ قابلتين للاشتقاق على $I$ وليكن $\lambda \in \R$. عندئذٍ تكون $u + v$
و $\lambda u$ و $uv$ قابلة للاشتقاق على $I$، وكذلك $\frac{u}{v}$ حيث
$v \neq 0$، ويكون
\[
(u+v)' = u' + v', \quad (\lambda u)' = \lambda u', \quad
(uv)' = u'v + uv', \quad
\left(\frac{u}{v}\right)' = \frac{u'v - uv'}{v^2}.
\]
\end{proposition}

\begin{proof}[برهان قاعدة الجداء]
اكتب معدل تغير $uv$ عند $a$ على الصورة
\[
\frac{u(a+h)v(a+h) - u(a)v(a)}{h}
= \frac{u(a+h) - u(a)}{h}\,v(a+h) + u(a)\,\frac{v(a+h) - v(a)}{h}.
\]
ولما $h \to 0$، يكون $v(a+h) \to v(a)$ \omterm{def:g12:limcont:continuity}{بالاتصال}
(\cref{prop:g12:deriv:diffcont})، إذن يؤول الطرف الأيمن إلى
$u'(a)v(a) + u(a)v'(a)$. ويُبرهن على القواعد الأخرى بالمثل.
\end{proof}

\begin{theorem}[قاعدة السلسلة]\label{thm:g12:deriv:chain}
\index{قاعدة السلسلة}
لتكن $u$ قابلة للاشتقاق على $I$ وقيمها في $J$، ولتكن $g$ قابلة للاشتقاق
على $J$. عندئذٍ تكون $g \circ u$ قابلة للاشتقاق على $I$ ويكون
\[
(g \circ u)' = (g' \circ u) \cdot u' .
\]
وخاصة، من أجل $u$ قابلة للاشتقاق:
\[
(u^n)' = n\,u'\,u^{n-1} \ (n \in \Z), \qquad
\left(\sqrt{u}\right)' = \frac{u'}{2\sqrt{u}} \ (u > 0), \qquad
\left(\eu^{u}\right)' = u'\,\eu^{u}.
\]
\end{theorem}

\begin{proof}[مخطط البرهان]
عندما يكون $u(a+h) \neq u(a)$ من أجل $h$ صغير، اكتب
\[
\frac{g(u(a+h)) - g(u(a))}{h}
= \frac{g(u(a+h)) - g(u(a))}{u(a+h) - u(a)} \cdot \frac{u(a+h) - u(a)}{h}.
\]
ولما $h \to 0$، يكون $u(a+h) \to u(a)$، إذن يؤول العامل الأول إلى $g'(u(a))$
ويؤول الثاني إلى $u'(a)$. (وعلى البرهان الكامل أن يعالج الحالة التي يكون فيها
$u(a+h) = u(a)$ من أجل $h$ صغير كما نشاء؛ وتُعالَج هذه النقطة التقنية في
الجامعة.)
\end{proof}

\begin{example}
المشتقات المألوفة، الصالحة على مجموعات التعريف الطبيعية:
\[
(x^n)' = nx^{n-1}, \quad
\left(\frac{1}{x}\right)' = -\frac{1}{x^2}, \quad
(\sqrt{x})' = \frac{1}{2\sqrt{x}}, \quad
(\cos x)' = -\sin x, \quad
(\sin x)' = \cos x .
\]
ويُبرهن على الأولى بالتراجع انطلاقًا من قاعدة الجداء، وعلى الأخيرتين في
\cref{ch:g12:trigo}.
\end{example}

\section{التغيرات والقيم الحدية}

\begin{theorem}[إشارة المشتقة والتغيرات]\label{thm:g12:deriv:variations}
لتكن $f$ قابلة للاشتقاق على \omterm{def:g10:numbers:interval}{فترة} $I$.
\begin{enumerate}
  \item تكون $f$ \omterm{def:g12:seq:monotonic}{متزايدة} على $I$ إذا وفقط إذا كان $f' \geq 0$ على $I$.
  \item تكون $f$ \omterm{def:g10:functions:variations}{متناقصة} على $I$ إذا وفقط إذا كان $f' \leq 0$ على $I$.
  \item إذا كان $f' > 0$ على $I$ عدا عند عدد منته من النقط التي
        تنعدم عندها، فإن $f$ \omterm{def:g12:seq:monotonic}{متزايدة} تمامًا على $I$.
\end{enumerate}
\end{theorem}

\begin{proof}[برهان جزئي]
إذا كانت $f$ \omterm{def:g12:seq:monotonic}{متزايدة}، فإن كل معدل تغير
$\frac{f(a+h)-f(a)}{h}$ يكون $\geq 0$، وبالانتقال إلى النهاية نجد
$f'(a) \geq 0$. أما الاستلزامات العكسية فتستند إلى مبرهنة التزايدات المنتهية،
المبرهَن عليها في الجامعة؛ وهي \emph{مقبولة في هذا المستوى}.
\end{proof}

\begin{definition}[القيمة الحدية المحلية]\label{def:g12:deriv:extremum}
تملك $f$ \emph{قيمة عظمى محلية}\index{قيمة حدية} عند $a$ إذا كان $f(x) \leq f(a)$ من أجل
كل $x$ بجوار $a$؛ وتُعرَّف القيم الصغرى المحلية بتناظر.
\end{definition}

\begin{proposition}[شرط المرتبة الأولى]\label{prop:g12:deriv:critical}
إذا كانت $f$ قابلة للاشتقاق على \omterm{def:g10:numbers:interval}{فترة} \emph{مفتوحة} $I$ وتملك \omterm{def:g12:deriv:extremum}{قيمة حدية}
محلية عند $a \in I$، فإن $f'(a) = 0$. والعكس خاطئ
($f(x) = x^3$ عند $a = 0$).
\end{proposition}

\begin{proof}
لنقل إن $a$ \omterm{def:g12:deriv:extremum}{قيمة عظمى محلية}. من أجل $h > 0$ صغير،
$\frac{f(a+h)-f(a)}{h} \leq 0$، إذن بجعل $h \to 0^+$ نجد $f'(a) \leq 0$؛
ومن أجل $h < 0$ يكون معدل التغير $\geq 0$، فنجد $f'(a) \geq 0$. ومنه
$f'(a) = 0$.
\end{proof}

\begin{method}[جدول التغيرات]\label{met:g12:deriv:table}
لدراسة \omterm{def:g11:func:function}{دالة} $f$: حدّد \omterm{def:g11:func:function}{مجموعة تعريفها} ونهاياتها عند الحدود؛
ثم احسب $f'$ وحلّلها إلى عوامل؛ ثم حدّد إشارة $f'$ على كل \omterm{def:g10:numbers:interval}{فترة} جزئية؛
وسجّل كل شيء في جدول تغيرات، مؤشرًا القيم الحدية؛ ثم استنتج عدد
حلول \omterm{def:g10:algebra:equation}{المعادلة} $f(x) = k$ باستعمال مبرهنة التقابل
(\cref{thm:g12:limcont:bijection}).
\end{method}

\section{التحدب}

\begin{definition}[الدالة المحدبة]\label{def:g12:deriv:convex}
تكون \omterm{def:g11:func:function}{الدالة} $f$ المعرَّفة على \omterm{def:g10:numbers:interval}{فترة} $I$ \emph{محدبة}\index{تحدب}
على $I$ إذا كان كل وتر من أوتار منحناها فوق \omterm{def:g10:functions:graph}{المنحنى}: أي من أجل كل
$a, b \in I$ وكل $t \in \intcc{0}{1}$،
\[
f\bigl((1-t)a + t b\bigr) \leq (1-t) f(a) + t f(b).
\]
وتكون \emph{مقعرة}\index{دالة مقعرة} إذا تحققت المتراجحة المعاكسة (أي إذا كانت $-f$ محدبة).
\end{definition}

\begin{theorem}[التحدب والمشتقات]\label{thm:g12:deriv:convexity}
لتكن $f$ قابلة للاشتقاق على \omterm{def:g10:numbers:interval}{فترة} $I$. العبارات التالية متكافئة:
\begin{enumerate}
  \item $f$ \omterm{def:g12:deriv:convex}{محدبة} على $I$؛
  \item $f'$ \omterm{def:g12:seq:monotonic}{متزايدة} على $I$؛
  \item \omterm{def:g10:functions:graph}{منحنى} $f$ فوق كل \omterm{def:g12:deriv:tangent}{مماس} من مماساته.
\end{enumerate}
وإذا كانت $f$ قابلة للاشتقاق مرتين، فهذه تكافئ أيضًا $f'' \geq 0$ على
$I$.
\end{theorem}

\begin{proof}[برهان (2) $\Rightarrow$ (3)]
ثبّت $a \in I$ وضع
$g(x) = f(x) - f(a) - f'(a)(x-a)$، وهي الفجوة بين \omterm{def:g10:functions:graph}{المنحنى} \omterm{def:g12:deriv:tangent}{والمماس}
عند $a$. عندئذٍ تكون $g$ قابلة للاشتقاق و $g'(x) = f'(x) - f'(a)$. وإذا كانت $f'$
\omterm{def:g12:seq:monotonic}{متزايدة}، فإن $g' \leq 0$ على $I \cap \intoc{-\infty}{a}$ و $g' \geq 0$ على
$I \cap \intco{a}{+\infty}$: أي إن $g$ تتناقص قبل $a$ وتتزايد بعدها،
فتبلغ $g$ قيمتها الصغرى $g(a) = 0$، ومنه $g \geq 0$.
أما الاستلزامات الباقية فمقبولة في هذا المستوى؛ والتكافؤ مع
$f'' \geq 0$ ينتج من \cref{thm:g12:deriv:variations} مطبَّقة على $f'$.
\end{proof}

\begin{definition}[نقطة الانعطاف]\label{def:g12:deriv:inflection}
النقطة التي يقطع عندها \omterm{def:g10:functions:graph}{منحنى} $f$ مماسه هي \emph{نقطة
انعطاف}\index{نقطة انعطاف}. ومن أجل $f$ قابلة للاشتقاق مرتين، تكون نقط
الانعطاف هي النقط التي تغيّر عندها $f''$ إشارتها.
\end{definition}

\begin{example}\label{ex:g12:deriv:cube}
من أجل $f(x) = x^3$ يكون $f''(x) = 6x$: فتكون $f$ \omterm{def:g12:deriv:convex}{مقعرة} على $\intoc{-\infty}{0}$،
\omterm{def:g12:deriv:convex}{ومحدبة} على $\intco{0}{+\infty}$، مع \omterm{def:g12:deriv:inflection}{نقطة انعطاف} عند المبدأ،
حيث يقطع \omterm{def:g10:functions:graph}{المنحنى} مماسه (وهو المحور $x$).
\end{example}

\begin{example}[متراجحات التحدب]\label{ex:g12:deriv:ineq}
\omterm{def:g11:func:function}{الدالة} الأسية \omterm{def:g12:deriv:convex}{محدبة} على $\R$ (فمشتقتها الثانية هي نفسها، وهي موجبة).
ومماسها عند $0$ هو $y = 1 + x$، إذن
\[
\eu^x \geq 1 + x \quad \text{من أجل كل } x \in \R .
\]
ومثل هذه \emph{المتراجحات المماسية} ناتج معياري من نواتج \omterm{def:g12:deriv:convex}{التحدب}.
\end{example}

\begin{method}[استعمال التحدب]\label{met:g12:deriv:convexity}
\begin{itemize}
  \item لتبرهن على متراجحة $f(x) \geq ax + b$: تعرّف على المستقيم بوصفه
        مماسًا \omterm{def:g11:func:function}{لدالة} \omterm{def:g12:deriv:convex}{محدبة} $f$ واستدعِ
        \cref{thm:g12:deriv:convexity}(3).
  \item لتحديد نقط الانعطاف: حل $f''(x) = 0$ \emph{ثم} تحقق
        من أن $f''$ تغيّر إشارتها هناك.
  \item وفي \omterm{def:g10:functions:table}{جدول التغيرات}، يشحذ \omterm{def:g12:deriv:convex}{التحدب} رسم \omterm{def:g10:functions:graph}{المنحنى}
        (في أي جهة ينحني).
\end{itemize}
\end{method}

النقطة ذات \omterm{def:g10:coordgeom:system}{الفاصلة} $a = 2$، وحدّد وضعي \omterm{def:g10:functions:graph}{المنحنى}
بالنسبة إلى هذا \omterm{def:g12:deriv:tangent}{المماس}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:deriv:3}
باستعمال تعريف \omterm{def:g12:deriv:derivative}{المشتقة}، احسب
\[
\lim_{x \to 0} \frac{(1+x)^{100} - 1}{x}
\qquad\text{و}\qquad
\lim_{h \to 0} \frac{\sqrt{4+h} - 2}{h}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:deriv:4}
ادرس \omterm{def:g11:func:function}{الدالة} $f(x) = \dfrac{x^2 + 3}{x - 1}$ على \omterm{def:g11:func:function}{مجموعة تعريفها}: النهايات،
\omterm{def:g12:deriv:derivative}{والمشتقة}، \omterm{def:g10:functions:table}{وجدول التغيرات}، والقيم الحدية المحلية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:deriv:5}
تُصنع علبة بلا غطاء من ورقة مقوّى مربعة طول ضلعها
$30\,$cm بقص مربعات متساوية طول ضلعها $x$ من الزوايا وطي
الجوانب إلى أعلى. حدّد $x$ الذي يعظّم حجم العلبة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:deriv:6}
لتكن $f(x) = x^4 - 4x^3 + 10$.
\begin{enumerate}
  \item احسب $f''$ وحدّد فترات \omterm{def:g12:deriv:convex}{تحدب} $f$ ونقط
        انعطافها.
  \item بيّن أن \omterm{def:g12:deriv:tangent}{المماس} عند \omterm{def:g12:deriv:inflection}{نقطة الانعطاف} ذات \omterm{def:g10:coordgeom:system}{الفاصلة} $2$
        يقطع \omterm{def:g10:functions:graph}{المنحنى} هناك.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:deriv:7}
باستعمال متراجحة مماسية، بيّن أنه من أجل كل $x > 0$،
\[
\sqrt{x} \leq \frac{x + 1}{2},
\]
وحدّد متى تتحقق المساواة. (إرشاد: \omterm{def:g11:quad:discriminant}{الجذر} التربيعي \omterm{def:g12:deriv:convex}{دالة مقعرة}.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:deriv:8}
لتكن $f$ \omterm{def:g12:deriv:convex}{محدبة} وقابلة للاشتقاق على $\R$، ولنفترض أن $f'$ تنعدم
عند نقطة $a$ ما. بيّن أن $f(a)$ هي \omterm{def:g10:functions:extrema}{القيمة الصغرى} \emph{الشاملة} \omterm{def:g11:func:function}{للدالة} $f$
على~$\R$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:deriv:9}
بيّن أنه من بين كل المستطيلات ذات المحيط الثابت $p$، للمربع
أكبر مساحة. ثم بيّن أنه من بين كل المستطيلات ذات المساحة الثابتة $A$،
للمربع أصغر محيط، وفسّر كيف ترتبط العبارتان
إحداهما بالأخرى.
\end{exercise}

\section{مسألة: حساب المنقذة}

\begin{problem}[{مسألة نهاية الأسبوع --- أسرع مسار إنقاذ يطيع
قانون سنيل في الضوء، والتحدب يصادق على كل أمثلية، و
علبة المشروب المثلى ارتفاعها يساوي عرضها بالضبط}]
\label{pb:g12:deriv:1}
ترى منقذة سابحًا في ورطة --- في الجهة القطرية من
الشاطئ، بعيدًا في الماء. والركض أسرع من السباحة:
فالخط المستقيم \emph{ليس} أسرع الطرق. والطريق الذي
يصغّر الزمن ينكسر عند خط الشاطئ، وقانون انكساره هو
بالضبط القانون الذي ينكسر به الضوء داخلًا الماء: فالطبيعة
تشتق هي أيضًا. وتدرّب هذه المسألة على قاعدة السلسلة
(\cref{thm:g12:deriv:chain})، وتنفّذ الإنقاذ، وتحصد
متراجحات \omterm{def:g12:deriv:convex}{التحدب} (\cref{thm:g12:deriv:convexity})، ثم
تصمّم علبة صفيح.

\medskip
\noindent\textbf{الجزء الأول --- التمكّن من قاعدة السلسلة.}
\begin{enumerate}
  \item اشتق: $\left(3x^2 + 1\right)^5$؛ \
        $\sqrt{x^2 + 9}$؛ \ $\dfrac{1}{x^2 + 1}$.
  \item أعطِ \omterm{def:g12:deriv:tangent}{المماس} \omterm{def:g10:functions:graph}{لمنحنى} $y = \sqrt{x^2 + 9}$ عند $x = 4$.
  \item ابنِ جدول تغيرات
        $f(x) = \dfrac{x}{x^2 + 1}$ على $\R$ (بما فيه القيم الحدية).
  \item من أجل $g(x) = x^3 - 3x^2 + 4$: \omterm{def:g10:functions:table}{جدول التغيرات} \emph{و}
        جدول \omterm{def:g12:deriv:convex}{التحدب} ($g''$)، مع \omterm{def:g12:deriv:inflection}{نقطة الانعطاف}
        (\cref{def:g12:deriv:inflection}).
  \item احسب \omterm{def:g12:deriv:tangent}{المماس} \omterm{def:g10:functions:graph}{للمنحنى} $g$ عند نقطة انعطافه،
        وبيّن أن $g(x) - (\text{المماس}) = (x - 1)^3$:
        فماذا يقول تغير الإشارة عن كيفية لقاء \omterm{def:g12:deriv:tangent}{المماس}
        \omterm{def:g10:functions:graph}{بالمنحنى} عند انعطاف؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الإنقاذ.}
خط الشاطئ على المحور $x$؛ وتقف المنقذة على الرمل عند
$A(0, 30)$، وينتظر السابح عند $B(40, -20)$ (بالمتر). و
تركض المنقذة بسرعة $5$ m/s على الرمل، وتسبح بسرعة $2$ m/s، وتدخل
الماء عند نقطة $(x, 0)$ من اختيارها.
\begin{enumerate}[resume]
  \item النمذجة: عبّر عن زمن الإنقاذ الكلي $T(x)$، وقل لماذا
        لا يستحق النظر إلا $x \in \intcc{0}{40}$.
  \item اشتق $T$ (وقاعدة السلسلة في العمل --- مرتين).
  \item لتكن $\theta_1$ زاوية مسار الركض مع
        \emph{العمود} على الشاطئ، ولتكن $\theta_2$ زاوية
        مسار السباحة. بيّن أن
        $\sin\theta_1 = \frac{x}{\sqrt{x^2 + 900}}$ وأن
        الشرط $T'(x) = 0$ يُقرأ
        \[
        \frac{\sin\theta_1}{5} = \frac{\sin\theta_2}{2}
        \]
        --- وهو \emph{قانون سنيل}، بالسرعتين محل
        سرعتي الضوء في الهواء والماء.
  \item تقطع القطعة المستقيمة $[AB]$ الشاطئ عند
        $x = 24$. احسب $T(24)$ و $T(40)$ (بالركض على
        الشاطئ ثم السباحة مباشرة) و $T(0)$: فهل الخط المستقيم
        الهندسي هو الأسرع؟ وهل أي من الاستراتيجيتين المتطرفتين
        كذلك؟
  \item حدّد نقطة الدخول المثلى \omterm{thm:g12:limcont:ivt}{بالتنصيف} على
        $T'$ (وهي $\approx 33.7$ m): أعطِ $x^*$ بدقة المتر
        والزمن القياسي $T(x^*)$ بدقة عُشر الثانية.
        فكم يوفّر الانكسار الأمثل بالقياس إلى الخط
        المستقيم؟
  \item لماذا تكون النقطة الحرجة \omterm{def:g10:functions:extrema}{قيمة صغرى} \emph{شاملة}؟
        (فكل حد من حدود $T$ محدب --- واقبل أن مجموع
        دالتين محدبتين محدب --- ثم طبّق
        \cref{exo:g12:deriv:8}.)
  \item ينص مبدأ فيرما على أن الضوء يسلك دائمًا
        مسار أقل \emph{زمن}. استنتج لماذا ينكسر شعاع ضوئي
        عند السطح بالضبط عندما يدخل الماء
        (حيث يكون الضوء أبطأ)، وسمِّ الملاحظة اليومية
        التي يفسّرها هذا عن قلم في كأس ماء.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- حصاد \omterm{def:g12:deriv:convex}{التحدب}.}
\begin{enumerate}[resume]
  \item بيّن أن $x \mapsto x^2$ \omterm{def:g12:deriv:convex}{محدبة}، وبرهن على
        متراجحة \omterm{prop:g10:coordgeom:midpoint}{المنتصف}
        $\left(\frac{a + b}{2}\right)^2 \leq
        \frac{a^2 + b^2}{2}$ مرتين: مرة انطلاقًا من \omterm{def:g12:deriv:convex}{التحدب}، ومرة
        \omterm{def:g10:algebra:expand}{بنشر} $(a - b)^2 \geq 0$.
  \item استنتج أن \emph{\omterm{def:g10:stats:mean}{المتوسط} التربيعي}
        $\sqrt{\frac{a^2 + b^2}{2}}$ يهيمن على \omterm{def:g11:stat:mean}{المتوسط
        الحسابي}، وركّب سلسلة السلسلة الكاملة ---
        التوافقي $\leq$ الهندسي $\leq$ الحسابي $\leq$
        التربيعي --- بالتحقق من الأربعة على $a = 2$ و $b = 8$.
  \item \omterm{def:g12:deriv:tangent}{المماس} تحت \omterm{def:g10:functions:graph}{المنحنى}: \omterm{def:g11:func:function}{الدالة} $x \mapsto \frac1x$ \omterm{def:g12:deriv:convex}{محدبة}
        على $\intoo{0}{+\infty}$؛ اكتب مماسها عند $1$ ثم
        استنتج المتراجحة
        $\frac1x \geq 2 - x$ من أجل كل $x > 0$. وأين تكون
        مساواة؟
  \item الانعطاف في الحياة: خلال وباء، يتبع
        العدد التراكمي للحالات \omterm{def:g10:functions:graph}{منحنى} على شكل حرف S. فماذا
        يحدث عند نقطة انعطافه، ولماذا يترقب علماء الأوبئة ذلك
        التاريخ؟ اربط بإشارة \omterm{def:g12:deriv:derivative}{المشتقة} الثانية على كل جهة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- العلبة المثلى.}
على علبة أسطوانية أن تسع $V = 330$ cm$^3$ بأقل
معدن: فالمساحة $S = 2\pi r^2 + 2\pi r h$، والقيد
$\pi r^2 h = V$.
\begin{enumerate}[resume]
  \item عبّر عن $S(r) = 2\pi r^2 + \dfrac{2V}{r}$، ثم
        اشتق، واحسب نصف القطر الأمثل و
        الارتفاع الأمثل من أجل $V = 330$ (بدقة المليمتر).
  \item برهن على القانون العام الأنيق: عند الأمثلية، يكون
        $h = 2r$ --- أي إن ارتفاع العلبة المثلى يساوي
        عرضها بالضبط.
  \item تحقق من $S''(r) > 0$ واستنتج (حسب \cref{exo:g12:deriv:8}
        مرة أخرى) أن الأمثلية شاملة. أما علب المشروبات
        الحقيقية فأطول من عرضها بوضوح:
        فسمِّ سببًا غير رياضي.
  \item الخاتمة --- خط أنابيب الأمثلة: انمذج، ثم اسأل
        \omterm{def:g12:deriv:derivative}{المشتقة}، ثم حل لإيجاد النقط الحرجة، ثم صادق \omterm{def:g12:deriv:convex}{بالتحدب}
        أو بجدول، ثم فسّر. طبّق القائمة على المنقذة،
        وعلى العلبة، وعلى أقوى عارضة في السنة الماضية
        (\cref{pb:g11:deriv:1}) --- وصُغ ما يقوله مبدأ فيرما
        عن الذي يشغّل هذا الخط أيضًا.
\end{enumerate}
\end{problem}
