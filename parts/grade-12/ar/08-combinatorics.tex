\chapter{التوافيق والعدّ}\label{ch:g12:comb}

التوافيق فنّ العدّ دون سرد. ويكفي مبدآها الابتدائيان
--- اجمع أحجام البدائل المنفصلة، واضرب أعداد
الاختيارات المستقلة --- لعدّ الترتيبات \omterm{def:g12:comb:permutation}{والتبديلات}
وأجزاء مجموعة منتهية، ويبلغان ذروتهما في مبرهنة ثنائي الحد.

\section{مبدآ العدّ}

نكتب $\abs{E}$ للدلالة على عدد عناصر (أي \emph{عدد العناصر}) مجموعة
منتهية $E$.

\begin{proposition}[مبدأ الجمع]\label{prop:g12:comb:addition}
إذا قُسّمت مجموعة منتهية $E$ إلى أجزاء $A_1, \dots, A_k$ (منفصلة
مثنى مثنى، واتحادها $E$)، فإن
\[
\abs{E} = \abs{A_1} + \abs{A_2} + \dots + \abs{A_k}.
\]
\end{proposition}

\begin{proposition}[مبدأ الضرب]\label{prop:g12:comb:multiplication}
إذا بُني شيء بتتابع $k$ اختيارًا، مع $n_1$ إمكانية من أجل
الاختيار الأول، و، \emph{مهما كانت الاختيارات \omterm{def:g10:functions:function}{السابقة}}، $n_i$ إمكانية من أجل
الاختيار رقم $i$، فإن عدد الأشياء المبنية هو
$n_1 \times n_2 \times \dots \times n_k$.
\end{proposition}

\begin{proof}
يُبرهن على العبارتين بالتراجع على $k$؛ والحالة $k = 2$ من
الثانية ترجع إلى عدّ جدول مستطيلي سطرًا سطرًا.
\end{proof}

\begin{example}
مطعم يقدّم 4 مقبلات و 6 أطباق رئيسية و 3 حلويات: أي $4 \times 6 \times 3 =
72$ وجبة مختلفة من ثلاثة أطباق.
\end{example}

\section{القوائم المرتبة والتبديلات والعوامل}

\begin{definition}[القوائم المرتبة ذات $k$ حدًا]\label{def:g12:comb:tuples}
\emph{القائمة المرتبة ذات $k$ حدًا} في مجموعة $E$ هي قائمة مرتبة $(x_1, \dots, x_k)$ من
عناصر $E$، مع السماح بالتكرار. وأما القائمة المرتبة ذات $k$ حدًا وعناصرها
\emph{متمايزة} فهي \emph{ترتيبة}\index{ترتيبة} لعدد $k$ من عناصر $E$.
\end{definition}

\begin{proposition}\label{prop:g12:comb:tuples}
ليكن $\abs E = n$. عدد القوائم المرتبة ذات $k$ حدًا في $E$ هو $n^k$. وعدد
ترتيبات $k$ من عناصر $E$ (حيث $0 \leq k \leq n$) هو
\[
n(n-1)(n-2)\cdots(n-k+1) = \frac{n!}{(n-k)!},
\]
حيث $n! = 1 \times 2 \times \dots \times n$ (مع $0! = 1$) هو
\emph{عاملي}\index{عاملي} العدد $n$.
\end{proposition}

\begin{proof}
بمبدأ الضرب: من أجل قائمة مرتبة ذات $k$ حدًا توجد $n$ إمكانية عند كل خطوة من
الخطوات $k$؛ ومن أجل \omterm{def:g12:comb:tuples}{ترتيبة}، توجد $n$ إمكانية من أجل $x_1$، ثم $n - 1$ من أجل
$x_2$ (لأن عنصرًا استُعمل)، \dots، و $n - k + 1$ من أجل $x_k$.
\end{proof}

\begin{definition}[التبديلة]\label{def:g12:comb:permutation}
\emph{التبديلة}\index{تبديلة} للمجموعة $E$ هي \omterm{def:g12:comb:tuples}{ترتيبة} لعناصر $E$ كلها
البالغ عددها $n$: أي \omterm{def:g10:coordgeom:system}{ترتيب} للمجموعة $E$. وحسب
\cref{prop:g12:comb:tuples} (الحالة $k = n$)، يكون عدد تبديلات مجموعة
فيها $n$ عنصرًا هو $n!$.
\end{definition}

\begin{example}
يمكن لخمسة عدّائين أن ينهوا سباقًا بعدد $5! = 120$ ترتيبًا مختلفًا. وعدد
منصات التتويج الممكنة (المراكز الثلاثة الأولى) هو $5 \times 4 \times 3 = 60$.
\end{example}

\section{التوفيقات والمعاملات الثنائية}

\begin{definition}[التوفيقات]\label{def:g12:comb:combination}
\emph{التوفيقة}\index{توفيقة} لعدد $k$ من عناصر $E$ هي جزء
من $E$ فيه $k$ عنصرًا (بلا \omterm{def:g10:coordgeom:system}{ترتيب} وبلا تكرار). ويُكتب عددها
$\dbinom{n}{k}$\index{معامل ثنائي}، ويُقرأ ``$n$ اختر $k$''.
\end{definition}

\begin{theorem}\label{thm:g12:comb:binomformula}
من أجل $0 \leq k \leq n$:
\[
\binom{n}{k} = \frac{n!}{k!\,(n-k)!} .
\]
\end{theorem}

\begin{proof}
عُدّ ترتيبات $k$ من عناصر $E$ بطريقتين. مباشرةً:
$\frac{n!}{(n-k)!}$. وبطريقة أخرى، اختر أولًا الجزء الحامل
($\binom nk$ طريقة)، ثم رتّبه ($k!$ طريقة)؛ فيعطي مبدأ الضرب
$\binom{n}{k}\,k!$. وبالمساواة بينهما،
$\binom nk = \frac{n!}{k!(n-k)!}$.
\end{proof}

\begin{proposition}[المتطابقات الأساسية]\label{prop:g12:comb:identities}
من أجل $0 \leq k \leq n$:
\[
\binom{n}{0} = \binom{n}{n} = 1, \qquad
\binom{n}{1} = n, \qquad
\binom{n}{k} = \binom{n}{n-k},
\]
و\emph{قاعدة باسكال}\index{مثلث باسكال}: من أجل $1 \leq k \leq n-1$،
\[
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}.
\]
\end{proposition}

\begin{proof}
يصح التناظر $\binom nk = \binom{n}{n-k}$ لأن أخذ المتممات
يقابل الأجزاء ذات $k$ عنصرًا بالأجزاء ذات $(n-k)$ عنصرًا، واحدًا بواحد. وأما
\omterm{prop:g12:comb:identities}{قاعدة باسكال}، فثبّت عنصرًا $a \in E$ ورتّب الأجزاء ذات $k$ عنصرًا في
تلك التي تحوي $a$ --- وتُنال بإضافة $a$ إلى جزء ذي $(k-1)$ عنصرًا
من $E \setminus \{a\}$، وعددها $\binom{n-1}{k-1}$ --- و
تلك التي تتجنب $a$، وهي الأجزاء ذات $k$ عنصرًا من
$E \setminus \{a\}$، وعددها $\binom{n-1}{k}$. ثم اختم بمبدأ
الجمع.
\end{proof}

وتولّد \omterm{prop:g12:comb:identities}{قاعدة باسكال} المعاملات سطرًا بعد سطر --- \emph{\omterm{prop:g11:binom:pascal}{مثلث
باسكال}}: فكل خانة مجموع الخانتين اللتين فوقها.

\begin{theorem}[مبرهنة ثنائي الحد]\label{thm:g12:comb:binomial}
\index{مبرهنة ثنائي الحد}
من أجل كل $a, b \in \R$ (أو $\C$) وكل $n \in \N$:
\[
(a+b)^n = \sum_{k=0}^{n} \binom{n}{k}\, a^{k}\, b^{\,n-k} .
\]
\end{theorem}

\begin{proof}
انشر الجداء $(a+b)(a+b)\cdots(a+b)$ ($n$ عاملًا): فكل حد من
\omterm{def:g10:algebra:expand}{النشر} يختار $a$ أو $b$ في كل عامل، فينتج $a^k b^{n-k}$ حيث
$k$ عدد العوامل التي تسهم بالعنصر $a$. وعدد طرائق اختيار
هذه العوامل $k$ من بين $n$ هو $\binom nk$، وهو إذن
معامل $a^k b^{n-k}$.
\end{proof}

\begin{corollary}\label{cor:g12:comb:sums}
$\displaystyle\sum_{k=0}^{n} \binom{n}{k} = 2^n$
\quad و \quad
$\displaystyle\sum_{k=0}^{n} (-1)^k\binom{n}{k} = 0$ \ (حيث $n \geq 1$).
\end{corollary}

\begin{proof}
خذ $a = b = 1$، ثم $a = -1$ و $b = 1$ في مبرهنة ثنائي الحد. وللمتطابقة
الأولى معنى مباشر أيضًا: فلمجموعة فيها $n$ عنصرًا $2^n$ جزءًا
(إذ كل عنصر داخل أو خارج: بمبدأ الضرب)، مرتَّبةً بالحجم.
\end{proof}

\begin{method}[اختيار النموذج المناسب]\label{met:g12:comb:model}
قبل العدّ، أجب عن سؤالين: \emph{هل \omterm{def:g10:coordgeom:system}{الترتيب} مهم؟} و
\emph{هل التكرار مسموح؟}
\begin{center}
\begin{tabular}{l|cc}
 & \omterm{def:g10:coordgeom:system}{الترتيب} مهم & \omterm{def:g10:coordgeom:system}{الترتيب} غير مهم\\
\hline
التكرار مسموح & $n^k$ (قوائم مرتبة) & (الجامعة)\\
لا تكرار & $\frac{n!}{(n-k)!}$ (ترتيبات) & $\binom nk$ (أجزاء)
\end{tabular}
\end{center}
وأما سحب كرات من كيس: \emph{بالإرجاع وبالترتيب} $\to$ قوائم مرتبة؛
و\emph{بلا إرجاع وبالترتيب} $\to$ ترتيبات؛ و\emph{حفنة دفعة
واحدة} $\to$ أجزاء.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:comb:1}
تتكوّن لوحة تسجيل من 2 حرف (A--Z)، ثم 3 أرقام، ثم 2 حرف.
فكم لوحة ممكنة؟ وكم منها بلا محرف مكرَّر؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:2}
احسب $\dbinom{8}{3}$ و $\dbinom{10}{8}$، وبسّط
$\dfrac{\binom{n}{2}}{\binom{n+1}{2}}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:3}
في قسم فيه 30 تلميذًا، يجب انتخاب لجنة من 4 تلاميذ، ثم
رئيس وأمين مال داخل اللجنة (ولا يمكن لشخص واحد أن يشغل المنصبين
معًا). فكم نتيجة ممكنة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:4}
انشر $(x + 2)^5$ و $(1 - x)^6$ باستعمال مبرهنة ثنائي الحد. وما
معامل $x^3$ في $(2x + 3)^7$؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:5}
تتكوّن يد البوكر المعيارية من 5 أوراق من رزمة فيها 52 ورقة.
\begin{enumerate}
  \item كم يدًا توجد؟
  \item كم يدًا تحوي آصًا واحدًا بالضبط؟ وكم يدًا تحوي آصًا واحدًا على الأقل؟
  \item كم يدًا تكون ``بيتًا كاملًا'' (ثلاث أوراق من رتبة واحدة، واثنتان من
        رتبة أخرى)؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:6}
كم جناسًا (إعادة \omterm{def:g10:coordgeom:system}{ترتيب} للحروف، ذا معنى أو بلا معنى) لكلمة
``حساب''؟ وكم لكلمة ``اللاما''؟
(إرشاد من أجل ``اللاما'': ضع حروف الألف الثلاثة أولًا.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:7}
برهن على المتطابقة $k\dbinom{n}{k} = n\dbinom{n-1}{k-1}$ (حيث $1 \leq k \leq n$)
بطريقتين: بالصيغة \omterm{prop:g12:comb:tuples}{العاملية}، وبعدّ الأزواج
(لجنة من $k$ شخصًا، ورئيسها) المختارة من $n$ شخصًا بطريقتين.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:8}
مسار في المستوي يذهب من $(0,0)$ إلى $(m, n)$ بخطوات واحدية نحو الشرق أو
نحو الشمال. بيّن أن عدد هذه المسارات هو $\dbinom{m+n}{m}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:comb:9}
برهن على \emph{متطابقة فاندرموند}: من أجل $0 \leq k \leq m + n$،
\[
\binom{m+n}{k} = \sum_{j=0}^{k} \binom{m}{j}\binom{n}{k-j},
\]
بعدّ الأجزاء ذات $k$ عنصرًا في مجموعة مقسومة إلى فريق من $m$ و
فريق من $n$. واستنتج أن
$\displaystyle\sum_{j=0}^{n}\binom{n}{j}^{\!2} = \binom{2n}{n}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:comb:10}
باستعمال مبرهنة ثنائي الحد، بيّن أنه من أجل كل $n \geq 1$،
\[
\sum_{k=1}^{n} k \binom{n}{k} = n\,2^{n-1}.
\]
(إرشاد: إما أن تشتق $(1+x)^n$، وإما أن تستعمل \cref{exo:g12:comb:7}.)
\end{exercise}

\section{مسألة: فنّ العدّ مرتين}

\begin{problem}[{مسألة نهاية الأسبوع --- النجوم والقضبان، والقبعات
المشوَّشة، ومتطابقات تُبرهن بعدّ الشيء نفسه بطريقتين}]
\label{pb:g12:comb:1}
أعمق حيلة في التوافيق بسيطة إلى حدّ نزع السلاح: عُدّ
المجموعة نفسها مرتين، بطريقتين مختلفتين، ثم ساوِ
بين الجوابين. وتتدرّب هذه المسألة على نماذج
\cref{met:g12:comb:model}، وتضيف تقنية لم يحتجها درس
الفصل --- \emph{النجوم والقضبان} في عدّ المثلجات ---
ثم تعدّ \emph{القبعات المشوَّشة} الشهيرة عدًّا مضبوطًا،
وتجد العدد $\frac1\eu$ ينتظر في قاع كومة
القبعات، في ظهوره الثالث في هذا الكتاب.

\medskip
\noindent\textbf{الجزء الأول --- اختيار النموذج.}
\begin{enumerate}
  \item عُدّ لوحات التسجيل المكوّنة من $2$ حرف يتبعهما
        $3$ أرقام؛ ثم أجناس كلمة ``اللاما''.
  \item من رزمة فيها $32$ ورقة، عُدّ الأيدي ذات $5$ أوراق؛ ثم
        الأيدي التي تحوي بالضبط $2$ من الآصات $4$.
  \item يمشي روبوت من $(0,0)$ إلى $(4,3)$ مستعملًا خطوات
        واحدية نحو اليمين أو نحو الأعلى فقط: فكم مسارًا؟ (رمّز المسار
        كلمةً بالحرفين R و U.)
  \item انشر $(1 + x)^4$ بمبرهنة ثنائي الحد
        (\cref{thm:g12:comb:binomial})؛ ثم قدّر عند
        $x = 1$ و $x = -1$: فأي متطابقتين بخصوص
        الأعداد $\binom nk$ تسقطان؟
  \item برهن بالعدّ المزدوج على أن
        $k\binom nk = n\binom{n-1}{k-1}$ (بعدّ
        اللجان ذات الرئيس بطريقتين)، واستنتج
        $\sum_{k=0}^{n} k\binom nk = n\,2^{n-1}$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- النجوم والقضبان.}
\begin{enumerate}[resume]
  \item محل مثلجات يبيع $4$ نكهات؛ وأنت تطلب $10$
        كرات (وقد تتكرر النكهات، \omterm{def:g10:coordgeom:system}{وترتيبها} في الكأس
        غير مهم). رمّز الطلب صفًا من $10$ نجوم
        (كرات) يفصلها $3$ قضبان (تغيرات النكهة)، ثم
        عُدّ الطلبات.
  \item عُدّ الثلاثيات من \omterm{def:g10:numbers:sets}{الأعداد الصحيحة} غير السالبة حيث
        $x + y + z = 12$.
  \item عُدّ الثلاثيات من \omterm{def:g10:numbers:sets}{الأعداد الصحيحة} \emph{الموجبة تمامًا} حيث
        $x + y + z = 12$ (عوّض $x = 1 + x'$، وهكذا).
  \item كم حدًا أحاديًا متمايزًا يظهر في \omterm{def:g10:algebra:expand}{نشر}
        $(a + b + c)^5$؟
  \item تحقق من سلامة المنهج: عُدّ طلبات $3$
        كرات من $2$ نكهة بالصيغة، ثم اسردها
        كلها وقارن.
  \item قل بالضبط أين دخلت عبارة ``الكرات متطابقة'' في
        الترميز --- ثم عُدّ ما يحدث بدلًا من ذلك إذا أُكلت
        الكرات \omterm{def:g10:coordgeom:system}{بالترتيب} (بمواضع متمايزة)، بقائمة تحقق
        \cref{met:g12:comb:model}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- القبعات المشوَّشة.}
\emph{التشويش} هو إعادة \omterm{def:g11:prob:rv}{توزيع} $n$ قبعة على أصحابها
$n$ بحيث \emph{لا أحد} ينال قبعته؛ وليكن
$D_n$ عددها. (وقد بيّن \Cref{pb:g11:prob:1} أن ضيفًا واحدًا
وسطيًا يستعيد قبعته --- والآن نعدّ الحفلات المنكوبة
تمامًا عدًّا مضبوطًا.)
\begin{enumerate}[resume]
  \item احسب $D_1$ و $D_2$ و $D_3$ بالسرد، و $D_4$
        بصبر (أو بذكاء).
  \item برّر العلاقة التراجعية
        $D_n = (n - 1)\left(D_{n-1} + D_{n-2}\right)$: فالضيف 1
        ينال قبعةً ما $k \neq 1$ ($n - 1$ اختيارًا)؛ ثم قسّم
        بحسب ما إذا كان الضيف $k$ ينال القبعة 1 أم لا.
        وتحقق من أنها تعيد إنتاج $D_4$، ثم احسب $D_5$.
  \item من أجل $n = 3$، برهن بمبدأ الاحتواء والاستبعاد (بطرح
        التوزيعات التي تثبّت قبعة واحدة على الأقل، ثم إعادة إضافة
        ما عُدّ زيادة) على أن
        $D_3 = 3!\left(1 - \frac{1}{1!} + \frac{1}{2!} -
        \frac{1}{3!}\right)$، ثم صُغ الصيغة العامة.
  \item احسب $\frac{D_5}{5!}$ وقارنه بالمقدار
        $\frac1\eu \approx 0.3679$: \omterm{def:g10:proba:distribution}{فاحتمال} أن تتشوّش حفلة كبيرة
        مخلوطة تشويشًا كاملًا هو
        $\frac1\eu$ --- وهو الظهور الثالث لهذا الثابت، بعد
        اليانصيب والسكرتيرة في \cref{pb:g12:exp:1}.
        (والسبب: أن صيغة السؤال 14 هي بداية متسلسلة
        شهيرة للمقدار $\eu^{-1}$، محكية في الكتب
        الجامعية.)
  \item تبادل الهدايا السري بين $10$ أصدقاء: تُسحب الأسماء
        عشوائيًا بانتظام. فما \omterm{def:g10:proba:distribution}{احتمال} أن تكون السحبة
        صحيحة (ألا يسحب أحد نفسه)، وكم إعادة سحب
        ينبغي للمجموعة أن تتوقع؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- عُدّ مرتين، واربح مرتين.}
\begin{enumerate}[resume]
  \item مبرهنة المصافحات: في أي حفلة، يعدّ جمع عدد
        الأيدي التي صافحها كل ضيف كلَّ مصافحة
        مرتين بالضبط. استنتج أن \emph{عدد الضيوف
        الذين صافحوا عددًا فرديًا من الأيدي زوجي دائمًا} ---
        وتحقق من معقولية الادعاء في حفلة من ثلاثة ضيوف.
  \item برهن على الجوهرة
        $1^3 + 2^3 + \dots + n^3 = (1 + 2 + \dots + n)^2$
        بالتراجع، وتحقق منها من أجل $n = 3$. (فمجموع غاوس
        الصغير، مربَّعًا، يعدّ المكعبات.)
  \item متطابقة فاندرموند (\cref{exo:g12:comb:9}) عبر
        المسارات: فسّر $\binom{2n}{n}$ على أنه مسارات شبكية
        من نمط السؤال 3 من $(0,0)$ إلى $(n,n)$، ثم اقطع كل
        مسار عند عبوره القطر المضاد، وفسّر
        كيف يظهر $\sum_j \binom nj^2$.
  \item الخاتمة --- حركات العدّاد الأربع، سطر واحد لكل حركة مع
        مثال من هذه المسألة: اضرب المراحل واجمع
        الحالات؛ ورمّز بذكاء (النجوم والقضبان، وكلمات المسارات)؛
        وعُدّ الشيء نفسه مرتين (اللجنة ذات الرئيس، والمصافحات)؛
        واطرح غير المرغوب وصحّح ما عُدّ زيادة (التشويشات). ولاحظ أين
        يذهب العدّ إلى العمل تاليًا: في الاحتمالات، وفي مسارات
        فصل المصفوفات والبيانات.
\end{enumerate}
\end{problem}
