\chapter{الدالة الأسية واللوغاريتم}\label{ch:g12:exp}

\omterm{thm:g12:exp:existence}{الدالة الأسية} هي \omterm{def:g11:func:function}{الدالة} الوحيدة المساوية لمشتقتها
والتي تأخذ القيمة $1$ عند $0$. وهي تحوّل المجاميع إلى جداءات؛ أما مقلوبها،
\omterm{def:g12:exp:ln}{اللوغاريتم النيبيري}، فيحوّل الجداءات إلى مجاميع. وهما معًا يصفان
كل ظاهرة يتناسب معدل تغيرها مع حجمها:
التفكك الإشعاعي، ونمو السكان، والفائدة المركبة.

\section{الدالة الأسية}

\begin{theorem}[الوجود والوحدانية]\label{thm:g12:exp:existence}
توجد \omterm{def:g11:func:function}{دالة} وحيدة قابلة للاشتقاق $f \colon \R \to \R$ بحيث
\[
f' = f \qquad\text{و}\qquad f(0) = 1 .
\]
وتُسمّى \emph{الدالة الأسية}\index{دالة أسية} وتُكتب
$\exp$، أو $x \mapsto \eu^x$.
\end{theorem}

\begin{proof}[برهان الوحدانية]
أولًا، \omterm{def:g11:func:function}{دالة} كهذه لا تنعدم أبدًا. وفعلًا، لتكن
$g(x) = f(x)f(-x)$؛ عندئذٍ
\[
g'(x) = f'(x)f(-x) - f(x)f'(-x) = f(x)f(-x) - f(x)f(-x) = 0,
\]
إذن $g$ ثابتة تساوي $g(0) = 1$: أي إنه من أجل كل $x$،
$f(x)f(-x) = 1$، وخاصة $f(x) \neq 0$.

والآن لتكن $f_1, f_2$ حلَّين ولتكن $h = \dfrac{f_1}{f_2}$ (وهذا مشروع
لأن $f_2$ لا تنعدم أبدًا). عندئذٍ
\[
h' = \frac{f_1' f_2 - f_1 f_2'}{f_2^2} = \frac{f_1 f_2 - f_1 f_2}{f_2^2} = 0,
\]
إذن $h$ ثابتة تساوي $h(0) = 1$، \emph{أي} $f_1 = f_2$.

أما \emph{الوجود} فمقبول في هذا المستوى (ويمكن الحصول عليه بواسطة
\omterm{def:g12:seq:sequence}{متتالية} \cref{exo:g12:seq:10}، أو بوصفه مقلوب \omterm{def:g12:exp:ln}{اللوغاريتم}
المبني بالتكامل في \cref{ch:g12:integ}).
\end{proof}

\begin{proposition}[المعادلة الدالية]\label{prop:g12:exp:funceq}
من أجل كل $x, y \in \R$ وكل $n \in \Z$:
\[
\eu^{x+y} = \eu^x\,\eu^y, \qquad
\eu^{-x} = \frac{1}{\eu^x}, \qquad
\eu^{x-y} = \frac{\eu^x}{\eu^y}, \qquad
\bigl(\eu^{x}\bigr)^n = \eu^{nx}.
\]
وفوق ذلك $\eu^x > 0$ من أجل كل $x$.
\end{proposition}

\begin{proof}
ثبّت $y$ واعتبر $\varphi(x) = \dfrac{\exp(x+y)}{\exp(x)\exp(y)}$. فبسطها
ومقامها، بوصفهما دالتين في $x$، حلّان \omterm{def:g10:algebra:equation}{للمعادلة}
$f' = f$ إلى غاية الثابت $\exp(y)$؛ واشتقاق $\varphi$ مباشرة
(بقاعدة خارج القسمة) يعطي $\varphi' = 0$، إذن
$\varphi \equiv \varphi(0) = 1$، وهذا يبرهن على المتطابقة الأولى. وأخذ
$y = -x$ يعطي الثانية (مع القيمة $\eu^0 = 1$)، ومنها تنتج
الثالثة. أما الرابعة فتراجع انطلاقًا من الأولى من أجل $n \geq 0$، ويُمدَّد
إلى $n < 0$ بالثانية.

وأما الموجبية: $\eu^x = \left(\eu^{x/2}\right)^2 \geq 0$ و $\eu^x \neq 0$
(كما بُيّن في \cref{thm:g12:exp:existence})، إذن $\eu^x > 0$.
\end{proof}

\begin{proposition}[التغيرات والنهايات]\label{prop:g12:exp:variations}
\omterm{thm:g12:exp:existence}{الدالة الأسية} \omterm{def:g12:seq:monotonic}{متزايدة} تمامًا \omterm{def:g12:deriv:convex}{ومحدبة}، و
\[
\lim_{x\to-\infty} \eu^x = 0, \qquad
\lim_{x\to+\infty} \eu^x = +\infty .
\]
\end{proposition}

\begin{proof}
$(\exp)' = \exp > 0$ يعطي التزايد التام؛ و $(\exp)'' = \exp > 0$ يعطي
\omterm{def:g12:deriv:convex}{التحدب}. وبالتحدب، $\eu^x \geq 1 + x$ (وهو \omterm{def:g12:deriv:tangent}{المماس} عند $0$، انظر
\cref{ex:g12:deriv:ineq})، إذن $\eu^x \to +\infty$ لما $x \to +\infty$
بالمقارنة. ثم $\eu^{x} = 1/\eu^{-x} \to 0$ لما $x \to -\infty$.
\end{proof}

\begin{theorem}[مقارنة النمو]\label{thm:g12:exp:growth}
\index{مقارنة النمو}
من أجل كل \omterm{def:g10:numbers:sets}{عدد صحيح} $n \geq 1$،
\[
\lim_{x\to+\infty} \frac{\eu^x}{x^n} = +\infty,
\qquad
\lim_{x\to-\infty} x^n \eu^x = 0 .
\]
وبالكلمات: تتغلب \omterm{thm:g12:exp:existence}{الدالة الأسية} على كل قوة للمتغير $x$.
\end{theorem}

\begin{proof}
من أجل $n = 1$: بتطبيق $\eu^t \geq 1 + t \geq t$ عند $t = x/2$،
\[
\frac{\eu^x}{x} = \frac{\left(\eu^{x/2}\right)^2}{x}
\geq \frac{(x/2)^2}{x} = \frac{x}{4} \xrightarrow[x\to+\infty]{} +\infty .
\]
ومن أجل $n$ عام، اكتب
\[
\frac{\eu^x}{x^n}
= \frac{1}{n^n}\left(\frac{\eu^{x/n}}{x/n}\right)^{n}
\xrightarrow[x\to+\infty]{} +\infty
\]
بحكم الحالة $n=1$ والتركيب. وتنتج النهاية عند $-\infty$ بالتعويض
$x \mapsto -x$:
$\abs{x^n \eu^x} = \frac{\abs{x}^n}{\eu^{\abs x}} \to 0$.
\end{proof}

\section{اللوغاريتم النيبيري}

\begin{definition}[اللوغاريتم النيبيري]\label{def:g12:exp:ln}
\omterm{thm:g12:exp:existence}{الدالة الأسية} \omterm{def:g12:limcont:continuity}{متصلة} \omterm{def:g12:seq:monotonic}{ومتزايدة} تمامًا من $\R$ على
$\intoo{0}{+\infty}$؛ فبمبرهنة التقابل
(\cref{thm:g12:limcont:bijection})، يكون \omterm{def:g10:algebra:equation}{للمعادلة} $\eu^x = y$ حل وحيد
من أجل كل $y > 0$. وهذا الحل هو \emph{اللوغاريتم
النيبيري}\index{لوغاريتم} للعدد $y$، ويُكتب $\ln y$. وهكذا
\[
\text{من أجل } x \in \R,\ y > 0: \qquad y = \eu^x \iff x = \ln y .
\]
وخاصة $\ln 1 = 0$ و $\ln \eu = 1$ و
$\eu^{\ln y} = y$ و $\ln(\eu^x) = x$.
\end{definition}

\begin{proposition}[الخواص الجبرية]\label{prop:g12:exp:lnalg}
من أجل كل $a, b > 0$ وكل $n \in \Z$:
\[
\ln(ab) = \ln a + \ln b, \quad
\ln\frac{1}{a} = -\ln a, \quad
\ln\frac{a}{b} = \ln a - \ln b, \quad
\ln(a^n) = n \ln a, \quad
\ln\sqrt{a} = \tfrac12 \ln a .
\]
\end{proposition}

\begin{proof}
$\eu^{\ln a + \ln b} = \eu^{\ln a}\eu^{\ln b} = ab$، وبأخذ $\ln$ للطرفين
نحصل على المتطابقة الأولى. وتنتج الأخريات بالآلية نفسها من
المتطابقات الموافقة في \cref{prop:g12:exp:funceq}.
\end{proof}

\begin{proposition}[الخواص التحليلية]\label{prop:g12:exp:lnanalytic}
\omterm{def:g11:func:function}{الدالة} $\ln$ قابلة للاشتقاق على $\intoo{0}{+\infty}$ ولها
\[
(\ln x)' = \frac{1}{x},
\]
وهي \omterm{def:g12:seq:monotonic}{متزايدة} تمامًا \omterm{def:g12:deriv:convex}{ومقعرة}، و
$\lim\limits_{x\to0^+} \ln x = -\infty$ و
$\lim\limits_{x\to+\infty} \ln x = +\infty$. وفوق ذلك، من أجل كل \omterm{def:g10:numbers:sets}{عدد صحيح}
$n \geq 1$،
\[
\lim_{x\to+\infty} \frac{\ln x}{x^{1/n}} = 0, \qquad
\lim_{x\to0^+} x \ln x = 0 .
\]
\end{proposition}

\begin{proof}
قابلية اشتقاق \omterm{def:g11:func:function}{الدالة} المقلوبة مقبولة في هذا المستوى؛ وبقبولها،
اشتق المتطابقة $\eu^{\ln x} = x$ بقاعدة السلسلة:
$(\ln x)'\,\eu^{\ln x} = 1$، إذن $(\ln x)' = \frac{1}{x} > 0$، ومنه التزايد
التام، و $(\ln x)'' = -\frac1{x^2} < 0$، ومنه التقعر. أما النهايتان عند
$0^+$ و $+\infty$ فتعكسان نهايتي $\exp$ عبر التقابل. وأما
\omterm{thm:g12:exp:growth}{مقارنة النمو}، فعوّض $x = \eu^{t}$:
$\frac{\ln x}{x} = \frac{t}{\eu^t} \to 0$ لما $t \to +\infty$ حسب
\cref{thm:g12:exp:growth}، وبالمثل من أجل النهايات الأخرى مع
$x = \eu^{-t}$، مما يعطي $x\ln x = -t\eu^{-t} \to 0$.
\end{proof}

\begin{method}[حل المعادلات بالدالتين $\exp$ و $\ln$]\label{met:g12:exp:equations}
كلتا الدالتين \omterm{def:g12:seq:monotonic}{متزايدة} تمامًا، فيمكن تطبيقهما على طرفي \omterm{def:g10:algebra:equation}{معادلة} أو متراجحة
(أو حذفهما منهما) دون تغيير الاتجاه:
\[
\eu^{u} = \eu^{v} \iff u = v, \qquad
\eu^{u} \leq \eu^{v} \iff u \leq v,
\]
وبالمثل من أجل $\ln$ على المقادير الموجبة. و\emph{تحقق دائمًا من مجموعات
التعريف أولًا} (\omterm{def:g11:func:function}{فالدالة} $\ln$ تقتضي وسائط موجبة). وأما المعادلات من الشكل
$a^x = b$ حيث $a>0$ و $a \neq 1$، فأعد كتابتها $a^x = \eu^{x\ln a}$ ثم حل
$x = \frac{\ln b}{\ln a}$.
\end{method}

\begin{example}\label{ex:g12:exp:halflife}
تتفكك \omterm{def:g10:proba:sample}{عينة} مشعّة وفق العلاقة $N(t) = N_0\,\eu^{-\lambda t}$. ويحقق
\emph{نصف عمرها} $T$ العلاقة $N(T) = N_0/2$، \emph{أي}
$\eu^{-\lambda T} = \frac12$، إذن $T = \frac{\ln 2}{\lambda}$: فنصف العمر
لا يتعلق بالكمية الابتدائية.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:exp:1}
بسّط
$\dfrac{\eu^{3x}\,\eu^{-x+1}}{\eu^{x}}$ و
$\ln\!\left(\dfrac{\eu^2\sqrt{\eu}}{\eu^{-3}}\right)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:exp:2}
حل في $\R$:
\[
\text{(أ) } \eu^{2x} - 3\eu^{x} + 2 = 0;
\qquad
\text{(ب) } \ln(x - 1) + \ln(x + 2) = \ln 4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:exp:3}
احسب النهايات:
\[
\lim_{x\to+\infty} \frac{\eu^x - x^2}{\eu^x + 1}, \qquad
\lim_{x\to+\infty} \frac{\ln x}{\sqrt x}, \qquad
\lim_{x\to0^+} x^2 \ln x, \qquad
\lim_{x\to+\infty} \bigl(x - \ln x\bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:4}
ادرس \omterm{def:g11:func:function}{الدالة} $f(x) = x\,\eu^{-x}$ على $\R$: التغيرات، والنهايات،
\omterm{def:g12:deriv:extremum}{والقيمة الحدية}؛ وبيّن أن لمنحناها \omterm{def:g12:deriv:inflection}{نقطة انعطاف} وأعطِ
إحداثياتها.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:5}
بيّن أنه من أجل كل $x > -1$، $\ln(1 + x) \leq x$، مع المساواة عند
$x = 0$ فقط. واستنتج أنه من أجل كل $n \geq 1$،
\[
\left(1 + \frac{1}{n}\right)^n \leq \eu \leq \left(1 - \frac{1}{n+1}\right)^{-(n+1)} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:6}
يُستثمر رأس مال $C_0$ بمعدل سنوي قدره $3\%$، والفائدة تُركَّب
كل سنة.
\begin{enumerate}
  \item عبّر عن رأس المال $C_n$ بعد $n$ سنة.
  \item بعد كم سنة يتضاعف رأس المال؟ أعطِ الجواب المضبوط
        باستعمال $\ln$، ثم قيمة عددية.
  \item قارن \omterm{def:g10:numbers:approx}{بالتقريب} ``$70$ مقسومًا على المعدل بالنسبة
        المئوية'' الذي يستعمله المصرفيون.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:7}
حل المتراجحة $\eu^{2x} - \eu^{x+1} > 0$، ثم المتراجحة
$\ln(x^2 - 1) \leq \ln(x + 5)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:exp:8}
لتكن $f(x) = \dfrac{\eu^x}{x}$ من أجل $x > 0$.
\begin{enumerate}
  \item ادرس تغيرات $f$ على $\intoo{0}{+\infty}$ وأعطِ
        قيمتها الصغرى.
  \item من أجل أي قيم للعدد $k$ يكون عدد حلول \omterm{def:g10:algebra:equation}{المعادلة} $\eu^x = kx$
        في $\intoo{0}{+\infty}$ مساويًا $0$ أو $1$ أو $2$؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:exp:9}
من أجل $n \geq 1$، لتكن $u_n = \left(1 + \frac1n\right)^n$.
\begin{enumerate}
  \item باستعمال \cref{exo:g12:exp:5}، بيّن أن $(u_n)$ \omterm{def:g12:seq:bounded}{محدودة من الأعلى}
        بالعدد~$\eu$.
  \item بيّن أن $\ln u_n = n \ln\left(1 + \frac1n\right) \to 1$، ثم
        استنتج أن $u_n \to \eu$.
        (إرشاد: تعرّف على معدل تغير \omterm{def:g11:func:function}{للدالة} $\ln$ عند $1$.)
\end{enumerate}
\end{exercise}

\section{مسألة: اللوغاريتم يروّض العالم}

\begin{problem}[{مسألة نهاية الأسبوع --- أزمنة التضاعف وقاعدة
72، وسلالم الزلازل والأحماض والبيانو، والثابت
$\eu$ يترك بصماته في كل مكان}]
\label{pb:g12:exp:1}
كل ما ينمو بمعدل \emph{نسبة مئوية} ثابت ينمو
نموًا أسيًا --- المدخرات، والبكتيريا، والأوبئة --- وكل ما
يمتد على قوى للعدد عشرة أكثر من أن تُدرَك --- طاقات الزلازل،
والحموضات، وشدّات الصوت --- يروّضه \omterm{def:g12:exp:ln}{اللوغاريتم}. وتحسب هذه
المسألة أزمنة التضاعف وتكشف قناع قاعدة المصرفيين
72، وتقرأ سلالم العالم اللوغاريتمية، وتجمع
البصمات التي يتركها العدد $\eu$ في كل مسرح جريمة
(\cref{prop:g12:exp:funceq} و \cref{thm:g12:exp:growth} و
\cref{met:g12:exp:equations}).

\medskip
\noindent\textbf{الجزء الأول --- التمكّن.}
\begin{enumerate}
  \item حل: $\eu^{2x} = 5$؛ \quad $\ln(3x - 1) = 2$؛ \quad
        $\eu^{2x} - 3\eu^x + 2 = 0$ (وهي تربيعية متنكرة).
  \item بسّط: $\dfrac{\ln 8}{\ln 2}$؛ \quad
        $\ln\!\left(\eu^3 \sqrt{\eu}\right)$؛ \quad
        $\eu^{\ln 5 - \ln 2}$.
  \item برهن على المتراجحة الأم: $\eu^x \geq 1 + x$ من أجل كل
        \omterm{def:g10:numbers:sets}{عدد حقيقي} $x$ (ادرس $f(x) = \eu^x - 1 - x$)، ثم استنتج
        $\ln(1 + u) \leq u$ من أجل $u > -1$.
  \item احسب
        $\lim_{x \to +\infty} x^2 \eu^{-x}$،
        و $\lim_{x \to +\infty} \frac{\ln x}{x}$
        (\cref{thm:g12:exp:growth})، و
        $\lim_{x \to 0} \frac{\eu^x - 1}{x}$ (تعرّف على
        \omterm{def:g12:deriv:derivative}{مشتقة}).
  \item ادرس $f(x) = x \ln x$ على $\intoo{0}{+\infty}$:
        التغيرات، \omterm{def:g10:functions:extrema}{والقيمة الصغرى}، والنهاية عند $0^+$ (وهي مقبولة:
        $x \ln x \to 0$). وهذه \omterm{def:g11:func:function}{الدالة} الصغيرة تقيس
        المعلومة والإنتروبيا عبر الكتب الجامعية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- أزمنة التضاعف وقاعدة
72.}
\begin{enumerate}[resume]
  \item تنمو مدخرات بمعدل $3\,\%$ في السنة. حل
        $1.03^n = 2$: فكم من الوقت يلزم لتضاعف رأس المال؟
  \item يقدّر المصرفيون زمن التضاعف بالمقدار
        $\frac{72}{\text{المعدل }\%}$. اختبر القاعدة عند
        $3\,\%$ و $6\,\%$ و $9\,\%$ في مواجهة القيمة المضبوطة
        $\frac{\ln 2}{\ln(1 + r)}$، ثم فسّرها: فمن أجل $r$
        صغير، $\ln(1 + r) \approx r$ (ومتراجحة السؤال 3
        نصف الحكاية)، إذن يكون الثابت المضبوط هو
        $100 \ln 2 \approx 69.3$ --- فلماذا يفضّل المصرفيون
        $72$؟
  \item تنقسم بكتيريا كل $20$ دقيقة:
        $p(t) = 2^{t/20}$ ($t$ بالدقائق). أعد كتابتها على الصورة
        $\eu^{\lambda t}$، ثم احسب $p$ بعد $24$ ساعة.
        فماذا يثبت الجواب (أكثر من $10^{21}$) عن
        النموذج --- وما الذي يوقف المستعمرات الحقيقية؟
  \item يغادر الكافيين الجسم بنصف عمر يقارب $5$
        ساعات. فبعد قهوة قدرها $100$ mg عند 15:00، كم
        يبقى عند 23:00؟ وفي أي ساعة ينزل تحت
        $10$ mg؟ (فللأرق \omterm{def:g12:exp:ln}{لوغاريتم}.)
  \item أورو واحد بفائدة سنوية قدرها $100\,\%$: احسب
        رأس المال في آخر السنة عند التركيب السنوي والشهري واليومي،
        وأعطِ السقف الذي برهن
        \cref{exo:g12:exp:9} على أنه لا يُكسر. فأي ثابت شهير
        هو نهاية الجشع الخالص؟
  \item لماذا يرسم العلماء $\ln(\text{الحالات})$ في مواجهة
        الزمن خلال الطور المبكر لوباء؟ وماذا يكشف
        الخط المستقيم على ذلك الرسم، وماذا يقيس ميله؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- سلالم العالم اللوغاريتمية.}
(اكتب $\log_{10} x = \frac{\ln x}{\ln 10}$.)
\begin{enumerate}[resume]
  \item مستوى الصوت بالديسيبل: $L = 10 \log_{10}(I/I_0)$. و
        تقيس محادثة $60$ dB، ويقيس حفل روك $120$ dB:
        فبأي معامل تختلف شدّتا الصوت؟
  \item ترتفع درجات الزلازل بمقدار $1$ عندما تُضرب
        السعة الزلزالية في $10$، وتتناسب الطاقة
        المحرَّرة مع السعة$^{3/2}$. قارن
        زلزالًا درجته $5$ بآخر درجته $7$: نسبة السعتين،
        ثم نسبة الطاقتين.
  \item الكيمياء: $\text{pH} = -\log_{10}[\mathrm{H^+}]$.
        عصير الليمون درجة حموضته $2$، والحليب $6.5$: فما
        نسبة تركيزَي الحمض فيهما؟
  \item الموسيقى: كل أوكتاف يضاعف \omterm{def:g10:stats:series}{التواتر}. فمن
        أخفض لا في البيانو ($27.5$ Hz) إلى أعلى دو فيه
        ($4\,186$ Hz)، كم أوكتافًا تمتد لوحة المفاتيح؟
        ولماذا تفضّل حواسنا --- السمع والبصر والإحساس بالزلازل
        --- السلالم اللوغاريتمية؟ (جملة واحدة.)
  \item المسطرة الحاسبة، آلة المهندسين الحاسبة طوال $350$
        سنة: عصوان مدرَّجان بحيث يتناسب \emph{الطول}
        حتى العلامة $x$ مع $\ln x$. فسّر كيف
        يضرب انزلاق عصا على الأخرى الأعداد،
        وسمِّ المتطابقة من
        \cref{prop:g12:exp:lnalg} التي تقوم بالعمل.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- بصمات $\eu$.}
\begin{enumerate}[resume]
  \item انطلاقًا من \cref{exo:g12:exp:8}: \omterm{def:g10:algebra:equation}{للمعادلة}
        $\eu^x = kx$ (حيث $x > 0$) عدد حلول يساوي $0$ أو $1$ أو $2$
        بحسب موضع $k$ بالنسبة إلى
        عتبة. أعد صياغة النتيجة --- وتحقق من
        الواقعة \omterm{def:g12:seq:arith-geom}{الهندسية} وراءها: المستقيم $y = \eu x$ هو
        بالضبط \omterm{def:g12:deriv:tangent}{المماس} \omterm{thm:g12:exp:existence}{للدالة الأسية} المار بالمبدأ.
  \item ثابت الفوز الضائع: مع $n$ تذكرة يانصيب
        \omterm{def:g10:proba:distribution}{احتمال} فوز كل منها $\frac1n$،
        $\P(\text{لا فوز}) = \left(1 - \frac1n\right)^n$.
        احسب نهايته بواسطة
        $n \ln\!\left(1 - \frac1n\right)$ (باستعمال النهاية
        المعيارية $\frac{\ln(1 + u)}{u} \to 1$)، ثم وفّق بينه و
        العدد الغامض $0.368$ في \cref{pb:g11:binom:1}.
  \item قاعدة $37\,\%$: لاختيار الأفضل من بين $n$
        مرشحًا تُجرى مقابلاتهم \omterm{def:g10:coordgeom:system}{بترتيب} عشوائي (بلا رجوع)،
        ترفض الاستراتيجية المثلى أول
        $\frac{n}{\eu} \approx 37\,\%$ منهم ثم تأخذ أول
        مرشح أفضل من كل السابقين --- فتنجح
        \omterm{def:g10:proba:distribution}{باحتمال} يقارب $\frac1\eu$. فمن أين يأتي المقداران
        $\frac1\eu$ في السؤالين 18 و 19 ---
        صُغ الآلية المشتركة (أحداث كثيرة مستقلة
        ضئيلة \omterm{def:g10:proba:distribution}{الاحتمال})، واحسب $\frac1\eu$ بثلاثة
        أرقام عشرية.
  \item الخاتمة --- صورة $\eu$: سقف التركيب
        (السؤال 10)؛ \omterm{def:g11:seq:arithmetic}{والأساس} الذي \omterm{def:g10:reffunc:affine}{ميل} مماسه عند $0$ يساوي
        $1$ بالضبط (السؤالان 3 و 4)؛ والنمو الذي لا
        يلحقه كثير حدود (السؤال 4)؛ وثابت الفوز
        الضائع والتوقف الأمثل (السؤالان 18--19)؛
        ومقلوبه $\ln$، الذي يحوّل الجداءات إلى مجاميع
        (السؤال 16) والعقود إلى بوصات (الجزء الثالث). جملة
        واحدة لكل بند.
\end{enumerate}
\end{problem}
