\chapter{الأعداد العقدية}\label{ch:g12:complex}

ليس \omterm{def:g10:algebra:equation}{للمعادلة} $x^2 = -1$ حل حقيقي. وتوسيع $\R$ بعدد جديد
واحد $\iu$ مربعه $-1$ ينتج حقل $\C$ للأعداد
العقدية، الذي يكون فيه حلول من أجل \emph{كل} \omterm{def:g10:algebra:equation}{معادلة} كثيرة حدود ---
والذي يشفّر حسابه هندسة المستوي: فتصير الانسحابات والدورانات
والتحاكيات جمعًا وضربًا.

\section{الصورة الجبرية}

\begin{definition}[الأعداد العقدية]\label{def:g12:complex:def}
مجموعة \emph{الأعداد العقدية}\index{عدد عقدي} هي
\[
\C = \{a + \iu b : a, b \in \R\},
\]
حيث $\iu$ رمز يخضع للقاعدة الوحيدة $\iu^2 = -1$؛ ويُجرى الجمع
والضرب كما يُجريان على كثيرات الحدود الحقيقية في $\iu$، مع ردّ
$\iu^2$ إلى $-1$. والعددان الحقيقيان $a$ و $b$ هما \emph{الجزء الحقيقي}\index{جزء حقيقي}
$\Rea(z)$ و\emph{الجزء التخيلي}\index{جزء تخيلي} $\Ima(z)$ للعدد $z = a + \iu b$؛ وهذه
العبارة هي \emph{الصورة الجبرية}\index{صورة جبرية} للعدد $z$، وهي وحيدة:
$a + \iu b = a' + \iu b'$ إذا وفقط إذا كان $a = a'$ و $b = b'$.
\end{definition}

\begin{example}
$(2 + 3\iu)(1 - \iu) = 2 - 2\iu + 3\iu - 3\iu^2 = 5 + \iu$.
\end{example}

\begin{definition}[المرافق]\label{def:g12:complex:conjugate}
\emph{مرافق}\index{مرافق} العدد $z = a + \iu b$ هو
$\conj{z} = a - \iu b$.
\end{definition}

\begin{proposition}[خواص الترافق]\label{prop:g12:complex:conjugate}
من أجل كل $z, w \in \C$:
\[
\conj{z + w} = \conj z + \conj w, \quad
\conj{zw} = \conj z\,\conj w, \quad
\conj{\conj z} = z, \quad
z + \conj z = 2\Rea(z), \quad
z - \conj z = 2\iu\,\Ima(z),
\]
و $z\conj z = a^2 + b^2 \geq 0$ من أجل $z = a + \iu b$. وفوق ذلك يكون $z \in \R$
إذا وفقط إذا كان $z = \conj z$.
\end{proposition}

\begin{proof}
كلها حسابات مباشرة انطلاقًا من الصور الجبرية؛ فمثلًا
$z\conj z = (a + \iu b)(a - \iu b) = a^2 - (\iu b)^2 = a^2 + b^2$.
\end{proof}

\begin{method}[قسمة الأعداد العقدية]\label{met:g12:complex:division}
لكتابة خارج قسمة على \omterm{def:g12:complex:def}{الصورة الجبرية}، اضرب البسط والمقام في
\omterm{def:g12:complex:conjugate}{مرافق} المقام:
\[
\frac{1}{2 + \iu} = \frac{2 - \iu}{(2+\iu)(2-\iu)} = \frac{2 - \iu}{5}
= \frac25 - \frac15\iu .
\]
ومنه فلكل \omterm{def:g12:complex:def}{عدد عقدي} غير معدوم مقلوب: أي إن $\C$
\emph{حقل}.
\end{method}

\section{المستوي العقدي والمقياس والعمدة}

\begin{definition}[اللاحقة والمقياس]\label{def:g12:complex:modulus}
في المستوي المزوَّد \omterm{def:g10:coordgeom:system}{بإحداثيات} متعامدة ومتجانسة، تكون النقطة $M(a, b)$
\emph{صورة} العدد $z = a + \iu b$، ويكون $z$ هو \emph{لاحقة}\index{لاحقة} النقطة $M$. و
\emph{مقياس}\index{مقياس} العدد $z$ هو المسافة من $M$ إلى المبدأ:
\[
\abs{z} = \sqrt{a^2 + b^2} = \sqrt{z\conj z}.
\]
\end{definition}

\begin{proposition}[خواص المقياس]\label{prop:g12:complex:modulus}
من أجل $z, w \in \C$:
\[
\abs{zw} = \abs z\,\abs w, \qquad
\abs{\tfrac zw} = \tfrac{\abs z}{\abs w}\ (w \neq 0), \qquad
\abs{\conj z} = \abs z, \qquad
\abs{z + w} \leq \abs{z} + \abs{w} \ \text{(متراجحة المثلث)}.
\]
وفوق ذلك، $\abs{z_B - z_A}$ هو المسافة بين النقطتين ذواتي اللاحقتين
$z_A$ و $z_B$.
\end{proposition}

\begin{proof}
$\abs{zw}^2 = zw\,\conj{zw} = z\conj z\, w \conj w = \abs z^2 \abs w^2$
يعطي الأولى؛ والثانية والثالثة مماثلتان. وأما متراجحة
المثلث:
\[
\abs{z+w}^2 = (z+w)(\conj z + \conj w)
= \abs z^2 + \abs w^2 + 2\Rea(z \conj w)
\leq \abs z^2 + \abs w^2 + 2\abs{z\conj w}
= \bigl(\abs z + \abs w\bigr)^2,
\]
باستعمال $\Rea(u) \leq \abs u$. وأما عبارة المسافة فهي صيغة
فيثاغورس مطبَّقة على \omterm{def:g10:coordgeom:system}{إحداثيات} $z_B - z_A$.
\end{proof}

\begin{definition}[العمدة والصورتان المثلثية والأسية]\label{def:g12:complex:argument}
ليكن $z \neq 0$. \emph{عمدة}\index{عمدة} العدد $z$، وتُكتب
$\Arg(z)$، هي أي قياس $\theta$ للزاوية من المحور الحقيقي الموجب
إلى نصف المستقيم $OM$؛ وهي معرَّفة إلى غاية $2k\pi$. وبكتابة $r = \abs z$،
\[
z = r(\cos\theta + \iu\sin\theta)
\qquad\text{(الصورة المثلثية)}.
\]
وبإدخال الترميز\index{صورة أسية}
\[
\eu^{\iu\theta} = \cos\theta + \iu\sin\theta,
\]
يصير هذا هو \emph{الصورة الأسية} $z = r\,\eu^{\iu\theta}$.
\end{definition}

\begin{theorem}\label{thm:g12:complex:expform}
من أجل كل $\theta, \theta' \in \R$:
\[
\eu^{\iu\theta}\,\eu^{\iu\theta'} = \eu^{\iu(\theta + \theta')},
\qquad
\conj{\eu^{\iu\theta}} = \eu^{-\iu\theta},
\qquad
\abs{\eu^{\iu\theta}} = 1 .
\]
ومنه، من أجل $z = r\eu^{\iu\theta}$ و $z' = r'\eu^{\iu\theta'}$
غير معدومين:
\[
zz' = rr'\,\eu^{\iu(\theta+\theta')},
\qquad
\frac{z}{z'} = \frac{r}{r'}\,\eu^{\iu(\theta - \theta')} :
\]
\emph{فالمقاييس تُضرب، والعمد تُجمع.}
\end{theorem}

\begin{proof}
المتطابقة الأولى \emph{هي} زوج صيغتي الجمع
(\cref{prop:g12:trigo:addition}):
\[
(\cos\theta + \iu\sin\theta)(\cos\theta' + \iu\sin\theta')
= (\cos\theta\cos\theta' - \sin\theta\sin\theta')
+ \iu(\sin\theta\cos\theta' + \cos\theta\sin\theta').
\]
وتنتج البقية بحساب مباشر.
\end{proof}

\begin{corollary}[صيغة دي موافر]\label{cor:g12:complex:demoivre}
\index{صيغة دي موافر}
من أجل $\theta \in \R$ و $n \in \Z$:
$\bigl(\cos\theta + \iu\sin\theta\bigr)^n = \cos n\theta + \iu \sin n\theta$.
\end{corollary}

\begin{proof}
بالتراجع على $n \geq 0$ باستعمال \cref{thm:g12:complex:expform}؛ ثم مدّد إلى
$n < 0$ بأخذ المقلوبات.
\end{proof}

\begin{method}[الانتقال بين الصور]\label{met:g12:complex:forms}
من الجبرية إلى الأسية: احسب $r = \abs z$، ثم أوجد $\theta$ حيث
$\cos\theta = \frac ar$ و $\sin\theta = \frac br$ (وحدّد الربع الصحيح
قبل استدعاء قوس \omterm{def:g12:trigo:cossin}{جيب التمام}). ومن الأسية إلى الجبرية: انشر
$r\cos\theta + \iu\, r\sin\theta$. واستعمل \omterm{def:g12:complex:def}{الصورة الجبرية} من أجل المجاميع، و
\omterm{def:g12:complex:argument}{الصورة الأسية} من أجل الجداءات والقوى وخوارج القسمة.
\end{method}

\begin{example}
$z = 1 + \iu$: $\abs z = \sqrt2$ و $\cos\theta = \sin\theta =
\frac{1}{\sqrt2}$، إذن $\theta = \frac\pi4$ و $z = \sqrt2\,\eu^{\iu\pi/4}$.
ومنه $z^{20} = 2^{10}\,\eu^{5\iu\pi} = -1024$.
\end{example}

\section{معادلات الدرجة الثانية وجذور الوحدة}

\begin{theorem}[المعادلات التربيعية ذات المعاملات الحقيقية]\label{thm:g12:complex:quadratic}
ليكن $a, b, c \in \R$ و $a \neq 0$ و $\Delta = b^2 - 4ac$. \omterm{def:g10:algebra:equation}{للمعادلة}
$az^2 + bz + c = 0$ حلول في $\C$:
\[
\begin{aligned}
&\Delta > 0:\ z = \frac{-b \pm \sqrt{\Delta}}{2a} \ (\text{حلان حقيقيان}); \qquad
\Delta = 0:\ z = \frac{-b}{2a} \ (\text{حل مضاعف}); \\
&\Delta < 0:\ z = \frac{-b \pm \iu\sqrt{-\Delta}}{2a} \ (\text{حلان مترافقان}).
\end{aligned}
\]
\end{theorem}

\begin{proof}
أكمل المربع:
$az^2 + bz + c = a\left(\left(z + \frac{b}{2a}\right)^2 -
\frac{\Delta}{4a^2}\right)$. فإذا كان $\Delta < 0$، فإن
$\frac{\Delta}{4a^2} = \left(\iu\frac{\sqrt{-\Delta}}{2a}\right)^2$، ويُحلَّل
فرق المربعين كالعادة.
\end{proof}

\begin{definition}[جذور الوحدة]\label{def:g12:complex:rootsofunity}
من أجل $n \geq 1$، \emph{جذور الوحدة من الرتبة $n$}\index{جذور الوحدة} هي
حلول \omterm{def:g10:algebra:equation}{المعادلة} $z^n = 1$.
\end{definition}

\begin{theorem}\label{thm:g12:complex:rootsofunity}
\omterm{def:g10:algebra:equation}{للمعادلة} $z^n = 1$ بالضبط $n$ حلًا:
\[
\omega_k = \eu^{2\iu k\pi/n}, \qquad k = 0, 1, \dots, n-1 .
\]
وصورها هي رؤوس مضلع منتظم ذي $n$ ضلعًا مرسوم داخل دائرة
الوحدة، ومجموعها $0$ من أجل $n \geq 2$.
\end{theorem}

\begin{proof}
كل عدد $\omega_k$ يحقق $\omega_k^n = \eu^{2\iu k\pi} = 1$، وهي متمايزة
مثنى مثنى لأن عمدها تقع في $\intco{0}{2\pi}$.
وبالعكس، إذا كان $z^n = 1$ فإن $\abs z^n = 1$، إذن $\abs z = 1$ (لأنه حقيقي
موجب) و $z = \eu^{\iu\theta}$ حيث $n\theta \equiv 0 \pmod{2\pi}$،
\emph{أي} $\theta = \frac{2k\pi}{n}$؛ وردّ $k$ بالباقي حسب $n$ يوقعنا بين
الأعداد $\omega_k$. والرؤوس متباعدة بالتساوي بزاوية $\frac{2\pi}{n}$:
أي مضلع منتظم ذو $n$ ضلعًا. وأخيرًا، مع $\omega = \omega_1$،
يكون $\omega \neq 1$ ويعطي المجموع الهندسي
\[
\sum_{k=0}^{n-1}\omega_k = \sum_{k=0}^{n-1}\omega^k
= \frac{\omega^n - 1}{\omega - 1} = 0 . \qedhere
\]
\end{proof}

$1 + j + j^2 = 0$.
\end{example}

\section{الأعداد العقدية وهندسة المستوي}

\begin{proposition}[القاموس الهندسي]\label{prop:g12:complex:geometry}
لتكن $A, B, C, D$ نقطًا لواحقها $a, b, c, d$ (حيث $a \neq b$
و $c \neq d$).
\begin{enumerate}
  \item الانسحاب \omterm{def:g11:vect:vector}{بالمتجهة} ذات \omterm{def:g12:complex:modulus}{اللاحقة} $t$ هو $z \mapsto z + t$.
  \item الدوران الذي مركزه $\omega$ (لاحقةً) وزاويته $\theta$ هو
        $z \mapsto \omega + \eu^{\iu\theta}(z - \omega)$.
  \item التحاكي الذي مركزه $\omega$ ونسبته
        $k \in \R^*$ هو $z \mapsto \omega + k(z - \omega)$.
  \item للعدد $\dfrac{d - c}{b - a}$ \omterm{def:g12:complex:modulus}{المقياس} $\dfrac{CD}{AB}$ وعمدته
        الزاوية بين المتجهتين $\vect{AB}$ و $\vect{CD}$. وخاصة،
        يكون $AB \perp CD$ إذا وفقط إذا كان $\frac{d-c}{b-a}$ تخيليًا
        صرفًا، وتكون $A, B, C$ على استقامة واحدة (حيث $C \ne A$ و $C\ne B$) إذا
        وفقط إذا كان $\frac{c-a}{b-a} \in \R$.
\end{enumerate}
\end{proposition}

\begin{proof}
تعيد (1) و (3) صياغة الصيغ \omterm{def:g10:coordgeom:system}{بالإحداثيات}. وأما (2)، فالتطبيق
$w \mapsto \eu^{\iu\theta} w$ يضرب المقاييس في $1$ ويضيف $\theta$ إلى
العمد: فهو الدوران بزاوية $\theta$ حول المبدأ؛ والاقتران
بالانسحاب الذي يرسل $\omega$ إلى $0$ يعطي المركز العام. وأما
(4)، فمقاييس خوارج القسمة وعمدها تُطرح
(\cref{thm:g12:complex:expform}).
\end{proof}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g12:complex:1}
اكتب على \omterm{def:g12:complex:def}{الصورة الجبرية}:
$(3 - 2\iu)(1 + 4\iu)$، \quad $\dfrac{2 + \iu}{1 - \iu}$، \quad
$\iu^{2027}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:complex:2}
حل في $\C$: (أ) $z^2 - 4z + 13 = 0$؛ (ب) $z^2 + z + 1 = 0$. وتحقق في كل
حالة من أن الحلين مترافقان واحسب جداءهما.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:complex:3}
ضع على \omterm{def:g12:complex:argument}{الصورة الأسية}: $z_1 = -1 + \iu$ و $z_2 = \sqrt3 - \iu$، ثم احسب
$\dfrac{z_1}{z_2}$ على \omterm{def:g12:complex:argument}{الصورة الأسية} ثم الجبرية. واستنتج القيمة
المضبوطة للمقدار $\cos\dfrac{11\pi}{12}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:4}
صِف هندسيًا مجموعة النقط $M$ ذات \omterm{def:g12:complex:modulus}{اللاحقة} $z$ التي تحقق:
(أ) $\abs{z - 2} = \abs{z + \iu}$؛\quad
(ب) $\abs{z - 1 - \iu} = 3$؛\quad
(ج) $\dfrac{z - 1}{z + 1}$ تخيلي صرف.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:5}
باستعمال صيغة دي موافر ومبرهنة ثنائي الحد، عبّر عن $\cos 3\theta$
بكثير حدود في $\cos\theta$، وعن $\sin 3\theta$ بدلالة
$\sin\theta$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:6}
\emph{(التخطيط.)} باستعمال
$\cos\theta = \frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}$، خطّط
$\cos^3\theta$ (أي اكتبه تركيبًا من المقادير $\cos k\theta$)، واستنتج
$\displaystyle\int_0^{\pi/2}\cos^3\theta\,\dd\theta$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:7}
لتكن $A$ و $B$ ذواتي اللاحقتين $a = 1 + \iu$ و $b = 3 + 2\iu$. حدّد
النقطتين $C$ اللتين تجعلان المثلث $ABC$ متساوي الأضلاع، بوصفهما صورتي $B$
بالدورانين اللذين مركزهما $A$ وزاويتاهما $\pm\frac{\pi}{3}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:8}
حل $z^4 = -4$ في $\C$ وارسم الحلول. وحلّل
$z^4 + 4$ إلى كثيري حدود تربيعيين بمعاملات حقيقية.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:9}
من أجل $\theta \in \intoo{0}{2\pi}$ و $n \in \N$، احسب
\[
S = 1 + \cos\theta + \cos 2\theta + \dots + \cos n\theta
\]
بجمع \omterm{def:g11:seq:geometric}{المتتالية الهندسية}
$\sum_{k=0}^n \eu^{\iu k\theta}$ وأخذ الأجزاء الحقيقية. (والجواب:
$S = \dfrac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}}
\cos\frac{n\theta}{2}$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:10}
ليكن $\omega = \eu^{2\iu\pi/5}$.
\begin{enumerate}
  \item برّر أن $1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$.
  \item ليكن $u = \omega + \omega^4$ و $v = \omega^2 + \omega^3$. بيّن أن
        $u + v = -1$ و $uv = -1$، واستنتج أن $u = \frac{-1+\sqrt5}{2}$.
  \item اختم بأن $\cos\dfrac{2\pi}{5} = \dfrac{\sqrt5 - 1}{4}$.
\end{enumerate}
\end{exercise}

\section{مسألة: المخمَّس في مرآة الوحدة}

\begin{problem}[{مسألة نهاية الأسبوع --- جذور الوحدة الخماسية
تحسب $\cos 72^\circ$ بالضبط، والنسبة الذهبية توقّع
النتيجة، والضرب يتبيّن أنه دوران}]
\label{pb:g12:complex:1}
كان إقليدس يستطيع إنشاء المخمَّس المنتظم، لكن سبب
\emph{استسلامه} للمسطرة والفرجار --- بينما لا تستسلم
الزاوية المتواضعة $20^\circ$ --- بقي خفيًا ألفي
سنة. وهو مختبئ في هذا الفصل: \omterm{def:g12:complex:rootsofunity}{فجذور الوحدة} الخماسية الخمسة
(\cref{thm:g12:complex:rootsofunity}) مجموعها صفر، وذلك السطر
الواحد من الجبر يعصر $\cos 72^\circ$ من \omterm{def:g10:algebra:equation}{معادلة}
تربيعية --- جذور تربيعية فقط، ومنه فهو قابل للإنشاء، وموقَّع،
بالطبع، بالنسبة الذهبية. وحول هذه الجوهرة: التمكّن، وطريق
دي موافر المختصر إلى المثلثات، والضرب مكشوف القناع بوصفه
هندسة.

\medskip
\noindent\textbf{الجزء الأول --- التمكّن.}
\begin{enumerate}
  \item احسب $(2 + \iu)(3 - \iu)$، \
        $\dfrac{1 + \iu}{1 - \iu}$
        (\cref{met:g12:complex:division})، و
        $\abs{3 + 4\iu}$.
  \item ضع على \omterm{def:g12:complex:argument}{الصورة الأسية}: $1 + \iu$؛ \ $-2$؛ \
        $1 - \iu\sqrt3$.
  \item احسب $(1 + \iu)^8$ باستعمال \omterm{def:g12:complex:argument}{الصورة الأسية}.
  \item حل $z^2 = \iu$، ثم $z^2 - 2z + 5 = 0$
        (\cref{thm:g12:complex:quadratic}).
  \item القاموس الهندسي
        (\cref{prop:g12:complex:geometry}): صِف التطبيق
        $z \mapsto \iu z$، والمجموعة
        $\abs{z - (1 + \iu)} = 2$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- خمسة جذور، ومخمَّس واحد.}
ليكن $\omega = \eu^{2\iu\pi/5}$ ولنعتبر
$1, \omega, \omega^2, \omega^3, \omega^4$.
\begin{enumerate}[resume]
  \item تحقق من أن هذه هي بالضبط حلول $z^5 = 1$،
        وصِف الشكل الذي ترسمه في المستوي.
  \item برهن على أن مجموعها صفر. (\omterm{def:g11:seq:geometric}{بمتتالية هندسية}
        --- أو بضرب المجموع في $1 - \omega$.)
  \item خذ الأجزاء الحقيقية واستنتج
        \[
        1 + 2\cos\frac{2\pi}{5} + 2\cos\frac{4\pi}{5} = 0 .
        \]
  \item ضع $c = \cos\frac{2\pi}{5}$ واستعمل صيغة الزاوية المضاعفة
        للمقدار $\cos\frac{4\pi}{5}$ لتحويل السؤال 8
        إلى \omterm{def:g10:algebra:equation}{معادلة} تربيعية في $c$؛ ثم حلّها و
        اختم بأن
        \[
        \cos 72^\circ = \frac{\sqrt5 - 1}{4} .
        \]
  \item تحقق عدديًا، ثم دع النسبة الذهبية توقّع
        عملها: بيّن أن $c = \frac{1}{2\varphi}$ حيث
        $\varphi = \frac{1 + \sqrt5}{2}$
        (\cref{pb:g10:algebra:1}) --- وصُغ الصدى الهندسي
        الشهير: ففي مخمَّس منتظم، يكون القطر
        $\varphi$ ضعف الضلع.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- طرق دي موافر المختصرة.}
\begin{enumerate}[resume]
  \item حكم الإنشاء: لا يحتاج $\cos 72^\circ$ إلا إلى
        جذور تربيعية (السؤال 9)، إذن المخمَّس
        قابل للإنشاء بالمسطرة والفرجار؛ أما $\cos 20^\circ$
        فيحقق تكعيبية لا تُكسر
        (\cref{pb:g12:trigo:1})، إذن لا يمكن تثليث الزاوية
        $60^\circ$. صُغ المحك الناشئ (الجذور
        التربيعية قابلة للبناء، والتكعيبية لا)، ونظريته الكاملة ---
        غاوس في التاسعة عشرة، والمضلع ذو السبعة عشر ضلعًا --- تعيش في
        الكتب الجامعية.
  \item حل $z^4 = -4$ \omterm{def:g12:complex:argument}{بالصورة الأسية} واستعد
        تحليل \cref{exo:g12:complex:8}
        $z^4 + 4 = (z^2 - 2z + 2)(z^2 + 2z + 2)$ بمزاوجة
        الجذور المترافقة.
  \item انشر $(\cos\theta + \iu\sin\theta)^3$ بمبرهنة
        ثنائي الحد وبصيغة دي موافر، واستخرج
        $\cos 3\theta = 4\cos^3\theta - 3\cos\theta$ من جديد في ثلاثة
        أسطر --- وقارن بطريق صيغ الجمع في
        \cref{pb:g12:trigo:1}.
  \item تحقق تحققًا مزدوجًا بواسطة \cref{exo:g12:complex:9}: من أجل
        $\theta = \frac{2\pi}{5}$ و $n = 4$، ماذا يساوي
        المجموع $1 + \cos\theta + \dots + \cos 4\theta$، و
        لماذا يتسق ذلك مع السؤال 8؟
  \item احسب جذور $8\iu$ التكعيبية الثلاثة على \omterm{def:g12:complex:def}{الصورة الجبرية}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الضرب هندسة.}
\begin{enumerate}[resume]
  \item صِف وصفًا كاملًا التطبيق $z \mapsto (1 + \iu)z$:
        فبأي زاوية يدير المستوي، وبأي معامل
        يكبّره، وماذا يفعل بمربع الوحدة؟
  \item برهن على $\abs{z_1 z_2} = \abs{z_1}\,\abs{z_2}$ بواسطة
        المرافقات (\cref{prop:g12:complex:modulus})، ثم
        احسب $(1+\iu)^n$ من أجل $n = 1, 2, 3, 4$ وصِف
        اللولب الذي ترسمه القوى.
  \item العدد $\mathrm{j} = \eu^{2\iu\pi/3}$: بيّن أن
        $1 + \mathrm{j} + \mathrm{j}^2 = 0$، وتحقق على المثلث
        $\left(1, \mathrm{j}, \mathrm{j}^2\right)$ من المحك الكلاسيكي:
        $a + \mathrm{j} b + \mathrm{j}^2 c = 0$ يصح من أجل (توجيه واحد من)
        مثلث متساوي الأضلاع $abc$.
  \item المبرهنة الأساسية في الجبر (وهي مقبولة:
        دالمبير--غاوس): كل كثير حدود يتحلّل تحللًا كاملًا
        على $\C$. استنتج النتيجة الواقعية --- أن كل
        كثير حدود حقيقي يتحلّل إلى قطع خطية وتربيعية حقيقية ---
        وأشر إلى السؤال 12 مثالًا عليها.
  \item الخاتمة --- صورة $\C$، جملة واحدة لكل بند: الأعداد
        تصير نقطًا؛ والضرب يصير
        دورانًا وتكبيرًا؛ \omterm{def:g12:complex:rootsofunity}{وجذور الوحدة} تصير مضلعات
        منتظمة؛ ودي موافر يحوّل القوى إلى متطابقات
        مثلثية؛ والانغلاق الجبري يضمن ألا تحتاج أي
        \omterm{def:g10:algebra:equation}{معادلة} إلى عالم أكبر. والخاتمة الصغرى: أي عددين
        شهيرين التقيا في السؤال 10؟
\end{enumerate}
\end{problem}
