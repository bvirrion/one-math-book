\chapter{Derivación y convexidad}\label{ch:g12:deriv}

La \omterm{def:g12:deriv:derivative}{derivada} mide la tasa de variación instantánea de una \omterm{def:g11:func:function}{función}; es la
\omterm{def:g10:reffunc:affine}{pendiente} de la recta \omterm{def:g12:deriv:tangent}{tangente} a su \omterm{def:g10:functions:graph}{gráfica}. Este capítulo repasa y amplía
las reglas de derivación, relaciona el signo de $f'$ con la variación de $f$
e introduce la \omterm{def:g12:deriv:convex}{convexidad}, gobernada por la \omterm{def:g12:deriv:derivative}{derivada} segunda.

\section{La derivada}

\begin{definition}[Derivada en un punto]\label{def:g12:deriv:derivative}
Sea $f$ definida en un \omterm{def:g10:numbers:interval}{intervalo} $I$ y sea $a \in I$. La \omterm{def:g11:func:function}{función} $f$ es
\emph{derivable en $a$}\index{derivada}\index{derivable} si el cociente
incremental
\[
\frac{f(a+h) - f(a)}{h}
\]
tiene límite finito cuando $h \to 0$. Ese límite se llama \emph{derivada de
$f$ en $a$} y se escribe $f'(a)$. Si $f$ es derivable en todos los puntos
de $I$, la \omterm{def:g11:func:function}{función} $f' \colon x \mapsto f'(x)$ es la \emph{derivada} de
$f$.
\end{definition}

\begin{definition}[Recta tangente]\label{def:g12:deriv:tangent}
Si $f$ es \omterm{def:g12:deriv:derivative}{derivable} en $a$, la \emph{tangente}\index{tangente} a la curva
de $f$ en el punto $(a, f(a))$ es la recta de \omterm{def:g10:algebra:equation}{ecuación}
\[
y = f(a) + f'(a)(x - a).
\]
\end{definition}

\begin{proposition}\label{prop:g12:deriv:diffcont}
Una \omterm{def:g11:func:function}{función} \omterm{def:g12:deriv:derivative}{derivable} en $a$ es \omterm{def:g12:limcont:continuity}{continua} en $a$. El recíproco es falso.
\end{proposition}

\begin{proof}
Para $h \neq 0$ pequeño, $f(a+h) - f(a) = h \cdot \frac{f(a+h)-f(a)}{h}$.
Cuando $h \to 0$, el segundo miembro tiende a $0 \times f'(a) = 0$, luego
$f(a+h) \to f(a)$. Para el recíproco, $x \mapsto \abs{x}$ es \omterm{def:g12:limcont:continuity}{continua} en
$0$, pero su cociente incremental en $0$ vale $\frac{\abs h}{h} = \pm 1$
según el signo de $h$, y no tiene límite.
\end{proof}

\subsection{Reglas de derivación}

\begin{proposition}[Operaciones]\label{prop:g12:deriv:operations}
Sean $u, v$ \omterm{def:g12:deriv:derivative}{derivables} en $I$ y sea $\lambda \in \R$. Entonces $u + v$,
$\lambda u$ y $uv$ son \omterm{def:g12:deriv:derivative}{derivables} en $I$, igual que $\frac{u}{v}$ allí
donde $v \neq 0$, y
\[
(u+v)' = u' + v', \quad (\lambda u)' = \lambda u', \quad
(uv)' = u'v + uv', \quad
\left(\frac{u}{v}\right)' = \frac{u'v - uv'}{v^2}.
\]
\end{proposition}

\begin{proof}[Demostración de la regla del producto]
Escribimos el cociente incremental de $uv$ en $a$ como
\[
\frac{u(a+h)v(a+h) - u(a)v(a)}{h}
= \frac{u(a+h) - u(a)}{h}\,v(a+h) + u(a)\,\frac{v(a+h) - v(a)}{h}.
\]
Cuando $h \to 0$, $v(a+h) \to v(a)$ por \omterm{def:g12:limcont:continuity}{continuidad}
(\cref{prop:g12:deriv:diffcont}), así que el segundo miembro tiende a
$u'(a)v(a) + u(a)v'(a)$. Las demás reglas se demuestran de forma análoga.
\end{proof}

\begin{theorem}[Regla de la cadena]\label{thm:g12:deriv:chain}
\index{regla de la cadena}
Sea $u$ \omterm{def:g12:deriv:derivative}{derivable} en $I$ con valores en $J$ y sea $g$ \omterm{def:g12:deriv:derivative}{derivable} en $J$.
Entonces $g \circ u$ es \omterm{def:g12:deriv:derivative}{derivable} en $I$ y
\[
(g \circ u)' = (g' \circ u) \cdot u' .
\]
En particular, para $u$ \omterm{def:g12:deriv:derivative}{derivable}:
\[
(u^n)' = n\,u'\,u^{n-1} \ (n \in \Z), \qquad
\left(\sqrt{u}\right)' = \frac{u'}{2\sqrt{u}} \ (u > 0), \qquad
\left(\eu^{u}\right)' = u'\,\eu^{u}.
\]
\end{theorem}

\begin{proof}[Esquema de demostración]
Cuando $u(a+h) \neq u(a)$ para $h$ pequeño, escribimos
\[
\frac{g(u(a+h)) - g(u(a))}{h}
= \frac{g(u(a+h)) - g(u(a))}{u(a+h) - u(a)} \cdot \frac{u(a+h) - u(a)}{h}.
\]
Cuando $h \to 0$, $u(a+h) \to u(a)$, así que el primer factor tiende a
$g'(u(a))$ y el segundo, a $u'(a)$. (Una demostración completa tiene que
tratar el caso en que $u(a+h) = u(a)$ para $h$ arbitrariamente pequeño;
ese punto técnico se trata en la universidad.)
\end{proof}

\begin{example}
Las \omterm{def:g12:deriv:derivative}{derivadas} usuales, válidas en sus \omterm{def:g11:func:function}{dominios} naturales:
\[
(x^n)' = nx^{n-1}, \quad
\left(\frac{1}{x}\right)' = -\frac{1}{x^2}, \quad
(\sqrt{x})' = \frac{1}{2\sqrt{x}}, \quad
(\cos x)' = -\sin x, \quad
(\sin x)' = \cos x .
\]
La primera se demuestra por inducción a partir de la regla del producto, y
las dos últimas, en el \cref{ch:g12:trigo}.
\end{example}

\section{Variación y extremos}

\begin{theorem}[Signo de la derivada y variación]\label{thm:g12:deriv:variations}
Sea $f$ \omterm{def:g12:deriv:derivative}{derivable} en un \omterm{def:g10:numbers:interval}{intervalo} $I$.
\begin{enumerate}
  \item $f$ es \omterm{def:g12:seq:monotonic}{creciente} en $I$ si y solo si $f' \geq 0$ en $I$.
  \item $f$ es \omterm{def:g10:functions:variations}{decreciente} en $I$ si y solo si $f' \leq 0$ en $I$.
  \item Si $f' > 0$ en $I$ salvo en un número finito de puntos donde se
        anula, entonces $f$ es estrictamente \omterm{def:g12:seq:monotonic}{creciente} en $I$.
\end{enumerate}
\end{theorem}

\begin{proof}[Demostración parcial]
Si $f$ es \omterm{def:g12:seq:monotonic}{creciente}, todos los cocientes incrementales
$\frac{f(a+h)-f(a)}{h}$ son $\geq 0$, y pasando al límite se obtiene
$f'(a) \geq 0$. Las implicaciones recíprocas se apoyan en el teorema del
valor medio, que se demuestra en la universidad; aquí se \emph{admiten a
este nivel}.
\end{proof}

\begin{definition}[Extremo relativo]\label{def:g12:deriv:extremum}
$f$ tiene un \emph{máximo relativo}\index{extremo} en $a$ si
$f(x) \leq f(a)$ para todo $x$ cercano a $a$; los \omterm{def:g10:functions:extrema}{mínimos} relativos se
definen de forma simétrica.
\end{definition}

\begin{proposition}[Condición de primer orden]\label{prop:g12:deriv:critical}
Si $f$ es \omterm{def:g12:deriv:derivative}{derivable} en un \omterm{def:g10:numbers:interval}{intervalo} \emph{abierto} $I$ y tiene un \omterm{def:g12:deriv:extremum}{extremo}
relativo en $a \in I$, entonces $f'(a) = 0$. El recíproco es falso
($f(x) = x^3$ en $a = 0$).
\end{proposition}

\begin{proof}
Supongamos que $a$ es un \omterm{def:g12:deriv:extremum}{máximo relativo}. Para $h > 0$ pequeño,
$\frac{f(a+h)-f(a)}{h} \leq 0$, así que haciendo $h \to 0^+$ se obtiene
$f'(a) \leq 0$; para $h < 0$ el cociente es $\geq 0$, lo que da
$f'(a) \geq 0$. Por tanto, $f'(a) = 0$.
\end{proof}

\begin{method}[Tabla de variación]\label{met:g12:deriv:table}
Para estudiar una \omterm{def:g11:func:function}{función} $f$: determina su \omterm{def:g11:func:function}{dominio} y sus límites en la
frontera; calcula $f'$ y factorízala; determina el signo de $f'$ en cada
subintervalo; recógelo todo en una \omterm{def:g10:functions:table}{tabla de variación}, marcando los
\omterm{def:g12:deriv:extremum}{extremos}; y deduce el número de soluciones de $f(x) = k$ con el teorema de
la biyección (\cref{thm:g12:limcont:bijection}).
\end{method}

\section{Convexidad}

\begin{definition}[Función convexa]\label{def:g12:deriv:convex}
Una \omterm{def:g11:func:function}{función} $f$ definida en un \omterm{def:g10:numbers:interval}{intervalo} $I$ es
\emph{convexa}\index{convexidad} en $I$ si todas las cuerdas de su \omterm{def:g10:functions:graph}{gráfica}
quedan por encima de la \omterm{def:g10:functions:graph}{gráfica}: para todos $a, b \in I$ y
$t \in \intcc{0}{1}$,
\[
f\bigl((1-t)a + t b\bigr) \leq (1-t) f(a) + t f(b).
\]
Es \emph{cóncava}\index{cóncava} si se cumple la desigualdad contraria
($-f$ convexa).
\end{definition}

\begin{theorem}[Convexidad y derivadas]\label{thm:g12:deriv:convexity}
Sea $f$ \omterm{def:g12:deriv:derivative}{derivable} en un \omterm{def:g10:numbers:interval}{intervalo} $I$. Las afirmaciones siguientes son
equivalentes:
\begin{enumerate}
  \item $f$ es \omterm{def:g12:deriv:convex}{convexa} en $I$;
  \item $f'$ es \omterm{def:g12:seq:monotonic}{creciente} en $I$;
  \item la \omterm{def:g10:functions:graph}{gráfica} de $f$ queda por encima de todas sus \omterm{def:g12:deriv:tangent}{tangentes}.
\end{enumerate}
Si $f$ es dos veces \omterm{def:g12:deriv:derivative}{derivable}, todas ellas equivalen además a
$f'' \geq 0$ en $I$.
\end{theorem}

\begin{proof}[Demostración de (2) $\Rightarrow$ (3)]
Fijemos $a \in I$ y tomemos $g(x) = f(x) - f(a) - f'(a)(x-a)$, la
separación entre la curva y la \omterm{def:g12:deriv:tangent}{tangente} en $a$. Entonces $g$ es \omterm{def:g12:deriv:derivative}{derivable} y
$g'(x) = f'(x) - f'(a)$. Si $f'$ es \omterm{def:g12:seq:monotonic}{creciente}, $g' \leq 0$ en
$I \cap \intoc{-\infty}{a}$ y $g' \geq 0$ en $I \cap \intco{a}{+\infty}$:
$g$ decrece antes de $a$ y crece después, así que $g$ alcanza su \omterm{def:g10:functions:extrema}{mínimo}
$g(a) = 0$, luego $g \geq 0$. Las demás implicaciones se admiten a este
nivel; la equivalencia con $f'' \geq 0$ se sigue del
\cref{thm:g12:deriv:variations} aplicado a $f'$.
\end{proof}

\begin{definition}[Punto de inflexión]\label{def:g12:deriv:inflection}
Un punto donde la curva de $f$ atraviesa su \omterm{def:g12:deriv:tangent}{tangente} es un \emph{punto de
inflexión}\index{punto de inflexión}. Si $f$ es dos veces \omterm{def:g12:deriv:derivative}{derivable}, los
puntos de inflexión son los puntos donde $f''$ cambia de signo.
\end{definition}

\begin{example}\label{ex:g12:deriv:cube}
$f(x) = x^3$ tiene $f''(x) = 6x$: $f$ es \omterm{def:g12:deriv:convex}{cóncava} en $\intoc{-\infty}{0}$ y
\omterm{def:g12:deriv:convex}{convexa} en $\intco{0}{+\infty}$, con un \omterm{def:g12:deriv:inflection}{punto de inflexión} en el origen,
donde la curva atraviesa su \omterm{def:g12:deriv:tangent}{tangente} (el eje $x$).
\end{example}

\begin{example}[Desigualdades por convexidad]\label{ex:g12:deriv:ineq}
La exponencial es \omterm{def:g12:deriv:convex}{convexa} en $\R$ (su \omterm{def:g12:deriv:derivative}{derivada} segunda es ella misma, que
es positiva). Su \omterm{def:g12:deriv:tangent}{tangente} en $0$ es $y = 1 + x$, así que
\[
\eu^x \geq 1 + x \quad \text{para todo } x \in \R .
\]
Estas \emph{desigualdades de la recta \omterm{def:g12:deriv:tangent}{tangente}} son un producto habitual de
la \omterm{def:g12:deriv:convex}{convexidad}.
\end{example}

\begin{method}[Usar la convexidad]\label{met:g12:deriv:convexity}
\begin{itemize}
  \item Para demostrar una desigualdad $f(x) \geq ax + b$: identifica la
        recta como una \omterm{def:g12:deriv:tangent}{tangente} de una $f$ \omterm{def:g12:deriv:convex}{convexa} e invoca el
        \cref{thm:g12:deriv:convexity}(3).
  \item Para localizar \omterm{def:g12:deriv:inflection}{puntos de inflexión}: resuelve $f''(x) = 0$ \emph{y}
        comprueba que $f''$ cambia de signo allí.
  \item En una \omterm{def:g10:functions:table}{tabla de variación}, la \omterm{def:g12:deriv:convex}{convexidad} afina el esbozo de la
        curva (hacia dónde se dobla).
\end{itemize}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:deriv:1}
Deriva las funciones siguientes en sus \omterm{def:g11:func:function}{dominios}:
\[
f(x) = x^3 - 5x + 2, \quad
g(x) = \frac{x}{x^2+1}, \quad
h(x) = (2x+1)^5, \quad
k(x) = \sqrt{x^2 + 1}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:deriv:2}
Da la \omterm{def:g10:algebra:equation}{ecuación} de la \omterm{def:g12:deriv:tangent}{tangente} a la curva de $f(x) = x^2 - 3x + 1$ en el
punto de \omterm{def:g10:coordgeom:system}{abscisa} $a = 2$ y determina las posiciones de la curva respecto de
esa \omterm{def:g12:deriv:tangent}{tangente}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:deriv:3}
Usando la definición de \omterm{def:g12:deriv:derivative}{derivada}, calcula
\[
\lim_{x \to 0} \frac{(1+x)^{100} - 1}{x}
\qquad\text{y}\qquad
\lim_{h \to 0} \frac{\sqrt{4+h} - 2}{h}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:deriv:4}
Estudia la \omterm{def:g11:func:function}{función} $f(x) = \dfrac{x^2 + 3}{x - 1}$ en su \omterm{def:g11:func:function}{dominio}: límites,
\omterm{def:g12:deriv:derivative}{derivada}, \omterm{def:g10:functions:table}{tabla de variación} y \omterm{def:g12:deriv:extremum}{extremos} relativos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:deriv:5}
Se fabrica una caja sin tapa a partir de una lámina cuadrada de cartón de
lado $30\,$cm, recortando cuadrados iguales de lado $x$ en las esquinas y
levantando los laterales. Determina el $x$ que maximiza el volumen de la
caja.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:deriv:6}
Sea $f(x) = x^4 - 4x^3 + 10$.
\begin{enumerate}
  \item Calcula $f''$ y determina los \omterm{def:g10:numbers:interval}{intervalos} de \omterm{def:g12:deriv:convex}{convexidad} y los \omterm{def:g12:deriv:inflection}{puntos
        de inflexión} de $f$.
  \item Demuestra que la \omterm{def:g12:deriv:tangent}{tangente} en el \omterm{def:g12:deriv:inflection}{punto de inflexión} de \omterm{def:g10:coordgeom:system}{abscisa} $2$
        atraviesa allí la curva.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:deriv:7}
Usando una desigualdad de la recta \omterm{def:g12:deriv:tangent}{tangente}, demuestra que para todo
$x > 0$,
\[
\sqrt{x} \leq \frac{x + 1}{2},
\]
e identifica cuándo se da la igualdad. (Indicación: la \omterm{def:g11:quad:discriminant}{raíz} cuadrada es
\omterm{def:g12:deriv:convex}{cóncava}.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:deriv:8}
Sea $f$ \omterm{def:g12:deriv:convex}{convexa} y \omterm{def:g12:deriv:derivative}{derivable} en $\R$, y supongamos que $f'$ se anula en
algún punto $a$. Demuestra que $f(a)$ es el \omterm{def:g10:functions:extrema}{mínimo} \emph{absoluto} de $f$
en~$\R$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:deriv:9}
Demuestra que, de entre todos los rectángulos de perímetro fijo $p$, el
cuadrado es el de mayor área. Demuestra después que, de entre todos los
rectángulos de área fija $A$, el cuadrado es el de menor perímetro, y
explica cómo se relacionan las dos afirmaciones.
\end{exercise}

\section{Problema: El cálculo del socorrista}

\begin{problem}[{Problema de fin de semana --- el rescate más rápido obedece
a la ley de Snell de la luz, la convexidad certifica todos los óptimos y la
lata de refresco ideal es exactamente tan alta como ancha}]
\label{pb:g12:deriv:1}
Un socorrista ve a un bañista en apuros, en diagonal playa abajo y mar
adentro. Correr es más rápido que nadar: la línea recta \emph{no} es el
camino más rápido. El \omterm{def:g10:stats:quartiles}{recorrido} que minimiza el tiempo se quiebra en la
orilla, y su ley de quiebro es exactamente la ley con la que se refracta la
luz al entrar en el agua: la naturaleza también deriva. Este problema
entrena la regla de la cadena (\cref{thm:g12:deriv:chain}), ejecuta el
rescate, cosecha las desigualdades de la \omterm{def:g12:deriv:convex}{convexidad}
(\cref{thm:g12:deriv:convexity}) y diseña una lata.

\medskip
\noindent\textbf{Parte I --- Soltura con la regla de la cadena.}
\begin{enumerate}
  \item Deriva: $\left(3x^2 + 1\right)^5$; \ $\sqrt{x^2 + 9}$; \
        $\dfrac{1}{x^2 + 1}$.
  \item Da la \omterm{def:g12:deriv:tangent}{tangente} a $y = \sqrt{x^2 + 9}$ en $x = 4$.
  \item Construye la \omterm{def:g10:functions:table}{tabla de variación} de $f(x) = \dfrac{x}{x^2 + 1}$ en
        $\R$ (con los \omterm{def:g12:deriv:extremum}{extremos}).
  \item Para $g(x) = x^3 - 3x^2 + 4$: \omterm{def:g10:functions:table}{tabla de variación} \emph{y} tabla de
        \omterm{def:g12:deriv:convex}{convexidad} ($g''$), con el \omterm{def:g12:deriv:inflection}{punto de inflexión}
        (\cref{def:g12:deriv:inflection}).
  \item Calcula la \omterm{def:g12:deriv:tangent}{tangente} a $g$ en su \omterm{def:g12:deriv:inflection}{punto de inflexión} y demuestra que
        $g(x) - (\text{tangente}) = (x - 1)^3$: ¿qué dice el cambio de
        signo sobre cómo se encuentra la \omterm{def:g12:deriv:tangent}{tangente} con la curva en una
        inflexión?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El rescate.}
La orilla va a lo largo del eje $x$; la socorrista está sobre la arena en
$A(0, 30)$ y el bañista espera en $B(40, -20)$ (en metros). La socorrista
corre a $5$ m/s sobre la arena, nada a $2$ m/s y entra en el agua por un
punto $(x, 0)$ que ella elige.
\begin{enumerate}[resume]
  \item Modelo: expresa el tiempo total de rescate $T(x)$ y di por qué solo
        merece la pena considerar $x \in \intcc{0}{40}$.
  \item Deriva $T$ (con la regla de la cadena en acción, dos veces).
  \item Sea $\theta_1$ el ángulo del tramo corriendo con la
        \emph{perpendicular} a la orilla y $\theta_2$ el del tramo a nado.
        Demuestra que $\sin\theta_1 = \frac{x}{\sqrt{x^2 + 900}}$ y que la
        condición $T'(x) = 0$ se lee
        \[
        \frac{\sin\theta_1}{5} = \frac{\sin\theta_2}{2}
        \]
        --- la \emph{ley de Snell}, con las dos velocidades en lugar de las
        velocidades de la luz en el aire y en el agua.
  \item El segmento recto $[AB]$ corta la orilla en $x = 24$. Calcula
        $T(24)$, $T(40)$ (correr por la playa y después nadar en línea
        recta) y $T(0)$: ¿es la recta geométrica lo más rápido? ¿Y alguna
        de las dos estrategias extremas?
  \item Localiza el punto de entrada óptimo por \omterm{thm:g12:limcont:ivt}{dicotomía} sobre $T'$ (está
        en torno a $33.7$ m): da $x^*$ al metro y el tiempo récord
        $T(x^*)$ con una décima de segundo. ¿Cuánto ahorra el quiebro
        óptimo frente a la línea recta?
  \item ¿Por qué el punto crítico es un \omterm{def:g10:functions:extrema}{mínimo} \emph{absoluto}? (Cada
        sumando de $T$ es convexo (admite que una suma de funciones
        \omterm{def:g12:deriv:convex}{convexas} es \omterm{def:g12:deriv:convex}{convexa}) y se aplica el \cref{exo:g12:deriv:8}.)
  \item El principio de Fermat afirma que la luz recorre siempre el camino
        de \emph{tiempo} \omterm{def:g10:functions:extrema}{mínimo}. Deduce por qué un rayo de luz se quiebra
        justo en la superficie al entrar en el agua (donde la luz es más
        lenta) y nombra la observación cotidiana que esto explica sobre un
        lápiz metido en un vaso de agua.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La cosecha de la \omterm{def:g12:deriv:convex}{convexidad}.}
\begin{enumerate}[resume]
  \item Demuestra que $x \mapsto x^2$ es \omterm{def:g12:deriv:convex}{convexa} y demuestra dos veces la
        desigualdad del \omterm{prop:g10:coordgeom:midpoint}{punto medio}
        $\left(\frac{a + b}{2}\right)^2 \leq \frac{a^2 + b^2}{2}$: una vez
        a partir de la \omterm{def:g12:deriv:convex}{convexidad} y otra desarrollando
        $(a - b)^2 \geq 0$.
  \item Deduce que la \emph{\omterm{def:g11:stat:mean}{media} cuadrática}
        $\sqrt{\frac{a^2 + b^2}{2}}$ domina a la \omterm{def:g11:stat:mean}{media} aritmética y monta
        la cadena completa de la serie (armónica $\leq$ geométrica $\leq$
        aritmética $\leq$ cuadrática), comprobando las cuatro con $a = 2$ y
        $b = 8$.
  \item \omterm{def:g12:deriv:tangent}{Tangente} por debajo de la curva: $x \mapsto \frac1x$ es \omterm{def:g12:deriv:convex}{convexa} en
        $\intoo{0}{+\infty}$; escribe su \omterm{def:g12:deriv:tangent}{tangente} en $1$ y deduce la
        desigualdad $\frac1x \geq 2 - x$ para todo $x > 0$. ¿Dónde es una
        igualdad?
  \item Inflexión en estado salvaje: durante una epidemia, el recuento
        acumulado de casos sigue una curva en forma de S. ¿Qué ocurre en su
        \omterm{def:g12:deriv:inflection}{punto de inflexión} y por qué es esa la fecha que vigilan los
        epidemiólogos? Relaciónalo con el signo de la \omterm{def:g12:deriv:derivative}{derivada} segunda a
        cada lado.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La lata ideal.}
Una lata cilíndrica debe contener $V = 330$ cm$^3$ con el menor metal
posible: superficie $S = 2\pi r^2 + 2\pi r h$ con la restricción
$\pi r^2 h = V$.
\begin{enumerate}[resume]
  \item Expresa $S(r) = 2\pi r^2 + \dfrac{2V}{r}$, deriva y calcula el
        radio y la altura óptimos para $V = 330$ (con precisión de
        milímetro).
  \item Demuestra la elegante ley general: en el óptimo, $h = 2r$; la lata
        ideal es exactamente tan alta como ancha.
  \item Comprueba que $S''(r) > 0$ y concluye (otra vez con el
        \cref{exo:g12:deriv:8}) que el óptimo es absoluto. Las latas de
        refresco reales son visiblemente más altas que anchas: da una \omterm{def:g11:seq:geometric}{razón}
        no matemática.
  \item Final: la cadena de montaje de la optimización: modelizar,
        preguntar a la \omterm{def:g12:deriv:derivative}{derivada}, resolver los puntos críticos, certificar
        con \omterm{def:g12:deriv:convex}{convexidad} o con una tabla, e interpretar. Recorre la lista con
        el socorrista, la lata y la viga más resistente del curso pasado
        (\cref{pb:g11:deriv:1}), y di qué afirma el principio de Fermat
        sobre quién más recorre esta cadena.
\end{enumerate}
\end{problem}
