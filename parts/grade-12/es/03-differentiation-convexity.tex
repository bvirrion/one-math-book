\chapter{Derivación y convexidad}\label{ch:g12:deriv}

La derivada Mide la tasa de cambio instantánea de una \omterm{def:g11:func:function}{función}; es
el \omterm{def:g10:reffunc:affine}{pendiente} del \omterm{def:g12:deriv:tangent}{tangente} línea a su grafo. Este capítulo revisa y amplía
las reglas de diferenciación, vincula el signo de $f'$ a las variaciones de $ f $,
y presenta convexidad, que se rige por la segunda derivada.

\section{La derivada}

\begin{definition}[Derivada en un punto]\label{def:g12:deriv:derivative}
Sea $f$ definida en un \omterm{def:g10:numbers:interval}{intervalo} $ I $ y $ a \in I $. La \omterm{def:g11:func:function}{función} $ f $ es
\emph{diferenciable es $ a $}\index{diferenciable en $ a $}\index{diferenciable} si el cociente de \omterm{def:g11:prob:variance}{diferencias}
\[
\frac{f(a+h) - f(a)}{h}
\]
tiene un límite finito como $ h \to 0$. Este límite se llama \emph{derivada de
$ f $ en $ a $} y escrito $ f'(a)$. Si $ f $ es diferenciable en cada punto de
$ I $, la \omterm{def:g11:func:function}{función} $ f' \colon x \mapsto f'(x)$ es el \emph{derivado} de
$ f $.
\end{definition}

\begin{definition}[recta tangente]\label{def:g12:deriv:tangent}
Si $ f $ es \omterm{def:g12:deriv:derivative}{diferenciable} en $ a $, el \emph{tangente}\index{tangente} al
curva de $ f $ en el punto $(a, f(a))$ es la linea de \omterm{def:g10:algebra:equation}{ecuación}
\[
y = f(a) + f'(a)(x - a).
\]
\end{definition}

\begin{proposition}\label{prop:g12:deriv:diffcont}
A \omterm{def:g11:func:function}{función} \omterm{def:g12:deriv:derivative}{diferenciable} en $ a $ es continua en $ a $. Lo contrario es falso.
\end{proposition}

\begin{proof}
Para $ h \neq 0$ pequeño, $ f(a+h) - f(a) = h \cdot \frac{f(a+h)-f(a)}{h}$. Como
$ h \to 0$, el lado derecho tiende a $0 \times f'(a) = 0$, entonces
$ f(a+h) \to f(a)$. Por el contrario, $ x \mapsto \abs{x}$ es continua en
$0$, pero su cociente de \omterm{def:g11:prob:variance}{diferencias} en $0$ es igual $\frac{\abs h}{h} = \pm 1$
dependiendo del signo de $ h $, y no tiene límite.
\end{proof}

\subsection{Reglas de diferenciación}

\begin{proposition}[Operaciones]\label{prop:g12:deriv:operations}
Sea $ u, v $ ser \omterm{def:g12:deriv:derivative}{diferenciable} en $ I $ y $\lambda \in \R $. Entonces $ u + v $,
$\lambda u $, $ uv $ son \omterm{def:g12:deriv:derivative}{diferenciable} en $ I $, como es $\frac{u}{v}$ dónde
$ v \neq 0$, y
\[
(u+v)' = u' + v', \quad (\lambda u)' = \lambda u', \quad
(uv)' = u'v + uv', \quad
\left(\frac{u}{v}\right)' = \frac{u'v - uv'}{v^2}.
\]
\end{proposition}

\begin{proof}[Prueba de la regla del producto]
Escribe el cociente de \omterm{def:g11:prob:variance}{diferencias} de $ uv $ en $ a $ como
\[
\frac{u(a+h)v(a+h) - u(a)v(a)}{h}
= \frac{u(a+h) - u(a)}{h}\,v(a+h) + u(a)\,\frac{v(a+h) - v(a)}{h}.
\]
Como $ h \to 0$, $ v(a+h) \to v(a)$ por continuidad
(\cref{prop:g12:deriv:diffcont}), por lo que el lado derecho tiende a
$ u'(a)v(a) + u(a)v'(a)$. Las demás reglas se demuestran de manera similar.
\end{proof}

\begin{theorem}[regla de la cadena]\label{thm:g12:deriv:chain}
\index{regla de la cadena}
Sea $ u $ ser \omterm{def:g12:deriv:derivative}{diferenciable} en $ I $ con valores en $ J $, y $ g $ \omterm{def:g12:deriv:derivative}{diferenciable}
en $ J $. Entonces $ g \circ u $ es \omterm{def:g12:deriv:derivative}{diferenciable} en $ I $ y
\[
(g \circ u)' = (g' \circ u) \cdot u' .
\]
En particular, para $ u $ \omterm{def:g12:deriv:derivative}{diferenciable}:
\[
(u^n)' = n\,u'\,u^{n-1} \ (n \in \Z), \qquad
\left(\sqrt{u}\right)' = \frac{u'}{2\sqrt{u}} \ (u > 0), \qquad
\left(\eu^{u}\right)' = u'\,\eu^{u}.
\]
\end{theorem}

\begin{proof}[Bosquejo de prueba]
cuando $ u(a+h) \neq u(a)$ para $ h $ pequeño, escribe
\[
\frac{g(u(a+h)) - g(u(a))}{h}
= \frac{g(u(a+h)) - g(u(a))}{u(a+h) - u(a)} \cdot \frac{u(a+h) - u(a)}{h}.
\]
Como $ h \to 0$, $ u(a+h) \to u(a)$, por lo que el primer factor tiende a $ g'(u(a))$ y
el segundo a $ u'(a)$. (Una prueba completa debe manejar el caso donde
$ u(a+h) = u(a)$ para $ h $ arbitrariamente pequeño; Este punto técnico es tratado en
universidad.)
\end{proof}

\begin{example}
lo habitual derivadas, válido en el natural \omterm{def:g11:func:function}{dominios}:
\[
(x^n)' = nx^{n-1}, \quad
\left(\frac{1}{x}\right)' = -\frac{1}{x^2}, \quad
(\sqrt{x})' = \frac{1}{2\sqrt{x}}, \quad
(\cos x)' = -\sin x, \quad
(\sin x)' = \cos x .
\]
El primero se prueba por inducción a partir de la regla del producto, y los dos últimos en
\cref{ch:g12:trigo}.
\end{example}

\section{Variaciones y extremos}

\begin{theorem}[Signo de la derivada y variaciones.]\label{thm:g12:deriv:variations}
Sea $ f $ ser \omterm{def:g12:deriv:derivative}{diferenciable} en un \omterm{def:g10:numbers:interval}{intervalo} $ I $.
\begin{enumerate}
  \item $ f $ es \omterm{def:g12:seq:monotonic}{creciente} en $ I $ si y solo si $ f' \geq 0$ en $ I $.
  \item $ f $ es \omterm{def:g10:functions:variations}{decreciente} en $ I $ si y solo si $ f' \leq 0$ en $ I $.
  \item Si $ f' > 0$ en $ I $ excepto en un número finito de puntos donde
        desaparece, entonces $ f $ es estrictamente \omterm{def:g12:seq:monotonic}{creciente} en $ I $.
\end{enumerate}
\end{theorem}

\begin{proof}[Prueba parcial]
si $ f $ es \omterm{def:g12:seq:monotonic}{creciente}, cada cociente de \omterm{def:g11:prob:variance}{diferencias}
$\frac{f(a+h)-f(a)}{h}$ es $\geq 0$, y pasar al límite da
$ f'(a) \geq 0$. Las implicaciones inversas descansan en la \omterm{def:g11:stat:mean}{media} teorema del valor,
probado en la universidad; ellos son \emph{admitido en este nivel}.
\end{proof}

\begin{definition}[Extremo local]\label{def:g12:deriv:extremum}
$ f $ tiene un \emph{máximo local}\index{máximo local} en $ a $ si $ f(x) \leq f(a)$ para
todos $ x $ cerca $ a $; los \omterm{def:g10:functions:extrema}{mínimos} locales se definen simétricamente.
\end{definition}

\begin{proposition}[Condición de primer orden]\label{prop:g12:deriv:critical}
Si $ f $ es \omterm{def:g12:deriv:derivative}{diferenciable} en un \emph{abierto} \omterm{def:g10:numbers:interval}{intervalo} $ I $ y tiene un local
extremo en $ a \in I $, entonces $ f'(a) = 0$. Lo contrario es falso
($ f(x) = x^3$ en $ a = 0$).
\end{proposition}

\begin{proof}
Decir $ a $ es un \omterm{def:g12:deriv:extremum}{máximo local}. Para $ h > 0$ pequeño,
$\frac{f(a+h)-f(a)}{h} \leq 0$, así que dejar $ h \to 0^+$ da $ f'(a) \leq 0$;
para $ h < 0$ el cociente es $\geq 0$, donación $ f'(a) \geq 0$. Por eso
$ f'(a) = 0$.
\end{proof}

\begin{method}[tabla de variaciones]\label{met:g12:deriv:table}
para estudiar una \omterm{def:g11:func:function}{función} $ f $: determinar su \omterm{def:g11:func:function}{dominio} y límites en la frontera;
calcular $ f'$ y factorizarlo; determinar el signo de $ f'$ en cada subintervalo;
grabar todo en un \omterm{def:g10:functions:table}{tabla de variaciones}, marcando extremos; deducir el número de
soluciones de $ f(x) = k $ usando el teorema de biyección
(\cref{thm:g12:limcont:bijection}).
\end{method}

\section{Convexidad}

\begin{definition}[función convexa]\label{def:g12:deriv:convex}
A \omterm{def:g11:func:function}{función} $ f $ definido en un \omterm{def:g10:numbers:interval}{intervalo} $ I $ es \emph{convexo}\index{convexo}
en $ I $ si cada acorde de su grafo se encuentra por encima del grafo: para todos
$ a, b \in I $ y $ t \in \intcc{0}{1}$,
\[
f\bigl((1-t)a + t b\bigr) \leq (1-t) f(a) + t f(b).
\]
Es \emph{cóncavo}\index{cóncavo} si se cumple la desigualdad inversa ($-f $ convexo).
\end{definition}

\begin{theorem}[Convexidad y derivadas]\label{thm:g12:deriv:convexity}
Sea $ f $ ser \omterm{def:g12:deriv:derivative}{diferenciable} en un \omterm{def:g10:numbers:interval}{intervalo} $ I $. Los siguientes son equivalentes:
\begin{enumerate}
  \item $ f $ es convexa en $ I $;
  \item $ f'$ es \omterm{def:g12:seq:monotonic}{creciente} en $ I $;
  \item el grafo de $ f $ se encuentra encima de cada uno de sus \omterm{def:g12:deriv:tangent}{tangentes}.
\end{enumerate}
Si $ f $ es dos veces \omterm{def:g12:deriv:derivative}{diferenciable}, estos también son equivalentes a $ f'' \geq 0$ en
$ I $.
\end{theorem}

\begin{proof}[Prueba de (2) $\Rightarrow $ (3)]
Arreglar $ a \in I $ y establecer
$ g(x) = f(x) - f(a) - f'(a)(x-a)$, la brecha entre la curva y la \omterm{def:g12:deriv:tangent}{tangente}
en $ a $. Entonces $ g $ es \omterm{def:g12:deriv:derivative}{diferenciable} y $ g'(x) = f'(x) - f'(a)$. Si $ f'$ es
\omterm{def:g12:seq:monotonic}{creciente}, $ g' \leq 0$ en $ I \cap \intoc{-\infty}{a}$ y $ g' \geq 0$ en
$ I \cap \intco{a}{+\infty}$: $ g $ disminuye antes $ a $ y aumenta después,
entonces $ g $ alcanza su \omterm{def:g10:functions:extrema}{mínimo} $ g(a) = 0$, por eso $ g \geq 0$.
Las implicaciones restantes se admiten a este nivel; la equivalencia con
$ f'' \geq 0$ sigue de \cref{thm:g12:deriv:variations} aplicado a $ f'$.
\end{proof}

\begin{definition}[Punto de inflexión]\label{def:g12:deriv:inflection}
Un punto donde la curva de $ f $ cruza su \omterm{def:g12:deriv:tangent}{tangente} es un \emph{inflexión
punto}\index{inflexión
punto}. Para $ f $ dos veces \omterm{def:g12:deriv:derivative}{diferenciable}, inflexión
Los puntos son los puntos donde $ f''$ cambia de signo.
\end{definition}

\begin{example}\label{ex:g12:deriv:cube}
$ f(x) = x^3$ tiene $ f''(x) = 6x $: $ f $ es cóncava en $\intoc{-\infty}{0}$,
convexa en $\intco{0}{+\infty}$, con un punto de inflexión en el origen,
donde la curva cruza su \omterm{def:g12:deriv:tangent}{tangente} (el $ x $-eje).
\end{example}

\begin{example}[Desigualdades de convexidad]\label{ex:g12:deriv:ineq}
La exponencial es convexa en $\R $ (es el segundo derivada es en sí mismo, positivo).
Es \omterm{def:g12:deriv:tangent}{tangente} en $0$ es $ y = 1 + x $, entonces
\[
\eu^x \geq 1 + x \quad \text{para todo } x \in \R .
\]
Semejante \emph{\omterm{def:g12:deriv:tangent}{tangente} desigualdades lineales} son un producto estándar de convexidad.
\end{example}

\begin{method}[Usando convexidad]\label{met:g12:deriv:convexity}
\begin{itemize}
  \item Para demostrar una desigualdad $ f(x) \geq ax + b $: identifica la línea como
        \omterm{def:g12:deriv:tangent}{tangente} de un convexa $ f $ e invocar
        \cref{thm:g12:deriv:convexity}(3).
  \item para localizar puntos de inflexión: resolver $ f''(x) = 0$ \emph{y} comprobar
        eso $ f''$ cambia de cartel allí.
  \item en un \omterm{def:g10:functions:table}{tabla de variaciones}, convexidad refina el boceto de la curva
        (en qué dirección se dobla).
\end{itemize}
\end{method}

\section{Problema: el cálculo del socorrista}

\begin{problem}[{Problema del fin de semana --- la ruta de rescate más rápida obedece
La ley de la luz de Snell, la convexidad certifica cada óptimo, y la
la lata de refresco ideal es exactamente tan alta como ancha}]
\label{pb:g12:deriv:1}
Un socorrista ve a un nadador en problemas, en diagonal por la
playa, en el agua. Correr es más rápido que nadar: el
línea recta es \emph{no} la ruta más rápida. La ruta que
minimiza las curvas de tiempo en la costa, y su ley de flexión es
Precisamente la ley por la que la luz se refracta al entrar en el agua: la naturaleza,
también \omterm{def:g11:prob:variance}{diferencia}. Este problema entrena la regla de la cadena.
(\cref{thm:g12:deriv:chain}), realiza el rescate, cosecha el
desigualdades de convexidad (\cref{thm:g12:deriv:convexity}), y
diseña una lata.

\medskip
\noindent\textbf{Part I --- Chain-rule fluency.}
\begin{enumerate}
  \item Diferenciar: $\left(3x^2 + 1\right)^5$; \
        $\sqrt{x^2 + 9}$; \ $\dfrac{1}{x^2 + 1}$.
