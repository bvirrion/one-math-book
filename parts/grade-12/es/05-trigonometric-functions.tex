\chapter{Funciones trigonométricas}\label{ch:g12:trigo}

Las funciones $\cos$ y $\sin$ convierten la geometría de la circunferencia
en análisis. Definidas enrollando la recta real sobre la circunferencia
unidad, son el arquetipo de las funciones periódicas, y sus \omterm{def:g12:deriv:derivative}{derivadas} las
convierten en soluciones de la \omterm{def:g10:algebra:equation}{ecuación} de las oscilaciones $y'' = -y$.

\section{La circunferencia unidad}

\begin{definition}[Coseno y seno]\label{def:g12:trigo:cossin}
Sea $\mathcal{C}$ la circunferencia de radio 1 centrada en el origen de un
\omterm{def:g10:coordgeom:system}{sistema de coordenadas} \omterm{def:g10:coordgeom:system}{ortonormal}. A cada real $t$ le asociamos el punto
$M(t)$ que se obtiene enrollando una longitud $\abs{t}$ sobre
$\mathcal{C}$ desde el punto $I(1,0)$, en sentido antihorario si
$t \geq 0$ y en sentido horario en caso contrario. Entonces
\emph{$\cos t$ y $\sin t$ son las coordenadas de
$M(t)$}\index{coseno}\index{seno}:
\[
M(t) = (\cos t, \sin t).
\]
\end{definition}

Como la circunferencia mide $2\pi$, el punto $M(t)$ no cambia cuando $t$
aumenta en $2\pi$; y, como $M(t)$ está sobre la circunferencia unidad:

\begin{proposition}[Identidades fundamentales]\label{prop:g12:trigo:identities}
Para todos $t \in \R$ y $k \in \Z$:
\[
\cos^2 t + \sin^2 t = 1, \qquad
\cos(t + 2k\pi) = \cos t, \qquad
\sin(t + 2k\pi) = \sin t,
\]
\[
\cos(-t) = \cos t, \qquad \sin(-t) = -\sin t,
\]
\[
\cos(\pi - t) = -\cos t, \quad \sin(\pi - t) = \sin t, \quad
\cos\left(\tfrac{\pi}{2} - t\right) = \sin t, \quad
\sin\left(\tfrac{\pi}{2} - t\right) = \cos t .
\]
\end{proposition}

\begin{proof}
La primera identidad es la \omterm{def:g10:algebra:equation}{ecuación} de la circunferencia unidad; las demás
expresan las simetrías de la circunferencia: $M(-t)$ es el simétrico de
$M(t)$ respecto del eje $x$, $M(\pi - t)$ lo es respecto del eje $y$ y
$M(\frac\pi2 - t)$, respecto de la recta $y = x$.
\end{proof}

Los valores que hay que saberse de memoria:
\begin{center}
\begin{tabular}{c|ccccc}
$t$ & $0$ & $\dfrac{\pi}{6}$ & $\dfrac{\pi}{4}$ & $\dfrac{\pi}{3}$ & $\dfrac{\pi}{2}$\\[6pt]
\hline\rule{0pt}{14pt}%
$\cos t$ & $1$ & $\dfrac{\sqrt3}{2}$ & $\dfrac{\sqrt2}{2}$ & $\dfrac12$ & $0$\\[6pt]
$\sin t$ & $0$ & $\dfrac12$ & $\dfrac{\sqrt2}{2}$ & $\dfrac{\sqrt3}{2}$ & $1$
\end{tabular}
\end{center}

\begin{proposition}[Fórmulas de adición]\label{prop:g12:trigo:addition}
Para todos $a, b \in \R$:
\begin{align*}
\cos(a+b) &= \cos a \cos b - \sin a \sin b, &
\sin(a+b) &= \sin a \cos b + \cos a \sin b,\\
\cos(a-b) &= \cos a \cos b + \sin a \sin b, &
\sin(a-b) &= \sin a \cos b - \cos a \sin b.
\end{align*}
En particular, $\cos 2a = 2\cos^2 a - 1 = 1 - 2\sin^2 a$ y
$\sin 2a = 2 \sin a \cos a$.
\end{proposition}

\begin{proof}
Los \omterm{def:g11:vect:vector}{vectores} $\vect{u} = (\cos a, \sin a)$ y $\vect{v} = (\cos b, \sin b)$
son unitarios y forman un ángulo $a - b$; su \omterm{def:g11:scal:dot}{producto escalar} vale a la vez
$\cos a\cos b + \sin a \sin b$ (por \omterm{def:g10:coordgeom:system}{coordenadas}) y
$\norm{\vect u}\,\norm{\vect v}\cos(a-b) = \cos(a - b)$ (por geometría),
que es la tercera fórmula. Sustituyendo $b$ por $-b$ se obtiene la primera;
las fórmulas del \omterm{def:g12:trigo:cossin}{seno} se siguen usando $\sin x = \cos(\frac\pi2 - x)$. Las
fórmulas del ángulo doble son el caso $b = a$ combinado con
$\cos^2 + \sin^2 = 1$.
\end{proof}

\section{Estudio analítico}

\begin{lemma}[Límite fundamental]\label{lem:g12:trigo:sinxx}
\[
\lim_{t \to 0} \frac{\sin t}{t} = 1
\qquad\text{y}\qquad
\lim_{t \to 0} \frac{\cos t - 1}{t} = 0 .
\]
\end{lemma}

\begin{proof}
Para $0 < t < \frac{\pi}{2}$, comparamos tres áreas en la circunferencia
unidad: el triángulo $OIM(t)$, el sector circular $OIM(t)$ y el triángulo
rectángulo de base $OI$ y altura $\tan t$:
\[
\frac{\sin t}{2} \leq \frac{t}{2} \leq \frac{\tan t}{2}.
\]

  \draw[->] (0,-0.25) -- (0,1.45) node[above] {$y$};

  \draw[omDef] (0,0) -- (1,0) -- (50:1) -- cycle;

  \node[left, font=\small] at (0.68,0.35) {$\sin t$};
  \node[omProp, right, font=\small] at (1,0.62) {$\tan t$};

\end{tikzpicture}

{\small Las tres áreas encajadas: el triángulo $OIM$ (en azul, de área
$\frac{\sin t}{2}$), el sector circular (limitado por el arco rojo, de área
$\frac{t}{2}$) y el triángulo rectángulo de altura $\tan t$ (en naranja, de
área $\frac{\tan t}{2}$).}
\end{omfigure}

La primera desigualdad da $\frac{\sin t}{t} \leq 1$; la segunda da
$\cos t \leq \frac{\sin t}{t}$. Cuando $t \to 0^+$, $\cos t \to 1$ (por
\omterm{def:g12:limcont:continuity}{continuidad}), y el teorema del sándwich da $\frac{\sin t}{t} \to 1$; la
paridad lo extiende a $t \to 0$. Para el segundo límite,
\[
\frac{\cos t - 1}{t} = \frac{\cos^2 t - 1}{t(\cos t + 1)}
= -\frac{\sin t}{t}\cdot\frac{\sin t}{\cos t + 1}
\xrightarrow[t\to0]{} -1 \cdot \frac{0}{2} = 0. \qedhere
\]
\end{proof}

\begin{theorem}[Derivadas del seno y del coseno]\label{thm:g12:trigo:derivatives}
Las funciones $\sin$ y $\cos$ son \omterm{def:g12:deriv:derivative}{derivables} en $\R$, con
\[
(\sin)' = \cos, \qquad (\cos)' = -\sin .
\]
Más en general, $(\sin(ax+b))' = a\cos(ax+b)$ y
$(\cos(ax+b))' = -a\sin(ax+b)$.
\end{theorem}

\begin{proof}
Por las fórmulas de adición,
\[
\frac{\sin(x+h) - \sin x}{h}
= \sin x\,\frac{\cos h - 1}{h} + \cos x\,\frac{\sin h}{h}
\xrightarrow[h\to0]{} \sin x \cdot 0 + \cos x \cdot 1 = \cos x,
\]
usando el \cref{lem:g12:trigo:sinxx}. El cálculo para $\cos$ es idéntico, y
la forma general se sigue de la regla de la cadena.
\end{proof}

\begin{proposition}[Variación]\label{prop:g12:trigo:variations}
En $\intcc{0}{\pi}$, $\cos$ decrece de $1$ a $-1$. En
$\intcc{-\frac\pi2}{\frac\pi2}$, $\sin$ crece de $-1$ a $1$. Las dos
funciones son $2\pi$-periódicas y tienen \omterm{def:g10:stats:quartiles}{recorrido} $\intcc{-1}{1}$; $\cos$
es par y $\sin$ es \omterm{def:g11:func:parity}{impar}.
\end{proposition}

\begin{proof}
En $\intoo{0}{\pi}$, $(\cos)' = -\sin < 0$, porque allí el punto $M(t)$
tiene ordenada positiva; y análogamente $(\sin)' = \cos > 0$ en
$\intoo{-\frac\pi2}{\frac\pi2}$. La periodicidad y la paridad vienen de la
\cref{prop:g12:trigo:identities}.
\end{proof}

\begin{method}[Resolver $\cos x = a$ y $\sin x = a$]\label{met:g12:trigo:equations}
Para $a \in \intcc{-1}{1}$, halla una solución particular $\alpha$ (de la
tabla de valores o con la calculadora). Entonces todas las soluciones son:
\[
\cos x = \cos\alpha \iff x = \alpha + 2k\pi \text{ o } x = -\alpha + 2k\pi,
\quad k \in \Z;
\]
\[
\sin x = \sin\alpha \iff x = \alpha + 2k\pi \text{ o } x = \pi - \alpha + 2k\pi,
\quad k \in \Z.
\]
Para las inecuaciones, localiza los arcos solución sobre la circunferencia
unidad y lee los \omterm{def:g10:numbers:interval}{intervalos} dentro de un periodo.
\end{method}

\begin{example}\label{ex:g12:trigo:equation}
Resolvamos $\cos x = \frac12$ en $\intoc{-\pi}{\pi}$: una solución
particular es $\frac\pi3$, así que las soluciones son $x = \frac\pi3$ y
$x = -\frac\pi3$. En todo $\R$: $x = \pm\frac\pi3 + 2k\pi$, $k \in \Z$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:trigo:1}
Calcula $\cos\frac{2\pi}{3}$, $\sin\frac{5\pi}{6}$, $\cos\frac{7\pi}{4}$ y
$\sin\left(-\frac{\pi}{3}\right)$ usando las identidades de la
\cref{prop:g12:trigo:identities}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:trigo:2}
Deriva $f(x) = \sin^2 x$, $g(x) = \cos(3x) + x\sin x$ y
$h(x) = \dfrac{\sin x}{2 + \cos x}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:trigo:3}
Resuelve en $\intoc{-\pi}{\pi}$ y después en $\R$:
\[
\text{(a) } \sin x = \frac{\sqrt3}{2};
\qquad
\text{(b) } \cos\left(2x\right) = \frac{\sqrt2}{2} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:4}
Usando las fórmulas de adición, calcula el valor exacto de
$\cos\frac{\pi}{12}$ y de $\sin\frac{\pi}{12}$. (Indicación:
$\frac{\pi}{12} = \frac{\pi}{3} - \frac{\pi}{4}$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:5}
Resuelve en $\intcc{0}{2\pi}$ la inecuación
$2\sin^2 x - \sin x - 1 \geq 0$. (Indicación: factoriza el polinomio de
segundo grado en $\sin x$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:6}
Calcula los límites
\[
\lim_{x\to0} \frac{\sin 3x}{x}, \qquad
\lim_{x\to0} \frac{1 - \cos x}{x^2}, \qquad
\lim_{x\to+\infty} x \sin\frac{1}{x}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:trigo:7}
Estudia la \omterm{def:g11:func:function}{función} $f(x) = x + \cos x$ en $\intcc{0}{2\pi}$: variación y
\omterm{def:g12:deriv:extremum}{extremos}. Demuestra que $f$ es \omterm{def:g12:seq:monotonic}{creciente} en $\R$ aunque $f'$ se anule en
infinitos puntos.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:trigo:8}
Sea $f(x) = A\cos x + B \sin x$ con $(A,B) \neq (0,0)$.
\begin{enumerate}
  \item Demuestra que $f$ se puede escribir $f(x) = R\cos(x - \varphi)$ con
        $R = \sqrt{A^2 + B^2}$ y un $\varphi$ adecuado.
  \item Deduce el \omterm{def:g10:functions:extrema}{máximo} y el \omterm{def:g10:functions:extrema}{mínimo} de $f$ y resuelve
        $\cos x + \sin x = 1$ en $\intoc{-\pi}{\pi}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:trigo:9}
Demuestra que $\sin$ y $\cos$ cumplen las dos la \omterm{def:g10:algebra:equation}{ecuación} diferencial
$y'' = -y$. Recíprocamente, sea $y$ una solución de $y'' = -y$ con
$y(0) = a$ e $y'(0) = b$; demuestra que $y = a\cos + b\sin$. (Indicación:
considera $g = (y - a\cos - b\sin)$ y la ``energía''
$E = g^2 + (g')^2$.)
\end{exercise}

\section{Problema: El secreto del péndulo}

\begin{problem}[{Problema de fin de semana --- un límite geométrico mueve
las derivadas del seno y del coseno, y de ellas sale toda la oscilación:
muelles, péndulos, pulsaciones y el metro que estuvo a punto de ser}]
\label{pb:g12:trigo:1}
Todos los relojes que han hecho tictac y todas las cuerdas que han sonado
obedecen a las mismas matemáticas: una fuerza recuperadora proporcional al
desplazamiento y, por tanto, la \omterm{def:g10:algebra:equation}{ecuación} $y'' = -\omega^2 y$ y, por tanto,
\omterm{def:g12:trigo:cossin}{cosenos}. En la base de todo ello se sienta un límite de aspecto inocente,
$\frac{\sin x}{x} \to 1$, prometido cursos atrás, usado por el
\cref{thm:g12:trigo:derivatives} y demostrado aquí con un dibujo. Este
problema sube desde ese límite hasta el vaivén de un péndulo de un metro y
explica por qué el metro no está definido por él, por muy poco.

\medskip
\noindent\textbf{Parte I --- El límite fundamental.}
\begin{enumerate}
  \item Calentamiento numérico: calcula $\frac{\sin x}{x}$ para $x = 0.1$ y
        $x = 0.01$ (¡en \omterm{def:g11:trigo:radian}{radianes}!).
  \item El sándwich, en la circunferencia unidad con
        $0 < x < \frac\pi2$: expresa por geometría elemental las áreas de
        (i) el triángulo de vértices $O$, $A(1,0)$ y el punto de la
        circunferencia $M(\cos x, \sin x)$; (ii) el sector circular $OAM$;
        (iii) el triángulo rectángulo $OAT$ con $T(1, \tan x)$.
  \item Del sándwich de áreas deduce $\sin x < x < \tan x$ y después
        $\cos x < \frac{\sin x}{x} < 1$, y concluye con el teorema del
        sándwich que $\lim_{x \to 0} \frac{\sin x}{x} = 1$ (la paridad se
        ocupa de $x < 0$).
  \item Deduce $\lim_{x \to 0} \frac{1 - \cos x}{x^2} = \frac12$
        (multiplica por el conjugado).
  \item Explica por qué este límite es el motor del
        \cref{thm:g12:trigo:derivatives} (¿cuánto vale $\sin'(0)$?) y por
        qué salda además una vieja deuda: ¿en qué sentido son los
        \emph{\omterm{def:g11:trigo:radian}{radianes}} la única unidad de ángulo con la que
        $\sin' = \cos$ se cumple sin constantes?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Movimiento armónico.}
\begin{enumerate}[resume]
  \item Comprueba con la regla de la cadena que
        $y(t) = R\cos(\omega t - \varphi)$ cumple $y'' = -\omega^2 y$.
  \item Una masa sujeta a un muelle cumple $y'' = -4y$, con $y(0) = 3$ e
        $y'(0) = 8$. Admitiendo (por el \cref{exo:g12:trigo:9}) que todas
        las soluciones son $A\cos 2t + B\sin 2t$, determina $A$ y $B$.
  \item Reescribe la solución como $R\cos(2t - \varphi)$
        (\cref{exo:g12:trigo:8}): da la amplitud $R$, el periodo y la fase
        $\varphi$ (con tres decimales).
  \item ¿En qué instante pasa la masa por primera vez por la posición de
        equilibrio $y = 0$?
  \item El péndulo: una lenteja colgada de un hilo de longitud $L$ cumple
        \emph{exactamente} $\theta'' = -\frac gL \sin\theta$, que no se
        resuelve con funciones elementales. Para oscilaciones pequeñas,
        sustituye $\sin\theta$ por $\theta$ (¡la Parte I lo justifica!) y
        deduce el periodo $T = 2\pi\sqrt{\frac Lg}$. Calcula $T$ para
        $L = 1$ m ($g = 9.81$): el famoso descubrimiento de Galileo. ¿De
        qué \emph{no} depende la fórmula?
  \item El péndulo de segundos: ¿qué longitud da exactamente $T = 2$ s? La
        Academia de Ciencias francesa de 1791 dudó entre esa longitud y la
        diezmillonésima parte del cuadrante del meridiano para definir el
        \emph{metro}: ¿en cuántos milímetros se diferencian los dos
        candidatos?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Las fórmulas de adición en acción.}
\begin{enumerate}[resume]
  \item Demuestra con las fórmulas de adición
        (\cref{prop:g12:trigo:addition}) que
        $\cos\left(x + \frac\pi2\right) = -\sin x$ y
        $\sin\left(x + \frac\pi2\right) = \cos x$: derivar es girar un
        cuarto de vuelta sobre la circunferencia; comprueba la coherencia
        con el \cref{thm:g12:trigo:derivatives}.
  \item Deduce la identidad de producto a suma
        $\cos a \cos b = \frac12\left(\cos(a - b) + \cos(a + b)\right)$ y
        la linealización $\cos^2 x = \frac{1 + \cos 2x}{2}$ (el capítulo de
        integración te lo agradecerá).
  \item Pulsaciones: deduce
        $\sin p + \sin q = 2 \sin\frac{p + q}{2}\cos\frac{p - q}{2}$ y
        aplícalo a dos diapasones de $440$ y $444$ Hz: ¿qué tono oye el
        oído y cuántas pulsaciones por segundo? (Así es como afinan las
        orquestas.)
  \item Deduce la identidad del ángulo triple
        $\cos 3x = 4\cos^3 x - 3\cos x$ (escribe $3x = 2x + x$). Tomando
        $x = 20^\circ$, ¿qué \omterm{def:g10:algebra:equation}{ecuación} de tercer grado debe cumplir
        $\cos 20^\circ$? (Que esa \omterm{def:g10:algebra:equation}{ecuación} no se pueda resolver solo con
        raíces cuadradas es precisamente la \omterm{def:g11:seq:geometric}{razón} de que la trisección del
        ángulo derrotara a la regla y el compás durante dos milenios; la
        historia completa se cuenta en los volúmenes universitarios.)
  \item Una corriente alterna vale
        $I(t) = 3\cos(100\pi t) + 3\sqrt3 \sin(100\pi t)$. Reescríbela como
        $R\cos(100\pi t - \varphi)$: ¿corriente de pico y fase?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El mundo que oscila.}
\begin{enumerate}[resume]
  \item Periodos: ¿cuál es el periodo (\omterm{def:g10:functions:extrema}{mínimo}) de $\sin^2 t$ (usa la
        pregunta 13)? ¿Y el de $\sin(2t) + \sin(3t)$?
  \item Un péndulo real se amortigua: $y(t) = \eu^{-t/10}\cos(2\pi t)$.
        Describe el movimiento, calcula el valor de la envolvente en
        $t = 10$ (aparece un viejo conocido, \cref{pb:g12:exp:1}) y nombra
        el capítulo donde se resuelven las ecuaciones con soluciones de ese
        tipo.
  \item ¿Por qué gobiernan los \omterm{def:g12:trigo:cossin}{senos} y los \omterm{def:g12:trigo:cossin}{cosenos} todas las vibraciones?
        Da en dos frases el diccionario de la física a las matemáticas
        (fuerza recuperadora proporcional al desplazamiento
        $\rightarrow$ qué \omterm{def:g10:algebra:equation}{ecuación} $\rightarrow$ qué soluciones) y di qué
        aporta a la afirmación el \cref{exo:g12:trigo:9}.
  \item Final: el arco del capítulo en cuatro pasos: un límite geométrico,
        dos \omterm{def:g12:deriv:derivative}{derivadas}, una \omterm{def:g10:algebra:equation}{ecuación} diferencial y toda la oscilación. Y el
        asterisco honrado: para oscilaciones grandes, el periodo verdadero
        del péndulo involucra una integral elíptica que se calcula a la
        velocidad del rayo con\,\dots\ la \omterm{def:g11:stat:mean}{media} aritmético-geométrica de
        Gauss, el postre del \cref{pb:g12:seq:1}. Los hilos de la serie se
        anudan solos.
\end{enumerate}
\end{problem}
