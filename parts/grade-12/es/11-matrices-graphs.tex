\chapter{Matrices y grafos}\label{ch:g12:matrix}

A matriz es una matriz rectangular de números, sumados y multiplicados por reglas
diseñado para que matriz El álgebra representa la composición de líneas.
transformaciones. Matrices resuelven sistemas lineales, unidad acoplada recurrente
sucesionesy contar paseos en redes: las matemáticas detrás de la búsqueda
motores y algoritmos de camino más corto.

\section{Álgebra matricial}

\begin{definition}[Matriz]\label{def:g12:matrix:matrix}
Un \emph{$m \times n $ matriz}\index{$ m \times n $ matriz} es una tabla de números reales con
$ m $ filas y $ n $ columnas:
$ A = (a_{ij})$, dónde $ a_{ij}$ es la entrada en la fila $ i $, columna $ j $.
Se suman dos matrices del mismo tamaño entrada por entrada, y
$\lambda A = (\lambda a_{ij})$.
\end{definition}

\begin{definition}[Producto de matriz]\label{def:g12:matrix:product}
Sea $ A $ ser $ m \times n $ y $ B $ ser $ n \times p $. el producto $ AB $ es el
$ m \times p $ matriz cuyo $(i,j)$ la entrada es
\[
(AB)_{ij} = \sum_{k=1}^{n} a_{ik} b_{kj}
\]
(la ``fila $ i $ de $ A $ columna de tiempos $ j $ de $ B $'' regla).
\end{definition}

\begin{example}
$\begin{pmatrix} 1 & 2\\ 3 & 4\end{pmatrix}
\begin{pmatrix} 0 & 1\\ 1 & 1\end{pmatrix}
= \begin{pmatrix} 2 & 3\\ 4 & 7\end{pmatrix}$,
\quad mientras \quad
$\begin{pmatrix} 0 & 1\\ 1 & 1\end{pmatrix}
\begin{pmatrix} 1 & 2\\ 3 & 4\end{pmatrix}
= \begin{pmatrix} 3 & 4\\ 4 & 6\end{pmatrix}$:
\emph{matriz la multiplicación no es conmutativa}.
\end{example}

\begin{proposition}[Reglas del álgebra matricial]\label{prop:g12:matrix:rules}
Siempre que los tamaños hagan que los productos tengan significado:
\[
(AB)C = A(BC), \qquad A(B + C) = AB + AC, \qquad (A+B)C = AC + BC,
\]
y el \emph{identidad matriz} $ I_n $ (unos en la diagonal, ceros en el resto)
satisface $ I_m A = A I_n = A $ para $ A $ de tamaño $ m \times n $.
\end{proposition}

\begin{proof}
Todas son verificaciones entrada por entrada de
\cref{def:g12:matrix:product}; asociatividad, la única no trivial,
equivale a intercambiar dos sumas finitas:
$\bigl((AB)C\bigr)_{ij} = \sum_l \left(\sum_k a_{ik}b_{kl}\right) c_{lj}
= \sum_k a_{ik} \left(\sum_l b_{kl} c_{lj}\right)
= \bigl(A(BC)\bigr)_{ij}$.
\end{proof}

\begin{definition}[Inverso]\label{def:g12:matrix:inverse}
un cuadrado matriz $ A $ de tamaño $ n $ es \emph{reversible}\index{reversible}
si hay un matriz $ B $ con $ AB = BA = I_n $; $ B $ es entonces único, escrito
$ A^{-1}$.
\end{definition}

\begin{proposition}[Inverso de un $2\times2$ matriz]\label{prop:g12:matrix:inverse2x2}
Sea $ A = \begin{pmatrix} a & b\\ c & d\end{pmatrix}$ y
$\det A = ad - bc $ (el \emph{determinante}\index{determinante}). Entonces $ A $ es
\omterm{def:g12:matrix:inverse}{reversible} si y solo si $\det A \neq 0$, en cuyo caso
\[
A^{-1} = \frac{1}{ad - bc}\begin{pmatrix} d & -b\\ -c & a\end{pmatrix}.
\]
\end{proposition}

\begin{proof}
Un cálculo da
$ A \begin{pmatrix} d & -b\\ -c & a\end{pmatrix}
= \begin{pmatrix} d & -b\\ -c & a\end{pmatrix} A = (ad - bc) I_2$; si
$ ad - bc \neq 0$, dividir. Por el contrario, si $ ad - bc = 0$, las columnas de $ A $
son proporcionales, al igual que las columnas de $ AB $ para cualquier $ B $; pero el
columnas de $ I_2$ no son proporcionales, entonces no $ B $ puede satisfacer $ AB = I_2$.
\end{proof}

\begin{method}[Sistemas lineales]\label{met:g12:matrix:systems}
el sistema
$\begin{cases} ax + by = e\\ cx + dy = f \end{cases}$
es el matriz \omterm{def:g10:algebra:equation}{ecuación} $ AX = Y $ con
$ X = \begin{pmatrix} x \\ y\end{pmatrix}$,
$ Y = \begin{pmatrix} e \\ f\end{pmatrix}$. Si $\det A \neq 0$, es único
la solución es $ X = A^{-1}Y $. El mismo formalismo maneja $ n $ ecuaciones en $ n $
incógnitas.
\end{method}

\section{Poderes matriciales y sucesiones recurrentes.}

\begin{definition}\label{def:g12:matrix:power}
por un cuadrado matriz $ A $ y $ k \in \N $, $ A^k = A \times \dots \times A $
($ k $ factores), con $ A^0 = I $.
\end{definition}

\begin{method}[Casos diagonal más nilpotente y diagonalizable]\label{met:g12:matrix:powers}
Dos formas estándar de calcular $ A^k $:
\begin{itemize}
  \item Si $ A = \lambda I + N $ dónde $ N^2 = 0$, el teorema del binomio
        (válido aquí desde $ I $ y $ N $ conmutar) se colapsa en dos términos:
        $ A^k = \lambda^k I + k \lambda^{k-1} N $.
  \item Si uno encuentra un \omterm{def:g12:matrix:inverse}{reversible} $ P $ con $ A = PDP^{-1}$ y $ D $ diagonal,
        entonces $ A^k = P D^k P^{-1}$, y $ D^k $ se computa entrada por entrada.
        (Encontrar tal $ P $ sistemáticamente es la teoría de
        \emph{diagonalización}, desarrollado en la universidad; en este nivel $ P $ es
        dado.)
\end{itemize}
\end{method}

\begin{example}[Sucesiones acopladas]\label{ex:g12:matrix:coupled}
Sea $ u_{n+1} = 3u_n + v_n $ y $ v_{n+1} = u_n + 3v_n $. Configuración
$ X_n = \begin{pmatrix} u_n\\ v_n \end{pmatrix}$ y
$ A = \begin{pmatrix} 3 & 1\\ 1 & 3\end{pmatrix}$, obtenemos
$ X_{n+1} = AX_n $, entonces $ X_n = A^n X_0$. el auxiliar sucesiones
$ s_n = u_n + v_n $ y $ d_n = u_n - v_n $ satisfacer $ s_{n+1} = 4s_n $ y
$ d_{n+1} = 2d_n $, entonces $ s_n = 4^n s_0$, $ d_n = 2^n d_0$ y
\[
u_n = \frac{4^n(u_0+v_0) + 2^n(u_0-v_0)}{2}, \qquad
v_n = \frac{4^n(u_0+v_0) - 2^n(u_0-v_0)}{2}.
\]
(Entre bastidores: $(1,1)$ y $(1,-1)$ son direcciones de \omterm{def:g11:vect:vector}{vectores} propios de $ A $.)
\end{example}

\section{Graficas y paseos}

\begin{definition}[Gráfico, matriz de adyacencia]\label{def:g12:matrix:graph}
A \emph{gráfico}\index{gráfico} consta de vértices $1, 2, \dots, n $ y bordes
unir ciertos pares de vértices (pares ordenados para un \emph{dirigido}
gráfico). Es \emph{matriz de adyacencia}\index{matriz de adyacencia} es el
$ n \times n $ matriz $ M $ con $ m_{ij} = 1$ si hay una ventaja de $ i $ a
$ j $, y $0$ de lo contrario. A \emph{caminar} de longitud $ k $ de $ i $ a $ j $ es un
\omterm{def:g12:seq:sequence}{sucesión} de $ k $ bordes consecutivos que van desde $ i $ a $ j $.
\end{definition}

  \node[circle, draw, inner sep=2.5pt] (v2) at (2.6,1.2) {$2$};
  \node[circle, draw, inner sep=2.5pt] (v3) at (2.6,-1.2) {$3$};
  \draw[-Stealth, omDef, thick] (v1) -- (v2);

\end{tikzpicture}

{\small un dirigido grafo y su \omterm{def:g12:matrix:graph}{matriz de adyacencia}

de $ i $ a $ j $.}
\end{omfigure}

\begin{theorem}[Contando paseos]\label{thm:g12:matrix:walks}
El número de caminatas de longitud. $ k $ desde el vértice $ i $ al vértice $ j $ es el
$(i,j)$ entrada de $ M^k $.
\end{theorem}

\begin{proof}
inducción en $ k $. Para $ k = 1$ esta es la definición de $ M $. asumir el
reclamo por $ k $. Un paseo de longitud $ k+1$ de $ i $ a $ j $ es un paseo de longitud
$ k $ de $ i $ a algún vértice $ l $, seguido de una ventaja de $ l $ a $ j $; por el
principios de suma y multiplicación, su número es
\[
\sum_{l=1}^{n} \bigl(M^k\bigr)_{il}\, m_{lj} = \bigl(M^{k+1}\bigr)_{ij}.
\qedhere
\]
\end{proof}

\begin{example}
para el triangulo grafo ($3$ vértices, todos los pares unidos),
$ M = \begin{pmatrix} 0&1&1\\ 1&0&1\\ 1&1&0\end{pmatrix}$ y
$ M^2 = \begin{pmatrix} 2&1&1\\ 1&2&1\\ 1&1&2\end{pmatrix}$: de cada vértice
hay $2$ paseos de longitud $2$ volver a sí mismo (a través de cualquiera de los vecinos) y
$1$ entre sí vértice.
\end{example}

\section{Problema: La matriz que conoce a Fibonacci (y la
clima)}

\begin{problem}[{Problema del fin de semana --- uno $2 \times 2$ matriz
lleva todo Fibonacci, una matriz de Markov pronostica el largo plazo
clima, y un vector propio vale mil millones de dólares}]
\label{pb:g12:matrix:1}
A matriz es una máquina que come un estado y devuelve el siguiente
--- y su \emph{potestades} por lo tanto, poseer futuros completos. esto
El problema se abre con la sorprendente matriz cuyos poderes enumeran el
Números de Fibonacci (y probar sus identidades en una línea cada uno),
luego ejecuta el clima como una cadena de Markov hasta su estado estacionario, y
cierra con el \omterm{def:g11:vect:vector}{vector} propio sobre el que se construyó un motor de búsqueda
(\cref{thm:g12:matrix:walks},
\cref{met:g12:matrix:powers}).

\medskip
\noindent\textbf{Part I --- Fluency.}
\begin{enumerate}
  \item Con $ A = \begin{pmatrix} 1 & 2\\ 3 & 4\end{pmatrix}$
        y $ B = \begin{pmatrix} 0 & 1\\ 1 & 0\end{pmatrix}$:
        calcular $ AB $ y $ BA $. ¿Veredicto sobre la conmutatividad?
  \item Invertir $\begin{pmatrix} 2 & 1\\ 5 & 3\end{pmatrix}$
        (\cref{prop:g12:matrix:inverse2x2}) y usar el inverso
        resolver $2x + y = 4$, $5x + 3y = 7$.
  \item Sea $ N = \begin{pmatrix} 0 & 1\\ 0 & 0\end{pmatrix}$:
        calcular $ N^2$, y deducir
        $(I + N)^n = I + nN $ por cada $ n $.
  \item el triangulo grafo (tres vértices, todos los pares unidos):
        escribe su \omterm{def:g12:matrix:graph}{matriz de adyacencia} $ A $, calcular $ A^3$, y
        interpretar las entradas diagonales
        (\cref{thm:g12:matrix:walks}).
  \item Para $ D = \begin{pmatrix} 2 & 0\\ 0 & \frac12
        \end{pmatrix}$: dar $ D^n $ y su comportamiento como
        $ n \to \infty $.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The Fibonacci matriz.}
Sea $ F = \begin{pmatrix} 1 & 1\\ 1 & 0\end{pmatrix}$ y deja
$ F_1 = F_2 = 1, F_3 = 2, \dots $ ser los números de Fibonacci de
\cref{pb:g11:seq:1}.
\begin{enumerate}[resume]
  \item Calcular $ F^2$, $ F^3$, $ F^4$ y conjeturar el general
        forma de $ F^n $ en términos de números de Fibonacci.
  \item Probar la conjetura
        $ F^n = \begin{pmatrix} F_{n+1} & F_n\\ F_n & F_{n-1}
        \end{pmatrix}$ por inducción.
  \item Llevar \omterm{def:g11:vect:det}{determinantes} de ambos lados (el \omterm{def:g11:vect:det}{determinante} de un
        producto es el producto del \omterm{def:g11:vect:det}{determinantes} --- compruébalo
        en $2 \times 2$ matrices si nunca lo has visto):
        deducir \emph{La identidad de Cassini}
        $ F_{n+1}F_{n-1} - F_n^2 = (-1)^n $ --- el
        Motor de cuadrado fugaz, probado en una línea.
  \item De $ F^{m+n} = F^m F^n $, lea la parte superior derecha
        entradas y derivar el \emph{fórmula de suma}
        \[
        F_{m+n} = F_{m+1} F_n + F_m F_{n-1} .
        \]
        Compruébalo para $ m = n = 3$.
  \item Deducir de la fórmula de la suma (inducción en $ k $)
        eso $ F_n $ dividir $ F_{kn}$, y verificar en
        $ F_3 \mid F_6$ y $ F_3 \mid F_9$.
  \item para calcular $ F_{100}$, no es necesario multiplicar $100$
        matrices: cuadrado repetidamente ($ F^2, F^4, F^8, \dots $)
        y combinar. Cuántos matriz las multiplicaciones son suficientes,
        y qué antiguo truco de multiplicación del Medio
        ¿El volumen escolar es este, promovido a matrices?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The weather machine.}
En una determinada ciudad: después de un día soleado, el siguiente es soleado con
\omterm{def:g10:proba:distribution}{probabilidad} $0.8$; después de un día lluvioso, soleado con \omterm{def:g10:proba:distribution}{probabilidad}
$0.4$. Codificar el día \omterm{def:g11:prob:rv}{distribución} como una columna
$\binom{p_{\text{sun}}}{p_{\text{rain}}}$ y la evolución por
\[
M = \begin{pmatrix} 0.8 & 0.4\\ 0.2 & 0.6 \end{pmatrix}.
\]
\begin{enumerate}[resume]
  \item Compruebe que cada columna de $ M $ sumas a $1$ y decir por qué
        cualquier máquina meteorológica debe tener esta propiedad.
  \item Hoy hace sol. Calcule el pronóstico para mañana y
        para el día siguiente.
  \item Encuentra el \emph{estado estable}: el \omterm{def:g11:prob:rv}{distribución} $v $ con
        $ Mv = v $ (y entradas que suman $1$). ¿Qué fracción de
        ¿Hay días soleados a largo plazo?
  \item Empezar desde un día lluvioso, $\binom01$ y aplicar $M $ cuatro
        veces, rastreando la distancia hasta el estado estable en
        cada paso. ¿En qué factor se reduce la brecha por paso?
        --- ¿Y qué tipo de convergencia es esta?
  \item PageRank en miniatura: tres páginas, con enlaces
        $ A \to B $, $ A \to C $, $ B \to C $, $ C \to A $. un azar
        El navegante sigue un enlace saliente de manera uniforme y aleatoria.
        Escribe la transición matriz, encuentre el estado estacionario y
        clasificar las páginas.
  \item Interpretar el ranking: ¿por qué? $ C $ puntuación tan alta como
        $ A $ para todo recibir enlaces de menos páginas --- ¿qué
        ¿Cómo mide realmente el estado estacionario? (PageRank real
        añade un factor de amortiguación para callejones sin salida y saltos; el
        La idea del \omterm{def:g11:vect:vector}{vector} propio es exactamente ésta.)
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Diagonal dividends.}
\begin{enumerate}[resume]
  \item Dos cantidades acopladas obedecen
        $ u_{n+1} = 3u_n + v_n $, $ v_{n+1} = u_n + 3v_n $, es decir\
        el matriz $ A $ de \cref{exo:g12:matrix:8}. Usando eso
        diagonalización del ejercicio
        ($ D = \operatorname{diag}(4, 2)$), dar el cerrado
        fórmula para $ u_n $ cuando $ u_0 = 1$, $ v_0 = 0$y comprobar
        contra el cálculo directo para $ n = 1, 2, 3$.
  \item En una o dos frases: ¿qué significa la diagonalización?
        \emph{hacer} a un sistema acoplado --- y en qué sentido es
        el estado estacionario de Markov de la pregunta 14 es un \omterm{def:g11:vect:vector}{vector} propio
        historia también?
  \item final --- el matrizLas tres caras de este fin de semana:
        contabilidad (sistemas e inversas), combinatoria
        (paseos y enlaces contados por poderes), y evolución
        (Fibonacci, el clima, la web --- lectura de futuros
        direcciones propias). Una frase cada uno, más la
        puntero hacia adelante: el álgebra lineal de los volúmenes universitarios
        hace de cada una de estas caras una teoría.
\end{enumerate}
\end{problem}
