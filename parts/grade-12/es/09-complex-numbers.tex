\chapter{Números complejos}\label{ch:g12:complex}

La \omterm{def:g10:algebra:equation}{ecuación} $x^2 = -1$ no tiene solución real. Ampliar $\R$ con un único
número nuevo $\iu$ cuyo cuadrado es $-1$ produce el cuerpo $\C$ de los
\omterm{def:g12:complex:def}{números complejos}, en el que \emph{toda} \omterm{def:g10:algebra:equation}{ecuación} polinómica tiene
soluciones y cuya aritmética codifica la geometría del plano: las
traslaciones, los giros y las homotecias se convierten en sumas y
productos.

\section{Forma binómica}

\begin{definition}[Números complejos]\label{def:g12:complex:def}
El conjunto de los \emph{números complejos}\index{número complejo} es
\[
\C = \{a + \iu b : a, b \in \R\},
\]
donde $\iu$ es un símbolo sujeto a la única regla $\iu^2 = -1$; la suma y el
producto se hacen como con polinomios reales en $\iu$, reduciendo $\iu^2$ a
$-1$. Los reales $a$ y $b$ son la \emph{parte real}\index{parte real}
$\Rea(z)$ y la \emph{parte imaginaria}\index{parte imaginaria} $\Ima(z)$ de
$z = a + \iu b$; esta expresión es la \emph{forma binómica}\index{forma
binómica} de $z$, y es única: $a + \iu b = a' + \iu b'$ si y solo si
$a = a'$ y $b = b'$.
\end{definition}

\begin{example}
$(2 + 3\iu)(1 - \iu) = 2 - 2\iu + 3\iu - 3\iu^2 = 5 + \iu$.
\end{example}

\begin{definition}[Conjugado]\label{def:g12:complex:conjugate}
El \emph{conjugado}\index{conjugado} de $z = a + \iu b$ es
$\conj{z} = a - \iu b$.
\end{definition}

\begin{proposition}[Propiedades de la conjugación]\label{prop:g12:complex:conjugate}
Para todos $z, w \in \C$:
\[
\conj{z + w} = \conj z + \conj w, \quad
\conj{zw} = \conj z\,\conj w, \quad
\conj{\conj z} = z, \quad
z + \conj z = 2\Rea(z), \quad
z - \conj z = 2\iu\,\Ima(z),
\]
y $z\conj z = a^2 + b^2 \geq 0$ para $z = a + \iu b$. Además, $z \in \R$ si
y solo si $z = \conj z$.
\end{proposition}

\begin{proof}
Todas son cálculos directos a partir de las \omterm{def:g12:complex:def}{formas binómicas}; por ejemplo,
$z\conj z = (a + \iu b)(a - \iu b) = a^2 - (\iu b)^2 = a^2 + b^2$.
\end{proof}

\begin{method}[Dividir números complejos]\label{met:g12:complex:division}
Para escribir un cociente en \omterm{def:g12:complex:def}{forma binómica}, multiplica el numerador y el
denominador por el \omterm{def:g12:complex:conjugate}{conjugado} del denominador:
\[
\frac{1}{2 + \iu} = \frac{2 - \iu}{(2+\iu)(2-\iu)} = \frac{2 - \iu}{5}
= \frac25 - \frac15\iu .
\]
Así pues, todo \omterm{def:g12:complex:def}{número complejo} no nulo tiene inverso: $\C$ es un
\emph{cuerpo}.
\end{method}

\section{El plano complejo, módulo y argumento}

\begin{definition}[Afijo, módulo]\label{def:g12:complex:modulus}
En el plano con \omterm{def:g10:coordgeom:system}{coordenadas} \omterm{def:g10:coordgeom:system}{ortonormales}, el punto $M(a, b)$ es la
\emph{\omterm{def:g10:functions:function}{imagen}} de $z = a + \iu b$, y $z$ es el \emph{afijo}\index{afijo} de
$M$. El \emph{módulo}\index{módulo} de $z$ es la distancia de $M$ al
origen:
\[
\abs{z} = \sqrt{a^2 + b^2} = \sqrt{z\conj z}.
\]
\end{definition}

\begin{proposition}[Propiedades del módulo]\label{prop:g12:complex:modulus}
Para $z, w \in \C$:
\[
\abs{zw} = \abs z\,\abs w, \quad
\abs{\tfrac zw} = \tfrac{\abs z}{\abs w}\ (w \neq 0), \quad
\abs{\conj z} = \abs z, \quad
\abs{z + w} \leq \abs{z} + \abs{w} \ \text{(desigualdad triangular)}.
\]
Además, $\abs{z_B - z_A}$ es la distancia entre los puntos de \omterm{def:g12:complex:modulus}{afijos} $z_A$
y $z_B$.
\end{proposition}

\begin{proof}
$\abs{zw}^2 = zw\,\conj{zw} = z\conj z\, w \conj w = \abs z^2 \abs w^2$ da
la primera; la segunda y la tercera son análogas. Para la desigualdad
triangular:
\[
\abs{z+w}^2 = (z+w)(\conj z + \conj w)
= \abs z^2 + \abs w^2 + 2\Rea(z \conj w)
\leq \abs z^2 + \abs w^2 + 2\abs{z\conj w}
= \bigl(\abs z + \abs w\bigr)^2,
\]
usando $\Rea(u) \leq \abs u$. La afirmación sobre la distancia es la
fórmula de Pitágoras aplicada a las \omterm{def:g10:coordgeom:system}{coordenadas} de $z_B - z_A$.
\end{proof}

\begin{definition}[Argumento, formas trigonométrica y exponencial]\label{def:g12:complex:argument}
Sea $z \neq 0$. Un \emph{argumento}\index{argumento} de $z$, que se escribe
$\Arg(z)$, es cualquier medida $\theta$ del ángulo que va del semieje real
positivo a la semirrecta $OM$; está definido salvo $2k\pi$. Escribiendo
$r = \abs z$,
\[
z = r(\cos\theta + \iu\sin\theta)
\qquad\text{(forma trigonométrica)}.
\]
Introduciendo la notación\index{forma exponencial}
\[
\eu^{\iu\theta} = \cos\theta + \iu\sin\theta,
\]
esto se convierte en la \emph{forma exponencial}
$z = r\,\eu^{\iu\theta}$.
\end{definition}

\begin{theorem}\label{thm:g12:complex:expform}
Para todos $\theta, \theta' \in \R$:
\[
\eu^{\iu\theta}\,\eu^{\iu\theta'} = \eu^{\iu(\theta + \theta')},
\qquad
\conj{\eu^{\iu\theta}} = \eu^{-\iu\theta},
\qquad
\abs{\eu^{\iu\theta}} = 1 .
\]
En consecuencia, para $z = r\eu^{\iu\theta}$ y $z' = r'\eu^{\iu\theta'}$ no
nulos:
\[
zz' = rr'\,\eu^{\iu(\theta+\theta')},
\qquad
\frac{z}{z'} = \frac{r}{r'}\,\eu^{\iu(\theta - \theta')} :
\]
\emph{los \omterm{def:g12:complex:modulus}{módulos} se multiplican y los \omterm{def:g12:complex:argument}{argumentos} se suman.}
\end{theorem}

\begin{proof}
La primera identidad \emph{es} la pareja de fórmulas de adición
(\cref{prop:g12:trigo:addition}):
\[
(\cos\theta + \iu\sin\theta)(\cos\theta' + \iu\sin\theta')
= (\cos\theta\cos\theta' - \sin\theta\sin\theta')
+ \iu(\sin\theta\cos\theta' + \cos\theta\sin\theta').
\]
Lo demás se sigue por cálculo directo.
\end{proof}

\begin{corollary}[Fórmula de De Moivre]\label{cor:g12:complex:demoivre}
\index{fórmula de De Moivre}
Para $\theta \in \R$ y $n \in \Z$:
$\bigl(\cos\theta + \iu\sin\theta\bigr)^n = \cos n\theta + \iu \sin n\theta$.
\end{corollary}

\begin{proof}
Inducción sobre $n \geq 0$ con el \cref{thm:g12:complex:expform}; se
extiende a $n < 0$ tomando inversos.
\end{proof}

\begin{method}[Cambiar de forma]\label{met:g12:complex:forms}
De la binómica a la exponencial: calcula $r = \abs z$ y halla después
$\theta$ con $\cos\theta = \frac ar$ y $\sin\theta = \frac br$ (localiza el
cuadrante correcto antes de invocar ningún arcocoseno). De la exponencial a
la binómica: desarrolla $r\cos\theta + \iu\, r\sin\theta$. Usa la \omterm{def:g12:complex:def}{forma
binómica} para las sumas y la exponencial para los productos, las potencias y
los cocientes.
\end{method}

\begin{example}
$z = 1 + \iu$: $\abs z = \sqrt2$ y
$\cos\theta = \sin\theta = \frac{1}{\sqrt2}$, luego $\theta = \frac\pi4$ y
$z = \sqrt2\,\eu^{\iu\pi/4}$. Por tanto,
$z^{20} = 2^{10}\,\eu^{5\iu\pi} = -1024$.
\end{example}

\section{Ecuaciones de segundo grado y raíces de la unidad}

\begin{theorem}[Ecuaciones de segundo grado con coeficientes reales]\label{thm:g12:complex:quadratic}
Sean $a, b, c \in \R$ con $a \neq 0$, y sea $\Delta = b^2 - 4ac$. La
\omterm{def:g10:algebra:equation}{ecuación} $az^2 + bz + c = 0$ tiene soluciones en $\C$:
\[
\begin{aligned}
&\Delta > 0:\ z = \frac{-b \pm \sqrt{\Delta}}{2a} \ (\text{dos reales}); \qquad
\Delta = 0:\ z = \frac{-b}{2a} \ (\text{doble}); \\
&\Delta < 0:\ z = \frac{-b \pm \iu\sqrt{-\Delta}}{2a} \ (\text{dos conjugadas}).
\end{aligned}
\]
\end{theorem}

\begin{proof}
Completamos el cuadrado:
$az^2 + bz + c = a\left(\left(z + \frac{b}{2a}\right)^2 -
\frac{\Delta}{4a^2}\right)$. Si $\Delta < 0$, entonces
$\frac{\Delta}{4a^2} = \left(\iu\frac{\sqrt{-\Delta}}{2a}\right)^2$, y la
\omterm{def:g11:seq:arithmetic}{diferencia} de cuadrados se factoriza como siempre.
\end{proof}

\begin{definition}[Raíces de la unidad]\label{def:g12:complex:rootsofunity}
Para $n \geq 1$, las \emph{raíces $n$-ésimas de la unidad}\index{raíces de
la unidad} son las soluciones de $z^n = 1$.
\end{definition}

\begin{theorem}\label{thm:g12:complex:rootsofunity}
La \omterm{def:g10:algebra:equation}{ecuación} $z^n = 1$ tiene exactamente $n$ soluciones:
\[
\omega_k = \eu^{2\iu k\pi/n}, \qquad k = 0, 1, \dots, n-1 .
\]
Sus imágenes son los vértices de un polígono regular de $n$ lados inscrito
en la circunferencia unidad y, para $n \geq 2$, su suma es $0$.
\end{theorem}

\begin{proof}
Cada $\omega_k$ cumple $\omega_k^n = \eu^{2\iu k\pi} = 1$, y son distintas
dos a dos, porque sus \omterm{def:g12:complex:argument}{argumentos} están en $\intco{0}{2\pi}$.
Recíprocamente, si $z^n = 1$, entonces $\abs z^n = 1$, luego $\abs z = 1$
(por ser real positivo) y $z = \eu^{\iu\theta}$ con
$n\theta \equiv 0 \pmod{2\pi}$, es decir, $\theta = \frac{2k\pi}{n}$;
reduciendo $k$ \omterm{def:g12:complex:modulus}{módulo} $n$ se cae entre los $\omega_k$. Los vértices están
igualmente espaciados por el ángulo $\frac{2\pi}{n}$: un polígono regular de
$n$ lados. Por último, con $\omega = \omega_1$, se tiene $\omega \neq 1$ y
la suma geométrica da
\[
\sum_{k=0}^{n-1}\omega_k = \sum_{k=0}^{n-1}\omega^k
= \frac{\omega^n - 1}{\omega - 1} = 0 . \qedhere
\]
\end{proof}

\end{example}

\section{Números complejos y geometría del plano}

\begin{proposition}[Diccionario geométrico]\label{prop:g12:complex:geometry}
Sean $A, B, C, D$ puntos de \omterm{def:g12:complex:modulus}{afijos} $a, b, c, d$ (con $a \neq b$ y
$c \neq d$).
\begin{enumerate}
  \item La traslación de \omterm{def:g11:vect:vector}{vector} de \omterm{def:g12:complex:modulus}{afijo} $t$ es $z \mapsto z + t$.
  \item El giro de centro $\omega$ (\omterm{def:g12:complex:modulus}{afijo}) y ángulo $\theta$ es
        $z \mapsto \omega + \eu^{\iu\theta}(z - \omega)$.
  \item La homotecia de centro $\omega$ y \omterm{def:g11:seq:geometric}{razón} $k \in \R^*$ es
        $z \mapsto \omega + k(z - \omega)$.
  \item $\dfrac{d - c}{b - a}$ tiene \omterm{def:g12:complex:modulus}{módulo} $\dfrac{CD}{AB}$ y \omterm{def:g12:complex:argument}{argumento}
        igual al ángulo que forman los \omterm{def:g11:vect:vector}{vectores} $\vect{AB}$ y $\vect{CD}$.
        En particular, $AB \perp CD$ si y solo si $\frac{d-c}{b-a}$ es
        imaginario puro, y $A, B, C$ están alineados ($C \ne A$,
        $C\ne B$) si y solo si $\frac{c-a}{b-a} \in \R$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) y (3) reformulan las fórmulas en \omterm{def:g10:coordgeom:system}{coordenadas}. Para (2), la aplicación
$w \mapsto \eu^{\iu\theta} w$ multiplica los \omterm{def:g12:complex:modulus}{módulos} por $1$ y suma
$\theta$ a los \omterm{def:g12:complex:argument}{argumentos}: es el giro de ángulo $\theta$ alrededor del
origen; conjugándola con la traslación que lleva $\omega$ a $0$ se obtiene
el centro general. Para (4), los \omterm{def:g12:complex:modulus}{módulos} y los \omterm{def:g12:complex:argument}{argumentos} de los cocientes
se restan (\cref{thm:g12:complex:expform}).
\end{proof}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:complex:1}
Escribe en \omterm{def:g12:complex:def}{forma binómica}:
$(3 - 2\iu)(1 + 4\iu)$, \quad $\dfrac{2 + \iu}{1 - \iu}$, \quad
$\iu^{2027}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:complex:2}
Resuelve en $\C$: (a) $z^2 - 4z + 13 = 0$; (b) $z^2 + z + 1 = 0$. Comprueba
en cada caso que las dos soluciones son conjugadas y calcula su producto.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:complex:3}
Pasa a \omterm{def:g12:complex:argument}{forma exponencial} $z_1 = -1 + \iu$ y $z_2 = \sqrt3 - \iu$, y calcula
$\dfrac{z_1}{z_2}$ en \omterm{def:g12:complex:argument}{forma exponencial} y después en \omterm{def:g12:complex:def}{forma binómica}. Deduce
el valor exacto de $\cos\dfrac{11\pi}{12}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:4}
Describe geométricamente el conjunto de los puntos $M$ de \omterm{def:g12:complex:modulus}{afijo} $z$ tales
que:
(a) $\abs{z - 2} = \abs{z + \iu}$;\quad
(b) $\abs{z - 1 - \iu} = 3$;\quad
(c) $\dfrac{z - 1}{z + 1}$ es imaginario puro.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:5}
Usando la fórmula de De Moivre y el teorema del binomio, expresa
$\cos 3\theta$ como un polinomio en $\cos\theta$, y $\sin 3\theta$ en
\omterm{def:g11:func:function}{función} de $\sin\theta$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:6}
\emph{(Linealización.)} Usando
$\cos\theta = \frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}$, lineariza
$\cos^3\theta$ (escríbelo como combinación de $\cos k\theta$) y deduce
$\displaystyle\int_0^{\pi/2}\cos^3\theta\,\dd\theta$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:7}
Sean $A$ y $B$ de \omterm{def:g12:complex:modulus}{afijos} $a = 1 + \iu$ y $b = 3 + 2\iu$. Determina los dos
puntos $C$ que hacen equilátero el triángulo $ABC$, como imágenes de $B$
mediante los giros de centro $A$ y ángulos $\pm\frac{\pi}{3}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:8}
Resuelve $z^4 = -4$ en $\C$ y representa las soluciones. Factoriza
$z^4 + 4$ como producto de dos polinomios de segundo grado con coeficientes
reales.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:9}
Para $\theta \in \intoo{0}{2\pi}$ y $n \in \N$, calcula
\[
S = 1 + \cos\theta + \cos 2\theta + \dots + \cos n\theta
\]
sumando la \omterm{def:g12:seq:arith-geom}{progresión geométrica} $\sum_{k=0}^n \eu^{\iu k\theta}$ y tomando
\omterm{def:g12:complex:def}{partes reales}. (Respuesta:
$S = \dfrac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}}
\cos\frac{n\theta}{2}$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:10}
Sea $\omega = \eu^{2\iu\pi/5}$.
\begin{enumerate}
  \item Justifica que $1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$.
  \item Sean $u = \omega + \omega^4$ y $v = \omega^2 + \omega^3$. Demuestra
        que $u + v = -1$ y $uv = -1$, y deduce que
        $u = \frac{-1+\sqrt5}{2}$.
  \item Concluye que $\cos\dfrac{2\pi}{5} = \dfrac{\sqrt5 - 1}{4}$.
\end{enumerate}
\end{exercise}

\section{Problema: El pentágono en el espejo de la unidad}

\begin{problem}[{Problema de fin de semana --- las raíces quintas de la
unidad calculan $\cos 72^\circ$ de forma exacta, el número áureo firma el
resultado y el producto resulta ser un giro}]
\label{pb:g12:complex:1}
Euclides sabía construir el pentágono regular, pero la \omterm{def:g11:seq:geometric}{razón} de \emph{por
qué} se rinde ante la regla y el compás (mientras que el humilde ángulo de
$20^\circ$ no lo hace) permaneció escondida durante dos mil años. Se
esconde en este capítulo: las cinco raíces quintas de la unidad
(\cref{thm:g12:complex:rootsofunity}) suman cero, y esa única línea de
álgebra exprime $\cos 72^\circ$ de una \omterm{def:g10:algebra:equation}{ecuación} de segundo grado: solo
raíces cuadradas y, por tanto, construible, y firmado, cómo no, por el
número áureo. Alrededor de esa joya: soltura, el atajo de De Moivre hacia la
trigonometría y el producto desenmascarado como geometría.

\medskip
\noindent\textbf{Parte I --- Soltura.}
\begin{enumerate}
  \item Calcula $(2 + \iu)(3 - \iu)$, \ $\dfrac{1 + \iu}{1 - \iu}$
        (\cref{met:g12:complex:division}) y $\abs{3 + 4\iu}$.
  \item Pasa a \omterm{def:g12:complex:argument}{forma exponencial}: $1 + \iu$; \ $-2$; \
        $1 - \iu\sqrt3$.
  \item Calcula $(1 + \iu)^8$ usando la \omterm{def:g12:complex:argument}{forma exponencial}.
  \item Resuelve $z^2 = \iu$ y después $z^2 - 2z + 5 = 0$
        (\cref{thm:g12:complex:quadratic}).
  \item Diccionario geométrico (\cref{prop:g12:complex:geometry}): describe
        la aplicación $z \mapsto \iu z$ y el conjunto
        $\abs{z - (1 + \iu)} = 2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Cinco raíces, un pentágono.}
Sea $\omega = \eu^{2\iu\pi/5}$ y consideremos
$1, \omega, \omega^2, \omega^3, \omega^4$.
\begin{enumerate}[resume]
  \item Comprueba que son exactamente las soluciones de $z^5 = 1$ y
        describe la figura que dibujan en el plano.
  \item Demuestra que su suma es cero. (\omterm{def:g12:seq:arith-geom}{Progresión geométrica}, o multiplica
        la suma por $1 - \omega$.)
  \item Toma \omterm{def:g12:complex:def}{partes reales} y deduce
        \[
        1 + 2\cos\frac{2\pi}{5} + 2\cos\frac{4\pi}{5} = 0 .
        \]
  \item Toma $c = \cos\frac{2\pi}{5}$ y usa la fórmula del ángulo doble
        para $\cos\frac{4\pi}{5}$ para convertir la pregunta 8 en una
        \omterm{def:g10:algebra:equation}{ecuación} de segundo grado en $c$; resuélvela y concluye que
        \[
        \cos 72^\circ = \frac{\sqrt5 - 1}{4} .
        \]
  \item Compruébalo numéricamente y deja que el número áureo firme su obra:
        demuestra que $c = \frac{1}{2\varphi}$, donde
        $\varphi = \frac{1 + \sqrt5}{2}$ (\cref{pb:g10:algebra:1}), y
        enuncia el famoso eco geométrico: en un pentágono regular, la
        diagonal es $\varphi$ veces el lado.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Los atajos de De Moivre.}
\begin{enumerate}[resume]
  \item El veredicto de constructibilidad: $\cos 72^\circ$ solo necesita
        raíces cuadradas (pregunta 9), así que el pentágono es construible
        con regla y compás; $\cos 20^\circ$ cumple una \omterm{def:g10:algebra:equation}{ecuación} de tercer
        grado irrompible (\cref{pb:g12:trigo:1}), así que el ángulo de
        $60^\circ$ no se puede trisecar. Enuncia el criterio que se
        vislumbra (las raíces cuadradas se construyen, las cúbicas no), cuya
        teoría completa (Gauss, a los 19 años, y el polígono de 17 lados)
        vive en los volúmenes universitarios.
  \item Resuelve $z^4 = -4$ en \omterm{def:g12:complex:argument}{forma exponencial} y recupera la
        factorización $z^4 + 4 = (z^2 - 2z + 2)(z^2 + 2z + 2)$ del
        \cref{exo:g12:complex:8} emparejando raíces conjugadas.
  \item Desarrolla $(\cos\theta + \iu\sin\theta)^3$ con el teorema del
        binomio y De Moivre, y vuelve a deducir
        $\cos 3\theta = 4\cos^3\theta - 3\cos\theta$ en tres líneas;
        compáralo con el camino de las fórmulas de adición del
        \cref{pb:g12:trigo:1}.
  \item Contrasta con el \cref{exo:g12:complex:9}: para
        $\theta = \frac{2\pi}{5}$ y $n = 4$, ¿cuánto vale la suma
        $1 + \cos\theta + \dots + \cos 4\theta$ y por qué es coherente con
        la pregunta 8?
  \item Calcula las tres raíces cúbicas de $8\iu$ en \omterm{def:g12:complex:def}{forma binómica}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El producto es geometría.}
\begin{enumerate}[resume]
  \item Describe por completo la aplicación $z \mapsto (1 + \iu)z$: ¿qué
        ángulo gira, por qué factor dilata y qué le hace al cuadrado
        unidad?
  \item Demuestra $\abs{z_1 z_2} = \abs{z_1}\,\abs{z_2}$ con \omterm{def:g12:complex:conjugate}{conjugados}
        (\cref{prop:g12:complex:modulus}), calcula después $(1+\iu)^n$ para
        $n = 1, 2, 3, 4$ y describe la espiral que trazan las potencias.
  \item El número $\mathrm{j} = \eu^{2\iu\pi/3}$: demuestra que
        $1 + \mathrm{j} + \mathrm{j}^2 = 0$ y comprueba sobre el triángulo
        $\left(1, \mathrm{j}, \mathrm{j}^2\right)$ el criterio clásico:
        $a + \mathrm{j} b + \mathrm{j}^2 c = 0$ se cumple para (una
        orientación de) un triángulo equilátero $abc$.
  \item El teorema fundamental del álgebra (que se admite:
        d'Alembert--Gauss): todo polinomio se factoriza por completo sobre
        $\C$. Deduce el corolario aplicable al mundo real (todo polinomio
        real se factoriza en piezas reales de primer y segundo grado) y
        señala la pregunta 12 como ejemplo.
  \item Final: el retrato de $\C$, una frase para cada cosa: los números se
        vuelven puntos; el producto se vuelve giro y dilatación; las \omterm{def:g12:complex:rootsofunity}{raíces
        de la unidad} se vuelven polígonos regulares; De Moivre convierte
        potencias en identidades trigonométricas; y la clausura algebraica
        garantiza que ninguna \omterm{def:g10:algebra:equation}{ecuación} necesitará nunca un mundo mayor.
        Coda: ¿qué dos números célebres se encontraron en la pregunta 10?
\end{enumerate}
\end{problem}
