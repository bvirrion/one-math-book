\chapter{Ecuaciones diferenciales}\label{ch:g12:diffeq}

A \omterm{def:g12:diffeq:ode}{ecuación diferencial} relaciona una \omterm{def:g11:func:function}{función} a su derivadas. física,
La química, la biología y la economía expresan sus leyes de esta forma: la tasa de
enfriamiento de un cuerpo, la desintegración de un núcleo radiactivo, el crecimiento de un
población son todas declaraciones sobre $y'$. Este capítulo resuelve completamente el
lineal de primer orden ecuaciones con coeficientes constantes.

\section{la ecuacion \texorpdfstring{$ y' = ay $}{y' = ay}}

\begin{definition}[Ecuación diferencial]\label{def:g12:diffeq:ode}
A \emph{ecuación diferencial}\index{ecuación diferencial} es un \omterm{def:g10:algebra:equation}{ecuación}
cuya incógnita es una \omterm{def:g11:func:function}{función} $ y $, involucrando $ y $ y su derivadas. A
\emph{solución} en un \omterm{def:g10:numbers:interval}{intervalo} $ I $ es un \omterm{def:g12:deriv:derivative}{diferenciable} \omterm{def:g11:func:function}{función} satisfactorio
el \omterm{def:g10:algebra:equation}{ecuación} en cada punto de $ I $.
\end{definition}

\begin{theorem}[Soluciones de $ y' = ay $]\label{thm:g12:diffeq:homogeneous}
Sea $ a \in \R $. Las soluciones en $\R $ del \omterm{def:g10:algebra:equation}{ecuación} $ y' = ay $ son exactamente
el funciones
\[
y(x) = C\,\eu^{ax}, \qquad C \in \R .
\]
Por cada par $(x_0, y_0)$ hay una solución única con
$ y(x_0) = y_0$.
\end{theorem}

\begin{proof}
Cada $ y = C\eu^{ax}$ satisface $ y' = Ca\,\eu^{ax} = ay $. Por el contrario, dejemos $ y $
ser cualquier solución y establecer $ z(x) = y(x)\,\eu^{-ax}$. Entonces
\[
z'(x) = y'(x)\,\eu^{-ax} - a\,y(x)\,\eu^{-ax}
= \bigl(y'(x) - a y(x)\bigr)\eu^{-ax} = 0,
\]
entonces $ z $ es constante, digamos $ C $, y $ y(x) = C\eu^{ax}$. La condición inicial
$ y(x_0) = y_0$ efectivo $ C = y_0 \eu^{-a x_0}$, únicamente.
\end{proof}

\begin{remark}
La exponencial es así \emph{caracterizado} por el más simple de todos
ecuaciones diferenciales: crecimiento proporcional al tamaño. Por eso aparece
en todas partes de la naturaleza.
\end{remark}

\begin{example}[desintegración radiactiva]\label{ex:g12:diffeq:decay}
Una cantidad radiactiva satisface $ N' = -\lambda N $ con $\lambda > 0$, entonces
$ N(t) = N_0\,\eu^{-\lambda t}$; ver \cref{ex:g12:exp:halflife} para el
vida \omterm{def:g11:stat:mean}{media}.
\end{example}

\section{la ecuacion \texorpdfstring{$ y' = ay + b $}{y' = ay + b}}

\begin{theorem}[Soluciones de $ y' = ay + b $]\label{thm:g12:diffeq:affine}
Sea $ a \neq 0$ y $ b \in \R $. Las soluciones en $\R $ de $ y' = ay + b $ son
exactamente el funciones
\[
y(x) = C\,\eu^{ax} - \frac{b}{a}, \qquad C \in \R .
\]
la constante \omterm{def:g11:func:function}{función} $ y_p = -\frac ba $ es el \emph{solución de equilibrio}.
por cada $(x_0, y_0)$ hay una solución única con $ y(x_0) = y_0$; si
$ a < 0$, toda solución tiende al equilibrio $-\frac ba $ como
$ x \to +\infty $.
\end{theorem}

\begin{proof}
la constante $ y_p = -\frac{b}{a}$ satisface $ y_p' = 0 = a y_p + b $. Ahora $ y $
es una solución si y sólo si
\[
(y - y_p)' = y' = ay + b = a(y - y_p) + \underbrace{a y_p + b}_{=\,0}
= a (y - y_p),
\]
es decir, si y sólo si $ z = y - y_p $ resuelve $ z' = az $. Por
\cref{thm:g12:diffeq:homogeneous}, $ z = C\eu^{ax}$, de donde la fórmula, la
existencia y la unicidad. Si $ a < 0$, $\eu^{ax} \to 0$ como
$ x \to +\infty $, entonces $ y(x) \to -\frac ba $.
\end{proof}

\begin{method}[resolviendo $ y' = ay + b $ con una condición inicial]\label{met:g12:diffeq:solve}
\begin{enumerate}
  \item Encuentra la solución de equilibrio. $ y_p = -\frac{b}{a}$ (resolver $ y' = 0$).
  \item Escribe la solución general. $ y = C\eu^{ax} + y_p $.
  \item Determinar $ C $ de la condición inicial.
  \item Compare el comportamiento a largo plazo con la intuición física (¿el
        solución converge al equilibrio?).
\end{enumerate}
La misma estrategia --- \emph{solución particular + solución general del
homogéneo \omterm{def:g10:algebra:equation}{ecuación}} --- se extiende a $ y' = ay + f(x)$: ver
\cref{prop:g12:diffeq:general}.
\end{method}

\begin{example}[Ley de enfriamiento de Newton]\label{ex:g12:diffeq:cooling}
Una taza de café en $80\,^\circ $ C se sienta en una habitación en $20\,^\circ $ C. La ley de Newton establece
que la temperatura $T $ satisface $ T' = -k(T - 20)$ para algunos $ k > 0$,
\emph{es decir.} $ T' = -kT + 20k $. El equilibrio es $20$, y
$ T(t) = 20 + 60\,\eu^{-kt}$: el café se enfría exponencialmente rápido a temperatura ambiente
temperatura.
\end{example}

\section{la ecuacion \texorpdfstring{$ y' = ay + f(x)$}{y' = ay + f(x)}}

\begin{proposition}[Estructura del conjunto de soluciones.]\label{prop:g12:diffeq:general}
Sea $ f $ ser continua en un \omterm{def:g10:numbers:interval}{intervalo} $ I $ y deja $ y_p $ ser uno
solución de
\[
y' = ay + f(x)
\]
en $ I $. Entonces las soluciones en $ I $ son exactamente los funciones
$ y = C\eu^{ax} + y_p $, $ C \in \R $.
\end{proposition}

\begin{proof}
Como en \cref{thm:g12:diffeq:affine}: $ y $ es una solución si y sólo si
$ z = y - y_p $ satisface
$ z' = (ay + f) - (ay_p + f) = az $, si y sólo si $ z = C\eu^{ax}$.
\end{proof}

\begin{method}[Adivinando una solución particular]\label{met:g12:diffeq:particular}
Busque una solución particular \emph{de la misma forma} como $ f $:
\begin{itemize}
  \item $ f $ polinomio de grado $ n $: prueba un polinomio de grado $ n $;
  \item $ f(x) = \alpha\,\eu^{kx}$ con $ k \neq a $: intentar $ y_p = \beta\,\eu^{kx}$;
  \item $ f(x) = \alpha\,\eu^{ax}$ (caso resonante): intenta
        $ y_p = \beta x\,\eu^{ax}$.
\end{itemize}
Sustituir en el \omterm{def:g10:algebra:equation}{ecuación} e identificar los coeficientes.
\end{method}

\begin{example}\label{ex:g12:diffeq:particular}
Resolver $ y' = 2y + 4x $. Intentar $ y_p = \alpha x + \beta $:
$\alpha = 2(\alpha x + \beta) + 4x $ para todo $ x $ efectivo
$2\alpha + 4 = 0$ y $\alpha = 2\beta $, entonces $\alpha = -2$, $\beta = -1$.
Solución general: $ y = C\eu^{2x} - 2x - 1$.
\end{example}

\end{exercise}

\begin{exercise}[$\star $]\label{exo:g12:diffeq:3}
verificar que $ y_p(x) = x\,\eu^{x}$ es una solución de $ y' = y + \eu^x $, y

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:diffeq:4}
Resolver $ y' = -2y + \eu^{x}$ con $ y(0) = 1$.
(Pista: busque una solución particular de la forma $\beta\,\eu^{x}$.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:diffeq:5}
El carbono 14 se desintegra con una vida \omterm{def:g11:stat:mean}{media} de $5730$ años. Un arqueológico \omterm{def:g10:proba:sample}{muestra}
contiene $60\%$ del carbono-14 de un organismo vivo. Estima su edad.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:diffeq:6}
Un tanque contiene $100$ L de agua pura. Salmuera que contiene $0.2$ kg de sal por
El litro fluye a $5$ L/min, la mezcla (mantenida uniforme) sale al mismo
tasa. Sea $ m(t)$ ser la masa de sal en el tanque en ese momento $ t $ (en minutos).
\begin{enumerate}
  \item justifica eso $ m' = 1 - \dfrac{m}{20}$.
  \item Resuelva y determine el límite de $ m(t)$ como $ t \to +\infty $.
        Interpretar.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:diffeq:7}
Un paracaidista de masa $80$ kg cae sujeto a la gravedad
($ g = 9.8\ \text{m/s}^2$) y la resistencia del aire es proporcional a la velocidad, de modo que
su velocidad satisface $ v' = g - \frac{k}{m}v $ con $ k = 16$ kg/s.
\begin{enumerate}
  \item Resuelve el \omterm{def:g10:algebra:equation}{ecuación} con $ v(0) = 0$.
  \item Calcular el \emph{velocidad terminal} $\lim_{t\to+\infty} v(t)$, y
        el tiempo necesario para llegar $95\%$ de ello.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g12:diffeq:8}
\emph{(Logístico \omterm{def:g10:algebra:equation}{ecuación}.)} una poblacion $ y(t) \in \intoo{0}{1}$ (como un
fracción de la población máxima) satisface
\[
y' = y(1 - y).
\]
\begin{enumerate}
  \item Sea $ z = \dfrac{1}{y}$. mostrar que $ z $ satisface el lineal \omterm{def:g10:algebra:equation}{ecuación}
        $ z' = -z + 1$.
  \item resolver para $ z $, entonces para $ y $, con $ y(0) = \frac{1}{10}$.
  \item mostrar que $ y(t) \to 1$ como $ t \to +\infty $ y dibuja la forma de
        la curva de solución.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g12:diffeq:9}
Sea $ y $ ser una solución de $ y' = ay + b $ y deja $ u_n = y(n)$ para $ n \in \N $.
mostrar que $(u_n)$ satisface una recurrencia aritmético-geométrica
$ u_{n+1} = q u_n + r $, y expresar $ q $ y $ r $ en términos de $ a $ y $ b $. que
hace la condición $\abs{q} < 1$ corresponden para el \omterm{def:g12:diffeq:ode}{ecuación diferencial}?
\end{exercise}

\section{Problema: el reloj dentro de las cosas}

\begin{problem}[{Problema del fin de semana: el carbono 14 data las cuevas,
La ley de enfriamiento de Newton multiplica un delito, y una pequeña ecuación desgasta
cuatro disfraces}]
\label{pb:g12:diffeq:1}
A \omterm{def:g12:diffeq:ode}{ecuación diferencial} es una ley de cambio; resolverlo convierte el
ley en un \emph{reloj}. lo mismo diminuto \omterm{def:g10:algebra:equation}{ecuación}
$ y' = ay + b $ (\cref{thm:g12:diffeq:affine}) garrapatas dentro
carbón prehistórico, café refrescante, gotas de lluvia que caen y
goteos hospitalarios --- y leer esos relojes es el problema de este problema.
negocio: fecha las pinturas de Lascaux, fija la hora de la muerte,
y audita sus propios supuestos como un buen científico.

\medskip
\noindent\textbf{Part I --- Fluency.}
\begin{enumerate}
  \item Resolver $ y' = 3y $, $ y(0) = 2$; entonces
        $ y' = -2y + 6$, $ y(0) = 0$
        (\cref{met:g12:diffeq:solve}).
  \item Vuelva a derivar el truco de unicidad detrás
        \cref{thm:g12:diffeq:homogeneous}: si $ y' = ay $, calcular
        la derivada de $ y(t)\,\eu^{-at}$ y concluir que
        \emph{cada} la solución es $ C\eu^{at}$.
  \item Para $ y' = -2y + 6$: encontrar la solución de equilibrio, y
        describir el destino de cualquier otra solución como
        $ t \to +\infty $. (El puntos fijos de
        \cref{pb:g11:seq:1}Las recurrencias, desaparecieron. continua.)
  \item Para la ley de decadencia $ y' = -\frac{y}{\tau}$: demostrar que el
        la vida \omterm{def:g11:stat:mean}{media} es $ t_{1/2} = \tau \ln 2$.
  \item Dibuje (o describa) la familia de soluciones de
        $ y' = -2y + 6$: ¿Qué soluciones comienzan arriba? $3$
        ¿hacer? Abajo $3$? En $3$?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Carbon-14.}
El tejido vivo mantiene una proporción constante de radiactivos.
carbono-14; al morir, la ingesta se detiene y el stock decae:
$ N' = -\lambda N $, con vida \omterm{def:g11:stat:mean}{media} $5\,730$ años.
\begin{enumerate}[resume]
  \item Calcular $\lambda $ (por año).
  \item Un hueso retiene $20\,\%$ de la proporción de vida: ¿cuántos años tiene?
        eso?
  \item El carbón de las cuevas pintadas de Lascaux conserva
        acerca de $15\,\%$: fechar las pinturas.
  \item el hombre de hielo Ötzi, encontrado en un glaciar alpino, midió
        acerca de $53\,\%$: salir con él (los arqueólogos dicen sobre
        $5\,300$ años --- ¿cómo te fue?).
  \item ¿Por qué el carbono 14 no puede datar a los dinosaurios? Calcular el
        fracción restante después de diez vidas \omterm{def:g11:stat:mean}{medias}, indique la
        fracción después $65$ millones de años como un poder de $2$,
        y concluir.
  \item Barras de error: si Ötzi's $53\,\%$ sólo es conocido por
        $\pm 1\,\%$, calcule el \omterm{def:g10:stats:quartiles}{rango} de edad. ¿Qué precisión de laboratorio
        compra qué precisión de citas?
\end{enumerate}

\medskip

Un cuerpo a temperatura $ T $ en un entorno a constante $ T_a $

\begin{enumerate}[resume]
  \item Resuelve el \omterm{def:g10:algebra:equation}{ecuación} (sustituto $ z = T - T_a $):

        \]

        luego el tiempo de espera hasta el bebible
        $55\,^\circ $ DO.
  \item La cuestión de la leche: tomar el café lo más caliente posible
        en diez minutos, ¿debe entrar la leche fría ahora o en
        el ultimo momento? Responde con el \omterm{def:g10:algebra:equation}{ecuación} (¿Qué hace?
        agregando leche a la brecha $T - T_a $, y ¿cómo funciona el
        ¿La brecha impulsa la pérdida?).
  \item Forense: encuentran un cuerpo a medianoche a las
        $30\,^\circ $ C en un $20\,^\circ $ sala C; una hora después

        momento de la muerte, recuperarse $k $ de las dos medidas,

\noindent\textbf{Part IV --- Four costumes and an S-curve.}
\begin{enumerate}[resume]
  \item la logística \omterm{def:g10:algebra:equation}{ecuación} de \cref{exo:g12:diffeq:8},

\end{problem}
