\chapter{Ecuaciones diferenciales}\label{ch:g12:diffeq}

Una \omterm{def:g12:diffeq:ode}{ecuación diferencial} relaciona una \omterm{def:g11:func:function}{función} con sus \omterm{def:g12:deriv:derivative}{derivadas}. La física,
la química, la biología y la economía expresan sus leyes en esa forma: el
ritmo de enfriamiento de un cuerpo, la desintegración de un núcleo
radiactivo y el crecimiento de una población son afirmaciones sobre $y'$.
Este capítulo resuelve por completo las ecuaciones lineales de primer orden
con coeficientes constantes.

\section{La ecuación \texorpdfstring{$y' = ay$}{y' = ay}}

\begin{definition}[Ecuación diferencial]\label{def:g12:diffeq:ode}
Una \emph{ecuación diferencial}\index{ecuación diferencial} es una \omterm{def:g10:algebra:equation}{ecuación}
cuya incógnita es una \omterm{def:g11:func:function}{función} $y$ y en la que aparecen $y$ y sus \omterm{def:g12:deriv:derivative}{derivadas}.
Una \emph{solución} en un \omterm{def:g10:numbers:interval}{intervalo} $I$ es una \omterm{def:g11:func:function}{función} \omterm{def:g12:deriv:derivative}{derivable} que cumple
la \omterm{def:g10:algebra:equation}{ecuación} en todos los puntos de $I$.
\end{definition}

\begin{theorem}[Soluciones de $y' = ay$]\label{thm:g12:diffeq:homogeneous}
Sea $a \in \R$. Las soluciones en $\R$ de la \omterm{def:g10:algebra:equation}{ecuación} $y' = ay$ son
exactamente las funciones
\[
y(x) = C\,\eu^{ax}, \qquad C \in \R .
\]
Para cada pareja $(x_0, y_0)$ hay una única solución con $y(x_0) = y_0$.
\end{theorem}

\begin{proof}
Cada $y = C\eu^{ax}$ cumple $y' = Ca\,\eu^{ax} = ay$. Recíprocamente, sea
$y$ una solución cualquiera y tomemos $z(x) = y(x)\,\eu^{-ax}$. Entonces
\[
z'(x) = y'(x)\,\eu^{-ax} - a\,y(x)\,\eu^{-ax}
= \bigl(y'(x) - a y(x)\bigr)\eu^{-ax} = 0,
\]
así que $z$ es constante, digamos $C$, y $y(x) = C\eu^{ax}$. La condición
inicial $y(x_0) = y_0$ obliga a $C = y_0 \eu^{-a x_0}$, de manera única.
\end{proof}

\begin{remark}
La exponencial queda así \emph{caracterizada} por la más sencilla de todas
las ecuaciones diferenciales: crecimiento proporcional al tamaño. Por eso
aparece por todas partes en la naturaleza.
\end{remark}

\begin{example}[Desintegración radiactiva]\label{ex:g12:diffeq:decay}
Una cantidad radiactiva cumple $N' = -\lambda N$ con $\lambda > 0$, luego
$N(t) = N_0\,\eu^{-\lambda t}$; véase el \cref{ex:g12:exp:halflife} para el
periodo de semidesintegración.
\end{example}

\section{La ecuación \texorpdfstring{$y' = ay + b$}{y' = ay + b}}

\begin{theorem}[Soluciones de $y' = ay + b$]\label{thm:g12:diffeq:affine}
Sean $a \neq 0$ y $b \in \R$. Las soluciones en $\R$ de $y' = ay + b$ son
exactamente las funciones
\[
y(x) = C\,\eu^{ax} - \frac{b}{a}, \qquad C \in \R .
\]
La \omterm{def:g11:func:function}{función} constante $y_p = -\frac ba$ es la \emph{solución de equilibrio}.
Para cada $(x_0, y_0)$ hay una única solución con $y(x_0) = y_0$; y si
$a < 0$, todas las soluciones tienden al equilibrio $-\frac ba$ cuando
$x \to +\infty$.
\end{theorem}

\begin{proof}
La constante $y_p = -\frac{b}{a}$ cumple $y_p' = 0 = a y_p + b$. Ahora
bien, $y$ es solución si y solo si
\[
(y - y_p)' = y' = ay + b = a(y - y_p) + \underbrace{a y_p + b}_{=\,0}
= a (y - y_p),
\]
es decir, si y solo si $z = y - y_p$ resuelve $z' = az$. Por el
\cref{thm:g12:diffeq:homogeneous}, $z = C\eu^{ax}$, de donde la fórmula, la
existencia y la unicidad. Si $a < 0$, $\eu^{ax} \to 0$ cuando
$x \to +\infty$, luego $y(x) \to -\frac ba$.
\end{proof}

\begin{method}[Resolver $y' = ay + b$ con una condición inicial]\label{met:g12:diffeq:solve}
\begin{enumerate}
  \item Halla la solución de equilibrio $y_p = -\frac{b}{a}$ (resolviendo
        $y' = 0$).
  \item Escribe la solución general $y = C\eu^{ax} + y_p$.
  \item Determina $C$ con la condición inicial.
  \item Contrasta el comportamiento a largo plazo con la intuición física
        (¿\omterm{def:g12:seq:limit}{converge} la solución al equilibrio?).
\end{enumerate}
La misma estrategia (\emph{una solución particular más la solución general
de la \omterm{def:g10:algebra:equation}{ecuación} homogénea}) se extiende a $y' = ay + f(x)$: véase la
\cref{prop:g12:diffeq:general}.
\end{method}

\begin{example}[Ley de enfriamiento de Newton]\label{ex:g12:diffeq:cooling}
Una taza de café a $80\,^\circ$C está en una habitación a $20\,^\circ$C. La
ley de Newton afirma que la temperatura $T$ cumple $T' = -k(T - 20)$ para
cierto $k > 0$, es decir, $T' = -kT + 20k$. El equilibrio es $20$ y
$T(t) = 20 + 60\,\eu^{-kt}$: el café se enfría exponencialmente deprisa
hasta la temperatura ambiente.
\end{example}

\section{La ecuación \texorpdfstring{$y' = ay + f(x)$}{y' = ay + f(x)}}

\begin{proposition}[Estructura del conjunto de soluciones]\label{prop:g12:diffeq:general}
Sea $f$ \omterm{def:g12:limcont:continuity}{continua} en un \omterm{def:g10:numbers:interval}{intervalo} $I$ y sea $y_p$ una solución particular de
\[
y' = ay + f(x)
\]
en $I$. Entonces las soluciones en $I$ son exactamente las funciones
$y = C\eu^{ax} + y_p$, con $C \in \R$.
\end{proposition}

\begin{proof}
Como en el \cref{thm:g12:diffeq:affine}: $y$ es solución si y solo si
$z = y - y_p$ cumple $z' = (ay + f) - (ay_p + f) = az$, si y solo si
$z = C\eu^{ax}$.
\end{proof}

\begin{method}[Adivinar una solución particular]\label{met:g12:diffeq:particular}
Busca una solución particular \emph{de la misma forma} que $f$:
\begin{itemize}
  \item $f$ polinómica de grado $n$: prueba con un polinomio de grado $n$;
  \item $f(x) = \alpha\,\eu^{kx}$ con $k \neq a$: prueba con
        $y_p = \beta\,\eu^{kx}$;
  \item $f(x) = \alpha\,\eu^{ax}$ (caso resonante): prueba con
        $y_p = \beta x\,\eu^{ax}$.
\end{itemize}
Sustituye en la \omterm{def:g10:algebra:equation}{ecuación} e identifica los coeficientes.
\end{method}

\begin{example}\label{ex:g12:diffeq:particular}
Resolvamos $y' = 2y + 4x$. Probamos con $y_p = \alpha x + \beta$: que
$\alpha = 2(\alpha x + \beta) + 4x$ para todo $x$ obliga a
$2\alpha + 4 = 0$ y $\alpha = 2\beta$, luego $\alpha = -2$ y $\beta = -1$.
Solución general: $y = C\eu^{2x} - 2x - 1$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:diffeq:1}
Resuelve en $\R$:
(a) $y' = 3y$ con $y(0) = 2$;\quad
(b) $2y' + y = 0$ con $y(0) = -1$;\quad
(c) $y' = -y + 5$ con $y(0) = 0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:diffeq:2}
Una población de bacterias crece a un ritmo proporcional a su tamaño y se
duplica cada $3$ horas. Escribe la \omterm{def:g12:diffeq:ode}{ecuación diferencial} que cumple la
población $N(t)$ y determina la constante de proporcionalidad.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:diffeq:3}
Comprueba que $y_p(x) = x\,\eu^{x}$ es solución de $y' = y + \eu^x$ y da
todas las soluciones en $\R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:4}
Resuelve $y' = -2y + \eu^{x}$ con $y(0) = 1$. (Indicación: busca una
solución particular de la forma $\beta\,\eu^{x}$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:5}
El carbono-14 se desintegra con un periodo de semidesintegración de $5730$
años. Una \omterm{def:g10:proba:sample}{muestra} arqueológica contiene el $60\,\%$ del carbono-14 de un
organismo vivo. Estima su edad.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:6}
Un depósito contiene $100$ L de agua pura. Entra salmuera con $0.2$ kg de
sal por litro a $5$ L/min, y la mezcla (que se mantiene uniforme) sale al
mismo ritmo. Sea $m(t)$ la masa de sal del depósito en el instante $t$ (en
minutos).
\begin{enumerate}
  \item Justifica que $m' = 1 - \dfrac{m}{20}$.
  \item Resuelve la \omterm{def:g10:algebra:equation}{ecuación} y determina el límite de $m(t)$ cuando
        $t \to +\infty$. Interprétalo.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:7}
Una paracaidista de $80$ kg cae sometida a la gravedad
($g = 9.8\ \text{m/s}^2$) y a una resistencia del aire proporcional a la
velocidad, de modo que su velocidad cumple $v' = g - \frac{k}{m}v$ con
$k = 16$ kg/s.
\begin{enumerate}
  \item Resuelve la \omterm{def:g10:algebra:equation}{ecuación} con $v(0) = 0$.
  \item Calcula la \emph{velocidad límite} $\lim_{t\to+\infty} v(t)$ y el
        tiempo necesario para alcanzar el $95\,\%$ de esa velocidad.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:diffeq:8}
\emph{(\omterm{def:g10:algebra:equation}{Ecuación} logística.)} Una población $y(t) \in \intoo{0}{1}$
(expresada como fracción de la población máxima) cumple
\[
y' = y(1 - y).
\]
\begin{enumerate}
  \item Sea $z = \dfrac{1}{y}$. Demuestra que $z$ cumple la \omterm{def:g10:algebra:equation}{ecuación} lineal
        $z' = -z + 1$.
  \item Resuelve para $z$ y después para $y$, con $y(0) = \frac{1}{10}$.
  \item Demuestra que $y(t) \to 1$ cuando $t \to +\infty$ y esboza la forma
        de la curva solución.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:diffeq:9}
Sea $y$ una solución de $y' = ay + b$ y sea $u_n = y(n)$ para $n \in \N$.
Demuestra que $(u_n)$ cumple una recurrencia aritmético-geométrica
$u_{n+1} = q u_n + r$ y expresa $q$ y $r$ en \omterm{def:g11:func:function}{función} de $a$ y $b$. ¿A qué
corresponde la condición $\abs{q} < 1$ para la \omterm{def:g12:diffeq:ode}{ecuación diferencial}?
\end{exercise}

\section{Problema: El reloj que llevan dentro las cosas}

\begin{problem}[{Problema de fin de semana --- el carbono-14 data las
cuevas, la ley de enfriamiento de Newton cronometra un crimen y una
ecuación diminuta se pone cuatro disfraces}]
\label{pb:g12:diffeq:1}
Una \omterm{def:g12:diffeq:ode}{ecuación diferencial} es una ley de cambio; resolverla convierte la ley
en un \emph{reloj}. La misma \omterm{def:g10:algebra:equation}{ecuación} diminuta $y' = ay + b$
(\cref{thm:g12:diffeq:affine}) hace tictac dentro del carbón prehistórico,
del café que se enfría, de las gotas de lluvia que caen y de los goteros de
hospital; y leer esos relojes es el oficio de este problema: data las
pinturas de Lascaux, fija una hora de la muerte y audita sus propias
hipótesis, como haría un buen científico.

\medskip
\noindent\textbf{Parte I --- Soltura.}
\begin{enumerate}
  \item Resuelve $y' = 3y$ con $y(0) = 2$; y después $y' = -2y + 6$ con
        $y(0) = 0$ (\cref{met:g12:diffeq:solve}).
  \item Vuelve a deducir el truco de unicidad que hay detrás del
        \cref{thm:g12:diffeq:homogeneous}: si $y' = ay$, calcula la
        \omterm{def:g12:deriv:derivative}{derivada} de $y(t)\,\eu^{-at}$ y concluye que \emph{toda} solución
        es $C\eu^{at}$.
  \item Para $y' = -2y + 6$: halla la solución de equilibrio y describe el
        destino de todas las demás soluciones cuando $t \to +\infty$. (Los
        \omterm{pb:g10:functions:1}{puntos fijos} de las recurrencias del \cref{pb:g11:seq:1}, vueltos
        \omterm{def:g12:limcont:continuity}{continuos}.)
  \item Para la ley de desintegración $y' = -\frac{y}{\tau}$: demuestra que
        el periodo de semidesintegración es $t_{1/2} = \tau \ln 2$.
  \item Esboza (o describe) la familia de soluciones de $y' = -2y + 6$:
        ¿qué hacen las soluciones que empiezan por encima de $3$? ¿Y las
        que empiezan por debajo? ¿Y la que empieza en $3$?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El carbono-14.}
Los tejidos vivos mantienen constante la proporción de carbono-14
radiactivo; al morir cesa la incorporación y las reservas se desintegran:
$N' = -\lambda N$, con periodo de semidesintegración de $5\,730$ años.
\begin{enumerate}[resume]
  \item Calcula $\lambda$ (por año).
  \item Un hueso conserva el $20\,\%$ de la proporción de un ser vivo:
        ¿cuántos años tiene?
  \item El carbón de las cuevas pintadas de Lascaux conserva alrededor del
        $15\,\%$: data las pinturas.
  \item El hombre de hielo Ötzi, encontrado en un glaciar alpino, medía
        alrededor del $53\,\%$: dátalo (los arqueólogos hablan de unos
        $5\,300$ años; ¿qué tal lo has hecho?).
  \item ¿Por qué no puede el carbono-14 datar dinosaurios? Calcula la
        fracción que queda tras diez periodos, expresa la fracción tras
        $65$ millones de años como potencia de $2$ y concluye.
  \item Barras de error: si el $53\,\%$ de Ötzi solo se conoce con un
        margen de $\pm 1\,\%$, calcula el \omterm{def:g10:numbers:interval}{intervalo} de edades. ¿Qué
        precisión en el laboratorio compra qué precisión en la datación?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El enfriamiento de Newton, y un
crimen.}
Un cuerpo a temperatura $T$ en un entorno a temperatura constante $T_a$ se
enfría según $T' = -k\,(T - T_a)$.
\begin{enumerate}[resume]
  \item Resuelve la \omterm{def:g10:algebra:equation}{ecuación} (sustituyendo $z = T - T_a$):
        \[
        T(t) = T_a + (T_0 - T_a)\,\eu^{-kt} .
        \]
  \item Un café servido a $90\,^\circ$C en una habitación a
        $20\,^\circ$C marca $70\,^\circ$C al cabo de $5$ minutos.
        Determina $k$ y después el tiempo de espera hasta los
        $55\,^\circ$C, que ya se pueden beber.
  \item La cuestión de la leche: para beber el café más caliente posible
        dentro de diez minutos, ¿hay que echar la leche fría ahora o en el
        último momento? Responde con la \omterm{def:g10:algebra:equation}{ecuación} (¿qué le hace a la
        \omterm{def:g11:seq:arithmetic}{diferencia} $T - T_a$ echar la leche, y cómo gobierna esa \omterm{def:g11:seq:arithmetic}{diferencia}
        la pérdida?).
  \item Investigación forense: se encuentra un cuerpo a medianoche a
        $30\,^\circ$C en una habitación a $20\,^\circ$C; una hora después
        marca $28\,^\circ$C. Suponiendo $37\,^\circ$C en el momento de la
        muerte, recupera $k$ a partir de las dos medidas y calcula la hora
        de la muerte.
  \item Audita el reloj del forense: nombra tres incumplimientos reales de
        las hipótesis del modelo y en qué dirección sesgaría cada uno la
        hora estimada de la muerte.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Cuatro disfraces y una curva en S.}
\begin{enumerate}[resume]
  \item La \omterm{def:g10:algebra:equation}{ecuación} logística del \cref{exo:g12:diffeq:8},
        $y' = y(1 - y)$ con $y(0) = 0.1$: usando la sustitución del
        ejercicio, deduce $y(t) = \dfrac{1}{1 + 9\eu^{-t}}$ y halla el
        instante del \omterm{def:g12:deriv:inflection}{punto de inflexión} $y = \frac12$: la fecha del
        epidemiólogo del \cref{pb:g12:deriv:1}, ahora calculable.
  \item Una gota de lluvia cumple $v' = 10 - \frac v2$ (la gravedad menos
        el rozamiento), con $v(0) = 0$. Halla la velocidad límite, resuelve
        para $v(t)$ y calcula cuándo alcanza la gota el $95\,\%$ de esa
        velocidad.
  \item Un gotero de hospital administra un fármaco a ritmo constante
        mientras el cuerpo lo elimina de forma proporcional:
        $c' = 4 - 0.5c$, con $c(0) = 0$. Halla la concentración de régimen
        estacionario y el tiempo necesario para llegar a la mitad. Enuncia
        en una frase el paralelismo con las recurrencias de préstamos del
        \cref{exo:g12:diffeq:9}.
  \item Final: una \omterm{def:g10:algebra:equation}{ecuación} y cuatro disfraces: desintegración,
        enfriamiento, caída y dosificación, todos $y' = ay + b$ con signos
        y nombres distintos. Recita el ciclo de modelización que este
        problema ha \omterm{def:g10:stats:quartiles}{recorrido} cuatro veces (ley $\to$ \omterm{def:g10:algebra:equation}{ecuación} $\to$
        solución $\to$ calibración $\to$ predicción $\to$ auditoría) y
        nombra el fenómeno que necesita $y''$ y el problema que ya se
        encontró con él.
\end{enumerate}
\end{problem}
