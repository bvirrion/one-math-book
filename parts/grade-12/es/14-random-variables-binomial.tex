\chapter{Variables aleatorias y distribución binomial}\label{ch:g12:randvar}

Una \omterm{def:g12:randvar:rv}{variable aleatoria} asocia un número a cada resultado de un
experimento: una ganancia, un recuento, una duración. Su \omterm{def:g12:randvar:exp}{esperanza} es el
promedio a largo plazo de los valores que produce, y su \omterm{def:g12:randvar:exp}{varianza} mide su
dispersión. La estrella de este capítulo es la \omterm{def:g12:randvar:binomial}{distribución binomial}, que
cuenta los éxitos en repeticiones \omterm{def:g12:condprob:indep}{independientes}. Las \omterm{def:g12:randvar:rv}{variables aleatorias}
y la \omterm{def:g12:randvar:binomial}{distribución binomial} aparecieron por primera vez en los
\cref{ch:g11:prob,ch:g11:binom}; este capítulo las repasa y las profundiza,
ahora que disponemos de las herramientas combinatorias del
\cref{ch:g12:comb}.

\section{Variables aleatorias discretas}

\begin{definition}[Variable aleatoria, distribución]\label{def:g12:randvar:rv}
Una \emph{variable aleatoria}\index{variable aleatoria} sobre un \omterm{def:g11:prob:model}{espacio
muestral} finito $\Omega$ es una \omterm{def:g11:func:function}{función} $X \colon \Omega \to \R$. Su
\emph{distribución}\index{distribución} (o ley) es el dato de sus valores
posibles $x_1, \dots, x_k$ y de las \omterm{def:g10:proba:distribution}{probabilidades}
\[
p_i = \P(X = x_i), \qquad \sum_{i=1}^{k} p_i = 1 .
\]
\end{definition}

\begin{definition}[Esperanza, varianza, desviación típica]\label{def:g12:randvar:exp}
La \emph{esperanza}\index{esperanza} de $X$ es
\[
\E(X) = \sum_{i=1}^{k} p_i\, x_i ,
\]
y su \emph{varianza}\index{varianza} y su \emph{\omterm{def:g11:stat:variance}{desviación típica}} son
\[
\V(X) = \E\bigl((X - \E(X))^2\bigr) = \sum_{i=1}^k p_i\bigl(x_i - \E(X)\bigr)^2,
\qquad
\sigma(X) = \sqrt{\V(X)} .
\]
\end{definition}

\begin{proposition}[Fórmula de König--Huygens]\label{prop:g12:randvar:konig}
\index{fórmula de König--Huygens}
$\V(X) = \E(X^2) - \E(X)^2$.
\end{proposition}

\begin{proof}
Escribimos $m = \E(X)$ y desarrollamos:
\[
\V(X) = \sum_i p_i (x_i^2 - 2m x_i + m^2)
= \E(X^2) - 2m\sum_i p_i x_i + m^2\sum_i p_i
= \E(X^2) - 2m^2 + m^2 . \qedhere
\]
\end{proof}

\begin{proposition}[Transformación afín]\label{prop:g12:randvar:affine}
Para $a, b \in \R$:
\[
\E(aX + b) = a\,\E(X) + b, \qquad
\V(aX + b) = a^2\,\V(X) .
\]
\end{proposition}

\begin{proof}
La primera es una reordenación de la suma que define la \omterm{def:g12:randvar:exp}{esperanza}. Para la
segunda, $aX + b$ se desvía de su \omterm{def:g11:stat:mean}{media} en $a(X - \E(X))$, y elevar al
cuadrado multiplica por $a^2$.
\end{proof}

\begin{example}[Juegos equitativos]\label{ex:g12:randvar:game}
Un juego cuesta $2$ euros; se lanza un dado y el jugador recibe el valor
que salga si es al menos $5$, y nada en caso contrario. La ganancia $G$
toma los valores $-2$ (con \omterm{def:g10:proba:distribution}{probabilidad} $\frac46$), $3$ ($\frac16$) y $4$
($\frac16$):
\[
\E(G) = \frac{-8 + 3 + 4}{6} = -\frac{1}{6} < 0 .
\]
En promedio, el jugador pierde $17$ céntimos por partida: el juego es
desfavorable (como lo son casi todos los juegos reales).
\end{example}

\section{Pruebas de Bernoulli y distribución binomial}

\begin{definition}[Distribución de Bernoulli]\label{def:g12:randvar:bernoulli}
Una \emph{prueba de Bernoulli}\index{prueba de Bernoulli} es un
experimento con dos resultados: \emph{éxito} (con \omterm{def:g10:proba:distribution}{probabilidad} $p$) y
\emph{fracaso} ($q = 1 - p$). El indicador $X$ del éxito ($X = 1$ si hay
éxito y $0$ si hay fracaso) sigue la \emph{\omterm{def:g12:randvar:rv}{distribución} de Bernoulli}
$\mathcal B(p)$:
\[
\E(X) = p, \qquad \V(X) = p(1-p) .
\]
(En efecto, $\E(X) = p$ y $\E(X^2) = p$, y König--Huygens da
$\V(X) = p - p^2$.)
\end{definition}

\begin{definition}[Distribución binomial]\label{def:g12:randvar:binomial}
Repetimos $n$ veces, de manera independiente, una \omterm{def:g12:randvar:bernoulli}{prueba de Bernoulli}, y
sea $X$ el número total de éxitos. La \omterm{def:g12:randvar:rv}{distribución} de $X$ es la
\emph{distribución binomial}\index{distribución binomial}
$\mathcal B(n, p)$.
\end{definition}

\begin{theorem}\label{thm:g12:randvar:binomial}
Si $X \sim \mathcal B(n, p)$, entonces, para $0 \leq k \leq n$:
\[
\P(X = k) = \binom{n}{k} p^k (1-p)^{n-k},
\]
y
\[
\E(X) = np, \qquad \V(X) = np(1-p) .
\]
\end{theorem}

\begin{proof}
Una \omterm{def:g12:seq:sequence}{sucesión} concreta de resultados con $k$ éxitos y $n-k$ fracasos tiene
\omterm{def:g10:proba:distribution}{probabilidad} $p^k(1-p)^{n-k}$ por \omterm{def:g12:condprob:indep}{independencia}; y el número de tales
sucesiones es el número de maneras de colocar los $k$ éxitos entre las $n$
pruebas, es decir, $\binom nk$ (\cref{ch:g12:comb}). Sumando sobre las
sucesiones sale la fórmula, y el teorema del binomio confirma que
$\sum_k \P(X = k) = (p + q)^n = 1$.

Para la \omterm{def:g12:randvar:exp}{esperanza}, usando $k\binom nk = n\binom{n-1}{k-1}$
(\cref{exo:g12:comb:7}):
\[
\E(X) = \sum_{k=1}^{n} k\binom nk p^k q^{n-k}
= np\sum_{k=1}^{n} \binom{n-1}{k-1} p^{k-1} q^{(n-1)-(k-1)}
= np\,(p + q)^{n-1} = np .
\]
La fórmula de la \omterm{def:g12:randvar:exp}{varianza} se demuestra igual con la identidad
$k(k-1)\binom nk = n(n-1)\binom{n-2}{k-2}$, que da
$\E(X(X-1)) = n(n-1)p^2$, de donde
$\V(X) = n(n-1)p^2 + np - (np)^2 = np(1-p)$. (Una demostración
estructural, la de la \omterm{def:g12:randvar:exp}{varianza} de una suma de variables \omterm{def:g12:condprob:indep}{independientes},
llegará con el \cref{thm:g12:sums:variance}.)
\end{proof}

\begin{method}[Reconocer una situación binomial]\label{met:g12:randvar:recognize}
Comprueba los tres ingredientes antes de escribir
$X \sim \mathcal B(n,p)$: un \emph{número fijo} $n$ de pruebas; \emph{dos
resultados} por prueba con la \emph{misma} \omterm{def:g10:proba:distribution}{probabilidad} de éxito $p$; e
\emph{\omterm{def:g12:condprob:indep}{independencia}} de las pruebas (muestreo \emph{con} reposición, o
sobre una población grande). Usa después
\[
\P(X \geq 1) = 1 - (1-p)^n
\]
para ``al menos un éxito'', y una calculadora o tablas acumuladas para
$\P(X \leq k)$ en general.
\end{method}

\begin{example}\label{ex:g12:randvar:atleastone}
¿Cuántas veces hay que lanzar un dado para tener al menos un $99\,\%$ de
\omterm{def:g10:proba:distribution}{probabilidad} de sacar un seis? Con
$X \sim \mathcal B\left(n, \frac16\right)$:
$\P(X \geq 1) = 1 - \left(\frac56\right)^n \geq 0.99$ significa
$\left(\frac56\right)^n \leq 0.01$, es decir,
$n \geq \frac{\ln 0.01}{\ln(5/6)} \approx 25.3$: a partir de $n = 26$
lanzamientos.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:randvar:1}
Una \omterm{def:g12:randvar:rv}{variable aleatoria} $X$ toma los valores $-1, 0, 2, 5$ con
\omterm{def:g10:proba:distribution}{probabilidades} $0.3, 0.2, 0.4, 0.1$. Calcula $\E(X)$, $\V(X)$ y
$\sigma(X)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:randvar:2}
Un examen tipo test tiene $10$ preguntas, cada una con $4$ opciones, de las
que una es correcta. Un estudiante responde al azar de manera uniforme e
independiente. Sea $X$ el número de respuestas correctas.
\begin{enumerate}
  \item Da la \omterm{def:g12:randvar:rv}{distribución} de $X$, $\E(X)$ y $\sigma(X)$.
  \item Calcula $\P(X = 0)$, $\P(X = 5)$ y $\P(X \geq 1)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:randvar:3}
Una compañía asegura a $n = 400$ clientes; cada uno presenta un parte
durante el año con \omterm{def:g10:proba:distribution}{probabilidad} $p = 0.05$, de manera independiente. Sea
$X$ el número de partes. Identifica la \omterm{def:g12:randvar:rv}{distribución} de $X$ y calcula su
\omterm{def:g12:randvar:exp}{esperanza} y su \omterm{def:g11:stat:variance}{desviación típica}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:4}
En el juego del \cref{ex:g12:randvar:game}, el organizador quiere que el
juego sea equitativo ($\E(G) = 0$) cambiando el precio de entrada $c$.
Halla $c$. Calcula la \omterm{def:g12:randvar:exp}{varianza} de la ganancia en esa versión equitativa;
¿es lo mismo ``equitativo'' que ``sin riesgo''?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:5}
Una jugadora de baloncesto encesta los tiros libres con \omterm{def:g10:proba:distribution}{probabilidad}
$0.7$. Lanza $8$ veces (tiros \omterm{def:g12:condprob:indep}{independientes}). Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de
que enceste: exactamente $6$; al menos $6$; al menos una vez. ¿Cuál es el
número de canastas más probable?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:6}
Una aerolínea sabe que cada pasajero con reserva se presenta con
\omterm{def:g10:proba:distribution}{probabilidad} $0.9$, de manera independiente. Un vuelo tiene $100$ plazas y
la aerolínea vende $104$ billetes. Expresa, mediante una \omterm{def:g12:randvar:binomial}{distribución
binomial}, la \omterm{def:g10:proba:distribution}{probabilidad} de que se presenten más pasajeros que plazas hay,
y acótala numéricamente con una calculadora (da la expresión exacta).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:randvar:7}
Sea $X \sim \mathcal B(n, p)$. Demuestra que
\[
\frac{\P(X = k+1)}{\P(X = k)} = \frac{n-k}{k+1}\cdot\frac{p}{1-p},
\]
y deduce que la \omterm{def:g12:randvar:rv}{distribución} crece hasta $k^* = \floor{(n+1)p}$ y decrece
después (la \emph{moda} de la binomial).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:randvar:8}
\emph{(San Petersburgo, domesticado.)} Se lanza una moneda equilibrada
hasta que sale cara, pero como mucho $10$ veces. Sea $N$ el número de
lanzamientos empleados; el jugador recibe $2^N$ euros si ha salido cara, y
$0$ en caso contrario.
\begin{enumerate}
  \item Da la \omterm{def:g12:randvar:rv}{distribución} de $N$ restringida a los resultados ganadores:
        $\P(\text{la primera cara sale en el lanzamiento } k) = 2^{-k}$
        para $1 \leq k \leq 10$, y comprueba la \omterm{def:g10:proba:distribution}{probabilidad} total de
        ganar.
  \item Calcula la ganancia esperada. ¿En qué se convertiría sin el tope de
        $10$ lanzamientos?
\end{enumerate}
\end{exercise}

\section{Problema: El vuelo con sobreventa}

\begin{problem}[{Problema de fin de semana --- las aerolíneas venden más
plazas de las que tienen, las fábricas aceptan lotes que apenas han
inspeccionado y la distribución binomial arbitra las dos cosas}]
\label{pb:g12:randvar:1}
Una aerolínea con $100$ plazas vende alegremente $105$ billetes: en torno
al $10\,\%$ de los pasajeros no se presenta nunca, y las plazas vacías no
dan dinero. ¿Hasta dónde puede apretar la aerolínea antes de que los
pasajeros a los que deniega el embarque se coman el beneficio? La
\omterm{def:g12:randvar:binomial}{distribución binomial} (\cref{thm:g12:randvar:binomial}) responde con
decimales, y la misma maquinaria inspecciona lotes de fábrica, pone precio
a las rifas y mantiene solventes a las aseguradoras. Decidir cuando algo se
repite: ese es el oficio diario de la binomial.

\medskip
\noindent\textbf{Parte I --- Soltura.}
\begin{enumerate}
  \item Da la tabla completa de la \omterm{def:g12:randvar:rv}{distribución} del número $X$ de seises en
        tres lanzamientos de un dado.
  \item Calcula $\E(X)$ y $\V(X)$ (\cref{def:g12:randvar:exp}).
  \item Un puesto de feria cobra $1$ euro por tres lanzamientos y paga $2$
        euros por cada seis obtenido. Calcula la ganancia esperada: ¿juego
        equitativo?
  \item Práctica con la lista de comprobación
        (\cref{met:g12:randvar:recognize}): ¿es binomial el número de
        corazones en $5$ cartas sacadas \emph{sin} reposición? ¿Y con
        reposición? Justifícalo.
  \item Para $X \sim \mathcal B(20,\ 0.3)$: da $\E(X)$ y $\sigma(X)$, y
        después $\E(S)$ y $\sigma(S)$ para la puntuación $S = 5X - 10$
        (\cref{prop:g12:randvar:affine}).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El vuelo con sobreventa.}
Plazas: $100$. Billetes vendidos: $105$. Cada titular de un billete se
presenta con \omterm{def:g10:proba:distribution}{probabilidad} $0.9$, de manera independiente; sea $X$ el número
de los que se presentan.
\begin{enumerate}[resume]
  \item Modelo: justifica $X \sim \mathcal B(105,\ 0.9)$ con la lista de
        comprobación, y confiesa el punto más débil del modelo (¿de verdad
        son \omterm{def:g12:condprob:indep}{independientes} las ausencias? piensa en los grupos y en las
        tormentas).
  \item Calcula $\E(X)$ y $\sigma(X)$.
  \item Se deniega el embarque a alguien cuando $X \geq 101$: escribe
        $\P(X \geq 101)$ como suma explícita de cinco términos binomiales.
  \item Evalúa la suma (con calculadora): ¿en qué fracción de los vuelos se
        queda en tierra al menos un pasajero?
  \item El dinero: los cinco billetes de más aportan
        $5 \times 200 = 1\,000$ euros; y cada pasajero al que se deniega el
        embarque cuesta $800$ euros de indemnización. Calcula el número
        esperado de pasajeros dejados en tierra,
        $\sum_{k=101}^{105} (k - 100)\,\P(X = k)$, la indemnización
        esperada y el veredicto sobre la política de la aerolínea.
  \item Diseño: el regulador tolera
        $\P(\text{denegación de embarque}) < 5\,\%$. Probando con
        $n = 105, 106, 107, \dots$ billetes, halla el mayor $n$ permitido.
  \item El atajo de la campana: para $n = 105$, calcula la \omterm{pb:g11:stat:1}{puntuación z} del
        umbral de denegación, $z = \frac{100.5 - \E(X)}{\sigma(X)}$, y
        compara la estimación tosca (``unas $2\sigma$, así que en torno al
        $2$--$3\,\%$ en la cola superior'') con tu respuesta exacta. (La
        curva que hay detrás de este atajo es la estrella de los capítulos
        siguientes.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La puerta de la fábrica.}
Un lote de piezas se acepta si una \omterm{def:g10:proba:sample}{muestra} de $20$ contiene como mucho una
pieza defectuosa.
\begin{enumerate}[resume]
  \item Si la tasa real de piezas defectuosas es del $2\,\%$ (un lote
        honrado), calcula la \omterm{def:g10:proba:distribution}{probabilidad} de que se acepte.
  \item Si la tasa es del $10\,\%$ (un lote malo), vuelve a calcularla.
  \item Nombra los dos riesgos del procedimiento (el lote honrado
        rechazado; el lote malo aceptado), lee sus valores en las preguntas
        13 y 14 y di qué único cambio mejora los dos a la vez, y a qué
        precio.
  \item Detrás de esa mejora: demuestra que la \emph{frecuencia} observada
        de piezas defectuosas $\frac Xn$ en una \omterm{def:g10:proba:sample}{muestra} de tamaño $n$ tiene
        \omterm{def:g11:stat:variance}{desviación típica} $\sqrt{\frac{p(1-p)}{n}}$, y concluye cómo escala
        la precisión con el tamaño de la \omterm{def:g10:proba:sample}{muestra} (una vieja conocida: el
        $\frac{1}{\sqrt n}$ del año 10).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Juegos, rifas y reservas.}
\begin{enumerate}[resume]
  \item El juego de San Petersburgo con tope del \cref{exo:g12:randvar:8}
        tiene una ganancia esperada de $10$ euros. Contrástalo en un
        párrafo con la paradoja sin tope que apareció en el
        \cref{pb:g11:prob:1}: ¿qué domestica exactamente el tope y qué
        precio sería ahora equitativo?
  \item Una rifa benéfica vende $200$ papeletas a $2$ euros; premios: una
        cesta de $100$ euros y dos de $50$ euros. Calcula la ganancia
        esperada de una papeleta y el margen de la rifa; compáralo con el
        margen de la banca en la ruleta europea, de
        $\frac{1}{37} \approx 2.7\,\%$. ¿Por qué le sale gratis a la rifa?
  \item Una aseguradora tiene $1\,000$ pólizas \omterm{def:g12:condprob:indep}{independientes}: \omterm{def:g10:proba:distribution}{probabilidad}
        de siniestro $0.01$ cada una, cuantía del siniestro $10\,000$ euros
        y prima $130$ euros. Calcula el beneficio anual esperado \emph{y}
        la \omterm{def:g11:stat:variance}{desviación típica} del total de siniestros. Compara los dos
        números: ¿a qué obliga esa comparación a las aseguradoras reales?
  \item Final: la binomial como árbitro de decisiones: la lista de
        comprobación para reconocerla; la brújula $\E \pm \sigma$; las
        \omterm{def:g10:proba:distribution}{probabilidades} de cola como precio del riesgo; y las dos
        advertencias permanentes (la \omterm{def:g12:condprob:indep}{independencia} es una afirmación sobre
        el modelo, y lo que arruina son las colas raras, no las \omterm{def:g11:stat:mean}{medias}).
        Una frase para cada cosa, más la mirada hacia delante: la ley de
        los grandes números y la campana, en los capítulos siguientes,
        completan el reglamento del árbitro.
\end{enumerate}
\end{problem}
