\chapter{Variables aleatorias y distribución binomial}\label{ch:g12:randvar}

A variable aleatoria asigna un número a cada resultado de un experimento: a
ganancia, una cuenta, una duración. Es \omterm{def:g12:randvar:exp}{esperanza} es el promedio de largo plazo de la
valores que produce, su varianza mide su propagación. La estrella de este
el capitulo es el \omterm{def:g12:randvar:rv}{distribución} binomial, que cuenta los éxitos en repetidas
\omterm{def:g12:condprob:indep}{independientes} ensayos. \omterm{def:g12:randvar:rv}{variables aleatorias} y el \omterm{def:g12:randvar:rv}{distribución} binomial fueron
se conocieron por primera vez en \cref{ch:g11:prob,ch:g11:binom}; este capítulo revisa y
los profundiza, con las herramientas combinatorias de \cref{ch:g12:comb} ahora
disponible.

\section{Variables aleatorias discretas}

\begin{definition}[Variable aleatoria, distribución]\label{def:g12:randvar:rv}
A \emph{variables aleatorias}\index{variables aleatorias} en un finito \omterm{def:g10:proba:events}{espacio muestral}
$\Omega $ es una \omterm{def:g11:func:function}{función} $ X \colon \Omega \to \R $. Es
\emph{distribución}\index{distribución} (o ley) son los datos de su posible
valores $ x_1, \dots, x_k $ y de las \omterm{def:g10:proba:distribution}{probabilidades}
\[
p_i = \P(X = x_i), \qquad \sum_{i=1}^{k} p_i = 1 .
\]
\end{definition}

\begin{definition}[Esperanza, varianza, desviación estándar]\label{def:g12:randvar:exp}
El \emph{esperanza}\index{esperanza} de $ X $ es
\[
\E(X) = \sum_{i=1}^{k} p_i\, x_i ,
\]
es \emph{diferencia}\index{diferencia} y \emph{desviación estándar} son
\[
\V(X) = \E\bigl((X - \E(X))^2\bigr) = \sum_{i=1}^k p_i\bigl(x_i - \E(X)\bigr)^2,
\qquad
\sigma(X) = \sqrt{\V(X)} .
\]
\end{definition}

\begin{proposition}[Fórmula de König-Huygens]\label{prop:g12:randvar:konig}
\index{Fórmula de König-Huygens}
$\V(X) = \E(X^2) - \E(X)^2$.
\end{proposition}

\begin{proof}
Escribir $ m = \E(X)$ y expandir:
\[
\V(X) = \sum_i p_i (x_i^2 - 2m x_i + m^2)
= \E(X^2) - 2m\sum_i p_i x_i + m^2\sum_i p_i
= \E(X^2) - 2m^2 + m^2 . \qedhere
\]
\end{proof}

\begin{proposition}[Transformación afín]\label{prop:g12:randvar:affine}
Para $ a, b \in \R $:
\[
\E(aX + b) = a\,\E(X) + b, \qquad
\V(aX + b) = a^2\,\V(X) .
\]
\end{proposition}

\begin{proof}
La primera es una reordenación de la suma definitoria. Para el segundo, $ aX + b $
se desvía de su \omterm{def:g11:stat:mean}{media} por $ a(X - \E(X))$, y elevando al cuadrado se multiplica por
$ a^2$.
\end{proof}

\begin{example}[Juegos justos]\label{ex:g12:randvar:game}
Un juego cuesta $2$ euros; Se lanza un dado y el jugador recibe el valor.
se \omterm{def:g10:proba:sample}{muestra} si es al menos $5$, nada más. la ganancia $ G $ toma valores
$-2$ (\omterm{def:g10:proba:distribution}{probabilidad} $\frac46$), $3$ ($\frac16$), $4$ ($\frac16$):
\[
\E(G) = \frac{-8 + 3 + 4}{6} = -\frac{1}{6} < 0 .
\]
En promedio, el jugador pierde $17$ centavos por juego: el juego es desfavorable
(como lo son la mayoría de los juegos reales).
\end{example}

\section{Ensayos de Bernoulli y distribución binomial.}

\begin{definition}[Distribución de Bernoulli]\label{def:g12:randvar:bernoulli}
A \emph{prueba de Bernoulli}\index{prueba de Bernoulli} es un experimento con dos
resultados, \emph{éxito} (\omterm{def:g10:proba:distribution}{probabilidad} $ p $) y \emph{falla}
($ q = 1 - p $). El indicador $ X $ de éxito ($ X = 1$ sobre el éxito, $0$ en
fracaso) sigue el \emph{bernoulli \omterm{def:g12:randvar:rv}{distribución}} $\mathcal B(p)$:
\[
\E(X) = p, \qquad \V(X) = p(1-p) .
\]
(En efecto $\E(X) = p $, $\E(X^2) = p $, y König--Huygens da
$\V(X) = p - p^2$.)
\end{definition}

\begin{definition}[Distribución binomial]\label{def:g12:randvar:binomial}
repetir un \omterm{def:g12:randvar:bernoulli}{prueba de Bernoulli} $ n $ veces de forma \omterm{def:g12:condprob:indep}{independiente} y dejar $ X $ ser el total
número de éxitos. El \omterm{def:g12:randvar:rv}{distribución} de $ X $ es el \emph{binomio
distribución}\index{binomio
distribución} $\mathcal B(n, p)$.
\end{definition}

\begin{theorem}\label{thm:g12:randvar:binomial}
Si $ X \sim \mathcal B(n, p)$, entonces para $0 \leq k \leq n $:
\[
\P(X = k) = \binom{n}{k} p^k (1-p)^{n-k},
\]
y
\[
\E(X) = np, \qquad \V(X) = np(1-p) .
\]
\end{theorem}

\begin{proof}
un especificado \omterm{def:g12:seq:sequence}{sucesión} de resultados con $ k $ éxitos y $ n-k $ fracasos tiene
\omterm{def:g10:proba:distribution}{probabilidad} $ p^k(1-p)^{n-k}$ por independencia; el número de tales sucesiones
es el número de formas de colocar el $ k $ éxitos entre los $ n $ ensayos,
es decir $\binom nk $ (\cref{ch:g12:comb}). resumiendo el sucesiones da
la fórmula --- y el teorema del binomio lo confirma
$\sum_k \P(X = k) = (p + q)^n = 1$.

Para la \omterm{def:g12:randvar:exp}{esperanza}, usando $ k\binom nk = n\binom{n-1}{k-1}$
(\cref{exo:g12:comb:7}):
\[
\E(X) = \sum_{k=1}^{n} k\binom nk p^k q^{n-k}
= np\sum_{k=1}^{n} \binom{n-1}{k-1} p^{k-1} q^{(n-1)-(k-1)}
= np\,(p + q)^{n-1} = np .
\]
La varianza La fórmula se demuestra de manera similar con la identidad.
$ k(k-1)\binom nk = n(n-1)\binom{n-2}{k-2}$, donación
$\E(X(X-1)) = n(n-1)p^2$, de donde
$\V(X) = n(n-1)p^2 + np - (np)^2 = np(1-p)$. (Una prueba estructural --- la
varianza de una suma de \omterm{def:g12:condprob:indep}{independientes} variables --- viene con
\cref{thm:g12:sums:variance}.)
\end{proof}

\begin{method}[Reconocer una situación binomial]\label{met:g12:randvar:recognize}
Comprueba los tres ingredientes antes de escribir. $ X \sim \mathcal B(n,p)$: a
\emph{numero fijo} $ n $ de juicios; \emph{dos resultados} por ensayo con el
\emph{mismo} éxito \omterm{def:g10:proba:distribution}{probabilidad} $ p $; \emph{independencia} de ensayos
(muestreo \emph{con} reemplazo, o de una población grande). Entonces usa
\[
\P(X \geq 1) = 1 - (1-p)^n
\]
para "al menos un éxito", y una calculadora o tablas acumulativas para
generales $\P(X \leq k)$.
\end{method}

\begin{example}\label{ex:g12:randvar:atleastone}
¿Cuántas veces se debe lanzar un dado para tener al menos una $99\%$ posibilidad de
sacar un seis? Con $ X \sim \mathcal B\left(n, \frac16\right)$:
$\P(X \geq 1) = 1 - \left(\frac56\right)^n \geq 0.99$ medio
$\left(\frac56\right)^n \leq 0.01$, \emph{es decir.}
$ n \geq \frac{\ln 0.01}{\ln(5/6)} \approx 25.3$: de $ n = 26$ sigue adelante.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g12:randvar:1}
A variable aleatoria $ X $ toma los valores $-1, 0, 2, 5$ con \omterm{def:g10:proba:distribution}{probabilidades}
$0.3, 0.2, 0.4, 0.1$. Calcular $\E(X)$, $\V(X)$ y $\sigma(X)$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g12:randvar:2}
Una prueba de opción múltiple tiene $10$ preguntas, cada una con $4$ opciones, una de
lo cual es correcto. Un estudiante responde uniformemente al azar, de forma \omterm{def:g12:condprob:indep}{independiente}.
Sea $ X $ Sea el número de respuestas correctas.
\begin{enumerate}
  \item Dale el \omterm{def:g12:randvar:rv}{distribución} de $ X $, $\E(X)$ y $\sigma(X)$.
  \item Calcular $\P(X = 0)$, $\P(X = 5)$ y $\P(X \geq 1)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g12:randvar:3}
Una compañía de seguros asegura $ n = 400$ clientela; cada uno presenta un reclamo durante
el año con \omterm{def:g10:proba:distribution}{probabilidad} $ p = 0.05$, de forma \omterm{def:g12:condprob:indep}{independiente}. Sea $ X $ ser el numero
de reclamaciones. Identificar el \omterm{def:g12:randvar:rv}{distribución} de $ X $ y calcular su \omterm{def:g12:randvar:exp}{esperanza} y
desviación estándar.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:randvar:4}
en el juego de \cref{ex:g12:randvar:game}, el organizador quiere un juego limpio
($\E(G) = 0$) cambiando el precio de entrada $ c $. Encontrar $ c $. Calcular el

¿"sin riesgos"?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:randvar:5}
Un jugador de baloncesto anota tiros libres con \omterm{def:g10:proba:distribution}{probabilidad} $0.7$. ella dispara
$8$ veces (\omterm{def:g12:condprob:indep}{independientes} disparos). Calcular el \omterm{def:g10:proba:distribution}{probabilidad} que ella anota:
exactamente $6$; al menos $6$; al menos una vez. ¿Cuál es el número más probable?
de puntuaciones?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:randvar:6}
Una aerolínea sabe que cada pasajero reservado se presenta con \omterm{def:g10:proba:distribution}{probabilidad}
$0.9$, de forma \omterm{def:g12:condprob:indep}{independiente}. un vuelo tiene $100$ asientos y la aerolínea vende $104$

Aparecen más pasajeros que asientos, y lo unimos numéricamente usando un

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:randvar:7}
Sea $ X \sim \mathcal B(n, p)$. mostrar que
\[
\frac{\P(X = k+1)}{\P(X = k)} = \frac{n-k}{k+1}\cdot\frac{p}{1-p},
\]
y deducir que el \omterm{def:g12:randvar:rv}{distribución} aumenta hasta
$ k^* = \floor{(n+1)p}$ y disminuye después (el \emph{modo} de la
binomio).
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g12:randvar:8}

        $1 \leq k \leq 10$ y verifique el total \omterm{def:g10:proba:distribution}{probabilidad} de ganar.
  \item Calcule el pago esperado. ¿Qué sería sin la gorra?
        de $10$ lanzamientos?
\end{enumerate}
\end{exercise}

\section{Problema: el vuelo con overbooking}

\begin{problem}[{Problema del fin de semana: las aerolíneas venden más asientos que
tienen, las fábricas aceptan lotes que apenas inspeccionaron y el
árbitros de distribución binomial ambos}]
\label{pb:g12:randvar:1}
una aerolínea con $100$ asientos felizmente se vende $105$ entradas: sobre
$10\,\%$ de los pasajeros nunca se presentan y los asientos vacíos ganan

comerse las ganancias? El \omterm{def:g12:randvar:rv}{distribución} binomial
(\cref{thm:g12:randvar:binomial}) respuestas al decimal --- y
la misma maquinaria inspecciona lotes de fábrica, cotiza sorteos y
