\chapter{Sucesiones}\label{ch:g12:seq}

Una \omterm{def:g12:seq:sequence}{sucesión} es una lista de \omterm{def:g10:numbers:sets}{números reales} indexada por los \omterm{def:g10:numbers:sets}{números
naturales}. Las sucesiones modelizan evoluciones discretas (poblaciones
contadas año a año, saldos de una cuenta bancaria, aproximaciones sucesivas
de un número) y sus límites son el primer encuentro serio con el infinito.
Este capítulo monta el vocabulario, el principio de inducción y los
teoremas fundamentales de convergencia.

\section{El razonamiento por inducción}

\begin{theorem}[Principio de inducción]\label{thm:g12:seq:induction}
Sea $P(n)$ una propiedad que depende de un entero $n$ y sea
$n_0 \in \N$. Si
\begin{enumerate}
  \item \emph{(caso base)} $P(n_0)$ es cierta y
  \item \emph{(paso de inducción)} para todo $n \geq n_0$, $P(n)$ implica
        $P(n+1)$,
\end{enumerate}
entonces $P(n)$ es cierta para todo $n \geq n_0$.
\end{theorem}

\begin{proof}
Supongamos, por reducción al absurdo, que el conjunto $A$ de los enteros
$n \geq n_0$ para los que $P(n)$ es falsa no es vacío. Entonces $A$ tiene
un elemento \omterm{def:g10:functions:extrema}{mínimo} $m$.\footnote{Todo subconjunto no vacío de $\N$ tiene un
elemento \omterm{def:g10:functions:extrema}{mínimo}; esta propiedad de $\N$ se toma como axioma.} Como
$P(n_0)$ es cierta, $m > n_0$, luego $m - 1 \geq n_0$ y $m-1 \notin A$, es
decir, $P(m-1)$ es cierta. El paso de inducción aplicado a $n = m-1$
muestra entonces que $P(m)$ es cierta, en contradicción con $m \in A$.
\end{proof}

\begin{example}\label{ex:g12:seq:bernoulli}
Demostremos la \emph{desigualdad de Bernoulli}: para todo real $a > 0$ y
todo $n \in \N$,
\[
(1+a)^n \geq 1 + na.
\]
\emph{Caso base.} Para $n = 0$, los dos miembros valen $1$.
\emph{Paso de inducción.} Supongamos $(1+a)^n \geq 1+na$ para cierto
$n \in \N$. Como $1 + a > 0$, multiplicar los dos miembros por $1+a$
conserva la desigualdad:
\[
(1+a)^{n+1} \geq (1+na)(1+a) = 1 + (n+1)a + na^2 \geq 1 + (n+1)a .
\]
Por inducción, la desigualdad vale para todo $n \in \N$.
\end{example}

\begin{method}[Escribir una demostración por inducción]\label{met:g12:seq:induction}
Explicita siempre la propiedad $P(n)$ antes de empezar. Una demostración
completa tiene tres partes visibles: el caso base, el paso de inducción
(``suponemos $P(n)$; demostramos $P(n+1)$'') y la conclusión invocando el
principio de inducción. El error más frecuente es demostrar el paso de
inducción sin usar nunca la hipótesis $P(n)$: si eso ocurre, o la
demostración está mal, o no hacía falta inducción.
\end{method}

\section{Vocabulario de las sucesiones}

\begin{definition}[Sucesión]\label{def:g12:seq:sequence}
Una \emph{sucesión}\index{sucesión} es una \omterm{def:g11:func:function}{función} $u \colon \N \to \R$ (o
desde $\{n \in \N : n \geq n_0\}$ hacia $\R$). La \omterm{def:g10:functions:function}{imagen} de $n$ se escribe
$u_n$, y la sucesión misma, $(u_n)_{n\in\N}$ o simplemente $(u_n)$.
\end{definition}

Una \omterm{def:g12:seq:sequence}{sucesión} se puede definir de forma \emph{explícita}, mediante una
fórmula $u_n = f(n)$, o por \emph{recurrencia}, mediante su primer término
y una relación $u_{n+1} = f(u_n)$.

\begin{definition}[Monotonía]\label{def:g12:seq:monotonic}
Una \omterm{def:g12:seq:sequence}{sucesión} $(u_n)$ es \emph{creciente}\index{sucesión!creciente} si
$u_{n+1} \geq u_n$ para todo $n$, \emph{\omterm{def:g10:functions:variations}{decreciente}} si
$u_{n+1} \leq u_n$ para todo $n$, y \emph{monótona} si es creciente o
\omterm{def:g10:functions:variations}{decreciente}. Es \emph{estrictamente} creciente (resp.\ \omterm{def:g10:functions:variations}{decreciente}) cuando
las desigualdades son estrictas.
\end{definition}

\begin{method}[Estudiar la monotonía de una sucesión]\label{met:g12:seq:monotonicity}
Tres técnicas habituales:
\begin{enumerate}
  \item estudiar el signo de $u_{n+1} - u_n$;
  \item si todos los términos son positivos, comparar
        $\dfrac{u_{n+1}}{u_n}$ con $1$;
  \item si $u_n = f(n)$ con $f$ definida en $\intco{0}{+\infty}$, usar la
        variación de $f$.
\end{enumerate}
\end{method}

\begin{definition}[Sucesión acotada]\label{def:g12:seq:bounded}
Una \omterm{def:g12:seq:sequence}{sucesión} $(u_n)$ está \emph{acotada superiormente}\index{sucesión!acotada}
si existe $M \in \R$ con $u_n \leq M$ para todo $n$; está \emph{acotada
inferiormente} si existe $m \in \R$ con $u_n \geq m$ para todo $n$; y está
\emph{acotada} si se cumplen las dos cosas.
\end{definition}

\subsection{Progresiones aritméticas y geométricas}

\begin{definition}[Progresiones aritméticas y geométricas]\label{def:g12:seq:arith-geom}
Una \omterm{def:g12:seq:sequence}{sucesión} $(u_n)$ es una \emph{progresión
aritmética}\index{sucesión!aritmética} de \omterm{def:g11:seq:arithmetic}{diferencia} $r$ si
$u_{n+1} = u_n + r$ para todo $n$, y una \emph{progresión
geométrica}\index{sucesión!geométrica} de \omterm{def:g11:seq:geometric}{razón} $q$ si $u_{n+1} = q\,u_n$
para todo $n$.
\end{definition}

\begin{proposition}[Forma explícita y sumas]\label{prop:g12:seq:arith-geom}
Sea $n \in \N$.
\begin{enumerate}
  \item Si $(u_n)$ es una \omterm{def:g12:seq:arith-geom}{progresión aritmética} de \omterm{def:g11:seq:arithmetic}{diferencia} $r$, entonces
        $u_n = u_0 + nr$ y
        \[
        u_0 + u_1 + \dots + u_n = (n+1)\,\frac{u_0 + u_n}{2}.
        \]
  \item Si $(u_n)$ es una \omterm{def:g12:seq:arith-geom}{progresión geométrica} de \omterm{def:g11:seq:geometric}{razón} $q \neq 1$,
        entonces $u_n = u_0\, q^n$ y
        \[
        u_0 + u_1 + \dots + u_n = u_0\,\frac{1 - q^{n+1}}{1 - q}.
        \]
\end{enumerate}
\end{proposition}

\begin{proof}
Las formas explícitas se siguen de inducciones inmediatas. Para la suma
aritmética, escribimos $S = u_0 + \dots + u_n$ y le sumamos la misma suma
escrita al revés: cada una de las $n+1$ columnas suma $u_0 + u_n$, luego
$2S = (n+1)(u_0+u_n)$. Para la suma geométrica, calculamos $S - qS$: todos
los términos se cancelan por parejas salvo el primero y el último, así que
$(1-q)S = u_0(1 - q^{n+1})$.
\end{proof}

\section{Límite de una sucesión}

\begin{definition}[Sucesión convergente]\label{def:g12:seq:limit}
Una \omterm{def:g12:seq:sequence}{sucesión} $(u_n)$ \emph{converge}\index{límite!de una sucesión} al
\omterm{def:g10:numbers:sets}{número real} $\ell$ si todo \omterm{def:g10:numbers:interval}{intervalo} abierto que contiene a $\ell$ contiene
todos los términos $u_n$ a partir de cierto índice. Escribimos entonces
$\lim\limits_{n\to+\infty} u_n = \ell$.

Equivalentemente: para todo $\varepsilon > 0$ existe $N \in \N$ tal que,
para todo $n \geq N$, $\abs{u_n - \ell} \leq \varepsilon$.
\end{definition}

\begin{definition}[Divergencia a infinito]\label{def:g12:seq:infinity}
La \omterm{def:g12:seq:sequence}{sucesión} $(u_n)$ \emph{tiende a $+\infty$} si para todo $A \in \R$
existe $N \in \N$ tal que $u_n \geq A$ para todo $n \geq N$. Escribimos
$\lim\limits_{n\to+\infty} u_n = +\infty$; la definición de
$\lim u_n = -\infty$ es análoga. De una \omterm{def:g12:seq:sequence}{sucesión} que no \omterm{def:g12:seq:limit}{converge} se dice
que \emph{diverge}.
\end{definition}

\begin{remark}
Una \omterm{def:g12:seq:sequence}{sucesión} puede divergir sin tender a $\pm\infty$: la \omterm{def:g12:seq:sequence}{sucesión}
$u_n = (-1)^n$ solo toma los valores $1$ y $-1$ y no tiene límite.
\end{remark}

\begin{proposition}[Unicidad del límite]\label{prop:g12:seq:uniqueness}
Si $(u_n)$ \omterm{def:g12:seq:limit}{converge}, su límite es único.
\end{proposition}

\begin{proof}
Supongamos $u_n \to \ell$ y $u_n \to \ell'$ con $\ell \neq \ell'$, digamos
$\ell < \ell'$. Tomemos $\varepsilon = \frac{\ell' - \ell}{3} > 0$. A
partir de cierto índice, $\abs{u_n - \ell} \leq \varepsilon$ y
$\abs{u_n - \ell'} \leq \varepsilon$, luego
\[
\ell' - \ell \leq \abs{\ell' - u_n} + \abs{u_n - \ell}
\leq 2\varepsilon = \tfrac{2}{3}(\ell' - \ell) < \ell' - \ell,
\]
lo que es una contradicción.
\end{proof}

\begin{proposition}[Operaciones con límites]\label{prop:g12:seq:operations}
Sean $(u_n)$ y $(v_n)$ sucesiones de límites $\ell$ y $\ell'$ (finitos o
infinitos). Entonces, siempre que el segundo miembro no sea una
indeterminación,
\[
\lim (u_n + v_n) = \ell + \ell', \qquad
\lim (u_n v_n) = \ell\,\ell', \qquad
\lim \frac{u_n}{v_n} = \frac{\ell}{\ell'}.
\]
Las indeterminaciones son $(+\infty) + (-\infty)$, $0 \times \infty$,
$\frac{\infty}{\infty}$ y $\frac{0}{0}$.
\end{proposition}

\begin{proof}
Demostramos la regla de la suma para límites finitos; los demás casos son
análogos y se dejan como ejercicio. Sea $\varepsilon > 0$. Existen
$N_1, N_2$ tales que $\abs{u_n - \ell} \leq \varepsilon/2$ para
$n \geq N_1$ y $\abs{v_n - \ell'} \leq \varepsilon/2$ para $n \geq N_2$.
Para $n \geq \max(N_1, N_2)$, la desigualdad triangular da
\[
\abs{(u_n + v_n) - (\ell + \ell')}
\leq \abs{u_n - \ell} + \abs{v_n - \ell'} \leq \varepsilon. \qedhere
\]
\end{proof}

\begin{method}[Levantar una indeterminación]\label{met:g12:seq:indeterminate}
Ante una indeterminación, saca como factor el término dominante. Por
ejemplo,
\[
n^2 - n = n^2\left(1 - \tfrac{1}{n}\right) \xrightarrow[n\to+\infty]{} +\infty,
\qquad
\frac{2n^2+1}{n^2 - n} = \frac{2 + 1/n^2}{1 - 1/n}
\xrightarrow[n\to+\infty]{} 2 .
\]
\end{method}

\section{Teoremas de convergencia}

\begin{theorem}[Teoremas de comparación y del sándwich]\label{thm:g12:seq:squeeze}
Sean $(u_n)$, $(v_n)$, $(w_n)$ sucesiones.
\begin{enumerate}
  \item Si $u_n \leq v_n$ a partir de cierto índice y $u_n \to +\infty$,
        entonces $v_n \to +\infty$.
  \item \emph{(Teorema del sándwich)}\index{teorema del sándwich} Si
        $u_n \leq v_n \leq w_n$ a partir de cierto índice y $(u_n)$ y
        $(w_n)$ convergen las dos al mismo límite $\ell$, entonces $(v_n)$
        \omterm{def:g12:seq:limit}{converge} a~$\ell$.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} Sea $A \in \R$. Como $u_n \to +\infty$, existe $N$ con
$u_n \geq A$ para $n \geq N$; agrandando $N$ si hace falta,
$v_n \geq u_n \geq A$ para $n \geq N$.

\emph{2.} Sea $\varepsilon > 0$. A partir de cierto índice se cumplen a la
vez $\ell - \varepsilon \leq u_n$ y $w_n \leq \ell + \varepsilon$, luego
$\ell - \varepsilon \leq u_n \leq v_n \leq w_n \leq \ell + \varepsilon$, es
decir, $\abs{v_n - \ell} \leq \varepsilon$.
\end{proof}

\begin{example}
Para todo $n \geq 1$,
$-\frac{1}{n} \leq \frac{(-1)^n}{n} \leq \frac{1}{n}$, y las dos cotas
tienden a $0$; por tanto, $\frac{(-1)^n}{n} \to 0$.
\end{example}

\begin{theorem}[Teorema de la convergencia monótona]\label{thm:g12:seq:monotone}
Una \omterm{def:g12:seq:sequence}{sucesión} \omterm{def:g12:seq:monotonic}{creciente} y \omterm{def:g12:seq:bounded}{acotada superiormente} \omterm{def:g12:seq:limit}{converge}. Una \omterm{def:g12:seq:sequence}{sucesión}
\omterm{def:g10:functions:variations}{decreciente} y \omterm{def:g12:seq:bounded}{acotada inferiormente} \omterm{def:g12:seq:limit}{converge}. Una \omterm{def:g12:seq:sequence}{sucesión} \omterm{def:g12:seq:monotonic}{creciente} y no
\omterm{def:g12:seq:bounded}{acotada superiormente} tiende a $+\infty$.
\end{theorem}

\begin{proof}[Demostración parcial]
Demostramos la tercera afirmación. Sea $(u_n)$ \omterm{def:g12:seq:monotonic}{creciente} y no \omterm{def:g12:seq:bounded}{acotada
superiormente}, y sea $A \in \R$. Como $A$ no es una cota superior, existe
$N$ con $u_N \geq A$; por monotonía, $u_n \geq u_N \geq A$ para todo
$n \geq N$. Luego $u_n \to +\infty$.

Las dos afirmaciones de convergencia se apoyan en la propiedad del
supremo de $\R$; se \emph{admiten a este nivel} (y se demuestran en el
primer año universitario).
\end{proof}

\begin{remark}
El teorema garantiza la \emph{existencia} del límite, pero no da su valor.
Una \omterm{def:g12:seq:sequence}{sucesión} \omterm{def:g12:seq:monotonic}{creciente} \omterm{def:g12:seq:bounded}{acotada superiormente} por $M$ \omterm{def:g12:seq:limit}{converge} a algún
$\ell \leq M$, no necesariamente a $M$.
\end{remark}

\begin{theorem}[Límite de las progresiones geométricas]\label{thm:g12:seq:geometric}
Sea $q \in \R$.
\begin{enumerate}
  \item Si $q > 1$, entonces $q^n \to +\infty$.
  \item Si $q = 1$, entonces $q^n \to 1$.
  \item Si $\abs{q} < 1$, entonces $q^n \to 0$.
  \item Si $q \leq -1$, entonces $(q^n)$ diverge y no tiene límite.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} Escribimos $q = 1 + a$ con $a > 0$. La desigualdad de Bernoulli
(\cref{ex:g12:seq:bernoulli}) da $q^n \geq 1 + na \to +\infty$, y
concluimos por comparación (\cref{thm:g12:seq:squeeze}).

\emph{2.} Inmediato.

\emph{3.} Si $q = 0$, la afirmación es clara. Si no, de $\abs{q} < 1$ se
sigue $1/\abs{q} > 1$, luego $(1/\abs{q})^n \to +\infty$ por el punto 1, y
por tanto $\abs{q}^n \to 0$; y $-\abs{q}^n \leq q^n \leq \abs{q}^n$ permite
concluir por el teorema del sándwich.

\emph{4.} Para $q \leq -1$, $(q^{2n})$ toma valores $\geq 1$ mientras que
$(q^{2n+1})$ toma valores $\leq -1$: ningún límite único puede atraer a
las dos subsucesiones.
\end{proof}

\begin{method}[Sucesiones recurrentes $u_{n+1} = f(u_n)$]\label{met:g12:seq:recurrent}
Para estudiar una \omterm{def:g12:seq:sequence}{sucesión} definida por $u_{n+1} = f(u_n)$:
\begin{enumerate}
  \item demuestra por inducción que $(u_n)$ se mantiene en un \omterm{def:g10:numbers:interval}{intervalo} $I$
        en el que $f$ se comporta bien (y, a menudo, que $(u_n)$ es
        \omterm{def:g12:seq:monotonic}{monótona});
  \item deduce la convergencia del teorema de la convergencia \omterm{def:g12:seq:monotonic}{monótona};
  \item pasa al límite en la relación $u_{n+1} = f(u_n)$: si $f$ es
        continua y $u_n \to \ell \in I$, entonces $\ell$ cumple
        $f(\ell) = \ell$ (véase el \cref{ch:g12:limcont}); resuelve esa
        \omterm{def:g10:algebra:equation}{ecuación} y elige la \omterm{def:g11:quad:discriminant}{raíz} correcta.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g12:seq:sqrt2}
Sean $u_0 = 2$ y $u_{n+1} = \frac{1}{2}\left(u_n + \frac{2}{u_n}\right)$.
Se comprueba por inducción que $u_n \geq \sqrt{2}$ para todo $n$ (la
desigualdad $\frac{1}{2}(x + 2/x) \geq \sqrt{2}$ para $x>0$ equivale a
$(x - \sqrt2)^2 \geq 0$) y después que $(u_n)$ es \omterm{def:g10:functions:variations}{decreciente}, ya que
\[
u_{n+1}-u_n=\frac{2-u_n^2}{2u_n}\leq 0 .
\]
\omterm{def:g10:functions:variations}{Decreciente} y \omterm{def:g12:seq:bounded}{acotada inferiormente}, $(u_n)$ \omterm{def:g12:seq:limit}{converge} a cierto
$\ell \geq \sqrt{2}$, que debe cumplir $\ell = \frac{1}{2}(\ell + 2/\ell)$,
es decir, $\ell^2 = 2$. Por tanto, $u_n \to \sqrt{2}$. Es el algoritmo de
Herón, ya usado por los babilonios; su convergencia es extremadamente
rápida ($u_3$ da ya $\sqrt 2$ con ocho decimales exactos).
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:seq:1}
Demuestra por inducción que para todo $n \in \N$,
\[
1^2 + 2^2 + \dots + n^2 = \frac{n(n+1)(2n+1)}{6}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:2}
Estudia la monotonía de las sucesiones definidas para $n \geq 1$ por
\[
a_n = \frac{n+1}{n}, \qquad
b_n = \frac{2^n}{n}, \qquad
c_n = n^2 - 10n .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:3}
Calcula los límites de las sucesiones de términos generales
\[
u_n = \frac{3n^2 - n + 1}{2n^2 + 5}, \qquad
v_n = \sqrt{n+1} - \sqrt{n}, \qquad
w_n = \frac{2^n - 3^n}{3^n + 1}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:4}
Sea $(u_n)$ la \omterm{def:g12:seq:arith-geom}{progresión aritmética} con $u_0 = 5$ y \omterm{def:g11:seq:arithmetic}{diferencia} $r = 3$, y
$(v_n)$ la \omterm{def:g12:seq:arith-geom}{progresión geométrica} con $v_0 = 8$ y \omterm{def:g11:seq:geometric}{razón} $q = \frac{1}{2}$.
Calcula $u_n$, $v_n$, $\sum_{k=0}^{n} u_k$ y $\sum_{k=0}^{n} v_k$, y los
límites de las cuatro expresiones cuando $n \to +\infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:5}
Usando el teorema del sándwich, calcula
\[
\lim_{n\to+\infty} \frac{n + \cos n}{n + 1}
\qquad\text{y}\qquad
\lim_{n\to+\infty} \frac{n!}{n^n},
\]
donde $n! = 1 \times 2 \times \dots \times n$. Para el segundo límite,
acota $\frac{n!}{n^n}$ por un término de una \omterm{def:g12:seq:arith-geom}{progresión geométrica}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:6}
Sean $u_0 = 0$ y $u_{n+1} = \sqrt{u_n + 2}$ para todo $n \in \N$.
\begin{enumerate}
  \item Demuestra por inducción que $0 \leq u_n \leq 2$ para todo $n$.
  \item Demuestra que $(u_n)$ es \omterm{def:g12:seq:monotonic}{creciente}.
  \item Deduce que $(u_n)$ \omterm{def:g12:seq:limit}{converge} y determina su límite.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:7}
Un paciente toma una dosis de $1$ unidad de un medicamento cada mañana. En
cada periodo de 24 horas, el organismo elimina el $40\,\%$ del medicamento
presente. Sea $u_n$ la cantidad de medicamento en el cuerpo justo después
de la dosis del día $n$, de modo que $u_0 = 1$.
\begin{enumerate}
  \item Justifica que $u_{n+1} = 0.6\,u_n + 1$.
  \item Sea $v_n = u_n - 2.5$. Demuestra que $(v_n)$ es una \omterm{def:g12:seq:arith-geom}{progresión
        geométrica} y deduce una fórmula explícita para $u_n$.
  \item Determina la cantidad de medicamento en el cuerpo a largo plazo.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:8}
Sea $(u_n)$ definida por $u_0 = 3$ y $u_{n+1} = \frac{4u_n - 1}{u_n + 2}$.
\begin{enumerate}
  \item Demuestra por inducción que $u_n > 1$ para todo $n \in \N$.
  \item Demuestra que $v_n = \dfrac{1}{u_n - 1}$ define una \omterm{def:g12:seq:arith-geom}{progresión
        aritmética}.
  \item Deduce fórmulas explícitas para $v_n$ y $u_n$, y el límite de
        $(u_n)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:seq:9}
Para $n \geq 1$, sea
$H_n = 1 + \frac{1}{2} + \frac{1}{3} + \dots + \frac{1}{n}$.
\begin{enumerate}
  \item Demuestra que para todo $n \geq 1$, $H_{2n} - H_n \geq \frac{1}{2}$.
  \item Deduce que $H_{2^k} \geq 1 + \frac{k}{2}$ para todo $k \in \N$ y
        concluye que $H_n \to +\infty$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:seq:10}
\emph{(Sucesiones adyacentes.)} Dos sucesiones $(a_n)$ y $(b_n)$ son
\emph{adyacentes} si $(a_n)$ es \omterm{def:g12:seq:monotonic}{creciente}, $(b_n)$ es \omterm{def:g10:functions:variations}{decreciente} y
$b_n - a_n \to 0$.
\begin{enumerate}
  \item Demuestra que para todo $n$, $a_n \leq b_n$. (Indicación: estudia
        la monotonía de $(b_n - a_n)$.)
  \item Demuestra que dos sucesiones adyacentes convergen las dos, y al
        mismo límite.
  \item Aplicación: demuestra que las sucesiones
        $a_n = \sum_{k=0}^{n} \frac{1}{k!}$ y
        $b_n = a_n + \frac{1}{n \cdot n!}$ ($n \geq 1$) son adyacentes. (Su
        límite común es el número $\eu$, que se estudia en el
        \cref{ch:g12:exp}.)
\end{enumerate}
\end{exercise}

\section{Problema: La sucesión de Herón, por fin juzgada}

\begin{problem}[{Problema de fin de semana --- la inducción certifica, la
convergencia monótona sentencia y la receta de $\sqrt2$ de hace dos mil
años consigue por fin su demostración (con la maravillosa media de Gauss
de postre)}]
\label{pb:g12:seq:1}
Tres veces se ha encontrado esta serie con la receta de Herón (haz la \omterm{def:g11:stat:mean}{media}
de la estimación con $2/\text{estimación}$) y tres veces solo pudo
\emph{observar} que la receta funciona. Este capítulo posee por fin los
instrumentos del juicio: la inducción (\cref{thm:g12:seq:induction}), el
teorema de la convergencia \omterm{def:g12:seq:monotonic}{monótona} (\cref{thm:g12:seq:monotone}) y los
límites de recurrencias. El veredicto, y la velocidad certificada, ocupan
el corazón de este problema; a su alrededor, las trampas clásicas de la
inducción, la divergencia más lenta de las matemáticas y la convergencia
más rápida que Gauss encontró jamás.

\medskip
\noindent\textbf{Parte I --- Calentamiento con la inducción.}
\begin{enumerate}
  \item Demuestra por inducción que
        $1 + 3 + 5 + \dots + (2n - 1) = n^2$ (la escalera de los números
        \omterm{def:g11:func:parity}{impares}, dibujada en el volumen anterior, ahora certificada).
  \item Demuestra por inducción que $2^n > n$ para todo $n \in \N$.
  \item Demuestra por inducción la desigualdad de Bernoulli: para
        $x \geq 0$ y $n \in \N$, $(1 + x)^n \geq 1 + nx$.
  \item La trampa clásica: ``todas las canicas tienen el mismo color:
        cierto para una canica; y si $n$ canicas cualesquiera son siempre
        monocromas, entonces, entre $n + 1$ canicas, las $n$ primeras
        comparten color y las $n$ últimas también, así que todas las
        $n + 1$ lo comparten''. Cualquier niño sabe que la conclusión es
        absurda: encuentra el paso exacto donde se rompe la inducción.
  \item Demuestra por inducción que $4^n - 1$ es divisible entre $3$ para
        todo $n \in \N$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El juicio a Herón.}
Sean $x_0 = 2$ y $x_{n+1} = \dfrac12\left(x_n + \dfrac{2}{x_n}\right)$.
\begin{enumerate}[resume]
  \item Calcula $x_1$, $x_2$ y $x_3$ como fracciones exactas (viejos
        conocidos).
  \item Demuestra la identidad clave
        \[
        x_{n+1}^2 - 2 = \left(\frac{x_n^2 - 2}{2x_n}\right)^{\!2}
        \geq 0,
        \]
        y deduce por inducción que $x_n > 0$ y $x_n^2 > 2$ para todo $n$.
  \item Demuestra que $(x_n)$ es estrictamente \omterm{def:g10:functions:variations}{decreciente} (calcula
        $x_{n+1} - x_n$ y usa la pregunta 7).
  \item Invoca el teorema de la convergencia \omterm{def:g12:seq:monotonic}{monótona}: ¿por qué \omterm{def:g12:seq:limit}{converge}
        $(x_n)$ a cierto límite $L \geq 1$?
  \item Identifica el límite: pasa al límite en la recurrencia
        (\cref{prop:g12:seq:operations}) y concluye que $L = \sqrt2$.
        Enuncia el veredicto histórico: tras dos mil años de servicio fiel,
        queda \emph{demostrado} que la receta de Herón \omterm{def:g12:seq:limit}{converge}.
  \item La velocidad certificada: con $e_n = x_n - \sqrt2$, demuestra que
        \[
        e_{n+1} = \frac{e_n^2}{2 x_n} ,
        \]
        y deduce $e_{n+1} \leq \frac{e_n^2}{2\sqrt2}$: el error se eleva al
        cuadrado en cada paso; la duplicación de cifras observada desde el
        volumen anterior, ahora convertida en teorema.
  \item Confírmalo numéricamente: calcula $e_0, e_1, e_2, e_3$ (a partir de
        la pregunta 6) y comprueba que cada $\frac{e_{n+1}}{e_n^2}$ está
        cerca de $\frac{1}{2x_n}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La divergencia más lenta.}
\begin{enumerate}[resume]
  \item El \cref{exo:g12:seq:9} demostró que
        $H_{2^k} \geq 1 + \frac k2$ para las sumas armónicas. ¿Cuántos
        términos garantizan $H_n > 10$? (Sirve una potencia de dos;
        asómbrate de su tamaño.)
  \item Por contraste, las sumas geométricas
        $1 + \frac12 + \frac14 + \dots + \frac{1}{2^n} = 2 - \frac{1}{2^n}$
        convergen a $2$ (\cref{thm:g12:seq:geometric}): la intuición de la
        tableta de chocolate del volumen anterior, por fin como enunciado
        sobre límites. Escribe la demostración de dos líneas.
  \item Entre las dos: demuestra que las sumas
        $S_n = 1 + \frac{1}{4} + \frac{1}{9} + \dots + \frac{1}{n^2}$
        convergen, acotando
        $\frac{1}{k^2} \leq \frac{1}{k(k-1)} = \frac{1}{k-1} - \frac{1}{k}$
        (para $k \geq 2$), telescopando y aplicando la convergencia
        \omterm{def:g12:seq:monotonic}{monótona}. (El límite, $\frac{\pi^2}{6}$, es uno de los milagros de
        Euler, que se demuestra en los volúmenes universitarios.)
  \item Enuncia en dos frases la moraleja de las preguntas 13 a 15: ¿qué
        decide, sobre la convergencia de las sumas, el hecho de que ``los
        términos tiendan a $0$''? ¿Y qué no decide?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La \omterm{def:g11:stat:mean}{media} aritmético-geométrica de
Gauss.}
Sean $a_0 = 1$, $b_0 = 2$ y
\[
a_{n+1} = \sqrt{a_n b_n},
\qquad
b_{n+1} = \frac{a_n + b_n}{2} .
\]
\begin{enumerate}[resume]
  \item Calcula $a_1, b_1, a_2, b_2$ (con cinco decimales). ¿Qué observas
        sobre la velocidad?
  \item Demuestra que $a_n \leq b_n$ para todo $n$ (la desigualdad entre
        las \omterm{def:g11:stat:mean}{medias} aritmética y geométrica, con la que nos hemos encontrado
        a lo largo de toda la serie), que $(a_n)$ crece y que $(b_n)$
        decrece.
  \item Demuestra que $b_{n+1} - a_{n+1} \leq \frac{b_n - a_n}{2}$
        (factoriza $b_{n+1} - a_{n+1} =
        \frac{(\sqrt{b_n} - \sqrt{a_n})^2}{2}$ y compara) y concluye con el
        \cref{exo:g12:seq:10} que las dos sucesiones son adyacentes:
        comparten un límite común $M(1, 2)$, la \emph{\omterm{def:g11:stat:mean}{media}
        aritmético-geométrica}.
  \item Calcula $M(1, 2)$ con seis decimales (¿cuántas iteraciones te han
        hecho falta?). El 30 de mayo de 1799, Gauss calculó
        $M(1, \sqrt2)$ con once decimales, reconoció
        $\frac{\pi}{M(1,\sqrt2)}$ como una integral conocida y escribió que
        se había abierto un ``nuevo campo del análisis''; así fue: las
        integrales elípticas, que se cuentan en los volúmenes
        universitarios. Cierra con las velocidades de convergencia
        observadas en este problema, de la más lenta a la más rápida.
\end{enumerate}
\end{problem}
