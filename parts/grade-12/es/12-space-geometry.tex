\section{Vectores en el espacio}

\omterm{def:g11:vect:vector}{vectores} en el espacio obedecen las mismas reglas que en el plano: suman, escalan,
y $\vect{AB} = \vect{CD}$ exactamente cuando $ ABDC $ es un paralelogramo. en un
\omterm{def:g10:coordgeom:system}{sistema de coordenadas} $(O; \vec\imath, \vec\jmath, \vec k)$, cada \omterm{def:g11:vect:vector}{vector} tiene
\omterm{def:g10:coordgeom:system}{coordenadas} $(x, y, z)$.

\begin{definition}[Colinealidad, coplanaridad]\label{def:g12:space:collinear}
Dos \omterm{def:g11:vect:vector}{vectores} son \emph{\omterm{def:g11:vect:collinear}{colineal}} si uno es múltiplo del otro. Tres
\omterm{def:g11:vect:vector}{vectores} $\vec u, \vec v, \vec w $ son \emph{coplanar}\index{coplanar} si uno
de ellos se puede escribir como una combinación de los otros dos, digamos
$\vec w = a\vec u + b\vec v $.
\end{definition}

\begin{proposition}\label{prop:g12:space:basis}
Si $\vec u, \vec v, \vec w $ no son \omterm{def:g12:space:collinear}{coplanares}, cada \omterm{def:g11:vect:vector}{vector} del espacio
se descompone únicamente como $ x\vec u + y\vec v + z\vec w $.
\end{proposition}

\begin{proof}[Idea de prueba]
A través de la punta del \omterm{def:g11:vect:vector}{vector} para descomponerse, trazar la línea paralela a
$\vec w $; se encuentra con el plano de $\vec u, \vec v $ en un solo punto, que
\omterm{def:g12:arith:divides}{divide} el \omterm{def:g11:vect:vector}{vector} en un componente en ese plano (exclusivamente $ x\vec u + y\vec
v $, geometría plana) y un componente a lo largo $\vec w $. Singularidad: una \omterm{def:g11:prob:variance}{diferencia}
de dos descomposiciones expresaría una de las \omterm{def:g11:vect:vector}{vectores} en términos del otro
dos, contradiciendo la no coplanaridad.
\end{proof}

\section{Rectas y planos}

\begin{definition}[Representación paramétrica de una línea.]\label{def:g12:space:line}
La línea a través $ A(x_A, y_A, z_A)$ con \omterm{def:g11:vect:direction}{vector de direccion}
$\vec u (a, b, c) \neq \vec 0$ es el conjunto de puntos
\[
M(x_A + ta,\; y_A + tb,\; z_A + tc), \qquad t \in \R .
\]
\end{definition}

\begin{definition}[Plano]\label{def:g12:space:plane}
El plano a través $ A $ dirigido por dos no \omterm{def:g11:vect:collinear}{colineales} \omterm{def:g11:vect:vector}{vectores}
$\vec u, \vec v $ es el conjunto de puntos $ M $ con
$\vect{AM} = s\vec u + t\vec v $, $(s, t) \in \R^2$.
\end{definition}

\begin{proposition}[Posiciones relativas]\label{prop:g12:space:positions}
Dos planos distintos son paralelos o se cruzan en una línea. Una línea y un
plano son paralelos (posiblemente contenidos) o se cruzan en una sola
punto. Dos líneas en el espacio pueden ser intersectantes, estrictamente paralelas, idénticas,
o \emph{sesgar} (no \omterm{def:g12:space:collinear}{coplanar}).
\end{proposition}

\begin{proof}[Boceto]
Éstas son declaraciones de incidencia; El análisis de cada caso reduce, en \omterm{def:g10:coordgeom:system}{coordenadas},
para resolver un sistema lineal y contando sus soluciones --- ver
\cref{met:g12:space:intersections}.
\end{proof}

\section{Producto escalar}

\begin{definition}[Producto escalar]\label{def:g12:space:dot}
El \emph{producto escalar}\index{producto escalar} de $\vec u $ y $\vec v $ es
\[
\vec u \cdot \vec v = \tfrac12\left(\norm{\vec u + \vec v}^2
- \norm{\vec u}^2 - \norm{\vec v}^2\right),
\]
dónde $\norm{\vec u}$ es la longitud de $\vec u $. si ambos \omterm{def:g11:vect:vector}{vectores} son
distinto de cero, $\vec u \cdot \vec v = \norm{\vec u}\,\norm{\vec v}\cos\theta $
dónde $\theta $ es el ángulo entre ellos; y en un \emph{\omterm{def:g10:coordgeom:system}{ortonormal}}
\omterm{def:g10:coordgeom:system}{sistema de coordenadas},
\[
\vec u \cdot \vec v = xx' + yy' + zz' .
\]
Dos \omterm{def:g11:vect:vector}{vectores} son \emph{\omterm{def:g11:scal:orthogonal}{ortogonal}} si su producto escalar es $0$.
\end{definition}

\begin{proposition}[Propiedades]\label{prop:g12:space:dotprops}
El \omterm{def:g12:space:dot}{producto escalar} es simétrico ($\vec u\cdot\vec v = \vec v\cdot\vec u $),
bilineal ($(a\vec u + b\vec v)\cdot \vec w = a\,\vec u\cdot\vec w +
b\,\vec v\cdot \vec w $), y
$\vec u \cdot \vec u = \norm{\vec u}^2 \geq 0$.
\end{proposition}

\begin{proof}
en un sistema \omterm{def:g10:coordgeom:system}{ortonormal}, las tres propiedades son inmediatas en la fórmula
$ xx' + yy' + zz'$, que a su vez se deriva de la fórmula definitoria y de la
Cálculo pitagórico de longitudes:
$\norm{\vec u}^2 = x^2 + y^2 + z^2$.
\end{proof}

\begin{definition}[vectores normales]\label{def:g12:space:normal}
A \emph{vectores normales}\index{vectores normales} de un plano $\mathcal P $ es un
distinto de cero \omterm{def:g11:vect:vector}{vector} \omterm{def:g11:scal:orthogonal}{ortogonal} a cada \omterm{def:g11:vect:vector}{vector} acostado en $\mathcal P $ (eso
basta que sea \omterm{def:g11:scal:orthogonal}{ortogonal} a dos no \omterm{def:g11:vect:collinear}{colineales} direcciones de
$\mathcal P $).
\end{definition}

\begin{theorem}[Ecuación cartesiana de un plano.]\label{thm:g12:space:planeeq}
en un \omterm{def:g10:coordgeom:system}{ortonormal} \omterm{def:g10:coordgeom:system}{sistema de coordenadas}, el plano a través
$ A $ con \omterm{def:g11:vect:vector}{vector} normal $\vec n(a, b, c)$ tiene \omterm{def:g10:algebra:equation}{ecuación}
\[
a x + b y + c z + d = 0,
\]
dónde $ d = -(ax_A + by_A + cz_A)$. Por el contrario, cada uno de esos \omterm{def:g10:algebra:equation}{ecuación} con
$(a,b,c) \neq (0,0,0)$ define un plano con \omterm{def:g11:vect:vector}{vector} normal $(a, b, c)$.
\end{theorem}

\begin{proof}
$ M \in \mathcal P \iff \vect{AM} \perp \vec n \iff
a(x - x_A) + b(y - y_A) + c(z - z_A) = 0$, que se expande hasta el \omterm{def:g10:algebra:equation}{ecuación}.
Por el contrario, elija cualquier punto de solución $ A $ del \omterm{def:g10:algebra:equation}{ecuación}; lo mismo
El cálculo al revés \omterm{def:g10:proba:sample}{muestra} que el conjunto solución es
$\{M : \vect{AM}\cdot\vec n = 0\}$, el plano a través $ A $ normal a
$\vec n $.
\end{proof}

\begin{method}[Intersecciones en la práctica]\label{met:g12:space:intersections}
\begin{itemize}
  \item \emph{Línea $\cap $ plano}: sustituir el paramétrico \omterm{def:g10:coordgeom:system}{coordenadas} de
        la línea en el plano \omterm{def:g10:algebra:equation}{ecuación}; resolver para $ t $ (una solución: una
        punto; $0 = $ distinto de cero: paralelo; $0 = 0$: línea contenida).
  \item \emph{Plano $\cap $ plano}: resuelve el sistema de los dos ecuaciones,
        parametrizar por una coordenada libre; la solución es una línea (o la
        Los planos son paralelos cuando \omterm{def:g12:space:normal}{vectores normales} son \omterm{def:g11:vect:collinear}{colineales}).
  \item \emph{Comprobaciones de ortogonalidad}: línea $\perp $ plano si su dirección
        es \omterm{def:g11:vect:collinear}{colineales} con lo normal; dos planos son perpendiculares si y sólo si
        lo normal es \omterm{def:g11:scal:orthogonal}{ortogonal}.
\end{itemize}
\end{method}

\begin{proposition}[Distancia de un punto a un plano]\label{prop:g12:space:distance}
\index{distancia!punto al plano}
en un sistema \omterm{def:g10:coordgeom:system}{ortonormal}, la distancia desde $ M_0(x_0, y_0, z_0)$ al
plano $\mathcal P : ax + by + cz + d = 0$ es
\[
\operatorname{dist}(M_0, \mathcal P)
= \frac{\abs{ax_0 + by_0 + cz_0 + d}}{\sqrt{a^2 + b^2 + c^2}}.
\]
\end{proposition}

\begin{proof}
Sea $ H $ ser el \omterm{def:g11:scal:orthogonal}{ortogonal} proyección de $ M_0$ en $\mathcal P $: el pie de
la perpendicular, entonces $\vect{HM_0}$ es \omterm{def:g11:vect:collinear}{colineales} con la unidad normal
$\frac{\vec n}{\norm{\vec n}}$, y la distancia es
$\abs{\vect{HM_0}\cdot \frac{\vec n}{\norm{\vec n}}}$. Para cualquier
$ H \in \mathcal P $,
$\vect{HM_0}\cdot\vec n = ax_0 + by_0 + cz_0 - (ax_H + by_H + cz_H)
= ax_0 + by_0 + cz_0 + d $ (usando el \omterm{def:g10:algebra:equation}{ecuación} en $ H $), de donde la fórmula
después de dividir por $\norm{\vec n} = \sqrt{a^2+b^2+c^2}$.
\end{proof}

\begin{example}\label{ex:g12:space:distance}
Distancia del origen al plano. $ x + 2y + 2z - 6 = 0$:
$\frac{\abs{-6}}{\sqrt{1 + 4 + 4}} = \frac63 = 2$.
\end{example}

\section{Ejercicios}

\end{exercise}

\begin{exercise}[$\star $]\label{exo:g12:space:3}

\begin{exercise}[$\star\star $]\label{exo:g12:space:4}
Considere la línea $\mathcal D $ a través de $ A(1, 2, 0)$ con dirección

Determinar $\mathcal D \cap \mathcal P $.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:space:5}
Sea $\mathcal P : x - y + 2z = 1$ y $\mathcal Q : 2x + y + z = 4$.
mostrar que $\mathcal P $ y $\mathcal Q $ se cruzan en una recta y dan un

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:space:6}
Demuestre que las líneas

son sesgados (no \omterm{def:g12:space:collinear}{coplanares}).
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:space:7}

\begin{enumerate}
  \item Calcula la distancia desde $ M_0(3, 1, 1)$ a $\mathcal P $.
  \item determinar el \omterm{def:g10:coordgeom:system}{coordenadas} del \omterm{def:g11:scal:orthogonal}{ortogonal} proyección $ H $ de

\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g12:space:8}
el cubo $ ABCDEFGH $ tiene lado $1$; lugar \omterm{def:g10:coordgeom:system}{coordenadas} de modo que
$ A(0,0,0)$, $ B(1,0,0)$, $ D(0,1,0)$, $ E(0,0,1)$ (con

  \item Demuestre que la diagonal $(AG)$ es \omterm{def:g11:scal:orthogonal}{ortogonal} al plano $(BDE)$.
  \item Calcula la distancia desde $ A $ al plano $(BDE)$, y el punto
        donde $(AG)$ lo perfora.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g12:space:9}
\emph{(Volumen de un tetraedro).} Sea $ A(1,1,1)$, $ B(2,1,0)$, $ C(0,2,1)$

\begin{enumerate}
  \item verificar que $\vect{AB}$, $\vect{AC}$, $\vect{AD}$ no son \omterm{def:g12:space:collinear}{coplanares}
        (los puntos forman un tetraedro genuino).
  \item Encuentra un \omterm{def:g11:vect:vector}{vector} $\vec n $ \omterm{def:g11:scal:orthogonal}{ortogonal} a ambos $\vect{BC}$ y

  \item Calcula la distancia desde $ A $ al plano $(BCD)$ y el área de
        el triangulo $ BCD $, usando la fórmula
        $\text{Area} = \frac12\sqrt{\norm{\vec u}^2\norm{\vec v}^2 -
        (\vec u \cdot \vec v)^2}$ para el triángulo formado por
        $\vec u, \vec v $.
  \item Deducir el volumen del tetraedro.
        ($ V = \frac13 \times \text{base area} \times \text{height}$).
\end{enumerate}
\end{exercise}

  \item Calcular el \omterm{def:g11:scal:orthogonal}{ortogonal} proyección de $ B $ sobre $ P $
        (caminar desde $ B $ por la normal hasta el plano), y
        comprueba que tu punto satisface $ P$'s \omterm{def:g10:algebra:equation}{ecuación}.
  \item Final --- el kit del topógrafo espacial: líneas paramétricas
        para perforar, \omterm{def:g12:space:normal}{vectores normales} para ángulos y distancias,
        plano ecuaciones para secciones; y el cubo como
        patio de juegos donde produjeron un hexágono, un
        tetraedro, el ángulo del diamante y un par de sesgos
        líneas. Una frase cada uno.
\end{enumerate}
\end{problem}
