\chapter{Vectores, rectas y planos en el espacio}\label{ch:g12:space}

La geometría del espacio se vuelve calculable en cuanto disponemos de
\omterm{def:g11:vect:vector}{vectores} y \omterm{def:g10:coordgeom:system}{coordenadas}: las rectas reciben representaciones paramétricas,
los planos reciben ecuaciones cartesianas y el \omterm{def:g12:space:dot}{producto escalar} mide
ángulos y distancias. Este capítulo construye esa caja de herramientas y la
usa para resolver los problemas clásicos de \omterm{def:g10:numbers:interunion}{intersección} y de distancia.

\section{Vectores en el espacio}

Los \omterm{def:g11:vect:vector}{vectores} del espacio obedecen a las mismas reglas que los del plano: se
suman, se multiplican por un número y $\vect{AB} = \vect{CD}$ exactamente
cuando $ABDC$ es un paralelogramo. En un \omterm{def:g10:coordgeom:system}{sistema de coordenadas}
$(O; \vec\imath, \vec\jmath, \vec k)$, todo \omterm{def:g11:vect:vector}{vector} tiene \omterm{def:g10:coordgeom:system}{coordenadas}
$(x, y, z)$.

\begin{definition}[Vectores colineales y coplanarios]\label{def:g12:space:collinear}
Dos \omterm{def:g11:vect:vector}{vectores} son \emph{colineales} si uno es múltiplo del otro. Tres
\omterm{def:g11:vect:vector}{vectores} $\vec u, \vec v, \vec w$ son \emph{coplanarios}\index{coplanario}
si uno de ellos se puede escribir como combinación de los otros dos,
digamos $\vec w = a\vec u + b\vec v$.
\end{definition}

\begin{proposition}\label{prop:g12:space:basis}
Si $\vec u, \vec v, \vec w$ no son \omterm{def:g12:space:collinear}{coplanarios}, todo \omterm{def:g11:vect:vector}{vector} del espacio se
descompone de manera única como $x\vec u + y\vec v + z\vec w$.
\end{proposition}

\begin{proof}[Idea de la demostración]
Por el \omterm{def:g12:deriv:extremum}{extremo} del \omterm{def:g11:vect:vector}{vector} que queremos descomponer, trazamos la recta
paralela a $\vec w$; corta al plano de $\vec u$ y $\vec v$ en un único
punto, que separa el \omterm{def:g11:vect:vector}{vector} en una componente contenida en ese plano
(escrita de manera única como $x\vec u + y\vec v$, por geometría plana) y
una componente en la dirección de $\vec w$. Unicidad: la \omterm{def:g11:seq:arithmetic}{diferencia} de dos
descomposiciones expresaría uno de los \omterm{def:g11:vect:vector}{vectores} en \omterm{def:g11:func:function}{función} de los otros
dos, en contra de que no sean \omterm{def:g12:space:collinear}{coplanarios}.
\end{proof}

\section{Rectas y planos}

\begin{definition}[Representación paramétrica de una recta]\label{def:g12:space:line}
La recta que pasa por $A(x_A, y_A, z_A)$ con \omterm{def:g11:vect:direction}{vector director}
$\vec u (a, b, c) \neq \vec 0$ es el conjunto de los puntos
\[
M(x_A + ta,\; y_A + tb,\; z_A + tc), \qquad t \in \R .
\]
\end{definition}

\begin{definition}[Plano]\label{def:g12:space:plane}
El plano que pasa por $A$ y está dirigido por dos \omterm{def:g11:vect:vector}{vectores} no \omterm{def:g12:space:collinear}{colineales}
$\vec u, \vec v$ es el conjunto de los puntos $M$ con
$\vect{AM} = s\vec u + t\vec v$, con $(s, t) \in \R^2$.
\end{definition}

\begin{proposition}[Posiciones relativas]\label{prop:g12:space:positions}
Dos planos distintos o son paralelos o se cortan en una recta. Una recta y
un plano o son paralelos (posiblemente con la recta contenida en el plano)
o se cortan en un único punto. Dos rectas del espacio pueden cortarse, ser
estrictamente paralelas, ser la misma o \emph{cruzarse} (sin ser
coplanarias).
\end{proposition}

\begin{proof}[Esquema]
Son enunciados de incidencia; cada análisis de casos se reduce, en
\omterm{def:g10:coordgeom:system}{coordenadas}, a resolver un \omterm{def:g10:lines:system}{sistema lineal} y a contar sus soluciones; véase
el \cref{met:g12:space:intersections}.
\end{proof}

\section{Producto escalar}

\begin{definition}[Producto escalar]\label{def:g12:space:dot}
El \emph{producto escalar}\index{producto escalar} de $\vec u$ y $\vec v$
es
\[
\vec u \cdot \vec v = \tfrac12\left(\norm{\vec u + \vec v}^2
- \norm{\vec u}^2 - \norm{\vec v}^2\right),
\]
donde $\norm{\vec u}$ es la longitud de $\vec u$. Si los dos \omterm{def:g11:vect:vector}{vectores} son
no nulos, $\vec u \cdot \vec v = \norm{\vec u}\,\norm{\vec v}\cos\theta$,
donde $\theta$ es el ángulo que forman; y en un \omterm{def:g10:coordgeom:system}{sistema de coordenadas}
\emph{\omterm{def:g10:coordgeom:system}{ortonormal}},
\[
\vec u \cdot \vec v = xx' + yy' + zz' .
\]
Dos \omterm{def:g11:vect:vector}{vectores} son \emph{\omterm{def:g11:scal:orthogonal}{ortogonales}} si su producto escalar es $0$.
\end{definition}

\begin{proposition}[Propiedades]\label{prop:g12:space:dotprops}
El \omterm{def:g12:space:dot}{producto escalar} es simétrico ($\vec u\cdot\vec v = \vec v\cdot\vec u$),
bilineal ($(a\vec u + b\vec v)\cdot \vec w = a\,\vec u\cdot\vec w +
b\,\vec v\cdot \vec w$) y cumple
$\vec u \cdot \vec u = \norm{\vec u}^2 \geq 0$.
\end{proposition}

\begin{proof}
En un \omterm{def:g10:coordgeom:system}{sistema ortonormal}, las tres propiedades son inmediatas sobre la
fórmula $xx' + yy' + zz'$, que a su vez se sigue de la fórmula que define
el \omterm{def:g12:space:dot}{producto escalar} y del cálculo pitagórico de las longitudes:
$\norm{\vec u}^2 = x^2 + y^2 + z^2$.
\end{proof}

\begin{definition}[Vector normal]\label{def:g12:space:normal}
Un \emph{vector normal}\index{vector normal} de un plano $\mathcal P$ es un
\omterm{def:g11:vect:vector}{vector} no nulo ortogonal a todos los \omterm{def:g11:vect:vector}{vectores} contenidos en $\mathcal P$
(basta con que sea ortogonal a dos direcciones no \omterm{def:g12:space:collinear}{colineales} de
$\mathcal P$).
\end{definition}

\begin{theorem}[Ecuación cartesiana de un plano]\label{thm:g12:space:planeeq}
En un \omterm{def:g10:coordgeom:system}{sistema de coordenadas} \omterm{def:g10:coordgeom:system}{ortonormal}, el plano que pasa por $A$ con
\omterm{def:g12:space:normal}{vector normal} $\vec n(a, b, c)$ tiene por \omterm{def:g10:algebra:equation}{ecuación}
\[
a x + b y + c z + d = 0,
\]
donde $d = -(ax_A + by_A + cz_A)$. Recíprocamente, toda \omterm{def:g10:algebra:equation}{ecuación} de esa
forma con $(a,b,c) \neq (0,0,0)$ define un plano de \omterm{def:g12:space:normal}{vector normal}
$(a, b, c)$.
\end{theorem}

\begin{proof}
$M \in \mathcal P \iff \vect{AM} \perp \vec n \iff
a(x - x_A) + b(y - y_A) + c(z - z_A) = 0$, que desarrollado da la \omterm{def:g10:algebra:equation}{ecuación}.
Recíprocamente, tomamos un punto solución cualquiera $A$ de la \omterm{def:g10:algebra:equation}{ecuación}; el
mismo cálculo al revés muestra que el conjunto de soluciones es
$\{M : \vect{AM}\cdot\vec n = 0\}$, el plano que pasa por $A$ y es normal a
$\vec n$.
\end{proof}

\begin{method}[Intersecciones en la práctica]\label{met:g12:space:intersections}
\begin{itemize}
  \item \emph{Recta $\cap$ plano}: sustituye las \omterm{def:g10:coordgeom:system}{coordenadas} paramétricas
        de la recta en la \omterm{def:g10:algebra:equation}{ecuación} del plano y despeja $t$ (una solución:
        un punto; $0 = $ algo no nulo: son paralelos; $0 = 0$: la recta
        está contenida en el plano).
  \item \emph{Plano $\cap$ plano}: resuelve el sistema de las dos
        ecuaciones parametrizando por una coordenada libre; la solución es
        una recta (o bien los planos son paralelos, cuando los \omterm{def:g12:space:normal}{vectores
        normales} son \omterm{def:g12:space:collinear}{colineales}).
  \item \emph{Comprobaciones de ortogonalidad}: una recta es perpendicular
        a un plano si y solo si su dirección es colineal con el \omterm{def:g12:space:normal}{vector
        normal}; y dos planos son perpendiculares si y solo si sus \omterm{def:g12:space:normal}{vectores
        normales} son \omterm{def:g11:scal:orthogonal}{ortogonales}.
\end{itemize}
\end{method}

\begin{proposition}[Distancia de un punto a un plano]\label{prop:g12:space:distance}
\index{distancia!de un punto a un plano}
En un \omterm{def:g10:coordgeom:system}{sistema ortonormal}, la distancia de $M_0(x_0, y_0, z_0)$ al plano
$\mathcal P : ax + by + cz + d = 0$ es
\[
\operatorname{dist}(M_0, \mathcal P)
= \frac{\abs{ax_0 + by_0 + cz_0 + d}}{\sqrt{a^2 + b^2 + c^2}}.
\]
\end{proposition}

\begin{proof}
Sea $H$ la proyección ortogonal de $M_0$ sobre $\mathcal P$: el pie de la
perpendicular, de modo que $\vect{HM_0}$ es colineal con el \omterm{def:g12:space:normal}{vector normal}
unitario $\frac{\vec n}{\norm{\vec n}}$ y la distancia vale
$\abs{\vect{HM_0}\cdot \frac{\vec n}{\norm{\vec n}}}$. Para cualquier
$H \in \mathcal P$,
$\vect{HM_0}\cdot\vec n = ax_0 + by_0 + cz_0 - (ax_H + by_H + cz_H)
= ax_0 + by_0 + cz_0 + d$ (usando la \omterm{def:g10:algebra:equation}{ecuación} en $H$), de donde sale la
fórmula tras dividir entre $\norm{\vec n} = \sqrt{a^2+b^2+c^2}$.
\end{proof}

\begin{example}\label{ex:g12:space:distance}
Distancia del origen al plano $x + 2y + 2z - 6 = 0$:
$\frac{\abs{-6}}{\sqrt{1 + 4 + 4}} = \frac63 = 2$.
\end{example}

\section{Ejercicios}

En todos los ejercicios, el \omterm{def:g10:coordgeom:system}{sistema de coordenadas} es \omterm{def:g10:coordgeom:system}{ortonormal}.

\begin{exercise}[$\star$]\label{exo:g12:space:1}
Sean $A(1, 0, 2)$, $B(3, 1, 1)$ y $C(2, -1, 3)$. ¿Están alineados los
puntos $A$, $B$ y $C$? Da una representación paramétrica de la recta
$(AB)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:space:2}
Determina una \omterm{def:g10:algebra:equation}{ecuación} cartesiana del plano que pasa por $A(1, 1, 0)$ con
\omterm{def:g12:space:normal}{vector normal} $\vec n(2, -1, 3)$. ¿Pertenece a él el punto $B(0, 2, 1)$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:space:3}
Calcula el ángulo en $A$ (redondeado al grado) del triángulo de vértices
$A(0,0,0)$, $B(1, 1, 0)$ y $C(1, 0, 1)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:space:4}
Considera la recta $\mathcal D$ que pasa por $A(1, 2, 0)$ con dirección
$\vec u(1, -1, 2)$, y el plano $\mathcal P : 2x + y - z + 1 = 0$.
Determina $\mathcal D \cap \mathcal P$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:space:5}
Sean $\mathcal P : x - y + 2z = 1$ y $\mathcal Q : 2x + y + z = 4$.
Demuestra que $\mathcal P$ y $\mathcal Q$ se cortan en una recta y da una
representación paramétrica de esa recta.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:space:6}
Demuestra que las rectas
\[
\mathcal D_1 : (x, y, z) = (1 + t,\; 2t,\; 3 - t), \qquad
\mathcal D_2 : (x, y, z) = (2 + s,\; 1 + s,\; 1 + 2s)
\]
se cruzan (no son coplanarias).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:space:7}
Sea $\mathcal P : 2x - y + 2z - 3 = 0$.
\begin{enumerate}
  \item Calcula la distancia de $M_0(3, 1, 1)$ a $\mathcal P$.
  \item Determina las \omterm{def:g10:coordgeom:system}{coordenadas} de la proyección ortogonal $H$ de $M_0$
        sobre $\mathcal P$ y comprueba la distancia $M_0H$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:space:8}
El cubo $ABCDEFGH$ tiene lado $1$; elige \omterm{def:g10:coordgeom:system}{coordenadas} de modo que
$A(0,0,0)$, $B(1,0,0)$, $D(0,1,0)$, $E(0,0,1)$ (con $C = B + D$,
$F = B + E$, $G = B + D + E$ y $H = D + E$ como \omterm{def:g11:vect:vector}{vectores} desde $A$).
\begin{enumerate}
  \item Demuestra que la diagonal $(AG)$ es ortogonal al plano $(BDE)$.
  \item Calcula la distancia de $A$ al plano $(BDE)$ y el punto por el que
        $(AG)$ lo atraviesa.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:space:9}
\emph{(Volumen de un tetraedro.)} Sean $A(1,1,1)$, $B(2,1,0)$, $C(0,2,1)$ y
$D(2,2,2)$.
\begin{enumerate}
  \item Comprueba que $\vect{AB}$, $\vect{AC}$ y $\vect{AD}$ no son
        \omterm{def:g12:space:collinear}{coplanarios} (los puntos forman un tetraedro de verdad).
  \item Halla un \omterm{def:g11:vect:vector}{vector} $\vec n$ ortogonal a la vez a $\vect{BC}$ y a
        $\vect{BD}$, y deduce una \omterm{def:g10:algebra:equation}{ecuación} cartesiana del plano $(BCD)$.
  \item Calcula la distancia de $A$ al plano $(BCD)$ y el área del
        triángulo $BCD$, usando la fórmula
        $\text{Área} = \frac12\sqrt{\norm{\vec u}^2\norm{\vec v}^2 -
        (\vec u \cdot \vec v)^2}$ para el triángulo generado por
        $\vec u$ y $\vec v$.
  \item Deduce el volumen del tetraedro
        ($V = \frac13 \times \text{área de la base} \times
        \text{altura}$).
\end{enumerate}
\end{exercise}

\section{Problema: Cortar el cubo en rodajas}

\begin{problem}[{Problema de fin de semana --- un plano corta un cubo
formando un hexágono perfecto, un tetraedro regular se esconde en las
esquinas y el ángulo de la diagonal da forma a todos los diamantes}]
\label{pb:g12:space:1}
Al cortar un cubo uno espera cuadrados y rectángulos; y, sin embargo, un
corte famoso produce un \emph{hexágono regular}, y cuatro cortes hábiles en
las esquinas dejan atrás un \emph{tetraedro regular}. Las dos sorpresas, y
el ángulo de $109.5^\circ$ que veneran los átomos de carbono, caen ante la
caja de herramientas de este capítulo: rectas paramétricas, ecuaciones de
planos y \omterm{def:g12:space:dot}{producto escalar} (\cref{thm:g12:space:planeeq},
\cref{prop:g12:space:distance}). Usa en todo momento el cubo del
\cref{exo:g12:space:8}: lado $1$, $A(0,0,0)$, $B(1,0,0)$, $D(0,1,0)$,
$E(0,0,1)$, $C(1,1,0)$, $F(1,0,1)$, $H(0,1,1)$, $G(1,1,1)$.

\medskip
\noindent\textbf{Parte I --- Soltura.}
\begin{enumerate}
  \item Escribe la \omterm{def:g10:algebra:equation}{ecuación} del plano que pasa por $B(1,0,0)$ con \omterm{def:g12:space:normal}{vector
        normal} $(1,1,1)$, una representación paramétrica de la recta que
        pasa por $A$ dirigida por $(1,1,1)$ y su punto de \omterm{def:g10:numbers:interunion}{intersección}.
  \item Calcula la distancia de $A$ a ese plano
        (\cref{prop:g12:space:distance}).
  \item ¿Son \omterm{def:g11:scal:orthogonal}{ortogonales} los \omterm{def:g11:vect:vector}{vectores} $(1,-1,0)$ y $(1,1,-2)$?
  \item ¿Posición relativa de los planos $x + y + z = 1$ y
        $2x + 2y + 2z = 5$? ¿Y de la recta que pasa por $A$ dirigida por
        $(1,1,0)$ respecto del primer plano?
  \item Enumera los \omterm{prop:g10:coordgeom:midpoint}{puntos medios} de las seis aristas
        $[BC], [CD], [DH], [HE], [EF], [FB]$ del cubo y comprueba que los
        seis cumplen la \omterm{def:g10:algebra:equation}{ecuación} $x + y + z = \frac32$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La sección hexagonal.}
Sea $P$ el plano $x + y + z = \frac32$.
\begin{enumerate}[resume]
  \item Demuestra que $P$ es perpendicular a la diagonal grande $(AG)$ y
        pasa por el centro del cubo.
  \item Calcula las distancias entre \omterm{prop:g10:coordgeom:midpoint}{puntos medios} consecutivos de la
        pregunta 5 al recorrer la sección: ¿qué encuentras?
  \item Calcula el ángulo del polígono de la sección en uno de sus vértices
        (dos \omterm{def:g11:vect:vector}{vectores} de aristas consecutivas y un \omterm{def:g12:space:dot}{producto escalar}).
        Concluye: la sección es un \emph{hexágono regular}.
  \item Calcula el perímetro y el área del hexágono, y compara el área con
        la de una cara del cubo. (Para que conste: la mayor sección plana
        del cubo unidad es el rectángulo diagonal $1 \times \sqrt2$, de
        área $\sqrt2$; el hexágono regular se lleva la plata.)
  \item ¿Por dónde atraviesa la diagonal $(AG)$ al plano $P$? Comprueba que
        el punto de paso es el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[AG]$.
  \item Desliza el corte: describe las secciones $x + y + z = c$ a medida
        que $c$ crece de $0$ a $3$; calcula la sección para
        $c = \frac12$ (¿qué polígono?, ¿de qué tamaño?) y narra la
        metamorfosis que pasa por tu hexágono en $c = \frac32$.
  \item ¿Por qué ningún plano puede cortar un cubo en un polígono de
        \emph{siete} lados? (¿Dónde tiene que estar cada lado de la
        sección?)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El tetraedro escondido.}
\begin{enumerate}[resume]
  \item Calcula el volumen del tetraedro de esquina $ABDE$ (base, altura y
        la fórmula del tercio que el volumen anterior admitió sin
        demostrar; el \cref{exo:g12:space:9} calcula volúmenes así en
        general).
  \item Los cuatro vértices $B$, $D$, $E$, $G$: calcula las seis distancias
        dos a dos y concluye que $BDEG$ es un tetraedro \emph{regular}.
        Obtén después su volumen restando al cubo los tetraedros de las
        esquinas.
  \item Calcula la distancia del centro del cubo al plano $(BDE)$.
  \item El ángulo del diamante: calcula el ángulo que forman las diagonales
        $\vect{AG}$ y $\vect{BH}$ del cubo. Su suplementario, de unos
        $109.5^\circ$, es el ángulo entre enlaces en el diamante y en el
        metano; ¿por qué eligen cuatro pares de electrones alrededor de un
        átomo de carbono los vértices del tetraedro de la pregunta 14?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Genuinamente tridimensional.}
\begin{enumerate}[resume]
  \item Halla por dónde atraviesa la recta que pasa por $B(1,0,0)$ dirigida
        por $(0,1,1)$ el plano $P$ del hexágono.
  \item Demuestra que las rectas $(AB)$ y $(EG)$ ni se cortan ni son
        paralelas: \emph{se cruzan}, el fenómeno que no puede ocurrir en un
        plano.
  \item Calcula la proyección ortogonal de $B$ sobre $P$ (camina desde $B$
        siguiendo la normal hasta llegar al plano) y comprueba que tu punto
        cumple la \omterm{def:g10:algebra:equation}{ecuación} de $P$.
  \item Final: el equipo del topógrafo del espacio: rectas paramétricas
        para atravesar, \omterm{def:g12:space:normal}{vectores normales} para ángulos y distancias y
        ecuaciones de planos para las secciones; y el cubo como terreno de
        juego donde produjeron un hexágono, un tetraedro regular, el ángulo
        del diamante y un par de rectas que se cruzan. Una frase para cada
        cosa.
\end{enumerate}
\end{problem}
