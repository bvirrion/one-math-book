\chapter{Aritmética}\label{ch:g12:arith}

La aritmética estudia los \omterm{def:g10:numbers:sets}{números enteros}: la \omterm{def:g12:arith:divides}{divisibilidad}, los \omterm{def:g12:arith:prime}{números
primos}, los restos. Durante mucho tiempo se la consideró la más pura de las
matemáticas puras y hoy protege todos los pagos por internet: el
criptosistema RSA se apoya en los teoremas de Bézout, Gauss y Fermat que se
demuestran en este capítulo.

\section{Divisibilidad y división euclídea}

\begin{definition}[Divisibilidad]\label{def:g12:arith:divides}
Sean $a, b \in \Z$. Decimos que $b$ \emph{divide} a
$a$\index{divisibilidad}, y escribimos $b \mid a$, si existe $k \in \Z$ con
$a = kb$. También decimos que $a$ es \emph{múltiplo} de $b$.
\end{definition}

\begin{proposition}\label{prop:g12:arith:divprops}
Si $c \mid a$ y $c \mid b$, entonces $c$ \omterm{def:g12:arith:divides}{divide} a toda combinación entera
$au + bv$ ($u, v \in \Z$). Si $a \mid b$ y $b \mid a$ con $a,b \in \N$,
entonces $a = b$. Si $a \mid b$ y $b \neq 0$, entonces
$\abs a \leq \abs b$.
\end{proposition}

\begin{proof}
Escribimos $a = kc$ y $b = lc$: entonces $au + bv = (ku + lv)c$. Los demás
puntos se siguen de $\abs{a} = \abs{k}\,\abs{b}$ con $\abs k \geq 1$ cuando
$b = ka \neq 0$.
\end{proof}

\begin{theorem}[División euclídea]\label{thm:g12:arith:euclid}
\index{división euclídea}
Sean $a \in \Z$ y $b \in \N^*$. Existe una única pareja
$(q, r) \in \Z \times \N$ tal que
\[
a = bq + r \qquad\text{y}\qquad 0 \leq r < b .
\]
$q$ es el \emph{cociente} y $r$, el \emph{resto}.
\end{theorem}

\begin{proof}
\emph{Existencia.} El conjunto de los múltiplos de $b$ que no superan a $a$
tiene un elemento \omterm{def:g10:functions:extrema}{máximo} $bq$ (no es vacío y está acotado superiormente);
tomamos $r = a - bq$. Por maximalidad, $b(q+1) > a$, luego $0 \leq r < b$.
\emph{Unicidad.} Si $bq + r = bq' + r'$ con $0 \leq r, r' < b$, entonces
$b(q - q') = r' - r$ y $\abs{r' - r} < b$: un múltiplo de $b$ de \omterm{def:g10:numbers:abs}{valor
absoluto} menor que $b$ tiene que ser $0$, luego $r = r'$ y $q = q'$.
\end{proof}

\section{Congruencias}

\begin{definition}[Congruencia]\label{def:g12:arith:congruence}
Sea $n \in \N^*$. Dos enteros $a, b$ son \emph{congruentes módulo
$n$}\index{congruencia}, y se escribe $a \equiv b \pmod n$, si
$n \mid (a - b)$; equivalentemente, si $a$ y $b$ dan el mismo resto en la
\omterm{thm:g12:arith:euclid}{división euclídea} entre $n$.
\end{definition}

\begin{proposition}[Compatibilidad con las operaciones]\label{prop:g12:arith:congops}
Si $a \equiv b \pmod n$ y $c \equiv d \pmod n$, entonces
\[
a + c \equiv b + d, \qquad
ac \equiv bd, \qquad
a^k \equiv b^k \ (k \in \N) \pmod n .
\]
\end{proposition}

\begin{proof}
$n$ \omterm{def:g12:arith:divides}{divide} a $(a-b) + (c-d) = (a+c) - (b+d)$, y
$ac - bd = a(c - d) + d(a - b)$ también es múltiplo de $n$. La regla de las
potencias se sigue por inducción a partir de la del producto.
\end{proof}

\begin{method}[Calcular potencias módulo $n$]\label{met:g12:arith:powers}
Para calcular $a^k \bmod n$, reduce la base \omterm{def:g12:complex:modulus}{módulo} $n$, busca después una
potencia pequeña de $a$ congruente con $\pm1$ y úsala para colapsar el
exponente. Por ejemplo, $2^{100} \bmod 7$: como $2^3 = 8 \equiv 1 \pmod 7$
y $100 = 3\times33 + 1$,
\[
2^{100} = \left(2^{3}\right)^{33} \times 2 \equiv 1^{33}\times 2 = 2 \pmod 7 .
\]
\end{method}

\section{Máximo común divisor, Bézout y Gauss}

\begin{definition}[Máximo común divisor]\label{def:g12:arith:gcd}
Sean $a, b$ enteros no nulos a la vez. El \emph{máximo común
divisor}\index{máximo común divisor} $\gcd(a, b)$ es el mayor entero que
\omterm{def:g12:arith:divides}{divide} a la vez a $a$ y a $b$. Cuando $\gcd(a,b) = 1$, se dice que $a$ y
$b$ son \emph{primos entre sí}\index{primos entre sí}.
\end{definition}

\begin{proposition}[Algoritmo de Euclides]\label{prop:g12:arith:euclidalgo}
\index{algoritmo de Euclides}
Si $a = bq + r$ (con $b \neq 0$), entonces $\gcd(a, b) = \gcd(b, r)$. Iterar
la \omterm{thm:g12:arith:euclid}{división euclídea} calcula, por tanto, $\gcd(a,b)$: el \omterm{def:g12:arith:gcd}{máximo común
divisor} es el último resto no nulo.
\end{proposition}

\begin{proof}
Todo divisor común de $a$ y $b$ \omterm{def:g12:arith:divides}{divide} a $r = a - bq$
(\cref{prop:g12:arith:divprops}) y es, por tanto, divisor común de $b$ y
$r$; y recíprocamente, puesto que $a = bq + r$. Las dos parejas tienen los
mismos divisores comunes y, por tanto, el mismo \omterm{def:g12:arith:gcd}{máximo común divisor}. El
algoritmo termina porque los restos forman una \omterm{def:g12:seq:sequence}{sucesión} estrictamente
\omterm{def:g10:functions:variations}{decreciente} de enteros no negativos.
\end{proof}

\begin{example}\label{ex:g12:arith:euclidalgo}
$\gcd(252, 198)$: $252 = 198 + 54$; $198 = 3\times54 + 36$;
$54 = 36 + 18$; $36 = 2 \times 18 + 0$. Por tanto,
$\gcd(252,198) = 18$.
\end{example}

\begin{theorem}[Identidad de Bézout]\label{thm:g12:arith:bezout}
\index{identidad de Bézout}
Sean $a, b$ enteros no nulos a la vez y sea $d = \gcd(a,b)$. Existen
$u, v \in \Z$ tales que
\[
au + bv = d .
\]
En particular, $a$ y $b$ son \omterm{def:g12:arith:gcd}{primos entre sí} si y solo si $au + bv = 1$
para ciertos enteros $u, v$.
\end{theorem}

\begin{proof}
Recorremos hacia atrás el \omterm{prop:g12:arith:euclidalgo}{algoritmo de Euclides}: cada resto es una
combinación entera de los dos anteriores, y los datos iniciales $a, b$ son
combinaciones de sí mismos; por sustitución descendente, el último resto no
nulo $d$ es una combinación entera de $a$ y $b$. (En el
\cref{ex:g12:arith:euclidalgo}:
$18 = 54 - 36 = 54 - (198 - 3\times54) = 4\times54 - 198 =
4(252 - 198) - 198 = 4\times252 - 5\times198$.)

Para la equivalencia: si $\gcd(a,b) = 1$, Bézout proporciona $u, v$;
recíprocamente, todo divisor común de $a$ y $b$ \omterm{def:g12:arith:divides}{divide} a $au + bv = 1$, lo
que obliga a $\gcd(a,b) = 1$.
\end{proof}

\begin{theorem}[Lema de Gauss]\label{thm:g12:arith:gauss}
\index{lema de Gauss}
Sean $a, b, c \in \Z$. Si $a \mid bc$ y $\gcd(a, b) = 1$, entonces
$a \mid c$.
\end{theorem}

\begin{proof}
Bézout da $au + bv = 1$; multiplicamos por $c$: $acu + bcv = c$. Los dos
términos del primer miembro son múltiplos de $a$ (el segundo porque
$a \mid bc$), luego $c$ también lo es.
\end{proof}

\begin{corollary}\label{cor:g12:arith:coprimeprod}
Si $a \mid c$, $b \mid c$ y $\gcd(a,b) = 1$, entonces $ab \mid c$.
\end{corollary}

\begin{proof}
Escribimos $c = ak$. De $b \mid ak$ y $\gcd(a,b)=1$, el lema de Gauss da
$b \mid k$, digamos $k = bl$; entonces $c = abl$.
\end{proof}

\section{Números primos}

\begin{definition}[Primo]\label{def:g12:arith:prime}
Un entero $p \geq 2$ es \emph{primo}\index{número primo} si sus únicos
divisores positivos son $1$ y $p$.
\end{definition}

\begin{proposition}\label{prop:g12:arith:primedivides}
Todo entero $n \geq 2$ tiene un divisor \omterm{def:g12:arith:prime}{primo}; y si $n$ no es \omterm{def:g12:arith:prime}{primo}, tiene
un divisor \omterm{def:g12:arith:prime}{primo} $\leq \sqrt n$. Si un \omterm{def:g12:arith:prime}{primo} $p$ \omterm{def:g12:arith:divides}{divide} a un producto $ab$,
entonces $p \mid a$ o $p \mid b$ (\emph{lema de Euclides}).
\end{proposition}

\begin{proof}
El menor divisor $d \geq 2$ de $n$ es \omterm{def:g12:arith:prime}{primo} (cualquier divisor propio de
$d$ sería un divisor menor de $n$). Si $n = de$ es compuesto con
$2 \leq d \leq e$, entonces $d^2 \leq de = n$, luego $d \leq \sqrt n$. Para
el lema de Euclides: si $p \nmid a$, entonces $\gcd(p, a) = 1$ (los únicos
divisores de $p$ son $1$ y $p$), y el lema de Gauss da $p \mid b$.
\end{proof}

\begin{theorem}[Euclides]\label{thm:g12:arith:infiniteprimes}
Hay infinitos \omterm{def:g12:arith:prime}{números primos}.
\end{theorem}

\begin{proof}
Dada una lista finita cualquiera $p_1, \dots, p_k$ de \omterm{def:g12:arith:prime}{primos}, consideramos
$N = p_1 p_2 \cdots p_k + 1$. Algún \omterm{def:g12:arith:prime}{primo} $p$ \omterm{def:g12:arith:divides}{divide} a $N$; pero ningún
$p_i$ \omterm{def:g12:arith:divides}{divide} a $N$ (el resto es $1$), así que $p$ es un \omterm{def:g12:arith:prime}{primo} que no está en
la lista. Ninguna lista finita agota los \omterm{def:g12:arith:prime}{primos}.
\end{proof}

\begin{theorem}[Teorema fundamental de la aritmética]\label{thm:g12:arith:fta}
\index{teorema fundamental de la aritmética}
Todo entero $n \geq 2$ es producto de \omterm{def:g12:arith:prime}{primos}, y esa factorización es única
salvo el orden de los factores:
\[
n = p_1^{\alpha_1} p_2^{\alpha_2} \cdots p_r^{\alpha_r},
\qquad p_1 < p_2 < \dots < p_r \text{ primos},\ \alpha_i \geq 1 .
\]
\end{theorem}

\begin{proof}
\emph{Existencia}, por inducción fuerte: si $n$ es \omterm{def:g12:arith:prime}{primo}, él mismo es su
factorización; si no, $n = de$ con $2 \leq d, e < n$, y los dos se
factorizan por la hipótesis de inducción. \emph{Unicidad}: supongamos
$p_1\cdots p_s = q_1 \cdots q_t$ (\omterm{def:g12:arith:prime}{primos}, con repeticiones permitidas). Por
el lema de Euclides, $p_1$ \omterm{def:g12:arith:divides}{divide} a algún $q_j$ y, al ser \omterm{def:g12:arith:prime}{primo},
$p_1 = q_j$; simplificamos y repetimos. Las dos factorizaciones coinciden
término a término.
\end{proof}

\begin{theorem}[Pequeño teorema de Fermat]\label{thm:g12:arith:fermat}
\index{pequeño teorema de Fermat}
Sea $p$ \omterm{def:g12:arith:prime}{primo} y sea $a \in \Z$ con $p \nmid a$. Entonces
\[
a^{p-1} \equiv 1 \pmod p .
\]
Para todo $a \in \Z$ (sin suponer que sean \omterm{def:g12:arith:gcd}{primos entre sí}),
$a^p \equiv a \pmod p$.
\end{theorem}

\begin{proof}
Consideremos los $p - 1$ enteros $a, 2a, 3a, \dots, (p-1)a$ \omterm{def:g12:complex:modulus}{módulo} $p$.
Ninguno es $\equiv 0$ (si $p \mid ka$ con $1 \leq k \leq p-1$, el lema de
Euclides obliga a $p \mid k$, imposible) y son distintos dos a dos \omterm{def:g12:complex:modulus}{módulo}
$p$ (si $ka \equiv la$, entonces $p \mid (k - l)a$, luego $p \mid k - l$ y
$k = l$). Por tanto, \omterm{def:g12:complex:modulus}{módulo} $p$ son los números $1, 2, \dots, p-1$ en algún
orden. Multiplicando todas las \omterm{def:g12:arith:congruence}{congruencias}:
\[
a^{p-1}\,(p-1)! \equiv (p-1)! \pmod p .
\]
Como $p$ no \omterm{def:g12:arith:divides}{divide} a ninguno de los $1, \dots, p-1$, aplicar repetidamente
el lema de Euclides permite cancelar $(p-1)!$ y queda
$a^{p-1} \equiv 1$. La segunda forma se sigue multiplicando por $a$ (y es
trivial cuando $p \mid a$).
\end{proof}

\begin{example}[Aplicación a la criptografía]\label{ex:g12:arith:rsa}
El teorema de Fermat hace reversible la exponenciación \omterm{def:g12:complex:modulus}{módulo} $n$ cuando
los exponentes se eligen adecuadamente: ese es el corazón del criptosistema
\emph{RSA}. Con $p, q$ \omterm{def:g12:arith:prime}{primos} grandes y $n = pq$, se publican $n$ y un
exponente $e$; cifrar es $x \mapsto x^e \bmod n$. Descifrar exige un
exponente $d$ con $ed \equiv 1 \pmod{(p-1)(q-1)}$, que solo puede calcular
quien conozca $p$ y $q$; y recuperar $p$ y $q$ a partir de $n$ significa
\omterm{def:g10:algebra:expand}{factorizar} un número de cientos de cifras, cosa que ningún algoritmo
conocido hace en un tiempo razonable.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:arith:1}
Calcula el cociente y el resto de la \omterm{thm:g12:arith:euclid}{división euclídea} de $2026$ entre
$17$, y de $-2026$ entre $17$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:arith:2}
¿Cuál es el resto de $7^{100}$ \omterm{def:g12:complex:modulus}{módulo} $10$? (¿Cuál es la última cifra de
$7^{100}$?)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:arith:3}
Usando el \omterm{prop:g12:arith:euclidalgo}{algoritmo de Euclides}, calcula $\gcd(1071, 462)$ y halla enteros
$u, v$ con $1071u + 462v = \gcd(1071, 462)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:arith:4}
Demuestra que para todo $n \in \Z$, $n^2$ es congruente con $0$ o con $1$
\omterm{def:g12:complex:modulus}{módulo} $4$. Deduce que un entero $\equiv 3 \pmod 4$ nunca es suma de dos
cuadrados.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:arith:5}
Demuestra que para todo $n \in \N$, $n(n+1)(2n+1)$ es divisible entre $6$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:arith:6}
Resuelve en $\Z$ la \omterm{def:g12:arith:congruence}{congruencia} $5x \equiv 3 \pmod{11}$. (Indicación: halla
el inverso de $5$ \omterm{def:g12:complex:modulus}{módulo} $11$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:arith:7}
Resuelve en $\Z \times \Z$ la \omterm{def:g10:algebra:equation}{ecuación} diofántica
\[
17x - 40y = 1,
\]
y describe después todas las soluciones de $17x - 40y = 6$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:arith:8}
Demuestra que $\sqrt2$ es irracional usando la unicidad de la factorización
en \omterm{def:g12:arith:prime}{primos} (compara el exponente de $2$ en los dos miembros de
$a^2 = 2b^2$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:arith:9}
Sea $p$ un \omterm{def:g12:arith:prime}{primo}.
\begin{enumerate}
  \item Demuestra que para $1 \leq k \leq p - 1$, $p$ \omterm{def:g12:arith:divides}{divide} a
        $\dbinom{p}{k}$. (Indicación: usa
        $k\binom pk = p\binom{p-1}{k-1}$, \cref{exo:g12:comb:7}, y el lema
        de Gauss.)
  \item Deduce, por inducción sobre $a \geq 0$, otra demostración del
        pequeño teorema de Fermat en la forma $a^p \equiv a \pmod p$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:arith:10}
\emph{(Problema chino de los restos.)} Halla todos los enteros $n$ tales
que
\[
n \equiv 2 \pmod 3, \qquad n \equiv 3 \pmod 5, \qquad n \equiv 2 \pmod 7 .
\]
(Indicación: resuelve las dos primeras condiciones e incorpora después la
tercera; los coeficientes de Bézout ayudan.)
\end{exercise}

\section{Problema: Códigos secretos y dígitos de control}

\begin{problem}[{Problema de fin de semana --- las congruencias custodian
todos los códigos de barras y todas las tarjetas de crédito, y el pequeño
teorema de Fermat maneja la cerradura de los secretos del mundo}]
\label{pb:g12:arith:1}
G.\,H.~Hardy presumía en 1940 de que la teoría de números estaba ``sin
mancillar'' por las aplicaciones. Ochenta años después, cada pitido de un
código de barras, cada pago con tarjeta y cada mensaje cifrado lo
contradicen, y con exactamente las herramientas de este capítulo: las
\omterm{def:g12:arith:congruence}{congruencias} (\cref{prop:g12:arith:congops}), los inversos de Bézout
(\cref{thm:g12:arith:bezout}) y el pequeño teorema de Fermat
(\cref{exo:g12:arith:9}). Este problema comprueba los códigos, fuerza una
versión de juguete de la cerradura y descubre por qué aguanta la de
verdad.

\medskip
\noindent\textbf{Parte I --- Soltura con las \omterm{def:g12:arith:congruence}{congruencias}.}
\begin{enumerate}
  \item Calcula $2026 \bmod 7$; y después la última cifra de $7^{100}$
        (halla el ciclo de las potencias de $7$ \omterm{def:g12:complex:modulus}{módulo} $10$).
  \item Exponenciación rápida (\cref{met:g12:arith:powers}): calcula
        $5^{117} \bmod 13$ (parte de $5^2 \equiv -1$).
  \item Resuelve $3x \equiv 5 \pmod 7$.
  \item Aplica el \omterm{prop:g12:arith:euclidalgo}{algoritmo de Euclides} a $(97, 35)$, sustituye hacia atrás
        para hallar enteros $u, v$ con $97u + 35v = 1$ y deduce el inverso
        de $35$ \omterm{def:g12:complex:modulus}{módulo} $97$.
  \item Enuncia con precisión cuándo es $a$ invertible \omterm{def:g12:complex:modulus}{módulo} $n$ y qué
        teorema entrega el inverso.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Dígitos de control.}
\begin{enumerate}[resume]
  \item ISBN-10: las diez cifras $d_1 \dots d_{10}$ del código de un libro
        deben cumplir
        $10d_1 + 9d_2 + \dots + 2d_9 + 1d_{10} \equiv 0 \pmod{11}$.
        Comprueba el ISBN real $0\,306\,40615\,2$.
  \item Demuestra que el esquema ISBN detecta \emph{todos} los errores de
        una sola cifra: si una cifra cambia en $d \not\equiv 0$, la suma
        ponderada cambia en $w d$ con $1 \leq w \leq 10$; ¿por qué eso
        nunca puede ser $\equiv 0 \pmod{11}$ (\cref{thm:g12:arith:gauss})?
  \item Demuestra que también detecta cualquier trasposición de dos cifras
        adyacentes (distintas). Explica después el secreto del diseño: ¿qué
        propiedad del $11$ hizo funcionar las dos demostraciones y qué
        podría salir mal con el \omterm{def:g12:complex:modulus}{módulo} $10$?
  \item Los códigos de barras EAN-13 ponderan las cifras con
        $1, 3, 1, 3, \dots$ \omterm{def:g12:complex:modulus}{módulo} $10$. Calcula el dígito de control que
        completa $978\,2940199\,05$. ¿Qué trasposiciones adyacentes
        \emph{no} detecta el EAN? (¿Cuándo es
        $2(a - b) \equiv 0 \pmod{10}$?)
  \item Las tarjetas de crédito usan el esquema de Luhn: desde la derecha,
        se dobla una cifra de cada dos (restando $9$ cuando el doble pasa
        de $9$), se suma todo y se exige un múltiplo de $10$. Comprueba el
        número de prueba $4539\,1488\,0343\,6467$.
  \item En una frase: ¿qué le compró el \omterm{def:g12:complex:modulus}{módulo} \omterm{def:g12:arith:prime}{primo} al ISBN que el EAN y
        Luhn, encadenados al $10$, no pueden tener?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La cerradura de Fermat.}
\begin{enumerate}[resume]
  \item Una trampa antes del tesoro: calcula $2^{10} \bmod 341$, deduce
        $2^{340} \bmod 341$ y factoriza después $341$. ¿Qué dice este
        ejemplo (un \emph{pseudoprimo de Fermat}) sobre usar el pequeño
        teorema de Fermat como test de primalidad?
  \item RSA en miniatura: tomamos $p = 3$ y $q = 11$, de modo que $n = 33$
        y $(p-1)(q-1) = 20$; el exponente público es $e = 3$. Halla el
        exponente privado $d$ con $3d \equiv 1 \pmod{20}$ (el método de la
        pregunta 4).
  \item Cifra el mensaje $m = 4$: calcula $c = m^3 \bmod 33$.
  \item Descifra: calcula $c^d \bmod 33$ (usa $c \equiv -2 \pmod{33}$) y
        recupera el mensaje.
  \item Por qué funciona siempre el descifrado: demuestra que
        $m^{21} \equiv m$ tanto \omterm{def:g12:complex:modulus}{módulo} $3$ como \omterm{def:g12:complex:modulus}{módulo} $11$ (el pequeño
        teorema de Fermat en cada mundo) y concluye \omterm{def:g12:complex:modulus}{módulo} $33$ (el
        \cref{thm:g12:arith:gauss} pega las dos \omterm{def:g12:arith:congruence}{congruencias}). ¿Dónde entró
        la forma especial $1 + 20k$ de $21 = ed$?
  \item La seguridad de la cerradura: todo el mundo conoce $n$ y $e$;
        recuperar $d$ exige $(p-1)(q-1)$ y, por tanto, los factores de $n$.
        Nuestro $33$ se factoriza de un vistazo; ¿por qué el mismo esquema,
        con un $n$ de seiscientas cifras, protege a los bancos del mundo?
        (Una frase sobre la asimetría entre multiplicar y \omterm{def:g10:algebra:expand}{factorizar}.)
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Clásicos.}
\begin{enumerate}[resume]
  \item El viejo recuento chino de soldados (compara con el
        \cref{exo:g12:arith:10}): un número de soldados deja resto $2$ al
        formar de tres en tres y resto $3$ al formar de cinco en cinco.
        Halla todos los recuentos posibles y explica por qué la respuesta
        es única \omterm{def:g12:complex:modulus}{módulo} $15$.
  \item Demostraciones de una línea, por fin: de $10 \equiv 1 \pmod 9$,
        demuestra que todo número es congruente con la suma de sus cifras
        \omterm{def:g12:complex:modulus}{módulo} $9$; y de $10 \equiv -1 \pmod{11}$, deduce la regla de la
        suma alternada para el $11$. (El volumen anterior las demostró con
        álgebra explícita; admira la compresión.)
  \item Final: Hardy contra el código de barras: repasa la caja de
        herramientas del capítulo (la aritmética de \omterm{def:g12:arith:congruence}{congruencias}, los
        inversos de Bézout, el pequeño teorema de Fermat y el pegado de
        \omterm{def:g12:complex:modulus}{módulos} \omterm{def:g12:arith:gcd}{primos entre sí}) y dónde encajó cada una en este problema;
        y da después el veredicto moderno sobre lo de ``sin mancillar''.
\end{enumerate}
\end{problem}
