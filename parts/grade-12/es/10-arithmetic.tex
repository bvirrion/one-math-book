La aritmética estudia la números enteros: \omterm{def:g12:arith:divides}{divisibilidad}, numeros primos, restos.
Considerada durante mucho tiempo la más pura de las matemáticas puras, ahora protege todas las aplicaciones en línea.
pago: el criptosistema RSA se basa en los teoremas de Bézout, Gauss y

\section{Divisibilidad y división euclidiana}

\begin{definition}[Divisibilidad]\label{def:g12:arith:divides}
Sea $a, b \in \Z $. decimos $ b $ \emph{divide} $ a $\index{divisibilidad}, escrito
$ b \mid a $, si existe $ k \in \Z $ con $ a = kb $. También decimos $ a $ es un
\emph{múltiple} de $ b $.
\end{definition}

\begin{proposition}\label{prop:g12:arith:divprops}
Si $ c \mid a $ y $ c \mid b $, entonces $ c $ dividir cada entero combinación
$ au + bv $ ($ u, v \in \Z $). Si $ a \mid b $ y $ b \mid a $ con
$ a,b \in \N $, entonces $ a = b $. Si $ a \mid b $ y $ b \neq 0$, entonces $\abs a \leq \abs b $.
\end{proposition}

\begin{proof}
Escribir $ a = kc $, $ b = lc $: entonces $ au + bv = (ku + lv)c $. Los otros puntos
seguir desde $\abs{a} = \abs{k}\,\abs{b}$ con $\abs k \geq 1$ cuando
$ b = ka \neq 0$.
\end{proof}

\begin{theorem}[división euclidiana]\label{thm:g12:arith:euclid}
\index{división euclidiana}
Sea $ a \in \Z $ y $ b \in \N^*$. Existe un par único
$(q, r) \in \Z \times \N $ tal que
\[
a = bq + r \qquad\text{and}\qquad 0 \leq r < b .
\]
$ q $ es el \emph{cociente} y $ r $ el \emph{resto}.
\end{theorem}

\begin{proof}
\emph{Existencia.} El conjunto de múltiplos de $ b $ sin exceder $ a $ tiene un
elemento más grande $ bq $ (no está vacío y \omterm{def:g12:seq:bounded}{delimitado arriba}); colocar $ r = a - bq $.
Por maximalidad, $ b(q+1) > a $, entonces $0 \leq r < b $.
\emph{Unicidad.} Si $ bq + r = bq' + r'$ con $0 \leq r, r' < b $, entonces
$ b(q - q') = r' - r $ y $\abs{r' - r} < b $: un múltiplo de $ b $ de \omterm{def:g10:numbers:abs}{valor absoluto} menos que $ b $ debe ser $0$, entonces $ r = r'$ y $ q = q'$.
\end{proof}

\section{Congruencias}

\begin{definition}[Congruencia]\label{def:g12:arith:congruence}
Sea $ n \in \N^*$. Dos números enteros $ a, b $ son \emph{módulo congruente
$ n $}\index{módulo congruente
$ n $}, escrito $ a \equiv b \pmod n $, si $ n \mid (a - b)$
--- equivalentemente, si $ a $ y $ b $ tener el mismo resto en el euclidiano
división por $ n $.
\end{definition}

\begin{proposition}[Compatibilidad con operaciones]\label{prop:g12:arith:congops}
Si $ a \equiv b \pmod n $ y $ c \equiv d \pmod n $, entonces
\[
a + c \equiv b + d, \qquad
ac \equiv bd, \qquad
a^k \equiv b^k \ (k \in \N) \pmod n .
\]
\end{proposition}

\begin{proof}
$ n $ dividir $(a-b) + (c-d) = (a+c) - (b+d)$, y
$ ac - bd = a(c - d) + d(a - b)$ también es múltiplo de $ n $. La regla del poder
sigue por inducción de la regla del producto.
\end{proof}

\begin{method}[Módulo de potencias informáticas $ n $]\label{met:g12:arith:powers}
para calcular $ a^k \bmod n $, reducir el \omterm{def:g12:complex:modulus}{módulo} base $ n $, luego busca un pequeño
poder de $ a $ congruente con $\pm1$ y utilícelo para colapsar el exponente. Por ejemplo $2^{100} \bmod 7$: desde $2^3 = 8 \equiv 1 \pmod 7$ y
$100 = 3\times33 + 1$,
\[
2^{100} = \left(2^{3}\right)^{33} \times 2 \equiv 1^{33}\times 2 = 2 \pmod 7 .
\]
\end{method}

\section{MCD, Bézout y Gauss}

\begin{definition}[MCD]\label{def:g12:arith:gcd}
Sea $ a, b $ ser números enteros, no ambos cero. El \emph{alcalde común
divisor}\index{alcalde común
divisor} $\gcd(a, b)$ es el mas grande entero dividiendo ambos $ a $
y $ b $. Cuando $\gcd(a,b) = 1$, $ a $ y $ b $ se dice que son
\emph{coprime}\index{coprime}.
\end{definition}

\begin{proposition}[algoritmo de euclides]\label{prop:g12:arith:euclidalgo}
\index{algoritmo de euclides}
Si $ a = bq + r $ ($ b \neq 0$), entonces $\gcd(a, b) = \gcd(b, r)$. Iterando el
\omterm{thm:g12:arith:euclid}{división euclidiana} por lo tanto calcula $\gcd(a,b)$: el mcd es el ultimo
resto distinto de cero.
\end{proposition}

\begin{proof}
Cualquier divisor común de $ a $ y $ b $ dividir $ r = a - bq $
(\cref{prop:g12:arith:divprops}), por lo tanto es un divisor común de $ b $ y $ r $;
y a la inversa, ya que $ a = bq + r $. Los dos pares tienen el mismo común
divisores, entonces lo mismo mcd. El algoritmo termina porque los restos
formar una estricta \omterm{def:g10:functions:variations}{decreciente} \omterm{def:g12:seq:sequence}{sucesión} de no negativo números enteros.
\end{proof}

\begin{example}\label{ex:g12:arith:euclidalgo}
$\gcd(252, 198)$: $252 = 198 + 54$; $198 = 3\times54 + 36$;
$54 = 36 + 18$; $36 = 2 \times 18 + 0$. Por eso $\gcd(252,198) = 18$.
\end{example}

\begin{theorem}[La identidad de Bézout]\label{thm:g12:arith:bezout}
\index{La identidad de Bézout}
Sea $ a, b $ ser números enteros, no ambos cero, y $ d = \gcd(a,b)$. existen
$ u, v \in \Z $ tal que
\[
au + bv = d .
\]
En particular, $ a $ y $ b $ son \omterm{def:g12:arith:gcd}{coprime} si y solo si $ au + bv = 1$ para algunos
números enteros $ u, v $.
\end{theorem}

\begin{proof}
Correr \omterm{prop:g12:arith:euclidalgo}{algoritmo de euclides} al revés: cada resto es un entero combinación
de los dos anteriores, y los datos iniciales $ a, b $ son combinaciones de
ellos mismos; mediante sustitución descendente, el último resto distinto de cero $ d $ es un
entero combinación de $ a $ y $ b $. (En \cref{ex:g12:arith:euclidalgo}:
$18 = 54 - 36 = 54 - (198 - 3\times54) = 4\times54 - 198 =
4(252 - 198) - 198 = 4\times252 - 5\times198$.)

Para la equivalencia: si $\gcd(a,b) = 1$, Bézout ofrece $ u, v $; por el contrario,
cualquier divisor común de $ a $ y $ b $ dividir $ au + bv = 1$, forzando
$\gcd(a,b) = 1$.
\end{proof}

\begin{theorem}[Lema de Gauss]\label{thm:g12:arith:gauss}
\index{Lema de Gauss}
Sea $ a, b, c \in \Z $. Si $ a \mid bc $ y $\gcd(a, b) = 1$, entonces $ a \mid c $.
\end{theorem}

\begin{proof}
Bézout da $ au + bv = 1$; multiplicar por $ c $: $ acu + bcv = c $. Ambos términos de
el lado izquierdo son múltiplos de $ a $ (el segundo porque $ a \mid bc $), por lo tanto
también lo es $ c $.
\end{proof}

\begin{corollary}\label{cor:g12:arith:coprimeprod}
Si $ a \mid c $, $ b \mid c $ y $\gcd(a,b) = 1$, entonces $ ab \mid c $.
\end{corollary}

\begin{proof}
Escribir $ c = ak $. De $ b \mid ak $ y $\gcd(a,b)=1$, Gauss da
$ b \mid k $, decir $ k = bl $; entonces $ c = abl $.
\end{proof}

\section{Números primos}

\begin{definition}[Principal]\label{def:g12:arith:prime}
Un entero $ p \geq 2$ es \emph{principal}\index{principal} si es solo
los divisores positivos son $1$ y $ p $.
\end{definition}

\begin{proposition}\label{prop:g12:arith:primedivides}
Cada entero $ n \geq 2$ tiene un \omterm{def:g12:arith:prime}{principal} divisor; si $ n $ no es \omterm{def:g12:arith:prime}{principal}, tiene un
\omterm{def:g12:arith:prime}{principal} divisor $\leq \sqrt n $. si un \omterm{def:g12:arith:prime}{principal} $ p $ dividir un producto $ ab $, entonces
$ p \mid a $ o $ p \mid b $ (\emph{Lema de Euclides}).
\end{proposition}

\begin{proof}
El divisor más pequeño $ d \geq 2$ de $ n $ es \omterm{def:g12:arith:prime}{principal} (cualquier divisor propio de $ d $
sería un divisor menor de $ n $). Si $ n = de $ es compuesto con
$2 \leq d \leq e $, entonces $ d^2 \leq de = n $, entonces $ d \leq \sqrt n $. Para Euclides
lema: si $ p \nmid a $, entonces $\gcd(p, a) = 1$ (los únicos divisores de $ p $ son
$1$ y $ p $), y el lema de Gauss da $ p \mid b $.
\end{proof}

\begin{theorem}[Euclides]\label{thm:g12:arith:infiniteprimes}
Hay infinitos primos.
\end{theorem}

\begin{proof}
Dada cualquier lista finita $ p_1, \dots, p_k $ de primos, considerar
$ N = p_1 p_2 \cdots p_k + 1$. Alguno \omterm{def:g12:arith:prime}{principal} $ p $ dividir $ N $; pero no $ p_i $
dividir $ N $ (el resto es $1$), entonces $ p $ es un \omterm{def:g12:arith:prime}{principal} no en la lista. No
lista finita agota el primos.
\end{proof}

\begin{theorem}[Teorema fundamental de la aritmética]\label{thm:g12:arith:fta}
\index{teorema fundamental de la aritmética}
Cada entero $ n \geq 2$ es un producto de primos, y esta \omterm{def:g10:algebra:expand}{factorización} es
único hasta el orden de los factores:
\[
n = p_1^{\alpha_1} p_2^{\alpha_2} \cdots p_r^{\alpha_r},
\qquad p_1 < p_2 < \dots < p_r \text{ primes},\ \alpha_i \geq 1 .
\]
\end{theorem}

\begin{proof}
\emph{Existencia}, por fuerte inducción: $ n $ \omterm{def:g12:arith:prime}{principal} es su propia \omterm{def:g10:algebra:expand}{factorización};
de lo contrario $ n = de $ con $2 \leq d, e < n $, y ambos factorizan por la inducción
hipótesis. \emph{Unicidad}: suponer
$ p_1\cdots p_s = q_1 \cdots q_t $ (primos, con repeticiones permitidas). Por
El lema de Euclides, $ p_1$ dividir alguno $ q_j $, y siendo \omterm{def:g12:arith:prime}{principal}, $ p_1 = q_j $;
cancelar y repetir. Las dos factorizaciones coinciden término por término.
\end{proof}

\begin{theorem}[El pequeño teorema de Fermat]\label{thm:g12:arith:fermat}
\index{El pequeño teorema de Fermat}
Sea $ p $ ser \omterm{def:g12:arith:prime}{principal} y $ a \in \Z $ con $ p \nmid a $. Entonces
\[
a^{p-1} \equiv 1 \pmod p .
\]
por cada $ a \in \Z $ (no se asume coprimalidad), $ a^p \equiv a \pmod p $.
\end{theorem}

\begin{proof}
Considere el $ p - 1$ números enteros $ a, 2a, 3a, \dots, (p-1)a $ \omterm{def:g12:complex:modulus}{módulo} $ p $. Ninguno
es $\equiv 0$ (si $ p \mid ka $ con $1 \leq k \leq p-1$, las fuerzas del lema de Euclides
$ p \mid k $, imposible), y son \omterm{def:g12:complex:modulus}{módulos} distintos por pares $ p $ (si
$ ka \equiv la $, entonces $ p \mid (k - l)a $, entonces $ p \mid k - l $, entonces $ k = l $).
Por lo tanto, \omterm{def:g12:complex:modulus}{módulo} $ p $, son los numeros $1, 2, \dots, p-1$ en algún orden.
Multiplicando todo congruencias:
\[
a^{p-1}\,(p-1)! \equiv (p-1)! \pmod p .
\]
Desde $ p $ dividir ninguno de $1, \dots, p-1$, uso repetido del lema de Euclides
permite cancelar $(p-1)!$, partida $ a^{p-1} \equiv 1$. la segunda forma
sigue multiplicando por $ a $ (y es trivial cuando $ p \mid a $).
\end{proof}

\begin{example}[Aplicación a la criptografía]\label{ex:g12:arith:rsa}
El teorema de Fermat hace \omterm{def:g12:complex:modulus}{módulo} de exponenciación $ n $ reversible cuando el
los exponentes se eligen adecuadamente --- el corazón del \emph{RSA} criptosistema.
Con $ p, q $ grande primos y $ n = pq $, uno publica $ n $ y un exponente
$ e $; el cifrado es $ x \mapsto x^e \bmod n $. Descifrar requiere un exponente
$ d $ con $ ed \equiv 1 \pmod{(p-1)(q-1)}$, que solo alguien lo sabe $ p $
y $ q $ puede calcular --- y recuperar $ p, q $ de $ n $ medio \omterm{def:g10:algebra:expand}{factorización} un
número de cientos de dígitos, algo que ningún algoritmo conocido hace en términos razonables.
tiempo.
\end{example}

\section{Problema: códigos secretos y dígitos de control}

\begin{problem}[{Problema del fin de semana --- las congruencias protegen cada
código de barras y tarjeta de crédito, y el pequeño teorema de Fermat ejecuta el
bloquear los secretos del mundo}]
\label{pb:g12:arith:1}
GRAMO.\,H.~Hardy se jactó en 1940 de que la teoría de números era "inmaculada"
por aplicaciones. Ochenta años después, cada código de barras emite un pitido, cada
pago con tarjeta de crédito y cada mensaje cifrado lo contradice
--- con exactamente las herramientas de este capítulo: congruencias
(\cref{prop:g12:arith:congops}), Bézout inversas
(\cref{thm:g12:arith:bezout}) y el pequeño teorema de Fermat
(\cref{exo:g12:arith:9}). Este problema verifica los códigos, rompe un
versión de juguete de la cerradura y aprende por qué la cerradura real se mantiene firme.

\medskip

        deducir la inversa de $35$ \omterm{def:g12:complex:modulus}{módulo} $97$.

        \omterm{def:g12:complex:modulus}{módulos}) y dónde cada uno encajó en su lugar en este
        problema; luego dé el veredicto moderno sobre
        ``inmaculado''.
\end{enumerate}
