\chapter{Sumas de variables aleatorias y la ley de los grandes números}\label{ch:g12:sums}

¿Por qué acaban ganando siempre los casinos y por qué funcionan las
encuestas? Porque los promedios de muchas cantidades aleatorias
\omterm{def:g12:sums:indep}{independientes} fluctúan cada vez menos. Este capítulo lo demuestra:
linealidad de la \omterm{def:g12:randvar:exp}{esperanza}, aditividad de la \omterm{def:g12:randvar:exp}{varianza} para variables
\omterm{def:g12:sums:indep}{independientes}, desigualdad de Bienaymé--Chebyshev y ley de los grandes
números.

\section{Sumas de variables aleatorias}

Dadas dos \omterm{def:g12:randvar:rv}{variables aleatorias} $X, Y$ sobre el mismo \omterm{def:g11:prob:model}{espacio muestral}
finito $\Omega$, la suma $X + Y$ es la \omterm{def:g12:randvar:rv}{variable aleatoria}
$\omega \mapsto X(\omega) + Y(\omega)$.

\begin{theorem}[Linealidad de la esperanza]\label{thm:g12:sums:linearity}
\index{esperanza!linealidad}
Para cualesquiera \omterm{def:g12:randvar:rv}{variables aleatorias} $X, Y$ sobre $\Omega$ y
$a, b \in \R$:
\[
\E(X + Y) = \E(X) + \E(Y), \qquad \E(aX + b) = a\E(X) + b .
\]
\emph{No hace falta ninguna hipótesis de \omterm{def:g12:condprob:indep}{independencia}.}
\end{theorem}

\begin{proof}
Escribimos la \omterm{def:g12:randvar:exp}{esperanza} como suma sobre los resultados: como
$\P(X = x) = \sum_{\omega : X(\omega) = x} \P(\{\omega\})$, agrupar
términos da $\E(X) = \sum_{\omega \in \Omega} \P(\{\omega\})\,X(\omega)$.
Entonces
\[
\E(X + Y) = \sum_{\omega} \P(\{\omega\})\bigl(X(\omega) + Y(\omega)\bigr)
= \sum_{\omega} \P(\{\omega\})X(\omega)
+ \sum_{\omega} \P(\{\omega\})Y(\omega) = \E(X) + \E(Y). \qedhere
\]
\end{proof}

\begin{definition}[Variables aleatorias independientes]\label{def:g12:sums:indep}
$X$ e $Y$ son \emph{independientes}\index{variable aleatoria!independiente}
si para todos los valores $x, y$:
\[
\P(X = x \text{ e } Y = y) = \P(X = x)\,\P(Y = y).
\]
Varias variables $X_1, \dots, X_n$ son independientes si esta regla del
producto vale para cualquier elección de valores de cualquier subfamilia.
\end{definition}

\begin{proposition}\label{prop:g12:sums:prodexp}
Si $X$ e $Y$ son \omterm{def:g12:sums:indep}{independientes}, $\E(XY) = \E(X)\,\E(Y)$.
\end{proposition}

\begin{proof}
\[
\begin{aligned}
\E(XY) &= \sum_{x, y} xy\;\P(X = x \text{ e } Y = y)
= \sum_{x, y} xy\,\P(X=x)\P(Y=y) \\
&= \Bigl(\sum_x x\P(X=x)\Bigr)\Bigl(\sum_y y\P(Y=y)\Bigr). \qedhere
\end{aligned}
\]
\end{proof}

\begin{theorem}[Varianza de una suma]\label{thm:g12:sums:variance}
\index{varianza!de una suma}
Si $X$ e $Y$ son \emph{\omterm{def:g12:sums:indep}{independientes}}, entonces
\[
\V(X + Y) = \V(X) + \V(Y).
\]
Más en general, para $X_1, \dots, X_n$ \omterm{def:g12:sums:indep}{independientes}:
$\V(X_1 + \dots + X_n) = \V(X_1) + \dots + \V(X_n)$.
\end{theorem}

\begin{proof}
Usando König--Huygens (\cref{prop:g12:randvar:konig}) y la linealidad:
\begin{align*}
\V(X + Y) &= \E\bigl((X+Y)^2\bigr) - \bigl(\E X + \E Y\bigr)^2\\
&= \E(X^2) + 2\E(XY) + \E(Y^2)
- \E(X)^2 - 2\E(X)\E(Y) - \E(Y)^2\\
&= \V(X) + \V(Y) + 2\bigl(\E(XY) - \E(X)\E(Y)\bigr),
\end{align*}
y el último paréntesis se anula para variables \omterm{def:g12:sums:indep}{independientes}
(\cref{prop:g12:sums:prodexp}). El caso general se sigue por inducción.
\end{proof}

\begin{example}[La binomial, revisitada]\label{ex:g12:sums:binomial}
Una variable binomial $X \sim \mathcal B(n, p)$ es una suma
$X = X_1 + \dots + X_n$ de $n$ variables de Bernoulli \omterm{def:g12:sums:indep}{independientes}. Por
tanto, estructuralmente:
\[
\E(X) = np, \qquad \V(X) = np(1-p),
\]
con lo que recuperamos el \cref{thm:g12:randvar:binomial} sin ningún
cálculo.
\end{example}

\begin{proposition}[Media muestral]\label{prop:g12:sums:mean}
Sean $X_1, \dots, X_n$ \omterm{def:g12:sums:indep}{independientes} y con la misma \omterm{def:g12:randvar:rv}{distribución} que $X$
(\omterm{def:g12:randvar:exp}{esperanza} $\mu$, \omterm{def:g12:randvar:exp}{varianza} $\sigma^2$), y sea
$M_n = \frac{X_1 + \dots + X_n}{n}$ su \emph{media muestral}\index{media
muestral}. Entonces
\[
\E(M_n) = \mu, \qquad \V(M_n) = \frac{\sigma^2}{n}, \qquad
\sigma(M_n) = \frac{\sigma}{\sqrt n}.
\]
\end{proposition}

\begin{proof}
La linealidad da $\E(M_n) = \frac{n\mu}{n} = \mu$; la \omterm{def:g12:condprob:indep}{independencia} da
$\V(X_1 + \dots + X_n) = n\sigma^2$, y dividir entre $n$ multiplica la
\omterm{def:g12:randvar:exp}{varianza} por $\frac{1}{n^2}$ (\cref{prop:g12:randvar:affine}).
\end{proof}

El $\frac{\sigma}{\sqrt n}$ es la \emph{ley de la \omterm{def:g11:quad:discriminant}{raíz} cuadrada}
fundamental: para reducir a la mitad las fluctuaciones de un promedio hay
que cuadruplicar el tamaño de la \omterm{def:g10:proba:sample}{muestra}.

\section{Desigualdades de concentración}

\begin{theorem}[Desigualdad de Markov]\label{thm:g12:sums:markov}
\index{desigualdad de Markov}
Si $X \geq 0$ y $a > 0$:
\[
\P(X \geq a) \leq \frac{\E(X)}{a}.
\]
\end{theorem}

\begin{proof}
En $\E(X) = \sum_i p_i x_i$ (con todos los términos no negativos),
conservamos solo los términos con $x_i \geq a$: cada uno vale al menos
$a\,p_i$, luego
$\E(X) \geq a \sum_{x_i \geq a} p_i = a\,\P(X \geq a)$.
\end{proof}

\begin{theorem}[Desigualdad de Bienaymé--Chebyshev]\label{thm:g12:sums:chebyshev}
\index{desigualdad de Bienaymé--Chebyshev}
Para cualquier \omterm{def:g12:randvar:rv}{variable aleatoria} $X$ y cualquier $\varepsilon > 0$:
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
\leq \frac{\V(X)}{\varepsilon^2}.
\]
\end{theorem}

\begin{proof}
Aplicamos la desigualdad de Markov a la variable no negativa
$Y = (X - \E(X))^2$ con $a = \varepsilon^2$:
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
= \P(Y \geq \varepsilon^2)
\leq \frac{\E(Y)}{\varepsilon^2} = \frac{\V(X)}{\varepsilon^2}. \qedhere
\]
\end{proof}

\begin{theorem}[Ley de los grandes números]\label{thm:g12:sums:lln}
\index{ley de los grandes números}
Sean $X_1, \dots, X_n$ \omterm{def:g12:sums:indep}{independientes} y con la misma \omterm{def:g12:randvar:rv}{distribución}
(\omterm{def:g12:randvar:exp}{esperanza} $\mu$, \omterm{def:g12:randvar:exp}{varianza} $\sigma^2$) y sea $M_n$ su \omterm{prop:g12:sums:mean}{media muestral}. Para
todo $\varepsilon > 0$:
\[
\P\bigl(\abs{M_n - \mu} \geq \varepsilon\bigr)
\leq \frac{\sigma^2}{n\,\varepsilon^2}
\xrightarrow[n \to +\infty]{} 0 .
\]
La \omterm{prop:g12:sums:mean}{media muestral} se concentra alrededor de la \omterm{def:g12:randvar:exp}{esperanza}.
\end{theorem}

\begin{proof}
Bienaymé--Chebyshev aplicada a $M_n$, cuya \omterm{def:g12:randvar:exp}{esperanza} es $\mu$ y cuya
\omterm{def:g12:randvar:exp}{varianza} es $\frac{\sigma^2}{n}$ (\cref{prop:g12:sums:mean}).
\end{proof}

\begin{remark}
Este teorema es el puente entre la teoría de la \omterm{def:g10:proba:distribution}{probabilidad} y la
estadística: la \emph{frecuencia} de un \omterm{def:g11:prob:model}{suceso} a lo largo de muchas
repeticiones \omterm{def:g12:sums:indep}{independientes} se acerca a su \emph{\omterm{def:g10:proba:distribution}{probabilidad}} (basta tomar
como $X_i$ el indicador del \omterm{def:g11:prob:model}{suceso}, con lo que $\mu = p$). Justifica
estimar una \omterm{def:g10:proba:distribution}{probabilidad} por simulación (\emph{método de Montecarlo}) y una
proporción de la población mediante una encuesta, y además de manera
cuantitativa: véase el \cref{ch:g12:contdist}.
\end{remark}

\begin{method}[Usar Bienaymé--Chebyshev]\label{met:g12:sums:chebyshev}
Para garantizar $\P(\abs{M_n - \mu} \geq \varepsilon) \leq \alpha$ basta
con que $n \geq \frac{\sigma^2}{\alpha\,\varepsilon^2}$. La cota es burda
(las fluctuaciones reales suelen ser mucho menores), pero perfectamente
general: no exige nada sobre la \omterm{def:g12:randvar:rv}{distribución} más allá de que tenga \omterm{def:g12:randvar:exp}{varianza}
finita.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:sums:1}
Se lanzan dos dados equilibrados; sea $S$ la suma. Usando la linealidad (¡y
no la \omterm{def:g12:randvar:rv}{distribución} de $S$!), calcula $\E(S)$; calcula después $\V(S)$
usando la \omterm{def:g12:condprob:indep}{independencia}, sabiendo que un solo dado equilibrado tiene
\omterm{def:g12:randvar:exp}{varianza} $\frac{35}{12}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sums:2}
Sea $X \sim \mathcal B(100,\ 0.5)$. Acota $\P(X \geq 75)$ con la
desigualdad de Markov y después $\P(\abs{X - 50} \geq 25)$ con
Bienaymé--Chebyshev. Compara.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:sums:3}
Se lanza una moneda equilibrada $n$ veces y $F_n$ denota la frecuencia de
caras. ¿Cómo de grande tiene que ser $n$ para que, según la cota de
Bienaymé--Chebyshev,
$\P\bigl(\abs{F_n - 0.5} \geq 0.05\bigr) \leq 0.05$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:4}
Una inversora reparte su capital a partes iguales entre $n$ activos
\omterm{def:g12:sums:indep}{independientes}, cada uno con rentabilidad esperada $\mu = 5\,\%$ y
\omterm{def:g11:stat:variance}{desviación típica} $\sigma = 20\,\%$. Calcula la \omterm{def:g12:randvar:exp}{esperanza} y la \omterm{def:g11:stat:variance}{desviación
típica} de la rentabilidad $M_n$ de la cartera, y el número de activos
necesario para bajar la \omterm{def:g11:stat:variance}{desviación típica} por debajo del $4\,\%$. ¿Qué
principio financiero ilustra esto?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:5}
Sean $X$ e $Y$ \omterm{def:g12:sums:indep}{independientes}, las dos uniformes sobre $\{1, 2, 3\}$.
\begin{enumerate}
  \item Da la \omterm{def:g12:randvar:rv}{distribución} de $S = X + Y$ y calcula $\E(S)$ y $\V(S)$
        directamente a partir de ella.
  \item Recupera los dos valores por linealidad y aditividad de la
        \omterm{def:g12:randvar:exp}{varianza}.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:6}
Demuestra que la aditividad de la \omterm{def:g12:randvar:exp}{varianza} puede fallar sin \omterm{def:g12:condprob:indep}{independencia}:
calcula $\V(X + Y)$ para $Y = X$ y compáralo con $\V(X) + \V(Y)$. ¿Para qué
variables $X$ se cumple $\V(2X) = 2\V(X)$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:sums:7}
Se sospecha que un dado está trucado. Se lanza $1200$ veces y saca seis
$260$ veces ($f = 0.2167$ en lugar de $\frac16 \approx 0.1667$). Bajo la
hipótesis de que el dado es equilibrado, acota
$\P\bigl(\abs{F_n - \frac16} \geq 0.05\bigr)$ con Bienaymé--Chebyshev y
discute si la hipótesis de equilibrio es plausible.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:sums:8}
\emph{(Una desigualdad mejor para la moneda.)} Sean
$X \sim \mathcal B(n,\ p)$ y $F_n = \frac Xn$.
\begin{enumerate}
  \item Demuestra que $p(1-p) \leq \frac14$ para $p \in \intcc{0}{1}$.
  \item Deduce la cota, independiente de la \omterm{def:g12:randvar:rv}{distribución},
        $\P\bigl(\abs{F_n - p} \geq \varepsilon\bigr) \leq
        \frac{1}{4n\varepsilon^2}$.
  \item ¿A cuántas personas hay que encuestar para que la frecuencia
        observada quede a menos de $3$ puntos de la proporción verdadera
        con \omterm{def:g10:proba:distribution}{probabilidad} al menos $95\,\%$, según esta cota? (Las encuestas
        reales usan estimaciones más finas, pero el orden de magnitud es
        correcto.)
\end{enumerate}
\end{exercise}

\section{Problema: La banca siempre gana}

\begin{problem}[{Problema de fin de semana --- la ley de los grandes
números explica los casinos, las encuestas y los seguros, y de paso demuele
la falacia del jugador}]
\label{pb:g12:sums:1}
Un jugador de ruleta, después de $100$ tiradas, va \emph{ganando} casi
tantas veces como no. El casino, que gestiona un millón de tiradas, va
ganando con una certeza que ningún tribunal pondría en duda. El mismo
juego, la misma ventaja minúscula: la \omterm{def:g11:seq:arithmetic}{diferencia} es $\sqrt n$, y es el
asunto de este capítulo (\cref{prop:g12:sums:mean},
\cref{thm:g12:sums:chebyshev}, \cref{thm:g12:sums:lln}). Este problema
lleva la contabilidad del casino, dimensiona una encuesta electoral, pone
precio a la diversificación y desmonta la falacia más cara de la historia
del juego.

\medskip
\noindent\textbf{Parte I --- Soltura.}
\begin{enumerate}
  \item Dos dados \omterm{def:g12:sums:indep}{independientes}: calcula $\V(X + Y)$
        (\cref{thm:g12:sums:variance}).
  \item Lanza $100$ dados \omterm{def:g12:sums:indep}{independientes} y promedia los resultados: da la
        \omterm{def:g12:randvar:exp}{esperanza} y la \omterm{def:g11:stat:variance}{desviación típica} de la \omterm{prop:g12:sums:mean}{media muestral}
        (\cref{prop:g12:sums:mean}; para un solo dado,
        $\sigma = \sqrt{35/12} \approx 1.71$).
  \item Una \omterm{def:g12:randvar:rv}{variable aleatoria} no negativa tiene \omterm{def:g12:randvar:exp}{esperanza} $2$. ¿Qué dice
        la desigualdad de Markov (\cref{thm:g12:sums:markov}) sobre
        $\P(X \geq 10)$?
  \item Acota $\P\left(\abs{\bar X_{100} - 3.5} \geq 0.5\right)$ para la
        \omterm{def:g11:stat:mean}{media} de los dados de la pregunta 2 con Bienaymé--Chebyshev.
  \item Las \omterm{def:g12:randvar:exp}{esperanzas} se suman siempre; las \omterm{def:g12:randvar:exp}{varianzas}, solo con
        \omterm{def:g12:condprob:indep}{independencia}: exhibe $X$ e $Y$ dependientes (indicación:
        $Y = -X$) para los que $\V(X + Y) \neq \V(X) + \V(Y)$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La contabilidad de la banca.}
Ruleta europea, apuesta de $1$ euro al rojo: se gana $+1$ con \omterm{def:g10:proba:distribution}{probabilidad}
$\frac{18}{37}$ y se pierde $-1$ con \omterm{def:g10:proba:distribution}{probabilidad} $\frac{19}{37}$.
\begin{enumerate}[resume]
  \item Calcula la \omterm{def:g12:randvar:exp}{esperanza} y la \omterm{def:g11:stat:variance}{desviación típica} de la ganancia de una
        apuesta.
  \item Un jugador hace $100$ apuestas \omterm{def:g12:sums:indep}{independientes}; sea $G$ la ganancia
        total. Calcula $\E(G)$ y $\sigma(G)$, y después el cociente
        $\frac{\abs{\E(G)}}{\sigma(G)}$. ¿Qué significa una deriva de un
        cuarto de \omterm{def:g11:stat:variance}{desviación típica} para las posibilidades que tiene el
        jugador de terminar la noche ganando?
  \item El casino ve $1\,000\,000$ de apuestas. Calcula la \omterm{def:g12:randvar:exp}{esperanza} y la
        \omterm{def:g11:stat:variance}{desviación típica} de su recaudación total, y el mismo cociente.
        Interpreta el número $27$.
  \item Certifícalo con Chebyshev: acota la \omterm{def:g10:proba:distribution}{probabilidad} de que el casino
        \emph{pierda} dinero a lo largo del millón de apuestas.
  \item Sistemas de apuestas: doblar tras las pérdidas, retirarse cuando se
        va ganando\,\dots\ Usando la linealidad de la \omterm{def:g12:randvar:exp}{esperanza}
        (\cref{thm:g12:sums:linearity}) sobre la \omterm{def:g12:seq:sequence}{sucesión} (posiblemente
        aleatoria) de apuestas, explica por qué \emph{toda} estrategia en
        un juego con ventaja negativa tiene ganancia esperada negativa.
        ¿Qué violaría un sistema ganador?
  \item Enuncia qué promete la ley de los grandes números
        (\cref{thm:g12:sums:lln}) sobre la frecuencia del rojo, y qué
        \emph{no} promete sobre la siguiente tirada después de diez rojos
        seguidos. Nombra la falacia.
  \item El punto más sutil: la ley funciona por \emph{dilución}, no por
        compensación. Si los rojos acaban una noche $100$ por encima de lo
        esperado, ese excedente no se ``devuelve'' nunca: calcula qué le
        pasa en cambio a la \emph{fracción} $\frac{100}{n}$ cuando
        $n = 10^6$, y reescribe en una frase el error de la falacia del
        jugador.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Encuestas.}
\begin{enumerate}[resume]
  \item A partir del \cref{exo:g12:sums:8}: ¿a cuántas personas hay que
        encuestar para que la frecuencia observada caiga a menos de $3$
        puntos de la verdad con \omterm{def:g10:proba:distribution}{probabilidad} al menos $95\,\%$, según la
        cota de Chebyshev independiente de la \omterm{def:g12:randvar:rv}{distribución}?
  \item Los institutos de sondeos reales usan unas $1\,100$ personas para
        un margen de $\pm 3$ puntos al $95\,\%$: su fórmula es
        $n \approx \frac{1.96^2 \times p(1-p)}{\varepsilon^2}$ con
        $p(1-p) \leq \frac14$. Evalúala y explica la \omterm{def:g11:seq:arithmetic}{diferencia} con la
        pregunta 13 (¿qué es lo que Chebyshev no sabe sobre la forma de las
        fluctuaciones?).
  \item Para reducir a la mitad el margen de error, ¿cómo tiene que crecer
        la \omterm{def:g10:proba:sample}{muestra}? Partiendo de $\pm 3$ puntos con $1\,100$ personas, ¿qué
        \omterm{def:g10:proba:sample}{muestra} da $\pm 1.5$ puntos?
  \item El clásico contraintuitivo: el tamaño de \omterm{def:g10:proba:sample}{muestra} necesario no ha
        usado en ningún momento el tamaño de la población; $1\,100$
        personas bastan para una ciudad o para un continente. Señala el
        lugar del modelo donde el tamaño de la población está ausente y da
        la analogía culinaria que emplean los encuestadores.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Diversificación.}
\begin{enumerate}[resume]
  \item La aseguradora del \cref{pb:g12:randvar:1} gana $30n$ euros en
        \omterm{def:g12:randvar:exp}{esperanza} con $n$ pólizas, con una \omterm{def:g11:stat:variance}{desviación típica} de los
        siniestros de unos $315\sqrt n$. ¿A partir de qué $n$ supera por
        fin el beneficio esperado a una \omterm{def:g11:stat:variance}{desviación típica} de los
        siniestros? ¿Qué está haciendo el $\sqrt n$ por la aseguradora?
  \item La rentabilidad anual de una acción fluctúa con
        $\sigma = 20\,\%$. Reparte el dinero a partes iguales entre $25$
        acciones \emph{\omterm{def:g12:sums:indep}{independientes}} como esa: calcula la $\sigma$ de la
        cartera. En finanzas se dice que la diversificación es la única
        comida gratis; ¿en qué consiste exactamente esa comida?
  \item Supón ahora que las $25$ acciones están perfectamente
        correlacionadas (se mueven todas a la vez): ¿cuál es la $\sigma$ de
        la cartera? Compara los dos \omterm{def:g12:deriv:extremum}{extremos} y di qué hipótesis alquila en
        secreto todo \omterm{def:g12:complex:argument}{argumento} a favor de la diversificación, y qué pasó
        cuando falló en todo el sistema en 2008.
  \item Final: la sinfonía del $\sqrt n$: las sumas fluctúan como
        $\sqrt n$ mientras sus \omterm{def:g11:stat:mean}{medias} se asientan como
        $\frac{1}{\sqrt n}$; toca el tema a lo largo de los cuatro sectores
        de este problema (casino, encuestas, seguros y carteras), nombra a
        los dos infractores habituales (la falacia del jugador y la
        \omterm{def:g12:condprob:indep}{independencia} fingida) y da la mirada hacia delante: la
        \emph{forma} de las fluctuaciones, la campana, es la estrella
        \omterm{def:g12:limcont:continuity}{continua} del capítulo siguiente, y su teorema completo corona los
        volúmenes universitarios.
\end{enumerate}
\end{problem}
