\section{Sumas de variables aleatorias}

dados dos \omterm{def:g12:randvar:rv}{variables aleatorias} $X, Y $ en el mismo finito \omterm{def:g10:proba:events}{espacio muestral}
$\Omega $, la suma $ X + Y $ es el variable aleatoria
$\omega \mapsto X(\omega) + Y(\omega)$.

\begin{theorem}[Linealidad de esperanza]\label{thm:g12:sums:linearity}
\index{esperanza!linealidad}
A pesar de \omterm{def:g12:randvar:rv}{variables aleatorias} $ X, Y $ en $\Omega $ y $ a, b \in \R $:
\[
\E(X + Y) = \E(X) + \E(Y), \qquad \E(aX + b) = a\E(X) + b .
\]
\emph{No independencia Se necesita una suposición.}
\end{theorem}

\begin{proof}
Escribe la \omterm{def:g12:randvar:exp}{esperanza} como una suma de resultados: desde
$\P(X = x) = \sum_{\omega : X(\omega) = x} \P(\{\omega\})$, términos de agrupación
da $\E(X) = \sum_{\omega \in \Omega} \P(\{\omega\})\,X(\omega)$. Entonces
\[
\E(X + Y) = \sum_{\omega} \P(\{\omega\})\bigl(X(\omega) + Y(\omega)\bigr)
= \sum_{\omega} \P(\{\omega\})X(\omega)
+ \sum_{\omega} \P(\{\omega\})Y(\omega) = \E(X) + \E(Y). \qedhere
\]
\end{proof}

\begin{definition}[Variables aleatorias independientes]\label{def:g12:sums:indep}
$ X $ y $ Y $ son \emph{independiente}\index{independiente} si
para todos los valores $ x, y $:
\[
\P(X = x \text{ and } Y = y) = \P(X = x)\,\P(Y = y).
\]
Varias variables $ X_1, \dots, X_n $ son independientes si esta regla del producto
es válido para cada elección de valores de cada subfamilia.
\end{definition}

\begin{proposition}\label{prop:g12:sums:prodexp}
Si $ X $ y $ Y $ son \omterm{def:g12:sums:indep}{independientes}, $\E(XY) = \E(X)\,\E(Y)$.
\end{proposition}

\begin{proof}
\[
\begin{aligned}
\E(XY) &= \sum_{x, y} xy\;\P(X = x \text{ and } Y = y)
= \sum_{x, y} xy\,\P(X=x)\P(Y=y) \\
&= \Bigl(\sum_x x\P(X=x)\Bigr)\Bigl(\sum_y y\P(Y=y)\Bigr). \qedhere
\end{aligned}
\]
\end{proof}

\begin{theorem}[Variación de una suma]\label{thm:g12:sums:variance}
\index{varianza! de una suma}
Si $ X $ y $ Y $ son \emph{\omterm{def:g12:sums:indep}{independientes}}, entonces
\[
\V(X + Y) = \V(X) + \V(Y).
\]
De manera más general, para \omterm{def:g12:sums:indep}{independientes} $ X_1, \dots, X_n $:
$\V(X_1 + \dots + X_n) = \V(X_1) + \dots + \V(X_n)$.
\end{theorem}

\begin{proof}
Usando König--Huygens (\cref{prop:g12:randvar:konig}) y linealidad:
\begin{align*}
\V(X + Y) &= \E\bigl((X+Y)^2\bigr) - \bigl(\E X + \E Y\bigr)^2\\
&= \E(X^2) + 2\E(XY) + \E(Y^2)
- \E(X)^2 - 2\E(X)\E(Y) - \E(Y)^2\\
&= \V(X) + \V(Y) + 2\bigl(\E(XY) - \E(X)\E(Y)\bigr),
\end{align*}
y el último corchete desaparece para \omterm{def:g12:sums:indep}{independientes} variables
(\cref{prop:g12:sums:prodexp}). El caso general se sigue por inducción.
\end{proof}

\begin{example}[Binomio revisitado]\label{ex:g12:sums:binomial}
Una variable binomial $ X \sim \mathcal B(n, p)$ es una suma
$ X = X_1 + \dots + X_n $ de $ n $ \omterm{def:g12:sums:indep}{independientes} Variables de Bernoulli. Por lo tanto,
estructuralmente:
\[
\E(X) = np, \qquad \V(X) = np(1-p),
\]
recuperante \cref{thm:g12:randvar:binomial} sin ningún cálculo.
\end{example}

\begin{proposition}[media muestral]\label{prop:g12:sums:mean}
Sea $ X_1, \dots, X_n $ ser \omterm{def:g12:sums:indep}{independientes} con el mismo \omterm{def:g12:randvar:rv}{distribución} como $ X $
(\omterm{def:g12:randvar:exp}{esperanza} $\mu $, varianza $\sigma^2$), y dejar
$ M_n = \frac{X_1 + \dots + X_n}{n}$ ser su \emph{muestra
media}\index{muestra
media}. Entonces
\[
\E(M_n) = \mu, \qquad \V(M_n) = \frac{\sigma^2}{n}, \qquad
\sigma(M_n) = \frac{\sigma}{\sqrt n}.
\]
\end{proposition}

\begin{proof}
La linealidad da $\E(M_n) = \frac{n\mu}{n} = \mu $; independencia da
$\V(X_1 + \dots + X_n) = n\sigma^2$ y dividiendo por $n $ escala el
varianza por $\frac{1}{n^2}$ (\cref{prop:g12:randvar:affine}).
\end{proof}

El $\frac{\sigma}{\sqrt n}$ es lo fundamental \emph{ley de la \omterm{def:g11:quad:discriminant}{raíz} cuadrada}: a
reducir a la mitad las fluctuaciones de un promedio, cuadruplicar las \omterm{def:g10:proba:sample}{muestra} tamaño.

\section{Desigualdades de concentración}

\begin{theorem}[La desigualdad de Markov]\label{thm:g12:sums:markov}
\index{La desigualdad de Markov}
Si $ X \geq 0$ y $ a > 0$:
\[
\P(X \geq a) \leq \frac{\E(X)}{a}.
\]
\end{theorem}

\begin{proof}
En $\E(X) = \sum_i p_i x_i $ (todos los términos no negativos), conserve sólo los términos
con $ x_i \geq a $: cada uno es al menos $ a\,p_i $, entonces
$\E(X) \geq a \sum_{x_i \geq a} p_i = a\,\P(X \geq a)$.
\end{proof}

\begin{theorem}[Bienaymé--Chebyshev desigualdad]\label{thm:g12:sums:chebyshev}
\index{Bienaymé--Chebyshev desigualdad}
Para cualquier variable aleatoria $ X $ y cualquier $\varepsilon > 0$:
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
\leq \frac{\V(X)}{\varepsilon^2}.
\]
\end{theorem}

\begin{proof}
Aplicar la desigualdad de Markov a la variable no negativa.
$ Y = (X - \E(X))^2$ con $ a = \varepsilon^2$:
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
= \P(Y \geq \varepsilon^2)
\leq \frac{\E(Y)}{\varepsilon^2} = \frac{\V(X)}{\varepsilon^2}. \qedhere
\]
\end{proof}

\begin{theorem}[ley de los grandes números]\label{thm:g12:sums:lln}
\index{ley de los grandes números}
Sea $ X_1, \dots, X_n $ ser \omterm{def:g12:sums:indep}{independientes} con el mismo \omterm{def:g12:randvar:rv}{distribución}
(\omterm{def:g12:randvar:exp}{esperanza} $\mu $, varianza $\sigma^2$) y $ M_n $ su \omterm{def:g10:proba:sample}{muestra} de medios. Para
cada $\varepsilon > 0$:
\[
\P\bigl(\abs{M_n - \mu} \geq \varepsilon\bigr)
\leq \frac{\sigma^2}{n\,\varepsilon^2}
\xrightarrow[n \to +\infty]{} 0 .
\]
El \omterm{def:g10:proba:sample}{muestra} de medios se concentra alrededor del \omterm{def:g12:randvar:exp}{esperanza}.
\end{theorem}

\begin{proof}
Bienaymé--Chebyshev solicitó $ M_n $, cuyo \omterm{def:g12:randvar:exp}{esperanza} es $\mu $ y cuyo
varianza es $\frac{\sigma^2}{n}$ (\cref{prop:g12:sums:mean}).
\end{proof}

\begin{remark}
Este teorema es el puente entre \omterm{def:g10:proba:distribution}{probabilidad} teoría y estadística: la
\emph{\omterm{def:g10:stats:series}{frecuencia}} de un suceso sobre muchos \omterm{def:g12:sums:indep}{independientes} enfoques de repeticiones
es \emph{\omterm{def:g10:proba:distribution}{probabilidad}} (llevar $ X_i $ el indicador de la suceso, de modo que
$\mu = p $). Justifica estimar una \omterm{def:g10:proba:distribution}{probabilidad} por simulación
(\emph{Método Montecarlo}) y una proporción de población según una encuesta ---
cuantitativamente: ver \cref{ch:g12:contdist}.
\end{remark}

\begin{method}[Usando Bienaymé--Chebyshev]\label{met:g12:sums:chebyshev}
para garantizar $\P(\abs{M_n - \mu} \geq \varepsilon) \leq \alpha $, eso
basta con que $ n \geq \frac{\sigma^2}{\alpha\,\varepsilon^2}$. El límite es
crudo (las fluctuaciones reales suelen ser mucho menores) pero perfectamente general:
no requiere nada sobre el \omterm{def:g12:randvar:rv}{distribución} más allá de lo finito varianza.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g12:sums:1}
Se lanzan dos dados justos; dejar $ S $ sea la suma. Usando linealidad (no la
\omterm{def:g12:randvar:rv}{distribución} de $ S $!), calcular $\E(S)$; luego calcular $\V(S)$ usando

\end{exercise}

\begin{exercise}[$\star $]\label{exo:g12:sums:2}
Sea $ X \sim \mathcal B(100,\ 0.5)$. Atado

\end{exercise}

\begin{exercise}[$\star $]\label{exo:g12:sums:3}
Se lanza una moneda justa $ n $ tiempos y $ F_n $ denota el \omterm{def:g10:stats:series}{frecuencia} de cabezas.

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:sums:4}

\begin{exercise}[$\star\star $]\label{exo:g12:sums:5}

        cada diversificación \omterm{def:g12:complex:argument}{argumento} alquila en secreto ---
        y qué ocurrió cuando falló en todo el sistema en 2008.
  \item final --- el $\sqrt n $ sinfonía: las sumas fluctúan como
        $\sqrt n $ mientras sus medios se asientan como
        $\frac{1}{\sqrt n}$; reproducir el tema a través de los cuatro
        industrias de este problema (casino, encuestas, seguros,
        portafolios), nombre a los dos abusadores permanentes (los
        falacia del jugador y falso independencia), y dar la
        puntero hacia adelante: el \emph{forma} de las fluctuaciones
        --- la campana --- es el siguiente capitulo continua
        estrella, y su teorema completo corona la universidad
        volúmenes.
\end{enumerate}
\end{problem}
