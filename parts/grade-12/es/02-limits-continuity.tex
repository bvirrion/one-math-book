\chapter{Límites y continuidad}\label{ch:g12:limcont}

Este capítulo amplía la noción de límite desde sucesiones a funciones de un
variable real, e introduce continuidad. El resultado central es el
teorema del valor intermedio, que convierte la información cualitativa (``$f $ es
continua y cambia de signo'') en la existencia de soluciones de ecuaciones.

\section{Límites de funciones}

A lo largo de, $ f $ es una \omterm{def:g11:func:function}{función} definido en un \omterm{def:g10:numbers:interval}{intervalo} o un \omterm{def:g10:numbers:interunion}{unión} de \omterm{def:g10:numbers:interval}{intervalos}
$ D \subseteq \R $.

\begin{definition}[Límite al infinito]\label{def:g12:limcont:liminf}
Sea $f$ definida en un \omterm{def:g10:numbers:interval}{intervalo} $\intco{a}{+\infty}$ y $\ell \in \R $. nosotros
di eso $ f $ \emph{tiende a $\ell $ es $+\infty $}\index{tiende a $\ell $ en $+\infty $},
escrito $\lim\limits_{x\to+\infty} f(x) = \ell $, si cada abierto \omterm{def:g10:numbers:interval}{intervalo}
que contiene $\ell $ contiene todos los valores $ f(x)$ para $ x $ suficientemente grande:
por cada $\varepsilon > 0$ existe $ A $ tal que para todos $ x \geq A $,
$\abs{f(x) - \ell} \leq \varepsilon $.

decimos $ f $ tiende a $+\infty $ en $+\infty $ si por cada $ M \in \R $ allí
existe $ A $ tal que $ f(x) \geq M $ para todo $ x \geq A $. Límites en $-\infty $
y límites iguales a $-\infty $ se definen de manera análoga.
\end{definition}

\begin{definition}[Limitar en un punto]\label{def:g12:limcont:limpoint}
Sea $ a \in \R $ y deja $ f $ definirse cerca $ a $ (excepto posiblemente en $ a $). nosotros
di eso $ f $ tiende a $\ell \in \R $ en $ a $, escrito
$\lim\limits_{x \to a} f(x) = \ell $, si por cada $\varepsilon > 0$ allí
existe $\delta > 0$ tal que para todos $ x \in D $ con
$\abs{x - a} \leq \delta $, $\abs{f(x) - \ell} \leq \varepsilon $.

decimos $ f $ tiende a $+\infty $ en $ a $ si por cada $ M $ hay $\delta > 0$
tal que $ f(x) \geq M $ cuando sea $ x \in D $ y $\abs{x - a} \leq \delta $.
Límites unilaterales ($ x \to a^+$, $ x \to a^-$) restringir $ x $ a $ x > a $ o
$ x < a $.
\end{definition}

\begin{example}
$\lim\limits_{x\to+\infty} \frac{1}{x} = 0$,
$\lim\limits_{x\to 0^+} \frac{1}{x} = +\infty $,
$\lim\limits_{x\to 0^-} \frac{1}{x} = -\infty $. La \omterm{def:g11:func:function}{función}
$ x \mapsto \frac1x $ no tiene límite (bilateral) en $0$.
\end{example}

\begin{definition}[Asíntotas]\label{def:g12:limcont:asymptote}
la linea $ y = \ell $ es un \emph{asíntota horizontal}\index{asíntota horizontal} de la
curva de $ f $ en $+\infty $ (resp.\ $-\infty $) si
$\lim_{x\to+\infty} f(x) = \ell $ (resp.\ en $-\infty $). la linea $ x = a $ es un
\emph{asíntota vertical}\index{asíntota vertical} si $ f $ tiene un límite unilateral infinito en $ a $.
\end{definition}

\begin{proposition}[Operaciones sobre límites]\label{prop:g12:limcont:operations}
las reglas de \cref{prop:g12:seq:operations} (sumas, productos, cocientes) se mantienen
palabra por palabra para los límites de funciones, en un punto o en el infinito, con el mismo
cuatro formas indeterminadas $(+\infty)+(-\infty)$, $0\times\infty $,
$\frac{\infty}{\infty}$, $\frac{0}{0}$.
\end{proposition}

\begin{proof}
Las pruebas son idénticas a las \omterm{def:g12:seq:sequence}{sucesión} caso, reemplazando ``para $ n \geq N $''
por ``para $ x \geq A $'' o ``para $\abs{x-a} \leq \delta $''.
\end{proof}

\begin{theorem}[Límite de una composición]\label{thm:g12:limcont:composition}
Sea $ f, g $ ser funciones y $ a, b, c $ ser números reales o $\pm\infty $. Si
$\lim\limits_{x \to a} f(x) = b $ y $\lim\limits_{y \to b} g(y) = c $, entonces
\[
\lim_{x \to a} g\bigl(f(x)\bigr) = c .
\]
La misma afirmación es válida con un \omterm{def:g12:seq:sequence}{sucesión} en lugar de $ f $: si
$ u_n \to b $ y $\lim_{y\to b} g(y) = c $, entonces $ g(u_n) \to c $.
\end{theorem}

\begin{proof}
Llevar $ b, c $ finito (los otros casos son análogos). Sea $\varepsilon > 0$.
Hay $\delta > 0$ tal que $\abs{y - b} \leq \delta $ implica
$\abs{g(y) - c} \leq \varepsilon $; y hay $\delta' > 0$ tal que
$\abs{x - a} \leq \delta'$ implica $\abs{f(x) - b} \leq \delta $. Encadenando el
dos da $\abs{g(f(x)) - c} \leq \varepsilon $ para
$\abs{x - a} \leq \delta'$.
\end{proof}

\begin{theorem}[Comparación y compresión]\label{thm:g12:limcont:squeeze}
Sea $ f, g, h $ ser funciones y $ a \in \R \cup \{\pm\infty\}$.
\begin{enumerate}
  \item Si $ f \leq g $ cerca $ a $ y $ f(x) \to +\infty $ como $ x \to a $, entonces
        $ g(x) \to +\infty $.
  \item Si $ f \leq g \leq h $ cerca $ a $ y $ f, h $ tienden al mismo finito
        límite $\ell $ en $ a $, entonces $ g(x) \to \ell $ como $ x \to a $.
\end{enumerate}
\end{theorem}

\begin{proof}
Mismo argumento que para sucesiones (\cref{thm:g12:seq:squeeze}).
\end{proof}

\begin{method}[Límites informáticos en la práctica.]\label{met:g12:limcont:limits}
\begin{itemize}
  \item Polinomios y racionales funciones en $\pm\infty $: factorizar el
        mayor poder de $ x $; el límite es el de la relación de avance
        términos.
  \item $\frac{0}{0}$ en un punto $ a $: factor $(x-a)$ fuera del numerador y
        denominador, o reconocer un derivada
        $\lim_{x\to a}\frac{f(x)-f(a)}{x-a} = f'(a)$ (\cref{ch:g12:deriv}).
  \item Raíces cuadradas: multiplicar por la expresión conjugada.
  \item Factores oscilantes ($\cos $, $\sin $): utilice el teorema de compresión.
\end{itemize}
\end{method}

\section{Continuidad}

\begin{definition}[Continuidad]\label{def:g12:limcont:continuity}
Sea $f$ definida en un \omterm{def:g10:numbers:interval}{intervalo} $ I $ y $ a \in I $. La \omterm{def:g11:func:function}{función} $ f $ es
\emph{continuo en $ a $}\index{continuo en $ a $} si
$\lim\limits_{x \to a} f(x) = f(a)$. Es \emph{continuo encendido $ I $}\index{continuo encendido $ I $} si es
continuo en cada punto de $ I $.
\end{definition}

Intuitivamente, el grafo de un continua \omterm{def:g11:func:function}{función} en un \omterm{def:g10:numbers:interval}{intervalo} se puede dibujar
sin levantar el bolígrafo.

\begin{proposition}[Continuidad de funciones habituales.]\label{prop:g12:limcont:usual}
Polinomios racionales funciones, $ x \mapsto \sqrt{x}$, $ x \mapsto \abs{x}$,
$\cos $, $\sin $, $\exp $ y $\ln $ son continua en cada \omterm{def:g10:numbers:interval}{intervalo} contenido
en su \omterm{def:g11:func:function}{dominio}. Sumas, productos, cocientes (donde se definan) y composiciones
de continua funciones son continua.
\end{proposition}

\begin{proof}[Prueba parcial]
La estabilidad bajo sumas, productos, cocientes y composición se deriva de la
teoremas correspondientes sobre límites. continuidad de $ x \mapsto x $ y de constante
funciones es inmediato de la definición, por lo tanto polinomios y racionales
funciones son continua. continuidad del resto habitual funciones es
probados en los capítulos donde se construyen o admiten.
\end{proof}

\begin{example}
El \emph{piso \omterm{def:g11:func:function}{función}} $ x \mapsto \floor{x}$ (el mayor entero
$\leq x $) no es continua en cualquier entero $ n $: su límite izquierdo es
$ n - 1$ mientras $\floor{n} = n $.
\end{example}

\begin{proposition}[Sucesiones y continuidad]\label{prop:g12:limcont:seqcont}
Si $ u_n \to \ell $ y $ f $ es continua en $\ell $, entonces $ f(u_n) \to f(\ell)$.
En particular, si un \omterm{def:g12:seq:sequence}{sucesión} definido por $ u_{n+1} = f(u_n)$ \omterm{def:g12:seq:limit}{converger} a
$\ell $ y $ f $ es continua en $\ell $, entonces $ f(\ell) = \ell $.
\end{proposition}

\begin{proof}
El primer reclamo es \cref{thm:g12:limcont:composition} aplicado con el
\omterm{def:g12:seq:sequence}{sucesión} $(u_n)$. Para el segundo, pase al límite a ambos lados de
$ u_{n+1} = f(u_n)$: el lado izquierdo tiende a $\ell $, el lado derecho a
$ f(\ell)$y los límites son únicos.
\end{proof}

\section{El teorema del valor intermedio}

\begin{theorem}[Teorema del valor intermedio]\label{thm:g12:limcont:ivt}
\index{teorema del valor intermedio}
Sea $ f $ ser continua en $\intcc{a}{b}$ y deja $ k $ ser cualquier número entre
$ f(a)$ y $ f(b)$. Entonces existe $ c \in \intcc{a}{b}$ tal que
$ f(c) = k $.
\end{theorem}

{\small El teorema del valor intermedio: a continua \omterm{def:g10:numbers:interunion}{unión} de curvas

$ y = k $ tumbado entre $ f(a)$ y $ f(b)$.}
\end{omfigure}

\begin{proof}
Reemplazo $ f $ por $ f - k $ (y por $ k - f $ si es necesario), podemos suponer
$ f(a) \leq 0 \leq f(b)$ y busca un cero de $ f $. Construimos dos
sucesiones por \emph{dicotomía}\index{dicotomía}: colocar $ a_0 = a $, $ b_0 = b $;
dado $ a_n \leq b_n $ con $ f(a_n) \leq 0 \leq f(b_n)$, dejar
$ m = \frac{a_n + b_n}{2}$ y definir
\[
(a_{n+1}, b_{n+1}) =
\begin{cases}
(a_n, m) & \text{if } f(m) \geq 0,\\
(m, b_n) & \text{if } f(m) < 0.
\end{cases}
\]
En ambos casos $ f(a_{n+1}) \leq 0 \leq f(b_{n+1})$, la \omterm{def:g12:seq:sequence}{sucesión} $(a_n)$ es
\omterm{def:g12:seq:monotonic}{creciente}, $(b_n)$ es \omterm{def:g10:functions:variations}{decreciente}, y
$ b_n - a_n = \frac{b-a}{2^n} \to 0$: el sucesiones son adyacentes (ver
\cref{exo:g12:seq:10}) y convergen a un límite común
$ c \in \intcc{a}{b}$. Por continuidad, $ f(a_n) \to f(c)$ y
$ f(b_n) \to f(c)$; pasando al limite en $ f(a_n) \leq 0$ y
$ f(b_n) \geq 0$ da $ f(c) \leq 0 \leq f(c)$, entonces $ f(c) = 0$.
\end{proof}

\begin{remark}
El \omterm{thm:g12:limcont:ivt}{dicotomía} La prueba es un algoritmo: produce resultados arbitrariamente precisos.
aproximaciones de una solución, reduciendo a la mitad el error en cada paso. Así es como un
la computadora puede resolver $ f(x) = 0$ sabiendo sólo cómo evaluar $ f $.
\end{remark}

\begin{theorem}[Teorema de biyección]\label{thm:g12:limcont:bijection}
Sea $ f $ ser continua y \emph{estrictamente monótona} en un \omterm{def:g10:numbers:interval}{intervalo} $ I $. entonces
por cada $ k $ estrictamente entre los límites de $ f $ en los puntos finales de $ I $, el
\omterm{def:g10:algebra:equation}{ecuación} $ f(x) = k $ tiene un \emph{único} solución en $ I $.
\end{theorem}

\begin{proof}
Existencia: elegir $ a, b \in I $ con $ k $ entre $ f(a)$ y $ f(b)$ (posible
ya que los valores de $ f $ acercarse a sus límites de punto final), y aplicar el
teorema del valor intermedio en $\intcc{a}{b}$. Unicidad: si $ x < y $, estricto
la monotonicidad da $ f(x) \neq f(y)$, por lo que dos puntos distintos no pueden ser ambos
soluciones.
\end{proof}

\begin{method}[resolviendo $ f(x) = k $ con el teorema de biyección]\label{met:g12:limcont:bijection}
para probar que $ f(x) = k $ tiene exactamente una solución en un \omterm{def:g10:numbers:interval}{intervalo}:
\begin{enumerate}
  \item establecer un \omterm{def:g10:functions:table}{tabla de variaciones} de $ f $ (signo de $ f'$, ver
        \cref{ch:g12:deriv});
  \item en cada uno \omterm{def:g10:numbers:interval}{intervalo} dónde $ f $ es continua y estrictamente monótona,
        comparar $ k $ con los valores o límites del punto final;
  \item aplicar el teorema de biyección en cada uno de estos \omterm{def:g10:numbers:interval}{intervalo} y contar el
        soluciones.
\end{enumerate}
una calculadora o \omterm{thm:g12:limcont:ivt}{dicotomía} luego localiza cada solución tan finamente como se desee.
\end{method}

\begin{example}\label{ex:g12:limcont:cubic}
El \omterm{def:g10:algebra:equation}{ecuación} $ x^3 + x = 1$ tiene una solución real única. En efecto
$ f(x) = x^3 + x $ es continua y estrictamente \omterm{def:g12:seq:monotonic}{creciente} en $\R $ (como suma de
estrictamente funciones \omterm{def:g12:seq:monotonic}{crecientes}), $\lim_{-\infty} f = -\infty $ y
$\lim_{+\infty} f = +\infty $, por lo que el teorema de biyección se aplica con $ k = 1$.
Desde $ f(0.6) = 0.816 < 1$ y $ f(0.7) = 1.043 > 1$, la solución está en
$\intoo{0.6}{0.7}$.
\end{example}

\section{Problema: No se puede cruzar un río sin mojarse}

\begin{problem}[{Problema del fin de semana --- el valor intermedio
teorema fuerza puntos fijos, antípodas de igual temperatura y
tablas estables: tres teoremas que parecen imposibles a partir de una idea}]
\label{pb:g12:limcont:1}
Arrugar un mapa de Francia y dejarlo caer en cualquier lugar del suelo francés: algunos
punto del mapa se encuentra \emph{exactamente} encima del lugar donde
representa. En cada instante, dos puntos diametralmente opuestos
del ecuador tienen \emph{exactamente} la misma temperatura. y un
Una mesa cuadrada que se tambalea sobre un terreno lleno de baches siempre se puede estabilizar
\emph{giratorio} él. Tres trucos, un teorema: a continua
La cantidad no puede pasar de negativo a positivo sin desaparecer.
(\cref{thm:g12:limcont:ivt}). Este problema entrena límites, entonces
construye las tres maravillas --- y sabe exactamente cuál es el teorema
no promete.

\medskip
\noindent\textbf{Part I --- Limit fluency.}
\begin{enumerate}
  \item Calcular $\lim_{x \to +\infty} \dfrac{3x^2 - x}{x^2 + 5}$,
        \ $\lim_{x \to +\infty} \dfrac{2x + 1}{x^2 + 1}$, \ y
        $\lim_{x \to +\infty} (x^3 - x^2)$
        (\cref{met:g12:limcont:limits}).
  \item Dale el asíntotas de
        $ f(x) = \dfrac{3x - 1}{x + 2}$
        (\cref{def:g12:limcont:asymptote}), calculando el
        límites pertinentes.
  \item Calcular $\lim_{x \to 3} \dfrac{x^2 - 9}{x - 3}$
        (factor) y
        $\lim_{x \to 0} \dfrac{\sqrt{x + 1} - 1}{x}$
        (conjugado).
  \item Por el teorema de la compresión
        (\cref{thm:g12:limcont:squeeze}): calcular
        $\lim_{x \to 0} x \sin\frac1x $ y
        $\lim_{x \to +\infty} \frac{\sin x}{x}$.
  \item Por composición y términos dominantes:

\end{enumerate}
