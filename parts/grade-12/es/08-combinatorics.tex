\chapter{Combinatoria y conteo}\label{ch:g12:comb}

La combinatoria es el arte de contar sin enumerar. Sus dos principios
elementales (sumar los tamaños de alternativas disjuntas y multiplicar los
números de elecciones independientes) bastan para contar las variaciones,
las \omterm{def:g12:comb:permutation}{permutaciones} y los subconjuntos de un conjunto finito, y culminan en el
teorema del binomio.

\section{Los dos principios de conteo}

Escribimos $\abs{E}$ para el número de elementos (el \emph{cardinal}) de un
conjunto finito $E$.

\begin{proposition}[Principio de la suma]\label{prop:g12:comb:addition}
Si un conjunto finito $E$ se reparte en subconjuntos $A_1, \dots, A_k$
(disjuntos dos a dos y de \omterm{def:g10:numbers:interunion}{unión} $E$), entonces
\[
\abs{E} = \abs{A_1} + \abs{A_2} + \dots + \abs{A_k}.
\]
\end{proposition}

\begin{proposition}[Principio del producto]\label{prop:g12:comb:multiplication}
Si un objeto se construye mediante una \omterm{def:g12:seq:sequence}{sucesión} de $k$ elecciones, con
$n_1$ opciones para la primera y, \emph{sean cuales sean las elecciones
anteriores}, $n_i$ opciones para la $i$-ésima, entonces el número de objetos
construidos es $n_1 \times n_2 \times \dots \times n_k$.
\end{proposition}

\begin{proof}
Las dos afirmaciones se demuestran por inducción sobre $k$; el caso
$k = 2$ de la segunda equivale a contar por filas una tabla rectangular.
\end{proof}

\begin{example}
Un restaurante ofrece 4 entrantes, 6 platos principales y 3 postres:
$4 \times 6 \times 3 = 72$ menús distintos de tres platos.
\end{example}

\section{Tuplas, permutaciones, factoriales}

\begin{definition}[$k$-tuplas]\label{def:g12:comb:tuples}
Una \emph{$k$-tupla} de un conjunto $E$ es una lista ordenada
$(x_1, \dots, x_k)$ de elementos de $E$, con repeticiones permitidas. Una
$k$-tupla de elementos \emph{distintos} es una
\emph{variación}\index{variación} de $k$ elementos de $E$.
\end{definition}

\begin{proposition}\label{prop:g12:comb:tuples}
Sea $\abs E = n$. El número de $k$-tuplas de $E$ es $n^k$. El número de
variaciones de $k$ elementos de $E$ ($0 \leq k \leq n$) es
\[
n(n-1)(n-2)\cdots(n-k+1) = \frac{n!}{(n-k)!},
\]
donde $n! = 1 \times 2 \times \dots \times n$ (y $0! = 1$) es el
\emph{factorial}\index{factorial} de $n$.
\end{proposition}

\begin{proof}
Principio del producto: para una $k$-tupla hay $n$ opciones en cada uno de
los $k$ pasos; para una \omterm{def:g12:comb:tuples}{variación}, $n$ opciones para $x_1$, después $n - 1$
para $x_2$ (ya se ha usado un elemento), \dots, y $n - k + 1$ para $x_k$.
\end{proof}

\begin{definition}[Permutación]\label{def:g12:comb:permutation}
Una \emph{permutación}\index{permutación} de $E$ es una \omterm{def:g12:comb:tuples}{variación} de los $n$
elementos de $E$: una ordenación de $E$. Por la
\cref{prop:g12:comb:tuples} (caso $k = n$), el número de permutaciones de un
conjunto de $n$ elementos es $n!$.
\end{definition}

\begin{example}
Cinco corredores pueden terminar una carrera en $5! = 120$ órdenes
distintos. El número de podios posibles (los tres primeros puestos) es
$5 \times 4 \times 3 = 60$.
\end{example}

\section{Combinaciones y números combinatorios}

\begin{definition}[Combinaciones]\label{def:g12:comb:combination}
Una \emph{combinación}\index{combinación} de $k$ elementos de $E$ es un
subconjunto de $E$ con $k$ elementos (sin orden y sin repetición). Su número
se escribe $\dbinom{n}{k}$\index{número combinatorio}, y se lee ``$n$ sobre
$k$''.
\end{definition}

\begin{theorem}\label{thm:g12:comb:binomformula}
Para $0 \leq k \leq n$:
\[
\binom{n}{k} = \frac{n!}{k!\,(n-k)!} .
\]
\end{theorem}

\begin{proof}
Contamos de dos maneras las variaciones de $k$ elementos de $E$.
Directamente: $\frac{n!}{(n-k)!}$. De otro modo: elegimos primero el
subconjunto subyacente ($\binom nk$ maneras) y lo ordenamos después ($k!$
maneras); el principio del producto da $\binom{n}{k}\,k!$. Igualando,
$\binom nk = \frac{n!}{k!(n-k)!}$.
\end{proof}

\begin{proposition}[Identidades básicas]\label{prop:g12:comb:identities}
Para $0 \leq k \leq n$:
\[
\binom{n}{0} = \binom{n}{n} = 1, \qquad
\binom{n}{1} = n, \qquad
\binom{n}{k} = \binom{n}{n-k},
\]
y la \emph{regla de Pascal}\index{triángulo de Pascal}: para
$1 \leq k \leq n-1$,
\[
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}.
\]
\end{proposition}

\begin{proof}
La simetría $\binom nk = \binom{n}{n-k}$ se cumple porque tomar
complementarios empareja uno a uno los subconjuntos de $k$ elementos con los
de $n-k$ elementos. Para la \omterm{prop:g12:comb:identities}{regla de Pascal}, fijamos un elemento $a \in E$ y
clasificamos los subconjuntos de $k$ elementos en los que contienen a $a$
(que se obtienen añadiendo $a$ a un subconjunto de $k-1$ elementos de
$E \setminus \{a\}$, y hay $\binom{n-1}{k-1}$) y los que no lo contienen,
que son los subconjuntos de $k$ elementos de $E \setminus \{a\}$, en número
$\binom{n-1}{k}$. Concluimos con el principio de la suma.
\end{proof}

La \omterm{prop:g12:comb:identities}{regla de Pascal} genera los coeficientes fila a fila: es el
\emph{\omterm{prop:g11:binom:pascal}{triángulo de Pascal}}, en el que cada entrada es la suma de las dos que
tiene encima.

\begin{theorem}[Teorema del binomio]\label{thm:g12:comb:binomial}
\index{teorema del binomio}
Para todos $a, b \in \R$ (o $\C$) y todo $n \in \N$:
\[
(a+b)^n = \sum_{k=0}^{n} \binom{n}{k}\, a^{k}\, b^{\,n-k} .
\]
\end{theorem}

\begin{proof}
Desarrollamos el producto $(a+b)(a+b)\cdots(a+b)$ ($n$ factores): cada
término del desarrollo elige $a$ o $b$ en cada factor y produce
$a^k b^{n-k}$, donde $k$ es el número de factores que aportan $a$. El número
de maneras de elegir esos $k$ factores entre $n$ es $\binom nk$, que es, por
tanto, el coeficiente de $a^k b^{n-k}$.
\end{proof}

\begin{corollary}\label{cor:g12:comb:sums}
$\displaystyle\sum_{k=0}^{n} \binom{n}{k} = 2^n$
\quad y \quad
$\displaystyle\sum_{k=0}^{n} (-1)^k\binom{n}{k} = 0$ \ ($n \geq 1$).
\end{corollary}

\begin{proof}
Tomamos $a = b = 1$ y después $a = -1$, $b = 1$ en el teorema del binomio.
La primera identidad tiene además un significado directo: un conjunto de $n$
elementos tiene $2^n$ subconjuntos (cada elemento está dentro o fuera:
principio del producto), clasificados por tamaño.
\end{proof}

\begin{method}[Elegir el modelo adecuado]\label{met:g12:comb:model}
Antes de contar, responde a dos preguntas: \emph{¿importa el orden?} y
\emph{¿se permiten repeticiones?}
\begin{center}
\begin{tabular}{l|cc}
 & importa el orden & el orden da igual\\
\hline
con repeticiones & $n^k$ (tuplas) & (universidad)\\
sin repeticiones & $\frac{n!}{(n-k)!}$ (variaciones) & $\binom nk$ (subconjuntos)
\end{tabular}
\end{center}
Extraer bolas de una urna: \emph{con devolución y en orden} $\to$ tuplas;
\emph{sin devolución y en orden} $\to$ variaciones; \emph{un puñado de golpe}
$\to$ subconjuntos.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:comb:1}
Una matrícula consta de 2 letras (A--Z), después 3 cifras y después 2
letras. ¿Cuántas matrículas son posibles? ¿Cuántas no repiten ningún
carácter?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:2}
Calcula $\dbinom{8}{3}$ y $\dbinom{10}{8}$, y simplifica
$\dfrac{\binom{n}{2}}{\binom{n+1}{2}}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:3}
En una clase de 30 alumnos hay que elegir una comisión de 4 alumnos y,
dentro de la comisión, un presidente y un tesorero (una misma persona no
puede ocupar los dos cargos). ¿Cuántos resultados son posibles?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:4}
Desarrolla $(x + 2)^5$ y $(1 - x)^6$ con el teorema del binomio. ¿Cuál es el
coeficiente de $x^3$ en $(2x + 3)^7$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:5}
Una mano de póquer consta de 5 cartas de una baraja de 52.
\begin{enumerate}
  \item ¿Cuántas manos hay?
  \item ¿Cuántas manos contienen exactamente un as? ¿Y al menos un as?
  \item ¿Cuántas manos son ``full'' (tres cartas de un mismo valor y dos de
        otro)?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:6}
¿Cuántos anagramas (reordenaciones de las letras, con sentido o sin él)
tiene la palabra ROMA? ¿Y la palabra BANANA? (Indicación para BANANA:
coloca primero las tres aes.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:7}
Demuestra la identidad $k\dbinom{n}{k} = n\dbinom{n-1}{k-1}$
($1 \leq k \leq n$) de dos maneras: con la fórmula de los \omterm{prop:g12:comb:tuples}{factoriales} y
contando de dos formas las parejas (comisión de $k$ personas, su presidente)
elegidas entre $n$ personas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:8}
Un camino del plano va de $(0,0)$ a $(m, n)$ mediante pasos unitarios hacia
el este o hacia el norte. Demuestra que el número de esos caminos es
$\dbinom{m+n}{m}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:comb:9}
Demuestra la \emph{identidad de Vandermonde}: para $0 \leq k \leq m + n$,
\[
\binom{m+n}{k} = \sum_{j=0}^{k} \binom{m}{j}\binom{n}{k-j},
\]
contando los subconjuntos de $k$ elementos de un conjunto repartido en un
grupo de $m$ y otro de $n$. Deduce que
$\displaystyle\sum_{j=0}^{n}\binom{n}{j}^{\!2} = \binom{2n}{n}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:comb:10}
Usando el teorema del binomio, demuestra que para todo $n \geq 1$,
\[
\sum_{k=1}^{n} k \binom{n}{k} = n\,2^{n-1}.
\]
(Indicación: deriva $(1+x)^n$, o usa el \cref{exo:g12:comb:7}.)
\end{exercise}

\section{Problema: El arte de contar dos veces}

\begin{problem}[{Problema de fin de semana --- estrellas y barras, sombreros
desarreglados e identidades demostradas contando una misma cosa de dos
maneras}]
\label{pb:g12:comb:1}
El truco más profundo de la combinatoria es de una sencillez desarmante:
contar dos veces la misma colección por dos métodos distintos e igualar las
respuestas. Este problema practica los modelos del
\cref{met:g12:comb:model}, añade una técnica que el curso del capítulo no
necesitó (las \emph{estrellas y barras} del recuento de helados), cuenta
después con exactitud los famosos \emph{sombreros desarreglados} y encuentra
al número $\frac1\eu$ esperando al fondo del montón de sombreros: su tercera
aparición en este libro.

\medskip
\noindent\textbf{Parte I --- Elegir el modelo.}
\begin{enumerate}
  \item Cuenta las matrículas formadas por $2$ letras seguidas de $3$
        cifras; y después los anagramas de BANANA.
  \item De una baraja de $32$ cartas, cuenta las manos de $5$ cartas; y
        después las manos que contienen exactamente $2$ de los $4$ ases.
  \item Un robot camina de $(0,0)$ a $(4,3)$ usando solo pasos unitarios a
        la derecha o hacia arriba: ¿cuántos caminos hay? (Codifica cada
        camino como una palabra en D y A.)
  \item Desarrolla $(1 + x)^4$ con el teorema del binomio
        (\cref{thm:g12:comb:binomial}) y evalúa después en $x = 1$ y en
        $x = -1$: ¿qué dos identidades sobre los números $\binom nk$ caen
        solas?
  \item Demuestra por doble conteo que $k\binom nk = n\binom{n-1}{k-1}$
        (cuenta de dos maneras las comisiones con presidente) y deduce que
        $\sum_{k=0}^{n} k\binom nk = n\,2^{n-1}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Estrellas y barras.}
\begin{enumerate}[resume]
  \item Una heladería vende $4$ sabores y pides $10$ bolas (los sabores se
        pueden repetir y el orden en la tarrina da igual). Codifica un
        pedido como una fila de $10$ estrellas (las bolas) separadas por $3$
        barras (los cambios de sabor) y cuenta los pedidos.
  \item Cuenta las ternas de enteros no negativos con $x + y + z = 12$.
  \item Cuenta las ternas de enteros \emph{positivos} con $x + y + z = 12$
        (sustituye $x = 1 + x'$, etc.).
  \item ¿Cuántos monomios distintos aparecen en el desarrollo de
        $(a + b + c)^5$?
  \item Comprueba el método: cuenta con la fórmula los pedidos de $3$ bolas
        con $2$ sabores, enuméralos todos después y compara.
  \item Di exactamente en qué punto entró en la codificación que ``las bolas
        son idénticas'', y cuenta qué pasaría si las bolas se comieran en
        orden (posiciones distintas), con la lista de comprobación del
        \cref{met:g12:comb:model}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Los sombreros desarreglados.}
Un \emph{desarreglo} es un reparto de $n$ sombreros a sus $n$ dueños en el
que \emph{nadie} recibe el suyo; sea $D_n$ el número de desarreglos. (El
\cref{pb:g11:prob:1} mostró que, de \omterm{def:g11:stat:mean}{media}, un invitado recupera su propio
sombrero; ahora contamos con exactitud las fiestas totalmente aciagas.)
\begin{enumerate}[resume]
  \item Calcula $D_1$, $D_2$ y $D_3$ enumerando, y $D_4$ con paciencia (o
        con astucia).
  \item Justifica la recurrencia $D_n = (n - 1)\left(D_{n-1} +
        D_{n-2}\right)$: el invitado 1 recibe algún sombrero $k \neq 1$
        ($n - 1$ opciones); separa los casos según si el invitado $k$ recibe
        el sombrero 1 o no. Comprueba que reproduce $D_4$ y calcula $D_5$.
  \item Para $n = 3$, demuestra por inclusión-exclusión (restando los
        repartos que fijan al menos un sombrero y devolviendo los excesos)
        que
        $D_3 = 3!\left(1 - \frac{1}{1!} + \frac{1}{2!} -
        \frac{1}{3!}\right)$, y enuncia la fórmula general.
  \item Calcula $\frac{D_5}{5!}$ y compáralo con
        $\frac1\eu \approx 0.3679$: la \omterm{def:g10:proba:distribution}{probabilidad} de que una fiesta
        grande se desarregle por completo es $\frac1\eu$, la tercera
        aparición estelar de esta constante, tras la lotería y la
        secretaria del \cref{pb:g12:exp:1}. (¿Por qué? La fórmula de la
        pregunta 14 es el comienzo de una serie famosa para $\eu^{-1}$, que
        se cuenta en los volúmenes universitarios.)
  \item Amigo invisible entre $10$ amigos: los nombres se sacan al azar de
        manera uniforme. ¿Cuál es la \omterm{def:g10:proba:distribution}{probabilidad} de que el sorteo sea
        válido (nadie se saca a sí mismo) y cuántas repeticiones del sorteo
        debe esperar el grupo?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Contar dos veces, ganar dos veces.}
\begin{enumerate}[resume]
  \item El lema de los apretones de manos: en cualquier fiesta, sumar sobre
        los invitados el número de manos que estrechó cada uno cuenta cada
        apretón exactamente dos veces. Deduce que \emph{el número de
        invitados que estrecharon un número \omterm{def:g11:func:parity}{impar} de manos es siempre par},
        y comprueba que la afirmación tiene sentido en una fiesta de tres
        invitados.
  \item Demuestra la joya
        $1^3 + 2^3 + \dots + n^3 = (1 + 2 + \dots + n)^2$ por inducción y
        compruébala para $n = 3$. (La suma del pequeño Gauss, elevada al
        cuadrado, cuenta cubos.)
  \item La identidad de Vandermonde (\cref{exo:g12:comb:9}) mediante
        caminos: interpreta $\binom{2n}{n}$ como los caminos reticulares del
        tipo de la pregunta 3 de $(0,0)$ a $(n,n)$, corta cada camino en su
        cruce con la antidiagonal y explica cómo aparece
        $\sum_j \binom nj^2$.
  \item Final: los cuatro movimientos del que cuenta, una línea para cada
        uno con un ejemplo de este problema: multiplicar etapas y sumar
        casos; codificar con astucia (estrellas y barras, palabras de
        caminos); contar dos veces lo mismo (la comisión con presidente, los
        apretones de manos); y restar lo indeseado corrigiendo los excesos
        (los desarreglos). Y fíjate en dónde se pone a trabajar el conteo a
        continuación: en la \omterm{def:g10:proba:distribution}{probabilidad} y en los caminos del capítulo de
        matrices y grafos.
\end{enumerate}
\end{problem}
