\chapter{Probabilidad condicional e independencia}\label{ch:g12:condprob}

La \omterm{def:g10:proba:distribution}{probabilidad} cuantifica la incertidumbre; la \omterm{def:g12:condprob:cond}{probabilidad condicional}
cuantifica cómo la modifica la información. Enterarse de que ha ocurrido un
\omterm{def:g11:prob:model}{suceso} $B$ reconfigura las \omterm{def:g10:proba:distribution}{probabilidades} de todos los demás \omterm{def:g11:prob:model}{sucesos}: un
mecanismo formalizado por Bayes y maltratado a diario en los tribunales y
en los periódicos. Este capítulo establece las reglas del condicionamiento
y el significado exacto de la \omterm{def:g12:condprob:indep}{independencia}.

\section{Espacios de probabilidad (recordatorio)}

Un experimento con un número finito de resultados se modeliza mediante un
\emph{\omterm{def:g11:prob:model}{espacio muestral}} $\Omega$ (el conjunto de los resultados) y una
\omterm{def:g10:proba:distribution}{probabilidad} $\P$ que asigna a cada \omterm{def:g11:prob:model}{suceso} $A \subseteq \Omega$ un número
$\P(A) \in \intcc{0}{1}$, aditiva sobre uniones disjuntas y con
$\P(\Omega) = 1$. Recordemos las reglas básicas:
\[
\P(\bar A) = 1 - \P(A),
\qquad
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B).
\]
Cuando todos los resultados son igual de probables,
$\P(A) = \frac{\abs A}{\abs\Omega}$, y calcular \omterm{def:g10:proba:distribution}{probabilidades} se reduce a
las técnicas de recuento del \cref{ch:g12:comb}.

\section{Probabilidad condicional}

\begin{definition}[Probabilidad condicional]\label{def:g12:condprob:cond}
Sea $B$ un \omterm{def:g11:prob:model}{suceso} con $\P(B) > 0$. La \emph{probabilidad de $A$ condicionada
a $B$}\index{probabilidad condicional} es
\[
\pcond{B}{A} = \frac{\P(A \cap B)}{\P(B)} .
\]
\end{definition}

La aplicación $A \mapsto \pcond BA$ es a su vez una \omterm{def:g10:proba:distribution}{probabilidad} (valen
todas las reglas); representa el nuevo estado de conocimiento de quien se
ha enterado de que $B$ ha ocurrido.

\begin{proposition}[Regla del producto]\label{prop:g12:condprob:product}
Para \omterm{def:g11:prob:model}{sucesos} de \omterm{def:g10:proba:distribution}{probabilidad} no nula:
\[
\P(A \cap B) = \P(B)\,\pcond{B}{A} = \P(A)\,\pcond{A}{B},
\]
y, más en general,
$\P(A_1 \cap A_2 \cap A_3) = \P(A_1)\,\pcond{A_1}{A_2}\,
\pcond{A_1 \cap A_2}{A_3}$, etcétera.
\end{proposition}

\begin{proof}
Basta reordenar la definición; la fórmula en cadena se obtiene iterando.
\end{proof}

\begin{theorem}[Ley de la probabilidad total]\label{thm:g12:condprob:total}
\index{ley de la probabilidad total}
Sean $B_1, \dots, B_n$ una partición de $\Omega$ en \omterm{def:g11:prob:model}{sucesos} de \omterm{def:g10:proba:distribution}{probabilidad}
no nula. Para todo \omterm{def:g11:prob:model}{suceso} $A$:
\[
\P(A) = \sum_{i=1}^{n} \P(B_i)\,\pcond{B_i}{A}.
\]
\end{theorem}

\begin{proof}
Los conjuntos $A \cap B_i$ son disjuntos dos a dos y su \omterm{def:g10:numbers:interunion}{unión} es $A$, luego
$\P(A) = \sum_i \P(A \cap B_i) = \sum_i \P(B_i)\pcond{B_i}{A}$ por la regla
del producto.
\end{proof}

\begin{method}[Árboles de probabilidad]\label{met:g12:condprob:tree}
Un \omterm{met:g10:proba:tree}{diagrama de árbol} organiza las \omterm{def:g12:condprob:cond}{probabilidades condicionales}: cada rama
lleva la \omterm{def:g10:proba:distribution}{probabilidad} del \omterm{def:g11:prob:model}{suceso} siguiente \emph{condicionada} al camino
\omterm{def:g10:stats:quartiles}{recorrido} hasta ahí.
\begin{itemize}
  \item La \omterm{def:g10:proba:distribution}{probabilidad} de una hoja (un camino completo) es el
        \emph{producto} de las \omterm{def:g10:proba:distribution}{probabilidades} de sus ramas (regla del
        producto).
  \item La \omterm{def:g10:proba:distribution}{probabilidad} de un \omterm{def:g11:prob:model}{suceso} es la \emph{suma} de las
        \omterm{def:g10:proba:distribution}{probabilidades} de las hojas que lo realizan (\omterm{def:g10:proba:distribution}{probabilidad} total).
  \item Las \omterm{def:g10:proba:distribution}{probabilidades} de las ramas que salen de un mismo nodo suman
        $1$.
\end{itemize}
\end{method}

\begin{theorem}[Fórmula de Bayes]\label{thm:g12:condprob:bayes}
\index{fórmula de Bayes}
Sean $B_1, \dots, B_n$ una partición de $\Omega$ como antes y $A$ un \omterm{def:g11:prob:model}{suceso}
con $\P(A) > 0$. Entonces
\[
\pcond{A}{B_j} =
\frac{\P(B_j)\,\pcond{B_j}{A}}{\sum_{i=1}^{n} \P(B_i)\,\pcond{B_i}{A}} .
\]
\end{theorem}

\begin{proof}
$\pcond{A}{B_j} = \frac{\P(A \cap B_j)}{\P(A)}$; se desarrolla el numerador
con la regla del producto y el denominador con la \omterm{def:g10:proba:distribution}{probabilidad} total.
\end{proof}

\begin{example}[Prueba de cribado]\label{ex:g12:condprob:test}
Una enfermedad afecta al $1\,\%$ de una población. Una prueba la detecta
con \omterm{def:g10:proba:distribution}{probabilidad} $0.99$ (sensibilidad) y da un falso positivo con
\omterm{def:g10:proba:distribution}{probabilidad} $0.05$. Dada una prueba positiva, la \omterm{def:g10:proba:distribution}{probabilidad} de estar
realmente enfermo es
\[
\frac{0.01 \times 0.99}{0.01\times0.99 + 0.99\times0.05}
= \frac{0.0099}{0.0099 + 0.0495} = \frac{1}{6} \approx 0.17 .
\]
Pese a lo fiable que es la prueba, cinco de cada seis positivos son falsos,
y eso es porque la enfermedad es rara. Confundir $\pcond{A}{B}$ con
$\pcond{B}{A}$ es la \emph{falacia del fiscal}.
\end{example}

\section{Independencia}

\begin{definition}[Sucesos independientes]\label{def:g12:condprob:indep}
Dos \omterm{def:g11:prob:model}{sucesos} $A$ y $B$ son \emph{independientes}\index{independencia} si
\[
\P(A \cap B) = \P(A)\,\P(B) .
\]
Cuando $\P(B) > 0$, esto equivale a $\pcond{B}{A} = \P(A)$: saber que ha
ocurrido $B$ no cambia la \omterm{def:g10:proba:distribution}{probabilidad} de $A$.
\end{definition}

\begin{proposition}\label{prop:g12:condprob:indepcompl}
Si $A$ y $B$ son \omterm{def:g12:condprob:indep}{independientes}, también lo son $A$ y $\bar B$ (y $\bar A$
y $\bar B$).
\end{proposition}

\begin{proof}
$\P(A \cap \bar B) = \P(A) - \P(A \cap B) = \P(A) - \P(A)\P(B)
= \P(A)\bigl(1 - \P(B)\bigr) = \P(A)\,\P(\bar B)$.
\end{proof}

\begin{remark}
No confundas \emph{\omterm{def:g12:condprob:indep}{independientes}} ($\P(A\cap B) = \P(A)\P(B)$) con
\emph{\omterm{def:g10:proba:operations}{incompatibles}} ($A \cap B = \varnothing$). Dos \omterm{def:g10:proba:operations}{sucesos incompatibles}
de \omterm{def:g10:proba:distribution}{probabilidad} no nula nunca son \omterm{def:g12:condprob:indep}{independientes}: saber que ha ocurrido uno
garantiza que el otro no.
\end{remark}

\begin{definition}[Repeticiones independientes]\label{def:g12:condprob:repet}
Cuando un experimento se repite $n$ veces de modo que el resultado de cada
prueba no influye en los demás, la \omterm{def:g10:proba:distribution}{probabilidad} de una \omterm{def:g12:seq:sequence}{sucesión} concreta de
resultados es el producto de las \omterm{def:g10:proba:distribution}{probabilidades} individuales. Este es el
modelo que hay detrás de la \omterm{def:g11:prob:rv}{distribución} binomial
(\cref{ch:g12:randvar}).
\end{definition}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:condprob:1}
Se saca una carta de una baraja estándar de 52 cartas. Calcula la
\omterm{def:g10:proba:distribution}{probabilidad} de que sea un rey sabiendo que es una figura (jota, reina o
rey). ¿Son \omterm{def:g12:condprob:indep}{independientes} los \omterm{def:g11:prob:model}{sucesos} ``ser rey'' y ``ser de corazones''?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:condprob:2}
Una urna contiene 5 bolas rojas y 3 azules. Se sacan dos bolas
sucesivamente \emph{sin} reposición.
\begin{enumerate}
  \item Dibuja el \omterm{met:g10:proba:tree}{árbol} de \omterm{def:g10:proba:distribution}{probabilidad}.
  \item Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de que las dos sean rojas y la \omterm{def:g10:proba:distribution}{probabilidad}
        de que la segunda sea roja.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:condprob:3}
Se lanzan dos dados equilibrados. Sean $A$ = ``la suma es $7$'', $B$ = ``el
primer dado saca $3$'' y $C$ = ``la suma es $6$''. Determina si $A$ y $B$
son \omterm{def:g12:condprob:indep}{independientes} y, después, si lo son $B$ y $C$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:4}
Una fábrica tiene tres máquinas que producen, respectivamente, el
$50\,\%$, el $30\,\%$ y el $20\,\%$ de la producción total, con tasas de
piezas defectuosas del $1\,\%$, el $2\,\%$ y el $4\,\%$.
\begin{enumerate}
  \item ¿Qué proporción de la producción es defectuosa?
  \item Una pieza elegida al azar es defectuosa. ¿Cuál es la \omterm{def:g10:proba:distribution}{probabilidad}
        de que provenga de la tercera máquina?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:5}
En el \cref{ex:g12:condprob:test}, ¿con qué prevalencia $p$ de la
enfermedad (en lugar del $1\,\%$) significaría una prueba positiva al menos
un $90\,\%$ de \omterm{def:g10:proba:distribution}{probabilidad} de estar enfermo? Resuelve la inecuación y
coméntala.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:6}
Una moneda trucada saca cara con \omterm{def:g10:proba:distribution}{probabilidad} $p \in \intoo{0}{1}$. Se
lanza tres veces, y los lanzamientos son \omterm{def:g12:condprob:indep}{independientes}.
\begin{enumerate}
  \item Calcula la \omterm{def:g10:proba:distribution}{probabilidad} del \omterm{def:g11:prob:model}{suceso} $E$ = ``exactamente dos caras''.
  \item ¿Para qué $p$ es máxima $\P(E)$?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:condprob:7}
\emph{(Monty Hall.)} Un premio se esconde detrás de una de tres puertas.
Eliges una puerta; el presentador, que sabe dónde está el premio, abre una
de las dos puertas restantes y revela siempre una vacía (eligiendo al azar
si las dos están vacías), y te ofrece cambiar a la otra puerta cerrada.
Usando la fórmula de Bayes, calcula la \omterm{def:g10:proba:distribution}{probabilidad} de ganar si cambias y
la \omterm{def:g10:proba:distribution}{probabilidad} de ganar si no cambias.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:condprob:8}
Un canal de información transmite bits. El ruido invierte cada bit con
\omterm{def:g10:proba:distribution}{probabilidad} $\varepsilon = 0.1$, de manera independiente. Para proteger un
bit, se envía tres veces y se descodifica por mayoría.
\begin{enumerate}
  \item Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de que el bit descodificado sea erróneo.
  \item La palabra recibida es $101$. ¿Cuál es la \omterm{def:g10:proba:distribution}{probabilidad} de que el
        bit enviado fuera un $1$? (Supón que el $0$ y el $1$ se envían con
        la misma \omterm{def:g10:proba:distribution}{probabilidad}.)
\end{enumerate}
\end{exercise}

\section{Problema: El médico, el juez y el filtro de correo basura}

\begin{problem}[{Problema de fin de semana --- la fórmula de Bayes contra
la falacia de la tasa base: por qué una prueba del $99\,\%$ puede
significar un $2\,\%$, por qué una coincidencia de uno entre un millón no
condena a nadie y cómo piensa tu buzón}]
\label{pb:g12:condprob:1}
Una prueba que detecta el $99\,\%$ de los casos da positivo, y la
\omterm{def:g10:proba:distribution}{probabilidad} real de estar enfermo es inferior al $2\,\%$. Aparece un
perfil de ADN que coincide con una persona entre un millón, y el
sospechoso es probablemente inocente. Las dos afirmaciones son correctas,
las dos han sido mal juzgadas por médicos y por tribunales, y las dos se
resuelven con una línea de este capítulo: la fórmula de Bayes
(\cref{thm:g12:condprob:bayes}). Este problema la entrena con urnas y la
lleva después a la consulta, a la sala del tribunal y a la bandeja de
entrada.

\medskip
\noindent\textbf{Parte I --- Soltura.}
\begin{enumerate}
  \item Una moneda equilibrada elige la urna I ($2$ rojas, $1$ azul) o la
        urna II ($1$ roja, $3$ azules), y después se saca una bola. Calcula
        $\P(\text{roja})$ con la ley de la \omterm{def:g10:proba:distribution}{probabilidad} total
        (\cref{thm:g12:condprob:total}).
  \item La bola es roja: calcula
        $\P(\text{urna I} \mid \text{roja})$.
  \item Un lanzamiento de un dado: ¿son \omterm{def:g12:condprob:indep}{independientes} los \omterm{def:g11:prob:model}{sucesos} ``par''
        y ``$\leq 4$'' (\cref{def:g12:condprob:indep})? ¿Y ``par'' y
        ``$\leq 3$''?
  \item Dos detectores de humo \omterm{def:g12:condprob:indep}{independientes} se disparan, cada uno con
        \omterm{def:g10:proba:distribution}{probabilidad} $0.9$, en caso de incendio. Usando la
        \cref{prop:g12:condprob:indepcompl}, calcula la \omterm{def:g10:proba:distribution}{probabilidad} de que
        se dispare al menos uno.
  \item Dos cartas sin reposición: calcula $\P(\text{dos reyes})$ con la
        regla del producto (\cref{prop:g12:condprob:product}).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El médico.}
Una enfermedad afecta a $1$ persona de cada $1\,000$. La prueba detecta al
$99\,\%$ de los enfermos (sensibilidad), pero también salta con el
$5\,\%$ de los sanos (falsos positivos).
\begin{enumerate}[resume]
  \item Frecuencias naturales: imagina $100\,000$ personas. ¿Cuántas están
        enfermas y cuántas sanas? ¿Cuántos positivos produce cada grupo y
        cuántos positivos hay en total?
  \item A partir de esos recuentos, calcula
        $\P(\text{enfermo} \mid \text{positivo})$. Encaja el golpe.
  \item Vuelve a calcularlo con la fórmula de Bayes y comprueba que
        coincide.
  \item En las encuestas, la mayoría de los médicos responde ``en torno al
        $99\,\%$''. Nombra las dos \omterm{def:g12:condprob:cond}{probabilidades condicionales} que se
        están confundiendo y explica por qué la minúscula tasa base manda
        sobre la respuesta verdadera.
  \item Una segunda prueba, independiente, también da positivo.
        Actualiza: tomando el $1.9\,\%$ como nueva \omterm{def:g10:proba:distribution}{probabilidad} previa,
        calcula $\P(\text{enfermo} \mid \text{dos positivos})$. ¿Qué está
        haciendo la \emph{acumulación} de indicios?
  \item Calcula $\P(\text{enfermo} \mid \text{positivo})$ si la misma
        prueba se aplica solo a un grupo de riesgo con una prevalencia del
        $5\,\%$. Deduce la moraleja de salud pública sobre el cribado
        masivo de enfermedades raras.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El juez.}
\begin{enumerate}[resume]
  \item Un perfil de ADN hallado en la escena del crimen coincide con una
        persona entre un millón. El sospechoso, encontrado al buscar en una
        base de datos de los $10$ millones de habitantes de la ciudad,
        coincide. ¿Número esperado de coincidencias inocentes en la ciudad?
        Sabiendo solo lo de la coincidencia, ¿cuánto vale aproximadamente
        $\P(\text{inocente} \mid \text{coincidencia})$?
  \item Nombra la falacia (¿qué \omterm{def:g10:proba:distribution}{probabilidad} citó el fiscal y cuál
        necesitaba el tribunal?) y su paralelismo exacto con la pregunta 9.
  \item La trampa de la \omterm{def:g12:condprob:indep}{independencia}: dos desgracias raras en una misma
        familia se elevaron una vez al cuadrado hasta dar ``una entre $73$
        millones'', tratándolas como \omterm{def:g12:condprob:indep}{independientes}, y condenaron a una
        madre inocente (el caso Sally Clark). Si los \omterm{def:g11:prob:model}{sucesos} comparten
        factores de riesgo familiares de modo que el segundo, dado el
        primero, es $5$ veces más probable que la tasa base, recalcula el
        orden de magnitud y enuncia la lección sobre multiplicar
        \omterm{def:g10:proba:distribution}{probabilidades}.
  \item La defensa también puede hacer trampas: ``coinciden diez inocentes,
        así que la coincidencia no demuestra nada'' ignora todo lo demás.
        ¿Qué hace la fórmula de Bayes con \emph{varios} indicios y qué
        única frase debería conocer todo miembro de un jurado?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El filtro de correo basura.}
\begin{enumerate}[resume]
  \item El $40\,\%$ de tu correo es basura. La palabra ``lotería'' aparece
        en el $15\,\%$ del correo basura y en el $0.5\,\%$ del correo
        honrado. Un mensaje la contiene: calcula
        $\P(\text{basura} \mid \text{lotería})$.
  \item Repetición correctora de errores (continuación del
        \cref{exo:g12:condprob:8}): con \omterm{def:g10:proba:distribution}{probabilidad} de inversión
        $\varepsilon = 0.1$, ¿con qué \omterm{def:g10:proba:distribution}{probabilidad} falla la
        descodificación por mayoría de \emph{cinco} copias? Compárala con
        la del código de tres copias.
  \item Un bit atraviesa $4$ repetidores ruidosos \omterm{def:g12:condprob:indep}{independientes}, cada uno
        de los cuales lo invierte con \omterm{def:g10:proba:distribution}{probabilidad} $0.1$: llega correcto
        exactamente cuando el número de inversiones es par. Calcula esa
        \omterm{def:g10:proba:distribution}{probabilidad} de dos maneras: sumando los términos binomiales y con
        la elegante fórmula $\frac{1 + (1 - 2\varepsilon)^4}{2}$.
  \item Los filtros de correo basura de verdad multiplican los indicios de
        muchas palabras como si fueran \omterm{def:g12:condprob:indep}{independientes} (Bayes ``ingenuo'').
        ¿Por qué es falsa esa hipótesis para palabras como ``ganar'' y
        ``lotería'' y por qué funciona bien el filtro aun así? (Una o dos
        frases.)
  \item Final: Bayes como la aritmética de la creencia: \omterm{def:g10:proba:distribution}{probabilidad} previa
        por verosimilitud, se renormaliza y sale la \omterm{def:g10:proba:distribution}{probabilidad}
        posterior. Repasa los tres veredictos (consulta, tribunal y bandeja
        de entrada) en una frase cada uno, y enumera después las tres
        trampas que ha destapado este problema: las tasas base olvidadas,
        la \omterm{def:g12:condprob:indep}{independencia} fingida y los indicios leídos de uno en uno.
\end{enumerate}
\end{problem}
