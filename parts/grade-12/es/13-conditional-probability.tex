\section{Espacios de probabilidad (recordatorio)}

Un experimento con un número finito de resultados se modela mediante un \emph{\omterm{def:g11:prob:model}{muestra
espacio}} $\Omega $ (el conjunto de resultados) y un \omterm{def:g10:proba:distribution}{probabilidad} $\P $ asignando a
cada uno suceso $ A \subseteq \Omega $ un numero $\P(A) \in \intcc{0}{1}$, aditivo
sobre disjunto sindicatos y con $\P(\Omega) = 1$. Recuerde las reglas básicas:
\[
\P(\bar A) = 1 - \P(A),
\qquad
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B).
\]
Cuando todos los resultados son igualmente probables,
$\P(A) = \frac{\abs A}{\abs\Omega}$ --- y las \omterm{def:g10:proba:distribution}{probabilidades} de cálculo reducen
a las técnicas de conteo de \cref{ch:g12:comb}.

\section{Probabilidad condicional}

\begin{definition}[Probabilidad condicional]\label{def:g12:condprob:cond}
Sea $ B $ frijol suceso con $\P(B) > 0$. El \emph{probabilidad de $ A $ friso
$ B $}\index{probabilidad de $ A $ dado
$ B $} es
\[
\pcond{B}{A} = \frac{\P(A \cap B)}{\P(B)} .
\]
\end{definition}

el mapa $ A \mapsto \pcond BA $ es en si mismo un \omterm{def:g10:proba:distribution}{probabilidad} (se aplican todas las reglas); eso
representa el nuevo estado de conocimiento de alguien que ha aprendido que $ B $
ocurrió.

\begin{proposition}[regla de multiplicacion]\label{prop:g12:condprob:product}
Para sucesos con \omterm{def:g10:proba:distribution}{probabilidades} distintas de cero:
\[
\P(A \cap B) = \P(B)\,\pcond{B}{A} = \P(A)\,\pcond{A}{B},
\]
y más en general
$\P(A_1 \cap A_2 \cap A_3) = \P(A_1)\,\pcond{A_1}{A_2}\,
\pcond{A_1 \cap A_2}{A_3}$, etc.
\end{proposition}

\begin{proof}
Reorganizar la definición; la fórmula de la cadena sigue iterando.
\end{proof}

\begin{theorem}[Ley de probabilidad total]\label{thm:g12:condprob:total}
\index{ley de probabilidad total}
Sea $ B_1, \dots, B_n $ ser una partición de $\Omega $ en sucesos de distinto de cero
\omterm{def:g10:proba:distribution}{probabilidad}. por cada suceso $ A $:
\[
\P(A) = \sum_{i=1}^{n} \P(B_i)\,\pcond{B_i}{A}.
\]
\end{theorem}

\begin{proof}
los conjuntos $ A \cap B_i $ son pares disjuntos con \omterm{def:g10:numbers:interunion}{unión} $ A $, entonces
$\P(A) = \sum_i \P(A \cap B_i) = \sum_i \P(B_i)\pcond{B_i}{A}$ por el
regla de multiplicación.
\end{proof}

\begin{method}[árboles de probabilidad]\label{met:g12:condprob:tree}
A diagrama de arbol organiza \omterm{def:g10:proba:distribution}{probabilidades} condicionales: cada rama lleva la
\omterm{def:g10:proba:distribution}{probabilidad} del próximo suceso \emph{dado} el camino hasta ahora.
\begin{itemize}
  \item El \omterm{def:g10:proba:distribution}{probabilidad} de una hoja (un camino completo) es el \emph{producto} de
        las \omterm{def:g10:proba:distribution}{probabilidades} a lo largo de sus ramas (regla de multiplicación).
  \item El \omterm{def:g10:proba:distribution}{probabilidad} de un suceso es el \emph{suma} de las \omterm{def:g10:proba:distribution}{probabilidades}
        de las hojas al darse cuenta (total \omterm{def:g10:proba:distribution}{probabilidad}).
  \item Las \omterm{def:g10:proba:distribution}{probabilidades} de que las ramas salgan de un nodo suman $1$.
\end{itemize}
\end{method}

\begin{theorem}[fórmula de bayes]\label{thm:g12:condprob:bayes}
\index{fórmula de bayes}
Sea $ B_1, \dots, B_n $ ser una partición de $\Omega $ como arriba y $ A $ un suceso
con $\P(A) > 0$. Entonces
\[
\pcond{A}{B_j} =
\frac{\P(B_j)\,\pcond{B_j}{A}}{\sum_{i=1}^{n} \P(B_i)\,\pcond{B_i}{A}} .
\]
\end{theorem}

\begin{proof}
$\pcond{A}{B_j} = \frac{\P(A \cap B_j)}{\P(A)}$; ampliar el numerador en
la regla de la multiplicación y el denominador por el total \omterm{def:g10:proba:distribution}{probabilidad}.
\end{proof}

\begin{example}[Prueba de detección]\label{ex:g12:condprob:test}
Una enfermedad afecta $1\%$ de una población. Una prueba lo detecta con \omterm{def:g10:proba:distribution}{probabilidad}
$0.99$ (sensibilidad) y da un falso positivo con \omterm{def:g10:proba:distribution}{probabilidad} $0.05$.
Ante una prueba positiva, el \omterm{def:g10:proba:distribution}{probabilidad} de tener realmente la enfermedad es
\[
\frac{0.01 \times 0.99}{0.01\times0.99 + 0.99\times0.05}
= \frac{0.0099}{0.0099 + 0.0495} = \frac{1}{6} \approx 0.17 .
\]
A pesar de la precisión de la prueba, cinco de seis positivos son falsos --- porque
la enfermedad es rara. confuso $\pcond{A}{B}$ con $\pcond{B}{A}$ es el
\emph{falacia del fiscal}.
\end{example}

\section{Independencia}

\begin{definition}[Eventos independientes]\label{def:g12:condprob:indep}
Dos sucesos $ A $ y $ B $ son \emph{independiente}\index{independiente} si
\[
\P(A \cap B) = \P(A)\,\P(B) .
\]
Cuando $\P(B) > 0$, esto equivale a $\pcond{B}{A} = \P(A)$: sabiendo $ B $
no cambia el \omterm{def:g10:proba:distribution}{probabilidad} de $ A $.
\end{definition}

\begin{proposition}\label{prop:g12:condprob:indepcompl}
Si $ A $ y $ B $ son \omterm{def:g12:condprob:indep}{independientes}, también lo son $ A $ y $\bar B $ (y $\bar A $ y
$\bar B $).
\end{proposition}

\begin{proof}
$\P(A \cap \bar B) = \P(A) - \P(A \cap B) = \P(A) - \P(A)\P(B)
= \P(A)\bigl(1 - \P(B)\bigr) = \P(A)\,\P(\bar B)$.
\end{proof}

\begin{remark}
no confundas \emph{\omterm{def:g12:condprob:indep}{independientes}} ($\P(A\cap B) = \P(A)\P(B)$) con
\emph{\omterm{def:g10:proba:operations}{incompatibles}} ($ A \cap B = \varnothing $). Dos \omterm{def:g10:proba:operations}{incompatibles} sucesos de
distinto de cero \omterm{def:g10:proba:distribution}{probabilidad} nunca son \omterm{def:g12:condprob:indep}{independientes}: saber que ocurrió uno garantiza
el otro no.
\end{remark}

\begin{definition}[repeticiones independientes]\label{def:g12:condprob:repet}
Cuando se repite un experimento $ n $ veces para que el resultado de cada prueba
no influye en los demás, el \omterm{def:g10:proba:distribution}{probabilidad} de un especificado \omterm{def:g12:seq:sequence}{sucesión} de
Los resultados son el producto de las \omterm{def:g10:proba:distribution}{probabilidades} individuales. este es el modelo
subyacente al binomio \omterm{def:g11:prob:rv}{distribución} (\cref{ch:g12:randvar}).
\end{definition}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g12:condprob:1}

\end{exercise}

\begin{exercise}[$\star $]\label{exo:g12:condprob:2}
Una urna contiene 5 bolas rojas y 3 azules. Se extraen dos bolas seguidas
\emph{sin} reemplazo.
\begin{enumerate}
  \item Dibuja el \omterm{def:g10:proba:distribution}{probabilidad} \omterm{met:g10:proba:tree}{árbol}.
  \item Calcular el \omterm{def:g10:proba:distribution}{probabilidad} que ambos son rojos, y el \omterm{def:g10:proba:distribution}{probabilidad} eso
        el segundo es rojo.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g12:condprob:3}
Se lanzan dos dados justos. Sea $ A $ = ``la suma es $7$'', $ B $ = ``el primero
morir \omterm{def:g10:proba:sample}{muestra} $3$'', $ C $ = ``la suma es $6$''. Determinar si $ A $ y $ B $
son \omterm{def:g12:condprob:indep}{independientes}, entonces si $ B $ y $ C $ son.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:condprob:4}
Una fábrica tiene tres máquinas que producen respectivamente. $50\%$, $30\%$ y
$20\%$ de la producción total, con tasas de defectos $1\%$, $2\%$ y $4\%$.
\begin{enumerate}
  \item ¿Qué proporción de la producción es defectuosa?
  \item Un artículo elegido al azar es defectuoso. cual es el \omterm{def:g10:proba:distribution}{probabilidad} eso
        vino de la tercera máquina?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:condprob:5}
En \cref{ex:g12:condprob:test}, ¿a qué prevalencia de la enfermedad $ p $ (en lugar de
$1\%$) sería una prueba positiva \omterm{def:g11:stat:mean}{media} al menos un $90\%$ posibilidades de enfermedad? resolver
la desigualdad y comentar.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g12:condprob:6}
Una moneda sesgada sale cara con \omterm{def:g10:proba:distribution}{probabilidad} $ p \in \intoo{0}{1}$. es
lanzado tres veces, siendo los lanzamientos \omterm{def:g12:condprob:indep}{independientes}.
\begin{enumerate}
  \item Calcular el \omterm{def:g10:proba:distribution}{probabilidad} del suceso $ E $ = ``exactamente dos cabezas''.
  \item para cual $ p $ es $\P(E)$ ¿\omterm{def:g10:functions:extrema}{máximo}?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g12:condprob:7}
\emph{(Monty Hall.)} Un premio se esconde detrás de una de las tres puertas. tu eliges un
puerta; el anfitrión, que sabe dónde está el premio, abre uno de los dos
puertas restantes, siempre revelando una vacía (eligiendo al azar si ambas
están vacías) y le ofrece pasar a la otra puerta cerrada. Usando
Fórmula de Bayes, calcule la \omterm{def:g10:proba:distribution}{probabilidad} de ganar si cambias, y si
no lo haces.
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g12:condprob:8}
Un canal de información transmite bits. Cada bit es invertido por el ruido con
\omterm{def:g10:proba:distribution}{probabilidad} $\varepsilon = 0.1$, de forma \omterm{def:g12:condprob:indep}{independiente}. Para proteger un poco, se envía.
tres veces y decodificado por mayoría.
\begin{enumerate}
  \item Calcular el \omterm{def:g10:proba:distribution}{probabilidad} que el bit decodificado es incorrecto.
  \item La palabra recibida es $101$. cual es el \omterm{def:g10:proba:distribution}{probabilidad} que el enviado
        poco fue $1$? (Asumir $0$ y $1$ se envían con igual \omterm{def:g10:proba:distribution}{probabilidad}.)
\end{enumerate}
\end{exercise}
