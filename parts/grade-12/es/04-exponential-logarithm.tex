\chapter{Exponencial y logaritmo}\label{ch:g12:exp}

La \omterm{thm:g12:exp:existence}{función exponencial} es la única \omterm{def:g11:func:function}{función} igual a su propia \omterm{def:g12:deriv:derivative}{derivada} y que
vale $1$ en $0$. Convierte sumas en productos; su inversa, el \omterm{def:g12:exp:ln}{logaritmo
neperiano}, convierte productos en sumas. Juntas describen todos los
fenómenos cuya tasa de variación es proporcional a su tamaño: la
desintegración radiactiva, el crecimiento de poblaciones, el interés
compuesto.

\section{La función exponencial}

\begin{theorem}[Existencia y unicidad]\label{thm:g12:exp:existence}
Existe una única \omterm{def:g11:func:function}{función} \omterm{def:g12:deriv:derivative}{derivable} $f \colon \R \to \R$ tal que
\[
f' = f \qquad\text{y}\qquad f(0) = 1 .
\]
Se llama \emph{función exponencial}\index{exponencial} y se escribe
$\exp$, o bien $x \mapsto \eu^x$.
\end{theorem}

\begin{proof}[Demostración de la unicidad]
En primer lugar, una \omterm{def:g11:func:function}{función} así no se anula nunca. En efecto, sea
$g(x) = f(x)f(-x)$; entonces
\[
g'(x) = f'(x)f(-x) - f(x)f'(-x) = f(x)f(-x) - f(x)f(-x) = 0,
\]
así que $g$ es constante e igual a $g(0) = 1$: para todo $x$,
$f(x)f(-x) = 1$ y, en particular, $f(x) \neq 0$.

Sean ahora $f_1, f_2$ dos soluciones y $h = \dfrac{f_1}{f_2}$ (legítimo,
porque $f_2$ no se anula nunca). Entonces
\[
h' = \frac{f_1' f_2 - f_1 f_2'}{f_2^2} = \frac{f_1 f_2 - f_1 f_2}{f_2^2} = 0,
\]
así que $h$ es constante e igual a $h(0) = 1$, es decir, $f_1 = f_2$.

La \emph{existencia} se admite a este nivel (se puede obtener mediante la
\omterm{def:g12:seq:sequence}{sucesión} del \cref{exo:g12:seq:10}, o como inversa del \omterm{def:g12:exp:ln}{logaritmo}
construido por integración en el \cref{ch:g12:integ}).
\end{proof}

\begin{proposition}[Ecuación funcional]\label{prop:g12:exp:funceq}
Para todos $x, y \in \R$ y $n \in \Z$:
\[
\eu^{x+y} = \eu^x\,\eu^y, \qquad
\eu^{-x} = \frac{1}{\eu^x}, \qquad
\eu^{x-y} = \frac{\eu^x}{\eu^y}, \qquad
\bigl(\eu^{x}\bigr)^n = \eu^{nx}.
\]
Además, $\eu^x > 0$ para todo $x$.
\end{proposition}

\begin{proof}
Fijemos $y$ y consideremos
$\varphi(x) = \dfrac{\exp(x+y)}{\exp(x)\exp(y)}$. Su numerador y su
denominador, como funciones de $x$, son los dos soluciones de $f' = f$
salvo la constante $\exp(y)$; derivando $\varphi$ directamente (regla del
cociente) se obtiene $\varphi' = 0$, luego
$\varphi \equiv \varphi(0) = 1$, lo que demuestra la primera identidad.
Tomando $y = -x$ se obtiene la segunda (con el valor $\eu^0 = 1$), y de ahí
se sigue la tercera. La cuarta es una inducción a partir de la primera para
$n \geq 0$, extendida a $n < 0$ mediante la segunda.

Positividad: $\eu^x = \left(\eu^{x/2}\right)^2 \geq 0$ y $\eu^x \neq 0$
(visto en el \cref{thm:g12:exp:existence}), luego $\eu^x > 0$.
\end{proof}

\begin{proposition}[Variación y límites]\label{prop:g12:exp:variations}
La exponencial es estrictamente \omterm{def:g12:seq:monotonic}{creciente} y \omterm{def:g12:deriv:convex}{convexa}, y
\[
\lim_{x\to-\infty} \eu^x = 0, \qquad
\lim_{x\to+\infty} \eu^x = +\infty .
\]
\end{proposition}

\begin{proof}
$(\exp)' = \exp > 0$ da el crecimiento estricto; $(\exp)'' = \exp > 0$ da
la \omterm{def:g12:deriv:convex}{convexidad}. Por la \omterm{def:g12:deriv:convex}{convexidad}, $\eu^x \geq 1 + x$ (la \omterm{def:g12:deriv:tangent}{tangente} en $0$,
véase el \cref{ex:g12:deriv:ineq}), así que $\eu^x \to +\infty$ cuando
$x \to +\infty$ por comparación. Y entonces
$\eu^{x} = 1/\eu^{-x} \to 0$ cuando $x \to -\infty$.
\end{proof}

\begin{theorem}[Comparación de crecimientos]\label{thm:g12:exp:growth}
\index{comparación de crecimientos}
Para todo entero $n \geq 1$,
\[
\lim_{x\to+\infty} \frac{\eu^x}{x^n} = +\infty,
\qquad
\lim_{x\to-\infty} x^n \eu^x = 0 .
\]
En palabras: la exponencial gana a todas las potencias de $x$.
\end{theorem}

\begin{proof}
Para $n = 1$: aplicando $\eu^t \geq 1 + t \geq t$ en $t = x/2$,
\[
\frac{\eu^x}{x} = \frac{\left(\eu^{x/2}\right)^2}{x}
\geq \frac{(x/2)^2}{x} = \frac{x}{4} \xrightarrow[x\to+\infty]{} +\infty .
\]
Para $n$ general, escribimos
\[
\frac{\eu^x}{x^n}
= \frac{1}{n^n}\left(\frac{\eu^{x/n}}{x/n}\right)^{n}
\xrightarrow[x\to+\infty]{} +\infty
\]
por el caso $n=1$ y la composición. El límite en $-\infty$ se sigue con la
sustitución $x \mapsto -x$:
$\abs{x^n \eu^x} = \frac{\abs{x}^n}{\eu^{\abs x}} \to 0$.
\end{proof}

\section{El logaritmo neperiano}

\begin{definition}[Logaritmo neperiano]\label{def:g12:exp:ln}
La exponencial es \omterm{def:g12:limcont:continuity}{continua} y estrictamente \omterm{def:g12:seq:monotonic}{creciente} de $\R$ sobre
$\intoo{0}{+\infty}$; por el teorema de la biyección
(\cref{thm:g12:limcont:bijection}), para todo $y > 0$ la \omterm{def:g10:algebra:equation}{ecuación}
$\eu^x = y$ tiene una única solución. Esa solución es el \emph{logaritmo
neperiano}\index{logaritmo} de $y$, que se escribe $\ln y$. Así pues,
\[
\text{para } x \in \R,\ y > 0: \qquad y = \eu^x \iff x = \ln y .
\]
En particular, $\ln 1 = 0$, $\ln \eu = 1$, $\eu^{\ln y} = y$ y
$\ln(\eu^x) = x$.
\end{definition}

\begin{proposition}[Propiedades algebraicas]\label{prop:g12:exp:lnalg}
Para todos $a, b > 0$ y $n \in \Z$:
\[
\ln(ab) = \ln a + \ln b, \quad
\ln\frac{1}{a} = -\ln a, \quad
\ln\frac{a}{b} = \ln a - \ln b, \quad
\ln(a^n) = n \ln a, \quad
\ln\sqrt{a} = \tfrac12 \ln a .
\]
\end{proposition}

\begin{proof}
$\eu^{\ln a + \ln b} = \eu^{\ln a}\eu^{\ln b} = ab$, y tomando $\ln$ en los
dos miembros se obtiene la primera identidad. Las demás se siguen por el
mismo mecanismo de las identidades correspondientes de la
\cref{prop:g12:exp:funceq}.
\end{proof}

\begin{proposition}[Propiedades analíticas]\label{prop:g12:exp:lnanalytic}
La \omterm{def:g11:func:function}{función} $\ln$ es \omterm{def:g12:deriv:derivative}{derivable} en $\intoo{0}{+\infty}$ con
\[
(\ln x)' = \frac{1}{x},
\]
es estrictamente \omterm{def:g12:seq:monotonic}{creciente} y \omterm{def:g12:deriv:convex}{cóncava}, y
$\lim\limits_{x\to0^+} \ln x = -\infty$,
$\lim\limits_{x\to+\infty} \ln x = +\infty$. Además, para todo entero
$n \geq 1$,
\[
\lim_{x\to+\infty} \frac{\ln x}{x^{1/n}} = 0, \qquad
\lim_{x\to0^+} x \ln x = 0 .
\]
\end{proposition}

\begin{proof}
La derivabilidad de la \omterm{def:g11:func:function}{función} inversa se admite a este nivel; admitiéndola,
derivamos la identidad $\eu^{\ln x} = x$ con la regla de la cadena:
$(\ln x)'\,\eu^{\ln x} = 1$, luego $(\ln x)' = \frac{1}{x} > 0$, de donde el
crecimiento estricto, y $(\ln x)'' = -\frac1{x^2} < 0$, de donde la
concavidad. Los límites en $0^+$ y en $+\infty$ reflejan los de $\exp$ a
través de la biyección. Para la \omterm{thm:g12:exp:growth}{comparación de crecimientos}, sustituimos
$x = \eu^{t}$: $\frac{\ln x}{x} = \frac{t}{\eu^t} \to 0$ cuando
$t \to +\infty$ por el \cref{thm:g12:exp:growth}, y análogamente para el
otro límite con $x = \eu^{-t}$, que da $x\ln x = -t\eu^{-t} \to 0$.
\end{proof}

\begin{method}[Resolver ecuaciones con $\exp$ y $\ln$]\label{met:g12:exp:equations}
Las dos funciones son estrictamente \omterm{def:g12:seq:monotonic}{crecientes}, así que se pueden aplicar
(o quitar) en los dos miembros de una \omterm{def:g10:algebra:equation}{ecuación} o de una inecuación sin
cambiar su sentido:
\[
\eu^{u} = \eu^{v} \iff u = v, \qquad
\eu^{u} \leq \eu^{v} \iff u \leq v,
\]
y lo mismo con $\ln$ sobre cantidades positivas. \emph{Comprueba siempre
antes los \omterm{def:g11:func:function}{dominios}} ($\ln$ exige argumentos positivos). Para ecuaciones del
tipo $a^x = b$ con $a>0$ y $a \neq 1$, reescribe $a^x = \eu^{x\ln a}$ y
resuelve $x = \frac{\ln b}{\ln a}$.
\end{method}

\begin{example}\label{ex:g12:exp:halflife}
Una \omterm{def:g10:proba:sample}{muestra} radiactiva se desintegra según $N(t) = N_0\,\eu^{-\lambda t}$.
Su \emph{periodo de semidesintegración} $T$ cumple $N(T) = N_0/2$, es
decir, $\eu^{-\lambda T} = \frac12$, luego $T = \frac{\ln 2}{\lambda}$: el
periodo no depende de la cantidad inicial.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g12:exp:1}
Simplifica $\dfrac{\eu^{3x}\,\eu^{-x+1}}{\eu^{x}}$ y
$\ln\!\left(\dfrac{\eu^2\sqrt{\eu}}{\eu^{-3}}\right)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:exp:2}
Resuelve en $\R$:
\[
\text{(a) } \eu^{2x} - 3\eu^{x} + 2 = 0;
\qquad
\text{(b) } \ln(x - 1) + \ln(x + 2) = \ln 4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:exp:3}
Calcula los límites:
\[
\lim_{x\to+\infty} \frac{\eu^x - x^2}{\eu^x + 1}, \qquad
\lim_{x\to+\infty} \frac{\ln x}{\sqrt x}, \qquad
\lim_{x\to0^+} x^2 \ln x, \qquad
\lim_{x\to+\infty} \bigl(x - \ln x\bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:4}
Estudia la \omterm{def:g11:func:function}{función} $f(x) = x\,\eu^{-x}$ en $\R$: variación, límites y
\omterm{def:g12:deriv:extremum}{extremo}; demuestra que su curva tiene un \omterm{def:g12:deriv:inflection}{punto de inflexión} y da sus
\omterm{def:g10:coordgeom:system}{coordenadas}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:5}
Demuestra que para todo $x > -1$ se cumple $\ln(1 + x) \leq x$, con igualdad
solo en $x = 0$. Deduce que para todo $n \geq 1$,
\[
\left(1 + \frac{1}{n}\right)^n \leq \eu \leq \left(1 - \frac{1}{n+1}\right)^{-(n+1)} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:6}
Se invierte un capital $C_0$ a un tipo anual del $3\,\%$, con los intereses
capitalizados cada año.
\begin{enumerate}
  \item Expresa el capital $C_n$ al cabo de $n$ años.
  \item ¿Al cabo de cuántos años se duplica el capital? Da la respuesta
        exacta con $\ln$ y después un valor numérico.
  \item Compárala con la \omterm{def:g10:numbers:approx}{aproximación} ``$70$ dividido entre el tipo en tanto
        por ciento'' que usan los banqueros.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:exp:7}
Resuelve la inecuación $\eu^{2x} - \eu^{x+1} > 0$ y después la inecuación
$\ln(x^2 - 1) \leq \ln(x + 5)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:exp:8}
Sea $f(x) = \dfrac{\eu^x}{x}$ para $x > 0$.
\begin{enumerate}
  \item Estudia la variación de $f$ en $\intoo{0}{+\infty}$ y da su \omterm{def:g10:functions:extrema}{mínimo}.
  \item ¿Para qué valores de $k$ tiene la \omterm{def:g10:algebra:equation}{ecuación} $\eu^x = kx$ $0$, $1$ o
        $2$ soluciones en $\intoo{0}{+\infty}$?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:exp:9}
Para $n \geq 1$, sea $u_n = \left(1 + \frac1n\right)^n$.
\begin{enumerate}
  \item Usando el \cref{exo:g12:exp:5}, demuestra que $(u_n)$ está \omterm{def:g12:seq:bounded}{acotada
        superiormente} por~$\eu$.
  \item Demuestra que $\ln u_n = n \ln\left(1 + \frac1n\right) \to 1$ y
        deduce que $u_n \to \eu$. (Indicación: reconoce un cociente
        incremental de $\ln$ en $1$.)
\end{enumerate}
\end{exercise}

\section{Problema: El logaritmo domestica el mundo}

\begin{problem}[{Problema de fin de semana --- tiempos de duplicación y la
regla del 72, las escalas de los terremotos, los ácidos y los pianos, y la
constante $\eu$ dejando huellas por todas partes}]
\label{pb:g12:exp:1}
Todo lo que crece un \emph{porcentaje} fijo crece exponencialmente (los
ahorros, las bacterias, las epidemias), y todo lo que abarca demasiadas
potencias de diez para poder abarcarlo con la vista (energías de terremotos,
acideces, intensidades sonoras) lo domestica un \omterm{def:g12:exp:ln}{logaritmo}. Este problema
calcula tiempos de duplicación y desenmascara la regla del 72 de los
banqueros, lee las escalas logarítmicas del mundo y recoge las huellas que
el número $\eu$ deja en todas las escenas del crimen
(\cref{prop:g12:exp:funceq}, \cref{thm:g12:exp:growth},
\cref{met:g12:exp:equations}).

\medskip
\noindent\textbf{Parte I --- Soltura.}
\begin{enumerate}
  \item Resuelve: $\eu^{2x} = 5$; \quad $\ln(3x - 1) = 2$; \quad
        $\eu^{2x} - 3\eu^x + 2 = 0$ (una \omterm{def:g10:algebra:equation}{ecuación} de segundo grado
        disfrazada).
  \item Simplifica: $\dfrac{\ln 8}{\ln 2}$; \quad
        $\ln\!\left(\eu^3 \sqrt{\eu}\right)$; \quad
        $\eu^{\ln 5 - \ln 2}$.
  \item Demuestra la desigualdad madre: $\eu^x \geq 1 + x$ para todo $x$
        real (estudia $f(x) = \eu^x - 1 - x$) y deduce
        $\ln(1 + u) \leq u$ para $u > -1$.
  \item Calcula $\lim_{x \to +\infty} x^2 \eu^{-x}$,
        $\lim_{x \to +\infty} \frac{\ln x}{x}$
        (\cref{thm:g12:exp:growth}) y
        $\lim_{x \to 0} \frac{\eu^x - 1}{x}$ (reconoce una \omterm{def:g12:deriv:derivative}{derivada}).
  \item Estudia $f(x) = x \ln x$ en $\intoo{0}{+\infty}$: variación, \omterm{def:g10:functions:extrema}{mínimo}
        y el límite en $0^+$ (se admite: $x \ln x \to 0$). Esta pequeña
        \omterm{def:g11:func:function}{función} mide la información y la entropía a lo largo de los
        volúmenes universitarios.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Tiempos de duplicación y la regla del
72.}
\begin{enumerate}[resume]
  \item Unos ahorros crecen un $3\,\%$ anual. Resuelve $1.03^n = 2$:
        ¿cuánto tarda en duplicarse el capital?
  \item Los banqueros estiman el tiempo de duplicación como
        $\frac{72}{\text{tipo en }\%}$. Prueba la regla al $3\,\%$, al
        $6\,\%$ y al $9\,\%$ frente al valor exacto
        $\frac{\ln 2}{\ln(1 + r)}$ y explícala después: para $r$ pequeño,
        $\ln(1 + r) \approx r$ (la desigualdad de la pregunta 3 es la mitad
        de la historia), así que la constante exacta es
        $100 \ln 2 \approx 69.3$. ¿Por qué prefieren los banqueros el
        $72$?
  \item Una bacteria se divide cada $20$ minutos: $p(t) = 2^{t/20}$ (con
        $t$ en minutos). Reescríbelo como $\eu^{\lambda t}$ y calcula $p$
        al cabo de $24$ horas. La respuesta (más de $10^{21}$) ¿qué
        demuestra sobre el modelo, y qué es lo que detiene a las colonias
        reales?
  \item La cafeína abandona el cuerpo con una semivida de unas $5$ horas.
        Tras un café de $100$ mg a las 15:00, ¿cuánta queda a las 23:00? ¿A
        qué hora baja de $10$ mg? (El insomnio tiene un \omterm{def:g12:exp:ln}{logaritmo}.)
  \item Un euro a un interés anual del $100\,\%$: calcula el capital a fin
        de año con capitalización anual, mensual y diaria, y da el techo
        que el \cref{exo:g12:exp:9} demostró infranqueable. ¿Qué constante
        famosa es el límite de la pura codicia?
  \item ¿Por qué representan los científicos $\ln(\text{casos})$ frente al
        tiempo durante la fase inicial de una epidemia? ¿Qué revela una
        recta en esa representación y qué mide su \omterm{def:g10:reffunc:affine}{pendiente}?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Las escalas logarítmicas del mundo.}
(Escribimos $\log_{10} x = \frac{\ln x}{\ln 10}$.)
\begin{enumerate}[resume]
  \item Nivel sonoro en decibelios: $L = 10 \log_{10}(I/I_0)$. Una
        conversación mide $60$ dB y un concierto de rock, $120$ dB: ¿por
        qué factor se diferencian sus intensidades sonoras?
  \item La magnitud de un terremoto sube en $1$ cuando la amplitud sísmica
        se multiplica por $10$, y la energía liberada escala como
        amplitud$^{3/2}$. Compara un terremoto de magnitud $5$ con uno de
        magnitud $7$: \omterm{def:g11:seq:geometric}{razón} de amplitudes y después \omterm{def:g11:seq:geometric}{razón} de energías.
  \item Química: $\text{pH} = -\log_{10}[\mathrm{H^+}]$. El zumo de limón
        tiene pH $2$ y la leche, pH $6.5$: ¿cuál es la \omterm{def:g11:seq:geometric}{razón} de sus
        concentraciones de ácido?
  \item Música: cada octava duplica la frecuencia. Del la más grave del
        piano ($27.5$ Hz) al do más agudo ($4\,186$ Hz), ¿cuántas octavas
        abarca el teclado? ¿Y por qué prefieren nuestros sentidos (el oído,
        la vista, la percepción de los terremotos) las escalas
        logarítmicas? (Una frase.)
  \item La regla de cálculo, la calculadora de los ingenieros durante $350$
        años: dos reglas graduadas de modo que la \emph{longitud} hasta la
        marca $x$ sea proporcional a $\ln x$. Explica cómo deslizar una
        regla sobre la otra multiplica números y nombra la identidad de la
        \cref{prop:g12:exp:lnalg} que hace el trabajo.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Las huellas de $\eu$.}
\begin{enumerate}[resume]
  \item Del \cref{exo:g12:exp:8}: la \omterm{def:g10:algebra:equation}{ecuación} $\eu^x = kx$ ($x > 0$) tiene
        $0$, $1$ o $2$ soluciones según la posición de $k$ respecto de un
        umbral. Reformula el resultado y comprueba el hecho geométrico que
        hay detrás: la recta $y = \eu x$ es exactamente la \omterm{def:g12:deriv:tangent}{tangente} a la
        exponencial que pasa por el origen.
  \item La constante del casi: con $n$ boletos de lotería de \omterm{def:g10:proba:distribution}{probabilidad}
        de premio $\frac1n$ cada uno,
        $\P(\text{ningún premio}) = \left(1 - \frac1n\right)^n$. Calcula su
        límite mediante $n \ln\!\left(1 - \frac1n\right)$ (usa el límite
        estándar $\frac{\ln(1 + u)}{u} \to 1$) y concílialo con el
        misterioso $0.368$ del \cref{pb:g11:binom:1}.
  \item La regla del $37\,\%$: para elegir al mejor de $n$ candidatos
        entrevistados en orden aleatorio (sin poder volver atrás), la
        estrategia óptima rechaza a los primeros
        $\frac{n}{\eu} \approx 37\,\%$ y se queda después con el primer
        candidato mejor que todos los anteriores, con una \omterm{def:g10:proba:distribution}{probabilidad} de
        acierto de en torno a $\frac1\eu$. ¿De dónde salen los dos
        $\frac1\eu$ de las preguntas 18 y 19? Enuncia el mecanismo común
        (muchos \omterm{def:g11:prob:model}{sucesos} independientes de \omterm{def:g10:proba:distribution}{probabilidad} pequeña) y calcula
        $\frac1\eu$ con tres decimales.
  \item Final: el retrato de $\eu$: el techo de la capitalización (pregunta
        10); la base cuya \omterm{def:g12:deriv:tangent}{tangente} en $0$ tiene \omterm{def:g10:reffunc:affine}{pendiente} exactamente $1$
        (preguntas 3 y 4); el crecimiento que ningún polinomio alcanza
        (pregunta 4); la constante de los ``por poco'' y de la parada
        óptima (preguntas 18 y 19); y su inversa $\ln$, que convierte
        productos en sumas (pregunta 16) y décadas en centímetros (Parte
        III). Una frase para cada cosa.
\end{enumerate}
\end{problem}
