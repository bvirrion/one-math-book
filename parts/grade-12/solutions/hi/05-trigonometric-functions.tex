\begin{solution}{exo:g12:trigo:1}
$\cos\frac{2\pi}{3} = \cos\left(\pi - \frac\pi3\right) = -\cos\frac\pi3
= -\frac12$।

$\sin\frac{5\pi}{6} = \sin\left(\pi - \frac\pi6\right) = \sin\frac\pi6
= \frac12$।

$\cos\frac{7\pi}{4} = \cos\left(2\pi - \frac{\pi}{4}\right)
= \cos\left(-\frac\pi4\right) = \frac{\sqrt2}{2}$।

$\sin\left(-\frac\pi3\right) = -\sin\frac\pi3 = -\frac{\sqrt3}{2}$।
\end{solution}

\begin{solution}{exo:g12:trigo:2}
$f'(x) = 2\sin x\cos x = \sin 2x$ (शृंखला नियम)।

$g'(x) = -3\sin(3x) + \sin x + x\cos x$ (शृंखला और गुणनफल के नियम)।

भागफल का नियम:
\[
h'(x) = \frac{\cos x\,(2 + \cos x) - \sin x\,(-\sin x)}{(2+\cos x)^2}
= \frac{2\cos x + \cos^2 x + \sin^2 x}{(2+\cos x)^2}
= \frac{2\cos x + 1}{(2+\cos x)^2}.
\]
\end{solution}

\begin{solution}{exo:g12:trigo:3}
\emph{(a)} $\sin x = \frac{\sqrt3}{2}$ का एक विशेष हल $\frac\pi3$ है। व्यापक हल: $x = \frac\pi3 + 2k\pi$ या $x = \pi - \frac\pi3 + 2k\pi = \frac{2\pi}{3} + 2k\pi$। $\intoc{-\pi}{\pi}$ में:
$x \in \left\{\frac\pi3, \frac{2\pi}{3}\right\}$।

\emph{(b)} $\cos 2x = \frac{\sqrt2}{2} = \cos\frac\pi4$ से $2x = \pm\frac\pi4 + 2k\pi$ मिलता है, इसलिए $x = \pm\frac\pi8 + k\pi$। $\intoc{-\pi}{\pi}$ में: $x \in \left\{-\frac{7\pi}{8}, -\frac\pi8, \frac\pi8, \frac{7\pi}{8}\right\}$।
\end{solution}

\begin{solution}{exo:g12:trigo:4}
$a = \frac\pi3$, $b = \frac\pi4$ के साथ:
\[
\cos\frac{\pi}{12} = \cos a\cos b + \sin a \sin b
= \frac12\cdot\frac{\sqrt2}{2} + \frac{\sqrt3}{2}\cdot\frac{\sqrt2}{2}
= \frac{\sqrt2 + \sqrt6}{4},
\]
\[
\sin\frac{\pi}{12} = \sin a\cos b - \cos a \sin b
= \frac{\sqrt3}{2}\cdot\frac{\sqrt2}{2} - \frac12\cdot\frac{\sqrt2}{2}
= \frac{\sqrt6 - \sqrt2}{4}.
\]
\end{solution}

\begin{solution}{exo:g12:trigo:5}
$s = \sin x$ रखिए: $2s^2 - s - 1 = (2s + 1)(s - 1) \geq 0$ ठीक तब जब $s \leq -\frac12$ या $s = 1$ हो।

$\sin x = 1$ से $x = \frac\pi2$ मिलता है। $\sin x \leq -\frac12$: इकाई वृत्त पर क्षैतिज रेखा $y = -\frac12$ के नीचे का चाप,
जो $\intcc{0}{2\pi}$ में $x \in \intcc{\frac{7\pi}{6}}{\frac{11\pi}{6}}$ है। अतः
\[
S = \left\{\frac{\pi}{2}\right\} \cup
\intcc{\frac{7\pi}{6}}{\frac{11\pi}{6}} .
\]
\end{solution}

\begin{solution}{exo:g12:trigo:6}
$\dfrac{\sin 3x}{x} = 3\,\dfrac{\sin 3x}{3x} \to 3 \cdot 1 = 3$ ($u = 3x \to 0$ के साथ संयोजन)।

$1 - \cos x = \dfrac{\sin^2 x}{1 + \cos x}$ का उपयोग करने पर:
\[
\frac{1 - \cos x}{x^2}
= \left(\frac{\sin x}{x}\right)^{\!2} \frac{1}{1 + \cos x}
\xrightarrow[x\to0]{} 1 \cdot \frac12 = \frac12 .
\]

$u = \frac1x \to 0^+$ प्रतिस्थापित करने पर: $x \sin\frac1x = \frac{\sin u}{u} \to 1$।
\end{solution}

\begin{solution}{exo:g12:trigo:7}
सभी $x$ के लिए $f'(x) = 1 - \sin x \geq 0$, और समता ठीक तब जब $\sin x = 1$ हो, \emph{अर्थात्} $x = \frac\pi2 + 2k\pi$। $\intcc{0}{2\pi}$
पर $f$ $f(0) = 1$ से $f(2\pi) = 2\pi + 1$ तक बढ़ता है ($\frac{\pi}{2}$ पर $f'$ का शून्य अलग-थलग है, इसलिए
वृद्धि निरंतर भी है); \omterm{def:g10:functions:extrema}{न्यूनतम} $0$ पर $1$, \omterm{def:g10:functions:extrema}{अधिकतम} $2\pi$ पर $2\pi + 1$। $\R$ पर हर
जगह $f' \geq 0$ है और शून्य केवल अलग-थलग बिंदुओं पर, इसलिए \cref{thm:g12:deriv:variations} से $f$ (निरंतर)
\omterm{def:g12:seq:monotonic}{वर्धमान} है, यद्यपि $f'$ हर $\frac\pi2 + 2k\pi$ पर लुप्त हो जाता है।
\end{solution}

\begin{solution}{exo:g12:trigo:8}
\emph{1.} मान लीजिए $R = \sqrt{A^2 + B^2} > 0$ है। बिंदु $\left(\frac AR, \frac BR\right)$ इकाई वृत्त पर है, इसलिए ऐसा $\varphi$ है
कि $\cos\varphi = \frac AR$ और $\sin\varphi = \frac BR$ हों। तब
\[
R\cos(x - \varphi) = R\bigl(\cos x\cos\varphi + \sin x \sin\varphi\bigr)
= A\cos x + B\sin x = f(x).
\]

\emph{2.} चूँकि $\cos$ का परिसर $\intcc{-1}{1}$ है, इसलिए $f$ का \omterm{def:g10:functions:extrema}{अधिकतम} $R$ और \omterm{def:g10:functions:extrema}{न्यूनतम}
$-R$ है। $\cos x + \sin x$ के लिए: $A = B = 1$, $R = \sqrt2$, $\varphi = \frac\pi4$, इसलिए \omterm{def:g10:algebra:equation}{समीकरण} $\sqrt2\cos\left(x - \frac\pi4\right) = 1$ बन जाता है,
\emph{अर्थात्} $\cos\left(x - \frac\pi4\right) = \frac{\sqrt2}{2}$, जिससे $x - \frac\pi4 = \pm\frac\pi4 + 2k\pi$ मिलता है: $\intoc{-\pi}{\pi}$ में $x \in \left\{0, \frac\pi2\right\}$।
\end{solution}

\begin{solution}{exo:g12:trigo:9}
$(\sin)'' = (\cos)' = -\sin$ और $(\cos)'' = (-\sin)' = -\cos$: दोनों $y'' = -y$ को हल करते हैं।

मान लीजिए $g = y - a\cos - b\sin$ है। तब $g'' = -g$ (अवकलन की रैखिकता), $g(0) = y(0) - a = 0$ और $g'(0) = y'(0) - b = 0$। $E = g^2 + (g')^2$ रखिए। तब
\[
E' = 2g g' + 2g' g'' = 2g g' - 2g' g = 0,
\]
इसलिए $E$ अचर है और $E(0) = 0$ के बराबर। दो वर्गों का योग सर्वत्र लुप्त होने से
$g = 0$ अनिवार्य हो जाता है, अतः $y = a\cos + b\sin$।
\end{solution}

\begin{solution}{pb:g12:trigo:1}
\textbf{1.} $0.99833\ldots$ और $0.99998\ldots$: घिसटकर $1$ की ओर।

\textbf{2.} त्रिभुज $OAM$: आधार $1$, ऊँचाई $\sin x$: क्षेत्रफल $\frac{\sin x}{2}$।
त्रिज्यखंड $OAM$: इकाई वृत्त-चक्रिका का $\frac{x}{2\pi}$ अंश: क्षेत्रफल $\frac x2$। त्रिभुज
$OAT$: आधार $1$, ऊँचाई $\tan x$: क्षेत्रफल $\frac{\tan x}{2}$।

\textbf{3.} त्रिभुज त्रिज्यखंड के भीतर बैठा है और त्रिज्यखंड बड़े त्रिभुज के
भीतर: $\frac{\sin x}{2} < \frac x2 < \frac{\tan x}{2}$, अर्थात् $\sin x < x < \tan x$। $\sin x < x$ से $\frac{\sin x}{x} < 1$; और $x < \tan x = \frac{\sin x}{\cos x}$ से, $\frac{\cos x}{x} > 0$ से गुणा करने
पर, $\cos x < \frac{\sin x}{x}$। $x \to 0^+$ पर $\cos x \to 1$: शिकंजे में, $\frac{\sin x}{x} \to 1$; और $\frac{\sin x}{x}$ सम है, इसलिए द्विपक्षीय
सीमा $1$ है।

\textbf{4.} $\frac{1 - \cos x}{x^2} =
\frac{1 - \cos^2 x}{x^2 (1 + \cos x)} =
\left(\frac{\sin x}{x}\right)^2 \frac{1}{1 + \cos x}
\to 1 \times \frac12$।

\textbf{5.} $\sin'(0) = \lim_{h \to 0} \frac{\sin h - 0}{h} =
1$: \omterm{def:g12:trigo:cossin}{ज्या} और \omterm{def:g12:trigo:cossin}{कोज्या} का पूरा अवकलन इसी एक सीमा से बहता है। अंशों
में नापने पर सीमा $\frac{\pi}{180}$ होती, और हर \omterm{def:g12:deriv:derivative}{अवकलज} उसी अचर को घसीटता: \omterm{def:g11:trigo:radian}{रेडियन} ठीक वही
मात्रक है जो दोलन के कलन को स्वच्छ बना देता है।

\textbf{6.} $y' = -R\omega\sin(\omega t - \varphi)$, $y'' = -R\omega^2\cos(\omega t - \varphi) = -\omega^2 y$।

\textbf{7.} $\omega = 2$। $y(0) = A = 3$; $y'(0) = 2B = 8$: $B = 4$: $y = 3\cos 2t + 4\sin 2t$।

\textbf{8.} $R = \sqrt{9 + 16} = 5$; आवर्तकाल $\frac{2\pi}{2} = \pi$; कला $\varphi = \arctan\frac43 \approx 0.927$: $y = 5\cos(2t - 0.927)$।

\textbf{9.} $5\cos(2t - \varphi) = 0$ पहली बार तब जब $2t - \varphi = \frac\pi2$: $t = \frac{\pi/2 + 0.927}{2} \approx 1.25$ s।

\textbf{10.} $\sin\theta \approx \theta$ के साथ: $\theta'' = -\frac gL \theta$: $\omega = \sqrt{\frac gL}$ वाली सरल आवर्त गति, आवर्तकाल $T = \frac{2\pi}{\omega} = 2\pi\sqrt{\frac Lg}$।
$L = 1$ के लिए: $T \approx 2.006$ s। आवर्तकाल आयाम पर निर्भर नहीं करता (छोटे झूलों के लिए)
--- गैलीलियो की समकालिकता, वही तथ्य जो लोलक-घड़ियों को संभव बनाता है।

\textbf{11.} $L = g\left(\frac{T}{2\pi}\right)^2 =
\frac{9.81}{\pi^2} \approx 0.994$ m: सेकंड-लोलक याम्योत्तर वाले मीटर से लगभग $6$ mm छोटा
है। अकादमी ने याम्योत्तर चुना (गुरुत्व जगह-जगह बदलता है, जो लोलक को धोखा दे
देता है); यदि $g$ बाल भर बड़ा होता, तो हमारा मीटर टिक-टिक करता।

\textbf{12.} $\cos\left(x + \frac\pi2\right) =
\cos x \cos\frac\pi2 - \sin x \sin\frac\pi2 = -\sin x$; $\sin\left(x + \frac\pi2\right) = \cos x$। इसलिए \omterm{def:g12:trigo:cossin}{ज्या} का अवकलन \omterm{def:g12:trigo:cossin}{ज्या} को एक चौथाई चक्कर खिसका
देता है ($\cos$), फिर दोबारा ($-\sin$), फिर दोबारा, और चार क़दमों में घर वापसी:
अवकलन वृत्त को घुमा देता है।

\textbf{13.} $\cos(a \mp b)$ के दोनों प्रसार जोड़ने पर: $\cos(a-b) + \cos(a+b) = 2\cos a\cos b$। $a = b = x$ के साथ: $1 + \cos 2x = 2\cos^2 x$,
अर्थात् $\cos^2 x = \frac{1 + \cos 2x}{2}$।

\textbf{14.} $p = \frac{p+q}{2} + \frac{p-q}{2}$ और उसके दर्पण $q$ को खोलकर जोड़ने पर: $\sin p + \sin q = 2\sin\frac{p+q}{2}\cos\frac{p-q}{2}$। $440$ और
$444$ Hz के लिए: $442$ Hz का एक स्वर, जिसकी प्रबलता $\cos(2\pi \cdot 2t)$ से मॉडुलित होती है
--- और प्रबलता प्रति सेकंड $4$ बार चोटी छूती है (अन्वालोप का \omterm{def:g10:numbers:abs}{निरपेक्ष मान}):
यानी चार विस्पंद प्रति सेकंड, जो स्वरित्रों के मिल जाने पर लुप्त हो जाते हैं।

\textbf{15.} $\cos 3x = \cos 2x\cos x - \sin 2x \sin x =
(2\cos^2 x - 1)\cos x - 2\sin^2 x\cos x = 4\cos^3 x -
3\cos x$। $x = 20^\circ$ पर: $\cos 60^\circ = \frac12$, इसलिए $c = \cos 20^\circ$ $8c^3 - 6c - 1 = 0$ को संतुष्ट करता है ---
एक ऐसा त्रिघात जिसका कोई वर्गमूल वाला हल नहीं: $20^\circ$ की रचना, अर्थात् $60^\circ$
का त्रिभाजन, पटरी और परकार की पहुँच से बाहर है।

\textbf{16.} $R = \sqrt{9 + 27} = 6$; $\tan\varphi = \frac{3\sqrt3}{3} = \sqrt3$: $\varphi = \frac\pi3$: $I(t) = 6\cos\left(100\pi t -
\frac\pi3\right)$: शिखर $6$ ऐम्पियर, कला-विलंब $60^\circ$।

\textbf{17.} $\sin^2 t = \frac{1 - \cos 2t}{2}$: आवर्तकाल $\pi$। $\sin 2t$ का आवर्तकाल $\pi$ है और $\sin 3t$ का
$\frac{2\pi}{3}$: योग सबसे छोटे उभयनिष्ठ गुणज $2\pi$ के बाद दोहराता है।

\textbf{18.} सिकुड़ते अन्वालोप $\pm\eu^{-t/10}$ के भीतर आवर्तकाल $1$ का दोलन: हर झूला
लगभग $10\,\%$ नीचा। $t = 10$ पर अन्वालोप $\eu^{-1} \approx 0.368$ है --- फिर वही \cref{pb:g12:exp:1} वाला अचर। ऐसे
अवमंदित हलों वाले \omterm{def:g10:algebra:equation}{समीकरण} ($y'' + a y' + b y = 0$) अवकल \omterm{def:g10:algebra:equation}{समीकरणों} वाले अध्याय में हल होते हैं।

\textbf{19.} विस्थापन के समानुपाती पीछे खींचने वाला बल $y'' = -\omega^2 y$ देता है; और \cref{exo:g12:trigo:9}
दिखाता है कि उस \omterm{def:g10:algebra:equation}{समीकरण} के हल \emph{ठीक} $\cos$ और $\sin$ के संयोजन ही हैं ---
अस्तित्व दिखाकर, अद्वितीयता अभ्यास से: कंपन के पास ज्यावक्रीय होने के सिवा
कोई चारा नहीं।

\textbf{20.} एक सीमा ($\frac{\sin x}{x} \to 1$, दबे हुए क्षेत्रफलों से), दो \omterm{def:g12:deriv:derivative}{अवकलज} ($\sin' = \cos$, $\cos' = -\sin$),
एक \omterm{def:g10:algebra:equation}{समीकरण} ($y'' = -\omega^2 y$), और घड़ियों, तारों, धाराओं तथा ज्वारों का पूरा संसार।
तारांकन: बड़े झूलों के लिए $T = 4\sqrt{\frac Lg} \cdot
\frac{\pi/2}{M(1, \cos(\theta_0/2))}$ --- \cref{pb:g12:seq:1} का समांतर--गुणोत्तर \omterm{def:g11:stat:mean}{माध्य}
दीर्घवृत्तीय समाकल को मुट्ठी भर पुनरावृत्तियों में निकाल देता है: गाउस की
डायरी और गैलीलियो का लोलक, बस एक सूत्र की दूरी पर।
\end{solution}
