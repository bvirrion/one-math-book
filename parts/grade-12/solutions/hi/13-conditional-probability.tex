\section*{अध्याय \ref{ch:g12:condprob} --- सप्रतिबंध प्रायिकता और स्वतंत्रता}

\begin{solution}{exo:g12:condprob:1}
तस्वीर के $12$ पत्ते हैं, जिनमें $4$ बादशाह हैं: $\pcond{\text{तस्वीर का पत्ता}}{\text{बादशाह}} = \frac{4}{12} = \frac13$।

\omterm{def:g12:condprob:indep}{स्वतंत्रता}: $\P(\text{बादशाह} \cap \text{पान}) = \frac{1}{52}$ (पान का बादशाह), और $\P(\text{बादशाह})\,\P(\text{पान}) = \frac{4}{52}\times\frac{13}{52}
= \frac{1}{52}$। बराबर: \omterm{def:g10:proba:events}{घटनाएँ} \omterm{def:g12:condprob:indep}{स्वतंत्र} हैं।
\end{solution}

\begin{solution}{exo:g12:condprob:2}
\emph{1.} पहली शाखा: लाल $\frac58$, नीली $\frac38$; दूसरी शाखाएँ (बिना वापसी): लाल के
बाद लाल $\frac47$ / नीली $\frac37$; नीली के बाद लाल $\frac57$ / नीली $\frac27$।

\emph{2.} $\P(RR) = \frac58 \times \frac47 = \frac{5}{14}$. कुल \omterm{def:g10:proba:distribution}{प्रायिकता} से
\[
\P(\text{दूसरी लाल}) = \frac58\cdot\frac47 + \frac38\cdot\frac57
= \frac{20 + 15}{56} = \frac58 .
\]
(पहली बार जैसा ही --- सममिति से दूसरी गेंद कलश की कोई एकसमान यादृच्छिक गेंद
ही है।)
\end{solution}

\begin{solution}{exo:g12:condprob:3}
$\P(A) = \frac{6}{36} = \frac16$ (छह युग्मों का योग $7$ है), $\P(B) = \frac16$, और $A \cap B = \{(3,4)\}$ की \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac{1}{36} = \P(A)\P(B)$ है: $A$
और $B$ \omterm{def:g12:condprob:indep}{स्वतंत्र} हैं। (योग $7$ ख़ास है: पहला पासा चाहे जो हो, दूसरे का ठीक
एक मान वही योग देता है।)

$\P(C) = \frac{5}{36}$ और $B \cap C = \{(3,3)\}$: $\P(B \cap C) = \frac{1}{36} \neq \frac16 \times \frac{5}{36}
= \frac{5}{216}$। \omterm{def:g12:condprob:indep}{स्वतंत्र} नहीं।
\end{solution}

\begin{solution}{exo:g12:condprob:4}
\emph{1.} मशीन के अनुसार विभाजन के साथ कुल \omterm{def:g10:proba:distribution}{प्रायिकता}:
\[
\P(D) = 0.5\times0.01 + 0.3\times0.02 + 0.2\times0.04
= 0.005 + 0.006 + 0.008 = 0.019 = 1.9\% .
\]

\emph{2.} बेज़: $\pcond{D}{M_3} = \dfrac{0.008}{0.019} = \dfrac{8}{19} \approx 0.42$. कुल उत्पादन का केवल पाँचवाँ हिस्सा बनाने वाली मशीन
दोषों के $40\%$ से अधिक के लिए ज़िम्मेदार है।
\end{solution}

\begin{solution}{exo:g12:condprob:5}
शर्त है
\[
\frac{0.99\,p}{0.99\,p + 0.05\,(1-p)} \geq 0.9 .
\]
हर धनात्मक है, इसलिए यह $0.99p \geq 0.891p + 0.045 - 0.045p$ बन जाती है, \emph{अर्थात्} $0.144\,p \geq 0.045$, \emph{अर्थात्}
\[
p \geq \frac{0.045}{0.144} = 0.3125 .
\]
अकेला परीक्षण $90\%$ की निश्चितता तभी देता है जब रोग परीक्षित समष्टि के $31\%$ से
अधिक को पहले से हो --- और इसीलिए विरले रोगों की व्यापक छानबीन में पुष्टिकारक
परीक्षण अनिवार्य हैं।
\end{solution}

\begin{solution}{exo:g12:condprob:6}
\emph{1.} $E$ को तीन क्रम साकार करते हैं (HHT, HTH, THH), और \omterm{def:g12:condprob:indep}{स्वतंत्रता} से
हर एक की \omterm{def:g10:proba:distribution}{प्रायिकता} $p^2(1-p)$ है: $\P(E) = 3p^2(1 - p)$।

\emph{2.} $f(p) = 3p^2 - 3p^3$ का $f'(p) = 6p - 9p^2 = 3p(2 - 3p)$ है, जो $\intoo{0}{1}$ पर पहले धनात्मक और फिर ऋणात्मक है: \omterm{def:g10:functions:extrema}{अधिकतम}
$p = \frac23$ पर, जहाँ $\P(E) = 3\cdot\frac49\cdot\frac13 = \frac49$।
\end{solution}

\begin{solution}{exo:g12:condprob:7}
मान लीजिए आपने दरवाज़ा 1 चुना और संचालक ने दरवाज़ा 3 खोला (\omterm{def:g11:prob:model}{घटना} $H_3$)। $B_i$ =
``पुरस्कार दरवाज़े $i$ के पीछे'' के साथ $\P(B_i) = \frac13$ और
\[
\pcond{B_1}{H_3} = \frac12, \qquad
\pcond{B_2}{H_3} = 1, \qquad
\pcond{B_3}{H_3} = 0
\]
(यदि पुरस्कार आपके ही दरवाज़े के पीछे हो तो संचालक 2 और 3 में से यादृच्छिक
चुनाव करता है; और यदि वह दरवाज़े 2 के पीछे हो तो उसे 3 खोलना ही पड़ता है)।
बेज़:
\[
\pcond{H_3}{B_1}
= \frac{\frac13\cdot\frac12}{\frac13\cdot\frac12 + \frac13\cdot1 + 0}
= \frac{1/6}{1/2} = \frac13,
\qquad
\pcond{H_3}{B_2} = \frac{\frac13}{\frac12} = \frac23 .
\]
टिके रहने पर \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac13$ से जीत होती है और बदलने पर $\frac23$ से।
\end{solution}

\begin{solution}{exo:g12:condprob:8}
\emph{1.} बहुमत तब ग़लत होता है जब तीनों प्रतियों में से कम से कम दो पलट जाएँ:
\[
3\varepsilon^2(1-\varepsilon) + \varepsilon^3
= 3(0.01)(0.9) + 0.001 = 0.028,
\]
जो $\varepsilon = 0.1$ से कहीं छोटा है: पुनरावृत्ति वाला कूट काम करता है।

\emph{2.} यदि $1$ भेजा गया हो ($111$ के रूप में), तो $101$ पाने के लिए ठीक एक
पलटाव चाहिए: \omterm{def:g10:proba:distribution}{प्रायिकता} $\varepsilon(1-\varepsilon)^2 = 0.081$। और यदि $0$ भेजा गया हो ($000$), तो दो पलटाव
चाहिए: $\varepsilon^2(1-\varepsilon) = 0.009$। बराबर पूर्व \omterm{def:g10:proba:distribution}{प्रायिकताओं} के साथ बेज़:
\[
\pcond{101}{\,1 \text{ भेजा गया}} = \frac{0.081}{0.081 + 0.009} = 0.9 .
\]
बहुमत वाला गूढ़वाचन ($101 \mapsto 1$) सचमुच अधिक संभावित अनुमान है।
\end{solution}

\begin{solution}{pb:g12:condprob:1}
\textbf{1.} $\P(\text{लाल}) = \frac12 \cdot \frac23 + \frac12
\cdot \frac14 = \frac13 + \frac18 = \frac{11}{24}$।

\textbf{2.} $\P(\text{I} \mid \text{लाल}) =
\dfrac{\frac12 \cdot \frac23}{\frac{11}{24}} =
\dfrac{1/3}{11/24} = \dfrac{8}{11}$।

\textbf{3.} $\P(\text{सम} \cap \leq 4) = \P(\{2, 4\}) =
\frac13$ और $\P(\text{सम})\P(\leq 4) = \frac12 \cdot
\frac23 = \frac13$: \omterm{def:g12:condprob:indep}{स्वतंत्र} (अप्रत्याशित रूप से!)। $\leq 3$ के लिए: $\P(\{2\}) = \frac16$
बनाम $\frac12 \cdot \frac12 = \frac14$: परावलंबी।

\textbf{4.} \omterm{def:g12:condprob:indep}{स्वतंत्र} \omterm{def:g10:proba:events}{घटनाओं} के \omterm{def:g10:proba:operations}{पूरक} भी \omterm{def:g12:condprob:indep}{स्वतंत्र} होते हैं, इसलिए
\[
\P(\text{कोई नहीं}) = 0.1 \times 0.1 = 0.01,
\qquad
\P(\text{कम से कम एक}) = 0.99 .
\]

\textbf{5.} $\frac{4}{52} \times \frac{3}{51} =
\frac{1}{221}$।

\textbf{6.} रोगग्रस्त: $100$; स्वस्थ: $99\,900$। धनात्मक: $99$ सत्य ($100$ का 99\,\%)
और $4\,995$ मिथ्या ($99\,900$ का 5\,\%): कुल $5\,094$ धनात्मक।

\textbf{7.} $\P(\text{बीमार} \mid +) = \frac{99}{5\,094}
\approx 1.9\,\%$: परीक्षण जिन लोगों पर झंडी उठाता है उनमें पचास में एक से भी
कम सचमुच रोगी है।

\textbf{8.} $\dfrac{0.001 \times 0.99}{0.001 \times 0.99 +
0.999 \times 0.05} = \dfrac{0.00099}{0.05094} \approx 0.019$: गिनतियाँ और सूत्र दोनों मेल खाते हैं।

\textbf{9.} गड्डमड्ड: $\P(+ \mid \text{बीमार}) = 99\,\%$ (परीक्षण की प्रचारित गुणवत्ता) और $\P(\text{बीमार} \mid +)$ (रोगी का
असली प्रश्न)। $\frac{1}{1000}$ की आधार-दर पर रोगी इतने विरले हैं कि विशाल स्वस्थ बहुमत
से रिसा हुआ $5\,\%$ भी सच्चे धनात्मकों को पचास-एक के अनुपात में डुबो देता है।

\textbf{10.} $\dfrac{0.019 \times 0.99}{0.019 \times 0.99 +
0.981 \times 0.05} \approx 28\,\%$. हर \omterm{def:g12:condprob:indep}{स्वतंत्र} धनात्मक परिणाम संभावनाओं को उसी गुणक से गुणा
कर देता है: प्रमाण जुड़ते जाते हैं, और तीसरा धनात्मक $90\,\%$ से भी आगे ले जाता
--- और इसीलिए पुष्टिकारक परीक्षण होते हैं।

\textbf{11.} $\dfrac{0.05 \times 0.99}{0.05 \times 0.99 +
0.95 \times 0.05} \approx 51\,\%$: उच्च-जोखिम समूह पर वही परीक्षण सूचनाप्रद है। सीख: वहीं
छानबीन कीजिए जहाँ आधार-दर पर्याप्त हो; विरले रोग की व्यापक छानबीन हज़ारों
झूठे ख़तरे गढ़ देती है।

\textbf{12.} प्रत्याशित निर्दोष मेल: $10^7 \times 10^{-6} = 10$. लगभग $11$ मेल खाने वाले लोगों में
($10$ निर्दोष, $1$ दोषी --- यदि अपराधी नगर में हो), \emph{केवल मेल से} मिला
संदिग्ध लगभग $\frac{10}{11} \approx
91\,\%$ \omterm{def:g10:proba:distribution}{प्रायिकता} से निर्दोष है।

\textbf{13.} अभियोजक ने $\P(\text{मेल} \mid \text{निर्दोष}) = 10^{-6}$ उद्धृत की; अदालत को $\P(\text{निर्दोष} \mid \text{मेल})$ चाहिए थी --- यानी
प्रश्न 9 की वही गड़बड़ी, शपथ लेकर। (और जिस असली मामले ने इन भूलों को प्रसिद्ध
किया, उसमें एक दूसरी चूक ने बात और बिगाड़ दी: प्रश्न 14 देखिए।)

\textbf{14.} भोला वर्ग: $\left(\frac{1}{8500}\right)^2
\approx \frac{1}{72 \text{ मिलियन}}$. और यदि पहली के बाद दूसरी दुर्घटना पाँच गुना
अधिक संभावित हो: $\frac{1}{8500} \times \frac{5}{8500} \approx
\frac{1}{14.5 \text{ मिलियन}}$ --- यानी पाँच गुना बड़ी, और तब भी यह अनदेखा करते हुए
कि \emph{वैकल्पिक} परिकल्पना को भी बेज़ से तोलना चाहिए। सीख: \omterm{def:g10:proba:distribution}{प्रायिकताओं} का
गुणा \omterm{def:g12:condprob:indep}{स्वतंत्रता} माँगता है, और साझे कारण (आनुवंशिकी, परिवेश) उसे नष्ट कर देते
हैं।

\textbf{15.} बेज़ \emph{सारे} प्रमाणों के संभाव्यता-अनुपात गुणा करके एक ही
पश्च \omterm{def:g10:proba:distribution}{प्रायिकता} बना देता है: मेल, अनुपस्थिति का साक्ष्य, रेशे --- हर एक
संभावनाएँ खिसकाता है। जूरी-सदस्य का वाक्य: \emph{पूछिए कि दोनों परिकल्पनाओं के
अधीन प्रमाण कितना संभावित है, और किसी \omterm{def:g12:condprob:cond}{सप्रतिबंध प्रायिकता} को उसकी उल्टी कभी मत
समझिए}।

\textbf{16.} $\dfrac{0.4 \times 0.15}{0.4 \times 0.15 +
0.6 \times 0.005} = \dfrac{0.060}{0.063} \approx 95\,\%$ स्पैम: एक ही शब्द लगभग फ़ैसला कर देता है।

\textbf{17.} पाँच प्रतियाँ तब विफल होती हैं जब $3$, $4$ या $5$ पलटें:
$\binom53 \varepsilon^3(1-\varepsilon)^2 + \binom54
\varepsilon^4 (1 - \varepsilon) + \varepsilon^5 \approx
0.0086$ --- जबकि तीन प्रतियों के लिए $0.028$: पुनरावृत्ति विश्वसनीयता ख़रीदती है,
और क़ीमत चौड़ाई से चुकाती है।

\textbf{18.} सम पलटाव: $\binom40 0.9^4 + \binom42 0.9^2
0.1^2 + \binom44 0.1^4 = 0.6561 + 0.0486 + 0.0001 = 0.7048$; और $\frac{1 + 0.8^4}{2} = \frac{1 + 0.4096}{2} = 0.7048$: तरकीब वाला सूत्र हर प्रसारक से होकर
अभिनति $(1 - 2\varepsilon)$ का पीछा करता जाता है।

\textbf{19.} स्पैम वाले शब्द झुंड में चलते हैं --- जिस संदेश में ``लॉटरी'' है
उसमें ``जीत'' भी बहुत संभावित है, इसलिए उनका संयुक्त प्रमाण उतना प्रबल नहीं है
जितना गुणनफल दिखाता है। छन्नी फिर भी काम करती है क्योंकि यह बढ़ा-चढ़ाकर आकलन
प्रायः \emph{सही दिशा} में धकेलता है: वर्गीकरण को $50\,\%$ का सही पक्ष चाहिए,
सही पश्च \omterm{def:g10:proba:distribution}{प्रायिकता} नहीं।

\textbf{20.} चिकित्सालय: विरला रोग किसी प्रबल परीक्षण को दुर्बल फ़ैसले में बदल
देता है --- आधार-दर से नवीनीकरण कीजिए, फिर प्रमाण जोड़ते जाइए। अदालत: किसी
बड़े नगर में विरला मेल नगर पर अभियोग लगाता है, उस आदमी पर नहीं --- संभाव्यताएँ
फ़ैसले नहीं होतीं। डाकख़ाना: अनेक दुर्बल सुराग़, गुणा किए हुए, डाक छाँट देते हैं
--- भोलेपन के बावजूद। और तीनों जाल: पूर्व \omterm{def:g10:proba:distribution}{प्रायिकता} भूलिए और मूर्ख बनिए; नक़ली
\omterm{def:g12:condprob:indep}{स्वतंत्रता} मानिए और बहुत बड़े मूर्ख बनिए; और प्रमाण टुकड़ों में पढ़िए तथा कभी
जान ही मत पाइए कि आप जानते क्या हैं।
\end{solution}
