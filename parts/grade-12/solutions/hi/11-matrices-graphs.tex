\begin{solution}{exo:g12:matrix:1}
\[
A + B = \begin{pmatrix} 3 & 2\\ 1 & 2\end{pmatrix}, \quad
AB = \begin{pmatrix} 4 & 2\\ 1 & 1\end{pmatrix}, \quad
BA = \begin{pmatrix} 2 & 4\\ 1 & 3\end{pmatrix}, \quad
A^2 = \begin{pmatrix} 1 & 4\\ 0 & 1\end{pmatrix}.
\]
ध्यान दीजिए कि $AB \neq BA$।
\end{solution}

\begin{solution}{exo:g12:matrix:2}
$\det A = 6 - 5 = 1 \neq 0$: $A^{-1} = \begin{pmatrix} 3 & -5\\ -1 & 2 \end{pmatrix}$।
$\det B = 12 - 12 = 0$: $B$ \omterm{def:g12:matrix:inverse}{प्रतिलोमनीय} नहीं है।
\end{solution}

\begin{solution}{exo:g12:matrix:3}
निकाय $AX = Y$ है, जहाँ $A$ \cref{exo:g12:matrix:2} जैसा है और $Y = \begin{pmatrix} 1\\ 2\end{pmatrix}$:
\[
X = A^{-1}Y = \begin{pmatrix} 3 & -5\\ -1 & 2\end{pmatrix}
\begin{pmatrix} 1\\ 2\end{pmatrix}
= \begin{pmatrix} -7\\ 3\end{pmatrix}:
\qquad x = -7,\ y = 3 .
\]
\end{solution}

\begin{solution}{exo:g12:matrix:4}
\emph{1.} $N^2 = \begin{pmatrix} 0&1\\0&0\end{pmatrix}
\begin{pmatrix} 0&1\\0&0\end{pmatrix} = 0$, और $I$ हर \omterm{def:g12:matrix:matrix}{आव्यूह} के साथ क्रमविनिमेय है।

\emph{2.} चूँकि दोनों पद क्रमविनिमेय हैं, इसलिए द्विपद प्रमेय लागू होती है और
$N^2$ वाले सभी पद लुप्त हो जाते हैं:
\[
A^k = (2I + N)^k = 2^k I + k\,2^{k-1} N
= \begin{pmatrix} 2^k & k\,2^{k-1}\\ 0 & 2^k \end{pmatrix}.
\]
$k = 2$ के लिए जाँच: $A^2 = \begin{pmatrix} 2&1\\0&2\end{pmatrix}^2
= \begin{pmatrix} 4&4\\0&4\end{pmatrix}$, और सूत्र $2^2 = 4$, $2 \times 2 = 4$ देता है। \checkmark
\end{solution}

\begin{solution}{exo:g12:matrix:5}
\emph{आगमन:} $n = 1$ के लिए: $A^1 = \begin{pmatrix} 0&1\\1&1\end{pmatrix}
= \begin{pmatrix} F_0 & F_1\\ F_1 & F_2\end{pmatrix}$। मान लीजिए सूत्र $n$ के लिए सही है; तब
\[
A^{n+1} = A^n A
= \begin{pmatrix} F_{n-1} & F_n\\ F_n & F_{n+1}\end{pmatrix}
\begin{pmatrix} 0 & 1\\ 1 & 1\end{pmatrix}
= \begin{pmatrix} F_n & F_{n-1} + F_n\\ F_{n+1} & F_n + F_{n+1}\end{pmatrix}
= \begin{pmatrix} F_n & F_{n+1}\\ F_{n+1} & F_{n+2}\end{pmatrix}.
\]
\emph{सर्वसमिका:} $2\times2$ \omterm{def:g12:matrix:matrix}{आव्यूहों} के लिए प्रसार करने पर $\det(MN) = \det M \det N$ दिखता है; अतः $\det(A^n) = (\det A)^n = (-1)^n$,
और $\det A^n = F_{n-1}F_{n+1} - F_n^2$। (यही \emph{कासीनी सर्वसमिका} है।)
\end{solution}

\begin{solution}{exo:g12:matrix:6}
\emph{1.} शीर्षों को $1, 2, 3$ क्रम में रखने पर:
\[
M = \begin{pmatrix} 0&1&1\\ 0&0&1\\ 1&0&0\end{pmatrix}, \quad
M^2 = \begin{pmatrix} 1&0&1\\ 1&0&0\\ 0&1&1\end{pmatrix}, \quad
M^3 = \begin{pmatrix} 1&1&1\\ 1&0&1\\ 1&0&1 \end{pmatrix}.
\]

\emph{2.} $\bigl(M^3\bigr)_{11} = 1$: शीर्ष $1$ पर लंबाई $3$ की ठीक एक बंद चाल है, अर्थात् $1 \to 2 \to 3 \to 1$।
(चाल $1 \to 3 \to 1$ केवल लंबाई $2$ की है, और $1 \to 3$ फिर $3\to1$ फिर $1\to3$ $3$ पर समाप्त
होती है।)
\end{solution}

\begin{solution}{exo:g12:matrix:7}
\emph{1.} $a_{n+1} = 0.8a_n + 0.3b_n$, $b_{n+1} = 0.2a_n + 0.7b_n$: $M = \begin{pmatrix} 0.8 & 0.3\\ 0.2 & 0.7\end{pmatrix}$।

\emph{2.} $MX = X$ से $0.8a + 0.3b = a$ मिलता है, \emph{अर्थात्} $0.3b = 0.2a$, इसलिए $b = \frac23 a$; और $a + b = 1$ के
साथ: $a = 0.6$, $b = 0.4$।

\emph{3.} $b_n = 1 - a_n$ का उपयोग करने पर: $a_{n+1} = 0.8a_n + 0.3(1 - a_n) = 0.5a_n + 0.3$, इसलिए
\[
c_{n+1} = a_{n+1} - 0.6 = 0.5a_n + 0.3 - 0.6 = 0.5(a_n - 0.6) = 0.5\,c_n .
\]
अतः $c_n = 0.5^n c_0 \to 0$: आरंभिक बँटवारा चाहे जो हो, $a_n \to 0.6$ और $b_n \to 0.4$।
\end{solution}

\begin{solution}{exo:g12:matrix:8}
\emph{1.} $\det P = -2$, इसलिए $P^{-1} = -\frac12\begin{pmatrix} -1 & -1\\ -1 & 1\end{pmatrix}
= \frac12\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}$। तब
\[
AP = \begin{pmatrix} 4 & 2\\ 4 & -2 \end{pmatrix}, \qquad
D = P^{-1}AP = \frac12\begin{pmatrix} 1&1\\1&-1\end{pmatrix}
\begin{pmatrix} 4&2\\4&-2\end{pmatrix}
= \begin{pmatrix} 4 & 0\\ 0 & 2\end{pmatrix}.
\]

\emph{2.} $A = PDP^{-1}$ से तत्काल आगमन $D^n = \begin{pmatrix} 4^n & 0\\ 0 & 2^n\end{pmatrix}$ वाला $A^n = PD^nP^{-1}$ देता है, इसलिए
\[
A^n = P D^n P^{-1}
= \begin{pmatrix} 4^n & 2^n\\ 4^n & -2^n\end{pmatrix}\cdot
\frac12\begin{pmatrix} 1&1\\1&-1\end{pmatrix}
= \frac12\begin{pmatrix} 4^n + 2^n & 4^n - 2^n\\
4^n - 2^n & 4^n + 2^n\end{pmatrix}.
\]
$X_0 = \begin{pmatrix} u_0\\v_0\end{pmatrix}$ पर $A^n$ लगाने से ठीक \cref{ex:g12:matrix:coupled} के सूत्र फिर मिल जाते हैं।
\end{solution}

\begin{solution}{pb:g12:matrix:1}
\textbf{1.} $AB = \begin{pmatrix} 2 & 1\\ 4 & 3\end{pmatrix}$ और $BA = \begin{pmatrix} 3 & 4\\ 1 & 2\end{pmatrix}$: \omterm{def:g12:matrix:matrix}{आव्यूहों} का गुणन क्रमविनिमेय नहीं है --- $B$ दाईं
ओर से स्तंभ बदलता है और बाईं ओर से पंक्तियाँ।

\textbf{2.} \omterm{def:g11:vect:det}{सारणिक} $6 - 5 = 1$: प्रतिलोम $\begin{pmatrix} 3 & -1\\ -5 & 2\end{pmatrix}$। उसे $\binom{4}{7}$ पर लगाने पर: $x = 12 - 7 = 5$, $y = -20 + 14 = -6$।

\textbf{3.} $N^2 = 0$। फिर आगमन से $(I + N)^n = I + nN$: $(I + nN)(I + N) = I + (n+1)N + nN^2 = I + (n+1)N$।

\textbf{4.} $A = \begin{pmatrix} 0&1&1\\ 1&0&1\\ 1&1&0
\end{pmatrix}$, और $A^3$ की विकर्ण प्रविष्टियाँ $2$ हैं: हर शीर्ष से लंबाई
$3$ की ठीक दो बंद चालें (त्रिभुज दक्षिणावर्त या वामावर्त चला हुआ) --- गिनती
वाली प्रमेय काम पर।

\textbf{5.} $D^n = \begin{pmatrix} 2^n & 0\\ 0 & 2^{-n}
\end{pmatrix}$: एक दिशा फट पड़ती है, दूसरी मर जाती है --- विकर्ण की नियतियाँ
स्वतंत्र \omterm{def:g11:seq:geometric}{गुणोत्तर अनुक्रम} हैं।

\textbf{6.} $F^2 = \begin{pmatrix} 2 & 1\\ 1 & 1\end{pmatrix}$, $F^3 = \begin{pmatrix} 3 & 2\\ 2 & 1\end{pmatrix}$, $F^4 = \begin{pmatrix} 5 & 3\\ 3 & 2\end{pmatrix}$: हर जगह फिबोनाच्ची; अनुमान वैसा ही जैसा बताया
गया।

\textbf{7.} यदि $F^n = \begin{pmatrix} F_{n+1} & F_n\\ F_n &
F_{n-1}\end{pmatrix}$ हो, तो
\[
F^{n+1} = F^n F =
\begin{pmatrix} F_{n+1} + F_n & F_{n+1}\\
F_n + F_{n-1} & F_n \end{pmatrix}
= \begin{pmatrix} F_{n+2} & F_{n+1}\\ F_{n+1} & F_n
\end{pmatrix} :
\]
वंशानुगति; और आधार-स्थिति $n = 1$ स्वयं $F$ है, जिसमें परिपाटी $F_0 = 0$ का उपयोग
होता है (जो पुनरावृत्ति को पीछे की ओर भी बढ़ा देती है)।

\textbf{8.} $\det F = -1$, इसलिए $\det(F^n) = (\det F)^n = (-1)^n$; और सीधे $\det(F^n) = F_{n+1}F_{n-1} - F_n^2$: कासीनी, एक पंक्ति में। ($2 \times 2$
\omterm{def:g11:vect:det}{सारणिकों} के लिए गुणनफल का नियम पाँच मिनट का सुखद प्रसार है।)

\textbf{9.} $F^m F^n$ की ऊपर-दाईं प्रविष्टि: $F_{m+1}F_n + F_m F_{n-1}$; $F^{m+n}$ की ऊपर-दाईं: $F_{m+n}$।
$m = n = 3$ के लिए: $F_4 F_3 + F_3 F_2 = 3 \times 2 + 2 \times 1 = 8 = F_6$।

\textbf{10.} $k = 1$ के लिए: तुच्छ। यदि $F_n \mid F_{kn}$ हो, तो $m = kn$ के साथ योग सूत्र:
$F_{(k+1)n} = F_{kn+1}F_n + F_{kn}F_{n-1}$: दोनों पद $F_n$ के \omterm{def:g12:arith:divides}{गुणज} हैं। इसलिए सभी $k$ के लिए $F_n \mid F_{kn}$: जाँचिए कि
$F_3 = 2$ $F_6 = 8$ और $F_9 = 34$ को \omterm{def:g12:arith:divides}{विभाजित} करता है।

\textbf{11.} $F^{100} = F^{64} F^{32} F^4$: सात बार वर्ग ($F^2, F^4, \dots, F^{64}$) और दो बार जोड़-मिलान --- यानी
निन्यानबे के बदले नौ गुणन। यह मिस्र के लिपिकों की दुगुना करने वाली सारणी की
तरकीब ही है, जो संख्याओं से उठकर \omterm{def:g12:matrix:matrix}{आव्यूहों} पर आ गई: $100$ को द्विआधारी में
लिखिए और जिन दुगुनों की ज़रूरत हो उन्हीं को गुणा कीजिए।

\textbf{12.} $0.8 + 0.2 = 1$ और $0.4 + 0.6 = 1$: कल कोई न कोई मौसम होगा \emph{ही} --- हर स्तंभ
एक पूरा \omterm{def:g10:proba:distribution}{प्रायिकता} \omterm{def:g11:prob:rv}{बंटन} है, इसलिए \omterm{def:g10:proba:distribution}{प्रायिकता} संरक्षित रहती है।

\textbf{13.} कल: $\binom{0.8}{0.2}$। परसों: $M\binom{0.8}{0.2} = \binom{0.72}{0.28}$।

\textbf{14.} $v = \binom{s}{r}$, $s + r = 1$ के साथ $Mv = v$: $0.8s + 0.4r = s$ से $0.4r = 0.2s$, $s = 2r$ मिलते हैं:
$v = \binom{2/3}{1/3}$। दीर्घ काल में तीन में से दो दिन धूप वाले होते हैं --- आज चाहे जैसा
दिखे।

\textbf{15.} $\binom01$ से: धूप वाले घटक $0.4$, $0.56$, $0.624$, $0.6496$; और $\frac23$ से
अंतर: $0.267$, $0.107$, $0.043$, $0.017$ --- हर क़दम अंतर को ठीक $0.4$ गुना कर देता
है (मशीन का दूसरा अभिलक्षणिक मान): स्थायी अवस्था पर \omterm{def:g12:seq:arith-geom}{गुणोत्तर} अभिसरण।

\textbf{16.} स्तंभ ($A$, $B$, $C$ से): $P = \begin{pmatrix} 0 & 0 & 1\\ \frac12 & 0 & 0\\
\frac12 & 1 & 0\end{pmatrix}$। स्थायी अवस्था: $v_A = v_C$,
$v_B = \frac{v_A}{2}$, $v_C = \frac{v_A}{2} + v_B$; और योग $1$ होने पर: $v = \left(\frac25, \frac15, \frac25\right)$। श्रेणी: $A$ तथा $C$ पहले
स्थान पर बराबरी में, और $B$ अंत में।

\textbf{17.} $C$ $B$ का \emph{सारा} आवागमन और $A$ का आधा पाता है, और वह
सब कुछ $A$ को लौटा देता है: स्थायी अवस्था यह नापती है कि पाठक \emph{समय
कहाँ बिताता है}, न कि उसकी ओर कितनी कड़ियाँ आती हैं --- किसी लोकप्रिय पृष्ठ
से आई एक कड़ी उजाड़ पृष्ठों की कई कड़ियों पर भारी पड़ती है। यही पुनरावर्ती
भारण उस खोज-इंजन का संस्थापक विचार है; और अवमंदन मकड़-जालों तथा बंद गलियों को
सँभालता है।

\textbf{18.} $A^n = P D^n P^{-1}$ से $u_n = \frac{4^n + 2^n}{2}$ मिलता है (और $v_n = \frac{4^n - 2^n}{2}$)। जाँच: $u_1 = 3$, $u_2 = 10$, $u_3 = 36$;
सीधे: $(1,0) \to (3,1) \to (10,6) \to
(36, 28)$: मेल।

\textbf{19.} विकर्णीकरण ऐसे \omterm{def:g10:coordgeom:system}{निर्देशांकों} पर ले जाता है जिनमें युग्मित तंत्र
टूटकर स्वतंत्र \omterm{def:g11:seq:geometric}{गुणोत्तर अनुक्रमों} में बिखर जाता है --- हर अभिलक्षणिक मान अपनी
दौड़ अलग दौड़ता है। मार्कोव की स्थायी अवस्था अभिलक्षणिक मान $1$ का अभिलक्षणिक
\omterm{def:g11:vect:vector}{सदिश} है, और प्रश्न 15 की अभिसरण-दर अगला अभिलक्षणिक मान: मौसम की मशीन शुरू से
ही अभिलक्षणिक कहानी थी।

\textbf{20.} बही-खाता: कोई निकाय एक ही आव्यूह-समीकरण है, जो एक ही प्रतिलोम से
हल हो जाता है। क्रमचय-संचय: \omterm{def:g12:matrix:graph}{आसन्नता आव्यूह} की घातें चालें, कड़ियाँ और संबंध
गिन देती हैं। विकास: मशीन की घातें अवस्थाओं को उनकी नियतियों तक ले जाती हैं,
और अभिलक्षणिक दिशाएँ (फिबोनाच्ची की स्वर्ण दिशा, मौसम की स्थायी अवस्था, जाल का
श्रेणी-सदिश) ही वे नियतियाँ हैं। स्नातक खंडों का रैखिक बीजगणित ठीक इसी का
विज्ञान है।
\end{solution}
