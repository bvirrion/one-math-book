\begin{solution}{exo:g12:integ:1}
\[
\int_1^2 \left(3x^2 - \frac{1}{x^2}\right)\dd x
= \left[x^3 + \frac1x\right]_1^2 = \left(8 + \tfrac12\right) - 2 = \frac{13}{2}.
\]
\[
\int_0^{\pi/2} \cos t\,\dd t = \bigl[\sin t\bigr]_0^{\pi/2} = 1 .
\]
$u = 2t+1$, $\frac{1}{2t+1} = \frac12\,\frac{u'}{u}$ के साथ:
\[
\int_0^1 \frac{\dd t}{2t+1} = \frac12\bigl[\ln(2t+1)\bigr]_0^1
= \frac{\ln 3}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:2}
$x\,\eu^{x^2} = \frac12\,(x^2)'\,\eu^{x^2}$ के \omterm{def:g12:integ:primitive}{प्रतिअवकलज} $\frac12\eu^{x^2} + c$ हैं। शर्त $F(0) = 1$ से $\frac12 + c = 1$ मिलता है, इसलिए
\[
F(x) = \frac{\eu^{x^2} + 1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:3}
\[
\mu = \frac{1}{\pi}\int_0^\pi \sin t\,\dd t
= \frac{1}{\pi}\bigl[-\cos t\bigr]_0^\pi = \frac{2}{\pi} \approx 0.64 .
\]
\omterm{def:g10:functions:graph}{आलेख} की दृष्टि से: आधार $\intcc{0}{\pi}$ और ऊँचाई $\frac{2}{\pi}$ वाले आयत का क्षेत्रफल ज्या-वक्र के
मेहराब जितना ही है।
\end{solution}

\begin{solution}{exo:g12:integ:4}
$u' = 1$, $v = \ln t$ लीजिए, इसलिए $u = t$, $v' = \frac1t$:
\[
\int_1^{\eu} \ln t\,\dd t = \bigl[t\ln t\bigr]_1^{\eu} - \int_1^{\eu} 1\,\dd t
= \eu - (\eu - 1) = 1 .
\]
$u' = \sin t$, $v = t$ लीजिए, इसलिए $u = -\cos t$, $v' = 1$:
\[
\int_0^\pi t\sin t\,\dd t = \bigl[-t\cos t\bigr]_0^\pi + \int_0^\pi \cos t\,\dd t
= \pi + 0 = \pi .
\]
\end{solution}

\begin{solution}{exo:g12:integ:5}
वक्र वहाँ मिलते हैं जहाँ $x^2 = x + 2$, \emph{अर्थात्} $x \in \{-1, 2\}$, और $\intcc{-1}{2}$ पर रेखा \omterm{def:g11:quad:parabola}{परवलय} के
ऊपर है। क्षेत्रफल है
\[
\int_{-1}^{2} \bigl(x + 2 - x^2\bigr)\dd x
= \left[\frac{x^2}{2} + 2x - \frac{x^3}{3}\right]_{-1}^{2}
= \frac{10}{3} - \left(-\frac{7}{6}\right) = \frac{9}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:6}
\emph{1.} $I = \bigl[\ln(1+t)\bigr]_0^1 = \ln 2$।

\emph{2.}
\[
I_n + I_{n+1} = \int_0^1 \frac{t^n(1 + t)}{1+t}\,\dd t
= \int_0^1 t^n \,\dd t = \frac{1}{n+1}.
\]
$\intcc{0}{1}$ पर $0 \leq \dfrac{t^n}{1+t} \leq t^n$, इसलिए तुलना से $0 \leq I_n \leq \frac{1}{n+1}$।

\emph{3.} पुनरावृत्ति से, क्रमिक निरसन वाले आगमन द्वारा,
\[
I_n = (-1)^n\left(I_0 - \left(1 - \frac12 + \dots + \frac{(-1)^{n-1}}{n}\right)\right),
\]
\emph{अर्थात्}
$\displaystyle\sum_{k=1}^{n} \frac{(-1)^{k-1}}{k} = \ln 2 - (-1)^n I_n$
($I_0 = \ln 2$ याद कीजिए)। चूँकि $\abs{I_n} \leq \frac{1}{n+1} \to 0$ है, इसलिए एकांतर हरात्मक योग $\ln 2$ की ओर जाता है।
\end{solution}

\begin{solution}{exo:g12:integ:7}
दूरी:
\[
d = \int_0^{100} 30\left(1 - \eu^{-t/50}\right)\dd t
= 30\left[t + 50\,\eu^{-t/50}\right]_0^{100}
= 30\left(100 + 50\eu^{-2} - 50\right)
= 1500 + 1500\,\eu^{-2}.
\]
संख्याओं में $d \approx 1500 + 203 = 1703$ m। औसत चाल: $\frac{d}{100} = 15\bigl(1 + \eu^{-2}\bigr) \approx 17.0$ m/s।
\end{solution}

\begin{solution}{exo:g12:integ:8}
\emph{1.} $S_n$ चौड़ाई $\frac1n$ और ऊँचाइयों $f\left(\frac kn\right)$, $k = 1, \dots, n$ वाले $n$ आयतों का कुल
क्षेत्रफल है, जहाँ $f(t) = \frac{1}{1+t}$: यह $\intcc{0}{1}$ पर $f$ के नीचे के क्षेत्रफल का ``दाएँ सिरे''
वाला आयत-सन्निकटन है।

\emph{2.} $f$ \omterm{def:g10:functions:variations}{ह्रासमान} है, इसलिए हर \omterm{def:g10:numbers:interval}{अंतराल} $\intcc{\frac{k-1}{n}}{\frac kn}$ पर $f\left(\frac kn\right) \leq f(t) \leq f\left(\frac{k-1}{n}\right)$; समाकलन करके $k$
पर जोड़ने पर:
\[
S_n \leq \int_0^1 f(t)\,\dd t \leq S_n + \frac1n\bigl(f(0) - f(1)\bigr)
= S_n + \frac{1}{2n}.
\]
(अभ्यास के प्रदर्शन का बीच वाला पद, $f(0) - f(1) = 1 - \frac12$ के साथ, ठीक यही परिबंध है।) अतः
$0 \leq \ln 2 - S_n \leq \frac{1}{2n} \to 0$, इसलिए $S_n \to \ln 2$।
\end{solution}

\begin{solution}{exo:g12:integ:9}
\emph{1.} $W_0 = \int_0^{\pi/2} \dd t = \frac{\pi}{2}$ और $W_1 = \bigl[-\cos t\bigr]_0^{\pi/2} = 1$।

\emph{2.} $u' = \sin t$, $v = \sin^{n+1} t$ के साथ \omterm{thm:g12:integ:ibp}{खंडशः समाकलन} कीजिए, इसलिए $u = -\cos t$, $v' = (n+1)\sin^n t \cos t$:
\[
W_{n+2} = \bigl[-\cos t \sin^{n+1} t\bigr]_0^{\pi/2}
+ (n+1)\int_0^{\pi/2} \cos^2 t\,\sin^n t\,\dd t
= (n+1)\int_0^{\pi/2} (1 - \sin^2 t)\sin^n t\,\dd t,
\]
जहाँ कोष्ठक दोनों सिरों पर लुप्त हो जाता है। अतः $W_{n+2} = (n+1)(W_n - W_{n+2})$, \emph{अर्थात्} $W_{n+2} = \frac{n+1}{n+2} W_n$।

\emph{3.} $W_2 = \frac12 W_0 = \frac{\pi}{4}$, $W_3 = \frac23 W_1 = \frac23$, $W_4 = \frac34 W_2 = \frac{3\pi}{16}$। $\intoo{0}{\frac\pi2}$ पर $0 < \sin t < 1$, इसलिए $\sin^{n+1} t \leq \sin^n t$, और भीतर कड़ी असमिका के
साथ; समाकलन करने पर $0 < W_{n+1} \leq W_n$ (वस्तुतः $<$): $(W_n)$ \omterm{def:g10:functions:variations}{ह्रासमान} और धनात्मक है।
\end{solution}

\begin{solution}{pb:g12:integ:1}
\textbf{1.} $\left[x^3 - x^2 + x\right]_0^1 = 1$; \quad $\left[\ln x\right]_1^{\eu} = 1$; \quad $\left[\sin x\right]_0^{\pi/2} = 1$।

\textbf{2.} $\left[\frac12 \eu^{x^2}\right]_0^1 =
\frac{\eu - 1}{2}$; \quad $\left[\ln(x^2 + 1)\right]_0^1 = \ln 2$।

\textbf{3.} $\int_0^1 x\eu^x = \left[x\eu^x\right]_0^1 -
\int_0^1 \eu^x = \eu - (\eu - 1) = 1$। और $\int_1^{\eu} \ln x = \left[x\ln x\right]_1^{\eu} -
\int_1^{\eu} 1 = \eu - (\eu - 1) = 1$।

\textbf{4.} $\frac{1}{\pi}\int_0^\pi \sin x\,\dd x =
\frac{2}{\pi} \approx 0.64$: ज्या-मेहराब की औसत ऊँचाई $\frac2\pi$ है, जो आधे से साफ़ अधिक है ---
मेहराब अपनी चोटी के पास मोटा है।

\textbf{5.} $\int_0^1 (x - x^2)\,\dd x = \frac12 - \frac13 =
\frac16$।

\textbf{6.} $L_n = \frac1n \sum_{k=0}^{n-1}
\left(\frac kn\right)^2$ और $U_n = \frac1n \sum_{k=1}^{n} \left(\frac kn\right)^2$: चौड़ाई $\frac1n$ वाले आयत, जिनकी ऊँचाइयाँ बाएँ सिरे
(वक्र के नीचे) या दाएँ सिरे (ऊपर) पर पढ़ी गई हैं।

\textbf{7.} $n = 1$ के लिए सत्य ($1 = \frac{1 \cdot 2 \cdot
3}{6}$)। यदि वह $n$ के लिए सही हो, तो $(n+1)^2$
जोड़ने पर
\[
\frac{n(n+1)(2n+1)}{6} + (n+1)^2
= \frac{(n+1)\left(2n^2 + n + 6n + 6\right)}{6}
= \frac{(n+1)(n+2)(2n+3)}{6},
\]
जो $n + 1$ पर वही सूत्र है: वंशानुगति, हो गया।

\textbf{8.} $U_n = \frac{1}{n^3} \cdot
\frac{n(n+1)(2n+1)}{6} = \frac{(1 + \frac1n)(2 + \frac1n)}{6}
\to \frac26 = \frac13$, और $L_n = U_n - \frac1n \to \frac13$: अपनी सीढ़ियों के बीच दबा हुआ क्षेत्रफल ठीक $\frac13$
है।

\textbf{9.} $\int_0^1 x^2\,\dd x =
\left[\frac{x^3}{3}\right]_0^1 = \frac13$: एक ही पंक्ति। मूल प्रमेय अनंत शिकंजे को एक ही \omterm{def:g12:integ:primitive}{प्रतिअवकलज}
के मान-निर्धारण में बदल देती है --- इसीलिए उसे \omterm{def:g11:quad:discriminant}{मूल} कहा जाता है।

\textbf{10.} खंड: $\int_{-1}^{1}(1 - x^2)\,\dd x =
2 - \frac23 = \frac43$। आधार जीवा (लंबाई $2$, ऊँचाई $1$ पर) और शीर्ष
$(0,0)$ वाला त्रिभुज: क्षेत्रफल $\frac12 \times 2 \times 1 = 1$। अनुपात: $\frac43$ --- ठीक आर्किमिडीज़ की
प्रमेय, उसी मशीन से सत्यापित जो उनके पास नहीं थी।

\textbf{11.} $T\left(1 + \frac14 + \frac{1}{16} + \dots\right)
= T \cdot \frac{1}{1 - \frac14} = \frac43 T$ (अनुपात $\frac14$ वाली \omterm{def:g12:seq:arith-geom}{गुणोत्तर} श्रेणी): $T = 1$ के साथ खंड का
क्षेत्रफल फिर $\frac43$। आर्किमिडीज़ ने यह श्रेणी शुद्ध ज्यामितीय तर्क से जोड़ी
थी --- चॉकलेट जैसी चौथाई-चौथाई कुतरन, सीमाओं के शब्द बनने से दो सहस्राब्दी
पहले।

\textbf{12.} निःशेषण को कोई चतुर ज्यामितीय स्व-समरूपता चाहिए और वह स्थिति-दर-स्थिति
यथार्थ क्षेत्रफल देती है; रीमान को केवल एकदिष्टता चाहिए और वह योग निकालने की
क़ीमत पर सार्वभौमिक परिभाषा देता है; और मूल प्रमेय को \omterm{def:g12:deriv:derivative}{अवकलजों} का पूरा भवन चाहिए
तथा वह उसकी भरपाई क्षेत्रफलों को एक-पंक्ति की गणना बनाकर कर देती है।

\textbf{13.} $A(1) = 0$ और $A'(x) = \frac1x$ (मूल प्रमेय): $A$ $\intoo{0}{+\infty}$ पर $\frac1x$ का वह
\omterm{def:g12:integ:primitive}{प्रतिअवकलज} है जो $1$ पर लुप्त होता है --- और वही ठीक \omterm{def:g12:exp:ln}{प्राकृत लघुगणक} है।

\textbf{14.} $g'(x) = \frac{a}{ax} - \frac1x = 0$: $g$ अचर है; $x = 1$ पर: $g(1) = A(a) - A(1) = A(a)$। अतः सभी $x$ के लिए $A(ax) = A(a) + A(x)$,
और $x = b$ के साथ: $\ln(ab) = \ln a + \ln b$। \omterm{def:g11:func:reference}{अतिपरवलय} का क्षेत्रफल क्षैतिज पैमाने की परवाह नहीं
करता: $t$ को $a$ गुना खींचने पर $\frac1t$ $a$ गुना दब जाता है, और क्षेत्रफल
बच जाता है --- \omterm{def:g12:exp:ln}{लघुगणकों} का जुड़ना मापन की सममिति ही है।

\textbf{15.} दोहराने पर: $\ln(a^n) = n\ln a$ ($n$ पर आगमन)। और $0 = \ln 1 = \ln\left(a \cdot \frac1a\right) =
\ln a + \ln\frac1a$: $\ln\frac1a = -\ln a$।

\textbf{16.} $\frac{1}{11} + \frac{1}{12} + \dots +
\frac{1}{20} \approx 0.669$, जबकि $\ln 2 \approx 0.693$: $\frac{1}{1+t}$ के नीचे दस आयत, और पहले ही बहुत पास।
हरात्मक श्रेणी अपसरित होती है, पर $n$ से $2n$ तक उसके \emph{टुकड़े} $\ln 2$
पर आ बैठते हैं: घिसटता हुआ अपसरण, जिसे एक क्षेत्रफल नाप लेता है।

\textbf{17.} दूरी: $\int_0^{10} 3t^2\,\dd t = 1000$ m। \omterm{def:g11:stat:mean}{माध्य} चाल: $100$ m/s। $v = 100$ वाला क्षण: $3t^2 = 100$:
$t = \sqrt{\frac{100}{3}} \approx 5.77$ s।

\textbf{18.} द्रव्यमान: $\int_0^4 (2 + x)\,\dd x = 8 + 8 = 16$ kg। आघूर्ण: $\int_0^4 (2x + x^2)\,\dd x = 16 + \frac{64}{3} =
\frac{112}{3}$। द्रव्यमान केंद्र: $\frac{112/3}{16} = \frac73 \approx 2.33$ m ---
भारी दाहिने सिरे ने उसे बीच से आगे धकेल दिया, जैसा अंतर्ज्ञान माँगता है।

\textbf{19.} $W_0 = \frac\pi2$; $W_1 = \left[-\cos
t\right]_0^{\pi/2} = 1$; और $W_2 = \int_0^{\pi/2} \frac{1 - \cos 2t}{2}\,\dd t =
\frac\pi4$।

\textbf{20.} क्षेत्रफल: परिभाषा, जो सीढ़ियों के बीच दबी है। संचयन: चालों से
दूरियाँ, घनत्वों से द्रव्यमान --- \omterm{def:g12:integ:area}{समाकल} एक चलते जोड़ के रूप में। \omterm{def:g12:integ:primitive}{प्रतिअवकलज}:
मूल प्रमेय, जो आर्किमिडीज़ के धैर्य में से न्यूटन और लाइबनिज़ का छोटा रास्ता
है। और मोती: $\ln$ का बटन $\frac1t$ के नीचे का क्षेत्रफल है, और उस क्षेत्रफल की
मापन-सममिति ही सर्वसमिका $\ln(ab) = \ln a + \ln b$ है।
\end{solution}
