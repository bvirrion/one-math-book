\section*{अध्याय \ref{ch:g12:sums} --- यादृच्छिक चरों के योग और बृहत् संख्याओं का नियम}

\begin{solution}{exo:g12:sums:1}
हर पासे की \omterm{def:g12:randvar:exp}{प्रत्याशा} $\frac{1 + 2 + \dots + 6}{6} = \frac{7}{2}$ है, इसलिए रैखिकता से $\E(S) = \frac72 + \frac72 = 7$। पासे \omterm{def:g12:sums:indep}{स्वतंत्र} हैं, अतः
$\V(S) = \frac{35}{12} + \frac{35}{12} = \frac{35}{6} \approx 5.83$।
\end{solution}

\begin{solution}{exo:g12:sums:2}
$\E(X) = 50$ और $\V(X) = 100 \times 0.25 = 25$।

मार्कोव ($X \geq 0$): $\P(X \geq 75) \leq \frac{50}{75} = \frac23$।

ब्येनेमे--चेबिशेव: $\P(\abs{X - 50} \geq 25) \leq \frac{25}{25^2} = 0.04$ --- और यह तो \emph{द्विपक्षीय} \omterm{def:g11:prob:model}{घटना} को \omterm{def:g12:seq:bounded}{परिबद्ध} करता
है, जिसका आधा हिस्सा $X \geq 75$ है। यहाँ चेबिशेव कहीं पैना है क्योंकि वह केवल \omterm{def:g11:stat:mean}{माध्य}
नहीं, \omterm{def:g12:randvar:exp}{प्रसरण} का भी उपयोग करता है। ($\P(X \geq 75)$ का सच्चा मान $10^{-6}$ से छोटा है: दोनों
परिबंध मोटे हैं।)
\end{solution}

\begin{solution}{exo:g12:sums:3}
$F_n$ $\left(\frac12\right)$ वाले $n$ बर्नूली चरों का \omterm{prop:g12:sums:mean}{प्रतिदर्श माध्य} है, जिनका \omterm{def:g12:randvar:exp}{प्रसरण} $\sigma^2 = \frac14$ है।
परिबंध
\[
\P\left(\abs{F_n - 0.5} \geq 0.05\right)
\leq \frac{1/4}{n \times 0.05^2} = \frac{100}{n}
\]
$n \geq 2000$ होते ही $\leq 0.05$ हो जाता है।
\end{solution}

\begin{solution}{exo:g12:sums:4}
\cref{prop:g12:sums:mean} से $\E(M_n) = 5\%$ (विविधीकरण प्रत्याशित प्रतिफल नहीं बदलता) और $\sigma(M_n) = \frac{20\%}{\sqrt n}$। $\frac{20}{\sqrt n} < 4$ माँगने
पर $\sqrt n > 5$, \emph{अर्थात्} $n \geq 26$ मिलता है। यही \emph{विविधीकरण} का सिद्धांत है:
पूँजी को \omterm{def:g12:sums:indep}{स्वतंत्र} जोखिमों पर फैला देने से जोखिम (\omterm{def:g11:stat:variance}{मानक विचलन}) $\sqrt n$ से बँट जाता
है और प्रत्याशित प्रतिफल घटता नहीं।
\end{solution}

\begin{solution}{exo:g12:sums:5}
\emph{1.} $9$ समसंभावी युग्मों को गिनने पर: $S$ मान $2, 3, 4, 5, 6$ लेता है, जिनकी
\omterm{def:g10:proba:distribution}{प्रायिकताएँ} $\frac19, \frac29, \frac39, \frac29, \frac19$ हैं। अतः $\E(S) = \frac{2 + 6 + 12 + 10 + 6}{9} = 4$ और $\E(S^2) = \frac{4 + 18 + 48 + 50 + 36}{9} = \frac{156}{9} = \frac{52}{3}$, इसलिए $\V(S) = \frac{52}{3} - 16 = \frac43$।

\emph{2.} एक चर के लिए: $\E(X) = 2$, $\E(X^2) = \frac{1 + 4 + 9}{3} = \frac{14}{3}$, $\V(X) = \frac{14}{3} - 4 = \frac23$। तब $\E(S) = 2 + 2 = 4$ और, \omterm{def:g12:condprob:indep}{स्वतंत्रता} से, $\V(S) = \frac23 + \frac23 = \frac43$।
वही मान।
\end{solution}

\begin{solution}{exo:g12:sums:6}
$Y = X$ के साथ: $\V(X + Y) = \V(2X) = 4\V(X)$, जबकि $\V(X) + \V(Y) = 2\V(X)$। दोनों केवल तभी मेल खाते हैं जब $\V(X) = 0$ हो,
\emph{अर्थात्} जब $X$ अचर हो --- इसलिए योज्यता सचमुच \omterm{def:g12:condprob:indep}{स्वतंत्रता} माँगती है
(यहाँ $X$ स्वयं पर \omterm{def:g10:functions:extrema}{अधिकतम} रूप से निर्भर है)।
\end{solution}

\begin{solution}{exo:g12:sums:7}
न्यायसंगतता के अधीन $F_n$ $p = \frac16$, $\sigma^2 = \frac16\cdot\frac56 = \frac{5}{36}$ वाले $n = 1200$ बर्नूली चरों का \omterm{def:g11:stat:mean}{माध्य} है:
\[
\P\left(\abs{F_n - \tfrac16} \geq 0.05\right)
\leq \frac{5/36}{1200 \times 0.0025} = \frac{5}{108} \approx 0.046 .
\]
प्रेक्षित विचलन ठीक $0.05$ है: ऐसी \omterm{def:g11:prob:model}{घटना} जिसे न्यायसंगत पासा अधिक से अधिक $4.6\%$
\omterm{def:g10:proba:distribution}{प्रायिकता} से देता है --- और चूँकि चेबिशेव बहुत ही रूढ़िवादी है, सच्ची \omterm{def:g10:proba:distribution}{प्रायिकता}
कहीं छोटी है। न्यायसंगतता की परिकल्पना विश्वसनीय नहीं है; पासा बहुत संभवतः
भारित है।
\end{solution}

\begin{solution}{exo:g12:sums:8}
\emph{1.} $p(1-p) = \frac14 - \left(p - \frac12\right)^2 \leq \frac14$।

\emph{2.} $F_n$ की \omterm{def:g12:randvar:exp}{प्रत्याशा} $p$ और \omterm{def:g12:randvar:exp}{प्रसरण} $\frac{p(1-p)}{n} \leq \frac{1}{4n}$ है; ब्येनेमे--चेबिशेव देता है
\[
\P\bigl(\abs{F_n - p} \geq \varepsilon\bigr)
\leq \frac{p(1-p)}{n\varepsilon^2} \leq \frac{1}{4n\varepsilon^2},
\]
जो अज्ञात $p$ चाहे जो हो, वैध रहता है।

\emph{3.} $\varepsilon = 0.03$ और स्तर $0.05$ के साथ:
\[
\frac{1}{4n(0.03)^2} \leq 0.05
\iff n \geq \frac{1}{4 \times 0.0009 \times 0.05} \approx 5556 .
\]
इस मोटे परिबंध से लगभग $5600$ लोग पर्याप्त हैं (चिरपरिचित प्रसामान्य-सन्निकटन
वाला उत्तर $1100$ के पास है, \cref{ch:g12:contdist} देखिए)।
\end{solution}

\begin{solution}{pb:g12:sums:1}
\textbf{1.} $V(X + Y) = V(X) + V(Y) = \frac{35}{12} +
\frac{35}{12} = \frac{35}{6}$ (\omterm{def:g12:condprob:indep}{स्वतंत्रता})।

\textbf{2.} $\E(\bar X) = 3.5$; $\sigma(\bar X) = \frac{1.71}{\sqrt{100}} \approx 0.17$: सौ पासों का औसत $3.5$ से पाँचवें हिस्से भर अंक के भीतर
चिपका रहता है।

\textbf{3.} $\P(X \geq 10) \leq \frac{2}{10} = 0.2$।

\textbf{4.} $\P \leq \frac{V(\bar X)}{0.5^2} =
\frac{35/1200}{0.25} \approx 0.117$: अधिक से अधिक लगभग $12\,\%$।

\textbf{5.} $Y = -X$ के साथ: $V(X + Y) = V(0) = 0$, जबकि $V(X) + V(Y) = 2V(X) > 0$: पूरी तरह विपरीत-सहसंबंधित चर एक-दूसरे
को काट देते हैं --- \omterm{def:g12:randvar:exp}{प्रसरण} का जुड़ना \omterm{def:g12:condprob:indep}{स्वतंत्रता} का विशेषाधिकार है।

\textbf{6.} $\E = \frac{18}{37} - \frac{19}{37} =
-\frac{1}{37} \approx -0.027$; $\sigma = \sqrt{1 - \left(\frac{1}{37}\right)^2} \approx
1.00$।

\textbf{7.} $\E(G) = -2.70$, $\sigma(G) = 10 \times 1.00 =
10$: बहाव केवल $0.27\sigma$ है। रात का शोर बढ़त को बौना कर देता है
--- जुआरियों का एक बड़ा अल्पमत (घंटी के अनुसार मोटे तौर पर $40\,\%$) जीतकर बाहर
निकलता है, और ठीक यही उन्हें लौटाता रहता है।

\textbf{8.} जुआघर की वसूली: \omterm{def:g12:randvar:exp}{प्रत्याशा} $+27\,027$ यूरो, \omterm{def:g11:stat:variance}{मानक विचलन} $\approx 1\,000$: लाभ शून्य
से $27$ \omterm{def:g11:stat:variance}{मानक विचलन} ऊपर बैठा है। $27\sigma$ पर ``जुआघर इस वर्ष हार भी सकता है'' कोई
जोखिम नहीं, केवल पूर्णांकन की त्रुटि है।

\textbf{9.} जुआरियों के लिए $\P(\text{हानि}) = \P(G \geq 27\,027$ $)$ $\leq \frac{10^6 \times 1}{27\,027^2} \approx
0.0014$: पुस्तक की सबसे भोंथरी असमिका भी
जुआघर को $99.86\,\%$ पर गारंटी दे देती है --- सच्चाई तो खगोलीय रूप से और भी मज़बूत
है।

\textbf{10.} दाँव पर लगा हर यूरो, चाहे कभी और चाहे कैसे भी चुना गया हो, अपना
$-\frac{1}{37}$ प्रत्याशित प्रतिफल देता है; और रैखिकता से कुल प्रत्याशित लाभ $-\frac{1}{37} \times
(\text{कुल दाँव})$ है, जो
कुछ भी दाँव पर लगाने वाली हर रणनीति के लिए ऋणात्मक है। कोई जिताऊ पद्धति
\omterm{def:g12:randvar:exp}{प्रत्याशा} की रैखिकता का उल्लंघन करेगी --- बुरे दाँवों को चतुराई से क्रम में
लगाने से वे अच्छे नहीं हो जाते। (दुगुना करने वाली पद्धतियाँ बस अनेक छोटी जीतें
देकर विरली विनाशकारी हार ख़रीदती हैं।)

\textbf{11.} प्रमेय: $n$ चक्करों पर लाल की \emph{\omterm{def:g10:stats:series}{आपेक्षिक बारंबारता}}
(\omterm{def:g10:proba:distribution}{प्रायिकता} में) $\frac{18}{37}$ पर \omterm{def:g12:seq:limit}{अभिसरित} होती है। वह चक्कर $n + 1$ के बारे में कुछ नहीं
कहती: पहिये की कोई स्मृति नहीं है, और दस लाल के बाद भी लाल की संभावना $\frac{18}{37}$
ही रहती है। इसके उलट मानना ही \emph{जुआरी की भ्रांति} है, और प्रश्न 12 दिखाता
है कि शृंखलाओं के साथ असल में होता क्या है।

\textbf{12.} $100$ लालों का अधिशेष लौटाया नहीं जाता --- आगे के चक्कर
न्यायसंगत प्रतियाँ हैं और प्रत्याशित अधिशेष $100$ ही रहता है। पर $\frac{100}{10^6} = 0.0001$: एक
अंक का सौवाँ हिस्सा। भ्रांति की भूल एक वाक्य में: \emph{बृहत् संख्याओं का
नियम बीते संयोगों को नए परीक्षणों के महासागर में घोल देता है; वह पहिये को
क़र्ज़ वसूलने कभी नहीं भेजता।}

\textbf{13.} $n \geq \frac{1}{4 \times 0.05 \times 0.03^2}
\approx 5\,556$ लोग।

\textbf{14.} $\frac{1.96^2 \times 0.25}{0.03^2} \approx
1\,067$: पाँच गुना कम। चेबिशेव \emph{हर} \omterm{def:g12:randvar:rv}{बंटन} के लिए सही रहता है
और अपनी सार्वभौमिकता की क़ीमत ढील देकर चुकाता है; सर्वेक्षकों का अचर $1.96$
उतार-चढ़ाव की असली घंटी-आकृति से आता है, जो कहीं कसकर सिमटती है।

\textbf{15.} सीमा $\propto \frac{1}{\sqrt n}$: उसे आधा करने पर \omterm{def:g10:proba:sample}{प्रतिदर्श} चौगुना हो जाता है ---
$\pm 1.5$ अंकों के लिए लगभग $4\,400$ लोग। यथार्थता वर्ग वाली क़ीमत पर ख़रीदी जाती
है।

\textbf{16.} प्रतिरूप सफलता-प्रायिकता $p$ वाले $n$ \omterm{def:g12:sums:indep}{स्वतंत्र} निकाले हैं ---
और समष्टि का आकार $N$ कहीं आता ही नहीं (किसी बड़ी, अच्छी तरह मिली हुई
समष्टि के नन्हे अंश का प्रतिदर्शन ही तो \omterm{def:g12:condprob:indep}{स्वतंत्रता} में समाया है)। सर्वेक्षकों
की उपमा: सूप चखने के लिए एक अच्छी तरह हिलाया हुआ चम्मच पर्याप्त है --- चाहे
पतीला दस लोगों का हो या दस हज़ार का। पेंच हिलाने में है --- अभिनत प्रतिदर्श-ढाँचा,
जैसा माध्यमिक विद्यालय खंड के उस सर्वेक्षण में था जिसने पूरे देश को छका दिया
--- पतीले में नहीं।

\textbf{17.} $315\sqrt n < 30n \iff \sqrt n > 10.5 \iff
n > 110$: सौ बीमा-पत्रों से आगे बहाव शोर से आगे निकल जाता है, और हर
अगला बीमा-पत्र $\sqrt n$ वाले नियम से अंतर और चौड़ा कर देता है --- साझा करना
\emph{ही} यह धंधा है।

\textbf{18.} \omterm{def:g12:sums:indep}{स्वतंत्र} बराबर हिस्से: $\sigma_{\text{pf}} =
\frac{20\,\%}{\sqrt{25}} = 4\,\%$: वही प्रत्याशित प्रतिफल, और
उतार-चढ़ाव पाँचवाँ हिस्सा। दावत यही है: \omterm{def:g12:randvar:exp}{प्रत्याशा} में शून्य क़ीमत पर
जोखिम-कमी --- \omterm{def:g12:sums:indep}{स्वतंत्र} यादृच्छिकता का औसत ले लेना।

\textbf{19.} पूरी तरह सहसंबंधित: निवेश-संचय $25$ वेशों में एक ही शेयर है:
$\sigma = 20\,\%$, कोई कमी नहीं। विविधीकरण \emph{\omterm{def:g12:condprob:indep}{स्वतंत्रता}} किराए पर लेता है; और जब कोई
संकट सब कुछ सहसंबंधित कर देता है (2008: मकानों के सारे दाँव एक ही दाँव थे),
तब $\sqrt n$ वाला कवच ठीक उसी समय उड़ जाता है जब उसकी सबसे अधिक ज़रूरत होती है।

\textbf{20.} योग $n$ की तरह बहते हैं और $\sqrt n$ की तरह उतार-चढ़ाव करते हैं,
इसलिए \omterm{def:g11:stat:mean}{माध्य} $\frac{1}{\sqrt n}$ की तरह जम जाते हैं: जुआघर दस लाख चक्करों पर बहाव जमा कर
लेता है; सर्वेक्षक फ़ोन के बजट में $\frac{1}{\sqrt{1100}}$ का शोर ख़रीद लेता है; बीमाकर्ता $n > 110$
पर अपने ही दावों से आगे निकल जाता है; और निवेशक जोखिम को $\sqrt{25}$ से बाँट लेता
है। दुरुपयोगकर्ता: वह जुआरी जो क़र्ज़ में विश्वास करता है (वहाँ केवल तनुकरण
है), और वह वित्तकर्मी जो \omterm{def:g12:condprob:indep}{स्वतंत्रता} में विश्वास करता है (वहाँ कभी-कभी केवल एक
ही दाँव होता है)। उतार-चढ़ाव की सार्वभौमिक आकृति --- घंटी --- \omterm{def:g12:limcont:continuity}{संतत} अध्याय का
सितारा है, और उसकी प्रमेय, केंद्रीय सीमा प्रमेय, स्नातक खंडों की \omterm{def:g10:proba:distribution}{प्रायिकता} का
शिखर।
\end{solution}
