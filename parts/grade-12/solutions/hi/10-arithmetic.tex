\section*{अध्याय \ref{ch:g12:arith} --- अंकगणित}

\begin{solution}{exo:g12:arith:1}
$17 \times 119 = 2023$, इसलिए $2026 = 17 \times 119 + 3$: भागफल $119$, शेषफल $3$। $-2026$ के लिए: $-2026 = 17\times(-120) + 14$ (सचमुच $17 \times 120 = 2040$ और $2040 - 2026 = 14$):
भागफल $-120$, शेषफल $14$ (शेषफल $\intco{0}{17}$ में होना चाहिए, इसलिए वह $-3$
\emph{नहीं} है)।
\end{solution}

\begin{solution}{exo:g12:arith:2}
$10$ के सापेक्ष: $7^2 = 49 \equiv 9 \equiv -1$। अतः $7^{100} = \left(7^2\right)^{50} \equiv (-1)^{50} = 1 \pmod{10}$: $7^{100}$ का अंतिम अंक $1$ है।
\end{solution}

\begin{solution}{exo:g12:arith:3}
यूक्लिड: $1071 = 2\times462 + 147$; $462 = 3\times147 + 21$; $147 = 7\times21 + 0$। इसलिए $\gcd = 21$।

पीछे प्रतिस्थापन: $21 = 462 - 3\times147 = 462 - 3(1071 - 2\times462)
= 7\times462 - 3\times1071$। इस प्रकार $u = -3$, $v = 7$: $1071\times(-3) + 462\times7 = 21$।
\end{solution}

\begin{solution}{exo:g12:arith:4}
हर \omterm{def:g10:numbers:sets}{पूर्णांक} $\equiv 0, 1, 2$ या $3 \pmod 4$ है, और वर्ग करने पर: $0^2 \equiv 0$, $1^2 \equiv 1$, $2^2 = 4 \equiv 0$, $3^2 = 9 \equiv 1$। इसलिए
$n^2 \equiv 0$ या $1 \pmod 4$। तब दो वर्गों का योग $0 + 0$, $0 + 1$ या $1 + 1$ के सर्वांगसम होता है,
\emph{अर्थात्} $0$, $1$ या $2 \pmod 4$ के --- कभी $3$ के नहीं।
\end{solution}

\begin{solution}{exo:g12:arith:5}
$2$ से \omterm{def:g12:arith:divides}{विभाज्यता}: $n$ और $n + 1$ में से एक सम है। $3$ से \omterm{def:g12:arith:divides}{विभाज्यता}: यदि $n \equiv 0$
हो तो $3 \mid n$; यदि $n \equiv 1 \pmod 3$ हो तो $2n + 1 \equiv 3 \equiv 0$; और यदि $n \equiv 2$ हो तो $n + 1 \equiv 0$। हर स्थिति में $3$
गुणनफल को \omterm{def:g12:arith:divides}{विभाजित} करता है। चूँकि $\gcd(2,3) = 1$ है, इसलिए \cref{cor:g12:arith:coprimeprod} $6 \mid n(n+1)(2n+1)$ दे देता है। (इससे
यह भी फिर सिद्ध हो जाता है कि \cref{exo:g12:seq:1} के वर्गों का योग $\frac{n(n+1)(2n+1)}{6}$ \omterm{def:g10:numbers:sets}{पूर्णांक} है।)
\end{solution}

\begin{solution}{exo:g12:arith:6}
हम $11$ के सापेक्ष $5$ का प्रतिलोम ढूँढ़ते हैं: आज़माने से (या बेज़ू से)
$5 \times 9 = 45 = 44 + 1 \equiv 1 \pmod{11}$। \omterm{def:g12:arith:congruence}{सर्वांगसमता} को $9$ से गुणा करने पर:
\[
x \equiv 9 \times 3 = 27 \equiv 5 \pmod{11}.
\]
हल \omterm{def:g10:numbers:sets}{पूर्णांक} $x = 5 + 11k$, $k \in \Z$ हैं। (जाँच: $5\times5 = 25 \equiv 3 \pmod{11}$।)
\end{solution}

\begin{solution}{exo:g12:arith:7}
$\gcd(17, 40) = 1$, इसलिए हल मौजूद हैं। यूक्लिड: $40 = 2\times17 + 6$; $17 = 2\times6 + 5$; $6 = 5 + 1$। पीछे प्रतिस्थापन करने
पर: $1 = 6 - 5 = 6 - (17 - 2\times6) = 3\times6 - 17
= 3(40 - 2\times17) - 17 = 3\times40 - 7\times17$। अतः $17\times(-7) - 40\times(-3) = 1$: विशेष हल $(x_0, y_0) = (-7, -3)$।

$17x - 40y = 1$ का व्यापक हल: विशेष संबंध घटाने पर $17(x + 7) = 40(y + 3)$; और चूँकि $\gcd(17, 40) = 1$ है, गाउस $40 \mid x + 7$
देते हैं, इसलिए $x = -7 + 40k$ और फिर $y = -3 + 17k$, $k \in \Z$ (और ये सब जाँच में खरे उतरते हैं)।

$17x - 40y = 6$ के लिए विशेष हल को $6$ से गुणा कीजिए: $(x_1, y_1) = (-42, -18)$, और उसी तर्क से
\[
x = -42 + 40k, \qquad y = -18 + 17k, \qquad k \in \Z .
\]
(जैसे $k = 2$: $x = 38$, $y = 16$; और सचमुच $17\times38 - 40\times16
= 646 - 640 = 6$।)
\end{solution}

\begin{solution}{exo:g12:arith:8}
मान लीजिए $a, b \in \N^*$ के साथ $\sqrt2 = \frac ab$ है; तब $a^2 = 2b^2$। किसी वर्ग के \omterm{def:g12:arith:prime}{अभाज्य} \omterm{def:g10:algebra:expand}{गुणनखंडन} में हर
घातांक सम होता है; इसलिए $a^2$ में $2$ का घातांक सम है, जबकि $2b^2$ में वह
विषम है (किसी सम संख्या से एक अधिक)। एक ही \omterm{def:g10:numbers:sets}{पूर्णांक} के दो \omterm{def:g10:algebra:expand}{गुणनखंडनों} में $2$
के घातांक भिन्न होना \cref{thm:g12:arith:fta} की अद्वितीयता का खंडन है। अतः ऐसी कोई भिन्न है ही
नहीं: $\sqrt2 \notin \Q$।
\end{solution}

\begin{solution}{exo:g12:arith:9}
\emph{1.} $k\binom pk = p \binom{p-1}{k-1}$ से $p$ $k\binom pk$ को \omterm{def:g12:arith:divides}{विभाजित} करता है। $1 \leq k \leq p-1$ के लिए $p \nmid k$ और $p$
के \omterm{def:g12:arith:prime}{अभाज्य} होने से $\gcd(p, k) = 1$ मिलता है, इसलिए गाउस की प्रमेयिका $p \mid \binom pk$ दे देती है।

\emph{2.} $a$ पर आगमन। $a = 0$ के लिए: $0^p \equiv 0$। मान लीजिए $a^p \equiv a \pmod p$ है। द्विपद
प्रमेय से
\[
(a+1)^p = \sum_{k=0}^{p} \binom pk a^k
\equiv a^p + 1 \pmod p,
\]
जहाँ बिंदु 1 के अनुसार बीच के सारे पद $p$ के सापेक्ष लुप्त हो जाते हैं।
आगमन-परिकल्पना से $(a+1)^p \equiv a + 1 \pmod p$। इससे सभी $a \in \N$ के लिए $a^p \equiv a$ सिद्ध हो जाता है, और
स्थिति $a < 0$ उपयुक्त धनात्मक प्रतिनिधि के रूप में $a \equiv a + kp$ लिखने से निकल आती है।
\end{solution}

\begin{solution}{exo:g12:arith:10}
$n \equiv 2 \pmod 3$ और $n \equiv 3 \pmod 5$: $n = 2 + 3s$ लिखिए; तब $2 + 3s \equiv 3 \pmod 5$, \emph{अर्थात्} $3s \equiv 1 \pmod 5$। $5$ के सापेक्ष
$3$ का प्रतिलोम $2$ है ($3\times2 = 6 \equiv 1$), इसलिए $s \equiv 2 \pmod 5$, मान लीजिए $s = 2 + 5t$, और $n = 8 + 15t$:
पहली दो शर्तों का अर्थ $n \equiv 8 \pmod{15}$ है।

$n \equiv 2 \pmod 7$ जोड़ने पर: $8 + 15t \equiv 2 \pmod 7$, और $15 \equiv 1 \pmod 7$, इसलिए $t \equiv -6 \equiv 1 \pmod 7$, मान लीजिए $t = 1 + 7u$। अतः $n = 23 + 105u$:
\[
n \equiv 23 \pmod{105}.
\]
(जाँच: $23 = 3\times7 + 2 = 5\times4 + 3 = 7\times3 + 2$।)
\end{solution}

\begin{solution}{pb:g12:arith:1}
\textbf{1.} $2026 = 289 \times 7 + 3$: $2026 \equiv 3
\pmod 7$। $10$ के सापेक्ष $7$ की घातें: $7, 9, 3, 1$, जिनका चक्र
$4$ लंबा है; $100 \equiv 0 \pmod 4$: $7^{100}$ का अंतिम अंक $1$ है।

\textbf{2.} $5^2 = 25 \equiv -1 \pmod{13}$, इसलिए $5^{116} = \left(5^2\right)^{58} \equiv (-1)^{58} = 1$ और $5^{117} \equiv 5 \pmod{13}$।

\textbf{3.} $7$ के सापेक्ष $3$ का प्रतिलोम $5$ है ($15 \equiv 1$): $x \equiv 5 \times 5 = 25 \equiv 4 \pmod 7$।

\textbf{4.} $97 = 2 \times 35 + 27$; $35 = 27 + 8$; $27 = 3 \times 8 + 3$; $8 = 2 \times 3 + 2$; $3 = 2 + 1$। पीछे प्रतिस्थापन करने पर:
$1 = 97 \times 13 + 35 \times (-36)$। इसलिए $35 \times (-36) \equiv 1 \pmod{97}$: $35$ का प्रतिलोम $-36 \equiv 61 \pmod{97}$ है।

\textbf{5.} $a$ $n$ के सापेक्ष ठीक तब प्रतिलोमनीय है जब $\gcd(a, n) = 1$ हो: बेज़ू
$au + nv = 1$ दे देते हैं, अर्थात् $au \equiv 1$; और इसके उलट किसी प्रतिलोम के होने से \omterm{def:g12:arith:gcd}{महत्तम
समापवर्तक} को $1$ \omterm{def:g12:arith:divides}{विभाजित} करना ही पड़ता है।

\textbf{6.} $0{\cdot}10 + 3{\cdot}9 + 0{\cdot}8 + 6{\cdot}7 +
4{\cdot}6 + 0{\cdot}5 + 6{\cdot}4 + 1{\cdot}3 + 5{\cdot}2 +
2{\cdot}1 = 132 = 12 \times 11 \equiv 0 \pmod{11}$: वैध।

\textbf{7.} योग $1 \leq w \leq 10$ और $1 \leq \abs d \leq 9$ वाले $wd$ से बदल जाता है: चूँकि $11$ \omterm{def:g12:arith:prime}{अभाज्य}
है और किसी भी गुणनखंड को \omterm{def:g12:arith:divides}{विभाजित} नहीं करता, इसलिए वह गुणनफल को भी \omterm{def:g12:arith:divides}{विभाजित} नहीं
कर सकता (\cref{thm:g12:arith:gauss} / \cref{prop:g12:arith:primedivides}): बदला हुआ योग फिर कभी $\equiv 0$ नहीं होता: हर एक-अंकीय
त्रुटि पर ख़तरे की घंटी बज जाती है।

\textbf{8.} सटे हुए अंकों $a, b$ (भार $w + 1, w$) की अदला-बदली योग को $a \neq b$ वाले
$(w+1)b + wa - (w+1)a - wb = b - a \not\equiv 0$ से बदल देती है: पकड़ी गई। रहस्य $11$ की \emph{अभाज्यता} है: $10$ के
सापेक्ष $5 \times 2$ जैसे गुणनफल बिना किसी गुणनखंड के शून्य हुए लुप्त हो जाते हैं,
इसलिए $\pm 2$ की भार-$5$ वाली त्रुटि (या कोई अभागी अदला-बदली) छिपकर निकल सकती
थी।

\textbf{9.} बारहों अंकों का भारित योग: $119$; जाँच-अंक को उसे $10$ के \omterm{def:g12:arith:divides}{गुणज}
तक पूरा करना है: $1$ (पूरा कूट $978\,2940199\,051$)। EAN उन सटी हुई अदला-बदलियों में चूक
जाता है जहाँ $2(a - b) \equiv 0 \pmod{10}$, अर्थात् $\abs{a - b} = 5$: मान लीजिए $2$ और $7$ की अदला-बदली बिना
पकड़ में आए निकल जाती है --- यही मिलनसार \omterm{def:g12:complex:modulus}{मापांक} $10$ की क़ीमत है।

\textbf{10.} दाईं ओर से हर दूसरा अंक दुगुना करके और मोड़कर ($16 \to 7$ इत्यादि)
योग $80 \equiv 0
\pmod{10}$ बनता है: जाँच-कार्ड वैध निकलता है।

\textbf{11.} अभाज्य मापांक के साथ हर भार प्रतिलोमनीय होता है, इसलिए
\emph{सारी} एकल त्रुटियाँ और \emph{सारी} सटी हुई अदला-बदलियाँ पकड़ में आ जाती
हैं --- यही ISBN की विलासिता है; $10$ वाले \omterm{def:g12:complex:modulus}{मापांक} की योजनाएँ मनुष्य के
मित्रवत अंक बनाए रखती हैं और बदले में एक छोटा-सा अंधा कोना स्वीकार कर लेती
हैं।

\textbf{12.} $2^{10} = 1024 = 3 \times 341 + 1 \equiv 1
\pmod{341}$, अतः $2^{340} = \left(2^{10}\right)^{34} \equiv
1$। फिर भी $341 = 11 \times 31$ भाज्य है: वह आधार $2$ पर फ़र्मा
की परीक्षा पास कर लेता है और फिर भी \omterm{def:g12:arith:prime}{अभाज्य} नहीं है। सीख: फ़र्मा की \omterm{def:g12:arith:congruence}{सर्वांगसमता}
आवश्यक है, पर्याप्त नहीं --- अभाज्यता की जाँच को और पैने औज़ार चाहिए (और वे
उसे स्नातक खंडों में मिलते भी हैं)।

\textbf{13.} $3d \equiv 1 \pmod{20}$: $d = 7$ ($21 = 20 + 1$)।

\textbf{14.} $c = 4^3 = 64 \equiv 31 \pmod{33}$।

\textbf{15.} $31 \equiv -2$: $(-2)^7 = -128$, और $-128 + 4 \times 33 = 4$: गूढ़ पाठ का गूढ़वाचन $m = 4$ देता है। ताला
घूम जाता है।

\textbf{16.} $3$ के सापेक्ष: यदि $3 \nmid m$ हो, तो $m^2 \equiv 1$ (फ़र्मा), इसलिए $m^{21} = m \cdot \left(m^2\right)^{10} \equiv m$;
और यदि $3 \mid m$ हो, तो दोनों पक्ष $\equiv 0$ हैं। $11$ के सापेक्ष: $m^{10} \equiv 1$ या $11 \mid m$,
और $m^{21} = m \cdot \left(m^{10}\right)^2 \equiv m$। $3$ और $11$ दोनों $m^{21} - m$ को \omterm{def:g12:arith:divides}{विभाजित} करते हैं, और \omterm{def:g12:arith:gcd}{सह-अभाज्य} होने
के कारण उनका गुणनफल $33$ भी (गाउस): $m^{21} \equiv m \pmod{33}$। घातांक $ed = 21 = 1 + 20k$ ऐसा गढ़ा गया था कि
दोनों फ़र्मा-घातांक ($2$ और $10$, जो $20$ को \omterm{def:g12:arith:divides}{विभाजित} करते हैं) ग़ायब हो
जाएँ।

\textbf{17.} $300$ अंकों के दो \omterm{def:g12:arith:prime}{अभाज्यों} को गुणा करने में माइक्रोसेकंड लगता
है; पर उनके गुणनफल से उन्हें वापस पाना हर ज्ञात कलनविधि और संसार के सारे
संगणकों को हरा देता है --- ताला एकतरफ़ा गली है। (हमारा $n = 33$ वही गली खिलौने
के पैमाने पर है, जो दोनों दिशाओं में चलने लायक़ है।)

\textbf{18.} शेषफल आज़माने से (या बेज़ू से बनाकर): $n \equiv 8 \pmod{15}$: गिनतियाँ $8, 23, 38, 53, \dots$।
$15$ के सापेक्ष अद्वितीयता: दो हल $3$ के और $5$ के \omterm{def:g12:arith:divides}{गुणज} से भिन्न होते
हैं, अतः $15$ के भी ($3$ और $5$ \omterm{def:g12:arith:gcd}{सह-अभाज्य}, गाउस)। $1000$ सैनिकों वाला
सेनापति तीन झटपट क़तारों से ``$8$'' घोषित कर देता है --- यही प्राचीन
सिर-गिनती की तरकीब है।

\textbf{19.} $10 \equiv 1 \pmod 9$ से $10^k \equiv 1$ मिलता है, इसलिए $\sum d_k 10^k \equiv \sum d_k$: कोई संख्या और उसके अंकों
का योग $9$ के सापेक्ष (और $3$ के सापेक्ष भी) सर्वांगसम होते हैं। और $10 \equiv -1 \pmod{11}$
से $\sum d_k 10^k \equiv \sum (-1)^k d_k$ मिलता है: यानी एकांतर वाला नियम। बचपन के दो नियम, एक-एक पंक्ति में।

\textbf{20.} सर्वांगसमताओं ने शेषफलों को एक अंकगणित बना दिया (भाग I); बेज़ू
ने वे प्रतिलोम ढाले जो रैखिक सर्वांगसमताएँ और RSA का $d$ हल कर देते हैं
(प्रश्न 4, 13); फ़र्मा की लघु प्रमेय ने ताला खोला और बंद किया (प्रश्न
15--16); और सह-अभाज्य मापांकों को जोड़ने ने सैनिक गिने तथा उपपत्ति पूरी की
(प्रश्न 16, 18)। हार्डी पर फ़ैसला: उन्हें ज्ञात सबसे शुद्ध प्रमेय आज हर
ख़रीद की पहरेदारी करती है --- समय मिल जाए तो शुद्धता ही सबसे अधिक अनुप्रयोग
योग्य वस्तु है।
\end{solution}
