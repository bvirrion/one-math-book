\section*{अध्याय \ref{ch:g12:exp} --- चरघातांकी फलन और लघुगणक}

\begin{solution}{exo:g12:exp:1}
$\dfrac{\eu^{3x}\,\eu^{-x+1}}{\eu^{x}} = \eu^{3x - x + 1 - x} = \eu^{x+1}$।

$\ln\!\left(\dfrac{\eu^2\sqrt{\eu}}{\eu^{-3}}\right)
= 2 + \tfrac12 + 3 = \tfrac{11}{2}$।
\end{solution}

\begin{solution}{exo:g12:exp:2}
\emph{(a)} $X = \eu^x > 0$ रखिए: $X^2 - 3X + 2 = 0$ से $X = 1$ या $X = 2$ मिलता है, और दोनों धनात्मक हैं, इसलिए
$x = 0$ या $x = \ln 2$।

\emph{(b)} \omterm{def:g11:func:function}{प्रांत}: $x - 1 > 0$ और $x + 2 > 0$, इसलिए $x > 1$। \omterm{def:g10:algebra:equation}{समीकरण} $\ln\bigl((x-1)(x+2)\bigr) = \ln 4$ बन जाता है, अतः
$(x-1)(x+2) = 4$, \emph{अर्थात्} $x^2 + x - 6 = 0$, इसलिए $x = 2$ या $x = -3$। \omterm{def:g11:func:function}{प्रांत} में केवल $x = 2$ है: $S = \{2\}$।
\end{solution}

\begin{solution}{exo:g12:exp:3}
$\eu^x$ से भाग दीजिए: $\dfrac{1 - x^2\eu^{-x}}{1 + \eu^{-x}} \to \dfrac{1-0}{1+0} = 1$, जहाँ $x^2 \eu^{-x} \to 0$ (\cref{thm:g12:exp:growth}) का उपयोग हुआ है।

\cref{prop:g12:exp:lnanalytic} से $\dfrac{\ln x}{\sqrt x} \to 0$ (स्थिति $n = 2$)।

$x^2 \ln x = x \cdot (x \ln x) \to 0 \times 0 = 0$।

$x - \ln x = x\left(1 - \frac{\ln x}{x}\right) \to +\infty$, क्योंकि $\frac{\ln x}{x} \to 0$।
\end{solution}

\begin{solution}{exo:g12:exp:4}
$f'(x) = \eu^{-x} - x\eu^{-x} = (1 - x)\eu^{-x}$, जिसका चिह्न $1 - x$ जैसा है: $f$ $\intoc{-\infty}{1}$ पर बढ़ता है और $\intco{1}{+\infty}$ पर घटता है, तथा
उसका वैश्विक \omterm{def:g10:functions:extrema}{अधिकतम} $f(1) = \eu^{-1}$ है।

सीमाएँ: $x \to +\infty$ पर $f(x) = \frac{x}{\eu^x} \to 0$ (\cref{thm:g12:exp:growth}); और $x \to -\infty$ पर $x \to -\infty$ तथा $\eu^{-x} \to +\infty$, इसलिए $f(x) \to -\infty$।

$f''(x) = -\eu^{-x} - (1-x)\eu^{-x} = (x - 2)\eu^{-x}$, जो $x = 2$ पर चिह्न बदलता है: $\bigl(2,\, 2\eu^{-2}\bigr)$ पर \omterm{def:g12:deriv:inflection}{नति-परिवर्तन बिंदु}।
\end{solution}

\begin{solution}{exo:g12:exp:5}
मान लीजिए $\intoo{-1}{+\infty}$ पर $g(x) = x - \ln(1+x)$ है। तब $g'(x) = 1 - \frac{1}{1+x} = \frac{x}{1+x}$, जो $\intoo{-1}{0}$ पर ऋणात्मक और $\intoo{0}{+\infty}$ पर धनात्मक है:
$g$ अपना \omterm{def:g10:functions:extrema}{न्यूनतम} $g(0) = 0$ पाता है, इसलिए $g \geq 0$, और समता केवल $0$ पर। अतः $\ln(1+x) \leq x$।

इसे $x = \frac1n$ के साथ लगाइए: $n\ln\left(1 + \frac1n\right) \leq 1$, इसलिए $\left(1+\frac1n\right)^n = \eu^{\,n\ln(1+1/n)} \leq \eu$। और $x = -\frac{1}{n+1} > -1$ के साथ लगाइए: $\ln\left(1 - \frac{1}{n+1}\right) \leq -\frac{1}{n+1}$, इसलिए
$-(n+1)\ln\left(1 - \frac{1}{n+1}\right) \geq 1$ तथा $\left(1 - \frac{1}{n+1}\right)^{-(n+1)} \geq \eu$।
\end{solution}

\begin{solution}{exo:g12:exp:6}
\emph{1.} हर वर्ष पूँजी $1.03$ से गुणा हो जाती है: $C_n = C_0 (1.03)^n$।

\emph{2.} $C_n \geq 2C_0 \iff (1.03)^n \geq 2 \iff n \ln 1.03 \geq \ln 2
\iff n \geq \frac{\ln 2}{\ln 1.03} \approx 23.45$: पूँजी $24$ वर्षों बाद दुगुनी हो जाती है।

\emph{3.} $\frac{70}{3} \approx 23.3$, जो यथार्थ $\frac{\ln 2}{\ln 1.03}$ के पास है। नियम इसलिए चलता है कि छोटे $r$ के
लिए $\ln(1 + r) \approx r$, इसलिए $\frac{\ln 2}{\ln(1+r)} \approx \frac{0.693}{r} \approx \frac{70}{100r}$।
\end{solution}

\begin{solution}{exo:g12:exp:7}
$\eu^{2x} > \eu^{x+1} \iff 2x > x + 1 \iff x > 1$ (\omterm{thm:g12:exp:existence}{चरघातांकी फलन} निरंतर \omterm{def:g12:seq:monotonic}{वर्धमान} है): $S = \intoo{1}{+\infty}$।

दूसरी असमिका का \omterm{def:g11:func:function}{प्रांत}: $x^2 - 1 > 0$ और $x + 5 > 0$, \emph{अर्थात्} $x \in \intoo{-5}{-1} \cup \intoo{1}{+\infty}$। इस \omterm{def:g11:func:function}{प्रांत} पर $\ln$
के निरंतर \omterm{def:g12:seq:monotonic}{वर्धमान} होने से
\[
\ln(x^2-1) \leq \ln(x+5) \iff x^2 - 1 \leq x + 5 \iff x^2 - x - 6 \leq 0
\iff x \in \intcc{-2}{3}.
\]
\omterm{def:g11:func:function}{प्रांत} से \omterm{def:g10:numbers:interunion}{प्रतिच्छेदन} लेने पर: $S = \intco{-2}{-1} \cup \intoc{1}{3}$।
\end{solution}

\begin{solution}{exo:g12:exp:8}
\emph{1.} $f'(x) = \dfrac{\eu^x x - \eu^x}{x^2} = \dfrac{(x-1)\eu^x}{x^2}$, जो $\intoo{0}{1}$ पर ऋणात्मक और $\intoo{1}{+\infty}$ पर धनात्मक है: \omterm{def:g10:functions:extrema}{न्यूनतम} $f(1) = \eu$।
सीमाएँ: $0^+$ पर $f \to +\infty$ (अंश $\to 1$, हर $\to 0^+$) और $+\infty$ पर (\cref{thm:g12:exp:growth})।

\emph{2.} $x > 0$ के लिए $\eu^x = kx \iff f(x) = k$। \omterm{def:g10:functions:table}{परिवर्तन-सारणी} से ($+\infty \searrow \eu \nearrow +\infty$, और हर ओर \omterm{def:g12:limcont:continuity}{संतत} तथा
निरंतर \omterm{def:g12:seq:monotonic}{एकदिष्ट}): $k < \eu$ के लिए कोई हल नहीं; $k = \eu$ के लिए ठीक एक ($x = 1$); और
$k > \eu$ के लिए ठीक दो (हर \omterm{def:g10:numbers:interval}{अंतराल} पर एकैकी आच्छादन प्रमेय से एक $\intoo{0}{1}$ में और एक
$\intoo{1}{+\infty}$ में)।
\end{solution}

\begin{solution}{exo:g12:exp:9}
\emph{1.} \cref{exo:g12:exp:5} की पहली असमिका से सीधे: $u_n = \eu^{\,n\ln(1+1/n)} \leq \eu^1 = \eu$।

\emph{2.} लिखिए
\[
\ln u_n = n \ln\!\left(1 + \frac1n\right)
= \frac{\ln\!\left(1 + \frac1n\right) - \ln 1}{\frac1n},
\]
जो $1$ बिंदु पर वृद्धि $h = \frac1n \to 0$ के साथ $\ln$ का अंतर-भागफल है। चूँकि $\ln$
$1$ पर \omterm{def:g12:deriv:derivative}{अवकलनीय} है और उसका \omterm{def:g12:deriv:derivative}{अवकलज} $1$ है, इसलिए $\ln u_n \to 1$। और $\exp$ के \omterm{def:g12:limcont:continuity}{सांतत्य}
से (\cref{prop:g12:limcont:seqcont}) $u_n = \eu^{\ln u_n} \to \eu^1 = \eu$।
\end{solution}

\begin{solution}{pb:g12:exp:1}
\textbf{1.} $x = \frac{\ln 5}{2}$. \quad $3x - 1 = \eu^2$: $x = \frac{\eu^2 + 1}{3}$. \quad $u = \eu^x$ के साथ: $u^2 - 3u + 2 = (u - 1)(u - 2) = 0$: $x = 0$ या
$x = \ln 2$।

\textbf{2.} $\frac{3\ln 2}{\ln 2} = 3$; \quad $3 + \frac12 = \frac72$; \quad $\frac52$।

\textbf{3.} $f'(x) = \eu^x - 1$: $0$ से पहले ऋणात्मक, बाद में धनात्मक: \omterm{def:g10:functions:extrema}{न्यूनतम} $f(0) = 0$,
इसलिए हर जगह $f \geq 0$: $\eu^x \geq 1 + x$। $x = \ln(1 + u)$ प्रतिस्थापित करने पर: $1 + u \geq 1 + \ln(1 + u)$, अर्थात् $\ln(1 + u) \leq u$।

\textbf{4.} $x^2\eu^{-x} = \frac{x^2}{\eu^x} \to 0$ और $\frac{\ln x}{x} \to 0$: \omterm{thm:g12:exp:existence}{चरघातांकी फलन} घातों को कुचल देता है, और घातें
\omterm{def:g12:exp:ln}{लघुगणकों} को। और $\frac{\eu^x - 1}{x} =
\frac{\eu^x - \eu^0}{x - 0} \to \exp'(0) = 1$।

\textbf{5.} $f'(x) = \ln x + 1$: $\eu^{-1}$ पर शून्य, उससे पहले ऋणात्मक, बाद में धनात्मक:
\omterm{def:g10:functions:extrema}{न्यूनतम} $f\!\left(\frac1\eu\right) = -\frac1\eu$; और $0^+$ पर $x \ln x \to 0$: वक्र मूल बिंदु से निकलता है, $-\frac1\eu$ तक
डुबकी लगाता है, और फिर चढ़ जाता है।

\textbf{6.} $n = \frac{\ln 2}{\ln 1.03} \approx 23.4$ वर्ष।

\textbf{7.} यथार्थ: $23.4$, $11.9$, $8.0$ वर्ष; 72 का नियम: $24$, $12$,
$8$। चूँकि $\ln(1+r) \approx r$, इसलिए $\frac{\ln 2}{\ln(1+r)} \approx \frac{0.693}{r}$, अर्थात् $\frac{69.3}{\text{\% में दर}}$; बैंकर $72$ तक पूर्णांकित कर
लेते हैं क्योंकि वह $2, 3, 4, 6, 8, 9, 12$ से बहुत सुंदर तरह से बँट जाता है --- मौखिक गणित एक
दशमलव की यथार्थता से बाज़ी मार ले जाता है।

\textbf{8.} $2^{t/20} = \eu^{t \ln 2 / 20}$: प्रति मिनट $\lambda = \frac{\ln 2}{20}$। $24 \times 60 = 1440$ मिनट बाद: $2^{72} \approx 4.7 \times
10^{21}$ जीवाणु --- यानी
एक ही दिन में एक कोशिका से पृथ्वी की रेत के कणों से भी अधिक। यह प्रतिरूप तभी
तक ईमानदार है जब तक भोजन और जगह हैं: असली वृद्धि मुड़कर लॉजिस्टिक S-वक्र बन
जाती है (\cref{pb:g12:deriv:1} का महामारी-विज्ञानी वाला वक्र)।

\textbf{9.} 23:00 बजे ($8$ घंटे): $100 \times 2^{-8/5} \approx 33$ mg। $10$ mg से नीचे: $2^{-t/5} < 0.1$ से
$t > 5\,\frac{\ln 10}{\ln 2} \approx
16.6$ घंटे मिलते हैं: यानी अगली सुबह लगभग 7:40 --- तीन बजे की एस्प्रेसो की
पूँछ लंबी होती है।

\textbf{10.} वार्षिक: $2$। मासिक: $\left(1 + \frac{1}{12}\right)^{12} \approx 2.613$। दैनिक: $\approx 2.715$। छत: $\eu = 2.71828\ldots$ (\cref{exo:g12:exp:9})
--- लगातार चक्रवृद्धि पर लोभ \omterm{def:g12:seq:limit}{अभिसरित} हो जाता है।

\textbf{11.} यदि रोगी चरघातांकी रूप से बढ़ें, $c(t) = c_0
\eu^{\lambda t}$, तो $\ln c(t) = \ln c_0 + \lambda t$: \omterm{def:g10:reffunc:affine}{प्रवणता} $\lambda$
वाली सीधी रेखा --- यानी वृद्धि-दर। लघुगणकीय \omterm{def:g10:functions:graph}{आलेख} पर सीधा हिस्सा चरघातांकी
चरण \emph{ही} है, और उसकी तीखी ढाल महामारी की लय है।

\textbf{12.} $120 - 60 = 60$ dB का अर्थ है $10\log_{10}(I_2/I_1) = 60$: समारोह बातचीत से $10^6$ --- यानी दस
लाख गुना --- अधिक तीव्र है।

\textbf{13.} आयाम: $10^{7-5} = 100$। ऊर्जा: $100^{3/2} = 1000$: परिमाण के दो अंक ऊर्जा में तीन कोटियाँ
छिपाए बैठे हैं।

\textbf{14.} $10^{6.5 - 2} = 10^{4.5} \approx 32\,000$: नींबू का रस दूध से तीस हज़ार गुना अधिक अम्लीय है --- pH
रसायन की खाइयों को जेब में समा जाने वाले मापक्रम में दबा देता है।

\textbf{15.} $\log_2\!\frac{4186}{27.5} =
\frac{\ln(4186/27.5)}{\ln 2} \approx 7.25$: पियानो सात सप्तक से ज़रा ही अधिक में फैला है। इंद्रियाँ
उद्दीपनों के \emph{अनुपात} पर प्रतिक्रिया करती हैं --- तीव्रता दुगुनी होना एक
क़दम जैसा लगता है, चाहे शुरुआत कहीं से भी हो --- इसलिए बोध लघुगणकीय मापक्रम
पर बना है, और उसके लिए गढ़े गए मात्रक भी।

\textbf{16.} $a$ वाली पट्टी को $b$ की स्थिति वाली पट्टी से सिरा-से-सिरा
जोड़ने पर लंबाइयाँ $\ln a + \ln b$ जुड़ जाती हैं, और उस कुल लंबाई पर बैठा अंकन $\eu^{\ln a + \ln b} = ab$
पढ़ता है: गणक-पट्टी \omterm{def:g12:exp:ln}{लघुगणक} जोड़कर गुणनफल निकाल देती है --- $\ln(ab) = \ln a + \ln b$ (\cref{prop:g12:exp:lnalg}) जो
लकड़ी में उकेरा गया है।

\textbf{17.} $\intoo{0}{+\infty}$ पर $\frac{\eu^x}{x}$ का \omterm{def:g10:functions:extrema}{न्यूनतम} $x = 1$ पर $\eu$ है: $k < \eu$ के लिए कोई
हल नहीं, $k = \eu$ के लिए ठीक एक, $k > \eu$ के लिए दो। \omterm{def:g12:deriv:tangent}{स्पर्श रेखा} से जाँच: $a = 1$ पर
$\eu^x$ की \omterm{def:g12:deriv:tangent}{स्पर्श रेखा} $y = \eu + \eu(x - 1) = \eu x$ है: वह मूल बिंदु से होकर जाती है --- वही सीमा-रेखा,
जो $(1, \eu)$ पर वक्र को छूकर निकल जाती है।

\textbf{18.} $n \ln\!\left(1 - \frac1n\right) =
\frac{\ln(1 - 1/n)}{-1/n} \times (-1) \to -1$, इसलिए $\left(1 - \frac1n\right)^n \to \eu^{-1} \approx 0.368$: \cref{pb:g11:binom:1} का $0.999^{1000} \approx 0.368$, समझाया हुआ --- लॉटरी का
``बस चूक गए'' वाला अचर $\frac1\eu$ है।

\textbf{19.} दोनों ``अनेक स्वतंत्र \omterm{def:g10:proba:events}{घटनाएँ}, और हर एक अकेले में असंभाव्य'' की
सीमांत आकृति हैं: आकार $\frac1n$ की $n$ संभावनाओं में से \emph{एक भी} न चलने
की संभावना $\eu^{-1}$ की ओर जाती है, और सचिव वाला नियम अपनी अस्वीकार-खिड़की इस तरह
सुर में बाँधता है कि सफलता उसी अचर पर सिमट आए। $\frac1\eu \approx 0.368$।

\textbf{20.} चक्रवृद्धि की छत; वह अकेला आधार जिसकी $0$ पर \omterm{def:g12:deriv:tangent}{स्पर्श रेखा} की
\omterm{def:g10:reffunc:affine}{प्रवणता} $1$ है (और इसीलिए कलन उसे इतना चाहता है); वह वृद्धि जो हर घात से
आगे निकल जाती है; वह अचर जहाँ ``बस चूक गए'' और इष्टतम ठहराव आ बैठते हैं; और
\omterm{def:g12:exp:ln}{लघुगणक} --- गुणा जो जोड़ बन गया, और संसार के सबसे जंगली परिसर जो एक पटरी पर
मोड़ दिए गए।
\end{solution}
