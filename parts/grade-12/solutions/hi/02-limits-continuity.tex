\section*{अध्याय \ref{ch:g12:limcont} --- सीमाएँ और सांतत्य}

\begin{solution}{exo:g12:limcont:1}
$x^3$ से भाग देने पर: $\dfrac{2x^3 - x + 1}{5x^3 + x^2} = \dfrac{2 - 1/x^2 + 1/x^3}{5 + 1/x}
\to \dfrac{2}{5}$।

$x \to -\infty$ पर $x^2 + 3x - 1 = x^2(1 + 3/x - 1/x^2) \to +\infty$, क्योंकि $x^2 \to +\infty$ और कोष्ठक $1$ की ओर जाता है।

$x \to 2^+$ पर अंश $3 > 0$ की ओर और हर $0^+$ की ओर जाता है, इसलिए $\dfrac{x+1}{x-2} \to +\infty$।

$x > 0$ के लिए $-\dfrac1x \leq \dfrac{\cos x}{x} \leq \dfrac1x$, इसलिए संपीडन प्रमेय से सीमा $0$ है।
\end{solution}

\begin{solution}{exo:g12:limcont:2}
\emph{1.} $2x^2 - 3x + 1 = (x-1)(2x-1)$ (\omterm{def:g11:quad:discriminant}{मूल} $x=1$ $2 - 3 + 1 = 0$ से स्पष्ट है)। $x \neq 1$ के लिए $x - 1$ काटिए: $f(x) = 2x - 1$।

\emph{2.} अतः $\lim_{x\to1} f(x) = 1$। $f(1) = 1$ रखने पर $f$ $\R$ पर बहुपद \omterm{def:g11:func:function}{फलन} $x \mapsto 2x-1$ तक बढ़ जाता है,
जो $1$ पर \omterm{def:g12:limcont:continuity}{संतत} है।
\end{solution}

\begin{solution}{exo:g12:limcont:3}
$f(x) = \dfrac{3x - 1}{x + 2} = \dfrac{3(x+2) - 7}{x+2}
= 3 - \dfrac{7}{x+2}$ लिखिए। $x \to \pm\infty$ पर $f(x) \to 3$: रेखा $y = 3$ दोनों अनंतों पर \omterm{def:g12:limcont:asymptote}{क्षैतिज अनन्तस्पर्शी} है।
$x \to (-2)^\pm$ पर $f(x) \to \mp\infty$: रेखा $x = -2$ \omterm{def:g12:limcont:asymptote}{ऊर्ध्वाधर अनन्तस्पर्शी} है। वक्र एक \omterm{def:g11:func:reference}{अतिपरवलय} है, जो
$x > -2$ के लिए $y=3$ से नीचे और $x < -2$ के लिए उससे ऊपर है।
\end{solution}

\begin{solution}{exo:g12:limcont:4}
संयुग्मी से गुणा कीजिए:
\[
\sqrt{x^2 + x} - x = \frac{x^2 + x - x^2}{\sqrt{x^2+x} + x}
= \frac{x}{x\left(\sqrt{1 + 1/x} + 1\right)}
= \frac{1}{\sqrt{1 + 1/x} + 1}
\xrightarrow[x\to+\infty]{} \frac{1}{2}.
\]
इसी तरह
\[
\frac{\sqrt{1+x} - 1}{x} = \frac{(1+x) - 1}{x\left(\sqrt{1+x} + 1\right)}
= \frac{1}{\sqrt{1+x} + 1} \xrightarrow[x\to0]{} \frac{1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:limcont:5}
\emph{1.} चूँकि $\sin x \geq -1$ है, इसलिए $f(x) \geq x - 1 \to +\infty$, और तुलना से निष्कर्ष निकल आता है। फिर भी
$f'(x) = 1 + \cos x$ $\pi$ के हर विषम गुणज पर लुप्त हो जाता है, और हर \omterm{def:g10:numbers:interval}{अंतराल} $\intco{A}{+\infty}$ पर \omterm{def:g11:func:function}{फलन} कड़ी
वृद्धि वाले \omterm{def:g10:numbers:interval}{अंतरालों} के बीच बदलता रहता है; वह \omterm{def:g12:seq:monotonic}{वर्धमान} है पर निरंतर नहीं, और
इससे भी बड़ी बात यह कि उसकी वृद्धि की आवश्यकता ही नहीं: तुलना पर्याप्त है।

\emph{2.} $x > 1$ के लिए $x$ से भाग दीजिए:
\[
g(x) = \frac{1 + \frac{\sin x}{x}}{1 - \frac{\sin x}{x}},
\]
और संपीडन प्रमेय से $\frac{\sin x}{x} \to 0$ ($\abs{\sin x} \leq 1$)। अतः $g(x) \to \frac{1+0}{1-0} = 1$।
\end{solution}

\begin{solution}{exo:g12:limcont:6}
$f(x) = x^3 - 3x + 1$ \omterm{def:g12:limcont:continuity}{संतत} है, और $f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$: $f$ $\intoc{-\infty}{-1}$ पर बढ़ता है, $\intcc{-1}{1}$ पर घटता है, $\intco{1}{+\infty}$ पर बढ़ता
है, तथा उसका स्थानीय \omterm{def:g10:functions:extrema}{अधिकतम} $f(-1) = 3 > 0$ और स्थानीय \omterm{def:g10:functions:extrema}{न्यूनतम} $f(1) = -1 < 0$ है, $\lim_{-\infty} f = -\infty$, $\lim_{+\infty} f = +\infty$।

एकदिष्टता के तीनों \omterm{def:g10:numbers:interval}{अंतरालों} में से हर एक पर $f$ \omterm{def:g12:limcont:continuity}{संतत} और निरंतर \omterm{def:g12:seq:monotonic}{एकदिष्ट} है,
और $0$ सिरों के मानों/सीमाओं के कड़ाई से बीच पड़ता है, इसलिए एकैकी आच्छादन
प्रमेय हर \omterm{def:g10:numbers:interval}{अंतराल} में ठीक एक शून्य देती है: कुल तीन शून्य। चिह्नों की जाँच:
$f(-2) = -1 < 0 < f(-1) = 3$, इसलिए $\intoo{-2}{-1}$ में एक शून्य; $f(0) = 1 > 0 > f(1) = -1$, $\intoo{0}{1}$ में एक शून्य; $f(1) = -1 < 0 < f(2) = 3$, $\intoo{1}{2}$ में एक
शून्य।
\end{solution}

\begin{solution}{exo:g12:limcont:7}
मान लीजिए $g(x) = f(x) - x$ है, जो $\intcc{0}{1}$ पर \omterm{def:g12:limcont:continuity}{संतत} है। चूँकि $f(0) \in \intcc{0}{1}$ है, इसलिए $g(0) = f(0) \geq 0$; और चूँकि
$f(1) \in \intcc{0}{1}$ है, इसलिए $g(1) = f(1) - 1 \leq 0$। मध्यवर्ती मान प्रमेय से $g$ किसी $c \in \intcc{0}{1}$ पर लुप्त हो जाता
है, अर्थात् $f(c) = c$।
\end{solution}

\begin{solution}{exo:g12:limcont:8}
मान लीजिए $u(t)$ और $d(t)$ समय $t \in \intcc{8}{20}$ पर चढ़ाई और उतराई के दिन की स्थितियाँ हैं
(रास्ते के तल से दूरी के रूप में नापी गईं)। दोनों \omterm{def:g12:limcont:continuity}{संतत} हैं, $u(8) = 0$, $u(20) = L$
(रास्ते की लंबाई), $d(8) = L$, $d(20) = 0$। \omterm{def:g11:func:function}{फलन} $g = u - d$ \omterm{def:g12:limcont:continuity}{संतत} है, जिसमें $g(8) = -L \leq 0$ और $g(20) = L \geq 0$ हैं,
इसलिए मध्यवर्ती मान प्रमेय से किसी समय $c$ पर $g(c) = 0$: समय $c$ पर दोनों दिनों
की स्थितियाँ मेल खा जाती हैं।
\end{solution}

\begin{solution}{exo:g12:limcont:9}
\emph{1.} $f'(x) = 5x^4 + 3x^2 + 1 > 0$, इसलिए $f$ $\R$ पर निरंतर \omterm{def:g12:seq:monotonic}{वर्धमान} है; और $\lim_{\pm\infty} f = \pm\infty$ तथा \omterm{def:g12:limcont:continuity}{सांतत्य}
के साथ एकैकी आच्छादन प्रमेय एक अद्वितीय वास्तविक शून्य $\alpha$ देती है। चूँकि
$f(0) = -1 < 0$ और $f(1) = 2 > 0$ हैं, इसलिए $\alpha \in \intoo{0}{1}$।

\emph{2.} $\intcc{0}{1}$ से शुरू करके ऐसा \omterm{def:g10:numbers:interval}{अंतराल} $\intcc{a_n}{b_n}$ बनाए रखिए जिसमें $f(a_n) < 0 < f(b_n)$ हो; हर
क़दम पर \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} $m$ पर $f$ का मान निकालिए और वही अर्ध-अंतराल रखिए जिस
पर चिह्न-परिवर्तन बना रहे। $n$ क़दमों के बाद \omterm{def:g10:numbers:interval}{अंतराल} की लंबाई $2^{-n}$ हो जाती
है, इसलिए \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} $\alpha$ का $2^{-(n+1)}$ तक \omterm{def:g10:numbers:approx}{सन्निकटन} देता है। हमें $2^{-(n+1)} \leq 10^{-6}$ चाहिए,
अर्थात् $n + 1 \geq \frac{6}{\log_{10} 2} \approx 19.93$: $n = 20$ पुनरावृत्तियाँ पर्याप्त हैं।
\end{solution}

\begin{solution}{pb:g12:limcont:1}
\textbf{1.} हावी पद: $\frac{3x^2}{x^2} \to 3$; $\frac{2x}{x^2} = \frac2x \to 0$; $x^3 - x^2 = x^3\left(1 - \frac1x\right) \to +\infty$।

\textbf{2.} $\lim_{x \to \pm\infty} f = 3$: \omterm{def:g12:limcont:asymptote}{क्षैतिज अनन्तस्पर्शी} $y = 3$; $\lim_{x \to (-2)^\pm} f = \mp\infty$: \omterm{def:g12:limcont:asymptote}{ऊर्ध्वाधर
अनन्तस्पर्शी} $x = -2$।

\textbf{3.} $\frac{(x-3)(x+3)}{x - 3} = x + 3 \to 6$। और $\frac{\sqrt{x+1} - 1}{x} = \frac{x}{x(\sqrt{x+1} + 1)} =
\frac{1}{\sqrt{x+1} + 1} \to \frac12$।

\textbf{4.} $\abs{x \sin\frac1x} \leq \abs x \to 0$, इसलिए सीमा $0$ है। और $\abs{\frac{\sin x}{x}} \leq \frac1x \to 0$: सीमा $0$।

\textbf{5.} $\frac{\sqrt{4x^2 + x}}{x} =
\sqrt{4 + \frac1x} \to \sqrt4 = 2$ (वर्गमूल का \omterm{def:g12:limcont:continuity}{सांतत्य}, संयोजन)।

\textbf{6.} $f(x) = x^3 + x$ \omterm{def:g12:limcont:continuity}{संतत} है और $f(1) = 2 < 5 < 10 = f(2)$: मध्यवर्ती मान प्रमेय से किसी $c \in \intoo{1}{2}$ के लिए
$f(c) = 5$।

\textbf{7.} $f'(x) = 3x^2 + 1 > 0$: $\R$ पर निरंतर \omterm{def:g12:seq:monotonic}{वर्धमान}, \omterm{def:g12:limcont:continuity}{संतत}, और सीमाएँ $\mp\infty$: एकैकी
आच्छादन प्रमेय से हर मान --- विशेष रूप से $5$ --- ठीक एक बार प्राप्त होता
है।

\textbf{8.} $p(x) = x^3\left(1 + \frac ax + \frac{b}{x^2} +
\frac{c}{x^3}\right)$: $x \to +\infty$ पर $p \to +\infty$; और $x \to -\infty$ पर $p \to -\infty$। इसलिए किसी विशाल
\omterm{def:g10:numbers:interval}{अंतराल} $\intcc{-M}{M}$ पर $p(-M) < 0 < p(M)$: मध्यवर्ती मान प्रमेय \omterm{def:g11:quad:discriminant}{मूल} दे देती है। वही हावी-पद वाला
तर्क किसी भी विषम घात के लिए ज्यों का त्यों चलता है --- विषम घातें दोनों
अनंतों पर विपरीत चिह्न बनाए रखती हैं।

\textbf{9.} $n$ बार आधा करने पर लंबाई $2^{-n}$ का \omterm{def:g10:numbers:interval}{अंतराल} बचता है; $2^{-20} \approx 9.5 \times 10^{-7} < 10^{-6}$: बीस
क़दम। न्यूटन हर क़दम पर सही अंक दुगुने कर देता है: एक सही अंक से छह तक पहुँचने
में लगभग $3$--$4$ क़दम --- \omterm{def:g11:quad:discriminant}{मूल} ढूँढ़ने की दुनिया के कछुआ और शिकारी कुत्ता।

\textbf{10.} $g$ $\intcc{-1}{1}$ पर \omterm{def:g12:limcont:continuity}{संतत} नहीं है: वह $0$ पर फट पड़ता है, जो उस
\omterm{def:g10:numbers:interval}{अंतराल} में है। मध्यवर्ती मान प्रमेय का निष्कर्ष उसकी परिकल्पना के साथ ही उड़
जाता है --- \omterm{def:g12:limcont:continuity}{सांतत्य} सदा \emph{पूरे \omterm{def:g10:numbers:interval}{अंतराल} पर} जाँचिए, केवल सिरों पर नहीं।

\textbf{11.} $g(x) = f(x) - x$ \omterm{def:g12:limcont:continuity}{संतत} है; $g(0) = f(0) \geq 0$ (मान $\intcc{0}{1}$ में हैं) और $g(1) = f(1) - 1 \leq 0$। मध्यवर्ती मान
प्रमेय से किसी $c$ के लिए $g(c) = 0$: $f(c) = c$।

\textbf{12.} देश को रेखाखंड $\intcc{0}{1}$ मानकर आदर्शीकरण कीजिए: $f$ हर स्थान को उस
स्थान पर भेजता है जो मसले हुए नक़्शे पर ठीक उसके नीचे दिखता है; मसलने से नक़्शा
देश के ऊपर ही रहता है, इसलिए $f$ $\intcc{0}{1}$ में गिरता है, और वह \omterm{def:g12:limcont:continuity}{संतत} है (काग़ज़
फटता नहीं)। \omterm{pb:g10:functions:1}{स्थिर बिंदु} $c$ वही जगह है जो ठीक अपने ही \omterm{def:g10:functions:function}{प्रतिबिंब} के ऊपर है
--- ईमानदार द्विविमीय नक़्शे को ब्राउवर की प्रमेय सँभालती है।

\textbf{13.} $g(0) + g(\pi) = \left(T(0) - T(\pi)\right) +
\left(T(\pi) - T(2\pi)\right) = T(0) - T(2\pi) = 0$। इसलिए $g(0)$ और $g(\pi)$ परस्पर विपरीत हैं (या दोनों शून्य):
दोनों ही स्थितियों में \omterm{def:g12:limcont:continuity}{संतत} $g$ किसी $t^* \in \intcc{0}{\pi}$ पर लुप्त हो जाता है: $T(t^*) = T(t^* + \pi)$ ---
यानी भूमध्य रेखा पर कहीं न कहीं, हर एक क्षण, दो प्रतिध्रुवी बिंदुओं का ताप
एक ही होता है।

\textbf{14.} मेज़ को एक चौथाई चक्कर घुमाने से दोनों विकर्णों की भूमिकाएँ बदल
जाती हैं: जो पाए ज़मीन दबा रहे थे और जिनमें से एक लटका था, उनकी स्थितियाँ आपस
में बदल जाती हैं, इसलिए लटकने की ऊँचाई --- जिसे ``चुने हुए विकर्ण के \omterm{def:g10:numbers:interval}{अंतराल}''
के रूप में चिह्न सहित गिना जाए --- $\theta$ और $\theta + \frac\pi2$ पर विपरीत चिह्न ले लेती है।
\omterm{def:g12:limcont:continuity}{संतत} ज़मीन $h$ को \omterm{def:g12:limcont:continuity}{संतत} बना देती है, और मध्यवर्ती मान प्रमेय $h(\theta^*) = 0$ वाला कोई
कोण $\theta^*$ सौंप देती है: चारों पाए ज़मीन पर। (आयताकार मेज़ या सीढ़ीदार ज़मीन के
साथ ऐसा सौभाग्य नहीं।)

\textbf{15.} ढाँचा: जिन दो राशियों को मिलाना है उनका \emph{अंतर} बनाइए ($f(x) - x$;
$T(t) - T(t + \pi)$; पदयात्री के लिए चढ़ाई-स्थिति घटा उतराई-स्थिति; मेज़ के लिए चिह्नित
\omterm{def:g10:numbers:interval}{अंतराल}); जाँचिए कि वह दोनों चिह्नों के मान लेता है (प्रायः सिरों की किसी
सममिति से); और किसी \omterm{def:g12:limcont:continuity}{संतत} \omterm{def:g11:func:function}{फलन} पर मध्यवर्ती मान प्रमेय का हवाला दीजिए। मेल
अनिवार्य।

\textbf{16.} खुले प्रश्न: हल \emph{कितने} हैं (इसका उत्तर निरंतर एकदिष्टता और
एकैकी आच्छादन प्रमेय ने दिया, प्रश्न 7) और वे \emph{कहाँ} हैं (इसका उत्तर
\omterm{thm:g12:limcont:ivt}{द्विभाजन} या न्यूटन ने दिया, प्रश्न 9)। मध्यवर्ती मान प्रमेय शुद्ध अस्तित्व का
भविष्यवक्ता है।

\textbf{17.} $3 - \frac{7}{x + 2}$: \omterm{def:g12:limcont:asymptote}{ऊर्ध्वाधर अनन्तस्पर्शी} $x = -2$, क्षैतिज $y = 3$; $-2$ के
दोनों ओर \omterm{def:g11:func:function}{फलन} \omterm{def:g12:seq:monotonic}{वर्धमान} है (जहाँ परिभाषित है वहाँ $-\frac{7}{x+2}$ बढ़ता है); वक्र आधार
\omterm{def:g11:func:reference}{अतिपरवलय} $-\frac7x$ ही है, जो $2$ बाईं ओर और $3$ ऊपर खिसका हुआ है --- दो
शाखाएँ अनन्तस्पर्शियों की क्रॉस से चिपकी हुई।

\textbf{18.} $x \neq 1$ के लिए $f(x) = x + 1$, इसलिए $\lim_{x \to 1} f = 2$: $f(1) = 2$ रखने पर \omterm{def:g12:limcont:continuity}{सांतत्य} लौट आता
है। असांतत्य कोई उछाल नहीं, केवल एक छेद था --- अकेला छूटा हुआ बिंदु भर दीजिए
और वक्र भर जाता है: \emph{हटाने योग्य}।

\textbf{19.} क्रमागत \omterm{def:g10:numbers:sets}{पूर्णांकों} के बीच के हर \omterm{def:g10:numbers:interval}{अंतराल} पर \omterm{def:g12:limcont:continuity}{संतत}; और हर \omterm{def:g10:numbers:sets}{पूर्णांक}
$n$ पर वह $n - 1$ से $n$ तक उछल जाता है। $a = 0.9$ के साथ $b = 1.1$: $\lfloor a \rfloor = 0 < \frac12 <
1 = \lfloor b \rfloor$, फिर भी
$\lfloor x \rfloor = \frac12$ का कोई हल नहीं --- उछाल $\frac12$ के ऊपर से छलाँग लगा जाता है, और ठीक यही
मध्यवर्ती मान प्रमेय \omterm{def:g12:limcont:continuity}{संतत} \omterm{def:g10:functions:function}{फलनों} को करने से मना करती है।

\textbf{20.} स्थानीय चेहरा: हर बिंदु पर मान $=$ सीमा --- हटाने योग्य छेद
(प्रश्न 18) ने उसका उल्लंघन इलाज-योग्य ढंग से किया, और महत्तम \omterm{def:g10:numbers:sets}{पूर्णांक} \omterm{def:g11:func:function}{फलन} के
उछालों (प्रश्न 19) ने लाइलाज ढंग से। वैश्विक चेहरा: किसी \omterm{def:g10:numbers:interval}{अंतराल} पर सभी
मध्यवर्ती मान छू लिए जाते हैं --- \omterm{def:g11:quad:discriminant}{मूल} के प्रमाणपत्र (6--8), \omterm{pb:g10:functions:1}{स्थिर बिंदु},
भूमध्य रेखा, मेज़ और पदयात्री, सब यहीं रहते हैं। नारे का औपचारिक नाम:
\emph{मध्यवर्ती मान प्रमेय}।
\end{solution}
