\begin{solution}{exo:g12:complex:2}
\emph{(a)} $\Delta = 16 - 52 = -36$: $z = \dfrac{4 \pm 6\iu}{2} = 2 \pm 3\iu$। \omterm{def:g12:complex:conjugate}{संयुग्मी} युग्म; गुणनफल $= 4 + 9 = 13 = \frac ca$। \emph{(b)} $\Delta = 1 - 4 = -3$: $z = \dfrac{-1 \pm \iu\sqrt3}{2}$

\end{solution}

\begin{solution}{exo:g12:complex:3}
$\abs{z_1} = \sqrt2$, \omterm{def:g12:complex:argument}{कोणांक} $\frac{3\pi}{4}$ (दूसरा चतुर्थांश): $z_1 = \sqrt2\,\eu^{3\iu\pi/4}$। $\abs{z_2} = 2$, \omterm{def:g12:complex:argument}{कोणांक} $-\frac\pi6$: $z_2 = 2\,\eu^{-\iu\pi/6}$। अतः
\[
\frac{z_1}{z_2} = \frac{\sqrt2}{2}\,\eu^{\iu\left(\frac{3\pi}{4} + \frac{\pi}{6}\right)}
= \frac{\sqrt2}{2}\,\eu^{11\iu\pi/12}.
\]
बीजगणित से,
\[
\frac{z_1}{z_2} = \frac{-1+\iu}{\sqrt3 - \iu}
= \frac{(-1+\iu)(\sqrt3+\iu)}{4}
= \frac{-\sqrt3 - 1 + \iu(\sqrt3 - 1)}{4}.
\]
\omterm{def:g12:complex:def}{वास्तविक भाग} मिलाने पर: $\frac{\sqrt2}{2}\cos\frac{11\pi}{12} = \frac{-\sqrt3-1}{4}$, इसलिए
\[
\cos\frac{11\pi}{12} = -\frac{\sqrt6 + \sqrt2}{4}.
\]
\end{solution}

\begin{solution}{exo:g12:complex:4}
\emph{(a)} $\abs{z - 2} = \abs{z - (-\iu)}$ का अर्थ है कि $M$ $A(2, 0)$ और $B(0, -1)$ से समान दूरी पर है: यानी $[AB]$
का \emph{लंब समद्विभाजक}।

\emph{(b)} बिंदु $(1,1)$ से दूरी $3$: केंद्र $1 + \iu$ और त्रिज्या $3$ वाला
\emph{वृत्त}।

\emph{(c)} \cref{prop:g12:complex:geometry}(4) से $\frac{z-1}{z+1}$ का विशुद्ध काल्पनिक होना बताता है कि $A(1,0)$ से $M$
तक और $B(-1,0)$ से $M$ तक के \omterm{def:g11:vect:vector}{सदिश} \omterm{def:g11:scal:orthogonal}{लांबिक} हैं: $M$ रेखाखंड $[AB]$ को समकोण पर देखता
है। समुच्चय \emph{व्यास $[AB]$ वाला वृत्त} (इकाई वृत्त) है, \emph{पर बिंदु $A$
और $B$ स्वयं हटाकर}।
\end{solution}

\begin{solution}{exo:g12:complex:5}
दे म्वाव्र: $\cos3\theta + \iu\sin3\theta = (\cos\theta + \iu\sin\theta)^3$। $c = \cos\theta$, $s = \sin\theta$ के साथ द्विपद प्रमेय से प्रसार करने पर:
\[
(c + \iu s)^3 = c^3 + 3\iu c^2 s - 3 c s^2 - \iu s^3
= (c^3 - 3cs^2) + \iu(3c^2 s - s^3).
\]
भाग मिलाकर और $s^2 = 1 - c^2$, $c^2 = 1 - s^2$ का उपयोग करके:
\[
\cos3\theta = 4\cos^3\theta - 3\cos\theta,
\qquad
\sin3\theta = 3\sin\theta - 4\sin^3\theta .
\]
\end{solution}

\begin{solution}{exo:g12:complex:6}
\[
\cos^3\theta = \left(\frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}\right)^{\!3}
= \frac{\eu^{3\iu\theta} + 3\eu^{\iu\theta} + 3\eu^{-\iu\theta} + \eu^{-3\iu\theta}}{8}
= \frac{\cos3\theta + 3\cos\theta}{4}.
\]
अतः
\[
\int_0^{\pi/2}\cos^3\theta\,\dd\theta
= \frac14\left[\frac{\sin3\theta}{3} + 3\sin\theta\right]_0^{\pi/2}
= \frac14\left(-\frac13 + 3\right) = \frac{2}{3}.
\]
(\cref{exo:g12:integ:9} के $W_3 = \frac23$ के अनुकूल, क्योंकि सममिति $\theta \mapsto \frac\pi2 - \theta$ से $\int_0^{\pi/2}\cos^3 = \int_0^{\pi/2}\sin^3$।)
\end{solution}

\begin{solution}{exo:g12:complex:7}
$b - a = 2 + \iu$ और $\eu^{\pm\iu\pi/3} = \frac12 \pm \iu\frac{\sqrt3}{2}$ के साथ $c = a + \eu^{\pm\iu\pi/3}(b - a)$:
\[
\eu^{\iu\pi/3}(2+\iu) = \left(\tfrac12 + \iu\tfrac{\sqrt3}{2}\right)(2+\iu)
= \left(1 - \tfrac{\sqrt3}{2}\right) + \iu\left(\tfrac12 + \sqrt3\right),
\]
इसलिए $c_1 = \left(2 - \frac{\sqrt3}{2}\right) +
\iu\left(\frac32 + \sqrt3\right)$, और इसी तरह कोण $-\frac\pi3$ के साथ: $c_2 = \left(2 + \frac{\sqrt3}{2}\right) + \iu\left(\frac32 - \sqrt3\right)$।
\end{solution}

\begin{solution}{exo:g12:complex:8}
$-4 = 4\,\eu^{\iu\pi}$, इसलिए $z^4 = -4$ के हल
\[
z_k = \sqrt2\,\eu^{\iu\left(\frac\pi4 + \frac{k\pi}{2}\right)},
\quad k = 0, 1, 2, 3,
\]
हैं, \emph{अर्थात्} $1 + \iu$, $-1 + \iu$, $-1 - \iu$, $1 - \iu$: परित्रिज्या $\sqrt 2$ वाले किसी वर्ग
के शीर्ष। \omterm{def:g12:complex:conjugate}{संयुग्मी} \omterm{def:g11:quad:discriminant}{मूलों} की जोड़ी बनाने पर
\[
z^4 + 4 = \bigl(z - (1{+}\iu)\bigr)\bigl(z - (1{-}\iu)\bigr)
\bigl(z + (1{-}\iu)\bigr)\bigl(z + (1{+}\iu)\bigr)
= (z^2 - 2z + 2)(z^2 + 2z + 2).
\]
\end{solution}

\begin{solution}{exo:g12:complex:9}
$q = \eu^{\iu\theta} \neq 1$ के साथ:
\[
\sum_{k=0}^{n} \eu^{\iu k\theta} = \frac{\eu^{\iu(n+1)\theta} - 1}{\eu^{\iu\theta} - 1}
= \frac{\eu^{\iu(n+1)\theta/2}}{\eu^{\iu\theta/2}}
\cdot\frac{\eu^{\iu(n+1)\theta/2} - \eu^{-\iu(n+1)\theta/2}}{\eu^{\iu\theta/2} - \eu^{-\iu\theta/2}}
= \eu^{\iu n\theta/2}\,
\frac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}},
\]
जहाँ $\eu^{\iu\alpha} - \eu^{-\iu\alpha} = 2\iu\sin\alpha$ का उपयोग हुआ है (\emph{अर्ध-कोण \omterm{def:g10:algebra:expand}{गुणनखंडन}})। \omterm{def:g12:complex:def}{वास्तविक भाग} लेने पर:
\[
S = \frac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}}\,
\cos\frac{n\theta}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:complex:10}
\emph{1.} $\omega$ $1$ से भिन्न $5$-वाँ इकाई-मूल है, इसलिए पाँचों \omterm{def:g11:quad:discriminant}{मूलों} का योग
लुप्त हो जाता है (\cref{thm:g12:complex:rootsofunity}), जिसका अर्थ $1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$ है।

\emph{2.} बिंदु 1 से $u + v = \omega + \omega^2 + \omega^3 + \omega^4 = -1$। गुणनफल के लिए $\omega^5 = 1$ का उपयोग करने पर:
\[
uv = (\omega + \omega^4)(\omega^2 + \omega^3)
= \omega^3 + \omega^4 + \omega^6 + \omega^7
= \omega^3 + \omega^4 + \omega + \omega^2 = -1 .
\]
इसलिए $u, v$ $X^2 + X - 1 = 0$ के \omterm{def:g11:quad:discriminant}{मूल} हैं: $\left\{\frac{-1+\sqrt5}{2}, \frac{-1-\sqrt5}{2}\right\}$। अब $u = \omega + \conj\omega = 2\cos\frac{2\pi}{5} > 0$ (क्योंकि $\frac{2\pi}{5} < \frac{\pi}{2}$), अतः $u = \frac{-1 + \sqrt5}{2}$।

\emph{3.} $\cos\frac{2\pi}{5} = \frac u2 = \frac{\sqrt5 - 1}{4}$।
\end{solution}

\begin{solution}{pb:g12:complex:1}
\textbf{1.} $(2 + \iu)(3 - \iu) = 6 - 2\iu + 3\iu + 1 =
7 + \iu$. \quad $\frac{1 + \iu}{1 - \iu} =
\frac{(1 + \iu)^2}{2} = \frac{2\iu}{2} = \iu$. \quad $\abs{3 + 4\iu} = 5$।

\textbf{2.} $1 + \iu = \sqrt2\,\eu^{\iu\pi/4}$; $-2 = 2\eu^{\iu\pi}$; $1 - \iu\sqrt3 = 2\eu^{-\iu\pi/3}$।

\textbf{3.} $(1 + \iu)^8 = \left(\sqrt2\right)^8
\eu^{8\iu\pi/4} = 16\,\eu^{2\iu\pi} = 16$।

\textbf{4.} $z^2 = \eu^{\iu\pi/2}$: $z = \pm\eu^{\iu\pi/4} = \pm\frac{\sqrt2}{2}(1 + \iu)$। और $\Delta = 4 - 20 = -16$: $z = 1 \pm 2\iu$।

\textbf{5.} $z \mapsto \iu z$: मूल बिंदु के परितः एक चौथाई चक्कर का घूर्णन। $\abs{z - (1 + \iu)} = 2$: केंद्र
$1 + \iu$ और त्रिज्या $2$ वाला वृत्त।

\textbf{6.} $\left(\omega^k\right)^5 = \eu^{2\iu\pi k} = 1$, और पाँचों बिंदु $\eu^{2\iu k\pi/5}$ भिन्न हैं: यानी $z^5 = 1$ के पाँचों हल।
वे इकाई वृत्त पर $72^\circ$ के बराबर कोणों पर बैठे हैं: एक सम पंचभुज, जिसका एक
शीर्ष $1$ पर है।

\textbf{7.} $S = 1 + \omega + \dots + \omega^4$ $(1 - \omega)S = 1 - \omega^5 = 0$ को संतुष्ट करता है, और $\omega \neq 1$: $S = 0$। (पंचभुज का
संतुलन-बिंदु उसका केंद्र है।)

\textbf{8.} $\omega^4 = \overline\omega$ और $\omega^3 = \overline{\omega^2}$: संयुग्मियों की जोड़ी बनाने पर योग का \omterm{def:g12:complex:def}{वास्तविक भाग}
$1 + 2\cos\frac{2\pi}{5} + 2\cos\frac{4\pi}{5} = 0$ पढ़ा जाता है।

\textbf{9.} $\cos\frac{4\pi}{5} = 2c^2 - 1$, इसलिए $1 + 2c + 4c^2 - 2 = 0$, अर्थात् $4c^2 + 2c - 1 = 0$: $c = \frac{-1 + \sqrt5}{4}$ (धनात्मक \omterm{def:g11:quad:discriminant}{मूल}, क्योंकि
$72^\circ$ न्यून कोण है): $\cos 72^\circ = \frac{\sqrt5 - 1}{4}$।

\textbf{10.} संख्याओं में $c \approx 0.30902$: कैलकुलेटर से मेल। और $\frac{1}{2\varphi} = \frac{1}{1 + \sqrt5} =
\frac{\sqrt5 - 1}{4}$ (हर का परिमेयकरण):
$c = \frac{1}{2\varphi}$. पंचभुज आरपार स्वर्णिम है: उसका विकर्ण उसकी भुजा को $\varphi$ के अनुपात
में काटता है --- पंचकोण-तारे की अंतहीन स्व-समरूपता।

\textbf{11.} वर्गमूल परकार से रचनीय हैं (माध्यमिक विद्यालय खंड की \omterm{def:g12:seq:arith-geom}{गुणोत्तर}
\omterm{def:g11:stat:mean}{माध्य} वाली अर्धवृत्त मशीन), इसलिए जो संख्या \omterm{def:g10:numbers:sets}{परिमेय संख्याओं} से केवल वर्गमूलों
द्वारा बनी हो --- जैसे $\cos 72^\circ$ --- वह रचनीय है; घनमूल रचनीय नहीं हैं, और $\cos 20^\circ$
को एक घनमूल चाहिए: पंचभुज झुक जाता है, त्रिभाजन डटा रहता है। गाउस की कसौटी हर
सम बहुभुज का फ़ैसला कर देती है --- उन्नीस वर्ष की आयु में उनके 17-भुज ने ही
उन्हें गणित चुनने पर विवश किया।

\textbf{12.} $-4 = 4\eu^{\iu\pi}$: चौथे \omterm{def:g11:quad:discriminant}{मूलों} का \omterm{def:g12:complex:modulus}{मापांक} $\sqrt2$ है और \omterm{def:g12:complex:argument}{कोणांक} $\frac{\pi}{4} + k\frac{\pi}{2}$: चारों
संख्याएँ $\pm 1 \pm \iu$। संयुग्मियों की जोड़ी बनाने पर: $(z - 1 - \iu)(z - 1 + \iu) = z^2 - 2z + 2$ और $(z + 1 - \iu)(z + 1 + \iu) = z^2 + 2z + 2$: यही \cref{exo:g12:complex:8} का
\omterm{def:g10:algebra:expand}{गुणनखंडन} है।

\textbf{13.} $(\cos\theta + \iu\sin\theta)^3 = \cos^3\theta +
3\iu\cos^2\theta\sin\theta - 3\cos\theta\sin^2\theta -
\iu\sin^3\theta$; दे म्वाव्र से यह $\cos 3\theta + \iu \sin 3\theta$ के बराबर है। \omterm{def:g12:complex:def}{वास्तविक भाग}: $\cos 3\theta = \cos^3\theta - 3\cos\theta(1 - \cos^2\theta)
= 4\cos^3\theta - 3\cos\theta$
--- और \omterm{def:g12:complex:def}{काल्पनिक भाग} $\sin 3\theta$ मुफ़्त में दे देते हैं।

\textbf{14.} यह $1 + \omega + \dots + \omega^4 = 0$ का \omterm{def:g12:complex:def}{वास्तविक भाग} है: योग $0$ के बराबर है --- ठीक
प्रश्न 8, \cref{exo:g12:complex:9} की मूल-निधि से देखा हुआ।

\textbf{15.} $8\iu = 8\eu^{\iu\pi/2}$: \omterm{def:g11:quad:discriminant}{मूल} $2\eu^{\iu\pi/6} = \sqrt3 + \iu$; $2\eu^{5\iu\pi/6} = -\sqrt3 + \iu$; $2\eu^{3\iu\pi/2} = -2\iu$।

\textbf{16.} $1 + \iu = \sqrt2\,\eu^{\iu\pi/4}$: प्रतिचित्रण समतल को $45^\circ$ से घुमाता है और $\sqrt2$ गुना बड़ा
कर देता है। इकाई वर्ग भुजा $\sqrt2$ वाला तिरछा वर्ग बन जाता है --- क्षेत्रफल में
दुगुना, और आठवें हिस्से भर घूमा हुआ।

\textbf{17.} $\abs{z_1 z_2}^2 = z_1 z_2 \overline{z_1 z_2} =
z_1\overline{z_1}\, z_2\overline{z_2} =
\abs{z_1}^2\abs{z_2}^2$: वर्गमूल लीजिए। $1 + \iu$ की घातें: $1 + \iu$, $2\iu$, $-2 + 2\iu$, $-4$:
हर एक $45^\circ$ और घूमी हुई तथा $\sqrt2$ गुना लंबी --- मापांकों $\sqrt2, 2, 2\sqrt2, 4, \dots$ से होकर बाहर
की ओर जाता एक सर्पिल।

\textbf{18.} $\mathrm{j}$ $1$ से भिन्न इकाई का तीसरा \omterm{def:g11:quad:discriminant}{मूल} है, इसलिए ($n = 3$ के साथ
प्रश्न 7 का तर्क) $1 + \mathrm{j} + \mathrm{j}^2 = 0$। मानक त्रिभुज पर कसौटी: $a + \mathrm{j}b + \mathrm{j}^2 c = 1 +
\mathrm{j}\cdot\mathrm{j} + \mathrm{j}^2 \cdot \mathrm{j}^2 =
1 + \mathrm{j}^2 + \mathrm{j}^4 = 1 + \mathrm{j}^2 +
\mathrm{j} = 0$: संतुष्ट, जैसा समबाहु
त्रिभुज $\left(1, \mathrm{j}, \mathrm{j}^2\right)$ माँगता है।

\textbf{19.} $\C$ पर कोई वास्तविक बहुपद रैखिक गुणनखंडों में बँट जाता है;
उसके अवास्तविक \omterm{def:g11:quad:discriminant}{मूल} \omterm{def:g12:complex:conjugate}{संयुग्मी} युग्मों में आते हैं (\omterm{def:g10:algebra:equation}{समीकरण} का संयुग्मन कीजिए),
और हर युग्म गुणा होकर एक वास्तविक द्विघात $z^2 - 2\operatorname{Re}(z_0)\,z + \abs{z_0}^2$ बन जाता है: अतः घात-$1$ और
घात-$2$ के टुकड़ों में वास्तविक \omterm{def:g10:algebra:expand}{गुणनखंडन} --- प्रश्न 12 ने उसे $z^4 + 4$ पर करके
दिखाया।

\textbf{20.} बिंदु; घूर्णन-और-मापन; सम बहुभुज; प्रसार से त्रिकोणमितीय
सर्वसमिकाएँ; और एक ऐसा बंद संसार जहाँ घात $n$ के हर \omterm{def:g10:algebra:equation}{समीकरण} के $n$ \omterm{def:g11:quad:discriminant}{मूल}
होते हैं। समापन-पद: प्रश्न 10 में $\pi$ के पंचभुज ने स्वर्ण अनुपात से हाथ
मिलाया --- यूक्लिड के दोनों सबसे प्रसिद्ध अतिथि, जिनका परिचय अंततः शून्य के
बराबर पाँच तीरों के योग ने कराया।
\end{solution}
