\section*{अध्याय \ref{ch:g12:contdist} --- संतत यादृच्छिक चर}

\begin{solution}{exo:g12:contdist:1}
$\P(X \leq 5) = \frac{5}{15} = \frac13$; $\P(4 \leq X \leq 10) = \frac{6}{15} = \frac25$; $\E(X) = \frac{15}{2} = 7.5$ मिनट; $\sigma(X) = \frac{15}{\sqrt{12}} = \frac{5\sqrt3}{2} \approx 4.3$ मिनट।
\end{solution}

\begin{solution}{exo:g12:contdist:2}
$f \geq 0$ और $\int_0^2 \frac38 t^2\,\dd t = \frac38\left[\frac{t^3}{3}\right]_0^2
= \frac38 \times \frac83 = 1$: यानी \omterm{def:g12:contdist:density}{घनत्व}।
\[
\E(X) = \int_0^2 \frac38 t^3\,\dd t = \frac38 \times \frac{16}{4} = \frac32,
\qquad
\P(X \geq 1) = \int_1^2 \frac38 t^2\,\dd t = \frac{8 - 1}{8} = \frac78 .
\]
\end{solution}

\begin{solution}{exo:g12:contdist:3}
\emph{1.} $\E(X) = \frac{1}{0.2} = 5$ वर्ष; $\P(X > 5) = \eu^{-0.2\times5} = \eu^{-1} \approx 0.37$।

\emph{2.} स्मृतिहीनता (\cref{thm:g12:contdist:memoryless}) से $\pcond{X > 3}{X > 8} = \P(X > 5) = \eu^{-1} \approx 0.37$: तीन वर्ष की सेवा से कुछ नहीं बदलता।
\end{solution}

\begin{solution}{exo:g12:contdist:4}
$\P(X > m) = \eu^{-\lambda m} = \frac12$ से $m = \frac{\ln 2}{\lambda} \approx \frac{0.69}{\lambda}$ मिलता है, जो $\E(X) = \frac1\lambda$ से छोटा है। चरघातांकी \omterm{def:g12:contdist:density}{घनत्व} की दाहिनी पूँछ लंबी
होती है: कुछ असामान्य रूप से लंबे जीवनकाल \emph{\omterm{def:g11:stat:mean}{माध्य}} को \emph{\omterm{def:g11:stat:median}{माध्यिका}} से
ऊपर खींच ले जाते हैं, जबकि \omterm{def:g11:stat:median}{माध्यिका} वही मान है जिससे आधी समष्टि आगे निकलती है।
(यही \cref{ex:g12:exp:halflife} की अर्धायु है: आधे परमाणु उसे पार कर जाते हैं, यद्यपि औसत जीवनकाल
उससे लंबा है।)
\end{solution}

\begin{solution}{exo:g12:contdist:5}
$168 = \mu - \sigma$ और $182 = \mu + \sigma$: लगभग $68\%$। $189 = \mu + 2\sigma$: उससे ऊपर बचे हुए $100 - 95.4 = 4.6\%$ का आधा पड़ता है, यानी
लगभग $2.3\%$। $154 = \mu - 3\sigma$: लगभग $\frac{100 - 99.7}{2} = 0.15\%$।
\end{solution}

\begin{solution}{exo:g12:contdist:6}
$\P(X < 500) \leq 2.5\%$ का अर्थ है कि $500$ को \omterm{def:g11:stat:mean}{माध्य} से कम से कम दो \omterm{def:g11:stat:variance}{मानक विचलन} नीचे बैठना चाहिए
(प्रसामान्य $\mu - 2\sigma$ से नीचे $2.3\% \approx 2.5\%$ छोड़ता है; प्रचलित $1.96$ के साथ तर्क बिलकुल वही
रहता है):
\[
\mu - 2\sigma \geq 500 \iff \mu \geq 500 + 8 = 508 \text{ g}.
\]
मशीन को औसतन $508$ g पर बिठाना पड़ेगा --- और इस गारंटी की क़ीमत प्रति थैला $8$
g ``मुफ़्त'' माल है।
\end{solution}

\begin{solution}{exo:g12:contdist:7}
\emph{1.} $p = \frac16$, $n = 1200$ के साथ: $\sqrt{\frac{p(1-p)}{n}} = \sqrt{\frac{5/36}{1200}} \approx 0.0108$, इसलिए $95\%$ उतार-चढ़ाव \omterm{def:g10:numbers:interval}{अंतराल}
\[
\frac16 \pm 1.96 \times 0.0108 \approx \intcc{0.146}{0.188}.
\]
है।

\emph{2.} प्रेक्षित \omterm{def:g10:stats:series}{आपेक्षिक बारंबारता} $\frac{260}{1200} \approx 0.217$ उससे बहुत बाहर है: न्यायसंगतता की
परिकल्पना $5\%$ स्तर पर अस्वीकार हो जाती है। विचलन लगभग $4.6$ \omterm{def:g11:stat:variance}{मानक विचलन} है ---
प्रसामान्य \omterm{def:g10:numbers:approx}{सन्निकटन} में इसकी \omterm{def:g10:proba:distribution}{प्रायिकता} $10^{-6}$ की कोटि की होती है, जो
ब्येनेमे--चेबिशेव के $\leq 4.6\%$ वाले परिबंध (\cref{exo:g12:sums:7}) से कहीं अधिक निर्णायक है।
\end{solution}

\begin{solution}{exo:g12:contdist:8}
\emph{1.} सीमा $\frac{1}{\sqrt n} \leq 0.02$ के लिए $n \geq 2500$ लोग चाहिए।

\emph{2.} $1$ अंक के अंतर वाले स्कोर अलग बताने के लिए सीमा $0.5$ अंक से कहीं
नीचे होनी चाहिए, जिसके लिए सरलीकृत सूत्र से $n \geq \left(\frac{1}{0.005}\right)^2 =
40\,000$ चाहिए --- जो अधिकांश
सर्वेक्षणों के लिए पहले ही अव्यावहारिक है। और उससे भी बुरा: सांख्यिकीय सीमा
केवल \emph{प्रतिदर्शन} की त्रुटि गिनती है; व्यवस्थित अभिनतियाँ (अप्रतिनिधि
\omterm{def:g10:proba:sample}{प्रतिदर्श}, उत्तर न देना, अंतिम क्षण के बदलाव) $n$ बढ़ने पर सिकुड़ती नहीं और
प्रायः एक अंक से अधिक होती हैं। $1$-अंक की दौड़ सचमुच इतनी क़रीबी है कि उसका
फ़ैसला नहीं किया जा सकता।
\end{solution}

\begin{solution}{exo:g12:contdist:9}
\emph{1.} चूँकि $\alpha > 0$, इसलिए $c \leq \alpha X + \beta \leq d
\iff \frac{c - \beta}{\alpha} \leq X \leq \frac{d - \beta}{\alpha}$, अतः
\[
\P(c \leq Y \leq d)
= \int_{(c-\beta)/\alpha}^{(d-\beta)/\alpha} f(t)\,\dd t .
\]
प्रतिस्थापन $y = \alpha t + \beta$ (अर्थात् क्षेत्रफल को $y$ चर में पढ़ना, $t = \frac{y - \beta}{\alpha}$, $\dd t = \frac{\dd y}{\alpha}$) इसे
$\int_c^d \frac{1}{\alpha} f\left(\frac{y-\beta}{\alpha}\right)\dd y$ में बदल देता है: इसलिए \omterm{def:g11:func:function}{फलन} $g(y) = \frac1\alpha f\left(\frac{y-\beta}{\alpha}\right)$ $Y$ का \omterm{def:g12:contdist:density}{घनत्व} है।

\emph{2.} $f = \varphi$, $\alpha = \sigma$, $\beta = \mu$ के साथ:
\[
g(y) = \frac{1}{\sigma}\,\varphi\!\left(\frac{y - \mu}{\sigma}\right)
= \frac{1}{\sigma\sqrt{2\pi}}\,
\exp\left(-\frac{(y-\mu)^2}{2\sigma^2}\right),
\]
और यही $\mathcal N(\mu, \sigma^2) = \mu + \sigma\,\mathcal N(0,1)$ का \omterm{def:g12:contdist:density}{घनत्व} है।
\end{solution}

\begin{solution}{pb:g12:contdist:1}
\textbf{1.} $\P(\text{प्रतीक्षा} \leq 10) = \frac{10}{30} =
\frac13$; $\E = 15$ मिनट; $\sigma = \frac{30}{\sqrt{12}} \approx 8.7$ मिनट।

\textbf{2.} $\int_0^1 2x\,\dd x = 1$: \omterm{def:g12:contdist:density}{घनत्व}। $\E(X) = \int_0^1 2x^2\,\dd x = \frac23$; $\P\left(X > \frac12\right) = \int_{1/2}^1 2x\,\dd x =
1 - \frac14 = \frac34$।

\textbf{3.} $\P(T > 15) = \eu^{-15/10} \approx 0.22$। \omterm{def:g11:stat:median}{माध्यिका}: $\eu^{-m/10} = \frac12$: $m = 10\ln 2 \approx 6.9$ मिनट --- जो \omterm{def:g11:stat:mean}{माध्य} $10$ से कम है,
क्योंकि चरघातांकी की लंबी दाहिनी पूँछ (विरली विशाल प्रतीक्षाएँ) \omterm{def:g11:stat:mean}{माध्य} को ऊपर
घसीट ले जाती है और \omterm{def:g11:stat:median}{माध्यिका} को ``प्रारूपिक'' प्रतीक्षा पर ही छोड़ देती है।

\textbf{4.} स्मृतिहीनता से $\P(T > 35 \mid T > 20) = \P(T > 15) \approx 0.22$: पहले ही बिता चुके बीस मिनट कुछ नहीं ख़रीदते
--- धारा बूढ़ी नहीं होती।

\textbf{5.} (a) चरघातांकी (रेडियोधर्मी क्षय ही वह प्रतिरूप है जिसके लिए
स्मृतिहीनता गढ़ी गई); (b) $\intcc{0}{8}$ पर एकसमान (यादृच्छिक खिसकाव वाली समय-सारणी);
(c) चरघातांकी; (d) इनमें से कोई नहीं: घिसा हुआ बल्ब अगले घंटे में नए बल्ब से
\emph{अधिक} संभावना से जलना बंद करता है, इसलिए $\P(T > s + t \mid T > s) < \P(T > t)$: बुढ़ाना स्मृतिहीनता प्रमेय
का खंडन है।

\textbf{6.} $68\,\%$, $95\,\%$, $99.7\,\%$: तीनों संकेत-अंतराल।

\textbf{7.} $z = \frac{190 - 172}{8} = 2.25$: ऊपरी पूँछ $\approx 1.2\,\%$।

\textbf{8.} $z = 2$: लगभग $2.3\,\%$ --- यानी मोटे तौर पर $44$ में एक व्यक्ति।

\textbf{9.} विनिर्देश $\pm 2\sigma$ पर बैठा है: लगभग $95.4\,\%$ पास होते हैं और $4.6\,\%$ अस्वीकार
--- जो बड़े पैमाने पर विनाशकारी है। $\pm 6\sigma$ पर विफलता-दर घटकर लगभग दो प्रति
\emph{अरब} रह जाती है: छह सिग्मा वह अधिकार ख़रीद लेता है कि बड़े पैमाने पर
उत्पादन हो और बड़े पैमाने पर निरीक्षण न करना पड़े।

\textbf{10.} $\sigma = 5$; सांतत्य-सुधार के साथ $\P \approx \P\left(\abs Z \leq \frac{5.5}{5}\right) =
\P(\abs Z \leq 1.1) \approx 0.73$। प्रमेय \cref{pb:g11:binom:1} के गॉल्टन पटल को
चिकना कर देती है: ख़ानों की सीढ़ी \omterm{def:g12:limcont:continuity}{संतत} घंटी बन जाती है।

\textbf{11.} $n \approx \frac{1.96^2 \times 0.25}{0.005^2}
\approx 38\,400$ लोग --- और उस आकार पर \emph{ग़ैर}-प्रतिदर्शन त्रुटियाँ
(ढाँचा, इनकार, झूठ) सीमा को बौना कर देती हैं: एक अंक की दौड़ ईमानदार सर्वेक्षण
की पहुँच से बाहर है, और इसीलिए गंभीर संस्थान तब कहते हैं ``फ़ैसला करना संभव
नहीं''।

\textbf{12.} स्मृतिहीनता: आपके पहुँचने के क्षण से बची हुई प्रतीक्षा \omterm{def:g11:stat:mean}{माध्य} $10$
वाली चरघातांकी है --- यानी पूरे $10$ मिनट, $5$ नहीं। समय-सारणी का ``माध्य
अंतराल $10$'' भ्रम में डाल देता है: आपकी प्रतीक्षा का \omterm{def:g12:randvar:rv}{बंटन} पूरे एक \omterm{def:g10:numbers:interval}{अंतराल}
जैसा ही है।

\textbf{13.} यादृच्छिक आगमन \omterm{def:g10:numbers:interval}{अंतराल} की लंबाई के समानुपात में गिरते हैं: $\P(\text{लंबा अंतराल}) = \frac{15}{20} = \frac34$।
प्रत्याशित प्रतीक्षा: $\frac34 \times \frac{15}{2} + \frac14 \times
\frac52 = 5.625 + 0.625 = 6.25$ मिनट --- भोले $5$ से अधिक, यद्यपि माध्य अंतराल
$10$ है। अपराधी: \emph{लंबाई-अभिनत प्रतिदर्शन} --- लंबे \omterm{def:g10:numbers:interval}{अंतराल} अधिक यात्री
पकड़ लेते हैं।

\textbf{14.} पीछे और आगे की प्रतीक्षाएँ दोनों \omterm{def:g11:stat:mean}{माध्य} $10$ वाली चरघातांकी हैं
(आपके आगमन से स्मृतिहीनता दोनों दिशाओं में चलती है): इसलिए आपको समेटने वाले
\omterm{def:g10:numbers:interval}{अंतराल} का औसत $20$ मिनट होता है --- प्रारूपिक \omterm{def:g10:numbers:interval}{अंतराल} का दुगुना। निरीक्षण
विरोधाभास एक वाक्य में: \emph{आप संयोग से जिस \omterm{def:g10:numbers:interval}{अंतराल} का निरीक्षण करते हैं वह
प्रारूपिक \omterm{def:g10:numbers:interval}{अंतराल} नहीं है, क्योंकि आपके किसी बड़े \omterm{def:g10:numbers:interval}{अंतराल} में गिरने की संभावना
अधिक थी।}

\textbf{15.} औसत कक्षा: $\frac{9 \times 10 + 110}{10} =
20$ विद्यार्थी। विद्यार्थी-भोगा औसत: $\frac{90 \times 10 + 110 \times 110}{200} = \frac{13\,000}
{200} = 65$
विद्यार्थी। विवरणिका $20$ छापती है; और पाँच में से तीन विद्यार्थी $110$ वाली
कक्षा में बैठते हैं तथा $65$ जीते हैं।

\textbf{16.} लोकप्रिय लोग अनेक मित्र-सूचियों में आते हैं, इसलिए ``एक मित्र''
का प्रतिदर्शन मिलनसारों की ओर लंबाई-अभिनत होता है --- आपके मित्र सामाजिक
समय-सारणी के लंबे अंतरालों से निकाले जाते हैं।

\textbf{17.} $t \geq 0$ के लिए: $\P(T > t) = \P(-10\ln U > t) = \P\left(U <
\eu^{-t/10}\right) = \eu^{-t/10}$: ठीक चरघातांकी का उत्तरजीविता \omterm{def:g11:func:function}{फलन} --- एक ही
\omterm{def:g12:exp:ln}{लघुगणक} संगणक के एकसमान शोर को किसी भी प्रतीक्षा-काल में बदल देता है।

\textbf{18.} हर एकसमान चर का \omterm{def:g11:stat:mean}{माध्य} $\frac12$ और \omterm{def:g12:randvar:exp}{प्रसरण} $\frac{1}{12}$ है: बारह के योग का
\omterm{def:g11:stat:mean}{माध्य} $6$ और \omterm{def:g12:randvar:exp}{प्रसरण} $1$ --- इसलिए नुस्ख़ा \omterm{def:g11:stat:mean}{माध्य} $0$ और \omterm{def:g12:randvar:exp}{प्रसरण} $1$
देता है। बारह \omterm{def:g12:sums:indep}{स्वतंत्र} छोटे धक्के जोड़े हुए: वही गॉल्टन तंत्र, और दे
म्वाव्र--लाप्लास के आशीर्वाद से आयतचित्र आँख को पहले ही घंटी-सा दिखता है।

\textbf{19.} प्रतिरूप के बारे में: बाज़ार अनेक \emph{\omterm{def:g12:sums:indep}{स्वतंत्र}} छोटे कारणों के
योग नहीं हैं --- घबराहट सब कुछ सहसंबंधित कर देती है (पिछली समस्या का 2008 वाला
पाठ) और ऐसी मोटी पूँछें पैदा करती है जो किसी प्रसामान्य \omterm{def:g12:contdist:density}{घनत्व} के पास हैं ही
नहीं। जब आँकड़े $10^{-137}$ फुसफुसाएँ, तब सांख्यिकीविद् का पाठ उत्तर देता है:
\emph{प्रतिरूप अस्वीकार हुआ है, वह दिन नहीं}।

\textbf{20.} एकसमान: सभी स्थान बराबर, यानी ईमानदार ``मैं केवल सीमाएँ जानता
हूँ''। चरघातांकी: वह ख़तरा जो कभी बूढ़ा नहीं होता --- क्षय, आगमन, चटकें।
प्रसामान्य: अनेक छोटे \omterm{def:g12:sums:indep}{स्वतंत्र} कारणों की लोकतांत्रिक घंटी --- लंबाइयाँ,
त्रुटियाँ, औसत। डंक: आप जिसका प्रतिदर्शन करते हैं वह इस बात से अभिनत होता है
कि आप उससे टकराए कैसे --- बसें, कक्षाएँ, मित्र। और चाप: जिस बच्चे ने कक्षा 6
में रस के डिब्बे गिने थे, उसने दस खंडों की समस्याओं के बाद एक वक्र से पूरी भीड़
नाप ली है; और यह समझाने के लिए कि घंटी हर चीज़ के लिए क्यों बजती है, एप्सिलॉन,
\omterm{def:g12:integ:area}{समाकल} तथा केंद्रीय सीमा प्रमेय स्नातक खंडों में प्रतीक्षा कर रहे हैं।
\end{solution}
