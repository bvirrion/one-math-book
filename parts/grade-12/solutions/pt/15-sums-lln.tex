\section*{Capítulo \ref{ch:g12:sums} --- Somas de variáveis aleatórias e a lei dos grandes números}

\begin{solution}{exo:g12:sums:1}
Cada dado tem \omterm{def:g12:randvar:exp}{esperança}
$\frac{1 + 2 + \dots + 6}{6} = \frac{7}{2}$, logo, por linearidade,
$\E(S) = \frac72 + \frac72 = 7$. Os dados são \omterm{def:g12:sums:indep}{independentes}, logo
$\V(S) = \frac{35}{12} + \frac{35}{12} = \frac{35}{6} \approx 5.83$.
\end{solution}

\begin{solution}{exo:g12:sums:2}
$\E(X) = 50$ e $\V(X) = 100 \times 0.25 = 25$.

Markov ($X \geq 0$): $\P(X \geq 75) \leq \frac{50}{75} = \frac23$.

Bienaymé--Chebyshev:
$\P(\abs{X - 50} \geq 25) \leq \frac{25}{25^2} = 0.04$ --- e isso limita
até o \omterm{def:g11:prob:model}{evento} \emph{bilateral}, do qual $X \geq 75$ é a metade. Chebyshev é
bem mais fina aqui porque explora a \omterm{def:g12:randvar:exp}{variância}, e não apenas a \omterm{def:g11:stat:mean}{média}. (O
valor verdadeiro de $\P(X \geq 75)$ é menor que $10^{-6}$: as duas cotas
são grosseiras.)
\end{solution}

\begin{solution}{exo:g12:sums:3}
$F_n$ é a \omterm{prop:g12:sums:mean}{média amostral} de $n$ variáveis de
Bernoulli$\left(\frac12\right)$, de \omterm{def:g12:randvar:exp}{variância} $\sigma^2 = \frac14$. A cota
\[
\P\left(\abs{F_n - 0.5} \geq 0.05\right)
\leq \frac{1/4}{n \times 0.05^2} = \frac{100}{n}
\]
é $\leq 0.05$ assim que $n \geq 2000$.
\end{solution}

\begin{solution}{exo:g12:sums:4}
Pela \cref{prop:g12:sums:mean}, $\E(M_n) = 5\%$ (a diversificação não
altera o retorno esperado) e
$\sigma(M_n) = \frac{20\%}{\sqrt n}$. Exigir
$\frac{20}{\sqrt n} < 4$ dá $\sqrt n > 5$, \emph{isto é}, $n \geq 26$.
Este é o princípio da \emph{diversificação}: repartir o capital entre
riscos \omterm{def:g12:sums:indep}{independentes} divide o risco (o \omterm{def:g11:stat:variance}{desvio padrão}) por $\sqrt n$ sem
reduzir o retorno esperado.
\end{solution}

\begin{solution}{exo:g12:sums:5}
\emph{1.} Contando os $9$ pares igualmente prováveis: $S$ assume os
valores $2, 3, 4, 5, 6$ com \omterm{def:g10:proba:distribution}{probabilidades}
$\frac19, \frac29, \frac39, \frac29, \frac19$. Logo
$\E(S) = \frac{2 + 6 + 12 + 10 + 6}{9} = 4$ e
$\E(S^2) = \frac{4 + 18 + 48 + 50 + 36}{9} = \frac{156}{9} = \frac{52}{3}$,
de modo que $\V(S) = \frac{52}{3} - 16 = \frac43$.

\emph{2.} Uma variável: $\E(X) = 2$,
$\E(X^2) = \frac{1 + 4 + 9}{3} = \frac{14}{3}$,
$\V(X) = \frac{14}{3} - 4 = \frac23$. Então $\E(S) = 2 + 2 = 4$ e, por
\omterm{def:g12:condprob:indep}{independência}, $\V(S) = \frac23 + \frac23 = \frac43$. Os mesmos valores.
\end{solution}

\begin{solution}{exo:g12:sums:6}
Com $Y = X$: $\V(X + Y) = \V(2X) = 4\V(X)$, enquanto
$\V(X) + \V(Y) = 2\V(X)$. Os dois só coincidem quando $\V(X) = 0$,
\emph{isto é}, quando $X$ é constante --- de modo que a aditividade exige
mesmo a \omterm{def:g12:condprob:indep}{independência} (aqui $X$ é maximamente dependente de si mesma).
\end{solution}

\begin{solution}{exo:g12:sums:7}
Sob a hipótese de honestidade, $F_n$ é a \omterm{def:g11:stat:mean}{média} de $n = 1200$ variáveis de
Bernoulli com $p = \frac16$, $\sigma^2 = \frac16\cdot\frac56
= \frac{5}{36}$:
\[
\P\left(\abs{F_n - \tfrac16} \geq 0.05\right)
\leq \frac{5/36}{1200 \times 0.0025} = \frac{5}{108} \approx 0.046 .
\]
O desvio observado é exatamente $0.05$: um \omterm{def:g11:prob:model}{evento} que um dado honesto
produz com \omterm{def:g10:proba:distribution}{probabilidade} de no máximo $4.6\%$ --- e, como Chebyshev é
muito conservadora, a \omterm{def:g10:proba:distribution}{probabilidade} verdadeira é muito menor. A hipótese
de honestidade não é plausível; o dado está muito provavelmente viciado.
\end{solution}

\begin{solution}{exo:g12:sums:8}
\emph{1.} $p(1-p) = \frac14 - \left(p - \frac12\right)^2 \leq \frac14$.

\emph{2.} $F_n$ tem \omterm{def:g12:randvar:exp}{esperança} $p$ e \omterm{def:g12:randvar:exp}{variância}
$\frac{p(1-p)}{n} \leq \frac{1}{4n}$; Bienaymé--Chebyshev dá
\[
\P\bigl(\abs{F_n - p} \geq \varepsilon\bigr)
\leq \frac{p(1-p)}{n\varepsilon^2} \leq \frac{1}{4n\varepsilon^2},
\]
válido qualquer que seja o $p$ desconhecido.

\emph{3.} Com $\varepsilon = 0.03$ e nível $0.05$:
\[
\frac{1}{4n(0.03)^2} \leq 0.05
\iff n \geq \frac{1}{4 \times 0.0009 \times 0.05} \approx 5556 .
\]
Cerca de $5600$ pessoas bastam por essa cota grosseira (a resposta
clássica pela \omterm{def:g10:numbers:approx}{aproximação} normal fica mais perto de $1100$, veja o
\cref{ch:g12:contdist}).
\end{solution}

\begin{solution}{pb:g12:sums:1}
\textbf{1.} $V(X + Y) = V(X) + V(Y) = \frac{35}{12} +
\frac{35}{12} = \frac{35}{6}$ (\omterm{def:g12:condprob:indep}{independência}).

\textbf{2.} $\E(\bar X) = 3.5$;
$\sigma(\bar X) = \frac{1.71}{\sqrt{100}} \approx 0.17$: a
\omterm{def:g11:stat:mean}{média} de cem dados abraça $3.5$ a menos de um quinto de ponto.

\textbf{3.} $\P(X \geq 10) \leq \frac{2}{10} = 0.2$.

\textbf{4.} $\P \leq \frac{V(\bar X)}{0.5^2} =
\frac{35/1200}{0.25} \approx 0.117$: no máximo uns
$12\,\%$.

\textbf{5.} Com $Y = -X$: $V(X + Y) = V(0) = 0$, enquanto
$V(X) + V(Y) = 2V(X) > 0$: variáveis perfeitamente
anticorrelacionadas se cancelam --- somar \omterm{def:g12:randvar:exp}{variâncias} é um
privilégio da \omterm{def:g12:condprob:indep}{independência}.

\textbf{6.} $\E = \frac{18}{37} - \frac{19}{37} =
-\frac{1}{37} \approx -0.027$;
$\sigma = \sqrt{1 - \left(\frac{1}{37}\right)^2} \approx
1.00$.

\textbf{7.} $\E(G) = -2.70$, $\sigma(G) = 10 \times 1.00 =
10$: a deriva é de apenas $0.27\sigma$. O ruído da noite
esmaga a vantagem --- uma grande minoria de apostadores (uns
$40\,\%$, diz o sino) sai ganhando, e é exatamente isso que os
faz voltar.

\textbf{8.} Faturamento do cassino: \omterm{def:g12:randvar:exp}{esperança} $+27\,027$
euros, \omterm{def:g11:stat:variance}{desvio padrão} $\approx 1\,000$: o lucro fica $27$
\omterm{def:g11:stat:variance}{desvios padrão} acima de zero. A $27\sigma$, ``o cassino pode
perder este ano'' não é um risco, é um erro de
arredondamento.

\textbf{9.} $\P(\text{perda}) = \P(G \geq 27\,027$ para os
apostadores$)$ $\leq \frac{10^6 \times 1}{27\,027^2} \approx
0.0014$: até a mais tosca desigualdade do livro garante a casa
a $99.86\,\%$ --- a verdade é astronomicamente mais forte.

\textbf{10.} Cada euro apostado, quando e como quer que seja
escolhido, tem retorno esperado de $-\frac{1}{37}$ de si
mesmo; por linearidade, o ganho total esperado é
$-\frac{1}{37} \times (\text{total apostado})$, negativo para
toda estratégia que aposte alguma coisa. Um sistema vencedor
violaria a linearidade da \omterm{def:g12:randvar:exp}{esperança} --- nenhum encadeamento
esperto de apostas ruins faz uma boa. (Os sistemas de
dobramento apenas trocam muitas vitórias pequenas por raras
perdas catastróficas.)

\textbf{11.} O teorema: a \emph{\omterm{def:g10:stats:series}{frequência}} do vermelho ao
longo de $n$ rodadas \omterm{def:g12:seq:limit}{converge} (em \omterm{def:g10:proba:distribution}{probabilidade}) para
$\frac{18}{37}$. Ele nada diz sobre a rodada $n + 1$: a roleta
não tem memória e, depois de dez vermelhos, a chance de
vermelho continua sendo $\frac{18}{37}$. Acreditar no
contrário é a \emph{falácia do apostador}, e a questão 12
mostra o que de fato acontece com as \omterm{def:g12:seq:sequence}{sequências}.

\textbf{12.} O excedente de $100$ vermelhos não é devolvido
--- as rodadas futuras são cópias honestas, e o excedente
esperado continua $100$. Mas $\frac{100}{10^6} = 0.0001$: um
centésimo de ponto. O erro da falácia em uma frase: \emph{a
lei dos grandes números dilui os acidentes passados em um
oceano de novos ensaios; ela nunca manda a roleta cobrar
dívidas.}

\textbf{13.} $n \geq \frac{1}{4 \times 0.05 \times 0.03^2}
\approx 5\,556$ pessoas.

\textbf{14.} $\frac{1.96^2 \times 0.25}{0.03^2} \approx
1\,067$: cinco vezes menos. Chebyshev vale para \emph{toda}
\omterm{def:g12:randvar:rv}{distribuição} e paga sua universalidade com folga; a constante
$1.96$ dos institutos de pesquisa vem da forma de sino real
das flutuações, que se concentra muito mais.

\textbf{15.} Margem $\propto \frac{1}{\sqrt n}$: reduzi-la à
metade quadruplica a \omterm{def:g10:proba:sample}{amostra} --- cerca de $4\,400$ pessoas
para $\pm 1.5$ ponto. A precisão se compra a preços
quadráticos.

\textbf{16.} O modelo é de $n$ retiradas \omterm{def:g12:sums:indep}{independentes} com
\omterm{def:g10:proba:distribution}{probabilidade} de sucesso $p$ --- o tamanho $N$ da população
nunca aparece (amostrar uma fração minúscula de uma população
grande e bem misturada é o que a \omterm{def:g12:condprob:indep}{independência} codifica). A
analogia dos institutos: para provar a sopa, uma colherada bem
mexida basta --- sirva a panela dez ou dez mil pessoas. O
problema está na mexida --- um cadastro de amostragem
enviesado, como na pesquisa do volume do ensino fundamental
que enganou um país --- e não na panela.

\textbf{17.} $315\sqrt n < 30n \iff \sqrt n > 10.5 \iff
n > 110$: passadas cem apólices, a deriva supera o ruído, e
cada apólice a mais alarga a diferença pela lei do $\sqrt n$
--- o agrupamento \emph{é} o modelo de negócio.

\textbf{18.} Partes iguais e \omterm{def:g12:sums:indep}{independentes}:
$\sigma_{\text{cart}} = \frac{20\,\%}{\sqrt{25}} = 4\,\%$:
mesmo retorno esperado, um quinto da flutuação. O almoço:
redução de risco a custo zero em \omterm{def:g12:randvar:exp}{esperança} --- a \omterm{def:g11:stat:mean}{média} de
aleatoriedades \omterm{def:g12:sums:indep}{independentes}.

\textbf{19.} Perfeitamente correlacionadas: a carteira é uma
única ação em $25$ fantasias: $\sigma = 20\,\%$, redução
nenhuma. A diversificação aluga a \emph{\omterm{def:g12:condprob:indep}{independência}};
quando uma crise correlaciona tudo (2008: todas as apostas
imobiliárias eram uma só aposta), a proteção do $\sqrt n$
evapora exatamente quando é necessária.

\textbf{20.} As somas derivam como $n$ e flutuam como
$\sqrt n$, de modo que as \omterm{def:g11:stat:mean}{médias} se assentam como
$\frac{1}{\sqrt n}$: o cassino embolsa a deriva ao longo de um
milhão de rodadas; o instituto de pesquisa compra
$\frac{1}{\sqrt{1100}}$ de ruído com um orçamento de
telefonemas; a seguradora ultrapassa os próprios sinistros em
$n > 110$; a investidora divide o risco por $\sqrt{25}$.
Abusadores: o apostador que acredita em dívidas (só existe
diluição) e o financista que acredita em \omterm{def:g12:condprob:indep}{independência} (às
vezes só existe uma aposta). A forma universal das flutuações
--- o sino --- é a estrela do capítulo \omterm{def:g12:limcont:continuity}{contínuo}, e seu
teorema, o teorema central do limite, é o cume da
\omterm{def:g10:proba:distribution}{probabilidade} dos volumes de graduação.
\end{solution}
