\section*{Capítulo \ref{ch:g12:limcont} --- Limites e continuidade}

\begin{solution}{exo:g12:limcont:1}
Dividindo por $x^3$:
$\dfrac{2x^3 - x + 1}{5x^3 + x^2} = \dfrac{2 - 1/x^2 + 1/x^3}{5 + 1/x}
\to \dfrac{2}{5}$.

$x^2 + 3x - 1 = x^2(1 + 3/x - 1/x^2) \to +\infty$ quando
$x \to -\infty$, pois $x^2 \to +\infty$ e o parêntese tende a $1$.

Quando $x \to 2^+$, o numerador tende a $3 > 0$ e o denominador a $0^+$,
logo $\dfrac{x+1}{x-2} \to +\infty$.

Para $x > 0$, $-\dfrac1x \leq \dfrac{\cos x}{x} \leq \dfrac1x$, de modo
que o limite é $0$ pelo teorema do confronto.
\end{solution}

\begin{solution}{exo:g12:limcont:2}
\emph{1.} $2x^2 - 3x + 1 = (x-1)(2x-1)$ (a \omterm{def:g11:quad:discriminant}{raiz} $x=1$ salta à vista de
$2 - 3 + 1 = 0$). Para $x \neq 1$, cancele $x - 1$: $f(x) = 2x - 1$.

\emph{2.} Logo $\lim_{x\to1} f(x) = 1$. Definir $f(1) = 1$ estende $f$ à
\omterm{def:g11:func:function}{função} polinomial $x \mapsto 2x-1$ em $\R$, que é \omterm{def:g12:limcont:continuity}{contínua} em $1$.
\end{solution}

\begin{solution}{exo:g12:limcont:3}
Escreva $f(x) = \dfrac{3x - 1}{x + 2} = \dfrac{3(x+2) - 7}{x+2}
= 3 - \dfrac{7}{x+2}$. Quando $x \to \pm\infty$, $f(x) \to 3$: a reta
$y = 3$ é \omterm{def:g12:limcont:asymptote}{assíntota horizontal} nos dois infinitos. Quando
$x \to (-2)^\pm$, $f(x) \to \mp\infty$: a reta $x = -2$ é \omterm{def:g12:limcont:asymptote}{assíntota
vertical}. A curva é uma \omterm{def:g11:func:reference}{hipérbole}, abaixo de $y=3$ para $x > -2$ e acima
dela para $x < -2$.
\end{solution}

\begin{solution}{exo:g12:limcont:4}
Multiplique pelo conjugado:
\[
\sqrt{x^2 + x} - x = \frac{x^2 + x - x^2}{\sqrt{x^2+x} + x}
= \frac{x}{x\left(\sqrt{1 + 1/x} + 1\right)}
= \frac{1}{\sqrt{1 + 1/x} + 1}
\xrightarrow[x\to+\infty]{} \frac{1}{2}.
\]
Do mesmo modo,
\[
\frac{\sqrt{1+x} - 1}{x} = \frac{(1+x) - 1}{x\left(\sqrt{1+x} + 1\right)}
= \frac{1}{\sqrt{1+x} + 1} \xrightarrow[x\to0]{} \frac{1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:limcont:5}
\emph{1.} Como $\sin x \geq -1$, $f(x) \geq x - 1 \to +\infty$, e
concluímos por comparação. E, no entanto, $f'(x) = 1 + \cos x$ anula-se em
todo múltiplo ímpar de $\pi$, e em todo \omterm{def:g10:numbers:interval}{intervalo} $\intco{A}{+\infty}$ a
\omterm{def:g11:func:function}{função} alterna trechos de crescimento estrito; ela é \omterm{def:g12:seq:monotonic}{crescente}, mas não
estritamente, e --- o que mais importa --- seu crescimento nem é
necessário: a comparação basta.

\emph{2.} Para $x > 1$, divida por $x$:
\[
g(x) = \frac{1 + \frac{\sin x}{x}}{1 - \frac{\sin x}{x}},
\]
e $\frac{\sin x}{x} \to 0$ pelo teorema do confronto
($\abs{\sin x} \leq 1$). Logo $g(x) \to \frac{1+0}{1-0} = 1$.
\end{solution}

\begin{solution}{exo:g12:limcont:6}
$f(x) = x^3 - 3x + 1$ é \omterm{def:g12:limcont:continuity}{contínua}, com
$f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$: $f$ cresce em $\intoc{-\infty}{-1}$,
decresce em $\intcc{-1}{1}$ e cresce em $\intco{1}{+\infty}$, com \omterm{def:g10:functions:extrema}{máximo}
local $f(-1) = 3 > 0$ e \omterm{def:g10:functions:extrema}{mínimo} local $f(1) = -1 < 0$,
$\lim_{-\infty} f = -\infty$ e $\lim_{+\infty} f = +\infty$.

Em cada um dos três \omterm{def:g10:numbers:interval}{intervalos} de monotonicidade, $f$ é \omterm{def:g12:limcont:continuity}{contínua},
estritamente \omterm{def:g12:seq:monotonic}{monótona}, e $0$ está estritamente entre os valores ou
limites nos extremos, de modo que o teorema da bijeção dá exatamente um
zero por \omterm{def:g10:numbers:interval}{intervalo}: três zeros ao todo. Verificações de sinal:
$f(-2) = -1 < 0 < f(-1) = 3$, logo um zero em $\intoo{-2}{-1}$;
$f(0) = 1 > 0 > f(1) = -1$, um zero em $\intoo{0}{1}$;
$f(1) = -1 < 0 < f(2) = 3$, um zero em $\intoo{1}{2}$.
\end{solution}

\begin{solution}{exo:g12:limcont:7}
Seja $g(x) = f(x) - x$, \omterm{def:g12:limcont:continuity}{contínua} em $\intcc{0}{1}$. Como
$f(0) \in \intcc{0}{1}$, $g(0) = f(0) \geq 0$; como
$f(1) \in \intcc{0}{1}$, $g(1) = f(1) - 1 \leq 0$. Pelo teorema do valor
intermediário, $g$ se anula em algum $c \in \intcc{0}{1}$, isto é,
$f(c) = c$.
\end{solution}

\begin{solution}{exo:g12:limcont:8}
Sejam $u(t)$ e $d(t)$ as posições (medidas como distância desde a base da
trilha) no instante $t \in \intcc{8}{20}$, na subida e na descida. As duas
são \omterm{def:g12:limcont:continuity}{contínuas}, com $u(8) = 0$, $u(20) = L$ (o comprimento da trilha),
$d(8) = L$ e $d(20) = 0$. A \omterm{def:g11:func:function}{função} $g = u - d$ é \omterm{def:g12:limcont:continuity}{contínua} com
$g(8) = -L \leq 0$ e $g(20) = L \geq 0$, de modo que $g(c) = 0$ para algum
instante $c$, pelo teorema do valor intermediário: no instante $c$, as
posições dos dois dias coincidem.
\end{solution}

\begin{solution}{exo:g12:limcont:9}
\emph{1.} $f'(x) = 5x^4 + 3x^2 + 1 > 0$, de modo que $f$ é estritamente
\omterm{def:g12:seq:monotonic}{crescente} em $\R$; com $\lim_{\pm\infty} f = \pm\infty$ e \omterm{def:g12:limcont:continuity}{continuidade}, o
teorema da bijeção dá um único zero real $\alpha$. Como $f(0) = -1 < 0$ e
$f(1) = 2 > 0$, $\alpha \in \intoo{0}{1}$.

\emph{2.} Mantenha um \omterm{def:g10:numbers:interval}{intervalo} $\intcc{a_n}{b_n}$ com
$f(a_n) < 0 < f(b_n)$, partindo de $\intcc{0}{1}$; a cada passo avalie $f$
no \omterm{prop:g10:coordgeom:midpoint}{ponto médio} $m$ e conserve a metade em que a mudança de sinal persiste.
Após $n$ passos o \omterm{def:g10:numbers:interval}{intervalo} tem comprimento $2^{-n}$, de modo que o \omterm{prop:g10:coordgeom:midpoint}{ponto
médio} aproxima $\alpha$ com erro $2^{-(n+1)}$. Precisamos de
$2^{-(n+1)} \leq 10^{-6}$, isto é,
$n + 1 \geq \frac{6}{\log_{10} 2} \approx 19.93$: $n = 20$ iterações
bastam.
\end{solution}

\begin{solution}{pb:g12:limcont:1}
\textbf{1.} Termos dominantes: $\frac{3x^2}{x^2} \to 3$;
$\frac{2x}{x^2} = \frac2x \to 0$;
$x^3 - x^2 = x^3\left(1 - \frac1x\right) \to +\infty$.

\textbf{2.} $\lim_{x \to \pm\infty} f = 3$: \omterm{def:g12:limcont:asymptote}{assíntota horizontal}
$y = 3$; $\lim_{x \to (-2)^\pm} f = \mp\infty$: \omterm{def:g12:limcont:asymptote}{assíntota
vertical} $x = -2$.

\textbf{3.} $\frac{(x-3)(x+3)}{x - 3} = x + 3 \to 6$. E
$\frac{\sqrt{x+1} - 1}{x} = \frac{x}{x(\sqrt{x+1} + 1)} =
\frac{1}{\sqrt{x+1} + 1} \to \frac12$.

\textbf{4.} $\abs{x \sin\frac1x} \leq \abs x \to 0$, logo o
limite é $0$. E $\abs{\frac{\sin x}{x}} \leq \frac1x \to 0$:
limite $0$.

\textbf{5.} $\frac{\sqrt{4x^2 + x}}{x} =
\sqrt{4 + \frac1x} \to \sqrt4 = 2$ (\omterm{def:g12:limcont:continuity}{continuidade} da raiz
quadrada, composição).

\textbf{6.} $f(x) = x^3 + x$ é \omterm{def:g12:limcont:continuity}{contínua} com
$f(1) = 2 < 5 < 10 = f(2)$: pelo teorema do valor
intermediário, $f(c) = 5$ para algum $c \in \intoo{1}{2}$.

\textbf{7.} $f'(x) = 3x^2 + 1 > 0$: estritamente \omterm{def:g12:seq:monotonic}{crescente} em
$\R$, \omterm{def:g12:limcont:continuity}{contínua}, com limites $\mp\infty$: pelo teorema da
bijeção, todo valor --- em particular $5$ --- é atingido
exatamente uma vez.

\textbf{8.} $p(x) = x^3\left(1 + \frac ax + \frac{b}{x^2} +
\frac{c}{x^3}\right)$: quando $x \to +\infty$, $p \to +\infty$;
quando $x \to -\infty$, $p \to -\infty$. Assim, em algum
\omterm{def:g10:numbers:interval}{intervalo} enorme $\intcc{-M}{M}$ vale $p(-M) < 0 < p(M)$: o
teorema do valor intermediário entrega uma \omterm{def:g11:quad:discriminant}{raiz}. O mesmo
argumento de termo dominante funciona literalmente para
qualquer grau ímpar --- as potências ímpares mantêm sinais
opostos nos dois infinitos.

\textbf{9.} Dividir ao meio $n$ vezes deixa um \omterm{def:g10:numbers:interval}{intervalo} de
comprimento $2^{-n}$; $2^{-20} \approx 9.5 \times 10^{-7} <
10^{-6}$: vinte passos. Newton dobra os dígitos corretos a cada
passo: partindo de um dígito correto, cerca de $3$ a $4$ passos
chegam a seis --- a tartaruga e o galgo da busca de raízes.

\textbf{10.} $g$ não é \omterm{def:g12:limcont:continuity}{contínua} em $\intcc{-1}{1}$: ela explode
em $0$, que o \omterm{def:g10:numbers:interval}{intervalo} contém. A conclusão do teorema evapora
junto com sua hipótese --- confira sempre a \omterm{def:g12:limcont:continuity}{continuidade}
\emph{no \omterm{def:g10:numbers:interval}{intervalo} inteiro}, e não só nos extremos.

\textbf{11.} $g(x) = f(x) - x$ é \omterm{def:g12:limcont:continuity}{contínua};
$g(0) = f(0) \geq 0$ (os valores estão em $\intcc{0}{1}$) e
$g(1) = f(1) - 1 \leq 0$. Pelo teorema do valor intermediário,
$g(c) = 0$ para algum $c$: $f(c) = c$.

\textbf{12.} Idealizando o país como o segmento
$\intcc{0}{1}$: $f$ leva cada lugar ao lugar mostrado
diretamente abaixo dele no mapa amassado; amassar mantém o mapa
sobre o país, de modo que $f$ cai em $\intcc{0}{1}$, e ela é
\omterm{def:g12:limcont:continuity}{contínua} (o papel não se rasga). O \omterm{pb:g10:functions:1}{ponto fixo} $c$ é o ponto
que fica precisamente acima da própria \omterm{def:g10:functions:function}{imagem} --- o teorema de
Brouwer trata do mapa honesto, em duas dimensões.

\textbf{13.} $g(0) + g(\pi) = \left(T(0) - T(\pi)\right) +
\left(T(\pi) - T(2\pi)\right) = T(0) - T(2\pi) = 0$. Assim,
$g(0)$ e $g(\pi)$ são opostos (ou ambos nulos): de qualquer
modo $g$, \omterm{def:g12:limcont:continuity}{contínua}, anula-se em algum
$t^* \in \intcc{0}{\pi}$: $T(t^*) = T(t^* + \pi)$ --- em algum
lugar do equador, dois pontos antípodas concordam na
temperatura, a cada instante.

\textbf{14.} Girar a mesa um quarto de volta troca os papéis
das duas diagonais: o par de pés que pressionava o chão e o par
em que um pé pairava trocam de situação, de modo que a altura
suspensa --- contada com sinal, como ``folga da diagonal
distinguida'' --- assume sinais opostos em $\theta$ e em
$\theta + \frac\pi2$. Um chão \omterm{def:g12:limcont:continuity}{contínuo} torna $h$ \omterm{def:g12:limcont:continuity}{contínua}, e o
teorema do valor intermediário entrega um ângulo $\theta^*$ com
$h(\theta^*) = 0$: quatro pés no chão. (Nada dessa sorte com
uma mesa retangular ou com chão em degraus.)

\textbf{15.} Modelo: construa a \emph{diferença} das duas
grandezas que você quer fazer coincidir ($f(x) - x$;
$T(t) - T(t + \pi)$; posição na subida menos posição na descida,
para a caminhante; a folga com sinal, para a mesa); verifique
que ela assume valores dos dois sinais (muitas vezes por uma
simetria entre as extremidades); invoque o teorema do valor
intermediário sobre uma \omterm{def:g11:func:function}{função} \omterm{def:g12:limcont:continuity}{contínua}. A coincidência fica
forçada.

\textbf{16.} Perguntas em aberto: \emph{quantas} soluções
(respondida pela monotonicidade estrita e pelo teorema da
bijeção, questão 7) e \emph{onde} elas estão (respondida pela
\omterm{thm:g12:limcont:ivt}{dicotomia} ou por Newton, questão 9). O teorema do valor
intermediário é um oráculo puro de existência.

\textbf{17.} $3 - \frac{7}{x + 2}$: \omterm{def:g12:limcont:asymptote}{assíntota vertical}
$x = -2$, horizontal $y = 3$; de cada lado de $-2$ a \omterm{def:g11:func:function}{função} é
\omterm{def:g12:seq:monotonic}{crescente} ($-\frac{7}{x+2}$ cresce onde está definida); a curva
é a \omterm{def:g11:func:reference}{hipérbole} de referência $-\frac7x$ deslocada $2$ para a
esquerda e $3$ para cima --- dois ramos abraçando o cruzamento
das \omterm{def:g12:limcont:asymptote}{assíntotas}.

\textbf{18.} Para $x \neq 1$: $f(x) = x + 1$, logo
$\lim_{x \to 1} f = 2$: definir $f(1) = 2$ restaura a
\omterm{def:g12:limcont:continuity}{continuidade}. A descontinuidade era um furo, não um salto ---
preencha o único ponto que falta e a curva cicatriza:
\emph{removível}.

\textbf{19.} \omterm{def:g12:limcont:continuity}{Contínua} em todo \omterm{def:g10:numbers:interval}{intervalo} entre inteiros
consecutivos; em cada inteiro $n$ ela salta de $n - 1$ para
$n$. Com $a = 0.9$ e $b = 1.1$:
$\lfloor a \rfloor = 0 < \frac12 < 1 = \lfloor b \rfloor$ e,
ainda assim, $\lfloor x \rfloor = \frac12$ não tem solução ---
o salto se teletransporta por cima de $\frac12$, que é
precisamente o que o teorema proíbe às funções \omterm{def:g12:limcont:continuity}{contínuas}.

\textbf{20.} Face local: valor $=$ limite em cada ponto --- o
furo removível (questão 18) a violou de modo curável, e os
saltos da \omterm{def:g11:func:function}{função} piso (questão 19), de modo incurável. Face
global: em um \omterm{def:g10:numbers:interval}{intervalo}, todos os valores intermediários são
visitados --- os certificados de \omterm{def:g11:quad:discriminant}{raiz} (6--8), o \omterm{pb:g10:functions:1}{ponto fixo}, o
equador, a mesa e a caminhante moram todos aqui. O nome oficial
do lema: o \emph{teorema do valor intermediário}.
\end{solution}
