\section*{Capítulo \ref{ch:g12:deriv} --- Derivadas e convexidade}

\begin{solution}{exo:g12:deriv:1}
$f'(x) = 3x^2 - 5$.

Regra do quociente:
$g'(x) = \dfrac{(x^2+1) - x\cdot 2x}{(x^2+1)^2} = \dfrac{1 - x^2}{(x^2+1)^2}$.

Regra da cadeia com $u = 2x+1$:
$h'(x) = 5 \cdot 2 \cdot (2x+1)^4 = 10(2x+1)^4$.

Regra da cadeia com $u = x^2 + 1$:
$k'(x) = \dfrac{2x}{2\sqrt{x^2+1}} = \dfrac{x}{\sqrt{x^2+1}}$.
\end{solution}

\begin{solution}{exo:g12:deriv:2}
$f'(x) = 2x - 3$, logo $f(2) = -1$ e $f'(2) = 1$: a \omterm{def:g12:deriv:tangent}{tangente} é
$y = -1 + (x - 2) = x - 3$. A diferença é
\[
f(x) - (x - 3) = x^2 - 4x + 4 = (x-2)^2 \geq 0 ,
\]
de modo que a curva fica acima da \omterm{def:g12:deriv:tangent}{tangente} em toda parte, tocando-a apenas
em $x = 2$ (como se esperava: $f$ é \omterm{def:g12:deriv:convex}{convexa}).
\end{solution}

\begin{solution}{exo:g12:deriv:3}
Os dois são quocientes de diferenças. Com $f(x) = (1+x)^{100}$ em $0$:
$f'(x) = 100(1+x)^{99}$, de modo que o limite é $f'(0) = 100$. Com
$g(x) = \sqrt{x}$ em $4$: $g'(x) = \frac{1}{2\sqrt{x}}$, de modo que o
limite é $g'(4) = \frac14$.
\end{solution}

\begin{solution}{exo:g12:deriv:4}
\omterm{def:g11:func:function}{Domínio} $\R \setminus \{1\}$. Limites: em $\pm\infty$,
$f(x) \sim x \to \pm\infty$; em $1^\pm$, o numerador $\to 4 > 0$ e o
denominador $\to 0^\pm$, de modo que $f \to \pm\infty$; a reta $x = 1$ é
\omterm{def:g12:limcont:asymptote}{assíntota vertical}.

\[
f'(x) = \frac{2x(x-1) - (x^2+3)}{(x-1)^2} = \frac{x^2 - 2x - 3}{(x-1)^2}
= \frac{(x-3)(x+1)}{(x-1)^2}.
\]
Logo $f' > 0$ em $\intoo{-\infty}{-1}$ e $\intoo{3}{+\infty}$, e
$f' < 0$ em $\intoo{-1}{1}$ e $\intoo{1}{3}$. A \omterm{def:g11:func:function}{função} cresce até um
\omterm{def:g12:deriv:extremum}{máximo local} $f(-1) = -2$, decresce até $-\infty$, salta para $+\infty$
depois da \omterm{def:g12:limcont:asymptote}{assíntota}, decresce até um \omterm{def:g10:functions:extrema}{mínimo} local $f(3) = 6$ e depois
cresce.
\end{solution}

\begin{solution}{exo:g12:deriv:5}
Para $x \in \intoo{0}{15}$, a caixa tem base quadrada de lado $30 - 2x$ e
altura $x$, de modo que
\[
V(x) = x(30 - 2x)^2 .
\]
$V'(x) = (30-2x)^2 + x \cdot 2(30-2x)(-2) = (30-2x)(30 - 2x - 4x)
= (30-2x)(30-6x)$. Em $\intoo{0}{15}$, $30 - 2x > 0$, de modo que $V'$ tem
o sinal de $30 - 6x$: positiva antes de $x = 5$ e negativa depois. O
volume é \omterm{def:g10:functions:extrema}{máximo} para $x = 5$ cm, dando
$V(5) = 5 \times 20^2 = 2000\ \text{cm}^3$.
\end{solution}

\begin{solution}{exo:g12:deriv:6}
\emph{1.} $f'(x) = 4x^3 - 12x^2$ e $f''(x) = 12x^2 - 24x = 12x(x-2)$.
Assim, $f'' > 0$ em $\intoo{-\infty}{0}$ e em $\intoo{2}{+\infty}$
(\omterm{def:g12:deriv:convex}{convexa}), e $f'' < 0$ em $\intoo{0}{2}$ (\omterm{def:g12:deriv:convex}{côncava}). $f''$ muda de sinal em
$0$ e em $2$: os dois são \omterm{def:g12:deriv:inflection}{pontos de inflexão}.

\emph{2.} Em $a = 2$: $f(2) = 16 - 32 + 10 = -6$,
$f'(2) = 32 - 48 = -16$, \omterm{def:g12:deriv:tangent}{tangente} $y = -6 - 16(x-2) = -16x + 26$. A
diferença é
\[
g(x) = f(x) + 16x - 26 = x^4 - 4x^3 + 16x - 16 .
\]
Como $g(2) = g'(2) = g''(2) = 0$, $(x-2)^3$ divide $g$; a divisão dá
$g(x) = (x-2)^3(x+2)$. Perto de $x = 2$, $x + 2 > 0$, de modo que $g$ tem
o sinal de $(x-2)^3$: negativo antes de $2$ e positivo depois. A curva
atravessa a \omterm{def:g12:deriv:tangent}{tangente}: $2$ é de fato um \omterm{def:g12:deriv:inflection}{ponto de inflexão}.
\end{solution}

\begin{solution}{exo:g12:deriv:7}
A \omterm{def:g11:func:function}{função} $f(x) = \sqrt{x}$ satisfaz $f''(x) = -\frac{1}{4}x^{-3/2} < 0$ em
$\intoo{0}{+\infty}$: ela é \omterm{def:g12:deriv:convex}{côncava}, de modo que seu \omterm{def:g10:functions:graph}{gráfico} fica
\emph{abaixo} de cada \omterm{def:g12:deriv:tangent}{tangente}. A \omterm{def:g12:deriv:tangent}{tangente} em $a = 1$ é
\[
y = f(1) + f'(1)(x - 1) = 1 + \tfrac12 (x-1) = \frac{x+1}{2},
\]
donde $\sqrt{x} \leq \frac{x+1}{2}$ para todo $x > 0$. A igualdade
significa que a curva toca a \omterm{def:g12:deriv:tangent}{tangente}, o que só acontece no ponto de
tangência $x = 1$. (Equivalentemente:
$\left(\sqrt x - 1\right)^2 \geq 0$.)
\end{solution}

\begin{solution}{exo:g12:deriv:8}
Pelo \cref{thm:g12:deriv:convexity}, o \omterm{def:g10:functions:graph}{gráfico} de uma \omterm{def:g11:func:function}{função} \omterm{def:g12:deriv:convex}{convexa} e
\omterm{def:g12:deriv:derivative}{derivável} fica acima de cada uma de suas \omterm{def:g12:deriv:tangent}{tangentes}. A \omterm{def:g12:deriv:tangent}{tangente} em $a$ é
horizontal ($f'(a) = 0$), de \omterm{def:g10:algebra:equation}{equação} $y = f(a)$. Logo $f(x) \geq f(a)$
para todo $x \in \R$: $f(a)$ é o \omterm{def:g10:functions:extrema}{mínimo} global.
\end{solution}

\begin{solution}{exo:g12:deriv:9}
\emph{Perímetro fixo.} Um retângulo de lados $x$ e $\frac{p}{2} - x$
($0 < x < \frac p2$) tem área $S(x) = x\left(\frac p2 - x\right)$. Então
$S'(x) = \frac p2 - 2x$ anula-se em $x = \frac p4$, sendo positiva antes e
negativa depois: o \omterm{def:g10:functions:extrema}{máximo} está em $x = \frac{p}{4}$, onde os dois lados
valem $\frac p4$ --- um quadrado.

\emph{Área fixa.} Lados $x$ e $\frac Ax$ dão perímetro
$P(x) = 2\left(x + \frac Ax\right)$, com
$P'(x) = 2\left(1 - \frac{A}{x^2}\right)$, negativa para $x < \sqrt A$ e
positiva depois: \omterm{def:g10:functions:extrema}{mínimo} em $x = \sqrt{A}$, de novo um quadrado.

\emph{Relação.} As duas afirmações são duais. Suponha que um retângulo não
quadrado minimizasse o perímetro com área fixa $A$; o quadrado de mesmo
perímetro teria área $> A$ pela primeira afirmação, e reduzi-lo por
homotetia até a área $A$ diminuiria estritamente seu perímetro ---
contradizendo a minimalidade. Cada afirmação, portanto, implica a outra.
\end{solution}

\begin{solution}{pb:g12:deriv:1}
\textbf{1.} $30x\left(3x^2 + 1\right)^4$; \quad
$\dfrac{x}{\sqrt{x^2 + 9}}$; \quad
$-\dfrac{2x}{\left(x^2 + 1\right)^2}$.

\textbf{2.} $y(4) = 5$, \omterm{def:g10:reffunc:affine}{coeficiente angular} $\frac45$: \omterm{def:g12:deriv:tangent}{tangente}
$y = 5 + \frac45(x - 4)$, isto é, $y = \frac45 x + \frac95$.

\textbf{3.} $f'(x) = \frac{(x^2 + 1) - x \cdot 2x}{(x^2+1)^2}
= \frac{1 - x^2}{(x^2 + 1)^2}$: negativa, positiva, negativa ao
atravessar $-1$ e $1$: \omterm{def:g10:functions:extrema}{mínimo} $f(-1) = -\frac12$, \omterm{def:g10:functions:extrema}{máximo}
$f(1) = \frac12$, com \omterm{def:g12:limcont:asymptote}{assíntota horizontal} $0$ dos dois lados.

\textbf{4.} $g'(x) = 3x^2 - 6x = 3x(x - 2)$: \omterm{def:g12:seq:monotonic}{crescente},
\omterm{def:g10:functions:variations}{decrescente}, \omterm{def:g12:seq:monotonic}{crescente} ao atravessar $0$ e $2$; \omterm{def:g12:deriv:extremum}{máximo local}
$g(0) = 4$, \omterm{def:g10:functions:extrema}{mínimo} local $g(2) = 0$. $g''(x) = 6x - 6$: \omterm{def:g12:deriv:convex}{côncava}
antes de $1$ e \omterm{def:g12:deriv:convex}{convexa} depois: inflexão em $(1, 2)$.

\textbf{5.} \omterm{def:g12:deriv:tangent}{Tangente} em $1$: \omterm{def:g10:reffunc:affine}{coeficiente angular} $g'(1) = -3$:
$y = 2 - 3(x - 1) = -3x + 5$. Diferença:
$x^3 - 3x^2 + 4 - (-3x + 5) = x^3 - 3x^2 + 3x - 1 =
(x - 1)^3$, que muda de sinal em $1$: a \omterm{def:g12:deriv:tangent}{tangente}
\emph{atravessa} a curva --- o comportamento característico de
um \omterm{def:g12:deriv:inflection}{ponto de inflexão}, em que a curva troca de lado.

\textbf{6.} $T(x) = \dfrac{\sqrt{x^2 + 900}}{5} +
\dfrac{\sqrt{(40 - x)^2 + 400}}{2}$. Entrar antes de $0$ ou
depois de $40$ alonga \emph{os dois} trechos: não faz sentido.

\textbf{7.} $T'(x) = \dfrac{x}{5\sqrt{x^2 + 900}} -
\dfrac{40 - x}{2\sqrt{(40 - x)^2 + 400}}$.

\textbf{8.} No triângulo da corrida, o lado ao longo da praia
mede $x$ e a hipotenusa $\sqrt{x^2 + 900}$: o \omterm{def:g11:trigo:cossin}{seno} do ângulo com
a perpendicular é exatamente o quociente deles (cateto oposto
sobre hipotenusa); do mesmo modo,
$\sin\theta_2 = \frac{40 - x}{\sqrt{(40-x)^2 + 400}}$. Então
$T'(x) = 0$ lê-se
$\frac{\sin\theta_1}{5} = \frac{\sin\theta_2}{2}$ --- a lei de
Snell com velocidades $5$ e $2$.

\textbf{9.} $T(24) \approx 20.5$ s; $T(40) = 10 + 10 = 20$ s;
$T(0) = 6 + \sqrt{2000}/2 \approx 28.4$ s. A linha reta perde
até para ``correr o caminho todo e depois nadar em linha reta''
--- e as duas perdem para o trajeto refratado.

\textbf{10.} A \omterm{thm:g12:limcont:ivt}{dicotomia} em $T'$ (negativa em $24$, positiva em
$40$) \omterm{def:g12:seq:limit}{converge} para $x^* \approx 34$ m, com
$T(x^*) \approx 19.5$ s: cerca de um segundo mais rápido que a
linha reta --- a diferença entre um resgate e uma tragédia.

\textbf{11.} O tempo de cada trecho é \omterm{def:g11:func:function}{função} \omterm{def:g12:deriv:convex}{convexa} de $x$ (um
ramo de $\sqrt{\text{trinômio}}$), de modo que $T$ é \omterm{def:g12:deriv:convex}{convexa}, e
um ponto crítico de uma \omterm{def:g11:func:function}{função} \omterm{def:g12:deriv:convex}{convexa} e \omterm{def:g12:deriv:derivative}{derivável} é seu \omterm{def:g10:functions:extrema}{mínimo}
global (\cref{exo:g12:deriv:8}): $x^*$ não é apenas
estacionário, é imbatível.

\textbf{12.} A luz é mais lenta na água; pelo princípio de
Fermat, o caminho ar-água mais rápido se dobra na superfície
exatamente segundo a lei de Snell --- assim, os raios de um
lápis submerso chegam ao olho dobrados e o cérebro, que
extrapola em linha reta, vê o lápis quebrado na linha d'água.

\textbf{13.} $(x^2)'' = 2 > 0$: \omterm{def:g12:deriv:convex}{convexa}. \omterm{def:g12:deriv:convex}{Convexidade} no \omterm{prop:g10:coordgeom:midpoint}{ponto
médio}: $f\left(\frac{a+b}{2}\right) \leq
\frac{f(a) + f(b)}{2}$, que é a afirmação. Na mão:
$\frac{a^2 + b^2}{2} - \left(\frac{a+b}{2}\right)^2 =
\frac{(a - b)^2}{4} \geq 0$.

\textbf{14.} Extraindo raízes quadradas (\omterm{def:g11:func:monotone}{função crescente}) na
questão 13: $\frac{a + b}{2} \leq \sqrt{\frac{a^2 + b^2}{2}}$.
Cadeia completa em $2, 8$: harmônica
$\frac{2 \cdot 16}{10} = 3.2$; geométrica $\sqrt{16} = 4$;
aritmética $5$; quadrática $\sqrt{34} \approx 5.83$: cada \omterm{def:g11:stat:mean}{média}
se curva diante da seguinte.

\textbf{15.} \omterm{def:g12:deriv:tangent}{Tangente} a $\frac1x$ em $1$: valor $1$, \omterm{def:g10:reffunc:affine}{coeficiente
angular} $-1$: $y = 2 - x$. Curvas \omterm{def:g12:deriv:convex}{convexas} ficam acima de suas
\omterm{def:g12:deriv:tangent}{tangentes} (\cref{thm:g12:deriv:convexity}):
$\frac1x \geq 2 - x$ para todo $x > 0$, com igualdade
exatamente no ponto de contato $x = 1$.

\textbf{16.} Na inflexão, o número \emph{diário} de casos (a
\omterm{def:g12:deriv:derivative}{derivada}) atinge o pico: o crescimento deixa de acelerar e
começa a desacelerar --- o primeiro sinal matemático de que a
onda está virando, muito antes de os próprios números caírem.
Antes dela, $f'' > 0$ (cada dia pior que o anterior); depois
dela, $f'' < 0$ (ainda crescendo, mas mais devagar).

\textbf{17.} $S'(r) = 4\pi r - \frac{2V}{r^2} = 0$ dá
$r^3 = \frac{V}{2\pi} = \frac{330}{2\pi}$:
$r \approx 3.74$ cm, e
$h = \frac{330}{\pi r^2} \approx 7.49$ cm.

\textbf{18.} $h = \frac{V}{\pi r^2} = \frac{2\pi r^3}{\pi r^2}
= 2r$ no ótimo: a altura é igual ao diâmetro, qualquer que seja
o volume.

\textbf{19.} $S''(r) = 4\pi + \frac{4V}{r^3} > 0$: \omterm{def:g12:deriv:convex}{convexa}, de
modo que o raio crítico é o \omterm{def:g10:functions:extrema}{mínimo} global. As latas reais são
mais altas para a pegada, o empilhamento, a presença na
prateleira e o metal mais grosso do topo e do fundo --- a
otimização sempre minimiza exatamente o que se escreveu, e não
o que se quis dizer.

\textbf{20.} Salva-vidas: modelar $T(x)$, derivar, Snell,
\omterm{def:g12:deriv:convex}{convexidade}, um segundo economizado. Lata: modelar $S(r)$,
derivar, $h = 2r$, \omterm{def:g12:deriv:convex}{convexidade}, metal economizado. Viga
(\cref{pb:g11:deriv:1}): modelar $S(w)$, derivar,
$d = w\sqrt2$, quadro, rigidez conquistada. E, pelo princípio
de Fermat, a própria luz percorre essa linha de montagem em
cada superfície que encontra --- a natureza foi a primeira
otimizadora.
\end{solution}
