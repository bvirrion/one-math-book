\section*{Capítulo \ref{ch:g12:condprob} --- Probabilidade condicional e independência}

\begin{solution}{exo:g12:condprob:1}
Há $12$ figuras, das quais $4$ são reis:
$\pcond{\text{figura}}{\text{rei}} = \frac{4}{12} = \frac13$.

\omterm{def:g12:condprob:indep}{Independência}: $\P(\text{rei} \cap \text{copas}) = \frac{1}{52}$ (o rei de
copas), e
$\P(\text{rei})\,\P(\text{copas}) = \frac{4}{52}\times\frac{13}{52}
= \frac{1}{52}$. Iguais: os \omterm{def:g11:prob:model}{eventos} são \omterm{def:g12:condprob:indep}{independentes}.
\end{solution}

\begin{solution}{exo:g12:condprob:2}
\emph{1.} Primeiro ramo: vermelha $\frac58$, azul $\frac38$; segundos
ramos (sem reposição): depois de vermelha, vermelha $\frac47$ / azul
$\frac37$; depois de azul, vermelha $\frac57$ / azul $\frac27$.

\emph{2.} $\P(VV) = \frac58 \times \frac47 = \frac{5}{14}$. Pela
\omterm{def:g10:proba:distribution}{probabilidade} total,
\[
\P(\text{segunda vermelha}) = \frac58\cdot\frac47 + \frac38\cdot\frac57
= \frac{20 + 15}{56} = \frac58 .
\]
(O mesmo valor da primeira retirada --- por simetria, a segunda bola é uma
bola da urna escolhida uniformemente ao acaso.)
\end{solution}

\begin{solution}{exo:g12:condprob:3}
$\P(A) = \frac{6}{36} = \frac16$ (seis pares somam $7$),
$\P(B) = \frac16$, e $A \cap B = \{(3,4)\}$ tem \omterm{def:g10:proba:distribution}{probabilidade}
$\frac{1}{36} = \P(A)\P(B)$: $A$ e $B$ são \omterm{def:g12:condprob:indep}{independentes}. (A soma $7$ é
especial: qualquer que seja o primeiro dado, exatamente um valor do
segundo a produz.)

$\P(C) = \frac{5}{36}$ e $B \cap C = \{(3,3)\}$:
$\P(B \cap C) = \frac{1}{36} \neq \frac16 \times \frac{5}{36}
= \frac{5}{216}$. Não \omterm{def:g12:condprob:indep}{independentes}.
\end{solution}

\begin{solution}{exo:g12:condprob:4}
\emph{1.} \omterm{def:g10:proba:distribution}{Probabilidade} total com a partição por máquina:
\[
\P(D) = 0.5\times0.01 + 0.3\times0.02 + 0.2\times0.04
= 0.005 + 0.006 + 0.008 = 0.019 = 1.9\% .
\]

\emph{2.} Bayes:
$\pcond{D}{M_3} = \dfrac{0.008}{0.019} = \dfrac{8}{19} \approx 0.42$. A
máquina que produz apenas um quinto da produção responde por mais de
$40\%$ dos defeitos.
\end{solution}

\begin{solution}{exo:g12:condprob:5}
A condição é
\[
\frac{0.99\,p}{0.99\,p + 0.05\,(1-p)} \geq 0.9 .
\]
O denominador é positivo, de modo que isso equivale a
$0.99p \geq 0.891p + 0.045 - 0.045p$, \emph{isto é},
$0.144\,p \geq 0.045$, \emph{isto é},
\[
p \geq \frac{0.045}{0.144} = 0.3125 .
\]
O teste sozinho alcança $90\%$ de certeza apenas se a doença já atinge
mais de $31\%$ da população testada --- e é por isso que a triagem em
massa de doenças raras exige testes de confirmação.
\end{solution}

\begin{solution}{exo:g12:condprob:6}
\emph{1.} Três \omterm{def:g12:seq:sequence}{sequências} realizam $E$ (cara-cara-coroa, cara-coroa-cara,
coroa-cara-cara), cada uma de \omterm{def:g10:proba:distribution}{probabilidade} $p^2(1-p)$ por \omterm{def:g12:condprob:indep}{independência}:
$\P(E) = 3p^2(1 - p)$.

\emph{2.} $f(p) = 3p^2 - 3p^3$ tem $f'(p) = 6p - 9p^2 = 3p(2 - 3p)$,
positiva e depois negativa em $\intoo{0}{1}$: \omterm{def:g10:functions:extrema}{máximo} em $p = \frac23$,
onde $\P(E) = 3\cdot\frac49\cdot\frac13 = \frac49$.
\end{solution}

\begin{solution}{exo:g12:condprob:7}
Digamos que você escolheu a porta 1 e o apresentador abriu a porta 3
(\omterm{def:g11:prob:model}{evento} $H_3$). Com $B_i$ = ``prêmio atrás da porta $i$'',
$\P(B_i) = \frac13$ e
\[
\pcond{B_1}{H_3} = \frac12, \qquad
\pcond{B_2}{H_3} = 1, \qquad
\pcond{B_3}{H_3} = 0
\]
(se o prêmio está atrás da sua porta, o apresentador escolhe ao acaso
entre as portas 2 e 3; se está atrás da porta 2, ele é obrigado a abrir a
3). Bayes:
\[
\pcond{H_3}{B_1}
= \frac{\frac13\cdot\frac12}{\frac13\cdot\frac12 + \frac13\cdot1 + 0}
= \frac{1/6}{1/2} = \frac13,
\qquad
\pcond{H_3}{B_2} = \frac{\frac13}{\frac12} = \frac23 .
\]
Ficar ganha com \omterm{def:g10:proba:distribution}{probabilidade} $\frac13$; trocar, com \omterm{def:g10:proba:distribution}{probabilidade}
$\frac23$.
\end{solution}

\begin{solution}{exo:g12:condprob:8}
\emph{1.} A maioria erra quando ao menos duas das três cópias são
invertidas:
\[
3\varepsilon^2(1-\varepsilon) + \varepsilon^3
= 3(0.01)(0.9) + 0.001 = 0.028,
\]
bem menor que $\varepsilon = 0.1$: a codificação por repetição funciona.

\emph{2.} Se $1$ foi enviado (como $111$), receber $101$ exige exatamente
uma inversão: \omterm{def:g10:proba:distribution}{probabilidade} $\varepsilon(1-\varepsilon)^2 = 0.081$. Se $0$
foi enviado ($000$), são necessárias duas inversões:
$\varepsilon^2(1-\varepsilon) = 0.009$. Bayes com \omterm{def:g10:proba:distribution}{probabilidades} a priori
iguais:
\[
\pcond{101}{\,\text{enviado }1} = \frac{0.081}{0.081 + 0.009} = 0.9 .
\]
A decodificação por maioria ($101 \mapsto 1$) é de fato o palpite mais
provável.
\end{solution}

\begin{solution}{pb:g12:condprob:1}
\textbf{1.} $\P(\text{vermelha}) = \frac12 \cdot \frac23 +
\frac12 \cdot \frac14 = \frac13 + \frac18 = \frac{11}{24}$.

\textbf{2.} $\P(\text{I} \mid \text{vermelha}) =
\dfrac{\frac12 \cdot \frac23}{\frac{11}{24}} =
\dfrac{1/3}{11/24} = \dfrac{8}{11}$.

\textbf{3.} $\P(\text{par} \cap \leq 4) = \P(\{2, 4\}) =
\frac13$ e $\P(\text{par})\P(\leq 4) = \frac12 \cdot
\frac23 = \frac13$: \omterm{def:g12:condprob:indep}{independentes} (inesperadamente!). Para
$\leq 3$: $\P(\{2\}) = \frac16$ contra
$\frac12 \cdot \frac12 = \frac14$: dependentes.

\textbf{4.} Os \omterm{def:g10:proba:operations}{complementares} de \omterm{def:g11:prob:model}{eventos} \omterm{def:g12:condprob:indep}{independentes} são
eles mesmos \omterm{def:g12:condprob:indep}{independentes}, logo
\[
\P(\text{nenhum}) = 0.1 \times 0.1 = 0.01,
\qquad
\P(\text{ao menos um}) = 0.99 .
\]

\textbf{5.} $\frac{4}{52} \times \frac{3}{51} =
\frac{1}{221}$.

\textbf{6.} Doentes: $100$; saudáveis: $99\,900$. Positivos:
$99$ verdadeiros ($99\,\%$ de $100$) e $4\,995$ falsos
($5\,\%$ de $99\,900$): $5\,094$ positivos ao todo.

\textbf{7.} $\P(\text{doente} \mid +) = \frac{99}{5\,094}
\approx 1.9\,\%$: entre as pessoas que o teste aponta, menos
de uma em cinquenta está de fato doente.

\textbf{8.} $\dfrac{0.001 \times 0.99}{0.001 \times 0.99 +
0.999 \times 0.05} = \dfrac{0.00099}{0.05094} \approx 0.019$:
as contagens e a fórmula concordam.

\textbf{9.} Confundem-se: $\P(+ \mid \text{doente}) =
99\,\%$ (a qualidade anunciada do teste) com
$\P(\text{doente} \mid +)$ (a pergunta real do paciente). Com
uma taxa de base de $\frac{1}{1000}$, os doentes são tão raros
que mesmo um vazamento de $5\,\%$ da enorme maioria saudável
afoga os positivos verdadeiros na proporção de cinquenta para
um.

\textbf{10.} $\dfrac{0.019 \times 0.99}{0.019 \times 0.99 +
0.981 \times 0.05} \approx 28\,\%$. Cada positivo
independente multiplica as chances pelo mesmo fator: as
evidências se acumulam, e um terceiro positivo passaria de
$90\,\%$ --- e é por isso que existem os testes
confirmatórios.

\textbf{11.} $\dfrac{0.05 \times 0.99}{0.05 \times 0.99 +
0.95 \times 0.05} \approx 51\,\%$: em um grupo de risco, o
mesmo teste é informativo. Moral: faça a triagem onde a taxa
de base é substancial; rastrear em massa uma condição rara
fabrica alarmes falsos aos milhares.

\textbf{12.} Número esperado de coincidências inocentes:
$10^7 \times 10^{-6} = 10$. Entre as cerca de $11$ pessoas
compatíveis ($10$ inocentes, $1$ culpado --- se o autor está
na cidade), o suspeito encontrado \emph{apenas pela
coincidência} é inocente com \omterm{def:g10:proba:distribution}{probabilidade} de cerca de
$\frac{10}{11} \approx 91\,\%$.

\textbf{13.} O promotor citou
$\P(\text{coincidência} \mid \text{inocente}) = 10^{-6}$; o
tribunal precisava de
$\P(\text{inocente} \mid \text{coincidência})$ --- a confusão
da questão 9 sob juramento. (E, no caso real que popularizou
esses erros, um segundo equívoco agravou tudo: veja a questão
14.)

\textbf{14.} Quadrado ingênuo: $\left(\frac{1}{8500}\right)^2
\approx \frac{1}{72 \text{ milhões}}$. Com a segunda tragédia
cinco vezes mais provável dada a primeira:
$\frac{1}{8500} \times \frac{5}{8500} \approx
\frac{1}{14.5 \text{ milhões}}$ --- cinco vezes maior, e
ainda ignorando que a hipótese \emph{alternativa} também tem
de ser pesada por Bayes. Lição: multiplicar \omterm{def:g10:proba:distribution}{probabilidades}
exige \omterm{def:g12:condprob:indep}{independência}, e causas comuns (genética, ambiente) a
destroem.

\textbf{15.} Bayes multiplica as razões de verossimilhança de
\emph{todas} as evidências em uma única \omterm{def:g10:proba:distribution}{probabilidade} a
posteriori: a coincidência, o álibi, as fibras, cada uma
deslocando as chances. A frase do jurado: \emph{pergunte
quão provável é a evidência sob as duas hipóteses e nunca
tome uma condicional pela sua inversa}.

\textbf{16.} $\dfrac{0.4 \times 0.15}{0.4 \times 0.15 +
0.6 \times 0.005} = \dfrac{0.060}{0.063} \approx 95\,\%$ de
spam: uma única palavra quase decide a questão.

\textbf{17.} Cinco cópias falham quando $3$, $4$ ou $5$ se
invertem:
$\binom53 \varepsilon^3(1-\varepsilon)^2 + \binom54
\varepsilon^4 (1 - \varepsilon) + \varepsilon^5 \approx
0.0086$ --- contra $0.028$ para três cópias: a repetição
compra confiabilidade ao preço de banda.

\textbf{18.} Número par de inversões:
$\binom40 0.9^4 + \binom42 0.9^2 0.1^2 + \binom44 0.1^4
= 0.6561 + 0.0486 + 0.0001 = 0.7048$; e
$\frac{1 + 0.8^4}{2} = \frac{1 + 0.4096}{2} = 0.7048$: a
fórmula esperta acompanha o viés $(1 - 2\varepsilon)$ ao
longo de cada repetidor.

\textbf{19.} As palavras de spam andam em bando --- uma
mensagem com ``sorteio'' muito provavelmente contém
``ganhe'', de modo que a evidência conjunta é mais fraca do
que o produto pretende. O filtro ainda funciona porque o
exagero costuma empurrar na \emph{direção certa}: classificar
exige o lado correto dos $50\,\%$, não a \omterm{def:g10:proba:distribution}{probabilidade} a
posteriori correta.

\textbf{20.} Consultório: uma condição rara transforma um
teste forte em um veredicto fraco --- atualize a partir da
taxa de base e depois acumule. Tribunal: uma coincidência
rara em uma cidade grande indicia a cidade, não o homem ---
verossimilhanças não são veredictos. Caixa de entrada: muitos
indícios fracos, multiplicados, separam a correspondência ---
mesmo ingenuamente. As três armadilhas: esquecer a
\omterm{def:g10:proba:distribution}{probabilidade} a priori e ser enganado; fingir \omterm{def:g12:condprob:indep}{independência} e
ser muito enganado; ler as evidências aos pedaços e nunca
saber o que se sabe.
\end{solution}
