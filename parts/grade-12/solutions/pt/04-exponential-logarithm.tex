\section*{Capítulo \ref{ch:g12:exp} --- Exponencial e logaritmo}

\begin{solution}{exo:g12:exp:1}
$\dfrac{\eu^{3x}\,\eu^{-x+1}}{\eu^{x}} = \eu^{3x - x + 1 - x} = \eu^{x+1}$.

$\ln\!\left(\dfrac{\eu^2\sqrt{\eu}}{\eu^{-3}}\right)
= 2 + \tfrac12 + 3 = \tfrac{11}{2}$.
\end{solution}

\begin{solution}{exo:g12:exp:2}
\emph{(a)} Ponha $X = \eu^x > 0$: $X^2 - 3X + 2 = 0$ dá $X = 1$ ou
$X = 2$, ambos positivos, logo $x = 0$ ou $x = \ln 2$.

\emph{(b)} \omterm{def:g11:func:function}{Domínio}: $x - 1 > 0$ e $x + 2 > 0$, logo $x > 1$. A \omterm{def:g10:algebra:equation}{equação} vira
$\ln\bigl((x-1)(x+2)\bigr) = \ln 4$, portanto $(x-1)(x+2) = 4$,
\emph{isto é}, $x^2 + x - 6 = 0$, logo $x = 2$ ou $x = -3$. Só $x = 2$
está no \omterm{def:g11:func:function}{domínio}: $S = \{2\}$.
\end{solution}

\begin{solution}{exo:g12:exp:3}
Divida por $\eu^x$:
$\dfrac{1 - x^2\eu^{-x}}{1 + \eu^{-x}} \to \dfrac{1-0}{1+0} = 1$, usando
$x^2 \eu^{-x} \to 0$ (\cref{thm:g12:exp:growth}).

$\dfrac{\ln x}{\sqrt x} \to 0$ pela \cref{prop:g12:exp:lnanalytic} (caso
$n = 2$).

$x^2 \ln x = x \cdot (x \ln x) \to 0 \times 0 = 0$.

$x - \ln x = x\left(1 - \frac{\ln x}{x}\right) \to +\infty$, pois
$\frac{\ln x}{x} \to 0$.
\end{solution}

\begin{solution}{exo:g12:exp:4}
$f'(x) = \eu^{-x} - x\eu^{-x} = (1 - x)\eu^{-x}$, com o sinal de
$1 - x$: $f$ cresce em $\intoc{-\infty}{1}$ e decresce em
$\intco{1}{+\infty}$, com \omterm{def:g10:functions:extrema}{máximo} global $f(1) = \eu^{-1}$.

Limites: quando $x \to +\infty$, $f(x) = \frac{x}{\eu^x} \to 0$
(\cref{thm:g12:exp:growth}); quando $x \to -\infty$, $x \to -\infty$ e
$\eu^{-x} \to +\infty$, logo $f(x) \to -\infty$.

$f''(x) = -\eu^{-x} - (1-x)\eu^{-x} = (x - 2)\eu^{-x}$, que muda de sinal
em $x = 2$: \omterm{def:g12:deriv:inflection}{ponto de inflexão} em $\bigl(2,\, 2\eu^{-2}\bigr)$.
\end{solution}

\begin{solution}{exo:g12:exp:5}
Seja $g(x) = x - \ln(1+x)$ em $\intoo{-1}{+\infty}$. Então
$g'(x) = 1 - \frac{1}{1+x} = \frac{x}{1+x}$, negativa em
$\intoo{-1}{0}$ e positiva em $\intoo{0}{+\infty}$: $g$ atinge seu \omterm{def:g10:functions:extrema}{mínimo}
$g(0) = 0$, logo $g \geq 0$, com igualdade só em $0$. Portanto
$\ln(1+x) \leq x$.

Aplique isso com $x = \frac1n$:
$n\ln\left(1 + \frac1n\right) \leq 1$, logo
$\left(1+\frac1n\right)^n = \eu^{\,n\ln(1+1/n)} \leq \eu$. Aplique com
$x = -\frac{1}{n+1} > -1$:
$\ln\left(1 - \frac{1}{n+1}\right) \leq -\frac{1}{n+1}$, logo
$-(n+1)\ln\left(1 - \frac{1}{n+1}\right) \geq 1$ e
$\left(1 - \frac{1}{n+1}\right)^{-(n+1)} \geq \eu$.
\end{solution}

\begin{solution}{exo:g12:exp:6}
\emph{1.} Cada ano multiplica o capital por $1.03$:
$C_n = C_0 (1.03)^n$.

\emph{2.} $C_n \geq 2C_0 \iff (1.03)^n \geq 2 \iff n \ln 1.03 \geq \ln 2
\iff n \geq \frac{\ln 2}{\ln 1.03} \approx 23.45$: o capital dobra após
$24$ anos.

\emph{3.} $\frac{70}{3} \approx 23.3$, próximo do valor exato
$\frac{\ln 2}{\ln 1.03}$. A regra funciona porque
$\ln(1 + r) \approx r$ para $r$ pequeno, de modo que
$\frac{\ln 2}{\ln(1+r)} \approx \frac{0.693}{r} \approx \frac{70}{100r}$.
\end{solution}

\begin{solution}{exo:g12:exp:7}
$\eu^{2x} > \eu^{x+1} \iff 2x > x + 1 \iff x > 1$ (a exponencial é
estritamente \omterm{def:g12:seq:monotonic}{crescente}): $S = \intoo{1}{+\infty}$.

\omterm{def:g11:func:function}{Domínio} da segunda inequação: $x^2 - 1 > 0$ e $x + 5 > 0$, \emph{isto é},
$x \in \intoo{-5}{-1} \cup \intoo{1}{+\infty}$. Nesse \omterm{def:g11:func:function}{domínio}, sendo
$\ln$ estritamente \omterm{def:g12:seq:monotonic}{crescente},
\[
\ln(x^2-1) \leq \ln(x+5) \iff x^2 - 1 \leq x + 5 \iff x^2 - x - 6 \leq 0
\iff x \in \intcc{-2}{3}.
\]
Intersectando com o \omterm{def:g11:func:function}{domínio}:
$S = \intco{-2}{-1} \cup \intoc{1}{3}$.
\end{solution}

\begin{solution}{exo:g12:exp:8}
\emph{1.} $f'(x) = \dfrac{\eu^x x - \eu^x}{x^2} =
\dfrac{(x-1)\eu^x}{x^2}$, negativa em $\intoo{0}{1}$ e positiva em
$\intoo{1}{+\infty}$: \omterm{def:g10:functions:extrema}{mínimo} $f(1) = \eu$. Limites: $f \to +\infty$ em
$0^+$ (numerador $\to 1$, denominador $\to 0^+$) e em $+\infty$
(\cref{thm:g12:exp:growth}).

\emph{2.} Para $x > 0$, $\eu^x = kx \iff f(x) = k$. Pelo \omterm{def:g10:functions:table}{quadro de
variação} ($+\infty \searrow \eu \nearrow +\infty$, \omterm{def:g12:limcont:continuity}{contínua} e
estritamente \omterm{def:g12:seq:monotonic}{monótona} de cada lado): nenhuma solução para $k < \eu$;
exatamente uma ($x = 1$) para $k = \eu$; exatamente duas para $k > \eu$
(uma em $\intoo{0}{1}$ e outra em $\intoo{1}{+\infty}$, pelo teorema da
bijeção em cada \omterm{def:g10:numbers:interval}{intervalo}).
\end{solution}

\begin{solution}{exo:g12:exp:9}
\emph{1.} Direto da primeira desigualdade do \cref{exo:g12:exp:5}:
$u_n = \eu^{\,n\ln(1+1/n)} \leq \eu^1 = \eu$.

\emph{2.} Escreva
\[
\ln u_n = n \ln\!\left(1 + \frac1n\right)
= \frac{\ln\!\left(1 + \frac1n\right) - \ln 1}{\frac1n},
\]
um quociente de diferenças de $\ln$ no ponto $1$ com incremento
$h = \frac1n \to 0$. Como $\ln$ é \omterm{def:g12:deriv:derivative}{derivável} em $1$ com \omterm{def:g12:deriv:derivative}{derivada} $1$,
$\ln u_n \to 1$. Pela \omterm{def:g12:limcont:continuity}{continuidade} de $\exp$
(\cref{prop:g12:limcont:seqcont}), $u_n = \eu^{\ln u_n} \to \eu^1 = \eu$.
\end{solution}

\begin{solution}{pb:g12:exp:1}
\textbf{1.} $x = \frac{\ln 5}{2}$. \quad
$3x - 1 = \eu^2$: $x = \frac{\eu^2 + 1}{3}$. \quad Com
$u = \eu^x$: $u^2 - 3u + 2 = (u - 1)(u - 2) = 0$: $x = 0$ ou
$x = \ln 2$.

\textbf{2.} $\frac{3\ln 2}{\ln 2} = 3$; \quad
$3 + \frac12 = \frac72$; \quad $\frac52$.

\textbf{3.} $f'(x) = \eu^x - 1$: negativa antes de $0$ e
positiva depois: \omterm{def:g10:functions:extrema}{mínimo} $f(0) = 0$, de modo que $f \geq 0$ em
toda parte: $\eu^x \geq 1 + x$. Substituindo
$x = \ln(1 + u)$: $1 + u \geq 1 + \ln(1 + u)$, isto é,
$\ln(1 + u) \leq u$.

\textbf{4.} $x^2\eu^{-x} = \frac{x^2}{\eu^x} \to 0$ e
$\frac{\ln x}{x} \to 0$: as exponenciais esmagam as potências,
e as potências esmagam os \omterm{def:g12:exp:ln}{logaritmos}. E
$\frac{\eu^x - 1}{x} = \frac{\eu^x - \eu^0}{x - 0} \to
\exp'(0) = 1$.

\textbf{5.} $f'(x) = \ln x + 1$: nula em $\eu^{-1}$, negativa
antes e positiva depois: \omterm{def:g10:functions:extrema}{mínimo}
$f\!\left(\frac1\eu\right) = -\frac1\eu$; e $x \ln x \to 0$ em
$0^+$: a curva sai da origem, mergulha até $-\frac1\eu$ e sobe.

\textbf{6.} $n = \frac{\ln 2}{\ln 1.03} \approx 23.4$ anos.

\textbf{7.} Exato: $23.4$, $11.9$, $8.0$ anos; regra de 72:
$24$, $12$, $8$. Como $\ln(1+r) \approx r$,
$\frac{\ln 2}{\ln(1+r)} \approx \frac{0.693}{r}$, isto é,
$\frac{69.3}{\text{taxa em }\%}$; os banqueiros arredondam para
$72$ porque ele se divide lindamente por $2, 3, 4, 6, 8, 9,
12$ --- a aritmética mental vale mais que uma casa decimal de
precisão.

\textbf{8.} $2^{t/20} = \eu^{t \ln 2 / 20}$:
$\lambda = \frac{\ln 2}{20}$ por minuto. Após
$24 \times 60 = 1440$ minutos:
$2^{72} \approx 4.7 \times 10^{21}$ bactérias --- mais que os
grãos de areia da Terra, a partir de uma célula em um dia. O
modelo só é honesto enquanto houver comida e espaço: o
crescimento real se dobra na curva logística em S (a curva do
epidemiologista do \cref{pb:g12:deriv:1}).

\textbf{9.} Às 23h ($8$ horas):
$100 \times 2^{-8/5} \approx 33$ mg. Abaixo de $10$ mg:
$2^{-t/5} < 0.1$ dá $t > 5\,\frac{\ln 10}{\ln 2} \approx
16.6$ h: por volta das 7h40 da manhã seguinte --- o cafezinho
das três tem cauda longa.

\textbf{10.} Anual: $2$. Mensal:
$\left(1 + \frac{1}{12}\right)^{12} \approx 2.613$. Diária:
$\approx 2.715$. O teto: $\eu = 2.71828\ldots$
(\cref{exo:g12:exp:9}) --- capitalizando continuamente, a
ganância \omterm{def:g12:seq:limit}{converge}.

\textbf{11.} Se os casos crescem exponencialmente,
$c(t) = c_0 \eu^{\lambda t}$, então
$\ln c(t) = \ln c_0 + \lambda t$: uma reta de \omterm{def:g10:reffunc:affine}{coeficiente
angular} $\lambda$ --- a taxa de crescimento. Um trecho retilíneo
no \omterm{def:g10:functions:graph}{gráfico} logarítmico \emph{é} a fase exponencial, e sua
inclinação é o compasso da epidemia.

\textbf{12.} $120 - 60 = 60$ dB significa
$10\log_{10}(I_2/I_1) = 60$: o show é $10^6$ --- um milhão de
vezes --- mais intenso que a conversa.

\textbf{13.} Amplitude: $10^{7-5} = 100$. Energia:
$100^{3/2} = 1000$: dois pontos de magnitude escondem três
ordens de grandeza em energia.

\textbf{14.} $10^{6.5 - 2} = 10^{4.5} \approx 32\,000$: o suco
de limão é trinta mil vezes mais ácido que o leite --- o pH
comprime os abismos da química em uma escala de bolso.

\textbf{15.} $\log_2\!\frac{4186}{27.5} =
\frac{\ln(4186/27.5)}{\ln 2} \approx 7.25$: um piano abrange
pouco mais de sete oitavas. Os sentidos respondem a
\emph{razões} de estímulo --- dobrar a intensidade parece um
degrau, seja qual for o ponto de partida ---, de modo que a
percepção é construída sobre uma escala logarítmica, e as
unidades que inventamos para ela também.

\textbf{16.} Encostar a régua de $a$ na posição de $b$ soma os
comprimentos $\ln a + \ln b$, e a graduação que se encontra
nesse comprimento total marca
$\eu^{\ln a + \ln b} = ab$: a régua de cálculo faz produtos
somando \omterm{def:g12:exp:ln}{logaritmos} --- $\ln(ab) = \ln a + \ln b$
(\cref{prop:g12:exp:lnalg}) entalhado em madeira.

\textbf{17.} O \omterm{def:g10:functions:extrema}{mínimo} de $\frac{\eu^x}{x}$ em
$\intoo{0}{+\infty}$ é $\eu$, em $x = 1$: nenhuma solução para
$k < \eu$, exatamente uma para $k = \eu$ e duas para
$k > \eu$. Verificação da \omterm{def:g12:deriv:tangent}{tangente}: em $a = 1$ a \omterm{def:g12:deriv:tangent}{tangente} a
$\eu^x$ é $y = \eu + \eu(x - 1) = \eu x$: ela passa pela origem
--- a reta do limiar, tocando a curva em $(1, \eu)$.

\textbf{18.} $n \ln\!\left(1 - \frac1n\right) =
\frac{\ln(1 - 1/n)}{-1/n} \times (-1) \to -1$, logo
$\left(1 - \frac1n\right)^n \to \eu^{-1} \approx 0.368$: o
$0.999^{1000} \approx 0.368$ do \cref{pb:g11:binom:1},
explicado --- a constante do quase da loteria é $\frac1\eu$.

\textbf{19.} Os dois são a forma-limite de ``muitos \omterm{def:g11:prob:model}{eventos}
independentes, cada um individualmente improvável'': a chance
de que \emph{nenhuma} de $n$ oportunidades de tamanho
$\frac1n$ se realize tende a $\eu^{-1}$, e a regra da
secretária ajusta sua janela de rejeição para que o sucesso se
concentre nessa mesma constante. $\frac1\eu \approx 0.368$.

\textbf{20.} O teto da capitalização; a única base cuja
\omterm{def:g12:deriv:tangent}{tangente} em $0$ tem \omterm{def:g10:reffunc:affine}{coeficiente angular} $1$ (razão pela qual o
cálculo a adora); o crescimento que ultrapassa toda potência; a
constante em que os quases e a parada ótima se acomodam; e o
\omterm{def:g12:exp:ln}{logaritmo} --- a multiplicação virada adição, e as faixas mais
selvagens do mundo dobradas sobre uma régua.
\end{solution}
