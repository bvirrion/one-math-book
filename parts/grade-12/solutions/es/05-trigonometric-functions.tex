\section*{Capítulo \ref{ch:g12:trigo} --- Funciones trigonométricas}

\begin{solution}{exo:g12:trigo:1}
$\cos\frac{2\pi}{3} = \cos\left(\pi - \frac\pi3\right) = -\cos\frac\pi3
= -\frac12$.

$\sin\frac{5\pi}{6} = \sin\left(\pi - \frac\pi6\right) = \sin\frac\pi6
= \frac12$.

$\cos\frac{7\pi}{4} = \cos\left(2\pi - \frac{\pi}{4}\right)
= \cos\left(-\frac\pi4\right) = \frac{\sqrt2}{2}$.

$\sin\left(-\frac\pi3\right) = -\sin\frac\pi3 = -\frac{\sqrt3}{2}$.
\end{solution}

\begin{solution}{exo:g12:trigo:2}
$ f'(x) = 2\sin x\cos x = \sin 2x $ (regla de la cadena).

$ g'(x) = -3\sin(3x) + \sin x + x\cos x $ (reglas de cadena y producto).

Regla del cociente:
\[
h'(x) = \frac{\cos x\,(2 + \cos x) - \sin x\,(-\sin x)}{(2+\cos x)^2}
= \frac{2\cos x + \cos^2 x + \sin^2 x}{(2+\cos x)^2}
= \frac{2\cos x + 1}{(2+\cos x)^2}.
\]
\end{solution}

\begin{solution}{exo:g12:trigo:3}
\emph{(a)} una solución particular de $\sin x = \frac{\sqrt3}{2}$ es
$\frac\pi3$. Soluciones generales: $ x = \frac\pi3 + 2k\pi $ o
$ x = \pi - \frac\pi3 + 2k\pi = \frac{2\pi}{3} + 2k\pi $. En
$\intoc{-\pi}{\pi}$: $ x \in \left\{\frac\pi3, \frac{2\pi}{3}\right\}$.

\emph{(b)} $\cos 2x = \frac{\sqrt2}{2} = \cos\frac\pi4$ da
$2x = \pm\frac\pi4 + 2k\pi $, entonces $ x = \pm\frac\pi8 + k\pi $. En
$\intoc{-\pi}{\pi}$:
$ x \in \left\{-\frac{7\pi}{8}, -\frac\pi8, \frac\pi8, \frac{7\pi}{8}\right\}$.
\end{solution}

\begin{solution}{exo:g12:trigo:4}
Con $ a = \frac\pi3$, $ b = \frac\pi4$:
\[
\cos\frac{\pi}{12} = \cos a\cos b + \sin a \sin b
= \frac12\cdot\frac{\sqrt2}{2} + \frac{\sqrt3}{2}\cdot\frac{\sqrt2}{2}
= \frac{\sqrt2 + \sqrt6}{4},
\]
\[
\sin\frac{\pi}{12} = \sin a\cos b - \cos a \sin b
= \frac{\sqrt3}{2}\cdot\frac{\sqrt2}{2} - \frac12\cdot\frac{\sqrt2}{2}
= \frac{\sqrt6 - \sqrt2}{4}.
\]
\end{solution}

\begin{solution}{exo:g12:trigo:5}
Colocar $ s = \sin x $: $2s^2 - s - 1 = (2s + 1)(s - 1) \geq 0$ si y si
$ s \leq -\frac12$ o $ s = 1$.

$\sin x = 1$ da $ x = \frac\pi2$. $\sin x \leq -\frac12$: en la unidad
círculo, el arco debajo de la línea horizontal $ y = -\frac12$, que en
$\intcc{0}{2\pi}$ es $ x \in \intcc{\frac{7\pi}{6}}{\frac{11\pi}{6}}$.
Por eso
\[
S = \left\{\frac{\pi}{2}\right\} \cup
\intcc{\frac{7\pi}{6}}{\frac{11\pi}{6}} .
\]
\end{solution}

\begin{solution}{exo:g12:trigo:6}
$\dfrac{\sin 3x}{x} = 3\,\dfrac{\sin 3x}{3x} \to 3 \cdot 1 = 3$
(composición con $ u = 3x \to 0$).

Usando $1 - \cos x = \dfrac{\sin^2 x}{1 + \cos x}$:
\[
\frac{1 - \cos x}{x^2}
= \left(\frac{\sin x}{x}\right)^{\!2} \frac{1}{1 + \cos x}
\xrightarrow[x\to0]{} 1 \cdot \frac12 = \frac12 .
\]

Sustituyendo $ u = \frac1x \to 0^+$:
$ x \sin\frac1x = \frac{\sin u}{u} \to 1$.
\end{solution}

\begin{solution}{exo:g12:trigo:7}
$ f'(x) = 1 - \sin x \geq 0$ para todo $ x $, con igualdad exactamente cuando
$\sin x = 1$, \emph{es decir.} $ x = \frac\pi2 + 2k\pi $. En $\intcc{0}{2\pi}$, $ f $
es \omterm{def:g12:seq:monotonic}{creciente} de $ f(0) = 1$ a $ f(2\pi) = 2\pi + 1$ (el cero de $ f'$ en
$\frac{\pi}{2}$ es aislado, por lo que el aumento es par estricto); \omterm{def:g10:functions:extrema}{mínimo} $1$ en
$0$, \omterm{def:g10:functions:extrema}{máximo} $2\pi + 1$ en $2\pi $. En $\R $, $ f' \geq 0$ en todas partes con solo
ceros aislados, entonces $ f $ es (estrictamente) \omterm{def:g12:seq:monotonic}{creciente} por
\cref{thm:g12:deriv:variations}, para todo $ f'$ desaparece en cada
$\frac\pi2 + 2k\pi $.
\end{solution}

\begin{solution}{exo:g12:trigo:8}
\emph{1.} Sea $ R = \sqrt{A^2 + B^2} > 0$. el punto
$\left(\frac AR, \frac BR\right)$ se encuentra en el círculo unitario, por lo que existe
$\varphi $ con $\cos\varphi = \frac AR $ y $\sin\varphi = \frac BR $. Entonces
\[
R\cos(x - \varphi) = R\bigl(\cos x\cos\varphi + \sin x \sin\varphi\bigr)
= A\cos x + B\sin x = f(x).
\]

\emph{2.} Desde $\cos $ tiene \omterm{def:g10:stats:quartiles}{rango} $\intcc{-1}{1}$, $ f $ tiene \omterm{def:g10:functions:extrema}{máximo} $ R $ y
\omterm{def:g10:functions:extrema}{mínimo} $-R $. Para $\cos x + \sin x $: $ A = B = 1$, $ R = \sqrt2$,
$\varphi = \frac\pi4$, entonces el \omterm{def:g10:algebra:equation}{ecuación} se convierte
$\sqrt2\cos\left(x - \frac\pi4\right) = 1$, \emph{es decir.}
$\cos\left(x - \frac\pi4\right) = \frac{\sqrt2}{2}$, donación
$ x - \frac\pi4 = \pm\frac\pi4 + 2k\pi $: en $\intoc{-\pi}{\pi}$,
$ x \in \left\{0, \frac\pi2\right\}$.
\end{solution}

\begin{solution}{exo:g12:trigo:9}
$(\sin)'' = (\cos)' = -\sin $ y $(\cos)'' = (-\sin)' = -\cos $: ambos resuelven
$ y'' = -y $.

Sea $ g = y - a\cos - b\sin $. Entonces $ g'' = -g $ (linealidad de diferenciación),
$ g(0) = y(0) - a = 0$ y $ g'(0) = y'(0) - b = 0$. Colocar
$ E = g^2 + (g')^2$. Entonces
\[
E' = 2g g' + 2g' g'' = 2g g' - 2g' g = 0,
\]
entonces $ E $ es constante igual a $ E(0) = 0$. Una suma de dos cuadrados que desaparece.
fuerzas idénticas $ g = 0$, por eso $ y = a\cos + b\sin $.
\end{solution}

\begin{solution}{pb:g12:trigo:1}
\textbf{1.} $0.99833\ldots $ y $0.99998\ldots $: arrastrándose hacia
$1$.

\textbf{2.} Triángulo $ OAM $: base $1$, altura $\sin x $: área
$\frac{\sin x}{2}$. Sector $ OAM $: fracción
$\frac{x}{2\pi}$ del disco unitario: área $\frac x2$. Triángulo
$ OAT $: base $1$, altura $\tan x $: área $\frac{\tan x}{2}$.

\textbf{3.} El triángulo se encuentra dentro del sector dentro del gran
triángulo: $\frac{\sin x}{2} < \frac x2 < \frac{\tan x}{2}$,
es decir.\ $\sin x < x < \tan x $. De $\sin x < x $:
$\frac{\sin x}{x} < 1$; y de
$ x < \tan x = \frac{\sin x}{\cos x}$, multiplicando por
$\frac{\cos x}{x} > 0$: $\cos x < \frac{\sin x}{x}$. Como
$ x \to 0^+$, $\cos x \to 1$: apretado,
$\frac{\sin x}{x} \to 1$; y $\frac{\sin x}{x}$ es par, entonces
el límite bilateral es $1$.

\textbf{4.} $\frac{1 - \cos x}{x^2} =
\frac{1 - \cos^2 x}{x^2 (1 + \cos x)} =
\left(\frac{\sin x}{x}\right)^2 \frac{1}{1 + \cos x}
\to 1 \times \frac12$.

\textbf{5.} $\sin'(0) = \lim_{h \to 0} \frac{\sin h - 0}{h} =
1$: toda la diferenciación de \omterm{def:g11:trigo:cossin}{señor} y \omterm{def:g11:trigo:cossin}{coseño} fluye desde
éste límite. Medido en grados, el límite sería
$\frac{\pi}{180}$, y cada derivada arrastraría eso
constante: radianes son exactamente la unidad que hace el calculo
de oscilación limpia.

\textbf{6.} $ y' = -R\omega\sin(\omega t - \varphi)$,
$ y'' = -R\omega^2\cos(\omega t - \varphi) = -\omega^2 y $.

\textbf{7.} $\omega = 2$. $ y(0) = A = 3$;
$ y'(0) = 2B = 8$: $ B = 4$: $ y = 3\cos 2t + 4\sin 2t $.

\textbf{8.} $ R = \sqrt{9 + 16} = 5$; período
$\frac{2\pi}{2} = \pi $; fase
$\varphi = \arctan\frac43 \approx 0.927$:
$ y = 5\cos(2t - 0.927)$.

\textbf{9.} $5\cos(2t - \varphi) = 0$ primero cuando
$2t - \varphi = \frac\pi2$:
$ t = \frac{\pi/2 + 0.927}{2} \approx 1.25$ s.

\textbf{10.} Con $\sin\theta \approx \theta $:
$\theta'' = -\frac gL \theta $: armónico con
$\omega = \sqrt{\frac gL}$, período
$ T = \frac{2\pi}{\omega} = 2\pi\sqrt{\frac Lg}$. Para
$ L = 1$: $ T \approx 2.006$ s. El período no depende de
la amplitud (para pequeñas oscilaciones) --- isocronismo de Galileo,
el hecho que hace posibles los relojes de péndulo.

\textbf{11.} $ L = g\left(\frac{T}{2\pi}\right)^2 =
\frac{9.81}{\pi^2} \approx 0.994$ m: el péndulo de segundos es
acerca de $6$ mm por debajo del metro meridiano. La Academia eligió
el meridiano (la gravedad varía de un lugar a otro, lo que delata
el péndulo); tenía $ g $ Si hubiera sido un poco más grande, nuestro medidor
garrapata.

\textbf{12.} $\cos\left(x + \frac\pi2\right) =
\cos x \cos\frac\pi2 - \sin x \sin\frac\pi2 = -\sin x $;
$\sin\left(x + \frac\pi2\right) = \cos x $. tan diferenciador
\omterm{def:g11:trigo:cossin}{señor} da \omterm{def:g11:trigo:cossin}{señor} desplazado un cuarto de vuelta ($\cos $), y otra vez
($-\sin $), y otra vez, y a casa en cuatro pasos: diferenciación
gira el círculo.

\textbf{13.} Sumando las dos expansiones de $\cos(a \mp b)$:
$\cos(a-b) + \cos(a+b) = 2\cos a\cos b $. Con $ a = b = x $:
$1 + \cos 2x = 2\cos^2 x $, es decir.\
$\cos^2 x = \frac{1 + \cos 2x}{2}$.

\textbf{14.} Con $ p = \frac{p+q}{2} + \frac{p-q}{2}$ y
$ q $ su espejo, en expansión y añadiendo:
$\sin p + \sin q = 2\sin\frac{p+q}{2}\cos\frac{p-q}{2}$. Para
$440$ y $444$ Hz: un tono en $442$ Hz cuyo volumen es
modulado por $\cos(2\pi \cdot 2t)$ --- el volumen alcanza su punto \omterm{def:g10:functions:extrema}{máximo} $4$
veces por segundo (la velocidad del sobre \omterm{def:g10:numbers:abs}{valor absoluto}): cuatro tiempos
un segundo, que desaparece cuando las bifurcaciones coinciden.

\textbf{15.} $\cos 3x = \cos 2x\cos x - \sin 2x \sin x =
(2\cos^2 x - 1)\cos x - 2\sin^2 x\cos x = 4\cos^3 x -
3\cos x $. En $ x = 20^\circ $: $\cos 60^\circ = \frac12$, entonces
$ c = \cos 20^\circ $ satisface $8c^3 - 6c - 1 = 0$ --- un cúbico
sin \omterm{def:g11:quad:discriminant}{raíz} cuadrada solución: construir $20^\circ $, es decir\
trisectando $60^\circ $, está más allá de la regla y el compás.

\textbf{16.} $ R = \sqrt{9 + 27} = 6$;
$\tan\varphi = \frac{3\sqrt3}{3} = \sqrt3$:
$\varphi = \frac\pi3$: $ I(t) = 6\cos\left(100\pi t -
\frac\pi3\right)$: cima $6$ amperios, desfase $60^\circ $.

\textbf{17.} $\sin^2 t = \frac{1 - \cos 2t}{2}$: período
$\pi $. $\sin 2t $ tiene periodo $\pi $, $\sin 3t $ tiene
$\frac{2\pi}{3}$: la suma se repite después del \omterm{def:g10:functions:extrema}{mínimo} común
múltiples, $2\pi $.

\textbf{18.} Una oscilación de período $1$ dentro de la contracción
sobre $\pm\eu^{-t/10}$: cada movimiento sobre $10\,\%$ más bajo. En
$ t = 10$ el sobre es $\eu^{-1} \approx 0.368$ --- el
constante de \cref{pb:g12:exp:1} una vez más. ecuaciones con tal
soluciones amortiguadas ($ y'' + a y' + b y = 0$) se resuelven en el
diferencial ecuaciones capítulo.

\textbf{19.} Una fuerza que retrocede proporcionalmente a la
el desplazamiento da $ y'' = -\omega^2 y $; y
\cref{exo:g12:trigo:9} \omterm{def:g10:proba:sample}{muestra} las soluciones de eso \omterm{def:g10:algebra:equation}{ecuación}
son \emph{exactamente} las combinaciones de $\cos $ y $\sin $ ---
existencia por exhibición, unicidad por el ejercicio: vibración
no tiene más remedio que ser sinusoidal.

\textbf{20.} Un límite ($\frac{\sin x}{x} \to 1$, por
áreas intercaladas), dos derivadas ($\sin' = \cos $,
$\cos' = -\sin $), uno \omterm{def:g10:algebra:equation}{ecuación} ($ y'' = -\omega^2 y $), un mundo
de relojes, cuerdas, corrientes y mareas. Asterisco: para ancho
columpios, $ T = 4\sqrt{\frac Lg} \cdot
\frac{\pi/2}{M(1, \cos(\theta_0/2))}$ --- el
aritmética--geométrica \omterm{def:g11:stat:mean}{media} de \cref{pb:g12:seq:1} calcula el
Integral elíptica en un puñado de iteraciones: el diario de Gauss.
entrada y el péndulo de Galileo, una fórmula aparte.
\end{solution}
