\section*{Capítulo \ref{ch:g12:trigo} --- Funciones trigonométricas}

\begin{solution}{exo:g12:trigo:1}
$\cos\frac{2\pi}{3} = \cos\left(\pi - \frac\pi3\right) = -\cos\frac\pi3
= -\frac12$.

$\sin\frac{5\pi}{6} = \sin\left(\pi - \frac\pi6\right) = \sin\frac\pi6
= \frac12$.

$\cos\frac{7\pi}{4} = \cos\left(2\pi - \frac{\pi}{4}\right)
= \cos\left(-\frac\pi4\right) = \frac{\sqrt2}{2}$.

$\sin\left(-\frac\pi3\right) = -\sin\frac\pi3 = -\frac{\sqrt3}{2}$.
\end{solution}

\begin{solution}{exo:g12:trigo:2}
$f'(x) = 2\sin x\cos x = \sin 2x$ (regla de la cadena).

$g'(x) = -3\sin(3x) + \sin x + x\cos x$ (reglas de la cadena y del
producto).

Regla del cociente:
\[
h'(x) = \frac{\cos x\,(2 + \cos x) - \sin x\,(-\sin x)}{(2+\cos x)^2}
= \frac{2\cos x + \cos^2 x + \sin^2 x}{(2+\cos x)^2}
= \frac{2\cos x + 1}{(2+\cos x)^2}.
\]
\end{solution}

\begin{solution}{exo:g12:trigo:3}
\emph{(a)} Una solución particular de $\sin x = \frac{\sqrt3}{2}$ es
$\frac\pi3$. Soluciones generales: $x = \frac\pi3 + 2k\pi$ o
$x = \pi - \frac\pi3 + 2k\pi = \frac{2\pi}{3} + 2k\pi$. En
$\intoc{-\pi}{\pi}$: $x \in \left\{\frac\pi3, \frac{2\pi}{3}\right\}$.

\emph{(b)} $\cos 2x = \frac{\sqrt2}{2} = \cos\frac\pi4$ da
$2x = \pm\frac\pi4 + 2k\pi$, luego $x = \pm\frac\pi8 + k\pi$. En
$\intoc{-\pi}{\pi}$:
$x \in \left\{-\frac{7\pi}{8}, -\frac\pi8, \frac\pi8, \frac{7\pi}{8}\right\}$.
\end{solution}

\begin{solution}{exo:g12:trigo:4}
Con $a = \frac\pi3$ y $b = \frac\pi4$:
\[
\cos\frac{\pi}{12} = \cos a\cos b + \sin a \sin b
= \frac12\cdot\frac{\sqrt2}{2} + \frac{\sqrt3}{2}\cdot\frac{\sqrt2}{2}
= \frac{\sqrt2 + \sqrt6}{4},
\]
\[
\sin\frac{\pi}{12} = \sin a\cos b - \cos a \sin b
= \frac{\sqrt3}{2}\cdot\frac{\sqrt2}{2} - \frac12\cdot\frac{\sqrt2}{2}
= \frac{\sqrt6 - \sqrt2}{4}.
\]
\end{solution}

\begin{solution}{exo:g12:trigo:5}
Tomamos $s = \sin x$: $2s^2 - s - 1 = (2s + 1)(s - 1) \geq 0$ si y solo si
$s \leq -\frac12$ o $s = 1$.

$\sin x = 1$ da $x = \frac\pi2$. $\sin x \leq -\frac12$: sobre la
circunferencia unidad, el arco por debajo de la recta horizontal
$y = -\frac12$, que en $\intcc{0}{2\pi}$ es
$x \in \intcc{\frac{7\pi}{6}}{\frac{11\pi}{6}}$. Por tanto,
\[
S = \left\{\frac{\pi}{2}\right\} \cup
\intcc{\frac{7\pi}{6}}{\frac{11\pi}{6}} .
\]
\end{solution}

\begin{solution}{exo:g12:trigo:6}
$\dfrac{\sin 3x}{x} = 3\,\dfrac{\sin 3x}{3x} \to 3 \cdot 1 = 3$ (por
composición con $u = 3x \to 0$).

Usando $1 - \cos x = \dfrac{\sin^2 x}{1 + \cos x}$:
\[
\frac{1 - \cos x}{x^2}
= \left(\frac{\sin x}{x}\right)^{\!2} \frac{1}{1 + \cos x}
\xrightarrow[x\to0]{} 1 \cdot \frac12 = \frac12 .
\]

Sustituyendo $u = \frac1x \to 0^+$:
$x \sin\frac1x = \frac{\sin u}{u} \to 1$.
\end{solution}

\begin{solution}{exo:g12:trigo:7}
$f'(x) = 1 - \sin x \geq 0$ para todo $x$, con igualdad exactamente cuando
$\sin x = 1$, es decir, $x = \frac\pi2 + 2k\pi$. En $\intcc{0}{2\pi}$, $f$
crece de $f(0) = 1$ a $f(2\pi) = 2\pi + 1$ (el cero de $f'$ en
$\frac{\pi}{2}$ está aislado, así que el crecimiento es incluso estricto);
\omterm{def:g10:functions:extrema}{mínimo} $1$ en $0$ y \omterm{def:g10:functions:extrema}{máximo} $2\pi + 1$ en $2\pi$. En $\R$, $f' \geq 0$ en
todas partes con ceros solo aislados, así que $f$ es (estrictamente)
\omterm{def:g12:seq:monotonic}{creciente} por el \cref{thm:g12:deriv:variations}, aunque $f'$ se anule en
todos los $\frac\pi2 + 2k\pi$.
\end{solution}

\begin{solution}{exo:g12:trigo:8}
\emph{1.} Sea $R = \sqrt{A^2 + B^2} > 0$. El punto
$\left(\frac AR, \frac BR\right)$ está en la circunferencia unidad, así que
existe $\varphi$ con $\cos\varphi = \frac AR$ y $\sin\varphi = \frac BR$.
Entonces
\[
R\cos(x - \varphi) = R\bigl(\cos x\cos\varphi + \sin x \sin\varphi\bigr)
= A\cos x + B\sin x = f(x).
\]

\emph{2.} Como el \omterm{def:g10:stats:quartiles}{recorrido} de $\cos$ es $\intcc{-1}{1}$, $f$ tiene \omterm{def:g10:functions:extrema}{máximo}
$R$ y \omterm{def:g10:functions:extrema}{mínimo} $-R$. Para $\cos x + \sin x$: $A = B = 1$, $R = \sqrt2$ y
$\varphi = \frac\pi4$, así que la \omterm{def:g10:algebra:equation}{ecuación} queda
$\sqrt2\cos\left(x - \frac\pi4\right) = 1$, es decir,
$\cos\left(x - \frac\pi4\right) = \frac{\sqrt2}{2}$, lo que da
$x - \frac\pi4 = \pm\frac\pi4 + 2k\pi$: en $\intoc{-\pi}{\pi}$,
$x \in \left\{0, \frac\pi2\right\}$.
\end{solution}

\begin{solution}{exo:g12:trigo:9}
$(\sin)'' = (\cos)' = -\sin$ y $(\cos)'' = (-\sin)' = -\cos$: las dos
resuelven $y'' = -y$.

Sea $g = y - a\cos - b\sin$. Entonces $g'' = -g$ (por la linealidad de la
derivación), $g(0) = y(0) - a = 0$ y $g'(0) = y'(0) - b = 0$. Tomemos
$E = g^2 + (g')^2$. Entonces
\[
E' = 2g g' + 2g' g'' = 2g g' - 2g' g = 0,
\]
así que $E$ es constante e igual a $E(0) = 0$. Una suma de dos cuadrados
que se anula idénticamente obliga a $g = 0$, luego
$y = a\cos + b\sin$.
\end{solution}

\begin{solution}{pb:g12:trigo:1}
\textbf{1.} $0.99833\ldots$ y $0.99998\ldots$: acercándose a $1$.

\textbf{2.} Triángulo $OAM$: base $1$ y altura $\sin x$: área
$\frac{\sin x}{2}$. Sector $OAM$: la fracción $\frac{x}{2\pi}$ del círculo
unidad: área $\frac x2$. Triángulo $OAT$: base $1$ y altura $\tan x$: área
$\frac{\tan x}{2}$.

\textbf{3.} El triángulo está dentro del sector y este, dentro del
triángulo grande: $\frac{\sin x}{2} < \frac x2 < \frac{\tan x}{2}$, es
decir, $\sin x < x < \tan x$. De $\sin x < x$ se sigue
$\frac{\sin x}{x} < 1$; y de $x < \tan x = \frac{\sin x}{\cos x}$,
multiplicando por $\frac{\cos x}{x} > 0$, se sigue
$\cos x < \frac{\sin x}{x}$. Cuando $x \to 0^+$, $\cos x \to 1$: por el
sándwich, $\frac{\sin x}{x} \to 1$; y $\frac{\sin x}{x}$ es par, así que el
límite bilateral también es $1$.

\textbf{4.} $\frac{1 - \cos x}{x^2} = \frac{1 - \cos^2 x}{x^2 (1 + \cos x)}
= \left(\frac{\sin x}{x}\right)^2 \frac{1}{1 + \cos x} \to
1 \times \frac12$.

\textbf{5.} $\sin'(0) = \lim_{h \to 0} \frac{\sin h - 0}{h} = 1$: toda la
derivación del \omterm{def:g12:trigo:cossin}{seno} y del \omterm{def:g12:trigo:cossin}{coseno} mana de ese único límite. Medido en
grados, el límite sería $\frac{\pi}{180}$ y todas las \omterm{def:g12:deriv:derivative}{derivadas}
arrastrarían esa constante: los \omterm{def:g11:trigo:radian}{radianes} son exactamente la unidad que deja
limpio el cálculo de las oscilaciones.

\textbf{6.} $y' = -R\omega\sin(\omega t - \varphi)$ e
$y'' = -R\omega^2\cos(\omega t - \varphi) = -\omega^2 y$.

\textbf{7.} $\omega = 2$. $y(0) = A = 3$; $y'(0) = 2B = 8$: $B = 4$:
$y = 3\cos 2t + 4\sin 2t$.

\textbf{8.} $R = \sqrt{9 + 16} = 5$; periodo $\frac{2\pi}{2} = \pi$; fase
$\varphi = \arctan\frac43 \approx 0.927$: $y = 5\cos(2t - 0.927)$.

\textbf{9.} $5\cos(2t - \varphi) = 0$ por primera vez cuando
$2t - \varphi = \frac\pi2$: $t = \frac{\pi/2 + 0.927}{2} \approx 1.25$ s.

\textbf{10.} Con $\sin\theta \approx \theta$:
$\theta'' = -\frac gL \theta$: armónico con $\omega = \sqrt{\frac gL}$ y
periodo $T = \frac{2\pi}{\omega} = 2\pi\sqrt{\frac Lg}$. Para $L = 1$:
$T \approx 2.006$ s. El periodo no depende de la amplitud (con
oscilaciones pequeñas): el isocronismo de Galileo, el hecho que hace
posibles los relojes de péndulo.

\textbf{11.} $L = g\left(\frac{T}{2\pi}\right)^2 = \frac{9.81}{\pi^2}
\approx 0.994$ m: el péndulo de segundos se queda unos $6$ mm por debajo
del metro del meridiano. La Academia eligió el meridiano (la gravedad varía
de un lugar a otro y delata al péndulo); si $g$ hubiera sido un pelo mayor,
nuestro metro haría tictac.

\textbf{12.} $\cos\left(x + \frac\pi2\right) =
\cos x \cos\frac\pi2 - \sin x \sin\frac\pi2 = -\sin x$;
$\sin\left(x + \frac\pi2\right) = \cos x$. Así que derivar el \omterm{def:g12:trigo:cossin}{seno} lo
desplaza un cuarto de vuelta ($\cos$), y otra vez ($-\sin$), y otra, y
vuelve a casa en cuatro pasos: derivar hace girar la circunferencia.

\textbf{13.} Sumando los dos desarrollos de $\cos(a \mp b)$:
$\cos(a-b) + \cos(a+b) = 2\cos a\cos b$. Con $a = b = x$:
$1 + \cos 2x = 2\cos^2 x$, es decir,
$\cos^2 x = \frac{1 + \cos 2x}{2}$.

\textbf{14.} Con $p = \frac{p+q}{2} + \frac{p-q}{2}$ y $q$ su reflejo,
desarrollando y sumando:
$\sin p + \sin q = 2\sin\frac{p+q}{2}\cos\frac{p-q}{2}$. Para $440$ y
$444$ Hz: un tono de $442$ Hz cuya intensidad queda modulada por
$\cos(2\pi \cdot 2t)$; la intensidad alcanza su \omterm{def:g10:functions:extrema}{máximo} $4$ veces por
segundo (por el \omterm{def:g10:numbers:abs}{valor absoluto} de la envolvente): cuatro pulsaciones por
segundo, que se apagan cuando los diapasones coinciden.

\textbf{15.} $\cos 3x = \cos 2x\cos x - \sin 2x \sin x =
(2\cos^2 x - 1)\cos x - 2\sin^2 x\cos x = 4\cos^3 x - 3\cos x$. En
$x = 20^\circ$: $\cos 60^\circ = \frac12$, luego $c = \cos 20^\circ$ cumple
$8c^3 - 6c - 1 = 0$: una \omterm{def:g10:algebra:equation}{ecuación} de tercer grado sin solución por raíces
cuadradas; construir $20^\circ$, es decir, trisecar $60^\circ$, queda fuera
del alcance de la regla y el compás.

\textbf{16.} $R = \sqrt{9 + 27} = 6$;
$\tan\varphi = \frac{3\sqrt3}{3} = \sqrt3$: $\varphi = \frac\pi3$:
$I(t) = 6\cos\left(100\pi t - \frac\pi3\right)$: pico de $6$ amperios y
desfase de $60^\circ$.

\textbf{17.} $\sin^2 t = \frac{1 - \cos 2t}{2}$: periodo $\pi$.
$\sin 2t$ tiene periodo $\pi$ y $\sin 3t$, $\frac{2\pi}{3}$: la suma se
repite tras el \omterm{def:g10:functions:extrema}{mínimo} común múltiplo, $2\pi$.

\textbf{18.} Una oscilación de periodo $1$ dentro de la envolvente
menguante $\pm\eu^{-t/10}$: cada vaivén, un $10\,\%$ más bajo. En $t = 10$
la envolvente vale $\eu^{-1} \approx 0.368$: otra vez la constante del
\cref{pb:g12:exp:1}. Las ecuaciones con soluciones amortiguadas así
($y'' + a y' + b y = 0$) se resuelven en el capítulo de ecuaciones
diferenciales.

\textbf{19.} Una fuerza que tira hacia atrás proporcionalmente al
desplazamiento da $y'' = -\omega^2 y$; y el \cref{exo:g12:trigo:9} muestra
que las soluciones de esa \omterm{def:g10:algebra:equation}{ecuación} son \emph{exactamente} las combinaciones
de $\cos$ y $\sin$: existencia por exhibición y unicidad por el ejercicio;
a la vibración no le queda más remedio que ser sinusoidal.

\textbf{20.} Un límite ($\frac{\sin x}{x} \to 1$, por áreas encajadas), dos
\omterm{def:g12:deriv:derivative}{derivadas} ($\sin' = \cos$, $\cos' = -\sin$), una \omterm{def:g10:algebra:equation}{ecuación}
($y'' = -\omega^2 y$) y un mundo de relojes, cuerdas, corrientes y mareas.
Asterisco: para oscilaciones amplias,
$T = 4\sqrt{\frac Lg} \cdot \frac{\pi/2}{M(1, \cos(\theta_0/2))}$: la \omterm{def:g11:stat:mean}{media}
aritmético-geométrica del \cref{pb:g12:seq:1} calcula la integral elíptica
en un puñado de iteraciones; la entrada del diario de Gauss y el péndulo de
Galileo, a una fórmula de distancia.
\end{solution}
