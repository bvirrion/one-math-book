\section*{Capítulo \ref{ch:g12:sums} --- Sumas de variables aleatorias y la ley de los grandes números}

\begin{solution}{exo:g12:sums:1}
Cada dado tiene \omterm{def:g12:randvar:exp}{esperanza}
$\frac{1 + 2 + \dots + 6}{6} = \frac{7}{2}$, entonces por linealidad
$\E(S) = \frac72 + \frac72 = 7$. los dados son \omterm{def:g12:sums:indep}{independientes}, entonces
$\V(S) = \frac{35}{12} + \frac{35}{12} = \frac{35}{6} \approx 5.83$.
\end{solution}

\begin{solution}{exo:g12:sums:2}
$\E(X) = 50$ y $\V(X) = 100 \times 0.25 = 25$.

Markov ($ X \geq 0$): $\P(X \geq 75) \leq \frac{50}{75} = \frac23$.

Bienaymé--Chebyshev:
$\P(\abs{X - 50} \geq 25) \leq \frac{25}{25^2} = 0.04$ --- y esto incluso
limita el \emph{bilateral} suceso, de los cuales $ X \geq 75$ es la mitad. Chebyshev
es mucho más nítido aquí porque explota la varianza, no sólo la \omterm{def:g11:stat:mean}{media}.
(El verdadero valor de $\P(X \geq 75)$ es más pequeño que $10^{-6}$: ambos límites
son toscos.)
\end{solution}

\begin{solution}{exo:g12:sums:3}
$ F_n $ es el \omterm{def:g10:proba:sample}{muestra} de medios de $ n $ bernoulli$\left(\frac12\right)$ variables,
de varianza $\sigma^2 = \frac14$. el atado
\[
\P\left(\abs{F_n - 0.5} \geq 0.05\right)
\leq \frac{1/4}{n \times 0.05^2} = \frac{100}{n}
\]
es $\leq 0.05$ tan pronto como $ n \geq 2000$.
\end{solution}

\begin{solution}{exo:g12:sums:4}
Por \cref{prop:g12:sums:mean}, $\E(M_n) = 5\%$ (la diversificación no
cambiar el rendimiento esperado) y
$\sigma(M_n) = \frac{20\%}{\sqrt n}$. Requerir
$\frac{20}{\sqrt n} < 4$ da $\sqrt n > 5$, \emph{es decir.} $ n \geq 26$.
Este es el principio de \emph{diversificación}: \omterm{def:g12:randvar:rv}{distribución} del capital
\omterm{def:g12:sums:indep}{independientes} riesgos dividir el riesgo (desviación estándar) por $\sqrt n $
sin reducir el rendimiento esperado.
\end{solution}

\begin{solution}{exo:g12:sums:5}
\emph{1.} contando el $9$ pares igualmente probables: $ S $ toma los valores
$2, 3, 4, 5, 6$ con \omterm{def:g10:proba:distribution}{probabilidades}
$\frac19, \frac29, \frac39, \frac29, \frac19$. Por eso
$\E(S) = \frac{2 + 6 + 12 + 10 + 6}{9} = 4$ y
$\E(S^2) = \frac{4 + 18 + 48 + 50 + 36}{9} = \frac{156}{9} = \frac{52}{3}$,
entonces $\V(S) = \frac{52}{3} - 16 = \frac43$.

\emph{2.} Una variable: $\E(X) = 2$,
$\E(X^2) = \frac{1 + 4 + 9}{3} = \frac{14}{3}$,
$\V(X) = \frac{14}{3} - 4 = \frac23$. Entonces $\E(S) = 2 + 2 = 4$ y, por
independencia, $\V(S) = \frac23 + \frac23 = \frac43$. Mismos valores.
\end{solution}

\begin{solution}{exo:g12:sums:6}
Con $ Y = X $: $\V(X + Y) = \V(2X) = 4\V(X)$, mientras
$\V(X) + \V(Y) = 2\V(X)$. Los dos se ponen de \omterm{def:g12:comb:tuples}{acuerdo} sólo cuando $\V(X) = 0$, \emph{es decir.}
cuando $ X $ es constante --- por lo que la aditividad realmente requiere independencia
(aquí $ X $ depende al \omterm{def:g10:functions:extrema}{máximo} de sí mismo).
\end{solution}

\begin{solution}{exo:g12:sums:7}
Bajo justicia, $ F_n $ es la \omterm{def:g11:stat:mean}{media} de $ n = 1200$ variables de Bernoulli con
$ p = \frac16$, $\sigma^2 = \frac16\cdot\frac56 = \frac{5}{36}$:
\[
\P\left(\abs{F_n - \tfrac16} \geq 0.05\right)
\leq \frac{5/36}{1200 \times 0.0025} = \frac{5}{108} \approx 0.046 .
\]
La desviación observada es exactamente $0.05$: un suceso que produce un dado justo
con \omterm{def:g10:proba:distribution}{probabilidad} a lo sumo $4.6\%$ --- y como Chebyshev es muy
conservador, el verdadero \omterm{def:g10:proba:distribution}{probabilidad} es mucho más pequeño. La hipótesis de la justicia
no es plausible; Es muy probable que el dado esté cargado.
\end{solution}

\begin{solution}{exo:g12:sums:8}
\emph{1.} $ p(1-p) = \frac14 - \left(p - \frac12\right)^2 \leq \frac14$.

\emph{2.} $ F_n $ tiene \omterm{def:g12:randvar:exp}{esperanza} $ p $ y varianza
$\frac{p(1-p)}{n} \leq \frac{1}{4n}$; Bienaymé--Chebyshev da
\[
\P\bigl(\abs{F_n - p} \geq \varepsilon\bigr)
\leq \frac{p(1-p)}{n\varepsilon^2} \leq \frac{1}{4n\varepsilon^2},
\]
válido sea lo que sea lo desconocido $ p $.

\emph{3.} Con $\varepsilon = 0.03$ y nivel $0.05$:
\[
\frac{1}{4n(0.03)^2} \leq 0.05
\iff n \geq \frac{1}{4 \times 0.0009 \times 0.05} \approx 5556 .
\]
Acerca de $5600$ gente se basta con este tosco límite (el clásico
\omterm{def:g10:numbers:approx}{aproximación} normal la respuesta está más cerca $1100$, ver
\cref{ch:g12:contdist}).
\end{solution}
