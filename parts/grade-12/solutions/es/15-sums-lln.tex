\section*{Capítulo \ref{ch:g12:sums} --- Sumas de variables aleatorias y la ley de los grandes números}

\begin{solution}{exo:g12:sums:1}
Cada dado tiene \omterm{def:g12:randvar:exp}{esperanza} $\frac{1 + 2 + \dots + 6}{6} = \frac{7}{2}$, así
que por linealidad $\E(S) = \frac72 + \frac72 = 7$. Los dados son
\omterm{def:g12:sums:indep}{independientes}, luego
$\V(S) = \frac{35}{12} + \frac{35}{12} = \frac{35}{6} \approx 5.83$.
\end{solution}

\begin{solution}{exo:g12:sums:2}
$\E(X) = 50$ y $\V(X) = 100 \times 0.25 = 25$.

Markov ($X \geq 0$): $\P(X \geq 75) \leq \frac{50}{75} = \frac23$.

Bienaymé--Chebyshev:
$\P(\abs{X - 50} \geq 25) \leq \frac{25}{25^2} = 0.04$, y esto acota
incluso el \omterm{def:g11:prob:model}{suceso} \emph{bilateral}, del que $X \geq 75$ es la mitad. Aquí
Chebyshev es mucho más fina porque aprovecha la \omterm{def:g12:randvar:exp}{varianza}, y no solo la
\omterm{def:g11:stat:mean}{media}. (El valor verdadero de $\P(X \geq 75)$ es menor que $10^{-6}$: las
dos cotas son burdas.)
\end{solution}

\begin{solution}{exo:g12:sums:3}
$F_n$ es la \omterm{prop:g12:sums:mean}{media muestral} de $n$ variables de Bernoulli de parámetro
$\frac12$, de \omterm{def:g12:randvar:exp}{varianza} $\sigma^2 = \frac14$. La cota
\[
\P\left(\abs{F_n - 0.5} \geq 0.05\right)
\leq \frac{1/4}{n \times 0.05^2} = \frac{100}{n}
\]
es $\leq 0.05$ en cuanto $n \geq 2000$.
\end{solution}

\begin{solution}{exo:g12:sums:4}
Por la \cref{prop:g12:sums:mean}, $\E(M_n) = 5\,\%$ (diversificar no cambia
la rentabilidad esperada) y $\sigma(M_n) = \frac{20\,\%}{\sqrt n}$. Exigir
$\frac{20}{\sqrt n} < 4$ da $\sqrt n > 5$, es decir, $n \geq 26$. Este es
el principio de la \emph{diversificación}: repartir el capital entre
riesgos \omterm{def:g12:sums:indep}{independientes} divide el riesgo (la \omterm{def:g11:stat:variance}{desviación típica}) entre
$\sqrt n$ sin reducir la rentabilidad esperada.
\end{solution}

\begin{solution}{exo:g12:sums:5}
\emph{1.} Contando las $9$ parejas equiprobables: $S$ toma los valores
$2, 3, 4, 5, 6$ con \omterm{def:g10:proba:distribution}{probabilidades}
$\frac19, \frac29, \frac39, \frac29, \frac19$. Por tanto,
$\E(S) = \frac{2 + 6 + 12 + 10 + 6}{9} = 4$ y
$\E(S^2) = \frac{4 + 18 + 48 + 50 + 36}{9} = \frac{156}{9} = \frac{52}{3}$,
luego $\V(S) = \frac{52}{3} - 16 = \frac43$.

\emph{2.} Para una sola variable: $\E(X) = 2$,
$\E(X^2) = \frac{1 + 4 + 9}{3} = \frac{14}{3}$ y
$\V(X) = \frac{14}{3} - 4 = \frac23$. Entonces $\E(S) = 2 + 2 = 4$ y, por
\omterm{def:g12:condprob:indep}{independencia}, $\V(S) = \frac23 + \frac23 = \frac43$. Los mismos valores.
\end{solution}

\begin{solution}{exo:g12:sums:6}
Con $Y = X$: $\V(X + Y) = \V(2X) = 4\V(X)$, mientras que
$\V(X) + \V(Y) = 2\V(X)$. Los dos coinciden solo cuando $\V(X) = 0$, es
decir, cuando $X$ es constante; así que la aditividad exige de verdad la
\omterm{def:g12:condprob:indep}{independencia} (aquí $X$ depende de sí misma todo lo que se puede depender).
\end{solution}

\begin{solution}{exo:g12:sums:7}
Suponiendo el dado equilibrado, $F_n$ es la \omterm{def:g11:stat:mean}{media} de $n = 1200$ variables
de Bernoulli con $p = \frac16$ y
$\sigma^2 = \frac16\cdot\frac56 = \frac{5}{36}$:
\[
\P\left(\abs{F_n - \tfrac16} \geq 0.05\right)
\leq \frac{5/36}{1200 \times 0.0025} = \frac{5}{108} \approx 0.046 .
\]
La desviación observada es exactamente $0.05$: un \omterm{def:g11:prob:model}{suceso} que un dado
equilibrado produce con \omterm{def:g10:proba:distribution}{probabilidad} como mucho del $4.6\,\%$; y como
Chebyshev es muy conservadora, la \omterm{def:g10:proba:distribution}{probabilidad} verdadera es mucho menor. La
hipótesis de equilibrio no es plausible; lo más probable, con mucho, es que
el dado esté trucado.
\end{solution}

\begin{solution}{exo:g12:sums:8}
\emph{1.} $p(1-p) = \frac14 - \left(p - \frac12\right)^2 \leq \frac14$.

\emph{2.} $F_n$ tiene \omterm{def:g12:randvar:exp}{esperanza} $p$ y \omterm{def:g12:randvar:exp}{varianza}
$\frac{p(1-p)}{n} \leq \frac{1}{4n}$; Bienaymé--Chebyshev da
\[
\P\bigl(\abs{F_n - p} \geq \varepsilon\bigr)
\leq \frac{p(1-p)}{n\varepsilon^2} \leq \frac{1}{4n\varepsilon^2},
\]
válido sea cual sea el $p$ desconocido.

\emph{3.} Con $\varepsilon = 0.03$ y nivel $0.05$:
\[
\frac{1}{4n(0.03)^2} \leq 0.05
\iff n \geq \frac{1}{4 \times 0.0009 \times 0.05} \approx 5556 .
\]
Con esta cota burda bastan unas $5600$ personas (la respuesta clásica por
\omterm{def:g10:numbers:approx}{aproximación} normal está más cerca de $1100$; véase el
\cref{ch:g12:contdist}).
\end{solution}

\begin{solution}{pb:g12:sums:1}
\textbf{1.} $\V(X + Y) = \V(X) + \V(Y) = \frac{35}{12} +
\frac{35}{12} = \frac{35}{6}$ (\omterm{def:g12:condprob:indep}{independencia}).

\textbf{2.} $\E(\bar X) = 3.5$;
$\sigma(\bar X) = \frac{1.71}{\sqrt{100}} \approx 0.17$: la \omterm{def:g11:stat:mean}{media} de cien
dados se pega al $3.5$ con menos de una quinta parte de punto de holgura.

\textbf{3.} $\P(X \geq 10) \leq \frac{2}{10} = 0.2$.

\textbf{4.} $\P \leq \frac{\V(\bar X)}{0.5^2} = \frac{35/1200}{0.25}
\approx 0.117$: como mucho, en torno al $12\,\%$.

\textbf{5.} Con $Y = -X$: $\V(X + Y) = \V(0) = 0$, mientras que
$\V(X) + \V(Y) = 2\V(X) > 0$: las variables perfectamente
anticorrelacionadas se cancelan; sumar \omterm{def:g12:randvar:exp}{varianzas} es un privilegio de la
\omterm{def:g12:condprob:indep}{independencia}.

\textbf{6.} $\E = \frac{18}{37} - \frac{19}{37} = -\frac{1}{37}
\approx -0.027$;
$\sigma = \sqrt{1 - \left(\frac{1}{37}\right)^2} \approx 1.00$.

\textbf{7.} $\E(G) = -2.70$ y $\sigma(G) = 10 \times 1.00 = 10$: la deriva
es de solo $0.27\sigma$. El ruido de la noche aplasta a la ventaja: una
amplia minoría de jugadores (en torno al $40\,\%$, según dice la campana)
sale ganando, que es exactamente lo que hace que vuelvan.

\textbf{8.} Recaudación del casino: \omterm{def:g12:randvar:exp}{esperanza} $+27\,027$ euros y \omterm{def:g11:stat:variance}{desviación
típica} $\approx 1\,000$: el beneficio está $27$ desviaciones típicas por
encima de cero. A $27\sigma$, ``el casino podría perder este año'' no es un
riesgo, es un error de redondeo.

\textbf{9.} $\P(\text{pérdida}) = \P(G \geq 27\,027$ para los
jugadores$)$ $\leq \frac{10^6 \times 1}{27\,027^2} \approx 0.0014$: hasta
la más tosca de las desigualdades del libro garantiza a la banca un
$99.86\,\%$; la verdad es astronómicamente más fuerte.

\textbf{10.} Cada euro apostado, se apueste cuando se apueste y como se
apueste, tiene una rentabilidad esperada de $-\frac{1}{37}$ de sí mismo;
por linealidad, la ganancia total esperada es $-\frac{1}{37} \times
(\text{total apostado})$, negativa para toda estrategia que apueste algo.
Un sistema ganador violaría la linealidad de la \omterm{def:g12:randvar:exp}{esperanza}: ninguna
secuencia ingeniosa de apuestas malas produce una buena. (Los sistemas de
doblar se limitan a cambiar muchas ganancias pequeñas por pérdidas raras y
catastróficas.)

\textbf{11.} El teorema dice que la \emph{frecuencia} del rojo a lo largo
de $n$ tiradas \omterm{def:g12:seq:limit}{converge} (en \omterm{def:g10:proba:distribution}{probabilidad}) a $\frac{18}{37}$. No dice nada
de la tirada $n + 1$: la ruleta no tiene memoria, y tras diez rojos la
\omterm{def:g10:proba:distribution}{probabilidad} de rojo sigue siendo $\frac{18}{37}$. Creer lo contrario es la
\emph{falacia del jugador}, y la pregunta 12 muestra qué le pasa de verdad
a las rachas.

\textbf{12.} El excedente de $100$ rojos no se devuelve: las tiradas
futuras son copias equitativas y el excedente esperado sigue siendo $100$.
Pero $\frac{100}{10^6} = 0.0001$: una centésima de punto. El error de la
falacia en una frase: \emph{la ley de los grandes números diluye los
accidentes pasados en un océano de pruebas nuevas; nunca manda a la ruleta
a cobrar deudas.}

\textbf{13.} $n \geq \frac{1}{4 \times 0.05 \times 0.03^2} \approx 5\,556$
personas.

\textbf{14.} $\frac{1.96^2 \times 0.25}{0.03^2} \approx 1\,067$: cinco
veces menos. Chebyshev vale para \emph{cualquier} \omterm{def:g12:randvar:rv}{distribución} y paga su
universalidad con holgura; el $1.96$ de los encuestadores sale de la forma
real, de campana, de las fluctuaciones, que se concentran mucho más.

\textbf{15.} El margen es proporcional a $\frac{1}{\sqrt n}$: reducirlo a
la mitad cuadruplica la \omterm{def:g10:proba:sample}{muestra}, unas $4\,400$ personas para
$\pm 1.5$ puntos. La precisión se paga a precios cuadráticos.

\textbf{16.} El modelo consiste en $n$ extracciones \omterm{def:g12:sums:indep}{independientes} con
\omterm{def:g10:proba:distribution}{probabilidad} de éxito $p$: el tamaño $N$ de la población no aparece jamás
(muestrear una fracción minúscula de una población grande y bien mezclada
es justamente lo que codifica la \omterm{def:g12:condprob:indep}{independencia}). La analogía de los
encuestadores: para probar la sopa basta una cucharada bien removida, tanto
si la olla da de comer a diez como a diez mil. La trampa está en el
removido (un marco de muestreo sesgado, como la encuesta del volumen
anterior que engañó a todo un país), no en la olla.

\textbf{17.} $315\sqrt n < 30n \iff \sqrt n > 10.5 \iff n > 110$: a partir
de un centenar de pólizas, la deriva adelanta al ruido, y cada póliza
adicional ensancha la \omterm{def:g11:seq:arithmetic}{diferencia} por la ley del $\sqrt n$; agrupar riesgos
\emph{es} el modelo de negocio.

\textbf{18.} Partes iguales e \omterm{def:g12:sums:indep}{independientes}:
$\sigma_{\text{cartera}} = \frac{20\,\%}{\sqrt{25}} = 4\,\%$: la misma
rentabilidad esperada con una quinta parte de la fluctuación. La comida
gratis: reducir el riesgo sin coste alguno en \omterm{def:g12:randvar:exp}{esperanza}, por el simple
hecho de promediar azares \omterm{def:g12:sums:indep}{independientes}.

\textbf{19.} Perfectamente correlacionadas: la cartera es una sola acción
con $25$ disfraces: $\sigma = 20\,\%$, sin reducción ninguna. La
diversificación alquila la \emph{\omterm{def:g12:condprob:indep}{independencia}}; y cuando una crisis
correlaciona todo (2008: todas las apuestas inmobiliarias eran una sola
apuesta), la protección del $\sqrt n$ se evapora exactamente cuando hace
falta.

\textbf{20.} Las sumas derivan como $n$ y fluctúan como $\sqrt n$, así que
las \omterm{def:g11:stat:mean}{medias} se asientan como $\frac{1}{\sqrt n}$: el casino ingresa la
deriva a lo largo de un millón de tiradas; el encuestador compra
$\frac{1}{\sqrt{1100}}$ de ruido con un presupuesto de llamadas; la
aseguradora crece más deprisa que sus propios siniestros a partir de
$n > 110$; la inversora divide el riesgo entre $\sqrt{25}$. Los
infractores: el jugador que cree en deudas (solo hay dilución) y el
financiero que cree en la \omterm{def:g12:condprob:indep}{independencia} (a veces solo hay una apuesta). La
forma universal de las fluctuaciones, la campana, es la estrella del
capítulo \omterm{def:g12:limcont:continuity}{continuo}, y su teorema, el teorema central del límite, es la cima
de la \omterm{def:g10:proba:distribution}{probabilidad} de los volúmenes universitarios.
\end{solution}
