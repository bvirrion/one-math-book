\section*{Capítulo \ref{ch:g12:diffeq} --- Ecuaciones diferenciales}

\begin{solution}{exo:g12:diffeq:1}
\emph{(a)} $y = C\eu^{3x}$; $y(0) = 2$ da $y = 2\eu^{3x}$.

\emph{(b)} $y' = -\frac12 y$, luego $y = C\eu^{-x/2}$; $y(0) = -1$ da
$y = -\eu^{-x/2}$.

\emph{(c)} Equilibrio $y_p = 5$; $y = C\eu^{-x} + 5$; $y(0) = 0$ da
$C = -5$: $y = 5\left(1 - \eu^{-x}\right)$.
\end{solution}

\begin{solution}{exo:g12:diffeq:2}
$N' = kN$, luego $N(t) = N_0 \eu^{kt}$. Duplicarse en $3$ horas significa
$\eu^{3k} = 2$, es decir,
\[
k = \frac{\ln 2}{3} \approx 0.231\ \text{h}^{-1}.
\]
\end{solution}

\begin{solution}{exo:g12:diffeq:3}
$y_p'(x) = \eu^x + x\eu^x = y_p(x) + \eu^x$: $y_p$ es una solución
particular (es el caso resonante del \cref{met:g12:diffeq:particular}). Por
la \cref{prop:g12:diffeq:general}, las soluciones son
$y = C\eu^{x} + x\eu^{x} = (C + x)\,\eu^x$, con $C \in \R$.
\end{solution}

\begin{solution}{exo:g12:diffeq:4}
Probamos con $y_p = \beta\eu^x$: $\beta\eu^x = -2\beta\eu^x + \eu^x$ da
$3\beta = 1$, luego $y_p = \frac13\eu^x$. Solución general
$y = C\eu^{-2x} + \frac13\eu^{x}$; la condición $y(0) = 1$ da
$C = \frac23$:
\[
y(x) = \frac{2}{3}\,\eu^{-2x} + \frac{1}{3}\,\eu^{x}.
\]
\end{solution}

\begin{solution}{exo:g12:diffeq:5}
$N(t) = N_0\,\eu^{-\lambda t}$ con $\lambda = \frac{\ln 2}{5730}$
(\cref{ex:g12:exp:halflife}). Resolvemos $\eu^{-\lambda t} = 0.6$:
\[
t = \frac{\ln(1/0.6)}{\lambda} = 5730\,\frac{\ln(5/3)}{\ln 2}
\approx 5730 \times \frac{0.5108}{0.6931} \approx 4220 \text{ años}.
\]
\end{solution}

\begin{solution}{exo:g12:diffeq:6}
\emph{1.} Entra sal a \omterm{def:g11:seq:geometric}{razón} de $0.2 \times 5 = 1$ kg/min. La salida lleva
una concentración de $\frac{m}{100}$ kg/L a $5$ L/min, es decir,
$\frac{m}{20}$ kg/min. Por tanto, $m' = 1 - \frac{m}{20}$.

\emph{2.} Equilibrio $m_p = 20$; $m(t) = 20 + C\eu^{-t/20}$, y $m(0) = 0$
da $C = -20$:
\[
m(t) = 20\left(1 - \eu^{-t/20}\right) \xrightarrow[t\to+\infty]{} 20 \text{ kg}.
\]
A la larga, la concentración del depósito iguala a la de la salmuera
entrante: $0.2$ kg/L $\times$ $100$ L $= 20$ kg.
\end{solution}

\begin{solution}{exo:g12:diffeq:7}
\emph{1.} $v' = g - \frac{k}{m} v$ con $\frac km = \frac{16}{80} = 0.2$.
Equilibrio $v_\infty = \frac{mg}{k} = \frac{9.8}{0.2} = 49$ m/s;
$v(t) = 49 + C\eu^{-0.2t}$, y $v(0) = 0$ da
\[
v(t) = 49\left(1 - \eu^{-0.2 t}\right).
\]

\emph{2.} Velocidad límite $49$ m/s ($\approx 176$ km/h). Queremos
$1 - \eu^{-0.2t} = 0.95$, es decir, $\eu^{-0.2t} = 0.05$:
\[
t = \frac{\ln 20}{0.2} \approx \frac{3.00}{0.2} \approx 15 \text{ s}.
\]
\end{solution}

\begin{solution}{exo:g12:diffeq:8}
\emph{1.} $z = \frac1y$ da $z' = -\frac{y'}{y^2}
= -\frac{y(1-y)}{y^2} = -\frac{1-y}{y} = -\frac1y + 1 = -z + 1$.

\emph{2.} Equilibrio $z_p = 1$, luego $z(t) = 1 + C\eu^{-t}$. De
$y(0) = \frac{1}{10}$ se sigue $z(0) = 10$, luego $C = 9$ y
\[
y(t) = \frac{1}{1 + 9\,\eu^{-t}} .
\]

\emph{3.} Cuando $t \to +\infty$, $9\eu^{-t} \to 0$ e $y(t) \to 1$. La
curva es la clásica logística en forma de S (una \emph{sigmoide}):
crecimiento lento al principio ($y$ pequeño, $y' \approx y$), crecimiento
\omterm{def:g10:functions:extrema}{máximo} cuando $y = \frac12$ (donde $y' = y(1-y)$ es \omterm{def:g10:functions:extrema}{máximo}) y saturación
después hacia la capacidad de carga $1$.
\end{solution}

\begin{solution}{exo:g12:diffeq:9}
Por el \cref{thm:g12:diffeq:affine}, $y(x) = C\eu^{ax} - \frac ba$. Por
tanto,
\[
u_{n+1} = C\eu^{a(n+1)} - \frac ba
= \eu^{a}\left(C\eu^{an} - \frac ba\right) + \frac ba\left(\eu^a - 1\right)
= q\,u_n + r
\]
con $q = \eu^{a}$ y $r = \frac{b}{a}\left(\eu^{a} - 1\right)$. La condición
$\abs q < 1$ significa $\eu^a < 1$, es decir, $a < 0$: exactamente la
condición bajo la cual las soluciones de la \omterm{def:g12:diffeq:ode}{ecuación diferencial} convergen
al equilibrio $-\frac ba$; y, en efecto, el \omterm{pb:g10:functions:1}{punto fijo} de la recurrencia es
$\frac{r}{1 - q} = -\frac{b}{a}$.
\end{solution}

\begin{solution}{pb:g12:diffeq:1}
\textbf{1.} $y = 2\eu^{3t}$. Para la segunda: equilibrio $3$, luego
$y = 3 + C\eu^{-2t}$ con $y(0) = 0$: $C = -3$:
$y = 3\left(1 - \eu^{-2t}\right)$.

\textbf{2.} $\left(y\eu^{-at}\right)' = y'\eu^{-at} - ay\eu^{-at} =
(y' - ay)\eu^{-at} = 0$: el producto es una constante $C$, luego
$y = C\eu^{at}$; ninguna solución se escapa.

\textbf{3.} Equilibrio: $y \equiv 3$. Todas las demás soluciones son
$3 + C\eu^{-2t}$ con $C \neq 0$: la exponencial muere y la solución se
desliza hasta $3$; un \omterm{pb:g10:functions:1}{punto fijo} estable, el gemelo \omterm{def:g12:limcont:continuity}{continuo} de los \omterm{pb:g10:functions:1}{puntos
fijos} de las recurrencias del \cref{pb:g11:seq:1}.

\textbf{4.} $y = y_0\eu^{-t/\tau}$ se reduce a la mitad cuando
$\eu^{-t/\tau} = \frac12$: $t = \tau\ln 2$.

\textbf{5.} Todas las soluciones son traslaciones verticales de la caída
hacia $3$: las que empiezan por encima bajan hasta $3$, las que empiezan
por debajo suben hasta $3$ y la solución constante $3$ se queda quieta; un
embudo alrededor del equilibrio.

\textbf{6.} $\lambda = \frac{\ln 2}{5730} \approx 1.21 \times 10^{-4}$ por
año.

\textbf{7.} $\eu^{-\lambda t} = 0.2$:
$t = \frac{\ln 5}{\lambda} \approx 13\,300$ años.

\textbf{8.} $t = \frac{\ln(1/0.15)}{\lambda} \approx 15\,700$ años: los
toros de Lascaux son del final de la última glaciación.

\textbf{9.} $t = \frac{\ln(1/0.53)}{\lambda} \approx 5\,250$ años: a menos
de una vida humana del valor de los arqueólogos; el reloj funciona.

\textbf{10.} Diez periodos dejan $2^{-10} \approx 0.1\,\%$: al borde de lo
medible. Tras $65$ millones de años, la fracción es
$2^{-65\,000\,000/5\,730} \approx 2^{-11\,344}$: no queda ni un átomo de
las reservas originales en ningún fósil; los dinosaurios se datan con
relojes más lentos (el de potasio-argón y sus parientes).

\textbf{11.} Un $52\,\%$ da unos $5\,405$ años y un $54\,\%$, unos
$5\,095$: el $\pm 1\,\%$ de la química se convierte en unos $\pm 155$ años
de historia; la precisión del laboratorio es la precisión del cartel del
museo.

\textbf{12.} $z = T - T_a$ cumple $z' = -kz$:
$z = (T_0 - T_a)\eu^{-kt}$, y $T = T_a + z$.

\textbf{13.} La \omterm{def:g11:seq:arithmetic}{diferencia} inicial es $T_0 - T_a = 70$; a los cinco
minutos, la \omterm{def:g11:seq:arithmetic}{diferencia} es $70 - 20 = 50$, luego $50 = 70\eu^{-5k}$:
$k = \frac{\ln(70/50)}{5} \approx 0.067$ por minuto. Bebible:
$55 - 20 = 35 = 70\eu^{-kt}$: $t = \frac{\ln 2}{k} \approx 10.3$ minutos.

\textbf{14.} El ritmo de pérdida es proporcional a la \omterm{def:g11:seq:arithmetic}{diferencia}
$T - T_a$: el café ardiendo pierde calor más deprisa. Echar la leche
\emph{ahora} recorta la \omterm{def:g11:seq:arithmetic}{diferencia} de inmediato, así que se pierde menos
calor en los diez minutos; echarla al final deja que el café se enfríe
antes a toda velocidad. Para la taza más caliente: la leche, primero. (Los
anfitriones impacientes tienen la termodinámica del revés.)

\textbf{15.} \omterm{def:g11:seq:arithmetic}{Diferencias} respecto del ambiente: a medianoche, $10$; una
hora después, $8$: $\eu^{-k} = 0.8$:
$k = \ln\frac{10}{8} \approx 0.223$ por hora. En el momento de la muerte la
\omterm{def:g11:seq:arithmetic}{diferencia} era $17$ y decayó hasta $10$ a medianoche:
$s = \frac{\ln(17/10)}{k} \approx 2.4$ horas. Hora de la muerte: sobre las
$21{:}40$.

\textbf{16.} Una habitación caldeada o con corriente ($T_a$ no constante)
tuerce el reloj en un sentido u otro; la ropa o la masa corporal cambian
$k$ (las tablas forenses lo corrigen), y una $k$ equivocada reescala todo
el \omterm{def:g10:numbers:interval}{intervalo}; y un cuerpo trasladado desde otro lugar reinicia $T_a$ a
\omterm{def:g11:stat:mean}{media} desintegración y finge una muerte más temprana o más tardía. La
\omterm{def:g10:algebra:equation}{ecuación} es honrada; el riesgo lo llevan las hipótesis.

\textbf{17.} $z = \frac1y$ cumple $z' = 1 - z$: $z = 1 + 9\eu^{-t}$ (de
$z(0) = 10$), luego $y = \frac{1}{1 + 9\eu^{-t}}$. Inflexión en
$y = \frac12$: $9\eu^{-t} = 1$: $t = \ln 9 \approx 2.2$; la fecha del giro
de la epidemia, directamente de la fórmula.

\textbf{18.} Velocidad límite: $v' = 0$ en $v = 20$ m/s. Solución:
$v(t) = 20\left(1 - \eu^{-t/2}\right)$. Y el $0.95$:
$\eu^{-t/2} = 0.05$: $t = 2\ln 20 \approx 6$ s; las gotas de lluvia
alcanzan su velocidad de crucero en segundos, y por eso la lluvia no mata.

\textbf{19.} Régimen estacionario: $c = 8$. La mitad:
$4 = 8\left(1 - \eu^{-t/2}\right)$: $\eu^{-t/2} = \frac12$:
$t = 2\ln 2 \approx 1.4$ horas. El gotero es el gemelo \omterm{def:g12:limcont:continuity}{continuo} del
préstamo: entrada constante y salida proporcional; el
\cref{exo:g12:diffeq:9} hace exacto el diccionario.

\textbf{20.} Desintegración ($a < 0$, $b = 0$), enfriamiento
($y = T - T_a$), caída ($a < 0$, $b > 0$: acercamiento al equilibrio desde
abajo) y dosificación (lo mismo, en una vena): una \omterm{def:g10:algebra:equation}{ecuación}, cuatro mundos.
El ciclo: enunciar la ley de cambio; escribir la \omterm{def:g10:algebra:equation}{ecuación}; resolver
(\cref{met:g12:diffeq:solve}); calibrar las constantes con las medidas;
predecir; y auditar después las hipótesis (pregunta 16). La oscilación
necesita $y'' = -\omega^2 y$, con la que ya nos encontramos, que ya
resolvimos y que ya pusimos a oscilar en el \cref{pb:g12:trigo:1}.
\end{solution}
