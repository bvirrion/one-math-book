\section*{Capítulo \ref{ch:g12:condprob} --- Probabilidad condicional e independencia}

\begin{solution}{exo:g12:condprob:1}
Hay $12$ figuras, de las cuales $4$ son reyes:
$\pcond{\text{figura}}{\text{rey}} = \frac{4}{12} = \frac13$.

\omterm{def:g12:condprob:indep}{Independencia}:
$\P(\text{rey} \cap \text{corazones}) = \frac{1}{52}$ (el rey de
corazones), y
$\P(\text{rey})\,\P(\text{corazones}) = \frac{4}{52}\times\frac{13}{52}
= \frac{1}{52}$. Coinciden: los \omterm{def:g11:prob:model}{sucesos} son \omterm{def:g12:condprob:indep}{independientes}.
\end{solution}

\begin{solution}{exo:g12:condprob:2}
\emph{1.} Primeras ramas: roja $\frac58$, azul $\frac38$; segundas ramas
(sin reposición): tras roja, roja $\frac47$ / azul $\frac37$; tras azul,
roja $\frac57$ / azul $\frac27$.

\emph{2.} $\P(RR) = \frac58 \times \frac47 = \frac{5}{14}$. Por la
\omterm{def:g10:proba:distribution}{probabilidad} total,
\[
\P(\text{la segunda es roja}) = \frac58\cdot\frac47
+ \frac38\cdot\frac57 = \frac{20 + 15}{56} = \frac58 .
\]
(Lo mismo que en la primera extracción: por simetría, la segunda bola es
una bola de la urna elegida al azar de manera uniforme.)
\end{solution}

\begin{solution}{exo:g12:condprob:3}
$\P(A) = \frac{6}{36} = \frac16$ (seis parejas suman $7$),
$\P(B) = \frac16$, y $A \cap B = \{(3,4)\}$ tiene \omterm{def:g10:proba:distribution}{probabilidad}
$\frac{1}{36} = \P(A)\P(B)$: $A$ y $B$ son \omterm{def:g12:condprob:indep}{independientes}. (La suma $7$ es
especial: sea cual sea el primer dado, exactamente un valor del segundo la
produce.)

$\P(C) = \frac{5}{36}$ y $B \cap C = \{(3,3)\}$:
$\P(B \cap C) = \frac{1}{36} \neq \frac16 \times \frac{5}{36}
= \frac{5}{216}$. No son \omterm{def:g12:condprob:indep}{independientes}.
\end{solution}

\begin{solution}{exo:g12:condprob:4}
\emph{1.} \omterm{def:g10:proba:distribution}{Probabilidad} total con la partición por máquinas:
\[
\P(D) = 0.5\times0.01 + 0.3\times0.02 + 0.2\times0.04
= 0.005 + 0.006 + 0.008 = 0.019 = 1.9\,\% .
\]

\emph{2.} Bayes:
$\pcond{D}{M_3} = \dfrac{0.008}{0.019} = \dfrac{8}{19} \approx 0.42$. La
máquina que produce solo una quinta parte del total explica más del
$40\,\%$ de las piezas defectuosas.
\end{solution}

\begin{solution}{exo:g12:condprob:5}
La condición es
\[
\frac{0.99\,p}{0.99\,p + 0.05\,(1-p)} \geq 0.9 .
\]
El denominador es positivo, así que esto se lee
$0.99p \geq 0.891p + 0.045 - 0.045p$, es decir,
$0.144\,p \geq 0.045$, es decir,
\[
p \geq \frac{0.045}{0.144} = 0.3125 .
\]
La prueba, por sí sola, alcanza el $90\,\%$ de certeza únicamente si la
enfermedad afecta ya a más del $31\,\%$ de la población examinada; por eso
el cribado masivo de enfermedades raras exige pruebas de confirmación.
\end{solution}

\begin{solution}{exo:g12:condprob:6}
\emph{1.} Tres sucesiones realizan $E$ (cara-cara-cruz, cara-cruz-cara y
cruz-cara-cara), cada una de \omterm{def:g10:proba:distribution}{probabilidad} $p^2(1-p)$ por \omterm{def:g12:condprob:indep}{independencia}:
$\P(E) = 3p^2(1 - p)$.

\emph{2.} $f(p) = 3p^2 - 3p^3$ tiene $f'(p) = 6p - 9p^2 = 3p(2 - 3p)$,
positiva y luego negativa en $\intoo{0}{1}$: \omterm{def:g10:functions:extrema}{máximo} en $p = \frac23$, donde
$\P(E) = 3\cdot\frac49\cdot\frac13 = \frac49$.
\end{solution}

\begin{solution}{exo:g12:condprob:7}
Supongamos que has elegido la puerta 1 y que el presentador ha abierto la
puerta 3 (\omterm{def:g11:prob:model}{suceso} $H_3$). Con $B_i$ = ``el premio está detrás de la puerta
$i$'', $\P(B_i) = \frac13$ y
\[
\pcond{B_1}{H_3} = \frac12, \qquad
\pcond{B_2}{H_3} = 1, \qquad
\pcond{B_3}{H_3} = 0
\]
(si el premio está detrás de tu puerta, el presentador elige al azar entre
las puertas 2 y 3; y si está detrás de la puerta 2, está obligado a abrir
la 3). Bayes:
\[
\pcond{H_3}{B_1}
= \frac{\frac13\cdot\frac12}{\frac13\cdot\frac12 + \frac13\cdot1 + 0}
= \frac{1/6}{1/2} = \frac13,
\qquad
\pcond{H_3}{B_2} = \frac{\frac13}{\frac12} = \frac23 .
\]
Quedarse gana con \omterm{def:g10:proba:distribution}{probabilidad} $\frac13$; cambiar, con \omterm{def:g10:proba:distribution}{probabilidad}
$\frac23$.
\end{solution}

\begin{solution}{exo:g12:condprob:8}
\emph{1.} La mayoría se equivoca cuando se invierten al menos dos de las
tres copias:
\[
3\varepsilon^2(1-\varepsilon) + \varepsilon^3
= 3(0.01)(0.9) + 0.001 = 0.028,
\]
mucho menor que $\varepsilon = 0.1$: la codificación por repetición
funciona.

\emph{2.} Si se envió un $1$ (como $111$), recibir $101$ exige exactamente
una inversión: \omterm{def:g10:proba:distribution}{probabilidad} $\varepsilon(1-\varepsilon)^2 = 0.081$. Si se
envió un $0$ ($000$), hacen falta dos inversiones:
$\varepsilon^2(1-\varepsilon) = 0.009$. Bayes con \omterm{def:g10:proba:distribution}{probabilidades} previas
iguales:
\[
\pcond{101}{\,\text{se envió }1} = \frac{0.081}{0.081 + 0.009} = 0.9 .
\]
La descodificación por mayoría ($101 \mapsto 1$) es, en efecto, la
conjetura más probable.
\end{solution}

\begin{solution}{pb:g12:condprob:1}
\textbf{1.} $\P(\text{roja}) = \frac12 \cdot \frac23 + \frac12
\cdot \frac14 = \frac13 + \frac18 = \frac{11}{24}$.

\textbf{2.} $\P(\text{I} \mid \text{roja}) =
\dfrac{\frac12 \cdot \frac23}{\frac{11}{24}} =
\dfrac{1/3}{11/24} = \dfrac{8}{11}$.

\textbf{3.} $\P(\text{par} \cap \leq 4) = \P(\{2, 4\}) = \frac13$ y
$\P(\text{par})\P(\leq 4) = \frac12 \cdot \frac23 = \frac13$:
\omterm{def:g12:condprob:indep}{independientes} (¡inesperadamente!). Para $\leq 3$: $\P(\{2\}) = \frac16$
frente a $\frac12 \cdot \frac12 = \frac14$: dependientes.

\textbf{4.} Los complementarios de \omterm{def:g11:prob:model}{sucesos} \omterm{def:g12:condprob:indep}{independientes} son a su vez
\omterm{def:g12:condprob:indep}{independientes}, así que
\[
\P(\text{ninguno}) = 0.1 \times 0.1 = 0.01,
\qquad
\P(\text{al menos uno}) = 0.99 .
\]

\textbf{5.} $\frac{4}{52} \times \frac{3}{51} = \frac{1}{221}$.

\textbf{6.} Enfermas: $100$; sanas: $99\,900$. Positivos: $99$ verdaderos
(el $99\,\%$ de $100$) y $4\,995$ falsos (el $5\,\%$ de $99\,900$):
$5\,094$ positivos en total.

\textbf{7.} $\P(\text{enfermo} \mid +) = \frac{99}{5\,094}
\approx 1.9\,\%$: de las personas que la prueba señala, menos de una de
cada cincuenta está realmente enferma.

\textbf{8.} $\dfrac{0.001 \times 0.99}{0.001 \times 0.99 +
0.999 \times 0.05} = \dfrac{0.00099}{0.05094} \approx 0.019$: los
recuentos y la fórmula coinciden.

\textbf{9.} Se confunde $\P(+ \mid \text{enfermo}) = 99\,\%$ (la calidad
que anuncia la prueba) con $\P(\text{enfermo} \mid +)$ (la pregunta que se
hace el paciente). Con una tasa base de $\frac{1}{1000}$, los enfermos son
tan escasos que hasta una fuga del $5\,\%$ de la enorme mayoría sana
sepulta a los positivos verdaderos en proporción de cincuenta a uno.

\textbf{10.} $\dfrac{0.019 \times 0.99}{0.019 \times 0.99 +
0.981 \times 0.05} \approx 28\,\%$. Cada positivo independiente multiplica
las \omterm{def:g10:proba:distribution}{probabilidades} relativas por el mismo factor: los indicios se acumulan,
y un tercer positivo pasaría del $90\,\%$; por eso existen las pruebas de
confirmación.

\textbf{11.} $\dfrac{0.05 \times 0.99}{0.05 \times 0.99 +
0.95 \times 0.05} \approx 51\,\%$: sobre un grupo de riesgo, la misma
prueba sí informa. Moraleja: hay que cribar donde la tasa base sea
apreciable; cribar masivamente una enfermedad rara fabrica falsas alarmas
por millares.

\textbf{12.} Número esperado de coincidencias inocentes:
$10^7 \times 10^{-6} = 10$. Entre las aproximadamente $11$ personas que
coinciden ($10$ inocentes y $1$ culpable, si el culpable está en la
ciudad), el sospechoso hallado \emph{solo} por la coincidencia es inocente
con \omterm{def:g10:proba:distribution}{probabilidad} de en torno a $\frac{10}{11} \approx 91\,\%$.

\textbf{13.} El fiscal citó
$\P(\text{coincidencia} \mid \text{inocente}) = 10^{-6}$; el tribunal
necesitaba $\P(\text{inocente} \mid \text{coincidencia})$: la confusión de
la pregunta 9, pero bajo juramento. (Y en el caso real que popularizó estos
errores, un segundo desatino lo agravó: véase la pregunta 14.)

\textbf{14.} Elevar al cuadrado ingenuamente:
$\left(\frac{1}{8500}\right)^2 \approx \frac{1}{72 \text{ millones}}$. Con
la segunda desgracia cinco veces más probable dada la primera:
$\frac{1}{8500} \times \frac{5}{8500} \approx
\frac{1}{14.5 \text{ millones}}$: cinco veces mayor, y sigue sin tenerse en
cuenta que la hipótesis \emph{alternativa} también tiene que pesarse con
Bayes. Lección: multiplicar \omterm{def:g10:proba:distribution}{probabilidades} exige \omterm{def:g12:condprob:indep}{independencia}, y las
causas compartidas (la genética, el entorno) la destruyen.

\textbf{15.} Bayes multiplica los cocientes de verosimilitud de
\emph{todos} los indicios hasta obtener una única \omterm{def:g10:proba:distribution}{probabilidad} posterior:
la coincidencia, la coartada, las fibras, y cada uno desplaza las
\omterm{def:g10:proba:distribution}{probabilidades} relativas. La frase para el jurado: \emph{pregúntate qué
probable es el indicio bajo cada una de las dos hipótesis y no confundas
nunca una \omterm{def:g12:condprob:cond}{probabilidad condicional} con la contraria}.

\textbf{16.} $\dfrac{0.4 \times 0.15}{0.4 \times 0.15 +
0.6 \times 0.005} = \dfrac{0.060}{0.063} \approx 95\,\%$ de que sea
basura: una sola palabra casi lo decide.

\textbf{17.} Cinco copias fallan cuando se invierten $3$, $4$ o $5$:
$\binom53 \varepsilon^3(1-\varepsilon)^2 + \binom54
\varepsilon^4 (1 - \varepsilon) + \varepsilon^5 \approx 0.0086$, frente al
$0.028$ de tres copias: la repetición compra fiabilidad al precio de ancho
de banda.

\textbf{18.} Número par de inversiones:
$\binom40 0.9^4 + \binom42 0.9^2 0.1^2 + \binom44 0.1^4
= 0.6561 + 0.0486 + 0.0001 = 0.7048$; y
$\frac{1 + 0.8^4}{2} = \frac{1 + 0.4096}{2} = 0.7048$: la fórmula astuta
sigue el sesgo $(1 - 2\varepsilon)$ a través de cada repetidor.

\textbf{19.} Las palabras del correo basura viajan en manada: un mensaje
con ``lotería'' contiene muy probablemente ``ganar'', así que su indicio
conjunto es más débil de lo que finge el producto. El filtro sigue
funcionando porque la exageración suele empujar en la \emph{dirección
correcta}: clasificar necesita el lado correcto del $50\,\%$, no la
\omterm{def:g10:proba:distribution}{probabilidad} posterior exacta.

\textbf{20.} Consulta: una enfermedad rara convierte una prueba fuerte en
un veredicto débil; actualiza desde la tasa base y después acumula.
Tribunal: una coincidencia rara en una ciudad grande acusa a la ciudad, no
al hombre; las verosimilitudes no son veredictos. Bandeja de entrada:
muchos indicios débiles, multiplicados, ordenan el correo, incluso de
manera ingenua. Las tres trampas: olvida la \omterm{def:g10:proba:distribution}{probabilidad} previa y te
engañarán; finge \omterm{def:g12:condprob:indep}{independencia} y te engañarán muchísimo más; lee los
indicios de uno en uno y nunca sabrás lo que sabes.
\end{solution}
