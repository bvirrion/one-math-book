\begin{solution}{exo:g12:integ:1}
\[
\int_1^2 \left(3x^2 - \frac{1}{x^2}\right)\dd x
= \left[x^3 + \frac1x\right]_1^2 = \left(8 + \tfrac12\right) - 2 = \frac{13}{2}.
\]
\[
\int_0^{\pi/2} \cos t\,\dd t = \bigl[\sin t\bigr]_0^{\pi/2} = 1 .
\]
Con $u = 2t+1$, $\frac{1}{2t+1} = \frac12\,\frac{u'}{u}$:
\[
\int_0^1 \frac{\dd t}{2t+1} = \frac12\bigl[\ln(2t+1)\bigr]_0^1
= \frac{\ln 3}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:4}
Llevar $ u' = 1$, $ v = \ln t $, entonces $ u = t $, $ v' = \frac1t $:
\[
\int_1^{\eu} \ln t\,\dd t = \bigl[t\ln t\bigr]_1^{\eu} - \int_1^{\eu} 1\,\dd t
= \eu - (\eu - 1) = 1 .
\]
Llevar $ u' = \sin t $, $ v = t $, entonces $ u = -\cos t $, $ v' = 1$:
\[
\int_0^\pi t\sin t\,\dd t = \bigl[-t\cos t\bigr]_0^\pi + \int_0^\pi \cos t\,\dd t
= \pi + 0 = \pi .
\]
\end{solution}

\begin{solution}{exo:g12:integ:5}
Las curvas se cruzan donde $ x^2 = x + 2$, \emph{es decir.} $ x \in \{-1, 2\}$, y
en $\intcc{-1}{2}$ la línea está por encima del \omterm{def:g11:quad:parabola}{parábola}. El área es
\[
\int_{-1}^{2} \bigl(x + 2 - x^2\bigr)\dd x
= \left[\frac{x^2}{2} + 2x - \frac{x^3}{3}\right]_{-1}^{2}
= \frac{10}{3} - \left(-\frac{7}{6}\right) = \frac{9}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:6}
\emph{1.} $ I = \bigl[\ln(1+t)\bigr]_0^1 = \ln 2$.

\emph{2.}
\[
I_n + I_{n+1} = \int_0^1 \frac{t^n(1 + t)}{1+t}\,\dd t
= \int_0^1 t^n \,\dd t = \frac{1}{n+1}.
\]
En $\intcc{0}{1}$, $0 \leq \dfrac{t^n}{1+t} \leq t^n $, entonces en comparación
$0 \leq I_n \leq \frac{1}{n+1}$.

\emph{3.} A partir de la recurrencia, por una inducción telescópica,
\[
I_n = (-1)^n\left(I_0 - \left(1 - \frac12 + \dots + \frac{(-1)^{n-1}}{n}\right)\right),
\]
\emph{es decir.}
$\displaystyle\sum_{k=1}^{n} \frac{(-1)^{k-1}}{k} = \ln 2 - (-1)^n I_n $
(recordar $ I_0 = \ln 2$). Desde $\abs{I_n} \leq \frac{1}{n+1} \to 0$, el
la suma armónica alterna tiende a $\ln 2$.
\end{solution}

\begin{solution}{exo:g12:integ:7}
Distancia:
\[
d = \int_0^{100} 30\left(1 - \eu^{-t/50}\right)\dd t
= 30\left[t + 50\,\eu^{-t/50}\right]_0^{100}
= 30\left(100 + 50\eu^{-2} - 50\right)
= 1500 + 1500\,\eu^{-2}.
\]
Numéricamente $ d \approx 1500 + 203 = 1703$ metro. Velocidad \omterm{def:g11:stat:mean}{media}:
$\frac{d}{100} = 15\bigl(1 + \eu^{-2}\bigr) \approx 17.0$ EM.
\end{solution}

\begin{solution}{exo:g12:integ:8}
\emph{1.} $ S_n $ es el área total de $ n $ rectángulos de ancho $\frac1n $ y
alturas $ f\left(\frac kn\right)$, $ k = 1, \dots, n $, dónde
$ f(t) = \frac{1}{1+t}$: es el rectángulo del ``extremo derecho'' \omterm{def:g10:numbers:approx}{aproximación}
del área bajo $ f $ en $\intcc{0}{1}$.

\emph{2.} $ f $ es \omterm{def:g10:functions:variations}{decreciente}, así en cada \omterm{def:g10:numbers:interval}{intervalo}
$\intcc{\frac{k-1}{n}}{\frac kn}$,
$ f\left(\frac kn\right) \leq f(t) \leq f\left(\frac{k-1}{n}\right)$;
integrando y sumando $ k $:
\[
S_n \leq \int_0^1 f(t)\,\dd t \leq S_n + \frac1n\bigl(f(0) - f(1)\bigr)
= S_n + \frac{1}{2n}.
\]
(El término medio de la visualización del ejercicio, con
$ f(0) - f(1) = 1 - \frac12$, es exactamente este límite.) Por lo tanto
$0 \leq \ln 2 - S_n \leq \frac{1}{2n} \to 0$, entonces $ S_n \to \ln 2$.
\end{solution}

\begin{solution}{exo:g12:integ:9}
\emph{1.} $ W_0 = \int_0^{\pi/2} \dd t = \frac{\pi}{2}$ y
$ W_1 = \bigl[-\cos t\bigr]_0^{\pi/2} = 1$.

\emph{2.} Integrar por partes con $ u' = \sin t $, $ v = \sin^{n+1} t $, entonces
$ u = -\cos t $, $ v' = (n+1)\sin^n t \cos t $:
\[
W_{n+2} = \bigl[-\cos t \sin^{n+1} t\bigr]_0^{\pi/2}
+ (n+1)\int_0^{\pi/2} \cos^2 t\,\sin^n t\,\dd t
= (n+1)\int_0^{\pi/2} (1 - \sin^2 t)\sin^n t\,\dd t,
\]
el soporte desaparece en ambos extremos. Por eso
$ W_{n+2} = (n+1)(W_n - W_{n+2})$, \emph{es decir.}
$ W_{n+2} = \frac{n+1}{n+2} W_n $.

\emph{3.} $ W_2 = \frac12 W_0 = \frac{\pi}{4}$,
$ W_3 = \frac23 W_1 = \frac23$,
$ W_4 = \frac34 W_2 = \frac{3\pi}{16}$.
En $\intoo{0}{\frac\pi2}$, $0 < \sin t < 1$, entonces
$\sin^{n+1} t \leq \sin^n t $ con estricta desigualdad en el interior;
integrando, $0 < W_{n+1} \leq W_n $ (En realidad $<$): $(W_n)$ es \omterm{def:g10:functions:variations}{decreciente} y
positivo.
\end{solution}

\begin{solution}{pb:g12:integ:1}
\textbf{1.} $\left[x^3 - x^2 + x\right]_0^1 = 1$; \quad
$\left[\ln x\right]_1^{\eu} = 1$; \quad
$\left[\sin x\right]_0^{\pi/2} = 1$.

\textbf{2.} $\left[\frac12 \eu^{x^2}\right]_0^1 =
\frac{\eu - 1}{2}$; \quad
$\left[\ln(x^2 + 1)\right]_0^1 = \ln 2$.

\textbf{3.} $\int_0^1 x\eu^x = \left[x\eu^x\right]_0^1 -
\int_0^1 \eu^x = \eu - (\eu - 1) = 1$. Y
$\int_1^{\eu} \ln x = \left[x\ln x\right]_1^{\eu} -
\int_1^{\eu} 1 = \eu - (\eu - 1) = 1$.

\textbf{4.} $\frac{1}{\pi}\int_0^\pi \sin x\,\dd x =
\frac{2}{\pi} \approx 0.64$: la altura promedio del \omterm{def:g11:trigo:cossin}{señor}
el arco es $\frac2\pi $, notablemente más de la mitad --- el
El arco es grueso cerca de su parte superior.

\textbf{5.} $\int_0^1 (x - x^2)\,\dd x = \frac12 - \frac13 =
\frac16$.

\textbf{6.} $ L_n = \frac1n \sum_{k=0}^{n-1}
\left(\frac kn\right)^2$ y
$ U_n = \frac1n \sum_{k=1}^{n} \left(\frac kn\right)^2$:
rectángulos de ancho $\frac1n $ con alturas leídas a la izquierda
(debajo de la curva) o puntos finales derechos (arriba).

\textbf{7.} cierto para $ n = 1$ ($1 = \frac{1 \cdot 2 \cdot
3}{6}$). Si se mantiene por $ n $, luego agregando $(n+1)^2$:
\[
\frac{n(n+1)(2n+1)}{6} + (n+1)^2
= \frac{(n+1)\left(2n^2 + n + 6n + 6\right)}{6}
= \frac{(n+1)(n+2)(2n+3)}{6},
\]
la fórmula en $ n + 1$: herencia, hecho.

\textbf{8.} $ U_n = \frac{1}{n^3} \cdot
\frac{n(n+1)(2n+1)}{6} = \frac{(1 + \frac1n)(2 + \frac1n)}{6}
\to \frac26 = \frac13$, y $ L_n = U_n - \frac1n \to \frac13$:
apretada entre sus escaleras, el área es exactamente
$\frac13$.

\textbf{9.} $\int_0^1 x^2\,\dd x =
\left[\frac{x^3}{3}\right]_0^1 = \frac13$: una línea. el
El teorema fundamental convierte una compresión infinita en una única.
antiderivada evaluación --- por eso se llama
fundamentales.

\textbf{10.} Segmento: $\int_{-1}^{1}(1 - x^2)\,\dd x =
2 - \frac23 = \frac43$. Triángulo con base la cuerda (longitud
$2$, en altura $1$) y ápice $(0,0)$: área
$\frac12 \times 2 \times 1 = 1$. Relación: $\frac43$ --- exactamente
El teorema de Arquímedes, verificado por la máquina que le faltaba.

\textbf{11.} $ T\left(1 + \frac14 + \frac{1}{16} + \dots\right)
= T \cdot \frac{1}{1 - \frac14} = \frac43 T $ (serie geométrica
de proporción $\frac14$): con $ T = 1$, el área del segmento
$\frac43$ de nuevo. Arquímedes resumió la serie de forma puramente
argumento geométrico --- los cuartos de la barra de chocolate
amable, dos milenios antes de que los límites fueran palabras.

\textbf{12.} El agotamiento necesita una geometría inteligente
autosimilitud y da áreas exactas caso por caso; Riemann
sólo necesita monotonicidad y da una definición universal en
el precio de calcular sumas; el teorema fundamental necesita el
todo el edificio de derivadas y lo paga haciendo áreas
cálculos de una sola línea.

\textbf{13.} $ A(1) = 0$ y $ A'(x) = \frac1x $ (fundamental
teorema): $ A $ es el primitiva de $\frac1x $ en
$\intoo{0}{+\infty}$ desapareciendo en $1$ --- que es precisamente
el logaritmo natural.

\textbf{14.} $ g'(x) = \frac{a}{ax} - \frac1x = 0$: $ g $ es
constante; en $ x = 1$: $ g(1) = A(a) - A(1) = A(a)$. Por eso
$ A(ax) = A(a) + A(x)$ para todo $ x $, y con $ x = b $:
$\ln(ab) = \ln a + \ln b $. El hipérbolaAl área no le importa
sobre la escala horizontal: estiramiento $ t $ por $ a $ calabazas
$\frac1t $ por $ a $, y el área sobrevive --- adición de
logaritmos es una simetría de escala.

\textbf{15.} Iterando: $\ln(a^n) = n\ln a $ (inducción en
$ n $). Y $0 = \ln 1 = \ln\left(a \cdot \frac1a\right) =
\ln a + \ln\frac1a $: $\ln\frac1a = -\ln a $.

\textbf{16.} $\frac{1}{11} + \frac{1}{12} + \dots +
\frac{1}{20} \approx 0.669$, contra $\ln 2 \approx 0.693$:
diez rectángulos debajo $\frac{1}{1+t}$, ya cerca. el
La serie armónica diverge, pero su \emph{rebanadas} de $ n $ a
$2n $ decidirse por $\ln 2$: divergencia a paso lento, medida por un
área.

\textbf{17.} Distancia: $\int_0^{10} 3t^2\,\dd t = 1000$ metro.
\omterm{def:g11:stat:mean}{media} velocidad: $100$ EM. Instantáneo con $ v = 100$:
$3t^2 = 100$: $ t = \sqrt{\frac{100}{3}} \approx 5.77$ s.

\textbf{18.} Masa: $\int_0^4 (2 + x)\,\dd x = 8 + 8 = 16$ kg.
Momento: $\int_0^4 (2x + x^2)\,\dd x = 16 + \frac{64}{3} =
\frac{112}{3}$. Centro de masa:
$\frac{112/3}{16} = \frac73 \approx 2.33$ m --- empujado más allá
el centro por el extremo derecho, más pesado, como exige la intuición.

\textbf{19.} $ W_0 = \frac\pi2$; $ W_1 = \left[-\cos
t\right]_0^{\pi/2} = 1$; y
$ W_2 = \int_0^{\pi/2} \frac{1 - \cos 2t}{2}\,\dd t =
\frac\pi4$.

\textbf{20.} Área: la definición, apretujada por las escaleras.
Acumulación: distancias de velocidades, masas de densidades ---
el \omterm{def:g12:integ:area}{integral} como total acumulado. Antiderivada: el
teorema fundamental, el atajo de Newton y Leibniz a través de
La paciencia de Arquímedes. La perla: la $\ln $ El botón es el área.
bajo $\frac1t $, y la simetría de escala de esa área \emph{es}
la identidad $\ln(ab) = \ln a + \ln b$.
\end{solution}
