\section*{Capítulo \ref{ch:g12:integ} --- Integración}

\begin{solution}{exo:g12:integ:1}
\[
\int_1^2 \left(3x^2 - \frac{1}{x^2}\right)\dd x
= \left[x^3 + \frac1x\right]_1^2 = \left(8 + \tfrac12\right) - 2 = \frac{13}{2}.
\]
\[
\int_0^{\pi/2} \cos t\,\dd t = \bigl[\sin t\bigr]_0^{\pi/2} = 1 .
\]
Con $u = 2t+1$, $\frac{1}{2t+1} = \frac12\,\frac{u'}{u}$:
\[
\int_0^1 \frac{\dd t}{2t+1} = \frac12\bigl[\ln(2t+1)\bigr]_0^1
= \frac{\ln 3}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:2}
$x\,\eu^{x^2} = \frac12\,(x^2)'\,\eu^{x^2}$ tiene por \omterm{def:g12:integ:primitive}{primitivas}
$\frac12\eu^{x^2} + c$. La condición $F(0) = 1$ da $\frac12 + c = 1$, luego
\[
F(x) = \frac{\eu^{x^2} + 1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:3}
\[
\mu = \frac{1}{\pi}\int_0^\pi \sin t\,\dd t
= \frac{1}{\pi}\bigl[-\cos t\bigr]_0^\pi = \frac{2}{\pi} \approx 0.64 .
\]
Gráficamente, el rectángulo de base $\intcc{0}{\pi}$ y altura
$\frac{2}{\pi}$ tiene la misma área que el arco de la curva del \omterm{def:g12:trigo:cossin}{seno}.
\end{solution}

\begin{solution}{exo:g12:integ:4}
Tomamos $u' = 1$ y $v = \ln t$, así que $u = t$ y $v' = \frac1t$:
\[
\int_1^{\eu} \ln t\,\dd t = \bigl[t\ln t\bigr]_1^{\eu} - \int_1^{\eu} 1\,\dd t
= \eu - (\eu - 1) = 1 .
\]
Tomamos $u' = \sin t$ y $v = t$, así que $u = -\cos t$ y $v' = 1$:
\[
\int_0^\pi t\sin t\,\dd t = \bigl[-t\cos t\bigr]_0^\pi + \int_0^\pi \cos t\,\dd t
= \pi + 0 = \pi .
\]
\end{solution}

\begin{solution}{exo:g12:integ:5}
Las curvas se cortan donde $x^2 = x + 2$, es decir, en
$x \in \{-1, 2\}$, y en $\intcc{-1}{2}$ la recta queda por encima de la
\omterm{def:g11:quad:parabola}{parábola}. El área es
\[
\int_{-1}^{2} \bigl(x + 2 - x^2\bigr)\dd x
= \left[\frac{x^2}{2} + 2x - \frac{x^3}{3}\right]_{-1}^{2}
= \frac{10}{3} - \left(-\frac{7}{6}\right) = \frac{9}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:6}
\emph{1.} $I = \bigl[\ln(1+t)\bigr]_0^1 = \ln 2$.

\emph{2.}
\[
I_n + I_{n+1} = \int_0^1 \frac{t^n(1 + t)}{1+t}\,\dd t
= \int_0^1 t^n \,\dd t = \frac{1}{n+1}.
\]
En $\intcc{0}{1}$, $0 \leq \dfrac{t^n}{1+t} \leq t^n$, así que por
comparación $0 \leq I_n \leq \frac{1}{n+1}$.

\emph{3.} De la recurrencia, mediante una inducción telescópica,
\[
I_n = (-1)^n\left(I_0 - \left(1 - \frac12 + \dots + \frac{(-1)^{n-1}}{n}\right)\right),
\]
es decir,
$\displaystyle\sum_{k=1}^{n} \frac{(-1)^{k-1}}{k} = \ln 2 - (-1)^n I_n$
(recordemos que $I_0 = \ln 2$). Como $\abs{I_n} \leq \frac{1}{n+1} \to 0$,
la suma armónica alternada tiende a $\ln 2$.
\end{solution}

\begin{solution}{exo:g12:integ:7}
Distancia:
\[
d = \int_0^{100} 30\left(1 - \eu^{-t/50}\right)\dd t
= 30\left[t + 50\,\eu^{-t/50}\right]_0^{100}
= 30\left(100 + 50\eu^{-2} - 50\right)
= 1500 + 1500\,\eu^{-2}.
\]
Numéricamente, $d \approx 1500 + 203 = 1703$ m. Velocidad \omterm{def:g11:stat:mean}{media}:
$\frac{d}{100} = 15\bigl(1 + \eu^{-2}\bigr) \approx 17.0$ m/s.
\end{solution}

\begin{solution}{exo:g12:integ:8}
\emph{1.} $S_n$ es el área total de $n$ rectángulos de anchura $\frac1n$ y
alturas $f\left(\frac kn\right)$, $k = 1, \dots, n$, donde
$f(t) = \frac{1}{1+t}$: es la \omterm{def:g10:numbers:approx}{aproximación} por rectángulos ``por el \omterm{def:g12:deriv:extremum}{extremo}
derecho'' del área bajo $f$ en $\intcc{0}{1}$.

\emph{2.} $f$ es \omterm{def:g10:functions:variations}{decreciente}, así que en cada \omterm{def:g10:numbers:interval}{intervalo}
$\intcc{\frac{k-1}{n}}{\frac kn}$ se cumple
$f\left(\frac kn\right) \leq f(t) \leq f\left(\frac{k-1}{n}\right)$;
integrando y sumando sobre $k$:
\[
S_n \leq \int_0^1 f(t)\,\dd t \leq S_n + \frac1n\bigl(f(0) - f(1)\bigr)
= S_n + \frac{1}{2n}.
\]
(El término central del enunciado, con $f(0) - f(1) = 1 - \frac12$, es
exactamente esa cota.) Por tanto,
$0 \leq \ln 2 - S_n \leq \frac{1}{2n} \to 0$, luego $S_n \to \ln 2$.
\end{solution}

\begin{solution}{exo:g12:integ:9}
\emph{1.} $W_0 = \int_0^{\pi/2} \dd t = \frac{\pi}{2}$ y
$W_1 = \bigl[-\cos t\bigr]_0^{\pi/2} = 1$.

\emph{2.} Integramos por partes con $u' = \sin t$ y $v = \sin^{n+1} t$, así
que $u = -\cos t$ y $v' = (n+1)\sin^n t \cos t$:
\[
W_{n+2} = \bigl[-\cos t \sin^{n+1} t\bigr]_0^{\pi/2}
+ (n+1)\int_0^{\pi/2} \cos^2 t\,\sin^n t\,\dd t
= (n+1)\int_0^{\pi/2} (1 - \sin^2 t)\sin^n t\,\dd t,
\]
porque el corchete se anula en los dos \omterm{def:g12:deriv:extremum}{extremos}. Por tanto,
$W_{n+2} = (n+1)(W_n - W_{n+2})$, es decir,
$W_{n+2} = \frac{n+1}{n+2} W_n$.

\emph{3.} $W_2 = \frac12 W_0 = \frac{\pi}{4}$,
$W_3 = \frac23 W_1 = \frac23$ y $W_4 = \frac34 W_2 = \frac{3\pi}{16}$. En
$\intoo{0}{\frac\pi2}$, $0 < \sin t < 1$, así que
$\sin^{n+1} t \leq \sin^n t$ con desigualdad estricta en el interior;
integrando, $0 < W_{n+1} \leq W_n$ (de hecho, $<$): $(W_n)$ es \omterm{def:g10:functions:variations}{decreciente}
y positiva.
\end{solution}

\begin{solution}{pb:g12:integ:1}
\textbf{1.} $\left[x^3 - x^2 + x\right]_0^1 = 1$; \quad
$\left[\ln x\right]_1^{\eu} = 1$; \quad
$\left[\sin x\right]_0^{\pi/2} = 1$.

\textbf{2.} $\left[\frac12 \eu^{x^2}\right]_0^1 = \frac{\eu - 1}{2}$; \quad
$\left[\ln(x^2 + 1)\right]_0^1 = \ln 2$.

\textbf{3.} $\int_0^1 x\eu^x = \left[x\eu^x\right]_0^1 - \int_0^1 \eu^x =
\eu - (\eu - 1) = 1$. Y
$\int_1^{\eu} \ln x = \left[x\ln x\right]_1^{\eu} - \int_1^{\eu} 1 =
\eu - (\eu - 1) = 1$.

\textbf{4.} $\frac{1}{\pi}\int_0^\pi \sin x\,\dd x = \frac{2}{\pi} \approx
0.64$: la altura \omterm{def:g11:stat:mean}{media} del arco del \omterm{def:g12:trigo:cossin}{seno} es $\frac2\pi$, bastante más de la
mitad; el arco es ancho cerca de su cima.

\textbf{5.} $\int_0^1 (x - x^2)\,\dd x = \frac12 - \frac13 = \frac16$.

\textbf{6.} $L_n = \frac1n \sum_{k=0}^{n-1} \left(\frac kn\right)^2$ y
$U_n = \frac1n \sum_{k=1}^{n} \left(\frac kn\right)^2$: rectángulos de
anchura $\frac1n$ con alturas leídas en el \omterm{def:g12:deriv:extremum}{extremo} izquierdo (por debajo de
la curva) o en el derecho (por encima).

\textbf{7.} Cierto para $n = 1$ ($1 = \frac{1 \cdot 2 \cdot 3}{6}$). Si
vale para $n$, al añadir $(n+1)^2$:
\[
\frac{n(n+1)(2n+1)}{6} + (n+1)^2
= \frac{(n+1)\left(2n^2 + n + 6n + 6\right)}{6}
= \frac{(n+1)(n+2)(2n+3)}{6},
\]
que es la fórmula en $n + 1$: paso de inducción hecho.

\textbf{8.} $U_n = \frac{1}{n^3} \cdot \frac{n(n+1)(2n+1)}{6} =
\frac{(1 + \frac1n)(2 + \frac1n)}{6} \to \frac26 = \frac13$, y
$L_n = U_n - \frac1n \to \frac13$: encajada entre sus escaleras, el área es
exactamente $\frac13$.

\textbf{9.} $\int_0^1 x^2\,\dd x = \left[\frac{x^3}{3}\right]_0^1 =
\frac13$: una línea. El teorema fundamental convierte un encajonamiento
infinito en una única evaluación de \omterm{def:g12:integ:primitive}{primitiva}; por eso se llama
fundamental.

\textbf{10.} Segmento: $\int_{-1}^{1}(1 - x^2)\,\dd x = 2 - \frac23 =
\frac43$. Triángulo de base la cuerda (de longitud $2$, a altura $1$) y
vértice $(0,0)$: área $\frac12 \times 2 \times 1 = 1$. \omterm{def:g11:seq:geometric}{Razón}: $\frac43$,
exactamente el teorema de Arquímedes, comprobado con la máquina de la que
él carecía.

\textbf{11.} $T\left(1 + \frac14 + \frac{1}{16} + \dots\right) =
T \cdot \frac{1}{1 - \frac14} = \frac43 T$ (serie geométrica de \omterm{def:g11:seq:geometric}{razón}
$\frac14$): con $T = 1$, otra vez el área $\frac43$ del segmento.
Arquímedes sumó la serie con un argumento puramente geométrico: los
mordiscos de cuarto en cuarto del estilo de la tableta de chocolate, dos
milenios antes de que existiera la palabra ``límite''.

\textbf{12.} La exhausción necesita una autosemejanza geométrica ingeniosa
y da áreas exactas caso por caso; Riemann solo necesita monotonía y da una
definición universal, al precio de calcular sumas; y el teorema fundamental
necesita todo el edificio de las \omterm{def:g12:deriv:derivative}{derivadas} y lo devuelve convirtiendo las
áreas en cálculos de una línea.

\textbf{13.} $A(1) = 0$ y $A'(x) = \frac1x$ (teorema fundamental): $A$ es
la \omterm{def:g12:integ:primitive}{primitiva} de $\frac1x$ en $\intoo{0}{+\infty}$ que se anula en $1$, que
es precisamente el \omterm{def:g12:exp:ln}{logaritmo neperiano}.

\textbf{14.} $g'(x) = \frac{a}{ax} - \frac1x = 0$: $g$ es constante; en
$x = 1$: $g(1) = A(a) - A(1) = A(a)$. Por tanto,
$A(ax) = A(a) + A(x)$ para todo $x$ y, con $x = b$:
$\ln(ab) = \ln a + \ln b$. Al área bajo la \omterm{def:g11:func:reference}{hipérbola} le da igual la escala
horizontal: estirar $t$ por $a$ aplasta $\frac1t$ por $a$ y el área
sobrevive; sumar \omterm{def:g12:exp:ln}{logaritmos} es una simetría de escala.

\textbf{15.} Iterando: $\ln(a^n) = n\ln a$ (por inducción sobre $n$). Y
$0 = \ln 1 = \ln\left(a \cdot \frac1a\right) = \ln a + \ln\frac1a$:
$\ln\frac1a = -\ln a$.

\textbf{16.} $\frac{1}{11} + \frac{1}{12} + \dots + \frac{1}{20} \approx
0.669$, frente a $\ln 2 \approx 0.693$: diez rectángulos bajo
$\frac{1}{1+t}$ y ya están cerca. La serie armónica diverge, pero sus
\emph{tramos} de $n$ a $2n$ se asientan en $\ln 2$: divergencia a paso de
tortuga, medida por un área.

\textbf{17.} Distancia: $\int_0^{10} 3t^2\,\dd t = 1000$ m. Velocidad
\omterm{def:g11:stat:mean}{media}: $100$ m/s. Instante con $v = 100$: $3t^2 = 100$:
$t = \sqrt{\frac{100}{3}} \approx 5.77$ s.

\textbf{18.} Masa: $\int_0^4 (2 + x)\,\dd x = 8 + 8 = 16$ kg. Momento:
$\int_0^4 (2x + x^2)\,\dd x = 16 + \frac{64}{3} = \frac{112}{3}$. Centro de
masas: $\frac{112/3}{16} = \frac73 \approx 2.33$ m, empujado más allá del
\omterm{prop:g10:coordgeom:midpoint}{punto medio} por el \omterm{def:g12:deriv:extremum}{extremo} derecho, que es el más pesado, como exige la
intuición.

\textbf{19.} $W_0 = \frac\pi2$; $W_1 = \left[-\cos t\right]_0^{\pi/2} = 1$;
y $W_2 = \int_0^{\pi/2} \frac{1 - \cos 2t}{2}\,\dd t = \frac\pi4$.

\textbf{20.} Área: la definición, encajonada entre escaleras. Acumulación:
distancias a partir de velocidades y masas a partir de densidades; la
\omterm{def:g12:integ:area}{integral} como total acumulado. \omterm{def:g12:integ:primitive}{Primitiva}: el teorema fundamental, el atajo
de Newton y Leibniz frente a la paciencia de Arquímedes. La perla: la tecla
$\ln$ es el área bajo $\frac1t$, y la simetría de escala de esa área
\emph{es} la identidad $\ln(ab) = \ln a + \ln b$.
\end{solution}
