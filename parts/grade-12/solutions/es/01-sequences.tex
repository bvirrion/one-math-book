\section*{Capítulo \ref{ch:g12:seq} --- Sucesiones}

\begin{solution}{exo:g12:seq:1}
Sea $P(n)$ la propiedad $\sum_{k=1}^{n} k^2 = \frac{n(n+1)(2n+1)}{6}$.
\emph{Caso base:} para $n = 0$ los dos miembros valen $0$ (suma vacía).
\emph{Paso de inducción:} supongamos $P(n)$. Entonces
\[
\sum_{k=1}^{n+1} k^2 = \frac{n(n+1)(2n+1)}{6} + (n+1)^2
= \frac{(n+1)\bigl(n(2n+1) + 6(n+1)\bigr)}{6}
= \frac{(n+1)(2n^2 + 7n + 6)}{6}.
\]
Como $2n^2 + 7n + 6 = (n+2)(2n+3)$, esto es
$\frac{(n+1)(n+2)(2(n+1)+1)}{6}$, que es $P(n+1)$. Por inducción, $P(n)$
vale para todo $n$.
\end{solution}

\begin{solution}{exo:g12:seq:2}
$a_{n+1} - a_n = \frac{n+2}{n+1} - \frac{n+1}{n}
= \frac{n(n+2) - (n+1)^2}{n(n+1)} = \frac{-1}{n(n+1)} < 0$: $(a_n)$ es
estrictamente \omterm{def:g10:functions:variations}{decreciente}.

$(b_n)$ tiene términos positivos y
$\frac{b_{n+1}}{b_n} = \frac{2^{n+1}}{n+1}\cdot\frac{n}{2^n} =
\frac{2n}{n+1} \geq 1 \iff 2n \geq n+1 \iff n \geq 1$: $(b_n)$ es \omterm{def:g12:seq:monotonic}{creciente}
(estrictamente para $n \geq 2$).

$c_{n+1} - c_n = (n+1)^2 - 10(n+1) - n^2 + 10n = 2n - 9$, que es negativo
para $n \leq 4$ y positivo para $n \geq 5$: $(c_n)$ decrece hasta
$c_5 = -25$, su \omterm{def:g10:functions:extrema}{mínimo}, y crece después. No es \omterm{def:g12:seq:monotonic}{monótona}.
\end{solution}

\begin{solution}{exo:g12:seq:3}
Sacamos factor común el término dominante:
\[
u_n = \frac{n^2(3 - 1/n + 1/n^2)}{n^2(2 + 5/n^2)} \xrightarrow[n\to+\infty]{} \frac{3}{2}.
\]
Multiplicamos por el conjugado:
\[
v_n = \frac{(n+1) - n}{\sqrt{n+1} + \sqrt{n}} = \frac{1}{\sqrt{n+1}+\sqrt{n}}
\xrightarrow[n\to+\infty]{} 0.
\]
Dividimos numerador y denominador entre $3^n$:
\[
w_n = \frac{(2/3)^n - 1}{1 + (1/3)^n} \xrightarrow[n\to+\infty]{} \frac{0-1}{1+0} = -1,
\]
usando $\lim q^n = 0$ para $\abs{q} < 1$.
\end{solution}

\begin{solution}{exo:g12:seq:4}
$u_n = 5 + 3n \to +\infty$ y $v_n = 8 \cdot (1/2)^n = 2^{3-n} \to 0$. Las
sumas son
\[
\sum_{k=0}^{n} u_k = (n+1)\,\frac{5 + (5+3n)}{2} = \frac{(n+1)(10+3n)}{2}
\xrightarrow[n\to+\infty]{} +\infty,
\]
\[
\sum_{k=0}^{n} v_k = 8\,\frac{1 - (1/2)^{n+1}}{1 - 1/2}
= 16\left(1 - \left(\tfrac{1}{2}\right)^{n+1}\right)
\xrightarrow[n\to+\infty]{} 16 .
\]
\end{solution}

\begin{solution}{exo:g12:seq:5}
Como $-1 \leq \cos n \leq 1$,
\[
\frac{n-1}{n+1} \leq \frac{n + \cos n}{n+1} \leq 1,
\]
y $\frac{n-1}{n+1} \to 1$, así que el límite es $1$ por el teorema del
sándwich.

Para el segundo límite, escribimos
\[
0 \leq \frac{n!}{n^n}
= \frac{1}{n}\cdot\frac{2}{n}\cdots\frac{n}{n}
\leq \frac{1}{n},
\]
porque cada factor $\frac{k}{n}$ con $2 \leq k \leq n$ es como mucho $1$.
Como $\frac1n \to 0$, el teorema del sándwich da $\frac{n!}{n^n} \to 0$.
(La cota geométrica sugerida también funciona: cada factor con
$k \leq n/2$ es como mucho $\frac12$, lo que da la cota más fina
$(1/2)^{\floor{n/2}}$.)
\end{solution}

\begin{solution}{exo:g12:seq:6}
\emph{1.} $u_0 = 0 \in \intcc{0}{2}$. Si $0 \leq u_n \leq 2$, entonces
$2 \leq u_n + 2 \leq 4$, luego $\sqrt{2} \leq u_{n+1} \leq 2$; en
particular, $0 \leq u_{n+1} \leq 2$. Por inducción, la propiedad vale para
todo $n$.

\emph{2.} $u_{n+1} - u_n = \sqrt{u_n + 2} - u_n$. Para
$x \in \intcc{0}{2}$, $\sqrt{x+2} \geq x \iff x + 2 \geq x^2 \iff
(2-x)(x+1) \geq 0$, que es cierto. Por tanto, $(u_n)$ es \omterm{def:g12:seq:monotonic}{creciente}.

\emph{3.} \omterm{def:g12:seq:monotonic}{Creciente} y \omterm{def:g12:seq:bounded}{acotada superiormente} por $2$, $(u_n)$ \omterm{def:g12:seq:limit}{converge} a
cierto $\ell \in \intcc{0}{2}$. Pasando al límite en
$u_{n+1} = \sqrt{u_n + 2}$ (la aplicación $x \mapsto \sqrt{x+2}$ es
continua) se obtiene $\ell = \sqrt{\ell + 2}$, luego
$\ell^2 - \ell - 2 = 0$, es decir, $\ell \in \{-1, 2\}$. Como
$\ell \geq 0$, $\lim u_n = 2$.
\end{solution}

\begin{solution}{exo:g12:seq:7}
\emph{1.} Entre dos dosis se elimina el $40\,\%$ del medicamento, así que
la cantidad $u_n$ pasa a ser $0.6\,u_n$; la dosis siguiente añade $1$
unidad: $u_{n+1} = 0.6\,u_n + 1$.

\emph{2.} $v_{n+1} = u_{n+1} - 2.5 = 0.6\,u_n + 1 - 2.5 = 0.6(u_n - 2.5)
= 0.6\,v_n$: $(v_n)$ es una \omterm{def:g12:seq:arith-geom}{progresión geométrica} de \omterm{def:g11:seq:geometric}{razón} $0.6$ y primer
término $v_0 = 1 - 2.5 = -1.5$. Por tanto,
$v_n = -1.5 \times 0.6^n$ y
\[
u_n = 2.5 - 1.5 \times 0.6^n .
\]

\emph{3.} Como $0.6^n \to 0$, $u_n \to 2.5$: la cantidad de medicamento se
estabiliza en $2.5$ unidades.
\end{solution}

\begin{solution}{exo:g12:seq:8}
\emph{1.} $u_0 = 3 > 1$. Si $u_n > 1$, entonces $u_n + 2 > 0$ y
\[
u_{n+1} - 1 = \frac{4u_n - 1 - u_n - 2}{u_n + 2} = \frac{3(u_n - 1)}{u_n + 2} > 0 .
\]
Por inducción, $u_n > 1$ para todo $n$ (y, en particular,
$u_n + 2 \neq 0$, así que la \omterm{def:g12:seq:sequence}{sucesión} está bien definida).

\emph{2.} Usando la identidad anterior,
\[
v_{n+1} = \frac{1}{u_{n+1} - 1} = \frac{u_n + 2}{3(u_n - 1)}
= \frac{(u_n - 1) + 3}{3(u_n - 1)} = \frac{1}{3} + v_n .
\]
Así que $(v_n)$ es una \omterm{def:g12:seq:arith-geom}{progresión aritmética} de \omterm{def:g11:seq:arithmetic}{diferencia} $\frac13$ con
$v_0 = \frac{1}{u_0 - 1} = \frac12$.

\emph{3.} $v_n = \frac12 + \frac{n}{3}$, luego
$u_n = 1 + \frac{1}{v_n} = 1 + \frac{6}{3 + 2n}$. Como $v_n \to +\infty$,
$u_n \to 1$.
\end{solution}

\begin{solution}{exo:g12:seq:9}
\emph{1.} $H_{2n} - H_n = \sum_{k=n+1}^{2n} \frac{1}{k}$ es una suma de
$n$ términos, cada uno al menos $\frac{1}{2n}$; por tanto,
$H_{2n} - H_n \geq n \cdot \frac{1}{2n} = \frac12$.

\emph{2.} Por inducción sobre $k$: $H_{2^0} = H_1 = 1 \geq 1$. Si
$H_{2^k} \geq 1 + \frac{k}{2}$, aplicando el punto 1 con $n = 2^k$,
\[
H_{2^{k+1}} \geq H_{2^k} + \frac12 \geq 1 + \frac{k+1}{2}.
\]
La \omterm{def:g12:seq:sequence}{sucesión} $(H_n)$ es \omterm{def:g12:seq:monotonic}{creciente} (cada paso añade $\frac{1}{n+1} > 0$) y
la subsucesión $H_{2^k}$ no está \omterm{def:g12:seq:bounded}{acotada}, así que $(H_n)$ no está \omterm{def:g12:seq:bounded}{acotada
superiormente}. \omterm{def:g12:seq:monotonic}{Creciente} y no \omterm{def:g12:seq:bounded}{acotada}, tiende a $+\infty$
(\cref{thm:g12:seq:monotone}).
\end{solution}

\begin{solution}{exo:g12:seq:10}
\emph{1.} La \omterm{def:g12:seq:sequence}{sucesión} $d_n = b_n - a_n$ cumple
$d_{n+1} - d_n = (b_{n+1} - b_n) - (a_{n+1} - a_n) \leq 0$, luego $(d_n)$
es \omterm{def:g10:functions:variations}{decreciente}; y como $d_n \to 0$, se obtiene $d_n \geq 0$ para todo $n$
(una \omterm{def:g12:seq:sequence}{sucesión} \omterm{def:g10:functions:variations}{decreciente} con un término negativo se quedaría por debajo
de él para siempre, lo que impediría el límite $0$). Por tanto,
$a_n \leq b_n$.

\emph{2.} De $a_n \leq b_n \leq b_0$, la \omterm{def:g12:seq:sequence}{sucesión} \omterm{def:g12:seq:monotonic}{creciente} $(a_n)$ está
\omterm{def:g12:seq:bounded}{acotada superiormente}, luego \omterm{def:g12:seq:limit}{converge} a cierto $\ell$. Análogamente,
$(b_n)$, \omterm{def:g10:functions:variations}{decreciente} y \omterm{def:g12:seq:bounded}{acotada inferiormente} por $a_0$, \omterm{def:g12:seq:limit}{converge} a cierto
$\ell'$. Entonces $\ell' - \ell = \lim (b_n - a_n) = 0$, luego
$\ell = \ell'$.

\emph{3.} $(a_n)$ es (estrictamente) \omterm{def:g12:seq:monotonic}{creciente}, ya que
$a_{n+1} - a_n = \frac{1}{(n+1)!} > 0$. Para $(b_n)$,
\[
b_{n+1} - b_n = \frac{1}{(n+1)!} + \frac{1}{(n+1)(n+1)!} - \frac{1}{n\,n!}
= \frac{n(n+1) + n - (n+1)^2}{n(n+1)(n+1)!}
= \frac{-1}{n(n+1)(n+1)!} < 0 ,
\]
así que $(b_n)$ es \omterm{def:g10:functions:variations}{decreciente}. Por último,
$b_n - a_n = \frac{1}{n\,n!} \to 0$. Las dos sucesiones son adyacentes y,
por tanto, convergen a un límite común.
\end{solution}

\begin{solution}{pb:g12:seq:1}
\textbf{1.} Cierto para $n = 1$ ($1 = 1^2$). Si
$1 + 3 + \dots + (2n - 1) = n^2$, al añadir el \omterm{def:g11:func:parity}{impar} siguiente:
$n^2 + (2n + 1) = (n + 1)^2$: paso de inducción hecho. Por inducción, es
cierto para todo $n \geq 1$.

\textbf{2.} $2^0 = 1 > 0$. Si $2^n > n$, entonces
$2^{n+1} = 2 \cdot 2^n > 2n \geq n + 1$ para $n \geq 1$ (y $n = 0$ se
comprueba directamente): paso hecho.

\textbf{3.} $n = 0$: $1 \geq 1$. Si $(1 + x)^n \geq 1 + nx$, multiplicamos
por $1 + x \geq 1 > 0$:
$(1 + x)^{n+1} \geq (1 + nx)(1 + x) = 1 + (n + 1)x + nx^2
\geq 1 + (n + 1)x$.

\textbf{4.} El paso de $n = 1$ a $n = 2$: entre dos canicas, ``las $n$
primeras'' y ``las $n$ últimas'' son dos canicas sueltas y
\emph{disjuntas}; ninguna canica común tiende un puente entre los dos
grupos, así que nada obliga a que sus colores coincidan. El argumento del
paso de inducción exige en silencio que los dos grupos se solapen, cosa
que solo ocurre a partir de $n \geq 2$; y con el caso base $n = 1$ la
cadena no arranca nunca.

\textbf{5.} $4^0 - 1 = 0 = 3 \times 0$. Si $4^n - 1 = 3k$, entonces
$4^{n+1} - 1 = 4(4^n - 1) + 3 = 3(4k + 1)$: paso hecho.

\textbf{6.} $x_1 = \frac32$, $x_2 = \frac{17}{12}$,
$x_3 = \frac{577}{408}$.

\textbf{7.} $x_{n+1}^2 - 2 = \frac{(x_n^2 + 2)^2 - 8x_n^2}{4x_n^2} =
\frac{(x_n^2 - 2)^2}{4 x_n^2}$: un cuadrado dividido entre algo positivo,
luego $\geq 0$, y $> 0$ siempre que $x_n^2 \neq 2$. Inducción:
$x_0 = 2 > 0$ con $x_0^2 = 4 > 2$; si $x_n > 0$ y $x_n^2 > 2$, entonces
$x_{n+1}$ (\omterm{def:g11:stat:mean}{media} de positivos) es positivo y $x_{n+1}^2 - 2 > 0$.

\textbf{8.} $x_{n+1} - x_n = \frac{2 - x_n^2}{2x_n} < 0$ por la pregunta
7: estrictamente \omterm{def:g10:functions:variations}{decreciente}.

\textbf{9.} \omterm{def:g10:functions:variations}{Decreciente} y \omterm{def:g12:seq:bounded}{acotada inferiormente} (por $1$, ya que
$x_n^2 > 2 > 1$ y $x_n > 0$): por el teorema de la convergencia \omterm{def:g12:seq:monotonic}{monótona},
$(x_n)$ \omterm{def:g12:seq:limit}{converge} a cierto $L \geq 1$.

\textbf{10.} Los límites respetan el álgebra: de
$x_{n+1} = \frac12\left(x_n + \frac{2}{x_n}\right)$ y
$x_n \to L \geq 1 > 0$ se obtiene $L = \frac12\left(L + \frac2L\right)$,
luego $L^2 = 2$ y, al ser $L$ positivo, $L = \sqrt2$. Veredicto:
convergencia demostrada y límite identificado; Herón, absuelto con
honores.

\textbf{11.} $x_{n+1} - \sqrt2 = \frac{x_n^2 - 2\sqrt2\,x_n + 2}{2x_n} =
\frac{(x_n - \sqrt2)^2}{2x_n}$: exactamente
$e_{n+1} = \frac{e_n^2}{2x_n}$, y $x_n > \sqrt2$ da
$e_{n+1} \leq \frac{e_n^2}{2\sqrt2}$. Error al cuadrado: cada paso duplica
el número de decimales correctos, tal como se venía observando desde
cursos atrás.

\textbf{12.} $e_0 \approx 0.5858$, $e_1 \approx 0.0858$,
$e_2 \approx 0.00245$, $e_3 \approx 2.1 \times 10^{-6}$. Cocientes:
$\frac{e_1}{e_0^2} \approx 0.25 = \frac{1}{2x_0}$;
$\frac{e_2}{e_1^2} \approx 0.333 = \frac{1}{2x_1}$;
$\frac{e_3}{e_2^2} \approx 0.353 \approx \frac{1}{2x_2}$: el teorema en
acción.

\textbf{13.} $H_{2^{18}} \geq 1 + 9 = 10$: hacen falta unos
$260\,000$ términos ($2^{18} = 262\,144$) solo para pasar de $10$;
divergencia a paso de tortuga (y $H_n > 100$ necesitaría más términos que
átomos hay en cualquier biblioteca).

\textbf{14.} $S_n = 2 - \frac{1}{2^n}$ (suma geométrica) y
$\frac{1}{2^n} \to 0$ (\cref{thm:g12:seq:geometric}), luego $S_n \to 2$:
la tableta de chocolate mordida sin fin tiende a la tableta entera sin
alcanzarla nunca, ahora en el lenguaje oficial de los límites.

\textbf{15.} Para $k \geq 2$:
$\frac{1}{k^2} \leq \frac{1}{k(k-1)} = \frac{1}{k-1} - \frac1k$, luego
$S_n \leq 1 + \left(1 - \frac1n\right) < 2$: \omterm{def:g12:seq:monotonic}{creciente} y \omterm{def:g12:seq:bounded}{acotada
superiormente} y, por tanto, convergente (convergencia \omterm{def:g12:seq:monotonic}{monótona}). Euler le
puso nombre más tarde al límite: $\frac{\pi^2}{6}$.

\textbf{16.} Que los términos tiendan a $0$ es \emph{necesario} para que
las sumas se estabilicen, pero no decide nada: los términos armónicos
$\frac1n \to 0$ y, sin embargo, las sumas se disparan; los términos
$\frac{1}{n^2} \to 0$ y las sumas convergen. Toda la cuestión está en lo
\emph{deprisa} que se apagan los términos; esa es la teoría de las series,
que se construye en los volúmenes universitarios.

\textbf{17.} $a_1 = \sqrt2 \approx 1.41421$, $b_1 = 1.5$;
$a_2 \approx 1.45648$, $b_2 \approx 1.45711$: con dos iteraciones ya
coinciden en tres decimales; una velocidad asombrosa.

\textbf{18.} $b_{n+1} - a_{n+1} = \frac{a_n + b_n}{2} - \sqrt{a_n b_n} =
\frac{(\sqrt{b_n} - \sqrt{a_n})^2}{2} \geq 0$: las \omterm{def:g11:stat:mean}{medias} mantienen el
orden. $(a_n)$ crece:
$a_{n+1} = \sqrt{a_n b_n} \geq \sqrt{a_n \cdot a_n} = a_n$; y $(b_n)$
decrece por simetría.

\textbf{19.} $\frac{b_{n+1} - a_{n+1}}{b_n - a_n} =
\frac{(\sqrt{b_n} - \sqrt{a_n})^2}
{2(\sqrt{b_n} - \sqrt{a_n})(\sqrt{b_n} + \sqrt{a_n})}
= \frac{\sqrt{b_n} - \sqrt{a_n}}{2(\sqrt{b_n} + \sqrt{a_n})}
\leq \frac12$: la separación se reduce al menos a la mitad, luego
$b_n - a_n \to 0$; junto con la pregunta 18, las sucesiones son adyacentes
y comparten un límite $M(1, 2)$.

\textbf{20.} La tercera iteración da
$a_3 \approx b_3 \approx 1.456791$: $M(1, 2) \approx 1.456791$ en tres
vueltas de manivela (la separación se \emph{eleva} aproximadamente
\emph{al cuadrado}, como en Herón). Clasificación de velocidades de las
sucesiones de este problema, de la más lenta a la más rápida: las sumas
armónicas (divergencia glacial), las sumas geométricas (error reducido a
la mitad en cada paso) y Herón junto con la \omterm{def:g11:stat:mean}{media} aritmético-geométrica
(error elevado al cuadrado en cada paso). Y fue la velocidad sobrenatural
de esa \omterm{def:g11:stat:mean}{media} la que le dijo a Gauss que había dado con un filón nuevo del
análisis.
\end{solution}
