\section*{Capítulo \ref{ch:g12:limcont} --- Límites y continuidad}

\begin{solution}{exo:g12:limcont:1}
Dividiendo entre $x^3$:
$\dfrac{2x^3 - x + 1}{5x^3 + x^2} = \dfrac{2 - 1/x^2 + 1/x^3}{5 + 1/x}
\to \dfrac{2}{5}$.

$x^2 + 3x - 1 = x^2(1 + 3/x - 1/x^2) \to +\infty$ cuando $x \to -\infty$,
porque $x^2 \to +\infty$ y el paréntesis tiende a $1$.

Cuando $x \to 2^+$, el numerador tiende a $3 > 0$ y el denominador a
$0^+$, luego $\dfrac{x+1}{x-2} \to +\infty$.

Para $x > 0$, $-\dfrac1x \leq \dfrac{\cos x}{x} \leq \dfrac1x$, así que el
límite es $0$ por el teorema del sándwich.
\end{solution}

\begin{solution}{exo:g12:limcont:2}
\emph{1.} $2x^2 - 3x + 1 = (x-1)(2x-1)$ (la \omterm{def:g11:quad:discriminant}{raíz} $x=1$ se ve a simple
vista, porque $2 - 3 + 1 = 0$). Para $x \neq 1$, simplificamos $x - 1$:
$f(x) = 2x - 1$.

\emph{2.} Por tanto, $\lim_{x\to1} f(x) = 1$. Definiendo $f(1) = 1$ se
prolonga $f$ a la \omterm{def:g11:func:function}{función} polinómica $x \mapsto 2x-1$ sobre $\R$, que es
\omterm{def:g12:limcont:continuity}{continua} en $1$.
\end{solution}

\begin{solution}{exo:g12:limcont:3}
Escribimos $f(x) = \dfrac{3x - 1}{x + 2} = \dfrac{3(x+2) - 7}{x+2}
= 3 - \dfrac{7}{x+2}$. Cuando $x \to \pm\infty$, $f(x) \to 3$: la recta
$y = 3$ es \omterm{def:g12:limcont:asymptote}{asíntota horizontal} en los dos infinitos. Cuando
$x \to (-2)^\pm$, $f(x) \to \mp\infty$: la recta $x = -2$ es \omterm{def:g12:limcont:asymptote}{asíntota
vertical}. La curva es una \omterm{def:g11:func:reference}{hipérbola}, por debajo de $y=3$ para $x > -2$ y
por encima para $x < -2$.
\end{solution}

\begin{solution}{exo:g12:limcont:4}
Multiplicamos por el conjugado:
\[
\sqrt{x^2 + x} - x = \frac{x^2 + x - x^2}{\sqrt{x^2+x} + x}
= \frac{x}{x\left(\sqrt{1 + 1/x} + 1\right)}
= \frac{1}{\sqrt{1 + 1/x} + 1}
\xrightarrow[x\to+\infty]{} \frac{1}{2}.
\]
Del mismo modo,
\[
\frac{\sqrt{1+x} - 1}{x} = \frac{(1+x) - 1}{x\left(\sqrt{1+x} + 1\right)}
= \frac{1}{\sqrt{1+x} + 1} \xrightarrow[x\to0]{} \frac{1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:limcont:5}
\emph{1.} Como $\sin x \geq -1$, $f(x) \geq x - 1 \to +\infty$, y
concluimos por comparación. Sin embargo, $f'(x) = 1 + \cos x$ se anula en
todos los múltiplos \omterm{def:g11:func:parity}{impares} de $\pi$, y en cada \omterm{def:g10:numbers:interval}{intervalo}
$\intco{A}{+\infty}$ la \omterm{def:g11:func:function}{función} alterna tramos de crecimiento estricto: es
\omterm{def:g12:seq:monotonic}{creciente}, pero no estrictamente y, sobre todo, ese crecimiento no hace
falta: basta con la comparación.

\emph{2.} Para $x > 1$, dividimos entre $x$:
\[
g(x) = \frac{1 + \frac{\sin x}{x}}{1 - \frac{\sin x}{x}},
\]
y $\frac{\sin x}{x} \to 0$ por el teorema del sándwich
($\abs{\sin x} \leq 1$). Por tanto, $g(x) \to \frac{1+0}{1-0} = 1$.
\end{solution}

\begin{solution}{exo:g12:limcont:6}
$f(x) = x^3 - 3x + 1$ es \omterm{def:g12:limcont:continuity}{continua}, con
$f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$: $f$ crece en $\intoc{-\infty}{-1}$,
decrece en $\intcc{-1}{1}$ y crece en $\intco{1}{+\infty}$, con \omterm{def:g10:functions:extrema}{máximo}
relativo $f(-1) = 3 > 0$ y \omterm{def:g10:functions:extrema}{mínimo} relativo $f(1) = -1 < 0$,
$\lim_{-\infty} f = -\infty$ y $\lim_{+\infty} f = +\infty$.

En cada uno de los tres \omterm{def:g10:numbers:interval}{intervalos} de monotonía, $f$ es \omterm{def:g12:limcont:continuity}{continua} y
estrictamente \omterm{def:g12:seq:monotonic}{monótona}, y el $0$ está estrictamente entre los valores o
límites de los extremos, así que el teorema de la biyección da exactamente
un cero por \omterm{def:g10:numbers:interval}{intervalo}: tres ceros en total. Comprobación de signos:
$f(-2) = -1 < 0 < f(-1) = 3$, luego hay un cero en $\intoo{-2}{-1}$;
$f(0) = 1 > 0 > f(1) = -1$, un cero en $\intoo{0}{1}$; y
$f(1) = -1 < 0 < f(2) = 3$, un cero en $\intoo{1}{2}$.
\end{solution}

\begin{solution}{exo:g12:limcont:7}
Sea $g(x) = f(x) - x$, \omterm{def:g12:limcont:continuity}{continua} en $\intcc{0}{1}$. Como
$f(0) \in \intcc{0}{1}$, $g(0) = f(0) \geq 0$; y como
$f(1) \in \intcc{0}{1}$, $g(1) = f(1) - 1 \leq 0$. Por el teorema de los
valores intermedios, $g$ se anula en algún $c \in \intcc{0}{1}$, es decir,
$f(c) = c$.
\end{solution}

\begin{solution}{exo:g12:limcont:8}
Sean $u(t)$ y $d(t)$ las posiciones (medidas como distancia desde el
comienzo del camino) en el instante $t \in \intcc{8}{20}$ en la subida y
en la bajada. Las dos son \omterm{def:g12:limcont:continuity}{continuas}, con $u(8) = 0$, $u(20) = L$ (la
longitud del camino), $d(8) = L$ y $d(20) = 0$. La \omterm{def:g11:func:function}{función} $g = u - d$ es
\omterm{def:g12:limcont:continuity}{continua} con $g(8) = -L \leq 0$ y $g(20) = L \geq 0$, así que $g(c) = 0$
para algún instante $c$ por el teorema de los valores intermedios: en el
instante $c$, las posiciones de los dos días coinciden.
\end{solution}

\begin{solution}{exo:g12:limcont:9}
\emph{1.} $f'(x) = 5x^4 + 3x^2 + 1 > 0$, así que $f$ es estrictamente
\omterm{def:g12:seq:monotonic}{creciente} en $\R$; con $\lim_{\pm\infty} f = \pm\infty$ y la \omterm{def:g12:limcont:continuity}{continuidad},
el teorema de la biyección da un único cero real $\alpha$. Como
$f(0) = -1 < 0$ y $f(1) = 2 > 0$, $\alpha \in \intoo{0}{1}$.

\emph{2.} Mantenemos un \omterm{def:g10:numbers:interval}{intervalo} $\intcc{a_n}{b_n}$ con
$f(a_n) < 0 < f(b_n)$, partiendo de $\intcc{0}{1}$; en cada paso evaluamos
$f$ en el \omterm{prop:g10:coordgeom:midpoint}{punto medio} $m$ y conservamos la mitad en la que persiste el
cambio de signo. Tras $n$ pasos, el \omterm{def:g10:numbers:interval}{intervalo} mide $2^{-n}$, así que su
\omterm{prop:g10:coordgeom:midpoint}{punto medio} aproxima $\alpha$ con error menor que $2^{-(n+1)}$.
Necesitamos $2^{-(n+1)} \leq 10^{-6}$, es decir,
$n + 1 \geq \frac{6}{\log_{10} 2} \approx 19.93$: bastan $n = 20$
iteraciones.
\end{solution}

\begin{solution}{pb:g12:limcont:1}
\textbf{1.} Términos dominantes: $\frac{3x^2}{x^2} \to 3$;
$\frac{2x}{x^2} = \frac2x \to 0$;
$x^3 - x^2 = x^3\left(1 - \frac1x\right) \to +\infty$.

\textbf{2.} $\lim_{x \to \pm\infty} f = 3$: \omterm{def:g12:limcont:asymptote}{asíntota horizontal} $y = 3$;
$\lim_{x \to (-2)^\pm} f = \mp\infty$: \omterm{def:g12:limcont:asymptote}{asíntota vertical} $x = -2$.

\textbf{3.} $\frac{(x-3)(x+3)}{x - 3} = x + 3 \to 6$. Y
$\frac{\sqrt{x+1} - 1}{x} = \frac{x}{x(\sqrt{x+1} + 1)} =
\frac{1}{\sqrt{x+1} + 1} \to \frac12$.

\textbf{4.} $\abs{x \sin\frac1x} \leq \abs x \to 0$, luego el límite es
$0$. Y $\abs{\frac{\sin x}{x}} \leq \frac1x \to 0$: límite $0$.

\textbf{5.} $\frac{\sqrt{4x^2 + x}}{x} = \sqrt{4 + \frac1x} \to
\sqrt4 = 2$ (\omterm{def:g12:limcont:continuity}{continuidad} de la \omterm{def:g11:quad:discriminant}{raíz} cuadrada, por composición).

\textbf{6.} $f(x) = x^3 + x$ es \omterm{def:g12:limcont:continuity}{continua} con
$f(1) = 2 < 5 < 10 = f(2)$: por el teorema de los valores intermedios,
$f(c) = 5$ para algún $c \in \intoo{1}{2}$.

\textbf{7.} $f'(x) = 3x^2 + 1 > 0$: estrictamente \omterm{def:g12:seq:monotonic}{creciente} en $\R$,
\omterm{def:g12:limcont:continuity}{continua} y con límites $\mp\infty$: por el teorema de la biyección, todo
valor (en particular, el $5$) se alcanza exactamente una vez.

\textbf{8.} $p(x) = x^3\left(1 + \frac ax + \frac{b}{x^2} +
\frac{c}{x^3}\right)$: cuando $x \to +\infty$, $p \to +\infty$; y cuando
$x \to -\infty$, $p \to -\infty$. Así que en algún \omterm{def:g10:numbers:interval}{intervalo} enorme
$\intcc{-M}{M}$ se cumple $p(-M) < 0 < p(M)$: el teorema de los valores
intermedios entrega una \omterm{def:g11:quad:discriminant}{raíz}. El mismo argumento del término dominante
vale literalmente para cualquier grado \omterm{def:g11:func:parity}{impar}: las potencias \omterm{def:g11:func:parity}{impares}
conservan signos opuestos en los dos infinitos.

\textbf{9.} Partir por la mitad $n$ veces deja un \omterm{def:g10:numbers:interval}{intervalo} de longitud
$2^{-n}$; y $2^{-20} \approx 9.5 \times 10^{-7} < 10^{-6}$: veinte pasos.
Newton duplica las cifras correctas en cada paso: partiendo de una cifra
correcta, unos $3$ o $4$ pasos llegan a seis; la tortuga y el galgo de la
búsqueda de raíces.

\textbf{10.} $g$ no es \omterm{def:g12:limcont:continuity}{continua} en $\intcc{-1}{1}$: explota en $0$, que
está dentro del \omterm{def:g10:numbers:interval}{intervalo}. La conclusión del teorema se evapora junto con
su hipótesis: comprueba siempre la \omterm{def:g12:limcont:continuity}{continuidad} \emph{en todo el
\omterm{def:g10:numbers:interval}{intervalo}}, no solo en los extremos.

\textbf{11.} $g(x) = f(x) - x$ es \omterm{def:g12:limcont:continuity}{continua}; $g(0) = f(0) \geq 0$ (los
valores están en $\intcc{0}{1}$) y $g(1) = f(1) - 1 \leq 0$. Por el
teorema de los valores intermedios, $g(c) = 0$ para algún $c$:
$f(c) = c$.

\textbf{12.} Idealizando el país como el segmento $\intcc{0}{1}$: $f$
lleva cada lugar al lugar que aparece justo debajo en el mapa arrugado;
arrugar mantiene el mapa sobre el país, así que $f$ aterriza en
$\intcc{0}{1}$, y es \omterm{def:g12:limcont:continuity}{continua} (el papel no se rompe). El \omterm{pb:g10:functions:1}{punto fijo} $c$ es
el punto que queda precisamente encima de su propia \omterm{def:g10:functions:function}{imagen}; el teorema de
Brouwer se ocupa del mapa honrado, en dos dimensiones.

\textbf{13.} $g(0) + g(\pi) = \left(T(0) - T(\pi)\right) +
\left(T(\pi) - T(2\pi)\right) = T(0) - T(2\pi) = 0$. Así que $g(0)$ y
$g(\pi)$ son opuestos (o los dos nulos): en cualquier caso, $g$, que es
\omterm{def:g12:limcont:continuity}{continua}, se anula en algún $t^* \in \intcc{0}{\pi}$:
$T(t^*) = T(t^* + \pi)$; en algún lugar del ecuador, dos puntos antípodas
coinciden en temperatura, en todos y cada uno de los instantes.

\textbf{14.} Girar la mesa un cuarto de vuelta intercambia los papeles de
las dos diagonales: la pareja de patas que apretaba el suelo y la pareja
en la que una pata quedaba en el aire cambian de situación, así que la
altura de la pata suelta (contada con signo, como ``holgura de la diagonal
distinguida'') toma signos opuestos en $\theta$ y en
$\theta + \frac\pi2$. Un suelo \omterm{def:g12:limcont:continuity}{continuo} hace \omterm{def:g12:limcont:continuity}{continua} a $h$, y el teorema
de los valores intermedios entrega un ángulo $\theta^*$ con
$h(\theta^*) = 0$: las cuatro patas en el suelo. (Con una mesa rectangular
o con un suelo escalonado no hay tanta suerte.)

\textbf{15.} Plantilla: construye la \emph{\omterm{def:g11:seq:arithmetic}{diferencia}} de las dos
magnitudes que quieres hacer coincidir ($f(x) - x$; $T(t) - T(t + \pi)$;
posición de subida menos posición de bajada para la senderista; la holgura
con signo para la mesa); comprueba que toma valores de los dos signos (a
menudo mediante una simetría entre los extremos); e invoca el teorema de
los valores intermedios sobre una \omterm{def:g11:func:function}{función} \omterm{def:g12:limcont:continuity}{continua}. La coincidencia queda
forzada.

\textbf{16.} Preguntas abiertas: \emph{cuántas} soluciones hay (la
responden la monotonía estricta y el teorema de la biyección, pregunta 7)
y \emph{dónde} están (lo responden la \omterm{thm:g12:limcont:ivt}{dicotomía} o Newton, pregunta 9). El
teorema de los valores intermedios es un oráculo de pura existencia.

\textbf{17.} $3 - \frac{7}{x + 2}$: \omterm{def:g12:limcont:asymptote}{asíntota vertical} $x = -2$ y
horizontal $y = 3$; a cada lado de $-2$ la \omterm{def:g11:func:function}{función} es \omterm{def:g12:seq:monotonic}{creciente}
($-\frac{7}{x+2}$ crece donde está definida); la curva es la \omterm{def:g11:func:reference}{hipérbola} de
referencia $-\frac7x$ desplazada $2$ a la izquierda y $3$ hacia arriba:
dos ramas abrazadas al cruce de las \omterm{def:g12:limcont:asymptote}{asíntotas}.

\textbf{18.} Para $x \neq 1$: $f(x) = x + 1$, luego
$\lim_{x \to 1} f = 2$: definiendo $f(1) = 2$ se restaura la \omterm{def:g12:limcont:continuity}{continuidad}.
La discontinuidad era un agujero, no un salto: se rellena el único punto
que falta y la curva cicatriza; es \emph{evitable}.

\textbf{19.} Es \omterm{def:g12:limcont:continuity}{continua} en cada \omterm{def:g10:numbers:interval}{intervalo} entre dos enteros consecutivos;
y en cada entero $n$ salta de $n - 1$ a $n$. Con $a = 0.9$ y $b = 1.1$:
$\lfloor a \rfloor = 0 < \frac12 < 1 = \lfloor b \rfloor$ y, sin embargo,
$\lfloor x \rfloor = \frac12$ no tiene solución: el salto se teletransporta
por encima de $\frac12$, que es precisamente lo que el teorema de los
valores intermedios prohíbe a las funciones \omterm{def:g12:limcont:continuity}{continuas}.

\textbf{20.} Cara local: valor $=$ límite en cada punto; el agujero
evitable (pregunta 18) la violaba con remedio y los saltos de la parte
entera (pregunta 19), sin remedio. Cara global: en un \omterm{def:g10:numbers:interval}{intervalo} se visitan
todos los valores intermedios; ahí viven los certificados de raíces
(preguntas 6 a 8), el \omterm{pb:g10:functions:1}{punto fijo}, el ecuador, la mesa y la senderista. El
nombre oficial del eslogan: el \emph{teorema de los valores intermedios}.
\end{solution}
