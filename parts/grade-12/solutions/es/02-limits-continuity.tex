\section*{Capítulo \ref{ch:g12:limcont} --- Límites y continuidad}

\begin{solution}{exo:g12:limcont:1}
Dividiendo por $x^3$:
$\dfrac{2x^3 - x + 1}{5x^3 + x^2} = \dfrac{2 - 1/x^2 + 1/x^3}{5 + 1/x}
\to \dfrac{2}{5}$.

$ x^2 + 3x - 1 = x^2(1 + 3/x - 1/x^2) \to +\infty $ como $ x \to -\infty $, desde
$ x^2 \to +\infty $ y el soporte tiende a $1$.

Como $ x \to 2^+$, el numerador tiende a $3 > 0$ y el denominador a $0^+$,
entonces $\dfrac{x+1}{x-2} \to +\infty $.

Para $ x > 0$, $-\dfrac1x \leq \dfrac{\cos x}{x} \leq \dfrac1x $, entonces el límite
es $0$ por el teorema de compresión.
\end{solution}

\begin{solution}{exo:g12:limcont:2}
\emph{1.} $2x^2 - 3x + 1 = (x-1)(2x-1)$ (el \omterm{def:g11:quad:discriminant}{raíz} $ x=1$ es evidente a partir de
$2 - 3 + 1 = 0$). Para $ x \neq 1$, Cancelar $ x - 1$: $ f(x) = 2x - 1$.

\emph{2.} Por eso $\lim_{x\to1} f(x) = 1$. Configuración $ f(1) = 1$ se extiende $ f $ a
el polinomio \omterm{def:g11:func:function}{función} $ x \mapsto 2x-1$ en $\R $, que es continua en $1$.
\end{solution}

\begin{solution}{exo:g12:limcont:3}
Escribir $ f(x) = \dfrac{3x - 1}{x + 2} = \dfrac{3(x+2) - 7}{x+2}
= 3 - \dfrac{7}{x+2}$. Como $ x \to \pm\infty $, $ f(x) \to 3$: la línea $ y = 3$
es un \omterm{def:g12:limcont:asymptote}{asíntota horizontal} en ambos infinitos. Como $ x \to (-2)^\pm $,
$ f(x) \to \mp\infty $: la línea $ x = -2$ es un \omterm{def:g12:limcont:asymptote}{asíntota vertical}. la curva es
un \omterm{def:g11:func:reference}{hipérbola}, abajo $ y=3$ para $ x > -2$ y encima para $ x < -2$.
\end{solution}

\begin{solution}{exo:g12:limcont:4}
Multiplicar por el conjugado:
\[
\sqrt{x^2 + x} - x = \frac{x^2 + x - x^2}{\sqrt{x^2+x} + x}
= \frac{x}{x\left(\sqrt{1 + 1/x} + 1\right)}
= \frac{1}{\sqrt{1 + 1/x} + 1}
\xrightarrow[x\to+\infty]{} \frac{1}{2}.
\]
Similarmente,
\[
\frac{\sqrt{1+x} - 1}{x} = \frac{(1+x) - 1}{x\left(\sqrt{1+x} + 1\right)}
= \frac{1}{\sqrt{1+x} + 1} \xrightarrow[x\to0]{} \frac{1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:limcont:5}
\emph{1.} Desde $\sin x \geq -1$, $ f(x) \geq x - 1 \to +\infty $, y nosotros
concluir por comparación. Todavía $ f'(x) = 1 + \cos x $ se desvanece en cada momento
múltiplo de $\pi $, y en cada \omterm{def:g10:numbers:interval}{intervalo} $\intco{A}{+\infty}$ la \omterm{def:g11:func:function}{función}
alterna entre \omterm{def:g10:numbers:interval}{intervalos} de estricto aumento; es \omterm{def:g12:seq:monotonic}{creciente} pero no
estrictamente, y lo que es más importante, su crecimiento no es necesario: basta con comparar.

\emph{2.} Para $ x > 1$, dividir por $ x $:
\[
g(x) = \frac{1 + \frac{\sin x}{x}}{1 - \frac{\sin x}{x}},
\]
y $\frac{\sin x}{x} \to 0$ por el teorema de compresión
($\abs{\sin x} \leq 1$). Por eso $ g(x) \to \frac{1+0}{1-0} = 1$.
\end{solution}

\begin{solution}{exo:g12:limcont:6}
$ f(x) = x^3 - 3x + 1$ es continua, con $ f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$:
$ f $ aumenta en $\intoc{-\infty}{-1}$, disminuye en $\intcc{-1}{1}$,
aumenta en $\intco{1}{+\infty}$, con locales \omterm{def:g10:functions:extrema}{máximo} $ f(-1) = 3 > 0$ y
locales \omterm{def:g10:functions:extrema}{mínimo} $ f(1) = -1 < 0$, $\lim_{-\infty} f = -\infty $,
$\lim_{+\infty} f = +\infty $.

En cada una de las tres monotonías \omterm{def:g10:numbers:interval}{intervalos}, $ f $ es continua, estrictamente
monótona, y $0$ se encuentra estrictamente entre los valores/límites del punto final, por lo que el
El teorema de biyección da exactamente un cero por \omterm{def:g10:numbers:interval}{intervalo}: tres ceros en total.
Firmar cheques: $ f(-2) = -1 < 0 < f(-1) = 3$, entonces un cero en
$\intoo{-2}{-1}$; $ f(0) = 1 > 0 > f(1) = -1$, un cero en $\intoo{0}{1}$;
$ f(1) = -1 < 0 < f(2) = 3$, un cero en $\intoo{1}{2}$.
\end{solution}

\begin{solution}{exo:g12:limcont:7}
Sea $ g(x) = f(x) - x $, continua en $\intcc{0}{1}$. Desde
$ f(0) \in \intcc{0}{1}$, $ g(0) = f(0) \geq 0$; desde $ f(1) \in \intcc{0}{1}$,
$ g(1) = f(1) - 1 \leq 0$. Según el teorema del valor intermedio, $ g $ desaparece en
algunos $ c \in \intcc{0}{1}$, es decir.\ $ f(c) = c $.
\end{solution}

\begin{solution}{exo:g12:limcont:8}
Sea $ u(t)$ y $ d(t)$ ser las posiciones (medidas como distancia desde la parte inferior
del sendero) en el momento $ t \in \intcc{8}{20}$ en el camino hacia arriba y en el camino
abajo. Ambos son continua, $ u(8) = 0$, $ u(20) = L $ (la longitud del sendero),
$ d(8) = L $, $ d(20) = 0$. La \omterm{def:g11:func:function}{función} $ g = u - d $ es continua con
$ g(8) = -L \leq 0$ y $ g(20) = L \geq 0$, entonces $ g(c) = 0$ por algún tiempo $ c $ por
el teorema del valor intermedio: en el tiempo $ c $, las posiciones de ambos días coinciden.
\end{solution}

\begin{solution}{exo:g12:limcont:9}
\emph{1.} $ f'(x) = 5x^4 + 3x^2 + 1 > 0$, entonces $ f $ es estrictamente \omterm{def:g12:seq:monotonic}{creciente} en
$\R $; con $\lim_{\pm\infty} f = \pm\infty $ y continuidad, la biyección
teorema da un cero real único $\alpha $. Desde $ f(0) = -1 < 0$ y
$ f(1) = 2 > 0$, $\alpha \in \intoo{0}{1}$.

\emph{2.} Mantener un \omterm{def:g10:numbers:interval}{intervalo} $\intcc{a_n}{b_n}$ con
$ f(a_n) < 0 < f(b_n)$, a partir de $\intcc{0}{1}$; en cada paso evaluar
$ f $ al \omterm{prop:g10:coordgeom:midpoint}{punto medio} $ m $ y mantener el \omterm{def:g10:numbers:interval}{intervalo} medio en el que el signo cambia
persiste. Después $ n $ pasos el \omterm{def:g10:numbers:interval}{intervalo} tiene longitud $2^{-n}$, entonces el \omterm{prop:g10:coordgeom:midpoint}{punto medio}
se aproxima $\alpha $ hacia dentro $2^{-(n+1)}$. necesitamos
$2^{-(n+1)} \leq 10^{-6}$, es decir.\ $ n + 1 \geq \frac{6}{\log_{10} 2} \approx 19.93$:
$ n = 20$ las iteraciones son suficientes.
\end{solution}

\begin{solution}{pb:g12:limcont:1}
\textbf{1.} Términos dominantes: $\frac{3x^2}{x^2} \to 3$;
$\frac{2x}{x^2} = \frac2x \to 0$;
$ x^3 - x^2 = x^3\left(1 - \frac1x\right) \to +\infty $.

\textbf{2.} $\lim_{x \to \pm\infty} f = 3$: horizontales
asíntota $ y = 3$; $\lim_{x \to (-2)^\pm} f = \mp\infty $:
\omterm{def:g12:limcont:asymptote}{asíntota vertical} $ x = -2$.

\textbf{3.} $\frac{(x-3)(x+3)}{x - 3} = x + 3 \to 6$. Y
$\frac{\sqrt{x+1} - 1}{x} = \frac{x}{x(\sqrt{x+1} + 1)} =
\frac{1}{\sqrt{x+1} + 1} \to \frac12$.

\textbf{4.} $\abs{x \sin\frac1x} \leq \abs x \to 0$, entonces el
el límite es $0$. Y
$\abs{\frac{\sin x}{x}} \leq \frac1x \to 0$: límite $0$.

\textbf{5.} $\frac{\sqrt{4x^2 + x}}{x} =
\sqrt{4 + \frac1x} \to \sqrt4 = 2$ (continuidad de la plaza
\omterm{def:g11:quad:discriminant}{raíz}, composición).

\textbf{6.} $ f(x) = x^3 + x $ es continua con
$ f(1) = 2 < 5 < 10 = f(2)$: por el IVT, $ f(c) = 5$ para algunos
$ c \in \intoo{1}{2}$.

\textbf{7.} $ f'(x) = 3x^2 + 1 > 0$: estrictamente \omterm{def:g12:seq:monotonic}{creciente} en
$\R $, continua, con límites $\mp\infty $: por la biyección
teorema, cada valor --- en particular $5$ --- se logra
exactamente una vez.

\textbf{8.} $ p(x) = x^3\left(1 + \frac ax + \frac{b}{x^2} +
\frac{c}{x^3}\right)$: como $ x \to +\infty $, $ p \to +\infty $; como
$ x \to -\infty $, $ p \to -\infty $. Así que en algunos enormes \omterm{def:g10:numbers:interval}{intervalo}
$\intcc{-M}{M}$, $ p(-M) < 0 < p(M)$: el IVT ofrece una \omterm{def:g11:quad:discriminant}{raíz}.
El mismo argumento del término dominante funciona textualmente para cualquier
grado --- los poderes \omterm{def:g11:func:parity}{impares} mantienen signos opuestos en los dos
infinitos.

\textbf{9.} Reducir a la mitad $ n $ veces deja un \omterm{def:g10:numbers:interval}{intervalo} de longitud
$2^{-n}$; $2^{-20} \approx 9.5 \times 10^{-7} < 10^{-6}$:
veinte pasos. Newton duplica los dígitos correctos en cada paso: desde
un dígito correcto, aproximadamente $3$--$4$ los pasos llegan a seis --- el
la tortuga y el galgo de la búsqueda de raíces.

\textbf{10.} $ g $ no es continua en $\intcc{-1}{1}$: eso
explota en $0$, que el \omterm{def:g10:numbers:interval}{intervalo} contiene. Los IVT
la conclusión se evapora con su hipótesis --- siempre verifique
continuidad \emph{en general \omterm{def:g10:numbers:interval}{intervalo}}, no sólo en el
puntos finales.

\textbf{11.} $ g(x) = f(x) - x $ es continua;
$ g(0) = f(0) \geq 0$ (los valores se encuentran en $\intcc{0}{1}$) y
$ g(1) = f(1) - 1 \leq 0$. Por el IVT, $ g(c) = 0$ para algunos $ c $:
$ f(c) = c $.

\textbf{12.} Idealizando al país como segmento
$\intcc{0}{1}$: $ f $ envía cada lugar al lugar mostrado
directamente debajo de él en el mapa arrugado; arrugar mantiene el
mapa sobre el país, por lo que $ f $ aterriza en $\intcc{0}{1}$, y es continua (el papel no se rompe). El \omterm{pb:g10:functions:1}{punto fijo} $ c $ es
el lugar que se encuentra precisamente encima de su propio \omterm{def:g10:functions:function}{imagen} --- Brouwer
El teorema maneja el mapa bidimensional honesto.

\textbf{13.} $ g(0) + g(\pi) = \left(T(0) - T(\pi)\right) +
\left(T(\pi) - T(2\pi)\right) = T(0) - T(2\pi) = 0$. Entonces
$ g(0)$ y $ g(\pi)$ son opuestos (o ambos cero): en cualquier sentido
$ g $, continua, desaparece en algún $ t^* \in \intcc{0}{\pi}$:
$ T(t^*) = T(t^* + \pi)$ --- en algún lugar del ecuador, dos
Los puntos antípodas coinciden en la temperatura, en cada uno de ellos.
momento.

\textbf{14.} Girar la mesa un cuarto de vuelta.
los roles de las dos diagonales: el par de patas que presionaban
el suelo y la pareja que (uno de ellos) flotaba se intercambian
situaciones, por lo que la altura flotante --- contaba con su signo
como ``espacio de la diagonal distinguida'' --- toma lo contrario
señales en $\theta $ y $\theta + \frac\pi2$. continua suelo
hace $ h $ continua, y el IVT entrega un ángulo
$\theta^*$ con $ h(\theta^*) = 0$: cuatro patas abajo. (No hay tal
suerte con una mesa rectangular o un suelo escalonado.)

\textbf{15.} Plantilla: construir el \emph{\omterm{def:g11:prob:variance}{diferencia}} de los dos
cantidades que desea que coincidan ($ f(x) - x $;
$ T(t) - T(t + \pi)$; posición arriba menos posición abajo para el
caminante; el espacio firmado para la mesa); comprueba que toma valores de
ambos signos (a menudo mediante una simetría de los extremos); invocar el IVT
en un continua \omterm{def:g11:func:function}{función}. Coincidencia forzada.

\textbf{16.} Preguntas abiertas: \emph{cuántos} soluciones
(respuesta por monotonicidad estricta y el teorema de biyección,
pregunta 7) y \emph{dónde} son (respondidos por \omterm{thm:g12:limcont:ivt}{dicotomía}
o Newton, pregunta 9). El IVT es un oráculo de existencia pura.

\textbf{17.} $3 - \frac{7}{x + 2}$: \omterm{def:g12:limcont:asymptote}{asíntota vertical}
$ x = -2$, horizontal $ y = 3$; a cada lado de $-2$ el
\omterm{def:g11:func:function}{función} es \omterm{def:g12:seq:monotonic}{creciente} ($-\frac{7}{x+2}$ aumenta donde
definido); la curva es la referencia \omterm{def:g11:func:reference}{hipérbola}
$-\frac7x $ desplazado a la izquierda $2$ y arriba $3$ --- dos ramas
abrazando al asíntota cruz.

\textbf{18.} Para $ x \neq 1$: $ f(x) = x + 1$, entonces
$\lim_{x \to 1} f = 2$: configuración $ f(1) = 2$ restaura
continuidad. La discontinuidad fue un pinchazo, no un salto ---
llene el único punto que falta y la curva sanará:
\emph{desmontable}.

\textbf{19.} continua en cada \omterm{def:g10:numbers:interval}{intervalo} entre consecutivos
números enteros; en cada entero $ n $ salta de $ n - 1$ a $ n $.
Con $ a = 0.9$, $ b = 1.1$: $\lfloor a \rfloor = 0 < \frac12 <
1 = \lfloor b \rfloor $, todavía $\lfloor x \rfloor = \frac12$ tiene
no hay solución --- el salto se teletransporta $\frac12$, que es
precisamente lo que prohíbe la IVT continua funciones hacer.

\textbf{20.} Cara local: valor $=$ límite en cada punto ---
la punción removible (pregunta 18) lo violó de forma curable, la
saltos del suelo (pregunta 19) de forma irremediable. Cara global: en un
\omterm{def:g10:numbers:interval}{intervalo}, se visitan todos los valores intermedios --- el \omterm{def:g11:quad:discriminant}{raíz}
certificados (6--8), el \omterm{pb:g10:functions:1}{punto fijo}, el ecuador, la mesa
y el excursionista todos viven aquí. El nombre oficial del eslogan: el
\emph{teorema del valor intermedio}.
\end{solution}
