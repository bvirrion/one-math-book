\begin{solution}{exo:g12:complex:1}
$(3-2\iu)(1+4\iu) = 3 + 12\iu - 2\iu - 8\iu^2 = 11 + 10\iu$.

$\dfrac{2+\iu}{1-\iu} = \dfrac{(2+\iu)(1+\iu)}{(1-\iu)(1+\iu)}
= \dfrac{1 + 3\iu}{2} = \dfrac12 + \dfrac32\iu$.

$\iu^{2027}$: بما أن $\iu^4 = 1$ و $2027 = 4 \times 506 + 3$، فإن
$\iu^{2027} = \iu^3 = -\iu$.
\end{solution}

\begin{solution}{exo:g12:complex:2}
\emph{(أ)} $\Delta = 16 - 52 = -36$:
$z = \dfrac{4 \pm 6\iu}{2} = 2 \pm 3\iu$. وهما زوج مترافق؛ وجداؤهما
$= 4 + 9 = 13 = \frac ca$. \emph{(ب)} $\Delta = 1 - 4 = -3$:
$z = \dfrac{-1 \pm \iu\sqrt3}{2}$ (وهما جذرا الوحدة التكعيبيان $j$ و
$\conj j$)؛ وجداؤهما $= \frac{1 + 3}{4} = 1$.
\end{solution}

\begin{solution}{exo:g12:complex:3}
$\abs{z_1} = \sqrt2$، \omterm{def:g12:complex:argument}{والعمدة} $\frac{3\pi}{4}$ (الربع الثاني):
$z_1 = \sqrt2\,\eu^{3\iu\pi/4}$.
و $\abs{z_2} = 2$، \omterm{def:g12:complex:argument}{والعمدة} $-\frac\pi6$:
$z_2 = 2\,\eu^{-\iu\pi/6}$. ومنه
\[
\frac{z_1}{z_2} = \frac{\sqrt2}{2}\,\eu^{\iu\left(\frac{3\pi}{4} + \frac{\pi}{6}\right)}
= \frac{\sqrt2}{2}\,\eu^{11\iu\pi/12}.
\]
وجبريًا،
\[
\frac{z_1}{z_2} = \frac{-1+\iu}{\sqrt3 - \iu}
= \frac{(-1+\iu)(\sqrt3+\iu)}{4}
= \frac{-\sqrt3 - 1 + \iu(\sqrt3 - 1)}{4}.
\]
وبمطابقة الأجزاء الحقيقية:
$\frac{\sqrt2}{2}\cos\frac{11\pi}{12} = \frac{-\sqrt3-1}{4}$، إذن
\[
\cos\frac{11\pi}{12} = -\frac{\sqrt6 + \sqrt2}{4}.
\]
\end{solution}

\begin{solution}{exo:g12:complex:4}
\emph{(أ)} $\abs{z - 2} = \abs{z - (-\iu)}$ يعني أن $M$ على البعد نفسه من
$A(2, 0)$ ومن $B(0, -1)$: أي \emph{محور القطعة} $[AB]$.

\emph{(ب)} البعد $3$ عن النقطة $(1,1)$: أي \emph{الدائرة} التي مركزها
$1 + \iu$ ونصف قطرها $3$.

\emph{(ج)} حسب \cref{prop:g12:complex:geometry}(4)، كون $\frac{z-1}{z+1}$ تخيليًا
صرفًا يعني أن المتجهتين من $A(1,0)$ إلى $M$ ومن $B(-1,0)$ إلى $M$
متعامدتان: أي إن $M$ ترى القطعة $[AB]$ تحت زاوية قائمة. والمجموعة هي
\emph{الدائرة ذات القطر $[AB]$} (أي دائرة الوحدة) \emph{محذوفًا منها
النقطتان $A$ و $B$} نفساهما.
\end{solution}

\begin{solution}{exo:g12:complex:5}
بصيغة دي موافر: $\cos3\theta + \iu\sin3\theta = (\cos\theta + \iu\sin\theta)^3$.
وبالنشر بمبرهنة ثنائي الحد مع $c = \cos\theta$ و $s = \sin\theta$:
\[
(c + \iu s)^3 = c^3 + 3\iu c^2 s - 3 c s^2 - \iu s^3
= (c^3 - 3cs^2) + \iu(3c^2 s - s^3).
\]
وبمطابقة الأجزاء واستعمال $s^2 = 1 - c^2$ و $c^2 = 1 - s^2$:
\[
\cos3\theta = 4\cos^3\theta - 3\cos\theta,
\qquad
\sin3\theta = 3\sin\theta - 4\sin^3\theta .
\]
\end{solution}

\begin{solution}{exo:g12:complex:6}
\[
\cos^3\theta = \left(\frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}\right)^{\!3}
= \frac{\eu^{3\iu\theta} + 3\eu^{\iu\theta} + 3\eu^{-\iu\theta} + \eu^{-3\iu\theta}}{8}
= \frac{\cos3\theta + 3\cos\theta}{4}.
\]
ومنه
\[
\int_0^{\pi/2}\cos^3\theta\,\dd\theta
= \frac14\left[\frac{\sin3\theta}{3} + 3\sin\theta\right]_0^{\pi/2}
= \frac14\left(-\frac13 + 3\right) = \frac{2}{3}.
\]
(وهذا متسق مع $W_3 = \frac23$ في \cref{exo:g12:integ:9}، لأن
$\int_0^{\pi/2}\cos^3 = \int_0^{\pi/2}\sin^3$ بحكم التناظر
$\theta \mapsto \frac\pi2 - \theta$.)
\end{solution}

\begin{solution}{exo:g12:complex:7}
$c = a + \eu^{\pm\iu\pi/3}(b - a)$ مع $b - a = 2 + \iu$ و
$\eu^{\pm\iu\pi/3} = \frac12 \pm \iu\frac{\sqrt3}{2}$:
\[
\eu^{\iu\pi/3}(2+\iu) = \left(\tfrac12 + \iu\tfrac{\sqrt3}{2}\right)(2+\iu)
= \left(1 - \tfrac{\sqrt3}{2}\right) + \iu\left(\tfrac12 + \sqrt3\right),
\]
إذن $c_1 = \left(2 - \frac{\sqrt3}{2}\right) +
\iu\left(\frac32 + \sqrt3\right)$، وبالمثل مع الزاوية
$-\frac\pi3$:
$c_2 = \left(2 + \frac{\sqrt3}{2}\right) + \iu\left(\frac32 - \sqrt3\right)$.
\end{solution}

\begin{solution}{exo:g12:complex:8}
$-4 = 4\,\eu^{\iu\pi}$، إذن تكون حلول $z^4 = -4$ هي
\[
z_k = \sqrt2\,\eu^{\iu\left(\frac\pi4 + \frac{k\pi}{2}\right)},
\quad k = 0, 1, 2, 3,
\]
\emph{أي} $1 + \iu$ و $-1 + \iu$ و $-1 - \iu$ و $1 - \iu$: وهي رؤوس
مربع نصف قطر دائرته المحيطة $\sqrt 2$. وبتجميع الجذور المترافقة،
\[
z^4 + 4 = \bigl(z - (1{+}\iu)\bigr)\bigl(z - (1{-}\iu)\bigr)
\bigl(z + (1{-}\iu)\bigr)\bigl(z + (1{+}\iu)\bigr)
= (z^2 - 2z + 2)(z^2 + 2z + 2).
\]
\end{solution}

\begin{solution}{exo:g12:complex:9}
مع $q = \eu^{\iu\theta} \neq 1$:
\[
\sum_{k=0}^{n} \eu^{\iu k\theta} = \frac{\eu^{\iu(n+1)\theta} - 1}{\eu^{\iu\theta} - 1}
= \frac{\eu^{\iu(n+1)\theta/2}}{\eu^{\iu\theta/2}}
\cdot\frac{\eu^{\iu(n+1)\theta/2} - \eu^{-\iu(n+1)\theta/2}}{\eu^{\iu\theta/2} - \eu^{-\iu\theta/2}}
= \eu^{\iu n\theta/2}\,
\frac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}},
\]
باستعمال $\eu^{\iu\alpha} - \eu^{-\iu\alpha} = 2\iu\sin\alpha$ (وهو
\emph{تحليل نصف الزاوية}). وبأخذ الأجزاء الحقيقية:
\[
S = \frac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}}\,
\cos\frac{n\theta}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:complex:10}
\emph{1.} $\omega$ \omterm{def:g11:quad:discriminant}{جذر} للوحدة من الرتبة $5$ مخالف للعدد $1$، إذن ينعدم مجموع
الجذور الخمسة كلها (\cref{thm:g12:complex:rootsofunity})، وهذا يُقرأ
$1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$.

\emph{2.} $u + v = \omega + \omega^2 + \omega^3 + \omega^4 = -1$ حسب النقطة 1.
وأما الجداء، فباستعمال $\omega^5 = 1$:
\[
uv = (\omega + \omega^4)(\omega^2 + \omega^3)
= \omega^3 + \omega^4 + \omega^6 + \omega^7
= \omega^3 + \omega^4 + \omega + \omega^2 = -1 .
\]
إذن $u, v$ جذرا \omterm{def:g10:algebra:equation}{المعادلة} $X^2 + X - 1 = 0$:
$\left\{\frac{-1+\sqrt5}{2}, \frac{-1-\sqrt5}{2}\right\}$. والآن
$u = \omega + \conj\omega = 2\cos\frac{2\pi}{5} > 0$ (لأن
$\frac{2\pi}{5} < \frac{\pi}{2}$)، إذن $u = \frac{-1 + \sqrt5}{2}$.

\emph{3.} $\cos\frac{2\pi}{5} = \frac u2 = \frac{\sqrt5 - 1}{4}$.
\end{solution}

\begin{solution}{pb:g12:complex:1}
\textbf{1.} $(2 + \iu)(3 - \iu) = 6 - 2\iu + 3\iu + 1 =
7 + \iu$. \quad $\frac{1 + \iu}{1 - \iu} =
\frac{(1 + \iu)^2}{2} = \frac{2\iu}{2} = \iu$. \quad
$\abs{3 + 4\iu} = 5$.

\textbf{2.} $1 + \iu = \sqrt2\,\eu^{\iu\pi/4}$؛
$-2 = 2\eu^{\iu\pi}$؛
$1 - \iu\sqrt3 = 2\eu^{-\iu\pi/3}$.

\textbf{3.} $(1 + \iu)^8 = \left(\sqrt2\right)^8
\eu^{8\iu\pi/4} = 16\,\eu^{2\iu\pi} = 16$.

\textbf{4.} $z^2 = \eu^{\iu\pi/2}$:
$z = \pm\eu^{\iu\pi/4} = \pm\frac{\sqrt2}{2}(1 + \iu)$. و
$\Delta = 4 - 20 = -16$: أي $z = 1 \pm 2\iu$.

\textbf{5.} $z \mapsto \iu z$: دوران بربع دورة
حول المبدأ. و $\abs{z - (1 + \iu)} = 2$: الدائرة التي
مركزها $1 + \iu$ ونصف قطرها $2$.

\textbf{6.} $\left(\omega^k\right)^5 = \eu^{2\iu\pi k} = 1$،
والنقط الخمس $\eu^{2\iu k\pi/5}$ متمايزة: فهي حلول
$z^5 = 1$ الخمسة كلها. وهي تجلس على دائرة الوحدة بزوايا
متساوية قدرها $72^\circ$: أي مخمَّس منتظم أحد رؤوسه عند
$1$.

\textbf{7.} $S = 1 + \omega + \dots + \omega^4$ يحقق
$(1 - \omega)S = 1 - \omega^5 = 0$، و $\omega \neq 1$: إذن
$S = 0$. (فنقطة توازن المخمَّس هي مركزه.)

\textbf{8.} $\omega^4 = \overline\omega$ و
$\omega^3 = \overline{\omega^2}$: وبمزاوجة المترافقات، يُقرأ \omterm{def:g12:complex:def}{الجزء
الحقيقي} من المجموع
$1 + 2\cos\frac{2\pi}{5} + 2\cos\frac{4\pi}{5} = 0$.

\textbf{9.} $\cos\frac{4\pi}{5} = 2c^2 - 1$، إذن
$1 + 2c + 4c^2 - 2 = 0$، أي $4c^2 + 2c - 1 = 0$:
$c = \frac{-1 + \sqrt5}{4}$ (وهو \omterm{def:g11:quad:discriminant}{الجذر} الموجب، لأن
$72^\circ$ زاوية حادة):
$\cos 72^\circ = \frac{\sqrt5 - 1}{4}$.

\textbf{10.} عدديًا $c \approx 0.30902$: وهذا يوافق
الآلة الحاسبة. و $\frac{1}{2\varphi} = \frac{1}{1 + \sqrt5} =
\frac{\sqrt5 - 1}{4}$ (بجعل المقام ناطقًا): $c = \frac{1}{2\varphi}$.
فالمخمَّس ذهبي من كل جهاته: إذ يقطع قطره
ضلعه بالنسبة $\varphi$ --- وهو التشابه الذاتي اللانهائي
للنجمة الخماسية.

\textbf{11.} الجذور التربيعية قابلة للإنشاء بالفرجار (بآلة
نصف الدائرة \omterm{def:g10:stats:mean}{للمتوسط} الهندسي في الكتاب السابق)، إذن العدد المبني
من الناطقة بالجذور التربيعية وحدها
--- مثل $\cos 72^\circ$ --- قابل للإنشاء؛ أما الجذور التكعيبية فليست
كذلك، و $\cos 20^\circ$ يحتاج إلى واحد منها: فالمخمَّس يستسلم،
والتثليث يقاوم. ومحك غاوس يحسم كل مضلع
منتظم --- وقد جعله مضلعه ذو السبعة عشر ضلعًا في التاسعة عشرة يختار
الرياضيات.

\textbf{12.} $-4 = 4\eu^{\iu\pi}$: فللجذور الرابعة
\omterm{def:g12:complex:modulus}{المقياس} $\sqrt2$ والعمد
$\frac{\pi}{4} + k\frac{\pi}{2}$: أي الأعداد الأربعة
$\pm 1 \pm \iu$. وبمزاوجة المترافقات:
$(z - 1 - \iu)(z - 1 + \iu) = z^2 - 2z + 2$ و
$(z + 1 - \iu)(z + 1 + \iu) = z^2 + 2z + 2$: وهو
تحليل \cref{exo:g12:complex:8}.

\textbf{13.} $(\cos\theta + \iu\sin\theta)^3 = \cos^3\theta +
3\iu\cos^2\theta\sin\theta - 3\cos\theta\sin^2\theta -
\iu\sin^3\theta$؛ وبصيغة دي موافر يساوي هذا
$\cos 3\theta + \iu \sin 3\theta$. وبأخذ الأجزاء الحقيقية:
$\cos 3\theta = \cos^3\theta - 3\cos\theta(1 - \cos^2\theta)
= 4\cos^3\theta - 3\cos\theta$ --- وتسلّم الأجزاء التخيلية
$\sin 3\theta$ بلا مقابل.

\textbf{14.} هو \omterm{def:g12:complex:def}{الجزء الحقيقي} من
$1 + \omega + \dots + \omega^4 = 0$: فالمجموع يساوي $0$ ---
أي بالضبط السؤال 8، منظورًا إليه عبر نواة
\cref{exo:g12:complex:9}.

\textbf{15.} $8\iu = 8\eu^{\iu\pi/2}$: فالجذور
$2\eu^{\iu\pi/6} = \sqrt3 + \iu$؛
و $2\eu^{5\iu\pi/6} = -\sqrt3 + \iu$؛
و $2\eu^{3\iu\pi/2} = -2\iu$.

\textbf{16.} $1 + \iu = \sqrt2\,\eu^{\iu\pi/4}$: فالتطبيق
يدير المستوي بزاوية $45^\circ$ ويكبّره بالمعامل $\sqrt2$. و
يصير مربع الوحدة مربعًا مائلًا طول ضلعه $\sqrt2$ ---
مضاعف المساحة، مدارًا ثمن دورة.

\textbf{17.} $\abs{z_1 z_2}^2 = z_1 z_2 \overline{z_1 z_2} =
z_1\overline{z_1}\, z_2\overline{z_2} =
\abs{z_1}^2\abs{z_2}^2$: خذ الجذور التربيعية. وأما قوى
$1 + \iu$: $1 + \iu$ و $2\iu$ و $-2 + 2\iu$ و $-4$: فكل منها
أبعد بدورة قدرها $45^\circ$ وأطول بمعامل $\sqrt2$ --- أي لولب
خارج عبر المقاييس
$\sqrt2, 2, 2\sqrt2, 4, \dots$

\textbf{18.} $\mathrm{j}$ \omterm{def:g11:quad:discriminant}{جذر} ثالث للوحدة
مخالف للعدد $1$، إذن (بحجة السؤال 7 مع $n = 3$)
$1 + \mathrm{j} + \mathrm{j}^2 = 0$. وأما المحك على المثلث
المعياري: $a + \mathrm{j}b + \mathrm{j}^2 c = 1 +
\mathrm{j}\cdot\mathrm{j} + \mathrm{j}^2 \cdot \mathrm{j}^2 =
1 + \mathrm{j}^2 + \mathrm{j}^4 = 1 + \mathrm{j}^2 +
\mathrm{j} = 0$: فهو محقَّق، كما يقتضي المثلث المتساوي الأضلاع
$\left(1, \mathrm{j}, \mathrm{j}^2\right)$.

\textbf{19.} على $\C$، ينشطر كثير حدود حقيقي إلى عوامل
خطية؛ وتأتي جذوره غير الحقيقية أزواجًا مترافقة (بمرافقة
\omterm{def:g10:algebra:equation}{المعادلة})، ويُضرب كل زوج فيعطي تربيعية حقيقية
$z^2 - 2\operatorname{Re}(z_0)\,z + \abs{z_0}^2$: ومنه التحليل
الحقيقي إلى قطع من الدرجة $1$ والدرجة $2$ ---
وقد نفّذه السؤال 12 على $z^4 + 4$.

\textbf{20.} نقط؛ ودوران وتكبير؛ ومضلعات منتظمة؛
ومتطابقات مثلثية \omterm{def:g10:algebra:expand}{بالنشر}؛ وعالم مغلق
يكون فيه لكل \omterm{def:g10:algebra:equation}{معادلة} درجتها $n$ جذورها $n$. والخاتمة الصغرى:
في السؤال 10، صافح مخمَّس $\pi$ النسبة
الذهبية --- وهما أشهر ضيفَي إقليدس، وقد قُدّم أحدهما إلى الآخر أخيرًا
بمجموع خمسة أسهم يساوي صفرًا.
\end{solution}
