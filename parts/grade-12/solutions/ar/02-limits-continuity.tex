\section*{الفصل \ref{ch:g12:limcont} --- النهايات والاتصال}

\begin{solution}{exo:g12:limcont:1}
بالقسمة على $x^3$:
$\dfrac{2x^3 - x + 1}{5x^3 + x^2} = \dfrac{2 - 1/x^2 + 1/x^3}{5 + 1/x}
\to \dfrac{2}{5}$.

$x^2 + 3x - 1 = x^2(1 + 3/x - 1/x^2) \to +\infty$ لما $x \to -\infty$، لأن
$x^2 \to +\infty$ والقوس يؤول إلى $1$.

ولما $x \to 2^+$، يؤول البسط إلى $3 > 0$ ويؤول المقام إلى $0^+$،
إذن $\dfrac{x+1}{x-2} \to +\infty$.

ومن أجل $x > 0$، يكون $-\dfrac1x \leq \dfrac{\cos x}{x} \leq \dfrac1x$، إذن النهاية
هي $0$ بمبرهنة الحصر.
\end{solution}

\begin{solution}{exo:g12:limcont:2}
\emph{1.} $2x^2 - 3x + 1 = (x-1)(2x-1)$ (\omterm{def:g11:quad:discriminant}{والجذر} $x=1$ ظاهر من
$2 - 3 + 1 = 0$). ومن أجل $x \neq 1$، اختصر $x - 1$: فيكون $f(x) = 2x - 1$.

\emph{2.} ومنه $\lim_{x\to1} f(x) = 1$. ووضع $f(1) = 1$ يمدّد $f$ إلى
\omterm{def:g11:func:function}{دالة} كثير الحدود $x \mapsto 2x-1$ على $\R$، وهي \omterm{def:g12:limcont:continuity}{متصلة} عند $1$.
\end{solution}

\begin{solution}{exo:g12:limcont:3}
اكتب $f(x) = \dfrac{3x - 1}{x + 2} = \dfrac{3(x+2) - 7}{x+2}
= 3 - \dfrac{7}{x+2}$. فلما $x \to \pm\infty$، يكون $f(x) \to 3$: إذن المستقيم $y = 3$
\omterm{def:g12:limcont:asymptote}{مقارب} أفقي عند اللانهايتين. ولما $x \to (-2)^\pm$،
$f(x) \to \mp\infty$: إذن المستقيم $x = -2$ \omterm{def:g12:limcont:asymptote}{مقارب} شاقولي. \omterm{def:g10:functions:graph}{والمنحنى}
\omterm{def:g11:func:reference}{قطع زائد}، يقع تحت $y=3$ من أجل $x > -2$ وفوقه من أجل $x < -2$.
\end{solution}

\begin{solution}{exo:g12:limcont:4}
اضرب في المرافق:
\[
\sqrt{x^2 + x} - x = \frac{x^2 + x - x^2}{\sqrt{x^2+x} + x}
= \frac{x}{x\left(\sqrt{1 + 1/x} + 1\right)}
= \frac{1}{\sqrt{1 + 1/x} + 1}
\xrightarrow[x\to+\infty]{} \frac{1}{2}.
\]
وبالمثل،
\[
\frac{\sqrt{1+x} - 1}{x} = \frac{(1+x) - 1}{x\left(\sqrt{1+x} + 1\right)}
= \frac{1}{\sqrt{1+x} + 1} \xrightarrow[x\to0]{} \frac{1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:limcont:5}
\emph{1.} بما أن $\sin x \geq -1$، فإن $f(x) \geq x - 1 \to +\infty$، ونستنتج
بالمقارنة. ومع ذلك فإن $f'(x) = 1 + \cos x$ تنعدم عند كل مضاعف
فردي للعدد $\pi$، وعلى كل \omterm{def:g10:numbers:interval}{فترة} $\intco{A}{+\infty}$ تتناوب \omterm{def:g11:func:function}{الدالة}
بين فترات تزايد تام؛ فهي \omterm{def:g12:seq:monotonic}{متزايدة} لكن ليس
تمامًا، والأهم أن تزايدها ليس لازمًا: فالمقارنة تكفي.

\emph{2.} من أجل $x > 1$، اقسم على $x$:
\[
g(x) = \frac{1 + \frac{\sin x}{x}}{1 - \frac{\sin x}{x}},
\]
و $\frac{\sin x}{x} \to 0$ بمبرهنة الحصر
($\abs{\sin x} \leq 1$). ومنه $g(x) \to \frac{1+0}{1-0} = 1$.
\end{solution}

\begin{solution}{exo:g12:limcont:6}
$f(x) = x^3 - 3x + 1$ \omterm{def:g12:limcont:continuity}{متصلة}، مع $f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$:
فتتزايد $f$ على $\intoc{-\infty}{-1}$، وتتناقص على $\intcc{-1}{1}$،
وتتزايد على $\intco{1}{+\infty}$، مع \omterm{def:g10:functions:extrema}{قيمة عظمى} محلية $f(-1) = 3 > 0$ و
\omterm{def:g10:functions:extrema}{قيمة صغرى} محلية $f(1) = -1 < 0$، و $\lim_{-\infty} f = -\infty$،
و $\lim_{+\infty} f = +\infty$.

وعلى كل \omterm{def:g10:numbers:interval}{فترة} من فترات الرتابة الثلاث، تكون $f$ \omterm{def:g12:limcont:continuity}{متصلة} \omterm{def:g12:seq:monotonic}{ورتيبة}
تمامًا، ويقع $0$ تمامًا بين القيم أو النهايات عند الطرفين، إذن
تعطي مبرهنة التقابل صفرًا واحدًا بالضبط في كل \omterm{def:g10:numbers:interval}{فترة}: أي ثلاثة أصفار في المجموع.
والتحقق من الإشارات: $f(-2) = -1 < 0 < f(-1) = 3$، إذن صفر واحد في
$\intoo{-2}{-1}$؛ و $f(0) = 1 > 0 > f(1) = -1$، فصفر واحد في $\intoo{0}{1}$؛
و $f(1) = -1 < 0 < f(2) = 3$، فصفر واحد في $\intoo{1}{2}$.
\end{solution}

\begin{solution}{exo:g12:limcont:7}
لتكن $g(x) = f(x) - x$، وهي \omterm{def:g12:limcont:continuity}{متصلة} على $\intcc{0}{1}$. وبما أن
$f(0) \in \intcc{0}{1}$، فإن $g(0) = f(0) \geq 0$؛ وبما أن $f(1) \in \intcc{0}{1}$،
فإن $g(1) = f(1) - 1 \leq 0$. وبمبرهنة القيم الوسطى، تنعدم $g$ عند
$c \in \intcc{0}{1}$ ما، أي $f(c) = c$.
\end{solution}

\begin{solution}{exo:g12:limcont:8}
ليكن $u(t)$ و $d(t)$ الموضعين (مقيسين بالمسافة من أسفل
الدرب) عند اللحظة $t \in \intcc{8}{20}$ في الصعود وفي
النزول. وكلاهما متصل، مع $u(8) = 0$ و $u(20) = L$ (وهو طول الدرب)
و $d(8) = L$ و $d(20) = 0$. \omterm{def:g11:func:function}{والدالة} $g = u - d$ \omterm{def:g12:limcont:continuity}{متصلة} مع
$g(8) = -L \leq 0$ و $g(20) = L \geq 0$، إذن $g(c) = 0$ عند لحظة $c$ ما بمبرهنة
القيم الوسطى: أي إن الموضعين في اليومين يتطابقان عند اللحظة $c$.
\end{solution}

\begin{solution}{exo:g12:limcont:9}
\emph{1.} $f'(x) = 5x^4 + 3x^2 + 1 > 0$، إذن $f$ \omterm{def:g12:seq:monotonic}{متزايدة} تمامًا على
$\R$؛ ومع $\lim_{\pm\infty} f = \pm\infty$ \omterm{def:g12:limcont:continuity}{والاتصال}، تعطي مبرهنة
التقابل صفرًا حقيقيًا وحيدًا $\alpha$. وبما أن $f(0) = -1 < 0$ و
$f(1) = 2 > 0$، فإن $\alpha \in \intoo{0}{1}$.

\emph{2.} احتفظ \omterm{def:g10:numbers:interval}{بفترة} $\intcc{a_n}{b_n}$ حيث
$f(a_n) < 0 < f(b_n)$، انطلاقًا من $\intcc{0}{1}$؛ وعند كل خطوة احسب
قيمة $f$ عند \omterm{prop:g10:coordgeom:midpoint}{المنتصف} $m$ واحتفظ بنصف \omterm{def:g10:numbers:interval}{الفترة} الذي يستمر فيه تغير
الإشارة. وبعد $n$ خطوة يصير طول \omterm{def:g10:numbers:interval}{الفترة} $2^{-n}$، إذن يقرّب \omterm{prop:g10:coordgeom:midpoint}{المنتصف}
$\alpha$ بدقة $2^{-(n+1)}$. ونحتاج إلى
$2^{-(n+1)} \leq 10^{-6}$، أي $n + 1 \geq \frac{6}{\log_{10} 2} \approx 19.93$:
إذن $n = 20$ تكرارًا تكفي.
\end{solution}

\begin{solution}{pb:g12:limcont:1}
\textbf{1.} الحدود المهيمنة: $\frac{3x^2}{x^2} \to 3$؛
$\frac{2x}{x^2} = \frac2x \to 0$؛
$x^3 - x^2 = x^3\left(1 - \frac1x\right) \to +\infty$.

\textbf{2.} $\lim_{x \to \pm\infty} f = 3$: \omterm{def:g12:limcont:asymptote}{فالمقارب}
الأفقي $y = 3$؛ و $\lim_{x \to (-2)^\pm} f = \mp\infty$:
\omterm{def:g12:limcont:asymptote}{فالمقارب} الشاقولي $x = -2$.

\textbf{3.} $\frac{(x-3)(x+3)}{x - 3} = x + 3 \to 6$. و
$\frac{\sqrt{x+1} - 1}{x} = \frac{x}{x(\sqrt{x+1} + 1)} =
\frac{1}{\sqrt{x+1} + 1} \to \frac12$.

\textbf{4.} $\abs{x \sin\frac1x} \leq \abs x \to 0$، إذن
النهاية هي $0$. و
$\abs{\frac{\sin x}{x}} \leq \frac1x \to 0$: فالنهاية $0$.

\textbf{5.} $\frac{\sqrt{4x^2 + x}}{x} =
\sqrt{4 + \frac1x} \to \sqrt4 = 2$ (\omterm{def:g12:limcont:continuity}{باتصال} \omterm{def:g11:quad:discriminant}{الجذر}
التربيعي والتركيب).

\textbf{6.} $f(x) = x^3 + x$ \omterm{def:g12:limcont:continuity}{متصلة} مع
$f(1) = 2 < 5 < 10 = f(2)$: فبمبرهنة القيم الوسطى، $f(c) = 5$ من أجل
$c \in \intoo{1}{2}$ ما.

\textbf{7.} $f'(x) = 3x^2 + 1 > 0$: فهي \omterm{def:g12:seq:monotonic}{متزايدة} تمامًا على
$\R$، \omterm{def:g12:limcont:continuity}{ومتصلة}، ونهايتاها $\mp\infty$: فبمبرهنة
التقابل، تُبلَغ كل قيمة --- وخاصة $5$ ---
مرة واحدة بالضبط.

\textbf{8.} $p(x) = x^3\left(1 + \frac ax + \frac{b}{x^2} +
\frac{c}{x^3}\right)$: فلما $x \to +\infty$، يكون $p \to +\infty$؛ ولما
$x \to -\infty$، يكون $p \to -\infty$. إذن على \omterm{def:g10:numbers:interval}{فترة} ضخمة ما
$\intcc{-M}{M}$، يكون $p(-M) < 0 < p(M)$: فتسلّم مبرهنة القيم الوسطى جذرًا.
وتصح حجة الحد المهيمن نفسها حرفيًا من أجل أي درجة
\omterm{def:g11:func:parity}{فردية} --- فالقوى \omterm{def:g11:func:parity}{الفردية} تحتفظ بإشارتين متعاكستين عند
اللانهايتين.

\textbf{9.} التنصيف $n$ مرة يترك \omterm{def:g10:numbers:interval}{فترة} طولها
$2^{-n}$؛ و $2^{-20} \approx 9.5 \times 10^{-7} < 10^{-6}$:
أي عشرون خطوة. أما نيوتن فيضاعف الأرقام الصحيحة عند كل خطوة: فمن
رقم صحيح واحد، تبلغ $3$--$4$ خطوات ستة أرقام --- إنهما
السلحفاة والسلوقي في اقتناص الجذور.

\textbf{10.} $g$ ليست \omterm{def:g12:limcont:continuity}{متصلة} على $\intcc{-1}{1}$: فهي
تنفجر عند $0$، وهو داخل \omterm{def:g10:numbers:interval}{الفترة}. فخاتمة مبرهنة القيم
الوسطى تتبخر مع فرضيتها --- فتحقق دائمًا من
\omterm{def:g12:limcont:continuity}{الاتصال} \emph{على \omterm{def:g10:numbers:interval}{الفترة} كلها}، لا عند الطرفين
وحدهما.

\textbf{11.} $g(x) = f(x) - x$ \omterm{def:g12:limcont:continuity}{متصلة}؛ و
$g(0) = f(0) \geq 0$ (لأن القيم تقع في $\intcc{0}{1}$) و
$g(1) = f(1) - 1 \leq 0$. فبمبرهنة القيم الوسطى، $g(c) = 0$ من أجل $c$ ما:
أي $f(c) = c$.

\textbf{12.} بمثلنة البلد على أنه القطعة
$\intcc{0}{1}$: ترسل $f$ كل موضع إلى الموضع المبيَّن
تحته مباشرة على الخريطة المكوّرة؛ والتكوير يبقي
الخريطة فوق البلد، إذن تقع صورة $f$ في $\intcc{0}{1}$، وهي
\omterm{def:g12:limcont:continuity}{متصلة} (فالورق لا يتمزق). \omterm{pb:g10:functions:1}{والنقطة الثابتة} $c$ هي
الموضع الواقع بالضبط فوق صورته --- ومبرهنة براور
تعالج الخريطة الأمينة ذات البعدين.

\textbf{13.} $g(0) + g(\pi) = \left(T(0) - T(\pi)\right) +
\left(T(\pi) - T(2\pi)\right) = T(0) - T(2\pi) = 0$. إذن
$g(0)$ و $g(\pi)$ متعاكسان (أو معدومان معًا): وفي الحالتين
تنعدم $g$، وهي \omterm{def:g12:limcont:continuity}{متصلة}، عند $t^* \in \intcc{0}{\pi}$ ما:
$T(t^*) = T(t^* + \pi)$ --- أي إن ثمة، في مكان ما من خط الاستواء، نقطتين
متقابلتين تتفقان في درجة الحرارة، في كل
لحظة.

\textbf{14.} إدارة الطاولة بربع دورة تبادل
دوري القطرين: فزوج الأرجل الذي كان يضغط
الأرض وزوج الأرجل الذي كانت إحداهما تحوم يتبادلان
الوضعين، إذن يأخذ ارتفاع التحويم --- محسوبًا بإشارته
بوصفه ``فجوة القطر \omterm{def:g11:quad:discriminant}{المميز}'' --- إشارتين متعاكستين
عند $\theta$ وعند $\theta + \frac\pi2$. والأرض \omterm{def:g12:limcont:continuity}{المتصلة}
تجعل $h$ \omterm{def:g12:limcont:continuity}{متصلة}، وتسلّم مبرهنة القيم الوسطى زاوية
$\theta^*$ حيث $h(\theta^*) = 0$: فالأرجل الأربع على الأرض. (ولا
حظّ كهذا مع طاولة مستطيلة أو أرض ذات درجات.)

\textbf{15.} القالب: ابنِ \emph{فرق} المقدارين
اللذين تريد لهما أن يتطابقا ($f(x) - x$؛
و $T(t) - T(t + \pi)$؛ وموضع الصعود ناقص موضع النزول عند
المتنزّهة؛ والفجوة ذات الإشارة عند الطاولة)؛ ثم تحقق من أنه يأخذ قيمًا
بالإشارتين (وغالبًا عبر تناظر بين الطرفين)؛ ثم استدعِ مبرهنة القيم الوسطى
على \omterm{def:g12:limcont:continuity}{دالة متصلة}. فيُفرَض التطابق.

\textbf{16.} السؤالان المفتوحان: \emph{كم} حلًا
(وتجيب عنه الرتابة التامة ومبرهنة التقابل، في
السؤال 7) و\emph{أين} تقع الحلول (ويجيب عنه التنصيف
أو نيوتن، في السؤال 9). فمبرهنة القيم الوسطى عرّافة وجود خالصة.

\textbf{17.} $3 - \frac{7}{x + 2}$: \omterm{def:g12:limcont:asymptote}{فالمقارب} الشاقولي
$x = -2$، والأفقي $y = 3$؛ وعلى كل جهة من $-2$ تكون
\omterm{def:g11:func:monotone}{الدالة متزايدة} (لأن $-\frac{7}{x+2}$ \omterm{def:g12:seq:monotonic}{متزايدة} حيث
تُعرَّف)؛ \omterm{def:g10:functions:graph}{والمنحنى} هو \omterm{def:g11:func:reference}{القطع الزائد} المرجعي
$-\frac7x$ منسحبًا $2$ إلى اليسار و $3$ إلى الأعلى --- فرعان
يعانقان \omterm{def:g10:numbers:interunion}{تقاطع} المقاربين.

\textbf{18.} من أجل $x \neq 1$: $f(x) = x + 1$، إذن
$\lim_{x \to 1} f = 2$: فوضع $f(1) = 2$ يعيد
\omterm{def:g12:limcont:continuity}{الاتصال}. فالانقطاع كان ثقبًا لا قفزة ---
فاملأ النقطة المفقودة الوحيدة يلتئم \omterm{def:g10:functions:graph}{المنحنى}:
انقطاع \emph{قابل للرفع}.

\textbf{19.} \omterm{def:g12:limcont:continuity}{متصلة} على كل \omterm{def:g10:numbers:interval}{فترة} بين عددين صحيحين
متتاليين؛ وعند كل \omterm{def:g10:numbers:sets}{عدد صحيح} $n$ تقفز من $n - 1$ إلى $n$.
ومع $a = 0.9$ و $b = 1.1$: $\lfloor a \rfloor = 0 < \frac12 <
1 = \lfloor b \rfloor$، ومع ذلك $\lfloor x \rfloor = \frac12$ ليس له
حل --- فالقفزة تنتقل آنيًا فوق $\frac12$، وهذا بالضبط
ما تمنع مبرهنة القيم الوسطى الدوال \omterm{def:g12:limcont:continuity}{المتصلة} من فعله.

\textbf{20.} الوجه المحلي: القيمة $=$ النهاية عند كل نقطة ---
وقد خرقه الثقب القابل للرفع (السؤال 18) خرقًا قابلًا للشفاء، وخرقته
قفزات الجزء الصحيح (السؤال 19) خرقًا لا شفاء له. والوجه الشامل: على
\omterm{def:g10:numbers:interval}{فترة}، تُزار كل القيم الوسطى --- وهنا تعيش شهادات
الجذور (6--8)، \omterm{pb:g10:functions:1}{والنقطة الثابتة}، وخط الاستواء، والطاولة،
والمتنزّهة. واسم الشعار الرسمي: \emph{مبرهنة
القيم الوسطى}.
\end{solution}
