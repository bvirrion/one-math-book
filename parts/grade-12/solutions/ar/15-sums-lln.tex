\section*{الفصل \ref{ch:g12:sums} --- مجاميع المتغيرات العشوائية وقانون الأعداد الكبيرة}

\begin{solution}{exo:g12:sums:1}
لكل حجر نرد \omterm{def:g12:randvar:exp}{الأمل الرياضي}
$\frac{1 + 2 + \dots + 6}{6} = \frac{7}{2}$، إذن بالخطية
$\E(S) = \frac72 + \frac72 = 7$. وحجرا النرد مستقلان، إذن
$\V(S) = \frac{35}{12} + \frac{35}{12} = \frac{35}{6} \approx 5.83$.
\end{solution}

\begin{solution}{exo:g12:sums:2}
$\E(X) = 50$ و $\V(X) = 100 \times 0.25 = 25$.

وبماركوف (حيث $X \geq 0$): $\P(X \geq 75) \leq \frac{50}{75} = \frac23$.

وببيينيمي--تشيبيشيف:
$\P(\abs{X - 50} \geq 25) \leq \frac{25}{25^2} = 0.04$ --- وهذا يحصر حتى
\omterm{def:g11:prob:model}{الحادثة} \emph{من الجهتين}، و $X \geq 75$ نصفها. فمتراجحة تشيبيشيف
أحدّ بكثير هنا لأنها تستثمر \omterm{def:g12:randvar:exp}{التباين} لا \omterm{def:g10:stats:mean}{المتوسط} وحده.
(والقيمة الحقيقية للمقدار $\P(X \geq 75)$ أصغر من $10^{-6}$: فالحاصرتان
خامّتان.)
\end{solution}

\begin{solution}{exo:g12:sums:3}
$F_n$ \omterm{def:g10:stats:mean}{متوسط} \omterm{def:g10:proba:sample}{عينة} من $n$ متغير برنولي$\left(\frac12\right)$،
\omterm{def:g12:randvar:exp}{تباين} كل منها $\sigma^2 = \frac14$. والحاصرة
\[
\P\left(\abs{F_n - 0.5} \geq 0.05\right)
\leq \frac{1/4}{n \times 0.05^2} = \frac{100}{n}
\]
تكون $\leq 0.05$ متى كان $n \geq 2000$.
\end{solution}

\begin{solution}{exo:g12:sums:4}
حسب \cref{prop:g12:sums:mean}، يكون $\E(M_n) = 5\%$ (فالتنويع لا
يغيّر المردود المأمول) و
$\sigma(M_n) = \frac{20\%}{\sqrt n}$. واشتراط
$\frac{20}{\sqrt n} < 4$ يعطي $\sqrt n > 5$، \emph{أي} $n \geq 26$.
وهذا هو مبدأ \emph{التنويع}: \omterm{def:g12:randvar:rv}{فتوزيع} رأس المال على
أخطار مستقلة \omterm{def:g12:arith:divides}{يقسم} الخطر (\omterm{def:g11:stat:variance}{الانحراف المعياري}) على $\sqrt n$
دون إنقاص المردود المأمول.
\end{solution}

\begin{solution}{exo:g12:sums:5}
\emph{1.} بعدّ الأزواج $9$ المتساوية \omterm{def:g10:proba:distribution}{الاحتمال}: يأخذ $S$ القيم
$2, 3, 4, 5, 6$ بالاحتمالات
$\frac19, \frac29, \frac39, \frac29, \frac19$. ومنه
$\E(S) = \frac{2 + 6 + 12 + 10 + 6}{9} = 4$ و
$\E(S^2) = \frac{4 + 18 + 48 + 50 + 36}{9} = \frac{156}{9} = \frac{52}{3}$،
إذن $\V(S) = \frac{52}{3} - 16 = \frac43$.

\emph{2.} من أجل متغير واحد: $\E(X) = 2$ و
$\E(X^2) = \frac{1 + 4 + 9}{3} = \frac{14}{3}$ و
$\V(X) = \frac{14}{3} - 4 = \frac23$. ثم $\E(S) = 2 + 2 = 4$، وبالاستقلال
$\V(S) = \frac23 + \frac23 = \frac43$. أي القيمتان نفساهما.
\end{solution}

\begin{solution}{exo:g12:sums:6}
مع $Y = X$: $\V(X + Y) = \V(2X) = 4\V(X)$، بينما
$\V(X) + \V(Y) = 2\V(X)$. ولا يتفقان إلا عندما $\V(X) = 0$، \emph{أي}
عندما يكون $X$ ثابتًا --- فالجمعية تقتضي \omterm{def:g12:condprob:indep}{الاستقلال} فعلًا
(وهنا يكون $X$ مرتبطًا بنفسه أشد ارتباط).
\end{solution}

\begin{solution}{exo:g12:sums:7}
في ظل عدم الغش، يكون $F_n$ \omterm{def:g10:stats:mean}{متوسط} $n = 1200$ متغير برنولي حيث
$p = \frac16$، و $\sigma^2 = \frac16\cdot\frac56 = \frac{5}{36}$:
\[
\P\left(\abs{F_n - \tfrac16} \geq 0.05\right)
\leq \frac{5/36}{1200 \times 0.0025} = \frac{5}{108} \approx 0.046 .
\]
والانحراف الملاحَظ يساوي $0.05$ بالضبط: وهي \omterm{def:g11:prob:model}{حادثة} ينتجها حجر نرد غير مغشوش
\omterm{def:g10:proba:distribution}{باحتمال} لا يتجاوز $4.6\%$ --- وبما أن متراجحة تشيبيشيف متحفظة
جدًا، \omterm{def:g10:proba:distribution}{فالاحتمال} الحقيقي أصغر بكثير. إذن فرضية عدم الغش
غير معقولة؛ والحجر مغشوش على الأرجح.
\end{solution}

\begin{solution}{exo:g12:sums:8}
\emph{1.} $p(1-p) = \frac14 - \left(p - \frac12\right)^2 \leq \frac14$.

\emph{2.} للمتغير $F_n$ \omterm{def:g12:randvar:exp}{الأمل الرياضي} $p$ \omterm{def:g12:randvar:exp}{والتباين}
$\frac{p(1-p)}{n} \leq \frac{1}{4n}$؛ وتعطي متراجحة بيينيمي--تشيبيشيف
\[
\P\bigl(\abs{F_n - p} \geq \varepsilon\bigr)
\leq \frac{p(1-p)}{n\varepsilon^2} \leq \frac{1}{4n\varepsilon^2},
\]
وهي صالحة مهما كان $p$ المجهول.

\emph{3.} مع $\varepsilon = 0.03$ والمستوى $0.05$:
\[
\frac{1}{4n(0.03)^2} \leq 0.05
\iff n \geq \frac{1}{4 \times 0.0009 \times 0.05} \approx 5556 .
\]
فنحو $5600$ شخص يكفون بهذه الحاصرة الخام (والجواب الكلاسيكي
\omterm{def:g10:numbers:approx}{بالتقريب} الطبيعي أقرب إلى $1100$، انظر
\cref{ch:g12:contdist}).
\end{solution}

\begin{solution}{pb:g12:sums:1}
\textbf{1.} $V(X + Y) = V(X) + V(Y) = \frac{35}{12} +
\frac{35}{12} = \frac{35}{6}$ (\omterm{def:g12:condprob:indep}{بالاستقلال}).

\textbf{2.} $\E(\bar X) = 3.5$؛
$\sigma(\bar X) = \frac{1.71}{\sqrt{100}} \approx 0.17$: \omterm{def:g10:stats:mean}{فمتوسط}
المئة حجر يعانق $3.5$ ضمن خُمس نقطة.

\textbf{3.} $\P(X \geq 10) \leq \frac{2}{10} = 0.2$.

\textbf{4.} $\P \leq \frac{V(\bar X)}{0.5^2} =
\frac{35/1200}{0.25} \approx 0.117$: أي نحو
$12\,\%$ على الأكثر.

\textbf{5.} مع $Y = -X$: $V(X + Y) = V(0) = 0$، بينما
$V(X) + V(Y) = 2V(X) > 0$: فالمتغيران المتعاكسان تمامًا
يتلاشيان --- فجمع التباينات امتياز \omterm{def:g12:condprob:indep}{للاستقلال}.

\textbf{6.} $\E = \frac{18}{37} - \frac{19}{37} =
-\frac{1}{37} \approx -0.027$؛
$\sigma = \sqrt{1 - \left(\frac{1}{37}\right)^2} \approx
1.00$.

\textbf{7.} $\E(G) = -2.70$ و $\sigma(G) = 10 \times 1.00 =
10$: فالانزياح ليس إلا $0.27\sigma$. فضجيج الليلة يقزّم
الأفضلية --- وأقلية كبيرة من المقامرين (نحو $40\,\%$،
كما يقول الجرس) يخرجون رابحين، وهذا بالضبط ما يبقيهم
عائدين.

\textbf{8.} حصيلة دار القمار: \omterm{def:g12:randvar:exp}{الأمل الرياضي} $+27\,027$ أورو،
\omterm{def:g11:stat:variance}{والانحراف المعياري} $\approx 1\,000$: فالربح يجلس على بعد $27$
انحرافًا معياريًا فوق الصفر. وعند $27\sigma$، لا تكون عبارة ``قد تخسر
دار القمار هذه السنة'' خطرًا، بل خطأ \omterm{def:g10:numbers:approx}{تقريب}.

\textbf{9.} $\P(\text{خسارة}) = \P(G \geq 27\,027$ من أجل
المقامرين$)$ $\leq \frac{10^6 \times 1}{27\,027^2} \approx
0.0014$: فحتى أفدح متراجحة في الكتاب تضمن
لبيت اللعب $99.86\,\%$ --- والحقيقة أقوى
بصورة فلكية.

\textbf{10.} كل أورو يُراهن به، متى وكيفما اختير،
له مردود مأمول قدره $-\frac{1}{37}$ من نفسه؛ وبالخطية
يكون الربح الكلي المأمول $-\frac{1}{37} \times
(\text{مجموع الرهان})$، وهو سالب من أجل كل استراتيجية
تراهن بشيء. والنظام الرابح سينتهك خطية
\omterm{def:g12:randvar:exp}{الأمل الرياضي} --- فلا \omterm{def:g10:coordgeom:system}{ترتيب} بارع لرهانات سيئة يصنع
رهانًا جيدًا. (وأنظمة المضاعفة لا تفعل إلا أن تستبدل بمكاسب صغيرة كثيرة
خسائر كارثية نادرة.)

\textbf{11.} المبرهنة: \emph{\omterm{def:g10:stats:series}{تواتر}} الأحمر على
$n$ دورة يتقارب (احتماليًا) إلى $\frac{18}{37}$. وهي
لا تقول شيئًا عن الدورة $n + 1$: فالعجلة بلا ذاكرة، و
بعد عشرة أحمر يظل \omterm{def:g10:proba:distribution}{احتمال} الأحمر $\frac{18}{37}$.
والاعتقاد بغير ذلك هو \emph{مغالطة المقامر}، و
السؤال 12 يبيّن ما يحدث فعلًا للسلاسل.

\textbf{12.} فائض $100$ أحمر لا يُردّ ---
فالدورات المقبلة نسخ عادلة، والفائض المأمول يبقى $100$.
لكن $\frac{100}{10^6} = 0.0001$: أي جزءًا من مئة من النقطة.
وخطأ المغالطة في جملة واحدة: \emph{قانون الأعداد
الكبيرة يخفّف حوادث الماضي في محيط من التجارب الجديدة؛ وهو
لا يرسل العجلة أبدًا لتحصيل ديون.}

\textbf{13.} $n \geq \frac{1}{4 \times 0.05 \times 0.03^2}
\approx 5\,556$ شخصًا.

\textbf{14.} $\frac{1.96^2 \times 0.25}{0.03^2} \approx
1\,067$: أي أقل بخمس مرات. فمتراجحة تشيبيشيف تصح من أجل \emph{كل}
\omterm{def:g12:randvar:rv}{توزيع} وتدفع ثمن كونيتها ارتخاءً؛ أما ثابت
المستطلعين $1.96$ فيأتي من الشكل الجرسي الفعلي
للتذبذبات، وهو يتركّز أشد بكثير.

\textbf{15.} الهامش $\propto \frac{1}{\sqrt n}$: فتنصيفه
يربّع \omterm{def:g10:proba:sample}{العينة} --- أي نحو $4\,400$ شخص من أجل
$\pm 1.5$ نقطة. فالدقة تُشترى بأسعار تربيعية.

\textbf{16.} النموذج هو $n$ سحبة مستقلة \omterm{def:g10:proba:distribution}{باحتمال} نجاح
$p$ --- ولا يظهر حجم المجتمع $N$ أبدًا
(فمعاينة جزء ضئيل من مجتمع كبير جيد الخلط
هي ما يشفّره \omterm{def:g12:condprob:indep}{الاستقلال}). وتشبيه المستطلعين: لتذوق
الحساء، تكفي ملعقة واحدة بعد تقليب جيد ---
سواء أكانت القدر تكفي عشرة أم عشرة آلاف. والمصيدة هي
التقليب --- أي إطار معاينة منحاز، كما في استطلاع الكتاب
السابق الذي خدع بلدًا --- لا القدر.

\textbf{17.} $315\sqrt n < 30n \iff \sqrt n > 10.5 \iff
n > 110$: فبعد مئة عقد يسبق الانزياح
الضجيج، وكل عقد إضافي يوسّع الفجوة بقانون
$\sqrt n$ --- فالتجميع \emph{هو} نموذج العمل.

\textbf{18.} حصص متساوية مستقلة: $\sigma_{\text{محفظة}} =
\frac{20\,\%}{\sqrt{25}} = 4\,\%$: أي المردود المأمول نفسه، و
خُمس التذبذب. والغداء: تقليص الخطر بلا كلفة
في \omterm{def:g12:randvar:exp}{الأمل الرياضي} --- بأخذ \omterm{def:g10:stats:mean}{متوسط} عشوائيات مستقلة.

\textbf{19.} وعند الترابط التام: تكون المحفظة سهمًا واحدًا
في $25$ ثوبًا: $\sigma = 20\,\%$، فلا تقليص أصلًا.
فالتنويع يستأجر \emph{\omterm{def:g12:condprob:indep}{الاستقلال}}؛ وحين تربط أزمةٌ كلَّ شيء
(2008: كانت كل رهانات الإسكان رهانًا واحدًا)،
تتبخر حماية $\sqrt n$ بالضبط حين تكون
مطلوبة.

\textbf{20.} المجاميع تنزاح مثل $n$ وتتذبذب مثل
$\sqrt n$، فتستقر المتوسطات مثل $\frac{1}{\sqrt n}$: فتصرف
دار القمار الانزياح على مليون دورة؛ ويشتري المستطلع
$\frac{1}{\sqrt{1100}}$ من الضجيج بميزانية هاتف؛ و
يتجاوز المؤمّن مطالباته عند $n > 110$؛ و
\omterm{def:g12:arith:divides}{يقسم} المستثمر الخطر على $\sqrt{25}$. وأما المعتديان: فالمقامر
الذي يؤمن بالديون (وليس ثمة إلا تخفيف)، و
المالي الذي يؤمن \omterm{def:g12:condprob:indep}{بالاستقلال} (وليس ثمة أحيانًا
إلا رهان واحد). والشكل الكوني للتذبذبات ---
أي الجرس --- هو نجم الفصل المتصل، ومبرهنته،
مبرهنة النهاية المركزية، هي قمة احتمالات الكتب
الجامعية.
\end{solution}
