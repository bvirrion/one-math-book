\section*{الفصل \ref{ch:g12:arith} --- الحساب}

\begin{solution}{exo:g12:arith:1}
$17 \times 119 = 2023$، إذن $2026 = 17 \times 119 + 3$: فخارج القسمة $119$
والباقي $3$. وأما من أجل $-2026$: $-2026 = 17\times(-120) + 14$ (وفعلًا
$17 \times 120 = 2040$ و $2040 - 2026 = 14$): فخارج القسمة $-120$ والباقي
$14$ (إذ يجب أن يقع الباقي في $\intco{0}{17}$، فهو \emph{ليس} $-3$).
\end{solution}

\begin{solution}{exo:g12:arith:2}
بترديد $10$: $7^2 = 49 \equiv 9 \equiv -1$. ومنه
$7^{100} = \left(7^2\right)^{50} \equiv (-1)^{50} = 1 \pmod{10}$: فآخر رقم
في $7^{100}$ هو $1$.
\end{solution}

\begin{solution}{exo:g12:arith:3}
\omterm{prop:g12:arith:euclidalgo}{بخوارزمية إقليدس}: $1071 = 2\times462 + 147$؛ و $462 = 3\times147 + 21$؛
و $147 = 7\times21 + 0$. إذن $\gcd = 21$.

وبالتعويض بالمقلوب: $21 = 462 - 3\times147 = 462 - 3(1071 - 2\times462)
= 7\times462 - 3\times1071$. ومنه $u = -3$ و $v = 7$:
$1071\times(-3) + 462\times7 = 21$.
\end{solution}

\begin{solution}{exo:g12:arith:4}
كل \omterm{def:g10:numbers:sets}{عدد صحيح} $\equiv 0, 1, 2$ أو $3 \pmod 4$، وبالتربيع:
$0^2 \equiv 0$ و $1^2 \equiv 1$ و $2^2 = 4 \equiv 0$ و $3^2 = 9 \equiv 1$. إذن
$n^2 \equiv 0$ أو $1 \pmod 4$. ومجموع مربعين يوافق عندئذٍ
$0 + 0$ أو $0 + 1$ أو $1 + 1$، \emph{أي} $0$ أو $1$ أو $2 \pmod 4$ ---
ولا يوافق $3$ أبدًا.
\end{solution}

\begin{solution}{exo:g12:arith:5}
\omterm{def:g12:arith:divides}{قابلية القسمة} على $2$: من بين $n$ و $n + 1$، واحد زوجي. \omterm{def:g12:arith:divides}{وقابلية القسمة} على
$3$: إذا كان $n \equiv 0$، فإن $3 \mid n$؛ وإذا كان $n \equiv 1 \pmod 3$، فإن
$2n + 1 \equiv 3 \equiv 0$؛ وإذا كان $n \equiv 2$، فإن $n + 1 \equiv 0$. وفي كل
الحالات \omterm{def:g12:arith:divides}{يقسم} $3$ الجداء. وبما أن $\gcd(2,3) = 1$، تعطي
\cref{cor:g12:arith:coprimeprod} أن $6 \mid n(n+1)(2n+1)$.
(وهذا يعيد أيضًا البرهان على أن $\frac{n(n+1)(2n+1)}{6}$، وهو مجموع مربعات
\cref{exo:g12:seq:1}، \omterm{def:g10:numbers:sets}{عدد صحيح}.)
\end{solution}

\begin{solution}{exo:g12:arith:6}
نبحث عن مقلوب $5$ بترديد $11$: فبالتجريب (أو بمساواة بيزو)،
$5 \times 9 = 45 = 44 + 1 \equiv 1 \pmod{11}$. وبضرب \omterm{def:g12:arith:congruence}{الموافقة} في
$9$:
\[
x \equiv 9 \times 3 = 27 \equiv 5 \pmod{11}.
\]
والحلول هي \omterm{def:g10:numbers:sets}{الأعداد الصحيحة} $x = 5 + 11k$ حيث $k \in \Z$. (وللتحقق:
$5\times5 = 25 \equiv 3 \pmod{11}$.)
\end{solution}

\begin{solution}{exo:g12:arith:7}
$\gcd(17, 40) = 1$، إذن توجد حلول. وبخوارزمية إقليدس:
$40 = 2\times17 + 6$؛ و $17 = 2\times6 + 5$؛ و $6 = 5 + 1$. وبالتعويض بالمقلوب:
$1 = 6 - 5 = 6 - (17 - 2\times6) = 3\times6 - 17
= 3(40 - 2\times17) - 17 = 3\times40 - 7\times17$. ومنه
$17\times(-7) - 40\times(-3) = 1$: أي الحل الخاص
$(x_0, y_0) = (-7, -3)$.

والحل العام \omterm{def:g10:algebra:equation}{للمعادلة} $17x - 40y = 1$: بطرح العلاقة الخاصة،
$17(x + 7) = 40(y + 3)$؛ وبما أن $\gcd(17, 40) = 1$، تعطي مبرهنة غاوس المساعدة
أن $40 \mid x + 7$، إذن $x = -7 + 40k$ ثم $y = -3 + 17k$ حيث $k \in \Z$
(وكلها تتحقق).

وأما من أجل $17x - 40y = 6$، فاضرب الحل الخاص في $6$:
$(x_1, y_1) = (-42, -18)$، ويعطي الاستدلال نفسه
\[
x = -42 + 40k, \qquad y = -18 + 17k, \qquad k \in \Z .
\]
(فمثلًا $k = 2$: $x = 38$ و $y = 16$؛ وفعلًا $17\times38 - 40\times16
= 646 - 640 = 6$.)
\end{solution}

\begin{solution}{exo:g12:arith:8}
لنفترض $\sqrt2 = \frac ab$ حيث $a, b \in \N^*$؛ عندئذٍ $a^2 = 2b^2$. وفي
تحليل مربع إلى عوامل أولية، يكون كل أس زوجيًا؛ إذن أس
$2$ في $a^2$ زوجي، بينما هو فردي في $2b^2$ (فهو أكبر بواحد من عدد
زوجي). وتحليلان للعدد نفسه بأسّين مختلفين للعدد
$2$ يناقضان الوحدانية في \cref{thm:g12:arith:fta}. ومنه لا يوجد
كسر كهذا: $\sqrt2 \notin \Q$.
\end{solution}

\begin{solution}{exo:g12:arith:9}
\emph{1.} من $k\binom pk = p \binom{p-1}{k-1}$، \omterm{def:g12:arith:divides}{يقسم} $p$ المقدار
$k\binom pk$. ومن أجل $1 \leq k \leq p-1$، يكون $p \nmid k$ ويعطي كون $p$ \omterm{def:g12:arith:prime}{أوليًا}
أن $\gcd(p, k) = 1$، إذن تعطي مبرهنة غاوس المساعدة أن $p \mid \binom pk$.

\emph{2.} بالتراجع على $a$. من أجل $a = 0$: $0^p \equiv 0$. ولنفترض
$a^p \equiv a \pmod p$. فبمبرهنة ثنائي الحد،
\[
(a+1)^p = \sum_{k=0}^{p} \binom pk a^k
\equiv a^p + 1 \pmod p,
\]
إذ تنعدم كل الحدود الوسطى بترديد $p$ حسب النقطة 1. وبفرضية
التراجع، $(a+1)^p \equiv a + 1 \pmod p$. وهذا يبرهن على أن
$a^p \equiv a$ من أجل كل $a \in \N$، وتنتج الحالة $a < 0$ بكتابة
$a \equiv a + kp$ من أجل ممثل موجب مناسب.
\end{solution}

\begin{solution}{exo:g12:arith:10}
$n \equiv 2 \pmod 3$ و $n \equiv 3 \pmod 5$: اكتب $n = 2 + 3s$؛ عندئذٍ
$2 + 3s \equiv 3 \pmod 5$، \emph{أي} $3s \equiv 1 \pmod 5$. ومقلوب
$3$ بترديد $5$ هو $2$ (لأن $3\times2 = 6 \equiv 1$)، إذن $s \equiv 2 \pmod 5$،
وليكن $s = 2 + 5t$، فيكون $n = 8 + 15t$: أي إن الشرطين الأولين يعنيان
$n \equiv 8 \pmod{15}$.

وبإضافة $n \equiv 2 \pmod 7$: $8 + 15t \equiv 2 \pmod 7$، و
$15 \equiv 1 \pmod 7$، إذن $t \equiv -6 \equiv 1 \pmod 7$، وليكن
$t = 1 + 7u$. ومنه $n = 23 + 105u$:
\[
n \equiv 23 \pmod{105}.
\]
(وللتحقق: $23 = 3\times7 + 2 = 5\times4 + 3 = 7\times3 + 2$.)
\end{solution}

\begin{solution}{pb:g12:arith:1}
\textbf{1.} $2026 = 289 \times 7 + 3$: $2026 \equiv 3
\pmod 7$. وقوى $7$ بترديد $10$: $7, 9, 3, 1$، ودورتها
طولها $4$؛ و $100 \equiv 0 \pmod 4$: فآخر رقم في
$7^{100}$ هو $1$.

\textbf{2.} $5^2 = 25 \equiv -1 \pmod{13}$، إذن
$5^{116} = \left(5^2\right)^{58} \equiv (-1)^{58} = 1$ و
$5^{117} \equiv 5 \pmod{13}$.

\textbf{3.} مقلوب $3$ بترديد $7$ هو $5$
(لأن $15 \equiv 1$): $x \equiv 5 \times 5 = 25 \equiv 4 \pmod 7$.

\textbf{4.} $97 = 2 \times 35 + 27$؛ و $35 = 27 + 8$؛
و $27 = 3 \times 8 + 3$؛ و $8 = 2 \times 3 + 2$؛ و $3 = 2 + 1$.
وبالتعويض بالمقلوب: $1 = 97 \times 13 + 35 \times (-36)$. إذن
$35 \times (-36) \equiv 1 \pmod{97}$: فمقلوب $35$ هو
$-36 \equiv 61 \pmod{97}$.

\textbf{5.} يكون $a$ قابلًا للقلب بترديد $n$ بالضبط عندما يكون
$\gcd(a, n) = 1$: فمساواة بيزو تعطي $au + nv = 1$، أي
$au \equiv 1$؛ وبالعكس، وجود مقلوب يفرض أن \omterm{def:g12:arith:divides}{يقسم} \omterm{def:g12:arith:gcd}{القاسم المشترك الأكبر}
$1$.

\textbf{6.} $0{\cdot}10 + 3{\cdot}9 + 0{\cdot}8 + 6{\cdot}7 +
4{\cdot}6 + 0{\cdot}5 + 6{\cdot}4 + 1{\cdot}3 + 5{\cdot}2 +
2{\cdot}1 = 132 = 12 \times 11 \equiv 0 \pmod{11}$: صحيحة.

\textbf{7.} يتغيّر المجموع بالمقدار $wd$ حيث
$1 \leq w \leq 10$ و $1 \leq \abs d \leq 9$: وبما أن $11$
أولي ولا \omterm{def:g12:arith:divides}{يقسم} أيًا من العاملين، فلا يمكنه أن \omterm{def:g12:arith:divides}{يقسم} الجداء
(\cref{thm:g12:arith:gauss} / \cref{prop:g12:arith:primedivides}):
فلا يعود المجموع المتغيّر $\equiv 0$ أبدًا: أي إن كل خطأ في رقم
واحد يشغّل الإنذار.

\textbf{8.} تبادل رقمين متجاورين $a, b$ (وزناهما
$w + 1, w$) يغيّر المجموع بالمقدار
$(w+1)b + wa - (w+1)a - wb = b - a \not\equiv 0$ من أجل
$a \neq b$: فيُكشف. والسرّ هو \emph{أولية}
العدد $11$: فبترديد $10$، تنعدم جداءات مثل $5 \times 2$ مع أن
كلا العاملين غير معدوم، فقد ينزلق خطأ قدره $\pm 2$ عند الوزن $5$ (أو
تبادل غير محظوظ) دون أن يُكشف.

\textbf{9.} المجموع المرجَّح للأرقام الاثني عشر: $119$؛ و
يجب أن يكمله رقم التحقق إلى مضاعف للعدد $10$: أي $1$ (والشيفرة
الكاملة $978\,2940199\,051$). ويفوت الترميز EAN التبادلات المتجاورة
التي فيها $2(a - b) \equiv 0 \pmod{10}$، أي
$\abs{a - b} = 5$: فتبادل $2$ و $7$ مثلًا يمر
دون أن يُرى --- وهو ثمن الترديد الودود $10$.

\textbf{10.} بمضاعفة كل رقم ثانٍ من اليمين وطيّه
($16 \to 7$، وهكذا)، يبلغ المجموع $80 \equiv 0
\pmod{10}$: فتُقبل البطاقة الاختبارية.

\textbf{11.} مع ترديد أولي يكون كل وزن قابلًا للقلب،
فتُكشف \emph{كل} الأخطاء المفردة و\emph{كل} التبادلات
المتجاورة --- وهي رفاهية ISBN؛ أما مخططات الترديد $10$
فتحتفظ بأرقام ودودة للإنسان وتقبل بقعة عمياء قصيرة.

\textbf{12.} $2^{10} = 1024 = 3 \times 341 + 1 \equiv 1
\pmod{341}$، ومنه $2^{340} = \left(2^{10}\right)^{34} \equiv
1$. ومع ذلك فإن $341 = 11 \times 31$ مركّب: فهو يجتاز اختبار
فيرما عند \omterm{def:g11:seq:arithmetic}{الأساس} $2$ وهو ليس \omterm{def:g12:arith:prime}{أوليًا}. والعبرة: أن \omterm{def:g12:arith:congruence}{موافقة}
فيرما لازمة لا كافية --- فاختبار الأولية
يحتاج إلى أدوات أحدّ (وينالها، في الكتب
الجامعية).

\textbf{13.} $3d \equiv 1 \pmod{20}$: أي $d = 7$
(لأن $21 = 20 + 1$).

\textbf{14.} $c = 4^3 = 64 \equiv 31 \pmod{33}$.

\textbf{15.} $31 \equiv -2$: فإن $(-2)^7 = -128$، و
$-128 + 4 \times 33 = 4$: فيفكّ النص المعمّى إلى
$m = 4$. والقفل يدور.

\textbf{16.} بترديد $3$: إذا كان $3 \nmid m$، فإن
$m^2 \equiv 1$ (بفيرما)، إذن
$m^{21} = m \cdot \left(m^2\right)^{10} \equiv m$؛ وإذا كان
$3 \mid m$، فالطرفان $\equiv 0$. وبترديد $11$:
$m^{10} \equiv 1$ أو $11 \mid m$، و
$m^{21} = m \cdot \left(m^{10}\right)^2 \equiv m$. \omterm{def:g12:arith:divides}{فيقسم} كل من $3$
و $11$ المقدار $m^{21} - m$، وبما أنهما \omterm{def:g12:arith:gcd}{أوليان فيما بينهما} فإن جداءهما
$33$ يقسمه أيضًا (بمبرهنة غاوس المساعدة): $m^{21} \equiv m \pmod{33}$. وقد بُني
الأس $ed = 21 = 1 + 20k$ بحيث يختفي أسّا فيرما
($2$ و $10$، وكلاهما \omterm{def:g12:arith:divides}{يقسم} $20$).

\textbf{17.} ضرب عددين أوليين طول كل منهما $300$ رقمًا يستغرق
ميكروثانية؛ أما استعادتهما من جداءهما فتهزم كل
خوارزمية معروفة وكل حواسيب العالم --- فالقفل
طريق باتجاه واحد. (وعددنا $n = 33$ هو الطريق \omterm{def:g12:complex:modulus}{بمقياس} اللعبة،
يُمشى في الاتجاهين.)

\textbf{18.} باختبار البواقي (أو بالبناء بمساواة بيزو):
$n \equiv 8 \pmod{15}$: فالأعداد $8, 23, 38, 53, \dots$
والوحدانية بترديد $15$: فحلان يختلفان بمضاعف
للعدد $3$ وللعدد $5$، ومنه للعدد $15$ (لأن $3$ و $5$ \omterm{def:g12:arith:gcd}{أوليان فيما بينهما}، بغاوس).
والقائد الذي معه $1000$ جندي يعلن ``$8$'' بثلاثة
اصطفافات سريعة --- وهي حيلة عدّ الرؤوس القديمة.

\textbf{19.} $10 \equiv 1 \pmod 9$ يعطي $10^k \equiv 1$، إذن
$\sum d_k 10^k \equiv \sum d_k$: أي إن العدد ومجموع أرقامه
متوافقان بترديد $9$ (وبترديد $3$). و
$10 \equiv -1 \pmod{11}$ يعطي
$\sum d_k 10^k \equiv \sum (-1)^k d_k$: وهي القاعدة المتناوبة.
قاعدتان من الطفولة، سطر واحد لكل منهما.

\textbf{20.} حوّلت الموافقات البواقي إلى حساب
(الجزء الأول)؛ وسكّت مساواة بيزو المقلوبات التي تحل الموافقات
الخطية وتعطي $d$ في نظام RSA (السؤالان 4 و 13)؛ وفتحت مبرهنة فيرما
الصغرى القفل وأغلقته (السؤالان 15--16)؛ ولصق الترديدين الأوليين
فيما بينهما عدّ الجنود وأتمّ البرهان
(السؤالان 16 و 18). والحكم على هاردي: فأنقى مبرهنة
عرفها صارت اليوم تحرس كل شراء --- فالنقاء، مع الوقت،
هو أكثر ما يقبل التطبيق.
\end{solution}
