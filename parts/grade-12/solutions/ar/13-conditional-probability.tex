\section*{الفصل \ref{ch:g12:condprob} --- الاحتمال الشرطي والاستقلال}

\begin{solution}{exo:g12:condprob:1}
توجد $12$ ورقة صورة، منها $4$ ملوك:
$\pcond{\text{صورة}}{\text{ملك}} = \frac{4}{12} = \frac13$.

وأما \omterm{def:g12:condprob:indep}{الاستقلال}: $\P(\text{ملك} \cap \text{قلب}) = \frac{1}{52}$ (وهو ملك
القلوب)، و
$\P(\text{ملك})\,\P(\text{قلب}) = \frac{4}{52}\times\frac{13}{52}
= \frac{1}{52}$. فهما متساويان: إذن الحادثتان مستقلتان.
\end{solution}

\begin{solution}{exo:g12:condprob:2}
\emph{1.} الفرع الأول: حمراء $\frac58$، وزرقاء $\frac38$؛ والفروع الثانية
(بلا إرجاع): بعد الحمراء، حمراء $\frac47$ / زرقاء $\frac37$؛ وبعد
الزرقاء، حمراء $\frac57$ / زرقاء $\frac27$.

\emph{2.} $\P(RR) = \frac58 \times \frac47 = \frac{5}{14}$. وبالاحتمالات
الكلية،
\[
\P(\text{الثانية حمراء}) = \frac58\cdot\frac47 + \frac38\cdot\frac57
= \frac{20 + 15}{56} = \frac58 .
\]
(وهو المقدار نفسه كما في السحبة الأولى --- فبالتناظر، الكرة الثانية كرة
عشوائية منتظمة من الكيس.)
\end{solution}

\begin{solution}{exo:g12:condprob:3}
$\P(A) = \frac{6}{36} = \frac16$ (فستة أزواج مجموعها $7$)، و
$\P(B) = \frac16$، و $A \cap B = \{(3,4)\}$ احتمالها
$\frac{1}{36} = \P(A)\P(B)$: إذن $A$ و $B$ مستقلتان. (فالمجموع $7$
خاص: إذ مهما كان الحجر الأول، توجد قيمة واحدة بالضبط للثاني تعطيه.)

و $\P(C) = \frac{5}{36}$ و $B \cap C = \{(3,3)\}$:
$\P(B \cap C) = \frac{1}{36} \neq \frac16 \times \frac{5}{36}
= \frac{5}{216}$. إذن ليستا \omterm{def:g12:condprob:indep}{مستقلتين}.
\end{solution}

\begin{solution}{exo:g12:condprob:4}
\emph{1.} بالاحتمالات الكلية مع التجزئة حسب الآلة:
\[
\P(D) = 0.5\times0.01 + 0.3\times0.02 + 0.2\times0.04
= 0.005 + 0.006 + 0.008 = 0.019 = 1.9\% .
\]

\emph{2.} وببايز:
$\pcond{D}{M_3} = \dfrac{0.008}{0.019} = \dfrac{8}{19} \approx 0.42$. فالآلة
التي تنتج خُمس الإنتاج فقط مسؤولة عن أكثر من $40\%$ من
العيوب.
\end{solution}

\begin{solution}{exo:g12:condprob:5}
الشرط هو
\[
\frac{0.99\,p}{0.99\,p + 0.05\,(1-p)} \geq 0.9 .
\]
والمقام موجب، فيُقرأ هذا
$0.99p \geq 0.891p + 0.045 - 0.045p$، \emph{أي} $0.144\,p \geq 0.045$،
\emph{أي}
\[
p \geq \frac{0.045}{0.144} = 0.3125 .
\]
فلا يبلغ الاختبار وحده يقين $90\%$ إلا إذا كان المرض يصيب أصلًا
أكثر من $31\%$ من المجموعة المفحوصة --- ولهذا يقتضي المسح الشامل
للأمراض النادرة اختبارات تأكيدية.
\end{solution}

\begin{solution}{exo:g12:condprob:6}
\emph{1.} ثلاث متتاليات تحقق $E$ (صص ظ، ص ظ ص، ظ صص)، \omterm{def:g10:proba:distribution}{احتمال} كل منها
$p^2(1-p)$ \omterm{def:g12:condprob:indep}{بالاستقلال}: $\P(E) = 3p^2(1 - p)$.

\emph{2.} $f(p) = 3p^2 - 3p^3$ لها $f'(p) = 6p - 9p^2 = 3p(2 - 3p)$،
وهي موجبة ثم سالبة على $\intoo{0}{1}$: \omterm{def:g10:functions:extrema}{فالقيمة العظمى} عند $p = \frac23$، حيث
$\P(E) = 3\cdot\frac49\cdot\frac13 = \frac49$.
\end{solution}

\begin{solution}{exo:g12:condprob:7}
لنقل إنك اخترت الباب 1 وفتح المقدّم الباب 3 (وهي \omterm{def:g11:prob:model}{الحادثة} $H_3$). ومع
$B_i$ = ``الجائزة خلف الباب $i$''، يكون $\P(B_i) = \frac13$ و
\[
\pcond{B_1}{H_3} = \frac12, \qquad
\pcond{B_2}{H_3} = 1, \qquad
\pcond{B_3}{H_3} = 0
\]
(فإذا كانت الجائزة خلف بابك، اختار المقدّم بين البابين 2 و 3
عشوائيًا؛ وإذا كانت خلف الباب 2، اضطر إلى فتح الباب 3). وببايز:
\[
\pcond{H_3}{B_1}
= \frac{\frac13\cdot\frac12}{\frac13\cdot\frac12 + \frac13\cdot1 + 0}
= \frac{1/6}{1/2} = \frac13,
\qquad
\pcond{H_3}{B_2} = \frac{\frac13}{\frac12} = \frac23 .
\]
فالبقاء يربح \omterm{def:g10:proba:distribution}{باحتمال} $\frac13$، والتبديل \omterm{def:g10:proba:distribution}{باحتمال}
$\frac23$.
\end{solution}

\begin{solution}{exo:g12:condprob:8}
\emph{1.} تخطئ الأغلبية عندما تُقلب نسختان على الأقل من النسخ
الثلاث:
\[
3\varepsilon^2(1-\varepsilon) + \varepsilon^3
= 3(0.01)(0.9) + 0.001 = 0.028,
\]
وهو أصغر بكثير من $\varepsilon = 0.1$: فالترميز التكراري يعمل.

\emph{2.} إذا أُرسل $1$ (على صورة $111$)، فاستقبال $101$ يقتضي قلبة واحدة
بالضبط: \omterm{def:g10:proba:distribution}{باحتمال} $\varepsilon(1-\varepsilon)^2 = 0.081$. وإذا أُرسل $0$
(على صورة $000$)، لزمت قلبتان: $\varepsilon^2(1-\varepsilon) = 0.009$.
وببايز مع قبْليين متساويين:
\[
\pcond{101}{\,\text{أُرسل }1} = \frac{0.081}{0.081 + 0.009} = 0.9 .
\]
ففكّ الترميز بالأغلبية ($101 \mapsto 1$) هو التخمين الأرجح فعلًا.
\end{solution}

\begin{solution}{pb:g12:condprob:1}
\textbf{1.} $\P(\text{حمراء}) = \frac12 \cdot \frac23 + \frac12
\cdot \frac14 = \frac13 + \frac18 = \frac{11}{24}$.

\textbf{2.} $\P(\text{I} \mid \text{حمراء}) =
\dfrac{\frac12 \cdot \frac23}{\frac{11}{24}} =
\dfrac{1/3}{11/24} = \dfrac{8}{11}$.

\textbf{3.} $\P(\text{زوجي} \cap \leq 4) = \P(\{2, 4\}) =
\frac13$ و $\P(\text{زوجي})\P(\leq 4) = \frac12 \cdot
\frac23 = \frac13$: إذن مستقلتان (على غير المتوقع!). ومن أجل
$\leq 3$: $\P(\{2\}) = \frac16$ في مواجهة
$\frac12 \cdot \frac12 = \frac14$: إذن مرتبطتان.

\textbf{4.} متممتا الحادثتين
\omterm{def:g12:condprob:indep}{المستقلتين}
مستقلتان هما أيضًا، إذن
\[
\P(\text{لا شيء}) = 0.1 \times 0.1 = 0.01,
\qquad
\P(\text{واحد على الأقل}) = 0.99 .
\]

\textbf{5.} $\frac{4}{52} \times \frac{3}{51} =
\frac{1}{221}$.

\textbf{6.} المرضى: $100$؛ والأصحاء: $99\,900$. والإيجابيات:
$99$ صحيحة (99\,\% من $100$) و $4\,995$ كاذبة (5\,\% من
$99\,900$): أي $5\,094$ إيجابيًا في المجموع.

\textbf{7.} $\P(\text{مريض} \mid +) = \frac{99}{5\,094}
\approx 1.9\,\%$: فمن بين من يؤشّر عليهم الاختبار، أقل من
واحد من كل خمسين مريض فعلًا.

\textbf{8.} $\dfrac{0.001 \times 0.99}{0.001 \times 0.99 +
0.999 \times 0.05} = \dfrac{0.00099}{0.05094} \approx 0.019$:
فالأعداد والصيغة متفقتان.

\textbf{9.} الخلط: بين $\P(+ \mid \text{مريض}) = 99\,\%$
(وهي جودة الاختبار المعلَنة) و
$\P(\text{مريض} \mid +)$ (وهو سؤال المريض الفعلي). ومع
معدل قاعدي قدره $\frac{1}{1000}$، يكون المرضى نادرين إلى حدّ أن
تسرّبًا قدره $5\,\%$ من الأغلبية الصحيحة الضخمة يغرق
الإيجابيات الصحيحة خمسين إلى واحد.

\textbf{10.} $\dfrac{0.019 \times 0.99}{0.019 \times 0.99 +
0.981 \times 0.05} \approx 28\,\%$. فكل إيجابي مستقل
يضرب الحظوظ في المعامل نفسه: فالأدلة تتراكم،
وإيجابي ثالث سيدفعها فوق $90\,\%$ --- ولهذا
توجد الاختبارات التأكيدية.

\textbf{11.} $\dfrac{0.05 \times 0.99}{0.05 \times 0.99 +
0.95 \times 0.05} \approx 51\,\%$: فعلى مجموعة عالية الخطر يكون
الاختبار نفسه مفيدًا. والعبرة: امسح حيث يكون المعدل القاعدي
كبيرًا؛ أما المسح الشامل لحالة نادرة فيصنع
إنذارات كاذبة بالآلاف.

\textbf{12.} العدد المأمول للتطابقات البريئة:
$10^7 \times 10^{-6} = 10$. ومن بين نحو $11$ شخصًا
مطابقًا ($10$ أبرياء و $1$ مذنب --- إذا كان الجاني في
المدينة)، يكون المشتبه به الذي وُجد \emph{بالتطابق وحده}
بريئًا \omterm{def:g10:proba:distribution}{باحتمال} نحو $\frac{10}{11} \approx
91\,\%$.

\textbf{13.} اقتبس المدّعي العام
$\P(\text{تطابق} \mid \text{بريء}) = 10^{-6}$؛ أما المحكمة فاحتاجت
إلى $\P(\text{بريء} \mid \text{تطابق})$ --- وهو خلط السؤال
9 تحت القسم. (وفي القضية الحقيقية التي شهّرت هذه الأخطاء،
ضاعفت غلطة ثانية الأمر:
انظر السؤال 14.)

\textbf{14.} التربيع الساذج: $\left(\frac{1}{8500}\right)^2
\approx \frac{1}{72 \text{ مليون}}$. ومع كون المأساة
الثانية أرجح خمس مرات علمًا بالأولى:
$\frac{1}{8500} \times \frac{5}{8500} \approx
\frac{1}{14.5 \text{ مليون}}$ --- أي أكبر بخمس مرات، ومع ذلك
ما زال يتجاهل أن الفرضية \emph{البديلة} يجب أن
تُوزَن ببايز أيضًا. والدرس: أن ضرب الاحتمالات
يقتضي \omterm{def:g12:condprob:indep}{الاستقلال}، وأن الأسباب المشتركة (الوراثة و
البيئة) تدمّره.

\textbf{15.} تضرب صيغة بايز نسب الأرجحية من أجل \emph{كل}
الأدلة في بعدي واحد: التطابق، و
الحجة الغيابية، والألياف، وكل منها يزيح الحظوظ. وجملة المحلَّف:
\emph{اسأل ما \omterm{def:g10:proba:distribution}{احتمال} \omterm{pb:g11:quad:1}{الدليل} في ظل كلتا
الفرضيتين، ولا تخلط أبدًا بين \omterm{def:g12:condprob:cond}{الاحتمال الشرطي} وعكسه}.

\textbf{16.} $\dfrac{0.4 \times 0.15}{0.4 \times 0.15 +
0.6 \times 0.005} = \dfrac{0.060}{0.063} \approx 95\,\%$
مزعج: فكلمة واحدة تكاد تحسم الأمر.

\textbf{17.} تفشل النسخ الخمس عندما تُقلب $3$ أو $4$ أو $5$:
$\binom53 \varepsilon^3(1-\varepsilon)^2 + \binom54
\varepsilon^4 (1 - \varepsilon) + \varepsilon^5 \approx
0.0086$ --- في مواجهة $0.028$ من أجل ثلاث نسخ: فالتكرار
يشتري الموثوقية بثمن عرض الحزمة.

\textbf{18.} القلبات \omterm{def:g11:func:parity}{الزوجية}: $\binom40 0.9^4 + \binom42 0.9^2
0.1^2 + \binom44 0.1^4 = 0.6561 + 0.0486 + 0.0001 = 0.7048$؛
و $\frac{1 + 0.8^4}{2} = \frac{1 + 0.4096}{2} = 0.7048$:
فالصيغة الحيلة تتتبع الانحياز $(1 - 2\varepsilon)$
عبر كل مرحّل.

\textbf{19.} الكلمات المزعجة تسافر في قطعان --- فرسالة فيها
``يانصيب'' يرجح جدًا أن تحوي ``اربح''، فيكون دليلهما
المشترك أضعف مما يدّعيه الجداء. ومع ذلك يعمل المرشِّح
لأن المبالغة تدفع عادةً في \emph{الاتجاه الصحيح}:
فالتصنيف يحتاج إلى الجهة الصحيحة
من $50\,\%$، لا إلى البعدي الصحيح.

\textbf{20.} العيادة: تحوّل حالة نادرة اختبارًا قويًا
إلى حكم ضعيف --- فحدِّث انطلاقًا من المعدل القاعدي، ثم
راكم. والمحكمة: تطابق نادر في مدينة كبيرة يدين
المدينة لا الرجل --- فالأرجحيات ليست أحكامًا.
وصندوق البريد: أدلة ضعيفة كثيرة، مضروبةً، تفرز البريد --- ولو
على نحو ساذج. والمصائد الثلاث: انسَ القبْلي تُخدَع؛
وزيّف \omterm{def:g12:condprob:indep}{الاستقلال} تُخدَع كثيرًا؛ واقرأ الأدلة قطعة قطعة
فلا تعرف أبدًا ما تعرفه.
\end{solution}
