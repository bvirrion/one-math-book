\section*{الفصل \ref{ch:g12:contdist} --- المتغيرات العشوائية المتصلة}

\begin{solution}{exo:g12:contdist:1}
$\P(X \leq 5) = \frac{5}{15} = \frac13$؛
$\P(4 \leq X \leq 10) = \frac{6}{15} = \frac25$؛
$\E(X) = \frac{15}{2} = 7.5$ min؛
$\sigma(X) = \frac{15}{\sqrt{12}} = \frac{5\sqrt3}{2} \approx 4.3$ min.
\end{solution}

\begin{solution}{exo:g12:contdist:2}
$f \geq 0$ و
$\int_0^2 \frac38 t^2\,\dd t = \frac38\left[\frac{t^3}{3}\right]_0^2
= \frac38 \times \frac83 = 1$: إذن \omterm{def:g12:contdist:density}{كثافة}.
\[
\E(X) = \int_0^2 \frac38 t^3\,\dd t = \frac38 \times \frac{16}{4} = \frac32,
\qquad
\P(X \geq 1) = \int_1^2 \frac38 t^2\,\dd t = \frac{8 - 1}{8} = \frac78 .
\]
\end{solution}

\begin{solution}{exo:g12:contdist:3}
\emph{1.} $\E(X) = \frac{1}{0.2} = 5$ سنة؛
و $\P(X > 5) = \eu^{-0.2\times5} = \eu^{-1} \approx 0.37$.

\emph{2.} بانعدام الذاكرة (\cref{thm:g12:contdist:memoryless})،
$\pcond{X > 3}{X > 8} = \P(X > 5) = \eu^{-1} \approx 0.37$: فسنوات الخدمة
الثلاث لا تغيّر شيئًا.
\end{solution}

\begin{solution}{exo:g12:contdist:4}
$\P(X > m) = \eu^{-\lambda m} = \frac12$ يعطي
$m = \frac{\ln 2}{\lambda} \approx \frac{0.69}{\lambda}$، وهو أصغر من
$\E(X) = \frac1\lambda$. فللكثافة الأسية ذيل أيمن طويل: إذ
تجرّ بضعة أعمار طويلة غير نموذجية \emph{\omterm{def:g10:stats:mean}{المتوسط}} فوق
\emph{\omterm{def:g11:stat:median}{الوسيط}}، وهو القيمة التي يتجاوزها نصف المجتمع. (وهذا هو
نصف العمر في \cref{ex:g12:exp:halflife}: فنصف الذرات ينجو منه، مع
أن العمر الوسطي أطول.)
\end{solution}

\begin{solution}{exo:g12:contdist:5}
$168 = \mu - \sigma$ و $182 = \mu + \sigma$: أي نحو $68\%$.
و $189 = \mu + 2\sigma$: وفوقه يقع نصف الباقي
$100 - 95.4 = 4.6\%$، أي نحو $2.3\%$.
و $154 = \mu - 3\sigma$: أي نحو $\frac{100 - 99.7}{2} = 0.15\%$.
\end{solution}

\begin{solution}{exo:g12:contdist:6}
$\P(X < 500) \leq 2.5\%$ يعني أن $500$ يجب أن يجلس على بعد انحرافين
معياريين على الأقل تحت \omterm{def:g10:stats:mean}{المتوسط} (\omterm{def:g12:contdist:normal}{فالتوزيع الطبيعي} يترك $2.3\% \approx 2.5\%$ تحت
$\mu - 2\sigma$؛ ومع الثابت الاصطلاحي $1.96$ يكون الاستدلال مطابقًا):
\[
\mu - 2\sigma \geq 500 \iff \mu \geq 500 + 8 = 508 \text{ g}.
\]
فيجب ضبط الآلة على $508$ g وسطيًا --- وثمن الضمانة
هو $8$ g من المنتج ``المجاني'' في كل كيس.
\end{solution}

\begin{solution}{exo:g12:contdist:7}
\emph{1.} مع $p = \frac16$ و $n = 1200$:
$\sqrt{\frac{p(1-p)}{n}} = \sqrt{\frac{5/36}{1200}} \approx 0.0108$، إذن
مجال التذبذب عند $95\%$ هو
\[
\frac16 \pm 1.96 \times 0.0108 \approx \intcc{0.146}{0.188}.
\]

\emph{2.} \omterm{def:g10:stats:series}{التواتر} الملاحَظ $\frac{260}{1200} \approx 0.217$ يقع بعيدًا
خارجه: إذن تُرفض فرضية عدم الغش عند المستوى $5\%$. و
الانحراف نحو $4.6$ انحرافًا معياريًا --- وهو، في \omterm{def:g10:numbers:approx}{تقريب}
طبيعي، \omterm{def:g10:proba:distribution}{احتمال} من رتبة $10^{-6}$، وهو أحسم بكثير من
الحاصرة $\leq 4.6\%$ التي أعطتها متراجحة بيينيمي--تشيبيشيف
(\cref{exo:g12:sums:7}).
\end{solution}

\begin{solution}{exo:g12:contdist:8}
\emph{1.} الهامش $\frac{1}{\sqrt n} \leq 0.02$ يقتضي
$n \geq 2500$ شخصًا.

\emph{2.} للتمييز بين نتيجتين تفصل بينهما $1$ نقطة، يجب أن يكون الهامش دون
$0.5$ نقطة بكثير، مما يقتضي $n \geq \left(\frac{1}{0.005}\right)^2 =
40\,000$ بالصيغة المبسَّطة --- وهذا غير عملي أصلًا لمعظم الاستطلاعات.
والأسوأ أن الهامش الإحصائي لا يفسّر إلا خطأ \emph{المعاينة}؛
أما الانحيازات المنهجية (العينات غير الممثِّلة، وعدم الاستجابة،
والتحولات في اللحظة الأخيرة) فلا تنكمش مع نمو $n$ وتتجاوز عادةً نقطة واحدة.
فسباق فارقه $1$ نقطة قريب جدًا فعلًا بحيث لا يُحسم.
\end{solution}

\begin{solution}{exo:g12:contdist:9}
\emph{1.} بما أن $\alpha > 0$، فإن $c \leq \alpha X + \beta \leq d
\iff \frac{c - \beta}{\alpha} \leq X \leq \frac{d - \beta}{\alpha}$، إذن
\[
\P(c \leq Y \leq d)
= \int_{(c-\beta)/\alpha}^{(d-\beta)/\alpha} f(t)\,\dd t .
\]
والتعويض $y = \alpha t + \beta$ (أي بقراءة المساحة في المتغير
$y$، حيث $t = \frac{y - \beta}{\alpha}$ و $\dd t = \frac{\dd y}{\alpha}$)
يحوّل هذا إلى
$\int_c^d \frac{1}{\alpha} f\left(\frac{y-\beta}{\alpha}\right)\dd y$: \omterm{def:g11:func:function}{فالدالة}
$g(y) = \frac1\alpha f\left(\frac{y-\beta}{\alpha}\right)$ \omterm{def:g12:contdist:density}{كثافة}
للمتغير $Y$.

\emph{2.} مع $f = \varphi$ و $\alpha = \sigma$ و $\beta = \mu$:
\[
g(y) = \frac{1}{\sigma}\,\varphi\!\left(\frac{y - \mu}{\sigma}\right)
= \frac{1}{\sigma\sqrt{2\pi}}\,
\exp\left(-\frac{(y-\mu)^2}{2\sigma^2}\right),
\]
وهي إذن \omterm{def:g12:contdist:density}{كثافة} المتغير
$\mathcal N(\mu, \sigma^2) = \mu + \sigma\,\mathcal N(0,1)$.
\end{solution}

\begin{solution}{pb:g12:contdist:1}
\textbf{1.} $\P(\text{الانتظار} \leq 10) = \frac{10}{30} =
\frac13$؛ و $\E = 15$ min؛
و $\sigma = \frac{30}{\sqrt{12}} \approx 8.7$ min.

\textbf{2.} $\int_0^1 2x\,\dd x = 1$: إذن \omterm{def:g12:contdist:density}{كثافة}.
و $\E(X) = \int_0^1 2x^2\,\dd x = \frac23$؛
و $\P\left(X > \frac12\right) = \int_{1/2}^1 2x\,\dd x =
1 - \frac14 = \frac34$.

\textbf{3.} $\P(T > 15) = \eu^{-15/10} \approx 0.22$. \omterm{def:g11:stat:median}{والوسيط}:
$\eu^{-m/10} = \frac12$: أي $m = 10\ln 2 \approx 6.9$ min ---
وهو أقل من \omterm{def:g10:stats:mean}{المتوسط} $10$ لأن ذيل \omterm{def:g12:contdist:expo}{التوزيع الأسي} الأيمن
الطويل (الانتظارات الضخمة النادرة) يجرّ \omterm{def:g10:stats:mean}{المتوسط} إلى الأعلى بينما يترك
\omterm{def:g11:stat:median}{الوسيط} عند الانتظار ``النموذجي''.

\textbf{4.} بانعدام الذاكرة،
$\P(T > 35 \mid T > 20) = \P(T > 15) \approx 0.22$: فالعشرون
دقيقة التي مضت لا تشتري شيئًا --- إذ لا يشيخ التدفق.

\textbf{5.} (أ) أسي (فالتفكك الإشعاعي هو النموذج
الذي اختُرع من أجله انعدام الذاكرة)؛ (ب) منتظم على
$\intcc{0}{8}$ (وهو جدول مواعيد بإزاحة عشوائية)؛ (ج)
أسي؛ (د) لا هذا ولا ذاك: فالمصباح البالي \emph{أرجح}
عطبًا في الساعة القادمة من الجديد، إذن
$\P(T > s + t \mid T > s) < \P(T > t)$: فالبلى يناقض
مبرهنة انعدام الذاكرة.

\textbf{6.} $68\,\%$ و $95\,\%$ و $99.7\,\%$: وهي المجالات
المعلَمية الثلاثة.

\textbf{7.} $z = \frac{190 - 172}{8} = 2.25$: فالذيل الأعلى
$\approx 1.2\,\%$.

\textbf{8.} $z = 2$: أي نحو $2.3\,\%$ --- ونحو شخص واحد
من $44$.

\textbf{9.} تجلس المواصفة عند $\pm 2\sigma$: فينجح نحو
$95.4\,\%$ ويُرفض $4.6\,\%$ --- وهذا مدمّر على نطاق واسع. وعند
$\pm 6\sigma$ يهبط معدل الفشل إلى نحو جزأين في
\emph{المليار}: فستة سيغما تشتري حق الإنتاج الضخم
بلا فحص ضخم.

\textbf{10.} $\sigma = 5$؛ ومع تصحيح \omterm{def:g12:limcont:continuity}{الاتصال}،
$\P \approx \P\left(\abs Z \leq \frac{5.5}{5}\right) =
\P(\abs Z \leq 1.1) \approx 0.73$. وتملّس المبرهنة
لوح غالتون في \cref{pb:g11:binom:1}: فيصير درج الخانات
جرسًا متصلًا.

\textbf{11.} $n \approx \frac{1.96^2 \times 0.25}{0.005^2}
\approx 38\,400$ شخصًا --- وعند ذلك الحجم تقزّم
الأخطاء \emph{غير} المعاينية (الإطار، والرفض، والكذب)
الهامش: فسباق فارقه نقطة واحدة وراء قدرة الاستطلاع الأمين، ولهذا
تقول المعاهد الجادة عندئذٍ ``أقرب من أن يُحسم''.

\textbf{12.} بانعدام الذاكرة: من لحظة وصولك، يكون
الانتظار المتبقي أسيًا \omterm{def:g10:stats:mean}{بمتوسط} $10$ --- أي $10$
دقائق كاملة، لا $5$. وعبارة ``\omterm{def:g10:stats:mean}{متوسط} الفجوة $10$'' في جدول المواعيد
مضلِّلة: فانتظارك له \omterm{def:g12:randvar:rv}{التوزيع} نفسه الذي لفجوة كاملة.

\textbf{13.} تحطّ الوصولات العشوائية بالتناسب مع طول
الفجوة: $\P(\text{فجوة طويلة}) = \frac{15}{20} = \frac34$.
والانتظار المأمول: $\frac34 \times \frac{15}{2} + \frac14 \times
\frac52 = 5.625 + 0.625 = 6.25$ min --- أي أكثر من الساذج
$5$، مع أن \omterm{def:g10:stats:mean}{متوسط} الفجوة $10$. والجاني: \emph{المعاينة المنحازة
بالطول} --- فالفجوات الطويلة تصطاد ركابًا أكثر.

\textbf{14.} الانتظاران إلى الوراء وإلى الأمام أسيان كلاهما
\omterm{def:g10:stats:mean}{بمتوسط} $10$ (فانعدام الذاكرة يسري في الاتجاهين من
لحظة وصولك): فالفجوة التي تحويك متوسطها $20$ دقيقة
--- أي ضعف الفجوة النموذجية. ومفارقة المعاينة في جملة واحدة:
\emph{\omterm{def:g10:numbers:interval}{الفترة} التي تصادف أن تعاينها ليست \omterm{def:g10:numbers:interval}{فترة}
نموذجية، لأنك كنت أرجح وقوعًا في \omterm{def:g10:numbers:interval}{فترة}
كبيرة.}

\textbf{15.} القسم الوسطي: $\frac{9 \times 10 + 110}{10} =
20$ طالبًا. \omterm{def:g10:stats:mean}{والمتوسط} الذي يعيشه الطالب:
$\frac{90 \times 10 + 110 \times 110}{200} = \frac{13\,000}
{200} = 65$ طالبًا. والكتيّب يطبع $20$؛ وثلاثة من كل خمسة
طلاب يجلسون في القسم ذي $110$ طلاب ويعيشون العدد $65$.

\textbf{16.} يظهر الأشخاص المشهورون في قوائم أصدقاء كثيرة، فتكون
معاينة ``صديق'' منحازة بالطول نحو الاجتماعيين
--- فأصدقاؤك مسحوبون من الفجوات الطويلة في جدول
المواعيد الاجتماعي.

\textbf{17.} من أجل $t \geq 0$:
$\P(T > t) = \P(-10\ln U > t) = \P\left(U <
\eu^{-t/10}\right) = \eu^{-t/10}$: أي بالضبط \omterm{def:g11:func:function}{دالة} بقاء
\omterm{def:g12:contdist:expo}{التوزيع الأسي} --- \omterm{def:g12:exp:ln}{فلوغاريتم} واحد يحوّل ضجيج الحاسوب
المنتظم إلى أي زمن انتظار.

\textbf{18.} لكل متغير منتظم \omterm{def:g10:stats:mean}{المتوسط} $\frac12$ \omterm{def:g12:randvar:exp}{والتباين}
$\frac{1}{12}$: فلمجموع اثني عشر منها \omterm{def:g10:stats:mean}{المتوسط} $6$ \omterm{def:g12:randvar:exp}{والتباين}
$1$ --- إذن تخرج الوصفة \omterm{def:g10:stats:mean}{بمتوسط} $0$ \omterm{def:g12:randvar:exp}{وتباين} $1$. اثنتا عشرة
دفعة صغيرة مستقلة مجموعة: أي آلية غالتون، وببركة
مبرهنة دي موافر--لابلاس يكون المدرَّج التكراري جرسي
الشكل للعين أصلًا.

\textbf{19.} عن النموذج: فالأسواق ليست مجاميع أسباب
صغيرة \emph{مستقلة} كثيرة --- فالذعر يربط
كل شيء (وهو درس سنة 2008 في المسألة \omterm{def:g10:functions:function}{السابقة}) و
ينتج ذيولًا سمينة لا تملكها أي \omterm{def:g12:contdist:density}{كثافة} طبيعية. وحين تهمس المعطيات
بالعدد $10^{-137}$، تجيب عقيدة الإحصائي:
\emph{النموذج هو المرفوض، لا اليوم}.

\textbf{20.} المنتظم: كل المواضع سواء، وهو قول أمين ``لا
أعرف إلا الحدود''. والأسي: الخطر الذي لا
يشيخ --- التفككات، والوصولات، والنقرات. والطبيعي: جرس
الأسباب الصغيرة المستقلة الكثيرة الديمقراطي --- الأطوال، والأخطاء،
والمتوسطات. واللسعة: ما تعاينه منحاز بكيفية اصطدامك
به --- الحافلات، والأقسام، والأصدقاء. وأما القوس: فالطفل
الذي عدّ علب العصير في الكتاب السابق قد قاس، بعد عشرة مجلدات
من المسائل، الجمهور \omterm{def:g10:functions:graph}{بمنحنى}؛ وإبسيلون،
\omterm{def:g12:integ:area}{والتكامل}، ومبرهنة النهاية المركزية تنتظر في الكتب
الجامعية لتفسّر \emph{لماذا} يقرع الجرس
لكل شيء.
\end{solution}
