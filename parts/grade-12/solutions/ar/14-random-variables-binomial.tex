\section*{الفصل \ref{ch:g12:randvar} --- المتغيرات العشوائية والتوزيع الثنائي}

\begin{solution}{exo:g12:randvar:1}
$\E(X) = -0.3 + 0 + 0.8 + 0.5 = 1$.
و $\E(X^2) = 0.3 \times 1 + 0 + 0.4\times4 + 0.1\times25 = 4.4$، إذن بصيغة
كونيغ--هويغنز $\V(X) = 4.4 - 1 = 3.4$ و
$\sigma(X) = \sqrt{3.4} \approx 1.84$.
\end{solution}

\begin{solution}{exo:g12:randvar:2}
\emph{1.} الأسئلة $10$ تجارب برنولي مستقلة حيث
$p = \frac14$: $X \sim \mathcal B\left(10, \frac14\right)$،
و $\E(X) = 2.5$،
و $\sigma(X) = \sqrt{10 \times \frac14 \times \frac34} = \sqrt{1.875}
\approx 1.37$.

\emph{2.} $\P(X = 0) = \left(\frac34\right)^{10} \approx 0.056$؛
\[
\P(X = 5) = \binom{10}{5}\left(\frac14\right)^5\left(\frac34\right)^5
\approx 0.058;
\quad
\P(X \geq 1) = 1 - \left(\tfrac34\right)^{10} \approx 0.944 .
\]
\end{solution}

\begin{solution}{exo:g12:randvar:3}
تجارب متماثلة مستقلة: $X \sim \mathcal B(400,\ 0.05)$، إذن
$\E(X) = 20$ مطالبة و
\[
\sigma(X) = \sqrt{400 \times 0.05 \times 0.95} = \sqrt{19} \approx 4.4 .
\]
\end{solution}

\begin{solution}{exo:g12:randvar:4}
المبلغ المقبوض $P$ يحقق
$\E(P) = \frac{5 + 6}{6} = \frac{11}{6}$، إذن السعر العادل هو
$c = \frac{11}{6} \approx 1.83$ أورو. والربح العادل $G = P - \frac{11}6$
يأخذ القيم $-\frac{11}{6}, \frac{19}{6}, \frac{25}{6}$ بالاحتمالات
$\frac46, \frac16, \frac16$:
\[
\V(G) = \E(G^2)
= \frac{4 \times 121 + 361 + 625}{6 \times 36} = \frac{1470}{216}
= \frac{245}{36} \approx 6.8,
\qquad \sigma(G) \approx 2.6 .
\]
فللعبة العادلة ربح \emph{وسطي} معدوم لكن نتائجها تظل تتذبذب:
فالعادل ليس خاليًا من الخطر.
\end{solution}

\begin{solution}{exo:g12:randvar:5}
$X \sim \mathcal B(8,\ 0.7)$.
\[
\P(X = 6) = \binom86 (0.7)^6(0.3)^2 \approx 0.296 ;
\]
\[
\P(X \geq 6) = \P(6) + \P(7) + \P(8)
\approx 0.296 + 8(0.7)^7(0.3) + (0.7)^8
\approx 0.296 + 0.198 + 0.058 = 0.552 ;
\]
$\P(X \geq 1) = 1 - (0.3)^8 \approx 0.99993$. والمنوال:
$(n+1)p = 6.3$، إذن القيمة الأرجح هي $k^* = 6$
(\cref{exo:g12:randvar:7}).
\end{solution}

\begin{solution}{exo:g12:randvar:6}
عدد المسافرين الحاضرين هو $X \sim \mathcal B(104,\ 0.9)$؛ و
يكون الحجز زائدًا عندما $X \geq 101$:
\[
\P(X \geq 101) = \sum_{k=101}^{104}\binom{104}{k}(0.9)^k(0.1)^{104-k}
\approx 0.006 .
\]
فبيع تذاكر أكثر من المقاعد بنسبة $4\%$ يسبّب \omterm{def:g11:prob:model}{حادثة} في نحو
$0.6\%$ فقط من الرحلات --- وهذا هو اقتصاد الحجز الزائد.
\end{solution}

\begin{solution}{exo:g12:randvar:7}
\[
\frac{\P(X = k+1)}{\P(X = k)}
= \frac{\binom{n}{k+1}}{\binom nk}\cdot\frac{p}{1-p}
= \frac{n - k}{k + 1}\cdot\frac{p}{1-p},
\]
باستعمال $\binom{n}{k+1} = \binom nk \frac{n-k}{k+1}$. وهذه النسبة $\geq 1$
إذا وفقط إذا كان $(n-k)p \geq (k+1)(1-p)$ إذا وفقط إذا كان $np - k p \geq k - kp + 1 - p$ إذا وفقط إذا كان
$k \leq (n+1)p - 1$. إذن تتزايد الاحتمالات تمامًا ما دام
$k + 1 \leq (n+1)p$ وتتناقص بعدها: \omterm{def:g10:functions:extrema}{فالقيمة العظمى} تُبلَغ عند
$k^* = \floor{(n+1)p}$ (وتتقاسمها مع $k^* - 1$ عندما يكون $(n+1)p$ عددًا
صحيحًا).
\end{solution}

\begin{solution}{exo:g12:randvar:8}
\emph{1.} أول صورة عند الرمية $k$ يعني $k-1$ ظهرًا ثم صورة:
\omterm{def:g10:proba:distribution}{باحتمال} $\left(\frac12\right)^{k-1}\cdot\frac12 = 2^{-k}$، من أجل
$1 \leq k \leq 10$. \omterm{def:g10:proba:distribution}{والاحتمال} الكلي للربح
$\sum_{k=1}^{10} 2^{-k} = 1 - 2^{-10} = \frac{1023}{1024}$ (فلا تُخسر اللعبة
إلا عند عشرة أظهر \omterm{def:g12:seq:sequence}{متتالية}).

\emph{2.} المدفوع المأمول:
\[
\sum_{k=1}^{10} 2^k \cdot 2^{-k} = \sum_{k=1}^{10} 1 = 10 \text{ أورو}.
\]
ولولا السقف، لتباعد المجموع $\sum_{k\geq1} 1$: فيكون المدفوع المأمول
لانهائيًا، مع أن اللعبة تدفع دائمًا \omterm{def:g10:numbers:approx}{تقريبًا} مبلغًا صغيرًا --- وهي
\emph{مفارقة سان بطرسبورغ} الشهيرة، التي تبيّن أن \omterm{def:g12:randvar:exp}{الأمل الرياضي} وحده
لا يقيس قيمة لعبة.
\end{solution}

\begin{solution}{pb:g12:randvar:1}
\textbf{1.} $X \sim \mathcal B\!\left(3, \frac16\right)$:
$\P(X = 0) = \frac{125}{216}$،
$\P(X = 1) = \frac{75}{216}$،
$\P(X = 2) = \frac{15}{216}$،
$\P(X = 3) = \frac{1}{216}$.

\textbf{2.} $\E(X) = 3 \times \frac16 = \frac12$؛
$V(X) = 3 \times \frac16 \times \frac56 = \frac{5}{12}$.

\textbf{3.} المدفوع المأمول $2\,\E(X) = 1$ أورو في مواجهة
رهان قدره $1$ أورو: فالربح $0$ --- أي عادلة تمامًا (وهو أمر نادر).

\textbf{4.} بلا إرجاع تكون السحبات مرتبطة (إذ تتعلق
حظوظ الورقة الثانية بالأولى): إذن ليس ثنائيًا ---
فخانة \omterm{def:g12:condprob:indep}{الاستقلال} في قائمة التحقق تسقط. وأما مع الإرجاع: فعدد
ثابت $n = 5$، \omterm{def:g10:proba:distribution}{واحتمال} ثابت $p = \frac14$، وسحبات مستقلة: إذن
ثنائي.

\textbf{5.} $\E(X) = 6$ و $\sigma(X) = \sqrt{4.2} \approx
2.05$. و $\E(S) = 5 \times 6 - 10 = 20$؛
و $\sigma(S) = 5\sigma(X) \approx 10.25$.

\textbf{6.} عدد ثابت $n = 105$ من التجارب (وهي التذاكر)، وكل منها
حضور أو عدم حضور \omterm{def:g10:proba:distribution}{بالاحتمال} نفسه $p = 0.9$، مفترضًا فيها \omterm{def:g12:condprob:indep}{الاستقلال}:
إذن ثنائي. وأما الاعتراف: فالأسر تفوّت الرحلات
\emph{معًا} والعواصف تفرّغ طائرات بكاملها --- \omterm{def:g12:condprob:indep}{فالاستقلال}
هو قفزة إيمان النموذج، وحالات عدم الحضور المترابطة تجعل
الذيول أسمن مما يعد به \omterm{def:g12:randvar:binomial}{التوزيع الثنائي}.

\textbf{7.} $\E(X) = 94.5$؛
و $\sigma(X) = \sqrt{105 \times 0.9 \times 0.1} =
\sqrt{9.45} \approx 3.07$.

\textbf{8.} $\P(X \geq 101) = \sum_{k=101}^{105}
\binom{105}{k} 0.9^k\, 0.1^{105 - k}$.

\textbf{9.} $\approx 0.0167$: أي نحو رحلة واحدة من ستين
تُنزل أحدًا أصلًا.

\textbf{10.} $\E(\text{المُنزَلين}) \approx 0.023$ مسافر في
الرحلة الواحدة: فالتعويض المأمول $\approx 19$ أورو --- في مواجهة
$1\,000$ أورو من الدخل الإضافي. فالحجز الزائد بخمس تذاكر
مربح بصورة ساحقة؛ ولهذا تفعله كل شركة طيران.

\textbf{11.} من جدول احتمالات الذيول:
$n = 105$: $1.7\,\%$؛ و $n = 106$: $4.0\,\%$؛ و $n = 107$:
$8.1\,\%$. فأكبر بيع ممتثل هو $n = 106$ تذكرة.

\textbf{12.} $z = \frac{100.5 - 94.5}{3.07} \approx 1.95$:
فالعتبة تقع على بعد انحرافين معياريين فوق \omterm{def:g10:stats:mean}{المتوسط}،
وقاعدة \omterm{def:g10:functions:graph}{منحنى} الجرس التقريبية (``الذيل الأعلى $2\sigma$
$\approx 2.5\,\%$'') تحطّ قريبًا من القيمة المضبوطة $1.7\,\%$ ---
\omterm{def:g10:functions:graph}{فالمنحنى} الأملس الذي يظلّل \omterm{def:g12:randvar:binomial}{التوزيع الثنائي} هو بطل
الفصول القادمة.

\textbf{13.} $0.98^{20} + 20 \times 0.02 \times 0.98^{19}
\approx 0.94$: فالدفعة النزيهة تمر في $94\,\%$ من الحالات.

\textbf{14.} $0.9^{20} + 20 \times 0.1 \times 0.9^{19}
\approx 0.39$: فالدفعة الرديئة ما زالت تتسلل في $39\,\%$ من
الحالات.

\textbf{15.} خطر المنتج: دفعة جيدة مرفوضة
($\approx 6\,\%$)؛ وخطر المستهلك: دفعة رديئة مقبولة
($\approx 39\,\%$). \omterm{def:g10:proba:sample}{وعينة} أكبر (مع عتبة قبول
متناسبة) تقلّص كليهما --- بثمن مزيد من
الفحص: فللجودة خط ميزانية.

\textbf{16.} $V\!\left(\frac Xn\right) = \frac{V(X)}{n^2} =
\frac{np(1-p)}{n^2} = \frac{p(1-p)}{n}$: \omterm{def:g11:stat:variance}{فالانحراف المعياري}
$\sqrt{\frac{p(1-p)}{n}}$. ربّع حجم \omterm{def:g10:proba:sample}{العينة}، ينتصف
الضجيج: وهو قانون $\frac{1}{\sqrt n}$ في مجالات التذبذب
التي رأيناها من قبل، مبرهَنًا عليه الآن.

\textbf{17.} يحدّ السقف المدفوع عند $2^{10} = 1024$
أورو، فتسهم كل جولة من الجولات العشر بمقدار $1$ أورو بالضبط
من \omterm{def:g12:randvar:exp}{الأمل الرياضي}: أي $\E = 10$. أما \omterm{def:g12:randvar:exp}{الأمل الرياضي} اللانهائي للعبة
غير المسقوفة فقد جاء كله من مدفوعات نادرة ندرة فلكية
وضخمة ضخامة فلكية؛ والسقف يعترف بأن لا مصرف
يدفع $2^{50}$ أورو. وسعر تذكرة عادل للعبة المسقوفة:
$10$ أورو --- وفجأةً لا مفارقة عند أحد.

\textbf{18.} الجائزة المأمولة للتذكرة الواحدة:
$\frac{100 + 2 \times 50}{200} = 1$ أورو في مواجهة تذكرة سعرها $2$
أورو: فالهامش $50\,\%$ --- أي ثمانية عشر ضعف هامش الروليت
البالغ $2.7\,\%$. ويفلت اليانصيب بفعلته لأن لاعبيه
يتبرعون عن علم: و``الخسارة'' هي المقصد.

\textbf{19.} المطالبات المأمولة: $1\,000 \times 0.01 \times
10\,000 = 100\,000$؛ والأقساط $130\,000$: فالربح المأمول
$30\,000$ أورو. \omterm{def:g11:stat:variance}{والانحراف المعياري} لمجموع المطالبات:
$10\,000 \times \sqrt{1\,000 \times 0.01 \times 0.99}
\approx 31\,500$ أورو --- فسنة سيئة عادية واحدة تلتهم
الربح المأمول كله. ومن هنا الاحتياطيات الرأسمالية، وإعادة التأمين،
والمحافظ الأكبر بكثير من ألف عقد: فالمؤمّنون
يعيشون على قانون الأعداد الكبيرة ويحتفظون باحتياطيات في مواجهة
بطئه.

\textbf{20.} قائمة التحقق أولًا --- عدد ثابت $n$، و $p$ نفسه، و
\omterm{def:g12:condprob:indep}{استقلال} \emph{مدَّعى} --- وإلا فلا \omterm{def:g12:randvar:binomial}{توزيع ثنائي} أصلًا. ثم
أبحر بالبوصلة $\E \pm \sigma$: فالمتوسطات تحدّد الموضع، والانحرافات تحذّر.
ثم سعّر الذيول: فالإنزال، والدفعات الرديئة، والسنوات المدمّرة كلها
تعيش وراء $2\sigma$. والتحذيران: \omterm{def:g12:condprob:indep}{الاستقلال} وعد
النمذجة لا وعد العالم؛ \omterm{def:g12:randvar:exp}{والأمل الرياضي} يتجاهل بالضبط
ما يدمّرك. وأما صفحات كتاب القواعد الناقصة --- بأي سرعة
تستقر التواترات، وأي شكل تأخذه التذبذبات ---
فهي الفصلان التاليان.
\end{solution}
