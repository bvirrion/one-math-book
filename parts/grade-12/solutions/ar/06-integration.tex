\begin{solution}{exo:g12:integ:1}
\[
\int_1^2 \left(3x^2 - \frac{1}{x^2}\right)\dd x
= \left[x^3 + \frac1x\right]_1^2 = \left(8 + \tfrac12\right) - 2 = \frac{13}{2}.
\]
\[
\int_0^{\pi/2} \cos t\,\dd t = \bigl[\sin t\bigr]_0^{\pi/2} = 1 .
\]
ومع $u = 2t+1$، $\frac{1}{2t+1} = \frac12\,\frac{u'}{u}$:
\[
\int_0^1 \frac{\dd t}{2t+1} = \frac12\bigl[\ln(2t+1)\bigr]_0^1
= \frac{\ln 3}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:2}
\omterm{def:g11:func:function}{للدالة} $x\,\eu^{x^2} = \frac12\,(x^2)'\,\eu^{x^2}$ الدوال الأصلية
$\frac12\eu^{x^2} + c$. والشرط $F(0) = 1$ يعطي $\frac12 + c = 1$،
إذن
\[
F(x) = \frac{\eu^{x^2} + 1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:3}
\[
\mu = \frac{1}{\pi}\int_0^\pi \sin t\,\dd t
= \frac{1}{\pi}\bigl[-\cos t\bigr]_0^\pi = \frac{2}{\pi} \approx 0.64 .
\]
وبيانيًا، للمستطيل ذي القاعدة $\intcc{0}{\pi}$ والارتفاع
$\frac{2}{\pi}$ المساحة نفسها التي لقوس \omterm{def:g10:functions:graph}{منحنى} \omterm{def:g11:trigo:cossin}{الجيب}.
\end{solution}

\begin{solution}{exo:g12:integ:4}
خذ $u' = 1$ و $v = \ln t$، فيكون $u = t$ و $v' = \frac1t$:
\[
\int_1^{\eu} \ln t\,\dd t = \bigl[t\ln t\bigr]_1^{\eu} - \int_1^{\eu} 1\,\dd t
= \eu - (\eu - 1) = 1 .
\]
وخذ $u' = \sin t$ و $v = t$، فيكون $u = -\cos t$ و $v' = 1$:
\[
\int_0^\pi t\sin t\,\dd t = \bigl[-t\cos t\bigr]_0^\pi + \int_0^\pi \cos t\,\dd t
= \pi + 0 = \pi .
\]
\end{solution}

\begin{solution}{exo:g12:integ:5}
يتقاطع المنحنيان حيث $x^2 = x + 2$، \emph{أي} $x \in \{-1, 2\}$، و
على $\intcc{-1}{2}$ يقع المستقيم فوق \omterm{def:g11:quad:parabola}{القطع المكافئ}. والمساحة هي
\[
\int_{-1}^{2} \bigl(x + 2 - x^2\bigr)\dd x
= \left[\frac{x^2}{2} + 2x - \frac{x^3}{3}\right]_{-1}^{2}
= \frac{10}{3} - \left(-\frac{7}{6}\right) = \frac{9}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:6}
\emph{1.} $I = \bigl[\ln(1+t)\bigr]_0^1 = \ln 2$.

\emph{2.}
\[
I_n + I_{n+1} = \int_0^1 \frac{t^n(1 + t)}{1+t}\,\dd t
= \int_0^1 t^n \,\dd t = \frac{1}{n+1}.
\]
وعلى $\intcc{0}{1}$، $0 \leq \dfrac{t^n}{1+t} \leq t^n$، إذن بالمقارنة
$0 \leq I_n \leq \frac{1}{n+1}$.

\emph{3.} ومن العلاقة التراجعية، بتراجع تلسكوبي،
\[
I_n = (-1)^n\left(I_0 - \left(1 - \frac12 + \dots + \frac{(-1)^{n-1}}{n}\right)\right),
\]
\emph{أي}
$\displaystyle\sum_{k=1}^{n} \frac{(-1)^{k-1}}{k} = \ln 2 - (-1)^n I_n$
(وتذكّر أن $I_0 = \ln 2$). وبما أن $\abs{I_n} \leq \frac{1}{n+1} \to 0$، فإن
المجموع التوافقي المتناوب يؤول إلى $\ln 2$.
\end{solution}

\begin{solution}{exo:g12:integ:7}
المسافة:
\[
d = \int_0^{100} 30\left(1 - \eu^{-t/50}\right)\dd t
= 30\left[t + 50\,\eu^{-t/50}\right]_0^{100}
= 30\left(100 + 50\eu^{-2} - 50\right)
= 1500 + 1500\,\eu^{-2}.
\]
وعدديًا $d \approx 1500 + 203 = 1703$ m. والسرعة المتوسطة:
$\frac{d}{100} = 15\bigl(1 + \eu^{-2}\bigr) \approx 17.0$ m/s.
\end{solution}

\begin{solution}{exo:g12:integ:8}
\emph{1.} $S_n$ هو المساحة الكلية لعدد $n$ من المستطيلات عرض كل منها $\frac1n$ و
ارتفاعاتها $f\left(\frac kn\right)$ و $k = 1, \dots, n$، حيث
$f(t) = \frac{1}{1+t}$: أي إنه \omterm{def:g10:numbers:approx}{تقريب} المساحة تحت $f$ على $\intcc{0}{1}$
بمستطيلات ``الطرف الأيمن''.

\emph{2.} \omterm{def:g11:func:function}{الدالة} $f$ \omterm{def:g10:functions:variations}{متناقصة}، إذن على كل \omterm{def:g10:numbers:interval}{فترة}
$\intcc{\frac{k-1}{n}}{\frac kn}$،
$f\left(\frac kn\right) \leq f(t) \leq f\left(\frac{k-1}{n}\right)$؛
وبالتكامل والجمع على $k$:
\[
S_n \leq \int_0^1 f(t)\,\dd t \leq S_n + \frac1n\bigl(f(0) - f(1)\bigr)
= S_n + \frac{1}{2n}.
\]
(والحد الأوسط في عبارة التمرين، مع
$f(0) - f(1) = 1 - \frac12$، هو بالضبط هذه الحاصرة.) ومنه
$0 \leq \ln 2 - S_n \leq \frac{1}{2n} \to 0$، إذن $S_n \to \ln 2$.
\end{solution}

\begin{solution}{exo:g12:integ:9}
\emph{1.} $W_0 = \int_0^{\pi/2} \dd t = \frac{\pi}{2}$ و
$W_1 = \bigl[-\cos t\bigr]_0^{\pi/2} = 1$.

\emph{2.} كامل بالتجزئة مع $u' = \sin t$ و $v = \sin^{n+1} t$، فيكون
$u = -\cos t$ و $v' = (n+1)\sin^n t \cos t$:
\[
W_{n+2} = \bigl[-\cos t \sin^{n+1} t\bigr]_0^{\pi/2}
+ (n+1)\int_0^{\pi/2} \cos^2 t\,\sin^n t\,\dd t
= (n+1)\int_0^{\pi/2} (1 - \sin^2 t)\sin^n t\,\dd t,
\]
إذ ينعدم القوس عند الطرفين معًا. ومنه
$W_{n+2} = (n+1)(W_n - W_{n+2})$، \emph{أي}
$W_{n+2} = \frac{n+1}{n+2} W_n$.

\emph{3.} $W_2 = \frac12 W_0 = \frac{\pi}{4}$، و
$W_3 = \frac23 W_1 = \frac23$، و
$W_4 = \frac34 W_2 = \frac{3\pi}{16}$.
وعلى $\intoo{0}{\frac\pi2}$، يكون $0 < \sin t < 1$، إذن
$\sin^{n+1} t \leq \sin^n t$ مع متراجحة تامة في الداخل؛
وبالتكامل، $0 < W_{n+1} \leq W_n$ (وهي $<$ في الواقع): إذن $(W_n)$ \omterm{def:g10:functions:variations}{متناقصة} و
موجبة.
\end{solution}

\begin{solution}{pb:g12:integ:1}
\textbf{1.} $\left[x^3 - x^2 + x\right]_0^1 = 1$؛ \quad
$\left[\ln x\right]_1^{\eu} = 1$؛ \quad
$\left[\sin x\right]_0^{\pi/2} = 1$.

\textbf{2.} $\left[\frac12 \eu^{x^2}\right]_0^1 =
\frac{\eu - 1}{2}$؛ \quad
$\left[\ln(x^2 + 1)\right]_0^1 = \ln 2$.

\textbf{3.} $\int_0^1 x\eu^x = \left[x\eu^x\right]_0^1 -
\int_0^1 \eu^x = \eu - (\eu - 1) = 1$. و
$\int_1^{\eu} \ln x = \left[x\ln x\right]_1^{\eu} -
\int_1^{\eu} 1 = \eu - (\eu - 1) = 1$.

\textbf{4.} $\frac{1}{\pi}\int_0^\pi \sin x\,\dd x =
\frac{2}{\pi} \approx 0.64$: فالارتفاع \omterm{def:g10:stats:mean}{المتوسط} لقوس
\omterm{def:g11:trigo:cossin}{الجيب} هو $\frac2\pi$، وهو أكبر من النصف بوضوح --- إذ
القوس سمين قرب قمته.

\textbf{5.} $\int_0^1 (x - x^2)\,\dd x = \frac12 - \frac13 =
\frac16$.

\textbf{6.} $L_n = \frac1n \sum_{k=0}^{n-1}
\left(\frac kn\right)^2$ و
$U_n = \frac1n \sum_{k=1}^{n} \left(\frac kn\right)^2$:
مستطيلات عرضها $\frac1n$ وارتفاعاتها مقروءة عند الطرف الأيسر
(تحت \omterm{def:g10:functions:graph}{المنحنى}) أو الأيمن (فوقه).

\textbf{7.} صحيحة من أجل $n = 1$ ($1 = \frac{1 \cdot 2 \cdot
3}{6}$). وإذا صحت من أجل $n$، فبإضافة $(n+1)^2$:
\[
\frac{n(n+1)(2n+1)}{6} + (n+1)^2
= \frac{(n+1)\left(2n^2 + n + 6n + 6\right)}{6}
= \frac{(n+1)(n+2)(2n+3)}{6},
\]
وهي الصيغة عند $n + 1$: فالوراثة تامة، وقد تمّ المطلوب.

\textbf{8.} $U_n = \frac{1}{n^3} \cdot
\frac{n(n+1)(2n+1)}{6} = \frac{(1 + \frac1n)(2 + \frac1n)}{6}
\to \frac26 = \frac13$، و $L_n = U_n - \frac1n \to \frac13$:
فبحصرها بين درجيها، تكون المساحة بالضبط
$\frac13$.

\textbf{9.} $\int_0^1 x^2\,\dd x =
\left[\frac{x^3}{3}\right]_0^1 = \frac13$: سطر واحد. فالمبرهنة
الأساسية تحوّل حصرًا لانهائيًا إلى تقدير واحد
\omterm{def:g12:integ:primitive}{لدالة أصلية} --- ولهذا تُسمّى
أساسية.

\textbf{10.} القطعة: $\int_{-1}^{1}(1 - x^2)\,\dd x =
2 - \frac23 = \frac43$. والمثلث الذي قاعدته الوتر (وطوله
$2$، على الارتفاع $1$) ورأسه $(0,0)$: مساحته
$\frac12 \times 2 \times 1 = 1$. والنسبة: $\frac43$ --- وهي بالضبط
مبرهنة أرخميدس، متحقَّقًا منها بالآلة التي لم يملكها.

\textbf{11.} $T\left(1 + \frac14 + \frac{1}{16} + \dots\right)
= T \cdot \frac{1}{1 - \frac14} = \frac43 T$ (متسلسلة \omterm{def:g12:seq:arith-geom}{هندسية}
أساسها $\frac14$): ومع $T = 1$، تكون مساحة القطعة
$\frac43$ من جديد. وقد جمع أرخميدس المتسلسلة بحجة
\omterm{def:g12:seq:arith-geom}{هندسية} خالصة --- بقضمات الربع من نمط لوح
الشوكولاتة، قبل ألفي سنة من صيرورة النهايات كلمات.

\textbf{12.} يحتاج الاستنفاد إلى تشابه ذاتي هندسي
بارع ويعطي مساحات مضبوطة حالة بحالة؛ ولا يحتاج ريمان
إلا إلى الرتابة ويعطي تعريفًا كونيًا بثمن
حساب المجاميع؛ وتحتاج المبرهنة الأساسية إلى
صرح المشتقات كله وتردّه بجعل المساحات
حسابات من سطر واحد.

\textbf{13.} $A(1) = 0$ و $A'(x) = \frac1x$ (بالمبرهنة
الأساسية): \omterm{def:g11:func:function}{فالدالة} $A$ هي \omterm{def:g12:integ:primitive}{الدالة الأصلية} \omterm{def:g11:func:function}{للدالة} $\frac1x$ على
$\intoo{0}{+\infty}$ المنعدمة عند $1$ --- وهي بالضبط
\omterm{def:g12:exp:ln}{اللوغاريتم النيبيري}.

\textbf{14.} $g'(x) = \frac{a}{ax} - \frac1x = 0$: إذن $g$
ثابتة؛ وعند $x = 1$: $g(1) = A(a) - A(1) = A(a)$. ومنه
$A(ax) = A(a) + A(x)$ من أجل كل $x$، ومع $x = b$:
$\ln(ab) = \ln a + \ln b$. فمساحة \omterm{def:g11:func:reference}{القطع الزائد} لا تبالي
بالسلّم الأفقي: فمطّ $t$ بالمعامل $a$ يسحق
$\frac1t$ بالمعامل $a$، وتنجو المساحة --- فجمع
اللوغاريتمات تناظر سلّمي.

\textbf{15.} بالتكرار: $\ln(a^n) = n\ln a$ (بالتراجع على
$n$). و $0 = \ln 1 = \ln\left(a \cdot \frac1a\right) =
\ln a + \ln\frac1a$: $\ln\frac1a = -\ln a$.

\textbf{16.} $\frac{1}{11} + \frac{1}{12} + \dots +
\frac{1}{20} \approx 0.669$، في مواجهة $\ln 2 \approx 0.693$:
عشرة مستطيلات تحت $\frac{1}{1+t}$، قريبة أصلًا. فالمتسلسلة
التوافقية متباعدة، لكن \emph{شرائحها} من $n$ إلى
$2n$ تستقر عند $\ln 2$: تباعد بخطوة السلحفاة، مقيسًا
بمساحة.

\textbf{17.} المسافة: $\int_0^{10} 3t^2\,\dd t = 1000$ m.
والسرعة المتوسطة: $100$ m/s. واللحظة التي فيها $v = 100$:
$3t^2 = 100$: $t = \sqrt{\frac{100}{3}} \approx 5.77$ s.

\textbf{18.} الكتلة: $\int_0^4 (2 + x)\,\dd x = 8 + 8 = 16$ kg.
والعزم: $\int_0^4 (2x + x^2)\,\dd x = 16 + \frac{64}{3} =
\frac{112}{3}$. ومركز الكتلة:
$\frac{112/3}{16} = \frac73 \approx 2.33$ m --- مدفوعًا وراء
\omterm{prop:g10:coordgeom:midpoint}{المنتصف} بالطرف الأيمن الأثقل، كما يقتضي الحدس.

\textbf{19.} $W_0 = \frac\pi2$؛ $W_1 = \left[-\cos
t\right]_0^{\pi/2} = 1$؛ و
$W_2 = \int_0^{\pi/2} \frac{1 - \cos 2t}{2}\,\dd t =
\frac\pi4$.

\textbf{20.} المساحة: التعريف، محصورًا بالدرجات.
والتراكم: المسافات من السرعات، والكتل من الكثافات ---
\omterm{def:g12:integ:area}{فالتكامل} مجموع جارٍ. \omterm{def:g12:integ:primitive}{والدالة الأصلية}:
المبرهنة الأساسية، وهي طريق نيوتن ولايبنتز المختصر عبر
صبر أرخميدس. واللؤلؤة: زر $\ln$ هو المساحة
تحت $\frac1t$، وتناظر تلك المساحة السلّمي \emph{هو}
المتطابقة $\ln(ab) = \ln a + \ln b$.
\end{solution}
