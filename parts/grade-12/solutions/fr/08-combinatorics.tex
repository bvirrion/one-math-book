\section*{Chapitre \ref{ch:g12:comb} --- Combinatoire et dénombrement}

\begin{solution}{exo:g12:comb:1}
Principe de multiplication~:
$26^2 \times 10^3 \times 26^2 = 26^4 \times 10^3 = 456\,976\,000$.

Sans caractère répété, les quatre lettres doivent être distinctes
($26 \times 25 \times 24 \times 23$ façons, en remplissant les positions
des lettres dans l'ordre) et les trois chiffres distincts
($10 \times 9 \times 8$)~:
\[
26 \times 25 \times 24 \times 23 \times 10 \times 9 \times 8
= 358\,800 \times 720 = 258\,336\,000 .
\]
\end{solution}

\begin{solution}{exo:g12:comb:2}
$\dbinom83 = \dfrac{8 \times 7 \times 6}{3!} = 56$~;
$\dbinom{10}{8} = \dbinom{10}{2} = \dfrac{10 \times 9}{2} = 45$~;
\[
\frac{\binom n2}{\binom{n+1}2}
= \frac{n(n-1)/2}{(n+1)n/2} = \frac{n-1}{n+1}.
\]
\end{solution}

\begin{solution}{exo:g12:comb:3}
Choisir le comité~: $\binom{30}{4}$ façons. Puis choisir président et
trésorier parmi les 4, dans l'ordre~: $4 \times 3 = 12$ façons. Total
\[
\binom{30}{4} \times 12 = 27\,405 \times 12 = 328\,860 .
\]
\end{solution}

\begin{solution}{exo:g12:comb:4}
\[
(x+2)^5 = x^5 + 10x^4 + 40x^3 + 80x^2 + 80x + 32 ,
\]
\[
(1-x)^6 = 1 - 6x + 15x^2 - 20x^3 + 15x^4 - 6x^5 + x^6 .
\]
Dans $(2x+3)^7$, le terme en $x^3$ est
$\binom{7}{3}(2x)^3\,3^4 = 35 \times 8 \times 81\, x^3$~: le coefficient
est $22\,680$.
\end{solution}

\begin{solution}{exo:g12:comb:5}
\emph{1.} $\dbinom{52}{5} = 2\,598\,960$.

\emph{2.} Exactement un as~: le choisir ($4$ façons) et compléter avec
$4$ non-as~: $4 \times \binom{48}{4} = 4 \times 194\,580 = 778\,320$.
Au moins un as~: dénombrement \omterm{def:g10:proba:operations}{complémentaire},
$\binom{52}{5} - \binom{48}{5} = 2\,598\,960 - 1\,712\,304 = 886\,656$.

\emph{3.} Choisir la valeur du brelan ($13$), ses couleurs
($\binom43 = 4$), la valeur de la paire ($12$ restantes), ses couleurs
($\binom42 = 6$)~:
$13 \times 4 \times 12 \times 6 = 3744$.
\end{solution}

\begin{solution}{exo:g12:comb:6}
MATH a 4 lettres distinctes~: $4! = 24$ anagrammes.

BANANA a 6 lettres~: trois A, deux N, un B. Choisir les positions des A
($\binom63$), puis des N parmi le reste ($\binom32$), le B prend la
dernière place~:
\[
\binom{6}{3}\binom{3}{2} = 20 \times 3 = 60 .
\]
(De façon équivalente $\frac{6!}{3!\,2!\,1!} = 60$.)
\end{solution}

\begin{solution}{exo:g12:comb:7}
\emph{Algébriquement~:}
\[
k\binom nk = \frac{k\,n!}{k!(n-k)!} = \frac{n!}{(k-1)!\,(n-k)!}
= n\,\frac{(n-1)!}{(k-1)!\bigl((n-1)-(k-1)\bigr)!} = n\binom{n-1}{k-1}.
\]
\emph{Par double dénombrement~:} compter les couples (comité de $k$,
président dedans). Soit choisir le comité ($\binom nk$) puis son
président ($k$)~: $k\binom nk$ couples. Soit choisir d'abord le président
($n$ options) puis les $k-1$ autres membres parmi les $n-1$ restants~:
$n\binom{n-1}{k-1}$ couples.
\end{solution}

\begin{solution}{exo:g12:comb:8}
Un chemin se compose d'exactement $m + n$ pas, dont $m$ sont Est et $n$
sont Nord~; il est entièrement déterminé par l'ensemble des instants
(parmi les $m+n$) où l'on avance vers l'Est. Il y a $\binom{m+n}{m}$
tels choix.
\end{solution}

\begin{solution}{exo:g12:comb:9}
Scinder un ensemble de $m + n$ personnes en un groupe $A$ de $m$ et un
groupe $B$ de $n$. Une partie à $k$ éléments contient un certain nombre
$j$ de membres de $A$ ($0 \leq j \leq k$) et $k - j$ membres de $B$~; pour
$j$ fixé il y a $\binom mj \binom{n}{k-j}$ telles parties, et le principe
d'addition sur $j$ donne l'identité de Vandermonde.

Avec $m = n = k$~:
\[
\binom{2n}{n} = \sum_{j=0}^n \binom nj \binom{n}{n-j}
= \sum_{j=0}^n \binom nj^{2},
\]
en utilisant la symétrie $\binom{n}{n-j} = \binom nj$.
\end{solution}

\begin{solution}{exo:g12:comb:10}
\emph{Via l'\cref{exo:g12:comb:7}~:}
\[
\sum_{k=1}^n k\binom nk = \sum_{k=1}^n n\binom{n-1}{k-1}
= n\sum_{j=0}^{n-1}\binom{n-1}{j} = n\,2^{n-1},
\]
par le \cref{cor:g12:comb:sums}. \emph{Via dérivation~:} dériver
$(1+x)^n = \sum_k \binom nk x^k$ donne
$n(1+x)^{n-1} = \sum_k k \binom nk x^{k-1}$~; évaluer en $x = 1$.
\end{solution}

\begin{solution}{pb:g12:comb:1}
\textbf{1.} $26^2 \times 10^3 = 676\,000$ plaques. BANANE~: $6$
lettres avec le A répété trois fois et le N deux fois~:
$\frac{6!}{3!\,2!} = 60$ anagrammes.

\textbf{2.} $\binom{32}{5} = 201\,376$ mains~;
$\binom42 \binom{28}{3} = 6 \times 3\,276 = 19\,656$ avec
exactement deux as.

\textbf{3.} Un chemin est un mot comportant $4$ D et $3$ H~: on
choisit la place des H, soit $\binom73 = 35$.

\textbf{4.} $(1+x)^4 = 1 + 4x + 6x^2 + 4x^3 + x^4$. En $x = 1$~:
$\sum_k \binom nk = 2^n$~; en $x = -1$~:
$\sum_k (-1)^k \binom nk = 0$ --- sommes des lignes et sommes
alternées des lignes du \omterm{prop:g11:binom:pascal}{triangle de Pascal}.

\textbf{5.} Comités de $k$ personnes munis d'un président, choisis
parmi $n$~: on choisit le comité puis son président
($\binom nk \times k$), ou bien le président puis les autres membres
($n \times \binom{n-1}{k-1}$)~: les deux comptages coïncident. En
sommant sur $k$, le membre de droite donne
$n \sum_j \binom{n-1}{j} = n\,2^{n-1}$.

\textbf{6.} Une rangée de $10$ étoiles et $3$ barres code la
commande (les boules du parfum 1 avant la première barre, etc.)~; la
rangée compte $13$ symboles et est déterminée par la position des
barres~: $\binom{13}{3} = 286$ commandes.

\textbf{7.} $12$ étoiles, $2$ barres~: $\binom{14}{2} = 91$.

\textbf{8.} Avec $x', y', z' \geq 0$ et $x' + y' + z' = 9$~:
$\binom{11}{2} = 55$.

\textbf{9.} Un monôme $a^i b^j c^k$ avec $i + j + k = 5$~:
$\binom72 = 21$.

\textbf{10.} Par la formule~: $3$ étoiles, $1$ barre, soit
$\binom41 = 4$~; par énumération~: $(3,0)$, $(2,1)$, $(1,2)$,
$(0,3)$~: cela concorde.

\textbf{11.} L'indiscernabilité est entrée au moment où l'on a
déclaré qu'une commande n'était rien d'autre que les
\emph{effectifs} par parfum --- les étoiles ne portent aucun nom. Si
les boules sont mangées dans l'ordre, chacune des $10$ positions
distinctes choisit librement un parfum~: $4^{10} = 1\,048\,576$
\omterm{def:g12:seq:sequence}{suites} --- un autre modèle et un autre monde
(\cref{met:g12:comb:model}~: toujours se demander \emph{ordonné~?
distinct~? répétitions autorisées~?}).

\textbf{12.} $D_1 = 0$~; $D_2 = 1$ (l'échange)~; $D_3 = 2$ (les
deux cycles de longueur $3$)~; $D_4 = 9$.

\textbf{13.} L'invité 1 reçoit le chapeau $k \neq 1$~: $n - 1$
choix. Si l'invité $k$ reçoit le chapeau 1, les $n - 2$ invités
restants dérangent leurs propres chapeaux~: $D_{n-2}$ façons. Si
l'invité $k$ ne reçoit \emph{pas} le chapeau 1, on renomme le
chapeau 1 en «~chapeau interdit de l'invité $k$~»~: les $n - 1$
invités restants dérangent~: $D_{n-1}$ façons. D'où
$D_n = (n-1)(D_{n-1} + D_{n-2})$. Vérification~:
$D_4 = 3(2 + 1) = 9$~; et $D_5 = 4(9 + 2) = 44$.

\textbf{14.} Sur les $3! = 6$ distributions, on retranche celles
qui fixent au moins un chapeau~: trois fixent un chapeau donné
($2!$ chacune, soit $3 \times 2 = 6$), ce qui compte deux fois les
paires ($3$ paires, $1!$ chacune) qu'il faut rajouter, et il faut
retrancher de nouveau l'identité ($1$)~:
$D_3 = 6 - 6 + 3 - 1 = 2$, c'est-à-dire
$3!\left(1 - 1 + \frac12 - \frac16\right) = 2$. En général,
$D_n = n!\sum_{k=0}^{n} \frac{(-1)^k}{k!}$.

\textbf{15.} $\frac{D_5}{120} = \frac{44}{120} \approx 0.3667$,
déjà proche de $\frac1\eu \approx 0.3679$~: la somme alternée
$1 - 1 + \frac{1}{2!} - \frac{1}{3!} + \dots$ marche vers
$\eu^{-1}$. Les chapeaux d'une grande soirée se dérangent
complètement environ $36.8\,\%$ du temps --- la constante de la
loterie et du recrutement, troisième observation.

\textbf{16.} $\P(\text{valide}) = \frac{D_{10}}{10!} \approx
0.368$. Chaque nouveau tirage réussit avec la \omterm{def:g10:proba:distribution}{probabilité}
$\approx \frac1\eu$, donc le nombre espéré de tirages vaut environ
$\eu \approx 2.7$~: prévoir trois passages du chapeau.

\textbf{17.} Chaque poignée de main contribue pour $2$ au total des
degrés, donc la somme des nombres de poignées de tous les invités
est paire. Or une somme d'entiers n'est paire que si le nombre de
termes impairs est pair~: les invités impairs vont par nombre pair.
(À trois invités~: aucun profil possible n'a exactement une ou trois
entrées \omterm{def:g11:func:parity}{impaires} --- vérifier les quatre graphes possibles.)

\textbf{18.} $n = 1$~: $1 = 1$. Si
$1^3 + \dots + n^3 = \left(\frac{n(n+1)}{2}\right)^2$, alors en
ajoutant $(n+1)^3$~:
\[
\frac{n^2(n+1)^2}{4} + (n+1)^3
= \frac{(n+1)^2\left(n^2 + 4n + 4\right)}{4}
= \left(\frac{(n+1)(n+2)}{2}\right)^{\!2} :
\]
hérédité. Pour $n = 3$~: $1 + 8 + 27 = 36 = 6^2$.

\textbf{19.} Un chemin menant à $(n, n)$ effectue $2n$ pas et
traverse l'antidiagonale $x + y = n$ en exactement un point du
réseau, $(j, n - j)$~; la première moitié est un chemin comportant
$j$ pas D parmi $n$ ($\binom nj$ choix), et la seconde moitié, lue à
rebours, de même ($\binom nj$ encore, par symétrie). En sommant sur
le point de traversée~: $\binom{2n}{n} = \sum_j \binom nj^2$ ---
l'identité de Vandermonde, dessinée.

\textbf{20.} Multiplier les étapes, additionner les cas~: les
plaques et les mains de cartes. Coder~: les chemins en mots DH, les
commandes en étoiles et barres. Compter deux fois~: les comités avec
président, les poignées de main, les chemins coupés en leur \omterm{prop:g10:coordgeom:midpoint}{milieu}.
Retrancher et corriger~: les chapeaux dérangés, avec $\frac1\eu$
pour résidu. Prochaines étapes~: ces dénombrements sous les
fractions des \omterm{def:g10:proba:distribution}{probabilités}, et la puissance de comptage de chemins
des matrices d'adjacence, deux chapitres plus loin.
\end{solution}
