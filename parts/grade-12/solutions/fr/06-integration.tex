\section*{Chapitre \ref{ch:g12:integ} --- Intégration}

\begin{solution}{exo:g12:integ:1}
\[
\int_1^2 \left(3x^2 - \frac{1}{x^2}\right)\dd x
= \left[x^3 + \frac1x\right]_1^2 = \left(8 + \tfrac12\right) - 2 = \frac{13}{2}.
\]
\[
\int_0^{\pi/2} \cos t\,\dd t = \bigl[\sin t\bigr]_0^{\pi/2} = 1 .
\]
Avec $u = 2t+1$, $\frac{1}{2t+1} = \frac12\,\frac{u'}{u}$~:
\[
\int_0^1 \frac{\dd t}{2t+1} = \frac12\bigl[\ln(2t+1)\bigr]_0^1
= \frac{\ln 3}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:2}
$x\,\eu^{x^2} = \frac12\,(x^2)'\,\eu^{x^2}$ a pour \omterm{def:g12:integ:primitive}{primitives}
$\frac12\eu^{x^2} + c$. La condition $F(0) = 1$ donne $\frac12 + c = 1$,
donc
\[
F(x) = \frac{\eu^{x^2} + 1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:3}
\[
\mu = \frac{1}{\pi}\int_0^\pi \sin t\,\dd t
= \frac{1}{\pi}\bigl[-\cos t\bigr]_0^\pi = \frac{2}{\pi} \approx 0.64 .
\]
Graphiquement, le rectangle de base $\intcc{0}{\pi}$ et de hauteur
$\frac{2}{\pi}$ a la même aire que l'arche de la courbe du \omterm{def:g12:trigo:cossin}{sinus}.
\end{solution}

\begin{solution}{exo:g12:integ:4}
Prendre $u' = 1$, $v = \ln t$, donc $u = t$, $v' = \frac1t$~:
\[
\int_1^{\eu} \ln t\,\dd t = \bigl[t\ln t\bigr]_1^{\eu} - \int_1^{\eu} 1\,\dd t
= \eu - (\eu - 1) = 1 .
\]
Prendre $u' = \sin t$, $v = t$, donc $u = -\cos t$, $v' = 1$~:
\[
\int_0^\pi t\sin t\,\dd t = \bigl[-t\cos t\bigr]_0^\pi + \int_0^\pi \cos t\,\dd t
= \pi + 0 = \pi .
\]
\end{solution}

\begin{solution}{exo:g12:integ:5}
Les courbes se coupent où $x^2 = x + 2$, \emph{c.-à-d.} $x \in \{-1, 2\}$,
et sur $\intcc{-1}{2}$ la droite est au-dessus de la \omterm{def:g11:quad:parabola}{parabole}. L'aire est
\[
\int_{-1}^{2} \bigl(x + 2 - x^2\bigr)\dd x
= \left[\frac{x^2}{2} + 2x - \frac{x^3}{3}\right]_{-1}^{2}
= \frac{10}{3} - \left(-\frac{7}{6}\right) = \frac{9}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:6}
\emph{1.} $I = \bigl[\ln(1+t)\bigr]_0^1 = \ln 2$.

\emph{2.}
\[
I_n + I_{n+1} = \int_0^1 \frac{t^n(1 + t)}{1+t}\,\dd t
= \int_0^1 t^n \,\dd t = \frac{1}{n+1}.
\]
Sur $\intcc{0}{1}$, $0 \leq \dfrac{t^n}{1+t} \leq t^n$, donc par
comparaison $0 \leq I_n \leq \frac{1}{n+1}$.

\emph{3.} À partir de la récurrence, par une récurrence télescopique,
\[
I_n = (-1)^n\left(I_0 - \left(1 - \frac12 + \dots + \frac{(-1)^{n-1}}{n}\right)\right),
\]
\emph{c.-à-d.}
$\displaystyle\sum_{k=1}^{n} \frac{(-1)^{k-1}}{k} = \ln 2 - (-1)^n I_n$
(rappel~: $I_0 = \ln 2$). Comme $\abs{I_n} \leq \frac{1}{n+1} \to 0$, la
somme harmonique alternée tend vers $\ln 2$.
\end{solution}

\begin{solution}{exo:g12:integ:7}
Distance~:
\[
d = \int_0^{100} 30\left(1 - \eu^{-t/50}\right)\dd t
= 30\left[t + 50\,\eu^{-t/50}\right]_0^{100}
= 30\left(100 + 50\eu^{-2} - 50\right)
= 1500 + 1500\,\eu^{-2}.
\]
Numériquement $d \approx 1500 + 203 = 1703$~m. Vitesse \omterm{def:g11:stat:mean}{moyenne}~:
$\frac{d}{100} = 15\bigl(1 + \eu^{-2}\bigr) \approx 17.0$~m/s.
\end{solution}

\begin{solution}{exo:g12:integ:8}
\emph{1.} $S_n$ est l'aire totale de $n$ rectangles de largeur $\frac1n$ et
de hauteurs $f\left(\frac kn\right)$, $k = 1, \dots, n$, où
$f(t) = \frac{1}{1+t}$~: c'est l'\omterm{def:g10:numbers:approx}{approximation} par rectangles «~à extrémité
droite~» de l'aire sous $f$ sur $\intcc{0}{1}$.

\emph{2.} $f$ est \omterm{def:g10:functions:variations}{décroissante}, donc sur chaque \omterm{def:g10:numbers:interval}{intervalle}
$\intcc{\frac{k-1}{n}}{\frac kn}$,
$f\left(\frac kn\right) \leq f(t) \leq f\left(\frac{k-1}{n}\right)$~;
en intégrant et en sommant sur $k$~:
\[
S_n \leq \int_0^1 f(t)\,\dd t \leq S_n + \frac1n\bigl(f(0) - f(1)\bigr)
= S_n + \frac{1}{2n}.
\]
(Le terme intermédiaire de l'affichage de l'exercice, avec
$f(0) - f(1) = 1 - \frac12$, est exactement cette borne.) D'où
$0 \leq \ln 2 - S_n \leq \frac{1}{2n} \to 0$, donc $S_n \to \ln 2$.
\end{solution}

\begin{solution}{exo:g12:integ:9}
\emph{1.} $W_0 = \int_0^{\pi/2} \dd t = \frac{\pi}{2}$ et
$W_1 = \bigl[-\cos t\bigr]_0^{\pi/2} = 1$.

\emph{2.} \omterm{thm:g12:integ:ibp}{Intégration par parties} avec $u' = \sin t$,
$v = \sin^{n+1} t$, donc $u = -\cos t$,
$v' = (n+1)\sin^n t \cos t$~:
\[
W_{n+2} = \bigl[-\cos t \sin^{n+1} t\bigr]_0^{\pi/2}
+ (n+1)\int_0^{\pi/2} \cos^2 t\,\sin^n t\,\dd t
= (n+1)\int_0^{\pi/2} (1 - \sin^2 t)\sin^n t\,\dd t,
\]
le crochet s'annulant aux deux extrémités. D'où
$W_{n+2} = (n+1)(W_n - W_{n+2})$, \emph{c.-à-d.}
$W_{n+2} = \frac{n+1}{n+2} W_n$.

\emph{3.} $W_2 = \frac12 W_0 = \frac{\pi}{4}$,
$W_3 = \frac23 W_1 = \frac23$,
$W_4 = \frac34 W_2 = \frac{3\pi}{16}$.
Sur $\intoo{0}{\frac\pi2}$, $0 < \sin t < 1$, donc
$\sin^{n+1} t \leq \sin^n t$ avec inégalité stricte à l'intérieur~; en
intégrant, $0 < W_{n+1} \leq W_n$ (en fait $<$)~: $(W_n)$ est \omterm{def:g10:functions:variations}{décroissante}
et positive.
\end{solution}

\begin{solution}{pb:g12:integ:1}
\textbf{1.} $\left[x^3 - x^2 + x\right]_0^1 = 1$~; \quad
$\left[\ln x\right]_1^{\eu} = 1$~; \quad
$\left[\sin x\right]_0^{\pi/2} = 1$.

\textbf{2.} $\left[\frac12 \eu^{x^2}\right]_0^1 =
\frac{\eu - 1}{2}$~; \quad $\left[\ln(x^2 + 1)\right]_0^1 =
\ln 2$.

\textbf{3.} $\int_0^1 x\eu^x = \left[x\eu^x\right]_0^1 -
\int_0^1 \eu^x = \eu - (\eu - 1) = 1$. Et
$\int_1^{\eu} \ln x = \left[x\ln x\right]_1^{\eu} -
\int_1^{\eu} 1 = \eu - (\eu - 1) = 1$.

\textbf{4.} $\frac{1}{\pi}\int_0^\pi \sin x\,\dd x =
\frac{2}{\pi} \approx 0.64$~: la hauteur \omterm{def:g11:stat:mean}{moyenne} de l'arche de
\omterm{def:g12:trigo:cossin}{sinus} vaut $\frac2\pi$, sensiblement plus que la moitié --- l'arche
est large près de son sommet.

\textbf{5.} $\int_0^1 (x - x^2)\,\dd x = \frac12 - \frac13 =
\frac16$.

\textbf{6.} $L_n = \frac1n \sum_{k=0}^{n-1}
\left(\frac kn\right)^2$ et
$U_n = \frac1n \sum_{k=1}^{n} \left(\frac kn\right)^2$~: des
rectangles de largeur $\frac1n$ dont la hauteur est lue à
l'extrémité gauche (sous la courbe) ou droite (au-dessus).

\textbf{7.} Vrai pour $n = 1$ ($1 = \frac{1 \cdot 2 \cdot 3}{6}$).
Si la formule vaut pour $n$, alors en ajoutant $(n+1)^2$~:
\[
\frac{n(n+1)(2n+1)}{6} + (n+1)^2
= \frac{(n+1)\left(2n^2 + n + 6n + 6\right)}{6}
= \frac{(n+1)(n+2)(2n+3)}{6},
\]
c'est-à-dire la formule au rang $n + 1$~: hérédité, c'est fait.

\textbf{8.} $U_n = \frac{1}{n^3} \cdot
\frac{n(n+1)(2n+1)}{6} = \frac{(1 + \frac1n)(2 + \frac1n)}{6}
\to \frac26 = \frac13$, et $L_n = U_n - \frac1n \to \frac13$~:
encadrée par ses escaliers, l'aire vaut exactement $\frac13$.

\textbf{9.} $\int_0^1 x^2\,\dd x =
\left[\frac{x^3}{3}\right]_0^1 = \frac13$~: une ligne. Le théorème
fondamental convertit un encadrement infini en une seule évaluation
de \omterm{def:g12:integ:primitive}{primitive} --- et c'est pourquoi on le dit fondamental.

\textbf{10.} Segment~: $\int_{-1}^{1}(1 - x^2)\,\dd x =
2 - \frac23 = \frac43$. Triangle de base la corde (de longueur $2$,
à la hauteur $1$) et de sommet $(0,0)$~: aire
$\frac12 \times 2 \times 1 = 1$. Rapport~: $\frac43$ ---
exactement le théorème d'Archimède, vérifié par la machine qui lui
manquait.

\textbf{11.} $T\left(1 + \frac14 + \frac{1}{16} + \dots\right)
= T \cdot \frac{1}{1 - \frac14} = \frac43 T$ (série géométrique de
raison $\frac14$)~: avec $T = 1$, on retrouve l'aire $\frac43$ du
segment. Archimède sommait la série par un argument purement
géométrique --- les bouchées d'un quart, à la manière de la tablette
de chocolat, deux millénaires avant que le mot «~limite~» n'existe.

\textbf{12.} L'exhaustion réclame une astucieuse autosimilitude
géométrique et donne des aires exactes au cas par cas~; Riemann ne
réclame que la monotonie et donne une définition universelle, au
prix du calcul de sommes~; le théorème fondamental réclame tout
l'édifice des \omterm{def:g12:deriv:derivative}{dérivées} et le rembourse en faisant des aires des
calculs d'une ligne.

\textbf{13.} $A(1) = 0$ et $A'(x) = \frac1x$ (théorème
fondamental)~: $A$ est la \omterm{def:g12:integ:primitive}{primitive} de $\frac1x$ sur
$\intoo{0}{+\infty}$ qui s'annule en $1$ --- c'est-à-dire
précisément le \omterm{def:g12:exp:ln}{logarithme népérien}.

\textbf{14.} $g'(x) = \frac{a}{ax} - \frac1x = 0$~: $g$ est
constante~; en $x = 1$, $g(1) = A(a) - A(1) = A(a)$. Donc
$A(ax) = A(a) + A(x)$ pour tout $x$, et avec $x = b$~:
$\ln(ab) = \ln a + \ln b$. L'aire sous l'\omterm{def:g11:func:reference}{hyperbole} se moque de
l'échelle horizontale~: dilater $t$ d'un facteur $a$ écrase
$\frac1t$ du même facteur, et l'aire survit --- l'addition des
\omterm{def:g12:exp:ln}{logarithmes} est une symétrie d'échelle.

\textbf{15.} Par itération~: $\ln(a^n) = n\ln a$ (récurrence sur
$n$). Et $0 = \ln 1 = \ln\left(a \cdot \frac1a\right) =
\ln a + \ln\frac1a$, donc $\ln\frac1a = -\ln a$.

\textbf{16.} $\frac{1}{11} + \frac{1}{12} + \dots +
\frac{1}{20} \approx 0.669$, contre $\ln 2 \approx 0.693$~: dix
rectangles sous $\frac{1}{1+t}$, et déjà proches. La série
harmonique diverge, mais ses \emph{tranches} de $n$ à $2n$ se
stabilisent sur $\ln 2$~: une divergence au pas de tortue, mesurée
par une aire.

\textbf{17.} Distance~: $\int_0^{10} 3t^2\,\dd t = 1000$ m. Vitesse
\omterm{def:g11:stat:mean}{moyenne}~: $100$ m/s. Instant où $v = 100$~: $3t^2 = 100$, donc
$t = \sqrt{\frac{100}{3}} \approx 5.77$ s.

\textbf{18.} Masse~: $\int_0^4 (2 + x)\,\dd x = 8 + 8 = 16$ kg.
Moment~: $\int_0^4 (2x + x^2)\,\dd x = 16 + \frac{64}{3} =
\frac{112}{3}$. Centre de masse~:
$\frac{112/3}{16} = \frac73 \approx 2.33$ m --- repoussé au-delà du
\omterm{prop:g10:coordgeom:midpoint}{milieu} par l'extrémité droite plus lourde, comme l'intuition
l'exige.

\textbf{19.} $W_0 = \frac\pi2$~; $W_1 = \left[-\cos
t\right]_0^{\pi/2} = 1$~; et
$W_2 = \int_0^{\pi/2} \frac{1 - \cos 2t}{2}\,\dd t = \frac\pi4$.

\textbf{20.} Aire~: la définition, encadrée par des escaliers.
Accumulation~: des distances à partir de vitesses, des masses à
partir de densités --- l'\omterm{def:g12:integ:area}{intégrale} comme total courant. \omterm{def:g12:integ:primitive}{Primitive}~:
le théorème fondamental, le raccourci de Newton et de Leibniz à
travers la patience d'Archimède. La perle~: la touche $\ln$ est
l'aire sous $\frac1t$, et la symétrie d'échelle de cette aire
\emph{est} l'identité $\ln(ab) = \ln a + \ln b$.
\end{solution}
