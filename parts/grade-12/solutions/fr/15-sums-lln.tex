\section*{Chapitre \ref{ch:g12:sums} --- Sommes de variables aléatoires et loi des grands nombres}

\begin{solution}{exo:g12:sums:1}
Chaque dé a pour \omterm{def:g12:randvar:exp}{espérance}
$\frac{1 + 2 + \dots + 6}{6} = \frac{7}{2}$, donc par linéarité
$\E(S) = \frac72 + \frac72 = 7$. Les dés sont \omterm{def:g12:condprob:indep}{indépendants}, donc
$\V(S) = \frac{35}{12} + \frac{35}{12} = \frac{35}{6} \approx 5.83$.
\end{solution}

\begin{solution}{exo:g12:sums:2}
$\E(X) = 50$ et $\V(X) = 100 \times 0.25 = 25$.

Markov ($X \geq 0$)~: $\P(X \geq 75) \leq \frac{50}{75} = \frac23$.

Bienaymé--Tchebychev~:
$\P(\abs{X - 50} \geq 25) \leq \frac{25}{25^2} = 0.04$ --- et cela borne
même l'\omterm{def:g11:prob:model}{événement} \emph{bilatéral}, dont $X \geq 75$ est la moitié.
Tchebychev est bien plus fin ici car il exploite la \omterm{def:g12:randvar:exp}{variance}, pas seulement
la \omterm{def:g11:stat:mean}{moyenne}. (La vraie valeur de $\P(X \geq 75)$ est inférieure à
$10^{-6}$~: les deux bornes sont grossières.)
\end{solution}

\begin{solution}{exo:g12:sums:3}
$F_n$ est la \omterm{prop:g12:sums:mean}{moyenne empirique} de $n$ variables de Bernoulli
$\left(\frac12\right)$, de \omterm{def:g12:randvar:exp}{variance} $\sigma^2 = \frac14$. La borne
\[
\P\left(\abs{F_n - 0.5} \geq 0.05\right)
\leq \frac{1/4}{n \times 0.05^2} = \frac{100}{n}
\]
est $\leq 0.05$ dès que $n \geq 2000$.
\end{solution}

\begin{solution}{exo:g12:sums:4}
Par la \cref{prop:g12:sums:mean}, $\E(M_n) = 5\%$ (la diversification ne
change pas le rendement espéré) et
$\sigma(M_n) = \frac{20\%}{\sqrt n}$. Exiger
$\frac{20}{\sqrt n} < 4$ donne $\sqrt n > 5$, \emph{c.-à-d.} $n \geq 26$.
C'est le principe de \emph{diversification}~: répartir le capital sur des
risques \omterm{def:g12:condprob:indep}{indépendants} divise le risque (\omterm{def:g11:stat:variance}{écart type}) par $\sqrt n$ sans
réduire le rendement espéré.
\end{solution}

\begin{solution}{exo:g12:sums:5}
\emph{1.} En comptant les $9$ couples équiprobables~: $S$ prend les
valeurs $2, 3, 4, 5, 6$ avec \omterm{def:g10:proba:distribution}{probabilités}
$\frac19, \frac29, \frac39, \frac29, \frac19$. D'où
$\E(S) = \frac{2 + 6 + 12 + 10 + 6}{9} = 4$ et
$\E(S^2) = \frac{4 + 18 + 48 + 50 + 36}{9} = \frac{156}{9} = \frac{52}{3}$,
donc $\V(S) = \frac{52}{3} - 16 = \frac43$.

\emph{2.} Une variable~: $\E(X) = 2$,
$\E(X^2) = \frac{1 + 4 + 9}{3} = \frac{14}{3}$,
$\V(X) = \frac{14}{3} - 4 = \frac23$. Puis $\E(S) = 2 + 2 = 4$ et, par
\omterm{def:g12:condprob:indep}{indépendance}, $\V(S) = \frac23 + \frac23 = \frac43$. Mêmes valeurs.
\end{solution}

\begin{solution}{exo:g12:sums:6}
Avec $Y = X$~: $\V(X + Y) = \V(2X) = 4\V(X)$, tandis que
$\V(X) + \V(Y) = 2\V(X)$. Les deux coïncident seulement lorsque
$\V(X) = 0$, \emph{c.-à-d.} lorsque $X$ est constante --- donc
l'additivité exige vraiment l'\omterm{def:g12:condprob:indep}{indépendance} (ici $X$ est maximalement
dépendante d'elle-même).
\end{solution}

\begin{solution}{exo:g12:sums:7}
Sous l'hypothèse d'équilibre, $F_n$ est la \omterm{def:g11:stat:mean}{moyenne} de $n = 1200$ variables
de Bernoulli avec $p = \frac16$,
$\sigma^2 = \frac16\cdot\frac56 = \frac{5}{36}$~:
\[
\P\left(\abs{F_n - \tfrac16} \geq 0.05\right)
\leq \frac{5/36}{1200 \times 0.0025} = \frac{5}{108} \approx 0.046 .
\]
L'écart observé est exactement $0.05$~: un \omterm{def:g11:prob:model}{événement} qu'un dé équilibré
produit avec \omterm{def:g10:proba:distribution}{probabilité} au plus $4.6\%$ --- et comme Tchebychev est très
conservateur, la vraie \omterm{def:g10:proba:distribution}{probabilité} est bien plus petite. L'hypothèse
d'équilibre n'est pas plausible~; le dé est très probablement pipé.
\end{solution}

\begin{solution}{exo:g12:sums:8}
\emph{1.} $p(1-p) = \frac14 - \left(p - \frac12\right)^2 \leq \frac14$.

\emph{2.} $F_n$ a pour \omterm{def:g12:randvar:exp}{espérance} $p$ et pour \omterm{def:g12:randvar:exp}{variance}
$\frac{p(1-p)}{n} \leq \frac{1}{4n}$~; Bienaymé--Tchebychev donne
\[
\P\bigl(\abs{F_n - p} \geq \varepsilon\bigr)
\leq \frac{p(1-p)}{n\varepsilon^2} \leq \frac{1}{4n\varepsilon^2},
\]
valable quel que soit le $p$ inconnu.

\emph{3.} Avec $\varepsilon = 0.03$ et niveau $0.05$~:
\[
\frac{1}{4n(0.03)^2} \leq 0.05
\iff n \geq \frac{1}{4 \times 0.0009 \times 0.05} \approx 5556 .
\]
Environ $5600$ personnes suffisent par cette borne grossière (la réponse
classique par \omterm{def:g10:numbers:approx}{approximation} normale est plus proche de $1100$, voir le
\cref{ch:g12:contdist}).
\end{solution}

\begin{solution}{pb:g12:sums:1}
\textbf{1.} $V(X + Y) = V(X) + V(Y) = \frac{35}{12} +
\frac{35}{12} = \frac{35}{6}$ (par \omterm{def:g12:condprob:indep}{indépendance}).

\textbf{2.} $\E(\bar X) = 3.5$~;
$\sigma(\bar X) = \frac{1.71}{\sqrt{100}} \approx 0.17$~: la
\omterm{def:g11:stat:mean}{moyenne} de cent dés serre $3.5$ à un cinquième de point près.

\textbf{3.} $\P(X \geq 10) \leq \frac{2}{10} = 0.2$.

\textbf{4.} $\P \leq \frac{V(\bar X)}{0.5^2} =
\frac{35/1200}{0.25} \approx 0.117$~: au plus $12\,\%$ environ.

\textbf{5.} Avec $Y = -X$~: $V(X + Y) = V(0) = 0$, alors que
$V(X) + V(Y) = 2V(X) > 0$~: des variables parfaitement
anticorrélées se compensent --- l'additivité des \omterm{def:g12:randvar:exp}{variances} est un
privilège de l'\omterm{def:g12:condprob:indep}{indépendance}.

\textbf{6.} $\E = \frac{18}{37} - \frac{19}{37} = -\frac{1}{37}
\approx -0.027$~;
$\sigma = \sqrt{1 - \left(\frac{1}{37}\right)^2} \approx 1.00$.

\textbf{7.} $\E(G) = -2.70$, $\sigma(G) = 10 \times 1.00 = 10$~:
la dérive ne vaut que $0.27\sigma$. Le bruit d'une soirée écrase
l'avantage de la banque --- une large minorité de joueurs (environ
$40\,\%$, dit la cloche) repart gagnante, et c'est précisément ce
qui les fait revenir.

\textbf{8.} Recette du casino~: \omterm{def:g12:randvar:exp}{espérance} $+27\,027$ euros, \omterm{def:g11:stat:variance}{écart
type} environ $1\,000$~: le bénéfice se situe $27$ \omterm{def:g11:stat:variance}{écarts types}
au-dessus de zéro. À $27\sigma$, «~le casino pourrait perdre cette
année~» n'est pas un risque, c'est une erreur d'arrondi.

\textbf{9.} $\P(\text{perte}) = \P(G \geq 27\,027$ pour les
joueurs$)$ $\leq \frac{10^6 \times 1}{27\,027^2} \approx 0.0014$~:
même l'inégalité la plus grossière du livre garantit la banque à
$99.86\,\%$ --- la vérité est astronomiquement plus forte.

\textbf{10.} Chaque euro misé, quand et comme que ce soit, a un
rendement espéré de $-\frac{1}{37}$ de lui-même~; par linéarité, le
gain total espéré vaut $-\frac{1}{37} \times (\text{total misé})$,
négatif pour toute stratégie qui mise quelque chose. Une martingale
gagnante violerait la linéarité de l'\omterm{def:g12:randvar:exp}{espérance} --- aucun
ordonnancement astucieux de mauvais paris n'en fait un bon. (Les
martingales de doublement échangent simplement beaucoup de petits
gains contre de rares pertes catastrophiques.)

\textbf{11.} Le théorème~: la \emph{\omterm{def:g10:stats:series}{fréquence}} du rouge sur $n$
coups \omterm{def:g12:seq:limit}{converge} (en \omterm{def:g10:proba:distribution}{probabilité}) vers $\frac{18}{37}$. Il ne dit
rien du coup $n + 1$~: la roue n'a pas de mémoire, et après dix
rouges la \omterm{def:g10:proba:distribution}{probabilité} du rouge vaut toujours $\frac{18}{37}$.
Croire le contraire, c'est l'\emph{illusion du joueur}, et la
question 12 montre ce qu'il advient réellement des séries.

\textbf{12.} L'excédent de $100$ rouges n'est pas remboursé --- les
coups à venir sont des copies équitables, et l'excédent espéré reste
$100$. Mais $\frac{100}{10^6} = 0.0001$~: un centième de point.
L'erreur du sophisme en une phrase~: \emph{la loi des grands nombres
dilue les accidents passés dans un océan d'épreuves nouvelles~; elle
n'envoie jamais la roue recouvrer des dettes.}

\textbf{13.} $n \geq \frac{1}{4 \times 0.05 \times 0.03^2}
\approx 5\,556$ personnes.

\textbf{14.} $\frac{1.96^2 \times 0.25}{0.03^2} \approx 1\,067$~:
cinq fois moins. Tchebychev vaut pour \emph{toutes} les \omterm{def:g12:randvar:rv}{lois} et paie
son universalité par du mou~; la constante $1.96$ des sondeurs vient
de la vraie forme en cloche des fluctuations, qui se concentre bien
davantage.

\textbf{15.} La marge est proportionnelle à $\frac{1}{\sqrt n}$~:
la diviser par deux quadruple l'\omterm{def:g10:proba:sample}{échantillon} --- environ $4\,400$
personnes pour $\pm 1.5$ point. La précision s'achète à prix
quadratique.

\textbf{16.} Le modèle est celui de $n$ tirages \omterm{def:g12:condprob:indep}{indépendants} de
\omterm{def:g10:proba:distribution}{probabilité} de succès $p$ --- la taille $N$ de la population n'y
apparaît jamais (sonder une infime fraction d'une grande population
bien brassée, voilà ce que code l'\omterm{def:g12:condprob:indep}{indépendance}). L'analogie des
sondeurs~: pour goûter la soupe, une cuillerée bien remuée suffit
--- que la marmite serve dix personnes ou dix mille. Le piège est le
brassage --- un plan de sondage biaisé, comme dans le sondage du
volume précédent qui a trompé tout un pays --- et non la marmite.

\textbf{17.} $315\sqrt n < 30n \iff \sqrt n > 10.5 \iff n > 110$~:
au-delà d'une centaine de contrats, la dérive distance le bruit, et
chaque contrat supplémentaire creuse l'écart selon la \omterm{def:g12:randvar:rv}{loi} en
$\sqrt n$ --- la mutualisation \emph{est} le modèle économique.

\textbf{18.} Parts égales \omterm{def:g12:sums:indep}{indépendantes}~: $\sigma_{\text{pf}} =
\frac{20\,\%}{\sqrt{25}} = 4\,\%$~: même rendement espéré, un
cinquième de la fluctuation. Le repas~: une réduction du risque à
coût nul en \omterm{def:g12:randvar:exp}{espérance} --- moyenner du hasard indépendant.

\textbf{19.} Parfaitement corrélées~: le portefeuille n'est qu'une
seule action en $25$ costumes, donc $\sigma = 20\,\%$, sans la
moindre réduction. La diversification loue l'\emph{\omterm{def:g12:condprob:indep}{indépendance}}~;
lorsqu'une crise corrèle tout (2008~: tous les paris immobiliers
n'en faisaient qu'un), la protection en $\sqrt n$ s'évapore
exactement au moment où l'on en a besoin.

\textbf{20.} Les sommes dérivent comme $n$ et fluctuent comme
$\sqrt n$, si bien que les \omterm{def:g11:stat:mean}{moyennes} se stabilisent comme
$\frac{1}{\sqrt n}$~: le casino encaisse la dérive sur un million
de coups~; le sondeur achète $\frac{1}{\sqrt{1100}}$ de bruit pour
un budget de téléphone~; l'assureur dépasse ses propres sinistres
dès $n > 110$~; l'investisseur divise le risque par $\sqrt{25}$.
Contrevenants~: le joueur qui croit aux dettes (il n'y a que
dilution) et le financier qui croit à l'\omterm{def:g12:condprob:indep}{indépendance} (il n'y a
parfois qu'un seul pari). La forme universelle des fluctuations ---
la cloche --- est la vedette du chapitre continu qui vient, et son
théorème, le théorème central limite, est le sommet des
\omterm{def:g10:proba:distribution}{probabilités} des volumes universitaires.
\end{solution}
