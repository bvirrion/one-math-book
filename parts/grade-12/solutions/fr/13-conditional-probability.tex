\section*{Chapitre \ref{ch:g12:condprob} --- Probabilité conditionnelle et indépendance}

\begin{solution}{exo:g12:condprob:1}
Il y a $12$ figures, dont $4$ rois~:
$\pcond{\text{figure}}{\text{roi}} = \frac{4}{12} = \frac13$.

\omterm{def:g12:condprob:indep}{Indépendance}~: $\P(\text{roi} \cap \text{cœur}) = \frac{1}{52}$ (le roi
de cœur), et
$\P(\text{roi})\,\P(\text{cœur}) = \frac{4}{52}\times\frac{13}{52}
= \frac{1}{52}$. Égaux~: les \omterm{def:g11:prob:model}{événements} sont \omterm{def:g12:condprob:indep}{indépendants}.
\end{solution}

\begin{solution}{exo:g12:condprob:2}
\emph{1.} Première branche~: rouge $\frac58$, bleue $\frac38$~; secondes
branches (sans remise)~: après rouge, rouge $\frac47$ / bleue $\frac37$~;
après bleue, rouge $\frac57$ / bleue $\frac27$.

\emph{2.} $\P(RR) = \frac58 \times \frac47 = \frac{5}{14}$. Par les
\omterm{def:g10:proba:distribution}{probabilités} totales,
\[
\P(\text{2e rouge}) = \frac58\cdot\frac47 + \frac38\cdot\frac57
= \frac{20 + 15}{56} = \frac58 .
\]
(La même que le premier tirage --- par symétrie, la seconde boule est une
boule uniformément aléatoire de l'urne.)
\end{solution}

\begin{solution}{exo:g12:condprob:3}
$\P(A) = \frac{6}{36} = \frac16$ (six couples de somme $7$),
$\P(B) = \frac16$, et $A \cap B = \{(3,4)\}$ a pour \omterm{def:g10:proba:distribution}{probabilité}
$\frac{1}{36} = \P(A)\P(B)$~: $A$ et $B$ sont \omterm{def:g12:condprob:indep}{indépendants}. (La somme $7$
est spéciale~: quel que soit le premier dé, exactement une valeur du
second la donne.)

$\P(C) = \frac{5}{36}$ et $B \cap C = \{(3,3)\}$~:
$\P(B \cap C) = \frac{1}{36} \neq \frac16 \times \frac{5}{36}
= \frac{5}{216}$. Non \omterm{def:g12:condprob:indep}{indépendants}.
\end{solution}

\begin{solution}{exo:g12:condprob:4}
\emph{1.} \omterm{def:g10:proba:distribution}{Probabilités} totales avec la partition par machine~:
\[
\P(D) = 0.5\times0.01 + 0.3\times0.02 + 0.2\times0.04
= 0.005 + 0.006 + 0.008 = 0.019 = 1.9\% .
\]

\emph{2.} Bayes~:
$\pcond{D}{M_3} = \dfrac{0.008}{0.019} = \dfrac{8}{19} \approx 0.42$. La
machine produisant seulement un cinquième de la production représente plus
de $40\%$ des défauts.
\end{solution}

\begin{solution}{exo:g12:condprob:5}
La condition est
\[
\frac{0.99\,p}{0.99\,p + 0.05\,(1-p)} \geq 0.9 .
\]
Le dénominateur est positif, donc cela s'écrit
$0.99p \geq 0.891p + 0.045 - 0.045p$, \emph{c.-à-d.} $0.144\,p \geq 0.045$,
\emph{c.-à-d.}
\[
p \geq \frac{0.045}{0.144} = 0.3125 .
\]
Le test seul atteint $90\%$ de certitude seulement si la maladie touche
déjà plus de $31\%$ de la population testée --- d'où la nécessité de tests
de confirmation pour le dépistage de masse des maladies rares.
\end{solution}

\begin{solution}{exo:g12:condprob:6}
\emph{1.} Trois \omterm{def:g12:seq:sequence}{suites} réalisent $E$ (PPF, PFP, FPP), chacune de
\omterm{def:g10:proba:distribution}{probabilité} $p^2(1-p)$ par \omterm{def:g12:condprob:indep}{indépendance}~: $\P(E) = 3p^2(1 - p)$.

\emph{2.} $f(p) = 3p^2 - 3p^3$ a $f'(p) = 6p - 9p^2 = 3p(2 - 3p)$,
positive puis négative sur $\intoo{0}{1}$~: \omterm{def:g10:functions:extrema}{maximum} en $p = \frac23$, où
$\P(E) = 3\cdot\frac49\cdot\frac13 = \frac49$.
\end{solution}

\begin{solution}{exo:g12:condprob:7}
Disons que vous avez choisi la porte~1 et que l'animateur a ouvert la
porte~3 (\omterm{def:g11:prob:model}{événement} $H_3$). Avec $B_i$ = «~prix derrière la porte $i$~»,
$\P(B_i) = \frac13$ et
\[
\pcond{B_1}{H_3} = \frac12, \qquad
\pcond{B_2}{H_3} = 1, \qquad
\pcond{B_3}{H_3} = 0
\]
(si le prix est derrière votre porte, l'animateur choisit au hasard entre
les portes~2 et~3~; s'il est derrière la porte~2, il est forcé d'ouvrir la
$3$). Bayes~:
\[
\pcond{H_3}{B_1}
= \frac{\frac13\cdot\frac12}{\frac13\cdot\frac12 + \frac13\cdot1 + 0}
= \frac{1/6}{1/2} = \frac13,
\qquad
\pcond{H_3}{B_2} = \frac{\frac13}{\frac12} = \frac23 .
\]
Rester gagne avec \omterm{def:g10:proba:distribution}{probabilité} $\frac13$, basculer avec \omterm{def:g10:proba:distribution}{probabilité}
$\frac23$.
\end{solution}

\begin{solution}{exo:g12:condprob:8}
\emph{1.} La majorité est fausse lorsque au moins deux des trois copies
sont inversées~:
\[
3\varepsilon^2(1-\varepsilon) + \varepsilon^3
= 3(0.01)(0.9) + 0.001 = 0.028,
\]
bien plus petit que $\varepsilon = 0.1$~: le codage par répétition
fonctionne.

\emph{2.} Si $1$ a été envoyé (comme $111$), recevoir $101$ exige
exactement une inversion~: \omterm{def:g10:proba:distribution}{probabilité}
$\varepsilon(1-\varepsilon)^2 = 0.081$. Si $0$ a été envoyé ($000$),
deux inversions sont nécessaires~: $\varepsilon^2(1-\varepsilon) = 0.009$.
Bayes avec prioris égaux~:
\[
\pcond{101}{\,\text{envoyé }1} = \frac{0.081}{0.081 + 0.009} = 0.9 .
\]
Le décodage majoritaire ($101 \mapsto 1$) est bien le guess le plus
probable.
\end{solution}

\begin{solution}{pb:g12:condprob:1}
\textbf{1.} $\P(\text{rouge}) = \frac12 \cdot \frac23 + \frac12
\cdot \frac14 = \frac13 + \frac18 = \frac{11}{24}$.

\textbf{2.} $\P(\text{I} \mid \text{rouge}) =
\dfrac{\frac12 \cdot \frac23}{\frac{11}{24}} =
\dfrac{1/3}{11/24} = \dfrac{8}{11}$.

\textbf{3.} $\P(\text{pair} \cap \leq 4) = \P(\{2, 4\}) =
\frac13$ et $\P(\text{pair})\P(\leq 4) = \frac12 \cdot
\frac23 = \frac13$~: les \omterm{def:g11:prob:model}{événements} sont \omterm{def:g12:condprob:indep}{indépendants} (contre toute
attente~!). Pour «~$\leq 3$~»~: $\P(\{2\}) = \frac16$ contre
$\frac12 \cdot \frac12 = \frac14$~: ils sont dépendants.

\textbf{4.} Les \omterm{def:g11:prob:model}{événements} contraires d'\omterm{def:g11:prob:model}{événements} \omterm{def:g12:condprob:indep}{indépendants}
sont eux-mêmes \omterm{def:g12:condprob:indep}{indépendants}, donc
\[
\P(\text{aucun}) = 0.1 \times 0.1 = 0.01,
\qquad
\P(\text{au moins un}) = 0.99 .
\]

\textbf{5.} $\frac{4}{52} \times \frac{3}{51} = \frac{1}{221}$.

\textbf{6.} Malades~: $100$~; bien portants~: $99\,900$.
Positifs~: $99$ vrais ($99\,\%$ de $100$) et $4\,995$ faux
($5\,\%$ de $99\,900$), soit $5\,094$ positifs en tout.

\textbf{7.} $\P(\text{malade} \mid +) = \frac{99}{5\,094}
\approx 1.9\,\%$~: parmi les personnes que le test signale, moins
d'une sur cinquante est réellement malade.

\textbf{8.} $\dfrac{0.001 \times 0.99}{0.001 \times 0.99 +
0.999 \times 0.05} = \dfrac{0.00099}{0.05094} \approx 0.019$~: les
effectifs et la formule concordent.

\textbf{9.} On confond $\P(+ \mid \text{malade}) = 99\,\%$ (la
qualité annoncée du test) avec $\P(\text{malade} \mid +)$ (la
question que se pose réellement le patient). Avec un taux de base
de $\frac{1}{1000}$, les malades sont si rares que même une fuite
de $5\,\%$ de l'immense majorité bien portante submerge les vrais
positifs à cinquante contre un.

\textbf{10.} $\dfrac{0.019 \times 0.99}{0.019 \times 0.99 +
0.981 \times 0.05} \approx 28\,\%$. Chaque positif indépendant
multiplie la cote par le même facteur~: les indices s'accumulent,
et un troisième positif dépasserait $90\,\%$ --- raison d'être des
tests de confirmation.

\textbf{11.} $\dfrac{0.05 \times 0.99}{0.05 \times 0.99 +
0.95 \times 0.05} \approx 51\,\%$~: sur un groupe à risque, le même
test devient informatif. Morale~: dépister là où le taux de base
est appréciable~; dépister massivement une affection rare fabrique
des fausses alertes par milliers.

\textbf{12.} Nombre attendu de correspondances innocentes~:
$10^7 \times 10^{-6} = 10$. Parmi la petite douzaine de personnes
qui correspondent ($10$ innocentes et $1$ coupable --- si le
coupable est bien dans la ville), le suspect trouvé \emph{par la
seule correspondance} est innocent avec une \omterm{def:g10:proba:distribution}{probabilité} d'environ
$\frac{10}{11} \approx 91\,\%$.

\textbf{13.} L'accusation a cité
$\P(\text{correspondance} \mid \text{innocent}) = 10^{-6}$~; le
tribunal avait besoin de
$\P(\text{innocent} \mid \text{correspondance})$ --- la confusion
de la question 9, prêtée sous serment. (Et dans l'affaire réelle
qui a popularisé ces erreurs, une seconde bévue est venue
s'ajouter~: voir la question 14.)

\textbf{14.} Élévation au carré naïve~:
$\left(\frac{1}{8500}\right)^2 \approx \frac{1}{72 \text{
millions}}$. Avec un second drame cinq fois plus probable sachant
le premier~: $\frac{1}{8500} \times \frac{5}{8500} \approx
\frac{1}{14.5 \text{ millions}}$ --- cinq fois plus grand, et
encore sans tenir compte du fait que l'hypothèse \emph{alternative}
doit elle aussi être pesée par Bayes. Leçon~: multiplier des
\omterm{def:g10:proba:distribution}{probabilités} exige l'\omterm{def:g12:condprob:indep}{indépendance}, et des causes communes
(génétique, environnement) la détruisent.

\textbf{15.} Bayes multiplie les rapports de vraisemblance de
\emph{tous} les indices en une seule \omterm{def:g10:proba:distribution}{probabilité} a posteriori~: la
correspondance, l'alibi, les fibres, chacun déplaçant la cote. La
phrase du juré~: \emph{se demander quelle est la \omterm{def:g10:proba:distribution}{probabilité} des
indices sous chacune des deux hypothèses, et ne jamais prendre une
\omterm{def:g12:condprob:cond}{probabilité conditionnelle} pour sa réciproque}.

\textbf{16.} $\dfrac{0.4 \times 0.15}{0.4 \times 0.15 +
0.6 \times 0.005} = \dfrac{0.060}{0.063} \approx 95\,\%$ de spam~:
un seul mot tranche presque.

\textbf{17.} Cinq copies échouent lorsque $3$, $4$ ou $5$ bits
s'inversent~:
$\binom53 \varepsilon^3(1-\varepsilon)^2 + \binom54
\varepsilon^4 (1 - \varepsilon) + \varepsilon^5 \approx 0.0086$ ---
contre $0.028$ pour trois copies~: la répétition achète de la
fiabilité au prix du débit.

\textbf{18.} Nombre pair d'inversions~: $\binom40 0.9^4 +
\binom42 0.9^2 0.1^2 + \binom44 0.1^4 = 0.6561 + 0.0486 + 0.0001 =
0.7048$~; et $\frac{1 + 0.8^4}{2} = \frac{1 + 0.4096}{2} =
0.7048$~: la formule astucieuse suit le biais $(1 - 2\varepsilon)$
de relais en relais.

\textbf{19.} Les mots de spam voyagent en bande --- un message
contenant «~loterie~» contient très probablement «~gagner~», si
bien que leur indice conjoint est plus faible que ne le prétend le
produit. Le filtre fonctionne malgré tout parce que l'exagération
pousse en général dans le \emph{bon sens}~: classer demande le bon
côté de $50\,\%$, pas la bonne \omterm{def:g10:proba:distribution}{probabilité} a posteriori.

\textbf{20.} Clinique~: une affection rare transforme un test
puissant en verdict faible --- partir du taux de base, puis
accumuler. Tribunal~: une correspondance rare dans une grande ville
met en cause la ville, non l'homme --- une vraisemblance n'est pas
un verdict. Boîte de réception~: de nombreux indices faibles,
multipliés, trient le courrier --- même naïvement. Les trois
pièges~: oublier l'a priori et se laisser tromper~; feindre
l'\omterm{def:g12:condprob:indep}{indépendance} et se laisser tromper bien davantage~; lire les
indices un par un et ne jamais savoir ce que l'on sait.
\end{solution}
