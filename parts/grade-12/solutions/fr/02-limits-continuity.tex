\section*{Chapitre \ref{ch:g12:limcont} --- Limites et continuité}

\begin{solution}{exo:g12:limcont:1}
En divisant par $x^3$~:
$\dfrac{2x^3 - x + 1}{5x^3 + x^2} = \dfrac{2 - 1/x^2 + 1/x^3}{5 + 1/x}
\to \dfrac{2}{5}$.

$x^2 + 3x - 1 = x^2(1 + 3/x - 1/x^2) \to +\infty$ lorsque
$x \to -\infty$, car $x^2 \to +\infty$ et le crochet tend vers $1$.

Lorsque $x \to 2^+$, le numérateur tend vers $3 > 0$ et le dénominateur
vers $0^+$, donc $\dfrac{x+1}{x-2} \to +\infty$.

Pour $x > 0$, $-\dfrac1x \leq \dfrac{\cos x}{x} \leq \dfrac1x$, donc la
limite est $0$ par le théorème des gendarmes.
\end{solution}

\begin{solution}{exo:g12:limcont:2}
\emph{1.} $2x^2 - 3x + 1 = (x-1)(2x-1)$ (la \omterm{def:g11:quad:discriminant}{racine} $x=1$ est apparente
d'après $2 - 3 + 1 = 0$). Pour $x \neq 1$, simplifier $x - 1$~:
$f(x) = 2x - 1$.

\emph{2.} D'où $\lim_{x\to1} f(x) = 1$. Poser $f(1) = 1$ prolonge $f$ en
la \omterm{def:g11:func:function}{fonction} polynomiale $x \mapsto 2x-1$ sur $\R$, \omterm{def:g12:limcont:continuity}{continue} en $1$.
\end{solution}

\begin{solution}{exo:g12:limcont:3}
Écrire $f(x) = \dfrac{3x - 1}{x + 2} = \dfrac{3(x+2) - 7}{x+2}
= 3 - \dfrac{7}{x+2}$. Lorsque $x \to \pm\infty$, $f(x) \to 3$~: la
droite $y = 3$ est une \omterm{def:g12:limcont:asymptote}{asymptote horizontale} aux deux infinis. Lorsque
$x \to (-2)^\pm$, $f(x) \to \mp\infty$~: la droite $x = -2$ est une
\omterm{def:g12:limcont:asymptote}{asymptote verticale}. La courbe est une \omterm{def:g11:func:reference}{hyperbole}, sous $y=3$ pour
$x > -2$ et au-dessus pour $x < -2$.
\end{solution}

\begin{solution}{exo:g12:limcont:4}
Multiplier par le conjugué~:
\[
\sqrt{x^2 + x} - x = \frac{x^2 + x - x^2}{\sqrt{x^2+x} + x}
= \frac{x}{x\left(\sqrt{1 + 1/x} + 1\right)}
= \frac{1}{\sqrt{1 + 1/x} + 1}
\xrightarrow[x\to+\infty]{} \frac{1}{2}.
\]
De même,
\[
\frac{\sqrt{1+x} - 1}{x} = \frac{(1+x) - 1}{x\left(\sqrt{1+x} + 1\right)}
= \frac{1}{\sqrt{1+x} + 1} \xrightarrow[x\to0]{} \frac{1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:limcont:5}
\emph{1.} Comme $\sin x \geq -1$, $f(x) \geq x - 1 \to +\infty$, et on
conclut par comparaison. Pourtant $f'(x) = 1 + \cos x$ s'annule en tout
multiple impair de $\pi$, et sur tout \omterm{def:g10:numbers:interval}{intervalle} $\intco{A}{+\infty}$ la
\omterm{def:g11:func:function}{fonction} alterne entre des \omterm{def:g10:numbers:interval}{intervalles} de croissance stricte~; elle est
\omterm{def:g12:seq:monotonic}{croissante} mais non strictement, et surtout sa croissance n'est pas
nécessaire~: la comparaison suffit.

\emph{2.} Pour $x > 1$, diviser par $x$~:
\[
g(x) = \frac{1 + \frac{\sin x}{x}}{1 - \frac{\sin x}{x}},
\]
et $\frac{\sin x}{x} \to 0$ par le théorème des gendarmes
($\abs{\sin x} \leq 1$). D'où $g(x) \to \frac{1+0}{1-0} = 1$.
\end{solution}

\begin{solution}{exo:g12:limcont:6}
$f(x) = x^3 - 3x + 1$ est \omterm{def:g12:limcont:continuity}{continue}, avec $f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$~:
$f$ croît sur $\intoc{-\infty}{-1}$, décroît sur $\intcc{-1}{1}$, croît
sur $\intco{1}{+\infty}$, avec \omterm{def:g10:functions:extrema}{maximum} local $f(-1) = 3 > 0$ et \omterm{def:g10:functions:extrema}{minimum}
local $f(1) = -1 < 0$, $\lim_{-\infty} f = -\infty$,
$\lim_{+\infty} f = +\infty$.

Sur chacun des trois \omterm{def:g10:numbers:interval}{intervalles} de monotonie, $f$ est \omterm{def:g12:limcont:continuity}{continue},
strictement \omterm{def:g11:func:monotone}{monotone}, et $0$ est strictement entre les valeurs/limites
aux extrémités, donc le théorème de la bijection donne exactement un zéro
par \omterm{def:g10:numbers:interval}{intervalle}~: trois zéros au total. Contrôles de signe~:
$f(-2) = -1 < 0 < f(-1) = 3$, donc un zéro dans $\intoo{-2}{-1}$~;
$f(0) = 1 > 0 > f(1) = -1$, un zéro dans $\intoo{0}{1}$~;
$f(1) = -1 < 0 < f(2) = 3$, un zéro dans $\intoo{1}{2}$.
\end{solution}

\begin{solution}{exo:g12:limcont:7}
Soit $g(x) = f(x) - x$, \omterm{def:g12:limcont:continuity}{continue} sur $\intcc{0}{1}$. Comme
$f(0) \in \intcc{0}{1}$, $g(0) = f(0) \geq 0$~; comme
$f(1) \in \intcc{0}{1}$, $g(1) = f(1) - 1 \leq 0$. Par le théorème des
valeurs intermédiaires, $g$ s'annule en un certain $c \in \intcc{0}{1}$,
c'est-à-dire $f(c) = c$.
\end{solution}

\begin{solution}{exo:g12:limcont:8}
Soient $u(t)$ et $d(t)$ les positions (mesurées comme distance depuis le
bas du sentier) à l'instant $t \in \intcc{8}{20}$ à la montée et à la
descente. Toutes deux sont \omterm{def:g12:limcont:continuity}{continues}, $u(8) = 0$, $u(20) = L$ (la
longueur du sentier), $d(8) = L$, $d(20) = 0$. La \omterm{def:g11:func:function}{fonction} $g = u - d$
est \omterm{def:g12:limcont:continuity}{continue} avec $g(8) = -L \leq 0$ et $g(20) = L \geq 0$, donc
$g(c) = 0$ pour un certain instant $c$ par le théorème des valeurs
intermédiaires~: à l'instant $c$, les positions des deux jours
coïncident.
\end{solution}

\begin{solution}{exo:g12:limcont:9}
\emph{1.} $f'(x) = 5x^4 + 3x^2 + 1 > 0$, donc $f$ est strictement
\omterm{def:g12:seq:monotonic}{croissante} sur $\R$~; avec $\lim_{\pm\infty} f = \pm\infty$ et la
\omterm{def:g12:limcont:continuity}{continuité}, le théorème de la bijection donne un unique zéro réel
$\alpha$. Comme $f(0) = -1 < 0$ et $f(1) = 2 > 0$,
$\alpha \in \intoo{0}{1}$.

\emph{2.} Maintenir un \omterm{def:g10:numbers:interval}{intervalle} $\intcc{a_n}{b_n}$ avec
$f(a_n) < 0 < f(b_n)$, en partant de $\intcc{0}{1}$~; à chaque étape
évaluer $f$ au \omterm{prop:g10:coordgeom:midpoint}{milieu} $m$ et garder le \omterm{def:g10:numbers:interval}{demi-intervalle} où le changement
de signe persiste. Après $n$ étapes l'\omterm{def:g10:numbers:interval}{intervalle} a pour longueur
$2^{-n}$, donc le \omterm{prop:g10:coordgeom:midpoint}{milieu} approxime $\alpha$ à $2^{-(n+1)}$ près. Il faut
$2^{-(n+1)} \leq 10^{-6}$, c'est-à-dire
$n + 1 \geq \frac{6}{\log_{10} 2} \approx 19.93$~: $n = 20$ itérations
suffisent.
\end{solution}

\begin{solution}{pb:g12:limcont:1}
\textbf{1.} Termes dominants~: $\frac{3x^2}{x^2} \to 3$~;
$\frac{2x}{x^2} = \frac2x \to 0$~;
$x^3 - x^2 = x^3\left(1 - \frac1x\right) \to +\infty$.

\textbf{2.} $\lim_{x \to \pm\infty} f = 3$~: \omterm{def:g12:limcont:asymptote}{asymptote horizontale}
$y = 3$~; $\lim_{x \to (-2)^\pm} f = \mp\infty$~: \omterm{def:g12:limcont:asymptote}{asymptote
verticale} $x = -2$.

\textbf{3.} $\frac{(x-3)(x+3)}{x - 3} = x + 3 \to 6$. Et
$\frac{\sqrt{x+1} - 1}{x} = \frac{x}{x(\sqrt{x+1} + 1)} =
\frac{1}{\sqrt{x+1} + 1} \to \frac12$.

\textbf{4.} $\abs{x \sin\frac1x} \leq \abs x \to 0$, donc la limite
vaut $0$. Et $\abs{\frac{\sin x}{x}} \leq \frac1x \to 0$~: limite
$0$.

\textbf{5.} $\frac{\sqrt{4x^2 + x}}{x} =
\sqrt{4 + \frac1x} \to \sqrt4 = 2$ (\omterm{def:g12:limcont:continuity}{continuité} de la racine carrée,
composition).

\textbf{6.} $f(x) = x^3 + x$ est \omterm{def:g12:limcont:continuity}{continue} et vérifie
$f(1) = 2 < 5 < 10 = f(2)$~: par le théorème des valeurs
intermédiaires, $f(c) = 5$ pour un certain $c \in \intoo{1}{2}$.

\textbf{7.} $f'(x) = 3x^2 + 1 > 0$~: $f$ est \omterm{def:g12:limcont:continuity}{continue} et
strictement \omterm{def:g12:seq:monotonic}{croissante} sur $\R$, de limites $\mp\infty$~: par le
théorème de la bijection, toute valeur --- en particulier $5$ ---
est atteinte exactement une fois.

\textbf{8.} $p(x) = x^3\left(1 + \frac ax + \frac{b}{x^2} +
\frac{c}{x^3}\right)$~: quand $x \to +\infty$, $p \to +\infty$~;
quand $x \to -\infty$, $p \to -\infty$. Donc sur un \omterm{def:g10:numbers:interval}{intervalle}
$\intcc{-M}{M}$ assez grand, $p(-M) < 0 < p(M)$~: le théorème des
valeurs intermédiaires livre une \omterm{def:g11:quad:discriminant}{racine}. Le même argument de terme
dominant vaut mot pour mot pour tout degré impair --- les puissances
\omterm{def:g11:func:parity}{impaires} gardent des signes opposés aux deux infinis.

\textbf{9.} Après $n$ partages en deux, il reste un \omterm{def:g10:numbers:interval}{intervalle} de
longueur $2^{-n}$~; or $2^{-20} \approx 9.5 \times 10^{-7} <
10^{-6}$~: vingt pas. Newton double les décimales exactes à chaque
pas~: en partant d'une décimale exacte, environ $3$ ou $4$ pas
suffisent pour en atteindre six --- la tortue et le lévrier de la
recherche de \omterm{def:g11:quad:discriminant}{racines}.

\textbf{10.} $g$ n'est pas \omterm{def:g12:limcont:continuity}{continue} sur $\intcc{-1}{1}$~: elle
explose en $0$, que l'\omterm{def:g10:numbers:interval}{intervalle} contient. La conclusion du théorème
s'évapore avec son hypothèse --- toujours vérifier la \omterm{def:g12:limcont:continuity}{continuité}
\emph{sur l'\omterm{def:g10:numbers:interval}{intervalle} tout entier}, et pas seulement aux
extrémités.

\textbf{11.} $g(x) = f(x) - x$ est \omterm{def:g12:limcont:continuity}{continue}~;
$g(0) = f(0) \geq 0$ (les valeurs sont dans $\intcc{0}{1}$) et
$g(1) = f(1) - 1 \leq 0$. Par le théorème des valeurs
intermédiaires, $g(c) = 0$ pour un certain $c$, c'est-à-dire
$f(c) = c$.

\textbf{12.} En idéalisant le pays par le segment $\intcc{0}{1}$~:
$f$ envoie chaque lieu sur le lieu figuré juste au-dessous de lui
sur la carte froissée~; froisser garde la carte au-dessus du pays,
donc $f$ arrive dans $\intcc{0}{1}$, et $f$ est \omterm{def:g12:limcont:continuity}{continue} (le papier
ne se déchire pas). Le \omterm{pb:g10:functions:1}{point fixe} $c$ est l'endroit situé
précisément au-dessus de sa propre \omterm{def:g10:functions:function}{image} --- le théorème de Brouwer
traite honnêtement la carte en dimension deux.

\textbf{13.} $g(0) + g(\pi) = \left(T(0) - T(\pi)\right) +
\left(T(\pi) - T(2\pi)\right) = T(0) - T(2\pi) = 0$. Donc $g(0)$ et
$g(\pi)$ sont opposés (ou tous deux nuls)~: dans les deux cas $g$,
\omterm{def:g12:limcont:continuity}{continue}, s'annule en un $t^* \in \intcc{0}{\pi}$, c'est-à-dire
$T(t^*) = T(t^* + \pi)$ --- quelque part sur l'équateur, deux points
antipodaux s'accordent sur la température, et cela à chaque
instant.

\textbf{14.} Faire tourner la table d'un quart de tour échange les
rôles des deux diagonales~: la paire de pieds qui appuyait sur le
sol et celle dont un pied planait échangent leurs situations, si
bien que la hauteur du pied en l'air --- comptée avec son signe
comme «~écart de la diagonale distinguée~» --- prend des signes
opposés en $\theta$ et en $\theta + \frac\pi2$. Un sol continu rend
$h$ \omterm{def:g12:limcont:continuity}{continue}, et le théorème des valeurs intermédiaires fournit un
angle $\theta^*$ tel que $h(\theta^*) = 0$~: les quatre pieds à
terre. (Aucune chance de ce genre avec une table rectangulaire ou un
sol à marches.)

\textbf{15.} Patron~: construire la \emph{différence} des deux
grandeurs que l'on veut faire coïncider ($f(x) - x$~;
$T(t) - T(t + \pi)$~; position à la montée moins position à la
descente pour le randonneur~; l'écart signé pour la table)~;
vérifier qu'elle prend des valeurs des deux signes (souvent grâce à
une symétrie des extrémités)~; invoquer le théorème des valeurs
intermédiaires sur une \omterm{def:g11:func:function}{fonction} \omterm{def:g12:limcont:continuity}{continue}. La coïncidence est
forcée.

\textbf{16.} Questions laissées ouvertes~: \emph{combien} de
solutions (réglé par la stricte monotonie et le théorème de la
bijection, question 7) et \emph{où} elles se trouvent (réglé par la
\omterm{thm:g12:limcont:ivt}{dichotomie} ou par Newton, question 9). Le théorème des valeurs
intermédiaires est un pur oracle d'existence.

\textbf{17.} $3 - \frac{7}{x + 2}$~: \omterm{def:g12:limcont:asymptote}{asymptote verticale}
$x = -2$, \omterm{def:g12:limcont:asymptote}{asymptote horizontale} $y = 3$~; de part et d'autre de
$-2$ la \omterm{def:g11:func:function}{fonction} est \omterm{def:g12:seq:monotonic}{croissante} ($-\frac{7}{x+2}$ croît là où elle
est définie)~; la courbe est l'\omterm{def:g11:func:reference}{hyperbole} de référence $-\frac7x$
translatée de $2$ vers la gauche et de $3$ vers le haut --- deux
branches serrant le croisement des \omterm{def:g12:limcont:asymptote}{asymptotes}.

\textbf{18.} Pour $x \neq 1$~: $f(x) = x + 1$, donc
$\lim_{x \to 1} f = 2$~: poser $f(1) = 2$ rétablit la \omterm{def:g12:limcont:continuity}{continuité}.
La discontinuité était une piqûre, non un saut --- on comble le
point manquant et la courbe se referme~: elle est
\emph{éliminable}.

\textbf{19.} \omterm{def:g12:limcont:continuity}{Continue} sur tout \omterm{def:g10:numbers:interval}{intervalle} compris entre deux entiers
consécutifs~; en chaque entier $n$ elle saute de $n - 1$ à $n$.
Avec $a = 0.9$ et $b = 1.1$~: $\lfloor a \rfloor = 0 < \frac12 <
1 = \lfloor b \rfloor$, et pourtant
$\lfloor x \rfloor = \frac12$ n'a aucune solution --- le saut
téléporte par-dessus $\frac12$, ce que le théorème des valeurs
intermédiaires interdit précisément aux \omterm{def:g11:func:function}{fonctions} \omterm{def:g12:limcont:continuity}{continues}.

\textbf{20.} Visage local~: valeur $=$ limite en chaque point ---
la piqûre éliminable (question 18) le violait de façon guérissable,
les sauts de la partie entière (question 19) de façon incurable.
Visage global~: sur un \omterm{def:g10:numbers:interval}{intervalle}, toutes les valeurs
intermédiaires sont atteintes --- les certificats de \omterm{def:g11:quad:discriminant}{racines}
(questions 6 à 8), le \omterm{pb:g10:functions:1}{point fixe}, l'équateur, la table et le
randonneur vivent tous ici. Le nom officiel du slogan~: le
\emph{théorème des valeurs intermédiaires}.
\end{solution}
