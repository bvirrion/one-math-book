\section*{Hoofdstuk \ref{ch:g12:arith} --- Getaltheorie}

\begin{solution}{exo:g12:arith:1}
$17 \times 119 = 2023$, dus $2026 = 17 \times 119 + 3$: quotiënt $119$, rest
$3$. Voor $-2026$: $-2026 = 17\times(-120) + 14$ (inderdaad is
$17 \times 120 = 2040$ en $2040 - 2026 = 14$): quotiënt $-120$, rest $14$ (de
rest moet in $\intco{0}{17}$ liggen, dus ze is \emph{niet} $-3$).
\end{solution}

\begin{solution}{exo:g12:arith:2}
Modulo $10$: $7^2 = 49 \equiv 9 \equiv -1$. Dus
$7^{100} = \left(7^2\right)^{50} \equiv (-1)^{50} = 1 \pmod{10}$: het laatste
cijfer van $7^{100}$ is $1$.
\end{solution}

\begin{solution}{exo:g12:arith:3}
Euclides: $1071 = 2\times462 + 147$; $462 = 3\times147 + 21$;
$147 = 7\times21 + 0$. Dus $\gcd = 21$.

Achterwaartse substitutie: $21 = 462 - 3\times147 = 462 - 3(1071 - 2\times462)
= 7\times462 - 3\times1071$. Dus $u = -3$ en $v = 7$:
$1071\times(-3) + 462\times7 = 21$.
\end{solution}

\begin{solution}{exo:g12:arith:4}
Elk \omterm{def:g10:numbers:sets}{geheel getal} is $\equiv 0, 1, 2$ of $3 \pmod 4$, en na kwadrateren:
$0^2 \equiv 0$, $1^2 \equiv 1$, $2^2 = 4 \equiv 0$, $3^2 = 9 \equiv 1$. Dus is
$n^2 \equiv 0$ of $1 \pmod 4$. Een som van twee kwadraten is dan congruent met
$0 + 0$, $0 + 1$ of $1 + 1$, dat wil zeggen met $0$, $1$ of $2 \pmod 4$ ---
nooit met $3$.
\end{solution}

\begin{solution}{exo:g12:arith:5}
\omterm{def:g12:arith:divides}{Deelbaarheid} door $2$: van $n$ en $n + 1$ is er één even. \omterm{def:g12:arith:divides}{Deelbaarheid} door
$3$: is $n \equiv 0$, dan geldt $3 \mid n$; is $n \equiv 1 \pmod 3$, dan is
$2n + 1 \equiv 3 \equiv 0$; is $n \equiv 2$, dan is $n + 1 \equiv 0$. In alle
gevallen \omterm{def:g12:arith:divides}{deelt} $3$ het product. Omdat $\gcd(2,3) = 1$, geeft
\cref{cor:g12:arith:coprimeprod} dat $6 \mid n(n+1)(2n+1)$.
(Daarmee is opnieuw bewezen dat $\frac{n(n+1)(2n+1)}{6}$, de som van de
kwadraten uit \cref{exo:g12:seq:1}, een \omterm{def:g10:numbers:sets}{geheel getal} is.)
\end{solution}

\begin{solution}{exo:g12:arith:6}
We zoeken de inverse van $5$ modulo $11$: proberen (of Bézout) geeft
$5 \times 9 = 45 = 44 + 1 \equiv 1 \pmod{11}$. Vermenigvuldig de \omterm{def:g12:arith:congruence}{congruentie}
met $9$:
\[
x \equiv 9 \times 3 = 27 \equiv 5 \pmod{11}.
\]
De oplossingen zijn de \omterm{def:g10:numbers:sets}{gehele getallen} $x = 5 + 11k$, $k \in \Z$. (Controle:
$5\times5 = 25 \equiv 3 \pmod{11}$.)
\end{solution}

\begin{solution}{exo:g12:arith:7}
$\gcd(17, 40) = 1$, dus er zijn oplossingen. Euclides:
$40 = 2\times17 + 6$; $17 = 2\times6 + 5$; $6 = 5 + 1$. Achterwaarts
substitueren: $1 = 6 - 5 = 6 - (17 - 2\times6) = 3\times6 - 17
= 3(40 - 2\times17) - 17 = 3\times40 - 7\times17$. Dus
$17\times(-7) - 40\times(-3) = 1$: de bijzondere oplossing
$(x_0, y_0) = (-7, -3)$.

Algemene oplossing van $17x - 40y = 1$: trek de bijzondere betrekking af, dan
is $17(x + 7) = 40(y + 3)$; omdat $\gcd(17, 40) = 1$, geeft Gauss
$40 \mid x + 7$, dus $x = -7 + 40k$ en vervolgens $y = -3 + 17k$, $k \in \Z$
(en die voldoen allemaal).

Voor $17x - 40y = 6$ vermenigvuldig je de bijzondere oplossing met $6$:
$(x_1, y_1) = (-42, -18)$, en dezelfde redenering geeft
\[
x = -42 + 40k, \qquad y = -18 + 17k, \qquad k \in \Z .
\]
(Bijvoorbeeld $k = 2$: $x = 38$, $y = 16$; inderdaad is
$17\times38 - 40\times16 = 646 - 640 = 6$.)
\end{solution}

\begin{solution}{exo:g12:arith:8}
Onderstel $\sqrt2 = \frac ab$ met $a, b \in \N^*$; dan is $a^2 = 2b^2$. In de
priemontbinding van een kwadraat is elke exponent even; de exponent van $2$ in
$a^2$ is dus even, terwijl hij in $2b^2$ oneven is (één meer dan een even
getal). Twee ontbindingen van hetzelfde gehele getal met verschillende
exponenten van $2$ zijn in tegenspraak met de uniciteit uit
\cref{thm:g12:arith:fta}. Zo'n breuk bestaat dus niet: $\sqrt2 \notin \Q$.
\end{solution}

\begin{solution}{exo:g12:arith:9}
\emph{1.} Uit $k\binom pk = p \binom{p-1}{k-1}$ volgt dat $p$ het getal
$k\binom pk$ \omterm{def:g12:arith:divides}{deelt}. Voor $1 \leq k \leq p-1$ geven $p \nmid k$ en $p$ \omterm{def:g12:arith:prime}{priem}
dat $\gcd(p, k) = 1$, dus levert het lemma van Gauss $p \mid \binom pk$.

\emph{2.} Inductie op $a$. Voor $a = 0$: $0^p \equiv 0$. Onderstel
$a^p \equiv a \pmod p$. Volgens het binomium van Newton is
\[
(a+1)^p = \sum_{k=0}^{p} \binom pk a^k
\equiv a^p + 1 \pmod p,
\]
want alle tussenliggende termen verdwijnen modulo $p$ volgens punt 1. Volgens
de inductiehypothese is dus $(a+1)^p \equiv a + 1 \pmod p$. Daarmee is
$a^p \equiv a$ bewezen voor alle $a \in \N$, en het geval $a < 0$ volgt door
$a \equiv a + kp$ met een geschikte positieve vertegenwoordiger te schrijven.
\end{solution}

\begin{solution}{exo:g12:arith:10}
$n \equiv 2 \pmod 3$ en $n \equiv 3 \pmod 5$: schrijf $n = 2 + 3s$; dan is
$2 + 3s \equiv 3 \pmod 5$, dat wil zeggen $3s \equiv 1 \pmod 5$. De inverse
van $3$ modulo $5$ is $2$ ($3\times2 = 6 \equiv 1$), dus $s \equiv 2 \pmod 5$,
zeg $s = 2 + 5t$, en $n = 8 + 15t$: de twee eerste voorwaarden betekenen
$n \equiv 8 \pmod{15}$.

Voeg $n \equiv 2 \pmod 7$ toe: $8 + 15t \equiv 2 \pmod 7$, en
$15 \equiv 1 \pmod 7$, dus $t \equiv -6 \equiv 1 \pmod 7$, zeg
$t = 1 + 7u$. Dus $n = 23 + 105u$:
\[
n \equiv 23 \pmod{105}.
\]
(Controle: $23 = 3\times7 + 2 = 5\times4 + 3 = 7\times3 + 2$.)
\end{solution}

\begin{solution}{pb:g12:arith:1}
\textbf{1.} $2026 = 289 \times 7 + 3$: $2026 \equiv 3 \pmod 7$. Machten van
$7$ mod $10$: $7, 9, 3, 1$, een cyclus van lengte $4$; en
$100 \equiv 0 \pmod 4$: het laatste cijfer van $7^{100}$ is $1$.

\textbf{2.} $5^2 = 25 \equiv -1 \pmod{13}$, dus
$5^{116} = \left(5^2\right)^{58} \equiv (-1)^{58} = 1$ en
$5^{117} \equiv 5 \pmod{13}$.

\textbf{3.} De inverse van $3$ modulo $7$ is $5$ ($15 \equiv 1$):
$x \equiv 5 \times 5 = 25 \equiv 4 \pmod 7$.

\textbf{4.} $97 = 2 \times 35 + 27$; $35 = 27 + 8$; $27 = 3 \times 8 + 3$;
$8 = 2 \times 3 + 2$; $3 = 2 + 1$. Achterwaarts substitueren:
$1 = 97 \times 13 + 35 \times (-36)$. Dus $35 \times (-36) \equiv 1
\pmod{97}$: de inverse van $35$ is $-36 \equiv 61 \pmod{97}$.

\textbf{5.} $a$ is inverteerbaar modulo $n$ precies wanneer
$\gcd(a, n) = 1$: Bézout levert $au + nv = 1$, dat wil zeggen
$au \equiv 1$; omgekeerd dwingt een inverse de ggd om $1$ te delen.

\textbf{6.} $0{\cdot}10 + 3{\cdot}9 + 0{\cdot}8 + 6{\cdot}7 +
4{\cdot}6 + 0{\cdot}5 + 6{\cdot}4 + 1{\cdot}3 + 5{\cdot}2 +
2{\cdot}1 = 132 = 12 \times 11 \equiv 0 \pmod{11}$: geldig.

\textbf{7.} De som verandert met $wd$ met $1 \leq w \leq 10$ en
$1 \leq \abs d \leq 9$: omdat $11$ \omterm{def:g12:arith:prime}{priem} is en geen van beide factoren \omterm{def:g12:arith:divides}{deelt},
kan het het product niet delen
(\cref{thm:g12:arith:gauss} / \cref{prop:g12:arith:primedivides}). De
gewijzigde som is dus nooit opnieuw $\equiv 0$: elke fout in één cijfer laat
het alarm afgaan.

\textbf{8.} Het verwisselen van naburige cijfers $a$ en $b$ (gewichten
$w + 1$ en $w$) verandert de som met
$(w+1)b + wa - (w+1)a - wb = b - a \not\equiv 0$ zodra $a \neq b$: opgespoord.
Het geheim is dat $11$ \emph{\omterm{def:g12:arith:prime}{priem}} is: modulo $10$ verdwijnen producten zoals
$5 \times 2$ zonder dat een van beide factoren nul is, zodat een fout van
$\pm 2$ op een gewicht $5$ (of een ongelukkige verwisseling) erdoor kan
glippen.

\textbf{9.} De gewogen som van de twaalf cijfers is $119$; het controlecijfer
moet ze aanvullen tot een veelvoud van $10$: dat is $1$ (volledige code
$978\,2940199\,051$). EAN mist de verwisselingen van naburige cijfers met
$2(a - b) \equiv 0 \pmod{10}$, dat wil zeggen $\abs{a - b} = 5$: een $2$ en
een $7$ verwisselen glipt er ongezien door --- de prijs van de vriendelijke
\omterm{def:g12:complex:modulus}{modulus} $10$.

\textbf{10.} Verdubbel je van rechts af elk tweede cijfer en vouw je terug
($16 \to 7$, enzovoort), dan komt de som op $80 \equiv 0 \pmod{10}$: de
testkaart is geldig.

\textbf{11.} Met een priemmodulus is elk gewicht inverteerbaar, zodat
\emph{alle} fouten in één cijfer en \emph{alle} verwisselingen van naburige
cijfers gevat worden --- de luxe van het ISBN; schema's modulo $10$ houden
mensvriendelijke cijfers en nemen een kleine blinde vlek erbij.

\textbf{12.} $2^{10} = 1024 = 3 \times 341 + 1 \equiv 1 \pmod{341}$, dus
$2^{340} = \left(2^{10}\right)^{34} \equiv 1$. En toch is
$341 = 11 \times 31$ samengesteld: het doorstaat de test van Fermat met
grondtal $2$ zonder \omterm{def:g12:arith:prime}{priem} te zijn. Moraal: de \omterm{def:g12:arith:congruence}{congruentie} van Fermat is
noodzakelijk, niet voldoende --- priemtesten vragen scherper gereedschap (en
krijgt dat, in de universitaire volumes).

\textbf{13.} $3d \equiv 1 \pmod{20}$: $d = 7$ ($21 = 20 + 1$).

\textbf{14.} $c = 4^3 = 64 \equiv 31 \pmod{33}$.

\textbf{15.} $31 \equiv -2$: $(-2)^7 = -128$, en $-128 + 4 \times 33 = 4$: de
versleutelde tekst ontsleutelt tot $m = 4$. Het slot draait.

\textbf{16.} Modulo $3$: geldt $3 \nmid m$, dan is $m^2 \equiv 1$ (Fermat),
dus $m^{21} = m \cdot \left(m^2\right)^{10} \equiv m$; geldt $3 \mid m$, dan
zijn beide leden $\equiv 0$. Modulo $11$: $m^{10} \equiv 1$ of $11 \mid m$, en
$m^{21} = m \cdot \left(m^{10}\right)^2 \equiv m$. Zowel $3$ als $11$ \omterm{def:g12:arith:divides}{deelt}
dus $m^{21} - m$, en omdat ze \omterm{def:g12:arith:gcd}{relatief priem} zijn, \omterm{def:g12:arith:divides}{deelt} hun product $33$ het
ook (Gauss): $m^{21} \equiv m \pmod{33}$. De exponent $ed = 21 = 1 + 20k$ was
zo gebouwd dat beide exponenten van Fermat ($2$ en $10$, delers van $20$)
verdwijnen.

\textbf{17.} Twee priemgetallen van $300$ cijfers vermenigvuldigen kost een
microseconde; ze uit hun product terugvinden verslaat elk bekend algoritme en
alle computers van de wereld --- het slot is een eenrichtingsstraat. (Onze
$n = 33$ is diezelfde straat op speelgoedschaal, in beide richtingen te
belopen.)

\textbf{18.} Door de resten te toetsen (of met Bézout op te bouwen):
$n \equiv 8 \pmod{15}$, dus de aantallen $8, 23, 38, 53, \dots$ Uniciteit
modulo $15$: twee oplossingen verschillen een veelvoud van $3$ en van $5$, dus
van $15$ ($3$ en $5$ zijn \omterm{def:g12:arith:gcd}{relatief priem}, Gauss). De generaal met $1000$
soldaten kondigt met drie snelle opstellingen ``$8$'' aan --- de aloude truc
om koppen te tellen.

\textbf{19.} $10 \equiv 1 \pmod 9$ geeft $10^k \equiv 1$, dus
$\sum d_k 10^k \equiv \sum d_k$: een getal en zijn cijfersom zijn congruent
modulo $9$ (en modulo $3$). En $10 \equiv -1 \pmod{11}$ geeft
$\sum d_k 10^k \equiv \sum (-1)^k d_k$: de alternerende regel. Twee regels uit
de kindertijd, elk in één lijn.

\textbf{20.} \omterm{def:g12:arith:congruence}{Congruenties} maakten van resten een rekenkunde (Deel I); Bézout
sloeg de inversen die lineaire \omterm{def:g12:arith:congruence}{congruenties} en de $d$ van RSA oplossen (vragen
4 en 13); de kleine stelling van Fermat opende en sloot het slot (vragen 15 en
16); het lijmen van relatief prieme moduli telde de soldaten en maakte het
bewijs af (vragen 16 en 18). Het oordeel over Hardy: de zuiverste stelling die
hij kende bewaakt nu elke aankoop --- zuiverheid is, mits tijd, het meest
toepasbare wat er is.
\end{solution}
