\section*{Hoofdstuk \ref{ch:g12:complex} --- Complexe getallen}

\begin{solution}{exo:g12:complex:1}
$(3-2\iu)(1+4\iu) = 3 + 12\iu - 2\iu - 8\iu^2 = 11 + 10\iu$.

$\dfrac{2+\iu}{1-\iu} = \dfrac{(2+\iu)(1+\iu)}{(1-\iu)(1+\iu)}
= \dfrac{1 + 3\iu}{2} = \dfrac12 + \dfrac32\iu$.

$\iu^{2027}$: omdat $\iu^4 = 1$ en $2027 = 4 \times 506 + 3$, is
$\iu^{2027} = \iu^3 = -\iu$.
\end{solution}

\begin{solution}{exo:g12:complex:2}
\emph{(a)} $\Delta = 16 - 52 = -36$:
$z = \dfrac{4 \pm 6\iu}{2} = 2 \pm 3\iu$. Een toegevoegd paar; het product is
$4 + 9 = 13 = \frac ca$. \emph{(b)} $\Delta = 1 - 4 = -3$:
$z = \dfrac{-1 \pm \iu\sqrt3}{2}$ (de derde \omterm{def:g12:complex:rootsofunity}{eenheidswortels} $j$ en
$\conj j$); het product is $\frac{1 + 3}{4} = 1$.
\end{solution}

\begin{solution}{exo:g12:complex:3}
$\abs{z_1} = \sqrt2$, \omterm{def:g12:complex:argument}{argument} $\frac{3\pi}{4}$ (tweede kwadrant):
$z_1 = \sqrt2\,\eu^{3\iu\pi/4}$.
$\abs{z_2} = 2$, \omterm{def:g12:complex:argument}{argument} $-\frac\pi6$: $z_2 = 2\,\eu^{-\iu\pi/6}$. Dus
\[
\frac{z_1}{z_2} = \frac{\sqrt2}{2}\,\eu^{\iu\left(\frac{3\pi}{4} + \frac{\pi}{6}\right)}
= \frac{\sqrt2}{2}\,\eu^{11\iu\pi/12}.
\]
Algebraïsch:
\[
\frac{z_1}{z_2} = \frac{-1+\iu}{\sqrt3 - \iu}
= \frac{(-1+\iu)(\sqrt3+\iu)}{4}
= \frac{-\sqrt3 - 1 + \iu(\sqrt3 - 1)}{4}.
\]
Door de reële delen gelijk te stellen:
$\frac{\sqrt2}{2}\cos\frac{11\pi}{12} = \frac{-\sqrt3-1}{4}$, dus
\[
\cos\frac{11\pi}{12} = -\frac{\sqrt6 + \sqrt2}{4}.
\]
\end{solution}

\begin{solution}{exo:g12:complex:4}
\emph{(a)} $\abs{z - 2} = \abs{z - (-\iu)}$ betekent dat $M$ even ver ligt van
$A(2, 0)$ als van $B(0, -1)$: de \emph{middelloodlijn} van $[AB]$.

\emph{(b)} Afstand $3$ tot het punt $(1,1)$: de \emph{cirkel} met middelpunt
$1 + \iu$ en straal $3$.

\emph{(c)} Volgens \cref{prop:g12:complex:geometry}(4) betekent
$\frac{z-1}{z+1}$ zuiver imaginair dat de \omterm{def:g11:vect:vector}{vectoren} van $A(1,0)$ naar $M$ en
van $B(-1,0)$ naar $M$ \omterm{def:g11:scal:orthogonal}{orthogonaal} zijn: $M$ ziet het lijnstuk $[AB]$ onder
een rechte hoek. De verzameling is de \emph{cirkel met middellijn $[AB]$} (de
eenheidscirkel) \emph{zonder de punten $A$ en $B$ zelf}.
\end{solution}

\begin{solution}{exo:g12:complex:5}
De Moivre: $\cos3\theta + \iu\sin3\theta = (\cos\theta + \iu\sin\theta)^3$.
\omterm{def:g10:algebra:expand}{Uitwerken} met het binomium van Newton, met $c = \cos\theta$ en
$s = \sin\theta$:
\[
(c + \iu s)^3 = c^3 + 3\iu c^2 s - 3 c s^2 - \iu s^3
= (c^3 - 3cs^2) + \iu(3c^2 s - s^3).
\]
Door de delen gelijk te stellen en $s^2 = 1 - c^2$, $c^2 = 1 - s^2$ te
gebruiken:
\[
\cos3\theta = 4\cos^3\theta - 3\cos\theta,
\qquad
\sin3\theta = 3\sin\theta - 4\sin^3\theta .
\]
\end{solution}

\begin{solution}{exo:g12:complex:6}
\[
\cos^3\theta = \left(\frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}\right)^{\!3}
= \frac{\eu^{3\iu\theta} + 3\eu^{\iu\theta} + 3\eu^{-\iu\theta} + \eu^{-3\iu\theta}}{8}
= \frac{\cos3\theta + 3\cos\theta}{4}.
\]
Dus
\[
\int_0^{\pi/2}\cos^3\theta\,\dd\theta
= \frac14\left[\frac{\sin3\theta}{3} + 3\sin\theta\right]_0^{\pi/2}
= \frac14\left(-\frac13 + 3\right) = \frac{2}{3}.
\]
(In overeenstemming met $W_3 = \frac23$ in \cref{exo:g12:integ:9}, want
$\int_0^{\pi/2}\cos^3 = \int_0^{\pi/2}\sin^3$ wegens de symmetrie
$\theta \mapsto \frac\pi2 - \theta$.)
\end{solution}

\begin{solution}{exo:g12:complex:7}
$c = a + \eu^{\pm\iu\pi/3}(b - a)$ met $b - a = 2 + \iu$ en
$\eu^{\pm\iu\pi/3} = \frac12 \pm \iu\frac{\sqrt3}{2}$:
\[
\eu^{\iu\pi/3}(2+\iu) = \left(\tfrac12 + \iu\tfrac{\sqrt3}{2}\right)(2+\iu)
= \left(1 - \tfrac{\sqrt3}{2}\right) + \iu\left(\tfrac12 + \sqrt3\right),
\]
dus $c_1 = \left(2 - \frac{\sqrt3}{2}\right) +
\iu\left(\frac32 + \sqrt3\right)$, en net zo met de hoek $-\frac\pi3$:
$c_2 = \left(2 + \frac{\sqrt3}{2}\right) + \iu\left(\frac32 - \sqrt3\right)$.
\end{solution}

\begin{solution}{exo:g12:complex:8}
$-4 = 4\,\eu^{\iu\pi}$, dus de oplossingen van $z^4 = -4$ zijn
\[
z_k = \sqrt2\,\eu^{\iu\left(\frac\pi4 + \frac{k\pi}{2}\right)},
\quad k = 0, 1, 2, 3,
\]
dat wil zeggen $1 + \iu$, $-1 + \iu$, $-1 - \iu$, $1 - \iu$: de hoekpunten van
een vierkant met omgeschreven straal $\sqrt 2$. Door toegevoegde \omterm{def:g11:quad:discriminant}{wortels} te
koppelen:
\[
z^4 + 4 = \bigl(z - (1{+}\iu)\bigr)\bigl(z - (1{-}\iu)\bigr)
\bigl(z + (1{-}\iu)\bigr)\bigl(z + (1{+}\iu)\bigr)
= (z^2 - 2z + 2)(z^2 + 2z + 2).
\]
\end{solution}

\begin{solution}{exo:g12:complex:9}
Met $q = \eu^{\iu\theta} \neq 1$:
\[
\sum_{k=0}^{n} \eu^{\iu k\theta} = \frac{\eu^{\iu(n+1)\theta} - 1}{\eu^{\iu\theta} - 1}
= \frac{\eu^{\iu(n+1)\theta/2}}{\eu^{\iu\theta/2}}
\cdot\frac{\eu^{\iu(n+1)\theta/2} - \eu^{-\iu(n+1)\theta/2}}{\eu^{\iu\theta/2} - \eu^{-\iu\theta/2}}
= \eu^{\iu n\theta/2}\,
\frac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}},
\]
waarbij $\eu^{\iu\alpha} - \eu^{-\iu\alpha} = 2\iu\sin\alpha$ gebruikt wordt
(het \emph{afsplitsen van de halve hoek}). De reële delen nemen geeft
\[
S = \frac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}}\,
\cos\frac{n\theta}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:complex:10}
\emph{1.} $\omega$ is een vijfde eenheidswortel verschillend van $1$, dus de
som van alle vijf de \omterm{def:g11:quad:discriminant}{wortels} is nul
(\cref{thm:g12:complex:rootsofunity}), en dat luidt
$1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$.

\emph{2.} $u + v = \omega + \omega^2 + \omega^3 + \omega^4 = -1$ volgens
punt 1. Voor het product, met $\omega^5 = 1$:
\[
uv = (\omega + \omega^4)(\omega^2 + \omega^3)
= \omega^3 + \omega^4 + \omega^6 + \omega^7
= \omega^3 + \omega^4 + \omega + \omega^2 = -1 .
\]
Dus $u$ en $v$ zijn de \omterm{def:g11:quad:discriminant}{wortels} van $X^2 + X - 1 = 0$:
$\left\{\frac{-1+\sqrt5}{2}, \frac{-1-\sqrt5}{2}\right\}$. Nu is
$u = \omega + \conj\omega = 2\cos\frac{2\pi}{5} > 0$ (want
$\frac{2\pi}{5} < \frac{\pi}{2}$), dus $u = \frac{-1 + \sqrt5}{2}$.

\emph{3.} $\cos\frac{2\pi}{5} = \frac u2 = \frac{\sqrt5 - 1}{4}$.
\end{solution}

\begin{solution}{pb:g12:complex:1}
\textbf{1.} $(2 + \iu)(3 - \iu) = 6 - 2\iu + 3\iu + 1 = 7 + \iu$. \quad
$\frac{1 + \iu}{1 - \iu} = \frac{(1 + \iu)^2}{2} = \frac{2\iu}{2} = \iu$.
\quad $\abs{3 + 4\iu} = 5$.

\textbf{2.} $1 + \iu = \sqrt2\,\eu^{\iu\pi/4}$; $-2 = 2\eu^{\iu\pi}$;
$1 - \iu\sqrt3 = 2\eu^{-\iu\pi/3}$.

\textbf{3.} $(1 + \iu)^8 = \left(\sqrt2\right)^8 \eu^{8\iu\pi/4} =
16\,\eu^{2\iu\pi} = 16$.

\textbf{4.} $z^2 = \eu^{\iu\pi/2}$:
$z = \pm\eu^{\iu\pi/4} = \pm\frac{\sqrt2}{2}(1 + \iu)$. En
$\Delta = 4 - 20 = -16$: $z = 1 \pm 2\iu$.

\textbf{5.} $z \mapsto \iu z$: een draaiing over een kwartslag om de
oorsprong. $\abs{z - (1 + \iu)} = 2$: de cirkel met middelpunt $1 + \iu$ en
straal $2$.

\textbf{6.} $\left(\omega^k\right)^5 = \eu^{2\iu\pi k} = 1$, en de vijf punten
$\eu^{2\iu k\pi/5}$ zijn verschillend: dat zijn dus alle vijf de oplossingen
van $z^5 = 1$. Ze liggen op de eenheidscirkel op gelijke hoeken van
$72^\circ$: een regelmatige vijfhoek met een hoekpunt in $1$.

\textbf{7.} $S = 1 + \omega + \dots + \omega^4$ voldoet aan
$(1 - \omega)S = 1 - \omega^5 = 0$, en $\omega \neq 1$: dus $S = 0$. (Het
zwaartepunt van de vijfhoek is haar middelpunt.)

\textbf{8.} $\omega^4 = \overline\omega$ en $\omega^3 = \overline{\omega^2}$:
door de toegevoegden te koppelen luidt het \omterm{def:g12:complex:def}{reële deel} van de som
$1 + 2\cos\frac{2\pi}{5} + 2\cos\frac{4\pi}{5} = 0$.

\textbf{9.} $\cos\frac{4\pi}{5} = 2c^2 - 1$, dus $1 + 2c + 4c^2 - 2 = 0$, dat
wil zeggen $4c^2 + 2c - 1 = 0$: $c = \frac{-1 + \sqrt5}{4}$ (de positieve
\omterm{def:g11:quad:discriminant}{wortel}, want $72^\circ$ is scherp): $\cos 72^\circ = \frac{\sqrt5 - 1}{4}$.

\textbf{10.} Numeriek is $c \approx 0.30902$: dat klopt met het rekentoestel.
En $\frac{1}{2\varphi} = \frac{1}{1 + \sqrt5} = \frac{\sqrt5 - 1}{4}$ (maak de
noemer wortelvrij): $c = \frac{1}{2\varphi}$. De vijfhoek is door en door
gulden: haar diagonaal verhoudt zich tot haar zijde als $\varphi$ --- de
eindeloze zelfgelijkvormigheid van het pentagram.

\textbf{11.} Vierkantswortels zijn met de passer construeerbaar (de
halvecirkelmachine van het \omterm{def:g12:seq:arith-geom}{meetkundig} \omterm{def:g11:stat:mean}{gemiddelde} uit het onderbouwvolume),
dus een getal dat uit \omterm{def:g10:numbers:sets}{rationale getallen} met alleen vierkantswortels gebouwd
wordt --- zoals $\cos 72^\circ$ --- is construeerbaar; derdemachtswortels zijn
dat niet, en $\cos 20^\circ$ heeft er één nodig: de vijfhoek geeft zich
gewonnen, de driedeling houdt stand. Het criterium van Gauss beslist over elke
regelmatige veelhoek --- zijn zeventienhoek, op zijn negentiende, deed hem
voor de wiskunde kiezen.

\textbf{12.} $-4 = 4\eu^{\iu\pi}$: de vierdemachtswortels hebben \omterm{def:g12:complex:modulus}{modulus}
$\sqrt2$ en \omterm{def:g12:complex:argument}{argumenten} $\frac{\pi}{4} + k\frac{\pi}{2}$: de vier getallen
$\pm 1 \pm \iu$. Toegevoegden koppelen:
$(z - 1 - \iu)(z - 1 + \iu) = z^2 - 2z + 2$ en
$(z + 1 - \iu)(z + 1 + \iu) = z^2 + 2z + 2$: de ontbinding van
\cref{exo:g12:complex:8}.

\textbf{13.} $(\cos\theta + \iu\sin\theta)^3 = \cos^3\theta +
3\iu\cos^2\theta\sin\theta - 3\cos\theta\sin^2\theta - \iu\sin^3\theta$;
volgens De Moivre is dit $\cos 3\theta + \iu \sin 3\theta$. Reële delen:
$\cos 3\theta = \cos^3\theta - 3\cos\theta(1 - \cos^2\theta)
= 4\cos^3\theta - 3\cos\theta$ --- en de imaginaire delen leveren
$\sin 3\theta$ er gratis bij.

\textbf{14.} Het is het \omterm{def:g12:complex:def}{reële deel} van $1 + \omega + \dots + \omega^4 = 0$: de
som is $0$ --- precies vraag 8, gezien door de kern van
\cref{exo:g12:complex:9}.

\textbf{15.} $8\iu = 8\eu^{\iu\pi/2}$: de \omterm{def:g11:quad:discriminant}{wortels} zijn
$2\eu^{\iu\pi/6} = \sqrt3 + \iu$; $2\eu^{5\iu\pi/6} = -\sqrt3 + \iu$;
$2\eu^{3\iu\pi/2} = -2\iu$.

\textbf{16.} $1 + \iu = \sqrt2\,\eu^{\iu\pi/4}$: de afbeelding draait het vlak
over $45^\circ$ en vergroot het met factor $\sqrt2$. Het eenheidsvierkant
wordt een schuin vierkant met zijde $\sqrt2$ --- verdubbeld in oppervlakte,
een achtste slag gedraaid.

\textbf{17.} $\abs{z_1 z_2}^2 = z_1 z_2 \overline{z_1 z_2} =
z_1\overline{z_1}\, z_2\overline{z_2} = \abs{z_1}^2\abs{z_2}^2$: neem de
vierkantswortels. Machten van $1 + \iu$: $1 + \iu$, $2\iu$, $-2 + 2\iu$, $-4$:
telkens $45^\circ$ verder gedraaid en $\sqrt2$ keer langer --- een
naar buiten lopende spiraal met moduli $\sqrt2, 2, 2\sqrt2, 4, \dots$

\textbf{18.} $\mathrm{j}$ is een derde eenheidswortel verschillend van $1$,
dus (het \omterm{def:g12:complex:argument}{argument} van vraag 7 met $n = 3$) is
$1 + \mathrm{j} + \mathrm{j}^2 = 0$. Criterium op de standaarddriehoek:
$a + \mathrm{j}b + \mathrm{j}^2 c = 1 + \mathrm{j}\cdot\mathrm{j} +
\mathrm{j}^2 \cdot \mathrm{j}^2 = 1 + \mathrm{j}^2 + \mathrm{j}^4 =
1 + \mathrm{j}^2 + \mathrm{j} = 0$: voldaan, zoals de gelijkzijdige driehoek
$\left(1, \mathrm{j}, \mathrm{j}^2\right)$ vraagt.

\textbf{19.} Over $\C$ valt een reële veelterm uiteen in factoren van graad
$1$; haar niet-reële \omterm{def:g11:quad:discriminant}{wortels} komen in toegevoegde paren voor (voeg de
\omterm{def:g10:algebra:equation}{vergelijking} toe), en elk paar vermenigvuldigt tot een reële
tweedegraadsfactor $z^2 - 2\operatorname{Re}(z_0)\,z + \abs{z_0}^2$: vandaar
de reële ontbinding in stukken van graad $1$ en $2$ --- vraag 12 voerde ze uit
op $z^4 + 4$.

\textbf{20.} Punten; draaien en vergroten; regelmatige veelhoeken;
goniometrische identiteiten door uitwerking; en een gesloten wereld waarin
elke \omterm{def:g10:algebra:equation}{vergelijking} van graad $n$ haar $n$ \omterm{def:g11:quad:discriminant}{wortels} heeft. Coda: in vraag 10 gaf
de vijfhoek van $\pi$ de hand aan de gulden snede --- de twee beroemdste
gasten van Euclides, eindelijk aan elkaar voorgesteld door een som van vijf
pijlen die nul is.
\end{solution}
