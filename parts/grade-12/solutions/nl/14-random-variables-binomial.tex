\section*{Hoofdstuk \ref{ch:g12:randvar} --- Toevalsvariabelen en de binomiale verdeling}

\begin{solution}{exo:g12:randvar:1}
$\E(X) = -0.3 + 0 + 0.8 + 0.5 = 1$.
$\E(X^2) = 0.3 \times 1 + 0 + 0.4\times4 + 0.1\times25 = 4.4$, dus geeft
König--Huygens $\V(X) = 4.4 - 1 = 3.4$ en
$\sigma(X) = \sqrt{3.4} \approx 1.84$.
\end{solution}

\begin{solution}{exo:g12:randvar:2}
\emph{1.} De $10$ vragen zijn onafhankelijke \omterm{def:g12:randvar:bernoulli}{bernoulli-experimenten} met
$p = \frac14$: $X \sim \mathcal B\left(10, \frac14\right)$,
$\E(X) = 2.5$ en
$\sigma(X) = \sqrt{10 \times \frac14 \times \frac34} = \sqrt{1.875}
\approx 1.37$.

\emph{2.} $\P(X = 0) = \left(\frac34\right)^{10} \approx 0.056$;
\[
\P(X = 5) = \binom{10}{5}\left(\frac14\right)^5\left(\frac34\right)^5
\approx 0.058;
\quad
\P(X \geq 1) = 1 - \left(\tfrac34\right)^{10} \approx 0.944 .
\]
\end{solution}

\begin{solution}{exo:g12:randvar:3}
Onafhankelijke identieke proeven: $X \sim \mathcal B(400,\ 0.05)$, dus
$\E(X) = 20$ claims en
\[
\sigma(X) = \sqrt{400 \times 0.05 \times 0.95} = \sqrt{19} \approx 4.4 .
\]
\end{solution}

\begin{solution}{exo:g12:randvar:4}
De ontvangen uitbetaling $P$ voldoet aan $\E(P) = \frac{5 + 6}{6} =
\frac{11}{6}$, dus is de eerlijke prijs $c = \frac{11}{6} \approx 1.83$ euro.
De eerlijke winst $G = P - \frac{11}6$ neemt de waarden
$-\frac{11}{6}, \frac{19}{6}, \frac{25}{6}$ aan met de kansen
$\frac46, \frac16, \frac16$:
\[
\V(G) = \E(G^2)
= \frac{4 \times 121 + 361 + 625}{6 \times 36} = \frac{1470}{216}
= \frac{245}{36} \approx 6.8,
\qquad \sigma(G) \approx 2.6 .
\]
Een eerlijk spel heeft een \emph{\omterm{def:g11:stat:mean}{gemiddelde}} winst nul, maar zijn uitkomsten
schommelen nog altijd: eerlijk is niet hetzelfde als zonder risico.
\end{solution}

\begin{solution}{exo:g12:randvar:5}
$X \sim \mathcal B(8,\ 0.7)$.
\[
\P(X = 6) = \binom86 (0.7)^6(0.3)^2 \approx 0.296 ;
\]
\[
\P(X \geq 6) = \P(6) + \P(7) + \P(8)
\approx 0.296 + 8(0.7)^7(0.3) + (0.7)^8
\approx 0.296 + 0.198 + 0.058 = 0.552 ;
\]
$\P(X \geq 1) = 1 - (0.3)^8 \approx 0.99993$. De modus:
$(n+1)p = 6.3$, dus is de waarschijnlijkste waarde $k^* = 6$
(\cref{exo:g12:randvar:7}).
\end{solution}

\begin{solution}{exo:g12:randvar:6}
Het aantal passagiers dat opdaagt is $X \sim \mathcal B(104,\ 0.9)$; de
vlucht is overboekt zodra $X \geq 101$:
\[
\P(X \geq 101) = \sum_{k=101}^{104}\binom{104}{k}(0.9)^k(0.1)^{104-k}
\approx 0.006 .
\]
$4\%$ meer tickets verkopen dan er plaatsen zijn, veroorzaakt maar op ongeveer
$0.6\%$ van de vluchten een incident --- de economie achter het overboeken.
\end{solution}

\begin{solution}{exo:g12:randvar:7}
\[
\frac{\P(X = k+1)}{\P(X = k)}
= \frac{\binom{n}{k+1}}{\binom nk}\cdot\frac{p}{1-p}
= \frac{n - k}{k + 1}\cdot\frac{p}{1-p},
\]
met $\binom{n}{k+1} = \binom nk \frac{n-k}{k+1}$. Die verhouding is $\geq 1$
als en slechts als $(n-k)p \geq (k+1)(1-p)$, dat wil zeggen
$np - k p \geq k - kp + 1 - p$, dus $k \leq (n+1)p - 1$. De kansen stijgen dus
strikt zolang $k + 1 \leq (n+1)p$ en dalen daarna: het \omterm{def:g10:functions:extrema}{maximum} wordt bereikt
in $k^* = \floor{(n+1)p}$ (gedeeld met $k^* - 1$ wanneer $(n+1)p$ een \omterm{def:g10:numbers:sets}{geheel
getal} is).
\end{solution}

\begin{solution}{exo:g12:randvar:8}
\emph{1.} Het eerste kruis bij worp $k$ betekent $k-1$ keer munt en dan kruis:
\omterm{def:g10:proba:distribution}{kans} $\left(\frac12\right)^{k-1}\cdot\frac12 = 2^{-k}$, voor
$1 \leq k \leq 10$. Totale \omterm{def:g10:proba:distribution}{kans} om te winnen:
$\sum_{k=1}^{10} 2^{-k} = 1 - 2^{-10} = \frac{1023}{1024}$ (het spel gaat
alleen verloren bij tien keer munt na elkaar).

\emph{2.} Verwachte uitbetaling:
\[
\sum_{k=1}^{10} 2^k \cdot 2^{-k} = \sum_{k=1}^{10} 1 = 10 \text{ euro}.
\]
Zonder de grens divergeert de som $\sum_{k\geq1} 1$: de verwachte uitbetaling
is oneindig, hoewel het spel bijna altijd een klein bedrag uitkeert --- de
beroemde \emph{paradox van Sint-Petersburg}, die toont dat de
\omterm{def:g12:randvar:exp}{verwachtingswaarde} alleen de waarde van een spel niet meet.
\end{solution}

\begin{solution}{pb:g12:randvar:1}
\textbf{1.} $X \sim \mathcal B\!\left(3, \frac16\right)$:
$\P(X = 0) = \frac{125}{216}$,
$\P(X = 1) = \frac{75}{216}$,
$\P(X = 2) = \frac{15}{216}$,
$\P(X = 3) = \frac{1}{216}$.

\textbf{2.} $\E(X) = 3 \times \frac16 = \frac12$;
$V(X) = 3 \times \frac16 \times \frac56 = \frac{5}{12}$.

\textbf{3.} Verwachte uitbetaling $2\,\E(X) = 1$ euro tegenover een inzet van
$1$ euro: winst $0$ --- precies eerlijk (een zeldzaamheid).

\textbf{4.} Zonder teruglegging zijn de trekkingen afhankelijk (de kansen van
de tweede kaart hangen van de eerste af): niet binomiaal --- het vakje
``\omterm{def:g12:condprob:indep}{onafhankelijkheid}'' van de checklist blijft leeg. Met teruglegging: vast
$n = 5$, constante $p = \frac14$, onafhankelijke trekkingen: binomiaal.

\textbf{5.} $\E(X) = 6$ en $\sigma(X) = \sqrt{4.2} \approx 2.05$.
$\E(S) = 5 \times 6 - 10 = 20$; $\sigma(S) = 5\sigma(X) \approx 10.25$.

\textbf{6.} Een vast aantal $n = 105$ proeven (de tickets), elk een
komen/niet-komen met dezelfde $p = 0.9$, verondersteld \omterm{def:g12:condprob:indep}{onafhankelijk}:
binomiaal. De bekentenis: gezinnen missen een vlucht \emph{samen} en stormen
maken hele vliegtuigen leeg --- de \omterm{def:g12:condprob:indep}{onafhankelijkheid} is de geloofssprong van
het model, en gecorreleerde afwezigheden maken de staarten dikker dan de
\omterm{def:g12:randvar:binomial}{binomiale verdeling} belooft.

\textbf{7.} $\E(X) = 94.5$; $\sigma(X) = \sqrt{105 \times 0.9 \times 0.1} =
\sqrt{9.45} \approx 3.07$.

\textbf{8.} $\P(X \geq 101) = \sum_{k=101}^{105}
\binom{105}{k} 0.9^k\, 0.1^{105 - k}$.

\textbf{9.} $\approx 0.0167$: ongeveer één vlucht op zestig weigert iemand.

\textbf{10.} $\E(\text{geweigerd}) \approx 0.023$ passagiers per vlucht:
verwachte vergoeding $\approx 19$ euro --- tegenover $1\,000$ euro extra
inkomsten. Met vijf overboeken is overweldigend winstgevend; vandaar dat elke
maatschappij het doet.

\textbf{11.} Uit de tabel van de staartkansen: $n = 105$: $1.7\,\%$;
$n = 106$: $4.0\,\%$; $n = 107$: $8.1\,\%$. De grootste toegelaten verkoop is
$n = 106$ tickets.

\textbf{12.} $z = \frac{100.5 - 94.5}{3.07} \approx 1.95$: de drempel ligt
twee \omterm{def:g11:stat:variance}{standaardafwijkingen} boven het \omterm{def:g11:stat:mean}{gemiddelde}, en de vuistregel van de
klokkromme (``bovenstaart voorbij $2\sigma$ $\approx 2.5\,\%$'') komt dicht bij
het exacte $1.7\,\%$ --- de gladde kromme die de \omterm{def:g12:randvar:binomial}{binomiale verdeling}
schaduwt, is de hoofdrolspeler van de komende hoofdstukken.

\textbf{13.} $0.98^{20} + 20 \times 0.02 \times 0.98^{19} \approx 0.94$: de
eerlijke partij raakt in $94\,\%$ van de gevallen door de keuring.

\textbf{14.} $0.9^{20} + 20 \times 0.1 \times 0.9^{19} \approx 0.39$: de
slechte partij glipt er nog altijd in $39\,\%$ van de gevallen door.

\textbf{15.} Het risico van de producent: een goede partij wordt geweigerd
($\approx 6\,\%$); het risico van de klant: een slechte partij wordt aanvaard
($\approx 39\,\%$). Een grotere \omterm{def:g10:proba:sample}{steekproef} (met een evenredige
aanvaardingsdrempel) verkleint beide --- ten koste van meer controle:
kwaliteit heeft een begrotingslijn.

\textbf{16.} $V\!\left(\frac Xn\right) = \frac{V(X)}{n^2} =
\frac{np(1-p)}{n^2} = \frac{p(1-p)}{n}$: \omterm{def:g11:stat:variance}{standaardafwijking}
$\sqrt{\frac{p(1-p)}{n}}$. Verviervoudig de \omterm{def:g10:proba:sample}{steekproef} en je halveert de ruis:
de wet in $\frac{1}{\sqrt n}$ van de schommelingsintervallen uit het eerste
jaar van dit volume, nu afgeleid.

\textbf{17.} De grens beperkt de uitbetaling tot $2^{10} = 1024$ euro, zodat
elk van de tien ronden precies $1$ euro aan \omterm{def:g12:randvar:exp}{verwachtingswaarde} bijdraagt:
$\E = 10$. De oneindige \omterm{def:g12:randvar:exp}{verwachtingswaarde} van het onbegrensde spel kwam
volledig van astronomisch zeldzame, astronomisch grote uitbetalingen; de grens
bekent dat geen enkele bank $2^{50}$ euro uitkeert. Een eerlijke ticketprijs
voor het begrensde spel: $10$ euro --- en plots is niemand nog in de war.

\textbf{18.} Verwacht prijzengeld per ticket:
$\frac{100 + 2 \times 50}{200} = 1$ euro tegenover een ticket van $2$ euro:
marge $50\,\%$ --- achttien keer de $2.7\,\%$ van de roulette. De tombola komt
ermee weg omdat haar spelers welbewust schenken: het ``verlies'' is de
bedoeling.

\textbf{19.} Verwachte schade: $1\,000 \times 0.01 \times 10\,000 =
100\,000$; premies $130\,000$: verwachte winst $30\,000$ euro.
\omterm{def:g11:stat:variance}{Standaardafwijking} van de totale schade:
$10\,000 \times \sqrt{1\,000 \times 0.01 \times 0.99} \approx 31\,500$ euro
--- één gewoon slecht jaar verslindt de hele verwachte winst. Vandaar
kapitaalreserves, herverzekering en portefeuilles die veel groter zijn dan
duizend polissen: verzekeraars leven van de wet van de grote aantallen en
houden reserves aan tegen haar traagheid.

\textbf{20.} Eerst de checklist --- vast $n$, dezelfde $p$, \omterm{def:g12:condprob:indep}{onafhankelijkheid}
\emph{beweerd} --- of er is helemaal geen \omterm{def:g12:randvar:binomial}{binomiale verdeling}. Navigeer daarna
op $\E \pm \sigma$: \omterm{def:g11:stat:mean}{gemiddelden} situeren, afwijkingen waarschuwen. Prijs
vervolgens de staarten: geweigerde passagiers, slechte partijen en rampjaren
wonen allemaal voorbij $2\sigma$. Voorbehouden: de \omterm{def:g12:condprob:indep}{onafhankelijkheid} is de
belofte van wie het model maakt, niet die van de wereld; en de
\omterm{def:g12:randvar:exp}{verwachtingswaarde} negeert precies wat je ondergang wordt. De ontbrekende
bladzijden van het reglement --- hoe snel \omterm{def:g10:stats:series}{frequenties} tot rust komen, en welke
vorm de schommelingen aannemen --- zijn de volgende twee hoofdstukken.
\end{solution}
