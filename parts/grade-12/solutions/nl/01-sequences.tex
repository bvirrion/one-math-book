\section*{Hoofdstuk \ref{ch:g12:seq} --- Rijen}

\begin{solution}{exo:g12:seq:1}
Zij $P(n)$ de uitspraak $\sum_{k=1}^{n} k^2 = \frac{n(n+1)(2n+1)}{6}$.
\emph{Basisgeval:} voor $n = 0$ zijn beide leden $0$ (lege som).
\emph{Inductiestap:} neem $P(n)$ aan. Dan is
\[
\sum_{k=1}^{n+1} k^2 = \frac{n(n+1)(2n+1)}{6} + (n+1)^2
= \frac{(n+1)\bigl(n(2n+1) + 6(n+1)\bigr)}{6}
= \frac{(n+1)(2n^2 + 7n + 6)}{6}.
\]
Omdat $2n^2 + 7n + 6 = (n+2)(2n+3)$, is dat
$\frac{(n+1)(n+2)(2(n+1)+1)}{6}$, en dat is $P(n+1)$. Met inductie geldt
$P(n)$ dus voor alle $n$.
\end{solution}

\begin{solution}{exo:g12:seq:2}
$a_{n+1} - a_n = \frac{n+2}{n+1} - \frac{n+1}{n}
= \frac{n(n+2) - (n+1)^2}{n(n+1)} = \frac{-1}{n(n+1)} < 0$: $(a_n)$ is
strikt \omterm{def:g10:functions:variations}{dalend}.

$(b_n)$ heeft positieve termen en
$\frac{b_{n+1}}{b_n} = \frac{2^{n+1}}{n+1}\cdot\frac{n}{2^n} =
\frac{2n}{n+1} \geq 1 \iff 2n \geq n+1 \iff n \geq 1$: $(b_n)$ is \omterm{def:g12:seq:monotonic}{stijgend}
(strikt vanaf $n \geq 2$).

$c_{n+1} - c_n = (n+1)^2 - 10(n+1) - n^2 + 10n = 2n - 9$, negatief voor
$n \leq 4$ en positief voor $n \geq 5$: $(c_n)$ daalt tot $c_5 = -25$, haar
\omterm{def:g10:functions:extrema}{minimum}, en stijgt daarna. Ze is niet monotoon.
\end{solution}

\begin{solution}{exo:g12:seq:3}
Haal de overheersende termen buiten haakjes:
\[
u_n = \frac{n^2(3 - 1/n + 1/n^2)}{n^2(2 + 5/n^2)} \xrightarrow[n\to+\infty]{} \frac{3}{2}.
\]
Vermenigvuldig met de toegevoegde uitdrukking:
\[
v_n = \frac{(n+1) - n}{\sqrt{n+1} + \sqrt{n}} = \frac{1}{\sqrt{n+1}+\sqrt{n}}
\xrightarrow[n\to+\infty]{} 0.
\]
Deel teller en noemer door $3^n$:
\[
w_n = \frac{(2/3)^n - 1}{1 + (1/3)^n} \xrightarrow[n\to+\infty]{} \frac{0-1}{1+0} = -1,
\]
met $\lim q^n = 0$ voor $\abs{q} < 1$.
\end{solution}

\begin{solution}{exo:g12:seq:4}
$u_n = 5 + 3n \to +\infty$ en $v_n = 8 \cdot (1/2)^n = 2^{3-n} \to 0$. De
sommen zijn
\[
\sum_{k=0}^{n} u_k = (n+1)\,\frac{5 + (5+3n)}{2} = \frac{(n+1)(10+3n)}{2}
\xrightarrow[n\to+\infty]{} +\infty,
\]
\[
\sum_{k=0}^{n} v_k = 8\,\frac{1 - (1/2)^{n+1}}{1 - 1/2}
= 16\left(1 - \left(\tfrac{1}{2}\right)^{n+1}\right)
\xrightarrow[n\to+\infty]{} 16 .
\]
\end{solution}

\begin{solution}{exo:g12:seq:5}
Omdat $-1 \leq \cos n \leq 1$, geldt
\[
\frac{n-1}{n+1} \leq \frac{n + \cos n}{n+1} \leq 1,
\]
en $\frac{n-1}{n+1} \to 1$, zodat de limiet met de insluitstelling $1$ is.

Voor de tweede limiet schrijf je
\[
0 \leq \frac{n!}{n^n}
= \frac{1}{n}\cdot\frac{2}{n}\cdots\frac{n}{n}
\leq \frac{1}{n},
\]
want elke factor $\frac{k}{n}$ met $2 \leq k \leq n$ is hoogstens $1$. Omdat
$\frac1n \to 0$, geeft de insluitstelling $\frac{n!}{n^n} \to 0$. (De
voorgestelde meetkundige begrenzing werkt ook: elke factor met $k \leq n/2$
is hoogstens $\frac12$, wat de sterkere grens $(1/2)^{\floor{n/2}}$ geeft.)
\end{solution}

\begin{solution}{exo:g12:seq:6}
\emph{1.} $u_0 = 0 \in \intcc{0}{2}$. Is $0 \leq u_n \leq 2$, dan is
$2 \leq u_n + 2 \leq 4$, dus $\sqrt{2} \leq u_{n+1} \leq 2$; in het bijzonder
is $0 \leq u_{n+1} \leq 2$. Met inductie geldt de eigenschap voor alle $n$.

\emph{2.} $u_{n+1} - u_n = \sqrt{u_n + 2} - u_n$. Voor $x \in \intcc{0}{2}$
geldt $\sqrt{x+2} \geq x \iff x + 2 \geq x^2 \iff (2-x)(x+1) \geq 0$, en dat
is waar. Bijgevolg is $(u_n)$ \omterm{def:g12:seq:monotonic}{stijgend}.

\emph{3.} \omterm{def:g12:seq:monotonic}{Stijgend} en \omterm{def:g12:seq:bounded}{naar boven begrensd} door $2$ \omterm{def:g12:seq:limit}{convergeert} $(u_n)$ naar
een zekere $\ell \in \intcc{0}{2}$. Overgaan naar de limiet in
$u_{n+1} = \sqrt{u_n + 2}$ (de afbeelding $x \mapsto \sqrt{x+2}$ is continu)
geeft $\ell = \sqrt{\ell + 2}$, dus $\ell^2 - \ell - 2 = 0$, dus
$\ell \in \{-1, 2\}$. Omdat $\ell \geq 0$, is $\lim u_n = 2$.
\end{solution}

\begin{solution}{exo:g12:seq:7}
\emph{1.} Tussen twee dosissen wordt $40\%$ van het middel verwijderd, dus
wordt de hoeveelheid $u_n$ herleid tot $0.6\,u_n$; de volgende dosis voegt
$1$ eenheid toe: $u_{n+1} = 0.6\,u_n + 1$.

\emph{2.} $v_{n+1} = u_{n+1} - 2.5 = 0.6\,u_n + 1 - 2.5 = 0.6(u_n - 2.5)
= 0.6\,v_n$: $(v_n)$ is \omterm{def:g12:seq:arith-geom}{meetkundig} met \omterm{def:g11:seq:geometric}{reden} $0.6$ en eerste term
$v_0 = 1 - 2.5 = -1.5$. Bijgevolg is $v_n = -1.5 \times 0.6^n$ en
\[
u_n = 2.5 - 1.5 \times 0.6^n .
\]

\emph{3.} Omdat $0.6^n \to 0$, is $u_n \to 2.5$: de hoeveelheid middel
stabiliseert op $2.5$ eenheden.
\end{solution}

\begin{solution}{exo:g12:seq:8}
\emph{1.} $u_0 = 3 > 1$. Is $u_n > 1$, dan is $u_n + 2 > 0$ en
\[
u_{n+1} - 1 = \frac{4u_n - 1 - u_n - 2}{u_n + 2} = \frac{3(u_n - 1)}{u_n + 2} > 0 .
\]
Met inductie is $u_n > 1$ voor alle $n$ (en in het bijzonder is
$u_n + 2 \neq 0$, zodat de \omterm{def:g12:seq:sequence}{rij} goed gedefinieerd is).

\emph{2.} Met de identiteit hierboven:
\[
v_{n+1} = \frac{1}{u_{n+1} - 1} = \frac{u_n + 2}{3(u_n - 1)}
= \frac{(u_n - 1) + 3}{3(u_n - 1)} = \frac{1}{3} + v_n .
\]
Dus is $(v_n)$ \omterm{def:g12:seq:arith-geom}{rekenkundig} met \omterm{def:g11:seq:arithmetic}{verschil} $\frac13$ en
$v_0 = \frac{1}{u_0 - 1} = \frac12$.

\emph{3.} $v_n = \frac12 + \frac{n}{3}$, dus
$u_n = 1 + \frac{1}{v_n} = 1 + \frac{6}{3 + 2n}$. Omdat $v_n \to +\infty$,
is $u_n \to 1$.
\end{solution}

\begin{solution}{exo:g12:seq:9}
\emph{1.} $H_{2n} - H_n = \sum_{k=n+1}^{2n} \frac{1}{k}$ is een som van $n$
termen die elk minstens $\frac{1}{2n}$ zijn; bijgevolg is
$H_{2n} - H_n \geq n \cdot \frac{1}{2n} = \frac12$.

\emph{2.} Met inductie op $k$: $H_{2^0} = H_1 = 1 \geq 1$. Is
$H_{2^k} \geq 1 + \frac{k}{2}$, dan geeft punt 1 met $n = 2^k$ dat
\[
H_{2^{k+1}} \geq H_{2^k} + \frac12 \geq 1 + \frac{k+1}{2}.
\]
De \omterm{def:g12:seq:sequence}{rij} $(H_n)$ is \omterm{def:g12:seq:monotonic}{stijgend} (elke stap voegt $\frac{1}{n+1} > 0$ toe) en de
deelrij $H_{2^k}$ is onbegrensd, dus is $(H_n)$ niet \omterm{def:g12:seq:bounded}{naar boven begrensd}.
\omterm{def:g12:seq:monotonic}{Stijgend} en onbegrensd streeft ze naar $+\infty$
(\cref{thm:g12:seq:monotone}).
\end{solution}

\begin{solution}{exo:g12:seq:10}
\emph{1.} De \omterm{def:g12:seq:sequence}{rij} $d_n = b_n - a_n$ voldoet aan
$d_{n+1} - d_n = (b_{n+1} - b_n) - (a_{n+1} - a_n) \leq 0$, dus is $(d_n)$
\omterm{def:g10:functions:variations}{dalend}; omdat $d_n \to 0$, volgt $d_n \geq 0$ voor alle $n$ (een dalende \omterm{def:g12:seq:sequence}{rij}
met een negatieve term zou er voor altijd onder blijven, wat de limiet $0$
verhindert). Bijgevolg is $a_n \leq b_n$.

\emph{2.} Uit $a_n \leq b_n \leq b_0$ volgt dat de stijgende \omterm{def:g12:seq:sequence}{rij} $(a_n)$
\omterm{def:g12:seq:bounded}{naar boven begrensd} is, zodat ze naar een zekere $\ell$ \omterm{def:g12:seq:limit}{convergeert}. Net zo
\omterm{def:g12:seq:limit}{convergeert} $(b_n)$, \omterm{def:g10:functions:variations}{dalend} en naar beneden begrensd door $a_0$, naar een
zekere $\ell'$. Dan is $\ell' - \ell = \lim (b_n - a_n) = 0$, dus
$\ell = \ell'$.

\emph{3.} $(a_n)$ is (strikt) \omterm{def:g12:seq:monotonic}{stijgend}, want
$a_{n+1} - a_n = \frac{1}{(n+1)!} > 0$. Voor $(b_n)$:
\[
b_{n+1} - b_n = \frac{1}{(n+1)!} + \frac{1}{(n+1)(n+1)!} - \frac{1}{n\,n!}
= \frac{n(n+1) + n - (n+1)^2}{n(n+1)(n+1)!}
= \frac{-1}{n(n+1)(n+1)!} < 0 ,
\]
dus is $(b_n)$ \omterm{def:g10:functions:variations}{dalend}. Ten slotte is $b_n - a_n = \frac{1}{n\,n!} \to 0$. De
twee \omterm{def:g12:seq:sequence}{rijen} zijn aangrenzend en convergeren dus naar een gemeenschappelijke
limiet.
\end{solution}

\begin{solution}{pb:g12:seq:1}
\textbf{1.} Waar voor $n = 1$ ($1 = 1^2$). Is
$1 + 3 + \dots + (2n - 1) = n^2$, dan geeft het volgende oneven getal
erbij optellen: $n^2 + (2n + 1) = (n + 1)^2$: de erfelijkheid. Met inductie
is de uitspraak waar voor alle $n \geq 1$.

\textbf{2.} $2^0 = 1 > 0$. Is $2^n > n$, dan is
$2^{n+1} = 2 \cdot 2^n > 2n \geq n + 1$ voor $n \geq 1$ (en $n = 0$ klopt
rechtstreeks): erfelijkheid, klaar.

\textbf{3.} $n = 0$: $1 \geq 1$. Is $(1 + x)^n \geq 1 + nx$, vermenigvuldig
dan met $1 + x \geq 1 > 0$:
$(1 + x)^{n+1} \geq (1 + nx)(1 + x) = 1 + (n + 1)x + nx^2 \geq 1 + (n + 1)x$.

\textbf{4.} De stap van $n = 1$ naar $n = 2$: bij twee knikkers zijn ``de
eerste $n$'' en ``de laatste $n$'' twee \emph{disjuncte} losse knikkers ---
geen enkele gemeenschappelijke knikker overbrugt de twee groepen, dus dwingt
niets hun kleuren om overeen te stemmen. Het erfelijkheidsargument vraagt
stilzwijgend dat de twee groepen overlappen, en dat geldt pas vanaf
$n \geq 2$; met het basisgeval $n = 1$ komt de ketting nooit op gang.

\textbf{5.} $4^0 - 1 = 0 = 3 \times 0$. Is $4^n - 1 = 3k$, dan is
$4^{n+1} - 1 = 4(4^n - 1) + 3 = 3(4k + 1)$: erfelijkheid.

\textbf{6.} $x_1 = \frac32$, $x_2 = \frac{17}{12}$,
$x_3 = \frac{577}{408}$.

\textbf{7.} $x_{n+1}^2 - 2 = \frac{(x_n^2 + 2)^2 - 8x_n^2}{4x_n^2} =
\frac{(x_n^2 - 2)^2}{4 x_n^2}$: een kwadraat gedeeld door iets positiefs,
dus $\geq 0$, en $> 0$ zodra $x_n^2 \neq 2$. Inductie: $x_0 = 2 > 0$ met
$x_0^2 = 4 > 2$; en is $x_n > 0$ met $x_n^2 > 2$, dan is $x_{n+1}$ (een
\omterm{def:g11:stat:mean}{gemiddelde} van positieve getallen) positief en $x_{n+1}^2 - 2 > 0$.

\textbf{8.} $x_{n+1} - x_n = \frac{2 - x_n^2}{2x_n} < 0$ volgens vraag 7:
strikt \omterm{def:g10:functions:variations}{dalend}.

\textbf{9.} \omterm{def:g10:functions:variations}{Dalend} en naar beneden begrensd (door $1$, want $x_n^2 > 2 > 1$
en $x_n > 0$): volgens de stelling van de monotone convergentie \omterm{def:g12:seq:limit}{convergeert}
$(x_n)$ naar een zekere $L \geq 1$.

\textbf{10.} Limieten respecteren de algebra: uit
$x_{n+1} = \frac12\left(x_n + \frac{2}{x_n}\right)$ en
$x_n \to L \geq 1 > 0$ volgt $L = \frac12\left(L + \frac2L\right)$, dus
$L^2 = 2$, en omdat $L$ positief is, $L = \sqrt2$. Vonnis: convergentie
bewezen, limiet bepaald --- Heron met lof vrijgesproken.

\textbf{11.} $x_{n+1} - \sqrt2 = \frac{x_n^2 - 2\sqrt2\,x_n + 2}{2x_n} =
\frac{(x_n - \sqrt2)^2}{2x_n}$: precies $e_{n+1} = \frac{e_n^2}{2x_n}$, en
$x_n > \sqrt2$ geeft $e_{n+1} \leq \frac{e_n^2}{2\sqrt2}$. Gekwadrateerde
fout: elke stap verdubbelt het aantal juiste decimalen, zoals sinds jaar 9
waargenomen.

\textbf{12.} $e_0 \approx 0.5858$, $e_1 \approx 0.0858$,
$e_2 \approx 0.00245$, $e_3 \approx 2.1 \times 10^{-6}$. Verhoudingen:
$\frac{e_1}{e_0^2} \approx 0.25 = \frac{1}{2x_0}$;
$\frac{e_2}{e_1^2} \approx 0.333 = \frac{1}{2x_1}$;
$\frac{e_3}{e_2^2} \approx 0.353 \approx \frac{1}{2x_2}$: de stelling aan het
werk.

\textbf{13.} $H_{2^{18}} \geq 1 + 9 = 10$: ongeveer $260\,000$ termen
($2^{18} = 262\,144$) om alleen al $10$ te passeren --- divergentie in
slakkengang (en $H_n > 100$ zou meer termen vragen dan er atomen in eender
welke bibliotheek zitten).

\textbf{14.} $S_n = 2 - \frac{1}{2^n}$ (meetkundige som), en
$\frac{1}{2^n} \to 0$ (\cref{thm:g12:seq:geometric}), dus $S_n \to 2$: de
eindeloos afgebeten chocoladereep streeft naar het geheel zonder het ooit te
bereiken --- nu in de officiële taal van de limieten.

\textbf{15.} Voor $k \geq 2$ is
$\frac{1}{k^2} \leq \frac{1}{k(k-1)} = \frac{1}{k-1} - \frac1k$, dus
$S_n \leq 1 + \left(1 - \frac1n\right) < 2$: \omterm{def:g12:seq:monotonic}{stijgend} en \omterm{def:g12:seq:bounded}{naar boven begrensd},
dus convergent (monotone convergentie). Euler doopte de limiet later:
$\frac{\pi^2}{6}$.

\textbf{16.} Dat de termen naar $0$ streven is \emph{noodzakelijk} opdat de
sommen tot rust zouden komen, maar het beslist niets: de harmonische termen
$\frac1n \to 0$ en toch ontploffen de sommen; de termen
$\frac{1}{n^2} \to 0$ en de sommen convergeren. \emph{Hoe snel} de termen
uitsterven is de hele vraag --- de theorie van de reeksen, opgebouwd in de
universitaire volumes.

\textbf{17.} $a_1 = \sqrt2 \approx 1.41421$, $b_1 = 1.5$;
$a_2 \approx 1.45648$, $b_2 \approx 1.45711$: na twee iteraties stemmen ze al
op drie decimalen overeen --- duizelingwekkende snelheid.

\textbf{18.} $b_{n+1} - a_{n+1} = \frac{a_n + b_n}{2} - \sqrt{a_n b_n} =
\frac{(\sqrt{b_n} - \sqrt{a_n})^2}{2} \geq 0$: de \omterm{def:g11:stat:mean}{gemiddelden} blijven
geordend. $(a_n)$ stijgt:
$a_{n+1} = \sqrt{a_n b_n} \geq \sqrt{a_n \cdot a_n} = a_n$; $(b_n)$ daalt
symmetrisch.

\textbf{19.} $\frac{b_{n+1} - a_{n+1}}{b_n - a_n} =
\frac{(\sqrt{b_n} - \sqrt{a_n})^2}
{2(\sqrt{b_n} - \sqrt{a_n})(\sqrt{b_n} + \sqrt{a_n})}
= \frac{\sqrt{b_n} - \sqrt{a_n}}{2(\sqrt{b_n} + \sqrt{a_n})} \leq \frac12$:
de kloof halveert minstens, dus $b_n - a_n \to 0$; samen met vraag 18 zijn de
\omterm{def:g12:seq:sequence}{rijen} aangrenzend en delen ze een limiet $M(1, 2)$.

\textbf{20.} De derde iteratie geeft $a_3 \approx b_3 \approx 1.456791$: dus
$M(1, 2) \approx 1.456791$ na drie slagen aan de zwengel (de kloof wordt
ruwweg \emph{gekwadrateerd}, net als bij Heron). Rangschikking van de
snelheden in deze opgave, van traagst naar snelst: de harmonische sommen
(divergentie op gletsjertempo), de meetkundige sommen (de fout halveert elke
stap), Heron en het \omterm{def:g12:seq:arith-geom}{rekenkundig-meetkundig} \omterm{def:g11:stat:mean}{gemiddelde} (de fout wordt elke
stap gekwadrateerd) --- en het was die onaardse snelheid die Gauss vertelde
dat hij een nieuwe ader in de analyse had aangeboord.
\end{solution}
