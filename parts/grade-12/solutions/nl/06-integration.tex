\section*{Hoofdstuk \ref{ch:g12:integ} --- Integreren}

\begin{solution}{exo:g12:integ:1}
\[
\int_1^2 \left(3x^2 - \frac{1}{x^2}\right)\dd x
= \left[x^3 + \frac1x\right]_1^2 = \left(8 + \tfrac12\right) - 2 = \frac{13}{2}.
\]
\[
\int_0^{\pi/2} \cos t\,\dd t = \bigl[\sin t\bigr]_0^{\pi/2} = 1 .
\]
Met $u = 2t+1$ is $\frac{1}{2t+1} = \frac12\,\frac{u'}{u}$:
\[
\int_0^1 \frac{\dd t}{2t+1} = \frac12\bigl[\ln(2t+1)\bigr]_0^1
= \frac{\ln 3}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:2}
$x\,\eu^{x^2} = \frac12\,(x^2)'\,\eu^{x^2}$ heeft als \omterm{def:g12:integ:primitive}{primitieve functies}
$\frac12\eu^{x^2} + c$. De voorwaarde $F(0) = 1$ geeft $\frac12 + c = 1$,
dus
\[
F(x) = \frac{\eu^{x^2} + 1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:3}
\[
\mu = \frac{1}{\pi}\int_0^\pi \sin t\,\dd t
= \frac{1}{\pi}\bigl[-\cos t\bigr]_0^\pi = \frac{2}{\pi} \approx 0.64 .
\]
Grafisch heeft de rechthoek met basis $\intcc{0}{\pi}$ en hoogte
$\frac{2}{\pi}$ dezelfde oppervlakte als de boog van de sinuskromme.
\end{solution}

\begin{solution}{exo:g12:integ:4}
Neem $u' = 1$ en $v = \ln t$, dus $u = t$ en $v' = \frac1t$:
\[
\int_1^{\eu} \ln t\,\dd t = \bigl[t\ln t\bigr]_1^{\eu} - \int_1^{\eu} 1\,\dd t
= \eu - (\eu - 1) = 1 .
\]
Neem $u' = \sin t$ en $v = t$, dus $u = -\cos t$ en $v' = 1$:
\[
\int_0^\pi t\sin t\,\dd t = \bigl[-t\cos t\bigr]_0^\pi + \int_0^\pi \cos t\,\dd t
= \pi + 0 = \pi .
\]
\end{solution}

\begin{solution}{exo:g12:integ:5}
De krommen snijden elkaar waar $x^2 = x + 2$, dus in $x \in \{-1, 2\}$, en op
$\intcc{-1}{2}$ ligt de rechte boven de \omterm{def:g11:quad:parabola}{parabool}. De oppervlakte is
\[
\int_{-1}^{2} \bigl(x + 2 - x^2\bigr)\dd x
= \left[\frac{x^2}{2} + 2x - \frac{x^3}{3}\right]_{-1}^{2}
= \frac{10}{3} - \left(-\frac{7}{6}\right) = \frac{9}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:integ:6}
\emph{1.} $I = \bigl[\ln(1+t)\bigr]_0^1 = \ln 2$.

\emph{2.}
\[
I_n + I_{n+1} = \int_0^1 \frac{t^n(1 + t)}{1+t}\,\dd t
= \int_0^1 t^n \,\dd t = \frac{1}{n+1}.
\]
Op $\intcc{0}{1}$ is $0 \leq \dfrac{t^n}{1+t} \leq t^n$, dus door
\omterm{def:g10:algebra:equation}{vergelijking} $0 \leq I_n \leq \frac{1}{n+1}$.

\emph{3.} Uit de recursie volgt, met een telescoperende inductie,
\[
I_n = (-1)^n\left(I_0 - \left(1 - \frac12 + \dots + \frac{(-1)^{n-1}}{n}\right)\right),
\]
dus $\displaystyle\sum_{k=1}^{n} \frac{(-1)^{k-1}}{k} = \ln 2 - (-1)^n I_n$
(met $I_0 = \ln 2$). Omdat $\abs{I_n} \leq \frac{1}{n+1} \to 0$, streeft de
alternerende harmonische som naar $\ln 2$.
\end{solution}

\begin{solution}{exo:g12:integ:7}
Afstand:
\[
d = \int_0^{100} 30\left(1 - \eu^{-t/50}\right)\dd t
= 30\left[t + 50\,\eu^{-t/50}\right]_0^{100}
= 30\left(100 + 50\eu^{-2} - 50\right)
= 1500 + 1500\,\eu^{-2}.
\]
In getallen is $d \approx 1500 + 203 = 1703$ m. \omterm{def:g11:stat:mean}{Gemiddelde} snelheid:
$\frac{d}{100} = 15\bigl(1 + \eu^{-2}\bigr) \approx 17.0$ m/s.
\end{solution}

\begin{solution}{exo:g12:integ:8}
\emph{1.} $S_n$ is de totale oppervlakte van $n$ rechthoeken met breedte
$\frac1n$ en hoogten $f\left(\frac kn\right)$ voor $k = 1, \dots, n$, waarbij
$f(t) = \frac{1}{1+t}$: het is de rechthoekbenadering ``met het rechteruiteinde''
van de oppervlakte onder $f$ op $\intcc{0}{1}$.

\emph{2.} $f$ is \omterm{def:g10:functions:variations}{dalend}, dus geldt op elk \omterm{def:g10:numbers:interval}{interval}
$\intcc{\frac{k-1}{n}}{\frac kn}$ dat
$f\left(\frac kn\right) \leq f(t) \leq f\left(\frac{k-1}{n}\right)$;
integreren en over $k$ sommeren geeft
\[
S_n \leq \int_0^1 f(t)\,\dd t \leq S_n + \frac1n\bigl(f(0) - f(1)\bigr)
= S_n + \frac{1}{2n}.
\]
(De middelste term uit de opgave, met $f(0) - f(1) = 1 - \frac12$, is precies
die grens.) Bijgevolg is $0 \leq \ln 2 - S_n \leq \frac{1}{2n} \to 0$, dus
$S_n \to \ln 2$.
\end{solution}

\begin{solution}{exo:g12:integ:9}
\emph{1.} $W_0 = \int_0^{\pi/2} \dd t = \frac{\pi}{2}$ en
$W_1 = \bigl[-\cos t\bigr]_0^{\pi/2} = 1$.

\emph{2.} Integreer partieel met $u' = \sin t$ en $v = \sin^{n+1} t$, dus
$u = -\cos t$ en $v' = (n+1)\sin^n t \cos t$:
\[
W_{n+2} = \bigl[-\cos t \sin^{n+1} t\bigr]_0^{\pi/2}
+ (n+1)\int_0^{\pi/2} \cos^2 t\,\sin^n t\,\dd t
= (n+1)\int_0^{\pi/2} (1 - \sin^2 t)\sin^n t\,\dd t,
\]
want de haak verdwijnt in beide uiteinden. Bijgevolg is
$W_{n+2} = (n+1)(W_n - W_{n+2})$, dus
$W_{n+2} = \frac{n+1}{n+2} W_n$.

\emph{3.} $W_2 = \frac12 W_0 = \frac{\pi}{4}$,
$W_3 = \frac23 W_1 = \frac23$, $W_4 = \frac34 W_2 = \frac{3\pi}{16}$. Op
$\intoo{0}{\frac\pi2}$ is $0 < \sin t < 1$, dus
$\sin^{n+1} t \leq \sin^n t$, met strikte ongelijkheid binnenin; integreren
geeft $0 < W_{n+1} \leq W_n$ (zelfs $<$): $(W_n)$ is \omterm{def:g10:functions:variations}{dalend} en positief.
\end{solution}

\begin{solution}{pb:g12:integ:1}
\textbf{1.} $\left[x^3 - x^2 + x\right]_0^1 = 1$; \quad
$\left[\ln x\right]_1^{\eu} = 1$; \quad
$\left[\sin x\right]_0^{\pi/2} = 1$.

\textbf{2.} $\left[\frac12 \eu^{x^2}\right]_0^1 = \frac{\eu - 1}{2}$; \quad
$\left[\ln(x^2 + 1)\right]_0^1 = \ln 2$.

\textbf{3.} $\int_0^1 x\eu^x = \left[x\eu^x\right]_0^1 - \int_0^1 \eu^x =
\eu - (\eu - 1) = 1$. En
$\int_1^{\eu} \ln x = \left[x\ln x\right]_1^{\eu} - \int_1^{\eu} 1 =
\eu - (\eu - 1) = 1$.

\textbf{4.} $\frac{1}{\pi}\int_0^\pi \sin x\,\dd x = \frac{2}{\pi} \approx
0.64$: de \omterm{def:g11:stat:mean}{gemiddelde} hoogte van de sinusboog is $\frac2\pi$, merkbaar meer
dan een half --- de boog is dik bij haar top.

\textbf{5.} $\int_0^1 (x - x^2)\,\dd x = \frac12 - \frac13 = \frac16$.

\textbf{6.} $L_n = \frac1n \sum_{k=0}^{n-1} \left(\frac kn\right)^2$ en
$U_n = \frac1n \sum_{k=1}^{n} \left(\frac kn\right)^2$: rechthoeken met
breedte $\frac1n$ waarvan de hoogte in het linkeruiteinde (onder de kromme)
of het rechteruiteinde (erboven) wordt afgelezen.

\textbf{7.} Waar voor $n = 1$ ($1 = \frac{1 \cdot 2 \cdot 3}{6}$). Geldt ze
voor $n$, dan geeft $(n+1)^2$ erbij optellen
\[
\frac{n(n+1)(2n+1)}{6} + (n+1)^2
= \frac{(n+1)\left(2n^2 + n + 6n + 6\right)}{6}
= \frac{(n+1)(n+2)(2n+3)}{6},
\]
de formule voor $n + 1$: erfelijkheid, klaar.

\textbf{8.} $U_n = \frac{1}{n^3} \cdot \frac{n(n+1)(2n+1)}{6} =
\frac{(1 + \frac1n)(2 + \frac1n)}{6} \to \frac26 = \frac13$, en
$L_n = U_n - \frac1n \to \frac13$: ingeklemd tussen haar trappen is de
oppervlakte precies $\frac13$.

\textbf{9.} $\int_0^1 x^2\,\dd x = \left[\frac{x^3}{3}\right]_0^1 =
\frac13$: één regel. De hoofdstelling zet een oneindige insluiting om in het
invullen van één \omterm{def:g12:integ:primitive}{primitieve functie} --- en daarom heet ze de hoofdstelling.

\textbf{10.} Segment: $\int_{-1}^{1}(1 - x^2)\,\dd x = 2 - \frac23 =
\frac43$. Driehoek met als basis de koorde (lengte $2$, op hoogte $1$) en top
$(0,0)$: oppervlakte $\frac12 \times 2 \times 1 = 1$. Verhouding: $\frac43$
--- precies de stelling van Archimedes, nagegaan met de machine die hij niet
had.

\textbf{11.} $T\left(1 + \frac14 + \frac{1}{16} + \dots\right) = T \cdot
\frac{1}{1 - \frac14} = \frac43 T$ (meetkundige reeks met \omterm{def:g11:seq:geometric}{reden} $\frac14$):
met $T = 1$ opnieuw de oppervlakte $\frac43$ van het segment. Archimedes
sommeerde de reeks met een zuiver \omterm{def:g12:seq:arith-geom}{meetkundig} argument --- de kwarthappen van
het chocoladereeptype, twee millennia voordat limieten woorden waren.

\textbf{12.} De uitputting vraagt een vernuftige meetkundige gelijkvormigheid
en geeft geval per geval exacte oppervlakten; Riemann vraagt alleen monotonie
en geeft een universele definitie, tegen de prijs van sommen berekenen; en de
hoofdstelling vraagt het hele bouwwerk van de \omterm{def:g12:deriv:derivative}{afgeleiden} en betaalt dat terug
door oppervlakten tot berekeningen van één regel te maken.

\textbf{13.} $A(1) = 0$ en $A'(x) = \frac1x$ (hoofdstelling): $A$ is de
\omterm{def:g12:integ:primitive}{primitieve functie} van $\frac1x$ op $\intoo{0}{+\infty}$ die in $1$
verdwijnt --- en dat is precies de \omterm{def:g12:exp:ln}{natuurlijke logaritme}.

\textbf{14.} $g'(x) = \frac{a}{ax} - \frac1x = 0$: $g$ is constant; in
$x = 1$ is $g(1) = A(a) - A(1) = A(a)$. Bijgevolg is $A(ax) = A(a) + A(x)$
voor alle $x$, en met $x = b$: $\ln(ab) = \ln a + \ln b$. De oppervlakte
onder de \omterm{def:g11:func:reference}{hyperbool} trekt zich niets aan van de horizontale schaal: $t$ met
$a$ uitrekken drukt $\frac1t$ met $a$ samen, en de oppervlakte overleeft ---
\omterm{def:g12:exp:ln}{logaritmen} optellen is een schaalsymmetrie.

\textbf{15.} Herhalen geeft $\ln(a^n) = n\ln a$ (inductie op $n$). En
$0 = \ln 1 = \ln\left(a \cdot \frac1a\right) = \ln a + \ln\frac1a$, dus
$\ln\frac1a = -\ln a$.

\textbf{16.} $\frac{1}{11} + \frac{1}{12} + \dots + \frac{1}{20} \approx
0.669$, tegenover $\ln 2 \approx 0.693$: tien rechthoeken onder
$\frac{1}{1+t}$, en al dichtbij. De harmonische reeks divergeert, maar haar
\emph{plakken} van $n$ tot $2n$ zetten zich vast op $\ln 2$: divergentie op
slakkentempo, gemeten door een oppervlakte.

\textbf{17.} Afstand: $\int_0^{10} 3t^2\,\dd t = 1000$ m. \omterm{def:g11:stat:mean}{Gemiddelde}
snelheid: $100$ m/s. Het ogenblik met $v = 100$: $3t^2 = 100$, dus
$t = \sqrt{\frac{100}{3}} \approx 5.77$ s.

\textbf{18.} Massa: $\int_0^4 (2 + x)\,\dd x = 8 + 8 = 16$ kg. Moment:
$\int_0^4 (2x + x^2)\,\dd x = 16 + \frac{64}{3} = \frac{112}{3}$.
Massamiddelpunt: $\frac{112/3}{16} = \frac73 \approx 2.33$ m --- voorbij het
\omterm{prop:g10:coordgeom:midpoint}{midden} geduwd door het zwaardere rechteruiteinde, zoals de intuïtie eist.

\textbf{19.} $W_0 = \frac\pi2$; $W_1 = \left[-\cos t\right]_0^{\pi/2} = 1$;
en $W_2 = \int_0^{\pi/2} \frac{1 - \cos 2t}{2}\,\dd t = \frac\pi4$.

\textbf{20.} Oppervlakte: de definitie, ingeklemd door trappen. Opstapeling:
afstanden uit snelheden, massa's uit dichtheden --- de \omterm{def:g12:integ:area}{integraal} als lopend
totaal. \omterm{def:g12:integ:primitive}{Primitieve functie}: de hoofdstelling, de sluipweg van Newton en
Leibniz door het geduld van Archimedes heen. De parel: de toets $\ln$ is de
oppervlakte onder $\frac1t$, en de schaalsymmetrie van die oppervlakte
\emph{is} de identiteit $\ln(ab) = \ln a + \ln b$.
\end{solution}
