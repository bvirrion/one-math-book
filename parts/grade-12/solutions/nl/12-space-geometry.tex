\section*{Hoofdstuk \ref{ch:g12:space} --- Vectoren, rechten en vlakken in de ruimte}

\begin{solution}{exo:g12:space:1}
$\vect{AB}(2, 1, -1)$ en $\vect{AC}(1, -1, 1)$ zijn niet evenredig
($\frac21 \neq \frac{1}{-1}$), dus de punten liggen niet op één rechte. De
rechte $(AB)$:
\[
(x, y, z) = (1 + 2t,\; t,\; 2 - t), \qquad t \in \R .
\]
\end{solution}

\begin{solution}{exo:g12:space:2}
$2(x - 1) - (y - 1) + 3(z - 0) = 0$, dat wil zeggen
\[
2x - y + 3z - 1 = 0 .
\]
Voor $B(0,2,1)$: $0 - 2 + 3 - 1 = 0$: ja, $B$ ligt in het vlak.
\end{solution}

\begin{solution}{exo:g12:space:3}
$\vect{AB}(1,1,0)$ en $\vect{AC}(1,0,1)$:
\[
\cos\widehat{A}
= \frac{\vect{AB}\cdot\vect{AC}}{\norm{\vect{AB}}\,\norm{\vect{AC}}}
= \frac{1}{\sqrt2\,\sqrt2} = \frac12,
\]
dus $\widehat A = 60^\circ$.
\end{solution}

\begin{solution}{exo:g12:space:4}
De punten van $\mathcal D$: $(1 + t,\, 2 - t,\, 2t)$. Invullen in de
\omterm{def:g10:algebra:equation}{vergelijking} van $\mathcal P$:
\[
2(1+t) + (2 - t) - 2t + 1 = 5 - t = 0 \iff t = 5 .
\]
Eén enkel snijpunt: $(6, -3, 10)$.
\end{solution}

\begin{solution}{exo:g12:space:5}
De \omterm{def:g12:space:normal}{normaalvectoren} $(1, -1, 2)$ en $(2, 1, 1)$ zijn niet \omterm{def:g11:vect:collinear}{collineair}, dus de
vlakken snijden elkaar volgens een rechte. De twee \omterm{def:g10:algebra:equation}{vergelijkingen} optellen
geeft $3x + 3z = 5$, dus $x = \frac53 - z$; de eerste \omterm{def:g10:algebra:equation}{vergelijking} geeft dan
$y = x + 2z - 1 = \frac23 + z$. Met $z = t$:
\[
(x, y, z) = \left(\frac53 - t,\; \frac23 + t,\; t\right), \qquad t \in \R,
\]
een rechte met \omterm{def:g11:vect:direction}{richtingsvector} $(-1, 1, 1)$. (Beide \omterm{def:g10:algebra:equation}{vergelijkingen} kloppen
identiek.)
\end{solution}

\begin{solution}{exo:g12:space:6}
De richtingen $\vec u_1(1, 2, -1)$ en $\vec u_2(1, 1, 2)$ zijn niet
\omterm{def:g11:vect:collinear}{collineair}, dus de rechten zijn niet evenwijdig. Snijden zouden ze alleen als
\[
1 + t = 2 + s, \qquad 2t = 1 + s, \qquad 3 - t = 1 + 2s .
\]
De twee eerste geven $t = 1 + s$ en $2(1+s) = 1 + s$, dus $s = -1$ en
$t = 0$; maar dan luidt de derde $3 = -1$, ongerijmd. Geen gemeenschappelijk
punt en niet evenwijdig: de rechten zijn kruisend.
\end{solution}

\begin{solution}{exo:g12:space:7}
\emph{1.} $\operatorname{dist} = \dfrac{\abs{2\cdot3 - 1 + 2\cdot1 - 3}}
{\sqrt{4 + 1 + 4}} = \dfrac{4}{3}$.

\emph{2.} $H = M_0 + t\,\vec n$ met $\vec n(2, -1, 2)$, waarbij $t$ zo gekozen
wordt dat $H \in \mathcal P$:
\[
2(3 + 2t) - (1 - t) + 2(1 + 2t) - 3 = 4 + 9t = 0 \iff t = -\frac49 .
\]
Dus $H\left(\frac{19}{9}, \frac{13}{9}, \frac19\right)$, en
\[
M_0H = \norm{t\,\vec n} = \frac49 \times 3 = \frac43,
\]
in overeenstemming met de afstandsformule.
\end{solution}

\begin{solution}{exo:g12:space:8}
\omterm{def:g10:coordgeom:system}{Coördinaten}: $G(1,1,1)$, en het vlak $(BDE)$ gaat door $B(1,0,0)$,
$D(0,1,0)$ en $E(0,0,1)$.

\emph{1.} Het vlak $(BDE)$ heeft als \omterm{def:g10:algebra:equation}{vergelijking} $x + y + z = 1$ (voldaan
door de drie punten, die niet op één rechte liggen), dus is $\vec n(1,1,1)$
er een \omterm{def:g12:space:normal}{normaalvector} van. Omdat $\vect{AG} = (1,1,1) = \vec n$, is de
diagonaal $(AG)$ \omterm{def:g11:scal:orthogonal}{orthogonaal} met het vlak $(BDE)$.

\emph{2.} Afstand vanaf $A(0,0,0)$:
$\frac{\abs{0 + 0 + 0 - 1}}{\sqrt3} = \frac{1}{\sqrt3} = \frac{\sqrt3}{3}$.
De rechte $(AG)$ is $(t, t, t)$; ze ontmoet het vlak zodra $3t = 1$: in het
punt $\left(\frac13, \frac13, \frac13\right)$, het zwaartepunt van de
driehoek $BDE$.
\end{solution}

\begin{solution}{exo:g12:space:9}
\emph{1.} $\vect{AB}(1, 0, -1)$, $\vect{AC}(-1, 1, 0)$ en
$\vect{AD}(1, 1, 1)$. Zou $\vect{AD} = a\vect{AB} + b\vect{AC}$ gelden, dan
geven de \omterm{def:g10:coordgeom:system}{coördinaten} $a - b = 1$, $b = 1$ en $-a = 1$: de twee eerste dwingen
$a = 2$ af, in tegenspraak met $a = -1$. Niet \omterm{def:g12:space:collinear}{coplanair}.

\emph{2.} $\vect{BC}(-2, 1, 1)$ en $\vect{BD}(0, 1, 2)$. Los
$\vec n \cdot \vect{BC} = \vec n \cdot \vect{BD} = 0$ op: uit $b + 2c = 0$
volgt $b = -2c$; dan geeft $-2a - 2c + c = 0$ dat $c = -2a$. Met $a = 1$:
$\vec n(1, 4, -2)$. Het vlak door $B(2,1,0)$:
\[
x + 4y - 2z - 6 = 0 .
\]

\emph{3.} Afstand vanaf $A(1,1,1)$:
$\dfrac{\abs{1 + 4 - 2 - 6}}{\sqrt{21}} = \dfrac{3}{\sqrt{21}}$.
Oppervlakte van $BCD$: $\norm{\vect{BC}}^2 = 6$, $\norm{\vect{BD}}^2 = 5$ en
$\vect{BC}\cdot\vect{BD} = 3$, dus
\[
\text{Opp} = \frac12\sqrt{6 \times 5 - 9} = \frac{\sqrt{21}}{2}.
\]

\emph{4.}
\[
V = \frac13 \times \frac{\sqrt{21}}{2} \times \frac{3}{\sqrt{21}}
= \frac12 .
\]
\end{solution}

\begin{solution}{pb:g12:space:1}
\textbf{1.} Het vlak: $x + y + z = 1$. De rechte: $(t, t, t)$. Het snijpunt:
$3t = 1$, dus $\left(\frac13, \frac13, \frac13\right)$.

\textbf{2.} $\dfrac{\abs{0 + 0 + 0 - 1}}{\sqrt3} = \dfrac{1}{\sqrt3} =
\dfrac{\sqrt3}{3}$.

\textbf{3.} $1 - 1 + 0 = 0$: \omterm{def:g11:scal:orthogonal}{orthogonaal}.

\textbf{4.} Dezelfde normaal $(1,1,1)$, maar $\frac52 \neq 1$: strikt
evenwijdige vlakken. De rechte: haar richting geeft
$(1,1,0) \cdot (1,1,1) = 2 \neq 0$: niet evenwijdig met het vlak, dus ze prikt
er in één punt doorheen.

\textbf{5.} De \omterm{prop:g10:coordgeom:midpoint}{middens}: $\left(1, \frac12, 0\right)$,
$\left(\frac12, 1, 0\right)$, $\left(0, 1, \frac12\right)$,
$\left(0, \frac12, 1\right)$, $\left(\frac12, 0, 1\right)$ en
$\left(1, 0, \frac12\right)$: elk heeft $\frac32$ als som van zijn
\omterm{def:g10:coordgeom:system}{coördinaten}.

\textbf{6.} De normaal van $P$ is $(1,1,1)$, de richting van $(AG)$:
loodrecht. Het middelpunt $\left(\frac12,\frac12,\frac12\right)$ heeft
coördinatensom $\frac32$: het ligt op $P$.

\textbf{7.} Opeenvolgende \omterm{prop:g10:coordgeom:midpoint}{middens} verschillen over \omterm{def:g11:vect:vector}{vectoren} zoals
$\left(-\frac12, \frac12, 0\right)$: telkens lengte $\frac{\sqrt2}{2}$ ---
zes gelijke zijden.

\textbf{8.} In $\left(\frac12, 1, 0\right)$: de zijdevectoren
$\left(\frac12, -\frac12, 0\right)$ en $\left(-\frac12, 0, \frac12\right)$
hebben \omterm{def:g12:space:dot}{scalair product} $-\frac14$ en \omterm{def:g11:scal:dot}{normen} $\frac{\sqrt2}{2}$, dus
$\cos = -\frac12$: de hoek is $120^\circ$. Gelijke zijden en gelijke hoeken
van $120^\circ$: een regelmatige zeshoek.

\textbf{9.} Omtrek $6 \times \frac{\sqrt2}{2} = 3\sqrt2 \approx 4.24$.
Oppervlakte $6 \times \frac{\sqrt3}{4} \times \frac12 = \frac{3\sqrt3}{4}
\approx 1.30$: groter dan een zijvlak (oppervlakte $1$) --- een snede groter
dan elke zijde van de doos; alleen de diagonale rechthoek
($\sqrt2 \approx 1.41$) doet beter.

\textbf{10.} $(t,t,t)$ ontmoet $P$ bij $t = \frac12$: het middelpunt --- en
dat is het \omterm{prop:g10:coordgeom:midpoint}{midden} van $[AG]$ ($A$ bij $t = 0$, $G$ bij $t = 1$).

\textbf{11.} Voor $c = \frac12$ snijdt het vlak de drie ribben bij de hoek van
$A$, in $\left(\frac12,0,0\right)$, $\left(0,\frac12,0\right)$ en
$\left(0,0,\frac12\right)$: een gelijkzijdige driehoek met zijde
$\frac{\sqrt2}{2}$. Naarmate $c$ groeit, groeit die driehoek, worden zijn
hoeken afgeknot tot een zeshoek (regelmatig precies bij $c = \frac32$), en
krimpt het \omterm{def:g10:functions:function}{beeld} daarna symmetrisch terug tot een driehoek bij de hoek van
$G$: de gedaanteverwisseling van een geslepen kristal.

\textbf{12.} Elke zijde van de \omterm{def:g10:numbers:interunion}{doorsnede} is de snijding van het snijvlak met
één \emph{zijvlak} van de kubus, en een vlak ontmoet een zijvlak (een vlakke
veelhoek) in ten hoogste één lijnstuk: dus ten hoogste $6$ zijden. Zeven is
onmogelijk; drie tot en met zes komen alle voor (de vragen 8 en 11 tonen er
twee van).

\textbf{13.} Grondvlak $ABD$: een rechthoekige driehoek met oppervlakte
$\frac12$; hoogte $AE = 1$: volume
$\frac13 \times \frac12 \times 1 = \frac16$.

\textbf{14.} Elk paar uit $B, D, E, G$ verschilt in precies twee \omterm{def:g10:coordgeom:system}{coördinaten}
over $1$: alle zes de afstanden zijn $\sqrt2$ --- regelmatig. De kubus valt
uiteen in $BDEG$ plus vier hoekviervlakken congruent met $ABDE$: volume
$1 - 4 \times \frac16 = \frac13$.

\textbf{15.} Het vlak $(BDE)$: $x + y + z = 1$; het middelpunt
$\left(\frac12,\frac12,\frac12\right)$: afstand
$\frac{\abs{\frac32 - 1}}{\sqrt3} = \frac{1}{2\sqrt3} = \frac{\sqrt3}{6}$.

\textbf{16.} $\vect{AG} = (1,1,1)$ en $\vect{BH} = (-1,1,1)$:
$\cos\theta = \frac{-1 + 1 + 1}{3} = \frac13$, dus
$\theta \approx 70.5^\circ$, met supplement $\approx 109.5^\circ$. Vier
elektronenparen stoten elkaar af en spreiden zich zo ver mogelijk rond het
koolstofatoom: het optimum zijn de hoekpunten van het regelmatige viervlak,
waarvan de richtingen van het middelpunt naar de hoekpunten elkaar onder
precies die hoek ontmoeten --- diamant is vraag 14, uitgekristalliseerd.

\textbf{17.} $(1, t, t)$ ligt op $P$ zodra $1 + 2t = \frac32$, dus
$t = \frac14$: het punt $\left(1, \frac14, \frac14\right)$.

\textbf{18.} $(AB)$: de punten $(t, 0, 0)$; $(EG)$: de punten $(s, s, 1)$.
Snijden zou $0 = 1$ vragen in de derde coördinaat: nooit. Evenwijdig zijn zou
vragen dat $(1,0,0)$ en $(1,1,0)$ \omterm{def:g11:vect:collinear}{collineair} zijn: nee. Ze snijden elkaar
niet en zijn niet evenwijdig: kruisend --- twee gangen op verschillende
verdiepingen.

\textbf{19.} $B + t(1,1,1) = (1 + t, t, t)$ ligt op $P$ zodra
$1 + 3t = \frac32$, dus $t = \frac16$: de projectie
$\left(\frac76, \frac16, \frac16\right)$, met coördinatensom $\frac32$: op
$P$, zoals gevraagd.

\textbf{20.} Parametervoorstellingen prikken door (de vragen 1, 10 en 17);
\omterm{def:g12:space:normal}{normaalvectoren} meten hoeken en afstanden (de vragen 2, 8, 15 en 16);
\omterm{def:g10:algebra:equation}{vergelijkingen} van vlakken snijden \omterm{def:g10:numbers:interunion}{doorsneden} uit (de gedaanteverwisseling van
driehoek naar zeshoek). En het speelterrein betaalde het bezoek terug: een
regelmatige zeshoek in een doos van vierkanten, een regelmatig viervlak in de
hoeken, de hoek van diamant, en twee rechten die elkaar negeren --- meetkunde
die alleen de ruimte kan bieden.
\end{solution}
