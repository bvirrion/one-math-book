\section*{Hoofdstuk \ref{ch:g12:limcont} --- Limieten en continuïteit}

\begin{solution}{exo:g12:limcont:1}
Delen door $x^3$:
$\dfrac{2x^3 - x + 1}{5x^3 + x^2} = \dfrac{2 - 1/x^2 + 1/x^3}{5 + 1/x}
\to \dfrac{2}{5}$.

$x^2 + 3x - 1 = x^2(1 + 3/x - 1/x^2) \to +\infty$ voor $x \to -\infty$, want
$x^2 \to +\infty$ en de haak streeft naar $1$.

Voor $x \to 2^+$ streeft de teller naar $3 > 0$ en de noemer naar $0^+$, dus
$\dfrac{x+1}{x-2} \to +\infty$.

Voor $x > 0$ is $-\dfrac1x \leq \dfrac{\cos x}{x} \leq \dfrac1x$, zodat de
limiet met de insluitstelling $0$ is.
\end{solution}

\begin{solution}{exo:g12:limcont:2}
\emph{1.} $2x^2 - 3x + 1 = (x-1)(2x-1)$ (de \omterm{def:g11:quad:discriminant}{wortel} $x=1$ springt in het oog
uit $2 - 3 + 1 = 0$). Voor $x \neq 1$ schrap je $x - 1$: $f(x) = 2x - 1$.

\emph{2.} Bijgevolg is $\lim_{x\to1} f(x) = 1$. Met $f(1) = 1$ breid je $f$
uit tot de veeltermfunctie $x \mapsto 2x-1$ op $\R$, die in $1$ \omterm{def:g12:limcont:continuity}{continu} is.
\end{solution}

\begin{solution}{exo:g12:limcont:3}
Schrijf $f(x) = \dfrac{3x - 1}{x + 2} = \dfrac{3(x+2) - 7}{x+2}
= 3 - \dfrac{7}{x+2}$. Voor $x \to \pm\infty$ is $f(x) \to 3$: de rechte
$y = 3$ is in beide oneindigheden een \omterm{def:g12:limcont:asymptote}{horizontale asymptoot}. Voor
$x \to (-2)^\pm$ is $f(x) \to \mp\infty$: de rechte $x = -2$ is een
\omterm{def:g12:limcont:asymptote}{verticale asymptoot}. De kromme is een \omterm{def:g11:func:reference}{hyperbool}, onder $y=3$ voor $x > -2$ en
erboven voor $x < -2$.
\end{solution}

\begin{solution}{exo:g12:limcont:4}
Vermenigvuldig met de toegevoegde uitdrukking:
\[
\sqrt{x^2 + x} - x = \frac{x^2 + x - x^2}{\sqrt{x^2+x} + x}
= \frac{x}{x\left(\sqrt{1 + 1/x} + 1\right)}
= \frac{1}{\sqrt{1 + 1/x} + 1}
\xrightarrow[x\to+\infty]{} \frac{1}{2}.
\]
Net zo geldt
\[
\frac{\sqrt{1+x} - 1}{x} = \frac{(1+x) - 1}{x\left(\sqrt{1+x} + 1\right)}
= \frac{1}{\sqrt{1+x} + 1} \xrightarrow[x\to0]{} \frac{1}{2}.
\]
\end{solution}

\begin{solution}{exo:g12:limcont:5}
\emph{1.} Omdat $\sin x \geq -1$, is $f(x) \geq x - 1 \to +\infty$, en
besluiten we door \omterm{def:g10:algebra:equation}{vergelijking}. En toch verdwijnt $f'(x) = 1 + \cos x$ in elk
oneven veelvoud van $\pi$, en wisselt de \omterm{def:g11:func:function}{functie} op elk \omterm{def:g10:numbers:interval}{interval}
$\intco{A}{+\infty}$ af tussen intervallen van strikte stijging; ze is
\omterm{def:g12:seq:monotonic}{stijgend} maar niet strikt, en vooral: haar groei is niet nodig, want de
\omterm{def:g10:algebra:equation}{vergelijking} volstaat.

\emph{2.} Deel voor $x > 1$ door $x$:
\[
g(x) = \frac{1 + \frac{\sin x}{x}}{1 - \frac{\sin x}{x}},
\]
en $\frac{\sin x}{x} \to 0$ volgens de insluitstelling
($\abs{\sin x} \leq 1$). Bijgevolg is $g(x) \to \frac{1+0}{1-0} = 1$.
\end{solution}

\begin{solution}{exo:g12:limcont:6}
$f(x) = x^3 - 3x + 1$ is \omterm{def:g12:limcont:continuity}{continu}, met $f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$: $f$
stijgt op $\intoc{-\infty}{-1}$, daalt op $\intcc{-1}{1}$ en stijgt op
$\intco{1}{+\infty}$, met lokaal \omterm{def:g10:functions:extrema}{maximum} $f(-1) = 3 > 0$ en lokaal \omterm{def:g10:functions:extrema}{minimum}
$f(1) = -1 < 0$, en $\lim_{-\infty} f = -\infty$,
$\lim_{+\infty} f = +\infty$.

Op elk van de drie intervallen van monotonie is $f$ \omterm{def:g12:limcont:continuity}{continu} en strikt
monotoon, en ligt $0$ strikt tussen de waarden of limieten in de uiteinden,
zodat de bijectiestelling per \omterm{def:g10:numbers:interval}{interval} precies één nulpunt geeft: in totaal
drie nulpunten. Tekencontroles: $f(-2) = -1 < 0 < f(-1) = 3$, dus één nulpunt
in $\intoo{-2}{-1}$; $f(0) = 1 > 0 > f(1) = -1$, één nulpunt in
$\intoo{0}{1}$; $f(1) = -1 < 0 < f(2) = 3$, één nulpunt in $\intoo{1}{2}$.
\end{solution}

\begin{solution}{exo:g12:limcont:7}
Zij $g(x) = f(x) - x$, \omterm{def:g12:limcont:continuity}{continu} op $\intcc{0}{1}$. Omdat
$f(0) \in \intcc{0}{1}$, is $g(0) = f(0) \geq 0$; omdat
$f(1) \in \intcc{0}{1}$, is $g(1) = f(1) - 1 \leq 0$. Volgens de
tussenwaardestelling verdwijnt $g$ in een zekere $c \in \intcc{0}{1}$, dus is
$f(c) = c$.
\end{solution}

\begin{solution}{exo:g12:limcont:8}
Zij $u(t)$ en $d(t)$ de posities (gemeten als afstand vanaf de voet van het
pad) op tijdstip $t \in \intcc{8}{20}$ tijdens de klim en tijdens de afdaling.
Beide zijn \omterm{def:g12:limcont:continuity}{continu}, met $u(8) = 0$, $u(20) = L$ (de lengte van het pad),
$d(8) = L$ en $d(20) = 0$. De \omterm{def:g11:func:function}{functie} $g = u - d$ is \omterm{def:g12:limcont:continuity}{continu} met
$g(8) = -L \leq 0$ en $g(20) = L \geq 0$, dus is $g(c) = 0$ voor een zeker
tijdstip $c$ volgens de tussenwaardestelling: op tijdstip $c$ vallen de
posities van beide dagen samen.
\end{solution}

\begin{solution}{exo:g12:limcont:9}
\emph{1.} $f'(x) = 5x^4 + 3x^2 + 1 > 0$, dus is $f$ strikt \omterm{def:g12:seq:monotonic}{stijgend} op $\R$;
met $\lim_{\pm\infty} f = \pm\infty$ en de \omterm{def:g12:limcont:continuity}{continuïteit} geeft de
bijectiestelling een uniek reëel nulpunt $\alpha$. Omdat $f(0) = -1 < 0$ en
$f(1) = 2 > 0$, is $\alpha \in \intoo{0}{1}$.

\emph{2.} Houd een \omterm{def:g10:numbers:interval}{interval} $\intcc{a_n}{b_n}$ bij met
$f(a_n) < 0 < f(b_n)$, vertrekkend van $\intcc{0}{1}$; evalueer bij elke stap
$f$ in het \omterm{prop:g10:coordgeom:midpoint}{midden} $m$ en houd de helft over waarop de tekenwissel blijft
bestaan. Na $n$ stappen heeft het \omterm{def:g10:numbers:interval}{interval} lengte $2^{-n}$, zodat het \omterm{prop:g10:coordgeom:midpoint}{midden}
$\alpha$ benadert op $2^{-(n+1)}$ na. We hebben $2^{-(n+1)} \leq 10^{-6}$
nodig, dus $n + 1 \geq \frac{6}{\log_{10} 2} \approx 19.93$: $n = 20$
iteraties volstaan.
\end{solution}

\begin{solution}{pb:g12:limcont:1}
\textbf{1.} Overheersende termen: $\frac{3x^2}{x^2} \to 3$;
$\frac{2x}{x^2} = \frac2x \to 0$;
$x^3 - x^2 = x^3\left(1 - \frac1x\right) \to +\infty$.

\textbf{2.} $\lim_{x \to \pm\infty} f = 3$: \omterm{def:g12:limcont:asymptote}{horizontale asymptoot} $y = 3$;
$\lim_{x \to (-2)^\pm} f = \mp\infty$: \omterm{def:g12:limcont:asymptote}{verticale asymptoot} $x = -2$.

\textbf{3.} $\frac{(x-3)(x+3)}{x - 3} = x + 3 \to 6$. En
$\frac{\sqrt{x+1} - 1}{x} = \frac{x}{x(\sqrt{x+1} + 1)} =
\frac{1}{\sqrt{x+1} + 1} \to \frac12$.

\textbf{4.} $\abs{x \sin\frac1x} \leq \abs x \to 0$, dus is de limiet $0$. En
$\abs{\frac{\sin x}{x}} \leq \frac1x \to 0$: limiet $0$.

\textbf{5.} $\frac{\sqrt{4x^2 + x}}{x} = \sqrt{4 + \frac1x} \to \sqrt4 = 2$
(\omterm{def:g12:limcont:continuity}{continuïteit} van de vierkantswortel, samenstelling).

\textbf{6.} $f(x) = x^3 + x$ is \omterm{def:g12:limcont:continuity}{continu} met $f(1) = 2 < 5 < 10 = f(2)$:
volgens de tussenwaardestelling is $f(c) = 5$ voor een zekere
$c \in \intoo{1}{2}$.

\textbf{7.} $f'(x) = 3x^2 + 1 > 0$: strikt \omterm{def:g12:seq:monotonic}{stijgend} op $\R$, \omterm{def:g12:limcont:continuity}{continu}, met
limieten $\mp\infty$: volgens de bijectiestelling wordt elke waarde --- en in
het bijzonder $5$ --- precies één keer aangenomen.

\textbf{8.} $p(x) = x^3\left(1 + \frac ax + \frac{b}{x^2} +
\frac{c}{x^3}\right)$: voor $x \to +\infty$ is $p \to +\infty$; voor
$x \to -\infty$ is $p \to -\infty$. Dus geldt op een reusachtig \omterm{def:g10:numbers:interval}{interval}
$\intcc{-M}{M}$ dat $p(-M) < 0 < p(M)$: de tussenwaardestelling levert een
\omterm{def:g11:quad:discriminant}{wortel}. Hetzelfde argument met de overheersende term werkt woordelijk voor
elke oneven graad --- oneven machten houden in de twee oneindigheden
tegengestelde tekens.

\textbf{9.} $n$ keer halveren laat een \omterm{def:g10:numbers:interval}{interval} van lengte $2^{-n}$ over; en
$2^{-20} \approx 9.5 \times 10^{-7} < 10^{-6}$: twintig stappen. Newton
verdubbelt per stap het aantal juiste cijfers: vanaf één juist cijfer
volstaan ongeveer $3$ à $4$ stappen voor zes --- de schildpad en de windhond
van het wortelzoeken.

\textbf{10.} $g$ is niet \omterm{def:g12:limcont:continuity}{continu} op $\intcc{-1}{1}$: ze ontploft in $0$, en
dat punt ligt in het \omterm{def:g10:numbers:interval}{interval}. Het besluit van de tussenwaardestelling
verdampt samen met haar hypothese --- ga de \omterm{def:g12:limcont:continuity}{continuïteit} altijd \emph{op het
hele \omterm{def:g10:numbers:interval}{interval}} na, niet alleen in de uiteinden.

\textbf{11.} $g(x) = f(x) - x$ is \omterm{def:g12:limcont:continuity}{continu}; $g(0) = f(0) \geq 0$ (de waarden
liggen in $\intcc{0}{1}$) en $g(1) = f(1) - 1 \leq 0$. Volgens de
tussenwaardestelling is $g(c) = 0$ voor een zekere $c$: $f(c) = c$.

\textbf{12.} Idealiseer het land als het lijnstuk $\intcc{0}{1}$: $f$ stuurt
elke plek naar de plek die er op de verfrommelde kaart recht onder wordt
getoond; verfrommelen houdt de kaart boven het land, dus belandt $f$ in
$\intcc{0}{1}$, en ze is \omterm{def:g12:limcont:continuity}{continu} (papier scheurt niet). Het vaste punt $c$ is
de plek die precies boven haar eigen \omterm{def:g10:functions:function}{beeld} ligt --- de stelling van Brouwer
behandelt de eerlijke tweedimensionale kaart.

\textbf{13.} $g(0) + g(\pi) = \left(T(0) - T(\pi)\right) +
\left(T(\pi) - T(2\pi)\right) = T(0) - T(2\pi) = 0$. Dus zijn $g(0)$ en
$g(\pi)$ tegengesteld (of allebei nul): hoe dan ook verdwijnt de continue $g$
in een zekere $t^* \in \intcc{0}{\pi}$, zodat $T(t^*) = T(t^* + \pi)$ ---
ergens op de evenaar zijn twee antipodale punten het over de temperatuur
eens, op elk ogenblik.

\textbf{14.} De tafel een kwartslag draaien verwisselt de rollen van de twee
diagonalen: het paar poten dat op de grond drukte en het paar waarvan er één
zweefde wisselen van situatie, zodat de zweefhoogte --- met haar teken geteld
als ``opening van de aangewezen diagonaal'' --- in $\theta$ en
$\theta + \frac\pi2$ tegengestelde tekens aanneemt. Continue grond maakt $h$
\omterm{def:g12:limcont:continuity}{continu}, en de tussenwaardestelling levert een hoek $\theta^*$ met
$h(\theta^*) = 0$: vier poten op de grond. (Bij een rechthoekige tafel of bij
grond met treden heb je die \omterm{def:g10:proba:distribution}{kans} niet.)

\textbf{15.} Het sjabloon: bouw het \emph{\omterm{def:g11:seq:arithmetic}{verschil}} van de twee grootheden
die je wilt laten samenvallen ($f(x) - x$; $T(t) - T(t + \pi)$; de positie
omhoog min de positie omlaag bij de wandelaar; de getekende opening bij de
tafel); ga na dat het waarden van beide tekens aanneemt (vaak via een
symmetrie van de uiteinden); en roep de tussenwaardestelling in op een
continue \omterm{def:g11:func:function}{functie}. Het samenvallen is afgedwongen.

\textbf{16.} Open vragen: \emph{hoeveel} oplossingen (beantwoord door de
strikte monotonie en de bijectiestelling, vraag 7) en \emph{waar} ze liggen
(beantwoord door \omterm{thm:g12:limcont:ivt}{bisectie} of Newton, vraag 9). De tussenwaardestelling is een
zuiver orakel van bestaan.

\textbf{17.} $3 - \frac{7}{x + 2}$: \omterm{def:g12:limcont:asymptote}{verticale asymptoot} $x = -2$, horizontale
$y = 3$; aan elke kant van $-2$ is de \omterm{def:g11:func:function}{functie} \omterm{def:g12:seq:monotonic}{stijgend} ($-\frac{7}{x+2}$
stijgt waar ze gedefinieerd is); de kromme is de referentiehyperbool
$-\frac7x$, $2$ naar links en $3$ omhoog geschoven --- twee takken die zich
tegen het kruis van de asymptoten aan vlijen.

\textbf{18.} Voor $x \neq 1$ is $f(x) = x + 1$, dus $\lim_{x \to 1} f = 2$:
met $f(1) = 2$ herstel je de \omterm{def:g12:limcont:continuity}{continuïteit}. De discontinuïteit was een gaatje,
geen sprong --- vul het ene ontbrekende punt op en de kromme geneest:
\emph{ophefbaar}.

\textbf{19.} \omterm{def:g12:limcont:continuity}{Continu} op elk \omterm{def:g10:numbers:interval}{interval} tussen opeenvolgende \omterm{def:g10:numbers:sets}{gehele getallen}; in
elk \omterm{def:g10:numbers:sets}{geheel getal} $n$ springt ze van $n - 1$ naar $n$. Met $a = 0.9$ en
$b = 1.1$ is $\lfloor a \rfloor = 0 < \frac12 < 1 = \lfloor b \rfloor$, en
toch heeft $\lfloor x \rfloor = \frac12$ geen oplossing --- de sprong
teleporteert over $\frac12$ heen, en precies dat verbiedt de
tussenwaardestelling aan continue \omterm{def:g11:func:function}{functies}.

\textbf{20.} Plaatselijk gezicht: waarde $=$ limiet in elk punt --- het
ophefbare gaatje (vraag 18) schond het geneeslijk, de sprongen van de
entierfunctie (vraag 19) ongeneeslijk. Globaal gezicht: op een \omterm{def:g10:numbers:interval}{interval} worden
alle tussenliggende waarden bezocht --- de certificaten voor \omterm{def:g11:quad:discriminant}{wortels} (6--8),
het vaste punt, de evenaar, de tafel en de wandelaar wonen hier allemaal. De
officiële naam van de leuze: de \emph{tussenwaardestelling}.
\end{solution}
