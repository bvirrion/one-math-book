\section*{Hoofdstuk \ref{ch:g12:exp} --- Exponentiële functie en logaritme}

\begin{solution}{exo:g12:exp:1}
$\dfrac{\eu^{3x}\,\eu^{-x+1}}{\eu^{x}} = \eu^{3x - x + 1 - x} = \eu^{x+1}$.

$\ln\!\left(\dfrac{\eu^2\sqrt{\eu}}{\eu^{-3}}\right)
= 2 + \tfrac12 + 3 = \tfrac{11}{2}$.
\end{solution}

\begin{solution}{exo:g12:exp:2}
\emph{(a)} Stel $X = \eu^x > 0$: $X^2 - 3X + 2 = 0$ geeft $X = 1$ of
$X = 2$, allebei positief, dus $x = 0$ of $x = \ln 2$.

\emph{(b)} \omterm{def:g11:func:function}{Domein}: $x - 1 > 0$ en $x + 2 > 0$, dus $x > 1$. De \omterm{def:g10:algebra:equation}{vergelijking}
wordt $\ln\bigl((x-1)(x+2)\bigr) = \ln 4$, dus $(x-1)(x+2) = 4$, dus
$x^2 + x - 6 = 0$, waaruit $x = 2$ of $x = -3$. Alleen $x = 2$ ligt in het
\omterm{def:g11:func:function}{domein}: $S = \{2\}$.
\end{solution}

\begin{solution}{exo:g12:exp:3}
Deel door $\eu^x$: $\dfrac{1 - x^2\eu^{-x}}{1 + \eu^{-x}} \to
\dfrac{1-0}{1+0} = 1$, met $x^2 \eu^{-x} \to 0$
(\cref{thm:g12:exp:growth}).

$\dfrac{\ln x}{\sqrt x} \to 0$ volgens \cref{prop:g12:exp:lnanalytic} (geval
$n = 2$).

$x^2 \ln x = x \cdot (x \ln x) \to 0 \times 0 = 0$.

$x - \ln x = x\left(1 - \frac{\ln x}{x}\right) \to +\infty$, want
$\frac{\ln x}{x} \to 0$.
\end{solution}

\begin{solution}{exo:g12:exp:4}
$f'(x) = \eu^{-x} - x\eu^{-x} = (1 - x)\eu^{-x}$, met het teken van
$1 - x$: $f$ stijgt op $\intoc{-\infty}{1}$ en daalt op
$\intco{1}{+\infty}$, met globaal \omterm{def:g10:functions:extrema}{maximum} $f(1) = \eu^{-1}$.

Limieten: voor $x \to +\infty$ is $f(x) = \frac{x}{\eu^x} \to 0$
(\cref{thm:g12:exp:growth}); voor $x \to -\infty$ is $x \to -\infty$ en
$\eu^{-x} \to +\infty$, dus $f(x) \to -\infty$.

$f''(x) = -\eu^{-x} - (1-x)\eu^{-x} = (x - 2)\eu^{-x}$, dat in $x = 2$ van
teken verandert: \omterm{def:g12:deriv:inflection}{buigpunt} in $\bigl(2,\, 2\eu^{-2}\bigr)$.
\end{solution}

\begin{solution}{exo:g12:exp:5}
Zij $g(x) = x - \ln(1+x)$ op $\intoo{-1}{+\infty}$. Dan is
$g'(x) = 1 - \frac{1}{1+x} = \frac{x}{1+x}$, negatief op $\intoo{-1}{0}$ en
positief op $\intoo{0}{+\infty}$: $g$ bereikt haar \omterm{def:g10:functions:extrema}{minimum} $g(0) = 0$, dus is
$g \geq 0$ met gelijkheid alleen in $0$. Bijgevolg is $\ln(1+x) \leq x$.

Pas dat toe met $x = \frac1n$: $n\ln\left(1 + \frac1n\right) \leq 1$, dus
$\left(1+\frac1n\right)^n = \eu^{\,n\ln(1+1/n)} \leq \eu$. Pas het toe met
$x = -\frac{1}{n+1} > -1$: $\ln\left(1 - \frac{1}{n+1}\right) \leq
-\frac{1}{n+1}$, dus $-(n+1)\ln\left(1 - \frac{1}{n+1}\right) \geq 1$ en
$\left(1 - \frac{1}{n+1}\right)^{-(n+1)} \geq \eu$.
\end{solution}

\begin{solution}{exo:g12:exp:6}
\emph{1.} Elk jaar vermenigvuldigt het kapitaal met $1.03$:
$C_n = C_0 (1.03)^n$.

\emph{2.} $C_n \geq 2C_0 \iff (1.03)^n \geq 2 \iff n \ln 1.03 \geq \ln 2
\iff n \geq \frac{\ln 2}{\ln 1.03} \approx 23.45$: het kapitaal verdubbelt na
$24$ jaar.

\emph{3.} $\frac{70}{3} \approx 23.3$, dicht bij de exacte
$\frac{\ln 2}{\ln 1.03}$. De regel werkt omdat $\ln(1 + r) \approx r$ voor
kleine $r$, zodat $\frac{\ln 2}{\ln(1+r)} \approx \frac{0.693}{r} \approx
\frac{70}{100r}$.
\end{solution}

\begin{solution}{exo:g12:exp:7}
$\eu^{2x} > \eu^{x+1} \iff 2x > x + 1 \iff x > 1$ (de \omterm{thm:g12:exp:existence}{exponentiële functie} is
strikt \omterm{def:g12:seq:monotonic}{stijgend}): $S = \intoo{1}{+\infty}$.

\omterm{def:g11:func:function}{Domein} van de tweede ongelijkheid: $x^2 - 1 > 0$ en $x + 5 > 0$, dus
$x \in \intoo{-5}{-1} \cup \intoo{1}{+\infty}$. Op dat \omterm{def:g11:func:function}{domein} geldt, omdat
$\ln$ strikt \omterm{def:g12:seq:monotonic}{stijgend} is,
\[
\ln(x^2-1) \leq \ln(x+5) \iff x^2 - 1 \leq x + 5 \iff x^2 - x - 6 \leq 0
\iff x \in \intcc{-2}{3}.
\]
Doorsnijden met het \omterm{def:g11:func:function}{domein}: $S = \intco{-2}{-1} \cup \intoc{1}{3}$.
\end{solution}

\begin{solution}{exo:g12:exp:8}
\emph{1.} $f'(x) = \dfrac{\eu^x x - \eu^x}{x^2} = \dfrac{(x-1)\eu^x}{x^2}$,
negatief op $\intoo{0}{1}$ en positief op $\intoo{1}{+\infty}$: \omterm{def:g10:functions:extrema}{minimum}
$f(1) = \eu$. Limieten: $f \to +\infty$ in $0^+$ (teller $\to 1$, noemer
$\to 0^+$) en in $+\infty$ (\cref{thm:g12:exp:growth}).

\emph{2.} Voor $x > 0$ is $\eu^x = kx \iff f(x) = k$. Uit de \omterm{def:g10:functions:table}{verlooptabel}
($+\infty \searrow \eu \nearrow +\infty$, \omterm{def:g12:limcont:continuity}{continu} en strikt monotoon aan elke
kant): geen oplossing voor $k < \eu$; precies één ($x = 1$) voor $k = \eu$;
precies twee voor $k > \eu$ (één in $\intoo{0}{1}$ en één in
$\intoo{1}{+\infty}$, volgens de bijectiestelling op elk \omterm{def:g10:numbers:interval}{interval}).
\end{solution}

\begin{solution}{exo:g12:exp:9}
\emph{1.} Rechtstreeks uit de eerste ongelijkheid van
\cref{exo:g12:exp:5}: $u_n = \eu^{\,n\ln(1+1/n)} \leq \eu^1 = \eu$.

\emph{2.} Schrijf
\[
\ln u_n = n \ln\!\left(1 + \frac1n\right)
= \frac{\ln\!\left(1 + \frac1n\right) - \ln 1}{\frac1n},
\]
een differentiequotiënt van $\ln$ in het punt $1$ met stap
$h = \frac1n \to 0$. Omdat $\ln$ in $1$ \omterm{def:g12:deriv:derivative}{differentieerbaar} is met \omterm{def:g12:deriv:derivative}{afgeleide}
$1$, is $\ln u_n \to 1$. Door de \omterm{def:g12:limcont:continuity}{continuïteit} van $\exp$
(\cref{prop:g12:limcont:seqcont}) is $u_n = \eu^{\ln u_n} \to \eu^1 = \eu$.
\end{solution}

\begin{solution}{pb:g12:exp:1}
\textbf{1.} $x = \frac{\ln 5}{2}$. \quad $3x - 1 = \eu^2$:
$x = \frac{\eu^2 + 1}{3}$. \quad Met $u = \eu^x$:
$u^2 - 3u + 2 = (u - 1)(u - 2) = 0$: $x = 0$ of $x = \ln 2$.

\textbf{2.} $\frac{3\ln 2}{\ln 2} = 3$; \quad $3 + \frac12 = \frac72$; \quad
$\frac52$.

\textbf{3.} $f'(x) = \eu^x - 1$: negatief vóór $0$, positief erna: \omterm{def:g10:functions:extrema}{minimum}
$f(0) = 0$, dus $f \geq 0$ overal: $\eu^x \geq 1 + x$. Invullen van
$x = \ln(1 + u)$ geeft $1 + u \geq 1 + \ln(1 + u)$, dus
$\ln(1 + u) \leq u$.

\textbf{4.} $x^2\eu^{-x} = \frac{x^2}{\eu^x} \to 0$ en
$\frac{\ln x}{x} \to 0$: exponentiëlen verpletteren machten, machten
verpletteren \omterm{def:g12:exp:ln}{logaritmen}. En
$\frac{\eu^x - 1}{x} = \frac{\eu^x - \eu^0}{x - 0} \to \exp'(0) = 1$.

\textbf{5.} $f'(x) = \ln x + 1$: nul in $\eu^{-1}$, negatief ervóór en
positief erna: \omterm{def:g10:functions:extrema}{minimum} $f\!\left(\frac1\eu\right) = -\frac1\eu$; en
$x \ln x \to 0$ in $0^+$: de kromme vertrekt uit de oorsprong, duikt naar
$-\frac1\eu$, en klimt daarna weg.

\textbf{6.} $n = \frac{\ln 2}{\ln 1.03} \approx 23.4$ jaar.

\textbf{7.} Exact: $23.4$, $11.9$ en $8.0$ jaar; met de regel van 72: $24$,
$12$ en $8$. Omdat $\ln(1+r) \approx r$, is
$\frac{\ln 2}{\ln(1+r)} \approx \frac{0.693}{r}$, dus
$\frac{69.3}{\text{rentevoet in }\%}$; bankiers ronden op tot $72$ omdat dat
prachtig deelbaar is door $2, 3, 4, 6, 8, 9$ en $12$ --- hoofdrekenen wint
het van één decimaal nauwkeurigheid.

\textbf{8.} $2^{t/20} = \eu^{t \ln 2 / 20}$: $\lambda = \frac{\ln 2}{20}$ per
minuut. Na $24 \times 60 = 1440$ minuten:
$2^{72} \approx 4.7 \times 10^{21}$ bacteriën --- meer dan er zandkorrels op
Aarde liggen, uit één cel in één dag. Het model is alleen eerlijk zolang er
voedsel en ruimte zijn: echte groei buigt af naar de logistische S-kromme
(de kromme van de epidemioloog uit \cref{pb:g12:deriv:1}).

\textbf{9.} Om 23 uur (na $8$ uur): $100 \times 2^{-8/5} \approx 33$ mg.
Onder $10$ mg: uit $2^{-t/5} < 0.1$ volgt
$t > 5\,\frac{\ln 10}{\ln 2} \approx 16.6$ u: rond 7.40 uur de volgende
ochtend --- de espresso van drie uur heeft een lange staart.

\textbf{10.} Jaarlijks: $2$. Maandelijks:
$\left(1 + \frac{1}{12}\right)^{12} \approx 2.613$. Dagelijks:
$\approx 2.715$. Het plafond: $\eu = 2.71828\ldots$
(\cref{exo:g12:exp:9}) --- bij continue kapitalisatie \omterm{def:g12:seq:limit}{convergeert} de
hebzucht.

\textbf{11.} Groeien de gevallen exponentieel, $c(t) = c_0 \eu^{\lambda t}$,
dan is $\ln c(t) = \ln c_0 + \lambda t$: een rechte met
\omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $\lambda$ --- de groeisnelheid. Een recht stuk op de
logaritmische \omterm{def:g10:functions:graph}{grafiek} \emph{is} de exponentiële fase, en haar steilheid is
het tempo van de epidemie.

\textbf{12.} $120 - 60 = 60$ dB betekent $10\log_{10}(I_2/I_1) = 60$: het
concert is $10^6$ --- een miljoen keer --- intenser dan het gesprek.

\textbf{13.} Amplitude: $10^{7-5} = 100$. Energie: $100^{3/2} = 1000$: twee
magnitudepunten verbergen drie grootteordes aan energie.

\textbf{14.} $10^{6.5 - 2} = 10^{4.5} \approx 32\,000$: citroensap is
dertigduizend keer zuurder dan melk --- de pH perst de kloven van de
scheikunde samen tot een zakformaat.

\textbf{15.} $\log_2\!\frac{4186}{27.5} = \frac{\ln(4186/27.5)}{\ln 2}
\approx 7.25$: een piano overspant net iets meer dan zeven octaven. Zintuigen
reageren op \emph{verhoudingen} van prikkels --- de intensiteit verdubbelen
voelt als één stap, ongeacht het beginniveau --- zodat de waarneming op een
logaritmische schaal gebouwd is, en de eenheden die we ervoor bedachten
evenzeer.

\textbf{16.} Het latje voor $a$ kop aan staart tegen de stand voor $b$
leggen telt de lengten $\ln a + \ln b$ op, en de ijking op die totale lengte
leest $\eu^{\ln a + \ln b} = ab$: de rekenliniaal berekent producten door
\omterm{def:g12:exp:ln}{logaritmen} op te tellen --- $\ln(ab) = \ln a + \ln b$
(\cref{prop:g12:exp:lnalg}), in palmhout gekerfd.

\textbf{17.} Het \omterm{def:g10:functions:extrema}{minimum} van $\frac{\eu^x}{x}$ op $\intoo{0}{+\infty}$ is
$\eu$, in $x = 1$: geen oplossing voor $k < \eu$, precies één voor $k = \eu$,
en twee voor $k > \eu$. Controle met de \omterm{def:g12:deriv:tangent}{raaklijn}: in $a = 1$ is de \omterm{def:g12:deriv:tangent}{raaklijn}
aan $\eu^x$ gelijk aan $y = \eu + \eu(x - 1) = \eu x$: ze gaat door de
oorsprong --- de drempelrechte, die de kromme in $(1, \eu)$ scheert.

\textbf{18.} $n \ln\!\left(1 - \frac1n\right) = \frac{\ln(1 - 1/n)}{-1/n}
\times (-1) \to -1$, dus $\left(1 - \frac1n\right)^n \to \eu^{-1} \approx
0.368$: de $0.999^{1000} \approx 0.368$ uit \cref{pb:g11:binom:1},
verklaard --- de constante van het net-niet bij de loterij is $\frac1\eu$.

\textbf{19.} Beide zijn de grensvorm van ``veel onafhankelijke
gebeurtenissen, elk op zich onwaarschijnlijk'': de \omterm{def:g10:proba:distribution}{kans} dat \emph{geen} van
$n$ kansen van grootte $\frac1n$ afgaat, streeft naar $\eu^{-1}$, en de
secretaresseregel stemt haar verwerpingsvenster zo af dat het succes zich
rond diezelfde constante concentreert. $\frac1\eu \approx 0.368$.

\textbf{20.} Het plafond van de kapitalisatie; het unieke grondtal waarvan de
\omterm{def:g12:deriv:tangent}{raaklijn} in $0$ \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $1$ heeft (en daarom houdt de analyse
zoveel van $\eu$); de groei die elke macht voorbijsteekt; de constante waar
het net-niet en het optimale stoppen zich vastzetten; en haar inverse $\ln$,
die vermenigvuldigen tot optellen maakt en de wildste bereiken van de wereld
op een meetlat vouwt.
\end{solution}
