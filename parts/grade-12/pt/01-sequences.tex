\chapter{Sequências}\label{ch:g12:seq}

Uma \omterm{def:g12:seq:sequence}{sequência} é uma lista de \omterm{def:g10:numbers:sets}{números reais} indexada pelos naturais. As
\omterm{def:g12:seq:sequence}{sequências} modelam evoluções discretas --- populações contadas ano a ano,
saldos de uma conta bancária, \omterm{def:g10:numbers:approx}{aproximações} sucessivas de um número --- e
seus limites são o primeiro encontro sério com o infinito. Este capítulo
monta o vocabulário, o princípio da indução e os teoremas fundamentais de
convergência.

\section{Raciocínio por indução}

\begin{theorem}[Princípio da indução]\label{thm:g12:seq:induction}
Seja $P(n)$ uma afirmação que depende de um inteiro $n$, e seja
$n_0 \in \N$. Se
\begin{enumerate}
  \item \emph{(caso base)} $P(n_0)$ é verdadeira e
  \item \emph{(passo indutivo)} para todo $n \geq n_0$, $P(n)$ implica
        $P(n+1)$,
\end{enumerate}
então $P(n)$ é verdadeira para todo $n \geq n_0$.
\end{theorem}

\begin{proof}
Suponha, por absurdo, que o conjunto $A$ dos inteiros $n \geq n_0$ para os
quais $P(n)$ é falsa seja não vazio. Então $A$ tem um menor elemento
$m$.\footnote{Todo subconjunto não vazio de $\N$ tem um menor elemento;
essa propriedade de $\N$ é tomada como axioma.} Como $P(n_0)$ é
verdadeira, $m > n_0$, logo $m - 1 \geq n_0$ e $m-1 \notin A$, isto é,
$P(m-1)$ é verdadeira. O passo indutivo aplicado a $n = m-1$ mostra então
que $P(m)$ é verdadeira, contradizendo $m \in A$.
\end{proof}

\begin{example}\label{ex:g12:seq:bernoulli}
Vamos demonstrar a \emph{desigualdade de Bernoulli}: para todo real
$a > 0$ e todo $n \in \N$,
\[
(1+a)^n \geq 1 + na.
\]
\emph{Caso base.} Para $n = 0$, os dois lados valem $1$.
\emph{Passo indutivo.} Suponha $(1+a)^n \geq 1+na$ para algum
$n \in \N$. Como $1 + a > 0$, multiplicar os dois lados por $1+a$ preserva
a desigualdade:
\[
(1+a)^{n+1} \geq (1+na)(1+a) = 1 + (n+1)a + na^2 \geq 1 + (n+1)a .
\]
Por indução, a desigualdade vale para todo $n \in \N$.
\end{example}

\begin{method}[Escrever uma demonstração por indução]\label{met:g12:seq:induction}
Deixe sempre a afirmação $P(n)$ explícita antes de começar. Uma
demonstração completa tem três partes visíveis: o caso base, o passo
indutivo (``suponha $P(n)$; demonstramos $P(n+1)$'') e a conclusão
invocando o princípio da indução. O erro mais comum é demonstrar o passo
indutivo sem nunca usar a hipótese $P(n)$: se isso acontecer, ou a
demonstração está errada, ou a indução não era necessária.
\end{method}

\section{Vocabulário das sequências}

\begin{definition}[Sequência]\label{def:g12:seq:sequence}
Uma \emph{sequência}\index{sequência} é uma \omterm{def:g11:func:function}{função} $u \colon \N \to \R$
(ou de $\{n \in \N : n \geq n_0\}$ em $\R$). A \omterm{def:g10:functions:function}{imagem} de $n$ escreve-se
$u_n$, e a própria sequência, $(u_n)_{n\in\N}$ ou simplesmente $(u_n)$.
\end{definition}

Uma \omterm{def:g12:seq:sequence}{sequência} pode ser definida de forma \emph{explícita}, por uma fórmula
$u_n = f(n)$, ou por \emph{recorrência}, pelo seu primeiro termo e uma
relação $u_{n+1} = f(u_n)$.

\begin{definition}[Monotonicidade]\label{def:g12:seq:monotonic}
Uma \omterm{def:g12:seq:sequence}{sequência} $(u_n)$ é \emph{crescente}\index{sequência!crescente} se
$u_{n+1} \geq u_n$ para todo $n$, \emph{\omterm{def:g10:functions:variations}{decrescente}} se
$u_{n+1} \leq u_n$ para todo $n$, e \emph{monótona} se for crescente ou
\omterm{def:g10:functions:variations}{decrescente}. Ela é \emph{estritamente} crescente (resp.\ \omterm{def:g10:functions:variations}{decrescente})
quando as desigualdades são estritas.
\end{definition}

\begin{method}[Estudar a monotonicidade de uma sequência]\label{met:g12:seq:monotonicity}
Três técnicas padrão:
\begin{enumerate}
  \item estudar o sinal de $u_{n+1} - u_n$;
  \item se todos os termos são positivos, comparar
        $\dfrac{u_{n+1}}{u_n}$ com $1$;
  \item se $u_n = f(n)$ com $f$ definida em $\intco{0}{+\infty}$, usar a
        variação de $f$.
\end{enumerate}
\end{method}

\begin{definition}[Sequência limitada]\label{def:g12:seq:bounded}
Uma \omterm{def:g12:seq:sequence}{sequência} $(u_n)$ é \emph{limitada
superiormente}\index{sequência!limitada} se existe $M \in \R$ com
$u_n \leq M$ para todo $n$; \emph{limitada inferiormente} se existe
$m \in \R$ com $u_n \geq m$ para todo $n$; e \emph{limitada} se as duas
coisas valem.
\end{definition}

\subsection{Progressões aritméticas e geométricas}

\begin{definition}[Progressões aritméticas e geométricas]\label{def:g12:seq:arith-geom}
Uma \omterm{def:g12:seq:sequence}{sequência} $(u_n)$ é uma \emph{progressão
aritmética}\index{sequência!aritmética} de razão $r$ se
$u_{n+1} = u_n + r$ para todo $n$, e uma \emph{progressão
geométrica}\index{sequência!geométrica} de razão $q$ se $u_{n+1} = q\,u_n$
para todo $n$.
\end{definition}

\begin{proposition}[Forma explícita e somas]\label{prop:g12:seq:arith-geom}
Seja $n \in \N$.
\begin{enumerate}
  \item Se $(u_n)$ é uma \omterm{def:g12:seq:arith-geom}{progressão aritmética} de razão $r$, então
        $u_n = u_0 + nr$ e
        \[
        u_0 + u_1 + \dots + u_n = (n+1)\,\frac{u_0 + u_n}{2}.
        \]
  \item Se $(u_n)$ é uma \omterm{def:g12:seq:arith-geom}{progressão geométrica} de razão $q \neq 1$, então
        $u_n = u_0\, q^n$ e
        \[
        u_0 + u_1 + \dots + u_n = u_0\,\frac{1 - q^{n+1}}{1 - q}.
        \]
\end{enumerate}
\end{proposition}

\begin{proof}
As formas explícitas seguem de induções imediatas. Para a soma
aritmética, escreva $S = u_0 + \dots + u_n$ e some a mesma soma escrita em
ordem inversa: cada uma das $n+1$ somas de coluna vale $u_0 + u_n$, logo
$2S = (n+1)(u_0+u_n)$. Para a soma geométrica, calcule $S - qS$: todos os
termos se cancelam aos pares, exceto o primeiro e o último, de modo que
$(1-q)S = u_0(1 - q^{n+1})$.
\end{proof}

\section{Limite de uma sequência}

\begin{definition}[Sequência convergente]\label{def:g12:seq:limit}
Uma \omterm{def:g12:seq:sequence}{sequência} $(u_n)$ \emph{converge}\index{limite!de uma sequência} para
o \omterm{def:g10:numbers:sets}{número real} $\ell$ se todo \omterm{def:g10:numbers:interval}{intervalo} aberto que contém $\ell$ contém
todos os termos $u_n$ a partir de um certo índice. Escrevemos então
$\lim\limits_{n\to+\infty} u_n = \ell$.

Equivalentemente: para todo $\varepsilon > 0$, existe $N \in \N$ tal que,
para todo $n \geq N$, $\abs{u_n - \ell} \leq \varepsilon$.
\end{definition}

\begin{definition}[Divergência para o infinito]\label{def:g12:seq:infinity}
A \omterm{def:g12:seq:sequence}{sequência} $(u_n)$ \emph{tende a $+\infty$} se, para todo $A \in \R$,
existe $N \in \N$ tal que $u_n \geq A$ para todo $n \geq N$. Escrevemos
$\lim\limits_{n\to+\infty} u_n = +\infty$; a definição de
$\lim u_n = -\infty$ é análoga. Uma \omterm{def:g12:seq:sequence}{sequência} que não \omterm{def:g12:seq:limit}{converge} é dita
\emph{divergente}.
\end{definition}

\begin{remark}
Uma \omterm{def:g12:seq:sequence}{sequência} pode divergir sem tender a $\pm\infty$: a \omterm{def:g12:seq:sequence}{sequência}
$u_n = (-1)^n$ só assume os valores $1$ e $-1$ e não tem limite.
\end{remark}

\begin{proposition}[Unicidade do limite]\label{prop:g12:seq:uniqueness}
Se $(u_n)$ \omterm{def:g12:seq:limit}{converge}, seu limite é único.
\end{proposition}

\begin{proof}
Suponha $u_n \to \ell$ e $u_n \to \ell'$ com $\ell \neq \ell'$, digamos
$\ell < \ell'$. Ponha $\varepsilon = \frac{\ell' - \ell}{3} > 0$. A partir
de um certo índice, $\abs{u_n - \ell} \leq \varepsilon$ e
$\abs{u_n - \ell'} \leq \varepsilon$, logo
\[
\ell' - \ell \leq \abs{\ell' - u_n} + \abs{u_n - \ell}
\leq 2\varepsilon = \tfrac{2}{3}(\ell' - \ell) < \ell' - \ell,
\]
uma contradição.
\end{proof}

\begin{proposition}[Operações com limites]\label{prop:g12:seq:operations}
Sejam $(u_n)$ e $(v_n)$ \omterm{def:g12:seq:sequence}{sequências} de limites $\ell$ e $\ell'$ (finitos ou
infinitos). Então, sempre que o lado direito não for uma forma
indeterminada,
\[
\lim (u_n + v_n) = \ell + \ell', \qquad
\lim (u_n v_n) = \ell\,\ell', \qquad
\lim \frac{u_n}{v_n} = \frac{\ell}{\ell'}.
\]
As formas indeterminadas são $(+\infty) + (-\infty)$,
$0 \times \infty$, $\frac{\infty}{\infty}$ e $\frac{0}{0}$.
\end{proposition}

\begin{proof}
Demonstramos a regra da soma para limites finitos; os demais casos são
análogos e ficam como exercício. Seja $\varepsilon > 0$. Existem
$N_1, N_2$ tais que $\abs{u_n - \ell} \leq \varepsilon/2$ para
$n \geq N_1$ e $\abs{v_n - \ell'} \leq \varepsilon/2$ para $n \geq N_2$.
Para $n \geq \max(N_1, N_2)$, a desigualdade triangular dá
\[
\abs{(u_n + v_n) - (\ell + \ell')}
\leq \abs{u_n - \ell} + \abs{v_n - \ell'} \leq \varepsilon. \qedhere
\]
\end{proof}

\begin{method}[Levantar uma forma indeterminada]\label{met:g12:seq:indeterminate}
Diante de uma forma indeterminada, coloque o termo dominante em
evidência. Por exemplo,
\[
n^2 - n = n^2\left(1 - \tfrac{1}{n}\right) \xrightarrow[n\to+\infty]{} +\infty,
\qquad
\frac{2n^2+1}{n^2 - n} = \frac{2 + 1/n^2}{1 - 1/n}
\xrightarrow[n\to+\infty]{} 2 .
\]
\end{method}

\section{Teoremas de convergência}

\begin{theorem}[Teoremas de comparação e do confronto]\label{thm:g12:seq:squeeze}
Sejam $(u_n)$, $(v_n)$, $(w_n)$ \omterm{def:g12:seq:sequence}{sequências}.
\begin{enumerate}
  \item Se $u_n \leq v_n$ a partir de certo índice e $u_n \to +\infty$,
        então $v_n \to +\infty$.
  \item \emph{(Teorema do confronto)}\index{teorema do confronto} Se
        $u_n \leq v_n \leq w_n$ a partir de certo índice e $(u_n)$ e
        $(w_n)$ convergem ambas para o mesmo limite $\ell$, então $(v_n)$
        \omterm{def:g12:seq:limit}{converge} para~$\ell$.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} Seja $A \in \R$. Como $u_n \to +\infty$, existe $N$ com
$u_n \geq A$ para $n \geq N$; aumentando $N$ se preciso,
$v_n \geq u_n \geq A$ para $n \geq N$.

\emph{2.} Seja $\varepsilon > 0$. A partir de um certo índice valem
$\ell - \varepsilon \leq u_n$ e $w_n \leq \ell + \varepsilon$, logo
$\ell - \varepsilon \leq u_n \leq v_n \leq w_n \leq \ell + \varepsilon$,
isto é, $\abs{v_n - \ell} \leq \varepsilon$.
\end{proof}

\begin{example}
Para todo $n \geq 1$,
$-\frac{1}{n} \leq \frac{(-1)^n}{n} \leq \frac{1}{n}$, e as duas cotas
tendem a $0$; logo $\frac{(-1)^n}{n} \to 0$.
\end{example}

\begin{theorem}[Teorema da convergência monótona]\label{thm:g12:seq:monotone}
Uma \omterm{def:g12:seq:sequence}{sequência} \omterm{def:g12:seq:monotonic}{crescente} e \omterm{def:g12:seq:bounded}{limitada superiormente} \omterm{def:g12:seq:limit}{converge}. Uma \omterm{def:g12:seq:sequence}{sequência}
\omterm{def:g10:functions:variations}{decrescente} e \omterm{def:g12:seq:bounded}{limitada inferiormente} \omterm{def:g12:seq:limit}{converge}. Uma \omterm{def:g12:seq:sequence}{sequência} \omterm{def:g12:seq:monotonic}{crescente} que
não é \omterm{def:g12:seq:bounded}{limitada superiormente} tende a $+\infty$.
\end{theorem}

\begin{proof}[Demonstração parcial]
Demonstramos a terceira afirmação. Seja $(u_n)$ \omterm{def:g12:seq:monotonic}{crescente} e não \omterm{def:g12:seq:bounded}{limitada
superiormente}, e seja $A \in \R$. Como $A$ não é cota superior, existe $N$
com $u_N \geq A$; pela monotonicidade, $u_n \geq u_N \geq A$ para todo
$n \geq N$. Logo $u_n \to +\infty$.

As duas afirmações de convergência apoiam-se na propriedade do supremo de
$\R$; elas são \emph{admitidas neste nível} (e demonstradas no primeiro
ano de graduação).
\end{proof}

\begin{remark}
O teorema garante a \emph{existência} do limite, mas não fornece seu
valor. Uma \omterm{def:g12:seq:sequence}{sequência} \omterm{def:g12:seq:monotonic}{crescente} e \omterm{def:g12:seq:bounded}{limitada superiormente} por $M$ \omterm{def:g12:seq:limit}{converge}
para algum $\ell \leq M$, não necessariamente para $M$.
\end{remark}

\begin{theorem}[Limite de progressões geométricas]\label{thm:g12:seq:geometric}
Seja $q \in \R$.
\begin{enumerate}
  \item Se $q > 1$, então $q^n \to +\infty$.
  \item Se $q = 1$, então $q^n \to 1$.
  \item Se $\abs{q} < 1$, então $q^n \to 0$.
  \item Se $q \leq -1$, então $(q^n)$ diverge e não tem limite.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} Escreva $q = 1 + a$ com $a > 0$. A desigualdade de Bernoulli
(\cref{ex:g12:seq:bernoulli}) dá $q^n \geq 1 + na \to +\infty$, e
concluímos por comparação (\cref{thm:g12:seq:squeeze}).

\emph{2.} Imediato.

\emph{3.} Se $q = 0$ a afirmação é clara. Caso contrário,
$\abs{q} < 1$ dá $1/\abs{q} > 1$, logo $(1/\abs{q})^n \to +\infty$ pelo
item 1, portanto $\abs{q}^n \to 0$, e
$-\abs{q}^n \leq q^n \leq \abs{q}^n$ permite concluir pelo teorema do
confronto.

\emph{4.} Para $q \leq -1$, $(q^{2n})$ assume valores $\geq 1$, enquanto
$(q^{2n+1})$ assume valores $\leq -1$: nenhum limite único pode atrair as
duas subsequências.
\end{proof}

\begin{method}[Sequências recorrentes $u_{n+1} = f(u_n)$]\label{met:g12:seq:recurrent}
Para estudar uma \omterm{def:g12:seq:sequence}{sequência} definida por $u_{n+1} = f(u_n)$:
\begin{enumerate}
  \item demonstre por indução que $(u_n)$ permanece em um \omterm{def:g10:numbers:interval}{intervalo} $I$ no
        qual $f$ se comporta bem (e, muitas vezes, que $(u_n)$ é
        \omterm{def:g12:seq:monotonic}{monótona});
  \item deduza a convergência do teorema da convergência \omterm{def:g12:seq:monotonic}{monótona};
  \item passe ao limite na relação $u_{n+1} = f(u_n)$: se $f$ é contínua e
        $u_n \to \ell \in I$, então $\ell$ satisfaz $f(\ell) = \ell$ (veja
        o \cref{ch:g12:limcont}); resolva essa \omterm{def:g10:algebra:equation}{equação} e escolha a \omterm{def:g11:quad:discriminant}{raiz}
        certa.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g12:seq:sqrt2}
Sejam $u_0 = 2$ e $u_{n+1} = \frac{1}{2}\left(u_n +
\frac{2}{u_n}\right)$. Verifica-se por indução que $u_n \geq \sqrt{2}$
para todo $n$ (a desigualdade $\frac{1}{2}(x + 2/x) \geq \sqrt{2}$ para
$x>0$ equivale a $(x - \sqrt2)^2 \geq 0$) e, em seguida, que $(u_n)$ é
\omterm{def:g10:functions:variations}{decrescente}, pois
\[
u_{n+1}-u_n=\frac{2-u_n^2}{2u_n}\leq 0 .
\]
\omterm{def:g10:functions:variations}{Decrescente} e \omterm{def:g12:seq:bounded}{limitada inferiormente}, $(u_n)$ \omterm{def:g12:seq:limit}{converge} para algum
$\ell \geq \sqrt{2}$, que deve satisfazer
$\ell = \frac{1}{2}(\ell + 2/\ell)$, isto é, $\ell^2 = 2$. Logo
$u_n \to \sqrt{2}$. Esse é o algoritmo de Heron, já usado pelos
babilônios; sua convergência é extremamente rápida ($u_3$ já dá $\sqrt 2$
com oito casas decimais corretas).
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:seq:1}
Demonstre por indução que, para todo $n \in \N$,
\[
1^2 + 2^2 + \dots + n^2 = \frac{n(n+1)(2n+1)}{6}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:2}
Estude a monotonicidade das \omterm{def:g12:seq:sequence}{sequências} definidas, para $n \geq 1$, por
\[
a_n = \frac{n+1}{n}, \qquad
b_n = \frac{2^n}{n}, \qquad
c_n = n^2 - 10n .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:3}
Calcule os limites das \omterm{def:g12:seq:sequence}{sequências} de termos gerais
\[
u_n = \frac{3n^2 - n + 1}{2n^2 + 5}, \qquad
v_n = \sqrt{n+1} - \sqrt{n}, \qquad
w_n = \frac{2^n - 3^n}{3^n + 1}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:seq:4}
Seja $(u_n)$ a \omterm{def:g12:seq:arith-geom}{progressão aritmética} com $u_0 = 5$ e razão $r = 3$, e
$(v_n)$ a \omterm{def:g12:seq:arith-geom}{progressão geométrica} com $v_0 = 8$ e razão $q = \frac{1}{2}$.
Calcule $u_n$, $v_n$, $\sum_{k=0}^{n} u_k$ e $\sum_{k=0}^{n} v_k$, e os
limites das quatro expressões quando $n \to +\infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:5}
Usando o teorema do confronto, calcule
\[
\lim_{n\to+\infty} \frac{n + \cos n}{n + 1}
\qquad\text{e}\qquad
\lim_{n\to+\infty} \frac{n!}{n^n},
\]
em que $n! = 1 \times 2 \times \dots \times n$. Para o segundo limite,
limite $\frac{n!}{n^n}$ por um termo de uma \omterm{def:g12:seq:arith-geom}{progressão geométrica}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:6}
Sejam $u_0 = 0$ e $u_{n+1} = \sqrt{u_n + 2}$ para todo $n \in \N$.
\begin{enumerate}
  \item Demonstre por indução que $0 \leq u_n \leq 2$ para todo $n$.
  \item Mostre que $(u_n)$ é \omterm{def:g12:seq:monotonic}{crescente}.
  \item Deduza que $(u_n)$ \omterm{def:g12:seq:limit}{converge} e determine seu limite.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:7}
Um paciente toma uma dose de $1$ unidade de um medicamento toda manhã. A
cada período de 24 horas, o organismo elimina $40\%$ do medicamento
presente. Seja $u_n$ a quantidade de medicamento no organismo logo após a
dose do dia $n$, de modo que $u_0 = 1$.
\begin{enumerate}
  \item Justifique que $u_{n+1} = 0.6\,u_n + 1$.
  \item Seja $v_n = u_n - 2.5$. Mostre que $(v_n)$ é uma \omterm{def:g12:seq:arith-geom}{progressão
        geométrica} e deduza uma fórmula explícita para $u_n$.
  \item Determine a quantidade de medicamento no organismo a longo prazo.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:seq:8}
Seja $(u_n)$ definida por $u_0 = 3$ e
$u_{n+1} = \frac{4u_n - 1}{u_n + 2}$.
\begin{enumerate}
  \item Mostre por indução que $u_n > 1$ para todo $n \in \N$.
  \item Mostre que $v_n = \dfrac{1}{u_n - 1}$ define uma \omterm{def:g12:seq:arith-geom}{progressão
        aritmética}.
  \item Deduza fórmulas explícitas para $v_n$ e $u_n$, e o limite de
        $(u_n)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:seq:9}
Para $n \geq 1$, seja
$H_n = 1 + \frac{1}{2} + \frac{1}{3} + \dots + \frac{1}{n}$.
\begin{enumerate}
  \item Mostre que, para todo $n \geq 1$, $H_{2n} - H_n \geq \frac{1}{2}$.
  \item Deduza que $H_{2^k} \geq 1 + \frac{k}{2}$ para todo $k \in \N$ e
        conclua que $H_n \to +\infty$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:seq:10}
\emph{(\omterm{def:g12:seq:sequence}{Sequências} adjacentes.)} Duas \omterm{def:g12:seq:sequence}{sequências} $(a_n)$ e $(b_n)$ são
\emph{adjacentes} se $(a_n)$ é \omterm{def:g12:seq:monotonic}{crescente}, $(b_n)$ é \omterm{def:g10:functions:variations}{decrescente} e
$b_n - a_n \to 0$.
\begin{enumerate}
  \item Mostre que, para todo $n$, $a_n \leq b_n$. (Sugestão: estude a
        monotonicidade de $(b_n - a_n)$.)
  \item Mostre que \omterm{def:g12:seq:sequence}{sequências} adjacentes convergem ambas para o mesmo
        limite.
  \item Aplicação: mostre que as \omterm{def:g12:seq:sequence}{sequências}
        $a_n = \sum_{k=0}^{n} \frac{1}{k!}$ e
        $b_n = a_n + \frac{1}{n \cdot n!}$ ($n \geq 1$) são adjacentes.
        (Seu limite comum é o número $\eu$, estudado no
        \cref{ch:g12:exp}.)
\end{enumerate}
\end{exercise}

\section{Problema: A sequência de Heron, enfim julgada}

\begin{problem}[{Problema de fim de semana --- a indução certifica, a
convergência monótona sentencia, e a receita de dois mil anos para
$\sqrt2$ finalmente ganha sua demonstração (com a maravilhosa média
de Gauss de sobremesa)}]
\label{pb:g12:seq:1}
Três vezes esta série encontrou a receita de Heron --- tire a
\omterm{def:g11:stat:mean}{média} entre o palpite e $2/\text{palpite}$ --- e três vezes só
pôde \emph{observar} que a receita funciona. Este capítulo enfim
possui os instrumentos do julgamento: a indução
(\cref{thm:g12:seq:induction}), o teorema da convergência \omterm{def:g12:seq:monotonic}{monótona}
(\cref{thm:g12:seq:monotone}) e os limites de recorrências. O
veredicto, e a velocidade certificada, ocupam o coração deste
problema; em volta dele, as armadilhas clássicas da indução, a
divergência mais lenta da matemática e a convergência mais rápida
que Gauss encontrou.

\medskip
\noindent\textbf{Parte I --- Aquecimentos de indução.}
\begin{enumerate}
  \item Demonstre por indução:
        $1 + 3 + 5 + \dots + (2n - 1) = n^2$ (a escada dos
        números ímpares, desenhada no volume do ensino
        fundamental, agora certificada).
  \item Demonstre por indução que $2^n > n$ para todo $n \in \N$.
  \item Demonstre a desigualdade de Bernoulli por indução: para
        $x \geq 0$ e $n \in \N$, $(1 + x)^n \geq 1 + nx$.
  \item A armadilha clássica: ``todas as bolinhas de gude têm a
        mesma cor --- verdade para uma bolinha; e, se quaisquer
        $n$ bolinhas são sempre monocromáticas, então entre
        $n + 1$ bolinhas as $n$ primeiras partilham uma cor, as
        $n$ últimas partilham uma cor, logo todas as $n + 1$
        partilham''. Toda criança sabe que a conclusão é absurda:
        encontre o passo exato em que a indução quebra.
  \item Demonstre por indução que $4^n - 1$ é divisível por $3$
        para todo $n \in \N$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O julgamento de Heron.}
Sejam $x_0 = 2$ e $x_{n+1} = \dfrac12\left(x_n +
\dfrac{2}{x_n}\right)$.
\begin{enumerate}[resume]
  \item Calcule $x_1$, $x_2$, $x_3$ como frações exatas (velhos
        conhecidos).
  \item Demonstre a identidade-chave
        \[
        x_{n+1}^2 - 2 = \left(\frac{x_n^2 - 2}{2x_n}\right)^{\!2}
        \geq 0,
        \]
        e deduza por indução que $x_n > 0$ e $x_n^2 > 2$ para
        todo $n$.
  \item Mostre que $(x_n)$ é estritamente \omterm{def:g10:functions:variations}{decrescente} (calcule
        $x_{n+1} - x_n$ e use a questão 7).
  \item Invoque o teorema da convergência \omterm{def:g12:seq:monotonic}{monótona}: por que
        $(x_n)$ \omterm{def:g12:seq:limit}{converge} para algum limite $L \geq 1$?
  \item Identifique o limite: passe a recorrência ao limite
        (\cref{prop:g12:seq:operations}) e conclua que
        $L = \sqrt2$. Enuncie o veredicto histórico: depois de
        dois mil anos de serviço fiel, está \emph{demonstrado}
        que a receita de Heron \omterm{def:g12:seq:limit}{converge}.
  \item A velocidade certificada: com $e_n = x_n - \sqrt2$,
        demonstre que
        \[
        e_{n+1} = \frac{e_n^2}{2 x_n} ,
        \]
        e deduza $e_{n+1} \leq \frac{e_n^2}{2\sqrt2}$: o erro é
        elevado ao quadrado a cada passo --- a duplicação de
        dígitos observada desde o volume do ensino fundamental,
        agora um teorema.
  \item Confirme numericamente: calcule $e_0, e_1, e_2, e_3$ (a
        partir da questão 6) e verifique que cada
        $\frac{e_{n+1}}{e_n^2}$ está próximo de
        $\frac{1}{2x_n}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A divergência mais lenta.}
\begin{enumerate}[resume]
  \item O \cref{exo:g12:seq:9} demonstrou
        $H_{2^k} \geq 1 + \frac k2$ para as somas harmônicas.
        Quantos termos garantem $H_n > 10$? (Uma potência de dois
        resolve; espante-se com o tamanho dela.)
  \item Em contraste, as somas geométricas
        $1 + \frac12 + \frac14 + \dots + \frac{1}{2^n} =
        2 - \frac{1}{2^n}$ convergem para $2$
        (\cref{thm:g12:seq:geometric}): a intuição da barra de
        chocolate do volume do ensino fundamental, enfim como
        afirmação sobre limites. Escreva a demonstração de duas
        linhas.
  \item Entre as duas: mostre que as somas
        $S_n = 1 + \frac{1}{4} + \frac{1}{9} + \dots +
        \frac{1}{n^2}$ convergem, limitando
        $\frac{1}{k^2} \leq \frac{1}{k(k-1)} =
        \frac{1}{k-1} - \frac{1}{k}$ (para $k \geq 2$),
        telescopando e aplicando a convergência \omterm{def:g12:seq:monotonic}{monótona}. (O
        limite, $\frac{\pi^2}{6}$, é um dos milagres de Euler,
        demonstrado nos volumes de graduação.)
  \item Enuncie em duas frases a moral das questões 13--15: o que
        ``os termos tendem a $0$'' decide sobre a convergência
        das somas --- e o que não decide?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A \omterm{def:g11:stat:mean}{média} aritmético-geométrica de
Gauss.}
Sejam $a_0 = 1$, $b_0 = 2$ e
\[
a_{n+1} = \sqrt{a_n b_n},
\qquad
b_{n+1} = \frac{a_n + b_n}{2} .
\]
\begin{enumerate}[resume]
  \item Calcule $a_1, b_1, a_2, b_2$ (cinco casas decimais). O
        que você observa quanto à velocidade?
  \item Mostre que $a_n \leq b_n$ para todo $n$ (a desigualdade
        entre as \omterm{def:g11:stat:mean}{médias} aritmética e geométrica, encontrada ao
        longo de toda esta série), que $(a_n)$ cresce e que
        $(b_n)$ decresce.
  \item Mostre que $b_{n+1} - a_{n+1} \leq \frac{b_n - a_n}{2}$
        (fatore $b_{n+1} - a_{n+1} =
        \frac{(\sqrt{b_n} - \sqrt{a_n})^2}{2}$ e compare) e
        conclua, com o \cref{exo:g12:seq:10}, que as duas
        \omterm{def:g12:seq:sequence}{sequências} são adjacentes: elas partilham um limite comum
        $M(1, 2)$, a \emph{\omterm{def:g11:stat:mean}{média} aritmético-geométrica}.
  \item Calcule $M(1, 2)$ com seis casas decimais (de quantas
        iterações você precisou?). Em 30 de maio de 1799, Gauss
        calculou $M(1, \sqrt2)$ com onze casas decimais,
        reconheceu $\frac{\pi}{M(1,\sqrt2)}$ como uma integral
        conhecida e escreveu que se abrira um ``novo campo da
        análise'' --- e de fato se abrira: as integrais
        elípticas, contadas nos volumes de graduação. Encerre com
        as velocidades de convergência observadas neste problema,
        da mais lenta à mais rápida.
\end{enumerate}
\end{problem}
