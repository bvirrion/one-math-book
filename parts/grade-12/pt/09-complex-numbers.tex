\chapter{Números complexos}\label{ch:g12:complex}

A \omterm{def:g10:algebra:equation}{equação} $x^2 = -1$ não tem solução real. Ampliar $\R$ com um único número
novo $\iu$, cujo quadrado é $-1$, produz o corpo $\C$ dos \omterm{def:g12:complex:def}{números
complexos}, no qual \emph{toda} \omterm{def:g10:algebra:equation}{equação} polinomial tem soluções --- e cuja
aritmética codifica a geometria do plano: translações, rotações e
dilatações viram adições e multiplicações.

\section{Forma algébrica}

\begin{definition}[Números complexos]\label{def:g12:complex:def}
O conjunto dos \emph{números complexos}\index{número complexo} é
\[
\C = \{a + \iu b : a, b \in \R\},
\]
em que $\iu$ é um símbolo sujeito à única regra $\iu^2 = -1$; a adição e a
multiplicação são feitas como para polinômios reais em $\iu$, reduzindo
$\iu^2$ a $-1$. Os reais $a$ e $b$ são a \emph{parte real}\index{parte
real} $\Rea(z)$ e a \emph{parte imaginária}\index{parte imaginária}
$\Ima(z)$ de $z = a + \iu b$; essa expressão é a \emph{forma
algébrica}\index{forma algébrica} de $z$, e ela é única:
$a + \iu b = a' + \iu b'$ se, e somente se, $a = a'$ e $b = b'$.
\end{definition}

\begin{example}
$(2 + 3\iu)(1 - \iu) = 2 - 2\iu + 3\iu - 3\iu^2 = 5 + \iu$.
\end{example}

\begin{definition}[Conjugado]\label{def:g12:complex:conjugate}
O \emph{conjugado}\index{conjugado} de $z = a + \iu b$ é
$\conj{z} = a - \iu b$.
\end{definition}

\begin{proposition}[Propriedades da conjugação]\label{prop:g12:complex:conjugate}
Para todos $z, w \in \C$:
\[
\conj{z + w} = \conj z + \conj w, \quad
\conj{zw} = \conj z\,\conj w, \quad
\conj{\conj z} = z, \quad
z + \conj z = 2\Rea(z), \quad
z - \conj z = 2\iu\,\Ima(z),
\]
e $z\conj z = a^2 + b^2 \geq 0$ para $z = a + \iu b$. Além disso,
$z \in \R$ se, e somente se, $z = \conj z$.
\end{proposition}

\begin{proof}
Todas são cálculos diretos a partir das \omterm{def:g12:complex:def}{formas algébricas}; por exemplo,
$z\conj z = (a + \iu b)(a - \iu b) = a^2 - (\iu b)^2 = a^2 + b^2$.
\end{proof}

\begin{method}[Dividir números complexos]\label{met:g12:complex:division}
Para escrever um quociente na \omterm{def:g12:complex:def}{forma algébrica}, multiplique numerador e
denominador pelo \omterm{def:g12:complex:conjugate}{conjugado} do denominador:
\[
\frac{1}{2 + \iu} = \frac{2 - \iu}{(2+\iu)(2-\iu)} = \frac{2 - \iu}{5}
= \frac25 - \frac15\iu .
\]
Todo \omterm{def:g12:complex:def}{número complexo} não nulo tem, portanto, um inverso: $\C$ é um
\emph{corpo}.
\end{method}

\section{O plano complexo, módulo e argumento}

\begin{definition}[Afixo, módulo]\label{def:g12:complex:modulus}
No plano com \omterm{def:g10:coordgeom:system}{coordenadas} \omterm{def:g10:coordgeom:system}{ortonormais}, o ponto $M(a, b)$ é a \emph{\omterm{def:g10:functions:function}{imagem}}
de $z = a + \iu b$, e $z$ é o \emph{afixo}\index{afixo} de $M$. O
\emph{módulo}\index{módulo} de $z$ é a distância de $M$ à origem:
\[
\abs{z} = \sqrt{a^2 + b^2} = \sqrt{z\conj z}.
\]
\end{definition}

\begin{proposition}[Propriedades do módulo]\label{prop:g12:complex:modulus}
Para $z, w \in \C$:
\[
\abs{zw} = \abs z\,\abs w, \qquad
\abs{\tfrac zw} = \tfrac{\abs z}{\abs w}\ (w \neq 0), \qquad
\abs{\conj z} = \abs z,
\]
\[
\abs{z + w} \leq \abs{z} + \abs{w} \quad
\text{(desigualdade triangular)}.
\]
Além disso, $\abs{z_B - z_A}$ é a distância entre os pontos de \omterm{def:g12:complex:modulus}{afixos}
$z_A$ e $z_B$.
\end{proposition}

\begin{proof}
$\abs{zw}^2 = zw\,\conj{zw} = z\conj z\, w \conj w = \abs z^2 \abs w^2$ dá
a primeira; a segunda e a terceira são análogas. Para a desigualdade
triangular:
\[
\abs{z+w}^2 = (z+w)(\conj z + \conj w)
= \abs z^2 + \abs w^2 + 2\Rea(z \conj w)
\leq \abs z^2 + \abs w^2 + 2\abs{z\conj w}
= \bigl(\abs z + \abs w\bigr)^2,
\]
usando $\Rea(u) \leq \abs u$. A afirmação sobre a distância é a fórmula de
Pitágoras aplicada às \omterm{def:g10:coordgeom:system}{coordenadas} de $z_B - z_A$.
\end{proof}

\begin{definition}[Argumento, formas trigonométrica e exponencial]\label{def:g12:complex:argument}
Seja $z \neq 0$. Um \emph{argumento}\index{argumento} de $z$, escrito
$\Arg(z)$, é qualquer medida $\theta$ do ângulo do semieixo real positivo
até a semirreta $OM$; ele é definido a menos de $2k\pi$. Escrevendo
$r = \abs z$,
\[
z = r(\cos\theta + \iu\sin\theta)
\qquad\text{(forma trigonométrica)}.
\]
Introduzindo a notação\index{forma exponencial}
\[
\eu^{\iu\theta} = \cos\theta + \iu\sin\theta,
\]
isso vira a \emph{forma exponencial} $z = r\,\eu^{\iu\theta}$.
\end{definition}

\begin{theorem}\label{thm:g12:complex:expform}
Para todos $\theta, \theta' \in \R$:
\[
\eu^{\iu\theta}\,\eu^{\iu\theta'} = \eu^{\iu(\theta + \theta')},
\qquad
\conj{\eu^{\iu\theta}} = \eu^{-\iu\theta},
\qquad
\abs{\eu^{\iu\theta}} = 1 .
\]
Por consequência, para $z = r\eu^{\iu\theta}$ e
$z' = r'\eu^{\iu\theta'}$ não nulos:
\[
zz' = rr'\,\eu^{\iu(\theta+\theta')},
\qquad
\frac{z}{z'} = \frac{r}{r'}\,\eu^{\iu(\theta - \theta')} :
\]
\emph{os \omterm{def:g12:complex:modulus}{módulos} se multiplicam, os \omterm{def:g12:complex:argument}{argumentos} se somam.}
\end{theorem}

\begin{proof}
A primeira identidade \emph{é} o par de fórmulas de adição
(\cref{prop:g12:trigo:addition}):
\[
(\cos\theta + \iu\sin\theta)(\cos\theta' + \iu\sin\theta')
= (\cos\theta\cos\theta' - \sin\theta\sin\theta')
+ \iu(\sin\theta\cos\theta' + \cos\theta\sin\theta').
\]
O restante segue por cálculo direto.
\end{proof}

\begin{corollary}[Fórmula de De Moivre]\label{cor:g12:complex:demoivre}
\index{fórmula de De Moivre}
Para $\theta \in \R$ e $n \in \Z$:
$\bigl(\cos\theta + \iu\sin\theta\bigr)^n = \cos n\theta + \iu \sin n\theta$.
\end{corollary}

\begin{proof}
Indução em $n \geq 0$ usando o \cref{thm:g12:complex:expform}; estenda a
$n < 0$ tomando inversos.
\end{proof}

\begin{method}[Passar de uma forma a outra]\label{met:g12:complex:forms}
Da algébrica para a exponencial: calcule $r = \abs z$ e depois encontre
$\theta$ com $\cos\theta = \frac ar$ e $\sin\theta = \frac br$ (localize o
quadrante correto antes de invocar um arco \omterm{def:g12:trigo:cossin}{cosseno}). Da exponencial para a
algébrica: expanda $r\cos\theta + \iu\, r\sin\theta$. Use a \omterm{def:g12:complex:def}{forma algébrica}
para somas e a exponencial para produtos, potências e quocientes.
\end{method}

\begin{example}
$z = 1 + \iu$: $\abs z = \sqrt2$ e
$\cos\theta = \sin\theta = \frac{1}{\sqrt2}$, logo $\theta = \frac\pi4$ e
$z = \sqrt2\,\eu^{\iu\pi/4}$. Assim,
$z^{20} = 2^{10}\,\eu^{5\iu\pi} = -1024$.
\end{example}

\section{Equações do segundo grau e raízes da unidade}

\begin{theorem}[Equações do segundo grau com coeficientes reais]\label{thm:g12:complex:quadratic}
Sejam $a, b, c \in \R$, $a \neq 0$, e $\Delta = b^2 - 4ac$. A \omterm{def:g10:algebra:equation}{equação}
$az^2 + bz + c = 0$ tem soluções em $\C$:
\[
\begin{aligned}
&\Delta > 0:\ z = \frac{-b \pm \sqrt{\Delta}}{2a} \ (\text{duas reais}); \qquad
\Delta = 0:\ z = \frac{-b}{2a} \ (\text{dupla}); \\
&\Delta < 0:\ z = \frac{-b \pm \iu\sqrt{-\Delta}}{2a} \ (\text{duas conjugadas}).
\end{aligned}
\]
\end{theorem}

\begin{proof}
Complete o quadrado:
$az^2 + bz + c = a\left(\left(z + \frac{b}{2a}\right)^2 -
\frac{\Delta}{4a^2}\right)$. Se $\Delta < 0$, então
$\frac{\Delta}{4a^2} = \left(\iu\frac{\sqrt{-\Delta}}{2a}\right)^2$, e a
diferença de quadrados se fatora como de costume.
\end{proof}

\begin{definition}[Raízes da unidade]\label{def:g12:complex:rootsofunity}
Para $n \geq 1$, as \emph{raízes $n$-ésimas da unidade}\index{raízes da
unidade} são as soluções de $z^n = 1$.
\end{definition}

\begin{theorem}\label{thm:g12:complex:rootsofunity}
A \omterm{def:g10:algebra:equation}{equação} $z^n = 1$ tem exatamente $n$ soluções:
\[
\omega_k = \eu^{2\iu k\pi/n}, \qquad k = 0, 1, \dots, n-1 .
\]
Suas \omterm{def:g10:functions:function}{imagens} são os vértices de um polígono regular de $n$ lados inscrito
no círculo de raio $1$ e, para $n \geq 2$, sua soma é $0$.
\end{theorem}

\begin{proof}
Cada $\omega_k$ satisfaz $\omega_k^n = \eu^{2\iu k\pi} = 1$, e eles são
dois a dois distintos, pois seus \omterm{def:g12:complex:argument}{argumentos} estão em $\intco{0}{2\pi}$.
Reciprocamente, se $z^n = 1$, então $\abs z^n = 1$, logo $\abs z = 1$
(real positivo) e $z = \eu^{\iu\theta}$ com
$n\theta \equiv 0 \pmod{2\pi}$, \emph{isto é},
$\theta = \frac{2k\pi}{n}$; reduzir $k$ módulo $n$ leva aos $\omega_k$. Os
vértices estão igualmente espaçados pelo ângulo $\frac{2\pi}{n}$: um
polígono regular de $n$ lados. Por fim, com $\omega = \omega_1$,
$\omega \neq 1$ e a soma geométrica dá
\[
\sum_{k=0}^{n-1}\omega_k = \sum_{k=0}^{n-1}\omega^k
= \frac{\omega^n - 1}{\omega - 1} = 0 . \qedhere
\]
\end{proof}

\end{example}

\section{Números complexos e geometria plana}

\begin{proposition}[Dicionário geométrico]\label{prop:g12:complex:geometry}
Sejam $A, B, C, D$ pontos de \omterm{def:g12:complex:modulus}{afixos} $a, b, c, d$ ($a \neq b$,
$c \neq d$).
\begin{enumerate}
  \item A translação pelo \omterm{def:g11:vect:vector}{vetor} de \omterm{def:g12:complex:modulus}{afixo} $t$ é $z \mapsto z + t$.
  \item A rotação de centro $\omega$ (\omterm{def:g12:complex:modulus}{afixo}) e ângulo $\theta$ é
        $z \mapsto \omega + \eu^{\iu\theta}(z - \omega)$.
  \item A homotetia de centro $\omega$ e razão $k \in \R^*$ é
        $z \mapsto \omega + k(z - \omega)$.
  \item $\dfrac{d - c}{b - a}$ tem módulo $\dfrac{CD}{AB}$ e \omterm{def:g12:complex:argument}{argumento}
        igual ao ângulo entre os \omterm{def:g11:vect:vector}{vetores} $\vect{AB}$ e $\vect{CD}$. Em
        particular, $AB \perp CD$ se, e somente se, $\frac{d-c}{b-a}$ é
        imaginário puro, e $A, B, C$ são \omterm{def:g11:vect:collinear}{colineares} ($C \ne A$,
        $C\ne B$) se, e somente se, $\frac{c-a}{b-a} \in \R$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) e (3) reenunciam as fórmulas em \omterm{def:g10:coordgeom:system}{coordenadas}. Para (2), a aplicação
$w \mapsto \eu^{\iu\theta} w$ multiplica \omterm{def:g12:complex:modulus}{módulos} por $1$ e soma $\theta$
aos \omterm{def:g12:complex:argument}{argumentos}: é a rotação de ângulo $\theta$ em torno da origem;
conjugar pela translação que leva $\omega$ a $0$ dá o centro geral. Para
(4), os \omterm{def:g12:complex:modulus}{módulos} e os \omterm{def:g12:complex:argument}{argumentos} de quocientes se subtraem
(\cref{thm:g12:complex:expform}).
\end{proof}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:complex:1}
Escreva na \omterm{def:g12:complex:def}{forma algébrica}: $(3 - 2\iu)(1 + 4\iu)$, \quad
$\dfrac{2 + \iu}{1 - \iu}$, \quad $\iu^{2027}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:complex:2}
Resolva em $\C$: (a) $z^2 - 4z + 13 = 0$; (b) $z^2 + z + 1 = 0$. Verifique
em cada caso que as duas soluções são conjugadas e calcule seu produto.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:complex:3}
Coloque na \omterm{def:g12:complex:argument}{forma exponencial}: $z_1 = -1 + \iu$, $z_2 = \sqrt3 - \iu$, e
calcule $\dfrac{z_1}{z_2}$ na \omterm{def:g12:complex:argument}{forma exponencial} e depois na algébrica.
Deduza o valor exato de $\cos\dfrac{11\pi}{12}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:4}
Descreva geometricamente o conjunto dos pontos $M$ de \omterm{def:g12:complex:modulus}{afixo} $z$ tais que:
(a) $\abs{z - 2} = \abs{z + \iu}$;\quad
(b) $\abs{z - 1 - \iu} = 3$;\quad
(c) $\dfrac{z - 1}{z + 1}$ é imaginário puro.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:5}
Usando a fórmula de De Moivre e o teorema binomial, exprima
$\cos 3\theta$ como polinômio em $\cos\theta$, e $\sin 3\theta$ em \omterm{def:g11:func:function}{função}
de $\sin\theta$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:6}
\emph{(Linearização.)} Usando
$\cos\theta = \frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}$, linearize
$\cos^3\theta$ (escreva-o como combinação de $\cos k\theta$) e deduza
$\displaystyle\int_0^{\pi/2}\cos^3\theta\,\dd\theta$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:complex:7}
Sejam $A$ e $B$ de \omterm{def:g12:complex:modulus}{afixos} $a = 1 + \iu$ e $b = 3 + 2\iu$. Determine os dois
pontos $C$ que tornam o triângulo $ABC$ equilátero, como \omterm{def:g10:functions:function}{imagens} de $B$
pelas rotações de centro $A$ e ângulos $\pm\frac{\pi}{3}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:8}
Resolva $z^4 = -4$ em $\C$ e marque as soluções no plano. Fatore
$z^4 + 4$ em dois polinômios do segundo grau com coeficientes reais.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:9}
Para $\theta \in \intoo{0}{2\pi}$ e $n \in \N$, calcule
\[
S = 1 + \cos\theta + \cos 2\theta + \dots + \cos n\theta
\]
somando a \omterm{def:g12:seq:arith-geom}{progressão geométrica} $\sum_{k=0}^n \eu^{\iu k\theta}$ e tomando
\omterm{def:g12:complex:def}{partes reais}. (Resposta:
$S = \dfrac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}}
\cos\frac{n\theta}{2}$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:complex:10}
Seja $\omega = \eu^{2\iu\pi/5}$.
\begin{enumerate}
  \item Justifique que $1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$.
  \item Sejam $u = \omega + \omega^4$ e $v = \omega^2 + \omega^3$. Mostre
        que $u + v = -1$ e $uv = -1$, e deduza que
        $u = \frac{-1+\sqrt5}{2}$.
  \item Conclua que
        $\cos\dfrac{2\pi}{5} = \dfrac{\sqrt5 - 1}{4}$.
\end{enumerate}
\end{exercise}

\section{Problema: O pentágono no espelho da unidade}

\begin{problem}[{Problema de fim de semana --- as raízes quintas da
unidade calculam $\cos 72^\circ$ exatamente, a razão áurea assina o
resultado, e a multiplicação revela-se uma rotação}]
\label{pb:g12:complex:1}
Euclides sabia construir o pentágono regular, mas a razão
\emph{pela qual} ele cede à régua e ao compasso --- enquanto o
humilde ângulo de $20^\circ$ não cede --- ficou escondida por dois
mil anos. Ela se esconde neste capítulo: as cinco raízes quintas
da unidade (\cref{thm:g12:complex:rootsofunity}) somam zero, e essa
única linha de álgebra espreme $\cos 72^\circ$ de uma \omterm{def:g10:algebra:equation}{equação} do
segundo grau --- só raízes quadradas, logo construtível, e
assinada, é claro, pela razão áurea. Em torno dessa joia:
fluência, o atalho de De Moivre para a trigonometria e a
multiplicação desmascarada como geometria.

\medskip
\noindent\textbf{Parte I --- Fluência.}
\begin{enumerate}
  \item Calcule $(2 + \iu)(3 - \iu)$, \
        $\dfrac{1 + \iu}{1 - \iu}$
        (\cref{met:g12:complex:division}) e $\abs{3 + 4\iu}$.
  \item Coloque na \omterm{def:g12:complex:argument}{forma exponencial}: $1 + \iu$; \ $-2$; \
        $1 - \iu\sqrt3$.
  \item Calcule $(1 + \iu)^8$ usando a \omterm{def:g12:complex:argument}{forma exponencial}.
  \item Resolva $z^2 = \iu$ e depois $z^2 - 2z + 5 = 0$
        (\cref{thm:g12:complex:quadratic}).
  \item Dicionário geométrico
        (\cref{prop:g12:complex:geometry}): descreva a aplicação
        $z \mapsto \iu z$ e o conjunto
        $\abs{z - (1 + \iu)} = 2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Cinco raízes, um pentágono.}
Seja $\omega = \eu^{2\iu\pi/5}$ e considere
$1, \omega, \omega^2, \omega^3, \omega^4$.
\begin{enumerate}[resume]
  \item Verifique que essas são exatamente as soluções de
        $z^5 = 1$ e descreva a figura que elas desenham no
        plano.
  \item Demonstre que a soma delas é zero. (\omterm{def:g12:seq:arith-geom}{Progressão
        geométrica} --- ou multiplique a soma por $1 - \omega$.)
  \item Tome \omterm{def:g12:complex:def}{partes reais} e deduza
        \[
        1 + 2\cos\frac{2\pi}{5} + 2\cos\frac{4\pi}{5} = 0 .
        \]
  \item Ponha $c = \cos\frac{2\pi}{5}$ e use a fórmula do arco
        duplo para $\cos\frac{4\pi}{5}$ a fim de converter a
        questão 8 em uma \omterm{def:g10:algebra:equation}{equação} do segundo grau em $c$;
        resolva-a e conclua que
        \[
        \cos 72^\circ = \frac{\sqrt5 - 1}{4} .
        \]
  \item Verifique numericamente e deixe a razão áurea assinar o
        trabalho: mostre que $c = \frac{1}{2\varphi}$, com
        $\varphi = \frac{1 + \sqrt5}{2}$
        (\cref{pb:g10:algebra:1}) --- e enuncie o famoso eco
        geométrico: num pentágono regular, a diagonal vale
        $\varphi$ vezes o lado.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Os atalhos de De Moivre.}
\begin{enumerate}[resume]
  \item O veredicto da construção: $\cos 72^\circ$ só precisa de
        raízes quadradas (questão 9), de modo que o pentágono é
        construtível com régua e compasso; já $\cos 20^\circ$
        satisfaz uma cúbica indomável
        (\cref{pb:g12:trigo:1}), de modo que o ângulo de
        $60^\circ$ não pode ser trissectado. Enuncie o critério
        que emerge (raízes quadradas são construtíveis, raízes
        cúbicas não), cuja teoria completa --- Gauss, aos 19
        anos, e o polígono de 17 lados --- mora nos volumes de
        graduação.
  \item Resolva $z^4 = -4$ pela \omterm{def:g12:complex:argument}{forma exponencial} e recupere a
        fatoração do \cref{exo:g12:complex:8},
        $z^4 + 4 = (z^2 - 2z + 2)(z^2 + 2z + 2)$, emparelhando
        raízes conjugadas.
  \item Expanda $(\cos\theta + \iu\sin\theta)^3$ com o teorema
        binomial e De Moivre, e rededuza
        $\cos 3\theta = 4\cos^3\theta - 3\cos\theta$ em três
        linhas --- compare com o caminho das fórmulas de adição
        do \cref{pb:g12:trigo:1}.
  \item Confronte com o \cref{exo:g12:complex:9}: para
        $\theta = \frac{2\pi}{5}$ e $n = 4$, quanto vale a soma
        $1 + \cos\theta + \dots + \cos 4\theta$, e por que isso é
        coerente com a questão 8?
  \item Calcule as três raízes cúbicas de $8\iu$ na \omterm{def:g12:complex:def}{forma
        algébrica}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A multiplicação é geometria.}
\begin{enumerate}[resume]
  \item Descreva por completo a aplicação
        $z \mapsto (1 + \iu)z$: de que ângulo ela gira, por que
        fator ela dilata e o que faz com o quadrado unitário?
  \item Demonstre $\abs{z_1 z_2} = \abs{z_1}\,\abs{z_2}$ com
        \omterm{def:g12:complex:conjugate}{conjugados} (\cref{prop:g12:complex:modulus}), depois
        calcule $(1+\iu)^n$ para $n = 1, 2, 3, 4$ e descreva a
        espiral que as potências traçam.
  \item O número $\mathrm{j} = \eu^{2\iu\pi/3}$: mostre que
        $1 + \mathrm{j} + \mathrm{j}^2 = 0$ e verifique no
        triângulo $\left(1, \mathrm{j}, \mathrm{j}^2\right)$ o
        critério clássico: $a + \mathrm{j} b + \mathrm{j}^2 c = 0$
        vale para (uma orientação de) um triângulo equilátero
        $abc$.
  \item O teorema fundamental da álgebra (admitido:
        d'Alembert e Gauss): todo polinômio se fatora
        completamente sobre $\C$. Deduza o corolário do mundo
        real --- todo polinômio real se fatora em pedaços reais
        de graus $1$ e $2$ --- e aponte a questão 12 como
        exemplo.
  \item Final --- o retrato de $\C$, uma frase para cada: os
        números viram pontos; a multiplicação vira
        rotação-com-dilatação; as \omterm{def:g12:complex:rootsofunity}{raízes da unidade} viram
        polígonos regulares; De Moivre transforma potências em
        identidades trigonométricas; e o fecho algébrico garante
        que nenhuma \omterm{def:g10:algebra:equation}{equação} jamais precisará de um mundo maior.
        Coda: que dois números célebres se encontraram na
        questão 10?
\end{enumerate}
\end{problem}
