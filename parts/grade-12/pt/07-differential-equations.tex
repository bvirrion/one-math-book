\chapter{Equações diferenciais}\label{ch:g12:diffeq}

Uma \omterm{def:g12:diffeq:ode}{equação diferencial} relaciona uma \omterm{def:g11:func:function}{função} com suas \omterm{def:g12:deriv:derivative}{derivadas}. A física,
a química, a biologia e a economia exprimem suas leis nessa forma: a taxa
de resfriamento de um corpo, o decaimento de um núcleo radioativo, o
crescimento de uma população são todos afirmações sobre $y'$. Este
capítulo resolve por completo as \omterm{def:g10:algebra:equation}{equações} lineares de primeira ordem com
coeficientes constantes.

\section{A equação \texorpdfstring{$y' = ay$}{y' = ay}}

\begin{definition}[Equação diferencial]\label{def:g12:diffeq:ode}
Uma \emph{equação diferencial}\index{equação diferencial} é uma \omterm{def:g10:algebra:equation}{equação}
cuja incógnita é uma \omterm{def:g11:func:function}{função} $y$ e que envolve $y$ e suas \omterm{def:g12:deriv:derivative}{derivadas}. Uma
\emph{solução} em um \omterm{def:g10:numbers:interval}{intervalo} $I$ é uma \omterm{def:g11:func:function}{função} \omterm{def:g12:deriv:derivative}{derivável} que satisfaz a
\omterm{def:g10:algebra:equation}{equação} em todo ponto de $I$.
\end{definition}

\begin{theorem}[Soluções de $y' = ay$]\label{thm:g12:diffeq:homogeneous}
Seja $a \in \R$. As soluções em $\R$ da \omterm{def:g10:algebra:equation}{equação} $y' = ay$ são exatamente as
funções
\[
y(x) = C\,\eu^{ax}, \qquad C \in \R .
\]
Para cada par $(x_0, y_0)$ existe uma única solução com $y(x_0) = y_0$.
\end{theorem}

\begin{proof}
Cada $y = C\eu^{ax}$ satisfaz $y' = Ca\,\eu^{ax} = ay$. Reciprocamente,
seja $y$ uma solução qualquer e ponha $z(x) = y(x)\,\eu^{-ax}$. Então
\[
z'(x) = y'(x)\,\eu^{-ax} - a\,y(x)\,\eu^{-ax}
= \bigl(y'(x) - a y(x)\bigr)\eu^{-ax} = 0,
\]
de modo que $z$ é constante, digamos $C$, e $y(x) = C\eu^{ax}$. A condição
inicial $y(x_0) = y_0$ obriga $C = y_0 \eu^{-a x_0}$, de modo único.
\end{proof}

\begin{remark}
A exponencial é, assim, \emph{caracterizada} pela mais simples de todas as
\omterm{def:g12:diffeq:ode}{equações diferenciais}: crescimento proporcional ao tamanho. É por isso que
ela aparece em toda parte na natureza.
\end{remark}

\begin{example}[Decaimento radioativo]\label{ex:g12:diffeq:decay}
Uma quantidade radioativa satisfaz $N' = -\lambda N$ com $\lambda > 0$,
logo $N(t) = N_0\,\eu^{-\lambda t}$; veja o
\cref{ex:g12:exp:halflife} para a meia-vida.
\end{example}

\section{A equação \texorpdfstring{$y' = ay + b$}{y' = ay + b}}

\begin{theorem}[Soluções de $y' = ay + b$]\label{thm:g12:diffeq:affine}
Sejam $a \neq 0$ e $b \in \R$. As soluções em $\R$ de $y' = ay + b$ são
exatamente as funções
\[
y(x) = C\,\eu^{ax} - \frac{b}{a}, \qquad C \in \R .
\]
A \omterm{def:g11:func:function}{função} constante $y_p = -\frac ba$ é a \emph{solução de equilíbrio}. Para
cada $(x_0, y_0)$ existe uma única solução com $y(x_0) = y_0$; se $a < 0$,
toda solução tende ao equilíbrio $-\frac ba$ quando $x \to +\infty$.
\end{theorem}

\begin{proof}
A constante $y_p = -\frac{b}{a}$ satisfaz $y_p' = 0 = a y_p + b$. Agora
$y$ é solução se, e somente se,
\[
(y - y_p)' = y' = ay + b = a(y - y_p) + \underbrace{a y_p + b}_{=\,0}
= a (y - y_p),
\]
isto é, se e somente se $z = y - y_p$ resolve $z' = az$. Pelo
\cref{thm:g12:diffeq:homogeneous}, $z = C\eu^{ax}$, donde a fórmula, a
existência e a unicidade. Se $a < 0$, $\eu^{ax} \to 0$ quando
$x \to +\infty$, de modo que $y(x) \to -\frac ba$.
\end{proof}

\begin{method}[Resolver $y' = ay + b$ com condição inicial]\label{met:g12:diffeq:solve}
\begin{enumerate}
  \item Encontre a solução de equilíbrio $y_p = -\frac{b}{a}$ (resolva
        $y' = 0$).
  \item Escreva a solução geral $y = C\eu^{ax} + y_p$.
  \item Determine $C$ pela condição inicial.
  \item Confronte o comportamento a longo prazo com a intuição física (a
        solução \omterm{def:g12:seq:limit}{converge} para o equilíbrio?).
\end{enumerate}
A mesma estratégia --- \emph{solução particular mais solução geral da
\omterm{def:g10:algebra:equation}{equação} homogênea} --- estende-se a $y' = ay + f(x)$: veja a
\cref{prop:g12:diffeq:general}.
\end{method}

\begin{example}[Lei do resfriamento de Newton]\label{ex:g12:diffeq:cooling}
Uma xícara de café a $80\,^\circ$C está numa sala a $20\,^\circ$C. A lei
de Newton afirma que a temperatura $T$ satisfaz $T' = -k(T - 20)$ para
algum $k > 0$, \emph{isto é}, $T' = -kT + 20k$. O equilíbrio é $20$, e
$T(t) = 20 + 60\,\eu^{-kt}$: o café esfria exponencialmente depressa até a
temperatura ambiente.
\end{example}

\section{A equação \texorpdfstring{$y' = ay + f(x)$}{y' = ay + f(x)}}

\begin{proposition}[Estrutura do conjunto de soluções]\label{prop:g12:diffeq:general}
Seja $f$ \omterm{def:g12:limcont:continuity}{contínua} em um \omterm{def:g10:numbers:interval}{intervalo} $I$ e seja $y_p$ uma solução particular
de
\[
y' = ay + f(x)
\]
em $I$. Então as soluções em $I$ são exatamente as funções
$y = C\eu^{ax} + y_p$, $C \in \R$.
\end{proposition}

\begin{proof}
Como no \cref{thm:g12:diffeq:affine}: $y$ é solução se, e somente se,
$z = y - y_p$ satisfaz $z' = (ay + f) - (ay_p + f) = az$, se e somente se
$z = C\eu^{ax}$.
\end{proof}

\begin{method}[Adivinhar uma solução particular]\label{met:g12:diffeq:particular}
Procure uma solução particular \emph{da mesma forma} que $f$:
\begin{itemize}
  \item $f$ polinomial de grau $n$: tente um polinômio de grau $n$;
  \item $f(x) = \alpha\,\eu^{kx}$ com $k \neq a$: tente
        $y_p = \beta\,\eu^{kx}$;
  \item $f(x) = \alpha\,\eu^{ax}$ (caso ressonante): tente
        $y_p = \beta x\,\eu^{ax}$.
\end{itemize}
Substitua na \omterm{def:g10:algebra:equation}{equação} e identifique os coeficientes.
\end{method}

\begin{example}\label{ex:g12:diffeq:particular}
Resolva $y' = 2y + 4x$. Tente $y_p = \alpha x + \beta$:
$\alpha = 2(\alpha x + \beta) + 4x$ para todo $x$ obriga
$2\alpha + 4 = 0$ e $\alpha = 2\beta$, logo $\alpha = -2$ e $\beta = -1$.
Solução geral: $y = C\eu^{2x} - 2x - 1$.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:diffeq:1}
Resolva em $\R$:
(a) $y' = 3y$ com $y(0) = 2$;\quad
(b) $2y' + y = 0$ com $y(0) = -1$;\quad
(c) $y' = -y + 5$ com $y(0) = 0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:diffeq:2}
Uma população de bactérias cresce a uma taxa proporcional ao seu tamanho,
dobrando a cada $3$ horas. Escreva a \omterm{def:g12:diffeq:ode}{equação diferencial} satisfeita pela
população $N(t)$ e determine a constante de proporcionalidade.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:diffeq:3}
Verifique que $y_p(x) = x\,\eu^{x}$ é solução de $y' = y + \eu^x$ e dê
todas as soluções em $\R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:4}
Resolva $y' = -2y + \eu^{x}$ com $y(0) = 1$. (Sugestão: procure uma solução
particular da forma $\beta\,\eu^{x}$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:5}
O carbono-14 decai com meia-vida de $5730$ anos. Uma \omterm{def:g10:proba:sample}{amostra} arqueológica
contém $60\%$ do carbono-14 de um organismo vivo. Estime sua idade.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:6}
Um tanque contém $100$ L de água pura. Uma salmoura com $0.2$ kg de sal por
litro entra a $5$ L/min, e a mistura (mantida uniforme) sai à mesma vazão.
Seja $m(t)$ a massa de sal no tanque no instante $t$ (em minutos).
\begin{enumerate}
  \item Justifique que $m' = 1 - \dfrac{m}{20}$.
  \item Resolva e determine o limite de $m(t)$ quando $t \to +\infty$.
        Interprete.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:diffeq:7}
Uma paraquedista de $80$ kg cai sujeita à gravidade
($g = 9.8\ \text{m/s}^2$) e a uma resistência do ar proporcional à
velocidade, de modo que sua velocidade satisfaz
$v' = g - \frac{k}{m}v$ com $k = 16$ kg/s.
\begin{enumerate}
  \item Resolva a \omterm{def:g10:algebra:equation}{equação} com $v(0) = 0$.
  \item Calcule a \emph{velocidade terminal} $\lim_{t\to+\infty} v(t)$ e o
        tempo necessário para atingir $95\%$ dela.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:diffeq:8}
\emph{(\omterm{def:g10:algebra:equation}{Equação} logística.)} Uma população $y(t) \in \intoo{0}{1}$ (como
fração da população máxima) satisfaz
\[
y' = y(1 - y).
\]
\begin{enumerate}
  \item Seja $z = \dfrac{1}{y}$. Mostre que $z$ satisfaz a \omterm{def:g10:algebra:equation}{equação} linear
        $z' = -z + 1$.
  \item Resolva em $z$ e depois em $y$, com $y(0) = \frac{1}{10}$.
  \item Mostre que $y(t) \to 1$ quando $t \to +\infty$ e esboce a forma da
        curva solução.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:diffeq:9}
Seja $y$ uma solução de $y' = ay + b$ e seja $u_n = y(n)$ para
$n \in \N$. Mostre que $(u_n)$ satisfaz uma recorrência
aritmético-geométrica $u_{n+1} = q u_n + r$ e exprima $q$ e $r$ em \omterm{def:g11:func:function}{função}
de $a$ e $b$. A que corresponde a condição $\abs{q} < 1$ para a \omterm{def:g12:diffeq:ode}{equação
diferencial}?
\end{exercise}

\section{Problema: O relógio dentro das coisas}

\begin{problem}[{Problema de fim de semana --- o carbono-14 data as
cavernas, a lei do resfriamento de Newton cronometra um crime, e uma
equaçãozinha veste quatro trajes}]
\label{pb:g12:diffeq:1}
Uma \omterm{def:g12:diffeq:ode}{equação diferencial} é uma lei de mudança; resolvê-la transforma
a lei em um \emph{relógio}. A mesma equaçãozinha $y' = ay + b$
(\cref{thm:g12:diffeq:affine}) faz tique-taque dentro de carvão
pré-histórico, de café esfriando, de gotas de chuva caindo e de
soros hospitalares --- e ler esses relógios é o ofício deste
problema: ele data as pinturas de Lascaux, fixa uma hora da morte e
audita as próprias hipóteses, como faz um bom cientista.

\medskip
\noindent\textbf{Parte I --- Fluência.}
\begin{enumerate}
  \item Resolva $y' = 3y$, $y(0) = 2$; e depois
        $y' = -2y + 6$, $y(0) = 0$
        (\cref{met:g12:diffeq:solve}).
  \item Rededuza o truque de unicidade por trás do
        \cref{thm:g12:diffeq:homogeneous}: se $y' = ay$, calcule
        a \omterm{def:g12:deriv:derivative}{derivada} de $y(t)\,\eu^{-at}$ e conclua que
        \emph{toda} solução é $C\eu^{at}$.
  \item Para $y' = -2y + 6$: encontre a solução de equilíbrio e
        descreva o destino de todas as demais soluções quando
        $t \to +\infty$. (Os \omterm{pb:g10:functions:1}{pontos fixos} das recorrências do
        \cref{pb:g11:seq:1}, agora \omterm{def:g12:limcont:continuity}{contínuos}.)
  \item Para a lei de decaimento $y' = -\frac{y}{\tau}$: mostre
        que a meia-vida é $t_{1/2} = \tau \ln 2$.
  \item Esboce (ou descreva) a família de soluções de
        $y' = -2y + 6$: o que fazem as soluções que começam acima
        de $3$? E abaixo de $3$? E em $3$?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Carbono-14.}
O tecido vivo mantém uma proporção constante de carbono-14
radioativo; na morte a entrada cessa e o estoque decai:
$N' = -\lambda N$, com meia-vida de $5\,730$ anos.
\begin{enumerate}[resume]
  \item Calcule $\lambda$ (por ano).
  \item Um osso retém $20\,\%$ da proporção do vivo: qual é a
        idade dele?
  \item Carvão das cavernas pintadas de Lascaux retém cerca de
        $15\,\%$: date as pinturas.
  \item O homem do gelo, Ötzi, encontrado em uma geleira alpina,
        media cerca de $53\,\%$: date-o (os arqueólogos falam em
        cerca de $5\,300$ anos --- como foi o seu resultado?).
  \item Por que o carbono-14 não pode datar dinossauros? Calcule
        a fração restante após dez meias-vidas, enuncie a fração
        após $65$ milhões de anos como potência de $2$ e
        conclua.
  \item Barras de erro: se os $53\,\%$ de Ötzi só são conhecidos
        com precisão de $\pm 1\,\%$, calcule a faixa de idade.
        Que precisão de laboratório compra que precisão de
        datação?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O resfriamento de Newton e um
crime.}
Um corpo à temperatura $T$ em um ambiente à temperatura constante
$T_a$ esfria segundo $T' = -k\,(T - T_a)$.
\begin{enumerate}[resume]
  \item Resolva a \omterm{def:g10:algebra:equation}{equação} (substitua $z = T - T_a$):
        \[
        T(t) = T_a + (T_0 - T_a)\,\eu^{-kt} .
        \]
  \item Um café servido a $90\,^\circ$C em uma sala a
        $20\,^\circ$C marca $70\,^\circ$C após $5$ minutos.
        Determine $k$ e, em seguida, o tempo de espera até os
        $55\,^\circ$C em que se pode beber.
  \item A questão do leite: para tomar o café mais quente
        possível daqui a dez minutos, o leite frio deve ser
        acrescentado agora ou no último momento? Responda com a
        \omterm{def:g10:algebra:equation}{equação} (o que acrescentar leite faz com a diferença
        $T - T_a$, e como essa diferença comanda a perda?).
  \item Perícia: um corpo é encontrado à meia-noite a
        $30\,^\circ$C em uma sala a $20\,^\circ$C; uma hora
        depois ele marca $28\,^\circ$C. Supondo $37\,^\circ$C no
        momento da morte, recupere $k$ com as duas medições e
        calcule a hora da morte.
  \item Audite o relógio do perito: cite três violações reais das
        hipóteses do modelo e o sentido em que cada uma
        enviesaria a hora da morte estimada.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Quatro trajes e uma curva em S.}
\begin{enumerate}[resume]
  \item A \omterm{def:g10:algebra:equation}{equação} logística do \cref{exo:g12:diffeq:8},
        $y' = y(1 - y)$ com $y(0) = 0.1$: usando a substituição
        do exercício, deduza
        $y(t) = \dfrac{1}{1 + 9\eu^{-t}}$ e encontre o instante
        do \omterm{def:g12:deriv:inflection}{ponto de inflexão} $y = \frac12$ --- a data do
        epidemiologista do \cref{pb:g12:deriv:1}, agora
        calculável.
  \item Uma gota de chuva obedece a $v' = 10 - \frac v2$
        (gravidade menos arrasto), com $v(0) = 0$. Encontre a
        velocidade terminal, resolva em $v(t)$ e calcule quando a
        gota atinge $95\,\%$ da velocidade terminal.
  \item Um soro hospitalar libera um medicamento a taxa constante
        enquanto o organismo o elimina proporcionalmente:
        $c' = 4 - 0.5c$, com $c(0) = 0$. Encontre a concentração
        de regime e o tempo para alcançar metade dela. Enuncie em
        uma frase o paralelo com as recorrências de empréstimo
        do \cref{exo:g12:diffeq:9}.
  \item Final --- uma \omterm{def:g10:algebra:equation}{equação}, quatro trajes: decaimento,
        resfriamento, queda, dosagem, todos $y' = ay + b$ com
        sinais e nomes diferentes. Recite o ciclo de modelagem
        que este problema percorreu quatro vezes (lei $\to$
        \omterm{def:g10:algebra:equation}{equação} $\to$ solução $\to$ calibração $\to$ previsão
        $\to$ auditoria) e nomeie o fenômeno que exige $y''$ ---
        e que problema já o encontrou.
\end{enumerate}
\end{problem}
