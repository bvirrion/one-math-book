\chapter{Análise combinatória e contagem}\label{ch:g12:comb}

A análise combinatória é a arte de contar sem listar. Seus dois princípios
elementares --- somar os tamanhos de alternativas disjuntas, multiplicar os
números de escolhas independentes --- bastam para contar os \omterm{def:g12:comb:tuples}{arranjos}, as
\omterm{def:g12:comb:permutation}{permutações} e os subconjuntos de um conjunto finito, e culminam no teorema
binomial.

\section{Os dois princípios de contagem}

Escrevemos $\abs{E}$ para o número de elementos (a \emph{cardinalidade}) de
um conjunto finito $E$.

\begin{proposition}[Princípio aditivo]\label{prop:g12:comb:addition}
Se um conjunto finito $E$ é repartido em subconjuntos $A_1, \dots, A_k$
(dois a dois disjuntos, com \omterm{def:g10:numbers:interunion}{união} $E$), então
\[
\abs{E} = \abs{A_1} + \abs{A_2} + \dots + \abs{A_k}.
\]
\end{proposition}

\begin{proposition}[Princípio multiplicativo]\label{prop:g12:comb:multiplication}
Se um objeto é construído por uma sucessão de $k$ escolhas, com $n_1$
opções para a primeira e, \emph{quaisquer que tenham sido as escolhas
anteriores}, $n_i$ opções para a $i$-ésima, então o número de objetos
construídos é $n_1 \times n_2 \times \dots \times n_k$.
\end{proposition}

\begin{proof}
Os dois enunciados se demonstram por indução em $k$; o caso $k = 2$ do
segundo equivale a contar por linhas um quadro retangular.
\end{proof}

\begin{example}
Um restaurante oferece 4 entradas, 6 pratos principais e 3 sobremesas:
$4 \times 6 \times 3 = 72$ refeições diferentes de três pratos.
\end{example}

\section{Sequências, permutações, fatoriais}

\begin{definition}[Sequências de $k$ elementos]\label{def:g12:comb:tuples}
Uma \emph{\omterm{def:g12:seq:sequence}{sequência} de $k$ elementos} de um conjunto $E$ é uma lista
ordenada $(x_1, \dots, x_k)$ de elementos de $E$, com repetições
permitidas. Uma \omterm{def:g12:seq:sequence}{sequência} de $k$ elementos \emph{distintos} é um
\emph{arranjo}\index{arranjo} de $k$ elementos de $E$.
\end{definition}

\begin{proposition}\label{prop:g12:comb:tuples}
Seja $\abs E = n$. O número de \omterm{def:g12:seq:sequence}{sequências} de $k$ elementos de $E$ é $n^k$.
O número de \omterm{def:g12:comb:tuples}{arranjos} de $k$ elementos de $E$ ($0 \leq k \leq n$) é
\[
n(n-1)(n-2)\cdots(n-k+1) = \frac{n!}{(n-k)!},
\]
em que $n! = 1 \times 2 \times \dots \times n$ (e $0! = 1$) é o
\emph{fatorial}\index{fatorial} de $n$.
\end{proposition}

\begin{proof}
Princípio multiplicativo: para uma \omterm{def:g12:seq:sequence}{sequência} há $n$ opções em cada um dos
$k$ passos; para um \omterm{def:g12:comb:tuples}{arranjo}, $n$ opções para $x_1$, depois $n - 1$ para
$x_2$ (um elemento já foi usado), \dots, $n - k + 1$ para $x_k$.
\end{proof}

\begin{definition}[Permutação]\label{def:g12:comb:permutation}
Uma \emph{permutação}\index{permutação} de $E$ é um \omterm{def:g12:comb:tuples}{arranjo} de todos os $n$
elementos de $E$: uma ordenação de $E$. Pela
\cref{prop:g12:comb:tuples} (caso $k = n$), o número de permutações de um
conjunto de $n$ elementos é $n!$.
\end{definition}

\begin{example}
Cinco corredores podem terminar uma prova em $5! = 120$ ordens diferentes.
O número de pódios possíveis (três primeiros lugares) é
$5 \times 4 \times 3 = 60$.
\end{example}

\section{Combinações e coeficientes binomiais}

\begin{definition}[Combinações]\label{def:g12:comb:combination}
Uma \emph{combinação}\index{combinação} de $k$ elementos de $E$ é um
subconjunto de $E$ com $k$ elementos (sem ordem, sem repetição). Seu número
escreve-se $\dbinom{n}{k}$\index{coeficiente binomial}, lido ``$n$ escolhe
$k$''.
\end{definition}

\begin{theorem}\label{thm:g12:comb:binomformula}
Para $0 \leq k \leq n$:
\[
\binom{n}{k} = \frac{n!}{k!\,(n-k)!} .
\]
\end{theorem}

\begin{proof}
Conte de duas maneiras os \omterm{def:g12:comb:tuples}{arranjos} de $k$ elementos de $E$. Diretamente:
$\frac{n!}{(n-k)!}$. Alternativamente, escolha primeiro o subconjunto
subjacente ($\binom nk$ maneiras) e depois ordene-o ($k!$ maneiras); o
princípio multiplicativo dá $\binom{n}{k}\,k!$. Igualando,
$\binom nk = \frac{n!}{k!(n-k)!}$.
\end{proof}

\begin{proposition}[Identidades básicas]\label{prop:g12:comb:identities}
Para $0 \leq k \leq n$:
\[
\binom{n}{0} = \binom{n}{n} = 1, \qquad
\binom{n}{1} = n, \qquad
\binom{n}{k} = \binom{n}{n-k},
\]
e a \emph{relação de Pascal}\index{triângulo de Pascal}: para
$1 \leq k \leq n-1$,
\[
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}.
\]
\end{proposition}

\begin{proof}
A simetria $\binom nk = \binom{n}{n-k}$ vale porque tomar \omterm{def:g10:proba:operations}{complementares}
casa um a um os subconjuntos de $k$ elementos com os de $n-k$ elementos.
Para a \omterm{prop:g12:comb:identities}{relação de Pascal}, fixe um elemento $a \in E$ e separe os
subconjuntos de $k$ elementos entre os que contêm $a$ --- obtidos juntando
$a$ a um subconjunto de $k-1$ elementos de $E \setminus \{a\}$, dos quais
há $\binom{n-1}{k-1}$ --- e os que evitam $a$, que são os subconjuntos de
$k$ elementos de $E \setminus \{a\}$, em número de $\binom{n-1}{k}$.
Conclua pelo princípio aditivo.
\end{proof}

A \omterm{prop:g12:comb:identities}{relação de Pascal} gera os coeficientes linha a linha --- o
\emph{\omterm{prop:g11:binom:pascal}{triângulo de Pascal}}: cada entrada é a soma das duas acima dela.

\begin{theorem}[Teorema binomial]\label{thm:g12:comb:binomial}
\index{teorema binomial}
Para todos $a, b \in \R$ (ou $\C$) e $n \in \N$:
\[
(a+b)^n = \sum_{k=0}^{n} \binom{n}{k}\, a^{k}\, b^{\,n-k} .
\]
\end{theorem}

\begin{proof}
Expanda o produto $(a+b)(a+b)\cdots(a+b)$ ($n$ fatores): cada termo da
expansão escolhe $a$ ou $b$ em cada fator, produzindo $a^k b^{n-k}$, em que
$k$ é o número de fatores que contribuíram com $a$. O número de maneiras de
escolher esses $k$ fatores entre $n$ é $\binom nk$, que é, portanto, o
coeficiente de $a^k b^{n-k}$.
\end{proof}

\begin{corollary}\label{cor:g12:comb:sums}
$\displaystyle\sum_{k=0}^{n} \binom{n}{k} = 2^n$
\quad e \quad
$\displaystyle\sum_{k=0}^{n} (-1)^k\binom{n}{k} = 0$ \ ($n \geq 1$).
\end{corollary}

\begin{proof}
Tome $a = b = 1$ e depois $a = -1$, $b = 1$ no teorema binomial. A primeira
identidade também tem sentido direto: um conjunto de $n$ elementos tem
$2^n$ subconjuntos (cada elemento entra ou não: princípio multiplicativo),
separados por tamanho.
\end{proof}

\begin{method}[Escolher o modelo certo]\label{met:g12:comb:model}
Antes de contar, responda a duas perguntas: \emph{a ordem importa?} e
\emph{são permitidas repetições?}
\begin{center}
\begin{tabular}{l|cc}
 & a ordem importa & a ordem não importa\\
\hline
com repetição & $n^k$ (\omterm{def:g12:seq:sequence}{sequências}) & (graduação)\\
sem repetição & $\frac{n!}{(n-k)!}$ (\omterm{def:g12:comb:tuples}{arranjos}) & $\binom nk$ (subconjuntos)
\end{tabular}
\end{center}
Retirar bolas de uma urna: \emph{com reposição, em ordem} $\to$ \omterm{def:g12:seq:sequence}{sequências};
\emph{sem reposição, em ordem} $\to$ \omterm{def:g12:comb:tuples}{arranjos}; \emph{um punhado de uma vez}
$\to$ subconjuntos.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:comb:1}
Uma placa de veículo é formada por 2 letras (A--Z), depois 3 algarismos e
depois 2 letras. Quantas placas são possíveis? Quantas não têm nenhum
caractere repetido?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:2}
Calcule $\dbinom{8}{3}$, $\dbinom{10}{8}$ e simplifique
$\dfrac{\binom{n}{2}}{\binom{n+1}{2}}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:3}
Em uma turma de 30 alunos, deve-se eleger uma comissão de 4 alunos e,
dentro dela, um presidente e um tesoureiro (uma mesma pessoa não pode
ocupar os dois cargos). Quantos resultados são possíveis?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:comb:4}
Expanda $(x + 2)^5$ e $(1 - x)^6$ usando o teorema binomial. Qual é o
coeficiente de $x^3$ em $(2x + 3)^7$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:5}
Uma mão de pôquer padrão é formada por 5 cartas de um baralho de 52 cartas.
\begin{enumerate}
  \item Quantas mãos existem?
  \item Quantas mãos contêm exatamente um ás? E pelo menos um ás?
  \item Quantas mãos são ``full houses'' (três cartas de um valor e duas de
        outro)?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:6}
Quantos anagramas (rearranjos das letras, com ou sem sentido) tem a palavra
MATE? E a palavra BANANA? (Sugestão para BANANA: coloque primeiro os três
A.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:7}
Demonstre a identidade $k\dbinom{n}{k} = n\dbinom{n-1}{k-1}$
($1 \leq k \leq n$) de duas maneiras: pela fórmula com \omterm{prop:g12:comb:tuples}{fatoriais} e contando
de duas maneiras os pares (comissão de $k$ pessoas, seu presidente)
escolhidos entre $n$ pessoas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:comb:8}
Um caminho no plano vai de $(0,0)$ até $(m, n)$ por passos unitários para
leste ou para o norte. Mostre que o número desses caminhos é
$\dbinom{m+n}{m}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:comb:9}
Demonstre a \emph{identidade de Vandermonde}: para $0 \leq k \leq m + n$,
\[
\binom{m+n}{k} = \sum_{j=0}^{k} \binom{m}{j}\binom{n}{k-j},
\]
contando os subconjuntos de $k$ elementos de um conjunto dividido em um
grupo de $m$ e um grupo de $n$. Deduza que
$\displaystyle\sum_{j=0}^{n}\binom{n}{j}^{\!2} = \binom{2n}{n}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:comb:10}
Usando o teorema binomial, mostre que, para todo $n \geq 1$,
\[
\sum_{k=1}^{n} k \binom{n}{k} = n\,2^{n-1}.
\]
(Sugestão: derive $(1+x)^n$ ou use o \cref{exo:g12:comb:7}.)
\end{exercise}

\section{Problema: A arte de contar duas vezes}

\begin{problem}[{Problema de fim de semana --- estrelas e barras,
chapéus desarranjados e identidades demonstradas contando uma mesma
coisa de dois modos}]
\label{pb:g12:comb:1}
O truque mais profundo da combinatória é desarmadoramente simples:
conte a mesma coleção duas vezes, por dois métodos diferentes, e
iguale as respostas. Este problema pratica os modelos do
\cref{met:g12:comb:model}, acrescenta uma técnica de que o curso do
capítulo não precisou --- as \emph{estrelas e barras} da contagem
de sorvetes ---, conta em seguida os famosos \emph{chapéus
desarranjados} de modo exato e encontra o número $\frac1\eu$ à
espera no fundo da pilha de chapéus, sua terceira aparição neste
livro.

\medskip
\noindent\textbf{Parte I --- Escolher o modelo.}
\begin{enumerate}
  \item Conte as placas formadas por $2$ letras seguidas de $3$
        algarismos; depois os anagramas de BANANA.
  \item De um baralho de $32$ cartas, conte as mãos de $5$
        cartas; depois as mãos que contêm exatamente $2$ dos $4$
        ases.
  \item Um robô caminha de $(0,0)$ até $(4,3)$ usando apenas
        passos unitários para a direita ou para cima: quantos
        caminhos? (Codifique um caminho como uma palavra em D e
        C.)
  \item Expanda $(1 + x)^4$ pelo teorema binomial
        (\cref{thm:g12:comb:binomial}); depois calcule em
        $x = 1$ e $x = -1$: que duas identidades sobre os números
        $\binom nk$ caem daí?
  \item Demonstre por contagem dupla que
        $k\binom nk = n\binom{n-1}{k-1}$ (conte
        comissões-com-presidente de dois modos) e deduza
        $\sum_{k=0}^{n} k\binom nk = n\,2^{n-1}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Estrelas e barras.}
\begin{enumerate}[resume]
  \item Uma sorveteria vende $4$ sabores; você pede $10$ bolas
        (os sabores podem se repetir, e a ordem na taça é
        irrelevante). Codifique um pedido como uma fileira de
        $10$ estrelas (bolas) separadas por $3$ barras (trocas de
        sabor) e conte os pedidos.
  \item Conte as ternas de inteiros não negativos com
        $x + y + z = 12$.
  \item Conte as ternas de inteiros \emph{positivos} com
        $x + y + z = 12$ (substitua $x = 1 + x'$ etc.).
  \item Quantos monômios distintos aparecem na expansão de
        $(a + b + c)^5$?
  \item Teste de sanidade do método: conte pela fórmula os
        pedidos de $3$ bolas com $2$ sabores, depois liste todos
        e compare.
  \item Diga exatamente onde ``as bolas são idênticas'' entrou na
        codificação --- e conte o que acontece se as bolas forem
        comidas em ordem (posições distintas), com a lista de
        verificação do \cref{met:g12:comb:model}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Os chapéus desarranjados.}
Um \emph{desarranjo} é uma redistribuição de $n$ chapéus aos seus
$n$ donos em que \emph{ninguém} recebe o próprio chapéu; seja
$D_n$ o número deles. (O \cref{pb:g11:prob:1} mostrou que, em
\omterm{def:g11:stat:mean}{média}, um convidado recupera o próprio chapéu --- agora contamos
exatamente as festas totalmente azaradas.)
\begin{enumerate}[resume]
  \item Calcule $D_1$, $D_2$, $D_3$ listando, e $D_4$ com
        paciência (ou com esperteza).
  \item Justifique a recorrência
        $D_n = (n - 1)\left(D_{n-1} + D_{n-2}\right)$: o
        convidado 1 recebe algum chapéu $k \neq 1$ ($n - 1$
        escolhas); separe conforme o convidado $k$ receba ou não
        o chapéu 1. Verifique que ela reproduz $D_4$ e calcule
        $D_5$.
  \item Para $n = 3$, demonstre por inclusão e exclusão
        (subtraia as distribuições que fixam pelo menos um
        chapéu e devolva as contagens excedentes) que
        $D_3 = 3!\left(1 - \frac{1}{1!} + \frac{1}{2!} -
        \frac{1}{3!}\right)$, e enuncie a fórmula geral.
  \item Calcule $\frac{D_5}{5!}$ e compare com
        $\frac1\eu \approx 0.3679$: a \omterm{def:g10:proba:distribution}{probabilidade} de uma festa
        grande se desarranjar por completo é $\frac1\eu$ --- a
        terceira participação dessa constante, depois da loteria
        e da secretária do \cref{pb:g12:exp:1}. (Motivo: a
        fórmula da questão 14 é o começo de uma série famosa para
        $\eu^{-1}$, contada nos volumes de graduação.)
  \item Amigo secreto entre $10$ amigos: os nomes são sorteados
        de modo uniforme. Qual é a \omterm{def:g10:proba:distribution}{probabilidade} de o sorteio ser
        válido (ninguém tirar a si mesmo) e quantos re-sorteios o
        grupo deve esperar?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Contar duas vezes, ganhar duas
vezes.}
\begin{enumerate}[resume]
  \item O lema dos apertos de mão: em qualquer festa, somar sobre
        os convidados o número de mãos que cada um apertou conta
        cada aperto exatamente duas vezes. Deduza que \emph{o
        número de convidados que apertaram um número ímpar de
        mãos é sempre par} --- e verifique que a afirmação faz
        sentido numa festa de três convidados.
  \item Demonstre por indução a joia
        $1^3 + 2^3 + \dots + n^3 = (1 + 2 + \dots + n)^2$ e
        verifique-a para $n = 3$. (A soma do pequeno Gauss, ao
        quadrado, conta cubos.)
  \item A identidade de Vandermonde (\cref{exo:g12:comb:9}) por
        caminhos: interprete $\binom{2n}{n}$ como caminhos
        reticulados do tipo da questão 3 de $(0,0)$ até $(n,n)$,
        corte cada caminho no cruzamento da antidiagonal e
        explique como $\sum_j \binom nj^2$ aparece.
  \item Final --- os quatro movimentos do contador, uma linha
        para cada com um exemplo deste problema: multiplicar
        etapas e somar casos; codificar com esperteza (estrelas e
        barras, palavras de caminho); contar a mesma coisa duas
        vezes (comissão com presidente, apertos de mão);
        subtrair o indesejado e corrigir os excessos
        (desarranjos). E note para onde a contagem vai trabalhar
        em seguida: a \omterm{def:g10:proba:distribution}{probabilidade} e os caminhos do capítulo de
        matrizes e grafos.
\end{enumerate}
\end{problem}
