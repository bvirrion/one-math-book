\chapter{Probabilidade condicional e independência}\label{ch:g12:condprob}

A \omterm{def:g10:proba:distribution}{probabilidade} quantifica a incerteza; a \omterm{def:g12:condprob:cond}{probabilidade condicional}
quantifica como a informação a modifica. Saber que um \omterm{def:g11:prob:model}{evento} $B$ ocorreu
remodela as \omterm{def:g10:proba:distribution}{probabilidades} de todos os outros \omterm{def:g11:prob:model}{eventos} --- um mecanismo
formalizado por Bayes e mal empregado diariamente em tribunais e jornais.
Este capítulo estabelece as regras do condicionamento e o sentido exato da
\omterm{def:g12:condprob:indep}{independência}.

\section{Espaços de probabilidade (lembrete)}

Um experimento com um número finito de resultados é modelado por um
\emph{\omterm{def:g11:prob:model}{espaço amostral}} $\Omega$ (o conjunto dos resultados) e uma
\omterm{def:g10:proba:distribution}{probabilidade} $\P$ que atribui a cada \omterm{def:g11:prob:model}{evento} $A \subseteq \Omega$ um
número $\P(A) \in \intcc{0}{1}$, aditivo sobre \omterm{def:g10:numbers:interunion}{uniões} disjuntas e com
$\P(\Omega) = 1$. Lembre as regras básicas:
\[
\P(\bar A) = 1 - \P(A),
\qquad
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B).
\]
Quando todos os resultados são igualmente prováveis,
$\P(A) = \frac{\abs A}{\abs\Omega}$ --- e calcular \omterm{def:g10:proba:distribution}{probabilidades} se reduz
às técnicas de contagem do \cref{ch:g12:comb}.

\section{Probabilidade condicional}

\begin{definition}[Probabilidade condicional]\label{def:g12:condprob:cond}
Seja $B$ um \omterm{def:g11:prob:model}{evento} com $\P(B) > 0$. A \emph{probabilidade de $A$ dado
$B$}\index{probabilidade condicional} é
\[
\pcond{B}{A} = \frac{\P(A \cap B)}{\P(B)} .
\]
\end{definition}

A aplicação $A \mapsto \pcond BA$ é ela mesma uma \omterm{def:g10:proba:distribution}{probabilidade} (valem
todas as regras); ela representa o novo estado de conhecimento de quem
soube que $B$ ocorreu.

\begin{proposition}[Regra do produto]\label{prop:g12:condprob:product}
Para \omterm{def:g11:prob:model}{eventos} de \omterm{def:g10:proba:distribution}{probabilidade} não nula:
\[
\P(A \cap B) = \P(B)\,\pcond{B}{A} = \P(A)\,\pcond{A}{B},
\]
e, mais geralmente,
$\P(A_1 \cap A_2 \cap A_3) = \P(A_1)\,\pcond{A_1}{A_2}\,
\pcond{A_1 \cap A_2}{A_3}$, e assim por diante.
\end{proposition}

\begin{proof}
Reorganize a definição; a fórmula em cadeia decorre da iteração.
\end{proof}

\begin{theorem}[Lei da probabilidade total]\label{thm:g12:condprob:total}
\index{lei da probabilidade total}
Sejam $B_1, \dots, B_n$ uma partição de $\Omega$ em \omterm{def:g11:prob:model}{eventos} de
\omterm{def:g10:proba:distribution}{probabilidade} não nula. Para todo \omterm{def:g11:prob:model}{evento} $A$:
\[
\P(A) = \sum_{i=1}^{n} \P(B_i)\,\pcond{B_i}{A}.
\]
\end{theorem}

\begin{proof}
Os conjuntos $A \cap B_i$ são dois a dois disjuntos e têm \omterm{def:g10:numbers:interunion}{união} $A$, de
modo que $\P(A) = \sum_i \P(A \cap B_i) = \sum_i \P(B_i)\pcond{B_i}{A}$
pela regra do produto.
\end{proof}

\begin{method}[Árvores de probabilidade]\label{met:g12:condprob:tree}
Um \omterm{met:g10:proba:tree}{diagrama de árvore} organiza as \omterm{def:g10:proba:distribution}{probabilidades} condicionais: cada ramo
carrega a \omterm{def:g10:proba:distribution}{probabilidade} do \omterm{def:g11:prob:model}{evento} seguinte \emph{dado} o caminho
percorrido até ali.
\begin{itemize}
  \item A \omterm{def:g10:proba:distribution}{probabilidade} de uma folha (um caminho completo) é o
        \emph{produto} das \omterm{def:g10:proba:distribution}{probabilidades} ao longo de seus ramos (regra do
        produto).
  \item A \omterm{def:g10:proba:distribution}{probabilidade} de um \omterm{def:g11:prob:model}{evento} é a \emph{soma} das \omterm{def:g10:proba:distribution}{probabilidades}
        das folhas que o realizam (\omterm{def:g10:proba:distribution}{probabilidade} total).
  \item As \omterm{def:g10:proba:distribution}{probabilidades} dos ramos que saem de um mesmo nó somam $1$.
\end{itemize}
\end{method}

\begin{theorem}[Fórmula de Bayes]\label{thm:g12:condprob:bayes}
\index{fórmula de Bayes}
Sejam $B_1, \dots, B_n$ uma partição de $\Omega$ como acima e $A$ um
\omterm{def:g11:prob:model}{evento} com $\P(A) > 0$. Então
\[
\pcond{A}{B_j} =
\frac{\P(B_j)\,\pcond{B_j}{A}}{\sum_{i=1}^{n} \P(B_i)\,\pcond{B_i}{A}} .
\]
\end{theorem}

\begin{proof}
$\pcond{A}{B_j} = \frac{\P(A \cap B_j)}{\P(A)}$; desenvolva o numerador
pela regra do produto e o denominador pela \omterm{def:g10:proba:distribution}{probabilidade} total.
\end{proof}

\begin{example}[Teste de triagem]\label{ex:g12:condprob:test}
Uma doença atinge $1\%$ de uma população. Um teste a detecta com
\omterm{def:g10:proba:distribution}{probabilidade} $0.99$ (sensibilidade) e dá um falso positivo com
\omterm{def:g10:proba:distribution}{probabilidade} $0.05$. Dado um teste positivo, a \omterm{def:g10:proba:distribution}{probabilidade} de realmente
ter a doença é
\[
\frac{0.01 \times 0.99}{0.01\times0.99 + 0.99\times0.05}
= \frac{0.0099}{0.0099 + 0.0495} = \frac{1}{6} \approx 0.17 .
\]
Apesar da precisão do teste, cinco em cada seis positivos são falsos ---
porque a doença é rara. Confundir $\pcond{A}{B}$ com $\pcond{B}{A}$ é a
\emph{falácia do promotor}.
\end{example}

\section{Independência}

\begin{definition}[Eventos independentes]\label{def:g12:condprob:indep}
Dois \omterm{def:g11:prob:model}{eventos} $A$ e $B$ são \emph{independentes}\index{independência} se
\[
\P(A \cap B) = \P(A)\,\P(B) .
\]
Quando $\P(B) > 0$, isso equivale a $\pcond{B}{A} = \P(A)$: saber que $B$
ocorreu não altera a \omterm{def:g10:proba:distribution}{probabilidade} de $A$.
\end{definition}

\begin{proposition}\label{prop:g12:condprob:indepcompl}
Se $A$ e $B$ são \omterm{def:g12:condprob:indep}{independentes}, também o são $A$ e $\bar B$ (e $\bar A$ e
$\bar B$).
\end{proposition}

\begin{proof}
$\P(A \cap \bar B) = \P(A) - \P(A \cap B) = \P(A) - \P(A)\P(B)
= \P(A)\bigl(1 - \P(B)\bigr) = \P(A)\,\P(\bar B)$.
\end{proof}

\begin{remark}
Não confunda \emph{\omterm{def:g12:condprob:indep}{independentes}} ($\P(A\cap B) = \P(A)\P(B)$) com
\emph{\omterm{def:g10:proba:operations}{incompatíveis}} ($A \cap B = \varnothing$). Dois \omterm{def:g11:prob:model}{eventos}
\omterm{def:g10:proba:operations}{incompatíveis} de \omterm{def:g10:proba:distribution}{probabilidade} não nula nunca são \omterm{def:g12:condprob:indep}{independentes}: saber que
um ocorreu garante que o outro não ocorreu.
\end{remark}

\begin{definition}[Repetições independentes]\label{def:g12:condprob:repet}
Quando um experimento é repetido $n$ vezes de modo que o resultado de cada
tentativa não influencie os demais, a \omterm{def:g10:proba:distribution}{probabilidade} de uma \omterm{def:g12:seq:sequence}{sequência}
determinada de resultados é o produto das \omterm{def:g10:proba:distribution}{probabilidades} individuais. Esse
é o modelo por trás da \omterm{def:g11:prob:rv}{distribuição} binomial (\cref{ch:g12:randvar}).
\end{definition}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g12:condprob:1}
Uma carta é retirada de um baralho de 52 cartas. Calcule a \omterm{def:g10:proba:distribution}{probabilidade}
de que seja um rei, sabendo que é uma figura (valete, dama ou rei). Os
\omterm{def:g11:prob:model}{eventos} ``rei'' e ``copas'' são \omterm{def:g12:condprob:indep}{independentes}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:condprob:2}
Uma urna contém 5 bolas vermelhas e 3 azuis. Duas bolas são retiradas em
\omterm{def:g12:seq:sequence}{sequência}, \emph{sem} reposição.
\begin{enumerate}
  \item Desenhe a \omterm{met:g10:proba:tree}{árvore} de \omterm{def:g10:proba:distribution}{probabilidades}.
  \item Calcule a \omterm{def:g10:proba:distribution}{probabilidade} de que as duas sejam vermelhas e a
        \omterm{def:g10:proba:distribution}{probabilidade} de que a segunda seja vermelha.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g12:condprob:3}
Dois dados honestos são lançados. Sejam $A$ = ``a soma é $7$'',
$B$ = ``o primeiro dado mostra $3$'', $C$ = ``a soma é $6$''. Determine se
$A$ e $B$ são \omterm{def:g12:condprob:indep}{independentes} e, em seguida, se $B$ e $C$ o são.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:4}
Uma fábrica tem três máquinas que produzem, respectivamente, $50\%$,
$30\%$ e $20\%$ da produção total, com taxas de defeito de $1\%$, $2\%$ e
$4\%$.
\begin{enumerate}
  \item Que proporção da produção é defeituosa?
  \item Um item escolhido ao acaso é defeituoso. Qual é a \omterm{def:g10:proba:distribution}{probabilidade} de
        que venha da terceira máquina?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:5}
No \cref{ex:g12:condprob:test}, para que prevalência $p$ da doença (em vez
de $1\%$) um teste positivo significaria pelo menos $90\%$ de chance de
doença? Resolva a inequação e comente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g12:condprob:6}
Uma moeda viciada dá cara com \omterm{def:g10:proba:distribution}{probabilidade} $p \in \intoo{0}{1}$. Ela é
lançada três vezes, sendo os lançamentos \omterm{def:g12:condprob:indep}{independentes}.
\begin{enumerate}
  \item Calcule a \omterm{def:g10:proba:distribution}{probabilidade} do \omterm{def:g11:prob:model}{evento} $E$ = ``exatamente duas caras''.
  \item Para que $p$ a \omterm{def:g10:proba:distribution}{probabilidade} $\P(E)$ é máxima?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:condprob:7}
\emph{(Monty Hall.)} Um prêmio está escondido atrás de uma de três portas.
Você escolhe uma porta; o apresentador, que sabe onde está o prêmio, abre
uma das duas portas restantes, sempre revelando uma vazia (escolhendo ao
acaso se as duas estiverem vazias), e oferece a você a troca pela outra
porta fechada. Usando a fórmula de Bayes, calcule a \omterm{def:g10:proba:distribution}{probabilidade} de
ganhar se você trocar e se não trocar.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g12:condprob:8}
Um canal de informação transmite bits. Cada bit é invertido pelo ruído com
\omterm{def:g10:proba:distribution}{probabilidade} $\varepsilon = 0.1$, de maneira independente. Para proteger
um bit, ele é enviado três vezes e decodificado por maioria.
\begin{enumerate}
  \item Calcule a \omterm{def:g10:proba:distribution}{probabilidade} de o bit decodificado estar errado.
  \item A palavra recebida é $101$. Qual é a \omterm{def:g10:proba:distribution}{probabilidade} de o bit
        enviado ter sido $1$? (Suponha que $0$ e $1$ sejam enviados com
        \omterm{def:g10:proba:distribution}{probabilidades} iguais.)
\end{enumerate}
\end{exercise}

\section{Problema: o médico, o juiz e o filtro de spam}

\begin{problem}[{Problema de fim de semana --- a fórmula de Bayes
contra a falácia da taxa de base: por que um teste de $99\,\%$
pode significar $2\,\%$, por que uma coincidência de um em um
milhão não condena ninguém, e como sua caixa de entrada pensa}]
\label{pb:g12:condprob:1}
Um teste que pega $99\,\%$ dos casos volta positivo --- e a
\omterm{def:g10:proba:distribution}{probabilidade} verdadeira de doença fica abaixo de $2\,\%$. Um
perfil de DNA que coincide com uma pessoa em um milhão é
encontrado --- e o suspeito é provavelmente inocente. As duas
afirmações estão corretas, as duas já foram julgadas errado por
médicos e por tribunais, e as duas se resolvem com uma linha
deste capítulo: a fórmula de Bayes
(\cref{thm:g12:condprob:bayes}). Este problema a treina em
urnas, depois no consultório, no tribunal e na caixa de entrada.

\medskip
\noindent\textbf{Parte I --- Fluência.}
\begin{enumerate}
  \item Uma moeda honesta escolhe a urna I ($2$ vermelhas, $1$
        azul) ou a urna II ($1$ vermelha, $3$ azuis), e então
        uma bola é retirada. Calcule
        $\P(\text{vermelha})$ pela lei da \omterm{def:g10:proba:distribution}{probabilidade} total
        (\cref{thm:g12:condprob:total}).
  \item A bola é vermelha: calcule $\P(\text{urna I} \mid
        \text{vermelha})$.
  \item Um lançamento de dado: os \omterm{def:g11:prob:model}{eventos} ``par'' e
        ``$\leq 4$'' são \omterm{def:g12:condprob:indep}{independentes}
        (\cref{def:g12:condprob:indep})? E ``par'' e
        ``$\leq 3$''?
  \item Dois detectores de fumaça \omterm{def:g12:condprob:indep}{independentes} disparam, cada
        um, com \omterm{def:g10:proba:distribution}{probabilidade} $0.9$ em um incêndio. Usando a
        \cref{prop:g12:condprob:indepcompl}, calcule a
        \omterm{def:g10:proba:distribution}{probabilidade} de que ao menos um dispare.
  \item Duas cartas sem reposição: calcule
        $\P(\text{dois reis})$ com a regra do produto
        (\cref{prop:g12:condprob:product}).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O médico.}
Uma doença atinge $1$ pessoa em $1\,000$. O teste detecta
$99\,\%$ dos doentes (sensibilidade), mas também dispara em
$5\,\%$ dos saudáveis (falsos positivos).
\begin{enumerate}[resume]
  \item \omterm{def:g10:stats:series}{Frequências} naturais: imagine $100\,000$ pessoas.
        Quantas estão doentes, quantas saudáveis? Quantos
        positivos cada grupo produz, e quantos positivos ao
        todo?
  \item A partir das contagens, calcule
        $\P(\text{doente} \mid \text{positivo})$. Absorva o
        choque.
  \item Recalcule com a fórmula de Bayes e confira a
        concordância.
  \item Em pesquisas, a maioria dos médicos responde ``uns
        $99\,\%$''. Nomeie as duas \omterm{def:g10:proba:distribution}{probabilidades} condicionais
        que estão sendo confundidas e explique por que a
        minúscula taxa de base determina a resposta correta.
  \item Um segundo teste, independente, também volta positivo.
        Atualize: tomando $1.9\,\%$ como nova \omterm{def:g10:proba:distribution}{probabilidade} a
        priori, calcule
        $\P(\text{doente} \mid \text{dois positivos})$. O que o
        \emph{acúmulo} de evidências está fazendo?
  \item Calcule $\P(\text{doente} \mid \text{positivo})$ se o
        mesmo teste for usado apenas em um grupo de risco com
        prevalência de $5\,\%$. Deduza a moral de saúde pública
        sobre a triagem em massa de condições raras.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O juiz.}
\begin{enumerate}[resume]
  \item Um perfil de DNA da cena do crime coincide com uma
        pessoa em um milhão. O suspeito, encontrado por uma
        busca em um banco de dados dos $10$ milhões de
        habitantes da cidade, coincide. Número esperado de
        coincidências inocentes na cidade? Dada apenas a
        coincidência, quanto vale, aproximadamente,
        $\P(\text{inocente} \mid \text{coincidência})$?
  \item Nomeie a falácia --- que \omterm{def:g10:proba:distribution}{probabilidade} o promotor citou
        e qual delas o tribunal precisava? --- e seu paralelo
        exato com a questão 9.
  \item A armadilha da \omterm{def:g12:condprob:indep}{independência}: duas tragédias raras em
        uma mesma família já foram elevadas ao quadrado até
        ``uma em $73$ milhões'' por serem tratadas como
        \omterm{def:g12:condprob:indep}{independentes}, condenando uma mãe inocente (o caso
        Sally Clark). Se os \omterm{def:g11:prob:model}{eventos} compartilham fatores de
        risco familiares, de modo que o segundo, dado o
        primeiro, seja $5$ vezes mais provável que a taxa de
        base, recalcule a ordem de grandeza --- e enuncie a
        lição sobre multiplicar \omterm{def:g10:proba:distribution}{probabilidades}.
  \item A defesa também pode trapacear: ``dez inocentes
        coincidem, logo a coincidência não prova nada'' ignora
        todo o resto. O que a fórmula de Bayes faz com
        \emph{várias} peças de evidência, e que frase única
        todo jurado deveria saber?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O filtro de spam.}
\begin{enumerate}[resume]
  \item Seu correio é $40\,\%$ spam. A palavra ``sorteio''
        aparece em $15\,\%$ dos spams e em $0.5\,\%$ das
        mensagens honestas. Uma mensagem a contém: calcule
        $\P(\text{spam} \mid \text{sorteio})$.
  \item Repetição corretora de erros
        (\cref{exo:g12:condprob:8} continuado): com
        \omterm{def:g10:proba:distribution}{probabilidade} de inversão $\varepsilon = 0.1$, a
        decodificação por maioria de \emph{cinco} cópias falha
        com que \omterm{def:g10:proba:distribution}{probabilidade}? Compare com o código de três
        cópias.
  \item Um bit atravessa $4$ repetidores ruidosos
        \omterm{def:g12:condprob:indep}{independentes}, cada um invertendo-o com \omterm{def:g10:proba:distribution}{probabilidade}
        $0.1$: ele chega correto exatamente quando o número de
        inversões é par. Calcule essa \omterm{def:g10:proba:distribution}{probabilidade} de duas
        maneiras: somando os termos binomiais e pela elegante
        fórmula $\frac{1 + (1 - 2\varepsilon)^4}{2}$.
  \item Os filtros de spam de verdade multiplicam as evidências
        de muitas palavras como se fossem \omterm{def:g12:condprob:indep}{independentes} (Bayes
        ``ingênuo''). Por que essa hipótese é falsa para
        palavras como ``ganhe'' e ``sorteio'' --- e por que o
        filtro funciona bem mesmo assim? (Uma ou duas frases.)
  \item Finale --- Bayes como a aritmética da crença: a priori
        $\times$ verossimilhança, renormalize, a posteriori.
        Reconte os três veredictos (consultório, tribunal,
        caixa de entrada) em uma frase cada, e depois liste as
        três armadilhas que este problema expôs: taxas de base
        esquecidas, \omterm{def:g12:condprob:indep}{independência} falsa e evidências lidas uma
        de cada vez.
\end{enumerate}
\end{problem}
