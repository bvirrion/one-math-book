\chapter{الأشكال والزوايا القائمة}\label{ch:g3:shapes}

المربعات والمستطيلات والمثلثات والدوائر: أصدقاء قدامى. ينظر إليها
هذا الفصل بعين المهندس --- فما الذي يجعل المربع مربعًا بالضبط؟ ---
ويقدّم الأداة التي تفصل في الأمر: الكوس
وزاويته القائمة.

\section{الزاوية القائمة}

\begin{definition}[الزاوية القائمة]\label{def:g3:shapes:rightangle}
\emph{الزاوية القائمة}\index{زاوية قائمة} هي زاوية ركن ورقة، أو
ركن الكوس. والمستقيمان اللذان يصنعان زاوية قائمة
\emph{متعامدان}. وفي الأشكال، تُعلَّم الزاوية القائمة بمربع صغير
في الركن.
\end{definition}

\begin{method}[التحقّق من زاوية قائمة]\label{met:g3:shapes:check}
\begin{enumerate}
  \item ضع ركن الكوس تمامًا على الزاوية المراد
        اختبارها؛
  \item وأزلق أحد حرفَي الكوس على أحد ضلعَي الزاوية؛
  \item فإن انطبق الضلع الآخر تمامًا على الحرف الآخر، \omterm{def:g3:shapes:rightangle}{فالزاوية
        قائمة} --- وإن انفتحت أوسع أو أضيق، فليست
        قائمة.
\end{enumerate}
\end{method}

\section{الأشكال وقواعدها}

\begin{definition}[المربع والمستطيل والمثلث]\label{def:g3:shapes:shapes}
\begin{itemize}
  \item \emph{المستطيل}\index{مستطيل} له $4$ أضلاع و $4$
        \omterm{def:g3:shapes:rightangle}{زوايا قائمة}؛ وضلعاه المتقابلان متساويان في الطول.
  \item و\emph{المربع}\index{مربع} مستطيل أضلاعه $4$
        متساوية كلها في الطول.
  \item و\emph{المثلث}\index{مثلث} له $3$ أضلاع؛ و\emph{المثلث
        القائم} فيه \omterm{def:g3:shapes:rightangle}{زاوية قائمة} واحدة.
  \item و\emph{الدائرة}\index{دائرة} تُرسم بالفرجار: وكل
        نقطها على المسافة نفسها (وهي \emph{نصف القطر}\index{نصف قطر}) من
        \emph{المركز}.
\end{itemize}
\end{definition}

\begin{example}[لماذا العلامات مهمة]\label{ex:g3:shapes:marks}
الشكل المرسوم مائلًا قليلًا قد \emph{يبدو} مربعًا وهو ليس كذلك.
والعلامات تقول الحقيقة: $4$ مربعات \omterm{def:g3:shapes:rightangle}{زوايا قائمة} $+$ $4$
شُرَط $=$ مربع حقيقي. فحين ترسم، ضع العلامات؛ وحين تقرأ شكلًا،
لا تثق إلا بالعلامات والقياسات المعطاة.
\end{example}

\section{الرسم على الورق المربّعات}

\begin{method}[الإنشاءات على الشبكة]\label{met:g3:shapes:grid}
الورق المربّعات يقدّم \omterm{def:g3:shapes:rightangle}{زوايا قائمة} مجانًا: فخطوطه متعامدة.
\begin{enumerate}
  \item لرسم مستطيل $5$ في $3$: اختر نقطة تقاطع،
        وعُدّ $5$ مربعات يمينًا و $3$ مربعات فوق، ثم أغلق الشكل
        على خطوط الشبكة؛
  \item وللتحقّق من تساوي الأطوال، عُدّ المربعات؛
  \item والخطوات القُطرية ترسم خطوطًا مائلة: فالسير «$1$ يمينًا
        و $1$ فوق» مرارًا يرسم خطًّا زاويته $45^\circ$.
\end{enumerate}
\end{method}

\begin{example}[برنامج إنشاء]\label{ex:g3:shapes:program}
يمكن وصف شكل بتعليمات:
«1. اُرسم مربعًا $ABCD$ طول ضلعه $4$ سم. 2. اُرسم الدائرة التي
مركزها $A$ وتمرّ بالنقطة $B$.» وباتّباع البرنامج سطرًا سطرًا،
يستطيع أي أحد أن يعيد رسم الشكل نفسه --- وكتابته تلزمك بتسمية
كل نقطة. جرّبه مع زميل: أحدكما يكتب والآخر
يرسم.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g3:shapes:1}
في الصف، جِد (بلا قياس) ثلاث \omterm{def:g3:shapes:rightangle}{زوايا قائمة} وزاوية واحدة
ليست قائمة. وكيف \emph{تتحقّق} من كل واحدة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:2}
اُرسم مستطيلًا $6$ سم في $4$ سم بالمسطرة والكوس. وعلّم
زواياه القائمة وأضلاعه المتساوية.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:3}
اُرسم مربعًا طول ضلعه $5$ سم. فكم \omterm{def:g3:shapes:rightangle}{زاوية قائمة} له؟
وكم ضلعًا متساويًا؟ وعلّم كل شيء.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:4}
على الورق المربّعات، اُرسم: مربعًا $4 \times 4$؛ ومستطيلًا
$7 \times 2$؛ ومثلثًا قائمًا ضلعا قائمته $3$ و $5$ مربعات.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:5}
صواب أم خطأ؟ اِشرح.
«كل مربع مستطيل». \quad
«كل مستطيل مربع». \quad
«يمكن أن تكون في مثلث زاويتان قائمتان». (حاول أن ترسمه!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:6}
اُرسم دائرة نصف قطرها $4$ سم. وعلّم مركزها $O$، ونقطة $A$
على الدائرة، وارسم \omterm{def:g3:shapes:shapes}{نصف القطر} $[OA]$. فكم يبلغ $OA$؟ وإن كانت $B$
نقطة أخرى من الدائرة، فكم يبلغ $OB$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:7}
اِتبع هذا البرنامج: «1. اُرسم مستطيلًا $ABCD$ فيه $AB = 6$ سم
و $BC = 3$ سم. 2. علّم النقطة $M$ في منتصف $[AB]$.
3. صِل $M$ بالنقطتين $C$ و $D$.» فإلى أي أشكال انقسم
المستطيل؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:8}
صنّف الأشكال: أيّها فيه \omterm{def:g3:shapes:rightangle}{زاوية قائمة} واحدة على الأقل؟
مربع؛ ومستطيل (غير مربع)؛ ومثلث قائم؛ ودائرة؛ وحرف
اللام مرسومًا بخطوط غليظة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:9}
على الورق المربّعات، اُرسم مربعًا مائلًا أضلاعه تسير «$2$
يمينًا و $1$ فوق» (مع الدورات الموافقة). وعُدّ المربعات لتقنع
نفسك بأن الأضلاع الأربعة متساوية في الطول.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:10}
اُكتب برنامج إنشاء (تعليمات مرقّمة، ونقاط مسمّاة)
لهذا الشكل: مربع طول ضلعه $3$ سم يجلس فوق مستطيل
$3$ سم $\times$ $2$ سم، كبيت فوق طابقه الأرضي. واختبر برنامجك
على زميل.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:11}
كم مربعًا تجد في شبكة $3 \times 3$ من المربعات
الصغيرة؟ (عُدّ مربعات $1 \times 1$، ومربعات $2 \times 2$،
ومربع $3 \times 3$.)
\end{exercise}
