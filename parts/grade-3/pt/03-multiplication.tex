\chapter{Multiplicação}\label{ch:g3:mult}

Três saquinhos de sete bolinhas: dava para somar $7 + 7 + 7$, mas há
um jeito mais rápido --- a multiplicação, a operação da adição
repetida. Este capítulo constrói as tabuadas a partir de desenhos e
depois ensina a conta armada.

\section{O que significa multiplicar}

\begin{definition}[Multiplicação]\label{def:g3:mult:def}
O \omterm{def:g2:mult:def}{produto} $3 \times 7$ quer dizer ``três vezes sete'':
\[
3 \times 7 = 7 + 7 + 7 = 21 .
\]
No desenho, $3 \times 7$ é um retângulo de pontos: $3$ fileiras de
$7$ pontos.
\end{definition}

  \end{scope}

{\small Os mesmos $21$ pontos, contados por fileiras ou por colunas:
a ordem de um \omterm{def:g2:mult:def}{produto} não importa. Saber $3 \times 7$ dá
$7 \times 3$ de graça --- metade das tabuadas a menos para
decorar!}
\end{omfigure}

\begin{proposition}[Regras que ajudam]\label{prop:g3:mult:rules}
\begin{enumerate}
  \item A ordem não importa: $a \times b = b \times a$.
  \item Multiplicar por $1$ não muda nada; multiplicar por $0$ dá
        $0$.
  \item Multiplicar por $10$ acrescenta um \omterm{def:g1:counting:zero}{zero}: $34 \times 10 =
        340$; por $100$, dois \omterm{def:g1:counting:zero}{zeros}.
  \item Uma tabuada pode ser reconstruída a partir das vizinhas:
        $6 \times 8 = 5 \times 8 + 8 = 40 + 8 = 48$.
\end{enumerate}
\end{proposition}
\admitted

\section{Multiplicação armada}

\begin{method}[Multiplicar por um número de um algarismo]\label{met:g3:mult:column}
Para calcular $234 \times 6$:
\begin{enumerate}
  \item multiplique as unidades: $4 \times 6 = 24$: escreva $4$, vai
        $2$;
  \item multiplique as dezenas e some o vai-um:
        $3 \times 6 + 2 = 20$: escreva $0$, vai $2$;
  \item multiplique as centenas e some o vai-um:
        $2 \times 6 + 2 = 14$: escreva $14$.
\end{enumerate}
\[
\begin{array}{r}
2\,3\,4 \\
\times \quad 6 \\
\hline
1\,4\,0\,4
\end{array}
\]
Estime para conferir: $234$ está perto de $200$, e
$200 \times 6 = 1\,200$: a resposta $1\,404$ é plausível.
\end{method}

\begin{example}[Decompor para multiplicar de cabeça]\label{ex:g3:mult:split}
A conta armada decompõe o número por casas, sem dizer. De cabeça,
faça o mesmo:
\[
23 \times 4 = (20 + 3) \times 4 = 20 \times 4 + 3 \times 4
= 80 + 12 = 92 .
\]
E com um vizinho amigável: $99 \times 5 = 100 \times 5 - 5 = 495$.
\end{example}

\section{A multiplicação em problemas}

\begin{example}\label{ex:g3:mult:problem}
Uma van leva $9$ passageiros. Quantos passageiros $27$ vans podem
levar?
\begin{enumerate}
  \item É uma multiplicação: $27$ grupos de $9$.
  \item $27 \times 9 = 243$ (armando a conta, ou
        $27 \times 9 = 27 \times 10 - 27 = 270 - 27$).
  \item Frase: as vans podem levar $243$ passageiros.
\end{enumerate}
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:mult:1}
Escreva como multiplicação e depois calcule:
$8 + 8 + 8 + 8$; \quad $5 + 5 + 5$; \quad
$20 + 20 + 20 + 20 + 20$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:2}
Recite e complete: $6 \times 7$; \ $8 \times 4$; \ $9 \times 9$;
\ $7 \times 8$; \ $6 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:3}
Desenhe um retângulo de pontos para $4 \times 6$ e depois um para
$6 \times 4$. O que você percebe nas duas contagens?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:4}
Calcule usando as regras da \cref{prop:g3:mult:rules}:
$56 \times 10$; \quad $8 \times 100$; \quad $73 \times 0$; \quad
$45 \times 1$; \quad $30 \times 10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:5}
Arme a conta e calcule: $132 \times 3$; \quad $217 \times 4$; \quad
$408 \times 7$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:6}
Calcule de cabeça decompondo (como no \cref{ex:g3:mult:split}):
$31 \times 5$; \quad $42 \times 3$; \quad $25 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:7}
Calcule usando um vizinho amigável: $99 \times 7$; \quad
$101 \times 6$; \quad $98 \times 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:8}
Estime primeiro e depois calcule: $389 \times 5$ (estime com $400$);
$612 \times 8$ (estime com $600$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:9}
Uma caixa de ovos leva $6$ ovos. Quantos ovos há em $38$ caixas?
Escreva a operação, calcule e responda com uma frase.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:10}
Uma sala de aula tem $8$ fileiras de $4$ carteiras; em cada carteira
sentam $2$ alunos. Quantos alunos a sala comporta? (Duas
multiplicações.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:11}
Encontre todas as maneiras de arrumar $24$ cadeiras em fileiras
iguais (fileiras de $1$, de $2$, \dots). Quais arrumações são
possíveis? (Use o quadro das tabuadas: procure o $24$ dentro dele.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:12}
Zoe calcula $47 \times 6$ assim: $40 \times 6 = 240$,
$7 \times 6 = 42$, e então $240 + 42 = 282$. Lia calcula
$47 \times 6 = 50 \times 6 - 3 \times 6 = 300 - 18 = 282$. Explique
cada método. Qual você prefere para $38 \times 4$? Calcule das duas
maneiras.
\end{exercise}
