\chapter{Medidas}\label{ch:g3:measure}

Que comprimento tem, quanto pesa, quanto cabe, que horas são? Medir
transforma o mundo em números --- desde que todo mundo use as mesmas
unidades. Este capítulo apresenta as unidades de comprimento, \omterm{def:g2:measure:units}{massa} e
\omterm{def:g2:measure:units}{capacidade}, e ensina a ler as horas no relógio.

\section{Comprimentos}

\begin{definition}[Unidades de comprimento]\label{def:g3:measure:length}
A unidade básica de comprimento é o \emph{metro} (m). Para
comprimentos pequenos: o centímetro (cm) e o milímetro (mm); para
distâncias longas: o quilômetro (km).
\[
1 \text{ m} = 100 \text{ cm}, \qquad
1 \text{ cm} = 10 \text{ mm}, \qquad
1 \text{ km} = 1\,000 \text{ m}.
\]
\end{definition}

depois do $3$.}
\end{omfigure}

\begin{example}[Escolher a unidade certa]\label{ex:g3:measure:choose}
Uma formiga: cerca de $5$ mm. Um lápis: cerca de $18$ cm. Uma porta:
cerca de $2$ m. Uma caminhada até o povoado vizinho: cerca de $3$
km. Dizer que ``o comprimento de um lápis é $18$'' não significa
nada --- a unidade faz parte da resposta.
\end{example}

\begin{method}[Converter unidades]\label{met:g3:measure:convert}
Para converter, multiplique ou divida por $10$, $100$ ou $1\,000$:
\begin{enumerate}
  \item de uma unidade grande para uma pequena, \emph{multiplique}:
        $3$ m $= 3 \times 100 = 300$ cm;
  \item de uma unidade pequena para uma grande, \emph{divida}:
        $250$ cm $= 250 \div 100 = 2$ m e $50$ cm;
  \item para somar comprimentos, converta-os antes para a
        \emph{mesma} unidade: $1$ m $+ 35$ cm $= 100 + 35 = 135$ cm.
\end{enumerate}
\end{method}

\section{Massas e capacidades}

\begin{definition}[Unidades de massa e de capacidade]\label{def:g3:measure:mass}
As \omterm{def:g2:measure:units}{massas} são medidas em \emph{gramas} (g) e \emph{quilogramas}
(kg); as \omterm{def:g2:measure:units}{capacidades} (quanto cabe num recipiente), em \emph{litros}
(L) e \emph{centilitros} (cL):
\[
1 \text{ kg} = 1\,000 \text{ g},
\qquad
1 \text{ L} = 100 \text{ cL}.
\]
\end{definition}

\begin{example}\label{ex:g3:measure:mass}
Uma carta: cerca de $20$ g. Um pacote de farinha: $1$ kg. Uma
banheira: cerca de $150$ L. Um copo: cerca de $20$ cL. Para
equilibrar uma balança com um pacote de $1$ kg de um lado e $650$ g
de maçãs do outro, acrescente $1\,000 - 650 = 350$ g do lado das
maçãs.
\end{example}

\section{Ler o relógio}

\begin{method}[Ler as horas]\label{met:g3:measure:clock}
Num relógio de ponteiros:
\begin{enumerate}
  \item o ponteiro \emph{pequeno} dá a hora: leia o último número por
        que ele passou;
  \item o ponteiro \emph{grande} dá os minutos: cada marca grande vale
        $5$ minutos, então o ponteiro grande no $3$ quer dizer $15$
        minutos;
  \item passada a meia hora, também se pode ler ``para a hora
        seguinte'': $10{:}50$ é ``dez para as onze''.
\end{enumerate}
\end{method}

\begin{example}[Durações]\label{ex:g3:measure:duration}
O filme começa às $14{:}40$ e termina às $16{:}10$. Quanto tempo ele
dura? Conte em dois pulos:
\begin{enumerate}
  \item de $14{:}40$ até $15{:}00$: $20$ minutos;
  \item de $15{:}00$ até $16{:}10$: $1$ hora e $10$ minutos.
\end{enumerate}
Total: $1$ h $30$ min. Cuidado: uma hora tem $60$ minutos, então
$16{:}10$ menos $14{:}40$ não se calcula como $1610 - 1440$!
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:measure:1}
Que unidade você usaria (mm, cm, m ou km) para: a largura de uma
unha; a altura de um colega; o comprimento do pátio da escola; a
distância entre duas cidades?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:2}
Converta: $5$ m em cm; \quad $70$ mm em cm; \quad $4$ km em m;
\quad $600$ cm em m.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:3}
Meça com a sua régua (com precisão de milímetro) os lados de um
triângulo desenhado por um colega e calcule o comprimento total do
contorno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:4}
Calcule, convertendo primeiro: $2$ m $+ 45$ cm (em cm); \quad
$1$ km $+ 250$ m (em m); \quad $3$ cm $+ 8$ mm (em mm).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:5}
Converta: $3$ kg em g; \quad $2\,500$ g em kg e g; \quad
$4$ L em cL; \quad $350$ cL em L e cL.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:6}
Uma receita pede $750$ g de farinha. Mona tem um pacote de $1$ kg.
Ela tem farinha suficiente? Quanto vai sobrar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:7}
Leia as horas: o ponteiro pequeno entre $2$ e $3$ e o grande no $6$;
o ponteiro pequeno no $9$ e o grande no $12$; o ponteiro pequeno
quase no $5$ e o grande no $10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:8}
A aula começa às $8{:}30$ e a manhã termina às $11{:}45$. Quanto
tempo dura a manhã? (Conte em pulos, como no
\cref{ex:g3:measure:duration}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:9}
O trem parte às $9{:}50$ e a viagem dura $45$ minutos. A que horas
ele chega? (Pule primeiro até $10{:}00$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measure:10}
Uma garrafa leva $1$ L. Paulo enche copos de $20$ cL. Quantos copos
cheios ele consegue servir? (Converta primeiro.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measure:11}
Lena caminha $600$ m até a escola, e a mesma coisa na volta, todos os
dias. Quantos metros ela caminha numa semana de $5$ dias de aula? Dê
a resposta em metros e depois em quilômetros.
\end{exercise}
