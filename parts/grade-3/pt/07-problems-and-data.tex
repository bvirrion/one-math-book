\chapter{Resolução de problemas}\label{ch:g3:problems}

Saber as quatro operações é uma coisa; saber \emph{qual delas} uma
história pede é outra. Este capítulo dá um método para os problemas
escritos --- os mesmos quatro passos todas as vezes --- e ensina a
ler informações em tabelas e pictogramas.

\section{Os quatro passos}

\begin{method}[Resolver um problema escrito]\label{met:g3:problems:steps}
\begin{enumerate}
  \item \emph{Leia} o problema duas vezes; diga com as suas palavras
        o que se pergunta e o que se sabe.
  \item \emph{Escolha} a operação: faça um esboço rápido ou um
        desenho de barras se a escolha não for evidente.
  \item \emph{Calcule}, armando a conta ou de cabeça, com uma
        estimativa como rede de segurança.
  \item \emph{Responda} com uma frase completa, incluindo a unidade
        --- e veja se a resposta faz sentido (uma criança não pesa
        $300$ kg).
\end{enumerate}
\end{method}

\begin{example}[Que operação?]\label{ex:g3:problems:which}
As palavras dão a pista da operação; o esboço decide:
\begin{itemize}
  \item \emph{juntar / ao todo}: adição;
  \item \emph{tirar / quantos a mais / o que falta}: subtração;
  \item \emph{tantos grupos de tantos / tantas vezes}:
        multiplicação;
  \item \emph{repartir igualmente / quantos grupos}: divisão.
\end{itemize}
\end{example}

\begin{example}[Um problema de duas etapas]\label{ex:g3:problems:twostep}
``Maia compra $3$ pacotes de $6$ garrafas e bebe $2$ garrafas.
Quantas garrafas sobram?''
\begin{enumerate}
  \item O que se pergunta: o número de garrafas que sobram. O que se
        sabe: $3$ pacotes de $6$; $2$ bebidas.
  \item Esboço: $3$ grupos de $6$ e depois tirar $2$. Duas
        operações!
  \item Cálculo: $3 \times 6 = 18$ e depois $18 - 2 = 16$.
  \item Frase: sobram $16$ garrafas.
\end{enumerate}
Nunca pare depois da primeira operação: releia a pergunta para ver
se a história já acabou.
\end{example}

\begin{example}[Um número inútil]\label{ex:g3:problems:trap}
``Um padeiro, que tem $52$ anos, assa $120$ pãezinhos e vende $85$.
Quantos pãezinhos sobram?'' A idade está ali para distrair: a
resposta só precisa de $120 - 85 = 35$ pãezinhos. Antes de calcular,
separe sempre os números: quais deles a pergunta realmente usa?
\end{example}

\section{Ler tabelas e pictogramas}

\begin{example}[Uma tabela]\label{ex:g3:problems:table}
Livros emprestados na biblioteca:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cccc}
 & gibis & romances & ciências & total \\
\hline
semana 1 & $14$ & $9$ & $6$ & $29$ \\
semana 2 & $11$ & $12$ & $8$ & $31$ \\
\end{tabular}
\end{center}
Leitura: semana 2, romances: $12$. Cálculo a partir da tabela: nas
duas semanas juntas foram emprestados $14 + 11 = 25$ gibis; a semana
mais movimentada foi a 2 ($31 > 29$).
\end{example}

\begin{example}[Um pictograma]\label{ex:g3:problems:pictogram}
Num pictograma, um símbolo vale uma quantidade fixa --- leia a
legenda primeiro!

\begin{omfigure}
\begin{tikzpicture}[scale=0.62]
  \node[left, font=\small] at (0,2.4) {segunda};
  \foreach \i in {0,1,2} \fill[omDef] (0.5+\i*0.8,2.4) circle (0.26);
  \node[left, font=\small] at (0,1.2) {terça};
  \foreach \i in {0,1,2,3,4} \fill[omDef] (0.5+\i*0.8,1.2) circle (0.26);
  \node[left, font=\small] at (0,0) {quarta};
  \foreach \i in {0,1} \fill[omDef] (0.5+\i*0.8,0) circle (0.26);
  \fill[omDef] (0.5+1.6,0) circle (0.26);
  \node[right, font=\small] at (5.4,1.2)
    {legenda: uma bolinha $= 4$ bolos vendidos};
\end{tikzpicture}

{\small Um pictograma: segunda tem $3$ bolinhas, então
$3 \times 4 = 12$ bolos; terça, $5$ bolinhas, $20$ bolos; quarta,
$3$ bolinhas, $12$ bolos.}
\end{omfigure}
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:problems:1}
Para cada história, apenas \emph{diga} a operação (não calcule):
um rebanho de $48$ ovelhas se junta a um de $37$; \ $56$ cerejas
repartidas entre $7$ crianças; \ $9$ caixas de $8$ lápis; \ Ali tinha
$71$ figurinhas e deu $18$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:2}
Resolva com os quatro passos: ``Um estacionamento tem $246$ carros de
manhã; ao meio-dia chegam mais $158$. Quantos carros há agora?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:3}
Resolva: ``Uma corda de $90$ m é cortada em pedaços de $9$ m. Quantos
pedaços?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:4}
Resolva: ``Cada ingresso custa $8$. Quanto custam $26$ ingressos?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:5}
Faça um desenho de barras e depois resolva: ``Emma tem $64$ adesivos
e Hugo tem $29$ a menos. Quantos Hugo tem?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:6}
Duas etapas: ``Um engradado leva $45$ maçãs. Um quitandeiro recebe
$6$ engradados e vende $178$ maçãs. Quantas maçãs sobram?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:7}
Usando a tabela do \cref{ex:g3:problems:table}: quantos livros de
ciências foram emprestados nas duas semanas juntas? Quantos romances
a mais na semana 2 do que na semana 1?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:8}
Usando o pictograma do \cref{ex:g3:problems:pictogram}: quantos
bolos foram vendidos nos três dias juntos? Em que dia se venderam
mais bolos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:9}
Este problema tem um número inútil --- encontre-o e depois resolva:
``Um livro de $210$ páginas custa $12$. Léo lê $35$ páginas por dia.
De quantos dias ele precisa para terminar o livro?''
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:problems:10}
``Uma família de $2$ adultos e $3$ crianças visita o museu. O
ingresso de adulto custa $9$ e o de criança, $4$. Eles pagam com uma
nota de $50$. Quanto recebem de troco?'' (Três etapas: adultos,
crianças e depois o troco.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:problems:11}
Invente o seu próprio problema de duas etapas cuja resposta seja
$100$, escreva a solução com os quatro passos e teste-o com um
colega.
\end{exercise}
