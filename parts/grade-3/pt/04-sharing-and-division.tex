\chapter{Partilha e divisão}\label{ch:g3:division}

Vinte bolinhas repartidas igualmente entre quatro crianças: quantas
para cada uma? A divisão responde a todas as perguntas de partilha
--- e também às de agrupamento --- e é a gêmea da multiplicação: uma
desfaz a outra.

\section{As duas caras da divisão}

\begin{definition}[Divisão]\label{def:g3:division:def}
A divisão $20 \div 4$ responde a dois tipos de pergunta:
\begin{itemize}
  \item \emph{repartir}: $20$ bolinhas repartidas entre $4$ crianças
        --- quantas bolinhas para \emph{cada uma}? ($5$ para cada);
  \item \emph{agrupar}: $20$ bolinhas colocadas em saquinhos de $4$
        --- quantos \emph{saquinhos}? ($5$ saquinhos).
\end{itemize}
As duas perguntas têm a mesma resposta, porque as duas desfazem a
mesma multiplicação: $4 \times 5 = 20$.
\end{definition}

\begin{method}[Dividir com as tabuadas]\label{met:g3:division:tables}
Para calcular $42 \div 6$, pergunte: ``$6$ vezes \emph{quanto} dá
$42$?'' A tabuada do $6$ responde: $6 \times 7 = 42$, logo
$42 \div 6 = 7$. Toda pergunta de divisão é uma pergunta de
multiplicação lida ao contrário.
\end{method}

\section{Quando a conta não é exata}

\begin{definition}[Resto]\label{def:g3:division:remainder}
Repartir $23$ bolinhas entre $4$ crianças dá $5$ bolinhas para cada
uma --- e $3$ bolinhas \emph{que sobram}, poucas demais para mais
uma rodada. Escrevemos
\[
23 = 4 \times 5 + 3 ,
\]
e dizemos: o \emph{quociente}\index{quociente} é $5$ e o
\emph{resto}\index{resto} é $3$. O resto é sempre menor que o número
de crianças --- senão, seria possível fazer mais uma rodada de
partilha.
\end{definition}

\end{omfigure}

\begin{example}\label{ex:g3:division:remainder}
Divida $58$ por $7$. Percorra a tabuada do $7$: $7 \times 8 = 56$ é
o maior \omterm{def:g2:mult:def}{produto} que cabe abaixo de $58$ ($7 \times 9 = 63$ passa).
Então
\[
58 = 7 \times 8 + 2 :
\]
\omterm{def:g3:division:remainder}{quociente} $8$, \omterm{def:g3:division:remainder}{resto} $2$. Conferência: $56 + 2 = 58$, e $2 < 7$.
\end{example}

\begin{method}[Interpretar o resto]\label{met:g3:division:interpret}
Depois de dividir, volte à história:
\begin{enumerate}
  \item ``quantos para cada um / quantos grupos completos?'' --- o
        \omterm{def:g3:division:remainder}{quociente};
  \item ``quantos sobram?'' --- o \omterm{def:g3:division:remainder}{resto};
  \item ``quantas caixas para guardar tudo?'' --- o \omterm{def:g3:division:remainder}{quociente}
        \emph{mais um}, se o \omterm{def:g3:division:remainder}{resto} não for \omterm{def:g1:counting:zero}{zero}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:division:interpret}
$58$ alunos vão andar de canoa, $7$ por canoa. $58 = 7 \times 8 +
2$: oito canoas ficam cheias e sobram $2$ alunos --- que também
precisam de canoa. Logo, é preciso reservar $9$ canoas.
\end{example}

\section{Metades, terças e quartas partes}

\begin{definition}[Metade, terça parte, quarta parte]\label{def:g3:division:half}
A \emph{metade}\index{metade} de um número é o resultado de dividi-lo por $2$; a
\emph{terça parte}, por $3$; a \emph{quarta parte}\index{quarta parte}, por $4$. Por
exemplo, a metade de $18$ é $9$, a terça parte de $18$ é $6$ e a
quarta parte de $36$ é $9$.
\end{definition}

\begin{example}[Dobros e metades andam juntos]\label{ex:g3:division:double}
A \omterm{def:g3:division:half}{metade} de $46$: como $46 = 40 + 6$, a \omterm{def:g3:division:half}{metade} de cada parte dá
$20 + 3 = 23$. Confira com o dobro: $23 + 23 = 46$. Saber os dobros
de cor ($25 \to 50$, $45 \to 90$, $150 \to 300$) torna as \omterm{def:g3:division:half}{metades}
instantâneas.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:division:1}
Calcule usando as tabuadas ao contrário: $36 \div 4$; \quad
$63 \div 9$; \quad $40 \div 5$; \quad $56 \div 8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:2}
Para cada divisão, escreva a igualdade $a = b \times q + r$ e dê o
\omterm{def:g3:division:remainder}{quociente} e o \omterm{def:g3:division:remainder}{resto}: $17 \div 5$; \quad $30 \div 4$; \quad
$50 \div 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:3}
$28$ bolinhas são repartidas entre $6$ crianças. Quantas bolinhas
para cada uma, e quantas sobram? Desenhe a partilha se isso ajudar.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:4}
Calcule: a \omterm{def:g3:division:half}{metade} de $68$; a terça parte de $27$; a \omterm{def:g3:division:half}{quarta parte} de
$48$; a \omterm{def:g3:division:half}{metade} de $150$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:5}
Verdadeiro ou falso? Confira com uma multiplicação.
``$45 \div 5 = 9$''; \quad ``$52 \div 6 = 8$ e \omterm{def:g3:division:remainder}{resto} $4$'';
\quad ``$70 \div 8 = 9$ e \omterm{def:g3:division:remainder}{resto} $2$''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:6}
$32$ fotos são coladas num álbum, $4$ por página. Quantas páginas são
usadas? De que cara da divisão se trata (repartir ou agrupar)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:7}
Uma corda de $75$ cm é cortada em pedaços de $9$ cm. Quantos pedaços
completos, e qual o comprimento do pedacinho que sobra?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:8}
$50$ crianças vão a um passeio em carros de $4$ lugares. Quantos
carros são necessários? (Cuidado: todo mundo tem de ir!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:9}
Descubra os números que faltam: $? \div 7 = 6$; \quad
$54 \div ? = 6$; \quad a \omterm{def:g3:division:half}{metade} de $?$ é $35$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:division:10}
Amina reparte os seus $47$ adesivos entre os $5$ amigos, o mais
igualmente possível, e fica com o que sobrar. Quantos adesivos cada
amigo recebe, e quantos Amina guarda?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:division:11}
Quando um número é dividido por $6$, quais são todos os \omterm{def:g3:division:remainder}{restos}
possíveis? Qual é o maior número cuja divisão por $6$ tem \omterm{def:g3:division:remainder}{quociente}
$7$?
\end{exercise}
