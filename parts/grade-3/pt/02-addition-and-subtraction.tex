\chapter{Adição e subtração}\label{ch:g3:addsub}

Somar é juntar; subtrair é tirar, ou descobrir o que está faltando.
Este capítulo treina as duas contas armadas --- com os seus famosos
vai-um e empréstimos --- e os truques de cálculo mental que muitas
vezes dispensam a conta armada.

\section{Adição armada}

\begin{method}[Somar em coluna]\label{met:g3:addsub:add}
\begin{enumerate}
  \item Escreva os números um embaixo do outro, unidades sob
        unidades, dezenas sob dezenas (alinhe pela \emph{direita});
  \item some as unidades; se a \omterm{def:g1:addition:def}{soma} passar de $9$, escreva o
        algarismo das unidades e \emph{leve} a dezena para a coluna
        seguinte (um $1$ pequeno);
  \item continue com as dezenas, as centenas, \dots, somando o
        vai-um a cada vez.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:addsub:add}
Calcule $457 + 368$:
\[
\begin{array}{r}
{\scriptstyle 1}\ \ {\scriptstyle 1}\ \ \phantom{0} \\[-3pt]
4\,5\,7 \\
+\ 3\,6\,8 \\
\hline
8\,2\,5
\end{array}
\]
Unidades: $7 + 8 = 15$: escreva $5$, vai $1$. Dezenas:
$5 + 6 + 1 = 12$: escreva $2$, vai $1$. Centenas: $4 + 3 + 1 = 8$.
Resultado: $825$.
\end{example}

\begin{example}[Adição mental]\label{ex:g3:addsub:mental}
Procure números amigos antes de armar a conta:
\begin{itemize}
  \item \emph{formar uma dezena}: $47 + 8 = 47 + 3 + 5 = 50 + 5 =
        55$;
  \item \emph{saltar por partes}: $256 + 130 = 256 + 100 + 30 = 386$;
  \item \emph{reordenar}: $17 + 59 + 3 = (17 + 3) + 59 = 79$.
\end{itemize}
\end{example}

\section{Subtração armada}

\begin{method}[Subtrair em coluna]\label{met:g3:addsub:sub}
\begin{enumerate}
  \item Escreva o número maior em cima, alinhado à direita;
  \item subtraia coluna por coluna, começando pelas unidades;
  \item se o algarismo de cima for pequeno demais, peça
        \emph{emprestado}: tome um da casa seguinte da linha de cima
        (ele vale dez na coluna atual);
  \item \emph{confira}: o resultado mais o número subtraído tem de
        devolver o número de cima.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:addsub:sub}
Calcule $603 - 147$:
\[
\begin{array}{r}
6\,0\,3 \\
-\ 1\,4\,7 \\
\hline
4\,5\,6
\end{array}
\]
Unidades: $3 - 7$ é impossível; peça emprestado passando adiante:
$603$ vira $5$ centenas, $9$ dezenas, $13$ unidades. Aí
$13 - 7 = 6$; dezenas $9 - 4 = 5$; centenas $5 - 1 = 4$. Resultado:
$456$. Conferência: $456 + 147 = 603$.
\end{example}

\begin{omfigure}

desmanchada em dez dezenas, e uma dessas dezenas em dez unidades. A

$6\,\vert\,0\,\vert\,3$ vira $5\,\vert\,9\,\vert\,13$, e agora toda
coluna consegue subtrair.}
\end{omfigure}

\begin{example}[A subtração como parte que falta]\label{ex:g3:addsub:missing}
``Lila tem $35$ bolinhas e Tomás tem $52$: quantas Tomás tem a
mais?'' é a subtração $52 - 35$. De cabeça, conte \emph{para cima} a
partir de $35$: de $35 \to 40$ vão $5$, depois de $40 \to 52$ vão
$12$; ao todo $5 + 12 = 17$. Contar para cima costuma ser mais fácil
que pedir emprestado.
\end{example}

  \draw[-Stealth] (-0.2,0) -- (10.4,0);
  \foreach \x/\l in {1.0/{35}, 3.5/{40}, 9.5/{52}}
    { \draw (\x,-0.08) -- (\x,0.08);
      \node[below, font=\small] at (\x,-0.1) {$\l$}; }
  \draw[-Stealth, omDef, thick] (1.0,0.25) .. controls (2.2,0.9) ..
    (3.5,0.25);
  \node[omDef, font=\small] at (2.25,0.92) {$+5$};
  \draw[-Stealth, omThm, thick] (3.5,0.25) .. controls (6.5,1.2) ..
    (9.5,0.25);
  \node[omThm, font=\small] at (6.5,1.1) {$+12$};
\end{tikzpicture}

{\small Calcular $52 - 35$ contando para cima: primeiro até o
amigável $40$, depois até $52$. A resposta é o total dos pulos:
$5 + 12 = 17$.}
\end{omfigure}

\section{Ordens de grandeza}

\begin{method}[Estimar antes de calcular]\label{met:g3:addsub:estimate}
Antes de qualquer conta armada, arredonde os números para valores
amigáveis e calcule uma \emph{estimativa}. Depois de calcular,
compare: um resultado longe da estimativa é sinal de erro.
\end{method}

\begin{example}
Para $487 + 312$: a estimativa é $500 + 300 = 800$. O resultado da
conta armada, $799$, está perto: é plausível. Se você tivesse
encontrado $1\,099$ (um alinhamento errado), a estimativa teria
avisado.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:addsub:1}
Arme a conta e calcule: $346 + 253$; \quad $478 + 265$; \quad
$1\,536 + 2\,876$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:2}
Arme a conta e depois confira com uma adição: $587 - 234$; \quad
$805 - 348$; \quad $1\,000 - 463$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:3}
Calcule de cabeça formando dezenas: $38 + 7$; \quad $56 + 9$; \quad
$45 + 28$ (some $30$ e tire $2$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:4}
Calcule de cabeça por saltos: $340 + 220$; \quad $675 + 210$; \quad
$512 - 300$; \quad $840 - 220$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:5}
Calcule $63 - 47$ contando para cima (como no
\cref{ex:g3:addsub:missing}) e depois confira armando a conta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:6}
Estime primeiro (arredondando para centenas) e depois calcule
exatamente: $492 + 218$; \quad $913 - 388$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:7}
Uma escola tem $348$ meninas e $296$ meninos. Quantos alunos ao
todo? Escreva a operação, calcule e responda com uma frase.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:8}
Um livro tem $224$ páginas; Nora já leu $178$. Quantas páginas
faltam para ela terminar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:9}
Complete os buracos (trabalhe coluna por coluna):
\[
\begin{array}{r}
3\,?\,7 \\
+\ ?\,2\,? \\
\hline
8\,6\,2
\end{array}
\]
Há uma única maneira de preencher os três buracos? Explique.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:addsub:10}
Uma padaria vendeu $145$ pães de manhã e $89$ à tarde; na hora de
fechar sobraram $17$ pães sem vender. Quantos pães foram assados
naquele dia? (Duas etapas.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:addsub:11}
Téo diz: ``para calcular $500 - 199$, eu calculo $500 - 200$ e
depois somo $1$.'' Ele tem razão? Calcule das duas maneiras e
explique o truque dele.
\end{exercise}
