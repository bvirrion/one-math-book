\chapter{Números até 10\,000}\label{ch:g3:numbers}

Mil, dois mil, três mil\,\dots\ os números grandes são construídos a
partir dos pequenos, com uma ideia formidável: a \emph{posição} de um
algarismo diz o valor dele. Este capítulo ensina a ler, escrever,
comparar e ordenar os números até $10\,000$.

\section{Ler e escrever números}

\begin{definition}[Valor posicional]\label{def:g3:numbers:place}
Um número é escrito com os algarismos $0, 1, 2, \dots, 9$. Lendo da
direita para a esquerda, as casas são: unidades, dezenas, centenas,
unidades de milhar. Em $3\,254$:
\[
3\,254 = 3 \text{ milhares} + 2 \text{ centenas} + 5 \text{ dezenas}
+ 4 \text{ unidades}.
\]
\end{definition}

\begin{example}[O zero que guarda a casa]\label{ex:g3:numbers:zero}
``Quatro mil e sete'' escreve-se $4\,007$: um $4$ nos milhares, um
$7$ nas unidades e \emph{\omterm{def:g1:counting:zero}{zeros}} no meio para manter cada algarismo na
sua casa certa. Sem eles, $47$ seria um número bem diferente!
\end{example}

\begin{example}[Trocar]\label{ex:g3:numbers:exchange}
Dez unidades formam uma dezena; dez dezenas formam uma centena; dez
centenas formam um milhar. Então $10$ centenas $= 1\,000$, e $32$
dezenas $= 320$: o número $320$ contém $32$ dezenas (e não apenas o
algarismo $2$!).
\end{example}

\section{Comparar e ordenar}

\begin{method}[Comparar dois números]\label{met:g3:numbers:compare}
\begin{enumerate}
  \item Conte os algarismos: mais algarismos significa número maior
        ($1\,002 > 998$).
  \item Com a mesma quantidade de algarismos: compare algarismo por
        algarismo, começando pela \emph{esquerda}. A primeira
        \omterm{ex:g1:subtraction:difference}{diferença} decide: $5\,614 > 5\,589$ porque $6 > 5$ nas
        centenas.
  \item Escreva a resposta com os sinais $<$ (menor que) ou $>$
        (maior que): o lado aberto sempre olha para o número maior.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:numbers:order}
Ordene do menor para o maior: $780$, $8\,700$, $807$, $87$. Primeiro
pela quantidade de algarismos: $87$ (dois), depois $780$ e $807$
(três), depois $8\,700$ (quatro). Entre $780$ e $807$: compare os
algarismos das centenas, $7 < 8$, então $780 < 807$. Ordem final:
\[
87 < 780 < 807 < 8\,700 .
\]
\end{example}

\section{Pares e ímpares}

\begin{definition}[Par, ímpar]\label{def:g3:numbers:evenodd}
Um número é \emph{par}\index{número par} quando pode ser repartido em
duas partes inteiras iguais --- o seu algarismo das unidades é $0$,
$2$, $4$, $6$ ou $8$. Caso contrário, ele é
\emph{ímpar}\index{número ímpar} (algarismo das unidades $1$, $3$,
$5$, $7$ ou $9$).
\end{definition}

\begin{example}
$46$ é par ($46 = 23 + 23$); $47$ é \omterm{def:g3:numbers:evenodd}{ímpar} (duas partes iguais
exigiriam meio objeto). Só o \emph{último} algarismo importa:
$3\,578$ é par e $2\,341$ é \omterm{def:g3:numbers:evenodd}{ímpar}.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g3:numbers:1}
Escreva em algarismos: ``dois mil trezentos e quarenta e cinco'';
``seis mil e cinquenta''; ``nove mil e um''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:2}
Escreva por extenso: $1\,208$; \quad $4\,070$; \quad $9\,999$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:3}
Em $7\,382$: qual algarismo está na casa das dezenas? E na casa dos
milhares? Quantas \emph{centenas} o número contém ao todo (cuidado:
não é só o algarismo das centenas --- pense na
\cref{ex:g3:numbers:exchange})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:4}
Decomponha como na \cref{def:g3:numbers:place}:
$5\,631$; \quad $2\,046$; \quad $8\,500$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:5}
Copie e complete com $<$ ou $>$:
\[
456 \;?\; 465, \qquad
1\,203 \;?\; 989, \qquad
6\,090 \;?\; 6\,009, \qquad
9\,999 \;?\; 10\,000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:6}
Ordene do menor para o maior: $530$; \ $3\,500$; \ $353$; \ $5\,033$;
\ $503$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:7}
Numa reta numérica graduada de cem em cem, de $0$ até $1\,000$,
marque (aproximadamente) os números $150$, $480$, $820$ e $990$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:8}
Par ou \omterm{def:g3:numbers:evenodd}{ímpar}? \ $34$; \ $57$; \ $70$; \ $2\,481$; \ $6\,548$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:9}
Conte por etapas: a partir de $2\,970$, conte cinco etapas de $10$;
a partir de $860$, quatro etapas de $100$; a partir de $9\,996$,
quatro etapas de $1$. O que você percebe no fim da última?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:numbers:10}
Usando cada um dos algarismos $3$, $7$, $0$, $5$ exatamente uma vez,
escreva o maior número possível de quatro algarismos e depois o
menor (um número não pode começar por $0$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:numbers:11}
Sou um número de três algarismos. Os meus algarismos são $2$, $5$ e
$8$, sou par e sou maior que $800$. Quem sou eu?
\end{exercise}
