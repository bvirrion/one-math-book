\chapter{Optellen en aftrekken}\label{ch:g3:addsub}

Optellen is samenvoegen; aftrekken is wegnemen, of zoeken wat er
ontbreekt. In dit hoofdstuk oefen je de twee manieren van cijferen ---
met hun beroemde onthouden en lenen --- en de hoofdrekentrucs die het
cijferen vaak overbodig maken.

\section{Cijferend optellen}

\begin{method}[Onder elkaar optellen]\label{met:g3:addsub:add}
\begin{enumerate}
  \item Schrijf de getallen onder elkaar, eenheden onder eenheden,
        tientallen onder tientallen (lijn \emph{rechts} uit);
  \item tel de eenheden op; komt de \omterm{def:g1:addition:def}{som} voorbij $9$, schrijf dan het
        cijfer van de eenheden en \emph{onthoud} het tiental voor de
        volgende kolom (een kleine $1$);
  \item ga verder met de tientallen, de honderdtallen, \dots, en tel
        elke keer het onthoudcijfer mee.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:addsub:add}
Reken $457 + 368$ uit:
\[
\begin{array}{r}
{\scriptstyle 1}\ \ {\scriptstyle 1}\ \ \phantom{0} \\[-3pt]
4\,5\,7 \\
+\ 3\,6\,8 \\
\hline
8\,2\,5
\end{array}
\]
Eenheden: $7 + 8 = 15$: schrijf $5$, onthoud $1$. Tientallen:
$5 + 6 + 1 = 12$: schrijf $2$, onthoud $1$. Honderdtallen:
$4 + 3 + 1 = 8$. Uitkomst: $825$.
\end{example}

\begin{example}[Optellen uit het hoofd]\label{ex:g3:addsub:mental}
Zoek naar vriendelijke getallen voordat je gaat cijferen:
\begin{itemize}
  \item \emph{maak een tiental}: $47 + 8 = 47 + 3 + 5 = 50 + 5 = 55$;
  \item \emph{spring in delen}: $256 + 130 = 256 + 100 + 30 = 386$;
  \item \emph{verwissel de volgorde}: $17 + 59 + 3 = (17 + 3) + 59 = 79$.
\end{itemize}
\end{example}

\section{Cijferend aftrekken}

\begin{method}[Onder elkaar aftrekken]\label{met:g3:addsub:sub}
\begin{enumerate}
  \item Schrijf het grootste getal bovenaan, rechts uitgelijnd;
  \item trek kolom voor kolom af, te beginnen bij de eenheden;
  \item is het bovenste cijfer te klein, dan \emph{leen} je: haal er
        één weg bij de volgende plaats van de bovenste rij (die is tien
        waard in deze kolom);
  \item \emph{controleer}: de uitkomst plus het afgetrokken getal moet
        het bovenste getal teruggeven.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:addsub:sub}
Reken $603 - 147$ uit:
\[
\begin{array}{r}
6\,0\,3 \\
-\ 1\,4\,7 \\
\hline
4\,5\,6
\end{array}
\]
Eenheden: $3 - 7$ kan niet; leen door: $603$ wordt $5$ honderdtallen,
$9$ tientallen, $13$ eenheden. Dan $13 - 7 = 6$; tientallen
$9 - 4 = 5$; honderdtallen $5 - 1 = 4$. Uitkomst: $456$. Controle:
$456 + 147 = 603$.
\end{example}

\begin{example}[Aftrekken als ontbrekend deel]\label{ex:g3:addsub:missing}
``Lila heeft $35$ knikkers, Tom heeft er $52$: hoeveel heeft Tom er
meer?'' is de aftrekking $52 - 35$. Uit het hoofd tel je \emph{door}
vanaf $35$: $35 \to 40$ is $5$, en $40 \to 52$ is $12$; samen
$5 + 12 = 17$. Doortellen is vaak makkelijker dan lenen.
\end{example}

\section{Grootteordes}

\begin{method}[Schatten voordat je rekent]\label{met:g3:addsub:estimate}
Rond voor elke cijferberekening de getallen af op vriendelijke getallen
en maak een \emph{schatting}. Vergelijk daarna: een uitkomst die ver van
de schatting ligt, wijst op een fout.
\end{method}

\begin{example}
Voor $487 + 312$ is de schatting $500 + 300 = 800$. De uitkomst van het
cijferen, $799$, ligt er dicht bij: aannemelijk. Had je $1\,099$
gevonden (een kolom verkeerd uitgelijnd), dan had de schatting je
gewaarschuwd.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g3:addsub:1}
Tel onder elkaar op: $346 + 253$; \quad $478 + 265$; \quad
$1\,536 + 2\,876$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:2}
Trek onder elkaar af en controleer met een optelling: $587 - 234$; \quad
$805 - 348$; \quad $1\,000 - 463$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:3}
Reken uit het hoofd door tientallen te maken: $38 + 7$; \quad $56 + 9$;
\quad $45 + 28$ (tel $30$ op en haal er $2$ af).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:4}
Reken uit het hoofd met sprongen: $340 + 220$; \quad $675 + 210$; \quad
$512 - 300$; \quad $840 - 220$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:5}
Reken $63 - 47$ uit met doortellen (zoals in
\cref{ex:g3:addsub:missing}) en controleer daarna met cijferen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:6}
Schat eerst (rond af op honderdtallen) en reken dan precies uit:
$492 + 218$; \quad $913 - 388$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:7}
Een school heeft $348$ meisjes en $296$ jongens. Hoeveel leerlingen zijn
dat in totaal? Schrijf de bewerking op, reken uit en antwoord met een
zin.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:8}
Een boek heeft $224$ bladzijden; Nora heeft er $178$ gelezen. Hoeveel
bladzijden moet ze nog lezen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:9}
Vul de gaten in (kolom voor kolom):
\[
\begin{array}{r}
3\,?\,7 \\
+\ ?\,2\,? \\
\hline
8\,6\,2
\end{array}
\]
Kunnen de drie gaten maar op één manier gevuld worden? Leg uit.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:addsub:10}
Een bakkerij verkocht 's ochtends $145$ croissants en 's middags $89$;
bij sluitingstijd waren er $17$ croissants over. Hoeveel croissants zijn
er die dag gebakken? (Twee stappen.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:addsub:11}
Theo zegt: ``om $500 - 199$ uit te rekenen, reken ik $500 - 200$ uit en
tel er dan $1$ bij op.'' Heeft hij gelijk? Reken het op beide manieren
uit en leg zijn truc uit.
\end{exercise}
