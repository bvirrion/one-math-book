\chapter{Vermenigvuldigen}\label{ch:g3:mult}

Drie zakjes met zeven knikkers: je kunt $7 + 7 + 7$ optellen, maar het
kan sneller --- met vermenigvuldigen, de bewerking van het herhaald
optellen. In dit hoofdstuk bouwen we de tafels op uit plaatjes en leer je
daarna het cijferen.

\section{Wat vermenigvuldigen betekent}

\begin{definition}[Vermenigvuldigen]\label{def:g3:mult:def}
Het \omterm{def:g2:mult:def}{product} $3 \times 7$ betekent ``drie keer zeven'':
\[
3 \times 7 = 7 + 7 + 7 = 21 .
\]
In een plaatje is $3 \times 7$ een rechthoek van stippen: $3$ rijen van
$7$ stippen.
\end{definition}

      \fill[omDef] (\c,-\r) circle (0.27);

  $3 \times 7 = 21$};

  \end{scope}
\end{tikzpicture}

{\small Dezelfde $21$ stippen, per rij of per kolom geteld: de volgorde
van een \omterm{def:g2:mult:def}{product} maakt niets uit. Wie $3 \times 7$ kent, krijgt
$7 \times 3$ er gratis bij --- de helft van de tafels te leren!}
\end{omfigure}

\begin{proposition}[Regels die helpen]\label{prop:g3:mult:rules}
\begin{enumerate}
  \item De volgorde maakt niets uit: $a \times b = b \times a$.
  \item Met $1$ vermenigvuldigen verandert niets; met $0$
        vermenigvuldigen geeft $0$.
  \item Met $10$ vermenigvuldigen zet er een \omterm{def:g1:counting:zero}{nul} achter:
        $34 \times 10 = 340$; met $100$ twee nullen.
  \item Een tafel kun je uit zijn buren opbouwen:
        $6 \times 8 = 5 \times 8 + 8 = 40 + 8 = 48$.
\end{enumerate}
\end{proposition}
\admitted

\section{Cijferend vermenigvuldigen}

\begin{method}[Met een getal van één cijfer vermenigvuldigen]\label{met:g3:mult:column}
Om $234 \times 6$ uit te rekenen:
\begin{enumerate}
  \item vermenigvuldig de eenheden: $4 \times 6 = 24$: schrijf $4$,
        onthoud $2$;
  \item vermenigvuldig de tientallen en tel het onthoudcijfer erbij:
        $3 \times 6 + 2 = 20$: schrijf $0$, onthoud $2$;
  \item vermenigvuldig de honderdtallen en tel het onthoudcijfer erbij:
        $2 \times 6 + 2 = 14$: schrijf $14$.
\end{enumerate}
\[
\begin{array}{r}
2\,3\,4 \\
\times \quad 6 \\
\hline
1\,4\,0\,4
\end{array}
\]
Controleer met een schatting: $234$ ligt dicht bij $200$, en
$200 \times 6 = 1\,200$: de uitkomst $1\,404$ is aannemelijk.
\end{method}

\begin{example}[Splitsen om uit het hoofd te rekenen]\label{ex:g3:mult:split}
Het cijferen splitst het getal in stilte naar plaatswaarden. Doe uit het
hoofd hetzelfde:
\[
23 \times 4 = (20 + 3) \times 4 = 20 \times 4 + 3 \times 4
= 80 + 12 = 92 .
\]
En met een vriendelijk buurgetal: $99 \times 5 = 100 \times 5 - 5 =
495$.
\end{example}

\end{omfigure}

\section{Vermenigvuldigen in verhaaltjes}

\begin{example}\label{ex:g3:mult:problem}
In een minibus passen $9$ passagiers. Hoeveel passagiers kunnen $27$
minibussen vervoeren?
\begin{enumerate}
  \item Het is een vermenigvuldiging: $27$ groepjes van $9$.
  \item $27 \times 9 = 243$ (met cijferen, of
        $27 \times 9 = 27 \times 10 - 27 = 270 - 27$).
  \item Antwoordzin: de minibussen kunnen $243$ passagiers vervoeren.
\end{enumerate}
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g3:mult:1}
Schrijf als vermenigvuldiging en reken uit:
$8 + 8 + 8 + 8$; \quad $5 + 5 + 5$; \quad
$20 + 20 + 20 + 20 + 20$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:2}
Zeg op en vul aan: $6 \times 7$; \ $8 \times 4$; \ $9 \times 9$;
\ $7 \times 8$; \ $6 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:3}
Teken een rechthoek van stippen voor $4 \times 6$, en daarna een voor
$6 \times 4$. Wat valt je op aan de twee aantallen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:4}
Reken uit met de regels van \cref{prop:g3:mult:rules}:
$56 \times 10$; \quad $8 \times 100$; \quad $73 \times 0$; \quad
$45 \times 1$; \quad $30 \times 10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:5}
Vermenigvuldig cijferend: $132 \times 3$; \quad $217 \times 4$; \quad
$408 \times 7$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:6}
Reken uit het hoofd door te splitsen (zoals in \cref{ex:g3:mult:split}):
$31 \times 5$; \quad $42 \times 3$; \quad $25 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:7}
Reken met een vriendelijk buurgetal: $99 \times 7$; \quad
$101 \times 6$; \quad $98 \times 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:8}
Schat eerst en reken dan uit: $389 \times 5$ (schat met $400$);
$612 \times 8$ (schat met $600$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:9}
In een eierdoos gaan $6$ eieren. Hoeveel eieren zitten er in $38$
dozen? Schrijf de bewerking op, reken uit en antwoord met een zin.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:10}
In een klaslokaal staan $8$ rijen van $4$ tafels; aan elke tafel zitten
$2$ leerlingen. Hoeveel leerlingen kunnen er in het lokaal zitten?
(Twee vermenigvuldigingen.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:11}
Zoek alle manieren om $24$ stoelen in gelijke rijen te zetten (rijen van
$1$, van $2$, \dots). Welke opstellingen kunnen? (Gebruik de tafelkaart:
zoek $24$ erin op.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:12}
Zoë rekent $47 \times 6$ zo uit: $40 \times 6 = 240$,
$7 \times 6 = 42$, en dan $240 + 42 = 282$. Léa rekent
$47 \times 6 = 50 \times 6 - 3 \times 6 = 300 - 18 = 282$. Leg beide
manieren uit. Welke gebruik je liever voor $38 \times 4$? Reken het op
beide manieren uit.
\end{exercise}
