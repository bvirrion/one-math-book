\chapter{Meten}\label{ch:g3:measure}

Hoe lang, hoe zwaar, hoeveel gaat erin, hoe laat is het? Meten maakt van
de wereld getallen --- op voorwaarde dat iedereen dezelfde eenheden
gebruikt. Dit hoofdstuk laat de eenheden van lengte, \omterm{def:g2:measure:units}{massa} en \omterm{def:g2:measure:units}{inhoud} zien,
en leert je de klok lezen.

\section{Lengtes}

\begin{definition}[Lengte-eenheden]\label{def:g3:measure:length}
De basiseenheid van lengte is de \emph{meter} (m). Voor kleine lengtes:
de centimeter (cm) en de millimeter (mm); voor grote afstanden: de
kilometer (km).
\[
1 \text{ m} = 100 \text{ cm}, \qquad
1 \text{ cm} = 10 \text{ mm}, \qquad
1 \text{ km} = 1\,000 \text{ m}.
\]
\end{definition}

\begin{example}[De juiste eenheid kiezen]\label{ex:g3:measure:choose}
Een mier: ongeveer $5$ mm. Een potlood: ongeveer $18$ cm. Een deur:
ongeveer $2$ m. Een wandeling naar het volgende dorp: ongeveer $3$ km.
Zeggen dat ``de lengte van een potlood $18$ is'' betekent niets --- de
eenheid hoort bij het antwoord.
\end{example}

\begin{method}[Eenheden omzetten]\label{met:g3:measure:convert}
Om om te zetten, vermenigvuldig of deel je door $10$, $100$ of $1\,000$:
\begin{enumerate}
  \item van een grote eenheid naar een kleine: \emph{vermenigvuldigen}:
        $3$ m $= 3 \times 100 = 300$ cm;
  \item van een kleine eenheid naar een grote: \emph{delen}:
        $250$ cm $= 250 \div 100 = 2$ m en $50$ cm;
  \item om lengtes op te tellen, zet je ze eerst in \emph{dezelfde}
        eenheid om: $1$ m $+ 35$ cm $= 100 + 35 = 135$ cm.
\end{enumerate}
\end{method}

\section{Massa's en inhouden}

\begin{definition}[Eenheden van massa en inhoud]\label{def:g3:measure:mass}
\omterm{def:g2:measure:units}{Massa}'s meet je in \emph{grammen} (g) en \emph{kilogrammen} (kg),
\omterm{def:g2:measure:units}{inhouden} (hoeveel er in een bak gaat) in \emph{liters} (L) en
\emph{centiliters} (cL):
\[
1 \text{ kg} = 1\,000 \text{ g},
\qquad
1 \text{ L} = 100 \text{ cL}.
\]
\end{definition}

\begin{example}\label{ex:g3:measure:mass}
Een brief: ongeveer $20$ g. Een zak bloem: $1$ kg. Een badkuip: ongeveer
$150$ L. Een glas: ongeveer $20$ cL. Om een weegschaal met een zak van
$1$ kg in evenwicht te brengen met $650$ g appels, leg je er nog
$1\,000 - 650 = 350$ g appels bij.
\end{example}

\section{De klok lezen}

\begin{method}[De tijd lezen]\label{met:g3:measure:clock}
Op een klok met wijzers:
\begin{enumerate}
  \item de \emph{korte} wijzer geeft het uur: lees het getal dat hij het
        laatst voorbij is;
  \item de \emph{lange} wijzer geeft de minuten: elk groot streepje is
        $5$ minuten, dus een lange wijzer op de $3$ betekent $15$
        minuten;
  \item na half kun je ook naar het volgende uur toe lezen: $10{:}50$ is
        ``tien voor elf''.
\end{enumerate}
\end{method}

\begin{example}[Tijdsduren]\label{ex:g3:measure:duration}
De film begint om $14{:}40$ en eindigt om $16{:}10$. Hoe lang duurt hij?
Tel in twee sprongen:
\begin{enumerate}
  \item van $14{:}40$ tot $15{:}00$: $20$ minuten;
  \item van $15{:}00$ tot $16{:}10$: $1$ uur en $10$ minuten.
\end{enumerate}
Samen: $1$ u $30$ min. Pas op: een uur heeft $60$ minuten, dus
$16{:}10$ min $14{:}40$ reken je niet uit als $1610 - 1440$!
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g3:measure:1}
Welke eenheid gebruik je (mm, cm, m of km) voor: de breedte van een
spijker; de lengte van een klasgenoot; de lengte van het schoolplein; de
afstand tussen twee steden?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:2}
Zet om: $5$ m in cm; \quad $70$ mm in cm; \quad $4$ km in m; \quad
$600$ cm in m.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:3}
Meet met je lat (tot op de millimeter) de zijden van een driehoek die een
klasgenoot heeft getekend, en reken de totale lengte van de rand uit.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:4}
Reken uit, na eerst om te zetten: $2$ m $+ 45$ cm (in cm); \quad
$1$ km $+ 250$ m (in m); \quad $3$ cm $+ 8$ mm (in mm).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:5}
Zet om: $3$ kg in g; \quad $2\,500$ g in kg en g; \quad
$4$ L in cL; \quad $350$ cL in L en cL.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:6}
Een recept vraagt $750$ g bloem. Mona heeft een zak van $1$ kg. Heeft ze
genoeg? Hoeveel houdt ze over?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:7}
Lees de tijden: de korte wijzer staat tussen $2$ en $3$, de lange wijzer
op de $6$; de korte wijzer op de $9$, de lange wijzer op de $12$; de
korte wijzer bijna op de $5$, de lange wijzer op de $10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:8}
De school begint om $8{:}30$ en de ochtend eindigt om $11{:}45$. Hoe lang
duurt de ochtend? (Tel in sprongen, zoals in
\cref{ex:g3:measure:duration}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:9}
De trein vertrekt om $9{:}50$ en de rit duurt $45$ minuten. Hoe laat komt
hij aan? (Spring eerst naar $10{:}00$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measure:10}
In een fles gaat $1$ L. Paul vult glazen van $20$ cL. Hoeveel volle
glazen kan hij inschenken? (Zet eerst om.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measure:11}
Lena loopt elke dag $600$ m naar school, en weer terug. Hoeveel meter
loopt ze in een schoolweek van $5$ dagen? Geef het antwoord in meters, en
daarna in kilometers.
\end{exercise}
