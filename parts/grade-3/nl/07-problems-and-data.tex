\chapter{Vraagstukken oplossen}\label{ch:g3:problems}

De vier bewerkingen kennen is één ding; weten \emph{welke} een verhaal
nodig heeft, is iets anders. Dit hoofdstuk geeft een aanpak voor
vraagstukken --- elke keer dezelfde vier stappen --- en leert je gegevens
uit tabellen en pictogrammen halen.

\section{De vier stappen}

\begin{method}[Een vraagstuk oplossen]\label{met:g3:problems:steps}
\begin{enumerate}
  \item \emph{Lees} het vraagstuk twee keer; zeg in je eigen woorden wat
        er gevraagd wordt en wat je weet.
  \item \emph{Kies} de bewerking: maak een snelle schets of strookjes als
        de keuze niet duidelijk is.
  \item \emph{Reken} uit, cijferend of uit het hoofd, met een schatting
        als vangnet.
  \item \emph{Antwoord} met een volledige zin, met de eenheid erbij ---
        en ga na of het antwoord zinnig is (een kind weegt geen $300$
        kg).
\end{enumerate}
\end{method}

\begin{example}[Welke bewerking?]\label{ex:g3:problems:which}
De woorden geven een hint, de schets beslist:
\begin{itemize}
  \item \emph{samenvoegen / in totaal}: optellen;
  \item \emph{wegnemen / hoeveel meer / wat ontbreekt}: aftrekken;
  \item \emph{zoveel groepjes van zoveel / zoveel keer}:
        vermenigvuldigen;
  \item \emph{eerlijk verdelen / hoeveel groepjes}: delen.
\end{itemize}
\end{example}

\begin{example}[Een vraagstuk in twee stappen]\label{ex:g3:problems:twostep}
``Maya koopt $3$ pakken van $6$ flesjes en drinkt er $2$ op. Hoeveel
flesjes blijven er over?''
\begin{enumerate}
  \item Gevraagd: het aantal flesjes dat overblijft. Bekend: $3$ pakken
        van $6$; $2$ opgedronken.
  \item Schets: $3$ groepjes van $6$, en dan $2$ eraf. Twee bewerkingen!
  \item Uitrekenen: $3 \times 6 = 18$, en dan $18 - 2 = 16$.
  \item Antwoordzin: er blijven $16$ flesjes over.
\end{enumerate}
Stop nooit na de eerste bewerking: lees de vraag opnieuw om te zien of het
verhaal al af is.
\end{example}

\begin{example}[Een getal te veel]\label{ex:g3:problems:trap}
``Een bakker van $52$ jaar bakt $120$ broodjes en verkoopt er $85$.
Hoeveel broodjes blijven er over?'' De leeftijd staat er om je af te
leiden: voor het antwoord heb je alleen $120 - 85 = 35$ broodjes nodig.
Sorteer voor het rekenen altijd de getallen: welke gebruikt de vraag echt?
\end{example}

\section{Tabellen en pictogrammen lezen}

\begin{example}[Een tabel]\label{ex:g3:problems:table}
Uitgeleende boeken in de bibliotheek:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cccc}
 & strips & romans & non-fictie & totaal \\
\hline
week 1 & $14$ & $9$ & $6$ & $29$ \\
week 2 & $11$ & $12$ & $8$ & $31$ \\
\end{tabular}
\end{center}
Aflezen: week 2, romans: $12$. Rekenen met de tabel: in de twee weken
samen zijn er $14 + 11 = 25$ strips uitgeleend; de drukste week was week 2
($31 > 29$).
\end{example}

\begin{example}[Een pictogram]\label{ex:g3:problems:pictogram}
In een pictogram staat één symbool voor een vast aantal --- lees eerst de
legenda!

\begin{omfigure}
\begin{tikzpicture}[scale=0.62]
  \node[left, font=\small] at (0,2.4) {maandag};
  \foreach \i in {0,1,2} \fill[omDef] (0.5+\i*0.8,2.4) circle (0.26);
  \node[left, font=\small] at (0,1.2) {dinsdag};
  \foreach \i in {0,1,2,3,4} \fill[omDef] (0.5+\i*0.8,1.2) circle (0.26);
  \node[left, font=\small] at (0,0) {woensdag};
  \foreach \i in {0,1} \fill[omDef] (0.5+\i*0.8,0) circle (0.26);
  \fill[omDef] (0.5+1.6,0) circle (0.26);
  \node[right, font=\small] at (5.4,1.2)
    {legenda: één stip $= 4$ verkochte gebakjes};
\end{tikzpicture}

{\small Een pictogram: maandag heeft $3$ stippen, dus $3 \times 4 = 12$
gebakjes; dinsdag $5$ stippen, $20$ gebakjes; woensdag $3$ stippen, $12$
gebakjes.}
\end{omfigure}
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g3:problems:1}
\emph{Noem} bij elk verhaal alleen de bewerking (reken niet uit): een
kudde van $48$ schapen sluit zich aan bij een kudde van $37$; \ $56$
kersen verdeeld over $7$ kinderen; \ $9$ dozen van $8$ potloden; \ Ali had
$71$ kaarten en gaf er $18$ weg.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:2}
Los op met de vier stappen: ``Op een parkeerterrein staan 's ochtends
$246$ auto's; rond het middaguur komen er $158$ bij. Hoeveel auto's staan
er nu?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:3}
Los op: ``Een touw van $90$ m wordt in stukken van $9$ m geknipt. Hoeveel
stukken krijg je?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:4}
Los op: ``Kaartjes kosten $8$ per stuk. Wat kosten $26$ kaartjes?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:5}
Teken strookjes en los dan op: ``Emma heeft $64$ stickers, Hugo heeft er
$29$ minder. Hoeveel heeft Hugo?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:6}
Twee stappen: ``In een krat gaan $45$ appels. Een groenteboer krijgt $6$
kratten en verkoopt $178$ appels. Hoeveel appels blijven over?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:7}
Gebruik de tabel van \cref{ex:g3:problems:table}: hoeveel non-fictieboeken
zijn er in de twee weken samen uitgeleend? Hoeveel romans meer in week 2
dan in week 1?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:8}
Gebruik het pictogram van \cref{ex:g3:problems:pictogram}: hoeveel
gebakjes zijn er in de drie dagen samen verkocht? Op welke dag zijn er de
meeste verkocht?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:problems:9}
In dit vraagstuk staat een getal te veel --- zoek het en los dan op: ``Een
boek van $210$ bladzijden kost $12$. Leo leest $35$ bladzijden per dag.
Hoeveel dagen heeft hij nodig om het boek uit te lezen?''
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:problems:10}
``Een gezin van $2$ volwassenen en $3$ kinderen gaat naar het museum. Een
kaartje voor een volwassene kost $9$, een kinderkaartje $4$. Ze betalen
met een biljet van $50$. Hoeveel wisselgeld krijgen ze?'' (Drie stappen:
volwassenen, kinderen, en dan het wisselgeld.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:problems:11}
Verzin zelf een vraagstuk in twee stappen met $100$ als antwoord, schrijf
de oplossing met de vier stappen op, en test het bij een klasgenoot.
\end{exercise}
