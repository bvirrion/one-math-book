\chapter{Getallen tot 10\,000}\label{ch:g3:numbers}

Duizend, tweeduizend, drieduizend\,\dots\ grote getallen zijn opgebouwd
uit kleine, met één prachtig idee: de \emph{plaats} van een cijfer
bepaalt zijn waarde. In dit hoofdstuk leer je getallen tot $10\,000$
lezen, schrijven, vergelijken en ordenen.

\section{Getallen lezen en schrijven}

\begin{definition}[Plaatswaarde]\label{def:g3:numbers:place}
Een getal schrijf je met de cijfers $0, 1, 2, \dots, 9$. Van rechts naar
links zijn de plaatsen: eenheden, tientallen, honderdtallen,
duizendtallen. In $3\,254$:
\[
3\,254 = 3 \text{ duizendtallen} + 2 \text{ honderdtallen}
+ 5 \text{ tientallen} + 4 \text{ eenheden}.
\]
\end{definition}

\begin{example}[De nul die een plaats bezet houdt]\label{ex:g3:numbers:zero}
``Vierduizend zeven'' schrijf je als $4\,007$: een $4$ bij de
duizendtallen, een $7$ bij de eenheden, en \emph{nullen} ertussen om elk
cijfer op zijn juiste plaats te houden. Zonder die nullen zou $47$ een
heel ander getal zijn!
\end{example}

\begin{example}[Wisselen]\label{ex:g3:numbers:exchange}
Tien eenheden maken één tiental; tien tientallen maken één honderdtal;
tien honderdtallen maken één duizendtal. Dus $10$ honderdtallen
$= 1\,000$, en $32$ tientallen $= 320$: het getal $320$ bevat $32$
tientallen (niet alleen het cijfer $2$!).
\end{example}

\section{Vergelijken en ordenen}

\begin{method}[Twee getallen vergelijken]\label{met:g3:numbers:compare}
\begin{enumerate}
  \item Tel de cijfers: meer cijfers betekent een groter getal
        ($1\,002 > 998$).
  \item \omterm{def:g3:numbers:evenodd}{Even} veel cijfers: vergelijk cijfer voor cijfer, te beginnen
        \emph{links}. Het eerste \omterm{ex:g1:subtraction:difference}{verschil} beslist:
        $5\,614 > 5\,589$, want bij de honderdtallen is $6 > 5$.
  \item Schrijf het antwoord met de tekens $<$ (kleiner dan) of $>$
        (groter dan): de open kant wijst altijd naar het grootste
        getal.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:numbers:order}
Zet op volgorde, van klein naar groot: $780$, $8\,700$, $807$, $87$.
Eerst op het aantal cijfers: $87$ (twee), dan $780$ en $807$ (drie), dan
$8\,700$ (vier). Tussen $780$ en $807$: vergelijk de cijfers van de
honderdtallen, $7 < 8$, dus $780 < 807$. De volgorde is:
\[
87 < 780 < 807 < 8\,700 .
\]
\end{example}

\section{Even en oneven}

\begin{definition}[Even, oneven]\label{def:g3:numbers:evenodd}
Een getal is \emph{even}\index{even getal} als het in twee gelijke hele
delen te splitsen is --- het cijfer van de eenheden is dan $0$, $2$,
$4$, $6$ of $8$. Anders is het getal \emph{oneven}\index{oneven getal}
(cijfer van de eenheden $1$, $3$, $5$, $7$ of $9$).
\end{definition}

\begin{example}
$46$ is \omterm{def:g3:numbers:evenodd}{even} ($46 = 23 + 23$); $47$ is \omterm{def:g3:numbers:evenodd}{oneven} (bij twee gelijke delen
zou je een half voorwerp nodig hebben). Alleen het \emph{laatste} cijfer
telt: $3\,578$ is \omterm{def:g3:numbers:evenodd}{even}, $2\,341$ is \omterm{def:g3:numbers:evenodd}{oneven}.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g3:numbers:1}
Schrijf in cijfers: ``tweeduizend driehonderdvijfenveertig'';
``zesduizend vijftig''; ``negenduizend één''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:2}
Schrijf in woorden: $1\,208$; \quad $4\,070$; \quad $9\,999$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:3}
In $7\,382$: welk cijfer staat op de plaats van de tientallen? En op die
van de duizendtallen? Hoeveel \emph{honderdtallen} bevat het getal in
totaal (pas op: niet alleen het cijfer van de honderdtallen --- denk aan
\cref{ex:g3:numbers:exchange})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:4}
Splits zoals in \cref{def:g3:numbers:place}:
$5\,631$; \quad $2\,046$; \quad $8\,500$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:5}
Neem over en vul in met $<$ of $>$:
\[
456 \;?\; 465, \qquad
1\,203 \;?\; 989, \qquad
6\,090 \;?\; 6\,009, \qquad
9\,999 \;?\; 10\,000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:6}
Zet op volgorde, van klein naar groot: $530$; \ $3\,500$; \ $353$;
\ $5\,033$; \ $503$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:7}
Zet op een getallenlijn met streepjes per honderd, van $0$ tot
$1\,000$, de getallen $150$, $480$, $820$ en $990$ ongeveer op hun
plaats.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:8}
\omterm{def:g3:numbers:evenodd}{Even} of \omterm{def:g3:numbers:evenodd}{oneven}? \ $34$; \ $57$; \ $70$; \ $2\,481$; \ $6\,548$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:9}
Tel met stappen: vanaf $2\,970$ vijf stappen van $10$; vanaf $860$ vier
stappen van $100$; vanaf $9\,996$ vier stappen van $1$. Wat valt je aan
het einde van de laatste rij op?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:numbers:10}
Gebruik elk van de cijfers $3$, $7$, $0$, $5$ precies één keer en
schrijf het grootste getal van vier cijfers, en daarna het kleinste (een
getal mag niet met $0$ beginnen).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:numbers:11}
Ik ben een getal van drie cijfers. Mijn cijfers zijn $2$, $5$ en $8$, ik
ben \omterm{def:g3:numbers:evenodd}{even}, en ik ben groter dan $800$. Wie ben ik?
\end{exercise}
