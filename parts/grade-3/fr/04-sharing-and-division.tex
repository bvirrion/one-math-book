\chapter{Partage et division}\label{ch:g3:division}

Vingt billes partagées équitablement entre quatre enfants~:
combien chacun~? La division répond à toutes les questions de
partage --- et aux questions de groupement aussi --- et c'est la
jumelle de la multiplication~: l'une défait l'autre.

\section{Deux visages de la division}

\begin{definition}[Division]\label{def:g3:division:def}
La division $20 \div 4$ répond à deux sortes de questions~:
\begin{itemize}
  \item \emph{partage}~: $20$ billes partagées entre $4$ enfants ---
        combien de billes \emph{chacun}~? ($5$ chacun)~;
  \item \emph{groupement}~: $20$ billes mises en sacs de $4$ ---
        combien de \emph{sacs}~? ($5$ sacs).
\end{itemize}
Les deux questions ont la même réponse, car toutes deux défont
la même multiplication~: $4 \times 5 = 20$.
\end{definition}

\begin{method}[Diviser avec les tables]\label{met:g3:division:tables}
Pour calculer $42 \div 6$, se demander~: \guillemotleft{} $6$ fois
\emph{quoi} fait $42$~? \guillemotright{} La table de $6$ répond~:
$6 \times 7 = 42$, donc $42 \div 6 = 7$. Chaque question de division
est une question de multiplication lue à l'envers.
\end{method}

\section{Quand ça ne tombe pas juste}

\begin{definition}[Reste]\label{def:g3:division:remainder}
Partager $23$ billes entre $4$ enfants donne $5$ billes chacun ---
et $3$ billes \emph{en trop}, pas assez pour un autre tour. On
écrit
\[
23 = 4 \times 5 + 3 ,
\]
et on dit~: le \emph{quotient}\index{quotient} est $5$, le
\emph{reste}\index{reste} est $3$. Le reste est toujours plus petit
que le nombre d'enfants --- sinon un autre tour de partage serait
possible.
\end{definition}

\begin{example}\label{ex:g3:division:remainder}
Diviser $58$ par $7$. Parcourir la table de $7$~:
$7 \times 8 = 56$ est le plus grand \omterm{def:g2:mult:def}{produit} qui tient sous $58$
($7 \times 9 = 63$ est trop grand). Donc
\[
58 = 7 \times 8 + 2 :
\]
\omterm{def:g3:division:remainder}{quotient} $8$, \omterm{def:g3:division:remainder}{reste} $2$. Vérification~: $56 + 2 = 58$, et
$2 < 7$.
\end{example}

\begin{method}[Interpréter le reste]\label{met:g3:division:interpret}
Après la division, revenir à l'histoire~:
\begin{enumerate}
  \item \guillemotleft{} combien chacun / combien de groupes
        complets~? \guillemotright{} --- le \omterm{def:g3:division:remainder}{quotient}~;
  \item \guillemotleft{} combien en trop~? \guillemotright{} --- le
        \omterm{def:g3:division:remainder}{reste}~;
  \item \guillemotleft{} combien de boîtes pour tout contenir~?
        \guillemotright{} --- le \omterm{def:g3:division:remainder}{quotient} \emph{plus un} si le \omterm{def:g3:division:remainder}{reste}
        n'est pas \omterm{def:g1:counting:zero}{zéro}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:division:interpret}
$58$ élèves vont en canoë, $7$ par canoë.
$58 = 7 \times 8 + 2$~: huit canoës sont pleins, et $2$ élèves
restent --- ils ont besoin d'un canoë aussi. Donc $9$ canoës
doivent être réservés.
\end{example}

\section{Moitiés, tiers, quarts}

\begin{definition}[Moitié, tiers, quart]\label{def:g3:division:half}
La \emph{moitié}\index{moitié} d'un nombre est le résultat de sa division par
$2$~; le \emph{tiers}, par $3$~; le \emph{quart}\index{quart}, par $4$. Par
exemple la moitié de $18$ est $9$, le tiers de $18$ est $6$, le
quart de $36$ est $9$.
\end{definition}

\begin{example}[Doubles et moitiés vont ensemble]\label{ex:g3:division:double}
La \omterm{def:g3:division:half}{moitié} de $46$~: comme $46 = 40 + 6$, la \omterm{def:g3:division:half}{moitié} de chaque
partie donne $20 + 3 = 23$. Vérifier avec le double~:
$23 + 23 = 46$. Connaître les doubles par cœur ($25 \to 50$,
$45 \to 90$, $150 \to 300$) rend les \omterm{def:g3:division:half}{moitiés} instantanées.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g3:division:1}
Calcule en utilisant les tables à l'envers~: $36 \div 4$~; \quad
$63 \div 9$~; \quad $40 \div 5$~; \quad $56 \div 8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:2}
Pour chaque division, écris l'égalité $a = b \times q + r$ et
donne le \omterm{def:g3:division:remainder}{quotient} et le \omterm{def:g3:division:remainder}{reste}~: $17 \div 5$~; \quad $30 \div 4$~;
\quad $50 \div 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:3}
$28$ billes sont partagées entre $6$ enfants. Combien de billes
chacun, et combien en trop~? Dessine le partage si cela aide.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:4}
Calcule~: la \omterm{def:g3:division:half}{moitié} de $68$~; le tiers de $27$~; le \omterm{def:g3:division:half}{quart} de
$48$~; la \omterm{def:g3:division:half}{moitié} de $150$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:5}
Vrai ou faux~? Vérifie avec une multiplication.
\guillemotleft{} $45 \div 5 = 9$ \guillemotright{}~; \quad
\guillemotleft{} $52 \div 6 = 8$ \omterm{def:g3:division:remainder}{reste} $4$ \guillemotright{}~;
\quad \guillemotleft{} $70 \div 8 = 9$ \omterm{def:g3:division:remainder}{reste} $2$
\guillemotright{}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:6}
$32$ photos sont collées dans un album, $4$ par page. Combien de
pages sont utilisées~? Quel visage de la division est-ce (partage
ou groupement)~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:7}
Une corde de $75$~cm est coupée en morceaux de $9$~cm. Combien de
morceaux complets, et quelle longueur reste-t-il~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:8}
$50$ enfants partent en voyage dans des voitures de $4$ places.
Combien de voitures faut-il~? (Attention~: tout le monde doit
monter~!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:9}
Trouve les nombres manquants~: $? \div 7 = 6$~; \quad
$54 \div ? = 6$~; \quad la \omterm{def:g3:division:half}{moitié} de $?$ est $35$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:division:10}
Amina partage ses $47$ autocollants entre ses $5$ amis, le plus
équitablement possible, et garde le \omterm{def:g3:division:remainder}{reste} pour elle. Combien
d'autocollants chaque ami reçoit-il, et combien Amina en
garde-t-elle~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:division:11}
Quand on divise un nombre par $6$, quels sont tous les \omterm{def:g3:division:remainder}{restes}
possibles~? Quel est le plus grand nombre dont la division par $6$ a
pour \omterm{def:g3:division:remainder}{quotient} $7$~?
\end{exercise}
