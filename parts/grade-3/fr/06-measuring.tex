\chapter{Mesures}\label{ch:g3:measure}

Combien de long, combien de lourd, combien ça contient, quelle
heure est-il~? Mesurer transforme le monde en nombres --- à
condition que tout le monde utilise les mêmes unités. Ce chapitre
présente les unités de longueur, de \omterm{def:g2:measure:units}{masse} et de \omterm{def:g2:measure:units}{contenance}, et
apprend à lire une horloge.

\section{Longueurs}

\begin{definition}[Unités de longueur]\label{def:g3:measure:length}
L'unité de base de longueur est le \emph{mètre} (m). Pour les
petites longueurs~: le centimètre (cm) et le millimètre (mm)~;
pour les longues distances~: le kilomètre (km).
\[
1 \text{ m} = 100 \text{ cm}, \qquad
1 \text{ cm} = 10 \text{ mm}, \qquad
1 \text{ km} = 1\,000 \text{ m}.
\]
\end{definition}

petites marques pour les millimètres. L'objet commence au trait

\end{omfigure}

\begin{example}[Choisir la bonne unité]\label{ex:g3:measure:choose}
Une fourmi~: environ $5$~mm. Un crayon~: environ $18$~cm. Une
porte~: environ $2$~m. Une marche jusqu'au village voisin~: environ
$3$~km. Dire \guillemotleft{} la longueur d'un crayon est $18$
\guillemotright{} ne veut rien dire --- l'unité fait partie de la
réponse.
\end{example}

\begin{method}[Convertir des unités]\label{met:g3:measure:convert}
Pour convertir, multiplier ou diviser par $10$, $100$ ou $1\,000$~:
\begin{enumerate}
  \item d'une grande unité vers une petite, \emph{multiplier}~:
        $3$~m $= 3 \times 100 = 300$~cm~;
  \item d'une petite unité vers une grande, \emph{diviser}~:
        $250$~cm $= 250 \div 100 = 2$~m et $50$~cm~;
  \item pour additionner des longueurs, les convertir d'abord à la
        \emph{même} unité~: $1$~m $+ 35$~cm $= 100 + 35 =
        135$~cm.
\end{enumerate}
\end{method}

\section{Masses et contenances}

\begin{definition}[Unités de masse et de contenance]\label{def:g3:measure:mass}
Les \omterm{def:g2:measure:units}{masses} se mesurent en \emph{grammes} (g) et \emph{kilogrammes}
(kg), les \omterm{def:g2:measure:units}{contenances} (ce qu'un récipient peut contenir) en
\emph{litres} (L) et \emph{centilitres} (cL)~:
\[
1 \text{ kg} = 1\,000 \text{ g},
\qquad
1 \text{ L} = 100 \text{ cL}.
\]
\end{definition}

\begin{example}\label{ex:g3:measure:mass}
Une lettre~: environ $20$~g. Un sac de farine~: $1$~kg. Une baignoire~:
environ $150$~L. Un verre~: environ $20$~cL. Pour équilibrer une
balance portant un sac de $1$~kg contre $650$~g de pommes, ajouter
$1\,000 - 650 = 350$~g du côté des pommes.
\end{example}

\section{Lire l'horloge}

\begin{method}[Lire l'heure]\label{met:g3:measure:clock}
Sur une horloge à aiguilles~:
\begin{enumerate}
  \item la \emph{petite} aiguille donne l'heure~: lire le nombre
        qu'elle a dépassé en dernier~;
  \item la \emph{grande} aiguille donne les minutes~: chaque grande
        marque vaut $5$ minutes, donc une grande aiguille sur le $3$
        signifie $15$ minutes~;
  \item après la demie, on peut aussi lire \guillemotleft{} avant
        l'heure suivante \guillemotright{}~: $10{:}50$ c'est
        \guillemotleft{} onze heures moins dix \guillemotright{}.
\end{enumerate}
\end{method}

{\small Trois horloges~: petite aiguille bleue pour les heures,

\begin{example}[Durées]\label{ex:g3:measure:duration}
Le film commence à $14{:}40$ et se termine à $16{:}10$. Combien
dure-t-il~? Compter en deux sauts~:
\begin{enumerate}
  \item de $14{:}40$ à $15{:}00$~: $20$ minutes~;
  \item de $15{:}00$ à $16{:}10$~: $1$ heure $10$ minutes.
\end{enumerate}
Total~: $1$~h $30$~min. Attention~: une heure a $60$ minutes, donc
$16{:}10$ moins $14{:}40$ ne se calcule pas comme $1610 - 1440$~!
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g3:measure:1}
Quelle unité utiliserais-tu (mm, cm, m ou km) pour~: la largeur
d'un clou~; la taille d'un camarade~; la longueur de la cour~; la
distance entre deux villes~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:2}
Convertis~: $5$~m en cm~; \quad $70$~mm en cm~; \quad $4$~km en m~;
\quad $600$~cm en m.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:3}
Mesure avec ta règle (au millimètre près) les côtés d'un
triangle dessiné par un camarade, et calcule la longueur totale de
son contour.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:4}
Calcule, en convertissant d'abord~: $2$~m $+ 45$~cm (en cm)~; \quad
$1$~km $+ 250$~m (en m)~; \quad $3$~cm $+ 8$~mm (en mm).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:5}
Convertis~: $3$~kg en g~; \quad $2\,500$~g en kg et g~; \quad
$4$~L en cL~; \quad $350$~cL en L et cL.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:6}
Une recette demande $750$~g de farine. Mona a un sac de $1$~kg. En
a-t-elle assez~? Combien en restera-t-il~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:7}
Lis les heures~: la petite aiguille est entre $2$ et $3$, la grande
sur le $6$~; la petite aiguille sur le $9$, la grande sur le $12$~;
la petite aiguille presque sur le $5$, la grande sur le $10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:8}
L'école commence à $8{:}30$ et la matinée se termine à
$11{:}45$. Combien dure la matinée~? (Compte par sauts, comme dans
\cref{ex:g3:measure:duration}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:9}
Le train part à $9{:}50$ et le trajet dure $45$ minutes. À
quelle heure arrive-t-il~? (Saute d'abord jusqu'à $10{:}00$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measure:10}
Une bouteille contient $1$~L. Paul remplit des verres de $20$~cL.
Combien de verres pleins peut-il verser~? (Convertis d'abord.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measure:11}
Léna marche $600$~m pour aller à l'école, et autant pour
revenir, chaque jour. Combien de mètres marche-t-elle en une
semaine scolaire de $5$~jours~? Donne la réponse en mètres, puis
en kilomètres.
\end{exercise}
