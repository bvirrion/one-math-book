\chapter{Figures et angles droits}\label{ch:g3:shapes}

Carrés, rectangles, triangles, cercles~: les figures sont de
vieilles amies. Ce chapitre les regarde avec l'œil du
géomètre --- qu'est-ce qui fait exactement d'un carré un
carré~? --- et présente l'outil qui décide~: l'équerre et
son \omterm{def:g3:shapes:rightangle}{angle droit}.

\section{L'angle droit}

\begin{definition}[Angle droit]\label{def:g3:shapes:rightangle}
Un \emph{angle droit}\index{angle droit} est l'angle d'un coin de
feuille de papier, ou de l'équerre. Deux droites qui forment un
angle droit sont \emph{perpendiculaires}. Dans les figures, un angle
droit se marque par un petit carré dans le coin.
\end{definition}

\begin{method}[Vérifier un angle droit]\label{met:g3:shapes:check}
\begin{enumerate}
  \item Placer le coin de l'équerre exactement sur le coin à
        tester~;
  \item glisser un bord de l'équerre le long d'un côté de
        l'angle~;
  \item si l'autre côté de l'angle suit exactement l'autre bord,
        l'angle est droit --- s'il s'ouvre plus large ou plus
        étroit, ce n'est pas un \omterm{def:g3:shapes:rightangle}{angle droit}.
\end{enumerate}
\end{method}

\section{Les figures et leurs règles}

\begin{definition}[Carré, rectangle, triangle]\label{def:g3:shapes:shapes}
\begin{itemize}
  \item Un \emph{rectangle}\index{rectangle} a $4$ côtés et $4$
        \omterm{def:g3:shapes:rightangle}{angles droits}~; ses côtés opposés ont la même
        longueur.
  \item Un \emph{carré}\index{carré} est un rectangle dont les
        $4$ côtés ont tous la même longueur.
  \item Un \emph{triangle}\index{triangle} a $3$ côtés~; un
        \emph{triangle rectangle} a un \omterm{def:g3:shapes:rightangle}{angle droit}.
  \item Un \emph{cercle}\index{cercle} se trace au compas~: tous ses
        points sont à la même distance (le \emph{rayon}\index{rayon}) du
        \emph{centre}.
\end{itemize}
\end{definition}

\begin{example}[Pourquoi les marques comptent]\label{ex:g3:shapes:marks}
Une figure un peu penchée peut \emph{ressembler} à un carré
sans l'être. Les marques disent la vérité~: $4$ petits carrés
d'\omterm{def:g3:shapes:rightangle}{angle droit} $+$ $4$ traits d'égalité $=$ un vrai carré.
Quand on dessine, mettre les marques~; quand on lit une figure, ne
se fier qu'aux marques et aux mesures données.
\end{example}

\section{Dessiner sur du papier quadrillé}

\begin{method}[Constructions sur un quadrillage]\label{met:g3:shapes:grid}
Le papier quadrillé offre des \omterm{def:g3:shapes:rightangle}{angles droits} gratuits~: ses lignes
sont perpendiculaires.
\begin{enumerate}
  \item Pour dessiner un rectangle $5$ sur $3$~: choisir un croisement,
        compter $5$ cases à droite, $3$ cases vers le haut, et
        fermer la figure le long des lignes du quadrillage~;
  \item pour vérifier des longueurs égales, compter les cases~;
  \item les pas en diagonale font des lignes penchées~: aller
        \guillemotleft{} $1$ à droite, $1$ en haut \guillemotright{}
        répétément trace une ligne à $45^\circ$.
\end{enumerate}
\end{method}

\begin{example}[Un programme de construction]\label{ex:g3:shapes:program}
Une figure peut se décrire par des instructions~:
\guillemotleft{} 1. Dessiner un carré $ABCD$ de côté $4$~cm.
2. Dessiner le cercle de centre $A$ passant par $B$.
\guillemotright{} En suivant un programme ligne par ligne, n'importe
qui peut redessiner la même figure --- et en écrire un force à
nommer chaque point. Essaie avec un camarade~: l'un écrit, l'autre
dessine.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g3:shapes:1}
Dans la classe, trouve (sans mesurer) trois \omterm{def:g3:shapes:rightangle}{angles droits} et un
angle qui n'est pas droit. Comment peux-tu \emph{vérifier} chacun~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:2}
Dessine un rectangle de $6$~cm sur $4$~cm à la règle et à
l'équerre. Marque ses \omterm{def:g3:shapes:rightangle}{angles droits} et ses côtés égaux.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:3}
Dessine un carré de côté $5$~cm. Combien d'\omterm{def:g3:shapes:rightangle}{angles droits}
a-t-il~? Combien de côtés égaux~? Marque tout.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:4}
Sur du papier quadrillé, dessine~: un carré $4 \times 4$~; un
rectangle $7 \times 2$~; un triangle rectangle de côtés $3$ et
$5$ cases.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:5}
Vrai ou faux~? Explique.
\guillemotleft{} Tout carré est un rectangle. \guillemotright{}
\quad \guillemotleft{} Tout rectangle est un carré.
\guillemotright{} \quad \guillemotleft{} Un triangle peut avoir deux
\omterm{def:g3:shapes:rightangle}{angles droits}. \guillemotright{} (Essaie d'en dessiner un~!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:6}
Dessine un cercle de \omterm{def:g3:shapes:shapes}{rayon} $4$~cm. Marque son centre $O$, un point
$A$ sur le cercle, et trace le \omterm{def:g3:shapes:shapes}{rayon} $[OA]$. Quelle est la longueur
de $OA$~? Et si $B$ est un autre point du cercle, quelle est la
longueur de $OB$~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:7}
Suis ce programme~: \guillemotleft{} 1. Dessine un rectangle $ABCD$
avec $AB = 6$~cm et $BC = 3$~cm. 2. Marque le point $M$ au milieu de
$[AB]$. 3. Relie $M$ à $C$ et à $D$. \guillemotright{} En
quelles figures le rectangle a-t-il été découpé~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:8}
Classe les figures~: lesquelles ont au moins un \omterm{def:g3:shapes:rightangle}{angle droit}~? Un
carré~; un rectangle (non carré)~; un triangle rectangle~; un
cercle~; la lettre L dessinée avec des traits épais.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:9}
Sur du papier quadrillé, dessine un carré penché dont les
côtés vont \guillemotleft{} $2$ à droite, $1$ en haut
\guillemotright{} (et les tours correspondants). Compte les cases
pour te convaincre que les quatre côtés ont la même longueur.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:10}
Écris un programme de construction (instructions numérotées,
points nommés) pour~: un carré de côté $3$~cm posé sur un
rectangle $3$~cm $\times$ $2$~cm, comme une maison sur son rez-de-
chaussée. Teste ton programme sur un camarade.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:11}
Combien de carrés peux-tu trouver dans une grille $3 \times 3$ de
petits carrés~? (Compte les $1 \times 1$, les $2 \times 2$ et les
$3 \times 3$.)
\end{exercise}
