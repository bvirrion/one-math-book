\section*{Chapitre \ref{ch:g3:shapes} --- Figures et angles droits}

\begin{solution}{exo:g3:shapes:1}
\omterm{def:g3:shapes:rightangle}{Angles droits} typiques~: les coins du tableau, d'une feuille de
papier, de la porte. Un angle non droit~: l'ouverture d'une paire de
ciseaux, les aiguilles de l'horloge à $1$~heure. Chacun se
vérifie à l'équerre (\cref{met:g3:shapes:check}) --- jamais
à l'œil seul.
\end{solution}

\begin{solution}{exo:g3:shapes:2}
Construction~: un côté de $6$~cm, \omterm{def:g3:shapes:rightangle}{angles droits} aux deux
extrémités avec l'équerre, côtés de $4$~cm, fermer la
figure. Marques~: $4$ petits carrés dans les coins, un trait sur
chaque côté de $6$~cm, un double trait sur chaque côté de
$4$~cm.
\end{solution}

\begin{solution}{exo:g3:shapes:3}
Un carré a $4$ \omterm{def:g3:shapes:rightangle}{angles droits} et $4$ côtés égaux --- tous
marqués.
\end{solution}

\begin{solution}{exo:g3:shapes:4}
Constructions sur quadrillage~; le quadrillage garantit les \omterm{def:g3:shapes:rightangle}{angles
droits}, et le comptage des cases garantit les longueurs.
\end{solution}

\begin{solution}{exo:g3:shapes:5}
\guillemotleft{} Tout carré est un rectangle \guillemotright{}~:
\emph{vrai} --- il a $4$ \omterm{def:g3:shapes:rightangle}{angles droits}, c'est tout ce qu'un
rectangle demande.

\guillemotleft{} Tout rectangle est un carré \guillemotright{}~:
\emph{faux} --- un rectangle $6 \times 4$ a des côtés inégaux.

\guillemotleft{} Un triangle peut avoir deux \omterm{def:g3:shapes:rightangle}{angles droits}
\guillemotright{}~: \emph{faux} --- les deux côtés tirés des
deux \omterm{def:g3:shapes:rightangle}{angles droits} seraient perpendiculaires à la même base,
donc ne se rencontreraient jamais pour fermer le triangle.
\end{solution}

\begin{solution}{exo:g3:shapes:6}
$OA = 4$~cm (le \omterm{def:g3:shapes:shapes}{rayon}). $OB = 4$~cm aussi~: \emph{tout} point du
cercle est à la même distance du centre.
\end{solution}

\begin{solution}{exo:g3:shapes:7}
Le rectangle est découpé en trois triangles~: deux triangles
rectangles ($AMD$ et $MBC$, rectangles en $A$ et $B$) et le triangle
du milieu $MCD$.
\end{solution}

\begin{solution}{exo:g3:shapes:8}
Avec au moins un \omterm{def:g3:shapes:rightangle}{angle droit}~: le carré, le rectangle, le triangle
rectangle, et la lettre L (ses coins intérieurs et extérieurs).
Le cercle n'a aucun coin.
\end{solution}

\begin{solution}{exo:g3:shapes:9}
Chaque côté croise \guillemotleft{} $2$ à droite, $1$ en haut
\guillemotright{} (ou les versions tournées \guillemotleft{} $1$
à droite, $2$ en bas \guillemotright{}, etc.)~: chaque côté est
la diagonale d'un bloc $2 \times 1$ de cases, donc les quatre ont la
même longueur.
\end{solution}

\begin{solution}{exo:g3:shapes:10}
Un programme correct~: \guillemotleft{} 1. Dessiner un rectangle
$ABCD$ avec $AB = 3$~cm (bas) et $BC = 2$~cm. 2. Sur le côté
$[DC]$, dessiner le carré $DCEF$ de côté $3$~cm, à
l'extérieur du rectangle. \guillemotright{} Tout programme clair,
numéroté et complet est correct --- le tester en le suivant mot
à mot.
\end{solution}

\begin{solution}{exo:g3:shapes:11}
Carrés $1 \times 1$~: $9$. Carrés $2 \times 2$~: $4$.
$3 \times 3$~: $1$. Total~: $9 + 4 + 1 = 14$ carrés.
\end{solution}
