\section*{अध्याय \ref{ch:g3:mult} --- गुणा}

\begin{solution}{exo:g3:mult:1}
$8 + 8 + 8 + 8 = 4 \times 8 = 32$; \quad
$5 + 5 + 5 = 3 \times 5 = 15$; \quad
$20 \times 5 = 100$।
\end{solution}

\begin{solution}{exo:g3:mult:2}

\end{solution}

\begin{solution}{exo:g3:mult:3}
दोनों आयतों में वही $24$ बिंदु हैं --- एक दूसरे को घुमाकर बना
है: $4 \times 6 = 6 \times 4$।
\end{solution}

\begin{solution}{exo:g3:mult:6}
$31 \times 5 = 30 \times 5 + 5 = 155$; \quad
$42 \times 3 = 120 + 6 = 126$; \quad
$25 \times 6 = 20 \times 6 + 5 \times 6 = 120 + 30 = 150$।
\end{solution}

\begin{solution}{exo:g3:mult:7}
$99 \times 7 = 700 - 7 = 693$; \quad
$101 \times 6 = 600 + 6 = 606$; \quad
$98 \times 5 = 500 - 10 = 490$।
\end{solution}

\begin{solution}{exo:g3:mult:8}

\end{solution}

\begin{solution}{exo:g3:mult:9}

\begin{solution}{exo:g3:mult:10}
डेस्क: $8 \times 4 = 32$। विद्यार्थी: $32 \times 2 = 64$। कमरे
में $64$ विद्यार्थी बैठ सकते हैं।
\end{solution}

\begin{solution}{exo:g3:mult:11}
पहाड़ों के वर्ग में $24$, $1 \times 24$, $2 \times 12$,
$3 \times 8$, $4 \times 6$ के रूप में दिखता है --- और यही जोड़े
उलटे भी। संभव सजावटें: $24$ की $1$ पंक्ति, $12$-$12$ की $2$
पंक्तियाँ, $8$-$8$ की $3$, $6$-$6$ की $4$, $4$-$4$ की $6$,
$3$-$3$ की $8$, $2$-$2$ की $12$, $1$-$1$ की $24$।
\end{solution}

\begin{solution}{exo:g3:mult:12}
ज़ोई $47$ को $40 + 7$ में तोड़ती है (मन में वही खानों वाला
तरीक़ा); लेआ $47$ की जगह मिलनसार $50$ रखती है और ज़्यादा गिने
गए $3 \times 6$ को घटाकर सुधार लेती है। $38 \times 4$ के लिए:
ज़ोई का तरीक़ा, $120 + 32 = 152$; लेआ का तरीक़ा,
$40 \times 4 - 2 \times 4 = 160 - 8 = 152$। दोनों काम करते
हैं --- यहाँ लेआ का तेज़ है, क्योंकि $38$, $40$ के पास है।

\end{solution}
