\section*{अध्याय \ref{ch:g3:measure} --- मापना}

\begin{solution}{exo:g3:measure:1}
कील की चौड़ाई: mm। सहपाठी की ऊँचाई: cm (या m)। विद्यालय का मैदान: m।
क़स्बों के बीच की दूरी: km।
\end{solution}

\begin{solution}{exo:g3:measure:2}
$5$ m $= 500$ cm; \quad $70$ mm $= 7$ cm; \quad
$4$ km $= 4\,000$ m; \quad $600$ cm $= 6$ m।
\end{solution}

\begin{solution}{exo:g3:measure:3}
उत्तर त्रिभुज पर निर्भर है; सीमा की कुल लंबाई तीनों मापी गई
भुजाओं का \omterm{def:g1:addition:def}{योगफल} है (इस कुल को \emph{परिमाप} कहते हैं, जिसे
\cref{ch:g4:measure} में पढ़ेंगे)।
\end{solution}

\begin{solution}{exo:g3:measure:4}
$2$ m $+ 45$ cm $= 200 + 45 = 245$ cm।

$1$ km $+ 250$ m $= 1\,000 + 250 = 1\,250$ m।

$3$ cm $+ 8$ mm $= 30 + 8 = 38$ mm।
\end{solution}

\begin{solution}{exo:g3:measure:5}
$3$ kg $= 3\,000$ g; \quad $2\,500$ g $= 2$ kg और $500$ g; \quad
$4$ L $= 400$ cL; \quad $350$ cL $= 3$ L और $50$ cL।
\end{solution}

\begin{solution}{exo:g3:measure:6}
$1$ kg $= 1\,000$ g $> 750$ g: हाँ, काफ़ी है, और
$1\,000 - 750 = 250$ g बचेगा।
\end{solution}

\begin{solution}{exo:g3:measure:7}
छोटी सुई $2$ और $3$ के बीच, लंबी सुई $6$ पर: $2{:}30$।
छोटी सुई $9$ पर, लंबी सुई $12$ पर: $9{:}00$। छोटी सुई लगभग
$5$ पर, लंबी सुई $10$ पर: $4{:}50$ (``पाँच बजने में दस मिनट''
--- घंटे की सुई अभी $5$ तक पहुँची नहीं है)।
\end{solution}

\begin{solution}{exo:g3:measure:8}
$8{:}30$ से $9{:}00$ तक: $30$ मिनट; $9{:}00$ से $11{:}45$ तक:
$2$ घंटे $45$ मिनट। कुल: $3$ घंटे $15$ मिनट।
\end{solution}

\begin{solution}{exo:g3:measure:9}
$9{:}50$ से $10{:}00$ तक: $10$ मिनट; $10{:}00$ के बाद
$45 - 10 = 35$ मिनट बाक़ी। पहुँचने का समय: $10{:}35$।
\end{solution}

\begin{solution}{exo:g3:measure:10}
$1$ L $= 100$ cL, और $100 = 20 \times 5$: पॉल $5$ पूरे
गिलास भर सकता है।
\end{solution}

\begin{solution}{exo:g3:measure:11}
हर दिन: $600 + 600 = 1\,200$ m। पाँच दिनों में:
$1\,200 \times 5 = 6\,000$ m $= 6$ km।
\end{solution}
