\section*{अध्याय \ref{ch:g3:numbers} --- 10\,000 तक की संख्याएँ}

\begin{solution}{exo:g3:numbers:1}
$2\,345$; \quad $6\,050$; \quad $9\,001$।
\end{solution}

\begin{solution}{exo:g3:numbers:2}
$1\,208$: एक हज़ार दो सौ आठ। \quad
$4\,070$: चार हज़ार सत्तर। \quad
$9\,999$: नौ हज़ार नौ सौ निन्यानवे।
\end{solution}

\begin{solution}{exo:g3:numbers:3}
दहाई का अंक: $8$। हज़ार का अंक: $7$। कुल सैकड़े:
$73$ (क्योंकि $7\,382 = 73$ सैकड़ा $+ 82$, यानी
$73 \times 100 + 82$)।
\end{solution}

\begin{solution}{exo:g3:numbers:4}
$5\,631 = 5$ हज़ार $+ 6$ सैकड़ा $+ 3$ दहाई $+ 1$ इकाई।

$2\,046 = 2$ हज़ार $+ 0$ सैकड़ा $+ 4$ दहाई $+ 6$ इकाई।

$8\,500 = 8$ हज़ार $+ 5$ सैकड़ा।
\end{solution}

\begin{solution}{exo:g3:numbers:5}
$456 < 465$ (दहाई: $5 < 6$); \quad
$1\,203 > 989$ (चार अंक तीन पर भारी); \quad
$6\,090 > 6\,009$ (दहाई: $9 > 0$); \quad
$9\,999 < 10\,000$।
\end{solution}

\begin{solution}{exo:g3:numbers:6}
$353 < 503 < 530 < 3\,500 < 5\,033$।
\end{solution}

\begin{solution}{exo:g3:numbers:7}
$150$, $100$ और $200$ के ठीक बीच बैठता है; $480$, $500$ से
ज़रा पहले; $820$, $800$ के ज़रा बाद; $990$ लगभग $1\,000$ पर।
\end{solution}

\begin{solution}{exo:g3:numbers:8}
सम: $34$, $70$, $6\,548$ (आख़िरी अंक $4$, $0$, $8$)। विषम: $57$,
$2\,481$ (आख़िरी अंक $7$, $1$)।
\end{solution}

\begin{solution}{exo:g3:numbers:9}
$2\,970$ से: $2\,980$, $2\,990$, $3\,000$, $3\,010$, $3\,020$।

$860$ से: $960$, $1\,060$, $1\,160$, $1\,260$।

$9\,996$ से: $9\,997$, $9\,998$, $9\,999$, $10\,000$ --- आख़िरी
कदम पाँच अंकों की संख्या में पहुँचा देता है: इस अध्याय की सीमा!
\end{solution}

\begin{solution}{exo:g3:numbers:10}
सबसे बड़ी: $7\,530$ (अंक घटते क्रम में)। सबसे छोटी: $3\,057$
(पहले सबसे छोटा शून्येतर अंक, फिर बाक़ी बढ़ते क्रम में --- $0$
ठीक $3$ के बाद)।
\end{solution}

\begin{solution}{exo:g3:numbers:11}
$800$ से बड़ी: सैकड़े का अंक $8$ है। सम: इकाई का अंक
$2$ ही हो सकता है ($2$, $5$, $8$ में से हर एक एक बार)। तो संख्या
$852$ है।
\end{solution}
