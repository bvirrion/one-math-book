\section*{الفصل \ref{ch:g3:numbers} --- الأعداد حتى 10\,000}

\begin{solution}{exo:g3:numbers:1}
$2\,345$; \quad $6\,050$; \quad $9\,001$.
\end{solution}

\begin{solution}{exo:g3:numbers:2}
$1\,208$: ألف ومئتان وثمانية. \quad
$4\,070$: أربعة آلاف وسبعون. \quad
$9\,999$: تسعة آلاف وتسعمئة وتسعة وتسعون.
\end{solution}

\begin{solution}{exo:g3:numbers:3}
رقم العشرات: $8$. ورقم الآلاف: $7$. والمئات التي يحتويها في
\omterm{def:g1:addition:def}{المجموع}: $73$ (لأن $7\,382 = 73$ مئة $+ 82$، أي
$73 \times 100 + 82$).
\end{solution}

\begin{solution}{exo:g3:numbers:4}
$5\,631 = 5$ آلاف $+ 6$ مئات $+ 3$ عشرات $+ 1$ واحد.

$2\,046 = 2$ ألفان $+ 0$ مئات $+ 4$ عشرات $+ 6$ آحاد.

$8\,500 = 8$ آلاف $+ 5$ مئات.
\end{solution}

\begin{solution}{exo:g3:numbers:5}
$456 < 465$ (العشرات: $5 < 6$)؛ \quad
$1\,203 > 989$ (أربعة أرقام تغلب ثلاثة)؛ \quad
$6\,090 > 6\,009$ (العشرات: $9 > 0$)؛ \quad
$9\,999 < 10\,000$.
\end{solution}

\begin{solution}{exo:g3:numbers:6}
$353 < 503 < 530 < 3\,500 < 5\,033$.
\end{solution}

\begin{solution}{exo:g3:numbers:7}
$150$ يقع في منتصف ما بين $100$ و $200$؛ و $480$ قبل
$500$ بقليل؛ و $820$ بعد $800$ بقليل؛ و $990$ يكاد يبلغ $1\,000$.
\end{solution}

\begin{solution}{exo:g3:numbers:8}
الزوجية: $34$ و $70$ و $6\,548$ (الرقم الأخير $4$ و $0$ و $8$).
والفردية: $57$ و $2\,481$ (الرقم الأخير $7$ و $1$).
\end{solution}

\begin{solution}{exo:g3:numbers:9}
من $2\,970$: $2\,980$، $2\,990$، $3\,000$، $3\,010$، $3\,020$.

ومن $860$: $960$، $1\,060$، $1\,160$، $1\,260$.

ومن $9\,996$: $9\,997$، $9\,998$، $9\,999$، $10\,000$ --- فالخطوة
الأخيرة تعبر إلى عدد من خمسة أرقام: وهو حدّ فصلنا كله!
\end{solution}

\begin{solution}{exo:g3:numbers:10}
الأكبر: $7\,530$ (الأرقام بترتيب تناقصي). والأصغر: $3\,057$
(أصغر رقم غير \omterm{def:g1:counting:zero}{الصفر} أولًا، ثم البقية تصاعديًّا --- والرقم $0$
مباشرة بعد $3$).
\end{solution}

\begin{solution}{exo:g3:numbers:11}
أكبر من $800$: فرقم المئات هو $8$. وزوجي: فرقم الآحاد
يجب أن يكون $2$ (باستعمال كل من $2$ و $5$ و $8$ مرة). إذن العدد هو
$852$.
\end{solution}
