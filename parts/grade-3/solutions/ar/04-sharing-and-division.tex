\section*{الفصل \ref{ch:g3:division} --- التقسيم والقسمة}

\begin{solution}{exo:g3:division:1}
$36 \div 4 = 9$ (لأن $4 \times 9 = 36$)؛ \quad
$63 \div 9 = 7$؛ \quad $40 \div 5 = 8$؛ \quad $56 \div 8 = 7$.
\end{solution}

\begin{solution}{exo:g3:division:2}
$17 = 5 \times 3 + 2$: \omterm{def:g3:division:remainder}{خارج القسمة} $3$، \omterm{def:g3:division:remainder}{والباقي} $2$.

$30 = 4 \times 7 + 2$: \omterm{def:g3:division:remainder}{خارج القسمة} $7$، \omterm{def:g3:division:remainder}{والباقي} $2$.

$50 = 6 \times 8 + 2$: \omterm{def:g3:division:remainder}{خارج القسمة} $8$، \omterm{def:g3:division:remainder}{والباقي} $2$.
\end{solution}

\begin{solution}{exo:g3:division:3}
$28 = 6 \times 4 + 4$: ينال كل طفل $4$ كرات، وتبقى $4$ كرات
زائدة.
\end{solution}

\begin{solution}{exo:g3:division:4}
نصف $68$: $34$. وثلث $27$: $9$. وربع $48$: $12$. ونصف
$150$: $75$.
\end{solution}

\begin{solution}{exo:g3:division:5}
«$45 \div 5 = 9$»: صواب، لأن $5 \times 9 = 45$.

«$52 \div 6 = 8$ \omterm{def:g3:division:remainder}{والباقي} $4$»: صواب، لأن $6 \times 8 + 4 = 52$ و
$4 < 6$.

«$70 \div 8 = 9$ \omterm{def:g3:division:remainder}{والباقي} $2$»: خطأ --- لأن $8 \times 9 + 2 = 74$،
لا $70$. والصواب: $70 = 8 \times 8 + 6$.
\end{solution}

\begin{solution}{exo:g3:division:6}
$32 \div 4 = 8$ صفحات. وهذا \emph{تجميع}: فالصور تُوضع في
مجموعات من $4$، ونعدّ المجموعات.
\end{solution}

\begin{solution}{exo:g3:division:7}
$75 = 9 \times 8 + 3$: ثماني قطع كاملة، ويفضل منه
$3$ سم.
\end{solution}

\begin{solution}{exo:g3:division:8}
$50 = 4 \times 12 + 2$: اثنتا عشرة سيارة تحمل $48$ طفلًا، ويبقى $2$
من الأطفال يحتاجان إلى ركوب. فتلزم $13$ سيارة.
\end{solution}

\begin{solution}{exo:g3:division:9}
$? = 7 \times 6 = 42$. \quad
$54 \div 9 = 6$، إذن القاسم الناقص هو $9$. \quad
والعدد الذي نصفه $35$ هو $70$.
\end{solution}

\begin{solution}{exo:g3:division:10}
$47 = 5 \times 9 + 2$: تنال كل صديقة $9$ ملصقات، وتحتفظ أمينة
\omterm{def:g3:division:remainder}{بالباقي}، أي $2$ من الملصقات.
\end{solution}

\begin{solution}{exo:g3:division:11}
عند القسمة على $6$، يكون \omterm{def:g3:division:remainder}{الباقي} أصغر من $6$ دائمًا: فهو
$0$ أو $1$ أو $2$ أو $3$ أو $4$ أو $5$. وأكبر عدد خارج قسمته $7$
يستعمل أكبر باقٍ: $6 \times 7 + 5 = 47$.
\end{solution}
