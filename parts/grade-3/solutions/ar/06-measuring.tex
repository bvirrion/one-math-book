\section*{الفصل \ref{ch:g3:measure} --- القياس}

\begin{solution}{exo:g3:measure:1}
عرض المسمار: مم. وطول الزميل: سم (أو م). وساحة المدرسة: م.
والمسافة بين المدينتين: كم.
\end{solution}

\begin{solution}{exo:g3:measure:2}
$5$ م $= 500$ سم؛ \quad $70$ مم $= 7$ سم؛ \quad
$4$ كم $= 4\,000$ م؛ \quad $600$ سم $= 6$ م.
\end{solution}

\begin{solution}{exo:g3:measure:3}
الجواب يتوقف على المثلث؛ والطول الكلي للحدّ هو
\omterm{def:g1:addition:def}{مجموع} الأضلاع الثلاثة المقيسة (ويسمّى هذا \omterm{def:g1:addition:def}{المجموع}
\emph{المحيط}، ويُدرس في \cref{ch:g4:measure}).
\end{solution}

\begin{solution}{exo:g3:measure:4}
$2$ م $+ 45$ سم $= 200 + 45 = 245$ سم.

$1$ كم $+ 250$ م $= 1\,000 + 250 = 1\,250$ م.

$3$ سم $+ 8$ مم $= 30 + 8 = 38$ مم.
\end{solution}

\begin{solution}{exo:g3:measure:5}
$3$ كغ $= 3\,000$ غ؛ \quad $2\,500$ غ $= 2$ كغ و $500$ غ؛ \quad
$4$ ل $= 400$ سل؛ \quad $350$ سل $= 3$ ل و $50$ سل.
\end{solution}

\begin{solution}{exo:g3:measure:6}
$1$ كغ $= 1\,000$ غ $> 750$ غ: نعم يكفيها، ويبقى
$1\,000 - 750 = 250$ غ.
\end{solution}

\begin{solution}{exo:g3:measure:7}
العقرب القصير بين $2$ و $3$، والطويل على $6$: $2{:}30$.
والقصير على $9$، والطويل على $12$: $9{:}00$. والقصير يكاد
يبلغ $5$، والطويل على $10$: $4{:}50$ («الخامسة إلا عشرًا» ---
فعقرب الساعات لم يبلغ $5$ بعد).
\end{solution}

\begin{solution}{exo:g3:measure:8}
من $8{:}30$ إلى $9{:}00$: $30$ دقيقة؛ ومن $9{:}00$ إلى $11{:}45$:
$2$ ساعة و $45$ دقيقة. \omterm{def:g1:addition:def}{والمجموع}: $3$ ساعات و $15$ دقيقة.
\end{solution}

\begin{solution}{exo:g3:measure:9}
من $9{:}50$ إلى $10{:}00$: $10$ دقائق؛ وتبقى
$45 - 10 = 35$ دقيقة بعد $10{:}00$. فالوصول: $10{:}35$.
\end{solution}

\begin{solution}{exo:g3:measure:10}
$1$ ل $= 100$ سل، و $100 = 20 \times 5$: فيستطيع سليم أن يصبّ $5$
كؤوس كاملة.
\end{solution}

\begin{solution}{exo:g3:measure:11}
في كل يوم: $600 + 600 = 1\,200$ م. وفي خمسة أيام:
$1\,200 \times 5 = 6\,000$ م $= 6$ كم.
\end{solution}
