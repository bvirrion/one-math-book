\section*{Capítulo \ref{ch:g3:numbers} --- Números até 10\,000}

\begin{solution}{exo:g3:numbers:1}
$2\,345$; \quad $6\,050$; \quad $9\,001$.
\end{solution}

\begin{solution}{exo:g3:numbers:2}
$1\,208$: mil duzentos e oito. \quad
$4\,070$: quatro mil e setenta. \quad
$9\,999$: nove mil novecentos e noventa e nove.
\end{solution}

\begin{solution}{exo:g3:numbers:3}
Algarismo das dezenas: $8$. Algarismo dos milhares: $7$. Centenas
contidas ao todo: $73$ (porque $7\,382 = 73$ centenas $+ 82$, isto
é, $73 \times 100 + 82$).
\end{solution}

\begin{solution}{exo:g3:numbers:4}
$5\,631 = 5$ milhares $+ 6$ centenas $+ 3$ dezenas $+ 1$ unidade.

$2\,046 = 2$ milhares $+ 0$ centenas $+ 4$ dezenas $+ 6$ unidades.

$8\,500 = 8$ milhares $+ 5$ centenas.
\end{solution}

\begin{solution}{exo:g3:numbers:5}
$456 < 465$ (dezenas: $5 < 6$); \quad
$1\,203 > 989$ (quatro algarismos vencem três); \quad
$6\,090 > 6\,009$ (dezenas: $9 > 0$); \quad
$9\,999 < 10\,000$.
\end{solution}

\begin{solution}{exo:g3:numbers:6}
$353 < 503 < 530 < 3\,500 < 5\,033$.
\end{solution}

\begin{solution}{exo:g3:numbers:7}
$150$ fica na metade entre $100$ e $200$; $480$, logo antes de
$500$; $820$, logo depois de $800$; $990$, quase em $1\,000$.
\end{solution}

\begin{solution}{exo:g3:numbers:8}
Pares: $34$, $70$, $6\,548$ (último algarismo $4$, $0$, $8$).
\omterm{def:g3:numbers:evenodd}{Ímpares}: $57$, $2\,481$ (último algarismo $7$, $1$).
\end{solution}

\begin{solution}{exo:g3:numbers:9}
A partir de $2\,970$: $2\,980$, $2\,990$, $3\,000$, $3\,010$,
$3\,020$.

A partir de $860$: $960$, $1\,060$, $1\,160$, $1\,260$.

A partir de $9\,996$: $9\,997$, $9\,998$, $9\,999$, $10\,000$ --- a
última etapa entra num número de cinco algarismos: o limite de todo
o nosso capítulo!
\end{solution}

\begin{solution}{exo:g3:numbers:10}
Maior: $7\,530$ (algarismos em ordem decrescente). Menor: $3\,057$ (o
menor algarismo não nulo na frente e o resto em ordem crescente ---
o $0$ logo depois do $3$).
\end{solution}

\begin{solution}{exo:g3:numbers:11}
Maior que $800$: o algarismo das centenas é $8$. Par: o algarismo das
unidades tem de ser $2$ (usando $2$, $5$ e $8$ uma vez cada). Logo o
número é $852$.
\end{solution}
