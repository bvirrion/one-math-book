\section*{Capítulo \ref{ch:g3:division} --- Partilha e divisão}

\begin{solution}{exo:g3:division:1}
$36 \div 4 = 9$ (pois $4 \times 9 = 36$); \quad
$63 \div 9 = 7$; \quad $40 \div 5 = 8$; \quad $56 \div 8 = 7$.
\end{solution}

\begin{solution}{exo:g3:division:2}
$17 = 5 \times 3 + 2$: \omterm{def:g3:division:remainder}{quociente} $3$, \omterm{def:g3:division:remainder}{resto} $2$.

$30 = 4 \times 7 + 2$: \omterm{def:g3:division:remainder}{quociente} $7$, \omterm{def:g3:division:remainder}{resto} $2$.

$50 = 6 \times 8 + 2$: \omterm{def:g3:division:remainder}{quociente} $8$, \omterm{def:g3:division:remainder}{resto} $2$.
\end{solution}

\begin{solution}{exo:g3:division:3}
$28 = 6 \times 4 + 4$: cada criança recebe $4$ bolinhas e sobram $4$
bolinhas.
\end{solution}

\begin{solution}{exo:g3:division:4}
\omterm{def:g3:division:half}{Metade} de $68$: $34$. Terça parte de $27$: $9$. \omterm{def:g3:division:half}{Quarta parte} de
$48$: $12$. \omterm{def:g3:division:half}{Metade} de $150$: $75$.
\end{solution}

\begin{solution}{exo:g3:division:5}
``$45 \div 5 = 9$'': verdadeiro, $5 \times 9 = 45$.

``$52 \div 6 = 8$ e \omterm{def:g3:division:remainder}{resto} $4$'': verdadeiro, $6 \times 8 + 4 = 52$ e
$4 < 6$.

``$70 \div 8 = 9$ e \omterm{def:g3:division:remainder}{resto} $2$'': falso --- $8 \times 9 + 2 = 74$, e
não $70$. O certo é $70 = 8 \times 8 + 6$.
\end{solution}

\begin{solution}{exo:g3:division:6}
$32 \div 4 = 8$ páginas. É um \emph{agrupamento}: as fotos são
postas em grupos de $4$ e contamos os grupos.
\end{solution}

\begin{solution}{exo:g3:division:7}
$75 = 9 \times 8 + 3$: oito pedaços completos e um pedacinho de
$3$ cm que sobra.
\end{solution}

\begin{solution}{exo:g3:division:8}
$50 = 4 \times 12 + 2$: doze carros levam $48$ crianças e ainda
faltam $2$ crianças para acomodar. São necessários $13$ carros.
\end{solution}

\begin{solution}{exo:g3:division:9}
$? = 7 \times 6 = 42$. \quad
$54 \div 9 = 6$, então o divisor que falta é $9$. \quad
O número cuja \omterm{def:g3:division:half}{metade} é $35$ é $70$.
\end{solution}

\begin{solution}{exo:g3:division:10}
$47 = 5 \times 9 + 2$: cada amigo recebe $9$ adesivos, e Amina fica
com o resto, $2$ adesivos.
\end{solution}

\begin{solution}{exo:g3:division:11}
Na divisão por $6$, o \omterm{def:g3:division:remainder}{resto} é sempre menor que $6$: pode ser $0$,
$1$, $2$, $3$, $4$ ou $5$. O maior número com \omterm{def:g3:division:remainder}{quociente} $7$ usa o
maior \omterm{def:g3:division:remainder}{resto}: $6 \times 7 + 5 = 47$.
\end{solution}
