\section*{Capítulo \ref{ch:g3:problems} --- Resolução de problemas}

\begin{solution}{exo:g3:problems:1}
Rebanhos que se juntam: adição. Repartir cerejas: divisão. Caixas de
lápis: multiplicação. Figurinhas dadas: subtração.
\end{solution}

\begin{solution}{exo:g3:problems:2}
Pergunta-se: o número de carros agora. Sabe-se: $246$ e depois mais
$158$. Operação: adição. Cálculo: $246 + 158 = 404$. Frase: há agora
$404$ carros no estacionamento.
\end{solution}

\begin{solution}{exo:g3:problems:3}
Divisão de agrupamento: $90 \div 9 = 10$. A corda dá $10$ pedaços.
\end{solution}

\begin{solution}{exo:g3:problems:4}
Multiplicação: $26 \times 8 = 208$. Os ingressos custam $208$.
\end{solution}

\begin{solution}{exo:g3:problems:5}
Barra de comparação: a barra de Emma vale $64$, a de Hugo é $29$ mais
curta. Subtração: $64 - 29 = 35$. Hugo tem $35$ adesivos.
\end{solution}

\begin{solution}{exo:g3:problems:6}
Recebidas: $6 \times 45 = 270$ maçãs. Sobram:
$270 - 178 = 92$ maçãs.
\end{solution}

\begin{solution}{exo:g3:problems:7}
Livros de ciências: $6 + 8 = 14$. Romances: $12 - 9 = 3$ a mais na
semana 2.
\end{solution}

\begin{solution}{exo:g3:problems:8}
Segunda $12$, terça $20$, quarta $12$: total $12 + 20 + 12 = 44$
bolos. O melhor dia foi terça.
\end{solution}

\begin{solution}{exo:g3:problems:9}
O número inútil é o preço ($12$). Divisão: $210 \div 35 = 6$ (pois
$35 \times 6 = 210$). Léo precisa de $6$ dias.
\end{solution}

\begin{solution}{exo:g3:problems:10}
Adultos: $2 \times 9 = 18$. Crianças: $3 \times 4 = 12$. Preço
total: $18 + 12 = 30$. Troco: $50 - 30 = 20$. Eles recebem $20$ de
volta.
\end{solution}

\begin{solution}{exo:g3:problems:11}
Há muitas respostas corretas --- por exemplo: ``Uma florista faz $8$
buquês de $15$ flores e depois retira $20$ flores murchas. Quantas
flores sobram?'' Solução: $8 \times 15 = 120$ e depois
$120 - 20 = 100$.
\end{solution}
