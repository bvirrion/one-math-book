\section*{Hoofdstuk \ref{ch:g3:addsub} --- Optellen en aftrekken}

\begin{solution}{exo:g3:addsub:1}
$346 + 253 = 599$; \quad
$478 + 265 = 743$; \quad
$1\,536 + 2\,876 = 4\,412$.
\end{solution}

\begin{solution}{exo:g3:addsub:2}
$587 - 234 = 353$ (controle: $353 + 234 = 587$).

$805 - 348 = 457$ (controle: $457 + 348 = 805$).

$1\,000 - 463 = 537$ (controle: $537 + 463 = 1\,000$).
\end{solution}

\begin{solution}{exo:g3:addsub:3}
$38 + 7 = 38 + 2 + 5 = 45$; \quad
$56 + 9 = 56 + 4 + 5 = 65$; \quad
$45 + 28 = 45 + 30 - 2 = 73$.
\end{solution}

\begin{solution}{exo:g3:addsub:4}

\begin{solution}{exo:g3:addsub:5}
Van $47$ naar $50$: $3$; van $50$ naar $63$: $13$; samen
$3 + 13 = 16$. Het cijferen bevestigt het: $63 - 47 = 16$.
\end{solution}

\begin{solution}{exo:g3:addsub:6}
$492 + 218$: schatting $500 + 200 = 700$; precies $710$.

$913 - 388$: schatting $900 - 400 = 500$; precies $525$.
\end{solution}

\begin{solution}{exo:g3:addsub:7}

\end{solution}

\begin{solution}{exo:g3:addsub:8}
$224 - 178 = 46$. Nora moet nog $46$ bladzijden lezen.
\end{solution}

\begin{solution}{exo:g3:addsub:9}
Eenheden: $7 + \text{?} = 2$ kan niet, dus $7 + 5 = 12$: in het gat bij
de eenheden komt $5$, en je onthoudt $1$. Tientallen:
$\text{?} + 2 + 1 = 6$: in het gat bij de tientallen komt $3$ (niets te
onthouden). Honderdtallen: $3 + \text{?} = 8$: in het gat bij de
honderdtallen komt $5$. Dus $337 + 525 = 862$, en bij elk gat paste
precies één cijfer --- er is maar één manier.
\end{solution}

\begin{solution}{exo:g3:addsub:10}
Verkocht: $145 + 89 = 234$ croissants. Gebakken: verkocht plus de rest,
$234 + 17 = 251$ croissants.
\end{solution}

\begin{solution}{exo:g3:addsub:11}
Hij heeft gelijk: $500 - 200 = 300$, plus $1$ geeft $301$; en met
cijferen is $500 - 199$ ook $301$. Door $200$ weg te halen haal je er
\emph{één te veel} weg, dus moet je er één teruggeven --- een
vriendelijk buurgetal aftrekken en dan verbeteren is vaak sneller dan
lenen.
\end{solution}
