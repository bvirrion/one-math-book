\section*{Hoofdstuk \ref{ch:g3:problems} --- Vraagstukken oplossen}

\begin{solution}{exo:g3:problems:1}
Kuddes die samengaan: optelling. Kersen verdelen: deling. Dozen potloden:
vermenigvuldiging. Kaarten weggeven: aftrekking.
\end{solution}

\begin{solution}{exo:g3:problems:2}
Gevraagd: het aantal auto's nu. Bekend: $246$, en dan $158$ erbij.
Bewerking: optelling. Uitrekenen: $246 + 158 = 404$. Antwoordzin: er staan
nu $404$ auto's op het parkeerterrein.
\end{solution}

\begin{solution}{exo:g3:problems:3}
Een deling waarbij je groepeert: $90 \div 9 = 10$. Het touw geeft $10$
stukken.
\end{solution}

\begin{solution}{exo:g3:problems:4}
Vermenigvuldiging: $26 \times 8 = 208$. De kaartjes kosten $208$.
\end{solution}

\begin{solution}{exo:g3:problems:5}
Vergelijkingsstrookjes: het strookje van Emma is $64$, dat van Hugo is
$29$ korter. Aftrekking: $64 - 29 = 35$. Hugo heeft $35$ stickers.
\end{solution}

\begin{solution}{exo:g3:problems:6}
Ontvangen: $6 \times 45 = 270$ appels. Over: $270 - 178 = 92$ appels.
\end{solution}

\begin{solution}{exo:g3:problems:7}
Non-fictieboeken: $6 + 8 = 14$. Romans: $12 - 9 = 3$ meer in week 2.
\end{solution}

\begin{solution}{exo:g3:problems:8}
Maandag $12$, dinsdag $20$, woensdag $12$: in totaal
$12 + 20 + 12 = 44$ gebakjes. De beste dag was dinsdag.
\end{solution}

\begin{solution}{exo:g3:problems:9}
Het getal te veel is de prijs ($12$). Deling: $210 \div 35 = 6$ (want
$35 \times 6 = 210$). Leo heeft $6$ dagen nodig.
\end{solution}

\begin{solution}{exo:g3:problems:10}
Volwassenen: $2 \times 9 = 18$. Kinderen: $3 \times 4 = 12$. Totaalprijs:
$18 + 12 = 30$. Wisselgeld: $50 - 30 = 20$. Ze krijgen $20$ terug.
\end{solution}

\begin{solution}{exo:g3:problems:11}
Er zijn veel goede antwoorden --- bijvoorbeeld: ``Een bloemist maakt $8$
bossen van $15$ bloemen en haalt er daarna $20$ verwelkte bloemen uit.
Hoeveel bloemen blijven over?'' Oplossing: $8 \times 15 = 120$, en dan
$120 - 20 = 100$.
\end{solution}
