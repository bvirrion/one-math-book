\section*{Capítulo \ref{ch:g3:numbers} --- Los números hasta 10\,000}

\begin{solution}{exo:g3:numbers:1}
$2\,345$; \quad $6\,050$; \quad $9\,001$.
\end{solution}

\begin{solution}{exo:g3:numbers:2}
$1\,208$: mil doscientos ocho. \quad
$4\,070$: cuatro mil setenta. \quad
$9\,999$: nueve mil novecientos noventa y nueve.
\end{solution}

\begin{solution}{exo:g3:numbers:3}
Cifra de las decenas: $8$. Cifra de los millares: $7$. Centenas contenidas en total:
$73$ (porque $7\,382 = 73$ centenas $+ 82$, es decir,
$73 \times 100 + 82$).
\end{solution}

\begin{solution}{exo:g3:numbers:4}
$5\,631 = 5$ millares $+ 6$ centenas $+ 3$ decenas $+ 1$ unidad.

$2\,046 = 2$ millares $+ 0$ centenas $+ 4$ decenas $+ 6$ unidades.

$8\,500 = 8$ millares $+ 5$ centenas.
\end{solution}

\begin{solution}{exo:g3:numbers:5}
$456 < 465$ (decenas: $5 < 6$); \quad
$1\,203 > 989$ (cuatro cifras ganan a tres); \quad
$6\,090 > 6\,009$ (decenas: $9 > 0$); \quad
$9\,999 < 10\,000$.
\end{solution}

\begin{solution}{exo:g3:numbers:6}
$353 < 503 < 530 < 3\,500 < 5\,033$.
\end{solution}

\begin{solution}{exo:g3:numbers:7}
$150$ queda a mitad de camino entre $100$ y $200$; $480$ justo antes de
$500$; $820$ justo después de $800$; $990$ casi en $1\,000$.
\end{solution}

\begin{solution}{exo:g3:numbers:8}
\omterm{def:g3:numbers:evenodd}{Pares}: $34$, $70$ y $6\,548$ (última cifra $4$, $0$, $8$). \omterm{def:g3:numbers:evenodd}{Impares}: $57$
y $2\,481$ (última cifra $7$, $1$).
\end{solution}

\begin{solution}{exo:g3:numbers:9}
Desde $2\,970$: $2\,980$, $2\,990$, $3\,000$, $3\,010$, $3\,020$.

Desde $860$: $960$, $1\,060$, $1\,160$, $1\,260$.

Desde $9\,996$: $9\,997$, $9\,998$, $9\,999$, $10\,000$ --- el último
paso entra en un número de cinco cifras: ¡el límite de todo el capítulo!
\end{solution}

\begin{solution}{exo:g3:numbers:10}
El mayor: $7\,530$ (cifras en orden decreciente). El menor: $3\,057$
(la cifra más pequeña distinta de \omterm{def:g1:counting:zero}{cero} primero y luego las demás en orden creciente --- con el $0$
justo detrás del $3$).
\end{solution}

\begin{solution}{exo:g3:numbers:11}
Mayor que $800$: la cifra de las centenas es $8$. \omterm{def:g3:numbers:evenodd}{Par}: la cifra de las unidades
tiene que ser $2$ (usando $2$, $5$ y $8$ una vez cada una). Así que el número es
$852$.
\end{solution}
