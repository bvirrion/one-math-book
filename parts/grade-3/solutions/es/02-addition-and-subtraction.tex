\section*{Capítulo \ref{ch:g3:addsub} --- La suma y la resta}

\begin{solution}{exo:g3:addsub:1}
$346 + 253 = 599$; \quad
$478 + 265 = 743$; \quad
$1\,536 + 2\,876 = 4\,412$.
\end{solution}

\begin{solution}{exo:g3:addsub:2}
$587 - 234 = 353$ (comprobación: $353 + 234 = 587$).

$805 - 348 = 457$ (comprobación: $457 + 348 = 805$).

$1\,000 - 463 = 537$ (comprobación: $537 + 463 = 1\,000$).
\end{solution}

\begin{solution}{exo:g3:addsub:3}

\end{solution}

\begin{solution}{exo:g3:addsub:5}
Del $47$ al $50$: $3$; del $50$ al $63$: $13$; en total,
$3 + 13 = 16$. Las columnas lo confirman: $63 - 47 = 16$.
\end{solution}

\begin{solution}{exo:g3:addsub:9}
Unidades: $7 + \text{?} = 2$ es imposible, así que $7 + 5 = 12$: el hueco de las
unidades es $5$ y se lleva $1$. Decenas: $\text{?} + 2 + 1 = 6$: el hueco de las decenas
es $3$ (sin llevada). Centenas: $3 + \text{?} = 8$: el hueco de las centenas es
$5$. Así, $337 + 525 = 862$, y cada hueco admitía una sola
cifra posible --- solo hay una manera.
\end{solution}

\begin{solution}{exo:g3:addsub:10}
Vendidos: $145 + 89 = 234$ cruasanes. Horneados: los vendidos más los no vendidos,
$234 + 17 = 251$ cruasanes.
\end{solution}

\begin{solution}{exo:g3:addsub:11}
Tiene razón: $500 - 200 = 300$ y, sumando $1$, da $301$; en columna,
$500 - 199 = 301$ también. Quitar $200$ quita \emph{uno de más},
así que hay que devolverlo --- restar una vecina amable y
corregir suele ser más rápido que pedir prestado.
\end{solution}
