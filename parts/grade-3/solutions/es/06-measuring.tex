\section*{Capítulo \ref{ch:g3:measure} --- Medir}

\begin{solution}{exo:g3:measure:1}
Ancho de una uña: mm. Altura de un compañero: cm (o m). Patio del colegio: m.
Distancia entre pueblos: km.
\end{solution}

\begin{solution}{exo:g3:measure:2}
$5$ m $= 500$ cm; \quad $70$ mm $= 7$ cm; \quad
$4$ km $= 4\,000$ m; \quad $600$ cm $= 6$ m.
\end{solution}

\begin{solution}{exo:g3:measure:3}
Depende del triángulo; la longitud total del contorno es la
\omterm{def:g1:addition:def}{suma} de los tres lados medidos (ese total se llama
\emph{perímetro} y se estudia en el \cref{ch:g4:measure}).
\end{solution}

\begin{solution}{exo:g3:measure:4}
$2$ m $+ 45$ cm $= 200 + 45 = 245$ cm.

$1$ km $+ 250$ m $= 1\,000 + 250 = 1\,250$ m.

$3$ cm $+ 8$ mm $= 30 + 8 = 38$ mm.
\end{solution}

\begin{solution}{exo:g3:measure:5}
$3$ kg $= 3\,000$ g; \quad $2\,500$ g $= 2$ kg y $500$ g; \quad
$4$ L $= 400$ cL; \quad $350$ cL $= 3$ L y $50$ cL.
\end{solution}

\begin{solution}{exo:g3:measure:6}
$1$ kg $= 1\,000$ g $> 750$ g: sí, le llega, y le sobrarán
$1\,000 - 750 = 250$ g.
\end{solution}

\begin{solution}{exo:g3:measure:7}
Aguja corta entre el $2$ y el $3$ y larga en el $6$: $2{:}30$.
Aguja corta en el $9$ y larga en el $12$: $9{:}00$. Aguja corta casi en
el $5$ y larga en el $10$: $4{:}50$ («las cinco menos diez» --- la aguja de las horas
aún no ha llegado al $5$).
\end{solution}

\begin{solution}{exo:g3:measure:8}
De las $8{:}30$ a las $9{:}00$: $30$ min; de las $9{:}00$ a las $11{:}45$:
$2$ h $45$ min. Total: $3$ h $15$ min.
\end{solution}

\begin{solution}{exo:g3:measure:9}
De las $9{:}50$ a las $10{:}00$: $10$ min; después de las $10{:}00$ quedan
$45 - 10 = 35$ min. Llegada: $10{:}35$.
\end{solution}

\begin{solution}{exo:g3:measure:10}
$1$ L $= 100$ cL y $100 = 20 \times 5$: Pablo puede servir $5$ vasos
llenos.
\end{solution}

\begin{solution}{exo:g3:measure:11}
Cada día: $600 + 600 = 1\,200$ m. En cinco días:
$1\,200 \times 5 = 6\,000$ m $= 6$ km.
\end{solution}
