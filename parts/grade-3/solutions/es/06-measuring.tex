\section*{Capítulo \ref{ch:g3:measure} --- Medidas}

\begin{solution}{exo:g3:measure:1}
Ancho de clavo: mm. Altura del compañero: cm (o m). Patio de la escuela: m.
Distancia entre pueblos: km.
\end{solution}

\begin{solution}{exo:g3:measure:2}
$5$ metro $= 500$ centímetro; \quad $70$ milímetros $= 7$ centímetro; \quad
$4$ kilómetros $= 4\,000$ metro; \quad $600$ centímetro $= 6$ metro.
\end{solution}

\begin{solution}{exo:g3:measure:3}
Las respuestas dependen del triángulo; la longitud total de la frontera es la
\omterm{def:g1:addition:def}{suma} de los tres lados medidos (este total se llama
\emph{perímetro}, estudiado en \cref{ch:g4:measure}).
\end{solution}

\begin{solution}{exo:g3:measure:4}
$2$ metro $+ 45$ centímetro $= 200 + 45 = 245$ centímetro.

$1$ kilómetros $+ 250$ metro $= 1\,000 + 250 = 1\,250$ metro.

$3$ centímetro $+ 8$ milímetros $= 30 + 8 = 38$ mm.
\end{solution}

\begin{solution}{exo:g3:measure:5}
$3$ kilos $= 3\,000$ gramo; \quad $2\,500$ gramo $= 2$ kilos y $500$ gramo; \quad
$4$ l $= 400$ CL; \quad $350$ CL $= 3$ Tierra $50$ CL.
\end{solution}

\begin{solution}{exo:g3:measure:6}
$1$ kilos $= 1\,000$ gramo $> 750$ g: si, ella tiene suficiente, y
$1\,000 - 750 = 250$ quedará g.
\end{solution}

\begin{solution}{exo:g3:measure:7}
mano corta entre $2$ y $3$, mano larga sobre el $6$: $2{:}30$.
mano corta $9$, mano larga en $12$: $9{:}00$. Mano corta casi en
$5$, mano larga en $10$: $4{:}50$ ("cinco menos diez'' --- la manecilla de las horas
no ha llegado a la $5$ todavía).
\end{solution}

\begin{solution}{exo:g3:measure:8}
De $8{:}30$ a $9{:}00$: $30$ mín; de $9{:}00$ a $11{:}45$:
$2$ h $45$ mín. Total: $3$ h $15$ mín.
\end{solution}

\begin{solution}{exo:g3:measure:9}
De $9{:}50$ a $10{:}00$: $10$ mín; quedan
$45 - 10 = 35$ minutos después $10{:}00$. Llegada: $10{:}35$.
\end{solution}

\begin{solution}{exo:g3:measure:10}
$1$ l $= 100$ cl, y $100 = 20 \times 5$: Paul puede servir $5$ lleno
gafas.
\end{solution}

\begin{solution}{exo:g3:measure:11}
Cada día: $600 + 600 = 1\,200$ metro. En cinco días:
$1\,200 \times 5 = 6\,000$ metro $= 6$ km.
\end{solution}
