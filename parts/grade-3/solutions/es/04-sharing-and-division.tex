\section*{Capítulo \ref{ch:g3:division} --- Repartir y dividir}

\begin{solution}{exo:g3:division:1}
$36 \div 4 = 9$ (porque $4 \times 9 = 36$); \quad
$63 \div 9 = 7$; \quad $40 \div 5 = 8$; \quad $56 \div 8 = 7$.
\end{solution}

\begin{solution}{exo:g3:division:2}
$17 = 5 \times 3 + 2$: \omterm{def:g3:division:remainder}{cociente} $3$, \omterm{def:g3:division:remainder}{resto} $2$.

$30 = 4 \times 7 + 2$: \omterm{def:g3:division:remainder}{cociente} $7$, \omterm{def:g3:division:remainder}{resto} $2$.

$50 = 6 \times 8 + 2$: \omterm{def:g3:division:remainder}{cociente} $8$, \omterm{def:g3:division:remainder}{resto} $2$.
\end{solution}

\begin{solution}{exo:g3:division:3}
$28 = 6 \times 4 + 4$: cada niño recibe $4$ canicas y sobran
$4$ canicas.
\end{solution}

\begin{solution}{exo:g3:division:4}
La \omterm{def:g3:division:half}{mitad} de $68$: $34$. El tercio de $27$: $9$. El \omterm{def:g3:division:half}{cuarto} de $48$: $12$. La \omterm{def:g3:division:half}{mitad}
de $150$: $75$.
\end{solution}

\begin{solution}{exo:g3:division:5}
«$45 \div 5 = 9$»: verdadero, $5 \times 9 = 45$.

«$52 \div 6 = 8$ y \omterm{def:g3:division:remainder}{resto} $4$»: verdadero, $6 \times 8 + 4 = 52$ y
$4 < 6$.

«$70 \div 8 = 9$ y \omterm{def:g3:division:remainder}{resto} $2$»: falso --- $8 \times 9 + 2 = 74$,
no $70$. Lo correcto: $70 = 8 \times 8 + 6$.
\end{solution}

\begin{solution}{exo:g3:division:6}
$32 \div 4 = 8$ páginas. Es un \emph{agrupamiento}: las fotos se colocan
en grupos de $4$ y contamos los grupos.
\end{solution}

\begin{solution}{exo:g3:division:7}
$75 = 9 \times 8 + 3$: ocho trozos completos y un pedazo sobrante de
$3$ cm.
\end{solution}

\begin{solution}{exo:g3:division:8}
$50 = 4 \times 12 + 2$: doce coches llevan a $48$ niños y quedan $2$
niños sin sitio. Hacen falta $13$ coches.
\end{solution}

\begin{solution}{exo:g3:division:9}
$? = 7 \times 6 = 42$. \quad
$54 \div 9 = 6$, así que el divisor que falta es $9$. \quad
El número cuya \omterm{def:g3:division:half}{mitad} es $35$ es $70$.
\end{solution}

\begin{solution}{exo:g3:division:10}
$47 = 5 \times 9 + 2$: cada amigo recibe $9$ pegatinas y Amina
se queda con el \omterm{def:g3:division:remainder}{resto}, $2$ pegatinas.
\end{solution}

\begin{solution}{exo:g3:division:11}
Al dividir entre $6$, el \omterm{def:g3:division:remainder}{resto} es siempre menor que $6$: puede ser
$0$, $1$, $2$, $3$, $4$ o $5$. El mayor número con \omterm{def:g3:division:remainder}{cociente} $7$
usa el mayor \omterm{def:g3:division:remainder}{resto}: $6 \times 7 + 5 = 47$.
\end{solution}
