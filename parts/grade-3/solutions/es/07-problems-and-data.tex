\section*{Capítulo \ref{ch:g3:problems} --- Resolver problemas}

\begin{solution}{exo:g3:problems:1}
Rebaños que se juntan: \omterm{def:g1:addition:def}{suma}. Repartir cerezas: división. Cajas de
lápices: multiplicación. Cromos regalados: resta.
\end{solution}

\begin{solution}{exo:g3:problems:2}
Se pregunta: cuántos coches hay ahora. Se sabe: $246$ y luego $158$ más.
Operación: \omterm{def:g1:addition:def}{suma}. Cálculo: $246 + 158 = 404$. Frase: ahora hay
$404$ coches en el aparcamiento.
\end{solution}

\begin{solution}{exo:g3:problems:3}
División de agrupamiento: $90 \div 9 = 10$. De la cuerda salen $10$ trozos.
\end{solution}

\begin{solution}{exo:g3:problems:4}
Multiplicación: $26 \times 8 = 208$. Las entradas cuestan $208$.
\end{solution}

\begin{solution}{exo:g3:problems:5}
Esquema de comparación: la barra de Elena, $64$; la de Hugo, $29$ más corta.
Resta: $64 - 29 = 35$. Hugo tiene $35$ pegatinas.
\end{solution}

\begin{solution}{exo:g3:problems:6}
Recibidas: $6 \times 45 = 270$ manzanas. Quedan:
$270 - 178 = 92$ manzanas.
\end{solution}

\begin{solution}{exo:g3:problems:7}
Libros de ciencia: $6 + 8 = 14$. Novelas: $12 - 9 = 3$ más en la semana 2.
\end{solution}

\begin{solution}{exo:g3:problems:8}
Lunes $12$, martes $20$, miércoles $12$: en total,
$12 + 20 + 12 = 44$ pasteles. El mejor día fue el martes.
\end{solution}

\begin{solution}{exo:g3:problems:9}
El número que sobra es el precio ($12$). División:
$210 \div 35 = 6$ (porque $35 \times 6 = 210$). Leo necesita $6$ días.
\end{solution}

\begin{solution}{exo:g3:problems:10}
Adultos: $2 \times 9 = 18$. Niños: $3 \times 4 = 12$. Precio total:
$18 + 12 = 30$. Cambio: $50 - 30 = 20$. Les devuelven $20$.
\end{solution}

\begin{solution}{exo:g3:problems:11}
Hay muchas respuestas correctas --- por ejemplo: «Una florista hace $8$ ramos
de $15$ flores y luego quita $20$ flores marchitas. ¿Cuántas flores
quedan?» Solución: $8 \times 15 = 120$ y después $120 - 20 = 100$.
\end{solution}
