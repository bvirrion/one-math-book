\section*{Bab \ref{ch:g3:measure} --- Mengukur}

\begin{solution}{exo:g3:measure:1}
Lebar paku: mm. Tinggi teman sekelas: cm (atau m). Halaman sekolah:
m. Jarak antara dua kota: km.
\end{solution}

\begin{solution}{exo:g3:measure:2}
$5$ m $= 500$ cm; \quad $70$ mm $= 7$ cm; \quad
$4$ km $= 4\,000$ m; \quad $600$ cm $= 6$ m.
\end{solution}

\begin{solution}{exo:g3:measure:3}
Jawabannya bergantung pada segitiganya; panjang seluruh tepinya
adalah \omterm{def:g1:addition:def}{jumlah} ketiga sisi yang diukur (\omterm{def:g1:addition:def}{jumlah} ini disebut
\emph{keliling}, yang dipelajari pada \cref{ch:g4:measure}).
\end{solution}

\begin{solution}{exo:g3:measure:4}
$2$ m $+ 45$ cm $= 200 + 45 = 245$ cm.

$1$ km $+ 250$ m $= 1\,000 + 250 = 1\,250$ m.

$3$ cm $+ 8$ mm $= 30 + 8 = 38$ mm.
\end{solution}

\begin{solution}{exo:g3:measure:5}
$3$ kg $= 3\,000$ g; \quad $2\,500$ g $= 2$ kg dan $500$ g; \quad
$4$ L $= 400$ cL; \quad $350$ cL $= 3$ L dan $50$ cL.
\end{solution}

\begin{solution}{exo:g3:measure:6}
$1$ kg $= 1\,000$ g $> 750$ g: ya, tepungnya cukup, dan
$1\,000 - 750 = 250$ g akan tersisa.
\end{solution}

\begin{solution}{exo:g3:measure:7}
Jarum pendek antara $2$ dan $3$, jarum panjang di angka $6$:
$2{:}30$. Jarum pendek di $9$, jarum panjang di $12$: $9{:}00$.
Jarum pendek hampir di $5$, jarum panjang di $10$: $4{:}50$ (``pukul
lima kurang sepuluh'' --- jarum jamnya belum mencapai angka $5$).
\end{solution}

\begin{solution}{exo:g3:measure:8}
Dari $8{:}30$ ke $9{:}00$: $30$ menit; dari $9{:}00$ ke $11{:}45$:
$2$ jam $45$ menit. Seluruhnya: $3$ jam $15$ menit.
\end{solution}

\begin{solution}{exo:g3:measure:9}
Dari $9{:}50$ ke $10{:}00$: $10$ menit; masih tersisa
$45 - 10 = 35$ menit setelah $10{:}00$. Tibanya: $10{:}35$.
\end{solution}

\begin{solution}{exo:g3:measure:10}
$1$ L $= 100$ cL, dan $100 = 20 \times 5$: Pandu dapat menuangkan
$5$ gelas penuh.
\end{solution}

\begin{solution}{exo:g3:measure:11}
Setiap hari: $600 + 600 = 1\,200$ m. Dalam lima hari:
$1\,200 \times 5 = 6\,000$ m $= 6$ km.

\end{solution}
