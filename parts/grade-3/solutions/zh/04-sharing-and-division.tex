\section*{第 \ref{ch:g3:division} 章——分享与除法}

\begin{solution}{exo:g3:division:1}
$36 \div 4 = 9$（因为 $4 \times 9 = 36$）；\quad
$63 \div 9 = 7$; \quad $40 \div 5 = 8$; \quad $56 \div 8 = 7$.
\end{solution}

\begin{solution}{exo:g3:division:2}
$17 = 5 \times 3 + 2$：商 $3$，\omterm{def:g3:division:remainder}{余数} $2$。

$30 = 4 \times 7 + 2$：商 $7$，\omterm{def:g3:division:remainder}{余数} $2$。

$50 = 6 \times 8 + 2$：商 $8$，\omterm{def:g3:division:remainder}{余数} $2$。
\end{solution}

\begin{solution}{exo:g3:division:3}
$28 = 6 \times 4 + 4$：每个孩子分得 $4$ 颗弹珠，
还剩 $4$ 颗。
\end{solution}

\begin{solution}{exo:g3:division:4}
$68$ 的\omterm{def:g3:division:half}{一半}是 $34$。$27$ 的三分之一是 $9$。$48$ 的\omterm{def:g3:division:half}{四分之一}是 $12$。
$150$ 的\omterm{def:g3:division:half}{一半}是 $75$。
\end{solution}

\begin{solution}{exo:g3:division:5}
“$45 \div 5 = 9$”：正确，$5 \times 9 = 45$。

“$52 \div 6 = 8$ 余 $4$”：正确，$6 \times 8 + 4 = 52$，并且
$4 < 6$。

“$70 \div 8 = 9$ 余 $2$”：错误——$8 \times 9 + 2 = 74$，
不是 $70$。正确的是 $70 = 8 \times 8 + 6$。
\end{solution}

\begin{solution}{exo:g3:division:6}
$32 \div 4 = 8$ 页。这是\emph{分组}：照片
每 $4$ 张分为一组，再数出共有几组。
\end{solution}

\begin{solution}{exo:g3:division:7}
$75 = 9 \times 8 + 3$：有 8 段完整的，还剩一小段，
长 $3$ 厘米。
\end{solution}

\begin{solution}{exo:g3:division:8}
$50 = 4 \times 12 + 2$：12 辆车共载 $48$ 名儿童，还有 $2$ 名
儿童需要搭车，共需 $13$ 辆车。
\end{solution}

\begin{solution}{exo:g3:division:9}
$? = 7 \times 6 = 42$. \quad
$54 \div 9 = 6$，所以缺少的除数是 $9$。\quad
\omterm{def:g3:division:half}{一半}是 $35$ 的数是 $70$。
\end{solution}

\begin{solution}{exo:g3:division:10}
$47 = 5 \times 9 + 2$：每位朋友分得 $9$ 张贴纸，
阿米娜留下余下的 $2$ 张贴纸。
\end{solution}

\begin{solution}{exo:g3:division:11}
除以 $6$ 时，\omterm{def:g3:division:remainder}{余数}总小于 $6$，
它可以是 $0$、$1$、$2$、$3$、$4$ 或 $5$。商为 $7$ 时，
最大的数要取最大的\omterm{def:g3:division:remainder}{余数}：$6 \times 7 + 5 = 47$。
\end{solution}
