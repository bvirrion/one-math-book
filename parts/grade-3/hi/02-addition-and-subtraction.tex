\chapter{जोड़ और घटाव}\label{ch:g3:addsub}

जोड़ना यानी मिलाना; घटाना यानी निकालना, या यह पता करना कि कितना
कम है। यह अध्याय खानों वाले दोनों तरीक़े सिखाता है --- अपने
मशहूर हासिल और उधार के साथ --- और वे मानसिक तरकीबें भी, जिनसे
अक्सर खाने बनाने की ज़रूरत ही नहीं पड़ती।

\section{खानों में जोड़}

\begin{method}[खानों में जोड़ना]\label{met:g3:addsub:add}
\begin{enumerate}
  \item संख्याएँ एक के नीचे एक लिखो, इकाई के नीचे इकाई, दहाई
        के नीचे दहाई (\emph{दाईं} ओर से मिलाकर);
  \item इकाइयाँ जोड़ो; \omterm{def:g1:addition:def}{योगफल} $9$ से आगे निकले तो उसका इकाई अंक
        लिखो और दहाई को अगले खाने में \emph{हासिल} भेजो (छोटा $1$);
  \item इसी तरह दहाई, सैकड़ा, \dots\ करते जाओ, हर बार हासिल
        जोड़ते हुए।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:addsub:add}
$457 + 368$ निकालो:
\[
\begin{array}{r}
{\scriptstyle 1}\ \ {\scriptstyle 1}\ \ \phantom{0} \\[-3pt]
4\,5\,7 \\
+\ 3\,6\,8 \\
\hline
8\,2\,5
\end{array}
\]
इकाई: $7 + 8 = 15$: $5$ लिखो, $1$ हासिल। दहाई:
$5 + 6 + 1 = 12$: $2$ लिखो, $1$ हासिल। सैकड़ा: $4 + 3 + 1 = 8$।
नतीजा: $825$।
\end{example}

\begin{example}[मन-ही-मन जोड़]\label{ex:g3:addsub:mental}
खानों तक पहुँचने से पहले मिलनसार संख्याएँ ढूँढ़ो:
\begin{itemize}
  \item \emph{दहाई बनाओ}: $47 + 8 = 47 + 3 + 5 = 50 + 5 = 55$;
  \item \emph{हिस्सों में छलाँग}: $256 + 130 = 256 + 100 + 30 = 386$;
  \item \emph{क्रम बदलो}: $17 + 59 + 3 = (17 + 3) + 59 = 79$।
\end{itemize}
\end{example}

\section{खानों में घटाव}

\begin{method}[खानों में घटाना]\label{met:g3:addsub:sub}
\begin{enumerate}
  \item बड़ी संख्या ऊपर लिखो, दाईं ओर से मिलाकर;
  \item इकाई से शुरू करके खाना-दर-खाना घटाओ;
  \item ऊपर का अंक छोटा पड़े तो \emph{उधार} लो: ऊपर वाली पंक्ति
        के अगले स्थान से एक लो (इस खाने में वह दस के बराबर
        होता है);
  \item \emph{जाँचो}: नतीजे में घटाई गई संख्या जोड़ने पर ऊपर
        वाली संख्या वापस मिलनी चाहिए।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:addsub:sub}
$603 - 147$ निकालो:
\[
\begin{array}{r}
6\,0\,3 \\
-\ 1\,4\,7 \\
\hline
4\,5\,6
\end{array}
\]
इकाई: $3 - 7$ नहीं हो सकता; उधार लेकर आगे बढ़ो: $603$ बन जाता है
$5$ सैकड़ा, $9$ दहाई, $13$ इकाई। फिर $13 - 7 = 6$;
दहाई $9 - 4 = 5$; सैकड़ा $5 - 1 = 4$। नतीजा: $456$।
जाँच: $456 + 147 = 603$।
\end{example}

बनता है, और उनमें से एक दहाई खुलकर दस इकाई। राशि वही, कपड़े
नए --- $6\,\vert\,0\,\vert\,3$ बन जाता है
$5\,\vert\,9\,\vert\,13$, और अब हर खाने में घटाव हो सकता है।}

\begin{example}[घटाव यानी लापता हिस्सा]\label{ex:g3:addsub:missing}
``लीला के पास $35$ कंचे हैं, टॉम के पास $52$: टॉम के पास कितने
ज़्यादा?'' यह घटाव $52 - 35$ है। मन-ही-मन $35$ से \emph{आगे}
गिनो: $35 \to 40$ पर $5$, फिर $40 \to 52$ पर $12$; कुल
$5 + 12 = 17$। आगे गिनना अक्सर उधार लेने से आसान होता है।
\end{example}

  \draw[-Stealth, omDef, thick] (1.0,0.25) .. controls (2.2,0.9) ..

तक। उत्तर छलाँगों का जोड़ है:
$5 + 12 = 17$।}
\end{omfigure}

\section{परिमाण की कोटि}

\begin{method}[गणना से पहले अंदाज़ा]\label{met:g3:addsub:estimate}
खानों में कोई भी गणना करने से पहले संख्याओं को मिलनसार संख्याओं
तक गोल करके एक \emph{अंदाज़ा} लगा लो। गणना के बाद तुलना करो:
अंदाज़े से बहुत दूर का नतीजा किसी ग़लती का संकेत है।
\end{method}

\begin{example}
$487 + 312$ के लिए अंदाज़ा $500 + 300 = 800$ है। खानों का
नतीजा $799$ उसके पास है: भरोसेमंद। अगर $1\,099$ मिला होता
(खाने ग़लत मिलाने पर), तो अंदाज़ा चेतावनी दे देता।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:addsub:1}
खानों में निकालो: $346 + 253$; \quad $478 + 265$; \quad
$1\,536 + 2\,876$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:2}
खानों में निकालो, फिर जोड़ से जाँचो: $587 - 234$; \quad
$805 - 348$; \quad $1\,000 - 463$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:3}
दहाई बनाकर मन-ही-मन निकालो: $38 + 7$; \quad $56 + 9$; \quad
$45 + 28$ ($30$ जोड़ो, $2$ घटाओ)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:4}
छलाँगों से मन-ही-मन निकालो: $340 + 220$; \quad $675 + 210$; \quad
$512 - 300$; \quad $840 - 220$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:5}
$63 - 47$ को आगे गिनकर निकालो (जैसे
\cref{ex:g3:addsub:missing} में), फिर खानों में जाँचो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:6}
पहले अंदाज़ा लगाओ (सैकड़ों तक गोल करो), फिर ठीक-ठीक निकालो:
$492 + 218$; \quad $913 - 388$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:7}
एक विद्यालय में $348$ लड़कियाँ और $296$ लड़के हैं। कुल कितने
विद्यार्थी? संक्रिया लिखो, निकालो, और एक वाक्य में उत्तर दो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:8}
एक किताब में $224$ पन्ने हैं; नोरा ने $178$ पढ़ लिए हैं। कितने
पन्ने पढ़ने बाक़ी हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:addsub:9}
ख़ाली जगह भरो (खाना-दर-खाना काम करो):
\[
\begin{array}{r}
3\,?\,7 \\
+\ ?\,2\,? \\
\hline
8\,6\,2
\end{array}
\]
क्या तीनों जगह भरने का सिर्फ़ एक ही तरीक़ा है? समझाओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:addsub:10}
एक नानबाई ने सुबह $145$ बन और दोपहर में $89$ बेचे; दुकान बंद
करते समय $17$ बन बिना बिके बचे। उस दिन कुल कितने बन बने थे?
(दो कदम।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:addsub:11}
थियो कहता है: ``$500 - 199$ निकालने के लिए मैं $500 - 200$
निकालता हूँ और फिर $1$ जोड़ देता हूँ।'' क्या वह सही है? दोनों तरीक़ों से निकालो, और उसकी तरकीब समझाओ।
\end{exercise}
