\chapter{मापना}\label{ch:g3:measure}

कितना लंबा, कितना भारी, कितना समाता है, कितने बज गए? मापना
दुनिया को संख्याओं में बदल देता है --- बशर्ते सब एक ही मात्रक
इस्तेमाल करें। यह अध्याय लंबाई, \omterm{def:g2:measure:units}{द्रव्यमान} और \omterm{def:g2:measure:units}{धारिता} के मात्रक
बताता है, और घड़ी पढ़ना सिखाता है।

\section{लंबाइयाँ}

\begin{definition}[लंबाई के मात्रक]\label{def:g3:measure:length}
लंबाई का बुनियादी मात्रक \emph{मीटर} (m) है। छोटी लंबाइयों के
लिए: सेंटीमीटर (cm) और मिलीमीटर (mm); लंबी दूरियों के लिए:
किलोमीटर (km)।
\[
1 \text{ m} = 100 \text{ cm}, \qquad
1 \text{ cm} = 10 \text{ mm}, \qquad
1 \text{ km} = 1\,000 \text{ m}.
\]
\end{definition}

लिए छोटे। वस्तु $0$ के निशान से शुरू होती है --- मापपट्टी के
किनारे से नहीं! --- और $3$ के बाद के छठे छोटे निशान पर ख़त्म होती है।}
\end{omfigure}

\begin{example}[सही मात्रक चुनना]\label{ex:g3:measure:choose}
एक चींटी: लगभग $5$ mm। एक पेंसिल: लगभग $18$ cm। एक दरवाज़ा:
लगभग $2$ m। पड़ोस के गाँव तक की चहलक़दमी: लगभग $3$ km।
``पेंसिल की लंबाई $18$ है'' कहने का कोई अर्थ नहीं --- मात्रक उत्तर का हिस्सा है।
\end{example}

\begin{method}[मात्रक बदलना]\label{met:g3:measure:convert}
बदलने के लिए $10$, $100$ या $1\,000$ से गुणा या भाग करो:
\begin{enumerate}
  \item बड़े मात्रक से छोटे में जाने पर \emph{गुणा} करो:
        $3$ m $= 3 \times 100 = 300$ cm;
  \item छोटे मात्रक से बड़े में जाने पर \emph{भाग} दो:
        $250$ cm $= 250 \div 100 = 2$ m और $50$ cm;
  \item लंबाइयाँ जोड़ने से पहले उन्हें \emph{एक ही} मात्रक में बदलो:
        $1$ m $+ 35$ cm $= 100 + 35 = 135$ cm।
\end{enumerate}
\end{method}

\section{द्रव्यमान और धारिता}

\begin{definition}[द्रव्यमान और धारिता के मात्रक]\label{def:g3:measure:mass}
\omterm{def:g2:measure:units}{द्रव्यमान} \emph{ग्राम} (g) और \emph{किलोग्राम} (kg) में मापा
जाता है, \omterm{def:g2:measure:units}{धारिता} (कोई पात्र कितना समाता है) \emph{लीटर} (L) और
\emph{सेंटीलीटर} (cL) में:
\[
1 \text{ kg} = 1\,000 \text{ g},
\qquad
1 \text{ L} = 100 \text{ cL}.
\]
\end{definition}

\begin{example}\label{ex:g3:measure:mass}
एक चिट्ठी: लगभग $20$ g। आटे का थैला: $1$ kg। एक टब: लगभग
$150$ L। एक गिलास: लगभग $20$ cL। $1$ kg के थैले के सामने
$650$ g सेब रखकर तराज़ू बराबर करना हो, तो सेबों की ओर
$1\,000 - 650 = 350$ g और रखो।
\end{example}

\section{घड़ी पढ़ना}

\begin{method}[समय पढ़ना]\label{met:g3:measure:clock}
सुइयों वाली घड़ी पर:
\begin{enumerate}
  \item \emph{छोटी} सुई घंटा बताती है: उसने आख़िरी बार जो
        संख्या पार की है, वही पढ़ो;
  \item \emph{लंबी} सुई मिनट बताती है: हर बड़ा निशान $5$
        मिनट का है, इसलिए $3$ पर लंबी सुई का मतलब $15$ मिनट;
  \item आधे घंटे के बाद इसे ``अगले घंटे में कितना बाक़ी'' के रूप
        में भी पढ़ा जाता है: $10{:}50$ यानी ``ग्यारह बजने में दस मिनट''।
\end{enumerate}
\end{method}

\begin{example}[अवधि]\label{ex:g3:measure:duration}
फ़िल्म $14{:}40$ पर शुरू होती है और $16{:}10$ पर ख़त्म। वह कितनी
लंबी है? दो छलाँगों में गिनो:
\begin{enumerate}
  \item $14{:}40$ से $15{:}00$ तक: $20$ मिनट;
  \item $15{:}00$ से $16{:}10$ तक: $1$ घंटा $10$ मिनट।
\end{enumerate}
कुल: $1$ घंटा $30$ मिनट। सावधान: एक घंटे में $60$ मिनट होते
हैं, इसलिए $16{:}10$ में से $14{:}40$ घटाना $1610 - 1440$ जैसा नहीं है!
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:measure:1}
किस मात्रक का उपयोग करोगे (mm, cm, m या km): एक कील की चौड़ाई;
किसी सहपाठी की ऊँचाई; विद्यालय के मैदान की लंबाई; दो क़स्बों के
बीच की दूरी?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:2}
बदलो: $5$ m को cm में; \quad $70$ mm को cm में; \quad $4$ km को m में;
\quad $600$ cm को m में।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:3}
किसी सहपाठी के बनाए त्रिभुज की भुजाएँ अपनी मापपट्टी से (नज़दीकी
मिलीमीटर तक) मापो, और उसकी सीमा की कुल लंबाई
निकालो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:4}
पहले बदलकर निकालो: $2$ m $+ 45$ cm (cm में); \quad
$1$ km $+ 250$ m (m में); \quad $3$ cm $+ 8$ mm (mm में)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:5}
बदलो: $3$ kg को g में; \quad $2\,500$ g को kg और g में; \quad
$4$ L को cL में; \quad $350$ cL को L और cL में।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:6}
एक पाकविधि में $750$ g आटा चाहिए। मोना के पास $1$ kg का थैला
है। क्या उसके पास काफ़ी है? कितना बचेगा?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:7}
समय पढ़ो: छोटी सुई $2$ और $3$ के बीच, लंबी सुई $6$ पर; छोटी
सुई $9$ पर, लंबी सुई $12$ पर; छोटी सुई लगभग $5$ पर, लंबी सुई
$10$ पर।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:8}
विद्यालय $8{:}30$ पर शुरू होता है और सुबह की पाली $11{:}45$ पर
ख़त्म होती है। सुबह कितनी लंबी है? (छलाँगों में गिनो, जैसे
\cref{ex:g3:measure:duration} में।)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:9}
रेलगाड़ी $9{:}50$ पर चलती है और सफ़र $45$ मिनट का है। वह किस
समय पहुँचेगी? (पहले $10{:}00$ तक छलाँग लगाओ।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measure:10}
एक बोतल में $1$ L आता है। पॉल $20$ cL के गिलास भरता है। वह
कितने पूरे गिलास भर सकता है? (पहले मात्रक बदलो।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measure:11}
लेना रोज़ विद्यालय तक $600$ m चलती है, और वापस भी। $5$ दिनों के
विद्यालय-सप्ताह में वह कितने मीटर चलती है? उत्तर मीटर में दो,
फिर किलोमीटर में।
\end{exercise}
