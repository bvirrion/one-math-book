\chapter{गुणा}\label{ch:g3:mult}

सात-सात कंचों के तीन थैले: तुम $7 + 7 + 7$ जोड़ सकते हो, पर एक
तेज़ रास्ता भी है --- गुणा, यानी बार-बार जोड़ने की संक्रिया।
यह अध्याय पहले चित्रों से पहाड़े बनाता है, फिर खानों वाला
तरीक़ा सिखाता है।

\section{गुणा का मतलब}

\begin{definition}[गुणा]\label{def:g3:mult:def}
\omterm{def:g2:mult:def}{गुणनफल} $3 \times 7$ का मतलब है ``तीन गुना सात'':
\[
3 \times 7 = 7 + 7 + 7 = 21 .
\]
चित्र में $3 \times 7$ बिंदुओं का एक आयत है: $7$-$7$ बिंदुओं
की $3$ पंक्तियाँ।
\end{definition}

{\small वही $21$ बिंदु, पंक्तियों से गिनो या स्तंभों से:
\omterm{def:g2:mult:def}{गुणनफल} में क्रम से फ़र्क़ नहीं पड़ता। $3 \times 7$ पता हो तो
$7 \times 3$ मुफ़्त में मिल जाता है --- आधे पहाड़े ही याद करने हैं!}
\end{omfigure}

\begin{proposition}[काम आने वाले नियम]\label{prop:g3:mult:rules}
\begin{enumerate}
  \item क्रम से फ़र्क़ नहीं पड़ता: $a \times b = b \times a$।
  \item $1$ से गुणा करने पर कुछ नहीं बदलता; $0$ से गुणा करने पर
        $0$ मिलता है।
  \item $10$ से गुणा करने पर एक \omterm{def:g1:counting:zero}{शून्य} जुड़ जाता है:
        $34 \times 10 = 340$; $100$ से, दो \omterm{def:g1:counting:zero}{शून्य}।
  \item किसी पहाड़े को उसके पड़ोसियों से दोबारा बनाया जा सकता है:
        $6 \times 8 = 5 \times 8 + 8 = 40 + 8 = 48$।
\end{enumerate}
\end{proposition}
\admitted

\section{खानों में गुणा}

\begin{method}[एक अंक की संख्या से गुणा]\label{met:g3:mult:column}
$234 \times 6$ निकालने के लिए:
\begin{enumerate}
  \item इकाई का गुणा: $4 \times 6 = 24$: $4$ लिखो, $2$ हासिल;
  \item दहाई का गुणा करके हासिल जोड़ो:
        $3 \times 6 + 2 = 20$: $0$ लिखो, $2$ हासिल;
  \item सैकड़े का गुणा करके हासिल जोड़ो:
        $2 \times 6 + 2 = 14$: $14$ लिखो।
\end{enumerate}
\[
\begin{array}{r}
2\,3\,4 \\
\times \quad 6 \\
\hline
1\,4\,0\,4
\end{array}
\]
जाँच के लिए अंदाज़ा: $234$, $200$ के पास है, और
$200 \times 6 = 1\,200$: उत्तर $1\,404$ भरोसेमंद है।
\end{method}

\begin{example}[मन-ही-मन गुणा के लिए तोड़ना]\label{ex:g3:mult:split}
खानों वाला तरीक़ा चुपके से संख्या को स्थानों के हिसाब से तोड़ता
है। मन-ही-मन भी वही करो:
\[
23 \times 4 = (20 + 3) \times 4 = 20 \times 4 + 3 \times 4
= 80 + 12 = 92 .
\]
और मिलनसार पड़ोसी के साथ: $99 \times 5 = 100 \times 5 - 5 =
495$।
\end{example}

\section{समस्याओं में गुणा}

\begin{example}\label{ex:g3:mult:problem}
एक मिनीबस में $9$ यात्री बैठते हैं। $27$ मिनीबसों में कितने
यात्री बैठ सकते हैं?
\begin{enumerate}
  \item यह गुणा है: $9$-$9$ के $27$ समूह।
  \item $27 \times 9 = 243$ (खानों में, या
        $27 \times 9 = 27 \times 10 - 27 = 270 - 27$)।
  \item वाक्य: मिनीबसों में $243$ यात्री बैठ सकते हैं।
\end{enumerate}
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:mult:1}
गुणा के रूप में लिखो, फिर निकालो:
$8 + 8 + 8 + 8$; \quad $5 + 5 + 5$; \quad
$20 + 20 + 20 + 20 + 20$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:2}
बोलो और पूरा करो: $6 \times 7$; \ $8 \times 4$; \ $9 \times 9$;
\ $7 \times 8$; \ $6 \times 6$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:3}
$4 \times 6$ के लिए बिंदुओं का आयत बनाओ, फिर $6 \times 4$ के
लिए। दोनों गिनतियों में क्या दिखाई देता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:4}
\cref{prop:g3:mult:rules} के नियमों से निकालो:
$56 \times 10$; \quad $8 \times 100$; \quad $73 \times 0$; \quad
$45 \times 1$; \quad $30 \times 10$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:5}
खानों में निकालो: $132 \times 3$; \quad $217 \times 4$; \quad
$408 \times 7$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:6}
तोड़कर मन-ही-मन निकालो (जैसे \cref{ex:g3:mult:split} में):
$31 \times 5$; \quad $42 \times 3$; \quad $25 \times 6$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:7}
मिलनसार पड़ोसी की मदद से निकालो: $99 \times 7$; \quad
$101 \times 6$; \quad $98 \times 5$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:8}
पहले अंदाज़ा लगाओ, फिर निकालो: $389 \times 5$ ($400$ से अंदाज़ा);
$612 \times 8$ ($600$ से अंदाज़ा)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:9}
अंडों के एक डिब्बे में $6$ अंडे आते हैं। $38$ डिब्बों में कितने
अंडे हैं? संक्रिया लिखो, निकालो, एक वाक्य में उत्तर दो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:10}
एक कक्षा में $4$-$4$ डेस्कों की $8$ पंक्तियाँ हैं; हर डेस्क पर
$2$ विद्यार्थी बैठते हैं। कमरे में कितने बैठ सकते हैं? (दो गुणा।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:11}
$24$ कुर्सियों को बराबर पंक्तियों में लगाने के सारे तरीक़े ढूँढ़ो
($1$-$1$ की पंक्तियाँ, $2$-$2$ की, \dots)। कौन-कौन-सी सजावटें
संभव हैं? (पहाड़ों के वर्ग में $24$ ढूँढ़ो।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:12}
ज़ोई $47 \times 6$ ऐसे निकालती है: $40 \times 6 = 240$,
$7 \times 6 = 42$, और फिर $240 + 42 = 282$। लेआ ऐसे निकालती है:
$47 \times 6 = 50 \times 6 - 3 \times 6 = 300 - 18 = 282$। दोनों
तरीक़े समझाओ। $38 \times 4$ के लिए तुम्हें कौन-सा पसंद है? उसे
दोनों तरीक़ों से निकालो।
\end{exercise}
