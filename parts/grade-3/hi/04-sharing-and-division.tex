\chapter{बाँटना और भाग}\label{ch:g3:division}

बीस कंचे चार बच्चों में बराबर-बराबर बाँटे गए: हर एक को कितने?
भाग बाँटने के सारे सवालों का उत्तर देता है --- और समूह बनाने के
सवालों का भी --- और वह गुणा का जुड़वाँ है: एक दूसरे को उलट
देता है।

\section{भाग के दो रूप}

\begin{definition}[भाग]\label{def:g3:division:def}
भाग $20 \div 4$ दो तरह के सवालों का उत्तर देता है:
\begin{itemize}
  \item \emph{बाँटना}: $20$ कंचे $4$ बच्चों में बाँटे गए ---
        \emph{हर एक} को कितने कंचे? (हर एक को $5$);
  \item \emph{समूह बनाना}: $20$ कंचे $4$-$4$ के थैलों में रखे
        गए --- कितने \emph{थैले}? ($5$ थैले)।
\end{itemize}
दोनों सवालों का उत्तर एक ही है, क्योंकि दोनों उसी गुणा को उलटते
हैं: $4 \times 5 = 20$।
\end{definition}

\begin{tikzpicture}[scale=0.55]
  \foreach \g in {0,1,2,3} {
    \draw[omExo, rounded corners] (\g*4.9-0.65,-0.7) rectangle
      (\g*4.9+3.85,0.7);
    \foreach \c in {0,...,4} \fill[omDef] (\g*4.9+\c*0.8,0) circle (0.27);
  }
  \node[below, font=\small] at (8.95,-1.15)
    {$20$ कंचे, $4$ बराबर समूह: हर समूह में $5$};
\end{tikzpicture}

{\small $20$ को $4$ बराबर समूहों में बाँटना। इसके बजाय $4$-$4$ के
समूह बनाने पर $5$ समूह मिलते --- वही भाग।}
\end{omfigure}

\begin{method}[पहाड़ों से भाग]\label{met:g3:division:tables}
$42 \div 6$ निकालने के लिए पूछो: ``$6$ गुना \emph{क्या} $42$
बनाता है?'' $6$ का पहाड़ा उत्तर देता है: $6 \times 7 = 42$,
इसलिए $42 \div 6 = 7$। भाग का हर सवाल उलटा पढ़ा हुआ गुणा का
सवाल है।
\end{method}

\section{जब बराबर न बँटे}

\begin{definition}[शेषफल]\label{def:g3:division:remainder}
$23$ कंचे $4$ बच्चों में बाँटने पर हर एक को $5$ कंचे मिलते हैं ---
और $3$ कंचे \emph{बच जाते हैं}, जो एक और बार बाँटने के लिए
काफ़ी नहीं। हम लिखते हैं
\[
23 = 4 \times 5 + 3 ,
\]
और कहते हैं: \emph{भागफल}\index{भागफल} $5$ है,
\emph{शेषफल}\index{शेषफल} $3$ है। शेषफल हमेशा बच्चों की
संख्या से छोटा होता है --- नहीं तो एक बार और बाँटा जा
सकता।
\end{definition}

      (\g*4.9+3.85,0.7);
    \foreach \c in {0,...,4} \fill[omDef] (\g*4.9+\c*0.8,0) circle (0.27);

  \node[below, font=\small] at (10.2,-1.15)
    {$23 = 4 \times 5 + 3$: भागफल $5$, शेषफल $3$ (लाल में)};
\end{tikzpicture}

{\small $23$ को $4$ में बाँटना: $5$ बार बाँटने के बाद $3$ कंचे
बचते हैं --- छठी बार के लिए बहुत कम।}
\end{omfigure}

\begin{example}\label{ex:g3:division:remainder}
$58$ में $7$ का भाग दो। $7$ का पहाड़ा देखो:
$7 \times 8 = 56$ सबसे बड़ा \omterm{def:g2:mult:def}{गुणनफल} है जो $58$ के नीचे समाता है
($7 \times 9 = 63$ बहुत ज़्यादा है)। तो
\[
58 = 7 \times 8 + 2 :
\]
\omterm{def:g3:division:remainder}{भागफल} $8$, \omterm{def:g3:division:remainder}{शेषफल} $2$। जाँच: $56 + 2 = 58$, और $2 < 7$।
\end{example}

\begin{method}[शेषफल का मतलब]\label{met:g3:division:interpret}
भाग देने के बाद कहानी पर लौटो:
\begin{enumerate}
  \item ``हर एक को कितने / कितने पूरे समूह?'' --- \omterm{def:g3:division:remainder}{भागफल};
  \item ``कितने बच गए?'' --- \omterm{def:g3:division:remainder}{शेषफल};
  \item ``सब कुछ रखने के लिए कितने डिब्बे?'' --- \omterm{def:g3:division:remainder}{भागफल}, और
        \omterm{def:g3:division:remainder}{शेषफल} \omterm{def:g1:counting:zero}{शून्य} न हो तो उसमें \emph{एक और} जोड़ो।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:division:interpret}
$58$ विद्यार्थी डोंगी चलाने जाते हैं, हर डोंगी में $7$।
$58 = 7 \times 8 + 2$: आठ डोंगियाँ भर जाती हैं, और $2$
विद्यार्थी बचते हैं --- उन्हें भी डोंगी चाहिए। तो $9$ डोंगियाँ
लेनी होंगी।
\end{example}

\section{आधा, तिहाई, चौथाई}

\begin{definition}[आधा, तिहाई, चौथाई]\label{def:g3:division:half}
किसी संख्या का \emph{आधा}\index{आधा} उसमें $2$ का भाग देने पर मिलता है;
\emph{तिहाई}, $3$ का भाग देने पर; \emph{चौथाई}\index{चौथाई}, $4$ का। जैसे
$18$ का आधा $9$ है, $18$ की तिहाई $6$ है, और $36$ की चौथाई
$9$ है।
\end{definition}

\begin{example}[दुगुना और आधा साथ-साथ चलते हैं]\label{ex:g3:division:double}
$46$ का \omterm{def:g3:division:half}{आधा}: चूँकि $46 = 40 + 6$, हर हिस्से का \omterm{def:g3:division:half}{आधा} लेने पर
$20 + 3 = 23$ मिलता है। दुगुने से जाँचो: $23 + 23 = 46$। दुगुने
याद हों ($25 \to 50$, $45 \to 90$, $150 \to 300$) तो आधे तुरंत
मिल जाते हैं।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:division:1}
पहाड़े उलटे पढ़कर निकालो: $36 \div 4$; \quad $63 \div 9$;
\quad $40 \div 5$; \quad $56 \div 8$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:2}
हर भाग के लिए संबंध $a = b \times q + r$ लिखो और \omterm{def:g3:division:remainder}{भागफल} तथा
\omterm{def:g3:division:remainder}{शेषफल} बताओ: $17 \div 5$; \quad $30 \div 4$; \quad
$50 \div 6$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:3}
$28$ कंचे $6$ बच्चों में बाँटे जाते हैं। हर एक को कितने कंचे,
और कितने बच जाते हैं? मदद के लिए बँटवारे का चित्र बना लो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:4}
निकालो: $68$ का \omterm{def:g3:division:half}{आधा}; $27$ की तिहाई; $48$ की \omterm{def:g3:division:half}{चौथाई};
$150$ का \omterm{def:g3:division:half}{आधा}।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:5}
सही या ग़लत? गुणा से जाँचो।
``$45 \div 5 = 9$''; \quad ``$52 \div 6 = 8$, \omterm{def:g3:division:remainder}{शेषफल} $4$'';
\quad ``$70 \div 8 = 9$, \omterm{def:g3:division:remainder}{शेषफल} $2$''।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:6}
$32$ तस्वीरें एक एलबम में चिपकाई जाती हैं, हर पन्ने पर $4$। कितने
पन्ने लगते हैं? यह भाग का कौन-सा रूप है (बाँटना या समूह बनाना)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:7}
$75$ cm की रस्सी $9$-$9$ cm के टुकड़ों में काटी जाती है। कितने
पूरे टुकड़े बनते हैं, और बचा हुआ टुकड़ा कितना लंबा है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:8}
$50$ बच्चे $4$-$4$ सीटों वाली कारों में सैर पर जाते हैं। कितनी
कारें चाहिए? (सावधान: सबको जाना है!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:9}
ग़ायब संख्याएँ ढूँढ़ो: $? \div 7 = 6$; \quad
$54 \div ? = 6$; \quad जिसका \omterm{def:g3:division:half}{आधा} $35$ है, वह संख्या।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:division:10}
अमीना अपने $47$ स्टिकर अपने $5$ दोस्तों में जितना हो सके उतने
बराबर बाँटती है, और बचे हुए अपने पास रख लेती है। हर दोस्त को
कितने स्टिकर मिलते हैं, और अमीना के पास कितने रहते हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:division:11}
किसी संख्या में $6$ का भाग देने पर कौन-कौन-से
\omterm{def:g3:division:remainder}{शेषफल} संभव हैं? वह सबसे बड़ी संख्या कौन-सी है जिसमें $6$ का भाग
देने पर \omterm{def:g3:division:remainder}{भागफल} $7$ आता है?
\end{exercise}
