\chapter{आकार और समकोण}\label{ch:g3:shapes}

वर्ग, आयत, त्रिभुज, वृत्त: ये आकार पुराने दोस्त हैं। यह अध्याय
उन्हें ज्यामिति की नज़र से देखता है --- वर्ग को वर्ग ठीक-ठीक
क्या बनाता है? --- और वह औज़ार सामने लाता है जो फ़ैसला करता है:
\omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका और उसका \omterm{def:g3:shapes:rightangle}{समकोण}।

\section{समकोण}

\begin{definition}[समकोण]\label{def:g3:shapes:rightangle}
\emph{समकोण}\index{समकोण} काग़ज़ की शीट का, या समकोण पट्टिका
का, कोने वाला कोण है। जो दो रेखाएँ समकोण बनाती हैं वे
\emph{लंब} कहलाती हैं। आकृतियों में समकोण कोने में एक छोटे
वर्ग से अंकित किया जाता है।
\end{definition}

\begin{method}[समकोण की जाँच]\label{met:g3:shapes:check}
\begin{enumerate}
  \item जिस कोने की जाँच करनी है, ठीक उस पर \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका का
        कोना रखो;
  \item पट्टिका का एक किनारा कोण की एक भुजा के साथ-साथ सरकाओ;
  \item अगर कोण की दूसरी भुजा ठीक दूसरे किनारे के साथ-साथ चले,
        तो कोण \omterm{def:g3:shapes:rightangle}{समकोण} है --- अगर वह ज़्यादा या कम खुला हो, तो
        नहीं।
\end{enumerate}
\end{method}

\section{आकार और उनके नियम}

\begin{definition}[वर्ग, आयत, त्रिभुज]\label{def:g3:shapes:shapes}
\begin{itemize}
  \item \emph{आयत}\index{आयत} की $4$ भुजाएँ और $4$
        \omterm{def:g3:shapes:rightangle}{समकोण} होते हैं; उसकी आमने-सामने की भुजाओं की लंबाई बराबर होती है।
  \item \emph{वर्ग}\index{वर्ग} वह आयत है जिसकी चारों
        भुजाओं की लंबाई बराबर हो।
  \item \emph{त्रिभुज}\index{त्रिभुज} की $3$ भुजाएँ होती हैं;
        \emph{\omterm{def:g3:shapes:rightangle}{समकोण} त्रिभुज} में एक \omterm{def:g3:shapes:rightangle}{समकोण} होता है।
  \item \emph{वृत्त}\index{वृत्त} परकार से खींचा जाता है: उसके
        सारे बिंदु \emph{केंद्र} से एक ही दूरी पर होते हैं, और
        यह दूरी \emph{त्रिज्या}\index{त्रिज्या} कहलाती है।
\end{itemize}
\end{definition}

\begin{example}[चिह्न क्यों ज़रूरी हैं]\label{ex:g3:shapes:marks}
ज़रा-सा तिरछा बना आकार वर्ग जैसा \emph{दिख} सकता है, पर हो न।
सच चिह्न बताते हैं: $4$ समकोण-चिह्न $+$ $4$ लकीरें $=$ असली
वर्ग। बनाते समय चिह्न लगाओ; आकृति पढ़ते समय केवल चिह्नों और दी
गई मापों पर भरोसा करो।
\end{example}

\section{जाल वाले काग़ज़ पर रचना}

\begin{method}[जाल पर रचनाएँ]\label{met:g3:shapes:grid}
जाल वाला काग़ज़ मुफ़्त में \omterm{def:g3:shapes:rightangle}{समकोण} देता है: उसकी रेखाएँ आपस में लंब हैं।
\begin{enumerate}
  \item $5$ गुणा $3$ का आयत बनाने के लिए: एक कटान-बिंदु चुनो,
        $5$ खाने दाईं ओर, $3$ खाने ऊपर गिनो, और जाल की रेखाओं
        के साथ आकार बंद कर दो;
  \item बराबर लंबाइयाँ जाँचने के लिए खाने गिनो;
  \item तिरछे कदम तिरछी रेखाएँ बनाते हैं: बार-बार ``$1$ दाईं
        ओर, $1$ ऊपर'' चलने पर $45^\circ$ की रेखा बनती है।
\end{enumerate}
\end{method}

देती हैं।}
\end{omfigure}

\begin{example}[रचना का कार्यक्रम]\label{ex:g3:shapes:program}
किसी आकृति को निर्देशों से बताया जा सकता है:
``1. $4$ cm भुजा वाला वर्ग $ABCD$ बनाओ। 2. $A$ को केंद्र मानकर
$B$ से होकर जाने वाला वृत्त बनाओ।'' कार्यक्रम की एक-एक पंक्ति
का पालन करके कोई भी वही आकृति दोबारा बना सकता है --- और ऐसा
कार्यक्रम लिखने पर हर बिंदु का नाम रखना पड़ता ही है। किसी साथी
के साथ आज़माओ: एक लिखे, दूसरा बनाए।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g3:shapes:1}
कक्षा में (बिना नापे) तीन \omterm{def:g3:shapes:rightangle}{समकोण} और एक ऐसा कोण ढूँढ़ो जो \omterm{def:g3:shapes:rightangle}{समकोण}
न हो। हर एक की \emph{जाँच} कैसे करोगे?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:2}
मापपट्टी और \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका से $6$ cm गुणा $4$ cm का आयत बनाओ।
उसके \omterm{def:g3:shapes:rightangle}{समकोण} और बराबर भुजाएँ अंकित करो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:3}
$5$ cm भुजा वाला वर्ग बनाओ। उसमें कितने \omterm{def:g3:shapes:rightangle}{समकोण} हैं?
कितनी बराबर भुजाएँ? सब कुछ अंकित करो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:4}
जाल वाले काग़ज़ पर बनाओ: $4 \times 4$ का वर्ग; $7 \times 2$
का आयत; $3$ और $5$ खानों की भुजाओं वाला \omterm{def:g3:shapes:rightangle}{समकोण} त्रिभुज।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:5}
सही या ग़लत? समझाओ।
``हर वर्ग एक आयत है।'' \quad
``हर आयत एक वर्ग है।'' \quad
``त्रिभुज में दो \omterm{def:g3:shapes:rightangle}{समकोण} हो सकते हैं।'' (बनाकर देखो!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:6}
$4$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} का वृत्त बनाओ। उसका केंद्र $O$ अंकित करो, वृत्त
पर एक बिंदु $A$ लो, और \omterm{def:g3:shapes:shapes}{त्रिज्या} $[OA]$ खींचो। $OA$ कितना लंबा
है? और $B$ वृत्त का कोई दूसरा बिंदु हो, तो $OB$ कितना लंबा है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:7}
इस कार्यक्रम का पालन करो: ``1. $AB = 6$ cm और $BC = 3$ cm
वाला आयत $ABCD$ बनाओ। 2. $[AB]$ के बीच में बिंदु $M$ अंकित करो।
3. $M$ को $C$ और $D$ से जोड़ो।'' आयत किन आकारों में कट
गया?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:8}
आकारों को छाँटो: इनमें से किनमें कम से कम एक \omterm{def:g3:shapes:rightangle}{समकोण} है?
एक वर्ग; एक (वर्ग न होने वाला) आयत; एक \omterm{def:g3:shapes:rightangle}{समकोण} त्रिभुज; एक वृत्त;
मोटी लकीरों से बना अक्षर L।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:9}
जाल वाले काग़ज़ पर एक तिरछा वर्ग बनाओ जिसकी भुजाएँ ``$2$ दाईं
ओर, $1$ ऊपर'' चलती हों (और उसी के मुड़े हुए रूप)। खाने गिनकर
पक्का करो कि चारों भुजाएँ बराबर लंबी हैं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:10}
एक रचना का कार्यक्रम लिखो (क्रम से निर्देश, नाम वाले बिंदु):
$3$ cm $\times$ $2$ cm के आयत के ऊपर $3$ cm भुजा वाला वर्ग,
जैसे भूतल पर बना घर। अपना कार्यक्रम किसी सहपाठी पर
आज़माओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:11}
छोटे वर्गों के $3 \times 3$ जाल में कुल कितने वर्ग ढूँढ़ सकते
हो? ($1 \times 1$, $2 \times 2$ और $3 \times 3$ वाले
गिनो।)
\end{exercise}
