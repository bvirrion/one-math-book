\chapter{Los números hasta 10\,000}\label{ch:g3:numbers}

Mil, dos mil, tres mil\,\dots\ los números grandes se
construyen a partir de los pequeños con una idea estupenda: el \emph{lugar} de una
cifra dice su valor. Este capítulo enseña a leer, escribir,
comparar y ordenar los números hasta $10\,000$.

\section{Leer y escribir números}

\begin{definition}[Valor posicional]\label{def:g3:numbers:place}
Un número se escribe con las cifras $0, 1, 2, \dots, 9$. Leyendo desde
la derecha, los lugares son: unidades, decenas, centenas y millares. En
$3\,254$:
\[
3\,254 = 3 \text{ millares} + 2 \text{ centenas} + 5 \text{ decenas}
+ 4 \text{ unidades}.
\]
\end{definition}

\begin{example}[El cero que guarda un lugar]\label{ex:g3:numbers:zero}
«Cuatro mil siete» se escribe $4\,007$: un $4$ en los
millares, un $7$ en las unidades y \emph{\omterm{def:g1:counting:zero}{ceros}} en medio para mantener
cada cifra en su lugar. ¡Sin ellos, $47$ sería un número muy
distinto!
\end{example}

\begin{example}[Cambiar]\label{ex:g3:numbers:exchange}
Diez unidades hacen una decena; diez decenas, una centena; diez centenas,
un millar. Así, $10$ centenas $= 1\,000$ y $32$ decenas
$= 320$: el número $320$ contiene $32$ decenas (no solo la cifra
$2$!).
\end{example}

\section{Comparar y ordenar}

\begin{method}[Comparar dos números]\label{met:g3:numbers:compare}
\begin{enumerate}
  \item cuenta las cifras: más cifras significa número mayor
        ($1\,002 > 998$).
  \item con el mismo número de cifras: compara cifra a cifra empezando por
        la \emph{izquierda}. La primera \omterm{ex:g1:subtraction:difference}{diferencia} decide:
        $5\,614 > 5\,589$ porque $6 > 5$ en las centenas;
  \item escribe la respuesta con los signos $<$ (menor que) o $>$
        (mayor que): el lado abierto mira siempre al número mayor.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:numbers:order}
Ordena de menor a mayor: $780$, $8\,700$, $807$, $87$.
Primero por el número de cifras: $87$ (dos), luego $780$ y $807$
(tres) y después $8\,700$ (cuatro). Entre $780$ y $807$ comparamos las
cifras de las centenas, $7 < 8$, así que $780 < 807$. Orden final:
\[
87 < 780 < 807 < 8\,700 .
\]
\end{example}

\section{Pares e impares}

\begin{definition}[Par, impar]\label{def:g3:numbers:evenodd}
Un número es \emph{par}\index{número par} cuando se puede repartir en
dos partes enteras iguales --- su cifra de las unidades es $0$, $2$, $4$, $6$ u
$8$. En caso contrario es \emph{impar}\index{número impar} (cifra de las unidades $1$,
$3$, $5$, $7$ o $9$).
\end{definition}

\begin{example}
$46$ es \omterm{def:g3:numbers:evenodd}{par} ($46 = 23 + 23$); $47$ es \omterm{def:g3:numbers:evenodd}{impar} (dos partes iguales
necesitarían medio objeto). Solo cuenta la \emph{última} cifra: $3\,578$ es
\omterm{def:g3:numbers:evenodd}{par} y $2\,341$ es \omterm{def:g3:numbers:evenodd}{impar}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g3:numbers:1}
Escribe con cifras: «dos mil trescientos cuarenta y cinco»; «seis
mil cincuenta»; «nueve mil uno».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:2}
Escribe con letras: $1\,208$; \quad $4\,070$; \quad $9\,999$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:3}
En $7\,382$: ¿qué cifra ocupa el lugar de las decenas? ¿Y el de los millares?
¿Cuántas \emph{centenas} contiene el número en total
(cuidado: no solo la cifra de las centenas --- piensa en el
\cref{ex:g3:numbers:exchange})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:4}
Descompón como en la \cref{def:g3:numbers:place}:
$5\,631$; \quad $2\,046$; \quad $8\,500$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:5}
Copia y completa con $<$ o $>$:
\[
456 \;?\; 465, \qquad
1\,203 \;?\; 989, \qquad
6\,090 \;?\; 6\,009, \qquad
9\,999 \;?\; 10\,000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:6}
Ordena de menor a mayor: $530$; \ $3\,500$; \ $353$; \ $5\,033$;
\ $503$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:7}
En una recta graduada de cien en cien del $0$ al $1\,000$, sitúa
(aproximadamente) los números $150$, $480$, $820$ y $990$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:8}
¿\omterm{def:g3:numbers:evenodd}{Par} o \omterm{def:g3:numbers:evenodd}{impar}? \ $34$; \ $57$; \ $70$; \ $2\,481$; \ $6\,548$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:numbers:9}
Cuenta a pasos: desde $2\,970$, cinco pasos de $10$; desde $860$,
cuatro pasos de $100$; desde $9\,996$, cuatro pasos de $1$.
¿Qué observas al final de esta última?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:numbers:10}
Usando cada una de las cifras $3$, $7$, $0$ y $5$ exactamente una vez, escribe el
mayor número posible de cuatro cifras y después el menor (un número
no puede empezar por $0$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:numbers:11}
Soy un número de tres cifras. Mis cifras son $2$, $5$ y $8$, soy \omterm{def:g3:numbers:evenodd}{par}
y soy mayor que $800$. ¿Quién soy?
\end{exercise}
