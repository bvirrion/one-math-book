\chapter{La multiplicación}\label{ch:g3:mult}

Tres bolsas de siete canicas: podrías sumar $7 + 7 + 7$, pero hay
un camino más rápido --- la multiplicación, la operación de la \omterm{def:g1:addition:def}{suma} repetida.
Este capítulo construye las tablas de multiplicar a partir de dibujos y luego
enseña el método en columna.

\section{Qué significa multiplicar}

\begin{definition}[La multiplicación]\label{def:g3:mult:def}
El \omterm{def:g2:mult:def}{producto} $3 \times 7$ significa «tres veces siete»:
\[
3 \times 7 = 7 + 7 + 7 = 21 .
\]
En un dibujo, $3 \times 7$ es un rectángulo de puntos: $3$ filas de $7$
puntos.
\end{definition}

  \end{scope}
\end{tikzpicture}

{\small Los mismos $21$ puntos, contados por filas o por columnas: el orden
de un \omterm{def:g2:mult:def}{producto} no importa. Saber $3 \times 7$ regala
$7 \times 3$ --- ¡la mitad de las tablas que aprender!}
\end{omfigure}

\begin{proposition}[Reglas que ayudan]\label{prop:g3:mult:rules}
\begin{enumerate}
  \item el orden no importa: $a \times b = b \times a$;
  \item multiplicar por $1$ no cambia nada; multiplicar por $0$ da
        $0$.
  \item multiplicar por $10$ añade un \omterm{def:g1:counting:zero}{cero}: $34 \times 10 = 340$;
        por $100$, dos \omterm{def:g1:counting:zero}{ceros};
  \item una tabla se puede reconstruir a partir de sus vecinas:
        $6 \times 8 = 5 \times 8 + 8 = 40 + 8 = 48$.
\end{enumerate}
\end{proposition}
\admitted

\section{La multiplicación en columna}

\begin{method}[Multiplicar por un número de una cifra]\label{met:g3:mult:column}
Para calcular $234 \times 6$:
\begin{enumerate}
  \item multiplica las unidades: $4 \times 6 = 24$: escribe $4$ y lleva $2$;
  \item multiplica las decenas y \omterm{def:g1:addition:def}{suma} la llevada:
        $3 \times 6 + 2 = 20$: escribe $0$ y lleva $2$;
  \item multiplica las centenas y \omterm{def:g1:addition:def}{suma} la llevada:
        $2 \times 6 + 2 = 14$: escribe $14$.
\end{enumerate}
\[
\begin{array}{r}
2\,3\,4 \\
\times \quad 6 \\
\hline
1\,4\,0\,4
\end{array}
\]
Estimación para comprobar: $234$ está cerca de $200$ y
$200 \times 6 = 1\,200$: la respuesta $1\,404$ es verosímil.
\end{method}

\begin{example}[Descomponer para multiplicar de cabeza]\label{ex:g3:mult:split}
El método en columna descompone en secreto el número por lugares. De cabeza, haz
lo mismo:
\[
23 \times 4 = (20 + 3) \times 4 = 20 \times 4 + 3 \times 4
= 80 + 12 = 92 .
\]
Y con una vecina amable: $99 \times 5 = 100 \times 5 - 5 =
495$.
\end{example}

\section{La multiplicación en los problemas}

\begin{example}\label{ex:g3:mult:problem}
Un microbús lleva $9$ pasajeros. ¿Cuántos pasajeros pueden llevar $27$
microbuses?
\begin{enumerate}
  \item es una multiplicación: $27$ grupos de $9$;
  \item $27 \times 9 = 243$ (en columna o
        $27 \times 9 = 27 \times 10 - 27 = 270 - 27$).
  \item frase: los microbuses pueden llevar $243$ pasajeros.
\end{enumerate}
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g3:mult:1}
Escribe como multiplicación y calcula:
$8 + 8 + 8 + 8$; \quad $5 + 5 + 5$; \quad
$20 + 20 + 20 + 20 + 20$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:2}
Recita y completa: $6 \times 7$; \ $8 \times 4$; \ $9 \times 9$;
\ $7 \times 8$; \ $6 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:3}
Dibuja un rectángulo de puntos para $4 \times 6$ y otro para $6 \times 4$.
¿Qué observas al comparar los dos recuentos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:4}
Calcula usando las reglas de la \cref{prop:g3:mult:rules}:
$56 \times 10$; \quad $8 \times 100$; \quad $73 \times 0$; \quad
$45 \times 1$; \quad $30 \times 10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:5}
Calcula en columna: $132 \times 3$; \quad $217 \times 4$; \quad
$408 \times 7$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:6}
Calcula de cabeza descomponiendo (como en el \cref{ex:g3:mult:split}):
$31 \times 5$; \quad $42 \times 3$; \quad $25 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:7}
Calcula con una vecina amable: $99 \times 7$; \quad
$101 \times 6$; \quad $98 \times 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:8}
Estima primero y después calcula: $389 \times 5$ (estima con $400$);
$612 \times 8$ (estima con $600$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:mult:9}
Una huevera lleva $6$ huevos. ¿Cuántos huevos hay en $38$ hueveras? Escribe
la operación, calcula y responde con una frase.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:10}
Un aula tiene $8$ filas de $4$ pupitres y en cada pupitre se sientan $2$ alumnos.
¿Cuántos alumnos caben en el aula? (Dos multiplicaciones.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:11}
Busca todas las maneras de colocar $24$ sillas en filas iguales (filas de
$1$, de $2$, \dots). ¿Qué disposiciones son posibles? (Usa el cuadro
de las tablas: busca el $24$ dentro.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:mult:12}
Zoe calcula $47 \times 6$ así: $40 \times 6 = 240$,
$7 \times 6 = 42$ y después $240 + 42 = 282$. Lea calcula
$47 \times 6 = 50 \times 6 - 3 \times 6 = 300 - 18 = 282$. Explica
cada método. ¿Cuál prefieres para $38 \times 4$? Calcúlalo de las dos
maneras.
\end{exercise}
