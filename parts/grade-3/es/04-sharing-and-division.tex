\chapter{Repartir y dividir}\label{ch:g3:division}

Veinte canicas repartidas a partes iguales entre cuatro niños: ¿cuántas para cada uno?
La división responde a todas las preguntas de reparto --- y también a las de
agrupamiento --- y es la gemela de la multiplicación: una deshace
a la otra.

\section{Las dos caras de la división}

\begin{definition}[La división]\label{def:g3:division:def}
La división $20 \div 4$ responde a dos tipos de preguntas:
\begin{itemize}
  \item \emph{repartir}: $20$ canicas repartidas entre $4$ niños ---
        ¿cuántas canicas para \emph{cada uno}? ($5$ para cada uno);
  \item \emph{agrupar}: $20$ canicas metidas en bolsas de $4$ --- ¿cuántas
        \emph{bolsas}? ($5$ bolsas).
\end{itemize}
Las dos preguntas tienen la misma respuesta, porque las dos deshacen la misma
multiplicación: $4 \times 5 = 20$.
\end{definition}

\begin{method}[Dividir con las tablas]\label{met:g3:division:tables}
Para calcular $42 \div 6$, pregunta: «¿$6$ por \emph{cuánto} da $42$?»
La tabla del $6$ responde: $6 \times 7 = 42$, así que $42 \div 6 = 7$.
Toda pregunta de división es una pregunta de multiplicación leída
al revés.
\end{method}

\section{Cuando no es exacta}

\begin{definition}[El resto]\label{def:g3:division:remainder}
Repartir $23$ canicas entre $4$ niños da $5$ canicas para cada uno ---
y \emph{sobran} $3$ canicas, que no llegan para otra ronda. Escribimos
\[
23 = 4 \times 5 + 3 ,
\]
y decimos: el \emph{cociente}\index{cociente} es $5$ y el
\emph{resto}\index{resto} es $3$. El resto es siempre
menor que el número de niños --- si no, se podría hacer otra ronda
de reparto.
\end{definition}

\begin{example}\label{ex:g3:division:remainder}
Divide $58$ entre $7$. Recorre la tabla del $7$:
$7 \times 8 = 56$ es el mayor \omterm{def:g2:mult:def}{producto} que cabe por debajo de $58$
($7 \times 9 = 63$ se pasa). Por tanto,
\[
58 = 7 \times 8 + 2 :
\]
\omterm{def:g3:division:remainder}{cociente} $8$ y \omterm{def:g3:division:remainder}{resto} $2$. Comprobación: $56 + 2 = 58$, y $2 < 7$.
\end{example}

\begin{method}[Interpretar el resto]\label{met:g3:division:interpret}
Después de dividir, vuelve a la historia:
\begin{enumerate}
  \item «¿cuántos para cada uno / cuántos grupos completos?» --- el \omterm{def:g3:division:remainder}{cociente};
  \item «¿cuántos sobran?» --- el \omterm{def:g3:division:remainder}{resto};
  \item «¿cuántas cajas hacen falta para guardarlo todo?» --- el \omterm{def:g3:division:remainder}{cociente}
        \emph{más uno} si el \omterm{def:g3:division:remainder}{resto} no es \omterm{def:g1:counting:zero}{cero}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g3:division:interpret}
$58$ alumnos van en canoa, $7$ por canoa. $58 = 7 \times 8 + 2$:
ocho canoas van llenas y quedan $2$ alumnos --- que también necesitan
canoa. Así que hay que reservar $9$ canoas.
\end{example}

\section{Mitades, tercios y cuartos}

\begin{definition}[Mitad, tercio, cuarto]\label{def:g3:division:half}
La \emph{mitad}\index{mitad} de un número es el resultado de dividirlo entre $2$; el
\emph{tercio}, entre $3$; el \emph{cuarto}\index{cuarto}, entre $4$. Por ejemplo, la
mitad de $18$ es $9$, el tercio de $18$ es $6$ y el cuarto de $36$
es $9$.
\end{definition}

\begin{example}[Los dobles y las mitades van juntos]\label{ex:g3:division:double}
La \omterm{def:g3:division:half}{mitad} de $46$: como $46 = 40 + 6$, la \omterm{def:g3:division:half}{mitad} de cada parte da
$20 + 3 = 23$. Comprobación con el doble: $23 + 23 = 46$. Saberse los
dobles de memoria ($25 \to 50$, $45 \to 90$, $150 \to 300$) hace
instantáneas las \omterm{def:g3:division:half}{mitades}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g3:division:1}
Calcula usando las tablas al revés: $36 \div 4$; \quad $63 \div 9$;
\quad $40 \div 5$; \quad $56 \div 8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:2}
Para cada división, escribe la igualdad $a = b \times q + r$ y di
el \omterm{def:g3:division:remainder}{cociente} y el \omterm{def:g3:division:remainder}{resto}: $17 \div 5$; \quad $30 \div 4$; \quad
$50 \div 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:3}
Se reparten $28$ canicas entre $6$ niños. ¿Cuántas canicas para cada uno
y cuántas sobran? Dibuja el reparto si te ayuda.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:4}
Calcula: la \omterm{def:g3:division:half}{mitad} de $68$; el tercio de $27$; el \omterm{def:g3:division:half}{cuarto} de $48$;
la \omterm{def:g3:division:half}{mitad} de $150$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:5}
¿Verdadero o falso? Compruébalo con una multiplicación.
«$45 \div 5 = 9$»; \quad «$52 \div 6 = 8$ y \omterm{def:g3:division:remainder}{resto} $4$»;
\quad «$70 \div 8 = 9$ y \omterm{def:g3:division:remainder}{resto} $2$».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:6}
Se pegan $32$ fotos en un álbum, $4$ por página. ¿Cuántas páginas se
usan? ¿Qué \omterm{def:g2:shapes:solids}{cara} de la división es esta (repartir o agrupar)?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:7}
Una cuerda de $75$ cm se corta en trozos de $9$ cm. ¿Cuántos trozos completos salen
y cuánto mide el pedazo que sobra?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:8}
$50$ niños van de excursión en coches de $4$ plazas. ¿Cuántos coches hacen
falta? (Cuidado: ¡todos tienen que ir!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:division:9}
Halla los números que faltan: $? \div 7 = 6$; \quad
$54 \div ? = 6$; \quad la \omterm{def:g3:division:half}{mitad} de $?$ es $35$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:division:10}
Amina reparte sus $47$ pegatinas entre sus $5$ amigos, lo más equitativamente
posible, y se queda con lo que sobra. ¿Cuántas pegatinas recibe cada
amigo y cuántas se queda Amina?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:division:11}
Cuando se divide un número entre $6$, ¿cuáles son todos los
\omterm{def:g3:division:remainder}{restos} posibles? ¿Cuál es el mayor número cuya división entre $6$ tiene
\omterm{def:g3:division:remainder}{cociente} $7$?
\end{exercise}
