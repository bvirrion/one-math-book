\chapter{Figuras y ángulos rectos}\label{ch:g3:shapes}

Cuadrados, rectángulos, triángulos, círculos: las figuras son viejas conocidas.
Este capítulo las mira con ojos de geómetra --- ¿qué hace exactamente
que un cuadrado sea un cuadrado? --- y presenta la herramienta que lo decide:
la escuadra y su \omterm{def:g3:shapes:rightangle}{ángulo recto}.

\section{El ángulo recto}

\begin{definition}[Ángulo recto]\label{def:g3:shapes:rightangle}
Un \emph{ángulo recto}\index{ángulo recto} es el ángulo de la esquina de una
hoja de papel o de la escuadra. Dos rectas que forman un ángulo recto
son \emph{perpendiculares}. En las figuras, un ángulo recto se marca con un
cuadradito en la esquina.
\end{definition}

\begin{method}[Comprobar un ángulo recto]\label{met:g3:shapes:check}
\begin{enumerate}
  \item coloca la esquina de la escuadra exactamente sobre la esquina que
        quieres comprobar;
  \item desliza un borde de la escuadra a lo largo de un lado del ángulo;
  \item si el otro lado del ángulo sigue exactamente el otro
        borde, el ángulo es recto --- si se abre más o menos,
        no lo es.
\end{enumerate}
\end{method}

\section{Las figuras y sus reglas}

\begin{definition}[Cuadrado, rectángulo, triángulo]\label{def:g3:shapes:shapes}
\begin{itemize}
  \item un \emph{rectángulo}\index{rectángulo} tiene $4$ lados y $4$
        \omterm{def:g3:shapes:rightangle}{ángulos rectos}; sus lados opuestos miden lo mismo;
  \item un \emph{cuadrado}\index{cuadrado} es un rectángulo cuyos $4$ lados
        miden todos lo mismo;
  \item un \emph{triángulo}\index{triángulo} tiene $3$ lados; un
        \emph{triángulo rectángulo} tiene un \omterm{def:g3:shapes:rightangle}{ángulo recto};
  \item un \emph{círculo}\index{círculo} se traza con el compás: todos
        sus puntos están a la misma distancia (el \emph{radio}\index{radio}) del
        \emph{centro}.
\end{itemize}
\end{definition}

\begin{example}[Por qué importan las marcas]\label{ex:g3:shapes:marks}
Una figura dibujada un poco inclinada puede \emph{parecer} un cuadrado sin
serlo. Las marcas dicen la verdad: $4$ cuadraditos de \omterm{def:g3:shapes:rightangle}{ángulo recto} $+$ $4$
trazos $=$ un cuadrado de verdad. Cuando dibujes, pon las marcas; cuando
leas una figura, fíate solo de las marcas y de las medidas dadas.
\end{example}

\section{Dibujar en papel cuadriculado}

\begin{method}[Construcciones en la cuadrícula]\label{met:g3:shapes:grid}
El papel cuadriculado regala \omterm{def:g3:shapes:rightangle}{ángulos rectos}: sus líneas son perpendiculares.
\begin{enumerate}
  \item para dibujar un rectángulo de $5$ por $3$: elige un cruce,
        cuenta $5$ casillas a la derecha y $3$ hacia arriba, y cierra la figura
        siguiendo las líneas de la cuadrícula;
  \item para comprobar longitudes iguales, cuenta casillas;
  \item los pasos en diagonal dan líneas inclinadas: repetir «$1$ a la derecha, $1$
        arriba» traza una recta de $45^\circ$.
\end{enumerate}
\end{method}

\begin{example}[Un programa de construcción]\label{ex:g3:shapes:program}
Una figura se puede describir con instrucciones:
«1. Traza un cuadrado $ABCD$ de lado $4$ cm. 2. Traza la circunferencia de
centro $A$ que pasa por $B$.» Siguiendo un programa línea a línea,
cualquiera puede volver a dibujar la misma figura --- y escribirlo obliga a
nombrar todos los puntos. Pruébalo con un compañero: uno escribe y el otro
dibuja.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g3:shapes:1}
En clase, busca (sin medir) tres \omterm{def:g3:shapes:rightangle}{ángulos rectos} y
un ángulo que no lo sea. ¿Cómo puedes \emph{comprobar} cada uno?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:2}
Dibuja un rectángulo de $6$ cm por $4$ cm con regla y escuadra. Marca
sus \omterm{def:g3:shapes:rightangle}{ángulos rectos} y sus lados iguales.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:3}
Dibuja un cuadrado de lado $5$ cm. ¿Cuántos \omterm{def:g3:shapes:rightangle}{ángulos rectos} tiene?
¿Cuántos lados iguales? Márcalo todo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:4}
En papel cuadriculado, dibuja: un cuadrado de $4 \times 4$; un rectángulo de
$7 \times 2$; un triángulo rectángulo con catetos de $3$ y $5$ casillas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:5}
¿Verdadero o falso? Explícalo.
«Todo cuadrado es un rectángulo.» \quad
«Todo rectángulo es un cuadrado.» \quad
«Un triángulo puede tener dos \omterm{def:g3:shapes:rightangle}{ángulos rectos}.» (¡Intenta dibujarlo!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:6}
Traza una circunferencia de \omterm{def:g3:shapes:shapes}{radio} $4$ cm. Marca su centro $O$, un punto $A$ sobre
la circunferencia y traza el \omterm{def:g3:shapes:shapes}{radio} $[OA]$. ¿Cuánto mide $OA$? Y si $B$
es otro punto de la circunferencia, ¿cuánto mide $OB$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:7}
Sigue este programa: «1. Traza un rectángulo $ABCD$ con $AB = 6$ cm
y $BC = 3$ cm. 2. Marca el punto $M$ en el medio de $[AB]$.
3. Une $M$ con $C$ y con $D$.» ¿En qué figuras ha quedado dividido
el rectángulo?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:shapes:8}
Clasifica las figuras: ¿cuáles de estas tienen al menos un \omterm{def:g3:shapes:rightangle}{ángulo recto}?
Un cuadrado; un rectángulo (no cuadrado); un triángulo rectángulo; un círculo; la
letra L dibujada con trazos gruesos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:9}
En papel cuadriculado, dibuja un cuadrado inclinado cuyos lados vayan «$2$ a la derecha, $1$
arriba» (y los giros correspondientes). Cuenta casillas para convencerte de
que los cuatro lados miden lo mismo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:10}
Escribe un programa de construcción (instrucciones numeradas, puntos con nombre)
para: un cuadrado de lado $3$ cm apoyado sobre un rectángulo de
$3$ cm $\times$ $2$ cm, como una casa sobre su planta baja. Prueba tu
programa con un compañero.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:shapes:11}
¿Cuántos cuadrados sabes encontrar en una cuadrícula de $3 \times 3$
casillas? (Cuenta los de $1 \times 1$, los de $2 \times 2$ y el de
$3 \times 3$.)
\end{exercise}
