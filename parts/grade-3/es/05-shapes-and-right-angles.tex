\chapter{Figuras y ángulos rectos}\label{ch:g3:shapes}

\omterm{def:g3:shapes:shapes}{Cuadrados}, rectángulos, triángulos, círculos: las formas son viejos amigos.
Este capítulo los analiza con ojos de geómetra: ¿qué es exactamente?
hace que un cuadrado sea un cuadrado? --- e introduce la herramienta que decide:
la escuadra y sus \omterm{def:g3:shapes:rightangle}{ángulo recto}.

\section{El ángulo correcto}

\begin{definition}[Ángulo recto]\label{def:g3:shapes:rightangle}
A \emph{ángulo recto}\index{ángulo recto} es el ángulo de esquina de un
hoja de papel, o de la escuadra. Dos líneas que forman un ángulo recto.
son \emph{perpendicular}. En las figuras, un ángulo recto está marcado con un
pequeña plaza en la esquina.
\end{definition}

\begin{method}[Comprobando un ángulo recto]\label{met:g3:shapes:check}
\begin{enumerate}
  \item Coloque la esquina de la escuadra exactamente en la esquina para
        prueba;
  \item deslice un borde de la escuadra a lo largo de un lado del ángulo;
  \item si el otro lado del ángulo corre exactamente a lo largo del otro
        borde, el ángulo es correcto --- si se abre más o menos,
        no lo es.
\end{enumerate}
\end{method}

\section{Las formas y sus reglas.}

\begin{definition}[Cuadrado, rectángulo, triángulo]\label{def:g3:shapes:shapes}
\begin{itemize}
  \item A \emph{rectángulo}\index{rectángulo} tiene $4$ lados y $4$
        ángulos rectos; sus lados opuestos tienen la misma longitud.
  \item A \emph{cuadrados}\index{cuadrados} es un rectángulo cuyo $4$ lados
        todos tienen la misma longitud.
  \item A \emph{triángulo}\index{triángulo} tiene $3$ lados; a
        \emph{triángulo rectángulo} tiene uno \omterm{def:g3:shapes:rightangle}{ángulo recto}.
  \item A \emph{circulo}\index{circulo} se dibuja con el compás: todo
        sus puntos están a la misma distancia (el \emph{radio}\index{radio}) de
        el \emph{centro}.
\end{itemize}
\end{definition}

\begin{example}[¿Por qué son importantes las notas?]\label{ex:g3:shapes:marks}
Una forma dibujada ligeramente inclinada puede \emph{mirar} como un cuadrado sin
siendo uno. Las marcas dicen la verdad: $4$ \omterm{def:g3:shapes:shapes}{cuadrados} en \omterm{def:g3:shapes:rightangle}{ángulo recto} $+$ $4$
marcas de verificación $=$ una verdadera plaza. Cuando dibujes, pon las marcas; cuando tu
lea una figura, confíe sólo en las marcas y las medidas dadas.
\end{example}

\section{Dibujo sobre papel cuadriculado}

\begin{method}[Construcciones en una cuadrícula]\label{met:g3:shapes:grid}
Ofertas de papel cuadriculado gratis ángulos rectos: sus líneas son perpendiculares.
\begin{enumerate}
  \item Para dibujar un rectángulo $5$ por $3$: elija un punto de cruce,
        contar $5$ \omterm{def:g3:shapes:shapes}{cuadrados} a la derecha, $3$ cuadra y cierra la forma
        a lo largo de las líneas de la cuadrícula;
  \item para comprobar longitudes iguales, cuente los \omterm{def:g3:shapes:shapes}{cuadrados};
  \item los pasos diagonales forman líneas inclinadas: yendo ``$1$ bien, $1$ arriba''
        dibuja repetidamente un $45^\circ $ línea.
\end{enumerate}
\end{method}

\begin{example}[Un programa de construcción]\label{ex:g3:shapes:program}
Una figura se puede describir mediante instrucciones:
``1. dibujar un cuadrado $ ABCD $ de lado $4$ centímetro. 2. Dibuja el círculo con
centro $ A $ pasando por $ B $.'' Siguiendo un programa línea por línea,
cualquiera puede volver a dibujar la misma figura, y escribir una te obliga a
nombrar cada punto. Pruébalo con un compañero: uno escribe, el otro
dibuja.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g3:shapes:1}
En el aula, encuentre (sin medir) tres ángulos rectos y
un ángulo que no es correcto. ¿Cómo puedes \emph{controlar} ¿cada persona?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:shapes:2}
Dibuja un rectángulo de $6$ centímetros por $4$ cm con regla y escuadra. marca
es ángulos rectos y sus lados iguales.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:shapes:3}
dibujar un cuadrado de lado $5$ centímetro. Cuántos ángulos rectos tiene?
¿Cuantos lados iguales? Marca todo.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:shapes:4}
En papel cuadriculado, dibuja: un $4 \times 4$ cuadrado; a $7 \times 2$
rectángulo; un triángulo rectángulo con catetos de $3$ y $5$ cuadrícula.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:shapes:5}
¿Verdadero o falso? Explicar.
"Cada cuadrado es un rectángulo." \quad
"Todo rectángulo es un cuadrado". \quad
"Un triángulo puede tener dos ángulos rectos.'' (¡Intenta dibujar uno!)
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:shapes:6}
Dibuja un círculo de \omterm{def:g3:shapes:shapes}{radio} $4$ centímetro. Marca su centro $ O $, un punto $ A $ en
el círculo y dibuja el \omterm{def:g3:shapes:shapes}{radio} $[OA]$. cuanto dura $ OA $? y si $ B $
es otro punto del círculo, ¿cuánto mide? $ OB $?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:shapes:7}
Siga este programa: ``1. dibujar un rectángulo $ ABCD $ con $ AB = 6$ centímetros
y $ a.\,C. = 3$ centímetro. 2. Marca el punto $ M $ a mediados de $[AB]$.
3. Únete $ M $ a $ C $ y a $ D $.'' ¿Qué formas ha tenido el rectángulo?
¿cortar?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:shapes:8}
Ordena las formas: ¿cuáles de ellas tienen al menos una? \omterm{def:g3:shapes:rightangle}{ángulo recto}?
Un cuadrado; un rectángulo (no cuadrado); un triángulo rectángulo; un círculo; el
Letra L dibujada con trazos gruesos.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g3:shapes:9}
En papel cuadriculado, dibuja un cuadrado inclinado cuyos lados digan "$2$ bien, $1$
arriba'' (y los giros correspondientes). Cuenta \omterm{def:g3:shapes:shapes}{cuadrados} para convencerte
los cuatro lados tienen la misma longitud.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g3:shapes:10}
Escribir un programa de construcción (instrucciones numeradas, puntos nombrados)
para: un cuadrado de lado $3$ cm sentado encima de un rectángulo
$3$ centímetro $\times $ $2$ cm, como una casa en su planta baja. prueba tu
programa en un compañero de clase.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g3:shapes:11}
¿Cuántos \omterm{def:g3:shapes:shapes}{cuadrados} puedes encontrar en un $3 \times 3$ cuadrícula de pequeños
\omterm{def:g3:shapes:shapes}{cuadrados}? (Cuente el $1 \times 1$, el $2 \times 2$ y el
$3 \times 3$ unos.)
\end{exercise}
