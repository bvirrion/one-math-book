\chapter{Medir}\label{ch:g3:measure}

¿Cuánto mide, cuánto pesa, cuánto cabe, qué hora es? Medir
convierte el mundo en números --- siempre que todos usemos las mismas
unidades. Este capítulo presenta las unidades de longitud, \omterm{def:g2:measure:units}{masa} y \omterm{def:g2:measure:units}{capacidad},
y enseña a leer el reloj.

\section{Longitudes}

\begin{definition}[Unidades de longitud]\label{def:g3:measure:length}
La unidad básica de longitud es el \emph{metro} (m). Para longitudes pequeñas:
el centímetro (cm) y el milímetro (mm); para distancias largas: el
kilómetro (km).
\[
1 \text{ m} = 100 \text{ cm}, \qquad
1 \text{ cm} = 10 \text{ mm}, \qquad
1 \text{ km} = 1\,000 \text{ m}.
\]
\end{definition}

  \draw[thick] (-0.35,0) rectangle (8.35,0.9);

\end{omfigure}

\begin{example}[Elegir la unidad adecuada]\label{ex:g3:measure:choose}
Una hormiga: unos $5$ mm. Un lápiz: unos $18$ cm. Una puerta: unos $2$ m.
Un paseo hasta el pueblo vecino: unos $3$ km. Decir «el lápiz mide
$18$» no significa nada --- la unidad forma parte de la respuesta.
\end{example}

\begin{method}[Convertir unidades]\label{met:g3:measure:convert}
Para convertir, multiplica o divide entre $10$, $100$ o $1\,000$:
\begin{enumerate}
  \item de una unidad grande a una pequeña, \emph{multiplica}:
        $3$ m $= 3 \times 100 = 300$ cm;
  \item de una unidad pequeña a una grande, \emph{divide}:
        $250$ cm $= 250 \div 100 = 2$ m y $50$ cm;
  \item para sumar longitudes, conviértelas antes a la \emph{misma} unidad:
        $1$ m $+ 35$ cm $= 100 + 35 = 135$ cm.
\end{enumerate}
\end{method}

\section{Masas y capacidades}

\begin{definition}[Unidades de masa y de capacidad]\label{def:g3:measure:mass}
Las \omterm{def:g2:measure:units}{masas} se miden en \emph{gramos} (g) y \emph{kilogramos} (kg), y las
\omterm{def:g2:measure:units}{capacidades} (lo que cabe en un recipiente) en \emph{litros} (L) y
\emph{centilitros} (cL):
\[
1 \text{ kg} = 1\,000 \text{ g},
\qquad
1 \text{ L} = 100 \text{ cL}.
\]
\end{definition}

\begin{example}\label{ex:g3:measure:mass}
Una carta: unos $20$ g. Un paquete de harina: $1$ kg. Una bañera: unos
$150$ L. Un vaso: unos $20$ cL. Para equilibrar una balanza que sostiene un
paquete de $1$ kg frente a $650$ g de manzanas, hay que añadir $1\,000 - 650 = 350$ g
del lado de las manzanas.
\end{example}

\section{Leer el reloj}

\begin{method}[Leer la hora]\label{met:g3:measure:clock}
En un reloj de agujas:
\begin{enumerate}
  \item la aguja \emph{corta} da la hora: lee el último número que ha
        pasado;
  \item la aguja \emph{larga} da los minutos: cada marca grande son $5$
        minutos, así que la aguja larga en el $3$ significa $15$ minutos;
  \item pasada la media, también se puede leer «para la hora siguiente»:
        las $10{:}50$ son «las once menos diez».
\end{enumerate}
\end{method}

\begin{example}[Duraciones]\label{ex:g3:measure:duration}
La película empieza a las $14{:}40$ y termina a las $16{:}10$. ¿Cuánto dura?
Cuenta en dos saltos:
\begin{enumerate}
  \item de las $14{:}40$ a las $15{:}00$: $20$ minutos;
  \item de las $15{:}00$ a las $16{:}10$: $1$ hora y $10$ minutos.
\end{enumerate}
Total: $1$ h $30$ min. Cuidado: una hora tiene $60$ minutos, ¡así que
$16{:}10$ menos $14{:}40$ no se calcula como $1610 - 1440$!
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g3:measure:1}
¿Qué unidad usarías (mm, cm, m o km) para: el ancho de una uña;
la altura de un compañero; el largo del patio del colegio; la
distancia entre dos pueblos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:2}
Convierte: $5$ m en cm; \quad $70$ mm en cm; \quad $4$ km en m;
\quad $600$ cm en m.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:3}
Mide con tu regla (con precisión de milímetro) los lados de un
triángulo dibujado por un compañero y calcula la longitud total de su
contorno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:4}
Calcula, convirtiendo primero: $2$ m $+ 45$ cm (en cm); \quad
$1$ km $+ 250$ m (en m); \quad $3$ cm $+ 8$ mm (en mm).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:5}
Convierte: $3$ kg en g; \quad $2\,500$ g en kg y g; \quad
$4$ L en cL; \quad $350$ cL en L y cL.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:6}
Una receta necesita $750$ g de harina. Mona tiene un paquete de $1$ kg. ¿Le
llega? ¿Cuánto le sobrará?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:7}
Lee las horas: la aguja corta entre el $2$ y el $3$ y la larga
en el $6$; la aguja corta en el $9$ y la larga en el $12$;
la aguja corta casi en el $5$ y la larga en el $10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:8}
Las clases empiezan a las $8{:}30$ y la mañana termina a las $11{:}45$. ¿Cuánto
dura la mañana? (Cuenta a saltos, como en el
\cref{ex:g3:measure:duration}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g3:measure:9}
El tren sale a las $9{:}50$ y el viaje dura $45$ minutos. ¿A qué
hora llega? (Salta primero a las $10{:}00$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measure:10}
Una botella contiene $1$ L. Pablo llena vasos de $20$ cL. ¿Cuántos vasos
llenos puede servir? (Convierte primero.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g3:measure:11}
Lena camina $600$ m hasta el colegio, y otro tanto de vuelta, todos los días. ¿Cuántos metros
camina en una semana de $5$ días de clase? Da la respuesta en metros
y después en kilómetros.
\end{exercise}
