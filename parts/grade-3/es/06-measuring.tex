\chapter{Medidas}\label{ch:g3:measure}

¿Cuánto tiempo, cuánto pesa, cuánto aguanta, qué tan tarde es? Medición
convierte el mundo en números, siempre que todos usen el mismo
unidades. Este capítulo presenta las unidades de longitud, \omterm{def:g2:measure:units}{masa} y \omterm{def:g2:measure:units}{capacidad},
y enseña a leer un reloj.

\section{Longitudes}

\begin{definition}[unidades de longitud]\label{def:g3:measure:length}
La unidad básica de longitud es la \emph{metro} (metro). Para longitudes pequeñas:
el centímetro (cm) y el milímetro (mm); para largas distancias: el
kilómetro (km).
\[
1 \text{ m} = 100 \text{ cm}, \qquad
1 \text{ cm} = 10 \text{ mm}, \qquad
1 \text{ km} = 1\,000 \text{ m}.
\]
\end{definition}

\begin{example}[Elegir la unidad adecuada]\label{ex:g3:measure:choose}
Una hormiga: aproximadamente $5$ mm. Un lápiz: aproximadamente $18$ centímetro. Una puerta: aproximadamente $2$ metro.
Un paseo hasta el siguiente pueblo: unos $3$ km. Dando "la longitud de un
lápiz es $18$'' no significa nada --- la unidad es parte de la respuesta.
\end{example}

\begin{method}[Conversión de unidades]\label{met:g3:measure:convert}
Para convertir, multiplicar o dividir por $10$, $100$ o $1\,000$:
\begin{enumerate}
  \item de una unidad grande a una pequeña, \emph{multiplicar}:
        $3$ metro $= 3 \times 100 = 300$ centímetro;
  \item de una unidad pequeña a una grande, \emph{divide}:
        $250$ centímetro $= 250 \div 100 = 2$ m y $50$ centímetro;
  \item para agregar longitudes, conviértalas al \emph{mismo} unidad primero:
        $1$ metro $+ 35$ centímetro $= 100 + 35 = 135$ centímetro.
\end{enumerate}
\end{method}

\section{Masas y capacidades}

\begin{definition}[Unidades de masa y capacidad.]\label{def:g3:measure:mass}
Las \omterm{def:g2:measure:units}{masas} se miden en \emph{gramos} (g) y \emph{kilogramos} (kg),
\omterm{def:g2:measure:units}{capacidades} (cuánto cabe en un contenedor) en \emph{litros} (Tierra
\emph{centilitros} (CL):
\[
1 \text{ kg} = 1\,000 \text{ g},
\qquad
1 \text{ L} = 100 \text{ cL}.
\]
\end{definition}

\begin{example}\label{ex:g3:measure:mass}
Una carta: sobre $20$ gramo. Una bolsa de harina: $1$ kilogramos. Una bañera: aproximadamente
$150$ L. Un vaso: aproximadamente $20$ CL. Para equilibrar una balanza que sostiene una
$1$ kg saco contra $650$ g de manzanas, agregar $1\,000 - 650 = 350$ vamos
el lado de las manzanas.
\end{example}

\section{leyendo el reloj}

\begin{method}[leyendo la hora]\label{met:g3:measure:clock}
En un reloj con manecillas:
\begin{enumerate}
  \item el \emph{corto} la manecilla da la hora: lee el número que tiene
        pasó último;
  \item el \emph{largo} mano da los minutos: cada marca grande es $5$
        minutos, por lo que una mano larga en el $3$ \omterm{def:g3:division:half}{medio} $15$ minutos;
  \item después de la media hora también se puede leer ``hasta la hora siguiente'':
        $10{:}50$ son "las once menos diez".
\end{enumerate}
\end{method}

\begin{example}[Duraciones]\label{ex:g3:measure:duration}
La película comienza en $14{:}40$ y termina en $16{:}10$. ¿Cuánto dura?
Cuente en dos saltos:
\begin{enumerate}
  \item de $14{:}40$ a $15{:}00$: $20$ minutos;
  \item de $15{:}00$ a $16{:}10$: $1$ hora $10$ minutos.
\end{enumerate}
Total: $1$ h $30$ mín. Cuidado: ha pasado una hora $60$ minutos, entonces
$16{:}10$ menos $14{:}40$ no se puede calcular como $1610 - 1440$!
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g3:measure:1}
¿Qué unidad usarías (mm, cm, mo km) para: el ancho de un clavo;
la altura de un compañero de clase; la longitud del patio de la escuela; el
¿Distancia entre dos pueblos?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:measure:2}
Convertir: $5$ m en cm; \quad $70$ mm en cm; \quad $4$ km en m;
\quad $600$ cm en m.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:measure:3}
Mide con tu regla (al milímetro más cercano) los lados de un
triángulo dibujado por un compañero de clase y calcular la longitud total de su
frontera.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:measure:4}
Calcule, convirtiendo primero: $2$ metro $+ 45$ centímetros (en centímetros); \quad
$1$ kilómetros $+ 250$ m (en m); \quad $3$ centímetro $+ 8$ mm (en mm).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:measure:5}
Convertir: $3$ kg en g; \quad $2\,500$ g en kg y g; \quad
$4$ L en cl; \quad $350$ cl en L y cl.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:measure:6}
una receta necesita $750$ g de harina. Mona tiene un $1$ bolsa de kilos. ella tiene
suficiente? ¿Cuánto quedará?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:measure:7}
Lea los tiempos: la mano corta está entre $2$ y $3$, la mano larga
en el $6$; la mano corta en el $9$, la mano larga
en el $12$;
la mano corta casi en el $5$, la mano larga
en el $10$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:measure:8}
La escuela comienza a las $8{:}30$ y la mañana termina a las $11{:}45$. cuanto tiempo
es la mañana? (Cuente en saltos, como en
\cref{ex:g3:measure:duration}.)
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g3:measure:9}
El tren sale a las $9{:}50$ y el paseo dura $45$ minutos. ¿En qué?
hora llega? (Salte a $10{:}00$ primero.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g3:measure:10}
una botella sostiene $1$ L. Paul llena vasos de $20$ CL. cuantos llenos
¿Puede servir vasos? (Convierta primero).
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g3:measure:11}
lena camina $600$ Voy a la escuela y regreso todos los días. cuantos metros
ella camina en un $5$-día escolar semana? Da la respuesta en metros,
luego en kilómetros.
\end{exercise}
