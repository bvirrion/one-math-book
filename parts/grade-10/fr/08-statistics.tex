\chapter{Statistique descriptive}\label{ch:g10:stats}

La statistique résume une grande collection de nombres --- notes,
salaires, températures --- par quelques indicateurs bien choisis. Ce
chapitre introduit les deux familles d'indicateurs~: ceux qui localisent
le \emph{centre} des données (\omterm{def:g10:stats:mean}{moyenne}, \omterm{def:g10:stats:median}{médiane}) et ceux qui mesurent leur
\emph{dispersion} (\omterm{def:g10:stats:quartiles}{étendue}, \omterm{def:g10:stats:quartiles}{quartiles}). Un traitement plus approfondi,
incluant l'écart type, vient dans le \cref{ch:g11:stat}.

\section{Données et fréquences}

\begin{definition}[Série statistique]\label{def:g10:stats:series}
Une \emph{série statistique}\index{série statistique} est une liste de
valeurs $x_1, x_2, \dots, x_N$ observées sur une population de taille
$N$. Lorsqu'une valeur $x$ apparaît $n$ fois, $n$ est son
\emph{effectif} et le quotient
\[
f = \frac{n}{N}
\]
est sa \emph{fréquence}\index{fréquence} (souvent exprimée en
pourcentage). Les fréquences de toutes les valeurs s'additionnent à $1$.
\end{definition}

\begin{example}\label{ex:g10:stats:frequencies}
Les notes de $20$ élèves à un quiz sur $5$ points~:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|cccccc}
note $x$ & $0$ & $1$ & $2$ & $3$ & $4$ & $5$ \\
\hline
effectif $n$ & $1$ & $2$ & $4$ & $6$ & $5$ & $2$ \\
\omterm{def:g10:stats:series}{fréquence} $f$ & $0.05$ & $0.10$ & $0.20$ & $0.30$ & $0.25$ & $0.10$ \\
\end{tabular}
\end{center}
Vérification~: $1 + 2 + 4 + 6 + 5 + 2 = 20$ et les \omterm{def:g10:stats:series}{fréquences} somment à
$1$.
\end{example}

\section{Indicateurs de position}

\begin{definition}[Moyenne]\label{def:g10:stats:mean}
La \emph{moyenne}\index{moyenne} de la série est
\[
\bar x = \frac{x_1 + x_2 + \dots + x_N}{N}.
\]
Si les valeurs $v_1, \dots, v_k$ apparaissent avec les effectifs
$n_1, \dots, n_k$, le même nombre se calcule comme la \emph{moyenne
pondérée}
\[
\bar x = \frac{n_1 v_1 + n_2 v_2 + \dots + n_k v_k}{n_1 + n_2 + \dots + n_k}.
\]
\end{definition}

\begin{example}\label{ex:g10:stats:mean}
Pour le quiz de l'\cref{ex:g10:stats:frequencies}~:
\[
\bar x = \frac{1 \times 0 + 2 \times 1 + 4 \times 2 + 6 \times 3
+ 5 \times 4 + 2 \times 5}{20}
= \frac{0 + 2 + 8 + 18 + 20 + 10}{20}
= \frac{58}{20} = 2.9 .
\]
\end{example}

\begin{definition}[Médiane]\label{def:g10:stats:median}
La \emph{médiane}\index{médiane} est une valeur qui partage la série
\emph{triée} en deux moitiés~: au moins la moitié des valeurs sont
$\leq$ la médiane, et au moins la moitié sont $\geq$ à elle. En
pratique, trier les $N$ valeurs~;
\begin{itemize}
  \item si $N$ est impair, la médiane est la valeur du \omterm{prop:g10:coordgeom:midpoint}{milieu}, en
        position $\frac{N+1}{2}$~;
  \item si $N$ est pair, prendre le \omterm{prop:g10:coordgeom:midpoint}{milieu} des deux valeurs centrales, en
        positions $\frac N2$ et $\frac N2 + 1$.
\end{itemize}
\end{definition}

\begin{example}\label{ex:g10:stats:median}
Série triée de $7$ prix de maisons (en milliers)~: $120$, $150$, $160$,
$180$, $210$, $240$, $900$. La \omterm{def:g10:stats:median}{médiane} est la $4$-ième valeur, $180$. La
\omterm{def:g10:stats:mean}{moyenne} est $\frac{1960}{7} = 280$ --- plus grande que $6$ des $7$
prix~! Une valeur extrême ($900$) tire la \omterm{def:g10:stats:mean}{moyenne} bien plus que la
\omterm{def:g10:stats:median}{médiane}~: la \omterm{def:g10:stats:median}{médiane} est \emph{robuste}, la \omterm{def:g10:stats:mean}{moyenne} ne l'est pas.
\end{example}

\section{Indicateurs de dispersion}

\begin{definition}[Étendue, quartiles]\label{def:g10:stats:quartiles}
Pour une série triée~:
\begin{itemize}
  \item l'\emph{étendue}\index{étendue} est la différence entre la plus
        grande et la plus petite valeur~;
  \item le \emph{premier quartile}\index{quartile} $Q_1$ est la plus
        petite valeur telle qu'au moins un quart ($25\,\%$) des valeurs
        sont $\leq Q_1$~; le \emph{troisième quartile} $Q_3$ est la plus
        petite valeur telle qu'au moins les trois quarts ($75\,\%$) des
        valeurs sont $\leq Q_3$~;
  \item l'\emph{écart interquartile}\index{écart interquartile} $Q_3 - Q_1$ mesure la dispersion de
        la moitié centrale des données.
\end{itemize}
\end{definition}

\begin{method}[Trouver les quartiles]\label{met:g10:stats:quartiles}
Pour une série de $N$ valeurs triées~:
\begin{enumerate}
  \item calculer $\frac N4$~; s'il n'est pas entier, l'arrondir
        \emph{au-dessus} à l'entier suivant~; $Q_1$ est la valeur en
        cette position~;
  \item calculer $\frac{3N}{4}$~; arrondir au-dessus si besoin~; $Q_3$
        est la valeur en cette position.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:stats:quartiles}
Prendre les $11$ valeurs triées
\[
2,\ 3,\ 5,\ 5,\ 6,\ 7,\ 8,\ 9,\ 9,\ 10,\ 12 .
\]
\omterm{def:g10:stats:median}{Médiane}~: position $\frac{11+1}{2} = 6$, donc la \omterm{def:g10:stats:median}{médiane} est $7$.
$Q_1$~: $\frac{11}{4} = 2.75$, arrondir à $3$~; la $3$-ième valeur est
$5$.
$Q_3$~: $\frac{33}{4} = 8.25$, arrondir à $9$~; la $9$-ième valeur est
$9$.
\omterm{def:g10:stats:quartiles}{Étendue}~: $12 - 2 = 10$~; \omterm{def:g10:stats:quartiles}{écart interquartile}~: $9 - 5 = 4$.
\end{example}

\begin{example}[Comparer deux groupes]\label{ex:g10:stats:comparison}
Deux classes passent le même test (notes sur $20$)~:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccc}
 & \omterm{def:g10:stats:median}{médiane} & $Q_1$ & $Q_3$ \\
\hline
classe A & $12$ & $10$ & $14$ \\
classe B & $12$ & $6$ & $17$ \\
\end{tabular}
\end{center}
Même \omterm{def:g10:stats:median}{médiane}, dispersions très différentes~: la classe A est homogène
(la moitié de ses notes dans $\intcc{10}{14}$), la classe B est
hétérogène (sa moitié centrale s'étend sur $\intcc{6}{17}$). Les
indicateurs de position seuls ne disent jamais toute l'histoire.
\end{example}

\begin{remark}
Ni la \omterm{def:g10:stats:mean}{moyenne} ni la \omterm{def:g10:stats:median}{médiane} n'est «~meilleure~»~: la \omterm{def:g10:stats:mean}{moyenne} utilise
chaque valeur (et est nécessaire pour calculer des totaux), la \omterm{def:g10:stats:median}{médiane}
résiste aux valeurs extrêmes. Un résumé sérieux d'une série donne au
moins un indicateur de position \emph{et} un de dispersion.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g10:stats:1}
Voici le nombre de livres lus en un an par $15$ élèves~:
\[
0,\ 1,\ 1,\ 2,\ 2,\ 2,\ 3,\ 3,\ 4,\ 4,\ 5,\ 6,\ 7,\ 8,\ 12 .
\]
Calculer la \omterm{def:g10:stats:mean}{moyenne}, la \omterm{def:g10:stats:median}{médiane} et l'\omterm{def:g10:stats:quartiles}{étendue} de cette série.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:2}
Un dé a été lancé $50$ fois~:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|cccccc}
issue & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ \\
\hline
effectif & $7$ & $9$ & $8$ & $10$ & $6$ & $10$ \\
\end{tabular}
\end{center}
Calculer la \omterm{def:g10:stats:series}{fréquence} de chaque issue et la \omterm{def:g10:stats:mean}{moyenne} des issues.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:3}
Trouver la \omterm{def:g10:stats:median}{médiane}, $Q_1$ et $Q_3$ de la série triée
\[
1,\ 2,\ 4,\ 4,\ 5,\ 6,\ 6,\ 7,\ 8,\ 9,\ 9,\ 11 \qquad (N = 12),
\]
et dessiner le diagramme en boîte correspondant.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:4}
La \omterm{def:g10:stats:mean}{moyenne} de $24$ notes de test est $11.5$. Un $25$-ième élève passe le
test et obtient $19$. Quelle est la nouvelle \omterm{def:g10:stats:mean}{moyenne} de la classe~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:5}
Un élève a une \omterm{def:g10:stats:mean}{moyenne} de $12$ après $5$ tests, tous de même poids.
Quelle note au $6$-ième test porterait la \omterm{def:g10:stats:mean}{moyenne} à $13$~? Une \omterm{def:g10:stats:mean}{moyenne} de
$14$ est-elle atteignable (notes sur $20$)~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:6}
Inventer une série de $7$ valeurs positives ou nulles dont la \omterm{def:g10:stats:median}{médiane} est
$10$ et dont la \omterm{def:g10:stats:mean}{moyenne} est plus grande que $20$~; puis une dont la
\omterm{def:g10:stats:median}{médiane} est $10$ et dont la \omterm{def:g10:stats:mean}{moyenne} est plus petite que $6$. Pourquoi la
\omterm{def:g10:stats:mean}{moyenne} d'une telle série ne peut-elle jamais être plus petite que
$\frac{40}{7}$~? Que montrent ces exemples~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:7}
Dans une entreprise, les $9$ employés gagnent $1800$ chacun par mois et
le directeur gagne $12000$.
\begin{enumerate}
  \item Calculer le salaire moyen et le salaire médian.
  \item Quel indicateur décrit le mieux un salaire «~typique~» ici~?
        Pourquoi~?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:8}
Une classe de $12$ garçons a une taille \omterm{def:g10:stats:mean}{moyenne} de $168$~cm~; les $18$
filles de la même classe ont une taille \omterm{def:g10:stats:mean}{moyenne} de $163$~cm. Calculer la
taille \omterm{def:g10:stats:mean}{moyenne} de toute la classe. (Attention~: ce n'est \emph{pas} le
\omterm{prop:g10:coordgeom:midpoint}{milieu} de $168$ et $163$ --- pondérer les \omterm{def:g10:stats:mean}{moyennes} par les tailles des
groupes.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:stats:9}
Une série de $N$ valeurs $x_1, \dots, x_N$ a pour \omterm{def:g10:stats:mean}{moyenne} $\bar x$.
Chaque valeur est transformée en $y_i = a x_i + b$ pour des nombres
fixes $a$ et $b$.
\begin{enumerate}
  \item Montrer que la \omterm{def:g10:stats:mean}{moyenne} de la nouvelle série est $a \bar x + b$.
  \item Les températures d'une semaine, en degrés Celsius, ont pour
        \omterm{def:g10:stats:mean}{moyenne} $20$. Quelle est leur \omterm{def:g10:stats:mean}{moyenne} en degrés Fahrenheit
        ($F = 1.8\,C + 32$)~?
\end{enumerate}
\end{exercise}

\section{Problème~: l'individu moyen n'existe pas}

\begin{problem}[{Devoir du week-end --- moyenne contre médiane~:
valeurs aberrantes, barèmes rehaussés, l'entrepôt sur la route, et le
cockpit conçu pour personne}]
\label{pb:g10:stats:1}
En 1950, l'armée de l'air américaine mesura $4\,000$ pilotes sur dix
dimensions corporelles et construisit le cockpit de l'\emph{homme
moyen}. Il allait --- comme un lieutenant s'en aperçut en comptant
--- à peu près à personne. La statistique résume des foules par des
nombres uniques, et chaque résumé dit la vérité sur quelque chose et
ment sur autre chose. Ce problème traduit la \omterm{def:g10:stats:mean}{moyenne} et la \omterm{def:g10:stats:median}{médiane}
(\cref{def:g10:stats:mean}, \cref{def:g10:stats:median}) devant le
tribunal, laisse chacune montrer ce qu'elle sait faire de mieux, et
referme le dossier du cockpit.

\medskip
\noindent\textbf{Partie I --- Deux résumés à la barre.}
\begin{enumerate}
  \item Une rue vend sept maisons, en milliers d'euros~: $200$,
        $210$, $220$, $230$, $240$, $250$ --- et un manoir à
        $1\,200$. Calculer le prix moyen et le prix médian. Lequel
        des deux nombres décrit «~une maison de cette rue~»~?
  \item Retirer le manoir et recalculer les deux. De combien chaque
        résumé s'est-il déplacé~? Énoncer la morale sur les valeurs
        aberrantes.
  \item Un tableau de recensement~: sur $100$ familles, $20$ n'ont
        pas d'enfant, $25$ en ont un, $30$ en ont deux, $15$ en ont
        trois, $10$ en ont quatre. Calculer le nombre moyen
        d'enfants. Aucune famille «~n'a $1.7$ enfant~» --- quelle
        quantité réelle la \omterm{def:g10:stats:mean}{moyenne} code-t-elle donc~? (La
        multiplier par $100$.)
  \item Pour les quinze notes
        $4, 6, 7, 8, 9, 10, 10, 11, 12, 12, 13, 14, 15, 17, 19$~:
        déterminer la \omterm{def:g10:stats:median}{médiane} et les \omterm{def:g10:stats:quartiles}{quartiles} $Q_1$, $Q_3$
        (\cref{met:g10:stats:quartiles}).
  \item Pour la même série, donner l'\omterm{def:g10:stats:quartiles}{étendue} et l'\omterm{def:g10:stats:quartiles}{écart
        interquartile}. Que mesure chacun~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Barèmes rehaussés et classes
regroupées.}
\begin{enumerate}[resume]
  \item D'après l'\cref{exo:g10:stats:9}~: translater toutes les
        valeurs de $+b$ translate \omterm{def:g10:stats:mean}{moyenne}, \omterm{def:g10:stats:median}{médiane} et \omterm{def:g10:stats:quartiles}{quartiles} de
        $+b$~; les multiplier par $a$ les multiplie par $a$.
        Qu'advient-il de l'\omterm{def:g10:stats:quartiles}{étendue} et de l'\omterm{def:g10:stats:quartiles}{écart interquartile} lors
        d'une translation~? Lors d'une multiplication~? Justifier.
  \item Un contrôle a pour \omterm{def:g10:stats:mean}{moyenne} $9$ (sur $20$) et l'enseignant
        veut une \omterm{def:g10:stats:mean}{moyenne} de $12$. Deux barèmes sont proposés~:
        ajouter $3$ points à tout le monde, ou multiplier chaque
        note par $\frac43$. Calculer le sort d'un élève à $6$ et
        d'un élève à $18$ sous chaque barème. Lequel favorise qui
        --- et lequel se heurte à un problème de plafond~?
  \item La classe A ($20$ élèves) a pour \omterm{def:g10:stats:mean}{moyenne} $12$~; la classe B
        ($30$ élèves) a pour \omterm{def:g10:stats:mean}{moyenne} $10$. Calculer la \omterm{def:g10:stats:mean}{moyenne} des
        $50$ élèves réunis, et expliquer pourquoi elle ne vaut pas
        $11$ (la leçon des \omterm{def:g10:stats:mean}{moyennes} pondérées du devoir du week-end
        sur la \omterm{def:g10:stats:mean}{moyenne} harmonique, du volume précédent, transposée
        en salle de classe).
  \item Montrer que les \omterm{def:g10:stats:median}{médianes} refusent ce jeu~: calculer les
        \omterm{def:g10:stats:median}{médianes} de $A = \{1, 2, 3\}$ et $B = \{0, 10\}$, puis la
        \omterm{def:g10:stats:median}{médiane} de la liste réunie, et comparer avec la \omterm{def:g10:stats:mean}{moyenne} des
        deux \omterm{def:g10:stats:median}{médianes} pondérée par les effectifs.
  \item Patinage artistique~: cinq juges notent $5.2$, $5.3$,
        $5.4$, $5.5$, $9.9$. Calculer la \omterm{def:g10:stats:mean}{moyenne}, la \omterm{def:g10:stats:median}{médiane} et la
        \emph{\omterm{def:g10:stats:mean}{moyenne} élaguée} (retirer la note la plus basse et la
        plus haute, faire la \omterm{def:g10:stats:mean}{moyenne} du reste). Pourquoi la plupart
        des sports jugés élaguent-ils~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Lire une distribution.}
\begin{enumerate}[resume]
  \item Deux usines produisent des boulons de longueur \omterm{def:g10:stats:mean}{moyenne}
        $50$ mm. Usine A~: $Q_1 = 49.9$, $Q_3 = 50.1$~; usine B~:
        $Q_1 = 48$, $Q_3 = 52$. Il vous faut des boulons
        interchangeables~: quelle usine, et quel indicateur a
        tranché~?
  \item Dans un cabinet médical, l'attente \omterm{def:g10:stats:median}{médiane} est de $10$
        minutes mais l'attente \omterm{def:g10:stats:mean}{moyenne} est de $25$. Quelle forme de
        distribution ce couple trahit-il~? Construire une série de
        cinq valeurs ayant exactement cette \omterm{def:g10:stats:median}{médiane} et cette
        \omterm{def:g10:stats:mean}{moyenne}.
  \item «~Votre bébé est au 75\textsuperscript{e} centile pour la
        taille~» --- que signifie cette phrase~? Et tous les
        enfants d'une ville peuvent-ils être «~au-dessus de la
        \omterm{def:g10:stats:mean}{moyenne}~»~? Au-dessus de la \omterm{def:g10:stats:median}{médiane}~? Expliquer la
        différence.
  \item Classe A~: \omterm{def:g10:functions:extrema}{minimum} $4$, $Q_1\,8$, \omterm{def:g10:stats:median}{médiane} $11$, $Q_3\,13$,
        \omterm{def:g10:functions:extrema}{maximum} $17$. Classe B~: \omterm{def:g10:functions:extrema}{minimum} $2$, $Q_1\,9$, \omterm{def:g10:stats:median}{médiane}
        $11$, $Q_3\,12$, \omterm{def:g10:functions:extrema}{maximum} $19$. Comparer~: note typique,
        homogénéité de la moitié centrale, turbulence des extrêmes.
  \item La liste de contrôle du journaliste~: un titre annonce
        «~salaire moyen chez TechCorp~: $5\,000$ euros~». Énumérer
        les trois questions que ce problème vous a appris à poser
        avant de croire que ce titre dit quoi que ce soit d'un
        employé typique.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Le dossier du cockpit.}
\begin{enumerate}[resume]
  \item Fabriquer sa propre pièce à conviction~: construire cinq
        prix de maisons de \omterm{def:g10:stats:mean}{moyenne} $300$ et de \omterm{def:g10:stats:median}{médiane} $200$ (en
        milliers d'euros).
  \item La \omterm{def:g10:stats:mean}{moyenne} est le point d'équilibre~: démontrer à partir de
        la définition que les écarts à la \omterm{def:g10:stats:mean}{moyenne} se compensent
        toujours, $\sum (x_i - \bar x) = 0$, et le vérifier sur la
        série de la question 16.
  \item La \omterm{def:g10:stats:median}{médiane} minimise les tournées~: des boutiques se
        trouvent aux kilomètres $1$, $3$, $4$, $9$, $13$ d'une
        route, et un entrepôt doit minimiser la \emph{distance
        totale} aux cinq boutiques. Calculer ce total pour un
        entrepôt placé à la \omterm{def:g10:stats:median}{médiane} ($4$), à la \omterm{def:g10:stats:mean}{moyenne} ($6$), puis
        en $5$. Qui gagne~? (Le fait est général~: la \omterm{def:g10:stats:median}{médiane}
        minimise la distance totale en \omterm{def:g10:numbers:abs}{valeur absolue} --- essayer
        n'importe quel autre emplacement.)
  \item Le cockpit, résolu~: supposons qu'un pilote ait $30\,\%$ de
        chances d'être «~proche de la \omterm{def:g10:stats:mean}{moyenne}~» sur une dimension
        corporelle, indépendamment d'une dimension à l'autre sur
        les $10$. Estimer la probabilité d'être proche de la
