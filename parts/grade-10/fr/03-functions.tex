\chapter{Fonctions}\label{ch:g10:functions}

Une \omterm{def:g10:functions:function}{fonction} est une machine qui prend un nombre en entrée et produit un
nombre en sortie, toujours la même sortie pour la même entrée. Les
\omterm{def:g10:functions:function}{fonctions} sont les objets centraux de l'analyse que l'on construira au
cours des prochaines années (\cref{ch:g11:quad} et suivants)~; ce chapitre
met en place le vocabulaire --- \omterm{def:g10:functions:function}{ensemble de définition}, \omterm{def:g10:functions:function}{image}, \omterm{def:g10:functions:graph}{graphique},
variations --- et entraîne la compétence essentielle de lire des
informations sur un \omterm{def:g10:functions:graph}{graphique}.

\section{Vocabulaire~: images et antécédents}

\begin{definition}[Fonction]\label{def:g10:functions:function}
Soit $D$ un ensemble de \omterm{def:g10:numbers:sets}{nombres réels}. Une \emph{fonction}\index{fonction}
$f$ définie sur $D$ associe à chaque nombre $x \in D$ exactement un
\omterm{def:g10:numbers:sets}{nombre réel}, noté $f(x)$ et appelé l'\emph{image}\index{image} de $x$.
L'ensemble $D$ est l'\emph{ensemble de définition}\index{ensemble de
définition} de $f$. On écrit
\[
f : x \longmapsto f(x).
\]
Si $f(x) = y$, alors $x$ est un \emph{antécédent}\index{antécédent} de
$y$.
\end{definition}

\begin{remark}
Chaque $x$ dans l'\omterm{def:g10:functions:function}{ensemble de définition} a exactement une \omterm{def:g10:functions:function}{image}, mais un
nombre $y$ peut avoir plusieurs \omterm{def:g10:functions:function}{antécédents}, ou aucun. Pour
$f(x) = x^2$~: l'\omterm{def:g10:functions:function}{image} de $3$ est $9$, et $9$ a deux \omterm{def:g10:functions:function}{antécédents}, $3$ et
$-3$~; le nombre $-4$ n'a pas d'\omterm{def:g10:functions:function}{antécédent}.
\end{remark}

\begin{example}[Trouver un ensemble de définition]\label{ex:g10:functions:domain}
Lorsqu'une \omterm{def:g10:functions:function}{fonction} est donnée par une formule, son \omterm{def:g10:functions:function}{ensemble de
définition} est l'ensemble des $x$ pour lesquels la formule a un sens.
\begin{itemize}
  \item $f(x) = 3x^2 - 5x + 1$ a un sens pour tout $x$~: l'\omterm{def:g10:functions:function}{ensemble de
        définition} est $\R$.
  \item $g(x) = \dfrac{1}{x - 2}$ exige $x - 2 \neq 0$~: l'\omterm{def:g10:functions:function}{ensemble de
        définition} est tous les réels sauf $2$.
  \item $h(x) = \sqrt{x - 3}$ exige $x - 3 \geq 0$~: l'\omterm{def:g10:functions:function}{ensemble de
        définition} est $\intco{3}{+\infty}$.
\end{itemize}
\end{example}

\begin{example}[Calculer images et antécédents]\label{ex:g10:functions:images}
Soit $f(x) = x^2 - 4x + 3$, définie sur $\R$.

\emph{\omterm{def:g10:functions:function}{Image} de $5$~:} substituer $x = 5$~:
$f(5) = 25 - 20 + 3 = 8$.

\emph{\omterm{def:g10:functions:function}{Antécédents} de $3$~:} résoudre $f(x) = 3$~:
\[
x^2 - 4x + 3 = 3
\quad\Longleftrightarrow\quad
x^2 - 4x = 0
\quad\Longleftrightarrow\quad
x(x - 4) = 0,
\]
donc les \omterm{def:g10:functions:function}{antécédents} de $3$ sont $0$ et $4$.
\end{example}

\section{Le graphique d'une fonction}

\begin{definition}[Graphique]\label{def:g10:functions:graph}
Dans un repère, le \emph{graphique}\index{graphique} (ou la
\emph{courbe}) de $f$ est l'ensemble de tous les points $(x, f(x))$ pour
$x$ dans l'\omterm{def:g10:functions:function}{ensemble de définition}. Autrement dit, un point $(x, y)$
appartient au graphique exactement lorsque $y = f(x)$.
\end{definition}

\end{axis}
\end{tikzpicture}

jusqu'à l'axe vertical pour lire $f(3) = 1.25$.}
\end{omfigure}

\begin{method}[Lire un graphique]\label{met:g10:functions:reading}
Sur le \omterm{def:g10:functions:graph}{graphique} de $f$~:
\begin{enumerate}
  \item \emph{\omterm{def:g10:functions:function}{image} de $a$~:} monter verticalement depuis $x = a$ sur
        l'axe horizontal jusqu'à la courbe, puis horizontalement jusqu'à
        l'axe vertical~; la valeur lue est $f(a)$~;
  \item \emph{\omterm{def:g10:functions:function}{antécédents} de $b$~:} tracer la droite horizontale $y = b$~;
        les \omterm{def:g10:functions:function}{antécédents} sont les abscisses de tous ses points
        d'\omterm{def:g10:numbers:interunion}{intersection} avec la courbe~;
  \item \emph{solutions de $f(x) = k$~:} même chose que les \omterm{def:g10:functions:function}{antécédents} de
        $k$~;
  \item \emph{solutions de $f(x) \leq k$~:} les $x$ pour lesquels la
        courbe est sur ou sous la droite $y = k$.
\end{enumerate}
\end{method}

\end{axis}
\end{tikzpicture}

{\small Résoudre $f(x) = k$ graphiquement~: les solutions $x_1$ et $x_2$
sont les abscisses des points d'\omterm{def:g10:numbers:interunion}{intersection} de la courbe avec la droite
horizontale $y = k$. Ici $f(x) \leq k$ est réalisé sur
$\intcc{x_1}{x_2}$, où la courbe est sous la droite.}
\end{omfigure}

\begin{example}\label{ex:g10:functions:membership}
Le point $A(2, 5)$ appartient-il au \omterm{def:g10:functions:graph}{graphique} de $f(x) = x^2 + 1$~?
Calculer $f(2) = 4 + 1 = 5$~: oui, puisque $f(2)$ égale l'ordonnée de
$A$. Le point $B(3, 8)$ n'est pas sur le \omterm{def:g10:functions:graph}{graphique}, car
$f(3) = 10 \neq 8$.
\end{example}

\section{Variations d'une fonction}

\begin{definition}[Croissante, décroissante]\label{def:g10:functions:variations}
Soit $f$ définie sur un \omterm{def:g10:numbers:interval}{intervalle} $I$.
\begin{itemize}
  \item $f$ est \emph{croissante}\index{fonction croissante} sur $I$
        lorsque pour tous $u, v \in I$, si $u < v$ alors
        $f(u) \leq f(v)$~: les sorties croissent avec les entrées, et le
        \omterm{def:g10:functions:graph}{graphique} monte de gauche à droite.
  \item $f$ est \emph{décroissante}\index{fonction décroissante} sur $I$
        lorsque $u < v$ implique $f(u) \geq f(v)$~: le \omterm{def:g10:functions:graph}{graphique} descend
        de gauche à droite.
\end{itemize}
Avec des inégalités strictes $f(u) < f(v)$ (resp.\ $>$) on dit
\emph{strictement} croissante (resp.\ décroissante).
\end{definition}

\begin{definition}[Tableau de variations]\label{def:g10:functions:table}
Un \emph{tableau de variations}\index{tableau de variations} résume sur
quels \omterm{def:g10:numbers:interval}{intervalles} $f$ croît et décroît, avec des flèches $\nearrow$ et
$\searrow$, et enregistre les valeurs de $f$ aux points de retournement.
\end{definition}

\begin{example}\label{ex:g10:functions:table}
Considérons la \omterm{def:g10:functions:function}{fonction} grapquée ci-dessous sur $\intcc{-2}{4}$.

{\small Une \omterm{def:g10:functions:function}{fonction} définie sur $\intcc{-2}{4}$, \omterm{def:g10:functions:variations}{croissante} puis

\end{omfigure}

Lecture du \omterm{def:g10:functions:graph}{graphique}~: $f$ croît de $f(-2) = -1.5$ jusqu'à son \omterm{def:g10:functions:extrema}{maximum}
$f(1) = 3$, puis décroît jusqu'à $f(4) = -1.5$. Son \omterm{def:g10:functions:table}{tableau de variations}
est
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccc}

\hline

\end{tabular}
\end{center}
\end{example}

\begin{example}[Démontrer une variation]\label{ex:g10:functions:proof}
Montrer que $f(x) = 3x + 1$ est strictement \omterm{def:g10:functions:variations}{croissante} sur $\R$.
Prendre $u < v$ quelconques et comparer les \omterm{def:g10:functions:function}{images}~:
\[
f(v) - f(u) = (3v + 1) - (3u + 1) = 3(v - u) > 0,
\]
car $v - u > 0$. Donc $f(u) < f(v)$~: la \omterm{def:g10:functions:function}{fonction} est strictement
\omterm{def:g10:functions:variations}{croissante}. Le même calcul avec un coefficient négatif, par exemple
$g(x) = -2x + 5$, donne $g(v) - g(u) = -2(v-u) < 0$~: strictement
\omterm{def:g10:functions:variations}{décroissante}.
\end{example}

\section{Extrema}

\begin{definition}[Maximum, minimum]\label{def:g10:functions:extrema}
Soit $f$ définie sur $D$ et $a \in D$. La valeur $f(a)$ est le
\emph{maximum}\index{maximum} de $f$ sur $D$ lorsque $f(x) \leq f(a)$
pour tout $x \in D$~; c'est le \emph{minimum}\index{minimum} lorsque
$f(x) \geq f(a)$ pour tout $x \in D$. Sur un \omterm{def:g10:functions:graph}{graphique}, ce sont les points
les plus hauts et les plus bas de la courbe.
\end{definition}

\begin{example}\label{ex:g10:functions:extremum}
Montrer que $f(x) = (x - 1)^2 + 2$ a pour \omterm{def:g10:functions:extrema}{minimum} $2$ sur $\R$, atteint
en $x = 1$. Pour tout $x$, le carré $(x-1)^2$ est $\geq 0$, donc
$f(x) = (x-1)^2 + 2 \geq 2$~; et $f(1) = 0 + 2 = 2$, donc la valeur $2$ est
effectivement atteinte. Ces deux faits ensemble prouvent que $2$ est le
\omterm{def:g10:functions:extrema}{minimum}.
\end{example}

en pointillés $y = 2$, et la touche en $x = 1$~: le \omterm{def:g10:functions:extrema}{minimum} de $f$ est
$2$.}
\end{omfigure}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g10:functions:1}
Soit $f(x) = 2x^2 - 3x + 1$. Calculer les \omterm{def:g10:functions:function}{images} de $0$, $2$, $-1$ et
$\frac12$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:2}
Donner l'\omterm{def:g10:functions:function}{ensemble de définition} de chaque \omterm{def:g10:functions:function}{fonction}~:
\[
f(x) = 5x - 2, \qquad
g(x) = \frac{3}{x + 4}, \qquad
h(x) = \sqrt{2x - 6}, \qquad
k(x) = \frac{1}{x^2 + 1}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:3}
Soit $f(x) = x^2 - 2x$. Trouver tous les \omterm{def:g10:functions:function}{antécédents} de $0$, de $3$, et
de $-1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:4}
Le point $A(-1, 4)$ appartient-il au \omterm{def:g10:functions:graph}{graphique} de
$f(x) = x^2 - 2x + 1$~? Et le point $B(2, 1)$~? Justifier par un calcul.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:5}
Une \omterm{def:g10:functions:function}{fonction} $g$ définie sur $\intcc{-3}{5}$ a le \omterm{def:g10:functions:table}{tableau de variations}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccccc}
$x$ & $-3$ & & $0$ & & $3$ & & $5$ \\
\hline
$g$ & $1$ & $\searrow$ & $-2$ & $\nearrow$ & $4$ & $\searrow$ & $0$ \\
\end{tabular}
\end{center}
\begin{enumerate}
  \item Quels sont le \omterm{def:g10:functions:extrema}{maximum} et le \omterm{def:g10:functions:extrema}{minimum} de $g$ sur $\intcc{-3}{5}$~?
  \item Comparer $g(-1)$ et $g(-0.5)$ sans connaître leurs valeurs.
  \item Combien de solutions l'\omterm{def:g10:algebra:equation}{équation} $g(x) = 0$ a-t-elle au plus sur
        chaque \omterm{def:g10:numbers:interval}{intervalle} du tableau~?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:6}
Soit $f(x) = -2x + 7$. Montrer, en comparant $f(u)$ et $f(v)$ pour
$u < v$, que $f$ est strictement \omterm{def:g10:functions:variations}{décroissante} sur $\R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:7}
Montrer que la \omterm{def:g10:functions:function}{fonction} $f(x) = x^2 + 6x + 11$ a un \omterm{def:g10:functions:extrema}{minimum} sur $\R$, et
donner sa valeur et où il est atteint. (Indication~: écrire
$f(x) = (x + 3)^2 + c$ pour la bonne constante $c$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:8}
Un jardin rectangulaire a un périmètre de $40$~m. Soit $x$ sa largeur, en
mètres.
\begin{enumerate}
  \item Exprimer la longueur, puis l'aire $A(x)$, en \omterm{def:g10:functions:function}{fonction} de $x$. Pour
        quels $x$ cela a-t-il un sens~?
  \item Calculer $A(4)$, $A(8)$, $A(10)$, $A(12)$ et $A(16)$. Que
        conjecturer sur la forme d'aire maximale~?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:9}
Soit $f(x) = \dfrac{1}{x^2 + 1}$, définie sur $\R$.
\begin{enumerate}
  \item Montrer que $0 < f(x) \leq 1$ pour tout réel $x$.
  \item $f$ a-t-elle un \omterm{def:g10:functions:extrema}{maximum} sur $\R$~? Un \omterm{def:g10:functions:extrema}{minimum}~? Justifier.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:functions:10}
Soit $f(x) = x^2$ sur $\R$.
\begin{enumerate}
  \item Soit $0 \leq u < v$. Factoriser $f(v) - f(u)$ et en déduire que
        $f$ est strictement \omterm{def:g10:functions:variations}{croissante} sur $\intco{0}{+\infty}$.
  \item Adapter l'argument pour montrer que $f$ est strictement
        \omterm{def:g10:functions:variations}{décroissante} sur $\intoc{-\infty}{0}$.
\end{enumerate}
\end{exercise}

\section{Problème~: des machines nourries de leur propre sortie}

\begin{problem}[{Devoir du week-end --- points fixes, itération et
trois machines célèbres~: l'améliorateur de Héron, l'élévateur au
carré et les grêlons non résolus}]
\label{pb:g10:functions:1}
Une \omterm{def:g10:functions:function}{fonction} est une machine~: un nombre entre, un nombre sort. Les
expériences les plus intéressantes consistent à lui redonner à manger
\emph{sa propre sortie}, encore et encore --- et deux questions
dominent alors~: le procédé se stabilise-t-il quelque part (en un
\emph{point fixe}\index{point fixe}), et comment les variations de la
machine l'y conduisent-elles~? L'une des machines de ce problème
polit des racines carrées depuis deux mille ans~; une autre garde le
plus célèbre problème ouvert que les mathématiques élémentaires aient
à offrir.

\medskip
\noindent\textbf{Partie I --- La machine de Héron.}
On considère la \omterm{def:g10:functions:function}{fonction}
\[
f(x) = \frac12\left(x + \frac2x\right).
\]
\begin{enumerate}
  \item Donner l'\omterm{def:g10:functions:function}{ensemble de définition} de $f$, et calculer $f(1)$,
        $f(2)$ et $f\!\left(\frac32\right)$ sous forme de fractions
        exactes.
  \item Déterminer tous les \omterm{def:g10:functions:function}{antécédents} de $\frac32$~: résoudre
        $f(x) = \frac32$ (chasser le dénominateur et employer les
        méthodes du \cref{pb:g10:algebra:1}).
  \item Un \emph{\omterm{pb:g10:functions:1}{point fixe}} de $f$ est un nombre que la machine
        laisse inchangé~: $f(x) = x$. Déterminer les deux \omterm{pb:g10:functions:1}{points
        fixes} de $f$. Quelle vieille connaissance garde le \omterm{pb:g10:functions:1}{point fixe}
        positif~?
  \item En partant de $x = 1$, redonner trois fois à la machine sa
        propre sortie et noter les résultats exacts. Quelle suite du
        devoir du week-end sur l'irrationalité, dans le volume
        précédent, vient-on de reconstruire --- et qu'était la
        «~recette de Héron~», dans le langage des \omterm{def:g10:functions:function}{fonctions}~?
  \item Sur une même figure, tracer l'allure de la courbe de $f$
        (pour $x > 0$) et la droite $y = x$. Où sont les \omterm{pb:g10:functions:1}{points fixes}
        sur ce dessin~? (Comparer avec le devoir du week-end sur les
        deux thermomètres, dans le volume précédent~: la température
        qui se lit de la même façon dans les deux échelles reposait
        sur la même idée.)
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Pourquoi la machine ne peut pas
manquer sa cible.}
\begin{enumerate}[resume]
  \item Pour $0 < u < v$, montrer par réduction au même dénominateur
        que
        \[
        f(v) - f(u) = \frac{(v - u)(uv - 2)}{2uv} ,
        \]
        et en déduire les variations de $f$ sur
        $\intoo{0}{+\infty}$~: \omterm{def:g10:functions:variations}{décroissante} jusqu'à un certain point,
        puis \omterm{def:g10:functions:variations}{croissante} --- lequel~? (Même technique qu'à
        l'\cref{exo:g10:functions:10}.)
  \item En déduire que sur $\intoo{0}{+\infty}$ la \omterm{def:g10:functions:function}{fonction} $f$
        admet un \omterm{def:g10:functions:extrema}{minimum}, valant exactement $\sqrt2$ et atteint en
        $\sqrt2$. Que cela dit-il de \emph{toute} sortie de la
        machine (entrées positives)~?
  \item Montrer que si $x > \sqrt2$, alors $\sqrt2 < f(x) < x$~: une
        estimation trop grande est toujours améliorée, jamais
        dépassée. (Pour $f(x) < x$, comparer $\frac2x$ et $x$.)
  \item Rassembler les questions 6 à 8 dans le \omterm{def:g10:functions:table}{tableau de variations}
        de $f$ sur $\intoo{0}{+\infty}$
        (\cref{def:g10:functions:table}), \omterm{def:g10:functions:extrema}{minimum} compris.
  \item Une deuxième machine~: $g(x) = x^2$. Ses \omterm{pb:g10:functions:1}{points fixes} sont
        $0$ et $1$ (le vérifier). Itérer $g$ quatre fois à partir de
        $x = 0.9$, puis quatre fois à partir de $x = 1.1$
        (calculatrice, trois décimales). L'un des \omterm{pb:g10:functions:1}{points fixes}
        attire, l'autre repousse~: lequel est lequel~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- La machine à grêlons.}
Sur les entiers naturels, on pose $h(n) = \frac n2$ si $n$ est pair,
et $h(n) = 3n + 1$ si $n$ est impair. Les nombres rebondissent sous
$h$ comme des grêlons dans un nuage d'orage.
\begin{enumerate}[resume]
  \item Calculer le vol complet de $7$ sous $h$, jusqu'à $1$. Combien
        d'étapes faut-il, et à quelle hauteur monte-t-il~?
  \item Amorcer le vol de $27$ et calculer dix étapes. (Son vol
        complet dure $111$ étapes et culmine à $9\,232$ --- pour un
        point de départ égal à $27$.) Quelle leçon $7$ et $27$
        donnent-ils sur les règles simples~?
  \item Expliquer pourquoi tout vol qui atteint une puissance de $2$
        retombe droit sur $1$, et ce qui se passe en $1$ (calculer
        $h(1)$, $h(4)$, $h(2)$). La \omterm{def:g10:functions:function}{fonction} $h$ a-t-elle un \omterm{pb:g10:functions:1}{point
        fixe} parmi les entiers strictement positifs~? Que forme
        plutôt la fin de tout vol observé~?
  \item La \emph{conjecture de Collatz} affirme que tout nombre de
        départ finit par atteindre $1$. Les ordinateurs l'ont
        vérifiée bien au-delà de $10^{20}$~; personne ne l'a
        démontrée. Quelles sont les deux sortes de découvertes qui la
        \emph{réfuteraient}~? Et pourquoi une vérification massive ne
        clôt-elle toujours pas l'affaire (le devoir du week-end sur
        l'oracle aux allumettes, dans le volume précédent, avait déjà
        sonné cette alarme)~?
  \item «~Les mathématiques ne sont pas encore mûres pour de telles
        questions~», disait Paul Erd\H{o}s de cette conjecture. En
        une phrase~: qu'est-ce qui sépare la machine à grêlons de
        celle de Héron, où les questions 6 à 8 ont tout réglé~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Le mode d'emploi des machines.}
\begin{enumerate}[resume]
  \item \omterm{def:g10:functions:function}{Ensembles de définition} sous forme d'\omterm{def:g10:numbers:interval}{intervalles} (le langage
        du \cref{pb:g10:numbers:1})~: donner les \omterm{def:g10:functions:function}{ensembles de
        définition} de $x \mapsto \sqrt{x - 3}$, \
        $x \mapsto \frac{1}{x^2 - 9}$, \
        $x \mapsto \sqrt{9 - x^2}$.
  \item Retour au $f$ de Héron~: déterminer les \omterm{def:g10:functions:function}{antécédents} exacts de
        $3$ (résoudre $f(x) = 3$ par mise sous forme canonique).
  \item On lance une pierre vers le haut~: sa hauteur vaut
        $H(t) = 20t - 5t^2$ mètres après $t$ secondes. Sur quel
        \omterm{def:g10:numbers:interval}{intervalle} de temps la formule a-t-elle un sens physique~?
        Mettre sous forme canonique pour trouver la hauteur maximale
        et l'instant où elle est atteinte
        (\cref{def:g10:functions:extrema}), puis donner le \omterm{def:g10:functions:table}{tableau de
        variations}.
  \item Pendant combien de temps la pierre est-elle à au moins
        $15$ m de hauteur~? (Résoudre $H(t) \geq 15$ à l'aide d'un
        \omterm{def:g10:algebra:signtable}{tableau de signes}, \cref{met:g10:algebra:signtable}.)
  \item Pour finir~: ce problème a fait tourner quatre machines ---
        l'améliorateur de Héron, l'élévateur au carré, les grêlons,
        la hauteur de la pierre. En deux ou trois phrases, énoncer ce
        que toute \omterm{def:g10:functions:function}{fonction} possède (un \omterm{def:g10:functions:function}{ensemble de définition}, une
        règle, une courbe) et les deux grandes questions que ce
        problème a posées à chacune (où l'itération se
        stabilise-t-elle~? comment la \omterm{def:g10:functions:function}{fonction} varie-t-elle~?) ---
        questions que les chapitres suivants reprennent une à une.
\end{enumerate}
\end{problem}
