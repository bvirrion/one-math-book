\chapter{Algebra: vergelijkingen en ongelijkheden}\label{ch:g10:algebra}

Algebra is de kunst van het rekenen met letters. Dit hoofdstuk herneemt en
verscherpt de twee basisbewegingen --- \omterm{def:g10:algebra:expand}{uitwerken} en \omterm{def:g10:algebra:expand}{ontbinden} --- en gebruikt
ze om \omterm{def:g10:algebra:equation}{vergelijkingen} en ongelijkheden stelselmatig op te lossen, met de
\omterm{def:g10:algebra:signtable}{tekentabel} als centraal gereedschap.

\section{Uitwerken en ontbinden}

\begin{definition}[Uitwerken, ontbinden]\label{def:g10:algebra:expand}
Een uitdrukking \emph{uitwerken}\index{uitwerken} betekent producten omzetten
in sommen; \emph{ontbinden}\index{ontbinden in factoren} is het omgekeerde,
een som omzetten in een product. De basisregel is de distributiviteit:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g10:algebra:expand}
Werk $(2x + 3)(x - 5)$ stap voor stap uit:
\begin{align*}
(2x + 3)(x - 5)
&= 2x \times x + 2x \times (-5) + 3 \times x + 3 \times (-5) \\
&= 2x^2 - 10x + 3x - 15 \\
&= 2x^2 - 7x - 15 .
\end{align*}
\end{example}

\begin{theorem}[Merkwaardige producten]\label{thm:g10:algebra:identities}
Voor alle \omterm{def:g10:numbers:sets}{reële getallen} $a$ en $b$ geldt
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
Elk ervan is een rechtstreekse uitwerking. Voor het eerste:
\[
(a+b)^2 = (a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2 .
\]
Het tweede volgt door $b$ door $-b$ te vervangen:
$(a-b)^2 = a^2 + 2a(-b) + (-b)^2 = a^2 - 2ab + b^2$. En voor het derde:
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$.
\end{proof}

\begin{example}\label{ex:g10:algebra:identities}
Heen gebruikt (\omterm{def:g10:algebra:expand}{uitwerken}) en terug (\omterm{def:g10:algebra:expand}{ontbinden}):
\begin{align*}
(3x + 2)^2 &= 9x^2 + 12x + 4,
& x^2 - 25 &= (x+5)(x-5), \\
(x - 4)^2 &= x^2 - 8x + 16,
& 4x^2 + 4x + 1 &= (2x + 1)^2 .
\end{align*}
Ook het hoofdrekenen vaart er wel bij: $101 \times 99 = (100+1)(100-1) =
10000 - 1 = 9999$.
\end{example}

\begin{method}[Een uitdrukking ontbinden]\label{met:g10:algebra:factor}
Probeer, in deze volgorde:
\begin{enumerate}
  \item \emph{Gemeenschappelijke factor:} zoek een factor die in elke term
        voorkomt en haal hem buiten haakjes, bijvoorbeeld
        $6x^2 + 4x = 2x(3x + 2)$. Die factor mag een hele haak zijn:
        $(x+1)(2x-3) + (x+1)(x+7) = (x+1)(3x+4)$.
  \item \emph{Merkwaardig product:} herken $a^2 - b^2$ of
        $a^2 \pm 2ab + b^2$, bijvoorbeeld $9x^2 - 16 = (3x)^2 - 4^2 =
        (3x+4)(3x-4)$.
  \item Combineer beide, en controleer het resultaat door het weer uit te
        werken.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:factorsteps}
Ontbind $E = (2x + 1)^2 - (x - 3)^2$. Dit is een verschil van twee kwadraten
$a^2 - b^2$ met $a = 2x+1$ en $b = x-3$:
\begin{align*}
E &= \bigl[(2x+1) + (x-3)\bigr] \times \bigl[(2x+1) - (x-3)\bigr] \\
  &= (3x - 2)(2x + 1 - x + 3) \\
  &= (3x - 2)(x + 4).
\end{align*}
Let op de tweede haak: $x - 3$ aftrekken betekent $x$ aftrekken \emph{en}
$3$ optellen.
\end{example}

\section{Vergelijkingen}

\begin{definition}[Vergelijking, oplossing]\label{def:g10:algebra:equation}
Een \emph{vergelijking}\index{vergelijking} is een gelijkheid met een
onbekend getal $x$ erin; een \emph{oplossing} is een waarde van $x$ die de
gelijkheid waar maakt. De vergelijking oplossen betekent \emph{alle}
oplossingen vinden.
\end{definition}

Twee \omterm{def:g10:algebra:equation}{vergelijkingen} heten \emph{gelijkwaardig} wanneer ze precies dezelfde
oplossingen hebben. Bij beide leden hetzelfde getal optellen, of beide leden
met eenzelfde getal \emph{verschillend van nul} vermenigvuldigen, levert een
gelijkwaardige \omterm{def:g10:algebra:equation}{vergelijking}.

\begin{example}[Vergelijking van de eerste graad]\label{ex:g10:algebra:firstdegree}
Los $5x - 7 = 2x + 8$ op, stap voor stap:
\begin{align*}
5x - 7 &= 2x + 8 \\
5x - 2x &= 8 + 7 && \text{(tel $7 - 2x$ bij beide leden op)} \\
3x &= 15 \\
x &= 5 && \text{(deel beide leden door $3$).}
\end{align*}
De enige oplossing is $5$. Controle: $5 \times 5 - 7 = 18$ en
$2 \times 5 + 8 = 18$.
\end{example}

\begin{theorem}[Nulproductregel]\label{thm:g10:algebra:zeroproduct}
Een product van \omterm{def:g10:numbers:sets}{reële getallen} is nul dan en slechts dan als minstens één van
zijn factoren nul is:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ of } \ B = 0 .
\]
\end{theorem}

\begin{proof}
Is $A = 0$ of $B = 0$, dan is het product duidelijk $0$. Omgekeerd, stel dat
$AB = 0$ en $A \neq 0$. Beide leden door $A$ delen geeft $B = 0$.
\end{proof}

\begin{method}[Oplossen door te ontbinden]\label{met:g10:algebra:zeroproduct}
Om een \omterm{def:g10:algebra:equation}{vergelijking} op te lossen waarvan het rechterlid $0$ is en het
linkerlid ontbonden kan worden:
\begin{enumerate}
  \item breng alles naar links, zodat de \omterm{def:g10:algebra:equation}{vergelijking} $E = 0$ luidt;
  \item ontbind $E$ in een product van factoren van de eerste graad;
  \item pas de nulproductregel toe en los elke kleine \omterm{def:g10:algebra:equation}{vergelijking} op;
  \item verzamel alle gevonden oplossingen.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:zeroproduct}
Los $(x - 2)(3x + 6) = 0$ op: ofwel $x - 2 = 0$, wat $x = 2$ geeft, ofwel
$3x + 6 = 0$, wat $x = -2$ geeft. Oplossingen: $-2$ en $2$.

Los $x^2 = 9x$ op. Deel \emph{niet} door $x$ (dat kan nul zijn!); breng alles
naar één lid en ontbind:
\[
x^2 - 9x = 0
\quad\Longleftrightarrow\quad
x(x - 9) = 0
\quad\Longleftrightarrow\quad
x = 0 \ \text{ of } \ x = 9 .
\]
\end{example}

\begin{example}[Vergelijking met een quotiënt]\label{ex:g10:algebra:quotient}
Los $\dfrac{x - 1}{x + 2} = 0$ op. Een quotiënt is nul precies wanneer de
teller nul is \emph{en} de noemer niet. Hier geeft $x - 1 = 0$ dat $x = 1$,
en $1 + 2 = 3 \neq 0$: de enige oplossing is $1$. De waarde $x = -2$ is
\emph{verboden} (deling door nul) en moet altijd eerst worden uitgesloten.
\end{example}

\section{Ongelijkheden en tekentabellen}

\begin{proposition}[Rekenregels voor ongelijkheden]\label{prop:g10:algebra:ineqrules}
Zij $a \leq b$.
\begin{enumerate}
  \item Voor elke $c$: $a + c \leq b + c$ (optellen behoudt de orde).
  \item Is $c > 0$: $ac \leq bc$ (vermenigvuldigen met een positief getal
        behoudt de orde).
  \item Is $c < 0$: $ac \geq bc$ (vermenigvuldigen met een negatief getal
        \emph{keert} de orde om).
\end{enumerate}
\end{proposition}

\begin{proof}
1. $(b + c) - (a + c) = b - a \geq 0$.
2. $bc - ac = (b-a)c$ is het product van twee niet-negatieve getallen, dus
$bc \geq ac$.
3. Nu is $(b-a)c \leq 0$ als product van een niet-negatief en een negatief
getal, dus $bc \leq ac$.
\end{proof}

\begin{example}\label{ex:g10:algebra:firstineq}
Los $-2x + 3 > 9$ op:
\begin{align*}
-2x + 3 &> 9 \\
-2x &> 6 && \text{(trek 3 af)} \\
x &< -3 && \text{(deel door $-2 < 0$: keer de ongelijkheid om).}
\end{align*}
De oplossingsverzameling is $\intoo{-\infty}{-3}$.
\end{example}

\begin{definition}[Tekentabel]\label{def:g10:algebra:signtable}
Een \emph{tekentabel}\index{tekentabel} noteert, met één rij per factor, het
teken ($+$ of $-$) van elke factor van een uitdrukking terwijl $x$ heel $\R$
doorloopt, met een $0$ bij elke waarde waar de factor verdwijnt; een laatste
rij geeft het teken van het product, kolom per kolom volgens de tekenregel.
\end{definition}

\begin{example}\label{ex:g10:algebra:signtable}
Het teken van $P(x) = (x - 2)(3 - x)$. De factor $x - 2$ verdwijnt in $2$ en
is daarna positief; de factor $3 - x$ verdwijnt in $3$ en is daarvóór
positief:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccc}
$x$ & $-\infty$ \quad & $2$ & & $3$ & \quad $+\infty$ \\
\hline
$x - 2$ & $-$ & $0$ & $+$ & & $+$ \\
$3 - x$ & $+$ & & $+$ & $0$ & $-$ \\
\hline
$P(x)$ & $-$ & $0$ & $+$ & $0$ & $-$ \\
\end{tabular}
\end{center}
Dus is $P(x) > 0$ precies op $\intoo{2}{3}$, en $P(x) \leq 0$ op
$\intoc{-\infty}{2} \cup \intco{3}{+\infty}$.
\end{example}

\begin{method}[Een ongelijkheid oplossen met een tekentabel]\label{met:g10:algebra:signtable}
Om een ongelijkheid als $E(x) \geq 0$ of $E(x) < 0$ op te lossen:
\begin{enumerate}
  \item breng alles naar het linkerlid, zodat rechts $0$ staat
        (vermenigvuldig beide leden nooit met een uitdrukking waarvan het
        teken onbekend is);
  \item ontbind het linkerlid, of breng het op één noemer als het een
        quotiënt is;
  \item bouw de \omterm{def:g10:algebra:signtable}{tekentabel} met één rij per factor (zet bij een quotiënt de
        verboden waarden met een dubbele streep in de laatste rij);
  \item lees de intervallen af waar het gevraagde teken geldt, en ga voor
        elke grens na of ze meetelt.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:quotientineq}
Los $\dfrac{x + 1}{x - 3} \leq 0$ op. De teller verdwijnt in $-1$, de noemer
in $3$ (verboden waarde). Het quotiënt is negatief wanneer beide een
tegengesteld teken hebben, en dat gebeurt voor $x$ tussen $-1$ en $3$. De
grens waar de \emph{teller} verdwijnt telt mee, de verboden waarde niet: de
oplossingsverzameling is $\intco{-1}{3}$.
\end{example}

\section{Opgave: van Babylon tot de gulden snede}

\begin{problem}[{Weekendopgave --- som-en-productvraagstukken, het kwadraat
aanvullen zoals al-Chwarizmi het tekende, en de vergelijking
$x^2 = x + 1$}]
\label{pb:g10:algebra:1}
Vierduizend jaar geleden losten Babylonische schrijvers dit op in klei:
\emph{vind twee getallen als je hun som en hun product kent}. Hun truc ---
zoek de getallen symmetrisch rond de halve som --- kraakt vandaag nog elke
vierkantsvergelijking, onder haar moderne naam \emph{kwadraatafsplitsing}.
Deze opgave leert de truc, kijkt toe hoe al-Chwarizmi letterlijk een vierkant
aanvult, en zet de methode aan het werk op de beroemdste verhouding van
allemaal.

\medskip
\noindent\textbf{Deel I --- Het Babylonische vraagstuk.}
\begin{enumerate}
  \item Opwarming (\cref{thm:g10:algebra:identities}): ontbind
        $x^2 - 10x + 25$, daarna $9x^2 - 49$, en ten slotte --- door een
        merkwaardig product met een verschil van kwadraten te combineren ---
        $x^2 + 6x + 9 - 4$.
  \item Los $(x + 1)(x + 5) = 0$ en $(3x - 7)(3x + 7) = 0$ op
        (\cref{thm:g10:algebra:zeroproduct}).
  \item Het vraagstuk van de schrijver: twee getallen hebben som $10$ en
        product $21$. Schrijf ze in Babylonische stijl als $5 - t$ en
        $5 + t$ (symmetrisch rond de halve som), bepaal $t$ en daarna de twee
        getallen.
  \item Dezelfde methode: som $14$, product $33$.
  \item Het algemene recept: toon aan dat de twee getallen bij som $s$ en
        product $p$ gelijk zijn aan
        \[
        \frac s2 - t
        \quad\text{en}\quad
        \frac s2 + t,
        \qquad\text{waarbij}\quad
        t^2 = \left(\frac s2\right)^2 - p ,
        \]
        en formuleer de voorwaarde op $s$ en $p$ opdat het vraagstuk een
        oplossing heeft.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het kwadraat aanvullen.}
\begin{enumerate}[resume]
  \item Werk $(x - a)(x - b)$ uit en besluit: de oplossingen van
        $x^2 - sx + p = 0$ zijn precies de paren getallen met som $s$ en
        product $p$. (De schrijvers losten vierkantsvergelijkingen op zonder
        het te weten.)
  \item Ga na dat $x^2 - 10x + 21 = (x - 5)^2 - 4$, en los
        $x^2 - 10x + 21 = 0$ op door het rechterlid als verschil van
        kwadraten te \omterm{def:g10:algebra:expand}{ontbinden}. Vergelijk met vraag 3.
  \item Los $x^2 + 6x - 7 = 0$ op door het kwadraat aan te vullen.
  \item Vul het kwadraat aan in $x^2 + 4x + 5$, en leg uit waarom de
        \omterm{def:g10:algebra:equation}{vergelijking} $x^2 + 4x + 5 = 0$ \emph{geen} reële oplossing heeft.
        Formuleer het algemene criterium: wanneer verdwijnt een uitdrukking
        van de tweede graad, eenmaal geschreven als $(x + h)^2 + k$?
  \item De naam is letterlijk bedoeld. In het Bagdad van de negende eeuw
        tekende al-Chwarizmi $x^2 + 6x = 7$ als een vierkant met zijde $x$
        met aan twee zijden een rechthoek $3 \times x$ geplakt, en
        \emph{vulde het vierkant aan} met de ontbrekende hoek $3 \times 3$.
        Teken de figuur en leg uit hoe ze de \omterm{def:g10:algebra:equation}{vergelijking} omzet in
        $(x + 3)^2 = 16$ --- en los daarna op.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Tekens, tabellen en twee oude beloften.}
\begin{enumerate}[resume]
  \item Bouw de \omterm{def:g10:algebra:signtable}{tekentabel} van $(x - 3)(x + 2)$
        (\cref{def:g10:algebra:signtable}) en los $(x - 3)(x + 2) < 0$ op.
  \item Los $x^2 \leq 10x - 21$ op (breng alles naar één lid en hergebruik
        vraag 7).
  \item Los $\dfrac{x - 1}{x + 2} \geq 0$ op met een \omterm{def:g10:algebra:signtable}{tekentabel}, met oog voor
        de verboden waarde.
  \item Een oude belofte ingelost: voor de boer uit de weekendopgave over de
        omheining van koningin Dido in het onderbouwvolume, met $20$ m
        omheining, voldoet de oppervlakte van een wei $x \times (10 - x)$ aan
        \[
        A(x) = x(10 - x) = 25 - (x - 5)^2 .
        \]
        Ga de identiteit na, en lees er een \emph{bewijs} uit af van wat in
        jaar 6 nog maar een vermoeden kon zijn: de grootste oppervlakte, en
        de unieke $x$ die haar bereikt.
  \item Nog een: bewijs dat $x + \dfrac1x \geq 2$ voor elke $x > 0$, met
        gelijkheid alleen in $x = 1$. (Vermenigvuldig met $x$, breng samen,
        herken een kwadraat --- de ongelijkheid tussen het rekenkundig en het
        meetkundig gemiddelde uit de weekendopgave over de betrekkingen van
        Euclides in het onderbouwvolume, in een nieuw kleedje.)
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De gulden snede.}
Verdeel een lijnstuk van lengte $1$ in een groot stuk $x$ en een klein stuk
$1 - x$ zó dat \emph{het geheel zich tot het grote stuk verhoudt als het
grote stuk tot het kleine}: $\frac{1}{x} = \frac{x}{1 - x}$.
\begin{enumerate}[resume]
  \item Toon aan dat de voorwaarde luidt: $x^2 + x - 1 = 0$.
  \item Los op door het kwadraat aan te vullen, en houd de wortel over die
        een lengte is: $x = \frac{\sqrt5 - 1}{2} \approx 0.618$.
  \item Het \emph{gulden getal} is $\varphi = \frac1x =
        \frac{2}{\sqrt5 - 1}$. Maak de noemer wortelvrij (vermenigvuldig
        teller en noemer met $\sqrt5 + 1$: een merkwaardig product aan het
        werk) en toon aan dat $\varphi = \frac{\sqrt5 + 1}{2} \approx 1.618$.
  \item Ga de twee magische identiteiten van $\varphi$ \emph{exact} na:
        \[
        \varphi^2 = \varphi + 1,
        \qquad
        \frac{1}{\varphi} = \varphi - 1 .
        \]
  \item Slotstuk: leg uit waarom $\varphi$ \omterm{ex:g10:numbers:classify}{irrationaal} is (de weekendopgave
        over irrationaliteit in het onderbouwvolume levert het sleutelfeit
        over $\sqrt5$), en bereken daarna de verhoudingen $\frac53$,
        $\frac85$, $\frac{13}{8}$, $\frac{21}{13}$ van opeenvolgende
        getallen van Hemachandra en Fibonacci (de weekendopgave over het
        tellen van ritmes in het onderbouwvolume) tot op drie decimalen. Wat
        lijken ze na te jagen? (Het bewijs dat ze het inhalen hoort bij de
        hoofdstukken over rijen.)
\end{enumerate}
\end{problem}
