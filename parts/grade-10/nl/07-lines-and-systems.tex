\chapter{Vergelijkingen van rechten en lineaire stelsels}\label{ch:g10:lines}

Een rechte in het coördinatenvlak wordt door een \omterm{def:g10:algebra:equation}{vergelijking} beschreven, en
twee rechten samen vormen een stelsel van \omterm{def:g10:algebra:equation}{vergelijkingen} waarvan de
oplossingen hun snijpunten zijn. Dit hoofdstuk verbindt drie talen ---
de meetkundige (rechten), de algebraïsche (\omterm{def:g10:algebra:equation}{vergelijkingen}) en de functionele
(\omterm{def:g10:reffunc:affine}{affiene functies} uit \cref{ch:g10:reffunc}) --- en leert de twee gebruikelijke
manieren om een \omterm{def:g10:lines:system}{lineair stelsel} op te lossen.

\section{De vergelijking van een rechte}

\begin{theorem}[Herleide vergelijking]\label{thm:g10:lines:reduced}
In een \omterm{def:g10:coordgeom:system}{assenstelsel} geldt:
\begin{enumerate}
  \item elke niet-verticale rechte heeft een \omterm{def:g10:algebra:equation}{vergelijking} van de vorm
        \[
        y = mx + p ,
        \]
        met $m$ (de \emph{richtingscoëfficiënt}\index{richtingscoëfficiënt})
        en $p$ (de $y$-afsnede) ondubbelzinnig bepaald --- dit is de
        \emph{herleide vergelijking}\index{herleide vergelijking} van de
        rechte;
  \item elke verticale rechte heeft een \omterm{def:g10:algebra:equation}{vergelijking} $x = c$;
  \item een punt ligt op de rechte precies wanneer zijn \omterm{def:g10:coordgeom:system}{coördinaten} aan de
        \omterm{def:g10:algebra:equation}{vergelijking} voldoen.
\end{enumerate}
\end{theorem}
\admitted

\begin{proposition}[Richtingscoëfficiënt uit twee punten]\label{prop:g10:lines:slope}
De \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} van de (niet-verticale) rechte door $A(x_A, y_A)$ en
$B(x_B, y_B)$, met $x_A \neq x_B$, is
\[
m = \frac{y_B - y_A}{x_B - x_A}
\qquad
\text{(``verandering van $y$ gedeeld door verandering van $x$'').}
\]
\end{proposition}

\begin{proof}
Beide punten voldoen aan $y = mx + p$, dus is
$y_B - y_A = (mx_B + p) - (mx_A + p) = m(x_B - x_A)$; deel door
$x_B - x_A \neq 0$.
\end{proof}

\begin{method}[De vergelijking van een rechte door twee punten]\label{met:g10:lines:through}
Gegeven $A$ en $B$ met $x_A \neq x_B$:
\begin{enumerate}
  \item bereken de \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt}
        $m = \dfrac{y_B - y_A}{x_B - x_A}$;
  \item schrijf $y = mx + p$ en vul de \omterm{def:g10:coordgeom:system}{coördinaten} van $A$ (of $B$) in om $p$
        te vinden;
  \item controleer de gevonden \omterm{def:g10:algebra:equation}{vergelijking} met het \emph{andere} punt.
\end{enumerate}
Is $x_A = x_B$, dan is de rechte verticaal: haar \omterm{def:g10:algebra:equation}{vergelijking} is $x = x_A$.
\end{method}

\begin{example}\label{ex:g10:lines:through}
De rechte door $A(1, 5)$ en $B(3, 1)$. \omterm{def:g10:reffunc:affine}{Richtingscoëfficiënt}:
$m = \frac{1 - 5}{3 - 1} = \frac{-4}{2} = -2$. Dan is $y = -2x + p$; het punt
$A$ geeft $5 = -2 \times 1 + p$, dus $p = 7$. \omterm{def:g10:algebra:equation}{Vergelijking}: $y = -2x + 7$.
Controle met $B$: $-2 \times 3 + 7 = 1$.
\end{example}

\begin{proposition}[Evenwijdige rechten]\label{prop:g10:lines:parallel}
Twee niet-verticale rechten zijn evenwijdig dan en slechts dan als ze
dezelfde \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} hebben. Twee verschillende evenwijdige rechten
hebben geen gemeenschappelijk punt; twee rechten met verschillende
\omterm{def:g10:reffunc:affine}{richtingscoëfficiënten} hebben er precies één.
\end{proposition}

\begin{proof}
Zij $y = mx + p$ en $y = m'x + p'$ de twee rechten. Een gemeenschappelijk
punt lost $mx + p = m'x + p'$ op, dus $(m - m')x = p' - p$. Is $m \neq m'$,
dan heeft dat precies één oplossing $x = \frac{p' - p}{m - m'}$: één
snijpunt. Is $m = m'$, dan luidt de \omterm{def:g10:algebra:equation}{vergelijking} $0 = p' - p$: geen oplossing
wanneer $p \neq p'$ (verschillende evenwijdige rechten), en elke $x$ wanneer
$p = p'$ (dezelfde rechte). Evenwijdigheid (geen snijpunt hebben, of
samenvallen) is dus gelijkwaardig met $m = m'$.
\end{proof}

\begin{remark}[Richtingsvector]
De rechte $y = mx + p$ klimt $m$ eenheden per eenheid naar rechts, zodat de
\omterm{def:g10:vectors:vector}{vector} $\vec u\,(1, m)$ een \emph{richtingsvector}\index{richtingsvector} van
de rechte is: de rechte is evenwijdig met $\vec u$. Twee rechten zijn
evenwijdig precies wanneer hun richtingsvectoren \omterm{def:g10:vectors:collinear}{collineair} zijn
(\cref{thm:g10:vectors:det}).
\end{remark}

\section{Lineaire stelsels}

\begin{definition}[Lineair stelsel]\label{def:g10:lines:system}
Een \emph{stelsel van twee lineaire vergelijkingen}\index{lineair stelsel} in
de onbekenden $x$ en $y$ is een paar \omterm{def:g10:algebra:equation}{vergelijkingen}
\[
\begin{cases}
a x + b y = c \\
a' x + b' y = c'
\end{cases}
\]
waaraan \emph{tegelijk} voldaan moet zijn. Een \emph{oplossing} is een paar
$(x, y)$ dat aan beide voldoet. Elke \omterm{def:g10:algebra:equation}{vergelijking} beschrijft een rechte, dus
het stelsel oplossen betekent de \omterm{def:g10:numbers:interunion}{doorsnede} van twee rechten bepalen.
\end{definition}

\begin{method}[Substitutie]\label{met:g10:lines:substitution}
\begin{enumerate}
  \item Los één \omterm{def:g10:algebra:equation}{vergelijking} op naar één onbekende --- kies de gemakkelijkste,
        bijvoorbeeld $y$ in \omterm{def:g10:functions:function}{functie} van $x$;
  \item vul die uitdrukking in de \emph{andere} \omterm{def:g10:algebra:equation}{vergelijking} in, die nu
        alleen nog $x$ bevat;
  \item los op naar $x$ en bereken daarna $y$ uit stap 1;
  \item controleer het paar in beide oorspronkelijke \omterm{def:g10:algebra:equation}{vergelijkingen}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:lines:substitution}
Los $\begin{cases} x + 2y = 8 \\ 3x - y = 3 \end{cases}$ op met substitutie.
De tweede \omterm{def:g10:algebra:equation}{vergelijking} geeft $y = 3x - 3$. Invullen in de eerste:
\[
x + 2(3x - 3) = 8
\ \Longleftrightarrow\
x + 6x - 6 = 8
\ \Longleftrightarrow\
7x = 14
\ \Longleftrightarrow\
x = 2 ,
\]
en dan $y = 3 \times 2 - 3 = 3$. De oplossing is het paar $(2, 3)$.
Controle: $2 + 2 \times 3 = 8$ en $3 \times 2 - 3 = 3$.
\end{example}

\begin{method}[Eliminatie]\label{met:g10:lines:elimination}
\begin{enumerate}
  \item Vermenigvuldig elke \omterm{def:g10:algebra:equation}{vergelijking} met een goed gekozen getal, zodat
        één onbekende in de twee \omterm{def:g10:algebra:equation}{vergelijkingen} tegengestelde coëfficiënten
        krijgt;
  \item tel de \omterm{def:g10:algebra:equation}{vergelijkingen} op: die onbekende verdwijnt;
  \item los de overblijvende \omterm{def:g10:algebra:equation}{vergelijking} met één onbekende op en vul terug in
        om de andere onbekende te vinden;
  \item controleer het paar in beide oorspronkelijke \omterm{def:g10:algebra:equation}{vergelijkingen}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:lines:elimination}
Los $\begin{cases} 2x + 3y = 7 \\ 5x - 2y = 8 \end{cases}$ op met eliminatie.
Om $y$ weg te werken, vermenigvuldig je de eerste \omterm{def:g10:algebra:equation}{vergelijking} met $2$ en de
tweede met $3$:
\[
\begin{cases}
4x + 6y = 14 \\
15x - 6y = 24
\end{cases}
\]
Optellen geeft $19x = 38$, dus $x = 2$. Invullen in $2x + 3y = 7$:
$4 + 3y = 7$, dus $y = 1$. Oplossing: $(2, 1)$. Controle in de tweede
\omterm{def:g10:algebra:equation}{vergelijking}: $5 \times 2 - 2 \times 1 = 8$.
\end{example}

\begin{remark}[Hoeveel oplossingen?]
Net als twee rechten heeft een \omterm{def:g10:lines:system}{lineair stelsel} in het algemeen precies één
oplossing (niet-evenwijdige rechten), en bij uitzondering geen enkele
(verschillende evenwijdige rechten) of oneindig veel (twee keer dezelfde
rechte). Het criterium van \cref{thm:g10:vectors:det} beslist: het stelsel
heeft precies dan één oplossing wanneer $ab' - a'b \neq 0$.
\end{remark}

\begin{example}[Een stelsel opstellen]\label{ex:g10:lines:modeling}
Drie schriften en twee pennen kosten $8$; één schrift en vier pennen kosten
$6$. Zij $x$ de prijs van een schrift en $y$ die van een pen:
\[
\begin{cases}
3x + 2y = 8 \\
x + 4y = 6 .
\end{cases}
\]
Uit de tweede \omterm{def:g10:algebra:equation}{vergelijking} volgt $x = 6 - 4y$; invullen geeft
$3(6 - 4y) + 2y = 8$, dus $18 - 10y = 8$, waaruit $y = 1$ en daarna $x = 2$.
Een schrift kost $2$ en een pen kost $1$.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g10:lines:1}
Lees voor elke rechte de \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} en de $y$-afsnede af, en schets
haar:
\[
y = 2x - 3, \qquad
y = -\tfrac13 x + 1, \qquad
y = 4, \qquad
x = -2 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:lines:2}
Ligt het punt $A(2, 7)$ op de rechte $y = 3x + 1$? En $B(-1, -1)$? En het
punt $C(4, 13)$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:lines:3}
Zoek de \omterm{thm:g10:lines:reduced}{herleide vergelijking} van de rechte door:
\[
\text{(a) } A(0, 2) \text{ en } B(3, 8);
\qquad
\text{(b) } A(-1, 5) \text{ en } B(2, -4);
\qquad
\text{(c) } A(4, 1) \text{ en } B(4, 6).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:lines:4}
Welke van de rechten $y = 3x - 1$, $y = -3x + 2$, $y = 3x + 5$,
$y = \frac{6x + 2}{2}$ zijn onderling evenwijdig? Welke vallen in
werkelijkheid samen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:lines:5}
Los op met substitutie:
\[
\begin{cases}
y = 2x - 1 \\
3x + y = 9
\end{cases}
\qquad\text{en daarna}\qquad
\begin{cases}
x - y = 4 \\
2x + 3y = 3 .
\end{cases}
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:lines:6}
Los op met eliminatie:

\begin{cases}
x + y = 10 \\
x - y = 4
\end{cases}
\qquad\text{en daarna}\qquad
\begin{cases}
3x + 2y = 12 \\
2x + 5y = 8 .
\end{cases}
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:lines:7}
Bepaal, zonder ze op te lossen, hoeveel oplossingen elk stelsel heeft:

\begin{cases}
2x - y = 3 \\
-4x + 2y = -6
\end{cases}
\qquad
\begin{cases}
2x - y = 3 \\
-4x + 2y = 1
\end{cases}
\qquad
\begin{cases}
2x - y = 3 \\
x + y = 0 .
\end{cases}
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:lines:8}
Voor een schooltoneel werden tweehonderd kaartjes verkocht, sommige aan $5$
(kinderen) en sommige aan $9$ (volwassenen), voor een totaal van $1352$.
Hoeveel kaartjes van elke soort zijn er verkocht?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:lines:9}
Zij $d$ de rechte $y = 2x - 3$ en $A(1, 4)$.
\begin{enumerate}
  \item Geef de \omterm{def:g10:algebra:equation}{vergelijking} van de rechte $d'$ die evenwijdig is met $d$ en
        door $A$ gaat.
  \item Bereken het snijpunt van $d'$ met de $x$-as.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:lines:10}
Zij $A(0, 0)$, $B(6, 2)$ en $C(2, 4)$.
\begin{enumerate}
  \item Zoek de \omterm{def:g10:algebra:equation}{vergelijkingen} van de zwaartelijnen van de driehoek $ABC$ uit
        $A$ en uit $B$ (een zwaartelijn verbindt een hoekpunt met het \omterm{prop:g10:coordgeom:midpoint}{midden}
        van de overstaande zijde).
  \item Bereken hun snijpunt $G$, en ga na dat $G$ ook op de derde
        zwaartelijn ligt. (Je zou moeten vinden dat de \omterm{def:g10:coordgeom:system}{coördinaten} van $G$ de
        gemiddelden zijn van die van $A$, $B$, $C$.)
\end{enumerate}
\end{exercise}

\section{Opgave: één oplossing, geen, of oneindig veel}

\begin{problem}[{Weekendopgave --- twee rechten hebben drie mogelijke
lotsbestemmingen, een determinant beslist ertussen, en het oudste
leerboek ter wereld wist het al}]
\label{pb:g10:lines:1}
Twee lineaire \omterm{def:g10:algebra:equation}{vergelijkingen}, twee onbekenden: het paar rechten dat ze
tekenen kan elkaar één keer snijden, nooit, of één en dezelfde rechte zijn ---
en \emph{elk} \omterm{def:g10:lines:system}{lineair stelsel} erft een van die drie lotsbestemmingen. Deze
opgave brengt ze in kaart, ontmoet het getal dat beslist (een oude bekende
uit het hoofdstuk over \omterm{def:g10:vectors:vector}{vectoren}), lost de tweeduizend jaar oude fazanten en
konijnen van de Chinese \emph{Negen Hoofdstukken} op, en eindigt bij de
hoekpunten van gebieden --- de eerste stap van de optimalisatie die vandaag
elke luchtvaartmaatschappij en elke fabriek gebruikt.

\medskip
\noindent\textbf{Deel I --- De rechte, vlot.}
\begin{enumerate}
  \item Zoek de \omterm{thm:g10:lines:reduced}{herleide vergelijking} van de rechte door $(1, 2)$ en
        $(3, 8)$ (\cref{met:g10:lines:through}).
  \item Welke twee van $y = 3x - 1$, $y = 3x + 4$ en $y = -x + 7$ zijn
        evenwijdig (\cref{prop:g10:lines:parallel}), en waar ontmoeten de
        niet-evenwijdige paren elkaar? (Bereken één snijpunt.)
  \item De rechte door $(2, 1)$ en $(2, 5)$: waarom heeft ze geen \omterm{thm:g10:lines:reduced}{herleide
        vergelijking} $y = mx + p$, en wat is haar \omterm{def:g10:algebra:equation}{vergelijking}?
  \item Ligt het punt $(4, 11)$ op de rechte $y = 3x - 1$? En $(-1, -5)$?
  \item De rechte door $(0, 3)$ met \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $-2$: geef haar
        \omterm{def:g10:algebra:equation}{vergelijking}, haar twee snijpunten met de assen, en de oppervlakte van
        de driehoek die ze uit het eerste kwadrant snijdt.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Drie lotsbestemmingen.}
\begin{enumerate}[resume]
  \item Los op met substitutie (\cref{met:g10:lines:substitution}):
        $\begin{cases} y = 2x - 3 \\ 3x + y = 7 \end{cases}$
  \item Los op met eliminatie (\cref{met:g10:lines:elimination}):
        $\begin{cases} 2x + 3y = 8 \\ 5x - 3y = -1 \end{cases}$
  \item Bepaal --- en duid met rechten --- het lot van de twee stelsels
        \[
        \begin{cases} 2x - y = 3 \\ 4x - 2y = 6 \end{cases}
        \qquad\text{en}\qquad
        \begin{cases} 2x - y = 3 \\ 4x - 2y = 1 . \end{cases}
        \]
        Formuleer de volledige driedeling: snijdend, evenwijdig, samenvallend
        --- één oplossing, geen, oneindig veel.
  \item Voor het algemene stelsel $ax + by = e$, $cx + dy = f$ is $ad - bc$
        het beslissende getal --- dezelfde determinant als in de toets op
        collineariteit uit \cref{pb:g10:vectors:1}. Verklaar het toeval (welke
        twee \omterm{def:g10:vectors:vector}{vectoren} zijn \omterm{def:g10:vectors:collinear}{collineair} precies wanneer de rechten evenwijdig
        zijn of samenvallen?), en bereken $ad - bc$ voor de drie stelsels van
        de vragen 7 en 8.
  \item Voor welke waarde van $m$ heeft
        $\begin{cases} x + my = 3 \\ 2x + 6y = 7 \end{cases}$ geen oplossing?
        Los het stelsel op voor alle andere waarden van $m$\,\dots\ of leg
        minstens uit hoe je dat zou doen.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De Negen Hoofdstukken.}
\begin{enumerate}[resume]
  \item Uit de \emph{Negen Hoofdstukken over de Rekenkunst} (China, zo'n 2000
        jaar geleden): in een kooi zitten fazanten en konijnen --- samen $35$
        koppen en $94$ poten. Hoeveel van elk?
  \item Een sapkraam mengt een siroop met $60\,\%$ fruit met een drank met
        $20\,\%$ fruit om $30$ L met $35\,\%$ fruit te krijgen. Hoeveel liter
        van elk?
  \item Een getal van twee cijfers heeft cijfersom $12$; de cijfers verwisselen
        verlaagt het met $18$. Zoek het, en gebruik de cijferalgebra uit de
        weekendopgave daarover in het onderbouwvolume om het stelsel op te
        stellen.
  \item In de bakkerij kosten $3$ koffies en $2$ croissants $9.10$ euro; $2$
        koffies en $3$ croissants kosten $8.90$ euro. Los het elegant op: wat
        vertellen het \emph{optellen} en het \emph{aftrekken} van de twee
        \omterm{def:g10:algebra:equation}{vergelijkingen} je elk rechtstreeks?
  \item Een marktkraam beweert: $2$ appels $+ 1$ peer voor $5$ euro, en $4$
        appels $+ 2$ peren voor $9$ euro. Los op --- of liever: leg uit wat
        het lot van dit stelsel over de rekenkunde van de kraam onthult.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De hoekpunten beslissen.}
\begin{enumerate}[resume]
  \item De rechte $y = 3x - 1$ verdeelt het vlak in twee. Aan welke kant ligt
        de oorsprong? Beschrijf de verzameling $y > 3x - 1$ en hoe je een
        punt eraan toetst.
  \item Teken het gebied bepaald door de vier voorwaarden
        \[
        x \geq 0, \qquad y \geq 0, \qquad x + y \leq 4,
        \qquad y \leq 2x + 1,
        \]
        en bereken zijn vier hoekpunten.
  \item De winst van een atelier is $P = 3x + 2y$ op het gebied van vraag 17.
        Bereken $P$ in de vier hoekpunten. Neem aan dat het \omterm{def:g10:functions:extrema}{maximum} van zo'n
        lineaire grootheid op zo'n gebied altijd in een hoekpunt bereikt wordt
        (stel je de rechten $3x + 2y = $ constant voor die over het gebied
        strijken), en geef het optimale productieplan.
  \item De twee taxi's uit de weekendopgave over de twee thermometers in het
        onderbouwvolume, nu als stelsel: schrijf
        $\begin{cases} y = 1.5x + 3 \\ y = 2x \end{cases}$ op, los het op en
        duid de oplossing.
  \item Slotstuk, het woordenboek: één \omterm{def:g10:algebra:equation}{vergelijking} $\leftrightarrow$ één
        rechte; één stelsel $\leftrightarrow$ twee rechten met drie
        lotsbestemmingen, beslist door $ad - bc$; veel ongelijkheden
        $\leftrightarrow$ een gebied waarvan de hoekpunten het optimum dragen.
        Zeg het in drie zinnen --- je hebt zojuist de inkomhal bezocht van de
        \emph{lineaire algebra} en het \emph{lineair programmeren}, allebei in
        de universitaire volumes volledig opgebouwd.
\end{enumerate}
