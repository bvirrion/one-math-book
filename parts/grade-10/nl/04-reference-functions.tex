\chapter{Referentiefuncties}\label{ch:g10:reffunc}

Een handvol eenvoudige \omterm{def:g10:functions:function}{functies} --- de affiene, het kwadraat, het omgekeerde,
de vierkantswortel, de derde macht --- duikt overal op, alleen of gecombineerd.
Wie hun \omterm{def:g10:functions:graph}{grafieken} en hun verloop uit het hoofd kent, herleidt veel vraagstukken
tot een snelle schets. Dit hoofdstuk bestudeert ze een voor een en bewijst hun
verloop met de vergelijkingstechniek van \cref{ch:g10:functions}.

\section{Affiene functies}

\begin{definition}[Affiene functie]\label{def:g10:reffunc:affine}
Een \emph{affiene functie}\index{affiene functie} is een \omterm{def:g10:functions:function}{functie} van de vorm
\[
f(x) = mx + p,
\]
waarbij $m$ en $p$ vaste \omterm{def:g10:numbers:sets}{reële getallen} zijn. Haar \omterm{def:g10:functions:graph}{grafiek} is een rechte:
$m$ is de \emph{richtingscoëfficiënt}\index{richtingscoëfficiënt} en $p$ de
\emph{$y$-afsnede}\index{y-afsnede@$y$-afsnede} (de rechte snijdt de
verticale as in $(0, p)$). Is $p = 0$, dan heet de \omterm{def:g10:functions:function}{functie} \emph{lineair}:
$f(x) = mx$ drukt evenredigheid uit.
\end{definition}

\begin{proposition}[Richtingscoëfficiënt en verloop]\label{prop:g10:reffunc:affine}
Zij $f(x) = mx + p$.
\begin{enumerate}
  \item Voor elke twee verschillende invoeren $u \neq v$ geldt
        $m = \dfrac{f(v) - f(u)}{v - u}$ (de \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} is de
        verandering van de uitvoer per eenheid verandering van de invoer).
  \item Is $m > 0$, dan is $f$ strikt \omterm{def:g10:functions:variations}{stijgend} op $\R$; is $m < 0$, dan
        strikt \omterm{def:g10:functions:variations}{dalend}; is $m = 0$, dan constant.
\end{enumerate}
\end{proposition}

\begin{proof}
1. Bereken $f(v) - f(u) = (mv + p) - (mu + p) = m(v - u)$, en deel daarna
door $v - u \neq 0$.

2. Is $u < v$, dan heeft $f(v) - f(u) = m(v-u)$ het teken van $m$, want
$v - u > 0$: een positieve $m$ geeft $f(u) < f(v)$ (\omterm{def:g10:functions:variations}{stijgend}), een negatieve
$m$ geeft $f(u) > f(v)$ (\omterm{def:g10:functions:variations}{dalend}).
\end{proof}

\begin{example}\label{ex:g10:reffunc:affine}
Zoek de \omterm{def:g10:reffunc:affine}{affiene functie} waarvan de \omterm{def:g10:functions:graph}{grafiek} door $A(1, 3)$ en $B(4, 9)$ gaat.
De \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} is
\[
m = \frac{9 - 3}{4 - 1} = \frac{6}{3} = 2 .
\]
Dan is $f(x) = 2x + p$, en $f(1) = 3$ geeft $2 + p = 3$, dus $p = 1$:
$f(x) = 2x + 1$. Controle met $B$: $f(4) = 9$.
\end{example}

\section{De kwadraatfunctie}

\begin{proposition}[Kwadraatfunctie]\label{prop:g10:reffunc:square}
De \omterm{def:g10:functions:function}{functie} $f(x) = x^2$, gedefinieerd op $\R$:
\begin{enumerate}
  \item is strikt \omterm{def:g10:functions:variations}{dalend} op $\intoc{-\infty}{0}$ en strikt \omterm{def:g10:functions:variations}{stijgend} op
        $\intco{0}{+\infty}$, met \omterm{def:g10:functions:extrema}{minimum} $0$ in $x = 0$;
  \item voldoet aan $f(-x) = f(x)$ voor alle $x$: haar \omterm{def:g10:functions:graph}{grafiek}, een
        \emph{parabool}\index{parabool}, is symmetrisch om de verticale as.
\end{enumerate}
\end{proposition}

\begin{proof}
1. Zij $0 \leq u < v$. Dan is
\[
f(v) - f(u) = v^2 - u^2 = (v - u)(v + u) > 0,
\]
want $v - u > 0$ en $v + u > 0$: $f$ is strikt \omterm{def:g10:functions:variations}{stijgend} op
$\intco{0}{+\infty}$. Is $u < v \leq 0$, dan is $v - u > 0$ maar
$v + u < 0$, dus $f(v) - f(u) < 0$: strikt \omterm{def:g10:functions:variations}{dalend}. Ten slotte is
$x^2 \geq 0 = f(0)$ voor alle $x$.

2. $(-x)^2 = x^2$; dus liggen de punten $(x, x^2)$ en $(-x, x^2)$, elkaars
spiegelbeeld om de verticale as, allebei op de \omterm{def:g10:functions:graph}{grafiek}.
\end{proof}

\begin{remark}[Kwadraten en orde]
Doordat de kwadraatfunctie aan de negatieve kant daalt, \emph{keert}
kwadrateren de orde van negatieve getallen om: $-3 < -2$ maar $9 > 4$.
Kwadrateer beide leden van een ongelijkheid dus nooit zonder de tekens na te
gaan.
\end{remark}

\section{De omgekeerde functie}

\begin{proposition}[Omgekeerde functie]\label{prop:g10:reffunc:inverse}
De \omterm{def:g10:functions:function}{functie} $f(x) = \dfrac1x$, gedefinieerd voor $x \neq 0$:
\begin{enumerate}
  \item is strikt \omterm{def:g10:functions:variations}{dalend} op $\intoo{-\infty}{0}$ en strikt \omterm{def:g10:functions:variations}{dalend} op
        $\intoo{0}{+\infty}$;
  \item voldoet aan $f(-x) = -f(x)$: haar \omterm{def:g10:functions:graph}{grafiek}, een
        \emph{hyperbool}\index{hyperbool}, is symmetrisch om de oorsprong.
\end{enumerate}
\end{proposition}

\begin{proof}
1. Zij $0 < u < v$. Dan is
\[
f(v) - f(u) = \frac1v - \frac1u = \frac{u - v}{uv} < 0,
\]
want $u - v < 0$ en $uv > 0$: strikt \omterm{def:g10:functions:variations}{dalend} op $\intoo{0}{+\infty}$. Op
$\intoo{-\infty}{0}$ heeft hetzelfde quotiënt opnieuw $u - v < 0$ en
$uv > 0$ (product van twee negatieve getallen), dus daalt $f$ ook daar.

2. $\frac{1}{-x} = -\frac1x$.
\end{proof}

\begin{remark}
Opgelet: $\frac1x$ is \emph{niet} \omterm{def:g10:functions:variations}{dalend} op haar hele \omterm{def:g10:functions:function}{domein}. Van $u = -1$
naar $v = 1$ springt de waarde van $-1$ omhoog naar $1$. De twee takken
moeten apart bestudeerd worden.
\end{remark}

\section{Vierkantswortel en derde macht}

\begin{proposition}[Vierkantswortelfunctie]\label{prop:g10:reffunc:sqrt}
De \omterm{def:g10:functions:function}{functie} $f(x) = \sqrt{x}$, gedefinieerd op $\intco{0}{+\infty}$, is
strikt \omterm{def:g10:functions:variations}{stijgend}.
\end{proposition}

\begin{proof}
Zij $0 \leq u < v$. Vermenigvuldig en deel door de \emph{toegevoegde}
uitdrukking:
\[
\sqrt v - \sqrt u
= \frac{(\sqrt v - \sqrt u)(\sqrt v + \sqrt u)}{\sqrt v + \sqrt u}
= \frac{v - u}{\sqrt v + \sqrt u} > 0,
\]
want $v - u > 0$ en $\sqrt v + \sqrt u > 0$ (merk op dat $v > 0$, dus
$\sqrt v > 0$).
\end{proof}

\begin{proposition}[Derdemachtsfunctie]\label{prop:g10:reffunc:cube}
De \omterm{def:g10:functions:function}{functie} $f(x) = x^3$, gedefinieerd op $\R$, is strikt \omterm{def:g10:functions:variations}{stijgend}, en haar
\omterm{def:g10:functions:graph}{grafiek} is symmetrisch om de oorsprong.
\end{proposition}
\admitted

\begin{proposition}[$x$, $x^2$ en $\sqrt x$ vergeleken]\label{prop:g10:reffunc:compare}
Voor $0 < x < 1$:\ \ $x^2 < x < \sqrt x$.\quad
Voor $x > 1$:\ \ $\sqrt x < x < x^2$.\quad
In $x = 0$ en $x = 1$ zijn de drie waarden gelijk.
\end{proposition}

\begin{proof}
Stel $0 < x < 1$. Vermenigvuldigen van $x < 1$ met $x > 0$ geeft $x^2 < x$.
Vervolgens $x < \sqrt x$: omdat beide leden positief zijn en kwadrateren
\omterm{def:g10:functions:variations}{stijgend} is op de positieve getallen, is die ongelijkheid gelijkwaardig met
$x^2 < x$, wat we net bewezen. Voor $x > 1$ geeft $x > 1$ vermenigvuldigd
met $x$ dat $x^2 > x$, en $\sqrt x < x$ volgt met hetzelfde
kwadrateerargument.
\end{proof}

\begin{method}[Beelden vergelijken]\label{met:g10:reffunc:compare}
Om $f(a)$ en $f(b)$ te vergelijken voor een referentiefunctie $f$:
\begin{enumerate}
  \item plaats $a$ en $b$ op een \omterm{def:g10:numbers:interval}{interval} waar het verloop van $f$ bekend is;
  \item stijgt $f$ daar, dan staan de \omterm{def:g10:functions:function}{beelden} in dezelfde volgorde als de
        invoeren; daalt $f$, dan is de volgorde omgekeerd;
  \item liggen $a$ en $b$ niet in hetzelfde \omterm{def:g10:numbers:interval}{interval} van monotonie, regel dan
        eerst de tekens (bijvoorbeeld bij het kwadraat: een negatieve en een
        positieve invoer).
\end{enumerate}
\end{method}

\begin{example}
Vergelijk $\dfrac{1}{2.3}$ en $\dfrac{1}{2.4}$: beide invoeren liggen in
$\intoo{0}{+\infty}$, waar de omgekeerde \omterm{def:g10:functions:function}{functie} daalt, en $2.3 < 2.4$, dus
$\dfrac{1}{2.3} > \dfrac{1}{2.4}$. Vergelijk $(-1.2)^2$ en $(-1.5)^2$: op
$\intoc{-\infty}{0}$ daalt het kwadraat, en $-1.5 < -1.2$, dus
$(-1.5)^2 > (-1.2)^2$ --- en inderdaad is $2.25 > 1.44$.
\end{example}

\qquad

\begin{exercise}[$\star\star\star$]\label{exo:g10:reffunc:10}

\end{exercise}

\section{Opgave: de natuurwetten spreken in referentiefuncties}

\begin{problem}[{Weekendopgave --- remafstanden groeien als $v^2$, hefbomen
gehoorzamen aan $1/d$, planeten volgen $a^{3/2}$: natuurkunde lezen met de
referentiefuncties}]
\label{pb:g10:reffunc:1}
Sla een natuurkundeboek open en op elke bladzijde duiken dezelfde paar
\omterm{def:g10:functions:graph}{grafieken} op: de parabool, de hyperbool, de wortelkromme. De natuur schrijft
haar wetten met de referentiefuncties van dit hoofdstuk --- en hun vormen
kennen (\cref{prop:g10:reffunc:square}, \cref{prop:g10:reffunc:inverse},
\cref{prop:g10:reffunc:sqrt}, \cref{prop:g10:reffunc:cube}) volstaat om een
auto af te remmen, een hefboom in evenwicht te brengen en de planeten te
klokken. Het slotstuk is de derde wet van Kepler, rechtstreeks van de
gegevens van het zonnestelsel afgelezen.

\medskip
\noindent\textbf{Deel I --- De kwadratische wet van het remmen.}
Een vuistregel voor de remafstand van een auto op een droge weg:
$d = \left(\frac{v}{10}\right)^{2}$ meter, bij een snelheid $v$ km/u.
\begin{enumerate}
  \item Bereken de remafstanden bij $30$, $50$, $90$ en $130$ km/u.
  \item Met hoeveel wordt de remafstand vermenigvuldigd als je de snelheid
        verdubbelt? Verklaar het uit het schaalgedrag van de
        kwadraatfunctie, en ga het na op $50 \to 100$ km/u.
  \item Forensisch onderzoek: de remsporen meten $50$ m. Van welke snelheid
        beschuldigt de formule de bestuurder (op de km/u nauwkeurig)? Welke
        referentiefunctie gaf het antwoord --- en op welk \omterm{def:g10:functions:function}{domein} is de
        aflezing ondubbelzinnig (\cref{prop:g10:reffunc:sqrt})?
  \item Vergelijk de extra afstand die \emph{één} km/u meer kost, bij lage en
        bij hoge snelheid: bereken $d(31) - d(30)$ en $d(91) - d(90)$. Welke
        eigenschap van de parabool (\cref{prop:g10:reffunc:square})
        illustreren de twee antwoorden?
  \item Echt stoppen vraagt ook reactietijd --- ongeveer één seconde, waarin
        de auto $0.28v$ meter aflegt:
        $D(v) = 0.28v + \left(\frac{v}{10}\right)^2$. Bereken $D(50)$ en
        $D(130)$. Welke van de twee termen --- de affiene of de kwadratische
        --- overheerst bij stadssnelheid, en welke op de snelweg?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De hyperbolen van het dagelijks leven.}
\begin{enumerate}[resume]
  \item Een hefboom is in evenwicht wanneer kracht $\times$ afstand aan beide
        kanten gelijk is. Een kind van $60$ kg zit op $2$ m van het
        draaipunt; de kracht die aan de andere kant op afstand $d$ evenwicht
        maakt, is $F(d) = \frac{120}{d}$ (in kilogramequivalenten). Bereken
        $F(0.5)$, $F(1)$, $F(3)$.
  \item Wat zegt het verloop van de omgekeerde \omterm{def:g10:functions:function}{functie}
        (\cref{prop:g10:reffunc:inverse}) over hefbomen --- en waarom schuilt
        de grootspraak van Archimedes (``geef mij een steunpunt en ik verplaats
        de Aarde'') in de staart van de hyperbool? Wat verbiedt $d = 0$?
  \item Geluid en licht doven uit met het \emph{kwadraat} van de afstand: op
        $1$ m van een lamp is de intensiteit $100$ eenheden; geef ze op $2$,
        $3$ en $10$ m. Verklaar de exponent $2$ met een bol: over welke
        oppervlakte heeft het licht zich op afstand $d$ verspreid (de
        weekendopgave over de grafsteen van Archimedes in het
        onderbouwvolume)?
  \item Een bus voor de schoolreis kost $600$ euro, gelijk verdeeld over $n$
        deelnemers. Zet de kostprijs per hoofd in een tabel voor
        $n = 10, 20, 30$, zoek hoeveel deelnemers ze op $25$ euro of minder
        brengen, en zeg wat de vorm van de hyperbool belooft --- en weigert
        --- naarmate $n$ groeit.
  \item $x$ affiches drukken kost $2x + 3$ euro ($3$ euro opstartkosten).
        Toon aan dat de \emph{gemiddelde} kostprijs per affiche
        $a(x) = 2 + \frac3x$ is, beschrijf het verloop ervan, en duid de
        horizontale asymptoot economisch. (Vergelijk met de algebra van
        \cref{exo:g10:reffunc:10}.)
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De harmonie van Kepler.}
Noteer voor elke planeet $a$ voor haar gemiddelde afstand tot de Zon (in
astronomische eenheden, Aarde $= 1$) en $T$ voor haar omlooptijd (in jaren).
Gegevens: Mars $a = 1.52$, $T = 1.88$; Jupiter $a = 5.20$, $T = 11.86$;
Saturnus $a = 9.54$, $T = 29.4$.
\begin{enumerate}[resume]
  \item Bereken $a^3$ en $T^2$ voor de Aarde, Mars en Jupiter. Wat merkte
        Kepler in 1618 op?
  \item Formuleer de wet als een \omterm{def:g10:functions:function}{functie}: $T = \sqrt{a^3} = a\sqrt a$. Toets
        ze op Saturnus.
  \item Twee voorspellingen: een planetoïde draait op $a = 4$ AE --- zoek
        haar omlooptijd (de getallen komen mooi uit); een komeet keert elke
        $27$ jaar terug --- zoek haar gemiddelde afstand (zoek een volkomen
        derde macht).
  \item Welke drie referentiefuncties vlecht de wet samen? En haar
        schaalregel: wat gebeurt er met $T$ als $a$ met $4$ vermenigvuldigd
        wordt? (Ga het na met de planetoïde tegenover de Aarde.)
  \item Kepler las deze wet af in de gegevens van Tycho Brahe, zeventig jaar
        vóór de gravitatie van Newton haar verklaarde. In één zin: wat zegt
        die episode over de kracht van het herkennen van een referentiefunctie
        in een tabel met getallen?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De schildpad en de haas.}
\begin{enumerate}[resume]
  \item Rangschik $\sqrt x$, $x$ en $x^2$ op $\intoo{0}{1}$ en op
        $\intoo{1}{+\infty}$ (\cref{prop:g10:reffunc:compare}), en ga beide
        rangschikkingen na in $x = 0.25$ en $x = 4$.
  \item Los $\sqrt x > x$ volledig op (kwadrateer voor $x \geq 0$ met de
        nodige zorg en eindig met een \omterm{def:g10:algebra:signtable}{tekentabel}).
  \item Neem de derde macht erbij: rangschik $x$, $x^2$, $x^3$ in $x = 0.5$
        en in $x = 2$, en zoek alle punten waar de kwadraatfunctie en de
        derdemachtsfunctie elkaar snijden.
  \item Twee spaarplannen keren na $t$ jaar $100\sqrt t$ euro uit (plan A) of
        $10 t^2$ euro (plan B). Toon aan dat B A voorbijsteekt wanneer
        $t^3 = 100$, en geef het omslagpunt op de maand nauwkeurig. Welke
        moraal over wortels tegenover machten op lange termijn?
  \item Slotstuk: zeg voor elke wet uit deze opgave --- het remmen ($v^2$),
        de hefboom ($1/d$), de lamp ($1/d^2$), Kepler ($a^{3/2}$), plan A
        ($\sqrt t$) --- in één bijzin welk kenmerk van haar
        referentiegrafiek de fysische betekenis draagt (steeds steilere
        stijging, verticale asymptoot, uitdijende bol, schaalexponent,
        afvlakkende groei). Besluit: de referentiefuncties zijn het alfabet;
        de natuur schrijft ermee.
\end{enumerate}
\end{problem}
