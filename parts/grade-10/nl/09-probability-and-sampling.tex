\chapter{Kansrekening en steekproeven}\label{ch:g10:proba}

De kansrekening kent getallen toe aan het toeval. Vertrekkend van experimenten
met even waarschijnlijke uitkomsten --- dobbelstenen, munten, kaarten --- bouwt
dit hoofdstuk de basisregels om kansen van gebeurtenissen te berekenen, voert
het boomdiagrammen in voor experimenten in twee stappen, en eindigt het bij het
verband tussen kansen en waargenomen \omterm{def:g10:stats:series}{frequenties}. De theorie loopt door in
\cref{ch:g11:prob}.

\section{Toevalsexperimenten en gebeurtenissen}

\begin{definition}[Uitkomsten en gebeurtenissen]\label{def:g10:proba:events}
Een \emph{toevalsexperiment}\index{toevalsexperiment} is een experiment
waarvan het resultaat niet te voorspellen is (met een dobbelsteen gooien). De
verzameling van alle mogelijke resultaten is de
\emph{uitkomstenverzameling}\index{uitkomstenverzameling}
$\Omega = \{\omega_1, \dots, \omega_n\}$; haar elementen zijn de
\emph{uitkomsten}. Een \emph{gebeurtenis}\index{gebeurtenis} is een
verzameling uitkomsten; een gebeurtenis \emph{doet zich voor} wanneer het
resultaat van het experiment ertoe behoort.
\end{definition}

\begin{definition}[Kansverdeling]\label{def:g10:proba:distribution}
Door aan elke uitkomst $\omega_i$ een getal $p_i \geq 0$ toe te kennen met
$p_1 + \dots + p_n = 1$, leg je een \emph{kans}\index{kans} vast: de kans
$\P(A)$ van een \omterm{def:g10:proba:events}{gebeurtenis} $A$ is de som van de $p_i$ van de uitkomsten in
$A$. In het bijzonder is $\P(\Omega) = 1$, en heeft de onmogelijke
\omterm{def:g10:proba:events}{gebeurtenis} $\varnothing$ kans $0$.
\end{definition}

\begin{proposition}[Even waarschijnlijke uitkomsten]\label{prop:g10:proba:uniform}
Zijn de $n$ uitkomsten even waarschijnlijk (elke $p_i = \frac1n$), dan geldt
voor elke \omterm{def:g10:proba:events}{gebeurtenis} $A$:
\[
\P(A) = \frac{\text{aantal uitkomsten in } A}{n}
= \frac{\text{gunstige gevallen}}{\text{mogelijke gevallen}} .
\]
\end{proposition}

\begin{proof}
$A$ bevat een zeker aantal $k$ uitkomsten, elk met \omterm{def:g10:proba:distribution}{kans} $\frac1n$, dus is
$\P(A) = \frac kn$.
\end{proof}

\begin{example}\label{ex:g10:proba:die}
Gooi met een eerlijke dobbelsteen: $\Omega = \{1, 2, 3, 4, 5, 6\}$, waarbij
elke uitkomst \omterm{def:g10:proba:distribution}{kans} $\frac16$ heeft. De \omterm{def:g10:proba:events}{gebeurtenis} $A$: ``het resultaat is
even'' is $A = \{2, 4, 6\}$, dus $\P(A) = \frac36 = \frac12$. De \omterm{def:g10:proba:events}{gebeurtenis}
$B$: ``het resultaat is minstens $5$'' is $B = \{5, 6\}$, dus
$\P(B) = \frac26 = \frac13$.
\end{example}

\section{Gebeurtenissen combineren}

\begin{definition}[Doorsnede, vereniging, complement]\label{def:g10:proba:operations}
Zij $A$ en $B$ gebeurtenissen.
\begin{itemize}
  \item $A \cap B$ (``$A$ en $B$'') doet zich voor wanneer beide zich
        voordoen;
  \item $A \cup B$ (``$A$ of $B$'') doet zich voor wanneer minstens één zich
        voordoet;
  \item het \emph{complement}\index{complement} $\bar A$ (``niet $A$'') doet
        zich precies voor wanneer $A$ zich niet voordoet;
  \item $A$ en $B$ heten \emph{onverenigbaar}\index{onverenigbaar} wanneer ze
        zich niet samen kunnen voordoen: $A \cap B = \varnothing$.
\end{itemize}
\end{definition}

\begin{theorem}[Somregels]\label{thm:g10:proba:addition}
Voor alle gebeurtenissen $A$ en $B$ geldt
\[
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B),
\qquad
\P(\bar A) = 1 - \P(A) .
\]
Voor onverenigbare gebeurtenissen vereenvoudigt de eerste regel tot
$\P(A \cup B) = \P(A) + \P(B)$.
\end{theorem}

\begin{proof}
In $\P(A) + \P(B)$ wordt de \omterm{def:g10:proba:distribution}{kans} van elke uitkomst van $A \cup B$ één keer
opgeteld, \emph{behalve} die van $A \cap B$, die twee keer meetellen --- één
keer in $A$ en één keer in $B$. $\P(A \cap B)$ aftrekken herstelt die dubbele
telling. Voor het \omterm{def:g10:proba:operations}{complement}: $A$ en $\bar A$ zijn \omterm{def:g10:proba:operations}{onverenigbaar} en
$A \cup \bar A = \Omega$, dus $\P(A) + \P(\bar A) = \P(\Omega) = 1$.
\end{proof}

\begin{example}\label{ex:g10:proba:cards}
Trek één kaart uit een standaardspel van $52$ kaarten. Zij $A$: ``de kaart is
een harten'' ($13$ kaarten) en $B$: ``de kaart is een heer'' ($4$ kaarten).
Dan is $A \cap B$ ``hartenheer'' ($1$ kaart), en
\[
\P(A \cup B) = \frac{13}{52} + \frac{4}{52} - \frac{1}{52} = \frac{16}{52}
= \frac{4}{13}.
\]
De complementregel is vaak de snelste weg: de \omterm{def:g10:proba:distribution}{kans} dat de kaart \emph{geen}
harten is, is $1 - \frac{13}{52} = \frac34$.
\end{example}

\section{Boomdiagrammen}

\begin{method}[Experimenten in twee stappen]\label{met:g10:proba:tree}
Voor een experiment dat in twee stappen verloopt:
\begin{enumerate}
  \item teken een \emph{boomdiagram}\index{boomdiagram}: één tak per mogelijk
        resultaat van de eerste stap, en vanuit elk daarvan één tak per
        resultaat van de tweede stap, met op elke tak de bijhorende \omterm{def:g10:proba:distribution}{kans} (de
        takken die uit één knoop vertrekken tellen op tot $1$);
  \item de \omterm{def:g10:proba:distribution}{kans} van een \emph{pad} (een blad van de boom) is het product van
        de kansen langs zijn takken;
  \item de \omterm{def:g10:proba:distribution}{kans} van een \omterm{def:g10:proba:events}{gebeurtenis} is de som van de kansen van de paden die
        haar verwezenlijken.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:proba:tree}
Een urne bevat $3$ rode en $2$ blauwe ballen. Trek één bal, leg hem terug en
trek opnieuw. Elke trekking geeft rood (R) met \omterm{def:g10:proba:distribution}{kans} $\frac35$ en blauw (B)
met \omterm{def:g10:proba:distribution}{kans} $\frac25$.

De \omterm{def:g10:proba:distribution}{kans} om twee ballen van dezelfde kleur te krijgen is
$\P(\text{RR}) + \P(\text{BB}) = \frac{9}{25} + \frac{4}{25} =
\frac{13}{25}$; de \omterm{def:g10:proba:distribution}{kans} op minstens één blauwe is
$1 - \P(\text{RR}) = 1 - \frac{9}{25} = \frac{16}{25}$ (complementregel).
\end{example}

\begin{remark}
Zonder teruglegging hangen de kansen van de tweede trekking van de eerste af:
in de urne hierboven blijven er na een rode bal nog $2$ rode en $2$ blauwe
over, zodat de tweede tak $\frac24$ en $\frac24$ draagt. De boommethode werkt
onveranderd; alleen de getallen op de takken van het tweede niveau
veranderen.
\end{remark}

\section{Frequenties en steekproeven}

Wordt een \omterm{def:g10:proba:events}{toevalsexperiment} vele keren herhaald, dan nadert de waargenomen
\omterm{def:g10:stats:series}{frequentie} van een \omterm{def:g10:proba:events}{gebeurtenis} haar \omterm{def:g10:proba:distribution}{kans} --- en daarom is de kansrekening
bruikbaar om de werkelijkheid te beschrijven.

\begin{definition}[Steekproef]\label{def:g10:proba:sample}
Een \emph{steekproef}\index{steekproef} van omvang $n$ is het resultaat van
hetzelfde experiment $n$ keer onafhankelijk te herhalen. De \emph{waargenomen
\omterm{def:g10:stats:series}{frequentie}} van een \omterm{def:g10:proba:events}{gebeurtenis} in de steekproef is het aantal keren dat ze
zich voordoet, gedeeld door $n$.
\end{definition}

\begin{example}\label{ex:g10:proba:sampling}
Een eerlijke munt geeft kop met \omterm{def:g10:proba:distribution}{kans} $0.5$. Hem $100$ keer opgooien geeft
zelden precies $50$ keer kop: steekproeven \emph{schommelen}. Typische
reeksen van $100$ worpen geven \omterm{def:g10:stats:series}{frequenties} als $0.46$, $0.53$, $0.49$ ---
dicht bij $0.5$, maar niet gelijk. Grotere steekproeven schommelen minder:
bij $10\,000$ worpen liggen de waargenomen \omterm{def:g10:stats:series}{frequenties} meestal binnen
ongeveer $0.01$ van $0.5$.
\end{example}

\begin{remark}[Schommelingsinterval]
Een handige vuistregel, in \cref{ch:g11:binom} verantwoord: voor een
\omterm{def:g10:proba:sample}{steekproef} van omvang $n$ van een \omterm{def:g10:proba:events}{gebeurtenis} met \omterm{def:g10:proba:distribution}{kans} $p$ (met $n$ redelijk
groot) ligt de waargenomen \omterm{def:g10:stats:series}{frequentie} in het \omterm{def:g10:numbers:interval}{interval}
\[
\intcc{p - \frac{1}{\sqrt n}}{\,p + \frac{1}{\sqrt n}}
\]
in ruwweg $95\%$ van de steekproeven. Valt een waargenomen \omterm{def:g10:stats:series}{frequentie} ver
buiten dat \omterm{def:g10:numbers:interval}{interval}, twijfel dan aan de waarde van $p$: dat is het
vertrekpunt van het statistisch toetsen.
\end{remark}

\begin{example}\label{ex:g10:proba:test}
Een fabriek beweert dat maar $10\%$ van haar onderdelen defect is
($p = 0.1$). In een \omterm{def:g10:proba:sample}{steekproef} van $n = 400$ onderdelen zijn er $64$ defect:
waargenomen \omterm{def:g10:stats:series}{frequentie} $\frac{64}{400} = 0.16$. Het schommelingsinterval is
$\intcc{0.1 - \frac{1}{20}}{0.1 + \frac{1}{20}} = \intcc{0.05}{0.15}$, en
$0.16$ valt erbuiten: de \omterm{def:g10:proba:sample}{steekproef} werpt ernstige twijfel op de bewering.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g10:proba:1}
Er wordt met een eerlijke dobbelsteen gegooid. Bereken de \omterm{def:g10:proba:distribution}{kans} van elke
\omterm{def:g10:proba:events}{gebeurtenis}: ``een $6$ gooien''; ``een oneven getal gooien''; ``hoogstens $4$
gooien''; ``een $7$ gooien''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:2}
Een zak bevat $5$ groene, $3$ gele en $2$ zwarte knikkers; er wordt er
willekeurig één getrokken. Bereken de \omterm{def:g10:proba:distribution}{kans} dat hij groen is; dat hij niet
zwart is; dat hij groen of geel is.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:3}
In een klas van $30$ leerlingen volgen er $18$ Spaans, $10$ Duits en $4$
allebei. Er wordt willekeurig een leerling gekozen. Bereken met de somregel
de \omterm{def:g10:proba:distribution}{kans} dat die leerling minstens één van de twee talen volgt, en daarna de
\omterm{def:g10:proba:distribution}{kans} dat hij geen van beide volgt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:4}
Een eerlijke munt wordt twee keer opgegooid. Teken het \omterm{met:g10:proba:tree}{boomdiagram} van het
experiment en bereken de \omterm{def:g10:proba:distribution}{kans} op twee keer kop; op precies één keer kop; op
minstens één keer kop.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:5}
Een urne bevat $4$ rode en $6$ witte ballen. Er worden na elkaar twee ballen
getrokken \emph{zonder} teruglegging. Teken het \omterm{met:g10:proba:tree}{boomdiagram} met de juiste
kansen op het tweede niveau, en bereken de \omterm{def:g10:proba:distribution}{kans} om twee rode ballen te
trekken, en daarna om twee ballen van verschillende kleur te trekken.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:6}
Er wordt met twee eerlijke dobbelstenen gegooid en de som wordt genoteerd.
\begin{enumerate}
  \item Leg uit waarom je als \omterm{def:g10:proba:events}{uitkomstenverzameling} de $36$ even
        waarschijnlijke paren $(1,1), (1,2), \dots, (6,6)$ mag nemen.
  \item Bereken de \omterm{def:g10:proba:distribution}{kans} dat de som $7$ is; dat ze $12$ is; dat ze hoogstens
        $4$ is.
  \item Welke som is het waarschijnlijkst?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:7}
Een meerkeuzevraag biedt $4$ antwoorden, waarvan er één juist is. Een
leerling beantwoordt $2$ zulke onafhankelijke vragen volstrekt willekeurig.
Bereken de \omterm{def:g10:proba:distribution}{kans} op twee juiste antwoorden, op precies één juist antwoord, en
op geen enkel juist antwoord. Ga na dat de drie kansen optellen tot $1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:8}
Een politicus beweert $50\%$ steun te hebben. In een peiling bij $n = 900$
willekeurig gekozen kiezers steunen er $396$ hem.
\begin{enumerate}
  \item Bereken de waargenomen \omterm{def:g10:stats:series}{frequentie} en het schommelingsinterval rond
        $p = 0.5$ voor $n = 900$.
  \item Werpt de peiling twijfel op de bewering?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:proba:9}
Meespelen aan een spel kost $2$: je gooit met een eerlijke dobbelsteen en
wint $6$ bij een $6$, $3$ bij een $5$, en niets in de overige gevallen.
\begin{enumerate}
  \item Bereken de \omterm{def:g10:proba:distribution}{kans} op elk gewonnen bedrag ($6$, $3$, $0$).
  \item Hoeveel keer verwacht je elke uitkomst ongeveer over $600$ spellen?
        Bereken de totale verwachte winst en het totale verwachte verlies:
        loont het spel?
\end{enumerate}
\end{exercise}

\section{Opgave: de drie deuren die duizend wiskundigen om de tuin leidden}

\begin{problem}[{Weekendopgave --- de paradox van Monty Hall en haar neven:
hoe informatie de kans verandert, en hoe steekproeven beweringen van
wishful thinking scheiden}]
\label{pb:g10:proba:1}
In 1990 publiceerde columniste Marilyn vos Savant een raadsel over een
quizmaster, drie deuren, een auto en twee geiten --- en beantwoordde het
correct. Tienduizend lezers, onder wie honderden doctors in de wiskunde,
schreven haar dat ze zich vergiste. Deze opgave wapent je met boomdiagrammen
(\cref{met:g10:proba:tree}), \omterm{def:g10:proba:operations}{complementen} en de somregels
(\cref{thm:g10:proba:addition}), laat je de ruzie zelf beslechten, en eindigt
met het gereedschap dat elk zulk geschil in de praktijk beslecht: de
schommelende \omterm{def:g10:proba:sample}{steekproef}.

\medskip
\noindent\textbf{Deel I --- De regels, opgewarmd.}
\begin{enumerate}
  \item Eén kaart uit een standaardspel van $52$ kaarten: bereken
        $P(\text{heer of harten})$ met de somregel --- en zeg waarom
        $\frac{4}{52} + \frac{13}{52}$ alleen fout is.
  \item Drie eerlijke muntworpen: bereken $P(\text{minstens één keer kop})$
        via het \omterm{def:g10:proba:operations}{complement}.
  \item Een zak bevat $3$ rode en $2$ blauwe knikkers; er worden er twee
        getrokken zonder teruglegging. Bereken met een \omterm{met:g10:proba:tree}{boomdiagram}
        $P(\text{twee rode})$ en $P(\text{één van elke kleur})$.
  \item Vervolledig de verdeling van vraag 3 met $P(\text{twee blauwe})$ en
        ga na dat de drie kansen optellen tot $1$.
  \item Iemand meldt voor twee \emph{onverenigbare} gebeurtenissen
        $P(A) = 0.6$ en $P(B) = 0.5$. Ontmasker hem met één regel rekenwerk.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De drie deuren.}
Het spel: achter één van drie deuren staat een auto, achter de twee andere
staan geiten. Jij kiest een deur (zeg deur 1). De quizmaster --- die
\emph{weet} waar de auto staat --- opent een van de twee andere deuren, altijd
met een geit erachter, en biedt je de \omterm{def:g10:proba:distribution}{kans} om naar de overblijvende gesloten
deur te wisselen.
\begin{enumerate}[resume]
  \item Voordat je iets berekent: helpt wisselen, schaadt het, of maakt het
        niet uit? Schrijf je onderbuikgevoel op --- eerlijk.
  \item Nu het \omterm{met:g10:proba:tree}{boomdiagram}, per geval naargelang de werkelijke plaats van de
        auto (elk $\frac13$): staat de auto achter jouw deur, dan verliest
        wisselen; staat hij achter een van de andere twee, wat moet de
        quizmaster dan doen, en wat wint wisselen dan? Bereken
        $P(\text{winst door te wisselen})$.
  \item De versie van één regel: wisselen wint precies wanneer je eerste
        keuze fout was. Besluit opnieuw.
  \item En nu de beslissende variant --- ``Monty Val'': de quizmaster weet
        \emph{niet} waar de auto staat, opent willekeurig een van de andere
        deuren, en onthult bij toeval een geit. Noem de zes even
        waarschijnlijke scenario's (plaats van de auto $\times$ geopende
        deur), gooi die weg waarin de auto onthuld wordt, en bereken de \omterm{def:g10:proba:distribution}{kans}
        dat wisselen wint \emph{onder de overblijvende scenario's}. Wat
        veranderde er, en waarom doet de kennis van de quizmaster ertoe?
  \item De verbeelding op gang: honderd deuren, jij kiest er één, de wetende
        quizmaster opent $98$ geitendeuren. Wisselen? Met welke \omterm{def:g10:proba:distribution}{kans} op
        winst?
  \item Je staat tegenover een scepticus die, net als de duizend
        briefschrijvers, op ``fiftyfifty'' blijft hameren. Beschrijf een
        simulatie met kaarten (drie kaarten, veel rondes, een vriend als
        quizmaster) en zeg welk aandeel gewonnen wisselingen je na $300$
        rondes verwacht --- met het schommelingsinterval uit dit hoofdstuk als
        marge.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Neven van de paradox.}
\begin{enumerate}[resume]
  \item Een gezin heeft twee kinderen; je verneemt dat \emph{minstens één
        ervan een jongen is}. Noem de even waarschijnlijke samenstellingen
        van een gezin met twee kinderen die met die informatie verenigbaar
        zijn, en bereken $P(\text{twee jongens})$.
  \item Hetzelfde gezin, maar de informatie luidt nu: \emph{het oudste kind
        is een jongen}. Bereken opnieuw. Waarom verschillen de twee
        antwoorden, en wat delen ze met de twee Monty's?
  \item Drie eerlijke munten: bereken $P(\text{alle drie gelijk})$. De
        wiskundige d'Alembert pleitte in 1754 voor een ander antwoord: ``ofwel
        zijn ze alle drie gelijk, ofwel niet --- één geval op twee''. Noem de
        precieze fout, met \cref{prop:g10:proba:uniform} bij de hand.
  \item Meespelen aan een spel kost $2$ euro: gooi met een dobbelsteen, win
        $6$ euro bij een zes, $3$ euro bij een vijf, en niets in de overige
        gevallen (\cref{exo:g10:proba:9}). Welke \omterm{def:g10:stats:mean}{gemiddelde} winst per spel
        voorspelt het tellen over $600$ spellen? Een verzekeraar past
        dezelfde wiskunde toe: een \omterm{def:g10:proba:distribution}{kans} van $1\,\%$ per jaar op een schade van
        $10\,000$ euro --- wat is de \emph{eerlijke} premie, en waarom ligt de
        werkelijke premie erboven?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Steekproeven als scheidsrechter.}
\begin{enumerate}[resume]
  \item Een munt wordt $100$ keer opgegooid. Beslis met het
        schommelingsinterval $p \pm \frac{1}{\sqrt n}$ welke van deze
        uitkomsten de wenkbrauwen doen fronsen: $55$ keer kop; $70$ keer kop.
  \item Een peiling bij $1\,000$ kiezers geeft een kandidaat $52\,\%$.
        Bereken het schommelingsinterval rond $50\,\%$ en zeg of de peiling
        \emph{bewijst} dat de kandidaat voorstaat.
  \item Een fabriek beweert $4\,\%$ defecte onderdelen te maken. Een
        \omterm{def:g10:proba:sample}{steekproef} van $400$ toont er $28$ defect ($7\,\%$); een latere
        \omterm{def:g10:proba:sample}{steekproef} van $2\,500$ toont er $175$ (opnieuw $7\,\%$). Toets de
        bewering aan elke \omterm{def:g10:proba:sample}{steekproef}. Waarom verschillen de oordelen?
  \item Twee foutenbronnen achtervolgen elk onderzoek: \emph{vertekening}
        van de \omterm{def:g10:proba:sample}{steekproef} (de weekendopgave over hoe gegevens liegen in het
        onderbouwvolume) en \emph{schommeling} van de \omterm{def:g10:proba:sample}{steekproef} (dit
        hoofdstuk). Welke van beide krimpt naarmate $n$ groeit, en welke
        overleeft elke steekproefomvang? Telkens één zin.
  \item Slotstuk --- het lijstje van de kansrekenaar, één regel per punt:
        controleer dat de uitkomsten even waarschijnlijk zijn vóór je telt
        (vraag 15); boomdiagrammen en \omterm{def:g10:proba:operations}{complementen} voor meerdere stappen en
        voor ``minstens'' (vragen 2--3); \emph{informatie verandert kansen, en
        de manier waarop je ze verkreeg doet ertoe} (vragen 7--14);
        steekproeven schommelen met ongeveer $\frac{1}{\sqrt n}$, en simulatie
        spreekt recht (vragen 11, 16--18).
\end{enumerate}
\end{problem}
