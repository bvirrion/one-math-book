\chapter{Analytische meetkunde}\label{ch:g10:coordgeom}

Het grote idee van Descartes was punten door getallenparen te beschrijven,
zodat meetkundige vraagstukken berekeningen worden. Met slechts twee formules
--- die voor het \omterm{prop:g10:coordgeom:midpoint}{midden} en die voor de afstand --- bewijs je dat een driehoek
gelijkbenig is, dat een vierhoek een parallellogram is of dat drie punten op
één rechte liggen, zonder één hulplijn te tekenen.

\section{Coördinaten in het vlak}

\begin{definition}[Assenstelsel]\label{def:g10:coordgeom:system}
Een \emph{assenstelsel}\index{assenstelsel} van het vlak bestaat uit een
oorsprong $O$ en twee geijkte assen door $O$: de horizontale $x$-as en de
verticale $y$-as. Elk punt $M$ heeft dan een uniek paar
\emph{coördinaten}\index{coördinaten} $(x, y)$: zijn
\emph{abscis}\index{abscis} $x$ en zijn \emph{ordinaat}\index{ordinaat} $y$.
Het stelsel heet \emph{orthonormaal}\index{orthonormaal assenstelsel} wanneer
de assen loodrecht op elkaar staan en dezelfde lengte-eenheid dragen. Alle
stelsels in dit hoofdstuk zijn orthonormaal.
\end{definition}

\section{Het midden van een lijnstuk}

\begin{proposition}[Formule voor het midden]\label{prop:g10:coordgeom:midpoint}
Zij $A(x_A, y_A)$ en $B(x_B, y_B)$. Het \emph{midden}\index{midden} $I$ van
het lijnstuk $[AB]$ heeft \omterm{def:g10:coordgeom:system}{coördinaten}
\[
I\left(\frac{x_A + x_B}{2},\ \frac{y_A + y_B}{2}\right).
\]
\end{proposition}

\begin{proof}
Bekijk de horizontale \omterm{def:g10:coordgeom:system}{coördinaten}. Van $A$ naar $B$ verandert de \omterm{def:g10:coordgeom:system}{abscis} met
$x_B - x_A$; het punt halverwege heeft dus \omterm{def:g10:coordgeom:system}{abscis}
\[
x_A + \frac{x_B - x_A}{2} = \frac{2x_A + x_B - x_A}{2}
= \frac{x_A + x_B}{2},
\]
het gemiddelde van de twee abscissen. Dezelfde berekening geldt voor de
ordinaten.
\end{proof}

\begin{example}\label{ex:g10:coordgeom:midpoint}
Met $A(-1, 4)$ en $B(5, -2)$ is het \omterm{prop:g10:coordgeom:midpoint}{midden} van $[AB]$
\[
I\left(\frac{-1 + 5}{2},\ \frac{4 + (-2)}{2}\right) = I(2, 1).
\]
De formule werkt ook achterwaarts: is $A(-1, 4)$ en is het \omterm{prop:g10:coordgeom:midpoint}{midden} $I(2, 1)$,
dan voldoet $B$ aan $\frac{-1 + x_B}{2} = 2$ en $\frac{4 + y_B}{2} = 1$, dus
$x_B = 5$ en $y_B = -2$: het punt $B$ is het \emph{spiegelbeeld} van $A$ om
$I$.
\end{example}

\section{Afstand tussen twee punten}

\begin{theorem}[Afstandsformule]\label{thm:g10:coordgeom:distance}
In een \omterm{def:g10:coordgeom:system}{orthonormaal assenstelsel} is de afstand tussen $A(x_A, y_A)$ en
$B(x_B, y_B)$ gelijk aan
\[
AB = \sqrt{(x_B - x_A)^2 + (y_B - y_A)^2}.
\]
\end{theorem}

\begin{proof}
Zij $C$ het punt $(x_B, y_A)$: het ligt op dezelfde hoogte als $A$ en heeft
dezelfde \omterm{def:g10:coordgeom:system}{abscis} als $B$, zodat de driehoek $ACB$ een rechte hoek in $C$
heeft. Zijn rechthoekszijden zijn een horizontaal en een verticaal lijnstuk,
met lengten $AC = \abs{x_B - x_A}$ en $CB = \abs{y_B - y_A}$. Volgens de
stelling van Pythagoras is
\[
AB^2 = AC^2 + CB^2 = (x_B - x_A)^2 + (y_B - y_A)^2,
\]
en de vierkantswortel trekken (beide leden zijn niet-negatief) geeft de
formule.
\end{proof}

\end{tikzpicture}

{\small De afstandsformule is de stelling van Pythagoras, toegepast op de
rechthoekige driehoek $ACB$ die op een horizontale en een verticale
rechthoekszijde staat.}
\end{omfigure}

\begin{example}\label{ex:g10:coordgeom:distance}
Met $A(1, 1)$ en $B(5, 3)$:
\[
AB = \sqrt{(5-1)^2 + (3-1)^2} = \sqrt{16 + 4} = \sqrt{20} = 2\sqrt 5 .
\]
Afstanden houd je gewoonlijk in exacte vorm (met wortels); rond pas op het
allerlaatst af als er een decimaal antwoord nodig is.
\end{example}

\begin{remark}
De volgorde van de punten doet er niet toe: $(x_A - x_B)^2 =
(x_B - x_A)^2$. Maar vergeet de kwadraten niet! De afstand is \emph{niet}
$\abs{x_B - x_A} + \abs{y_B - y_A}$.
\end{remark}

\section{De formules meetkundig gebruiken}

\begin{method}[De aard van een driehoek]\label{met:g10:coordgeom:triangle}
Zijn drie punten $A$, $B$, $C$ door hun \omterm{def:g10:coordgeom:system}{coördinaten} gegeven:
\begin{enumerate}
  \item bereken met de afstandsformule de drie kwadraten $AB^2$, $AC^2$,
        $BC^2$ (met kwadraten werken spaart wortels uit);
  \item twee gelijke kwadraten $\Rightarrow$ de driehoek is
        \emph{gelijkbenig};
  \item is het grootste kwadraat de som van de twee andere, dan is de driehoek
        \emph{rechthoekig} in het hoekpunt tegenover de langste zijde
        (omgekeerde stelling van Pythagoras).
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:coordgeom:triangle}
Zij $A(0, 1)$, $B(4, 3)$ en $C(2, -3)$. Dan is
\begin{align*}
AB^2 &= (4-0)^2 + (3-1)^2 = 16 + 4 = 20, \\
AC^2 &= (2-0)^2 + (-3-1)^2 = 4 + 16 = 20, \\
BC^2 &= (2-4)^2 + (-3-3)^2 = 4 + 36 = 40 .
\end{align*}
Ten eerste is $AB^2 = AC^2$, dus $AB = AC$: de driehoek is gelijkbenig in
$A$. Bovendien is $AB^2 + AC^2 = 40 = BC^2$, zodat de driehoek volgens de
omgekeerde stelling van Pythagoras ook rechthoekig is in $A$.
\end{example}

\begin{method}[Een parallellogram herkennen]\label{met:g10:coordgeom:parallelogram}
Een vierhoek $ABCD$ is precies dan een parallellogram wanneer zijn
diagonalen $[AC]$ en $[BD]$ hetzelfde \omterm{prop:g10:coordgeom:midpoint}{midden} hebben. Bereken dus beide
\omterm{prop:g10:coordgeom:midpoint}{middens} met de formule voor het \omterm{prop:g10:coordgeom:midpoint}{midden} en vergelijk ze.
\end{method}

\begin{example}\label{ex:g10:coordgeom:parallelogram}
Zij $A(-1, 0)$, $B(2, 2)$, $C(5, 1)$ en $D(2, -1)$. Het \omterm{prop:g10:coordgeom:midpoint}{midden} van $[AC]$ is
$\left(\frac{-1+5}{2}, \frac{0+1}{2}\right) = (2, \frac12)$; het \omterm{prop:g10:coordgeom:midpoint}{midden} van
$[BD]$ is $\left(\frac{2+2}{2}, \frac{2-1}{2}\right) = (2, \frac12)$. Ze
vallen samen, dus is $ABCD$ een parallellogram.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:1}
Zij $A(2, 5)$ en $B(-4, 1)$. Bereken de \omterm{def:g10:coordgeom:system}{coördinaten} van het \omterm{prop:g10:coordgeom:midpoint}{midden} van
$[AB]$ en de afstand $AB$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:2}
Zij $A(3, -2)$ en $I(1, 2)$. Zoek de \omterm{def:g10:coordgeom:system}{coördinaten} van het punt $B$ waarvoor
$I$ het \omterm{prop:g10:coordgeom:midpoint}{midden} van $[AB]$ is.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:3}
Welk van de punten $P(4, 1)$, $Q(-3, 2)$ en $R(0, -5)$ ligt het dichtst bij
de oorsprong $O(0,0)$? Antwoord met exacte afstanden.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:4}
Zet de punten $A(1, 2)$, $B(5, 4)$, $C(7, 0)$ op ruitjespapier uit, en toon
daarna met een berekening aan dat de driehoek $ABC$ gelijkbenig is. Is hij
rechthoekig?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:5}
Zij $A(-2, 1)$, $B(1, 3)$, $C(4, 1)$ en $D(1, -1)$.
\begin{enumerate}
  \item Toon aan dat $ABCD$ een parallellogram is.
  \item Bereken $AB$ en $BC$. Is $ABCD$ een ruit (alle zijden gelijk)?
  \item Bereken $AC$ en $BD$. Is $ABCD$ een rechthoek (gelijke diagonalen)?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:6}
De cirkel met middelpunt $\Omega(2, 1)$ en straal $5$ bestaat uit alle
punten op afstand $5$ van $\Omega$. Welke van de punten $A(5, 5)$,
$B(-1, -3)$, $C(6, 2)$ liggen op die cirkel? Welke erbinnen? Welke
erbuiten?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:7}
Zij $A(1, 1)$ en $B(7, 5)$. Zoek alle punten $M(x, 0)$ van de $x$-as die op
gelijke afstand van $A$ en $B$ liggen. (Schrijf $MA^2 = MB^2$ en los op naar
$x$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:8}
Zij $A(0, 3)$, $B(4, 1)$.
\begin{enumerate}
  \item Bereken de \omterm{def:g10:coordgeom:system}{coördinaten} van het \omterm{prop:g10:coordgeom:midpoint}{midden} $I$ van $[AB]$.
  \item Toon aan dat $M(2, 2) = I$, en ga daarna met $MA$ en $MB$ na dat $M$
        op gelijke afstand van $A$ en $B$ ligt.
  \item Zoek een tweede punt, op de $y$-as, dat op gelijke afstand van $A$ en
        $B$ ligt.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:9}
Gegeven zijn de punten $A(-1, -1)$, $B(3, 1)$ en $C(11, 5)$. Bereken $AB$,
$BC$ en $AC$, en leid af dat $A$, $B$, $C$ op één rechte liggen. (Drie punten
liggen precies dan op één rechte wanneer de grootste van de drie afstanden de
som van de twee andere is.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:coordgeom:10}
Zij $A(a, 0)$ en $B(0, b)$ met $a, b > 0$, en zij $I$ het \omterm{prop:g10:coordgeom:midpoint}{midden} van $[AB]$.
Toon met een berekening aan dat $OI = IA = IB$, waarbij $O$ de oorsprong is.
(Dit bewijst een klassieke stelling: in een rechthoekige driehoek ligt het
\omterm{prop:g10:coordgeom:midpoint}{midden} van de schuine zijde op gelijke afstand van de drie hoekpunten.)
\end{exercise}

\section{Opgave: de brug van Descartes --- cirkelvergelijkingen en hoe je
te weten komt waar je bent}

\begin{problem}[{Weekendopgave --- de cirkel wordt een vergelijking, oude
stellingen worden berekeningen, en drie bakens bepalen een punt:
trilateratie met algebra}]
\label{pb:g10:coordgeom:1}
In 1637 bouwde René Descartes een brug tussen twee continenten: elke kromme
van de meetkunde werd een \emph{\omterm{def:g10:algebra:equation}{vergelijking}}, elke meetkundige vraag een
berekening. Deze opgave loopt de brug in beide richtingen over --- ze leidt
de \omterm{def:g10:algebra:equation}{vergelijking} van de cirkel af uit de afstandsformule
(\cref{thm:g10:coordgeom:distance}), bewijst klassieke stellingen opnieuw in
drie regels algebra, en eindigt waar de brug vandaag het drukst bereden wordt:
een positie berekenen uit afstandssignalen, het platte hart van gps.

\medskip
\noindent\textbf{Deel I --- De cirkel krijgt een \omterm{def:g10:algebra:equation}{vergelijking}.}
\begin{enumerate}
  \item Een punt $M(x, y)$ ligt precies dan op de cirkel met middelpunt
        $\Omega(2, 1)$ en straal $5$ wanneer $\Omega M = 5$. Kwadrateer die
        voorwaarde om de \omterm{def:g10:algebra:equation}{vergelijking} van de cirkel te krijgen:
        \[
        (x - 2)^2 + (y - 1)^2 = 25 .
        \]
  \item Toets het lidmaatschap: welke van $A(5, 5)$, $B(6, 4)$, $D(4, 4)$
        liggen op de cirkel, erbinnen, erbuiten?
  \item Een cirkel in vermomming: vul de kwadraten aan
        (\cref{pb:g10:algebra:1}) in
        \[
        x^2 + y^2 - 6x + 4y - 12 = 0
        \]
        en geef middelpunt en straal.
  \item Doe hetzelfde met $x^2 + y^2 - 6x + 4y + 14 = 0$. Wat onthult de
        aangevulde vorm? Formuleer het criterium: wanneer beschrijft
        $(x - h)^2 + (y - k)^2 = c$ een cirkel, wanneer één enkel punt, en
        wanneer helemaal niets?
  \item Snijd de cirkel van vraag 3 met de horizontale rechte $y = 3$: vul
        in en los op. Hoeveel punten, en hoe heet zo'n rechte meetkundig?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Oude stellingen in drie regels.}
\begin{enumerate}[resume]
  \item De cirkelstelling uit het onderbouwvolume (een rechte hoek in elke
        halve cirkel), opnieuw bewezen met algebra: zij $A(-r, 0)$ en
        $B(r, 0)$ de uiteinden van een middellijn en $M(x, y)$ een
        willekeurig punt van de cirkel $x^2 + y^2 = r^2$. Bereken
        $MA^2 + MB^2$ en besluit met de omgekeerde stelling van Pythagoras
        dat $\widehat{AMB}$ recht is.
  \item De zwaartelijnstelling: toon voor $A(-a, 0)$, $B(a, 0)$ (zodat het
        \omterm{prop:g10:coordgeom:midpoint}{midden} van $[AB]$ de oorsprong $I$ is) aan dat voor elk punt
        $M(x, y)$ geldt:
        \[
        MA^2 + MB^2 = 2\,MI^2 + \frac{AB^2}{2} .
        \]
  \item Leid de meetkundige plaats af van de punten $M$ met
        $MA^2 + MB^2 = 26$ wanneer $AB = 6$: welke kromme, welk middelpunt,
        welke straal?
  \item Een andere meetkundige plaats: $A(0,0)$, $B(3,0)$; zoek alle punten
        met $MA = 2\,MB$. (Kwadrateer, werk uit, vul de kwadraten aan: er
        verschijnt een beroemde cirkel --- Apollonius kende hem zonder
        \omterm{def:g10:coordgeom:system}{coördinaten}.)
  \item In één of twee zinnen: wat verandert de brug van Descartes aan de
        \emph{manier} waarop stellingen bewezen worden? Vergelijk vraag 6 met
        het rechthoekbewijs van de cirkelstelling uit het onderbouwvolume.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Drie bakens vinden je.}
Je positie $M(x, y)$ is onbekend. Baken $P(0, 0)$ meet je afstand als $5$;
baken $Q(6, 0)$ meet eveneens $5$.
\begin{enumerate}[resume]
  \item Schrijf de twee cirkelvergelijkingen op, trek ze van elkaar af en
        kijk hoe de kwadraten wegvallen: welke eenvoudige \omterm{def:g10:algebra:equation}{vergelijking} blijft
        over? Combineer met één cirkel om de \emph{twee} kandidaatposities te
        vinden.
  \item Een derde baken $R(0, 8)$ meet je afstand als $5$. Bereken zijn
        afstand tot elke kandidaat en beslis waar je bent.
  \item Leg uit waarom het aftrekken van twee cirkelvergelijkingen
        \emph{altijd} de \omterm{def:g10:algebra:equation}{vergelijking} van een rechte oplevert (welke termen
        vallen weg?), en wat die rechte meetkundig is wanneer de cirkels
        elkaar in twee punten snijden.
  \item Echte satellietplaatsbepaling werkt in de ruimte en met onvolmaakte
        klokken: elke satelliet geeft (via de looptijd van het signaal) één
        afstand, en dus één bol. Hoeveel onbekenden heeft een ontvanger
        (positie \emph{én} zijn eigen klokfout), en waarom luistert gps
        daarom naar minstens \emph{vier} satellieten?
  \item Nauwkeurigheid: stel dat de afstand van baken $P$ in werkelijkheid
        $5.1$ is in plaats van $5$ (al de rest ongewijzigd). Doe de
        aftrekking van vraag 11 opnieuw om de nieuwe $x$ te vinden, en
        kwantificeer hoever de schatting opschoof. (Vergelijk met de
        foutenbegroting van \cref{pb:g10:numbers:1}.)
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De gereedschapskist van de landmeter.}
\begin{enumerate}[resume]
  \item De kortste weg langs het water: een kamp in $A(1, 5)$, een schuur in
        $B(7, 3)$, en een rechte rivier langs de as $y = 0$. Om van $A$ naar
        de rivier en dan naar $B$ te gaan met zo weinig mogelijk stappen:
        spiegel $B$ over de rivier naar $B'$, en zoek waar het lijnstuk
        $[AB']$ de rivier kruist. Geef het snijpunt en de kortste totale
        lengte. (Het idee is de biljartoefening uit het onderbouwvolume, nu
        volledig berekenbaar.)
  \item Bevestig de minimaliteit met een concurrent: bereken de totale weg
        via $Q(3, 0)$ en ga na dat ze het van jouw antwoord verliest.
  \item Bepaal de vierhoek $A(1,1)$, $B(4,2)$, $C(5,5)$, $D(2,4)$ volledig:
        parallellogram? ruit? rechthoek? vierkant?
        (\cref{met:g10:coordgeom:parallelogram},
        \cref{met:g10:coordgeom:triangle} --- vergelijk ook de diagonalen.)
  \item Zoek bij $A(-1, 2)$, $B(3, 4)$, $C(6, 0)$ het punt $D$ waarvoor
        $ABCD$ een parallellogram is (de truc met het \omterm{prop:g10:coordgeom:midpoint}{midden} van de
        diagonalen uit de weekendopgave over halve draaiingen in het
        onderbouwvolume, nu in formules).
  \item Slotstuk: drie gereedschappen --- de formule voor het \omterm{prop:g10:coordgeom:midpoint}{midden}, de
        afstandsformule en het aftrekken van cirkelvergelijkingen. Noem voor
        elk het soort vraag dat het in deze opgave beslechtte, en zeg wat de
        twee volgende hoofdstukken aan de kist toevoegen (richtingen en
        \omterm{def:g10:reffunc:affine}{richtingscoëfficiënten}: vectoren en \omterm{def:g10:algebra:equation}{vergelijkingen} van rechten).
\end{enumerate}
\end{problem}
