\chapter{الأعداد ومجموعات الأعداد}\label{ch:g10:numbers}

تبدأ الرياضيات بالأعداد، وليست الأعداد كلها من نوع واحد: أعداد العدّ،
والأعداد السالبة، والكسور، وأعداد مثل $\sqrt 2$ أو $\pi$ لا يقدر أي كسر
على التعبير عنها. يرتّب هذا الفصل تلك الأعداد في عائلات متداخلة، ويقدّم
الفترات لوصف أجزاء من مستقيم الأعداد، ويستعمل \omterm{def:g10:numbers:abs}{القيمة المطلقة} لقياس
المسافات بين الأعداد.

\section{عائلات الأعداد}

\begin{definition}[مجموعات الأعداد]\label{def:g10:numbers:sets}
\begin{itemize}
  \item $\N$ هي مجموعة \emph{الأعداد الطبيعية}\index{عدد طبيعي}:
        $0, 1, 2, 3, \dots$
  \item $\Z$ هي مجموعة \emph{الأعداد الصحيحة}\index{عدد صحيح}: $\dots, -2, -1,
        0, 1, 2, \dots$
  \item $\Q$ هي مجموعة \emph{الأعداد الناطقة}\index{عدد ناطق}:
        كل الخارجات $\frac{p}{q}$ حيث $p \in \Z$، $q \in \N$،
        $q \neq 0$.
  \item $\R$ هي مجموعة \emph{الأعداد الحقيقية}\index{عدد حقيقي}: كل
        الأعداد التي يمكن وضعها على مستقيم الأعداد.
\end{itemize}
\end{definition}

\begin{notation}
يُقرأ الرمز $\in$ ``ينتمي إلى'': $3 \in \N$، $-5 \in \Z$،
$\frac23 \in \Q$. ويُقرأ الرمز $\subset$ ``محتواة في'': كل
\omterm{def:g10:numbers:sets}{عدد طبيعي} \omterm{def:g10:numbers:sets}{عدد صحيح}، وكل \omterm{def:g10:numbers:sets}{عدد صحيح} \omterm{def:g10:numbers:sets}{عدد ناطق}
(مثلًا $-5 = \frac{-5}{1}$)، وكل \omterm{def:g10:numbers:sets}{عدد ناطق} \omterm{def:g10:numbers:sets}{عدد حقيقي}، إذن
\[
\N \subset \Z \subset \Q \subset \R .
\]
وتُبطل الشرطة المائلة الرمز: $\frac12 \notin \Z$.
\end{notation}

\begin{example}\label{ex:g10:numbers:classify}
لنضع بضعة أعداد في أصغر عائلة تحتويها.
\begin{itemize}
  \item $\frac{28}{4} = 7$، إذن $\frac{28}{4} \in \N$ رغم أنه مكتوب
        على صورة كسر: اختزل دائمًا قبل كل شيء.
  \item $-3.5 = -\frac{35}{10} = -\frac72$ \omterm{def:g10:numbers:sets}{عدد ناطق} لكنه ليس عددًا صحيحًا.
  \item $0.333\dots$ (الرقم $3$ يتكرر بلا انقطاع) يساوي $\frac13$،
        وهو \omterm{def:g10:numbers:sets}{عدد ناطق}.
  \item $\sqrt 2$ و $\pi$ عددان حقيقيان غير ناطقين، كما سنرى بعد قليل
        في حالة $\sqrt 2$؛ وتُسمّى هذه الأعداد
        \emph{صمّاء}\index{عدد أصم}.
\end{itemize}
\end{example}

\begin{theorem}[العدد $\sqrt 2$ أصمّ]\label{thm:g10:numbers:sqrt2}
العدد $\sqrt 2$ ليس ناطقًا: ما من كسر لعددين صحيحين مربعه
$2$.
\end{theorem}

\begin{proof}
نبرهن بالخلف، على خطوات صغيرة.
\begin{enumerate}
  \item لنفترض أن $\sqrt 2 = \frac{p}{q}$ حيث $p$ و $q$ عددان صحيحان
        موجبان، والكسر مختزل تمامًا، أي أن $p$ و $q$ ليسا زوجيين معًا.
  \item بتربيع الطرفين نجد $2 = \frac{p^2}{q^2}$، أي
        $p^2 = 2q^2$. إذن $p^2$ زوجي.
  \item لو كان $p$ فرديًا، وليكن $p = 2k+1$، لكان
        $p^2 = 4k^2 + 4k + 1$ فرديًا. وبما أن $p^2$ زوجي، فإن $p$
        زوجي بالضرورة: $p = 2k$ من أجل \omterm{def:g10:numbers:sets}{عدد صحيح} $k$.
  \item بالتعويض: $(2k)^2 = 2q^2$، إذن $4k^2 = 2q^2$، إذن
        $q^2 = 2k^2$. وبالحجة نفسها المستعملة في الخطوة 3، فإن $q$ زوجي.
  \item صار $p$ و $q$ زوجيين معًا، وهذا يناقض الخطوة 1. إذن
        الافتراض مستحيل: $\sqrt 2$ عدد أصمّ.
\end{enumerate}
\end{proof}

\begin{remark}
\omterm{def:g10:numbers:sets}{الأعداد الناطقة} هي بالضبط الأعداد التي يتوقف تمثيلها العشري
(مثل $\frac72 = 3.5$) أو ينتهي إلى تكرار المقطع نفسه بلا انقطاع
(مثل $\frac13 = 0.333\dots$ أو $\frac{1}{7} = 0.142857\,142857\dots$).
أما الأعداد \omterm{ex:g10:numbers:classify}{الصمّاء} مثل $\sqrt2 = 1.41421356\dots$ فأرقامها العشرية
لا تستقر أبدًا على نمط متكرر. ونقبل هذا التوصيف في هذا
المستوى.
\end{remark}

\section{مستقيم الأعداد والفترات}

تملأ \omterm{def:g10:numbers:sets}{الأعداد الحقيقية} مستقيمًا: باختيار مبدأ $0$ ووحدة للطول، يقابل
كل \omterm{def:g10:numbers:sets}{عدد حقيقي} نقطة واحدة بالضبط.

\begin{definition}[الفترة]\label{def:g10:numbers:interval}
ليكن $a$ و $b$ عددين حقيقيين حيث $a < b$.
\emph{الفترة}\index{فترة} هي مجموعة كل \omterm{def:g10:numbers:sets}{الأعداد الحقيقية} المحصورة بين
طرفين. تدل العلامة المربعة على أن الطرف مُدرَج، وتدل العلامة الهلالية
على أنه مستبعَد:
\begin{center}
\begin{tabular}{c|c}
الترميز & الوصف \\
\hline
$\intcc{a}{b}$ & $a \leq x \leq b$ (الطرفان مُدرَجان) \\
$\intoo{a}{b}$ & $a < x < b$ (الطرفان مستبعَدان) \\
$\intco{a}{b}$ & $a \leq x < b$ \\
$\intoc{a}{b}$ & $a < x \leq b$ \\
$\intco{a}{+\infty}$ & $x \geq a$ \\
$\intoo{-\infty}{b}$ & $x < b$ \\
\end{tabular}
\end{center}
الرمزان $-\infty$ و $+\infty$ (``ما لا نهاية'') ليسا عددين، بل هما
طريقة للقول إن الفترة تمضي بلا حد؛ والعلامة المجاورة لهما هلالية
دائمًا. والمستقيم كله $\R$ هو الفترة
$\intoo{-\infty}{+\infty}$.
\end{definition}

\begin{definition}[التقاطع والاتحاد]\label{def:g10:numbers:interunion}
لتكن $I$ و $J$ مجموعتين من \omterm{def:g10:numbers:sets}{الأعداد الحقيقية}.
\emph{التقاطع}\index{تقاطع} $I \cap J$ (``$I$ و $J$'')
هو مجموعة الأعداد التي تنتمي إليهما معًا؛ و\emph{الاتحاد}\index{اتحاد}
$I \cup J$ (``$I$ أو $J$'') هو مجموعة الأعداد التي تنتمي إلى إحداهما
على الأقل.
\end{definition}

\begin{example}\label{ex:g10:numbers:interunion}
لنأخذ $I = \intcc{-1}{3}$ و $J = \intoo{1}{5}$. ارسمهما على المستقيم
نفسه: يتداخلان بين $1$ و $3$. إذن
\[
I \cap J = \intoc{1}{3},
\qquad
I \cup J = \intco{-1}{5}.
\]
لاحظ العلامات: $1 \notin J$ إذن $1 \notin I \cap J$، لكن $3 \in I$ و
$3 \in J$، إذن $3 \in I \cap J$.
\end{example}

\begin{method}[العمل بالفترات]\label{met:g10:numbers:intervals}
لإيجاد \omterm{def:g10:numbers:interunion}{تقاطع} فترتين أو اتحادهما:
\begin{enumerate}
  \item ارسم مستقيم الأعداد وعلّم الأطراف الأربعة؛
  \item ظلّل \omterm{def:g10:numbers:interval}{الفترة} الأولى فوق المستقيم والثانية تحته؛
  \item \omterm{def:g10:numbers:interunion}{التقاطع} حيث يتراكب التظليلان، \omterm{def:g10:numbers:interunion}{والاتحاد} حيث يوجد
        تظليل واحد على الأقل؛
  \item حدّد كل علامة باختبار ما إذا كان الطرف نفسه ينتمي إلى
        المجموعتين معًا (\omterm{def:g10:numbers:interunion}{التقاطع}) أو إلى إحداهما على الأقل (\omterm{def:g10:numbers:interunion}{الاتحاد}).
\end{enumerate}
\end{method}

\section{القيمة المطلقة والمسافة}

\begin{definition}[القيمة المطلقة]\label{def:g10:numbers:abs}
\emph{القيمة المطلقة}\index{قيمة مطلقة} \omterm{def:g10:numbers:sets}{لعدد حقيقي} $x$ هي
\[
\abs{x} =
\begin{cases}
x & \text{إذا كان } x \geq 0, \\
-x & \text{إذا كان } x < 0.
\end{cases}
\]
مثلًا $\abs{7} = 7$ و $\abs{-4} = 4$. وفي كل الحالات
$\abs{x} \geq 0$.
\end{definition}

\begin{proposition}[المسافة على المستقيم]\label{prop:g10:numbers:distance}
من أجل كل عددين حقيقيين $a$ و $b$، المسافة بين النقطتين $a$ و
$b$ على مستقيم الأعداد هي $\abs{b - a}$. وبوجه خاص، $\abs{x}$ هي
المسافة من $x$ إلى $0$.
\end{proposition}

\begin{proof}
إذا كان $b \geq a$، فالمسافة من $a$ إلى $b$ هي $b - a \geq 0$، وهي
تساوي $\abs{b-a}$. وإذا كان $b < a$، فالمسافة هي $a - b = -(b - a) > 0$،
وهي مرة أخرى $\abs{b - a}$ حسب تعريف \omterm{def:g10:numbers:abs}{القيمة المطلقة}.
\end{proof}

\begin{proposition}[القيمة المطلقة والفترات]\label{prop:g10:numbers:absinterval}
ليكن $a$ عددًا حقيقيًا و $r > 0$. عندئذٍ
\[
\abs{x - a} \leq r
\quad\text{إذا وفقط إذا}\quad
x \in \intcc{a - r}{a + r}.
\]
\end{proposition}

\begin{proof}
تعني $\abs{x-a} \leq r$ أن المسافة من $x$ إلى $a$ لا تتجاوز $r$،
أي أن $x$ لا يبعد عن $a$ بأكثر من $r$ في أي من الجهتين. والأعداد
التي تحقق ذلك هي بالضبط الأعداد المحصورة بين $a - r$ و $a + r$،
مع إدراج الطرفين.
\end{proof}

\begin{example}\label{ex:g10:numbers:abs}
حل $\abs{x - 3} \leq 2$. الحلول هي الأعداد التي تبعد عن $3$ بمسافة
لا تتجاوز $2$: \omterm{def:g10:numbers:interval}{الفترة} $\intcc{1}{5}$. وبالعكس، \omterm{def:g10:numbers:interval}{الفترة}
$\intcc{-1}{7}$ مركزها $\frac{-1+7}{2} = 3$ ونصف قطرها
$\frac{7-(-1)}{2} = 4$، إذن توصف بالمتراجحة $\abs{x - 3} \leq 4$.
\end{example}

\section{التقريبات}

الأعداد \omterm{ex:g10:numbers:classify}{الصمّاء}، بل ومعظم الكسور أيضًا، لا يمكن كتابتها بالضبط
بعدد منتهٍ من الأرقام العشرية، لذلك نقرّبها عمليًا.

\begin{definition}[التقريب بدقة معطاة]\label{def:g10:numbers:approx}
يكون العدد $d$ \emph{تقريبًا}\index{تقريب} للعدد $x$ بدقة
$10^{-n}$ إذا كان $\abs{x - d} \leq 10^{-n}$. ويعطي بترُ التمثيل العشري
بعد الرقم ذي الرتبة $n$ أو تدويرُه عنده تقريبين من هذا
النوع.
\end{definition}

\begin{example}
انطلاقًا من $\pi = 3.14159\,26\dots$: البتر $3.141$ والتدوير
$3.142$ تقريبان للعدد $\pi$ بدقة $10^{-3}$. ويبعد التدوير
بأقل من $\frac12 \times 10^{-3}$، أما البتر فلا يضمن سوى $10^{-3}$.
وكتابة $3.141 \leq \pi \leq 3.142$ \emph{تحصر} $\pi$ بين طرفين عشريين.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g10:numbers:1}
أعطِ لكل عدد أصغرَ المجموعات $\N$، $\Z$، $\Q$، $\R$ التي ينتمي
إليها:
\[
\frac{15}{3}, \qquad -7, \qquad \frac{22}{7}, \qquad \sqrt{9}, \qquad
\sqrt{10}, \qquad -2.4, \qquad 0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:2}
اكتب كل عبارة بترميز الفترات، ثم ارسمها على مستقيم
الأعداد: (أ) $-2 \leq x < 5$؛ (ب) $x > 3$؛ (ج) $x \leq -1$؛
(د) المسافة من $x$ إلى $2$ لا تتجاوز $3$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:3}
احسب $I \cap J$ و $I \cup J$ من أجل
\[
\text{(أ) } I = \intcc{-3}{2},\ J = \intco{0}{4};
\qquad
\text{(ب) } I = \intoo{-\infty}{1},\ J = \intco{-2}{+\infty}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:4}
احسب دون آلة حاسبة:
\[
\abs{-6}, \qquad \abs{4 - 9}, \qquad \abs{-3 - 5}, \qquad
\abs{\sqrt 2 - 1}, \qquad \abs{1 - \sqrt 2}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:5}
حل المعادلات والمتراجحات، وأعطِ مجموعات الحلول:
\[
\abs{x} = 5, \qquad \abs{x - 1} = 3, \qquad \abs{x - 4} \leq 1, \qquad
\abs{x + 2} < 3 .
\]
(لاحظ أن $\abs{x+2} = \abs{x - (-2)}$ مسافة إلى $-2$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:6}
صف كل \omterm{def:g10:numbers:interval}{فترة} بمتراجحة على الصورة

\intcc{2}{8}, \qquad \intoo{-5}{1}, \qquad \intcc{-7}{-3}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:7}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:8}

  \item خارج قسمة عددين صحيحين \omterm{def:g10:numbers:sets}{عدد صحيح}.
  \item مجموع عددين ناطقين \omterm{def:g10:numbers:sets}{عدد ناطق}.
  \item مجموع \omterm{def:g10:numbers:sets}{عدد ناطق} وعدد أصمّ عدد

\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:9}
باستعمال $1.414 \leq \sqrt 2 \leq 1.415$، احصر الأعداد $2\sqrt2$ و

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:numbers:10}
كيّف برهان \cref{thm:g10:numbers:sqrt2} لتبيّن أن $\sqrt 3$ عدد
أصمّ. (استبدل كلمة ``زوجي'' بعبارة ``مضاعف للعدد $3$'': تحقق أولًا من أنه إذا

$3$، وهي $0$ و $1$ و $2$.)
\end{exercise}

\section{مسألة: بين أي عددين}

\begin{problem}[{مسألة نهاية الأسبوع --- تتشابك الأعداد الناطقة
والصمّاء: كل فترة، مهما صغرت، تحوي عددًا لا نهائيًا من كل
منهما، ولا يستطيع أي قياس أن يميّز بينهما}]
\label{pb:g10:numbers:1}
تبدو \omterm{def:g10:numbers:sets}{الأعداد الناطقة} حشدًا (كل الكسور!) وتبدو الأعداد
\omterm{ex:g10:numbers:classify}{الصمّاء} استثناءات نادرة ($\sqrt2$، $\pi$). تكشف هذه المسألة
الصورة الحقيقية: العائلتان \emph{تتشابكان} تشابكًا دقيقًا
إلى حدّ أن كل \omterm{def:g10:numbers:interval}{فترة} من مستقيم الأعداد، مهما كانت
مجهرية، تحوي عددًا لا نهائيًا من كل منهما --- مع نتائج
غريبة، منها هذه: ما من قياس فيزيائي، مهما بلغت
دقته، يستطيع أن يقرر أطولٌ ما ناطق أم لا.

\medskip
\noindent\textbf{الجزء الأول --- الممالك الأربع.}
\begin{enumerate}
  \item عيّن لكل عدد أصغرَ المجموعات $\N$،
        $\Z$، $\Q$، $\R$ التي تحتويه
        (\cref{def:g10:numbers:sets}): $-7$؛ \ $\frac{13}{4}$؛ \
        $\sqrt{16}$؛ \ $0.121212\ldots$ (تذكّر
        مسألة نهاية الأسبوع عن العشريات الدورية في الكتاب السابق)؛ \ $\sqrt8$؛ \ $\pi$ (اقبل
        كونه أصمّ --- فذلك لا يُبرهن إلا في الكتب
        الجامعية).
  \item برهن على أن $\Q$ مستقرة بالجمع
        والضرب: إذا كان $x = \frac pq$ و $y = \frac rs$
        عددين ناطقين، فاكتب $x + y$ و $xy$ على صورة كسر واحد.
  \item استنتج بالخلف: (أ) مجموع \omterm{def:g10:numbers:sets}{عدد ناطق} وعدد
        أصمّ عدد أصمّ؛ (ب) جداء \omterm{def:g10:numbers:sets}{عدد ناطق}
        \emph{غير معدوم} في عدد أصمّ عدد أصمّ.
  \item بيّن أن مجموعة الأعداد \omterm{ex:g10:numbers:classify}{الصمّاء} ليست مستقرة
        \emph{بأيٍّ} من العمليتين: أعطِ عددين أصمّين
        مجموعهما ناطق، وعددين آخرين جداؤهما ناطق.
  \item حدّد موضع $\sqrt2 + \sqrt8$ و $\sqrt2 \times \sqrt8$ في
        الممالك الأربع (اختزل $\sqrt8$ أولًا،
        بطريقة الكتاب السابق).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- \omterm{def:g10:numbers:sets}{الأعداد الناطقة} كثيفة.}
\begin{enumerate}[resume]
  \item أوجد عددًا ناطقًا محصورًا تمامًا بين $3.47$ و $3.48$؛ ثم
        عددًا ثانيًا؛ ثم صف كيف تنتج منها ما شئت
        (تكبير مسألة نهاية الأسبوع عن انعدام العدد التالي في الكتاب السابق، وقد صار
        البرهان في المتناول).
  \item المبرهنة العامة. ليكن $a < b$ عددين حقيقيين أيًّا كانا،
        فارقهما $g = b - a > 0$. اختر $n$ بحيث
        $10^{-n} < g$، وانظر في مضاعفات $10^{-n}$
        (الشبكة العشرية ذات الخطوة $10^{-n}$). فسّر لماذا
        تقع نقطة واحدة على الأقل من الشبكة بين $a$ و $b$ تمامًا،
        واستنتج: \emph{كل \omterm{def:g10:numbers:interval}{فترة} ذات طول موجب
        تحوي عددًا ناطقًا}.
  \item ارفع الاستنتاج درجة: كل \omterm{def:g10:numbers:interval}{فترة} كهذه تحوي
        \emph{عددًا لا نهائيًا} من \omterm{def:g10:numbers:sets}{الأعداد الناطقة}. (طبّق السؤال 7
        مرة أخرى، داخل \omterm{def:g10:numbers:interval}{فترة} أصغر.)
  \item والآن الأعداد \omterm{ex:g10:numbers:classify}{الصمّاء}: من أجل $a < b$، خذ عددًا ناطقًا $r$
        داخلها تمامًا (السؤال 7)، وانظر في الأعداد
        $r + \frac{\sqrt2}{10^k}$. مستعينًا بالسؤال 3، بيّن أنها
        \omterm{ex:g10:numbers:classify}{صمّاء}، وأنها تبقى في \omterm{def:g10:numbers:interval}{الفترة} من أجل $k$ كبير
        بما يكفي: \emph{كل \omterm{def:g10:numbers:interval}{فترة} تحوي أيضًا
        عددًا لا نهائيًا من الأعداد \omterm{ex:g10:numbers:classify}{الصمّاء}}.
  \item لغزان كلاسيكيان يُحسمان: هل يوجد أصغر
        \omterm{def:g10:numbers:sets}{عدد حقيقي} موجب؟ وهل يوجد \omterm{def:g10:numbers:sets}{عدد حقيقي} ``يأتي مباشرة بعد
        $3$''؟ أجب عنهما بمبرهنة السؤال 7، وحيِّ
        الصيغة الطفولية للسؤال
        (مسألة نهاية الأسبوع عن انعدام العدد التالي في الكتاب السابق).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- \omterm{def:g10:numbers:abs}{القيمة المطلقة}، هندسة
$\R$.}
\begin{enumerate}[resume]
  \item حل، وعبّر عن الحلول بفترات أو باتحادات
        (\cref{prop:g10:numbers:absinterval}):
        $\abs{x - 5} = 2$؛ \quad $\abs{x - 5} < 2$؛ \quad
        $\abs{x + 1} \geq 3$.
  \item يجب أن يُصنع مكبس بقطر $80$ مم ضمن
        سماحية $0.05$ مم. اكتب الشرط \omterm{def:g10:numbers:abs}{بقيمة
        مطلقة}، ثم على صورة \omterm{def:g10:numbers:interval}{فترة}. مكبسان قياسهما
        $79.97$ و $80.06$ مم: ما الحكم فيهما؟
  \item متراجحة المثلث على المستقيم:
        $\abs{a + b} \leq \abs a + \abs b$. تحقق منها في حالتي
        $(a, b) = (3, -5)$ و $(-2, -7)$، وبرهن عليها حين يكون $a$ و
        $b$ من الإشارة نفسها وحين يكونان من إشارتين
        مختلفتين، وبيّن بالضبط متى تتحقق المساواة.
  \item حل $\abs{x - 2} = \abs{x + 4}$ بقراءتها مساواةً بين
        مسافتين على المستقيم. أي نقطة من
        مسألة نهاية الأسبوع عن المرآتين في الكتاب السابق حسبتَ للتوّ، ببعد
        واحد أقل؟
  \item بسّط $\sqrt{x^2}$ --- بحذر. اختبر صيغتك
        من أجل $x = 3$ و $x = -3$، ثم استعملها لحل
        $x^2 < 9$ \omterm{def:g10:numbers:abs}{بقيمة مطلقة}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- ما لا يستطيع أي قياس أن يحسمه.}
\begin{enumerate}[resume]
  \item يقيس فيزيائي قضيبًا: $1.41421 \pm 0.00001$ م.
        هل يستطيع قياس كهذا --- هذا القياس أو قياس
        أدق منه في المستقبل --- أن \emph{يبرهن} يومًا على أن طول
        القضيب أصمّ؟ أو على أنه ناطق؟ استعمل السؤالين 7
        و 9 على \omterm{def:g10:numbers:interval}{فترة} السماحية، وفسّر لماذا لا يحسم
        كون العدد أصمّ إلا \emph{البرهان} (كما في القطر، مسألة نهاية الأسبوع عن الصمم في الكتاب السابق).
  \item بتور $\sqrt2$، وهي $1.4$ و $1.41$ و $1.414$ و $1.4142,\dots$،
        كلها أعداد ناطقة. ماذا تبيّن عن
        مدى التصاق $\Q$ بكل عدد أصمّ؟
        صُغ الواقعة العامة.
  \item حساب الفترات: قيمتان مقيستان $a = 2.5 \pm 0.1$
        و $b = 1.2 \pm 0.1$. احصر $a + b$ و $a \times b$
        بين طرفين مؤكدين. أي العمليتين تُتلف
        الدقة أكثر؟
  \item تقول قاعدة المهندسين إن أخطاء القياسات
        المستقلة ``تتجمع'' في المجاميع: عبّر بمتراجحة
        المثلث (السؤال 13) عن سبب كون خطأ $a + b$ لا
        يتجاوز مجموع الخطأين --- فالمتراجحة هي
        قاعدة الهندسة التطبيقية.
  \item الخاتمة، في فقرة قصيرة: اجمع صورة
        مستقيم الأعداد التي أرستها هذه المسألة
        وسابقاتها --- لا وجود لعدد تالٍ
        (مسألة نهاية الأسبوع عن انعدام العدد التالي في الكتاب السابق)، \omterm{def:g10:numbers:sets}{والأعداد الناطقة} $=$ العشريات
        الدورية (مسألة نهاية الأسبوع عن العشريات الدورية في الكتاب السابق)، والعائلتان كثيفتان
        (السؤالان 8 و 9)، والقياس عاجز إلى الأبد
        (السؤال 16). واختم بالإشارة التي لا يستطيع هذا الفصل
        برهانها بعد: بمعنى دقيق، الأعداد \omterm{ex:g10:numbers:classify}{الصمّاء} \emph{أكثر
        بكثير} من الناطقة --- والكتب الجامعية
        تعدّ اللانهايات.
\end{enumerate}
\end{problem}
