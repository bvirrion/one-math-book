\chapter{الإحصاء الوصفي}\label{ch:g10:stats}

يلخّص الإحصاء مجموعة كبيرة من الأعداد --- العلامات، والرواتب،
ودرجات الحرارة --- ببضعة مؤشرات مختارة جيدًا. يقدّم هذا الفصل
عائلتَي المؤشرات: تلك التي تحدد \emph{مركز} المعطيات
(\omterm{def:g10:stats:mean}{المتوسط الحسابي}، \omterm{def:g10:stats:median}{والوسيط}) وتلك التي تقيس \emph{تشتتها} (\omterm{def:g10:stats:quartiles}{المدى}،
والربيعيات). أما المعالجة الأعمق، بما فيها الانحراف المعياري، فتأتي في
\cref{ch:g11:stat}.

\section{المعطيات والتواترات}

\begin{definition}[السلسلة الإحصائية]\label{def:g10:stats:series}
\emph{السلسلة الإحصائية}\index{سلسلة إحصائية} قائمة من القيم
$x_1, x_2, \dots, x_N$ مرصودة على مجتمع حجمه $N$. وعندما تظهر القيمة
$x$ عددًا $n$ من المرات، يكون $n$ هو \emph{تكرارها} ويكون الخارج
\[
f = \frac{n}{N}
\]
هو \emph{تواترها}\index{تواتر} (ويُعبَّر عنه غالبًا بنسبة
مئوية). ومجموع تواترات كل القيم يساوي $1$.
\end{definition}

\begin{example}\label{ex:g10:stats:frequencies}
علامات $20$ تلميذًا في اختبار قصير من $5$ نقاط:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|cccccc}
العلامة $x$ & $0$ & $1$ & $2$ & $3$ & $4$ & $5$ \\
\hline
التكرار $n$ & $1$ & $2$ & $4$ & $6$ & $5$ & $2$ \\
\omterm{def:g10:stats:series}{التواتر} $f$ & $0.05$ & $0.10$ & $0.20$ & $0.30$ & $0.25$ & $0.10$ \\
\end{tabular}
\end{center}
تحقق: $1 + 2 + 4 + 6 + 5 + 2 = 20$ ومجموع التواترات $1$.
\end{example}

\section{مؤشرات المركز}

\begin{definition}[المتوسط الحسابي]\label{def:g10:stats:mean}
\emph{المتوسط الحسابي}\index{متوسط حسابي} للسلسلة هو
\[
\bar x = \frac{x_1 + x_2 + \dots + x_N}{N}.
\]
وإذا ظهرت القيم $v_1, \dots, v_k$ بتكرارات $n_1, \dots, n_k$، حُسب
العدد نفسه على صورة \emph{المتوسط المرجّح}
\[
\bar x = \frac{n_1 v_1 + n_2 v_2 + \dots + n_k v_k}{n_1 + n_2 + \dots + n_k}.
\]
\end{definition}

\begin{example}\label{ex:g10:stats:mean}
من أجل الاختبار في \cref{ex:g10:stats:frequencies}:
\[
\bar x = \frac{1 \times 0 + 2 \times 1 + 4 \times 2 + 6 \times 3
+ 5 \times 4 + 2 \times 5}{20}
= \frac{0 + 2 + 8 + 18 + 20 + 10}{20}
= \frac{58}{20} = 2.9 .
\]
\end{example}

\begin{definition}[الوسيط]\label{def:g10:stats:median}
\emph{الوسيط}\index{وسيط} قيمة تقسم السلسلة \emph{المرتَّبة}
نصفين: فنصف القيم على الأقل $\leq$ الوسيط، و
نصفها على الأقل $\geq$ الوسيط. وعمليًا، رتّب القيم $N$؛
\begin{itemize}
  \item إذا كان $N$ فرديًا، فالوسيط هو القيمة الوسطى، في الرتبة
        $\frac{N+1}{2}$؛
  \item وإذا كان $N$ زوجيًا، فخذ \omterm{prop:g10:coordgeom:midpoint}{منتصف} القيمتين الوسطيين، في
        الرتبتين $\frac N2$ و $\frac N2 + 1$.
\end{itemize}
\end{definition}

\begin{example}\label{ex:g10:stats:median}
سلسلة مرتَّبة لأسعار $7$ منازل (بالآلاف): $120$ و $150$ و $160$ و
$180$ و $210$ و $240$ و $900$. \omterm{def:g10:stats:median}{الوسيط} هو القيمة الرابعة، $180$. أما \omterm{def:g10:stats:mean}{المتوسط
الحسابي} فهو $\frac{1960}{7} = 280$ --- وهو أكبر من $6$ من الأسعار السبعة!
فقيمة متطرفة واحدة ($900$) تجرّ \omterm{def:g10:stats:mean}{المتوسط} أكثر بكثير من \omterm{def:g10:stats:median}{الوسيط}:
\omterm{def:g10:stats:median}{فالوسيط} \emph{صامد} \omterm{def:g10:stats:mean}{والمتوسط} ليس كذلك.
\end{example}

\section{مؤشرات التشتت}

\begin{definition}[المدى والربيعيات]\label{def:g10:stats:quartiles}
من أجل سلسلة مرتَّبة:
\begin{itemize}
  \item \emph{المدى}\index{مدى} هو الفرق بين
        أكبر القيم وأصغرها؛
  \item و\emph{الربيعي الأول}\index{ربيعي} $Q_1$ هو أصغر
        قيمة تجعل ربع القيم على الأقل ($25\%$)
        $\leq Q_1$؛ و\emph{الربيعي الثالث} $Q_3$ هو أصغر قيمة
        تجعل ثلاثة أرباع القيم على الأقل ($75\%$)
        $\leq Q_3$؛
  \item و\emph{المدى الربيعي}\index{مدى ربيعي} $Q_3 - Q_1$ يقيس تشتت
        النصف المركزي من المعطيات.
\end{itemize}
\end{definition}

\begin{method}[إيجاد الربيعيين]\label{met:g10:stats:quartiles}
من أجل سلسلة من $N$ قيمة مرتَّبة:
\begin{enumerate}
  \item احسب $\frac N4$؛ وإذا لم يكن عددًا صحيحًا فقرّبه \emph{صعودًا}
        إلى \omterm{def:g10:numbers:sets}{العدد الصحيح} التالي؛ و $Q_1$ هو القيمة في تلك الرتبة؛
  \item احسب $\frac{3N}{4}$؛ وقرّب صعودًا عند اللزوم؛ و $Q_3$ هو القيمة في
        تلك الرتبة.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:stats:quartiles}
خذ القيم $11$ المرتَّبة
\[
2,\ 3,\ 5,\ 5,\ 6,\ 7,\ 8,\ 9,\ 9,\ 10,\ 12 .
\]
\omterm{def:g10:stats:median}{الوسيط}: الرتبة $\frac{11+1}{2} = 6$، إذن \omterm{def:g10:stats:median}{الوسيط} $7$.
و $Q_1$: $\frac{11}{4} = 2.75$، نقرّب صعودًا إلى $3$؛ والقيمة الثالثة هي $5$.
و $Q_3$: $\frac{33}{4} = 8.25$، نقرّب صعودًا إلى $9$؛ والقيمة التاسعة هي $9$.
\omterm{def:g10:stats:quartiles}{المدى}: $12 - 2 = 10$؛ \omterm{def:g10:stats:quartiles}{والمدى الربيعي}: $9 - 5 = 4$.
\end{example}

\begin{example}[مقارنة فوجين]\label{ex:g10:stats:comparison}
يخضع فوجان للاختبار نفسه (العلامات من $20$):
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccc}
 & \omterm{def:g10:stats:median}{الوسيط} & $Q_1$ & $Q_3$ \\
\hline
الفوج أ & $12$ & $10$ & $14$ \\
الفوج ب & $12$ & $6$ & $17$ \\
\end{tabular}
\end{center}
\omterm{def:g10:stats:median}{الوسيط} نفسه، والتشتت مختلف جدًّا: فالفوج أ متجانس (نصف
علاماته في $\intcc{10}{14}$)، والفوج ب غير متجانس (فنصفه المركزي
يمتد على $\intcc{6}{17}$). فمؤشرات المركز وحدها لا تحكي القصة
كاملة أبدًا.
\end{example}

\begin{remark}
ليس \omterm{def:g10:stats:mean}{المتوسط} ``أفضل'' من \omterm{def:g10:stats:median}{الوسيط} ولا العكس: \omterm{def:g10:stats:mean}{فالمتوسط} يستعمل كل قيمة
(وهو لازم لحساب المجاميع)، \omterm{def:g10:stats:median}{والوسيط} يقاوم القيم المتطرفة. و
التلخيص الجاد لسلسلة يعطي مؤشر مركز واحدًا على الأقل
\emph{ومؤشر} تشتت واحدًا.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g10:stats:1}
إليك أعداد الكتب التي قرأها $15$ تلميذًا في سنة:
\[
0,\ 1,\ 1,\ 2,\ 2,\ 2,\ 3,\ 3,\ 4,\ 4,\ 5,\ 6,\ 7,\ 8,\ 12 .
\]
احسب \omterm{def:g10:stats:mean}{المتوسط الحسابي} \omterm{def:g10:stats:median}{والوسيط} \omterm{def:g10:stats:quartiles}{والمدى} لهذه السلسلة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:2}
رُمي حجر نرد $50$ مرة:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|cccccc}
النتيجة & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ \\
\hline
التكرار & $7$ & $9$ & $8$ & $10$ & $6$ & $10$ \\
\end{tabular}
\end{center}
احسب \omterm{def:g10:stats:series}{تواتر} كل نتيجة والنتيجة المتوسطة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:3}
أوجد \omterm{def:g10:stats:median}{الوسيط} و $Q_1$ و $Q_3$ للسلسلة المرتَّبة
\[
1,\ 2,\ 4,\ 4,\ 5,\ 6,\ 6,\ 7,\ 8,\ 9,\ 9,\ 11 \qquad (N = 12),
\]
وارسم المخطط الصندوقي الموافق.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:4}
\omterm{def:g10:stats:mean}{متوسط} $24$ علامة اختبار هو $11.5$. ويجتاز تلميذ خامس وعشرون الاختبار
فيحصل على $19$. ما \omterm{def:g10:stats:mean}{المتوسط} الجديد للفوج؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:5}
حصل تلميذ على \omterm{def:g10:stats:mean}{متوسط} $12$ بعد $5$ اختبارات متساوية الوزن.
ما العلامة التي ترفع \omterm{def:g10:stats:mean}{المتوسط} إلى $13$ في الاختبار السادس؟ وهل \omterm{def:g10:stats:mean}{متوسط}
$14$ قابل للبلوغ (والعلامات من $20$)؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:6}
ابتكر سلسلة من $7$ قيم غير سالبة وسيطها $10$ ومتوسطها
أكبر من $20$؛ ثم سلسلة وسيطها $10$ ومتوسطها
أصغر من $6$. ولماذا لا يمكن \omterm{def:g10:stats:mean}{لمتوسط} سلسلة كهذه أن يكون أصغر من
$\frac{40}{7}$؟ وماذا تبيّن هذه الأمثلة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:7}
في شركة، يتقاضى الموظفون $9$ مبلغ $1800$ لكل منهم شهريًا ويتقاضى
المدير $12000$.
\begin{enumerate}
  \item احسب الراتب \omterm{def:g10:stats:mean}{المتوسط} والراتب \omterm{def:g10:stats:median}{الوسيط}.
  \item أي المؤشرين يصف الراتب ``النموذجي'' هنا وصفًا أفضل؟ ولماذا؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:8}
\omterm{def:g10:stats:mean}{متوسط} طول $12$ فتى في فوج هو $168$ سم؛ \omterm{def:g10:stats:mean}{ومتوسط} طول $18$ فتاة في
الفوج نفسه هو $163$ سم. احسب \omterm{def:g10:stats:mean}{متوسط} طول
الفوج كله. (انتبه: فهو \emph{ليس} \omterm{prop:g10:coordgeom:midpoint}{منتصف} $168$ و $163$
--- بل رجّح المتوسطين بحجمي المجموعتين.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:stats:9}
سلسلة من $N$ قيمة $x_1, \dots, x_N$ متوسطها $\bar x$. وتُحوَّل كل قيمة
إلى $y_i = a x_i + b$ من أجل عددين ثابتين $a$ و $b$.
\begin{enumerate}
  \item بيّن أن \omterm{def:g10:stats:mean}{متوسط} السلسلة الجديدة هو $a \bar x + b$.
  \item \omterm{def:g10:stats:mean}{متوسط} درجات حرارة أسبوع، بالدرجات المئوية، هو $20$.
        فما متوسطها بدرجات فهرنهايت
        ($F = 1.8\,C + 32$)؟
\end{enumerate}
\end{exercise}

\section{مسألة: الإنسان المتوسط لا وجود له}

\begin{problem}[{مسألة نهاية الأسبوع --- المتوسط في مواجهة الوسيط: القيم
الشاذة، ومنحنيات التصحيح، والمستودع على الطريق، وقمرة القيادة المصممة
لأحد}]
\label{pb:g10:stats:1}
سنة 1950 قاس سلاح الجو الأمريكي $4\,000$ طيار في عشرة
أبعاد جسدية وبنى قمرة القيادة على مقاس \emph{الرجل \omterm{def:g10:stats:mean}{المتوسط}}. وقد
ناسبت --- كما اكتشف ملازم بالعدّ --- لا أحد
\omterm{def:g10:numbers:approx}{تقريبًا}. يلخّص الإحصاء الحشود بأعداد مفردة، وكل
تلخيص يقول الحقيقة عن شيء ويكذب عن شيء
آخر. تسوق هذه المسألة \omterm{def:g10:stats:mean}{المتوسط} \omterm{def:g10:stats:median}{والوسيط}
(\cref{def:g10:stats:mean} و \cref{def:g10:stats:median}) إلى المحكمة،
وتدع كلًّا منهما يُظهر ما يحسنه، وتغلق قضية قمرة القيادة.

\medskip
\noindent\textbf{الجزء الأول --- تلخيصان في قفص الاتهام.}
\begin{enumerate}
  \item يبيع شارع سبعة منازل، بآلاف اليورو:
        $200$ و $210$ و $220$ و $230$ و $240$ و $250$ --- وقصر
        واحد ثمنه $1\,200$. احسب السعر \omterm{def:g10:stats:mean}{المتوسط} والسعر
        \omterm{def:g10:stats:median}{الوسيط}. أي عدد يصف ``منزلًا في هذا الشارع''؟
  \item احذف القصر وأعد حساب الاثنين. بكم انتقل
        كل تلخيص؟ وبيّن المغزى بشأن القيم الشاذة.
  \item جدول إحصاء سكاني: من بين $100$ أسرة، $20$ بلا
        أطفال، و $25$ لها طفل واحد، و $30$ لها طفلان، و $15$ لها ثلاثة،
        و $10$ لها أربعة. احسب العدد \omterm{def:g10:stats:mean}{المتوسط} للأطفال. ولا
        أسرة ``لها $1.7$ طفل'' --- فأي مقدار حقيقي
        يشفّره \omterm{def:g10:stats:mean}{المتوسط} إذن؟ (اضربه في $100$.)
  \item من أجل العلامات الخمس عشرة
        $4, 6, 7, 8, 9, 10, 10, 11, 12, 12, 13, 14, 15, 17, 19$:
        أوجد \omterm{def:g10:stats:median}{الوسيط} والربيعيين $Q_1$ و $Q_3$
        (\cref{met:g10:stats:quartiles}).
  \item من أجل السلسلة نفسها، أعطِ \omterm{def:g10:stats:quartiles}{المدى} و
        \omterm{def:g10:stats:quartiles}{المدى الربيعي}. وماذا يقيس كل منهما؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- منحنيات التصحيح والأفواج المدمجة.}
\begin{enumerate}[resume]
  \item انطلاقًا من \cref{exo:g10:stats:9}: إزاحة كل قيمة بمقدار
        $+b$ تزيح \omterm{def:g10:stats:mean}{المتوسط} \omterm{def:g10:stats:median}{والوسيط} والربيعيين بمقدار $+b$؛ والضرب
        في $a$ يضربها في $a$. فماذا يحدث \omterm{def:g10:stats:quartiles}{للمدى} وللمدى
        \omterm{def:g10:stats:quartiles}{الربيعي} عند الإزاحة؟ وعند الضرب؟
        برّر.
  \item لاختبار \omterm{def:g10:stats:mean}{متوسط} $9$ (من $20$)، ويريد الأستاذ
        \omterm{def:g10:stats:mean}{متوسط} $12$. واقتُرح منحنيان للتصحيح: إضافة $3$ نقاط
        للجميع، أو ضرب كل علامة في $\frac43$. احسب
        مصير تلميذ عند $6$ وآخر عند $18$ في كل
        \omterm{def:g10:functions:graph}{منحنى}. أيهما يحابي مَن --- وأيهما فيه مشكلة سقف؟
  \item للفوج أ ($20$ تلميذًا) \omterm{def:g10:stats:mean}{متوسط} $12$؛ وللفوج ب ($30$
        تلميذًا) \omterm{def:g10:stats:mean}{متوسط} $10$. احسب \omterm{def:g10:stats:mean}{متوسط} التلاميذ $50$
        مجتمعين، وفسّر لماذا لا يساوي $11$
        (درس \omterm{def:g10:stats:mean}{المتوسط} المرجّح في مسألة نهاية الأسبوع عن \omterm{def:g10:stats:mean}{المتوسط} التوافقي في الكتاب السابق، في
        قاعة الدرس).
  \item بيّن أن الوسائط ترفض هذه اللعبة: احسب وسيطَي
        $A = \{1, 2, 3\}$ و $B = \{0, 10\}$، ثم
        \omterm{def:g10:stats:median}{وسيط} القائمة المدمجة، وقارن \omterm{def:g10:stats:mean}{بالمتوسط}
        المرجّح بالحجمين للوسيطين.
  \item التزلج الفني: يعطي خمسة حكام العلامات $5.2$ و $5.3$ و $5.4$ و
        $5.5$ و $9.9$. احسب \omterm{def:g10:stats:mean}{المتوسط} \omterm{def:g10:stats:median}{والوسيط} و
        \emph{\omterm{def:g10:stats:mean}{المتوسط} المشذّب} (احذف الأصغر والأكبر، وتوسّط
        الباقي). ولماذا تشذّب معظم الرياضات المحكَّمة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- قراءة التوزيعات.}
\begin{enumerate}[resume]
  \item ينتج مصنعان براغي \omterm{def:g10:stats:mean}{متوسط} طولها
        $50$ مم. المصنع أ: $Q_1 = 49.9$ و $Q_3 = 50.1$؛ والمصنع
        ب: $Q_1 = 48$ و $Q_3 = 52$. وأنت تحتاج إلى براغٍ قابلة
        للتبادل: فأي مصنع، وأي مؤشر حسم الأمر؟
  \item في عيادة، الانتظار \omterm{def:g10:stats:median}{الوسيط} $10$ دقائق لكن
        الانتظار \omterm{def:g10:stats:mean}{المتوسط} $25$. أي شكل من المعطيات يفضحه هذا
        الزوج؟ ابنِ مجموعة معطيات من خمس قيم لها هذا
        \omterm{def:g10:stats:median}{الوسيط} وهذا \omterm{def:g10:stats:mean}{المتوسط} بالضبط.
  \item ``طفلك في المئيني الخامس والسبعين في الطول'' ---
        ماذا تعني هذه الجملة؟ وهل يمكن أن يكون كل طفل في
        بلدة ``فوق \omterm{def:g10:stats:mean}{المتوسط}''؟ وفوق \omterm{def:g10:stats:median}{الوسيط}؟ فسّر
        الفرق.
  \item الفوج أ: الأصغر $4$، و $Q_1\,8$، \omterm{def:g10:stats:median}{والوسيط} $11$، و $Q_3\,13$، والأكبر
        $17$. والفوج ب: الأصغر $2$، و $Q_1\,9$، \omterm{def:g10:stats:median}{والوسيط} $11$، و $Q_3\,12$،
        والأكبر $19$. قارن: العلامة النموذجية، وتجانس
        النصف المركزي، وجموح القيم المتطرفة.
  \item قائمة الصحفي: يقول عنوان ``الراتب
        \omterm{def:g10:stats:mean}{المتوسط} في شركة التقنية: $5\,000$ يورو''. اذكر الأسئلة الثلاثة
        التي علّمتك هذه المسألة أن تطرحها قبل
        تصديق أن العنوان يقول شيئًا عن موظف نموذجي.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- قضية قمرة القيادة.}
\begin{enumerate}[resume]
  \item ابنِ دليلك بنفسك: كوّن خمسة أسعار منازل
        متوسطها $300$ ووسيطها $200$ (بآلاف اليورو).
  \item \omterm{def:g10:stats:mean}{المتوسط} هو نقطة التوازن: برهن انطلاقًا من التعريف
        على أن الانحرافات عن \omterm{def:g10:stats:mean}{المتوسط} يلغي بعضها بعضًا دائمًا،
        $\sum (x_i - \bar x) = 0$، وتحقق من ذلك على مجموعة معطياتك
        في السؤال 16.
  \item \omterm{def:g10:stats:median}{الوسيط} هو مقلّل المشاوير: تقع متاجر عند
        الكيلومترات $1$ و $3$ و $4$ و $9$ و $13$ من طريق، ويجب أن
        يقلّل مستودع \emph{المسافة الكلية} إلى
        المتاجر الخمسة. احسب تلك المسافة الكلية لمستودع عند
        \omterm{def:g10:stats:median}{الوسيط} ($4$)، وعند \omterm{def:g10:stats:mean}{المتوسط} ($6$)، وعند $5$. مَن يفوز؟
        (والأمر عام: \omterm{def:g10:stats:median}{فالوسيط} يقلّل مجموع المسافات المطلقة ---
        اختبر أي موضع آخر.)
  \item قمرة القيادة، محسومة: لنفترض أن لطيار احتمال $30\,\%$
        بأن يكون ``قريبًا من \omterm{def:g10:stats:mean}{المتوسط}'' في بُعد جسدي
        واحد، وذلك على نحو مستقل في $10$ أبعاد.
        قدّر احتمال أن يكون قريبًا من \omterm{def:g10:stats:mean}{المتوسط} في
        \emph{الأبعاد العشرة كلها} --- ووفّق بين النتيجة وما
        وجده الملازم في $4\,000$ طيار. ماذا بنى سلاح
        الجو بدل قمرة الرجل \omterm{def:g10:stats:mean}{المتوسط}؟
  \item الخاتمة --- دليل الاستعمال، سطر لكل حالة: حين تكون المجاميع
        هي المهمة (ميزانية، أو محصول)، استعمل \dots؛
        وحين يهم الفرد النموذجي (الرواتب، وأزمنة
        الانتظار)، استعمل \dots؛ وحين تهم عدالة التشتت،
        اذكر \dots؛ وحين يهدد التلوّث، استعمل
        \dots. أغلق القضية بعنوان المسألة.
\end{enumerate}
\end{problem}
