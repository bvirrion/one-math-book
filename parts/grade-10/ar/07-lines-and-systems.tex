\chapter{معادلات المستقيمات والجمل الخطية}\label{ch:g10:lines}

يوصف المستقيم في المستوي الإحداثي \omterm{def:g10:algebra:equation}{بمعادلة}، ويشكّل
مستقيمان معًا جملة معادلات حلولها هي نقط
تقاطعهما. يربط هذا الفصل بين ثلاث لغات --- الهندسية
(المستقيمات)، والجبرية (المعادلات)، والدالية (الدوال التآلفية في
\cref{ch:g10:reffunc}) --- ويعلّم الطريقتين المعتادتين لحل
\omterm{def:g10:lines:system}{جملة خطية}.

\section{معادلة مستقيم}

\begin{theorem}[المعادلة المختزلة]\label{thm:g10:lines:reduced}
في \omterm{def:g10:coordgeom:system}{نظام إحداثيات}:
\begin{enumerate}
  \item لكل مستقيم غير شاقولي \omterm{def:g10:algebra:equation}{معادلة} على الصورة
        \[
        y = mx + p ,
        \]
        حيث يتحدد $m$ (\emph{الميل}\index{ميل}) و $p$
        (\emph{\omterm{def:g10:reffunc:affine}{الترتيب عند المبدأ}}) تحديدًا وحيدًا --- وهذه هي
        \emph{المعادلة المختزلة}\index{معادلة مختزلة} للمستقيم؛
  \item ولكل مستقيم شاقولي \omterm{def:g10:algebra:equation}{معادلة} $x = c$؛
  \item وتقع نقطة على المستقيم بالضبط عندما تحقق إحداثياتها
        \omterm{def:g10:algebra:equation}{المعادلة}.
\end{enumerate}
\end{theorem}
\admitted

\begin{proposition}[الميل بدلالة نقطتين]\label{prop:g10:lines:slope}
\omterm{def:g10:reffunc:affine}{ميل} المستقيم (غير الشاقولي) المار بالنقطتين $A(x_A, y_A)$ و
$B(x_B, y_B)$، حيث $x_A \neq x_B$، هو
\[
m = \frac{y_B - y_A}{x_B - x_A}
\qquad
\text{(``تغير $y$ على تغير $x$'').}
\]
\end{proposition}

\begin{proof}
تحقق النقطتان $y = mx + p$، إذن $y_B - y_A = (mx_B + p) - (mx_A + p)
= m(x_B - x_A)$؛ اقسم على $x_B - x_A \neq 0$.
\end{proof}

\begin{method}[إيجاد معادلة مستقيم يمر بنقطتين]\label{met:g10:lines:through}
إذا أُعطيت $A$ و $B$ حيث $x_A \neq x_B$:
\begin{enumerate}
  \item احسب \omterm{def:g10:reffunc:affine}{الميل} $m = \dfrac{y_B - y_A}{x_B - x_A}$؛
  \item اكتب $y = mx + p$ وعوّض \omterm{def:g10:coordgeom:system}{إحداثيات} $A$ (أو $B$)
        لإيجاد $p$؛
  \item تحقق من \omterm{def:g10:algebra:equation}{المعادلة} النهائية بالنقطة \emph{الأخرى}.
\end{enumerate}
وإذا كان $x_A = x_B$، فالمستقيم شاقولي ومعادلته $x = x_A$.
\end{method}

\begin{example}\label{ex:g10:lines:through}
المستقيم المار بالنقطتين $A(1, 5)$ و $B(3, 1)$. \omterm{def:g10:reffunc:affine}{الميل}:
$m = \frac{1 - 5}{3 - 1} = \frac{-4}{2} = -2$. عندئذٍ $y = -2x + p$؛ وتعطي
النقطة $A$ أن $5 = -2 \times 1 + p$، إذن $p = 7$. \omterm{def:g10:algebra:equation}{والمعادلة}:
$y = -2x + 7$. تحقق بالنقطة $B$: $-2 \times 3 + 7 = 1$.
\end{example}

\begin{proposition}[المستقيمات المتوازية]\label{prop:g10:lines:parallel}
يكون مستقيمان غير شاقوليين متوازيين إذا وفقط إذا كان لهما \omterm{def:g10:reffunc:affine}{الميل}
نفسه. وليس لمستقيمين متوازيين متمايزين أي نقطة مشتركة؛ ولمستقيمين
مختلفي \omterm{def:g10:reffunc:affine}{الميل} نقطة مشتركة واحدة بالضبط.
\end{proposition}

\begin{proof}
ليكن $y = mx + p$ و $y = m'x + p'$ هما المستقيمين. النقطة المشتركة
تحل $mx + p = m'x + p'$، أي $(m - m')x = p' - p$. فإذا كان $m \neq m'$،
كان لهذه \omterm{def:g10:algebra:equation}{المعادلة} حل وحيد $x = \frac{p' - p}{m - m'}$: أي نقطة
\omterm{def:g10:numbers:interunion}{تقاطع} واحدة. وإذا كان $m = m'$، صارت \omterm{def:g10:algebra:equation}{المعادلة} $0 = p' - p$: فلا
حل عندما $p \neq p'$ (متوازيان متمايزان)، وكل $x$ حل عندما $p = p'$
(المستقيم نفسه). إذن التوازي (انعدام \omterm{def:g10:numbers:interunion}{التقاطع}، أو التطابق)
يكافئ $m = m'$.
\end{proof}

\begin{remark}[متجهة التوجيه]
يصعد المستقيم $y = mx + p$ بمقدار $m$ وحدة لكل وحدة نحو اليمين، إذن
\omterm{def:g10:vectors:vector}{المتجهة} $\vec u\,(1, m)$ هي \emph{متجهة
توجيه}\index{متجهة توجيه} للمستقيم: أي أن المستقيم يوازي
$\vec u$. ويكون مستقيمان متوازيين بالضبط عندما تكون متجهتا توجيههما
\omterm{def:g10:vectors:collinear}{مستقيميتين} (\cref{thm:g10:vectors:det}).
\end{remark}

\section{الجمل الخطية}

\begin{definition}[الجملة الخطية]\label{def:g10:lines:system}
\emph{جملة معادلتين خطيتين}\index{جملة خطية} بالمجهولين
$x$ و $y$ هي زوج من المعادلتين
\[
\begin{cases}
a x + b y = c \\
a' x + b' y = c'
\end{cases}
\]
يجب أن يتحققا \emph{في آنٍ واحد}. و\emph{الحل} هو زوج
$(x, y)$ يحقق المعادلتين معًا. وتصف كل \omterm{def:g10:algebra:equation}{معادلة} مستقيمًا، إذن حل
الجملة هو إيجاد \omterm{def:g10:numbers:interunion}{تقاطع} مستقيمين.
\end{definition}

\begin{method}[التعويض]\label{met:g10:lines:substitution}
\begin{enumerate}
  \item حل إحدى المعادلتين بالنسبة إلى أحد المجهولين --- واختر الأسهل، مثلًا
        عبّر عن $y$ بدلالة $x$؛
  \item عوّض هذه العبارة في \omterm{def:g10:algebra:equation}{المعادلة} \emph{الأخرى}، فتصير
        لا تحوي سوى $x$؛
  \item حل بالنسبة إلى $x$، ثم احسب $y$ من الخطوة 1؛
  \item تحقق من الزوج في المعادلتين الأصليتين معًا.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:lines:substitution}
حل $\begin{cases} x + 2y = 8 \\ 3x - y = 3 \end{cases}$ بالتعويض.
تعطي \omterm{def:g10:algebra:equation}{المعادلة} الثانية $y = 3x - 3$. عوّض في
الأولى:
\[
x + 2(3x - 3) = 8
\ \Longleftrightarrow\
x + 6x - 6 = 8
\ \Longleftrightarrow\
7x = 14
\ \Longleftrightarrow\
x = 2 ,
\]
ثم $y = 3 \times 2 - 3 = 3$. والحل هو الزوج $(2, 3)$.
تحقق: $2 + 2 \times 3 = 8$ و $3 \times 2 - 3 = 3$.
\end{example}

\begin{method}[الحذف]\label{met:g10:lines:elimination}
\begin{enumerate}
  \item اضرب كل \omterm{def:g10:algebra:equation}{معادلة} في عدد مختار جيدًا بحيث يصير لأحد المجهولين
        معاملان متقابلان في المعادلتين؛
  \item اجمع المعادلتين: فيختفي ذلك المجهول؛
  \item حل \omterm{def:g10:algebra:equation}{المعادلة} الباقية ذات المجهول الواحد، ثم عوّض للخلف
        لإيجاد المجهول الآخر؛
  \item تحقق من الزوج في المعادلتين الأصليتين معًا.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:lines:elimination}
حل $\begin{cases} 2x + 3y = 7 \\ 5x - 2y = 8 \end{cases}$ بالحذف.
لحذف $y$، اضرب \omterm{def:g10:algebra:equation}{المعادلة} الأولى في $2$ و
الثانية في $3$:
\[
\begin{cases}
4x + 6y = 14 \\
15x - 6y = 24
\end{cases}
\]
وبجمعهما: $19x = 38$، إذن $x = 2$. وبالتعويض في $2x + 3y = 7$:
$4 + 3y = 7$، إذن $y = 1$. والحل: $(2, 1)$. تحقق في \omterm{def:g10:algebra:equation}{المعادلة}
الثانية: $5 \times 2 - 2 \times 1 = 8$.
\end{example}

\begin{remark}[كم حلًّا؟]
مثل مستقيمين، \omterm{def:g10:lines:system}{للجملة الخطية} حل وحيد في الحالة العامة
(مستقيمان غير متوازيين)، وليس لها حل في حالة استثنائية (مستقيمان متوازيان متمايزان)، أو
لها عدد لا نهائي من الحلول (المستقيم نفسه مرتين). ويحسم محك
\cref{thm:g10:vectors:det} الأمر: فللجملة حل وحيد
بالضبط عندما $ab' - a'b \neq 0$.
\end{remark}

\begin{example}[النمذجة بجملة]\label{ex:g10:lines:modeling}
ثلاثة دفاتر وقلمان تكلّف $8$؛ ودفتر واحد وأربعة أقلام تكلّف
$6$. ليكن $x$ ثمن الدفتر و $y$ ثمن القلم:
\[
\begin{cases}
3x + 2y = 8 \\
x + 4y = 6 .
\end{cases}
\]
من \omterm{def:g10:algebra:equation}{المعادلة} الثانية، $x = 6 - 4y$؛ وبالتعويض،
$3(6 - 4y) + 2y = 8$، إذن $18 - 10y = 8$، فيكون $y = 1$ ثم
$x = 2$. إذن يكلّف الدفتر $2$ ويكلّف القلم $1$.
\end{example}

\section{مسألة: حل واحد، أو لا حل، أو عدد لا نهائي من الحلول}

\begin{problem}[{مسألة نهاية الأسبوع --- لمستقيمين ثلاثة مصائر
ممكنة، ومحدد يفصل بينها، وأقدم كتاب مدرسي في العالم
كان يعرف ذلك}]
\label{pb:g10:lines:1}
معادلتان خطيتان ومجهولان: زوج المستقيمات الذي ترسمانه
قد يتقاطع مرة، أو لا يتقاطع أبدًا، أو يكون مستقيمًا واحدًا ---
و\emph{كل} \omterm{def:g10:lines:system}{جملة خطية} ترث أحد هذه المصائر الثلاثة.
تصنّف هذه المسألة تلك المصائر، وتلتقي بالعدد الذي يحسمها (وهو
معرفة قديمة من فصل المتجهات)، وتحل مسألة
الدرّاج والأرانب التي يبلغ عمرها ألفي سنة في
كتاب \emph{الفصول التسعة}، وتنتهي بزوايا المناطق --- وهي
الخطوة الأولى في الاستمثال الذي تستعمله كل شركة طيران وكل مصنع
اليوم.

\medskip
\noindent\textbf{الجزء الأول --- المستقيم بطلاقة.}
\begin{enumerate}
  \item أوجد \omterm{thm:g10:lines:reduced}{المعادلة المختزلة} للمستقيم المار بالنقطتين
        $(1, 2)$ و $(3, 8)$
        (\cref{met:g10:lines:through}).
  \item من بين $y = 3x - 1$ و $y = 3x + 4$ و $y = -x + 7$: أي
        مستقيمين متوازيان
        (\cref{prop:g10:lines:parallel})، وأين تلتقي الأزواج
        غير المتوازية؟ (احسب تقاطعًا واحدًا.)
  \item المستقيم المار بالنقطتين $(2, 1)$ و $(2, 5)$: لماذا ليست له
        \omterm{thm:g10:lines:reduced}{معادلة مختزلة} $y = mx + p$، وما
        معادلته؟
  \item هل النقطة $(4, 11)$ على المستقيم $y = 3x - 1$؟ وماذا عن
        $(-1, -5)$؟
  \item المستقيم المار بالنقطة $(0, 3)$ وميله $-2$: أعطِ
        معادلته، ونقطتي تقاطعه مع المحورين، ومساحة
        المثلث الذي يقتطعه من الربع الأول.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- ثلاثة مصائر.}
\begin{enumerate}[resume]
  \item حل بالتعويض
        (\cref{met:g10:lines:substitution}):
        $\begin{cases} y = 2x - 3 \\ 3x + y = 7 \end{cases}$
  \item حل بالحذف
        (\cref{met:g10:lines:elimination}):
        $\begin{cases} 2x + 3y = 8 \\ 5x - 3y = -1 \end{cases}$
  \item صنّف --- وفسّر بالمستقيمات --- الجملتين
        \[
        \begin{cases} 2x - y = 3 \\ 4x - 2y = 6 \end{cases}
        \qquad\text{و}\qquad
        \begin{cases} 2x - y = 3 \\ 4x - 2y = 1 . \end{cases}
        \]
        وبيّن الثلاثية الكاملة: \omterm{def:g10:numbers:interunion}{تقاطع}، أو توازٍ، أو
        تطابق --- حل واحد، أو لا حل، أو عدد لا نهائي.
  \item من أجل الجملة العامة $ax + by = e$ و $cx + dy = f$،
        يكون العدد الحاسم هو $ad - bc$ --- وهو المحدد
        نفسه في اختبار الاستقامية في
        \cref{pb:g10:vectors:1}. فسّر هذا التوافق (أي
        متجهتين تكونان \omterm{def:g10:vectors:collinear}{مستقيميتين} بالضبط عندما يكون المستقيمان
        متوازيين أو متطابقين؟)، واحسب $ad - bc$ من أجل
        الجمل الثلاث في السؤالين 7 و 8.
  \item من أجل أي قيمة للعدد $m$ لا يكون للجملة
        $\begin{cases} x + my = 3 \\ 2x + 6y = 7 \end{cases}$
        حل؟ وحل الجملة من أجل كل القيم الأخرى
        للعدد $m$\,\dots\ أو فسّر على الأقل كيف تفعل ذلك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الفصول التسعة.}
\begin{enumerate}[resume]
  \item من \emph{الفصول التسعة في فن الرياضيات}
        (الصين، قبل نحو 2000 سنة): قفص يحوي دراجًا
        وأرانب --- $35$ رأسًا و $94$ رجلًا إجمالًا. كم
        من كل نوع؟
  \item يمزج كشك عصير شرابًا فاكهته $60\,\%$ مع
        مشروب فاكهته $20\,\%$ للحصول على $30$ لترًا فاكهته
        $35\,\%$. كم لترًا من كل منهما؟
  \item عدد من رقمين مجموع رقميه $12$؛ وتبديل
        رقميه ينقصه $18$. أوجده، مستعملًا جبر الأرقام
        في مسألة نهاية الأسبوع عن جبر الأرقام في الكتاب السابق لتكوين الجملة.
  \item في المخبز، تكلّف $3$ قهوات و $2$ كرواسان
        $9.10$ يورو؛ وتكلّف $2$ قهوة و $3$ كرواسان
        $8.90$ يورو. حل بأناقة: ماذا يخبرك \emph{جمع}
        المعادلتين و\emph{طرحهما} مباشرةً؟
  \item يدّعي بائع في سوق: $2$ تفاحة $+ 1$ إجاصة بسعر
        $5$ يورو، و $4$ تفاحات $+ 2$ إجاصة بسعر $9$ يورو.
        حل --- أو بالأحرى فسّر ما يكشفه مصير الجملة
        عن حساب البائع.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الزوايا تحكم.}
\begin{enumerate}[resume]
  \item يقسم المستقيم $y = 3x - 1$ المستوي قسمين. في أي
        جهة يقع المبدأ؟ صف المجموعة
        $y > 3x - 1$ وكيف تختبر نقطة إزاءها.
  \item ارسم المنطقة المعرَّفة بالقيود الأربعة
        \[
        x \geq 0, \qquad y \geq 0, \qquad x + y \leq 4,
        \qquad y \leq 2x + 1,
        \]
        واحسب زواياها الأربع.
  \item ربح ورشة هو $P = 3x + 2y$ على منطقة
        السؤال 17. احسب $P$ عند الزوايا الأربع. وبقبول
        أن \omterm{def:g10:functions:extrema}{القيمة العظمى} لمقدار خطي كهذا على منطقة كهذه
        تُبلغ دائمًا عند زاوية (تخيّل المستقيمات $3x + 2y = c$، من أجل $c$ ثابت، وهي تكنس المنطقة)، أعطِ خطة
        الإنتاج المثلى.
  \item سيارتا الأجرة في مسألة نهاية الأسبوع عن الميزانين الحراريين في الكتاب السابق على صورة جملة:
        اكتب وحل
        $\begin{cases} y = 1.5x + 3 \\ y = 2x \end{cases}$ و
        فسّر الحل.
  \item الخاتمة، وهي القاموس: \omterm{def:g10:algebra:equation}{معادلة} واحدة $\leftrightarrow$
        مستقيم واحد؛ وجملة واحدة $\leftrightarrow$ مستقيمان لهما
        ثلاثة مصائر، يحسمها $ad - bc$؛ ومتراجحات كثيرة
        $\leftrightarrow$ منطقة تحمل زواياها
        القيمة المثلى. صُغ ذلك في ثلاث جمل --- فقد جُلت للتوّ
        في بهو \emph{الجبر الخطي} و\emph{البرمجة
        الخطية}، وكلاهما يُبنى كاملًا في الكتب الجامعية.
\end{enumerate}
\end{problem}
