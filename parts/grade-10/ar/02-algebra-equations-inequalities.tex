\chapter{الجبر: المعادلات والمتراجحات}\label{ch:g10:algebra}

الجبر فن الحساب بالحروف. يراجع هذا الفصل الحركتين الأساسيتين ---
\omterm{def:g10:algebra:expand}{النشر} \omterm{def:g10:algebra:expand}{والتحليل إلى عوامل} --- ويصقلهما، ثم يستعملهما في حل المعادلات
والمتراجحات على نحو منهجي، \omterm{def:g10:algebra:signtable}{وجدول الإشارة} هو الأداة المركزية فيه.

\section{النشر والتحليل إلى عوامل}

\begin{definition}[النشر والتحليل إلى عوامل]\label{def:g10:algebra:expand}
\emph{نشرُ}\index{نشر} عبارةٍ هو تحويل الجداءات إلى
مجاميع؛ و\emph{تحليلُها إلى عوامل}\index{تحليل إلى عوامل} هو العكس،
أي تحويل المجموع إلى جداء. والقاعدة الأساسية هي التوزيعية:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g10:algebra:expand}
انشر $(2x + 3)(x - 5)$ خطوة خطوة:
\begin{align*}
(2x + 3)(x - 5)
&= 2x \times x + 2x \times (-5) + 3 \times x + 3 \times (-5) \\
&= 2x^2 - 10x + 3x - 15 \\
&= 2x^2 - 7x - 15 .
\end{align*}
\end{example}

\begin{theorem}[المتطابقات الشهيرة]\label{thm:g10:algebra:identities}
من أجل كل عددين حقيقيين $a$ و $b$:
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
كل واحدة منها \omterm{def:g10:algebra:expand}{نشر} مباشر. فمن أجل الأولى:
\[
(a+b)^2 = (a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2 .
\]
وتنتج الثانية بتعويض $-b$ عن $b$: $(a-b)^2 = a^2 + 2a(-b) +
(-b)^2 = a^2 - 2ab + b^2$. ومن أجل الثالثة:
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$.
\end{proof}

\begin{example}\label{ex:g10:algebra:identities}
مستعملة طردًا (نشرًا) وعكسًا (تحليلًا إلى عوامل):
\begin{align*}
(3x + 2)^2 &= 9x^2 + 12x + 4,
& x^2 - 25 &= (x+5)(x-5), \\
(x - 4)^2 &= x^2 - 8x + 16,
& 4x^2 + 4x + 1 &= (2x + 1)^2 .
\end{align*}
والحساب الذهني يستفيد منها أيضًا: $101 \times 99 = (100+1)(100-1) =
10000 - 1 = 9999$.
\end{example}

\begin{method}[تحليل عبارة إلى عوامل]\label{met:g10:algebra:factor}
جرّب، بهذا الترتيب:
\begin{enumerate}
  \item \emph{العامل المشترك:} ابحث عن عامل موجود في كل الحدود
        وأخرجه، مثلًا $6x^2 + 4x = 2x(3x + 2)$. وقد يكون العامل المشترك
        قوسًا كاملًا: $(x+1)(2x-3) + (x+1)(x+7) = (x+1)(3x+4)$.
  \item \emph{المتطابقة الشهيرة:} تعرّف على $a^2 - b^2$، أو على
        $a^2 \pm 2ab + b^2$، مثلًا $9x^2 - 16 = (3x)^2 - 4^2 =
        (3x+4)(3x-4)$.
  \item ازدواج الطريقتين، وتحقق من النتيجة بنشرها من جديد.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:factorsteps}
حلّل $E = (2x + 1)^2 - (x - 3)^2$ إلى عوامل. إنه فرق مربعين
$a^2 - b^2$ حيث $a = 2x+1$ و $b = x-3$:
\begin{align*}
E &= \bigl[(2x+1) + (x-3)\bigr] \times \bigl[(2x+1) - (x-3)\bigr] \\
  &= (3x - 2)(2x + 1 - x + 3) \\
  &= (3x - 2)(x + 4).
\end{align*}
انتبه إلى القوس الثاني: طرح $x - 3$ يعني طرح $x$
\emph{وإضافة} $3$.
\end{example}

\section{المعادلات}

\begin{definition}[المعادلة، الحل]\label{def:g10:algebra:equation}
\emph{المعادلة}\index{معادلة} مساواة تحوي عددًا مجهولًا
$x$؛ و\emph{الحل} قيمة للمجهول $x$ تجعل المساواة
صحيحة. وحل المعادلة هو إيجاد \emph{كل} حلولها.
\end{definition}

تكون معادلتان \emph{متكافئتين} إذا كان لهما المجموعة نفسها من
الحلول. وإضافة العدد نفسه إلى الطرفين، أو ضرب الطرفين
في العدد \emph{غير المعدوم} نفسه، تعطي \omterm{def:g10:algebra:equation}{معادلة} مكافئة.

\begin{example}[معادلة من الدرجة الأولى]\label{ex:g10:algebra:firstdegree}
حل $5x - 7 = 2x + 8$، حركةً حركةً:
\begin{align*}
5x - 7 &= 2x + 8 \\
5x - 2x &= 8 + 7 && \text{(نضيف $7 - 2x$ إلى الطرفين)} \\
3x &= 15 \\
x &= 5 && \text{(نقسم الطرفين على $3$).}
\end{align*}
الحل الوحيد هو $5$. تحقق: $5 \times 5 - 7 = 18$ و
$2 \times 5 + 8 = 18$.
\end{example}

\begin{theorem}[قاعدة الجداء المعدوم]\label{thm:g10:algebra:zeroproduct}
يكون جداء عددين حقيقيين معدومًا إذا وفقط إذا كان أحد
عامليه على الأقل معدومًا:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ أو } \ B = 0 .
\]
\end{theorem}

\begin{proof}
إذا كان $A = 0$ أو $B = 0$، فالجداء $0$ بداهةً. وبالعكس، لنفترض أن
$AB = 0$ و $A \neq 0$. بقسمة الطرفين على $A$ نجد $B = 0$.
\end{proof}

\begin{method}[الحل بالتحليل إلى عوامل]\label{met:g10:algebra:zeroproduct}
لحل \omterm{def:g10:algebra:equation}{معادلة} طرفها الأيمن $0$ ويمكن تحليل طرفها الأيسر
إلى عوامل:
\begin{enumerate}
  \item انقل كل شيء إلى اليسار حتى تصير \omterm{def:g10:algebra:equation}{المعادلة} $E = 0$؛
  \item حلّل $E$ إلى جداء عوامل من الدرجة الأولى؛
  \item طبّق قاعدة الجداء المعدوم وحل كل \omterm{def:g10:algebra:equation}{معادلة} صغيرة؛
  \item اجمع كل الحلول التي وجدتها.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:zeroproduct}
حل $(x - 2)(3x + 6) = 0$: إما $x - 2 = 0$ فيكون $x = 2$، وإما
$3x + 6 = 0$ فيكون $x = -2$. الحلول: $-2$ و $2$.

حل $x^2 = 9x$. \emph{لا} تقسم على $x$ (فقد يكون معدومًا!)؛ انقل
كل شيء إلى طرف واحد وحلّل إلى عوامل:
\[
x^2 - 9x = 0
\quad\Longleftrightarrow\quad
x(x - 9) = 0
\quad\Longleftrightarrow\quad
x = 0 \ \text{ أو } \ x = 9 .
\]
\end{example}

\begin{example}[معادلة ذات خارج قسمة]\label{ex:g10:algebra:quotient}
حل $\dfrac{x - 1}{x + 2} = 0$. يكون الخارج معدومًا بالضبط عندما يكون
بسطه معدومًا \emph{ومقامه} غير معدوم. هنا $x - 1 = 0$
يعطي $x = 1$، و $1 + 2 = 3 \neq 0$: فالحل الوحيد هو $1$. أما القيمة
$x = -2$ فهي \emph{ممنوعة} (قسمة على صفر) ويجب استبعادها
دائمًا في البداية.
\end{example}

\section{المتراجحات وجداول الإشارة}

\begin{proposition}[قواعد المتراجحات]\label{prop:g10:algebra:ineqrules}
ليكن $a \leq b$.
\begin{enumerate}
  \item من أجل أي $c$: $a + c \leq b + c$ (الإضافة تحفظ الترتيب).
  \item إذا كان $c > 0$: $ac \leq bc$ (الضرب في عدد موجب
        يحفظ الترتيب).
  \item إذا كان $c < 0$: $ac \geq bc$ (الضرب في عدد سالب
        \emph{يقلب} الترتيب).
\end{enumerate}
\end{proposition}

\begin{proof}
1. $(b + c) - (a + c) = b - a \geq 0$.
2. $bc - ac = (b-a)c$ جداء عددين غير سالبين، إذن
$bc \geq ac$.
3. وهنا يكون $(b-a)c \leq 0$ لأنه جداء عدد غير سالب في عدد
سالب، إذن $bc \leq ac$.
\end{proof}

\begin{example}\label{ex:g10:algebra:firstineq}
حل $-2x + 3 > 9$:
\begin{align*}
-2x + 3 &> 9 \\
-2x &> 6 && \text{(نطرح 3)} \\
x &< -3 && \text{(نقسم على $-2 < 0$: ينقلب اتجاه المتراجحة).}
\end{align*}
مجموعة الحلول هي $\intoo{-\infty}{-3}$.
\end{example}

\begin{definition}[جدول الإشارة]\label{def:g10:algebra:signtable}
\emph{جدول الإشارة}\index{جدول الإشارة} يسجل، في سطر لكل عامل،
إشارة ($+$ أو $-$) كل عامل من عوامل عبارة عندما يمسح $x$ المجموعة $\R$،
مع $0$ عند كل قيمة ينعدم عندها العامل؛ ويعطي سطر أخير
إشارة الجداء، بتطبيق قاعدة الإشارات عمودًا عمودًا.
\end{definition}

\begin{example}\label{ex:g10:algebra:signtable}
إشارة $P(x) = (x - 2)(3 - x)$. ينعدم العامل $x - 2$ عند $2$ ويكون
موجبًا بعدها؛ وينعدم العامل $3 - x$ عند $3$ ويكون موجبًا قبلها:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccc}
$x$ & $-\infty$ \quad & $2$ & & $3$ & \quad $+\infty$ \\
\hline
$x - 2$ & $-$ & $0$ & $+$ & & $+$ \\
$3 - x$ & $+$ & & $+$ & $0$ & $-$ \\
\hline
$P(x)$ & $-$ & $0$ & $+$ & $0$ & $-$ \\
\end{tabular}
\end{center}
إذن $P(x) > 0$ على $\intoo{2}{3}$ بالضبط، و $P(x) \leq 0$ على
$\intoc{-\infty}{2} \cup \intco{3}{+\infty}$.
\end{example}

\begin{method}[حل متراجحة بجدول إشارة]\label{met:g10:algebra:signtable}
لحل متراجحة مثل $E(x) \geq 0$ أو $E(x) < 0$:
\begin{enumerate}
  \item انقل كل شيء إلى الطرف الأيسر حتى يصير الطرف الأيمن
        $0$ (ولا تضرب الطرفين أبدًا في عبارة إشارتها
        مجهولة)؛
  \item حلّل الطرف الأيسر إلى عوامل، أو وحّد مقاماته إذا
        كان خارج قسمة؛
  \item ابنِ \omterm{def:g10:algebra:signtable}{جدول الإشارة} بسطر لكل عامل (وفي حالة خارج القسمة، علّم
        القيم الممنوعة بشرطة مزدوجة في السطر الأخير)؛
  \item اقرأ الفترات التي تتحقق فيها الإشارة المطلوبة، متحققًا
        من إدراج كل طرف أو استبعاده.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:quotientineq}
حل $\dfrac{x + 1}{x - 3} \leq 0$. ينعدم البسط عند $-1$، وينعدم
المقام عند $3$ (قيمة ممنوعة). ويكون الخارج سالبًا حين تكون
إشارتاهما مختلفتين، وذلك من أجل $x$ المحصور بين $-1$ و $3$.
وبإدراج الطرف الذي ينعدم عنده \emph{البسط} واستبعاد
القيمة الممنوعة: مجموعة الحلول هي $\intco{-1}{3}$.
\end{example}

\section{مسألة: من بابل إلى النسبة الذهبية}

\begin{problem}[{مسألة نهاية الأسبوع --- مسائل المجموع والجداء،
وإتمام المربع كما رسمه الخوارزمي، والمعادلة
$x^2 = x + 1$}]
\label{pb:g10:algebra:1}
قبل أربعة آلاف سنة، كان كتبة بابل يحلون هذه المسألة على الطين:
\emph{أوجد عددين تعرف مجموعهما وجداءهما}. وحيلتهم ---
البحث عن العددين متناظرين حول نصف المجموع --- ما زالت
تفكّ كل \omterm{def:g10:algebra:equation}{معادلة} من الدرجة الثانية اليوم، تحت اسمها
الحديث: \emph{إتمام المربع}. تتعلم هذه المسألة الحيلة،
وتشاهد الخوارزمي يتم مربعًا بالمعنى الحرفي،
وتشغّل الطريقة على أشهر النسب جميعًا.

\medskip
\noindent\textbf{الجزء الأول --- المسألة البابلية.}
\begin{enumerate}
  \item تحمية (\cref{thm:g10:algebra:identities}): حلّل
        $x^2 - 10x + 25$ إلى عوامل، ثم $9x^2 - 49$، ثم --- بمزج
        متطابقة بفرق مربعين ---
        $x^2 + 6x + 9 - 4$.
  \item حل $(x + 1)(x + 5) = 0$ و
        $(3x - 7)(3x + 7) = 0$
        (\cref{thm:g10:algebra:zeroproduct}).
  \item مسألة الكاتب: عددان مجموعهما $10$
        وجداؤهما $21$. على منوال بابل، اكتبهما $5 - t$
        و $5 + t$ (متناظرين حول نصف المجموع) وأوجد $t$،
        ثم العددين.
  \item الطريقة نفسها: المجموع $14$ والجداء $33$.
  \item الوصفة العامة: من أجل مجموع $s$ وجداء $p$، بيّن
        أن العددين هما
        \[
        \frac s2 - t
        \quad\text{و}\quad
        \frac s2 + t,
        \qquad\text{حيث}\quad
        t^2 = \left(\frac s2\right)^2 - p ,
        \]
        وبيّن الشرط على $s$ و $p$ ليكون للمسألة
        حل.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- إتمام المربع.}
\begin{enumerate}[resume]
  \item انشر $(x - a)(x - b)$ واستنتج: حلول
        $x^2 - sx + p = 0$ هي بالضبط أزواج الأعداد ذات
        المجموع $s$ والجداء $p$. (كان الكتبة يحلون
        معادلات من الدرجة الثانية دون أن يدروا.)
  \item تحقق من أن $x^2 - 10x + 21 = (x - 5)^2 - 4$، ثم حل
        $x^2 - 10x + 21 = 0$ بتحليل الطرف الأيمن على صورة
        فرق مربعين. قارن بالسؤال 3.
  \item حل $x^2 + 6x - 7 = 0$ بإتمام المربع.
  \item أتمّ المربع في $x^2 + 4x + 5$، وفسّر لماذا
        \emph{ليس} \omterm{def:g10:algebra:equation}{للمعادلة} $x^2 + 4x + 5 = 0$ حل
        حقيقي. وبيّن المحك العام: متى تنعدم عبارة من الدرجة
        الثانية بعد كتابتها على الصورة
        $(x + h)^2 + k$؟
  \item الاسم حرفي. في بغداد القرن التاسع،
        رسم الخوارزمي $x^2 + 6x = 7$ مربعًا ضلعه $x$
        ألصق بضلعين منه مستطيلين بعداهما $3 \times x$،
        ثم \emph{أتمّ المربع} بالزاوية الناقصة
        $3 \times 3$. ارسم الشكل وفسّر كيف
        يحوّل \omterm{def:g10:algebra:equation}{المعادلة} إلى $(x + 3)^2 = 16$ --- ثم حل.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الإشارات والجداول ووعدان
قديمان.}
\begin{enumerate}[resume]
  \item ابنِ جدول إشارة $(x - 3)(x + 2)$
        (\cref{def:g10:algebra:signtable}) وحل
        $(x - 3)(x + 2) < 0$.
  \item حل $x^2 \leq 10x - 21$ (انقل كل شيء إلى طرف واحد
        وأعد استعمال السؤال 7).
  \item حل $\dfrac{x - 1}{x + 2} \geq 0$ بجدول إشارة،
        منتبهًا إلى القيمة الممنوعة.
  \item وعد قديم يُوفى: بالنسبة إلى فلّاح
        مسألة سياج الملكة ديدو في الكتاب السابق، ومعه $20$ م من السياج، فإن مساحة
        حظيرة بعداها $x \times (10 - x)$ تحقق
        \[
        A(x) = x(10 - x) = 25 - (x - 5)^2 .
        \]
        تحقق من المتطابقة، واقرأ فيها \emph{برهانًا} على
        ما لم يكن ممكنًا من قبل سوى تخمينه: المساحة العظمى،
        وقيمة $x$ الوحيدة التي تبلغها.
  \item ووعد آخر: برهن على أن $x + \dfrac1x \geq 2$ من أجل كل
        $x > 0$، مع تحقق المساواة عند $x = 1$ فقط. (اضرب في
        $x$، واجمع الحدود، وتعرّف على مربع --- إنها
        متراجحة المتوسطين الحسابي والهندسي في
        مسألة نهاية الأسبوع عن علاقات إقليدس في الكتاب السابق، في حلة جديدة.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- المقطع الذهبي.}
اقطع قطعة طولها $1$ إلى جزء كبير $x$ وجزء صغير
$1 - x$ بحيث تكون
\emph{نسبة الكل إلى الكبير كنسبة الكبير إلى الصغير}:
$\frac{1}{x} = \frac{x}{1 - x}$.
\begin{enumerate}[resume]
  \item بيّن أن الشرط يُكتب $x^2 + x - 1 = 0$.
  \item حل بإتمام المربع، واحتفظ بالجذر الذي
        يصلح طولًا: $x = \frac{\sqrt5 - 1}{2} \approx 0.618$.
  \item \emph{النسبة الذهبية} هي
        $\varphi = \frac1x = \frac{2}{\sqrt5 - 1}$. اجعل المقام
        ناطقًا (اضرب البسط والمقام في
        $\sqrt5 + 1$، وهي متطابقة في العمل) وبيّن أن
        $\varphi = \frac{\sqrt5 + 1}{2} \approx 1.618$.
  \item تحقق \emph{بالضبط} من متطابقتَي
        $\varphi$ السحريتين:
        \[
        \varphi^2 = \varphi + 1,
        \qquad
        \frac{1}{\varphi} = \varphi - 1 .
        \]
  \item الخاتمة: فسّر لماذا $\varphi$ عدد أصمّ
        (تعطي مسألة نهاية الأسبوع عن الصمم في الكتاب السابق الواقعة الأساسية عن
        $\sqrt5$)، ثم احسب النسب $\frac53$ و
        $\frac85$ و $\frac{13}{8}$ و $\frac{21}{13}$ بين
        أعداد هيماشندرا--فيبوناتشي المتتالية
        (مسألة نهاية الأسبوع عن عدّ الأوزان في الكتاب السابق) بثلاثة أرقام عشرية. ماذا
        يبدو أنها تطارد؟ (والبرهان على أنها تدركه
        يخص فصول المتتاليات.)
\end{enumerate}
\end{problem}
