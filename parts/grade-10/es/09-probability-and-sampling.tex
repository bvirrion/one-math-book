\chapter{Probabilidad y muestreo}\label{ch:g10:proba}

\omterm{def:g10:proba:distribution}{probabilidad} asigna números al azar. A partir de experimentos con
resultados igualmente probables --- dados, monedas, cartas --- este capítulo construye el
reglas básicas para calcular \omterm{def:g10:proba:distribution}{probabilidades} de sucesos, presenta \omterm{met:g10:proba:tree}{árbol}
diagramas para experimentos de dos pasos, y termina con el vínculo entre
\omterm{def:g10:proba:distribution}{probabilidades} y \omterm{def:g10:stats:series}{frecuencias} observadas. La teoría continúa en
\cref{ch:g11:prob}.

\section{Experimentos y eventos aleatorios.}

\begin{definition}[Resultados y eventos]\label{def:g10:proba:events}
A \emph{experimento aleatorio}\index{experimento aleatorio} es un experimento cuyo
El resultado no se puede predecir (tirar un dado). El conjunto de todos los posibles.
resultados es el \emph{espacio muestral}\index{espacio muestral}
$\Omega = \{\omega_1, \dots, \omega_n\}$; sus elementos son los
\emph{resultados}. Un \emph{evento}\index{evento} es un conjunto de resultados; un
evento \emph{ocurre} cuando el resultado del experimento le pertenece.
\end{definition}

\begin{definition}[Distribución de probabilidad]\label{def:g10:proba:distribution}
Asignar a cada resultado $\omega_i $ un numero $ p_i \geq 0$, con
$ p_1 + \dots + p_n = 1$, define un \emph{probabilidad}\index{probabilidad}:
la probabilidad $\P(A)$ de un suceso $ A $ es la suma de $ p_i $ de la
resultados en $ A $. En particular $\P(\Omega) = 1$ y lo imposible suceso
$\varnothing $ tiene probabilidad $0$.
\end{definition}

\begin{proposition}[Resultados igualmente probables]\label{prop:g10:proba:uniform}
si el $ n $ los resultados son igualmente probables (cada $ p_i = \frac1n $), entonces para
cada suceso $ A $:
\[
\P(A) = \frac{\text{number of outcomes in } A}{n}
= \frac{\text{favorable cases}}{\text{possible cases}} .
\]
\end{proposition}

\begin{proof}
$ A $ contiene algún número $ k $ de resultados, cada uno de \omterm{def:g10:proba:distribution}{probabilidad} $\frac1n $,
entonces $\P(A) = \frac kn $.
\end{proof}

\begin{example}\label{ex:g10:proba:die}
Tira un dado justo: $\Omega = \{1, 2, 3, 4, 5, 6\}$, teniendo cada resultado
\omterm{def:g10:proba:distribution}{probabilidad} $\frac16$. El suceso $ A $: ``el resultado es par'' es
$ A = \{2, 4, 6\}$, entonces $\P(A) = \frac36 = \frac12$. El suceso $ B $: ``el
el resultado es al menos $5$'' es $ B = \{5, 6\}$, entonces $\P(B) = \frac26 =
\frac13$.
\end{example}

\section{Combinando eventos}

\begin{definition}[Intersección, unión, complemento.]\label{def:g10:proba:operations}
Sea $ A $ y $ B $ ser sucesos.
\begin{itemize}
  \item $ A \cap B $ (``$ A $ y $ B $'') ocurre cuando ocurren ambas cosas;
  \item $ A \cup B $ (``$ A $ o $ B $'') ocurre cuando ocurre al menos uno;
  \item el \emph{complemento}\index{complemento} $\bar A $ (``no $ A $'')
        ocurre exactamente cuando $ A $ no lo hace;
  \item $ A $ y $ B $ son \emph{incompatible}\index{incompatible} cuando no pueden ocurrir
        juntos: $ A \cap B = \varnothing $.
\end{itemize}
\end{definition}

\begin{theorem}[Reglas de suma]\label{thm:g10:proba:addition}
A pesar de sucesos $ A $ y $ B $:
\[
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B),
\qquad
\P(\bar A) = 1 - \P(A) .
\]
Para \omterm{def:g10:proba:operations}{incompatibles} sucesos, la primera regla se simplifica a
$\P(A \cup B) = \P(A) + \P(B)$.
\end{theorem}

\begin{proof}
sumando $\P(A) + \P(B)$ agrega el \omterm{def:g10:proba:distribution}{probabilidad} de cada resultado de
$ A \cup B $ una vez, \emph{excepto} los de $ A \cap B $, que se cuentan
dos veces --- una vez en $ A $ y una vez en $ B $. Restando $\P(A \cap B)$ reparaciones
el doble cómputo. Para el \omterm{def:g10:proba:operations}{complemento}: $ A $ y $\bar A $ son \omterm{def:g10:proba:operations}{incompatibles}
y $ A \cup \bar A = \Omega $, entonces $\P(A) + \P(\bar A) = \P(\Omega) = 1$.
\end{proof}

\begin{example}\label{ex:g10:proba:cards}
Roba una carta de un estándar. $52$-baraja de cartas. Sea $ A $: ``la tarjeta es una
corazón '' ($13$ tarjetas) y $ B $: ``la carta es un rey'' ($4$ tarjetas). Entonces
$ A \cap B $ es ``el rey de corazones'' ($1$ tarjeta), y
\[
\P(A \cup B) = \frac{13}{52} + \frac{4}{52} - \frac{1}{52} = \frac{16}{52}
= \frac{4}{13}.
\]
El \omterm{def:g10:proba:operations}{complemento} La regla es a menudo la ruta más rápida: el \omterm{def:g10:proba:distribution}{probabilidad} que el
la tarjeta es \emph{no} un corazón es $1 - \frac{13}{52} = \frac34$.
\end{example}

\section{Diagramas de árbol}

\begin{method}[Experimentos de dos pasos]\label{met:g10:proba:tree}
Para un experimento realizado en dos pasos:
\begin{enumerate}
  \item dibujar un \emph{árbol}\index{árbol}: una sucursal por posible
        resultado del primer paso, luego de cada uno de estos, una rama por
        resultado del segundo paso, escribir el \omterm{def:g10:proba:distribution}{probabilidad} en cada rama
        (las ramas que salen de cualquier nodo deben sumar $1$);
  \item el \omterm{def:g10:proba:distribution}{probabilidad} de un \emph{camino} (una hoja de la \omterm{met:g10:proba:tree}{árbol}) es el
        producto de las \omterm{def:g10:proba:distribution}{probabilidades} a lo largo de sus ramas;
  \item el \omterm{def:g10:proba:distribution}{probabilidad} de un suceso es la suma de las \omterm{def:g10:proba:distribution}{probabilidades} de
        los caminos que lo realizan.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:proba:tree}
Una urna contiene $3$ bolas rojas y $2$ bolas azules. Saca una bola, ponla.
Atrás, dibuja de nuevo. Cada sorteo da rojo (R) con \omterm{def:g10:proba:distribution}{probabilidad} $\frac35$ y
azul (B) con \omterm{def:g10:proba:distribution}{probabilidad} $\frac25$.

El \omterm{def:g10:proba:distribution}{probabilidad} de sacar dos bolas del mismo color es
$\P(\text{RR}) + \P(\text{BB}) = \frac{9}{25} + \frac{4}{25} =
\frac{13}{25}$; el \omterm{def:g10:proba:distribution}{probabilidad} de al menos un azul es
$1 - \P(\text{RR}) = 1 - \frac{9}{25} = \frac{16}{25}$ (\omterm{def:g10:proba:operations}{complemento} regla).
\end{example}

\begin{remark}
Sin reemplazo, las \omterm{def:g10:proba:distribution}{probabilidades} del segundo empate dependen de la
primero: con la urna de arriba, después de sacar una bola roja, solo $2$ rojo y $2$
el azul permanece, por lo que la segunda rama lleva $\frac24$ y $\frac24$. El
\omterm{met:g10:proba:tree}{árbol} el método funciona sin cambios; solo los números en las sucursales de segundo nivel
cambiar.
\end{remark}

\section{Frecuencias y muestreo}

cuando un \omterm{def:g10:proba:events}{experimento aleatorio} se repite muchas veces, lo observado \omterm{def:g10:stats:series}{frecuencia}
de un suceso se acerca a su \omterm{def:g10:proba:distribution}{probabilidad} --- esta es la razón \omterm{def:g10:proba:distribution}{probabilidad} es
útil para describir el mundo real.

\begin{definition}[Muestra]\label{def:g10:proba:sample}
A \emph{muestra}\index{muestra} de tamaño $ n $ es el resultado de repetir el
mismo experimento $ n $ veces de forma independiente. El \emph{observado \omterm{def:g10:stats:series}{frecuencia}} de
un suceso en la muestra es el número de ocurrencias dividido por $ n $.
\end{definition}

\begin{example}\label{ex:g10:proba:sampling}
Una moneda justa da cara con \omterm{def:g10:proba:distribution}{probabilidad} $0.5$. tirándolo $100$ veces
rara vez da exactamente $50$ cabezas: \omterm{def:g10:proba:sample}{muestras} \emph{fluctuar}. Ejecuciones típicas de
$100$ los lanzamientos dan \omterm{def:g10:stats:series}{frecuencias} como $0.46$, $0.53$, $0.49$ --- cerca de
$0.5$, pero no iguales. Más grande \omterm{def:g10:proba:sample}{muestras} fluctuar menos: con $10\,000$
lanzamientos, las \omterm{def:g10:stats:series}{frecuencias} observadas generalmente están dentro de aproximadamente $0.01$ de $0.5$.
\end{example}

\begin{remark}[Intervalo de fluctuación]
Una regla práctica útil, justificada en \cref{ch:g11:binom}: para un \omterm{def:g10:proba:sample}{muestra} de
tamaño $ n $ de un suceso de \omterm{def:g10:proba:distribution}{probabilidad} $ p $ (con $ n $ razonablemente grande), el
observado \omterm{def:g10:stats:series}{frecuencia} se encuentra en el \omterm{def:g10:numbers:interval}{intervalo}
\[
\intcc{p - \frac{1}{\sqrt n}}{\,p + \frac{1}{\sqrt n}}
\]
en aproximadamente $95\%$ de \omterm{def:g10:proba:sample}{muestras}. Si se observa \omterm{def:g10:stats:series}{frecuencia} cae muy afuera
esto \omterm{def:g10:numbers:interval}{intervalo}, dudar del valor de $ p $: este es el punto de partida de
pruebas estadísticas.
\end{remark}

\begin{example}\label{ex:g10:proba:test}
Una fábrica afirma que sólo $10\%$ de sus piezas están defectuosas
($ p = 0.1$). en un \omterm{def:g10:proba:sample}{muestra} de $ n = 400$ regiones, $64$ están defectuosos: observados
\omterm{def:g10:stats:series}{frecuencia} $\frac{64}{400} = 0.16$. la fluctuación \omterm{def:g10:numbers:interval}{intervalo} es
$\intcc{0.1 - \frac{1}{20}}{0.1 + \frac{1}{20}} = \intcc{0.05}{0.15}$,
y $0.16$ cae fuera de él: el \omterm{def:g10:proba:sample}{muestra} arroja serias dudas sobre la afirmación.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g10:proba:1}
Se lanza un dado justo. Calcular el \omterm{def:g10:proba:distribution}{probabilidad} de cada uno suceso: ``obtener un
$6$''; ``obteniendo un número impar''; ``obteniendo como \omterm{def:g10:functions:extrema}{máximo} $4$''; ``obtener un
$7$''.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g10:proba:2}
una bolsa contiene $5$ verde, $3$ amarillo y $2$ canicas negras; una canica es
extraído al azar. Calcular el \omterm{def:g10:proba:distribution}{probabilidad} que sea verde; que no lo es
negro; que sea verde o amarillo.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g10:proba:3}
en una clase de $30$ estudiantes, $18$ estudiar español, $10$ estudiar alemán y
$4$ estudiar ambos. Se elige un estudiante al azar. Usando la regla de la suma,
calcular el \omterm{def:g10:proba:distribution}{probabilidad} que este estudiante estudie al menos uno de los dos
idiomas, entonces el \omterm{def:g10:proba:distribution}{probabilidad} que no estudien ninguno de los dos.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g10:proba:4}
Se lanza dos veces una moneda justa. Dibuja el \omterm{met:g10:proba:tree}{árbol} del experimento y calcular
el \omterm{def:g10:proba:distribution}{probabilidad} de sacar dos cabezas; exactamente una cabeza; al menos uno
cabeza.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g10:proba:5}
Una urna contiene $4$ rojo y $6$ bolas blancas. Se extraen dos bolas una después
el otro \emph{sin} reemplazo. Dibuja el \omterm{met:g10:proba:tree}{árbol} con lo correcto
\omterm{def:g10:proba:distribution}{probabilidades} en el segundo nivel y calcular las \omterm{def:g10:proba:distribution}{probabilidad} de dibujo
dos bolas rojas, luego de sacar dos bolas de diferentes colores.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g10:proba:6}
Se lanzan dos dados justos y se registra su suma.
\begin{enumerate}
  \item Explique por qué el \omterm{def:g10:proba:events}{espacio muestral} puede tomarse como el $36$ igualmente
        pares probables $(1,1), (1,2), \dots, (6,6)$.
  \item Calcular el \omterm{def:g10:proba:distribution}{probabilidad} que la suma es $7$; que es $12$;
        que es como mucho $4$.
  \item ¿Qué suma es la más probable?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g10:proba:7}
Una pregunta de opción múltiple ofrece $4$ respuestas, una correcta. un estudiante
respuestas $2$ tales preguntas independientes son completamente aleatorias. Calcular el
\omterm{def:g10:proba:distribution}{probabilidad} de responder ambas correctamente, exactamente una correctamente, ninguna
correctamente. Comprueba que las tres \omterm{def:g10:proba:distribution}{probabilidades} suman $1$.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g10:proba:8}
Un político afirma $50\%$ aprobación. en una encuesta de $ n = 900$ al azar
votantes elegidos, $396$ aprobar.
\begin{enumerate}
  \item Calcular lo observado \omterm{def:g10:stats:series}{frecuencia} y la fluctuación \omterm{def:g10:numbers:interval}{intervalo}
        alrededor $ p = 0.5$ para $ n = 900$.
  \item ¿La encuesta arroja dudas sobre esta afirmación?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g10:proba:9}
Un juego cuesta $2$ para jugar: tiras un dado justo y ganas $6$ si tiras un
$6$, $3$ si tiras un $5$, nada más.
\begin{enumerate}
  \item Calcular el \omterm{def:g10:proba:distribution}{probabilidad} de cada cantidad ganada ($6$, $3$, $0$).
  \item Encima $600$ juegos, ¿aproximadamente cuántos de cada resultado esperas?
        Calcule las ganancias y pérdidas totales esperadas: ¿vale la pena el juego?
        jugando?
\end{enumerate}
\end{exercise}

\section{Problema: Las tres puertas que engañaron a mil
matemáticos}

\begin{problem}[{Problema del fin de semana --- la paradoja de Monty Hall y
sus primos: cómo la información cambia la probabilidad y cómo las muestras
separar afirmaciones de ilusiones}]
\label{pb:g10:proba:1}
En 1990, la columnista Marilyn vos Savant publicó un acertijo sobre
un presentador de concurso, tres puertas, un coche y dos cabras --- y
Respondió correctamente. Diez mil lectores, entre ellos cientos
de doctorados en matemáticas, escribió para insistir en que estaba equivocada. esto
El problema te arma \omterm{met:g10:proba:tree}{árboles} (\cref{met:g10:proba:tree}),
\omterm{def:g10:proba:operations}{complementos} y las reglas de suma
(\cref{thm:g10:proba:addition}), le permite resolver el argumento
usted mismo, y termina con la herramienta que arbitra cada uno de esos
disputa en la práctica: la fluctuación \omterm{def:g10:proba:sample}{muestra}.

\medskip
\noindent\textbf{Part I --- The rules, warmed up.}
\begin{enumerate}
  \item Una tarjeta de un estándar $52$-baraja de cartas: calcular
        $ P(\text{king or heart})$ con la regla de la suma ---
        y decir por qué $\frac{4}{52} + \frac{13}{52}$ solo es
        mal.
  \item Tres lanzamientos de moneda justos: cálculo $ P(\text{at least one
        head})$ por el \omterm{def:g10:proba:operations}{complemento}.
  \item una bolsa contiene $3$ rojo y $2$ canicas azules; dos estan dibujados
        sin reemplazo. con un \omterm{met:g10:proba:tree}{árbol}, calcular
        $ P(\text{two reds})$ y $ P(\text{one of each color})$.
  \item Completa la distribución de la pregunta 3 con
        $ P(\text{two blues})$ y comprueba las tres \omterm{def:g10:proba:distribution}{probabilidades}
        suma a $1$.
  \item Alguien informa, para dos. \emph{\omterm{def:g10:proba:operations}{incompatibles}} sucesos,
        $ P(A) = 0.6$ y $ P(B) = 0.5$. Condenarlos con uno
        línea de aritmética.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The three doors.}
El juego: un coche se esconde detrás de una de las tres puertas, detrás unas cabras
los otros dos. Eliges una puerta (digamos la puerta 1). El anfitrión --- quien
\emph{sabe} donde esta el auto --- abre uno de los otros dos
puertas, siempre revelando una cabra, y te ofrece la oportunidad de
cambie a la puerta cerrada restante.
\begin{enumerate}[resume]
  \item Antes de calcular algo: ¿el cambio ayuda, perjudica o
        no importa? Escriba su respuesta instintiva, honestamente.
  \item Ahora el \omterm{met:g10:proba:tree}{árbol}, por casos sobre la verdadera posición del coche
        (cada uno $\frac13$): si el coche está detrás de tu puerta,
        el cambio pierde; Si está detrás de cualquiera de las otras puertas, ¿qué
        ¿Qué debe hacer el anfitrión y qué gana entonces el cambio?
        Calcular $ P(\text{win by switching})$.
  \item La versión de una sola línea: cambiar gana exactamente cuando su
        La primera elección fue incorrecta. Concluir de nuevo.
  \item Ahora la variante crucial: ``Monty Fall'': el anfitrión
        hace \emph{no} saber dónde está el coche, abre uno de los
        Otras puertas al azar y por suerte revela una cabra. Lista
        los seis escenarios igualmente probables (posición del coche $\times $
        puerta abierta), descartar aquellas en las que se revela el coche,
        y calcular el \omterm{def:g10:proba:distribution}{probabilidad} ese cambio gana
        \emph{entre los escenarios supervivientes}. ¿Qué cambió y
        ¿Por qué es importante el conocimiento del anfitrión?
  \item La bomba de la intuición: cien puertas, eliges una, la
        sabiendo que el anfitrión se abre $98$ puertas de cabra. ¿Cambiar? Con qué
        \omterm{def:g10:proba:distribution}{probabilidad} de ganar?
  \item Te enfrentas a un escéptico que, como los mil que escriben cartas,
        insiste en "50--50". Describir una simulación basada en tarjetas.
        (tres cartas, muchas rondas, un amigo como anfitrión) y di
        ¿Qué proporción de victorias en el cambio esperas después? $300$
        rondas --- con la fluctuación \omterm{def:g10:numbers:interval}{intervalo} del capitulo
        como su margen.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Cousins of the paradox.}
\begin{enumerate}[resume]
  \item Una familia tiene dos hijos; aprendes eso \emph{al menos
        uno es un niño}. Enumere los dos hijos igualmente probables
        configuraciones compatibles con la información, y
        calcular $ P(\text{two boys})$.
  \item Misma familia, pero la información ahora es: \emph{el mayor
        el niño es un niño}. Vuelva a calcular. ¿Por qué las dos respuestas?
        Se diferencian y ¿qué comparten con los dos Monty?
  \item Tres monedas justas: calcular $ P(\text{all three alike})$.
        El matemático d'Alembert defendió en 1754 otra
        respuesta: ``o son todos iguales o no lo son ---
        un caso de cada dos''. Nombra el error exacto, con
        \cref{prop:g10:proba:uniform} en la mano.
  \item Un juego cuesta $2$ euros: tira un dado, gana $6$ euros en un
        seis, $3$ euros por cinco, nada más
        (\cref{exo:g10:proba:9}). Encima $600$ juegos, que
        ¿Qué ganancia promedio por juego predice el conteo? un
        aseguradora juega las mismas matemáticas: un $1\,\%$ anualmente
        posibilidad de un $10\,000$-reclamo en euros --- ¿cuál es el
        \emph{justo} prima, y ¿por qué la prima real
        ¿superarlo?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- \omterm{def:g10:proba:sample}{muestras} as arbiters.}
\begin{enumerate}[resume]
  \item Se lanza una moneda $100$ veces. Usando la fluctuación
        \omterm{def:g10:numbers:interval}{intervalo} $ p \pm \frac{1}{\sqrt n}$, decide cuál de
        Estos resultados deberían sorprender: $55$ cabezas; $70$
        cabezas.
  \item una encuesta de $1\,000$ los votantes dan un candidato $52\,\%$.
        Calcular la fluctuación \omterm{def:g10:numbers:interval}{intervalo} alrededor $50\,\%$ y
        decir si la encuesta \emph{prueba} el candidato es
        adelante.
  \item Una fábrica afirma $4\,\%$ piezas defectuosas. A \omterm{def:g10:proba:sample}{muestra} de
        $400$ \omterm{def:g10:proba:sample}{muestra} $28$ defectuoso ($7\,\%$); un mas tarde \omterm{def:g10:proba:sample}{muestra} de
        $2\,500$ \omterm{def:g10:proba:sample}{muestra} $175$ ($7\,\%$ de nuevo). Pruebe el reclamo
        contra cada uno \omterm{def:g10:proba:sample}{muestra}. ¿Por qué difieren los veredictos?
  \item Dos fuentes de error acechan en todas las encuestas: el muestreo
        \emph{inclinación} (el problema del fin de semana de cómo se encuentran los datos del volumen de la escuela secundaria) y el muestreo
        \emph{fluctuación} (este capítulo). ¿Cuál se encoge como
        $ n $ crece, y cuál sobrevive a cualquier \omterm{def:g10:proba:sample}{muestra} ¿tamaño? uno
        frase cada uno.
  \item Final --- la lista de verificación del probabilista, una línea por
        ítem: resultados igualmente probables antes de contar
        (pregunta 15); \omterm{met:g10:proba:tree}{árboles} y \omterm{def:g10:proba:operations}{complementos} para varios pasos y
        ``al menos'' (preguntas 2--3); \emph{información
        cambia las \omterm{def:g10:proba:distribution}{probabilidades}, y cómo se obtuvo importa}
        (preguntas 7--14); \omterm{def:g10:proba:sample}{muestras} fluctúan aproximadamente
        $\frac{1}{\sqrt n}$ y arbitrajes de simulación
        (preguntas 11, 16--18).
\end{enumerate}
\end{problem}
