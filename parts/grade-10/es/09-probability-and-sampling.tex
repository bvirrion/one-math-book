\chapter{Probabilidad y muestreo}\label{ch:g10:proba}

La \omterm{def:g10:proba:distribution}{probabilidad} pone números al azar. Partiendo de experimentos con
resultados equiprobables (dados, monedas, cartas), este capítulo construye
las reglas básicas para calcular \omterm{def:g10:proba:distribution}{probabilidades} de \omterm{def:g10:proba:events}{sucesos}, introduce los
\omterm{met:g10:proba:tree}{diagramas de árbol} para experimentos en dos etapas y termina con el
vínculo entre \omterm{def:g10:proba:distribution}{probabilidades} y frecuencias observadas. La teoría continúa
en el \cref{ch:g11:prob}.

\section{Experimentos aleatorios y sucesos}

\begin{definition}[Resultados y sucesos]\label{def:g10:proba:events}
Un \emph{experimento aleatorio}\index{experimento aleatorio} es un
experimento cuyo resultado no se puede predecir (lanzar un dado). El
conjunto de todos los resultados posibles es el \emph{espacio
muestral}\index{espacio muestral} $\Omega = \{\omega_1, \dots, \omega_n\}$;
sus elementos son los \emph{resultados}. Un \emph{suceso}\index{suceso} es
un conjunto de resultados; un suceso \emph{ocurre} cuando el resultado del
experimento le pertenece.
\end{definition}

\begin{definition}[Distribución de probabilidad]\label{def:g10:proba:distribution}
Asignar a cada resultado $\omega_i$ un número $p_i \geq 0$, con
$p_1 + \dots + p_n = 1$, define una
\emph{probabilidad}\index{probabilidad}: la probabilidad $\P(A)$ de un
\omterm{def:g10:proba:events}{suceso} $A$ es la suma de los $p_i$ de los resultados que hay en $A$. En
particular, $\P(\Omega) = 1$ y el \omterm{def:g10:proba:events}{suceso} imposible $\varnothing$ tiene
probabilidad $0$.
\end{definition}

\begin{proposition}[Resultados equiprobables]\label{prop:g10:proba:uniform}
Si los $n$ resultados son equiprobables (cada $p_i = \frac1n$), entonces
para todo \omterm{def:g10:proba:events}{suceso} $A$:
\[
\P(A) = \frac{\text{número de resultados de } A}{n}
= \frac{\text{casos favorables}}{\text{casos posibles}} .
\]
\end{proposition}

\begin{proof}
$A$ contiene un cierto número $k$ de resultados, cada uno de \omterm{def:g10:proba:distribution}{probabilidad}
$\frac1n$, luego $\P(A) = \frac kn$.
\end{proof}

\begin{example}\label{ex:g10:proba:die}
Lanzamos un dado no trucado: $\Omega = \{1, 2, 3, 4, 5, 6\}$, y cada
resultado tiene \omterm{def:g10:proba:distribution}{probabilidad} $\frac16$. El \omterm{def:g10:proba:events}{suceso} $A$: ``el resultado es
par'' es $A = \{2, 4, 6\}$, luego $\P(A) = \frac36 = \frac12$. El \omterm{def:g10:proba:events}{suceso}
$B$: ``el resultado es al menos $5$'' es $B = \{5, 6\}$, luego
$\P(B) = \frac26 = \frac13$.
\end{example}

\section{Combinar sucesos}

\begin{definition}[Intersección, unión, suceso contrario]\label{def:g10:proba:operations}
Sean $A$ y $B$ dos \omterm{def:g10:proba:events}{sucesos}.
\begin{itemize}
  \item $A \cap B$ (``$A$ y $B$'') ocurre cuando ocurren los dos;
  \item $A \cup B$ (``$A$ o $B$'') ocurre cuando ocurre al menos uno;
  \item el \emph{suceso contrario}\index{suceso contrario} $\bar A$ (``no
        $A$'') ocurre exactamente cuando no ocurre $A$;
  \item $A$ y $B$ son \emph{incompatibles}\index{sucesos incompatibles}
        cuando no pueden ocurrir a la vez: $A \cap B = \varnothing$.
\end{itemize}
\end{definition}

\begin{theorem}[Reglas de la suma]\label{thm:g10:proba:addition}
Para todos los \omterm{def:g10:proba:events}{sucesos} $A$ y $B$:
\[
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B),
\qquad
\P(\bar A) = 1 - \P(A) .
\]
Para \omterm{def:g10:proba:operations}{sucesos incompatibles}, la primera regla se simplifica a
$\P(A \cup B) = \P(A) + \P(B)$.
\end{theorem}

\begin{proof}
Al sumar $\P(A) + \P(B)$ se añade una vez la \omterm{def:g10:proba:distribution}{probabilidad} de cada
resultado de $A \cup B$, \emph{salvo} los de $A \cap B$, que se cuentan
dos veces: una en $A$ y otra en $B$. Restar $\P(A \cap B)$ repara ese
doble recuento. Para el \omterm{def:g10:proba:operations}{suceso contrario}: $A$ y $\bar A$ son
\omterm{def:g10:proba:operations}{incompatibles} y $A \cup \bar A = \Omega$, luego
$\P(A) + \P(\bar A) = \P(\Omega) = 1$.
\end{proof}

regla de la suma.}
\end{omfigure}

\begin{example}\label{ex:g10:proba:cards}
Sacamos una carta de una baraja de $52$ cartas. Sea $A$: ``la carta es de
corazones'' ($13$ cartas) y $B$: ``la carta es un rey'' ($4$ cartas).
Entonces $A \cap B$ es ``el rey de corazones'' ($1$ carta), y
\[
\P(A \cup B) = \frac{13}{52} + \frac{4}{52} - \frac{1}{52} = \frac{16}{52}
= \frac{4}{13}.
\]
La regla del \omterm{def:g10:proba:operations}{suceso contrario} suele ser el camino más rápido: la
\omterm{def:g10:proba:distribution}{probabilidad} de que la carta \emph{no} sea de corazones es
$1 - \frac{13}{52} = \frac34$.
\end{example}

\section{Diagramas de árbol}

\begin{method}[Experimentos en dos etapas]\label{met:g10:proba:tree}
Para un experimento que se realiza en dos etapas:
\begin{enumerate}
  \item dibuja un \emph{árbol}\index{diagrama de árbol}: una rama por cada
        resultado posible de la primera etapa y, desde cada una de ellas,
        una rama por cada resultado de la segunda, escribiendo la
        \omterm{def:g10:proba:distribution}{probabilidad} sobre cada rama (las ramas que salen de un mismo nudo
        deben sumar $1$);
  \item la \omterm{def:g10:proba:distribution}{probabilidad} de un \emph{camino} (una hoja del \omterm{met:g10:proba:tree}{árbol}) es el
        producto de las \omterm{def:g10:proba:distribution}{probabilidades} de sus ramas;
  \item la \omterm{def:g10:proba:distribution}{probabilidad} de un \omterm{def:g10:proba:events}{suceso} es la suma de las \omterm{def:g10:proba:distribution}{probabilidades} de
        los caminos que lo realizan.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:proba:tree}
Una urna contiene $3$ bolas rojas y $2$ azules. Sacamos una bola, la
devolvemos y sacamos otra. Cada extracción da roja (R) con \omterm{def:g10:proba:distribution}{probabilidad}
$\frac35$ y azul (A) con \omterm{def:g10:proba:distribution}{probabilidad} $\frac25$.

\omterm{def:g10:proba:distribution}{probabilidades} de los caminos suman $1$.}
\end{omfigure}

La \omterm{def:g10:proba:distribution}{probabilidad} de obtener dos bolas del mismo color es
$\P(\text{RR}) + \P(\text{AA}) = \frac{9}{25} + \frac{4}{25} =
\frac{13}{25}$; la \omterm{def:g10:proba:distribution}{probabilidad} de obtener al menos una azul es
$1 - \P(\text{RR}) = 1 - \frac{9}{25} = \frac{16}{25}$ (regla del \omterm{def:g10:proba:operations}{suceso
contrario}).
\end{example}

\begin{remark}
Sin devolución, las \omterm{def:g10:proba:distribution}{probabilidades} de la segunda extracción dependen de la
primera: con la urna anterior, después de sacar una bola roja solo quedan
$2$ rojas y $2$ azules, así que las ramas de la segunda etapa llevan
$\frac24$ y $\frac24$. El método del \omterm{met:g10:proba:tree}{árbol} funciona igual; solo cambian
los números de las ramas del segundo nivel.
\end{remark}

\section{Frecuencias y muestreo}

Cuando un \omterm{def:g10:proba:events}{experimento aleatorio} se repite muchas veces, la frecuencia
observada de un \omterm{def:g10:proba:events}{suceso} se acerca a su \omterm{def:g10:proba:distribution}{probabilidad}; por eso la
\omterm{def:g10:proba:distribution}{probabilidad} sirve para describir el mundo real.

\begin{definition}[Muestra]\label{def:g10:proba:sample}
Una \emph{muestra}\index{muestra} de tamaño $n$ es el resultado de repetir
$n$ veces el mismo experimento de forma independiente. La
\emph{frecuencia observada} de un \omterm{def:g10:proba:events}{suceso} en la muestra es el número de
veces que ocurre dividido entre $n$.
\end{definition}

\begin{example}\label{ex:g10:proba:sampling}
Una moneda no trucada da cara con \omterm{def:g10:proba:distribution}{probabilidad} $0.5$. Lanzarla $100$ veces
rara vez da exactamente $50$ caras: las \omterm{def:g10:proba:sample}{muestras} \emph{fluctúan}. Series
típicas de $100$ lanzamientos dan frecuencias como $0.46$, $0.53$ o
$0.49$: cerca de $0.5$, pero no iguales. Las \omterm{def:g10:proba:sample}{muestras} grandes fluctúan
menos: con $10\,000$ lanzamientos, las frecuencias observadas suelen estar
a menos de $0.01$ de $0.5$.
\end{example}

\begin{remark}[Intervalo de fluctuación]
Una regla práctica útil, que se justifica en el \cref{ch:g11:binom}: para
una \omterm{def:g10:proba:sample}{muestra} de tamaño $n$ de un \omterm{def:g10:proba:events}{suceso} de \omterm{def:g10:proba:distribution}{probabilidad} $p$ (con $n$
razonablemente grande), la frecuencia observada está en el \omterm{def:g10:numbers:interval}{intervalo}
\[
\intcc{p - \frac{1}{\sqrt n}}{\,p + \frac{1}{\sqrt n}}
\]
en aproximadamente el $95\,\%$ de las \omterm{def:g10:proba:sample}{muestras}. Si una frecuencia
observada cae muy fuera de este \omterm{def:g10:numbers:interval}{intervalo}, duda del valor de $p$: ese es
el punto de partida de los contrastes estadísticos.
\end{remark}

\begin{example}\label{ex:g10:proba:test}
Una fábrica afirma que solo el $10\,\%$ de sus piezas son defectuosas
($p = 0.1$). En una \omterm{def:g10:proba:sample}{muestra} de $n = 400$ piezas hay $64$ defectuosas:
frecuencia observada $\frac{64}{400} = 0.16$. El \omterm{def:g10:numbers:interval}{intervalo} de fluctuación
es $\intcc{0.1 - \frac{1}{20}}{0.1 + \frac{1}{20}} = \intcc{0.05}{0.15}$,
y $0.16$ cae fuera: la \omterm{def:g10:proba:sample}{muestra} pone seriamente en duda la afirmación.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:proba:1}
Se lanza un dado no trucado. Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de cada \omterm{def:g10:proba:events}{suceso}:
``sacar un $6$''; ``sacar un número impar''; ``sacar como mucho $4$'';
``sacar un $7$''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:2}
Una bolsa contiene $5$ canicas verdes, $3$ amarillas y $2$ negras; se saca
una canica al azar. Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de que sea verde; de que no
sea negra; de que sea verde o amarilla.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:3}
En una clase de $30$ alumnos, $18$ estudian inglés, $10$ estudian alemán y
$4$ estudian los dos idiomas. Se elige un alumno al azar. Con la regla de
la suma, calcula la \omterm{def:g10:proba:distribution}{probabilidad} de que estudie al menos uno de los dos
idiomas y después la de que no estudie ninguno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:4}
Se lanza dos veces una moneda no trucada. Dibuja el \omterm{met:g10:proba:tree}{árbol} del experimento
y calcula la \omterm{def:g10:proba:distribution}{probabilidad} de sacar dos caras; exactamente una cara; al
menos una cara.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:5}
Una urna contiene $4$ bolas rojas y $6$ blancas. Se sacan dos bolas, una
tras otra y \emph{sin} devolución. Dibuja el \omterm{met:g10:proba:tree}{árbol} con las \omterm{def:g10:proba:distribution}{probabilidades}
correctas en el segundo nivel y calcula la \omterm{def:g10:proba:distribution}{probabilidad} de sacar dos bolas
rojas y después la de sacar dos bolas de colores distintos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:6}
Se lanzan dos dados no trucados y se anota su suma.
\begin{enumerate}
  \item Explica por qué se puede tomar como \omterm{def:g10:proba:events}{espacio muestral} las $36$
        parejas equiprobables $(1,1), (1,2), \dots, (6,6)$.
  \item Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de que la suma sea $7$; de que sea $12$;
        de que sea como mucho $4$.
  \item ¿Cuál es la suma más probable?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:7}
Una pregunta de tipo test ofrece $4$ respuestas, una correcta. Un alumno
responde completamente al azar a $2$ preguntas independientes de este
tipo. Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de acertar las dos, de acertar exactamente
una y de no acertar ninguna. Comprueba que las tres \omterm{def:g10:proba:distribution}{probabilidades} suman
$1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:8}
Un político afirma tener un $50\,\%$ de aprobación. En una encuesta a
$n = 900$ votantes elegidos al azar, $396$ lo aprueban.
\begin{enumerate}
  \item Calcula la frecuencia observada y el \omterm{def:g10:numbers:interval}{intervalo} de fluctuación
        alrededor de $p = 0.5$ para $n = 900$.
  \item ¿Pone la encuesta en duda la afirmación?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:proba:9}
Jugar cuesta $2$: lanzas un dado no trucado y ganas $6$ si sacas un $6$,
$3$ si sacas un $5$ y nada en los demás casos.
\begin{enumerate}
  \item Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de cada cantidad ganada ($6$, $3$, $0$).
  \item A lo largo de $600$ partidas, ¿cuántas veces esperas cada
        resultado? Calcula la ganancia y la pérdida totales esperadas:
        ¿merece la pena jugar?
\end{enumerate}
\end{exercise}

\section{Problema: Las tres puertas que engañaron a mil matemáticos}

\begin{problem}[{Problema de fin de semana --- la paradoja de Monty Hall
y sus primas: cómo la información cambia la probabilidad y cómo las
muestras separan las afirmaciones de los buenos deseos}]
\label{pb:g10:proba:1}
En 1990, la columnista Marilyn vos Savant publicó un acertijo sobre un
presentador de concursos, tres puertas, un coche y dos cabras, y lo
respondió correctamente. Diez mil lectores, entre ellos cientos de
doctores en matemáticas, le escribieron para insistir en que se
equivocaba. Este problema te arma con \omterm{met:g10:proba:tree}{árboles}
(\cref{met:g10:proba:tree}), \omterm{def:g10:proba:operations}{sucesos contrarios} y las reglas de la suma
(\cref{thm:g10:proba:addition}), te deja zanjar tú mismo la discusión y
termina con la herramienta que arbitra en la práctica todas las disputas
de este tipo: la \omterm{def:g10:proba:sample}{muestra} que fluctúa.

\medskip
\noindent\textbf{Parte I --- Las reglas, en calentamiento.}
\begin{enumerate}
  \item Una carta de una baraja de $52$: calcula
        $P(\text{rey o corazones})$ con la regla de la suma y di por qué
        $\frac{4}{52} + \frac{13}{52}$ a secas está mal.
  \item Tres lanzamientos de una moneda no trucada: calcula
        $P(\text{al menos una cara})$ mediante el \omterm{def:g10:proba:operations}{suceso contrario}.
  \item Una bolsa tiene $3$ canicas rojas y $2$ azules; se sacan dos sin
        devolución. Con un \omterm{met:g10:proba:tree}{árbol}, calcula $P(\text{dos rojas})$ y
        $P(\text{una de cada color})$.
  \item Completa la distribución de la pregunta 3 con
        $P(\text{dos azules})$ y comprueba que las tres \omterm{def:g10:proba:distribution}{probabilidades}
        suman $1$.
  \item Alguien informa, para dos \omterm{def:g10:proba:events}{sucesos} \emph{\omterm{def:g10:proba:operations}{incompatibles}}, de que
        $P(A) = 0.6$ y $P(B) = 0.5$. Desmóntalo con una línea de
        aritmética.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Las tres puertas.}
El juego: un coche se esconde tras una de tres puertas y hay cabras tras
las otras dos. Eliges una puerta (digamos la 1). El presentador, que
\emph{sabe} dónde está el coche, abre una de las otras dos puertas y
siempre enseña una cabra, y te ofrece la oportunidad de cambiar a la
puerta cerrada que queda.
\begin{enumerate}[resume]
  \item Antes de calcular nada: ¿cambiar ayuda, perjudica o da igual?
        Anota tu respuesta instintiva, con honestidad.
  \item Ahora el \omterm{met:g10:proba:tree}{árbol}, por casos según la posición real del coche (cada
        una con \omterm{def:g10:proba:distribution}{probabilidad} $\frac13$): si el coche está tras tu puerta,
        cambiar pierde; si está tras cualquiera de las otras dos, ¿qué
        tiene que hacer el presentador y qué gana entonces cambiar?
        Calcula $P(\text{ganar cambiando})$.
  \item La versión de una línea: cambiar gana exactamente cuando tu
        primera elección fue equivocada. Concluye de nuevo.
  \item Ahora la variante crucial, el ``Monty distraído'': el presentador
        \emph{no} sabe dónde está el coche, abre una de las otras puertas
        al azar y, por suerte, enseña una cabra. Enumera los seis
        escenarios equiprobables (posición del coche $\times$ puerta
        abierta), descarta aquellos en los que se descubre el coche y
        calcula la \omterm{def:g10:proba:distribution}{probabilidad} de ganar cambiando \emph{entre los
        escenarios supervivientes}. ¿Qué ha cambiado y por qué importa lo
        que sabe el presentador?
  \item El acelerador de intuición: cien puertas, eliges una y el
        presentador, que sabe, abre $98$ puertas con cabra. ¿Cambias? ¿Con
        qué \omterm{def:g10:proba:distribution}{probabilidad} de ganar?
  \item Te enfrentas a un escéptico que, como los mil autores de cartas,
        insiste en el ``50--50''. Describe una simulación con cartas (tres
        cartas, muchas rondas y un amigo de presentador) y di qué
        proporción de victorias al cambiar esperas al cabo de $300$
        rondas, tomando como margen el \omterm{def:g10:numbers:interval}{intervalo} de fluctuación del
        capítulo.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Primas de la paradoja.}
\begin{enumerate}[resume]
  \item Una familia tiene dos hijos; te enteras de que \emph{al menos uno
        es un chico}. Enumera las configuraciones equiprobables de dos
        hijos compatibles con esa información y calcula
        $P(\text{dos chicos})$.
  \item La misma familia, pero ahora la información es: \emph{el hijo
        mayor es un chico}. Vuelve a calcular. ¿Por qué difieren las dos
        respuestas y qué tienen en común con los dos Monty?
  \item Tres monedas no trucadas: calcula $P(\text{las tres iguales})$. El
        matemático d'Alembert defendía en 1754 otra respuesta: ``o son
        todas iguales o no lo son: un caso de dos''. Nombra el error
        exacto, con la \cref{prop:g10:proba:uniform} en la mano.
  \item Jugar cuesta $2$ euros: lanzas un dado y ganas $6$ euros con un
        seis, $3$ euros con un cinco y nada en los demás casos
        (\cref{exo:g10:proba:9}). A lo largo de $600$ partidas, ¿qué
        ganancia \omterm{def:g10:stats:mean}{media} por partida predice el recuento? Una aseguradora
        hace las mismas matemáticas: un $1\,\%$ de \omterm{def:g10:proba:distribution}{probabilidad} anual de
        un siniestro de $10\,000$ euros. ¿Cuál es la prima \emph{justa} y
        por qué la prima real la supera?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Las \omterm{def:g10:proba:sample}{muestras} como árbitro.}
\begin{enumerate}[resume]
  \item Se lanza una moneda $100$ veces. Usando el \omterm{def:g10:numbers:interval}{intervalo} de
        fluctuación $p \pm \frac{1}{\sqrt n}$, decide cuál de estos
        resultados debería levantar sospechas: $55$ caras; $70$ caras.
  \item Una encuesta a $1\,000$ votantes da a un candidato el $52\,\%$.
        Calcula el \omterm{def:g10:numbers:interval}{intervalo} de fluctuación alrededor del $50\,\%$ y di si
        la encuesta \emph{demuestra} que el candidato va por delante.
  \item Una fábrica afirma tener un $4\,\%$ de piezas defectuosas. Una
        \omterm{def:g10:proba:sample}{muestra} de $400$ da $28$ defectuosas ($7\,\%$); una \omterm{def:g10:proba:sample}{muestra}
        posterior de $2\,500$ da $175$ (otra vez el $7\,\%$). Contrasta la
        afirmación con cada \omterm{def:g10:proba:sample}{muestra}. ¿Por qué difieren los veredictos?
  \item Dos fuentes de error acechan a toda encuesta: el \emph{sesgo} de
        muestreo (problema de fin de semana del volumen anterior sobre
        cómo mienten los datos) y la \emph{fluctuación} de muestreo (este
        capítulo). ¿Cuál se encoge al crecer $n$ y cuál sobrevive a
        cualquier tamaño de \omterm{def:g10:proba:sample}{muestra}? Una frase para cada una.
  \item Final, la lista de comprobación del probabilista, una línea por
        punto: comprobar que los resultados son equiprobables antes de
        contar (pregunta 15); \omterm{met:g10:proba:tree}{árboles} y \omterm{def:g10:proba:operations}{sucesos contrarios} para los
        experimentos en varias etapas y para los ``al menos'' (preguntas 2
        y 3); \emph{la información cambia las \omterm{def:g10:proba:distribution}{probabilidades}, y cómo se ha
        obtenido importa} (preguntas 7 a 14); las \omterm{def:g10:proba:sample}{muestras} fluctúan
        aproximadamente $\frac{1}{\sqrt n}$, y la simulación arbitra
        (preguntas 11, 16 a 18).
\end{enumerate}
\end{problem}
