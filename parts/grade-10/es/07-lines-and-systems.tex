\chapter{Ecuaciones de rectas y sistemas lineales}\label{ch:g10:lines}

Una línea recta en el plano coordenado se describe mediante una \omterm{def:g10:algebra:equation}{ecuación}, y
dos rectas juntas forman un sistema de ecuaciones cuyas soluciones son sus
\omterm{def:g10:numbers:interunion}{intersección} agujas. Este capítulo conecta tres lenguajes: geométrico.
(líneas), algebraico (ecuaciones) y funcional (funciones afines de
\cref{ch:g10:reffunc}) --- y enseña las dos formas estándar de resolver un
sistema lineal.

\section{La ecuación de una recta.}

\begin{theorem}[Ecuación reducida]\label{thm:g10:lines:reduced}
en un \omterm{def:g10:coordgeom:system}{sistema de coordenadas}:
\begin{enumerate}
  \item Cada línea no vertical tiene un \omterm{def:g10:algebra:equation}{ecuación} de la forma
        \[
        y = mx + p ,
        \]
        con $m $ (el \emph{pendiente}\index{pendiente}) y $ p $ (el
        \emph{$ y $-interceptar}) determinado de forma única --- este es el
        \emph{ecuación reducida}\index{ecuación reducida} de la línea;
  \item Cada línea vertical tiene un \omterm{def:g10:algebra:equation}{ecuación} $ x = c $;
  \item un punto se encuentra en la recta exactamente cuando su \omterm{def:g10:coordgeom:system}{coordenadas} satisfacer el
        \omterm{def:g10:algebra:equation}{ecuación}.
\end{enumerate}
\end{theorem}
\admitted

\begin{proposition}[Pendiente a través de dos puntos]\label{prop:g10:lines:slope}
El \omterm{def:g10:reffunc:affine}{pendiente} de la línea (no vertical) que pasa por $ A(x_A, y_A)$ y
$ B(x_B, y_B)$, con $ x_A \neq x_B $, es
\[
m = \frac{y_B - y_A}{x_B - x_A}
\qquad
\text{(``change in $ y $ over change in $ x $'').}
\]
\end{proposition}

\begin{proof}
Ambos puntos satisfacen $ y = mx + p $, entonces $ y_B - y_A = (mx_B + p) - (mx_A + p)
= m(x_B - x_A)$; dividir por $ x_B - x_A \neq 0$.
\end{proof}

\begin{method}[Encontrar la ecuación de una recta que pasa por dos puntos.]\label{met:g10:lines:through}
Dado $ A $ y $ B $ con $ x_A \neq x_B $:
\begin{enumerate}
  \item calcular el \omterm{def:g10:reffunc:affine}{pendiente} $ m = \dfrac{y_B - y_A}{x_B - x_A}$;
  \item escribir $ y = mx + p $ y sustituir el \omterm{def:g10:coordgeom:system}{coordenadas} de $ A $ (o $ B $)
        encontrar $ p $;
  \item comprobar la final \omterm{def:g10:algebra:equation}{ecuación} con el \emph{otro} punto.
\end{enumerate}
Si $ x_A = x_B $, la línea es vertical: su \omterm{def:g10:algebra:equation}{ecuación} es $ x = x_A $.
\end{method}

\begin{example}\label{ex:g10:lines:through}
Línea a través $ A(1, 5)$ y $ B(3, 1)$. \omterm{def:g10:reffunc:affine}{pendiente}:
$ m = \frac{1 - 5}{3 - 1} = \frac{-4}{2} = -2$. Entonces $ y = -2x + p $; el
punto $ A $ da $5 = -2 \times 1 + p $, entonces $ p = 7$. \omterm{def:g10:algebra:equation}{ecuación}:
$ y = -2x + 7$. Consulte con $ B $: $-2 \times 3 + 7 = 1$.
\end{example}

\begin{proposition}[Líneas paralelas]\label{prop:g10:lines:parallel}
Dos rectas no verticales son paralelas si y sólo si tienen las mismas
\omterm{def:g10:reffunc:affine}{pendiente}. Dos líneas paralelas distintas no tienen un punto común; dos líneas con
diferente \omterm{def:g10:reffunc:affine}{pendientes} tener exactamente uno.
\end{proposition}

\begin{proof}
Sea $ y = mx + p $ y $ y = m'x + p'$ sean las dos lineas. Un punto común
resuelve $ mx + p = m'x + p'$, es decir.\ $(m - m')x = p' - p $. Si $ m \neq m'$,
esto tiene exactamente una solución $ x = \frac{p' - p}{m - m'}$: uno
\omterm{def:g10:numbers:interunion}{intersección} punto. Si $ m = m'$, el \omterm{def:g10:algebra:equation}{ecuación} lee $0 = p' - p $: no
solución cuando $ p \neq p'$ (paralelos distintos), cada $ x $ cuando $ p = p'$
(misma línea). Paralelismo (al no tener \omterm{def:g10:numbers:interunion}{intersección}, o ser igual) es por lo tanto
equivalente a $ m = m'$.
\end{proof}

\begin{remark}[Vector de dirección]
la linea $ y = mx + p $ sube $ m $ unidades para cada unidad a la derecha, por lo que
\omterm{def:g10:vectors:vector}{vector} $\vec u\,(1, m)$ es un \emph{dirección
vector}\index{dirección
vector} de la recta: la recta es paralela a
$\vec u $. Dos rectas son paralelas exactamente cuando su \omterm{prop:g10:lines:parallel}{dirección vectores} son
\omterm{def:g10:vectors:collinear}{colineales} (\cref{thm:g10:vectors:det}).
\end{remark}

\section{Sistemas lineales}

\begin{definition}[sistema lineal]\label{def:g10:lines:system}
A \emph{sistema de dos ecuaciones lineales}\index{sistema de dos ecuaciones lineales} en el
incógnitas $ x $ y $ y $ es un par de ecuaciones
\[
\begin{cases}
a x + b y = c \\
a' x + b' y = c'
\end{cases}
\]
estar satisfecho \emph{simultáneamente}. A \emph{solución} es un par
$(x, y)$ satisfaciendo a ambos. Cada \omterm{def:g10:algebra:equation}{ecuación} describe una recta, por lo que resolver la
sistema significa encontrar el \omterm{def:g10:numbers:interunion}{intersección} de dos líneas.
\end{definition}

\begin{method}[Sustitución]\label{met:g10:lines:substitution}
\begin{enumerate}
  \item resolver uno \omterm{def:g10:algebra:equation}{ecuación} para uno desconocido --- elija el más fácil, por ejemplo\
        expresar $ y $ en términos de $ x $;
  \item sustituir esta expresión en la \emph{otro} \omterm{def:g10:algebra:equation}{ecuación}, que
        ahora contiene solo $ x $;
  \item resolver para $ x $, luego calcular $ y $ desde el paso 1;
  \item comprobar el par en ambos originales ecuaciones.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:lines:substitution}
Resolver $\begin{cases} x + 2y = 8 \\ 3x - y = 3 \end{cases}$ por
sustitución. el segundo \omterm{def:g10:algebra:equation}{ecuación} da $ y = 3x - 3$. Sustituir en el
primero:
\[
x + 2(3x - 3) = 8
\ \Longleftrightarrow\
x + 6x - 6 = 8
\ \Longleftrightarrow\
7x = 14
\ \Longleftrightarrow\
x = 2 ,
\]
entonces $ y = 3 \times 2 - 3 = 3$. La solución es el par. $(2, 3)$.
Controlar: $2 + 2 \times 3 = 8$ y $3 \times 2 - 3 = 3$.
\end{example}

\begin{method}[Eliminación]\label{met:g10:lines:elimination}
\begin{enumerate}
  \item Multiplica cada uno \omterm{def:g10:algebra:equation}{ecuación} por un número bien elegido para que un desconocido
        obtiene coeficientes opuestos en los dos ecuaciones;
  \item agregar el ecuaciones: esa incógnita desaparece;
  \item resolver la incógnita restante \omterm{def:g10:algebra:equation}{ecuación}, luego sustituya de nuevo a
        encontrar al otro desconocido;
  \item comprobar el par en ambos originales ecuaciones.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:lines:elimination}
Resolver $\begin{cases} 2x + 3y = 7 \\ 5x - 2y = 8 \end{cases}$ por
eliminación. para eliminar $ y $, multiplica el primero \omterm{def:g10:algebra:equation}{ecuación} por $2$ y el
segundo por $3$:
\[
\begin{cases}
4x + 6y = 14 \\
15x - 6y = 24
\end{cases}
\]
Agregándolos: $19x = 38$, entonces $ x = 2$. Sustituyendo en $2x + 3y = 7$:
$4 + 3y = 7$, entonces $ y = 1$. Solución: $(2, 1)$. Consulta en el segundo
\omterm{def:g10:algebra:equation}{ecuación}: $5 \times 2 - 2 \times 1 = 8$.
\end{example}

\begin{remark}[¿Cuántas soluciones?]
Como dos líneas, una sistema lineal tiene exactamente una solución en general
(líneas no paralelas), y excepcionalmente ninguna (líneas paralelas distintas) o
infinitamente muchos (dos veces la misma línea). El criterio de
\cref{thm:g10:vectors:det} decide: el sistema tiene una solución única
exactamente cuando $ ab' - a'b \neq 0$.
\end{remark}

\begin{example}[Modelado con un sistema]\label{ex:g10:lines:modeling}
Tres cuadernos y dos bolígrafos cuestan $8$; cuesta un cuaderno y cuatro bolígrafos
$6$. Sea $ x $ ser el precio de un cuaderno y $ y $ el de un bolígrafo:
\[
\begin{cases}
3x + 2y = 8 \\
x + 4y = 6 .
\end{cases}
\]
desde el segundo \omterm{def:g10:algebra:equation}{ecuación}, $ x = 6 - 4y $; sustituyendo,
$3(6 - 4y) + 2y = 8$, entonces $18 - 10y = 8$, donación $ y = 1$ y luego
$ x = 2$. Un cuaderno cuesta $2$ y un bolígrafo cuesta $1$.
\end{example}

y = -\tfrac13 x + 1, \qquad
y = 4, \qquad
x = -2 .

\end{exercise}

\begin{exercise}[$\star $]\label{exo:g10:lines:4}

\begin{exercise}[$\star\star $]\label{exo:g10:lines:9}

  \item Calcular el \omterm{def:g10:numbers:interunion}{intersección} punto de $ d'$ con el $ x $-eje.

\begin{exercise}[$\star\star\star $]\label{exo:g10:lines:10}
Sea $ A(0, 0)$, $ B(6, 2)$ y $ C(2, 4)$.
\begin{enumerate}
  \item Encuentra el ecuaciones de las medianas del triángulo $ ABC $ emitido
        de $ A $ y de $ B $ (una mediana une un vértice a la \omterm{prop:g10:coordgeom:midpoint}{punto medio} de

  \item Calcular su \omterm{def:g10:numbers:interunion}{intersección} punto $ G $, y comprueba que
        $ G $ también está en la tercera mediana. (Deberías encontrar que el
        \omterm{def:g10:coordgeom:system}{coordenadas} de $ G $ son los promedios de los de $ A $, $ B $, $ C $.)
\end{enumerate}
\end{exercise}

\section{Problema: una solución, ninguna o infinitas soluciones}

\begin{problem}[{Problema del fin de semana --- dos líneas tienen tres posibles
destinos, un determinante decide entre ellos y el mundo.
el libro de texto más antiguo ya lo sabía}]
\label{pb:g10:lines:1}
dos lineales ecuaciones, dos incógnitas: el par de rectas que dibujan
puede cruzarse una vez, nunca, o ser la misma línea --- y
\emph{cada} sistema lineal hereda uno de estos tres destinos.
Este problema los clasifica, cumple con el número que decide (un
viejo conocido de la \omterm{def:g10:vectors:vector}{vectores} capítulo), resuelve el
faisanes y conejos chinos de dos mil años de antigüedad
\emph{Nueve capítulos}, y termina con esquinas de regiones --- el
Primer paso de la optimización utilizada por cada aerolínea y fábrica.
hoy.

\medskip
\noindent\textbf{Part I --- The line, fluently.}
\begin{enumerate}
  \item Encuentra el \omterm{thm:g10:lines:reduced}{ecuación reducida} de la línea a través
        $(1, 2)$ y $(3, 8)$
        (\cref{met:g10:lines:through}).
  \item Entre $ y = 3x - 1$, $ y = 3x + 4$ y $ y = -x + 7$: cual
        dos rectas son paralelas
        (\cref{prop:g10:lines:parallel}), y ¿dónde están los
        ¿Se encuentran pares no paralelos? (Calcule uno \omterm{def:g10:numbers:interunion}{intersección}.)
  \item La línea a través $(2, 1)$ y $(2, 5)$: ¿por qué tiene
        no \omterm{thm:g10:lines:reduced}{ecuación reducida} $ y = mx + p $, y cual es su
        \omterm{def:g10:algebra:equation}{ecuación}?
  \item es el punto $(4, 11)$ en la linea $ y = 3x - 1$? Y
        $(-1, -5)$?
  \item La línea a través $(0, 3)$ con \omterm{def:g10:reffunc:affine}{pendiente} $-2$: dar su
        \omterm{def:g10:algebra:equation}{ecuación}, sus dos ejes se interceptan y el área del
        triángulo que corta del primer cuadrante.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Three destinies.}
\begin{enumerate}[resume]
  \item Resolver por sustitución
        (\cref{met:g10:lines:substitution}):
        $\begin{cases} y = 2x - 3 \\ 3x + y = 7 \end{cases}$
  \item Resolver por eliminación
        (\cref{met:g10:lines:elimination}):
        $\begin{cases} 2x + 3y = 8 \\ 5x - 3y = -1 \end{cases}$
  \item Clasifica --- e interpreta con líneas --- los dos
        sistemas
        \[
        \begin{cases} 2x - y = 3 \\ 4x - 2y = 6 \end{cases}
        \qquad\text{and}\qquad
        \begin{cases} 2x - y = 3 \\ 4x - 2y = 1 . \end{cases}
        \]
        Indique la tricotomía completa: cruzada, paralela,
        idéntico --- una solución, ninguna, infinitas.
  \item Para el sistema general $ ax + by = e $, $ cx + dy = f $,
        el número decisivo es $ ad - bc $ --- lo mismo
        determinante como prueba de colinealidad de
        \cref{pb:g10:vectors:1}. Explique la coincidencia (que
        dos \omterm{def:g10:vectors:vector}{vectores} son \omterm{def:g10:vectors:collinear}{colineales} exactamente cuando las líneas están
        ¿paralelos o idénticos?), y evaluar $ ad - bc $ para el
        tres sistemas de preguntas 7 y 8.
  \item ¿Para qué valor de $ m $ hace
        $\begin{cases} x + my = 3 \\ 2x + 6y = 7 \end{cases}$
        no tiene solución? Resuelva el sistema para todos los demás valores.
        de $ m $\,\dots\ o al menos explica cómo lo harías.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The Nine Chapters.}
\begin{enumerate}[resume]
  \item Desde \emph{Nueve capítulos sobre el arte matemático}
        (China, hace unos 2000 años): una jaula contiene faisanes
        y conejos --- $35$ cabezas y $94$ piernas en total. como
        muchos de cada uno?
  \item Un puesto de jugos mezcla un $60\,\%$-almíbar de frutas con
        $20\,\%$-bebida de frutas para obtener $30$ L en $35\,\%$
        fruta. ¿Cuántos litros de cada uno?
  \item Un número de dos dígitos tiene suma de dígitos. $12$; intercambiando su
        dígitos lo disminuye en $18$. Encuéntralo usando el dígito
        álgebra del problema de fin de semana de álgebra de dígitos del volumen de la escuela secundaria para configurar el sistema.
  \item En la panadería, $3$ cafés y $2$ costo de los croissants
        $9.10$ euros; $2$ cafés y $3$ costo de los croissants
        $8.90$ euros. Resuelve elegantemente: ¿qué hacer? \emph{agregando} el
        dos ecuaciones y \emph{restando} cada uno te dice
        directamente?
  \item Un puesto en el mercado afirma: $2$ manzanas $+ 1$ pera para
        $5$ euros, y $4$ manzanas $+ 2$ peras para $9$ euros.
        Resolver --- o mejor dicho, explicar cuál es el destino del sistema
        revela sobre la aritmética del puesto.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Corners rule.}
\begin{enumerate}[resume]
  \item la linea $ y = 3x - 1$ divide el plano en dos. cual
        ¿De qué lado está el origen? describir el conjunto
        $ y > 3x - 1$ y cómo se contrasta un punto en su contra.
  \item Dibujar la región definida por las cuatro restricciones.
        \[
        x \geq 0, \qquad y \geq 0, \qquad x + y \leq 4,
        \qquad y \leq 2x + 1,
        \]
        y calcular sus cuatro puntos vértices.
  \item El beneficio de un taller es $ P = 3x + 2y $ en la región de
        pregunta 17. Evaluar $ P $ en las cuatro esquinas. Admitiendo
        que el \omterm{def:g10:functions:extrema}{máximo} de tal cantidad lineal en tal
        La región siempre se alcanza en una esquina (imagínese las líneas
        $3x + 2y = $ barrido constante), dan la óptima
        plano de producción.
  \item Los dos taxis del problema de fin de semana de dos termómetros del volumen de la escuela secundaria como sistema:
        escribe y resuelve
        $\begin{cases} y = 1.5x + 3 \\ y = 2x \end{cases}$ y
        Interpretar la solución.
  \item Final, el diccionario: uno \omterm{def:g10:algebra:equation}{ecuación} $\leftrightarrow $
        una línea; un sistema $\leftrightarrow $ dos líneas con
        tres destinos, decididos por $ ad - bc $; muchas desigualdades
        $\leftrightarrow$ una región cuyos rincones llevan la
        óptimo. Exprésalo en tres frases: acabas de
        recorrió el hall de entrada de \emph{álgebra lineal} y
        \emph{programación lineal}, ambos construidos íntegramente en el
        volúmenes universitarios.
\end{enumerate}
\end{problem}
