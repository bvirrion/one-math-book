\chapter{Ecuaciones de rectas y sistemas lineales}\label{ch:g10:lines}

Una recta del plano de \omterm{def:g10:coordgeom:system}{coordenadas} se describe mediante una \omterm{def:g10:algebra:equation}{ecuación}, y dos
rectas juntas forman un sistema de ecuaciones cuyas soluciones son sus
puntos de corte. Este capítulo conecta tres lenguajes (el geométrico de las
rectas, el algebraico de las ecuaciones y el funcional de las funciones
afines del \cref{ch:g10:reffunc}) y enseña las dos maneras habituales de
resolver un \omterm{def:g10:lines:system}{sistema lineal}.

\section{La ecuación de una recta}

\begin{theorem}[Ecuación explícita]\label{thm:g10:lines:reduced}
En un \omterm{def:g10:coordgeom:system}{sistema de coordenadas}:
\begin{enumerate}
  \item toda recta no vertical tiene una \omterm{def:g10:algebra:equation}{ecuación} de la forma
        \[
        y = mx + p ,
        \]
        con $m$ (la \emph{pendiente}\index{pendiente}) y $p$ (la
        \emph{\omterm{def:g10:reffunc:affine}{ordenada en el origen}}) determinados de manera única; es la
        \emph{ecuación explícita}\index{ecuación explícita} de la recta;
  \item toda recta vertical tiene una \omterm{def:g10:algebra:equation}{ecuación} $x = c$;
  \item un punto está sobre la recta exactamente cuando sus \omterm{def:g10:coordgeom:system}{coordenadas}
        satisfacen la \omterm{def:g10:algebra:equation}{ecuación}.
\end{enumerate}
\end{theorem}
\admitted

\begin{proposition}[Pendiente a partir de dos puntos]\label{prop:g10:lines:slope}
La \omterm{def:g10:reffunc:affine}{pendiente} de la recta (no vertical) que pasa por $A(x_A, y_A)$ y
$B(x_B, y_B)$, con $x_A \neq x_B$, es
\[
m = \frac{y_B - y_A}{x_B - x_A}
\qquad
\text{(``lo que sube $y$ dividido entre lo que avanza $x$'').}
\]
\end{proposition}

\begin{proof}
Los dos puntos cumplen $y = mx + p$, luego
$y_B - y_A = (mx_B + p) - (mx_A + p) = m(x_B - x_A)$; dividimos entre
$x_B - x_A \neq 0$.
\end{proof}

\begin{method}[Hallar la ecuación de la recta que pasa por dos puntos]\label{met:g10:lines:through}
Dados $A$ y $B$ con $x_A \neq x_B$:
\begin{enumerate}
  \item calcula la \omterm{def:g10:reffunc:affine}{pendiente} $m = \dfrac{y_B - y_A}{x_B - x_A}$;
  \item escribe $y = mx + p$ y sustituye las \omterm{def:g10:coordgeom:system}{coordenadas} de $A$ (o de
        $B$) para hallar $p$;
  \item comprueba la \omterm{def:g10:algebra:equation}{ecuación} final con el \emph{otro} punto.
\end{enumerate}
Si $x_A = x_B$, la recta es vertical: su \omterm{def:g10:algebra:equation}{ecuación} es $x = x_A$.
\end{method}

\begin{example}\label{ex:g10:lines:through}
Recta que pasa por $A(1, 5)$ y $B(3, 1)$. \omterm{def:g10:reffunc:affine}{Pendiente}:
$m = \frac{1 - 5}{3 - 1} = \frac{-4}{2} = -2$. Entonces $y = -2x + p$; el
punto $A$ da $5 = -2 \times 1 + p$, luego $p = 7$. \omterm{def:g10:algebra:equation}{Ecuación}:
$y = -2x + 7$. Comprobación con $B$: $-2 \times 3 + 7 = 1$.
\end{example}

\begin{proposition}[Rectas paralelas]\label{prop:g10:lines:parallel}
Dos rectas no verticales son paralelas si y solo si tienen la misma
\omterm{def:g10:reffunc:affine}{pendiente}. Dos rectas paralelas distintas no tienen ningún punto común;
dos rectas de \omterm{def:g10:reffunc:affine}{pendientes} distintas tienen exactamente uno.
\end{proposition}

\begin{proof}
Sean $y = mx + p$ e $y = m'x + p'$ las dos rectas. Un punto común resuelve
$mx + p = m'x + p'$, es decir, $(m - m')x = p' - p$. Si $m \neq m'$, esto
tiene exactamente una solución, $x = \frac{p' - p}{m - m'}$: un punto de
corte. Si $m = m'$, la \omterm{def:g10:algebra:equation}{ecuación} queda $0 = p' - p$: sin solución cuando
$p \neq p'$ (paralelas distintas) y con todo $x$ cuando $p = p'$ (la misma
recta). Así pues, el paralelismo (no cortarse, o coincidir) equivale a
$m = m'$.
\end{proof}

\begin{remark}[Vector director]
La recta $y = mx + p$ sube $m$ unidades por cada unidad hacia la derecha,
así que el \omterm{def:g10:vectors:vector}{vector} $\vec u\,(1, m)$ es un \emph{vector
director}\index{vector director} de la recta: la recta es paralela a
$\vec u$. Dos rectas son paralelas exactamente cuando sus \omterm{def:g10:vectors:vector}{vectores}
directores son \omterm{def:g10:vectors:collinear}{colineales} (\cref{thm:g10:vectors:det}).
\end{remark}

\section{Sistemas lineales}

\begin{definition}[Sistema lineal]\label{def:g10:lines:system}
Un \emph{sistema de dos ecuaciones lineales}\index{sistema lineal} con las
incógnitas $x$ e $y$ es una pareja de ecuaciones
\[
\begin{cases}
a x + b y = c \\
a' x + b' y = c'
\end{cases}
\]
que deben cumplirse \emph{simultáneamente}. Una \emph{solución} es una
pareja $(x, y)$ que cumple las dos. Cada \omterm{def:g10:algebra:equation}{ecuación} describe una recta, así
que resolver el sistema es hallar el corte de dos rectas.
\end{definition}

\begin{method}[Sustitución]\label{met:g10:lines:substitution}
\begin{enumerate}
  \item Despeja una incógnita en una de las ecuaciones, la que resulte más
        cómoda; por ejemplo, expresa $y$ en \omterm{def:g10:functions:function}{función} de $x$;
  \item sustituye esa expresión en la \emph{otra} \omterm{def:g10:algebra:equation}{ecuación}, que pasa a
        contener solo $x$;
  \item resuelve en $x$ y calcula después $y$ con el paso 1;
  \item comprueba la pareja en las dos ecuaciones de partida.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:lines:substitution}
Resolvamos $\begin{cases} x + 2y = 8 \\ 3x - y = 3 \end{cases}$ por
sustitución. La segunda \omterm{def:g10:algebra:equation}{ecuación} da $y = 3x - 3$. Sustituimos en la
primera:
\[
x + 2(3x - 3) = 8
\ \Longleftrightarrow\
x + 6x - 6 = 8
\ \Longleftrightarrow\
7x = 14
\ \Longleftrightarrow\
x = 2 ,
\]
y entonces $y = 3 \times 2 - 3 = 3$. La solución es la pareja $(2, 3)$.
Comprobación: $2 + 2 \times 3 = 8$ y $3 \times 2 - 3 = 3$.
\end{example}

\begin{method}[Reducción]\label{met:g10:lines:elimination}
\begin{enumerate}
  \item Multiplica cada \omterm{def:g10:algebra:equation}{ecuación} por un número bien elegido para que una
        incógnita quede con coeficientes opuestos en las dos ecuaciones;
  \item suma las ecuaciones: esa incógnita desaparece;
  \item resuelve la \omterm{def:g10:algebra:equation}{ecuación} con una sola incógnita que queda y sustituye
        hacia atrás para hallar la otra;
  \item comprueba la pareja en las dos ecuaciones de partida.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:lines:elimination}
Resolvamos $\begin{cases} 2x + 3y = 7 \\ 5x - 2y = 8 \end{cases}$ por
reducción. Para eliminar $y$, multiplicamos la primera \omterm{def:g10:algebra:equation}{ecuación} por $2$ y
la segunda por $3$:
\[
\begin{cases}
4x + 6y = 14 \\
15x - 6y = 24
\end{cases}
\]
Sumándolas: $19x = 38$, luego $x = 2$. Sustituyendo en $2x + 3y = 7$:
$4 + 3y = 7$, luego $y = 1$. Solución: $(2, 1)$. Comprobación en la
segunda \omterm{def:g10:algebra:equation}{ecuación}: $5 \times 2 - 2 \times 1 = 8$.
\end{example}

\begin{remark}[¿Cuántas soluciones?]
Igual que dos rectas, un \omterm{def:g10:lines:system}{sistema lineal} tiene en general exactamente una
solución (rectas no paralelas) y, excepcionalmente, ninguna (rectas
paralelas distintas) o infinitas (dos veces la misma recta). El criterio
del \cref{thm:g10:vectors:det} decide: el sistema tiene solución única
exactamente cuando $ab' - a'b \neq 0$.
\end{remark}

\begin{example}[Modelizar con un sistema]\label{ex:g10:lines:modeling}
Tres cuadernos y dos bolígrafos cuestan $8$; un cuaderno y cuatro
bolígrafos cuestan $6$. Sea $x$ el precio de un cuaderno e $y$ el de un
bolígrafo:
\[
\begin{cases}
3x + 2y = 8 \\
x + 4y = 6 .
\end{cases}
\]
De la segunda \omterm{def:g10:algebra:equation}{ecuación}, $x = 6 - 4y$; sustituyendo,
$3(6 - 4y) + 2y = 8$, luego $18 - 10y = 8$, de donde $y = 1$ y después
$x = 2$. Un cuaderno cuesta $2$ y un bolígrafo, $1$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:lines:1}
Para cada recta, indica la \omterm{def:g10:reffunc:affine}{pendiente} y la \omterm{def:g10:reffunc:affine}{ordenada en el origen}, y
represéntala:
\[
y = 2x - 3, \qquad
y = -\tfrac13 x + 1, \qquad
y = 4, \qquad
x = -2 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:lines:2}
¿Pertenece el punto $A(2, 7)$ a la recta $y = 3x + 1$? ¿Y $B(-1, -1)$? ¿Y
el punto $C(4, 13)$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:lines:3}
Halla la \omterm{thm:g10:lines:reduced}{ecuación explícita} de la recta que pasa por:
\[
\text{(a) } A(0, 2) \text{ y } B(3, 8);
\qquad
\text{(b) } A(-1, 5) \text{ y } B(2, -4);
\qquad
\text{(c) } A(4, 1) \text{ y } B(4, 6).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:lines:4}
De entre las rectas $y = 3x - 1$, $y = -3x + 2$, $y = 3x + 5$,
$y = \frac{6x + 2}{2}$, ¿cuáles son paralelas entre sí? ¿Cuáles son en
realidad iguales?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:lines:5}
Resuelve por sustitución:
\[
\begin{cases}
y = 2x - 1 \\
3x + y = 9
\end{cases}
\qquad\text{y después}\qquad
\begin{cases}
x - y = 4 \\
2x + 3y = 3 .
\end{cases}
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:lines:6}
Resuelve por reducción:
\[
\begin{cases}
x + y = 10 \\
x - y = 4
\end{cases}
\qquad\text{y después}\qquad
\begin{cases}
3x + 2y = 12 \\
2x + 5y = 8 .
\end{cases}
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:lines:7}
Determina, sin resolverlos, cuántas soluciones tiene cada sistema:
\[
\begin{cases}
2x - y = 3 \\
-4x + 2y = -6
\end{cases}
\qquad
\begin{cases}
2x - y = 3 \\
-4x + 2y = 1
\end{cases}
\qquad
\begin{cases}
2x - y = 3 \\
x + y = 0 .
\end{cases}
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:lines:8}
Se vendieron doscientas entradas para una obra de teatro del instituto,
unas a $5$ (infantiles) y otras a $9$ (de adulto), con una recaudación
total de $1352$. ¿Cuántas entradas de cada clase se vendieron?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:lines:9}
Sea $d$ la recta $y = 2x - 3$ y sea $A(1, 4)$.
\begin{enumerate}
  \item Da la \omterm{def:g10:algebra:equation}{ecuación} de la recta $d'$ paralela a $d$ que pasa por $A$.
  \item Calcula el punto de corte de $d'$ con el eje $x$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:lines:10}
Sean $A(0, 0)$, $B(6, 2)$ y $C(2, 4)$.
\begin{enumerate}
  \item Halla las ecuaciones de las medianas del triángulo $ABC$ que
        parten de $A$ y de $B$ (una mediana une un vértice con el \omterm{prop:g10:coordgeom:midpoint}{punto
        medio} del lado opuesto).
  \item Calcula su punto de corte $G$ y comprueba que $G$ está también
        sobre la tercera mediana. (Deberías encontrar que las \omterm{def:g10:coordgeom:system}{coordenadas}
        de $G$ son las medias de las de $A$, $B$ y $C$.)
\end{enumerate}
\end{exercise}

\section{Problema: Una solución, ninguna o infinitas}

\begin{problem}[{Problema de fin de semana --- dos rectas tienen tres
destinos posibles, un determinante decide entre ellos y el libro de texto
más antiguo del mundo ya lo sabía}]
\label{pb:g10:lines:1}
Dos ecuaciones lineales, dos incógnitas: la pareja de rectas que dibujan
puede cortarse una vez, no cortarse nunca o ser una sola y misma recta, y
\emph{todo} \omterm{def:g10:lines:system}{sistema lineal} hereda uno de esos tres destinos. Este problema
los clasifica, presenta al número que decide (un viejo conocido del
capítulo de \omterm{def:g10:vectors:vector}{vectores}), resuelve los faisanes y conejos de hace dos mil años
de los \emph{Nueve capítulos} chinos y termina con los vértices de una
región: el primer paso de la optimización que hoy usan todas las compañías
aéreas y todas las fábricas.

\medskip
\noindent\textbf{Parte I --- La recta, con soltura.}
\begin{enumerate}
  \item Halla la \omterm{thm:g10:lines:reduced}{ecuación explícita} de la recta que pasa por $(1, 2)$ y
        $(3, 8)$ (\cref{met:g10:lines:through}).
  \item De entre $y = 3x - 1$, $y = 3x + 4$ e $y = -x + 7$: ¿qué dos
        rectas son paralelas (\cref{prop:g10:lines:parallel}) y dónde se
        cortan las parejas no paralelas? (Calcula uno de los cortes.)
  \item La recta que pasa por $(2, 1)$ y $(2, 5)$: ¿por qué no tiene
        \omterm{thm:g10:lines:reduced}{ecuación explícita} $y = mx + p$ y cuál es su \omterm{def:g10:algebra:equation}{ecuación}?
  \item ¿Está el punto $(4, 11)$ sobre la recta $y = 3x - 1$? ¿Y
        $(-1, -5)$?
  \item La recta que pasa por $(0, 3)$ con \omterm{def:g10:reffunc:affine}{pendiente} $-2$: da su \omterm{def:g10:algebra:equation}{ecuación},
        sus dos cortes con los ejes y el área del triángulo que recorta en
        el primer cuadrante.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Tres destinos.}
\begin{enumerate}[resume]
  \item Resuelve por sustitución (\cref{met:g10:lines:substitution}):
        $\begin{cases} y = 2x - 3 \\ 3x + y = 7 \end{cases}$
  \item Resuelve por reducción (\cref{met:g10:lines:elimination}):
        $\begin{cases} 2x + 3y = 8 \\ 5x - 3y = -1 \end{cases}$
  \item Clasifica, e interpreta con rectas, los dos sistemas
        \[
        \begin{cases} 2x - y = 3 \\ 4x - 2y = 6 \end{cases}
        \qquad\text{y}\qquad
        \begin{cases} 2x - y = 3 \\ 4x - 2y = 1 . \end{cases}
        \]
        Enuncia la tricotomía completa: rectas que se cortan, paralelas o
        idénticas; una solución, ninguna o infinitas.
  \item Para el sistema general $ax + by = e$, $cx + dy = f$, el número
        que decide es $ad - bc$: el mismo determinante que la prueba de
        colinealidad del \cref{pb:g10:vectors:1}. Explica la coincidencia
        (¿qué dos \omterm{def:g10:vectors:vector}{vectores} son \omterm{def:g10:vectors:collinear}{colineales} exactamente cuando las rectas
        son paralelas o idénticas?) y evalúa $ad - bc$ en los tres
        sistemas de las preguntas 7 y 8.
  \item ¿Para qué valor de $m$ no tiene solución
        $\begin{cases} x + my = 3 \\ 2x + 6y = 7 \end{cases}$? Resuelve el
        sistema para todos los demás valores de $m$\,\dots\ o, al menos,
        explica cómo lo harías.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Los Nueve capítulos.}
\begin{enumerate}[resume]
  \item De los \emph{Nueve capítulos sobre el arte matemático} (China,
        hace unos 2000 años): en una jaula hay faisanes y conejos, con
        $35$ cabezas y $94$ patas en total. ¿Cuántos hay de cada?
  \item Un puesto de zumos mezcla un jarabe con un $60\,\%$ de fruta y una
        bebida con un $20\,\%$ para obtener $30$ L al $35\,\%$ de fruta.
        ¿Cuántos litros de cada uno?
  \item Un número de dos cifras tiene suma de cifras $12$; al
        intercambiarlas, disminuye en $18$. Halla el número, usando para
        plantear el sistema el álgebra de las cifras del problema de fin
        de semana del volumen anterior sobre las cifras.
  \item En la panadería, $3$ cafés y $2$ cruasanes cuestan $9.10$ euros;
        $2$ cafés y $3$ cruasanes cuestan $8.90$ euros. Resuelve con
        elegancia: ¿qué te dicen directamente \emph{sumar} las dos
        ecuaciones y \emph{restarlas}?
  \item Un puesto del mercado anuncia: $2$ manzanas $+ 1$ pera por $5$
        euros, y $4$ manzanas $+ 2$ peras por $9$ euros. Resuelve o, mejor
        dicho, explica qué revela el destino del sistema sobre la
        aritmética del puesto.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Mandan los vértices.}
\begin{enumerate}[resume]
  \item La recta $y = 3x - 1$ parte el plano en dos. ¿De qué lado está el
        origen? Describe el conjunto $y > 3x - 1$ y cómo se comprueba si
        un punto pertenece a él.
  \item Dibuja la región definida por las cuatro restricciones
        \[
        x \geq 0, \qquad y \geq 0, \qquad x + y \leq 4,
        \qquad y \leq 2x + 1,
        \]
        y calcula sus cuatro vértices.
  \item El beneficio de un taller es $P = 3x + 2y$ sobre la región de la
        pregunta 17. Evalúa $P$ en los cuatro vértices. Admitiendo que el
        \omterm{def:g10:functions:extrema}{máximo} de una cantidad lineal así sobre una región de este tipo se
        alcanza siempre en un vértice (imagina las rectas
        $3x + 2y = $ constante barriendo el plano), da el plan de
        producción óptimo.
  \item Los dos taxis del problema de fin de semana del volumen anterior
        sobre los dos termómetros, como sistema: escribe y resuelve
        $\begin{cases} y = 1.5x + 3 \\ y = 2x \end{cases}$ e interpreta la
        solución.
  \item Final, el diccionario: una \omterm{def:g10:algebra:equation}{ecuación} $\leftrightarrow$ una recta;
        un sistema $\leftrightarrow$ dos rectas con tres destinos,
        decididos por $ad - bc$; muchas inecuaciones $\leftrightarrow$ una
        región cuyos vértices llevan el óptimo. Enúncialo en tres frases:
        acabas de recorrer el vestíbulo del \emph{álgebra lineal} y de la
        \emph{programación lineal}, que se construyen enteras en los
        volúmenes universitarios.
\end{enumerate}
\end{problem}
