\chapter{Números y conjuntos de números}\label{ch:g10:numbers}

Las matemáticas comienzan con los números, y no todos los números son del mismo tipo:
contar números, números negativos, fracciones y números como $\sqrt 2$
o $\pi $ que ninguna fracción puede expresar. Este capítulo los organiza en
familias anidadas, introduce \omterm{def:g10:numbers:interval}{intervalos} para describir partes del número
línea, y utiliza el \omterm{def:g10:numbers:abs}{valor absoluto} para medir distancias entre números.

\section{Las familias de números.}

\begin{definition}[Conjuntos de números]\label{def:g10:numbers:sets}
\begin{itemize}
  \item $\N $ es el conjunto de \emph{numeros naturales}\index{numeros naturales}:
        $0, 1, 2, 3, \dots $
  \item $\Z $ es el conjunto de \emph{numeros enteros}\index{numeros enteros}: $\dots, -2, -1,
        0, 1, 2, \dots $
  \item $\Q $ es el conjunto de \emph{numeros racionales}\index{numeros racionales}:
        todos los cocientes $\frac{p}{q}$ con $ p \in \Z $, $ q \in \N $ y
        $ q \neq 0$.
  \item $\R $ es el conjunto de \emph{numeros reales}\index{numeros reales}: todos los
        números que se pueden colocar en la recta numérica.
\end{itemize}
\end{definition}

\begin{notation}
El símbolo $\in $ dice ``pertenece a'': $3 \in \N $, $-5 \in \Z $,
$\frac23 \in \Q $. El símbolo $\subset $ dice ``está incluido en'': cada
número natural es un entero, cada entero es un número racional
(p.ej.\ $-5 = \frac{-5}{1}$), y cada número racional es real, entonces
\[
\N \subset \Z \subset \Q \subset \R .
\]
Una barra diagonal niega un símbolo: $\frac12 \notin \Z $.
\end{notation}

\begin{example}\label{ex:g10:numbers:classify}
Coloquemos algunos números en la familia más pequeña que los contenga.
\begin{itemize}
  \item $\frac{28}{4} = 7$, entonces $\frac{28}{4} \in \N $ para todo que es
        escrito como una fracción: siempre simplifica primero.
  \item $-3.5 = -\frac{35}{10} = -\frac72$ es racional pero no un entero.
  \item $0.333\dots $ (el dígito $3$ repitiendo para siempre) es igual $\frac13$,
        a número racional.
  \item $\sqrt 2$ y $\pi $ son reales pero no racionales, como estamos a punto de
        ver por $\sqrt 2$; tales números se llaman
        \emph{irracional}\index{irracional}.
\end{itemize}
\end{example}

\begin{theorem}[Irracionalidad de $\sqrt 2$]\label{thm:g10:numbers:sqrt2}
el numero $\sqrt 2$ no es racional: ninguna fracción de números enteros tiene cuadrado
$2$.
\end{theorem}

\begin{proof}
Razonamos por contradicción, en pequeños pasos.
\begin{enumerate}
  \item Suponer $\sqrt 2 = \frac{p}{q}$ dónde $ p $ y $ q $ son positivos
        números enteros, y la fracción está completamente simplificada, por lo que $ p $ y $ q $ son
        Ni siquiera ambos.
  \item Al elevar ambos lados al cuadrado se obtiene $2 = \frac{p^2}{q^2}$, eso es
        $ p^2 = 2q^2$. Entonces $ p^2$ es par.
  \item Si $ p $ eran extraños, digamos $ p = 2k+1$, entonces
        $ p^2 = 4k^2 + 4k + 1$ Sería impar. Desde $ p^2$ es par, $ p $ debe
        ser par: $ p = 2k $ para algunos entero $ k $.
  \item Sustituyendo: $(2k)^2 = 2q^2$, entonces $4k^2 = 2q^2$, entonces
        $ q^2 = 2k^2$. Con el mismo argumento que en el paso 3, $ q $ es par.
  \item Ahora $ p $ y $ q $ ambos son pares, contradiciendo el paso 1. El
        la suposición era imposible: $\sqrt 2$ es \omterm{ex:g10:numbers:classify}{irracional}.
\end{enumerate}
\end{proof}

\begin{remark}
números racionales son exactamente los números cuya expansión decimal ya sea
se detiene (como $\frac72 = 3.5$) o eventualmente repite el mismo bloque para siempre
(como $\frac13 = 0.333\dots $ o $\frac{1}{7} = 0.142857\,142857\dots $).
Números \omterm{ex:g10:numbers:classify}{irracionales} como $\sqrt2 = 1.41421356\dots $ tener dígitos decimales
que nunca caen en un patrón repetitivo. Admitimos esta caracterización.
a este nivel.
\end{remark}

\section{La recta numérica y los intervalos.}

\omterm{def:g10:numbers:sets}{numeros reales} llenar una línea: elegir un origen $0$ y una unidad de longitud,
cada número real corresponde exactamente a un punto.

\begin{definition}[Intervalo]\label{def:g10:numbers:interval}
Sea $ a $ y $ b $ ser números reales con $ a < b $. Un
\emph{intervalo}\index{intervalo} es el conjunto de todos números reales entre dos
límites. Un corchete significa que el límite está incluido, un paréntesis significa
queda excluido:
\begin{center}
\begin{tabular}{c|c}
notation & description \\
\hline
$\intcc{a}{b}$ & $ a \leq x \leq b $ (both ends included) \\
$\intoo{a}{b}$ & $ a < x < b $ (both ends excluded) \\
$\intco{a}{b}$ & $ a \leq x < b $ \\
$\intoc{a}{b}$ & $ a < x \leq b $ \\
$\intco{a}{+\infty}$ & $ x \geq a $ \\
$\intoo{-\infty}{b}$ & $ x < b $ \\
\end{tabular}
\end{center}
los simbolos $-\infty $ y $+\infty $ (``infinito'') no son números, sólo un
forma de decir que el intervalo continúa para siempre; el soporte al lado de
ellos es siempre un paréntesis. toda la linea $\R $ es el intervalo
$\intoo{-\infty}{+\infty}$.
\end{definition}

\begin{definition}[Intersección y unión]\label{def:g10:numbers:interunion}
Sea $ I $ y $ J $ ser dos conjuntos de números reales.
El \emph{intersección}\index{intersección} $ I \cap J $ (``$ I $ y $ J $'')
es el conjunto de números pertenecientes a ambos; el
\emph{unión}\index{unión} $ I \cup J $ (``$ I $ o $ J $'') es el conjunto de
números pertenecientes al menos a uno de ellos.
\end{definition}

\begin{example}\label{ex:g10:numbers:interunion}
Llevar $ I = \intcc{-1}{3}$ y $ J = \intoo{1}{5}$. Dibuja ambos en el mismo
línea: se superponen entre $1$ y $3$. Por lo tanto
\[
I \cap J = \intoc{1}{3},
\qquad
I \cup J = \intco{-1}{5}.
\]
Tenga en cuenta los corchetes: $1 \notin J $ entonces $1 \notin I \cap J $, pero $3 \in I $ y
$3 \in J $, entonces $3 \in I \cap J $.
\end{example}

\begin{method}[Trabajar con intervalos]\label{met:g10:numbers:intervals}
para encontrar el \omterm{def:g10:numbers:interunion}{intersección} o \omterm{def:g10:numbers:interunion}{unión} de dos \omterm{def:g10:numbers:interval}{intervalos}:
\begin{enumerate}
  \item dibuja la recta numérica y marca los cuatro límites;
  \item sombrea el primero \omterm{def:g10:numbers:interval}{intervalo} encima de la línea y el segundo debajo de ella;
  \item el \omterm{def:g10:numbers:interunion}{intersección} es donde las sombras se superponen, el \omterm{def:g10:numbers:interunion}{unión} es donde
        al menos un sombreado está presente;
  \item decidir cada paréntesis probando si el límite en sí pertenece a
        ambos conjuntos (\omterm{def:g10:numbers:interunion}{intersección}) o al menos a uno (\omterm{def:g10:numbers:interunion}{unión}).
\end{enumerate}
\end{method}

\section{Valor absoluto y distancia.}

\begin{definition}[valor absoluto]\label{def:g10:numbers:abs}
El \emph{valor absoluto}\index{valor absoluto} de un número real $ x $ es
\[
\abs{x} =
\begin{cases}
x & \text{if } x \geq 0, \\
-x & \text{if } x < 0.
\end{cases}
\]
Por ejemplo $\abs{7} = 7$ y $\abs{-4} = 4$. en todos los casos
$\abs{x} \geq 0$.
\end{definition}

\begin{proposition}[Distancia en la línea]\label{prop:g10:numbers:distance}
A pesar de números reales $ a $ y $ b $, la distancia entre los puntos $ a $ y
$ b $ en la recta numérica es $\abs{b - a}$. En particular $\abs{x}$ es el
distancia de $ x $ a $0$.
\end{proposition}

\begin{proof}
Si $ b \geq a $, la distancia desde $ a $ a $ b $ es $ b - a \geq 0$, que
es igual $\abs{b-a}$. Si $ b < a $, la distancia es $ a - b = -(b - a) > 0$,
que es otra vez $\abs{b - a}$ por definición de la \omterm{def:g10:numbers:abs}{valor absoluto}.
\end{proof}

\begin{proposition}[Valor absoluto e intervalos.]\label{prop:g10:numbers:absinterval}
Sea $ a $ ser un número real y $ r > 0$. Entonces
\[
\abs{x - a} \leq r
\quad\text{exactly when}\quad
x \in \intcc{a - r}{a + r}.
\]
\end{proposition}

\begin{proof}
$\abs{x-a} \leq r $ dice que la distancia desde $ x $ a $ a $ es como máximo $ r $,
es decir\ que $ x $ no se encuentra más allá de $ r $ de $ a $ a cada lado. el
los números que satisfacen esto son exactamente aquellos entre $ a - r $ y $ a + r $,
límites incluidos.
\end{proof}

\begin{example}\label{ex:g10:numbers:abs}
Resolver $\abs{x - 3} \leq 2$. Las soluciones son los números a distancia en
la mayoría $2$ de $3$: el \omterm{def:g10:numbers:interval}{intervalo} $\intcc{1}{5}$. Por el contrario, el \omterm{def:g10:numbers:interval}{intervalo}
$\intcc{-1}{7}$ tiene centro $\frac{-1+7}{2} = 3$ y radio
$\frac{7-(-1)}{2} = 4$, así lo describe $\abs{x - 3} \leq 4$.
\end{example}

\section{Aproximaciones}

Números \omterm{ex:g10:numbers:classify}{irracionales}, e incluso la mayoría de las fracciones, no se pueden escribir exactamente
con un número finito de dígitos decimales, por lo que en la práctica los aproximamos.

\begin{definition}[Aproximación a una precisión dada]\label{def:g10:numbers:approx}
un numero $ d $ es un \emph{aproximación}\index{aproximación} de $ x $ a
dentro $10^{-n}$ cuando $\abs{x - d} \leq 10^{-n}$. Truncar o redondear
la expansión decimal después de la $ n $-ésimo dígito ambos dan tal
aproximaciones.
\end{definition}

\begin{example}
De $\pi = 3.14159\,26\dots $: el truncamiento $3.141$ y el redondeo
$3.142$ son ambos aproximaciones de $\pi $ hacia dentro $10^{-3}$. el redondeo
está a una distancia menor que $\frac12 \times 10^{-3}$, el truncamiento solamente
garantías $10^{-3}$. Escribiendo $3.141 \leq \pi \leq 3.142$ \emph{marcos}
$\pi $ entre dos límites decimales.
\end{example}

\section{Problema: entre dos números cualesquiera}

\begin{problem}[{Problema del fin de semana --- racionales e irracionales
entrelazado: cada intervalo, por pequeño que sea, contiene infinitos
de cada uno, y ninguna medida puede distinguirlos}]
\label{pb:g10:numbers:1}
Los racionales parecen una multitud (¡todas las fracciones!) y los
\omterm{ex:g10:numbers:classify}{irracionales} como excepciones exóticas ($\sqrt2$, $\pi $). este problema
revela la verdadera imagen: las dos familias \emph{entrelazar} entonces
finamente que cada \omterm{def:g10:numbers:interval}{intervalo} de la recta numérica, sin embargo
microscópico, contiene infinitos de cada uno --- con extraños
consecuencias, como esta: no hay medición física, sin embargo
preciso, puede alguna vez decidir si una longitud es racional.

\medskip
\noindent\textbf{Part I --- The four kingdoms.}
\begin{enumerate}
  \item Para cada número, nombra el más pequeño de los conjuntos. $\N $,
        $\Z $, $\Q $, $\R $ conteniéndolo
        (\cref{def:g10:numbers:sets}): $-7$; \ $\frac{13}{4}$; \
        $\sqrt{16}$; \ $0.121212\ldots $ (recordar
        el problema de fin de semana con decimales repetidos del volumen de la escuela secundaria); \ $\sqrt8$; \ $\pi $ (admite su
        irracionalidad --- probada sólo en la universidad
        volúmenes).
  \item demostrar que $\Q $ es estable bajo adición y
        multiplicación: si $ x = \frac pq $ y $ y = \frac rs $ son
        racional, escribir $ x + y $ y $ xy $ como fracciones simples.
  \item Deducir por contradicción: (a) la suma de un racional y un
        \omterm{ex:g10:numbers:classify}{irracional} es \omterm{ex:g10:numbers:classify}{irracional}; (b) el producto de un
        \emph{distinto de cero} racional y un \omterm{ex:g10:numbers:classify}{irracional} es \omterm{ex:g10:numbers:classify}{irracional}.
  \item Demuestre que el conjunto de \omterm{ex:g10:numbers:classify}{irracionales} es estable bajo
        \emph{ni} operación: exhibición dos \omterm{ex:g10:numbers:classify}{irracionales} cuyo
        suma es racional, y dos cuyo producto es racional.
  \item Localizar $\sqrt2 + \sqrt8$ y $\sqrt2 \times \sqrt8$ en
        los cuatro reinos (simplificar $\sqrt8$ primero,
        método del volumen de la escuela secundaria).
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The rationals are dense.}
\begin{enumerate}[resume]
  \item Encuentra un racional estrictamente entre $3.47$ y $3.48$; entonces
        un segundo; luego describe cómo producir tantos como
        deseado (el zoom del problema del fin de semana sin número siguiente del volumen de Escuela Secundaria, ahora con un
        prueba a la vista).
  \item El teorema general. Sea $ a < b $ ser dos reales cualesquiera, con
        brecha $ g = b - a > 0$. Elegir $ n $ con
        $10^{-n} < g $ y considere los múltiplos de $10^{-n}$
        (la cuadrícula decimal del paso $10^{-n}$). Explique por qué en
        al menos un punto de la cuadrícula cae estrictamente entre $ a $ y $ b $,
        y concluir: \emph{cada \omterm{def:g10:numbers:interval}{intervalo} de longitud positiva
        contiene un número racional}.
  \item Actualice la conclusión: cada uno de esos \omterm{def:g10:numbers:interval}{intervalo} contiene
        \emph{infinitamente muchos} racionales. (Aplica la pregunta 7
        de nuevo, dentro de un más pequeño \omterm{def:g10:numbers:interval}{intervalo}.)
  \item Ahora el \omterm{ex:g10:numbers:classify}{irracionales}: dado $ a < b $, toma un racional $ r $
        estrictamente dentro (pregunta 7) y considere los números
        $ r + \frac{\sqrt2}{10^k}$. Utilizando la pregunta 3, muestre que
        son \omterm{ex:g10:numbers:classify}{irracional}, y eso para $ k $ lo suficientemente grandes como para que todavía
        mentir en el \omterm{def:g10:numbers:interval}{intervalo}: \emph{cada \omterm{def:g10:numbers:interval}{intervalo} también contiene
        infinitamente muchos \omterm{ex:g10:numbers:classify}{irracionales}}.
  \item Dos enigmas clásicos resueltos: ¿existe el más pequeño?
        positivo número real? A número real ``justo después
        $3$''? Responda ambas con el teorema de la pregunta 7, y
        saludo a la versión infantil
        (el problema del fin de semana sin número siguiente del volumen de la escuela secundaria).
\end{enumerate}

\medskip
\noindent\textbf{Part III --- \omterm{def:g10:numbers:abs}{valor absoluto}, the geometry of
$\R $.}
\begin{enumerate}[resume]
  \item Resuelva y exprese las soluciones como \omterm{def:g10:numbers:interval}{intervalos} o sindicatos
        (\cref{prop:g10:numbers:absinterval}):
        $\abs{x - 5} = 2$; \quad $\abs{x - 5} < 2$; \quad
        $\abs{x + 1} \geq 3$.
  \item Se debe fabricar un pistón para $80$ mm dentro de un
        tolerancia de $0.05$ mm. Escriba el requisito con un
        \omterm{def:g10:numbers:abs}{valor absoluto}, entonces como un \omterm{def:g10:numbers:interval}{intervalo}. Medida de dos pistones
        $79.97$ y $80.06$ mm: veredictos?
  \item La desigualdad del triángulo en la recta:
        $\abs{a + b} \leq \abs a + \abs b $. Compruébalo
        $(a, b) = (3, -5)$ y $(-2, -7)$, pruébalo cuando $ a $ y
        $ b $ tienen el mismo signo y cuando tienen opuesto
