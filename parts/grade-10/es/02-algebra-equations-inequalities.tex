\chapter{Álgebra: ecuaciones e inecuaciones}\label{ch:g10:algebra}

El álgebra es el arte de calcular con letras. Este capítulo repasa y afina
los dos movimientos básicos, \omterm{def:g10:algebra:expand}{desarrollar} y \omterm{def:g10:algebra:expand}{factorizar}, y los usa para
resolver ecuaciones e inecuaciones de forma sistemática, con la \omterm{def:g10:algebra:signtable}{tabla de
signos} como herramienta central.

\section{Desarrollar y factorizar}

\begin{definition}[Desarrollar, factorizar]\label{def:g10:algebra:expand}
\emph{Desarrollar}\index{desarrollar} una expresión es transformar
productos en sumas; \emph{factorizar}\index{factorizar} es lo contrario,
transformar una suma en un producto. La regla básica es la propiedad
distributiva:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g10:algebra:expand}
Desarrollemos $(2x + 3)(x - 5)$ paso a paso:
\begin{align*}
(2x + 3)(x - 5)
&= 2x \times x + 2x \times (-5) + 3 \times x + 3 \times (-5) \\
&= 2x^2 - 10x + 3x - 15 \\
&= 2x^2 - 7x - 15 .
\end{align*}
\end{example}

\begin{theorem}[Identidades notables]\label{thm:g10:algebra:identities}
Para todos los \omterm{def:g10:numbers:sets}{números reales} $a$ y $b$:
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
Cada una es un desarrollo directo. Para la primera:
\[
(a+b)^2 = (a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2 .
\]
La segunda se obtiene sustituyendo $b$ por $-b$: $(a-b)^2 = a^2 + 2a(-b) +
(-b)^2 = a^2 - 2ab + b^2$. Y para la tercera:
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$.
\end{proof}

\begin{example}\label{ex:g10:algebra:identities}
Usadas hacia delante (desarrollando) y hacia atrás (factorizando):
\begin{align*}
(3x + 2)^2 &= 9x^2 + 12x + 4,
& x^2 - 25 &= (x+5)(x-5), \\
(x - 4)^2 &= x^2 - 8x + 16,
& 4x^2 + 4x + 1 &= (2x + 1)^2 .
\end{align*}
El cálculo mental también sale ganando: $101 \times 99 = (100+1)(100-1) =
10000 - 1 = 9999$.
\end{example}

\begin{method}[Factorizar una expresión]\label{met:g10:algebra:factor}
Prueba, en este orden:
\begin{enumerate}
  \item \emph{Factor común:} localiza un factor presente en todos los
        términos y sácalo fuera, por ejemplo $6x^2 + 4x = 2x(3x + 2)$. El
        factor común puede ser un paréntesis entero:
        $(x+1)(2x-3) + (x+1)(x+7) = (x+1)(3x+4)$.
  \item \emph{Identidad notable:} reconoce $a^2 - b^2$ o
        $a^2 \pm 2ab + b^2$, por ejemplo $9x^2 - 16 = (3x)^2 - 4^2 =
        (3x+4)(3x-4)$.
  \item Combina las dos vías y comprueba el resultado desarrollándolo de
        nuevo.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:factorsteps}
Factoricemos $E = (2x + 1)^2 - (x - 3)^2$. Es una diferencia de cuadrados
$a^2 - b^2$ con $a = 2x+1$ y $b = x-3$:
\begin{align*}
E &= \bigl[(2x+1) + (x-3)\bigr] \times \bigl[(2x+1) - (x-3)\bigr] \\
  &= (3x - 2)(2x + 1 - x + 3) \\
  &= (3x - 2)(x + 4).
\end{align*}
Atención al segundo corchete: restar $x - 3$ es restar $x$ \emph{y} sumar
$3$.
\end{example}

\section{Ecuaciones}

\begin{definition}[Ecuación, solución]\label{def:g10:algebra:equation}
Una \emph{ecuación}\index{ecuación} es una igualdad en la que aparece un
número desconocido $x$; una \emph{solución} es un valor de $x$ que hace
cierta la igualdad. Resolver la ecuación es hallar \emph{todas} sus
soluciones.
\end{definition}

Dos ecuaciones son \emph{equivalentes} cuando tienen exactamente las
mismas soluciones. Sumar el mismo número a los dos miembros, o
multiplicarlos por un mismo número \emph{no nulo}, da una \omterm{def:g10:algebra:equation}{ecuación}
equivalente.

\begin{example}[Ecuación de primer grado]\label{ex:g10:algebra:firstdegree}
Resolvamos $5x - 7 = 2x + 8$, paso a paso:
\begin{align*}
5x - 7 &= 2x + 8 \\
5x - 2x &= 8 + 7 && \text{(sumamos $7 - 2x$ a los dos miembros)} \\
3x &= 15 \\
x &= 5 && \text{(dividimos los dos miembros entre $3$).}
\end{align*}
La única solución es $5$. Comprobación: $5 \times 5 - 7 = 18$ y
$2 \times 5 + 8 = 18$.
\end{example}

\begin{theorem}[Regla del producto nulo]\label{thm:g10:algebra:zeroproduct}
Un producto de \omterm{def:g10:numbers:sets}{números reales} es cero si y solo si al menos uno de sus
factores es cero:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ o } \ B = 0 .
\]
\end{theorem}

\begin{proof}
Si $A = 0$ o $B = 0$, el producto es evidentemente $0$. Recíprocamente,
supongamos que $AB = 0$ y $A \neq 0$. Dividiendo los dos miembros entre
$A$ se obtiene $B = 0$.
\end{proof}

\begin{method}[Resolver factorizando]\label{met:g10:algebra:zeroproduct}
Para resolver una \omterm{def:g10:algebra:equation}{ecuación} cuyo segundo miembro es $0$ y cuyo primer
miembro se puede \omterm{def:g10:algebra:expand}{factorizar}:
\begin{enumerate}
  \item pasa todo al primer miembro, de modo que la \omterm{def:g10:algebra:equation}{ecuación} quede
        $E = 0$;
  \item factoriza $E$ como producto de factores de primer grado;
  \item aplica la regla del producto nulo y resuelve cada \omterm{def:g10:algebra:equation}{ecuación}
        pequeña;
  \item reúne todas las soluciones encontradas.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:zeroproduct}
Resolvamos $(x - 2)(3x + 6) = 0$: o bien $x - 2 = 0$, que da $x = 2$, o
bien $3x + 6 = 0$, que da $x = -2$. Soluciones: $-2$ y $2$.

Resolvamos $x^2 = 9x$. \emph{No} dividas entre $x$ (¡podría ser cero!):
pasa todo a un miembro y factoriza:
\[
x^2 - 9x = 0
\quad\Longleftrightarrow\quad
x(x - 9) = 0
\quad\Longleftrightarrow\quad
x = 0 \ \text{ o } \ x = 9 .
\]
\end{example}

\begin{example}[Ecuación con cociente]\label{ex:g10:algebra:quotient}
Resolvamos $\dfrac{x - 1}{x + 2} = 0$. Un cociente es cero exactamente
cuando su numerador es cero \emph{y} su denominador no lo es. Aquí
$x - 1 = 0$ da $x = 1$, y $1 + 2 = 3 \neq 0$: la única solución es $1$. El
valor $x = -2$ está \emph{prohibido} (división entre cero) y hay que
excluirlo siempre desde el principio.
\end{example}

\section{Inecuaciones y tablas de signos}

\begin{proposition}[Reglas de las desigualdades]\label{prop:g10:algebra:ineqrules}
Sea $a \leq b$.
\begin{enumerate}
  \item Para cualquier $c$: $a + c \leq b + c$ (sumar conserva el orden).
  \item Si $c > 0$: $ac \leq bc$ (multiplicar por un número positivo
        conserva el orden).
  \item Si $c < 0$: $ac \geq bc$ (multiplicar por un número negativo
        \emph{invierte} el orden).
\end{enumerate}
\end{proposition}

\begin{proof}
1. $(b + c) - (a + c) = b - a \geq 0$.
2. $bc - ac = (b-a)c$ es el producto de dos números no negativos, luego
$bc \geq ac$.
3. Ahora $(b-a)c \leq 0$, por ser producto de un número no negativo por
uno negativo, luego $bc \leq ac$.
\end{proof}

\begin{example}\label{ex:g10:algebra:firstineq}
Resolvamos $-2x + 3 > 9$:
\begin{align*}
-2x + 3 &> 9 \\
-2x &> 6 && \text{(restamos 3)} \\
x &< -3 && \text{(dividimos entre $-2 < 0$: se invierte la desigualdad).}
\end{align*}
El conjunto de soluciones es $\intoo{-\infty}{-3}$.
\end{example}

\begin{definition}[Tabla de signos]\label{def:g10:algebra:signtable}
Una \emph{tabla de signos}\index{tabla de signos} recoge, en una fila por
factor, el signo ($+$ o $-$) de cada factor de una expresión cuando $x$
recorre $\R$, con un $0$ en cada valor donde el factor se anula; una
última fila da el signo del producto, aplicando la regla de los signos
columna a columna.
\end{definition}

\begin{example}\label{ex:g10:algebra:signtable}
Signo de $P(x) = (x - 2)(3 - x)$. El factor $x - 2$ se anula en $2$ y es
positivo después; el factor $3 - x$ se anula en $3$ y es positivo antes:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccc}
$x$ & $-\infty$ \quad & $2$ & & $3$ & \quad $+\infty$ \\
\hline
$x - 2$ & $-$ & $0$ & $+$ & & $+$ \\
$3 - x$ & $+$ & & $+$ & $0$ & $-$ \\
\hline
$P(x)$ & $-$ & $0$ & $+$ & $0$ & $-$ \\
\end{tabular}
\end{center}
Así pues, $P(x) > 0$ exactamente en $\intoo{2}{3}$, y $P(x) \leq 0$ en
$\intoc{-\infty}{2} \cup \intco{3}{+\infty}$.
\end{example}

\begin{method}[Resolver una inecuación con una tabla de signos]\label{met:g10:algebra:signtable}
Para resolver una inecuación del tipo $E(x) \geq 0$ o $E(x) < 0$:
\begin{enumerate}
  \item pasa todo al primer miembro, de modo que el segundo sea $0$ (nunca
        multipliques los dos miembros por una expresión de signo
        desconocido);
  \item factoriza el primer miembro o, si es un cociente, redúcelo a común
        denominador;
  \item construye la \omterm{def:g10:algebra:signtable}{tabla de signos} con una fila por factor (si hay
        cociente, marca los valores prohibidos con una doble barra en la
        última fila);
  \item lee los \omterm{def:g10:numbers:interval}{intervalos} donde se cumple el signo pedido, comprobando si
        cada extremo está incluido.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:quotientineq}
Resolvamos $\dfrac{x + 1}{x - 3} \leq 0$. El numerador se anula en $-1$ y
el denominador en $3$ (valor prohibido). El cociente es negativo cuando
los dos tienen signos opuestos, lo que ocurre para $x$ entre $-1$ y $3$.
Incluyendo el extremo donde se anula el \emph{numerador} y excluyendo el
valor prohibido, el conjunto de soluciones es $\intco{-1}{3}$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:algebra:1}
Desarrolla y simplifica:
\[
3(2x - 5) - 2(x - 7), \qquad
(x + 4)(2x - 1), \qquad
(3x - 2)^2, \qquad
(5 - x)(5 + x).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:algebra:2}
Factoriza:
\[
x^2 - 49, \qquad
x^2 + 10x + 25, \qquad
7x^2 - 14x, \qquad
16x^2 - 8x + 1 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:algebra:3}
Resuelve:
\[
4x + 9 = x - 3, \qquad
\frac{2x - 1}{3} = 5, \qquad
2(x - 3) = 2x + 1 .
\]
(Una de ellas no tiene solución: explica por qué.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:algebra:4}
Resuelve usando la regla del producto nulo:
\[
(x + 5)(2x - 8) = 0, \qquad
x^2 - 16 = 0, \qquad
x^2 + 6x + 9 = 0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:algebra:5}
Resuelve las inecuaciones y escribe los conjuntos de soluciones como
\omterm{def:g10:numbers:interval}{intervalos}:
\[
3x - 5 \leq 7, \qquad
-4x + 1 > 9, \qquad
2(x - 1) \geq 5x + 4 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:6}
Factoriza $E = (x - 1)(x + 3) - (x - 1)(2x - 5)$ y resuelve después
$E = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:7}
Resuelve $x^2 = 5x$ y después $(2x + 1)^2 = (x - 4)^2$. (Para la segunda,
pasa todo al primer miembro y factoriza la diferencia de cuadrados.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:8}
Con una \omterm{def:g10:algebra:signtable}{tabla de signos}, resuelve
\[
(x + 2)(4 - x) > 0,
\qquad
\frac{2x - 6}{x + 1} \geq 0 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:9}
Un servicio de vídeo bajo demanda cuesta $9$ al mes, más una cuota única
de alta de $15$; un rival cuesta $12$ al mes sin cuota de alta. ¿A partir
de cuántos meses sale más barato el primero en total? Plantea y resuelve
una inecuación.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:10}
Resuelve $\dfrac{x+3}{x-1} = 2$. (Multiplica los dos miembros por $x - 1$
después de excluir el valor prohibido, o pasa todo al primer miembro con
denominador común.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:algebra:11}
Sea $n$ un entero. Demuestra que $(n + 1)^2 - n^2$ es siempre impar y que
la diferencia entre los cuadrados de dos números impares consecutivos es
siempre múltiplo de $8$.
\end{exercise}

\section{Problema: De Babilonia al número áureo}

\begin{problem}[{Problema de fin de semana --- problemas de suma y
producto, completar el cuadrado tal como lo dibujaba al-Juarismi y la
ecuación $x^2 = x + 1$}]
\label{pb:g10:algebra:1}
Hace cuatro mil años, los escribas babilonios resolvían esto sobre
arcilla: \emph{hallar dos números conociendo su suma y su producto}. Su
truco, buscar los números simétricamente alrededor de la semisuma, sigue
hoy resolviendo cualquier \omterm{def:g10:algebra:equation}{ecuación} de segundo grado bajo su nombre
moderno: \emph{completar el cuadrado}. Este problema aprende el truco, ve
a al-Juarismi completar literalmente un cuadrado y pone el método a
trabajar sobre la proporción más célebre de todas.

\medskip
\noindent\textbf{Parte I --- El problema babilonio.}
\begin{enumerate}
  \item Calentamiento (\cref{thm:g10:algebra:identities}): factoriza
        $x^2 - 10x + 25$, después $9x^2 - 49$ y después, combinando una
        identidad con una diferencia de cuadrados, $x^2 + 6x + 9 - 4$.
  \item Resuelve $(x + 1)(x + 5) = 0$ y $(3x - 7)(3x + 7) = 0$
        (\cref{thm:g10:algebra:zeroproduct}).
  \item El problema del escriba: dos números tienen suma $10$ y producto
        $21$. Siguiendo a Babilonia, escríbelos como $5 - t$ y $5 + t$
        (simétricos alrededor de la semisuma) y halla $t$ y después los
        dos números.
  \item El mismo método: suma $14$, producto $33$.
  \item La receta general: para suma $s$ y producto $p$, demuestra que los
        dos números son
        \[
        \frac s2 - t
        \quad\text{y}\quad
        \frac s2 + t,
        \qquad\text{donde}\quad
        t^2 = \left(\frac s2\right)^2 - p ,
        \]
        y enuncia la condición sobre $s$ y $p$ para que el problema tenga
        solución.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Completar el cuadrado.}
\begin{enumerate}[resume]
  \item Desarrolla $(x - a)(x - b)$ y concluye: las soluciones de
        $x^2 - sx + p = 0$ son exactamente las parejas de números de suma
        $s$ y producto $p$. (Los escribas resolvían ecuaciones de segundo
        grado sin saberlo.)
  \item Comprueba que $x^2 - 10x + 21 = (x - 5)^2 - 4$ y resuelve
        $x^2 - 10x + 21 = 0$ factorizando el segundo miembro como
        diferencia de cuadrados. Compara con la pregunta 3.
  \item Resuelve $x^2 + 6x - 7 = 0$ completando el cuadrado.
  \item Completa el cuadrado en $x^2 + 4x + 5$ y explica por qué la
        \omterm{def:g10:algebra:equation}{ecuación} $x^2 + 4x + 5 = 0$ \emph{no} tiene solución real. Enuncia
        el criterio general: escrita en la forma $(x + h)^2 + k$, ¿cuándo
        se anula una expresión de segundo grado?
  \item El nombre es literal. En el Bagdad del siglo IX, al-Juarismi
        dibujaba $x^2 + 6x = 7$ como un cuadrado de lado $x$ con dos
        rectángulos de $3 \times x$ pegados a dos de sus lados, y
        \emph{completaba el cuadrado} con la esquina de $3 \times 3$ que
        faltaba. Haz el dibujo y explica cómo convierte la \omterm{def:g10:algebra:equation}{ecuación} en
        $(x + 3)^2 = 16$; resuélvela después.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Signos, tablas y dos promesas
antiguas.}
\begin{enumerate}[resume]
  \item Construye la \omterm{def:g10:algebra:signtable}{tabla de signos} de $(x - 3)(x + 2)$
        (\cref{def:g10:algebra:signtable}) y resuelve
        $(x - 3)(x + 2) < 0$.
  \item Resuelve $x^2 \leq 10x - 21$ (pasa todo a un miembro y reutiliza
        la pregunta 7).
  \item Resuelve $\dfrac{x - 1}{x + 2} \geq 0$ con una \omterm{def:g10:algebra:signtable}{tabla de signos},
        atento al valor prohibido.
  \item Una promesa antigua cumplida: para el granjero del problema del
        cercado de la reina Dido, en el volumen anterior, con $20$ m de
        valla, el área de un corral de $x \times (10 - x)$ cumple
        \[
        A(x) = x(10 - x) = 25 - (x - 5)^2 .
        \]
        Comprueba la identidad y lee en ella una \emph{demostración} de lo
        que entonces solo se podía conjeturar: el área máxima y el único
        $x$ que la alcanza.
  \item Otra: demuestra que $x + \dfrac1x \geq 2$ para todo $x > 0$, con
        igualdad solo en $x = 1$. (Multiplica por $x$, agrupa, reconoce un
        cuadrado; es la desigualdad entre las medias aritmética y
        geométrica del problema de fin de semana del volumen anterior
        sobre las relaciones de Euclides, con otro disfraz.)
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La sección áurea.}
Divide un segmento de longitud $1$ en una parte grande $x$ y una parte
pequeña $1 - x$ de manera que \emph{el todo sea a la parte grande como la
grande es a la pequeña}: $\frac{1}{x} = \frac{x}{1 - x}$.
\begin{enumerate}[resume]
  \item Demuestra que la condición se escribe $x^2 + x - 1 = 0$.
  \item Resuelve completando el cuadrado y quédate con la raíz que es una
        longitud: $x = \frac{\sqrt5 - 1}{2} \approx 0.618$.
  \item El \emph{número áureo} es $\varphi = \frac1x =
        \frac{2}{\sqrt5 - 1}$. Racionaliza el denominador (multiplica
        arriba y abajo por $\sqrt5 + 1$: una identidad en acción) y
        demuestra que $\varphi = \frac{\sqrt5 + 1}{2} \approx 1.618$.
  \item Comprueba \emph{exactamente} las dos identidades mágicas de
        $\varphi$:
        \[
        \varphi^2 = \varphi + 1,
        \qquad
        \frac{1}{\varphi} = \varphi - 1 .
        \]
  \item Final: explica por qué $\varphi$ es irracional (el problema de fin
        de semana del volumen anterior sobre la irracionalidad aporta el
        hecho clave sobre $\sqrt5$) y calcula después con tres decimales
        los cocientes $\frac53$, $\frac85$, $\frac{13}{8}$,
        $\frac{21}{13}$ de números de Hemachandra--Fibonacci consecutivos
        (problema de fin de semana del volumen anterior sobre el recuento
        de ritmos). ¿A qué parecen perseguir? (La demostración de que lo
        alcanzan pertenece a los capítulos sobre sucesiones.)
\end{enumerate}
\end{problem}
