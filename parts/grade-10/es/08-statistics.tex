\chapter{Estadística descriptiva}\label{ch:g10:stats}

Las estadísticas resumen una gran colección de números: grados, salarios,
temperaturas --- según unos pocos indicadores bien elegidos. Este capítulo presenta
las dos familias de indicadores: los que localizan el \emph{centro} de la
datos (\omterm{def:g10:stats:mean}{media}, \omterm{def:g10:stats:median}{mediana}) y aquellos que miden su \emph{dispersión} (recorrido,
cuartiles). Entra un tratamiento más profundo, que incluye la desviación estándar.
\cref{ch:g11:stat}.

\section{Datos y frecuencias}

\begin{definition}[Serie estadística]\label{def:g10:stats:series}
A \emph{serie estadística}\index{serie estadística} es una lista de valores
$x_1, x_2, \dots, x_N $ observado en una población de tamaño $ N $. cuando un valor
$ x $ aparece $ n $ veces, $ n $ es su \emph{efectivo} y el cociente
\[
f = \frac{n}{N}
\]
es su \emph{frecuencia}\index{frecuencia} (a menudo expresado como un
porcentaje). Las frecuencias de todos los valores suman $1$.
\end{definition}

\begin{example}\label{ex:g10:stats:frequencies}
las calificaciones de $20$ estudiantes en un $5$-prueba de puntos:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|cccccc}
grade $ x $ & $0$ & $1$ & $2$ & $3$ & $4$ & $5$ \\
\hline
count $ n $ & $1$ & $2$ & $4$ & $6$ & $5$ & $2$ \\
\omterm{def:g10:stats:series}{frecuencia} $ f $ & $0.05$ & $0.10$ & $0.20$ & $0.30$ & $0.25$ & $0.10$ \\
\end{tabular}
\end{center}
Controlar: $1 + 2 + 4 + 6 + 5 + 2 = 20$ y las \omterm{def:g10:stats:series}{frecuencias} suman $1$.
\end{example}

\section{Medidas de centro}

\begin{definition}[Media]\label{def:g10:stats:mean}
La \emph{media}\index{media} de la serie es
\[
\bar x = \frac{x_1 + x_2 + \dots + x_N}{N}.
\]
Si los valores $ v_1, \dots, v_k $ aparecen con efectivos $ n_1, \dots, n_k $, el
Se calcula el mismo número que la \emph{media ponderada}
\[
\bar x = \frac{n_1 v_1 + n_2 v_2 + \dots + n_k v_k}{n_1 + n_2 + \dots + n_k}.
\]
\end{definition}

\begin{example}\label{ex:g10:stats:mean}
Para el cuestionario de \cref{ex:g10:stats:frequencies}:
\[
\bar x = \frac{1 \times 0 + 2 \times 1 + 4 \times 2 + 6 \times 3
+ 5 \times 4 + 2 \times 5}{20}
= \frac{0 + 2 + 8 + 18 + 20 + 10}{20}
= \frac{58}{20} = 2.9 .
\]
\end{example}

\begin{definition}[Mediana]\label{def:g10:stats:median}
La \emph{mediana}\index{mediana} es un valor que divide el \emph{ordenado}
serie en dos mitades: al menos la mitad de los valores son $\leq $ la mediana, y
al menos la mitad son $\geq $ él. En la práctica, ordenar los $ N $ valores;
\begin{itemize}
  \item si $ N $ es impar, la mediana es el valor medio, en posición
        $\frac{N+1}{2}$;
  \item si $ N $ es par, toma el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de los dos valores medios, en
        posiciones $\frac N2$ y $\frac N2 + 1$.
\end{itemize}
\end{definition}

\begin{example}\label{ex:g10:stats:median}
serie ordenada de $7$ Precios de la vivienda (en miles): $120$, $150$, $160$,
$180$, $210$, $240$, $900$. La \omterm{def:g10:stats:median}{mediana} es el $4$ valor, $180$. La \omterm{def:g10:stats:mean}{media}
es $\frac{1960}{7} = 280$ --- más grande que $6$ del $7$ precios!
Un valor extremo ($900$) tira de la \omterm{def:g10:stats:mean}{media} mucho más que la \omterm{def:g10:stats:median}{mediana}: el
\omterm{def:g10:stats:median}{mediana} es \emph{robusto}, la \omterm{def:g10:stats:mean}{media} no lo es.
\end{example}

\section{Medidas de dispersión}

\begin{definition}[Rango, cuartiles]\label{def:g10:stats:quartiles}
Para una serie ordenada:
\begin{itemize}
  \item el \emph{rango}\index{rango} es la diferencia entre el
        valores más grandes y más pequeños;
  \item el \emph{primer cuartil}\index{primer cuartil} $ Q_1$ es el mas pequeño
        valor tal que al menos una cuarta parte ($25\%$) de los valores son
        $\leq Q_1$; el \emph{tercer cuartil} $ Q_3$ es el valor más pequeño
        tal que al menos tres cuartas partes ($75\%$) de los valores son
        $\leq Q_3$;
  \item el \emph{rango intercuartil}\index{rango intercuartil} $ Q_3 - Q_1$ mide la propagación de
        la mitad central de los datos.
\end{itemize}
\end{definition}

\begin{method}[Encontrar los cuartiles]\label{met:g10:stats:quartiles}
Para una serie de $ N $ valores ordenados:
\begin{enumerate}
  \item calcular $\frac N4$; si no es un entero redondearlo \emph{arriba} a
        el siguiente entero; $ Q_1$ es el valor en esa posición;
  \item calcular $\frac{3N}{4}$; redondear si es necesario; $ Q_3$ es el valor en
        esa posición.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:stats:quartiles}
Toma el $11$ valores ordenados
\[
2,\ 3,\ 5,\ 5,\ 6,\ 7,\ 8,\ 9,\ 9,\ 10,\ 12 .
\]
\omterm{def:g10:stats:median}{mediana}: posición $\frac{11+1}{2} = 6$, entonces la \omterm{def:g10:stats:median}{mediana} es $7$.
$ Q_1$: $\frac{11}{4} = 2.75$, redondear a $3$; el $3$ El tercer valor es $5$.
$ Q_3$: $\frac{33}{4} = 8.25$, redondear a $9$; el $9$ el valor es $9$.
recorrido: $12 - 2 = 10$; \omterm{def:g10:stats:quartiles}{rango} intercuartílico: $9 - 5 = 4$.
\end{example}

\begin{example}[Comparando dos grupos]\label{ex:g10:stats:comparison}
Dos clases toman el mismo examen (puntuaciones de $20$):
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccc}
 & \omterm{def:g10:stats:median}{mediana} & $ Q_1$ & $ Q_3$ \\
\hline
class A & $12$ & $10$ & $14$ \\
class B & $12$ & $6$ & $17$ \\
\end{tabular}
\end{center}
Mismo \omterm{def:g10:stats:median}{mediana}, diferenciales muy diferentes: la clase A es homogénea (la mitad de su
puntuaciones en $\intcc{10}{14}$), la clase B es heterogénea (su mitad central
se extiende $\intcc{6}{17}$). Los indicadores del centro por sí solos nunca dicen el todo
historia.
\end{example}

\end{remark}

\section{Problema: la persona promedio no existe}

\begin{problem}[{Problema del fin de semana --- media versus mediana: valores atípicos,
curvas niveladoras, el almacén en la carretera y la cabina diseñada
para nadie}]
\label{pb:g10:stats:1}
En 1950, la Fuerza Aérea de los EE. UU. midió $4\,000$ pilotos en diez cuerpos
dimensiones y construyó la cabina para el \emph{hombre promedio}. eso
equipado --- como descubrió un teniente al contar --- esencialmente
nadie. Las estadísticas resumen las multitudes con números únicos, y cada
resumen dice la verdad sobre algo y miente sobre algo
más. Este problema toma la \omterm{def:g10:stats:mean}{media} y la \omterm{def:g10:stats:median}{mediana}
(\cref{def:g10:stats:mean}, \cref{def:g10:stats:median}) a los tribunales,
deja que cada uno muestre lo que hace mejor y cierra la cabina.

\medskip
\noindent\textbf{Part I --- Two summaries on trial.}
\begin{enumerate}
  \item En una calle se venden siete casas, en miles de euros:
        $200$, $210$, $220$, $230$, $240$, $250$ --- y uno
        mansión en $1\,200$. Calcular la \omterm{def:g10:stats:mean}{media} y la \omterm{def:g10:stats:median}{mediana}
        precio. ¿Qué número describe "una casa en esta calle"?
  \item Retire la mansión y vuelva a calcular ambas. ¿Por cuánto hizo
        cada movimiento resumido? Establezca la moraleja sobre los valores atípicos.
  \item Una tabla censal: fuera de $100$ familias, $20$ no tener
        niño, $25$ tener uno, $30$ tener dos, $15$ tener tres,
        $10$ tener cuatro. Calcular la \omterm{def:g10:stats:mean}{media} número de niños. No
        familia ``tiene $1.7$ niños '' --- entonces, ¿qué cantidad real
        hace la \omterm{def:g10:stats:mean}{media} ¿codificar? (Multiplícalo por $100$.)
  \item Por los quince marcos
        $4, 6, 7, 8, 9, 10, 10, 11, 12, 12, 13, 14, 15, 17, 19$:
        encontrar la \omterm{def:g10:stats:median}{mediana} y el cuartiles $ Q_1$, $ Q_3$
        (\cref{met:g10:stats:quartiles}).
  \item Para la misma serie, dé la recorrido y el
        \omterm{def:g10:stats:quartiles}{rango} intercuartílico. ¿Qué mide cada uno?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Grading curves and merged classes.}
\begin{enumerate}[resume]
  \item De \cref{exo:g10:stats:9}: cambiando cada valor por
        $+b $ turnos \omterm{def:g10:stats:mean}{media}, \omterm{def:g10:stats:median}{mediana} y cuartiles por $+b $; escalando
        por $ a $ los escala por $ a $. ¿Qué pasa con el recorrido y
        el \omterm{def:g10:stats:quartiles}{rango} intercuartílico bajo un turno? ¿Bajo una escala?
        Justificar.
  \item una prueba tiene \omterm{def:g10:stats:mean}{media} $9$ (fuera de $20$) y el profesor quiere
        \omterm{def:g10:stats:mean}{media} $12$. Se proponen dos curvas: sumar $3$ apunta a
        todos, o multiplicar cada marca por $\frac43$. calcular
        el destino de un estudiante en $6$ y uno en $18$ debajo de cada
        curva. ¿Qué curva favorece a quién y cuál tiene una
        ¿Problema del techo?
  \item Clase A ($20$ estudiantes) tiene \omterm{def:g10:stats:mean}{media} $12$; clase B ($30$
        estudiantes) tiene \omterm{def:g10:stats:mean}{media} $10$. Calcular la \omterm{def:g10:stats:mean}{media} del $50$
        estudiantes juntos y explique por qué no es $11$
        (la lección de promedio ponderado de la \omterm{def:g10:stats:mean}{media} armónica problema de fin de semana del volumen de Escuela Secundaria, en
        el aula).
  \item mostrar que \omterm{def:g10:stats:median}{medianas} rechazar este juego: calcular el \omterm{def:g10:stats:median}{medianas}
        de $ A = \{1, 2, 3\}$ y $ B = \{0, 10\}$, entonces el
        \omterm{def:g10:stats:median}{mediana} de la lista fusionada y comparar con la
        promedio ponderado por tamaño de los dos \omterm{def:g10:stats:median}{medianas}.
  \item Patinaje artístico: puntuación de cinco jueces $5.2$, $5.3$, $5.4$,
        $5.5$, $9.9$. Calcular la \omterm{def:g10:stats:mean}{media}, la \omterm{def:g10:stats:median}{mediana}, y el
        \emph{recortado \omterm{def:g10:stats:mean}{media}} (elimine el promedio más bajo y más alto
        el resto). ¿Por qué la mayoría de los deportes juzgados recortan?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Reading distributions.}
\begin{enumerate}[resume]
  \item Dos fábricas producen pernos de \omterm{def:g10:stats:mean}{media} longitud

  \item En una clínica, la \omterm{def:g10:stats:median}{mediana} esperar es $10$ minutos pero el

  \item ``Su bebé está en el percentil 75 de altura'' ---
        ¿Qué significa esa frase? \omterm{def:g10:stats:mean}{media}? ¿Y puede cada hijo de una

  \item Clase A: mín. $4$, $ Q_1\,8$, \omterm{def:g10:stats:median}{mediana} $11$, $ Q_3\,13$, \omterm{def:g10:functions:extrema}{máximo}

  \item La lista de verificación del periodista: un titular dice "promedio

\noindent\textbf{Part IV --- The cockpit case.}
\begin{enumerate}[resume]
  \item Construya su propia evidencia: construya cinco precios de la vivienda
        con \omterm{def:g10:stats:mean}{media} $300$ y \omterm{def:g10:stats:median}{mediana} $200$ (miles de euros).
  \item La \omterm{def:g10:stats:mean}{media} es el punto de equilibrio: probar a partir de la definición
        que las desviaciones de la \omterm{def:g10:stats:mean}{media} cancelar siempre,
        $\sum (x_i - \bar x) = 0$ y verificarlo en su conjunto de datos
        de la pregunta 16.
  \item La \omterm{def:g10:stats:median}{mediana} es el minimizador de recados: las tiendas se sientan en
        kilómetros $1$, $3$, $4$, $9$, $13$ de un camino, y un
        El almacén debe minimizar el \emph{distancia total} al
        cinco tiendas. Calcule ese total para un almacén en el
        \omterm{def:g10:stats:median}{mediana} ($4$), en la \omterm{def:g10:stats:mean}{media} ($6$), y en $5$. ¿Quién gana?
        (Esto es general: la \omterm{def:g10:stats:median}{mediana} minimiza total absoluto

  \item La cabina, resuelta: supongamos que un piloto tiene un $30\,\%$
        probabilidad de estar "cerca del promedio" en un cuerpo

  \item Final --- la guía del usuario, una línea cada una: cuando totales

        reportar el \dots; cuando la contaminación amenace, utilice el
        \dots. Cierra el caso con el título del problema.
\end{enumerate}
\end{problem}
