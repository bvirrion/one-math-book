\chapter{Estadística descriptiva}\label{ch:g10:stats}

La estadística resume una colección grande de números (notas, salarios,
temperaturas) mediante unos pocos indicadores bien elegidos. Este capítulo
presenta las dos familias de indicadores: los que localizan el
\emph{centro} de los datos (\omterm{def:g10:stats:mean}{media}, \omterm{def:g10:stats:median}{mediana}) y los que miden su
\emph{dispersión} (\omterm{def:g10:stats:quartiles}{recorrido}, \omterm{def:g10:stats:quartiles}{cuartiles}). Un tratamiento más profundo, con
la desviación típica, llega en el \cref{ch:g11:stat}.

\section{Datos y frecuencias}

\begin{definition}[Serie estadística]\label{def:g10:stats:series}
Una \emph{serie estadística}\index{serie estadística} es una lista de
valores $x_1, x_2, \dots, x_N$ observados en una población de tamaño $N$.
Cuando un valor $x$ aparece $n$ veces, $n$ es su \emph{frecuencia
absoluta} y el cociente
\[
f = \frac{n}{N}
\]
es su \emph{frecuencia relativa}\index{frecuencia relativa} (que a menudo
se expresa como porcentaje). Las frecuencias relativas de todos los
valores suman $1$.
\end{definition}

\begin{example}\label{ex:g10:stats:frequencies}
Las notas de $20$ alumnos en una prueba sobre $5$ puntos:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|cccccc}
nota $x$ & $0$ & $1$ & $2$ & $3$ & $4$ & $5$ \\
\hline
frec.\ absoluta $n$ & $1$ & $2$ & $4$ & $6$ & $5$ & $2$ \\
frec.\ relativa $f$ & $0.05$ & $0.10$ & $0.20$ & $0.30$ & $0.25$ & $0.10$ \\
\end{tabular}
\end{center}
Comprobación: $1 + 2 + 4 + 6 + 5 + 2 = 20$ y las \omterm{def:g10:stats:series}{frecuencias relativas}
suman $1$.
\end{example}

\section{Medidas de centralización}

\begin{definition}[Media]\label{def:g10:stats:mean}
La \emph{media}\index{media} de la serie es
\[
\bar x = \frac{x_1 + x_2 + \dots + x_N}{N}.
\]
Si los valores $v_1, \dots, v_k$ aparecen con frecuencias absolutas
$n_1, \dots, n_k$, el mismo número se calcula como la \emph{media
ponderada}
\[
\bar x = \frac{n_1 v_1 + n_2 v_2 + \dots + n_k v_k}{n_1 + n_2 + \dots + n_k}.
\]
\end{definition}

\begin{example}\label{ex:g10:stats:mean}
Para la prueba del \cref{ex:g10:stats:frequencies}:
\[
\bar x = \frac{1 \times 0 + 2 \times 1 + 4 \times 2 + 6 \times 3
+ 5 \times 4 + 2 \times 5}{20}
= \frac{0 + 2 + 8 + 18 + 20 + 10}{20}
= \frac{58}{20} = 2.9 .
\]
\end{example}

\begin{definition}[Mediana]\label{def:g10:stats:median}
La \emph{mediana}\index{mediana} es un valor que parte en dos mitades la
serie \emph{ordenada}: al menos la mitad de los valores son $\leq$ que la
mediana y al menos la mitad son $\geq$ que ella. En la práctica, ordena
los $N$ valores;
\begin{itemize}
  \item si $N$ es impar, la mediana es el valor central, en la posición
        $\frac{N+1}{2}$;
  \item si $N$ es par, se toma el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de los dos valores
        centrales, en las posiciones $\frac N2$ y $\frac N2 + 1$.
\end{itemize}
\end{definition}

\begin{example}\label{ex:g10:stats:median}
Serie ordenada de $7$ precios de vivienda (en miles): $120$, $150$, $160$,
$180$, $210$, $240$, $900$. La \omterm{def:g10:stats:median}{mediana} es el $4$.\textsuperscript{o}
valor, $180$. La \omterm{def:g10:stats:mean}{media} es $\frac{1960}{7} = 280$, ¡mayor que $6$ de los
$7$ precios! Un valor extremo ($900$) arrastra a la \omterm{def:g10:stats:mean}{media} mucho más que a
la \omterm{def:g10:stats:median}{mediana}: la \omterm{def:g10:stats:median}{mediana} es \emph{robusta}; la \omterm{def:g10:stats:mean}{media}, no.
\end{example}

\section{Medidas de dispersión}

\begin{definition}[Recorrido, cuartiles]\label{def:g10:stats:quartiles}
Para una serie ordenada:
\begin{itemize}
  \item el \emph{recorrido}\index{recorrido} es la diferencia entre el
        mayor y el menor de los valores;
  \item el \emph{primer cuartil}\index{cuartil} $Q_1$ es el menor valor
        tal que al menos una cuarta parte ($25\,\%$) de los valores son
        $\leq Q_1$; el \emph{tercer cuartil} $Q_3$ es el menor valor tal
        que al menos tres cuartas partes ($75\,\%$) de los valores son
        $\leq Q_3$;
  \item el \emph{recorrido intercuartílico}\index{recorrido
        intercuartílico} $Q_3 - Q_1$ mide la dispersión de la mitad
        central de los datos.
\end{itemize}
\end{definition}

\begin{method}[Hallar los cuartiles]\label{met:g10:stats:quartiles}
Para una serie de $N$ valores ordenados:
\begin{enumerate}
  \item calcula $\frac N4$; si no es entero, redondéalo \emph{hacia
        arriba} al entero siguiente; $Q_1$ es el valor de esa posición;
  \item calcula $\frac{3N}{4}$; redondea hacia arriba si hace falta;
        $Q_3$ es el valor de esa posición.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:stats:quartiles}
Tomemos los $11$ valores ordenados
\[
2,\ 3,\ 5,\ 5,\ 6,\ 7,\ 8,\ 9,\ 9,\ 10,\ 12 .
\]
\omterm{def:g10:stats:median}{Mediana}: posición $\frac{11+1}{2} = 6$, así que la \omterm{def:g10:stats:median}{mediana} es $7$.
$Q_1$: $\frac{11}{4} = 2.75$, que se redondea a $3$; el $3.$\textsuperscript{er}
valor es $5$. $Q_3$: $\frac{33}{4} = 8.25$, que se redondea a $9$; el
$9.$\textsuperscript{o} valor es $9$. \omterm{def:g10:stats:quartiles}{Recorrido}: $12 - 2 = 10$; \omterm{def:g10:stats:quartiles}{recorrido
intercuartílico}: $9 - 5 = 4$.
\end{example}

\begin{example}[Comparar dos grupos]\label{ex:g10:stats:comparison}
Dos clases hacen el mismo examen (notas sobre $20$):
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccc}
 & \omterm{def:g10:stats:median}{mediana} & $Q_1$ & $Q_3$ \\
\hline
clase A & $12$ & $10$ & $14$ \\
clase B & $12$ & $6$ & $17$ \\
\end{tabular}
\end{center}
La misma \omterm{def:g10:stats:median}{mediana} y dispersiones muy distintas: la clase A es homogénea
(la mitad de sus notas están en $\intcc{10}{14}$) y la clase B es
heterogénea (su mitad central abarca $\intcc{6}{17}$). Los indicadores de
centralización, por sí solos, nunca cuentan toda la historia.
\end{example}

\begin{remark}
Ni la \omterm{def:g10:stats:mean}{media} ni la \omterm{def:g10:stats:median}{mediana} son ``mejores'': la \omterm{def:g10:stats:mean}{media} usa todos los valores
(y hace falta para calcular totales), la \omterm{def:g10:stats:median}{mediana} resiste los valores
extremos. Un resumen serio de una serie da al menos un indicador de
centralización \emph{y} uno de dispersión.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:stats:1}
Estos son los números de libros leídos en un año por $15$ alumnos:
\[
0,\ 1,\ 1,\ 2,\ 2,\ 2,\ 3,\ 3,\ 4,\ 4,\ 5,\ 6,\ 7,\ 8,\ 12 .
\]
Calcula la \omterm{def:g10:stats:mean}{media}, la \omterm{def:g10:stats:median}{mediana} y el \omterm{def:g10:stats:quartiles}{recorrido} de esta serie.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:2}
Se ha lanzado un dado $50$ veces:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|cccccc}
resultado & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ \\
\hline
frecuencia & $7$ & $9$ & $8$ & $10$ & $6$ & $10$ \\
\end{tabular}
\end{center}
Calcula la \omterm{def:g10:stats:series}{frecuencia relativa} de cada resultado y el resultado medio.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:3}
Halla la \omterm{def:g10:stats:median}{mediana}, $Q_1$ y $Q_3$ de la serie ordenada
\[
1,\ 2,\ 4,\ 4,\ 5,\ 6,\ 6,\ 7,\ 8,\ 9,\ 9,\ 11 \qquad (N = 12),
\]
y dibuja el diagrama de caja correspondiente.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:4}
La \omterm{def:g10:stats:mean}{media} de $24$ notas de un examen es $11.5$. Un $25$.\textsuperscript{o}
alumno hace el examen y saca $19$. ¿Cuál es la nueva \omterm{def:g10:stats:mean}{media} de la clase?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:5}
Un alumno tiene una \omterm{def:g10:stats:mean}{media} de $12$ después de $5$ exámenes, todos con el
mismo peso. ¿Qué nota en el $6$.\textsuperscript{o} examen subiría la
\omterm{def:g10:stats:mean}{media} a $13$? ¿Es alcanzable una \omterm{def:g10:stats:mean}{media} de $14$ (las notas son sobre
$20$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:6}
Inventa una serie de $7$ valores no negativos cuya \omterm{def:g10:stats:median}{mediana} sea $10$ y cuya
\omterm{def:g10:stats:mean}{media} sea mayor que $20$; después otra cuya \omterm{def:g10:stats:median}{mediana} sea $10$ y cuya \omterm{def:g10:stats:mean}{media}
sea menor que $6$. ¿Por qué la \omterm{def:g10:stats:mean}{media} de una serie así nunca puede ser
menor que $\frac{40}{7}$? ¿Qué muestran estos ejemplos?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:7}
En una empresa, los $9$ empleados cobran $1800$ cada uno al mes y el
director cobra $12000$.
\begin{enumerate}
  \item Calcula el salario medio y el salario mediano.
  \item ¿Qué indicador describe mejor un salario ``típico'' aquí? ¿Por
        qué?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:8}
Los $12$ chicos de una clase miden de \omterm{def:g10:stats:mean}{media} $168$ cm; las $18$ chicas de
la misma clase miden de \omterm{def:g10:stats:mean}{media} $163$ cm. Calcula la estatura \omterm{def:g10:stats:mean}{media} de toda
la clase. (Cuidado: \emph{no} es el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $168$ y $163$; hay que
ponderar las \omterm{def:g10:stats:mean}{medias} por los tamaños de los grupos.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:stats:9}
Una serie de $N$ valores $x_1, \dots, x_N$ tiene \omterm{def:g10:stats:mean}{media} $\bar x$. Cada
valor se transforma en $y_i = a x_i + b$ con $a$ y $b$ números fijos.
\begin{enumerate}
  \item Demuestra que la \omterm{def:g10:stats:mean}{media} de la nueva serie es $a \bar x + b$.
  \item Las temperaturas de una semana, en grados Celsius, tienen \omterm{def:g10:stats:mean}{media}
        $20$. ¿Cuál es su \omterm{def:g10:stats:mean}{media} en grados Fahrenheit
        ($F = 1.8\,C + 32$)?
\end{enumerate}
\end{exercise}

\section{Problema: La persona media no existe}

\begin{problem}[{Problema de fin de semana --- media contra mediana:
valores atípicos, curvas de notas, el almacén en la carretera y la cabina
diseñada para nadie}]
\label{pb:g10:stats:1}
En 1950, la Fuerza Aérea estadounidense midió a $4\,000$ pilotos en diez
dimensiones corporales y construyó la cabina para el \emph{hombre medio}.
Le servía, como descubrió un teniente contando, prácticamente a nadie. La
estadística resume multitudes con números sueltos, y cada resumen dice la
verdad sobre algo y miente sobre otra cosa. Este problema lleva a juicio a
la \omterm{def:g10:stats:mean}{media} y a la \omterm{def:g10:stats:median}{mediana} (\cref{def:g10:stats:mean},
\cref{def:g10:stats:median}), deja que cada una enseñe en qué es la mejor
y cierra el caso de la cabina.

\medskip
\noindent\textbf{Parte I --- Dos resúmenes en el banquillo.}
\begin{enumerate}
  \item En una calle se venden siete casas, en miles de euros: $200$,
        $210$, $220$, $230$, $240$, $250$ y un caserón de $1\,200$.
        Calcula el precio medio y el precio mediano. ¿Qué número describe
        ``una casa de esta calle''?
  \item Quita el caserón y vuelve a calcular los dos. ¿Cuánto se ha movido
        cada resumen? Enuncia la moraleja sobre los valores atípicos.
  \item Una tabla del censo: de $100$ familias, $20$ no tienen hijos, $25$
        tienen uno, $30$ tienen dos, $15$ tienen tres y $10$ tienen
        cuatro. Calcula el número medio de hijos. Ninguna familia ``tiene
        $1.7$ hijos'', así que ¿qué cantidad real codifica la \omterm{def:g10:stats:mean}{media}?
        (Multiplícala por $100$.)
  \item Para las quince notas
        $4, 6, 7, 8, 9, 10, 10, 11, 12, 12, 13, 14, 15, 17, 19$: halla la
        \omterm{def:g10:stats:median}{mediana} y los \omterm{def:g10:stats:quartiles}{cuartiles} $Q_1$ y $Q_3$
        (\cref{met:g10:stats:quartiles}).
  \item Para la misma serie, da el \omterm{def:g10:stats:quartiles}{recorrido} y el \omterm{def:g10:stats:quartiles}{recorrido
        intercuartílico}. ¿Qué mide cada uno?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Curvas de notas y clases fusionadas.}
\begin{enumerate}[resume]
  \item A partir del \cref{exo:g10:stats:9}: desplazar todos los valores
        en $+b$ desplaza en $+b$ la \omterm{def:g10:stats:mean}{media}, la \omterm{def:g10:stats:median}{mediana} y los \omterm{def:g10:stats:quartiles}{cuartiles};
        multiplicarlos por $a$ los multiplica por $a$. ¿Qué les ocurre al
        \omterm{def:g10:stats:quartiles}{recorrido} y al \omterm{def:g10:stats:quartiles}{recorrido intercuartílico} con un desplazamiento?
        ¿Y con un cambio de escala? Justifícalo.
  \item Un examen tiene \omterm{def:g10:stats:mean}{media} $9$ (sobre $20$) y el profesor quiere \omterm{def:g10:stats:mean}{media}
        $12$. Se proponen dos correcciones: sumar $3$ puntos a todo el
        mundo o multiplicar cada nota por $\frac43$. Calcula qué le ocurre
        a un alumno con $6$ y a otro con $18$ con cada corrección. ¿A
        quién favorece cada una y cuál tiene un problema de techo?
  \item La clase A ($20$ alumnos) tiene \omterm{def:g10:stats:mean}{media} $12$; la clase B ($30$
        alumnos) tiene \omterm{def:g10:stats:mean}{media} $10$. Calcula la \omterm{def:g10:stats:mean}{media} de los $50$ alumnos
        juntos y explica por qué no es $11$ (la lección de la \omterm{def:g10:stats:mean}{media}
        ponderada del problema de fin de semana del volumen anterior sobre
        la \omterm{def:g10:stats:mean}{media} armónica, llevada al aula).
  \item Comprueba que las \omterm{def:g10:stats:median}{medianas} se niegan a este juego: calcula las
        \omterm{def:g10:stats:median}{medianas} de $A = \{1, 2, 3\}$ y $B = \{0, 10\}$, después la
        \omterm{def:g10:stats:median}{mediana} de la lista fusionada, y compárala con la \omterm{def:g10:stats:mean}{media} ponderada
        por tamaños de las dos \omterm{def:g10:stats:median}{medianas}.
  \item Patinaje artístico: cinco jueces puntúan $5.2$, $5.3$, $5.4$,
        $5.5$, $9.9$. Calcula la \omterm{def:g10:stats:mean}{media}, la \omterm{def:g10:stats:median}{mediana} y la \emph{\omterm{def:g10:stats:mean}{media}
        recortada} (se descartan la nota más baja y la más alta y se
        promedian las demás). ¿Por qué recortan casi todos los deportes
        con jueces?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Leer distribuciones.}
\begin{enumerate}[resume]
  \item Dos fábricas producen tornillos de longitud \omterm{def:g10:stats:mean}{media} $50$ mm. Fábrica
        A: $Q_1 = 49.9$, $Q_3 = 50.1$; fábrica B: $Q_1 = 48$, $Q_3 = 52$.
        Necesitas tornillos intercambiables: ¿qué fábrica eliges y qué
        estadístico ha decidido?
  \item En un ambulatorio, la espera \omterm{def:g10:stats:median}{mediana} es de $10$ minutos, pero la
        espera \omterm{def:g10:stats:mean}{media} es de $25$. ¿Qué forma de los datos delata esta
        pareja de números? Construye un conjunto de cinco valores con
        exactamente esa \omterm{def:g10:stats:median}{mediana} y esa \omterm{def:g10:stats:mean}{media}.
  \item ``Tu bebé está en el percentil 75 de estatura'': ¿qué significa
        esa frase? ¿Y puede estar por encima de la \omterm{def:g10:stats:mean}{media} todo niño de una
        ciudad? ¿Y por encima de la \omterm{def:g10:stats:median}{mediana}? Explica la diferencia.
  \item Clase A: mín $4$, $Q_1\,8$, \omterm{def:g10:stats:median}{mediana} $11$, $Q_3\,13$, máx $17$.
        Clase B: mín $2$, $Q_1\,9$, \omterm{def:g10:stats:median}{mediana} $11$, $Q_3\,12$, máx $19$.
        Compara: nota típica, homogeneidad de la mitad central y
        salvajismo de los extremos.
  \item La lista del periodista: un titular dice ``salario medio en
        TechCorp: $5\,000$ euros''. Enumera las tres preguntas que este
        problema te ha enseñado a hacer antes de creer que el titular dice
        algo sobre un empleado típico.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El caso de la cabina.}
\begin{enumerate}[resume]
  \item Fabrica tus propias pruebas: construye cinco precios de vivienda
        con \omterm{def:g10:stats:mean}{media} $300$ y \omterm{def:g10:stats:median}{mediana} $200$ (miles de euros).
  \item La \omterm{def:g10:stats:mean}{media} es el punto de equilibrio: demuestra a partir de la
        definición que las desviaciones respecto de la \omterm{def:g10:stats:mean}{media} siempre se
        cancelan, $\sum (x_i - \bar x) = 0$, y compruébalo con tu conjunto
        de datos de la pregunta 16.
  \item La \omterm{def:g10:stats:median}{mediana} es la que minimiza los recados: hay tiendas en los
        kilómetros $1$, $3$, $4$, $9$ y $13$ de una carretera, y un
        almacén debe minimizar la \emph{distancia total} a las cinco
        tiendas. Calcula esa distancia total con el almacén en la \omterm{def:g10:stats:median}{mediana}
        ($4$), en la \omterm{def:g10:stats:mean}{media} ($6$) y en $5$. ¿Quién gana? (Es un hecho
        general: la \omterm{def:g10:stats:median}{mediana} minimiza la distancia total en \omterm{def:g10:numbers:abs}{valor absoluto};
        prueba con cualquier otro punto.)
  \item La cabina, resuelta: supón que un piloto tiene un $30\,\%$ de
        probabilidad de estar ``cerca de la \omterm{def:g10:stats:mean}{media}'' en una dimensión
        corporal, de forma independiente en las $10$ dimensiones. Estima
        la probabilidad de estar cerca de la \omterm{def:g10:stats:mean}{media} en \emph{las diez} y
        concilia el resultado con lo que encontró el teniente entre
        $4\,000$ pilotos. ¿Qué construyó la Fuerza Aérea en lugar de la
        cabina \omterm{def:g10:stats:mean}{media}?
  \item Final, el manual de uso, una línea para cada caso: cuando lo que
        importa son los totales (un presupuesto, una cosecha), usa
        la\,\dots; cuando importa el individuo típico (salarios, tiempos
        de espera), usa la\,\dots; cuando importa la equidad de la
        dispersión, informa de los\,\dots; cuando amenaza la
        contaminación, usa la\,\dots\ Cierra el caso con el título del
        problema.
\end{enumerate}
\end{problem}
