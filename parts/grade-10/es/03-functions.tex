\chapter{Funciones}\label{ch:g10:functions}

Una \omterm{def:g10:functions:function}{función} es una máquina que recibe un número y devuelve otro, siempre el
mismo resultado para la misma entrada. Las funciones son los objetos
centrales del análisis que irás construyendo en los próximos años (desde
el \cref{ch:g11:quad} en adelante); este capítulo monta el vocabulario
(\omterm{def:g10:functions:function}{dominio}, \omterm{def:g10:functions:function}{imagen}, \omterm{def:g10:functions:graph}{gráfica}, variación) y entrena la destreza esencial de
leer información en una \omterm{def:g10:functions:graph}{gráfica}.

\section{Vocabulario: imágenes y antiimágenes}

\begin{definition}[Función]\label{def:g10:functions:function}
Sea $D$ un conjunto de \omterm{def:g10:numbers:sets}{números reales}. Una \emph{función}\index{función}
$f$ definida en $D$ asocia a cada número $x \in D$ exactamente un \omterm{def:g10:numbers:sets}{número
real}, que se escribe $f(x)$ y se llama \emph{imagen}\index{imagen} de $x$.
El conjunto $D$ es el \emph{dominio}\index{dominio} de $f$. Escribimos
\[
f : x \longmapsto f(x).
\]
Si $f(x) = y$, se dice que $x$ es una
\emph{antiimagen}\index{antiimagen} de $y$.
\end{definition}

\begin{remark}
Cada $x$ del \omterm{def:g10:functions:function}{dominio} tiene exactamente una \omterm{def:g10:functions:function}{imagen}, pero un número $y$
puede tener varias antiimágenes o ninguna. Para $f(x) = x^2$: la \omterm{def:g10:functions:function}{imagen} de
$3$ es $9$, y $9$ tiene dos antiimágenes, $3$ y $-3$; el número $-4$ no
tiene ninguna.
\end{remark}

\begin{example}[Hallar un dominio]\label{ex:g10:functions:domain}
Cuando una \omterm{def:g10:functions:function}{función} viene dada por una fórmula, su \omterm{def:g10:functions:function}{dominio} es el conjunto
de los $x$ para los que la fórmula tiene sentido.
\begin{itemize}
  \item $f(x) = 3x^2 - 5x + 1$ tiene sentido para todo $x$: el \omterm{def:g10:functions:function}{dominio} es
        $\R$.
  \item $g(x) = \dfrac{1}{x - 2}$ exige $x - 2 \neq 0$: el \omterm{def:g10:functions:function}{dominio} es
        $\R$ salvo $2$.
  \item $h(x) = \sqrt{x - 3}$ exige $x - 3 \geq 0$: el \omterm{def:g10:functions:function}{dominio} es
        $\intco{3}{+\infty}$.
\end{itemize}
\end{example}

\begin{example}[Calcular imágenes y antiimágenes]\label{ex:g10:functions:images}
Sea $f(x) = x^2 - 4x + 3$, definida en $\R$.

\emph{\omterm{def:g10:functions:function}{Imagen} de $5$:} sustituimos $x = 5$:
$f(5) = 25 - 20 + 3 = 8$.

\emph{Antiimágenes de $3$:} resolvemos $f(x) = 3$:
\[
x^2 - 4x + 3 = 3
\quad\Longleftrightarrow\quad
x^2 - 4x = 0
\quad\Longleftrightarrow\quad
x(x - 4) = 0,
\]
así que las antiimágenes de $3$ son $0$ y $4$.
\end{example}

\section{La gráfica de una función}

\begin{definition}[Gráfica]\label{def:g10:functions:graph}
En un sistema de coordenadas, la \emph{gráfica}\index{gráfica} (o
\emph{curva}) de $f$ es el conjunto de todos los puntos $(x, f(x))$ con
$x$ en el \omterm{def:g10:functions:function}{dominio}. Dicho de otro modo, un punto $(x, y)$ está en la
gráfica exactamente cuando $y = f(x)$.
\end{definition}

    -- (axis cs:0,1.25);
  \fill[omThm] (axis cs:3,1.25) circle (1.5pt);
  \node[omThm, font=\small, above right] at (axis cs:3,1.2)
    {$(3,\,f(3))$};
\end{axis}
\end{tikzpicture}

{\small Cómo se lee una \omterm{def:g10:functions:function}{imagen} en una \omterm{def:g10:functions:graph}{gráfica}: se parte de $x = 3$ en el
eje horizontal, se sube verticalmente hasta la curva y se va después en
horizontal hasta el eje vertical para leer $f(3) = 1.25$.}
\end{omfigure}

\begin{method}[Leer una gráfica]\label{met:g10:functions:reading}
Sobre la \omterm{def:g10:functions:graph}{gráfica} de $f$:
\begin{enumerate}
  \item \emph{\omterm{def:g10:functions:function}{imagen} de $a$}: sube verticalmente desde $x = a$ en el eje
        horizontal hasta la curva y ve después en horizontal hasta el eje
        vertical; el valor que leas allí es $f(a)$;
  \item \emph{antiimágenes de $b$}: traza la recta horizontal $y = b$; las
        antiimágenes son las abscisas de todos sus puntos de corte con la
        curva;
  \item \emph{soluciones de $f(x) = k$}: son las antiimágenes de $k$;
  \item \emph{soluciones de $f(x) \leq k$}: los $x$ para los que la curva
        está sobre la recta $y = k$ o por debajo de ella.
\end{enumerate}
\end{method}

\begin{tikzpicture}
\begin{axis}[omaxis,

  \fill (axis cs:-2,2) circle (1.5pt);
  \fill (axis cs:2,2) circle (1.5pt);

  \draw[dashed] (axis cs:2,2) -- (axis cs:2,0);
\end{axis}
\end{tikzpicture}

{\small Resolución \omterm{def:g10:functions:graph}{gráfica} de $f(x) = k$: las soluciones $x_1$ y $x_2$ son
las abscisas de los puntos de corte de la curva con la recta horizontal
$y = k$. Aquí $f(x) \leq k$ se cumple en $\intcc{x_1}{x_2}$, donde la
curva está por debajo de la recta.}
\end{omfigure}

\begin{example}\label{ex:g10:functions:membership}
¿Está el punto $A(2, 5)$ en la \omterm{def:g10:functions:graph}{gráfica} de $f(x) = x^2 + 1$? Calculamos
$f(2) = 4 + 1 = 5$: sí, porque $f(2)$ coincide con la ordenada de $A$. El
punto $B(3, 8)$ no está en la \omterm{def:g10:functions:graph}{gráfica}, porque $f(3) = 10 \neq 8$.
\end{example}

\section{Variación de una función}

\begin{definition}[Creciente, decreciente]\label{def:g10:functions:variations}
Sea $f$ definida en un \omterm{def:g10:numbers:interval}{intervalo} $I$.
\begin{itemize}
  \item $f$ es \emph{creciente}\index{función creciente} en $I$ cuando,
        para todos $u, v \in I$, si $u < v$ entonces $f(u) \leq f(v)$: los
        resultados crecen con las entradas y la \omterm{def:g10:functions:graph}{gráfica} sube de izquierda
        a derecha.
  \item $f$ es \emph{decreciente}\index{función decreciente} en $I$ cuando
        $u < v$ implica $f(u) \geq f(v)$: la \omterm{def:g10:functions:graph}{gráfica} baja de izquierda a
        derecha.
\end{itemize}
Con desigualdades estrictas $f(u) < f(v)$ (resp.\ $>$) se dice
\emph{estrictamente} creciente (resp.\ decreciente).
\end{definition}

\begin{definition}[Tabla de variación]\label{def:g10:functions:table}
Una \emph{tabla de variación}\index{tabla de variación} resume en qué
\omterm{def:g10:numbers:interval}{intervalos} crece y decrece $f$, mediante las flechas $\nearrow$ y
$\searrow$, y recoge los valores de $f$ en los puntos donde cambia el
sentido.
\end{definition}

\begin{example}\label{ex:g10:functions:table}
Consideremos la \omterm{def:g10:functions:function}{función} cuya \omterm{def:g10:functions:graph}{gráfica} aparece abajo, en $\intcc{-2}{4}$.

\begin{omfigure}

  xmin=-2.8, xmax=4.6, ymin=-2.4, ymax=3.8,
  xtick={-2,1,4}, ytick={-1.5,3}, yticklabels={$-1.5$,$3$}]
  \addplot[omDef, thick, domain=-2:4, samples=120] {-0.5*(x-1)^2 + 3};
  \fill (axis cs:-2,-1.5) circle (1.5pt);

  \fill (axis cs:1,3) circle (1.5pt);
\end{axis}
\end{tikzpicture}

{\small Una \omterm{def:g10:functions:function}{función} definida en $\intcc{-2}{4}$, \omterm{def:g10:functions:variations}{creciente} primero y
\omterm{def:g10:functions:variations}{decreciente} después.}
\end{omfigure}

Leyendo la \omterm{def:g10:functions:graph}{gráfica}: $f$ crece desde $f(-2) = -1.5$ hasta su \omterm{def:g10:functions:extrema}{máximo}
$f(1) = 3$ y decrece después hasta $f(4) = -1.5$. Su \omterm{def:g10:functions:table}{tabla de variación} es
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccc}
$x$ & $-2$ & & $1$ & & $4$ \\
\hline
$f$ & $-1.5$ & $\nearrow$ & $3$ & $\searrow$ & $-1.5$ \\
\end{tabular}
\end{center}
\end{example}

\begin{example}[Demostrar una variación]\label{ex:g10:functions:proof}
Demostremos que $f(x) = 3x + 1$ es estrictamente \omterm{def:g10:functions:variations}{creciente} en $\R$.
Tomamos $u < v$ cualesquiera y comparamos las imágenes:
\[
f(v) - f(u) = (3v + 1) - (3u + 1) = 3(v - u) > 0,
\]
porque $v - u > 0$. Luego $f(u) < f(v)$: la \omterm{def:g10:functions:function}{función} es estrictamente
\omterm{def:g10:functions:variations}{creciente}. El mismo cálculo con pendiente negativa, por ejemplo
$g(x) = -2x + 5$, da $g(v) - g(u) = -2(v-u) < 0$: estrictamente
\omterm{def:g10:functions:variations}{decreciente}.
\end{example}

\section{Extremos}

\begin{definition}[Máximo, mínimo]\label{def:g10:functions:extrema}
Sea $f$ definida en $D$ y sea $a \in D$. El valor $f(a)$ es el
\emph{máximo}\index{máximo} de $f$ en $D$ cuando $f(x) \leq f(a)$ para
todo $x \in D$; es el \emph{mínimo}\index{mínimo} cuando
$f(x) \geq f(a)$ para todo $x \in D$. En una \omterm{def:g10:functions:graph}{gráfica} son el punto más alto
y el más bajo de la curva.
\end{definition}

\begin{example}\label{ex:g10:functions:extremum}
Demostremos que $f(x) = (x - 1)^2 + 2$ tiene \omterm{def:g10:functions:extrema}{mínimo} $2$ en $\R$, alcanzado
en $x = 1$. Para todo $x$, el cuadrado $(x-1)^2$ es $\geq 0$, luego
$f(x) = (x-1)^2 + 2 \geq 2$; y $f(1) = 0 + 2 = 2$, así que el valor $2$ se
alcanza de verdad. Las dos cosas juntas demuestran que $2$ es el \omterm{def:g10:functions:extrema}{mínimo}.
\end{example}

\begin{omfigure}
\begin{tikzpicture}
\begin{axis}[omaxis,
  width=7.6cm, height=5.2cm,
  xmin=-1.8, xmax=3.8, ymin=-0.9, ymax=6.4,
  xtick={1}, ytick={2}]
  \addplot[omDef, thick, domain=-1.0:3.0, samples=100] {(x-1)^2 + 2};
  \addplot[omThm, dashed, domain=-1.6:3.6] {2};
  \fill (axis cs:1,2) circle (1.5pt);
  \node[below right, font=\small] at (axis cs:1,2) {$(1,2)$};
\end{axis}
\end{tikzpicture}

{\small La curva de $f(x) = (x-1)^2 + 2$ nunca baja de la recta discontinua
$y = 2$ y la toca en $x = 1$: el \omterm{def:g10:functions:extrema}{mínimo} de $f$ es $2$.}
\end{omfigure}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:functions:1}
Sea $f(x) = 2x^2 - 3x + 1$. Calcula las imágenes de $0$, $2$, $-1$ y
$\frac12$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:2}
Da el \omterm{def:g10:functions:function}{dominio} de cada \omterm{def:g10:functions:function}{función}:
\[
f(x) = 5x - 2, \qquad
g(x) = \frac{3}{x + 4}, \qquad
h(x) = \sqrt{2x - 6}, \qquad
k(x) = \frac{1}{x^2 + 1}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:3}
Sea $f(x) = x^2 - 2x$. Halla todas las antiimágenes de $0$, de $3$ y de
$-1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:4}
¿Pertenece el punto $A(-1, 4)$ a la \omterm{def:g10:functions:graph}{gráfica} de $f(x) = x^2 - 2x + 1$? ¿Y
el punto $B(2, 1)$? Justifica con un cálculo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:5}
Una \omterm{def:g10:functions:function}{función} $g$ definida en $\intcc{-3}{5}$ tiene la \omterm{def:g10:functions:table}{tabla de variación}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccccc}
$x$ & $-3$ & & $0$ & & $3$ & & $5$ \\
\hline
$g$ & $1$ & $\searrow$ & $-2$ & $\nearrow$ & $4$ & $\searrow$ & $0$ \\
\end{tabular}
\end{center}
\begin{enumerate}
  \item ¿Cuáles son el \omterm{def:g10:functions:extrema}{máximo} y el \omterm{def:g10:functions:extrema}{mínimo} de $g$ en $\intcc{-3}{5}$?
  \item Compara $g(-1)$ y $g(-0.5)$ sin conocer sus valores.
  \item ¿Cuántas soluciones puede tener como mucho la \omterm{def:g10:algebra:equation}{ecuación} $g(x) = 0$
        en cada \omterm{def:g10:numbers:interval}{intervalo} de la tabla?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:6}
Sea $f(x) = -2x + 7$. Demuestra, comparando $f(u)$ y $f(v)$ para $u < v$,
que $f$ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:7}
Demuestra que la \omterm{def:g10:functions:function}{función} $f(x) = x^2 + 6x + 11$ tiene un \omterm{def:g10:functions:extrema}{mínimo} en $\R$ y
da su valor y el punto donde se alcanza. (Indicación: escribe
$f(x) = (x + 3)^2 + c$ con la constante $c$ adecuada.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:8}
Un jardín rectangular tiene perímetro $40$ m. Sea $x$ su anchura, en
metros.
\begin{enumerate}
  \item Expresa el largo y después el área $A(x)$ en \omterm{def:g10:functions:function}{función} de $x$. ¿Para
        qué valores de $x$ tiene sentido?
  \item Calcula $A(4)$, $A(8)$, $A(10)$, $A(12)$ y $A(16)$. ¿Qué
        conjeturas sobre la forma de área máxima?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:9}
Sea $f(x) = \dfrac{1}{x^2 + 1}$, definida en $\R$.
\begin{enumerate}
  \item Demuestra que $0 < f(x) \leq 1$ para todo $x$ real.
  \item ¿Tiene $f$ \omterm{def:g10:functions:extrema}{máximo} en $\R$? ¿Y \omterm{def:g10:functions:extrema}{mínimo}? Justifica la respuesta.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:functions:10}
Sea $f(x) = x^2$ en $\R$.
\begin{enumerate}
  \item Sea $0 \leq u < v$. Factoriza $f(v) - f(u)$ y deduce que $f$ es
        estrictamente \omterm{def:g10:functions:variations}{creciente} en $\intco{0}{+\infty}$.
  \item Adapta el argumento para demostrar que $f$ es estrictamente
        \omterm{def:g10:functions:variations}{decreciente} en $\intoc{-\infty}{0}$.
\end{enumerate}
\end{exercise}

\section{Problema: Máquinas alimentadas con su propio resultado}

\begin{problem}[{Problema de fin de semana --- puntos fijos, iteración y
tres máquinas célebres: la de Herón, la que eleva al cuadrado y la del
granizo, todavía sin resolver}]
\label{pb:g10:functions:1}
Una \omterm{def:g10:functions:function}{función} es una máquina: entra un número, sale un número. Los
experimentos más interesantes consisten en alimentar la máquina con
\emph{su propio resultado}, una y otra vez; entonces dominan dos
preguntas: ¿acaba el proceso en algún sitio (un \emph{\omterm{pb:g10:functions:1}{punto fijo}})?, ¿y
cómo lo lleva hasta allí la variación de la máquina? Una de las máquinas
de este problema lleva dos mil años puliendo raíces cuadradas; otra
custodia el problema sin resolver más famoso que ofrecen las matemáticas
elementales.

\medskip
\noindent\textbf{Parte I --- La máquina de Herón.}
Consideremos la \omterm{def:g10:functions:function}{función}
\[
f(x) = \frac12\left(x + \frac2x\right).
\]
\begin{enumerate}
  \item Da el \omterm{def:g10:functions:function}{dominio} de $f$ y calcula $f(1)$, $f(2)$ y
        $f\!\left(\frac32\right)$ como fracciones exactas.
  \item Halla todas las antiimágenes de $\frac32$: resuelve
        $f(x) = \frac32$ (quita el denominador y usa los métodos del
        \cref{pb:g10:algebra:1}).
  \item Un \emph{punto fijo}\index{punto fijo} de $f$ es un número al que
        la máquina no cambia: $f(x) = x$. Halla los dos \omterm{pb:g10:functions:1}{puntos fijos} de
        $f$. ¿Qué viejo conocido custodia el positivo?
  \item Partiendo de $x = 1$, alimenta la máquina con su propio resultado
        tres veces y anota los resultados exactos. ¿Qué sucesión del
        problema de fin de semana del volumen anterior sobre la
        irracionalidad has reconstruido? ¿Y qué era ``la receta de Herón''
        en el lenguaje de las funciones?
  \item En un mismo dibujo, esboza la \omterm{def:g10:functions:graph}{gráfica} de $f$ (para $x > 0$) y la
        recta diagonal $y = x$. ¿Dónde están los \omterm{pb:g10:functions:1}{puntos fijos} en ese
        dibujo? (Compara con el problema de fin de semana del volumen
        anterior sobre los dos termómetros: la temperatura que marcaba lo
        mismo en las dos escalas era la misma idea.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Por qué la máquina no puede fallar.}
\begin{enumerate}[resume]
  \item Para $0 < u < v$, demuestra, reduciendo a común denominador, que
        \[
        f(v) - f(u) = \frac{(v - u)(uv - 2)}{2uv} ,
        \]
        y deduce la variación de $f$ en $\intoo{0}{+\infty}$: \omterm{def:g10:functions:variations}{decreciente}
        hasta un punto y \omterm{def:g10:functions:variations}{creciente} después. ¿Qué punto? (Misma técnica
        que en el \cref{exo:g10:functions:10}.)
  \item Deduce que en $\intoo{0}{+\infty}$ la \omterm{def:g10:functions:function}{función} $f$ tiene un \omterm{def:g10:functions:extrema}{mínimo},
        que vale exactamente $\sqrt2$ y se alcanza en $\sqrt2$. ¿Qué dice
        esto sobre \emph{todos} los resultados de la máquina (con entradas
        positivas)?
  \item Demuestra que si $x > \sqrt2$, entonces $\sqrt2 < f(x) < x$: una
        estimación demasiado grande siempre mejora y nunca se pasa de
        largo. (Para $f(x) < x$, compara $\frac2x$ con $x$.)
  \item Reúne las preguntas 6 a 8 en la \omterm{def:g10:functions:table}{tabla de variación} de $f$ en
        $\intoo{0}{+\infty}$ (\cref{def:g10:functions:table}), \omterm{def:g10:functions:extrema}{mínimo}
        incluido.
  \item Una segunda máquina: $g(x) = x^2$. Sus \omterm{pb:g10:functions:1}{puntos fijos} son $0$ y $1$
        (compruébalo). Itera $g$ cuatro veces desde $x = 0.9$ y otras
        cuatro desde $x = 1.1$ (con calculadora, tres decimales). Un \omterm{pb:g10:functions:1}{punto
        fijo} atrae y el otro repele: ¿cuál es cuál?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La máquina del granizo.}
Sobre los \omterm{def:g10:numbers:sets}{números enteros} positivos definimos $h(n) = \frac n2$ si $n$ es
par y $h(n) = 3n + 1$ si $n$ es impar. Bajo $h$, los números rebotan como
granizos dentro de una nube de tormenta.
\begin{enumerate}[resume]
  \item Calcula el vuelo completo de $7$ bajo $h$, hasta llegar a $1$.
        ¿Cuántos pasos tarda y hasta qué altura sube?
  \item Empieza el vuelo de $27$ y calcula diez pasos. (Su vuelo completo
        dura $111$ pasos y llega hasta $9\,232$, partiendo de $27$.) ¿Qué
        lección sobre las reglas sencillas enseñan $7$ y $27$?
  \item Explica por qué todo vuelo que alcanza una potencia de $2$ se
        precipita directamente hasta $1$, y qué ocurre en $1$ (calcula
        $h(1)$, $h(4)$, $h(2)$). ¿Tiene $h$ algún \omterm{pb:g10:functions:1}{punto fijo} entre los
        enteros positivos? ¿Qué forma en su lugar el final de todos los
        vuelos observados?
  \item La \emph{conjetura de Collatz} afirma que todo número de partida
        acaba llegando a $1$. Los ordenadores la han verificado mucho más
        allá de $10^{20}$; nadie la ha demostrado. ¿Qué dos tipos de
        hallazgo la \emph{refutarían}? ¿Y por qué una verificación masiva
        sigue sin cerrar el caso (el problema de fin de semana del volumen
        anterior sobre el oráculo de las cerillas ya tocó esta campana)?
  \item ``Las matemáticas todavía no están maduras para estas
        preguntas'', dijo Paul Erd\H{o}s de esta conjetura. En una frase:
        ¿qué separa la máquina del granizo de la de Herón, donde las
        preguntas 6 a 8 lo resolvieron todo?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Manual de uso de las máquinas.}
\begin{enumerate}[resume]
  \item \omterm{def:g10:functions:function}{Dominios} como \omterm{def:g10:numbers:interval}{intervalos} (en el lenguaje del
        \cref{pb:g10:numbers:1}): da los \omterm{def:g10:functions:function}{dominios} de
        $x \mapsto \sqrt{x - 3}$, \
        $x \mapsto \frac{1}{x^2 - 9}$, \
        $x \mapsto \sqrt{9 - x^2}$.
  \item Volvamos a la $f$ de Herón: halla las antiimágenes exactas de $3$
        (resuelve $f(x) = 3$ completando el cuadrado).
  \item Se lanza una piedra hacia arriba: su altura es
        $H(t) = 20t - 5t^2$ metros al cabo de $t$ segundos. ¿En qué
        \omterm{def:g10:numbers:interval}{intervalo} de tiempo tiene sentido físico la fórmula? Completa el
        cuadrado para hallar la altura máxima y el instante en que se
        alcanza (\cref{def:g10:functions:extrema}) y da la \omterm{def:g10:functions:table}{tabla de
        variación}.
  \item ¿Durante cuánto tiempo está la piedra al menos a $15$ m de altura?
        (Resuelve $H(t) \geq 15$ con una \omterm{def:g10:algebra:signtable}{tabla de signos},
        \cref{met:g10:algebra:signtable}.)
  \item Final: este problema ha puesto en marcha cuatro máquinas: la de
        Herón, la que eleva al cuadrado, la del granizo y la altura de la
        piedra. En dos o tres frases, di qué tiene toda \omterm{def:g10:functions:function}{función} (un
        \omterm{def:g10:functions:function}{dominio}, una regla, una \omterm{def:g10:functions:graph}{gráfica}) y cuáles son las dos grandes
        preguntas que este problema ha hecho a cada una (¿dónde acaba la
        iteración?, ¿cómo varía la \omterm{def:g10:functions:function}{función}?), preguntas que los capítulos
        siguientes retoman una a una.
\end{enumerate}
\end{problem}
