\chapter{Vectores}\label{ch:g10:vectors}

A \omterm{def:g10:vectors:vector}{vector} codifica un desplazamiento: una dirección y una longitud, independientemente de
el punto de partida. \omterm{def:g10:vectors:vector}{vectores} dar pruebas limpias de hechos geométricos y, una vez
\omterm{def:g10:coordgeom:system}{coordenadas} entre en escena, redúzcalos a pequeños cálculos. ellos
será una herramienta clave en \cref{ch:g10:lines} y, posteriormente, para el escalar
producto en \cref{ch:g11:scal}.

\section{Traducciones y vectores}

\begin{definition}[Vector]\label{def:g10:vectors:vector}
Dados dos puntos $A $ y $ B $, el \emph{vector}\index{vector} $\vect{AB}$
representa el desplazamiento de $ A $ a $ B $: tiene un
\emph{dirección} (la dirección de la línea $(AB)$, junto con la forma
de recorrerlo, desde $ A $ hacia $ B $) y un
\emph{longitud} (la distancia $ AB $, también escrito $\norm{\vect{AB}}$).
Dos vectores son \emph{igual} cuando codifican el mismo desplazamiento:
$\vect{AB} = \vect{CD}$ significa que aplicando los dos desplazamientos a cualquier
punto da el mismo resultado.
\end{definition}

\begin{proposition}[Vectores iguales y paralelogramos]\label{prop:g10:vectors:parallelogram}
$\vect{AB} = \vect{CD}$ exactamente cuando $ ABDC $ (¡en este orden!) es un
paralelogramo, posiblemente uno aplanado --- equivalentemente, cuando $[AD]$ y
$[BC]$ tener lo mismo \omterm{prop:g10:coordgeom:midpoint}{punto medio}.
\end{proposition}
\admitted

\begin{notation}
A \omterm{def:g10:vectors:vector}{vector} a menudo se nombra con una sola letra, $\vec u $ o $\vec v $, cuando el
los puntos finales no importan. El \emph{cero \omterm{def:g10:vectors:vector}{vector}} $\vec 0 = \vect{AA}$ es
el desplazamiento que no mueve nada. El \emph{opuesto} de
$\vect{AB}$ es $\vect{BA}$: misma longitud, en sentido opuesto. escribimos
$\vect{BA} = -\vect{AB}$.
\end{notation}

\section{Añadiendo vectores}

\begin{definition}[Suma de dos vectores]\label{def:g10:vectors:sum}
El \emph{suma}\index{suma} $\vec u + \vec v $ es el desplazamiento
obtenido realizando $\vec u $ entonces $\vec v $.
\end{definition}

\begin{theorem}[La relación de Chasles]\label{thm:g10:vectors:chasles}
Por tres puntos cualesquiera $ A $, $ B $, $ C $:
\[
\vect{AB} + \vect{BC} = \vect{AC}.
\]
\end{theorem}

\begin{proof}
Pasando de $ A $ a $ B $, entonces de $ B $ a $ C $, es un desplazamiento que toma
$ A $ a $ C $: es el desplazamiento codificado por $\vect{AC}$.
\end{proof}

\begin{remark}[regla del paralelogramo]
para sumar dos \omterm{def:g10:vectors:vector}{vectores} dibujado desde el mismo punto, $\vect{AB} + \vect{AC}$,
completar el paralelogramo $ ABDC $: el \omterm{def:g10:vectors:sum}{suma} es la diagonal $\vect{AD}$.
En efecto $\vect{AC} = \vect{BD}$, entonces
$\vect{AB} + \vect{AC} = \vect{AB} + \vect{BD} = \vect{AD}$ por Chasles'
relación.
\end{remark}

\begin{example}[Simplificando con Chasles]\label{ex:g10:vectors:chasles}
Simplificar $\vect{MN} + \vect{NP} + \vect{PQ}$: encadenando los desplazamientos
da $\vect{MQ}$. Simplificar $\vect{AB} - \vect{AC}$: reescribir el
diferencia como un \omterm{def:g10:vectors:sum}{suma} con lo contrario \omterm{def:g10:vectors:vector}{vector},
\[
\vect{AB} - \vect{AC} = \vect{AB} + \vect{CA}
= \vect{CA} + \vect{AB} = \vect{CB}.
\]
\end{example}

\section{Multiplicar un vector por un número}

\begin{definition}[múltiplo escalar]\label{def:g10:vectors:scalar}
Sea $\vec u $ ser distinto de cero \omterm{def:g10:vectors:vector}{vector} y $ k $ a número real. El \omterm{def:g10:vectors:vector}{vector}
$ k\vec u $\index{vectorial!escalar múltiplo} tiene la misma dirección que
$\vec u $, longitud $\abs{k} \times \norm{\vec u}$, y apunta en la misma dirección
como $\vec u $ si $ k > 0$, al revés si $ k < 0$. Para $ k = 0$,
$0\vec u = \vec 0$.
\end{definition}

\begin{definition}[Colinealidad]\label{def:g10:vectors:collinear}
Dos \omterm{def:g10:vectors:vector}{vectores} $\vec u $ y $\vec v $ son
\emph{colineal}\index{colineal} cuando tienen lo mismo
dirección, es decir, cuando $\vec v = k \vec u $ por algo real $ k $ (o uno de
ellos son $\vec 0$).
\end{definition}

\begin{remark}
La colinealidad es la \omterm{def:g10:vectors:vector}{vector} lenguaje para paralelismo y alineación:
\begin{itemize}
  \item las lineas $(AB)$ y $(CD)$ son paralelos exactamente cuando $\vect{AB}$
        y $\vect{CD}$ son \omterm{def:g10:vectors:collinear}{colineales};
  \item los puntos $ A $, $ B $, $ C $ están alineados exactamente cuando $\vect{AB}$ y
        $\vect{AC}$ son \omterm{def:g10:vectors:collinear}{colineales}.
\end{itemize}
\end{remark}

\section{Coordenadas de vectores}

\begin{definition}[Coordenadas de un vector]\label{def:g10:vectors:coords}
en un \omterm{def:g10:coordgeom:system}{sistema de coordenadas}, el \emph{\omterm{def:g10:coordgeom:system}{coordenadas}} del \omterm{def:g10:vectors:vector}{vector} $\vect{AB}$
son los números que describen el desplazamiento:
\[
\vect{AB}\,(x_B - x_A,\ y_B - y_A).
\]
A \omterm{def:g10:vectors:vector}{vector} $\vec u\,(a, b)$ mueve cada punto $ a $ unidades horizontalmente y
$ b $ unidades verticalmente.
\end{definition}

\begin{proposition}[Computando con coordenadas]\label{prop:g10:vectors:coordrules}
Sea $\vec u\,(a, b)$ y $\vec v\,(c, d)$, y $ k \in \R $.
\begin{enumerate}
  \item $\vec u = \vec v $ exactamente cuando $ a = c $ y $ b = d $;
  \item $\vec u + \vec v $ tiene \omterm{def:g10:coordgeom:system}{coordenadas} $(a + c,\ b + d)$;
  \item $ k\vec u $ tiene \omterm{def:g10:coordgeom:system}{coordenadas} $(ka,\ kb)$;
  \item $\norm{\vec u} = \sqrt{a^2 + b^2}$ (sistema \omterm{def:g10:coordgeom:system}{ortonormal}).
\end{enumerate}
\end{proposition}

\begin{proof}
1--3 reformula, coordenada por coordenada, lo que hacen los desplazamientos: por
instancia realizando $\vec u $ entonces $\vec v $ mueve un punto horizontalmente por
$ a $ entonces $ c $, por lo tanto por $ a + c $ en total. El punto 4 es la fórmula de la distancia.
de \cref{thm:g10:coordgeom:distance} aplicado a un segmento que representa
$\vec u $.
\end{proof}

\begin{theorem}[Criterio de colinealidad]\label{thm:g10:vectors:det}
Dos \omterm{def:g10:vectors:vector}{vectores} $\vec u\,(a, b)$ y $\vec v\,(c, d)$ son \omterm{def:g10:vectors:collinear}{colineales} si y
solo si
\[
ad - bc = 0 .
\]
\end{theorem}

\begin{proof}
Si $\vec v = k\vec u $, entonces $ c = ka $ y $ d = kb $, entonces
$ ad - bc = a(kb) - b(ka) = 0$. Lo mismo vale si $\vec u = k \vec v $ o si
ya sea \omterm{def:g10:vectors:vector}{vector} es cero.

Por el contrario, supongamos $ ad - bc = 0$ con $\vec u \neq \vec 0$, decir
$ a \neq 0$ (el caso $ b \neq 0$ es similar). Colocar $ k = \frac{c}{a}$; entonces
$ c = ka $, y $ ad = bc = b(ka)$ da, después de dividir por $ a \neq 0$,
$ d = kb $. Entonces $\vec v = k\vec u $.
\end{proof}

\begin{example}\label{ex:g10:vectors:det}
Son $\vec u\,(3, -2)$ y $\vec v\,(-6, 4)$ \omterm{def:g10:vectors:collinear}{colineales}? Calcular
$ ad - bc = 3 \times 4 - (-2)\times(-6) = 12 - 12 = 0$: si, y de hecho
$\vec v = -2\vec u $. Para $\vec u\,(3, -2)$ y $\vec w\,(1, 5)$:
$3 \times 5 - (-2) \times 1 = 17 \neq 0$: no \omterm{def:g10:vectors:collinear}{colineales}.
\end{example}

\begin{method}[Alineación y paralelismo]\label{met:g10:vectors:alignment}
Para demostrar que tres puntos $ A $, $ B $, $ C $ están alineados:
\begin{enumerate}
  \item calcular el \omterm{def:g10:coordgeom:system}{coordenadas} de $\vect{AB}$ y $\vect{AC}$;
  \item comprobar el criterio $ ad - bc = 0$ de \cref{thm:g10:vectors:det};
  \item concluir: el \omterm{def:g10:vectors:vector}{vectores} son \omterm{def:g10:vectors:collinear}{colineales} y compartir el punto $ A $, entonces
        Los tres puntos están alineados.
\end{enumerate}
El mismo cálculo con $\vect{AB}$ y $\vect{CD}$ prueba que el
lineas $(AB)$ y $(CD)$ son paralelos.
\end{method}

\begin{example}\label{ex:g10:vectors:alignment}
Sea $ A(1, 2)$, $ B(3, 5)$ y $ C(7, 11)$. Entonces $\vect{AB}\,(2, 3)$ y
$\vect{AC}\,(6, 9)$, y $2 \times 9 - 3 \times 6 = 18 - 18 = 0$: el
Los puntos están alineados (de hecho. $\vect{AC} = 3\vect{AB}$).
\end{example}

\section{Problema: el álgebra de las flechas}

\begin{problem}[{Problema del fin de semana --- Gimnasia Chasles, el
centroide re-probado en dos líneas, y fuerzas en equilibrio: ¿qué
los vectores son realmente para}]
\label{pb:g10:vectors:1}
El octavo grado demostró que las medianas de un triángulo se encuentran en dos tercios.
de su longitud --- con un ingenioso paralelogramo escondido en su interior
el triángulo (el problema del centroide del fin de semana del volumen de la escuela secundaria). \omterm{def:g10:vectors:vector}{vectores} reprobarlo en
dos líneas y reparte la concurrencia gratis. Eso es lo que
Esta nueva álgebra es para: los teoremas se convierten en cálculos con
flechas. Este problema entrena el cálculo (Chasles sobre todo),
ofrece la prueba de dos líneas, amplía Varignon
(\cref{exo:g10:vectors:10}), y termina donde \omterm{def:g10:vectors:vector}{vectores} nacieron:
fuerzas y velocidades.

\medskip
\noindent\textbf{Part I --- Chasles gymnastics.}
\begin{enumerate}
  \item Simplifica usando la relación de Chasles
        (\cref{thm:g10:vectors:chasles}):
        $\vect{AB} + \vect{BC} + \vect{CD}$; \quad
        $\vect{MA} - \vect{MB}$; \quad
        $\vect{AB} + \vect{CD} + \vect{BC} + \vect{DA}$.
  \item El truco del origen: demostrar que para \emph{cualquier} punto $ O $,
        $\vect{AB} = \vect{OB} - \vect{OA}$.
  \item Pruebe las dos caracterizaciones de la \omterm{prop:g10:coordgeom:midpoint}{punto medio} $ I $ de
        $[AB]$:
        \[
        \vect{IA} + \vect{IB} = \vec 0
        \qquad\text{and}\qquad
        \vect{OI} = \tfrac12\left(\vect{OA} + \vect{OB}\right)
        \ \text{for any } O .
        \]
  \item Usando $\vect{AB} = \vect{DC}$, vuelve a encontrar el cuarto vértice
        $ D $ del paralelogramo $ ABCD $ con $ A(-1, 2)$,
        $ B(3, 4)$, $ C(6, 0)$ --- y comparar con el
        respuesta de \omterm{prop:g10:coordgeom:midpoint}{punto medio} de \cref{pb:g10:coordgeom:1}.
  \item Agregar un fijo \omterm{def:g10:vectors:vector}{vector} $\vec u $ a cada punto del
        plano realiza qué transformación --- la que
        descubierto con dos espejos en el problema de fin de semana de dos espejos del volumen de Escuela Secundaria
        y dos medias vueltas en el problema de fin de semana de medias vueltas del volumen de la escuela secundaria? calcular
        el \omterm{def:g10:functions:function}{imagen} de $(1, 2)$ bajo $\vec u = (3, -1)$.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The centroid, third proof.}
Definir el punto $ G $ de un triangulo $ ABC $ por el elegante
condición
\[
\vect{GA} + \vect{GB} + \vect{GC} = \vec 0 .
\]
\begin{enumerate}[resume]
  \item Usando el truco del origen, demuestre que esta condición
        determina $ G $ únicamente:
        $\vect{OG} = \frac13\left(\vect{OA} + \vect{OB} +
        \vect{OC}\right)$ para cualquier $ O $ --- entonces el \omterm{def:g10:coordgeom:system}{coordenadas} de
        $ G $ son los \emph{promedios} de los vértices'
        (el problema del centroide del fin de semana del volumen de la escuela secundaria lo vio numéricamente).
  \item Probar $\vect{AG} = \frac13\left(\vect{AB} +
        \vect{AC}\right)$ y, escribiendo $K $ para el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de
        $[BC]$, $\vect{AK} = \frac12\left(\vect{AB} +
        \vect{AC}\right)$. Concluir:
        $\vect{AG} = \frac23\,\vect{AK}$ --- los dos tercios
        teorema, re-probado.
  \item ¿Por qué las medianas de $ B $ y de $ C $ pasar por
        el \emph{mismo} $ G $ sin más cálculos?
        Compare las tres pruebas disponibles de este teorema ---
        paralelogramo oculto de grado 8, \omterm{def:g10:coordgeom:system}{coordenadas}, \omterm{def:g10:vectors:vector}{vectores} ---
        en una frase cada uno.
  \item Verificación numérica: $ A(1, 1)$, $ B(5, 2)$, $ C(3, 6)$.
        Calcular $ G $, entonces $ K $, luego verificar
        $\vect{AG} = \frac23 \vect{AK}$ con la colinealidad
        criterio (\cref{thm:g10:vectors:det}).
  \item Lecturas de física $ G $ como el punto de equilibrio de tres iguales
        masas en los vértices. Duplicar la masa en $ A $ (masas
        $2, 1, 1$): el punto de equilibrio se convierte en
        $\vect{OG'} = \frac{2\vect{OA} + \vect{OB} +
        \vect{OC}}{4}$. Calcular $ G'$ para el triangulo de
        pregunta 9 y comparar su posición con $ G $.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Collinearity at work.}
\begin{enumerate}[resume]
  \item Son $\vec u(3, -2)$ y $\vec v(-6, 4)$ \omterm{def:g10:vectors:collinear}{colineales}? Y
        $\vec w(2, 5)$, $\vec z(4, 9)$? decidir con el
        criterio determinante.
  \item son los puntos $ A(1, 2)$, $ B(3, 5)$, $ C(7, 11)$
        ¿alineado?
  \item Más allá de Varignon (\cref{exo:g10:vectors:10}): en cualquier
        cuadrilátero $ ABCD $, el \emph{bimedianas} únete al
        puntos medios de lados opuestos ($[IK]$ y $[JL]$). Usando
        posición \omterm{def:g10:vectors:vector}{vectores} ($\vect{OI} = \frac12(\vect{OA} +
        \vect{OB})$, etc.), muestran que ambas bimedianas tienen la
        lo mismo \omterm{prop:g10:coordgeom:midpoint}{punto medio}, de posición \omterm{def:g10:vectors:vector}{vector}
        $\frac14\left(\vect{OA} + \vect{OB} + \vect{OC} +
        \vect{OD}\right)$: el centroide del cuadrilátero.
  \item Tales en una línea: $ M $ y $ N $ son los puntos definidos
        por $\vect{AM} = \frac13\vect{AB}$ y
        $\vect{AN} = \frac13\vect{AC}$. Calcular
        $\vect{MN}$ en términos de $\vect{BC}$ y concluir
        paralelismo y proporción.
  \item Generalice la pregunta 14 a una proporción arbitraria $ k $, y
        indicar en una frase cómo \omterm{def:g10:vectors:vector}{vector} el cálculo absorbe
        el \omterm{prop:g10:coordgeom:midpoint}{punto medio} teorema (el \omterm{prop:g10:coordgeom:midpoint}{punto medio} teorema)
        y Tales (teorema de Tales) a la vez.
\end{enumerate}

\noindent\textbf{Part IV --- Forces and velocities.}
\begin{enumerate}[resume]
  \item Un barco es empujado por una corriente de $3$ km/h hacia el este
        mientras remaba en $4$ km/h hacia el norte. dar el resultante
        velocidad \omterm{def:g10:vectors:vector}{vector}, su magnitud y describir la magnitud real
        curso. (Aparece un triple muy antiguo.)
  \item Sobre un anillo actúan tres fuerzas: $\vec u(2, 1)$,
        $\vec v(-3, 2)$, $\vec w(1, -3)$. Calcula la resultante.
        ¿Qué hace el anillo? ¿Y por qué es el \omterm{def:g10:vectors:sum}{suma} de las flechas
        a lo largo de cualquier \emph{cerrado} polígono siempre $\vec 0$
        (Chasles)?
  \item Dos fuerzas $\vec u(4, -1)$ y $\vec v(-1, 3)$ actuar sobre un
        gancho. ¿Qué tercera fuerza mantiene el gancho en equilibrio?
  \item Un nadador cruza un río en línea recta a
        $2$ EM; la corriente corre a $1.5$ EM. Dale el
        velocidad resultante y su magnitud. el río es
        $50$ m de ancho: ¿cuánto dura el cruce y cuánto tiempo
        ¿Qué tan abajo aterriza el nadador?
  \item Final: un objeto, tres disfraces --- desplazamiento
        flecha, par de \omterm{def:g10:coordgeom:system}{coordenadas}, fuerza o velocidad. Enumere el
        tres teoremas clásicos, este problema volvió a demostrarse con
        \omterm{def:g10:vectors:vector}{vector} álgebra y nombra la cantidad geométrica como
        \omterm{def:g10:vectors:vector}{vectores} de este capítulo aún no puede medir --- el
        herramienta que lo soluciona llega con el producto escalar, en
        grado 11.
\end{enumerate}
\end{problem}
