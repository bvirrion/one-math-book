\chapter{Vectores}\label{ch:g10:vectors}

Un \omterm{def:g10:vectors:vector}{vector} codifica un desplazamiento: una dirección y una longitud, sin
importar el punto de partida. Los \omterm{def:g10:vectors:vector}{vectores} dan demostraciones limpias de
hechos geométricos y, en cuanto entran las \omterm{def:g10:coordgeom:system}{coordenadas}, los reducen a
cálculos breves. Serán una herramienta clave en el \cref{ch:g10:lines} y,
más adelante, para el producto escalar del \cref{ch:g11:scal}.

\section{Traslaciones y vectores}

\begin{definition}[Vector]\label{def:g10:vectors:vector}
Dados dos puntos $A$ y $B$, el \emph{vector}\index{vector} $\vect{AB}$
representa el desplazamiento de $A$ a $B$: tiene una \emph{dirección} (la
de la recta $(AB)$, junto con el sentido en que se recorre, de $A$ hacia
$B$) y una \emph{longitud} (la distancia $AB$, que también se escribe
$\norm{\vect{AB}}$). Dos vectores son \emph{iguales} cuando codifican el
mismo desplazamiento: $\vect{AB} = \vect{CD}$ significa que aplicar los
dos desplazamientos a un punto cualquiera da el mismo resultado.
\end{definition}

\begin{proposition}[Vectores iguales y paralelogramos]\label{prop:g10:vectors:parallelogram}
$\vect{AB} = \vect{CD}$ exactamente cuando $ABDC$ (¡en este orden!) es un
paralelogramo, quizá aplastado; equivalentemente, cuando $[AD]$ y $[BC]$
tienen el mismo \omterm{prop:g10:coordgeom:midpoint}{punto medio}.
\end{proposition}
\admitted

\begin{notation}
Un \omterm{def:g10:vectors:vector}{vector} se nombra a menudo con una sola letra, $\vec u$ o $\vec v$,
cuando los extremos no importan. El \emph{\omterm{def:g10:vectors:vector}{vector} nulo}
$\vec 0 = \vect{AA}$ es el desplazamiento que no mueve nada. El
\emph{opuesto} de $\vect{AB}$ es $\vect{BA}$: la misma longitud, el
sentido contrario. Escribimos $\vect{BA} = -\vect{AB}$.
\end{notation}

\section{Suma de vectores}

\begin{definition}[Suma de dos vectores]\label{def:g10:vectors:sum}
La \emph{suma}\index{vector!suma} $\vec u + \vec v$ es el desplazamiento
que se obtiene efectuando $\vec u$ y después $\vec v$.
\end{definition}

\begin{theorem}[Relación de Chasles]\label{thm:g10:vectors:chasles}
Para tres puntos cualesquiera $A$, $B$, $C$:
\[
\vect{AB} + \vect{BC} = \vect{AC}.
\]
\end{theorem}

\begin{proof}
Ir de $A$ a $B$ y después de $B$ a $C$ es un desplazamiento que lleva $A$
hasta $C$: es el desplazamiento codificado por $\vect{AC}$.
\end{proof}

\begin{remark}[Regla del paralelogramo]
Para sumar dos \omterm{def:g10:vectors:vector}{vectores} trazados desde el mismo punto,
$\vect{AB} + \vect{AC}$, completa el paralelogramo $ABDC$: la \omterm{def:g10:vectors:sum}{suma} es la
diagonal $\vect{AD}$. En efecto, $\vect{AC} = \vect{BD}$, luego
$\vect{AB} + \vect{AC} = \vect{AB} + \vect{BD} = \vect{AD}$ por la
relación de Chasles.
\end{remark}

\begin{example}[Simplificar con Chasles]\label{ex:g10:vectors:chasles}
Simplifiquemos $\vect{MN} + \vect{NP} + \vect{PQ}$: encadenando los
desplazamientos queda $\vect{MQ}$. Simplifiquemos $\vect{AB} - \vect{AC}$:
reescribimos la diferencia como \omterm{def:g10:vectors:sum}{suma} con el \omterm{def:g10:vectors:vector}{vector} opuesto,
\[
\vect{AB} - \vect{AC} = \vect{AB} + \vect{CA}
= \vect{CA} + \vect{AB} = \vect{CB}.
\]
\end{example}

\section{Producto de un vector por un número}

\begin{definition}[Producto por un escalar]\label{def:g10:vectors:scalar}
Sean $\vec u$ un \omterm{def:g10:vectors:vector}{vector} no nulo y $k$ un \omterm{def:g10:numbers:sets}{número real}. El \omterm{def:g10:vectors:vector}{vector}
$k\vec u$\index{vector!producto por un escalar} tiene la misma dirección
que $\vec u$, longitud $\abs{k} \times \norm{\vec u}$ y el mismo sentido
que $\vec u$ si $k > 0$, o el contrario si $k < 0$. Para $k = 0$,
$0\vec u = \vec 0$.
\end{definition}

\begin{definition}[Colinealidad]\label{def:g10:vectors:collinear}
Dos \omterm{def:g10:vectors:vector}{vectores} $\vec u$ y $\vec v$ son
\emph{colineales}\index{vectores colineales} cuando tienen la misma
dirección, es decir, cuando $\vec v = k \vec u$ para algún real $k$ (o
cuando uno de ellos es $\vec 0$).
\end{definition}

\begin{remark}
La colinealidad es el lenguaje vectorial del paralelismo y de la
alineación:
\begin{itemize}
  \item las rectas $(AB)$ y $(CD)$ son paralelas exactamente cuando
        $\vect{AB}$ y $\vect{CD}$ son \omterm{def:g10:vectors:collinear}{colineales};
  \item los puntos $A$, $B$, $C$ están alineados exactamente cuando
        $\vect{AB}$ y $\vect{AC}$ son \omterm{def:g10:vectors:collinear}{colineales}.
\end{itemize}
\end{remark}

\section{Coordenadas de un vector}

\begin{definition}[Coordenadas de un vector]\label{def:g10:vectors:coords}
En un \omterm{def:g10:coordgeom:system}{sistema de coordenadas}, las \emph{\omterm{def:g10:coordgeom:system}{coordenadas}} del \omterm{def:g10:vectors:vector}{vector}
$\vect{AB}$ son los números que describen el desplazamiento:
\[
\vect{AB}\,(x_B - x_A,\ y_B - y_A).
\]
Un \omterm{def:g10:vectors:vector}{vector} $\vec u\,(a, b)$ mueve cada punto $a$ unidades en horizontal y
$b$ unidades en vertical.
\end{definition}

\begin{proposition}[Cálculo con coordenadas]\label{prop:g10:vectors:coordrules}
Sean $\vec u\,(a, b)$ y $\vec v\,(c, d)$, y sea $k \in \R$.
\begin{enumerate}
  \item $\vec u = \vec v$ exactamente cuando $a = c$ y $b = d$;
  \item $\vec u + \vec v$ tiene \omterm{def:g10:coordgeom:system}{coordenadas} $(a + c,\ b + d)$;
  \item $k\vec u$ tiene \omterm{def:g10:coordgeom:system}{coordenadas} $(ka,\ kb)$;
  \item $\norm{\vec u} = \sqrt{a^2 + b^2}$ (en un \omterm{def:g10:coordgeom:system}{sistema ortonormal}).
\end{enumerate}
\end{proposition}

\begin{proof}
Los puntos 1 a 3 reformulan, coordenada a coordenada, lo que hacen los
desplazamientos: por ejemplo, efectuar $\vec u$ y después $\vec v$ mueve
un punto en horizontal primero $a$ y luego $c$, o sea $a + c$ en total. El
punto 4 es la fórmula de la distancia del
\cref{thm:g10:coordgeom:distance} aplicada a un segmento que representa
$\vec u$.
\end{proof}

\begin{theorem}[Criterio de colinealidad]\label{thm:g10:vectors:det}
Dos \omterm{def:g10:vectors:vector}{vectores} $\vec u\,(a, b)$ y $\vec v\,(c, d)$ son \omterm{def:g10:vectors:collinear}{colineales} si y solo
si
\[
ad - bc = 0 .
\]
\end{theorem}

\begin{proof}
Si $\vec v = k\vec u$, entonces $c = ka$ y $d = kb$, luego
$ad - bc = a(kb) - b(ka) = 0$. Lo mismo ocurre si $\vec u = k \vec v$ o si
alguno de los dos \omterm{def:g10:vectors:vector}{vectores} es nulo.

Recíprocamente, supongamos $ad - bc = 0$ con $\vec u \neq \vec 0$,
digamos $a \neq 0$ (el caso $b \neq 0$ es análogo). Tomamos
$k = \frac{c}{a}$; entonces $c = ka$, y de $ad = bc = b(ka)$ se obtiene,
dividiendo entre $a \neq 0$, $d = kb$. Luego $\vec v = k\vec u$.
\end{proof}

\begin{example}\label{ex:g10:vectors:det}
¿Son \omterm{def:g10:vectors:collinear}{colineales} $\vec u\,(3, -2)$ y $\vec v\,(-6, 4)$? Calculamos
$ad - bc = 3 \times 4 - (-2)\times(-6) = 12 - 12 = 0$: sí, y en efecto
$\vec v = -2\vec u$. Para $\vec u\,(3, -2)$ y $\vec w\,(1, 5)$:
$3 \times 5 - (-2) \times 1 = 17 \neq 0$: no son \omterm{def:g10:vectors:collinear}{colineales}.
\end{example}

\begin{method}[Alineación y paralelismo]\label{met:g10:vectors:alignment}
Para demostrar que tres puntos $A$, $B$, $C$ están alineados:
\begin{enumerate}
  \item calcula las \omterm{def:g10:coordgeom:system}{coordenadas} de $\vect{AB}$ y de $\vect{AC}$;
  \item comprueba el criterio $ad - bc = 0$ del
        \cref{thm:g10:vectors:det};
  \item concluye: los \omterm{def:g10:vectors:vector}{vectores} son \omterm{def:g10:vectors:collinear}{colineales} y comparten el punto $A$,
        así que los tres puntos están alineados.
\end{enumerate}
El mismo cálculo con $\vect{AB}$ y $\vect{CD}$ demuestra que las rectas
$(AB)$ y $(CD)$ son paralelas.
\end{method}

\begin{example}\label{ex:g10:vectors:alignment}
Sean $A(1, 2)$, $B(3, 5)$ y $C(7, 11)$. Entonces $\vect{AB}\,(2, 3)$ y
$\vect{AC}\,(6, 9)$, y $2 \times 9 - 3 \times 6 = 18 - 18 = 0$: los puntos
están alineados (de hecho, $\vect{AC} = 3\vect{AB}$).
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g10:vectors:1}
Simplifica usando la relación de Chasles:
\[
\vect{AB} + \vect{BD}, \qquad
\vect{KL} + \vect{LM} + \vect{MK}, \qquad
\vect{AC} - \vect{BC}, \qquad
\vect{AB} + \vect{CA}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:vectors:2}
Sean $A(2, 1)$, $B(5, 3)$, $C(-1, 4)$. Calcula las \omterm{def:g10:coordgeom:system}{coordenadas} de
$\vect{AB}$, $\vect{BC}$ y $\vect{AC}$, y comprueba sobre las \omterm{def:g10:coordgeom:system}{coordenadas}
que $\vect{AB} + \vect{BC} = \vect{AC}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:vectors:3}
Sean $\vec u\,(2, -3)$ y $\vec v\,(-1, 4)$. Da las \omterm{def:g10:coordgeom:system}{coordenadas} de
$\vec u + \vec v$, $3\vec u$ y $2\vec u - \vec v$, y calcula
$\norm{\vec u}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:vectors:4}
En cada caso, di si $\vec u$ y $\vec v$ son \omterm{def:g10:vectors:collinear}{colineales}:
\[
\text{(a) } \vec u\,(4, 6),\ \vec v\,(6, 9);
\qquad
\text{(b) } \vec u\,(2, -5),\ \vec v\,(-4, 10);
\qquad
\text{(c) } \vec u\,(3, 1),\ \vec v\,(1, 3).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:vectors:5}
Sean $A(0, 2)$, $B(4, 0)$, $C(6, 3)$. Halla las \omterm{def:g10:coordgeom:system}{coordenadas} del punto $D$
tal que $ABCD$ sea un paralelogramo, es decir, tal que
$\vect{AB} = \vect{DC}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:vectors:6}
Sean $A(-2, 1)$, $B(1, 3)$, $C(4, -1)$.
\begin{enumerate}
  \item Calcula las \omterm{def:g10:coordgeom:system}{coordenadas} del punto $M$ tal que
        $\vect{AM} = \vect{AB} + \vect{AC}$.
  \item ¿Qué es el cuadrilátero $ABMC$? Justifícalo.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:vectors:7}
Sean $A(1, -1)$, $B(4, 1)$, $C(10, 5)$. ¿Están alineados los puntos $A$,
$B$, $C$? La misma pregunta para $A(0, 3)$, $B(2, 2)$, $C(8, -1)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:vectors:8}
Sean $A(1, 2)$, $B(5, 4)$, $C(6, 1)$, $D(2, -1)$. Demuestra que $(AB)$ y
$(CD)$ son paralelas y después que $ABCD$ es un paralelogramo. ¿Cuál de
las dos comprobaciones implica la otra?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:vectors:9}
Sea $ABC$ un triángulo y sea $I$ el punto definido por
$\vect{AI} = \frac23 \vect{AB}$. Con $A(0,0)$, $B(6, 3)$, $C(2, 5)$ (para
que los cálculos sean sencillos):
\begin{enumerate}
  \item calcula las \omterm{def:g10:coordgeom:system}{coordenadas} de $I$;
  \item sea $J$ tal que $\vect{CJ} = \frac23\vect{CB}$; calcula las
        \omterm{def:g10:coordgeom:system}{coordenadas} de $J$;
  \item demuestra que $(IJ)$ es paralela a $(AC)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:vectors:10}
Sea $ABCD$ un cuadrilátero cualquiera y sean $I$, $J$, $K$, $L$ los
\omterm{prop:g10:coordgeom:midpoint}{puntos medios} de $[AB]$, $[BC]$, $[CD]$, $[DA]$. Usando \omterm{def:g10:coordgeom:system}{coordenadas}
$A(x_A, y_A)$, etc., demuestra que $\vect{IJ} = \vect{LK}$ y concluye que
los \omterm{prop:g10:coordgeom:midpoint}{puntos medios} de los lados de \emph{cualquier} cuadrilátero forman un
paralelogramo.
\end{exercise}

\section{Problema: El álgebra de las flechas}

\begin{problem}[{Problema de fin de semana --- gimnasia con Chasles, el
baricentro demostrado en dos líneas y fuerzas en equilibrio: para qué
sirven de verdad los vectores}]
\label{pb:g10:vectors:1}
En el volumen anterior se demostró que las medianas de un triángulo se
cortan a dos tercios de su longitud, con un ingenioso paralelogramo
escondido dentro del triángulo (problema de fin de semana del volumen
anterior sobre el baricentro). Los \omterm{def:g10:vectors:vector}{vectores} lo vuelven a demostrar en dos
líneas y regalan además la concurrencia. Para eso sirve esta nueva
álgebra: los teoremas se convierten en cálculos con flechas. Este problema
entrena el cálculo (Chasles ante todo), entrega la demostración de dos
líneas, extiende Varignon (\cref{exo:g10:vectors:10}) y termina donde
nacieron los \omterm{def:g10:vectors:vector}{vectores}: las fuerzas y las velocidades.

\medskip
\noindent\textbf{Parte I --- Gimnasia con Chasles.}
\begin{enumerate}
  \item Simplifica usando la relación de Chasles
        (\cref{thm:g10:vectors:chasles}):
        $\vect{AB} + \vect{BC} + \vect{CD}$; \quad
        $\vect{MA} - \vect{MB}$; \quad
        $\vect{AB} + \vect{CD} + \vect{BC} + \vect{DA}$.
  \item El truco del origen: demuestra que para \emph{cualquier} punto
        $O$ se cumple $\vect{AB} = \vect{OB} - \vect{OA}$.
  \item Demuestra las dos caracterizaciones del \omterm{prop:g10:coordgeom:midpoint}{punto medio} $I$ de
        $[AB]$:
        \[
        \vect{IA} + \vect{IB} = \vec 0
        \qquad\text{y}\qquad
        \vect{OI} = \tfrac12\left(\vect{OA} + \vect{OB}\right)
        \ \text{para cualquier } O .
        \]
  \item Usando $\vect{AB} = \vect{DC}$, vuelve a hallar el cuarto vértice
        $D$ del paralelogramo $ABCD$ con $A(-1, 2)$, $B(3, 4)$, $C(6, 0)$,
        y compara con la respuesta obtenida mediante el \omterm{prop:g10:coordgeom:midpoint}{punto medio} en el
        \cref{pb:g10:coordgeom:1}.
  \item Sumar un \omterm{def:g10:vectors:vector}{vector} fijo $\vec u$ a todos los puntos del plano
        efectúa, ¿qué transformación? La que se descubrió con dos espejos
        en el problema de fin de semana del volumen anterior sobre los dos
        espejos y con dos medios giros en el problema de fin de semana del
        volumen anterior sobre los medios giros. Calcula la \omterm{def:g10:functions:function}{imagen} de
        $(1, 2)$ con $\vec u = (3, -1)$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El baricentro, tercera demostración.}
Definimos el punto $G$ de un triángulo $ABC$ por la elegante condición
\[
\vect{GA} + \vect{GB} + \vect{GC} = \vec 0 .
\]
\begin{enumerate}[resume]
  \item Con el truco del origen, demuestra que esta condición determina
        $G$ de forma única:
        $\vect{OG} = \frac13\left(\vect{OA} + \vect{OB} +
        \vect{OC}\right)$ para cualquier $O$; así que las \omterm{def:g10:coordgeom:system}{coordenadas} de
        $G$ son las \emph{medias} de las de los vértices (el problema de
        fin de semana del volumen anterior sobre el baricentro ya lo vio
        numéricamente).
  \item Demuestra que $\vect{AG} = \frac13\left(\vect{AB} +
        \vect{AC}\right)$ y que, llamando $K$ al \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[BC]$,
        $\vect{AK} = \frac12\left(\vect{AB} + \vect{AC}\right)$. Concluye:
        $\vect{AG} = \frac23\,\vect{AK}$; el teorema de los dos tercios,
        demostrado de nuevo.
  \item ¿Por qué las medianas desde $B$ y desde $C$ pasan por el
        \emph{mismo} $G$ sin ningún cálculo adicional? Compara en una
        frase cada una de las tres demostraciones disponibles de este
        teorema: el paralelogramo escondido, las \omterm{def:g10:coordgeom:system}{coordenadas} y los
        \omterm{def:g10:vectors:vector}{vectores}.
  \item Comprobación numérica: $A(1, 1)$, $B(5, 2)$, $C(3, 6)$. Calcula
        $G$, después $K$, y verifica $\vect{AG} = \frac23 \vect{AK}$ con
        el criterio de colinealidad (\cref{thm:g10:vectors:det}).
  \item La física lee $G$ como el punto de equilibrio de tres masas
        iguales situadas en los vértices. Duplica la masa en $A$ (masas
        $2, 1, 1$): el punto de equilibrio pasa a ser
        $\vect{OG'} = \frac{2\vect{OA} + \vect{OB} + \vect{OC}}{4}$.
        Calcula $G'$ para el triángulo de la pregunta 9 y compara su
        posición con la de $G$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La colinealidad en acción.}
\begin{enumerate}[resume]
  \item ¿Son \omterm{def:g10:vectors:collinear}{colineales} $\vec u(3, -2)$ y $\vec v(-6, 4)$? ¿Y
        $\vec w(2, 5)$ y $\vec z(4, 9)$? Decídelo con el criterio del
        determinante.
  \item ¿Están alineados los puntos $A(1, 2)$, $B(3, 5)$, $C(7, 11)$?
  \item Más allá de Varignon (\cref{exo:g10:vectors:10}): en cualquier
        cuadrilátero $ABCD$, las \emph{bimedianas} unen los \omterm{prop:g10:coordgeom:midpoint}{puntos medios}
        de lados opuestos ($[IK]$ y $[JL]$). Usando \omterm{def:g10:vectors:vector}{vectores} de posición
        ($\vect{OI} = \frac12(\vect{OA} + \vect{OB})$, etc.), demuestra
        que las dos bimedianas tienen el mismo \omterm{prop:g10:coordgeom:midpoint}{punto medio}, de \omterm{def:g10:vectors:vector}{vector} de
        posición $\frac14\left(\vect{OA} + \vect{OB} + \vect{OC} +
        \vect{OD}\right)$: el baricentro del cuadrilátero.
  \item Tales en una línea: $M$ y $N$ son los puntos definidos por
        $\vect{AM} = \frac13\vect{AB}$ y $\vect{AN} = \frac13\vect{AC}$.
        Calcula $\vect{MN}$ en \omterm{def:g10:functions:function}{función} de $\vect{BC}$ y deduce el
        paralelismo y la razón.
  \item Generaliza la pregunta 14 a una razón cualquiera $k$ y explica en
        una frase cómo el cálculo vectorial absorbe de golpe el teorema de
        la base media y el teorema de Tales.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Fuerzas y velocidades.}
\begin{enumerate}[resume]
  \item Una barca es empujada por una corriente de $3$ km/h hacia el este
        mientras rema a $4$ km/h hacia el norte. Da el \omterm{def:g10:vectors:vector}{vector} velocidad
        resultante, su módulo y describe el rumbo real. (Aparece un trío
        de números muy antiguo.)
  \item Sobre una anilla actúan tres fuerzas: $\vec u(2, 1)$,
        $\vec v(-3, 2)$, $\vec w(1, -3)$. Calcula la resultante. ¿Qué hace
        la anilla? ¿Y por qué la \omterm{def:g10:vectors:sum}{suma} de las flechas a lo largo de
        cualquier polígono \emph{cerrado} es siempre $\vec 0$ (Chasles)?
  \item Sobre un gancho actúan dos fuerzas, $\vec u(4, -1)$ y
        $\vec v(-1, 3)$. ¿Qué tercera fuerza mantiene el gancho en
        equilibrio?
  \item Una nadadora cruza un río apuntando en perpendicular a la orilla
        a $2$ m/s; la corriente va a $1.5$ m/s. Da la velocidad resultante
        y su módulo. El río mide $50$ m de ancho: ¿cuánto dura la travesía
        y a qué distancia río abajo llega la nadadora?
  \item Final: un mismo objeto con tres disfraces: flecha de
        desplazamiento, pareja de \omterm{def:g10:coordgeom:system}{coordenadas} y fuerza o velocidad.
        Enumera los tres teoremas clásicos que este problema ha vuelto a
        demostrar con álgebra vectorial y nombra la única magnitud
        geométrica que los \omterm{def:g10:vectors:vector}{vectores} de este capítulo todavía no saben
        medir; la herramienta que lo arregla llega con el producto
        escalar, el año que viene.
\end{enumerate}
\end{problem}
