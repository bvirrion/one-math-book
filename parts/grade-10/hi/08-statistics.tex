\chapter{वर्णनात्मक सांख्यिकी}\label{ch:g10:stats}

सांख्यिकी संख्याओं के किसी बड़े संग्रह --- अंक, वेतन, तापमान --- को कुछ
सोच-समझकर चुने गए सूचकों में समेट देती है। यह अध्याय सूचकों के दो परिवार
प्रस्तुत करता है: वे जो आँकड़ों का \emph{केंद्र} बताते हैं (\omterm{def:g10:stats:mean}{माध्य}, \omterm{def:g10:stats:median}{माध्यिका})
और वे जो उसका \emph{प्रसार} नापते हैं (\omterm{def:g10:stats:quartiles}{परास}, \omterm{def:g10:stats:quartiles}{चतुर्थक})। मानक विचलन सहित इसका
गहरा उपचार \cref{ch:g11:stat} में आता है।

\section{आँकड़े और आपेक्षिक बारंबारताएँ}

\begin{definition}[सांख्यिकीय श्रेणी]\label{def:g10:stats:series}
\emph{सांख्यिकीय श्रेणी}\index{सांख्यिकीय श्रेणी} $N$ आकार की किसी समष्टि पर
प्रेक्षित मानों $x_1, x_2, \dots, x_N$ की सूची है। जब कोई मान $x$ $n$ बार आता है, तब $n$ उसकी
\emph{बारंबारता} है और भागफल
\[
f = \frac{n}{N}
\]
उसकी \emph{आपेक्षिक बारंबारता}\index{आपेक्षिक बारंबारता} है (जिसे प्रायः
प्रतिशत में लिखा जाता है)। सभी मानों की आपेक्षिक बारंबारताओं का योग $1$ होता
है।
\end{definition}

\begin{example}\label{ex:g10:stats:frequencies}
$5$ अंकों की एक लघु परीक्षा में $20$ विद्यार्थियों के अंक:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|cccccc}
अंक $x$ & $0$ & $1$ & $2$ & $3$ & $4$ & $5$ \\
\hline
बारंबारता $n$ & $1$ & $2$ & $4$ & $6$ & $5$ & $2$ \\
\omterm{def:g10:stats:series}{आपेक्षिक बारंबारता} $f$ & $0.05$ & $0.10$ & $0.20$ & $0.30$ & $0.25$ & $0.10$ \\
\end{tabular}
\end{center}
जाँच: $1 + 2 + 4 + 6 + 5 + 2 = 20$, और आपेक्षिक बारंबारताओं का योग $1$ है।
\end{example}

\section{केंद्र के माप}

\begin{definition}[माध्य]\label{def:g10:stats:mean}
श्रेणी का \emph{माध्य}\index{माध्य} है
\[
\bar x = \frac{x_1 + x_2 + \dots + x_N}{N}.
\]
यदि मान $v_1, \dots, v_k$ बारंबारताओं $n_1, \dots, n_k$ के साथ आते हों, तो वही संख्या \emph{भारित
माध्य} के रूप में इस तरह निकाली जाती है
\[
\bar x = \frac{n_1 v_1 + n_2 v_2 + \dots + n_k v_k}{n_1 + n_2 + \dots + n_k}.
\]
\end{definition}

\begin{example}\label{ex:g10:stats:mean}
\cref{ex:g10:stats:frequencies} की परीक्षा के लिए:
\[
\bar x = \frac{1 \times 0 + 2 \times 1 + 4 \times 2 + 6 \times 3
+ 5 \times 4 + 2 \times 5}{20}
= \frac{0 + 2 + 8 + 18 + 20 + 10}{20}
= \frac{58}{20} = 2.9 .
\]
\end{example}

\begin{definition}[माध्यिका]\label{def:g10:stats:median}
\emph{माध्यिका}\index{माध्यिका} वह मान है जो \emph{क्रमबद्ध} श्रेणी को दो
हिस्सों में बाँट देता है: कम से कम आधे मान माध्यिका $\leq$ हैं और कम से कम
आधे उससे $\geq$ हैं। व्यवहार में $N$ मानों को क्रम में लगाइए; तब
\begin{itemize}
  \item यदि $N$ विषम हो, तो माध्यिका बीच वाला मान है, जो स्थान $\frac{N+1}{2}$ पर है;
  \item यदि $N$ सम हो, तो बीच वाले दोनों मानों का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} लीजिए, जो
        स्थान $\frac N2$ और $\frac N2 + 1$ पर हैं।
\end{itemize}
\end{definition}

\begin{example}\label{ex:g10:stats:median}
$7$ घरों के दामों की क्रमबद्ध श्रेणी (हज़ारों में): $120$, $150$, $160$,
$180$, $210$, $240$, $900$। \omterm{def:g10:stats:median}{माध्यिका} $4$-वाँ मान $180$ है। \omterm{def:g10:stats:mean}{माध्य} $\frac{1960}{7} = 280$ है ---
जो $7$ दामों में से $6$ से भी बड़ा है! एक चरम मान ($900$) \omterm{def:g10:stats:mean}{माध्य} को
\omterm{def:g10:stats:median}{माध्यिका} की तुलना में कहीं अधिक खींच लेता है: \omterm{def:g10:stats:median}{माध्यिका} \emph{सुदृढ़} है,
\omterm{def:g10:stats:mean}{माध्य} नहीं।
\end{example}

\section{प्रसार के माप}

\begin{definition}[परास, चतुर्थक]\label{def:g10:stats:quartiles}
किसी क्रमबद्ध श्रेणी के लिए:
\begin{itemize}
  \item \emph{परास}\index{परास} सबसे बड़े और सबसे छोटे मान का अंतर है;
  \item \emph{प्रथम चतुर्थक}\index{चतुर्थक} $Q_1$ वह सबसे छोटा मान है जिसके
        लिए कम से कम एक चौथाई ($25\%$) मान $\leq Q_1$ हों; \emph{तृतीय चतुर्थक}
        $Q_3$ वह सबसे छोटा मान है जिसके लिए कम से कम तीन चौथाई ($75\%$) मान
        $\leq Q_3$ हों;
  \item \emph{अंतर-चतुर्थक परास}\index{अंतर-चतुर्थक परास} $Q_3 - Q_1$ आँकड़ों के
        मध्य आधे भाग का प्रसार नापता है।
\end{itemize}
\end{definition}

\begin{method}[चतुर्थक ज्ञात करना]\label{met:g10:stats:quartiles}
$N$ क्रमबद्ध मानों की श्रेणी के लिए:
\begin{enumerate}
  \item $\frac N4$ निकालिए; यदि वह \omterm{def:g10:numbers:sets}{पूर्णांक} न हो तो उसे अगले \omterm{def:g10:numbers:sets}{पूर्णांक} तक
        \emph{ऊपर} पूर्णांकित कीजिए; उस स्थान का मान $Q_1$ है;
  \item $\frac{3N}{4}$ निकालिए; आवश्यकता हो तो ऊपर पूर्णांकित कीजिए; उस स्थान का मान
        $Q_3$ है।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:stats:quartiles}
$11$ क्रमबद्ध मान लीजिए:
\[
2,\ 3,\ 5,\ 5,\ 6,\ 7,\ 8,\ 9,\ 9,\ 10,\ 12 .
\]
\omterm{def:g10:stats:median}{माध्यिका}: स्थान $\frac{11+1}{2} = 6$, इसलिए \omterm{def:g10:stats:median}{माध्यिका} $7$ है।
$Q_1$: $\frac{11}{4} = 2.75$, ऊपर पूर्णांकित करके $3$; $3$-वाँ मान $5$ है।
$Q_3$: $\frac{33}{4} = 8.25$, ऊपर पूर्णांकित करके $9$; $9$-वाँ मान $9$ है।
\omterm{def:g10:stats:quartiles}{परास}: $12 - 2 = 10$; अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास}: $9 - 5 = 4$।
\end{example}

\begin{example}[दो समूहों की तुलना]\label{ex:g10:stats:comparison}
दो कक्षाएँ एक ही परीक्षा देती हैं ($20$ में से अंक):
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccc}
 & \omterm{def:g10:stats:median}{माध्यिका} & $Q_1$ & $Q_3$ \\
\hline
कक्षा A & $12$ & $10$ & $14$ \\
कक्षा B & $12$ & $6$ & $17$ \\
\end{tabular}
\end{center}
\omterm{def:g10:stats:median}{माध्यिका} एक ही, पर प्रसार बहुत भिन्न: कक्षा A समरूप है (उसके आधे अंक $\intcc{10}{14}$
में), कक्षा B विषम (उसका मध्य आधा भाग $\intcc{6}{17}$ में फैला है)। अकेले केंद्र के
सूचक कभी पूरी कहानी नहीं कहते।
\end{example}

\begin{remark}
न \omterm{def:g10:stats:mean}{माध्य} ``बेहतर'' है, न \omterm{def:g10:stats:median}{माध्यिका}: \omterm{def:g10:stats:mean}{माध्य} हर मान का उपयोग करता है (और कुल
निकालने के लिए वही चाहिए), जबकि \omterm{def:g10:stats:median}{माध्यिका} चरम मानों के आगे टिकी रहती है। किसी
श्रेणी का गंभीर सारांश कम से कम एक केंद्र-सूचक \emph{और} एक प्रसार-सूचक
अवश्य देता है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g10:stats:1}
$15$ विद्यार्थियों ने एक वर्ष में जितनी पुस्तकें पढ़ीं उनकी संख्याएँ ये
हैं:
\[
0,\ 1,\ 1,\ 2,\ 2,\ 2,\ 3,\ 3,\ 4,\ 4,\ 5,\ 6,\ 7,\ 8,\ 12 .
\]
इस श्रेणी का \omterm{def:g10:stats:mean}{माध्य}, \omterm{def:g10:stats:median}{माध्यिका} और \omterm{def:g10:stats:quartiles}{परास} निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:2}
एक पासा $50$ बार फेंका गया:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|cccccc}
परिणाम & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ \\
\hline
बारंबारता & $7$ & $9$ & $8$ & $10$ & $6$ & $10$ \\
\end{tabular}
\end{center}
हर परिणाम की \omterm{def:g10:stats:series}{आपेक्षिक बारंबारता} और \omterm{def:g10:stats:mean}{माध्य} परिणाम निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:3}
क्रमबद्ध श्रेणी
\[
1,\ 2,\ 4,\ 4,\ 5,\ 6,\ 6,\ 7,\ 8,\ 9,\ 9,\ 11 \qquad (N = 12),
\]
की \omterm{def:g10:stats:median}{माध्यिका}, $Q_1$ और $Q_3$ ज्ञात कीजिए और उसका बक्सा-आरेख खींचिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:stats:4}
$24$ परीक्षा-अंकों का \omterm{def:g10:stats:mean}{माध्य} $11.5$ है। $25$-वाँ विद्यार्थी परीक्षा देता है
और $19$ अंक पाता है। कक्षा का नया \omterm{def:g10:stats:mean}{माध्य} क्या है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:5}
$5$ परीक्षाओं के बाद, जिनका भार समान है, किसी विद्यार्थी का \omterm{def:g10:stats:mean}{माध्य} $12$
है। $6$-वीं परीक्षा में कितने अंक \omterm{def:g10:stats:mean}{माध्य} को $13$ तक ले जाएँगे? क्या $14$
का \omterm{def:g10:stats:mean}{माध्य} संभव है (अंक $20$ में से हैं)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:6}
$7$ अऋणात्मक मानों की ऐसी श्रेणी बनाइए जिसकी \omterm{def:g10:stats:median}{माध्यिका} $10$ हो और \omterm{def:g10:stats:mean}{माध्य}
$20$ से बड़ा; फिर ऐसी जिसकी \omterm{def:g10:stats:median}{माध्यिका} $10$ हो और \omterm{def:g10:stats:mean}{माध्य} $6$ से छोटा। ऐसी
श्रेणी का \omterm{def:g10:stats:mean}{माध्य} $\frac{40}{7}$ से छोटा कभी क्यों नहीं हो सकता? ये उदाहरण क्या दिखाते
हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:7}
किसी कंपनी में $9$ कर्मचारी प्रति माह $1800$ कमाते हैं और निदेशक $12000$।
\begin{enumerate}
  \item \omterm{def:g10:stats:mean}{माध्य} और \omterm{def:g10:stats:median}{माध्यिका} वेतन निकालिए।
  \item यहाँ ``प्रारूपिक'' वेतन का सबसे अच्छा वर्णन कौन-सा सूचक करता है?
        क्यों?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:stats:8}
$12$ लड़कों की एक कक्षा की \omterm{def:g10:stats:mean}{माध्य} लंबाई $168$ cm है; उसी कक्षा की $18$
लड़कियों की \omterm{def:g10:stats:mean}{माध्य} लंबाई $163$ cm है। पूरी कक्षा की \omterm{def:g10:stats:mean}{माध्य} लंबाई निकालिए।
(सावधान: वह $168$ और $163$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} \emph{नहीं} है --- \omterm{def:g10:stats:mean}{माध्यों} को समूह
के आकार से भारित कीजिए।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:stats:9}
$N$ मानों $x_1, \dots, x_N$ की श्रेणी का \omterm{def:g10:stats:mean}{माध्य} $\bar x$ है। हर मान को नियत संख्याओं $a$
और $b$ के लिए $y_i = a x_i + b$ में बदल दिया जाता है।
\begin{enumerate}
  \item दिखाइए कि नई श्रेणी का \omterm{def:g10:stats:mean}{माध्य} $a \bar x + b$ है।
  \item किसी सप्ताह के तापमानों का \omterm{def:g10:stats:mean}{माध्य} सेल्सियस में $20$ है। फ़ारेनहाइट
        में उनका \omterm{def:g10:stats:mean}{माध्य} क्या है ($F = 1.8\,C + 32$)?
\end{enumerate}
\end{exercise}

\section{समस्या: औसत व्यक्ति होता ही नहीं}

\begin{problem}[{सप्ताहांत समस्या --- माध्य बनाम माध्यिका: बहिर्मान,
अंक-सुधार, सड़क किनारे का गोदाम, और वह कॉकपिट जो किसी के लिए नहीं बना}]
\label{pb:g10:stats:1}
1950 में एक वायुसेना ने $4\,000$ वैमानिकों के शरीर की दस माप-राशियाँ नापीं और
कॉकपिट \emph{औसत आदमी} के लिए बनाया। एक लेफ़्टिनेंट ने गिनकर पाया कि वह
असल में लगभग किसी पर भी ठीक नहीं बैठता था। सांख्यिकी भीड़ को अकेली-अकेली
संख्याओं में समेटती है, और हर सारांश किसी बात का सच बताता है तथा किसी और बात
पर झूठ बोलता है। यह समस्या \omterm{def:g10:stats:mean}{माध्य} और \omterm{def:g10:stats:median}{माध्यिका} (\cref{def:g10:stats:mean}, \cref{def:g10:stats:median}) को कठघरे में
खड़ा करती है, दोनों को अपनी-अपनी सर्वोत्तम बात दिखाने देती है, और कॉकपिट का
मुक़दमा बंद करती है।

\medskip
\noindent\textbf{भाग I --- दो सारांश कठघरे में।}
\begin{enumerate}
  \item एक गली में सात घर बिकते हैं, हज़ार यूरो में: $200$, $210$, $220$,
        $230$, $240$, $250$ --- और एक हवेली $1\,200$ की। \omterm{def:g10:stats:mean}{माध्य} और \omterm{def:g10:stats:median}{माध्यिका}
        दाम निकालिए। ``इस गली का एक घर'' कौन-सी संख्या बताती है?
  \item हवेली हटाकर दोनों फिर से निकालिए। हर सारांश कितना खिसका? बहिर्मानों
        के बारे में सीख बताइए।
  \item जनगणना की एक सारणी: $100$ परिवारों में से $20$ के कोई संतान नहीं,
        $25$ के एक, $30$ के दो, $15$ के तीन और $10$ के चार हैं।
        संतानों की \omterm{def:g10:stats:mean}{माध्य} संख्या निकालिए। किसी परिवार में ``$1.7$ संतानें''
        नहीं होतीं --- तो \omterm{def:g10:stats:mean}{माध्य} कौन-सी वास्तविक राशि समेटे है? (उसे $100$
        से गुणा कीजिए।)
  \item पंद्रह अंकों $4, 6, 7, 8, 9, 10, 10, 11, 12, 12, 13, 14, 15, 17, 19$ के लिए \omterm{def:g10:stats:median}{माध्यिका} और \omterm{def:g10:stats:quartiles}{चतुर्थक} $Q_1$, $Q_3$ ज्ञात
        कीजिए (\cref{met:g10:stats:quartiles})।
  \item उसी श्रेणी के लिए \omterm{def:g10:stats:quartiles}{परास} और अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} बताइए। हर एक क्या
        नापता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- अंक-सुधार और मिली हुई कक्षाएँ।}
\begin{enumerate}[resume]
  \item \cref{exo:g10:stats:9} से: हर मान में $+b$ जोड़ने पर \omterm{def:g10:stats:mean}{माध्य}, \omterm{def:g10:stats:median}{माध्यिका} और \omterm{def:g10:stats:quartiles}{चतुर्थक} भी
        $+b$ खिसक जाते हैं; $a$ से गुणा करने पर वे $a$ गुने हो जाते
        हैं। खिसकाने पर \omterm{def:g10:stats:quartiles}{परास} और अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} का क्या होता है? गुणा
        करने पर? पुष्ट कीजिए।
  \item किसी परीक्षा का \omterm{def:g10:stats:mean}{माध्य} $9$ है ($20$ में से) और अध्यापक $12$
        \omterm{def:g10:stats:mean}{माध्य} चाहते हैं। दो सुधार सुझाए जाते हैं: सबको $3$ अंक जोड़ देना,
        या हर अंक को $\frac43$ से गुणा कर देना। हर सुधार के अधीन $6$ वाले और
        $18$ वाले विद्यार्थी का भाग्य निकालिए। कौन-सा सुधार किसके पक्ष में
        जाता है --- और किसमें ऊपरी सीमा की समस्या है?
  \item कक्षा A ($20$ विद्यार्थी) का \omterm{def:g10:stats:mean}{माध्य} $12$ है; कक्षा B ($30$
        विद्यार्थी) का \omterm{def:g10:stats:mean}{माध्य} $10$। दोनों को मिलाकर $50$ विद्यार्थियों का
        \omterm{def:g10:stats:mean}{माध्य} निकालिए, और समझाइए कि वह $11$ क्यों नहीं है (माध्यमिक
        विद्यालय खंड की हरात्मक-माध्य वाली सप्ताहांत समस्या का भारित-औसत
        वाला पाठ, अब कक्षा में)।
  \item दिखाइए कि माध्यिकाएँ यह खेल खेलने से मना कर देती हैं: $A = \{1, 2, 3\}$ और $B = \{0, 10\}$
        की माध्यिकाएँ निकालिए, फिर दोनों को मिलाकर बनी सूची की \omterm{def:g10:stats:median}{माध्यिका}, और
        उसकी तुलना दोनों माध्यिकाओं के आकार-भारित औसत से कीजिए।
  \item स्केटिंग: पाँच निर्णायक $5.2$, $5.3$, $5.4$, $5.5$, $9.9$ अंक देते
        हैं। \omterm{def:g10:stats:mean}{माध्य}, \omterm{def:g10:stats:median}{माध्यिका} और \emph{कटा-छँटा \omterm{def:g10:stats:mean}{माध्य}} (सबसे नीचे और सबसे
        ऊपर वाला हटाकर शेष का औसत) निकालिए। अधिकांश निर्णायक-आधारित खेल
        काट-छाँट क्यों करते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- बंटनों को पढ़ना।}
\begin{enumerate}[resume]
  \item दो कारख़ाने ऐसे बोल्ट बनाते हैं जिनकी \omterm{def:g10:stats:mean}{माध्य} लंबाई $50$ mm है।
        कारख़ाना A: $Q_1 = 49.9$, $Q_3 = 50.1$; कारख़ाना B: $Q_1 = 48$, $Q_3 = 52$। आपको आपस में
        बदले जा सकने वाले बोल्ट चाहिए: कौन-सा कारख़ाना, और किस सांख्यिकी ने
        निर्णय किया?
  \item किसी चिकित्सालय में प्रतीक्षा की \omterm{def:g10:stats:median}{माध्यिका} $10$ मिनट है पर \omterm{def:g10:stats:mean}{माध्य}
        प्रतीक्षा $25$ है। यह युग्म आँकड़ों की किस आकृति की चुगली करता है?
        ठीक इसी \omterm{def:g10:stats:median}{माध्यिका} और \omterm{def:g10:stats:mean}{माध्य} वाले पाँच मानों का एक आँकड़ा-समुच्चय
        बनाइए।
  \item ``आपका शिशु लंबाई में 75-वें शतमक पर है'' --- इस वाक्य का अर्थ
        क्या है? और क्या किसी नगर का हर बच्चा ``औसत से ऊपर'' हो सकता है?
        \omterm{def:g10:stats:median}{माध्यिका} से ऊपर? अंतर समझाइए।
  \item कक्षा A: \omterm{def:g10:functions:extrema}{न्यूनतम} $4$, $Q_1\,8$, \omterm{def:g10:stats:median}{माध्यिका} $11$, $Q_3\,13$, \omterm{def:g10:functions:extrema}{अधिकतम} $17$।
        कक्षा B: \omterm{def:g10:functions:extrema}{न्यूनतम} $2$, $Q_1\,9$, \omterm{def:g10:stats:median}{माध्यिका} $11$, $Q_3\,12$, \omterm{def:g10:functions:extrema}{अधिकतम} $19$।
        तुलना कीजिए: प्रारूपिक अंक, मध्य आधे भाग की समरूपता, और चरम मानों
        का उत्पात।
  \item पत्रकार की जाँच-सूची: एक सुर्ख़ी कहती है ``टेककॉर्प में औसत वेतन:
        $5\,000$ यूरो''। इस समस्या ने आपको जो तीन प्रश्न पूछना सिखाया है उन्हें
        गिनाइए, जिन्हें पूछे बिना यह मान लेना कि सुर्ख़ी किसी प्रारूपिक
        कर्मचारी के बारे में कुछ कह रही है, ठीक नहीं।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- कॉकपिट का मुक़दमा।}
\begin{enumerate}[resume]
  \item अपने प्रमाण स्वयं बनाइए: ऐसे पाँच घरों के दाम बनाइए जिनका \omterm{def:g10:stats:mean}{माध्य}
        $300$ और \omterm{def:g10:stats:median}{माध्यिका} $200$ हो (हज़ार यूरो में)।
  \item \omterm{def:g10:stats:mean}{माध्य} संतुलन-बिंदु है: परिभाषा से सिद्ध कीजिए कि \omterm{def:g10:stats:mean}{माध्य} से विचलन सदा
        कट जाते हैं, $\sum (x_i - \bar x) = 0$, और इसे प्रश्न 16 के अपने आँकड़ा-समुच्चय पर
        जाँचिए।
  \item \omterm{def:g10:stats:median}{माध्यिका} फेरे-न्यूनतमकारी है: किसी सड़क के $1$, $3$, $4$,
        $9$, $13$ किलोमीटर पर दुकानें हैं, और एक गोदाम को पाँचों दुकानों
        तक की \emph{कुल दूरी} \omterm{def:g10:functions:extrema}{न्यूनतम} करनी है। \omterm{def:g10:stats:median}{माध्यिका} ($4$) पर, \omterm{def:g10:stats:mean}{माध्य}
        ($6$) पर और $5$ पर बने गोदाम के लिए वह कुल दूरी निकालिए। कौन
        जीता? (यह सामान्य बात है: \omterm{def:g10:stats:median}{माध्यिका} कुल निरपेक्ष दूरी को \omterm{def:g10:functions:extrema}{न्यूनतम}
        करती है --- कोई भी दूसरा स्थान जाँच लीजिए।)
  \item कॉकपिट का निपटारा: मान लीजिए किसी वैमानिक के शरीर की एक माप-राशि
        पर ``औसत के पास'' होने की संभावना $30\,\%$ है, और दसों माप-राशियों पर
        यह स्वतंत्र रूप से लागू होती है ($10$ राशियाँ)। \emph{दसों} पर
        औसत के पास होने की संभावना का आकलन कीजिए --- और इस परिणाम को उससे
        मिलाइए जो लेफ़्टिनेंट ने $4\,000$ वैमानिकों में पाया था। वायुसेना ने
        औसत कॉकपिट के बदले क्या बनाया?
  \item समापन --- उपयोग-निर्देश, एक-एक पंक्ति में: जब कुल ही महत्त्वपूर्ण
        हो (बजट, फ़सल), तब \dots\ का उपयोग कीजिए; जब प्रारूपिक व्यक्ति
        महत्त्वपूर्ण हो (वेतन, प्रतीक्षा-काल), तब \dots\ का; जब प्रसार की
        समता महत्त्वपूर्ण हो, तब \dots\ बताइए; और जब दूषण का ख़तरा हो, तब
        \dots\ का उपयोग कीजिए। मुक़दमा समस्या के शीर्षक से बंद कीजिए।
\end{enumerate}
\end{problem}
