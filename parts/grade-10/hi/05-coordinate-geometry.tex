\chapter{निर्देशांक ज्यामिति}\label{ch:g10:coordgeom}

देकार्त का महान विचार यह था कि बिंदुओं को संख्या-युग्मों से बताया जाए, जिससे
ज्यामिति की समस्याएँ गणनाएँ बन जाएँ। केवल दो सूत्रों --- \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} और दूरी ---
से यह सिद्ध किया जा सकता है कि कोई त्रिभुज समद्विबाहु है, कोई चतुर्भुज समांतर
चतुर्भुज है, या तीन बिंदु संरेख हैं, और इसके लिए एक भी सहायक रेखा खींचने की
आवश्यकता नहीं पड़ती।

\section{समतल में निर्देशांक}

\begin{definition}[निर्देशांक-पद्धति]\label{def:g10:coordgeom:system}
समतल की \emph{निर्देशांक-पद्धति}\index{निर्देशांक-पद्धति} में एक मूल बिंदु
$O$ और $O$ से होकर जाने वाले दो अंकित अक्ष होते हैं: क्षैतिज $x$-अक्ष और
ऊर्ध्वाधर $y$-अक्ष। तब हर बिंदु $M$ के अपने अद्वितीय
\emph{निर्देशांक}\index{निर्देशांक} $(x, y)$ होते हैं: उसका
\emph{भुज}\index{भुज} $x$ और उसकी \emph{कोटि}\index{कोटि} $y$। पद्धति
\emph{लंबकोणिक}\index{लंबकोणिक पद्धति} तब कहलाती है जब अक्ष परस्पर लंब हों
और दोनों पर लंबाई का एक ही मात्रक हो। इस अध्याय की सभी पद्धतियाँ लंबकोणिक
हैं।
\end{definition}

\section{रेखाखंड का मध्यबिंदु}

\begin{proposition}[मध्यबिंदु सूत्र]\label{prop:g10:coordgeom:midpoint}
मान लीजिए $A(x_A, y_A)$ और $B(x_B, y_B)$ हैं। रेखाखंड $[AB]$ के
\emph{मध्यबिंदु}\index{मध्यबिंदु} $I$ के \omterm{def:g10:coordgeom:system}{निर्देशांक} हैं
\[
I\left(\frac{x_A + x_B}{2},\ \frac{y_A + y_B}{2}\right).
\]
\end{proposition}

\begin{proof}
क्षैतिज \omterm{def:g10:coordgeom:system}{निर्देशांकों} पर विचार कीजिए। $A$ से $B$ तक जाने में \omterm{def:g10:coordgeom:system}{भुज} $x_B - x_A$ बदलता
है; ठीक आधे रास्ते पड़ने वाले बिंदु का \omterm{def:g10:coordgeom:system}{भुज}
\[
x_A + \frac{x_B - x_A}{2} = \frac{2x_A + x_B - x_A}{2}
= \frac{x_A + x_B}{2},
\]
है, अर्थात् दोनों भुजों का माध्य। यही गणना कोटियों पर भी लागू होती है।
\end{proof}

\begin{example}\label{ex:g10:coordgeom:midpoint}
$A(-1, 4)$ और $B(5, -2)$ के साथ $[AB]$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} है
\[
I\left(\frac{-1 + 5}{2},\ \frac{4 + (-2)}{2}\right) = I(2, 1).
\]
यह सूत्र उल्टी दिशा में भी चलता है: यदि $A(-1, 4)$ हो और \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} $I(2, 1)$ हो, तो
$B$ $\frac{-1 + x_B}{2} = 2$ और $\frac{4 + y_B}{2} = 1$ को संतुष्ट करता है, इसलिए $x_B = 5$ और $y_B = -2$: बिंदु $B$ $I$
के परितः $A$ का \emph{सममित} बिंदु है।
\end{example}

\section{दो बिंदुओं के बीच की दूरी}

\begin{theorem}[दूरी सूत्र]\label{thm:g10:coordgeom:distance}
\omterm{def:g10:coordgeom:system}{लंबकोणिक पद्धति} में $A(x_A, y_A)$ और $B(x_B, y_B)$ के बीच की दूरी है
\[
AB = \sqrt{(x_B - x_A)^2 + (y_B - y_A)^2}.
\]
\end{theorem}

\begin{proof}
मान लीजिए $C$ बिंदु $(x_B, y_A)$ है: उसकी ऊँचाई $A$ जैसी है और \omterm{def:g10:coordgeom:system}{भुज} $B$ जैसा,
इसलिए त्रिभुज $ACB$ में $C$ पर समकोण है। उसकी भुजाएँ क्षैतिज और ऊर्ध्वाधर
रेखाखंड हैं, जिनकी लंबाइयाँ $AC = \abs{x_B - x_A}$ और $CB = \abs{y_B - y_A}$ हैं। पाइथागोरस प्रमेय से
\[
AB^2 = AC^2 + CB^2 = (x_B - x_A)^2 + (y_B - y_A)^2,
\]
और वर्गमूल लेने पर (दोनों पक्ष अऋणात्मक हैं) सूत्र मिल जाता है।
\end{proof}

त्रिभुज $ACB$ पर लगाई गई पाइथागोरस प्रमेय है।}

\begin{example}\label{ex:g10:coordgeom:distance}
$A(1, 1)$ और $B(5, 3)$ के साथ:
\[
AB = \sqrt{(5-1)^2 + (3-1)^2} = \sqrt{16 + 4} = \sqrt{20} = 2\sqrt 5 .
\]
दूरियाँ प्रायः यथार्थ (वर्गमूल वाले) रूप में ही रखी जाती हैं; यदि दशमलव
उत्तर चाहिए तो केवल अंत में पूर्णांकित कीजिए।
\end{example}

\begin{remark}
बिंदुओं के क्रम से कोई अंतर नहीं पड़ता: $(x_A - x_B)^2 =
(x_B - x_A)^2$। पर वर्ग लगाना मत भूलिए! दूरी
$\abs{x_B - x_A} + \abs{y_B - y_A}$ \emph{नहीं} है।
\end{remark}

\section{ज्यामिति में सूत्रों का उपयोग}

\begin{method}[त्रिभुज की प्रकृति]\label{met:g10:coordgeom:triangle}
तीन बिंदु $A$, $B$, $C$ उनके \omterm{def:g10:coordgeom:system}{निर्देशांकों} से दिए हों तो:
\begin{enumerate}
  \item दूरी सूत्र से तीनों वर्ग-लंबाइयाँ $AB^2$, $AC^2$, $BC^2$ निकालिए (वर्ग
        बनाए रखने से वर्गमूलों से बचाव होता है);
  \item दो वर्ग-लंबाइयाँ बराबर हों $\Rightarrow$ त्रिभुज \emph{समद्विबाहु} है;
  \item यदि सबसे बड़ी वर्ग-लंबाई शेष दोनों के योग के बराबर हो, तो पाइथागोरस
        प्रमेय के विलोम से त्रिभुज सबसे लंबी भुजा के सामने वाले शीर्ष पर
        \emph{समकोण} है।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:coordgeom:triangle}
मान लीजिए $A(0, 1)$, $B(4, 3)$ और $C(2, -3)$ हैं। तब
\begin{align*}
AB^2 &= (4-0)^2 + (3-1)^2 = 16 + 4 = 20, \\
AC^2 &= (2-0)^2 + (-3-1)^2 = 4 + 16 = 20, \\
BC^2 &= (2-4)^2 + (-3-3)^2 = 4 + 36 = 40 .
\end{align*}
पहले, $AB^2 = AC^2$, इसलिए $AB = AC$: त्रिभुज $A$ पर समद्विबाहु है। इसके अतिरिक्त $AB^2 + AC^2 = 40 = BC^2$,
इसलिए पाइथागोरस प्रमेय के विलोम से त्रिभुज $A$ पर समकोण भी है।
\end{example}

\begin{method}[समांतर चतुर्भुज पहचानना]\label{met:g10:coordgeom:parallelogram}
चतुर्भुज $ABCD$ समांतर चतुर्भुज ठीक तब है जब उसके विकर्णों $[AC]$ और $[BD]$ का
\omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} एक ही हो। अतः: \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} सूत्र से दोनों \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} निकालिए और
उनकी तुलना कीजिए।
\end{method}

\begin{example}\label{ex:g10:coordgeom:parallelogram}
मान लीजिए $A(-1, 0)$, $B(2, 2)$, $C(5, 1)$ और $D(2, -1)$ हैं। $[AC]$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} $\left(\frac{-1+5}{2}, \frac{0+1}{2}\right) = (2, \frac12)$ है; $[BD]$
का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} $\left(\frac{2+2}{2}, \frac{2-1}{2}\right) =
(2, \frac12)$ है। दोनों एक ही हैं, इसलिए $ABCD$ समांतर चतुर्भुज है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:1}
मान लीजिए $A(2, 5)$ और $B(-4, 1)$ हैं। $[AB]$ के \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} के \omterm{def:g10:coordgeom:system}{निर्देशांक} और दूरी $AB$
निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:2}
मान लीजिए $A(3, -2)$ और $I(1, 2)$ हैं। वह बिंदु $B$ ज्ञात कीजिए जिसके लिए $I$
$[AB]$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} हो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:3}
बिंदुओं $P(4, 1)$, $Q(-3, 2)$ और $R(0, -5)$ में से कौन-सा मूल बिंदु $O(0,0)$ के सबसे निकट
है? उत्तर यथार्थ दूरियों के साथ दीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:4}
आलेख-कागज़ पर बिंदु $A(1, 2)$, $B(5, 4)$, $C(7, 0)$ अंकित कीजिए, फिर गणना करके दिखाइए
कि त्रिभुज $ABC$ समद्विबाहु है। क्या वह समकोण भी है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:5}
मान लीजिए $A(-2, 1)$, $B(1, 3)$, $C(4, 1)$ और $D(1, -1)$ हैं।
\begin{enumerate}
  \item दिखाइए कि $ABCD$ समांतर चतुर्भुज है।
  \item $AB$ और $BC$ निकालिए। क्या $ABCD$ समचतुर्भुज है (सभी भुजाएँ
        बराबर)?
  \item $AC$ और $BD$ निकालिए। क्या $ABCD$ आयत है (विकर्ण बराबर)?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:6}
केंद्र $\Omega(2, 1)$ और त्रिज्या $5$ वाले वृत्त में वे सभी बिंदु हैं जो $\Omega$ से
$5$ दूरी पर हैं। बिंदुओं $A(5, 5)$, $B(-1, -3)$, $C(6, 2)$ में से कौन इस वृत्त पर हैं?
कौन उसके भीतर? कौन बाहर?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:7}
मान लीजिए $A(1, 1)$ और $B(7, 5)$ हैं। $x$-अक्ष के वे सभी बिंदु $M(x, 0)$ ज्ञात कीजिए
जो $A$ और $B$ से समान दूरी पर हैं। ($MA^2 = MB^2$ लिखिए और $x$ के लिए हल
कीजिए।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:8}
मान लीजिए $A(0, 3)$, $B(4, 1)$ हैं।
\begin{enumerate}
  \item $[AB]$ के \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} $I$ के \omterm{def:g10:coordgeom:system}{निर्देशांक} निकालिए।
  \item दिखाइए कि $M(2, 2) = I$, फिर $MA$, $MB$ निकालकर सत्यापित कीजिए कि $M$
        $A$ और $B$ से समान दूरी पर है।
  \item $y$-अक्ष पर एक दूसरा बिंदु ज्ञात कीजिए जो $A$ और $B$ से समान
        दूरी पर हो।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:9}
बिंदु $A(-1, -1)$, $B(3, 1)$ और $C(11, 5)$ दिए हैं। $AB$, $BC$ और $AC$ निकालिए और
निष्कर्ष निकालिए कि $A$, $B$, $C$ संरेख हैं। (तीन बिंदु संरेख ठीक तब
होते हैं जब तीनों दूरियों में सबसे बड़ी शेष दोनों के योग के बराबर हो।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:coordgeom:10}
मान लीजिए $A(a, 0)$ और $B(0, b)$ हैं, जहाँ $a, b > 0$ है, और मान लीजिए $I$ $[AB]$ का
\omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} है। गणना करके दिखाइए कि $OI = IA = IB$ है, जहाँ $O$ मूल बिंदु है। (यह
एक चिरपरिचित प्रमेय सिद्ध करता है: समकोण त्रिभुज में कर्ण का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु}
तीनों शीर्षों से समान दूरी पर होता है।)
\end{exercise}

\section{समस्या: देकार्त का पुल --- वृत्त के समीकरण और अपनी जगह पहचानना}

\begin{problem}[{सप्ताहांत समस्या --- वृत्त समीकरण बन जाता है, पुरानी
प्रमेय गणनाएँ बन जाती हैं, और तीन संकेत-स्तंभ किसी बिंदु का पता लगा लेते
हैं: बीजगणित से त्रिदूरी-निर्धारण}]
\label{pb:g10:coordgeom:1}
1637 में रने देकार्त ने दो महाद्वीपों के बीच एक पुल बनाया: ज्यामिति का हर
वक्र \emph{\omterm{def:g10:algebra:equation}{समीकरण}} बन गया, और हर ज्यामितीय प्रश्न एक गणना। यह समस्या उस पुल
पर दोनों दिशाओं में चलती है --- दूरी सूत्र (\cref{thm:g10:coordgeom:distance}) से वृत्त का \omterm{def:g10:algebra:equation}{समीकरण}
निकालती है, चिरपरिचित प्रमेयों को बीजगणित की तीन पंक्तियों में फिर सिद्ध
करती है, और वहाँ समाप्त होती है जहाँ आज इस पुल पर सबसे अधिक आवागमन है: दूरी
के संकेतों से स्थिति निकालना, जो GPS का समतल-पृथ्वी वाला हृदय है।

\medskip
\noindent\textbf{भाग I --- वृत्त को \omterm{def:g10:algebra:equation}{समीकरण} मिलता है।}
\begin{enumerate}
  \item बिंदु $M(x, y)$ केंद्र $\Omega(2, 1)$ और त्रिज्या $5$ वाले वृत्त पर ठीक तब
        होता है जब $\Omega M = 5$ हो। इस शर्त का वर्ग करके वृत्त का \omterm{def:g10:algebra:equation}{समीकरण}
        प्राप्त कीजिए:
        \[
        (x - 2)^2 + (y - 1)^2 = 25 .
        \]
  \item सदस्यता जाँचिए: $A(5, 5)$, $B(6, 4)$, $D(4, 4)$ में से कौन वृत्त पर हैं,
        कौन भीतर, कौन बाहर?
  \item भेस बदले हुए वृत्त: \[
        x^2 + y^2 - 6x + 4y - 12 = 0
        \] में वर्ग पूरे कीजिए (\cref{pb:g10:algebra:1}) और उसका
        केंद्र तथा त्रिज्या बताइए।
  \item यही $x^2 + y^2 - 6x + 4y + 14 = 0$ के साथ कीजिए। पूरा किया हुआ रूप क्या प्रकट करता है?
        कसौटी बताइए: $(x - h)^2 + (y - k)^2 = c$ कब एक वृत्त बताता है, कब एक अकेला बिंदु, और
        कब कुछ भी नहीं?
  \item प्रश्न 3 के वृत्त को क्षैतिज रेखा $y = 3$ से काटिए: प्रतिस्थापन
        करके हल कीजिए। कितने बिंदु मिले, और ऐसी रेखा का ज्यामितीय नाम
        क्या है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- पुरानी प्रमेय तीन पंक्तियों में।}
\begin{enumerate}[resume]
  \item माध्यमिक विद्यालय खंड की वृत्त-प्रमेय (हर अर्धवृत्त में एक समकोण)
        को बीजगणित से फिर सिद्ध कीजिए: मान लीजिए $A(-r, 0)$ और $B(r, 0)$ किसी व्यास
        के सिरे हैं और $M(x, y)$ वृत्त $x^2 + y^2 = r^2$ का कोई बिंदु। $MA^2 + MB^2$ निकालिए और
        पाइथागोरस के विलोम से निष्कर्ष निकालिए कि $\widehat{AMB}$ समकोण है।
  \item माध्यिका प्रमेय: $A(-a, 0)$, $B(a, 0)$ के साथ (जिससे $[AB]$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु}
        मूल बिंदु $I$ हो जाता है) दिखाइए कि हर बिंदु $M(x, y)$ के लिए
        \[
        MA^2 + MB^2 = 2\,MI^2 + \frac{AB^2}{2} .
        \]
  \item जब $AB = 6$ हो तब $MA^2 + MB^2 = 26$ वाले बिंदुओं $M$ का बिंदुपथ निकालिए:
        कौन-सा वक्र, कौन-सा केंद्र, कौन-सी त्रिज्या?
  \item एक अलग बिंदुपथ: $A(0,0)$, $B(3,0)$; $MA = 2\,MB$ वाले सभी बिंदु ज्ञात कीजिए।
        (वर्ग कीजिए, प्रसार कीजिए, वर्ग पूरे कीजिए: एक प्रसिद्ध वृत्त
        प्रकट होता है --- अपोलोनियस उसे \omterm{def:g10:coordgeom:system}{निर्देशांकों} के बिना ही जानता
        था।)
  \item एक-दो वाक्यों में: देकार्त का पुल इस बात में क्या बदल देता है कि
        प्रमेय \emph{कैसे} सिद्ध होती हैं? प्रश्न 6 की तुलना माध्यमिक
        विद्यालय खंड की वृत्त-प्रमेय (हर अर्धवृत्त में एक समकोण) की आयत
        वाली उपपत्ति से कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- तीन संकेत-स्तंभ आपको ढूँढ़ लेते हैं।}
आपकी स्थिति $M(x, y)$ अज्ञात है। संकेत-स्तंभ $P(0, 0)$ आपकी दूरी $5$ नापता है;
संकेत-स्तंभ $Q(6, 0)$ भी $5$ नापता है।
\begin{enumerate}[resume]
  \item दोनों वृत्त-समीकरण लिखिए, उन्हें घटाइए, और देखिए कि वर्ग कट जाते
        हैं: कौन-सा सरल \omterm{def:g10:algebra:equation}{समीकरण} बचता है? उसे किसी एक वृत्त के साथ मिलाकर
        \emph{दो} संभावित स्थितियाँ ज्ञात कीजिए।
  \item तीसरा संकेत-स्तंभ $R(0, 8)$ आपकी दूरी $5$ नापता है। हर संभावित
        स्थिति से उसकी दूरी निकालिए और तय कीजिए कि आप कहाँ हैं।
  \item समझाइए कि दो वृत्त-समीकरणों को घटाने पर \emph{हमेशा} एक रेखा का
        \omterm{def:g10:algebra:equation}{समीकरण} क्यों मिलता है (कौन-से पद कटते हैं?), और जब दोनों वृत्त
        दो बिंदुओं पर मिलते हों तब वह रेखा ज्यामितीय रूप से क्या है।
  \item वास्तविक उपग्रह-स्थिति-निर्धारण अंतरिक्ष में और अपूर्ण घड़ियों के
        साथ काम करता है: हर उपग्रह (संकेत के यात्रा-काल से) एक दूरी देता
        है, यानी एक गोला। किसी अभिग्राही के पास कितने अज्ञात होते हैं
        (स्थिति \emph{और} उसकी अपनी घड़ी की त्रुटि), और इसीलिए GPS कम से
        कम \emph{चार} उपग्रह क्यों सुनता है?
  \item यथार्थता: मान लीजिए संकेत-स्तंभ $P$ की दूरी वस्तुतः $5$ के
        बजाय $5.1$ है (शेष सब वैसा ही)। प्रश्न 11 का घटाव फिर से करके नया
        $x$ ज्ञात कीजिए, और बताइए कि आकलन कितना खिसका। (\cref{pb:g10:numbers:1} के
        त्रुटि-आकलन वाले विचारों से तुलना कीजिए।)
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- भूमापक का औज़ार-बक्सा।}
\begin{enumerate}[resume]
  \item पानी भरने का सबसे छोटा रास्ता: $A(1, 5)$ पर एक शिविर, $B(7, 3)$ पर एक
        गोशाला, और अक्ष $y = 0$ के साथ-साथ बहती एक सीधी नदी। $A$ से नदी
        तक और फिर $B$ तक कम से कम चलकर जाने के लिए: $B$ का नदी के
        आरपार परावर्तन $B'$ लीजिए और देखिए कि रेखाखंड $[AB']$ नदी को कहाँ
        काटता है। वह प्रतिच्छेद बिंदु और \omterm{def:g10:functions:extrema}{न्यूनतम} कुल लंबाई बताइए।
        (विचार वही है जो माध्यमिक विद्यालय खंड के बिलियर्ड-टप्पे वाले
        अभ्यास का था, पर अब पूरी तरह गणनीय।)
  \item एक प्रतिद्वंद्वी रास्ते पर न्यूनतमता की पुष्टि कीजिए: $Q(3, 0)$ होकर
        कुल रास्ता निकालिए और जाँचिए कि वह आपके उत्तर से हार जाता है।
  \item चतुर्भुज $A(1,1)$, $B(4,2)$, $C(5,5)$, $D(2,4)$ का पूरा वर्गीकरण कीजिए:
        समांतर चतुर्भुज? समचतुर्भुज? आयत? वर्ग? (\cref{met:g10:coordgeom:parallelogram}, \cref{met:g10:coordgeom:triangle} --- विकर्णों
        की भी तुलना कीजिए।)
  \item $A(-1, 2)$, $B(3, 4)$, $C(6, 0)$ दिए होने पर ऐसा $D$ ज्ञात कीजिए कि $ABCD$
        समांतर चतुर्भुज हो (माध्यमिक विद्यालय खंड की अर्ध-घूर्णन वाली
        सप्ताहांत समस्या की विकर्ण-मध्यबिंदु वाली तरकीब, अब सूत्रों में)।
  \item समापन: तीन औज़ार --- \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} सूत्र, दूरी सूत्र, और वृत्त-समीकरणों
        का घटाव। हर एक के लिए बताइए कि उसने इस समस्या में किस तरह का
        प्रश्न सुलझाया, और यह भी कि अगले दो अध्याय इस बक्से में क्या
        जोड़ते हैं (दिशाएँ और प्रवणताएँ: सदिश और रेखा-समीकरण)।
\end{enumerate}
\end{problem}
