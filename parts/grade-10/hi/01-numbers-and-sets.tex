\chapter{संख्याएँ और संख्या-समुच्चय}\label{ch:g10:numbers}

गणित की शुरुआत संख्याओं से होती है, और सभी संख्याएँ एक ही तरह की नहीं होतीं:
गिनती की संख्याएँ, ऋणात्मक संख्याएँ, भिन्न, और $\sqrt 2$ या $\pi$ जैसी संख्याएँ
जिन्हें कोई भिन्न व्यक्त नहीं कर सकती। यह अध्याय उन्हें एक-दूसरे में समाए हुए
परिवारों में व्यवस्थित करता है, संख्या रेखा के हिस्सों का वर्णन करने के लिए
\omterm{def:g10:numbers:interval}{अंतराल} प्रस्तुत करता है, और संख्याओं के बीच की दूरी नापने के लिए \omterm{def:g10:numbers:abs}{निरपेक्ष मान}
का उपयोग करता है।

\section{संख्याओं के परिवार}

\begin{definition}[संख्या-समुच्चय]\label{def:g10:numbers:sets}
\begin{itemize}
  \item $\N$ \emph{प्राकृत संख्याओं}\index{प्राकृत संख्या} का समुच्चय है:
        $0, 1, 2, 3, \dots$
  \item $\Z$ \emph{पूर्णांकों}\index{पूर्णांक} का समुच्चय है: $\dots, -2, -1,
        0, 1, 2, \dots$
  \item $\Q$ \emph{परिमेय संख्याओं}\index{परिमेय संख्या} का समुच्चय है:
        वे सभी भागफल $\frac{p}{q}$ जिनमें $p \in \Z$, $q \in \N$ और
        $q \neq 0$ हैं।
  \item $\R$ \emph{वास्तविक संख्याओं}\index{वास्तविक संख्या} का समुच्चय है:
        वे सभी संख्याएँ जिन्हें संख्या रेखा पर रखा जा सकता है।
\end{itemize}
\end{definition}

\begin{notation}
संकेत $\in$ को ``का अवयव है'' पढ़ा जाता है: $3 \in \N$, $-5 \in \Z$,
$\frac23 \in \Q$। संकेत $\subset$ को ``में अंतर्विष्ट है'' पढ़ा जाता है: हर \omterm{def:g10:numbers:sets}{प्राकृत संख्या}
\omterm{def:g10:numbers:sets}{पूर्णांक} है, हर \omterm{def:g10:numbers:sets}{पूर्णांक} \omterm{def:g10:numbers:sets}{परिमेय संख्या} है
(उदाहरणार्थ $-5 = \frac{-5}{1}$), और हर \omterm{def:g10:numbers:sets}{परिमेय संख्या} वास्तविक है, इसलिए
\[
\N \subset \Z \subset \Q \subset \R .
\]
तिरछी रेखा किसी संकेत का निषेध कर देती है: $\frac12 \notin \Z$।
\end{notation}

\begin{example}\label{ex:g10:numbers:classify}
आइए कुछ संख्याओं को उस सबसे छोटे परिवार में रखें जिसमें वे आती हैं।
\begin{itemize}
  \item $\frac{28}{4} = 7$, इसलिए $\frac{28}{4} \in \N$, यद्यपि वह भिन्न के
        रूप में लिखी गई है: पहले हमेशा सरल कीजिए।
  \item $-3.5 = -\frac{35}{10} = -\frac72$ परिमेय है, पर \omterm{def:g10:numbers:sets}{पूर्णांक} नहीं।
  \item $0.333\dots$ (अंक $3$ अनंत बार दोहराया गया) $\frac13$ के बराबर है,
        जो एक \omterm{def:g10:numbers:sets}{परिमेय संख्या} है।
  \item $\sqrt 2$ और $\pi$ वास्तविक हैं पर परिमेय नहीं, जैसा हम $\sqrt 2$ के लिए
        अभी देखने वाले हैं; ऐसी संख्याओं को
        \emph{अपरिमेय}\index{अपरिमेय संख्या} कहा जाता है।
\end{itemize}
\end{example}

\begin{theorem}[$\sqrt 2$ की अपरिमेयता]\label{thm:g10:numbers:sqrt2}
संख्या $\sqrt 2$ परिमेय नहीं है: \omterm{def:g10:numbers:sets}{पूर्णांकों} की किसी भी भिन्न का वर्ग $2$ नहीं होता।
\end{theorem}

\begin{proof}
हम छोटे-छोटे चरणों में विरोधाभास द्वारा तर्क करते हैं।
\begin{enumerate}
  \item मान लीजिए $\sqrt 2 = \frac{p}{q}$, जहाँ $p$ और $q$ धनात्मक \omterm{def:g10:numbers:sets}{पूर्णांक} हैं
        और भिन्न पूरी तरह सरल की जा चुकी है, इसलिए $p$ और $q$
        दोनों सम नहीं हैं।
  \item दोनों पक्षों का वर्ग करने पर $2 = \frac{p^2}{q^2}$ मिलता है, अर्थात्
        $p^2 = 2q^2$। तो $p^2$ सम है।
  \item यदि $p$ विषम होता, मान लीजिए $p = 2k+1$, तो
        $p^2 = 4k^2 + 4k + 1$ विषम होता। चूँकि $p^2$ सम है, $p$ का सम होना
        अनिवार्य है: किसी \omterm{def:g10:numbers:sets}{पूर्णांक} $k$ के लिए $p = 2k$।
  \item प्रतिस्थापन से: $(2k)^2 = 2q^2$, इसलिए $4k^2 = 2q^2$, इसलिए
        $q^2 = 2k^2$। चरण 3 के उसी तर्क से $q$ सम है।
  \item अब $p$ और $q$ दोनों सम हैं, जो चरण 1 का खंडन करता है।
        कल्पना असंभव थी: $\sqrt 2$ \omterm{ex:g10:numbers:classify}{अपरिमेय} है।
\end{enumerate}
\end{proof}

\begin{remark}
\omterm{def:g10:numbers:sets}{परिमेय संख्याएँ} ठीक वही संख्याएँ हैं जिनका दशमलव प्रसार या तो समाप्त हो
जाता है (जैसे $\frac72 = 3.5$) या अंततः एक ही खंड को अनंत बार दोहराता है
(जैसे $\frac13 = 0.333\dots$ या $\frac{1}{7} = 0.142857\,142857\dots$)।
$\sqrt2 = 1.41421356\dots$ जैसी \omterm{ex:g10:numbers:classify}{अपरिमेय} संख्याओं के दशमलव अंक कभी किसी दोहराव के ढर्रे में
नहीं बैठते। इस लक्षण-वर्णन को हम इस स्तर पर स्वीकार कर लेते हैं।
\end{remark}

\section{संख्या रेखा और अंतराल}

\omterm{def:g10:numbers:sets}{वास्तविक संख्याएँ} एक रेखा को भर देती हैं: मूल बिंदु $0$ और लंबाई का एक
मात्रक चुन लेने पर हर \omterm{def:g10:numbers:sets}{वास्तविक संख्या} ठीक एक बिंदु से मेल खाती है।

\begin{definition}[अंतराल]\label{def:g10:numbers:interval}
मान लीजिए $a$ और $b$ \omterm{def:g10:numbers:sets}{वास्तविक संख्याएँ} हैं और $a < b$ है।
\emph{अंतराल}\index{अंतराल} दो सीमाओं के बीच पड़ने वाली सभी \omterm{def:g10:numbers:sets}{वास्तविक
संख्याओं} का समुच्चय है। वर्ग कोष्ठक का अर्थ है कि सीमा सम्मिलित है, और
गोल कोष्ठक का अर्थ है कि वह बाहर है:
\begin{center}
\begin{tabular}{c|c}
संकेतन & विवरण \\
\hline
$\intcc{a}{b}$ & $a \leq x \leq b$ (दोनों सिरे सम्मिलित) \\
$\intoo{a}{b}$ & $a < x < b$ (दोनों सिरे बाहर) \\
$\intco{a}{b}$ & $a \leq x < b$ \\
$\intoc{a}{b}$ & $a < x \leq b$ \\
$\intco{a}{+\infty}$ & $x \geq a$ \\
$\intoo{-\infty}{b}$ & $x < b$ \\
\end{tabular}
\end{center}
संकेत $-\infty$ और $+\infty$ (``अनंत'') संख्याएँ नहीं हैं; वे केवल यह कहने का एक
तरीका हैं कि अंतराल अनंत तक चलता जाता है, और उनके साथ का कोष्ठक हमेशा
गोल होता है। पूरी रेखा $\R$ स्वयं अंतराल $\intoo{-\infty}{+\infty}$ है।
\end{definition}

\begin{definition}[प्रतिच्छेदन और सम्मिलन]\label{def:g10:numbers:interunion}
मान लीजिए $I$ और $J$ \omterm{def:g10:numbers:sets}{वास्तविक संख्याओं} के दो समुच्चय हैं।
\emph{प्रतिच्छेदन}\index{प्रतिच्छेदन} $I \cap J$ (``$I$ और $J$'') उन
संख्याओं का समुच्चय है जो दोनों में हैं; \emph{सम्मिलन}\index{सम्मिलन}
$I \cup J$ (``$I$ या $J$'') उन संख्याओं का समुच्चय है जो इनमें से कम से कम
एक में हैं।
\end{definition}

\begin{example}\label{ex:g10:numbers:interunion}
$I = \intcc{-1}{3}$ और $J = \intoo{1}{5}$ लीजिए। दोनों को एक ही रेखा पर खींचिए: वे $1$ और $3$
के बीच एक-दूसरे को ढकते हैं। इसलिए
\[
I \cap J = \intoc{1}{3},
\qquad
I \cup J = \intco{-1}{5}.
\]
कोष्ठकों पर ध्यान दीजिए: $1 \notin J$, इसलिए $1 \notin I \cap J$; पर $3 \in I$ और
$3 \in J$, इसलिए $3 \in I \cap J$।
\end{example}

\begin{method}[अंतरालों के साथ काम करना]\label{met:g10:numbers:intervals}
दो \omterm{def:g10:numbers:interval}{अंतरालों} का \omterm{def:g10:numbers:interunion}{प्रतिच्छेदन} या \omterm{def:g10:numbers:interunion}{सम्मिलन} निकालने के लिए:
\begin{enumerate}
  \item संख्या रेखा खींचिए और चारों सीमाएँ अंकित कीजिए;
  \item पहले \omterm{def:g10:numbers:interval}{अंतराल} को रेखा के ऊपर और दूसरे को रेखा के नीचे छायांकित कीजिए;
  \item \omterm{def:g10:numbers:interunion}{प्रतिच्छेदन} वहाँ है जहाँ दोनों छायाएँ एक-दूसरे को ढकती हैं, और
        \omterm{def:g10:numbers:interunion}{सम्मिलन} वहाँ है जहाँ कम से कम एक छाया मौजूद है;
  \item हर कोष्ठक का निर्णय यह जाँचकर कीजिए कि सीमा स्वयं दोनों समुच्चयों
        में है (\omterm{def:g10:numbers:interunion}{प्रतिच्छेदन}) या कम से कम एक में (\omterm{def:g10:numbers:interunion}{सम्मिलन})।
\end{enumerate}
\end{method}

\section{निरपेक्ष मान और दूरी}

\begin{definition}[निरपेक्ष मान]\label{def:g10:numbers:abs}
\omterm{def:g10:numbers:sets}{वास्तविक संख्या} $x$ का \emph{निरपेक्ष मान}\index{निरपेक्ष मान} है
\[
\abs{x} =
\begin{cases}
x & \text{यदि } x \geq 0, \\
-x & \text{यदि } x < 0.
\end{cases}
\]
उदाहरण के लिए $\abs{7} = 7$ और $\abs{-4} = 4$। हर स्थिति में
$\abs{x} \geq 0$।
\end{definition}

\begin{proposition}[रेखा पर दूरी]\label{prop:g10:numbers:distance}
सभी \omterm{def:g10:numbers:sets}{वास्तविक संख्याओं} $a$ और $b$ के लिए, संख्या रेखा पर बिंदुओं $a$
और $b$ के बीच की दूरी $\abs{b - a}$ है। विशेष रूप से $\abs{x}$, $x$ की $0$ से
दूरी है।
\end{proposition}

\begin{proof}
यदि $b \geq a$ है, तो $a$ से $b$ तक की दूरी $b - a \geq 0$ है, जो $\abs{b-a}$ के बराबर
है। यदि $b < a$ है, तो दूरी $a - b = -(b - a) > 0$ है, जो \omterm{def:g10:numbers:abs}{निरपेक्ष मान} की परिभाषा से फिर
$\abs{b - a}$ ही है।
\end{proof}

\begin{proposition}[निरपेक्ष मान और अंतराल]\label{prop:g10:numbers:absinterval}
मान लीजिए $a$ एक \omterm{def:g10:numbers:sets}{वास्तविक संख्या} है और $r > 0$ है। तब
\[
\abs{x - a} \leq r
\quad\text{ठीक तब जब}\quad
x \in \intcc{a - r}{a + r}.
\]
\end{proposition}

\begin{proof}
$\abs{x-a} \leq r$ का अर्थ है कि $x$ की $a$ से दूरी अधिक से अधिक $r$ है,
अर्थात् $x$ दोनों ओर $a$ से $r$ से अधिक दूर नहीं है। इसे संतुष्ट
करने वाली संख्याएँ ठीक वे हैं जो $a - r$ और $a + r$ के बीच हैं, सीमाएँ
सम्मिलित।
\end{proof}

  \draw[omDef, very thick] (1.5,0.02) -- (5.5,0.02);

\begin{example}\label{ex:g10:numbers:abs}
$\abs{x - 3} \leq 2$ हल कीजिए। हल वे संख्याएँ हैं जो $3$ से अधिक से अधिक $2$ की
दूरी पर हैं: \omterm{def:g10:numbers:interval}{अंतराल} $\intcc{1}{5}$। इसके उलट, \omterm{def:g10:numbers:interval}{अंतराल} $\intcc{-1}{7}$ का केंद्र $\frac{-1+7}{2} = 3$ और
त्रिज्या $\frac{7-(-1)}{2} = 4$ है, इसलिए उसे $\abs{x - 3} \leq 4$ द्वारा व्यक्त किया जाता है।
\end{example}

\section{सन्निकटन}

\omterm{ex:g10:numbers:classify}{अपरिमेय} संख्याओं को, और अधिकांश भिन्नों को भी, सीमित दशमलव अंकों में ठीक-ठीक
नहीं लिखा जा सकता, इसलिए व्यवहार में हम उनका \omterm{def:g10:numbers:approx}{सन्निकटन} करते हैं।

\begin{definition}[दी हुई यथार्थता तक सन्निकटन]\label{def:g10:numbers:approx}
संख्या $d$ को $x$ का $10^{-n}$ तक \emph{सन्निकटन}\index{सन्निकटन} कहते
हैं जब $\abs{x - d} \leq 10^{-n}$ हो। दशमलव प्रसार को $n$-वें अंक के बाद काट देने या पूर्णांकित
करने, दोनों से ऐसे सन्निकटन मिलते हैं।
\end{definition}

\begin{example}
$\pi = 3.14159\,26\dots$ से: कटाव $3.141$ और पूर्णांकन $3.142$ दोनों $\pi$ के $10^{-3}$ तक \omterm{def:g10:numbers:approx}{सन्निकटन}
हैं। पूर्णांकन $\frac12 \times 10^{-3}$ से कम दूरी पर है, जबकि कटाव केवल $10^{-3}$ की गारंटी देता
है। $3.141 \leq \pi \leq 3.142$ लिखना $\pi$ को दो दशमलव सीमाओं के बीच \emph{घेरना} है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g10:numbers:1}
हर संख्या के लिए $\N$, $\Z$, $\Q$, $\R$ में से वह सबसे छोटा समुच्चय
बताइए जिसमें वह आती है:
\[
\frac{15}{3}, \qquad -7, \qquad \frac{22}{7}, \qquad \sqrt{9}, \qquad
\sqrt{10}, \qquad -2.4, \qquad 0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:2}
हर कथन को \omterm{def:g10:numbers:interval}{अंतराल} संकेतन में लिखिए और फिर उसे संख्या रेखा पर खींचिए:
(a) $-2 \leq x < 5$; (b) $x > 3$; (c) $x \leq -1$;
(d) $x$ की $2$ से दूरी अधिक से अधिक $3$ है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:3}
निम्नलिखित के लिए $I \cap J$ और $I \cup J$ निकालिए:
\[
\text{(a) } I = \intcc{-3}{2},\ J = \intco{0}{4};
\qquad
\text{(b) } I = \intoo{-\infty}{1},\ J = \intco{-2}{+\infty}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:4}
कैलकुलेटर के बिना निकालिए:
\[
\abs{-6}, \qquad \abs{4 - 9}, \qquad \abs{-3 - 5}, \qquad
\abs{\sqrt 2 - 1}, \qquad \abs{1 - \sqrt 2}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:5}
निम्नलिखित समीकरणों और असमिकाओं को हल कीजिए और हल-समुच्चय बताइए:
\[
\abs{x} = 5, \qquad \abs{x - 1} = 3, \qquad \abs{x - 4} \leq 1, \qquad
\abs{x + 2} < 3 .
\]
(ध्यान दीजिए कि $\abs{x+2} = \abs{x - (-2)}$ वस्तुतः $-2$ से दूरी है।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:6}
हर \omterm{def:g10:numbers:interval}{अंतराल} को $\abs{x - a} \leq r$ या $\abs{x - a} < r$ के रूप की असमिका से व्यक्त कीजिए:
\[
\intcc{2}{8}, \qquad \intoo{-5}{1}, \qquad \intcc{-7}{-3}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:7}
दिखाइए कि $0.272727\dots$ (खंड $27$ अनंत बार दोहराया गया) परिमेय है।
(संकेत: उसे $x$ कहिए और $100x - x$ निकालिए।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:8}
सत्य या असत्य? हर उत्तर को तर्क या प्रतिउदाहरण से पुष्ट कीजिए।
\begin{enumerate}
  \item दो \omterm{def:g10:numbers:sets}{पूर्णांकों} का योग \omterm{def:g10:numbers:sets}{पूर्णांक} होता है।
  \item दो \omterm{def:g10:numbers:sets}{पूर्णांकों} का भागफल \omterm{def:g10:numbers:sets}{पूर्णांक} होता है।
  \item दो \omterm{def:g10:numbers:sets}{परिमेय संख्याओं} का योग परिमेय होता है।
  \item एक परिमेय और एक \omterm{ex:g10:numbers:classify}{अपरिमेय संख्या} का योग \omterm{ex:g10:numbers:classify}{अपरिमेय} होता है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:9}
$1.414 \leq \sqrt 2 \leq 1.415$ का उपयोग करके संख्याओं $2\sqrt2$, $\sqrt2 + 3$ और $-\sqrt2$ को दो दशमलव सीमाओं
के बीच घेरिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:numbers:10}
\cref{thm:g10:numbers:sqrt2} की उपपत्ति को ढालकर दिखाइए कि $\sqrt 3$ \omterm{ex:g10:numbers:classify}{अपरिमेय} है। (``सम'' के स्थान
पर ``$3$ का गुणज'' रखिए: पहले जाँचिए कि यदि $p^2$, $3$ का गुणज है तो
$p$ भी है --- इसके लिए $p$ को $3$ से भाग देने पर बचने वाले शेषफल
$0$, $1$, $2$ की जाँच कीजिए।)
\end{exercise}

\section{समस्या: किन्हीं दो संख्याओं के बीच}

\begin{problem}[{सप्ताहांत समस्या --- परिमेय और अपरिमेय संख्याएँ आपस में
गुँथी हुई हैं: हर अंतराल, चाहे कितना ही छोटा हो, दोनों में से अनंत संख्याएँ
रखता है, और कोई भी माप उन्हें कभी अलग नहीं पहचान सकता}]
\label{pb:g10:numbers:1}
\omterm{def:g10:numbers:sets}{परिमेय संख्याएँ} भीड़ जैसी लगती हैं (सारी भिन्नें!) और \omterm{ex:g10:numbers:classify}{अपरिमेय संख्याएँ}
विरले अपवाद जैसी ($\sqrt2$, $\pi$)। यह समस्या असली चित्र दिखाती है: ये दोनों
परिवार इतनी बारीकी से \emph{गुँथे} हुए हैं कि संख्या रेखा का हर \omterm{def:g10:numbers:interval}{अंतराल},
चाहे कितना ही सूक्ष्म हो, दोनों में से अनंत संख्याएँ रखता है --- और इसके
विचित्र परिणाम निकलते हैं, जैसे यह: कोई भी भौतिक माप, चाहे कितना ही
यथार्थ हो, यह कभी तय नहीं कर सकता कि कोई लंबाई परिमेय है या नहीं।

\medskip
\noindent\textbf{भाग I --- चार राज्य।}
\begin{enumerate}
  \item हर संख्या के लिए $\N$, $\Z$, $\Q$, $\R$ में से वह सबसे छोटा
        समुच्चय बताइए जिसमें वह आती है
        (\cref{def:g10:numbers:sets}): $-7$; \ $\frac{13}{4}$; \
        $\sqrt{16}$; \ $0.121212\ldots$ (माध्यमिक विद्यालय खंड की आवर्ती दशमलव वाली
        सप्ताहांत समस्या याद कीजिए); \ $\sqrt8$; \ $\pi$ (इसकी अपरिमेयता
        स्वीकार कर लीजिए --- वह केवल स्नातक खंडों में सिद्ध होती है)।
  \item सिद्ध कीजिए कि $\Q$ योग और गुणन के अंतर्गत बंद है: यदि $x = \frac pq$
        और $y = \frac rs$ परिमेय हैं, तो $x + y$ और $xy$ को एक-एक भिन्न के रूप
        में लिखिए।
  \item विरोधाभास द्वारा निष्कर्ष निकालिए: (a) एक परिमेय और एक \omterm{ex:g10:numbers:classify}{अपरिमेय
        संख्या} का योग \omterm{ex:g10:numbers:classify}{अपरिमेय} होता है; (b) एक \emph{शून्येतर} परिमेय और
        एक \omterm{ex:g10:numbers:classify}{अपरिमेय संख्या} का गुणनफल \omterm{ex:g10:numbers:classify}{अपरिमेय} होता है।
  \item दिखाइए कि \omterm{ex:g10:numbers:classify}{अपरिमेय} संख्याओं का समुच्चय इनमें से \emph{किसी भी}
        संक्रिया के अंतर्गत बंद नहीं है: ऐसी दो \omterm{ex:g10:numbers:classify}{अपरिमेय संख्याएँ} दीजिए
        जिनका योग परिमेय हो, और ऐसी दो जिनका गुणनफल परिमेय हो।
  \item $\sqrt2 + \sqrt8$ और $\sqrt2 \times \sqrt8$ को चारों राज्यों में स्थान दीजिए (पहले $\sqrt8$ को
        सरल कीजिए --- माध्यमिक विद्यालय खंड की विधि)।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- \omterm{def:g10:numbers:sets}{परिमेय संख्याएँ} सघन हैं।}
\begin{enumerate}[resume]
  \item $3.47$ और $3.48$ के ठीक बीच में एक \omterm{def:g10:numbers:sets}{परिमेय संख्या} खोजिए; फिर दूसरी;
        और फिर बताइए कि जितनी चाहें उतनी कैसे बनाई जा सकती हैं (माध्यमिक
        विद्यालय खंड की ``कोई अगली संख्या नहीं'' वाली सप्ताहांत समस्या का
        वही ज़ूम, अब उपपत्ति सामने रखकर)।
  \item सामान्य प्रमेय। मान लीजिए $a < b$ कोई दो \omterm{def:g10:numbers:sets}{वास्तविक संख्याएँ} हैं
        जिनका अंतर $g = b - a > 0$ है। ऐसा $n$ चुनिए कि $10^{-n} < g$ हो, और $10^{-n}$ के
        गुणजों पर विचार कीजिए (डग $10^{-n}$ वाली दशमलव जाली)। समझाइए कि
        कम से कम एक जाली-बिंदु $a$ और $b$ के ठीक बीच क्यों पड़ता है,
        और निष्कर्ष निकालिए: \emph{धनात्मक लंबाई का हर \omterm{def:g10:numbers:interval}{अंतराल} एक \omterm{def:g10:numbers:sets}{परिमेय
        संख्या} रखता है}।
  \item निष्कर्ष को और आगे बढ़ाइए: ऐसा हर \omterm{def:g10:numbers:interval}{अंतराल} \emph{अनंत} \omterm{def:g10:numbers:sets}{परिमेय
        संख्याएँ} रखता है। (प्रश्न 7 को एक छोटे \omterm{def:g10:numbers:interval}{अंतराल} के भीतर फिर से
        लगाइए।)
  \item अब \omterm{ex:g10:numbers:classify}{अपरिमेय संख्याएँ}: $a < b$ दिए होने पर उसके ठीक भीतर एक \omterm{def:g10:numbers:sets}{परिमेय
        संख्या} $r$ लीजिए (प्रश्न 7) और संख्याओं $r + \frac{\sqrt2}{10^k}$ पर विचार कीजिए।
        प्रश्न 3 का उपयोग करके दिखाइए कि वे \omterm{ex:g10:numbers:classify}{अपरिमेय} हैं, और यह भी कि
        पर्याप्त बड़े $k$ के लिए वे \omterm{def:g10:numbers:interval}{अंतराल} में ही रहती हैं:
        \emph{हर \omterm{def:g10:numbers:interval}{अंतराल} अनंत \omterm{ex:g10:numbers:classify}{अपरिमेय संख्याएँ} भी रखता है}।
  \item दो पुरानी पहेलियाँ सुलझीं: क्या कोई सबसे छोटी धनात्मक \omterm{def:g10:numbers:sets}{वास्तविक
        संख्या} है? क्या ``$3$ के ठीक बाद'' वाली कोई \omterm{def:g10:numbers:sets}{वास्तविक संख्या} है?
        दोनों का उत्तर प्रश्न 7 की प्रमेय से दीजिए, और उसके बचपन वाले रूप
        को नमन कीजिए (माध्यमिक विद्यालय खंड की ``कोई अगली संख्या नहीं''
        वाली सप्ताहांत समस्या)।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{def:g10:numbers:abs}{निरपेक्ष मान}, $\R$ की ज्यामिति।}
\begin{enumerate}[resume]
  \item हल कीजिए और हलों को \omterm{def:g10:numbers:interval}{अंतरालों} या \omterm{def:g10:numbers:interunion}{सम्मिलनों} के रूप में लिखिए
        (\cref{prop:g10:numbers:absinterval}):
        $\abs{x - 5} = 2$; \quad $\abs{x - 5} < 2$; \quad
        $\abs{x + 1} \geq 3$।
  \item एक पिस्टन $80$ mm का बनाया जाना है, जिसमें $0.05$ mm की सहनशीलता
        है। इस अपेक्षा को पहले \omterm{def:g10:numbers:abs}{निरपेक्ष मान} से और फिर \omterm{def:g10:numbers:interval}{अंतराल} के रूप में
        लिखिए। दो पिस्टन $79.97$ और $80.06$ mm नापे गए: क्या निर्णय है?
  \item रेखा पर त्रिभुज असमिका: $\abs{a + b} \leq \abs a + \abs b$। इसे $(a, b) = (3, -5)$ और $(-2, -7)$ पर जाँचिए,
        फिर इसे तब सिद्ध कीजिए जब $a$ और $b$ के चिह्न समान हों और तब
        भी जब वे विपरीत हों, और ठीक-ठीक बताइए कि समता कब होती है।
  \item $\abs{x - 2} = \abs{x + 4}$ को रेखा पर दूरियों की समता के रूप में पढ़कर हल कीजिए।
        माध्यमिक विद्यालय खंड की ``दो दर्पण'' वाली सप्ताहांत समस्या का
        कौन-सा बिंदु आपने अभी एक विमा नीचे निकाल लिया है?
  \item $\sqrt{x^2}$ को सरल कीजिए --- सावधानी से। अपने सूत्र को $x = 3$ और $x = -3$
        पर जाँचिए, और फिर उसका उपयोग करके $x^2 < 9$ को \omterm{def:g10:numbers:abs}{निरपेक्ष मान} की सहायता
        से हल कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- जो कोई माप तय नहीं कर सकता।}
\begin{enumerate}[resume]
  \item एक भौतिकशास्त्री एक छड़ नापता है: $1.41421 \pm 0.00001$ m. क्या ऐसा कोई भी माप
        --- यह वाला या भविष्य का कोई और यथार्थ माप --- कभी \emph{सिद्ध}
        कर सकता है कि छड़ की लंबाई \omterm{ex:g10:numbers:classify}{अपरिमेय} है? या कि वह परिमेय है?
        सहनशीलता के \omterm{def:g10:numbers:interval}{अंतराल} पर प्रश्न 7 और 9 लगाइए, और समझाइए कि
        अपरिमेयता का निपटारा केवल \emph{उपपत्ति} ही कर सकती है (जैसे
        विकर्ण के लिए, माध्यमिक विद्यालय खंड की अपरिमेयता वाली सप्ताहांत
        समस्या में)।
  \item $\sqrt2$ के कटाव $1.4$, $1.41$, $1.414$, $1.4142,\dots$ सभी परिमेय हैं। वे इस
        बारे में क्या दिखाते हैं कि $\Q$ हर \omterm{ex:g10:numbers:classify}{अपरिमेय संख्या} से कितना
        सटकर बैठता है? सामान्य तथ्य को शब्दों में रखिए।
  \item अंतराल-अंकगणित: नापे गए मान $a = 2.5 \pm 0.1$ और $b = 1.2 \pm 0.1$ हैं। $a + b$ और $a \times b$
        को निश्चित सीमाओं के बीच घेरिए। कौन-सी संक्रिया यथार्थता को
        अधिक बिगाड़ती है?
  \item अभियंताओं का एक नियम कहता है कि स्वतंत्र मापों की त्रुटियाँ योग
        के लिए ``जुड़ जाती हैं'': त्रिभुज असमिका (प्रश्न 13) से व्यक्त
        कीजिए कि $a + b$ की त्रुटि अधिक से अधिक त्रुटियों का योग क्यों है
        --- यह असमिका ही वह अभियांत्रिक नियम है।
  \item समापन, एक छोटे अनुच्छेद में: इस समस्या और इसकी पूर्वजों द्वारा
        बनाए गए वास्तविक रेखा के चित्र को जोड़िए --- कोई अगली संख्या नहीं
        (माध्यमिक विद्यालय खंड की ``कोई अगली संख्या नहीं'' वाली सप्ताहांत
        समस्या), \omterm{def:g10:numbers:sets}{परिमेय संख्याएँ} $=$ आवर्ती दशमलव (माध्यमिक विद्यालय
        खंड की आवर्ती दशमलव वाली सप्ताहांत समस्या), दोनों परिवार सघन
        (प्रश्न 8 और 9), माप सदा अनिर्णीत (प्रश्न 16)। अंत में वह संकेत
        दीजिए जिसे यह अध्याय अभी सिद्ध नहीं कर सकता: एक सुनिश्चित अर्थ
        में \omterm{ex:g10:numbers:classify}{अपरिमेय संख्याएँ} \omterm{def:g10:numbers:sets}{परिमेय संख्याओं} से \emph{कहीं अधिक} हैं ---
        स्नातक खंड अनंतताओं की गिनती करते हैं।
