\chapter{रेखाओं के समीकरण और रैखिक निकाय}\label{ch:g10:lines}

\omterm{def:g10:coordgeom:system}{निर्देशांक} समतल की सरल रेखा एक \omterm{def:g10:algebra:equation}{समीकरण} से बताई जाती है, और दो रेखाएँ मिलकर
\omterm{def:g10:algebra:equation}{समीकरणों} का एक निकाय बनाती हैं जिसके हल उनके प्रतिच्छेद बिंदु होते हैं। यह
अध्याय तीन भाषाओं को जोड़ता है --- ज्यामितीय (रेखाएँ), बीजगणितीय (\omterm{def:g10:algebra:equation}{समीकरण})
और फलनात्मक (\cref{ch:g10:reffunc} के \omterm{def:g10:reffunc:affine}{एकघात फलन}) --- और \omterm{def:g10:lines:system}{रैखिक निकाय} हल करने के दो मानक तरीक़े
सिखाता है।

\section{रेखा का समीकरण}

\begin{theorem}[संक्षिप्त समीकरण]\label{thm:g10:lines:reduced}
किसी \omterm{def:g10:coordgeom:system}{निर्देशांक-पद्धति} में:
\begin{enumerate}
  \item हर अ-ऊर्ध्वाधर रेखा का एक \omterm{def:g10:algebra:equation}{समीकरण} इस रूप का होता है
        \[
        y = mx + p ,
        \]
        जिसमें $m$ (\emph{प्रवणता}\index{प्रवणता}) और $p$
        (\emph{$y$-अंतःखंड}) अद्वितीय रूप से नियत हैं --- यही रेखा का
        \emph{संक्षिप्त समीकरण}\index{संक्षिप्त समीकरण} है;
  \item हर ऊर्ध्वाधर रेखा का \omterm{def:g10:algebra:equation}{समीकरण} $x = c$ होता है;
  \item कोई बिंदु रेखा पर ठीक तब होता है जब उसके \omterm{def:g10:coordgeom:system}{निर्देशांक} \omterm{def:g10:algebra:equation}{समीकरण} को
        संतुष्ट करें।
\end{enumerate}
\end{theorem}
\admitted

\begin{proposition}[दो बिंदुओं से प्रवणता]\label{prop:g10:lines:slope}
$A(x_A, y_A)$ और $B(x_B, y_B)$ से होकर जाने वाली (अ-ऊर्ध्वाधर) रेखा की \omterm{def:g10:reffunc:affine}{प्रवणता}, जहाँ $x_A \neq x_B$ है,
यह है:
\[
m = \frac{y_B - y_A}{x_B - x_A}
\qquad
\text{(``$x$ में परिवर्तन पर $y$ में परिवर्तन'')।}
\]
\end{proposition}

\begin{proof}
दोनों बिंदु $y = mx + p$ को संतुष्ट करते हैं, इसलिए $y_B - y_A = (mx_B + p) - (mx_A + p)
= m(x_B - x_A)$; अब $x_B - x_A \neq 0$ से भाग दीजिए।
\end{proof}

\begin{method}[दो बिंदुओं से जाने वाली रेखा का समीकरण ज्ञात करना]\label{met:g10:lines:through}
$A$ और $B$ दिए हों, जहाँ $x_A \neq x_B$ है:
\begin{enumerate}
  \item \omterm{def:g10:reffunc:affine}{प्रवणता} $m = \dfrac{y_B - y_A}{x_B - x_A}$ निकालिए;
  \item $y = mx + p$ लिखिए और $p$ ज्ञात करने के लिए $A$ (या $B$) के \omterm{def:g10:coordgeom:system}{निर्देशांक}
        प्रतिस्थापित कीजिए;
  \item अंतिम \omterm{def:g10:algebra:equation}{समीकरण} की जाँच \emph{दूसरे} बिंदु से कीजिए।
\end{enumerate}
यदि $x_A = x_B$ हो, तो रेखा ऊर्ध्वाधर है: उसका \omterm{def:g10:algebra:equation}{समीकरण} $x = x_A$ है।
\end{method}

\begin{example}\label{ex:g10:lines:through}
$A(1, 5)$ और $B(3, 1)$ से होकर जाने वाली रेखा। \omterm{def:g10:reffunc:affine}{प्रवणता}: $m = \frac{1 - 5}{3 - 1} = \frac{-4}{2} = -2$। फिर $y = -2x + p$; बिंदु $A$ से
$5 = -2 \times 1 + p$ मिलता है, इसलिए $p = 7$। \omterm{def:g10:algebra:equation}{समीकरण}: $y = -2x + 7$। $B$ से जाँच: $-2 \times 3 + 7 = 1$।
\end{example}

\begin{proposition}[समांतर रेखाएँ]\label{prop:g10:lines:parallel}
दो अ-ऊर्ध्वाधर रेखाएँ समांतर हैं यदि और केवल यदि उनकी \omterm{def:g10:reffunc:affine}{प्रवणता} एक ही हो। दो
भिन्न समांतर रेखाओं का कोई उभयनिष्ठ बिंदु नहीं होता; भिन्न प्रवणताओं वाली दो
रेखाओं का ठीक एक होता है।
\end{proposition}

\begin{proof}
मान लीजिए $y = mx + p$ और $y = m'x + p'$ दोनों रेखाएँ हैं। कोई उभयनिष्ठ बिंदु $mx + p = m'x + p'$ को हल करता
है, अर्थात् $(m - m')x = p' - p$ को। यदि $m \neq m'$ हो, तो इसका ठीक एक हल $x = \frac{p' - p}{m - m'}$ है: एक प्रतिच्छेद
बिंदु। यदि $m = m'$ हो, तो \omterm{def:g10:algebra:equation}{समीकरण} $0 = p' - p$ बन जाता है: $p \neq p'$ होने पर कोई हल नहीं
(भिन्न समांतर रेखाएँ), और $p = p'$ होने पर हर $x$ हल है (एक ही रेखा)। इस
प्रकार समांतरता (कोई प्रतिच्छेद न होना, या बराबर होना) $m = m'$ के तुल्य है।
\end{proof}

\begin{remark}[दिक् सदिश]
रेखा $y = mx + p$ दाईं ओर हर एक मात्रक पर $m$ मात्रक ऊपर चढ़ती है, इसलिए \omterm{def:g10:vectors:vector}{सदिश} $\vec u\,(1, m)$
रेखा का एक \emph{दिक् सदिश}\index{दिक् सदिश} है: रेखा $\vec u$ के समांतर है।
दो रेखाएँ समांतर ठीक तब हैं जब उनके दिक् \omterm{def:g10:vectors:vector}{सदिश} \omterm{def:g10:vectors:collinear}{संरेखी} हों (\cref{thm:g10:vectors:det})।
\end{remark}

\section{रैखिक निकाय}

\begin{definition}[रैखिक निकाय]\label{def:g10:lines:system}
अज्ञातों $x$ और $y$ में \emph{दो रैखिक समीकरणों का
निकाय}\index{रैखिक निकाय} \omterm{def:g10:algebra:equation}{समीकरणों} का एक ऐसा युग्म है
\[
\begin{cases}
a x + b y = c \\
a' x + b' y = c'
\end{cases}
\]
जिसे \emph{एक साथ} संतुष्ट करना है। \emph{हल} वह युग्म $(x, y)$ है जो दोनों को
संतुष्ट करे। हर \omterm{def:g10:algebra:equation}{समीकरण} एक रेखा बताता है, इसलिए निकाय हल करने का अर्थ है दो
रेखाओं का प्रतिच्छेद ज्ञात करना।
\end{definition}

\begin{method}[प्रतिस्थापन]\label{met:g10:lines:substitution}
\begin{enumerate}
  \item किसी एक \omterm{def:g10:algebra:equation}{समीकरण} को किसी एक अज्ञात के लिए हल कीजिए --- सबसे आसान वाला
        चुनिए, जैसे $y$ को $x$ के पदों में व्यक्त कीजिए;
  \item इस व्यंजक को \emph{दूसरे} \omterm{def:g10:algebra:equation}{समीकरण} में रखिए, जिसमें अब केवल $x$ बचता
        है;
  \item $x$ के लिए हल कीजिए, फिर चरण 1 से $y$ निकालिए;
  \item युग्म की जाँच दोनों मूल \omterm{def:g10:algebra:equation}{समीकरणों} में कीजिए।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:lines:substitution}
$\begin{cases} x + 2y = 8 \\ 3x - y = 3 \end{cases}$ को प्रतिस्थापन से हल कीजिए। दूसरे \omterm{def:g10:algebra:equation}{समीकरण} से $y = 3x - 3$ मिलता है। इसे पहले में
रखिए:
\[
x + 2(3x - 3) = 8
\ \Longleftrightarrow\
x + 6x - 6 = 8
\ \Longleftrightarrow\
7x = 14
\ \Longleftrightarrow\
x = 2 ,
\]
फिर $y = 3 \times 2 - 3 = 3$। हल युग्म $(2, 3)$ है। जाँच: $2 + 2 \times 3 = 8$ और $3 \times 2 - 3 = 3$।
\end{example}

\begin{method}[विलोपन]\label{met:g10:lines:elimination}
\begin{enumerate}
  \item हर \omterm{def:g10:algebra:equation}{समीकरण} को ऐसी सोची-समझी संख्या से गुणा कीजिए कि किसी एक अज्ञात
        के गुणांक दोनों \omterm{def:g10:algebra:equation}{समीकरणों} में परस्पर विपरीत हो जाएँ;
  \item \omterm{def:g10:algebra:equation}{समीकरणों} को जोड़िए: वह अज्ञात लुप्त हो जाता है;
  \item बचे हुए एक-अज्ञात \omterm{def:g10:algebra:equation}{समीकरण} को हल कीजिए, फिर पीछे रखकर दूसरा अज्ञात
        ज्ञात कीजिए;
  \item युग्म की जाँच दोनों मूल \omterm{def:g10:algebra:equation}{समीकरणों} में कीजिए।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:lines:elimination}
$\begin{cases} 2x + 3y = 7 \\ 5x - 2y = 8 \end{cases}$ को विलोपन से हल कीजिए। $y$ का विलोप करने के लिए पहले \omterm{def:g10:algebra:equation}{समीकरण} को $2$
से और दूसरे को $3$ से गुणा कीजिए:
\[
\begin{cases}
4x + 6y = 14 \\
15x - 6y = 24
\end{cases}
\]
इन्हें जोड़ने पर $19x = 38$, इसलिए $x = 2$। $2x + 3y = 7$ में रखने पर $4 + 3y = 7$, इसलिए $y = 1$। हल:
$(2, 1)$। दूसरे \omterm{def:g10:algebra:equation}{समीकरण} में जाँच: $5 \times 2 - 2 \times 1 = 8$।
\end{example}

\begin{remark}[कितने हल?]
दो रेखाओं की तरह \omterm{def:g10:lines:system}{रैखिक निकाय} का भी सामान्यतः ठीक एक हल होता है (असमांतर
रेखाएँ), और अपवाद स्वरूप कोई नहीं (भिन्न समांतर रेखाएँ) या अनंत (एक ही रेखा
दो बार)। \cref{thm:g10:vectors:det} की कसौटी निर्णय कर देती है: निकाय का अद्वितीय हल ठीक तब है जब
$ab' - a'b \neq 0$ हो।
\end{remark}

\begin{example}[निकाय से प्रतिरूपण]\label{ex:g10:lines:modeling}
तीन कॉपियों और दो क़लमों का दाम $8$ है; एक कॉपी और चार क़लमों का दाम $6$
है। मान लीजिए $x$ एक कॉपी का और $y$ एक क़लम का दाम है:
\[
\begin{cases}
3x + 2y = 8 \\
x + 4y = 6 .
\end{cases}
\]
दूसरे \omterm{def:g10:algebra:equation}{समीकरण} से $x = 6 - 4y$; प्रतिस्थापन करने पर $3(6 - 4y) + 2y = 8$, इसलिए $18 - 10y = 8$, जिससे $y = 1$
और फिर $x = 2$ मिलता है। कॉपी का दाम $2$ और क़लम का दाम $1$ है।
\end{example}

        \omterm{def:g10:algebra:equation}{समीकरण} ज्ञात कीजिए (माध्यिका किसी शीर्ष को सामने वाली भुजा के
        \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} से जोड़ती है)।
  \item उनका प्रतिच्छेद बिंदु $G$ निकालिए, और जाँचिए कि $G$ तीसरी
        माध्यिका पर भी है। (आपको यह मिलना चाहिए कि $G$ के \omterm{def:g10:coordgeom:system}{निर्देशांक}
        $A$, $B$, $C$ के \omterm{def:g10:coordgeom:system}{निर्देशांकों} के माध्य हैं।)
\end{enumerate}
\end{exercise}

\section{समस्या: एक हल, कोई नहीं, या अनंत}

\begin{problem}[{सप्ताहांत समस्या --- दो रेखाओं की तीन संभव नियतियाँ, एक
सारणिक उनमें से चुनाव करता है, और संसार की सबसे पुरानी पाठ्यपुस्तक यह पहले
से जानती थी}]
\label{pb:g10:lines:1}
दो रैखिक \omterm{def:g10:algebra:equation}{समीकरण}, दो अज्ञात: वे जो रेखा-युग्म खींचते हैं वह एक बार कट सकता
है, कभी नहीं कट सकता, या दोनों एक ही रेखा हो सकती हैं --- और \emph{हर} \omterm{def:g10:lines:system}{रैखिक
निकाय} इन तीन नियतियों में से एक पाता है। यह समस्या उनका वर्गीकरण करती है,
उस संख्या से मिलाती है जो निर्णय करती है (\omterm{def:g10:vectors:vector}{सदिशों} वाले अध्याय की पुरानी
परिचित), चीनी \emph{नौ अध्याय} की दो हज़ार साल पुरानी तीतर-और-ख़रगोश वाली
पहेली हल करती है, और क्षेत्रों के कोनों पर समाप्त होती है --- उस इष्टतमीकरण
का पहला क़दम जिसे आज हर विमान-कंपनी और हर कारख़ाना काम में लाता है।

\medskip
\noindent\textbf{भाग I --- रेखा, धाराप्रवाह।}
\begin{enumerate}
  \item $(1, 2)$ और $(3, 8)$ से होकर जाने वाली रेखा का \omterm{thm:g10:lines:reduced}{संक्षिप्त समीकरण} ज्ञात
        कीजिए (\cref{met:g10:lines:through})।
  \item $y = 3x - 1$, $y = 3x + 4$ और $y = -x + 7$ में से कौन-सी दो रेखाएँ समांतर हैं (\cref{prop:g10:lines:parallel}),
        और असमांतर युग्म कहाँ मिलते हैं? (एक प्रतिच्छेद निकालिए।)
  \item $(2, 1)$ और $(2, 5)$ से होकर जाने वाली रेखा: उसका कोई \omterm{thm:g10:lines:reduced}{संक्षिप्त समीकरण}
        $y = mx + p$ क्यों नहीं है, और उसका \omterm{def:g10:algebra:equation}{समीकरण} क्या है?
  \item क्या बिंदु $(4, 11)$ रेखा $y = 3x - 1$ पर है? और $(-1, -5)$?
  \item $(0, 3)$ से होकर जाने वाली और \omterm{def:g10:reffunc:affine}{प्रवणता} $-2$ वाली रेखा: उसका \omterm{def:g10:algebra:equation}{समीकरण},
        उसके दोनों अक्ष-अंतःखंड, और वह जो त्रिभुज प्रथम चतुर्थांश में से
        काटती है उसका क्षेत्रफल बताइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- तीन नियतियाँ।}
\begin{enumerate}[resume]
  \item प्रतिस्थापन (\cref{met:g10:lines:substitution}) से हल कीजिए:
        $\begin{cases} y = 2x - 3 \\ 3x + y = 7 \end{cases}$
  \item विलोपन (\cref{met:g10:lines:elimination}) से हल कीजिए:
        $\begin{cases} 2x + 3y = 8 \\ 5x - 3y = -1 \end{cases}$
  \item दोनों निकायों का वर्गीकरण कीजिए --- और रेखाओं से उसकी व्याख्या
        कीजिए:
        \[
        \begin{cases} 2x - y = 3 \\ 4x - 2y = 6 \end{cases}
        \qquad\text{और}\qquad
        \begin{cases} 2x - y = 3 \\ 4x - 2y = 1 . \end{cases}
        \]
        पूरा त्रिविभाजन बताइए: काटती हुईं, समांतर, समरूप --- एक हल, कोई
        नहीं, अनंत।
  \item सामान्य निकाय $ax + by = e$, $cx + dy = f$ के लिए निर्णायक संख्या $ad - bc$ है --- वही
        सारणिक जो \cref{pb:g10:vectors:1} की संरेखता-जाँच में था। इस संयोग की व्याख्या कीजिए
        (कौन-से दो \omterm{def:g10:vectors:vector}{सदिश} ठीक तब \omterm{def:g10:vectors:collinear}{संरेखी} होते हैं जब रेखाएँ समांतर या समरूप
        हों?), और प्रश्न 7 तथा 8 के तीनों निकायों के लिए $ad - bc$ का मान
        निकालिए।
  \item $m$ के किस मान के लिए
        $\begin{cases} x + my = 3 \\ 2x + 6y = 7 \end{cases}$
        का कोई हल नहीं है? $m$ के शेष सभी मानों के लिए निकाय हल
        कीजिए\,\dots\ या कम से कम बताइए कि आप कैसे करेंगे।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- नौ अध्याय।}
\begin{enumerate}[resume]
  \item \emph{गणितीय कला के नौ अध्याय} से (चीन, लगभग 2000 वर्ष पूर्व): एक
        पिंजरे में तीतर और ख़रगोश हैं --- कुल $35$ सिर और $94$ टाँगें।
        हर तरह के कितने हैं?
  \item एक रस की दुकान $60\,\%$-फल वाले शरबत को $20\,\%$-फल वाले पेय के साथ
        मिलाकर $35\,\%$ फल वाला $30$ L तैयार करती है। हर एक कितने लीटर?
  \item किसी दो अंकों की संख्या के अंकों का योग $12$ है; अंक आपस में
        बदलने पर वह $18$ घट जाती है। निकाय बनाने के लिए माध्यमिक विद्यालय
        खंड की अंक-बीजगणित वाली सप्ताहांत समस्या का बीजगणित काम में लाकर
        उसे ज्ञात कीजिए।
  \item बेकरी में $3$ कॉफ़ी और $2$ क्रोसां का दाम $9.10$ यूरो है; $2$
        कॉफ़ी और $3$ क्रोसां का दाम $8.90$ यूरो। सुंदर ढंग से हल कीजिए:
        दोनों \omterm{def:g10:algebra:equation}{समीकरणों} को \emph{जोड़ने} से और \emph{घटाने} से आपको सीधे-सीधे
        क्या-क्या पता चलता है?
  \item एक बाज़ार की दुकान का दावा है: $2$ सेब $+ 1$ नाशपाती $5$ यूरो
        में, और $4$ सेब $+ 2$ नाशपाती $9$ यूरो में। हल कीजिए --- या
        बल्कि यह बताइए कि निकाय की नियति उस दुकान के गणित के बारे में क्या
        उजागर करती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- कोनों का राज।}
\begin{enumerate}[resume]
  \item रेखा $y = 3x - 1$ समतल को दो भागों में बाँट देती है। मूल बिंदु किस ओर है?
        समुच्चय $y > 3x - 1$ का वर्णन कीजिए और बताइए कि किसी बिंदु की उसके प्रति
        जाँच कैसे की जाती है।
  \item चार प्रतिबंधों
        \[
        x \geq 0, \qquad y \geq 0, \qquad x + y \leq 4,
        \qquad y \leq 2x + 1,
        \]
        से परिभाषित क्षेत्र खींचिए और उसके चारों कोने के बिंदु निकालिए।
  \item प्रश्न 17 के क्षेत्र पर एक कारख़ाने का लाभ $P = 3x + 2y$ है। चारों कोनों पर
        $P$ का मान निकालिए। यह मानते हुए कि ऐसे क्षेत्र पर ऐसी रैखिक
        राशि का \omterm{def:g10:functions:extrema}{अधिकतम} सदा किसी कोने पर ही मिलता है (कल्पना कीजिए कि रेखाएँ
        $3x + 2y = c$ ($c$ अचर) आरपार बहती जा रही हैं), इष्टतम उत्पादन-योजना बताइए।
  \item माध्यमिक विद्यालय खंड की दो-थर्मामीटर वाली सप्ताहांत समस्या की दो
        टैक्सियाँ निकाय के रूप में: $\begin{cases} y = 1.5x + 3 \\ y = 2x \end{cases}$ लिखिए और हल कीजिए, और हल की
        व्याख्या कीजिए।
  \item समापन, शब्दकोश: एक \omterm{def:g10:algebra:equation}{समीकरण} $\leftrightarrow$ एक रेखा; एक निकाय $\leftrightarrow$ तीन
        नियतियों वाली दो रेखाएँ, जिनमें $ad - bc$ निर्णय करता है; अनेक असमिकाएँ
        $\leftrightarrow$ एक ऐसा क्षेत्र जिसके कोने इष्टतम उठाए रहते हैं। इसे तीन
        वाक्यों में लिखिए --- आपने अभी \emph{रैखिक बीजगणित} और \emph{रैखिक
        प्रोग्रामन} दोनों के प्रवेश-कक्ष का चक्कर लगाया है, और दोनों पूरे
        के पूरे स्नातक खंडों में बनते हैं।
\end{enumerate}
\end{problem}
