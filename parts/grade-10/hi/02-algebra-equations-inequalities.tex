\chapter{बीजगणित: समीकरण और असमिकाएँ}\label{ch:g10:algebra}

बीजगणित अक्षरों के साथ गणना करने की कला है। यह अध्याय दो बुनियादी चालों ---
\omterm{def:g10:algebra:expand}{प्रसार} और \omterm{def:g10:algebra:expand}{गुणनखंडन} --- को दोहराता और पैना करता है, और उनका उपयोग \omterm{def:g10:algebra:equation}{समीकरणों} तथा
असमिकाओं को व्यवस्थित ढंग से हल करने में करता है, जिसमें \omterm{def:g10:algebra:signtable}{चिह्न-सारणी} केंद्रीय
औज़ार है।

\section{प्रसार और गुणनखंडन}

\begin{definition}[प्रसार, गुणनखंडन]\label{def:g10:algebra:expand}
किसी व्यंजक का \emph{प्रसार}\index{प्रसार} करने का अर्थ है गुणनफलों को योगों
में बदलना; \emph{गुणनखंडन}\index{गुणनखंडन} इसका उल्टा है, अर्थात् योग को
गुणनफल में बदलना। बुनियादी नियम वितरण का नियम है:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g10:algebra:expand}
$(2x + 3)(x - 5)$ का चरण-दर-चरण \omterm{def:g10:algebra:expand}{प्रसार} कीजिए:
\begin{align*}
(2x + 3)(x - 5)
&= 2x \times x + 2x \times (-5) + 3 \times x + 3 \times (-5) \\
&= 2x^2 - 10x + 3x - 15 \\
&= 2x^2 - 7x - 15 .
\end{align*}
\end{example}

\begin{theorem}[उल्लेखनीय सर्वसमिकाएँ]\label{thm:g10:algebra:identities}
सभी \omterm{def:g10:numbers:sets}{वास्तविक संख्याओं} $a$ और $b$ के लिए:
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
हर एक सीधे \omterm{def:g10:algebra:expand}{प्रसार} से मिलती है। पहली के लिए:
\[
(a+b)^2 = (a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2 .
\]
दूसरी $b$ के स्थान पर $-b$ रखने से मिलती है: $(a-b)^2 = a^2 + 2a(-b) +
(-b)^2 = a^2 - 2ab + b^2$। तीसरी के लिए:
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$।
\end{proof}

\begin{example}\label{ex:g10:algebra:identities}
आगे की ओर (\omterm{def:g10:algebra:expand}{प्रसार}) और पीछे की ओर (\omterm{def:g10:algebra:expand}{गुणनखंडन}), दोनों तरह से प्रयोग:
\begin{align*}
(3x + 2)^2 &= 9x^2 + 12x + 4,
& x^2 - 25 &= (x+5)(x-5), \\
(x - 4)^2 &= x^2 - 8x + 16,
& 4x^2 + 4x + 1 &= (2x + 1)^2 .
\end{align*}
मौखिक गणना को भी लाभ होता है: $101 \times 99 = (100+1)(100-1) =
10000 - 1 = 9999$।
\end{example}

\begin{method}[किसी व्यंजक का गुणनखंडन]\label{met:g10:algebra:factor}
इसी क्रम में प्रयास कीजिए:
\begin{enumerate}
  \item \emph{उभयनिष्ठ गुणनखंड:} ऐसा गुणनखंड पहचानिए जो हर पद में मौजूद हो
        और उसे बाहर निकाल लीजिए, जैसे $6x^2 + 4x = 2x(3x + 2)$। उभयनिष्ठ गुणनखंड एक पूरा
        कोष्ठक भी हो सकता है: $(x+1)(2x-3) + (x+1)(x+7) = (x+1)(3x+4)$।
  \item \emph{उल्लेखनीय सर्वसमिका:} $a^2 - b^2$ या $a^2 \pm 2ab + b^2$ को पहचानिए, जैसे $9x^2 - 16 = (3x)^2 - 4^2 =
        (3x+4)(3x-4)$।
  \item दोनों को मिलाइए, और परिणाम का \omterm{def:g10:algebra:expand}{प्रसार} करके उसकी जाँच कीजिए।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:factorsteps}
$E = (2x + 1)^2 - (x - 3)^2$ का \omterm{def:g10:algebra:expand}{गुणनखंडन} कीजिए। यह दो वर्गों का अंतर $a^2 - b^2$ है, जिसमें $a = 2x+1$ और
$b = x-3$ हैं:
\begin{align*}
E &= \bigl[(2x+1) + (x-3)\bigr] \times \bigl[(2x+1) - (x-3)\bigr] \\
  &= (3x - 2)(2x + 1 - x + 3) \\
  &= (3x - 2)(x + 4).
\end{align*}
दूसरे कोष्ठक पर ध्यान दीजिए: $x - 3$ घटाने का अर्थ है $x$ घटाना \emph{और}
$3$ जोड़ना।
\end{example}

\section{समीकरण}

\begin{definition}[समीकरण, हल]\label{def:g10:algebra:equation}
\emph{समीकरण}\index{समीकरण} ऐसी समता है जिसमें एक अज्ञात संख्या $x$ हो;
\emph{हल} $x$ का वह मान है जो समता को सत्य बना देता है। समीकरण हल करने का
अर्थ है उसके \emph{सभी} हल खोज निकालना।
\end{definition}

दो \omterm{def:g10:algebra:equation}{समीकरण} \emph{तुल्य} कहलाते हैं जब उनके हल ठीक एक जैसे हों। दोनों पक्षों
में एक ही संख्या जोड़ने से, या दोनों पक्षों को एक ही \emph{शून्येतर} संख्या
से गुणा करने से, तुल्य \omterm{def:g10:algebra:equation}{समीकरण} मिलता है।

\begin{example}[प्रथम घात समीकरण]\label{ex:g10:algebra:firstdegree}
$5x - 7 = 2x + 8$ को एक-एक चाल करके हल कीजिए:
\begin{align*}
5x - 7 &= 2x + 8 \\
5x - 2x &= 8 + 7 && \text{(दोनों पक्षों में $7 - 2x$ जोड़ें)} \\
3x &= 15 \\
x &= 5 && \text{(दोनों पक्षों को $3$ से भाग दें)।}
\end{align*}
अकेला हल $5$ है। जाँच: $5 \times 5 - 7 = 18$ और $2 \times 5 + 8 = 18$।
\end{example}

\begin{theorem}[शून्य-गुणनफल नियम]\label{thm:g10:algebra:zeroproduct}
\omterm{def:g10:numbers:sets}{वास्तविक संख्याओं} का गुणनफल शून्य होता है यदि और केवल यदि उसका कम से कम एक
गुणनखंड शून्य हो:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ या } \ B = 0 .
\]
\end{theorem}

\begin{proof}
यदि $A = 0$ या $B = 0$ हो, तो गुणनफल स्पष्टतः $0$ है। इसके उलट, मान लीजिए $AB = 0$
और $A \neq 0$ है। दोनों पक्षों को $A$ से भाग देने पर $B = 0$ मिलता है।
\end{proof}

\begin{method}[गुणनखंडन द्वारा हल करना]\label{met:g10:algebra:zeroproduct}
ऐसा \omterm{def:g10:algebra:equation}{समीकरण} हल करने के लिए जिसका दाहिना पक्ष $0$ हो और जिसके बाएँ पक्ष का
\omterm{def:g10:algebra:expand}{गुणनखंडन} किया जा सके:
\begin{enumerate}
  \item सब कुछ बाईं ओर ले आइए ताकि \omterm{def:g10:algebra:equation}{समीकरण} $E = 0$ के रूप में आ जाए;
  \item $E$ का प्रथम घात गुणनखंडों के गुणनफल में \omterm{def:g10:algebra:expand}{गुणनखंडन} कीजिए;
  \item शून्य-गुणनफल नियम लगाइए और हर छोटे \omterm{def:g10:algebra:equation}{समीकरण} को हल कीजिए;
  \item मिले हुए सभी हलों को इकट्ठा कीजिए।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:zeroproduct}
$(x - 2)(3x + 6) = 0$ हल कीजिए: या तो $x - 2 = 0$, जिससे $x = 2$ मिलता है, या $3x + 6 = 0$, जिससे $x = -2$
मिलता है। हल: $-2$ और $2$।

$x^2 = 9x$ हल कीजिए। $x$ से भाग \emph{मत} दीजिए (वह शून्य हो सकता है!); सब कुछ
एक ओर लाइए और \omterm{def:g10:algebra:expand}{गुणनखंडन} कीजिए:
\[
x^2 - 9x = 0
\quad\Longleftrightarrow\quad
x(x - 9) = 0
\quad\Longleftrightarrow\quad
x = 0 \ \text{ या } \ x = 9 .
\]
\end{example}

\begin{example}[भागफल वाला समीकरण]\label{ex:g10:algebra:quotient}
$\dfrac{x - 1}{x + 2} = 0$ हल कीजिए। भागफल ठीक तब शून्य होता है जब उसका अंश शून्य हो \emph{और}
हर शून्य न हो। यहाँ $x - 1 = 0$ से $x = 1$ मिलता है, और $1 + 2 = 3 \neq 0$ है: एकमात्र हल $1$ है।
मान $x = -2$ \emph{वर्जित} है (शून्य से भाग) और उसे हमेशा पहले ही हटा देना
चाहिए।
\end{example}

\section{असमिकाएँ और चिह्न-सारणियाँ}

\begin{proposition}[असमिकाओं के नियम]\label{prop:g10:algebra:ineqrules}
मान लीजिए $a \leq b$ है।
\begin{enumerate}
  \item किसी भी $c$ के लिए: $a + c \leq b + c$ (जोड़ने से क्रम बना रहता है)।
  \item यदि $c > 0$ हो: $ac \leq bc$ (धनात्मक संख्या से गुणा करने पर क्रम बना रहता
        है)।
  \item यदि $c < 0$ हो: $ac \geq bc$ (ऋणात्मक संख्या से गुणा करने पर क्रम
        \emph{पलट} जाता है)।
\end{enumerate}
\end{proposition}

\begin{proof}
1. $(b + c) - (a + c) = b - a \geq 0$।
2. $bc - ac = (b-a)c$ दो अऋणात्मक संख्याओं का गुणनफल है, इसलिए $bc \geq ac$।
3. अब $(b-a)c \leq 0$ एक अऋणात्मक और एक ऋणात्मक संख्या का गुणनफल है, इसलिए $bc \leq ac$।
\end{proof}

\begin{example}\label{ex:g10:algebra:firstineq}
$-2x + 3 > 9$ हल कीजिए:
\begin{align*}
-2x + 3 &> 9 \\
-2x &> 6 && \text{($3$ घटाएँ)} \\
x &< -3 && \text{($-2 < 0$ से भाग दें: असमिका का चिह्न पलटें)।}
\end{align*}
हल-समुच्चय $\intoo{-\infty}{-3}$ है।
\end{example}

\begin{definition}[चिह्न-सारणी]\label{def:g10:algebra:signtable}
\emph{चिह्न-सारणी}\index{चिह्न-सारणी} में, हर गुणनखंड के लिए एक पंक्ति
लेकर, यह दर्ज किया जाता है कि जब $x$, $\R$ पर चलता है तब व्यंजक के हर
गुणनखंड का चिह्न ($+$ या $-$) क्या है; जिस मान पर गुणनखंड लुप्त होता है
वहाँ $0$ लिखा जाता है। अंतिम पंक्ति गुणनफल का चिह्न देती है, जो स्तंभ-दर-स्तंभ
चिह्नों के नियम से निकलता है।
\end{definition}

\begin{example}\label{ex:g10:algebra:signtable}
$P(x) = (x - 2)(3 - x)$ का चिह्न। गुणनखंड $x - 2$ $2$ पर लुप्त होता है और उसके बाद धनात्मक
रहता है; गुणनखंड $3 - x$ $3$ पर लुप्त होता है और उससे पहले धनात्मक रहता है:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccc}
$x$ & $-\infty$ \quad & $2$ & & $3$ & \quad $+\infty$ \\
\hline
$x - 2$ & $-$ & $0$ & $+$ & & $+$ \\
$3 - x$ & $+$ & & $+$ & $0$ & $-$ \\
\hline
$P(x)$ & $-$ & $0$ & $+$ & $0$ & $-$ \\
\end{tabular}
\end{center}
अतः $P(x) > 0$ ठीक $\intoo{2}{3}$ पर, और $P(x) \leq 0$ $\intoc{-\infty}{2} \cup \intco{3}{+\infty}$ पर।
\end{example}

\begin{method}[चिह्न-सारणी से असमिका हल करना]\label{met:g10:algebra:signtable}
$E(x) \geq 0$ या $E(x) < 0$ जैसी असमिका हल करने के लिए:
\begin{enumerate}
  \item सब कुछ बाएँ पक्ष में ले आइए, ताकि दाहिना पक्ष $0$ हो जाए (दोनों
        पक्षों को ऐसे व्यंजक से कभी गुणा न कीजिए जिसका चिह्न अज्ञात हो);
  \item बाएँ पक्ष का \omterm{def:g10:algebra:expand}{गुणनखंडन} कीजिए, या यदि वह भागफल है तो उसे उभयनिष्ठ हर
        पर लाइए;
  \item हर गुणनखंड के लिए एक पंक्ति लेकर \omterm{def:g10:algebra:signtable}{चिह्न-सारणी} बनाइए (भागफल के लिए
        अंतिम पंक्ति में वर्जित मानों को दोहरी खड़ी रेखा से अंकित कीजिए);
  \item वे \omterm{def:g10:numbers:interval}{अंतराल} पढ़ लीजिए जहाँ माँगा गया चिह्न मिलता है, और हर सीमा के
        लिए यह जाँचिए कि वह सम्मिलित है या नहीं।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:quotientineq}
$\dfrac{x + 1}{x - 3} \leq 0$ हल कीजिए। अंश $-1$ पर लुप्त होता है और हर $3$ पर (वर्जित मान)।
भागफल तब ऋणात्मक होता है जब दोनों के चिह्न विपरीत हों, और यह $x$ के
$-1$ और $3$ के बीच होने पर होता है। जिस सीमा पर \emph{अंश} लुप्त होता
है उसे सम्मिलित करके और वर्जित मान को हटाकर: हल-समुच्चय $\intco{-1}{3}$ है।
\end{example}

शून्य-गुणनफल नियम का उपयोग करके हल कीजिए:
\[
(x + 5)(2x - 8) = 0, \qquad
x^2 - 16 = 0, \qquad
x^2 + 6x + 9 = 0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:algebra:5}
असमिकाएँ हल कीजिए और हल-समुच्चयों को \omterm{def:g10:numbers:interval}{अंतरालों} के रूप में लिखिए:
\[
3x - 5 \leq 7, \qquad
-4x + 1 > 9, \qquad
2(x - 1) \geq 5x + 4 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:6}
$E = (x - 1)(x + 3) - (x - 1)(2x - 5)$ का \omterm{def:g10:algebra:expand}{गुणनखंडन} कीजिए, फिर $E = 0$ हल कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:7}
$x^2 = 5x$ हल कीजिए, फिर $(2x + 1)^2 = (x - 4)^2$। (दूसरे के लिए सब कुछ बाईं ओर लाइए और वर्गों के
अंतर का \omterm{def:g10:algebra:expand}{गुणनखंडन} कीजिए।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:8}
\omterm{def:g10:algebra:signtable}{चिह्न-सारणी} का उपयोग करके हल कीजिए:
\[
(x + 2)(4 - x) > 0,
\qquad
\frac{2x - 6}{x + 1} \geq 0 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:9}
एक स्ट्रीमिंग सेवा का शुल्क $9$ प्रति माह है, और उसके साथ $15$ का
एकमुश्त पंजीकरण शुल्क भी लगता है; एक प्रतिद्वंद्वी सेवा का शुल्क $12$
प्रति माह है और कोई पंजीकरण शुल्क नहीं। कितने महीनों के बाद पहली सेवा
कुल मिलाकर सस्ती पड़ने लगती है? असमिका बनाइए और हल कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:10}
$\dfrac{x+3}{x-1} = 2$ हल कीजिए। (वर्जित मान हटाने के बाद दोनों पक्षों को $x - 1$ से गुणा
कीजिए, या सब कुछ उभयनिष्ठ हर पर लेकर बाईं ओर ले आइए।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:algebra:11}
मान लीजिए $n$ एक \omterm{def:g10:numbers:sets}{पूर्णांक} है। दिखाइए कि $(n + 1)^2 - n^2$ सदा विषम होता है, और यह भी
कि दो क्रमागत विषम संख्याओं के वर्गों का अंतर सदा $8$ का गुणज होता है।
\end{exercise}

\section{समस्या: बेबीलोन से स्वर्ण अनुपात तक}

\begin{problem}[{सप्ताहांत समस्या --- योग-और-गुणनफल की समस्याएँ, अल-ख़्वारिज़्मी
के खींचे चित्र के अनुसार वर्ग पूरा करना, और समीकरण $x^2 = x + 1$}]
\label{pb:g10:algebra:1}
चार हज़ार साल पहले बेबीलोन के लिपिक मिट्टी की पट्टियों पर यही हल करते थे:
\emph{दो संख्याओं का योग और गुणनफल जानते हुए उन्हें ढूँढ़ना}। उनकी तरकीब ---
संख्याओं को आधे योग के दोनों ओर सममित रूप से खोजना --- आज भी हर द्विघात
\omterm{def:g10:algebra:equation}{समीकरण} को तोड़ देती है, अपने आधुनिक नाम \emph{वर्ग पूरा करना} के साथ। यह
समस्या पहले वह तरकीब सीखती है, फिर अल-ख़्वारिज़्मी को सचमुच एक वर्ग पूरा
करते देखती है, और अंत में उस विधि को सबसे प्रसिद्ध अनुपात पर काम में लगाती
है।

\medskip
\noindent\textbf{भाग I --- बेबीलोनी समस्या।}
\begin{enumerate}
  \item तैयारी (\cref{thm:g10:algebra:identities}): $x^2 - 10x + 25$ का \omterm{def:g10:algebra:expand}{गुणनखंडन} कीजिए, फिर $9x^2 - 49$ का, और फिर ---
        एक सर्वसमिका को वर्गों के अंतर के साथ मिलाकर --- $x^2 + 6x + 9 - 4$ का।
  \item $(x + 1)(x + 5) = 0$ और $(3x - 7)(3x + 7) = 0$ हल कीजिए (\cref{thm:g10:algebra:zeroproduct})।
  \item लिपिक की समस्या: दो संख्याओं का योग $10$ और गुणनफल $21$ है।
        बेबीलोन का अनुसरण करते हुए उन्हें $5 - t$ और $5 + t$ लिखिए (आधे योग के
        दोनों ओर सममित) और $t$ ज्ञात कीजिए, फिर दोनों संख्याएँ।
  \item वही विधि: योग $14$, गुणनफल $33$।
  \item सामान्य नुस्ख़ा: योग $s$ और गुणनफल $p$ के लिए दिखाइए कि दोनों
        संख्याएँ
        \[
        \frac s2 - t
        \quad\text{और}\quad
        \frac s2 + t,
        \qquad\text{जहाँ}\quad
        t^2 = \left(\frac s2\right)^2 - p ,
        \]
        हैं, और $s$ तथा $p$ पर वह शर्त बताइए जिससे समस्या का हल हो।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- वर्ग पूरा करना।}
\begin{enumerate}[resume]
  \item $(x - a)(x - b)$ का \omterm{def:g10:algebra:expand}{प्रसार} कीजिए और निष्कर्ष निकालिए: $x^2 - sx + p = 0$ के हल ठीक वही
        संख्या-युग्म हैं जिनका योग $s$ और गुणनफल $p$ है। (लिपिक बिना
        जाने द्विघात \omterm{def:g10:algebra:equation}{समीकरण} ही हल कर रहे थे।)
  \item जाँचिए कि $x^2 - 10x + 21 = (x - 5)^2 - 4$, और $x^2 - 10x + 21 = 0$ को दाहिने पक्ष का वर्गों के अंतर के रूप
        में \omterm{def:g10:algebra:expand}{गुणनखंडन} करके हल कीजिए। प्रश्न 3 से तुलना कीजिए।
  \item $x^2 + 6x - 7 = 0$ को वर्ग पूरा करके हल कीजिए।
  \item $x^2 + 4x + 5$ में वर्ग पूरा कीजिए, और समझाइए कि \omterm{def:g10:algebra:equation}{समीकरण} $x^2 + 4x + 5 = 0$ का कोई
        वास्तविक हल क्यों \emph{नहीं} है। सामान्य कसौटी बताइए: $(x + h)^2 + k$ के
        रूप में लिख लेने पर कोई द्विघात व्यंजक कब लुप्त होता है?
  \item नाम अक्षरशः सही है। नवीं सदी के बग़दाद में अल-ख़्वारिज़्मी ने $x^2 + 6x = 7$
        को भुजा $x$ वाले एक वर्ग के रूप में खींचा जिसकी दो भुजाओं से दो
        $3 \times x$ आयत चिपके थे, और छूटे हुए $3 \times 3$ कोने को जोड़कर उन्होंने
        \emph{वर्ग पूरा किया}। यह चित्र खींचिए और समझाइए कि यह \omterm{def:g10:algebra:equation}{समीकरण} को
        $(x + 3)^2 = 16$ में कैसे बदल देता है --- फिर हल कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- चिह्न, सारणियाँ और दो पुराने वादे।}
\begin{enumerate}[resume]
  \item $(x - 3)(x + 2)$ की \omterm{def:g10:algebra:signtable}{चिह्न-सारणी} बनाइए (\cref{def:g10:algebra:signtable}) और $(x - 3)(x + 2) < 0$ हल कीजिए।
  \item $x^2 \leq 10x - 21$ हल कीजिए (सब कुछ एक ओर लाइए और प्रश्न 7 फिर से काम में लाइए)।
  \item $\dfrac{x - 1}{x + 2} \geq 0$ को \omterm{def:g10:algebra:signtable}{चिह्न-सारणी} से हल कीजिए, वर्जित मान का ध्यान रखते हुए।
  \item एक पुराना वादा निभा: माध्यमिक विद्यालय खंड की रानी डिडो की बाड़
        वाली समस्या के किसान के पास $20$ m बाड़ है, और $x \times (10 - x)$ बाड़े का
        क्षेत्रफल
        \[
        A(x) = x(10 - x) = 25 - (x - 5)^2 .
        \]
        को संतुष्ट करता है। सर्वसमिका की जाँच कीजिए, और उससे उस बात की
        \emph{उपपत्ति} पढ़ लीजिए जिसका कक्षा 6 केवल अनुमान लगा सकी थी:
        अधिकतम क्षेत्रफल, और उसे प्राप्त करने वाला अकेला $x$।
  \item एक और: सिद्ध कीजिए कि हर $x > 0$ के लिए $x + \dfrac1x \geq 2$ है, और समता केवल $x = 1$
        पर होती है। ($x$ से गुणा कीजिए, पद इकट्ठे कीजिए, एक वर्ग पहचानिए
        --- माध्यमिक विद्यालय खंड की यूक्लिड-संबंधों वाली सप्ताहांत समस्या
        की समांतर--गुणोत्तर माध्य असमिका ही नए लिबास में।)
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- स्वर्ण विभाजन।}
लंबाई $1$ के एक रेखाखंड को एक बड़े भाग $x$ और एक छोटे भाग $1 - x$ में
इस तरह काटिए कि \emph{पूरा बड़े से उसी अनुपात में हो जिसमें बड़ा छोटे से
है}: $\frac{1}{x} = \frac{x}{1 - x}$।
\begin{enumerate}[resume]
  \item दिखाइए कि यह शर्त $x^2 + x - 1 = 0$ बन जाती है।
  \item वर्ग पूरा करके हल कीजिए, और वह मूल रखिए जो लंबाई हो सकता है: $x = \frac{\sqrt5 - 1}{2} \approx 0.618$।
  \item \emph{स्वर्ण अनुपात} $\varphi = \frac1x = \frac{2}{\sqrt5 - 1}$ है। हर का परिमेयकरण कीजिए (ऊपर और नीचे
        दोनों को $\sqrt5 + 1$ से गुणा कीजिए --- वही सर्वसमिका काम पर) और दिखाइए
        कि $\varphi = \frac{\sqrt5 + 1}{2} \approx 1.618$।
  \item $\varphi$ की दो जादुई सर्वसमिकाओं की \emph{यथार्थ} जाँच कीजिए:
        \[
        \varphi^2 = \varphi + 1,
        \qquad
        \frac{1}{\varphi} = \varphi - 1 .
        \]
  \item समापन: समझाइए कि $\varphi$ \omterm{ex:g10:numbers:classify}{अपरिमेय} क्यों है (माध्यमिक विद्यालय खंड की
        अपरिमेयता वाली सप्ताहांत समस्या $\sqrt5$ के बारे में मुख्य तथ्य देती
        है), फिर क्रमागत हेमचंद्र--फिबोनाच्ची संख्याओं (माध्यमिक विद्यालय
        खंड की लय-गणना वाली सप्ताहांत समस्या) के अनुपात $\frac53$, $\frac85$,
        $\frac{13}{8}$, $\frac{21}{13}$ को तीन दशमलव स्थानों तक निकालिए। वे किसका पीछा करते
        जान पड़ते हैं? (वे उसे पकड़ भी लेते हैं, इसकी उपपत्ति अनुक्रमों
        वाले अध्यायों की है।)
\end{enumerate}
\end{problem}
