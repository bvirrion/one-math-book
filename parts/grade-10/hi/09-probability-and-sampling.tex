\chapter{प्रायिकता और प्रतिदर्शन}\label{ch:g10:proba}

\omterm{def:g10:proba:distribution}{प्रायिकता} संयोग को संख्याएँ देती है। समान रूप से संभावित परिणामों वाले
प्रयोगों --- पासे, सिक्के, ताश --- से शुरू करके यह अध्याय \omterm{def:g10:proba:events}{घटनाओं} की
\omterm{def:g10:proba:distribution}{प्रायिकता} निकालने के बुनियादी नियम खड़े करता है, दो-चरणीय प्रयोगों के लिए
\omterm{met:g10:proba:tree}{वृक्ष आरेख} प्रस्तुत करता है, और \omterm{def:g10:proba:distribution}{प्रायिकताओं} तथा प्रेक्षित आपेक्षिक
बारंबारताओं के बीच के संबंध पर समाप्त होता है। सिद्धांत \cref{ch:g11:prob} में आगे बढ़ता
है।

\section{यादृच्छिक प्रयोग और घटनाएँ}

\begin{definition}[परिणाम और घटनाएँ]\label{def:g10:proba:events}
\emph{यादृच्छिक प्रयोग}\index{यादृच्छिक प्रयोग} वह प्रयोग है जिसके परिणाम
की भविष्यवाणी नहीं की जा सकती (जैसे पासा फेंकना)। सभी संभव परिणामों का
समुच्चय \emph{प्रतिदर्श समष्टि}\index{प्रतिदर्श समष्टि} $\Omega = \{\omega_1, \dots, \omega_n\}$ है; उसके अवयव
\emph{परिणाम} कहलाते हैं। \emph{घटना}\index{घटना} परिणामों का एक समुच्चय
है; कोई घटना \emph{घटित} तब होती है जब प्रयोग का परिणाम उसमें आता हो।
\end{definition}

\begin{definition}[प्रायिकता बंटन]\label{def:g10:proba:distribution}
हर परिणाम $\omega_i$ को एक संख्या $p_i \geq 0$ देना, जहाँ $p_1 + \dots + p_n = 1$ हो,
\emph{प्रायिकता}\index{प्रायिकता} को परिभाषित करता है: \omterm{def:g10:proba:events}{घटना} $A$ की
प्रायिकता $\P(A)$ $A$ के परिणामों की $p_i$ का योग है। विशेष रूप से $\P(\Omega) = 1$, और
असंभव \omterm{def:g10:proba:events}{घटना} $\varnothing$ की प्रायिकता $0$ है।
\end{definition}

\begin{proposition}[समसंभावी परिणाम]\label{prop:g10:proba:uniform}
यदि $n$ परिणाम समान रूप से संभावित हों (हर एक $p_i = \frac1n$), तो हर \omterm{def:g10:proba:events}{घटना} $A$ के
लिए:
\[
\P(A) = \frac{A \text{ में परिणामों की संख्या}}{n}
= \frac{\text{अनुकूल स्थितियाँ}}{\text{संभव स्थितियाँ}} .
\]
\end{proposition}

\begin{proof}
$A$ में कुछ $k$ परिणाम हैं, जिनमें से हर एक की \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac1n$ है, इसलिए
$\P(A) = \frac kn$।
\end{proof}

\begin{example}\label{ex:g10:proba:die}
एक न्यायसंगत पासा फेंकिए: $\Omega = \{1, 2, 3, 4, 5, 6\}$, और हर परिणाम की \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac16$ है। \omterm{def:g10:proba:events}{घटना}
$A$: ``परिणाम सम है'' $A = \{2, 4, 6\}$ है, इसलिए $\P(A) = \frac36 = \frac12$। \omterm{def:g10:proba:events}{घटना} $B$: ``परिणाम कम से कम
$5$ है'' $B = \{5, 6\}$ है, इसलिए $\P(B) = \frac26 =
\frac13$।
\end{example}

\section{घटनाओं को जोड़ना}

\begin{definition}[प्रतिच्छेदन, सम्मिलन, पूरक]\label{def:g10:proba:operations}
मान लीजिए $A$ और $B$ \omterm{def:g10:proba:events}{घटनाएँ} हैं।
\begin{itemize}
  \item $A \cap B$ (``$A$ और $B$'') तब घटित होती है जब दोनों घटित हों;
  \item $A \cup B$ (``$A$ या $B$'') तब घटित होती है जब कम से कम एक घटित हो;
  \item \emph{पूरक}\index{पूरक} $\bar A$ (``$A$ नहीं'') ठीक तब घटित होती है
        जब $A$ घटित न हो;
  \item $A$ और $B$ \emph{असंगत}\index{असंगत} तब हैं जब वे एक साथ घटित
        नहीं हो सकतीं: $A \cap B = \varnothing$।
\end{itemize}
\end{definition}

\begin{theorem}[योग के नियम]\label{thm:g10:proba:addition}
सभी \omterm{def:g10:proba:events}{घटनाओं} $A$ और $B$ के लिए:
\[
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B),
\qquad
\P(\bar A) = 1 - \P(A) .
\]
\omterm{def:g10:proba:operations}{असंगत} \omterm{def:g10:proba:events}{घटनाओं} के लिए पहला नियम $\P(A \cup B) = \P(A) + \P(B)$ बन जाता है।
\end{theorem}

\begin{proof}
$\P(A) + \P(B)$ जोड़ने पर $A \cup B$ के हर परिणाम की \omterm{def:g10:proba:distribution}{प्रायिकता} एक बार जुड़ती है,
\emph{सिवाय} $A \cap B$ के परिणामों के, जो दो बार गिने जाते हैं --- एक बार $A$
में और एक बार $B$ में। $\P(A \cap B)$ घटाने से यह दोहरी गिनती सुधर जाती है। \omterm{def:g10:proba:operations}{पूरक} के
लिए: $A$ और $\bar A$ \omterm{def:g10:proba:operations}{असंगत} हैं और $A \cup \bar A = \Omega$, इसलिए $\P(A) + \P(\bar A) = \P(\Omega) = 1$।
\end{proof}

  \node[omExo, font=\small] at (3.0,1.45) {$\Omega$};

जाता है, और यही योग के नियम में घटाव की व्याख्या करता है।}
\end{omfigure}

\begin{example}\label{ex:g10:proba:cards}
$52$ पत्तों की एक मानक गड्डी से एक पत्ता निकालिए। मान लीजिए $A$: ``पत्ता
पान का है'' ($13$ पत्ते) और $B$: ``पत्ता बादशाह है'' ($4$ पत्ते)। तब $A \cap B$
``पान का बादशाह'' है ($1$ पत्ता), और
\[
\P(A \cup B) = \frac{13}{52} + \frac{4}{52} - \frac{1}{52} = \frac{16}{52}
= \frac{4}{13}.
\]
\omterm{def:g10:proba:operations}{पूरक} का नियम प्रायः सबसे तेज़ रास्ता होता है: पत्ता पान का \emph{न} होने की
\omterm{def:g10:proba:distribution}{प्रायिकता} $1 - \frac{13}{52} = \frac34$ है।
\end{example}

\section{वृक्ष आरेख}

\begin{method}[दो-चरणीय प्रयोग]\label{met:g10:proba:tree}
दो चरणों में किए जाने वाले प्रयोग के लिए:
\begin{enumerate}
  \item एक \emph{वृक्ष}\index{वृक्ष आरेख} खींचिए: पहले चरण के हर संभव
        परिणाम के लिए एक शाखा, और फिर उनमें से हर एक से दूसरे चरण के हर
        परिणाम के लिए एक शाखा, और हर शाखा पर उसकी \omterm{def:g10:proba:distribution}{प्रायिकता} लिखिए (किसी भी
        गाँठ से निकलने वाली शाखाओं का योग $1$ होना चाहिए);
  \item किसी \emph{पथ} (\omterm{met:g10:proba:tree}{वृक्ष} के पत्ते) की \omterm{def:g10:proba:distribution}{प्रायिकता} उसकी शाखाओं की
        \omterm{def:g10:proba:distribution}{प्रायिकताओं} का गुणनफल है;
  \item किसी \omterm{def:g10:proba:events}{घटना} की \omterm{def:g10:proba:distribution}{प्रायिकता} उसे साकार करने वाले पथों की \omterm{def:g10:proba:distribution}{प्रायिकताओं} का
        योग है।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:proba:tree}
एक कलश में $3$ लाल और $2$ नीली गेंदें हैं। एक गेंद निकालिए, उसे वापस
रखिए, और फिर निकालिए। हर बार लाल (R) \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac35$ से और नीली (B)
\omterm{def:g10:proba:distribution}{प्रायिकता} $\frac25$ से मिलती है।

योग $1$ है।}

एक ही रंग की दो गेंदें मिलने की \omterm{def:g10:proba:distribution}{प्रायिकता} $\P(\text{RR}) + \P(\text{BB}) = \frac{9}{25} + \frac{4}{25} =
\frac{13}{25}$ है; कम से कम एक नीली मिलने
की \omterm{def:g10:proba:distribution}{प्रायिकता} $1 - \P(\text{RR}) = 1 - \frac{9}{25} = \frac{16}{25}$ है (\omterm{def:g10:proba:operations}{पूरक} का नियम)।
\end{example}

\begin{remark}
वापसी के बिना दूसरी बार निकालने की \omterm{def:g10:proba:distribution}{प्रायिकताएँ} पहली बार पर निर्भर करती हैं:
ऊपर वाले कलश में एक लाल गेंद निकल जाने के बाद केवल $2$ लाल और $2$ नीली
बचती हैं, इसलिए दूसरी शाखा पर $\frac24$ और $\frac24$ आते हैं। \omterm{met:g10:proba:tree}{वृक्ष} की विधि ज्यों की
त्यों काम करती है; केवल दूसरे स्तर की शाखाओं की संख्याएँ बदलती हैं।
\end{remark}

\section{आपेक्षिक बारंबारता और प्रतिदर्शन}

जब कोई \omterm{def:g10:proba:events}{यादृच्छिक प्रयोग} बहुत बार दोहराया जाता है, तब किसी \omterm{def:g10:proba:events}{घटना} की प्रेक्षित
\omterm{def:g10:stats:series}{आपेक्षिक बारंबारता} उसकी \omterm{def:g10:proba:distribution}{प्रायिकता} के पास पहुँच जाती है --- और इसीलिए
\omterm{def:g10:proba:distribution}{प्रायिकता} वास्तविक संसार का वर्णन करने में काम आती है।

\begin{definition}[प्रतिदर्श]\label{def:g10:proba:sample}
$n$ आकार का \emph{प्रतिदर्श}\index{प्रतिदर्श} एक ही प्रयोग को $n$ बार
स्वतंत्र रूप से दोहराने का परिणाम है। प्रतिदर्श में किसी \omterm{def:g10:proba:events}{घटना} की
\emph{प्रेक्षित \omterm{def:g10:stats:series}{आपेक्षिक बारंबारता}} उसके घटित होने की संख्या को $n$ से भाग
देकर मिलती है।
\end{definition}

\begin{example}\label{ex:g10:proba:sampling}
एक न्यायसंगत सिक्के से चित \omterm{def:g10:proba:distribution}{प्रायिकता} $0.5$ से आता है। उसे $100$ बार उछालने पर
ठीक $50$ चित विरले ही मिलते हैं: \omterm{def:g10:proba:sample}{प्रतिदर्श} \emph{उतार-चढ़ाव} करते हैं।
$100$ उछालों की प्रारूपिक शृंखलाएँ $0.46$, $0.53$, $0.49$ जैसी आपेक्षिक
बारंबारताएँ देती हैं --- $0.5$ के पास, पर बराबर नहीं। बड़े \omterm{def:g10:proba:sample}{प्रतिदर्श} कम
उतार-चढ़ाव करते हैं: $10\,000$ उछालों पर प्रेक्षित आपेक्षिक बारंबारताएँ प्रायः
$0.5$ के लगभग $0.01$ के भीतर रहती हैं।
\end{example}

\begin{remark}[उतार-चढ़ाव अंतराल]
एक उपयोगी मोटा नियम, जिसका औचित्य \cref{ch:g11:binom} में है: \omterm{def:g10:proba:distribution}{प्रायिकता} $p$ वाली \omterm{def:g10:proba:events}{घटना}
के $n$ आकार के \omterm{def:g10:proba:sample}{प्रतिदर्श} के लिए (जहाँ $n$ पर्याप्त बड़ा हो), प्रेक्षित
\omterm{def:g10:stats:series}{आपेक्षिक बारंबारता} लगभग $95\%$ \omterm{def:g10:proba:sample}{प्रतिदर्शों} में \omterm{def:g10:numbers:interval}{अंतराल}
\[
\intcc{p - \frac{1}{\sqrt n}}{\,p + \frac{1}{\sqrt n}}
\]
में रहती है। यदि कोई प्रेक्षित \omterm{def:g10:stats:series}{आपेक्षिक बारंबारता} इस \omterm{def:g10:numbers:interval}{अंतराल} से बहुत बाहर
गिरे, तो $p$ के मान पर संदेह कीजिए: यही सांख्यिकीय परीक्षण का आरंभ-बिंदु
है।
\end{remark}

\begin{example}\label{ex:g10:proba:test}
एक कारख़ाने का दावा है कि उसके केवल $10\%$ पुर्ज़े दोषपूर्ण हैं ($p = 0.1$)।
$n = 400$ पुर्ज़ों के एक \omterm{def:g10:proba:sample}{प्रतिदर्श} में $64$ दोषपूर्ण निकले: प्रेक्षित \omterm{def:g10:stats:series}{आपेक्षिक
बारंबारता} $\frac{64}{400} = 0.16$। उतार-चढ़ाव \omterm{def:g10:numbers:interval}{अंतराल} $\intcc{0.1 - \frac{1}{20}}{0.1 + \frac{1}{20}} = \intcc{0.05}{0.15}$ है, और $0.16$ उससे बाहर पड़ता है:
\omterm{def:g10:proba:sample}{प्रतिदर्श} उस दावे पर गंभीर संदेह खड़ा करता है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g10:proba:1}
एक न्यायसंगत पासा फेंका जाता है। हर \omterm{def:g10:proba:events}{घटना} की \omterm{def:g10:proba:distribution}{प्रायिकता} निकालिए: ``$6$
आना''; ``विषम संख्या आना''; ``अधिक से अधिक $4$ आना''; ``$7$ आना''।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:2}
एक थैले में $5$ हरे, $3$ पीले और $2$ काले कंचे हैं; एक कंचा यादृच्छिक
रूप से निकाला जाता है। निम्नलिखित \omterm{def:g10:proba:distribution}{प्रायिकताएँ} निकालिए: वह हरा है; वह काला नहीं
है; वह हरा या पीला है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:3}
$30$ विद्यार्थियों की कक्षा में $18$ स्पेनी पढ़ते हैं, $10$ जर्मन, और
$4$ दोनों। एक विद्यार्थी यादृच्छिक रूप से चुना जाता है। योग के नियम से
यह \omterm{def:g10:proba:distribution}{प्रायिकता} निकालिए कि वह दोनों भाषाओं में से कम से कम एक पढ़ता है, फिर यह
कि वह इनमें से कोई नहीं पढ़ता।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:4}
एक न्यायसंगत सिक्का दो बार उछाला जाता है। प्रयोग का \omterm{met:g10:proba:tree}{वृक्ष} खींचिए और ये
\omterm{def:g10:proba:distribution}{प्रायिकताएँ} निकालिए: दो चित आना; ठीक एक चित आना; कम से कम एक चित आना।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:5}
एक कलश में $4$ लाल और $6$ सफ़ेद गेंदें हैं। दो गेंदें एक के बाद एक,
\emph{बिना} वापसी के, निकाली जाती हैं। दूसरे स्तर पर सही \omterm{def:g10:proba:distribution}{प्रायिकताओं} के
साथ \omterm{met:g10:proba:tree}{वृक्ष} खींचिए, और दो लाल गेंदें निकलने की तथा फिर अलग-अलग रंगों की दो
गेंदें निकलने की \omterm{def:g10:proba:distribution}{प्रायिकता} निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:6}
दो न्यायसंगत पासे फेंके जाते हैं और उनका योग दर्ज किया जाता है।
\begin{enumerate}
  \item समझाइए कि \omterm{def:g10:proba:events}{प्रतिदर्श समष्टि} को $36$ समसंभावी युग्मों $(1,1), (1,2), \dots, (6,6)$ के रूप
        में क्यों लिया जा सकता है।
  \item यह \omterm{def:g10:proba:distribution}{प्रायिकता} निकालिए कि योग $7$ है; कि वह $12$ है; कि वह अधिक
        से अधिक $4$ है।
  \item कौन-सा योग सबसे अधिक संभावित है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:7}
एक बहुविकल्पीय प्रश्न में $4$ उत्तर दिए जाते हैं, जिनमें एक सही है। एक
विद्यार्थी ऐसे $2$ स्वतंत्र प्रश्नों के उत्तर पूरी तरह यादृच्छिक रूप से
देता है। दोनों सही होने की, ठीक एक सही होने की और एक भी सही न होने की
\omterm{def:g10:proba:distribution}{प्रायिकता} निकालिए। जाँचिए कि तीनों \omterm{def:g10:proba:distribution}{प्रायिकताओं} का योग $1$ है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:8}
एक नेता $50\%$ समर्थन का दावा करता है। यादृच्छिक रूप से चुने गए $n = 900$
मतदाताओं के सर्वेक्षण में $396$ समर्थन करते हैं।
\begin{enumerate}
  \item प्रेक्षित \omterm{def:g10:stats:series}{आपेक्षिक बारंबारता} और $n = 900$ के लिए $p = 0.5$ के आसपास का
        उतार-चढ़ाव \omterm{def:g10:numbers:interval}{अंतराल} निकालिए।
  \item क्या सर्वेक्षण उस दावे पर संदेह खड़ा करता है?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:proba:9}
एक खेल खेलने का शुल्क $2$ है: आप एक न्यायसंगत पासा फेंकते हैं और $6$
आने पर $6$ जीतते हैं, $5$ आने पर $3$, और अन्यथा कुछ नहीं।
\begin{enumerate}
  \item जीती जाने वाली हर राशि ($6$, $3$, $0$) की \omterm{def:g10:proba:distribution}{प्रायिकता} निकालिए।
  \item $600$ खेलों में आप हर परिणाम के लगभग कितनी बार आने की अपेक्षा करते
        हैं? कुल अपेक्षित लाभ और हानि निकालिए: क्या यह खेल खेलने लायक़ है?
\end{enumerate}
\end{exercise}

\section{समस्या: वे तीन दरवाज़े जिन्होंने हज़ार गणितज्ञों को छका दिया}

\begin{problem}[{सप्ताहांत समस्या --- मॉन्टी हॉल विरोधाभास और उसके भाई-बंधु:
सूचना प्रायिकता को कैसे बदल देती है, और प्रतिदर्श दावों को मनचाही सोच से कैसे
अलग करते हैं}]
\label{pb:g10:proba:1}
1990 में स्तंभकार मैरिलिन वॉस सैवंट ने एक खेल-कार्यक्रम के संचालक, तीन
दरवाज़ों, एक कार और दो बकरियों वाली पहेली छापी --- और उसका सही उत्तर दिया।
दस हज़ार पाठकों ने, जिनमें सैकड़ों गणित के शोध-उपाधिधारी थे, लिखकर ज़ोर
दिया कि वे ग़लत हैं। यह समस्या आपको \omterm{met:g10:proba:tree}{वृक्ष} (\cref{met:g10:proba:tree}), \omterm{def:g10:proba:operations}{पूरक} और योग के नियमों
(\cref{thm:g10:proba:addition}) से लैस करती है, विवाद स्वयं सुलझाने देती है, और उस औज़ार पर
समाप्त होती है जो व्यवहार में ऐसे हर झगड़े का पंच है: उतार-चढ़ाव करता
\omterm{def:g10:proba:sample}{प्रतिदर्श}।

\medskip
\noindent\textbf{भाग I --- नियम, गरमाहट के साथ।}
\begin{enumerate}
  \item $52$ पत्तों की मानक गड्डी से एक पत्ता: योग के नियम से $P(\text{बादशाह या पान})$
        निकालिए --- और बताइए कि अकेला $\frac{4}{52} + \frac{13}{52}$ ग़लत क्यों है।
  \item एक न्यायसंगत सिक्के के तीन उछाल: \omterm{def:g10:proba:operations}{पूरक} से $P(\text{कम से कम एक
        चित})$ निकालिए।
  \item एक थैले में $3$ लाल और $2$ नीले कंचे हैं; दो बिना वापसी के
        निकाले जाते हैं। \omterm{met:g10:proba:tree}{वृक्ष} से $P(\text{दो लाल})$ और $P(\text{प्रत्येक रंग का एक})$ निकालिए।
  \item प्रश्न 3 के बंटन को $P(\text{दो नीली})$ से पूरा कीजिए और जाँचिए कि तीनों
        \omterm{def:g10:proba:distribution}{प्रायिकताओं} का योग $1$ है।
  \item कोई व्यक्ति दो \emph{\omterm{def:g10:proba:operations}{असंगत}} \omterm{def:g10:proba:events}{घटनाओं} के लिए $P(A) = 0.6$ और $P(B) = 0.5$ बताता है।
        अंकगणित की एक पंक्ति से उसे दोषी सिद्ध कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- तीन दरवाज़े।}
खेल यह है: तीन दरवाज़ों में से एक के पीछे कार छिपी है और बाक़ी दो के पीछे
बकरियाँ। आप एक दरवाज़ा चुनते हैं (मान लीजिए दरवाज़ा 1)। संचालक --- जो
\emph{जानता} है कि कार कहाँ है --- बाक़ी दो में से एक दरवाज़ा खोलता है और
हमेशा बकरी दिखाता है, और फिर आपको बचे हुए बंद दरवाज़े पर बदल जाने का अवसर
देता है।
\begin{enumerate}[resume]
  \item कुछ भी निकालने से पहले: बदलने से लाभ होता है, हानि होती है, या कोई
        अंतर नहीं पड़ता? अपना पहला अनुमान लिख लीजिए --- ईमानदारी से।
  \item अब कार की सच्ची स्थिति पर स्थितियाँ बाँटकर \omterm{met:g10:proba:tree}{वृक्ष} बनाइए (हर एक
        $\frac13$): यदि कार आपके ही दरवाज़े के पीछे हो, तो बदलने पर हार;
        यदि वह बाक़ी दोनों में से किसी के पीछे हो, तो संचालक को क्या करना
        ही पड़ेगा, और तब बदलने पर क्या जीत होती है? $P(\text{बदलकर जीत})$ निकालिए।
  \item एक-पंक्ति वाला रूप: बदलने पर जीत ठीक तब होती है जब आपका पहला चुनाव
        ग़लत था। फिर से निष्कर्ष निकालिए।
  \item अब निर्णायक रूपांतर --- ``मॉन्टी गिर पड़ा'': संचालक \emph{नहीं}
        जानता कि कार कहाँ है, बाक़ी दरवाज़ों में से एक यादृच्छिक रूप से
        खोलता है, और संयोग से बकरी निकल आती है। छह समसंभावी परिदृश्य
        (कार की स्थिति $\times$ खोला गया दरवाज़ा) गिनाइए, उन्हें हटाइए जिनमें
        कार दिख जाती है, और \emph{बचे हुए परिदृश्यों में} बदलने पर जीतने
        की \omterm{def:g10:proba:distribution}{प्रायिकता} निकालिए। क्या बदला, और संचालक का जानना क्यों मायने
        रखता है?
  \item अंतर्ज्ञान का पंप: सौ दरवाज़े, आप एक चुनते हैं, जानकार संचालक $98$
        बकरी-दरवाज़े खोल देता है। बदलेंगे? जीतने की \omterm{def:g10:proba:distribution}{प्रायिकता} कितनी?
  \item आपके सामने एक संशयवादी है जो, उन हज़ार पत्र-लेखकों की तरह,
        ``50--50'' पर अड़ा है। ताश पर आधारित एक अनुकरण का वर्णन कीजिए (तीन
        पत्ते, अनेक दौर, और एक मित्र संचालक के रूप में) और बताइए कि $300$
        दौरों के बाद बदलने पर जीत का कितना अनुपात आप अपेक्षित करते हैं ---
        और अपने अंतर के लिए इस अध्याय का उतार-चढ़ाव \omterm{def:g10:numbers:interval}{अंतराल} लीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- विरोधाभास के भाई-बंधु।}
\begin{enumerate}[resume]
  \item किसी परिवार में दो संतानें हैं; आपको पता चलता है कि \emph{कम से कम
        एक लड़का है}। इस सूचना के अनुकूल दो-संतान वाली समसंभावी रचनाएँ
        गिनाइए और $P(\text{दो लड़के})$ निकालिए।
  \item वही परिवार, पर अब सूचना यह है: \emph{बड़ी संतान लड़का है}। फिर से
        निकालिए। दोनों उत्तर अलग क्यों हैं, और उनमें दोनों मॉन्टी वाली बात
        क्या साझी है?
  \item तीन न्यायसंगत सिक्के: $P(\text{तीनों एक जैसे})$ निकालिए। गणितज्ञ दालम्बेर ने 1754 में
        एक और उत्तर के पक्ष में तर्क दिया था: ``या तो तीनों एक जैसे होंगे
        या नहीं --- दो में से एक स्थिति।'' \cref{prop:g10:proba:uniform} हाथ में लेकर ठीक-ठीक बताइए
        कि त्रुटि क्या है।
  \item एक खेल का शुल्क $2$ यूरो है: पासा फेंकिए, छक्के पर $6$ यूरो
        जीतिए, पंजे पर $3$ यूरो, अन्यथा कुछ नहीं (\cref{exo:g10:proba:9})। $600$ खेलों
        में गिनती के अनुसार प्रति खेल कितना औसत लाभ निकलता है? बीमाकर्ता
        भी यही गणित खेलता है: $10\,000$ यूरो के दावे की $1\,\%$ वार्षिक संभावना
        --- \emph{न्यायसंगत} प्रीमियम क्या है, और वास्तविक प्रीमियम उससे
        अधिक क्यों होता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- \omterm{def:g10:proba:sample}{प्रतिदर्श} पंच के रूप में।}
\begin{enumerate}[resume]
  \item एक सिक्का $100$ बार उछाला जाता है। उतार-चढ़ाव \omterm{def:g10:numbers:interval}{अंतराल} $p \pm \frac{1}{\sqrt n}$ का
        उपयोग करके तय कीजिए कि इनमें से कौन-से परिणाम भौंहें चढ़ानी चाहिए:
        $55$ चित; $70$ चित।
  \item $1\,000$ मतदाताओं के एक सर्वेक्षण में किसी उम्मीदवार को $52\,\%$ मिलता
        है। $50\,\%$ के आसपास उतार-चढ़ाव \omterm{def:g10:numbers:interval}{अंतराल} निकालिए और बताइए कि क्या यह
        सर्वेक्षण \emph{सिद्ध} करता है कि उम्मीदवार आगे है।
  \item एक कारख़ाना $4\,\%$ दोषपूर्ण पुर्ज़ों का दावा करता है। $400$ का एक
        \omterm{def:g10:proba:sample}{प्रतिदर्श} $28$ दोषपूर्ण दिखाता है ($7\,\%$); बाद में $2\,500$ का एक
        \omterm{def:g10:proba:sample}{प्रतिदर्श} $175$ दिखाता है (फिर $7\,\%$)। हर \omterm{def:g10:proba:sample}{प्रतिदर्श} के विरुद्ध उस
        दावे की जाँच कीजिए। दोनों निर्णय अलग क्यों हैं?
  \item हर सर्वेक्षण पर दो त्रुटि-स्रोत मँडराते हैं: प्रतिदर्शन की
        \emph{अभिनति} (माध्यमिक विद्यालय खंड की ``आँकड़े कैसे झूठ बोलते
        हैं'' वाली सप्ताहांत समस्या) और प्रतिदर्शन का \emph{उतार-चढ़ाव}
        (यह अध्याय)। $n$ बढ़ने पर कौन-सा सिकुड़ता है, और कौन-सा किसी भी
        प्रतिदर्श-आकार पर बचा रहता है? एक-एक वाक्य में।
  \item समापन --- प्रायिकतावादी की जाँच-सूची, हर मद पर एक पंक्ति: गिनने
        से पहले परिणाम समसंभावी हों (प्रश्न 15); बहु-चरणीय और ``कम से कम''
        वाली स्थितियों के लिए \omterm{met:g10:proba:tree}{वृक्ष} और \omterm{def:g10:proba:operations}{पूरक} (प्रश्न 2--3); \emph{सूचना
        \omterm{def:g10:proba:distribution}{प्रायिकताएँ} बदल देती है, और वह किस तरह मिली यह भी मायने रखता है}
        (प्रश्न 7--14); \omterm{def:g10:proba:sample}{प्रतिदर्श} लगभग $\frac{1}{\sqrt n}$ उतार-चढ़ाव करते हैं, और
        अनुकरण पंच का काम करता है (प्रश्न 11, 16--18)।
\end{enumerate}
\end{problem}
