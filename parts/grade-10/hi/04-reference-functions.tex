\chapter{आधार फलन}\label{ch:g10:reffunc}

गिने-चुने कुछ सरल \omterm{def:g10:functions:function}{फलन} --- एकघात, वर्ग, व्युत्क्रम, वर्गमूल, घन --- हर जगह
दिखते हैं, अकेले या मिलकर। उनके \omterm{def:g10:functions:graph}{आलेख} और परिवर्तन कंठस्थ हों तो अनेक समस्याएँ
एक त्वरित रेखाचित्र भर रह जाती हैं। यह अध्याय एक-एक करके उनका अध्ययन करता है
और \cref{ch:g10:functions} की तुलना-तकनीक से उनके परिवर्तन सिद्ध करता है।

\section{एकघात फलन}

\begin{definition}[एकघात फलन]\label{def:g10:reffunc:affine}
\emph{एकघात फलन}\index{एकघात फलन} इस रूप का \omterm{def:g10:functions:function}{फलन} है
\[
f(x) = mx + p,
\]
जहाँ $m$ और $p$ नियत \omterm{def:g10:numbers:sets}{वास्तविक संख्याएँ} हैं। इसका \omterm{def:g10:functions:graph}{आलेख} एक सरल रेखा है: $m$
\emph{प्रवणता}\index{प्रवणता} है और $p$ \emph{$y$-अंतःखंड}\index{y@$y$-अंतःखंड}
(रेखा ऊर्ध्वाधर अक्ष को $(0, p)$ पर काटती है)। जब $p = 0$ हो तब \omterm{def:g10:functions:function}{फलन} \emph{रैखिक}
होता है: $f(x) = mx$ समानुपात व्यक्त करता है।
\end{definition}

\begin{proposition}[प्रवणता और परिवर्तन]\label{prop:g10:reffunc:affine}
मान लीजिए $f(x) = mx + p$ है।
\begin{enumerate}
  \item किन्हीं दो भिन्न निवेशों $u \neq v$ के लिए: $m = \dfrac{f(v) - f(u)}{v - u}$ (\omterm{def:g10:reffunc:affine}{प्रवणता} निवेश में एक
        मात्रक परिवर्तन पर निर्गम का परिवर्तन है)।
  \item यदि $m > 0$ हो, तो $f$ $\R$ पर निरंतर \omterm{def:g10:functions:variations}{वर्धमान} है; यदि $m < 0$ हो, तो
        निरंतर \omterm{def:g10:functions:variations}{ह्रासमान}; और यदि $m = 0$ हो, तो अचर।
\end{enumerate}
\end{proposition}

\begin{proof}
1. $f(v) - f(u) = (mv + p) - (mu + p) = m(v - u)$ निकालिए, फिर $v - u \neq 0$ से भाग दीजिए।

2. यदि $u < v$ हो, तो $f(v) - f(u) = m(v-u)$ का चिह्न $m$ के चिह्न जैसा है, क्योंकि $v - u > 0$:
धनात्मक $m$ से $f(u) < f(v)$ मिलता है (\omterm{def:g10:functions:variations}{वर्धमान}), और ऋणात्मक $m$ से $f(u) > f(v)$ (\omterm{def:g10:functions:variations}{ह्रासमान})।
\end{proof}

\begin{example}\label{ex:g10:reffunc:affine}
वह \omterm{def:g10:reffunc:affine}{एकघात फलन} ज्ञात कीजिए जिसका \omterm{def:g10:functions:graph}{आलेख} $A(1, 3)$ और $B(4, 9)$ से होकर जाता है। \omterm{def:g10:reffunc:affine}{प्रवणता}
है
\[
m = \frac{9 - 3}{4 - 1} = \frac{6}{3} = 2 .
\]
फिर $f(x) = 2x + p$, और $f(1) = 3$ से $2 + p = 3$ मिलता है, इसलिए $p = 1$: $f(x) = 2x + 1$। $B$ से जाँच: $f(4) = 9$।
\end{example}

\section{वर्ग फलन}

\begin{proposition}[वर्ग फलन]\label{prop:g10:reffunc:square}
$\R$ पर परिभाषित \omterm{def:g10:functions:function}{फलन} $f(x) = x^2$:
\begin{enumerate}
  \item $\intoc{-\infty}{0}$ पर निरंतर \omterm{def:g10:functions:variations}{ह्रासमान} और $\intco{0}{+\infty}$ पर निरंतर \omterm{def:g10:functions:variations}{वर्धमान} है, और $x = 0$ पर
        उसका \omterm{def:g10:functions:extrema}{न्यूनतम} $0$ है;
  \item सभी $x$ के लिए $f(-x) = f(x)$ संतुष्ट करता है: उसका \omterm{def:g10:functions:graph}{आलेख}, एक
        \emph{परवलय}\index{परवलय}, ऊर्ध्वाधर अक्ष के परितः सममित है।
\end{enumerate}
\end{proposition}

\begin{proof}
1. मान लीजिए $0 \leq u < v$ है। तब
\[
f(v) - f(u) = v^2 - u^2 = (v - u)(v + u) > 0,
\]
क्योंकि $v - u > 0$ और $v + u > 0$: $f$ $\intco{0}{+\infty}$ पर निरंतर \omterm{def:g10:functions:variations}{वर्धमान} है। यदि $u < v \leq 0$ हो, तो
$v - u > 0$ पर $v + u < 0$, इसलिए $f(v) - f(u) < 0$: निरंतर \omterm{def:g10:functions:variations}{ह्रासमान}। अंत में सभी $x$ के लिए $x^2 \geq 0 = f(0)$।

2. $(-x)^2 = x^2$; इसलिए बिंदु $(x, x^2)$ और $(-x, x^2)$, जो ऊर्ध्वाधर अक्ष के आरपार एक-दूसरे के
\omterm{def:g10:functions:function}{प्रतिबिंब} हैं, दोनों \omterm{def:g10:functions:graph}{आलेख} पर हैं।
\end{proof}

\begin{remark}[वर्ग और क्रम]
चूँकि वर्ग \omterm{def:g10:functions:function}{फलन} ऋणात्मक ओर घटता है, इसलिए वर्ग लेने पर ऋणात्मक संख्याओं का
क्रम \emph{पलट} जाता है: $-3 < -2$ पर $9 > 4$। चिह्न जाँचे बिना किसी असमिका के दोनों
पक्षों का वर्ग कभी मत कीजिए।
\end{remark}

\section{व्युत्क्रम फलन}

\begin{proposition}[व्युत्क्रम फलन]\label{prop:g10:reffunc:inverse}
$x \neq 0$ के लिए परिभाषित \omterm{def:g10:functions:function}{फलन} $f(x) = \dfrac1x$:
\begin{enumerate}
  \item $\intoo{-\infty}{0}$ पर निरंतर \omterm{def:g10:functions:variations}{ह्रासमान} है और $\intoo{0}{+\infty}$ पर भी निरंतर \omterm{def:g10:functions:variations}{ह्रासमान};
  \item $f(-x) = -f(x)$ संतुष्ट करता है: उसका \omterm{def:g10:functions:graph}{आलेख}, एक
        \emph{अतिपरवलय}\index{अतिपरवलय}, मूल बिंदु के परितः सममित है।
\end{enumerate}
\end{proposition}

\begin{proof}
1. मान लीजिए $0 < u < v$ है। तब
\[
f(v) - f(u) = \frac1v - \frac1u = \frac{u - v}{uv} < 0,
\]
क्योंकि $u - v < 0$ और $uv > 0$: $\intoo{0}{+\infty}$ पर निरंतर \omterm{def:g10:functions:variations}{ह्रासमान}। $\intoo{-\infty}{0}$ पर उसी भागफल में $u - v < 0$
और फिर $uv > 0$ है (दो ऋणात्मक संख्याओं का गुणनफल), इसलिए $f$ वहाँ भी घटता है।

2. $\frac{1}{-x} = -\frac1x$।
\end{proof}

\begin{remark}
सावधान: $\frac1x$ अपने पूरे \omterm{def:g10:functions:function}{प्रांत} पर \omterm{def:g10:functions:variations}{ह्रासमान} \emph{नहीं} है। $u = -1$ से $v = 1$ तक
जाने पर मान $-1$ से उछलकर $1$ हो जाता है। दोनों शाखाओं का अध्ययन अलग-अलग
करना चाहिए।
\end{remark}

\section{वर्गमूल और घन}

\begin{proposition}[वर्गमूल फलन]\label{prop:g10:reffunc:sqrt}
$\intco{0}{+\infty}$ पर परिभाषित \omterm{def:g10:functions:function}{फलन} $f(x) = \sqrt{x}$ निरंतर \omterm{def:g10:functions:variations}{वर्धमान} है।
\end{proposition}

\begin{proof}
मान लीजिए $0 \leq u < v$ है। \emph{संयुग्मी} से गुणा और भाग कीजिए:
\[
\sqrt v - \sqrt u
= \frac{(\sqrt v - \sqrt u)(\sqrt v + \sqrt u)}{\sqrt v + \sqrt u}
= \frac{v - u}{\sqrt v + \sqrt u} > 0,
\]
क्योंकि $v - u > 0$ और $\sqrt v + \sqrt u > 0$ (ध्यान दीजिए कि $v > 0$, इसलिए $\sqrt v > 0$)।
\end{proof}

\begin{proposition}[घन फलन]\label{prop:g10:reffunc:cube}
$\R$ पर परिभाषित \omterm{def:g10:functions:function}{फलन} $f(x) = x^3$ निरंतर \omterm{def:g10:functions:variations}{वर्धमान} है, और उसका \omterm{def:g10:functions:graph}{आलेख} मूल बिंदु के
परितः सममित है।
\end{proposition}
\admitted

\begin{proposition}[$x$, $x^2$ और $\sqrt x$ की तुलना]\label{prop:g10:reffunc:compare}
$0 < x < 1$ के लिए:\ \ $x^2 < x < \sqrt x$।\quad
$x > 1$ के लिए:\ \ $\sqrt x < x < x^2$।\quad
$x = 0$ और $x = 1$ पर तीनों मान बराबर हैं।
\end{proposition}

\begin{proof}
मान लीजिए $0 < x < 1$ है। $x < 1$ को $x > 0$ से गुणा करने पर $x^2 < x$ मिलता है। आगे,
$x < \sqrt x$: चूँकि दोनों पक्ष धनात्मक हैं और धनात्मक संख्याओं पर वर्ग लेना \omterm{def:g10:functions:variations}{वर्धमान}
है, यह असमिका $x^2 < x$ के तुल्य है, जो अभी सिद्ध हो चुकी। $x > 1$ के लिए $x > 1$
को $x$ से गुणा करने पर $x^2 > x$ मिलता है, और वर्ग वाले उसी तर्क से $\sqrt x < x$
निकलता है।
\end{proof}

\begin{method}[प्रतिबिंबों की तुलना]\label{met:g10:reffunc:compare}
किसी आधार \omterm{def:g10:functions:function}{फलन} $f$ के लिए $f(a)$ और $f(b)$ की तुलना करने के लिए:
\begin{enumerate}
  \item $a$ और $b$ को ऐसे \omterm{def:g10:numbers:interval}{अंतराल} में रखिए जहाँ $f$ के परिवर्तन ज्ञात
        हों;
  \item यदि $f$ वहाँ बढ़ता है, तो \omterm{def:g10:functions:function}{प्रतिबिंब} निवेशों के क्रम में ही रहते
        हैं; यदि $f$ घटता है, तो क्रम उलट जाता है;
  \item यदि $a$ और $b$ एकदिष्टता के एक ही \omterm{def:g10:numbers:interval}{अंतराल} में न हों, तो पहले
        चिह्नों को सँभालिए (जैसे वर्ग के लिए: एक ऋणात्मक और एक धनात्मक
        निवेश)।
\end{enumerate}
\end{method}

\begin{example}
$\dfrac{1}{2.3}$ और $\dfrac{1}{2.4}$ की तुलना कीजिए: दोनों निवेश $\intoo{0}{+\infty}$ में हैं, जहाँ व्युत्क्रम
\omterm{def:g10:functions:function}{फलन} घटता है, और $2.3 < 2.4$ है, इसलिए $\dfrac{1}{2.3} > \dfrac{1}{2.4}$। अब $(-1.2)^2$ और $(-1.5)^2$ की तुलना कीजिए:
$\intoc{-\infty}{0}$ पर वर्ग घटता है, और $-1.5 < -1.2$ है, इसलिए $(-1.5)^2 > (-1.2)^2$ --- और सचमुच $2.25 > 1.44$।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g10:reffunc:1}
हर \omterm{def:g10:reffunc:affine}{एकघात फलन} के लिए उसकी \omterm{def:g10:reffunc:affine}{प्रवणता} और $y$-अंतःखंड बताइए, और यह भी कि वह
\omterm{def:g10:functions:variations}{वर्धमान} है या \omterm{def:g10:functions:variations}{ह्रासमान}:
\[
f(x) = 3x - 2, \qquad
g(x) = -\tfrac12 x + 4, \qquad
h(x) = 7, \qquad
k(x) = -x .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:reffunc:2}
वह \omterm{def:g10:reffunc:affine}{एकघात फलन} ज्ञात कीजिए जिसका \omterm{def:g10:functions:graph}{आलेख} $(2, 1)$ और $(5, 10)$ से होकर जाता है; फिर वह
जो $(-1, 4)$ और $(3, -4)$ से होकर जाता है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:reffunc:3}

(-2.7)^2 \text{ और } (-2.8)^2; \qquad
\frac{1}{5.1} \text{ और } \frac{1}{5.2}; \qquad
\sqrt{17} \text{ और } \sqrt{15}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:reffunc:4}
वर्ग \omterm{def:g10:functions:function}{फलन} के \omterm{def:g10:functions:graph}{आलेख} का उपयोग करके $x^2 = 16$ हल कीजिए, फिर $x^2 \leq 16$, फिर $x^2 > 9$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:reffunc:5}
मान लीजिए $x \in \intoo{0}{1}$ है। संख्याओं $x$, $x^2$, $x^3$ और $\sqrt x$ को छोटे से बड़े
के क्रम में रखिए, और अपने उत्तर की जाँच $x = 0.25$ से कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:reffunc:6}
पहले \omterm{def:g10:functions:graph}{आलेख} से और फिर बीजगणित से हल कीजिए: $\dfrac1x = 2$, और $x > 0$ के लिए $\dfrac1x < 2$।
यदि $x < 0$ की भी अनुमति दे दें तो क्या बदलता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:reffunc:7}
यह जानते हुए कि वर्ग \omterm{def:g10:functions:function}{फलन} $\intoc{-\infty}{0}$ पर घटता है और $\intco{0}{+\infty}$ पर बढ़ता है, $x^2$ को
निम्नलिखित स्थितियों में घेरिए:
\[
\text{(a) } x \in \intcc{2}{5};
\qquad
\text{(b) } x \in \intcc{-3}{-1};
\qquad
\text{(c) } x \in \intcc{-2}{3}.
\]
(स्थिति (c) में सावधान रहिए: $x^2$ का \omterm{def:g10:functions:extrema}{न्यूनतम} किसी सिरे पर नहीं है।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:reffunc:8}
एक टैक्सी कंपनी $4$ का नियत शुल्क और उसके ऊपर $1.5$ प्रति किलोमीटर लेती
है; दूसरी कोई नियत शुल्क नहीं लेती और $2$ प्रति किलोमीटर लेती है। हर
क़ीमत को दूरी $x$ के \omterm{def:g10:reffunc:affine}{एकघात फलन} से व्यक्त कीजिए, दोनों रेखाएँ खींचिए, और
ज्ञात कीजिए कि किस दूरी से आगे पहली कंपनी सस्ती पड़ती है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:reffunc:9}
दिखाइए कि सभी $a, b \geq 0$ के लिए $\sqrt{a + b} \leq \sqrt a + \sqrt b$ है। (संकेत: दोनों पक्ष अऋणात्मक हैं,
इसलिए उनके वर्गों की तुलना कीजिए।) समता कब होती है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:reffunc:10}
मान लीजिए $x \neq 1$ के लिए परिभाषित $f(x) = \dfrac{2x + 1}{x - 1}$ है।
\begin{enumerate}
  \item दिखाइए कि सभी $x \neq 1$ के लिए $f(x) = 2 + \dfrac{3}{x - 1}$ है।
  \item व्युत्क्रम \omterm{def:g10:functions:function}{फलन} के परिवर्तनों से $\intoo{1}{+\infty}$ पर $f$ के परिवर्तन निकालिए।
\end{enumerate}
\end{exercise}

\section{समस्या: प्रकृति के नियम आधार फलनों में बोलते हैं}

\begin{problem}[{सप्ताहांत समस्या --- ब्रेक-दूरी $v^2$ की तरह बढ़ती है,
उत्तोलक $1/d$ मानते हैं, ग्रह $a^{3/2}$ का अनुसरण करते हैं: आधार \omterm{def:g10:functions:function}{फलनों} से भौतिकी
पढ़ना}]
\label{pb:g10:reffunc:1}
कोई भी भौतिकी की पुस्तक खोलिए और हर पृष्ठ पर वही चंद \omterm{def:g10:functions:graph}{आलेख} मिलेंगे: परवलय,
अतिपरवलय, मूल-वक्र। प्रकृति अपने नियम इसी अध्याय के आधार \omterm{def:g10:functions:function}{फलनों} से लिखती है
--- और उनकी आकृतियाँ जान लेना (\cref{prop:g10:reffunc:square}, \cref{prop:g10:reffunc:inverse}, \cref{prop:g10:reffunc:sqrt}, \cref{prop:g10:reffunc:cube}) कार रोकने,
उत्तोलक सँभालने और ग्रहों का समय नापने के लिए पर्याप्त है। भव्य समापन है
केप्लर का तीसरा नियम, जो सौरमंडल के आँकड़ों से सीधे पढ़ लिया जाता है।

\medskip
\noindent\textbf{भाग I --- ब्रेक लगाने का वर्ग नियम।}
सूखी सड़क पर कार की ब्रेक-दूरी के लिए एक मोटा नियम: $v$ km/h की चाल पर
$d = \left(\frac{v}{10}\right)^{2}$ मीटर।
\begin{enumerate}
  \item $30$, $50$, $90$ और $130$ km/h पर ब्रेक-दूरियाँ निकालिए।
  \item चाल दुगुनी करने पर ब्रेक-दूरी कितने गुनी हो जाती है? वर्ग \omterm{def:g10:functions:function}{फलन} के
        मापन-गुण से समझाइए, और $50 \to 100$ km/h पर जाँचिए।
  \item अपराध-विज्ञान: फिसलन के निशान $50$ m नापे गए। सूत्र चालक को
        किस चाल का दोषी ठहराता है (km/h तक)? किस आधार \omterm{def:g10:functions:function}{फलन} ने उत्तर दिया
        --- और किस \omterm{def:g10:functions:function}{प्रांत} पर यह पठन असंदिग्ध है (\cref{prop:g10:reffunc:sqrt})?
  \item \emph{एक} km/h अधिक होने से बढ़ने वाली अतिरिक्त दूरी की तुलना
        कम और अधिक चाल पर कीजिए: $d(31) - d(30)$ और $d(91) - d(90)$ निकालिए। परवलय का
        कौन-सा गुण (\cref{prop:g10:reffunc:square}) इन दोनों उत्तरों में झलकता है?
  \item वास्तविक रुकने में प्रतिक्रिया-काल भी जुड़ता है --- लगभग एक सेकंड,
        जिसमें कार $0.28v$ मीटर चल जाती है: $D(v) = 0.28v + \left(\frac{v}{10}\right)^2$। $D(50)$ और $D(130)$
        निकालिए। शहर की चाल पर दोनों पदों --- एकघात और वर्ग --- में
        कौन-सा हावी रहता है, और राजमार्ग पर कौन-सा?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- रोज़मर्रा के अतिपरवलय।}
\begin{enumerate}[resume]
  \item उत्तोलक तब संतुलित होता है जब बल $\times$ दूरी दोनों ओर बराबर हो।
        $60$ kg का एक बच्चा आधार से $2$ m दूर बैठता है; दूसरी ओर दूरी
        $d$ पर संतुलनकारी बल $F(d) = \frac{120}{d}$ है (किलोग्राम-तुल्य में)। $F(0.5)$, $F(1)$,
        $F(3)$ निकालिए।
  \item व्युत्क्रम \omterm{def:g10:functions:function}{फलन} के परिवर्तन (\cref{prop:g10:reffunc:inverse}) उत्तोलकों के बारे में क्या
        कहते हैं --- और आर्किमिडीज़ की डींग (``मुझे खड़े होने की जगह दो,
        मैं पृथ्वी हिला दूँगा'') अतिपरवलय की पूँछ में कैसे छिपी है? $d = 0$
        किस कारण वर्जित है?
  \item ध्वनि और प्रकाश दूरी के \emph{वर्ग} के साथ मंद पड़ते हैं: किसी
        दीये से $1$ m दूर तीव्रता $100$ मात्रक है; $2$, $3$ और $10$
        m पर उसे बताइए। घातांक $2$ को एक गोले से समझाइए: दूरी $d$ पर
        प्रकाश कितने क्षेत्रफल पर फैल चुका होता है (माध्यमिक विद्यालय खंड
        की आर्किमिडीज़ की समाधि वाली समस्या)?
  \item विद्यालय की सैर के लिए बस का ख़र्च $600$ यूरो है, जो $n$
        सहभागियों में बराबर बँटता है। $n = 10, 20, 30$ के लिए प्रति व्यक्ति ख़र्च की
        सारणी बनाइए, ज्ञात कीजिए कि कितने सहभागी होने पर वह $25$ यूरो या
        उससे कम रह जाता है, और बताइए कि $n$ बढ़ने पर अतिपरवलय की आकृति
        क्या वादा करती है --- और क्या करने से मना कर देती है।
  \item $x$ पोस्टर बनाने का ख़र्च $2x + 3$ यूरो है ($3$ यूरो का प्रारंभिक
        जमाव)। दिखाइए कि प्रति पोस्टर \emph{औसत} ख़र्च $a(x) = 2 + \frac3x$ है, उसके
        परिवर्तन बताइए, और उसकी क्षैतिज अनन्तस्पर्शी की आर्थिक व्याख्या
        कीजिए। (\cref{exo:g10:reffunc:10} के बीजगणित से तुलना कीजिए।)
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- केप्लर का सामंजस्य।}
हर ग्रह के लिए मान लीजिए $a$ सूर्य से उसकी माध्य दूरी है (खगोलीय मात्रकों
में, पृथ्वी $= 1$) और $T$ उसका आवर्तकाल (वर्षों में)। आँकड़े: मंगल $a = 1.52$,
$T = 1.88$; बृहस्पति $a = 5.20$, $T = 11.86$; शनि $a = 9.54$, $T = 29.4$।
\begin{enumerate}[resume]
  \item पृथ्वी, मंगल और बृहस्पति के लिए $a^3$ और $T^2$ निकालिए। 1618 में
        केप्लर ने क्या देखा था?
  \item नियम को \omterm{def:g10:functions:function}{फलन} के रूप में बताइए: $T = \sqrt{a^3} = a\sqrt a$। इसे शनि पर जाँचिए।
  \item दो भविष्यवाणियाँ: एक क्षुद्रग्रह $a = 4$ AU पर परिक्रमा करता है ---
        उसका आवर्तकाल ज्ञात कीजिए (संख्याएँ पूरी निकलती हैं); एक धूमकेतु
        हर $27$ वर्ष लौटता है --- उसकी माध्य दूरी ज्ञात कीजिए (कोई पूर्ण
        घन ढूँढ़िए)।
  \item यह नियम किन तीन आधार \omterm{def:g10:functions:function}{फलनों} को आपस में गूँथता है? और उसका मापन-नियम:
        $a$ को $4$ से गुणा करने पर $T$ का क्या होता है? (क्षुद्रग्रह
        की पृथ्वी से तुलना करके जाँचिए।)
  \item केप्लर ने यह नियम टाइको ब्राहे के आँकड़ों में न्यूटन के गुरुत्वाकर्षण
        द्वारा उसकी व्याख्या से सत्तर वर्ष पहले पढ़ लिया था। एक वाक्य में:
        संख्याओं की सारणी में आधार \omterm{def:g10:functions:function}{फलन} पहचान लेने की शक्ति के बारे में यह
        प्रसंग क्या कहता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- कछुआ और ख़रगोश।}
\begin{enumerate}[resume]
  \item $\sqrt x$, $x$ और $x^2$ को $\intoo{0}{1}$ पर और $\intoo{1}{+\infty}$ पर क्रम में रखिए
        (\cref{prop:g10:reffunc:compare}), और दोनों क्रमों की जाँच $x = 0.25$ और $x = 4$ पर कीजिए।
  \item $\sqrt x > x$ को पूरी तरह हल कीजिए ($x \geq 0$ के लिए सावधानी से वर्ग कीजिए
        और \omterm{def:g10:algebra:signtable}{चिह्न-सारणी} से समाप्त कीजिए)।
  \item अब घन भी लाइए: $x$, $x^2$, $x^3$ को $x = 0.5$ पर और $x = 2$ पर क्रम
        में रखिए, और वे सभी बिंदु ज्ञात कीजिए जहाँ वर्ग और घन \omterm{def:g10:functions:function}{फलन}
        एक-दूसरे को काटते हैं।
  \item दो बचत योजनाएँ $t$ वर्षों बाद $100\sqrt t$ यूरो (योजना A) या $10 t^2$
        यूरो (योजना B) देती हैं। दिखाइए कि B, A से आगे तब निकलती है जब
        $t^3 = 100$ हो, और पार करने का समय महीने तक बताइए। दीर्घ काल में मूलों
        और घातों के बारे में क्या सीख मिली?
  \item समापन: इस समस्या के हर नियम के लिए --- ब्रेक ($v^2$), उत्तोलक
        ($1/d$), दीया ($1/d^2$), केप्लर ($a^{3/2}$), योजना A ($\sqrt t$) --- एक-एक
        उपवाक्य में बताइए कि उसके आधार \omterm{def:g10:functions:graph}{आलेख} की कौन-सी विशेषता भौतिक अर्थ
        उठाए हुए है (तेज़ होती चढ़ाई, ऊर्ध्वाधर अनन्तस्पर्शी, फैलता गोला,
        मापन-घातांक, धीमी पड़ती वृद्धि)। निष्कर्ष: आधार \omterm{def:g10:functions:function}{फलन} वर्णमाला हैं;
        प्रकृति उन्हीं से लिखती है।
\end{enumerate}
\end{problem}
