\section*{Chapitre \ref{ch:g10:proba} --- Probabilités et échantillonnage}

\begin{solution}{exo:g10:proba:1}
Chacune des $6$ issues a pour \omterm{def:g10:proba:distribution}{probabilité} $\frac16$.
«~Obtenir un $6$~»~: $\frac16$.
«~Nombre impair~» $= \{1, 3, 5\}$~: $\frac36 = \frac12$.
«~Au plus $4$~» $= \{1,2,3,4\}$~: $\frac46 = \frac23$.
«~Obtenir un $7$~»~: \omterm{def:g10:proba:events}{événement} impossible, \omterm{def:g10:proba:distribution}{probabilité} $0$.
\end{solution}

\begin{solution}{exo:g10:proba:2}
Il y a $10$ billes équiprobables.
$\P(\text{verte}) = \frac{5}{10} = \frac12$.
$\P(\text{non noire}) = 1 - \frac{2}{10} = \frac{8}{10} = \frac45$.
Verte et jaune sont \omterm{def:g10:proba:operations}{incompatibles}~:
$\P(\text{verte ou jaune}) = \frac{5}{10} + \frac{3}{10} = \frac{8}{10}
= \frac45$.
\end{solution}

\begin{solution}{exo:g10:proba:3}
Soit $S$~: «~étudie l'espagnol~», $G$~: «~étudie l'allemand~». Alors
$\P(S) = \frac{18}{30}$, $\P(G) = \frac{10}{30}$,
$\P(S \cap G) = \frac{4}{30}$. Règle d'addition~:
\[
\P(S \cup G) = \frac{18}{30} + \frac{10}{30} - \frac{4}{30}
= \frac{24}{30} = \frac45 .
\]
\omterm{def:g10:proba:operations}{Complémentaire}~: $\P(\text{ni l'un ni l'autre}) = 1 - \frac45 =
\frac15$.
\end{solution}

\begin{solution}{exo:g10:proba:4}
L'\omterm{met:g10:proba:tree}{arbre} a quatre chemins FF, FP, PF, PP, chacun de \omterm{def:g10:proba:distribution}{probabilité}
$\frac12 \times \frac12 = \frac14$.
Deux faces~: $\frac14$. Exactement une face (FP ou PF)~:
$\frac24 = \frac12$.
Au moins une face~: $1 - \P(\text{PP}) = 1 - \frac14 = \frac34$.
\end{solution}

\begin{solution}{exo:g10:proba:5}
Premier tirage~: rouge avec \omterm{def:g10:proba:distribution}{probabilité} $\frac{4}{10}$, blanche
$\frac{6}{10}$.
Second tirage \emph{sans} remise~: après une rouge, $3$ rouges et $6$
blanches restent parmi $9$~; après une blanche, $4$ rouges et $5$
blanches parmi $9$.

Deux rouges~: $\frac{4}{10} \times \frac39 = \frac{12}{90} =
\frac{2}{15}$.

Couleurs différentes~: chemins RB et BR~:
\[
\frac{4}{10} \times \frac69 + \frac{6}{10} \times \frac49
= \frac{24}{90} + \frac{24}{90} = \frac{48}{90} = \frac{8}{15}.
\]
\end{solution}

\begin{solution}{exo:g10:proba:6}
\emph{1.} Chaque dé montre indépendamment chaque face avec \omterm{def:g10:proba:distribution}{probabilité}
$\frac16$, donc les $36$ couples ordonnés sont équiprobables~; la somme
est déterminée par le couple.

\emph{2.} Somme $7$~: les couples $(1,6), (2,5), (3,4), (4,3), (5,2),
(6,1)$, donc $\P = \frac{6}{36} = \frac16$. Somme $12$~: seulement
$(6,6)$, $\P = \frac{1}{36}$. Somme au plus $4$~: couples sommant à
$2$, $3$ ou $4$~: $(1,1)$~; $(1,2), (2,1)$~; $(1,3), (2,2), (3,1)$ ---
six couples, donc $\P = \frac{6}{36} = \frac16$.

\emph{3.} Le nombre de couples réalisant chaque somme croît de $1$
(somme $2$) à $6$ (somme $7$) puis décroît~: la somme la plus probable
est $7$.
\end{solution}

\begin{solution}{exo:g10:proba:7}
Chaque question est correctement répondue avec \omterm{def:g10:proba:distribution}{probabilité} $\frac14$,
indépendamment. \omterm{met:g10:proba:tree}{Arbre} à deux niveaux~:

Les deux correctes~: $\frac14 \times \frac14 = \frac{1}{16}$.

Exactement une correcte~: chemins correct--faux et faux--correct~:
$\frac14 \times \frac34 + \frac34 \times \frac14 = \frac{6}{16}
= \frac38$.

Aucune correcte~: $\frac34 \times \frac34 = \frac{9}{16}$.

Vérification~: $\frac{1}{16} + \frac{6}{16} + \frac{9}{16} = 1$.
\end{solution}

\begin{solution}{exo:g10:proba:8}
\emph{1.} \omterm{def:g10:stats:series}{Fréquence} observée~: $\frac{396}{900} = 0.44$. Avec
$n = 900$~: $\frac{1}{\sqrt{900}} = \frac{1}{30} \approx 0.033$, donc
l'\omterm{def:g10:numbers:interval}{intervalle} de fluctuation est environ $\intcc{0.467}{0.533}$.

\emph{2.} La \omterm{def:g10:stats:series}{fréquence} observée $0.44$ est hors de l'\omterm{def:g10:numbers:interval}{intervalle}~: des
\omterm{def:g10:proba:sample}{échantillons} de $900$ électeurs d'une population à $50\,\%$ d'approbation
s'écartent rarement autant. Le sondage jette un doute sérieux sur les
$50\,\%$ affirmés.
\end{solution}

\begin{solution}{exo:g10:proba:9}
\emph{1.} $\P(\text{gagner } 6) = \frac16$ (lancer un $6$),
$\P(\text{gagner } 3) = \frac16$ (lancer un $5$),
$\P(\text{gagner } 0) = \frac46 = \frac23$.

\emph{2.} Sur $600$ parties, on attend environ $100$ six, $100$ cinq,
$400$ autres lancers. Gains~: environ
$100 \times 6 + 100 \times 3 = 900$. Coût~: $600 \times 2 = 1200$.
Résultat net attendu~: $900 - 1200 = -300$, c'est-à-dire une perte
d'environ $0.5$ par partie en \omterm{def:g10:stats:mean}{moyenne} --- le jeu ne vaut pas la peine
d'être joué.
\end{solution}

\begin{solution}{pb:g10:proba:1}
\textbf{1.} $P(\text{roi ou cœur}) = \frac{4}{52} +
\frac{13}{52} - \frac{1}{52} = \frac{16}{52} = \frac{4}{13}$~: le
roi de cœur appartient aux deux \omterm{def:g10:proba:events}{événements} et ne doit pas être compté
deux fois --- d'où la soustraction
(\cref{thm:g10:proba:addition}).

\textbf{2.} $P(\text{aucun pile}) = \left(\frac12\right)^3 =
\frac18$, donc $P(\text{au moins un}) = \frac78$.

\textbf{3.} $P(RR) = \frac35 \times \frac24 = \frac{3}{10}$~;
$P(\text{une de chaque}) = \frac35 \times \frac24 + \frac25
\times \frac34 = \frac{6}{20} + \frac{6}{20} = \frac35$.

\textbf{4.} $P(BB) = \frac25 \times \frac14 = \frac1{10}$~; et
$\frac{3}{10} + \frac{6}{10} + \frac{1}{10} = 1$~: les feuilles d'un
\omterm{met:g10:proba:tree}{arbre} totalisent toujours $1$.

\textbf{5.} Des \omterm{def:g10:proba:events}{événements} \omterm{def:g10:proba:operations}{incompatibles} vérifient
$P(A \cup B) = P(A) + P(B) = 1.1 > 1$~: impossible --- une
\omterm{def:g10:proba:distribution}{probabilité} a été inventée quelque part.

\textbf{6.} (La plupart des gens, y compris les mille épistoliers,
répondent «~cela ne change rien, une chance sur deux~». Gardez votre
réponse pour la question 8.)

\textbf{7.} Voiture derrière la porte 1 (\omterm{def:g10:proba:distribution}{probabilité} $\frac13$)~:
l'animateur ouvre l'une ou l'autre des portes à chèvre~; changer
abandonne la voiture~: \emph{perdu}. Voiture derrière la porte 2
($\frac13$)~: l'animateur \emph{doit} ouvrir la porte 3~; changer
aboutit sur la porte 2~: \emph{gagné}. Voiture derrière la porte 3~:
cas symétrique, \emph{gagné}. Au total~:
$P(\text{gagner en changeant}) = \frac13 + \frac13 = \frac23$.

\textbf{8.} Votre premier choix est mauvais avec la \omterm{def:g10:proba:distribution}{probabilité}
$\frac23$ --- et c'est précisément alors que l'ouverture forcée de
l'animateur laisse la voiture derrière la porte restante~: changer
convertit chaque erreur initiale en victoire. Donc $\frac23$, quoi
qu'en ait dit l'intuition.

\textbf{9.} Six scénarios équiprobables~: voiture derrière $1$, $2$
ou $3$, animateur ouvrant la porte $2$ ou $3$ (jamais la vôtre).
L'ouverture dévoile la voiture dans deux d'entre eux (voiture en 2 /
ouvre 2, voiture en 3 / ouvre 3)~: on les écarte. Parmi les quatre
survivants --- voiture en 1 / ouvre 2, voiture en 1 / ouvre 3,
voiture en 2 / ouvre 3, voiture en 3 / ouvre 2 --- changer gagne dans
les deux derniers~: $\frac24 = \frac12$. Avec un animateur ignorant,
«~une chance sur deux~» \emph{est} la bonne réponse~: le paradoxe
tient tout entier au savoir de l'animateur, qui filtre les scénarios
de façon inégale.

\textbf{10.} Votre porte cache la voiture avec la \omterm{def:g10:proba:distribution}{probabilité}
$\frac{1}{100}$~; les $98$ ouvertures de l'animateur qui sait
canalisent les $\frac{99}{100}$ restants sur l'unique porte encore
fermée. Changez, et gagnez $99$ fois sur $100$.

\textbf{11.} Trois cartes (un as, deux jokers), un camarade qui
regarde et écarte toujours un joker parmi les deux cartes non
choisies~; jouer $300$ parties en changeant systématiquement et noter
les victoires. \omterm{def:g10:stats:series}{Fréquence} attendue $\frac23 \approx 0.667$, avec une
fluctuation $\pm\frac{1}{\sqrt{300}} \approx 0.058$~: on attend entre
$61\,\%$ et $72\,\%$ environ de victoires en changeant --- très loin
de toute «~chance sur deux~».

\textbf{12.} Configurations équiprobables (aîné--cadet)~: GG, GF,
FG, FF. «~Au moins un garçon~» conserve GG, GF, FG~:
$P(\text{GG}) = \frac13$.

\textbf{13.} «~L'aîné est un garçon~» conserve GG et GF~:
$P(\text{GG}) = \frac12$. Des informations différentes filtrent des
ensembles de scénarios différents --- exactement comme les deux
animateurs~: ce que l'on apprend compte, et \emph{la façon} dont on
l'a appris compte tout autant.

\textbf{14.} Toutes identiques~: PPP et FFF, soit
$\frac28 = \frac14$. L'erreur de d'Alembert~: ses deux «~cas~» ne
sont pas équiprobables --- «~pas toutes identiques~» regroupe six des
huit issues. La \cref{prop:g10:proba:uniform} ne dénombre que sur des
issues équiprobables.

\textbf{15.} Gain moyen par partie~: $6 \times \frac16 + 3
\times \frac16 = 1.50$ euro, contre une mise de $2$ euros~: une perte
\omterm{def:g10:stats:mean}{moyenne} de $0.50$ par partie, soit environ $300$ euros sur $600$
parties. La prime équitable égalise le versement moyen~:
$1\,\% \times 10\,000 = 100$ euros~; les primes réelles ajoutent les
frais et une marge de sécurité --- l'assureur doit survivre aux
mauvaises années, pas seulement aux années \omterm{def:g10:stats:mean}{moyennes}.

\textbf{16.} \omterm{def:g10:numbers:interval}{Intervalle}~: $0.5 \pm \frac{1}{\sqrt{100}} =
\intcc{0.4}{0.6}$. $55$ piles ($0.55$)~: à l'intérieur, rien à
signaler. $70$ piles ($0.70$)~: largement dehors --- il faut douter
de la pièce.

\textbf{17.} $\frac{1}{\sqrt{1000}} \approx 0.032$~: un candidat
réellement à $50\,\%$ obtiendrait typiquement un score dans
$\intcc{0.468}{0.532}$ --- et $0.52$ y tient confortablement. Le
sondage \emph{suggère} une avance mais ne prouve rien~: la marge
l'engloutit.

\textbf{18.} Premier \omterm{def:g10:proba:sample}{échantillon}~: \omterm{def:g10:numbers:interval}{intervalle}
$0.04 \pm \frac{1}{20} = \intcc{0}{0.09}$~; le $0.07$ observé est
dedans~: pas de verdict. Second~:
$0.04 \pm \frac{1}{50} = \intcc{0.02}{0.06}$~; le $0.07$ observé
tombe dehors~: l'annonce devient sérieusement douteuse. Même
\omterm{def:g10:stats:series}{fréquence}, instrument plus fin~: la précision croît comme $\sqrt n$.

\textbf{19.} La fluctuation décroît comme $\frac{1}{\sqrt n}$~:
quadrupler l'\omterm{def:g10:proba:sample}{échantillon} divise le bruit par deux. Le biais ---
interroger la mauvaise foule --- survit à tout $n$~: deux millions de
bulletins mal choisis restaient faux (devoir du week-end sur les
mensonges des données, dans le volume précédent).

\textbf{20.} (i) Vérifier que les issues sont équiprobables avant de
dénombrer --- d'Alembert l'avait oublié. (ii) Les \omterm{met:g10:proba:tree}{arbres} multiplient
le long des branches~; les «~au moins un~» capitulent devant
l'\omterm{def:g10:proba:events}{événement} contraire. (iii) L'information filtre les scénarios, et
sa \emph{source} change le filtre --- animateur savant contre
animateur ignorant, aîné-garçon contre au-moins-un-garçon. (iv) Les
\omterm{def:g10:stats:series}{fréquences} s'approchent des \omterm{def:g10:proba:distribution}{probabilités} à la vitesse
$\frac{1}{\sqrt n}$, si bien qu'une simulation patiente règle
n'importe quelle querelle --- y compris avec dix mille épistoliers en
colère.
\end{solution}
