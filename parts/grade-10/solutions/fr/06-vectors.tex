\section*{Chapitre \ref{ch:g10:vectors} --- Vecteurs}

\begin{solution}{exo:g10:vectors:1}
$\vect{AB} + \vect{BD} = \vect{AD}$ (Chasles).

$\vect{KL} + \vect{LM} + \vect{MK} = \vect{KK} = \vec 0$.

$\vect{AC} - \vect{BC} = \vect{AC} + \vect{CB} = \vect{AB}$.

$\vect{AB} + \vect{CA} = \vect{CA} + \vect{AB} = \vect{CB}$.
\end{solution}

\begin{solution}{exo:g10:vectors:2}
$\vect{AB}\,(5-2,\ 3-1) = (3, 2)$~; \quad
$\vect{BC}\,(-1-5,\ 4-3) = (-6, 1)$~; \quad
$\vect{AC}\,(-1-2,\ 4-1) = (-3, 3)$.
Vérification~: $(3, 2) + (-6, 1) = (-3, 3)$, coordonnée par
coordonnée.
\end{solution}

\begin{solution}{exo:g10:vectors:3}
$\vec u + \vec v = (2 + (-1),\ -3 + 4) = (1, 1)$.

$3\vec u = (6, -9)$.

$2\vec u - \vec v = (4 - (-1),\ -6 - 4) = (5, -10)$.

$\norm{\vec u} = \sqrt{2^2 + (-3)^2} = \sqrt{13}$.
\end{solution}

\begin{solution}{exo:g10:vectors:4}
(a)~$4 \times 9 - 6 \times 6 = 36 - 36 = 0$~: \omterm{def:g10:vectors:collinear}{colinéaires}
($\vec v = \frac32 \vec u$).

(b)~$2 \times 10 - (-5) \times (-4) = 20 - 20 = 0$~: \omterm{def:g10:vectors:collinear}{colinéaires}
($\vec v = -2\vec u$).

(c)~$3 \times 3 - 1 \times 1 = 8 \neq 0$~: non \omterm{def:g10:vectors:collinear}{colinéaires}.
\end{solution}

\begin{solution}{exo:g10:vectors:5}
$\vect{AB}\,(4, -2)$, et $\vect{DC}\,(6 - x_D,\ 3 - y_D)$. La condition
$\vect{AB} = \vect{DC}$ donne $6 - x_D = 4$ et $3 - y_D = -2$, donc
$D(2, 5)$.
\end{solution}

\begin{solution}{exo:g10:vectors:6}
\emph{1.} $\vect{AB}\,(3, 2)$ et $\vect{AC}\,(6, -2)$, donc
$\vect{AB} + \vect{AC} = (9, 0)$ et
$M = (-2 + 9,\ 1 + 0) = (7, 1)$.

\emph{2.} Par la règle du parallélogramme,
$\vect{AM} = \vect{AB} + \vect{AC}$ signifie que $ABMC$ est un
parallélogramme ($M$ est le quatrième sommet opposé à $A$). On peut
vérifier~: $\vect{AB}\,(3,2)$ et $\vect{CM}\,(7-4,\ 1-(-1)) = (3, 2)$
sont égaux.
\end{solution}

\begin{solution}{exo:g10:vectors:7}
Premier triplet~: $\vect{AB}\,(3, 2)$, $\vect{AC}\,(9, 6)$~;
$3 \times 6 - 2 \times 9 = 0$~: alignés ($\vect{AC} = 3\vect{AB}$).

Second triplet~: $\vect{AB}\,(2, -1)$, $\vect{AC}\,(8, -4)$~;
$2 \times (-4) - (-1) \times 8 = -8 + 8 = 0$~: alignés aussi
($\vect{AC} = 4\vect{AB}$).
\end{solution}

\begin{solution}{exo:g10:vectors:8}
$\vect{AB}\,(4, 2)$ et $\vect{DC}\,(6-2,\ 1-(-1)) = (4, 2)$~: les deux
\omterm{def:g10:vectors:vector}{vecteurs} sont \emph{égaux}, donc $ABCD$ est un parallélogramme, et en
particulier $(AB) \parallel (CD)$. La propriété de parallélogramme est
la plus forte~: elle implique le parallélisme (colinéarité des
\omterm{def:g10:vectors:vector}{vecteurs}), pas l'inverse.
\end{solution}

\begin{solution}{exo:g10:vectors:9}
\emph{1.} $\vect{AB}\,(6, 3)$, donc
$\vect{AI} = \frac23\vect{AB} = (4, 2)$ et $I(4, 2)$.

\emph{2.} $\vect{CB}\,(6-2,\ 3-5) = (4, -2)$, donc
$\vect{CJ} = \frac23\vect{CB} = \left(\frac83, -\frac43\right)$ et
$J\left(2 + \frac83,\ 5 - \frac43\right) = \left(\frac{14}{3},
\frac{11}{3}\right)$.

\emph{3.} $\vect{IJ}\,\left(\frac{14}{3} - 4,\ \frac{11}{3} - 2\right) =
\left(\frac23, \frac53\right)$ et $\vect{AC}\,(2, 5)$. Critère de
colinéarité~:
\[
\frac23 \times 5 - \frac53 \times 2 = \frac{10}{3} - \frac{10}{3} = 0 :
\]
les \omterm{def:g10:vectors:vector}{vecteurs} sont \omterm{def:g10:vectors:collinear}{colinéaires}, donc $(IJ)$ est parallèle à $(AC)$.
(Les deux points $I$ et $J$ sont situés aux deux tiers du côté vers
$B$~: le petit triangle $IBJ$ est une réduction de $ABC$.)
\end{solution}

\begin{solution}{exo:g10:vectors:10}
Les milieux sont
\[
I\left(\tfrac{x_A + x_B}{2}, \tfrac{y_A + y_B}{2}\right), \quad
J\left(\tfrac{x_B + x_C}{2}, \tfrac{y_B + y_C}{2}\right), \quad
K\left(\tfrac{x_C + x_D}{2}, \tfrac{y_C + y_D}{2}\right), \quad
L\left(\tfrac{x_D + x_A}{2}, \tfrac{y_D + y_A}{2}\right).
\]
Premières \omterm{def:g10:coordgeom:system}{coordonnées} des deux \omterm{def:g10:vectors:vector}{vecteurs}~:
\[
x_J - x_I = \frac{x_B + x_C}{2} - \frac{x_A + x_B}{2}
= \frac{x_C - x_A}{2},
\qquad
x_K - x_L = \frac{x_C + x_D}{2} - \frac{x_D + x_A}{2}
= \frac{x_C - x_A}{2},
\]
et le même calcul fonctionne pour les secondes \omterm{def:g10:coordgeom:system}{coordonnées}. Donc
$\vect{IJ} = \vect{LK}$ (tous deux égaux à $\frac12\vect{AC}$), et
$IJKL$ est un parallélogramme, quelle que soit l'allure du
quadrilatère $ABCD$.
\end{solution}

\begin{solution}{pb:g10:vectors:1}
\textbf{1.} $\vect{AB} + \vect{BC} + \vect{CD} = \vect{AD}$~;
\quad $\vect{MA} - \vect{MB} = \vect{MA} + \vect{BM} = \vect{BA}$~;
\quad $\vect{AB} + \vect{BC} + \vect{CD} + \vect{DA} = \vect{AA} =
\vec 0$ (en réordonnant la troisième \omterm{def:g10:vectors:sum}{somme}).

\textbf{2.} $\vect{OB} - \vect{OA} = \vect{AO} + \vect{OB} =
\vect{AB}$ par Chasles.

\textbf{3.} $I$ est le \omterm{prop:g10:coordgeom:midpoint}{milieu} de $[AB]$ exactement lorsque
$\vect{IA} = -\vect{IB}$, c'est-à-dire
$\vect{IA} + \vect{IB} = \vec 0$. Alors, pour tout $O$~:
$\vec 0 = \vect{IO} + \vect{OA} + \vect{IO} + \vect{OB}$, donc
$2\,\vect{OI} = \vect{OA} + \vect{OB}$ --- et réciproquement.

\textbf{4.} $\vect{AB} = (4, 2)$, et $D$ vérifie
$\vect{DC} = \vect{AB}$~: $D = C - (4, 2) = (2, -2)$ --- le même $D$
qu'au \cref{pb:g10:coordgeom:1}, obtenu une ligne plus vite.

\textbf{5.} La translation de \omterm{def:g10:vectors:vector}{vecteur} $\vec u$ --- exactement la
transformation produite par deux miroirs parallèles ou par deux
demi-tours. \omterm{def:g10:functions:function}{Image} de $(1, 2)$~: $(4, 1)$.

\textbf{6.} En développant chaque $\vect{GX}$ en
$\vect{GO} + \vect{OX}$~:
$3\,\vect{GO} + \vect{OA} + \vect{OB} + \vect{OC} = \vec 0$, donc
$\vect{OG} = \frac13(\vect{OA} + \vect{OB} + \vect{OC})$~: un tel
point existe et est unique, et ses \omterm{def:g10:coordgeom:system}{coordonnées} sont les moyennes de
celles des trois sommets.

\textbf{7.} $\vect{AB} + \vect{AC} =
(\vect{AG} + \vect{GB}) + (\vect{AG} + \vect{GC})
= 2\vect{AG} + (\vect{GB} + \vect{GC})
= 2\vect{AG} - \vect{GA} = 3\vect{AG}$, d'où
$\vect{AG} = \frac13(\vect{AB} + \vect{AC})$. La formule du \omterm{prop:g10:coordgeom:midpoint}{milieu}
(question 3, vue depuis $A$) donne
$\vect{AK} = \frac12(\vect{AB} + \vect{AC})$. En comparant~:
$\vect{AG} = \frac23\vect{AK}$ --- $G$ est sur la médiane $[AK]$, aux
deux tiers à partir de $A$.

\textbf{8.} La condition de définition traite $A$, $B$, $C$ de façon
identique~; le même calcul mené depuis $B$ ou depuis $C$ place donc
le \emph{même} $G$ aux deux tiers de ces médianes~: le concours ne
coûte rien. La démonstration par le parallélogramme était ingénieuse
mais taillée sur mesure~; les \omterm{def:g10:coordgeom:system}{coordonnées} calculent mais masquent~;
les \omterm{def:g10:vectors:vector}{vecteurs} rendent la symétrie visible et la démonstration tient en
deux lignes.

\textbf{9.} $G\left(\frac{1+5+3}{3}, \frac{1+2+6}{3}\right) =
(3, 3)$~; $K(4, 4)$~; $\vect{AG} = (2, 2)$ et
$\vect{AK} = (3, 3)$~: déterminant
$2 \times 3 - 2 \times 3 = 0$, \omterm{def:g10:vectors:collinear}{vecteurs colinéaires}, et en effet
$\vect{AG} = \frac23\vect{AK}$.

\textbf{10.} $G'\left(\frac{2 + 5 + 3}{4}, \frac{2 + 2 +
6}{4}\right) = (2.5,\ 2.5)$~: tiré de $G(3,3)$ vers le sommet alourdi
$A(1,1)$, comme il sied à un point d'équilibre.

\textbf{11.} $3 \times 4 - (-2)(-6) = 12 - 12 = 0$~: \omterm{def:g10:vectors:collinear}{colinéaires}
($\vec v = -2\vec u$). $2 \times 9 - 5 \times 4 = -2 \neq 0$~: non
\omterm{def:g10:vectors:collinear}{colinéaires}.

\textbf{12.} $\vect{AB} = (2, 3)$, $\vect{AC} = (6, 9)$~:
déterminant $2 \times 9 - 3 \times 6 = 0$~: les points sont alignés.

\textbf{13.} $\vect{OI} = \frac12(\vect{OA} + \vect{OB})$ et
$\vect{OK} = \frac12(\vect{OC} + \vect{OD})$, donc le \omterm{prop:g10:coordgeom:midpoint}{milieu} de
$[IK]$ a pour \omterm{def:g10:vectors:vector}{vecteur} position
$\frac12(\vect{OI} + \vect{OK}) = \frac14(\vect{OA} +
\vect{OB} + \vect{OC} + \vect{OD})$ --- et le calcul pour $[JL]$
donne exactement le même \omterm{def:g10:vectors:vector}{vecteur}. Les deux bimédianes (et, par
Varignon, les diagonales du parallélogramme $IJKL$) partagent toutes
ce centre~: le centre de gravité du quadrilatère.

\textbf{14.} $\vect{MN} = \vect{AN} - \vect{AM} =
\frac13\vect{AC} - \frac13\vect{AB} = \frac13\vect{BC}$~: donc
$(MN) \parallel (BC)$ et $MN = \frac13 BC$.

\textbf{15.} Avec $\vect{AM} = k\,\vect{AB}$ et
$\vect{AN} = k\,\vect{AC}$~: $\vect{MN} = k\,\vect{BC}$ --- une seule
ligne contient le théorème des milieux ($k = \frac12$) et le
théorème de Thalès ($k$ quelconque)~: le rapport traverse la
soustraction sans une égratignure.

\textbf{16.} Résultante $(3, 4)$, de norme $\sqrt{9 + 16} = 5$ km/h~:
le bateau file en réalité vers le nord-nord-est, le long de
l'hypoténuse du triangle $3$--$4$--$5$.

\textbf{17.} $\vec u + \vec v + \vec w = (0, 0)$~: l'anneau ne bouge
pas --- équilibre. Le long d'un polygone fermé
$A_1 A_2 \dots A_n A_1$, Chasles télescope la \omterm{def:g10:vectors:sum}{somme} en
$\vect{A_1 A_1} = \vec 0$~: faire le tour, c'est revenir.

\textbf{18.} $\vec w = -(\vec u + \vec v) = -(3, 2) = (-3, -2)$.

\textbf{19.} Résultante $(1.5,\ 2)$, de norme
$\sqrt{2.25 + 4} = 2.5$ m/s. Traversée~: $50$ m à $2$ m/s dans le
sens de la largeur, soit $25$ s~; dérive~: $1.5 \times 25 = 37.5$ m
en aval.

\textbf{20.} Redémontrés ce week-end~: le théorème des deux tiers
pour le centre de gravité (question 7), le centre des bimédianes et
de Varignon (question 13), Thalès et les milieux (questions 14 et
15). Ce que les flèches ne savent toujours pas faire~: mesurer des
\emph{angles} (et donc des longueurs dans des directions
quelconques) --- le produit scalaire de l'an prochain est précisément
l'instrument manquant.
\end{solution}
