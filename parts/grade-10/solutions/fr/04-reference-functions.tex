\section*{Chapitre \ref{ch:g10:reffunc} --- Fonctions de référence}

\begin{solution}{exo:g10:reffunc:1}
$f$~: coefficient $3$, \omterm{def:g10:reffunc:affine}{ordonnée à l'origine} $-2$, \omterm{def:g10:functions:variations}{croissante}. \quad
$g$~: coefficient $-\frac12$, \omterm{def:g10:reffunc:affine}{ordonnée à l'origine} $4$, \omterm{def:g10:functions:variations}{décroissante}.
\quad $h$~: coefficient $0$, \omterm{def:g10:reffunc:affine}{ordonnée à l'origine} $7$, constante.
\quad $k$~: coefficient $-1$, \omterm{def:g10:reffunc:affine}{ordonnée à l'origine} $0$, \omterm{def:g10:functions:variations}{décroissante}
(et linéaire).
\end{solution}

\begin{solution}{exo:g10:reffunc:2}
Par $(2,1)$ et $(5,10)$~: coefficient $m = \frac{10 - 1}{5 - 2} = 3$~;
puis $1 = 3 \times 2 + p$ donne $p = -5$~: $f(x) = 3x - 5$.

Par $(-1,4)$ et $(3,-4)$~: coefficient
$m = \frac{-4 - 4}{3 - (-1)} = \frac{-8}{4} = -2$~; puis
$4 = -2 \times (-1) + p$ donne $p = 2$~: $f(x) = -2x + 2$.
\end{solution}

\begin{solution}{exo:g10:reffunc:3}
$(3.1)^2 < (3.2)^2$~: le carré croît sur les positifs.

$(-2.7)^2 < (-2.8)^2$~: le carré décroît sur les négatifs et
$-2.8 < -2.7$.

$\frac{1}{5.1} > \frac{1}{5.2}$~: l'inverse décroît sur les positifs.

$\sqrt{17} > \sqrt{15}$~: la racine carrée croît.
\end{solution}

\begin{solution}{exo:g10:reffunc:4}
$x^2 = 16$~: deux solutions, $x = 4$ et $x = -4$ (la droite horizontale
$y = 16$ coupe la parabole deux fois).

$x^2 \leq 16$~: la parabole est sous la droite entre les deux points
d'\omterm{def:g10:numbers:interunion}{intersection}~: $x \in \intcc{-4}{4}$.

$x^2 > 9$~: la parabole est strictement au-dessus de $y = 9$ à
l'extérieur de $\intcc{-3}{3}$~:
$x \in \intoo{-\infty}{-3} \cup \intoo{3}{+\infty}$.
\end{solution}

\begin{solution}{exo:g10:reffunc:5}
Pour $0 < x < 1$~: multiplier $x < 1$ plusieurs fois par $x$ donne
$x^3 < x^2 < x$, et le \cref{prop:g10:reffunc:compare} donne
$x < \sqrt x$, donc
\[
x^3 < x^2 < x < \sqrt x .
\]
Vérification avec $x = 0.25$~: $x^3 = 0.015625$, $x^2 = 0.0625$,
$x = 0.25$, $\sqrt x = 0.5$.
\end{solution}

\begin{solution}{exo:g10:reffunc:6}
$\frac1x = 2$ a l'unique solution $x = \frac12$ (l'hyperbole rencontre
la droite horizontale $y = 2$ une fois, sur la branche positive).

Pour $x > 0$~: $\frac1x < 2$. Multiplier par $x > 0$~: $1 < 2x$, donc
$x > \frac12$~: ensemble des solutions $\intoo{\frac12}{+\infty}$.

Si $x < 0$ est autorisé~: tout $x$ négatif satisfait
$\frac1x < 0 < 2$, donc l'ensemble complet des solutions est
$\intoo{-\infty}{0} \cup \intoo{\frac12}{+\infty}$.
\end{solution}

\begin{solution}{exo:g10:reffunc:7}
(a)~Sur $\intcc{2}{5}$, toutes les entrées sont positives et le carré
croît~: $4 \leq x^2 \leq 25$.

(b)~Sur $\intcc{-3}{-1}$, le carré décroît~: le plus grand carré vient
de $-3$~: $1 \leq x^2 \leq 9$.

(c)~Sur $\intcc{-2}{3}$, l'\omterm{def:g10:numbers:interval}{intervalle} contient $0$, où le carré atteint
son \omterm{def:g10:functions:extrema}{minimum} $0$~; le \omterm{def:g10:functions:extrema}{maximum} est le plus grand de $(-2)^2 = 4$ et
$3^2 = 9$. Donc $0 \leq x^2 \leq 9$.
\end{solution}

\begin{solution}{exo:g10:reffunc:8}
Première compagnie~: $f(x) = 1.5x + 4$~; seconde~: $g(x) = 2x$. La
première est moins chère lorsque
\[
1.5x + 4 < 2x
\ \Longleftrightarrow\
4 < 0.5x
\ \Longleftrightarrow\
x > 8 .
\]
Au-delà de $8$~km, la première compagnie est moins chère~; à exactement
$8$~km les deux facturent $16$.
\end{solution}

\begin{solution}{exo:g10:reffunc:9}
Les deux membres sont positifs ou nuls, donc l'inégalité équivaut à
celle entre leurs carrés~:
\[
\left(\sqrt{a+b}\right)^2 = a + b
\qquad\text{et}\qquad
\left(\sqrt a + \sqrt b\right)^2 = a + 2\sqrt a \sqrt b + b .
\]
Comme $2\sqrt a\sqrt b \geq 0$, le second carré est au moins égal au
premier, ce qui prouve $\sqrt{a+b} \leq \sqrt a + \sqrt b$. L'égalité a
lieu exactement lorsque $\sqrt a \sqrt b = 0$, c'est-à-dire lorsque
$a = 0$ ou $b = 0$.
\end{solution}

\begin{solution}{exo:g10:reffunc:10}
\emph{1.} Mettre le second membre sur un dénominateur commun~:
\[
2 + \frac{3}{x-1} = \frac{2(x-1) + 3}{x - 1} = \frac{2x + 1}{x - 1}
= f(x).
\]

\emph{2.} Sur $\intoo{1}{+\infty}$, lorsque $x$ croît, $x - 1$ croît et
reste positif, donc $\frac{3}{x-1}$ décroît (\omterm{def:g10:functions:function}{fonction} inverse, multipliée
par la constante positive $3$), donc $f(x) = 2 + \frac{3}{x-1}$
décroît~: $f$ est strictement \omterm{def:g10:functions:variations}{décroissante} sur $\intoo{1}{+\infty}$.
\end{solution}

\begin{solution}{pb:g10:reffunc:1}
\textbf{1.} $d(30) = 3^2 = 9$ m~; $d(50) = 25$ m~; $d(90) = 81$ m~;
$d(130) = 169$ m.

\textbf{2.} Doubler $v$ double $\frac v{10}$ et multiplie son carré
par $4$~: $d(100) = 100$ m $= 4 \times d(50)$. Deux fois la vitesse,
quatre fois la distance --- l'effet d'échelle de la \omterm{def:g10:functions:function}{fonction} carré.

\textbf{3.} $\left(\frac{v}{10}\right)^2 = 50$ donne
$\frac{v}{10} = \sqrt{50}$, donc $v = 10\sqrt{50} \approx 71$ km/h.
C'est la racine carrée qui a répondu~; la lecture est sans ambiguïté
parce que les vitesses sont positives et que, sur
$\intco{0}{+\infty}$, la \omterm{def:g10:functions:function}{fonction} carré est strictement \omterm{def:g10:functions:variations}{croissante}
(\cref{prop:g10:reffunc:sqrt}~: un seul \omterm{def:g10:functions:function}{antécédent} positif).

\textbf{4.} $d(31) - d(30) = 9.61 - 9 = 0.61$ m, tandis que
$d(91) - d(90) = 82.81 - 81 = 1.81$ m~: le même $+1$ km/h coûte trois
fois plus de distance à haute vitesse. La parabole ne fait pas que
monter, elle \emph{se redresse} --- sa vitesse de montée croît avec
$x$.

\textbf{5.} $D(50) = 14 + 25 = 39$ m~; $D(130) = 36.4 + 169 =
205.4$ m. En ville, le terme affine (la réaction) pèse le plus~; sur
autoroute, le terme carré l'écrase --- l'affine croît régulièrement,
les carrés s'emballent.

\textbf{6.} $F(0.5) = 240$, $F(1) = 120$, $F(3) = 40$.

\textbf{7.} La \omterm{def:g10:functions:function}{fonction} inverse est \omterm{def:g10:functions:variations}{décroissante}~: plus on s'éloigne
du pivot, plus la force nécessaire est faible --- avec un bras de
levier assez long ($d$ énorme), n'importe quelle force, si petite
soit-elle, équilibre n'importe quelle charge~: la fanfaronnade
d'Archimède vit dans la queue de l'hyperbole, qui s'approche de $0$
sans jamais l'atteindre. Et $d = 0$ est la valeur interdite~: pas de
bras, pas de levier --- c'est l'asymptote verticale.

\textbf{8.} $\frac{100}{4} = 25$ à $2$ m,
$\frac{100}{9} \approx 11$ à $3$ m, $1$ à $10$ m. À la distance $d$,
la lumière s'est répandue sur une sphère d'aire $4\pi d^2$ (problème
de la pierre tombale d'Archimède, dans le volume précédent)~: la même
énergie divisée par une aire qui croît comme $d^2$ --- d'où l'inverse
du \emph{carré}.

\textbf{9.} $60$, $30$, $20$ euros par tête pour $n = 10, 20, 30$.
Pour $\frac{600}{n} \leq 25$~: il faut $n \geq 24$ participants.
L'hyperbole promet des parts toujours moins chères quand $n$ grandit
--- et refuse d'atteindre $0$~: le car n'est jamais gratuit.

\textbf{10.} $a(x) = \frac{2x + 3}{x} = 2 + \frac3x$~: \omterm{def:g10:functions:variations}{décroissante}
sur $\intoo{0}{+\infty}$ (la \omterm{def:g10:functions:function}{fonction} inverse translatée), et
s'approchant de l'asymptote $y = 2$. Économiquement~: répartir les
$3$ euros fixes sur davantage d'affiches pousse le coût unitaire vers
les $2$ euros incompressibles de matière --- des économies d'échelle,
avec un plancher infranchissable.

\textbf{11.} Terre~: $a^3 = 1$, $T^2 = 1$. Mars~:
$1.52^3 \approx 3.51$ et $1.88^2 \approx 3.53$. Jupiter~:
$5.20^3 \approx 140.6$ et $11.86^2 \approx 140.7$. La remarque de
Kepler~: $T^2 = a^3$ pour toutes les planètes --- une seule loi pour
tout le ciel.

\textbf{12.} $T = \sqrt{a^3}$~: pour Saturne,
$\sqrt{9.54^3} = \sqrt{868} \approx 29.5$ ans, contre $29.4$
observés~: la loi tient au dixième près.

\textbf{13.} Astéroïde~: $T = \sqrt{4^3} = \sqrt{64} = 8$ ans.
Comète~: $a^3 = 27^2 = 729 = 9^3$, donc $a = 9$ UA.

\textbf{14.} Le carré ($T^2$), le cube ($a^3$) et la racine carrée
(pour extraire $T$). Échelle~: multiplier $a$ par $4$ multiplie $T$
par $4\sqrt4 = 8$ --- ce que l'astéroïde confirme ($a$ quatre fois
celle de la Terre, $T$ huit fois).

\textbf{15.} Un tableau de nombres, confronté aux allures de quelques
\omterm{def:g10:functions:function}{fonctions} de référence, a livré une loi de l'univers soixante-dix ans
avant que quiconque sache \emph{pourquoi} elle tenait --- reconnaître
la \omterm{def:g10:functions:function}{fonction}, c'est déjà la moitié de la science.

\textbf{16.} Sur $\intoo{0}{1}$~: $x^2 < x < \sqrt x$~; sur
$\intoo{1}{+\infty}$~: $\sqrt x < x < x^2$. Vérifications~: en
$x = 0.25$, $0.0625 < 0.25 < 0.5$~; en $x = 4$, $2 < 4 < 16$.

\textbf{17.} Pour $x \geq 0$, les deux membres sont $\geq 0$, donc
$\sqrt x > x \iff x > x^2 \iff x(1 - x) > 0 \iff
x \in \intoo{0}{1}$.

\textbf{18.} En $x = 0.5$~: $x^3 = 0.125 < x^2 = 0.25 < x = 0.5$. En
$x = 2$~: $2 < 4 < 8$~: l'ordre s'inverse. Croisements de $x^2$ et
$x^3$~: $x^3 - x^2 = x^2(x - 1) = 0$, en $x = 0$ et $x = 1$.

\textbf{19.} $10t^2 = 100\sqrt t$ donne $t^2 = 10\sqrt t$~; en
élevant au carré, $t^4 = 100t$, donc $t^3 = 100$ et
$t = \sqrt[3]{100} \approx 4.64$ ans --- environ $4$ ans et $8$ mois.
(Les deux plans rapportent alors à peu près $215$ euros.) Morale~:
les racines partent vite, les puissances gagnent tous les marathons.

\textbf{20.} Freinage~: le \emph{redressement} de la parabole, c'est
le danger de la vitesse. Levier~: l'\emph{asymptote verticale} en $0$
et la longue queue font l'avantage du mécanicien. Lampe~: le $d^2$ de
la \emph{sphère qui s'étale} règle l'affaiblissement. Kepler~:
l'\emph{exposant} $\frac32$ est le tempo du système solaire. Plan
d'épargne A~: l'\emph{aplatissement} de la racine est le sort du
petit épargnant. Cinq courbes, cinq lois~: l'alphabet de la nature,
en effet.
\end{solution}
