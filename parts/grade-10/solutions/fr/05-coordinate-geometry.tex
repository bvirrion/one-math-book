\section*{Chapitre \ref{ch:g10:coordgeom} --- Géométrie repérée}

\begin{solution}{exo:g10:coordgeom:1}
\omterm{prop:g10:coordgeom:midpoint}{Milieu}~: $\left(\frac{2 + (-4)}{2}, \frac{5 + 1}{2}\right) = (-1, 3)$.
Distance~:
\[
AB = \sqrt{(-4 - 2)^2 + (1 - 5)^2} = \sqrt{36 + 16} = \sqrt{52}
= 2\sqrt{13}.
\]
\end{solution}

\begin{solution}{exo:g10:coordgeom:2}
$I$ est le \omterm{prop:g10:coordgeom:midpoint}{milieu} de $[AB]$ signifie $\frac{3 + x_B}{2} = 1$ et
$\frac{-2 + y_B}{2} = 2$, donc $x_B = -1$ et $y_B = 6$~: $B(-1, 6)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:3}
$OP = \sqrt{16 + 1} = \sqrt{17}$~; $OQ = \sqrt{9 + 4} = \sqrt{13}$~;
$OR = \sqrt{0 + 25} = 5 = \sqrt{25}$. Comme $13 < 17 < 25$, la racine
carrée étant \omterm{def:g10:functions:variations}{croissante}, $Q$ est le plus proche de l'origine.
\end{solution}

\begin{solution}{exo:g10:coordgeom:4}
Longueurs au carré~:
\begin{align*}
AB^2 &= (5-1)^2 + (4-2)^2 = 16 + 4 = 20, \\
BC^2 &= (7-5)^2 + (0-4)^2 = 4 + 16 = 20, \\
AC^2 &= (7-1)^2 + (0-2)^2 = 36 + 4 = 40 .
\end{align*}
$AB = BC$~: le triangle est isocèle en $B$. De plus
$AB^2 + BC^2 = 40 = AC^2$~: par la réciproque du théorème de Pythagore
il est aussi rectangle en $B$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:5}
\emph{1.} \omterm{prop:g10:coordgeom:midpoint}{Milieu} de $[AC]$~:
$\left(\frac{-2+4}{2}, \frac{1+1}{2}\right) = (1, 1)$~; \omterm{prop:g10:coordgeom:midpoint}{milieu} de
$[BD]$~: $\left(\frac{1+1}{2}, \frac{3+(-1)}{2}\right) = (1, 1)$. Même
\omterm{prop:g10:coordgeom:midpoint}{milieu}~: $ABCD$ est un parallélogramme.

\emph{2.} $AB^2 = 3^2 + 2^2 = 13$ et $BC^2 = 3^2 + (-4)^2 = 25$~:
$AB \neq BC$, ce n'est pas un losange.

\emph{3.} $AC^2 = 6^2 + 0^2 = 36$ et $BD^2 = 0^2 + (-4)^2 = 16$~:
$AC \neq BD$, ce n'est pas un rectangle. $ABCD$ est un simple
parallélogramme.
\end{solution}

\begin{solution}{exo:g10:coordgeom:6}
Calculer les distances au carré à $\Omega(2,1)$ et comparer à
$5^2 = 25$~:
\[
\Omega A^2 = 3^2 + 4^2 = 25, \qquad
\Omega B^2 = (-3)^2 + (-4)^2 = 25, \qquad
\Omega C^2 = 4^2 + 1^2 = 17 .
\]
$A$ et $B$ sont sur le cercle~; $C$ est à l'intérieur ($17 < 25$).
\end{solution}

\begin{solution}{exo:g10:coordgeom:7}
$MA^2 = (x - 1)^2 + 1$ et $MB^2 = (x - 7)^2 + 25$. En posant
$MA^2 = MB^2$~:
\[
x^2 - 2x + 2 = x^2 - 14x + 74
\ \Longleftrightarrow\
12x = 72
\ \Longleftrightarrow\
x = 6 .
\]
L'unique tel point est $M(6, 0)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:8}
\emph{1.} $I = \left(\frac{0+4}{2}, \frac{3+1}{2}\right) = (2, 2)$.

\emph{2.} $M(2,2)$ est bien $I$. Et
$MA^2 = (0-2)^2 + (3-2)^2 = 5$, $MB^2 = (4-2)^2 + (1-2)^2 = 5$~: le
\omterm{prop:g10:coordgeom:midpoint}{milieu} est équidistant des deux extrémités, comme attendu.

\emph{3.} Chercher $N(0, y)$ avec $NA^2 = NB^2$~:
\[
(3 - y)^2 = 16 + (1 - y)^2
\ \Longleftrightarrow\
9 - 6y = 16 + 1 - 2y
\ \Longleftrightarrow\
-4y = 8,
\]
donc $y = -2$~: le point $N(0, -2)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:9}
\begin{align*}
AB &= \sqrt{4^2 + 2^2} = \sqrt{20} = 2\sqrt5, \\
BC &= \sqrt{8^2 + 4^2} = \sqrt{80} = 4\sqrt5, \\
AC &= \sqrt{12^2 + 6^2} = \sqrt{180} = 6\sqrt5 .
\end{align*}
Alors $AB + BC = 2\sqrt5 + 4\sqrt5 = 6\sqrt5 = AC$~: l'inégalité
triangulaire est une égalité, donc $B$ est sur le segment $[AC]$ --- les
trois points sont alignés.
\end{solution}

\begin{solution}{exo:g10:coordgeom:10}
Le \omterm{prop:g10:coordgeom:midpoint}{milieu} de $[AB]$ est $I\left(\frac a2, \frac b2\right)$. Alors
\[
OI^2 = \frac{a^2}{4} + \frac{b^2}{4},
\qquad
IA^2 = \left(a - \frac a2\right)^2 + \left(0 - \frac b2\right)^2
= \frac{a^2}{4} + \frac{b^2}{4},
\]
et $IB^2 = \left(0 - \frac a2\right)^2 + \left(b - \frac b2\right)^2 =
\frac{a^2}{4} + \frac{b^2}{4}$ aussi. Les trois distances au carré sont
égales, donc $OI = IA = IB$~: le \omterm{prop:g10:coordgeom:midpoint}{milieu} de l'hypoténuse du triangle
rectangle $OAB$ est équidistant des trois sommets (c'est le centre de
son cercle circonscrit).
\end{solution}

\begin{solution}{pb:g10:coordgeom:1}
\textbf{1.} $\Omega M = 5 \iff \Omega M^2 = 25 \iff
(x - 2)^2 + (y - 1)^2 = 25$ par la formule de distance
(\cref{thm:g10:coordgeom:distance}) --- élever au carré est sans
danger, les deux membres étant positifs.

\textbf{2.} $A$~: $(5-2)^2 + (5-1)^2 = 9 + 16 = 25$~: sur le cercle.
$B$~: $16 + 9 = 25$~: sur le cercle également. $D$~:
$4 + 9 = 13 < 25$~: à l'intérieur.

\textbf{3.} $x^2 - 6x = (x - 3)^2 - 9$ et $y^2 + 4y = (y + 2)^2 - 4$,
donc l'\omterm{def:g10:algebra:equation}{équation} s'écrit $(x - 3)^2 + (y + 2)^2 = 25$~: centre
$(3, -2)$, rayon $5$.

\textbf{4.} $(x - 3)^2 + (y + 2)^2 = -1$~: une somme de carrés n'est
jamais négative --- aucun point ne convient~: l'ensemble vide. En
général, $(x - h)^2 + (y - k)^2 = c$ est un cercle de rayon $\sqrt c$
si $c > 0$, le point unique $(h, k)$ si $c = 0$, et l'ensemble vide
si $c < 0$.

\textbf{5.} Avec $y = 3$~: $(x - 3)^2 + 25 = 25$, donc
$(x - 3)^2 = 0$~: le point unique $(3, 3)$. Un seul point de
contact~: la droite est \emph{tangente} au cercle.

\textbf{6.} $MA^2 + MB^2 = (x + r)^2 + y^2 + (x - r)^2 + y^2
= 2x^2 + 2y^2 + 2r^2 = 2r^2 + 2r^2 = 4r^2$ (en utilisant
$x^2 + y^2 = r^2$). Comme $AB^2 = (2r)^2 = 4r^2$, le triangle $AMB$
vérifie $MA^2 + MB^2 = AB^2$~: il est rectangle en $M$ (réciproque de
Pythagore) --- trois lignes, comme promis.

\textbf{7.} $MA^2 + MB^2 = (x + a)^2 + y^2 + (x - a)^2 + y^2
= 2x^2 + 2y^2 + 2a^2 = 2\,MI^2 + 2a^2$, et
$\frac{AB^2}{2} = \frac{(2a)^2}{2} = 2a^2$. Identité démontrée.

\textbf{8.} $2\,MI^2 + \frac{36}{2} = 26$ donne $MI^2 = 4$~: le
cercle de centre le \omterm{prop:g10:coordgeom:midpoint}{milieu} $I$ et de rayon $2$.

\textbf{9.} $MA^2 = 4\,MB^2$~:
$x^2 + y^2 = 4\left((x - 3)^2 + y^2\right)$, c'est-à-dire
$3x^2 + 3y^2 - 24x + 36 = 0$, ou encore
$x^2 + y^2 - 8x + 12 = 0$~; en complétant les carrés,
$(x - 4)^2 + y^2 = 4$ --- le cercle de centre $(4, 0)$ et de rayon
$2$ (le \emph{cercle d'Apollonius} de rapport $2$).

\textbf{10.} Le pont convertit les constructions en calculs~: plus de
rectangle auxiliaire, plus de discussion de cas --- un
développement et le théorème tombe. Le prix à payer~: le calcul porte
moins d'intuition géométrique~; la démonstration classique montre
\emph{pourquoi}, la démonstration algébrique montre \emph{que} --- un
mathématicien de métier garde les deux.

\textbf{11.} $x^2 + y^2 = 25$ et $(x - 6)^2 + y^2 = 25$. Par
soustraction~: $x^2 - (x - 6)^2 = 0$, c'est-à-dire $12x - 36 = 0$,
donc $x = 3$. Puis $9 + y^2 = 25$~: $y = \pm 4$. Candidates
$(3, 4)$ et $(3, -4)$.

\textbf{12.} De $(3, 4)$ à $R(0, 8)$~: $\sqrt{9 + 16} = 5$ --- cela
concorde. De $(3, -4)$~: $\sqrt{9 + 144} \approx 12.4$ --- faux. Vous
êtes en $(3, 4)$.

\textbf{13.} Les deux \omterm{def:g10:algebra:equation}{équations} portent le même $x^2 + y^2$ (ou avec
des coefficients égaux après développement)~: la soustraction élimine
les carrés et laisse une \omterm{def:g10:algebra:equation}{équation} du premier degré --- une droite.
Lorsque les cercles se coupent en deux points, cette droite passe par
les deux points d'\omterm{def:g10:numbers:interunion}{intersection}~: c'est la droite de la corde commune.

\textbf{14.} Quatre inconnues~: trois \omterm{def:g10:coordgeom:system}{coordonnées}, plus l'erreur
d'horloge du récepteur (une montre bon marché, corrigée par les
mathématiques). Chaque satellite fournit une \omterm{def:g10:algebra:equation}{équation}, donc quatre
satellites donnent quatre \omterm{def:g10:algebra:equation}{équations} pour quatre inconnues --- des
satellites supplémentaires améliorant encore la précision.

\textbf{15.} La soustraction donne maintenant
$x = \frac{5.1^2 - 5^2 + 36}{12} = \frac{37.01}{12} \approx 3.084$~:
une erreur de $10$ cm sur une distance a déplacé l'estimation d'environ
$8$ cm ici. Les erreurs se propagent à travers chaque formule --- la
prudence de l'arithmétique des \omterm{def:g10:numbers:interval}{intervalles} du
\cref{pb:g10:numbers:1}, maintenant avec des \omterm{def:g10:coordgeom:system}{coordonnées}.

\textbf{16.} $B' (7, -3)$. La droite $(AB')$ a pour \omterm{def:g10:reffunc:affine}{coefficient
directeur} $\frac{-3 - 5}{7 - 1} = -\frac43$ et coupe $y = 0$ en
$x = 1 + \frac{5}{8} \times 6 = 4.75$~: on boit en
$P(4.75,\ 0)$. Longueur minimale~: $AP + PB = AP + PB' = AB' =
\sqrt{6^2 + 8^2} = 10$.

\textbf{17.} En passant par $Q(3, 0)$~:
$AQ + QB = \sqrt{4 + 25} + \sqrt{16 + 9} = \sqrt{29} + 5
\approx 10.39 > 10$. La ligne droite obtenue par symétrie l'emporte.

\textbf{18.} Milieux des diagonales~: $[AC]$~: $(3, 3)$~; $[BD]$~:
$(3, 3)$ --- égaux, donc parallélogramme. Côtés~:
$AB = \sqrt{10} = AD$~: losange. Diagonales~:
$AC = \sqrt{32} \neq BD = \sqrt{8}$~: ce n'est pas un rectangle, donc
pas un carré. Verdict~: un losange, et rien de plus.

\textbf{19.} $D = A + C - B = (-1 + 6 - 3,\ 2 + 0 - 4) = (2, -2)$~:
$[AC]$ et $[BD]$ ont alors le même \omterm{prop:g10:coordgeom:midpoint}{milieu}
$\left(\frac52, 1\right)$.

\textbf{20.} Formule du \omterm{prop:g10:coordgeom:midpoint}{milieu}~: parallélogrammes et centres
(questions 18 et 19). Formule de distance~: cercles, lieux
géométriques, natures de triangles, plus courts chemins (questions 1
à 9 et 16). Soustraction d'\omterm{def:g10:algebra:equation}{équations} de cercles~: la droite qui
localise une position (questions 11 à 15). Ce qui manque encore à la
trousse~: une algèbre propre des \emph{directions} --- les vecteurs
--- et des \emph{\omterm{def:g10:reffunc:affine}{coefficients directeurs}} --- \omterm{def:g10:algebra:equation}{équations} de droites et
systèmes~: les deux chapitres suivants fournissent exactement cela.
\end{solution}
