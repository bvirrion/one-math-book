\section*{Hoofdstuk \ref{ch:g10:reffunc} --- Referentiefuncties}

\begin{solution}{exo:g10:reffunc:1}
$f$: \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $3$, $y$-afsnede $-2$, \omterm{def:g10:functions:variations}{stijgend}. \quad
$g$: \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $-\frac12$, $y$-afsnede $4$, \omterm{def:g10:functions:variations}{dalend}. \quad
$h$: \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $0$, $y$-afsnede $7$, constant. \quad
$k$: \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $-1$, $y$-afsnede $0$, \omterm{def:g10:functions:variations}{dalend} (en lineair).
\end{solution}

\begin{solution}{exo:g10:reffunc:2}
Door $(2,1)$ en $(5,10)$: \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt}
$m = \frac{10 - 1}{5 - 2} = 3$; daarna geeft $1 = 3 \times 2 + p$ dat
$p = -5$: $f(x) = 3x - 5$.

Door $(-1,4)$ en $(3,-4)$: \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt}
$m = \frac{-4 - 4}{3 - (-1)} = \frac{-8}{4} = -2$; daarna geeft
$4 = -2 \times (-1) + p$ dat $p = 2$: $f(x) = -2x + 2$.
\end{solution}

\begin{solution}{exo:g10:reffunc:3}
$(3.1)^2 < (3.2)^2$: het kwadraat stijgt op de positieve getallen.

$(-2.7)^2 < (-2.8)^2$: het kwadraat daalt op de negatieve getallen en
$-2.8 < -2.7$.

$\frac{1}{5.1} > \frac{1}{5.2}$: het omgekeerde daalt op de positieve
getallen.

$\sqrt{17} > \sqrt{15}$: de vierkantswortel stijgt.
\end{solution}

\begin{solution}{exo:g10:reffunc:4}
$x^2 = 16$: twee oplossingen, $x = 4$ en $x = -4$ (de horizontale rechte
$y = 16$ snijdt de parabool twee keer).

$x^2 \leq 16$: de parabool ligt tussen de twee snijpunten onder de rechte:
$x \in \intcc{-4}{4}$.

$x^2 > 9$: de parabool ligt buiten $\intcc{-3}{3}$ strikt boven $y = 9$:
$x \in \intoo{-\infty}{-3} \cup \intoo{3}{+\infty}$.
\end{solution}

\begin{solution}{exo:g10:reffunc:5}
Voor $0 < x < 1$ geeft $x < 1$ herhaaldelijk met $x$ vermenigvuldigen dat
$x^3 < x^2 < x$, en \cref{prop:g10:reffunc:compare} geeft $x < \sqrt x$, dus
\[
x^3 < x^2 < x < \sqrt x .
\]
Controle met $x = 0.25$: $x^3 = 0.015625$, $x^2 = 0.0625$, $x = 0.25$,
$\sqrt x = 0.5$.
\end{solution}

\begin{solution}{exo:g10:reffunc:6}
$\frac1x = 2$ heeft de enige oplossing $x = \frac12$ (de hyperbool ontmoet
de horizontale rechte $y = 2$ één keer, op de positieve tak).

Voor $x > 0$: $\frac1x < 2$. Vermenigvuldigen met $x > 0$ geeft $1 < 2x$,
dus $x > \frac12$: oplossingsverzameling $\intoo{\frac12}{+\infty}$.

Laat je ook $x < 0$ toe, dan voldoet elke negatieve $x$ aan
$\frac1x < 0 < 2$, zodat de volledige oplossingsverzameling
$\intoo{-\infty}{0} \cup \intoo{\frac12}{+\infty}$ is.
\end{solution}

\begin{solution}{exo:g10:reffunc:7}
(a) Op $\intcc{2}{5}$ zijn alle invoeren positief en stijgt het kwadrateren:
$4 \leq x^2 \leq 25$.

(b) Op $\intcc{-3}{-1}$ daalt het kwadrateren: het grootste kwadraat komt
van $-3$: $1 \leq x^2 \leq 9$.

(c) Op $\intcc{-2}{3}$ bevat het \omterm{def:g10:numbers:interval}{interval} $0$, waar het kwadraat zijn
\omterm{def:g10:functions:extrema}{minimum} $0$ bereikt; het \omterm{def:g10:functions:extrema}{maximum} is de grootste van $(-2)^2 = 4$ en
$3^2 = 9$. Dus $0 \leq x^2 \leq 9$.
\end{solution}

\begin{solution}{exo:g10:reffunc:8}
Eerste bedrijf: $f(x) = 1.5x + 4$; tweede: $g(x) = 2x$. Het eerste is
goedkoper wanneer
\[
1.5x + 4 < 2x
\ \Longleftrightarrow\
4 < 0.5x
\ \Longleftrightarrow\
x > 8 .
\]
Voorbij $8$ km is het eerste bedrijf goedkoper; op precies $8$ km rekenen
beide $16$.
\end{solution}

\begin{solution}{exo:g10:reffunc:9}
Beide leden zijn niet-negatief, dus is de ongelijkheid gelijkwaardig met die
tussen hun kwadraten:
\[
\left(\sqrt{a+b}\right)^2 = a + b
\qquad\text{en}\qquad
\left(\sqrt a + \sqrt b\right)^2 = a + 2\sqrt a \sqrt b + b .
\]
Omdat $2\sqrt a\sqrt b \geq 0$, is het tweede kwadraat minstens het eerste,
wat $\sqrt{a+b} \leq \sqrt a + \sqrt b$ bewijst. Gelijkheid treedt precies op
wanneer $\sqrt a \sqrt b = 0$, dus wanneer $a = 0$ of $b = 0$.
\end{solution}

\begin{solution}{exo:g10:reffunc:10}
\emph{1.} Breng het rechterlid op één noemer:
\[
2 + \frac{3}{x-1} = \frac{2(x-1) + 3}{x - 1} = \frac{2x + 1}{x - 1}
= f(x).
\]

\emph{2.} Op $\intoo{1}{+\infty}$ groeit $x - 1$ mee met $x$ en blijft het
positief, dus daalt $\frac{3}{x-1}$ (de omgekeerde \omterm{def:g10:functions:function}{functie}, maal de
positieve constante $3$), en daalt dus ook $f(x) = 2 + \frac{3}{x-1}$: $f$
is strikt \omterm{def:g10:functions:variations}{dalend} op $\intoo{1}{+\infty}$.
\end{solution}

\begin{solution}{pb:g10:reffunc:1}
\textbf{1.} $d(30) = 3^2 = 9$ m; $d(50) = 25$ m; $d(90) = 81$ m;
$d(130) = 169$ m.

\textbf{2.} $v$ verdubbelen verdubbelt $\frac v{10}$ en vermenigvuldigt het
kwadraat ervan met $4$: $d(100) = 100$ m $= 4 \times d(50)$. Dubbele
snelheid, viervoudige afstand --- het schaalgedrag van de kwadraatfunctie.

\textbf{3.} $\left(\frac{v}{10}\right)^2 = 50$ geeft
$\frac{v}{10} = \sqrt{50}$, dus $v = 10\sqrt{50} \approx 71$ km/u. De
vierkantswortel gaf het antwoord; de aflezing is ondubbelzinnig omdat
snelheden positief zijn en de kwadraatfunctie op $\intco{0}{+\infty}$ strikt
klimt (\cref{prop:g10:reffunc:sqrt}: één positief \omterm{def:g10:functions:function}{origineel}).

\textbf{4.} $d(31) - d(30) = 9.61 - 9 = 0.61$ m, terwijl
$d(91) - d(90) = 82.81 - 81 = 1.81$ m: dezelfde $+1$ km/u kost bij hoge
snelheid drie keer meer afstand. De parabool stijgt niet alleen, ze wordt
\emph{steiler} --- haar klimtempo groeit met $x$.

\textbf{5.} $D(50) = 14 + 25 = 39$ m; $D(130) = 36.4 + 169 = 205.4$ m. In de
stad weegt de affiene term (de reactie) het zwaarst; op de snelweg
verplettert de kwadratische term hem --- affien groeit gestaag, kwadraten
lopen weg.

\textbf{6.} $F(0.5) = 240$, $F(1) = 120$, $F(3) = 40$.

\textbf{7.} De omgekeerde \omterm{def:g10:functions:function}{functie} daalt: hoe verder van het draaipunt, hoe
kleiner de nodige kracht --- met een lange genoeg hefboomarm ($d$ enorm)
houdt elke kracht, hoe klein ook, elke last in evenwicht: de grootspraak van
Archimedes leeft in de staart van de hyperbool, die $0$ nadert zonder het te
bereiken. En $d = 0$ is de verboden waarde: geen arm, geen hefboom --- de
verticale asymptoot.

\textbf{8.} $\frac{100}{4} = 25$ op $2$ m, $\frac{100}{9} \approx 11$ op
$3$ m, $1$ op $10$ m. Op afstand $d$ heeft het licht zich verspreid over een
bol met oppervlakte $4\pi d^2$ (de weekendopgave over de grafsteen van
Archimedes in het onderbouwvolume): dezelfde energie gedeeld door een
oppervlakte die als $d^2$ groeit --- vandaar het omgekeerde \emph{kwadraat}.

\textbf{9.} $60$, $30$, $20$ euro per hoofd voor $n = 10, 20, 30$. Uit
$\frac{600}{n} \leq 25$ volgt $n \geq 24$ deelnemers. De hyperbool belooft
almaar goedkopere aandelen naarmate $n$ groeit --- en weigert ooit $0$ te
bereiken: de bus wordt nooit gratis.

\textbf{10.} $a(x) = \frac{2x + 3}{x} = 2 + \frac3x$: \omterm{def:g10:functions:variations}{dalend} op
$\intoo{0}{+\infty}$ (de verschoven omgekeerde \omterm{def:g10:functions:function}{functie}), naderend tot de
asymptoot $y = 2$. Economisch: de vaste $3$ euro over meer affiches
spreiden duwt de kostprijs per stuk omlaag naar de onsamendrukbare $2$ euro
materiaal --- schaalvoordeel, met een harde ondergrens.

\textbf{11.} Aarde: $a^3 = 1$, $T^2 = 1$. Mars: $1.52^3 \approx 3.51$ en
$1.88^2 \approx 3.53$. Jupiter: $5.20^3 \approx 140.6$ en
$11.86^2 \approx 140.7$. Wat Kepler opmerkte: $T^2 = a^3$ voor elke planeet
--- één wet voor de hele hemel.

\textbf{12.} $T = \sqrt{a^3}$: voor Saturnus
$\sqrt{9.54^3} = \sqrt{868} \approx 29.5$ jaar, tegenover de waargenomen
$29.4$: de wet klopt op een tiende na.

\textbf{13.} Planetoïde: $T = \sqrt{4^3} = \sqrt{64} = 8$ jaar. Komeet:
$a^3 = 27^2 = 729 = 9^3$, dus $a = 9$ AE.

\textbf{14.} Het kwadraat ($T^2$), de derde macht ($a^3$) en de
vierkantswortel (om $T$ vrij te maken). Schaalregel: $a$ met $4$
vermenigvuldigen vermenigvuldigt $T$ met $4\sqrt4 = 8$ --- zoals de
planetoïde bevestigt ($a$ vier keer die van de Aarde, $T$ acht keer).

\textbf{15.} Een tabel met getallen, gelegd naast de vormen van een paar
referentiefuncties, leverde een wet van het heelal op, zeventig jaar voordat
iemand wist \emph{waarom} ze gold --- de \omterm{def:g10:functions:function}{functie} herkennen is de helft van
de wetenschap.

\textbf{16.} Op $\intoo{0}{1}$: $x^2 < x < \sqrt x$; op
$\intoo{1}{+\infty}$: $\sqrt x < x < x^2$. Controles: in $x = 0.25$ is
$0.0625 < 0.25 < 0.5$; in $x = 4$ is $2 < 4 < 16$.

\textbf{17.} Voor $x \geq 0$ zijn beide leden $\geq 0$, dus
$\sqrt x > x \iff x > x^2 \iff x(1 - x) > 0 \iff x \in \intoo{0}{1}$.

\textbf{18.} In $x = 0.5$: $x^3 = 0.125 < x^2 = 0.25 < x = 0.5$. In $x = 2$:
$2 < 4 < 8$: de volgorde keert om. Snijpunten van $x^2$ en $x^3$:
$x^3 - x^2 = x^2(x - 1) = 0$: in $x = 0$ en $x = 1$.

\textbf{19.} $10t^2 = 100\sqrt t$ geeft $t^2 = 10\sqrt t$; kwadrateren geeft
$t^4 = 100t$, dus $t^3 = 100$ en $t = \sqrt[3]{100} \approx 4.64$ jaar ---
ongeveer $4$ jaar en $8$ maanden. (Beide plannen keren dan ongeveer $215$
euro uit.) Moraal: wortels spurten vroeg, machten winnen elke marathon.

\textbf{20.} Remmen: het \emph{steiler worden} van de parabool is het gevaar
van snelheid. Hefboom: de \emph{verticale asymptoot} in $0$ en de lange
staart zijn het voordeel van de mecanicien. Lamp: de $d^2$ van de
\emph{uitdijende bol} bepaalt het uitdoven. Kepler: de \emph{exponent}
$\frac32$ is het tempo van het zonnestelsel. Plan A: het \emph{afvlakken}
van de wortel is het lot van de trage spaarder. Vijf \omterm{def:g10:functions:graph}{grafieken}, vijf wetten:
het alfabet van de natuur, inderdaad.
\end{solution}
