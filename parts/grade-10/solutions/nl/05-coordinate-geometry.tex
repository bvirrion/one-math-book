\section*{Hoofdstuk \ref{ch:g10:coordgeom} --- Analytische meetkunde}

\begin{solution}{exo:g10:coordgeom:1}
\omterm{prop:g10:coordgeom:midpoint}{Midden}: $\left(\frac{2 + (-4)}{2}, \frac{5 + 1}{2}\right) = (-1, 3)$.
Afstand:
\[
AB = \sqrt{(-4 - 2)^2 + (1 - 5)^2} = \sqrt{36 + 16} = \sqrt{52}
= 2\sqrt{13}.
\]
\end{solution}

\begin{solution}{exo:g10:coordgeom:2}
Dat $I$ het \omterm{prop:g10:coordgeom:midpoint}{midden} van $[AB]$ is, betekent $\frac{3 + x_B}{2} = 1$ en
$\frac{-2 + y_B}{2} = 2$, dus $x_B = -1$ en $y_B = 6$: $B(-1, 6)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:3}
$OP = \sqrt{16 + 1} = \sqrt{17}$; $OQ = \sqrt{9 + 4} = \sqrt{13}$;
$OR = \sqrt{0 + 25} = 5 = \sqrt{25}$. Omdat $13 < 17 < 25$ en de
vierkantswortel \omterm{def:g10:functions:variations}{stijgend} is, ligt $Q$ het dichtst bij de oorsprong.
\end{solution}

\begin{solution}{exo:g10:coordgeom:4}
Kwadraten van de lengten:
\begin{align*}
AB^2 &= (5-1)^2 + (4-2)^2 = 16 + 4 = 20, \\
BC^2 &= (7-5)^2 + (0-4)^2 = 4 + 16 = 20, \\
AC^2 &= (7-1)^2 + (0-2)^2 = 36 + 4 = 40 .
\end{align*}
$AB = BC$: de driehoek is gelijkbenig in $B$. Bovendien is
$AB^2 + BC^2 = 40 = AC^2$: volgens de omgekeerde stelling van Pythagoras is
hij ook rechthoekig in $B$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:5}
\emph{1.} \omterm{prop:g10:coordgeom:midpoint}{Midden} van $[AC]$: $\left(\frac{-2+4}{2}, \frac{1+1}{2}\right) =
(1, 1)$; \omterm{prop:g10:coordgeom:midpoint}{midden} van $[BD]$: $\left(\frac{1+1}{2},
\frac{3+(-1)}{2}\right) = (1, 1)$. Zelfde \omterm{prop:g10:coordgeom:midpoint}{midden}: $ABCD$ is een
parallellogram.

\emph{2.} $AB^2 = 3^2 + 2^2 = 13$ en $BC^2 = 3^2 + (-4)^2 = 25$:
$AB \neq BC$, dus geen ruit.

\emph{3.} $AC^2 = 6^2 + 0^2 = 36$ en $BD^2 = 0^2 + (-4)^2 = 16$:
$AC \neq BD$, dus geen rechthoek. $ABCD$ is een gewoon parallellogram.
\end{solution}

\begin{solution}{exo:g10:coordgeom:6}
Bereken de kwadraten van de afstanden tot $\Omega(2,1)$ en vergelijk met
$5^2 = 25$:
\[
\Omega A^2 = 3^2 + 4^2 = 25, \qquad
\Omega B^2 = (-3)^2 + (-4)^2 = 25, \qquad
\Omega C^2 = 4^2 + 1^2 = 17 .
\]
$A$ en $B$ liggen op de cirkel; $C$ ligt erbinnen ($17 < 25$).
\end{solution}

\begin{solution}{exo:g10:coordgeom:7}
$MA^2 = (x - 1)^2 + 1$ en $MB^2 = (x - 7)^2 + 25$. Uit $MA^2 = MB^2$ volgt
\[
x^2 - 2x + 2 = x^2 - 14x + 74
\ \Longleftrightarrow\
12x = 72
\ \Longleftrightarrow\
x = 6 .
\]
Het enige zulke punt is $M(6, 0)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:8}
\emph{1.} $I = \left(\frac{0+4}{2}, \frac{3+1}{2}\right) = (2, 2)$.

\emph{2.} $M(2,2)$ is inderdaad $I$. En $MA^2 = (0-2)^2 + (3-2)^2 = 5$,
$MB^2 = (4-2)^2 + (1-2)^2 = 5$: het \omterm{prop:g10:coordgeom:midpoint}{midden} ligt op gelijke afstand van de
twee uiteinden, zoals verwacht.

\emph{3.} Zoek $N(0, y)$ met $NA^2 = NB^2$:
\[
(3 - y)^2 = 16 + (1 - y)^2
\ \Longleftrightarrow\
9 - 6y = 16 + 1 - 2y
\ \Longleftrightarrow\
-4y = 8,
\]
dus $y = -2$: het punt $N(0, -2)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:9}
\begin{align*}
AB &= \sqrt{4^2 + 2^2} = \sqrt{20} = 2\sqrt5, \\
BC &= \sqrt{8^2 + 4^2} = \sqrt{80} = 4\sqrt5, \\
AC &= \sqrt{12^2 + 6^2} = \sqrt{180} = 6\sqrt5 .
\end{align*}
Nu is $AB + BC = 2\sqrt5 + 4\sqrt5 = 6\sqrt5 = AC$: de
driehoeksongelijkheid is een gelijkheid, dus ligt $B$ op het lijnstuk $[AC]$
--- de drie punten liggen op één rechte.
\end{solution}

\begin{solution}{exo:g10:coordgeom:10}
Het \omterm{prop:g10:coordgeom:midpoint}{midden} van $[AB]$ is $I\left(\frac a2, \frac b2\right)$. Dan is
\[
OI^2 = \frac{a^2}{4} + \frac{b^2}{4},
\qquad
IA^2 = \left(a - \frac a2\right)^2 + \left(0 - \frac b2\right)^2
= \frac{a^2}{4} + \frac{b^2}{4},
\]
en eveneens $IB^2 = \left(0 - \frac a2\right)^2 +
\left(b - \frac b2\right)^2 = \frac{a^2}{4} + \frac{b^2}{4}$. De drie
kwadraten zijn gelijk, dus $OI = IA = IB$: het \omterm{prop:g10:coordgeom:midpoint}{midden} van de schuine zijde
van de rechthoekige driehoek $OAB$ ligt op gelijke afstand van alle drie de
hoekpunten (het is het middelpunt van de omgeschreven cirkel).
\end{solution}

\begin{solution}{pb:g10:coordgeom:1}
\textbf{1.} $\Omega M = 5 \iff \Omega M^2 = 25 \iff
(x - 2)^2 + (y - 1)^2 = 25$ volgens de afstandsformule
(\cref{thm:g10:coordgeom:distance}) --- kwadrateren is onschadelijk, want
beide leden zijn positief.

\textbf{2.} $A$: $(5-2)^2 + (5-1)^2 = 9 + 16 = 25$: op de cirkel. $B$:
$16 + 9 = 25$: eveneens erop. $D$: $4 + 9 = 13 < 25$: erbinnen.

\textbf{3.} $x^2 - 6x = (x - 3)^2 - 9$ en $y^2 + 4y = (y + 2)^2 - 4$, zodat
de \omterm{def:g10:algebra:equation}{vergelijking} luidt $(x - 3)^2 + (y + 2)^2 = 25$: middelpunt $(3, -2)$,
straal $5$.

\textbf{4.} $(x - 3)^2 + (y + 2)^2 = -1$: een som van kwadraten is nooit
negatief --- geen enkel punt voldoet eraan: de lege verzameling. Algemeen is
$(x - h)^2 + (y - k)^2 = c$ een cirkel met straal $\sqrt c$ als $c > 0$, het
enkele punt $(h, k)$ als $c = 0$, en leeg als $c < 0$.

\textbf{5.} Met $y = 3$: $(x - 3)^2 + 25 = 25$, dus $(x - 3)^2 = 0$: het
enkele punt $(3, 3)$. Eén raakpunt: de rechte is een \emph{raaklijn} aan de
cirkel.

\textbf{6.} $MA^2 + MB^2 = (x + r)^2 + y^2 + (x - r)^2 + y^2
= 2x^2 + 2y^2 + 2r^2 = 2r^2 + 2r^2 = 4r^2$ (met $x^2 + y^2 = r^2$). Omdat
$AB^2 = (2r)^2 = 4r^2$, voldoet de driehoek $AMB$ aan
$MA^2 + MB^2 = AB^2$: rechthoekig in $M$ (omgekeerde stelling van
Pythagoras) --- drie regels, zoals beloofd.

\textbf{7.} $MA^2 + MB^2 = (x + a)^2 + y^2 + (x - a)^2 + y^2
= 2x^2 + 2y^2 + 2a^2 = 2\,MI^2 + 2a^2$, en
$\frac{AB^2}{2} = \frac{(2a)^2}{2} = 2a^2$. De identiteit is bewezen.

\textbf{8.} $2\,MI^2 + \frac{36}{2} = 26$ geeft $MI^2 = 4$: de cirkel met
als middelpunt het \omterm{prop:g10:coordgeom:midpoint}{midden} $I$ en straal $2$.

\textbf{9.} $MA^2 = 4\,MB^2$: $x^2 + y^2 = 4\left((x - 3)^2 + y^2\right)$,
dus $3x^2 + 3y^2 - 24x + 36 = 0$, of $x^2 + y^2 - 8x + 12 = 0$; aanvullen
geeft $(x - 4)^2 + y^2 = 4$ --- de cirkel met middelpunt $(4, 0)$ en straal
$2$ (de \emph{cirkel van Apollonius} met verhouding $2$).

\textbf{10.} De brug zet constructies om in berekeningen: geen hulprechthoek,
geen gevalsonderscheid --- één uitwerking en de stelling valt eruit. De
prijs: berekeningen dragen minder meetkundig inzicht; het klassieke bewijs
laat zien \emph{waarom}, het algebraïsche \emph{dat} --- een werkende
wiskundige houdt beide bij de hand.

\textbf{11.} $x^2 + y^2 = 25$ en $(x - 6)^2 + y^2 = 25$. Aftrekken geeft
$x^2 - (x - 6)^2 = 0$, dus $12x - 36 = 0$: $x = 3$. Daarna $9 + y^2 = 25$:
$y = \pm 4$. Kandidaten $(3, 4)$ en $(3, -4)$.

\textbf{12.} Van $(3, 4)$ naar $R(0, 8)$: $\sqrt{9 + 16} = 5$ --- dat klopt.
Van $(3, -4)$: $\sqrt{9 + 144} \approx 12.4$ --- fout. Je bevindt je in
$(3, 4)$.

\textbf{13.} Beide \omterm{def:g10:algebra:equation}{vergelijkingen} dragen dezelfde $x^2 + y^2$ (of na
\omterm{def:g10:algebra:expand}{uitwerken} met gelijke coëfficiënten): het aftrekken schrapt de kwadraten en
laat een \omterm{def:g10:algebra:equation}{vergelijking} van de eerste graad over --- een rechte. Snijden de
cirkels elkaar twee keer, dan gaat die rechte door beide snijpunten: ze
draagt de gemeenschappelijke koorde.

\textbf{14.} Vier onbekenden: drie \omterm{def:g10:coordgeom:system}{coördinaten} plus de klokfout van de
ontvanger (een goedkoop uurwerk, door de wiskunde gecorrigeerd). Elke
satelliet levert één \omterm{def:g10:algebra:equation}{vergelijking}, dus geven vier satellieten vier
\omterm{def:g10:algebra:equation}{vergelijkingen} voor vier onbekenden --- en meer satellieten verbeteren de
nauwkeurigheid.

\textbf{15.} Het aftrekken geeft nu
$x = \frac{5.1^2 - 5^2 + 36}{12} = \frac{37.01}{12} \approx 3.084$: een fout
van $10$ cm in één afstand verschoof de schatting hier ongeveer $8$ cm.
Fouten planten zich door elke formule voort --- de waarschuwing over rekenen
met intervallen uit \cref{pb:g10:numbers:1}, nu met \omterm{def:g10:coordgeom:system}{coördinaten}.

\textbf{16.} $B'(7, -3)$. De rechte $(AB')$ heeft \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt}
$\frac{-3 - 5}{7 - 1} = -\frac43$ en kruist $y = 0$ in
$x = 1 + \frac{5}{8} \times 6 = 4.75$: drink in $P(4.75,\ 0)$. Kortste
lengte: $AP + PB = AP + PB' = AB' = \sqrt{6^2 + 8^2} = 10$.

\textbf{17.} Via $Q(3, 0)$: $AQ + QB = \sqrt{4 + 25} + \sqrt{16 + 9} =
\sqrt{29} + 5 \approx 10.39 > 10$. De gespiegelde rechte lijn wint.

\textbf{18.} \omterm{prop:g10:coordgeom:midpoint}{Middens} van de diagonalen: $[AC]$: $(3, 3)$; $[BD]$: $(3, 3)$
--- gelijk: parallellogram. Zijden: $AB = \sqrt{10} = AD$: ruit.
Diagonalen: $AC = \sqrt{32} \neq BD = \sqrt{8}$: geen rechthoek, en dus ook
geen vierkant. Oordeel: een ruit, en niets meer.

\textbf{19.} $D = A + C - B = (-1 + 6 - 3,\ 2 + 0 - 4) = (2, -2)$: dan delen
$[AC]$ en $[BD]$ het \omterm{prop:g10:coordgeom:midpoint}{midden} $\left(\frac52, 1\right)$.

\textbf{20.} Formule voor het \omterm{prop:g10:coordgeom:midpoint}{midden}: parallellogrammen en middelpunten
(vragen 18, 19). Afstandsformule: cirkels, meetkundige plaatsen, de aard van
driehoeken, kortste wegen (vragen 1--9, 16). Aftrekken van
cirkelvergelijkingen: de rechte die een positie vastlegt (vragen 11--15). Wat
nog ontbreekt in de kist: een nette algebra van \emph{richtingen} ---
vectoren --- en van \emph{\omterm{def:g10:reffunc:affine}{richtingscoëfficiënten}} --- \omterm{def:g10:algebra:equation}{vergelijkingen} van
rechten en stelsels: de twee volgende hoofdstukken leveren precies dat.
\end{solution}
