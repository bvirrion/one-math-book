\section*{Hoofdstuk \ref{ch:g10:vectors} --- Vectoren}

\begin{solution}{exo:g10:vectors:1}
$\vect{AB} + \vect{BD} = \vect{AD}$ (Chasles).

$\vect{KL} + \vect{LM} + \vect{MK} = \vect{KK} = \vec 0$.

$\vect{AC} - \vect{BC} = \vect{AC} + \vect{CB} = \vect{AB}$.

$\vect{AB} + \vect{CA} = \vect{CA} + \vect{AB} = \vect{CB}$.
\end{solution}

\begin{solution}{exo:g10:vectors:2}
$\vect{AB}\,(5-2,\ 3-1) = (3, 2)$; \quad
$\vect{BC}\,(-1-5,\ 4-3) = (-6, 1)$; \quad
$\vect{AC}\,(-1-2,\ 4-1) = (-3, 3)$.
Controle: $(3, 2) + (-6, 1) = (-3, 3)$, coördinaat per coördinaat.
\end{solution}

\begin{solution}{exo:g10:vectors:3}
$\vec u + \vec v = (2 + (-1),\ -3 + 4) = (1, 1)$.

$3\vec u = (6, -9)$.

$2\vec u - \vec v = (4 - (-1),\ -6 - 4) = (5, -10)$.

$\norm{\vec u} = \sqrt{2^2 + (-3)^2} = \sqrt{13}$.
\end{solution}

\begin{solution}{exo:g10:vectors:4}
(a) $4 \times 9 - 6 \times 6 = 36 - 36 = 0$: \omterm{def:g10:vectors:collinear}{collineair}
($\vec v = \frac32 \vec u$).

(b) $2 \times 10 - (-5) \times (-4) = 20 - 20 = 0$: \omterm{def:g10:vectors:collinear}{collineair}
($\vec v = -2\vec u$).

(c) $3 \times 3 - 1 \times 1 = 8 \neq 0$: niet \omterm{def:g10:vectors:collinear}{collineair}.
\end{solution}

\begin{solution}{exo:g10:vectors:5}
$\vect{AB}\,(4, -2)$, en $\vect{DC}\,(6 - x_D,\ 3 - y_D)$. De voorwaarde
$\vect{AB} = \vect{DC}$ geeft $6 - x_D = 4$ en $3 - y_D = -2$, dus
$D(2, 5)$.
\end{solution}

\begin{solution}{exo:g10:vectors:6}
\emph{1.} $\vect{AB}\,(3, 2)$ en $\vect{AC}\,(6, -2)$, dus
$\vect{AB} + \vect{AC} = (9, 0)$ en $M = (-2 + 9,\ 1 + 0) = (7, 1)$.

\emph{2.} Volgens de parallellogramregel betekent
$\vect{AM} = \vect{AB} + \vect{AC}$ dat $ABMC$ een parallellogram is ($M$ is
het vierde hoekpunt, tegenover $A$). Ga na: $\vect{AB}\,(3,2)$ en
$\vect{CM}\,(7-4,\ 1-(-1)) = (3, 2)$ zijn gelijk.
\end{solution}

\begin{solution}{exo:g10:vectors:7}
Eerste drietal: $\vect{AB}\,(3, 2)$, $\vect{AC}\,(9, 6)$;
$3 \times 6 - 2 \times 9 = 0$: op één rechte ($\vect{AC} = 3\vect{AB}$).

Tweede drietal: $\vect{AB}\,(2, -1)$, $\vect{AC}\,(8, -4)$;
$2 \times (-4) - (-1) \times 8 = -8 + 8 = 0$: eveneens op één rechte
($\vect{AC} = 4\vect{AB}$).
\end{solution}

\begin{solution}{exo:g10:vectors:8}
$\vect{AB}\,(4, 2)$ en $\vect{DC}\,(6-2,\ 1-(-1)) = (4, 2)$: de twee
\omterm{def:g10:vectors:vector}{vectoren} zijn \emph{gelijk}, dus is $ABCD$ een parallellogram, en in het
bijzonder is $(AB) \parallel (CD)$. De parallellogrameigenschap is de
sterkste: ze houdt de evenwijdigheid (de collineariteit van de \omterm{def:g10:vectors:vector}{vectoren}) in,
niet omgekeerd.
\end{solution}

\begin{solution}{exo:g10:vectors:9}
\emph{1.} $\vect{AB}\,(6, 3)$, dus $\vect{AI} = \frac23\vect{AB} = (4, 2)$
en $I(4, 2)$.

\emph{2.} $\vect{CB}\,(6-2,\ 3-5) = (4, -2)$, dus
$\vect{CJ} = \frac23\vect{CB} = \left(\frac83, -\frac43\right)$ en
$J\left(2 + \frac83,\ 5 - \frac43\right) = \left(\frac{14}{3},
\frac{11}{3}\right)$.

\emph{3.} $\vect{IJ}\,\left(\frac{14}{3} - 4,\ \frac{11}{3} - 2\right) =
\left(\frac23, \frac53\right)$ en $\vect{AC}\,(2, 5)$. Criterium voor
collineariteit:
\[
\frac23 \times 5 - \frac53 \times 2 = \frac{10}{3} - \frac{10}{3} = 0 :
\]
de \omterm{def:g10:vectors:vector}{vectoren} zijn \omterm{def:g10:vectors:collinear}{collineair}, dus is $(IJ)$ evenwijdig met $(AC)$. (Zowel $I$
als $J$ ligt op een derde van de weg vanaf de zijde naar $B$: de kleine
driehoek $IBJ$ is een verkleining van $ABC$.)
\end{solution}

\begin{solution}{exo:g10:vectors:10}
De \omterm{prop:g10:coordgeom:midpoint}{middens} zijn
\[
I\left(\tfrac{x_A + x_B}{2}, \tfrac{y_A + y_B}{2}\right), \quad
J\left(\tfrac{x_B + x_C}{2}, \tfrac{y_B + y_C}{2}\right), \quad
K\left(\tfrac{x_C + x_D}{2}, \tfrac{y_C + y_D}{2}\right), \quad
L\left(\tfrac{x_D + x_A}{2}, \tfrac{y_D + y_A}{2}\right).
\]
De eerste \omterm{def:g10:coordgeom:system}{coördinaten} van de twee \omterm{def:g10:vectors:vector}{vectoren}:
\[
x_J - x_I = \frac{x_B + x_C}{2} - \frac{x_A + x_B}{2}
= \frac{x_C - x_A}{2},
\qquad
x_K - x_L = \frac{x_C + x_D}{2} - \frac{x_D + x_A}{2}
= \frac{x_C - x_A}{2},
\]
en dezelfde berekening werkt voor de tweede \omterm{def:g10:coordgeom:system}{coördinaten}. Dus is
$\vect{IJ} = \vect{LK}$ (beide gelijk aan $\frac12\vect{AC}$), en is $IJKL$
een parallellogram, hoe de vierhoek $ABCD$ er ook uitziet.
\end{solution}

\begin{solution}{pb:g10:vectors:1}
\textbf{1.} $\vect{AB} + \vect{BC} + \vect{CD} = \vect{AD}$; \quad
$\vect{MA} - \vect{MB} = \vect{MA} + \vect{BM} = \vect{BA}$; \quad
$\vect{AB} + \vect{BC} + \vect{CD} + \vect{DA} = \vect{AA} = \vec 0$ (na het
herschikken van de derde \omterm{def:g10:vectors:sum}{som}).

\textbf{2.} $\vect{OB} - \vect{OA} = \vect{AO} + \vect{OB} = \vect{AB}$
volgens Chasles.

\textbf{3.} $I$ is het \omterm{prop:g10:coordgeom:midpoint}{midden} van $[AB]$ precies wanneer
$\vect{IA} = -\vect{IB}$, dus wanneer $\vect{IA} + \vect{IB} = \vec 0$. Dan
is voor elke $O$: $\vec 0 = \vect{IO} + \vect{OA} + \vect{IO} + \vect{OB}$,
dus $2\,\vect{OI} = \vect{OA} + \vect{OB}$ --- en omgekeerd.

\textbf{4.} $\vect{AB} = (4, 2)$, en $D$ voldoet aan
$\vect{DC} = \vect{AB}$: $D = C - (4, 2) = (2, -2)$ --- dezelfde $D$ als in
\cref{pb:g10:coordgeom:1}, één regel sneller.

\textbf{5.} De translatie over de \omterm{def:g10:vectors:vector}{vector} $\vec u$ --- precies de
transformatie die twee evenwijdige spiegels of twee halve draaiingen
voortbrengen. \omterm{def:g10:functions:function}{Beeld} van $(1, 2)$: $(4, 1)$.

\textbf{6.} Werk elke $\vect{GX}$ uit als $\vect{GO} + \vect{OX}$:
$3\,\vect{GO} + \vect{OA} + \vect{OB} + \vect{OC} = \vec 0$, dus
$\vect{OG} = \frac13(\vect{OA} + \vect{OB} + \vect{OC})$: er is precies één
zo'n punt, en zijn \omterm{def:g10:coordgeom:system}{coördinaten} zijn de gemiddelden van die van de drie
hoekpunten.

\textbf{7.} $\vect{AB} + \vect{AC} = (\vect{AG} + \vect{GB}) +
(\vect{AG} + \vect{GC}) = 2\vect{AG} + (\vect{GB} + \vect{GC})
= 2\vect{AG} - \vect{GA} = 3\vect{AG}$, waaruit
$\vect{AG} = \frac13(\vect{AB} + \vect{AC})$. De formule voor het \omterm{prop:g10:coordgeom:midpoint}{midden}
(vraag 3, vanuit $A$ gezien) geeft
$\vect{AK} = \frac12(\vect{AB} + \vect{AC})$. Vergelijken levert
$\vect{AG} = \frac23\vect{AK}$ --- $G$ ligt op de zwaartelijn $[AK]$, op twee
derde van de weg vanaf $A$.

\textbf{8.} De definiërende voorwaarde behandelt $A$, $B$ en $C$ op dezelfde
manier, dus zet dezelfde berekening vanuit $B$ of vanuit $C$ \emph{dezelfde}
$G$ op twee derde van die zwaartelijnen: de concurrentie kost niets. Het
parallellogrambewijs was vernuftig maar op maat gemaakt; de \omterm{def:g10:coordgeom:system}{coördinaten}
rekenen maar verduisteren; de \omterm{def:g10:vectors:vector}{vectoren} maken de symmetrie zichtbaar en het
bewijs twee regels lang.

\textbf{9.} $G\left(\frac{1+5+3}{3}, \frac{1+2+6}{3}\right) = (3, 3)$;
$K(4, 4)$; $\vect{AG} = (2, 2)$ en $\vect{AK} = (3, 3)$: determinant
$2 \times 3 - 2 \times 3 = 0$, dus \omterm{def:g10:vectors:collinear}{collineair}, en inderdaad is
$\vect{AG} = \frac23\vect{AK}$.

\textbf{10.} $G'\left(\frac{2 + 5 + 3}{4}, \frac{2 + 2 + 6}{4}\right) =
(2.5,\ 2.5)$: van $G(3,3)$ weggetrokken naar het verzwaarde hoekpunt
$A(1,1)$, zoals een evenwichtspunt betaamt.

\textbf{11.} $3 \times 4 - (-2)(-6) = 12 - 12 = 0$: \omterm{def:g10:vectors:collinear}{collineair}
($\vec v = -2\vec u$). $2 \times 9 - 5 \times 4 = -2 \neq 0$: niet
\omterm{def:g10:vectors:collinear}{collineair}.

\textbf{12.} $\vect{AB} = (2, 3)$, $\vect{AC} = (6, 9)$: determinant
$2 \times 9 - 3 \times 6 = 0$: op één rechte.

\textbf{13.} $\vect{OI} = \frac12(\vect{OA} + \vect{OB})$ en
$\vect{OK} = \frac12(\vect{OC} + \vect{OD})$, dus heeft het \omterm{prop:g10:coordgeom:midpoint}{midden} van
$[IK]$ als plaatsvector
$\frac12(\vect{OI} + \vect{OK}) = \frac14(\vect{OA} + \vect{OB} +
\vect{OC} + \vect{OD})$ --- en de berekening voor $[JL]$ levert precies
dezelfde \omterm{def:g10:vectors:vector}{vector}. De twee bimedianen (en, volgens Varignon, ook de diagonalen
van het parallellogram $IJKL$) delen dus dit middelpunt: het zwaartepunt van
de vierhoek.

\textbf{14.} $\vect{MN} = \vect{AN} - \vect{AM} = \frac13\vect{AC} -
\frac13\vect{AB} = \frac13\vect{BC}$: dus $(MN) \parallel (BC)$ en
$MN = \frac13 BC$.

\textbf{15.} Met $\vect{AM} = k\,\vect{AB}$ en $\vect{AN} = k\,\vect{AC}$:
$\vect{MN} = k\,\vect{BC}$ --- één regel die de middenparallelstelling
($k = \frac12$) en de stelling van Thales (elke $k$) bevat: de verhouding
komt ongeschonden door het aftrekken.

\textbf{16.} Resultante $(3, 4)$, met grootte $\sqrt{9 + 16} = 5$ km/u: de
boot vaart in werkelijkheid noordoostwaarts, langs de schuine zijde van de
$3$--$4$--$5$-driehoek.

\textbf{17.} $\vec u + \vec v + \vec w = (0, 0)$: de ring blijft liggen ---
evenwicht. Langs een gesloten veelhoek $A_1 A_2 \dots A_n A_1$ schuift
Chasles de \omterm{def:g10:vectors:sum}{som} ineen tot $\vect{A_1 A_1} = \vec 0$: rondgaan brengt je terug.

\textbf{18.} $\vec w = -(\vec u + \vec v) = -(3, 2) = (-3, -2)$.

\textbf{19.} Resultante $(1.5,\ 2)$, met grootte $\sqrt{2.25 + 4} = 2.5$
m/s. De overtocht: $50$ m aan $2$ m/s dwars: $25$ s; de afdrijving:
$1.5 \times 25 = 37.5$ m stroomafwaarts.

\textbf{20.} Dit weekend opnieuw bewezen: de tweederdestelling voor het
zwaartepunt (vraag 7), het middelpunt van de bimedianen en Varignon (vraag
13), en de middenparallelstelling samen met Thales (vragen 14--15). Wat de
pijlen nog niet kunnen: \emph{hoeken} meten (en dus lengten in willekeurige
richtingen) --- het scalair product van jaar 11 is precies het ontbrekende
instrument.
\end{solution}
