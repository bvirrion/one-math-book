\section*{Hoofdstuk \ref{ch:g10:proba} --- Kansrekening en steekproeven}

\begin{solution}{exo:g10:proba:1}
Elk van de $6$ uitkomsten heeft \omterm{def:g10:proba:distribution}{kans} $\frac16$.
``Een $6$ gooien'': $\frac16$.
``Oneven getal'' $= \{1, 3, 5\}$: $\frac36 = \frac12$.
``Hoogstens $4$'' $= \{1,2,3,4\}$: $\frac46 = \frac23$.
``Een $7$ gooien'': onmogelijke \omterm{def:g10:proba:events}{gebeurtenis}, \omterm{def:g10:proba:distribution}{kans} $0$.
\end{solution}

\begin{solution}{exo:g10:proba:2}
Er zijn $10$ even waarschijnlijke knikkers.
$\P(\text{groen}) = \frac{5}{10} = \frac12$.
$\P(\text{niet zwart}) = 1 - \frac{2}{10} = \frac{8}{10} = \frac45$.
Groen en geel zijn \omterm{def:g10:proba:operations}{onverenigbaar}:
$\P(\text{groen of geel}) = \frac{5}{10} + \frac{3}{10} = \frac{8}{10}
= \frac45$.
\end{solution}

\begin{solution}{exo:g10:proba:3}
Zij $S$: ``volgt Spaans'' en $D$: ``volgt Duits''. Dan is
$\P(S) = \frac{18}{30}$, $\P(D) = \frac{10}{30}$ en
$\P(S \cap D) = \frac{4}{30}$. Somregel:
\[
\P(S \cup D) = \frac{18}{30} + \frac{10}{30} - \frac{4}{30}
= \frac{24}{30} = \frac45 .
\]
\omterm{def:g10:proba:operations}{Complement}: $\P(\text{geen van beide}) = 1 - \frac45 = \frac15$.
\end{solution}

\begin{solution}{exo:g10:proba:4}
Het \omterm{met:g10:proba:tree}{boomdiagram} heeft vier paden KK, KM, MK, MM, elk met \omterm{def:g10:proba:distribution}{kans}
$\frac12 \times \frac12 = \frac14$.
Twee keer kop: $\frac14$. Precies één keer kop (KM of MK):
$\frac24 = \frac12$. Minstens één keer kop:
$1 - \P(\text{MM}) = 1 - \frac14 = \frac34$.
\end{solution}

\begin{solution}{exo:g10:proba:5}
Eerste trekking: rood met \omterm{def:g10:proba:distribution}{kans} $\frac{4}{10}$, wit met \omterm{def:g10:proba:distribution}{kans} $\frac{6}{10}$.
Tweede trekking \emph{zonder} teruglegging: na een rode blijven er $3$ rode
en $6$ witte over van de $9$; na een witte $4$ rode en $5$ witte van de $9$.

Twee rode: $\frac{4}{10} \times \frac39 = \frac{12}{90} = \frac{2}{15}$.

Verschillende kleuren: de paden RW en WR:
\[
\frac{4}{10} \times \frac69 + \frac{6}{10} \times \frac49
= \frac{24}{90} + \frac{24}{90} = \frac{48}{90} = \frac{8}{15}.
\]
\end{solution}

\begin{solution}{exo:g10:proba:6}
\emph{1.} Elke dobbelsteen toont elk vlak onafhankelijk met \omterm{def:g10:proba:distribution}{kans} $\frac16$,
dus zijn de $36$ geordende paren even waarschijnlijk; het paar bepaalt de
som.

\emph{2.} Som $7$: de paren $(1,6), (2,5), (3,4), (4,3), (5,2), (6,1)$, dus
$\P = \frac{6}{36} = \frac16$. Som $12$: alleen $(6,6)$,
$\P = \frac{1}{36}$. Som hoogstens $4$: de paren met som $2$, $3$ of $4$:
$(1,1)$; $(1,2), (2,1)$; $(1,3), (2,2), (3,1)$ --- zes paren, dus
$\P = \frac{6}{36} = \frac16$.

\emph{3.} Het aantal paren dat een som bereikt groeit van $1$ (som $2$) naar
$6$ (som $7$) en daalt daarna: de waarschijnlijkste som is $7$.
\end{solution}

\begin{solution}{exo:g10:proba:7}
Elke vraag wordt onafhankelijk met \omterm{def:g10:proba:distribution}{kans} $\frac14$ juist beantwoord.
\omterm{met:g10:proba:tree}{Boomdiagram} met twee niveaus:

Allebei juist: $\frac14 \times \frac14 = \frac{1}{16}$.

Precies één juist: de paden juist--fout en fout--juist:
$\frac14 \times \frac34 + \frac34 \times \frac14 = \frac{6}{16} = \frac38$.

Geen enkel juist: $\frac34 \times \frac34 = \frac{9}{16}$.

Controle: $\frac{1}{16} + \frac{6}{16} + \frac{9}{16} = 1$.
\end{solution}

\begin{solution}{exo:g10:proba:8}
\emph{1.} Waargenomen \omterm{def:g10:stats:series}{frequentie}: $\frac{396}{900} = 0.44$. Met $n = 900$ is
$\frac{1}{\sqrt{900}} = \frac{1}{30} \approx 0.033$, zodat het
schommelingsinterval ongeveer $\intcc{0.467}{0.533}$ is.

\emph{2.} De waargenomen $0.44$ ligt buiten het \omterm{def:g10:numbers:interval}{interval}: steekproeven van
$900$ kiezers uit een bevolking met $50\%$ steun dwalen zelden zo ver af. De
peiling werpt ernstige twijfel op de beweerde $50\%$.
\end{solution}

\begin{solution}{exo:g10:proba:9}
\emph{1.} $\P(\text{win } 6) = \frac16$ (een $6$ gooien),
$\P(\text{win } 3) = \frac16$ (een $5$ gooien),
$\P(\text{win } 0) = \frac46 = \frac23$.

\emph{2.} Over $600$ spellen verwacht je ongeveer $100$ zessen, $100$
vijven en $400$ andere worpen. Winst: ongeveer
$100 \times 6 + 100 \times 3 = 900$. Kosten: $600 \times 2 = 1200$.
Verwacht nettoresultaat: $900 - 1200 = -300$, dus gemiddeld een verlies van
ongeveer $0.5$ per spel --- het spel loont niet.
\end{solution}

\begin{solution}{pb:g10:proba:1}
\textbf{1.} $P(\text{heer of harten}) = \frac{4}{52} + \frac{13}{52} -
\frac{1}{52} = \frac{16}{52} = \frac{4}{13}$: de hartenheer zit in beide
gebeurtenissen en mag niet twee keer geteld worden --- vandaar de aftrekking
(\cref{thm:g10:proba:addition}).

\textbf{2.} $P(\text{geen kop}) = \left(\frac12\right)^3 = \frac18$, dus is
$P(\text{minstens één}) = \frac78$.

\textbf{3.} $P(RR) = \frac35 \times \frac24 = \frac{3}{10}$;
$P(\text{één van elke}) = \frac35 \times \frac24 + \frac25 \times \frac34 =
\frac{6}{20} + \frac{6}{20} = \frac35$.

\textbf{4.} $P(BB) = \frac25 \times \frac14 = \frac1{10}$; en
$\frac{3}{10} + \frac{6}{10} + \frac{1}{10} = 1$: de bladeren van het
\omterm{met:g10:proba:tree}{boomdiagram} tellen altijd op tot $1$.

\textbf{5.} Onverenigbare gebeurtenissen voldoen aan
$P(A \cup B) = P(A) + P(B) = 1.1 > 1$: onmogelijk --- er is \omterm{def:g10:proba:distribution}{kans} verzonnen.

\textbf{6.} (De meeste mensen, de duizend briefschrijvers inbegrepen,
antwoorden ``het maakt niet uit, fiftyfifty''. Houd je antwoord bij voor
vraag 8.)

\textbf{7.} Auto achter deur 1 (\omterm{def:g10:proba:distribution}{kans} $\frac13$): de quizmaster opent een van
beide geitendeuren; wisselen laat de auto staan: \emph{verlies}. Auto achter
deur 2 ($\frac13$): de quizmaster \emph{moet} deur 3 openen; wisselen komt
op deur 2 uit: \emph{winst}. Auto achter deur 3: symmetrisch: \emph{winst}.
Samen: $P(\text{wisselen wint}) = \frac13 + \frac13 = \frac23$.

\textbf{8.} Je eerste keuze is fout met \omterm{def:g10:proba:distribution}{kans} $\frac23$ --- en precies dan
laat de gedwongen onthulling van de quizmaster de auto achter de
overblijvende deur staan: wisselen zet elke beginfout om in een winst.
$\frac23$, wat het onderbuikgevoel ook zei.

\textbf{9.} Zes even waarschijnlijke scenario's: auto achter $1$, $2$ of
$3$, quizmaster opent deur $2$ of $3$ (nooit die van jou). In twee ervan
toont de onthulling de auto (auto@2/opent 2, auto@3/opent 3): weggegooid.
Onder de vier overblijvende --- auto@1/opent 2, auto@1/opent 3, auto@2/opent
3, auto@3/opent 2 --- wint wisselen in de laatste twee:
$\frac24 = \frac12$. Bij een onwetende quizmaster \emph{is} fiftyfifty juist:
de paradox schuilt volledig in de kennis van de quizmaster, die de
scenario's ongelijk filtert.

\textbf{10.} Jouw keuze verbergt de auto met \omterm{def:g10:proba:distribution}{kans} $\frac{1}{100}$; de $98$
onthullingen van de wetende quizmaster trechteren de overblijvende
$\frac{99}{100}$ naar die ene gesloten deur. Wissel, en win $99$ keer op
$100$.

\textbf{11.} Drie kaarten (één aas, twee jokers), een vriend die gluurt en
altijd een joker weglegt uit de twee die jij niet koos; speel $300$ rondes,
wissel altijd, en noteer de winsten. Verwachte \omterm{def:g10:stats:series}{frequentie}
$\frac23 \approx 0.667$, met schommeling
$\pm\frac{1}{\sqrt{300}} \approx 0.058$: verwacht tussen ongeveer $61\,\%$ en
$72\,\%$ gewonnen wisselingen --- ver van elke ``fiftyfifty''.

\textbf{12.} Even waarschijnlijke samenstellingen (oudste--jongste): JJ, JM,
MJ, MM. ``Minstens één jongen'' houdt JJ, JM, MJ over:
$P(\text{JJ}) = \frac13$.

\textbf{13.} ``Het oudste is een jongen'' houdt JJ en JM over:
$P(\text{JJ}) = \frac12$. Verschillende informatie filtert verschillende
verzamelingen scenario's --- precies zoals bij de twee Monty's: wat je
verneemt doet ertoe, en \emph{hoe} je het vernam evenzeer.

\textbf{14.} Alle drie gelijk: KKK, MMM: $\frac28 = \frac14$. De fout van
d'Alembert: zijn twee ``gevallen'' zijn niet even waarschijnlijk --- ``niet
alle drie gelijk'' bundelt zes van de acht uitkomsten.
\Cref{prop:g10:proba:uniform} telt alleen over even waarschijnlijke
uitkomsten.

\textbf{15.} \omterm{def:g10:stats:mean}{Gemiddelde} winst per spel:
$6 \times \frac16 + 3 \times \frac16 = 1.50$ euro, tegenover een inzet van
$2$ euro: een gemiddeld verlies van $0.50$ per spel, ongeveer $300$ euro
over $600$ spellen. De eerlijke premie maakt de \omterm{def:g10:stats:mean}{gemiddelde} uitkering gelijk:
$1\,\% \times 10\,000 = 100$ euro; echte premies tellen er kosten en een
veiligheidsmarge bij --- de verzekeraar moet slechte jaren overleven, niet
alleen \omterm{def:g10:stats:mean}{gemiddelde}.

\textbf{16.} \omterm{def:g10:numbers:interval}{Interval}: $0.5 \pm \frac{1}{\sqrt{100}} = \intcc{0.4}{0.6}$.
$55$ keer kop ($0.55$): erbinnen, niets bijzonders. $70$ keer kop ($0.70$):
ver erbuiten --- wantrouw de munt.

\textbf{17.} $\frac{1}{\sqrt{1000}} \approx 0.032$: een eerlijke kandidaat op
$50\,\%$ zou typisch tussen $\intcc{0.468}{0.532}$ scoren --- en $0.52$ ligt
daar comfortabel binnen. De peiling \emph{suggereert} een voorsprong maar
bewijst niets: de marge slokt haar op.

\textbf{18.} Eerste \omterm{def:g10:proba:sample}{steekproef}: \omterm{def:g10:numbers:interval}{interval}
$0.04 \pm \frac{1}{20} = \intcc{0}{0.09}$; de waargenomen $0.07$ ligt
erbinnen: geen oordeel. Tweede: $0.04 \pm \frac{1}{50} = \intcc{0.02}{0.06}$;
de waargenomen $0.07$ valt erbuiten: de bewering staat nu ernstig ter
discussie. Dezelfde \omterm{def:g10:stats:series}{frequentie}, scherper instrument: de nauwkeurigheid groeit
als $\sqrt n$.

\textbf{19.} De schommeling krimpt als $\frac{1}{\sqrt n}$: verviervoudig de
\omterm{def:g10:proba:sample}{steekproef} en je halveert de ruis. Vertekening --- de verkeerde menigte
bevragen --- overleeft elke $n$: twee miljoen slecht gekozen stembiljetten
bleven fout (de weekendopgave over hoe gegevens liegen in het
onderbouwvolume).

\textbf{20.} (i) Ga na dat de uitkomsten even waarschijnlijk zijn vóór je
telt --- d'Alembert vergat het. (ii) Boomdiagrammen vermenigvuldigen langs de
takken; ``minstens één'' geeft zich over aan het \omterm{def:g10:proba:operations}{complement}. (iii) Informatie
filtert scenario's, en haar \emph{bron} verandert het filter --- de wetende
Monty tegenover de gevallen Monty, het oudste-jongen tegenover
een-of-ander-jongen. (iv) \omterm{def:g10:stats:series}{Frequenties} naderen kansen met snelheid
$\frac{1}{\sqrt n}$, zodat een geduldige simulatie elke ruzie beslecht ---
ook een met tienduizend boze briefschrijvers.
\end{solution}
