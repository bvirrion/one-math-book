\begin{solution}{exo:g10:functions:1}
$f(0) = 1$; \quad $f(2) = 8 - 6 + 1 = 3$; \quad
$f(-1) = 2 + 3 + 1 = 6$; \quad
$f\!\left(\tfrac12\right) = 2 \times \tfrac14 - \tfrac32 + 1 = \tfrac12 -
\tfrac32 + 1 = 0$.
\end{solution}

\begin{solution}{exo:g10:functions:2}
$f$: geen enkele voorwaarde, \omterm{def:g10:functions:function}{domein} $\R$.

$g$: vereist $x + 4 \neq 0$, \omterm{def:g10:functions:function}{domein} alle \omterm{def:g10:numbers:sets}{reële getallen} behalve $-4$.

$h$: vereist $2x - 6 \geq 0$, dus $x \geq 3$: \omterm{def:g10:functions:function}{domein} $\intco{3}{+\infty}$.

$k$: $x^2 + 1 \geq 1 > 0$ verdwijnt nooit, \omterm{def:g10:functions:function}{domein} $\R$.
\end{solution}

\begin{solution}{exo:g10:functions:3}
\emph{Originelen van $0$:} $x^2 - 2x = x(x-2) = 0$, dus $0$ en $2$.

\emph{Originelen van $3$:} $x^2 - 2x = 3$, dus $x^2 - 2x - 3 = 0$, dus
$(x-3)(x+1) = 0$ (ga na door uit te werken), dus $3$ en $-1$.

\emph{Originelen van $-1$:} $x^2 - 2x + 1 = (x - 1)^2 = 0$, dus het enige
\omterm{def:g10:functions:function}{origineel} $1$.
\end{solution}

\begin{solution}{exo:g10:functions:4}
$f(-1) = 1 + 2 + 1 = 4$: ja, $A(-1, 4)$ ligt op de \omterm{def:g10:functions:graph}{grafiek}.
$f(2) = 4 - 4 + 1 = 1$: ja, ook $B(2, 1)$ ligt op de \omterm{def:g10:functions:graph}{grafiek}.
\end{solution}

\begin{solution}{exo:g10:functions:5}
\emph{1.} De grootste waarde in de tabel is $4 = g(3)$ (\omterm{def:g10:functions:extrema}{maximum}), de
kleinste is $-2 = g(0)$ (\omterm{def:g10:functions:extrema}{minimum}).

\emph{2.} Op $\intcc{-3}{0}$ daalt de \omterm{def:g10:functions:function}{functie}, en $-1 < -0.5$, dus
$g(-1) \geq g(-0.5)$.

\emph{3.} Op elk \omterm{def:g10:numbers:interval}{interval} waar $g$ monotoon is, neemt ze elke waarde
hoogstens één keer aan, dus heeft $g(x) = 0$ hoogstens één oplossing per
\omterm{def:g10:numbers:interval}{interval}: hoogstens $3$ oplossingen in totaal. (Hier ligt $0$ tussen $-2$ en
$1$, tussen $-2$ en $4$, en tussen $0$ en $4$, dus zijn het er precies
drie.)
\end{solution}

\begin{solution}{exo:g10:functions:6}
Zij $u < v$. Dan is
\[
f(v) - f(u) = (-2v + 7) - (-2u + 7) = -2(v - u) < 0,
\]
want $v - u > 0$. Dus $f(v) < f(u)$: $f$ is strikt \omterm{def:g10:functions:variations}{dalend} op $\R$.
\end{solution}

\begin{solution}{exo:g10:functions:7}
Vul het kwadraat aan: $(x+3)^2 = x^2 + 6x + 9$, dus
\[
f(x) = x^2 + 6x + 11 = (x + 3)^2 + 2 .
\]
Voor elke $x$ is $(x+3)^2 \geq 0$, dus $f(x) \geq 2$; en
$f(-3) = 0 + 2 = 2$. Het \omterm{def:g10:functions:extrema}{minimum} van $f$ is $2$, bereikt in $x = -3$.
\end{solution}

\begin{solution}{exo:g10:functions:8}
\emph{1.} Lengte $+$ breedte $= 20$, dus is de lengte $20 - x$ en
\[
A(x) = x(20 - x).
\]
Beide afmetingen moeten positief zijn: $0 < x < 20$.

\emph{2.} $A(4) = 64$, $A(8) = 96$, $A(10) = 100$, $A(12) = 96$,
$A(16) = 64$. De waarden stijgen tot $x = 10$ en dalen daarna symmetrisch:
de oppervlakte lijkt het grootst voor $x = 10$, dus voor een
\emph{vierkante} tuin. (\cref{ch:g11:quad} bewijst dat vermoeden.)
\end{solution}

\begin{solution}{exo:g10:functions:9}
\emph{1.} Voor elke $x$ is $x^2 + 1 \geq 1 > 0$, dus is
$f(x) = \frac{1}{x^2+1}$ positief; en uit $x^2 + 1 \geq 1$ volgt door
omkeren (beide leden positief) dat $f(x) \leq 1$.

\emph{2.} \omterm{def:g10:functions:extrema}{Maximum}: $f(0) = 1$ en $f(x) \leq 1$ voor alle $x$, dus is $1$ het
\omterm{def:g10:functions:extrema}{maximum}, bereikt in $0$. \omterm{def:g10:functions:extrema}{Minimum}: $f(x) > 0$ geldt altijd, maar geen enkele
$x$ bereikt $0$ (een quotiënt van getallen verschillend van nul is zelf niet
nul), en $f$ neemt voor grote $x$ waarden aan die zo dicht bij $0$ liggen
als je wilt; dus heeft $f$ geen \omterm{def:g10:functions:extrema}{minimum}.
\end{solution}

\begin{solution}{exo:g10:functions:10}
\emph{1.} $f(v) - f(u) = v^2 - u^2 = (v-u)(v+u)$. Is $0 \leq u < v$, dan is
$v - u > 0$ en $v + u > 0$ (som van een niet-negatief en een positief
getal), dus $f(v) - f(u) > 0$: $f$ is strikt \omterm{def:g10:functions:variations}{stijgend} op
$\intco{0}{+\infty}$.

\emph{2.} Is $u < v \leq 0$, dan is $v - u > 0$ nog steeds, maar nu is
$v + u < 0$ (som van een niet-positief en een negatief getal), dus
$f(v) - f(u) < 0$: $f$ is strikt \omterm{def:g10:functions:variations}{dalend} op $\intoc{-\infty}{0}$.
\end{solution}

\begin{solution}{pb:g10:functions:1}
\textbf{1.} \omterm{def:g10:functions:function}{Domein}: $x \neq 0$ (deling door $x$). $f(1) = \frac12(1 + 2) =
\frac32$; $f(2) = \frac12(2 + 1) = \frac32$;
$f\!\left(\frac32\right) = \frac12\left(\frac32 + \frac43\right) =
\frac{17}{12}$.

\textbf{2.} $\frac12\left(x + \frac2x\right) = \frac32$ geeft
$x^2 + 2 = 3x$, dus $x^2 - 3x + 2 = (x - 1)(x - 2) = 0$: de originelen van
$\frac32$ zijn $1$ en $2$ --- zoals vraag 1 al aankondigde.

\textbf{3.} $f(x) = x$ geeft $x + \frac2x = 2x$, dus $\frac2x = x$ en
$x^2 = 2$: vaste punten $\sqrt2$ en $-\sqrt2$. Het positieve is de diagonaal
van het eenheidsvierkant, \omterm{ex:g10:numbers:classify}{irrationaal} volgens de weekendopgave over
irrationaliteit in het onderbouwvolume.

\textbf{4.} $1 \to \frac32 \to \frac{17}{12} \to \frac{577}{408}$: de
\omterm{def:g10:numbers:approx}{benaderingen} van $\sqrt2$ van Heron. Zijn recept ``neem het gemiddelde van
de schatting en $2/$schatting'' is precies \emph{het itereren van de \omterm{def:g10:functions:function}{functie}
$f$} --- en het getal dat het proces najaagt is het vaste punt van $f$.

\textbf{5.} De vaste punten zijn de snijpunten van de \omterm{def:g10:functions:graph}{grafiek} met de
diagonaal $y = x$: de uitvoer opnieuw invoeren laat het punt langs de
\omterm{def:g10:functions:graph}{grafiek} naar dat snijpunt wandelen. In de weekendopgave over de twee
thermometers in het onderbouwvolume was de aanwijzing $-40^\circ$ hetzelfde
idee van het kruisen van de diagonaal, voor de omrekeningsfunctie.

\textbf{6.} $f(v) - f(u) = \frac{v - u}{2} + \frac1v - \frac1u =
\frac{v - u}{2} + \frac{u - v}{uv}
= (v - u)\left(\frac12 - \frac{1}{uv}\right)
= \frac{(v - u)(uv - 2)}{2uv}$. Voor $0 < u < v \leq \sqrt2$ is $uv < 2$, is
de haak negatief en dus $f(v) < f(u)$: \omterm{def:g10:functions:variations}{dalend}. Voor $\sqrt2 \leq u < v$ is
$uv > 2$: \omterm{def:g10:functions:variations}{stijgend}. Keerpunt: $\sqrt2$.

\textbf{7.} \omterm{def:g10:functions:variations}{Dalend} vóór $\sqrt2$, \omterm{def:g10:functions:variations}{stijgend} erna: $f$ bereikt haar \omterm{def:g10:functions:extrema}{minimum} in
$\sqrt2$, met waarde
$f(\sqrt2) = \frac12\left(\sqrt2 + \frac{2}{\sqrt2}\right) = \sqrt2$.
Gevolg: elke uitvoer van de machine (bij positieve invoer) is minstens
$\sqrt2$ --- na één slag aan de zwengel wonen de schattingen in
$\intco{\sqrt2}{+\infty}$.

\textbf{8.} Is $x > \sqrt2$, dan is $x^2 > 2$, dus $\frac2x < x$, en het
gemiddelde $f(x)$ van $x$ en $\frac2x$ is kleiner dan $x$; en $f(x) \geq
\sqrt2$ volgens vraag 7, met gelijkheid alleen in $x = \sqrt2$. Dus is
$\sqrt2 < f(x) < x$: elke iteratie verbetert strikt zonder voorbij te
schieten --- de rij uit vraag 4 glijdt neer op $\sqrt2$.

\textbf{9.} Op $\intoo{0}{+\infty}$: \omterm{def:g10:functions:variations}{dalend} van $+\infty$ (dicht bij $0$)
tot het \omterm{def:g10:functions:extrema}{minimum} $\sqrt2$ in $x = \sqrt2$, daarna onbegrensd \omterm{def:g10:functions:variations}{stijgend}.

\textbf{10.} $g(x) = x$ geeft $x^2 = x$, dus $x(x - 1) = 0$: vaste punten
$0$ en $1$. Vanaf $0.9$: $0.81$, $0.656$, $0.430$, $0.185$ --- glijdend naar
$0$. Vanaf $1.1$: $1.21$, $1.464$, $2.144$, $4.595$ --- wegvluchtend naar
oneindig. Het vaste punt $0$ trekt aan, $1$ stoot af: een haarbreed verschil
bij de start, tegengestelde lotsbestemmingen.

\textbf{11.} $7 \to 22 \to 11 \to 34 \to 17 \to 52 \to 26 \to 13 \to 40 \to
20 \to 10 \to 5 \to 16 \to 8 \to 4 \to 2 \to 1$: zestien stappen, met een
piek op $52$.

\textbf{12.} $27 \to 82 \to 41 \to 124 \to 62 \to 31 \to 94 \to 47 \to 142
\to 71 \to 214 \to \dots$ --- na tien stappen nog altijd klimmend, op weg
naar een piek van $9\,232$ en een vlucht van $111$ stappen. De les: een
regel van twee regels kan gedrag voortbrengen dat niemand in één oogopslag
voorspelt --- buurgetallen ($26$, $27$, $28$) hebben wild uiteenlopende
vluchten.

\textbf{13.} Een macht van $2$ halveert zijn hele ladder af:
$2^k \to 2^{k-1} \to \dots \to 2 \to 1$. Onderaan is $h(1) = 4$, $h(4) = 2$,
$h(2) = 1$: de vlucht komt in de cyclus $4 \to 2 \to 1 \to 4$ terecht. Een
\omterm{pb:g10:functions:1}{vast punt} zou $\frac n2 = n$ vragen (alleen $n = 0$) of $3n + 1 = n$
(negatief): geen enkel onder de positieve \omterm{def:g10:numbers:sets}{gehele getallen} --- de
waargenomen vluchten eindigen niet in een \omterm{pb:g10:functions:1}{vast punt} maar in die kleine
\emph{cyclus}.

\textbf{14.} Een tegenvoorbeeld zou ofwel een vlucht zijn die eeuwig groeit
(nooit terug naar $1$), ofwel een tweede cyclus los van $4 \to 2 \to 1$.
Verificatie tot $10^{20}$ laat oneindig veel ongeteste getallen over ---
precies de les over de gebieden in de cirkel uit de weekendopgave over het
orakel met de luciferstokjes in het onderbouwvolume: overeenstemming in
eindig veel gevallen bewijst niets over alle.

\textbf{15.} Bij de machine van Heron hadden we \emph{structuur} --- een
ontbonden verschil (vraag 6) dat het verloop blootlegt, een \omterm{def:g10:functions:extrema}{minimum}, een
insluiting --- terwijl de pariteitsschakelaar van de hagelsteen geen enkel
zulk houvast biedt: niemand heeft de structuur gevonden die haar temt.

\textbf{16.} $\sqrt{x - 3}$: vereist $x - 3 \geq 0$: $\intco{3}{+\infty}$.
$\frac{1}{x^2 - 9}$: vereist $x \neq \pm 3$: $\R$ zonder $-3$ en $3$.
$\sqrt{9 - x^2}$: vereist $x^2 \leq 9$: $\intcc{-3}{3}$.

\textbf{17.} $f(x) = 3$ geeft $x + \frac2x = 6$, dus $x^2 - 6x + 2 = 0$, dus
$(x - 3)^2 = 7$: $x = 3 - \sqrt7$ of $x = 3 + \sqrt7$ (beide positief: twee
exacte originelen).

\textbf{18.} Fysische zin zolang de steen in de lucht is:
$H(t) = 5t(4 - t) \geq 0$ voor $t \in \intcc{0}{4}$. Kwadraat aanvullen:
$H(t) = 20 - 5(t - 2)^2$: \omterm{def:g10:functions:extrema}{maximum} $20$ m in $t = 2$ s. Verloop: \omterm{def:g10:functions:variations}{stijgend} op
$\intcc{0}{2}$ van $0$ tot $20$, \omterm{def:g10:functions:variations}{dalend} op $\intcc{2}{4}$ terug tot $0$.

\textbf{19.} $20t - 5t^2 \geq 15 \iff t^2 - 4t + 3 \leq 0 \iff
(t - 1)(t - 3) \leq 0 \iff t \in \intcc{1}{3}$: twee volle seconden boven
$15$ m.

\textbf{20.} Elke machine heeft een \omterm{def:g10:functions:function}{domein} (waar het voorschrift mag
werken), een voorschrift (de formule of de pariteitsschakelaar) en een
\omterm{def:g10:functions:graph}{grafiek} of tabel die haar in haar geheel toont. Aan elk stelden we de twee
grote vragen: \emph{waar leidt herhaling heen} (vaste punten, cycli,
aantrekking --- opgelost voor Heron, open voor de hagelsteen) en \emph{hoe
beweegt de uitvoer met de invoer mee} (verloop, extrema --- de top van de
steen, het \omterm{def:g10:functions:extrema}{minimum} van Heron). De volgende hoofdstukken verscherpen beide:
eerst de referentiefuncties, en in latere jaren de afgeleiden om het
verloop te meten en de limieten om te bevestigen waar iteraties belanden.
\end{solution}
