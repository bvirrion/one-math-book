\begin{solution}{exo:g10:functions:2}
$f$: sin ninguna restricción, \omterm{def:g10:functions:function}{dominio} $\R$.

$g$: exige $x + 4 \neq 0$, \omterm{def:g10:functions:function}{dominio} $\R$ salvo $-4$.

$h$: exige $2x - 6 \geq 0$, es decir, $x \geq 3$: \omterm{def:g10:functions:function}{dominio}

$k$: $x^2 + 1 \geq 1 > 0$ no se anula nunca, \omterm{def:g10:functions:function}{dominio} $\R$.
\end{solution}

\begin{solution}{exo:g10:functions:3}
\emph{Antiimágenes de $0$:} $x^2 - 2x = x(x-2) = 0$, luego $0$ y $2$.

\emph{Antiimágenes de $3$:} $x^2 - 2x = 3$, es decir, $x^2 - 2x - 3 = 0$,
o sea $(x-3)(x+1) = 0$ (compruébalo desarrollando), luego $3$ y $-1$.

\emph{Antiimágenes de $-1$:} $x^2 - 2x + 1 = (x - 1)^2 = 0$, luego la
única \omterm{def:g10:functions:function}{antiimagen} $1$.
\end{solution}

\begin{solution}{exo:g10:functions:4}

$f(2) = 4 - 4 + 1 = 1$: sí, $B(2, 1)$ también está en la \omterm{def:g10:functions:graph}{gráfica}.
\end{solution}

\begin{solution}{exo:g10:functions:5}
\emph{1.} El mayor valor de la tabla es $4 = g(3)$ (\omterm{def:g10:functions:extrema}{máximo}) y el menor es
$-2 = g(0)$ (\omterm{def:g10:functions:extrema}{mínimo}).

\emph{2.} En $\intcc{-3}{0}$ la \omterm{def:g10:functions:function}{función} decrece y $-1 < -0.5$, luego
$g(-1) \geq g(-0.5)$.

\emph{3.} En cada \omterm{def:g10:numbers:interval}{intervalo} de monotonía, $g$ toma cada valor como mucho
una vez, así que $g(x) = 0$ tiene como mucho una solución por \omterm{def:g10:numbers:interval}{intervalo}:
como mucho $3$ soluciones en total. (Aquí $0$ está entre $-2$ y $1$, entre
$-2$ y $4$, y entre $0$ y $4$, así que hay exactamente tres.)
\end{solution}

\begin{solution}{exo:g10:functions:6}
Sea $u < v$. Entonces
\[
f(v) - f(u) = (-2v + 7) - (-2u + 7) = -2(v - u) < 0,
\]
puesto que $v - u > 0$. Luego $f(v) < f(u)$: $f$ es estrictamente
\omterm{def:g10:functions:variations}{decreciente} en $\R$.
\end{solution}

\begin{solution}{exo:g10:functions:7}
Completamos el cuadrado: $(x+3)^2 = x^2 + 6x + 9$, luego
\[
f(x) = x^2 + 6x + 11 = (x + 3)^2 + 2 .
\]
Para todo $x$, $(x+3)^2 \geq 0$, luego $f(x) \geq 2$; y
$f(-3) = 0 + 2 = 2$. El \omterm{def:g10:functions:extrema}{mínimo} de $f$ es $2$, alcanzado en $x = -3$.
\end{solution}

\begin{solution}{exo:g10:functions:8}
\emph{1.} Largo $+$ ancho $= 20$, así que el largo es $20 - x$ y
\[
A(x) = x(20 - x).
\]
Las dos dimensiones han de ser positivas: $0 < x < 20$.

\emph{2.} $A(4) = 64$, $A(8) = 96$, $A(10) = 100$, $A(12) = 96$,
$A(16) = 64$. Los valores suben hasta $x = 10$ y bajan después de forma
simétrica: parece que el área es máxima para $x = 10$, es decir, para un
jardín \emph{cuadrado}. (El \cref{ch:g11:quad} demuestra esta conjetura.)
\end{solution}

\begin{solution}{exo:g10:functions:9}
\emph{1.} Para todo $x$: $x^2 + 1 \geq 1 > 0$, luego
$f(x) = \frac{1}{x^2+1}$ es positivo; y como $x^2 + 1 \geq 1$, al tomar
inversos (los dos miembros son positivos) se obtiene $f(x) \leq 1$.

\emph{2.} \omterm{def:g10:functions:extrema}{Máximo}: $f(0) = 1$ y $f(x) \leq 1$ para todo $x$, luego $1$ es
el \omterm{def:g10:functions:extrema}{máximo}, alcanzado en $0$. \omterm{def:g10:functions:extrema}{Mínimo}: $f(x) > 0$ siempre, pero ningún valor
de $x$ da $0$ (un cociente de números no nulos no es nulo) y $f$ toma
valores tan próximos a $0$ como se quiera para $x$ grande; así que $f$ no
tiene \omterm{def:g10:functions:extrema}{mínimo}.
\end{solution}

\begin{solution}{exo:g10:functions:10}
\emph{1.} $f(v) - f(u) = v^2 - u^2 = (v-u)(v+u)$. Si $0 \leq u < v$,
entonces $v - u > 0$ y $v + u > 0$ (suma de un número no negativo y uno
positivo), luego $f(v) - f(u) > 0$: $f$ es estrictamente \omterm{def:g10:functions:variations}{creciente} en
$\intco{0}{+\infty}$.

\emph{2.} Si $u < v \leq 0$, sigue siendo $v - u > 0$, pero ahora
$v + u < 0$ (suma de un número no positivo y uno negativo), luego
$f(v) - f(u) < 0$: $f$ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en
$\intoc{-\infty}{0}$.
\end{solution}

\begin{solution}{pb:g10:functions:1}
\textbf{1.} \omterm{def:g10:functions:function}{Dominio}: $x \neq 0$ (se divide entre $x$). $f(1) =
\frac12(1 + 2) = \frac32$; $f(2) = \frac12(2 + 1) = \frac32$;
$f\!\left(\frac32\right) = \frac12\left(\frac32 +
\frac43\right) = \frac{17}{12}$.

\textbf{2.} $\frac12\left(x + \frac2x\right) = \frac32$ da
$x^2 + 2 = 3x$, es decir, $x^2 - 3x + 2 = (x - 1)(x - 2) = 0$: las
antiimágenes de $\frac32$ son $1$ y $2$, tal como anunciaba la
pregunta 1.

\textbf{3.} $f(x) = x$ da $x + \frac2x = 2x$, luego $\frac2x = x$ y
$x^2 = 2$: los \omterm{pb:g10:functions:1}{puntos fijos} son $\sqrt2$ y $-\sqrt2$. El positivo es la
diagonal del cuadrado unidad, irracional por el problema de fin de semana
del volumen anterior sobre la irracionalidad.

\textbf{4.} $1 \to \frac32 \to \frac{17}{12} \to \frac{577}{408}$: las
aproximaciones de $\sqrt2$ de Herón. Su receta ``haz la media de la
estimación con $2/$estimación'' es exactamente \emph{iterar la \omterm{def:g10:functions:function}{función}
$f$}, y el número que persigue el proceso es el \omterm{pb:g10:functions:1}{punto fijo} de $f$.

\textbf{5.} Los \omterm{pb:g10:functions:1}{puntos fijos} son los cortes de la \omterm{def:g10:functions:graph}{gráfica} con la diagonal
$y = x$: volver a meter los resultados hace caminar el punto por la
\omterm{def:g10:functions:graph}{gráfica} hacia ese corte. En el problema de fin de semana del volumen
anterior sobre los dos termómetros, la lectura $-40^\circ$ era esa misma
idea de cortar la diagonal, para la \omterm{def:g10:functions:function}{función} de conversión.

\textbf{6.} $f(v) - f(u) = \frac{v - u}{2} + \frac1v - \frac1u
= \frac{v - u}{2} + \frac{u - v}{uv}
= (v - u)\left(\frac12 - \frac{1}{uv}\right)
= \frac{(v - u)(uv - 2)}{2uv}$. Para $0 < u < v \leq \sqrt2$: $uv < 2$, el
paréntesis es negativo y $f(v) < f(u)$: \omterm{def:g10:functions:variations}{decreciente}. Para
$\sqrt2 \leq u < v$: $uv > 2$: \omterm{def:g10:functions:variations}{creciente}. El punto de cambio es $\sqrt2$.

\textbf{7.} \omterm{def:g10:functions:variations}{Decreciente} antes de $\sqrt2$ y \omterm{def:g10:functions:variations}{creciente} después: $f$ alcanza
su \omterm{def:g10:functions:extrema}{mínimo} en $\sqrt2$, y vale
$f(\sqrt2) = \frac12\left(\sqrt2 + \frac{2}{\sqrt2}\right) = \sqrt2$.
Consecuencia: todo resultado de la máquina (con entrada positiva) es al
menos $\sqrt2$; tras una vuelta de manivela, las estimaciones viven en
$\intco{\sqrt2}{+\infty}$.

\textbf{8.} Si $x > \sqrt2$, entonces $x^2 > 2$, luego $\frac2x < x$ y la
media $f(x)$ de $x$ y $\frac2x$ es menor que $x$; además
$f(x) \geq \sqrt2$ por la pregunta 7, con igualdad solo en $x = \sqrt2$.
Así pues, $\sqrt2 < f(x) < x$: cada iteración mejora estrictamente sin
pasarse de largo, y la sucesión de la pregunta 4 se desliza hasta
$\sqrt2$.

\textbf{9.} En $\intoo{0}{+\infty}$: \omterm{def:g10:functions:variations}{decreciente} desde $+\infty$ (cerca de
$0$) hasta el \omterm{def:g10:functions:extrema}{mínimo} $\sqrt2$ en $x = \sqrt2$, y \omterm{def:g10:functions:variations}{creciente} después sin
límite.

\textbf{10.} $g(x) = x$ da $x^2 = x$, $x(x - 1) = 0$: \omterm{pb:g10:functions:1}{puntos fijos} $0$ y
$1$. Desde $0.9$: $0.81$, $0.656$, $0.430$, $0.185$, deslizándose hacia
$0$. Desde $1.1$: $1.21$, $1.464$, $2.144$, $4.595$, escapando hacia el
infinito. El \omterm{pb:g10:functions:1}{punto fijo} $0$ atrae y el $1$ repele: una diferencia de un
pelo al principio, destinos opuestos.

\textbf{11.} $7 \to 22 \to 11 \to 34 \to 17 \to 52 \to 26 \to
13 \to 40 \to 20 \to 10 \to 5 \to 16 \to 8 \to 4 \to 2 \to 1$: dieciséis
pasos, con una cima en $52$.

\textbf{12.} $27 \to 82 \to 41 \to 124 \to 62 \to 31 \to 94 \to
47 \to 142 \to 71 \to 214 \to \dots$: todavía subiendo tras diez pasos,
camino de una cima de $9\,232$ y de un vuelo de $111$ pasos. Lección: una
regla de dos líneas puede producir un comportamiento que nadie sabe
prever de un vistazo; números vecinos ($26$, $27$, $28$) tienen vuelos
disparatadamente distintos.

\textbf{13.} Una potencia de $2$ se va dividiendo entre dos por toda su
escalera: $2^k \to 2^{k-1} \to \dots \to 2 \to 1$. Abajo del todo,
$h(1) = 4$, $h(4) = 2$, $h(2) = 1$: el vuelo entra en el ciclo
$4 \to 2 \to 1 \to 4$. Un \omterm{pb:g10:functions:1}{punto fijo} exigiría $\frac n2 = n$ (solo
$n = 0$) o $3n + 1 = n$ (negativo): no hay ninguno entre los enteros
positivos; lo que forman los vuelos observados no es un \omterm{pb:g10:functions:1}{punto fijo}, sino
ese pequeño \emph{ciclo}.

\textbf{14.} Un contraejemplo sería o bien un vuelo que crece para
siempre (sin volver nunca a $1$), o bien un segundo ciclo disjunto de
$4 \to 2 \to 1$. Verificar hasta $10^{20}$ deja infinitos números sin
comprobar: exactamente la lección de las regiones del círculo del problema
de fin de semana del volumen anterior sobre el oráculo de las cerillas;
que algo se cumpla en un número finito de casos no demuestra nada sobre
todos.

\textbf{15.} En la máquina de Herón había \emph{estructura}: una
diferencia factorizada (pregunta 6) que revelaba la variación, un \omterm{def:g10:functions:extrema}{mínimo},
un cerco; la regla del granizo, que cambia según la paridad, no ofrece
ningún asidero parecido: nadie ha encontrado la estructura que la
domestique.

\textbf{16.} $\sqrt{x - 3}$: exige $x - 3 \geq 0$: $\intco{3}{+\infty}$.
$\frac{1}{x^2 - 9}$: exige $x \neq \pm 3$: $\R$ salvo $-3$ y $3$.
$\sqrt{9 - x^2}$: exige $x^2 \leq 9$: $\intcc{-3}{3}$.

\textbf{17.} $f(x) = 3$ da $x + \frac2x = 6$, luego $x^2 - 6x + 2 = 0$, es
decir, $(x - 3)^2 = 7$: $x = 3 - \sqrt7$ o $x = 3 + \sqrt7$ (los dos
positivos: dos antiimágenes exactas).

\textbf{18.} La fórmula tiene sentido físico mientras la piedra está en el
aire: $H(t) = 5t(4 - t) \geq 0$ para $t \in \intcc{0}{4}$. Completando el
cuadrado: $H(t) = 20 - 5(t - 2)^2$: \omterm{def:g10:functions:extrema}{máximo} de $20$ m en $t = 2$ s.
Variación: \omterm{def:g10:functions:variations}{creciente} en $\intcc{0}{2}$ de $0$ a $20$ y \omterm{def:g10:functions:variations}{decreciente} en
$\intcc{2}{4}$ de vuelta a $0$.

\textbf{19.} $20t - 5t^2 \geq 15 \iff t^2 - 4t + 3 \leq 0 \iff
(t - 1)(t - 3) \leq 0 \iff t \in \intcc{1}{3}$: dos segundos completos por
encima de $15$ m.

\textbf{20.} Cada máquina tiene un \omterm{def:g10:functions:function}{dominio} (donde la regla está
permitida), una regla (la fórmula o el cambio según la paridad) y una
\omterm{def:g10:functions:graph}{gráfica} o una tabla que la muestra entera. A cada una le hemos hecho las
dos grandes preguntas: \emph{adónde lleva la repetición} (\omterm{pb:g10:functions:1}{puntos fijos},
ciclos, atracción: resuelto para Herón, abierto para el granizo) y
\emph{cómo se mueve el resultado con la entrada} (variación, extremos: la
cima de la piedra, el \omterm{def:g10:functions:extrema}{mínimo} de Herón). Los capítulos siguientes afinan
las dos: primero las funciones de referencia y, en cursos posteriores, las
derivadas para medir la variación y los límites para certificar dónde
aterrizan las iteraciones.
\end{solution}
