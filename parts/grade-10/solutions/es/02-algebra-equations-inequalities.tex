\section*{Capítulo \ref{ch:g10:algebra} --- Álgebra: ecuaciones e inecuaciones}

\begin{solution}{exo:g10:algebra:1}
$3(2x-5) - 2(x-7) = 6x - 15 - 2x + 14 = 4x - 1$.

$(x+4)(2x-1) = 2x^2 - x + 8x - 4 = 2x^2 + 7x - 4$.

$(3x-2)^2 = 9x^2 - 12x + 4$.

$(5-x)(5+x) = 25 - x^2$.
\end{solution}

\begin{solution}{exo:g10:algebra:2}
$x^2 - 49 = (x+7)(x-7)$ (diferencia de cuadrados).

$x^2 + 10x + 25 = (x+5)^2$.

$7x^2 - 14x = 7x(x - 2)$ (factor común).

$16x^2 - 8x + 1 = (4x - 1)^2$.
\end{solution}

\begin{solution}{exo:g10:algebra:3}
$4x + 9 = x - 3$: $3x = -12$, luego $x = -4$.

$\frac{2x-1}{3} = 5$: $2x - 1 = 15$, luego $x = 8$.

$2(x-3) = 2x + 1$ se desarrolla como $2x - 6 = 2x + 1$, es decir,
$-6 = 1$: falso para todo $x$. No hay solución; los dos miembros definen
rectas paralelas que nunca se cortan.
\end{solution}

\begin{solution}{exo:g10:algebra:4}
$(x+5)(2x-8) = 0$: $x = -5$ o $x = 4$.

$x^2 - 16 = (x-4)(x+4) = 0$: $x = 4$ o $x = -4$.

$x^2 + 6x + 9 = (x+3)^2 = 0$: la única solución $x = -3$.
\end{solution}

\begin{solution}{exo:g10:algebra:5}
$3x - 5 \leq 7$: $3x \leq 12$, luego $x \leq 4$: conjunto de soluciones
$\intoc{-\infty}{4}$.

$-4x + 1 > 9$: $-4x > 8$; al dividir entre $-4$ se invierte la
desigualdad: $x < -2$: conjunto de soluciones $\intoo{-\infty}{-2}$.

$2(x-1) \geq 5x + 4$: $2x - 2 \geq 5x + 4$, luego $-3x \geq 6$, luego
$x \leq -2$: conjunto de soluciones $\intoc{-\infty}{-2}$.
\end{solution}

\begin{solution}{exo:g10:algebra:6}
El factor común es $(x-1)$:
\[
E = (x-1)\bigl[(x+3) - (2x - 5)\bigr] = (x-1)(x + 3 - 2x + 5)
= (x-1)(8 - x).
\]
Entonces $E = 0$ cuando $x = 1$ o $x = 8$ (regla del producto nulo).
\end{solution}

\begin{solution}{exo:g10:algebra:7}
$x^2 = 5x$: $x^2 - 5x = x(x - 5) = 0$, luego $x = 0$ o $x = 5$ (no
dividas nunca entre $x$: se perdería la solución $0$).

$(2x+1)^2 = (x-4)^2$: pasamos al primer miembro y factorizamos la
diferencia de cuadrados:
\[
(2x+1)^2 - (x-4)^2
= \bigl[(2x+1) + (x-4)\bigr]\bigl[(2x+1) - (x-4)\bigr]
= (3x - 3)(x + 5).
\]
Regla del producto nulo: $x = 1$ o $x = -5$.
\end{solution}

\begin{solution}{exo:g10:algebra:8}
$(x+2)(4-x) > 0$: el factor $x + 2$ se anula en $-2$ (positivo después) y
$4 - x$ se anula en $4$ (positivo antes). El producto es positivo
exactamente cuando los dos factores lo son, es decir, en $\intoo{-2}{4}$.

$\frac{2x-6}{x+1} \geq 0$: el numerador se anula en $3$ y el denominador
en $-1$ (prohibido). \omterm{def:g10:algebra:signtable}{Tabla de signos}: el cociente es positivo cuando las
dos partes tienen el mismo signo, es decir, para $x < -1$ o $x > 3$, y se
anula en $x = 3$. Conjunto de soluciones:
$\intoo{-\infty}{-1} \cup \intco{3}{+\infty}$.
\end{solution}

\begin{solution}{exo:g10:algebra:9}
Al cabo de $x$ meses, el primer servicio cuesta $9x + 15$ y el segundo
$12x$. El primero es más barato cuando
\[
9x + 15 < 12x
\ \Longleftrightarrow\
15 < 3x
\ \Longleftrightarrow\
x > 5 .
\]
A partir del sexto mes, el primer servicio sale más barato en total.
\end{solution}

\begin{solution}{exo:g10:algebra:10}
Valor prohibido: $x = 1$. Para $x \neq 1$, multiplicamos los dos miembros
por $x - 1$:
\[
x + 3 = 2(x - 1) = 2x - 2
\ \Longleftrightarrow\
x = 5 .
\]
Como $5 \neq 1$, la solución es $x = 5$. Comprobación:
$\frac{5+3}{5-1} = \frac84 = 2$.
\end{solution}

\begin{solution}{exo:g10:algebra:11}
$(n+1)^2 - n^2 = n^2 + 2n + 1 - n^2 = 2n + 1$, que es impar para todo
entero $n$.

Dos números impares consecutivos se pueden escribir $2n - 1$ y $2n + 1$.
Entonces
\[
(2n+1)^2 - (2n-1)^2
= \bigl[(2n+1)+(2n-1)\bigr]\bigl[(2n+1)-(2n-1)\bigr]
= 4n \times 2 = 8n,
\]
múltiplo de $8$.
\end{solution}

\begin{solution}{pb:g10:algebra:1}
\textbf{1.} $x^2 - 10x + 25 = (x - 5)^2$;
$9x^2 - 49 = (3x - 7)(3x + 7)$;
$x^2 + 6x + 9 - 4 = (x + 3)^2 - 2^2 = (x + 1)(x + 5)$.

\textbf{2.} $x = -1$ o $x = -5$; \quad $x = \frac73$ o
$x = -\frac73$.

\textbf{3.} $(5 - t)(5 + t) = 25 - t^2 = 21$ da $t^2 = 4$, $t = 2$: los
números son $3$ y $7$ (suma $10$, producto $21$: comprobado).

\textbf{4.} $(7 - t)(7 + t) = 49 - t^2 = 33$: $t = 4$, y los números son
$3$ y $11$.

\textbf{5.} $\left(\frac s2 - t\right)\left(\frac s2 + t\right)
= \left(\frac s2\right)^2 - t^2 = p$ obliga a que
$t^2 = \left(\frac s2\right)^2 - p$. Hay solución exactamente cuando esto
es $\geq 0$, es decir, cuando $p \leq \left(\frac s2\right)^2$: el
producto de dos números de suma dada nunca puede superar el cuadrado de la
semisuma (como ya sabía el problema del cercado de la reina Dido, en el
volumen anterior).

\textbf{6.} $(x - a)(x - b) = x^2 - (a + b)x + ab$: la \omterm{def:g10:algebra:equation}{ecuación}
$x^2 - sx + p = 0$ dice exactamente ``$x$ es uno de dos números de suma
$s$ y producto $p$''.

\textbf{7.} $(x-5)^2 - 4 = x^2 - 10x + 25 - 4 = x^2 - 10x + 21$.
Diferencia de cuadrados: $(x - 5 - 2)(x - 5 + 2) = (x - 7)(x - 3) = 0$:
soluciones $3$ y $7$, los números del escriba de la pregunta 3.

\textbf{8.} $x^2 + 6x - 7 = (x + 3)^2 - 16 = 0$ da $(x + 3)^2 = 16$,
$x + 3 = \pm 4$: $x = 1$ o $x = -7$.

\textbf{9.} $x^2 + 4x + 5 = (x + 2)^2 + 1 \geq 1 > 0$ para todo $x$: no
hay solución real. En la forma $(x + h)^2 + k$ hay soluciones exactamente
cuando $k \leq 0$ (entonces $(x+h)^2 = -k$ tiene las dos raíces
$-h \pm \sqrt{-k}$); si $k > 0$, la expresión se mantiene positiva.

\textbf{10.} El cuadrado de lado $x$ (área $x^2$) con dos rectángulos de
$3 \times x$ ($6x$ en total) forma una L de área $x^2 + 6x = 7$; la
esquina de $3 \times 3$ que falta (área $9$) completa un cuadrado grande
de lado $x + 3$. Sumando $9$ a los dos miembros: $(x + 3)^2 = 16$, luego
$x + 3 = 4$ (las longitudes son positivas) y $x = 1$. El álgebra y el
dibujo son el mismo gesto.

\textbf{11.} $(x-3)(x+2)$: negativo entre las raíces. Solución de $< 0$:
$x \in \intoo{-2}{3}$.

\textbf{12.} $x^2 - 10x + 21 \leq 0$, es decir, $(x - 3)(x - 7) \leq 0$:
entre las raíces, extremos incluidos: $x \in \intcc{3}{7}$.

\textbf{13.} Se anula en $x = 1$ y está prohibido $x = -2$; el cociente es
positivo fuera de esos valores:
$x \in \intoo{-\infty}{-2} \cup \intco{1}{+\infty}$.

\textbf{14.} $25 - (x - 5)^2 = 25 - x^2 + 10x - 25 = x(10 - x)$:
identidad comprobada. Como $(x-5)^2 \geq 0$ siempre, $A(x) \leq 25$, con
igualdad exactamente cuando $x = 5$: el corral cuadrado de $5 \times 5$,
con un máximo de $25$ m$^2$; la tabla del volumen anterior, ahora una
demostración de dos líneas.

\textbf{15.} Para $x > 0$, multiplicamos la afirmación por $x$:
$x^2 + 1 \geq 2x$, es decir, $x^2 - 2x + 1 = (x - 1)^2 \geq 0$, que es
cierto, con igualdad solo en $x = 1$. Al dividir de nuevo entre $x > 0$
todo se conserva (\cref{prop:g10:algebra:ineqrules}).

\textbf{16.} Multiplicando en cruz en $\frac1x = \frac{x}{1 - x}$ (todas
las longitudes son positivas): $x^2 = 1 - x$, es decir,
$x^2 + x - 1 = 0$.

\textbf{17.} $x^2 + x - 1 = \left(x + \frac12\right)^2 - \frac54 = 0$ da
$x + \frac12 = \pm\frac{\sqrt5}{2}$. La raíz positiva:
$x = \frac{\sqrt5 - 1}{2} \approx 0.618$.

\textbf{18.} $\varphi = \frac{2}{\sqrt5 - 1} =
\frac{2(\sqrt5 + 1)}{(\sqrt5 - 1)(\sqrt5 + 1)}
= \frac{2(\sqrt5 + 1)}{5 - 1} = \frac{\sqrt5 + 1}{2}
\approx 1.618$.

\textbf{19.} $\varphi^2 = \frac{(\sqrt5 + 1)^2}{4} =
\frac{6 + 2\sqrt5}{4} = \frac{3 + \sqrt5}{2}$, y
$\varphi + 1 = \frac{\sqrt5 + 3}{2}$: coinciden. Y
$\varphi - 1 = \frac{\sqrt5 - 1}{2} = x = \frac1\varphi$ (así se definió
$\varphi$). El número áureo es el número cuyo cuadrado suma uno y cuyo
inverso resta uno.

\textbf{20.} $\sqrt5$ es irracional (problema de fin de semana del volumen
anterior sobre la irracionalidad: $5$ no es un cuadrado perfecto), y
sumar $1$ y dividir después entre $2$ conserva la irracionalidad
(operaciones racionales, \cref{pb:g10:numbers:1}). Cocientes:
$\frac53 \approx 1.667$, $\frac85 = 1.600$, $\frac{13}{8} = 1.625$,
$\frac{21}{13} \approx 1.615$: alternativamente por encima y por debajo,
cerrándose sobre $\varphi = 1.618\ldots$ Los números del recuento de
ritmos (problema de fin de semana del volumen anterior sobre el recuento
de ritmos) persiguen en secreto la sección áurea; los capítulos sobre
sucesiones los pillarán en el acto.
\end{solution}
