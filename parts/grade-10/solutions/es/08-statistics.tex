\section*{Capítulo \ref{ch:g10:stats} --- Estadística descriptiva}

\begin{solution}{exo:g10:stats:1}
Suma: $0+1+1+2+2+2+3+3+4+4+5+6+7+8+12 = 60$, entonces la \omterm{def:g10:stats:mean}{media} es
$\frac{60}{15} = 4$. La serie ya está ordenada y $ N = 15$ es impar:
la \omterm{def:g10:stats:median}{mediana} es el $8$ valor, $3$. recorrido: $12 - 0 = 12$.
\end{solution}

\begin{solution}{exo:g10:stats:2}
\omterm{def:g10:stats:series}{Frecuencias}: $\frac{7}{50} = 0.14$, $\frac{9}{50} = 0.18$,
$\frac{8}{50} = 0.16$, $\frac{10}{50} = 0.20$, $\frac{6}{50} = 0.12$,
$\frac{10}{50} = 0.20$ (suma $1$). \omterm{def:g10:stats:mean}{media} resultado:
\[
\bar x = \frac{7 \times 1 + 9 \times 2 + 8 \times 3 + 10 \times 4
+ 6 \times 5 + 10 \times 6}{50}
= \frac{7 + 18 + 24 + 40 + 30 + 60}{50}
= \frac{179}{50} = 3.58 .
\]
\end{solution}

\begin{solution}{exo:g10:stats:3}
$ N = 12$ (par): la \omterm{def:g10:stats:median}{mediana} es el \omterm{prop:g10:coordgeom:midpoint}{punto medio} del $6$ th y $7$ th
valores, ambos iguales a $6$: \omterm{def:g10:stats:median}{mediana} $6$.

$ Q_1$: $\frac{12}{4} = 3$, entonces $ Q_1$ es el $3$ tercer valor: $4$.

$ Q_3$: $\frac{3 \times 12}{4} = 9$, entonces $ Q_3$ es el $9$ ésimo valor: $8$.

Diagrama de caja: caja de $4$ a $8$ con la \omterm{def:g10:stats:median}{mediana} bar en $6$, bigotes de
$1$ a $11$.
\end{solution}

\begin{solution}{exo:g10:stats:5}
Total actual: $5 \times 12 = 60$. A \omterm{def:g10:stats:mean}{media} de $13$ encima $6$ las pruebas requieren
un total de $78$, entonces el $6$ La puntuación debe ser $78 - 60 = 18$. A \omterm{def:g10:stats:mean}{media} de

\end{solution}

\begin{solution}{exo:g10:stats:6}
\omterm{def:g10:stats:mean}{media} más grande que $20$: por ejemplo
$1,\ 2,\ 3,\ 10,\ 40,\ 50,\ 60$ --- \omterm{def:g10:stats:median}{mediana} $10$ (el $4$º valor ordenado),
\omterm{def:g10:stats:mean}{media} $\frac{166}{7} \approx 23.7$.

\omterm{def:g10:stats:mean}{media} más pequeño que $6$: por ejemplo
$0,\ 0,\ 0,\ 10,\ 10,\ 10,\ 10$ --- \omterm{def:g10:stats:median}{mediana} $10$, \omterm{def:g10:stats:mean}{media}
$\frac{40}{7} \approx 5.7$.

La \omterm{def:g10:stats:mean}{media} no puedo ir abajo $\frac{40}{7}$: para la \omterm{def:g10:stats:median}{mediana} (el $4$º ordenado
valor) para ser $10$, los valores en posiciones $4$ a $7$ todos deben ser
$\geq 10$, por lo que el total es al menos $40$ y la \omterm{def:g10:stats:mean}{media} al menos
$\frac{40}{7}$ --- el segundo ejemplo es extremo.

Estos ejemplos muestran que la \omterm{def:g10:stats:mean}{media} y la \omterm{def:g10:stats:median}{mediana} son en gran medida independientes:
conocer uno dice poco del otro, por eso un buen resumen
informa ambos.
\end{solution}

\begin{solution}{exo:g10:stats:7}
\emph{1.} \omterm{def:g10:stats:mean}{media}:
$\frac{9 \times 1800 + 12000}{10} = \frac{16200 + 12000}{10} = 2820$.
\omterm{def:g10:stats:median}{mediana}: ordenado, el $10$ los salarios son nueve valores $1800$ entonces $12000$;
la \omterm{def:g10:stats:median}{mediana} es el \omterm{prop:g10:coordgeom:midpoint}{punto medio} del $5$ th y $6$º valores, ambos $1800$:
\omterm{def:g10:stats:median}{mediana} $1800$.

\emph{2.} La \omterm{def:g10:stats:median}{mediana} ($1800$) describe el salario típico: es lo que
$9$ empleados fuera de $10$ realmente ganar. La \omterm{def:g10:stats:mean}{media} ($2820$) está levantado
por el salario único alto y no iguala la nómina de nadie.
\end{solution}

\begin{solution}{exo:g10:stats:8}
Altura total: $12 \times 168 + 18 \times 163 = 2016 + 2934 = 4950$ centímetros para
$30$ estudiantes, por lo que la \omterm{def:g10:stats:mean}{media} es $\frac{4950}{30} = 165$ centímetro. esta mas cerca de
las chicas \omterm{def:g10:stats:mean}{media} porque son más numerosos (ponderados \omterm{def:g10:stats:mean}{media}, pesas
$12$ y $18$).
\end{solution}

\begin{solution}{exo:g10:stats:9}
\emph{1.} el nuevo \omterm{def:g10:stats:mean}{media} es
\[
\bar y = \frac{y_1 + \dots + y_N}{N}
= \frac{(a x_1 + b) + \dots + (a x_N + b)}{N}
= \frac{a(x_1 + \dots + x_N) + Nb}{N}
= a \bar x + b .
\]

\emph{2.} Con $ a = 1.8$ y $ b = 32$: \omterm{def:g10:stats:mean}{media}
$1.8 \times 20 + 32 = 68$ grados Fahrenheit.
\end{solution}

\begin{solution}{pb:g10:stats:1}
\textbf{1.} \omterm{def:g10:stats:mean}{media}: $\frac{2\,550}{7} \approx 364$ mil
euros; \omterm{def:g10:stats:median}{mediana}: el cuarto de los siete precios ordenados, $230$.
La \omterm{def:g10:stats:median}{mediana} describe la calle --- seis de las siete casas
no cuesta nada cerca $364$.

\textbf{2.} Sin la mansión: \omterm{def:g10:stats:mean}{media} $\frac{1\,350}{6} = 225$,
\omterm{def:g10:stats:median}{mediana} $\frac{220 + 230}{2} = 225$. La \omterm{def:g10:stats:mean}{media} cayó aproximadamente
$139$; la \omterm{def:g10:stats:median}{mediana} por $5$. Un valor extremo puede arrastrar la \omterm{def:g10:stats:mean}{media}
en cualquier lugar; la \omterm{def:g10:stats:median}{mediana} apenas se da cuenta --- es \emph{robusto}.

\textbf{3.} \omterm{def:g10:stats:mean}{media} $= \frac{0 \times 20 + 1 \times 25 + 2 \times
30 + 3 \times 15 + 4 \times 10}{100} = \frac{170}{100} = 1.7$.
Ninguna familia tiene $1.7$ niños, pero $1.7 \times 100 = 170$
\emph{es} el número total exacto de niños: la \omterm{def:g10:stats:mean}{media} codifica
el total, redistribuido equitativamente --- su verdadero talento.

\textbf{4.} $ N = 15$: \omterm{def:g10:stats:median}{mediana} $=$ octavo valor $= 11$;
$ Q_1 =$ 4to valor $= 8$; $ Q_3 =$ valor 12 $= 14$.

\textbf{5.} recorrido $19 - 4 = 15$: la extensión completa, extremos
incluido. \omterm{def:g10:stats:quartiles}{rango} intercuartílico $14 - 8 = 6$: el ancho de la
mitad \omterm{def:g10:stats:mean}{media} --- extendido sin el ruido de los extremos.

\textbf{6.} un turno $+b $ se mueve \emph{cada} valor, entonces
diferencias de valores --- recorrido, IQR --- no cambian:
$(x_i + b) - (x_j + b) = x_i - x_j $. una escala por $ a $
multiplica todas las diferencias por $ a $: recorrido y el IQR se escalan por
$\abs a $.

\textbf{7.} Curva aditiva: $6 \to 9$ y $18 \to 21$ --- el
débiles logros estudiantiles $50\,\%$, el fuerte $17\,\%$, y $21$
estalla el $20$-Escala de puntos sólo levemente. Multiplicativo:
$6 \to 8$ y $18 \to 24$ --- proporciones preservadas, pero $24$ es
imposible en un $20$-escala: el problema del techo. aditivo
las curvas aplanan la parte inferior; los multiplicativos explotan el
arriba.

\textbf{8.} Notas totales: $20 \times 12 + 30 \times 10 = 540$,
entonces los fusionados \omterm{def:g10:stats:mean}{media} es $\frac{540}{50} = 10.8$: clase B treinta
los estudiantes superan a los veinte de la clase A --- significa fusionarse por
\emph{ponderado} promedio, nunca por promedio simple.

\textbf{9.} \omterm{def:g10:stats:median}{mediana} de $ A $: $2$; de $ B $: $5$. Lista fusionada
$\{0, 1, 2, 3, 10\}$: \omterm{def:g10:stats:median}{mediana} $2$. El promedio ponderado por tamaño de
el \omterm{def:g10:stats:median}{medianas} sería $\frac{3 \times 2 + 2 \times 5}{5} = 3.2
\neq 2$: \omterm{def:g10:stats:median}{medianas} no lleva total, por lo que no existe una fórmula de fusión
--- hay que reordenar toda la lista.

\textbf{10.} \omterm{def:g10:stats:mean}{media}: $\frac{31.3}{5} = 6.26$. \omterm{def:g10:stats:median}{mediana}: $5.4$.
recortado \omterm{def:g10:stats:mean}{media}: $\frac{5.3 + 5.4 + 5.5}{3} = 5.4$. uno
juez entusiasta (o corrupto) movió la \omterm{def:g10:stats:mean}{media} por casi un
punto; la \omterm{def:g10:stats:median}{mediana} y el recortado \omterm{def:g10:stats:mean}{media} se encogió de hombros --- de ahí el
Reglas de recorte de deportes juzgados.

\textbf{11.} Fábrica A: mitad \omterm{def:g10:stats:mean}{media} dentro $0.2$ milímetros; fábrica B:
dentro $4$ mm. Mismo \omterm{def:g10:stats:mean}{media}, confiabilidad opuesta: compre a A. El
\omterm{def:g10:stats:quartiles}{rango} intercuartílico decidido --- significa solo comparar centros,
nunca coherencia.

\textbf{12.} \omterm{def:g10:stats:median}{mediana} muy por debajo \omterm{def:g10:stats:mean}{media} traiciona a un \emph{largo a la derecha
cola}: la mayoría espera poco tiempo, algunas catastróficas. Ejemplo:
$\{5, 5, 10, 10, 95\}$ --- \omterm{def:g10:stats:median}{mediana} $10$, \omterm{def:g10:stats:mean}{media}
$\frac{125}{5} = 25$.

\textbf{13.} significa $75\,\%$ de los bebés de esa edad son
más corto (y $25\,\%$ más alto): un \omterm{def:g10:stats:quartiles}{rango}, no un tamaño. ``Cada
niño por encima de la \omterm{def:g10:stats:mean}{media}'' es imposible --- si se exceden todos los valores
la \omterm{def:g10:stats:mean}{media}, su promedio superaría la \omterm{def:g10:stats:mean}{media}, que es en sí mismo:
contradicción. Pero \emph{casi} todos pueden ser: uno muy corto
El niño puede sostener la \omterm{def:g10:stats:mean}{media} debajo de todos los demás. Por encima del
\emph{\omterm{def:g10:stats:median}{mediana}}: nunca más de la mitad, por definición. \omterm{def:g10:stats:quartiles}{Rangos}
no puede ser halagado; los promedios pueden.

\textbf{14.} Marca típica: igual, \omterm{def:g10:stats:median}{mediana} $11$ para ambos.
Homogeneidad del medio: B más ajustado (IQR $3$ contra $5$).
Extremos: B más salvaje (recorrido $17$ contra $13$). B es una clase de
estudiantes similares más algunos valores atípicos; A está distribuido uniformemente.

\textbf{15.} (1) es $5\,000$ la \omterm{def:g10:stats:mean}{media} o la \omterm{def:g10:stats:median}{mediana} --- y puede
tengo la \omterm{def:g10:stats:median}{mediana}? (2) ¿Cuál es el diferencial (cuartiles), entonces yo
¿Sabes lo típico que es lo "típico"? (3) Quién fue contado --- muestra
tamaño y selección (el problema del fin de semana de cómo se encuentran los datos de la lección de encuesta del volumen de la escuela secundaria)?

\textbf{16.} Por ejemplo $100, 150, 200, 250, 800$: \omterm{def:g10:stats:median}{mediana}
$200$, \omterm{def:g10:stats:mean}{media} $\frac{1\,500}{5} = 300$.

\textbf{17.} $\sum (x_i - \bar x) = \sum x_i - N \bar x =
N\bar x - N\bar x = 0$. Controlar: $-200 - 150 - 100 - 50 + 500 =
0$. La \omterm{def:g10:stats:mean}{media} es el punto donde las desviaciones se equilibran --- el
centro de gravedad de los datos (el problema del fin de semana del centroide del volumen de la escuela secundaria $ G $, en
una dimensión).

\textbf{18.} Al \omterm{def:g10:stats:median}{mediana} $4$: $3 + 1 + 0 + 5 + 9 = 18$ km. en
la \omterm{def:g10:stats:mean}{media} $6$: $5 + 3 + 2 + 3 + 7 = 20$ km. En $5$:
$4 + 2 + 1 + 4 + 8 = 19$ km. La \omterm{def:g10:stats:median}{mediana} gana --- alejándose
desde allí siempre pasa más tiendas de las que se acerca, por lo que el
el total sólo puede crecer.

\textbf{19.} $0.3^{10} \approx 6 \times 10^{-6}$: alrededor de seis
pilotos \emph{por millón}. En $4\,000$ pilotos lo esperado
contar es $0.02$ --- cero, como descubrió el teniente Daniels. el aire
Force dejó de diseñar para el hombre promedio y creó
todo ajustable: asientos, pedales, correas --- diseño para el
\emph{dispersión}, no el centro.

\textbf{20.} Totales: el \emph{\omterm{def:g10:stats:mean}{media}}. Individuo típico: el
\emph{\omterm{def:g10:stats:median}{mediana}}. Equidad de propagación: el \emph{cuartiles} (y
RIQ). Contaminación: la \emph{recortado \omterm{def:g10:stats:mean}{media}}. y el cierre
línea: la persona promedio no existe --- pero la \omterm{def:g10:stats:median}{mediana}
la persona casi lo hace.
\end{solution}
