\section*{Capítulo \ref{ch:g10:coordgeom} --- Geometría analítica}

\begin{solution}{exo:g10:coordgeom:1}
\omterm{prop:g10:coordgeom:midpoint}{Punto medio}: $\left(\frac{2 + (-4)}{2}, \frac{5 + 1}{2}\right) = (-1, 3)$.
Distancia:
\[
AB = \sqrt{(-4 - 2)^2 + (1 - 5)^2} = \sqrt{36 + 16} = \sqrt{52}
= 2\sqrt{13}.
\]
\end{solution}

\begin{solution}{exo:g10:coordgeom:2}
Que $I$ sea el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[AB]$ significa $\frac{3 + x_B}{2} = 1$ y
$\frac{-2 + y_B}{2} = 2$, luego $x_B = -1$ e $y_B = 6$: $B(-1, 6)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:3}
$OP = \sqrt{16 + 1} = \sqrt{17}$; $OQ = \sqrt{9 + 4} = \sqrt{13}$;
$OR = \sqrt{0 + 25} = 5 = \sqrt{25}$. Como $13 < 17 < 25$ y la raíz
cuadrada es \omterm{def:g10:functions:variations}{creciente}, $Q$ es el más cercano al origen.
\end{solution}

\begin{solution}{exo:g10:coordgeom:4}
Cuadrados de las longitudes:
\begin{align*}
AB^2 &= (5-1)^2 + (4-2)^2 = 16 + 4 = 20, \\
BC^2 &= (7-5)^2 + (0-4)^2 = 4 + 16 = 20, \\
AC^2 &= (7-1)^2 + (0-2)^2 = 36 + 4 = 40 .
\end{align*}
$AB = BC$: el triángulo es isósceles en $B$. Además
$AB^2 + BC^2 = 40 = AC^2$: por el recíproco del teorema de Pitágoras
también es rectángulo en $B$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:5}
\emph{1.} \omterm{prop:g10:coordgeom:midpoint}{Punto medio} de $[AC]$:
$\left(\frac{-2+4}{2}, \frac{1+1}{2}\right) = (1, 1)$; \omterm{prop:g10:coordgeom:midpoint}{punto medio} de
$[BD]$: $\left(\frac{1+1}{2}, \frac{3+(-1)}{2}\right) = (1, 1)$. Mismo
\omterm{prop:g10:coordgeom:midpoint}{punto medio}: $ABCD$ es un paralelogramo.

\emph{2.} $AB^2 = 3^2 + 2^2 = 13$ y $BC^2 = 3^2 + (-4)^2 = 25$:
$AB \neq BC$, no es un rombo.

\emph{3.} $AC^2 = 6^2 + 0^2 = 36$ y $BD^2 = 0^2 + (-4)^2 = 16$:
$AC \neq BD$, no es un rectángulo. $ABCD$ es un paralelogramo sin más.
\end{solution}

\begin{solution}{exo:g10:coordgeom:6}
Calculamos los cuadrados de las distancias a $\Omega(2,1)$ y los
comparamos con $5^2 = 25$:
\[
\Omega A^2 = 3^2 + 4^2 = 25, \qquad
\Omega B^2 = (-3)^2 + (-4)^2 = 25, \qquad
\Omega C^2 = 4^2 + 1^2 = 17 .
\]
$A$ y $B$ están sobre la circunferencia; $C$ está dentro ($17 < 25$).
\end{solution}

\begin{solution}{exo:g10:coordgeom:7}
$MA^2 = (x - 1)^2 + 1$ y $MB^2 = (x - 7)^2 + 25$. Igualando
$MA^2 = MB^2$:
\[
x^2 - 2x + 2 = x^2 - 14x + 74
\ \Longleftrightarrow\
12x = 72
\ \Longleftrightarrow\
x = 6 .
\]
El único punto de ese tipo es $M(6, 0)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:8}
\emph{1.} $I = \left(\frac{0+4}{2}, \frac{3+1}{2}\right) = (2, 2)$.

\emph{2.} $M(2,2)$ es efectivamente $I$. Y
$MA^2 = (0-2)^2 + (3-2)^2 = 5$, $MB^2 = (4-2)^2 + (1-2)^2 = 5$: el \omterm{prop:g10:coordgeom:midpoint}{punto
medio} equidista de los dos extremos, como era de esperar.

\emph{3.} Buscamos $N(0, y)$ con $NA^2 = NB^2$:
\[
(3 - y)^2 = 16 + (1 - y)^2
\ \Longleftrightarrow\
9 - 6y = 16 + 1 - 2y
\ \Longleftrightarrow\
-4y = 8,
\]
luego $y = -2$: el punto $N(0, -2)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:9}
\begin{align*}
AB &= \sqrt{4^2 + 2^2} = \sqrt{20} = 2\sqrt5, \\
BC &= \sqrt{8^2 + 4^2} = \sqrt{80} = 4\sqrt5, \\
AC &= \sqrt{12^2 + 6^2} = \sqrt{180} = 6\sqrt5 .
\end{align*}
Entonces $AB + BC = 2\sqrt5 + 4\sqrt5 = 6\sqrt5 = AC$: la desigualdad
triangular es una igualdad, así que $B$ está sobre el segmento $[AC]$; los
tres puntos están alineados.
\end{solution}

\begin{solution}{exo:g10:coordgeom:10}
El \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[AB]$ es $I\left(\frac a2, \frac b2\right)$. Entonces
\[
OI^2 = \frac{a^2}{4} + \frac{b^2}{4},
\qquad
IA^2 = \left(a - \frac a2\right)^2 + \left(0 - \frac b2\right)^2
= \frac{a^2}{4} + \frac{b^2}{4},
\]
y también $IB^2 = \left(0 - \frac a2\right)^2 +
\left(b - \frac b2\right)^2 = \frac{a^2}{4} + \frac{b^2}{4}$. Los tres
cuadrados de las distancias coinciden, luego $OI = IA = IB$: el \omterm{prop:g10:coordgeom:midpoint}{punto
medio} de la hipotenusa del triángulo rectángulo $OAB$ equidista de los
tres vértices (es el centro de su circunferencia circunscrita).
\end{solution}

\begin{solution}{pb:g10:coordgeom:1}
\textbf{1.} $\Omega M = 5 \iff \Omega M^2 = 25 \iff
(x - 2)^2 + (y - 1)^2 = 25$ por la fórmula de la distancia
(\cref{thm:g10:coordgeom:distance}); elevar al cuadrado es inofensivo,
porque los dos miembros son positivos.

\textbf{2.} $A$: $(5-2)^2 + (5-1)^2 = 9 + 16 = 25$: está sobre la
circunferencia. $B$: $16 + 9 = 25$: también. $D$: $4 + 9 = 13 < 25$: está
dentro.

\textbf{3.} $x^2 - 6x = (x - 3)^2 - 9$ y $y^2 + 4y = (y + 2)^2 - 4$, así
que la \omterm{def:g10:algebra:equation}{ecuación} queda $(x - 3)^2 + (y + 2)^2 = 25$: centro $(3, -2)$,
radio $5$.

\textbf{4.} $(x - 3)^2 + (y + 2)^2 = -1$: una suma de cuadrados nunca es
negativa, así que ningún punto la cumple: el conjunto vacío. En general,
$(x - h)^2 + (y - k)^2 = c$ es una circunferencia de radio $\sqrt c$ si
$c > 0$, el único punto $(h, k)$ si $c = 0$ y el vacío si $c < 0$.

\textbf{5.} Con $y = 3$: $(x - 3)^2 + 25 = 25$, luego $(x - 3)^2 = 0$: el
único punto $(3, 3)$. Un solo punto de contacto: la recta es
\emph{tangente} a la circunferencia.

\textbf{6.} $MA^2 + MB^2 = (x + r)^2 + y^2 + (x - r)^2 + y^2
= 2x^2 + 2y^2 + 2r^2 = 2r^2 + 2r^2 = 4r^2$ (usando $x^2 + y^2 = r^2$).
Como $AB^2 = (2r)^2 = 4r^2$, el triángulo $AMB$ cumple
$MA^2 + MB^2 = AB^2$: es rectángulo en $M$ (recíproco de Pitágoras); tres
líneas, tal como se prometía.

\textbf{7.} $MA^2 + MB^2 = (x + a)^2 + y^2 + (x - a)^2 + y^2
= 2x^2 + 2y^2 + 2a^2 = 2\,MI^2 + 2a^2$, y
$\frac{AB^2}{2} = \frac{(2a)^2}{2} = 2a^2$. Identidad demostrada.

\textbf{8.} $2\,MI^2 + \frac{36}{2} = 26$ da $MI^2 = 4$: la circunferencia
de centro el \omterm{prop:g10:coordgeom:midpoint}{punto medio} $I$ y radio $2$.

\textbf{9.} $MA^2 = 4\,MB^2$:
$x^2 + y^2 = 4\left((x - 3)^2 + y^2\right)$, es decir,
$3x^2 + 3y^2 - 24x + 36 = 0$, o bien $x^2 + y^2 - 8x + 12 = 0$;
completando cuadrados, $(x - 4)^2 + y^2 = 4$: la circunferencia de centro
$(4, 0)$ y radio $2$ (la \emph{circunferencia de Apolonio} de razón $2$).

\textbf{10.} El puente convierte las construcciones en cálculos: sin
rectángulo auxiliar y sin distinción de casos, un desarrollo y el teorema
cae solo. El precio: los cálculos llevan menos intuición geométrica; la
demostración clásica muestra el \emph{porqué} y la algebraica, el
\emph{que}. Un matemático en activo conserva las dos.

\textbf{11.} $x^2 + y^2 = 25$ y $(x - 6)^2 + y^2 = 25$. Restando:
$x^2 - (x - 6)^2 = 0$, es decir, $12x - 36 = 0$: $x = 3$. Después
$9 + y^2 = 25$: $y = \pm 4$. Candidatas: $(3, 4)$ y $(3, -4)$.

\textbf{12.} De $(3, 4)$ a $R(0, 8)$: $\sqrt{9 + 16} = 5$: encaja. De
$(3, -4)$: $\sqrt{9 + 144} \approx 12.4$: no encaja. Estás en $(3, 4)$.

\textbf{13.} Las dos ecuaciones llevan el mismo $x^2 + y^2$ (o con
coeficientes iguales tras \omterm{def:g10:algebra:expand}{desarrollar}): al restar se cancelan los
cuadrados y queda una \omterm{def:g10:algebra:equation}{ecuación} de primer grado, o sea, una recta. Cuando
las circunferencias se cortan dos veces, esa recta pasa por los dos puntos
de corte: es la recta de la cuerda común.

\textbf{14.} Cuatro incógnitas: las tres \omterm{def:g10:coordgeom:system}{coordenadas} más el error del
reloj del receptor (un reloj barato, corregido por las matemáticas). Cada
satélite aporta una \omterm{def:g10:algebra:equation}{ecuación}, así que cuatro satélites dan cuatro
ecuaciones para cuatro incógnitas; más satélites mejoran la precisión.

\textbf{15.} La resta da ahora
$x = \frac{5.1^2 - 5^2 + 36}{12} = \frac{37.01}{12} \approx 3.084$: un
error de $10$ cm en una distancia ha desplazado la estimación unos $8$ cm.
Los errores se propagan por todas las fórmulas: la prudencia de la
aritmética de \omterm{def:g10:numbers:interval}{intervalos} del \cref{pb:g10:numbers:1}, ahora con
\omterm{def:g10:coordgeom:system}{coordenadas}.

\textbf{16.} $B'(7, -3)$. La recta $(AB')$ tiene \omterm{def:g10:reffunc:affine}{pendiente}
$\frac{-3 - 5}{7 - 1} = -\frac43$ y corta a $y = 0$ en
$x = 1 + \frac{5}{8} \times 6 = 4.75$: hay que beber en $P(4.75,\ 0)$.
Longitud mínima: $AP + PB = AP + PB' = AB' = \sqrt{6^2 + 8^2} = 10$.

\textbf{17.} Pasando por $Q(3, 0)$:
$AQ + QB = \sqrt{4 + 25} + \sqrt{16 + 9} = \sqrt{29} + 5 \approx 10.39 >
10$. Gana la línea recta reflejada.

\textbf{18.} \omterm{prop:g10:coordgeom:midpoint}{Puntos medios} de las diagonales: $[AC]$: $(3, 3)$; $[BD]$:
$(3, 3)$; iguales: paralelogramo. Lados: $AB = \sqrt{10} = AD$: rombo.
Diagonales: $AC = \sqrt{32} \neq BD = \sqrt{8}$: no es un rectángulo y,
por tanto, tampoco un cuadrado. Veredicto: un rombo, y nada más.

\textbf{19.} $D = A + C - B = (-1 + 6 - 3,\ 2 + 0 - 4) = (2, -2)$:
entonces $[AC]$ y $[BD]$ comparten el \omterm{prop:g10:coordgeom:midpoint}{punto medio}
$\left(\frac52, 1\right)$.

\textbf{20.} Fórmula del \omterm{prop:g10:coordgeom:midpoint}{punto medio}: paralelogramos y centros (preguntas
18 y 19). Fórmula de la distancia: circunferencias, lugares geométricos,
naturaleza de triángulos, caminos más cortos (preguntas 1 a 9 y 16).
Resta de ecuaciones de circunferencia: la recta que localiza una posición
(preguntas 11 a 15). Lo que falta en la caja: un álgebra limpia de las
\emph{direcciones} (los vectores) y de las \emph{\omterm{def:g10:reffunc:affine}{pendientes}} (las
ecuaciones de rectas y los sistemas); los dos capítulos siguientes
proporcionan exactamente eso.
\end{solution}
