\begin{solution}{exo:g10:coordgeom:1}
\omterm{prop:g10:coordgeom:midpoint}{punto medio}: $\left(\frac{2 + (-4)}{2}, \frac{5 + 1}{2}\right) = (-1, 3)$.
Distancia:
\[
AB = \sqrt{(-4 - 2)^2 + (1 - 5)^2} = \sqrt{36 + 16} = \sqrt{52}
= 2\sqrt{13}.
\]
\end{solution}

\begin{solution}{exo:g10:coordgeom:2}
$ I $ es el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[AB]$ medio $\frac{3 + x_B}{2} = 1$ y
$\frac{-2 + y_B}{2} = 2$, entonces $ x_B = -1$ y $ y_B = 6$: $ B(-1, 6)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:3}
$ OP = \sqrt{16 + 1} = \sqrt{17}$; $ OQ = \sqrt{9 + 4} = \sqrt{13}$;
$ OR = \sqrt{0 + 25} = 5 = \sqrt{25}$. Desde $13 < 17 < 25$, la plaza
ser raíz \omterm{def:g10:functions:variations}{creciente}, $ Q $ es el más cercano al origen.
\end{solution}

\begin{solution}{exo:g10:coordgeom:4}
Longitudes al cuadrado:
\begin{align*}
AB^2 &= (5-1)^2 + (4-2)^2 = 16 + 4 = 20, \\
BC^2 &= (7-5)^2 + (0-4)^2 = 4 + 16 = 20, \\
AC^2 &= (7-1)^2 + (0-2)^2 = 36 + 4 = 40 .
\end{align*}
$ AB = BC $: el triángulo es isósceles en $ B $. Además
$ AB^2 + BC^2 = 40 = AC^2$: por el inverso del teorema de Pitágoras
tiene un ángulo recto en $ B $ también.
\end{solution}

\begin{solution}{exo:g10:coordgeom:5}
\emph{1.} \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[AC]$:
$\left(\frac{-2+4}{2}, \frac{1+1}{2}\right) = (1, 1)$; \omterm{prop:g10:coordgeom:midpoint}{punto medio} de
$[BD]$: $\left(\frac{1+1}{2}, \frac{3+(-1)}{2}\right) = (1, 1)$. Mismo
\omterm{prop:g10:coordgeom:midpoint}{punto medio}: $ ABCD $ es un paralelogramo.

\emph{2.} $ AB^2 = 3^2 + 2^2 = 13$ y $ BC^2 = 3^2 + (-4)^2 = 25$:
$ AB \neq BC $, no un rombo.

\emph{3.} $ AC^2 = 6^2 + 0^2 = 36$ y $ BD^2 = 0^2 + (-4)^2 = 16$:
$ AC \neq BD $, no un rectángulo. $ ABCD $ es un paralelogramo simple.
\end{solution}

\begin{solution}{exo:g10:coordgeom:6}
Calcular distancias al cuadrado a $\Omega(2,1)$ y comparar con $5^2 = 25$:
\[
\Omega A^2 = 3^2 + 4^2 = 25, \qquad
\Omega B^2 = (-3)^2 + (-4)^2 = 25, \qquad
\Omega C^2 = 4^2 + 1^2 = 17 .
\]
$ A $ y $ B $ están en el círculo; $ C $ está dentro de él ($17 < 25$).
\end{solution}

\begin{solution}{exo:g10:coordgeom:7}
$ MA^2 = (x - 1)^2 + 1$ y $ MB^2 = (x - 7)^2 + 25$. Configuración
$ MA^2 = MB^2$:
\[
x^2 - 2x + 2 = x^2 - 14x + 74
\ \Longleftrightarrow\
12x = 72
\ \Longleftrightarrow\
x = 6 .
\]
El único punto de este tipo es $ M(6, 0)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:8}
\emph{1.} $ I = \left(\frac{0+4}{2}, \frac{3+1}{2}\right) = (2, 2)$.

\emph{2.} $ M(2,2)$ es de hecho $ I $. Y
$ MA^2 = (0-2)^2 + (3-2)^2 = 5$, $ MB^2 = (4-2)^2 + (1-2)^2 = 5$: el
\omterm{prop:g10:coordgeom:midpoint}{punto medio} es equidistante de los dos puntos finales, como se esperaba.

\emph{3.} Buscar $ N(0, y)$ con $ NA^2 = NB^2$:
\[
(3 - y)^2 = 16 + (1 - y)^2
\ \Longleftrightarrow\
9 - 6y = 16 + 1 - 2y
\ \Longleftrightarrow\
-4y = 8,
\]
entonces $ y = -2$: el punto $ N(0, -2)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:9}
\begin{align*}
AB &= \sqrt{4^2 + 2^2} = \sqrt{20} = 2\sqrt5, \\
BC &= \sqrt{8^2 + 4^2} = \sqrt{80} = 4\sqrt5, \\
AC &= \sqrt{12^2 + 6^2} = \sqrt{180} = 6\sqrt5 .
\end{align*}
Entonces $ AB + BC = 2\sqrt5 + 4\sqrt5 = 6\sqrt5 = AC $: el triangulo
la desigualdad es una igualdad, entonces $ B $ se encuentra en el segmento $[AC]$ --- el
Tres puntos están alineados.
\end{solution}

\begin{solution}{exo:g10:coordgeom:10}
El \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[AB]$ es $ I\left(\frac a2, \frac b2\right)$. Entonces
\[
OI^2 = \frac{a^2}{4} + \frac{b^2}{4},
\qquad
IA^2 = \left(a - \frac a2\right)^2 + \left(0 - \frac b2\right)^2
= \frac{a^2}{4} + \frac{b^2}{4},
\]
y $ IB^2 = \left(0 - \frac a2\right)^2 + \left(b - \frac b2\right)^2 =
\frac{a^2}{4} + \frac{b^2}{4}$ también. Las tres distancias al cuadrado son
igual, entonces $ OI = IA = IB $: el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de la hipotenusa de la derecha
triangulo $ OAB $ es equidistante de los tres vértices (es el centro
de su círculo circunscrito).
\end{solution}

\begin{solution}{pb:g10:coordgeom:1}
\textbf{1.} $\Omega M = 5 \iff \Omega M^2 = 25 \iff
(x - 2)^2 + (y - 1)^2 = 25$ por la fórmula de la distancia
(\cref{thm:g10:coordgeom:distance}) --- elevar al cuadrado es inofensivo,
ambas partes son positivas.

\textbf{2.} $ A $: $(5-2)^2 + (5-1)^2 = 9 + 16 = 25$: en el
círculo. $ B $: $16 + 9 = 25$: en eso también. $ D $: $4 + 9 = 13 < 25$:
adentro.

\textbf{3.} $ x^2 - 6x = (x - 3)^2 - 9$ y
$ y^2 + 4y = (y + 2)^2 - 4$, entonces el \omterm{def:g10:algebra:equation}{ecuación} lee
$(x - 3)^2 + (y + 2)^2 = 25$: centro $(3, -2)$, radio $5$.

\textbf{4.} $(x - 3)^2 + (y + 2)^2 = -1$: una suma de cuadrados es
nunca negativo --- ningún punto lo satisface: el conjunto vacío. En general $(x - h)^2 + (y - k)^2 = c $ es un circulo de radio
$\sqrt c $ si $ c > 0$, el único punto $(h, k)$ si $ c = 0$, y
vacío si $ c < 0$.

\textbf{5.} Con $ y = 3$: $(x - 3)^2 + 25 = 25$, entonces
$(x - 3)^2 = 0$: el único punto $(3, 3)$. Un punto de contacto:
la linea es \emph{tangente} al círculo.

\textbf{6.} $ MA^2 + MB^2 = (x + r)^2 + y^2 + (x - r)^2 + y^2
= 2x^2 + 2y^2 + 2r^2 = 2r^2 + 2r^2 = 4r^2$ (usando
$ x^2 + y^2 = r^2$). Desde $ AB^2 = (2r)^2 = 4r^2$, el triangulo
$ AMB $ satisface $ MA^2 + MB^2 = AB^2$: en ángulo recto en $ M $
(conversación de Pitágoras) --- tres líneas, como se prometió.

\textbf{7.} $ MA^2 + MB^2 = (x + a)^2 + y^2 + (x - a)^2 + y^2
= 2x^2 + 2y^2 + 2a^2 = 2\,MI^2 + 2a^2$, y
$\frac{AB^2}{2} = \frac{(2a)^2}{2} = 2a^2$. Identidad demostrada.

\textbf{8.} $2\,MI^2 + \frac{36}{2} = 26$ da $ MI^2 = 4$: el
círculo de centro el \omterm{prop:g10:coordgeom:midpoint}{punto medio} $ I $ y radio $2$.

\textbf{9.} $ MA^2 = 4\,MB^2$:
$ x^2 + y^2 = 4\left((x - 3)^2 + y^2\right)$, es decir.\
$3x^2 + 3y^2 - 24x + 36 = 0$, o
$ x^2 + y^2 - 8x + 12 = 0$: completando,
$(x - 4)^2 + y^2 = 4$ --- el círculo del centro $(4, 0)$ y
radio $2$ (el \emph{círculo de Apolonio} de proporción $2$).

\textbf{10.} El puente convierte las construcciones en
cálculos: sin rectángulo auxiliar, sin análisis de caso --- uno
expansión y el teorema desaparece. El precio: cálculos
tener menos conocimiento geométrico; la prueba clásica muestra
\emph{por qué}, el algebraico muestra \emph{eso} --- un trabajo
El matemático se queda con ambos.

\textbf{11.} $ x^2 + y^2 = 25$ y $(x - 6)^2 + y^2 = 25$.
Restando: $ x^2 - (x - 6)^2 = 0$, es decir.\ $12x - 36 = 0$:
$ x = 3$. Entonces $9 + y^2 = 25$: $ y = \pm 4$. Candidatos
$(3, 4)$ y $(3, -4)$.

\textbf{12.} De $(3, 4)$ a $ R(0, 8)$:
$\sqrt{9 + 16} = 5$ --- partidos. De $(3, -4)$:
$\sqrt{9 + 144} \approx 12.4$ --- equivocado. estas en $(3, 4)$.

\textbf{13.} Ambos ecuaciones llevar lo mismo $ x^2 + y^2$ (o con
coeficientes iguales después de la expansión): la resta cancela la
cuadra y deja un \omterm{def:g10:algebra:equation}{ecuación} de grado uno --- una línea. cuando
los círculos se encuentran dos veces, esa línea pasa por ambos encuentros
puntos: es la línea del acorde común.

\textbf{14.} Cuatro incógnitas: tres \omterm{def:g10:coordgeom:system}{coordenadas} más el
error del reloj del receptor (un reloj barato, corregido por el
matemáticas). Cada satélite suministra uno \omterm{def:g10:algebra:equation}{ecuación}, entonces cuatro
los satélites dan cuatro ecuaciones para cuatro incógnitas --- con más
satélites mejorando la precisión.

\textbf{15.} La resta ahora da
$ x = \frac{5.1^2 - 5^2 + 36}{12} = \frac{37.01}{12} \approx
3.084$: a $10$ error de cm en una distancia movió la estimación
acerca de $8$ cm aquí. Los errores se propagan a través de cada fórmula ---
la precaución aritmética de \omterm{def:g10:numbers:interval}{intervalos} de \cref{pb:g10:numbers:1},
ahora con \omterm{def:g10:coordgeom:system}{coordenadas}.

\textbf{16.} $ B' (7, -3)$. la linea $(AB')$ tiene \omterm{def:g10:reffunc:affine}{pendiente}
$\frac{-3 - 5}{7 - 1} = -\frac43$ y cruces $ y = 0$ en
$ x = 1 + \frac{5}{8} \times 6 = 4.75$: beber en
$ P(4.75,\ 0)$. Longitud mínima: $ AP + PB = AP + PB' = AB' =
\sqrt{6^2 + 8^2} = 10$.

\textbf{17.} A través de $ Q(3, 0)$:
$ AQ + QB = \sqrt{4 + 25} + \sqrt{16 + 9} = \sqrt{29} + 5
\approx 10.39 > 10$. La línea recta reflejada gana.

\textbf{18.} Diagonal puntos medios: $[AC]$: $(3, 3)$; $[BD]$:
$(3, 3)$ --- igual: paralelogramo. Lados:
$ AB = \sqrt{10} = AD $: rombo. Diagonales:
$ AC = \sqrt{32} \neq BD = \sqrt{8}$: no es un rectángulo, por lo tanto
no un cuadrado. Veredicto: un rombo y nada más.

\textbf{19.} $ D = A + C - B = (-1 + 6 - 3,\ 2 + 0 - 4) =
(2, -2)$: entonces $[AC]$ y $[BD]$ compartir el \omterm{prop:g10:coordgeom:midpoint}{punto medio}
$\left(\frac52, 1\right)$.

\textbf{20.} \omterm{prop:g10:coordgeom:midpoint}{punto medio} fórmula: paralelogramos y centros
(preguntas 18, 19). Fórmula de distancia: círculos, lugares geométricos, triángulos
naturalezas, caminos más cortos (preguntas 1-9, 16). Resta de
circulo ecuaciones: la línea que localiza una posición (preguntas
11--15). Falta en el kit: un álgebra limpia de
\emph{instrucciones} --- vectores --- y de \emph{\omterm{def:g10:reffunc:affine}{pendientes}} ---
línea ecuaciones y sistemas: los dos capítulos siguientes proporcionan
exactamente eso.
\end{solution}
