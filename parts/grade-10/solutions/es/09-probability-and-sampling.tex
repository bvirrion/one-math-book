\section*{Capítulo \ref{ch:g10:proba} --- Probabilidad y muestreo}

\begin{solution}{exo:g10:proba:1}
Cada uno de los $6$ resultados tiene \omterm{def:g10:proba:distribution}{probabilidad} $\frac16$.
``Sacar un $6$'': $\frac16$.
``Número impar'' $= \{1, 3, 5\}$: $\frac36 = \frac12$.
``Como mucho $4$'' $= \{1,2,3,4\}$: $\frac46 = \frac23$.
``Sacar un $7$'': \omterm{def:g10:proba:events}{suceso} imposible, \omterm{def:g10:proba:distribution}{probabilidad} $0$.
\end{solution}

\begin{solution}{exo:g10:proba:2}
Hay $10$ canicas equiprobables.
$\P(\text{verde}) = \frac{5}{10} = \frac12$.
$\P(\text{no negra}) = 1 - \frac{2}{10} = \frac{8}{10} = \frac45$.
Verde y amarilla son \omterm{def:g10:proba:operations}{incompatibles}:
$\P(\text{verde o amarilla}) = \frac{5}{10} + \frac{3}{10} = \frac{8}{10}
= \frac45$.
\end{solution}

\begin{solution}{exo:g10:proba:3}
Sea $I$: ``estudia inglés'' y $A$: ``estudia alemán''. Entonces
$\P(I) = \frac{18}{30}$, $\P(A) = \frac{10}{30}$ y
$\P(I \cap A) = \frac{4}{30}$. Regla de la suma:
\[
\P(I \cup A) = \frac{18}{30} + \frac{10}{30} - \frac{4}{30}
= \frac{24}{30} = \frac45 .
\]
\omterm{def:g10:proba:operations}{Suceso contrario}: $\P(\text{ninguno}) = 1 - \frac45 = \frac15$.
\end{solution}

\begin{solution}{exo:g10:proba:4}
El \omterm{met:g10:proba:tree}{árbol} tiene cuatro caminos: CC, CX, XC, XX, cada uno de \omterm{def:g10:proba:distribution}{probabilidad}
$\frac12 \times \frac12 = \frac14$.
Dos caras: $\frac14$. Exactamente una cara (CX o XC):
$\frac24 = \frac12$. Al menos una cara:
$1 - \P(\text{XX}) = 1 - \frac14 = \frac34$.
\end{solution}

\begin{solution}{exo:g10:proba:5}
Primera extracción: roja con \omterm{def:g10:proba:distribution}{probabilidad} $\frac{4}{10}$ y blanca con
$\frac{6}{10}$. Segunda extracción \emph{sin} devolución: después de una
roja quedan $3$ rojas y $6$ blancas entre $9$; después de una blanca, $4$
rojas y $5$ blancas entre $9$.

Dos rojas: $\frac{4}{10} \times \frac39 = \frac{12}{90} = \frac{2}{15}$.

Colores distintos: caminos RB y BR:
\[
\frac{4}{10} \times \frac69 + \frac{6}{10} \times \frac49
= \frac{24}{90} + \frac{24}{90} = \frac{48}{90} = \frac{8}{15}.
\]
\end{solution}

\begin{solution}{exo:g10:proba:6}
\emph{1.} Cada dado muestra cada cara con \omterm{def:g10:proba:distribution}{probabilidad} $\frac16$, de forma
independiente, así que las $36$ parejas ordenadas son equiprobables; la
suma queda determinada por la pareja.

\emph{2.} Suma $7$: las parejas $(1,6), (2,5), (3,4), (4,3), (5,2),
(6,1)$, luego $\P = \frac{6}{36} = \frac16$. Suma $12$: solo $(6,6)$,
$\P = \frac{1}{36}$. Suma como mucho $4$: parejas que suman $2$, $3$ o
$4$: $(1,1)$; $(1,2), (2,1)$; $(1,3), (2,2), (3,1)$, o sea, seis parejas,
luego $\P = \frac{6}{36} = \frac16$.

\emph{3.} El número de parejas que dan cada suma crece de $1$ (suma $2$) a
$6$ (suma $7$) y decrece después: la suma más probable es $7$.
\end{solution}

\begin{solution}{exo:g10:proba:7}
Cada pregunta se acierta con \omterm{def:g10:proba:distribution}{probabilidad} $\frac14$, de forma
independiente. \omterm{met:g10:proba:tree}{Árbol} de dos niveles:

Las dos correctas: $\frac14 \times \frac14 = \frac{1}{16}$.

Exactamente una correcta: caminos acierto--fallo y fallo--acierto:
$\frac14 \times \frac34 + \frac34 \times \frac14 = \frac{6}{16}
= \frac38$.

Ninguna correcta: $\frac34 \times \frac34 = \frac{9}{16}$.

Comprobación: $\frac{1}{16} + \frac{6}{16} + \frac{9}{16} = 1$.
\end{solution}

\begin{solution}{exo:g10:proba:8}
\emph{1.} Frecuencia observada: $\frac{396}{900} = 0.44$. Con $n = 900$:
$\frac{1}{\sqrt{900}} = \frac{1}{30} \approx 0.033$, así que el \omterm{def:g10:numbers:interval}{intervalo}
de fluctuación es aproximadamente $\intcc{0.467}{0.533}$.

\emph{2.} El $0.44$ observado queda fuera del \omterm{def:g10:numbers:interval}{intervalo}: las \omterm{def:g10:proba:sample}{muestras} de
$900$ votantes de una población con un $50\,\%$ de aprobación rara vez se
alejan tanto. La encuesta pone seriamente en duda el $50\,\%$ afirmado.
\end{solution}

\begin{solution}{exo:g10:proba:9}
\emph{1.} $\P(\text{ganar } 6) = \frac16$ (sacar un $6$),
$\P(\text{ganar } 3) = \frac16$ (sacar un $5$),
$\P(\text{ganar } 0) = \frac46 = \frac23$.

\emph{2.} A lo largo de $600$ partidas cabe esperar unos $100$ seises,
$100$ cincos y $400$ tiradas restantes. Ganancias: unos
$100 \times 6 + 100 \times 3 = 900$. Coste: $600 \times 2 = 1200$.
Resultado neto esperado: $900 - 1200 = -300$, es decir, una pérdida \omterm{def:g10:stats:mean}{media}
de unos $0.5$ por partida: no merece la pena jugar.
\end{solution}

\begin{solution}{pb:g10:proba:1}
\textbf{1.} $P(\text{rey o corazones}) = \frac{4}{52} +
\frac{13}{52} - \frac{1}{52} = \frac{16}{52} = \frac{4}{13}$: el rey de
corazones está en los dos \omterm{def:g10:proba:events}{sucesos} y no debe contarse dos veces; de ahí la
resta (\cref{thm:g10:proba:addition}).

\textbf{2.} $P(\text{ninguna cara}) = \left(\frac12\right)^3 = \frac18$,
luego $P(\text{al menos una}) = \frac78$.

\textbf{3.} $P(RR) = \frac35 \times \frac24 = \frac{3}{10}$;
$P(\text{una de cada}) = \frac35 \times \frac24 + \frac25
\times \frac34 = \frac{6}{20} + \frac{6}{20} = \frac35$.

\textbf{4.} $P(AA) = \frac25 \times \frac14 = \frac1{10}$; y
$\frac{3}{10} + \frac{6}{10} + \frac{1}{10} = 1$: las hojas del \omterm{met:g10:proba:tree}{árbol}
suman siempre $1$.

\textbf{5.} Los \omterm{def:g10:proba:operations}{sucesos incompatibles} cumplen $P(A \cup B) = P(A) +
P(B) = 1.1 > 1$: imposible; alguien se ha inventado \omterm{def:g10:proba:distribution}{probabilidad}.

\textbf{6.} (Casi todo el mundo, incluidos los mil autores de cartas,
responde ``da igual, 50--50''. Guarda tu respuesta para la pregunta 8.)

\textbf{7.} Coche tras la puerta 1 (\omterm{def:g10:proba:distribution}{probabilidad} $\frac13$): el
presentador abre cualquiera de las dos puertas con cabra y cambiar
abandona el coche: \emph{pierdes}. Coche tras la puerta 2 ($\frac13$): el
presentador \emph{tiene} que abrir la puerta 3 y cambiar aterriza en la
2: \emph{ganas}. Coche tras la puerta 3: simétrico: \emph{ganas}. En
total, $P(\text{ganar cambiando}) = \frac13 + \frac13 = \frac23$.

\textbf{8.} Tu primera elección es equivocada con \omterm{def:g10:proba:distribution}{probabilidad} $\frac23$
y, precisamente entonces, la apertura forzada del presentador deja el
coche tras la puerta restante: cambiar convierte cada error inicial en una
victoria. $\frac23$, dijera lo que dijera el instinto.

\textbf{9.} Seis escenarios equiprobables: el coche tras la $1$, la $2$ o
la $3$, y el presentador abre la puerta $2$ o la $3$ (nunca la tuya). En
dos de ellos se descubre el coche (coche en 2 / abre 2; coche en 3 / abre
3): se descartan. Entre los cuatro supervivientes (coche en 1 / abre 2,
coche en 1 / abre 3, coche en 2 / abre 3, coche en 3 / abre 2), cambiar
gana en los dos últimos: $\frac24 = \frac12$. Con un presentador
ignorante, el 50--50 \emph{sí} es correcto: la paradoja vive entera en lo
que sabe el presentador, que filtra los escenarios de forma desigual.

\textbf{10.} Tu elección esconde el coche con \omterm{def:g10:proba:distribution}{probabilidad}
$\frac{1}{100}$; las $98$ aperturas del presentador que sabe canalizan el
$\frac{99}{100}$ restante hacia la única puerta cerrada. Cambia y ganarás
$99$ veces de cada $100$.

\textbf{11.} Tres cartas (un as y dos comodines) y un amigo que mira y
descarta siempre un comodín de las dos que no has elegido; juega $300$
rondas cambiando siempre y anota las victorias. Frecuencia esperada
$\frac23 \approx 0.667$, con una fluctuación de
$\pm\frac{1}{\sqrt{300}} \approx 0.058$: cabe esperar entre un $61\,\%$ y
un $72\,\%$ de victorias al cambiar, muy lejos de cualquier ``50--50''.

\textbf{12.} Configuraciones equiprobables (mayor--menor): CC, CH, HC,
HH, escribiendo C para chico y H para chica. ``Al menos un chico''
conserva CC, CH y HC: $P(\text{CC}) = \frac13$.

\textbf{13.} ``El mayor es un chico'' conserva CC y CH:
$P(\text{CC}) = \frac12$. Informaciones distintas filtran conjuntos de
escenarios distintos, exactamente como hacían los dos Monty: importa lo
que averiguas y, otro tanto, \emph{cómo} lo averiguaste.

\textbf{14.} Todas iguales: CCC y XXX: $\frac28 = \frac14$. El error de
d'Alembert: sus dos ``casos'' no son equiprobables; ``no todas iguales''
agrupa seis de los ocho resultados. La
\cref{prop:g10:proba:uniform} solo cuenta sobre resultados
equiprobables.

\textbf{15.} Ganancia \omterm{def:g10:stats:mean}{media} por partida: $6 \times \frac16 + 3
\times \frac16 = 1.50$ euros, frente a una apuesta de $2$ euros: una
pérdida \omterm{def:g10:stats:mean}{media} de $0.50$ por partida, unos $300$ euros a lo largo de $600$
partidas. La prima justa iguala el pago medio:
$1\,\% \times 10\,000 = 100$ euros; las primas reales añaden gastos y un
margen de seguridad, porque la aseguradora tiene que sobrevivir a los
malos años, no solo a los años medios.

\textbf{16.} \omterm{def:g10:numbers:interval}{Intervalo}: $0.5 \pm \frac{1}{\sqrt{100}} =
\intcc{0.4}{0.6}$. $55$ caras ($0.55$): dentro, nada llamativo. $70$
caras ($0.70$): muy fuera; hay que dudar de la moneda.

\textbf{17.} $\frac{1}{\sqrt{1000}} \approx 0.032$: un candidato que
estuviera realmente en el $50\,\%$ saldría normalmente en
$\intcc{0.468}{0.532}$, y $0.52$ queda cómodamente dentro. La encuesta
\emph{sugiere} una ventaja, pero no demuestra nada: el margen se la traga.

\textbf{18.} Primera \omterm{def:g10:proba:sample}{muestra}: \omterm{def:g10:numbers:interval}{intervalo}
$0.04 \pm \frac{1}{20} = \intcc{0}{0.09}$; el $0.07$ observado está
dentro: sin veredicto. Segunda: $0.04 \pm \frac{1}{50} =
\intcc{0.02}{0.06}$; el $0.07$ observado cae fuera: ahora la afirmación
queda en serio entredicho. La misma frecuencia, con un instrumento más
fino: la precisión crece como $\sqrt n$.

\textbf{19.} La fluctuación se encoge como $\frac{1}{\sqrt n}$: cuadruplica
la \omterm{def:g10:proba:sample}{muestra} y reducirás el ruido a la mitad. El sesgo, preguntar al público
equivocado, sobrevive a cualquier $n$: dos millones de papeletas mal
elegidas siguieron estando mal (problema de fin de semana del volumen
anterior sobre cómo mienten los datos).

\textbf{20.} (i) Comprobar que los resultados son equiprobables antes de
contar; d'Alembert lo olvidó. (ii) Los \omterm{met:g10:proba:tree}{árboles} multiplican a lo largo de
las ramas, y los ``al menos uno'' se rinden ante el \omterm{def:g10:proba:operations}{suceso contrario}.
(iii) La información filtra escenarios, y su \emph{procedencia} cambia el
filtro: el Monty que sabe frente al Monty distraído, el hijo mayor frente
a algún hijo. (iv) Las frecuencias se acercan a las \omterm{def:g10:proba:distribution}{probabilidades} a
velocidad $\frac{1}{\sqrt n}$, así que una simulación paciente zanja
cualquier discusión, incluida una con diez mil autores de cartas
indignados.
\end{solution}
