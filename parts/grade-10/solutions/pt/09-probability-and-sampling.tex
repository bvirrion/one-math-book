\section*{Capítulo \ref{ch:g10:proba} --- Probabilidade e amostragem}

\begin{solution}{exo:g10:proba:1}
Cada um dos $6$ resultados tem \omterm{def:g10:proba:distribution}{probabilidade} $\frac16$.
``Obter um $6$'': $\frac16$.
``Número ímpar'' $= \{1, 3, 5\}$: $\frac36 = \frac12$.
``No máximo $4$'' $= \{1,2,3,4\}$: $\frac46 = \frac23$.
``Obter um $7$'': \omterm{def:g10:proba:events}{evento} impossível, \omterm{def:g10:proba:distribution}{probabilidade} $0$.
\end{solution}

\begin{solution}{exo:g10:proba:2}
Há $10$ bolinhas igualmente prováveis.
$\P(\text{verde}) = \frac{5}{10} = \frac12$.
$\P(\text{não preta}) = 1 - \frac{2}{10} = \frac{8}{10} = \frac45$.
Verde e amarela são \omterm{def:g10:proba:operations}{incompatíveis}:
$\P(\text{verde ou amarela}) = \frac{5}{10} + \frac{3}{10} =
\frac{8}{10} = \frac45$.
\end{solution}

\begin{solution}{exo:g10:proba:3}
Sejam $S$: ``estuda espanhol'' e $G$: ``estuda alemão''. Então
$\P(S) = \frac{18}{30}$, $\P(G) = \frac{10}{30}$,
$\P(S \cap G) = \frac{4}{30}$. Regra da adição:
\[
\P(S \cup G) = \frac{18}{30} + \frac{10}{30} - \frac{4}{30}
= \frac{24}{30} = \frac45 .
\]
\omterm{def:g10:proba:operations}{Complementar}: $\P(\text{nenhuma}) = 1 - \frac45 = \frac15$.
\end{solution}

\begin{solution}{exo:g10:proba:4}
A \omterm{met:g10:proba:tree}{árvore} tem quatro caminhos CaCa, CaCo, CoCa, CoCo, cada um de
\omterm{def:g10:proba:distribution}{probabilidade} $\frac12 \times \frac12 = \frac14$.
Duas caras: $\frac14$. Exatamente uma cara (CaCo ou CoCa):
$\frac24 = \frac12$. Pelo menos uma cara:
$1 - \P(\text{CoCo}) = 1 - \frac14 = \frac34$.
\end{solution}

\begin{solution}{exo:g10:proba:5}
Primeira retirada: vermelha com \omterm{def:g10:proba:distribution}{probabilidade} $\frac{4}{10}$, branca
$\frac{6}{10}$. Segunda retirada \emph{sem} reposição: depois de uma
vermelha restam $3$ vermelhas e $6$ brancas entre $9$; depois de uma
branca, $4$ vermelhas e $5$ brancas entre $9$.

Duas vermelhas:
$\frac{4}{10} \times \frac39 = \frac{12}{90} = \frac{2}{15}$.

Cores diferentes: caminhos VB e BV:
\[
\frac{4}{10} \times \frac69 + \frac{6}{10} \times \frac49
= \frac{24}{90} + \frac{24}{90} = \frac{48}{90} = \frac{8}{15}.
\]
\end{solution}

\begin{solution}{exo:g10:proba:6}
\emph{1.} Cada dado mostra cada face, de forma independente, com
\omterm{def:g10:proba:distribution}{probabilidade} $\frac16$, de modo que os $36$ pares ordenados são
igualmente prováveis; a soma fica determinada pelo par.

\emph{2.} Soma $7$: os pares $(1,6), (2,5), (3,4), (4,3), (5,2),
(6,1)$, logo $\P = \frac{6}{36} = \frac16$. Soma $12$: só $(6,6)$,
$\P = \frac{1}{36}$. Soma no máximo $4$: pares que somam $2$, $3$ ou
$4$: $(1,1)$; $(1,2), (2,1)$; $(1,3), (2,2), (3,1)$ --- seis pares,
logo $\P = \frac{6}{36} = \frac16$.

\emph{3.} O número de pares que produzem cada soma cresce de $1$ (soma
$2$) até $6$ (soma $7$) e depois decresce: a soma mais provável é $7$.
\end{solution}

\begin{solution}{exo:g10:proba:7}
Cada questão é acertada com \omterm{def:g10:proba:distribution}{probabilidade} $\frac14$, de forma
independente. \omterm{met:g10:proba:tree}{Árvore} de dois níveis:

As duas certas: $\frac14 \times \frac14 = \frac{1}{16}$.

Exatamente uma certa: caminhos certa--errada e errada--certa:
$\frac14 \times \frac34 + \frac34 \times \frac14 = \frac{6}{16}
= \frac38$.

Nenhuma certa: $\frac34 \times \frac34 = \frac{9}{16}$.

Verificação: $\frac{1}{16} + \frac{6}{16} + \frac{9}{16} = 1$.
\end{solution}

\begin{solution}{exo:g10:proba:8}
\emph{1.} \omterm{def:g10:stats:series}{Frequência} observada: $\frac{396}{900} = 0.44$. Com $n = 900$:
$\frac{1}{\sqrt{900}} = \frac{1}{30} \approx 0.033$, de modo que o
\omterm{def:g10:numbers:interval}{intervalo} de flutuação é cerca de $\intcc{0.467}{0.533}$.

\emph{2.} O valor observado $0.44$ está fora do \omterm{def:g10:numbers:interval}{intervalo}: \omterm{def:g10:proba:sample}{amostras} de
$900$ eleitores tiradas de uma população com $50\%$ de aprovação
raramente se afastam tanto. A pesquisa lança séria dúvida sobre os
$50\%$ alegados.
\end{solution}

\begin{solution}{exo:g10:proba:9}
\emph{1.} $\P(\text{ganhar } 6) = \frac16$ (sair um $6$),
$\P(\text{ganhar } 3) = \frac16$ (sair um $5$),
$\P(\text{ganhar } 0) = \frac46 = \frac23$.

\emph{2.} Em $600$ partidas, esperam-se cerca de $100$ seis, $100$ cinco
e $400$ outros resultados. Ganhos: cerca de
$100 \times 6 + 100 \times 3 = 900$. Custo: $600 \times 2 = 1200$.
Resultado líquido esperado: $900 - 1200 = -300$, isto é, uma perda \omterm{def:g10:stats:mean}{média}
de cerca de $0.5$ por partida --- não vale a pena jogar.
\end{solution}

\begin{solution}{pb:g10:proba:1}
\textbf{1.} $P(\text{rei ou copas}) = \frac{4}{52} +
\frac{13}{52} - \frac{1}{52} = \frac{16}{52} = \frac{4}{13}$: o
rei de copas está nos dois \omterm{def:g10:proba:events}{eventos} e não pode ser contado duas
vezes --- daí a subtração
(\cref{thm:g10:proba:addition}).

\textbf{2.} $P(\text{nenhuma cara}) = \left(\frac12\right)^3 =
\frac18$, logo $P(\text{pelo menos uma}) = \frac78$.

\textbf{3.} $P(\text{VV}) = \frac35 \times \frac24 =
\frac{3}{10}$; $P(\text{uma de cada}) = \frac35 \times \frac24 +
\frac25 \times \frac34 = \frac{6}{20} + \frac{6}{20} =
\frac35$.

\textbf{4.} $P(\text{AA}) = \frac25 \times \frac14 =
\frac1{10}$; e $\frac{3}{10} + \frac{6}{10} + \frac{1}{10} = 1$:
as folhas da \omterm{met:g10:proba:tree}{árvore} sempre somam $1$.

\textbf{5.} \omterm{def:g10:proba:events}{Eventos} \omterm{def:g10:proba:operations}{incompatíveis} obedecem a
$P(A \cup B) = P(A) + P(B) = 1.1 > 1$: impossível --- alguma
\omterm{def:g10:proba:distribution}{probabilidade} foi inventada.

\textbf{6.} (A maioria das pessoas, inclusive os mil autores das
cartas, responde ``tanto faz, 50--50''. Guarde sua resposta
para a questão 8.)

\textbf{7.} Carro atrás da porta 1 (\omterm{def:g10:proba:distribution}{probabilidade} $\frac13$): o
apresentador abre qualquer uma das portas com bode; trocar
abandona o carro: \emph{perde}. Carro atrás da porta 2
($\frac13$): o apresentador \emph{tem de} abrir a porta 3;
trocar cai na porta 2: \emph{ganha}. Carro atrás da porta 3:
simétrico: \emph{ganha}. Total:
$P(\text{trocar ganha}) = \frac13 + \frac13 = \frac23$.

\textbf{8.} Sua primeira escolha está errada com \omterm{def:g10:proba:distribution}{probabilidade}
$\frac23$ --- e é precisamente aí que a revelação forçada do
apresentador deixa o carro atrás da porta restante: trocar
converte todo erro inicial em vitória. $\frac23$, seja lá o que
tenha dito a intuição.

\textbf{9.} Seis cenários igualmente prováveis: carro atrás de
$1$, $2$ ou $3$, apresentador abre a porta $2$ ou a $3$ (nunca
a sua). A revelação mostra o carro em dois deles
(carro em 2/abre 2 e carro em 3/abre 3): descartados. Entre os
quatro sobreviventes --- carro em 1/abre 2, carro em 1/abre 3,
carro em 2/abre 3, carro em 3/abre 2 --- trocar ganha nos dois
últimos: $\frac24 = \frac12$. Com um apresentador ignorante,
50--50 \emph{está} correto: o paradoxo mora inteiramente no
conhecimento do apresentador, que filtra os cenários de modo
desigual.

\textbf{10.} Sua escolha guarda o carro com \omterm{def:g10:proba:distribution}{probabilidade}
$\frac{1}{100}$; as $98$ revelações do apresentador que sabe
canalizam os $\frac{99}{100}$ restantes para a única porta
fechada. Troque e ganhe $99$ vezes em $100$.

\textbf{11.} Três cartas (um ás e dois curingas), um amigo que
espia e sempre descarta um curinga dentre as duas que você não
escolheu; jogue $300$ rodadas trocando sempre e registre as
vitórias. \omterm{def:g10:stats:series}{Frequência} esperada $\frac23 \approx 0.667$, com
flutuação $\pm\frac{1}{\sqrt{300}} \approx 0.058$: espere entre
cerca de $61\,\%$ e $72\,\%$ de vitórias ao trocar --- muito
longe de qualquer ``50--50''.

\textbf{12.} Configurações igualmente prováveis (mais
velho--mais novo): MM, MF, FM, FF, escrevendo M para menino e F
para menina. ``Pelo menos um menino'' mantém MM, MF, FM:
$P(\text{MM}) = \frac13$.

\textbf{13.} ``O mais velho é menino'' mantém MM, MF:
$P(\text{MM}) = \frac12$. Informações diferentes filtram
conjuntos diferentes de cenários --- exatamente como fizeram os
dois Montys: o que você aprende importa, e \emph{como} você
aprendeu importa igualmente.

\textbf{14.} Todas iguais: CaCaCa e CoCoCo: $\frac28 = \frac14$.
O erro de d'Alembert: seus dois ``casos'' não são igualmente
prováveis --- ``nem todas iguais'' agrupa seis dos oito
resultados. A \cref{prop:g10:proba:uniform} só conta sobre
resultados igualmente prováveis.

\textbf{15.} Ganho médio por partida: $6 \times \frac16 + 3
\times \frac16 = 1.50$ euros, contra uma aposta de $2$ euros:
uma perda \omterm{def:g10:stats:mean}{média} de $0.50$ por partida, cerca de $300$ euros em
$600$ partidas. O prêmio justo iguala o pagamento médio:
$1\,\% \times 10\,000 = 100$ euros; os prêmios reais acrescentam
custos e uma margem de segurança --- a seguradora precisa
sobreviver aos anos ruins, não apenas aos anos médios.

\textbf{16.} \omterm{def:g10:numbers:interval}{Intervalo}: $0.5 \pm \frac{1}{\sqrt{100}} =
\intcc{0.4}{0.6}$. $55$ caras ($0.55$): dentro, nada de
notável. $70$ caras ($0.70$): bem fora --- desconfie da moeda.

\textbf{17.} $\frac{1}{\sqrt{1000}} \approx 0.032$: um candidato
realmente em $50\,\%$ apareceria tipicamente em
$\intcc{0.468}{0.532}$ --- e $0.52$ fica confortavelmente
dentro. A pesquisa \emph{sugere} uma vantagem, mas nada prova:
a margem a engole.

\textbf{18.} Primeira \omterm{def:g10:proba:sample}{amostra}: \omterm{def:g10:numbers:interval}{intervalo}
$0.04 \pm \frac{1}{20} = \intcc{0}{0.09}$; o valor observado
$0.07$ está dentro: nenhum veredicto. Segunda:
$0.04 \pm \frac{1}{50} = \intcc{0.02}{0.06}$; o valor observado
$0.07$ cai fora: a afirmação passa a ser seriamente duvidosa.
Mesma \omterm{def:g10:stats:series}{frequência}, instrumento mais fino: a precisão cresce como
$\sqrt n$.

\textbf{19.} A flutuação encolhe como $\frac{1}{\sqrt n}$:
quadruplique a \omterm{def:g10:proba:sample}{amostra} e reduza o ruído à metade. O viés ---
perguntar à multidão errada --- sobrevive a qualquer $n$: dois
milhões de cédulas mal escolhidas continuaram erradas (o
problema de fim de semana sobre como os dados mentem, no volume
do ensino fundamental).

\textbf{20.} (i) Verifique se os resultados são igualmente
prováveis antes de contar --- d'Alembert esqueceu. (ii) As
\omterm{met:g10:proba:tree}{árvores} multiplicam ao longo dos ramos; os ``pelo menos um''
rendem-se ao \omterm{def:g10:proba:operations}{complementar}. (iii) A informação filtra cenários,
e a \emph{origem} dela muda o filtro --- Monty que sabe contra
Monty distraído, o mais velho menino contra algum menino.
(iv) As \omterm{def:g10:stats:series}{frequências} se aproximam das \omterm{def:g10:proba:distribution}{probabilidades} à
velocidade $\frac{1}{\sqrt n}$, de modo que uma simulação
paciente encerra qualquer discussão --- inclusive uma com dez
mil autores de cartas indignados.
\end{solution}
