\section*{Capítulo \ref{ch:g10:stats} --- Estatística descritiva}

\begin{solution}{exo:g10:stats:1}
Soma: $0+1+1+2+2+2+3+3+4+4+5+6+7+8+12 = 60$, logo a \omterm{def:g10:stats:mean}{média} é
$\frac{60}{15} = 4$. A série já está ordenada e $N = 15$ é ímpar: a
\omterm{def:g10:stats:median}{mediana} é o $8^{\text{o}}$ valor, $3$. \omterm{def:g10:stats:quartiles}{Amplitude}: $12 - 0 = 12$.
\end{solution}

\begin{solution}{exo:g10:stats:2}
\omterm{def:g10:stats:series}{Frequências}: $\frac{7}{50} = 0.14$, $\frac{9}{50} = 0.18$,
$\frac{8}{50} = 0.16$, $\frac{10}{50} = 0.20$, $\frac{6}{50} = 0.12$,
$\frac{10}{50} = 0.20$ (soma $1$). Resultado médio:
\[
\bar x = \frac{7 \times 1 + 9 \times 2 + 8 \times 3 + 10 \times 4
+ 6 \times 5 + 10 \times 6}{50}
= \frac{7 + 18 + 24 + 40 + 30 + 60}{50}
= \frac{179}{50} = 3.58 .
\]
\end{solution}

\begin{solution}{exo:g10:stats:3}
$N = 12$ (par): a \omterm{def:g10:stats:median}{mediana} é o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} entre o $6^{\text{o}}$ e o
$7^{\text{o}}$ valores, ambos iguais a $6$: \omterm{def:g10:stats:median}{mediana} $6$.

$Q_1$: $\frac{12}{4} = 3$, logo $Q_1$ é o $3^{\text{o}}$ valor: $4$.

$Q_3$: $\frac{3 \times 12}{4} = 9$, logo $Q_3$ é o $9^{\text{o}}$ valor:
$8$.

Diagrama de caixa: caixa de $4$ a $8$ com a barra da \omterm{def:g10:stats:median}{mediana} em $6$ e
bigodes de $1$ até $11$.
\end{solution}

\begin{solution}{exo:g10:stats:4}
As $24$ notas somam $24 \times 11.5 = 276$. Com a nova nota, o total é
$276 + 19 = 295$ para $25$ alunos: nova \omterm{def:g10:stats:mean}{média} $\frac{295}{25} = 11.8$.
\end{solution}

\begin{solution}{exo:g10:stats:5}
Total atual: $5 \times 12 = 60$. Uma \omterm{def:g10:stats:mean}{média} de $13$ em $6$ provas exige um
total de $78$, de modo que a $6^{\text{a}}$ nota deve ser
$78 - 60 = 18$. Uma \omterm{def:g10:stats:mean}{média} de $14$ exigiria $84 - 60 = 24 > 20$: não é
alcançável.
\end{solution}

\begin{solution}{exo:g10:stats:6}
\omterm{def:g10:stats:mean}{Média} maior que $20$: por exemplo,
$1,\ 2,\ 3,\ 10,\ 40,\ 50,\ 60$ --- \omterm{def:g10:stats:median}{mediana} $10$ (o $4^{\text{o}}$ valor
ordenado), \omterm{def:g10:stats:mean}{média} $\frac{166}{7} \approx 23.7$.

\omterm{def:g10:stats:mean}{Média} menor que $6$: por exemplo,
$0,\ 0,\ 0,\ 10,\ 10,\ 10,\ 10$ --- \omterm{def:g10:stats:median}{mediana} $10$, \omterm{def:g10:stats:mean}{média}
$\frac{40}{7} \approx 5.7$.

A \omterm{def:g10:stats:mean}{média} não pode ficar abaixo de $\frac{40}{7}$: para que a \omterm{def:g10:stats:median}{mediana} (o
$4^{\text{o}}$ valor ordenado) seja $10$, os valores nas posições $4$ a
$7$ devem ser todos $\geq 10$, de modo que o total é pelo menos $40$ e a
\omterm{def:g10:stats:mean}{média} pelo menos $\frac{40}{7}$ --- o segundo exemplo é extremal.

Esses exemplos mostram que a \omterm{def:g10:stats:mean}{média} e a \omterm{def:g10:stats:median}{mediana} são em larga medida
independentes: conhecer uma diz pouco sobre a outra, e é por isso que um
bom resumo informa as duas.
\end{solution}

\begin{solution}{exo:g10:stats:7}
\emph{1.} \omterm{def:g10:stats:mean}{Média}:
$\frac{9 \times 1800 + 12000}{10} = \frac{16200 + 12000}{10} = 2820$.
\omterm{def:g10:stats:median}{Mediana}: ordenados, os $10$ salários são nove valores $1800$ e depois
$12000$; a \omterm{def:g10:stats:median}{mediana} é o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} entre o $5^{\text{o}}$ e o
$6^{\text{o}}$ valores, ambos $1800$: \omterm{def:g10:stats:median}{mediana} $1800$.

\emph{2.} A \omterm{def:g10:stats:median}{mediana} ($1800$) descreve o salário típico: é o que $9$ em
cada $10$ funcionários efetivamente ganham. A \omterm{def:g10:stats:mean}{média} ($2820$) é puxada
para cima pelo único salário alto e não corresponde ao contracheque de
ninguém.
\end{solution}

\begin{solution}{exo:g10:stats:8}
Altura total: $12 \times 168 + 18 \times 163 = 2016 + 2934 = 4950$ cm
para $30$ alunos, de modo que a \omterm{def:g10:stats:mean}{média} é $\frac{4950}{30} = 165$ cm. Ela
fica mais perto da \omterm{def:g10:stats:mean}{média} das meninas porque elas são mais numerosas
(\omterm{def:g10:stats:mean}{média} ponderada, com pesos $12$ e $18$).
\end{solution}

\begin{solution}{exo:g10:stats:9}
\emph{1.} A nova \omterm{def:g10:stats:mean}{média} é
\[
\bar y = \frac{y_1 + \dots + y_N}{N}
= \frac{(a x_1 + b) + \dots + (a x_N + b)}{N}
= \frac{a(x_1 + \dots + x_N) + Nb}{N}
= a \bar x + b .
\]

\emph{2.} Com $a = 1.8$ e $b = 32$: \omterm{def:g10:stats:mean}{média}
$1.8 \times 20 + 32 = 68$ graus Fahrenheit.
\end{solution}

\begin{solution}{pb:g10:stats:1}
\textbf{1.} \omterm{def:g10:stats:mean}{Média}: $\frac{2\,550}{7} \approx 364$ mil euros;
\omterm{def:g10:stats:median}{mediana}: o quarto dos sete preços ordenados, $230$. A \omterm{def:g10:stats:median}{mediana}
descreve a rua --- seis das sete casas não custam nada perto de
$364$.

\textbf{2.} Sem a mansão: \omterm{def:g10:stats:mean}{média} $\frac{1\,350}{6} = 225$,
\omterm{def:g10:stats:median}{mediana} $\frac{220 + 230}{2} = 225$. A \omterm{def:g10:stats:mean}{média} caiu cerca de
$139$; a \omterm{def:g10:stats:median}{mediana}, $5$. Um único valor extremo pode arrastar a
\omterm{def:g10:stats:mean}{média} para qualquer lugar; a \omterm{def:g10:stats:median}{mediana} mal percebe --- ela é
\emph{robusta}.

\textbf{3.} \omterm{def:g10:stats:mean}{Média} $= \frac{0 \times 20 + 1 \times 25 + 2 \times
30 + 3 \times 15 + 4 \times 10}{100} = \frac{170}{100} = 1.7$.
Nenhuma família tem $1.7$ filhos, mas $1.7 \times 100 = 170$
\emph{é} o número total exato de filhos: a \omterm{def:g10:stats:mean}{média} codifica o
total, redistribuído igualmente --- seu verdadeiro talento.

\textbf{4.} $N = 15$: \omterm{def:g10:stats:median}{mediana} $=$ $8^{\text{o}}$ valor $= 11$;
$Q_1 =$ $4^{\text{o}}$ valor $= 8$; $Q_3 =$ $12^{\text{o}}$
valor $= 14$.

\textbf{5.} \omterm{def:g10:stats:quartiles}{Amplitude} $19 - 4 = 15$: a dispersão total,
extremos incluídos. \omterm{def:g10:stats:quartiles}{Amplitude interquartil} $14 - 8 = 6$: a
largura da metade central --- dispersão sem o ruído dos
extremos.

\textbf{6.} Um deslocamento $+b$ move \emph{todos} os valores,
de modo que as diferenças entre valores --- \omterm{def:g10:stats:quartiles}{amplitude} e
\omterm{def:g10:stats:quartiles}{amplitude interquartil} --- não mudam:
$(x_i + b) - (x_j + b) = x_i - x_j$. Multiplicar por $a$
multiplica todas as diferenças por $a$: \omterm{def:g10:stats:quartiles}{amplitude} e \omterm{def:g10:stats:quartiles}{amplitude
interquartil} ficam multiplicadas por $\abs a$.

\textbf{7.} Correção aditiva: $6 \to 9$ e $18 \to 21$ --- o
aluno fraco ganha $50\,\%$, o forte $17\,\%$, e o $21$ estoura
a escala de $20$ apenas um pouco. Multiplicativa: $6 \to 8$ e
$18 \to 24$ --- as razões se preservam, mas $24$ é impossível
numa escala de $20$: o problema do teto. Correções aditivas
favorecem a base; as multiplicativas explodem o topo.

\textbf{8.} Total das notas: $20 \times 12 + 30 \times 10 =
540$, de modo que a \omterm{def:g10:stats:mean}{média} conjunta é $\frac{540}{50} = 10.8$:
os trinta alunos da turma B pesam mais que os vinte da turma A
--- \omterm{def:g10:stats:mean}{médias} se juntam por \omterm{def:g10:stats:mean}{média} \emph{ponderada}, nunca por
\omterm{def:g10:stats:mean}{média} simples.

\textbf{9.} \omterm{def:g10:stats:median}{Mediana} de $A$: $2$; de $B$: $5$. Lista reunida
$\{0, 1, 2, 3, 10\}$: \omterm{def:g10:stats:median}{mediana} $2$. A \omterm{def:g10:stats:mean}{média} ponderada pelos
tamanhos das \omterm{def:g10:stats:median}{medianas} daria
$\frac{3 \times 2 + 2 \times 5}{5} = 3.2 \neq 2$: as \omterm{def:g10:stats:median}{medianas}
não carregam total algum, logo não existe fórmula de junção ---
é preciso reordenar a lista inteira.

\textbf{10.} \omterm{def:g10:stats:mean}{Média}: $\frac{31.3}{5} = 6.26$. \omterm{def:g10:stats:median}{Mediana}: $5.4$.
\omterm{def:g10:stats:mean}{Média} aparada: $\frac{5.3 + 5.4 + 5.5}{3} = 5.4$. Um juiz
entusiasmado (ou corrupto) deslocou a \omterm{def:g10:stats:mean}{média} em quase um ponto;
a \omterm{def:g10:stats:median}{mediana} e a \omterm{def:g10:stats:mean}{média} aparada nem se abalaram --- daí as regras
de aparo dos esportes com júri.

\textbf{11.} Fábrica A: metade central dentro de $0.2$ mm;
fábrica B: dentro de $4$ mm. Mesma \omterm{def:g10:stats:mean}{média}, confiabilidade
oposta: compre da A. Quem decidiu foi a \omterm{def:g10:stats:quartiles}{amplitude interquartil}
--- \omterm{def:g10:stats:mean}{médias} sozinhas comparam centros, nunca a consistência.

\textbf{12.} \omterm{def:g10:stats:median}{Mediana} muito abaixo da \omterm{def:g10:stats:mean}{média} denuncia uma
\emph{cauda longa à direita}: a maioria das esperas é curta, e
algumas são catastróficas. Exemplo: $\{5, 5, 10, 10, 95\}$ ---
\omterm{def:g10:stats:median}{mediana} $10$, \omterm{def:g10:stats:mean}{média} $\frac{125}{5} = 25$.

\textbf{13.} Significa que $75\,\%$ dos bebês daquela idade são
mais baixos (e $25\,\%$ mais altos): é uma posição, não um
tamanho. ``Toda criança acima da \omterm{def:g10:stats:mean}{média}'' é impossível --- se
todos os valores superassem a \omterm{def:g10:stats:mean}{média}, a \omterm{def:g10:stats:mean}{média} deles superaria a
própria \omterm{def:g10:stats:mean}{média}: contradição. Mas \emph{quase} todas podem estar:
uma criança muito baixa segura a \omterm{def:g10:stats:mean}{média} abaixo de todas as
outras. Acima da \emph{\omterm{def:g10:stats:median}{mediana}}: nunca mais que a metade, por
definição. Posições não se lisonjeiam; \omterm{def:g10:stats:mean}{médias} sim.

\textbf{14.} Nota típica: a mesma, \omterm{def:g10:stats:median}{mediana} $11$ nas duas.
Homogeneidade do meio: B é mais compacta (\omterm{def:g10:stats:quartiles}{amplitude
interquartil} $3$ contra $5$). Extremos: B é mais selvagem
(\omterm{def:g10:stats:quartiles}{amplitude} $17$ contra $13$). B é uma turma de alunos
parecidos com alguns valores atípicos; A é uniformemente
espalhada.

\textbf{15.} (1) Esses $5\,000$ são a \omterm{def:g10:stats:mean}{média} ou a \omterm{def:g10:stats:median}{mediana} --- e
posso ver a \omterm{def:g10:stats:median}{mediana}? (2) Qual é a dispersão (quartis), para eu
saber o quanto o ``típico'' é típico? (3) Quem foi contado ---
tamanho e seleção da amostra (a lição das pesquisas do problema
de fim de semana sobre como os dados mentem, no volume do
ensino fundamental)?

\textbf{16.} Por exemplo, $100, 150, 200, 250, 800$: \omterm{def:g10:stats:median}{mediana}
$200$, \omterm{def:g10:stats:mean}{média} $\frac{1\,500}{5} = 300$.

\textbf{17.} $\sum (x_i - \bar x) = \sum x_i - N \bar x =
N\bar x - N\bar x = 0$. Verificação:
$-200 - 150 - 100 - 50 + 500 = 0$. A \omterm{def:g10:stats:mean}{média} é o ponto onde os
desvios se equilibram --- o centro de gravidade dos dados (o
$G$ do problema de fim de semana sobre o baricentro, no volume
do ensino fundamental, em uma dimensão).

\textbf{18.} Na \omterm{def:g10:stats:median}{mediana} $4$: $3 + 1 + 0 + 5 + 9 = 18$ km. Na
\omterm{def:g10:stats:mean}{média} $6$: $5 + 3 + 2 + 3 + 7 = 20$ km. Em $5$:
$4 + 2 + 1 + 4 + 8 = 19$ km. A \omterm{def:g10:stats:median}{mediana} vence --- afastar-se
dela sempre deixa para trás mais lojas do que aquelas de que se
aproxima, de modo que o total só pode crescer.

\textbf{19.} $0.3^{10} \approx 6 \times 10^{-6}$: cerca de seis
pilotos \emph{por milhão}. Em $4\,000$ pilotos o número
esperado é $0.02$ --- zero, como o tenente Daniels constatou. A
Força Aérea parou de projetar para o homem médio e tornou tudo
ajustável: assentos, pedais, cintos --- projetar para a
\emph{dispersão}, não para o centro.

\textbf{20.} Totais: a \emph{\omterm{def:g10:stats:mean}{média}}. Indivíduo típico: a
\emph{\omterm{def:g10:stats:median}{mediana}}. Justiça da dispersão: os \emph{quartis} (e a
\omterm{def:g10:stats:quartiles}{amplitude interquartil}). Contaminação: a \emph{\omterm{def:g10:stats:mean}{média} aparada}.
E a frase de encerramento: a pessoa \omterm{def:g10:stats:mean}{média} não existe --- mas a
pessoa \omterm{def:g10:stats:median}{mediana} quase existe.
\end{solution}
