\section*{Capítulo \ref{ch:g10:vectors} --- Vetores}

\begin{solution}{exo:g10:vectors:1}
$\vect{AB} + \vect{BD} = \vect{AD}$ (Chasles).

$\vect{KL} + \vect{LM} + \vect{MK} = \vect{KK} = \vec 0$.

$\vect{AC} - \vect{BC} = \vect{AC} + \vect{CB} = \vect{AB}$.

$\vect{AB} + \vect{CA} = \vect{CA} + \vect{AB} = \vect{CB}$.
\end{solution}

\begin{solution}{exo:g10:vectors:2}
$\vect{AB}\,(5-2,\ 3-1) = (3, 2)$; \quad
$\vect{BC}\,(-1-5,\ 4-3) = (-6, 1)$; \quad
$\vect{AC}\,(-1-2,\ 4-1) = (-3, 3)$.
Verificação: $(3, 2) + (-6, 1) = (-3, 3)$, coordenada a coordenada.
\end{solution}

\begin{solution}{exo:g10:vectors:3}
$\vec u + \vec v = (2 + (-1),\ -3 + 4) = (1, 1)$.

$3\vec u = (6, -9)$.

$2\vec u - \vec v = (4 - (-1),\ -6 - 4) = (5, -10)$.

$\norm{\vec u} = \sqrt{2^2 + (-3)^2} = \sqrt{13}$.
\end{solution}

\begin{solution}{exo:g10:vectors:4}
(a) $4 \times 9 - 6 \times 6 = 36 - 36 = 0$: \omterm{def:g10:vectors:collinear}{colineares}
($\vec v = \frac32 \vec u$).

(b) $2 \times 10 - (-5) \times (-4) = 20 - 20 = 0$: \omterm{def:g10:vectors:collinear}{colineares}
($\vec v = -2\vec u$).

(c) $3 \times 3 - 1 \times 1 = 8 \neq 0$: não são \omterm{def:g10:vectors:collinear}{colineares}.
\end{solution}

\begin{solution}{exo:g10:vectors:5}
$\vect{AB}\,(4, -2)$ e $\vect{DC}\,(6 - x_D,\ 3 - y_D)$. A condição
$\vect{AB} = \vect{DC}$ dá $6 - x_D = 4$ e $3 - y_D = -2$, logo
$D(2, 5)$.
\end{solution}

\begin{solution}{exo:g10:vectors:6}
\emph{1.} $\vect{AB}\,(3, 2)$ e $\vect{AC}\,(6, -2)$, logo
$\vect{AB} + \vect{AC} = (9, 0)$ e
$M = (-2 + 9,\ 1 + 0) = (7, 1)$.

\emph{2.} Pela regra do paralelogramo, $\vect{AM} = \vect{AB} +
\vect{AC}$ significa que $ABMC$ é um paralelogramo ($M$ é o quarto
vértice, oposto a $A$). Dá para conferir: $\vect{AB}\,(3,2)$ e
$\vect{CM}\,(7-4,\ 1-(-1)) = (3, 2)$ são iguais.
\end{solution}

\begin{solution}{exo:g10:vectors:7}
Primeira terna: $\vect{AB}\,(3, 2)$, $\vect{AC}\,(9, 6)$;
$3 \times 6 - 2 \times 9 = 0$: \omterm{def:g10:vectors:collinear}{colineares} ($\vect{AC} = 3\vect{AB}$).

Segunda terna: $\vect{AB}\,(2, -1)$, $\vect{AC}\,(8, -4)$;
$2 \times (-4) - (-1) \times 8 = -8 + 8 = 0$: também \omterm{def:g10:vectors:collinear}{colineares}
($\vect{AC} = 4\vect{AB}$).
\end{solution}

\begin{solution}{exo:g10:vectors:8}
$\vect{AB}\,(4, 2)$ e $\vect{DC}\,(6-2,\ 1-(-1)) = (4, 2)$: os dois
\omterm{def:g10:vectors:vector}{vetores} são \emph{iguais}, logo $ABCD$ é um paralelogramo e, em
particular, $(AB) \parallel (CD)$. A propriedade do paralelogramo é a
mais forte: ela implica o paralelismo (colinearidade dos \omterm{def:g10:vectors:vector}{vetores}), e não
o contrário.
\end{solution}

\begin{solution}{exo:g10:vectors:9}
\emph{1.} $\vect{AB}\,(6, 3)$, logo
$\vect{AI} = \frac23\vect{AB} = (4, 2)$ e $I(4, 2)$.

\emph{2.} $\vect{CB}\,(6-2,\ 3-5) = (4, -2)$, logo
$\vect{CJ} = \frac23\vect{CB} = \left(\frac83, -\frac43\right)$ e
$J\left(2 + \frac83,\ 5 - \frac43\right) = \left(\frac{14}{3},
\frac{11}{3}\right)$.

\emph{3.} $\vect{IJ}\,\left(\frac{14}{3} - 4,\ \frac{11}{3} - 2\right) =
\left(\frac23, \frac53\right)$ e $\vect{AC}\,(2, 5)$. Critério de
colinearidade:
\[
\frac23 \times 5 - \frac53 \times 2 = \frac{10}{3} - \frac{10}{3} = 0 :
\]
os \omterm{def:g10:vectors:vector}{vetores} são \omterm{def:g10:vectors:collinear}{colineares}, logo $(IJ)$ é paralela a $(AC)$. (Tanto $I$
quanto $J$ ficam a um terço do caminho a partir do lado rumo a $B$: o
pequeno triângulo $IBJ$ é uma redução de $ABC$.)
\end{solution}

\begin{solution}{exo:g10:vectors:10}
Os \omterm{prop:g10:coordgeom:midpoint}{pontos médios} são
\[
I\left(\tfrac{x_A + x_B}{2}, \tfrac{y_A + y_B}{2}\right), \quad
J\left(\tfrac{x_B + x_C}{2}, \tfrac{y_B + y_C}{2}\right), \quad
K\left(\tfrac{x_C + x_D}{2}, \tfrac{y_C + y_D}{2}\right), \quad
L\left(\tfrac{x_D + x_A}{2}, \tfrac{y_D + y_A}{2}\right).
\]
Primeiras \omterm{def:g10:coordgeom:system}{coordenadas} dos dois \omterm{def:g10:vectors:vector}{vetores}:
\[
x_J - x_I = \frac{x_B + x_C}{2} - \frac{x_A + x_B}{2}
= \frac{x_C - x_A}{2},
\qquad
x_K - x_L = \frac{x_C + x_D}{2} - \frac{x_D + x_A}{2}
= \frac{x_C - x_A}{2},
\]
e o mesmo cálculo vale para as segundas \omterm{def:g10:coordgeom:system}{coordenadas}. Portanto
$\vect{IJ} = \vect{LK}$ (ambos iguais a $\frac12\vect{AC}$), e $IJKL$ é
um paralelogramo, qualquer que seja o aspecto do quadrilátero $ABCD$.
\end{solution}

\begin{solution}{pb:g10:vectors:1}
\textbf{1.} $\vect{AB} + \vect{BC} + \vect{CD} = \vect{AD}$;
\quad $\vect{MA} - \vect{MB} = \vect{MA} + \vect{BM} =
\vect{BA}$; \quad $\vect{AB} + \vect{BC} + \vect{CD} +
\vect{DA} = \vect{AA} = \vec 0$ (reordenando a terceira \omterm{def:g10:vectors:sum}{soma}).

\textbf{2.} $\vect{OB} - \vect{OA} = \vect{AO} + \vect{OB} =
\vect{AB}$ por Chasles.

\textbf{3.} $I$ é o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$ exatamente quando
$\vect{IA} = -\vect{IB}$, isto é,
$\vect{IA} + \vect{IB} = \vec 0$. Então, para todo $O$:
$\vec 0 = \vect{IO} + \vect{OA} + \vect{IO} + \vect{OB}$, logo
$2\,\vect{OI} = \vect{OA} + \vect{OB}$ --- e reciprocamente.

\textbf{4.} $\vect{AB} = (4, 2)$, e $D$ satisfaz
$\vect{DC} = \vect{AB}$: $D = C - (4, 2) = (2, -2)$ --- o mesmo
$D$ do \cref{pb:g10:coordgeom:1}, uma linha mais rápido.

\textbf{5.} A translação de \omterm{def:g10:vectors:vector}{vetor} $\vec u$ --- exatamente a
transformação produzida por dois espelhos paralelos ou por duas
meias voltas. \omterm{def:g10:functions:function}{Imagem} de $(1, 2)$: $(4, 1)$.

\textbf{6.} Desdobrando cada $\vect{GX}$ como
$\vect{GO} + \vect{OX}$:
$3\,\vect{GO} + \vect{OA} + \vect{OB} + \vect{OC} = \vec 0$,
logo $\vect{OG} = \frac13(\vect{OA} + \vect{OB} + \vect{OC})$:
existe um e um só ponto assim, cujas \omterm{def:g10:coordgeom:system}{coordenadas} são as médias
das dos três vértices.

\textbf{7.} $\vect{AB} + \vect{AC} =
(\vect{AG} + \vect{GB}) + (\vect{AG} + \vect{GC})
= 2\vect{AG} + (\vect{GB} + \vect{GC})
= 2\vect{AG} - \vect{GA} = 3\vect{AG}$, donde
$\vect{AG} = \frac13(\vect{AB} + \vect{AC})$. A fórmula do
\omterm{prop:g10:coordgeom:midpoint}{ponto médio} (questão 3, vista de $A$) dá
$\vect{AK} = \frac12(\vect{AB} + \vect{AC})$. Comparando:
$\vect{AG} = \frac23\vect{AK}$ --- $G$ está sobre a mediana
$[AK]$, a dois terços do caminho a partir de $A$.

\textbf{8.} A condição que define $G$ trata $A$, $B$, $C$ de
modo idêntico, de modo que o mesmo cálculo feito a partir de
$B$ ou de $C$ coloca o \emph{mesmo} $G$ a dois terços dessas
medianas: a concorrência sai de graça. A demonstração pelo
paralelogramo era engenhosa, mas feita sob medida; as
\omterm{def:g10:coordgeom:system}{coordenadas} calculam, mas obscurecem; os \omterm{def:g10:vectors:vector}{vetores} tornam a
simetria visível e a demonstração cabe em duas linhas.

\textbf{9.} $G\left(\frac{1+5+3}{3}, \frac{1+2+6}{3}\right) =
(3, 3)$; $K(4, 4)$; $\vect{AG} = (2, 2)$ e
$\vect{AK} = (3, 3)$: determinante
$2 \times 3 - 2 \times 3 = 0$, \omterm{def:g10:vectors:collinear}{colineares}, e de fato
$\vect{AG} = \frac23\vect{AK}$.

\textbf{10.} $G'\left(\frac{2 + 5 + 3}{4}, \frac{2 + 2 +
6}{4}\right) = (2.5,\ 2.5)$: puxado de $G(3,3)$ na direção do
vértice de massa dobrada $A(1,1)$, como convém a um ponto de
equilíbrio.

\textbf{11.} $3 \times 4 - (-2)(-6) = 12 - 12 = 0$: \omterm{def:g10:vectors:collinear}{colineares}
($\vec v = -2\vec u$). $2 \times 9 - 5 \times 4 = -2 \neq 0$:
não são \omterm{def:g10:vectors:collinear}{colineares}.

\textbf{12.} $\vect{AB} = (2, 3)$, $\vect{AC} = (6, 9)$:
determinante $2 \times 9 - 3 \times 6 = 0$: \omterm{def:g10:vectors:collinear}{colineares}.

\textbf{13.} $\vect{OI} = \frac12(\vect{OA} + \vect{OB})$ e
$\vect{OK} = \frac12(\vect{OC} + \vect{OD})$, de modo que o
\omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[IK]$ tem \omterm{def:g10:vectors:vector}{vetor} posição
$\frac12(\vect{OI} + \vect{OK}) = \frac14(\vect{OA} +
\vect{OB} + \vect{OC} + \vect{OD})$ --- e a conta para $[JL]$
dá exatamente o mesmo \omterm{def:g10:vectors:vector}{vetor}. As duas bimedianas (e, por
Varignon, as diagonais do paralelogramo $IJKL$) partilham esse
centro: o baricentro do quadrilátero.

\textbf{14.} $\vect{MN} = \vect{AN} - \vect{AM} =
\frac13\vect{AC} - \frac13\vect{AB} = \frac13\vect{BC}$:
logo $(MN) \parallel (BC)$ e $MN = \frac13 BC$.

\textbf{15.} Com $\vect{AM} = k\,\vect{AB}$ e
$\vect{AN} = k\,\vect{AC}$: $\vect{MN} = k\,\vect{BC}$ --- uma
única linha que contém o teorema da base média
($k = \frac12$) e o de Tales (qualquer $k$): a razão atravessa
a subtração intacta.

\textbf{16.} Resultante $(3, 4)$, módulo
$\sqrt{9 + 16} = 5$ km/h: o barco anda de fato para nordeste,
puxando para o norte, ao longo da hipotenusa do triângulo
$3$--$4$--$5$.

\textbf{17.} $\vec u + \vec v + \vec w = (0, 0)$: o anel fica
parado --- equilíbrio. Ao longo de um polígono fechado
$A_1 A_2 \dots A_n A_1$, Chasles telescopa a \omterm{def:g10:vectors:sum}{soma} até
$\vect{A_1 A_1} = \vec 0$: dar a volta traz você de volta.

\textbf{18.} $\vec w = -(\vec u + \vec v) = -(3, 2) =
(-3, -2)$.

\textbf{19.} Resultante $(1.5,\ 2)$, módulo
$\sqrt{2.25 + 4} = 2.5$ m/s. Travessia: $50$ m a $2$ m/s na
direção da outra margem: $25$ s; deriva:
$1.5 \times 25 = 37.5$ m rio abaixo.

\textbf{20.} Redemonstrados neste fim de semana: o teorema dos
dois terços do baricentro (questão 7), o centro das
bimedianas e de Varignon (questão 13) e Tales com a base média
(questões 14--15). O que as setas ainda não conseguem fazer:
medir \emph{ângulos} (e, com eles, comprimentos em direções
arbitrárias) --- o produto escalar do ano 11 é exatamente o
instrumento que falta.
\end{solution}
