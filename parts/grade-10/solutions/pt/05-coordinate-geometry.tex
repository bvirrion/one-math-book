\section*{Capítulo \ref{ch:g10:coordgeom} --- Geometria analítica}

\begin{solution}{exo:g10:coordgeom:1}
\omterm{prop:g10:coordgeom:midpoint}{Ponto médio}: $\left(\frac{2 + (-4)}{2}, \frac{5 + 1}{2}\right) = (-1,
3)$. Distância:
\[
AB = \sqrt{(-4 - 2)^2 + (1 - 5)^2} = \sqrt{36 + 16} = \sqrt{52}
= 2\sqrt{13}.
\]
\end{solution}

\begin{solution}{exo:g10:coordgeom:2}
Dizer que $I$ é o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$ significa $\frac{3 + x_B}{2} = 1$
e $\frac{-2 + y_B}{2} = 2$, logo $x_B = -1$ e $y_B = 6$: $B(-1, 6)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:3}
$OP = \sqrt{16 + 1} = \sqrt{17}$; $OQ = \sqrt{9 + 4} = \sqrt{13}$;
$OR = \sqrt{0 + 25} = 5 = \sqrt{25}$. Como $13 < 17 < 25$ e a raiz
quadrada é \omterm{def:g10:functions:variations}{crescente}, $Q$ é o mais próximo da origem.
\end{solution}

\begin{solution}{exo:g10:coordgeom:4}
Quadrados dos comprimentos:
\begin{align*}
AB^2 &= (5-1)^2 + (4-2)^2 = 16 + 4 = 20, \\
BC^2 &= (7-5)^2 + (0-4)^2 = 4 + 16 = 20, \\
AC^2 &= (7-1)^2 + (0-2)^2 = 36 + 4 = 40 .
\end{align*}
$AB = BC$: o triângulo é isósceles em $B$. Além disso,
$AB^2 + BC^2 = 40 = AC^2$: pela recíproca do teorema de Pitágoras, ele
também é retângulo em $B$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:5}
\emph{1.} \omterm{prop:g10:coordgeom:midpoint}{Ponto médio} de $[AC]$:
$\left(\frac{-2+4}{2}, \frac{1+1}{2}\right) = (1, 1)$; \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de
$[BD]$: $\left(\frac{1+1}{2}, \frac{3+(-1)}{2}\right) = (1, 1)$. Mesmo
\omterm{prop:g10:coordgeom:midpoint}{ponto médio}: $ABCD$ é um paralelogramo.

\emph{2.} $AB^2 = 3^2 + 2^2 = 13$ e $BC^2 = 3^2 + (-4)^2 = 25$:
$AB \neq BC$, não é losango.

\emph{3.} $AC^2 = 6^2 + 0^2 = 36$ e $BD^2 = 0^2 + (-4)^2 = 16$:
$AC \neq BD$, não é retângulo. $ABCD$ é um paralelogramo comum.
\end{solution}

\begin{solution}{exo:g10:coordgeom:6}
Calcule os quadrados das distâncias a $\Omega(2,1)$ e compare com
$5^2 = 25$:
\[
\Omega A^2 = 3^2 + 4^2 = 25, \qquad
\Omega B^2 = (-3)^2 + (-4)^2 = 25, \qquad
\Omega C^2 = 4^2 + 1^2 = 17 .
\]
$A$ e $B$ estão sobre o círculo; $C$ está dentro dele ($17 < 25$).
\end{solution}

\begin{solution}{exo:g10:coordgeom:7}
$MA^2 = (x - 1)^2 + 1$ e $MB^2 = (x - 7)^2 + 25$. Impondo
$MA^2 = MB^2$:
\[
x^2 - 2x + 2 = x^2 - 14x + 74
\ \Longleftrightarrow\
12x = 72
\ \Longleftrightarrow\
x = 6 .
\]
O único ponto assim é $M(6, 0)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:8}
\emph{1.} $I = \left(\frac{0+4}{2}, \frac{3+1}{2}\right) = (2, 2)$.

\emph{2.} $M(2,2)$ é de fato $I$. E
$MA^2 = (0-2)^2 + (3-2)^2 = 5$, $MB^2 = (4-2)^2 + (1-2)^2 = 5$: o \omterm{prop:g10:coordgeom:midpoint}{ponto
médio} é equidistante das duas extremidades, como se esperava.

\emph{3.} Procure $N(0, y)$ com $NA^2 = NB^2$:
\[
(3 - y)^2 = 16 + (1 - y)^2
\ \Longleftrightarrow\
9 - 6y = 16 + 1 - 2y
\ \Longleftrightarrow\
-4y = 8,
\]
logo $y = -2$: o ponto $N(0, -2)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:9}
\begin{align*}
AB &= \sqrt{4^2 + 2^2} = \sqrt{20} = 2\sqrt5, \\
BC &= \sqrt{8^2 + 4^2} = \sqrt{80} = 4\sqrt5, \\
AC &= \sqrt{12^2 + 6^2} = \sqrt{180} = 6\sqrt5 .
\end{align*}
Então $AB + BC = 2\sqrt5 + 4\sqrt5 = 6\sqrt5 = AC$: a desigualdade
triangular vira igualdade, de modo que $B$ está sobre o segmento $[AC]$
--- os três pontos são colineares.
\end{solution}

\begin{solution}{exo:g10:coordgeom:10}
O \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$ é $I\left(\frac a2, \frac b2\right)$. Então
\[
OI^2 = \frac{a^2}{4} + \frac{b^2}{4},
\qquad
IA^2 = \left(a - \frac a2\right)^2 + \left(0 - \frac b2\right)^2
= \frac{a^2}{4} + \frac{b^2}{4},
\]
e $IB^2 = \left(0 - \frac a2\right)^2 + \left(b - \frac b2\right)^2 =
\frac{a^2}{4} + \frac{b^2}{4}$ também. Os três quadrados de distância
são iguais, logo $OI = IA = IB$: o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} da hipotenusa do
triângulo retângulo $OAB$ é equidistante dos três vértices (é o centro
de seu círculo circunscrito).
\end{solution}

\begin{solution}{pb:g10:coordgeom:1}
\textbf{1.} $\Omega M = 5 \iff \Omega M^2 = 25 \iff
(x - 2)^2 + (y - 1)^2 = 25$ pela fórmula da distância
(\cref{thm:g10:coordgeom:distance}) --- elevar ao quadrado é
inofensivo, já que os dois lados são positivos.

\textbf{2.} $A$: $(5-2)^2 + (5-1)^2 = 9 + 16 = 25$: sobre o
círculo. $B$: $16 + 9 = 25$: também sobre ele. $D$:
$4 + 9 = 13 < 25$: dentro.

\textbf{3.} $x^2 - 6x = (x - 3)^2 - 9$ e
$y^2 + 4y = (y + 2)^2 - 4$, de modo que a \omterm{def:g10:algebra:equation}{equação} fica
$(x - 3)^2 + (y + 2)^2 = 25$: centro $(3, -2)$, raio $5$.

\textbf{4.} $(x - 3)^2 + (y + 2)^2 = -1$: uma soma de quadrados
nunca é negativa --- nenhum ponto a satisfaz: o conjunto vazio.
Em geral, $(x - h)^2 + (y - k)^2 = c$ é um círculo de raio
$\sqrt c$ se $c > 0$, o único ponto $(h, k)$ se $c = 0$, e é
vazio se $c < 0$.

\textbf{5.} Com $y = 3$: $(x - 3)^2 + 25 = 25$, logo
$(x - 3)^2 = 0$: o único ponto $(3, 3)$. Um só ponto de
contato: a reta é \emph{tangente} ao círculo.

\textbf{6.} $MA^2 + MB^2 = (x + r)^2 + y^2 + (x - r)^2 + y^2
= 2x^2 + 2y^2 + 2r^2 = 2r^2 + 2r^2 = 4r^2$ (usando
$x^2 + y^2 = r^2$). Como $AB^2 = (2r)^2 = 4r^2$, o triângulo
$AMB$ satisfaz $MA^2 + MB^2 = AB^2$: é retângulo em $M$
(recíproca de Pitágoras) --- três linhas, como prometido.

\textbf{7.} $MA^2 + MB^2 = (x + a)^2 + y^2 + (x - a)^2 + y^2
= 2x^2 + 2y^2 + 2a^2 = 2\,MI^2 + 2a^2$, e
$\frac{AB^2}{2} = \frac{(2a)^2}{2} = 2a^2$. Identidade
demonstrada.

\textbf{8.} $2\,MI^2 + \frac{36}{2} = 26$ dá $MI^2 = 4$: o
círculo de centro no \omterm{prop:g10:coordgeom:midpoint}{ponto médio} $I$ e raio $2$.

\textbf{9.} $MA^2 = 4\,MB^2$:
$x^2 + y^2 = 4\left((x - 3)^2 + y^2\right)$, isto é,
$3x^2 + 3y^2 - 24x + 36 = 0$, ou
$x^2 + y^2 - 8x + 12 = 0$: completando os quadrados,
$(x - 4)^2 + y^2 = 4$ --- o círculo de centro $(4, 0)$ e raio
$2$ (o \emph{círculo de Apolônio} de razão $2$).

\textbf{10.} A ponte converte construções em contas: nenhum
retângulo auxiliar, nenhuma análise de casos --- uma expansão e
o teorema cai. O preço: as contas carregam menos intuição
geométrica; a demonstração clássica mostra \emph{por que}, e a
algébrica mostra \emph{que} --- um matemático em atividade
guarda as duas.

\textbf{11.} $x^2 + y^2 = 25$ e $(x - 6)^2 + y^2 = 25$.
Subtraindo: $x^2 - (x - 6)^2 = 0$, isto é, $12x - 36 = 0$:
$x = 3$. Então $9 + y^2 = 25$: $y = \pm 4$. Candidatos
$(3, 4)$ e $(3, -4)$.

\textbf{12.} De $(3, 4)$ até $R(0, 8)$: $\sqrt{9 + 16} = 5$ ---
confere. De $(3, -4)$: $\sqrt{9 + 144} \approx 12.4$ ---
errado. Você está em $(3, 4)$.

\textbf{13.} As duas \omterm{def:g10:algebra:equation}{equações} trazem o mesmo $x^2 + y^2$ (ou
com coeficientes iguais depois da expansão): a subtração
cancela os quadrados e deixa uma \omterm{def:g10:algebra:equation}{equação} de grau um --- uma
reta. Quando os círculos se encontram duas vezes, essa reta
passa pelos dois pontos de encontro: é a reta da corda comum.

\textbf{14.} Quatro incógnitas: três \omterm{def:g10:coordgeom:system}{coordenadas} mais o erro do
relógio do receptor (um relógio barato, corrigido pela
matemática). Cada satélite fornece uma \omterm{def:g10:algebra:equation}{equação}, de modo que
quatro satélites dão quatro \omterm{def:g10:algebra:equation}{equações} para quatro incógnitas ---
e mais satélites melhoram a precisão.

\textbf{15.} A subtração dá agora
$x = \frac{5.1^2 - 5^2 + 36}{12} = \frac{37.01}{12} \approx
3.084$: um erro de $10$ cm em uma distância deslocou a
estimativa cerca de $8$ cm aqui. Os erros se propagam por
todas as fórmulas --- a cautela da aritmética de \omterm{def:g10:numbers:interval}{intervalos} do
\cref{pb:g10:numbers:1}, agora com \omterm{def:g10:coordgeom:system}{coordenadas}.

\textbf{16.} $B'(7, -3)$. A reta $(AB')$ tem \omterm{def:g10:reffunc:affine}{coeficiente
angular} $\frac{-3 - 5}{7 - 1} = -\frac43$ e cruza $y = 0$ em
$x = 1 + \frac{5}{8} \times 6 = 4.75$: beba em
$P(4.75,\ 0)$. Comprimento \omterm{def:g10:functions:extrema}{mínimo}: $AP + PB = AP + PB' = AB' =
\sqrt{6^2 + 8^2} = 10$.

\textbf{17.} Passando por $Q(3, 0)$:
$AQ + QB = \sqrt{4 + 25} + \sqrt{16 + 9} = \sqrt{29} + 5
\approx 10.39 > 10$. A reta obtida por reflexão vence.

\textbf{18.} \omterm{prop:g10:coordgeom:midpoint}{Pontos médios} das diagonais: $[AC]$: $(3, 3)$;
$[BD]$: $(3, 3)$ --- iguais: paralelogramo. Lados:
$AB = \sqrt{10} = AD$: losango. Diagonais:
$AC = \sqrt{32} \neq BD = \sqrt{8}$: não é retângulo e,
portanto, não é quadrado. Veredicto: um losango, e nada mais.

\textbf{19.} $D = A + C - B = (-1 + 6 - 3,\ 2 + 0 - 4) =
(2, -2)$: aí $[AC]$ e $[BD]$ passam a ter o mesmo \omterm{prop:g10:coordgeom:midpoint}{ponto médio}
$\left(\frac52, 1\right)$.

\textbf{20.} Fórmula do \omterm{prop:g10:coordgeom:midpoint}{ponto médio}: paralelogramos e centros
(questões 18 e 19). Fórmula da distância: círculos, lugares
geométricos, natureza de triângulos, caminhos \omterm{def:g10:functions:extrema}{mínimos} (questões
1--9 e 16). Subtração de \omterm{def:g10:algebra:equation}{equações} de círculo: a reta que
localiza uma posição (questões 11--15). O que falta na caixa:
uma álgebra limpa das \emph{direções} --- os vetores --- e das
\emph{inclinações} --- \omterm{def:g10:algebra:equation}{equações} de retas e sistemas: os dois
próximos capítulos fornecem exatamente isso.
\end{solution}
