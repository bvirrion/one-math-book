\section*{الفصل \ref{ch:g10:vectors} --- المتجهات}

\begin{solution}{exo:g10:vectors:1}
$\vect{AB} + \vect{BD} = \vect{AD}$ (شال).

$\vect{KL} + \vect{LM} + \vect{MK} = \vect{KK} = \vec 0$.

$\vect{AC} - \vect{BC} = \vect{AC} + \vect{CB} = \vect{AB}$.

$\vect{AB} + \vect{CA} = \vect{CA} + \vect{AB} = \vect{CB}$.
\end{solution}

\begin{solution}{exo:g10:vectors:2}
$\vect{AB}\,(5-2,\ 3-1) = (3, 2)$؛ \quad
$\vect{BC}\,(-1-5,\ 4-3) = (-6, 1)$؛ \quad
$\vect{AC}\,(-1-2,\ 4-1) = (-3, 3)$.
تحقق: $(3, 2) + (-6, 1) = (-3, 3)$، إحداثيًا إحداثيًا.
\end{solution}

\begin{solution}{exo:g10:vectors:3}
$\vec u + \vec v = (2 + (-1),\ -3 + 4) = (1, 1)$.

$3\vec u = (6, -9)$.

$2\vec u - \vec v = (4 - (-1),\ -6 - 4) = (5, -10)$.

$\norm{\vec u} = \sqrt{2^2 + (-3)^2} = \sqrt{13}$.
\end{solution}

\begin{solution}{exo:g10:vectors:4}
(أ) $4 \times 9 - 6 \times 6 = 36 - 36 = 0$: مستقيميتان
($\vec v = \frac32 \vec u$).

(ب) $2 \times 10 - (-5) \times (-4) = 20 - 20 = 0$: مستقيميتان
($\vec v = -2\vec u$).

(ج) $3 \times 3 - 1 \times 1 = 8 \neq 0$: ليستا \omterm{def:g10:vectors:collinear}{مستقيميتين}.
\end{solution}

\begin{solution}{exo:g10:vectors:5}
$\vect{AB}\,(4, -2)$، و $\vect{DC}\,(6 - x_D,\ 3 - y_D)$. ويعطي الشرط
$\vect{AB} = \vect{DC}$ أن $6 - x_D = 4$ و $3 - y_D = -2$، إذن
$D(2, 5)$.
\end{solution}

\begin{solution}{exo:g10:vectors:6}
\emph{1.} $\vect{AB}\,(3, 2)$ و $\vect{AC}\,(6, -2)$، إذن
$\vect{AB} + \vect{AC} = (9, 0)$ و
$M = (-2 + 9,\ 1 + 0) = (7, 1)$.

\emph{2.} حسب قاعدة متوازي الأضلاع، تعني $\vect{AM} = \vect{AB} + \vect{AC}$
أن $ABMC$ متوازي أضلاع (فالنقطة $M$ هي الرأس الرابع المقابل
للرأس $A$). ويمكن التحقق: $\vect{AB}\,(3,2)$ و $\vect{CM}\,(7-4,\ 1-(-1)) =
(3, 2)$ متساويتان.
\end{solution}

\begin{solution}{exo:g10:vectors:7}
الثلاثي الأول: $\vect{AB}\,(3, 2)$ و $\vect{AC}\,(9, 6)$؛
$3 \times 6 - 2 \times 9 = 0$: على استقامة واحدة ($\vect{AC} = 3\vect{AB}$).

الثلاثي الثاني: $\vect{AB}\,(2, -1)$ و $\vect{AC}\,(8, -4)$؛
$2 \times (-4) - (-1) \times 8 = -8 + 8 = 0$: على استقامة واحدة أيضًا
($\vect{AC} = 4\vect{AB}$).
\end{solution}

\begin{solution}{exo:g10:vectors:8}
$\vect{AB}\,(4, 2)$ و $\vect{DC}\,(6-2,\ 1-(-1)) = (4, 2)$: فالمتجهتان
\emph{متساويتان}، إذن $ABCD$ متوازي أضلاع، وبوجه خاص
$(AB) \parallel (CD)$. وخاصية متوازي الأضلاع هي الأقوى: فهي
تستلزم التوازي (استقامية المتجهتين)، لا العكس.
\end{solution}

\begin{solution}{exo:g10:vectors:9}
\emph{1.} $\vect{AB}\,(6, 3)$، إذن
$\vect{AI} = \frac23\vect{AB} = (4, 2)$ و $I(4, 2)$.

\emph{2.} $\vect{CB}\,(6-2,\ 3-5) = (4, -2)$، إذن
$\vect{CJ} = \frac23\vect{CB} = \left(\frac83, -\frac43\right)$ و
$J\left(2 + \frac83,\ 5 - \frac43\right) = \left(\frac{14}{3},
\frac{11}{3}\right)$.

\emph{3.} $\vect{IJ}\,\left(\frac{14}{3} - 4,\ \frac{11}{3} - 2\right) =
\left(\frac23, \frac53\right)$ و $\vect{AC}\,(2, 5)$. ومحك
الاستقامية:
\[
\frac23 \times 5 - \frac53 \times 2 = \frac{10}{3} - \frac{10}{3} = 0 :
\]
\omterm{def:g10:vectors:collinear}{فالمتجهتان مستقيميتان}، إذن $(IJ)$ يوازي $(AC)$. (فكلٌّ من $I$ و
$J$ يقع على ثلث الطريق من الضلع نحو $B$: أي أن المثلث
الصغير $IBJ$ تصغير للمثلث $ABC$.)
\end{solution}

\begin{solution}{exo:g10:vectors:10}
المنتصفات هي
\[
I\left(\tfrac{x_A + x_B}{2}, \tfrac{y_A + y_B}{2}\right), \quad
J\left(\tfrac{x_B + x_C}{2}, \tfrac{y_B + y_C}{2}\right), \quad
K\left(\tfrac{x_C + x_D}{2}, \tfrac{y_C + y_D}{2}\right), \quad
L\left(\tfrac{x_D + x_A}{2}, \tfrac{y_D + y_A}{2}\right).
\]
والإحداثي الأول للمتجهتين:
\[
x_J - x_I = \frac{x_B + x_C}{2} - \frac{x_A + x_B}{2}
= \frac{x_C - x_A}{2},
\qquad
x_K - x_L = \frac{x_C + x_D}{2} - \frac{x_D + x_A}{2}
= \frac{x_C - x_A}{2},
\]
والحساب نفسه ينجح من أجل الإحداثي الثاني. إذن
$\vect{IJ} = \vect{LK}$ (وكلاهما يساوي $\frac12\vect{AC}$)، فيكون $IJKL$
متوازي أضلاع، أيًّا كان شكل الرباعي $ABCD$.
\end{solution}

\begin{solution}{pb:g10:vectors:1}
\textbf{1.} $\vect{AB} + \vect{BC} + \vect{CD} = \vect{AD}$؛
\quad $\vect{MA} - \vect{MB} = \vect{MA} + \vect{BM} =
\vect{BA}$؛ \quad $\vect{AB} + \vect{BC} + \vect{CD} +
\vect{DA} = \vect{AA} = \vec 0$ (بإعادة \omterm{def:g10:coordgeom:system}{ترتيب} \omterm{def:g10:vectors:sum}{المجموع} الثالث).

\textbf{2.} $\vect{OB} - \vect{OA} = \vect{AO} + \vect{OB} =
\vect{AB}$ حسب شال.

\textbf{3.} تكون $I$ \omterm{prop:g10:coordgeom:midpoint}{منتصف} $[AB]$ بالضبط عندما
$\vect{IA} = -\vect{IB}$، أي
$\vect{IA} + \vect{IB} = \vec 0$. عندئذٍ، من أجل أي $O$:
$\vec 0 = \vect{IO} + \vect{OA} + \vect{IO} + \vect{OB}$، إذن
$2\,\vect{OI} = \vect{OA} + \vect{OB}$ --- والعكس صحيح.

\textbf{4.} $\vect{AB} = (4, 2)$، و $D$ تحقق
$\vect{DC} = \vect{AB}$: $D = C - (4, 2) = (2, -2)$ --- وهي
$D$ نفسها في \cref{pb:g10:coordgeom:1}، لكن أسرع بسطر.

\textbf{5.} الانسحاب ذو \omterm{def:g10:vectors:vector}{المتجهة} $\vec u$ --- وهو بالضبط
التحويل الناتج عن مرآتين متوازيتين أو نصفي
دورة. وصورة $(1, 2)$: هي $(4, 1)$.

\textbf{6.} \omterm{def:g10:algebra:expand}{بنشر} كل $\vect{GX}$ على صورة
$\vect{GO} + \vect{OX}$:
$3\,\vect{GO} + \vect{OA} + \vect{OB} + \vect{OC} = \vec 0$،
إذن $\vect{OG} = \frac13(\vect{OA} + \vect{OB} + \vect{OC})$:
فهي نقطة واحدة لا غير، إحداثياتها متوسطات \omterm{def:g10:coordgeom:system}{إحداثيات}
الرؤوس الثلاثة.

\textbf{7.} $\vect{AB} + \vect{AC} =
(\vect{AG} + \vect{GB}) + (\vect{AG} + \vect{GC})
= 2\vect{AG} + (\vect{GB} + \vect{GC})
= 2\vect{AG} - \vect{GA} = 3\vect{AG}$، ومنه
$\vect{AG} = \frac13(\vect{AB} + \vect{AC})$. وتعطي صيغة
\omterm{prop:g10:coordgeom:midpoint}{المنتصف} (السؤال 3، منظورًا إليها من $A$)
$\vect{AK} = \frac12(\vect{AB} + \vect{AC})$. وبالمقارنة:
$\vect{AG} = \frac23\vect{AK}$ --- إذن تقع $G$ على المتوسط
$[AK]$، على ثلثي الطريق من $A$.

\textbf{8.} يعامل الشرط المعرِّف النقط $A$ و $B$ و $C$
معاملة واحدة، إذن يضع الحساب نفسه انطلاقًا من $B$ أو من $C$
النقطةَ $G$ \emph{نفسها} على ثلثي هذين المتوسطين:
فالتلاقي لا يكلّف شيئًا. أما برهان متوازي الأضلاع فكان
بارعًا لكنه خاص بالمسألة؛ \omterm{def:g10:coordgeom:system}{والإحداثيات} تحسب لكنها تحجب؛
أما المتجهات فتُظهر التناظر وتجعل البرهان سطرين.

\textbf{9.} $G\left(\frac{1+5+3}{3}, \frac{1+2+6}{3}\right) =
(3, 3)$؛ $K(4, 4)$؛ و $\vect{AG} = (2, 2)$ و
$\vect{AK} = (3, 3)$: المحدد
$2 \times 3 - 2 \times 3 = 0$، إذن مستقيميتان، وفعلًا
$\vect{AG} = \frac23\vect{AK}$.

\textbf{10.} $G'\left(\frac{2 + 5 + 3}{4}, \frac{2 + 2 +
6}{4}\right) = (2.5,\ 2.5)$: مشدودة من $G(3,3)$ نحو
الرأس المضاعف $A(1,1)$، كما ينبغي لنقطة توازن.

\textbf{11.} $3 \times 4 - (-2)(-6) = 12 - 12 = 0$: مستقيميتان
($\vec v = -2\vec u$). و$2 \times 9 - 5 \times 4 = -2 \neq 0$:
ليستا \omterm{def:g10:vectors:collinear}{مستقيميتين}.

\textbf{12.} $\vect{AB} = (2, 3)$ و $\vect{AC} = (6, 9)$:
المحدد $2 \times 9 - 3 \times 6 = 0$: على استقامة واحدة.

\textbf{13.} $\vect{OI} = \frac12(\vect{OA} + \vect{OB})$ و
$\vect{OK} = \frac12(\vect{OC} + \vect{OD})$، إذن \omterm{def:g10:vectors:vector}{متجهة} موضع \omterm{prop:g10:coordgeom:midpoint}{منتصف}
$[IK]$ هي
$\frac12(\vect{OI} + \vect{OK}) = \frac14(\vect{OA} +
\vect{OB} + \vect{OC} + \vect{OD})$ --- والحساب
من أجل $[JL]$ يعطي \omterm{def:g10:vectors:vector}{المتجهة} نفسها تمامًا. فالمتوسطان الثنائيان (وكذلك،
حسب فارينيون، قطرا متوازي الأضلاع $IJKL$) يتقاسمان
هذا المركز: أي مركز ثقل الرباعي.

\textbf{14.} $\vect{MN} = \vect{AN} - \vect{AM} =
\frac13\vect{AC} - \frac13\vect{AB} = \frac13\vect{BC}$:
إذن $(MN) \parallel (BC)$ و $MN = \frac13 BC$.

\textbf{15.} من أجل $\vect{AM} = k\,\vect{AB}$ و
$\vect{AN} = k\,\vect{AC}$: $\vect{MN} = k\,\vect{BC}$ --- أي
سطر واحد يحوي مبرهنة المنتصفين ($k = \frac12$) و
مبرهنة طاليس (من أجل أي $k$): فالنسبة تمر عبر الطرح
دون أن تُمسّ.

\textbf{16.} المحصلة $(3, 4)$، وطويلتها
$\sqrt{9 + 16} = 5$ كم/سا: فالقارب يسير فعليًا
نحو الشمال الشرقي مائلًا إلى الشمال، على وتر
المثلث $3$--$4$--$5$.

\textbf{17.} $\vec u + \vec v + \vec w = (0, 0)$: فالحلقة
تبقى في مكانها --- إنه التوازن. وعلى مضلع مغلق
$A_1 A_2 \dots A_n A_1$، تطوي علاقة شال \omterm{def:g10:vectors:sum}{المجموع} إلى
$\vect{A_1 A_1} = \vec 0$: فالدوران حول المضلع يعيدك إلى نقطة البداية.

\textbf{18.} $\vec w = -(\vec u + \vec v) = -(3, 2) =
(-3, -2)$.

\textbf{19.} المحصلة $(1.5,\ 2)$، وطويلتها
$\sqrt{2.25 + 4} = 2.5$ م/ثا. والعبور: $50$ م بسرعة $2$ م/ثا
عرضيًا: أي $25$ ثا؛ والانزياح: $1.5 \times 25 = 37.5$ م مع التيار.

\textbf{20.} أُعيد برهانه في نهاية الأسبوع هذه: مبرهنة ثلثي
مركز الثقل (السؤال 7)، ومركز المتوسطين الثنائيين عند فارينيون (السؤال
13)، وطاليس والمنتصفان (السؤالان 14--15). أما ما تعجز عنه الأسهم
حتى الآن فهو قياس \emph{الزوايا} (ومن ثم الأطوال في
اتجاهات كيفية) --- والجداء السلّمي في السنة المقبلة هو
الأداة الناقصة بالضبط.
\end{solution}
