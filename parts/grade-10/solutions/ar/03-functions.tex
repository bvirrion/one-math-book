\begin{solution}{exo:g10:functions:3}
\emph{سوابق $0$:} $x^2 - 2x = x(x-2) = 0$، إذن $0$ و $2$.

\emph{سوابق $3$:} $x^2 - 2x = 3$، أي $x^2 - 2x - 3 = 0$، أي
$(x-3)(x+1) = 0$ (تحقق \omterm{def:g10:algebra:expand}{بالنشر})، إذن $3$ و $-1$.

\emph{سوابق $-1$:} $x^2 - 2x + 1 = (x - 1)^2 = 0$، إذن \omterm{def:g10:functions:function}{السابقة}
الوحيدة $1$.
\end{solution}

\begin{solution}{exo:g10:functions:5}
\emph{1.} أكبر قيمة في الجدول هي $4 = g(3)$ (\omterm{def:g10:functions:extrema}{القيمة العظمى})، وأصغرها
هي $-2 = g(0)$ (\omterm{def:g10:functions:extrema}{القيمة الصغرى}).

\emph{2.} على $\intcc{-3}{0}$ تتناقص \omterm{def:g10:functions:function}{الدالة}، و
$-1 < -0.5$، إذن $g(-1) \geq g(-0.5)$.

\emph{3.} على كل \omterm{def:g10:numbers:interval}{فترة} رتابة، تأخذ $g$ كل قيمة مرة واحدة
على الأكثر، إذن \omterm{def:g10:algebra:equation}{للمعادلة} $g(x) = 0$ حل واحد على الأكثر في كل \omterm{def:g10:numbers:interval}{فترة}: أي $3$
حلول على الأكثر إجمالًا. (وهنا يقع $0$ بين $-2$ و $1$، وبين $-2$ و
$4$، وبين $0$ و $4$، فالحلول ثلاثة بالضبط.)
\end{solution}

\begin{solution}{exo:g10:functions:6}
ليكن $u < v$. عندئذٍ
\[
f(v) - f(u) = (-2v + 7) - (-2u + 7) = -2(v - u) < 0,
\]
لأن $v - u > 0$. إذن $f(v) < f(u)$: أي أن $f$ \omterm{def:g10:functions:variations}{متناقصة} تمامًا على $\R$.
\end{solution}

\begin{solution}{exo:g10:functions:7}
أتمّ المربع: $(x+3)^2 = x^2 + 6x + 9$، إذن
\[
f(x) = x^2 + 6x + 11 = (x + 3)^2 + 2 .
\]
ومن أجل كل $x$ يكون $(x+3)^2 \geq 0$، إذن $f(x) \geq 2$؛ و
$f(-3) = 0 + 2 = 2$. إذن \omterm{def:g10:functions:extrema}{القيمة الصغرى} \omterm{def:g10:functions:function}{للدالة} $f$ هي $2$، وتُبلغ عند $x = -3$.
\end{solution}

\begin{solution}{exo:g10:functions:8}
\emph{1.} الطول $+$ العرض $= 20$، إذن الطول هو $20 - x$ و
\[
A(x) = x(20 - x).
\]
ويجب أن يكون البعدان موجبين: $0 < x < 20$.

\emph{2.} $A(4) = 64$ و $A(8) = 96$ و $A(10) = 100$ و $A(12) = 96$ و
$A(16) = 64$. ترتفع القيم حتى $x = 10$ ثم تنخفض بتناظر:
فيبدو أن المساحة أكبر ما تكون من أجل $x = 10$، أي من أجل حديقة \emph{مربعة}.
(ويبرهن \cref{ch:g11:quad} على هذا التخمين.)
\end{solution}

\begin{solution}{exo:g10:functions:9}
\emph{1.} من أجل كل $x$: $x^2 + 1 \geq 1 > 0$، إذن $f(x) = \frac{1}{x^2+1}$
موجب، وبما أن $x^2 + 1 \geq 1$، فإن أخذ المقلوبين (والطرفان
موجبان) يعطي $f(x) \leq 1$.

\emph{2.} \omterm{def:g10:functions:extrema}{القيمة العظمى}: $f(0) = 1$ و $f(x) \leq 1$ من أجل كل $x$، إذن $1$ هي
\omterm{def:g10:functions:extrema}{القيمة العظمى}، وتُبلغ عند $0$. أما \omterm{def:g10:functions:extrema}{القيمة الصغرى}: فإن $f(x) > 0$ دائمًا، لكن ما من قيمة للمجهول $x$
تحقق $0$ (فخارج قسمة عددين غير معدومين غير معدوم)، وتأخذ $f$
قيمًا قريبة من $0$ بأي قدر نشاء من أجل $x$ كبير؛ إذن ليس \omterm{def:g10:functions:function}{للدالة} $f$ \omterm{def:g10:functions:extrema}{قيمة صغرى}.
\end{solution}

\begin{solution}{exo:g10:functions:10}
\emph{1.} $f(v) - f(u) = v^2 - u^2 = (v-u)(v+u)$. إذا كان $0 \leq u < v$، فإن
$v - u > 0$ و $v + u > 0$ (مجموع عدد غير سالب وعدد موجب)،
إذن $f(v) - f(u) > 0$: أي أن $f$ \omterm{def:g10:functions:variations}{متزايدة} تمامًا على $\intco{0}{+\infty}$.

\emph{2.} وإذا كان $u < v \leq 0$، فإن $v - u > 0$ كذلك، لكن
$v + u < 0$ الآن (مجموع عدد غير موجب وعدد سالب)، إذن
$f(v) - f(u) < 0$: أي أن $f$ \omterm{def:g10:functions:variations}{متناقصة} تمامًا على $\intoc{-\infty}{0}$.
\end{solution}

\begin{solution}{pb:g10:functions:1}
\textbf{1.} \omterm{def:g10:functions:function}{مجموعة التعريف}: $x \neq 0$ (قسمة على $x$). $f(1) =
\frac12(1 + 2) = \frac32$؛ $f(2) = \frac12(2 + 1) = \frac32$؛
$f\!\left(\frac32\right) = \frac12\left(\frac32 +
\frac43\right) = \frac{17}{12}$.

\textbf{2.} $\frac12\left(x + \frac2x\right) = \frac32$ يعطي
$x^2 + 2 = 3x$، أي $x^2 - 3x + 2 = (x - 1)(x - 2) = 0$: إذن
سابقتا $\frac32$ هما $1$ و $2$ --- كما أعلن
السؤال 1.

\textbf{3.} $f(x) = x$ يعطي $x + \frac2x = 2x$، إذن
$\frac2x = x$ و $x^2 = 2$: فالنقطتان الثابتتان $\sqrt2$ و $-\sqrt2$.
والموجبة منهما هي قطر مربع الوحدة، وهو عدد أصمّ
حسب مسألة نهاية الأسبوع عن الصمم في الكتاب السابق.

\textbf{4.} $1 \to \frac32 \to \frac{17}{12} \to
\frac{577}{408}$: وهي تقريبات هيرون للعدد $\sqrt2$. ووصفته
``خذ متوسط التخمين مع $2/$التخمين'' هي بالضبط
\emph{تكرار \omterm{def:g10:functions:function}{الدالة} $f$} --- والعدد الذي
يطارده المسار هو \omterm{pb:g10:functions:1}{النقطة الثابتة} \omterm{def:g10:functions:function}{للدالة} $f$.

\textbf{5.} النقطتان الثابتتان هما تقاطعا \omterm{def:g10:functions:graph}{المنحنى}
مع القطر $y = x$: فإعادة إطعام الخرج تمشي
بالنقطة على \omterm{def:g10:functions:graph}{المنحنى} نحو ذلك \omterm{def:g10:numbers:interunion}{التقاطع}. وفي
مسألة نهاية الأسبوع عن الميزانين الحراريين في الكتاب السابق، كانت القراءة $-40^\circ$ هي
فكرة \omterm{def:g10:numbers:interunion}{تقاطع} القطر نفسها من أجل \omterm{def:g10:functions:function}{دالة} التحويل.

\textbf{6.} $f(v) - f(u) = \frac{v - u}{2} + \frac1v - \frac1u
= \frac{v - u}{2} + \frac{u - v}{uv}
= (v - u)\left(\frac12 - \frac{1}{uv}\right)
= \frac{(v - u)(uv - 2)}{2uv}$. من أجل $0 < u < v \leq \sqrt2$:
يكون $uv < 2$، فالقوس سالب و $f(v) < f(u)$: \omterm{def:g10:functions:variations}{متناقصة}.
ومن أجل $\sqrt2 \leq u < v$: يكون $uv > 2$: \omterm{def:g10:functions:variations}{متزايدة}. ونقطة الانعطاف:
$\sqrt2$.

\textbf{7.} \omterm{def:g10:functions:variations}{متناقصة} قبل $\sqrt2$ \omterm{def:g10:functions:variations}{ومتزايدة} بعده: إذن تبلغ $f$
قيمتها الصغرى عند $\sqrt2$، وهي تساوي
$f(\sqrt2) = \frac12\left(\sqrt2 + \frac{2}{\sqrt2}\right)
= \sqrt2$. والنتيجة: كل خرج من الآلة (من أجل دخل
موجب) لا يقل عن $\sqrt2$ --- فبعد دورة واحدة من المرفع،
تعيش التخمينات في $\intco{\sqrt2}{+\infty}$.

\textbf{8.} إذا كان $x > \sqrt2$، فإن $x^2 > 2$، إذن
$\frac2x < x$ ويكون متوسط $f(x)$ للعددين $x$ و $\frac2x$
أصغر من $x$؛ و$f(x) \geq \sqrt2$ حسب السؤال 7، مع
تحقق المساواة عند $x = \sqrt2$ فقط. إذن
$\sqrt2 < f(x) < x$: فكل تكرار يحسّن تمامًا دون
تجاوز --- والمتتالية في السؤال 4 تنزلق نازلة إلى
$\sqrt2$.

\textbf{9.} على $\intoo{0}{+\infty}$: \omterm{def:g10:functions:variations}{متناقصة} من
$+\infty$ (قرب $0$) نزولًا إلى \omterm{def:g10:functions:extrema}{القيمة الصغرى} $\sqrt2$ عند
$x = \sqrt2$، ثم \omterm{def:g10:functions:variations}{متزايدة} بلا حد.

\textbf{10.} $g(x) = x$ يعطي $x^2 = x$ و $x(x - 1) = 0$: فالنقطتان
الثابتتان $0$ و $1$. انطلاقًا من $0.9$: $0.81$ و $0.656$ و $0.430$ و
$0.185$ --- انزلاق نحو $0$. وانطلاقًا من $1.1$: $1.21$ و $1.464$ و
$2.144$ و $4.595$ --- هروب إلى ما لا نهاية. \omterm{pb:g10:functions:1}{فالنقطة الثابتة} $0$
جاذبة و $1$ طاردة: فرق شعرة في البداية، ومصيران
متعاكسان.

\textbf{11.} $7 \to 22 \to 11 \to 34 \to 17 \to 52 \to 26 \to
13 \to 40 \to 20 \to 10 \to 5 \to 16 \to 8 \to 4 \to 2 \to 1$:
ست عشرة خطوة، وذروة عند $52$.

\textbf{12.} $27 \to 82 \to 41 \to 124 \to 62 \to 31 \to 94 \to
47 \to 142 \to 71 \to 214 \to \dots$ --- ما زال يتسلق بعد
عشر خطوات، في طريقه إلى ذروة عند $9\,232$ ورحلة من
$111$ خطوة. والدرس: قاعدة من سطرين قد تنتج سلوكًا لا يستطيع أحد
التنبؤ به بنظرة --- فالأعداد المتجاورة ($26$ و $27$ و $28$)
لها رحلات شديدة الاختلاف.

\textbf{13.} تتنصّف قوة العدد $2$ نزولًا على سلّمها كله:
$2^k \to 2^{k-1} \to \dots \to 2 \to 1$. وفي القاع،
$h(1) = 4$ و $h(4) = 2$ و $h(2) = 1$: فتدخل الرحلة الدورة
$4 \to 2 \to 1 \to 4$. \omterm{pb:g10:functions:1}{والنقطة الثابتة} تستلزم $\frac n2 = n$
(ولا يحقق ذلك سوى $n = 0$) أو $3n + 1 = n$ (وهو سالب): فلا \omterm{pb:g10:functions:1}{نقطة ثابتة} بين
\omterm{def:g10:numbers:sets}{الأعداد الصحيحة} الموجبة --- والرحلات المرصودة تنتهي لا عند \omterm{pb:g10:functions:1}{نقطة
ثابتة} بل في تلك \emph{الدورة} الصغيرة.

\textbf{14.} المثال المضاد إما رحلة تنمو
إلى الأبد (ولا تعود أبدًا إلى $1$) وإما دورة ثانية
منفصلة عن $4 \to 2 \to 1$. والتحقق حتى $10^{20}$
يترك عددًا لا نهائيًا من الأعداد غير المختبرة --- وهو بالضبط
درس مناطق الدائرة في مسألة نهاية الأسبوع عن عرّاف أعواد الثقاب في الكتاب السابق: فالتوافق على
عدد منته من الحالات لا يبرهن على شيء بشأنها جميعًا.

\textbf{15.} في آلة هيرون كان لدينا \emph{بنية} --- فرق
محلَّل إلى عوامل (السؤال 6) يكشف التغيرات، \omterm{def:g10:functions:extrema}{وقيمة
صغرى}، وحصر --- بينما لا تقدّم قاعدة آلة البَرَد المتبدّلة
بالزوجية أي مقبض كهذا: فما من أحد وجد البنية التي
تروّضها.

\textbf{16.} $\sqrt{x - 3}$: تستلزم $x - 3 \geq 0$:
$\intco{3}{+\infty}$. و$\frac{1}{x^2 - 9}$: تستلزم
$x \neq \pm 3$: أي $\R$ محرومة من $-3$ و $3$.
و $\sqrt{9 - x^2}$: تستلزم $x^2 \leq 9$: أي $\intcc{-3}{3}$.

\textbf{17.} $f(x) = 3$ يعطي $x + \frac2x = 6$، إذن
$x^2 - 6x + 2 = 0$، أي $(x - 3)^2 = 7$:
فيكون $x = 3 - \sqrt7$ أو $x = 3 + \sqrt7$ (وكلاهما موجب: سابقتان
مضبوطتان).

\textbf{18.} يكون للعبارة معنى فيزيائي ما دام الحجر محلقًا:
$H(t) = 5t(4 - t) \geq 0$ من أجل $t \in \intcc{0}{4}$. وبإتمام المربع:
$H(t) = 20 - 5(t - 2)^2$: أي ارتفاع أعظمي $20$ م عند $t = 2$ ثا.
والتغيرات: \omterm{def:g10:functions:variations}{متزايدة} على $\intcc{0}{2}$ من $0$ إلى $20$،
\omterm{def:g10:functions:variations}{ومتناقصة} على $\intcc{2}{4}$ عودًا إلى $0$.

\textbf{19.} $20t - 5t^2 \geq 15 \iff t^2 - 4t + 3 \leq 0 \iff
(t - 1)(t - 3) \leq 0 \iff t \in \intcc{1}{3}$: أي ثانيتان كاملتان
فوق $15$ م.

\textbf{20.} لكل آلة \omterm{def:g10:functions:function}{مجموعة تعريف} (حيث تُسمح
القاعدة)، وقاعدة (العبارة أو التبديل بالزوجية)، \omterm{def:g10:functions:graph}{ومنحنى}
أو جدول يعرضها كاملة. وسألنا كلًّا منها السؤالين
الكبيرين: \emph{إلى أين يقود التكرار} (نقط
ثابتة، ودورات، وجذب --- محسومة عند هيرون، مفتوحة عند
البَرَد) و\emph{كيف يتحرك الخرج مع الدخل}
(التغيرات والقيم الحدية --- ذروة الحجر، \omterm{def:g10:functions:extrema}{والقيمة الصغرى} عند هيرون).
والفصول المقبلة تصقل الأمرين: الدوال المرجعية أولًا،
ثم، في سنوات لاحقة، المشتقات لقياس التغير
والنهايات لتوثيق مستقر التكرارات.
\end{solution}
