\section*{الفصل \ref{ch:g10:algebra} --- الجبر: المعادلات والمتراجحات}

\begin{solution}{exo:g10:algebra:1}
$3(2x-5) - 2(x-7) = 6x - 15 - 2x + 14 = 4x - 1$.

$(x+4)(2x-1) = 2x^2 - x + 8x - 4 = 2x^2 + 7x - 4$.

$(3x-2)^2 = 9x^2 - 12x + 4$.

$(5-x)(5+x) = 25 - x^2$.
\end{solution}

\begin{solution}{exo:g10:algebra:2}
$x^2 - 49 = (x+7)(x-7)$ (فرق مربعين).

$x^2 + 10x + 25 = (x+5)^2$.

$7x^2 - 14x = 7x(x - 2)$ (عامل مشترك).

$16x^2 - 8x + 1 = (4x - 1)^2$.
\end{solution}

\begin{solution}{exo:g10:algebra:3}
$4x + 9 = x - 3$: نجد $3x = -12$، إذن $x = -4$.

$\frac{2x-1}{3} = 5$: نجد $2x - 1 = 15$، إذن $x = 8$.

$2(x-3) = 2x + 1$ تُنشر إلى $2x - 6 = 2x + 1$، أي $-6 = 1$: وهذا خطأ من أجل
كل $x$. لا حل لها --- فالطرفان يعرّفان مستقيمين متوازيين لا
يلتقيان أبدًا.
\end{solution}

\begin{solution}{exo:g10:algebra:4}
$(x+5)(2x-8) = 0$: $x = -5$ أو $x = 4$.

$x^2 - 16 = (x-4)(x+4) = 0$: $x = 4$ أو $x = -4$.

$x^2 + 6x + 9 = (x+3)^2 = 0$: الحل الوحيد $x = -3$.
\end{solution}

\begin{solution}{exo:g10:algebra:5}
$3x - 5 \leq 7$: نجد $3x \leq 12$، إذن $x \leq 4$: مجموعة الحلول
$\intoc{-\infty}{4}$.

$-4x + 1 > 9$: نجد $-4x > 8$، والقسمة على $-4$ تقلب المتراجحة:
$x < -2$: مجموعة الحلول $\intoo{-\infty}{-2}$.

$2(x-1) \geq 5x + 4$: نجد $2x - 2 \geq 5x + 4$، إذن $-3x \geq 6$، إذن
$x \leq -2$: مجموعة الحلول $\intoc{-\infty}{-2}$.
\end{solution}

\begin{solution}{exo:g10:algebra:6}
العامل المشترك هو $(x-1)$:
\[
E = (x-1)\bigl[(x+3) - (2x - 5)\bigr] = (x-1)(x + 3 - 2x + 5)
= (x-1)(8 - x).
\]
إذن $E = 0$ عندما $x = 1$ أو $x = 8$ (قاعدة الجداء المعدوم).
\end{solution}

\begin{solution}{exo:g10:algebra:7}
$x^2 = 5x$: نجد $x^2 - 5x = x(x - 5) = 0$، إذن $x = 0$ أو $x = 5$ (ولا
تقسم على $x$ أبدًا: فالحل $0$ سيضيع).

$(2x+1)^2 = (x-4)^2$: انقل إلى اليسار وحلّل فرق
المربعين:
\[
(2x+1)^2 - (x-4)^2
= \bigl[(2x+1) + (x-4)\bigr]\bigl[(2x+1) - (x-4)\bigr]
= (3x - 3)(x + 5).
\]
وبقاعدة الجداء المعدوم: $x = 1$ أو $x = -5$.
\end{solution}

\begin{solution}{exo:g10:algebra:8}
$(x+2)(4-x) > 0$: ينعدم العامل $x + 2$ عند $-2$ (ويكون موجبًا بعدها)،
وينعدم $4 - x$ عند $4$ (ويكون موجبًا قبلها). ويكون الجداء موجبًا
بالضبط عندما يكون العاملان موجبين معًا، أي على $\intoo{-2}{4}$.

$\frac{2x-6}{x+1} \geq 0$: ينعدم البسط عند $3$، وينعدم المقام عند
$-1$ (قيمة ممنوعة). \omterm{def:g10:algebra:signtable}{جدول الإشارة}: يكون الخارج موجبًا عندما يكون
للطرفين الإشارة نفسها، أي من أجل $x < -1$ أو $x > 3$؛ وينعدم عند $x = 3$.
مجموعة الحلول: $\intoo{-\infty}{-1} \cup \intco{3}{+\infty}$.
\end{solution}

\begin{solution}{exo:g10:algebra:9}
بعد $x$ شهرًا تكلّف الخدمة الأولى $9x + 15$ وتكلّف الثانية $12x$.
وتكون الأولى أرخص عندما
\[
9x + 15 < 12x
\ \Longleftrightarrow\
15 < 3x
\ \Longleftrightarrow\
x > 5 .
\]
ابتداءً من الشهر $6$ تصير الخدمة الأولى الخيار الأرخص إجمالًا.
\end{solution}

\begin{solution}{exo:g10:algebra:10}
القيمة الممنوعة: $x = 1$. ومن أجل $x \neq 1$، اضرب الطرفين في $x - 1$:
\[
x + 3 = 2(x - 1) = 2x - 2
\ \Longleftrightarrow\
x = 5 .
\]
وبما أن $5 \neq 1$، فالحل هو $x = 5$. تحقق:
$\frac{5+3}{5-1} = \frac84 = 2$.
\end{solution}

\begin{solution}{exo:g10:algebra:11}
$(n+1)^2 - n^2 = n^2 + 2n + 1 - n^2 = 2n + 1$، وهو فردي من أجل كل
\omterm{def:g10:numbers:sets}{عدد صحيح} $n$.

يمكن كتابة عددين فرديين متتاليين على الصورة $2n - 1$ و $2n + 1$. عندئذٍ
\[
(2n+1)^2 - (2n-1)^2
= \bigl[(2n+1)+(2n-1)\bigr]\bigl[(2n+1)-(2n-1)\bigr]
= 4n \times 2 = 8n,
\]
وهو مضاعف للعدد $8$.
\end{solution}

\begin{solution}{pb:g10:algebra:1}
\textbf{1.} $x^2 - 10x + 25 = (x - 5)^2$؛
$9x^2 - 49 = (3x - 7)(3x + 7)$؛
$x^2 + 6x + 9 - 4 = (x + 3)^2 - 2^2 = (x + 1)(x + 5)$.

\textbf{2.} $x = -1$ أو $x = -5$؛ \quad $x = \frac73$ أو
$x = -\frac73$.

\textbf{3.} $(5 - t)(5 + t) = 25 - t^2 = 21$ يعطي $t^2 = 4$، أي
$t = 2$: والعددان هما $3$ و $7$ (المجموع $10$، والجداء $21$:
تحقق).

\textbf{4.} $(7 - t)(7 + t) = 49 - t^2 = 33$: نجد $t = 4$، والعددان
$3$ و $11$.

\textbf{5.} $\left(\frac s2 - t\right)\left(\frac s2 + t\right)
= \left(\frac s2\right)^2 - t^2 = p$ يفرض
$t^2 = \left(\frac s2\right)^2 - p$. ويوجد حل بالضبط
عندما يكون هذا المقدار $\geq 0$، أي عندما
$p \leq \left(\frac s2\right)^2$ --- فجداء عددين
معلوم مجموعهما لا يمكن أن يتجاوز مربع نصف المجموع
(وهذا ما كانت تعرفه مسألة سياج الملكة ديدو في الكتاب السابق).

\textbf{6.} $(x - a)(x - b) = x^2 - (a + b)x + ab$: \omterm{def:g10:algebra:equation}{فالمعادلة}
$x^2 - sx + p = 0$ تقول بالضبط ``$x$ أحد عددين
مجموعهما $s$ وجداؤهما $p$''.

\textbf{7.} $(x-5)^2 - 4 = x^2 - 10x + 25 - 4 = x^2 - 10x +
21$. وبفرق المربعين:
$(x - 5 - 2)(x - 5 + 2) = (x - 7)(x - 3) = 0$: الحلان $3$
و $7$ --- وهما عددا الكاتب في السؤال 3.

\textbf{8.} $x^2 + 6x - 7 = (x + 3)^2 - 16 = 0$ يعطي
$(x + 3)^2 = 16$، أي $x + 3 = \pm 4$: فيكون $x = 1$ أو $x = -7$.

\textbf{9.} $x^2 + 4x + 5 = (x + 2)^2 + 1 \geq 1 > 0$ من أجل كل
$x$: فلا حل حقيقي. وعلى الصورة $(x + h)^2 + k$: توجد الحلول
بالضبط عندما $k \leq 0$ (وعندئذٍ \omterm{def:g10:algebra:equation}{للمعادلة} $(x+h)^2 = -k$ الجذران
$-h \pm \sqrt{-k}$)؛ أما عندما $k > 0$ فتبقى العبارة
موجبة.

\textbf{10.} المربع ذو الضلع $x$ (مساحته $x^2$) مع مستطيلين
بعداهما $3 \times x$ (مساحتهما $6x$ إجمالًا) يشكّل شكلًا على هيئة حرف
سلّم مساحته $x^2 + 6x = 7$؛ والزاوية الناقصة $3 \times 3$ (مساحتها $9$)
تتمّ مربعًا كبيرًا ضلعه $x + 3$. وبإضافة $9$ إلى
الطرفين: $(x + 3)^2 = 16$، إذن $x + 3 = 4$ (فالأطوال موجبة)
و $x = 1$. الجبر والرسم فعل واحد.

\textbf{11.} $(x-3)(x+2)$: سالب بين الجذرين. فحل
$< 0$ هو: $x \in \intoo{-2}{3}$.

\textbf{12.} $x^2 - 10x + 21 \leq 0$، أي
$(x - 3)(x - 7) \leq 0$: بين الجذرين، مع إدراج الطرفين:
$x \in \intcc{3}{7}$.

\textbf{13.} ينعدم عند $x = 1$، ويمنع عند $x = -2$؛ ويكون
الخارج موجبًا خارج الجذرين:
$x \in \intoo{-\infty}{-2} \cup \intco{1}{+\infty}$.

\textbf{14.} $25 - (x - 5)^2 = 25 - x^2 + 10x - 25 =
x(10 - x)$: المتطابقة متحققة. وبما أن $(x-5)^2 \geq 0$ دائمًا،
فإن $A(x) \leq 25$، مع تحقق المساواة عند $x = 5$ بالضبط: أي
الحظيرة المربعة $5 \times 5$، ومساحتها العظمى $25$ م$^2$ --- وهو
الجدول العددي القديم، وقد صار برهانًا من سطرين.

\textbf{15.} من أجل $x > 0$، اضرب الادعاء في $x$:
$x^2 + 1 \geq 2x$، أي $x^2 - 2x + 1 = (x - 1)^2 \geq 0$ ---
وهذا صحيح، مع تحقق المساواة عند $x = 1$ فقط. والقسمة من جديد على $x > 0$
تحفظ كل شيء (\cref{prop:g10:algebra:ineqrules}).

\textbf{16.} بالضرب التبادلي في $\frac1x = \frac{x}{1 - x}$
(وكل الأطوال موجبة): $x^2 = 1 - x$، أي
$x^2 + x - 1 = 0$.

\textbf{17.} $x^2 + x - 1 = \left(x + \frac12\right)^2 -
\frac54 = 0$ يعطي $x + \frac12 = \pm\frac{\sqrt5}{2}$. والجذر
الموجب: $x = \frac{\sqrt5 - 1}{2} \approx 0.618$.

\textbf{18.} $\varphi = \frac{2}{\sqrt5 - 1} =
\frac{2(\sqrt5 + 1)}{(\sqrt5 - 1)(\sqrt5 + 1)}
= \frac{2(\sqrt5 + 1)}{5 - 1} = \frac{\sqrt5 + 1}{2}
\approx 1.618$.

\textbf{19.} $\varphi^2 = \frac{(\sqrt5 + 1)^2}{4} =
\frac{6 + 2\sqrt5}{4} = \frac{3 + \sqrt5}{2}$، و
$\varphi + 1 = \frac{\sqrt5 + 3}{2}$: متساويان. و
$\varphi - 1 = \frac{\sqrt5 - 1}{2} = x = \frac1\varphi$ (وهكذا
عُرّف $\varphi$). فالنسبة الذهبية هي العدد الذي
يضيف مربعُه واحدًا ويطرح مقلوبُه واحدًا.

\textbf{20.} العدد $\sqrt5$ أصمّ (مسألة نهاية الأسبوع عن الصمم في الكتاب السابق: فالعدد $5$
ليس مربعًا كاملًا)، وإضافة $1$ ثم التنصيف يحفظان
كون العدد أصمّ (عمليات ناطقة، \cref{pb:g10:numbers:1}).
النسب: $\frac53 \approx 1.667$ و $\frac85 = 1.600$ و
$\frac{13}{8} = 1.625$ و $\frac{21}{13} \approx 1.615$:
تعلو وتنخفض بالتناوب، وتضيق حول
$\varphi = 1.618\ldots$ --- فأعداد الأوزان في
مسألة نهاية الأسبوع عن عدّ الأوزان في الكتاب السابق تطارد المقطع الذهبي في الخفاء؛
وفصول المتتاليات ستضبطها متلبسة.
\end{solution}
