\section*{الفصل \ref{ch:g10:coordgeom} --- الهندسة التحليلية}

\begin{solution}{exo:g10:coordgeom:1}
\omterm{prop:g10:coordgeom:midpoint}{المنتصف}: $\left(\frac{2 + (-4)}{2}, \frac{5 + 1}{2}\right) = (-1, 3)$.
المسافة:
\[
AB = \sqrt{(-4 - 2)^2 + (1 - 5)^2} = \sqrt{36 + 16} = \sqrt{52}
= 2\sqrt{13}.
\]
\end{solution}

\begin{solution}{exo:g10:coordgeom:2}
كون $I$ \omterm{prop:g10:coordgeom:midpoint}{منتصف} $[AB]$ يعني $\frac{3 + x_B}{2} = 1$ و
$\frac{-2 + y_B}{2} = 2$، إذن $x_B = -1$ و $y_B = 6$: أي $B(-1, 6)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:3}
$OP = \sqrt{16 + 1} = \sqrt{17}$؛ $OQ = \sqrt{9 + 4} = \sqrt{13}$؛
$OR = \sqrt{0 + 25} = 5 = \sqrt{25}$. وبما أن $13 < 17 < 25$، والجذر
التربيعي متزايد، فإن $Q$ هي الأقرب إلى المبدأ.
\end{solution}

\begin{solution}{exo:g10:coordgeom:4}
مربعات الأطوال:
\begin{align*}
AB^2 &= (5-1)^2 + (4-2)^2 = 16 + 4 = 20, \\
BC^2 &= (7-5)^2 + (0-4)^2 = 4 + 16 = 20, \\
AC^2 &= (7-1)^2 + (0-2)^2 = 36 + 4 = 40 .
\end{align*}
نجد $AB = BC$: فالمثلث متساوي الساقين في $B$. وعلاوة على ذلك
$AB^2 + BC^2 = 40 = AC^2$: فحسب عكس مبرهنة فيثاغورس يكون
قائم الزاوية في $B$ أيضًا.
\end{solution}

\begin{solution}{exo:g10:coordgeom:5}
\emph{1.} \omterm{prop:g10:coordgeom:midpoint}{منتصف} $[AC]$:
$\left(\frac{-2+4}{2}, \frac{1+1}{2}\right) = (1, 1)$؛ \omterm{prop:g10:coordgeom:midpoint}{ومنتصف}
$[BD]$: $\left(\frac{1+1}{2}, \frac{3+(-1)}{2}\right) = (1, 1)$. \omterm{prop:g10:coordgeom:midpoint}{والمنتصف}
نفسه: إذن $ABCD$ متوازي أضلاع.

\emph{2.} $AB^2 = 3^2 + 2^2 = 13$ و $BC^2 = 3^2 + (-4)^2 = 25$:
إذن $AB \neq BC$، وليس معيّنًا.

\emph{3.} $AC^2 = 6^2 + 0^2 = 36$ و $BD^2 = 0^2 + (-4)^2 = 16$:
إذن $AC \neq BD$، وليس مستطيلًا. إذن $ABCD$ متوازي أضلاع عادي.
\end{solution}

\begin{solution}{exo:g10:coordgeom:6}
احسب مربعات المسافات إلى $\Omega(2,1)$ وقارنها بالمقدار $5^2 = 25$:
\[
\Omega A^2 = 3^2 + 4^2 = 25, \qquad
\Omega B^2 = (-3)^2 + (-4)^2 = 25, \qquad
\Omega C^2 = 4^2 + 1^2 = 17 .
\]
النقطتان $A$ و $B$ على الدائرة؛ و $C$ داخلها ($17 < 25$).
\end{solution}

\begin{solution}{exo:g10:coordgeom:7}
$MA^2 = (x - 1)^2 + 1$ و $MB^2 = (x - 7)^2 + 25$. وبوضع
$MA^2 = MB^2$:
\[
x^2 - 2x + 2 = x^2 - 14x + 74
\ \Longleftrightarrow\
12x = 72
\ \Longleftrightarrow\
x = 6 .
\]
فالنقطة الوحيدة هي $M(6, 0)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:8}
\emph{1.} $I = \left(\frac{0+4}{2}, \frac{3+1}{2}\right) = (2, 2)$.

\emph{2.} النقطة $M(2,2)$ هي فعلًا $I$. و
$MA^2 = (0-2)^2 + (3-2)^2 = 5$ و $MB^2 = (4-2)^2 + (1-2)^2 = 5$: \omterm{prop:g10:coordgeom:midpoint}{فالمنتصف}
متساوي البعد عن الطرفين، كما هو متوقع.

\emph{3.} نبحث عن $N(0, y)$ تحقق $NA^2 = NB^2$:
\[
(3 - y)^2 = 16 + (1 - y)^2
\ \Longleftrightarrow\
9 - 6y = 16 + 1 - 2y
\ \Longleftrightarrow\
-4y = 8,
\]
إذن $y = -2$: أي النقطة $N(0, -2)$.
\end{solution}

\begin{solution}{exo:g10:coordgeom:9}
\begin{align*}
AB &= \sqrt{4^2 + 2^2} = \sqrt{20} = 2\sqrt5, \\
BC &= \sqrt{8^2 + 4^2} = \sqrt{80} = 4\sqrt5, \\
AC &= \sqrt{12^2 + 6^2} = \sqrt{180} = 6\sqrt5 .
\end{align*}
عندئذٍ $AB + BC = 2\sqrt5 + 4\sqrt5 = 6\sqrt5 = AC$: فمتراجحة
المثلث صارت مساواة، إذن تقع $B$ على القطعة $[AC]$ --- أي أن
النقط الثلاث على استقامة واحدة.
\end{solution}

\begin{solution}{exo:g10:coordgeom:10}
\omterm{prop:g10:coordgeom:midpoint}{منتصف} $[AB]$ هو $I\left(\frac a2, \frac b2\right)$. عندئذٍ
\[
OI^2 = \frac{a^2}{4} + \frac{b^2}{4},
\qquad
IA^2 = \left(a - \frac a2\right)^2 + \left(0 - \frac b2\right)^2
= \frac{a^2}{4} + \frac{b^2}{4},
\]
و $IB^2 = \left(0 - \frac a2\right)^2 + \left(b - \frac b2\right)^2 =
\frac{a^2}{4} + \frac{b^2}{4}$ كذلك. فمربعات المسافات الثلاثة
متساوية، إذن $OI = IA = IB$: أي أن \omterm{prop:g10:coordgeom:midpoint}{منتصف} وتر المثلث القائم
$OAB$ متساوي البعد عن الرؤوس الثلاثة (وهو مركز
دائرته المحيطة).
\end{solution}

\begin{solution}{pb:g10:coordgeom:1}
\textbf{1.} $\Omega M = 5 \iff \Omega M^2 = 25 \iff
(x - 2)^2 + (y - 1)^2 = 25$ حسب صيغة المسافة
(\cref{thm:g10:coordgeom:distance}) --- والتربيع غير مؤذٍ،
لأن الطرفين موجبان.

\textbf{2.} $A$: $(5-2)^2 + (5-1)^2 = 9 + 16 = 25$: على
الدائرة. $B$: $16 + 9 = 25$: عليها أيضًا. $D$: $4 + 9 = 13 < 25$:
داخلها.

\textbf{3.} $x^2 - 6x = (x - 3)^2 - 9$ و
$y^2 + 4y = (y + 2)^2 - 4$، إذن تُكتب \omterm{def:g10:algebra:equation}{المعادلة}
$(x - 3)^2 + (y + 2)^2 = 25$: المركز $(3, -2)$ ونصف القطر $5$.

\textbf{4.} $(x - 3)^2 + (y + 2)^2 = -1$: ومجموع مربعين
ليس سالبًا أبدًا --- فلا نقطة تحققها: المجموعة خالية. وبوجه
عام، تصف $(x - h)^2 + (y - k)^2 = c$ دائرة نصف قطرها
$\sqrt c$ إذا كان $c > 0$، والنقطة الوحيدة $(h, k)$ إذا كان $c = 0$،
وتصف المجموعة الخالية إذا كان $c < 0$.

\textbf{5.} من أجل $y = 3$: $(x - 3)^2 + 25 = 25$، إذن
$(x - 3)^2 = 0$: أي النقطة الوحيدة $(3, 3)$. نقطة تماس واحدة:
فالمستقيم \emph{مماس} للدائرة.

\textbf{6.} $MA^2 + MB^2 = (x + r)^2 + y^2 + (x - r)^2 + y^2
= 2x^2 + 2y^2 + 2r^2 = 2r^2 + 2r^2 = 4r^2$ (باستعمال
$x^2 + y^2 = r^2$). وبما أن $AB^2 = (2r)^2 = 4r^2$، فإن المثلث
$AMB$ يحقق $MA^2 + MB^2 = AB^2$: أي أنه قائم الزاوية في $M$
(عكس مبرهنة فيثاغورس) --- ثلاثة أسطر، كما وُعد.

\textbf{7.} $MA^2 + MB^2 = (x + a)^2 + y^2 + (x - a)^2 + y^2
= 2x^2 + 2y^2 + 2a^2 = 2\,MI^2 + 2a^2$، و
$\frac{AB^2}{2} = \frac{(2a)^2}{2} = 2a^2$. إذن المتطابقة مبرهنة.

\textbf{8.} $2\,MI^2 + \frac{36}{2} = 26$ يعطي $MI^2 = 4$: أي
الدائرة التي مركزها \omterm{prop:g10:coordgeom:midpoint}{المنتصف} $I$ ونصف قطرها $2$.

\textbf{9.} $MA^2 = 4\,MB^2$:
$x^2 + y^2 = 4\left((x - 3)^2 + y^2\right)$، أي
$3x^2 + 3y^2 - 24x + 36 = 0$، أو
$x^2 + y^2 - 8x + 12 = 0$: وبإتمام المربع،
$(x - 4)^2 + y^2 = 4$ --- أي الدائرة التي مركزها $(4, 0)$ و
نصف قطرها $2$ (\emph{دائرة أبولونيوس} ذات النسبة $2$).

\textbf{10.} يحوّل الجسر الإنشاءات إلى
حسابات: فلا مستطيل مساعد ولا مناقشة حالات --- \omterm{def:g10:algebra:expand}{نشر}
واحد وتسقط المبرهنة. والثمن: أن الحسابات
تحمل بصيرة هندسية أقل؛ فالبرهان الكلاسيكي يبيّن
\emph{لماذا}، والبرهان الجبري يبيّن \emph{أن} --- والرياضي
المحترف يحتفظ بكليهما.

\textbf{11.} $x^2 + y^2 = 25$ و $(x - 6)^2 + y^2 = 25$.
وبالطرح: $x^2 - (x - 6)^2 = 0$، أي $12x - 36 = 0$:
فيكون $x = 3$. ثم $9 + y^2 = 25$: فيكون $y = \pm 4$. والمرشحان
$(3, 4)$ و $(3, -4)$.

\textbf{12.} من $(3, 4)$ إلى $R(0, 8)$:
$\sqrt{9 + 16} = 5$ --- مطابق. ومن $(3, -4)$:
$\sqrt{9 + 144} \approx 12.4$ --- خطأ. إذن أنت في $(3, 4)$.

\textbf{13.} تحمل المعادلتان المقدار $x^2 + y^2$ نفسه (أو بمعاملات
متساوية بعد \omterm{def:g10:algebra:expand}{النشر}): فالطرح يلغي
المربعات ويترك \omterm{def:g10:algebra:equation}{معادلة} من الدرجة الأولى --- أي مستقيمًا. وعندما
تلتقي الدائرتان مرتين، يمر ذلك المستقيم بنقطتي
اللقاء: إنه مستقيم الوتر المشترك.

\textbf{14.} أربعة مجاهيل: \omterm{def:g10:coordgeom:system}{الإحداثيات} الثلاثة زائد
خطأ ساعة المستقبِل (وهي ساعة رخيصة تصححها
الرياضيات). ويعطي كل قمر \omterm{def:g10:algebra:equation}{معادلة} واحدة، فتعطي أربعة
أقمار أربع معادلات لأربعة مجاهيل --- وتحسّن الأقمار
الإضافية الدقة.

\textbf{15.} يعطي الطرح الآن
$x = \frac{5.1^2 - 5^2 + 36}{12} = \frac{37.01}{12} \approx
3.084$: أي أن خطأ $10$ سم في مسافة واحدة نقل التقدير
نحو $8$ سم هنا. فالأخطاء تنتشر عبر كل صيغة ---
إنه تحذير حساب الفترات في \cref{pb:g10:numbers:1}،
\omterm{def:g10:coordgeom:system}{بالإحداثيات} هذه المرة.

\textbf{16.} النقطة $B' (7, -3)$. \omterm{def:g10:reffunc:affine}{وميل} المستقيم $(AB')$ هو
$\frac{-3 - 5}{7 - 1} = -\frac43$ ويقطع $y = 0$ في
$x = 1 + \frac{5}{8} \times 6 = 4.75$: فاشرب عند
$P(4.75,\ 0)$. والطول الأصغري: $AP + PB = AP + PB' = AB' =
\sqrt{6^2 + 8^2} = 10$.

\textbf{17.} عبر $Q(3, 0)$:
$AQ + QB = \sqrt{4 + 25} + \sqrt{16 + 9} = \sqrt{29} + 5
\approx 10.39 > 10$. إذن يفوز المستقيم المنعكس.

\textbf{18.} منتصفا القطرين: $[AC]$: $(3, 3)$؛ و $[BD]$:
$(3, 3)$ --- متساويان: فهو متوازي أضلاع. والأضلاع:
$AB = \sqrt{10} = AD$: فهو معيّن. والقطران:
$AC = \sqrt{32} \neq BD = \sqrt{8}$: فليس مستطيلًا، وبالتالي
ليس مربعًا. الحكم: معيّن، ولا شيء أكثر.

\textbf{19.} $D = A + C - B = (-1 + 6 - 3,\ 2 + 0 - 4) =
(2, -2)$: عندئذٍ يتقاسم $[AC]$ و $[BD]$ \omterm{prop:g10:coordgeom:midpoint}{المنتصف}
$\left(\frac52, 1\right)$.

\textbf{20.} صيغة \omterm{prop:g10:coordgeom:midpoint}{المنتصف}: متوازيات الأضلاع والمراكز
(السؤالان 18 و 19). صيغة المسافة: الدوائر، والمحال الهندسية، وطبائع
المثلثات، وأقصر الطرق (الأسئلة 1--9 و 16). وطرح
معادلتي دائرتين: المستقيم الذي يحدد الموضع (الأسئلة
11--15). وما ينقص العدة: جبر نظيف
\emph{للاتجاهات} --- المتجهات --- و\emph{للميول} ---
معادلات المستقيمات والجمل: والفصلان التاليان يقدمان
ذلك بالضبط.
\end{solution}
