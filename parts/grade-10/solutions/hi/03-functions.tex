\end{solution}

\begin{solution}{exo:g10:functions:10}
\emph{1.} $f(v) - f(u) = v^2 - u^2 = (v-u)(v+u)$। यदि $0 \leq u < v$ हो, तो $v - u > 0$ और $v + u > 0$ (एक अऋणात्मक और एक धनात्मक
संख्या का योग), इसलिए $f(v) - f(u) > 0$: $f$ $\intco{0}{+\infty}$ पर निरंतर \omterm{def:g10:functions:variations}{वर्धमान} है।

\emph{2.} यदि $u < v \leq 0$ हो, तो $v - u > 0$ तब भी, पर अब $v + u < 0$ (एक अधनात्मक और एक
ऋणात्मक संख्या का योग), इसलिए $f(v) - f(u) < 0$: $f$ $\intoc{-\infty}{0}$ पर निरंतर \omterm{def:g10:functions:variations}{ह्रासमान} है।
\end{solution}

\begin{solution}{pb:g10:functions:1}
\textbf{1.} \omterm{def:g10:functions:function}{प्रांत}: $x \neq 0$ ($x$ से भाग)। $f(1) =
\frac12(1 + 2) = \frac32$; $f(2) = \frac12(2 + 1) = \frac32$; $f\!\left(\frac32\right) = \frac12\left(\frac32 +
\frac43\right) = \frac{17}{12}$।

\textbf{2.} $\frac12\left(x + \frac2x\right) = \frac32$ से $x^2 + 2 = 3x$ मिलता है, अर्थात् $x^2 - 3x + 2 = (x - 1)(x - 2) = 0$: $\frac32$ के
\omterm{def:g10:functions:function}{पूर्वप्रतिबिंब} $1$ और $2$ हैं --- जैसा प्रश्न 1 ने बताया था।

\textbf{3.} $f(x) = x$ से $x + \frac2x = 2x$ मिलता है, इसलिए $\frac2x = x$, $x^2 = 2$: \omterm{pb:g10:functions:1}{स्थिर बिंदु} $\sqrt2$
और $-\sqrt2$। धनात्मक वाला इकाई वर्ग का विकर्ण है, जो माध्यमिक विद्यालय खंड की
अपरिमेयता वाली सप्ताहांत समस्या से \omterm{ex:g10:numbers:classify}{अपरिमेय} है।

\textbf{4.} $1 \to \frac32 \to \frac{17}{12} \to
\frac{577}{408}$: $\sqrt2$ के हीरोन-सन्निकटन। उनका नुस्ख़ा ``अनुमान का
$2/$अनुमान के साथ औसत लीजिए'' ठीक \emph{\omterm{def:g10:functions:function}{फलन} $f$ की पुनरावृत्ति} है ---
और यह प्रक्रिया जिस संख्या का पीछा करती है वह $f$ का \omterm{pb:g10:functions:1}{स्थिर बिंदु} है।

\textbf{5.} \omterm{pb:g10:functions:1}{स्थिर बिंदु} \omterm{def:g10:functions:graph}{आलेख} के विकर्ण $y = x$ के साथ प्रतिच्छेद हैं: निर्गम
को वापस भीतर डालते जाना बिंदु को \omterm{def:g10:functions:graph}{आलेख} के साथ-साथ उसी प्रतिच्छेद की ओर चलाता
है। माध्यमिक विद्यालय खंड की दो-थर्मामीटर वाली सप्ताहांत समस्या में पठन
$-40^\circ$ रूपांतरण-फलन के लिए वही विकर्ण-काटने वाला विचार था।

\textbf{6.} $f(v) - f(u) = \frac{v - u}{2} + \frac1v - \frac1u
= \frac{v - u}{2} + \frac{u - v}{uv}
= (v - u)\left(\frac12 - \frac{1}{uv}\right)
= \frac{(v - u)(uv - 2)}{2uv}$। $0 < u < v \leq \sqrt2$ के लिए: $uv < 2$, कोष्ठक ऋणात्मक है, $f(v) < f(u)$: \omterm{def:g10:functions:variations}{ह्रासमान}।
$\sqrt2 \leq u < v$ के लिए: $uv > 2$: \omterm{def:g10:functions:variations}{वर्धमान}। मोड़ का बिंदु: $\sqrt2$।

\textbf{7.} $\sqrt2$ से पहले \omterm{def:g10:functions:variations}{ह्रासमान}, उसके बाद \omterm{def:g10:functions:variations}{वर्धमान}: $f$ अपना \omterm{def:g10:functions:extrema}{न्यूनतम}
$\sqrt2$ पर पाता है, जिसका मान $f(\sqrt2) = \frac12\left(\sqrt2 + \frac{2}{\sqrt2}\right)
= \sqrt2$ है। परिणाम: मशीन का हर निर्गम (धनात्मक
निवेश पर) कम से कम $\sqrt2$ है --- हैंडल एक बार घुमाते ही अनुमान $\intco{\sqrt2}{+\infty}$ में आ
बसते हैं।

\textbf{8.} यदि $x > \sqrt2$ हो, तो $x^2 > 2$, इसलिए $\frac2x < x$, और $x$ तथा $\frac2x$ का औसत
$f(x)$ $x$ से कम है; और प्रश्न 7 से $f(x) \geq \sqrt2$, जिसमें समता केवल $x = \sqrt2$ पर है।
इसलिए $\sqrt2 < f(x) < x$: हर पुनरावृत्ति कड़ाई से सुधार करती है और नीचे नहीं गिरती ---
प्रश्न 4 का अनुक्रम फिसलकर $\sqrt2$ पर आ बैठता है।

\textbf{9.} $\intoo{0}{+\infty}$ पर: $+\infty$ से (जो $0$ के पास है) घटकर $x = \sqrt2$ पर \omterm{def:g10:functions:extrema}{न्यूनतम}
$\sqrt2$ तक, फिर बिना किसी सीमा के बढ़ता हुआ।

\textbf{10.} $g(x) = x$ से $x^2 = x$, $x(x - 1) = 0$ मिलता है: \omterm{pb:g10:functions:1}{स्थिर बिंदु} $0$ और $1$।
$0.9$ से: $0.81$, $0.656$, $0.430$, $0.185$ --- $0$ की ओर फिसलते हुए। $1.1$ से:
$1.21$, $1.464$, $2.144$, $4.595$ --- अनंत की ओर भागते हुए। \omterm{pb:g10:functions:1}{स्थिर बिंदु} $0$
खींचता है, $1$ धकेलता है: आरंभ में बाल भर का अंतर, और नियति उलटी।

\textbf{11.} $7 \to 22 \to 11 \to 34 \to 17 \to 52 \to 26 \to
13 \to 40 \to 20 \to 10 \to 5 \to 16 \to 8 \to 4 \to 2 \to 1$: सोलह चरण, चोटी $52$ पर।

\textbf{12.} $27 \to 82 \to 41 \to 124 \to 62 \to 31 \to 94 \to
47 \to 142 \to 71 \to 214 \to \dots$ --- दस चरणों के बाद भी चढ़ता ही जा रहा है, $9\,232$ की चोटी
और $111$-चरणों की उड़ान की ओर। सीख: दो पंक्तियों का नियम ऐसा व्यवहार पैदा
कर सकता है जिसकी भविष्यवाणी एक नज़र में कोई नहीं कर सकता --- पड़ोसी संख्याओं
($26$, $27$, $28$) की उड़ानें बेतहाशा अलग हैं।

\textbf{13.} $2$ की कोई घात अपनी पूरी सीढ़ी आधी-आधी करके उतर आती है:
$2^k \to 2^{k-1} \to \dots \to 2 \to 1$। तल पर $h(1) = 4$, $h(4) = 2$, $h(2) = 1$: उड़ान चक्र $4 \to 2 \to 1 \to 4$ में प्रवेश कर जाती है।
\omterm{pb:g10:functions:1}{स्थिर बिंदु} के लिए $\frac n2 = n$ चाहिए (केवल $n = 0$) या $3n + 1 = n$ (ऋणात्मक): धनात्मक
\omterm{def:g10:numbers:sets}{पूर्णांकों} में कोई नहीं --- देखी गई उड़ानें किसी \omterm{pb:g10:functions:1}{स्थिर बिंदु} पर नहीं, उसी
छोटे \emph{चक्र} में समाप्त होती हैं।

\textbf{14.} प्रतिउदाहरण या तो ऐसी उड़ान होगी जो सदा बढ़ती जाए ($1$ पर
कभी न लौटे) या $4 \to 2 \to 1$ से असंयुक्त कोई दूसरा चक्र। $10^{20}$ तक की जाँच अनगिनत
संख्याएँ बिना जाँचे छोड़ देती है --- ठीक माध्यमिक विद्यालय खंड की
माचिस-भविष्यवक्ता वाली सप्ताहांत समस्या का वृत्त-क्षेत्रों वाला पाठ: सीमित
स्थितियों में मेल खा जाना सबके बारे में कुछ भी सिद्ध नहीं करता।

\textbf{15.} हीरोन की मशीन में हमारे पास \emph{संरचना} थी --- एक गुणनखंडित
अंतर (प्रश्न 6) जो परिवर्तन उजागर करता है, एक \omterm{def:g10:functions:extrema}{न्यूनतम}, एक शिकंजा --- जबकि
ओला-मशीन का सम-विषम बदलता नियम ऐसा कोई हत्था नहीं देता: उसे साधने वाली
संरचना अब तक किसी को नहीं मिली।

\textbf{16.} $\sqrt{x - 3}$: $x - 3 \geq 0$ चाहिए: $\intco{3}{+\infty}$। $\frac{1}{x^2 - 9}$: $x \neq \pm 3$ चाहिए: $\R$ में से
$-3$ और $3$ हटाकर। $\sqrt{9 - x^2}$: $x^2 \leq 9$ चाहिए: $\intcc{-3}{3}$।

\textbf{17.} $f(x) = 3$ से $x + \frac2x = 6$ मिलता है, इसलिए $x^2 - 6x + 2 = 0$, अर्थात् $(x - 3)^2 = 7$: $x = 3 - \sqrt7$ या
$x = 3 + \sqrt7$ (दोनों धनात्मक: दो यथार्थ \omterm{def:g10:functions:function}{पूर्वप्रतिबिंब})।

\textbf{18.} जब तक पत्थर हवा में है तभी तक भौतिक अर्थ बनता है: $t \in \intcc{0}{4}$ के लिए
$H(t) = 5t(4 - t) \geq 0$। पूरा करने पर: $H(t) = 20 - 5(t - 2)^2$: \omterm{def:g10:functions:extrema}{अधिकतम} $20$ m, $t = 2$ s पर। परिवर्तन: $\intcc{0}{2}$ पर
$0$ से $20$ तक \omterm{def:g10:functions:variations}{वर्धमान}, $\intcc{2}{4}$ पर घटकर वापस $0$ तक।

\textbf{19.} $20t - 5t^2 \geq 15 \iff t^2 - 4t + 3 \leq 0 \iff
(t - 1)(t - 3) \leq 0 \iff t \in \intcc{1}{3}$: पूरे दो सेकंड $15$ m से ऊपर।

\textbf{20.} हर मशीन के पास एक \omterm{def:g10:functions:function}{प्रांत} होता है (जहाँ नियम लागू होता है), एक
नियम (सूत्र या सम-विषम स्विच), और एक \omterm{def:g10:functions:graph}{आलेख} या सारणी जो उसे पूरा दिखा देती है।
हमने हर एक से वही दो बड़े प्रश्न पूछे: \emph{दोहराव कहाँ ले जाता है} (\omterm{pb:g10:functions:1}{स्थिर
बिंदु}, चक्र, आकर्षण --- हीरोन के लिए सुलझा, ओला-मशीन के लिए खुला) और
\emph{निवेश के साथ निर्गम कैसे चलता है} (परिवर्तन, चरम मान --- पत्थर की
चोटी, हीरोन का \omterm{def:g10:functions:extrema}{न्यूनतम})। आगे के अध्याय दोनों को और पैना करते हैं: पहले आधार
\omterm{def:g10:functions:function}{फलन}, और फिर आगे के वर्षों में परिवर्तन नापने के लिए अवकलज तथा यह प्रमाणित
करने के लिए सीमाएँ कि पुनरावृत्तियाँ कहाँ जा बैठती हैं।
\end{solution}
