\begin{solution}{exo:g10:numbers:2}
(a) $\intco{-2}{5}$: $-2$ पर भरा बिंदु, $5$ पर खोखला बिंदु।
(b) $\intoo{3}{+\infty}$: $3$ पर खोखला बिंदु, और दाईं ओर छायांकन।
(c) $\intoc{-\infty}{-1}$: बाईं ओर से $-1$ पर भरे बिंदु तक छायांकन।
(d) ``$x$ की $2$ से दूरी अधिक से अधिक $3$'' का अर्थ $\abs{x - 2} \leq 3$ है, अर्थात्
$x \in \intcc{-1}{5}$: $-1$ और $5$ पर भरे बिंदु।
\end{solution}

\begin{solution}{exo:g10:numbers:3}
(a) दोनों \omterm{def:g10:numbers:interval}{अंतराल} $0$ और $2$ के बीच एक-दूसरे को ढकते हैं: $I \cap J = \intcc{0}{2}$ ($0 \in J$ और
$2 \in I$, और दोनों दूसरे समुच्चय के भी अवयव हैं), और $I \cup J = \intco{-3}{4}$।

(b) $I \cap J$ उन $x$ का समुच्चय है जिनके लिए $-2 \leq x$ और $x < 1$ हों: $\intco{-2}{1}$। \omterm{def:g10:numbers:interunion}{सम्मिलन}
सब कुछ ढक लेता है: $I \cup J = \R$।
\end{solution}

\begin{solution}{exo:g10:numbers:5}
$\abs{x} = 5$: $0$ से दूरी $5$, इसलिए $x = 5$ या $x = -5$।

$\abs{x - 1} = 3$: $1$ से दूरी $3$, इसलिए $x = 4$ या $x = -2$।

$\abs{x - 4} \leq 1$: $4$ से दूरी अधिक से अधिक $1$, इसलिए $x \in \intcc{3}{5}$।

$\abs{x + 2} < 3$: $-2$ से दूरी $3$ से कड़ाई से कम, इसलिए $x \in \intoo{-5}{1}$।
\end{solution}

\begin{solution}{exo:g10:numbers:7}
मान लीजिए $x = 0.272727\dots$ तब $100x = 27.2727\dots$, और घटाने पर:

100x - x = 27.2727\dots - 0.2727\dots = 27,

\end{solution}

\begin{solution}{exo:g10:numbers:8}
\emph{1. सत्य:} \omterm{def:g10:numbers:sets}{पूर्णांकों} (धनात्मक या ऋणात्मक पूर्ण संख्याओं) को जोड़ने पर
सदा \omterm{def:g10:numbers:sets}{पूर्णांक} ही मिलता है।

\emph{2. असत्य:} $\frac{1}{2}$ \omterm{def:g10:numbers:sets}{पूर्णांकों} $1$ और $2$ का भागफल है और \omterm{def:g10:numbers:sets}{पूर्णांक} नहीं
है।

\emph{3. सत्य:} $\frac pq + \frac{p'}{q'} = \frac{pq' + p'q}{qq'}$ फिर से \omterm{def:g10:numbers:sets}{पूर्णांकों} का भागफल है (जिसका हर शून्येतर है)।

\emph{4. सत्य:} मान लीजिए $r$ परिमेय है, $t$ \omterm{ex:g10:numbers:classify}{अपरिमेय}, और $r + t = s$ परिमेय
होता। तब $t = s - r$ दो \omterm{def:g10:numbers:sets}{परिमेय संख्याओं} का अंतर होता, अतः परिमेय (3 से, $-r$ के
साथ लगाकर) --- विरोधाभास। इसलिए $r + t$ \omterm{ex:g10:numbers:classify}{अपरिमेय} है।
\end{solution}

\begin{solution}{exo:g10:numbers:9}
$1.414 \leq \sqrt2 \leq 1.415$ को $2 > 0$ से गुणा करने पर: $2.828 \leq 2\sqrt2 \leq 2.830$।

$3$ जोड़ने पर: $4.414 \leq \sqrt2 + 3 \leq 4.415$।

\end{solution}

\begin{solution}{exo:g10:numbers:10}
पहले सहायक तथ्य। $p$ को $3$ से भाग दीजिए: शेषफल $0$, $1$ या $2$
होगा, अर्थात् $p = 3k$, $p = 3k+1$ या $p = 3k+2$। वर्ग करने पर:
\[
(3k)^2 = 3(3k^2), \quad
(3k+1)^2 = 3(3k^2 + 2k) + 1, \quad
(3k+2)^2 = 3(3k^2 + 4k + 1) + 1 .
\]
इनमें केवल पहला $3$ का गुणज है: इसलिए यदि $p^2$ $3$ का गुणज हो, तो $p$
भी है।

अब मान लीजिए $\sqrt3 = \frac pq$ पूरी तरह सरल की जा चुकी है। वर्ग करने पर $p^2 = 3q^2$ मिलता है,
इसलिए $p^2$ $3$ का गुणज है, इसलिए $p = 3k$। तब $9k^2 = 3q^2$, इसलिए $q^2 = 3k^2$ और $q$
भी $3$ का गुणज है --- पर तब भिन्न $\frac pq$ पूरी तरह सरल नहीं थी: विरोधाभास।
इसलिए $\sqrt3$ \omterm{ex:g10:numbers:classify}{अपरिमेय} है।
\end{solution}

\begin{solution}{pb:g10:numbers:1}
\textbf{1.} $-7 \in \Z$; \ $\frac{13}{4} \in \Q$; \
$\sqrt{16} = 4 \in \N$; \ $0.121212\ldots = \frac{12}{99} =
\frac{4}{33} \in \Q$ (माध्यमिक विद्यालय खंड की आवर्ती दशमलव वाली सप्ताहांत
समस्या); \ $\sqrt8 = 2\sqrt2$ \omterm{ex:g10:numbers:classify}{अपरिमेय} है: सबसे छोटा समुच्चय $\R$; \ $\pi$: $\R$।

\textbf{2.} $\frac pq + \frac rs = \frac{ps + qr}{qs}$ और $\frac pq \times \frac rs = \frac{pr}{qs}$: शून्येतर हर वाले \omterm{def:g10:numbers:sets}{पूर्णांकों} के भागफल --- अतः
परिमेय।

\textbf{3.} (a) यदि $r$ परिमेय हो, $x$ \omterm{ex:g10:numbers:classify}{अपरिमेय}, और $r + x = q$ परिमेय होता,
तो $x = q - r$ \omterm{def:g10:numbers:sets}{परिमेय संख्याओं} का अंतर होता, अतः परिमेय (प्रश्न 2): विरोधाभास।
(b) यदि $r \neq 0$ हो और $r x = q$ परिमेय, तो $x = \frac qr$ परिमेय है: विरोधाभास।

\textbf{4.} $\sqrt2$ और $-\sqrt2$ \omterm{ex:g10:numbers:classify}{अपरिमेय} हैं और उनका योग $0$ है; $\sqrt2$ और $\sqrt2$
का गुणनफल $2$ है। \omterm{ex:g10:numbers:classify}{अपरिमेय} सामग्री से परिमेय परिणाम: \omterm{ex:g10:numbers:classify}{अपरिमेय संख्याएँ} कोई
बंद राज्य नहीं बनातीं।

\textbf{5.} $\sqrt8 = 2\sqrt2$, इसलिए $\sqrt2 + \sqrt8 = 3\sqrt2$: \omterm{ex:g10:numbers:classify}{अपरिमेय} (प्रश्न 3b, गुणक $3$)। और $\sqrt2 \times \sqrt8 = \sqrt{16} = 4 \in \N$:
दो \omterm{ex:g10:numbers:classify}{अपरिमेय} संख्याओं का गुणनफल सबसे छोटे राज्य में जा गिरता है।

\textbf{6.} $3.475$, फिर $3.471$ (या $3.4701$, $3.47001$, \dots): दशमलव अंक जोड़ते
जाने से दोनों के बीच नई-नई \omterm{def:g10:numbers:sets}{परिमेय संख्याएँ} अनंत काल तक बनती रहती हैं।

\textbf{7.} $10^{-n}$ के गुणज रेखा पर $10^{-n} < g$ की डग भरते चलते हैं। $a$ से कड़ाई
से बड़ा पहला गुणज --- वह मौजूद है, क्योंकि गुणज अंततः $a$ से आगे निकल
जाते हैं --- $a$ से अधिक से अधिक एक डग आगे है, अतः $a + g = b$ से पहले: वह $a$
और $b$ के ठीक बीच है। जाली-बिंदु दशमलव है, अतः परिमेय: धनात्मक लंबाई का
हर \omterm{def:g10:numbers:interval}{अंतराल} एक ऐसी संख्या रखता है।

\textbf{8.} $a$ और प्रश्न 7 में मिली \omterm{def:g10:numbers:sets}{परिमेय संख्या} $r_1$ के बीच (फिर
प्रश्न 7 से) एक परिमेय $r_2$ है; $a$ और $r_2$ के बीच एक परिमेय $r_3$; और
इसी तरह आगे: अनंत, सब अलग-अलग, और सब मूल अंतराल में।

\textbf{9.} $r + \frac{\sqrt2}{10^k}$ एक परिमेय और एक \omterm{ex:g10:numbers:classify}{अपरिमेय संख्या} का योग है ($\frac{\sqrt2}{10^k}$ प्रश्न 3b
से \omterm{ex:g10:numbers:classify}{अपरिमेय} है), अतः \omterm{ex:g10:numbers:classify}{अपरिमेय}। चूँकि $\frac{\sqrt2}{10^k} < \frac{2}{10^k}$ किसी भी सीमा से नीचे सिकुड़ जाता
है, इसलिए पर्याप्त बड़े $k$ के लिए $r + \frac{\sqrt2}{10^k}$ अब भी $b$ से पहले पड़ता है:
\omterm{def:g10:numbers:interval}{अंतराल} के भीतर एक \omterm{ex:g10:numbers:classify}{अपरिमेय संख्या} --- और $k$ बदलते जाने से अनंत।

\textbf{10.} कोई सबसे छोटी धनात्मक \omterm{def:g10:numbers:sets}{वास्तविक संख्या} नहीं: यदि $s > 0$ वह होती,
तो \omterm{def:g10:numbers:interval}{अंतराल} $(0, s)$ में भी एक \omterm{def:g10:numbers:sets}{परिमेय संख्या} होती (प्रश्न 7), जो धनात्मक और
$s$ से छोटी होती। $3$ के ठीक बाद कोई संख्या नहीं: कोई भी दावेदार $c > 3$
\omterm{def:g10:numbers:interval}{अंतराल} $(3, c)$ छोड़ देता है, जो रिक्त नहीं है --- माध्यमिक विद्यालय खंड का
``कोई अगली संख्या नहीं'' वाला खेल, अब $\R$ के बारे में एक प्रमेय।

\textbf{11.} $\abs{x - 5} = 2$: $x = 3$ या $x = 7$।
$\abs{x - 5} < 2$: $x \in \intoo{3}{7}$।
$\abs{x + 1} \geq 3$: $-1$ से दूरी कम से कम $3$: $x \in \intoc{-\infty}{-4} \cup \intco{2}{+\infty}$।

\textbf{12.} $\abs{d - 80} \leq 0.05$, अर्थात् $d \in \intcc{79.95}{80.05}$। $79.97$ वाला पिस्टन पास हो जाता है; $80.06$
वाला मिलीमीटर के सौवें हिस्से से चूक जाता है।

\textbf{13.} $(3, -5)$: $\abs{-2} = 2 \leq 8$। $(-2, -7)$: $\abs{-9} = 9 = 2 + 7$: समता। समान चिह्न: $\abs{a + b}$ दूरियों का
योग है, जो $\abs a + \abs b$ के बराबर है। विपरीत चिह्न: चाल पीछे मुड़ जाती है, $\abs{a + b}$
दूरियों का \emph{अंतर} है, जो उनके योग से कड़ाई से कम है (जब तक उनमें से एक
$0$ न हो)। समता ठीक तब जब $a$ और $b$ के चिह्न समान हों या उनमें से एक
शून्य हो।

\textbf{14.} हल वे बिंदु हैं जो $2$ और $-4$ से समान दूरी पर हैं:
मध्यबिंदु, $x = -1$। यही माध्यमिक विद्यालय खंड की दो-दर्पण वाली सप्ताहांत
समस्या का लंब समद्विभाजक है, जो एक विमा में सिमटकर एक अकेला बिंदु रह जाता
है।

\textbf{15.} $\sqrt{x^2} = \abs{x}$, $x$ नहीं: $x = -3$ के लिए $\sqrt{9} = 3 = \abs{-3} \neq -3$ होता है। तब $x^2 < 9$ का अर्थ
$\sqrt{x^2} < 3$ है, अर्थात् $\abs x < 3$: $x \in \intoo{-3}{3}$।

\textbf{16.} माप केवल इतना कहता है कि लंबाई \omterm{def:g10:numbers:interval}{अंतराल} $\intcc{1.41420}{1.41422}$ में है --- और
प्रश्न 7 तथा 9 से \emph{उस \omterm{def:g10:numbers:interval}{अंतराल} में अनंत परिमेय और अनंत \omterm{ex:g10:numbers:classify}{अपरिमेय संख्याएँ}
हैं}। यही बात हर भावी सहनशीलता पर लागू है, चाहे वह कितनी ही छोटी हो। कोई
माप इन दोनों परिवारों को कभी अलग नहीं कर सकता; यह केवल यथार्थ लंबाई के बारे
में कोई उपपत्ति ही कर सकती है --- जैसी इकाई वर्ग के विकर्ण के लिए थी
(माध्यमिक विद्यालय खंड की अपरिमेयता वाली सप्ताहांत समस्या)।

\textbf{17.} वे $\sqrt2$ के पास $10^{-1}, 10^{-2}, 10^{-3}, \dots$ से कम त्रुटि के साथ पहुँचते हैं: हर

सिद्धांत स्नातक खंडों में बनता है।
