\begin{solution}{exo:g10:vectors:4}

(b) $2 \times 10 - (-5) \times (-4) = 20 - 20 = 0$: \omterm{def:g10:vectors:collinear}{संरेखी} ($\vec v = -2\vec u$)।

(c) $3 \times 3 - 1 \times 1 = 8 \neq 0$: \omterm{def:g10:vectors:collinear}{संरेखी} नहीं।
\end{solution}

\begin{solution}{exo:g10:vectors:6}
\emph{1.} $\vect{AB}\,(3, 2)$ और $\vect{AC}\,(6, -2)$, इसलिए $\vect{AB} + \vect{AC} = (9, 0)$ और $M = (-2 + 9,\ 1 + 0) = (7, 1)$।

\emph{2.} समांतर चतुर्भुज नियम से $\vect{AM} = \vect{AB} + \vect{AC}$ का अर्थ है कि $ABMC$ समांतर चतुर्भुज
है ($M$ $A$ के सामने वाला चौथा शीर्ष है)। जाँच लीजिए: $\vect{AB}\,(3,2)$ और $\vect{CM}\,(7-4,\ 1-(-1)) =
(3, 2)$ बराबर
हैं।
\end{solution}

\begin{solution}{exo:g10:vectors:7}
पहला त्रिक: $\vect{AB}\,(3, 2)$, $\vect{AC}\,(9, 6)$; $3 \times 6 - 2 \times 9 = 0$: संरेख ($\vect{AC} = 3\vect{AB}$)।

दूसरा त्रिक: $\vect{AB}\,(2, -1)$, $\vect{AC}\,(8, -4)$; $2 \times (-4) - (-1) \times 8 = -8 + 8 = 0$: यह भी संरेख ($\vect{AC} = 4\vect{AB}$)।
\end{solution}

\begin{solution}{exo:g10:vectors:8}
$\vect{AB}\,(4, 2)$ और $\vect{DC}\,(6-2,\ 1-(-1)) = (4, 2)$: दोनों \omterm{def:g10:vectors:vector}{सदिश} \emph{बराबर} हैं, इसलिए $ABCD$ समांतर चतुर्भुज है,
और विशेष रूप से $(AB) \parallel (CD)$। समांतर चतुर्भुज वाला गुण अधिक प्रबल है: उससे
समांतरता (\omterm{def:g10:vectors:vector}{सदिशों} की संरेखता) निकल आती है, पर उल्टा नहीं।
\end{solution}

\begin{solution}{exo:g10:vectors:9}
\emph{1.} $\vect{AB}\,(6, 3)$, इसलिए $\vect{AI} = \frac23\vect{AB} = (4, 2)$ और $I(4, 2)$।

\emph{2.} $\vect{CB}\,(6-2,\ 3-5) = (4, -2)$, इसलिए $\vect{CJ} = \frac23\vect{CB} = \left(\frac83, -\frac43\right)$ और $J\left(2 + \frac83,\ 5 - \frac43\right) = \left(\frac{14}{3},
\frac{11}{3}\right)$।

\emph{3.} $\vect{IJ}\,\left(\frac{14}{3} - 4,\ \frac{11}{3} - 2\right) =
\left(\frac23, \frac53\right)$ और $\vect{AC}\,(2, 5)$। संरेखता की कसौटी:
\[
\frac23 \times 5 - \frac53 \times 2 = \frac{10}{3} - \frac{10}{3} = 0 :
\]
\omterm{def:g10:vectors:vector}{सदिश} \omterm{def:g10:vectors:collinear}{संरेखी} हैं, इसलिए $(IJ)$ $(AC)$ के समांतर है। ($I$ और $J$ दोनों भुजा से
$B$ की ओर एक-तिहाई दूरी पर बैठे हैं: छोटा त्रिभुज $IBJ$ $ABC$ का लघुरूप है।)
\end{solution}

\begin{solution}{exo:g10:vectors:10}
\omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} ये हैं:
\[
I\left(\tfrac{x_A + x_B}{2}, \tfrac{y_A + y_B}{2}\right), \quad
J\left(\tfrac{x_B + x_C}{2}, \tfrac{y_B + y_C}{2}\right), \quad
K\left(\tfrac{x_C + x_D}{2}, \tfrac{y_C + y_D}{2}\right), \quad
L\left(\tfrac{x_D + x_A}{2}, \tfrac{y_D + y_A}{2}\right).
\]
दोनों \omterm{def:g10:vectors:vector}{सदिशों} के पहले \omterm{def:g10:coordgeom:system}{निर्देशांक}:
\[
x_J - x_I = \frac{x_B + x_C}{2} - \frac{x_A + x_B}{2}
= \frac{x_C - x_A}{2},
\qquad
x_K - x_L = \frac{x_C + x_D}{2} - \frac{x_D + x_A}{2}
= \frac{x_C - x_A}{2},
\]
और वही गणना दूसरे \omterm{def:g10:coordgeom:system}{निर्देशांकों} पर भी चलती है। इसलिए $\vect{IJ} = \vect{LK}$ (दोनों $\frac12\vect{AC}$ के
बराबर), और $IJKL$ समांतर चतुर्भुज है, चाहे चतुर्भुज $ABCD$ कैसा भी दिखता हो।
\end{solution}

\begin{solution}{pb:g10:vectors:1}
\textbf{1.} $\vect{AB} + \vect{BC} + \vect{CD} = \vect{AD}$; \quad $\vect{MA} - \vect{MB} = \vect{MA} + \vect{BM} =
\vect{BA}$; \quad $\vect{AB} + \vect{BC} + \vect{CD} +
\vect{DA} = \vect{AA} = \vec 0$ (तीसरे \omterm{def:g10:vectors:sum}{योग} के पदों का क्रम बदलकर)।

\textbf{2.} शाल से $\vect{OB} - \vect{OA} = \vect{AO} + \vect{OB} =
\vect{AB}$।

\textbf{3.} $I$ $[AB]$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} ठीक तब है जब $\vect{IA} = -\vect{IB}$ हो, अर्थात् $\vect{IA} + \vect{IB} = \vec 0$। तब
किसी भी $O$ के लिए $\vec 0 = \vect{IO} + \vect{OA} + \vect{IO} + \vect{OB}$, इसलिए $2\,\vect{OI} = \vect{OA} + \vect{OB}$ --- और इसके उलट भी।

\textbf{4.} $\vect{AB} = (4, 2)$, और $D$ $\vect{DC} = \vect{AB}$ को संतुष्ट करता है: $D = C - (4, 2) = (2, -2)$ --- वही $D$ जो
\cref{pb:g10:coordgeom:1} में था, पर एक पंक्ति जल्दी।

\textbf{5.} \omterm{def:g10:vectors:vector}{सदिश} $\vec u$ का स्थानांतरण --- ठीक वही रूपांतरण जो दो समांतर
दर्पणों या दो अर्ध-घूर्णनों से बनता है। $(1, 2)$ का \omterm{def:g10:functions:function}{प्रतिबिंब}: $(4, 1)$।

\textbf{6.} हर $\vect{GX}$ को $\vect{GO} + \vect{OX}$ के रूप में खोलने पर: $3\,\vect{GO} + \vect{OA} + \vect{OB} + \vect{OC} = \vec 0$, इसलिए $\vect{OG} = \frac13(\vect{OA} + \vect{OB} + \vect{OC})$: ऐसा
एक और केवल एक बिंदु, जिसके \omterm{def:g10:coordgeom:system}{निर्देशांक} तीनों शीर्षों के \omterm{def:g10:coordgeom:system}{निर्देशांकों} के माध्य
हैं।

\textbf{7.} $\vect{AB} + \vect{AC} =
(\vect{AG} + \vect{GB}) + (\vect{AG} + \vect{GC})
= 2\vect{AG} + (\vect{GB} + \vect{GC})
= 2\vect{AG} - \vect{GA} = 3\vect{AG}$, जिससे $\vect{AG} = \frac13(\vect{AB} + \vect{AC})$। \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} सूत्र (प्रश्न 3, $A$ की ओर से
देखा गया) $\vect{AK} = \frac12(\vect{AB} + \vect{AC})$ देता है। तुलना करने पर: $\vect{AG} = \frac23\vect{AK}$ --- $G$ माध्यिका $[AK]$ पर
है, $A$ से दो-तिहाई दूरी पर।

\textbf{8.} परिभाषित करने वाली शर्त $A$, $B$, $C$ के साथ एक जैसा
व्यवहार करती है, इसलिए $B$ से या $C$ से की गई वही गणना \emph{उसी} $G$
को उन माध्यिकाओं के दो-तिहाई पर बिठा देती है: संगामिता मुफ़्त में। समांतर
चतुर्भुज वाली उपपत्ति चतुर थी पर विशेष अवसर की; \omterm{def:g10:coordgeom:system}{निर्देशांक} गणना तो करते हैं
पर बात धुँधली कर देते हैं; \omterm{def:g10:vectors:vector}{सदिश} सममिति को दिखा देते हैं और उपपत्ति दो
पंक्तियों की कर देते हैं।

\textbf{9.} $G\left(\frac{1+5+3}{3}, \frac{1+2+6}{3}\right) =
(3, 3)$; $K(4, 4)$; $\vect{AG} = (2, 2)$ और $\vect{AK} = (3, 3)$: सारणिक $2 \times 3 - 2 \times 3 = 0$, \omterm{def:g10:vectors:collinear}{संरेखी}, और सचमुच
$\vect{AG} = \frac23\vect{AK}$।

\textbf{10.} $G'\left(\frac{2 + 5 + 3}{4}, \frac{2 + 2 +
6}{4}\right) = (2.5,\ 2.5)$: $G(3,3)$ से खिसककर दुगुने द्रव्यमान वाले शीर्ष $A(1,1)$ की ओर,
जैसा किसी संतुलन-बिंदु को होना ही चाहिए।

\textbf{11.} $3 \times 4 - (-2)(-6) = 12 - 12 = 0$: \omterm{def:g10:vectors:collinear}{संरेखी} ($\vec v = -2\vec u$)। $2 \times 9 - 5 \times 4 = -2 \neq 0$: \omterm{def:g10:vectors:collinear}{संरेखी} नहीं।

\textbf{12.} $\vect{AB} = (2, 3)$, $\vect{AC} = (6, 9)$: सारणिक $2 \times 9 - 3 \times 6 = 0$: संरेख।

\textbf{13.} $\vect{OI} = \frac12(\vect{OA} + \vect{OB})$ और $\vect{OK} = \frac12(\vect{OC} + \vect{OD})$, इसलिए $[IK]$ के \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} का स्थिति \omterm{def:g10:vectors:vector}{सदिश} $\frac12(\vect{OI} + \vect{OK}) = \frac14(\vect{OA} +
\vect{OB} + \vect{OC} + \vect{OD})$
है --- और $[JL]$ के लिए की गई गणना ठीक वही \omterm{def:g10:vectors:vector}{सदिश} देती है। दोनों
द्वि-माध्यिकाएँ (और वारिन्यों से, समांतर चतुर्भुज $IJKL$ के विकर्ण) इसी
केंद्र को साझा करती हैं: वही चतुर्भुज का केन्द्रक है।

\textbf{14.} $\vect{MN} = \vect{AN} - \vect{AM} =
\frac13\vect{AC} - \frac13\vect{AB} = \frac13\vect{BC}$: इसलिए $(MN) \parallel (BC)$ और $MN = \frac13 BC$।

\textbf{15.} $\vect{AM} = k\,\vect{AB}$ और $\vect{AN} = k\,\vect{AC}$ के साथ: $\vect{MN} = k\,\vect{BC}$ --- एक ही पंक्ति जिसमें \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु}
प्रमेय ($k = \frac12$) और थेल्स (कोई भी $k$) दोनों समाए हैं: अनुपात घटाव में से
ज्यों का त्यों निकल आता है।

\textbf{16.} परिणामी $(3, 4)$, परिमाण $\sqrt{9 + 16} = 5$ km/h: नाव वस्तुतः $3$--$4$--$5$
त्रिभुज के कर्ण के साथ-साथ उत्तर-पूर्व की ओर उत्तर झुकी दिशा में चलती है।

\textbf{17.} $\vec u + \vec v + \vec w = (0, 0)$: छल्ला जहाँ है वहीं रहता है --- साम्यावस्था। किसी बंद
बहुभुज $A_1 A_2 \dots A_n A_1$ के अनुदिश शाल \omterm{def:g10:vectors:sum}{योग} को समेटकर $\vect{A_1 A_1} = \vec 0$ कर देता है: चक्कर लगाकर आप
वहीं लौट आते हैं।

\textbf{18.} $\vec w = -(\vec u + \vec v) = -(3, 2) =
(-3, -2)$।

\textbf{19.} परिणामी $(1.5,\ 2)$, परिमाण $\sqrt{2.25 + 4} = 2.5$ m/s। पार करना: $2$ m/s की आरपार
चाल से $50$ m: $25$ s; बहाव: धारा की दिशा में $1.5 \times 25 = 37.5$ m।

\textbf{20.} इस सप्ताहांत जो फिर सिद्ध हुआ: केन्द्रक की दो-तिहाई प्रमेय
(प्रश्न 7), द्वि-माध्यिका/वारिन्यों केंद्र (प्रश्न 13), थेल्स--मध्यबिंदु
(प्रश्न 14--15)। तीर अब भी जो नहीं कर सकते: \emph{कोण} नापना (और इस तरह किसी
भी दिशा में लंबाइयाँ) --- कक्षा 11 का अदिश गुणनफल ठीक वही छूटा हुआ यंत्र है।
\end{solution}
