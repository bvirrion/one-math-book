\section*{अध्याय \ref{ch:g10:stats} --- वर्णनात्मक सांख्यिकी}

\begin{solution}{exo:g10:stats:1}
योग: $0+1+1+2+2+2+3+3+4+4+5+6+7+8+12 = 60$, इसलिए \omterm{def:g10:stats:mean}{माध्य} $\frac{60}{15} = 4$ है। श्रेणी पहले से क्रमबद्ध है और $N = 15$ विषम है:
\omterm{def:g10:stats:median}{माध्यिका} $8$-वाँ मान $3$ है। \omterm{def:g10:stats:quartiles}{परास}: $12 - 0 = 12$।
\end{solution}

\begin{solution}{exo:g10:stats:2}
आपेक्षिक बारंबारताएँ: $\frac{7}{50} = 0.14$, $\frac{9}{50} = 0.18$, $\frac{8}{50} = 0.16$, $\frac{10}{50} = 0.20$, $\frac{6}{50} = 0.12$, $\frac{10}{50} = 0.20$ (योग $1$)। \omterm{def:g10:stats:mean}{माध्य}
परिणाम:
\[
\bar x = \frac{7 \times 1 + 9 \times 2 + 8 \times 3 + 10 \times 4
+ 6 \times 5 + 10 \times 6}{50}
= \frac{7 + 18 + 24 + 40 + 30 + 60}{50}
= \frac{179}{50} = 3.58 .
\]
\end{solution}

\begin{solution}{exo:g10:stats:3}
$N = 12$ (सम): \omterm{def:g10:stats:median}{माध्यिका} $6$-वें और $7$-वें मानों का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} है, और दोनों
$6$ हैं: \omterm{def:g10:stats:median}{माध्यिका} $6$।

$Q_1$: $\frac{12}{4} = 3$, इसलिए $Q_1$ $3$-वाँ मान है: $4$।

$Q_3$: $\frac{3 \times 12}{4} = 9$, इसलिए $Q_3$ $9$-वाँ मान है: $8$।

बक्सा-आरेख: बक्सा $4$ से $8$ तक, बीच में $6$ पर \omterm{def:g10:stats:median}{माध्यिका} की पट्टी, और
मूँछें $1$ से $11$ तक।
\end{solution}

\begin{solution}{exo:g10:stats:4}
$24$ अंकों का योग $24 \times 11.5 = 276$ है। नए अंक के साथ $25$ विद्यार्थियों का योग $276 + 19 = 295$ हो
जाता है: नया \omterm{def:g10:stats:mean}{माध्य} $\frac{295}{25} = 11.8$।
\end{solution}

\begin{solution}{exo:g10:stats:5}
अभी का योग: $5 \times 12 = 60$। $6$ परीक्षाओं पर $13$ का \omterm{def:g10:stats:mean}{माध्य} पाने के लिए योग $78$ चाहिए,
इसलिए $6$-वें अंक $78 - 60 = 18$ होने चाहिए। $14$ के \omterm{def:g10:stats:mean}{माध्य} के लिए $84 - 60 = 24 > 20$ चाहिए: वह
असंभव है।
\end{solution}

\begin{solution}{exo:g10:stats:6}
\omterm{def:g10:stats:mean}{माध्य} $20$ से बड़ा: जैसे $1,\ 2,\ 3,\ 10,\ 40,\ 50,\ 60$ --- \omterm{def:g10:stats:median}{माध्यिका} $10$ ($4$-वाँ क्रमबद्ध मान),
\omterm{def:g10:stats:mean}{माध्य} $\frac{166}{7} \approx 23.7$।

\omterm{def:g10:stats:mean}{माध्य} $6$ से छोटा: जैसे $0,\ 0,\ 0,\ 10,\ 10,\ 10,\ 10$ --- \omterm{def:g10:stats:median}{माध्यिका} $10$, \omterm{def:g10:stats:mean}{माध्य} $\frac{40}{7} \approx 5.7$।

\omterm{def:g10:stats:mean}{माध्य} $\frac{40}{7}$ से नीचे नहीं जा सकता: \omterm{def:g10:stats:median}{माध्यिका} ($4$-वाँ क्रमबद्ध मान) $10$ होने
के लिए स्थान $4$ से $7$ तक के सभी मानों का $\geq 10$ होना आवश्यक है, इसलिए योग
कम से कम $40$ है और \omterm{def:g10:stats:mean}{माध्य} कम से कम $\frac{40}{7}$ --- दूसरा उदाहरण चरम स्थिति है।

ये उदाहरण दिखाते हैं कि \omterm{def:g10:stats:mean}{माध्य} और \omterm{def:g10:stats:median}{माध्यिका} काफ़ी हद तक एक-दूसरे से स्वतंत्र
हैं: एक जान लेने से दूसरे के बारे में कम ही पता चलता है, और इसीलिए अच्छा
सारांश दोनों बताता है।
\end{solution}

\begin{solution}{exo:g10:stats:7}
\emph{1.} \omterm{def:g10:stats:mean}{माध्य}: $\frac{9 \times 1800 + 12000}{10} = \frac{16200 + 12000}{10} = 2820$। \omterm{def:g10:stats:median}{माध्यिका}: क्रम में लगाने पर $10$ वेतन नौ बार $1800$
और फिर $12000$ हैं; \omterm{def:g10:stats:median}{माध्यिका} $5$-वें और $6$-वें मानों का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} है, और
दोनों $1800$ हैं: \omterm{def:g10:stats:median}{माध्यिका} $1800$।

\emph{2.} \omterm{def:g10:stats:median}{माध्यिका} ($1800$) प्रारूपिक वेतन बताती है: $10$ में से $9$
कर्मचारी सचमुच यही कमाते हैं। \omterm{def:g10:stats:mean}{माध्य} ($2820$) को अकेला ऊँचा वेतन खींच ले जाता
है और वह किसी की भी वेतन-पर्ची से मेल नहीं खाता।
\end{solution}

\begin{solution}{exo:g10:stats:8}
कुल लंबाई: $30$ विद्यार्थियों के लिए $12 \times 168 + 18 \times 163 = 2016 + 2934 = 4950$ cm, इसलिए \omterm{def:g10:stats:mean}{माध्य} $\frac{4950}{30} = 165$ cm है। वह
लड़कियों के \omterm{def:g10:stats:mean}{माध्य} के अधिक पास है क्योंकि वे अधिक संख्या में हैं (भारित
\omterm{def:g10:stats:mean}{माध्य}, भार $12$ और $18$)।
\end{solution}

\begin{solution}{exo:g10:stats:9}
\emph{1.} नया \omterm{def:g10:stats:mean}{माध्य} है
\[
\bar y = \frac{y_1 + \dots + y_N}{N}
= \frac{(a x_1 + b) + \dots + (a x_N + b)}{N}
= \frac{a(x_1 + \dots + x_N) + Nb}{N}
= a \bar x + b .
\]

\emph{2.} $a = 1.8$ और $b = 32$ के साथ: \omterm{def:g10:stats:mean}{माध्य} $1.8 \times 20 + 32 = 68$ डिग्री फ़ारेनहाइट।
\end{solution}

\begin{solution}{pb:g10:stats:1}
\textbf{1.} \omterm{def:g10:stats:mean}{माध्य}: $\frac{2\,550}{7} \approx 364$ हज़ार यूरो; \omterm{def:g10:stats:median}{माध्यिका}: सात क्रमबद्ध दामों में चौथा,
$230$। गली का वर्णन \omterm{def:g10:stats:median}{माध्यिका} करती है --- सात में से छह घरों का दाम $364$ के
आसपास भी नहीं है।

\textbf{2.} हवेली के बिना: \omterm{def:g10:stats:mean}{माध्य} $\frac{1\,350}{6} = 225$, \omterm{def:g10:stats:median}{माध्यिका} $\frac{220 + 230}{2} = 225$। \omterm{def:g10:stats:mean}{माध्य} लगभग $139$
गिरा; \omterm{def:g10:stats:median}{माध्यिका} $5$। एक चरम मान \omterm{def:g10:stats:mean}{माध्य} को कहीं भी घसीट सकता है; \omterm{def:g10:stats:median}{माध्यिका}
उसे मुश्किल से ही देखती है --- वह \emph{सुदृढ़} है।

\textbf{3.} \omterm{def:g10:stats:mean}{माध्य} $= \frac{0 \times 20 + 1 \times 25 + 2 \times
30 + 3 \times 15 + 4 \times 10}{100} = \frac{170}{100} = 1.7$। किसी परिवार में $1.7$ संतानें नहीं हैं, पर $1.7 \times 100 = 170$
संतानों की ठीक-ठीक कुल संख्या \emph{है}: \omterm{def:g10:stats:mean}{माध्य} कुल को समान रूप से बाँटकर
अपने भीतर समेटे रखता है --- यही उसकी असली प्रतिभा है।

\textbf{4.} $N = 15$: \omterm{def:g10:stats:median}{माध्यिका} $=$ आठवाँ मान $= 11$; $Q_1 =$ चौथा मान $= 8$;
$Q_3 =$ बारहवाँ मान $= 14$।

\textbf{5.} \omterm{def:g10:stats:quartiles}{परास} $19 - 4 = 15$: पूरा प्रसार, चरम मान सहित। अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} $14 - 8 = 6$:
मध्य आधे भाग की चौड़ाई --- चरम मानों के शोर के बिना प्रसार।

\textbf{6.} $+b$ का खिसकाव \emph{हर} मान को हटा देता है, इसलिए मानों के
अंतर --- \omterm{def:g10:stats:quartiles}{परास}, अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} --- अपरिवर्तित रहते हैं: $(x_i + b) - (x_j + b) = x_i - x_j$। $a$ से
गुणा करने पर सभी अंतर $a$ गुने हो जाते हैं: \omterm{def:g10:stats:quartiles}{परास} और अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास}
$\abs a$ गुने हो जाते हैं।

\textbf{7.} जोड़ वाला सुधार: $6 \to 9$ और $18 \to 21$ --- कमज़ोर विद्यार्थी को $50\,\%$
मिलता है, तेज़ को $17\,\%$, और $21$ $20$-अंक के पैमाने को थोड़ा ही फोड़ता
है। गुणा वाला: $6 \to 8$ और $18 \to 24$ --- अनुपात बने रहते हैं, पर $20$ के पैमाने
पर $24$ असंभव है: वही ऊपरी सीमा की समस्या। जोड़ वाले सुधार नीचे वालों को
सँवारते हैं; गुणा वाले ऊपर वालों को उड़ा देते हैं।

\textbf{8.} कुल अंक: $20 \times 12 + 30 \times 10 = 540$, इसलिए मिला हुआ \omterm{def:g10:stats:mean}{माध्य} $\frac{540}{50} = 10.8$ है: कक्षा B के तीस
विद्यार्थी कक्षा A के बीस पर भारी पड़ते हैं --- \omterm{def:g10:stats:mean}{माध्य} \emph{भारित} औसत से
मिलते हैं, कभी साधारण औसत से नहीं।

\textbf{9.} $A$ की \omterm{def:g10:stats:median}{माध्यिका}: $2$; $B$ की: $5$। मिली हुई सूची $\{0, 1, 2, 3, 10\}$:
\omterm{def:g10:stats:median}{माध्यिका} $2$। माध्यिकाओं का आकार-भारित औसत $\frac{3 \times 2 + 2 \times 5}{5} = 3.2
\neq 2$ होता: माध्यिकाएँ कोई कुल
नहीं उठातीं, इसलिए उनके मिलाने का कोई सूत्र नहीं है --- पूरी सूची फिर से
क्रमबद्ध करनी पड़ती है।

\textbf{10.} \omterm{def:g10:stats:mean}{माध्य}: $\frac{31.3}{5} = 6.26$। \omterm{def:g10:stats:median}{माध्यिका}: $5.4$। कटा-छँटा \omterm{def:g10:stats:mean}{माध्य}: $\frac{5.3 + 5.4 + 5.5}{3} = 5.4$। एक
उत्साही (या भ्रष्ट) निर्णायक ने \omterm{def:g10:stats:mean}{माध्य} को लगभग एक अंक हिला दिया; \omterm{def:g10:stats:median}{माध्यिका} और
कटे-छँटे \omterm{def:g10:stats:mean}{माध्य} ने कंधे उचका दिए --- इसीलिए निर्णायक-आधारित खेलों में
काट-छाँट के नियम हैं।

\textbf{11.} कारख़ाना A: मध्य आधा भाग $0.2$ mm के भीतर; कारख़ाना B: $4$
mm के भीतर। \omterm{def:g10:stats:mean}{माध्य} एक ही, विश्वसनीयता उलटी: A से ख़रीदिए। निर्णय
अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} ने किया --- अकेले \omterm{def:g10:stats:mean}{माध्य} केंद्र मिलाते हैं, एकरूपता कभी
नहीं।

\textbf{12.} \omterm{def:g10:stats:median}{माध्यिका} का \omterm{def:g10:stats:mean}{माध्य} से बहुत नीचे होना \emph{लंबी दाहिनी पूँछ}
की चुगली करता है: अधिकांश प्रतीक्षाएँ छोटी, कुछ विनाशकारी। उदाहरण: $\{5, 5, 10, 10, 95\}$
--- \omterm{def:g10:stats:median}{माध्यिका} $10$, \omterm{def:g10:stats:mean}{माध्य} $\frac{125}{5} = 25$।

\textbf{13.} इसका अर्थ है कि उस आयु के $75\,\%$ शिशु उससे छोटे हैं (और $25\,\%$
बड़े): यह एक क्रमांक है, कोई नाप नहीं। ``हर बच्चा \omterm{def:g10:stats:mean}{माध्य} से ऊपर'' असंभव है
--- यदि सभी मान \omterm{def:g10:stats:mean}{माध्य} से बड़े होते, तो उनका औसत भी \omterm{def:g10:stats:mean}{माध्य} से बड़ा होता, जबकि
वह स्वयं \omterm{def:g10:stats:mean}{माध्य} है: विरोधाभास। पर \emph{लगभग} सभी हो सकते हैं: एक बहुत छोटा
बच्चा \omterm{def:g10:stats:mean}{माध्य} को बाक़ी सबसे नीचे रोक सकता है। \emph{\omterm{def:g10:stats:median}{माध्यिका}} से ऊपर:
परिभाषा से ही कभी आधे से अधिक नहीं। क्रमांकों की चापलूसी नहीं हो सकती;
औसतों की हो सकती है।

\textbf{14.} प्रारूपिक अंक: दोनों के लिए एक ही, \omterm{def:g10:stats:median}{माध्यिका} $11$। मध्य भाग की
समरूपता: B अधिक कसी हुई (अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} $3$ बनाम $5$)। चरम: B अधिक
उत्पाती (\omterm{def:g10:stats:quartiles}{परास} $17$ बनाम $13$)। B एक जैसे विद्यार्थियों की कक्षा है जिसमें
कुछ बहिर्मान हैं; A समान रूप से फैली है।

\textbf{15.} (1) $5\,000$ \omterm{def:g10:stats:mean}{माध्य} है या \omterm{def:g10:stats:median}{माध्यिका} --- और क्या मुझे \omterm{def:g10:stats:median}{माध्यिका} मिल
सकती है? (2) प्रसार क्या है (\omterm{def:g10:stats:quartiles}{चतुर्थक}), ताकि पता चले कि ``प्रारूपिक'' कितना
प्रारूपिक है? (3) किन्हें गिना गया --- प्रतिदर्श का आकार और चयन (माध्यमिक
विद्यालय खंड की ``आँकड़े कैसे झूठ बोलते हैं'' वाली सप्ताहांत समस्या का
सर्वेक्षण वाला पाठ)?

\textbf{16.} जैसे $100, 150, 200, 250, 800$: \omterm{def:g10:stats:median}{माध्यिका} $200$, \omterm{def:g10:stats:mean}{माध्य} $\frac{1\,500}{5} = 300$।

\textbf{17.} $\sum (x_i - \bar x) = \sum x_i - N \bar x =
N\bar x - N\bar x = 0$। जाँच: $-200 - 150 - 100 - 50 + 500 =
0$। \omterm{def:g10:stats:mean}{माध्य} वह बिंदु है जहाँ विचलन संतुलित हो
जाते हैं --- आँकड़ों का गुरुत्व केंद्र (माध्यमिक विद्यालय खंड की केन्द्रक
वाली सप्ताहांत समस्या का $G$, एक विमा में)।

\textbf{18.} \omterm{def:g10:stats:median}{माध्यिका} $4$ पर: $3 + 1 + 0 + 5 + 9 = 18$ km। \omterm{def:g10:stats:mean}{माध्य} $6$ पर: $5 + 3 + 2 + 3 + 7 = 20$ km। $5$
पर: $4 + 2 + 1 + 4 + 8 = 19$ km। \omterm{def:g10:stats:median}{माध्यिका} जीतती है --- उससे हटने पर हमेशा जितनी दुकानें पास
आती हैं उससे अधिक दूर चली जाती हैं, इसलिए कुल केवल बढ़ ही सकता है।

\textbf{19.} $0.3^{10} \approx 6 \times 10^{-6}$: \emph{प्रति दस लाख} लगभग छह वैमानिक। $4\,000$ वैमानिकों
में अपेक्षित संख्या $0.02$ है --- यानी शून्य, जैसा लेफ़्टिनेंट डेनियल्स ने
पाया। वायुसेना ने औसत आदमी के लिए बनाना बंद कर दिया और सब कुछ समायोज्य बना
दिया: सीटें, पैडल, पट्टियाँ --- रचना केंद्र के लिए नहीं, \emph{प्रसार} के
लिए।
