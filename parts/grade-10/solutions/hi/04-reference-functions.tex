\begin{solution}{exo:g10:reffunc:2}
$(2,1)$ और $(5,10)$ से होकर: \omterm{def:g10:reffunc:affine}{प्रवणता} $m = \frac{10 - 1}{5 - 2} = 3$; फिर $1 = 3 \times 2 + p$ से $p = -5$ मिलता है: $f(x) = 3x - 5$।

$(-1,4)$ और $(3,-4)$ से होकर: \omterm{def:g10:reffunc:affine}{प्रवणता} $m = \frac{-4 - 4}{3 - (-1)} =
\frac{-8}{4} = -2$; फिर $4 = -2 \times (-1) + p$ से $p = 2$ मिलता है: $f(x) = -2x + 2$।
\end{solution}

\begin{solution}{exo:g10:reffunc:4}
$x^2 = 16$: दो हल, $x = 4$ और $x = -4$ (क्षैतिज रेखा $y = 16$ परवलय को दो बार काटती है)।

$x^2 \leq 16$: परवलय दोनों प्रतिच्छेद बिंदुओं के बीच रेखा से नीचे है: $x \in \intcc{-4}{4}$।

$x^2 > 9$: परवलय $\intcc{-3}{3}$ के बाहर $y = 9$ से कड़ाई से ऊपर है: $x \in \intoo{-\infty}{-3} \cup \intoo{3}{+\infty}$।
\end{solution}

\begin{solution}{exo:g10:reffunc:5}
$0 < x < 1$ के लिए: $x < 1$ को बार-बार $x$ से गुणा करने पर $x^3 < x^2 < x$ मिलता है, और \cref{prop:g10:reffunc:compare} से
$x < \sqrt x$ मिलता है, इसलिए
\[
x^3 < x^2 < x < \sqrt x .
\]
$x = 0.25$ से जाँच: $x^3 = 0.015625$, $x^2 = 0.0625$, $x = 0.25$, $\sqrt x = 0.5$।
\end{solution}

\begin{solution}{exo:g10:reffunc:6}
$\frac1x = 2$ का अद्वितीय हल $x = \frac12$ है (अतिपरवलय क्षैतिज रेखा $y = 2$ से एक ही बार मिलता
है, धनात्मक शाखा पर)।

$x > 0$ के लिए: $\frac1x < 2$। $x > 0$ से गुणा करने पर $1 < 2x$, इसलिए $x > \frac12$: हल-समुच्चय $\intoo{\frac12}{+\infty}$।

यदि $x < 0$ की भी अनुमति हो: हर ऋणात्मक $x$ $\frac1x < 0 < 2$ को संतुष्ट करता है, इसलिए
पूरा हल-समुच्चय $\intoo{-\infty}{0} \cup \intoo{\frac12}{+\infty}$ है।
\end{solution}

\begin{solution}{exo:g10:reffunc:7}
(a) $\intcc{2}{5}$ पर सभी निवेश धनात्मक हैं और वर्ग करना \omterm{def:g10:functions:variations}{वर्धमान} है: $4 \leq x^2 \leq 25$।

(b) $\intcc{-3}{-1}$ पर वर्ग करना \omterm{def:g10:functions:variations}{ह्रासमान} है: सबसे बड़ा वर्ग $-3$ से आता है: $1 \leq x^2 \leq 9$।

(c) $\intcc{-2}{3}$ पर \omterm{def:g10:numbers:interval}{अंतराल} में $0$ आ जाता है, जहाँ वर्ग अपना \omterm{def:g10:functions:extrema}{न्यूनतम} $0$ पाता है;
\omterm{def:g10:functions:extrema}{अधिकतम} $(-2)^2 = 4$ और $3^2 = 9$ में से बड़ा वाला है। इसलिए $0 \leq x^2 \leq 9$।
\end{solution}

\begin{solution}{exo:g10:reffunc:8}
पहली कंपनी: $f(x) = 1.5x + 4$; दूसरी: $g(x) = 2x$। पहली तब सस्ती है जब
\[
1.5x + 4 < 2x
\ \Longleftrightarrow\
4 < 0.5x
\ \Longleftrightarrow\
x > 8 .
\]
$8$ km से आगे पहली कंपनी सस्ती है; ठीक $8$ km पर दोनों $16$ लेती हैं।
\end{solution}

\begin{solution}{exo:g10:reffunc:9}
दोनों पक्ष अऋणात्मक हैं, इसलिए यह असमिका उनके वर्गों वाली असमिका के तुल्य
है:
\[
\left(\sqrt{a+b}\right)^2 = a + b
\qquad\text{और}\qquad
\left(\sqrt a + \sqrt b\right)^2 = a + 2\sqrt a \sqrt b + b .
\]
चूँकि $2\sqrt a\sqrt b \geq 0$ है, दूसरा वर्ग कम से कम पहले जितना है, जिससे $\sqrt{a+b} \leq \sqrt a + \sqrt b$ सिद्ध हो जाता
है। समता ठीक तब है जब $\sqrt a \sqrt b = 0$ हो, अर्थात् जब $a = 0$ या $b = 0$ हो।
\end{solution}

\begin{solution}{exo:g10:reffunc:10}
\emph{1.} दाहिने पक्ष को उभयनिष्ठ हर पर लाइए:
\[
2 + \frac{3}{x-1} = \frac{2(x-1) + 3}{x - 1} = \frac{2x + 1}{x - 1}
= f(x).
\]

\emph{2.} $\intoo{1}{+\infty}$ पर, $x$ बढ़ने पर $x - 1$ बढ़ता है और धनात्मक बना रहता है,
इसलिए $\frac{3}{x-1}$ घटता है (व्युत्क्रम \omterm{def:g10:functions:function}{फलन}, धनात्मक अचर $3$ से गुणित), इसलिए
$f(x) = 2 + \frac{3}{x-1}$ घटता है: $f$ $\intoo{1}{+\infty}$ पर निरंतर \omterm{def:g10:functions:variations}{ह्रासमान} है।
\end{solution}

\begin{solution}{pb:g10:reffunc:1}
\textbf{1.} $d(30) = 3^2 = 9$ m; $d(50) = 25$ m; $d(90) =
81$ m; $d(130) = 169$ m।

\textbf{2.} $v$ दुगुना करने पर $\frac v{10}$ दुगुना होता है और उसका वर्ग $4$
गुना: $d(100) = 100$ m $= 4 \times d(50)$। चाल दुगुनी, दूरी चौगुनी --- वर्ग \omterm{def:g10:functions:function}{फलन} का मापन-गुण।

\textbf{3.} $\left(\frac{v}{10}\right)^2 = 50$ से $\frac{v}{10} = \sqrt{50}$ मिलता है, इसलिए $v = 10\sqrt{50} \approx 71$ km/h। उत्तर वर्गमूल ने
दिया; पठन असंदिग्ध है क्योंकि चालें धनात्मक होती हैं, और $\intco{0}{+\infty}$ पर वर्ग \omterm{def:g10:functions:function}{फलन}
कड़ाई से चढ़ता है (\cref{prop:g10:reffunc:sqrt}: एक ही धनात्मक \omterm{def:g10:functions:function}{पूर्वप्रतिबिंब})।

\textbf{4.} $d(31) - d(30) = 9.61 - 9 = 0.61$ m, जबकि $d(91) - d(90) = 82.81 - 81 = 1.81$ m: वही $+1$ km/h अधिक चाल पर तीन गुना
अधिक दूरी माँगता है। परवलय केवल चढ़ता ही नहीं, वह \emph{तीखा भी होता जाता
है} --- उसकी चढ़ाई की दर $x$ के साथ बढ़ती है।

\textbf{5.} $D(50) = 14 + 25 = 39$ m; $D(130) = 36.4 + 169 =
205.4$ m। शहर में एकघात (प्रतिक्रिया) वाला पद बड़ा
हिस्सा है; राजमार्ग पर वर्ग वाला पद उसे कुचल देता है --- एकघात एक-सी चाल से
बढ़ता है, वर्ग भाग खड़ा होता है।

\textbf{6.} $F(0.5) = 240$, $F(1) = 120$, $F(3) = 40$।

\textbf{7.} व्युत्क्रम \omterm{def:g10:functions:function}{फलन} घटता है: आधार से जितना दूर, उतना कम बल चाहिए ---
और पर्याप्त लंबी भुजा ($d$ विशाल) के साथ कितना ही छोटा बल कितने ही बड़े
भार को संतुलित कर सकता है: आर्किमिडीज़ की डींग अतिपरवलय की पूँछ में बसती है,
जो $0$ के पास पहुँचती है पर उस तक कभी नहीं पहुँचती। और $d = 0$ वर्जित मान
है: भुजा नहीं, तो उत्तोलक नहीं --- वही ऊर्ध्वाधर अनन्तस्पर्शी।

\textbf{8.} $2$ m पर $\frac{100}{4} = 25$, $3$ m पर $\frac{100}{9} \approx 11$, $10$ m पर $1$। दूरी $d$
पर प्रकाश $4\pi d^2$ क्षेत्रफल के गोले पर फैल चुका होता है (माध्यमिक विद्यालय
खंड की आर्किमिडीज़ की समाधि वाली समस्या): वही ऊर्जा $d^2$ की तरह बढ़ते
क्षेत्रफल से बँटती है --- इसीलिए व्युत्क्रम \emph{वर्ग}।

\textbf{9.} $n = 10, 20,
30$ के लिए प्रति व्यक्ति $60$, $30$, $20$ यूरो। $\frac{600}{n} \leq 25$ के
लिए: $n \geq 24$ सहभागी। अतिपरवलय $n$ बढ़ने पर लगातार सस्ता होता हिस्सा देने का
वादा करता है --- और $0$ तक पहुँचने से सदा मना कर देता है: बस कभी मुफ़्त
नहीं होती।

\textbf{10.} $a(x) = \frac{2x + 3}{x} = 2 + \frac3x$: $\intoo{0}{+\infty}$ पर \omterm{def:g10:functions:variations}{ह्रासमान} (व्युत्क्रम \omterm{def:g10:functions:function}{फलन} खिसका हुआ), और
अनन्तस्पर्शी $y = 2$ के पास पहुँचता हुआ। आर्थिक रूप से: नियत $3$ यूरो को
अधिक पोस्टरों पर फैलाने से प्रति नग ख़र्च घटकर सामग्री के अटल $2$ यूरो की
ओर चला जाता है --- पैमाने की मितव्ययिता, पर एक कठोर तल के साथ।

\textbf{11.} पृथ्वी: $a^3 = 1$, $T^2 = 1$। मंगल: $1.52^3 \approx 3.51$ और $1.88^2 \approx 3.53$। बृहस्पति: $5.20^3 \approx 140.6$
और $11.86^2 \approx 140.7$। केप्लर ने जो देखा: हर ग्रह के लिए $T^2 = a^3$ --- पूरे आकाश के लिए एक
ही नियम।

\textbf{12.} $T = \sqrt{a^3}$: शनि के लिए $\sqrt{9.54^3} = \sqrt{868} \approx 29.5$ वर्ष, प्रेक्षित $29.4$ के मुक़ाबले:
नियम दशमांश तक सही उतरता है।

\textbf{13.} क्षुद्रग्रह: $T = \sqrt{4^3} = \sqrt{64} = 8$ वर्ष। धूमकेतु: $a^3 = 27^2 = 729 = 9^3$, इसलिए $a = 9$ AU।

\textbf{14.} वर्ग ($T^2$), घन ($a^3$) और वर्गमूल ($T$ निकालने के लिए)।
मापन: $a$ को $4$ से गुणा करने पर $T$ $4\sqrt4 = 8$ गुना हो जाता है --- जैसा
क्षुद्रग्रह पुष्ट करता है ($a$ पृथ्वी का चार गुना, $T$ आठ गुना)।

\textbf{15.} संख्याओं की एक सारणी, जिसे चंद आधार \omterm{def:g10:functions:function}{फलनों} की आकृतियों से
मिलाया गया, ने ब्रह्मांड का एक नियम दे दिया --- और वह भी यह जानने से सत्तर
वर्ष पहले कि वह \emph{क्यों} सही है। \omterm{def:g10:functions:function}{फलन} पहचान लेना विज्ञान का आधा काम है।

\textbf{16.} $\intoo{0}{1}$ पर: $x^2 < x < \sqrt x$; $\intoo{1}{+\infty}$ पर: $\sqrt x < x < x^2$। जाँच: $x = 0.25$ पर $0.0625 < 0.25 < 0.5$; $x = 4$
पर $2 < 4 < 16$।

\textbf{17.} $x \geq 0$ के लिए दोनों पक्ष $\geq 0$ हैं, इसलिए $\sqrt x > x \iff x > x^2 \iff x(1 - x) > 0 \iff
x \in \intoo{0}{1}$।

\textbf{18.} $x = 0.5$ पर: $x^3 = 0.125 < x^2 = 0.25 < x =
0.5$। $x = 2$ पर: $2 < 4 < 8$: क्रम पलट जाता है। $x^2$ और
$x^3$ के प्रतिच्छेद: $x^3 - x^2 = x^2(x - 1) = 0$: $x = 0$ और $x = 1$ पर।

\textbf{19.} $10t^2 = 100\sqrt t$ से $t^2 = 10\sqrt t$ मिलता है; वर्ग करने पर $t^4 = 100t$, इसलिए $t^3 = 100$ और
$t = \sqrt[3]{100} \approx 4.64$ वर्ष --- लगभग $4$ वर्ष $8$ महीने। (तब दोनों योजनाएँ लगभग $215$
यूरो देती हैं।) सीख: मूल शुरू में दौड़ लगाते हैं, घातें हर मैराथन जीतती
हैं।

\textbf{20.} ब्रेक: परवलय का \emph{तीखा होते जाना} ही चाल का ख़तरा है।
उत्तोलक: $0$ पर \emph{ऊर्ध्वाधर अनन्तस्पर्शी} और लंबी पूँछ ही यंत्रकार

जाना} धीमे बचतकर्ता की नियति है। पाँच \omterm{def:g10:functions:graph}{आलेख}, पाँच नियम: प्रकृति की वर्णमाला
सचमुच।
