\chapter{Álgebra: equações e inequações}\label{ch:g10:algebra}

A álgebra é a arte de calcular com letras. Este capítulo revê e aprimora
os dois movimentos básicos --- \omterm{def:g10:algebra:expand}{expandir} e \omterm{def:g10:algebra:expand}{fatorar} --- e usa-os para
resolver \omterm{def:g10:algebra:equation}{equações} e inequações de modo sistemático, tendo o \omterm{def:g10:algebra:signtable}{quadro de
sinais} como ferramenta central.

\section{Expandir e fatorar}

\begin{definition}[Expandir, fatorar]\label{def:g10:algebra:expand}
\emph{Expandir}\index{expandir} uma expressão é transformar produtos em
somas; \emph{fatorar}\index{fatorar} é o inverso, transformar uma soma em
um produto. A regra básica é a distributividade:
\[
k(a + b) = ka + kb,
\qquad
(a + b)(c + d) = ac + ad + bc + bd .
\]
\end{definition}

\begin{example}\label{ex:g10:algebra:expand}
Expanda $(2x + 3)(x - 5)$ passo a passo:
\begin{align*}
(2x + 3)(x - 5)
&= 2x \times x + 2x \times (-5) + 3 \times x + 3 \times (-5) \\
&= 2x^2 - 10x + 3x - 15 \\
&= 2x^2 - 7x - 15 .
\end{align*}
\end{example}

\begin{theorem}[Produtos notáveis]\label{thm:g10:algebra:identities}
Para quaisquer \omterm{def:g10:numbers:sets}{números reais} $a$ e $b$:
\[
(a+b)^2 = a^2 + 2ab + b^2,
\qquad
(a-b)^2 = a^2 - 2ab + b^2,
\qquad
(a+b)(a-b) = a^2 - b^2 .
\]
\end{theorem}

\begin{proof}
Cada uma delas é uma expansão direta. Para a primeira:
\[
(a+b)^2 = (a+b)(a+b) = a^2 + ab + ba + b^2 = a^2 + 2ab + b^2 .
\]
A segunda segue trocando $b$ por $-b$: $(a-b)^2 = a^2 + 2a(-b) +
(-b)^2 = a^2 - 2ab + b^2$. E, para a terceira,
$(a+b)(a-b) = a^2 - ab + ba - b^2 = a^2 - b^2$.
\end{proof}

\begin{example}\label{ex:g10:algebra:identities}
Usados no sentido direto (expandindo) e no inverso (fatorando):
\begin{align*}
(3x + 2)^2 &= 9x^2 + 12x + 4,
& x^2 - 25 &= (x+5)(x-5), \\
(x - 4)^2 &= x^2 - 8x + 16,
& 4x^2 + 4x + 1 &= (2x + 1)^2 .
\end{align*}
O cálculo mental também ganha: $101 \times 99 = (100+1)(100-1) =
10000 - 1 = 9999$.
\end{example}

\begin{method}[Fatorar uma expressão]\label{met:g10:algebra:factor}
Tente, nesta ordem:
\begin{enumerate}
  \item \emph{Fator comum:} identifique um fator presente em todos os
        termos e coloque-o em evidência, por exemplo
        $6x^2 + 4x = 2x(3x + 2)$. O fator comum pode ser um parêntese
        inteiro: $(x+1)(2x-3) + (x+1)(x+7) = (x+1)(3x+4)$.
  \item \emph{Produto notável:} reconheça $a^2 - b^2$ ou
        $a^2 \pm 2ab + b^2$, por exemplo $9x^2 - 16 = (3x)^2 - 4^2 =
        (3x+4)(3x-4)$.
  \item Combine os dois e confira o resultado expandindo-o de volta.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:factorsteps}
Fatore $E = (2x + 1)^2 - (x - 3)^2$. Trata-se de uma diferença de dois
quadrados $a^2 - b^2$ com $a = 2x+1$ e $b = x-3$:
\begin{align*}
E &= \bigl[(2x+1) + (x-3)\bigr] \times \bigl[(2x+1) - (x-3)\bigr] \\
  &= (3x - 2)(2x + 1 - x + 3) \\
  &= (3x - 2)(x + 4).
\end{align*}
Atenção ao segundo colchete: subtrair $x - 3$ é subtrair $x$ \emph{e}
somar $3$.
\end{example}

\section{Equações}

\begin{definition}[Equação, solução]\label{def:g10:algebra:equation}
Uma \emph{equação}\index{equação} é uma igualdade que contém um número
desconhecido $x$; uma \emph{solução} é um valor de $x$ que torna a
igualdade verdadeira. Resolver a equação é encontrar \emph{todas} as suas
soluções.
\end{definition}

Duas \omterm{def:g10:algebra:equation}{equações} são \emph{equivalentes} quando têm exatamente as mesmas
soluções. Somar o mesmo número aos dois lados, ou multiplicar os dois
lados pelo mesmo número \emph{não nulo}, produz uma \omterm{def:g10:algebra:equation}{equação} equivalente.

\begin{example}[Equação do primeiro grau]\label{ex:g10:algebra:firstdegree}
Resolva $5x - 7 = 2x + 8$, um movimento de cada vez:
\begin{align*}
5x - 7 &= 2x + 8 \\
5x - 2x &= 8 + 7 && \text{(some $7 - 2x$ aos dois lados)} \\
3x &= 15 \\
x &= 5 && \text{(divida os dois lados por $3$).}
\end{align*}
A única solução é $5$. Verificação: $5 \times 5 - 7 = 18$ e
$2 \times 5 + 8 = 18$.
\end{example}

\begin{theorem}[Regra do produto nulo]\label{thm:g10:algebra:zeroproduct}
Um produto de \omterm{def:g10:numbers:sets}{números reais} é nulo se, e somente se, pelo menos um dos
seus fatores é nulo:
\[
A \times B = 0
\quad\Longleftrightarrow\quad
A = 0 \ \text{ ou } \ B = 0 .
\]
\end{theorem}

\begin{proof}
Se $A = 0$ ou $B = 0$, o produto é claramente $0$. Reciprocamente,
suponha $AB = 0$ e $A \neq 0$. Dividindo os dois lados por $A$, obtemos
$B = 0$.
\end{proof}

\begin{method}[Resolver fatorando]\label{met:g10:algebra:zeroproduct}
Para resolver uma \omterm{def:g10:algebra:equation}{equação} cujo lado direito é $0$ e cujo lado esquerdo
pode ser fatorado:
\begin{enumerate}
  \item passe tudo para a esquerda, de modo que a \omterm{def:g10:algebra:equation}{equação} fique $E = 0$;
  \item fatore $E$ em um produto de fatores do primeiro grau;
  \item aplique a regra do produto nulo e resolva cada \omterm{def:g10:algebra:equation}{equação} pequena;
  \item reúna todas as soluções encontradas.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:zeroproduct}
Resolva $(x - 2)(3x + 6) = 0$: ou $x - 2 = 0$, o que dá $x = 2$, ou
$3x + 6 = 0$, o que dá $x = -2$. Soluções: $-2$ e $2$.

Resolva $x^2 = 9x$. \emph{Não} divida por $x$ (ele pode ser nulo!); passe
tudo para um lado e fatore:
\[
x^2 - 9x = 0
\quad\Longleftrightarrow\quad
x(x - 9) = 0
\quad\Longleftrightarrow\quad
x = 0 \ \text{ ou } \ x = 9 .
\]
\end{example}

\begin{example}[Equação com quociente]\label{ex:g10:algebra:quotient}
Resolva $\dfrac{x - 1}{x + 2} = 0$. Um quociente é nulo exatamente quando
o numerador é nulo \emph{e} o denominador não é. Aqui $x - 1 = 0$ dá
$x = 1$, e $1 + 2 = 3 \neq 0$: a única solução é $1$. O valor $x = -2$ é
\emph{proibido} (divisão por zero) e deve sempre ser excluído de saída.
\end{example}

\section{Inequações e quadros de sinais}

\begin{proposition}[Regras das desigualdades]\label{prop:g10:algebra:ineqrules}
Seja $a \leq b$.
\begin{enumerate}
  \item Para qualquer $c$: $a + c \leq b + c$ (somar preserva a ordem).
  \item Se $c > 0$: $ac \leq bc$ (multiplicar por um número positivo
        preserva a ordem).
  \item Se $c < 0$: $ac \geq bc$ (multiplicar por um número negativo
        \emph{inverte} a ordem).
\end{enumerate}
\end{proposition}

\begin{proof}
1. $(b + c) - (a + c) = b - a \geq 0$.
2. $bc - ac = (b-a)c$ é o produto de dois números não negativos, logo
$bc \geq ac$.
3. Agora $(b-a)c \leq 0$, por ser o produto de um número não negativo por
um número negativo, logo $bc \leq ac$.
\end{proof}

\begin{example}\label{ex:g10:algebra:firstineq}
Resolva $-2x + 3 > 9$:
\begin{align*}
-2x + 3 &> 9 \\
-2x &> 6 && \text{(subtraia 3)} \\
x &< -3 && \text{(divida por $-2 < 0$: inverta a desigualdade).}
\end{align*}
O conjunto solução é $\intoo{-\infty}{-3}$.
\end{example}

\begin{definition}[Quadro de sinais]\label{def:g10:algebra:signtable}
Um \emph{quadro de sinais}\index{quadro de sinais} registra, uma linha por
fator, o sinal ($+$ ou $-$) de cada fator de uma expressão quando $x$
percorre $\R$, com um $0$ em cada valor onde o fator se anula; uma linha
final dá o sinal do produto, aplicando a regra dos sinais coluna a coluna.
\end{definition}

\begin{example}\label{ex:g10:algebra:signtable}
Sinal de $P(x) = (x - 2)(3 - x)$. O fator $x - 2$ anula-se em $2$ e é
positivo depois; o fator $3 - x$ anula-se em $3$ e é positivo antes:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccc}
$x$ & $-\infty$ \quad & $2$ & & $3$ & \quad $+\infty$ \\
\hline
$x - 2$ & $-$ & $0$ & $+$ & & $+$ \\
$3 - x$ & $+$ & & $+$ & $0$ & $-$ \\
\hline
$P(x)$ & $-$ & $0$ & $+$ & $0$ & $-$ \\
\end{tabular}
\end{center}
Assim, $P(x) > 0$ exatamente em $\intoo{2}{3}$, e $P(x) \leq 0$ em
$\intoc{-\infty}{2} \cup \intco{3}{+\infty}$.
\end{example}

\begin{method}[Resolver uma inequação com um quadro de sinais]\label{met:g10:algebra:signtable}
Para resolver uma inequação como $E(x) \geq 0$ ou $E(x) < 0$:
\begin{enumerate}
  \item passe tudo para o lado esquerdo, de modo que o lado direito seja
        $0$ (nunca multiplique os dois lados por uma expressão de sinal
        desconhecido);
  \item fatore o lado esquerdo, ou reduza-o a um denominador comum se for
        um quociente;
  \item monte o \omterm{def:g10:algebra:signtable}{quadro de sinais} com uma linha por fator (num quociente,
        marque os valores proibidos com uma dupla barra na última linha);
  \item leia os \omterm{def:g10:numbers:interval}{intervalos} em que o sinal pedido ocorre, verificando se
        cada extremo está incluído.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:algebra:quotientineq}
Resolva $\dfrac{x + 1}{x - 3} \leq 0$. O numerador anula-se em $-1$ e o
denominador em $3$ (valor proibido). O quociente é negativo quando os
dois têm sinais contrários, o que ocorre para $x$ entre $-1$ e $3$.
Incluindo o extremo em que o \emph{numerador} se anula e excluindo o
valor proibido: o conjunto solução é $\intco{-1}{3}$.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g10:algebra:1}
Expanda e simplifique:
\[
3(2x - 5) - 2(x - 7), \qquad
(x + 4)(2x - 1), \qquad
(3x - 2)^2, \qquad
(5 - x)(5 + x).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:algebra:2}
Fatore:
\[
x^2 - 49, \qquad
x^2 + 10x + 25, \qquad
7x^2 - 14x, \qquad
16x^2 - 8x + 1 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:algebra:3}
Resolva:
\[
4x + 9 = x - 3, \qquad
\frac{2x - 1}{3} = 5, \qquad
2(x - 3) = 2x + 1 .
\]
(Uma delas não tem solução --- explique por quê.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:algebra:4}
Resolva usando a regra do produto nulo:
\[
(x + 5)(2x - 8) = 0, \qquad
x^2 - 16 = 0, \qquad
x^2 + 6x + 9 = 0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:algebra:5}
Resolva as inequações e escreva os conjuntos solução como \omterm{def:g10:numbers:interval}{intervalos}:
\[
3x - 5 \leq 7, \qquad
-4x + 1 > 9, \qquad
2(x - 1) \geq 5x + 4 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:6}
Fatore $E = (x - 1)(x + 3) - (x - 1)(2x - 5)$ e resolva em seguida
$E = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:7}
Resolva $x^2 = 5x$ e depois $(2x + 1)^2 = (x - 4)^2$. (Na segunda, passe
tudo para a esquerda e fatore a diferença de quadrados.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:8}
Usando um \omterm{def:g10:algebra:signtable}{quadro de sinais}, resolva
\[
(x + 2)(4 - x) > 0,
\qquad
\frac{2x - 6}{x + 1} \geq 0 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:9}
Um serviço de streaming custa $9$ por mês, mais uma taxa única de
adesão de $15$; um concorrente custa $12$ por mês, sem taxa. A partir de
quantos meses o primeiro serviço passa a ser a escolha mais barata no
total? Monte e resolva uma inequação.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:algebra:10}
Resolva $\dfrac{x+3}{x-1} = 2$. (Multiplique os dois lados por $x - 1$
depois de excluir o valor proibido, ou passe tudo para a esquerda com
denominador comum.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:algebra:11}
Seja $n$ um inteiro. Mostre que $(n + 1)^2 - n^2$ é sempre ímpar e que a
diferença entre os quadrados de dois ímpares consecutivos é sempre
múltiplo de $8$.
\end{exercise}

\section{Problema: Da Babilônia à razão áurea}

\begin{problem}[{Problema de fim de semana --- problemas de soma e
produto, completar o quadrado como al-Khwarizmi o desenhou, e a
equação $x^2 = x + 1$}]
\label{pb:g10:algebra:1}
Há quatro mil anos, escribas babilônios resolviam isto em argila:
\emph{achar dois números conhecendo sua soma e seu produto}. O truque
deles --- procurar os números simetricamente em torno da metade da
soma --- ainda hoje resolve toda \omterm{def:g10:algebra:equation}{equação} do segundo grau, sob o nome
moderno de \emph{completar o quadrado}. Este problema aprende o truque,
vê al-Khwarizmi completar literalmente um quadrado e põe o método a
trabalhar na mais célebre de todas as proporções.

\medskip
\noindent\textbf{Parte I --- O problema babilônico.}
\begin{enumerate}
  \item Aquecimento (\cref{thm:g10:algebra:identities}): fatore
        $x^2 - 10x + 25$, depois $9x^2 - 49$, depois --- combinando
        um produto notável com uma diferença de quadrados ---
        $x^2 + 6x + 9 - 4$.
  \item Resolva $(x + 1)(x + 5) = 0$ e
        $(3x - 7)(3x + 7) = 0$
        (\cref{thm:g10:algebra:zeroproduct}).
  \item O problema do escriba: dois números têm soma $10$ e
        produto $21$. Seguindo a Babilônia, escreva-os como
        $5 - t$ e $5 + t$ (simétricos em torno da metade da soma)
        e encontre $t$, depois os dois números.
  \item Mesmo método: soma $14$, produto $33$.
  \item A receita geral: para soma $s$ e produto $p$, mostre que
        os dois números são
        \[
        \frac s2 - t
        \quad\text{e}\quad
        \frac s2 + t,
        \qquad\text{onde}\quad
        t^2 = \left(\frac s2\right)^2 - p ,
        \]
        e enuncie a condição sobre $s$ e $p$ para que o problema
        tenha solução.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Completar o quadrado.}
\begin{enumerate}[resume]
  \item Expanda $(x - a)(x - b)$ e conclua: as soluções de
        $x^2 - sx + p = 0$ são exatamente os pares de números com
        soma $s$ e produto $p$. (Os escribas resolviam \omterm{def:g10:algebra:equation}{equações}
        do segundo grau sem o saber.)
  \item Verifique que $x^2 - 10x + 21 = (x - 5)^2 - 4$ e resolva
        $x^2 - 10x + 21 = 0$ fatorando o lado direito como uma
        diferença de quadrados. Compare com a questão 3.
  \item Resolva $x^2 + 6x - 7 = 0$ completando o quadrado.
  \item Complete o quadrado em $x^2 + 4x + 5$ e explique por que
        a \omterm{def:g10:algebra:equation}{equação} $x^2 + 4x + 5 = 0$ \emph{não} tem solução real.
        Enuncie o critério geral: uma vez escrita na forma
        $(x + h)^2 + k$, quando uma expressão do segundo grau se
        anula?
  \item O nome é literal. Na Bagdá do século IX, al-Khwarizmi
        desenhou $x^2 + 6x = 7$ como um quadrado de lado $x$ com
        dois retângulos $3 \times x$ colados a dois de seus lados
        e \emph{completou o quadrado} com o canto $3 \times 3$
        que faltava. Faça a figura e explique como ela transforma
        a \omterm{def:g10:algebra:equation}{equação} em $(x + 3)^2 = 16$ --- e resolva.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Sinais, quadros e duas promessas
antigas.}
\begin{enumerate}[resume]
  \item Monte o \omterm{def:g10:algebra:signtable}{quadro de sinais} de $(x - 3)(x + 2)$
        (\cref{def:g10:algebra:signtable}) e resolva
        $(x - 3)(x + 2) < 0$.
  \item Resolva $x^2 \leq 10x - 21$ (passe tudo para um lado e
        reaproveite a questão 7).
  \item Resolva $\dfrac{x - 1}{x + 2} \geq 0$ com um \omterm{def:g10:algebra:signtable}{quadro de
        sinais}, atento ao valor proibido.
  \item Uma promessa antiga cumprida: para o fazendeiro do
        problema de fim de semana sobre a cerca da rainha Dido,
        no volume do ensino fundamental, com $20$ m de cerca, a
        área de um curral $x \times (10 - x)$ satisfaz
        \[
        A(x) = x(10 - x) = 25 - (x - 5)^2 .
        \]
        Verifique a identidade e leia nela uma
        \emph{demonstração} do que o ano 6 só podia conjecturar:
        a área máxima e o único $x$ que a atinge.
  \item Outra: prove que $x + \dfrac1x \geq 2$ para todo
        $x > 0$, com igualdade só em $x = 1$. (Multiplique por
        $x$, agrupe, reconheça um quadrado --- a desigualdade
        entre as médias aritmética e geométrica do problema de
        fim de semana sobre as relações de Euclides, no volume do
        ensino fundamental, com nova roupagem.)
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A seção áurea.}
Divida um segmento de comprimento $1$ em uma parte grande $x$ e
uma parte pequena $1 - x$ de modo que
\emph{o todo esteja para a parte grande assim como a grande está
para a pequena}: $\frac{1}{x} = \frac{x}{1 - x}$.
\begin{enumerate}[resume]
  \item Mostre que a condição se escreve $x^2 + x - 1 = 0$.
  \item Resolva completando o quadrado e fique com a raiz que é
        um comprimento: $x = \frac{\sqrt5 - 1}{2} \approx 0.618$.
  \item A \emph{razão áurea} é
        $\varphi = \frac1x = \frac{2}{\sqrt5 - 1}$. Racionalize o
        denominador (multiplique numerador e denominador por
        $\sqrt5 + 1$, um produto notável em ação) e mostre que
        $\varphi = \frac{\sqrt5 + 1}{2} \approx 1.618$.
  \item Verifique \emph{exatamente} as duas identidades mágicas
        de $\varphi$:
        \[
        \varphi^2 = \varphi + 1,
        \qquad
        \frac{1}{\varphi} = \varphi - 1 .
        \]
  \item Final: explique por que $\varphi$ é irracional (o
        problema de fim de semana sobre irracionalidade do volume
        do ensino fundamental fornece o fato-chave sobre
        $\sqrt5$) e calcule com três casas decimais as razões
        $\frac53$, $\frac85$, $\frac{13}{8}$, $\frac{21}{13}$ de
        números de Hemachandra--Fibonacci consecutivos (o
        problema de fim de semana sobre contagem de ritmos, no
        mesmo volume). O que elas parecem perseguir? (A
        demonstração de que alcançam o alvo pertence aos
        capítulos sobre sequências.)
