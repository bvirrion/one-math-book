\chapter{Funções}\label{ch:g10:functions}

Uma \omterm{def:g10:functions:function}{função} é uma máquina que recebe um número e devolve um número, sempre
a mesma saída para a mesma entrada. As funções são os objetos centrais da
análise que você construirá nos próximos anos (do \cref{ch:g11:quad} em
diante); este capítulo monta o vocabulário --- \omterm{def:g10:functions:function}{domínio}, \omterm{def:g10:functions:function}{imagem}, \omterm{def:g10:functions:graph}{gráfico},
variação --- e treina a habilidade essencial de ler informação em um
\omterm{def:g10:functions:graph}{gráfico}.

\section{Vocabulário: imagens e pré-imagens}

\begin{definition}[Função]\label{def:g10:functions:function}
Seja $D$ um conjunto de \omterm{def:g10:numbers:sets}{números reais}. Uma \emph{função}\index{função} $f$
definida em $D$ associa a cada número $x \in D$ exatamente um \omterm{def:g10:numbers:sets}{número
real}, escrito $f(x)$ e chamado de \emph{imagem}\index{imagem} de $x$. O
conjunto $D$ é o \emph{domínio}\index{domínio} de $f$. Escrevemos
\[
f : x \longmapsto f(x).
\]
Se $f(x) = y$, então $x$ é uma \emph{pré-imagem}\index{pré-imagem} de $y$.
\end{definition}

\begin{remark}
Cada $x$ do \omterm{def:g10:functions:function}{domínio} tem exatamente uma \omterm{def:g10:functions:function}{imagem}, mas um número $y$ pode ter
várias \omterm{def:g10:functions:function}{pré-imagens}, ou nenhuma. Para $f(x) = x^2$: a \omterm{def:g10:functions:function}{imagem} de $3$ é $9$,
e $9$ tem duas \omterm{def:g10:functions:function}{pré-imagens}, $3$ e $-3$; o número $-4$ não tem
\omterm{def:g10:functions:function}{pré-imagem}.
\end{remark}

\begin{example}[Determinar um domínio]\label{ex:g10:functions:domain}
Quando uma \omterm{def:g10:functions:function}{função} é dada por uma fórmula, seu \omterm{def:g10:functions:function}{domínio} é o conjunto dos
$x$ para os quais a fórmula faz sentido.
\begin{itemize}
  \item $f(x) = 3x^2 - 5x + 1$ faz sentido para todo $x$: o \omterm{def:g10:functions:function}{domínio} é
        $\R$.
  \item $g(x) = \dfrac{1}{x - 2}$ exige $x - 2 \neq 0$: o \omterm{def:g10:functions:function}{domínio} é o
        conjunto de todos os reais exceto $2$.
  \item $h(x) = \sqrt{x - 3}$ exige $x - 3 \geq 0$: o \omterm{def:g10:functions:function}{domínio} é
        $\intco{3}{+\infty}$.
\end{itemize}
\end{example}

\begin{example}[Calcular imagens e pré-imagens]\label{ex:g10:functions:images}
Seja $f(x) = x^2 - 4x + 3$, definida em $\R$.

\emph{\omterm{def:g10:functions:function}{Imagem} de $5$:} substitua $x = 5$:
$f(5) = 25 - 20 + 3 = 8$.

\emph{\omterm{def:g10:functions:function}{Pré-imagens} de $3$:} resolva $f(x) = 3$:
\[
x^2 - 4x + 3 = 3
\quad\Longleftrightarrow\quad
x^2 - 4x = 0
\quad\Longleftrightarrow\quad
x(x - 4) = 0,
\]
logo as \omterm{def:g10:functions:function}{pré-imagens} de $3$ são $0$ e $4$.
\end{example}

\section{O gráfico de uma função}

\begin{definition}[Gráfico]\label{def:g10:functions:graph}
Em um sistema de coordenadas, o \emph{gráfico}\index{gráfico} (ou
\emph{curva}) de $f$ é o conjunto de todos os pontos $(x, f(x))$ com $x$
no \omterm{def:g10:functions:function}{domínio}. Em outras palavras, um ponto $(x, y)$ está no gráfico
exatamente quando $y = f(x)$.
\end{definition}

\end{axis}

para ler $f(3) = 1.25$.}
\end{omfigure}

\begin{method}[Ler um gráfico]\label{met:g10:functions:reading}
No \omterm{def:g10:functions:graph}{gráfico} de $f$:
\begin{enumerate}
  \item \emph{\omterm{def:g10:functions:function}{imagem} de $a$}: siga verticalmente de $x = a$ no eixo
        horizontal até a curva e depois horizontalmente até o eixo
        vertical; o valor lido ali é $f(a)$;
  \item \emph{\omterm{def:g10:functions:function}{pré-imagens} de $b$}: trace a reta horizontal $y = b$; as
        \omterm{def:g10:functions:function}{pré-imagens} são as abscissas de todos os seus pontos de
        \omterm{def:g10:numbers:interunion}{interseção} com a curva;
  \item \emph{soluções de $f(x) = k$}: as mesmas que as \omterm{def:g10:functions:function}{pré-imagens} de
        $k$;
  \item \emph{soluções de $f(x) \leq k$}: os $x$ para os quais a curva
        está sobre a reta $y = k$ ou abaixo dela.
\end{enumerate}
\end{method}

  \draw[dashed] (axis cs:2,2) -- (axis cs:2,0);
\end{axis}
\end{tikzpicture}

{\small Resolver $f(x) = k$ graficamente: as soluções $x_1$ e $x_2$ são as
abscissas dos pontos de \omterm{def:g10:numbers:interunion}{interseção} da curva com a reta horizontal
$y = k$. Aqui $f(x) \leq k$ vale em $\intcc{x_1}{x_2}$, onde a curva está
abaixo da reta.}
\end{omfigure}

\begin{example}\label{ex:g10:functions:membership}
O ponto $A(2, 5)$ está no \omterm{def:g10:functions:graph}{gráfico} de $f(x) = x^2 + 1$? Calcule
$f(2) = 4 + 1 = 5$: sim, pois $f(2)$ é igual à ordenada de $A$. O ponto
$B(3, 8)$ não está no \omterm{def:g10:functions:graph}{gráfico}, porque $f(3) = 10 \neq 8$.
\end{example}

\section{Variação de uma função}

\begin{definition}[Crescente, decrescente]\label{def:g10:functions:variations}
Seja $f$ definida em um \omterm{def:g10:numbers:interval}{intervalo} $I$.
\begin{itemize}
  \item $f$ é \emph{crescente}\index{função crescente} em $I$ quando,
        para todos $u, v \in I$, $u < v$ implica $f(u) \leq f(v)$: as
        saídas crescem junto com as entradas, e o \omterm{def:g10:functions:graph}{gráfico} sobe da
        esquerda para a direita.
  \item $f$ é \emph{decrescente}\index{função decrescente} em $I$ quando
        $u < v$ implica $f(u) \geq f(v)$: o \omterm{def:g10:functions:graph}{gráfico} desce da esquerda
        para a direita.
\end{itemize}
Com desigualdades estritas $f(u) < f(v)$ (resp.\ $>$) dizemos
\emph{estritamente} crescente (resp.\ decrescente).
\end{definition}

\begin{definition}[Quadro de variação]\label{def:g10:functions:table}
Um \emph{quadro de variação}\index{quadro de variação} resume em quais
\omterm{def:g10:numbers:interval}{intervalos} $f$ cresce e decresce, com as setas $\nearrow$ e $\searrow$, e
registra os valores de $f$ nos pontos de virada.
\end{definition}

\begin{example}\label{ex:g10:functions:table}
Considere a \omterm{def:g10:functions:function}{função} cujo \omterm{def:g10:functions:graph}{gráfico} está abaixo, em $\intcc{-2}{4}$.

  xmin=-2.8, xmax=4.6, ymin=-2.4, ymax=3.8,
  xtick={-2,1,4}, ytick={-1.5,3}, yticklabels={$-1.5$,$3$}]
  \addplot[omDef, thick, domain=-2:4, samples=120] {-0.5*(x-1)^2 + 3};
  \fill (axis cs:-2,-1.5) circle (1.5pt);
  \fill (axis cs:4,-1.5) circle (1.5pt);
  \fill (axis cs:1,3) circle (1.5pt);
\end{axis}
\end{tikzpicture}

{\small Uma \omterm{def:g10:functions:function}{função} definida em $\intcc{-2}{4}$, \omterm{def:g10:functions:variations}{crescente} e depois
\omterm{def:g10:functions:variations}{decrescente}.}
\end{omfigure}

Lendo o \omterm{def:g10:functions:graph}{gráfico}: $f$ cresce de $f(-2) = -1.5$ até seu \omterm{def:g10:functions:extrema}{máximo} $f(1) = 3$ e
depois decresce até $f(4) = -1.5$. Seu \omterm{def:g10:functions:table}{quadro de variação} é
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccc}
$x$ & $-2$ & & $1$ & & $4$ \\
\hline
$f$ & $-1.5$ & $\nearrow$ & $3$ & $\searrow$ & $-1.5$ \\
\end{tabular}
\end{center}
\end{example}

\begin{example}[Demonstrar uma variação]\label{ex:g10:functions:proof}
Mostre que $f(x) = 3x + 1$ é estritamente \omterm{def:g10:functions:variations}{crescente} em $\R$. Tome
quaisquer $u < v$ e compare as \omterm{def:g10:functions:function}{imagens}:
\[
f(v) - f(u) = (3v + 1) - (3u + 1) = 3(v - u) > 0,
\]
pois $v - u > 0$. Logo $f(u) < f(v)$: a \omterm{def:g10:functions:function}{função} é estritamente \omterm{def:g10:functions:variations}{crescente}. O
mesmo cálculo com coeficiente angular negativo, por exemplo
$g(x) = -2x + 5$, dá $g(v) - g(u) = -2(v-u) < 0$: estritamente
\omterm{def:g10:functions:variations}{decrescente}.
\end{example}

\section{Extremos}

\begin{definition}[Máximo, mínimo]\label{def:g10:functions:extrema}
Seja $f$ definida em $D$ e seja $a \in D$. O valor $f(a)$ é o
\emph{máximo}\index{máximo} de $f$ em $D$ quando $f(x) \leq f(a)$ para
todo $x \in D$; é o \emph{mínimo}\index{mínimo} quando $f(x) \geq f(a)$
para todo $x \in D$. No \omterm{def:g10:functions:graph}{gráfico}, são o ponto mais alto e o ponto mais
baixo da curva.
\end{definition}

\begin{example}\label{ex:g10:functions:extremum}
Mostre que $f(x) = (x - 1)^2 + 2$ tem \omterm{def:g10:functions:extrema}{mínimo} $2$ em $\R$, atingido em
$x = 1$. Para todo $x$, o quadrado $(x-1)^2$ é $\geq 0$, logo
$f(x) = (x-1)^2 + 2 \geq 2$; e $f(1) = 0 + 2 = 2$, de modo que o valor $2$
é de fato atingido. Os dois fatos juntos provam que $2$ é o \omterm{def:g10:functions:extrema}{mínimo}.
\end{example}

\begin{omfigure}
\begin{tikzpicture}
\begin{axis}[omaxis,
  width=7.6cm, height=5.2cm,
  xmin=-1.8, xmax=3.8, ymin=-0.9, ymax=6.4,

  \addplot[omDef, thick, domain=-1.0:3.0, samples=100] {(x-1)^2 + 2};

tracejada $y = 2$ e a toca em $x = 1$: o \omterm{def:g10:functions:extrema}{mínimo} de $f$ é $2$.}
\end{omfigure}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g10:functions:1}
Seja $f(x) = 2x^2 - 3x + 1$. Calcule as \omterm{def:g10:functions:function}{imagens} de $0$, $2$, $-1$ e
$\frac12$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:2}
Dê o \omterm{def:g10:functions:function}{domínio} de cada \omterm{def:g10:functions:function}{função}:
\[
f(x) = 5x - 2, \qquad
g(x) = \frac{3}{x + 4}, \qquad
h(x) = \sqrt{2x - 6}, \qquad
k(x) = \frac{1}{x^2 + 1}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:3}
Seja $f(x) = x^2 - 2x$. Encontre todas as \omterm{def:g10:functions:function}{pré-imagens} de $0$, de $3$ e de
$-1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:4}
O ponto $A(-1, 4)$ pertence ao \omterm{def:g10:functions:graph}{gráfico} de $f(x) = x^2 - 2x + 1$? E o
ponto $B(2, 1)$? Justifique com um cálculo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:functions:5}
Uma \omterm{def:g10:functions:function}{função} $g$ definida em $\intcc{-3}{5}$ tem o \omterm{def:g10:functions:table}{quadro de variação}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{c|ccccccc}
$x$ & $-3$ & & $0$ & & $3$ & & $5$ \\
\hline
$g$ & $1$ & $\searrow$ & $-2$ & $\nearrow$ & $4$ & $\searrow$ & $0$ \\
\end{tabular}
\end{center}
\begin{enumerate}
  \item Quais são o \omterm{def:g10:functions:extrema}{máximo} e o \omterm{def:g10:functions:extrema}{mínimo} de $g$ em $\intcc{-3}{5}$?
  \item Compare $g(-1)$ e $g(-0.5)$ sem conhecer seus valores.
  \item Quantas soluções, no máximo, a \omterm{def:g10:algebra:equation}{equação} $g(x) = 0$ tem em cada
        \omterm{def:g10:numbers:interval}{intervalo} do quadro?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:6}
Seja $f(x) = -2x + 7$. Mostre, comparando $f(u)$ e $f(v)$ para $u < v$,
que $f$ é estritamente \omterm{def:g10:functions:variations}{decrescente} em $\R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:7}
Mostre que a \omterm{def:g10:functions:function}{função} $f(x) = x^2 + 6x + 11$ tem um \omterm{def:g10:functions:extrema}{mínimo} em $\R$ e dê seu
valor e onde ele é atingido. (Sugestão: escreva $f(x) = (x + 3)^2 + c$
para a constante $c$ adequada.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:8}
Um jardim retangular tem perímetro $40$ m. Seja $x$ a sua largura, em
metros.
\begin{enumerate}
  \item Exprima o comprimento e, em seguida, a área $A(x)$ em \omterm{def:g10:functions:function}{função} de
        $x$. Para quais $x$ isso faz sentido?
  \item Calcule $A(4)$, $A(8)$, $A(10)$, $A(12)$ e $A(16)$. O que você
        conjectura sobre a forma de maior área?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:functions:9}
Seja $f(x) = \dfrac{1}{x^2 + 1}$, definida em $\R$.
\begin{enumerate}
  \item Mostre que $0 < f(x) \leq 1$ para todo real $x$.
  \item $f$ tem \omterm{def:g10:functions:extrema}{máximo} em $\R$? E \omterm{def:g10:functions:extrema}{mínimo}? Justifique.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:functions:10}
Seja $f(x) = x^2$ em $\R$.
\begin{enumerate}
  \item Sejam $0 \leq u < v$. Fatore $f(v) - f(u)$ e deduza que $f$ é
        estritamente \omterm{def:g10:functions:variations}{crescente} em $\intco{0}{+\infty}$.
  \item Adapte o argumento para mostrar que $f$ é estritamente
        \omterm{def:g10:functions:variations}{decrescente} em $\intoc{-\infty}{0}$.
\end{enumerate}
\end{exercise}

\section{Problema: Máquinas alimentadas com a própria saída}

\begin{problem}[{Problema de fim de semana --- pontos fixos, iteração e
três máquinas famosas: a de Heron, a que eleva ao quadrado e a
granizo, ainda sem solução}]
\label{pb:g10:functions:1}
Uma \omterm{def:g10:functions:function}{função} é uma máquina: entra um número, sai um número. As
experiências mais interessantes alimentam a máquina com a
\emph{própria saída}, repetidamente --- e então duas perguntas
dominam: o processo se acomoda em algum lugar (um \emph{\omterm{pb:g10:functions:1}{ponto
fixo}})? E como a variação da máquina o conduz até lá? Uma das
máquinas deste problema poli raízes quadradas há dois mil anos;
outra guarda o mais famoso problema em aberto que a matemática
elementar tem a oferecer.

\medskip
\noindent\textbf{Parte I --- A máquina de Heron.}
Considere a \omterm{def:g10:functions:function}{função}
\[
f(x) = \frac12\left(x + \frac2x\right).
\]
\begin{enumerate}
  \item Dê o \omterm{def:g10:functions:function}{domínio} de $f$ e calcule $f(1)$, $f(2)$ e
        $f\!\left(\frac32\right)$ como frações exatas.
  \item Encontre todas as \omterm{def:g10:functions:function}{pré-imagens} de $\frac32$: resolva
        $f(x) = \frac32$ (elimine o denominador e use os métodos
        do \cref{pb:g10:algebra:1}).
  \item Um \emph{ponto fixo}\index{ponto fixo} de $f$ é um
        número que a máquina não altera: $f(x) = x$. Encontre os
        dois \omterm{pb:g10:functions:1}{pontos fixos} de $f$. Que velho conhecido guarda o
        positivo?
  \item Partindo de $x = 1$, alimente a máquina com sua própria
        saída três vezes e registre os resultados exatos. Que
        sequência do problema de fim de semana sobre
        irracionalidade, no volume do ensino fundamental, você
        reconstruiu --- e o que era a ``receita de Heron'', na
        linguagem das funções?
  \item Numa mesma figura, esboce o \omterm{def:g10:functions:graph}{gráfico} de $f$ (para $x > 0$)
        e a reta diagonal $y = x$. Onde estão os \omterm{pb:g10:functions:1}{pontos fixos}
        nessa figura? (Compare com o problema de fim de semana
        sobre os dois termômetros, no volume do ensino
        fundamental: a temperatura que se lê igual nas duas
        escalas era a mesma ideia.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Por que a máquina não erra.}
\begin{enumerate}[resume]
  \item Para $0 < u < v$, mostre, reduzindo ao mesmo
        denominador, que
        \[
        f(v) - f(u) = \frac{(v - u)(uv - 2)}{2uv} ,
        \]
        e deduza a variação de $f$ em $\intoo{0}{+\infty}$:
        \omterm{def:g10:functions:variations}{decrescente} até um ponto e \omterm{def:g10:functions:variations}{crescente} depois --- qual
        ponto? (Mesma técnica do
        \cref{exo:g10:functions:10}.)
  \item Deduza que em $\intoo{0}{+\infty}$ a \omterm{def:g10:functions:function}{função} $f$ tem um
        \omterm{def:g10:functions:extrema}{mínimo}, que vale exatamente $\sqrt2$, atingido em
        $\sqrt2$. O que isso diz sobre \emph{toda} saída da
        máquina (com entradas positivas)?
  \item Mostre que, se $x > \sqrt2$, então
        $\sqrt2 < f(x) < x$: um palpite grande demais é sempre
        melhorado, nunca ultrapassado. (Para $f(x) < x$, compare
        $\frac2x$ com $x$.)
  \item Reúna as questões 6--8 no \omterm{def:g10:functions:table}{quadro de variação} de $f$ em
        $\intoo{0}{+\infty}$
        (\cref{def:g10:functions:table}), \omterm{def:g10:functions:extrema}{mínimo} incluído.
  \item Uma segunda máquina: $g(x) = x^2$. Seus \omterm{pb:g10:functions:1}{pontos fixos} são
        $0$ e $1$ (verifique). Itere $g$ quatro vezes a partir de
        $x = 0.9$ e depois quatro vezes a partir de $x = 1.1$
        (calculadora, três casas decimais). Um dos \omterm{pb:g10:functions:1}{pontos fixos}
        atrai, o outro repele: qual é qual?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A máquina granizo.}
Nos \omterm{def:g10:numbers:sets}{números inteiros} positivos, defina: $h(n) = \frac n2$ se $n$
é par e $h(n) = 3n + 1$ se $n$ é ímpar. Os números saltam sob $h$
como pedras de granizo dentro de uma nuvem de tempestade.
\begin{enumerate}[resume]
  \item Calcule o voo completo de $7$ sob $h$, até chegar a $1$.
        Quantos passos ele leva e a que altura chega?
  \item Comece o voo de $27$ e calcule dez passos. (Seu voo
        completo dura $111$ passos e tem pico em $9\,232$ ---
        partindo de $27$.) Que lição $7$ e $27$ ensinam sobre
        regras simples?
  \item Explique por que todo voo que atinge uma potência de $2$
        despenca direto até $1$ e o que acontece em $1$ (calcule
        $h(1)$, $h(4)$, $h(2)$). A \omterm{def:g10:functions:function}{função} $h$ tem algum \omterm{pb:g10:functions:1}{ponto
        fixo} entre os inteiros positivos? O que o fim de todo voo
        observado forma em vez disso?
  \item A \emph{conjectura de Collatz} afirma que todo número
        inicial acaba chegando a $1$. Os computadores já a
        verificaram muito além de $10^{20}$; nenhum ser humano a
        demonstrou. Que dois tipos de descoberta a
        \emph{refutariam}? E por que a verificação em massa
        ainda assim não encerra o caso (o problema de fim de
        semana sobre o oráculo dos palitos, no volume do ensino
        fundamental, já tocou esse sino)?
  \item ``A matemática ainda não está madura para tais
        questões'', disse Paul Erd\H{o}s a respeito desta
        conjectura. Em uma frase: o que separa a máquina granizo
        da de Heron, em que as questões 6--8 resolveram tudo?
\end{enumerate}

\noindent\textbf{Parte IV --- O manual do usuário de máquinas.}
\begin{enumerate}[resume]
  \item \omterm{def:g10:functions:function}{Domínios} como \omterm{def:g10:numbers:interval}{intervalos} (a linguagem do
        \cref{pb:g10:numbers:1}): dê os \omterm{def:g10:functions:function}{domínios} de
        $x \mapsto \sqrt{x - 3}$, \
        $x \mapsto \frac{1}{x^2 - 9}$, \
        $x \mapsto \sqrt{9 - x^2}$.
  \item De volta ao $f$ de Heron: encontre as \omterm{def:g10:functions:function}{pré-imagens}
        exatas de $3$ (resolva $f(x) = 3$ completando o
        quadrado).
  \item Uma pedra é lançada para cima: sua altura é
        $H(t) = 20t - 5t^2$ metros após $t$ segundos. Em que
        \omterm{def:g10:numbers:interval}{intervalo} de tempo a fórmula faz sentido físico?
        Complete o quadrado para achar a altura máxima e o
        instante em que ocorre
        (\cref{def:g10:functions:extrema}) e dê o \omterm{def:g10:functions:table}{quadro de
        variação}.
  \item Durante quanto tempo a pedra fica a pelo menos $15$ m de
        altura? (Resolva $H(t) \geq 15$ com um \omterm{def:g10:algebra:signtable}{quadro de sinais},
        \cref{met:g10:algebra:signtable}.)
  \item Final: este problema pôs quatro máquinas em movimento ---
        a de Heron, a que eleva ao quadrado, a granizo e a altura
        da pedra. Em duas ou três frases, diga o que toda \omterm{def:g10:functions:function}{função}
        tem (um \omterm{def:g10:functions:function}{domínio}, uma regra, um \omterm{def:g10:functions:graph}{gráfico}) e as duas grandes
        perguntas que este problema fez a cada uma delas (onde a
        iteração se acomoda? como a \omterm{def:g10:functions:function}{função} varia?) ---
        perguntas que os próximos capítulos retomam uma a uma.
\end{enumerate}
