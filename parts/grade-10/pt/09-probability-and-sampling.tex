\chapter{Probabilidade e amostragem}\label{ch:g10:proba}

A \omterm{def:g10:proba:distribution}{probabilidade} atribui números ao acaso. Partindo de experimentos com
resultados igualmente prováveis --- dados, moedas, cartas ---, este
capítulo constrói as regras básicas para calcular \omterm{def:g10:proba:distribution}{probabilidades} de
\omterm{def:g10:proba:events}{eventos}, apresenta os \omterm{met:g10:proba:tree}{diagramas de árvore} para experimentos em duas
etapas e termina com a ligação entre \omterm{def:g10:proba:distribution}{probabilidades} e \omterm{def:g10:stats:series}{frequências}
observadas. A teoria continua no \cref{ch:g11:prob}.

\section{Experimentos aleatórios e eventos}

\begin{definition}[Resultados e eventos]\label{def:g10:proba:events}
Um \emph{experimento aleatório}\index{experimento aleatório} é um
experimento cujo resultado não pode ser previsto (lançar um dado). O
conjunto de todos os resultados possíveis é o \emph{espaço
amostral}\index{espaço amostral} $\Omega = \{\omega_1, \dots,
\omega_n\}$; seus elementos são os \emph{resultados}. Um
\emph{evento}\index{evento} é um conjunto de resultados; um evento
\emph{ocorre} quando o resultado do experimento pertence a ele.
\end{definition}

\begin{definition}[Distribuição de probabilidade]\label{def:g10:proba:distribution}
Atribuir a cada resultado $\omega_i$ um número $p_i \geq 0$, com
$p_1 + \dots + p_n = 1$, define uma
\emph{probabilidade}\index{probabilidade}: a probabilidade $\P(A)$ de um
\omterm{def:g10:proba:events}{evento} $A$ é a soma dos $p_i$ dos resultados contidos em $A$. Em
particular, $\P(\Omega) = 1$ e o \omterm{def:g10:proba:events}{evento} impossível $\varnothing$ tem
probabilidade $0$.
\end{definition}

\begin{proposition}[Resultados igualmente prováveis]\label{prop:g10:proba:uniform}
Se os $n$ resultados são igualmente prováveis (cada $p_i = \frac1n$),
então, para todo \omterm{def:g10:proba:events}{evento} $A$:
\[
\P(A) = \frac{\text{número de resultados em } A}{n}
= \frac{\text{casos favoráveis}}{\text{casos possíveis}} .
\]
\end{proposition}

\begin{proof}
$A$ contém um certo número $k$ de resultados, cada um de \omterm{def:g10:proba:distribution}{probabilidade}
$\frac1n$, logo $\P(A) = \frac kn$.
\end{proof}

\begin{example}\label{ex:g10:proba:die}
Lance um dado honesto: $\Omega = \{1, 2, 3, 4, 5, 6\}$, cada resultado
com \omterm{def:g10:proba:distribution}{probabilidade} $\frac16$. O \omterm{def:g10:proba:events}{evento} $A$: ``o resultado é par'' é
$A = \{2, 4, 6\}$, logo $\P(A) = \frac36 = \frac12$. O \omterm{def:g10:proba:events}{evento} $B$: ``o
resultado é pelo menos $5$'' é $B = \{5, 6\}$, logo
$\P(B) = \frac26 = \frac13$.
\end{example}

\section{Combinar eventos}

\begin{definition}[Interseção, união, complementar]\label{def:g10:proba:operations}
Sejam $A$ e $B$ \omterm{def:g10:proba:events}{eventos}.
\begin{itemize}
  \item $A \cap B$ (``$A$ e $B$'') ocorre quando os dois ocorrem;
  \item $A \cup B$ (``$A$ ou $B$'') ocorre quando pelo menos um ocorre;
  \item o \emph{complementar}\index{complementar} $\bar A$ (``não $A$'')
        ocorre exatamente quando $A$ não ocorre;
  \item $A$ e $B$ são \emph{incompatíveis}\index{incompatíveis} quando
        não podem ocorrer juntos: $A \cap B = \varnothing$.
\end{itemize}
\end{definition}

\begin{theorem}[Regras da adição]\label{thm:g10:proba:addition}
Para quaisquer \omterm{def:g10:proba:events}{eventos} $A$ e $B$:
\[
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B),
\qquad
\P(\bar A) = 1 - \P(A) .
\]
Para \omterm{def:g10:proba:events}{eventos} \omterm{def:g10:proba:operations}{incompatíveis}, a primeira regra simplifica-se para
$\P(A \cup B) = \P(A) + \P(B)$.
\end{theorem}

\begin{proof}
Somar $\P(A) + \P(B)$ acrescenta uma vez a \omterm{def:g10:proba:distribution}{probabilidade} de cada
resultado de $A \cup B$, \emph{exceto} os de $A \cap B$, que são contados
duas vezes --- uma em $A$ e outra em $B$. Subtrair $\P(A \cap B)$ repara
a contagem dupla. Para o \omterm{def:g10:proba:operations}{complementar}: $A$ e $\bar A$ são \omterm{def:g10:proba:operations}{incompatíveis} e
$A \cup \bar A = \Omega$, logo $\P(A) + \P(\bar A) = \P(\Omega) = 1$.
\end{proof}

\begin{example}\label{ex:g10:proba:cards}
Retire uma carta de um baralho comum de $52$ cartas. Sejam $A$: ``a carta
é de copas'' ($13$ cartas) e $B$: ``a carta é um rei'' ($4$ cartas).
Então $A \cap B$ é ``o rei de copas'' ($1$ carta), e
\[
\P(A \cup B) = \frac{13}{52} + \frac{4}{52} - \frac{1}{52} = \frac{16}{52}
= \frac{4}{13}.
\]
A regra do \omterm{def:g10:proba:operations}{complementar} costuma ser o caminho mais rápido: a
\omterm{def:g10:proba:distribution}{probabilidade} de a carta \emph{não} ser de copas é
$1 - \frac{13}{52} = \frac34$.
\end{example}

\section{Diagramas de árvore}

\begin{method}[Experimentos em duas etapas]\label{met:g10:proba:tree}
Para um experimento realizado em duas etapas:
\begin{enumerate}
  \item desenhe uma \emph{árvore}\index{diagrama de árvore}: um ramo por
        resultado possível da primeira etapa e, a partir de cada um
        deles, um ramo por resultado da segunda etapa, escrevendo a
        \omterm{def:g10:proba:distribution}{probabilidade} em cada ramo (os ramos que saem de um mesmo nó
        devem somar $1$);
  \item a \omterm{def:g10:proba:distribution}{probabilidade} de um \emph{caminho} (uma folha da \omterm{met:g10:proba:tree}{árvore}) é o
        produto das \omterm{def:g10:proba:distribution}{probabilidades} ao longo de seus ramos;
  \item a \omterm{def:g10:proba:distribution}{probabilidade} de um \omterm{def:g10:proba:events}{evento} é a soma das \omterm{def:g10:proba:distribution}{probabilidades} dos
        caminhos que o realizam.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:proba:tree}
Uma urna contém $3$ bolas vermelhas e $2$ bolas azuis. Retire uma bola,
devolva-a e retire outra. Cada retirada dá vermelha (V) com \omterm{def:g10:proba:distribution}{probabilidade}
$\frac35$ e azul (A) com \omterm{def:g10:proba:distribution}{probabilidade} $\frac25$.

quatro caminhos somam $1$.}
\end{omfigure}

A \omterm{def:g10:proba:distribution}{probabilidade} de obter duas bolas da mesma cor é
$\P(\text{VV}) + \P(\text{AA}) = \frac{9}{25} + \frac{4}{25} =
\frac{13}{25}$; a \omterm{def:g10:proba:distribution}{probabilidade} de pelo menos uma azul é
$1 - \P(\text{VV}) = 1 - \frac{9}{25} = \frac{16}{25}$ (regra do
\omterm{def:g10:proba:operations}{complementar}).
\end{example}

\begin{remark}
Sem reposição, as \omterm{def:g10:proba:distribution}{probabilidades} da segunda retirada dependem da
primeira: com a urna acima, depois de sair uma bola vermelha restam
apenas $2$ vermelhas e $2$ azuis, de modo que o segundo ramo carrega
$\frac24$ e $\frac24$. O método da \omterm{met:g10:proba:tree}{árvore} funciona sem alteração; apenas
mudam os números nos ramos do segundo nível.
\end{remark}

\section{Frequências e amostragem}

Quando um \omterm{def:g10:proba:events}{experimento aleatório} é repetido muitas vezes, a \omterm{def:g10:stats:series}{frequência}
observada de um \omterm{def:g10:proba:events}{evento} se aproxima da sua \omterm{def:g10:proba:distribution}{probabilidade} --- é por isso
que a \omterm{def:g10:proba:distribution}{probabilidade} serve para descrever o mundo real.

\begin{definition}[Amostra]\label{def:g10:proba:sample}
Uma \emph{amostra}\index{amostra} de tamanho $n$ é o resultado de repetir
o mesmo experimento $n$ vezes de forma independente. A \emph{\omterm{def:g10:stats:series}{frequência}
observada} de um \omterm{def:g10:proba:events}{evento} na amostra é o número de ocorrências dividido por
$n$.
\end{definition}

\begin{example}\label{ex:g10:proba:sampling}
Uma moeda honesta dá cara com \omterm{def:g10:proba:distribution}{probabilidade} $0.5$. Lançá-la $100$ vezes
raramente dá exatamente $50$ caras: as \omterm{def:g10:proba:sample}{amostras} \emph{flutuam}. Séries
típicas de $100$ lançamentos dão \omterm{def:g10:stats:series}{frequências} como $0.46$, $0.53$, $0.49$
--- próximas de $0.5$, mas não iguais. \omterm{def:g10:proba:sample}{Amostras} maiores flutuam menos:
com $10\,000$ lançamentos, as \omterm{def:g10:stats:series}{frequências} observadas costumam ficar a
menos de cerca de $0.01$ de $0.5$.
\end{example}

\begin{remark}[Intervalo de flutuação]
Uma regra prática útil, justificada no \cref{ch:g11:binom}: para uma
\omterm{def:g10:proba:sample}{amostra} de tamanho $n$ de um \omterm{def:g10:proba:events}{evento} de \omterm{def:g10:proba:distribution}{probabilidade} $p$ (com $n$
razoavelmente grande), a \omterm{def:g10:stats:series}{frequência} observada fica no \omterm{def:g10:numbers:interval}{intervalo}
\[
\intcc{p - \frac{1}{\sqrt n}}{\,p + \frac{1}{\sqrt n}}
\]
em cerca de $95\%$ das \omterm{def:g10:proba:sample}{amostras}. Se uma \omterm{def:g10:stats:series}{frequência} observada cair bem
fora desse \omterm{def:g10:numbers:interval}{intervalo}, desconfie do valor de $p$: esse é o ponto de
partida dos testes estatísticos.
\end{remark}

\begin{example}\label{ex:g10:proba:test}
Uma fábrica afirma que apenas $10\%$ de suas peças são defeituosas
($p = 0.1$). Em uma \omterm{def:g10:proba:sample}{amostra} de $n = 400$ peças, $64$ são defeituosas:
\omterm{def:g10:stats:series}{frequência} observada $\frac{64}{400} = 0.16$. O \omterm{def:g10:numbers:interval}{intervalo} de flutuação é
$\intcc{0.1 - \frac{1}{20}}{0.1 + \frac{1}{20}} = \intcc{0.05}{0.15}$, e
$0.16$ cai fora dele: a \omterm{def:g10:proba:sample}{amostra} lança séria dúvida sobre a afirmação.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g10:proba:1}
Um dado honesto é lançado. Calcule a \omterm{def:g10:proba:distribution}{probabilidade} de cada \omterm{def:g10:proba:events}{evento}:
``obter um $6$''; ``obter um número ímpar''; ``obter no máximo $4$'';
``obter um $7$''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:2}
Um saco contém $5$ bolinhas verdes, $3$ amarelas e $2$ pretas; retira-se
uma bolinha ao acaso. Calcule a \omterm{def:g10:proba:distribution}{probabilidade} de ela ser verde; de não
ser preta; de ser verde ou amarela.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:3}
Em uma turma de $30$ alunos, $18$ estudam espanhol, $10$ estudam alemão e
$4$ estudam os dois. Escolhe-se um aluno ao acaso. Usando a regra da
adição, calcule a \omterm{def:g10:proba:distribution}{probabilidade} de esse aluno estudar pelo menos uma das
duas línguas e depois a \omterm{def:g10:proba:distribution}{probabilidade} de não estudar nenhuma.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:proba:4}
Uma moeda honesta é lançada duas vezes. Desenhe a \omterm{met:g10:proba:tree}{árvore} do experimento e
calcule a \omterm{def:g10:proba:distribution}{probabilidade} de obter duas caras; exatamente uma cara; pelo
menos uma cara.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:5}
Uma urna contém $4$ bolas vermelhas e $6$ brancas. Duas bolas são
retiradas uma após a outra, \emph{sem} reposição. Desenhe a \omterm{met:g10:proba:tree}{árvore} com as
\omterm{def:g10:proba:distribution}{probabilidades} corretas no segundo nível e calcule a \omterm{def:g10:proba:distribution}{probabilidade} de
retirar duas bolas vermelhas e a de retirar duas bolas de cores
diferentes.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:6}
Dois dados honestos são lançados e sua soma é registrada.
\begin{enumerate}
  \item Explique por que o \omterm{def:g10:proba:events}{espaço amostral} pode ser tomado como os $36$
        pares igualmente prováveis $(1,1), (1,2), \dots, (6,6)$.
  \item Calcule a \omterm{def:g10:proba:distribution}{probabilidade} de a soma ser $7$; de ser $12$; de ser no
        máximo $4$.
  \item Qual soma é a mais provável?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:7}
Uma questão de múltipla escolha oferece $4$ respostas, uma correta. Um
aluno responde a $2$ questões independentes desse tipo completamente ao
acaso. Calcule a \omterm{def:g10:proba:distribution}{probabilidade} de acertar as duas, de acertar exatamente
uma e de não acertar nenhuma. Verifique que as três \omterm{def:g10:proba:distribution}{probabilidades} somam
$1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:proba:8}
Um político afirma ter $50\%$ de aprovação. Em uma pesquisa com $n = 900$
eleitores escolhidos ao acaso, $396$ aprovam.
\begin{enumerate}
  \item Calcule a \omterm{def:g10:stats:series}{frequência} observada e o \omterm{def:g10:numbers:interval}{intervalo} de flutuação em
        torno de $p = 0.5$ para $n = 900$.
  \item A pesquisa lança dúvida sobre a afirmação?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:proba:9}
Um jogo custa $2$ para ser jogado: você lança um dado honesto e ganha $6$
se sair $6$, $3$ se sair $5$, e nada nos demais casos.
\begin{enumerate}
  \item Calcule a \omterm{def:g10:proba:distribution}{probabilidade} de cada quantia ganha ($6$, $3$, $0$).
  \item Em $600$ partidas, quantos resultados de cada tipo você espera?
        Calcule o ganho e a perda totais esperados: vale a pena jogar?
\end{enumerate}
\end{exercise}

\section{Problema: As três portas que enganaram mil matemáticos}

\begin{problem}[{Problema de fim de semana --- o paradoxo de Monty
Hall e seus primos: como a informação muda a probabilidade e como as
amostras separam afirmações de ilusões}]
\label{pb:g10:proba:1}
Em 1990, a colunista Marilyn vos Savant publicou um enigma sobre um
apresentador de programa de auditório, três portas, um carro e dois
bodes --- e o respondeu corretamente. Dez mil leitores, entre eles
centenas de doutores em matemática, escreveram para insistir que
ela estava errada. Este problema arma você com \omterm{met:g10:proba:tree}{árvores}
(\cref{met:g10:proba:tree}), \omterm{def:g10:proba:operations}{complementares} e as regras da adição
(\cref{thm:g10:proba:addition}), permite que você mesmo encerre a
discussão e termina com a ferramenta que arbitra na prática toda
disputa desse tipo: a \omterm{def:g10:proba:sample}{amostra} que flutua.

\medskip
\noindent\textbf{Parte I --- As regras, aquecidas.}
\begin{enumerate}
  \item Uma carta de um baralho comum de $52$ cartas: calcule
        $P(\text{rei ou copas})$ com a regra da adição --- e diga
        por que $\frac{4}{52} + \frac{13}{52}$ sozinho está
        errado.
  \item Três lançamentos de uma moeda honesta: calcule
        $P(\text{pelo menos uma cara})$ pelo \omterm{def:g10:proba:operations}{complementar}.
  \item Um saco tem $3$ bolinhas vermelhas e $2$ azuis; duas são
        retiradas sem reposição. Com uma \omterm{met:g10:proba:tree}{árvore}, calcule
        $P(\text{duas vermelhas})$ e
        $P(\text{uma de cada cor})$.
  \item Complete a distribuição da questão 3 com
        $P(\text{duas azuis})$ e verifique que as três
        \omterm{def:g10:proba:distribution}{probabilidades} somam $1$.
  \item Alguém informa, para dois \omterm{def:g10:proba:events}{eventos} \emph{\omterm{def:g10:proba:operations}{incompatíveis}},
        $P(A) = 0.6$ e $P(B) = 0.5$. Condene-o com uma linha de
        aritmética.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- As três portas.}
O jogo: um carro se esconde atrás de uma de três portas, e há
bodes atrás das outras duas. Você escolhe uma porta (digamos, a
porta 1). O apresentador --- que \emph{sabe} onde está o carro ---
abre uma das outras duas portas, revelando sempre um bode, e lhe
oferece a chance de trocar para a porta fechada restante.
\begin{enumerate}[resume]
  \item Antes de calcular qualquer coisa: trocar ajuda, atrapalha
        ou é indiferente? Anote sua resposta intuitiva --- com
        honestidade.
  \item Agora a \omterm{met:g10:proba:tree}{árvore}, por casos segundo a verdadeira posição do
        carro (cada um com $\frac13$): se o carro está atrás da
        sua porta, trocar faz perder; se está atrás de qualquer
        das outras duas, o que o apresentador é obrigado a fazer,
        e o que trocar ganha então? Calcule
        $P(\text{ganhar trocando})$.
  \item A versão de uma linha: trocar ganha exatamente quando a
        sua primeira escolha estava errada. Conclua de novo.
  \item Agora a variante crucial --- ``Monty distraído'': o
        apresentador \emph{não} sabe onde está o carro, abre uma
        das outras portas ao acaso e, por sorte, revela um bode.
        Liste os seis cenários igualmente prováveis (posição do
        carro $\times$ porta aberta), descarte aqueles em que o
        carro é revelado e calcule a \omterm{def:g10:proba:distribution}{probabilidade} de trocar
        ganhar \emph{entre os cenários sobreviventes}. O que
        mudou, e por que o conhecimento do apresentador importa?
  \item A bomba de intuição: cem portas, você escolhe uma, e o
        apresentador que sabe abre $98$ portas com bodes. Trocar?
        Com que \omterm{def:g10:proba:distribution}{probabilidade} de ganhar?
  \item Você enfrenta um cético que, como os mil autores das
        cartas, insiste em ``50--50''. Descreva uma simulação com
        cartas (três cartas, muitas rodadas, um amigo como
        apresentador) e diga que proporção de vitórias ao trocar
        você espera após $300$ rodadas --- com o \omterm{def:g10:numbers:interval}{intervalo} de
        flutuação do capítulo como margem.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Primos do paradoxo.}
\begin{enumerate}[resume]
  \item Uma família tem dois filhos; você fica sabendo que
        \emph{pelo menos um é menino}. Liste as configurações
        igualmente prováveis de dois filhos compatíveis com essa
        informação e calcule $P(\text{dois meninos})$.
  \item Mesma família, mas a informação agora é: \emph{o filho
        mais velho é menino}. Recalcule. Por que as duas
        respostas diferem, e o que elas têm em comum com os dois
        Montys?
  \item Três moedas honestas: calcule $P(\text{as três iguais})$.
        O matemático d'Alembert argumentou em 1754 por outra
        resposta: ``ou são todas iguais, ou não são --- um caso
        em dois''. Nomeie o erro exato, com a
        \cref{prop:g10:proba:uniform} em mãos.
  \item Um jogo custa $2$ euros: lance um dado, ganhe $6$ euros
        com um seis, $3$ euros com um cinco, e nada nos demais
        casos (\cref{exo:g10:proba:9}). Em $600$ partidas, que
        ganho médio por partida a contagem prevê? Uma seguradora
        pratica a mesma matemática: uma chance anual de $1\,\%$
        de um sinistro de $10\,000$ euros --- qual é o prêmio
        \emph{justo} e por que o prêmio real o supera?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- As \omterm{def:g10:proba:sample}{amostras} como árbitros.}
\begin{enumerate}[resume]
  \item Uma moeda é lançada $100$ vezes. Usando o \omterm{def:g10:numbers:interval}{intervalo} de
        flutuação $p \pm \frac{1}{\sqrt n}$, decida qual destes
        resultados deve levantar suspeitas: $55$ caras; $70$
        caras.
  \item Uma pesquisa com $1\,000$ eleitores dá $52\,\%$ a um
        candidato. Calcule o \omterm{def:g10:numbers:interval}{intervalo} de flutuação em torno de
        $50\,\%$ e diga se a pesquisa \emph{prova} que o
        candidato está à frente.
  \item Uma fábrica afirma ter $4\,\%$ de peças defeituosas. Uma
        \omterm{def:g10:proba:sample}{amostra} de $400$ mostra $28$ defeituosas ($7\,\%$); uma
        \omterm{def:g10:proba:sample}{amostra} posterior de $2\,500$ mostra $175$ ($7\,\%$ de
        novo). Teste a afirmação contra cada \omterm{def:g10:proba:sample}{amostra}. Por que os
        veredictos diferem?
  \item Duas fontes de erro assombram toda pesquisa: o
        \emph{viés} de amostragem (o problema de fim de semana
        sobre como os dados mentem, no volume do ensino
        fundamental) e a \emph{flutuação} de amostragem (este
        capítulo). Qual delas encolhe quando $n$ cresce e qual
        sobrevive a qualquer tamanho de \omterm{def:g10:proba:sample}{amostra}? Uma frase para
        cada.
  \item Final --- a lista de verificação do probabilista, uma
        linha por item: resultados igualmente prováveis antes de
        contar (questão 15); \omterm{met:g10:proba:tree}{árvores} e \omterm{def:g10:proba:operations}{complementares} para vários
        passos e para os ``pelo menos'' (questões 2--3); \emph{a
        informação muda as \omterm{def:g10:proba:distribution}{probabilidades}, e o modo como foi
        obtida importa} (questões 7--14); as \omterm{def:g10:proba:sample}{amostras} flutuam
        cerca de $\frac{1}{\sqrt n}$, e a simulação arbitra
        (questões 11, 16--18).
\end{enumerate}
\end{problem}
