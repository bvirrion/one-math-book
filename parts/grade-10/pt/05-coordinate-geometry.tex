\chapter{Geometria analítica}\label{ch:g10:coordgeom}

A grande ideia de Descartes foi descrever pontos por pares de números,
transformando problemas de geometria em cálculos. Com apenas duas
fórmulas --- a do \omterm{prop:g10:coordgeom:midpoint}{ponto médio} e a da distância --- é possível provar que
um triângulo é isósceles, que um quadrilátero é um paralelogramo ou que
três pontos são colineares, sem traçar uma única linha auxiliar.

\section{Coordenadas no plano}

\begin{definition}[Sistema de coordenadas]\label{def:g10:coordgeom:system}
Um \emph{sistema de coordenadas}\index{sistema de coordenadas} do plano
consiste em uma origem $O$ e dois eixos graduados que passam por $O$: o
eixo horizontal $x$ e o eixo vertical $y$. Todo ponto $M$ tem então um
único par de \emph{coordenadas}\index{coordenadas} $(x, y)$: sua
\emph{abscissa}\index{abscissa} $x$ e sua
\emph{ordenada}\index{ordenada} $y$. O sistema é
\emph{ortonormal}\index{sistema ortonormal} quando os eixos são
perpendiculares e trazem a mesma unidade de comprimento. Todos os
sistemas deste capítulo são ortonormais.
\end{definition}

\section{Ponto médio de um segmento}

\begin{proposition}[Fórmula do ponto médio]\label{prop:g10:coordgeom:midpoint}
Sejam $A(x_A, y_A)$ e $B(x_B, y_B)$. O \emph{ponto médio}\index{ponto
médio} $I$ do segmento $[AB]$ tem \omterm{def:g10:coordgeom:system}{coordenadas}
\[
I\left(\frac{x_A + x_B}{2},\ \frac{y_A + y_B}{2}\right).
\]
\end{proposition}

\begin{proof}
Considere as \omterm{def:g10:coordgeom:system}{coordenadas} horizontais. Ao ir de $A$ a $B$, a \omterm{def:g10:coordgeom:system}{abscissa}
varia de $x_B - x_A$; o ponto que está na metade do caminho tem \omterm{def:g10:coordgeom:system}{abscissa}
\[
x_A + \frac{x_B - x_A}{2} = \frac{2x_A + x_B - x_A}{2}
= \frac{x_A + x_B}{2},
\]
a média das duas \omterm{def:g10:coordgeom:system}{abscissas}. O mesmo cálculo se aplica às \omterm{def:g10:coordgeom:system}{ordenadas}.
\end{proof}

\begin{example}\label{ex:g10:coordgeom:midpoint}
Com $A(-1, 4)$ e $B(5, -2)$, o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$ é
\[
I\left(\frac{-1 + 5}{2},\ \frac{4 + (-2)}{2}\right) = I(2, 1).
\]
A fórmula também funciona no sentido inverso: se $A(-1, 4)$ e o \omterm{prop:g10:coordgeom:midpoint}{ponto
médio} é $I(2, 1)$, então $B$ satisfaz $\frac{-1 + x_B}{2} = 2$ e
$\frac{4 + y_B}{2} = 1$, logo $x_B = 5$ e $y_B = -2$: o ponto $B$ é o
\emph{simétrico} de $A$ em relação a $I$.
\end{example}

\section{Distância entre dois pontos}

\begin{theorem}[Fórmula da distância]\label{thm:g10:coordgeom:distance}
Em um \omterm{def:g10:coordgeom:system}{sistema ortonormal}, a distância entre $A(x_A, y_A)$ e
$B(x_B, y_B)$ é
\[
AB = \sqrt{(x_B - x_A)^2 + (y_B - y_A)^2}.
\]
\end{theorem}

\begin{proof}
Seja $C$ o ponto $(x_B, y_A)$: ele tem a mesma altura de $A$ e a mesma
\omterm{def:g10:coordgeom:system}{abscissa} de $B$, de modo que o triângulo $ACB$ tem um ângulo reto em $C$.
Seus catetos são segmentos horizontal e vertical, de comprimentos
$AC = \abs{x_B - x_A}$ e $CB = \abs{y_B - y_A}$. Pelo teorema de
Pitágoras,
\[
AB^2 = AC^2 + CB^2 = (x_B - x_A)^2 + (y_B - y_A)^2,
\]
e extrair a raiz quadrada (os dois lados são não negativos) dá a
fórmula.
\end{proof}

  \draw[thin, gray!40] (-0.4,-0.4) grid (5.4,3.4);
  \draw[-Stealth] (-0.6,0) -- (5.6,0) node[below, font=\small] {$x$};

  \draw[omDef, thick] (A) -- (B);

  \fill (A) circle (2pt) node[above left, font=\small] {$A$};
  \fill (B) circle (2pt) node[above, font=\small] {$B$};
  \fill (C) circle (2pt) node[below right, font=\small] {$C$};
\end{tikzpicture}

{\small A fórmula da distância é o teorema de Pitágoras aplicado ao
triângulo retângulo $ACB$, construído sobre um cateto horizontal e um
cateto vertical.}
\end{omfigure}

\begin{example}\label{ex:g10:coordgeom:distance}
Com $A(1, 1)$ e $B(5, 3)$:
\[
AB = \sqrt{(5-1)^2 + (3-1)^2} = \sqrt{16 + 4} = \sqrt{20} = 2\sqrt 5 .
\]
As distâncias costumam ser mantidas na forma exata (com raiz); arredonde
somente no fim, se for preciso dar uma resposta decimal.
\end{example}

\begin{remark}
A ordem dos pontos não importa: $(x_A - x_B)^2 = (x_B - x_A)^2$. Mas não
esqueça os quadrados! A distância \emph{não} é
$\abs{x_B - x_A} + \abs{y_B - y_A}$.
\end{remark}

\section{Usar as fórmulas em geometria}

\begin{method}[Natureza de um triângulo]\label{met:g10:coordgeom:triangle}
Dados três pontos $A$, $B$, $C$ por suas \omterm{def:g10:coordgeom:system}{coordenadas}:
\begin{enumerate}
  \item calcule os três quadrados de comprimento $AB^2$, $AC^2$, $BC^2$
        com a fórmula da distância (guardar os quadrados evita raízes);
  \item dois quadrados iguais $\Rightarrow$ o triângulo é
        \emph{isósceles};
  \item se o maior dos quadrados for a soma dos outros dois, o triângulo
        é \emph{retângulo} no vértice oposto ao maior lado, pela recíproca
        do teorema de Pitágoras.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g10:coordgeom:triangle}
Sejam $A(0, 1)$, $B(4, 3)$ e $C(2, -3)$. Então
\begin{align*}
AB^2 &= (4-0)^2 + (3-1)^2 = 16 + 4 = 20, \\
AC^2 &= (2-0)^2 + (-3-1)^2 = 4 + 16 = 20, \\
BC^2 &= (2-4)^2 + (-3-3)^2 = 4 + 36 = 40 .
\end{align*}
Primeiro, $AB^2 = AC^2$, logo $AB = AC$: o triângulo é isósceles em $A$.
Além disso, $AB^2 + AC^2 = 40 = BC^2$, de modo que, pela recíproca do
teorema de Pitágoras, o triângulo também é retângulo em $A$.
\end{example}

\begin{method}[Reconhecer um paralelogramo]\label{met:g10:coordgeom:parallelogram}
Um quadrilátero $ABCD$ é um paralelogramo exatamente quando suas
diagonais $[AC]$ e $[BD]$ têm o mesmo \omterm{prop:g10:coordgeom:midpoint}{ponto médio}. Portanto: calcule os
dois \omterm{prop:g10:coordgeom:midpoint}{pontos médios} com a fórmula do \omterm{prop:g10:coordgeom:midpoint}{ponto médio} e compare-os.
\end{method}

\begin{example}\label{ex:g10:coordgeom:parallelogram}
Sejam $A(-1, 0)$, $B(2, 2)$, $C(5, 1)$ e $D(2, -1)$. O \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de
$[AC]$ é $\left(\frac{-1+5}{2}, \frac{0+1}{2}\right) = (2, \frac12)$; o
\omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[BD]$ é $\left(\frac{2+2}{2}, \frac{2-1}{2}\right) =
(2, \frac12)$. Eles coincidem, logo $ABCD$ é um paralelogramo.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:1}
Sejam $A(2, 5)$ e $B(-4, 1)$. Calcule as \omterm{def:g10:coordgeom:system}{coordenadas} do \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de
$[AB]$ e a distância $AB$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:2}
Sejam $A(3, -2)$ e $I(1, 2)$. Encontre as \omterm{def:g10:coordgeom:system}{coordenadas} do ponto $B$ tal
que $I$ seja o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:3}
Qual dos pontos $P(4, 1)$, $Q(-3, 2)$ e $R(0, -5)$ está mais próximo da
origem $O(0,0)$? Responda com distâncias exatas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:coordgeom:4}
Marque os pontos $A(1, 2)$, $B(5, 4)$, $C(7, 0)$ no papel quadriculado e
mostre em seguida, por cálculo, que o triângulo $ABC$ é isósceles. Ele é
retângulo?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:5}
Sejam $A(-2, 1)$, $B(1, 3)$, $C(4, 1)$ e $D(1, -1)$.
\begin{enumerate}
  \item Mostre que $ABCD$ é um paralelogramo.
  \item Calcule $AB$ e $BC$. $ABCD$ é um losango (todos os lados
        iguais)?
  \item Calcule $AC$ e $BD$. $ABCD$ é um retângulo (diagonais iguais)?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:6}
O círculo de centro $\Omega(2, 1)$ e raio $5$ é formado por todos os
pontos que estão a distância $5$ de $\Omega$. Quais dos pontos
$A(5, 5)$, $B(-1, -3)$, $C(6, 2)$ estão sobre esse círculo? Dentro dele?
Fora dele?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:7}
Sejam $A(1, 1)$ e $B(7, 5)$. Encontre todos os pontos $M(x, 0)$ do eixo
$x$ equidistantes de $A$ e $B$. (Escreva $MA^2 = MB^2$ e resolva em
$x$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:8}
Sejam $A(0, 3)$ e $B(4, 1)$.
\begin{enumerate}
  \item Calcule as \omterm{def:g10:coordgeom:system}{coordenadas} do \omterm{prop:g10:coordgeom:midpoint}{ponto médio} $I$ de $[AB]$.
  \item Mostre que $M(2, 2) = I$ e verifique em seguida, calculando $MA$
        e $MB$, que $M$ é equidistante de $A$ e $B$.
  \item Encontre um segundo ponto, sobre o eixo $y$, equidistante de $A$
        e $B$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:coordgeom:9}
São dados os pontos $A(-1, -1)$, $B(3, 1)$ e $C(11, 5)$. Calcule $AB$,
$BC$ e $AC$ e deduza que $A$, $B$, $C$ são colineares. (Três pontos são
colineares exatamente quando a maior das três distâncias é a soma das
outras duas.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:coordgeom:10}
Sejam $A(a, 0)$ e $B(0, b)$ com $a, b > 0$, e seja $I$ o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de
$[AB]$. Mostre por cálculo que $OI = IA = IB$, onde $O$ é a origem.
(Isso demonstra um teorema clássico: num triângulo retângulo, o \omterm{prop:g10:coordgeom:midpoint}{ponto
médio} da hipotenusa é equidistante dos três vértices.)
\end{exercise}

\section{Problema: A ponte de Descartes --- equações de círculos e
como descobrir onde você está}

\begin{problem}[{Problema de fim de semana --- o círculo vira uma
equação, teoremas antigos viram contas, e três balizas localizam um
ponto: trilateração pela álgebra}]
\label{pb:g10:coordgeom:1}
Em 1637, René Descartes construiu uma ponte entre dois
continentes: toda curva da geometria virou uma \emph{\omterm{def:g10:algebra:equation}{equação}} e
toda questão geométrica virou uma conta. Este problema atravessa
a ponte nos dois sentidos --- deduzindo a \omterm{def:g10:algebra:equation}{equação} do círculo a
partir da fórmula da distância
(\cref{thm:g10:coordgeom:distance}), redemonstrando teoremas
clássicos em três linhas de álgebra e terminando onde a ponte tem
hoje mais tráfego: no cálculo de uma posição a partir de sinais
de distância, o coração plano do GPS.

\medskip
\noindent\textbf{Parte I --- O círculo ganha uma \omterm{def:g10:algebra:equation}{equação}.}
\begin{enumerate}
  \item Um ponto $M(x, y)$ está no círculo de centro
        $\Omega(2, 1)$ e raio $5$ exatamente quando
        $\Omega M = 5$. Eleve essa condição ao quadrado para
        obter a \omterm{def:g10:algebra:equation}{equação} do círculo:
        \[
        (x - 2)^2 + (y - 1)^2 = 25 .
        \]
  \item Teste a pertinência: quais dos pontos $A(5, 5)$,
        $B(6, 4)$, $D(4, 4)$ estão sobre o círculo, dentro dele
        ou fora dele?
  \item Um círculo disfarçado: complete os quadrados
        (\cref{pb:g10:algebra:1}) em
        \[
        x^2 + y^2 - 6x + 4y - 12 = 0
        \]
        e dê seu centro e seu raio.
  \item Faça o mesmo com $x^2 + y^2 - 6x + 4y + 14 = 0$. O que a
        forma completada revela? Enuncie o critério: quando
        $(x - h)^2 + (y - k)^2 = c$ descreve um círculo, um único
        ponto ou coisa alguma?
  \item Intersecte o círculo da questão 3 com a reta horizontal
        $y = 3$: substitua e resolva. Quantos pontos, e qual é o
        nome geométrico de uma reta assim?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Teoremas antigos em três linhas.}
\begin{enumerate}[resume]
  \item O teorema do círculo do volume do ensino fundamental (um
        ângulo reto em toda semicircunferência), redemonstrado
        pela álgebra: sejam $A(-r, 0)$ e $B(r, 0)$ as
        extremidades de um diâmetro e $M(x, y)$ um ponto qualquer
        do círculo $x^2 + y^2 = r^2$. Calcule $MA^2 + MB^2$ e
        conclua, pela recíproca de Pitágoras, que
        $\widehat{AMB}$ é reto.
  \item O teorema da mediana: com $A(-a, 0)$, $B(a, 0)$ (de modo
        que o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$ é a origem $I$), mostre que,
        para todo ponto $M(x, y)$,
        \[
        MA^2 + MB^2 = 2\,MI^2 + \frac{AB^2}{2} .
        \]
  \item Deduza o lugar geométrico dos pontos $M$ com
        $MA^2 + MB^2 = 26$ quando $AB = 6$: que curva, qual
        centro, qual raio?
  \item Outro lugar geométrico: $A(0,0)$, $B(3,0)$; encontre
        todos os pontos com $MA = 2\,MB$. (Eleve ao quadrado,
        expanda, complete os quadrados: aparece um círculo
        famoso --- Apolônio já o conhecia sem \omterm{def:g10:coordgeom:system}{coordenadas}.)
  \item Em uma ou duas frases: o que a ponte de Descartes muda no
        \emph{modo} como os teoremas são demonstrados? Compare a
        questão 6 com a demonstração pelo retângulo do teorema do
        círculo do volume do ensino fundamental (um ângulo reto
        em toda semicircunferência).
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Três balizas encontram você.}
Sua posição $M(x, y)$ é desconhecida. A baliza $P(0, 0)$ mede sua
distância como $5$; a baliza $Q(6, 0)$ também mede $5$.
\begin{enumerate}[resume]
  \item Escreva as duas \omterm{def:g10:algebra:equation}{equações} de círculo, subtraia-as e veja
        os quadrados se cancelarem: que \omterm{def:g10:algebra:equation}{equação} simples
        sobrevive? Combine-a com um dos círculos para encontrar
        as \emph{duas} posições candidatas.
  \item Uma terceira baliza $R(0, 8)$ mede sua distância como
        $5$. Calcule sua distância a cada candidata e decida onde
        você está.
  \item Explique por que subtrair duas \omterm{def:g10:algebra:equation}{equações} de círculo
        \emph{sempre} produz a \omterm{def:g10:algebra:equation}{equação} de uma reta (que termos se
        cancelam?) e o que essa reta é, geometricamente, quando
        os círculos se cortam em dois pontos.
  \item O posicionamento por satélite real acontece no espaço e
        com relógios imperfeitos: cada satélite fornece (pelo
        tempo de percurso do sinal) uma distância, portanto uma
        esfera. Quantas incógnitas tem um receptor (a posição
        \emph{e} o erro do próprio relógio) e por que o GPS,
        assim, escuta pelo menos \emph{quatro} satélites?
  \item Precisão: suponha que a distância da baliza $P$ seja na
        verdade $5.1$ em vez de $5$ (tudo o mais inalterado).
        Refaça a subtração da questão 11 para achar o novo $x$ e
        quantifique quanto a estimativa se deslocou. (Compare com
        as ideias de orçamento de erro do
        \cref{pb:g10:numbers:1}.)
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A caixa de ferramentas do
agrimensor.}
\begin{enumerate}[resume]
  \item O caminho mais curto até a água: um acampamento em
        $A(1, 5)$, um celeiro em $B(7, 3)$ e um rio reto ao longo
        do eixo $y = 0$. Para ir de $A$ ao rio e depois a $B$
        andando o \omterm{def:g10:functions:extrema}{mínimo} possível: reflita $B$ em relação ao rio,
        obtendo $B'$, e encontre onde o segmento $[AB']$ cruza o
        rio. Dê o ponto de cruzamento e o comprimento total
        \omterm{def:g10:functions:extrema}{mínimo}. (A ideia é a do exercício do ricochete de
        bilhar, no volume do ensino fundamental, agora
        inteiramente calculável.)
  \item Confirme a minimalidade contra um concorrente: calcule o
        percurso total passando por $Q(3, 0)$ e verifique que ele
        perde para a sua resposta.
  \item Classifique completamente o quadrilátero $A(1,1)$,
        $B(4,2)$, $C(5,5)$, $D(2,4)$: paralelogramo? losango?
        retângulo? quadrado?
        (\cref{met:g10:coordgeom:parallelogram},
        \cref{met:g10:coordgeom:triangle} --- compare também as
        diagonais.)
  \item Dados $A(-1, 2)$, $B(3, 4)$, $C(6, 0)$, encontre $D$ tal
        que $ABCD$ seja um paralelogramo (o truque do \omterm{prop:g10:coordgeom:midpoint}{ponto médio}
        das diagonais, do problema de fim de semana sobre meias
        voltas, no volume do ensino fundamental, agora em
        fórmulas).
  \item Final: três ferramentas --- fórmula do \omterm{prop:g10:coordgeom:midpoint}{ponto médio},
        fórmula da distância e subtração de \omterm{def:g10:algebra:equation}{equações} de círculo.
        Para cada uma, diga que tipo de questão ela resolveu
        neste problema e o que os dois próximos capítulos
        acrescentam à caixa (direções e \omterm{def:g10:reffunc:affine}{coeficientes angulares}:
        vetores e \omterm{def:g10:algebra:equation}{equações} de retas).
\end{enumerate}
\end{problem}
