\chapter{Números e conjuntos numéricos}\label{ch:g10:numbers}

A matemática começa com os números, e nem todos os números são da mesma
espécie: os que servem para contar, os negativos, as frações e números
como $\sqrt 2$ ou $\pi$ que nenhuma fração consegue exprimir. Este
capítulo organiza-os em famílias encaixadas, apresenta os \omterm{def:g10:numbers:interval}{intervalos} para
descrever pedaços da reta numérica e usa o \omterm{def:g10:numbers:abs}{módulo} para medir distâncias
entre números.

\section{As famílias de números}

\begin{definition}[Conjuntos numéricos]\label{def:g10:numbers:sets}
\begin{itemize}
  \item $\N$ é o conjunto dos \emph{números naturais}\index{número natural}:
        $0, 1, 2, 3, \dots$
  \item $\Z$ é o conjunto dos \emph{números inteiros}\index{número inteiro}:
        $\dots, -2, -1, 0, 1, 2, \dots$
  \item $\Q$ é o conjunto dos \emph{números racionais}\index{número racional}:
        todos os quocientes $\frac{p}{q}$ com $p \in \Z$, $q \in \N$ e
        $q \neq 0$.
  \item $\R$ é o conjunto dos \emph{números reais}\index{número real}: todos
        os números que podem ser colocados sobre a reta numérica.
\end{itemize}
\end{definition}

\begin{notation}
O símbolo $\in$ lê-se ``pertence a'': $3 \in \N$, $-5 \in \Z$,
$\frac23 \in \Q$. O símbolo $\subset$ lê-se ``está contido em'': todo
\omterm{def:g10:numbers:sets}{número natural} é inteiro, todo inteiro é racional
(por exemplo, $-5 = \frac{-5}{1}$) e todo racional é real, de modo que
\[
\N \subset \Z \subset \Q \subset \R .
\]
Uma barra nega o símbolo: $\frac12 \notin \Z$.
\end{notation}

\begin{example}\label{ex:g10:numbers:classify}
Vamos situar alguns números na menor família que os contém.
\begin{itemize}
  \item $\frac{28}{4} = 7$, logo $\frac{28}{4} \in \N$ mesmo estando
        escrito como fração: simplifique sempre primeiro.
  \item $-3.5 = -\frac{35}{10} = -\frac72$ é racional, mas não é inteiro.
  \item $0.333\dots$ (o algarismo $3$ repetindo-se indefinidamente) vale
        $\frac13$, um \omterm{def:g10:numbers:sets}{número racional}.
  \item $\sqrt 2$ e $\pi$ são reais, mas não racionais, como veremos a
        seguir para $\sqrt 2$; tais números são ditos
        \emph{irracionais}\index{número irracional}.
\end{itemize}
\end{example}

\begin{theorem}[Irracionalidade de $\sqrt 2$]\label{thm:g10:numbers:sqrt2}
O número $\sqrt 2$ não é racional: nenhuma fração de inteiros tem
quadrado $2$.
\end{theorem}

\begin{proof}
Raciocinamos por absurdo, em pequenas etapas.
\begin{enumerate}
  \item Suponha $\sqrt 2 = \frac{p}{q}$, com $p$ e $q$ inteiros positivos
        e a fração totalmente simplificada, de modo que $p$ e $q$ não são
        ambos pares.
  \item Elevando os dois lados ao quadrado, $2 = \frac{p^2}{q^2}$, isto é,
        $p^2 = 2q^2$. Logo $p^2$ é par.
  \item Se $p$ fosse ímpar, digamos $p = 2k+1$, então
        $p^2 = 4k^2 + 4k + 1$ seria ímpar. Como $p^2$ é par, $p$ tem de
        ser par: $p = 2k$ para algum inteiro $k$.
  \item Substituindo: $(2k)^2 = 2q^2$, logo $4k^2 = 2q^2$, logo
        $q^2 = 2k^2$. Pelo mesmo argumento da etapa 3, $q$ é par.
  \item Agora $p$ e $q$ são ambos pares, o que contradiz a etapa 1. A
        hipótese era impossível: $\sqrt 2$ é irracional.
\end{enumerate}
\end{proof}

\begin{remark}
Os \omterm{def:g10:numbers:sets}{números racionais} são exatamente os números cuja representação decimal
ou termina (como $\frac72 = 3.5$) ou passa a repetir indefinidamente um
mesmo bloco (como $\frac13 = 0.333\dots$ ou
$\frac{1}{7} = 0.142857\,142857\dots$). Os \omterm{ex:g10:numbers:classify}{irracionais}, como
$\sqrt2 = 1.41421356\dots$, têm algarismos decimais que nunca caem em um
padrão periódico. Admitimos essa caracterização neste nível.
\end{remark}

\section{A reta numérica e os intervalos}

Os \omterm{def:g10:numbers:sets}{números reais} preenchem uma reta: escolhidos uma origem $0$ e uma
unidade de comprimento, cada \omterm{def:g10:numbers:sets}{número real} corresponde a exatamente um
ponto.

\begin{definition}[Intervalo]\label{def:g10:numbers:interval}
Sejam $a$ e $b$ \omterm{def:g10:numbers:sets}{números reais} com $a < b$. Um
\emph{intervalo}\index{intervalo} é o conjunto de todos os \omterm{def:g10:numbers:sets}{números reais}
compreendidos entre dois extremos. O colchete indica que o extremo está
incluído; o parêntese, que está excluído:
\begin{center}
\begin{tabular}{c|c}
notação & descrição \\
\hline
$\intcc{a}{b}$ & $a \leq x \leq b$ (ambos os extremos incluídos) \\
$\intoo{a}{b}$ & $a < x < b$ (ambos os extremos excluídos) \\
$\intco{a}{b}$ & $a \leq x < b$ \\
$\intoc{a}{b}$ & $a < x \leq b$ \\
$\intco{a}{+\infty}$ & $x \geq a$ \\
$\intoo{-\infty}{b}$ & $x < b$ \\
\end{tabular}
\end{center}
Os símbolos $-\infty$ e $+\infty$ (``infinito'') não são números, apenas
uma forma de dizer que o intervalo prossegue indefinidamente; ao lado
deles o sinal é sempre um parêntese. A reta inteira $\R$ é o intervalo
$\intoo{-\infty}{+\infty}$.
\end{definition}

\begin{definition}[Interseção e união]\label{def:g10:numbers:interunion}
Sejam $I$ e $J$ dois conjuntos de \omterm{def:g10:numbers:sets}{números reais}.
A \emph{interseção}\index{interseção} $I \cap J$ (``$I$ e $J$'') é o
conjunto dos números que pertencem a ambos; a
\emph{união}\index{união} $I \cup J$ (``$I$ ou $J$'') é o conjunto dos
números que pertencem a pelo menos um deles.
\end{definition}

\begin{example}\label{ex:g10:numbers:interunion}
Tome $I = \intcc{-1}{3}$ e $J = \intoo{1}{5}$. Desenhe os dois sobre a
mesma reta: eles se sobrepõem entre $1$ e $3$. Portanto
\[
I \cap J = \intoc{1}{3},
\qquad
I \cup J = \intco{-1}{5}.
\]
Repare nos extremos: $1 \notin J$, logo $1 \notin I \cap J$; mas $3 \in I$
e $3 \in J$, logo $3 \in I \cap J$.
\end{example}

\begin{method}[Trabalhar com intervalos]\label{met:g10:numbers:intervals}
Para achar a \omterm{def:g10:numbers:interunion}{interseção} ou a \omterm{def:g10:numbers:interunion}{união} de dois \omterm{def:g10:numbers:interval}{intervalos}:
\begin{enumerate}
  \item desenhe a reta numérica e marque os quatro extremos;
  \item hachure o primeiro \omterm{def:g10:numbers:interval}{intervalo} acima da reta e o segundo abaixo;
  \item a \omterm{def:g10:numbers:interunion}{interseção} é onde as duas hachuras se sobrepõem; a \omterm{def:g10:numbers:interunion}{união} é onde
        há pelo menos uma hachura;
  \item decida cada extremo testando se ele próprio pertence aos dois
        conjuntos (\omterm{def:g10:numbers:interunion}{interseção}) ou a pelo menos um (\omterm{def:g10:numbers:interunion}{união}).
\end{enumerate}
\end{method}

\section{Módulo e distância}

\begin{definition}[Módulo]\label{def:g10:numbers:abs}
O \emph{módulo}\index{módulo} de um \omterm{def:g10:numbers:sets}{número real} $x$ é
\[
\abs{x} =
\begin{cases}
x & \text{se } x \geq 0, \\
-x & \text{se } x < 0.
\end{cases}
\]
Por exemplo, $\abs{7} = 7$ e $\abs{-4} = 4$. Em todos os casos
$\abs{x} \geq 0$.
\end{definition}

\begin{proposition}[Distância na reta]\label{prop:g10:numbers:distance}
Para quaisquer \omterm{def:g10:numbers:sets}{números reais} $a$ e $b$, a distância entre os pontos $a$ e
$b$ da reta numérica é $\abs{b - a}$. Em particular, $\abs{x}$ é a
distância de $x$ a $0$.
\end{proposition}

\begin{proof}
Se $b \geq a$, a distância de $a$ a $b$ é $b - a \geq 0$, que é igual a
$\abs{b-a}$. Se $b < a$, a distância é $a - b = -(b - a) > 0$, que é de
novo $\abs{b - a}$ pela definição de \omterm{def:g10:numbers:abs}{módulo}.
\end{proof}

\begin{proposition}[Módulo e intervalos]\label{prop:g10:numbers:absinterval}
Sejam $a$ um \omterm{def:g10:numbers:sets}{número real} e $r > 0$. Então
\[
\abs{x - a} \leq r
\quad\text{exatamente quando}\quad
x \in \intcc{a - r}{a + r}.
\]
\end{proposition}

\begin{proof}
$\abs{x-a} \leq r$ diz que a distância de $x$ a $a$ é no máximo $r$, isto
é, que $x$ não se afasta mais do que $r$ de $a$, para nenhum dos dois
lados. Os números que cumprem isso são exatamente os que estão entre
$a - r$ e $a + r$, extremos incluídos.
\end{proof}

{\small A desigualdade $\abs{x - a} \leq r$ descreve o \omterm{def:g10:numbers:interval}{intervalo} dos

\end{omfigure}

\begin{example}\label{ex:g10:numbers:abs}
Resolva $\abs{x - 3} \leq 2$. As soluções são os números que estão a
distância no máximo $2$ de $3$: o \omterm{def:g10:numbers:interval}{intervalo} $\intcc{1}{5}$.
Reciprocamente, o \omterm{def:g10:numbers:interval}{intervalo} $\intcc{-1}{7}$ tem centro
$\frac{-1+7}{2} = 3$ e raio $\frac{7-(-1)}{2} = 4$, logo é descrito por
$\abs{x - 3} \leq 4$.
\end{example}

\section{Aproximações}

Os números \omterm{ex:g10:numbers:classify}{irracionais}, e mesmo a maioria das frações, não podem ser
escritos exatamente com um número finito de algarismos decimais; na
prática, portanto, nós os aproximamos.

\begin{definition}[Aproximação com precisão dada]\label{def:g10:numbers:approx}
Um número $d$ é uma \emph{aproximação}\index{aproximação} de $x$ com
precisão $10^{-n}$ quando $\abs{x - d} \leq 10^{-n}$. Truncar ou
arredondar a representação decimal após o $n$-ésimo algarismo produz, em
ambos os casos, aproximações desse tipo.
\end{definition}

\begin{example}
De $\pi = 3.14159\,26\dots$: a truncatura $3.141$ e o arredondamento
$3.142$ são ambos \omterm{def:g10:numbers:approx}{aproximações} de $\pi$ com precisão $10^{-3}$. O
arredondamento fica a distância menor que $\frac12 \times 10^{-3}$; a
truncatura garante apenas $10^{-3}$. Escrever
$3.141 \leq \pi \leq 3.142$ \emph{enquadra} $\pi$ entre dois limitantes
decimais.
\end{example}

Para cada número, indique o menor dos conjuntos $\N$, $\Z$, $\Q$, $\R$ ao

\frac{15}{3}, \qquad -7, \qquad \frac{22}{7}, \qquad \sqrt{9}, \qquad

\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:2}
Escreva cada afirmação em notação de \omterm{def:g10:numbers:interval}{intervalo} e depois represente-a na
reta numérica: (a) $-2 \leq x < 5$; (b) $x > 3$; (c) $x \leq -1$;
(d) a distância de $x$ a $2$ é no máximo $3$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:3}
Calcule $I \cap J$ e $I \cup J$ para
\[
\text{(a) } I = \intcc{-3}{2},\ J = \intco{0}{4};
\qquad
\text{(b) } I = \intoo{-\infty}{1},\ J = \intco{-2}{+\infty}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:4}
Calcule sem calculadora:
\[
\abs{-6}, \qquad \abs{4 - 9}, \qquad \abs{-3 - 5}, \qquad
\abs{\sqrt 2 - 1}, \qquad \abs{1 - \sqrt 2}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g10:numbers:5}
Resolva as equações e inequações e dê os conjuntos solução:
\[
\abs{x} = 5, \qquad \abs{x - 1} = 3, \qquad \abs{x - 4} \leq 1, \qquad
\abs{x + 2} < 3 .
\]
(Observe que $\abs{x+2} = \abs{x - (-2)}$ é uma distância a $-2$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:6}
Descreva cada \omterm{def:g10:numbers:interval}{intervalo} por uma desigualdade da forma
$\abs{x - a} \leq r$ ou $\abs{x - a} < r$:
\[
\intcc{2}{8}, \qquad \intoo{-5}{1}, \qquad \intcc{-7}{-3}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:7}
Mostre que $0.272727\dots$ (o bloco $27$ repetindo-se indefinidamente) é
racional. (Sugestão: chame-o de $x$ e calcule $100x - x$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:8}
Verdadeiro ou falso? Justifique cada resposta com um argumento ou um
contraexemplo.
\begin{enumerate}
  \item A soma de dois inteiros é um inteiro.
  \item O quociente de dois inteiros é um inteiro.
  \item A soma de dois \omterm{def:g10:numbers:sets}{números racionais} é racional.
  \item A soma de um \omterm{def:g10:numbers:sets}{número racional} com um \omterm{ex:g10:numbers:classify}{número irracional} é
        irracional.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g10:numbers:9}
Usando $1.414 \leq \sqrt 2 \leq 1.415$, enquadre os números $2\sqrt2$,
$\sqrt2 + 3$ e $-\sqrt2$ entre dois limitantes decimais.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g10:numbers:10}
Adapte a demonstração do \cref{thm:g10:numbers:sqrt2} para mostrar que
$\sqrt 3$ é irracional. (Troque ``par'' por ``múltiplo de $3$'': verifique
primeiro que, se $p^2$ é múltiplo de $3$, então $p$ também é, examinando
os restos $0$, $1$, $2$ da divisão de $p$ por $3$.)
\end{exercise}

\section{Problema: Entre dois números quaisquer}

\begin{problem}[{Problema de fim de semana --- racionais e irracionais
entrelaçam-se: todo intervalo, por menor que seja, contém uma infinidade
de cada um, e nenhuma medição consegue distingui-los}]
\label{pb:g10:numbers:1}
Os racionais parecem uma multidão (todas as frações!) e os \omterm{ex:g10:numbers:classify}{irracionais},
exceções exóticas ($\sqrt2$, $\pi$). Este problema revela o quadro
verdadeiro: as duas famílias \emph{entrelaçam-se} tão finamente que todo
\omterm{def:g10:numbers:interval}{intervalo} da reta numérica, por microscópico que seja, contém uma
infinidade de cada uma --- com consequências estranhas, como esta:
nenhuma medição física, por mais precisa, jamais poderá decidir se um
comprimento é racional.

\medskip
\noindent\textbf{Parte I --- Os quatro reinos.}
\begin{enumerate}
  \item Para cada número, nomeie o menor dos conjuntos $\N$,
        $\Z$, $\Q$, $\R$ que o contém
        (\cref{def:g10:numbers:sets}): $-7$; \ $\frac{13}{4}$; \
        $\sqrt{16}$; \ $0.121212\ldots$ (lembre-se do problema
        de fim de semana sobre dízimas periódicas do volume do
        ensino fundamental); \ $\sqrt8$; \ $\pi$ (admita sua
        irracionalidade --- demonstrada apenas nos volumes de
        graduação).
  \item Prove que $\Q$ é estável pela adição e pela
        multiplicação: se $x = \frac pq$ e $y = \frac rs$ são
        racionais, escreva $x + y$ e $xy$ como frações únicas.
  \item Deduza por absurdo: (a) a soma de um racional com um
        irracional é irracional; (b) o produto de um racional
        \emph{não nulo} por um irracional é irracional.
  \item Mostre que o conjunto dos \omterm{ex:g10:numbers:classify}{irracionais} não é estável por
        \emph{nenhuma} das duas operações: exiba dois
        \omterm{ex:g10:numbers:classify}{irracionais} cuja soma é racional e dois cujo produto é
        racional.
  \item Situe $\sqrt2 + \sqrt8$ e $\sqrt2 \times \sqrt8$ nos
        quatro reinos (simplifique $\sqrt8$ primeiro, pelo
        método do volume do ensino fundamental).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Os racionais são densos.}
\begin{enumerate}[resume]
  \item Encontre um racional estritamente entre $3.47$ e $3.48$;
        depois um segundo; depois descreva como produzir quantos
        se queira (o zoom do problema de fim de semana sobre o
        número seguinte que não existe, do volume do ensino
        fundamental, agora com uma demonstração à vista).
  \item O teorema geral. Sejam $a < b$ dois reais quaisquer, com
        folga $g = b - a > 0$. Escolha $n$ com
        $10^{-n} < g$ e considere os múltiplos de $10^{-n}$
        (a grade decimal de passo $10^{-n}$). Explique por que
        pelo menos um ponto da grade cai estritamente entre $a$ e
        $b$ e conclua: \emph{todo \omterm{def:g10:numbers:interval}{intervalo} de comprimento
        positivo contém um \omterm{def:g10:numbers:sets}{número racional}}.
  \item Reforce a conclusão: todo \omterm{def:g10:numbers:interval}{intervalo} desses contém
        \emph{uma infinidade} de racionais. (Aplique de novo a
        questão 7, dentro de um \omterm{def:g10:numbers:interval}{intervalo} menor.)
  \item Agora os \omterm{ex:g10:numbers:classify}{irracionais}: dados $a < b$, tome um racional $r$
        estritamente no interior (questão 7) e considere os
        números $r + \frac{\sqrt2}{10^k}$. Usando a questão 3,
        mostre que eles são \omterm{ex:g10:numbers:classify}{irracionais} e que, para $k$ grande o
        bastante, continuam dentro do \omterm{def:g10:numbers:interval}{intervalo}: \emph{todo
        \omterm{def:g10:numbers:interval}{intervalo} também contém uma infinidade de \omterm{ex:g10:numbers:classify}{irracionais}}.
  \item Dois enigmas clássicos resolvidos: existe um menor
        \omterm{def:g10:numbers:sets}{número real} positivo? Existe um real ``logo depois de
        $3$''? Responda aos dois com o teorema da questão 7 e
        saúde a versão da infância (o problema de fim de semana
        sobre o número seguinte que não existe, do volume do
        ensino fundamental).
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O \omterm{def:g10:numbers:abs}{módulo}, a geometria de
$\R$.}
\begin{enumerate}[resume]
  \item Resolva e exprima as soluções como \omterm{def:g10:numbers:interval}{intervalos} ou \omterm{def:g10:numbers:interunion}{uniões}
        (\cref{prop:g10:numbers:absinterval}):
        $\abs{x - 5} = 2$; \quad $\abs{x - 5} < 2$; \quad
        $\abs{x + 1} \geq 3$.
  \item Um pistão deve ser fabricado com $80$ mm dentro de uma
        tolerância de $0.05$ mm. Escreva a exigência com um
        \omterm{def:g10:numbers:abs}{módulo} e depois como \omterm{def:g10:numbers:interval}{intervalo}. Dois pistões medem
        $79.97$ e $80.06$ mm: quais são os veredictos?
  \item A desigualdade triangular na reta:
        $\abs{a + b} \leq \abs a + \abs b$. Verifique-a em
        $(a, b) = (3, -5)$ e $(-2, -7)$, demonstre-a quando $a$ e
        $b$ têm o mesmo sinal e quando têm sinais contrários, e
        diga exatamente quando vale a igualdade.
  \item Resolva $\abs{x - 2} = \abs{x + 4}$ lendo-a como uma
        igualdade de distâncias na reta. Que ponto do problema de
        fim de semana sobre os dois espelhos, do volume do ensino
        fundamental, você acabou de calcular, uma dimensão
        abaixo?
  \item Simplifique $\sqrt{x^2}$ --- com cuidado. Teste sua
        fórmula em $x = 3$ e $x = -3$ e use-a em seguida para
        resolver $x^2 < 9$ com um \omterm{def:g10:numbers:abs}{módulo}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O que nenhuma medição decide.}
\begin{enumerate}[resume]
  \item Um físico mede uma barra: $1.41421 \pm 0.00001$ m.
        Alguma medição desse tipo --- esta ou uma futura, mais
        precisa --- poderá algum dia \emph{provar} que o
        comprimento da barra é irracional? Ou que é racional? Use
        as questões 7 e 9 sobre o \omterm{def:g10:numbers:interval}{intervalo} de tolerância e
        explique por que só uma \emph{demonstração} (como a da
        diagonal, no problema de fim de semana sobre
        irracionalidade do volume do ensino fundamental) pode
        decidir a irracionalidade.
  \item As truncaturas $1.4$, $1.41$, $1.414$, $1.4142,\dots$
        de $\sqrt2$ são todas racionais. O que elas mostram sobre
        o quanto $\Q$ se comprime contra cada irracional?
        Formule o fato geral.
  \item Aritmética de \omterm{def:g10:numbers:interval}{intervalos}: valores medidos
        $a = 2.5 \pm 0.1$ e $b = 1.2 \pm 0.1$. Enquadre $a + b$ e
        $a \times b$ entre limitantes seguros. Qual das duas
        operações degrada mais a precisão?
  \item Uma regra de engenheiro diz que os erros de medições
        independentes ``somam-se'' nas somas: exprima com a
        desigualdade triangular (questão 13) por que o erro de
        $a + b$ é no máximo a soma dos erros --- a desigualdade
        \emph{é} a regra de engenharia.
  \item Final, em um parágrafo curto: reúna o retrato da reta
        real estabelecido por este problema e por seus
        antecessores --- nenhum número seguinte (o problema de
        fim de semana sobre o número seguinte que não existe, do
        volume do ensino fundamental), racionais $=$ decimais
        que param ou se repetem (o problema de fim de semana
        sobre dízimas periódicas do mesmo volume), as duas
        famílias densas (questões 8 e 9), a medição para sempre
        indecisa (questão 16). Termine com a provocação que este
        capítulo ainda não pode demonstrar: num sentido preciso
        há \emph{muitíssimo mais} \omterm{ex:g10:numbers:classify}{irracionais} do que racionais
        --- os volumes de graduação contam os infinitos.
\end{enumerate}
\end{problem}
