\chapter{Functies en hun verloop}\label{ch:g11:func}

Dit hoofdstuk ordent wat je over \omterm{def:g11:func:function}{functies} moet weten vóór de analyse: de
kleine familie \emph{\omterm{def:g11:func:reference}{referentiefuncties}} waarvan de \omterm{def:g10:functions:graph}{grafieken} uit het hoofd
gekend moeten zijn, de precieze betekenis van \emph{\omterm{def:g11:func:monotone}{stijgend}} en
\emph{\omterm{def:g10:functions:variations}{dalend}}, en hoe het verloop zich gedraagt wanneer \omterm{def:g11:func:function}{functies} gecombineerd
worden.

\section{Functies en krommen}

\begin{definition}[Functie, domein, kromme]\label{def:g11:func:function}
Een \emph{functie}\index{functie} $f$ kent aan elk getal $x$ van een
verzameling $D \subseteq \R$ (haar \emph{domein}\index{domein}) precies één
getal $f(x)$ toe. Haar \emph{kromme} (of \omterm{def:g10:functions:graph}{grafiek}) is de verzameling van de
punten $\bigl(x, f(x)\bigr)$ met $x \in D$.
\end{definition}

\begin{example}
De formule $f(x) = \frac{\sqrt{x+2}}{x-1}$ vereist $x + 2 \geq 0$ (voor de
vierkantswortel) en $x \neq 1$ (noemer verschillend van nul): het \omterm{def:g11:func:function}{domein} is
$\intco{-2}{1} \cup \intoo{1}{+\infty}$.
\end{example}

\index{referentiefuncties}
\[
x \mapsto x^2 \text{ (het kwadraat)}, \quad
x \mapsto x^3 \text{ (de derde macht)}, \quad
x \mapsto \sqrt{x}, \quad
x \mapsto \frac{1}{x}, \quad

\section{Verloop op een interval}

\begin{definition}[Stijgend, dalend]\label{def:g11:func:monotone}
Zij $f$ gedefinieerd op een \omterm{def:g10:numbers:interval}{interval} $I$. De \omterm{def:g11:func:function}{functie} $f$ heet
\emph{stijgend}\index{stijgende functie} op $I$ wanneer ze de orde behoudt:
\[
\text{voor alle } u, v \in I: \quad u < v \implies f(u) \leq f(v),
\]
en \emph{\omterm{def:g10:functions:variations}{dalend}} wanneer ze de orde omkeert ($u < v \implies f(u) \geq
f(v)$). Met strikte ongelijkheden rechts heet $f$ \emph{strikt} stijgend
(respectievelijk \omterm{def:g10:functions:variations}{dalend}). Een \omterm{def:g11:func:function}{functie} die op $I$ stijgend of \omterm{def:g10:functions:variations}{dalend} is, heet
\emph{monotoon} op $I$.
\end{definition}

\begin{method}[Een verloop uit de definitie bewijzen]\label{met:g11:func:variation}
Neem twee willekeurige getallen $u < v$ uit het \omterm{def:g10:numbers:interval}{interval} en bepaal het teken
van het verschil $f(v) - f(u)$, meestal door het te \emph{\omterm{def:g10:algebra:expand}{ontbinden}}. Is dat
teken constant op het \omterm{def:g10:numbers:interval}{interval}, dan mag je besluiten.
\end{method}

\begin{proposition}[Verloop van de referentiefuncties]\label{prop:g11:func:refvariations}
\begin{enumerate}
  \item $x^2$ is strikt \omterm{def:g10:functions:variations}{dalend} op $\intoc{-\infty}{0}$ en strikt \omterm{def:g11:func:monotone}{stijgend} op
        $\intco{0}{+\infty}$.
  \item $x^3$ is strikt \omterm{def:g11:func:monotone}{stijgend} op $\R$.
  \item $\sqrt x$ is strikt \omterm{def:g11:func:monotone}{stijgend} op $\intco{0}{+\infty}$.
  \item $\frac1x$ is strikt \omterm{def:g10:functions:variations}{dalend} op $\intoo{-\infty}{0}$ en op
        $\intoo{0}{+\infty}$ (maar \emph{niet} op hun \omterm{def:g10:numbers:interunion}{vereniging}).
  \item $\abs x$ is strikt \omterm{def:g10:functions:variations}{dalend} op $\intoc{-\infty}{0}$ en strikt \omterm{def:g11:func:monotone}{stijgend}
        op $\intco{0}{+\infty}$.
\end{enumerate}
\end{proposition}

\begin{proof}
Zij $u < v$. We \omterm{def:g10:algebra:expand}{ontbinden} telkens $f(v) - f(u)$.

\emph{1.} $v^2 - u^2 = (v-u)(v+u)$. Op $\intco{0}{+\infty}$: $v - u > 0$ en
$v + u > 0$ (want $v > u \geq 0$), dus $v^2 > u^2$. Op
$\intoc{-\infty}{0}$: $v + u < 0$, dus $v^2 < u^2$.

\emph{2.} $v^3 - u^3 = (v - u)(v^2 + uv + u^2)$, en de tweede factor is
gelijk aan $\left(v + \frac u2\right)^2 + \frac{3u^2}{4} > 0$ tenzij
$u = v = 0$. Bijgevolg is $v^3 > u^3$ zodra $v > u$.

\emph{3.} Voor $0 \leq u < v$:
$\sqrt v - \sqrt u = \frac{(\sqrt v - \sqrt u)(\sqrt v + \sqrt u)}
{\sqrt v + \sqrt u} = \frac{v - u}{\sqrt v + \sqrt u} > 0$.

\emph{4.} Voor $0 < u < v$: $\frac1v - \frac1u = \frac{u - v}{uv} < 0$,
want $u - v < 0$ en $uv > 0$; dezelfde berekening geldt voor $u < v < 0$. Op
de \omterm{def:g10:numbers:interunion}{vereniging} gaat de bewering niet op: $-1 < 1$, maar
$\frac{1}{-1} = -1 < 1 = \frac11$.

\emph{5.} Op $\intco{0}{+\infty}$ is $\abs x = x$ strikt \omterm{def:g11:func:monotone}{stijgend}; op
$\intoc{-\infty}{0}$ is $\abs x = -x$ strikt \omterm{def:g10:functions:variations}{dalend}.
\end{proof}

\begin{remark}
Punt 4 is een klassieke valkuil: ``\omterm{def:g10:functions:variations}{dalend} op $\intoo{-\infty}{0}$ en op
$\intoo{0}{+\infty}$'' is \emph{niet} hetzelfde als ``\omterm{def:g10:functions:variations}{dalend} op
$\R \setminus \{0\}$''. Verloop heeft alleen zin op intervallen.
\end{remark}

\begin{example}[Vergelijken zonder rekenen]
Omdat $\sqrt x$ strikt \omterm{def:g11:func:monotone}{stijgend} is, geldt $\sqrt{7} < \sqrt{8}$; omdat $x^2$
strikt \omterm{def:g10:functions:variations}{dalend} is op $\intoc{-\infty}{0}$, geldt $(-3)^2 > (-2)^2$.
Monotonie zet vergelijkingen van beelden om in \omterm{def:g10:algebra:equation}{vergelijkingen} van de
argumenten.
\end{example}

\section{Nieuwe functies uit oude}

\begin{proposition}[Transformaties]\label{prop:g11:func:transforms}
Zij $u$ een \omterm{def:g11:func:function}{functie} gedefinieerd op een \omterm{def:g10:numbers:interval}{interval} $I$ en $k \in \R$.
\begin{enumerate}
  \item $u + k$ heeft op $I$ \emph{hetzelfde} verloop als $u$; haar kromme is
        die van $u$, verticaal over $k$ verschoven.
  \item Voor $\lambda > 0$ heeft $\lambda u$ hetzelfde verloop als $u$; voor
        $\lambda < 0$ is het verloop omgekeerd.
  \item Is $u \geq 0$ op $I$, dan heeft $\sqrt u$ hetzelfde verloop als $u$.
  \item Is $u \neq 0$ en heeft $u$ een constant teken op $I$, dan heeft
        $\frac1u$ het \emph{tegengestelde} verloop van $u$.
\end{enumerate}
\end{proposition}

\begin{proof}
Zij $x_1 < x_2$ in $I$, en schrijf $s = u(x_1)$ en $t = u(x_2)$ voor de
\omterm{def:g10:functions:function}{beelden}.

\emph{1.} $(u+k)(x_2) - (u+k)(x_1) = t - s$: het verschil van de \omterm{def:g10:functions:function}{beelden}
verandert niet, en dus ook het teken ervan niet.

\emph{2.} $\lambda t - \lambda s = \lambda(t - s)$: het teken blijft behouden
als $\lambda > 0$ en keert om als $\lambda < 0$.

\emph{3.} Is $u$ \omterm{def:g11:func:monotone}{stijgend} en $x_1 < x_2$, dan is $0 \leq s \leq t$, en omdat
de vierkantswortel \omterm{def:g11:func:monotone}{stijgend} is (\cref{prop:g11:func:refvariations}) geldt
$\sqrt s \leq \sqrt t$. Hetzelfde argument werkt in het dalende geval.

\emph{4.} Is $u$ \omterm{def:g11:func:monotone}{stijgend} en $0 < s \leq t$ (of $s \leq t < 0$), dan is
$\frac1s \geq \frac1t$, want $x \mapsto \frac1x$ is \omterm{def:g10:functions:variations}{dalend} aan elke kant van
$0$: de orde keert om.
\end{proof}

\begin{example}\label{ex:g11:func:composite}
Bestudeer $f(x) = \sqrt{x^2 + 1}$ op $\R$. De binnenfunctie
$u(x) = x^2 + 1$ is positief, \omterm{def:g10:functions:variations}{dalend} op $\intoc{-\infty}{0}$ en \omterm{def:g11:func:monotone}{stijgend} op
$\intco{0}{+\infty}$ (een verschoven kwadraat). Volgens punt 3 heeft $f$
hetzelfde verloop: eerst \omterm{def:g10:functions:variations}{dalend}, dan \omterm{def:g11:func:monotone}{stijgend}, met \omterm{def:g10:functions:extrema}{minimum} $f(0) = 1$.
\end{example}

\section{Even en oneven functies}

\begin{definition}[Pariteit]\label{def:g11:func:parity}
Zij $f$ gedefinieerd op een \omterm{def:g11:func:function}{domein} $D$ dat symmetrisch is om $0$ (dus
$-x \in D$ zodra $x \in D$). De \omterm{def:g11:func:function}{functie} $f$ heet \emph{even}\index{even
functie} wanneer $f(-x) = f(x)$ voor alle $x \in D$, en
\emph{oneven}\index{oneven functie} wanneer $f(-x) = -f(x)$ voor alle
$x \in D$.
\end{definition}

\begin{proposition}[Meetkundige betekenis]\label{prop:g11:func:paritysym}
De kromme van een \omterm{def:g11:func:parity}{even functie} is symmetrisch om de $y$-as; de kromme van een
\omterm{def:g11:func:parity}{oneven functie} is symmetrisch om de oorsprong.
\end{proposition}

\begin{proof}
Is $f$ \omterm{def:g11:func:parity}{even} en ligt $(x, y)$ op de kromme (dus $y = f(x)$), dan is
$f(-x) = f(x) = y$: ook het spiegelpunt $(-x, y)$ ligt op de kromme. Is $f$
\omterm{def:g11:func:parity}{oneven}, dan is $f(-x) = -y$: het punt $(-x, -y)$, symmetrisch met $(x,y)$ om
de oorsprong, ligt op de kromme.
\end{proof}

\begin{example}
$x^2$ en $\abs x$ zijn \omterm{def:g11:func:parity}{even}; $x^3$ en $\frac1x$ zijn \omterm{def:g11:func:parity}{oneven};
$f(x) = x^2 + x$ is geen van beide: $f(-1) = 0$ terwijl $f(1) = 2$, dus is
$f(-1) \neq \pm f(1)$. Pariteit halveert het werk: een \omterm{def:g11:func:parity}{even} of \omterm{def:g11:func:parity}{oneven functie}
hoef je alleen op $\intco{0}{+\infty}$ te bestuderen.
\end{example}

f(x) = 3x^4 - x^2, \qquad

$f(5) = 0$. Hoeveel oplossingen heeft de \omterm{def:g10:algebra:equation}{vergelijking} $f(x) = 1$ op

\end{exercise}

g(x) = \frac{1}{x^2 + 1} \ \text{ op } \intco{0}{+\infty},

\begin{exercise}[$\star\star$]\label{exo:g11:func:7}

  \item Leid het verloop van $g(x) = \frac{1}{(x-2)^2+1}$ en haar \omterm{def:g10:functions:extrema}{maximum} af.

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:8}
De kromme van een \omterm{def:g11:func:function}{functie} $f$ gaat door de punten $(0, 1)$, $(2, 3)$ en

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:9}
Toon aan dat de \omterm{def:g11:func:function}{functie} $f(x) = \frac{2x+1}{x+1}$ te schrijven is als

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:10}
Waar of niet waar (verantwoord of geef een tegenvoorbeeld)?
\begin{enumerate}
  \item De som van twee \omterm{def:g11:func:monotone}{stijgende functies} op $I$ is \omterm{def:g11:func:monotone}{stijgend} op $I$.
  \item Het product van twee \omterm{def:g11:func:monotone}{stijgende functies} op $I$ is \omterm{def:g11:func:monotone}{stijgend} op $I$.
  \item Is $f$ \omterm{def:g11:func:monotone}{stijgend} op $I$, dan is $f^2$ ($= f \times f$) \omterm{def:g11:func:monotone}{stijgend} op
        $I$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:func:11}
Zij $f(x) = x + \frac{1}{x}$ voor $x > 0$.
\begin{enumerate}
  \item Toon aan dat voor $0 < u < v$ geldt:
        $f(v) - f(u) = (v - u)\,\frac{uv - 1}{uv}$.
  \item Leid af dat $f$ strikt \omterm{def:g10:functions:variations}{dalend} is op $\intoc{0}{1}$ en strikt
        \omterm{def:g11:func:monotone}{stijgend} op $\intco{1}{+\infty}$.
  \item Besluit dat $x + \frac1x \geq 2$ voor alle $x > 0$, met gelijkheid
        alleen in $x = 1$.
\end{enumerate}
\end{exercise}

\section{Opgave: de symmetriefabriek}

\begin{problem}[{Weekendopgave --- elke functie is een even deel plus een
oneven deel: pariteitsregels, het transformatiespel, en minima gewonnen
zonder afgeleiden}]
\label{pb:g11:func:1}
Sommige \omterm{def:g11:func:function}{functies} zijn spiegelsymmetrisch, andere symmetrisch onder een halve
draai, de meeste geen van beide --- en toch zegt een klein algebraïsch
wonder dat \emph{elke} \omterm{def:g11:func:function}{functie} op precies één manier de som is van een
spiegelsymmetrisch en een halvedraaisymmetrisch stuk. Deze opgave bewijst het
wonder, speelt het transformatiespel op de referentiekrommen
(\cref{prop:g11:func:transforms}), en sluit af met de ontbindingstechniek van
\cref{exo:g11:func:11}, die minimaliseringsvraagstukken wint een volledig
hoofdstuk voordat de afgeleide arriveert.

\medskip
\noindent\textbf{Deel I --- Vlot met pariteit.}
\begin{enumerate}
  \item \omterm{def:g11:func:parity}{Even}, \omterm{def:g11:func:parity}{oneven}, of geen van beide (\cref{def:g11:func:parity})?
        Verantwoord elk geval: $x^4 - 3x^2$; \ $x^3 - x$; \ $x^2 + x$; \
        $\abs x$; \ $\frac1x$.
  \item Formuleer de meetkundige betekenis van elk oordeel
        (\cref{prop:g11:func:paritysym}): welke symmetrie heeft de kromme van
        een \omterm{def:g11:func:parity}{even functie}? En die van een \omterm{def:g11:func:parity}{oneven functie}?
  \item Pariteit vermenigvuldigt als tekens: bewijs dat het product van twee
        \omterm{def:g11:func:parity}{oneven functies} \omterm{def:g11:func:parity}{even} is, en vul de hele vermenigvuldigingstabel aan
        (\omterm{def:g11:func:parity}{even} $\times$ \omterm{def:g11:func:parity}{even}, \omterm{def:g11:func:parity}{even} $\times$ \omterm{def:g11:func:parity}{oneven}, \omterm{def:g11:func:parity}{oneven} $\times$ \omterm{def:g11:func:parity}{oneven}).
  \item Snelle gevolgen: is $f$ \omterm{def:g11:func:parity}{oneven} en in $0$ gedefinieerd, wat is dan
        $f(0)$? Is $f$ \omterm{def:g11:func:parity}{even} met $f(3) = 7$, wat is dan $f(-3)$? En toon aan
        dat de enige \omterm{def:g11:func:function}{functie} die zowel \omterm{def:g11:func:parity}{even} als \omterm{def:g11:func:parity}{oneven} is, de nulfunctie is.
  \item Hoeveel van de kromme van een \omterm{def:g11:func:parity}{even functie} legt haar volledig vast?
        Geef een \omterm{def:g11:func:function}{functie} die noch \omterm{def:g11:func:parity}{even} noch \omterm{def:g11:func:parity}{oneven} is, en leg uit wat er
        misgaat.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De splitsingsstelling.}
Stel voor een \omterm{def:g11:func:function}{functie} $f$ gedefinieerd op een \omterm{def:g11:func:function}{domein} dat symmetrisch is om
$0$:
\[
E(x) = \frac{f(x) + f(-x)}{2},
\qquad
O(x) = \frac{f(x) - f(-x)}{2} .
\]
\begin{enumerate}[resume]
  \item Bereken $E$ en $O$ voor $f(x) = x^3 + x^2 + 1$, en ga na: $E$ \omterm{def:g11:func:parity}{even},
        $O$ \omterm{def:g11:func:parity}{oneven}, $E + O = f$.
  \item Bewijs de algemene uitspraak: voor elke $f$ is de \omterm{def:g11:func:function}{functie} $E$ \omterm{def:g11:func:parity}{even},
        is $O$ \omterm{def:g11:func:parity}{oneven}, en geldt $f = E + O$ --- \emph{elke \omterm{def:g11:func:function}{functie} splitst in
        een \omterm{def:g11:func:parity}{even} en een \omterm{def:g11:func:parity}{oneven} deel}.
  \item Bewijs dat de splitsing uniek is: is $f = E_1 + O_1 = E_2 + O_2$ met
        $E_i$ \omterm{def:g11:func:parity}{even} en $O_i$ \omterm{def:g11:func:parity}{oneven}, gebruik dan vraag 4 op $E_1 - E_2$.
  \item Splits $f(x) = \dfrac{1}{1 - x}$ (voor $x \neq \pm 1$): breng $E$ en
        $O$ op de gemeenschappelijke noemer $1 - x^2$ en vereenvoudig.
  \item Een vooruitblik, telkens in één zin: het \omterm{def:g11:func:parity}{even} en het \omterm{def:g11:func:parity}{oneven} deel van
        de exponentiële \omterm{def:g11:func:function}{functie} zullen de tweelingfuncties van jaar 12 zijn
        (de hyperbolische cosinus en sinus); en hetzelfde instinct --- splits
        een object in symmetrische stukken en behandel elk met zijn eigen
        symmetrie --- drijft in de universitaire volumes de analyse van geluid
        en signalen aan. Waarom is zo'n splitsing in het algemeen
        \emph{nuttig}?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Het transformatiespel.}
\begin{enumerate}[resume]
  \item Beschrijf, vertrekkend van de kromme van $x \mapsto x^2$, nauwkeurig
        de krommen van $(x - 3)^2$, \ $x^2 + 2$, \ $(x - 3)^2 + 2$, \
        $-x^2$, \ $2x^2$ (\cref{prop:g11:func:transforms}).
  \item Herleid $f(x) = \abs{x - 2} + \abs{x + 2}$ tot een stuksgewijze
        formule op de drie intervallen die $-2$ en $2$ afsnijden, en
        beschrijf haar kromme (ze heeft een beroemde vlakke bodem).
  \item Welke transformaties behouden het \omterm{def:g11:func:parity}{even} zijn? Beslis, met een bewijs
        of een tegenvoorbeeld: de horizontale verschuiving $f(x - 3)$; de
        verticale verschuiving $f(x) + 2$; de verticale rek $2 f(x)$ (alle
        toegepast op een \omterm{def:g11:func:parity}{even} $f$).
  \item Los $\abs{x - 2} + \abs{x + 2} = 6$ op met de stuksgewijze formule
        van vraag 12.
  \item Terugrekenen: een \omterm{def:g11:quad:parabola}{parabool} heeft top $(2, -3)$ en gaat door
        $(0, 1)$. Reconstrueer haar \omterm{def:g10:algebra:equation}{vergelijking} in \omterm{thm:g11:quad:canonical}{toppuntsvorm}.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Minima zonder afgeleiden.}
\begin{enumerate}[resume]
  \item Spiegel \cref{exo:g11:func:11} voor $g(x) = x + \dfrac4x$ op
        $\intoo{0}{+\infty}$: toon aan dat
        $g(v) - g(u) = (v - u)\,\dfrac{uv - 4}{uv}$, bepaal het \omterm{def:g10:functions:extrema}{minimum} en
        geef zijn waarde.
  \item Pas toe: welke rechthoek met oppervlakte $16$ heeft de kleinste
        omtrek? (Schrijf $P = 2\left(x + \frac{16}{x}\right)$ en gebruik de
        machine van vraag 16.) Een oude bekende besluit.
  \item Bepaal het verloop van $h(x) = \sqrt{x^2 + 1}$ op heel $\R$, door een
        samenstellingsargument op $\intco{0}{+\infty}$ met een
        pariteitsargument voor de rest te combineren.
  \item Het dichtstbijzijnde punt: zoek het punt van de kromme $y = \sqrt x$
        dat het dichtst bij $(3, 0)$ ligt. (Minimaliseer het \emph{kwadraat}
        van de afstand --- een kwadratische uitdrukking in $x$ --- en leg uit
        waarom het kwadraat minimaliseren geoorloofd is.)
  \item Slotstuk, de rondleiding door de fabriek: de referentiekrommen zijn
        de grondstof; de transformaties zijn de machines; de pariteit en de
        splitsing zijn de kwaliteitscontrole; het \omterm{def:g10:algebra:expand}{ontbinden} van het verschil
        is de meetbank. Telkens één zin --- en noem het krachtige gereedschap
        dat het volgende hoofdstuk levert.
\end{enumerate}
\end{problem}
