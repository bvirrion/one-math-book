\chapter{Afgeleiden: een eerste kennismaking}\label{ch:g11:deriv}

Hoe snel verandert een \omterm{def:g11:func:function}{functie}? Het antwoord in één punt is de
\emph{\omterm{def:g11:deriv:derivative}{afgeleide}}, het nuttigste getal uit de hele toegepaste wiskunde. Dit
hoofdstuk bouwt haar op uit \omterm{def:g10:stats:mean}{gemiddelde} veranderingssnelheden, berekent haar
voor de \omterm{def:g11:func:reference}{referentiefuncties}, en gebruikt haar teken om het verloop van een
\omterm{def:g11:func:function}{functie} af te lezen. Alles hier wordt in \cref{ch:g12:deriv} uitgebreid en
verdiept.

\section{De gemiddelde veranderingssnelheid}

\begin{definition}[Gemiddelde veranderingssnelheid]\label{def:g11:deriv:rate}
Zij $f$ gedefinieerd op een \omterm{def:g10:numbers:interval}{interval} $I$, en $a \neq b$ in $I$. De
\emph{gemiddelde veranderingssnelheid}\index{gemiddelde
veranderingssnelheid} van $f$ tussen $a$ en $b$ is
\[
\frac{f(b) - f(a)}{b - a}.
\]
Het is de \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} van de \emph{snijlijn}\index{snijlijn} door de
punten $\bigl(a, f(a)\bigr)$ en $\bigl(b, f(b)\bigr)$ van de kromme.
\end{definition}

\begin{example}
Een auto legt tussen 14 en 16 uur $140$ km af: zijn \omterm{def:g10:stats:mean}{gemiddelde} snelheid is
$\frac{140}{2} = 70$ km/u. De snelheidsmeter toont daarentegen op elk
ogenblik een \emph{ogenblikkelijke} snelheid --- en dat is precies het begrip
dat de \omterm{def:g11:deriv:derivative}{afgeleide} scherp maakt.
\end{example}

\section{De afgeleide in een punt}

\begin{definition}[Afgeleide in een punt]\label{def:g11:deriv:derivative}
Zij $f$ gedefinieerd op een \omterm{def:g10:numbers:interval}{interval} dat $a$ bevat. Vorm voor $h \neq 0$ de
\omterm{def:g11:deriv:rate}{gemiddelde veranderingssnelheid} tussen $a$ en $a + h$:
\[
r(h) = \frac{f(a+h) - f(a)}{h}.
\]
Nadert $r(h)$ één vast getal naarmate $h$ willekeurig dicht bij $0$ komt, dan
zeggen we dat $f$ \emph{differentieerbaar in $a$}\index{afgeleide}\index{differentieerbaar}
is; dat getal is de \emph{afgeleide van $f$ in $a$}, geschreven $f'(a)$:
\[
f'(a) = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}.
\]
\end{definition}

\begin{example}\label{ex:g11:deriv:atpoint}
Zij $f(x) = x^2$ en $a = 1$. Voor $h \neq 0$:
\[
\frac{f(1+h) - f(1)}{h}
= \frac{(1+h)^2 - 1}{h}
= \frac{1 + 2h + h^2 - 1}{h}
= \frac{h(2 + h)}{h}
= 2 + h .
\]
Nadert $h$ tot $0$, dan nadert dit tot $2$: de \omterm{def:g11:func:function}{functie} $x^2$ is
\omterm{def:g11:deriv:derivative}{differentieerbaar} in $1$ en $f'(1) = 2$. Let op de strategie:
\emph{vereenvoudig het quotiënt tot $h$ niet langer in een noemer staat}, en
laat dan $h \to 0$ gaan.
\end{example}

\begin{example}[Een punt zonder afgeleide]
Zij $f(x) = \abs{x}$ en $a = 0$. Dan is
$\frac{f(h) - f(0)}{h} = \frac{\abs h}{h}$, wat $1$ is voor $h > 0$ en $-1$
voor $h < 0$: er wordt geen enkel getal genaderd, en $\abs x$ is niet
\omterm{def:g11:deriv:derivative}{differentieerbaar} in $0$. Haar kromme heeft daar een \emph{knik}.
\end{example}

\section{De raaklijn}

\begin{definition}[Raaklijn]\label{def:g11:deriv:tangent}
Is $f$ \omterm{def:g11:deriv:derivative}{differentieerbaar} in $a$, dan is de \emph{raaklijn}\index{raaklijn}
aan de kromme in het punt $A = \bigl(a, f(a)\bigr)$ de rechte door $A$ met
\omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $f'(a)$. Haar \omterm{def:g10:algebra:equation}{vergelijking} is
\[
y = f(a) + f'(a)(x - a).
\]
\end{definition}

De \omterm{def:g11:deriv:tangent}{raaklijn} is de grensstand van de \omterm{def:g11:deriv:rate}{snijlijnen} door $A$: naarmate $h \to 0$
glijdt het tweede punt $\bigl(a+h, f(a+h)\bigr)$ langs de kromme naar $A$, en
nadert de \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $\frac{f(a+h)-f(a)}{h}$ van de \omterm{def:g11:deriv:rate}{snijlijn} tot
$f'(a)$.

\begin{example}\label{ex:g11:deriv:tangenteq}
Voor $f(x) = x^2$ in $a = 1$: $f(1) = 1$ en $f'(1) = 2$
(\cref{ex:g11:deriv:atpoint}), dus is de \omterm{def:g11:deriv:tangent}{raaklijn}
\[
y = 1 + 2(x - 1) = 2x - 1 .
\]
Dat is de rechte $d_2$ die in \cref{exo:g11:quad:11} met zuivere algebra
gevonden werd.
\end{example}

\section{Afgeleiden van de referentiefuncties}

\begin{proposition}\label{prop:g11:deriv:refderivatives}
Op hun \omterm{def:g11:func:function}{domeinen} van differentieerbaarheid geldt
\[
(c)' = 0, \quad
(x)' = 1, \quad
(x^2)' = 2x, \quad
(x^3)' = 3x^2, \quad
\left(\frac1x\right)' = -\frac{1}{x^2}, \quad
(\sqrt x\,)' = \frac{1}{2\sqrt x} \ (x > 0).
\]
Algemener is $(x^n)' = n\,x^{n-1}$ voor elk \omterm{def:g10:numbers:sets}{geheel getal} $n \geq 1$.
\end{proposition}

\begin{proof}
Elk bewijs volgt de strategie van \cref{ex:g11:deriv:atpoint}: vereenvoudig
de veranderingssnelheid en laat dan $h \to 0$ gaan. Kies $a$ in het \omterm{def:g11:func:function}{domein}.

\emph{Constante en identiteit.} $\frac{c - c}{h} = 0$ en
$\frac{(a+h) - a}{h} = 1$ voor alle $h$: de limieten zijn $0$ en $1$.

\emph{Kwadraat.} $\frac{(a+h)^2 - a^2}{h} = \frac{2ah + h^2}{h} = 2a + h
\to 2a$.

\emph{Derde macht.} $(a+h)^3 = a^3 + 3a^2h + 3ah^2 + h^3$, zodat de
veranderingssnelheid $3a^2 + 3ah + h^2 \to 3a^2$ is.

\emph{Omgekeerde.} Voor $a \neq 0$ en $h$ klein genoeg opdat $a + h \neq 0$:
\[
\frac{1}{h}\left(\frac{1}{a+h} - \frac{1}{a}\right)
= \frac{1}{h} \cdot \frac{a - (a+h)}{a(a+h)}
= \frac{-1}{a(a+h)}
\xrightarrow[h \to 0]{} -\frac{1}{a^2}.
\]

\emph{Vierkantswortel.} Vermenigvuldig voor $a > 0$ met de toegevoegde
uitdrukking:
\[
\frac{\sqrt{a+h} - \sqrt a}{h}
= \frac{(a + h) - a}{h\,(\sqrt{a+h} + \sqrt a)}
= \frac{1}{\sqrt{a+h} + \sqrt a}
\xrightarrow[h \to 0]{} \frac{1}{2\sqrt a}.
\]

\emph{Algemene macht.} Het patroon $2a$, $3a^2$ loopt door: werk je
$(a+h)^n$ uit, dan dragen alle termen na $a^n + n\,a^{n-1}h$ een $h^2$,
zodat de veranderingssnelheid tot $n\,a^{n-1}$ nadert. (Een volledige
inductie staat in \cref{ch:g12:deriv}.)
\end{proof}

\section{Rekenregels voor afgeleiden}

\begin{theorem}[Bewerkingen]\label{thm:g11:deriv:rules}
Zijn $u$ en $v$ \omterm{def:g11:deriv:derivative}{differentieerbaar} op $I$ en is $\lambda \in \R$, dan zijn
$u + v$, $\lambda u$ en $uv$ \omterm{def:g11:deriv:derivative}{differentieerbaar} op $I$, en $\frac{u}{v}$
overal waar $v \neq 0$, met
\[
(u+v)' = u' + v', \qquad
(\lambda u)' = \lambda u', \qquad
(uv)' = u'v + uv', \qquad
\left(\frac{u}{v}\right)' = \frac{u'v - uv'}{v^2}.
\]
\end{theorem}

\begin{proof}[Bewijs van de eerste drie regels]
Kies $a \in I$ en kort af: $\Delta u = u(a+h) - u(a)$,
$\Delta v = v(a+h) - v(a)$.

\emph{Som.} De veranderingssnelheid van $u + v$ is
$\frac{\Delta u + \Delta v}{h} = \frac{\Delta u}{h} + \frac{\Delta v}{h}
\to u'(a) + v'(a)$.

\emph{Veelvoud.} Die van $\lambda u$ is
$\lambda \frac{\Delta u}{h} \to \lambda u'(a)$.

\emph{Product.} Tel de gemengde term $u(a+h)\,v(a)$ erbij op en trek hem er
weer af:
\begin{align*}
\frac{u(a+h)v(a+h) - u(a)v(a)}{h}
&= \frac{u(a+h)v(a+h) - u(a+h)v(a) + u(a+h)v(a) - u(a)v(a)}{h}\\
&= u(a+h)\,\frac{\Delta v}{h} + v(a)\,\frac{\Delta u}{h}.
\end{align*}
Voor $h \to 0$: $\frac{\Delta v}{h} \to v'(a)$,
$\frac{\Delta u}{h} \to u'(a)$, en $u(a+h) \to u(a)$ (haar
veranderingssnelheid heeft een limiet, dus
$\Delta u = h \cdot \frac{\Delta u}{h} \to 0$). Bijgevolg nadert de
veranderingssnelheid tot $u(a)v'(a) + v(a)u'(a)$.

De quotiëntregel wordt op dezelfde manier bewezen; ze wordt \emph{op dit
niveau aangenomen} en in \cref{ch:g12:deriv} bewezen.
\end{proof}

\begin{example}\label{ex:g11:deriv:computations}
\emph{Veelterm.} $f(x) = 3x^3 - 5x^2 + 7x - 2$:
\[
f'(x) = 3 \times 3x^2 - 5 \times 2x + 7 = 9x^2 - 10x + 7 .
\]

\emph{Quotiënt.} $g(x) = \frac{2x+1}{x^2+1}$: met $u = 2x + 1$ en
$v = x^2 + 1$ is
\[
g'(x) = \frac{2(x^2+1) - (2x+1)\,2x}{(x^2+1)^2}
= \frac{-2x^2 - 2x + 2}{(x^2+1)^2}.
\]

\emph{Product.} $k(x) = x\sqrt x$ voor $x > 0$:
$k'(x) = 1 \times \sqrt x + x \times \frac{1}{2\sqrt x}
= \sqrt x + \frac{\sqrt x}{2} = \frac{3\sqrt x}{2}$.
\end{example}

\section{Het teken van de afgeleide en het verloop}

\begin{theorem}[Afgeleide en verloop]\label{thm:g11:deriv:variations}
Zij $f$ \omterm{def:g11:deriv:derivative}{differentieerbaar} op een \omterm{def:g10:numbers:interval}{interval} $I$.
\begin{enumerate}
  \item Is $f' \geq 0$ op $I$, dan is $f$ \omterm{def:g11:func:monotone}{stijgend} op $I$.
  \item Is $f' \leq 0$ op $I$, dan is $f$ \omterm{def:g10:functions:variations}{dalend} op $I$.
  \item Is $f' > 0$ op $I$ (op hoogstens eindig veel punten na, waar ze
        verdwijnt), dan is $f$ strikt \omterm{def:g11:func:monotone}{stijgend} op $I$.
\end{enumerate}
\end{theorem}

\begin{proof}
De intuïtie is duidelijk --- een kromme waarvan alle \omterm{def:g11:deriv:tangent}{raaklijnen} omhoog wijzen
moet klimmen --- maar een streng bewijs vraagt de middelwaardestelling. Het
resultaat wordt \emph{op dit niveau aangenomen}; zie
\cref{thm:g12:deriv:variations}.
\end{proof}

\begin{proposition}[Extrema]\label{prop:g11:deriv:extrema}
Verandert $f'$ in $a$ van teken, van $+$ naar $-$, dan heeft $f$ in $a$ een
lokaal \omterm{def:g10:functions:extrema}{maximum}; van $-$ naar $+$ een lokaal \omterm{def:g10:functions:extrema}{minimum}. In zo'n punt is
$f'(a) = 0$ en is de \omterm{def:g11:deriv:tangent}{raaklijn} horizontaal.
\end{proposition}

\begin{proof}
Is $f' \geq 0$ op een \omterm{def:g10:numbers:interval}{interval} dat in $a$ eindigt en $f' \leq 0$ op een
\omterm{def:g10:numbers:interval}{interval} dat in $a$ begint, dan stijgt $f$ tot $f(a)$ en daalt ze daarna
(\cref{thm:g11:deriv:variations}): $f(a)$ is in de buurt de grootste waarde.
Omdat $f'$ in $a$ van teken verandert en daar gedefinieerd is, is
$f'(a) = 0$.
\end{proof}

\begin{method}[Een functie bestuderen]\label{met:g11:deriv:study}
\begin{enumerate}
  \item Bereken $f'(x)$ en vereenvoudig --- bij voorkeur tot een ontbonden
        vorm.
  \item Bepaal het teken van $f'(x)$ op elk \omterm{def:g10:numbers:interval}{interval} (is $f'$ kwadratisch,
        gebruik dan \cref{thm:g11:quad:sign}).
  \item Noteer de resultaten in een \emph{\omterm{def:g10:functions:table}{verlooptabel}}: één rij voor het
        teken van $f'$, één rij pijlen voor $f$, met de waarden van $f$ in de
        keerpunten.
  \item Besluit: extrema, aantal oplossingen van $f(x) = k$, ongelijkheden.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g11:deriv:study}
Bestudeer $f(x) = x^3 - 3x + 1$ op $\R$. Eerst is
$f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$: een drieterm met \omterm{def:g11:quad:discriminant}{wortels} $-1$ en $1$,
positief daarbuiten. Dus is $f$ \omterm{def:g11:func:monotone}{stijgend} op $\intoc{-\infty}{-1}$, \omterm{def:g10:functions:variations}{dalend} op
$\intcc{-1}{1}$ en \omterm{def:g11:func:monotone}{stijgend} op $\intco{1}{+\infty}$, met een lokaal \omterm{def:g10:functions:extrema}{maximum}
$f(-1) = -1 + 3 + 1 = 3$ en een lokaal \omterm{def:g10:functions:extrema}{minimum} $f(1) = 1 - 3 + 1 = -1$.
\end{example}

\begin{example}[Optimalisering]\label{ex:g11:deriv:box}
Uit een blad van $12 \times 12$ cm maak je een open doos door uit elke hoek
een vierkant met zijde $x$ te knippen en de randen op te plooien. Het volume
is $V(x) = x(12 - 2x)^2$ voor $0 < x < 6$. Uitgewerkt is
$V(x) = 4x^3 - 48x^2 + 144x$, dus
\[
V'(x) = 12x^2 - 96x + 144 = 12(x^2 - 8x + 12) = 12(x - 2)(x - 6).
\]
Op $\intoo{0}{6}$ is de factor $x - 6$ negatief, zodat $V'(x)$ het teken van
$-(x-2)$ heeft: positief vóór $2$, negatief erna. Het volume is maximaal
voor $x = 2$ cm, met $V(2) = 2 \times 8^2 = 128$ cm$^3$.
\end{example}

\begin{exercise}[$\star$]\label{exo:g11:deriv:1}
Bereken met de definitie (veranderingssnelheid, en dan $h \to 0$) $f'(2)$
voor $f(x) = x^2$, en daarna $g'(1)$ voor $g(x) = x^2 + 3x$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:deriv:2}

f(x) = 4x^3 - 2x^2 + x - 7, \qquad
g(x) = 5\sqrt{x} + \frac{3}{x}, \qquad
h(x) = (2x+1)(x^2 - 3) .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:deriv:3}
Geef de \omterm{def:g10:algebra:equation}{vergelijking} van de \omterm{def:g11:deriv:tangent}{raaklijn} aan de kromme van $f$ in het punt met
\omterm{def:g10:coordgeom:system}{abscis} $a$:
\[
f(x) = x^2 - 3x, \ a = 2; \qquad
f(x) = \frac1x, \ a = 1; \qquad
f(x) = \sqrt x, \ a = 4 .

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:deriv:8}

\begin{exercise}[$\star\star$]\label{exo:g11:deriv:10}

\section{Opgave: de drie beroepen van de raaklijn}

\begin{problem}[{Weekendopgave --- de parabolische spiegel bewezen, de
wortelzoekmachine van Newton, en de sterkste balk in de boomstam}]
\label{pb:g11:deriv:1}
Een \omterm{def:g11:deriv:derivative}{afgeleide} tekent \omterm{def:g11:deriv:tangent}{raaklijnen}; dat klinkt als een bescheiden talent. Deze
opgave laat zien dat de \omterm{def:g11:deriv:tangent}{raaklijn} drie beroepen tegelijk uitoefent: ze bewijst
de koplampstelling die in \cref{pb:g11:quad:1} bleef openstaan (elke straal
evenwijdig met de as van een \omterm{def:g11:quad:parabola}{parabool} weerkaatst door het \omterm{pb:g11:quad:1}{brandpunt}), ze
drijft het snelste veelgebruikte algoritme aan om \omterm{def:g10:algebra:equation}{vergelijkingen} op te lossen
--- het zit in elke rekenmachine en elk neuraal netwerk --- en ze vindt de
sterkste rechthoekige balk die een ronde boomstam kan leveren.

\medskip
\noindent\textbf{Deel I --- Vlot met \omterm{def:g11:deriv:tangent}{raaklijnen}.}
\begin{enumerate}
  \item Leid $f(x) = 3x^2 - 5x + 1$ en $g(x) = x^3 - 12x$ af
        (\cref{thm:g11:deriv:rules}); en haal $\left(x^2\right)'$ in $x = 1$
        terug uit de definitie (veranderingssnelheid, en dan $h \to 0$).
  \item Zoek de \omterm{def:g11:deriv:tangent}{raaklijn} aan $y = x^3 - 12x$ in $a = 2$. Wat is er
        bijzonder aan, en wat signaleert dat
        (\cref{prop:g11:deriv:extrema})?
  \item Stel de algemene \omterm{def:g11:deriv:tangent}{raaklijn} aan de \omterm{def:g11:quad:parabola}{parabool} $y = x^2$ in het punt
        $(a, a^2)$ op:
        \[
        y = 2ax - a^2 .
        \]
  \item Waar snijdt die \omterm{def:g11:deriv:tangent}{raaklijn} de $y$-as? Leid het recept van de tekenaar
        af: om de \omterm{def:g11:deriv:tangent}{raaklijn} te tekenen in een punt $P$ op hoogte $h$ van de
        \omterm{def:g11:quad:parabola}{parabool}, verbind je $P$ met het punt van de as op een diepte\,\dots\
        onder de oorsprong. Formuleer het.
  \item Bouw de volledige \omterm{def:g10:functions:table}{verlooptabel} van $g(x) = x^3 - 12x$ (teken van
        $g'$, lokale extrema en hun waarden, \cref{met:g11:deriv:study}).
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De spiegelstelling.}
Zij $P(a, a^2)$ een punt van de \omterm{def:g11:quad:parabola}{parabool} $y = x^2$ (neem $a \neq 0$),
$F\left(0, \frac14\right)$ het \omterm{pb:g11:quad:1}{brandpunt}, en $T$ het punt waar de \omterm{def:g11:deriv:tangent}{raaklijn} in
$P$ de $y$-as snijdt (vraag 4). Denk eraan dat uit \cref{pb:g11:quad:1} volgt
dat $PF = a^2 + \frac14$.
\begin{enumerate}[resume]
  \item Bereken de afstand $FT$.
  \item Besluit dat de driehoek $FPT$ gelijkbenig is in $F$. Welke twee
        hoeken zijn bijgevolg gelijk?
  \item De invallende straal door $P$, evenwijdig met de as, is evenwijdig
        met de rechte $(TF)$ (beide verticaal). Toon met verwisselende hoeken
        aan dat de \omterm{def:g11:deriv:tangent}{raaklijn} in $P$ \emph{gelijke hoeken} maakt met de
        invallende verticale straal en met het lijnstuk $[PF]$.
  \item Natuurkunde: licht weerkaatst op een kromme zoals op haar \omterm{def:g11:deriv:tangent}{raaklijn},
        en vertrekt onder dezelfde hoek als waaronder het aankwam (de
        weerkaatsingswet). Besluit tot de \emph{spiegelstelling}: elke straal
        evenwijdig met de as weerkaatst door het \omterm{pb:g11:quad:1}{brandpunt} --- en, met de tijd
        omgekeerd, straalt een lampje in het \omterm{pb:g11:quad:1}{brandpunt} evenwijdig licht uit.
        De schotel en de koplamp van \cref{pb:g11:quad:1} zijn nu stellingen.
  \item Ga de gelijkbenige driehoek na met getallen in $a = 1$: bereken $T$,
        $FP$ en $FT$.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De machine van Newton.}
Om $f(x) = 0$ op te lossen stelde Newton voor: vertrek van een schatting
$x_0$, volg de \omterm{def:g11:deriv:tangent}{raaklijn} in $x_0$ omlaag tot aan de as, en neem haar snijpunt
met de as als volgende schatting.
\begin{enumerate}[resume]
  \item Leid de formule van de methode af:
        \[
        x_{n + 1} = x_n - \frac{f(x_n)}{f'(x_n)} .
        \]
  \item Laat ze lopen op $f(x) = x^2 - 2$ vanaf $x_0 = 1$: bereken $x_1$,
        $x_2$, $x_3$ als exacte breuken. Welke tweeduizend jaar oude rij ---
        en welke machine uit \cref{pb:g10:functions:1} --- heb je zojuist
        herbouwd? Bewijs de identiteit achter het toeval:
        \[
        x - \frac{x^2 - 2}{2x} = \frac12\left(x + \frac2x\right).
        \]
  \item Los $x^3 = 100$ op (het omslagpunt van de spaarplannen uit
        \cref{pb:g10:reffunc:1}): pas twee stappen van Newton toe op
        $f(x) = x^3 - 100$ vanaf $x_0 = 5$ (vier decimalen), en vergelijk met
        de $\sqrt[3]{100}$ van de rekenmachine.
  \item Sabotage: probeer de methode te starten op $f(x) = x^3 - 12x$ in
        $x_0 = 2$. Wat loopt er mis, en waarom (vraag 2)? Formuleer het
        praktische voorbehoud.
  \item In vraag 12 gingen de juiste decimalen ruwweg $1 \to 3 \to 6$.
        Vergelijk met bisectie (elke stap een \omterm{def:g10:numbers:interval}{interval} halveren) en zeg in
        één zin waarom rekenmachines, gps-ontvangers en het trainen van
        neurale netwerken alle op het idee van Newton draaien.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De sterkste balk.}
Uit een ronde boomstam met diameter $30$ cm wordt een rechthoekige balk
gezaagd met breedte $w$ en hoogte $d$, zodat $w^2 + d^2 = 900$. De stijfheid
van een balk is evenredig met $w d^2$ (de hoogte telt dubbel: balken staan op
hun kant!).
\begin{enumerate}[resume]
  \item Druk de stijfheid uit als \omterm{def:g11:func:function}{functie} van $w$ alleen:
        $S(w) = 900w - w^3$, op welk \omterm{def:g10:numbers:interval}{interval}?
  \item Leid af, bouw de \omterm{def:g10:functions:table}{verlooptabel}, en zoek de optimale $w$ en $d$
        (exacte waarden, en daarna op de millimeter).
  \item Toon aan dat de optimale verhoudingen voldoen aan $d = w\sqrt2$ ---
        de oude verhouding van de timmerlieden. (Extra weetje: de diameter
        van de stam in drie gelijke stukken verdelen en loodlijnen naar de
        cirkel oprichten construeert precies deze rechthoek.)
  \item De voor de hand liggende keuze is de \emph{vierkante} balk
        ($w = d$). Bereken haar stijfheid en die van de optimale balk, en
        geef de winst van de timmerman in procent.
  \item Slotstuk --- de drie beroepen van de \omterm{def:g11:deriv:tangent}{raaklijn}, telkens één zin:
        meetkunde (de spiegelstelling), numeriek rekenen (de machine van
        Newton), optimaliseren (de balk), alle aangedreven door hetzelfde
        object. En de vooruitblik: jaar 12 voegt de convexiteit toe --- aan
        welke kant van de \omterm{def:g11:deriv:tangent}{raaklijn} de kromme ligt --- om alle drie te
        verscherpen.
\end{enumerate}
\end{problem}
