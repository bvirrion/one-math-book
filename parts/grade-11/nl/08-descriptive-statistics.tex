\chapter{Beschrijvende statistiek}\label{ch:g11:stat}

De statistiek perst een lijst getallen samen tot enkele betekenisvolle
kengetallen: waar de gegevens liggen (\omterm{def:g11:stat:mean}{gemiddelde}, \omterm{def:g11:stat:median}{mediaan}) en hoe verspreid
ze zijn (\omterm{def:g11:stat:quartiles}{kwartielen}, \omterm{def:g11:stat:variance}{variantie}, \omterm{def:g11:stat:variance}{standaardafwijking}). Dit hoofdstuk werkt alles
uit op één doorlopend voorbeeld. Dezelfde kengetallen duiken opnieuw op bij
toevalsvariabelen in \cref{ch:g11:prob}, en hun kanswetten in
\cref{ch:g12:sums}.

Doorheen het hoofdstuk is onze gegevensreeks het puntenaantal (op $20$) van
een klas met $12$ leerlingen:
\[
8,\ 9,\ 10,\ 10,\ 11,\ 12,\ 12,\ 12,\ 13,\ 14,\ 16,\ 17 .
\]

\section{Gemiddelde en mediaan}

\begin{definition}[Gemiddelde]\label{def:g11:stat:mean}
Het \emph{gemiddelde}\index{gemiddelde} van de waarden
$x_1, x_2, \dots, x_n$ is
\[
\bar x = \frac{x_1 + x_2 + \dots + x_n}{n}.
\]
Komt de waarde $x_i$ voor met \omterm{def:g10:stats:series}{frequentie} $n_i$ (met
$n_1 + \dots + n_k = n$), dan is
$\bar x = \frac{n_1 x_1 + \dots + n_k x_k}{n}$.
\end{definition}

\begin{definition}[Mediaan]\label{def:g11:stat:median}
Een \emph{mediaan}\index{mediaan} van een gesorteerde gegevensreeks is een
waarde die haar in twee helften splitst: minstens de helft van de waarden is
kleiner dan of gelijk aan de mediaan, en minstens de helft is groter dan of
gelijk. In de praktijk sorteer je de $n$ waarden; is $n$ oneven, dan is de
mediaan de middelste waarde; is $n$ even, neem dan het \omterm{prop:g10:coordgeom:midpoint}{midden} van de twee
middelste waarden.
\end{definition}

\begin{example}\label{ex:g11:stat:meanmedian}
Voor onze gegevensreeks is de som $144$, dus $\bar x = \frac{144}{12} = 12$.
De twee middelste waarden (de $6$de en de $7$de van de gesorteerde lijst)
zijn allebei $12$, dus is ook de \omterm{def:g11:stat:median}{mediaan} $12$. \omterm{def:g11:stat:mean}{Gemiddelde} en \omterm{def:g11:stat:median}{mediaan} moeten
niet samenvallen: de \omterm{def:g11:stat:median}{mediaan} negeert \emph{hoe ver} de uiterste waarden
liggen, het \omterm{def:g11:stat:mean}{gemiddelde} niet. Vervang je het hoogste punt $17$ door $20$, dan
schuift het \omterm{def:g11:stat:mean}{gemiddelde} naar $12.25$, terwijl de \omterm{def:g11:stat:median}{mediaan} op $12$ blijft.
\end{example}

\section{Kwartielen}

\begin{definition}[Kwartielen, interkwartielafstand]\label{def:g11:stat:quartiles}
Het \emph{eerste kwartiel}\index{kwartiel} $Q_1$ is de kleinste waarde
waarvoor minstens een kwart van de gegevens $\leq Q_1$ is; het \emph{derde
kwartiel} $Q_3$ is de kleinste waarde waarvoor minstens drie kwart van de
gegevens $\leq Q_3$ is. De
\emph{interkwartielafstand}\index{interkwartielafstand} is $Q_3 - Q_1$: de
breedte van de middelste helft van de gegevens.
\end{definition}

\begin{example}
$n = 12$: een kwart is $3$, dus is $Q_1$ de $3$de gesorteerde waarde:
$Q_1 = 10$. Drie kwart is $9$, dus is $Q_3$ de $9$de waarde: $Q_3 = 13$. De
\omterm{def:g11:stat:quartiles}{interkwartielafstand} is $3$: de middelste helft van de klas ligt binnen $3$
punten.
\end{example}

\section{Variantie en standaardafwijking}

\begin{definition}[Variantie, standaardafwijking]\label{def:g11:stat:variance}
De \emph{variantie}\index{variantie} van de gegevensreeks is het \omterm{def:g11:stat:mean}{gemiddelde}
van de gekwadrateerde afwijkingen van het \omterm{def:g11:stat:mean}{gemiddelde}:
\[
V = \frac{1}{n}\sum_{i=1}^{n} (x_i - \bar x)^2,
\]
en de \emph{standaardafwijking}\index{standaardafwijking} is
$\sigma = \sqrt{V}$, uitgedrukt in dezelfde eenheid als de gegevens.
\end{definition}

\begin{proposition}[Rekenformule]\label{prop:g11:stat:shortcut}
De \omterm{def:g11:stat:variance}{variantie} is het \omterm{def:g11:stat:mean}{gemiddelde} van de kwadraten min het kwadraat van het
\omterm{def:g11:stat:mean}{gemiddelde}:
\[
V = \frac{1}{n}\sum_{i=1}^{n} x_i^2 \;-\; \bar x^{\,2}.
\]
\end{proposition}

\begin{proof}
Werk elke gekwadrateerde afwijking uit:
$(x_i - \bar x)^2 = x_i^2 - 2\bar x\, x_i + \bar x^2$. Optellen en door $n$
delen geeft
\[
V = \frac1n \sum x_i^2 - 2\bar x \cdot \underbrace{\frac1n \sum
x_i}_{=\ \bar x} + \frac1n \cdot n\bar x^2
= \frac1n\sum x_i^2 - 2\bar x^2 + \bar x^2
= \frac1n\sum x_i^2 - \bar x^2 .
\]
Dezelfde identiteit, voor toevalsvariabelen, is
\cref{prop:g11:prob:konig}.
\end{proof}

\begin{example}\label{ex:g11:stat:variance}
Voor onze gegevensreeks zijn de afwijkingen van $\bar x = 12$ gelijk aan
$-4$, $-3$, $-2$, $-2$, $-1$, $0$, $0$, $0$, $1$, $2$, $4$, $5$; hun
kwadraten tellen op tot
\[
16 + 9 + 4 + 4 + 1 + 0 + 0 + 0 + 1 + 4 + 16 + 25 = 80 .
\]
Bijgevolg is
\[
V = \frac{80}{12} = \frac{20}{3} \approx 6.67,
\qquad
\sigma = \sqrt{20/3} \approx 2.6 \text{ punten}.
\]
Ruwweg gezegd ligt een typisch resultaat ongeveer $2.6$ punten van het
\omterm{def:g11:stat:mean}{gemiddelde} af.
\end{example}

\begin{proposition}[Verandering van eenheid]\label{prop:g11:stat:affine}
Wordt elke waarde omgezet door $y_i = a x_i + b$, dan geldt
\[
\bar y = a \bar x + b,
\qquad
V_y = a^2\, V_x,
\qquad
\sigma_y = \abs{a}\,\sigma_x .
\]
\end{proposition}

\begin{proof}
$\bar y = \frac1n \sum (a x_i + b) = a\left(\frac1n\sum x_i\right) +
\frac1n \cdot nb = a\bar x + b$. Vervolgens is
$y_i - \bar y = a(x_i - \bar x)$: elke afwijking wordt met $a$ geschaald (de
verschuiving $b$ valt weg), dus elke gekwadrateerde afwijking met $a^2$, en
dus ook hun \omterm{def:g11:stat:mean}{gemiddelde}. De vierkantswortel trekken geeft
$\sigma_y = \abs a\,\sigma_x$.
\end{proof}

\begin{example}
Temperaturen die in graden Celsius genoteerd zijn met \omterm{def:g11:stat:mean}{gemiddelde} $20$ en
\omterm{def:g11:stat:variance}{standaardafwijking} $3$ zet je naar Fahrenheit om met $F = 1.8\,C + 32$: het
\omterm{def:g11:stat:mean}{gemiddelde} wordt $1.8 \times 20 + 32 = 68$ en de \omterm{def:g11:stat:variance}{standaardafwijking}
$1.8 \times 3 = 5.4$. Een gegevensreeks verschuiven verandert haar spreiding
niet; ze herschalen herschaalt de spreiding.
\end{example}

\begin{method}[Gegevensreeksen samenvatten en vergelijken]\label{met:g11:stat:compare}
Om twee gegevensreeksen te vergelijken zet je naast elkaar: het \omterm{def:g11:stat:mean}{gemiddelde}
(of de \omterm{def:g11:stat:median}{mediaan}) voor de \emph{ligging}, en de \omterm{def:g11:stat:variance}{standaardafwijking} (of de
\omterm{def:g11:stat:quartiles}{interkwartielafstand}) voor de \emph{spreiding}. Het paar (\omterm{def:g11:stat:median}{mediaan},
\omterm{def:g11:stat:quartiles}{interkwartielafstand}) is robuust tegen extreme waarden; het paar (\omterm{def:g11:stat:mean}{gemiddelde},
\omterm{def:g11:stat:variance}{standaardafwijking}) gebruikt elke waarde en voedt de kansrekening.
\end{method}

\begin{example}
Twee boogschutters schieten elk $10$ pijlen; hun scores hebben allebei
\omterm{def:g11:stat:mean}{gemiddelde} $8.5$, maar de \omterm{def:g11:stat:variance}{standaardafwijkingen} zijn $0.5$ en $2.1$. Gemiddeld
even sterk --- maar de eerste boogschutter is veel regelmatiger.
\end{example}

\section{Opgave: de z-score, een universele meetlat}

\begin{problem}[{Weekendopgave --- standaardafwijkingen vergelijken appelen
met peren, de ongelijkheid van Chebyshev maakt van $\sigma$ een garantie, en
het gemiddelde wint zijn eigen medaille}]
\label{pb:g11:stat:1}
Is $15/20$ op een moeilijk examen beter dan $13/20$ op een makkelijk? Is een
$23$ in een lijst kleine getallen een toevalligheid of een uitschieter? Ruwe
getallen kunnen dat niet zeggen --- maar gedeeld door de juiste
\omterm{def:g11:stat:variance}{standaardafwijking} spreken ze allemaal één taal, die van de
\emph{\omterm{pb:g11:stat:1}{z-score}}. Deze opgave beheerst de machinerie van de \omterm{def:g11:stat:variance}{variantie}
(\cref{def:g11:stat:variance}, \cref{prop:g11:stat:shortcut}), bouwt de
universele meetlat, en bewijst de merkwaardige ongelijkheid die van $\sigma$
meer maakt dan een beschrijvend getal: een garantie.

\medskip
\noindent\textbf{Deel I --- Mechanica van de \omterm{def:g11:stat:variance}{variantie}.}
\begin{enumerate}
  \item Bereken voor de gegevensreeks $4, 8, 6, 5, 2$ het \omterm{def:g11:stat:mean}{gemiddelde}, daarna
        de \omterm{def:g11:stat:variance}{variantie} rechtstreeks uit de definitie, en daarna $\sigma$.
  \item Bereken de \omterm{def:g11:stat:variance}{variantie} opnieuw met de rekenformule
        (\cref{prop:g11:stat:shortcut}) --- het \omterm{def:g11:stat:mean}{gemiddelde} van de kwadraten
        min het kwadraat van het \omterm{def:g11:stat:mean}{gemiddelde} --- en ga de overeenstemming na.
  \item Met \omterm{def:g10:stats:series}{frequenties}: de punten $8$, $10$, $14$ werden behaald door $3$,
        $5$ en $2$ leerlingen. Bereken het \omterm{def:g11:stat:mean}{gemiddelde}, de \omterm{def:g11:stat:variance}{variantie} en
        $\sigma$ (twee decimalen).
  \item Temperaturen hebben \omterm{def:g11:stat:mean}{gemiddelde} $20$ en $\sigma = 3$ (in graden
        Celsius). Geef het \omterm{def:g11:stat:mean}{gemiddelde} en $\sigma$ in Fahrenheit
        ($F = 1.8\,C + 32$; \cref{prop:g11:stat:affine}).
  \item Bewijs de twee regels die je zonet gebruikte: een constante $b$ bij
        elke waarde optellen verschuift het \omterm{def:g11:stat:mean}{gemiddelde} met $b$ en laat
        $\sigma$ ongemoeid; met $a$ vermenigvuldigen vermenigvuldigt het
        \omterm{def:g11:stat:mean}{gemiddelde} met $a$ en $\sigma$ met $\abs a$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De universele meetlat.}
De \emph{z-score}\index{z-score} van een waarde $x$ in een gegevensreeks met
\omterm{def:g11:stat:mean}{gemiddelde} $\bar x$ en \omterm{def:g11:stat:variance}{standaardafwijking} $\sigma$ is
$z = \dfrac{x - \bar x}{\sigma}$: het aantal \omterm{def:g11:stat:variance}{standaardafwijkingen} tussen $x$
en het \omterm{def:g11:stat:mean}{gemiddelde}.
\begin{enumerate}[resume]
  \item Examen A heeft \omterm{def:g11:stat:mean}{gemiddelde} $12$ en $\sigma = 2$; examen B heeft
        \omterm{def:g11:stat:mean}{gemiddelde} $8$ en $\sigma = 4$. Alice haalde $15$ op A, Bob $13$ op
        B. Bereken beide \omterm{pb:g11:stat:1}{z-scores}: wiens prestatie steekt verder boven de
        massa uit?
  \item Bereken de vijf \omterm{pb:g11:stat:1}{z-scores} van de gegevensreeks uit vraag 1, en ga na
        dat ze \omterm{def:g11:stat:mean}{gemiddelde} $0$ en \omterm{def:g11:stat:variance}{standaardafwijking} $1$ hebben. Leg met vraag
        5 uit waarom standaardiseren \emph{altijd} \omterm{def:g11:stat:mean}{gemiddelde} $0$ en
        $\sigma = 1$ oplevert.
  \item Een aanlegtest heeft \omterm{def:g11:stat:mean}{gemiddelde} $100$ en $\sigma = 15$. Welke ruwe
        score hoort bij $z = 2$? En wat is de \omterm{pb:g11:stat:1}{z-score} van $76$?
  \item Een kinderarts noteert het gewicht van een baby als ``$z = -2.5$''.
        Zeg wat dat betekent, en waarom groei in \omterm{pb:g11:stat:1}{z-scores} en niet in
        kilogram wordt gevolgd naarmate het kind ouder wordt.
  \item De uitschieter uit \cref{exo:g11:stat:10} (de reeks
        $1, 2, 2, 3, 3, 3, 4, 4, 5, 23$): bereken het \omterm{def:g11:stat:mean}{gemiddelde}, $\sigma$ en
        de \omterm{pb:g11:stat:1}{z-score} van $23$. De klassieke alarmdrempel is $\abs z \geq 3$:
        wat is het oordeel?
  \item Bereken $\sigma$ opnieuw \emph{zonder} de waarde $23$, en de \omterm{pb:g11:stat:1}{z-score}
        die $23$ zou krijgen tegenover het \omterm{def:g11:stat:mean}{gemiddelde} en de $\sigma$ van de
        schone gegevens. Wat deed de uitschieter met de meetlat die hem juist
        moest meten, en welke tegenzet stelt \cref{exo:g11:stat:10} voor?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De garantie van Chebyshev.}
\begin{enumerate}[resume]
  \item Bewijs de ongelijkheid van Bienaymé en Chebyshev voor een
        gegevensreeks: voldoet een aandeel $p$ van de waarden aan
        $\abs{x - \bar x} \geq k\sigma$, dan geeft het behouden van alleen
        die termen in de variantiesom
        \[
        \sigma^2 \geq p \cdot (k\sigma)^2,
        \qquad\text{en dus}\qquad
        p \leq \frac{1}{k^2} .
        \]
  \item Wat garandeert de ongelijkheid over het aandeel van eender welke
        gegevensreeks verder dan $2\sigma$ van haar \omterm{def:g11:stat:mean}{gemiddelde}? En verder dan
        $3\sigma$? Ga de uitspraak over $2\sigma$ na op de gegevensreeks uit
        vraag 1.
  \item Een fabriek maakt bouten met \omterm{def:g11:stat:mean}{gemiddelde} $50$ mm en $\sigma = 0.1$ mm.
        Wat garandeert Chebyshev over het aandeel bouten buiten
        $\intcc{49.7}{50.3}$? (In werkelijkheid doet de productie het veel
        beter --- de klokkromme van jaar 12 maakt van dit voorzichtige
        $\frac19$ een fractie van een procent.)
  \item Twee fondsen brengen allebei gemiddeld $5\,\%$ per jaar op, met
        $\sigma = 2\,\%$ voor fonds A en $\sigma = 12\,\%$ voor fonds B. Wat
        zegt Chebyshev voor elk fonds over het aandeel jaren met een
        rendement onder $-19\,\%$? Hoe noemt de financiële wereld $\sigma$?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De medaille van het \omterm{def:g11:stat:mean}{gemiddelde}, en een
waarschuwing.}
\begin{enumerate}[resume]
  \item Zij $f(c)$ het \omterm{def:g11:stat:mean}{gemiddelde} van de gekwadrateerde afwijkingen
        $(x_i - c)^2$ ten opzichte van een willekeurig middelpunt $c$. Bewijs
        de identiteit
        \[
        f(c) = \sigma^2 + (\bar x - c)^2 ,
        \]
        en besluit: het \omterm{def:g11:stat:mean}{gemiddelde} is de unieke minimalisator van de
        \omterm{def:g11:stat:mean}{gemiddelde} gekwadrateerde afwijking --- zijn medaille, tegenhanger van
        de minimalisering van absolute afwijkingen door de \omterm{def:g11:stat:median}{mediaan} in
        \cref{pb:g10:stats:1}.
  \item Ga de identiteit na op de gegevensreeks uit vraag 1 met $c = 6$:
        bereken $f(6)$ rechtstreeks en via de formule.
  \item Dezelfde kengetallen, andere gegevens: ga na dat
        $P = \{3, 5, 7, 9, 11\}$ en $Q = \{1, 7, 7, 7, 13\}$ hetzelfde
        \omterm{def:g11:stat:mean}{gemiddelde} maar niet dezelfde \omterm{def:g11:stat:variance}{variantie} hebben; bouw daarna een
        gegevensreeks van de vorm $\{7 - a, 7, 7, 7, 7 + a\}$ met precies het
        \omterm{def:g11:stat:mean}{gemiddelde} en de \omterm{def:g11:stat:variance}{variantie} van $P$ --- en toch een heel eigen vorm.
  \item Bewijs dat $\sigma = 0$ afdwingt dat alle waarden gelijk zijn. (Wat
        moet elke term van een som van kwadraten die nul is, zelf zijn?)
  \item Slotstuk --- de ladder van de statisticus, één regel per sport:
        \omterm{prop:g10:coordgeom:midpoint}{middens} (\omterm{def:g11:stat:mean}{gemiddelde}, \omterm{def:g11:stat:median}{mediaan}), spreidingen ($\sigma$,
        \omterm{def:g11:stat:quartiles}{interkwartielafstand}), de universele meetlat ($z$), de garantie
        (Chebyshev), en de waarschuwing van vraag 18: kengetallen persen
        samen, en samenpersen verliest --- met de klokkromme van jaar 12 die
        klaarstaat om \omterm{pb:g11:stat:1}{z-scores} in exacte kansen om te zetten.
\end{enumerate}
