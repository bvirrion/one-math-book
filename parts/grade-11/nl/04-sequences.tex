\chapter{Rijen: een eerste kennismaking}\label{ch:g11:seq}

Een \omterm{def:g11:seq:sequence}{rij} is een lijst getallen die door een voorschrift wordt voortgebracht: de
opeenvolgende saldi van een spaarrekening, de omvang van een populatie jaar na
jaar. Dit hoofdstuk bestudeert de twee families die de toepassingen
beheersen --- \emph{rekenkundige} \omterm{def:g11:seq:sequence}{rijen}, die met gelijke stappen groeien, en
\emph{meetkundige} \omterm{def:g11:seq:sequence}{rijen}, die met gelijke verhoudingen groeien. De strenge
theorie van de limieten wordt in \cref{ch:g12:seq} opgebouwd.

\section{Een rij vastleggen}

\begin{definition}[Rij]\label{def:g11:seq:sequence}
Een \emph{rij}\index{rij} $(u_n)$ kent aan elk \omterm{def:g10:numbers:sets}{geheel getal} $n \geq 0$ (of
$n \geq 1$) een \omterm{def:g10:numbers:sets}{reëel getal} $u_n$ toe, haar \emph{term met index $n$}. Een
rij kan gegeven worden
\begin{itemize}
  \item \emph{expliciet}, door een formule voor $u_n$ in \omterm{def:g11:func:function}{functie} van $n$:
        bijvoorbeeld $u_n = n^2 + 1$;
  \item \emph{recursief}, door haar eerste term en een voorschrift om van
        elke term naar de volgende te gaan: bijvoorbeeld $u_0 = 3$ en
        $u_{n+1} = 2u_n - 1$.
\end{itemize}
\end{definition}

\begin{example}
Voor $u_n = n^2 + 1$: $u_0 = 1$, $u_1 = 2$, $u_2 = 5$, en rechtstreeks
$u_{10} = 101$. Voor $u_0 = 3$, $u_{n+1} = 2u_n - 1$: $u_1 = 5$, $u_2 = 9$,
$u_3 = 17$ --- elke term heeft de vorige nodig; tot bij $u_{10}$ geraken
kost tien stappen (of een algemene formule, zie \cref{exo:g11:seq:11}).
\end{example}

\section{Rekenkundige rijen}

\begin{definition}[Rekenkundige rij]\label{def:g11:seq:arithmetic}
Een \omterm{def:g11:seq:sequence}{rij} heet \emph{rekenkundig}\index{rekenkundige rij} met
\emph{verschil}\index{verschil} $d$ wanneer elke term uit de vorige ontstaat
door er $d$ bij op te tellen:
\[
u_{n+1} = u_n + d \quad \text{voor alle } n.
\]
Gelijkwaardig: het verschil $u_{n+1} - u_n$ is constant en gelijk aan $d$.
\end{definition}

\begin{theorem}[Algemene term]\label{thm:g11:seq:arithgeneral}
Is $(u_n)$ \omterm{def:g11:seq:arithmetic}{rekenkundig} met eerste term $u_0$ en \omterm{def:g11:seq:arithmetic}{verschil} $d$, dan geldt
\[
u_n = u_0 + n\,d \quad \text{voor alle } n \geq 0,
\qquad\text{en algemener } u_n = u_p + (n - p)\,d .
\]
\end{theorem}

\begin{proof}
Om van $u_0$ tot $u_n$ te komen wordt het voorschrift ``tel $d$ op'' $n$ keer
toegepast: één stap geeft $u_1 = u_0 + d$, twee stappen geven
$u_2 = u_0 + 2d$, en na $n$ stappen heeft elke toepassing één $d$
bijgedragen, dus $u_n = u_0 + nd$. (Dat ``enzovoort'' wordt in
\cref{ch:g12:seq} met inductie streng gemaakt.) De algemene formule volgt door
de $n - p$ stappen van $u_p$ naar $u_n$ te tellen.
\end{proof}

\begin{theorem}[Som van opeenvolgende gehele getallen]\label{thm:g11:seq:intsum}
Voor elk \omterm{def:g10:numbers:sets}{geheel getal} $n \geq 1$ geldt
\[
1 + 2 + \dots + n = \frac{n(n+1)}{2}.
\]
Algemener is een som van opeenvolgende termen van een \omterm{def:g11:seq:arithmetic}{rekenkundige rij} gelijk
aan
\[
(\text{aantal termen}) \times
\frac{\text{eerste term} + \text{laatste term}}{2}.
\]
\end{theorem}

\begin{proof}
Schrijf de som $S$ twee keer op, de tweede keer in omgekeerde volgorde, en
tel kolom per kolom op:
\[
\begin{array}{ccccccccc}
S & = & 1 & + & 2 & + & \dots & + & n\\
S & = & n & + & (n-1) & + & \dots & + & 1\\
\hline
2S & = & (n+1) & + & (n+1) & + & \dots & + & (n+1)
\end{array}
\]
Er zijn $n$ kolommen, elk met som $n + 1$, dus $2S = n(n+1)$. Voor een
algemene \omterm{def:g11:seq:arithmetic}{rekenkundige rij} werkt dezelfde koppeling: eerste $+$ laatste $=$
tweede $+$ voorlaatste $= \dots$, want één stap vooruit aan de linkerkant
($+d$) wordt gecompenseerd door één stap achteruit aan de rechterkant ($-d$).
\end{proof}

\begin{example}
$1 + 2 + \dots + 100 = \frac{100 \times 101}{2} = 5050$. De som van de
oneven getallen $1 + 3 + \dots + 99$ ($50$ termen) is
$50 \times \frac{1 + 99}{2} = 2500$.
\end{example}

\section{Meetkundige rijen}

\begin{definition}[Meetkundige rij]\label{def:g11:seq:geometric}
Een \omterm{def:g11:seq:sequence}{rij} heet \emph{meetkundig}\index{meetkundige rij} met
\emph{reden}\index{reden} $q \neq 0$ wanneer elke term uit de vorige ontstaat
door met $q$ te vermenigvuldigen:
\[
u_{n+1} = q\,u_n \quad \text{voor alle } n.
\]
Gelijkwaardig, wanneer geen enkele term nul is: de verhouding
$\frac{u_{n+1}}{u_n}$ is constant en gelijk aan $q$.
\end{definition}

\begin{theorem}[Algemene term]\label{thm:g11:seq:geomgeneral}
Is $(u_n)$ \omterm{def:g11:seq:geometric}{meetkundig} met eerste term $u_0$ en \omterm{def:g11:seq:geometric}{reden} $q$, dan geldt
\[
u_n = u_0\, q^n \quad \text{voor alle } n \geq 0,
\qquad\text{en algemener } u_n = u_p\, q^{\,n-p} .
\]
\end{theorem}

\begin{proof}
Hetzelfde stappen tellen als in \cref{thm:g11:seq:arithgeneral}: van $u_0$
tot $u_n$ wordt het voorschrift ``vermenigvuldig met $q$'' $n$ keer
toegepast, wat een factor $q^n$ bijdraagt.
\end{proof}

\begin{theorem}[Meetkundige som]\label{thm:g11:seq:geomsum}
Voor elke reële $q \neq 1$ en elk \omterm{def:g10:numbers:sets}{geheel getal} $n \geq 0$ geldt
\[
1 + q + q^2 + \dots + q^n = \frac{1 - q^{\,n+1}}{1 - q}.
\]
\end{theorem}

\begin{proof}
Zij $S = 1 + q + \dots + q^n$. Vermenigvuldig met $q$:
$qS = q + q^2 + \dots + q^{n+1}$. Aftrekken geeft
\[
S - qS = \bigl(1 + q + \dots + q^n\bigr)
- \bigl(q + q^2 + \dots + q^{n+1}\bigr) = 1 - q^{\,n+1},
\]
want elke tussenliggende term komt in beide sommen één keer voor en valt
weg. Bijgevolg is $(1 - q)S = 1 - q^{\,n+1}$, en delen door $1 - q \neq 0$
geeft de formule.
\end{proof}

\begin{example}\label{ex:g11:seq:chessboard}
$1 + 2 + 4 + \dots + 2^{10} = \frac{1 - 2^{11}}{1 - 2} = 2^{11} - 1 =
2047$: rijstkorrels verdubbelen op de velden van een schaakbord overspoelt
elke graanschuur ruim vóór het $64$ste veld, waar het totaal
$2^{64} - 1 \approx 1.8 \times 10^{19}$ bedraagt.
\end{example}

\begin{method}[Het type van een rij herkennen]\label{met:g11:seq:recognize}
Bereken $u_{n+1} - u_n$ en vereenvoudig. Is het resultaat een constante $d$,
dan is de \omterm{def:g11:seq:sequence}{rij} \omterm{def:g11:seq:arithmetic}{rekenkundig}. Bereken anders $\frac{u_{n+1}}{u_n}$ (met termen
verschillend van nul) en vereenvoudig: een constante $q$ betekent
\omterm{def:g11:seq:geometric}{meetkundig}. Is geen van beide constant, dan behoort de \omterm{def:g11:seq:sequence}{rij} tot geen van beide
types --- besluit nooit uit alleen de eerste paar termen.
\end{method}

\begin{example}
Voor $u_n = 3 \times 5^n$ is
$\frac{u_{n+1}}{u_n} = \frac{3 \times 5^{n+1}}{3 \times 5^n} = 5$ voor alle
$n$: \omterm{def:g11:seq:geometric}{meetkundig} met \omterm{def:g11:seq:geometric}{reden} $5$. Voor $u_n = n^2$ is $u_1 - u_0 = 1$ maar
$u_2 - u_1 = 3$, en $\frac{u_1}{u_0}$ is niet eens gedefinieerd --- noch
\omterm{def:g11:seq:arithmetic}{rekenkundig}, noch \omterm{def:g11:seq:geometric}{meetkundig}.
\end{example}

\section{Monotonie}

\begin{definition}[Monotone rij]\label{def:g11:seq:monotone}
Een \omterm{def:g11:seq:sequence}{rij} $(u_n)$ heet \emph{\omterm{def:g11:func:monotone}{stijgend}} wanneer $u_{n+1} \geq u_n$ voor alle
$n$, en \emph{\omterm{def:g10:functions:variations}{dalend}} wanneer $u_{n+1} \leq u_n$ voor alle $n$.
\end{definition}

\begin{method}[Monotonie bestuderen]\label{met:g11:seq:monotonicity}
Bestudeer het teken van $u_{n+1} - u_n$. Voor \omterm{def:g11:seq:sequence}{rijen} met positieve termen mag
je in de plaats $\frac{u_{n+1}}{u_n}$ met $1$ vergelijken.
\end{method}

\begin{example}
Een \omterm{def:g11:seq:arithmetic}{rekenkundige rij} is \omterm{def:g11:func:monotone}{stijgend} wanneer $d \geq 0$ ($u_{n+1} - u_n = d$) en
\omterm{def:g10:functions:variations}{dalend} wanneer $d \leq 0$. Een \omterm{def:g11:seq:geometric}{meetkundige rij} met $u_0 > 0$ en $q > 1$ is
\omterm{def:g11:func:monotone}{stijgend}: $u_{n+1} - u_n = u_0 q^n (q - 1) > 0$; met $u_0 > 0$ en
$0 < q < 1$ is ze \omterm{def:g10:functions:variations}{dalend}.
\end{example}

\section{Gedrag op lange termijn, informeel}

Wat gebeurt er met $u_n$ wanneer $n$ heel groot wordt? Voor een \omterm{def:g11:seq:arithmetic}{rekenkundige
rij} met $d > 0$ overtreffen de termen $u_0 + nd$ uiteindelijk elk vast getal.
Voor een \omterm{def:g11:seq:geometric}{meetkundige rij} met $0 < q < 1$ krimpen de termen $u_0 q^n$ naar
$0$: herhaaldelijk met bijvoorbeeld $0.9$ vermenigvuldigen knaagt elke
beginwaarde weg. En voor $q > 1$ ontploffen de termen, zoals in
\cref{ex:g11:seq:chessboard}.

\begin{remark}
Die uitspraken laten zich volkomen precies maken --- ``de termen blijven
uiteindelijk binnen elke gegeven afstand van $0$'' --- en bewijzen. Dat is de
theorie van de \emph{limieten}, het openingsthema van \cref{ch:g12:seq}.
\end{remark}

\begin{exercise}[$\star$]\label{exo:g11:seq:1}
Bereken voor elke \omterm{def:g11:seq:sequence}{rij} $u_1$, $u_2$, $u_3$:
\[
u_n = \frac{n}{n+1}; \qquad
u_0 = 5,\ u_{n+1} = 3u_n - 2; \qquad
u_n = (-1)^n\,n .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:2}
$(u_n)$ is \omterm{def:g11:seq:arithmetic}{rekenkundig} met $u_0 = 7$ en $d = -3$. Bereken $u_{10}$ en
$u_{25}$. $(v_n)$ is \omterm{def:g11:seq:arithmetic}{rekenkundig} met $v_3 = 11$ en $v_8 = 26$. Zoek het
\omterm{def:g11:seq:arithmetic}{verschil} en $v_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:3}
$(u_n)$ is \omterm{def:g11:seq:geometric}{meetkundig} met $u_0 = 5$ en $q = 2$. Bereken $u_8$. $(v_n)$ is
\omterm{def:g11:seq:geometric}{meetkundig} met positieve termen, $v_2 = 12$ en $v_4 = 48$. Zoek de \omterm{def:g11:seq:geometric}{reden} en
$v_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:4}
Bereken
\[
1 + 2 + 3 + \dots + 500, \qquad
4 + 7 + 10 + \dots + 61, \qquad
1 + \frac12 + \frac14 + \dots + \frac{1}{2^{10}} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:5}
Bepaal of elke \omterm{def:g11:seq:sequence}{rij} \omterm{def:g11:seq:arithmetic}{rekenkundig} is, \omterm{def:g11:seq:geometric}{meetkundig}, of geen van beide:
\[
u_n = 4n - 1; \qquad
v_n = \frac{2^n}{3^{n+1}}; \qquad
w_n = n^2 + n .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:6}
Een theater telt $20$ rijen stoelen: $16$ stoelen in de eerste rij, en elke
rij heeft $2$ stoelen meer dan de vorige. Hoeveel stoelen staan er in de
laatste rij? En in het hele theater?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:7}
Een bacteriepopulatie verdubbelt elk uur; om 12 uur 's middags zijn er $500$
bacteriën. Hoeveel zijn er om 20 uur? Na hoeveel volle uren overtreft de
populatie voor het eerst één miljoen? (Los op door opeenvolgende machten van
$2$ te proberen.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:8}
Elke maand stort een spaarder $100$ euro op een rekening die $0.2\%$ rente
per maand geeft op het bestaande saldo (de rente wordt vlak vóór de storting
bijgeschreven). Zij $c_n$ het saldo vlak na de $n$-de storting, dus
$c_1 = 100$ en $c_{n+1} = 1.002\,c_n + 100$. Bereken $c_2$ en $c_3$, en leg
uit waarom $(c_n)$ noch \omterm{def:g11:seq:arithmetic}{rekenkundig} noch \omterm{def:g11:seq:geometric}{meetkundig} is.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:9}
Bestudeer de monotonie van de \omterm{def:g11:seq:sequence}{rijen}
\[
u_n = n^2 - 8n \ (n \geq 0), \qquad
v_n = \frac{3^n}{n!}\ (n \geq 1),
\]
waarbij $n! = 1 \times 2 \times \dots \times n$. (Vergelijk voor $(v_n)$
$\frac{v_{n+1}}{v_n}$ met $1$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:10}
De som van de eerste $n$ termen van een \omterm{def:g11:seq:arithmetic}{rekenkundige rij} met $u_0 = 3$ en
$d = 4$ is gelijk aan $903$. Zoek $n$. (Stel een kwadratische \omterm{def:g10:algebra:equation}{vergelijking} in
$n$ op en gebruik \cref{ch:g11:quad}.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:seq:11}
Zij $u_0 = 3$ en $u_{n+1} = 2u_n - 1$.
\begin{enumerate}
  \item Bereken $u_1, u_2, u_3$ en vermoed een formule voor $u_n$.
  \item Zij $v_n = u_n - 1$. Toon aan dat $(v_n)$ \omterm{def:g11:seq:geometric}{meetkundig} is, en geef haar
        \omterm{def:g11:seq:geometric}{reden} en haar eerste term.
  \item Leid een expliciete formule voor $u_n$ af en ga je vermoeden na.
\end{enumerate}
\end{exercise}

\section{Opgave: de toren van Brahma en de konijnen van Fibonacci}

\begin{problem}[{Weekendopgave --- twee legendarische recursies: de toren die
de wereld beëindigt, de rij die als goud groeit, en de hulptruc die leningen
temt}]
\label{pb:g11:seq:1}
Twee \omterm{def:g11:seq:sequence}{rijen} beheersen de folklore van de wiskunde. De ene telt de zetten van
de toren van Brahma --- vierenzestig gouden schijven waarvan de verhuizing,
zo zegt de legende, het einde van de wereld inluidt. De andere telt de
konijnen van Fibonacci en verbergt de gulden snede. Geen van beide is
\omterm{def:g11:seq:arithmetic}{rekenkundig}, geen van beide is \omterm{def:g11:seq:geometric}{meetkundig} --- en allebei geven ze zich over
aan de wapens van dit hoofdstuk: recursies, meetkundige sommen
(\cref{thm:g11:seq:geomsum}), en de hulprijtruc van \cref{exo:g11:seq:11},
die ook je hypotheek berekent.

\medskip
\noindent\textbf{Deel I --- De toren van Brahma.}
De puzzel: $n$ schijven van afnemende grootte liggen op stapel op pen A;
verplaats de hele stapel naar pen C, één schijf per keer, zonder ooit een
grotere schijf op een kleinere te leggen (pen B mag helpen). Zij $h_n$ het
kleinste aantal zetten.
\begin{enumerate}
  \item Speel het na (met munten) en noteer $h_1$, $h_2$, $h_3$.
  \item Leg de strategie achter de recursie $h_{n+1} = 2h_n + 1$ uit: wat moet
        er gebeuren vóór en na de verplaatsing van de grootste schijf?
  \item Los de recursie op met de truc van \cref{exo:g11:seq:11}: stel
        $v_n = h_n + 1$, toon aan dat $(v_n)$ \omterm{def:g11:seq:geometric}{meetkundig} is, en besluit dat
        $h_n = 2^n - 1$.
  \item De toren uit de legende heeft $64$ schijven, en de monniken
        verplaatsen één schijf per seconde. Schat met $2^{10} = 1024 \approx
        10^3$ de verhuistijd in jaren (een jaar telt ongeveer
        $3 \times 10^7$ seconden; vergelijk \cref{ex:g11:seq:chessboard},
        dezelfde reus in een ander verhaal). Moeten we ons zorgen maken?
  \item Waarom kan geen enkele strategie het onder $2^n - 1$ zetten doen?
        Beargumenteer dat \emph{elke} oplossing voldoet aan
        $h_{n+1} \geq 2 h_n + 1$: wat moet er waar zijn voor de bovenste $n$
        schijven vlak vóór en vlak na de zet van de onderste schijf?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Fibonacci.}
Definieer $F_1 = F_2 = 1$ en $F_{n+2} = F_{n+1} + F_n$ (elke term de som van
de twee ervoor --- het voorschrift voor het tellen van ritmes uit het
onderbouwvolume, nu met zijn Europese naam).
\begin{enumerate}[resume]
  \item Noem $F_1$ tot en met $F_{12}$.
  \item Toon aan dat $(F_n)$ noch \omterm{def:g11:seq:arithmetic}{rekenkundig} noch \omterm{def:g11:seq:geometric}{meetkundig} is, maar wel
        strikt \omterm{def:g11:func:monotone}{stijgend} vanaf $n = 2$ (\cref{met:g11:seq:monotonicity} en de
        recursie).
  \item Bewijs de somidentiteit
        \[
        F_1 + F_2 + \dots + F_n = F_{n+2} - 1
        \]
        door te telescoperen: schrijf elke $F_k$ als $F_{k+2} - F_{k+1}$ en
        kijk hoe de som ineenschuift. Ga ze na voor $n = 6$.
  \item Bewijs de kwadratenidentiteit
        $F_1^2 + F_2^2 + \dots + F_n^2 = F_n F_{n+1}$, telescoperend met
        $F_k F_{k+1} - F_{k-1} F_k = F_k^2$. Ga ze na voor $n = 4$.
        (\omterm{def:g10:functions:function}{Beeld}: vierkanten met zijden $1, 1, 2, 3, 5, \dots$ betegelen een
        rechthoek --- het skelet van de beroemde spiraal van Fibonacci.)
  \item De identiteit van Cassini luidt
        $F_{n+1} F_{n-1} - F_n^2 = (-1)^n$. Ga ze na voor $n = 4, 5, 6$ ---
        en herken de motor van de truc met het verdwijnende vierkant uit de
        oppervlakteopgave in het onderbouwvolume.
  \item Toon uit de recursie aan dat $F_{n+2} \geq 2 F_n$: Fibonacci
        verdubbelt minstens om de twee stappen --- ze groeit minstens zo snel
        als een \omterm{def:g11:seq:geometric}{meetkundige rij} met \omterm{def:g11:seq:geometric}{reden} $\sqrt2$.
  \item Bereken de verhoudingen $r_n = \frac{F_{n+1}}{F_n}$ voor $n = 3$ tot
        $10$ (drie decimalen). Neem aan dat ze zich op een limiet $L$
        vastzetten, laat de betrekking $r_{n+1} = 1 + \frac{1}{r_n}$ naar de
        limiet overgaan en los op: welk getal uit \cref{pb:g10:algebra:1}
        aanbidden de konijnen?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De hulptruc, aan het bankloket.}
\begin{enumerate}[resume]
  \item Veralgemeen \cref{exo:g11:seq:11}: stel voor $u_{n+1} = a\,u_n + b$
        met $a \neq 1$ het vaste punt $\ell = \frac{b}{1 - a}$. Toon aan dat
        $v_n = u_n - \ell$ \omterm{def:g11:seq:geometric}{meetkundig} is met \omterm{def:g11:seq:geometric}{reden} $a$, en besluit dat
        $u_n = a^n (u_0 - \ell) + \ell$.
  \item Een lening: $10\,000$ euro tegen $1\,\%$ rente per maand, met
        $300$ euro aflossing per maand, zodat de schuld voldoet aan
        $d_{n+1} = 1.01\,d_n - 300$. Pas vraag 13 toe (eerst het vaste punt!)
        om een expliciete formule voor $d_n$ te krijgen.
  \item Zoek met een rekenmachine de eerste maand waarin de schuld afbetaald
        is, en het totaal terugbetaalde bedrag. Hoeveel kostte het lenen
        zelf?
  \item Een stad met $50\,000$ inwoners groeit met $2\,\%$ per jaar en
        verwelkomt daarnaast $1\,000$ nieuwkomers: $p_{n+1} = 1.02\,p_n +
        1000$. Geef de expliciete formule en de bevolking na $10$ jaar.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De twee koninklijke families.}
\begin{enumerate}[resume]
  \item Bereken $1 + 2 + 3 + \dots + 1000$ (\cref{thm:g11:seq:intsum} --- de
        som van de kleine Gauss uit het onderbouwvolume, nu officieel), en
        $1 + 2 + 4 + \dots + 2^{19}$ (\cref{thm:g11:seq:geomsum}).
  \item Bereken de som van de rekenkundige rij $7, 12, 17, \dots, 502$
        (hoeveel termen?).
  \item Spaarplan: elke maand wordt $100$ euro gestort, met $0.5\,\%$ rente
        per maand; na de $n$-de storting is het saldo
        $100\left(1.005^{n-1} + \dots + 1.005 + 1\right)$. Bereken het saldo
        na $5$ jaar ($n = 60$).
  \item Slotstuk --- de gereedschapskist van de rijentemmer: expliciete
        tegenover recursieve beschrijvingen; de twee koninklijke families en
        hun somformules; de hulprij die affiene recursies in meetkundige
        omzet; en Fibonacci, eerste burger buiten beide families, vandaag
        getemd met identiteiten en wachtend op matrices (jaar 12) en limieten
        om volledig gevangen te worden. Telkens één zin.
\end{enumerate}
\end{problem}
