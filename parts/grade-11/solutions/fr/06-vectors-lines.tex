\section*{Chapitre \ref{ch:g11:vect} --- Vecteurs et droites dans le plan}

\begin{solution}{exo:g11:vect:1}
$\vect{AB}\,(3, 4)$, $\vect{AC}\,(-3, 2)$,
$\vect{AB} + \vect{AC}\,(0, 6)$,
$2\vect{AB} - 3\vect{AC}\,(6 + 9,\ 8 - 6) = (15, 2)$. Le \omterm{prop:g10:coordgeom:midpoint}{milieu} de
$[BC]$ est $\left(\frac{5 + (-1)}{2}, \frac{3+1}{2}\right) = (2, 2)$.
\end{solution}

\begin{solution}{exo:g11:vect:2}
$\det = 4 \times 3 - (-2) \times (-6) = 12 - 12 = 0$~: \omterm{def:g11:vect:collinear}{colinéaires} (en
effet $\vec v = -\frac12 \vec u$).

$\det = 2 \times 7 - 3 \times 5 = -1 \neq 0$~: non \omterm{def:g11:vect:collinear}{colinéaires}.

$\det = 0 \times (-8) - 0 \times 3 = 0$~: \omterm{def:g11:vect:collinear}{colinéaires} (tous deux
verticaux).
\end{solution}

\begin{solution}{exo:g11:vect:3}
Par $A(2,1)$, direction $(3, -1)$~: identifier $(-b, a) = (3, -1)$ donne
$a = -1$, $b = -3$, donc $-x - 3y + c = 0$~; en injectant $A$~:
$-2 - 3 + c = 0$, $c = 5$. En multipliant par $-1$~: $x + 3y - 5 = 0$.

Par $B(0,4)$ et $C(2,0)$~: $\vect{BC}\,(2, -4)$, donc $a = -4$,
$b = -2$~: $-4x - 2y + c = 0$~; en injectant $B$~: $-8 + c = 0$,
$c = 8$. En simplifiant par $-2$~: $2x + y - 4 = 0$. Vérification avec
$C$~: $4 + 0 - 4 = 0$.
\end{solution}

\begin{solution}{exo:g11:vect:4}
Un \omterm{def:g11:vect:direction}{vecteur directeur} est $(-b, a) = (2, 3)$. \omterm{def:g10:numbers:interunion}{Intersections}~: $x = 0$
donne $y = 3$, point $(0, 3)$~; $y = 0$ donne $x = -2$, point $(-2, 0)$.
Pour $P(2,6)$~: $3 \times 2 - 2 \times 6 + 6 = 0$, donc $P \in d$. Pour
$Q(1,4)$~: $3 - 8 + 6 = 1 \neq 0$, donc $Q \notin d$.
\end{solution}

\begin{solution}{exo:g11:vect:5}
$\vect{AB}\,(3, 6)$ et $\vect{AC}\,(-3, -6)$~:
$\det = 3 \times (-6) - (-3) \times 6 = 0$, donc $A$, $B$, $C$ sont
alignés.

$\vect{DE}\,(2, 3)$ et $\vect{DF}\,(5, 8)$~:
$\det = 2 \times 8 - 5 \times 3 = 1 \neq 0$~: $D$, $E$, $F$ ne sont pas
alignés.
\end{solution}

\begin{solution}{exo:g11:vect:6}
Premier couple~: en additionnant les deux \omterm{def:g10:algebra:equation}{équations},
$(2x + y - 7) + (x - y + 1) = 3x - 6 = 0$, donc $x = 2$, puis
$y = x + 1 = 3$~: elles se coupent en $(2, 3)$ (vérification~:
$4 + 3 - 7 = 0$).

Second couple~: $ab' - a'b = 1 \times 4 - (-2) \times (-2) = 0$, donc les
droites sont parallèles~; multiplier $e$ par $-2$ donne
$-2x + 4y - 6 = 0$, qui diffère de $e'$ ($-6 \neq 1$)~: strictement
parallèles, pas d'\omterm{def:g10:numbers:interunion}{intersection}.
\end{solution}

\begin{solution}{exo:g11:vect:7}
Pour $d: 2x - y + 3 = 0$~: direction $(-b, a) = (1, 2)$ et le point
$(0, 3)$ est sur $d$, donc $d = \{(t,\ 3 + 2t) : t \in \R\}$.

Pour la droite paramétrique par $(1, -3)$ de direction $(2, 1)$~:
$a = 1$, $b = -2$, donc $x - 2y + c = 0$~; en injectant $(1, -3)$~:
$1 + 6 + c = 0$, $c = -7$. \omterm{def:g10:algebra:equation}{Équation}~: $x - 2y - 7 = 0$ (vérification
$t = 1$, point $(3, -2)$~: $3 + 4 - 7 = 0$).
\end{solution}

\begin{solution}{exo:g11:vect:8}
Une droite parallèle a le même $(a, b)$~: $4x - 3y + c = 0$~; par
$P(1, -2)$~: $4 + 6 + c = 0$, donc $c = -10$ et la droite est
$4x - 3y - 10 = 0$.

Le parallélisme de $mx + 2y - 5 = 0$ avec $d$ exige
$ab' - a'b = 4 \times 2 - m \times (-3) = 8 + 3m = 0$, donc
$m = -\frac83$.
\end{solution}

\begin{solution}{exo:g11:vect:9}
Le \omterm{prop:g10:coordgeom:midpoint}{milieu} de $[AB]$ est $M\!\left(\frac{-1+3}{2}, \frac{0+2}{2}\right)
= (1, 1)$. La \omterm{def:g10:stats:median}{médiane} est la droite $(CM)$ avec $C(1, -4)$~: les deux
points ont pour \omterm{def:g10:coordgeom:system}{abscisse} $1$, donc la \omterm{def:g10:stats:median}{médiane} est la droite verticale
\[
x - 1 = 0 .
\]
\end{solution}

\begin{solution}{exo:g11:vect:10}
$\det(\vec u, \vec v) = m(m - 1) - 3 \times 2 = m^2 - m - 6$. Il
s'annule lorsque $m^2 - m - 6 = 0$~: $\Delta = 1 + 24 = 25$, \omterm{def:g11:quad:discriminant}{racines}
$m = \frac{1 \pm 5}{2}$, c'est-à-dire $m = 3$ ou $m = -2$.
\end{solution}

\begin{solution}{exo:g11:vect:11}
\emph{1.} $I = \left(\frac12, 0\right)$. Alors
$\vect{DI} = \left(\frac12 - 0,\ 0 - 1\right) = \left(\frac12, -1\right)$
et $\vect{DJ} = \frac23\vect{DI} = \left(\frac13, -\frac23\right)$,
donc
\[
J = D + \vect{DJ} = \left(\frac13,\ \frac13\right).
\]

\emph{2.} $\vect{AJ}\left(\frac13, \frac13\right)$ et
$\vect{AC}\,(1, 1)$~:
$\det = \frac13 \times 1 - 1 \times \frac13 = 0$, donc $A$, $J$, $C$
sont alignés --- en fait $\vect{AJ} = \frac13\vect{AC}$~: la droite
$(DI)$ coupe la diagonale $(AC)$ à exactement un tiers de sa longueur
depuis $A$.
\end{solution}

\begin{solution}{pb:g11:vect:1}
\textbf{1.} $x = 1 + 3t$, $y = 2 - t$. De la seconde on tire
$t = 2 - y$~; en substituant~: $x = 1 + 3(2 - y) = 7 - 3y$, d'où
l'\omterm{def:g10:algebra:equation}{équation} cartésienne $x + 3y - 7 = 0$.

\textbf{2.} Pour $ax + by + c = 0$, un \omterm{def:g11:vect:direction}{vecteur directeur} est
$(-b, a)$~: ici $(5, 2)$. Un point~: $y = 1$ donne $x = 1$, soit
$(1, 1)$. Représentation paramétrique~: $x = 1 + 5t$, $y = 1 + 2t$.

\textbf{3.} En reportant $x = 7 - 3y$ dans $2x - 5y + 3 = 0$~:
$14 - 6y - 5y + 3 = 0$, donc $y = \frac{17}{11}$ et
$x = 7 - \frac{51}{11} = \frac{26}{11}$~: le point
$\left(\frac{26}{11}, \frac{17}{11}\right)$.

\textbf{4.} Directions $(5, 2)$ et $(3, -1)$~:
$\det = 5 \times (-1) - 2 \times 3 = -11 \neq 0$~: les droites
se coupent, comme on l'a vu. Pour le second couple~: les directions
$(5, 2)$ et $(10, 4)$ ont $\det = 0$ (parallèles), et doubler
$2x - 5y + 3 = 0$ donne $-4x + 10y - 6 = 0 \neq -4x + 10y + 1
= 0$~: strictement parallèles.

\textbf{5.} $\vect{AB}(3, 2)$, $\vect{AC}(9, 6)$~:
$\det = 18 - 18 = 0$~: les points sont alignés.

\textbf{6.} $P$~: de $x = 2t$ et $y = 3 + t$ on tire
$y = 3 + \frac x2$. $Q$~: de $x = 10 - t$ on tire $t = 10 - x$,
donc $y = 2(10 - x) - 1 = 19 - 2x$. Croisement~:
$3 + \frac x2 = 19 - 2x$ donne $x = 6.4$ et $y = 6.2$~: les
trajectoires se croisent en $(6.4,\ 6.2)$.

\textbf{7.} $P$ y passe lorsque $2t = 6.4$, soit $t = 3.2$ h. $Q$
lorsque $10 - t = 6.4$, soit $t = 3.6$ h. Vingt-quatre minutes
d'écart~: pas de collision. Avertissement~: une carte marine montre
des \emph{trajectoires}~; seules les horloges du paramétrage disent
si deux mouvements occupent le même point au même instant.

\textbf{8.} $D^2(t) = (3t - 10)^2 + (4 - t)^2 =
10t^2 - 68t + 116$~: sommet en $t = \frac{68}{20} = 3.4$ h, où
$D^2 = 0.4$~: distance minimale d'approche
$D = \sqrt{0.4} \approx 0.63$ mille à $t = 3.4$.

\textbf{9.} Le règlement est violé~: $0.63 < 1$. $D^2(t) < 1$
s'écrit $10t^2 - 68t + 115 < 0$~: $\Delta = 4624 - 4600 = 24$,
\omterm{def:g11:quad:discriminant}{racines} $\frac{68 \pm \sqrt{24}}{20} = 3.4 \pm 0.245$~: les navires
sont trop proches de $t \approx 3.16$ à $t \approx 3.64$ ---
environ $29$ minutes d'infraction. L'un des deux doit changer de
cap.

\textbf{10.} Chaque coordonnée de chaque navire est une \omterm{def:g11:func:function}{fonction}
\emph{affine} de $t$, donc chaque différence de \omterm{def:g10:coordgeom:system}{coordonnées} est
affine, et $D^2$ --- somme de deux carrés de \omterm{def:g10:reffunc:affine}{fonctions affines} ---
est un polynôme du second degré (à coefficient dominant positif ou
nul). La distance minimale d'approche se lit donc toujours sur un
sommet.

\textbf{11.} Rejoindre $P$ à $t = 3$, c'est-à-dire en $(6, 6)$~:
$(3a, 3b) = (6, 6)$ donne $(a, b) = (2, 2)$, et une vitesse
$\sqrt{8} \approx 2.8$ nœuds.

\textbf{12.} $D(0, 1)$, $I\left(\frac12, 0\right)$~: direction
$\left(\frac12, -1\right)$, c'est-à-dire $(1, -2)$~: \omterm{def:g10:algebra:equation}{équation}
$2x + y = 1$.

\textbf{13.} $(AC)$~: $y = x$. Alors $2x + x = 1$, donc
$x = \frac13$~: le point $\left(\frac13, \frac13\right)$, au tiers
du chemin de $A$ vers $C$.

\textbf{14.} $B(1, 0)$, $K\left(\frac12, 1\right)$~: direction
$\left(-\frac12, 1\right)$, \omterm{def:g10:algebra:equation}{équation} $2x + y = 2$. Avec $y = x$~:
$x = \frac23$, soit le point $\left(\frac23, \frac23\right)$. Les
deux céviennes coupent $(AC)$ en ses deux points de trisection~:
théorème démontré, deux fois trois lignes de calcul.

\textbf{15.} Tout parallélogramme est envoyé sur le carré unité en
choisissant $A$ comme origine et $\vect{AB}$, $\vect{AD}$ comme
unités des deux axes. Ce choix conserve tout ce dont parle le
théorème --- milieux, alignements et rapports le long d'une droite
donnée (tout cela s'exprime avec des \omterm{def:g11:vect:vector}{vecteurs} et des scalaires) ---
si bien que démontrer l'énoncé dans le carré le démontre pour tout
parallélogramme. (Il ne transporterait en revanche \emph{ni} les
longueurs \emph{ni} les angles~: ceux-là, le \omterm{def:g10:coordgeom:system}{repère} les déforme.)

\textbf{16.} $E\left(1, \frac12\right)$~: la droite $(AE)$ a pour
\omterm{def:g10:algebra:equation}{équation} $y = \frac x2$. La diagonale $(DB)$, de $(0,1)$ à
$(1,0)$~: $y = 1 - x$. \omterm{def:g10:numbers:interunion}{Intersection}~: $\frac x2 = 1 - x$, donc
$x = \frac23$~: le point $\left(\frac23, \frac13\right)$ --- encore
un point de trisection, cette fois de l'autre diagonale. Les
céviennes des milieux aiment les tiers.

\textbf{17.} Les deux premières~: $x + y = 4$ et $2x - y = 2$
donnent $3x = 6$, soit $(2, 2)$. La troisième~: $2 - 4 = -2$~:
vérifiée. Elles sont concourantes en $(2, 2)$.

\textbf{18.} $mx - y + 2 - 3m = m(x - 3) + (2 - y) = 0$~: pour que
l'égalité vaille pour \emph{tout} $m$, il faut $x = 3$ et $y = 2$
--- et en effet toute $d_m$ contient $(3, 2)$. C'est un faisceau de
droites passant par un même point, une droite par \omterm{def:g10:reffunc:affine}{coefficient
directeur} $m$.

\textbf{19.} Parallèle à $y = 2x + 7$~: \omterm{def:g10:reffunc:affine}{coefficient directeur}
$m = 2$, soit la droite $2x - y - 4 = 0$. Passant par l'origine~:
$2 - 3m = 0$, soit $m = \frac23$.

\textbf{20.} Cartésienne~: on substitue un point et l'appartenance
est tranchée. Réduite~: \omterm{def:g10:reffunc:affine}{coefficient directeur} et \omterm{def:g10:reffunc:affine}{ordonnée à
l'origine} visibles, courbe immédiate. Paramétrique~: l'horloge est
incluse --- collisions, distance minimale d'approche, interception.
Le \omterm{def:g11:vect:det}{déterminant}~: un produit en croix, et le parallélisme est décidé.
Le \omterm{def:g10:coordgeom:system}{repère} bien choisi~: un parallélogramme devient un carré unité,
et un théorème classique de trisection devient trois \omterm{def:g10:numbers:interunion}{intersections}
de droites.
\end{solution}
