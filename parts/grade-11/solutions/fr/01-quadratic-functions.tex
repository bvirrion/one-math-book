\section*{Chapitre \ref{ch:g11:quad} --- Fonctions et équations du second degré}

\begin{solution}{exo:g11:quad:1}
$x^2-5x+6$~: $\Delta = 25 - 24 = 1$, \omterm{def:g11:quad:discriminant}{racines} $\frac{5 \pm 1}{2}$,
c'est-à-dire $2$ et $3$.

$2x^2+3x-2$~: $\Delta = 9 + 16 = 25$, \omterm{def:g11:quad:discriminant}{racines} $\frac{-3 \pm 5}{4}$,
c'est-à-dire $\frac12$ et $-2$.

$x^2+2x+5$~: $\Delta = 4 - 20 = -16 < 0$, pas de solution réelle.

$9x^2-6x+1$~: $\Delta = 36 - 36 = 0$, une (double) \omterm{def:g11:quad:discriminant}{racine}
$x_0 = \frac{6}{18} = \frac13$.
\end{solution}

\begin{solution}{exo:g11:quad:2}
$f(x) = x^2 - 6x + 11 = (x-3)^2 - 9 + 11 = (x-3)^2 + 2$~: sommet
$(3, 2)$.

$g(x) = -2x^2+4x+1 = -2(x^2 - 2x) + 1 = -2\bigl((x-1)^2 - 1\bigr) + 1
= -2(x-1)^2 + 3$~: sommet $(1, 3)$.

$h(x) = 3x^2 + 6x = 3(x^2 + 2x) = 3\bigl((x+1)^2 - 1\bigr)
= 3(x+1)^2 - 3$~: sommet $(-1, -3)$.
\end{solution}

\begin{solution}{exo:g11:quad:3}
$x^2 - 3x - 10$~: $\Delta = 9 + 40 = 49$, \omterm{def:g11:quad:discriminant}{racines}
$\frac{3 \pm 7}{2} = -2$ et $5$~; coefficient dominant positif, donc le
trinôme est négatif strictement entre les \omterm{def:g11:quad:discriminant}{racines}~: l'ensemble des
solutions est $\intoo{-2}{5}$.

$2x^2 + x + 3$~: $\Delta = 1 - 24 = -23 < 0$ et $a = 2 > 0$~: toujours
positif, l'ensemble des solutions est $\R$.

$-x^2 + 6x - 9 = -(x-3)^2$~: il est $\geq 0$ seulement là où
$(x-3)^2 \leq 0$, c'est-à-dire en $x = 3$. L'ensemble des solutions est
$\{3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:4}
$x^2 - 7x + 6$~: $1 - 7 + 6 = 0$, donc $1$ est \omterm{def:g11:quad:discriminant}{racine}~; le produit des
\omterm{def:g11:quad:discriminant}{racines} est $6$, donc l'autre \omterm{def:g11:quad:discriminant}{racine} est $6$.

$2x^2 + 5x - 7$~: $2 + 5 - 7 = 0$, donc $1$ est \omterm{def:g11:quad:discriminant}{racine}~; le produit est
$\frac{c}{a} = -\frac72$, donc l'autre \omterm{def:g11:quad:discriminant}{racine} est $-\frac72$.

$x^2 + 4x - 5$~: $1 + 4 - 5 = 0$, donc $1$ est \omterm{def:g11:quad:discriminant}{racine}~; le produit est
$-5$, donc l'autre \omterm{def:g11:quad:discriminant}{racine} est $-5$.
\end{solution}

\begin{solution}{exo:g11:quad:5}
Les deux nombres résolvent $x^2 - 14x + 45 = 0$
(\cref{prop:g11:quad:vieta})~; $\Delta = 196 - 180 = 16$, \omterm{def:g11:quad:discriminant}{racines}
$\frac{14 \pm 4}{2} = 9$ et $5$.

Pour le rectangle~: la moitié du périmètre donne longueur $+$ largeur
$= 17$, et l'aire donne leur produit $60$. Ils résolvent
$x^2 - 17x + 60 = 0$~; $\Delta = 289 - 240 = 49$, \omterm{def:g11:quad:discriminant}{racines}
$\frac{17 \pm 7}{2} = 12$ et $5$. Le rectangle est $12 \times 5$.
\end{solution}

\begin{solution}{exo:g11:quad:6}
$\Delta = (m-2)^2 - 4$. Exactement une solution lorsque $\Delta = 0$~:
$(m-2)^2 = 4$, donc $m - 2 = \pm 2$, c'est-à-dire $m = 0$ ou $m = 4$.
Aucune solution lorsque $\Delta < 0$~: $(m-2)^2 < 4$, c'est-à-dire
$-2 < m - 2 < 2$, soit $0 < m < 4$.
\end{solution}

\begin{solution}{exo:g11:quad:7}
Le sommet est en $t = -\frac{20}{2 \times (-5)} = 2$, et
$h(2) = -20 + 40 + 1 = 21$~: hauteur maximale $21$~m, atteinte après
$2$ secondes.

$h(t) \geq 16$ signifie $-5t^2 + 20t - 15 \geq 0$, c'est-à-dire (en
divisant par $-5$, ce qui inverse l'inégalité) $t^2 - 4t + 3 \leq 0$.
Les \omterm{def:g11:quad:discriminant}{racines} de $t^2-4t+3$ sont $1$ et $3$ ($1 - 4 + 3 = 0$, produit
$3$), donc $t^2 - 4t + 3 \leq 0$ exactement sur $\intcc{1}{3}$~: la
balle est à au moins $16$~m de hauteur entre les instants $t = 1$~s et
$t = 3$~s.
\end{solution}

\begin{solution}{exo:g11:quad:8}
Avec $X = x^2$~: $X^2 - 13X + 36 = 0$, $\Delta = 169 - 144 = 25$, donc
$X = \frac{13 \pm 5}{2} = 9$ ou $4$. Retour à $x$~: $x^2 = 9$ donne
$x = \pm 3$, et $x^2 = 4$ donne $x = \pm 2$. L'ensemble des solutions est
$\{-3, -2, 2, 3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:9}
La condition est $x^2 < x + 2$, c'est-à-dire $x^2 - x - 2 < 0$. Comme
$x^2 - x - 2 = (x+1)(x-2)$ (\omterm{def:g11:quad:discriminant}{racines} $-1$ et $2$), l'inégalité est
réalisée strictement entre les \omterm{def:g11:quad:discriminant}{racines}~: $x \in \intoo{-1}{2}$. Sur un
croquis, la \omterm{def:g11:quad:parabola}{parabole} $y = x^2$ passe sous la droite $y = x+2$ exactement
entre leurs deux points d'\omterm{def:g10:numbers:interunion}{intersection} $(-1, 1)$ et $(2, 4)$.
\end{solution}

\begin{solution}{exo:g11:quad:10}
Soit $x > 0$ tel que $x + \frac1x = \frac{13}{6}$. Multiplier par
$6x > 0$~: $6x^2 + 6 = 13x$, c'est-à-dire $6x^2 - 13x + 6 = 0$. Alors
$\Delta = 169 - 144 = 25$ et $x = \frac{13 \pm 5}{12}$, d'où
$x = \frac32$ ou $x = \frac23$. Les deux sont positifs et inverses l'un
de l'autre, comme attendu~: le nombre est $\frac32$ (ou $\frac23$).
\end{solution}

\begin{solution}{exo:g11:quad:11}
\emph{1.} Les \omterm{def:g10:numbers:interunion}{intersections} correspondent aux solutions de
$x^2 = mx - 1$, c'est-à-dire $x^2 - mx + 1 = 0$, dont le \omterm{def:g11:quad:discriminant}{discriminant} est
$\Delta = m^2 - 4$. Deux points lorsque $\abs{m} > 2$~; exactement un
lorsque $m = \pm 2$~; aucun lorsque $-2 < m < 2$.

\emph{2.} Pour $m = 2$~: $x^2 - 2x + 1 = (x-1)^2 = 0$, donc l'unique
point de contact est $(1, 1)$. Pour $m = -2$~: $(x+1)^2 = 0$, point de
contact $(-1, 1)$. Dans les deux cas la droite rencontre la \omterm{def:g11:quad:parabola}{parabole} en
un seul point sans la traverser --- c'est la tangente en ce point, et en
effet sa pente $m = \pm 2$ égale la valeur en $x = \pm 1$ de la dérivée
$2x$ de $x^2$ (\cref{ch:g11:deriv}).
\end{solution}

\begin{solution}{pb:g11:quad:1}
\textbf{1.} $x^2 - 5x + 6$~: $\Delta = 1$, \omterm{def:g11:quad:discriminant}{racines} $2$ et $3$.
$2x^2 + x - 6$~: $\Delta = 1 + 48 = 49$, \omterm{def:g11:quad:discriminant}{racines}
$\frac{-1 \pm 7}{4}$, soit $\frac32$ et $-2$. $x^2 + x + 1$~:
$\Delta = -3 < 0$~: aucune solution réelle.

\textbf{2.} $\Delta = 16 + 48 = 64$~: \omterm{def:g11:quad:discriminant}{racines} $3$ et $-1$, donc
$2x^2 - 4x - 6 = 2(x - 3)(x + 1)$~; \omterm{thm:g11:quad:canonical}{forme canonique}
$2(x - 1)^2 - 8$. La forme factorisée montre les \omterm{def:g11:quad:discriminant}{racines}, la \omterm{thm:g11:quad:canonical}{forme
canonique} le sommet $(1, -8)$, la forme développée l'\omterm{def:g10:reffunc:affine}{ordonnée à
l'origine} $-6$.

\textbf{3.} \omterm{def:g11:quad:discriminant}{Racines} $1$ et $3$, coefficient dominant négatif~: le
trinôme est $\geq 0$ \emph{entre} les \omterm{def:g11:quad:discriminant}{racines}, soit
$x \in \intcc{1}{3}$.

\textbf{4.} Les deux nombres sont les \omterm{def:g11:quad:discriminant}{racines} de
$x^2 - 7x + 12 = 0$~: $3$ et $4$. Si $2$ est \omterm{def:g11:quad:discriminant}{racine} de
$x^2 - 5x + c$, l'autre \omterm{def:g11:quad:discriminant}{racine} $r$ vérifie $2 + r = 5$ (la somme),
donc $r = 3$, et $c = 2 \times 3 = 6$ (le produit)~: Viète le lit
sans calcul.

\textbf{5.} $\Delta = m^2 - 36$~: \omterm{def:g11:quad:discriminant}{racine} double pour $m = \pm 6$
(\omterm{def:g11:quad:discriminant}{racine} $\mp 3$)~; aucune \omterm{def:g11:quad:discriminant}{racine} pour $-6 < m < 6$~; deux \omterm{def:g11:quad:discriminant}{racines}
pour $\abs m > 6$.

\textbf{6.} $PF^2 = x^2 + \left(x^2 - \frac14\right)^2
= x^2 + x^4 - \frac{x^2}{2} + \frac{1}{16}
= x^4 + \frac{x^2}{2} + \frac{1}{16}
= \left(x^2 + \frac14\right)^2$. D'où
$PF = x^2 + \frac14$ (les deux facteurs étant positifs).

\textbf{7.} La distance de $P(x, x^2)$ à la droite horizontale
$y = -\frac14$ vaut $x^2 + \frac14$ --- exactement $PF$. Tout point
de la \omterm{def:g11:quad:parabola}{parabole} est donc équidistant du \omterm{pb:g11:quad:1}{foyer} et de la \omterm{pb:g11:quad:1}{directrice}.

\textbf{8.} $PF^2 = x^2 + \left(y - \frac14\right)^2$ et
$\operatorname{dist}^2 = \left(y + \frac14\right)^2$. L'égalité
donne $x^2 + y^2 - \frac y2 + \frac{1}{16} = y^2 + \frac y2 +
\frac1{16}$, donc $x^2 = y$~: le lieu est exactement la \omterm{def:g11:quad:parabola}{parabole}. Un
point et une droite engendrent la courbe, comme un point et une
distance engendraient le cercle.

\textbf{9.} Pour $y = \frac{x^2}{4f}$, le même développement avec
$F(0, f)$ donne $PF^2 = x^2 + \left(\frac{x^2}{4f} - f\right)^2
= \left(\frac{x^2}{4f} + f\right)^2$~: la distance à $F$ égale la
distance à $y = -f$. Antenne~: $9 = \frac{30^2}{4f}$ donne
$4f = 100$, soit $f = 25$ cm~: le récepteur se suspend à un quart de
mètre au-dessus du sommet --- hors de la cuvette de $9$ cm, comme le
montrent les antennes réelles.

\textbf{10.} Un front d'onde qui arrive parallèlement à l'axe a
parcouru des distances égales pour atteindre le niveau de la
\omterm{pb:g11:quad:1}{directrice}~; l'équidistance convertit «~même distance à la
\omterm{pb:g11:quad:1}{directrice}~» en «~même distance à $F$~»~: tous les signaux
réfléchis arrivent en $F$ \emph{en phase}, et se renforcent --- c'est
là qu'il faut écouter. La démonstration honnête réclame le
\omterm{def:g10:reffunc:affine}{coefficient directeur} de la tangente en chaque point --- la dérivée
du \cref{ch:g11:deriv}, entrevue à l'\cref{exo:g11:quad:11}.

\textbf{11.} $y = x\left(1 - \frac{x}{20}\right)$~: chute en
$x = 20$ m. Sommet à mi-parcours~: $x = 10$, de hauteur
$y(10) = 10 - 5 = 5$ m.

\textbf{12.} $y(4) = 4 - \frac{16}{20} = 3.2$ m~: le ballon passe
au-dessus du mur de $3$ m avec de la marge, et s'écrase sur celui de
$4$ m.

\textbf{13.} $y(10) = 10 - \frac{100}{40} = 7.5$ m $> 7$~: le camion
passe à $10$ m du centre. $y(15) = 10 - \frac{225}{40} = 4.375$ m~:
il ne passe pas à $15$ m.

\textbf{14.} $R(p) = -2p^2 + 120p$~: sommet en
$p = \frac{120}{4} = 30$ euros, pour une recette
$R(30) = 30 \times 60 = 1\,800$ euros.

\textbf{15.} $y(320) = 143 \times \left(\frac{320}{640}\right)^2
= \frac{143}{4} \approx 36$ m au-dessus du point le plus bas.

\textbf{16.} $x^2 = mx - 1$ donne $x^2 - mx + 1 = 0$, avec
$\Delta = m^2 - 4$~: deux points pour $\abs m > 2$, aucun pour
$\abs m < 2$, tangence pour $m = \pm 2$ avec contact en
$x = \frac m2 = \pm 1$~: tangentes $y = 2x - 1$ en $(1, 1)$ et
$y = -2x - 1$ en $(-1, 1)$.

\textbf{17.} $x^2 = mx + 1$ donne $\Delta = m^2 + 4 > 0$ pour tout
$m$~: toujours deux \omterm{def:g10:numbers:interunion}{intersections}. Depuis un point intérieur, toute
droite traverse la courbe deux fois --- les tangentes ne se mènent
que de l'extérieur, exactement comme pour les cercles.

\textbf{18.} $y = \frac14$ coupe $y = x^2$ en $x = \pm\frac12$~:
corde de longueur $1$. Pour $y = \frac{x^2}{4f}$~: $y = f$ donne
$x^2 = 4f^2$, soit $x = \pm 2f$, d'où une longueur $4f$ --- la
largeur focale vaut quatre fois la distance focale (les concepteurs
d'antennes l'appellent le \emph{latus rectum}).

\textbf{19.} Un point de $y = x^2$ s'écrit $(a, a^2)$~; son \omterm{def:g10:functions:function}{image} est
$(X, Y) = (ka, ka^2)$, et $\frac{X^2}{k} = \frac{k^2a^2}{k} =
ka^2 = Y$~: la courbe \omterm{def:g10:functions:function}{image} est exactement $y = \frac{x^2}{k}$. En
choisissant $k = 4f$ on atteint toutes les \omterm{def:g11:quad:parabola}{paraboles}~: elles sont
toutes des agrandissements d'une seule courbe maîtresse. (Les cercles
sont eux aussi tous semblables --- mais pas les ellipses~:
l'aplatissement compte, l'agrandissement non.)

\textbf{20.} Une \omterm{def:g10:algebra:equation}{équation} à trois costumes lisibles~; le lieu des
points équidistants d'un point et d'une droite~; le réflecteur qui
rassemble en son \omterm{pb:g11:quad:1}{foyer} les rayons parallèles et le câble qui porte un
tablier uniforme~; la trajectoire de tout ballon lancé~; et, à un
agrandissement près, une courbe unique --- une seule \omterm{def:g11:quad:parabola}{parabole},
partout.
\end{solution}
