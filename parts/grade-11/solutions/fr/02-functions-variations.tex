\section*{Chapitre \ref{ch:g11:func} --- Fonctions et variations}

\begin{solution}{exo:g11:func:1}
$f$~: il faut $3 - x \geq 0$, \omterm{def:g11:func:function}{ensemble de définition}
$\intoc{-\infty}{3}$.

$g$~: il faut $x^2 - 4 \neq 0$, c'est-à-dire $x \neq \pm 2$~: \omterm{def:g11:func:function}{ensemble
de définition} $\R \setminus \{-2, 2\}$.

$h$~: il faut $x \geq 0$ et $x \neq 5$~: \omterm{def:g11:func:function}{ensemble de définition}
$\intco{0}{5} \cup \intoo{5}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:2}
$\sqrt{11} < \sqrt{13}$ car $\sqrt{}$ est \omterm{def:g11:func:monotone}{croissante} et $11 < 13$.

$(-2.1)^2 > (-2.05)^2$ car $x^2$ est \omterm{def:g10:functions:variations}{décroissante} sur
$\intoc{-\infty}{0}$ et $-2.1 < -2.05$.

$\frac{1}{0.9} > \frac{1}{1.1}$ car $\frac1x$ est \omterm{def:g10:functions:variations}{décroissante} sur
$\intoo{0}{+\infty}$ et $0.9 < 1.1$.

$(-1.01)^3 < (-0.99)^3$ car $x^3$ est \omterm{def:g11:func:monotone}{croissante} et $-1.01 < -0.99$.
\end{solution}

\begin{solution}{exo:g11:func:3}
$f(-x) = 3(-x)^4 - (-x)^2 = 3x^4 - x^2 = f(x)$~: \omterm{def:g11:func:parity}{paire}.

$g(-x) = -x^3 - x = -g(x)$~: \omterm{def:g11:func:parity}{impaire}.

$h(1) = 2$ et $h(-1) = 1 - 1 = 0$, donc $h(-1)$ n'est ni $h(1)$ ni
$-h(1)$~: ni \omterm{def:g11:func:parity}{paire} ni \omterm{def:g11:func:parity}{impaire}.

$k(-x) = \frac{-x}{x^2+1} = -k(x)$~: \omterm{def:g11:func:parity}{impaire}.
\end{solution}

\begin{solution}{exo:g11:func:4}
Lire le \omterm{def:g10:functions:table}{tableau de variations}. Sur $\intcc{-3}{1}$, $f$ monte de $-2$ à
$4$ et croise la valeur $1$ exactement une fois~; sur $\intcc{1}{5}$
elle descend de $4$ à $0$ et croise $1$ une fois encore~: l'\omterm{def:g10:algebra:equation}{équation}
$f(x) = 1$ a $2$ solutions. Le \omterm{def:g10:functions:extrema}{maximum} de $f$ est $f(1) = 4 < 5$, donc
$f(x) = 5$ n'a aucune solution.
\end{solution}

\begin{solution}{exo:g11:func:5}
Soit $2 \leq u < v$. Alors
\[
f(v) - f(u) = v^2 - 4v - u^2 + 4u = (v-u)(v+u) - 4(v-u)
= (v - u)(u + v - 4).
\]
Les deux facteurs sont positifs~: $v - u > 0$, et $u + v > 2 + 2 = 4$
car $u \geq 2$ et $v > 2$. Donc $f(v) > f(u)$~: $f$ est strictement
\omterm{def:g11:func:monotone}{croissante} sur $\intco{2}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:6}
$f = \sqrt{u}$ avec $u(x) = 5 - x$ \omterm{def:g10:functions:variations}{décroissante} et positive ou nulle sur
$\intoc{-\infty}{5}$~: $f$ est \omterm{def:g10:functions:variations}{décroissante} là
(\cref{prop:g11:func:transforms}).

$g = \frac1u$ avec $u(x) = x^2 + 1$ \omterm{def:g11:func:monotone}{croissante} et positive sur
$\intco{0}{+\infty}$~: $g$ est \omterm{def:g10:functions:variations}{décroissante} là.

$h = 3 - 2\sqrt x$~: $\sqrt x$ est \omterm{def:g11:func:monotone}{croissante}, multiplier par $-2 < 0$
inverse, ajouter $3$ ne change rien~: $h$ est \omterm{def:g10:functions:variations}{décroissante} sur
$\intco{0}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:7}
\emph{1.} $f$ est sous \omterm{thm:g11:quad:canonical}{forme canonique}~: \omterm{def:g10:functions:variations}{décroissante} sur
$\intoc{-\infty}{2}$, \omterm{def:g11:func:monotone}{croissante} sur $\intco{2}{+\infty}$, \omterm{def:g10:functions:extrema}{minimum}
$f(2) = -3$.

\emph{2.} $g = \frac{1}{u}$ avec $u(x) = (x-2)^2 + 1 = f(x) + 4$. Le
décalage de $4$ conserve les variations de $f$, et $u \geq 1 > 0$.
Prendre l'inverse les inverse~: $g$ est \omterm{def:g11:func:monotone}{croissante} sur
$\intoc{-\infty}{2}$ et \omterm{def:g10:functions:variations}{décroissante} sur $\intco{2}{+\infty}$, avec
\omterm{def:g10:functions:extrema}{maximum} $g(2) = \frac{1}{1} = 1$.
\end{solution}

\begin{solution}{exo:g11:func:8}
Toute courbe en forme de colline passant par les trois points convient.
Puis~: $f + 2$ passe par $(0,3)$, $(2,5)$, $(4,3)$, mêmes variations
(décalage vertical)~; $-f$ passe par $(0,-1)$, $(2,-3)$, $(4,-1)$ avec
des variations \emph{inversées} (le pic devient une vallée)~; $2f$
passe par $(0,2)$, $(2,6)$, $(4,2)$, mêmes variations mais amplifiées
verticalement.
\end{solution}

\begin{solution}{exo:g11:func:9}
$2 - \frac{1}{x+1} = \frac{2(x+1) - 1}{x+1} = \frac{2x+1}{x+1} = f(x)$.
Sur $\intoo{-1}{+\infty}$, $u(x) = x + 1$ est \omterm{def:g11:func:monotone}{croissante} et positive,
donc $\frac1u$ est \omterm{def:g10:functions:variations}{décroissante}, $-\frac1u$ est \omterm{def:g11:func:monotone}{croissante}, et ajouter
$2$ ne change rien~: $f$ est strictement \omterm{def:g11:func:monotone}{croissante} sur
$\intoo{-1}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:10}
\emph{1.~Vrai.} Si $u < v$, alors
$(f+g)(v) - (f+g)(u) = \bigl(f(v)-f(u)\bigr) + \bigl(g(v)-g(u)\bigr)$
est une somme de deux nombres positifs ou nuls.

\emph{2.~Faux.} $f(x) = g(x) = x$ sont \omterm{def:g11:func:monotone}{croissantes} sur $\R$, mais leur
produit $x^2$ n'est pas croissant sur $\R$ (il décroît sur
$\intoc{-\infty}{0}$). (L'assertion devient vraie si les deux \omterm{def:g11:func:function}{fonctions}
sont aussi positives ou nulles.)

\emph{3.~Faux.} Même contre-exemple~: $f(x) = x$ est \omterm{def:g11:func:monotone}{croissante} sur
$\R$ mais $f^2(x) = x^2$ ne l'est pas.
\end{solution}

\begin{solution}{exo:g11:func:11}
\emph{1.} Pour $0 < u < v$~:
\[
f(v) - f(u) = (v - u) + \frac1v - \frac1u
= (v - u) + \frac{u - v}{uv}
= (v - u)\left(1 - \frac{1}{uv}\right)
= (v - u)\,\frac{uv - 1}{uv}.
\]

\emph{2.} Toujours $v - u > 0$ et $uv > 0$. Si $0 < u < v \leq 1$,
alors $uv < 1$, donc $uv - 1 < 0$ et $f(v) < f(u)$~: $f$ est strictement
\omterm{def:g10:functions:variations}{décroissante} sur $\intoc{0}{1}$. Si $1 \leq u < v$, alors $uv > 1$ et
$f(v) > f(u)$~: $f$ est strictement \omterm{def:g11:func:monotone}{croissante} sur
$\intco{1}{+\infty}$.

\emph{3.} $f$ décroît avant $1$ et croît après, donc
$f(x) \geq f(1) = 2$ pour tout $x > 0$, avec égalité exactement en
$x = 1$. (Cette inégalité réapparaît sous de nombreuses formes~;
comparer l'\cref{ex:g11:quad:canonical} et les inégalités de tangente
du \cref{ch:g12:deriv}.)
\end{solution}

\begin{solution}{pb:g11:func:1}
\textbf{1.} $x^4 - 3x^2$~: \omterm{def:g11:func:parity}{paire} ($(-x)^4 - 3(-x)^2 = x^4 -
3x^2$). $x^3 - x$~: \omterm{def:g11:func:parity}{impaire}. $x^2 + x$~: ni l'une ni l'autre
($f(-1) = 0 \neq f(1) = 2$ et $\neq -f(1)$). $\abs x$~: \omterm{def:g11:func:parity}{paire}.
$\frac1x$~: \omterm{def:g11:func:parity}{impaire}.

\textbf{2.} \omterm{def:g11:func:parity}{Paire}~: la courbe est symétrique par rapport à l'axe des
ordonnées (un miroir). \omterm{def:g11:func:parity}{Impaire}~: symétrique par rapport à l'origine
(un demi-tour).

\textbf{3.} Si $f$ et $g$ sont \omterm{def:g11:func:parity}{impaires}~:
$(fg)(-x) = f(-x)g(-x) = (-f(x))(-g(x)) = (fg)(x)$~: le produit est
pair. Table~: \omterm{def:g11:func:parity}{paire} $\times$ \omterm{def:g11:func:parity}{paire} $=$ \omterm{def:g11:func:parity}{paire}~; \omterm{def:g11:func:parity}{paire} $\times$
\omterm{def:g11:func:parity}{impaire} $=$ \omterm{def:g11:func:parity}{impaire}~; \omterm{def:g11:func:parity}{impaire} $\times$ \omterm{def:g11:func:parity}{impaire} $=$ \omterm{def:g11:func:parity}{paire} --- la
règle des signes, transposée aux parités.

\textbf{4.} \omterm{def:g11:func:parity}{Impaire} en $0$~: $f(0) = -f(0)$, donc $f(0) = 0$.
\omterm{def:g11:func:parity}{Paire}~: $f(-3) = f(3) = 7$. À la fois \omterm{def:g11:func:parity}{paire} et \omterm{def:g11:func:parity}{impaire}~:
$f(x) = f(-x) = -f(x)$ pour tout $x$, donc $2f(x) = 0$~: $f$ est la
\omterm{def:g11:func:function}{fonction} nulle.

\textbf{5.} La moitié droite ($x \geq 0$) détermine entièrement une
\omterm{def:g11:func:parity}{fonction paire}~: la moitié gauche en est le miroir. Ni \omterm{def:g11:func:parity}{paire} ni
\omterm{def:g11:func:parity}{impaire}~: $x^2 + x$ (question 1) --- ses deux symétries échouent
en $x = 1$.

\textbf{6.} $E(x) = \frac{(x^3 + x^2 + 1) + (-x^3 + x^2 +
1)}{2} = x^2 + 1$ (\omterm{def:g11:func:parity}{paire}), $O(x) = \frac{(x^3 + x^2 + 1) -
(-x^3 + x^2 + 1)}{2} = x^3$ (\omterm{def:g11:func:parity}{impaire})~; leur somme vaut $f$.

\textbf{7.} $E(-x) = \frac{f(-x) + f(x)}{2} = E(x)$~: $E$ est \omterm{def:g11:func:parity}{paire}.
$O(-x) = \frac{f(-x) - f(x)}{2} = -O(x)$~: $O$ est \omterm{def:g11:func:parity}{impaire}. Et
$E(x) + O(x) = f(x)$ par addition directe~: toute \omterm{def:g11:func:function}{fonction} se
scinde.

\textbf{8.} De $E_1 + O_1 = E_2 + O_2$ on tire
$E_1 - E_2 = O_2 - O_1$. Le membre de gauche est pair, celui de
droite impair --- cette \omterm{def:g11:func:function}{fonction} est donc les deux à la fois, donc
nulle (question 4)~: $E_1 = E_2$ et $O_1 = O_2$. Une seule
décomposition.

\textbf{9.} $E(x) = \frac12\left(\frac{1}{1 - x} + \frac{1}{1 +
x}\right) = \frac12 \cdot \frac{(1 + x) + (1 - x)}{1 - x^2}
= \frac{1}{1 - x^2}$, et
$O(x) = \frac12 \cdot \frac{(1 + x) - (1 - x)}{1 - x^2}
= \frac{x}{1 - x^2}$. Vérification~: $E$ est \omterm{def:g11:func:parity}{paire}, $O$ est \omterm{def:g11:func:parity}{impaire},
et leur somme vaut $\frac{1 + x}{1 - x^2} =
\frac{1 + x}{(1 - x)(1 + x)} = \frac{1}{1 - x} = f(x)$.

\textbf{10.} Parce que chaque morceau obéit à une symétrie, la
moitié du travail (et la moitié des données) suffit pour chacun~: on
résout d'un côté, on recopie en miroir de l'autre --- et beaucoup
d'opérations (intégration, séries, analyse du signal) traitent les
morceaux symétriques bien plus simplement que les \omterm{def:g11:func:function}{fonctions} brutes.

\textbf{11.} $(x-3)^2$~: translatée de $3$ vers la droite.
$x^2 + 2$~: translatée de $2$ vers le haut. $(x-3)^2 + 2$~: les
deux. $-x^2$~: retournée vers le bas. $2x^2$~: dilatée verticalement
d'un facteur $2$ (cuvette plus étroite).

\textbf{12.} Pour $x \geq 2$~: $(x - 2) + (x + 2) = 2x$. Pour
$-2 \leq x \leq 2$~: $(2 - x) + (x + 2) = 4$. Pour $x \leq -2$~:
$-2x$. La courbe descend avec un \omterm{def:g10:reffunc:affine}{coefficient directeur} $-2$, court à
plat à la hauteur $4$ entre $-2$ et $2$, puis remonte avec un
\omterm{def:g10:reffunc:affine}{coefficient directeur} $2$~: une vallée à fond plat.

\textbf{13.} Translation verticale~: $f(x) + 2$ reste \omterm{def:g11:func:parity}{paire}
($f(-x) + 2 = f(x) + 2$). Dilatation verticale~: $2f(x)$ reste
\omterm{def:g11:func:parity}{paire}. Translation horizontale~: $(x - 3)^2$ n'est pas \omterm{def:g11:func:parity}{paire}
($f(-1) = 16 \neq 4 = f(1)$)~: glisser latéralement détruit le
miroir.

\textbf{14.} Sur le fond plat la valeur est $4 \neq 6$. Pour
$x \geq 2$~: $2x = 6$, donc $x = 3$. Pour $x \leq -2$~: $-2x = 6$,
donc $x = -3$. Solutions~: $\pm 3$.

\textbf{15.} $f(x) = a(x - 2)^2 - 3$ avec $f(0) = 4a - 3 = 1$~:
$a = 1$, donc $f(x) = (x - 2)^2 - 3$.

\textbf{16.} $g(v) - g(u) = (v - u) + 4\,\frac{u - v}{uv}
= (v - u)\,\frac{uv - 4}{uv}$~: négatif pour $u < v$ avec
$uv < 4$, positif avec $uv > 4$~: $g$ est \omterm{def:g10:functions:variations}{décroissante} sur
$\intoc{0}{2}$ et \omterm{def:g11:func:monotone}{croissante} sur $\intco{2}{+\infty}$~; \omterm{def:g10:functions:extrema}{minimum} en
$x = 2$, de valeur $g(2) = 4$.

\textbf{17.} Côtés $x$ et $\frac{16}{x}$~:
$P(x) = 2\left(x + \frac{16}{x}\right)$, minimal (machine de la
question 16 avec $4 \to 16$, point de retournement $\sqrt{16} = 4$)
en $x = 4$~: le \emph{carré} $4 \times 4$, de périmètre $16$ --- la
réponse de Didon, désormais démontrée pour toute longueur réelle de
côté.

\textbf{18.} Sur $\intco{0}{+\infty}$~: $x \mapsto x^2 + 1$ est
\omterm{def:g11:func:monotone}{croissante} et $\sqrt{\phantom{x}}$ l'est aussi, donc la composée est
\omterm{def:g11:func:monotone}{croissante}. Et $h(-x) = \sqrt{x^2 + 1} = h(x)$~: $h$ est \omterm{def:g11:func:parity}{paire}, donc
par symétrie en miroir elle décroît sur $\intoc{-\infty}{0}$.
\omterm{def:g10:functions:extrema}{Minimum} $h(0) = 1$.

\textbf{19.} Carré de la distance de $(x, \sqrt x)$ à $(3, 0)$~:
$d^2(x) = (x - 3)^2 + x = x^2 - 5x + 9$, minimal en
$x = \frac52$~: point le plus proche
$\left(\frac52, \sqrt{\frac52}\right)$, à la distance
$\sqrt{\frac{11}{4}} = \frac{\sqrt{11}}{2}$. Minimiser $d^2$ est
légitime parce que l'élévation au carré est strictement \omterm{def:g11:func:monotone}{croissante}
sur les nombres positifs~: $d$ et $d^2$ atteignent leur creux au même
endroit.

\textbf{20.} Matière première~: les courbes de référence, sues par
cœur. Machines~: translations, retournements et dilatations, qui
assemblent de nouvelles courbes à partir des anciennes. Contrôle
qualité~: la parité et la décomposition pair-impair, qui divisent par
deux toute étude. Banc de mesure~: la factorisation de
$f(v) - f(u)$, qui lit directement variations et \omterm{def:g10:functions:extrema}{minimums}. L'outil
de puissance du chapitre suivant~: la \emph{dérivée}, qui les lit
d'un coup d'œil.
\end{solution}
