\section*{Chapitre \ref{ch:g11:stat} --- Statistique descriptive}

\begin{solution}{exo:g11:stat:1}
$n = 8$, somme $71$~: \omterm{def:g11:stat:mean}{moyenne} $\bar x = \frac{71}{8} = 8.875$. \omterm{def:g11:stat:median}{Médiane}~:
\omterm{prop:g10:coordgeom:midpoint}{milieu} des $4$-ième et $5$-ième valeurs, $\frac{8 + 9}{2} = 8.5$.
\omterm{def:g11:stat:quartiles}{Quartiles}~: un quart de $8$ est $2$, donc $Q_1$ est la $2$-ième
valeur, $7$~; trois quarts est $6$, donc $Q_3$ est la $6$-ième valeur,
$10$. \omterm{def:g11:stat:quartiles}{Écart interquartile}~: $10 - 7 = 3$.
\end{solution}

\begin{solution}{exo:g11:stat:2}
Somme $40$, donc $\bar x = 5$. Somme des carrés~:
$4 + 16 + 16 + 16 + 25 + 25 + 49 + 81 = 232$. Par la formule de
König--Huygens,
\[
V = \frac{232}{8} - 5^2 = 29 - 25 = 4,
\qquad
\sigma = 2 .
\]
\end{solution}

\begin{solution}{exo:g11:stat:3}
\omterm{def:g11:stat:mean}{Moyenne}~:
\[
\bar x = \frac{1 \times 6 + 2 \times 9 + 3 \times 8 + 4 \times 10
+ 5 \times 9 + 6 \times 8}{50}
= \frac{181}{50} = 3.62 .
\]
\omterm{def:g11:stat:median}{Médiane}~: les effectifs cumulés sont $6, 15, 23, 33, \dots$, donc les
$25$-ième et $26$-ième issues triées sont toutes deux $4$~: la \omterm{def:g11:stat:median}{médiane}
est $4$.
\end{solution}

\begin{solution}{exo:g11:stat:4}
Le total des $15$ notes est $15 \times 11.2 = 168$~; avec le nouvel
arrivant, $\frac{168 + 18}{16} = \frac{186}{16} = 11.625$.
\end{solution}

\begin{solution}{exo:g11:stat:5}
$\frac{6 + 8 + x + 12 + 14}{5} = 10$ donne $40 + x = 50$, donc
$x = 10$. Les écarts de $6, 8, 10, 12, 14$ à la \omterm{def:g11:stat:mean}{moyenne} $10$ sont
$-4, -2, 0, 2, 4$, de carrés sommant à $40$~:
$V = \frac{40}{5} = 8$ et $\sigma = 2\sqrt2 \approx 2.83$.
\end{solution}

\begin{solution}{exo:g11:stat:6}
Par la \cref{prop:g11:stat:affine} avec $a = \frac12$, $b = 4$~: nouvelle
\omterm{def:g11:stat:mean}{moyenne} $\frac{12.4}{2} + 4 = 10.2$~; nouvel \omterm{def:g11:stat:variance}{écart type}
$\frac{2.5}{2} = 1.25$.
\end{solution}

\begin{solution}{exo:g11:stat:7}
La \omterm{def:g11:stat:mean}{moyenne} regroupée est la \omterm{def:g11:stat:mean}{moyenne} \emph{pondérée}
\[
\frac{20 \times 10 + 30 \times 15}{50} = \frac{650}{50} = 13,
\]
pas $12.5$~: le groupe B contribue plus de valeurs, donc tire la \omterm{def:g11:stat:mean}{moyenne}
regroupée vers la sienne.
\end{solution}

\begin{solution}{exo:g11:stat:8}
Les deux chaînes sont presque dans la cible en \omterm{def:g11:stat:mean}{moyenne} ($1.002$ et
$1.001$~kg), mais l'\omterm{def:g11:stat:variance}{écart type} de la chaîne~2 est trois fois plus grand~:
ses sacs varient beaucoup plus, produisant bien plus de sacs
manifestement sous- et sur-remplis. La chaîne~2 doit être traitée en
premier --- c'est la régularité, pas seulement la \omterm{def:g11:stat:mean}{moyenne}, qu'il faut
corriger.
\end{solution}

\begin{solution}{exo:g11:stat:9}
$\bar x = \frac{70}{10} = 7$~;
$V = \frac{540}{10} - 7^2 = 54 - 49 = 5$~;
$\sigma = \sqrt5 \approx 2.24$.
\end{solution}

\begin{solution}{exo:g11:stat:10}
\emph{1.} Avec $23$~: somme $50$, \omterm{def:g11:stat:mean}{moyenne} $5$~; \omterm{def:g11:stat:median}{médiane} $=$ \omterm{prop:g10:coordgeom:midpoint}{milieu} des
$5$-ième et $6$-ième valeurs $= 3$. Sans $23$~: somme $27$, \omterm{def:g11:stat:mean}{moyenne}
$3$~; \omterm{def:g11:stat:median}{médiane} ($5$-ième sur $9$ valeurs) $= 3$.

\emph{2.} Avec $23$ ($n = 10$)~: $Q_1$ est la $3$-ième valeur ($2$),
$Q_3$ la $8$-ième ($4$)~: \omterm{def:g11:stat:quartiles}{écart interquartile} $2$. Sans ($n = 9$)~:
$Q_1$ est la $3$-ième valeur ($2$), $Q_3$ la $7$-ième ($4$)~: encore
$2$.

\emph{3.} L'unique valeur aberrante tire la \omterm{def:g11:stat:mean}{moyenne} de $3$ à $5$ mais
laisse la \omterm{def:g11:stat:median}{médiane} et l'\omterm{def:g11:stat:quartiles}{écart interquartile} inchangés~: ces deux-là sont
\emph{robustes}. Lorsqu'un jeu de données peut contenir des valeurs
aberrantes, rapporter la \omterm{def:g11:stat:median}{médiane} et l'\omterm{def:g11:stat:quartiles}{écart interquartile} plutôt que la
\omterm{def:g11:stat:mean}{moyenne} et l'\omterm{def:g11:stat:variance}{écart type}.
\end{solution}

\begin{solution}{pb:g11:stat:1}
\textbf{1.} \omterm{def:g11:stat:mean}{Moyenne} $\bar x = \frac{25}{5} = 5$. Carrés des
écarts~: $1, 9, 1, 0, 9$~: \omterm{def:g11:stat:variance}{variance} $\frac{20}{5} = 4$, donc
$\sigma = 2$.

\textbf{2.} \omterm{def:g11:stat:mean}{Moyenne} des carrés~:
$\frac{16 + 64 + 36 + 25 + 4}{5} = 29$~; moins
$\bar x^2 = 25$~: \omterm{def:g11:stat:variance}{variance} $4$. Même réponse, moitié moins de
travail.

\textbf{3.} \omterm{def:g11:stat:mean}{Moyenne} $= \frac{3 \times 8 + 5 \times 10 + 2 \times
14}{10} = 10.2$~; \omterm{def:g11:stat:variance}{variance} $= \frac{3 \times 64 + 5 \times 100 +
2 \times 196}{10} - 10.2^2 = 108.4 - 104.04 = 4.36$~;
$\sigma \approx 2.09$.

\textbf{4.} \omterm{def:g11:stat:mean}{Moyenne}~: $1.8 \times 20 + 32 = 68\,^\circ$F~;
$\sigma$~: $1.8 \times 3 = 5.4\,^\circ$F --- la translation de
$+32$ n'étale rien du tout.

\textbf{5.} En ajoutant $b$~: chaque écart
$x_i + b - (\bar x + b) = x_i - \bar x$ est inchangé, donc la
\omterm{def:g11:stat:variance}{variance} aussi. En multipliant par $a$~: les écarts sont multipliés
par $a$, leurs carrés par $a^2$, la \omterm{def:g11:stat:variance}{variance} par $a^2$, et $\sigma$
par $\sqrt{a^2} = \abs a$.

\textbf{6.} $z_A = \frac{15 - 12}{2} = 1.5$~;
$z_B = \frac{13 - 8}{4} = 1.25$. Alice se détache davantage de son
groupe~: le $15$ du contrôle difficile bat le $13$ du contrôle
facile.

\textbf{7.} Cotes $z$~: $-0.5,\ 1.5,\ 0.5,\ 0,\ -1.5$~: \omterm{def:g11:stat:mean}{moyenne}
$0$, \omterm{def:g11:stat:variance}{écart type} $1$. En général, standardiser revient à retrancher
$\bar x$ (ce qui annule la \omterm{def:g11:stat:mean}{moyenne} sans toucher à $\sigma$) puis à
diviser par $\sigma$ (ce qui ramène le nouvel \omterm{def:g11:stat:variance}{écart type} à $1$)~: la
question 5, appliquée deux fois.

\textbf{8.} $z = 2$~: $100 + 2 \times 15 = 130$. Et
$z(76) = \frac{76 - 100}{15} = -1.6$.

\textbf{9.} Le bébé se situe $2.5$ \omterm{def:g11:stat:variance}{écarts types} au-dessous du poids
moyen \emph{des bébés de son âge} --- inquiétant pour sa cohorte.
Les kilogrammes bruts ne se comparent pas d'un âge à l'autre (tous
les bébés grossissent)~; les \omterm{pb:g11:stat:1}{cotes z} mesurent chaque enfant face à
sa propre classe d'âge, une seule règle pour tous les âges.

\textbf{10.} \omterm{def:g11:stat:mean}{Moyenne} $5$~; \omterm{def:g11:stat:mean}{moyenne} des carrés
$\frac{622}{10} = 62.2$, \omterm{def:g11:stat:variance}{variance} $37.2$, $\sigma \approx 6.10$~;
$z(23) = \frac{18}{6.10} \approx 2.95$~: juste sous le seuil
d'alerte --- le test cille à peine.

\textbf{11.} Sans le $23$~: \omterm{def:g11:stat:mean}{moyenne} $3$, \omterm{def:g11:stat:variance}{variance}
$\frac{93}{9} - 9 \approx 1.33$, $\sigma \approx 1.15$~; face à
cette règle assainie, $23$ obtient
$z \approx \frac{20}{1.15} \approx 17$~: un monstre. La valeur
aberrante avait gonflé à la fois la \omterm{def:g11:stat:mean}{moyenne} et $\sigma$, se
camouflant ainsi elle-même --- c'est pourquoi
l'\cref{exo:g11:stat:10} recommande des résumés robustes (\omterm{def:g11:stat:median}{médiane},
\omterm{def:g11:stat:quartiles}{écart interquartile}) lorsque la contamination menace.

\textbf{12.} Dans la somme
$\sigma^2 = \frac1N \sum (x_i - \bar x)^2$, ne gardons que les
termes vérifiant $\abs{x_i - \bar x} \geq k\sigma$~: il y en a $pN$,
chacun valant au moins $(k\sigma)^2$, donc
$\sigma^2 \geq \frac{1}{N} \cdot pN \cdot k^2\sigma^2 =
p\,k^2\sigma^2$~; en divisant par $k^2 \sigma^2$ (dispersion non
nulle)~: $p \leq \frac{1}{k^2}$.

\textbf{13.} Au-delà de $2\sigma$~: au plus $\frac14 = 25\,\%$ de
n'importe quelle série~; au-delà de $3\sigma$~: au plus
$\frac19 \approx 11\,\%$. Pour les données de la question 1~:
au-delà de $2\sigma = 4$ de la \omterm{def:g11:stat:mean}{moyenne} $5$ signifie $\leq 1$ ou
$\geq 9$, c'est-à-dire aucune des cinq valeurs ---
$0\,\% \leq 25\,\%$, comme promis.

\textbf{14.} Être hors de $\intcc{49.7}{50.3}$, c'est être au-delà
de $3\sigma$~: au plus $\frac19$ de la production, \emph{quelle que
soit} la forme de la distribution --- Tchebychev n'exige rien
d'autre qu'une \omterm{def:g11:stat:mean}{moyenne} et un $\sigma$. Une production en cloche se
concentre bien mieux (environ $0.3\,\%$ au-delà de $3\sigma$)~: la
loi normale, l'an prochain.

\textbf{15.} $-19\,\%$ se situe $24$ points sous la \omterm{def:g11:stat:mean}{moyenne} $5$.
Fonds B~: $24 = 2\sigma_B$, donc Tchebychev autorise jusqu'à
$\frac14$ des années aussi mauvaises. Fonds A~: $24 = 12\sigma_A$,
donc au plus $\frac{1}{144}$. Même rendement moyen, dangers sans
commune mesure --- $\sigma$ est ce que la finance appelle le
\emph{risque} (ou la volatilité).

\textbf{16.} Développons autour de $\bar x$~:
$(x_i - c)^2 = \left((x_i - \bar x) + (\bar x - c)\right)^2
= (x_i - \bar x)^2 + 2(\bar x - c)(x_i - \bar x) +
(\bar x - c)^2$. En moyennant, le terme du \omterm{prop:g10:coordgeom:midpoint}{milieu} disparaît (les
écarts à la \omterm{def:g11:stat:mean}{moyenne} se compensent), et il reste
$f(c) = \sigma^2 + (\bar x - c)^2$ --- minimal exactement en
$c = \bar x$, avec pour \omterm{def:g10:functions:extrema}{minimum} $\sigma^2$. La \omterm{def:g11:stat:mean}{moyenne} est le centre
des moindres carrés, comme la \omterm{def:g11:stat:median}{médiane} est le centre des moindres
écarts absolus (\cref{pb:g10:stats:1}).

\textbf{17.} Directement~: les écarts à $6$ valent
$-2, 2, 0, -1, -4$, de carrés $4, 4, 0, 1, 16$, donc
$f(6) = \frac{25}{5} = 5$. Par la formule~:
$\sigma^2 + (5 - 6)^2 = 4 + 1 = 5$. Les deux concordent.

\textbf{18.} Les deux séries ont pour \omterm{def:g11:stat:mean}{moyenne} $7$~; \omterm{def:g11:stat:variance}{variances}~:
$P$~: $\frac{16 + 4 + 0 + 4 + 16}{5} = 8$~;
$Q$~: $\frac{36 + 0 + 0 + 0 + 36}{5} = 14.4$~: deux dispersions
derrière une même \omterm{def:g11:stat:mean}{moyenne}. Pour $\{7 - a, 7, 7, 7, 7 + a\}$~: la
\omterm{def:g11:stat:variance}{variance} $\frac{2a^2}{5} = 8$ donne $a = \sqrt{20} \approx 4.47$~:
la série $\{2.53, 7, 7, 7, 11.47\}$ porte exactement la \omterm{def:g11:stat:mean}{moyenne} et
la \omterm{def:g11:stat:variance}{variance} de $P$ avec un profil complètement différent --- les
résumés compriment, et comprimer fait perdre.

\textbf{19.} $\sigma^2 = 0$ signifie qu'une somme de carrés
$(x_i - \bar x)^2$ s'annule~; les carrés étant $\geq 0$, une somme
nulle force chaque terme à être nul~: $x_i = \bar x$ pour tout $i$
--- toutes les valeurs sont égales.

\textbf{20.} Les centres disent où vivent les données~; les
dispersions, avec quel relâchement~; la \omterm{pb:g11:stat:1}{cote z} pose une seule
règle sur toutes les séries, tous les contrôles, tous les âges et
toutes les monnaies~; Tchebychev convertit $\sigma$ en garantie
inconditionnelle~; et la question 18 nous garde honnêtes --- jamais
une poignée de nombres ne contient toute l'histoire. L'an prochain,
la courbe en cloche affinera la règle en \omterm{def:g10:proba:distribution}{probabilités} exactes.
\end{solution}
