\section*{Chapitre \ref{ch:g11:binom} --- La loi binomiale}

\begin{solution}{exo:g11:binom:1}
Lignes $6$ et $7$~:
\[
1,\ 6,\ 15,\ 20,\ 15,\ 6,\ 1
\qquad\text{et}\qquad
1,\ 7,\ 21,\ 35,\ 35,\ 21,\ 7,\ 1 .
\]
D'où $\binom62 = 15$, $\binom73 = 35$, $\binom74 = 35$ (la symétrie
$\binom73 = \binom74$ reflète l'échange des succès et des échecs).
\end{solution}

\begin{solution}{exo:g11:binom:2}
Nombre fixe d'épreuves ($4$ lancers), deux issues par lancer (six ou
non, $p = \frac16$), lancers indépendants~:
$X \sim \mathcal B(4, \frac16)$.
\[
\P(X=0) = \left(\frac56\right)^4 = \frac{625}{1296} \approx 0.48,
\qquad
\P(X=1) = 4 \times \frac16\left(\frac56\right)^3 = \frac{500}{1296}
\approx 0.39,
\]
\[
\P(X \geq 2) = 1 - \frac{625 + 500}{1296} = \frac{171}{1296}
\approx 0.13 .
\]
\end{solution}

\begin{solution}{exo:g11:binom:3}
\emph{1.} Binomiale $\mathcal B(20, \frac12)$~: $n$ fixe, même $p$,
lancers indépendants.

\emph{2.} Pas binomiale~: les cartes sont distribuées \emph{sans
remise}, donc la \omterm{def:g10:proba:distribution}{probabilité} d'un as change d'une carte à l'autre et les
tirages ne sont pas indépendants.

\emph{3.} Binomiale $\mathcal B(7, 0.3)$ par l'hypothèse d'indépendance
énoncée.
\end{solution}

\begin{solution}{exo:g11:binom:4}
$\E(X) = 50 \times 0.2 = 10$~;
$\V(X) = 50 \times 0.2 \times 0.8 = 8$~;
$\sigma(X) = 2\sqrt2 \approx 2.83$.
\end{solution}

\begin{solution}{exo:g11:binom:5}
$X \sim \mathcal B(6, 0.7)$.
\[
\P(X = 4) = \binom64 (0.7)^4 (0.3)^2 = 15 \times 0.2401 \times 0.09
\approx 0.324 .
\]
\[
\P(X \geq 5) = \binom65 (0.7)^5(0.3) + (0.7)^6
= 6 \times 0.16807 \times 0.3 + 0.117649 \approx 0.420 .
\]
\end{solution}

\begin{solution}{exo:g11:binom:6}
$X \sim \mathcal B\left(8, \frac12\right)$~; chaque chemin a pour
\omterm{def:g10:proba:distribution}{probabilité} $\frac{1}{256}$, donc
\[
\P(X \geq 6) = \frac{\binom86 + \binom87 + \binom88}{256}
= \frac{28 + 8 + 1}{256} = \frac{37}{256} \approx 0.14 .
\]
Deviner fait réussir environ une fois sur sept.
\end{solution}

\begin{solution}{exo:g11:binom:7}
Le \omterm{def:g10:proba:operations}{complémentaire} de «~au moins une de chaque~» est «~les cinq
identiques~»~: toutes A ou toutes B, chacune de \omterm{def:g10:proba:distribution}{probabilité}
$\left(\frac12\right)^5 = \frac1{32}$. D'où
\[
\P(\text{un de chaque type}) = 1 - \frac{2}{32} = \frac{15}{16} .
\]
\end{solution}

\begin{solution}{exo:g11:binom:8}
$\P(X = 3) = p^3$ et $\P(X \geq 1) = 1 - (1-p)^3$. Résoudre
$p^3 = \frac{27}{64} = \left(\frac34\right)^3$ donne $p = \frac34$ (la
\omterm{def:g11:func:function}{fonction} cube est strictement \omterm{def:g11:func:monotone}{croissante}, \cref{ch:g11:func}, donc la
solution est unique).
\end{solution}

\begin{solution}{exo:g11:binom:9}
$\P(\text{au moins une face}) = 1 - \left(\frac12\right)^n$, donc la
condition est $\left(\frac12\right)^n < 0.01$, c'est-à-dire
$2^n > 100$. Comme $2^6 = 64$ et $2^7 = 128$~: à partir de $n = 7$
lancers.
\end{solution}

\begin{solution}{exo:g11:binom:10}
La somme de la ligne~$n$ compte tous les chemins de l'\omterm{met:g10:proba:tree}{arbre} à $n$
épreuves, classés par leur nombre de succès. Mais l'\omterm{met:g10:proba:tree}{arbre} double ses
chemins à chaque épreuve (chaque chemin se scinde en S et E), donc il a
$2^n$ chemins au total. D'où $\sum_{k} \binom nk = 2^n$.
Alternativement, par récurrence~: la ligne~$0$ somme à $1 = 2^0$, et la
règle de Pascal fait que chaque entrée de la ligne~$n$ contribue à
exactement deux entrées de la ligne~$n+1$, donc les sommes de lignes
doublent.
\end{solution}

\begin{solution}{exo:g11:binom:11}
\emph{1.} Sous l'affirmation, $X \sim \mathcal B(10, 0.6)$. En sommant
les premiers termes~:
\begin{align*}
\P(X \leq 3)
&= (0.4)^{10} + 10(0.6)(0.4)^9 + 45(0.6)^2(0.4)^8
+ 120(0.6)^3(0.4)^7\\
&\approx 0.0001 + 0.0016 + 0.0106 + 0.0425 = 0.0548 .
\end{align*}

\emph{2.} Un résultat au moins aussi extrême que celui observé ($3$
approbations ou moins) a une \omterm{def:g10:proba:distribution}{probabilité} d'environ $5.5\,\%$ --- juste
\emph{au-dessus} du seuil de $5\,\%$. En appliquant strictement la règle,
l'observation est (de justesse) compatible avec l'affirmation et on ne
la rejette pas. L'exemple montre à quel point les décisions limites sont
sensibles au choix du seuil~: avec une convention à $6\,\%$ la
conclusion s'inverserait.
\end{solution}

\begin{solution}{pb:g11:binom:1}
\textbf{1.} Lignes~: $1$~; $1\,1$~; $1\,2\,1$~; $1\,3\,3\,1$~;
$1\,4\,6\,4\,1$~; $1\,5\,10\,10\,5\,1$~;
$1\,6\,15\,20\,15\,6\,1$. Symétrie~: choisir les $k$ objets que
l'on prend est le même acte que choisir les $n - k$ que l'on
laisse.

\textbf{2.} $1 + 4 + 6 + 4 + 1 = 16 = 2^4$~;
$1 + 5 + 10 + 10 + 5 + 1 = 32 = 2^5$. Démonstration~: les
$\binom nk$ dénombrent les parties de chaque taille d'un ensemble à
$n$ éléments, et les parties sont au nombre de $2^n$ (chaque élément
est dedans ou dehors, indépendamment).

\textbf{3.} Comités de $k$ personnes choisies parmi $n + 1$, dont
l'une s'appelle Zoé~: ceux qui ne contiennent pas Zoé sont au nombre
de $\binom nk$ (on choisit les $k$ parmi les autres)~; ceux qui la
contiennent sont au nombre de $\binom{n}{k-1}$ (on choisit ses
$k - 1$ collègues). Total~: $\binom nk + \binom{n}{k-1}$.

\textbf{4.} Ligne $7$ du triangle~: $1\,7\,21\,35\,\dots$, d'où
$35$. Par la formule~: $\frac{7 \times 6 \times 5}{3 \times 2
\times 1} = 35$.

\textbf{5.} $1 + 3 + 6 + 10 = 20 = \binom63$. Cascade~:
$\binom63 = \binom52 + \binom53 = \binom52 + \binom42 +
\binom43 = \binom52 + \binom42 + \binom32 + \binom33$ --- chaque
application de la règle de Pascal détache une marche de l'escalier.

\textbf{6.} Nombre $n$ de rebonds fixé~; chaque rebond est une
\omterm{def:g11:binom:bernoulli}{épreuve de Bernoulli} indépendante de même paramètre
$p = \frac12$~; le numéro de case compte les succès (les rebonds à
droite)~: les trois cases de la
\cref{met:g11:binom:recognize} sont cochées, d'où
$\mathcal B\!\left(n, \frac12\right)$.

\textbf{7.} \omterm{def:g10:proba:distribution}{Probabilités} $\frac{1}{16}, \frac{4}{16},
\frac{6}{16}, \frac{4}{16}, \frac{1}{16}$ pour les cases
$0, \dots, 4$~: la case centrale $2$ est la plus chargée.

\textbf{8.} Effectifs attendus $= 1024 \times
\binom{10}{k}/1024 = \binom{10}{k}$~: case centrale
$\binom{10}{5} = 252$ billes~; case $7$~: $\binom{10}{7} = 120$~;
chaque case du bord~: $1$ bille. Un centre élevé qui redescend
symétriquement jusqu'à des bords ténus~: la cloche.

\textbf{9.} $\E = np = 5$~; $V = np(1 - p) = 2.5$~;
$\sigma \approx 1.58$. À moins de $2\sigma$~: les cases $2$ à $8$
portent
\[
\frac{45 + 120 + 210 + 252 + 210 + 120 + 45}{1024}
= \frac{1002}{1024} \approx 98\,\%
\]
des billes --- bien mieux que les $75\,\%$ tout-terrain de
Tchebychev (\cref{pb:g11:stat:1})~: les formes en cloche se
concentrent fort.

\textbf{10.} $\E = 6$, $V = 10 \times 0.6 \times 0.4 = 2.4$,
$\sigma \approx 1.55$~: le tas garde sa forme de cloche mais fait
glisser son sommet sur la case $6$ --- une planche inclinée est une
pièce truquée rendue visible.

\textbf{11.} Le numéro de case est une \emph{somme} de nombreuses
poussées aléatoires petites, indépendantes et de même taille --- et
de telles sommes s'organisent toujours en cloche~: la plupart des
poussées se compensent, les extrêmes exigent l'unanimité. Les
tailles, les erreurs de mesure et d'innombrables grandeurs
naturelles sont de même des sommes de nombreux petits effets
indépendants, ce qui explique que la même silhouette apparaisse
partout~; le théorème qui le certifie est le théorème central
limite.

\textbf{12.} Un balayage~: une équipe gagne les $4$ matchs~:
$2 \times \left(\frac12\right)^4 = \frac18$.

\textbf{13.} Sept matchs exigent un score de $3$--$3$ après six~:
$\binom63 \left(\frac12\right)^6 = \frac{20}{64} = \frac{5}{16}$.

\textbf{14.} Fin en $5$ matchs~: le vainqueur remporte le
cinquième et $3$ des quatre premiers~:
$2 \times \binom43 \left(\frac12\right)^5 = \frac14$. Fin en $6$~:
$2 \times \binom53 \left(\frac12\right)^6 = \frac{5}{16}$. \omterm{def:g11:prob:rv}{Loi} sur
$4, 5, 6, 7$~: $\frac18, \frac14, \frac{5}{16}, \frac{5}{16}$
(de somme $1$). Longueur espérée~:
$4 \cdot \frac18 + 5 \cdot \frac14 + 6 \cdot \frac{5}{16} +
7 \cdot \frac{5}{16} = 5.8125$ matchs. Les séries de six et sept
matchs sont les plus probables --- le suspense est inscrit dans le
format.

\textbf{15.} Victoire en $4$~: $0.6^4 = 0.1296$~; en $5$~:
$\binom43\,0.6^3 \times 0.4 \times 0.6 = 0.2074$~; en $6$~:
$\binom53\,0.6^3 \times 0.4^2 \times 0.6 = 0.2074$~; en $7$~:
$\binom63\,0.6^3 \times 0.4^3 \times 0.6 = 0.1659$. Total~: environ
$0.710$~: une équipe à $60\,\%$ par match remporte $71\,\%$ des
séries --- la série amplifie l'avantage.

\textbf{16.} Finale unique~: $60\,\%$. Meilleur des trois~:
$p^2 + 2p^2 q = 0.36 + 0.288 = 0.648$. L'échelle
$60\,\% \to 65\,\% \to 71\,\%$ se poursuit avec la longueur~:
davantage de matchs moyennent la chance (la loi des grands nombres
en miniature), si bien que les finales longues couronnent le talent
--- ce qui est précisément ce que les ligues vendent.

\textbf{17.} Pièce équilibrée~: $\E = 50$, $\sigma = \sqrt{25} =
5$~; $z = \frac{62 - 50}{5} = 2.4$~: au-delà de la convention des
$2\sigma$ --- la pièce mérite une enquête.

\textbf{18.} $\P(X = 0) = 0.98^{50} \approx 0.364$~;
$\P(X = 1) = 50 \times 0.02 \times 0.98^{49} \approx 0.372$~;
$\P(X = 2) = \binom{50}{2} 0.02^2 \times 0.98^{48} \approx
0.186$. Donc $\P(X \geq 3) \approx 1 - 0.922 = 0.078$, soit environ
$7.8\,\%$ --- au-dessus du seuil de $5\,\%$, ce qui ne suffit pas
(encore) à rejeter l'annonce~; un second lot mauvais changerait
l'histoire.

\textbf{19.} $\P(\text{au moins un gain}) = 1 - 0.999^{1000}
\approx 1 - 0.368 = 0.632$~: mille billets à une chance sur mille
ne donnent pas la certitude mais $63\,\%$. Le $0.368$ récurrent est
$\frac1e$ déguisé --- la constante $e$ fera son entrée officielle
l'an prochain.

\textbf{20.} Reconnaissance~: $n$ fixé, indépendance, $p$ constant
--- alors, et seulement alors, la \omterm{def:g11:prob:rv}{loi} est binomiale. Table~: le
\omterm{prop:g11:binom:pascal}{triangle de Pascal}, ligne $n$. Forme~: la cloche, symétrique pour
$p = \frac12$, décalée sinon. Boussole~: centre $np$, dispersion
$\sqrt{np(1-p)}$ --- les \omterm{pb:g11:stat:1}{cotes z} des décisions. Héritiers~: la
courbe en cloche continue dont les tas s'approchent, et la loi des
grands nombres qui explique pourquoi les grandes planches ne mentent
jamais.
\end{solution}
