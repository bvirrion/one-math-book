\section*{Capítulo \ref{ch:g11:prob} --- Probabilidade e variáveis aleatórias}

\begin{solution}{exo:g11:prob:1}
Modelo uniforme sobre $52$ cartões. Corações: $\frac{13}{52} = \frac14$. Reis:
$\frac{4}{52} = \frac{1}{13}$. Coração ou rei:
$\frac{13 + 4 - 1}{52} = \frac{16}{52} = \frac{4}{13}$ (o rei de
corações são contados uma vez). Nenhum: $1 - \frac{4}{13} = \frac{9}{13}$.
\end{solution}

\begin{solution}{exo:g11:prob:2}
$ S = 6$: os pares $(1,5), (2,4), (3,3), (4,2), (5,1)$, então
$\P(S = 6) = \frac{5}{36}$.

$ S \geq 10$: três pares para $10$, dois para $11$, um para $12$:
$\P(S \geq 10) = \frac{6}{36} = \frac16$.

$ S $ par: somas $2, 4, 6, 8, 10, 12$ ter $1 + 3 + 5 + 5 + 3 + 1 = 18$
pares: \omterm{def:g10:proba:distribution}{probabilidade} $\frac{18}{36} = \frac12$.
\end{solution}

\begin{solution}{exo:g11:prob:3}
As \omterm{def:g10:proba:distribution}{probabilidades} somam $1$: $ p = 1 - 0.3 - 0.2 - 0.4 = 0.1$. Então
\[
\E(X) = 0.3 \times (-1) + 0.2 \times 0 + 0.4 \times 2 + 0.1 \times 5
= -0.3 + 0.8 + 0.5 = 1 .
\]
\end{solution}

\begin{solution}{exo:g11:prob:4}
$\E(Y) = 2 \times 3 - 5 = 1$; $\V(Y) = 2^2 \times 4 = 16$;
$\sigma(Y) = 4$.
\end{solution}

\begin{solution}{exo:g11:prob:5}
Os ganhos são $0$, $2$, $5$ euros com \omterm{def:g10:proba:distribution}{probabilidades} $0.5$, $0.3$,
$0.2$; subtraindo o $2$-participação em euros, $ G $ leva os valores $-2, 0, 3$:
\begin{center}
\begin{tabular}{c|ccc}
$ g $ & $-2$ & $0$ & $3$\\
\hline
$\P(G=g)$ & $0.5$ & $0.3$ & $0.2$
\end{tabular}
\end{center}
$\E(G) = -1 + 0 + 0.6 = -0.4$: o jogador perde $40$ centavos por jogo em
médio --- desfavorável.
\end{solution}

\begin{solution}{exo:g11:prob:6}
Desenhando em ordem: $\P(X = 2) = \frac35 \times \frac24 = \frac{3}{10}$
(vermelho e depois vermelho); $\P(X = 0) = \frac25 \times \frac14 = \frac{1}{10}$
(azul depois azul); por isso
$\P(X = 1) = 1 - \frac{3}{10} - \frac{1}{10} = \frac{6}{10}$. Então
\[
\E(X) = 0 \times \frac1{10} + 1 \times \frac6{10} + 2 \times \frac3{10}
= \frac{12}{10} = 1.2 ,
\]
consistente com a intuição: $\frac35$ dos dois sorteios são vermelhos em \omterm{def:g11:stat:mean}{média}.
\end{solution}

\begin{solution}{exo:g11:prob:7}
$\E(G^2) = 0.5 \times 4 + 0.3 \times 0 + 0.2 \times 9 = 3.8$, então pelo
fórmula de atalho
$\V(G) = 3.8 - (-0.4)^2 = 3.8 - 0.16 = 3.64$ e
$\sigma(G) \approx 1.9$ euros.
\end{solution}

\begin{solution}{exo:g11:prob:8}
O benefício é $80$ menos a reclamação:
\begin{center}
\begin{tabular}{c|ccc}
$ b $ & $80$ & $-920$ & $-4920$\\
\hline
$\P(B=b)$ & $0.94$ & $0.05$ & $0.01$
\end{tabular}
\end{center}
$\E(B) = 75.2 - 46 - 49.2 = -20$: a esse preço a empresa \emph{perde}
$20$ euros por apólice, em \omterm{def:g11:stat:mean}{média}. O prêmio é muito baixo para os riscos
coberto (seria necessário exceder $100$ euros para uma \omterm{def:g11:prob:expectation}{esperança} positiva
benefício).
\end{solution}

\begin{solution}{exo:g11:prob:9}
A pontuação aleatória é $+1$ com \omterm{def:g10:proba:distribution}{probabilidade} $\frac14$ e $x$ com
\omterm{def:g10:proba:distribution}{probabilidade} $\frac34$:
\[
\E = \frac14 + \frac{3x}{4} = 0 \iff x = -\frac13 .
\]
Com a penalidade $-\frac13$, adivinhações aleatórias não trazem nada em \omterm{def:g11:stat:mean}{média}.
\end{solution}

\begin{solution}{exo:g11:prob:10}
Duas moedas: elas combinam (HH ou TT) com \omterm{def:g10:proba:distribution}{probabilidade} $\frac24 \times 2 =
\frac12$. O ganho do jogador é $+1$ ou $-1$ com \omterm{def:g10:proba:distribution}{probabilidades} iguais:
\omterm{def:g11:prob:expectation}{esperança} $0$ para ambos os jogadores --- o jogo é justo.

Três moedas: todas iguais (HHH ou TTT) com \omterm{def:g10:proba:distribution}{probabilidade} $\frac{2}{8} =
\frac14$. O jogador recebe $2$ com \omterm{def:g10:proba:distribution}{probabilidade} $\frac14$ e paga $1$
com \omterm{def:g10:proba:distribution}{probabilidade} $\frac34$:
$\E = \frac24 - \frac34 = -\frac14$. O jogo favorece Ann por $25$ centavos
por rodada; seria justo se ela pagasse $3$ euros por um jogo triplo.
\end{solution}

\begin{solution}{exo:g11:prob:11}
\emph{1.} O organizador recebe $1000 \times 5 = 5000$ euros e paga
fora $2000 + 5 \times 200 + 20 \times 25 = 3500$ euros ao todo: o lucro
é o \emph{constante} $1500$ (sem aleatoriedade quando todos os ingressos forem vendidos).
O ganho de um único jogador $ G $ (prêmio menos o $5$-bilhete em euros):
\begin{center}
\begin{tabular}{c|cccc}
$ g $ & $1995$ & $195$ & $20$ & $-5$\\
\hline\rule{0pt}{11pt}%
$\P(G=g)$ & $\dfrac{1}{1000}$ & $\dfrac{5}{1000}$ & $\dfrac{20}{1000}$
& $\dfrac{974}{1000}$
\end{tabular}
\end{center}

\emph{2.}
\[
\E(G) = \frac{1995 + 5 \times 195 + 20 \times 20 - 974 \times 5}{1000}
= \frac{1995 + 975 + 400 - 4870}{1000} = -1.5 .
\]
Cada jogador espera perder $1.50$ euros; sobre $1000$ jogadores que é
$1500$ euros --- exatamente o lucro do organizador. O saldo dos livros:
o que os jogadores perdem em \omterm{def:g11:stat:mean}{média} é precisamente o que o organizador
coleta.
\end{solution}

\begin{solution}{pb:g11:prob:1}
\textbf{1.} $ G $ leva $-2, -1, 0, 1, 2, 3$, cada um com
\omterm{def:g10:proba:distribution}{probabilidade} $\frac16$: $\E(G) = \frac{-2 - 1 + 0 + 1 + 2 +
3}{6} = \frac12$ euro.

\textbf{2.} $\E(G^2) = \frac{4 + 1 + 0 + 1 + 4 + 9}{6} =
\frac{19}{6}$; $ V(G) = \frac{19}{6} - \frac14 = \frac{35}{12}
\approx 2.92$; $\sigma(G) \approx 1.71$.

\textbf{3.} $\E(2G - 1) = 2 \times \frac12 - 1 = 0$;
$ V(2G - 1) = 4\,V(G) = \frac{35}{3}$.

\textbf{4.} $\E(\text{sum}) = 7$: preço justo $7$ euros. No $8$,
a perda esperada é $1$ euros por jogo.

\textbf{5.} Prêmio justo: $0.05 \times 800 = 40$ euros.
Carregando $60$: lucro esperado $20$ euros por apólice.

\textbf{6.} Divisão $5:3$ recompensa o \emph{passado} --- mas o
o pote pertence a quem ganhar o \emph{futuro}, e B pode
ainda vencer. Dar tudo para A trata uma vitória provável como certa ---
A pequena chance de B vale alguma coisa.

\textbf{7.} Rodadas para jogar: $3$; resultados (A-vitória/B-vitória por
rodada): AAA, AAB, ABA, ABB, BAA, BAB, BBA, BBB. O jogador B leva
a série apenas vencendo os três: sozinho o BBB. A vence o
série em $7$ de $8$ casos: divisão justa
$64 \times \frac78 = 56$ pistolas para A, $8$ para B.

\textbf{8.} Jogar as rodadas extras e inúteis muda
vitória de ninguém: assim que A tiver sua sexta vitória, as rodadas posteriores serão
cerimônia. Mas fixando o horizonte exatamente $3$ rodadas faz
tudo $8$ resultados igualmente prováveis --- necessidades de contagem honesta
histórias do mesmo tamanho, e preenchê-las não custa nada.

\textbf{9.} No $5$--$5$: uma rodada decisiva, $32$ cada. No
$5$--$4$: Um recebe $\frac{64 + 32}{2} = 48$. No $5$--$3$: o
próxima rodada leva a $64$ (A ganha) ou para o $5$--$4$
posição vale a pena $48$: A parte de A $\frac{64 + 48}{2} = 56$ ---
Fermat $56 : 8$, re-derivado andando para trás.

\textbf{10.} Uma necessidade $2$, B precisa $3$: jogar $4$ imaginário
rodadas ($16$ resultados). B leva a série com $3$ ou $4$
B-vitórias: $4 + 1 = 5$ resultados. Dividir:
$ B $: $64 \times \frac{5}{16} = 20$; $ A $: $44$.

\textbf{11.} O valor presente justo de um futuro incerto
a recompensa é sua \omterm{def:g11:prob:expectation}{esperança} --- \omterm{def:g11:stat:mean}{média} ponderada de \omterm{def:g10:proba:distribution}{probabilidade} de
o que pode vir. Seguros (pergunta 5) e finanças (preços
recompensas incertas) são esta sentença transformada em indústrias;
o mesmo acontece com os cassinos, do outro lado da mesa.

\textbf{12.} O pote sempre acaba em algum lugar: as duas ações
são variáveis aleatórias somando à constante $64$, então por
linearidade $\E_A + \E_B = 64$ --- ações justas esgotam o pote,
em cada pontuação.

\textbf{13.} Linearidade: $\E(X + Y) = 3.5 + 3.5 = 7$, sem mesa
--- verdadeiro mesmo para dados colados. \omterm{def:g11:prob:variance}{variância} de um dado:
$\E(X^2) - 3.5^2 = \frac{91}{6} - \frac{49}{4} = \frac{35}{12}$;
para dados independentes
$ V(X + Y) = \frac{35}{12} + \frac{35}{12} = \frac{35}{6}$.

\textbf{14.} Cada convidado fixo recupera seu próprio chapéu com
\omterm{def:g10:proba:distribution}{probabilidade} $\frac1n $ (seu chapéu é um entre $ n $). Resumindo
sobre o $ n $ convidados:
$\E(X) = n \times \frac1n = 1$ --- um convidado sortudo em \omterm{def:g11:stat:mean}{média},
se o partido tem dois chapéus ou dois mil.

\textbf{15.} $ n = 2$: atribuições: ambos certos ($ X = 2$) ou
trocado ($ X = 0$): $\E = 1$. $ n = 3$: as seis tarefas dão
$ X = 3, 1, 1, 1, 0, 0$: $\E = \frac{6}{6} = 1$.

\textbf{16.} O cálculo adicionou o \omterm{def:g11:prob:expectation}{esperança} do $ n $
indicadores individuais (``convidado $ i $ pegaram o chapéu''), e
linearidade acrescenta \omterm{def:g11:prob:expectation}{esperança} \emph{incondicionalmente} ---
dependência destruiria um produto ou um \omterm{def:g11:prob:variance}{variância}, mas nunca um
soma de \omterm{def:g11:prob:expectation}{esperança}. Esse é o superpoder da linearidade: não
perguntas de independência feitas.

\textbf{17.} Ambos os ganhos: $\E = 0.5 \times 10 - 5 = 0$ e
$0.005 \times 1000 - 5 = 0$. \omterm{def:g11:prob:variance}{variância} (do ganho):
R: $\E(G^2) = 0.5 \times 25 + 0.5 \times 25 = 25$;
B: $0.005 \times 995^2 + 0.995 \times 25 \approx 4\,975$:
$\sigma_A = 5$, $\sigma_B \approx 70.5$. As pessoas compram B: o que
eles compram não é \omterm{def:g11:prob:expectation}{esperança} (zero em ambos), mas o
\emph{espalhar} --- uma pequena chance de um direito de mudança de vida
cauda.

\textbf{18.} Primeira coroa ao virar $ n $ tem \omterm{def:g10:proba:distribution}{probabilidade}
$2^{-n}$ e paga $2^n $: cada $ n $ contribui
$2^{-n} \times 2^n = 1$ para a \omterm{def:g11:prob:expectation}{esperança}, o que é portanto
$1 + 1 + 1 + \dots $: infinito. No entanto, nenhuma pessoa sã paga
$1\,000$ jogar uma vez: com \omterm{def:g10:proba:distribution}{probabilidade} $\frac{15}{16}$ o
o pagamento é no \omterm{def:g10:functions:extrema}{máximo} $16$. \omterm{def:g11:prob:expectation}{esperança} resume o longo prazo
\emph{repetível} apostas; um jogo único na vida com um selvagem
cauda não é avaliada por seu \omterm{def:g11:stat:mean}{média}.

\textbf{19.} \omterm{def:g11:prob:expectation}{esperança}: segurar $-20$; fique nu
$-0.01 \times 1000 = -10$. A estratégia nua vence em \omterm{def:g11:stat:mean}{média}
--- para quem consegue absorver a perda e repetir a aposta
muitas vezes. Uma família que não consegue sobreviver $-1\,000$ não é
no negócio de \omterm{def:g11:stat:mean}{média}: vale a pena $10$ extra para excluir o
\omterm{def:g11:prob:variance}{variância}. \omterm{def:g11:stat:mean}{Média} das seguradoras; as famílias compram certeza.

\textbf{20.} Nasci dividindo uma panela entre dois impacientes
jogadores; fortalecido pela linearidade, que soma \omterm{def:g11:prob:expectation}{esperanças} sem
pedindo permissão de dependência; cego ao risco, que \omterm{def:g11:prob:variance}{variância}
--- as loterias, o jogo de Petersburgo, a política da família
--- deve medir em vez disso; e coroado no próximo ano pela lei de
números grandes, o que explica por que as \omterm{def:g11:stat:mean}{médias} sempre,
eventualmente, colete.
\end{solution}
