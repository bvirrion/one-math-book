\section*{Capítulo \ref{ch:g11:deriv} --- Derivadas: um primeiro curso}

\begin{solution}{exo:g11:deriv:1}
Para $f(x) = x^2$ em $a = 2$:
\[
\frac{(2+h)^2 - 4}{h} = \frac{4h + h^2}{h} = 4 + h
\xrightarrow[h\to0]{} 4,
\]
logo $f'(2) = 4$. Para $g(x) = x^2 + 3x$ em $a = 1$: $g(1) = 4$ e
\[
\frac{(1+h)^2 + 3(1+h) - 4}{h} = \frac{5h + h^2}{h} = 5 + h
\xrightarrow[h\to0]{} 5,
\]
logo $g'(1) = 5$.
\end{solution}

\begin{solution}{exo:g11:deriv:2}
$f'(x) = 12x^2 - 4x + 1$.

$g'(x) = \frac{5}{2\sqrt x} - \frac{3}{x^2}$ (em $\intoo{0}{+\infty}$).

$h'(x) = 2(x^2 - 3) + (2x+1)(2x) = 2x^2 - 6 + 4x^2 + 2x
= 6x^2 + 2x - 6$ (regra do produto).
\end{solution}

\begin{solution}{exo:g11:deriv:3}
$f(x) = x^2 - 3x$, $a = 2$: $f(2) = -2$, $f'(x) = 2x - 3$, logo
$f'(2) = 1$; \omterm{def:g11:deriv:tangent}{tangente} $y = -2 + (x - 2) = x - 4$.

$f(x) = \frac1x$, $a = 1$: $f(1) = 1$, $f'(1) = -1$; \omterm{def:g11:deriv:tangent}{tangente}
$y = 1 - (x - 1) = 2 - x$.

$f(x) = \sqrt x$, $a = 4$: $f(4) = 2$,
$f'(4) = \frac{1}{2\sqrt4} = \frac14$; \omterm{def:g11:deriv:tangent}{tangente}
$y = 2 + \frac14(x - 4) = \frac{x}{4} + 1$.
\end{solution}

\begin{solution}{exo:g11:deriv:4}
$f'(x) = \frac{(x-1) - (x+2)}{(x-1)^2} = \frac{-3}{(x-1)^2}$.

$g'(x) = \frac{2x(x^2+1) - x^2 \times 2x}{(x^2+1)^2}
= \frac{2x}{(x^2+1)^2}$.

$h'(x) = -\frac{2x+1}{(x^2+x+1)^2}$ (regra do quociente com $u = 1$).
\end{solution}

\begin{solution}{exo:g11:deriv:5}
$f'(x) = 3x^2 - 12x + 9 = 3(x^2 - 4x + 3) = 3(x-1)(x-3)$: positiva fora
de $\intcc{1}{3}$ e negativa dentro. Assim, $f$ é \omterm{def:g11:func:monotone}{crescente} em
$\intoc{-\infty}{1}$, \omterm{def:g10:functions:variations}{decrescente} em $\intcc{1}{3}$ e \omterm{def:g11:func:monotone}{crescente} em
$\intco{3}{+\infty}$, com \omterm{def:g10:functions:extrema}{máximo} local $f(1) = 1 - 6 + 9 - 2 = 2$ e
\omterm{def:g10:functions:extrema}{mínimo} local $f(3) = 27 - 54 + 27 - 2 = -2$.
\end{solution}

\begin{solution}{exo:g11:deriv:6}
Em $\intoo{-1}{+\infty}$:
\[
f'(x) = \frac{2x(x+1) - (x^2+3)}{(x+1)^2}
= \frac{x^2 + 2x - 3}{(x+1)^2}
= \frac{(x+3)(x-1)}{(x+1)^2}.
\]
Para $x > -1$, tanto $x + 3$ quanto $(x+1)^2$ são positivos, de modo que
$f'(x)$ tem o sinal de $x - 1$: $f$ é \omterm{def:g10:functions:variations}{decrescente} em $\intoc{-1}{1}$ e
\omterm{def:g11:func:monotone}{crescente} em $\intco{1}{+\infty}$, com \omterm{def:g10:functions:extrema}{mínimo}
$f(1) = \frac{4}{2} = 2$.
\end{solution}

\begin{solution}{exo:g11:deriv:7}
$f'(x) = 2x - 4$.

\emph{1.} \omterm{def:g11:deriv:tangent}{Tangente} horizontal: $f'(x) = 0$ em $x = 2$; o ponto é
$(2, f(2)) = (2, 1)$ (o vértice da \omterm{def:g11:quad:parabola}{parábola}).

\emph{2.} Paralela a $y = 2x + 1$ significa \omterm{def:g10:reffunc:affine}{coeficiente angular} $2$:
$2x - 4 = 2$, logo $x = 3$, o que dá o ponto $(3, f(3)) = (3, 2)$.
\end{solution}

\begin{solution}{exo:g11:deriv:8}
Com largura $x$ (dois lados cercados) e comprimento $60 - 2x$:
$A(x) = x(60 - 2x) = 60x - 2x^2$ em $\intoo{0}{30}$. Então
$A'(x) = 60 - 4x$, positiva antes de $x = 15$ e negativa depois: a área é
máxima para $x = 15$, dando um cercado $15 \times 30$ m de área
$450$ m$^2$. A fórmula do vértice
$\alpha = -\frac{60}{2 \times (-2)} = 15$ do \cref{ch:g11:quad} dá a
mesma resposta sem cálculo.
\end{solution}

\begin{solution}{exo:g11:deriv:9}
Seja $f(x) = \frac{x+1}{2} - \sqrt x$ em $\intoo{0}{+\infty}$. Então
\[
f'(x) = \frac12 - \frac{1}{2\sqrt x} = \frac{\sqrt x - 1}{2\sqrt x},
\]
que tem o sinal de $\sqrt x - 1$: negativa em $\intoo{0}{1}$ e positiva
em $\intoo{1}{+\infty}$. Assim, $f$ decresce e depois cresce, com \omterm{def:g10:functions:extrema}{mínimo}
$f(1) = 1 - 1 = 0$. Logo $f(x) \geq 0$ para todo $x > 0$, isto é,
$\sqrt x \leq \frac{x+1}{2}$, com igualdade somente em $x = 1$.
\end{solution}

\begin{solution}{exo:g11:deriv:10}
De $\pi r^2 h = 500$ vem $h = \frac{500}{\pi r^2}$, logo
\[
S(r) = 2\pi r^2 + 2\pi r \cdot \frac{500}{\pi r^2}
= 2\pi r^2 + \frac{1000}{r},
\qquad
S'(r) = 4\pi r - \frac{1000}{r^2}.
\]
$S'(r) = 0$ quando $4\pi r^3 = 1000$, isto é,
$r^3 = \frac{250}{\pi}$, logo $r = \sqrt[3]{250/\pi}$
($\approx 4.3$ cm). Como
$S'(r) = \frac{4\pi r^3 - 1000}{r^2}$ é negativa antes desse valor e
positiva depois, trata-se de um \omterm{def:g10:functions:extrema}{mínimo}.
\end{solution}

\begin{solution}{exo:g11:deriv:11}
A \omterm{def:g11:deriv:tangent}{tangente} a $y = x^2$ no ponto de \omterm{def:g10:coordgeom:system}{abscissa} $a$ é
$y = a^2 + 2a(x - a) = 2ax - a^2$. Ela passa por $P(1, -3)$ quando
$-3 = 2a - a^2$, isto é, $a^2 - 2a - 3 = 0$, cujas raízes são $a = 3$ e
$a = -1$ ($\Delta = 16$). Há, portanto, \emph{duas} \omterm{def:g11:deriv:tangent}{tangentes} por $P$:
\[
y = 6x - 9 \quad (a = 3),
\qquad
y = -2x - 1 \quad (a = -1).
\]
\end{solution}

\begin{solution}{exo:g11:deriv:12}
Pelo \cref{ex:g11:deriv:study}, $f$ cresce até o \omterm{def:g10:functions:extrema}{máximo} local
$f(-1) = 3$, decresce até o \omterm{def:g10:functions:extrema}{mínimo} local $f(1) = -1$ e depois cresce de
novo; para $x$ positivo grande (resp.\ negativo grande em \omterm{def:g10:numbers:abs}{módulo}), $x^3$
domina e $f$ assume valores arbitrariamente grandes (resp.\ pequenos).
Lendo as retas horizontais $y = k$ sobre o \omterm{def:g10:functions:table}{quadro de variação}:
\begin{itemize}
  \item $k > 3$ ou $k < -1$: a reta encontra a curva uma só vez ---
        $1$ solução;
  \item $k = 3$ ou $k = -1$: a reta toca um ponto de virada e cruza um
        outro ramo --- $2$ soluções;
  \item $-1 < k < 3$: a reta cruza os três ramos --- $3$ soluções.
\end{itemize}
\end{solution}

\begin{solution}{pb:g11:deriv:1}
\textbf{1.} $f'(x) = 6x - 5$; $g'(x) = 3x^2 - 12$. Pela
definição: $\frac{(1 + h)^2 - 1}{h} = 2 + h \to 2$: a \omterm{def:g11:deriv:derivative}{derivada}
de $x^2$ em $1$ é $2$.

\textbf{2.} $g(2) = 8 - 24 = -16$ e $g'(2) = 12 - 12 = 0$: a
\omterm{def:g11:deriv:tangent}{tangente} é a reta horizontal $y = -16$. Uma \omterm{def:g11:deriv:tangent}{tangente} horizontal
sinaliza um ponto crítico --- aqui um \omterm{def:g10:functions:extrema}{mínimo} local, como o
\omterm{def:g10:functions:table}{quadro de variação} da questão 5 confirma.

\textbf{3.} \omterm{def:g10:reffunc:affine}{Coeficiente angular} $2a$ em $(a, a^2)$:
$y = a^2 + 2a(x - a) = 2ax - a^2$.

\textbf{4.} Em $x = 0$: $y = -a^2$: a \omterm{def:g11:deriv:tangent}{tangente} corta o eixo à
profundidade $a^2$ abaixo da origem --- exatamente tão
\emph{abaixo} de zero quanto $P$ está acima dele. Receita: para
traçar a \omterm{def:g11:deriv:tangent}{tangente} em um ponto de altura $h$, ligue-o ao ponto
do eixo situado à profundidade $h$ abaixo da origem. (Nenhuma
\omterm{def:g11:deriv:derivative}{derivada} é necessária na prancheta.)

\textbf{5.} $g'(x) = 3(x^2 - 4)$: positiva em
$\intoo{-\infty}{-2}$, negativa em $\intoo{-2}{2}$ e positiva
adiante. $g$ cresce até um \omterm{def:g10:functions:extrema}{máximo} local $g(-2) = 16$, cai até
um \omterm{def:g10:functions:extrema}{mínimo} local $g(2) = -16$ e depois cresce.

\textbf{6.} $F\left(0, \frac14\right)$ e $T(0, -a^2)$:
$FT = \frac14 + a^2$.

\textbf{7.} $FP = a^2 + \frac14 = FT$: isósceles em $F$, logo os
ângulos da base são iguais: $\widehat{FTP} = \widehat{FPT}$.

\textbf{8.} O raio incidente em $P$ é vertical, portanto
paralelo à reta $(TF)$ (o eixo $y$). A \omterm{def:g11:deriv:tangent}{tangente} $(TP)$ corta as
duas paralelas, de modo que o ângulo entre o raio e a \omterm{def:g11:deriv:tangent}{tangente}
é igual a $\widehat{FTP}$ (ângulos alternos) --- que é igual a
$\widehat{FPT}$, o ângulo entre $[PF]$ e a \omterm{def:g11:deriv:tangent}{tangente}. Ângulos
iguais dos dois lados da \omterm{def:g11:deriv:tangent}{tangente} em $P$.

\textbf{9.} Pela lei da reflexão, o raio que chega
verticalmente a $P$ sai pela reta que faz o ângulo igual do
outro lado --- pela questão 8, ao longo de $[PF]$: pelo \omterm{pb:g11:quad:1}{foco},
qualquer que seja $a$. Ao contrário, a luz nascida em $F$ sai
de cada ponto do espelho paralela ao eixo. As antenas coletam,
os faróis iluminam: demonstrado.

\textbf{10.} $a = 1$: \omterm{def:g11:deriv:tangent}{tangente} $y = 2x - 1$, $T(0, -1)$;
$FP = \sqrt{1 + \left(1 - \frac14\right)^2} =
\sqrt{\frac{25}{16}} = \frac54$; $FT = \frac14 + 1 = \frac54$:
isósceles, como prometido.

\textbf{11.} A \omterm{def:g11:deriv:tangent}{tangente} em $x_n$ é
$y = f(x_n) + f'(x_n)(x - x_n)$; ela cruza $y = 0$ onde
$x = x_n - \frac{f(x_n)}{f'(x_n)}$.

\textbf{12.} $x_1 = 1 - \frac{-1}{2} = \frac32$;
$x_2 = \frac32 - \frac{1/4}{3} = \frac{17}{12}$;
$x_3 = \frac{577}{408}$: as \omterm{def:g10:numbers:approx}{aproximações} de $\sqrt2$ de Heron,
isto é, a máquina de \omterm{pb:g10:functions:1}{ponto fixo} do
\cref{pb:g10:functions:1}. Identidade:
$x - \frac{x^2 - 2}{2x} = \frac{2x^2 - x^2 + 2}{2x}
= \frac{x^2 + 2}{2x} = \frac12\left(x + \frac2x\right)$ --- a
receita de Heron \emph{é} o método de Newton aplicado a
$x^2 - 2$, dois milênios antes.

\textbf{13.} $x_1 = 5 - \frac{125 - 100}{75} = \frac{14}{3}
\approx 4.6667$; $x_2 \approx 4.6667 - \frac{1.63}{65.33}
\approx 4.6417$. Calculadora: $\sqrt[3]{100} = 4.6416\ldots$
--- quatro casas decimais corretas em dois passos.

\textbf{14.} $f'(2) = 0$: a \omterm{def:g11:deriv:tangent}{tangente} no ponto de partida é
horizontal (questão 2) e nunca cruza o eixo --- a fórmula
divide por zero. Ressalva: comece Newton longe dos pontos
críticos (e, mais geralmente, perto o bastante da \omterm{def:g11:quad:discriminant}{raiz} que se
procura).

\textbf{15.} A bisseção ganha um dígito binário por passo;
Newton praticamente \emph{dobra} o número de dígitos corretos a
cada passo ($1 \to 3 \to 6$ aqui). Quando cada passo é caro ---
milhões de parâmetros, navegação em tempo real ---, dobrar
vence avançar a passinhos, e é por isso que a reta \omterm{def:g11:deriv:tangent}{tangente}
roda dentro de tanto silício.

\textbf{16.} $d^2 = 900 - w^2$, logo
$S(w) = w(900 - w^2) = 900w - w^3$, para $0 < w < 30$.

\textbf{17.} $S'(w) = 900 - 3w^2 = 0$ em
$w = \sqrt{300} = 10\sqrt3 \approx 17.3$ cm ($S' > 0$ antes e
$< 0$ depois: um \omterm{def:g10:functions:extrema}{máximo}). Então
$d = \sqrt{600} = 10\sqrt6 \approx 24.5$ cm.

\textbf{18.} $\frac{d^2}{w^2} = \frac{600}{300} = 2$, logo
$d = w\sqrt2$: a altura é a largura vezes $\sqrt2$ --- a razão
do carpinteiro (e a construção do diâmetro dividido em terços a
entrega com um compasso, sem álgebra na obra).

\textbf{19.} Viga quadrada: $w = d = \sqrt{450} \approx 21.2$
cm, rigidez $w^3 = 450^{3/2} \approx 9\,546$. Viga ótima:
$S = 10\sqrt3 \times 600 = 6\,000\sqrt3 \approx 10\,392$.
Ganho: cerca de $8.9\,\%$ mais rígida a partir da mesma tora
--- a \omterm{def:g11:deriv:derivative}{derivada} paga a serraria.

\textbf{20.} Geometria: os ângulos iguais da \omterm{def:g11:deriv:tangent}{tangente}
transformaram um paraboloide em coletor de luz. Cálculo
numérico: escorregar pelas \omterm{def:g11:deriv:tangent}{tangentes} resolve \omterm{def:g10:algebra:equation}{equações} dobrando
os dígitos. Otimização: uma \omterm{def:g11:deriv:derivative}{derivada} nula encontrou o retângulo
mais forte da tora. Um objeto, três ofícios --- e a convexidade
do ano 12 dirá, além disso, de que lado de cada \omterm{def:g11:deriv:tangent}{tangente} a
curva se encontra.
\end{solution}
