\section*{Capítulo \ref{ch:g11:stat} --- Estatística descritiva}

\begin{solution}{exo:g11:stat:1}
$n = 8$, soma $71$: \omterm{def:g11:stat:mean}{média} $\bar x = \frac{71}{8} = 8.875$. \omterm{def:g11:stat:median}{Mediana}: \omterm{prop:g10:coordgeom:midpoint}{ponto
médio} entre o $4^{\text{o}}$ e o $5^{\text{o}}$ valores,
$\frac{8 + 9}{2} = 8.5$. Quartis: um quarto de $8$ é $2$, de modo que
$Q_1$ é o $2^{\text{o}}$ valor, $7$; três quartos é $6$, de modo que
$Q_3$ é o $6^{\text{o}}$ valor, $10$. \omterm{def:g11:stat:quartiles}{Amplitude interquartil}:
$10 - 7 = 3$.
\end{solution}

\begin{solution}{exo:g11:stat:2}
Soma $40$, logo $\bar x = 5$. Soma dos quadrados:
$4 + 16 + 16 + 16 + 25 + 25 + 49 + 81 = 232$. Pela fórmula abreviada,
\[
V = \frac{232}{8} - 5^2 = 29 - 25 = 4,
\qquad
\sigma = 2 .
\]
\end{solution}

\begin{solution}{exo:g11:stat:3}
\omterm{def:g11:stat:mean}{Média}:
\[
\bar x = \frac{1 \times 6 + 2 \times 9 + 3 \times 8 + 4 \times 10
+ 5 \times 9 + 6 \times 8}{50}
= \frac{181}{50} = 3.62 .
\]
\omterm{def:g11:stat:median}{Mediana}: as \omterm{def:g10:stats:series}{frequências} acumuladas são $6, 15, 23, 33, \dots$, de modo
que o $25^{\text{o}}$ e o $26^{\text{o}}$ resultados ordenados são ambos
$4$: a \omterm{def:g11:stat:median}{mediana} é $4$.
\end{solution}

\begin{solution}{exo:g11:stat:4}
O total das $15$ notas é $15 \times 11.2 = 168$; com o recém-chegado,
$\frac{168 + 18}{16} = \frac{186}{16} = 11.625$.
\end{solution}

\begin{solution}{exo:g11:stat:5}
$\frac{6 + 8 + x + 12 + 14}{5} = 10$ dá $40 + x = 50$, logo $x = 10$. Os
desvios de $6, 8, 10, 12, 14$ em relação à \omterm{def:g11:stat:mean}{média} $10$ são
$-4, -2, 0, 2, 4$, com quadrados somando $40$: $V = \frac{40}{5} = 8$ e
$\sigma = 2\sqrt2 \approx 2.83$.
\end{solution}

\begin{solution}{exo:g11:stat:6}
Pela \cref{prop:g11:stat:affine} com $a = \frac12$ e $b = 4$: nova \omterm{def:g11:stat:mean}{média}
$\frac{12.4}{2} + 4 = 10.2$; novo \omterm{def:g11:stat:variance}{desvio padrão}
$\frac{2.5}{2} = 1.25$.
\end{solution}

\begin{solution}{exo:g11:stat:7}
A \omterm{def:g11:stat:mean}{média} conjunta é a \omterm{def:g11:stat:mean}{média} \emph{ponderada}
\[
\frac{20 \times 10 + 30 \times 15}{50} = \frac{650}{50} = 13,
\]
e não $12.5$: o grupo B contribui com mais valores e, portanto, puxa a
\omterm{def:g11:stat:mean}{média} conjunta para o seu lado.
\end{solution}

\begin{solution}{exo:g11:stat:8}
As duas linhas estão quase no alvo em \omterm{def:g11:stat:mean}{média} ($1.002$ e $1.001$ kg), mas o
\omterm{def:g11:stat:variance}{desvio padrão} da linha 2 é três vezes maior: seus pacotes variam muito
mais, produzindo bem mais pacotes visivelmente cheios demais ou vazios
demais. A linha 2 deve ser tratada primeiro --- o que precisa de conserto
é a consistência, não apenas a \omterm{def:g11:stat:mean}{média}.
\end{solution}

\begin{solution}{exo:g11:stat:9}
$\bar x = \frac{70}{10} = 7$;
$V = \frac{540}{10} - 7^2 = 54 - 49 = 5$;
$\sigma = \sqrt5 \approx 2.24$.
\end{solution}

\begin{solution}{exo:g11:stat:10}
\emph{1.} Com o $23$: soma $50$, \omterm{def:g11:stat:mean}{média} $5$; \omterm{def:g11:stat:median}{mediana} $=$ \omterm{prop:g10:coordgeom:midpoint}{ponto médio} entre
o $5^{\text{o}}$ e o $6^{\text{o}}$ valores $= 3$. Sem o $23$: soma $27$,
\omterm{def:g11:stat:mean}{média} $3$; \omterm{def:g11:stat:median}{mediana} ($5^{\text{o}}$ de $9$ valores) $= 3$.

\emph{2.} Com o $23$ ($n = 10$): $Q_1$ é o $3^{\text{o}}$ valor ($2$) e
$Q_3$ o $8^{\text{o}}$ ($4$): \omterm{def:g11:stat:quartiles}{amplitude interquartil} $2$. Sem ele
($n = 9$): $Q_1$ é o $3^{\text{o}}$ valor ($2$) e $Q_3$ o $7^{\text{o}}$
($4$): ainda $2$.

\emph{3.} O único valor atípico arrasta a \omterm{def:g11:stat:mean}{média} de $3$ para $5$, mas
deixa a \omterm{def:g11:stat:median}{mediana} e a \omterm{def:g11:stat:quartiles}{amplitude interquartil} inalteradas: essas duas são
\emph{robustas}. Quando um conjunto de dados pode conter valores
aberrantes, informe a \omterm{def:g11:stat:median}{mediana} e a \omterm{def:g11:stat:quartiles}{amplitude interquartil}, e não a \omterm{def:g11:stat:mean}{média} e
o \omterm{def:g11:stat:variance}{desvio padrão}.
\end{solution}

\begin{solution}{pb:g11:stat:1}
\textbf{1.} \omterm{def:g11:stat:mean}{Média} $\bar x = \frac{25}{5} = 5$. Quadrados dos
desvios: $1, 9, 1, 0, 9$: \omterm{def:g11:stat:variance}{variância} $\frac{20}{5} = 4$, logo
$\sigma = 2$.

\textbf{2.} \omterm{def:g11:stat:mean}{Média} dos quadrados:
$\frac{16 + 64 + 36 + 25 + 4}{5} = 29$; menos
$\bar x^2 = 25$: \omterm{def:g11:stat:variance}{variância} $4$. Mesma resposta, metade do
trabalho.

\textbf{3.} \omterm{def:g11:stat:mean}{Média} $= \frac{3 \times 8 + 5 \times 10 + 2 \times
14}{10} = 10.2$; \omterm{def:g11:stat:variance}{variância} $= \frac{3 \times 64 + 5 \times 100 +
2 \times 196}{10} - 10.2^2 = 108.4 - 104.04 = 4.36$;
$\sigma \approx 2.09$.

\textbf{4.} \omterm{def:g11:stat:mean}{Média}: $1.8 \times 20 + 32 = 68\,^\circ$F;
$\sigma$: $1.8 \times 3 = 5.4\,^\circ$F --- o deslocamento
$+32$ não espalha nada.

\textbf{5.} Somando $b$: cada desvio
$x_i + b - (\bar x + b) = x_i - \bar x$ fica inalterado, e a
\omterm{def:g11:stat:variance}{variância} também. Multiplicando por $a$: os desvios ficam
multiplicados por $a$, seus quadrados por $a^2$, a \omterm{def:g11:stat:variance}{variância}
por $a^2$ e $\sigma$ por $\sqrt{a^2} = \abs a$.

\textbf{6.} $z_A = \frac{15 - 12}{2} = 1.5$;
$z_B = \frac{13 - 8}{4} = 1.25$. Alice se destaca mais em
relação ao seu grupo: o $15$ da prova difícil vence o $13$ da
prova fácil.

\textbf{7.} Escores $z$: $-0.5,\ 1.5,\ 0.5,\ 0,\ -1.5$: \omterm{def:g11:stat:mean}{média}
$0$, \omterm{def:g11:stat:variance}{desvio padrão} $1$. Em geral, padronizar subtrai $\bar x$
(o que zera a \omterm{def:g11:stat:mean}{média} sem tocar em $\sigma$) e depois divide por
$\sigma$ (o que torna o novo $\sigma$ igual a $1$): a questão 5
aplicada duas vezes.

\textbf{8.} $z = 2$: $100 + 2 \times 15 = 130$. E
$z(76) = \frac{76 - 100}{15} = -1.6$.

\textbf{9.} O bebê está $2.5$ \omterm{def:g11:stat:variance}{desvios padrão} abaixo do peso
médio \emph{dos bebês da sua idade} --- preocupantemente leve
para a sua faixa. Quilogramas brutos não podem ser comparados
entre idades (todos os bebês ganham peso); os \omterm{pb:g11:stat:1}{escores z} medem
cada criança contra o próprio grupo etário, uma régua só para
todas as idades.

\textbf{10.} \omterm{def:g11:stat:mean}{Média} $5$; \omterm{def:g11:stat:mean}{média} dos quadrados
$\frac{622}{10} = 62.2$, \omterm{def:g11:stat:variance}{variância} $37.2$,
$\sigma \approx 6.10$;
$z(23) = \frac{18}{6.10} \approx 2.95$: pouco abaixo do limiar
de alarme --- o teste mal pisca.

\textbf{11.} Sem o $23$: \omterm{def:g11:stat:mean}{média} $3$, \omterm{def:g11:stat:variance}{variância}
$\frac{93}{9} - 9 \approx 1.33$, $\sigma \approx 1.15$; contra
a régua limpa, o $23$ marca
$z \approx \frac{20}{1.15} \approx 17$: um monstro. O valor
atípico inflou tanto a \omterm{def:g11:stat:mean}{média} quanto $\sigma$, camuflando-se ---
e é por isso que o \cref{exo:g11:stat:10} recomenda resumos
robustos (\omterm{def:g11:stat:median}{mediana}, \omterm{def:g11:stat:quartiles}{amplitude interquartil}) quando há risco de
contaminação.

\textbf{12.} Na soma da \omterm{def:g11:stat:variance}{variância}
$\sigma^2 = \frac1N \sum (x_i - \bar x)^2$, mantenha apenas os
termos com $\abs{x_i - \bar x} \geq k\sigma$: há $pN$ deles,
cada um pelo menos $(k\sigma)^2$, logo
$\sigma^2 \geq \frac{1}{N} \cdot pN \cdot k^2\sigma^2 =
p\,k^2\sigma^2$: divida por $k^2 \sigma^2$ (dispersão não
nula): $p \leq \frac{1}{k^2}$.

\textbf{13.} Além de $2\sigma$: no máximo $\frac14 = 25\,\%$ de
qualquer conjunto de dados que seja; além de $3\sigma$: no
máximo $\frac19 \approx 11\,\%$. Os dados da questão 1: além de
$2\sigma = 4$ da \omterm{def:g11:stat:mean}{média} $5$ significa $\leq 1$ ou $\geq 9$:
nenhum dos cinco valores --- $0\,\% \leq 25\,\%$, como
prometido.

\textbf{14.} Fora de $\intcc{49.7}{50.3}$ significa além de
$3\sigma$: no máximo $\frac19$ da produção,
\emph{qualquer que seja} a forma da distribuição --- Chebyshev
não pede nada além de uma \omterm{def:g11:stat:mean}{média} e um $\sigma$. A produção em
forma de sino se concentra muito melhor (cerca de $0.3\,\%$
além de $3\sigma$): a lei normal, no ano 12.

\textbf{15.} $-19\,\%$ fica $24$ pontos abaixo da \omterm{def:g11:stat:mean}{média} $5$.
Fundo B: $24 = 2\sigma_B$: Chebyshev permite até $\frac14$ de
anos assim ruins. Fundo A: $24 = 12\sigma_A$: no máximo
$\frac{1}{144}$. Mesma \omterm{def:g11:stat:mean}{média}, perigos incomparáveis ---
$\sigma$ é o que as finanças chamam de \emph{risco} (ou
volatilidade).

\textbf{16.} Expanda em torno de $\bar x$:
$(x_i - c)^2 = \left((x_i - \bar x) + (\bar x - c)\right)^2
= (x_i - \bar x)^2 + 2(\bar x - c)(x_i - \bar x) +
(\bar x - c)^2$. Tirando a \omterm{def:g11:stat:mean}{média}: o termo do meio morre (os
desvios em relação à \omterm{def:g11:stat:mean}{média} somam zero), restando
$f(c) = \sigma^2 + (\bar x - c)^2$ --- \omterm{def:g10:functions:extrema}{mínimo} exatamente em
$c = \bar x$, com valor \omterm{def:g10:functions:extrema}{mínimo} $\sigma^2$. A \omterm{def:g11:stat:mean}{média} é o centro
de \omterm{def:g10:functions:extrema}{mínimos} quadrados, como a \omterm{def:g11:stat:median}{mediana} é o centro de \omterm{def:g10:functions:extrema}{mínimos}
desvios absolutos (\cref{pb:g10:stats:1}).

\textbf{17.} Diretamente: os desvios em relação a $6$ são
$-2, 2, 0, -1, -4$, com quadrados $4, 4, 0, 1, 16$:
$f(6) = \frac{25}{5} = 5$. Pela fórmula:
$\sigma^2 + (5 - 6)^2 = 4 + 1 = 5$. Concordam.

\textbf{18.} Os dois têm \omterm{def:g11:stat:mean}{média} $7$; \omterm{def:g11:stat:variance}{variâncias}:
$P$: $\frac{16 + 4 + 0 + 4 + 16}{5} = 8$;
$Q$: $\frac{36 + 0 + 0 + 0 + 36}{5} = 14.4$: dispersões
diferentes por trás de uma só \omterm{def:g11:stat:mean}{média}. Para
$\{7 - a, 7, 7, 7, 7 + a\}$: a \omterm{def:g11:stat:variance}{variância}
$\frac{2a^2}{5} = 8$ dá $a = \sqrt{20} \approx 4.47$: o
conjunto $\{2.53, 7, 7, 7, 11.47\}$ carrega exatamente a \omterm{def:g11:stat:mean}{média}
e a \omterm{def:g11:stat:variance}{variância} de $P$ com um perfil completamente diferente ---
os resumos comprimem, e comprimir perde.

\textbf{19.} $\sigma^2 = 0$ significa que uma soma de quadrados
$(x_i - \bar x)^2$ se anula; quadrados são $\geq 0$, de modo
que uma soma nula obriga cada termo a ser nulo:
$x_i = \bar x$ para todo $i$ --- todos os valores iguais.

\textbf{20.} Os centros dizem onde os dados moram; as
dispersões dizem com que folga; o \omterm{pb:g11:stat:1}{escore z} estende uma única
régua sobre todos os conjuntos, provas, idades e moedas;
Chebyshev converte $\sigma$ em garantia incondicional; e a
questão 18 nos mantém honestos --- nenhum punhado de números
jamais contém a história inteira. No ano que vem, a curva em
sino afia a régua até transformá-la em \omterm{def:g10:proba:distribution}{probabilidades} exatas.
\end{solution}
