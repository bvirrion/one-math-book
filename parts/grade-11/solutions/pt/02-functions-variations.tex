\section*{Capítulo \ref{ch:g11:func} --- Funções e variação}

\begin{solution}{exo:g11:func:1}
$f$: é preciso $3 - x \geq 0$, \omterm{def:g11:func:function}{domínio} $\intoc{-\infty}{3}$.

$g$: é preciso $x^2 - 4 \neq 0$, isto é, $x \neq \pm 2$: \omterm{def:g11:func:function}{domínio}
$\R \setminus \{-2, 2\}$.

$h$: é preciso $x \geq 0$ e $x \neq 5$: \omterm{def:g11:func:function}{domínio}
$\intco{0}{5} \cup \intoo{5}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:2}
$\sqrt{11} < \sqrt{13}$, pois $\sqrt{}$ é \omterm{def:g11:func:monotone}{crescente} e $11 < 13$.

$(-2.1)^2 > (-2.05)^2$, pois $x^2$ é \omterm{def:g10:functions:variations}{decrescente} em
$\intoc{-\infty}{0}$ e $-2.1 < -2.05$.

$\frac{1}{0.9} > \frac{1}{1.1}$, pois $\frac1x$ é \omterm{def:g10:functions:variations}{decrescente} em
$\intoo{0}{+\infty}$ e $0.9 < 1.1$.

$(-1.01)^3 < (-0.99)^3$, pois $x^3$ é \omterm{def:g11:func:monotone}{crescente} e $-1.01 < -0.99$.
\end{solution}

\begin{solution}{exo:g11:func:3}
$f(-x) = 3(-x)^4 - (-x)^2 = 3x^4 - x^2 = f(x)$: \omterm{def:g11:func:parity}{par}.

$g(-x) = -x^3 - x = -g(x)$: \omterm{def:g11:func:parity}{ímpar}.

$h(1) = 2$ e $h(-1) = 1 - 1 = 0$, de modo que $h(-1)$ não é nem $h(1)$
nem $-h(1)$: nem \omterm{def:g11:func:parity}{par} nem \omterm{def:g11:func:parity}{ímpar}.

$k(-x) = \frac{-x}{x^2+1} = -k(x)$: \omterm{def:g11:func:parity}{ímpar}.
\end{solution}

\begin{solution}{exo:g11:func:4}
Leia o \omterm{def:g10:functions:table}{quadro de variação}. Em $\intcc{-3}{1}$, $f$ sobe de $-2$ até $4$ e
cruza o valor $1$ exatamente uma vez; em $\intcc{1}{5}$ ela desce de $4$
até $0$ e cruza $1$ mais uma vez: a \omterm{def:g10:algebra:equation}{equação} $f(x) = 1$ tem $2$ soluções.
O \omterm{def:g10:functions:extrema}{máximo} de $f$ é $f(1) = 4 < 5$, de modo que $f(x) = 5$ não tem
solução.
\end{solution}

\begin{solution}{exo:g11:func:5}
Sejam $2 \leq u < v$. Então
\[
f(v) - f(u) = v^2 - 4v - u^2 + 4u = (v-u)(v+u) - 4(v-u)
= (v - u)(u + v - 4).
\]
Os dois fatores são positivos: $v - u > 0$ e $u + v > 2 + 2 = 4$, pois
$u \geq 2$ e $v > 2$. Logo $f(v) > f(u)$: $f$ é estritamente \omterm{def:g11:func:monotone}{crescente} em
$\intco{2}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:6}
$f = \sqrt{u}$ com $u(x) = 5 - x$ \omterm{def:g10:functions:variations}{decrescente} e não negativa em
$\intoc{-\infty}{5}$: $f$ é \omterm{def:g10:functions:variations}{decrescente} ali
(\cref{prop:g11:func:transforms}).

$g = \frac1u$ com $u(x) = x^2 + 1$ \omterm{def:g11:func:monotone}{crescente} e positiva em
$\intco{0}{+\infty}$: $g$ é \omterm{def:g10:functions:variations}{decrescente} ali.

$h = 3 - 2\sqrt x$: $\sqrt x$ é \omterm{def:g11:func:monotone}{crescente}, multiplicar por $-2 < 0$
inverte, e somar $3$ nada muda: $h$ é \omterm{def:g10:functions:variations}{decrescente} em
$\intco{0}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:7}
\emph{1.} $f$ já está na \omterm{thm:g11:quad:canonical}{forma canônica}: \omterm{def:g10:functions:variations}{decrescente} em
$\intoc{-\infty}{2}$, \omterm{def:g11:func:monotone}{crescente} em $\intco{2}{+\infty}$, \omterm{def:g10:functions:extrema}{mínimo}
$f(2) = -3$.

\emph{2.} $g = \frac{1}{u}$ com $u(x) = (x-2)^2 + 1 = f(x) + 4$. O
deslocamento de $4$ preserva a variação de $f$, e $u \geq 1 > 0$. Tomar o
inverso a inverte: $g$ é \omterm{def:g11:func:monotone}{crescente} em $\intoc{-\infty}{2}$ e \omterm{def:g10:functions:variations}{decrescente}
em $\intco{2}{+\infty}$, com \omterm{def:g10:functions:extrema}{máximo} $g(2) = \frac{1}{1} = 1$.
\end{solution}

\begin{solution}{exo:g11:func:8}
Qualquer curva em forma de morro passando pelos três pontos serve.
Então: $f + 2$ passa por $(0,3)$, $(2,5)$, $(4,3)$, com a mesma variação
(deslocamento vertical); $-f$ passa por $(0,-1)$, $(2,-3)$, $(4,-1)$ com
variação \emph{invertida} (o pico vira um vale); $2f$ passa por $(0,2)$,
$(2,6)$, $(4,2)$, com a mesma variação, mas amplificada na vertical.
\end{solution}

\begin{solution}{exo:g11:func:9}
$2 - \frac{1}{x+1} = \frac{2(x+1) - 1}{x+1} = \frac{2x+1}{x+1} = f(x)$.
Em $\intoo{-1}{+\infty}$, $u(x) = x + 1$ é \omterm{def:g11:func:monotone}{crescente} e positiva, logo
$\frac1u$ é \omterm{def:g10:functions:variations}{decrescente}, $-\frac1u$ é \omterm{def:g11:func:monotone}{crescente}, e somar $2$ nada muda:
$f$ é estritamente \omterm{def:g11:func:monotone}{crescente} em $\intoo{-1}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:10}
\emph{1. Verdadeiro.} Se $u < v$, então
$(f+g)(v) - (f+g)(u) = \bigl(f(v)-f(u)\bigr) + \bigl(g(v)-g(u)\bigr)$ é
uma soma de dois números não negativos.

\emph{2. Falso.} $f(x) = g(x) = x$ são \omterm{def:g11:func:monotone}{crescentes} em $\R$, mas seu
produto $x^2$ não é \omterm{def:g11:func:monotone}{crescente} em $\R$ (decresce em
$\intoc{-\infty}{0}$). (A afirmação passa a ser verdadeira se as duas
funções também forem não negativas.)

\emph{3. Falso.} Mesmo contraexemplo: $f(x) = x$ é \omterm{def:g11:func:monotone}{crescente} em $\R$,
mas $f^2(x) = x^2$ não é.
\end{solution}

\begin{solution}{exo:g11:func:11}
\emph{1.} Para $0 < u < v$:
\[
f(v) - f(u) = (v - u) + \frac1v - \frac1u
= (v - u) + \frac{u - v}{uv}
= (v - u)\left(1 - \frac{1}{uv}\right)
= (v - u)\,\frac{uv - 1}{uv}.
\]

\emph{2.} Sempre $v - u > 0$ e $uv > 0$. Se $0 < u < v \leq 1$, então
$uv < 1$, logo $uv - 1 < 0$ e $f(v) < f(u)$: $f$ é estritamente
\omterm{def:g10:functions:variations}{decrescente} em $\intoc{0}{1}$. Se $1 \leq u < v$, então $uv > 1$ e
$f(v) > f(u)$: $f$ é estritamente \omterm{def:g11:func:monotone}{crescente} em $\intco{1}{+\infty}$.

\emph{3.} $f$ decresce antes de $1$ e cresce depois, de modo que
$f(x) \geq f(1) = 2$ para todo $x > 0$, com igualdade exatamente em
$x = 1$. (Essa desigualdade reaparece sob muitos disfarces; compare com o
\cref{ex:g11:quad:canonical} e com as desigualdades da reta tangente do
\cref{ch:g12:deriv}.)
\end{solution}

\begin{solution}{pb:g11:func:1}
\textbf{1.} $x^4 - 3x^2$: \omterm{def:g11:func:parity}{par} ($(-x)^4 - 3(-x)^2 = x^4 -
3x^2$). $x^3 - x$: \omterm{def:g11:func:parity}{ímpar}. $x^2 + x$: nenhuma das duas
($f(-1) = 0 \neq f(1) = 2$ e $\neq -f(1)$). $\abs x$: \omterm{def:g11:func:parity}{par}.
$\frac1x$: \omterm{def:g11:func:parity}{ímpar}.

\textbf{2.} \omterm{def:g11:func:parity}{Par}: a curva é simétrica em relação ao eixo $y$
(espelho). \omterm{def:g11:func:parity}{Ímpar}: simétrica em relação à origem (meia volta).

\textbf{3.} Se $f$ e $g$ são \omterm{def:g11:func:parity}{ímpares}:
$(fg)(-x) = f(-x)g(-x) = (-f(x))(-g(x)) = (fg)(x)$: \omterm{def:g11:func:parity}{par}.
Tabela: \omterm{def:g11:func:parity}{par} $\times$ \omterm{def:g11:func:parity}{par} $=$ \omterm{def:g11:func:parity}{par}; \omterm{def:g11:func:parity}{par} $\times$ \omterm{def:g11:func:parity}{ímpar} $=$ \omterm{def:g11:func:parity}{ímpar};
\omterm{def:g11:func:parity}{ímpar} $\times$ \omterm{def:g11:func:parity}{ímpar} $=$ \omterm{def:g11:func:parity}{par} --- a regra dos sinais com
paridades.

\textbf{4.} \omterm{def:g11:func:parity}{Ímpar} em $0$: $f(0) = -f(0)$, logo $f(0) = 0$. \omterm{def:g11:func:parity}{Par}:
$f(-3) = f(3) = 7$. \omterm{def:g11:func:parity}{Par} e \omterm{def:g11:func:parity}{ímpar} ao mesmo tempo:
$f(x) = f(-x) = -f(x)$ para todo $x$, logo $2f(x) = 0$: $f$ é a
\omterm{def:g11:func:function}{função} nula.

\textbf{5.} A metade direita ($x \geq 0$) determina
completamente uma \omterm{def:g11:func:parity}{função par}: a metade esquerda é seu espelho.
Nem \omterm{def:g11:func:parity}{par} nem \omterm{def:g11:func:parity}{ímpar}: $x^2 + x$ (questão 1) --- suas duas
simetrias falham em $x = 1$.

\textbf{6.} $E(x) = \frac{(x^3 + x^2 + 1) + (-x^3 + x^2 +
1)}{2} = x^2 + 1$ (\omterm{def:g11:func:parity}{par}), $O(x) = \frac{(x^3 + x^2 + 1) -
(-x^3 + x^2 + 1)}{2} = x^3$ (\omterm{def:g11:func:parity}{ímpar}); sua soma é $f$.

\textbf{7.} $E(-x) = \frac{f(-x) + f(x)}{2} = E(x)$: \omterm{def:g11:func:parity}{par}.
$O(-x) = \frac{f(-x) - f(x)}{2} = -O(x)$: \omterm{def:g11:func:parity}{ímpar}. E
$E(x) + O(x) = f(x)$ por soma direta: toda \omterm{def:g11:func:function}{função} se
decompõe.

\textbf{8.} De $E_1 + O_1 = E_2 + O_2$ vem
$E_1 - E_2 = O_2 - O_1$. O lado esquerdo é \omterm{def:g11:func:parity}{par}, o direito é
\omterm{def:g11:func:parity}{ímpar} --- logo essa \omterm{def:g11:func:function}{função} é as duas coisas e, portanto, nula
(questão 4): $E_1 = E_2$ e $O_1 = O_2$. Uma só decomposição.

\textbf{9.} $E(x) = \frac12\left(\frac{1}{1 - x} + \frac{1}{1 +
x}\right) = \frac12 \cdot \frac{(1 + x) + (1 - x)}{1 - x^2}
= \frac{1}{1 - x^2}$, e
$O(x) = \frac12 \cdot \frac{(1 + x) - (1 - x)}{1 - x^2}
= \frac{x}{1 - x^2}$. Verificação: $E$ é \omterm{def:g11:func:parity}{par}, $O$ é \omterm{def:g11:func:parity}{ímpar} e a
soma delas é $\frac{1 + x}{1 - x^2} =
\frac{1 + x}{(1 - x)(1 + x)} = \frac{1}{1 - x} = f(x)$.

\textbf{10.} Porque cada parte obedece a uma simetria, metade do
trabalho (e metade dos dados) basta para cada uma: resolva de
um lado e espelhe para o outro --- e muitas operações
(integração, séries, análise de sinais) tratam pedaços
simétricos de modo muito mais simples do que funções brutas.

\textbf{11.} $(x-3)^2$: deslocada $3$ para a direita.
$x^2 + 2$: deslocada $2$ para cima. $(x-3)^2 + 2$: as duas
coisas. $-x^2$: virada de cabeça para baixo. $2x^2$: dilatada
verticalmente por $2$ (concha mais estreita).

\textbf{12.} Para $x \geq 2$: $(x - 2) + (x + 2) = 2x$. Para
$-2 \leq x \leq 2$: $(2 - x) + (x + 2) = 4$. Para $x \leq -2$:
$-2x$. A curva desce com \omterm{def:g10:reffunc:affine}{coeficiente angular} $-2$, segue plana
na altura $4$ entre $-2$ e $2$ e depois sobe com \omterm{def:g10:reffunc:affine}{coeficiente
angular} $2$: um vale de fundo plano.

\textbf{13.} Deslocamento vertical: $f(x) + 2$ continua \omterm{def:g11:func:parity}{par}
($f(-x) + 2 = f(x) + 2$). Dilatação vertical: $2f(x)$ continua
\omterm{def:g11:func:parity}{par}. Deslocamento horizontal: $(x - 3)^2$ não é \omterm{def:g11:func:parity}{par}
($f(-1) = 16 \neq 4 = f(1)$): escorregar para o lado destrói o
espelho.

\textbf{14.} No fundo plano o valor é $4 \neq 6$. Para
$x \geq 2$: $2x = 6$, $x = 3$. Para $x \leq -2$: $-2x = 6$,
$x = -3$. Soluções: $\pm 3$.

\textbf{15.} $f(x) = a(x - 2)^2 - 3$ com $f(0) = 4a - 3 = 1$:
$a = 1$, logo $f(x) = (x - 2)^2 - 3$.

\textbf{16.} $g(v) - g(u) = (v - u) + 4\,\frac{u - v}{uv}
= (v - u)\,\frac{uv - 4}{uv}$: negativo para $u < v$ com
$uv < 4$, positivo com $uv > 4$: \omterm{def:g10:functions:variations}{decrescente} em
$\intoc{0}{2}$, \omterm{def:g11:func:monotone}{crescente} em $\intco{2}{+\infty}$; \omterm{def:g10:functions:extrema}{mínimo} em
$x = 2$, de valor $g(2) = 4$.

\textbf{17.} Lados $x$ e $\frac{16}{x}$:
$P(x) = 2\left(x + \frac{16}{x}\right)$, minimizado (a máquina
da questão 16 com $4 \to 16$, ponto de virada $\sqrt{16} = 4$)
em $x = 4$: o \emph{quadrado} $4 \times 4$, de perímetro $16$
--- a resposta de Dido, agora demonstrada para todo comprimento
real de lado.

\textbf{18.} Em $\intco{0}{+\infty}$: $x \mapsto x^2 + 1$
cresce, e $\sqrt{\phantom{x}}$ cresce, de modo que a composta
cresce. E $h(-x) = \sqrt{x^2 + 1} = h(x)$: $h$ é \omterm{def:g11:func:parity}{par}, logo, por
simetria especular, decresce em $\intoc{-\infty}{0}$. \omterm{def:g10:functions:extrema}{Mínimo}
$h(0) = 1$.

\textbf{19.} Quadrado da distância de $(x, \sqrt x)$ a
$(3, 0)$: $d^2(x) = (x - 3)^2 + x = x^2 - 5x + 9$, \omterm{def:g10:functions:extrema}{mínimo} em
$x = \frac52$: ponto mais próximo
$\left(\frac52, \sqrt{\frac52}\right)$, distância
$\sqrt{\frac{11}{4}} = \frac{\sqrt{11}}{2}$. Minimizar $d^2$ é
legítimo porque elevar ao quadrado é estritamente \omterm{def:g11:func:monotone}{crescente} nos
números positivos: $d$ e $d^2$ atingem o fundo no mesmo ponto.

\textbf{20.} Matéria-prima: as curvas de referência, sabidas de
cor. Máquinas: deslocamentos, reflexões e dilatações que montam
\omterm{def:g10:functions:graph}{gráficos} novos a partir dos antigos. Controle de qualidade: a
paridade e a decomposição par--ímpar, que reduzem todo estudo
pela metade. Bancada de medição: \omterm{def:g10:algebra:expand}{fatorar} $f(v) - f(u)$, o que
lê variação e \omterm{def:g10:functions:extrema}{mínimos} diretamente. A ferramenta elétrica do
próximo capítulo: a \emph{derivada}, que os lê de relance.
\end{solution}
