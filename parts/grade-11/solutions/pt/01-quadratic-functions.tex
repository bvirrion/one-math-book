\section*{Capítulo \ref{ch:g11:quad} --- Funções e equações do segundo grau}

\begin{solution}{exo:g11:quad:1}
$x^2-5x+6$: $\Delta = 25 - 24 = 1$, raízes $\frac{5 \pm 1}{2}$, ou seja\ $2$
e $3$.

$2x^2+3x-2$: $\Delta = 9 + 16 = 25$, raízes $\frac{-3 \pm 5}{4}$, ou seja\
$\frac12$ e $-2$.

$ x^2+2x+5$: $\Delta = 4 - 20 = -16 < 0$, nenhuma solução real.

$9x^2-6x+1$: $\Delta = 36 - 36 = 0$, um (duplo) \omterm{def:g11:quad:discriminant}{raiz}
$ x_0 = \frac{6}{18} = \frac13$.
\end{solution}

\begin{solution}{exo:g11:quad:2}
$ f(x) = x^2 - 6x + 11 = (x-3)^2 - 9 + 11 = (x-3)^2 + 2$: vértice $(3, 2)$.

$ g(x) = -2x^2+4x+1 = -2(x^2 - 2x) + 1 = -2\bigl((x-1)^2 - 1\bigr) + 1
= -2(x-1)^2 + 3$: vértice $(1, 3)$.

$ h(x) = 3x^2 + 6x = 3(x^2 + 2x) = 3\bigl((x+1)^2 - 1\bigr)
= 3(x+1)^2 - 3$: vértice $(-1, -3)$.
\end{solution}

\begin{solution}{exo:g11:quad:3}
$ x^2 - 3x - 10$: $\Delta = 9 + 40 = 49$, raízes $\frac{3 \pm 7}{2} = -2$
e $5$; coeficiente líder positivo, então o trinômio é negativo
estritamente entre os raízes: o conjunto solução é $\intoo{-2}{5}$.

$2x^2 + x + 3$: $\Delta = 1 - 24 = -23 < 0$ e $ a = 2 > 0$: sempre
positivo, o conjunto solução é $\R $.

$-x^2 + 6x - 9 = -(x-3)^2$: isso é $\geq 0$ somente onde $(x-3)^2 \leq 0$,
isso é em $ x = 3$. O conjunto solução é $\{3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:4}
$ x^2 - 7x + 6$: $1 - 7 + 6 = 0$, então $1$ é um \omterm{def:g11:quad:discriminant}{raiz}; o produto do
raízes é $6$, então o outro \omterm{def:g11:quad:discriminant}{raiz} é $6$.

$2x^2 + 5x - 7$: $2 + 5 - 7 = 0$, então $1$ é um \omterm{def:g11:quad:discriminant}{raiz}; o produto é
$\frac{c}{a} = -\frac72$, então o outro \omterm{def:g11:quad:discriminant}{raiz} é $-\frac72$.

$ x^2 + 4x - 5$: $1 + 4 - 5 = 0$, então $1$ é um \omterm{def:g11:quad:discriminant}{raiz}; o produto é $-5$, então
o outro \omterm{def:g11:quad:discriminant}{raiz} é $-5$.
\end{solution}

\begin{solution}{exo:g11:quad:5}
Os dois números resolvem $ x^2 - 14x + 45 = 0$ (\cref{prop:g11:quad:vieta});
$\Delta = 196 - 180 = 16$, raízes $\frac{14 \pm 4}{2} = 9$ e $5$.

Para o retângulo: metade do perímetro dá comprimento $+$ largura $= 17$ e
a área dá seu produto $60$. Eles resolvem $ x^2 - 17x + 60 = 0$;
$\Delta = 289 - 240 = 49$, raízes $\frac{17 \pm 7}{2} = 12$ e $5$. O
retângulo é $12 \times 5$.
\end{solution}

\begin{solution}{exo:g11:quad:6}
$\Delta = (m-2)^2 - 4$. Exatamente uma solução quando $\Delta = 0$:
$(m-2)^2 = 4$, então $ m - 2 = \pm 2$, ou seja\ $ m = 0$ ou $ m = 4$. Nenhuma solução
quando $\Delta < 0$: $(m-2)^2 < 4$, ou seja\ $-2 < m - 2 < 2$, aquilo é
$0 < m < 4$.
\end{solution}

\begin{solution}{exo:g11:quad:7}
O vértice está em $ t = -\frac{20}{2 \times (-5)} = 2$, e
$ h(2) = -20 + 40 + 1 = 21$: altura máxima $21$ m, alcançado depois $2$
segundos.

$ h(t) \geq 16$ significa $-5t^2 + 20t - 15 \geq 0$, ou seja,\ (dividindo por $-5$,
que inverte a desigualdade) $ t^2 - 4t + 3 \leq 0$. O raízes de
$ t^2-4t+3$ são $1$ e $3$ ($1 - 4 + 3 = 0$, produto $3$), então
$ t^2 - 4t + 3 \leq 0$ exatamente em $\intcc{1}{3}$: a bola está pelo menos
$16$ estou alto entre os tempos $ t = 1$ areia $ t = 3$ S.
\end{solution}

\begin{solution}{exo:g11:quad:8}
Com $ X = x^2$: $ X^2 - 13X + 36 = 0$, $\Delta = 169 - 144 = 25$, então
$ X = \frac{13 \pm 5}{2} = 9$ ou $4$. Voltar para $x$: $ x^2 = 9$ dá
$ x = \pm 3$, e $ x^2 = 4$ dá $ x = \pm 2$. O conjunto solução é
$\{-3, -2, 2, 3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:9}
A condição é $ x^2 < x + 2$, ou seja\ $ x^2 - x - 2 < 0$. Desde
$ x^2 - x - 2 = (x+1)(x-2)$ (raízes $-1$ e $2$), a desigualdade é válida
estritamente entre os raízes: $ x \in \intoo{-1}{2}$. Num esboço, o
\omterm{def:g11:quad:parabola}{parábola} $ y = x^2$ mergulha abaixo da linha $ y = x+2$ exatamente entre os dois
\omterm{def:g10:numbers:interunion}{interseção} pontos $(-1, 1)$ e $(2, 4)$.
\end{solution}

\begin{solution}{exo:g11:quad:10}
Seja $ x > 0$ satisfazer $ x + \frac1x = \frac{13}{6}$. Multiplicando por $6x > 0$:
$6x^2 + 6 = 13x $, ou seja\ $6x^2 - 13x + 6 = 0$. Então
$\Delta = 169 - 144 = 25$ e $ x = \frac{13 \pm 5}{12}$, dando
$ x = \frac32$ ou $ x = \frac23$. Ambos são positivos e são recíprocos de
entre si, como esperado: o número é $\frac32$ (ou $\frac23$).
\end{solution}

\begin{solution}{exo:g11:quad:11}
\emph{1.} Interseções correspondem a soluções de $ x^2 = mx - 1$, ou seja\
$ x^2 - mx + 1 = 0$, cujo \omterm{def:g11:quad:discriminant}{discriminante} é $\Delta = m^2 - 4$. Dois pontos
quando $\abs{m} > 2$; exatamente um quando $ m = \pm 2$; nenhum quando $-2 < m < 2$.

\emph{2.} Para $ m = 2$: $ x^2 - 2x + 1 = (x-1)^2 = 0$, então o contato único
ponto é $(1, 1)$. Para $ m = -2$: $(x+1)^2 = 0$, ponto de contato $(-1, 1)$.
Em ambos os casos a linha atende ao \omterm{def:g11:quad:parabola}{parábola} em um único ponto sem
cruzando-o --- é a tangente ali, e de fato é coeficiente angular $ m = \pm 2$
é igual ao valor em $ x = \pm 1$ da derivada $2x $ de $ x^2$
(\cref{ch:g11:deriv}).
\end{solution}

\begin{solution}{pb:g11:quad:1}
\textbf{1.} $ x^2 - 5x + 6$: $\Delta = 1$, raízes $2$ e $3$.
$2x^2 + x - 6$: $\Delta = 1 + 48 = 49$, raízes
$\frac{-1 \pm 7}{4}$: $\frac32$ e $-2$. $ x^2 + x + 1$:
$\Delta = -3 < 0$: nenhuma solução real.

\textbf{2.} $\Delta = 16 + 48 = 64$: raízes $3$ e $-1$, então
$2x^2 - 4x - 6 = 2(x - 3)(x + 1)$; \omterm{thm:g11:quad:canonical}{forma canônica}
$2(x - 1)^2 - 8$. A forma fatorada mostra o raízes, o
\omterm{thm:g11:quad:canonical}{forma canônica} o vértice $(1, -8)$, a forma expandida do
$ y $-interceptar $-6$.

\textbf{3.} raízes $1$ e $3$, coeficiente líder negativo:
o trinômio é $\geq 0$ \emph{entre} o raízes:
$ x \in \intcc{1}{3}$.

\textbf{4.} Os números são os raízes de $ x^2 - 7x + 12 = 0$:
$3$ e $4$. Se $2$ é um \omterm{def:g11:quad:discriminant}{raiz} de $ x^2 - 5x + c $, o outro \omterm{def:g11:quad:discriminant}{raiz}
$ r $ satisfaz $2 + r = 5$ (soma), então $ r = 3$, e
$ c = 2 \times 3 = 6$ (produto): Viète lê.

\textbf{5.} $\Delta = m^2 - 36$: dobro \omterm{def:g11:quad:discriminant}{raiz} para $ m = \pm 6$
(\omterm{def:g11:quad:discriminant}{raiz} $\mp 3$); não \omterm{def:g11:quad:discriminant}{raiz} para $-6 < m < 6$; dois raízes para
$\abs m > 6$.

\textbf{6.} $ PF^2 = x^2 + \left(x^2 - \frac14\right)^2
= x^2 + x^4 - \frac{x^2}{2} + \frac{1}{16}
= x^4 + \frac{x^2}{2} + \frac{1}{16}
= \left(x^2 + \frac14\right)^2$. Por isso
$ PF = x^2 + \frac14$ (ambos os fatores positivos).

\textbf{7.} A distância de $ P(x, x^2)$ para a horizontal
linha $ y = -\frac14$ é $ x^2 + \frac14$ --- exatamente $ PF $. Cada
ponto do \omterm{def:g11:quad:parabola}{parábola} é equidistante do \omterm{pb:g11:quad:1}{foco} e o
\omterm{pb:g11:quad:1}{diretora}.

\textbf{8.} $ PF^2 = x^2 + \left(y - \frac14\right)^2$ e
$\operatorname{dist}^2 = \left(y + \frac14\right)^2$. Igualdade
dá $ x^2 + y^2 - \frac y2 + \frac{1}{16} = y^2 + \frac y2 +
\frac1{16}$, então $ x^2 = y $: o lugar geométrico é exatamente o \omterm{def:g11:quad:parabola}{parábola}.
Um ponto e uma linha geram a curva, assim como um ponto e uma linha
distância gerou o círculo.

\textbf{9.} Para $ y = \frac{x^2}{4f}$, a mesma expansão com
$ F(0, f)$ dá $ PF^2 = x^2 + \left(\frac{x^2}{4f} - f\right)^2
= \left(\frac{x^2}{4f} + f\right)^2$: distância até $ F $ é igual
distância para $ y = -f $. Prato: $9 = \frac{30^2}{4f}$ dá
$4f = 100$, $ f = 25$ cm: o receptor fica pendurado um quarto de metro
acima do vértice --- fora do $9$ tigela cm, como pratos reais
mostrar.

\textbf{10.} Uma frente de onda que chega paralelamente ao eixo tem
percorreu distâncias iguais para chegar ao diretoranível;
equidistância converte ``distância igual ao \omterm{pb:g11:quad:1}{diretora}'' em
``distância igual para $ F $'': todos os sinais refletidos chegam em
$ F $ \emph{no passo}, reforçando --- é onde ouvir.
A prova honesta precisa da tangente coeficiente angular em cada ponto ---
a derivada de \cref{ch:g11:deriv}, vislumbrado em
\cref{exo:g11:quad:11}.

\textbf{11.} $ y = x\left(1 - \frac{x}{20}\right)$: pousa em
$ x = 20$ m. Vértice intermediário: $ x = 10$, altura
$ y(10) = 10 - 5 = 5$ m.

\textbf{12.} $ y(4) = 4 - \frac{16}{20} = 3.2$ m: limpa $3$ eu
com espaço, bate em um $4$ minha parede.

\textbf{13.} $ y(10) = 10 - \frac{100}{40} = 7.5$ eu $> 7$: o
caminhão passa em $10$ estou fora do centro. $ y(15) = 10 - \frac{225}{40}
= 4.375$ m: não em $15$ m.

\textbf{14.} $ R(p) = -2p^2 + 120p $: vértice em
$ p = \frac{120}{4} = 30$ euros, receita
$ R(30) = 30 \times 60 = 1\,800$ euros.

\textbf{15.} $ y(320) = 143 \times \left(\frac{320}{640}\right)^2
= \frac{143}{4} \approx 36$ m acima do ponto mais baixo.

\textbf{16.} $ x^2 = mx - 1$ dá $ x^2 - mx + 1 = 0$,
$\Delta = m^2 - 4$: dois pontos para $\abs m > 2$, nenhum para
$\abs m < 2$, tangência para $ m = \pm 2$ com contato
$ x = \frac m2 = \pm 1$: tangentes $ y = 2x - 1$ no $(1, 1)$ e
$ y = -2x - 1$ no $(-1, 1)$.

\textbf{17.} $ x^2 = mx + 1$ dá $\Delta = m^2 + 4 > 0$ para
cada $ m $: dois cruzamentos sempre. De um ponto interior,
cada linha cruza a curva duas vezes --- apenas as tangentes são desenhadas
de fora, exatamente como nos círculos.

\textbf{18.} $ y = \frac14$ conhece $ y = x^2$ no $ x = \pm\frac12$:
comprimento do acorde $1$. Para $ y = \frac{x^2}{4f}$: $ y = f $ dá
$ x^2 = 4f^2$, $ x = \pm 2f $: comprimento $4f $ --- a largura focal é
quatro vezes a distância focal (os designers de pratos chamam isso de
lato reto).

\textbf{19.} Um ponto de $ y = x^2$ é $(a, a^2)$; isso é \omterm{def:g10:functions:function}{imagem} é
$(X, Y) = (ka, ka^2)$, e $\frac{X^2}{k} = \frac{k^2a^2}{k} =
ka^2 = Y $: o \omterm{def:g10:functions:function}{imagem} curva é exatamente $ y = \frac{x^2}{k}$.
Escolhendo $ k = 4f$ atinge todos \omterm{def:g11:quad:parabola}{parábola}: são todos
ampliações de uma curva mestre. (Os círculos são todos semelhantes
também --- mas as elipses não são: o squash é importante, o zoom não.)

\textbf{20.} Um equação com três fantasias legíveis; o
lugar geométrico de equidistância de um ponto e de uma reta; o refletor
que reúne raios paralelos em seu \omterm{pb:g11:quad:1}{foco} e o cabo que
carrega um baralho uniforme; a trajetória de cada bola lançada; e,
até a ampliação, uma única curva --- uma \omterm{def:g11:quad:parabola}{parábola},
em todos os lugares.
\end{solution}
