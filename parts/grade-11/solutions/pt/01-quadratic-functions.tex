\section*{Capítulo \ref{ch:g11:quad} --- Funções e equações do segundo grau}

\begin{solution}{exo:g11:quad:1}
$x^2-5x+6$: $\Delta = 25 - 24 = 1$, raízes $\frac{5 \pm 1}{2}$, isto é,
$2$ e $3$.

$2x^2+3x-2$: $\Delta = 9 + 16 = 25$, raízes $\frac{-3 \pm 5}{4}$, isto
é, $\frac12$ e $-2$.

$x^2+2x+5$: $\Delta = 4 - 20 = -16 < 0$, nenhuma solução real.

$9x^2-6x+1$: $\Delta = 36 - 36 = 0$, uma \omterm{def:g11:quad:discriminant}{raiz} (dupla)
$x_0 = \frac{6}{18} = \frac13$.
\end{solution}

\begin{solution}{exo:g11:quad:2}
$f(x) = x^2 - 6x + 11 = (x-3)^2 - 9 + 11 = (x-3)^2 + 2$: vértice
$(3, 2)$.

$g(x) = -2x^2+4x+1 = -2(x^2 - 2x) + 1 = -2\bigl((x-1)^2 - 1\bigr) + 1
= -2(x-1)^2 + 3$: vértice $(1, 3)$.

$h(x) = 3x^2 + 6x = 3(x^2 + 2x) = 3\bigl((x+1)^2 - 1\bigr)
= 3(x+1)^2 - 3$: vértice $(-1, -3)$.
\end{solution}

\begin{solution}{exo:g11:quad:3}
$x^2 - 3x - 10$: $\Delta = 9 + 40 = 49$, raízes
$\frac{3 \pm 7}{2} = -2$ e $5$; coeficiente dominante positivo, de modo
que o trinômio é negativo estritamente entre as raízes: o conjunto
solução é $\intoo{-2}{5}$.

$2x^2 + x + 3$: $\Delta = 1 - 24 = -23 < 0$ e $a = 2 > 0$: sempre
positivo, e o conjunto solução é $\R$.

$-x^2 + 6x - 9 = -(x-3)^2$: é $\geq 0$ apenas onde
$(x-3)^2 \leq 0$, isto é, em $x = 3$. O conjunto solução é $\{3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:4}
$x^2 - 7x + 6$: $1 - 7 + 6 = 0$, logo $1$ é \omterm{def:g11:quad:discriminant}{raiz}; o produto das raízes é
$6$, logo a outra \omterm{def:g11:quad:discriminant}{raiz} é $6$.

$2x^2 + 5x - 7$: $2 + 5 - 7 = 0$, logo $1$ é \omterm{def:g11:quad:discriminant}{raiz}; o produto é
$\frac{c}{a} = -\frac72$, logo a outra \omterm{def:g11:quad:discriminant}{raiz} é $-\frac72$.

$x^2 + 4x - 5$: $1 + 4 - 5 = 0$, logo $1$ é \omterm{def:g11:quad:discriminant}{raiz}; o produto é $-5$, logo
a outra \omterm{def:g11:quad:discriminant}{raiz} é $-5$.
\end{solution}

\begin{solution}{exo:g11:quad:5}
Os dois números resolvem $x^2 - 14x + 45 = 0$
(\cref{prop:g11:quad:vieta}); $\Delta = 196 - 180 = 16$, raízes
$\frac{14 \pm 4}{2} = 9$ e $5$.

Para o retângulo: metade do perímetro dá comprimento $+$ largura $= 17$,
e a área dá o produto deles, $60$. Eles resolvem $x^2 - 17x + 60 = 0$;
$\Delta = 289 - 240 = 49$, raízes $\frac{17 \pm 7}{2} = 12$ e $5$. O
retângulo é $12 \times 5$.
\end{solution}

\begin{solution}{exo:g11:quad:6}
$\Delta = (m-2)^2 - 4$. Exatamente uma solução quando $\Delta = 0$:
$(m-2)^2 = 4$, logo $m - 2 = \pm 2$, isto é, $m = 0$ ou $m = 4$. Nenhuma
solução quando $\Delta < 0$: $(m-2)^2 < 4$, isto é, $-2 < m - 2 < 2$, ou
seja, $0 < m < 4$.
\end{solution}

\begin{solution}{exo:g11:quad:7}
O vértice está em $t = -\frac{20}{2 \times (-5)} = 2$, e
$h(2) = -20 + 40 + 1 = 21$: altura máxima $21$ m, atingida após $2$
segundos.

$h(t) \geq 16$ significa $-5t^2 + 20t - 15 \geq 0$, isto é (dividindo
por $-5$, o que inverte a desigualdade), $t^2 - 4t + 3 \leq 0$. As
raízes de $t^2-4t+3$ são $1$ e $3$ ($1 - 4 + 3 = 0$, produto $3$), de
modo que $t^2 - 4t + 3 \leq 0$ exatamente em $\intcc{1}{3}$: a bola fica
a pelo menos $16$ m de altura entre os instantes $t = 1$ s e $t = 3$ s.
\end{solution}

\begin{solution}{exo:g11:quad:8}
Com $X = x^2$: $X^2 - 13X + 36 = 0$, $\Delta = 169 - 144 = 25$, logo
$X = \frac{13 \pm 5}{2} = 9$ ou $4$. De volta a $x$: $x^2 = 9$ dá
$x = \pm 3$, e $x^2 = 4$ dá $x = \pm 2$. O conjunto solução é
$\{-3, -2, 2, 3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:9}
A condição é $x^2 < x + 2$, isto é, $x^2 - x - 2 < 0$. Como
$x^2 - x - 2 = (x+1)(x-2)$ (raízes $-1$ e $2$), a desigualdade vale
estritamente entre as raízes: $x \in \intoo{-1}{2}$. Num esboço, a
\omterm{def:g11:quad:parabola}{parábola} $y = x^2$ mergulha abaixo da reta $y = x+2$ exatamente entre
seus dois pontos de \omterm{def:g10:numbers:interunion}{interseção} $(-1, 1)$ e $(2, 4)$.
\end{solution}

\begin{solution}{exo:g11:quad:10}
Seja $x > 0$ satisfazendo $x + \frac1x = \frac{13}{6}$. Multiplicando por
$6x > 0$: $6x^2 + 6 = 13x$, isto é, $6x^2 - 13x + 6 = 0$. Então
$\Delta = 169 - 144 = 25$ e $x = \frac{13 \pm 5}{12}$, o que dá
$x = \frac32$ ou $x = \frac23$. Os dois são positivos e inversos um do
outro, como se esperava: o número é $\frac32$ (ou $\frac23$).
\end{solution}

\begin{solution}{exo:g11:quad:11}
\emph{1.} As \omterm{def:g10:numbers:interunion}{interseções} correspondem às soluções de $x^2 = mx - 1$,
isto é, $x^2 - mx + 1 = 0$, cujo \omterm{def:g11:quad:discriminant}{discriminante} é $\Delta = m^2 - 4$.
Dois pontos quando $\abs{m} > 2$; exatamente um quando $m = \pm 2$;
nenhum quando $-2 < m < 2$.

\emph{2.} Para $m = 2$: $x^2 - 2x + 1 = (x-1)^2 = 0$, de modo que o
único ponto de contato é $(1, 1)$. Para $m = -2$: $(x+1)^2 = 0$, ponto
de contato $(-1, 1)$. Nos dois casos a reta encontra a \omterm{def:g11:quad:parabola}{parábola} em um
único ponto sem atravessá-la --- é a tangente ali, e de fato seu
\omterm{def:g10:reffunc:affine}{coeficiente angular} $m = \pm 2$ é igual ao valor em $x = \pm 1$ da
derivada $2x$ de $x^2$ (\cref{ch:g11:deriv}).
\end{solution}

\begin{solution}{pb:g11:quad:1}
\textbf{1.} $x^2 - 5x + 6$: $\Delta = 1$, raízes $2$ e $3$.
$2x^2 + x - 6$: $\Delta = 1 + 48 = 49$, raízes
$\frac{-1 \pm 7}{4}$: $\frac32$ e $-2$. $x^2 + x + 1$:
$\Delta = -3 < 0$: nenhuma solução real.

\textbf{2.} $\Delta = 16 + 48 = 64$: raízes $3$ e $-1$, logo
$2x^2 - 4x - 6 = 2(x - 3)(x + 1)$; \omterm{thm:g11:quad:canonical}{forma canônica}
$2(x - 1)^2 - 8$. A forma fatorada mostra as raízes, a \omterm{thm:g11:quad:canonical}{forma
canônica} o vértice $(1, -8)$ e a forma expandida a \omterm{def:g10:numbers:interunion}{interseção}
$-6$ com o eixo $y$.

\textbf{3.} Raízes $1$ e $3$, coeficiente dominante negativo: o
trinômio é $\geq 0$ \emph{entre} as raízes:
$x \in \intcc{1}{3}$.

\textbf{4.} Os números são as raízes de $x^2 - 7x + 12 = 0$:
$3$ e $4$. Se $2$ é \omterm{def:g11:quad:discriminant}{raiz} de $x^2 - 5x + c$, a outra \omterm{def:g11:quad:discriminant}{raiz} $r$
satisfaz $2 + r = 5$ (soma), logo $r = 3$, e
$c = 2 \times 3 = 6$ (produto): Viète lê tudo de imediato.

\textbf{5.} $\Delta = m^2 - 36$: \omterm{def:g11:quad:discriminant}{raiz} dupla para $m = \pm 6$
(\omterm{def:g11:quad:discriminant}{raiz} $\mp 3$); nenhuma \omterm{def:g11:quad:discriminant}{raiz} para $-6 < m < 6$; duas raízes
para $\abs m > 6$.

\textbf{6.} $PF^2 = x^2 + \left(x^2 - \frac14\right)^2
= x^2 + x^4 - \frac{x^2}{2} + \frac{1}{16}
= x^4 + \frac{x^2}{2} + \frac{1}{16}
= \left(x^2 + \frac14\right)^2$. Portanto
$PF = x^2 + \frac14$ (os dois fatores são positivos).

\textbf{7.} A distância de $P(x, x^2)$ à reta horizontal
$y = -\frac14$ é $x^2 + \frac14$ --- exatamente $PF$. Todo
ponto da \omterm{def:g11:quad:parabola}{parábola} é equidistante do \omterm{pb:g11:quad:1}{foco} e da \omterm{pb:g11:quad:1}{diretriz}.

\textbf{8.} $PF^2 = x^2 + \left(y - \frac14\right)^2$ e
$\operatorname{dist}^2 = \left(y + \frac14\right)^2$. A
igualdade dá $x^2 + y^2 - \frac y2 + \frac{1}{16} = y^2 +
\frac y2 + \frac1{16}$, logo $x^2 = y$: o lugar geométrico é
exatamente a \omterm{def:g11:quad:parabola}{parábola}. Um ponto e uma reta geram a curva, como
um ponto e uma distância geraram o círculo.

\textbf{9.} Para $y = \frac{x^2}{4f}$, a mesma expansão com
$F(0, f)$ dá $PF^2 = x^2 + \left(\frac{x^2}{4f} - f\right)^2
= \left(\frac{x^2}{4f} + f\right)^2$: a distância a $F$ é igual
à distância a $y = -f$. Antena: $9 = \frac{30^2}{4f}$ dá
$4f = 100$, $f = 25$ cm: o receptor fica um quarto de metro
acima do vértice --- fora da concha de $9$ cm, como se vê nas
antenas reais.

\textbf{10.} Uma frente de onda que chega paralela ao eixo
percorreu distâncias iguais até atingir o nível da \omterm{pb:g11:quad:1}{diretriz}; a
equidistância converte ``mesma distância à \omterm{pb:g11:quad:1}{diretriz}'' em ``mesma
distância a $F$'': todos os sinais refletidos chegam a $F$
\emph{em fase}, reforçando-se --- é ali que se deve escutar. A
demonstração honesta precisa do \omterm{def:g10:reffunc:affine}{coeficiente angular} da tangente
em cada ponto --- a derivada do \cref{ch:g11:deriv},
entrevista no \cref{exo:g11:quad:11}.

\textbf{11.} $y = x\left(1 - \frac{x}{20}\right)$: cai em
$x = 20$ m. Vértice na metade: $x = 10$, altura
$y(10) = 10 - 5 = 5$ m.

\textbf{12.} $y(4) = 4 - \frac{16}{20} = 3.2$ m: passa sobre um
muro de $3$ m com folga e bate num muro de $4$ m.

\textbf{13.} $y(10) = 10 - \frac{100}{40} = 7.5$ m $> 7$: o
caminhão passa a $10$ m do centro. $y(15) = 10 -
\frac{225}{40} = 4.375$ m: não passa a $15$ m.

\textbf{14.} $R(p) = -2p^2 + 120p$: vértice em
$p = \frac{120}{4} = 30$ euros, receita
$R(30) = 30 \times 60 = 1\,800$ euros.

\textbf{15.} $y(320) = 143 \times \left(\frac{320}{640}\right)^2
= \frac{143}{4} \approx 36$ m acima do ponto mais baixo.

\textbf{16.} $x^2 = mx - 1$ dá $x^2 - mx + 1 = 0$,
$\Delta = m^2 - 4$: dois pontos para $\abs m > 2$, nenhum para
$\abs m < 2$, tangência para $m = \pm 2$ com contato em
$x = \frac m2 = \pm 1$: tangentes $y = 2x - 1$ em $(1, 1)$ e
$y = -2x - 1$ em $(-1, 1)$.

\textbf{17.} $x^2 = mx + 1$ dá $\Delta = m^2 + 4 > 0$ para todo
$m$: sempre duas \omterm{def:g10:numbers:interunion}{interseções}. De um ponto interior, toda reta
atravessa a curva duas vezes --- tangentes só se traçam de
fora, exatamente como nos círculos.

\textbf{18.} $y = \frac14$ encontra $y = x^2$ em
$x = \pm\frac12$: corda de comprimento $1$. Para
$y = \frac{x^2}{4f}$: $y = f$ dá $x^2 = 4f^2$, $x = \pm 2f$:
comprimento $4f$ --- a largura focal é quatro vezes a distância
focal (os projetistas de antenas a chamam de latus rectum).

\textbf{19.} Um ponto de $y = x^2$ é $(a, a^2)$; sua \omterm{def:g10:functions:function}{imagem} é
$(X, Y) = (ka, ka^2)$, e $\frac{X^2}{k} = \frac{k^2a^2}{k} =
ka^2 = Y$: a curva \omterm{def:g10:functions:function}{imagem} é exatamente $y = \frac{x^2}{k}$.
Escolhendo $k = 4f$ alcança-se toda \omterm{def:g11:quad:parabola}{parábola}: todas são
ampliações de uma única curva-mestra. (Os círculos também são
todos semelhantes --- mas as elipses não: o achatamento importa,
a ampliação não.)

\textbf{20.} Uma \omterm{def:g10:algebra:equation}{equação} com três trajes legíveis; o lugar
geométrico da equidistância entre um ponto e uma reta; o
refletor que reúne no \omterm{pb:g11:quad:1}{foco} os raios paralelos e o cabo que
sustenta um tabuleiro uniforme; a trajetória de toda bola
lançada; e, a menos de ampliação, uma única curva --- uma só
\omterm{def:g11:quad:parabola}{parábola}, em toda parte.
\end{solution}
