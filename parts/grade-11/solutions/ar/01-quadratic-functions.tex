\section*{الفصل \ref{ch:g11:quad} --- الدوال والمعادلات التربيعية}

\begin{solution}{exo:g11:quad:1}
$x^2-5x+6$: $\Delta = 25 - 24 = 1$، والجذران $\frac{5 \pm 1}{2}$، أي $2$
و $3$.

$2x^2+3x-2$: $\Delta = 9 + 16 = 25$، والجذران $\frac{-3 \pm 5}{4}$، أي
$\frac12$ و $-2$.

$x^2+2x+5$: $\Delta = 4 - 20 = -16 < 0$، فلا حل حقيقي.

$9x^2-6x+1$: $\Delta = 36 - 36 = 0$، \omterm{def:g11:quad:discriminant}{وجذر} وحيد (مضاعف)
$x_0 = \frac{6}{18} = \frac13$.
\end{solution}

\begin{solution}{exo:g11:quad:2}
$f(x) = x^2 - 6x + 11 = (x-3)^2 - 9 + 11 = (x-3)^2 + 2$: الرأس $(3, 2)$.

$g(x) = -2x^2+4x+1 = -2(x^2 - 2x) + 1 = -2\bigl((x-1)^2 - 1\bigr) + 1
= -2(x-1)^2 + 3$: الرأس $(1, 3)$.

$h(x) = 3x^2 + 6x = 3(x^2 + 2x) = 3\bigl((x+1)^2 - 1\bigr)
= 3(x+1)^2 - 3$: الرأس $(-1, -3)$.
\end{solution}

\begin{solution}{exo:g11:quad:3}
$x^2 - 3x - 10$: $\Delta = 9 + 40 = 49$، والجذران $\frac{3 \pm 7}{2} = -2$
و $5$؛ ومعامل الحد الأعلى موجب، إذن يكون ثلاثي الحدود سالبًا
بين الجذرين تمامًا: فمجموعة الحلول هي $\intoo{-2}{5}$.

$2x^2 + x + 3$: $\Delta = 1 - 24 = -23 < 0$ و $a = 2 > 0$: فهو موجب
دائمًا، ومجموعة الحلول هي $\R$.

$-x^2 + 6x - 9 = -(x-3)^2$: وهو $\geq 0$ فقط حيث $(x-3)^2 \leq 0$،
أي عند $x = 3$. فمجموعة الحلول هي $\{3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:4}
$x^2 - 7x + 6$: لدينا $1 - 7 + 6 = 0$، إذن $1$ \omterm{def:g11:quad:discriminant}{جذر}؛ وجداء
الجذرين $6$، إذن \omterm{def:g11:quad:discriminant}{الجذر} الآخر $6$.

$2x^2 + 5x - 7$: لدينا $2 + 5 - 7 = 0$، إذن $1$ \omterm{def:g11:quad:discriminant}{جذر}؛ والجداء
$\frac{c}{a} = -\frac72$، إذن \omterm{def:g11:quad:discriminant}{الجذر} الآخر $-\frac72$.

$x^2 + 4x - 5$: لدينا $1 + 4 - 5 = 0$، إذن $1$ \omterm{def:g11:quad:discriminant}{جذر}؛ والجداء $-5$، إذن
\omterm{def:g11:quad:discriminant}{الجذر} الآخر $-5$.
\end{solution}

\begin{solution}{exo:g11:quad:5}
العددان جذرا $x^2 - 14x + 45 = 0$ (\cref{prop:g11:quad:vieta})؛
و $\Delta = 196 - 180 = 16$، والجذران $\frac{14 \pm 4}{2} = 9$ و $5$.

أما المستطيل: فنصف المحيط يعطي الطول $+$ العرض $= 17$، و
المساحة تعطي جداءهما $60$. وهما جذرا $x^2 - 17x + 60 = 0$؛
و $\Delta = 289 - 240 = 49$، والجذران $\frac{17 \pm 7}{2} = 12$ و $5$. إذن
المستطيل $12 \times 5$.
\end{solution}

\begin{solution}{exo:g11:quad:6}
$\Delta = (m-2)^2 - 4$. ويوجد حل وحيد عندما $\Delta = 0$:
أي $(m-2)^2 = 4$، إذن $m - 2 = \pm 2$، أي $m = 0$ أو $m = 4$. ولا يوجد حل
عندما $\Delta < 0$: أي $(m-2)^2 < 4$، أي $-2 < m - 2 < 2$، أي
$0 < m < 4$.
\end{solution}

\begin{solution}{exo:g11:quad:7}
الرأس عند $t = -\frac{20}{2 \times (-5)} = 2$، و
$h(2) = -20 + 40 + 1 = 21$: أي ارتفاع أعظمي $21$ م، يُبلغ بعد $2$
ثانية.

والشرط $h(t) \geq 16$ يعني $-5t^2 + 20t - 15 \geq 0$، أي (بالقسمة على $-5$،
وهي تقلب المتراجحة) $t^2 - 4t + 3 \leq 0$. وجذرا
$t^2-4t+3$ هما $1$ و $3$ (إذ $1 - 4 + 3 = 0$، والجداء $3$)، إذن
$t^2 - 4t + 3 \leq 0$ على $\intcc{1}{3}$ بالضبط: فالكرة على ارتفاع
$16$ م على الأقل بين اللحظتين $t = 1$ ثا و $t = 3$ ثا.
\end{solution}

\begin{solution}{exo:g11:quad:8}
بوضع $X = x^2$: $X^2 - 13X + 36 = 0$، و $\Delta = 169 - 144 = 25$، إذن
$X = \frac{13 \pm 5}{2} = 9$ أو $4$. وبالعودة إلى $x$: يعطي $x^2 = 9$ أن
$x = \pm 3$، ويعطي $x^2 = 4$ أن $x = \pm 2$. فمجموعة الحلول هي
$\{-3, -2, 2, 3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:9}
الشرط هو $x^2 < x + 2$، أي $x^2 - x - 2 < 0$. وبما أن
$x^2 - x - 2 = (x+1)(x-2)$ (والجذران $-1$ و $2$)، تتحقق المتراجحة
بين الجذرين تمامًا: $x \in \intoo{-1}{2}$. وعلى رسم، ينغمس
\omterm{def:g11:quad:parabola}{القطع المكافئ} $y = x^2$ تحت المستقيم $y = x+2$ بالضبط بين نقطتي
تقاطعهما $(-1, 1)$ و $(2, 4)$.
\end{solution}

\begin{solution}{exo:g11:quad:10}
ليكن $x > 0$ يحقق $x + \frac1x = \frac{13}{6}$. وبالضرب في $6x > 0$:
نجد $6x^2 + 6 = 13x$، أي $6x^2 - 13x + 6 = 0$. عندئذٍ
$\Delta = 169 - 144 = 25$ و $x = \frac{13 \pm 5}{12}$، فيكون
$x = \frac32$ أو $x = \frac23$. وكلاهما موجب وكل منهما مقلوب
الآخر، كما هو متوقع: فالعدد هو $\frac32$ (أو $\frac23$).
\end{solution}

\begin{solution}{exo:g11:quad:11}
\emph{1.} تقابل نقط \omterm{def:g10:numbers:interunion}{التقاطع} حلول $x^2 = mx - 1$، أي
$x^2 - mx + 1 = 0$، ومميزها $\Delta = m^2 - 4$. فنقطتان
عندما $\abs{m} > 2$؛ ونقطة واحدة بالضبط عندما $m = \pm 2$؛ ولا نقطة عندما $-2 < m < 2$.

\emph{2.} من أجل $m = 2$: $x^2 - 2x + 1 = (x-1)^2 = 0$، إذن نقطة التماس
الوحيدة هي $(1, 1)$. ومن أجل $m = -2$: $(x+1)^2 = 0$، ونقطة التماس $(-1, 1)$.
وفي الحالتين يلتقي المستقيم \omterm{def:g11:quad:parabola}{القطع المكافئ} في نقطة واحدة دون
أن يقطعه --- فهو المماس هناك، وفعلًا يساوي ميله $m = \pm 2$
قيمة مشتقة $x^2$، وهي $2x$، عند $x = \pm 1$
(\cref{ch:g11:deriv}).
\end{solution}

\begin{solution}{pb:g11:quad:1}
\textbf{1.} $x^2 - 5x + 6$: $\Delta = 1$، والجذران $2$ و $3$.
و $2x^2 + x - 6$: $\Delta = 1 + 48 = 49$، والجذران
$\frac{-1 \pm 7}{4}$: أي $\frac32$ و $-2$. و $x^2 + x + 1$:
$\Delta = -3 < 0$: فلا حل حقيقي.

\textbf{2.} $\Delta = 16 + 48 = 64$: الجذران $3$ و $-1$، إذن
$2x^2 - 4x - 6 = 2(x - 3)(x + 1)$؛ \omterm{thm:g11:quad:canonical}{والصورة النموذجية}
$2(x - 1)^2 - 8$. فالصورة المحلَّلة تُظهر الجذرين، و
\omterm{thm:g11:quad:canonical}{الصورة النموذجية} تُظهر الرأس $(1, -8)$، والصورة المنشورة تُظهر
\omterm{def:g10:reffunc:affine}{الترتيب عند المبدأ} $-6$.

\textbf{3.} الجذران $1$ و $3$، ومعامل الحد الأعلى سالب:
إذن ثلاثي الحدود $\geq 0$ \emph{بين} الجذرين:
$x \in \intcc{1}{3}$.

\textbf{4.} العددان جذرا $x^2 - 7x + 12 = 0$:
أي $3$ و $4$. وإذا كان $2$ جذرًا للعبارة $x^2 - 5x + c$، فإن \omterm{def:g11:quad:discriminant}{الجذر} الآخر
$r$ يحقق $2 + r = 5$ (المجموع)، إذن $r = 3$، و
$c = 2 \times 3 = 6$ (الجداء): يقرؤها فييت مباشرة.

\textbf{5.} $\Delta = m^2 - 36$: \omterm{def:g11:quad:discriminant}{جذر} مضاعف من أجل $m = \pm 6$
(\omterm{def:g11:quad:discriminant}{والجذر} $\mp 3$)؛ ولا \omterm{def:g11:quad:discriminant}{جذر} من أجل $-6 < m < 6$؛ وجذران من أجل
$\abs m > 6$.

\textbf{6.} $PF^2 = x^2 + \left(x^2 - \frac14\right)^2
= x^2 + x^4 - \frac{x^2}{2} + \frac{1}{16}
= x^4 + \frac{x^2}{2} + \frac{1}{16}
= \left(x^2 + \frac14\right)^2$. ومنه
$PF = x^2 + \frac14$ (والعاملان موجبان).

\textbf{7.} المسافة من $P(x, x^2)$ إلى المستقيم الأفقي
$y = -\frac14$ هي $x^2 + \frac14$ --- أي $PF$ بالضبط. فكل
نقطة من \omterm{def:g11:quad:parabola}{القطع المكافئ} متساوية البعد عن \omterm{pb:g11:quad:1}{البؤرة} وعن
\omterm{pb:g11:quad:1}{الدليل}.

\textbf{8.} $PF^2 = x^2 + \left(y - \frac14\right)^2$ و
$\operatorname{dist}^2 = \left(y + \frac14\right)^2$. وتعطي المساواة
$x^2 + y^2 - \frac y2 + \frac{1}{16} = y^2 + \frac y2 +
\frac1{16}$، إذن $x^2 = y$: فالمحل الهندسي هو \omterm{def:g11:quad:parabola}{القطع المكافئ} بالضبط.
نقطة واحدة ومستقيم واحد يولّدان \omterm{def:g10:functions:graph}{المنحنى}، كما ولّدت نقطة واحدة و
مسافة واحدة الدائرة.

\textbf{9.} من أجل $y = \frac{x^2}{4f}$، يعطي \omterm{def:g10:algebra:expand}{النشر} نفسه مع
$F(0, f)$ أن $PF^2 = x^2 + \left(\frac{x^2}{4f} - f\right)^2
= \left(\frac{x^2}{4f} + f\right)^2$: أي أن المسافة إلى $F$ تساوي
المسافة إلى $y = -f$. أما الطبق: فيعطي $9 = \frac{30^2}{4f}$
أن $4f = 100$، إذن $f = 25$ سم: فيعلّق المستقبِل على ربع متر
فوق الرأس --- أي خارج الوعاء ذي العمق $9$ سم، كما تبيّن الأطباق
الحقيقية.

\textbf{10.} تكون جبهة موجية واردة موازية للمحور قد
قطعت مسافات متساوية لتبلغ مستوى \omterm{pb:g11:quad:1}{الدليل}؛
ويحوّل تساوي البعد عبارة ``المسافة نفسها إلى \omterm{pb:g11:quad:1}{الدليل}'' إلى
``المسافة نفسها إلى $F$'': فتصل كل الإشارات المنعكسة إلى
$F$ \emph{متزامنة}، فيعزز بعضها بعضًا --- وهناك ينبغي الإصغاء.
أما البرهان الأمين فيحتاج إلى \omterm{def:g10:reffunc:affine}{ميل} المماس عند كل نقطة ---
أي إلى المشتقة في \cref{ch:g11:deriv}، الملموحة في
\cref{exo:g11:quad:11}.

\textbf{11.} $y = x\left(1 - \frac{x}{20}\right)$: تحط عند
$x = 20$ م. والرأس في \omterm{prop:g10:coordgeom:midpoint}{المنتصف}: $x = 10$، والارتفاع
$y(10) = 10 - 5 = 5$ م.

\textbf{12.} $y(4) = 4 - \frac{16}{20} = 3.2$ م: فتتجاوز جدار $3$ م
بمتسع، وترتطم بجدار $4$ م.

\textbf{13.} $y(10) = 10 - \frac{100}{40} = 7.5$ م $> 7$: فتمر
الشاحنة على بعد $10$ م عن المركز. أما $y(15) = 10 - \frac{225}{40}
= 4.375$ م: فلا تمر على بعد $15$ م.

\textbf{14.} $R(p) = -2p^2 + 120p$: الرأس عند
$p = \frac{120}{4} = 30$ يورو، والإيراد
$R(30) = 30 \times 60 = 1\,800$ يورو.

\textbf{15.} $y(320) = 143 \times \left(\frac{320}{640}\right)^2
= \frac{143}{4} \approx 36$ م فوق أخفض نقطة.

\textbf{16.} $x^2 = mx - 1$ يعطي $x^2 - mx + 1 = 0$، و
$\Delta = m^2 - 4$: فنقطتان من أجل $\abs m > 2$، ولا نقطة من أجل
$\abs m < 2$، وتماس من أجل $m = \pm 2$ مع نقطة تماس
$x = \frac m2 = \pm 1$: فالمماسان $y = 2x - 1$ عند $(1, 1)$ و
$y = -2x - 1$ عند $(-1, 1)$.

\textbf{17.} $x^2 = mx + 1$ يعطي $\Delta = m^2 + 4 > 0$ من أجل
كل $m$: أي تقاطعين دائمًا. فمن نقطة داخلية،
يقطع كل مستقيم \omterm{def:g10:functions:graph}{المنحنى} مرتين --- ولا تُرسم المماسات إلا
من الخارج، تمامًا كما في الدوائر.

\textbf{18.} يلتقي $y = \frac14$ \omterm{def:g10:functions:graph}{المنحنى} $y = x^2$ عند $x = \pm\frac12$:
فطول الوتر $1$. ومن أجل $y = \frac{x^2}{4f}$: يعطي $y = f$ أن
$x^2 = 4f^2$، أي $x = \pm 2f$: فالطول $4f$ --- أي أن العرض البؤري
أربعة أمثال البعد البؤري (ويسميه مصممو الأطباق
الوتر القائم).

\textbf{19.} نقطة من $y = x^2$ هي $(a, a^2)$؛ وصورتها
$(X, Y) = (ka, ka^2)$، و $\frac{X^2}{k} = \frac{k^2a^2}{k} =
ka^2 = Y$: \omterm{def:g10:functions:graph}{فمنحنى} الصورة هو بالضبط $y = \frac{x^2}{k}$.
واختيار $k = 4f$ يبلغ كل \omterm{def:g11:quad:parabola}{قطع مكافئ}: فكلها تكبيرات
\omterm{def:g10:functions:graph}{لمنحنى} أصلي واحد. (والدوائر كلها متشابهة أيضًا --- أما القطوع الناقصة فلا: إذ يهم التفلطح، ولا يهم التكبير.)

\textbf{20.} \omterm{def:g10:algebra:equation}{معادلة} بثلاثة أثواب مقروءة؛ ومحل تساوي
البعد عن نقطة وعن مستقيم؛ والعاكس
الذي يجمع الأشعة المتوازية في بؤرته والكابل الذي
يحمل سطحًا منتظمًا؛ ومسار كل كرة مقذوفة؛ و،
إلى حد التكبير، \omterm{def:g10:functions:graph}{منحنى} واحد --- \omterm{def:g11:quad:parabola}{قطع مكافئ} واحد،
في كل مكان.
\end{solution}
