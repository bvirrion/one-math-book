\section*{الفصل \ref{ch:g11:deriv} --- الاشتقاق: مدخل أول}

\begin{solution}{exo:g11:deriv:1}
من أجل $f(x) = x^2$ عند $a = 2$:
\[
\frac{(2+h)^2 - 4}{h} = \frac{4h + h^2}{h} = 4 + h
\xrightarrow[h\to0]{} 4,
\]
إذن $f'(2) = 4$. ومن أجل $g(x) = x^2 + 3x$ عند $a = 1$: لدينا $g(1) = 4$ و
\[
\frac{(1+h)^2 + 3(1+h) - 4}{h} = \frac{5h + h^2}{h} = 5 + h
\xrightarrow[h\to0]{} 5,
\]
إذن $g'(1) = 5$.
\end{solution}

\begin{solution}{exo:g11:deriv:2}
$f'(x) = 12x^2 - 4x + 1$.

$g'(x) = \frac{5}{2\sqrt x} - \frac{3}{x^2}$ (على $\intoo{0}{+\infty}$).

$h'(x) = 2(x^2 - 3) + (2x+1)(2x) = 2x^2 - 6 + 4x^2 + 2x
= 6x^2 + 2x - 6$ (قاعدة الجداء).
\end{solution}

\begin{solution}{exo:g11:deriv:3}
$f(x) = x^2 - 3x$، $a = 2$: لدينا $f(2) = -2$ و $f'(x) = 2x - 3$ إذن $f'(2) = 1$؛
\omterm{def:g11:deriv:tangent}{والمماس} $y = -2 + (x - 2) = x - 4$.

$f(x) = \frac1x$، $a = 1$: لدينا $f(1) = 1$ و $f'(1) = -1$؛ \omterm{def:g11:deriv:tangent}{والمماس}
$y = 1 - (x - 1) = 2 - x$.

$f(x) = \sqrt x$، $a = 4$: لدينا $f(4) = 2$ و $f'(4) = \frac{1}{2\sqrt4} =
\frac14$؛ \omterm{def:g11:deriv:tangent}{والمماس} $y = 2 + \frac14(x - 4) = \frac{x}{4} + 1$.
\end{solution}

\begin{solution}{exo:g11:deriv:4}
$f'(x) = \frac{(x-1) - (x+2)}{(x-1)^2} = \frac{-3}{(x-1)^2}$.

$g'(x) = \frac{2x(x^2+1) - x^2 \times 2x}{(x^2+1)^2}
= \frac{2x}{(x^2+1)^2}$.

$h'(x) = -\frac{2x+1}{(x^2+x+1)^2}$ (قاعدة الخارج مع $u = 1$).
\end{solution}

\begin{solution}{exo:g11:deriv:5}
$f'(x) = 3x^2 - 12x + 9 = 3(x^2 - 4x + 3) = 3(x-1)(x-3)$: موجب
خارج $\intcc{1}{3}$، وسالب داخلها. إذن $f$ \omterm{def:g11:func:monotone}{متزايدة} على
$\intoc{-\infty}{1}$، \omterm{def:g10:functions:variations}{ومتناقصة} على $\intcc{1}{3}$، \omterm{def:g11:func:monotone}{ومتزايدة} على
$\intco{3}{+\infty}$، ولها \omterm{def:g10:functions:extrema}{قيمة عظمى} محلية $f(1) = 1 - 6 + 9 - 2 = 2$ و
\omterm{def:g10:functions:extrema}{قيمة صغرى} محلية $f(3) = 27 - 54 + 27 - 2 = -2$.
\end{solution}

\begin{solution}{exo:g11:deriv:6}
على $\intoo{-1}{+\infty}$:
\[
f'(x) = \frac{2x(x+1) - (x^2+3)}{(x+1)^2}
= \frac{x^2 + 2x - 3}{(x+1)^2}
= \frac{(x+3)(x-1)}{(x+1)^2}.
\]
ومن أجل $x > -1$ يكون $x + 3$ و $(x+1)^2$ موجبين، إذن يحمل $f'(x)$
إشارة $x - 1$: فتكون $f$ \omterm{def:g10:functions:variations}{متناقصة} على $\intoc{-1}{1}$ \omterm{def:g11:func:monotone}{ومتزايدة} على
$\intco{1}{+\infty}$، وقيمتها الصغرى $f(1) = \frac{4}{2} = 2$.
\end{solution}

\begin{solution}{exo:g11:deriv:7}
$f'(x) = 2x - 4$.

\emph{1.} \omterm{def:g11:deriv:tangent}{المماس} الأفقي: $f'(x) = 0$ عند $x = 2$؛ والنقطة هي
$(2, f(2)) = (2, 1)$ (وهي رأس \omterm{def:g11:quad:parabola}{القطع المكافئ}).

\emph{2.} التوازي مع $y = 2x + 1$ يعني \omterm{def:g10:reffunc:affine}{الميل} $2$: أي $2x - 4 = 2$، إذن
$x = 3$، فتكون النقطة $(3, f(3)) = (3, 2)$.
\end{solution}

\begin{solution}{exo:g11:deriv:8}
بعرض $x$ (وضلعان مسيَّجان) وطول $60 - 2x$:
$A(x) = x(60 - 2x) = 60x - 2x^2$ على $\intoo{0}{30}$. عندئذٍ
$A'(x) = 60 - 4x$، وهو موجب قبل $x = 15$ وسالب بعدها: إذن تكون المساحة
أعظمية من أجل $x = 15$، فتعطي حظيرة $15 \times 30$ م مساحتها
$450$ م$^2$. وتعطي صيغة الرأس $\alpha = -\frac{60}{2 \times (-2)} = 15$
في \cref{ch:g11:quad} الجواب نفسه دون حساب تفاضل.
\end{solution}

\begin{solution}{exo:g11:deriv:9}
لتكن $f(x) = \frac{x+1}{2} - \sqrt x$ على $\intoo{0}{+\infty}$. عندئذٍ
\[
f'(x) = \frac12 - \frac{1}{2\sqrt x} = \frac{\sqrt x - 1}{2\sqrt x},
\]
وهو يحمل إشارة $\sqrt x - 1$: سالب على $\intoo{0}{1}$، وموجب
على $\intoo{1}{+\infty}$. إذن تتناقص $f$ ثم تتزايد، وقيمتها الصغرى
$f(1) = 1 - 1 = 0$. ومنه $f(x) \geq 0$ من أجل كل $x > 0$، أي
$\sqrt x \leq \frac{x+1}{2}$، مع تحقق المساواة عند $x = 1$ فقط.
\end{solution}

\begin{solution}{exo:g11:deriv:10}
من $\pi r^2 h = 500$ نجد $h = \frac{500}{\pi r^2}$، إذن
\[
S(r) = 2\pi r^2 + 2\pi r \cdot \frac{500}{\pi r^2}
= 2\pi r^2 + \frac{1000}{r},
\qquad
S'(r) = 4\pi r - \frac{1000}{r^2}.
\]
ولدينا $S'(r) = 0$ عندما $4\pi r^3 = 1000$، أي $r^3 = \frac{250}{\pi}$، إذن
$r = \sqrt[3]{250/\pi}$ ($\approx 4.3$ سم). وبما أن $S'(r) =
\frac{4\pi r^3 - 1000}{r^2}$ سالب قبل هذه القيمة وموجب
بعدها، فإنها \omterm{def:g10:functions:extrema}{قيمة صغرى}.
\end{solution}

\begin{solution}{exo:g11:deriv:11}
\omterm{def:g11:deriv:tangent}{المماس} \omterm{def:g10:functions:graph}{للمنحنى} $y = x^2$ عند النقطة ذات \omterm{def:g10:coordgeom:system}{الفاصلة} $a$ هو
$y = a^2 + 2a(x - a) = 2ax - a^2$. وهو يمر بالنقطة $P(1, -3)$ عندما
$-3 = 2a - a^2$، أي $a^2 - 2a - 3 = 0$، وجذراها $a = 3$ و
$a = -1$ ($\Delta = 16$). إذن يوجد \emph{مماسان} يمران
بالنقطة $P$:
\[
y = 6x - 9 \quad (a = 3),
\qquad
y = -2x - 1 \quad (a = -1).
\]
\end{solution}

\begin{solution}{exo:g11:deriv:12}
من \cref{ex:g11:deriv:study}، تتزايد $f$ حتى \omterm{def:g10:functions:extrema}{القيمة العظمى} المحلية
$f(-1) = 3$، ثم تتناقص إلى \omterm{def:g10:functions:extrema}{القيمة الصغرى} المحلية $f(1) = -1$، ثم تتزايد
من جديد؛ ومن أجل $x$ الموجب الكبير (أو السالب الكبير)، يهيمن $x^3$ فتأخذ $f$
قيمًا كبيرة (أو صغيرة) بأي قدر. وبقراءة المستقيمات الأفقية
$y = k$ على \omterm{def:g10:functions:table}{جدول التغيرات}:
\begin{itemize}
  \item إذا كان $k > 3$ أو $k < -1$: التقى المستقيم \omterm{def:g10:functions:graph}{المنحنى} مرة واحدة
        --- أي حل واحد؛
  \item وإذا كان $k = 3$ أو $k = -1$: لامس المستقيم نقطة انعطاف وقطع
        فرعًا آخر --- أي حلان؛
  \item وإذا كان $-1 < k < 3$: قطع المستقيم الفروع الثلاثة كلها --- أي ثلاثة
        حلول.
\end{itemize}
\end{solution}

\begin{solution}{pb:g11:deriv:1}
\textbf{1.} $f'(x) = 6x - 5$؛ و $g'(x) = 3x^2 - 12$. وبالتعريف:
$\frac{(1 + h)^2 - 1}{h} = 2 + h \to 2$: أي أن \omterm{def:g11:deriv:derivative}{مشتقة}
$x^2$ عند $1$ هي $2$.

\textbf{2.} $g(2) = 8 - 24 = -16$ و $g'(2) = 12 - 12 = 0$:
إذن \omterm{def:g11:deriv:tangent}{المماس} هو المستقيم الأفقي $y = -16$. \omterm{def:g11:deriv:tangent}{والمماس} الأفقي
يدل على نقطة حرجة --- وهي هنا \omterm{def:g10:functions:extrema}{قيمة صغرى} محلية، كما
يؤكد \omterm{def:g10:functions:table}{جدول التغيرات} في السؤال 5.

\textbf{3.} \omterm{def:g10:reffunc:affine}{الميل} $2a$ عند $(a, a^2)$:
$y = a^2 + 2a(x - a) = 2ax - a^2$.

\textbf{4.} عند $x = 0$: يكون $y = -a^2$: أي أن \omterm{def:g11:deriv:tangent}{المماس} يقطع
المحور على عمق $a^2$ تحت المبدأ --- أي على البعد نفسه
\emph{تحت} الصفر الذي تعلوه فيه $P$. والوصفة: لرسم
\omterm{def:g11:deriv:tangent}{المماس} عند نقطة ارتفاعها $h$، صِلها بنقطة المحور الواقعة على
عمق $h$ تحت المبدأ. (فلا حاجة إلى \omterm{def:g11:deriv:derivative}{مشتقة} على
لوح الرسم.)

\textbf{5.} $g'(x) = 3(x^2 - 4)$: موجب على
$\intoo{-\infty}{-2}$، وسالب على $\intoo{-2}{2}$، وموجب
بعدها. فتتزايد $g$ إلى \omterm{def:g10:functions:extrema}{قيمة عظمى} محلية $g(-2) = 16$، ثم تهبط
إلى \omterm{def:g10:functions:extrema}{قيمة صغرى} محلية $g(2) = -16$، ثم تتزايد.

\textbf{6.} $F\left(0, \frac14\right)$ و $T(0, -a^2)$:
$FT = \frac14 + a^2$.

\textbf{7.} $FP = a^2 + \frac14 = FT$: إذن متساوي الساقين في $F$، فتكون
زاويتا القاعدة متساويتين: $\widehat{FTP} = \widehat{FPT}$.

\textbf{8.} الشعاع الوارد عند $P$ شاقولي، إذن يوازي
المستقيم $(TF)$ (وهو المحور $y$). ويقطع \omterm{def:g11:deriv:tangent}{المماس} $(TP)$
المتوازيين معًا، إذن تساوي الزاوية بين الشعاع \omterm{def:g11:deriv:tangent}{والمماس}
الزاويةَ $\widehat{FTP}$ (زاويتان متبادلتان) --- وهي تساوي
$\widehat{FPT}$، أي الزاوية بين $[PF]$ \omterm{def:g11:deriv:tangent}{والمماس}.
إذن زاويتان متساويتان على جهتَي \omterm{def:g11:deriv:tangent}{المماس} عند $P$.

\textbf{9.} بقانون الانعكاس، يغادر الشعاع الواصل
شاقوليًا إلى $P$ على المستقيم الذي يصنع الزاوية المساوية
في الجهة الأخرى --- وهو، حسب السؤال 8، المستقيم $[PF]$: أي عبر
\omterm{pb:g11:quad:1}{البؤرة}، أيًّا كان $a$. وبعكس ذلك، يغادر الضوء المولود عند $F$
كل نقطة من المرآة موازيًا للمحور. فالأطباق تجمع،
والمصابيح الأمامية تبعث: وقد بُرهن على ذلك.

\textbf{10.} من أجل $a = 1$: \omterm{def:g11:deriv:tangent}{المماس} $y = 2x - 1$، و $T(0, -1)$؛ و
$FP = \sqrt{1 + \left(1 - \frac14\right)^2} =
\sqrt{\frac{25}{16}} = \frac54$؛ و $FT = \frac14 + 1 = \frac54$:
إذن متساوي الساقين، كما وُعد.

\textbf{11.} \omterm{def:g11:deriv:tangent}{المماس} عند $x_n$ هو
$y = f(x_n) + f'(x_n)(x - x_n)$؛ وهو يقطع $y = 0$ حيث
$x = x_n - \frac{f(x_n)}{f'(x_n)}$.

\textbf{12.} $x_1 = 1 - \frac{-1}{2} = \frac32$؛
و $x_2 = \frac32 - \frac{1/4}{3} = \frac{17}{12}$؛
و $x_3 = \frac{577}{408}$: أي تقريبات هيرون للعدد $\sqrt2$،
وهي آلة \omterm{pb:g10:functions:1}{النقطة الثابتة} في \cref{pb:g10:functions:1}.
والمتطابقة: $x - \frac{x^2 - 2}{2x} = \frac{2x^2 - x^2 + 2}{2x}
= \frac{x^2 + 2}{2x} = \frac12\left(x + \frac2x\right)$ ---
أي أن وصفة هيرون \emph{هي} طريقة نيوتن مطبقة على
$x^2 - 2$، قبل ألفي سنة.

\textbf{13.} $x_1 = 5 - \frac{125 - 100}{75} = \frac{14}{3}
\approx 4.6667$؛ و $x_2 \approx 4.6667 - \frac{1.63}{65.33}
\approx 4.6417$. وعلى الحاسبة: $\sqrt[3]{100} = 4.6416\ldots$ ---
أي أربعة أرقام عشرية صحيحة في خطوتين.

\textbf{14.} $f'(2) = 0$: \omterm{def:g11:deriv:tangent}{فالمماس} عند نقطة الانطلاق
أفقي (السؤال 2) ولا يقطع المحور أبدًا --- و
الصيغة تقسم على صفر. والتحذير: ابدأ نيوتن بعيدًا عن
النقط الحرجة (وبوجه أعم، قريبًا بما يكفي من \omterm{def:g11:quad:discriminant}{الجذر}
المطلوب).

\textbf{15.} يكسب التنصيف رقمًا ثنائيًا واحدًا في كل خطوة؛ أما نيوتن
فيكاد \emph{يضاعف} عدد الأرقام الصحيحة في كل خطوة
($1 \to 3 \to 6$ هنا). وعندما تكون كل خطوة مكلفة --- ملايين
الوسائط، والملاحة في الزمن الحقيقي --- تتفوق المضاعفة على
التقدم الزاحف، ولهذا يعمل \omterm{def:g11:deriv:tangent}{المماس} داخل كل هذا
السيليكون.

\textbf{16.} $d^2 = 900 - w^2$، إذن
$S(w) = w(900 - w^2) = 900w - w^3$، من أجل $0 < w < 30$.

\textbf{17.} $S'(w) = 900 - 3w^2 = 0$ عند
$w = \sqrt{300} = 10\sqrt3 \approx 17.3$ سم (إذ $S' > 0$ قبلها، و
$< 0$ بعدها: أي \omterm{def:g10:functions:extrema}{قيمة عظمى}). عندئذٍ
$d = \sqrt{600} = 10\sqrt6 \approx 24.5$ سم.

\textbf{18.} $\frac{d^2}{w^2} = \frac{600}{300} = 2$، إذن
$d = w\sqrt2$: أي أن العمق يساوي العرض مضروبًا في $\sqrt2$ ---
وهي نسبة النجار (وإنشاء تقسيم القطر إلى أثلاث
يعطيها بالفرجار، دون أي جبر في الورشة).

\textbf{19.} العارضة المربعة: $w = d = \sqrt{450} \approx 21.2$
سم، وصلابتها $w^3 = 450^{3/2} \approx 9\,546$. والعارضة المثلى:
$S = 10\sqrt3 \times 600 = 6\,000\sqrt3 \approx 10\,392$. والربح:
نحو $8.9\,\%$ صلابة إضافية من الجذع نفسه --- \omterm{def:g11:deriv:derivative}{فالمشتقة}
تدفع ثمن المنشرة.

\textbf{20.} الهندسة: حوّلت زاويتا \omterm{def:g11:deriv:tangent}{المماس} المتساويتان
مجسمًا مكافئًا إلى جامع للضوء. والحساب العددي: يحل الانزلاق على
المماسات المعادلات بسرعة تضاعف الأرقام.
والاستمثال: وجد انعدام \omterm{def:g11:deriv:derivative}{المشتقة} أقوى مستطيل في
الجذع. كائن واحد وثلاث مهن --- وسيخبرنا تحدب السنة
الأخيرة، فوق ذلك، بأي جهة من كل \omterm{def:g11:deriv:tangent}{مماس} يقع \omterm{def:g10:functions:graph}{المنحنى}.
\end{solution}
