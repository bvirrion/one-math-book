\section*{الفصل \ref{ch:g11:vect} --- المتجهات والمستقيمات في المستوي}

\begin{solution}{exo:g11:vect:1}
$\vect{AB}\,(3, 4)$ و $\vect{AC}\,(-3, 2)$ و
$\vect{AB} + \vect{AC}\,(0, 6)$ و
$2\vect{AB} - 3\vect{AC}\,(6 + 9,\ 8 - 6) = (15, 2)$. \omterm{prop:g10:coordgeom:midpoint}{ومنتصف}
$[BC]$ هو $\left(\frac{5 + (-1)}{2}, \frac{3+1}{2}\right) = (2, 2)$.
\end{solution}

\begin{solution}{exo:g11:vect:2}
$\det = 4 \times 3 - (-2) \times (-6) = 12 - 12 = 0$: مستقيميتان (وفعلًا
$\vec v = -\frac12 \vec u$).

$\det = 2 \times 7 - 3 \times 5 = -1 \neq 0$: ليستا \omterm{def:g11:vect:collinear}{مستقيميتين}.

$\det = 0 \times (-8) - 0 \times 3 = 0$: مستقيميتان (فكلتاهما شاقولية).
\end{solution}

\begin{solution}{exo:g11:vect:3}
بالمرور بالنقطة $A(2,1)$ والتوجيه $(3, -1)$: تعطي مطابقة $(-b, a) = (3, -1)$
أن $a = -1$ و $b = -3$، إذن $-x - 3y + c = 0$؛ وبتعويض $A$: $-2 - 3 + c =
0$، فيكون $c = 5$. وبالضرب في $-1$: $x + 3y - 5 = 0$.

وبالمرور بالنقطتين $B(0,4)$ و $C(2,0)$: $\vect{BC}\,(2, -4)$، إذن $a = -4$ و
$b = -2$: $-4x - 2y + c = 0$؛ وبتعويض $B$: $-8 + c = 0$، فيكون $c = 8$.
وبالتبسيط على $-2$: $2x + y - 4 = 0$. وتحقق بالنقطة $C$: $4 + 0 - 4 = 0$.
\end{solution}

\begin{solution}{exo:g11:vect:4}
\omterm{def:g11:vect:direction}{متجهة التوجيه} هي $(-b, a) = (2, 3)$. والتقاطعان: يعطي $x = 0$
أن $y = 3$، فالنقطة $(0, 3)$؛ ويعطي $y = 0$ أن $x = -2$، فالنقطة $(-2, 0)$. ومن أجل
$P(2,6)$: $3 \times 2 - 2 \times 6 + 6 = 0$، إذن $P \in d$. ومن أجل $Q(1,4)$:
$3 - 8 + 6 = 1 \neq 0$، إذن $Q \notin d$.
\end{solution}

\begin{solution}{exo:g11:vect:5}
$\vect{AB}\,(3, 6)$ و $\vect{AC}\,(-3, -6)$:
$\det = 3 \times (-6) - (-3) \times 6 = 0$، إذن $A$ و $B$ و $C$ على استقامة واحدة.

$\vect{DE}\,(2, 3)$ و $\vect{DF}\,(5, 8)$:
$\det = 2 \times 8 - 5 \times 3 = 1 \neq 0$: فليست $D$ و $E$ و $F$ على استقامة
واحدة.
\end{solution}

\begin{solution}{exo:g11:vect:6}
الزوج الأول: بجمع المعادلتين، $(2x + y - 7) + (x - y + 1) =
3x - 6 = 0$، إذن $x = 2$، ثم $y = x + 1 = 3$: فيلتقيان في $(2, 3)$
(تحقق: $4 + 3 - 7 = 0$).

والزوج الثاني: $ab' - a'b = 1 \times 4 - (-2) \times (-2) = 0$، إذن
المستقيمان متوازيان؛ وبضرب $e$ في $-2$ نجد $-2x + 4y - 6 = 0$،
وهي تختلف عن $e'$ ($-6 \neq 1$): إذن متوازيان تمامًا، ولا
\omterm{def:g10:numbers:interunion}{تقاطع}.
\end{solution}

\begin{solution}{exo:g11:vect:7}
من أجل $d: 2x - y + 3 = 0$: التوجيه $(-b, a) = (1, 2)$ والنقطة
$(0, 3)$ تقع على $d$، إذن $d = \{(t,\ 3 + 2t) : t \in \R\}$.

ومن أجل المستقيم الوسيطي المار بالنقطة $(1, -3)$ وتوجيهه $(2, 1)$:
$a = 1$ و $b = -2$، إذن $x - 2y + c = 0$؛ وبتعويض $(1, -3)$:
$1 + 6 + c = 0$، فيكون $c = -7$. \omterm{def:g10:algebra:equation}{والمعادلة}: $x - 2y - 7 = 0$ (وتحقق عند $t = 1$،
عند النقطة $(3, -2)$: $3 + 4 - 7 = 0$).
\end{solution}

\begin{solution}{exo:g11:vect:8}
للمستقيم الموازي القيمتان $(a, b)$ نفسهما: $4x - 3y + c = 0$؛ وبالمرور
بالنقطة $P(1, -2)$: $4 + 6 + c = 0$، إذن $c = -10$ والمستقيم هو
$4x - 3y - 10 = 0$.

ويستلزم توازي $mx + 2y - 5 = 0$ مع $d$ أن
$ab' - a'b = 4 \times 2 - m \times (-3) = 8 + 3m = 0$، إذن
$m = -\frac83$.
\end{solution}

\begin{solution}{exo:g11:vect:9}
\omterm{prop:g10:coordgeom:midpoint}{منتصف} $[AB]$ هو $M\!\left(\frac{-1+3}{2}, \frac{0+2}{2}\right)
= (1, 1)$. \omterm{def:g10:stats:mean}{والمتوسط} هو المستقيم $(CM)$ حيث $C(1, -4)$: ولكلتا النقطتين
\omterm{def:g10:coordgeom:system}{الفاصلة} $1$، إذن \omterm{def:g10:stats:mean}{المتوسط} هو المستقيم الشاقولي
\[
x - 1 = 0 .
\]
\end{solution}

\begin{solution}{exo:g11:vect:10}
$\det(\vec u, \vec v) = m(m - 1) - 3 \times 2 = m^2 - m - 6$. وينعدم
عندما $m^2 - m - 6 = 0$: $\Delta = 1 + 24 = 25$، وجذراها
$m = \frac{1 \pm 5}{2}$، أي $m = 3$ أو $m = -2$.
\end{solution}

\begin{solution}{exo:g11:vect:11}
\emph{1.} $I = \left(\frac12, 0\right)$. عندئذٍ
$\vect{DI} = \left(\frac12 - 0,\ 0 - 1\right) = \left(\frac12, -1\right)$
و $\vect{DJ} = \frac23\vect{DI} = \left(\frac13, -\frac23\right)$، إذن
\[
J = D + \vect{DJ} = \left(\frac13,\ \frac13\right).
\]

\emph{2.} $\vect{AJ}\left(\frac13, \frac13\right)$ و
$\vect{AC}\,(1, 1)$:
$\det = \frac13 \times 1 - 1 \times \frac13 = 0$، إذن $A$ و $J$ و $C$ على استقامة
واحدة --- وفي الواقع $\vect{AJ} = \frac13\vect{AC}$: أي أن المستقيم $(DI)$ يقطع
القطر $(AC)$ عند ثلث طوله بالضبط انطلاقًا من $A$.
\end{solution}

\begin{solution}{pb:g11:vect:1}
\textbf{1.} $x = 1 + 3t$ و $y = 2 - t$. ومن الثانية،
$t = 2 - y$؛ وبالتعويض: $x = 1 + 3(2 - y) = 7 - 3y$:
\omterm{def:g10:algebra:equation}{فالمعادلة} الديكارتية $x + 3y - 7 = 0$.

\textbf{2.} من أجل $ax + by + c = 0$، \omterm{def:g11:vect:direction}{متجهة التوجيه} هي
$(-b, a)$: أي هنا $(5, 2)$. ونقطة: يعطي $y = 1$ أن $x = 1$:
أي $(1, 1)$. والتمثيل الوسيطي: $x = 1 + 5t$ و $y = 1 + 2t$.

\textbf{3.} من $x = 7 - 3y$ في $2x - 5y + 3 = 0$:
$14 - 6y - 5y + 3 = 0$، إذن $y = \frac{17}{11}$ و
$x = 7 - \frac{51}{11} = \frac{26}{11}$: أي النقطة
$\left(\frac{26}{11}, \frac{17}{11}\right)$.

\textbf{4.} التوجيهان $(5, 2)$ و $(3, -1)$:
$\det = 5 \times (-1) - 2 \times 3 = -11 \neq 0$: إذن متقاطعان،
كما وجدنا. وفي الزوج الثاني: التوجيهان $(5, 2)$ و
$(10, 4)$ لهما $\det = 0$ (فهما متوازيان)، وبمضاعفة
$2x - 5y + 3 = 0$ نجد $-4x + 10y - 6 = 0 \neq -4x + 10y + 1
= 0$: إذن متوازيان تمامًا.

\textbf{5.} $\vect{AB}(3, 2)$ و $\vect{AC}(9, 6)$:
$\det = 18 - 18 = 0$: إذن على استقامة واحدة.

\textbf{6.} السفينة $P$: من $x = 2t$ و $y = 3 + t$: $y = 3 +
\frac x2$. والسفينة $Q$: من $x = 10 - t$ نجد $t = 10 - x$، إذن
$y = 2(10 - x) - 1 = 19 - 2x$. \omterm{def:g10:numbers:interunion}{والتقاطع}:
$3 + \frac x2 = 19 - 2x$ يعطي $x = 6.4$ و $y = 6.2$: إذن يتقاطع المساران
في $(6.4,\ 6.2)$.

\textbf{7.} تمر $P$ هناك عندما $2t = 6.4$: أي $t = 3.2$ سا. وتمر $Q$
عندما $10 - t = 6.4$: أي $t = 3.6$ سا. أي بفارق أربع وعشرين دقيقة: فلا
تصادم. والتحذير: تُظهر الخريطة \emph{المسارات}؛ ولا تقول
سوى الساعات الوسيطية هل تشغل حركتان النقطة نفسها
في اللحظة نفسها.

\textbf{8.} $D^2(t) = (3t - 10)^2 + (4 - t)^2 =
10t^2 - 68t + 116$: والرأس عند $t = \frac{68}{20} = 3.4$ سا،
حيث $D^2 = 0.4$: فأقرب اقتراب
$D = \sqrt{0.4} \approx 0.63$ \omterm{def:g10:reffunc:affine}{ميل} عند $t = 3.4$.

\textbf{9.} إذن مُنتهك: $0.63 < 1$. وتُقرأ $D^2(t) < 1$
$10t^2 - 68t + 115 < 0$: $\Delta = 4624 - 4600 = 24$، وجذراها
$\frac{68 \pm \sqrt{24}}{20} = 3.4 \pm 0.245$: فالسفينتان
قريبتان أكثر من اللازم من $t \approx 3.16$ إلى $t \approx 3.64$ --- أي نحو
$29$ دقيقة من المخالفة. وعلى إحداهما أن تغيّر مسارها.

\textbf{10.} كل إحداثي لكل سفينة \omterm{def:g11:func:function}{دالة} \emph{تآلفية}
في $t$، إذن كل فرق بين إحداثيين تآلفي،
و $D^2$ --- وهو مجموع مربعَي دالتين تآلفيتين --- مقدار
تربيعي (معامل حده الأعلى غير سالب). فأقرب
اقتراب يُقرأ دائمًا على رأس.

\textbf{11.} لقاء $P$ عند $t = 3$، أي عند $(6, 6)$:
يعطي $(3a, 3b) = (6, 6)$ أن $(a, b) = (2, 2)$، والسرعة
$\sqrt{8} \approx 2.8$ عقدة.

\textbf{12.} $D(0, 1)$ و $I\left(\frac12, 0\right)$: التوجيه
$\left(\frac12, -1\right)$، أي $(1, -2)$: \omterm{def:g10:algebra:equation}{فالمعادلة}
$2x + y = 1$.

\textbf{13.} المستقيم $(AC)$: $y = x$. عندئذٍ $2x + x = 1$:
أي $x = \frac13$: فالنقطة $\left(\frac13, \frac13\right)$، على
ثلث الطريق من $A$ إلى $C$.

\textbf{14.} $B(1, 0)$ و $K\left(\frac12, 1\right)$: التوجيه
$\left(-\frac12, 1\right)$، \omterm{def:g10:algebra:equation}{والمعادلة} $2x + y = 2$. ومع
$y = x$: $x = \frac23$: فالنقطة
$\left(\frac23, \frac23\right)$. إذن يقطع المستقيمان القطر $(AC)$ عند
نقطتي تثليثه: وقد بُرهن على المبرهنة، بثلاثة أسطر من الحساب مرتين.

\textbf{15.} يُحمل أي متوازي أضلاع على مربع الوحدة
باختيار $A$ مبدأً و $\vect{AB}$ و $\vect{AD}$ وحدتين
للمحورين. ويحفظ هذا الاختيار كل ما تتحدث عنه المبرهنة
--- المنتصفات، والاستقامة، والنسب على مستقيم معطى (وكلها
قابلة للتعبير بالمتجهات والسلّميات) --- فبرهان
العبارة في المربع يبرهن عليها من أجل كل متوازي أضلاع.
(أما الأطوال والزوايا \emph{فلا} تُنقل: فالمعلم يشوّهها.)

\textbf{16.} $E\left(1, \frac12\right)$: والمستقيم $(AE)$:
$y = \frac x2$. والقطر $(DB)$: من $(0,1)$ إلى $(1,0)$:
$y = 1 - x$. \omterm{def:g10:numbers:interunion}{والتقاطع}: $\frac x2 = 1 - x$: أي $x = \frac23$:
فالنقطة $\left(\frac23, \frac13\right)$ --- وهي مرة أخرى
نقطة تثليث، لكن للقطر الآخر. فمستقيمات المنتصفات
تعشق الأثلاث.

\textbf{17.} الأولان: يعطي $x + y = 4$ و $2x - y = 2$ أن
$3x = 6$: أي $(2, 2)$. والثالث: $2 - 4 = -2$: متحقق. إذن متلاقية
في $(2, 2)$.

\textbf{18.} $mx - y + 2 - 3m = m(x - 3) + (2 - y) = 0$: ولكي
تتحقق \omterm{def:g10:algebra:equation}{المعادلة} من أجل \emph{كل} $m$، خذ $x = 3$ و
$y = 2$ --- وفعلًا يحوي كل $d_m$ النقطة $(3, 2)$. إنها حزمة
مستقيمات مارة بنقطة واحدة، مستقيم لكل \omterm{def:g10:reffunc:affine}{ميل} $m$.

\textbf{19.} الموازي للمستقيم $y = 2x + 7$: \omterm{def:g10:reffunc:affine}{الميل} $m = 2$، فالمستقيم
$2x - y - 4 = 0$. والمار بالمبدأ: $2 - 3m = 0$:
أي $m = \frac23$.

\textbf{20.} الديكارتي: عوّض نقطة، فتُجاب مسألة الانتماء.
والمختزل: \omterm{def:g10:reffunc:affine}{الميل} \omterm{def:g10:reffunc:affine}{والترتيب عند المبدأ} ظاهران، \omterm{def:g10:functions:graph}{والمنحنى} فوري.
والوسيطي: الساعة مدمجة فيه --- فالتصادمات، وأقرب
اقتراب، والاعتراض. \omterm{def:g11:vect:det}{والمحدد}: ضربان
متقاطعان، ويُحسم التوازي. والمعلم المختار: يصير
متوازي الأضلاع مربع وحدة، وتصير مبرهنة تثليث
كلاسيكية ثلاثة تقاطعات مستقيمات.
\end{solution}
