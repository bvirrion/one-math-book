\section*{الفصل \ref{ch:g11:prob} --- الاحتمالات والمتغيرات العشوائية}

\begin{solution}{exo:g11:prob:1}
\omterm{def:g11:prob:model}{نموذج التساوي في الاحتمال} على $52$ ورقة. القلوب: $\frac{13}{52} = \frac14$. الملوك:
$\frac{4}{52} = \frac{1}{13}$. قلب أو ملك:
$\frac{13 + 4 - 1}{52} = \frac{16}{52} = \frac{4}{13}$ (فملك
القلوب يُعدّ مرة واحدة). ولا هذا ولا ذاك: $1 - \frac{4}{13} = \frac{9}{13}$.
\end{solution}

\begin{solution}{exo:g11:prob:2}
$S = 6$: الأزواج $(1,5), (2,4), (3,3), (4,2), (5,1)$، إذن
$\P(S = 6) = \frac{5}{36}$.

$S \geq 10$: ثلاثة أزواج من أجل $10$، واثنان من أجل $11$، وواحد من أجل $12$:
$\P(S \geq 10) = \frac{6}{36} = \frac16$.

$S$ زوجي: المجاميع $2, 4, 6, 8, 10, 12$ لها $1 + 3 + 5 + 5 + 3 + 1 = 18$
زوجًا: \omterm{def:g10:proba:distribution}{فالاحتمال} $\frac{18}{36} = \frac12$.
\end{solution}

\begin{solution}{exo:g11:prob:3}
مجموع الاحتمالات $1$: $p = 1 - 0.3 - 0.2 - 0.4 = 0.1$. ثم
\[
\E(X) = 0.3 \times (-1) + 0.2 \times 0 + 0.4 \times 2 + 0.1 \times 5
= -0.3 + 0.8 + 0.5 = 1 .
\]
\end{solution}

\begin{solution}{exo:g11:prob:4}
$\E(Y) = 2 \times 3 - 5 = 1$؛ $\V(Y) = 2^2 \times 4 = 16$؛
$\sigma(Y) = 4$.
\end{solution}

\begin{solution}{exo:g11:prob:5}
المكاسب هي $0$ و $2$ و $5$ أورو باحتمالات $0.5$ و $0.3$
و $0.2$؛ وبطرح الرهان البالغ $2$ أورو، يأخذ $G$ القيم $-2, 0, 3$:
\begin{center}
\begin{tabular}{c|ccc}
$g$ & $-2$ & $0$ & $3$\\
\hline
$\P(G=g)$ & $0.5$ & $0.3$ & $0.2$
\end{tabular}
\end{center}
$\E(G) = -1 + 0 + 0.6 = -0.4$: فاللاعب يخسر $40$ سنتًا في اللعبة الواحدة
وسطيًا --- أي لعبة غير مواتية.
\end{solution}

\begin{solution}{exo:g11:prob:6}
بالسحب المرتب: $\P(X = 2) = \frac35 \times \frac24 = \frac{3}{10}$
(حمراء ثم حمراء)؛ و $\P(X = 0) = \frac25 \times \frac14 = \frac{1}{10}$
(زرقاء ثم زرقاء)؛ ومنه
$\P(X = 1) = 1 - \frac{3}{10} - \frac{1}{10} = \frac{6}{10}$. ثم
\[
\E(X) = 0 \times \frac1{10} + 1 \times \frac6{10} + 2 \times \frac3{10}
= \frac{12}{10} = 1.2 ,
\]
وهذا متسق مع الحدس: إذ $\frac35$ من السحبتين حمراء وسطيًا.
\end{solution}

\begin{solution}{exo:g11:prob:7}
$\E(G^2) = 0.5 \times 4 + 0.3 \times 0 + 0.2 \times 9 = 3.8$، إذن حسب
الصيغة المختصرة
$\V(G) = 3.8 - (-0.4)^2 = 3.8 - 0.16 = 3.64$ و
$\sigma(G) \approx 1.9$ أورو.
\end{solution}

\begin{solution}{exo:g11:prob:8}
الربح هو $80$ ناقص المطالبة:
\begin{center}
\begin{tabular}{c|ccc}
$b$ & $80$ & $-920$ & $-4920$\\
\hline
$\P(B=b)$ & $0.94$ & $0.05$ & $0.01$
\end{tabular}
\end{center}
$\E(B) = 75.2 - 46 - 49.2 = -20$: فبهذا السعر \emph{تخسر} الشركة
$20$ أورو من كل عقد وسطيًا. فالقسط أدنى مما تقتضيه الأخطار
المغطاة (وكان عليه أن يتجاوز $100$ أورو ليكون الربح المأمول
موجبًا).
\end{solution}

\begin{solution}{exo:g11:prob:9}
النتيجة العشوائية هي $+1$ \omterm{def:g10:proba:distribution}{باحتمال} $\frac14$ و $x$
\omterm{def:g10:proba:distribution}{باحتمال} $\frac34$:
\[
\E = \frac14 + \frac{3x}{4} = 0 \iff x = -\frac13 .
\]
ومع العقوبة $-\frac13$، لا يجلب التخمين العشوائي شيئًا وسطيًا.
\end{solution}

\begin{solution}{exo:g11:prob:10}
قطعتان: تتطابقان (صورة صورة أو ظهر ظهر) \omterm{def:g10:proba:distribution}{باحتمال} $\frac24 \times 2 =
\frac12$. وربح اللاعب هو $+1$ أو $-1$ باحتمالين متساويين:
\omterm{def:g11:prob:expectation}{فالأمل الرياضي} $0$ لكلا اللاعبين --- إذن اللعبة عادلة.

ثلاث قطع: تتطابق كلها (ثلاث صور أو ثلاثة أظهر) \omterm{def:g10:proba:distribution}{باحتمال} $\frac{2}{8} =
\frac14$. ويقبض اللاعب $2$ \omterm{def:g10:proba:distribution}{باحتمال} $\frac14$ ويدفع $1$
\omterm{def:g10:proba:distribution}{باحتمال} $\frac34$:
$\E = \frac24 - \frac34 = -\frac14$. فاللعبة تحابي آن بمقدار $25$ سنتًا
في الجولة؛ وكانت ستصير عادلة لو دفعت $3$ أورو عند التطابق الثلاثي.
\end{solution}

\begin{solution}{exo:g11:prob:11}
\emph{1.} يقبض المنظّم $1000 \times 5 = 5000$ أورو ويدفع
$2000 + 5 \times 200 + 20 \times 25 = 3500$ أورو في المجموع: فربحه
هو المقدار \emph{الثابت} $1500$ (ولا عشوائية فيه متى بيعت كل التذاكر).
وأما ربح لاعب واحد $G$ (الجائزة ناقص التذكرة البالغة $5$ أورو):
\begin{center}
\begin{tabular}{c|cccc}
$g$ & $1995$ & $195$ & $20$ & $-5$\\
\hline\rule{0pt}{11pt}%
$\P(G=g)$ & $\dfrac{1}{1000}$ & $\dfrac{5}{1000}$ & $\dfrac{20}{1000}$
& $\dfrac{974}{1000}$
\end{tabular}
\end{center}

\emph{2.}
\[
\E(G) = \frac{1995 + 5 \times 195 + 20 \times 20 - 974 \times 5}{1000}
= \frac{1995 + 975 + 400 - 4870}{1000} = -1.5 .
\]
فكل لاعب يتوقع أن يخسر $1.50$ أورو؛ وعلى $1000$ لاعب يكون ذلك
$1500$ أورو --- وهو ربح المنظّم بالضبط. فالدفاتر متوازنة:
إذ ما يخسره اللاعبون وسطيًا هو بالضبط ما يجمعه المنظّم.
\end{solution}

\begin{solution}{pb:g11:prob:1}
\textbf{1.} يأخذ $G$ القيم $-2, -1, 0, 1, 2, 3$، كل منها
\omterm{def:g10:proba:distribution}{باحتمال} $\frac16$: $\E(G) = \frac{-2 - 1 + 0 + 1 + 2 +
3}{6} = \frac12$ أورو.

\textbf{2.} $\E(G^2) = \frac{4 + 1 + 0 + 1 + 4 + 9}{6} =
\frac{19}{6}$؛ $V(G) = \frac{19}{6} - \frac14 = \frac{35}{12}
\approx 2.92$؛ $\sigma(G) \approx 1.71$.

\textbf{3.} $\E(2G - 1) = 2 \times \frac12 - 1 = 0$؛
$V(2G - 1) = 4\,V(G) = \frac{35}{3}$.

\textbf{4.} $\E(\text{مجموع}) = 7$: فالسعر العادل $7$ أورو. وعند $8$،
تكون الخسارة المأمولة $1$ أورو في اللعبة الواحدة.

\textbf{5.} القسط العادل: $0.05 \times 800 = 40$ أورو.
وبتقاضي $60$: يكون الربح المأمول $20$ أورو من العقد الواحد.

\textbf{6.} القسمة $5:3$ تكافئ \emph{الماضي} --- لكن
الرهان ملك لمن يربح \emph{المستقبل}، وقد يربح B
بعد. وإعطاء كل شيء إلى A يعامل فوزًا محتملًا معاملة المؤكد ---
ففرصة B الصغيرة تساوي شيئًا.

\textbf{7.} الجولات التي تُلعب: $3$؛ والإمكانيات (فوز A/فوز B في كل
جولة): AAA و AAB و ABA و ABB و BAA و BAB و BBA و BBB. ولا يأخذ اللاعب B
السلسلة إلا بالفوز بالثلاث كلها: أي BBB وحدها. ويربح A
السلسلة في $7$ من $8$ حالات: فالقسمة العادلة
$64 \times \frac78 = 56$ بستولًا للاعب A و $8$ للاعب B.

\textbf{8.} لعب الجولات الزائدة العديمة الجدوى لا يغيّر
انتصار أحد: فمتى نالت A فوزها السادس، صارت الجولات اللاحقة
مراسم. لكن تثبيت الأفق عند $3$ جولات بالضبط يجعل
الإمكانيات $8$ كلها متساوية \omterm{def:g10:proba:distribution}{الاحتمال} --- فالعدّ الأمين يحتاج
إلى حكايات متساوية الطول، وحشوها لا يكلّف شيئًا.

\textbf{9.} عند $5$--$5$: جولة \omterm{def:g10:coordgeom:system}{فاصلة} واحدة، أي $32$ لكل منهما. وعند
$5$--$4$: تنال A $\frac{64 + 32}{2} = 48$. وعند $5$--$3$: تقود
الجولة التالية إلى $64$ (إذا ربحتها A) أو إلى الوضع
$5$--$4$ الذي يساوي $48$: فحصة A $\frac{64 + 48}{2} = 56$ ---
أي $56 : 8$ عند فيرما، مستخرجة من جديد بالسير إلى الوراء.

\textbf{10.} تحتاج A إلى $2$، ويحتاج B إلى $3$: فتُلعب $4$ جولات
خيالية ($16$ إمكانية). ويأخذ B السلسلة عند $3$ أو $4$
انتصارات: أي $4 + 1 = 5$ إمكانيات. والقسمة:
$B$: $64 \times \frac{5}{16} = 20$؛ $A$: $44$.

\textbf{11.} القيمة الحالية العادلة لمردود مستقبلي غير
مؤكد هي أمله الرياضي --- أي \omterm{def:g10:stats:mean}{متوسط} ما قد يأتي مرجَّحًا
بالاحتمالات. والتأمين (السؤال 5) والمالية (تسعير المردودات
غير المؤكدة) هما هذه الجملة وقد صارت صناعتين؛
وكذلك دور القمار، من الجهة الأخرى من الطاولة.

\textbf{12.} ينتهي الرهان دائمًا في مكان ما: فالحصتان
متغيران عشوائيان مجموعهما الثابت $64$، إذن بالخطية
$\E_A + \E_B = 64$ --- فالحصص العادلة تستنفد الرهان،
عند كل نتيجة.

\textbf{13.} بالخطية: $\E(X + Y) = 3.5 + 3.5 = 7$، بلا جدول
--- وهذا صحيح حتى لو كان حجرا النرد ملتصقين. \omterm{def:g11:prob:variance}{وتباين} حجر واحد:
$\E(X^2) - 3.5^2 = \frac{91}{6} - \frac{49}{4} = \frac{35}{12}$؛
ومن أجل حجرين مستقلين
$V(X + Y) = \frac{35}{12} + \frac{35}{12} = \frac{35}{6}$.

\textbf{14.} يستعيد كل ضيف ثابت قبعته \omterm{def:g10:proba:distribution}{باحتمال}
$\frac1n$ (فقبعته واحدة من بين $n$). وبالجمع
على الضيوف $n$:
$\E(X) = n \times \frac1n = 1$ --- أي ضيف محظوظ واحد وسطيًا،
سواء أكانت في الحفلة قبعتان أم ألفان.

\textbf{15.} $n = 2$: التوزيعان: كلاهما صحيح ($X = 2$) أو
متبادلان ($X = 0$): أي $\E = 1$. و $n = 3$: تعطي التوزيعات الستة
$X = 3, 1, 1, 1, 0, 0$: $\E = \frac{6}{6} = 1$.

\textbf{16.} جمع الحساب الآمال الرياضية للمؤشرات $n$
\omterm{def:g11:func:parity}{الفردية} (``الضيف $i$ نال قبعته'')، و
الخطية تجمع الآمال الرياضية \emph{بلا شروط} ---
فالتبعية تخرّب جداءً أو تباينًا، لكنها لا تخرّب أبدًا
مجموع آمال رياضية. تلك هي القوة الخارقة للخطية: فلا
تُسأل أسئلة عن الاستقلال.

\textbf{17.} الربحان: $\E = 0.5 \times 10 - 5 = 0$ و
$0.005 \times 1000 - 5 = 0$. والتباينان (للربح):
A: $\E(G^2) = 0.5 \times 25 + 0.5 \times 25 = 25$؛
B: $0.005 \times 995^2 + 0.995 \times 25 \approx 4\,975$:
أي $\sigma_A = 5$، و $\sigma_B \approx 70.5$. والناس يشترون B: فما
يشترونه ليس \omterm{def:g11:prob:expectation}{الأمل الرياضي} (المعدوم في كليهما) بل
\emph{التبدد} --- أي فرصة صغيرة في ذيل أيمن يغيّر الحياة.

\textbf{18.} أول ظهر عند الرمية $n$ احتماله
$2^{-n}$ ويدفع $2^n$: فكل $n$ يسهم بمقدار
$2^{-n} \times 2^n = 1$ في \omterm{def:g11:prob:expectation}{الأمل الرياضي}، الذي يساوي إذن
$1 + 1 + 1 + \dots$: أي لا نهائي. ومع ذلك لا يدفع عاقل
$1\,000$ للعب مرة واحدة: فباحتمال $\frac{15}{16}$ يكون
المدفوع $16$ على الأكثر. \omterm{def:g11:prob:expectation}{فالأمل الرياضي} يلخّص \omterm{def:g10:stats:quartiles}{المدى} الطويل
للرهانات \emph{القابلة للتكرار}؛ أما لعبة مرة واحدة في العمر ذات
ذيل جامح فلا يسعّرها متوسطها.

\textbf{19.} الأملان الرياضيان: التأمين $-20$؛ والمواجهة عاريًا
$-0.01 \times 1000 = -10$. فالاستراتيجية العارية تربح وسطيًا
--- لمن يقدر على امتصاص الخسارة وتكرار الرهان
مرات كثيرة. أما الأسرة التي لا تنجو من $-1\,000$ فليست
في تجارة المتوسطات: فهي تدفع $10$ زيادة لتمحو
\omterm{def:g11:prob:variance}{التباين}. المؤمّنون يأخذون المتوسطات؛ والأسر تشتري اليقين.

\textbf{20.} وُلد وهو يقسم رهانًا بين مقامرين
نافدَي الصبر؛ ومكّنته الخطية، التي تجمع الآمال دون أن
تستأذن التبعية؛ وهو أعمى عن الخطر، الذي على \omterm{def:g11:prob:variance}{التباين}
--- في اليانصيبات، ولعبة بطرسبورغ، وعقد الأسرة ---
أن يقيسه بدلًا منه؛ ويتوّجه في السنة المقبلة قانون الأعداد
الكبيرة، الذي يفسّر لماذا تحصّل المتوسطات دائمًا،
في نهاية المطاف.
\end{solution}
