\section*{الفصل \ref{ch:g11:stat} --- الإحصاء الوصفي}

\begin{solution}{exo:g11:stat:1}
$n = 8$، والمجموع $71$: \omterm{def:g10:stats:mean}{فالمتوسط} $\bar x = \frac{71}{8} = 8.875$. \omterm{def:g11:stat:median}{والوسيط}: \omterm{prop:g10:coordgeom:midpoint}{منتصف}
القيمتين رقم $4$ ورقم $5$، أي $\frac{8 + 9}{2} = 8.5$. والربيعيات: ربع
$8$ هو $2$، إذن $Q_1$ هي القيمة رقم $2$، أي $7$؛ وثلاثة الأرباع هي
$6$، إذن $Q_3$ هي القيمة رقم $6$، أي $10$. \omterm{def:g11:stat:quartiles}{والمدى الربيعي}:
$10 - 7 = 3$.
\end{solution}

\begin{solution}{exo:g11:stat:2}
المجموع $40$، إذن $\bar x = 5$. ومجموع المربعات:
$4 + 16 + 16 + 16 + 25 + 25 + 49 + 81 = 232$. وبالصيغة المختصرة،
\[
V = \frac{232}{8} - 5^2 = 29 - 25 = 4,
\qquad
\sigma = 2 .
\]
\end{solution}

\begin{solution}{exo:g11:stat:3}
\omterm{def:g10:stats:mean}{المتوسط}:
\[
\bar x = \frac{1 \times 6 + 2 \times 9 + 3 \times 8 + 4 \times 10
+ 5 \times 9 + 6 \times 8}{50}
= \frac{181}{50} = 3.62 .
\]
\omterm{def:g11:stat:median}{والوسيط}: التواترات المتراكمة هي $6, 15, 23, 33, \dots$، إذن
النتيجتان المرتَّبتان رقم $25$ ورقم $26$ كلتاهما $4$: \omterm{def:g11:stat:median}{فالوسيط} هو $4$.
\end{solution}

\begin{solution}{exo:g11:stat:4}
مجموع العلامات $15$ هو $15 \times 11.2 = 168$؛ ومع
الوافد الجديد، $\frac{168 + 18}{16} = \frac{186}{16} = 11.625$.
\end{solution}

\begin{solution}{exo:g11:stat:5}
$\frac{6 + 8 + x + 12 + 14}{5} = 10$ يعطي $40 + x = 50$، إذن $x = 10$.
وانحرافات $6, 8, 10, 12, 14$ عن \omterm{def:g10:stats:mean}{المتوسط} $10$ هي
$-4, -2, 0, 2, 4$، ومجموع مربعاتها $40$:
$V = \frac{40}{5} = 8$ و $\sigma = 2\sqrt2 \approx 2.83$.
\end{solution}

\begin{solution}{exo:g11:stat:6}
حسب \cref{prop:g11:stat:affine} مع $a = \frac12$ و $b = 4$: \omterm{def:g10:stats:mean}{المتوسط} الجديد
$\frac{12.4}{2} + 4 = 10.2$؛ \omterm{def:g11:stat:variance}{والانحراف المعياري} الجديد
$\frac{2.5}{2} = 1.25$.
\end{solution}

\begin{solution}{exo:g11:stat:7}
\omterm{def:g10:stats:mean}{متوسط} المجموعتين معًا هو \omterm{def:g10:stats:mean}{المتوسط} \emph{المرجَّح}
\[
\frac{20 \times 10 + 30 \times 15}{50} = \frac{650}{50} = 13,
\]
لا $12.5$: فالمجموعة B تسهم بعدد أكبر من القيم، فتجذب \omterm{def:g10:stats:mean}{المتوسط} المشترك
نحو متوسطها.
\end{solution}

\begin{solution}{exo:g11:stat:8}
كلا الخطين قريب من الهدف وسطيًا ($1.002$ و $1.001$ kg)، لكن
\omterm{def:g11:stat:variance}{الانحراف المعياري} للخط 2 أكبر بثلاث مرات: فأكياسه تتفاوت
أكثر بكثير، فينتج أكياسًا ناقصة وزائدة الملء بصورة أوضح. فالخط 2
هو الذي ينبغي أن يُعالَج أولًا --- فالانتظام، لا \omterm{def:g10:stats:mean}{المتوسط} وحده، هو ما
يحتاج إلى إصلاح.
\end{solution}

\begin{solution}{exo:g11:stat:9}
$\bar x = \frac{70}{10} = 7$؛
$V = \frac{540}{10} - 7^2 = 54 - 49 = 5$؛
$\sigma = \sqrt5 \approx 2.24$.
\end{solution}

\begin{solution}{exo:g11:stat:10}
\emph{1.} مع $23$: المجموع $50$ \omterm{def:g10:stats:mean}{والمتوسط} $5$؛ \omterm{def:g11:stat:median}{والوسيط} هو \omterm{prop:g10:coordgeom:midpoint}{منتصف} القيمتين رقم $5$ و
رقم $6$ $= 3$. وبدون $23$: المجموع $27$ \omterm{def:g10:stats:mean}{والمتوسط} $3$؛ \omterm{def:g11:stat:median}{والوسيط} (الخامسة من بين 9
قيم) $= 3$.

\emph{2.} مع $23$ ($n = 10$): $Q_1$ هي القيمة رقم $3$ (أي $2$)، و $Q_3$ هي
رقم $8$ (أي $4$): \omterm{def:g11:stat:quartiles}{فالمدى الربيعي} $2$. وبدونها ($n = 9$): $Q_1$ هي
القيمة رقم $3$ (أي $2$)، و $Q_3$ هي رقم $7$ (أي $4$): وهو $2$ أيضًا.

\emph{3.} تجرّ القيمة الشاذة الوحيدة \omterm{def:g10:stats:mean}{المتوسط} من $3$ إلى $5$ لكنها تترك
\omterm{def:g11:stat:median}{الوسيط} \omterm{def:g11:stat:quartiles}{والمدى الربيعي} دون تغيير: فهذان
\emph{صامدان}. وحين يُحتمل أن تحوي سلسلة قيمًا شاذة، فأعلن
\omterm{def:g11:stat:median}{الوسيط} \omterm{def:g11:stat:quartiles}{والمدى الربيعي} لا \omterm{def:g10:stats:mean}{المتوسط} \omterm{def:g11:stat:variance}{والانحراف
المعياري}.
\end{solution}

\begin{solution}{pb:g11:stat:1}
\textbf{1.} \omterm{def:g10:stats:mean}{المتوسط} $\bar x = \frac{25}{5} = 5$. ومربعات
الانحرافات: $1, 9, 1, 0, 9$: \omterm{def:g11:stat:variance}{فالتباين} $\frac{20}{5} = 4$، إذن
$\sigma = 2$.

\textbf{2.} \omterm{def:g10:stats:mean}{متوسط} المربعات:
$\frac{16 + 64 + 36 + 25 + 4}{5} = 29$؛ ناقص
$\bar x^2 = 25$: \omterm{def:g11:stat:variance}{فالتباين} $4$. الجواب نفسه بنصف العمل.

\textbf{3.} \omterm{def:g10:stats:mean}{المتوسط} $= \frac{3 \times 8 + 5 \times 10 + 2 \times
14}{10} = 10.2$؛ \omterm{def:g11:stat:variance}{والتباين} $= \frac{3 \times 64 + 5 \times 100 +
2 \times 196}{10} - 10.2^2 = 108.4 - 104.04 = 4.36$؛
و $\sigma \approx 2.09$.

\textbf{4.} \omterm{def:g10:stats:mean}{المتوسط}: $1.8 \times 20 + 32 = 68\,^\circ$F؛
و $\sigma$: $1.8 \times 3 = 5.4\,^\circ$F --- فالانسحاب $+32$
لا يبدّد شيئًا.

\textbf{5.} بإضافة $b$: يبقى كل انحراف $x_i + b - (\bar x + b)
= x_i - \bar x$ دون تغيير، فيبقى \omterm{def:g11:stat:variance}{التباين} كذلك. وبالضرب
في $a$: تُضرب الانحرافات في $a$، ومربعاتها في
$a^2$، \omterm{def:g11:stat:variance}{والتباين} في $a^2$، و $\sigma$ في
$\sqrt{a^2} = \abs a$.

\textbf{6.} $z_A = \frac{15 - 12}{2} = 1.5$؛
$z_B = \frac{13 - 8}{4} = 1.25$. فأليس تقف أعلى فوق
جمهورها: إذ يتغلب $15$ في الامتحان الأصعب على $13$ في الأسهل.

\textbf{7.} الدرجات المعيارية $z$: $-0.5,\ 1.5,\ 0.5,\ 0,\ -1.5$: \omterm{def:g10:stats:mean}{المتوسط}
$0$، \omterm{def:g11:stat:variance}{والانحراف المعياري} $1$. وعمومًا، تطرح المعيرة
$\bar x$ (فيصير \omterm{def:g10:stats:mean}{المتوسط} $0$ ولا يمسّ $\sigma$)
ثم تقسم على $\sigma$ (فيصير $\sigma$ الجديد مساويًا
$1$): أي السؤال 5 مرتين.

\textbf{8.} $z = 2$: $100 + 2 \times 15 = 130$. و
$z(76) = \frac{76 - 100}{15} = -1.6$.

\textbf{9.} الرضيع أدنى بمقدار $2.5$ انحرافًا معياريًا من
\omterm{def:g10:stats:mean}{متوسط} وزن \emph{الرضّع في سنّه} --- أي خفيف على نحو مقلق
بالنسبة إلى فوجه. ولا يمكن مقارنة الكيلوغرامات الخام عبر الأعمار (فكل
الرضّع يكتسبون وزنًا)؛ أما الدرجات المعيارية فتقيس كل طفل في مواجهة
فئته العمرية، فمسطرة واحدة لكل الأعمار.

\textbf{10.} \omterm{def:g10:stats:mean}{المتوسط} $5$؛ \omterm{def:g10:stats:mean}{ومتوسط} المربعات $\frac{622}{10} = 62.2$،
\omterm{def:g11:stat:variance}{والتباين} $37.2$، و $\sigma \approx 6.10$؛
و $z(23) = \frac{18}{6.10} \approx 2.95$: أي دون عتبة الإنذار
بقليل --- فالاختبار يكاد لا يرفّ له جفن.

\textbf{11.} بدون $23$: \omterm{def:g10:stats:mean}{المتوسط} $3$، \omterm{def:g11:stat:variance}{والتباين}
$\frac{93}{9} - 9 \approx 1.33$، و $\sigma \approx 1.15$؛
وفي مواجهة المسطرة النظيفة، تنال $23$ الدرجة
$z \approx \frac{20}{1.15} \approx 17$: وحش. فالقيمة الشاذة
نفخت \omterm{def:g10:stats:mean}{المتوسط} و $\sigma$ معًا، فتموّهت ---
ولهذا يوصي \cref{exo:g11:stat:10} بالملخصات الصامدة
(\omterm{def:g11:stat:median}{الوسيط} \omterm{def:g11:stat:quartiles}{والمدى الربيعي}) كلما لاح خطر التلوث.

\textbf{12.} في مجموع \omterm{def:g11:stat:variance}{التباين}
$\sigma^2 = \frac1N \sum (x_i - \bar x)^2$، احتفظ فقط بالحدود
التي تحقق $\abs{x_i - \bar x} \geq k\sigma$: وعددها $pN$
حدًا، كل منها لا يقل عن $(k\sigma)^2$، إذن
$\sigma^2 \geq \frac{1}{N} \cdot pN \cdot k^2\sigma^2 =
p\,k^2\sigma^2$: ثم اقسم على $k^2 \sigma^2$ (والتبدد غير معدوم):
$p \leq \frac{1}{k^2}$.

\textbf{13.} وراء $2\sigma$: على الأكثر $\frac14 = 25\,\%$ من
أي سلسلة كانت؛ ووراء $3\sigma$: على الأكثر
$\frac19 \approx 11\,\%$. وفي معطيات السؤال 1: ما وراء
$2\sigma = 4$ عن \omterm{def:g10:stats:mean}{المتوسط} $5$ يعني $\leq 1$ أو $\geq 9$:
ولا قيمة من القيم الخمس --- $0\,\% \leq 25\,\%$، كما وُعد.

\textbf{14.} الخروج عن $\intcc{49.7}{50.3}$ يعني ما وراء
$3\sigma$: أي على الأكثر $\frac19$ من الإنتاج، \emph{مهما}
كان شكل التوزيع --- فمتراجحة تشيبيشيف لا تطلب سوى
\omterm{def:g10:stats:mean}{متوسط} و $\sigma$. أما الإنتاج ذو الشكل الجرسي فيتركز أفضل
بكثير (نحو $0.3\,\%$ وراء $3\sigma$): وذلك هو القانون الطبيعي،
في السنة المقبلة.

\textbf{15.} تقع $-19\,\%$ على بعد $24$ نقطة تحت \omterm{def:g10:stats:mean}{المتوسط} $5$.
الصندوق B: $24 = 2\sigma_B$: فتسمح متراجحة تشيبيشيف بما يصل إلى
$\frac14$ من السنوات بهذا السوء. والصندوق A: $24 = 12\sigma_A$: أي
على الأكثر $\frac{1}{144}$. \omterm{def:g10:stats:mean}{المتوسط} نفسه، والأخطار غير قابلة للمقارنة ---
و $\sigma$ هو ما تسميه المالية \emph{الخطر} (أو التقلب).

\textbf{16.} انشر حول $\bar x$:
$(x_i - c)^2 = \left((x_i - \bar x) + (\bar x - c)\right)^2
= (x_i - \bar x)^2 + 2(\bar x - c)(x_i - \bar x) +
(\bar x - c)^2$. وبأخذ \omterm{def:g10:stats:mean}{المتوسط}: يموت الحد الأوسط (لأن مجموع الانحرافات
عن \omterm{def:g10:stats:mean}{المتوسط} معدوم)، فيبقى
$f(c) = \sigma^2 + (\bar x - c)^2$ --- وهو أصغري بالضبط عند
$c = \bar x$، وأصغريته $\sigma^2$. \omterm{def:g10:stats:mean}{فالمتوسط} هو
مركز المربعات الصغرى، كما أن \omterm{def:g11:stat:median}{الوسيط} هو مركز القيم المطلقة
الصغرى (\cref{pb:g10:stats:1}).

\textbf{17.} مباشرة: الانحرافات عن $6$ هي
$-2, 2, 0, -1, -4$، ومربعاتها $4, 4, 0, 1, 16$:
$f(6) = \frac{25}{5} = 5$. وبالصيغة:
$\sigma^2 + (5 - 6)^2 = 4 + 1 = 5$. فالتوافق حاصل.

\textbf{18.} لكليهما \omterm{def:g10:stats:mean}{المتوسط} $7$؛ وأما التباينان:
$P$: $\frac{16 + 4 + 0 + 4 + 16}{5} = 8$؛
$Q$: $\frac{36 + 0 + 0 + 0 + 36}{5} = 14.4$: فتبددان مختلفان
وراء \omterm{def:g10:stats:mean}{متوسط} واحد. ومن أجل $\{7 - a, 7, 7, 7, 7 + a\}$: يعطي \omterm{def:g11:stat:variance}{التباين}
$\frac{2a^2}{5} = 8$ أن $a = \sqrt{20} \approx 4.47$: فالسلسلة
$\{2.53, 7, 7, 7, 11.47\}$ تحمل \omterm{def:g10:stats:mean}{متوسط} $P$ وتباينها
بالضبط، بملمح مختلف تمامًا ---
فالملخصات تضغط، والضغط يُفقِد.

\textbf{19.} $\sigma^2 = 0$ يعني أن مجموع مربعات
$(x_i - \bar x)^2$ ينعدم؛ والمربعات $\geq 0$، فالمجموع المعدوم
يفرض انعدام كل حد: أي $x_i = \bar x$ من أجل كل $i$
--- فكل القيم متساوية.

\textbf{20.} المراكز تقول أين تعيش المعطيات؛ والتبددات تقول بأي
سعة؛ \omterm{pb:g11:stat:1}{والدرجة المعيارية} تمدّ مسطرة واحدة عبر كل السلاسل
والامتحانات والأعمار والعملات؛ ومتراجحة تشيبيشيف تحوّل $\sigma$ إلى
ضمانة غير مشروطة؛ والسؤال 18 يبقينا صادقين
--- فما من حفنة أعداد تحوي الحكاية كلها. وفي السنة
المقبلة يشحذ \omterm{def:g10:functions:graph}{منحنى} الجرس المسطرة فتصير احتمالات
مضبوطة.
\end{solution}
