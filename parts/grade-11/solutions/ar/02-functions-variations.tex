\begin{solution}{exo:g11:func:1}

$g$: يلزم $x^2 - 4 \neq 0$، أي $x \neq \pm 2$: \omterm{def:g11:func:function}{فمجموعة التعريف}

\end{solution}

\begin{solution}{exo:g11:func:2}
$\sqrt{11} < \sqrt{13}$ لأن $\sqrt{}$ \omterm{def:g11:func:monotone}{متزايدة} و $11 < 13$.

$(-2.1)^2 > (-2.05)^2$ لأن $x^2$ \omterm{def:g10:functions:variations}{متناقصة} على
$\intoc{-\infty}{0}$ و $-2.1 < -2.05$.

$\frac{1}{0.9} > \frac{1}{1.1}$ لأن $\frac1x$ \omterm{def:g10:functions:variations}{متناقصة} على
$\intoo{0}{+\infty}$ و $0.9 < 1.1$.

$(-1.01)^3 < (-0.99)^3$ لأن $x^3$ \omterm{def:g11:func:monotone}{متزايدة} و $-1.01 < -0.99$.
\end{solution}

\begin{solution}{exo:g11:func:3}
$f(-x) = 3(-x)^4 - (-x)^2 = 3x^4 - x^2 = f(x)$: \omterm{def:g11:func:parity}{زوجية}.

$g(-x) = -x^3 - x = -g(x)$: \omterm{def:g11:func:parity}{فردية}.

$h(1) = 2$ و $h(-1) = 1 - 1 = 0$، إذن $h(-1)$ لا يساوي $h(1)$ ولا
$-h(1)$: فليست \omterm{def:g11:func:parity}{زوجية} ولا \omterm{def:g11:func:parity}{فردية}.

$k(-x) = \frac{-x}{x^2+1} = -k(x)$: \omterm{def:g11:func:parity}{فردية}.
\end{solution}

\begin{solution}{exo:g11:func:4}
اقرأ \omterm{def:g10:functions:table}{جدول التغيرات}. على $\intcc{-3}{1}$ تتسلق $f$ من $-2$ إلى $4$
وتعبر القيمة $1$ مرة واحدة بالضبط؛ وعلى $\intcc{1}{5}$ تهبط
من $4$ إلى $0$ وتعبر $1$ مرة أخرى: إذن \omterm{def:g10:algebra:equation}{للمعادلة} $f(x) = 1$
حلان. \omterm{def:g10:functions:extrema}{والقيمة العظمى} \omterm{def:g11:func:function}{للدالة} $f$ هي $f(1) = 4 < 5$، إذن ليس \omterm{def:g10:algebra:equation}{للمعادلة} $f(x) = 5$
حل.
\end{solution}

\begin{solution}{exo:g11:func:5}
ليكن $2 \leq u < v$. عندئذٍ
\[
f(v) - f(u) = v^2 - 4v - u^2 + 4u = (v-u)(v+u) - 4(v-u)
= (v - u)(u + v - 4).
\]
والعاملان موجبان: إذ $v - u > 0$، و $u + v > 2 + 2 = 4$ لأن
$u \geq 2$ و $v > 2$. ومنه $f(v) > f(u)$: أي أن $f$ \omterm{def:g11:func:monotone}{متزايدة} تمامًا
على $\intco{2}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:6}
$f = \sqrt{u}$ حيث $u(x) = 5 - x$ \omterm{def:g10:functions:variations}{متناقصة} وغير سالبة على
$\intoc{-\infty}{5}$: إذن $f$ \omterm{def:g10:functions:variations}{متناقصة} هناك
(\cref{prop:g11:func:transforms}).

$g = \frac1u$ حيث $u(x) = x^2 + 1$ \omterm{def:g11:func:monotone}{متزايدة} وموجبة على
$\intco{0}{+\infty}$: إذن $g$ \omterm{def:g10:functions:variations}{متناقصة} هناك.

$h = 3 - 2\sqrt x$: \omterm{def:g11:func:function}{الدالة} $\sqrt x$ \omterm{def:g11:func:monotone}{متزايدة}، والضرب في $-2 < 0$
يقلب، وإضافة $3$ لا تغيّر شيئًا: إذن $h$ \omterm{def:g10:functions:variations}{متناقصة} على
$\intco{0}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:7}
\emph{1.} \omterm{def:g11:func:function}{الدالة} $f$ على \omterm{thm:g11:quad:canonical}{الصورة النموذجية}: \omterm{def:g10:functions:variations}{متناقصة} على $\intoc{-\infty}{2}$،
\omterm{def:g11:func:monotone}{ومتزايدة} على $\intco{2}{+\infty}$، وقيمتها الصغرى $f(2) = -3$.

\emph{2.} $g = \frac{1}{u}$ حيث $u(x) = (x-2)^2 + 1 = f(x) + 4$. و
الانسحاب بمقدار $4$ يحفظ تغيرات $f$، و $u \geq 1 > 0$. وأخذ
المقلوب يقلبها: إذن $g$ \omterm{def:g11:func:monotone}{متزايدة} على $\intoc{-\infty}{2}$ و
\omterm{def:g10:functions:variations}{متناقصة} على $\intco{2}{+\infty}$، وقيمتها العظمى
$g(2) = \frac{1}{1} = 1$.
\end{solution}

\begin{solution}{exo:g11:func:8}
يصلح أي \omterm{def:g10:functions:graph}{منحنى} على هيئة تلّ يمر بالنقط الثلاث. عندئذٍ:
يمر $f + 2$ بالنقط $(0,3)$ و $(2,5)$ و $(4,3)$، وله التغيرات نفسها
(انسحاب شاقولي)؛ ويمر $-f$ بالنقط $(0,-1)$ و $(2,-3)$ و $(4,-1)$ مع
تغيرات \emph{منقلبة} (فالقمة تصير قاعًا)؛ ويمر $2f$
بالنقط $(0,2)$ و $(2,6)$ و $(4,2)$، وله التغيرات نفسها لكن مضخَّمة
شاقوليًا.
\end{solution}

\begin{solution}{exo:g11:func:9}
$2 - \frac{1}{x+1} = \frac{2(x+1) - 1}{x+1} = \frac{2x+1}{x+1} = f(x)$.
وعلى $\intoo{-1}{+\infty}$، تكون $u(x) = x + 1$ \omterm{def:g11:func:monotone}{متزايدة} وموجبة، إذن
$\frac1u$ \omterm{def:g10:functions:variations}{متناقصة}، و $-\frac1u$ \omterm{def:g11:func:monotone}{متزايدة}، وإضافة $2$ لا تغيّر
شيئًا: إذن $f$ \omterm{def:g11:func:monotone}{متزايدة} تمامًا على $\intoo{-1}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:10}
\emph{1. صحيح.} إذا كان $u < v$، فإن
$(f+g)(v) - (f+g)(u) = \bigl(f(v)-f(u)\bigr) + \bigl(g(v)-g(u)\bigr)$ هو
مجموع عددين غير سالبين.

\emph{2. خطأ.} $f(x) = g(x) = x$ متزايدتان على $\R$، لكن
جداءهما $x^2$ ليس متزايدًا على $\R$ (فهو يتناقص على
$\intoc{-\infty}{0}$). (وتصير العبارة صحيحة إذا كانت الدالتان
غير سالبتين أيضًا.)

\emph{3. خطأ.} المثال المضاد نفسه: \omterm{def:g11:func:function}{فالدالة} $f(x) = x$ \omterm{def:g11:func:monotone}{متزايدة} على $\R$
لكن $f^2(x) = x^2$ ليست كذلك.
\end{solution}

\begin{solution}{exo:g11:func:11}
\emph{1.} من أجل $0 < u < v$:
\[
f(v) - f(u) = (v - u) + \frac1v - \frac1u
= (v - u) + \frac{u - v}{uv}
= (v - u)\left(1 - \frac{1}{uv}\right)
= (v - u)\,\frac{uv - 1}{uv}.
\]

\emph{2.} دائمًا $v - u > 0$ و $uv > 0$. فإذا كان $0 < u < v \leq 1$، فإن
$uv < 1$، إذن $uv - 1 < 0$ و $f(v) < f(u)$: أي أن $f$ \omterm{def:g10:functions:variations}{متناقصة} تمامًا
على $\intoc{0}{1}$. وإذا كان $1 \leq u < v$، فإن $uv > 1$ و $f(v) > f(u)$:
أي أن $f$ \omterm{def:g11:func:monotone}{متزايدة} تمامًا على $\intco{1}{+\infty}$.

\emph{3.} تتناقص $f$ قبل $1$ وتتزايد بعده، إذن
$f(x) \geq f(1) = 2$ من أجل كل $x > 0$، مع تحقق المساواة عند $x = 1$ بالضبط.
(وتعود هذه المتراجحة في صور كثيرة؛ قارن
\cref{ex:g11:quad:canonical} ومتراجحات المماس في
\cref{ch:g12:deriv}.)
\end{solution}

\begin{solution}{pb:g11:func:1}
\textbf{1.} $x^4 - 3x^2$: \omterm{def:g11:func:parity}{زوجية} ($(-x)^4 - 3(-x)^2 = x^4 -
3x^2$). و $x^3 - x$: \omterm{def:g11:func:parity}{فردية}. و $x^2 + x$: لا هذه ولا تلك ($f(-1) = 0 \neq
f(1) = 2$ و $\neq -f(1)$). و $\abs x$: \omterm{def:g11:func:parity}{زوجية}. و $\frac1x$: \omterm{def:g11:func:parity}{فردية}.

\textbf{2.} \omterm{def:g11:func:parity}{الزوجية}: \omterm{def:g10:functions:graph}{المنحنى} متناظر بالنسبة إلى المحور $y$
(مرآة). \omterm{def:g11:func:parity}{والفردية}: متناظر بالنسبة إلى المبدأ (نصف دورة).

\textbf{3.} إذا كانت $f$ و $g$ فرديتين:
$(fg)(-x) = f(-x)g(-x) = (-f(x))(-g(x)) = (fg)(x)$: أي \omterm{def:g11:func:parity}{زوجية}.
والجدول: \omterm{def:g11:func:parity}{زوجية} $\times$ \omterm{def:g11:func:parity}{زوجية} $=$ \omterm{def:g11:func:parity}{زوجية}؛ \omterm{def:g11:func:parity}{وزوجية} $\times$ \omterm{def:g11:func:parity}{فردية} $=$ \omterm{def:g11:func:parity}{فردية}؛
\omterm{def:g11:func:parity}{وفردية} $\times$ \omterm{def:g11:func:parity}{فردية} $=$ \omterm{def:g11:func:parity}{زوجية} --- إنها قاعدة الإشارات بالزوجيات.

\textbf{4.} \omterm{def:g11:func:parity}{الفردية} عند $0$: $f(0) = -f(0)$، إذن $f(0) = 0$. \omterm{def:g11:func:parity}{والزوجية}:
$f(-3) = f(3) = 7$. \omterm{def:g11:func:parity}{والزوجية} \omterm{def:g11:func:parity}{والفردية} معًا:
$f(x) = f(-x) = -f(x)$ من أجل كل $x$، إذن $2f(x) = 0$: أي أن $f$ هي
\omterm{def:g11:func:function}{الدالة} المعدومة.

\textbf{5.} النصف الأيمن ($x \geq 0$) يحدد \omterm{def:g11:func:parity}{الدالة الزوجية}
تحديدًا كاملًا: فالنصف الأيسر صورته بالمرآة. أما ما ليس زوجيًا
ولا فرديًا: فمثاله $x^2 + x$ (السؤال 1) --- إذ يسقط تناظراه عند
$x = 1$.

\textbf{6.} $E(x) = \frac{(x^3 + x^2 + 1) + (-x^3 + x^2 +
1)}{2} = x^2 + 1$ (\omterm{def:g11:func:parity}{زوجية})، و $O(x) = \frac{(x^3 + x^2 + 1) -
(-x^3 + x^2 + 1)}{2} = x^3$ (\omterm{def:g11:func:parity}{فردية})؛ ومجموعهما $f$.

\textbf{7.} $E(-x) = \frac{f(-x) + f(x)}{2} = E(x)$: \omterm{def:g11:func:parity}{زوجية}.
و $O(-x) = \frac{f(-x) - f(x)}{2} = -O(x)$: \omterm{def:g11:func:parity}{فردية}. و
$E(x) + O(x) = f(x)$ بالجمع المباشر: فكل \omterm{def:g11:func:function}{دالة}
تنقسم.

\textbf{8.} من $E_1 + O_1 = E_2 + O_2$:
$E_1 - E_2 = O_2 - O_1$. والطرف الأيسر زوجي والأيمن
فردي --- إذن هذه \omterm{def:g11:func:parity}{الدالة زوجية} \omterm{def:g11:func:parity}{وفردية} معًا، أي معدومة (السؤال 4):
فيكون $E_1 = E_2$ و $O_1 = O_2$. إذن الانقسام وحيد.

\textbf{9.} $E(x) = \frac12\left(\frac{1}{1 - x} + \frac{1}{1 +
x}\right) = \frac12 \cdot \frac{(1 + x) + (1 - x)}{1 - x^2}
= \frac{1}{1 - x^2}$، و
$O(x) = \frac12 \cdot \frac{(1 + x) - (1 - x)}{1 - x^2}
= \frac{x}{1 - x^2}$. وتحقق: $E$ \omterm{def:g11:func:parity}{زوجية} و $O$ \omterm{def:g11:func:parity}{فردية}، و
مجموعهما $\frac{1 + x}{1 - x^2} =
\frac{1 + x}{(1 - x)(1 + x)} = \frac{1}{1 - x} = f(x)$.

\textbf{10.} لأن كل قطعة تخضع لتناظر، يكفي نصف
العمل (ونصف المعطيات) لكل منهما: فتحل على جهة واحدة، ثم
تعكس على الأخرى --- كما أن عمليات كثيرة (التكامل، و
المتسلسلات، وتحليل الإشارات) تعالج القطع المتناظرة على نحو أبسط
بكثير من الدوال الخام.

\textbf{11.} $(x-3)^2$: منسحبة $3$ نحو اليمين. و $x^2 + 2$: منسحبة
$2$ نحو الأعلى. و $(x-3)^2 + 2$: الأمران معًا. و $-x^2$: مقلوبة رأسًا على عقب.
و $2x^2$: ممدَّدة شاقوليًا بمعامل $2$ (فالوعاء أضيق).

\textbf{12.} من أجل $x \geq 2$: $(x - 2) + (x + 2) = 2x$. ومن أجل
$-2 \leq x \leq 2$: $(2 - x) + (x + 2) = 4$. ومن أجل $x \leq -2$:
$-2x$. فينزل \omterm{def:g10:functions:graph}{المنحنى} \omterm{def:g10:reffunc:affine}{بميل} $-2$، ويسير مسطحًا على الارتفاع
$4$ بين $-2$ و $2$، ثم يتسلق \omterm{def:g10:reffunc:affine}{بميل} $2$: أي واد
ذو أرضية مسطحة.

\textbf{13.} الانسحاب الشاقولي: تبقى $f(x) + 2$ \omterm{def:g11:func:parity}{زوجية}
($f(-x) + 2 = f(x) + 2$). والتمديد الشاقولي: تبقى $2f(x)$
\omterm{def:g11:func:parity}{زوجية}. أما الانسحاب الأفقي: فإن $(x - 3)^2$ ليست \omterm{def:g11:func:parity}{زوجية}
($f(-1) = 16 \neq 4 = f(1)$): إذ يهدم الانزلاق الجانبي
المرآة.

\textbf{14.} على الأرضية المسطحة تكون القيمة $4 \neq 6$. ومن أجل
$x \geq 2$: $2x = 6$ فيكون $x = 3$. ومن أجل $x \leq -2$: $-2x = 6$ فيكون
$x = -3$. إذن الحلول: $\pm 3$.

\textbf{15.} $f(x) = a(x - 2)^2 - 3$ حيث $f(0) = 4a - 3 = 1$:
فيكون $a = 1$، إذن $f(x) = (x - 2)^2 - 3$.

\textbf{16.} $g(v) - g(u) = (v - u) + 4\,\frac{u - v}{uv}
= (v - u)\,\frac{uv - 4}{uv}$: سالب من أجل $u < v$ مع
$uv < 4$، وموجب مع $uv > 4$: إذن \omterm{def:g10:functions:variations}{متناقصة} على
$\intoc{0}{2}$، \omterm{def:g11:func:monotone}{ومتزايدة} على $\intco{2}{+\infty}$؛ \omterm{def:g10:functions:extrema}{والقيمة الصغرى} عند
$x = 2$، وقيمتها $g(2) = 4$.

\textbf{17.} الضلعان $x$ و $\frac{16}{x}$:
$P(x) = 2\left(x + \frac{16}{x}\right)$، ويُصغَّر (بآلة السؤال
16 مع $4 \to 16$، ونقطة الانعطاف $\sqrt{16} = 4$)
عند $x = 4$: أي \emph{المربع} $4 \times 4$، ومحيطه $16$ ---
وهو جواب ديدو، مبرهنًا عليه الآن من أجل كل طول ضلع حقيقي.

\textbf{18.} على $\intco{0}{+\infty}$: تتزايد $x \mapsto x^2 + 1$،
وتتزايد $\sqrt{\phantom{x}}$، إذن تتزايد
\omterm{def:g11:func:function}{الدالة} المركبة. و $h(-x) = \sqrt{x^2 + 1} = h(x)$: أي أن $h$
\omterm{def:g11:func:parity}{زوجية}، فتتناقص بالتناظر المرآتي على
$\intoc{-\infty}{0}$. \omterm{def:g10:functions:extrema}{والقيمة الصغرى} $h(0) = 1$.

\textbf{19.} مربع المسافة من النقطة $(x, \sqrt x)$ إلى $(3, 0)$:
$d^2(x) = (x - 3)^2 + x = x^2 - 5x + 9$، وهو أصغري عند
$x = \frac52$: فأقرب نقطة
$\left(\frac52, \sqrt{\frac52}\right)$، والمسافة
$\sqrt{\frac{11}{4}} = \frac{\sqrt{11}}{2}$. وتصغير $d^2$
مشروع لأن التربيع متزايد تمامًا على
الأعداد الموجبة: فتبلغ $d$ و $d^2$ قاعهما في الموضع نفسه.

\textbf{20.} المادة الخام: المنحنيات المرجعية، محفوظة عن ظهر قلب.
والآلات: الانسحابات والقلوب والتمديدات التي تركّب منحنيات جديدة
من قديمة. وضبط الجودة: \omterm{def:g11:func:parity}{الزوجية} والانقسام إلى زوجي وفردي،
وهما ينصّفان كل دراسة. ومنضدة القياس: تحليل
$f(v) - f(u)$ إلى عوامل، وهو يقرأ التغيرات والقيم الصغرى مباشرة. أما
الأداة القوية في الفصل التالي فهي \emph{المشتقة}، التي تقرؤها
بنظرة.
\end{solution}
