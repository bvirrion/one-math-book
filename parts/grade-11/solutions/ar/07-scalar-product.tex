\section*{الفصل \ref{ch:g11:scal} --- الجداء السلّمي في المستوي}

\begin{solution}{exo:g11:scal:1}
$(2,5)\cdot(3,-1) = 6 - 5 = 1$.

$(1,-4)\cdot(8,2) = 8 - 8 = 0$ (\omterm{def:g11:scal:orthogonal}{متجهتان متعامدتان}).

$\vec u\cdot\vec v = 3 \times 4 \times \cos\frac\pi3 = 12 \times \frac12
= 6$.
\end{solution}

\begin{solution}{exo:g11:scal:2}
$\vect{AB}\cdot\vect{AC} = 6 \times 6 \times \cos 60^\circ = 18$.

تصنع المتجهتان $\vect{AB}$ و $\vect{BC}$ زاوية قدرها
$180^\circ - 60^\circ = 120^\circ$ (ضعهما بنفس المبدأ عند $B$)، إذن
$\vect{AB}\cdot\vect{BC} = 36 \cos 120^\circ = -18$.
\end{solution}

\begin{solution}{exo:g11:scal:3}
$\vec u \cdot \vec v = 3(t - 8) + t = 4t - 24 = 0$ يعطي $t = 6$.

$\vec u - \vec v = (t - 4,\ 0)$، إذن
$\vec u \cdot (\vec u - \vec v) = t(t-4) + 0 = t^2 - 4t$، وهذا ينعدم
من أجل $t = 0$ أو $t = 4$.
\end{solution}

\begin{solution}{exo:g11:scal:4}
$\vec u \cdot \vec v = 3 + 2 = 5$، و $\norm{\vec u} = \sqrt5$،
و $\norm{\vec v} = \sqrt{10}$، إذن
\[
\cos\theta = \frac{5}{\sqrt5\,\sqrt{10}} = \frac{5}{5\sqrt2}
= \frac{\sqrt2}{2},
\]
ومنه $\theta = 45^\circ$ بالضبط.
\end{solution}

\begin{solution}{exo:g11:scal:5}
$a^2 = 16 + 36 - 2 \times 4 \times 6 \times \cos\frac{2\pi}{3}
= 52 - 48 \times \left(-\frac12\right) = 76$، إذن $a = \sqrt{76} =
2\sqrt{19} \approx 8.7$.

أما في المثلث الثاني، فقانون \omterm{def:g11:trigo:cossin}{جيب التمام} محلولًا بالنسبة إلى الزاوية يعطي
\[
\cos\widehat A = \frac{b^2 + c^2 - a^2}{2bc}
= \frac{25 + 16 - 49}{2 \times 5 \times 4} = -\frac{8}{40} = -\frac15,
\]
وهو سالب: إذن $\widehat A$ منفرجة.
\end{solution}

\begin{solution}{exo:g11:scal:6}
$\vect{AB}\,(4, -2)$ و $\vect{AC}\,(2, 4)$:
$\vect{AB}\cdot\vect{AC} = 8 - 8 = 0$، إذن الزاوية عند $A$ قائمة.
والأضلاع: $AB = \sqrt{16 + 4} = \sqrt{20}$، و $AC = \sqrt{4 + 16} =
\sqrt{20}$، و $\vect{BC}\,(-2, 6)$ يعطي $BC = \sqrt{40}$. وللتحقق من الاتساق:
$BC^2 = 40 = 20 + 20 = AB^2 + AC^2$، وهو قانون \omterm{def:g11:trigo:cossin}{جيب التمام} مع
$\cos\widehat A = 0$ (فيثاغورس).
\end{solution}

\begin{solution}{exo:g11:scal:7}
الصورة بالمركز ونصف القطر: $(x + 2)^2 + (y - 3)^2 = 25$.

أما الدائرة الثانية فأكمل المربعات:
\[
x^2 - 6x + y^2 + 2y = 6
\iff (x - 3)^2 - 9 + (y + 1)^2 - 1 = 6
\iff (x-3)^2 + (y+1)^2 = 16 :
\]
المركز $(3, -1)$ ونصف القطر $4$.
\end{solution}

\begin{solution}{exo:g11:scal:8}
تقع $M(x,y)$ على الدائرة عندما يكون $\vect{MA}\cdot\vect{MB} = 0$ مع
$\vect{MA}\,(-3 - x,\ -y)$ و $\vect{MB}\,(-x,\ 2 - y)$:
\[
(-3 - x)(-x) + (-y)(2 - y) = x^2 + 3x + y^2 - 2y = 0 .
\]
وأما المبدأ: $0 + 0 + 0 - 0 = 0$، إذن $O$ تقع على الدائرة ---
وهندسيًا، المتجهتان $\vect{OA}\,(-3,0)$ و $\vect{OB}\,(0,2)$ متعامدتان،
فترى $O$ القطعة $[AB]$ تحت زاوية قائمة. (وبإكمال المربعات:
المركز $\left(-\frac32, 1\right)$، ونصف القطر $\frac{\sqrt{13}}{2}$.)
\end{solution}

\begin{solution}{exo:g11:scal:9}
للعمود المار بالنقطة $P$ \omterm{def:g11:vect:direction}{متجهة توجيه} المستقيم $d$، أي
$(-1, 2)$، متجهةً ناظمة: فمعادلته $-x + 2y + c = 0$؛ وبالمرور
بالنقطة $P(1,3)$: $-1 + 6 + c = 0$، أي $c = -5$، فتكون $x - 2y + 5 = 0$.

\omterm{def:g10:numbers:interunion}{والتقاطع} مع $d$: من $2x + y = 7$ نجد $y = 7 - 2x$؛ وبالتعويض،
$x - 2(7 - 2x) + 5 = 5x - 9 = 0$، إذن $x = \frac95$ و $y = \frac{17}5$:
$H\left(\frac95, \frac{17}5\right)$. ثم
$\vect{PH}\left(\frac45, \frac25\right)$ و
\[
PH = \sqrt{\frac{16}{25} + \frac{4}{25}} = \frac{\sqrt{20}}{5}
= \frac{2\sqrt5}{5} \approx 0.89 .
\]
\end{solution}

\begin{solution}{exo:g11:scal:10}
بجمع
\[
\norm{\vec u + \vec v}^2 = \norm{\vec u}^2 + 2\vec u\cdot\vec v +
\norm{\vec v}^2
\quad\text{و}\quad
\norm{\vec u - \vec v}^2 = \norm{\vec u}^2 - 2\vec u\cdot\vec v +
\norm{\vec v}^2,
\]
تختصر الحدود المختلطة ويكون
$\norm{\vec u + \vec v}^2 + \norm{\vec u - \vec v}^2 = 2\norm{\vec u}^2 +
2\norm{\vec v}^2$. وفي متوازي أضلاع ضلعاه $\vec u$ و $\vec v$،
يكون القطران $\vec u + \vec v$ و $\vec u - \vec v$: فمجموع
مربعي القطرين يساوي مجموع مربعات الأضلاع
الأربعة.
\end{solution}

\begin{solution}{exo:g11:scal:11}
\emph{1.} بما أن $I$ هي \omterm{prop:g10:coordgeom:midpoint}{المنتصف}، فإن $\vect{IB} = -\vect{IA}$ و
$IA = 2$. ثم
\[
\vect{MA}\cdot\vect{MB}
= (\vect{MI} + \vect{IA})\cdot(\vect{MI} - \vect{IA})
= \norm{\vect{MI}}^2 - \norm{\vect{IA}}^2
= MI^2 - 4 .
\]

\emph{2.} يصير الشرط $\vect{MA}\cdot\vect{MB} = 12$ هو
$MI^2 = 16$، أي $MI = 4$: فالمجموعة هي الدائرة التي مركزها $I$ ونصف
قطرها $4$.
\end{solution}

\begin{solution}{pb:g11:scal:1}
\textbf{1.} $\vec u \cdot \vec v = -3 + 8 = 5$؛
و $\norm{\vec u} = 5$، و $\norm{\vec v} = \sqrt5$؛
و $\cos\theta = \frac{5}{5\sqrt5} = \frac{1}{\sqrt5}$:
$\theta \approx 63.4^\circ$.

\textbf{2.} $6 - 4k = 0$: إذن $k = \frac32$.

\textbf{3.} $W = 50 \times 10 \times \cos 60^\circ = 250$
جولًا: فوحده المركّبة الموازية للحركة تشتغل.

\textbf{4.} $c^2 = 25 + 49 - 2 \times 35 \times \frac12 = 39$:
إذن $c = \sqrt{39}$.

\textbf{5.} $\cos\gamma = \frac{16 + 36 - 64}{2 \times 4
\times 6} = -\frac14$: سالب، فالزاوية منفرجة:
$\gamma \approx 104.5^\circ$.

\textbf{6.} \omterm{def:g10:coordgeom:system}{بالإحداثيات}: $\vec u \cdot \vec v = \cos a \cos b +
\sin a \sin b$. وهندسيًا: متجهتان واحديتان،
$\vec u \cdot \vec v = 1 \times 1 \times \cos(a - b)$.
وبمساواة الحسابين للعدد نفسه:
$\cos(a - b) = \cos a \cos b + \sin a \sin b$.

\textbf{7.} $\cos(a + b) = \cos(a - (-b)) = \cos a \cos(-b) +
\sin a \sin(-b) = \cos a \cos b - \sin a \sin b$.

\textbf{8.} $\sin(a + b) = \cos\left(\frac\pi2 - a - b\right)
= \cos\left(\left(\frac\pi2 - a\right) - b\right)
= \cos\left(\frac\pi2 - a\right)\cos b +
\sin\left(\frac\pi2 - a\right)\sin b
= \sin a \cos b + \cos a \sin b$.

\textbf{9.} $\cos 15^\circ = \cos 45^\circ \cos 30^\circ +
\sin 45^\circ \sin 30^\circ = \frac{\sqrt2}{2} \cdot
\frac{\sqrt3}{2} + \frac{\sqrt2}{2} \cdot \frac12 =
\frac{\sqrt6 + \sqrt2}{4} \approx 0.966$. و
$\sin 75^\circ = \cos 15^\circ$: القيمة نفسها.

\textbf{10.} $b = a$ في السؤالين 7 و 8:
$\cos 2a = \cos^2 a - \sin^2 a = 2\cos^2 a - 1 = 1 - 2\sin^2 a$
(باستعمال $\cos^2 + \sin^2 = 1$)، و
$\sin 2a = 2 \sin a \cos a$.

\textbf{11.} $\sin^2 15^\circ = \frac{1 - \cos 30^\circ}{2}
= \frac{2 - \sqrt3}{4}$، إذن
$\sin 15^\circ = \frac{\sqrt{2 - \sqrt3}}{2}$. وبتربيع
$\frac{\sqrt6 - \sqrt2}{4}$:
$\frac{6 - 2\sqrt{12} + 2}{16} = \frac{8 - 4\sqrt3}{16} =
\frac{2 - \sqrt3}{4}$: العدد نفسه في ثوبين.

\textbf{12.} تتفكك $\vect{AP}$ إلى جزء موازٍ للمستقيم $d$
(لا تراه $\vec n$) وجزء عمودي على $d$، طوله هو
المسافة المطلوبة؛ والضرب سلّميًا في \omterm{def:g11:vect:vector}{المتجهة} الناظمة الواحدية
$\frac{\vec n}{\norm{\vec n}}$ يقتل الجزء الأول
ويقيس الثاني. ومع $A$ على $d$:
$\vect{AP} \cdot \vec n = a x_0 + b y_0 - (a x_A + b y_A) =
a x_0 + b y_0 + c$، ومنه الصيغة. مثال:
$\frac{\abs{15 + 24 - 12}}{5} = \frac{27}{5} = 5.4$.

\textbf{13.} نصف القطر $= 5.4$:
$(x - 5)^2 + (y - 6)^2 = 29.16$.

\textbf{14.} المسافة $= \frac{\abs{-12}}{5} = 2.4$. وأما المسقط:
فالعمود المار بالمبدأ هو $(3t, 4t)$؛ وعلى المستقيم:
$9t + 16t = 12$، أي $t = \frac{12}{25}$: فالمسقط
$\left(\frac{36}{25}, \frac{48}{25}\right)$، على مسافة
$5t = \frac{60}{25} = 2.4$. فالصيغة والحساب الصريح
متفقان.

\textbf{15.} القاعدة $AB = 6$، والارتفاع $=$ المسافة من $C$ إلى
$(AB)$ (وهو المحور $y = 0$): أي $5$؛ والمساحة $15$. والمستقيم $(AC)$:
$5x - 2y = 0$، و
$\operatorname{dist}(B, (AC)) = \frac{\abs{30}}{\sqrt{29}}$؛
ثم $\frac12 \times AC \times \frac{30}{\sqrt{29}} =
\frac12 \times \sqrt{29} \times \frac{30}{\sqrt{29}} = 15$:
المساحة نفسها، بأي من الارتفاعين.

\textbf{16.} $\operatorname{dist} = \frac{\abs{2 + 1 - 6}}
{\sqrt2} = \frac{3}{\sqrt2} \approx 2.12 > 2$: فالمستقيم يمر
بعيدًا عن الدائرة --- إذن منفصلان.

\textbf{17.} أدخل $I$:
$\vect{MA} \cdot \vect{MB} = (\vect{MI} + \vect{IA}) \cdot
(\vect{MI} + \vect{IB}) = MI^2 + \vect{MI} \cdot (\vect{IA} +
\vect{IB}) + \vect{IA} \cdot \vect{IB}$. ويموت الحد الأوسط
($\vect{IA} + \vect{IB} = \vec 0$)، و
$\vect{IA} \cdot \vect{IB} = -IA^2 = -\frac{AB^2}{4}$
(متجهتان متعاكستان طولهما $\frac{AB}2$).

\textbf{18.} $\vect{MA} \cdot \vect{MB} = 0 \iff MI^2 =
\frac{AB^2}{4} \iff MI = \frac{AB}{2}$: وهي الدائرة التي مركزها
$I$ ونصف قطرها $\frac{AB}2$ --- أي الدائرة ذات القطر
$[AB]$.

\textbf{19.} \omterm{def:g10:algebra:expand}{بنشر} كل زوج انطلاقًا من $H$ ($\vect{BC} =
\vect{HC} - \vect{HB}$، وهكذا):
\[
\vect{HA} \cdot \vect{HC} - \vect{HA} \cdot \vect{HB}
+ \vect{HB} \cdot \vect{HA} - \vect{HB} \cdot \vect{HC}
+ \vect{HC} \cdot \vect{HB} - \vect{HC} \cdot \vect{HA} = 0 :
\]
يختصر كل شيء أزواجًا --- فالمتطابقة صحيحة من أجل
\emph{أي} أربع نقط. وفي حالة $H$ التي عندنا، ينعدم الجداءان
السلّميان الأولان بالفرض، إذن
$\vect{HC} \cdot \vect{AB} = 0$: أي أن $H$ تقع على الارتفاع الصادر من
$C$. ثلاثة ارتفاعات، ونقطة واحدة.

\textbf{20.} جاءت صيغ الجمع من حساب \omterm{def:g11:scal:dot}{جداء سلّمي} واحد
بالوجه \emph{الإحداثي} وبوجه
\emph{المعيارين والزاوية}؛ وجاءت صيغة المسافة من
وجه \emph{الإسقاط}؛ وجاء \omterm{pb:g11:scal:1}{مركز التعامد} من
وجه \emph{الخطية الثنائية} (شال \omterm{def:g10:algebra:expand}{والنشر}). احسب المقدار
نفسه مرتين، وساوِ بين الحسابين، فتسقط مبرهنة --- وهذه هي صنعة \omterm{def:g11:scal:dot}{الجداء
السلّمي} كلها.
\end{solution}
