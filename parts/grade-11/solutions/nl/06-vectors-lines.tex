\section*{Hoofdstuk \ref{ch:g11:vect} --- Vectoren en rechten in het vlak}

\begin{solution}{exo:g11:vect:1}
$\vect{AB}\,(3, 4)$, $\vect{AC}\,(-3, 2)$,
$\vect{AB} + \vect{AC}\,(0, 6)$,
$2\vect{AB} - 3\vect{AC}\,(6 + 9,\ 8 - 6) = (15, 2)$. Het \omterm{prop:g10:coordgeom:midpoint}{midden} van $[BC]$
is $\left(\frac{5 + (-1)}{2}, \frac{3+1}{2}\right) = (2, 2)$.
\end{solution}

\begin{solution}{exo:g11:vect:2}
$\det = 4 \times 3 - (-2) \times (-6) = 12 - 12 = 0$: \omterm{def:g11:vect:collinear}{collineair} (en
inderdaad is $\vec v = -\frac12 \vec u$).

$\det = 2 \times 7 - 3 \times 5 = -1 \neq 0$: niet \omterm{def:g11:vect:collinear}{collineair}.

$\det = 0 \times (-8) - 0 \times 3 = 0$: \omterm{def:g11:vect:collinear}{collineair} (allebei verticaal).
\end{solution}

\begin{solution}{exo:g11:vect:3}
Door $A(2,1)$ met richting $(3, -1)$: uit $(-b, a) = (3, -1)$ volgt
$a = -1$ en $b = -3$, dus $-x - 3y + c = 0$; $A$ invullen geeft
$-2 - 3 + c = 0$, dus $c = 5$. Vermenigvuldigd met $-1$: $x + 3y - 5 = 0$.

Door $B(0,4)$ en $C(2,0)$: $\vect{BC}\,(2, -4)$, dus $a = -4$ en $b = -2$:
$-4x - 2y + c = 0$; $B$ invullen geeft $-8 + c = 0$, dus $c = 8$.
Vereenvoudigd door $-2$: $2x + y - 4 = 0$. Controle met $C$:
$4 + 0 - 4 = 0$.
\end{solution}

\begin{solution}{exo:g11:vect:4}
Een \omterm{def:g11:vect:direction}{richtingsvector} is $(-b, a) = (2, 3)$. Snijpunten: $x = 0$ geeft
$y = 3$, het punt $(0, 3)$; $y = 0$ geeft $x = -2$, het punt $(-2, 0)$. Voor
$P(2,6)$: $3 \times 2 - 2 \times 6 + 6 = 0$, dus $P \in d$. Voor $Q(1,4)$:
$3 - 8 + 6 = 1 \neq 0$, dus $Q \notin d$.
\end{solution}

\begin{solution}{exo:g11:vect:5}
$\vect{AB}\,(3, 6)$ en $\vect{AC}\,(-3, -6)$:
$\det = 3 \times (-6) - (-3) \times 6 = 0$, dus liggen $A$, $B$ en $C$ op
één rechte.

$\vect{DE}\,(2, 3)$ en $\vect{DF}\,(5, 8)$:
$\det = 2 \times 8 - 5 \times 3 = 1 \neq 0$: $D$, $E$ en $F$ liggen niet op
één rechte.
\end{solution}

\begin{solution}{exo:g11:vect:6}
Eerste paar: de twee \omterm{def:g10:algebra:equation}{vergelijkingen} optellen geeft
$(2x + y - 7) + (x - y + 1) = 3x - 6 = 0$, dus $x = 2$ en daarna
$y = x + 1 = 3$: ze ontmoeten elkaar in $(2, 3)$ (controle:
$4 + 3 - 7 = 0$).

Tweede paar: $ab' - a'b = 1 \times 4 - (-2) \times (-2) = 0$, dus zijn de
rechten evenwijdig; $e$ met $-2$ vermenigvuldigen geeft
$-2x + 4y - 6 = 0$, wat van $e'$ verschilt ($-6 \neq 1$): strikt
evenwijdig, geen snijpunt.
\end{solution}

\begin{solution}{exo:g11:vect:7}
Voor $d: 2x - y + 3 = 0$: richting $(-b, a) = (1, 2)$, en het punt $(0, 3)$
ligt op $d$, dus $d = \{(t,\ 3 + 2t) : t \in \R\}$.

Voor de parametrische rechte door $(1, -3)$ met richting $(2, 1)$: $a = 1$
en $b = -2$, dus $x - 2y + c = 0$; $(1, -3)$ invullen geeft
$1 + 6 + c = 0$, dus $c = -7$. \omterm{def:g10:algebra:equation}{Vergelijking}: $x - 2y - 7 = 0$ (controle met
$t = 1$, het punt $(3, -2)$: $3 + 4 - 7 = 0$).
\end{solution}

\begin{solution}{exo:g11:vect:8}
Een evenwijdige rechte heeft dezelfde $(a, b)$: $4x - 3y + c = 0$; door
$P(1, -2)$: $4 + 6 + c = 0$, dus $c = -10$ en de rechte is
$4x - 3y - 10 = 0$.

Evenwijdigheid van $mx + 2y - 5 = 0$ met $d$ vereist
$ab' - a'b = 4 \times 2 - m \times (-3) = 8 + 3m = 0$, dus
$m = -\frac83$.
\end{solution}

\begin{solution}{exo:g11:vect:9}
Het \omterm{prop:g10:coordgeom:midpoint}{midden} van $[AB]$ is $M\!\left(\frac{-1+3}{2}, \frac{0+2}{2}\right) =
(1, 1)$. De zwaartelijn is de rechte $(CM)$ met $C(1, -4)$: beide punten
hebben \omterm{def:g10:coordgeom:system}{abscis} $1$, dus is de zwaartelijn de verticale rechte
\[
x - 1 = 0 .
\]
\end{solution}

\begin{solution}{exo:g11:vect:10}
$\det(\vec u, \vec v) = m(m - 1) - 3 \times 2 = m^2 - m - 6$. Die verdwijnt
wanneer $m^2 - m - 6 = 0$: $\Delta = 1 + 24 = 25$, met \omterm{def:g11:quad:discriminant}{wortels}
$m = \frac{1 \pm 5}{2}$, dus $m = 3$ of $m = -2$.
\end{solution}

\begin{solution}{exo:g11:vect:11}
\emph{1.} $I = \left(\frac12, 0\right)$. Dan is
$\vect{DI} = \left(\frac12 - 0,\ 0 - 1\right) = \left(\frac12, -1\right)$ en
$\vect{DJ} = \frac23\vect{DI} = \left(\frac13, -\frac23\right)$, dus
\[
J = D + \vect{DJ} = \left(\frac13,\ \frac13\right).
\]

\emph{2.} $\vect{AJ}\left(\frac13, \frac13\right)$ en $\vect{AC}\,(1, 1)$:
$\det = \frac13 \times 1 - 1 \times \frac13 = 0$, dus liggen $A$, $J$ en $C$
op één rechte --- er geldt zelfs $\vect{AJ} = \frac13\vect{AC}$: de rechte
$(DI)$ snijdt de diagonaal $(AC)$ op precies een derde van haar lengte
vanaf $A$.
\end{solution}

\begin{solution}{pb:g11:vect:1}
\textbf{1.} $x = 1 + 3t$, $y = 2 - t$. Uit de tweede volgt $t = 2 - y$;
invullen geeft $x = 1 + 3(2 - y) = 7 - 3y$: de cartesische \omterm{def:g10:algebra:equation}{vergelijking}
$x + 3y - 7 = 0$.

\textbf{2.} Voor $ax + by + c = 0$ is $(-b, a)$ een \omterm{def:g11:vect:direction}{richtingsvector}: hier
$(5, 2)$. Een punt: $y = 1$ geeft $x = 1$, dus $(1, 1)$.
Parametervoorstelling: $x = 1 + 5t$, $y = 1 + 2t$.

\textbf{3.} $x = 7 - 3y$ invullen in $2x - 5y + 3 = 0$ geeft
$14 - 6y - 5y + 3 = 0$, dus $y = \frac{17}{11}$ en
$x = 7 - \frac{51}{11} = \frac{26}{11}$: het punt
$\left(\frac{26}{11}, \frac{17}{11}\right)$.

\textbf{4.} Richtingen $(5, 2)$ en $(3, -1)$:
$\det = 5 \times (-1) - 2 \times 3 = -11 \neq 0$: snijdend, zoals gevonden.
Voor het tweede paar: de richtingen $(5, 2)$ en $(10, 4)$ hebben
$\det = 0$ (evenwijdig), en $2x - 5y + 3 = 0$ verdubbelen geeft
$-4x + 10y - 6 = 0 \neq -4x + 10y + 1 = 0$: strikt evenwijdig.

\textbf{5.} $\vect{AB}(3, 2)$, $\vect{AC}(9, 6)$: $\det = 18 - 18 = 0$: op
één rechte.

\textbf{6.} $P$: uit $x = 2t$ en $y = 3 + t$ volgt $y = 3 + \frac x2$. $Q$:
uit $x = 10 - t$ volgt $t = 10 - x$, dus $y = 2(10 - x) - 1 = 19 - 2x$.
Kruising: $3 + \frac x2 = 19 - 2x$ geeft $x = 6.4$ en $y = 6.2$: de banen
kruisen elkaar in $(6.4,\ 6.2)$.

\textbf{7.} $P$ passeert daar wanneer $2t = 6.4$: $t = 3.2$ u. $Q$ wanneer
$10 - t = 6.4$: $t = 3.6$ u. Vierentwintig minuten uit elkaar: geen
aanvaring. Waarschuwing: een kaart toont \emph{banen}; alleen de
parametrische klokken zeggen of twee bewegingen op hetzelfde ogenblik
hetzelfde punt bezetten.

\textbf{8.} $D^2(t) = (3t - 10)^2 + (4 - t)^2 = 10t^2 - 68t + 116$: top in
$t = \frac{68}{20} = 3.4$ u, waar $D^2 = 0.4$: de kleinste nadering is
$D = \sqrt{0.4} \approx 0.63$ mijl, in $t = 3.4$.

\textbf{9.} Geschonden: $0.63 < 1$. $D^2(t) < 1$ luidt
$10t^2 - 68t + 115 < 0$: $\Delta = 4624 - 4600 = 24$, met \omterm{def:g11:quad:discriminant}{wortels}
$\frac{68 \pm \sqrt{24}}{20} = 3.4 \pm 0.245$: de schepen zitten te dicht
bij elkaar van $t \approx 3.16$ tot $t \approx 3.64$ --- ongeveer $29$
minuten inbreuk. Een van beide moet van koers veranderen.

\textbf{10.} Elke coördinaat van elk schip is een \emph{affiene} \omterm{def:g11:func:function}{functie} van
$t$, dus is elk \omterm{def:g11:seq:arithmetic}{verschil} van \omterm{def:g10:coordgeom:system}{coördinaten} affien, en is $D^2$ --- een som van
twee gekwadrateerde \omterm{def:g10:reffunc:affine}{affiene functies} --- kwadratisch (met niet-negatieve
kopcoëfficiënt). De kleinste nadering lees je dus altijd van een top af.

\textbf{11.} $P$ ontmoeten in $t = 3$, dus in $(6, 6)$: $(3a, 3b) = (6, 6)$
geeft $(a, b) = (2, 2)$, met snelheid $\sqrt{8} \approx 2.8$ knopen.

\textbf{12.} $D(0, 1)$ en $I\left(\frac12, 0\right)$: richting
$\left(\frac12, -1\right)$, dus $(1, -2)$: \omterm{def:g10:algebra:equation}{vergelijking} $2x + y = 1$.

\textbf{13.} $(AC)$: $y = x$. Dan geeft $2x + x = 1$ dat $x = \frac13$: het
punt $\left(\frac13, \frac13\right)$, op een derde van de weg van $A$ naar
$C$.

\textbf{14.} $B(1, 0)$ en $K\left(\frac12, 1\right)$: richting
$\left(-\frac12, 1\right)$, \omterm{def:g10:algebra:equation}{vergelijking} $2x + y = 2$. Met $y = x$:
$x = \frac23$: het punt $\left(\frac23, \frac23\right)$. De twee
hoektransversalen snijden $(AC)$ in haar twee derdepunten: de stelling is
bewezen, twee keer drie regels rekenwerk.

\textbf{15.} Elk parallellogram wordt op het eenheidsvierkant afgebeeld door
$A$ als oorsprong en $\vect{AB}$ en $\vect{AD}$ als de eenheden van de twee
assen te kiezen. Die keuze behoudt alles waarover de stelling spreekt ---
\omterm{prop:g10:coordgeom:midpoint}{middens}, het op één rechte liggen, en verhoudingen langs een gegeven rechte
(alle uit te drukken met \omterm{def:g11:vect:vector}{vectoren} en scalairen) --- zodat de uitspraak in het
vierkant bewijzen ze voor elk parallellogram bewijst. (Lengten of hoeken zou
ze \emph{niet} overdragen: die vervormt het \omterm{def:g10:coordgeom:system}{assenstelsel}.)

\textbf{16.} $E\left(1, \frac12\right)$: de rechte $(AE)$ is
$y = \frac x2$. De diagonaal $(DB)$ loopt van $(0,1)$ naar $(1,0)$:
$y = 1 - x$. Snijpunt: $\frac x2 = 1 - x$ geeft $x = \frac23$: het punt
$\left(\frac23, \frac13\right)$ --- opnieuw een derdepunt, nu van de andere
diagonaal. Hoektransversalen naar \omterm{prop:g10:coordgeom:midpoint}{middens} houden van derden.

\textbf{17.} De eerste twee: $x + y = 4$ en $2x - y = 2$ geven $3x = 6$:
$(2, 2)$. De derde: $2 - 4 = -2$: voldaan. Ze gaan alle drie door $(2, 2)$.

\textbf{18.} $mx - y + 2 - 3m = m(x - 3) + (2 - y) = 0$: opdat de
\omterm{def:g10:algebra:equation}{vergelijking} voor \emph{elke} $m$ zou gelden, neem je $x = 3$ en $y = 2$ ---
en inderdaad bevat elke $d_m$ het punt $(3, 2)$. Een bundel rechten door één
punt, één rechte per \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $m$.

\textbf{19.} Evenwijdig met $y = 2x + 7$: \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $m = 2$, dus
de rechte $2x - y - 4 = 0$. Door de oorsprong: $2 - 3m = 0$, dus
$m = \frac23$.

\textbf{20.} Cartesisch: vul een punt in, en het lidmaatschap is beslecht.
Herleid: \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} en $y$-afsnede zichtbaar, \omterm{def:g10:functions:graph}{grafiek} meteen.
Parametrisch: de klok zit ingebouwd --- aanvaringen, kleinste nadering,
onderschepping. De \omterm{def:g11:vect:det}{determinant}: één kruisvermenigvuldiging, en de
evenwijdigheid staat vast. Het gekozen \omterm{def:g10:coordgeom:system}{assenstelsel}: een parallellogram wordt
een eenheidsvierkant, en een klassieke stelling over verdeling in drieën
wordt het snijden van drie rechten.
\end{solution}
