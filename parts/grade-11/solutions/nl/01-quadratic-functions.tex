\section*{Hoofdstuk \ref{ch:g11:quad} --- Kwadratische functies en vergelijkingen}

\begin{solution}{exo:g11:quad:1}
$x^2-5x+6$: $\Delta = 25 - 24 = 1$, \omterm{def:g11:quad:discriminant}{wortels} $\frac{5 \pm 1}{2}$, dus $2$ en
$3$.

$2x^2+3x-2$: $\Delta = 9 + 16 = 25$, \omterm{def:g11:quad:discriminant}{wortels} $\frac{-3 \pm 5}{4}$, dus
$\frac12$ en $-2$.

$x^2+2x+5$: $\Delta = 4 - 20 = -16 < 0$, geen reële oplossing.

$9x^2-6x+1$: $\Delta = 36 - 36 = 0$, één (dubbele) \omterm{def:g11:quad:discriminant}{wortel}
$x_0 = \frac{6}{18} = \frac13$.
\end{solution}

\begin{solution}{exo:g11:quad:2}
$f(x) = x^2 - 6x + 11 = (x-3)^2 - 9 + 11 = (x-3)^2 + 2$: top $(3, 2)$.

$g(x) = -2x^2+4x+1 = -2(x^2 - 2x) + 1 = -2\bigl((x-1)^2 - 1\bigr) + 1
= -2(x-1)^2 + 3$: top $(1, 3)$.

$h(x) = 3x^2 + 6x = 3(x^2 + 2x) = 3\bigl((x+1)^2 - 1\bigr)
= 3(x+1)^2 - 3$: top $(-1, -3)$.
\end{solution}

\begin{solution}{exo:g11:quad:3}
$x^2 - 3x - 10$: $\Delta = 9 + 40 = 49$, \omterm{def:g11:quad:discriminant}{wortels}
$\frac{3 \pm 7}{2} = -2$ en $5$; de kopcoëfficiënt is positief, dus is de
drieterm strikt tussen de \omterm{def:g11:quad:discriminant}{wortels} negatief: de oplossingsverzameling is
$\intoo{-2}{5}$.

$2x^2 + x + 3$: $\Delta = 1 - 24 = -23 < 0$ en $a = 2 > 0$: altijd positief,
dus is de oplossingsverzameling $\R$.

$-x^2 + 6x - 9 = -(x-3)^2$: dat is alleen $\geq 0$ waar
$(x-3)^2 \leq 0$, dus in $x = 3$. De oplossingsverzameling is $\{3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:4}
$x^2 - 7x + 6$: $1 - 7 + 6 = 0$, dus is $1$ een \omterm{def:g11:quad:discriminant}{wortel}; het product van de
\omterm{def:g11:quad:discriminant}{wortels} is $6$, dus is de andere \omterm{def:g11:quad:discriminant}{wortel} $6$.

$2x^2 + 5x - 7$: $2 + 5 - 7 = 0$, dus is $1$ een \omterm{def:g11:quad:discriminant}{wortel}; het product is
$\frac{c}{a} = -\frac72$, dus is de andere \omterm{def:g11:quad:discriminant}{wortel} $-\frac72$.

$x^2 + 4x - 5$: $1 + 4 - 5 = 0$, dus is $1$ een \omterm{def:g11:quad:discriminant}{wortel}; het product is $-5$,
dus is de andere \omterm{def:g11:quad:discriminant}{wortel} $-5$.
\end{solution}

\begin{solution}{exo:g11:quad:5}
De twee getallen lossen $x^2 - 14x + 45 = 0$ op
(\cref{prop:g11:quad:vieta}); $\Delta = 196 - 180 = 16$, met \omterm{def:g11:quad:discriminant}{wortels}
$\frac{14 \pm 4}{2} = 9$ en $5$.

Voor de rechthoek: de halve omtrek geeft lengte $+$ breedte $= 17$, en de
oppervlakte geeft hun product $60$. Ze lossen $x^2 - 17x + 60 = 0$ op;
$\Delta = 289 - 240 = 49$, met \omterm{def:g11:quad:discriminant}{wortels} $\frac{17 \pm 7}{2} = 12$ en $5$. De
rechthoek meet $12 \times 5$.
\end{solution}

\begin{solution}{exo:g11:quad:6}
$\Delta = (m-2)^2 - 4$. Precies één oplossing wanneer $\Delta = 0$:
$(m-2)^2 = 4$, dus $m - 2 = \pm 2$, dus $m = 0$ of $m = 4$. Geen oplossing
wanneer $\Delta < 0$: $(m-2)^2 < 4$, dus $-2 < m - 2 < 2$, dus
$0 < m < 4$.
\end{solution}

\begin{solution}{exo:g11:quad:7}
De top ligt in $t = -\frac{20}{2 \times (-5)} = 2$, en
$h(2) = -20 + 40 + 1 = 21$: grootste hoogte $21$ m, na $2$ seconden bereikt.

$h(t) \geq 16$ betekent $-5t^2 + 20t - 15 \geq 0$, dus (na deling door $-5$,
wat de ongelijkheid omkeert) $t^2 - 4t + 3 \leq 0$. De \omterm{def:g11:quad:discriminant}{wortels} van
$t^2-4t+3$ zijn $1$ en $3$ ($1 - 4 + 3 = 0$, product $3$), zodat
$t^2 - 4t + 3 \leq 0$ precies op $\intcc{1}{3}$ geldt: de bal bevindt zich
tussen de tijdstippen $t = 1$ s en $t = 3$ s op minstens $16$ m hoogte.
\end{solution}

\begin{solution}{exo:g11:quad:8}
Met $X = x^2$: $X^2 - 13X + 36 = 0$, $\Delta = 169 - 144 = 25$, dus
$X = \frac{13 \pm 5}{2} = 9$ of $4$. Terug naar $x$: $x^2 = 9$ geeft
$x = \pm 3$, en $x^2 = 4$ geeft $x = \pm 2$. De oplossingsverzameling is
$\{-3, -2, 2, 3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:9}
De voorwaarde is $x^2 < x + 2$, dus $x^2 - x - 2 < 0$. Omdat
$x^2 - x - 2 = (x+1)(x-2)$ (\omterm{def:g11:quad:discriminant}{wortels} $-1$ en $2$), geldt de ongelijkheid
strikt tussen de \omterm{def:g11:quad:discriminant}{wortels}: $x \in \intoo{-1}{2}$. Op een schets duikt de
\omterm{def:g11:quad:parabola}{parabool} $y = x^2$ precies tussen haar twee snijpunten $(-1, 1)$ en $(2, 4)$
onder de rechte $y = x+2$.
\end{solution}

\begin{solution}{exo:g11:quad:10}
Zij $x > 0$ met $x + \frac1x = \frac{13}{6}$. Vermenigvuldigen met
$6x > 0$: $6x^2 + 6 = 13x$, dus $6x^2 - 13x + 6 = 0$. Dan is
$\Delta = 169 - 144 = 25$ en $x = \frac{13 \pm 5}{12}$, wat $x = \frac32$ of
$x = \frac23$ geeft. Beide zijn positief en elkaars omgekeerde, zoals
verwacht: het getal is $\frac32$ (of $\frac23$).
\end{solution}

\begin{solution}{exo:g11:quad:11}
\emph{1.} De snijpunten horen bij de oplossingen van $x^2 = mx - 1$, dus van
$x^2 - mx + 1 = 0$, met \omterm{def:g11:quad:discriminant}{discriminant} $\Delta = m^2 - 4$. Twee punten wanneer
$\abs{m} > 2$; precies één wanneer $m = \pm 2$; geen enkel wanneer
$-2 < m < 2$.

\emph{2.} Voor $m = 2$: $x^2 - 2x + 1 = (x-1)^2 = 0$, dus is het enige
raakpunt $(1, 1)$. Voor $m = -2$: $(x+1)^2 = 0$, raakpunt $(-1, 1)$. In
beide gevallen ontmoet de rechte de \omterm{def:g11:quad:parabola}{parabool} in één punt zonder haar te
kruisen --- het is daar de raaklijn, en inderdaad is haar
\omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $m = \pm 2$ gelijk aan de waarde in $x = \pm 1$ van de
afgeleide $2x$ van $x^2$ (\cref{ch:g11:deriv}).
\end{solution}

\begin{solution}{pb:g11:quad:1}
\textbf{1.} $x^2 - 5x + 6$: $\Delta = 1$, \omterm{def:g11:quad:discriminant}{wortels} $2$ en $3$.
$2x^2 + x - 6$: $\Delta = 1 + 48 = 49$, \omterm{def:g11:quad:discriminant}{wortels} $\frac{-1 \pm 7}{4}$:
$\frac32$ en $-2$. $x^2 + x + 1$: $\Delta = -3 < 0$: geen reële oplossing.

\textbf{2.} $\Delta = 16 + 48 = 64$: \omterm{def:g11:quad:discriminant}{wortels} $3$ en $-1$, dus is
$2x^2 - 4x - 6 = 2(x - 3)(x + 1)$; de \omterm{thm:g11:quad:canonical}{toppuntsvorm} is $2(x - 1)^2 - 8$. De
ontbonden vorm toont de \omterm{def:g11:quad:discriminant}{wortels}, de \omterm{thm:g11:quad:canonical}{toppuntsvorm} de top $(1, -8)$, en de
uitgewerkte vorm de $y$-afsnede $-6$.

\textbf{3.} \omterm{def:g11:quad:discriminant}{Wortels} $1$ en $3$, met negatieve kopcoëfficiënt: de drieterm is
$\geq 0$ \emph{tussen} de \omterm{def:g11:quad:discriminant}{wortels}: $x \in \intcc{1}{3}$.

\textbf{4.} De getallen zijn de \omterm{def:g11:quad:discriminant}{wortels} van $x^2 - 7x + 12 = 0$: $3$ en $4$.
Is $2$ een \omterm{def:g11:quad:discriminant}{wortel} van $x^2 - 5x + c$, dan voldoet de andere \omterm{def:g11:quad:discriminant}{wortel} $r$ aan
$2 + r = 5$ (som), dus $r = 3$, en $c = 2 \times 3 = 6$ (product): Viète
leest het zo af.

\textbf{5.} $\Delta = m^2 - 36$: een dubbele \omterm{def:g11:quad:discriminant}{wortel} voor $m = \pm 6$ (\omterm{def:g11:quad:discriminant}{wortel}
$\mp 3$); geen \omterm{def:g11:quad:discriminant}{wortel} voor $-6 < m < 6$; twee \omterm{def:g11:quad:discriminant}{wortels} voor $\abs m > 6$.

\textbf{6.} $PF^2 = x^2 + \left(x^2 - \frac14\right)^2
= x^2 + x^4 - \frac{x^2}{2} + \frac{1}{16}
= x^4 + \frac{x^2}{2} + \frac{1}{16}
= \left(x^2 + \frac14\right)^2$. Bijgevolg is $PF = x^2 + \frac14$ (beide
factoren zijn positief).

\textbf{7.} De afstand van $P(x, x^2)$ tot de horizontale rechte
$y = -\frac14$ is $x^2 + \frac14$ --- precies $PF$. Elk punt van de \omterm{def:g11:quad:parabola}{parabool}
ligt even ver van het \omterm{pb:g11:quad:1}{brandpunt} als van de \omterm{pb:g11:quad:1}{richtlijn}.

\textbf{8.} $PF^2 = x^2 + \left(y - \frac14\right)^2$ en
$\operatorname{dist}^2 = \left(y + \frac14\right)^2$. Gelijkstellen geeft
$x^2 + y^2 - \frac y2 + \frac{1}{16} = y^2 + \frac y2 + \frac1{16}$, dus
$x^2 = y$: de meetkundige plaats is precies de \omterm{def:g11:quad:parabola}{parabool}. Eén punt en één
rechte brengen de kromme voort, zoals één punt en één afstand de cirkel
voortbrachten.

\textbf{9.} Voor $y = \frac{x^2}{4f}$ geeft dezelfde uitwerking met
$F(0, f)$ dat $PF^2 = x^2 + \left(\frac{x^2}{4f} - f\right)^2
= \left(\frac{x^2}{4f} + f\right)^2$: de afstand tot $F$ is gelijk aan de
afstand tot $y = -f$. Schotel: $9 = \frac{30^2}{4f}$ geeft $4f = 100$, dus
$f = 25$ cm: de ontvanger hangt een kwart meter boven de top --- buiten de
$9$ cm diepe schotel, zoals echte antennes laten zien.

\textbf{10.} Een golffront dat evenwijdig met de as binnenkomt heeft gelijke
afstanden afgelegd om de hoogte van de \omterm{pb:g11:quad:1}{richtlijn} te bereiken; de gelijke
afstanden zetten ``even ver van de \omterm{pb:g11:quad:1}{richtlijn}'' om in ``even ver van $F$'':
alle weerkaatste signalen komen \emph{in de pas} in $F$ aan en versterken
elkaar --- daar moet je dus luisteren. Het eerlijke bewijs vraagt de
\omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} van de raaklijn in elk punt --- de afgeleide van
\cref{ch:g11:deriv}, waarvan \cref{exo:g11:quad:11} een glimp geeft.

\textbf{11.} $y = x\left(1 - \frac{x}{20}\right)$: hij komt neer in
$x = 20$ m. De top ligt halverwege: $x = 10$, met hoogte
$y(10) = 10 - 5 = 5$ m.

\textbf{12.} $y(4) = 4 - \frac{16}{20} = 3.2$ m: ruim over een muur van
$3$ m, maar knal tegen een muur van $4$ m.

\textbf{13.} $y(10) = 10 - \frac{100}{40} = 7.5$ m $> 7$: de vrachtwagen
raakt erdoor op $10$ m van het \omterm{prop:g10:coordgeom:midpoint}{midden}. $y(15) = 10 - \frac{225}{40} =
4.375$ m: op $15$ m niet.

\textbf{14.} $R(p) = -2p^2 + 120p$: top in $p = \frac{120}{4} = 30$ euro,
met opbrengst $R(30) = 30 \times 60 = 1\,800$ euro.

\textbf{15.} $y(320) = 143 \times \left(\frac{320}{640}\right)^2 =
\frac{143}{4} \approx 36$ m boven het laagste punt.

\textbf{16.} $x^2 = mx - 1$ geeft $x^2 - mx + 1 = 0$, met
$\Delta = m^2 - 4$: twee punten voor $\abs m > 2$, geen voor $\abs m < 2$,
en raking voor $m = \pm 2$ met raakpunt $x = \frac m2 = \pm 1$: de
raaklijnen $y = 2x - 1$ in $(1, 1)$ en $y = -2x - 1$ in $(-1, 1)$.

\textbf{17.} $x^2 = mx + 1$ geeft $\Delta = m^2 + 4 > 0$ voor elke $m$: dus
altijd twee snijpunten. Vanuit een inwendig punt kruist elke rechte de
kromme twee keer --- raaklijnen teken je alleen van buitenaf, net als bij
cirkels.

\textbf{18.} $y = \frac14$ ontmoet $y = x^2$ in $x = \pm\frac12$:
koordelengte $1$. Voor $y = \frac{x^2}{4f}$ geeft $y = f$ dat $x^2 = 4f^2$,
dus $x = \pm 2f$: lengte $4f$ --- de brandpuntsbreedte is vier keer de
brandpuntsafstand (schotelontwerpers noemen ze de latus rectum).

\textbf{19.} Een punt van $y = x^2$ is $(a, a^2)$; zijn \omterm{def:g10:functions:function}{beeld} is
$(X, Y) = (ka, ka^2)$, en $\frac{X^2}{k} = \frac{k^2a^2}{k} = ka^2 = Y$: de
beeldkromme is precies $y = \frac{x^2}{k}$. Met $k = 4f$ bereik je elke
\omterm{def:g11:quad:parabola}{parabool}: het zijn allemaal vergrotingen van één moederkromme. (Ook cirkels
zijn onderling gelijkvormig --- ellipsen niet: samendrukken telt, vergroten
niet.)

\textbf{20.} Een \omterm{def:g10:algebra:equation}{vergelijking} met drie leesbare kostuums; de meetkundige
plaats van de gelijke afstand tot een punt en een rechte; de weerkaatser die
evenwijdige stralen in zijn \omterm{pb:g11:quad:1}{brandpunt} bundelt en de kabel die een
gelijkmatig belast dek draagt; de baan van elke geworpen bal; en, op een
vergroting na, één enkele kromme --- één \omterm{def:g11:quad:parabola}{parabool}, overal.
\end{solution}
