\section*{Hoofdstuk \ref{ch:g11:binom} --- De binomiale verdeling}

\begin{solution}{exo:g11:binom:1}
De rijen $6$ en $7$:
\[
1,\ 6,\ 15,\ 20,\ 15,\ 6,\ 1
\qquad\text{en}\qquad
1,\ 7,\ 21,\ 35,\ 35,\ 21,\ 7,\ 1 .
\]
Bijgevolg is $\binom62 = 15$, $\binom73 = 35$ en $\binom74 = 35$ (de
symmetrie $\binom73 = \binom74$ weerspiegelt het verwisselen van successen en
mislukkingen).
\end{solution}

\begin{solution}{exo:g11:binom:2}
Vast aantal experimenten ($4$ worpen), twee uitkomsten per worp (zes of geen
zes, $p = \frac16$), en onafhankelijke worpen: $X \sim \mathcal B(4,
\frac16)$.
\[
\P(X=0) = \left(\frac56\right)^4 = \frac{625}{1296} \approx 0.48,
\qquad
\P(X=1) = 4 \times \frac16\left(\frac56\right)^3 = \frac{500}{1296}
\approx 0.39,
\]
\[
\P(X \geq 2) = 1 - \frac{625 + 500}{1296} = \frac{171}{1296}
\approx 0.13 .
\]
\end{solution}

\begin{solution}{exo:g11:binom:3}
\emph{1.} Binomiaal $\mathcal B(20, \frac12)$: vaste $n$, dezelfde $p$,
onafhankelijke worpen.

\emph{2.} Niet binomiaal: de kaarten worden \emph{zonder teruglegging}
gedeeld, zodat de \omterm{def:g10:proba:distribution}{kans} op een aas van kaart tot kaart verandert en de
trekkingen niet onafhankelijk zijn.

\emph{3.} Binomiaal $\mathcal B(7, 0.3)$, volgens de veronderstelde
onafhankelijkheid.
\end{solution}

\begin{solution}{exo:g11:binom:4}
$\E(X) = 50 \times 0.2 = 10$; $\V(X) = 50 \times 0.2 \times 0.8 = 8$;
$\sigma(X) = 2\sqrt2 \approx 2.83$.
\end{solution}

\begin{solution}{exo:g11:binom:5}
$X \sim \mathcal B(6, 0.7)$.
\[
\P(X = 4) = \binom64 (0.7)^4 (0.3)^2 = 15 \times 0.2401 \times 0.09
\approx 0.324 .
\]
\[
\P(X \geq 5) = \binom65 (0.7)^5(0.3) + (0.7)^6
= 6 \times 0.16807 \times 0.3 + 0.117649 \approx 0.420 .
\]
\end{solution}

\begin{solution}{exo:g11:binom:6}
$X \sim \mathcal B\left(8, \frac12\right)$; elk pad heeft \omterm{def:g10:proba:distribution}{kans}
$\frac{1}{256}$, dus
\[
\P(X \geq 6) = \frac{\binom86 + \binom87 + \binom88}{256}
= \frac{28 + 8 + 1}{256} = \frac{37}{256} \approx 0.14 .
\]
Gokken levert ongeveer één keer op zeven een voldoende op.
\end{solution}

\begin{solution}{exo:g11:binom:7}
Het \omterm{def:g10:proba:operations}{complement} van ``minstens één van elk'' is ``alle vijf gelijk'': alles A
of alles B, elk met \omterm{def:g10:proba:distribution}{kans} $\left(\frac12\right)^5 = \frac1{32}$. Bijgevolg is
\[
\P(\text{één van elke soort}) = 1 - \frac{2}{32} = \frac{15}{16} .
\]
\end{solution}

\begin{solution}{exo:g11:binom:8}
$\P(X = 3) = p^3$ en $\P(X \geq 1) = 1 - (1-p)^3$. Uit
$p^3 = \frac{27}{64} = \left(\frac34\right)^3$ volgt $p = \frac34$ (de
derdemachtsfunctie is strikt \omterm{def:g11:func:monotone}{stijgend}, \cref{ch:g11:func}, dus is de
oplossing uniek).
\end{solution}

\begin{solution}{exo:g11:binom:9}
$\P(\text{minstens één keer kop}) = 1 - \left(\frac12\right)^n$, dus luidt de
voorwaarde $\left(\frac12\right)^n < 0.01$, dus $2^n > 100$. Omdat
$2^6 = 64$ en $2^7 = 128$: vanaf $n = 7$ worpen.
\end{solution}

\begin{solution}{exo:g11:binom:10}
De som van rij $n$ telt alle paden van het \omterm{met:g10:proba:tree}{boomdiagram} met $n$ experimenten,
gesorteerd naar hun aantal successen. Maar het \omterm{met:g10:proba:tree}{boomdiagram} verdubbelt zijn
paden bij elk experiment (elk pad splitst in S en M), zodat het er in totaal
$2^n$ heeft. Bijgevolg is $\sum_{k} \binom nk = 2^n$. Anders, met inductie:
rij $0$ telt op tot $1 = 2^0$, en de regel van Pascal laat elk getal van rij
$n$ aan precies twee getallen van rij $n+1$ bijdragen, zodat de rijsommen
verdubbelen.
\end{solution}

\begin{solution}{exo:g11:binom:11}
\emph{1.} Onder de bewering is $X \sim \mathcal B(10, 0.6)$. De eerste termen
optellen geeft
\begin{align*}
\P(X \leq 3)
&= (0.4)^{10} + 10(0.6)(0.4)^9 + 45(0.6)^2(0.4)^8
+ 120(0.6)^3(0.4)^7\\
&\approx 0.0001 + 0.0016 + 0.0106 + 0.0425 = 0.0548 .
\end{align*}

\emph{2.} Een resultaat dat minstens even extreem is als het waargenomene
($3$ of minder steunbetuigingen) heeft \omterm{def:g10:proba:distribution}{kans} ongeveer $5.5\%$ --- net
\emph{boven} de drempel van $5\%$. Wie de regel strikt toepast, besluit dat
de waarneming (nipt) verenigbaar is met de bewering en verwerpt haar niet.
Het voorbeeld toont hoe gevoelig grensgevallen zijn voor de keuze van de
drempel: bij een afspraak van $6\%$ zou het besluit omslaan.
\end{solution}

\begin{solution}{pb:g11:binom:1}
\textbf{1.} De \omterm{def:g11:seq:sequence}{rijen}: $1$; $1\,1$; $1\,2\,1$; $1\,3\,3\,1$; $1\,4\,6\,4\,1$;
$1\,5\,10\,10\,5\,1$; $1\,6\,15\,20\,15\,6\,1$. De symmetrie: kiezen welke
$k$ voorwerpen je meeneemt is dezelfde handeling als kiezen welke $n - k$ je
achterlaat.

\textbf{2.} $1 + 4 + 6 + 4 + 1 = 16 = 2^4$;
$1 + 5 + 10 + 10 + 5 + 1 = 32 = 2^5$. Bewijs: de $\binom nk$ tellen de
deelverzamelingen van elke omvang van een verzameling met $n$ elementen, en
alle deelverzamelingen samen zijn er $2^n$ (elk element onafhankelijk erin of
eruit).

\textbf{3.} Commissies van $k$ personen, gekozen uit $n + 1$ mensen onder wie
Zoë: die zonder Zoë zijn er $\binom nk$ (kies alle $k$ onder de anderen); die
met Zoë zijn er $\binom{n}{k-1}$ (kies haar $k - 1$ collega's). Samen:
$\binom nk + \binom{n}{k-1}$.

\textbf{4.} \omterm{def:g11:seq:sequence}{Rij} $7$ van de driehoek: $1\,7\,21\,35\,\dots$: dus $35$. Met de
formule: $\frac{7 \times 6 \times 5}{3 \times 2 \times 1} = 35$.

\textbf{5.} $1 + 3 + 6 + 10 = 20 = \binom63$. De cascade:
$\binom63 = \binom52 + \binom53 = \binom52 + \binom42 + \binom43
= \binom52 + \binom42 + \binom32 + \binom33$ --- elke toepassing van de regel
van Pascal pelt één trede van de trap af.

\textbf{6.} Een vast aantal $n$ stuiten; elke stuit een onafhankelijk
\omterm{def:g11:binom:bernoulli}{bernoulli-experiment} met dezelfde $p = \frac12$; en het vakjesnummer telt de
successen (de stuiten naar rechts): alle drie de vakjes van
\cref{met:g11:binom:recognize} aangevinkt:
$\mathcal B\!\left(n, \frac12\right)$.

\textbf{7.} De kansen zijn $\frac{1}{16}, \frac{4}{16}, \frac{6}{16},
\frac{4}{16}, \frac{1}{16}$ voor de vakjes $0, \dots, 4$: het middelste vakje
$2$ raakt het drukst bezet.

\textbf{8.} De verwachte aantallen zijn
$1024 \times \binom{10}{k}/1024 = \binom{10}{k}$: het middelste vakje krijgt
$\binom{10}{5} = 252$ balletjes; vakje $7$ krijgt er $\binom{10}{7} = 120$;
en elk randvakje $1$. Een hoog \omterm{prop:g10:coordgeom:midpoint}{midden} dat symmetrisch wegzakt naar
flinterdunne randen: de klok.

\textbf{9.} $\E = np = 5$; $V = np(1 - p) = 2.5$; $\sigma \approx 1.58$.
Binnen $2\sigma$ dragen de vakjes $2$ tot $8$
\[
\frac{45 + 120 + 210 + 252 + 210 + 120 + 45}{1024}
= \frac{1002}{1024} \approx 98\,\%
\]
van de balletjes --- veel beter dan de universele $75\,\%$ van Chebyshev
(\cref{pb:g11:stat:1}): klokvormen concentreren zich hard.

\textbf{10.} $\E = 6$, $V = 10 \times 0.6 \times 0.4 = 2.4$,
$\sigma \approx 1.55$: de stapel houdt haar klokvorm maar schuift haar top
naar vakje $6$ --- een scheve plank is een verzwaarde munt, zichtbaar
gemaakt.

\textbf{11.} Het vakjesnummer is een \emph{som} van veel kleine,
onafhankelijke, even grote toevalsduwtjes --- en zulke sommen ordenen zich
altijd tot de klok: de meeste duwtjes heffen elkaar op, terwijl uitersten
eensgezindheid vragen. Lichaamslengten, meetfouten en talloze andere
natuurlijke grootheden zijn eveneens sommen van veel kleine onafhankelijke
effecten, en daarom duikt overal dezelfde silhouet op; de stelling die dat
bevestigt is de centrale limietstelling.

\textbf{12.} Een schoonveeg: één ploeg wint alle $4$:
$2 \times \left(\frac12\right)^4 = \frac18$.

\textbf{13.} Zeven wedstrijden vragen $3$--$3$ na zes:
$\binom63 \left(\frac12\right)^6 = \frac{20}{64} = \frac{5}{16}$.

\textbf{14.} Eindigt in $5$: de winnaar pakt wedstrijd 5 en $3$ van de eerste
$4$: $2 \times \binom43 \left(\frac12\right)^5 = \frac14$. Eindigt in $6$:
$2 \times \binom53 \left(\frac12\right)^6 = \frac{5}{16}$. De verdeling over
$4, 5, 6, 7$: $\frac18, \frac14, \frac{5}{16}, \frac{5}{16}$ (som $1$).
Verwachte lengte: $4 \cdot \frac18 + 5 \cdot \frac14 + 6 \cdot \frac{5}{16} +
7 \cdot \frac{5}{16} = 5.8125$ wedstrijden. Reeksen van zes en zeven
wedstrijden zijn het waarschijnlijkst --- de spanning zit in het format
ingebouwd.

\textbf{15.} Winst in $4$: $0.6^4 = 0.1296$; in $5$:
$\binom43\,0.6^3 \times 0.4 \times 0.6 = 0.2074$; in $6$:
$\binom53\,0.6^3 \times 0.4^2 \times 0.6 = 0.2074$; in $7$:
$\binom63\,0.6^3 \times 0.4^3 \times 0.6 = 0.1659$. Samen: ongeveer $0.710$:
een ploeg met $60\,\%$ per wedstrijd wint $71\,\%$ van de reeksen --- de
reeks versterkt de voorsprong.

\textbf{16.} Eén enkele finale: $60\,\%$. Reeks van drie:
$p^2 + 2p^2 q = 0.36 + 0.288 = 0.648$. De ladder
$60\,\% \to 65\,\% \to 71\,\%$ loopt door met de lengte: meer wedstrijden
middelen het geluk uit (de wet van de grote aantallen in het klein), zodat
lange finales de kunde bekronen --- en dat is precies wat competities
verkopen.

\textbf{17.} Eerlijke munt: $\E = 50$, $\sigma = \sqrt{25} = 5$;
$z = \frac{62 - 50}{5} = 2.4$: voorbij de afspraak van $2\sigma$ --- de munt
verdient een onderzoek.

\textbf{18.} $\P(X = 0) = 0.98^{50} \approx 0.364$;
$\P(X = 1) = 50 \times 0.02 \times 0.98^{49} \approx 0.372$;
$\P(X = 2) = \binom{50}{2} 0.02^2 \times 0.98^{48} \approx 0.186$. Dus is
$\P(X \geq 3) \approx 1 - 0.922 = 0.078$: ongeveer $7.8\,\%$ --- boven de
drempel van $5\,\%$, dus (nog) niet genoeg om de bewering te verwerpen; een
tweede slechte partij zou het verhaal veranderen.

\textbf{19.} $\P(\text{minstens één winnend lot}) = 1 - 0.999^{1000}
\approx 1 - 0.368 = 0.632$: duizend loten met \omterm{def:g10:proba:distribution}{kans} één op duizend geven geen
zekerheid maar $63\,\%$. De steeds terugkerende $0.368$ is $\frac1e$ in
vermomming --- de constante $e$ maakt haar officiële intrede in jaar 12.

\textbf{20.} Herkenning: vaste $n$, onafhankelijkheid, constante $p$ --- dan
en alleen dan binomiaal. Tabel: de \omterm{prop:g11:binom:pascal}{driehoek van Pascal}, rij $n$. Vorm: de
klok, symmetrisch voor $p = \frac12$ en verschoven daarbuiten. Kompas: het
\omterm{prop:g10:coordgeom:midpoint}{midden} $np$ en de spreiding $\sqrt{np(1-p)}$ --- de \omterm{pb:g11:stat:1}{z-scores} van
beslissingen. Erfgenamen: de gladde klokkromme die de stapels naderen, en de
wet van de grote aantallen die verklaart waarom grote planken nooit liegen.
\end{solution}
