\section*{Hoofdstuk \ref{ch:g11:trigo} --- Goniometrie: de eenheidscirkel}

\begin{solution}{exo:g11:trigo:1}
Naar \omterm{def:g11:trigo:radian}{radialen}: $30^\circ = \frac{\pi}{6}$, $45^\circ = \frac{\pi}{4}$,
$120^\circ = \frac{2\pi}{3}$, $270^\circ = \frac{3\pi}{2}$.

Naar graden: $\frac{\pi}{5} = 36^\circ$, $\frac{5\pi}{6} = 150^\circ$,
$\frac{7\pi}{4} = 315^\circ$, $\frac{2\pi}{9} = 40^\circ$.
\end{solution}

\begin{solution}{exo:g11:trigo:2}
$\frac{2\pi}{3}$ en $-\frac{\pi}{4}$ liggen al in $\intoc{-\pi}{\pi}$:
respectievelijk in het tweede en in het vierde kwadrant.

$\frac{17\pi}{6} - 2\pi = \frac{5\pi}{6}$: tweede kwadrant, dicht bij de
negatieve $x$-as.

$-\frac{7\pi}{2} + 4\pi = \frac{\pi}{2}$: de top van de cirkel, $(0, 1)$.
\end{solution}

\begin{solution}{exo:g11:trigo:3}
$\cos\frac{3\pi}{4} = \cos\left(\pi - \frac\pi4\right) = -\cos\frac\pi4
= -\frac{\sqrt2}{2}$.

$\sin\frac{5\pi}{6} = \sin\left(\pi - \frac\pi6\right) = \sin\frac\pi6
= \frac12$.

$\cos\left(-\frac\pi3\right) = \cos\frac\pi3 = \frac12$.

$\sin\frac{4\pi}{3} = \sin\left(\pi + \frac\pi3\right) = -\sin\frac\pi3
= -\frac{\sqrt3}{2}$.

$\cos\frac{11\pi}{6} = \cos\left(-\frac\pi6 + 2\pi\right) = \cos\frac\pi6
= \frac{\sqrt3}{2}$.
\end{solution}

\begin{solution}{exo:g11:trigo:4}
$\sin^2 t = 1 - \cos^2 t = 1 - \frac{25}{169} = \frac{144}{169}$, dus
$\sin t = \pm\frac{12}{13}$. Op $\intoo{-\frac\pi2}{0}$ (vierde kwadrant) is
de \omterm{def:g10:coordgeom:system}{ordinaat} negatief: $\sin t = -\frac{12}{13}$.
\end{solution}

\begin{solution}{exo:g11:trigo:5}
Met \cref{prop:g11:trigo:associated}:
\[
A = (-\cos x) + \cos x + \cos x - \cos x = 0 ,
\]
\[
B = \sin x + (-\sin x) + (-\sin x) = -\sin x .
\]
\end{solution}

\begin{solution}{exo:g11:trigo:6}
$\cos t = \frac{\sqrt3}{2}$: één oplossing is $\frac\pi6$, dus
$t = \pm\frac\pi6$ (de verschuivingen over $2k\pi$ verlaten het \omterm{def:g10:numbers:interval}{interval}).

$\sin t = \frac12$: $t = \frac\pi6$ of $t = \pi - \frac\pi6 =
\frac{5\pi}{6}$.

$\cos t = -1$: het enige punt met \omterm{def:g10:coordgeom:system}{abscis} $-1$ is $M(\pi)$, dus $t = \pi$.
\end{solution}

\begin{solution}{exo:g11:trigo:7}
$\sin t = -\frac{\sqrt3}{2}$: vanuit $a = -\frac\pi3$ zijn de families
$-\frac\pi3 + 2k\pi$ en $\pi + \frac\pi3 + 2k\pi$; in $\intco{0}{2\pi}$:
$t = \frac{4\pi}{3}$ en $t = \frac{5\pi}{3}$.

$\cos t = 0$: $t = \frac\pi2$ en $t = \frac{3\pi}{2}$.

$2\sin t + 1 = 0$ betekent $\sin t = -\frac12$: vanuit $a = -\frac\pi6$, in
$\intco{0}{2\pi}$: $t = \frac{7\pi}{6}$ en $t = \frac{11\pi}{6}$.
\end{solution}

\begin{solution}{exo:g11:trigo:8}
Volgens \cref{thm:g11:trigo:equations} betekent $\cos 2t = \cos t$ dat
$2t = t + 2k\pi$ of $2t = -t + 2k\pi$, dus $t = 2k\pi$ of
$t = \frac{2k\pi}{3}$ ($k \in \Z$). De eerste familie zit in de tweede
vervat (neem $k$ een veelvoud van $3$), zodat de oplossingsverzameling
\[
S = \left\{\frac{2k\pi}{3} : k \in \Z\right\}
\]
is --- op de cirkel de drie punten $M(0)$, $M\!\left(\frac{2\pi}{3}\right)$
en $M\!\left(\frac{4\pi}{3}\right)$.
\end{solution}

\begin{solution}{exo:g11:trigo:9}
Beide kwadraten \omterm{def:g10:algebra:expand}{uitwerken} en $\cos^2 x + \sin^2 x = 1$ gebruiken:
\[
(\cos^2 x + 2\sin x\cos x + \sin^2 x)
+ (\cos^2 x - 2\sin x\cos x + \sin^2 x)
= 1 + 1 = 2 .
\]
\end{solution}

\begin{solution}{exo:g11:trigo:10}
Rollen zonder slippen betekent dat de booglengte gelijk is aan de afgelegde
afstand: $r\theta = 150$ cm geeft $\theta = \frac{150}{30} = 5$ rad
$= 5 \times \frac{180}{\pi} \approx 286.5^\circ$. Eén volledige omwenteling
brengt de fiets de omtrek vooruit: $2\pi \times 30 = 60\pi \approx 188.5$
cm, ongeveer $1.88$ m.
\end{solution}

\begin{solution}{exo:g11:trigo:11}
Op de eenheidscirkel zijn de punten met \omterm{def:g10:coordgeom:system}{abscis} precies $\frac12$ de punten
$M\!\left(\pm\frac\pi3\right)$. De punten met \omterm{def:g10:coordgeom:system}{abscis} \emph{hoogstens}
$\frac12$ vormen de boog links van de verticale rechte $x = \frac12$, die je
doorloopt terwijl $t$ van $\frac\pi3$ tot $\pi$ gaat, of van $-\pi$ tot
$-\frac\pi3$. Bijgevolg is op $\intoc{-\pi}{\pi}$
\[
S = \intoc{-\pi}{-\frac{\pi}{3}} \cup \intcc{\frac{\pi}{3}}{\pi}.
\]
(De randpunten $\pm\frac\pi3$ tellen mee, want de ongelijkheid is ruim.)
\end{solution}

\begin{solution}{pb:g11:trigo:1}
\textbf{1.} $30^\circ = \frac{\pi}{6}$; $135^\circ = \frac{3\pi}{4}$;
$300^\circ = \frac{5\pi}{3}$. En $\frac{3\pi}{4} = 135^\circ$;
$\frac{5\pi}{6} = 150^\circ$.

\textbf{2.} Lengte $r\,t = 3 \times 2 = 6$ m. \omterm{def:g11:trigo:radian}{Radialen} maken van de
booglengte een gewoon product --- graden zouden in elke formule een
$\frac{\pi}{180}$ meeslepen.

\textbf{3.} Aantal \omterm{def:g11:trigo:radian}{radialen} $= \frac{\text{afstand}}{\text{straal}} =
\frac{1000}{0.30} \approx 3\,333$ rad; gedeeld door $2\pi$: ongeveer $530$
volledige omwentelingen.

\textbf{4.} $\cos\frac{2\pi}{3} = -\frac12$;
$\sin\left(-\frac\pi4\right) = -\frac{\sqrt2}{2}$;
$\frac{19\pi}{6} - 2\pi = \frac{7\pi}{6}$, dus
$\cos\frac{19\pi}{6} = \cos\frac{7\pi}{6} = -\frac{\sqrt3}{2}$.

\textbf{5.} $\cos^2 t = 1 - \frac{9}{25} = \frac{16}{25}$; in het tweede
kwadrant is de \omterm{def:g11:trigo:cossin}{cosinus} negatief: $\cos t = -\frac45$, en
$\tan t = \frac{3/5}{-4/5} = -\frac34$.

\textbf{6.} In $t$ minuten draait het rad over
$\frac{t}{30} \times 2\pi = \frac{\pi t}{15}$. In $t = 0$ is
$h = 65 - 60 = 5$ m --- het instapperron onderaan (ashoogte min straal). Het
minteken zet je \emph{onder} de as wanneer de gedraaide hoek $0$ is.

\textbf{7.} $h(7.5) = 65 - 60\cos\frac{\pi}{2} = 65$ m (ashoogte);
$h(15) = 65 + 60 = 125$ m (de top);
$h(20) = 65 - 60\cos\frac{4\pi}{3} = 65 + 30 = 95$ m.

\textbf{8.} $65 - 60\cos\frac{\pi t}{15} = 95$ geeft
$\cos\frac{\pi t}{15} = -\frac12$: de eerste oplossing is
$\frac{\pi t}{15} = \frac{2\pi}{3}$, dus $t = 10$ minuten.

\textbf{9.} $\cos\frac{\pi t}{15} \leq -\frac12$ voor
$\frac{\pi t}{15} \in \intcc{\frac{2\pi}{3}}{\frac{4\pi}{3}}$, dus
$t \in \intcc{10}{20}$: tien minuten van de rit boven $95$ m, symmetrisch
rond de top in $t = 15$.

\textbf{10.} Eerste minuut:
\[
h(1) - h(0) = 60\left(1 - \cos\frac{\pi}{15}\right)
\approx 1.3 \text{ m}.
\]
Tussen minuut $7$ en minuut $8$:
\[
60\left(\cos\frac{7\pi}{15} - \cos\frac{8\pi}{15}\right)
\approx 12.5 \text{ m}
\]
--- bijna tien keer meer. De hoogte verandert het snelst ter hoogte van de
as en staat bijna stil bij de bodem en de top: de sinusoïde is steil in haar
\omterm{prop:g10:coordgeom:midpoint}{midden} en vlak in haar uitersten.

\textbf{11.} Straal $= \frac{40 - 4}{2} = 18$ m, as op $4 + 18 = 22$ m,
periode $12$ min: $h(t) = 22 - 18\cos\left(\frac{\pi t}{6}\right)$.

\textbf{12.} \omterm{def:g10:functions:extrema}{Maximum} $7 + 3 = 10$ m wanneer $\sin\frac{\pi t}{6} = 1$: voor
het eerst in $t = 3$ (3 uur 's nachts); \omterm{def:g10:functions:extrema}{minimum} $4$ m voor het eerst in
$t = 9$; periode $\frac{2\pi}{\pi/6} = 12$ uur.

\textbf{13.} $\sin\frac{\pi t}{6} \geq \frac12$ voor
$\frac{\pi t}{6} \in \intcc{\frac\pi6}{\frac{5\pi}{6}}$: $t \in \intcc{1}{5}$
--- een venster van vier uur, van 1 tot 5 uur, dat één periode later opnieuw
opengaat, $t \in \intcc{13}{17}$ (van 13 tot 17 uur).

\textbf{14.} De spiegeling om de verticale as stuurt het punt met hoek $t$
naar het punt met hoek $\pi - t$ en behoudt de \omterm{def:g10:coordgeom:system}{ordinaat}:
$\sin(\pi - t) = \sin t$. De halve draai om het middelpunt stuurt $t$ naar
$\pi + t$ en keert beide \omterm{def:g10:coordgeom:system}{coördinaten} om: $\cos(\pi + t) = -\cos t$. En de
spiegeling om de diagonaal verwisselt \omterm{def:g10:coordgeom:system}{abscis} en \omterm{def:g10:coordgeom:system}{ordinaat}:
$\cos\left(\frac\pi2 - t\right) = \sin t$ --- de ``co'' van de
complementaire hoeken.

\textbf{15.} $\sin(-t) = -\sin t$: oneven (haar \omterm{def:g10:functions:graph}{grafiek} is symmetrisch onder
een halve draai); $\cos(-t) = \cos t$: even (spiegeling). En
$\sin t + t^3$ is een som van twee \omterm{def:g11:func:parity}{oneven functies}: oneven.

\textbf{16.} \omterm{def:g10:algebra:expand}{Ontbinden}: $(2\sin t + 1)(\sin t - 1) = 0$: $\sin t = 1$ geeft
$t = \frac\pi2$; $\sin t = -\frac12$ geeft $t = \frac{7\pi}{6}$ en
$t = \frac{11\pi}{6}$. Drie oplossingen op de cirkel.

\textbf{17.} $1$ rad $= \frac{180}{\pi} \approx 57.3^\circ$. Dus is
$\sin(1) \approx 0.841$ terwijl $\sin(1^\circ) \approx 0.0175$: een factor
van bijna $50$. Moraal: kijk naar het lampje van de modus voordat je een
goniometrische uitkomst vertrouwt.

\textbf{18.} \omterm{def:g10:coordgeom:system}{Abscis} en \omterm{def:g10:coordgeom:system}{ordinaat} zijn gelijk op de diagonaal:
$t = \frac\pi4$ en $t = \frac{5\pi}{4}$.

\textbf{19.} $\sin(0.1) = 0.09983\ldots$ tegenover $0.1$: een relatieve fout
van ongeveer $0.17\,\%$. Voor kleine hoeken vlijt de koorde zich tegen de
boog aan --- en de limiet $\frac{\sin t}{t} \to 1$ uit jaar 12 is precies
deze waarneming, tot stelling verheven.

\textbf{20.} \omterm{def:g11:trigo:radian}{Radialen}: hoek $=$ boog op de eenheidscirkel, zodat je met
hoeken rekent als met lengten. \omterm{def:g11:trigo:cossin}{Cosinus} en \omterm{def:g11:trigo:cossin}{sinus}: de \omterm{def:g10:coordgeom:system}{coördinaten} van de
wijzer --- schaduw op de grond, hoogte tegen de muur. Symmetrieën van de
cirkel: de hele catalogus van identiteiten, afgelezen van spiegelingen en
halve draaien. De \omterm{def:g10:algebra:equation}{vergelijkingen} $\cos t = c$: de dienstregeling van de klok,
met de cirkel die alle oplossingen tegelijk toont. Kostuums: het rad droeg
$65 - 60\cos$, het getij droeg $7 + 3\sin$ --- dezelfde klok, dezelfde
wiskunde.
\end{solution}
