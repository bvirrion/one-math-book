\section*{Hoofdstuk \ref{ch:g11:func} --- Functies en hun verloop}

\begin{solution}{exo:g11:func:1}
$f$: nodig is $3 - x \geq 0$, \omterm{def:g11:func:function}{domein} $\intoc{-\infty}{3}$.

$g$: nodig is $x^2 - 4 \neq 0$, dus $x \neq \pm 2$: \omterm{def:g11:func:function}{domein}
$\R \setminus \{-2, 2\}$.

$h$: nodig is $x \geq 0$ en $x \neq 5$: \omterm{def:g11:func:function}{domein}
$\intco{0}{5} \cup \intoo{5}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:2}
$\sqrt{11} < \sqrt{13}$, want de vierkantswortel is \omterm{def:g11:func:monotone}{stijgend} en
$11 < 13$.

$(-2.1)^2 > (-2.05)^2$, want $x^2$ is \omterm{def:g10:functions:variations}{dalend} op $\intoc{-\infty}{0}$ en
$-2.1 < -2.05$.

$\frac{1}{0.9} > \frac{1}{1.1}$, want $\frac1x$ is \omterm{def:g10:functions:variations}{dalend} op
$\intoo{0}{+\infty}$ en $0.9 < 1.1$.

$(-1.01)^3 < (-0.99)^3$, want $x^3$ is \omterm{def:g11:func:monotone}{stijgend} en $-1.01 < -0.99$.
\end{solution}

\begin{solution}{exo:g11:func:3}
$f(-x) = 3(-x)^4 - (-x)^2 = 3x^4 - x^2 = f(x)$: \omterm{def:g11:func:parity}{even}.

$g(-x) = -x^3 - x = -g(x)$: \omterm{def:g11:func:parity}{oneven}.

$h(1) = 2$ en $h(-1) = 1 - 1 = 0$, dus is $h(-1)$ noch $h(1)$ noch
$-h(1)$: noch \omterm{def:g11:func:parity}{even} noch \omterm{def:g11:func:parity}{oneven}.

$k(-x) = \frac{-x}{x^2+1} = -k(x)$: \omterm{def:g11:func:parity}{oneven}.
\end{solution}

\begin{solution}{exo:g11:func:4}
Lees de \omterm{def:g10:functions:table}{verlooptabel}. Op $\intcc{-3}{1}$ klimt $f$ van $-2$ naar $4$ en
passeert ze de waarde $1$ precies één keer; op $\intcc{1}{5}$ zakt ze van
$4$ naar $0$ en passeert ze $1$ nog een keer: de \omterm{def:g10:algebra:equation}{vergelijking} $f(x) = 1$
heeft $2$ oplossingen. Het \omterm{def:g10:functions:extrema}{maximum} van $f$ is $f(1) = 4 < 5$, dus heeft
$f(x) = 5$ geen oplossing.
\end{solution}

\begin{solution}{exo:g11:func:5}
Zij $2 \leq u < v$. Dan is
\[
f(v) - f(u) = v^2 - 4v - u^2 + 4u = (v-u)(v+u) - 4(v-u)
= (v - u)(u + v - 4).
\]
Beide factoren zijn positief: $v - u > 0$, en $u + v > 2 + 2 = 4$ want
$u \geq 2$ en $v > 2$. Bijgevolg is $f(v) > f(u)$: $f$ is strikt \omterm{def:g11:func:monotone}{stijgend} op
$\intco{2}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:6}
$f = \sqrt{u}$ met $u(x) = 5 - x$ \omterm{def:g10:functions:variations}{dalend} en niet-negatief op
$\intoc{-\infty}{5}$: $f$ is daar \omterm{def:g10:functions:variations}{dalend}
(\cref{prop:g11:func:transforms}).

$g = \frac1u$ met $u(x) = x^2 + 1$ \omterm{def:g11:func:monotone}{stijgend} en positief op
$\intco{0}{+\infty}$: $g$ is daar \omterm{def:g10:functions:variations}{dalend}.

$h = 3 - 2\sqrt x$: $\sqrt x$ is \omterm{def:g11:func:monotone}{stijgend}, vermenigvuldigen met $-2 < 0$
keert om, en $3$ optellen verandert niets: $h$ is \omterm{def:g10:functions:variations}{dalend} op
$\intco{0}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:7}
\emph{1.} $f$ staat in \omterm{thm:g11:quad:canonical}{toppuntsvorm}: \omterm{def:g10:functions:variations}{dalend} op $\intoc{-\infty}{2}$,
\omterm{def:g11:func:monotone}{stijgend} op $\intco{2}{+\infty}$, met \omterm{def:g10:functions:extrema}{minimum} $f(2) = -3$.

\emph{2.} $g = \frac{1}{u}$ met $u(x) = (x-2)^2 + 1 = f(x) + 4$. De
verschuiving met $4$ behoudt het verloop van $f$, en $u \geq 1 > 0$. Het
omgekeerde nemen keert het verloop om: $g$ is \omterm{def:g11:func:monotone}{stijgend} op
$\intoc{-\infty}{2}$ en \omterm{def:g10:functions:variations}{dalend} op $\intco{2}{+\infty}$, met \omterm{def:g10:functions:extrema}{maximum}
$g(2) = \frac{1}{1} = 1$.
\end{solution}

\begin{solution}{exo:g11:func:8}
Elke heuvelvormige kromme door de drie punten voldoet. Vervolgens: $f + 2$
gaat door $(0,3)$, $(2,5)$, $(4,3)$, met hetzelfde verloop (verticale
verschuiving); $-f$ gaat door $(0,-1)$, $(2,-3)$, $(4,-1)$ met
\emph{omgekeerd} verloop (de top wordt een dal); $2f$ gaat door $(0,2)$,
$(2,6)$, $(4,2)$, met hetzelfde verloop maar verticaal uitgerekt.
\end{solution}

\begin{solution}{exo:g11:func:9}
$2 - \frac{1}{x+1} = \frac{2(x+1) - 1}{x+1} = \frac{2x+1}{x+1} = f(x)$. Op
$\intoo{-1}{+\infty}$ is $u(x) = x + 1$ \omterm{def:g11:func:monotone}{stijgend} en positief, dus is
$\frac1u$ \omterm{def:g10:functions:variations}{dalend}, is $-\frac1u$ \omterm{def:g11:func:monotone}{stijgend}, en verandert $2$ optellen niets:
$f$ is strikt \omterm{def:g11:func:monotone}{stijgend} op $\intoo{-1}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:10}
\emph{1. Waar.} Is $u < v$, dan is
$(f+g)(v) - (f+g)(u) = \bigl(f(v)-f(u)\bigr) + \bigl(g(v)-g(u)\bigr)$ een
som van twee niet-negatieve getallen.

\emph{2. Niet waar.} $f(x) = g(x) = x$ zijn \omterm{def:g11:func:monotone}{stijgend} op $\R$, maar hun
product $x^2$ is niet \omterm{def:g11:func:monotone}{stijgend} op $\R$ (het daalt op $\intoc{-\infty}{0}$).
(De uitspraak wordt wel waar als beide \omterm{def:g11:func:function}{functies} ook niet-negatief zijn.)

\emph{3. Niet waar.} Hetzelfde tegenvoorbeeld: $f(x) = x$ is \omterm{def:g11:func:monotone}{stijgend} op
$\R$, maar $f^2(x) = x^2$ niet.
\end{solution}

\begin{solution}{exo:g11:func:11}
\emph{1.} Voor $0 < u < v$:
\[
f(v) - f(u) = (v - u) + \frac1v - \frac1u
= (v - u) + \frac{u - v}{uv}
= (v - u)\left(1 - \frac{1}{uv}\right)
= (v - u)\,\frac{uv - 1}{uv}.
\]

\emph{2.} Altijd is $v - u > 0$ en $uv > 0$. Is $0 < u < v \leq 1$, dan is
$uv < 1$, dus $uv - 1 < 0$ en $f(v) < f(u)$: $f$ is strikt \omterm{def:g10:functions:variations}{dalend} op
$\intoc{0}{1}$. Is $1 \leq u < v$, dan is $uv > 1$ en $f(v) > f(u)$: $f$ is
strikt \omterm{def:g11:func:monotone}{stijgend} op $\intco{1}{+\infty}$.

\emph{3.} $f$ daalt vóór $1$ en stijgt erna, dus is $f(x) \geq f(1) = 2$
voor alle $x > 0$, met gelijkheid precies in $x = 1$. (Deze ongelijkheid
duikt in vele gedaanten op; vergelijk \cref{ex:g11:quad:canonical} en de
raaklijnongelijkheden van \cref{ch:g12:deriv}.)
\end{solution}

\begin{solution}{pb:g11:func:1}
\textbf{1.} $x^4 - 3x^2$: \omterm{def:g11:func:parity}{even} ($(-x)^4 - 3(-x)^2 = x^4 - 3x^2$).
$x^3 - x$: \omterm{def:g11:func:parity}{oneven}. $x^2 + x$: geen van beide ($f(-1) = 0 \neq f(1) = 2$ en
ook $\neq -f(1)$). $\abs x$: \omterm{def:g11:func:parity}{even}. $\frac1x$: \omterm{def:g11:func:parity}{oneven}.

\textbf{2.} \omterm{def:g11:func:parity}{Even}: de kromme is symmetrisch om de $y$-as (spiegeling).
\omterm{def:g11:func:parity}{Oneven}: symmetrisch om de oorsprong (halve draai).

\textbf{3.} Zijn $f$ en $g$ \omterm{def:g11:func:parity}{oneven}, dan is
$(fg)(-x) = f(-x)g(-x) = (-f(x))(-g(x)) = (fg)(x)$: \omterm{def:g11:func:parity}{even}. De tabel: \omterm{def:g11:func:parity}{even}
$\times$ \omterm{def:g11:func:parity}{even} $=$ \omterm{def:g11:func:parity}{even}; \omterm{def:g11:func:parity}{even} $\times$ \omterm{def:g11:func:parity}{oneven} $=$ \omterm{def:g11:func:parity}{oneven}; \omterm{def:g11:func:parity}{oneven} $\times$
\omterm{def:g11:func:parity}{oneven} $=$ \omterm{def:g11:func:parity}{even} --- de tekenregel, maar dan met pariteiten.

\textbf{4.} \omterm{def:g11:func:parity}{Oneven} in $0$: $f(0) = -f(0)$, dus $f(0) = 0$. \omterm{def:g11:func:parity}{Even}:
$f(-3) = f(3) = 7$. Zowel \omterm{def:g11:func:parity}{even} als \omterm{def:g11:func:parity}{oneven}: $f(x) = f(-x) = -f(x)$ voor alle
$x$, dus $2f(x) = 0$: $f$ is de nulfunctie.

\textbf{5.} De rechterhelft ($x \geq 0$) legt een \omterm{def:g11:func:parity}{even functie} volledig
vast: de linkerhelft is haar spiegelbeeld. Noch \omterm{def:g11:func:parity}{even} noch \omterm{def:g11:func:parity}{oneven}:
$x^2 + x$ (vraag 1) --- allebei de symmetrieën falen in $x = 1$.

\textbf{6.} $E(x) = \frac{(x^3 + x^2 + 1) + (-x^3 + x^2 + 1)}{2} = x^2 + 1$
(\omterm{def:g11:func:parity}{even}), $O(x) = \frac{(x^3 + x^2 + 1) - (-x^3 + x^2 + 1)}{2} = x^3$
(\omterm{def:g11:func:parity}{oneven}); hun som is $f$.

\textbf{7.} $E(-x) = \frac{f(-x) + f(x)}{2} = E(x)$: \omterm{def:g11:func:parity}{even}.
$O(-x) = \frac{f(-x) - f(x)}{2} = -O(x)$: \omterm{def:g11:func:parity}{oneven}. En
$E(x) + O(x) = f(x)$ door rechtstreeks op te tellen: elke \omterm{def:g11:func:function}{functie} splitst.

\textbf{8.} Uit $E_1 + O_1 = E_2 + O_2$ volgt $E_1 - E_2 = O_2 - O_1$. Het
linkerlid is \omterm{def:g11:func:parity}{even}, het rechterlid \omterm{def:g11:func:parity}{oneven} --- die \omterm{def:g11:func:function}{functie} is dus allebei, en
bijgevolg nul (vraag 4): $E_1 = E_2$ en $O_1 = O_2$. Slechts één splitsing.

\textbf{9.} $E(x) = \frac12\left(\frac{1}{1 - x} + \frac{1}{1 + x}\right)
= \frac12 \cdot \frac{(1 + x) + (1 - x)}{1 - x^2} = \frac{1}{1 - x^2}$, en
$O(x) = \frac12 \cdot \frac{(1 + x) - (1 - x)}{1 - x^2} =
\frac{x}{1 - x^2}$. Controle: $E$ is \omterm{def:g11:func:parity}{even}, $O$ is \omterm{def:g11:func:parity}{oneven}, en hun som is
$\frac{1 + x}{1 - x^2} = \frac{1 + x}{(1 - x)(1 + x)} = \frac{1}{1 - x} =
f(x)$.

\textbf{10.} Doordat elk stuk aan een symmetrie gehoorzaamt, volstaat voor
elk de helft van het werk (en de helft van de gegevens): los het aan één
kant op en spiegel naar de andere --- en veel bewerkingen (integreren,
reeksen, signaalanalyse) behandelen symmetrische stukken veel eenvoudiger
dan ruwe \omterm{def:g11:func:function}{functies}.

\textbf{11.} $(x-3)^2$: $3$ naar rechts geschoven. $x^2 + 2$: $2$ omhoog
geschoven. $(x-3)^2 + 2$: allebei. $-x^2$: ondersteboven gekeerd. $2x^2$:
verticaal met $2$ uitgerekt (smallere schaal).

\textbf{12.} Voor $x \geq 2$: $(x - 2) + (x + 2) = 2x$. Voor
$-2 \leq x \leq 2$: $(2 - x) + (x + 2) = 4$. Voor $x \leq -2$: $-2x$. De
kromme daalt met \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $-2$, loopt tussen $-2$ en $2$ vlak op
hoogte $4$, en klimt daarna met \omterm{def:g10:reffunc:affine}{richtingscoëfficiënt} $2$: een dal met een
vlakke bodem.

\textbf{13.} Verticale verschuiving: $f(x) + 2$ blijft \omterm{def:g11:func:parity}{even}
($f(-x) + 2 = f(x) + 2$). Verticale rek: $2f(x)$ blijft \omterm{def:g11:func:parity}{even}. Horizontale
verschuiving: $(x - 3)^2$ is niet \omterm{def:g11:func:parity}{even} ($f(-1) = 16 \neq 4 = f(1)$):
zijwaarts schuiven vernielt de spiegeling.

\textbf{14.} Op de vlakke bodem is de waarde $4 \neq 6$. Voor $x \geq 2$:
$2x = 6$, dus $x = 3$. Voor $x \leq -2$: $-2x = 6$, dus $x = -3$.
Oplossingen: $\pm 3$.

\textbf{15.} $f(x) = a(x - 2)^2 - 3$ met $f(0) = 4a - 3 = 1$: $a = 1$, dus
$f(x) = (x - 2)^2 - 3$.

\textbf{16.} $g(v) - g(u) = (v - u) + 4\,\frac{u - v}{uv}
= (v - u)\,\frac{uv - 4}{uv}$: negatief voor $u < v$ met $uv < 4$, positief
met $uv > 4$: \omterm{def:g10:functions:variations}{dalend} op $\intoc{0}{2}$, \omterm{def:g11:func:monotone}{stijgend} op $\intco{2}{+\infty}$;
\omterm{def:g10:functions:extrema}{minimum} in $x = 2$, met waarde $g(2) = 4$.

\textbf{17.} Zijden $x$ en $\frac{16}{x}$:
$P(x) = 2\left(x + \frac{16}{x}\right)$, minimaal (de machine van vraag 16
met $4 \to 16$, keerpunt $\sqrt{16} = 4$) in $x = 4$: het \emph{vierkant}
$4 \times 4$, met omtrek $16$ --- het antwoord van Dido, nu bewezen voor elke
reële zijdelengte.

\textbf{18.} Op $\intco{0}{+\infty}$ stijgt $x \mapsto x^2 + 1$, en de
vierkantswortel stijgt, dus stijgt de samenstelling. En
$h(-x) = \sqrt{x^2 + 1} = h(x)$: $h$ is \omterm{def:g11:func:parity}{even}, dus daalt ze door de
spiegelsymmetrie op $\intoc{-\infty}{0}$. \omterm{def:g10:functions:extrema}{Minimum} $h(0) = 1$.

\textbf{19.} Het kwadraat van de afstand van het punt $(x, \sqrt x)$ tot
$(3, 0)$ is $d^2(x) = (x - 3)^2 + x = x^2 - 5x + 9$, minimaal in
$x = \frac52$: het dichtstbijzijnde punt is
$\left(\frac52, \sqrt{\frac52}\right)$, op afstand
$\sqrt{\frac{11}{4}} = \frac{\sqrt{11}}{2}$. Het kwadraat minimaliseren mag,
omdat kwadrateren strikt \omterm{def:g11:func:monotone}{stijgend} is op de positieve getallen: $d$ en $d^2$
bereiken hun laagste punt op dezelfde plaats.

\textbf{20.} Grondstof: de referentiekrommen, uit het hoofd gekend.
Machines: verschuivingen, spiegelingen en rekkingen die uit oude \omterm{def:g10:functions:graph}{grafieken}
nieuwe bouwen. Kwaliteitscontrole: de pariteit en de splitsing in een \omterm{def:g11:func:parity}{even}
en een \omterm{def:g11:func:parity}{oneven} deel, die elke studie halveren. Meetbank: het \omterm{def:g10:algebra:expand}{ontbinden} van
$f(v) - f(u)$, dat verloop en minima rechtstreeks afleest. Het krachtige
gereedschap van het volgende hoofdstuk: de \emph{afgeleide}, die ze in één
oogopslag afleest.
\end{solution}
