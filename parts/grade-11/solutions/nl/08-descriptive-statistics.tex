\section*{Hoofdstuk \ref{ch:g11:stat} --- Beschrijvende statistiek}

\begin{solution}{exo:g11:stat:1}
$n = 8$, som $71$: \omterm{def:g11:stat:mean}{gemiddelde} $\bar x = \frac{71}{8} = 8.875$. \omterm{def:g11:stat:median}{Mediaan}: het
\omterm{prop:g10:coordgeom:midpoint}{midden} van de $4$de en de $5$de waarde, $\frac{8 + 9}{2} = 8.5$.
\omterm{def:g11:stat:quartiles}{Kwartielen}: een kwart van $8$ is $2$, dus is $Q_1$ de $2$de waarde, $7$;
drie kwart is $6$, dus is $Q_3$ de $6$de waarde, $10$.
\omterm{def:g11:stat:quartiles}{Interkwartielafstand}: $10 - 7 = 3$.
\end{solution}

\begin{solution}{exo:g11:stat:2}
Som $40$, dus $\bar x = 5$. Som van de kwadraten:
$4 + 16 + 16 + 16 + 25 + 25 + 49 + 81 = 232$. Met de rekenformule:
\[
V = \frac{232}{8} - 5^2 = 29 - 25 = 4,
\qquad
\sigma = 2 .
\]
\end{solution}

\begin{solution}{exo:g11:stat:3}
\omterm{def:g11:stat:mean}{Gemiddelde}:
\[
\bar x = \frac{1 \times 6 + 2 \times 9 + 3 \times 8 + 4 \times 10
+ 5 \times 9 + 6 \times 8}{50}
= \frac{181}{50} = 3.62 .
\]
\omterm{def:g11:stat:median}{Mediaan}: de cumulatieve \omterm{def:g10:stats:series}{frequenties} zijn $6, 15, 23, 33, \dots$, zodat de
$25$ste en de $26$ste gesorteerde uitkomst allebei $4$ zijn: de \omterm{def:g11:stat:median}{mediaan} is
$4$.
\end{solution}

\begin{solution}{exo:g11:stat:4}
Het totaal van de $15$ resultaten is $15 \times 11.2 = 168$; met de
nieuwkomer wordt dat $\frac{168 + 18}{16} = \frac{186}{16} = 11.625$.
\end{solution}

\begin{solution}{exo:g11:stat:5}
$\frac{6 + 8 + x + 12 + 14}{5} = 10$ geeft $40 + x = 50$, dus $x = 10$. De
afwijkingen van $6, 8, 10, 12, 14$ ten opzichte van het \omterm{def:g11:stat:mean}{gemiddelde} $10$ zijn
$-4, -2, 0, 2, 4$, met kwadraten die optellen tot $40$: $V = \frac{40}{5} =
8$ en $\sigma = 2\sqrt2 \approx 2.83$.
\end{solution}

\begin{solution}{exo:g11:stat:6}
Volgens \cref{prop:g11:stat:affine} met $a = \frac12$ en $b = 4$: het nieuwe
\omterm{def:g11:stat:mean}{gemiddelde} is $\frac{12.4}{2} + 4 = 10.2$; de nieuwe \omterm{def:g11:stat:variance}{standaardafwijking} is
$\frac{2.5}{2} = 1.25$.
\end{solution}

\begin{solution}{exo:g11:stat:7}
Het samengevoegde \omterm{def:g11:stat:mean}{gemiddelde} is het \emph{gewogen} \omterm{def:g11:stat:mean}{gemiddelde}
\[
\frac{20 \times 10 + 30 \times 15}{50} = \frac{650}{50} = 13,
\]
en niet $12.5$: groep B levert meer waarden en trekt het gezamenlijke
\omterm{def:g11:stat:mean}{gemiddelde} dus naar zich toe.
\end{solution}

\begin{solution}{exo:g11:stat:8}
Beide lijnen zitten gemiddeld bijna op doel ($1.002$ en $1.001$ kg), maar de
\omterm{def:g11:stat:variance}{standaardafwijking} van lijn 2 is drie keer groter: haar zakken variëren veel
meer, wat veel merkbaarder te licht en te zwaar gevulde zakken oplevert. Lijn
2 moet als eerste aangepakt worden --- niet alleen het \omterm{def:g11:stat:mean}{gemiddelde}, maar
vooral de regelmaat vraagt herstel.
\end{solution}

\begin{solution}{exo:g11:stat:9}
$\bar x = \frac{70}{10} = 7$; $V = \frac{540}{10} - 7^2 = 54 - 49 = 5$;
$\sigma = \sqrt5 \approx 2.24$.
\end{solution}

\begin{solution}{exo:g11:stat:10}
\emph{1.} Met $23$: som $50$, \omterm{def:g11:stat:mean}{gemiddelde} $5$; \omterm{def:g11:stat:median}{mediaan} $=$ het \omterm{prop:g10:coordgeom:midpoint}{midden} van de
$5$de en de $6$de waarde $= 3$. Zonder $23$: som $27$, \omterm{def:g11:stat:mean}{gemiddelde} $3$;
\omterm{def:g11:stat:median}{mediaan} (de $5$de van $9$ waarden) $= 3$.

\emph{2.} Met $23$ ($n = 10$): $Q_1$ is de $3$de waarde ($2$) en $Q_3$ de
$8$ste ($4$): \omterm{def:g11:stat:quartiles}{interkwartielafstand} $2$. Zonder ($n = 9$): $Q_1$ is de $3$de
waarde ($2$) en $Q_3$ de $7$de ($4$): opnieuw $2$.

\emph{3.} De ene uitschieter sleurt het \omterm{def:g11:stat:mean}{gemiddelde} van $3$ naar $5$, maar
laat de \omterm{def:g11:stat:median}{mediaan} en de \omterm{def:g11:stat:quartiles}{interkwartielafstand} onaangeroerd: die twee zijn
\emph{robuust}. Kan een gegevensreeks afwijkende waarden bevatten, meld dan
de \omterm{def:g11:stat:median}{mediaan} en de \omterm{def:g11:stat:quartiles}{interkwartielafstand} in plaats van het \omterm{def:g11:stat:mean}{gemiddelde} en de
\omterm{def:g11:stat:variance}{standaardafwijking}.
\end{solution}

\begin{solution}{pb:g11:stat:1}
\textbf{1.} \omterm{def:g11:stat:mean}{Gemiddelde} $\bar x = \frac{25}{5} = 5$. Gekwadrateerde
afwijkingen: $1, 9, 1, 0, 9$: \omterm{def:g11:stat:variance}{variantie} $\frac{20}{5} = 4$, dus $\sigma = 2$.

\textbf{2.} \omterm{def:g11:stat:mean}{Gemiddelde} van de kwadraten:
$\frac{16 + 64 + 36 + 25 + 4}{5} = 29$; min $\bar x^2 = 25$: \omterm{def:g11:stat:variance}{variantie} $4$.
Hetzelfde antwoord, half het werk.

\textbf{3.} \omterm{def:g11:stat:mean}{Gemiddelde} $= \frac{3 \times 8 + 5 \times 10 + 2 \times 14}{10}
= 10.2$; \omterm{def:g11:stat:variance}{variantie} $= \frac{3 \times 64 + 5 \times 100 + 2 \times 196}{10} -
10.2^2 = 108.4 - 104.04 = 4.36$; $\sigma \approx 2.09$.

\textbf{4.} \omterm{def:g11:stat:mean}{Gemiddelde}: $1.8 \times 20 + 32 = 68\,^\circ$F; $\sigma$:
$1.8 \times 3 = 5.4\,^\circ$F --- de verschuiving $+32$ spreidt niets.

\textbf{5.} $b$ optellen: elke afwijking $x_i + b - (\bar x + b) =
x_i - \bar x$ blijft ongewijzigd, en dus ook de \omterm{def:g11:stat:variance}{variantie}. Met $a$
vermenigvuldigen: de afwijkingen worden met $a$ vermenigvuldigd, hun
kwadraten met $a^2$, de \omterm{def:g11:stat:variance}{variantie} met $a^2$, en $\sigma$ met
$\sqrt{a^2} = \abs a$.

\textbf{6.} $z_A = \frac{15 - 12}{2} = 1.5$; $z_B = \frac{13 - 8}{4} =
1.25$. Alice steekt verder boven haar groep uit: de $15$ op het moeilijke
examen verslaat de $13$ op het makkelijke.

\textbf{7.} De \omterm{pb:g11:stat:1}{z-scores} zijn $-0.5,\ 1.5,\ 0.5,\ 0,\ -1.5$: \omterm{def:g11:stat:mean}{gemiddelde} $0$,
\omterm{def:g11:stat:variance}{standaardafwijking} $1$. In het algemeen trekt standaardiseren eerst $\bar x$
af (waardoor het \omterm{def:g11:stat:mean}{gemiddelde} $0$ wordt en $\sigma$ onaangeroerd blijft) en
deelt het daarna door $\sigma$ (waardoor de nieuwe $\sigma$ gelijk aan $1$
wordt): twee keer vraag 5.

\textbf{8.} $z = 2$: $100 + 2 \times 15 = 130$. En
$z(76) = \frac{76 - 100}{15} = -1.6$.

\textbf{9.} De baby ligt $2.5$ \omterm{def:g11:stat:variance}{standaardafwijkingen} onder het \omterm{def:g11:stat:mean}{gemiddelde}
gewicht \emph{van baby's van zijn leeftijd} --- alarmerend licht voor zijn
groep. Ruwe kilogrammen laten zich niet over leeftijden heen vergelijken
(alle baby's komen aan); \omterm{pb:g11:stat:1}{z-scores} meten elk kind tegenover zijn eigen
leeftijdsgroep, één meetlat voor alle leeftijden.

\textbf{10.} \omterm{def:g11:stat:mean}{Gemiddelde} $5$; \omterm{def:g11:stat:mean}{gemiddelde} van de kwadraten
$\frac{622}{10} = 62.2$, \omterm{def:g11:stat:variance}{variantie} $37.2$, $\sigma \approx 6.10$;
$z(23) = \frac{18}{6.10} \approx 2.95$: net onder de alarmdrempel --- de
toets knippert nauwelijks.

\textbf{11.} Zonder $23$: \omterm{def:g11:stat:mean}{gemiddelde} $3$, \omterm{def:g11:stat:variance}{variantie}
$\frac{93}{9} - 9 \approx 1.33$, $\sigma \approx 1.15$; tegenover die schone
meetlat scoort $23$ een $z \approx \frac{20}{1.15} \approx 17$: een monster.
De uitschieter blies zowel het \omterm{def:g11:stat:mean}{gemiddelde} als $\sigma$ op en camoufleerde
zichzelf --- en daarom beveelt \cref{exo:g11:stat:10} robuuste kengetallen
(\omterm{def:g11:stat:median}{mediaan}, \omterm{def:g11:stat:quartiles}{interkwartielafstand}) aan zodra besmetting dreigt.

\textbf{12.} Houd in de variantiesom
$\sigma^2 = \frac1N \sum (x_i - \bar x)^2$ alleen de termen over met
$\abs{x_i - \bar x} \geq k\sigma$: dat zijn er $pN$, elk minstens
$(k\sigma)^2$, dus $\sigma^2 \geq \frac{1}{N} \cdot pN \cdot k^2\sigma^2 =
p\,k^2\sigma^2$: deel door $k^2 \sigma^2$ (met spreiding verschillend van
nul) en er blijft $p \leq \frac{1}{k^2}$ over.

\textbf{13.} Verder dan $2\sigma$: hoogstens $\frac14 = 25\,\%$ van eender
welke gegevensreeks; verder dan $3\sigma$: hoogstens
$\frac19 \approx 11\,\%$. De gegevens van vraag 1: verder dan
$2\sigma = 4$ van het \omterm{def:g11:stat:mean}{gemiddelde} $5$ betekent $\leq 1$ of $\geq 9$: geen
enkele van de vijf waarden --- $0\,\% \leq 25\,\%$, zoals beloofd.

\textbf{14.} Buiten $\intcc{49.7}{50.3}$ betekent verder dan $3\sigma$:
hoogstens $\frac19$ van de productie, \emph{welke} vorm de verdeling ook
heeft --- Chebyshev vraagt niets meer dan een \omterm{def:g11:stat:mean}{gemiddelde} en een $\sigma$. Een
klokvormige productie concentreert zich veel beter (ongeveer $0.3\,\%$
verder dan $3\sigma$): de normale verdeling, jaar 12.

\textbf{15.} $-19\,\%$ ligt $24$ punten onder het \omterm{def:g11:stat:mean}{gemiddelde} $5$. Fonds B:
$24 = 2\sigma_B$: Chebyshev laat tot $\frac14$ van de jaren zo slecht toe.
Fonds A: $24 = 12\sigma_A$: hoogstens $\frac{1}{144}$. Hetzelfde \omterm{def:g11:stat:mean}{gemiddelde},
onvergelijkbare gevaren --- $\sigma$ is wat de financiële wereld
\emph{risico} (of volatiliteit) noemt.

\textbf{16.} Werk uit rond $\bar x$:
$(x_i - c)^2 = \left((x_i - \bar x) + (\bar x - c)\right)^2
= (x_i - \bar x)^2 + 2(\bar x - c)(x_i - \bar x) + (\bar x - c)^2$.
Middelen: de middelste term sterft (de afwijkingen van het \omterm{def:g11:stat:mean}{gemiddelde} tellen
op tot nul), zodat $f(c) = \sigma^2 + (\bar x - c)^2$ overblijft --- minimaal
precies in $c = \bar x$, met \omterm{def:g10:functions:extrema}{minimum} $\sigma^2$. Het \omterm{def:g11:stat:mean}{gemiddelde} is het
kleinstekwadratenmiddelpunt, zoals de \omterm{def:g11:stat:median}{mediaan} het middelpunt van de kleinste
absolute afwijking is (\cref{pb:g10:stats:1}).

\textbf{17.} Rechtstreeks: de afwijkingen van $6$ zijn $-2, 2, 0, -1, -4$,
met kwadraten $4, 4, 0, 1, 16$: $f(6) = \frac{25}{5} = 5$. Via de formule:
$\sigma^2 + (5 - 6)^2 = 4 + 1 = 5$. Overeenstemming.

\textbf{18.} Beide hebben \omterm{def:g11:stat:mean}{gemiddelde} $7$; de \omterm{def:g11:stat:variance}{varianties} zijn
$P$: $\frac{16 + 4 + 0 + 4 + 16}{5} = 8$ en
$Q$: $\frac{36 + 0 + 0 + 0 + 36}{5} = 14.4$: verschillende spreidingen
achter één \omterm{def:g11:stat:mean}{gemiddelde}. Voor $\{7 - a, 7, 7, 7, 7 + a\}$ geeft de \omterm{def:g11:stat:variance}{variantie}
$\frac{2a^2}{5} = 8$ dat $a = \sqrt{20} \approx 4.47$: de reeks
$\{2.53, 7, 7, 7, 11.47\}$ draagt precies het \omterm{def:g11:stat:mean}{gemiddelde} en de \omterm{def:g11:stat:variance}{variantie} van
$P$ met een compleet ander profiel --- kengetallen persen samen, en
samenpersen verliest.

\textbf{19.} $\sigma^2 = 0$ betekent dat een som van kwadraten
$(x_i - \bar x)^2$ verdwijnt; kwadraten zijn $\geq 0$, dus dwingt een som
gelijk aan nul elke term nul te zijn: $x_i = \bar x$ voor alle $i$ --- alle
waarden gelijk.

\textbf{20.} \omterm{prop:g10:coordgeom:midpoint}{Middens} zeggen waar de gegevens wonen; spreidingen hoe los; de
\omterm{pb:g11:stat:1}{z-score} legt één meetlat over alle gegevensreeksen, examens, leeftijden en
munteenheden; Chebyshev zet $\sigma$ om in een onvoorwaardelijke garantie; en
vraag 18 houdt ons eerlijk --- geen handvol getallen bevat ooit het hele
verhaal. Volgend jaar verscherpt de klokkromme de meetlat tot exacte kansen.
\end{solution}
