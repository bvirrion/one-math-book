\section*{Capítulo \ref{ch:g11:quad} --- Funciones y ecuaciones de segundo grado}

\begin{solution}{exo:g11:quad:1}
$x^2-5x+6$: $\Delta = 25 - 24 = 1$, raíces $\frac{5 \pm 1}{2}$, es decir,
$2$ y $3$.

$2x^2+3x-2$: $\Delta = 9 + 16 = 25$, raíces $\frac{-3 \pm 5}{4}$, es
decir, $\frac12$ y $-2$.

$x^2+2x+5$: $\Delta = 4 - 20 = -16 < 0$, sin solución real.

$9x^2-6x+1$: $\Delta = 36 - 36 = 0$, una única \omterm{def:g11:quad:discriminant}{raíz} (doble),
$x_0 = \frac{6}{18} = \frac13$.
\end{solution}

\begin{solution}{exo:g11:quad:2}
$f(x) = x^2 - 6x + 11 = (x-3)^2 - 9 + 11 = (x-3)^2 + 2$: vértice $(3, 2)$.

$g(x) = -2x^2+4x+1 = -2(x^2 - 2x) + 1 = -2\bigl((x-1)^2 - 1\bigr) + 1
= -2(x-1)^2 + 3$: vértice $(1, 3)$.

$h(x) = 3x^2 + 6x = 3(x^2 + 2x) = 3\bigl((x+1)^2 - 1\bigr)
= 3(x+1)^2 - 3$: vértice $(-1, -3)$.
\end{solution}

\begin{solution}{exo:g11:quad:3}
$x^2 - 3x - 10$: $\Delta = 9 + 40 = 49$, raíces $\frac{3 \pm 7}{2} = -2$ y
$5$; el coeficiente principal es positivo, así que el trinomio es negativo
estrictamente entre las raíces: el conjunto de soluciones es
$\intoo{-2}{5}$.

$2x^2 + x + 3$: $\Delta = 1 - 24 = -23 < 0$ y $a = 2 > 0$: siempre
positivo, el conjunto de soluciones es $\R$.

$-x^2 + 6x - 9 = -(x-3)^2$: es $\geq 0$ solo donde $(x-3)^2 \leq 0$, es
decir, en $x = 3$. El conjunto de soluciones es $\{3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:4}
$x^2 - 7x + 6$: $1 - 7 + 6 = 0$, luego $1$ es \omterm{def:g11:quad:discriminant}{raíz}; el producto de las
raíces es $6$, así que la otra \omterm{def:g11:quad:discriminant}{raíz} es $6$.

$2x^2 + 5x - 7$: $2 + 5 - 7 = 0$, luego $1$ es \omterm{def:g11:quad:discriminant}{raíz}; el producto es
$\frac{c}{a} = -\frac72$, así que la otra \omterm{def:g11:quad:discriminant}{raíz} es $-\frac72$.

$x^2 + 4x - 5$: $1 + 4 - 5 = 0$, luego $1$ es \omterm{def:g11:quad:discriminant}{raíz}; el producto es $-5$,
así que la otra \omterm{def:g11:quad:discriminant}{raíz} es $-5$.
\end{solution}

\begin{solution}{exo:g11:quad:5}
Los dos números son las soluciones de $x^2 - 14x + 45 = 0$
(\cref{prop:g11:quad:vieta}); $\Delta = 196 - 180 = 16$ y las raíces son
$\frac{14 \pm 4}{2} = 9$ y $5$.

Para el rectángulo: la mitad del perímetro da largo $+$ ancho $= 17$, y el
área da su producto, $60$. Son las soluciones de $x^2 - 17x + 60 = 0$;
$\Delta = 289 - 240 = 49$, raíces $\frac{17 \pm 7}{2} = 12$ y $5$. El
rectángulo mide $12 \times 5$.
\end{solution}

\begin{solution}{exo:g11:quad:6}
$\Delta = (m-2)^2 - 4$. Hay exactamente una solución cuando $\Delta = 0$:
$(m-2)^2 = 4$, luego $m - 2 = \pm 2$, es decir, $m = 0$ o $m = 4$. No hay
solución cuando $\Delta < 0$: $(m-2)^2 < 4$, es decir, $-2 < m - 2 < 2$, o
sea $0 < m < 4$.
\end{solution}

\begin{solution}{exo:g11:quad:7}
El vértice está en $t = -\frac{20}{2 \times (-5)} = 2$, y
$h(2) = -20 + 40 + 1 = 21$: altura máxima de $21$ m, alcanzada a los $2$
segundos.

$h(t) \geq 16$ equivale a $-5t^2 + 20t - 15 \geq 0$, es decir (dividiendo
entre $-5$, lo que invierte la desigualdad), $t^2 - 4t + 3 \leq 0$. Las
raíces de $t^2-4t+3$ son $1$ y $3$ ($1 - 4 + 3 = 0$, producto $3$), así
que $t^2 - 4t + 3 \leq 0$ exactamente en $\intcc{1}{3}$: la pelota está al
menos a $16$ m de altura entre los instantes $t = 1$ s y $t = 3$ s.
\end{solution}

\begin{solution}{exo:g11:quad:8}
Con $X = x^2$: $X^2 - 13X + 36 = 0$, $\Delta = 169 - 144 = 25$, luego
$X = \frac{13 \pm 5}{2} = 9$ o $4$. Volviendo a $x$: $x^2 = 9$ da
$x = \pm 3$, y $x^2 = 4$ da $x = \pm 2$. El conjunto de soluciones es
$\{-3, -2, 2, 3\}$.
\end{solution}

\begin{solution}{exo:g11:quad:9}
La condición es $x^2 < x + 2$, es decir, $x^2 - x - 2 < 0$. Como
$x^2 - x - 2 = (x+1)(x-2)$ (raíces $-1$ y $2$), la desigualdad se cumple
estrictamente entre las raíces: $x \in \intoo{-1}{2}$. En un esbozo, la
\omterm{def:g11:quad:parabola}{parábola} $y = x^2$ se hunde por debajo de la recta $y = x+2$ exactamente
entre sus dos puntos de corte, $(-1, 1)$ y $(2, 4)$.
\end{solution}

\begin{solution}{exo:g11:quad:10}
Sea $x > 0$ tal que $x + \frac1x = \frac{13}{6}$. Multiplicando por
$6x > 0$: $6x^2 + 6 = 13x$, es decir, $6x^2 - 13x + 6 = 0$. Entonces
$\Delta = 169 - 144 = 25$ y $x = \frac{13 \pm 5}{12}$, de donde
$x = \frac32$ o $x = \frac23$. Los dos son positivos e inversos el uno del
otro, como cabía esperar: el número es $\frac32$ (o $\frac23$).
\end{solution}

\begin{solution}{exo:g11:quad:11}
\emph{1.} Los cortes corresponden a las soluciones de $x^2 = mx - 1$, es
decir, $x^2 - mx + 1 = 0$, cuyo \omterm{def:g11:quad:discriminant}{discriminante} es $\Delta = m^2 - 4$. Dos
puntos cuando $\abs{m} > 2$; exactamente uno cuando $m = \pm 2$; ninguno
cuando $-2 < m < 2$.

\emph{2.} Para $m = 2$: $x^2 - 2x + 1 = (x-1)^2 = 0$, así que el único
punto de contacto es $(1, 1)$. Para $m = -2$: $(x+1)^2 = 0$, punto de
contacto $(-1, 1)$. En los dos casos la recta corta a la \omterm{def:g11:quad:parabola}{parábola} en un
solo punto sin atravesarla: es la tangente allí, y en efecto su \omterm{def:g10:reffunc:affine}{pendiente}
$m = \pm 2$ coincide con el valor en $x = \pm 1$ de la derivada $2x$ de
$x^2$ (\cref{ch:g11:deriv}).
\end{solution}

\begin{solution}{pb:g11:quad:1}
\textbf{1.} $x^2 - 5x + 6$: $\Delta = 1$, raíces $2$ y $3$.
$2x^2 + x - 6$: $\Delta = 1 + 48 = 49$, raíces $\frac{-1 \pm 7}{4}$:
$\frac32$ y $-2$. $x^2 + x + 1$: $\Delta = -3 < 0$: sin solución real.

\textbf{2.} $\Delta = 16 + 48 = 64$: raíces $3$ y $-1$, luego
$2x^2 - 4x - 6 = 2(x - 3)(x + 1)$; \omterm{thm:g11:quad:canonical}{forma canónica} $2(x - 1)^2 - 8$. La
forma factorizada enseña las raíces; la canónica, el vértice $(1, -8)$; y
la desarrollada, el corte $-6$ con el eje $y$.

\textbf{3.} Raíces $1$ y $3$, coeficiente principal negativo: el trinomio
es $\geq 0$ \emph{entre} las raíces: $x \in \intcc{1}{3}$.

\textbf{4.} Los números son las raíces de $x^2 - 7x + 12 = 0$: $3$ y $4$.
Si $2$ es \omterm{def:g11:quad:discriminant}{raíz} de $x^2 - 5x + c$, la otra \omterm{def:g11:quad:discriminant}{raíz} $r$ cumple $2 + r = 5$ (la
suma), luego $r = 3$, y $c = 2 \times 3 = 6$ (el producto): Viète lo lee
de un vistazo.

\textbf{5.} $\Delta = m^2 - 36$: \omterm{def:g11:quad:discriminant}{raíz} doble para $m = \pm 6$ (la \omterm{def:g11:quad:discriminant}{raíz} es
$\mp 3$); ninguna \omterm{def:g11:quad:discriminant}{raíz} para $-6 < m < 6$; dos raíces para $\abs m > 6$.

\textbf{6.} $PF^2 = x^2 + \left(x^2 - \frac14\right)^2
= x^2 + x^4 - \frac{x^2}{2} + \frac{1}{16}
= x^4 + \frac{x^2}{2} + \frac{1}{16}
= \left(x^2 + \frac14\right)^2$. Por tanto,
$PF = x^2 + \frac14$ (los dos factores son positivos).

\textbf{7.} La distancia de $P(x, x^2)$ a la recta horizontal
$y = -\frac14$ es $x^2 + \frac14$, exactamente $PF$. Todo punto de la
\omterm{def:g11:quad:parabola}{parábola} equidista del \omterm{pb:g11:quad:1}{foco} y de la \omterm{pb:g11:quad:1}{directriz}.

\textbf{8.} $PF^2 = x^2 + \left(y - \frac14\right)^2$ y
$\operatorname{dist}^2 = \left(y + \frac14\right)^2$. La igualdad da
$x^2 + y^2 - \frac y2 + \frac{1}{16} = y^2 + \frac y2 + \frac1{16}$, luego
$x^2 = y$: el lugar geométrico es exactamente la \omterm{def:g11:quad:parabola}{parábola}. Un punto y una
recta generan la curva, igual que un punto y una distancia generaban la
circunferencia.

\textbf{9.} Para $y = \frac{x^2}{4f}$, el mismo desarrollo con $F(0, f)$
da $PF^2 = x^2 + \left(\frac{x^2}{4f} - f\right)^2
= \left(\frac{x^2}{4f} + f\right)^2$: la distancia a $F$ es igual a la
distancia a $y = -f$. Antena: $9 = \frac{30^2}{4f}$ da $4f = 100$,
$f = 25$ cm: el receptor cuelga a un cuarto de metro sobre el vértice, es
decir, fuera del casquete de $9$ cm, como se ve en las antenas reales.

\textbf{10.} Un frente de onda que llega paralelo al eje ha \omterm{def:g10:stats:quartiles}{recorrido}
distancias iguales para alcanzar el nivel de la \omterm{pb:g11:quad:1}{directriz}; la
equidistancia convierte ``igual distancia a la \omterm{pb:g11:quad:1}{directriz}'' en ``igual
distancia a $F$'': todas las señales reflejadas llegan a $F$ \emph{a la
vez} y se refuerzan; ahí es donde hay que escuchar. La demostración
honrada necesita la \omterm{def:g10:reffunc:affine}{pendiente} de la tangente en cada punto, o sea, la
derivada del \cref{ch:g11:deriv}, ya entrevista en el
\cref{exo:g11:quad:11}.

\textbf{11.} $y = x\left(1 - \frac{x}{20}\right)$: cae en $x = 20$ m. El
vértice está a mitad de camino: $x = 10$, con altura
$y(10) = 10 - 5 = 5$ m.

\textbf{12.} $y(4) = 4 - \frac{16}{20} = 3.2$ m: supera el muro de $3$ m
con margen y se estrella contra uno de $4$ m.

\textbf{13.} $y(10) = 10 - \frac{100}{40} = 7.5$ m $> 7$: el camión pasa a
$10$ m del centro. $y(15) = 10 - \frac{225}{40} = 4.375$ m: a $15$ m, no.

\textbf{14.} $R(p) = -2p^2 + 120p$: vértice en
$p = \frac{120}{4} = 30$ euros, con ingresos
$R(30) = 30 \times 60 = 1\,800$ euros.

\textbf{15.} $y(320) = 143 \times \left(\frac{320}{640}\right)^2
= \frac{143}{4} \approx 36$ m por encima del punto más bajo.

\textbf{16.} $x^2 = mx - 1$ da $x^2 - mx + 1 = 0$ y $\Delta = m^2 - 4$:
dos puntos para $\abs m > 2$, ninguno para $\abs m < 2$ y tangencia para
$m = \pm 2$, con contacto en $x = \frac m2 = \pm 1$: las tangentes son
$y = 2x - 1$ en $(1, 1)$ e $y = -2x - 1$ en $(-1, 1)$.

\textbf{17.} $x^2 = mx + 1$ da $\Delta = m^2 + 4 > 0$ para todo $m$:
siempre hay dos cortes. Desde un punto interior, toda recta atraviesa dos
veces la curva; las tangentes se trazan solo desde fuera, exactamente como
en las circunferencias.

\textbf{18.} $y = \frac14$ corta a $y = x^2$ en $x = \pm\frac12$: la
cuerda mide $1$. Para $y = \frac{x^2}{4f}$: $y = f$ da $x^2 = 4f^2$,
$x = \pm 2f$: longitud $4f$; la anchura focal es cuatro veces la distancia
focal (los diseñadores de antenas la llaman lado recto).

\textbf{19.} Un punto de $y = x^2$ es $(a, a^2)$; su \omterm{def:g10:functions:function}{imagen} es
$(X, Y) = (ka, ka^2)$, y $\frac{X^2}{k} = \frac{k^2a^2}{k} = ka^2 = Y$: la
curva \omterm{def:g10:functions:function}{imagen} es exactamente $y = \frac{x^2}{k}$. Eligiendo $k = 4f$ se
alcanza cualquier \omterm{def:g11:quad:parabola}{parábola}: todas son ampliaciones de una misma curva
maestra. (Las circunferencias también son todas semejantes entre sí; las
elipses, no: el achatamiento importa, el aumento no.)

\textbf{20.} Una \omterm{def:g10:algebra:equation}{ecuación} con tres disfraces legibles; el lugar geométrico
de los puntos que equidistan de un punto y de una recta; el reflector que
reúne los rayos paralelos en su \omterm{pb:g11:quad:1}{foco} y el cable que sostiene un tablero
uniforme; la trayectoria de cualquier pelota lanzada; y, salvo ampliación,
una sola curva: una \omterm{def:g11:quad:parabola}{parábola}, en todas partes.
\end{solution}
