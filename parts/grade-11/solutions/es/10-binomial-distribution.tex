\section*{Capítulo \ref{ch:g11:binom} --- La distribución binomial}

\begin{solution}{exo:g11:binom:1}
Filas $6$ y $7$:
\[
1,\ 6,\ 15,\ 20,\ 15,\ 6,\ 1
\qquad\text{y}\qquad
1,\ 7,\ 21,\ 35,\ 35,\ 21,\ 7,\ 1 .
\]
Por tanto, $\binom62 = 15$, $\binom73 = 35$ y $\binom74 = 35$ (la simetría
$\binom73 = \binom74$ refleja el intercambio de éxitos y fracasos).
\end{solution}

\begin{solution}{exo:g11:binom:2}
Número fijo de pruebas ($4$ lanzamientos), dos resultados por lanzamiento
(seis o no seis, con $p = \frac16$) y lanzamientos independientes:
$X \sim \mathcal B(4, \frac16)$.
\[
\P(X=0) = \left(\frac56\right)^4 = \frac{625}{1296} \approx 0.48,
\qquad
\P(X=1) = 4 \times \frac16\left(\frac56\right)^3 = \frac{500}{1296}
\approx 0.39,
\]
\[
\P(X \geq 2) = 1 - \frac{625 + 500}{1296} = \frac{171}{1296}
\approx 0.13 .
\]
\end{solution}

\begin{solution}{exo:g11:binom:3}
\emph{1.} Binomial $\mathcal B(20, \frac12)$: $n$ fijo, la misma $p$ y
lanzamientos independientes.

\emph{2.} No es binomial: las cartas se reparten \emph{sin devolución},
así que la \omterm{def:g10:proba:distribution}{probabilidad} de as cambia de carta en carta y las extracciones
no son independientes.

\emph{3.} Binomial $\mathcal B(7, 0.3)$ por la hipótesis de independencia
del enunciado.
\end{solution}

\begin{solution}{exo:g11:binom:4}
$\E(X) = 50 \times 0.2 = 10$; $\V(X) = 50 \times 0.2 \times 0.8 = 8$;
$\sigma(X) = 2\sqrt2 \approx 2.83$.
\end{solution}

\begin{solution}{exo:g11:binom:5}
$X \sim \mathcal B(6, 0.7)$.
\[
\P(X = 4) = \binom64 (0.7)^4 (0.3)^2 = 15 \times 0.2401 \times 0.09
\approx 0.324 .
\]
\[
\P(X \geq 5) = \binom65 (0.7)^5(0.3) + (0.7)^6
= 6 \times 0.16807 \times 0.3 + 0.117649 \approx 0.420 .
\]
\end{solution}

\begin{solution}{exo:g11:binom:6}
$X \sim \mathcal B\left(8, \frac12\right)$; todos los caminos tienen
\omterm{def:g10:proba:distribution}{probabilidad} $\frac{1}{256}$, luego
\[
\P(X \geq 6) = \frac{\binom86 + \binom87 + \binom88}{256}
= \frac{28 + 8 + 1}{256} = \frac{37}{256} \approx 0.14 .
\]
Respondiendo al azar se aprueba aproximadamente una vez de cada siete.
\end{solution}

\begin{solution}{exo:g11:binom:7}
El \omterm{def:g10:proba:operations}{suceso contrario} de ``al menos una de cada'' es ``las cinco iguales'':
todas A o todas B, cada caso con \omterm{def:g10:proba:distribution}{probabilidad}
$\left(\frac12\right)^5 = \frac1{32}$. Por tanto,
\[
\P(\text{una de cada clase}) = 1 - \frac{2}{32} = \frac{15}{16} .
\]
\end{solution}

\begin{solution}{exo:g11:binom:8}
$\P(X = 3) = p^3$ y $\P(X \geq 1) = 1 - (1-p)^3$. Resolviendo
$p^3 = \frac{27}{64} = \left(\frac34\right)^3$ se obtiene $p = \frac34$
(la \omterm{def:g11:func:function}{función} cubo es estrictamente \omterm{def:g11:func:monotone}{creciente}, \cref{ch:g11:func}, así que
la solución es única).
\end{solution}

\begin{solution}{exo:g11:binom:9}
$\P(\text{al menos una cara}) = 1 - \left(\frac12\right)^n$, así que la
condición es $\left(\frac12\right)^n < 0.01$, es decir, $2^n > 100$. Como
$2^6 = 64$ y $2^7 = 128$: a partir de $n = 7$ lanzamientos.
\end{solution}

\begin{solution}{exo:g11:binom:10}
La suma de la fila $n$ cuenta todos los caminos del \omterm{met:g10:proba:tree}{árbol} de $n$ pruebas,
clasificados por su número de éxitos. Pero el \omterm{met:g10:proba:tree}{árbol} duplica sus caminos en
cada prueba (cada camino se bifurca en E y F), así que tiene $2^n$ caminos
en total. Por tanto, $\sum_{k} \binom nk = 2^n$. Otra vía, por inducción:
la fila $0$ suma $1 = 2^0$, y la regla de Pascal hace que cada entrada de
la fila $n$ contribuya a exactamente dos entradas de la fila $n+1$, así
que las sumas de las filas se duplican.
\end{solution}

\begin{solution}{exo:g11:binom:11}
\emph{1.} Bajo esa afirmación, $X \sim \mathcal B(10, 0.6)$. Sumando los
primeros términos:
\begin{align*}
\P(X \leq 3)
&= (0.4)^{10} + 10(0.6)(0.4)^9 + 45(0.6)^2(0.4)^8
+ 120(0.6)^3(0.4)^7\\
&\approx 0.0001 + 0.0016 + 0.0106 + 0.0425 = 0.0548 .
\end{align*}

\emph{2.} Un resultado al menos tan extremo como el observado ($3$
aprobaciones o menos) tiene \omterm{def:g10:proba:distribution}{probabilidad} de en torno al $5.5\,\%$, justo
\emph{por encima} del umbral del $5\,\%$. Aplicando la regla al pie de la
letra, la observación es (por poco) compatible con la afirmación y no se
rechaza. El ejemplo muestra lo sensibles que son las decisiones al filo
del umbral: con un convenio del $6\,\%$ la conclusión se daría la vuelta.
\end{solution}

\begin{solution}{pb:g11:binom:1}
\textbf{1.} Filas: $1$; $1\,1$; $1\,2\,1$; $1\,3\,3\,1$;
$1\,4\,6\,4\,1$; $1\,5\,10\,10\,5\,1$; $1\,6\,15\,20\,15\,6\,1$.
Simetría: elegir qué $k$ objetos se toman es el mismo acto que elegir qué
$n - k$ se dejan.

\textbf{2.} $1 + 4 + 6 + 4 + 1 = 16 = 2^4$;
$1 + 5 + 10 + 10 + 5 + 1 = 32 = 2^5$. Demostración: los $\binom nk$
cuentan los subconjuntos de cada tamaño de un conjunto de $n$ elementos, y
los subconjuntos en total son $2^n$ (cada elemento dentro o fuera, de
forma independiente).

\textbf{3.} Comités de $k$ personas elegidas entre $n + 1$, una de las
cuales es Zoe: los que no la incluyen son $\binom nk$ (se eligen los $k$
entre los demás); los que sí la incluyen son $\binom{n}{k-1}$ (se eligen
sus $k - 1$ compañeros). Total: $\binom nk + \binom{n}{k-1}$.

\textbf{4.} Fila $7$ del triángulo: $1\,7\,21\,35\,\dots$: $35$. Con la
fórmula: $\frac{7 \times 6 \times 5}{3 \times 2 \times 1} = 35$.

\textbf{5.} $1 + 3 + 6 + 10 = 20 = \binom63$. Cascada:
$\binom63 = \binom52 + \binom53 = \binom52 + \binom42 + \binom43 =
\binom52 + \binom42 + \binom32 + \binom33$; cada aplicación de la regla de
Pascal pela un peldaño de la escalera.

\textbf{6.} Número fijo $n$ de rebotes; cada rebote es una \omterm{def:g11:binom:bernoulli}{prueba de
Bernoulli} independiente con la misma $p = \frac12$; y el número de casilla
cuenta los éxitos (los rebotes a la derecha): las tres casillas del
\cref{met:g11:binom:recognize} marcadas:
$\mathcal B\!\left(n, \frac12\right)$.

\textbf{7.} \omterm{def:g10:proba:distribution}{Probabilidades} $\frac{1}{16}, \frac{4}{16}, \frac{6}{16},
\frac{4}{16}, \frac{1}{16}$ para las casillas $0, \dots, 4$: la casilla
central, la $2$, es la más concurrida.

\textbf{8.} Recuentos esperados $= 1024 \times \binom{10}{k}/1024 =
\binom{10}{k}$: en la casilla central, $\binom{10}{5} = 252$ bolas; en la
casilla $7$, $\binom{10}{7} = 120$; y en cada casilla del borde, $1$ bola.
Un centro alto que cae simétricamente hasta unos bordes finísimos: la
campana.

\textbf{9.} $\E = np = 5$; $V = np(1 - p) = 2.5$; $\sigma \approx 1.58$.
Dentro de $2\sigma$: las casillas $2$ a $8$ recogen
\[
\frac{45 + 120 + 210 + 252 + 210 + 120 + 45}{1024}
= \frac{1002}{1024} \approx 98\,\%
\]
de las bolas, mucho mejor que el $75\,\%$ para todo uso de Chebyshev
(\cref{pb:g11:stat:1}): las campanas concentran con fuerza.

\textbf{10.} $\E = 6$, $V = 10 \times 0.6 \times 0.4 = 2.4$,
$\sigma \approx 1.55$: el montón conserva su forma de campana, pero
desliza su cima hasta la casilla $6$; un tablero inclinado es una moneda
trucada hecha visible.

\textbf{11.} El número de casilla es una \emph{suma} de muchos empujones
aleatorios pequeños, independientes y del mismo tamaño, y esas sumas se
organizan siempre en forma de campana: la mayoría de los empujones se
cancelan y los extremos exigen unanimidad. Las estaturas, los errores de
medida y un sinfín de magnitudes naturales son también sumas de muchos
efectos pequeños e independientes, y por eso aparece la misma silueta por
todas partes; el teorema que lo certifica es el teorema central del
límite.

\textbf{12.} Que acabe en cuatro partidos: uno de los dos equipos gana los
$4$: $2 \times \left(\frac12\right)^4 = \frac18$.

\textbf{13.} Llegar a siete partidos exige un $3$--$3$ tras seis:
$\binom63 \left(\frac12\right)^6 = \frac{20}{64} = \frac{5}{16}$.

\textbf{14.} Que acabe en $5$: el ganador se lleva el partido 5 y $3$ de
los $4$ primeros: $2 \times \binom43 \left(\frac12\right)^5 = \frac14$.
Que acabe en $6$: $2 \times \binom53 \left(\frac12\right)^6 =
\frac{5}{16}$. \omterm{def:g11:prob:rv}{Distribución} sobre $4, 5, 6, 7$:
$\frac18, \frac14, \frac{5}{16}, \frac{5}{16}$ (suma $1$). Duración
esperada: $4 \cdot \frac18 + 5 \cdot \frac14 + 6 \cdot \frac{5}{16} +
7 \cdot \frac{5}{16} = 5.8125$ partidos. Las eliminatorias de seis y siete
partidos son las más probables: el dramatismo viene de serie en el
formato.

\textbf{15.} Ganar en $4$: $0.6^4 = 0.1296$; en $5$:
$\binom43\,0.6^3 \times 0.4 \times 0.6 = 0.2074$; en $6$:
$\binom53\,0.6^3 \times 0.4^2 \times 0.6 = 0.2074$; en $7$:
$\binom63\,0.6^3 \times 0.4^3 \times 0.6 = 0.1659$. Total: unos $0.710$:
un equipo con un $60\,\%$ por partido gana el $71\,\%$ de las
eliminatorias; la eliminatoria amplifica la ventaja.

\textbf{16.} Final a partido único: $60\,\%$. Al mejor de $3$:
$p^2 + 2p^2 q = 0.36 + 0.288 = 0.648$. La escalera
$60\,\% \to 65\,\% \to 71\,\%$ sigue subiendo con la duración: más
partidos promedian la suerte (la ley de los grandes números en miniatura),
así que las finales largas coronan el talento, que es justo lo que venden
las ligas.

\textbf{17.} Moneda no trucada: $\E = 50$, $\sigma = \sqrt{25} = 5$;
$z = \frac{62 - 50}{5} = 2.4$: más allá del convenio de $2\sigma$; la
moneda merece una investigación.

\textbf{18.} $\P(X = 0) = 0.98^{50} \approx 0.364$;
$\P(X = 1) = 50 \times 0.02 \times 0.98^{49} \approx 0.372$;
$\P(X = 2) = \binom{50}{2} 0.02^2 \times 0.98^{48} \approx 0.186$. Así
que $\P(X \geq 3) \approx 1 - 0.922 = 0.078$: alrededor de un
$7.8\,\%$, por encima del umbral del $5\,\%$, así que todavía no basta
para rechazar la afirmación; un segundo lote malo cambiaría la historia.

\textbf{19.} $\P(\text{al menos un premio}) = 1 - 0.999^{1000} \approx
1 - 0.368 = 0.632$: mil boletos a uno entre mil no dan certeza, sino un
$63\,\%$. El $0.368$ que reaparece es $\frac1e$ disfrazado; la constante
$e$ hace su entrada oficial el año que viene.

\textbf{20.} Reconocimiento: $n$ fijo, independencia y $p$ constante;
entonces, y solo entonces, binomial. Tabla: el \omterm{prop:g11:binom:pascal}{triángulo de Pascal}, fila
$n$. Forma: la campana, simétrica si $p = \frac12$ y desplazada en los
demás casos. Brújula: centro $np$ y dispersión $\sqrt{np(1-p)}$, las
puntuaciones z de las decisiones. Herederas: la campana continua a la que
se acercan los montones y la ley de los grandes números, que explica por
qué los tableros grandes nunca mienten.
\end{solution}
