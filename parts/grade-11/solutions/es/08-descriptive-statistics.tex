\section*{Capítulo \ref{ch:g11:stat} --- Estadística descriptiva}

\begin{solution}{exo:g11:stat:1}
$n = 8$ y suma $71$: \omterm{def:g11:stat:mean}{media} $\bar x = \frac{71}{8} = 8.875$. \omterm{def:g11:stat:median}{Mediana}: \omterm{prop:g10:coordgeom:midpoint}{punto
medio} del $4$.\textsuperscript{o} y el $5$.\textsuperscript{o} valores,
$\frac{8 + 9}{2} = 8.5$. \omterm{def:g11:stat:quartiles}{Cuartiles}: la cuarta parte de $8$ es $2$, así que
$Q_1$ es el $2$.\textsuperscript{o} valor, $7$; tres cuartas partes son
$6$, así que $Q_3$ es el $6$.\textsuperscript{o} valor, $10$. \omterm{def:g11:stat:quartiles}{Recorrido
intercuartílico}: $10 - 7 = 3$.
\end{solution}

\begin{solution}{exo:g11:stat:2}
Suma $40$, luego $\bar x = 5$. Suma de cuadrados:
$4 + 16 + 16 + 16 + 25 + 25 + 49 + 81 = 232$. Por la fórmula abreviada,
\[
V = \frac{232}{8} - 5^2 = 29 - 25 = 4,
\qquad
\sigma = 2 .
\]
\end{solution}

\begin{solution}{exo:g11:stat:3}
\omterm{def:g11:stat:mean}{Media}:
\[
\bar x = \frac{1 \times 6 + 2 \times 9 + 3 \times 8 + 4 \times 10
+ 5 \times 9 + 6 \times 8}{50}
= \frac{181}{50} = 3.62 .
\]
\omterm{def:g11:stat:median}{Mediana}: las frecuencias acumuladas son $6, 15, 23, 33, \dots$, así que
los resultados $25$ y $26$ de la lista ordenada son los dos $4$: la
\omterm{def:g11:stat:median}{mediana} es $4$.
\end{solution}

\begin{solution}{exo:g11:stat:4}
El total de las $15$ notas es $15 \times 11.2 = 168$; con el recién
llegado, $\frac{168 + 18}{16} = \frac{186}{16} = 11.625$.
\end{solution}

\begin{solution}{exo:g11:stat:5}
$\frac{6 + 8 + x + 12 + 14}{5} = 10$ da $40 + x = 50$, luego $x = 10$. Las
desviaciones de $6, 8, 10, 12, 14$ respecto de la \omterm{def:g11:stat:mean}{media} $10$ son
$-4, -2, 0, 2, 4$, con cuadrados que suman $40$: $V = \frac{40}{5} = 8$ y
$\sigma = 2\sqrt2 \approx 2.83$.
\end{solution}

\begin{solution}{exo:g11:stat:6}
Por la \cref{prop:g11:stat:affine} con $a = \frac12$ y $b = 4$: nueva
\omterm{def:g11:stat:mean}{media} $\frac{12.4}{2} + 4 = 10.2$; nueva \omterm{def:g11:stat:variance}{desviación típica}
$\frac{2.5}{2} = 1.25$.
\end{solution}

\begin{solution}{exo:g11:stat:7}
La \omterm{def:g11:stat:mean}{media} conjunta es la \omterm{def:g11:stat:mean}{media} \emph{ponderada}
\[
\frac{20 \times 10 + 30 \times 15}{50} = \frac{650}{50} = 13,
\]
y no $12.5$: el grupo B aporta más valores, así que arrastra la \omterm{def:g11:stat:mean}{media}
conjunta hacia la suya.
\end{solution}

\begin{solution}{exo:g11:stat:8}
Las dos líneas están casi en el objetivo de \omterm{def:g11:stat:mean}{media} ($1.002$ y $1.001$ kg),
pero la \omterm{def:g11:stat:variance}{desviación típica} de la línea 2 es el triple: sus bolsas varían
mucho más y produce muchas más bolsas visiblemente escasas o pasadas de
peso. Hay que ocuparse primero de la línea 2: lo que hay que arreglar es
la regularidad, no solo la \omterm{def:g11:stat:mean}{media}.
\end{solution}

\begin{solution}{exo:g11:stat:9}
$\bar x = \frac{70}{10} = 7$; $V = \frac{540}{10} - 7^2 = 54 - 49 = 5$;
$\sigma = \sqrt5 \approx 2.24$.
\end{solution}

\begin{solution}{exo:g11:stat:10}
\emph{1.} Con el $23$: suma $50$, \omterm{def:g11:stat:mean}{media} $5$; \omterm{def:g11:stat:median}{mediana} $=$ \omterm{prop:g10:coordgeom:midpoint}{punto medio} del
$5$.\textsuperscript{o} y el $6$.\textsuperscript{o} valores $= 3$. Sin el
$23$: suma $27$, \omterm{def:g11:stat:mean}{media} $3$; \omterm{def:g11:stat:median}{mediana} (el $5$.\textsuperscript{o} de $9$
valores) $= 3$.

\emph{2.} Con el $23$ ($n = 10$): $Q_1$ es el $3$.\textsuperscript{er}
valor ($2$) y $Q_3$, el $8$.\textsuperscript{o} ($4$): \omterm{def:g11:stat:quartiles}{recorrido
intercuartílico} $2$. Sin él ($n = 9$): $Q_1$ es el
$3$.\textsuperscript{er} valor ($2$) y $Q_3$, el $7$.\textsuperscript{o}
($4$): sigue siendo $2$.

\emph{3.} Un único valor atípico arrastra la \omterm{def:g11:stat:mean}{media} de $3$ a $5$, pero deja
intactos la \omterm{def:g11:stat:median}{mediana} y el \omterm{def:g11:stat:quartiles}{recorrido intercuartílico}: esos dos son
\emph{robustos}. Cuando un conjunto de datos puede contener valores
aberrantes, informa de la \omterm{def:g11:stat:median}{mediana} y del \omterm{def:g11:stat:quartiles}{recorrido intercuartílico} antes
que de la \omterm{def:g11:stat:mean}{media} y la \omterm{def:g11:stat:variance}{desviación típica}.
\end{solution}

\begin{solution}{pb:g11:stat:1}
\textbf{1.} \omterm{def:g11:stat:mean}{Media} $\bar x = \frac{25}{5} = 5$. Desviaciones al cuadrado:
$1, 9, 1, 0, 9$: \omterm{def:g11:stat:variance}{varianza} $\frac{20}{5} = 4$, luego $\sigma = 2$.

\textbf{2.} \omterm{def:g11:stat:mean}{Media} de los cuadrados:
$\frac{16 + 64 + 36 + 25 + 4}{5} = 29$; menos $\bar x^2 = 25$: \omterm{def:g11:stat:variance}{varianza}
$4$. La misma respuesta con la mitad de trabajo.

\textbf{3.} \omterm{def:g11:stat:mean}{Media} $= \frac{3 \times 8 + 5 \times 10 + 2 \times
14}{10} = 10.2$; \omterm{def:g11:stat:variance}{varianza} $= \frac{3 \times 64 + 5 \times 100 +
2 \times 196}{10} - 10.2^2 = 108.4 - 104.04 = 4.36$;
$\sigma \approx 2.09$.

\textbf{4.} \omterm{def:g11:stat:mean}{Media}: $1.8 \times 20 + 32 = 68\,^\circ$F; $\sigma$:
$1.8 \times 3 = 5.4\,^\circ$F: el desplazamiento $+32$ no dispersa nada.

\textbf{5.} Al sumar $b$: cada desviación
$x_i + b - (\bar x + b) = x_i - \bar x$ no cambia, luego la \omterm{def:g11:stat:variance}{varianza}
tampoco. Al multiplicar por $a$: las desviaciones quedan multiplicadas por
$a$, sus cuadrados por $a^2$, la \omterm{def:g11:stat:variance}{varianza} por $a^2$ y $\sigma$ por
$\sqrt{a^2} = \abs a$.

\textbf{6.} $z_A = \frac{15 - 12}{2} = 1.5$;
$z_B = \frac{13 - 8}{4} = 1.25$. Alicia destaca más sobre su grupo: el
$15$ del examen difícil gana al $13$ del fácil.

\textbf{7.} Puntuaciones z: $-0.5,\ 1.5,\ 0.5,\ 0,\ -1.5$: \omterm{def:g11:stat:mean}{media} $0$ y
\omterm{def:g11:stat:variance}{desviación típica} $1$. En general, estandarizar resta $\bar x$ (la \omterm{def:g11:stat:mean}{media}
pasa a $0$ y $\sigma$ no se toca) y divide después entre $\sigma$ (la
nueva $\sigma$ pasa a valer $1$): la pregunta 5 aplicada dos veces.

\textbf{8.} $z = 2$: $100 + 2 \times 15 = 130$. Y
$z(76) = \frac{76 - 100}{15} = -1.6$.

\textbf{9.} El bebé está $2.5$ desviaciones típicas por debajo del peso
medio \emph{de los bebés de su edad}: alarmantemente ligero para su
cohorte. Los kilogramos en bruto no se pueden comparar entre edades (todos
los bebés ganan peso); las puntuaciones z miden a cada niño frente a su
propio grupo de edad, con una sola regla para todas las edades.

\textbf{10.} \omterm{def:g11:stat:mean}{Media} $5$; \omterm{def:g11:stat:mean}{media} de los cuadrados
$\frac{622}{10} = 62.2$, \omterm{def:g11:stat:variance}{varianza} $37.2$ y $\sigma \approx 6.10$;
$z(23) = \frac{18}{6.10} \approx 2.95$: justo por debajo del umbral de
alarma; el test apenas parpadea.

\textbf{11.} Sin el $23$: \omterm{def:g11:stat:mean}{media} $3$, \omterm{def:g11:stat:variance}{varianza}
$\frac{93}{9} - 9 \approx 1.33$ y $\sigma \approx 1.15$; frente a esa
regla limpia, el $23$ puntúa $z \approx \frac{20}{1.15} \approx 17$: un
monstruo. El valor atípico infló a la vez la \omterm{def:g11:stat:mean}{media} y $\sigma$ y se
camufló: por eso el \cref{exo:g11:stat:10} recomienda resúmenes robustos
(\omterm{def:g11:stat:median}{mediana}, \omterm{def:g11:stat:quartiles}{recorrido intercuartílico}) cuando amenaza la contaminación.

\textbf{12.} En la suma de la \omterm{def:g11:stat:variance}{varianza}
$\sigma^2 = \frac1N \sum (x_i - \bar x)^2$, conservamos solo los términos
con $\abs{x_i - \bar x} \geq k\sigma$: hay $pN$ de ellos y cada uno vale
al menos $(k\sigma)^2$, luego
$\sigma^2 \geq \frac{1}{N} \cdot pN \cdot k^2\sigma^2 =
p\,k^2\sigma^2$; dividiendo entre $k^2 \sigma^2$ (con dispersión no
nula), $p \leq \frac{1}{k^2}$.

\textbf{13.} Más allá de $2\sigma$: como mucho $\frac14 = 25\,\%$ de un
conjunto de datos cualquiera; más allá de $3\sigma$: como mucho
$\frac19 \approx 11\,\%$. Con los datos de la pregunta 1: más allá de
$2\sigma = 4$ de la \omterm{def:g11:stat:mean}{media} $5$ significa $\leq 1$ o $\geq 9$: ninguno de
los cinco valores, o sea, $0\,\% \leq 25\,\%$, como se prometía.

\textbf{14.} Estar fuera de $\intcc{49.7}{50.3}$ significa estar más allá
de $3\sigma$: como mucho $\frac19$ de la producción, \emph{sea cual sea}
la forma de la distribución; Chebyshev no pide nada más que una \omterm{def:g11:stat:mean}{media} y
una $\sigma$. Una producción en forma de campana se concentra mucho mejor
(alrededor de un $0.3\,\%$ más allá de $3\sigma$): la ley normal, el año
que viene.

\textbf{15.} El $-19\,\%$ está $24$ puntos por debajo de la \omterm{def:g11:stat:mean}{media} $5$.
Fondo B: $24 = 2\sigma_B$, así que Chebyshev permite hasta $\frac14$ de
años tan malos. Fondo A: $24 = 12\sigma_A$: como mucho $\frac{1}{144}$. La
misma \omterm{def:g11:stat:mean}{media}, peligros incomparables: $\sigma$ es lo que en finanzas se
llama \emph{riesgo} (o volatilidad).

\textbf{16.} Desarrollamos alrededor de $\bar x$:
$(x_i - c)^2 = \left((x_i - \bar x) + (\bar x - c)\right)^2
= (x_i - \bar x)^2 + 2(\bar x - c)(x_i - \bar x) + (\bar x - c)^2$. Al
promediar, el término central muere (las desviaciones respecto de la \omterm{def:g11:stat:mean}{media}
suman cero) y queda $f(c) = \sigma^2 + (\bar x - c)^2$, \omterm{def:g10:functions:extrema}{mínimo} exactamente
en $c = \bar x$, con \omterm{def:g10:functions:extrema}{mínimo} $\sigma^2$. La \omterm{def:g11:stat:mean}{media} es el centro de \omterm{def:g10:functions:extrema}{mínimos}
cuadrados, igual que la \omterm{def:g11:stat:median}{mediana} es el centro de \omterm{def:g10:functions:extrema}{mínimos} \omterm{def:g10:numbers:abs}{valores absolutos}
(\cref{pb:g10:stats:1}).

\textbf{17.} Directamente: las desviaciones respecto de $6$ son
$-2, 2, 0, -1, -4$, con cuadrados $4, 4, 0, 1, 16$:
$f(6) = \frac{25}{5} = 5$. Con la fórmula:
$\sigma^2 + (5 - 6)^2 = 4 + 1 = 5$. Coinciden.

\textbf{18.} Los dos tienen \omterm{def:g11:stat:mean}{media} $7$; \omterm{def:g11:stat:variance}{varianzas}: $P$:
$\frac{16 + 4 + 0 + 4 + 16}{5} = 8$; $Q$:
$\frac{36 + 0 + 0 + 0 + 36}{5} = 14.4$: dispersiones distintas tras una
misma \omterm{def:g11:stat:mean}{media}. Para $\{7 - a, 7, 7, 7, 7 + a\}$: la \omterm{def:g11:stat:variance}{varianza}
$\frac{2a^2}{5} = 8$ da $a = \sqrt{20} \approx 4.47$: el conjunto
$\{2.53, 7, 7, 7, 11.47\}$ tiene exactamente la \omterm{def:g11:stat:mean}{media} y la \omterm{def:g11:stat:variance}{varianza} de
$P$ con un perfil completamente distinto; los resúmenes comprimen, y
comprimir pierde.

\textbf{19.} $\sigma^2 = 0$ significa que se anula una suma de cuadrados
$(x_i - \bar x)^2$; los cuadrados son $\geq 0$, así que una suma nula
obliga a que todos los términos sean nulos: $x_i = \bar x$ para todo $i$,
o sea, todos los valores iguales.

\textbf{20.} Los centros dicen dónde vive el conjunto de datos; las
dispersiones, con qué holgura; la \omterm{pb:g11:stat:1}{puntuación z} tiende una misma regla
sobre todos los conjuntos de datos, exámenes, edades y monedas; Chebyshev
convierte $\sigma$ en una garantía incondicional; y la pregunta 18 nos
mantiene honrados: ningún puñado de números contiene nunca la historia
entera. El año que viene la campana afinará la regla hasta convertirla en
\omterm{def:g10:proba:distribution}{probabilidades} exactas.
\end{solution}
