\section*{Capítulo \ref{ch:g11:stat} --- Estadística descriptiva}

\begin{solution}{exo:g11:stat:1}
$n = 8$, suma $71$: \omterm{def:g11:stat:mean}{media} $\bar x = \frac{71}{8} = 8.875$. \omterm{def:g11:stat:median}{mediana}: \omterm{prop:g10:coordgeom:midpoint}{punto medio}
del $4$ th y $5$º valores, $\frac{8 + 9}{2} = 8.5$. cuartiles: un
cuarto de $8$ es $2$, entonces $ Q_1$ es el $2$ segundo valor, $7$; tres cuartos es
$6$, entonces $ Q_3$ es el $6$ valor, $10$. \omterm{def:g10:stats:quartiles}{rango} intercuartílico:
$10 - 7 = 3$.
\end{solution}

\begin{solution}{exo:g11:stat:2}
Suma $40$, entonces $\bar x = 5$. Suma de cuadrados:
$4 + 16 + 16 + 16 + 25 + 25 + 49 + 81 = 232$. Por la fórmula abreviada,
\[
V = \frac{232}{8} - 5^2 = 29 - 25 = 4,
\qquad
\sigma = 2 .
\]
\end{solution}

\begin{solution}{exo:g11:stat:3}
\omterm{def:g11:stat:mean}{media}:
\[
\bar x = \frac{1 \times 6 + 2 \times 9 + 3 \times 8 + 4 \times 10
+ 5 \times 9 + 6 \times 8}{50}
= \frac{181}{50} = 3.62 .
\]
\omterm{def:g11:stat:median}{mediana}: las \omterm{def:g10:stats:series}{frecuencias} acumuladas son $6, 15, 23, 33, \dots $, entonces el
$25$ th y $26$ Los resultados ordenados son ambos. $4$: la \omterm{def:g11:stat:median}{mediana} es $4$.
\end{solution}

\begin{solution}{exo:g11:stat:4}

\end{solution}

\begin{solution}{exo:g11:stat:5}
$\frac{6 + 8 + x + 12 + 14}{5} = 10$ da $40 + x = 50$, entonces $ x = 10$.
Las desviaciones de $6, 8, 10, 12, 14$ desde \omterm{def:g11:stat:mean}{media} $10$ son
$-4, -2, 0, 2, 4$, con cuadrados que suman $40$:
$ V = \frac{40}{5} = 8$ y $\sigma = 2\sqrt2 \approx 2.83$.
\end{solution}

\begin{solution}{exo:g11:stat:7}

\]

\begin{solution}{exo:g11:stat:8}
Ambas líneas están cerca del objetivo en promedio ($1.002$ y $1.001$ kilogramos), pero

\begin{solution}{exo:g11:stat:9}

$ V = \frac{540}{10} - 7^2 = 54 - 49 = 5$;

\end{solution}

\begin{solution}{exo:g11:stat:10}
\emph{1.} Con $23$: suma $50$, \omterm{def:g11:stat:mean}{media} $5$; \omterm{def:g11:stat:median}{mediana} = \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $5$ th y
$6$º valores $= 3$. Sin $23$: suma $27$, \omterm{def:g11:stat:mean}{media} $3$; \omterm{def:g11:stat:median}{mediana} (5 de 9
valores) $= 3$.

\emph{2.} Con $23$ ($ n = 10$): $ Q_1$ es el $3$ tercer valor ($2$), $ Q_3$ el
$8$ th ($4$): \omterm{def:g10:stats:quartiles}{rango} intercuartílico $2$. Sin ($ n = 9$): $ Q_1$ es el
$3$ tercer valor ($2$), $ Q_3$ el $7$ th ($4$): aún $2$.

\emph{3.} El único valor atípico arrastra al \omterm{def:g11:stat:mean}{media} de $3$ arriba a $5$ pero se va
la \omterm{def:g11:stat:median}{mediana} y el \omterm{def:g10:stats:quartiles}{rango} intercuartílico sin cambios: esos dos son
\emph{robusto}. Cuando un conjunto de datos pueda contener valores aberrantes, informar el
\omterm{def:g11:stat:median}{mediana} y \omterm{def:g10:stats:quartiles}{rango} intercuartílico en lugar de la \omterm{def:g11:stat:mean}{media} y desviación estándar.
\end{solution}

\begin{solution}{pb:g11:stat:1}
\textbf{1.} \omterm{def:g11:stat:mean}{media} $\bar x = \frac{25}{5} = 5$. Desviaciones
al cuadrado: $1, 9, 1, 0, 9$: varianza $\frac{20}{5} = 4$, entonces
$\sigma = 2$.

\textbf{2.} \omterm{def:g11:stat:mean}{media} de cuadrados:
$\frac{16 + 64 + 36 + 25 + 4}{5} = 29$; menos
$\bar x^2 = 25$: varianza $4$. Misma respuesta, la mitad del trabajo.

\textbf{3.} \omterm{def:g11:stat:mean}{media} $= \frac{3 \times 8 + 5 \times 10 + 2 \times
14}{10} = 10.2$; varianza $= \frac{3 \times 64 + 5 \times 100 +
2 \times 196}{10} - 10.2^2 = 108.4 - 104.04 = 4.36$;
$\sigma \approx 2.09$.

\textbf{4.} \omterm{def:g11:stat:mean}{media}: $1.8 \times 20 + 32 = 68\,^\circ $ F;
$\sigma $: $1.8 \times 3 = 5.4\,^\circ $ F --- el cambio $+32$
no difunde nada.

\textbf{5.} Añadiendo $ b $: cada desviación $ x_i + b - (\bar x + b)
= x_i - \bar x $ no cambia, por lo que la varianza es. multiplicando
por $ a $: las desviaciones se multiplican por $ a $, sus cuadrados por
$ a^2$, la varianza por $ a^2$, y $\sigma $ por
$\sqrt{a^2} = \abs a $.

\textbf{6.} $ z_A = \frac{15 - 12}{2} = 1.5$;
$ z_B = \frac{13 - 8}{4} = 1.25$. Alice está más arriba de ella.
multitud: los exámenes más difíciles $15$ supera los exámenes más fáciles $13$.

\textbf{7.} $ z $-montones: $-0.5,\ 1.5,\ 0.5,\ 0,\ -1.5$: \omterm{def:g11:stat:mean}{media}
$0$, desviación estándar $1$. En general, la estandarización
resta $\bar x $ (haciendo la \omterm{def:g11:stat:mean}{media} $0$, $\sigma $ intacto)
luego se divide por $\sigma $ (haciendo el nuevo $\sigma $ igual a
$1$): pregunta 5 dos veces.

\textbf{8.} $ z = 2$: $100 + 2 \times 15 = 130$. Y
$ z(76) = \frac{76 - 100}{15} = -1.6$.

\textbf{9.} el bebe es $2.5$ desviaciones estándar debajo del
\omterm{def:g11:stat:mean}{media} peso \emph{de bebés de su edad} --- alarmantemente ligero para
su cohorte. Los kilogramos brutos no se pueden comparar entre edades (todas
los bebés aumentan de peso); Defectuoso z medir a cada niño contra su
propio grupo de edad, un gobernante para todas las edades.

\textbf{10.} \omterm{def:g11:stat:mean}{media} $5$; \omterm{def:g11:stat:mean}{media} de cuadrados $\frac{622}{10} = 62.2$,
varianza $37.2$, $\sigma \approx 6.10$;
$ z(23) = \frac{18}{6.10} \approx 2.95$: justo debajo de la alarma
umbral --- la prueba apenas parpadea.

\textbf{11.} Sin $23$: \omterm{def:g11:stat:mean}{media} $3$, varianza
$\frac{93}{9} - 9 \approx 1.33$, $\sigma \approx 1.15$;
contra el gobernante limpio, $23$ montones
$ z \approx \frac{20}{1.15} \approx 17$: un monstruo. El valor atípico
infló tanto la \omterm{def:g11:stat:mean}{media} y $\sigma $, camuflándose ---
es por eso que \cref{exo:g11:stat:10} recomienda robusto
resúmenes (\omterm{def:g11:stat:median}{mediana}, IQR) cuando existe amenaza de contaminación.

\textbf{12.} En la varianza suma
$\sigma^2 = \frac1N \sum (x_i - \bar x)^2$, mantenga sólo el
términos con $\abs{x_i - \bar x} \geq k\sigma $: hay $ pN $
de ellos, cada uno al menos $(k\sigma)^2$, entonces
$\sigma^2 \geq \frac{1}{N} \cdot pN \cdot k^2\sigma^2 =
p\,k^2\sigma^2$: dividir por $ k^2 \sigma^2$ (spread distinto de cero):
$ p \leq \frac{1}{k^2}$.

\textbf{13.} Más allá de $2\sigma $: como mucho $\frac14 = 25\,\%$ de
cualquier conjunto de datos; más allá de $3\sigma $: como mucho
$\frac19 \approx 11\,\%$. Datos de la pregunta 1: más allá
$2\sigma = 4$ desde \omterm{def:g11:stat:mean}{media} $5$ medio $\leq 1$ o $\geq 9$:
ninguno de los cinco valores --- $0\,\% \leq 25\,\%$, como prometí.

\textbf{14.} Afuera $\intcc{49.7}{50.3}$ significa más allá
$3\sigma $: como mucho $\frac19$ de producción, \emph{lo que}
la forma de la distribución --- Chebyshev no pregunta más que una
\omterm{def:g11:stat:mean}{media} y un $\sigma $. La producción en forma de campana se concentra
mejor (sobre $0.3\,\%$ más allá de $3\sigma $): la ley normal,
grado 12.

\textbf{15.} $-19\,\%$ se sienta $24$ puntos debajo de la \omterm{def:g11:stat:mean}{media} $5$.
Fondo B: $24 = 2\sigma_B $: Chebyshev permite hasta
$\frac14$ de años tan malos. Fondo A: $24 = 12\sigma_A $: en
la mayoría $\frac{1}{144}$. El mismo promedio, peligros incomparables ---
$\sigma $ es lo que llaman las finanzas \emph{riesgo} (o volatilidad).

\textbf{16.} Expandir alrededor $\bar x $:
$(x_i - c)^2 = \left((x_i - \bar x) + (\bar x - c)\right)^2
= (x_i - \bar x)^2 + 2(\bar x - c)(x_i - \bar x) +
(\bar x - c)^2$. Promediando: el término medio muere (desviaciones
desde la \omterm{def:g11:stat:mean}{media} suma cero), dejando
$ f(c) = \sigma^2 + (\bar x - c)^2$ --- \omterm{def:g10:functions:extrema}{mínimo} exactamente en
$ c = \bar x $, con \omterm{def:g10:functions:extrema}{mínimo} $\sigma^2$. La \omterm{def:g11:stat:mean}{media} es el
centro de \omterm{def:g10:functions:extrema}{mínimos} cuadrados, como la \omterm{def:g11:stat:median}{mediana} es el menos absoluto
centro (\cref{pb:g10:stats:1}).

\textbf{17.} Directamente: desviaciones de $6$ son
$-2, 2, 0, -1, -4$, cuadrados $4, 4, 0, 1, 16$:
$ f(6) = \frac{25}{5} = 5$. Fórmula:
$\sigma^2 + (5 - 6)^2 = 4 + 1 = 5$. Acuerdo.

\textbf{18.} ambos tienen \omterm{def:g11:stat:mean}{media} $7$; variaciones:
$ P $: $\frac{16 + 4 + 0 + 4 + 16}{5} = 8$;
$ Q $: $\frac{36 + 0 + 0 + 0 + 36}{5} = 14.4$: diferentes diferenciales
detrás de uno \omterm{def:g11:stat:mean}{media}. Para $\{7 - a, 7, 7, 7, 7 + a\}$: varianza
$\frac{2a^2}{5} = 8$ da $ a = \sqrt{20} \approx 4.47$: el
conjunto de datos $\{2.53, 7, 7, 7, 11.47\}$ lleva el exacto \omterm{def:g11:stat:mean}{media} y
varianza de $ P $ con un perfil completamente diferente ---
los resúmenes se comprimen y la compresión se pierde.

\textbf{19.} $\sigma^2 = 0$ significa una suma de cuadrados
$(x_i - \bar x)^2$ desaparece; los cuadrados son $\geq 0$, entonces un cero
La suma obliga a que cada término sea cero: $ x_i = \bar x $ para todo $ i $
--- todos los valores iguales.

\textbf{20.} Los centros dicen dónde residen los datos; los diferenciales dicen cómo
libremente; el \omterm{pb:g11:stat:1}{puntuación z} establece una regla en todos los conjuntos de datos,
exámenes, edades y monedas; Chebyshev se convierte $\sigma$ en
una garantía incondicional; y la pregunta 18 nos mantiene honestos
--- ningún puñado de números contiene la historia completa. Siguiente
año en que la curva de campana afila la regla hasta obtener
\omterm{def:g10:proba:distribution}{probabilidades}.
\end{solution}
