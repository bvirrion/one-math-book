\section*{Capítulo \ref{ch:g11:prob} --- Probabilidad y variables aleatorias}

\begin{solution}{exo:g11:prob:1}
\omterm{def:g11:prob:model}{Modelo uniforme} sobre $52$ cartas. Corazones: $\frac{13}{52} = \frac14$.
Reyes: $\frac{4}{52} = \frac{1}{13}$. Corazones o rey:
$\frac{13 + 4 - 1}{52} = \frac{16}{52} = \frac{4}{13}$ (el rey de
corazones se cuenta una sola vez). Ninguna de las dos cosas:
$1 - \frac{4}{13} = \frac{9}{13}$.
\end{solution}

\begin{solution}{exo:g11:prob:2}
$S = 6$: las parejas $(1,5), (2,4), (3,3), (4,2), (5,1)$, luego
$\P(S = 6) = \frac{5}{36}$.

$S \geq 10$: tres parejas para $10$, dos para $11$ y una para $12$:
$\P(S \geq 10) = \frac{6}{36} = \frac16$.

$S$ par: las sumas $2, 4, 6, 8, 10, 12$ tienen
$1 + 3 + 5 + 5 + 3 + 1 = 18$ parejas: \omterm{def:g10:proba:distribution}{probabilidad}
$\frac{18}{36} = \frac12$.
\end{solution}

\begin{solution}{exo:g11:prob:3}
Las \omterm{def:g10:proba:distribution}{probabilidades} suman $1$: $p = 1 - 0.3 - 0.2 - 0.4 = 0.1$. Entonces
\[
\E(X) = 0.3 \times (-1) + 0.2 \times 0 + 0.4 \times 2 + 0.1 \times 5
= -0.3 + 0.8 + 0.5 = 1 .
\]
\end{solution}

\begin{solution}{exo:g11:prob:4}
$\E(Y) = 2 \times 3 - 5 = 1$; $\V(Y) = 2^2 \times 4 = 16$;
$\sigma(Y) = 4$.
\end{solution}

\begin{solution}{exo:g11:prob:5}
Lo ganado es $0$, $2$ o $5$ euros con \omterm{def:g10:proba:distribution}{probabilidades} $0.5$, $0.3$ y $0.2$;
restando la apuesta de $2$ euros, $G$ toma los valores $-2, 0, 3$:
\begin{center}
\begin{tabular}{c|ccc}
$g$ & $-2$ & $0$ & $3$\\
\hline
$\P(G=g)$ & $0.5$ & $0.3$ & $0.2$
\end{tabular}
\end{center}
$\E(G) = -1 + 0 + 0.6 = -0.4$: el jugador pierde $40$ céntimos por partida
de \omterm{def:g11:stat:mean}{media}; desfavorable.
\end{solution}

\begin{solution}{exo:g11:prob:6}
Extrayendo en orden: $\P(X = 2) = \frac35 \times \frac24 = \frac{3}{10}$
(roja y roja); $\P(X = 0) = \frac25 \times \frac14 = \frac{1}{10}$ (azul y
azul); por tanto,
$\P(X = 1) = 1 - \frac{3}{10} - \frac{1}{10} = \frac{6}{10}$. Entonces
\[
\E(X) = 0 \times \frac1{10} + 1 \times \frac6{10} + 2 \times \frac3{10}
= \frac{12}{10} = 1.2 ,
\]
coherente con la intuición: de \omterm{def:g11:stat:mean}{media}, $\frac35$ de las dos extracciones
son rojas.
\end{solution}

\begin{solution}{exo:g11:prob:7}
$\E(G^2) = 0.5 \times 4 + 0.3 \times 0 + 0.2 \times 9 = 3.8$, así que por
la fórmula abreviada $\V(G) = 3.8 - (-0.4)^2 = 3.8 - 0.16 = 3.64$ y
$\sigma(G) \approx 1.9$ euros.
\end{solution}

\begin{solution}{exo:g11:prob:8}
El beneficio son los $80$ menos la reclamación:
\begin{center}
\begin{tabular}{c|ccc}
$b$ & $80$ & $-920$ & $-4920$\\
\hline
$\P(B=b)$ & $0.94$ & $0.05$ & $0.01$
\end{tabular}
\end{center}
$\E(B) = 75.2 - 46 - 49.2 = -20$: a ese precio, la compañía \emph{pierde}
$20$ euros por póliza de \omterm{def:g11:stat:mean}{media}. La prima es demasiado baja para los
riesgos cubiertos (tendría que pasar de $100$ euros para que el beneficio
esperado fuera positivo).
\end{solution}

\begin{solution}{exo:g11:prob:9}
La puntuación aleatoria es $+1$ con \omterm{def:g10:proba:distribution}{probabilidad} $\frac14$ y $x$ con
\omterm{def:g10:proba:distribution}{probabilidad} $\frac34$:
\[
\E = \frac14 + \frac{3x}{4} = 0 \iff x = -\frac13 .
\]
Con la penalización $-\frac13$, responder al azar no aporta nada de
\omterm{def:g11:stat:mean}{media}.
\end{solution}

\begin{solution}{exo:g11:prob:10}
Dos monedas: coinciden (CC o XX) con \omterm{def:g10:proba:distribution}{probabilidad}
$\frac24 \times 2 = \frac12$. La ganancia del jugador es $+1$ o $-1$ con
la misma \omterm{def:g10:proba:distribution}{probabilidad}: \omterm{def:g11:prob:expectation}{esperanza} $0$ para los dos jugadores; el juego es
equitativo.

Tres monedas: las tres coinciden (CCC o XXX) con \omterm{def:g10:proba:distribution}{probabilidad}
$\frac{2}{8} = \frac14$. El jugador recibe $2$ con \omterm{def:g10:proba:distribution}{probabilidad} $\frac14$
y paga $1$ con \omterm{def:g10:proba:distribution}{probabilidad} $\frac34$:
$\E = \frac24 - \frac34 = -\frac14$. El juego favorece a Ana en $25$
céntimos por ronda; sería equitativo si pagase $3$ euros por un triple.
\end{solution}

\begin{solution}{exo:g11:prob:11}
\emph{1.} El organizador recibe $1000 \times 5 = 5000$ euros y reparte
$2000 + 5 \times 200 + 20 \times 25 = 3500$ euros en total: el beneficio
es la \emph{constante} $1500$ (no hay azar una vez vendidas todas las
papeletas). La ganancia $G$ de un jugador (premio menos los $5$ euros de
la papeleta):
\begin{center}
\begin{tabular}{c|cccc}
$g$ & $1995$ & $195$ & $20$ & $-5$\\
\hline\rule{0pt}{11pt}%
$\P(G=g)$ & $\dfrac{1}{1000}$ & $\dfrac{5}{1000}$ & $\dfrac{20}{1000}$
& $\dfrac{974}{1000}$
\end{tabular}
\end{center}

\emph{2.}
\[
\E(G) = \frac{1995 + 5 \times 195 + 20 \times 20 - 974 \times 5}{1000}
= \frac{1995 + 975 + 400 - 4870}{1000} = -1.5 .
\]
Cada jugador espera perder $1.50$ euros; con $1000$ jugadores son $1500$
euros, exactamente el beneficio del organizador. Las cuentas cuadran: lo
que los jugadores pierden de \omterm{def:g11:stat:mean}{media} es justo lo que el organizador
recauda.
\end{solution}

\begin{solution}{pb:g11:prob:1}
\textbf{1.} $G$ toma los valores $-2, -1, 0, 1, 2, 3$, cada uno con
\omterm{def:g10:proba:distribution}{probabilidad} $\frac16$: $\E(G) = \frac{-2 - 1 + 0 + 1 + 2 + 3}{6} =
\frac12$ euro.

\textbf{2.} $\E(G^2) = \frac{4 + 1 + 0 + 1 + 4 + 9}{6} = \frac{19}{6}$;
$V(G) = \frac{19}{6} - \frac14 = \frac{35}{12} \approx 2.92$;
$\sigma(G) \approx 1.71$.

\textbf{3.} $\E(2G - 1) = 2 \times \frac12 - 1 = 0$;
$V(2G - 1) = 4\,V(G) = \frac{35}{3}$.

\textbf{4.} $\E(\text{suma}) = 7$: precio justo, $7$ euros. A $8$, la
pérdida esperada es de $1$ euro por partida.

\textbf{5.} Prima justa: $0.05 \times 800 = 40$ euros. Cobrando $60$: un
beneficio esperado de $20$ euros por póliza.

\textbf{6.} Repartir $5:3$ premia el \emph{pasado}, pero el bote es de
quien gane el \emph{futuro}, y B todavía podría ganar. Dárselo todo a A
trata una victoria probable como si fuera segura: la pequeña opción de B
vale algo.

\textbf{7.} Rondas por jugar: $3$; resultados (victoria de A o de B por
ronda): AAA, AAB, ABA, ABB, BAA, BAB, BBA, BBB. B se lleva la serie solo
si gana las tres: únicamente BBB. A gana la serie en $7$ de los $8$ casos:
reparto justo, $64 \times \frac78 = 56$ pistolas para A y $8$ para B.

\textbf{8.} Jugar rondas de más, ya inútiles, no cambia la victoria de
nadie: una vez que A tiene su sexta victoria, las rondas posteriores son
pura ceremonia. Pero fijar el horizonte en exactamente $3$ rondas hace
equiprobables los $8$ resultados; contar con honradez exige historias de
la misma longitud, y alargarlas no cuesta nada.

\textbf{9.} En $5$--$5$: una ronda decisiva, $32$ para cada uno. En
$5$--$4$: A obtiene $\frac{64 + 32}{2} = 48$. En $5$--$3$: la ronda
siguiente lleva a $64$ (si la gana A) o a la posición $5$--$4$, que vale
$48$: la parte de A es $\frac{64 + 48}{2} = 56$; el $56 : 8$ de Fermat,
deducido caminando hacia atrás.

\textbf{10.} A necesita $2$ y B necesita $3$: se juegan $4$ rondas
imaginarias ($16$ resultados). B se lleva la serie con $3$ o $4$ victorias
suyas: $4 + 1 = 5$ resultados. Reparto:
B: $64 \times \frac{5}{16} = 20$; A: $44$.

\textbf{11.} El valor presente justo de un pago futuro incierto es su
\omterm{def:g11:prob:expectation}{esperanza}, o sea, el promedio de lo que puede venir ponderado por las
\omterm{def:g10:proba:distribution}{probabilidades}. Los seguros (pregunta 5) y las finanzas (valorar pagos
inciertos) son esa frase convertida en industrias; y los casinos también,
desde el otro lado de la mesa.

\textbf{12.} El bote acaba siempre en algún sitio: las dos partes son
\omterm{def:g11:prob:rv}{variables aleatorias} cuya suma es la constante $64$, así que por
linealidad $\E_A + \E_B = 64$; las partes justas agotan el bote, sea cual
sea el marcador.

\textbf{13.} Linealidad: $\E(X + Y) = 3.5 + 3.5 = 7$, sin tabla, y cierto
incluso con dos dados pegados. \omterm{def:g11:prob:variance}{Varianza} de un dado:
$\E(X^2) - 3.5^2 = \frac{91}{6} - \frac{49}{4} = \frac{35}{12}$; para
dados independientes,
$V(X + Y) = \frac{35}{12} + \frac{35}{12} = \frac{35}{6}$.

\textbf{14.} Cada invitado fijado recupera su propio sombrero con
\omterm{def:g10:proba:distribution}{probabilidad} $\frac1n$ (el suyo es uno entre $n$). Sumando sobre los $n$
invitados: $\E(X) = n \times \frac1n = 1$; un invitado afortunado de
\omterm{def:g11:stat:mean}{media}, tenga la fiesta dos sombreros o dos mil.

\textbf{15.} $n = 2$: los repartos son ``los dos aciertan'' ($X = 2$) o
``intercambiados'' ($X = 0$): $\E = 1$. $n = 3$: los seis repartos dan
$X = 3, 1, 1, 1, 0, 0$: $\E = \frac{6}{6} = 1$.

\textbf{16.} El cálculo sumó las \omterm{def:g11:prob:expectation}{esperanzas} de los $n$ indicadores
individuales (``el invitado $i$ recuperó su sombrero''), y la linealidad
suma \omterm{def:g11:prob:expectation}{esperanzas} \emph{sin condiciones}: la dependencia arruinaría un
producto o una \omterm{def:g11:prob:variance}{varianza}, pero nunca una suma de \omterm{def:g11:prob:expectation}{esperanzas}. Ese es el
superpoder de la linealidad: no hace preguntas sobre independencia.

\textbf{17.} Las dos ganancias: $\E = 0.5 \times 10 - 5 = 0$ y
$0.005 \times 1000 - 5 = 0$. \omterm{def:g11:prob:variance}{Varianzas} (de la ganancia): A:
$\E(G^2) = 0.5 \times 25 + 0.5 \times 25 = 25$; B:
$0.005 \times 995^2 + 0.995 \times 25 \approx 4\,975$: $\sigma_A = 5$ y
$\sigma_B \approx 70.5$. La gente compra el B: lo que compra no es
\omterm{def:g11:prob:expectation}{esperanza} (nula en los dos casos), sino \emph{dispersión}: una pequeña
opción de una cola derecha que cambia la vida.

\textbf{18.} Que la primera cruz salga en el lanzamiento $n$ tiene
\omterm{def:g10:proba:distribution}{probabilidad} $2^{-n}$ y paga $2^n$: cada $n$ aporta
$2^{-n} \times 2^n = 1$ a la \omterm{def:g11:prob:expectation}{esperanza}, que es por tanto
$1 + 1 + 1 + \dots$: infinita. Y, sin embargo, nadie en su sano juicio
paga $1\,000$ por jugar una vez: con \omterm{def:g10:proba:distribution}{probabilidad} $\frac{15}{16}$ el pago
es como mucho $16$. La \omterm{def:g11:prob:expectation}{esperanza} resume el largo plazo de apuestas
\emph{repetibles}; un juego único con una cola descontrolada no se valora
con su \omterm{def:g11:stat:mean}{media}.

\textbf{19.} \omterm{def:g11:prob:expectation}{Esperanzas}: asegurarse, $-20$; ir a pecho descubierto,
$-0.01 \times 1000 = -10$. La estrategia sin seguro gana de \omterm{def:g11:stat:mean}{media}, para
quien pueda absorber la pérdida y repetir la apuesta muchas veces. Una
familia que no sobrevive al $-1\,000$ no está en el negocio de promediar:
paga $10$ de más para borrar la \omterm{def:g11:prob:variance}{varianza}. Las aseguradoras promedian; las
familias compran certeza.

\textbf{20.} Nació repartiendo un bote entre dos jugadores impacientes;
la fortalece la linealidad, que suma \omterm{def:g11:prob:expectation}{esperanzas} sin pedirle permiso a la
dependencia (los sombreros); es ciega al riesgo, que tiene que medir la
\omterm{def:g11:prob:variance}{varianza} (las rifas, el juego de San Petersburgo, la póliza de la
familia); y la corona el año que viene la ley de los grandes números, que
explica por qué los promedios siempre acaban cobrando.
\end{solution}
