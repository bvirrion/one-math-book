\begin{solution}{exo:g11:func:1}
$f$: hace falta $3 - x \geq 0$, \omterm{def:g11:func:function}{dominio} $\intoc{-\infty}{3}$.

$g$: hace falta $x^2 - 4 \neq 0$, es decir, $x \neq \pm 2$: \omterm{def:g11:func:function}{dominio}
$\R \setminus \{-2, 2\}$.

$h$: hacen falta $x \geq 0$ y $x \neq 5$: \omterm{def:g11:func:function}{dominio}

\end{solution}

\begin{solution}{exo:g11:func:2}
$\sqrt{11} < \sqrt{13}$, porque la \omterm{def:g11:quad:discriminant}{raíz} cuadrada es \omterm{def:g11:func:monotone}{creciente} y
$11 < 13$.

$(-2.1)^2 > (-2.05)^2$, porque $x^2$ es \omterm{def:g10:functions:variations}{decreciente} en
$\intoc{-\infty}{0}$ y $-2.1 < -2.05$.

$\frac{1}{0.9} > \frac{1}{1.1}$, porque $\frac1x$ es \omterm{def:g10:functions:variations}{decreciente} en
$\intoo{0}{+\infty}$ y $0.9 < 1.1$.

$(-1.01)^3 < (-0.99)^3$, porque $x^3$ es \omterm{def:g11:func:monotone}{creciente} y $-1.01 < -0.99$.
\end{solution}

\begin{solution}{exo:g11:func:3}
$f(-x) = 3(-x)^4 - (-x)^2 = 3x^4 - x^2 = f(x)$: \omterm{def:g11:func:parity}{par}.

$g(-x) = -x^3 - x = -g(x)$: \omterm{def:g11:func:parity}{impar}.

$h(1) = 2$ y $h(-1) = 1 - 1 = 0$, así que $h(-1)$ no es ni $h(1)$ ni
$-h(1)$: ni \omterm{def:g11:func:parity}{par} ni \omterm{def:g11:func:parity}{impar}.

$k(-x) = \frac{-x}{x^2+1} = -k(x)$: \omterm{def:g11:func:parity}{impar}.
\end{solution}

\begin{solution}{exo:g11:func:4}
Leemos la \omterm{def:g10:functions:table}{tabla de variación}. En $\intcc{-3}{1}$, $f$ sube de $-2$ a $4$ y
cruza el valor $1$ exactamente una vez; en $\intcc{1}{5}$ baja de $4$ a
$0$ y cruza $1$ otra vez: la \omterm{def:g10:algebra:equation}{ecuación} $f(x) = 1$ tiene $2$ soluciones. El
\omterm{def:g10:functions:extrema}{máximo} de $f$ es $f(1) = 4 < 5$, así que $f(x) = 5$ no tiene solución.
\end{solution}

\begin{solution}{exo:g11:func:5}
Sea $2 \leq u < v$. Entonces
\[
f(v) - f(u) = v^2 - 4v - u^2 + 4u = (v-u)(v+u) - 4(v-u)
= (v - u)(u + v - 4).
\]
Los dos factores son positivos: $v - u > 0$ y $u + v > 2 + 2 = 4$, porque
$u \geq 2$ y $v > 2$. Por tanto, $f(v) > f(u)$: $f$ es estrictamente
\omterm{def:g11:func:monotone}{creciente} en $\intco{2}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:6}
$f = \sqrt{u}$ con $u(x) = 5 - x$ \omterm{def:g10:functions:variations}{decreciente} y no negativa en
$\intoc{-\infty}{5}$: $f$ es \omterm{def:g10:functions:variations}{decreciente} ahí
(\cref{prop:g11:func:transforms}).

$g = \frac1u$ con $u(x) = x^2 + 1$ \omterm{def:g11:func:monotone}{creciente} y positiva en
$\intco{0}{+\infty}$: $g$ es \omterm{def:g10:functions:variations}{decreciente} ahí.

$h = 3 - 2\sqrt x$: $\sqrt x$ es \omterm{def:g11:func:monotone}{creciente}, multiplicar por $-2 < 0$
invierte y sumar $3$ no cambia nada: $h$ es \omterm{def:g10:functions:variations}{decreciente} en
$\intco{0}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:7}
\emph{1.} $f$ ya está en \omterm{thm:g11:quad:canonical}{forma canónica}: \omterm{def:g10:functions:variations}{decreciente} en
$\intoc{-\infty}{2}$, \omterm{def:g11:func:monotone}{creciente} en $\intco{2}{+\infty}$, con \omterm{def:g10:functions:extrema}{mínimo}
$f(2) = -3$.

\emph{2.} $g = \frac{1}{u}$ con $u(x) = (x-2)^2 + 1 = f(x) + 4$. Sumar $4$
conserva la variación de $f$, y $u \geq 1 > 0$. Tomar el inverso la
invierte: $g$ es \omterm{def:g11:func:monotone}{creciente} en $\intoc{-\infty}{2}$ y \omterm{def:g10:functions:variations}{decreciente} en
$\intco{2}{+\infty}$, con \omterm{def:g10:functions:extrema}{máximo} $g(2) = \frac{1}{1} = 1$.
\end{solution}

\begin{solution}{exo:g11:func:8}
Sirve cualquier curva en forma de colina que pase por los tres puntos.
Entonces: $f + 2$ pasa por $(0,3)$, $(2,5)$, $(4,3)$, con la misma
variación (desplazamiento vertical); $-f$ pasa por $(0,-1)$, $(2,-3)$,
$(4,-1)$ con la variación \emph{invertida} (la cima se vuelve valle); y
$2f$ pasa por $(0,2)$, $(2,6)$, $(4,2)$, con la misma variación, pero
amplificada verticalmente.
\end{solution}

\begin{solution}{exo:g11:func:9}
$2 - \frac{1}{x+1} = \frac{2(x+1) - 1}{x+1} = \frac{2x+1}{x+1} = f(x)$. En
$\intoo{-1}{+\infty}$, $u(x) = x + 1$ es \omterm{def:g11:func:monotone}{creciente} y positiva, luego
$\frac1u$ es \omterm{def:g10:functions:variations}{decreciente}, $-\frac1u$ es \omterm{def:g11:func:monotone}{creciente} y sumar $2$ no cambia
nada: $f$ es estrictamente \omterm{def:g11:func:monotone}{creciente} en $\intoo{-1}{+\infty}$.
\end{solution}

\begin{solution}{exo:g11:func:10}
\emph{1. Verdadero.} Si $u < v$, entonces
$(f+g)(v) - (f+g)(u) = \bigl(f(v)-f(u)\bigr) + \bigl(g(v)-g(u)\bigr)$ es
suma de dos números no negativos.

\emph{2. Falso.} $f(x) = g(x) = x$ son \omterm{def:g11:func:monotone}{crecientes} en $\R$, pero su
producto $x^2$ no es \omterm{def:g11:func:monotone}{creciente} en $\R$ (decrece en $\intoc{-\infty}{0}$).
(El enunciado sí es cierto si además las dos funciones son no negativas.)

\emph{3. Falso.} El mismo contraejemplo: $f(x) = x$ es \omterm{def:g11:func:monotone}{creciente} en $\R$,
pero $f^2(x) = x^2$ no lo es.
\end{solution}

\begin{solution}{exo:g11:func:11}
\emph{1.} Para $0 < u < v$:
\[
f(v) - f(u) = (v - u) + \frac1v - \frac1u
= (v - u) + \frac{u - v}{uv}
= (v - u)\left(1 - \frac{1}{uv}\right)
= (v - u)\,\frac{uv - 1}{uv}.
\]

\emph{2.} Siempre $v - u > 0$ y $uv > 0$. Si $0 < u < v \leq 1$, entonces
$uv < 1$, luego $uv - 1 < 0$ y $f(v) < f(u)$: $f$ es estrictamente
\omterm{def:g10:functions:variations}{decreciente} en $\intoc{0}{1}$. Si $1 \leq u < v$, entonces $uv > 1$ y
$f(v) > f(u)$: $f$ es estrictamente \omterm{def:g11:func:monotone}{creciente} en $\intco{1}{+\infty}$.

\emph{3.} $f$ decrece antes de $1$ y crece después, así que
$f(x) \geq f(1) = 2$ para todo $x > 0$, con igualdad exactamente en
$x = 1$. (Esta desigualdad reaparece con muchos disfraces; compara con el
\cref{ex:g11:quad:canonical} y con las desigualdades de la recta tangente
del \cref{ch:g12:deriv}.)
\end{solution}

\begin{solution}{pb:g11:func:1}
\textbf{1.} $x^4 - 3x^2$: \omterm{def:g11:func:parity}{par} ($(-x)^4 - 3(-x)^2 = x^4 - 3x^2$).
$x^3 - x$: \omterm{def:g11:func:parity}{impar}. $x^2 + x$: ninguna de las dos ($f(-1) = 0 \neq
f(1) = 2$ y tampoco $= -f(1)$). $\abs x$: \omterm{def:g11:func:parity}{par}. $\frac1x$: \omterm{def:g11:func:parity}{impar}.

\textbf{2.} \omterm{def:g11:func:parity}{Par}: la curva es simétrica respecto del eje $y$ (espejo).
\omterm{def:g11:func:parity}{Impar}: simétrica respecto del origen (medio giro).

\textbf{3.} Si $f$ y $g$ son \omterm{def:g11:func:parity}{impares}:
$(fg)(-x) = f(-x)g(-x) = (-f(x))(-g(x)) = (fg)(x)$: \omterm{def:g11:func:parity}{par}. Tabla: \omterm{def:g11:func:parity}{par}
$\times$ \omterm{def:g11:func:parity}{par} $=$ \omterm{def:g11:func:parity}{par}; \omterm{def:g11:func:parity}{par} $\times$ \omterm{def:g11:func:parity}{impar} $=$ \omterm{def:g11:func:parity}{impar}; \omterm{def:g11:func:parity}{impar} $\times$ \omterm{def:g11:func:parity}{impar}
$=$ \omterm{def:g11:func:parity}{par}; la regla de los signos, con paridades.

\textbf{4.} \omterm{def:g11:func:parity}{Impar} en $0$: $f(0) = -f(0)$, luego $f(0) = 0$. \omterm{def:g11:func:parity}{Par}:
$f(-3) = f(3) = 7$. \omterm{def:g11:func:parity}{Par} e \omterm{def:g11:func:parity}{impar} a la vez: $f(x) = f(-x) = -f(x)$ para todo
$x$, luego $2f(x) = 0$: $f$ es la \omterm{def:g11:func:function}{función} nula.

\textbf{5.} La mitad derecha ($x \geq 0$) determina por completo una
\omterm{def:g11:func:parity}{función par}: la mitad izquierda es su reflejo. Ni \omterm{def:g11:func:parity}{par} ni \omterm{def:g11:func:parity}{impar}:
$x^2 + x$ (pregunta 1); sus dos simetrías fallan en $x = 1$.

\textbf{6.} $E(x) = \frac{(x^3 + x^2 + 1) + (-x^3 + x^2 + 1)}{2} =
x^2 + 1$ (\omterm{def:g11:func:parity}{par}) y $O(x) = \frac{(x^3 + x^2 + 1) - (-x^3 + x^2 + 1)}{2} =
x^3$ (\omterm{def:g11:func:parity}{impar}); su suma es $f$.

\textbf{7.} $E(-x) = \frac{f(-x) + f(x)}{2} = E(x)$: \omterm{def:g11:func:parity}{par}.
$O(-x) = \frac{f(-x) - f(x)}{2} = -O(x)$: \omterm{def:g11:func:parity}{impar}. Y
$E(x) + O(x) = f(x)$ sin más que sumar: toda \omterm{def:g11:func:function}{función} se descompone.

\textbf{8.} De $E_1 + O_1 = E_2 + O_2$ se sigue $E_1 - E_2 = O_2 - O_1$.
El primer miembro es \omterm{def:g11:func:parity}{par} y el segundo, \omterm{def:g11:func:parity}{impar}, así que esa \omterm{def:g11:func:function}{función} es las
dos cosas y, por tanto, nula (pregunta 4): $E_1 = E_2$ y $O_1 = O_2$. Solo
hay una descomposición.

\textbf{9.} $E(x) = \frac12\left(\frac{1}{1 - x} + \frac{1}{1 +
x}\right) = \frac12 \cdot \frac{(1 + x) + (1 - x)}{1 - x^2}
= \frac{1}{1 - x^2}$, y
$O(x) = \frac12 \cdot \frac{(1 + x) - (1 - x)}{1 - x^2}
= \frac{x}{1 - x^2}$. Comprobación: $E$ es \omterm{def:g11:func:parity}{par}, $O$ es \omterm{def:g11:func:parity}{impar} y su suma es
$\frac{1 + x}{1 - x^2} = \frac{1 + x}{(1 - x)(1 + x)} = \frac{1}{1 - x} =
f(x)$.

\textbf{10.} Porque cada pieza obedece una simetría, y entonces basta con
la mitad del trabajo (y con la mitad de los datos) para cada una:
resolver en un lado y reflejar hacia el otro; además, muchas operaciones
(la integración, las series, el análisis de señales) tratan las piezas
simétricas mucho más fácilmente que las funciones en bruto.

\textbf{11.} $(x-3)^2$: desplazada $3$ a la derecha. $x^2 + 2$:
desplazada $2$ hacia arriba. $(x-3)^2 + 2$: las dos cosas. $-x^2$: dada la
vuelta. $2x^2$: dilatada verticalmente por $2$ (cuenco más estrecho).

\textbf{12.} Para $x \geq 2$: $(x - 2) + (x + 2) = 2x$. Para
$-2 \leq x \leq 2$: $(2 - x) + (x + 2) = 4$. Para $x \leq -2$: $-2x$. La
curva baja con \omterm{def:g10:reffunc:affine}{pendiente} $-2$, corre plana a la altura $4$ entre $-2$ y
$2$ y sube después con \omterm{def:g10:reffunc:affine}{pendiente} $2$: un valle de fondo plano.

\textbf{13.} Desplazamiento vertical: $f(x) + 2$ sigue siendo \omterm{def:g11:func:parity}{par}
($f(-x) + 2 = f(x) + 2$). Dilatación vertical: $2f(x)$ sigue siendo \omterm{def:g11:func:parity}{par}.
Desplazamiento horizontal: $(x - 3)^2$ no es \omterm{def:g11:func:parity}{par}
($f(-1) = 16 \neq 4 = f(1)$): correrse de lado destruye el espejo.

\textbf{14.} En el fondo plano el valor es $4 \neq 6$. Para $x \geq 2$:
$2x = 6$, $x = 3$. Para $x \leq -2$: $-2x = 6$, $x = -3$. Soluciones:
$\pm 3$.

\textbf{15.} $f(x) = a(x - 2)^2 - 3$ con $f(0) = 4a - 3 = 1$: $a = 1$,
luego $f(x) = (x - 2)^2 - 3$.

\textbf{16.} $g(v) - g(u) = (v - u) + 4\,\frac{u - v}{uv}
= (v - u)\,\frac{uv - 4}{uv}$: negativo para $u < v$ con $uv < 4$ y
positivo con $uv > 4$; \omterm{def:g10:functions:variations}{decreciente} en $\intoc{0}{2}$ y \omterm{def:g11:func:monotone}{creciente} en
$\intco{2}{+\infty}$; \omterm{def:g10:functions:extrema}{mínimo} en $x = 2$, de valor $g(2) = 4$.

\textbf{17.} Lados $x$ y $\frac{16}{x}$:
$P(x) = 2\left(x + \frac{16}{x}\right)$, \omterm{def:g10:functions:extrema}{mínimo} (con la máquina de la
pregunta 16, cambiando $4$ por $16$, con punto de giro $\sqrt{16} = 4$) en
$x = 4$: el \emph{cuadrado} de $4 \times 4$, de perímetro $16$; la
respuesta de Dido, ahora demostrada para cualquier longitud real de lado.

\textbf{18.} En $\intco{0}{+\infty}$: $x \mapsto x^2 + 1$ crece y la \omterm{def:g11:quad:discriminant}{raíz}
cuadrada crece, así que la composición crece. Y
$h(-x) = \sqrt{x^2 + 1} = h(x)$: $h$ es \omterm{def:g11:func:parity}{par}, así que por simetría de
espejo decrece en $\intoc{-\infty}{0}$. \omterm{def:g10:functions:extrema}{Mínimo} $h(0) = 1$.

\textbf{19.} Distancia al cuadrado de $(x, \sqrt x)$ a $(3, 0)$:
$d^2(x) = (x - 3)^2 + x = x^2 - 5x + 9$, mínima en $x = \frac52$: el punto
más cercano es $\left(\frac52, \sqrt{\frac52}\right)$, a distancia
$\sqrt{\frac{11}{4}} = \frac{\sqrt{11}}{2}$. Minimizar $d^2$ es legítimo
porque elevar al cuadrado es estrictamente \omterm{def:g11:func:monotone}{creciente} sobre los números
positivos: $d$ y $d^2$ alcanzan su \omterm{def:g10:functions:extrema}{mínimo} en el mismo sitio.

\textbf{20.} Materia prima: las curvas de referencia, sabidas de memoria.
Máquinas: desplazamientos, simetrías y dilataciones que montan \omterm{def:g10:functions:graph}{gráficas}
nuevas a partir de las viejas. Control de calidad: la paridad y la
descomposición par--impar, que reducen a la mitad cualquier estudio. Banco
de medida: la factorización de $f(v) - f(u)$, que lee directamente la
variación y los \omterm{def:g10:functions:extrema}{mínimos}. Y la herramienta eléctrica del capítulo
siguiente: la \emph{derivada}, que los lee de un vistazo.
\end{solution}
