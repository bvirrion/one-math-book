\section*{अध्याय \ref{ch:g11:stat} --- वर्णनात्मक सांख्यिकी}

\begin{solution}{exo:g11:stat:1}
$n = 8$, योग $71$: \omterm{def:g11:stat:mean}{माध्य} $\bar x = \frac{71}{8} = 8.875$। \omterm{def:g11:stat:median}{माध्यिका}: $4$-वें और $5$-वें मानों का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु},
$\frac{8 + 9}{2} = 8.5$। \omterm{def:g11:stat:quartiles}{चतुर्थक}: $8$ का एक चौथाई $2$ है, इसलिए $Q_1$ $2$-वाँ मान है, $7$; तीन
चौथाई $6$ है, इसलिए $Q_3$ $6$-वाँ मान है, $10$। अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास}: $10 - 7 = 3$।
\end{solution}

\begin{solution}{exo:g11:stat:2}
योग $40$, इसलिए $\bar x = 5$। वर्गों का योग: $4 + 16 + 16 + 16 + 25 + 25 + 49 + 81 = 232$। संक्षिप्त सूत्र से
\[
V = \frac{232}{8} - 5^2 = 29 - 25 = 4,
\qquad
\sigma = 2 .
\]
\end{solution}

\begin{solution}{exo:g11:stat:3}
\omterm{def:g11:stat:mean}{माध्य}:
\[
\bar x = \frac{1 \times 6 + 2 \times 9 + 3 \times 8 + 4 \times 10
+ 5 \times 9 + 6 \times 8}{50}
= \frac{181}{50} = 3.62 .
\]
\omterm{def:g11:stat:median}{माध्यिका}: संचयी बारंबारताएँ $6, 15, 23, 33, \dots$ हैं, इसलिए $25$-वाँ और $26$-वाँ क्रमबद्ध
परिणाम दोनों $4$ हैं: \omterm{def:g11:stat:median}{माध्यिका} $4$ है।
\end{solution}

\begin{solution}{exo:g11:stat:4}
$15$ अंकों का कुल $15 \times 11.2 = 168$ है; नए विद्यार्थी के साथ $\frac{168 + 18}{16} = \frac{186}{16} = 11.625$।
\end{solution}

\begin{solution}{exo:g11:stat:5}
$\frac{6 + 8 + x + 12 + 14}{5} = 10$ से $40 + x = 50$ मिलता है, इसलिए $x = 10$। \omterm{def:g11:stat:mean}{माध्य} $10$ से $6, 8, 10, 12, 14$ के विचलन $-4, -2, 0, 2, 4$ हैं,
जिनके वर्गों का योग $40$ है: $V = \frac{40}{5} = 8$ और $\sigma = 2\sqrt2 \approx 2.83$।
\end{solution}

\begin{solution}{exo:g11:stat:6}
$a = \frac12$, $b = 4$ के साथ \cref{prop:g11:stat:affine} से: नया \omterm{def:g11:stat:mean}{माध्य} $\frac{12.4}{2} + 4 = 10.2$; नया \omterm{def:g11:stat:variance}{मानक विचलन} $\frac{2.5}{2} = 1.25$।
\end{solution}

\begin{solution}{exo:g11:stat:7}
मिला हुआ \omterm{def:g11:stat:mean}{माध्य} \emph{भारित} औसत है
\[
\frac{20 \times 10 + 30 \times 15}{50} = \frac{650}{50} = 13,
\]
न कि $12.5$: समूह B अधिक मान देता है, इसलिए वह मिले हुए \omterm{def:g11:stat:mean}{माध्य} को अपनी ओर खींच
लेता है।
\end{solution}

\begin{solution}{exo:g11:stat:8}
दोनों पंक्तियाँ औसतन लगभग लक्ष्य पर हैं ($1.002$ और $1.001$ kg), पर पंक्ति 2 का
\omterm{def:g11:stat:variance}{मानक विचलन} तीन गुना बड़ा है: उसके थैले कहीं अधिक बदलते हैं और कहीं अधिक
ध्यान खींचने वाले कम-भरे तथा अधिक-भरे थैले बनते हैं। पंक्ति 2 को पहले
सँभालना चाहिए --- सुधार की ज़रूरत केवल \omterm{def:g11:stat:mean}{माध्य} को नहीं, एकरूपता को है।
\end{solution}

\begin{solution}{exo:g11:stat:9}
$\bar x = \frac{70}{10} = 7$; $V = \frac{540}{10} - 7^2 = 54 - 49 = 5$; $\sigma = \sqrt5 \approx 2.24$।
\end{solution}

\begin{solution}{exo:g11:stat:10}
\emph{1.} $23$ के साथ: योग $50$, \omterm{def:g11:stat:mean}{माध्य} $5$; \omterm{def:g11:stat:median}{माध्यिका} = $5$-वें और $6$-वें
मानों का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} $= 3$। $23$ के बिना: योग $27$, \omterm{def:g11:stat:mean}{माध्य} $3$; \omterm{def:g11:stat:median}{माध्यिका} (9
मानों में पाँचवाँ) $= 3$।

\emph{2.} $23$ के साथ ($n = 10$): $Q_1$ $3$-वाँ मान है ($2$), $Q_3$ $8$-वाँ
($4$): अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} $2$। बिना ($n = 9$): $Q_1$ $3$-वाँ मान है ($2$),
$Q_3$ $7$-वाँ ($4$): फिर भी $2$।

\emph{3.} अकेला बहिर्मान \omterm{def:g11:stat:mean}{माध्य} को $3$ से $5$ तक घसीट ले जाता है पर \omterm{def:g11:stat:median}{माध्यिका}
और अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} ज्यों के त्यों रहते हैं: वे दोनों \emph{सुदृढ़} हैं। जब
किसी आँकड़ा-समुच्चय में विसंगत मान हो सकते हों, तब \omterm{def:g11:stat:mean}{माध्य} और \omterm{def:g11:stat:variance}{मानक विचलन} के
बजाय \omterm{def:g11:stat:median}{माध्यिका} और अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} बताइए।
\end{solution}

\begin{solution}{pb:g11:stat:1}
\textbf{1.} \omterm{def:g11:stat:mean}{माध्य} $\bar x = \frac{25}{5} = 5$। वर्ग किए गए विचलन: $1, 9, 1, 0, 9$: \omterm{def:g11:stat:variance}{प्रसरण} $\frac{20}{5} = 4$, इसलिए $\sigma = 2$।

\textbf{2.} वर्गों का \omterm{def:g11:stat:mean}{माध्य}: $\frac{16 + 64 + 36 + 25 + 4}{5} = 29$; घटा $\bar x^2 = 25$: \omterm{def:g11:stat:variance}{प्रसरण} $4$। वही उत्तर, आधे
काम में।

\textbf{3.} \omterm{def:g11:stat:mean}{माध्य} $= \frac{3 \times 8 + 5 \times 10 + 2 \times
14}{10} = 10.2$; \omterm{def:g11:stat:variance}{प्रसरण} $= \frac{3 \times 64 + 5 \times 100 +
2 \times 196}{10} - 10.2^2 = 108.4 - 104.04 = 4.36$; $\sigma \approx 2.09$।

\textbf{4.} \omterm{def:g11:stat:mean}{माध्य}: $1.8 \times 20 + 32 = 68\,^\circ$F; $\sigma$: $1.8 \times 3 = 5.4\,^\circ$F --- खिसकाव $+32$ कुछ भी नहीं
फैलाता।

\textbf{5.} $b$ जोड़ने पर: हर विचलन $x_i + b - (\bar x + b)
= x_i - \bar x$ अपरिवर्तित रहता है, इसलिए \omterm{def:g11:stat:variance}{प्रसरण}
भी। $a$ से गुणा करने पर: विचलन $a$ गुने हो जाते हैं, उनके वर्ग $a^2$
गुने, \omterm{def:g11:stat:variance}{प्रसरण} $a^2$ गुना, और $\sigma$ $\sqrt{a^2} = \abs a$ गुना।

\textbf{6.} $z_A = \frac{15 - 12}{2} = 1.5$; $z_B = \frac{13 - 8}{4} = 1.25$। ऐलिस अपनी भीड़ से अधिक ऊपर खड़ी है: कठिन परीक्षा
का $15$ आसान परीक्षा के $13$ को हरा देता है।

\textbf{7.} $z$-मान: $-0.5,\ 1.5,\ 0.5,\ 0,\ -1.5$: \omterm{def:g11:stat:mean}{माध्य} $0$, \omterm{def:g11:stat:variance}{मानक विचलन} $1$। सामान्य रूप से
मानकीकरण पहले $\bar x$ घटाता है (जिससे \omterm{def:g11:stat:mean}{माध्य} $0$ हो जाता है और $\sigma$ अछूता
रहता है) और फिर $\sigma$ से भाग देता है (जिससे नया $\sigma$ $1$ हो जाता है):
यानी प्रश्न 5, दो बार।

\textbf{8.} $z = 2$: $100 + 2 \times 15 = 130$। और $z(76) = \frac{76 - 100}{15} = -1.6$।

\textbf{9.} शिशु \emph{अपनी आयु के शिशुओं} के \omterm{def:g11:stat:mean}{माध्य} भार से $2.5$ \omterm{def:g11:stat:variance}{मानक विचलन}
नीचे है --- अपने समूह के लिए चिंताजनक रूप से हल्का। कच्चे किलोग्राम अलग-अलग
आयु के बीच तुलना नहीं करा सकते (हर शिशु भार बढ़ाता है); \omterm{pb:g11:stat:1}{z-मान} हर बच्चे को
उसके अपने आयु-समूह के सामने नापते हैं, और सब आयु के लिए एक ही पैमाना देते
हैं।

\textbf{10.} \omterm{def:g11:stat:mean}{माध्य} $5$; वर्गों का \omterm{def:g11:stat:mean}{माध्य} $\frac{622}{10} = 62.2$, \omterm{def:g11:stat:variance}{प्रसरण} $37.2$, $\sigma \approx 6.10$; $z(23) = \frac{18}{6.10} \approx 2.95$:
ख़तरे की सीमा से ज़रा नीचे --- परीक्षण मुश्किल से पलक झपकता है।

\textbf{11.} $23$ के बिना: \omterm{def:g11:stat:mean}{माध्य} $3$, \omterm{def:g11:stat:variance}{प्रसरण} $\frac{93}{9} - 9 \approx 1.33$, $\sigma \approx 1.15$; स्वच्छ पैमाने
के सामने $23$ का मान $z \approx \frac{20}{1.15} \approx 17$ बनता है: एक दैत्य। बहिर्मान ने \omterm{def:g11:stat:mean}{माध्य} और $\sigma$
दोनों को फुला दिया और स्वयं को छिपा लिया --- इसीलिए \cref{exo:g11:stat:10} दूषण का ख़तरा होने
पर सुदृढ़ सारांश (\omterm{def:g11:stat:median}{माध्यिका}, अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास}) की सलाह देता है।

\textbf{12.} \omterm{def:g11:stat:variance}{प्रसरण} के योग $\sigma^2 = \frac1N \sum (x_i - \bar x)^2$ में केवल वे पद रखिए जिनके लिए $\abs{x_i - \bar x} \geq k\sigma$ हो:
ऐसे $pN$ पद हैं, और हर एक कम से कम $(k\sigma)^2$ है, इसलिए $\sigma^2 \geq \frac{1}{N} \cdot pN \cdot k^2\sigma^2 =
p\,k^2\sigma^2$: $k^2 \sigma^2$ से भाग
दीजिए (प्रसार शून्येतर): $p \leq \frac{1}{k^2}$।

\textbf{13.} $2\sigma$ से आगे: किसी भी आँकड़ा-समुच्चय का अधिक से अधिक $\frac14 = 25\,\%$;
$3\sigma$ से आगे: अधिक से अधिक $\frac19 \approx 11\,\%$। प्रश्न 1 के आँकड़े: \omterm{def:g11:stat:mean}{माध्य} $5$ से $2\sigma = 4$
से आगे का अर्थ है $\leq 1$ या $\geq 9$: पाँचों में से कोई नहीं --- $0\,\% \leq 25\,\%$, जैसा
वादा था।

\textbf{14.} $\intcc{49.7}{50.3}$ के बाहर होने का अर्थ है $3\sigma$ से आगे: उत्पादन का अधिक से
अधिक $\frac19$, और बंटन की आकृति \emph{चाहे जो हो} --- चेबिशेव \omterm{def:g11:stat:mean}{माध्य} और $\sigma$ के
सिवा कुछ नहीं माँगता। घंटी-आकार वाला उत्पादन कहीं बेहतर सिमटता है ($3\sigma$ से
आगे लगभग $0.3\,\%$): यही प्रसामान्य नियम है, कक्षा 12 में।

\textbf{15.} $-19\,\%$ \omterm{def:g11:stat:mean}{माध्य} $5$ से $24$ अंक नीचे है। निधि B: $24 = 2\sigma_B$: चेबिशेव
इतने बुरे वर्षों को $\frac14$ तक की छूट देता है। निधि A: $24 = 12\sigma_A$: अधिक से अधिक
$\frac{1}{144}$। एक ही औसत, पर ख़तरे की तुलना ही नहीं हो सकती --- $\sigma$ को ही वित्त
में \emph{जोखिम} (या अस्थिरता) कहा जाता है।

\textbf{16.} $\bar x$ के परितः प्रसार कीजिए: $(x_i - c)^2 = \left((x_i - \bar x) + (\bar x - c)\right)^2
= (x_i - \bar x)^2 + 2(\bar x - c)(x_i - \bar x) +
(\bar x - c)^2$। औसत लेने पर बीच वाला पद मर
जाता है (\omterm{def:g11:stat:mean}{माध्य} से विचलनों का योग शून्य है), और $f(c) = \sigma^2 + (\bar x - c)^2$ बचता है --- जो ठीक $c = \bar x$
पर \omterm{def:g10:functions:extrema}{न्यूनतम} है, और उसका \omterm{def:g10:functions:extrema}{न्यूनतम} मान $\sigma^2$ है। \omterm{def:g11:stat:mean}{माध्य} न्यूनतम-वर्ग वाला केंद्र
है, जैसे \omterm{def:g11:stat:median}{माध्यिका} न्यूनतम-निरपेक्ष वाला केंद्र है (\cref{pb:g10:stats:1})।

\textbf{17.} सीधे: $6$ से विचलन $-2, 2, 0, -1, -4$ हैं, वर्ग $4, 4, 0, 1, 16$: $f(6) = \frac{25}{5} = 5$। सूत्र से:
$\sigma^2 + (5 - 6)^2 = 4 + 1 = 5$। मेल हो गया।

\textbf{18.} दोनों का \omterm{def:g11:stat:mean}{माध्य} $7$ है; \omterm{def:g11:stat:variance}{प्रसरण}: $P$: $\frac{16 + 4 + 0 + 4 + 16}{5} = 8$; $Q$: $\frac{36 + 0 + 0 + 0 + 36}{5} = 14.4$:
एक ही \omterm{def:g11:stat:mean}{माध्य} के पीछे अलग-अलग प्रसार। $\{7 - a, 7, 7, 7, 7 + a\}$ के लिए: \omterm{def:g11:stat:variance}{प्रसरण} $\frac{2a^2}{5} = 8$ से $a = \sqrt{20} \approx 4.47$
मिलता है: आँकड़ा-समुच्चय $\{2.53, 7, 7, 7, 11.47\}$ $P$ का ठीक वही \omterm{def:g11:stat:mean}{माध्य} और \omterm{def:g11:stat:variance}{प्रसरण} उठाए हुए है
पर उसका रूप बिलकुल अलग है --- सारांश दबाते हैं, और दबाव में कुछ खोता है।

\textbf{19.} $\sigma^2 = 0$ का अर्थ है कि वर्गों का योग $(x_i - \bar x)^2$ लुप्त हो जाता है; वर्ग
$\geq 0$ होते हैं, इसलिए शून्य योग हर पद को शून्य होने पर मजबूर कर देता है:
सभी $i$ के लिए $x_i = \bar x$ --- यानी सारे मान बराबर।

\textbf{20.} केंद्र बताते हैं कि आँकड़े कहाँ रहते हैं; प्रसार बताते हैं कि
कितने ढीले-ढाले; \omterm{pb:g11:stat:1}{z-मान} सभी आँकड़ा-समुच्चयों, परीक्षाओं, आयु-समूहों और
मुद्राओं पर एक ही पैमाना रख देता है; चेबिशेव $\sigma$ को एक बेशर्त गारंटी में
बदल देता है; और प्रश्न 18 हमें ईमानदार रखता है --- चंद संख्याएँ कभी पूरी
कहानी नहीं समेटतीं। अगले वर्ष घंटी-वक्र इस पैमाने को यथार्थ \omterm{def:g10:proba:distribution}{प्रायिकताओं} में
पैना कर देगा।
\end{solution}
