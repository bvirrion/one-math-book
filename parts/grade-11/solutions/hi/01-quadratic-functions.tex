\section*{अध्याय \ref{ch:g11:quad} --- द्विघात फलन और समीकरण}

\begin{solution}{exo:g11:quad:1}
$x^2-5x+6$: $\Delta = 25 - 24 = 1$, \omterm{def:g11:quad:discriminant}{मूल} $\frac{5 \pm 1}{2}$, अर्थात् $2$ और $3$।

$2x^2+3x-2$: $\Delta = 9 + 16 = 25$, \omterm{def:g11:quad:discriminant}{मूल} $\frac{-3 \pm 5}{4}$, अर्थात् $\frac12$ और $-2$।

$x^2+2x+5$: $\Delta = 4 - 20 = -16 < 0$, कोई वास्तविक हल नहीं।

$9x^2-6x+1$: $\Delta = 36 - 36 = 0$, एक (द्विगुण) \omterm{def:g11:quad:discriminant}{मूल} $x_0 = \frac{6}{18} = \frac13$।
\end{solution}

\begin{solution}{exo:g11:quad:2}
$f(x) = x^2 - 6x + 11 = (x-3)^2 - 9 + 11 = (x-3)^2 + 2$: शीर्ष $(3, 2)$।

$g(x) = -2x^2+4x+1 = -2(x^2 - 2x) + 1 = -2\bigl((x-1)^2 - 1\bigr) + 1
= -2(x-1)^2 + 3$: शीर्ष $(1, 3)$।

$h(x) = 3x^2 + 6x = 3(x^2 + 2x) = 3\bigl((x+1)^2 - 1\bigr)
= 3(x+1)^2 - 3$: शीर्ष $(-1, -3)$।
\end{solution}

\begin{solution}{exo:g11:quad:3}
$x^2 - 3x - 10$: $\Delta = 9 + 40 = 49$, \omterm{def:g11:quad:discriminant}{मूल} $\frac{3 \pm 7}{2} = -2$ और $5$; अग्र गुणांक धनात्मक है, इसलिए त्रिपद \omterm{def:g11:quad:discriminant}{मूलों} के
कड़ाई से बीच में ऋणात्मक है: हल-समुच्चय $\intoo{-2}{5}$ है।

$2x^2 + x + 3$: $\Delta = 1 - 24 = -23 < 0$ और $a = 2 > 0$: सदा धनात्मक, इसलिए हल-समुच्चय $\R$ है।

$-x^2 + 6x - 9 = -(x-3)^2$: वह $\geq 0$ केवल वहाँ है जहाँ $(x-3)^2 \leq 0$ हो, अर्थात् $x = 3$ पर। हल-समुच्चय $\{3\}$
है।
\end{solution}

\begin{solution}{exo:g11:quad:4}
$x^2 - 7x + 6$: $1 - 7 + 6 = 0$, इसलिए $1$ एक \omterm{def:g11:quad:discriminant}{मूल} है; \omterm{def:g11:quad:discriminant}{मूलों} का गुणनफल $6$ है, इसलिए दूसरा \omterm{def:g11:quad:discriminant}{मूल}
$6$ है।

$2x^2 + 5x - 7$: $2 + 5 - 7 = 0$, इसलिए $1$ एक \omterm{def:g11:quad:discriminant}{मूल} है; गुणनफल $\frac{c}{a} = -\frac72$ है, इसलिए दूसरा \omterm{def:g11:quad:discriminant}{मूल} $-\frac72$ है।

$x^2 + 4x - 5$: $1 + 4 - 5 = 0$, इसलिए $1$ एक \omterm{def:g11:quad:discriminant}{मूल} है; गुणनफल $-5$ है, इसलिए दूसरा \omterm{def:g11:quad:discriminant}{मूल} $-5$ है।
\end{solution}

\begin{solution}{exo:g11:quad:5}
दोनों संख्याएँ $x^2 - 14x + 45 = 0$ (\cref{prop:g11:quad:vieta}) को हल करती हैं; $\Delta = 196 - 180 = 16$, \omterm{def:g11:quad:discriminant}{मूल} $\frac{14 \pm 4}{2} = 9$ और $5$।

आयत के लिए: आधे परिमाप से लंबाई $+$ चौड़ाई $= 17$ मिलता है, और क्षेत्रफल से
उनका गुणनफल $60$। वे $x^2 - 17x + 60 = 0$ को हल करती हैं; $\Delta = 289 - 240 = 49$, \omterm{def:g11:quad:discriminant}{मूल} $\frac{17 \pm 7}{2} = 12$ और $5$। आयत $12 \times 5$
है।
\end{solution}

\begin{solution}{exo:g11:quad:6}
$\Delta = (m-2)^2 - 4$। ठीक एक हल तब जब $\Delta = 0$ हो: $(m-2)^2 = 4$, इसलिए $m - 2 = \pm 2$, अर्थात् $m = 0$ या $m = 4$। कोई
हल नहीं तब जब $\Delta < 0$ हो: $(m-2)^2 < 4$, अर्थात् $-2 < m - 2 < 2$, यानी $0 < m < 4$।
\end{solution}

\begin{solution}{exo:g11:quad:7}
शीर्ष $t = -\frac{20}{2 \times (-5)} = 2$ पर है, और $h(2) = -20 + 40 + 1 = 21$: \omterm{def:g10:functions:extrema}{अधिकतम} ऊँचाई $21$ m, जो $2$ सेकंड बाद मिलती है।

$h(t) \geq 16$ का अर्थ है $-5t^2 + 20t - 15 \geq 0$, अर्थात् ($-5$ से भाग देने पर, जिससे असमिका पलट जाती
है) $t^2 - 4t + 3 \leq 0$। $t^2-4t+3$ के \omterm{def:g11:quad:discriminant}{मूल} $1$ और $3$ हैं ($1 - 4 + 3 = 0$, गुणनफल $3$), इसलिए $t^2 - 4t + 3 \leq 0$ ठीक
$\intcc{1}{3}$ पर है: गेंद $t = 1$ s और $t = 3$ s के बीच कम से कम $16$ m ऊँचाई पर रहती है।
\end{solution}

\begin{solution}{exo:g11:quad:8}
$X = x^2$ के साथ: $X^2 - 13X + 36 = 0$, $\Delta = 169 - 144 = 25$, इसलिए $X = \frac{13 \pm 5}{2} = 9$ या $4$। $x$ पर लौटिए: $x^2 = 9$ से
$x = \pm 3$ मिलता है, और $x^2 = 4$ से $x = \pm 2$। हल-समुच्चय $\{-3, -2, 2, 3\}$ है।
\end{solution}

\begin{solution}{exo:g11:quad:9}
शर्त $x^2 < x + 2$ है, अर्थात् $x^2 - x - 2 < 0$। चूँकि $x^2 - x - 2 = (x+1)(x-2)$ है (\omterm{def:g11:quad:discriminant}{मूल} $-1$ और $2$), इसलिए
असमिका \omterm{def:g11:quad:discriminant}{मूलों} के कड़ाई से बीच में सही है: $x \in \intoo{-1}{2}$। रेखाचित्र पर \omterm{def:g11:quad:parabola}{परवलय} $y = x^2$ ठीक
अपने दोनों प्रतिच्छेद बिंदुओं $(-1, 1)$ और $(2, 4)$ के बीच रेखा $y = x+2$ से नीचे डूब
जाता है।
\end{solution}

\begin{solution}{exo:g11:quad:10}
मान लीजिए $x > 0$ $x + \frac1x = \frac{13}{6}$ को संतुष्ट करता है। $6x > 0$ से गुणा करने पर: $6x^2 + 6 = 13x$,
अर्थात् $6x^2 - 13x + 6 = 0$। तब $\Delta = 169 - 144 = 25$ और $x = \frac{13 \pm 5}{12}$, जिससे $x = \frac32$ या $x = \frac23$ मिलता है। दोनों
धनात्मक हैं और एक-दूसरे के व्युत्क्रम हैं, जैसी अपेक्षा थी: संख्या $\frac32$
(या $\frac23$) है।
\end{solution}

\begin{solution}{exo:g11:quad:11}
\emph{1.} प्रतिच्छेद $x^2 = mx - 1$ के हलों के संगत हैं, अर्थात् $x^2 - mx + 1 = 0$ के, जिसका
\omterm{def:g11:quad:discriminant}{विविक्तकर} $\Delta = m^2 - 4$ है। $\abs{m} > 2$ पर दो बिंदु; $m = \pm 2$ पर ठीक एक; $-2 < m < 2$ पर कोई नहीं।

\emph{2.} $m = 2$ के लिए: $x^2 - 2x + 1 = (x-1)^2 = 0$, इसलिए अकेला स्पर्श बिंदु $(1, 1)$ है। $m = -2$ के
लिए: $(x+1)^2 = 0$, स्पर्श बिंदु $(-1, 1)$। दोनों स्थितियों में रेखा \omterm{def:g11:quad:parabola}{परवलय} से एक ही
बिंदु पर मिलती है और उसे पार नहीं करती --- वह वहाँ स्पर्श रेखा है, और सचमुच
उसकी \omterm{def:g10:reffunc:affine}{प्रवणता} $m = \pm 2$ $x^2$ के अवकलज $2x$ के $x = \pm 1$ पर मान के बराबर है (\cref{ch:g11:deriv})।
\end{solution}

\begin{solution}{pb:g11:quad:1}
\textbf{1.} $x^2 - 5x + 6$: $\Delta = 1$, \omterm{def:g11:quad:discriminant}{मूल} $2$ और $3$। $2x^2 + x - 6$: $\Delta = 1 + 48 = 49$, \omterm{def:g11:quad:discriminant}{मूल} $\frac{-1 \pm 7}{4}$: $\frac32$
और $-2$। $x^2 + x + 1$: $\Delta = -3 < 0$: कोई वास्तविक हल नहीं।

\textbf{2.} $\Delta = 16 + 48 = 64$: \omterm{def:g11:quad:discriminant}{मूल} $3$ और $-1$, इसलिए $2x^2 - 4x - 6 = 2(x - 3)(x + 1)$; \omterm{thm:g11:quad:canonical}{मानक रूप} $2(x - 1)^2 - 8$।
गुणनखंडित रूप \omterm{def:g11:quad:discriminant}{मूल} दिखाता है, \omterm{thm:g11:quad:canonical}{मानक रूप} शीर्ष $(1, -8)$, और प्रसारित रूप
$y$-अंतःखंड $-6$।

\textbf{3.} \omterm{def:g11:quad:discriminant}{मूल} $1$ और $3$, अग्र गुणांक ऋणात्मक: त्रिपद \omterm{def:g11:quad:discriminant}{मूलों} के
\emph{बीच} $\geq 0$ है: $x \in \intcc{1}{3}$।

\textbf{4.} संख्याएँ $x^2 - 7x + 12 = 0$ के \omterm{def:g11:quad:discriminant}{मूल} हैं: $3$ और $4$। यदि $2$ $x^2 - 5x + c$ का
एक \omterm{def:g11:quad:discriminant}{मूल} हो, तो दूसरा \omterm{def:g11:quad:discriminant}{मूल} $r$ $2 + r = 5$ (योग) को संतुष्ट करता है, इसलिए $r = 3$,
और $c = 2 \times 3 = 6$ (गुणनफल): वियेत उसे यूँ ही पढ़ लेते हैं।

\textbf{5.} $\Delta = m^2 - 36$: $m = \pm 6$ के लिए द्विगुण \omterm{def:g11:quad:discriminant}{मूल} (\omterm{def:g11:quad:discriminant}{मूल} $\mp 3$); $-6 < m < 6$ के लिए कोई
\omterm{def:g11:quad:discriminant}{मूल} नहीं; $\abs m > 6$ के लिए दो \omterm{def:g11:quad:discriminant}{मूल}।

\textbf{6.} $PF^2 = x^2 + \left(x^2 - \frac14\right)^2
= x^2 + x^4 - \frac{x^2}{2} + \frac{1}{16}
= x^4 + \frac{x^2}{2} + \frac{1}{16}
= \left(x^2 + \frac14\right)^2$। अतः $PF = x^2 + \frac14$ (दोनों गुणनखंड धनात्मक)।

\textbf{7.} $P(x, x^2)$ की क्षैतिज रेखा $y = -\frac14$ से दूरी $x^2 + \frac14$ है --- ठीक $PF$।
\omterm{def:g11:quad:parabola}{परवलय} का हर बिंदु \omterm{pb:g11:quad:1}{नाभि} और \omterm{pb:g11:quad:1}{नियता} से समान दूरी पर है।

\textbf{8.} $PF^2 = x^2 + \left(y - \frac14\right)^2$ और $\operatorname{dist}^2 = \left(y + \frac14\right)^2$। समता से $x^2 + y^2 - \frac y2 + \frac{1}{16} = y^2 + \frac y2 +
\frac1{16}$ मिलता है, इसलिए $x^2 = y$: बिंदुपथ
ठीक वही \omterm{def:g11:quad:parabola}{परवलय} है। एक बिंदु और एक रेखा वक्र को जन्म देते हैं, जैसे एक बिंदु
और एक दूरी ने वृत्त को दिया था।

\textbf{9.} $y = \frac{x^2}{4f}$ के लिए $F(0, f)$ के साथ वही प्रसार $PF^2 = x^2 + \left(\frac{x^2}{4f} - f\right)^2
= \left(\frac{x^2}{4f} + f\right)^2$ देता है: $F$ से
दूरी $y = -f$ से दूरी के बराबर। तश्तरी: $9 = \frac{30^2}{4f}$ से $4f = 100$, $f = 25$ cm मिलता है:
अभिग्राही शीर्ष से चौथाई मीटर ऊपर लटकता है --- $9$ cm के कटोरे से बाहर,
जैसा असली तश्तरियों में दिखता है।

\textbf{10.} अक्ष के समांतर आता हुआ कोई तरंगाग्र \omterm{pb:g11:quad:1}{नियता} के स्तर तक पहुँचने में
बराबर दूरियाँ तय कर चुका होता है; समान-दूरी ``\omterm{pb:g11:quad:1}{नियता} से बराबर दूरी'' को
``$F$ से बराबर दूरी'' में बदल देती है: सभी परावर्तित संकेत $F$ पर
\emph{एक ताल में} पहुँचते हैं और एक-दूसरे को बल देते हैं --- सुनना वहीं
चाहिए। ईमानदार उपपत्ति के लिए हर बिंदु पर स्पर्श रेखा की \omterm{def:g10:reffunc:affine}{प्रवणता} चाहिए ---
\cref{ch:g11:deriv} का अवकलज, जिसकी झलक \cref{exo:g11:quad:11} में है।

\textbf{11.} $y = x\left(1 - \frac{x}{20}\right)$: $x = 20$ m पर गिरती है। शीर्ष ठीक बीच में: $x = 10$, ऊँचाई
$y(10) = 10 - 5 = 5$ m।

\textbf{12.} $y(4) = 4 - \frac{16}{20} = 3.2$ m: $3$ m की दीवार आराम से पार कर जाती है, पर $4$ m की
दीवार से टकरा जाती है।

\textbf{13.} $y(10) = 10 - \frac{100}{40} = 7.5$ m $> 7$: ट्रक केंद्र से $10$ m पर निकल जाता है। $y(15) = 10 - \frac{225}{40}
= 4.375$
m: $15$ m पर नहीं।

\textbf{14.} $R(p) = -2p^2 + 120p$: शीर्ष $p = \frac{120}{4} = 30$ यूरो पर, आय $R(30) = 30 \times 60 = 1\,800$ यूरो।

\textbf{15.} सबसे नीचे वाले बिंदु से $y(320) = 143 \times \left(\frac{320}{640}\right)^2
= \frac{143}{4} \approx 36$ m ऊपर।

\textbf{16.} $x^2 = mx - 1$ से $x^2 - mx + 1 = 0$, $\Delta = m^2 - 4$ मिलता है: $\abs m > 2$ के लिए दो बिंदु, $\abs m < 2$
के लिए कोई नहीं, और $m = \pm 2$ के लिए स्पर्श जिसमें स्पर्श बिंदु $x = \frac m2 = \pm 1$ है: स्पर्श
रेखाएँ $(1, 1)$ पर $y = 2x - 1$ और $(-1, 1)$ पर $y = -2x - 1$।

\textbf{17.} $x^2 = mx + 1$ से हर $m$ के लिए $\Delta = m^2 + 4 > 0$ मिलता है: सदा दो प्रतिच्छेद।
किसी भीतरी बिंदु से हर रेखा वक्र को दो बार काटती है --- स्पर्श रेखाएँ केवल
बाहर से खींची जाती हैं, ठीक जैसे वृत्तों के लिए।

\textbf{18.} $y = \frac14$ $y = x^2$ से $x = \pm\frac12$ पर मिलता है: जीवा की लंबाई $1$। $y = \frac{x^2}{4f}$
के लिए: $y = f$ से $x^2 = 4f^2$, $x = \pm 2f$ मिलता है: लंबाई $4f$ --- नाभीय चौड़ाई नाभीय
दूरी की चौगुनी है (तश्तरी बनाने वाले इसे नाभिलंब कहते हैं)।

\textbf{19.} $y = x^2$ का कोई बिंदु $(a, a^2)$ है; उसका \omterm{def:g10:functions:function}{प्रतिबिंब} $(X, Y) = (ka, ka^2)$ है, और $\frac{X^2}{k} = \frac{k^2a^2}{k} =
ka^2 = Y$:
प्रतिबिंब-वक्र ठीक $y = \frac{x^2}{k}$ है। $k = 4f$ चुनकर हर \omterm{def:g11:quad:parabola}{परवलय} तक पहुँचा जा सकता है: वे
सब एक ही मूल वक्र के बड़े किए हुए रूप हैं। (वृत्त भी सब समरूप हैं --- पर
दीर्घवृत्त नहीं: चपटाई मायने रखती है, ज़ूम नहीं।)

\textbf{20.} तीन पढ़े जा सकने वाले वेशों वाला एक \omterm{def:g10:algebra:equation}{समीकरण}; एक बिंदु और एक रेखा
से समान दूरी का बिंदुपथ; वह परावर्तक जो समांतर किरणें अपनी \omterm{pb:g11:quad:1}{नाभि} पर इकट्ठी कर
लेता है और वह रस्सा जो समान रूप से बँटा पाट उठाए रहता है; हर फेंकी गई गेंद का
पथ; और, बड़ा-छोटा करने की छूट देने पर, एक अकेला वक्र --- एक ही \omterm{def:g11:quad:parabola}{परवलय}, हर
जगह।
\end{solution}
