\begin{solution}{exo:g11:vect:6}

मिलती हैं (जाँच: $4 + 3 - 7 = 0$)।

दूसरा युग्म: $ab' - a'b = 1 \times 4 - (-2) \times (-2) = 0$, इसलिए रेखाएँ \omterm{def:g11:seq:arithmetic}{समांतर} हैं; $e$ को $-2$ से गुणा करने पर
$-2x + 4y - 6 = 0$ मिलता है, जो $e'$ से भिन्न है ($-6 \neq 1$): कड़ाई से \omterm{def:g11:seq:arithmetic}{समांतर}, कोई प्रतिच्छेद
नहीं।
\end{solution}

\begin{solution}{exo:g11:vect:7}
$d: 2x - y + 3 = 0$ के लिए: दिशा $(-b, a) = (1, 2)$, और बिंदु $(0, 3)$ $d$ पर है, इसलिए $d = \{(t,\ 3 + 2t) : t \in \R\}$।

$(1, -3)$ से होकर दिशा $(2, 1)$ वाली प्राचलिक रेखा के लिए: $a = 1$, $b = -2$, इसलिए $x - 2y + c = 0$;
$(1, -3)$ रखने पर: $1 + 6 + c = 0$, $c = -7$। \omterm{def:g10:algebra:equation}{समीकरण}: $x - 2y - 7 = 0$ ($t = 1$ से जाँच, बिंदु $(3, -2)$: $3 + 4 - 7 = 0$)।
\end{solution}

\begin{solution}{exo:g11:vect:8}

$mx + 2y - 5 = 0$ का $d$ के \omterm{def:g11:seq:arithmetic}{समांतर} होना $ab' - a'b = 4 \times 2 - m \times (-3) = 8 + 3m = 0$ माँगता है, इसलिए $m = -\frac83$।
\end{solution}

\begin{solution}{exo:g11:vect:10}

\end{solution}

\begin{solution}{exo:g11:vect:11}
\emph{1.} $I = \left(\frac12, 0\right)$। तब
$\vect{DI} = \left(\frac12 - 0,\ 0 - 1\right) = \left(\frac12, -1\right)$
और $\vect{DJ} = \frac23\vect{DI} = \left(\frac13, -\frac23\right)$, इसलिए
\[
J = D + \vect{DJ} = \left(\frac13,\ \frac13\right).
\]

\emph{2.} $\vect{AJ}\left(\frac13, \frac13\right)$ और $\vect{AC}\,(1, 1)$: $\det = \frac13 \times 1 - 1 \times \frac13 = 0$, इसलिए $A$, $J$, $C$ संरेख हैं ---
वस्तुतः $\vect{AJ} = \frac13\vect{AC}$: रेखा $(DI)$ विकर्ण $(AC)$ को $A$ से ठीक एक-तिहाई लंबाई पर
काटती है।
\end{solution}

\begin{solution}{pb:g11:vect:1}
\textbf{1.} $x = 1 + 3t$, $y = 2 - t$। दूसरे से $t = 2 - y$; प्रतिस्थापन करने पर: $x = 1 + 3(2 - y) = 7 - 3y$:
कार्तीय \omterm{def:g10:algebra:equation}{समीकरण} $x + 3y - 7 = 0$।

\textbf{2.} $ax + by + c = 0$ के लिए एक \omterm{def:g11:vect:direction}{दिक् सदिश} $(-b, a)$ है: यहाँ $(5, 2)$। एक बिंदु: $y = 1$
से $x = 1$ मिलता है: $(1, 1)$। प्राचलिक: $x = 1 + 5t$, $y = 1 + 2t$।

\textbf{3.} $2x - 5y + 3 = 0$ में $x = 7 - 3y$ से: $14 - 6y - 5y + 3 = 0$, इसलिए $y = \frac{17}{11}$ और $x = 7 - \frac{51}{11} = \frac{26}{11}$: बिंदु $\left(\frac{26}{11}, \frac{17}{11}\right)$।

\textbf{4.} दिशाएँ $(5, 2)$ और $(3, -1)$: $\det = 5 \times (-1) - 2 \times 3 = -11 \neq 0$: प्रतिच्छेदी, जैसा मिला था। दूसरे
युग्म के लिए: दिशाओं $(5, 2)$ और $(10, 4)$ का $\det = 0$ है (\omterm{def:g11:seq:arithmetic}{समांतर}), और $2x - 5y + 3 = 0$ को दुगुना
करने पर $-4x + 10y - 6 = 0 \neq -4x + 10y + 1
= 0$ मिलता है: कड़ाई से \omterm{def:g11:seq:arithmetic}{समांतर}।

\textbf{5.} $\vect{AB}(3, 2)$, $\vect{AC}(9, 6)$: $\det = 18 - 18 = 0$: संरेख।

\textbf{6.} $P$: $x = 2t$ से $y = 3 + t$: $y = 3 +
\frac x2$। $Q$: $x = 10 - t$ से: $t = 10 - x$, इसलिए
$y = 2(10 - x) - 1 = 19 - 2x$। कटान: $3 + \frac x2 = 19 - 2x$ से $x = 6.4$, $y = 6.2$ मिलता है: रास्ते $(6.4,\ 6.2)$ पर कटते हैं।

\textbf{7.} $P$ वहाँ तब गुज़रता है जब $2t = 6.4$ हो: $t = 3.2$ घंटे। $Q$ तब जब
$10 - t = 6.4$ हो: $t = 3.6$ घंटे। चौबीस मिनट का अंतर: कोई टक्कर नहीं। चेतावनी: नक्शा
\emph{रास्ते} दिखाता है; केवल प्राचलिक घड़ियाँ बताती हैं कि दो गतियाँ एक ही
क्षण एक ही बिंदु पर हैं या नहीं।

\textbf{8.} $D^2(t) = (3t - 10)^2 + (4 - t)^2 =
10t^2 - 68t + 116$: शीर्ष $t = \frac{68}{20} = 3.4$ घंटे पर, जहाँ $D^2 = 0.4$: निकटतम पहुँच $t = 3.4$ पर
$D = \sqrt{0.4} \approx 0.63$ मील।

\textbf{9.} उल्लंघन हुआ: $0.63 < 1$। $D^2(t) < 1$ का अर्थ है $10t^2 - 68t + 115 < 0$: $\Delta = 4624 - 4600 = 24$, \omterm{def:g11:quad:discriminant}{मूल} $\frac{68 \pm \sqrt{24}}{20} = 3.4 \pm 0.245$:
जहाज़ $t \approx 3.16$ से $t \approx 3.64$ तक बहुत पास रहते हैं --- लगभग $29$ मिनट का उल्लंघन।
उनमें से एक को रास्ता बदलना पड़ेगा।

\textbf{10.} हर जहाज़ का हर \omterm{def:g10:coordgeom:system}{निर्देशांक} $t$ का \emph{एकघात} \omterm{def:g11:func:function}{फलन} है, इसलिए
\omterm{def:g10:coordgeom:system}{निर्देशांकों} का हर अंतर भी एकघात है, और $D^2$ --- दो वर्ग किए गए \omterm{def:g10:reffunc:affine}{एकघात फलनों}
का योग --- एक द्विघात है (जिसका अग्र गुणांक अऋणात्मक है)। अतः निकटतम पहुँच
सदा किसी शीर्ष से ही पढ़ी जाती है।

\textbf{11.} $t = 3$ पर, अर्थात् $(6, 6)$ पर, $P$ से मिलना: $(3a, 3b) = (6, 6)$ से $(a, b) = (2, 2)$
मिलता है, चाल $\sqrt{8} \approx 2.8$ नॉट।

\textbf{12.} $D(0, 1)$, $I\left(\frac12, 0\right)$: दिशा $\left(\frac12, -1\right)$, अर्थात् $(1, -2)$: \omterm{def:g10:algebra:equation}{समीकरण} $2x + y = 1$।

\textbf{13.} $(AC)$: $y = x$। फिर $2x + x = 1$: $x = \frac13$: बिंदु $\left(\frac13, \frac13\right)$, जो $A$ से $C$
तक एक-तिहाई दूरी पर है।

\textbf{14.} $B(1, 0)$, $K\left(\frac12, 1\right)$: दिशा $\left(-\frac12, 1\right)$, \omterm{def:g10:algebra:equation}{समीकरण} $2x + y = 2$। $y = x$ के साथ: $x = \frac23$:
बिंदु $\left(\frac23, \frac23\right)$। दोनों चेवियन $(AC)$ को उसके दोनों त्रिभाजन बिंदुओं पर काटते हैं:
प्रमेय सिद्ध, दो बार तीन-तीन पंक्तियों के अंकगणित से।

\textbf{15.} किसी भी \omterm{def:g11:seq:arithmetic}{समांतर} चतुर्भुज को इकाई वर्ग पर ले जाया जा सकता है ---
इसके लिए $A$ को मूल बिंदु और $\vect{AB}$, $\vect{AD}$ को दोनों अक्षों के मात्रक चुन
लीजिए। वह चुनाव उन सब बातों को बनाए रखता है जिनकी प्रमेय बात करती है ---
\omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु}, संरेखण, और किसी दी हुई रेखा पर अनुपात (जो सब \omterm{def:g10:vectors:vector}{सदिशों} और अदिशों से
व्यक्त होते हैं) --- इसलिए वर्ग में कथन सिद्ध कर देना उसे हर \omterm{def:g11:seq:arithmetic}{समांतर} चतुर्भुज
के लिए सिद्ध कर देता है। (वह लंबाइयाँ या कोण \emph{नहीं} ढोता: उन्हें यह
ढाँचा बिगाड़ देता है।)

\textbf{16.} $E\left(1, \frac12\right)$: रेखा $(AE)$: $y = \frac x2$। विकर्ण $(DB)$: $(0,1)$ से $(1,0)$ तक:
$y = 1 - x$। प्रतिच्छेद: $\frac x2 = 1 - x$: $x = \frac23$: बिंदु $\left(\frac23, \frac13\right)$ --- फिर एक त्रिभाजन बिंदु, अब
दूसरे विकर्ण का। \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} वाले चेवियन तिहाइयों के प्रेमी हैं।

\textbf{17.} पहली दो: $x + y = 4$ और $2x - y = 2$ से $3x = 6$ मिलता है: $(2, 2)$। तीसरी: $2 - 4 = -2$:
संतुष्ट। $(2, 2)$ पर संगामी।

\textbf{18.} $mx - y + 2 - 3m = m(x - 3) + (2 - y) = 0$: \omterm{def:g10:algebra:equation}{समीकरण} \emph{हर} $m$ के लिए सही रहे, इसके लिए $x = 3$,
$y = 2$ लीजिए --- और सचमुच हर $d_m$ में $(3, 2)$ आता है। एक ही बिंदु से गुज़रती
रेखाओं का एक गुच्छा, हर \omterm{def:g10:reffunc:affine}{प्रवणता} $m$ के लिए एक रेखा।

\textbf{19.} $y = 2x + 7$ के \omterm{def:g11:seq:arithmetic}{समांतर}: \omterm{def:g10:reffunc:affine}{प्रवणता} $m = 2$, रेखा $2x - y - 4 = 0$। मूल बिंदु से होकर:
$2 - 3m = 0$: $m = \frac23$।

\textbf{20.} कार्तीय: कोई बिंदु रखिए, सदस्यता का उत्तर मिल गया। संक्षिप्त:
\omterm{def:g10:reffunc:affine}{प्रवणता} और अंतःखंड दिखते हैं, \omterm{def:g10:functions:graph}{आलेख} तुरंत। प्राचलिक: घड़ी भीतर ही बनी हुई ---
टक्करें, निकटतम पहुँच, अवरोधन। \omterm{def:g11:vect:det}{सारणिक}: एक तिर्यक गुणन, और समांतरता का फ़ैसला।
और चुना हुआ ढाँचा: \omterm{def:g11:seq:arithmetic}{समांतर} चतुर्भुज इकाई वर्ग बन जाता है, और एक चिरपरिचित
त्रिभाजन प्रमेय रेखाओं के तीन प्रतिच्छेदों में बदल जाती है।
\end{solution}
