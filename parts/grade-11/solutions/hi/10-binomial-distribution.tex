\section*{अध्याय \ref{ch:g11:binom} --- द्विपद बंटन}

\begin{solution}{exo:g11:binom:1}
पंक्तियाँ $6$ और $7$:
\[
1,\ 6,\ 15,\ 20,\ 15,\ 6,\ 1
\qquad\text{और}\qquad
1,\ 7,\ 21,\ 35,\ 35,\ 21,\ 7,\ 1 .
\]
अतः $\binom62 = 15$, $\binom73 = 35$, $\binom74 = 35$ (सममिति $\binom73 = \binom74$ सफलताओं और विफलताओं की अदला-बदली दर्शाती है)।
\end{solution}

\begin{solution}{exo:g11:binom:2}
परीक्षणों की नियत संख्या ($4$ फेंक), हर फेंक के दो परिणाम (छक्का या नहीं,
$p = \frac16$), और फेंकों का स्वतंत्र होना: $X \sim \mathcal B(4, \frac16)$।
\[
\P(X=0) = \left(\frac56\right)^4 = \frac{625}{1296} \approx 0.48,
\qquad
\P(X=1) = 4 \times \frac16\left(\frac56\right)^3 = \frac{500}{1296}
\approx 0.39,
\]
\[
\P(X \geq 2) = 1 - \frac{625 + 500}{1296} = \frac{171}{1296}
\approx 0.13 .
\]
\end{solution}

\begin{solution}{exo:g11:binom:3}
\emph{1.} द्विपद $\mathcal B(20, \frac12)$: नियत $n$, वही $p$, और स्वतंत्र उछाल।

\emph{2.} द्विपद नहीं: पत्ते \emph{बिना वापसी} के बाँटे जाते हैं, इसलिए
इक्के की \omterm{def:g10:proba:distribution}{प्रायिकता} पत्ते-दर-पत्ते बदलती है और निकाले स्वतंत्र नहीं हैं।

\emph{3.} बताई गई स्वतंत्रता की मान्यता से द्विपद $\mathcal B(7, 0.3)$।
\end{solution}

\begin{solution}{exo:g11:binom:4}
$\E(X) = 50 \times 0.2 = 10$; $\V(X) = 50 \times 0.2 \times 0.8 = 8$; $\sigma(X) = 2\sqrt2 \approx 2.83$।
\end{solution}

\begin{solution}{exo:g11:binom:5}
$X \sim \mathcal B(6, 0.7)$।
\[
\P(X = 4) = \binom64 (0.7)^4 (0.3)^2 = 15 \times 0.2401 \times 0.09
\approx 0.324 .
\]
\[
\P(X \geq 5) = \binom65 (0.7)^5(0.3) + (0.7)^6
= 6 \times 0.16807 \times 0.3 + 0.117649 \approx 0.420 .
\]
\end{solution}

\begin{solution}{exo:g11:binom:6}
$X \sim \mathcal B\left(8, \frac12\right)$; हर पथ की \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac{1}{256}$ है, इसलिए
\[
\P(X \geq 6) = \frac{\binom86 + \binom87 + \binom88}{256}
= \frac{28 + 8 + 1}{256} = \frac{37}{256} \approx 0.14 .
\]
अनुमान लगाने से लगभग सात में एक बार उत्तीर्ण हुआ जाता है।
\end{solution}

\begin{solution}{exo:g11:binom:7}
``हर तरह की कम से कम एक'' का \omterm{def:g10:proba:operations}{पूरक} ``पाँचों एक जैसे'' है: सभी A या सभी B, और
हर एक की \omterm{def:g10:proba:distribution}{प्रायिकता} $\left(\frac12\right)^5 =
\frac1{32}$। अतः
\[
\P(\text{प्रत्येक प्रकार का एक}) = 1 - \frac{2}{32} = \frac{15}{16} .
\]
\end{solution}

\begin{solution}{exo:g11:binom:8}
$\P(X = 3) = p^3$ और $\P(X \geq 1) = 1 - (1-p)^3$। $p^3 = \frac{27}{64} = \left(\frac34\right)^3$ हल करने पर $p = \frac34$ मिलता है (घन \omterm{def:g11:func:function}{फलन} निरंतर \omterm{def:g11:func:monotone}{वर्धमान} है, \cref{ch:g11:func},
इसलिए हल अद्वितीय है)।
\end{solution}

\begin{solution}{exo:g11:binom:9}
$\P(\text{कम से कम एक चित}) = 1 - \left(\frac12\right)^n$, इसलिए शर्त $\left(\frac12\right)^n < 0.01$ है, अर्थात् $2^n > 100$। चूँकि $2^6 = 64$ और $2^7 = 128$ हैं: $n = 7$ उछालों
से आगे।
\end{solution}

\begin{solution}{exo:g11:binom:10}
पंक्ति $n$ का योग $n$-परीक्षण वाले \omterm{met:g10:proba:tree}{वृक्ष} के सभी पथों को उनकी सफलताओं की
संख्या के अनुसार छाँटकर गिनता है। पर \omterm{met:g10:proba:tree}{वृक्ष} हर परीक्षण पर अपने पथ दुगुने कर
लेता है (हर पथ S और F में बँट जाता है), इसलिए उसमें कुल $2^n$ पथ हैं। अतः
$\sum_{k} \binom nk = 2^n$। वैकल्पिक रूप से, आगमन द्वारा: पंक्ति $0$ का योग $1 = 2^0$ है, और पास्कल का
नियम पंक्ति $n$ की हर प्रविष्टि को पंक्ति $n+1$ की ठीक दो प्रविष्टियों में
योगदान करा देता है, इसलिए पंक्ति-योग दुगुने होते जाते हैं।
\end{solution}

\begin{solution}{exo:g11:binom:11}
\emph{1.} दावे के अधीन $X \sim \mathcal B(10, 0.6)$। पहले पदों को जोड़ने पर:
\begin{align*}
\P(X \leq 3)
&= (0.4)^{10} + 10(0.6)(0.4)^9 + 45(0.6)^2(0.4)^8
+ 120(0.6)^3(0.4)^7\\
&\approx 0.0001 + 0.0016 + 0.0106 + 0.0425 = 0.0548 .
\end{align*}

\emph{2.} प्रेक्षित परिणाम जितने या उससे भी अधिक चरम परिणाम ($3$ या उससे कम
समर्थन) की \omterm{def:g10:proba:distribution}{प्रायिकता} लगभग $5.5\%$ है --- यानी $5\%$ की सीमा से ज़रा \emph{ऊपर}।
नियम को कड़ाई से लगाएँ तो प्रेक्षण (मुश्किल से ही सही) दावे के अनुकूल है और
उसे अस्वीकार नहीं किया जाता। यह उदाहरण दिखाता है कि सीमा-रेखा पर खड़े निर्णय
सीमा के चुनाव के प्रति कितने संवेदनशील होते हैं: $6\%$ की परिपाटी के साथ
निष्कर्ष पलट जाता।
\end{solution}

\begin{solution}{pb:g11:binom:1}
\textbf{1.} पंक्तियाँ: $1$; $1\,1$; $1\,2\,1$; $1\,3\,3\,1$; $1\,4\,6\,4\,1$; $1\,5\,10\,10\,5\,1$; $1\,6\,15\,20\,15\,6\,1$। सममिति: यह
चुनना कि कौन-सी $k$ वस्तुएँ ली जाएँ, ठीक वही काम है जो यह चुनना कि कौन-सी
$n - k$ पीछे छोड़ी जाएँ।

\textbf{2.} $1 + 4 + 6 + 4 + 1 = 16 = 2^4$; $1 + 5 + 10 + 10 + 5 + 1 = 32 = 2^5$। उपपत्ति: $\binom nk$ किसी $n$-अवयवी समुच्चय के हर आकार के
उपसमुच्चय गिनते हैं, और कुल मिलाकर उपसमुच्चय $2^n$ हैं (हर अवयव स्वतंत्र रूप
से भीतर या बाहर)।

\textbf{3.} $n + 1$ लोगों में से चुनी गई $k$ की समितियाँ, जिनमें एक ज़ो है: बिना
ज़ो वाली $\binom nk$ हैं (बाक़ी में से सारी $k$ चुनिए); ज़ो वाली $\binom{n}{k-1}$ हैं (उसके
$k - 1$ सहयोगी चुनिए)। कुल: $\binom nk + \binom{n}{k-1}$।

\textbf{4.} त्रिभुज की पंक्ति $7$: $1\,7\,21\,35\,\dots$: $35$। सूत्र से: $\frac{7 \times 6 \times 5}{3 \times 2 \times 1} =
35$।

\textbf{5.} $1 + 3 + 6 + 10 = 20 = \binom63$। झरना: $\binom63 = \binom52 + \binom53 = \binom52 + \binom42 +
\binom43 = \binom52 + \binom42 + \binom32 + \binom33$ --- पास्कल के नियम का हर प्रयोग सीढ़ी का एक पद
छील देता है।

\textbf{6.} उछालों की नियत संख्या $n$; हर उछाल वही $p = \frac12$ वाला एक स्वतंत्र
\omterm{def:g11:binom:bernoulli}{बर्नूली परीक्षण}; और ख़ाने की संख्या सफलताओं (दाईं ओर की उछालों) को गिनती है:
\cref{met:g11:binom:recognize} के तीनों ख़ाने पर निशान लग गया: $\mathcal B\!\left(n, \frac12\right)$।

\textbf{7.} ख़ानों $0, \dots, 4$ की \omterm{def:g10:proba:distribution}{प्रायिकताएँ} $\frac{1}{16}, \frac{4}{16},
\frac{6}{16}, \frac{4}{16}, \frac{1}{16}$ हैं: बीच वाला ख़ाना $2$ सबसे
भीड़ भरा है।

\textbf{8.} प्रत्याशित गिनतियाँ $= 1024 \times
\binom{10}{k}/1024 = \binom{10}{k}$: बीच वाला ख़ाना $\binom{10}{5} = 252$ गेंदें; ख़ाना
$7$: $\binom{10}{7} = 120$; और हर किनारे वाला ख़ाना: $1$ गेंद। ऊँचा केंद्र जो सममित रूप
से गिरकर फुसफुसाहट-भर पतले किनारों तक आता है: वही घंटी।

\textbf{9.} $\E = np = 5$; $V = np(1 - p) = 2.5$; $\sigma \approx 1.58$। $2\sigma$ के भीतर: ख़ाने $2$ से $8$ तक
गेंदों का \[
\frac{45 + 120 + 210 + 252 + 210 + 120 + 45}{1024}
= \frac{1002}{1024} \approx 98\,\%
\] उठाए हैं --- चेबिशेव की सर्वकाम $75\,\%$ (\cref{pb:g11:stat:1}) से कहीं बेहतर:
घंटी-आकार कसकर सिमटते हैं।

\textbf{10.} $\E = 6$, $V = 10 \times 0.6 \times 0.4 = 2.4$, $\sigma \approx 1.55$: ढेर अपनी घंटी-आकृति बनाए रखता है पर अपनी
चोटी सरकाकर ख़ाने $6$ पर ले जाता है --- झुका हुआ पटल दिखाई देने वाला
भारित सिक्का है।

\textbf{11.} ख़ाने की संख्या अनेक छोटे, स्वतंत्र और एक ही आकार के संयोग-धक्कों
का \emph{योग} है --- और ऐसे योग सदा अपने आप घंटी में सज जाते हैं: अधिकांश
धक्के आपस में कट जाते हैं, और चरम तक पहुँचने के लिए सबकी सहमति चाहिए।
लंबाइयाँ, मापन की त्रुटियाँ और अनगिनत प्राकृतिक राशियाँ भी इसी तरह अनेक छोटे
स्वतंत्र प्रभावों के योग हैं, और इसीलिए वही आकृति हर जगह दिखती है; इसे
प्रमाणित करने वाली प्रमेय केंद्रीय सीमा प्रमेय है।

\textbf{12.} सफ़ाया: एक ही टीम सारे $4$ जीत ले: $2 \times \left(\frac12\right)^4 = \frac18$।

\textbf{13.} सात मुक़ाबलों के लिए छह के बाद $3$--$3$ चाहिए: $\binom63 \left(\frac12\right)^6 = \frac{20}{64} =
\frac{5}{16}$।

\textbf{14.} $5$ में समाप्ति: विजेता पाँचवाँ मुक़ाबला और पहले $4$ में से
$3$ जीतता है: $2 \times \binom43 \left(\frac12\right)^5 = \frac14$। $6$ में समाप्ति: $2 \times \binom53 \left(\frac12\right)^6 = \frac{5}{16}$। $4, 5, 6, 7$ पर \omterm{def:g11:prob:rv}{बंटन}: $\frac18, \frac14, \frac{5}{16}, \frac{5}{16}$ (योग
$1$)। प्रत्याशित लंबाई: $4 \cdot \frac18 + 5 \cdot \frac14 + 6 \cdot
\frac{5}{16} + 7 \cdot \frac{5}{16} = 5.8125$ मुक़ाबले। छह और सात मुक़ाबलों वाली शृंखलाएँ
सबसे संभावित हैं --- नाटक इस प्रारूप में ही गूँथा हुआ है।

\textbf{15.} $4$ में जीत: $0.6^4 = 0.1296$; $5$ में: $\binom43\,0.6^3 \times 0.4 \times 0.6 = 0.2074$; $6$ में: $\binom53\,0.6^3 \times 0.4^2 \times 0.6 = 0.2074$; $7$
में: $\binom63\,0.6^3 \times 0.4^3 \times 0.6 = 0.1659$। कुल: लगभग $0.710$: प्रति मुक़ाबले $60\,\%$ वाली टीम $71\,\%$ शृंखलाएँ
जीतती है --- शृंखला बढ़त को बढ़ा देती है।

\textbf{16.} अकेला अंतिम मुक़ाबला: $60\,\%$। तीन में से दो: $p^2 + 2p^2 q = 0.36 + 0.288 = 0.648$। सीढ़ी $60\,\% \to 65\,\% \to 71\,\%$
लंबाई के साथ आगे बढ़ती जाती है: अधिक मुक़ाबले संयोग को औसत में मिला देते हैं
(छोटे पैमाने पर बृहत् संख्याओं का नियम), इसलिए लंबे फ़ाइनल कौशल को ताज
पहनाते हैं --- और लीग यही बेचती हैं।

\textbf{17.} न्यायसंगत सिक्का: $\E = 50$, $\sigma = \sqrt{25} = 5$; $z = \frac{62 - 50}{5} = 2.4$: $2\sigma$ की परिपाटी से आगे
--- सिक्का जाँच का हक़दार है।

\textbf{18.} $\P(X = 0) = 0.98^{50} \approx 0.364$; $\P(X = 1) = 50 \times 0.02 \times 0.98^{49} \approx 0.372$; $\P(X = 2) = \binom{50}{2} 0.02^2 \times 0.98^{48} \approx
0.186$। इसलिए $\P(X \geq 3) \approx 1 - 0.922 = 0.078$: लगभग $7.8\,\%$ --- $5\,\%$ की सीमा
से ऊपर, इसलिए दावे को अस्वीकार करने के लिए (अभी) पर्याप्त नहीं; पर एक और
ख़राब बैच कहानी बदल देगा।

\textbf{19.} $\P(\text{कम से कम एक जीत}) = 1 - 0.999^{1000}
\approx 1 - 0.368 = 0.632$: एक हज़ार में एक की दर से एक हज़ार टिकट निश्चितता नहीं,
$63\,\%$ देते हैं। बार-बार लौटता $0.368$ भेस बदले $\frac1e$ है --- अचर $e$ कक्षा 12
में औपचारिक प्रवेश करता है।

\textbf{20.} पहचान: नियत $n$, स्वतंत्रता, अचर $p$ --- तभी और केवल तभी,
द्विपद। सारणी: \omterm{prop:g11:binom:pascal}{पास्कल त्रिभुज}, पंक्ति $n$। आकृति: घंटी, $p = \frac12$ के लिए सममित
और अन्यथा खिसकी हुई। दिशा-सूई: केंद्र $np$, प्रसार $\sqrt{np(1-p)}$ --- निर्णयों के
\omterm{pb:g11:stat:1}{z-मान}। उत्तराधिकारी: वह चिकना घंटी-वक्र जिसके पास ढेर पहुँचते हैं, और बृहत्
संख्याओं का नियम, जो बताता है कि बड़े पटल कभी झूठ क्यों नहीं बोलते।
\end{solution}
