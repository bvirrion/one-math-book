\chapter{Vecteurs et droites dans le plan}\label{ch:g11:vect}

Les \omterm{def:g11:vect:vector}{vecteurs} encodent des déplacements~; les droites sont leurs
trajectoires. Le nouvel outil clé de ce chapitre est le
\emph{\omterm{def:g11:vect:det}{déterminant}}, un seul nombre qui décide si deux \omterm{def:g11:vect:vector}{vecteurs} sont
\omterm{def:g11:vect:collinear}{colinéaires} --- et, de là, produit des \omterm{def:g10:algebra:equation}{équations} de droites et tranche les
questions d'alignement et de parallélisme par pure computation. Les mêmes
idées s'étendent à l'espace dans le \cref{ch:g12:space}.

\section{Vecteurs~: un rappel}

\begin{definition}[Vecteur, coordonnées]\label{def:g11:vect:vector}
Un \emph{vecteur}\index{vecteur} $\vec u$ représente un déplacement~:
toutes les flèches de même direction, même sens et même longueur
représentent le même vecteur. Dans un \omterm{def:g10:coordgeom:system}{repère}, $\vec u\,(x, y)$ enregistre
le déplacement ($x$ horizontalement, $y$ verticalement)~; pour des points
$A(x_A, y_A)$ et $B(x_B, y_B)$,
\[
\vect{AB}\,\bigl(x_B - x_A,\ y_B - y_A\bigr).
\]
Les vecteurs s'additionnent coordonnée par coordonnée, et $\lambda\vec u$
a pour \omterm{def:g10:coordgeom:system}{coordonnées} $(\lambda x, \lambda y)$.
\end{definition}

\begin{proposition}[Milieu]\label{prop:g11:vect:midpoint}
Le \omterm{prop:g10:coordgeom:midpoint}{milieu} $M$ du segment $[AB]$ a pour \omterm{def:g10:coordgeom:system}{coordonnées}
$\left(\frac{x_A + x_B}{2},\ \frac{y_A + y_B}{2}\right)$.
\end{proposition}

\begin{proof}
$M$ est caractérisé par $\vect{AM} = \frac12\vect{AB}$~: coordonnée par
coordonnée, $x_M - x_A = \frac12(x_B - x_A)$, donc
$x_M = \frac{x_A + x_B}{2}$, et de même pour $y_M$.
\end{proof}

\section{Colinéarité et déterminant}

\begin{definition}[Vecteurs colinéaires]\label{def:g11:vect:collinear}
Deux \omterm{def:g11:vect:vector}{vecteurs} sont \emph{colinéaires}\index{vecteurs colinéaires} si l'un
est un multiple scalaire de l'autre~: $\vec v = \lambda \vec u$ pour un
certain $\lambda \in \R$ (ou $\vec u = \vec 0$). Géométriquement~: ils ont
la même direction, ou l'un d'eux est nul.
\end{definition}

\begin{definition}[Déterminant]\label{def:g11:vect:det}
Le \emph{déterminant}\index{déterminant} de $\vec u\,(x, y)$ et
$\vec v\,(x', y')$ est
\[
\det(\vec u, \vec v) =
\begin{vmatrix} x & x' \\ y & y' \end{vmatrix}
= x y' - x' y .
\]
\end{definition}

\begin{theorem}[Critère de colinéarité]\label{thm:g11:vect:collinearity}
Deux \omterm{def:g11:vect:vector}{vecteurs} $\vec u$ et $\vec v$ sont \omterm{def:g11:vect:collinear}{colinéaires} si et seulement si
$\det(\vec u, \vec v) = 0$.
\end{theorem}

\begin{proof}
($\Rightarrow$) Si $\vec v = \lambda\vec u$, alors $x' = \lambda x$ et
$y' = \lambda y$, donc
$\det(\vec u, \vec v) = x(\lambda y) - (\lambda x)y = 0$~; si
$\vec u = \vec 0$, le \omterm{def:g11:vect:det}{déterminant} est trivialement $0$.

($\Leftarrow$) Supposons $xy' - x'y = 0$ et $\vec u \neq \vec 0$, disons
$x \neq 0$ (le cas $y \neq 0$ est symétrique). Poser
$\lambda = \frac{x'}{x}$. Alors $x' = \lambda x$ par construction, et la
relation $xy' = x'y = \lambda x y$ donne, après division par $x \neq 0$,
$y' = \lambda y$. D'où $\vec v = \lambda \vec u$.
\end{proof}

\begin{example}\label{ex:g11:vect:collinear}
$\vec u\,(3, -2)$ et $\vec v\,(-6, 4)$~:
$\det = 3 \times 4 - (-6) \times (-2) = 12 - 12 = 0$, \omterm{def:g11:vect:collinear}{colinéaires} (en
effet $\vec v = -2\vec u$). $\vec u\,(3, -2)$ et $\vec w\,(2, 1)$~:
$\det = 3 + 4 = 7 \neq 0$, non \omterm{def:g11:vect:collinear}{colinéaires}.
\end{example}

\begin{method}[Alignement de trois points]\label{met:g11:vect:alignment}
$A$, $B$, $C$ sont alignés si et seulement si $\vect{AB}$ et $\vect{AC}$
sont \omterm{def:g11:vect:collinear}{colinéaires}~: calculer les deux \omterm{def:g11:vect:vector}{vecteurs} et vérifier
$\det\bigl(\vect{AB}, \vect{AC}\bigr) = 0$.
\end{method}

\section{Équations cartésiennes de droites}

\begin{definition}[Vecteur directeur]\label{def:g11:vect:direction}
Un \emph{vecteur directeur}\index{vecteur directeur} d'une droite $d$ est
tout \omterm{def:g11:vect:vector}{vecteur} non nul $\vec u$ tel que $\vect{AB}$ est \omterm{def:g11:vect:collinear}{colinéaire} à
$\vec u$ pour tous points $A, B$ de $d$.
\end{definition}

\begin{theorem}[Équation cartésienne]\label{thm:g11:vect:cartesian}
Une droite de \omterm{def:g11:vect:direction}{vecteur directeur} $\vec u\,(-b, a)$ admet une \omterm{def:g10:algebra:equation}{équation} de
la forme
\[
ax + by + c = 0
\qquad (a, b) \neq (0, 0).
\]
Réciproquement, toute telle \omterm{def:g10:algebra:equation}{équation} décrit une droite de \omterm{def:g11:vect:direction}{vecteur
directeur} $\vec u\,(-b, a)$.
\end{theorem}

\begin{proof}
Soit $d$ passant par $A(x_A, y_A)$ de direction $\vec u\,(-b, a)$. Un
point $M(x, y)$ est sur $d$ exactement lorsque $\vect{AM}$ est \omterm{def:g11:vect:collinear}{colinéaire}
à $\vec u$, c'est-à-dire (\cref{thm:g11:vect:collinearity})
\[
0 = \det\bigl(\vect{AM}, \vec u\bigr)
= (x - x_A)\,a - (-b)(y - y_A)
= ax + by - (a x_A + b y_A),
\]
une \omterm{def:g10:algebra:equation}{équation} de la forme annoncée avec $c = -(a x_A + b y_A)$.
Réciproquement, les solutions de $ax + by + c = 0$ avec, disons,
$b \neq 0$ sont les points $\left(x, -\frac{a x + c}{b}\right)$~: en
soustrayant deux d'entre eux, toute corde est \omterm{def:g11:vect:collinear}{colinéaire} à $(-b, a)$, et
une solution existe toujours --- c'est la droite passant par elle de
direction $(-b, a)$.
\end{proof}

\begin{method}[Droite par deux points]\label{met:g11:vect:twopoints}
Pour trouver une \omterm{def:g10:algebra:equation}{équation} de la droite $(AB)$~: calculer le \omterm{def:g11:vect:direction}{vecteur
directeur} $\vect{AB}\,(x_B - x_A, y_B - y_A)$~; l'identifier à $(-b, a)$,
c'est-à-dire prendre $a = y_B - y_A$ et $b = -(x_B - x_A)$~; déterminer
$c$ en injectant les \omterm{def:g10:coordgeom:system}{coordonnées} de $A$.
\end{method}

\begin{example}\label{ex:g11:vect:twopoints}
Droite par $A(1, 2)$ et $B(4, 3)$~: $\vect{AB}\,(3, 1)$, donc $a = 1$,
$b = -3$, et l'\omterm{def:g10:algebra:equation}{équation} est $x - 3y + c = 0$. En injectant $A$~:
$1 - 6 + c = 0$, donc $c = 5$. \omterm{def:g10:algebra:equation}{Équation}~: $x - 3y + 5 = 0$. Vérification
avec $B$~: $4 - 9 + 5 = 0$.
\end{example}

\begin{remark}
Lorsque $b \neq 0$, l'\omterm{def:g10:algebra:equation}{équation} peut se résoudre en $y$~:
$y = mx + p$ avec \omterm{def:g10:reffunc:affine}{coefficient directeur} $m = -\frac ab$. La forme
cartésienne $ax + by + c = 0$ est plus générale~: elle couvre aussi les
droites verticales ($b = 0$), qui n'ont pas de \omterm{def:g10:reffunc:affine}{coefficient directeur}.
\end{remark}

\section{Représentation paramétrique}

\begin{definition}[Représentation paramétrique]\label{def:g11:vect:parametric}
La droite passant par $A(x_A, y_A)$ de direction $\vec u\,(\alpha, \beta)$
est l'ensemble des points
\[
M\bigl(x_A + t\alpha,\ y_A + t\beta\bigr), \qquad t \in \R .
\]
Le nombre $t$ est le \emph{paramètre}~: chaque valeur de $t$ produit un
point de la droite, comme si l'on se déplaçait le long d'elle à vitesse
constante $\vec u$.
\end{definition}

\begin{example}
La droite de l'\cref{ex:g11:vect:twopoints} est aussi
$\{(1 + 3t,\ 2 + t) : t \in \R\}$~: $t = 0$ donne $A$, $t = 1$ donne $B$.
En éliminant $t$ ($t = y - 2$, donc $x = 1 + 3(y-2)$) on retrouve
$x - 3y + 5 = 0$.
\end{example}

\section{Positions relatives de deux droites}

\begin{proposition}[Parallèles ou sécantes]\label{prop:g11:vect:positions}
Deux droites de \omterm{def:g11:vect:direction}{vecteurs directeurs} $\vec u$ et $\vec v$ sont parallèles
si et seulement si $\det(\vec u, \vec v) = 0$. Sous forme d'\omterm{def:g10:algebra:equation}{équations}~:
$d : ax + by + c = 0$ et $d' : a'x + b'y + c' = 0$ sont parallèles si et
seulement si $ab' - a'b = 0$. Des droites non parallèles se coupent en
exactement un point.
\end{proposition}

\begin{proof}
Le parallélisme signifie des directions \omterm{def:g11:vect:collinear}{colinéaires}, ce qui est le
\cref{thm:g11:vect:collinearity}~; avec $\vec u\,(-b, a)$ et
$\vec v\,(-b', a')$, le \omterm{def:g11:vect:det}{déterminant} est $(-b)a' - (-b')a = ab' - a'b$.
Si les droites ne sont pas parallèles, résoudre les deux \omterm{def:g10:algebra:equation}{équations}
simultanément (par substitution ou combinaison) conduit à une unique
solution~: le système se comporte comme un système $2 \times 2$ à
\omterm{def:g11:vect:det}{déterminant} non nul.
\end{proof}

\begin{method}[Intersection de deux droites]\label{met:g11:vect:intersection}
Résoudre le système des deux \omterm{def:g10:algebra:equation}{équations}~: isoler une inconnue dans une
\omterm{def:g10:algebra:equation}{équation} et substituer dans l'autre, ou combiner les \omterm{def:g10:algebra:equation}{équations} pour
éliminer une inconnue. Toujours vérifier le point obtenu dans
\emph{les deux} \omterm{def:g10:algebra:equation}{équations}.
\end{method}

\begin{example}\label{ex:g11:vect:intersection}
Intersecter $d: x - 3y + 5 = 0$ et $d': 2x + y - 4 = 0$. De $d'$~:
$y = 4 - 2x$. En substituant dans $d$~:
\[
x - 3(4 - 2x) + 5 = 0
\iff x - 12 + 6x + 5 = 0
\iff 7x = 7
\iff x = 1,
\]
puis $y = 2$. Les droites se coupent en $(1, 2)$. Vérification dans
$d$~: $1 - 6 + 5 = 0$.
\end{example}

\section{Problème~: routes de collision et coordonnées bien choisies}

\begin{problem}[{Devoir du week-end --- se croiser n'est pas
entrer en collision~: distance minimale d'approche en mer, et des
théorèmes démontrés en choisissant le bon repère}]
\label{pb:g11:vect:1}
Les routes rectilignes de deux navires se croisent --- les navires
doivent-ils pour autant entrer en collision~? Non, sauf s'ils
atteignent le point de croisement \emph{au même instant}~: une
trajectoire est un ensemble de points, un mouvement est
paramétré, et confondre les deux a coulé de vrais navires. Ce
problème fait tourner un petit service de trafic maritime sur des
droites paramétrées (\cref{def:g11:vect:parametric}), puis passe à la
géométrie pure avec l'autre superpouvoir des \omterm{def:g10:coordgeom:system}{coordonnées}~:
\emph{les choisir} habilement, comme à l'\cref{exo:g11:vect:11},
pour que les théorèmes classiques se démontrent tout seuls.

\medskip
\noindent\textbf{Partie I --- La droite, de trois façons.}
\begin{enumerate}
  \item Donner une représentation paramétrique de la droite passant
        par $A(1, 2)$ et de \omterm{def:g11:vect:direction}{vecteur directeur} $\vec u(3, -1)$, puis
        éliminer le paramètre pour obtenir une \omterm{def:g10:algebra:equation}{équation}
        cartésienne.
  \item Pour la droite $2x - 5y + 3 = 0$~: lire un \omterm{def:g11:vect:direction}{vecteur
        directeur} (\cref{thm:g11:vect:cartesian}), trouver un point,
        et écrire une représentation paramétrique.
  \item Calculer le point d'\omterm{def:g10:numbers:interunion}{intersection} des deux droites des
        questions 1 et 2 (\cref{met:g11:vect:intersection}).
  \item Confirmer à l'aide du \omterm{def:g11:vect:det}{déterminant}
        (\cref{thm:g11:vect:collinearity}) que ces deux droites
        devaient se couper~; puis montrer que $2x - 5y + 3 = 0$ et
        $-4x + 10y + 1 = 0$ sont strictement parallèles.
  \item Les points $A(2, 1)$, $B(5, 3)$, $C(11, 7)$ sont-ils
        alignés (\cref{met:g11:vect:alignment})~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Le service de trafic maritime.}
À l'instant $t$ (en heures), le navire $P$ est en $(2t,\ 3 + t)$ et
le navire $Q$ en $(10 - t,\ 2t - 1)$ (distances en milles marins).
\begin{enumerate}[resume]
  \item Déterminer les \omterm{def:g10:algebra:equation}{équations} cartésiennes des deux
        \emph{trajectoires}, et calculer leur point de croisement.
  \item À quel instant chaque navire passe-t-il au point de
        croisement~? Y a-t-il collision~? Énoncer en une phrase
        l'avertissement général~: trajectoires qui se croisent
        contre mouvements qui se rencontrent.
  \item Calculer le carré de la distance $D^2(t)$ entre les navires
        à l'instant $t$, et le réduire à un trinôme du second degré
        en $t$. À quel instant les navires sont-ils au plus près, et
        à quelle distance passent-ils l'un de l'autre (technique du
        sommet, \cref{pb:g11:quad:1})~?
  \item Le règlement impose une séparation d'au moins $1$ mille.
        Est-elle violée~? Si oui, résoudre $D^2(t) < 1$ et donner la
        durée de l'infraction.
  \item Expliquer pourquoi, pour \emph{deux} navires quelconques
        suivant des routes rectilignes à vitesse constante, $D^2(t)$
        est toujours une \omterm{def:g11:func:function}{fonction} polynôme du second degré de $t$
        --- de sorte que la «~distance minimale d'approche~» est
        toujours un calcul de sommet.
  \item Interception~: une vedette part de l'origine à $t = 0$ avec
        le \omterm{def:g11:vect:vector}{vecteur} vitesse constant $(a, b)$ et doit rejoindre le
        navire $P$ exactement à $t = 3$. Calculer le couple
        $(a, b)$ nécessaire et la vitesse de la vedette.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Des \omterm{def:g10:coordgeom:system}{coordonnées} bien choisies.}
On travaille dans le \omterm{def:g10:coordgeom:system}{repère} de l'\cref{exo:g11:vect:11}~: le
parallélogramme $ABCD$ devient $A(0,0)$, $B(1,0)$, $C(1,1)$,
$D(0,1)$~; on note $I$ le \omterm{prop:g10:coordgeom:midpoint}{milieu} de $[AB]$ et $K$ le \omterm{prop:g10:coordgeom:midpoint}{milieu} de
$[CD]$.
\begin{enumerate}[resume]
  \item Calculer une \omterm{def:g10:algebra:equation}{équation} cartésienne de la droite $(DI)$.
  \item Couper $(DI)$ par la diagonale $(AC)$~: on retrouve le fait
        de l'\cref{exo:g11:vect:11} --- la coupure se situe
        exactement au tiers de la diagonale.
  \item Passons à la droite $(BK)$~: déterminer son \omterm{def:g10:algebra:equation}{équation}, la
        couper par $(AC)$, et conclure le théorème classique
        complet~: \emph{les céviennes $(DI)$ et $(BK)$ partagent la
        diagonale $(AC)$ en trois parts égales}.
  \item Pourquoi était-il légitime de démontrer le théorème dans ce
        seul \omterm{def:g10:coordgeom:system}{repère} particulier~? (Que conserve le choix des
        \omterm{def:g10:coordgeom:system}{coordonnées}, et de quoi l'énoncé parle-t-il~?) Répondre en
        deux ou trois phrases.
  \item Encore un partage en trois, pour s'exercer~: soit $E$ le
        \omterm{prop:g10:coordgeom:midpoint}{milieu} de $[BC]$. En quel point la droite $(AE)$ coupe-t-elle
        l'\emph{autre} diagonale $(DB)$~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Énigmes de position.}
\begin{enumerate}[resume]
  \item Les trois droites $x + y = 4$, \ $2x - y = 2$, \
        $x - 2y = -2$ sont-elles concourantes~? (En couper deux,
        tester la troisième.)
  \item Pour tout réel $m$, on note $d_m$ la droite
        $mx - y + 2 - 3m = 0$. Montrer que \emph{toutes} les $d_m$
        passent par un même \omterm{pb:g10:functions:1}{point fixe}. (Regrouper les termes en
        $m$.)
  \item Dans la famille $(d_m)$~: quel membre est parallèle à
        $y = 2x + 7$~? Lequel passe par l'origine~?
  \item Pour finir --- les trois costumes d'une droite~: cartésien
        (test d'appartenance en une substitution), réduit
        (\omterm{def:g10:reffunc:affine}{coefficient directeur} et courbe à vue), paramétrique
        (mouvement, horaires, interception)~; le \omterm{def:g11:vect:det}{déterminant} comme
        oracle du parallélisme~; et le \omterm{def:g10:coordgeom:system}{repère} bien choisi comme
        fabrique à théorèmes. Une phrase pour chacun.
\end{enumerate}
\end{problem}
