\chapter{La loi binomiale}\label{ch:g11:binom}

Répéter la même expérience oui/non plusieurs fois, indépendamment, et
compter les succès~: la \omterm{def:g11:prob:rv}{loi} résultante --- la \emph{binomiale} --- est la
plus importante de toutes les \omterm{def:g11:prob:rv}{lois} discrètes. Ce chapitre la construit
avec des \omterm{met:g10:proba:tree}{arbres} et le comptage de chemins~; la formule fermée pour les
comptages de chemins (avec factorielles) vient avec les outils de
dénombrement du \cref{ch:g12:comb}, et la \omterm{def:g11:prob:rv}{loi} est revue dans le
\cref{ch:g12:randvar}.

\section{Épreuves de Bernoulli}

\begin{definition}[Épreuve de Bernoulli]\label{def:g11:binom:bernoulli}
Une \emph{épreuve de Bernoulli}\index{épreuve de Bernoulli} est une
expérience à exactement deux issues~: \emph{succès}, de \omterm{def:g10:proba:distribution}{probabilité} $p$,
et \emph{échec}, de \omterm{def:g10:proba:distribution}{probabilité} $1 - p$. La \omterm{def:g11:prob:rv}{variable aléatoire} $X$ égale
à $1$ en cas de succès et $0$ en cas d'échec suit la \emph{\omterm{def:g11:prob:rv}{loi} de
Bernoulli} $\mathcal B(p)$~; alors
\[
\E(X) = p,
\qquad
\V(X) = p(1 - p).
\]
\end{definition}

\begin{proof}[Démonstration des deux formules]
$\E(X) = p \times 1 + (1-p) \times 0 = p$~; et comme $X^2 = X$ ($0$ et
$1$ sont leurs propres carrés), $\E(X^2) = p$, donc par la
\cref{prop:g11:prob:konig}, $\V(X) = p - p^2 = p(1-p)$.
\end{proof}

\begin{definition}[Épreuves indépendantes répétées]\label{def:g11:binom:repeated}
Répéter une \omterm{def:g11:binom:bernoulli}{épreuve de Bernoulli} $n$ fois \emph{indépendamment} signifie~:
l'issue de chaque épreuve n'a aucune influence sur les autres, et la
\omterm{def:g10:proba:distribution}{probabilité} de toute \omterm{def:g11:seq:sequence}{suite} complète d'issues est le \emph{produit} des
\omterm{def:g10:proba:distribution}{probabilités} le long du chemin correspondant de l'\omterm{met:g10:proba:tree}{arbre} --- $p$ pour
chaque succès, $1 - p$ pour chaque échec.
\end{definition}

\begin{example}\label{ex:g11:binom:threetrials}
Trois épreuves indépendantes de \omterm{def:g10:proba:distribution}{probabilité} de succès $p$. La \omterm{def:g11:seq:sequence}{suite} SES
(succès, échec, succès) a pour \omterm{def:g10:proba:distribution}{probabilité} $p(1-p)p = p^2(1-p)$ --- et
il en va de même pour \emph{toute} \omterm{def:g11:seq:sequence}{suite} avec exactement deux succès,
quelles que soient les positions~: seul le \emph{nombre} de S et de E
compte.
\end{example}

\section{Comptage de chemins et coefficients binomiaux}

\begin{definition}[Coefficient binomial]\label{def:g11:binom:coefficient}
Dans l'\omterm{met:g10:proba:tree}{arbre} de $n$ épreuves indépendantes, le \emph{coefficient
binomial}\index{coefficient binomial} $\binom{n}{k}$ (lu «~$n$ parmi
$k$~») est le nombre de chemins contenant exactement $k$ succès.
\end{definition}

\begin{example}
$\binom{3}{2} = 3$~: les chemins SSE, SES, ESS. De même
$\binom{3}{0} = 1$ (le chemin EEE), $\binom{3}{1} = 3$ et
$\binom{3}{3} = 1$. Par convention et par l'\omterm{met:g10:proba:tree}{arbre},
$\binom n0 = \binom nn = 1$ pour tout $n$.
\end{example}

\begin{proposition}[Règle de Pascal]\label{prop:g11:binom:pascal}
\index{triangle de Pascal}
Pour $1 \leq k \leq n - 1$~:
\[
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}.
\]
\end{proposition}

\begin{proof}
Classer les chemins de l'\omterm{met:g10:proba:tree}{arbre} à $n$ épreuves avec $k$ succès selon leur
\emph{dernière} épreuve. Ceux qui se terminent par un succès s'obtiennent
d'un chemin des $n-1$ premières épreuves avec $k - 1$ succès~: il y en a
$\binom{n-1}{k-1}$. Ceux qui se terminent par un échec prolongent un
chemin avec $k$ succès parmi les $n-1$ premières~: $\binom{n-1}{k}$.
Tout chemin est d'exactement l'un des deux types.
\end{proof}

La règle de Pascal engendre les coefficients ligne par ligne --- chaque
entrée est la somme des deux au-dessus d'elle~:
\[
\begin{array}{ccccccccccc}
&&&&&1&&&&&\\
&&&&1&&1&&&&\\
&&&1&&2&&1&&&\\
&&1&&3&&3&&1&&\\
&1&&4&&6&&4&&1&\\
1&&5&&10&&10&&5&&1
\end{array}
\]

\begin{remark}
Une formule fermée, $\binom nk = \frac{n!}{k!(n-k)!}$, avec une théorie
systématique du dénombrement, est établie dans le \cref{ch:g12:comb}. À
ce niveau, le \omterm{prop:g11:binom:pascal}{triangle de Pascal} calcule tout coefficient dont on a
besoin.
\end{remark}

\section{La loi binomiale}

\begin{theorem}[Loi binomiale]\label{thm:g11:binom:binomial}
Soit $X$ le nombre de succès dans $n$ \omterm{def:g11:binom:bernoulli}{épreuves de Bernoulli}
indépendantes de paramètre $p$. Alors $X$ suit la \emph{loi
binomiale}\index{loi binomiale} $\mathcal B(n, p)$~:
\[
\P(X = k) = \binom{n}{k}\, p^k (1-p)^{n-k},
\qquad k = 0, 1, \dots, n .
\]
\end{theorem}

\begin{proof}
L'\omterm{def:g11:prob:model}{événement} $X = k$ est la collection de tous les chemins avec
exactement $k$ succès. Chaque tel chemin a pour \omterm{def:g10:proba:distribution}{probabilité}
$p^k(1-p)^{n-k}$~: le produit le long du chemin contient $k$ facteurs
$p$ et $n - k$ facteurs $1-p$, dans un certain ordre
(\cref{def:g11:binom:repeated}). Il y a $\binom nk$ tels chemins
(\cref{def:g11:binom:coefficient}), et leurs \omterm{def:g10:proba:distribution}{probabilités} s'additionnent.
\end{proof}

\begin{example}\label{ex:g11:binom:quiz}
Un quiz a $5$ questions indépendantes, chacune à $4$ choix~; un élève
répond au hasard, donc chaque question est un succès avec
$p = \frac14$. Le nombre $X$ de bonnes réponses suit
$\mathcal B\left(5, \frac14\right)$, et, en utilisant la ligne~$5$ du
\omterm{prop:g11:binom:pascal}{triangle de Pascal}~:
\[
\P(X = 2) = \binom52 \left(\frac14\right)^2\left(\frac34\right)^3
= 10 \times \frac{1}{16} \times \frac{27}{64}
= \frac{270}{1024} \approx 0.26 .
\]
La \omterm{def:g10:proba:distribution}{probabilité} d'au moins une bonne réponse utilise le \omterm{def:g10:proba:operations}{complémentaire}~:
$\P(X \geq 1) = 1 - \P(X = 0) = 1 - \left(\frac34\right)^5 \approx
0.76$.
\end{example}

\begin{proposition}[Espérance et variance]\label{prop:g11:binom:expectation}
Si $X \sim \mathcal B(n, p)$~:
\[
\E(X) = np,
\qquad
\V(X) = np(1-p).
\]
\end{proposition}

\begin{proof}[Justification]
Écrire $X = X_1 + X_2 + \dots + X_n$, où $X_i$ vaut $1$ si la $i$-ième
épreuve réussit~: chaque $X_i$ est une variable de Bernoulli d'\omterm{def:g11:prob:expectation}{espérance}
$p$ (\cref{def:g11:binom:bernoulli}). Les \omterm{def:g11:stat:mean}{moyennes} s'additionnent ---
sommer les $n$ contributions donne $\E(X) = np$. Que les \emph{\omterm{def:g11:prob:variance}{variances}}
s'additionnent aussi pour des variables indépendantes est vrai mais plus
délicat~: la formule de \omterm{def:g11:prob:variance}{variance} est \emph{admise à ce niveau} et
prouvée dans le \cref{ch:g12:sums}.
\end{proof}

\begin{method}[Reconnaître une situation binomiale]\label{met:g11:binom:recognize}
Avant d'écrire $X \sim \mathcal B(n, p)$, vérifier trois ingrédients~:
un \emph{nombre fixe} $n$ d'épreuves, décidé à l'avance~; chaque épreuve
a \emph{deux issues} avec la \emph{même} \omterm{def:g10:proba:distribution}{probabilité} de succès $p$~; les
épreuves sont \emph{indépendantes} (avec remise, ou dispositifs
séparés). Tirer sans remise dans une petite population n'est \emph{pas}
binomial --- la \omterm{def:g10:proba:distribution}{probabilité} change à chaque tirage
(\cref{exo:g11:prob:6}).
\end{method}

\section{Échantillonnage~: l'observation est-elle surprenante~?}

La \omterm{def:g11:prob:rv}{loi} binomiale répond à une question très pratique~: \emph{si la
\omterm{def:g10:proba:distribution}{probabilité} de succès est vraiment $p$, quels comptes de succès sont
plausibles~?}

\begin{example}\label{ex:g11:binom:decision}
Une machine est censée produire au plus $10\,\%$ de pièces défectueuses.
Dans un lot de $10$ pièces, $4$ sont défectueuses. Malchance ou machine
cassée~? Si la machine est en bon état, le nombre de défectueuses suit
$\mathcal B(10, 0.1)$, et
\[
\P(X \geq 4) = 1 - \P(X \leq 3) \approx 1 - 0.987 = 0.013 :
\]
environ une chance sur $80$. Observer un \omterm{def:g11:prob:model}{événement} aussi improbable est
un signal fort --- on \emph{rejette} l'hypothèse que la machine fonctionne
encore à $10\,\%$, tout en gardant à l'esprit que la décision pourrait
être fausse avec une \omterm{def:g10:proba:distribution}{probabilité} d'environ $0.013$.
\end{example}

\begin{method}[Règle de décision à partir d'un modèle binomial]\label{met:g11:binom:decision}
Pour juger un nombre observé $k$ de succès contre une hypothèse
$X \sim \mathcal B(n, p)$~: calculer la \omterm{def:g10:proba:distribution}{probabilité}, sous l'hypothèse,
d'un résultat \emph{au moins aussi extrême} que $k$. Si cette
\omterm{def:g10:proba:distribution}{probabilité} est très petite (une convention courante~: sous $5\,\%$),
rejeter l'hypothèse~; sinon l'observation est compatible avec elle. Le
seuil est un choix, pas un théorème --- la statistique quantifie le
risque, et l'utilisateur l'accepte.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g11:binom:1}
Prolonger le \omterm{prop:g11:binom:pascal}{triangle de Pascal} jusqu'à la ligne~$7$, et donner les
valeurs de $\binom62$, $\binom{7}{3}$ et $\binom74$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:binom:2}
Un dé équilibré est lancé $4$ fois~; $X$ compte les six. Justifier que
$X \sim \mathcal B\left(4, \frac16\right)$ et calculer $\P(X = 0)$,
$\P(X = 1)$ et $\P(X \geq 2)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:binom:3}
Lequel des cas suivants est binomial~? Justifier.
\begin{enumerate}
  \item Nombre de faces dans $20$ lancers d'une pièce équilibrée.
  \item Nombre d'as dans $5$ cartes distribuées d'un jeu.
  \item Nombre de jours de pluie la semaine prochaine, si chaque jour est
        pluvieux avec \omterm{def:g10:proba:distribution}{probabilité} $0.3$ indépendamment.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:binom:4}
$X \sim \mathcal B(50, 0.2)$. Donner $\E(X)$, $\V(X)$ et $\sigma(X)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:5}
Un archer touche la cible avec \omterm{def:g10:proba:distribution}{probabilité} $0.7$ à chaque tir,
indépendamment. En $6$ tirs, calculer la \omterm{def:g10:proba:distribution}{probabilité} d'exactement $4$
touches, et d'au moins $5$ touches.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:6}
Un test vrai/faux a $8$ questions~; un élève devine chaque réponse.
Quelle est la \omterm{def:g10:proba:distribution}{probabilité} de réussir (au moins $6$ bonnes réponses)~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:7}
Chaque boîte de céréales achetée contient la figurine A ou la figurine B,
avec \omterm{def:g10:proba:distribution}{probabilité} $\frac12$ chacune, indépendamment. Un collectionneur
achète $5$ boîtes. Calculer la \omterm{def:g10:proba:distribution}{probabilité} qu'il obtienne au moins une
figurine de chaque type. (\omterm{def:g10:proba:operations}{Complémentaire}~: toutes A ou toutes B.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:8}
Un basketteur marque des lancers francs avec \omterm{def:g10:proba:distribution}{probabilité} $p$,
indépendamment. Soit $X \sim \mathcal B(3, p)$ le nombre de réussites en
trois tirs. Exprimer $\P(X = 3)$ et $\P(X \geq 1)$ en \omterm{def:g11:func:function}{fonction} de $p$,
et trouver pour quel $p$ la \omterm{def:g10:proba:distribution}{probabilité} de marquer les trois égale
$\frac{27}{64}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:9}
Combien de fois faut-il lancer une pièce équilibrée pour que la
\omterm{def:g10:proba:distribution}{probabilité} d'obtenir au moins une face dépasse $0.99$~?
(\omterm{def:g10:proba:operations}{Complémentaire}, puis essayer des valeurs successives de $n$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:10}
En utilisant la règle de Pascal (\cref{prop:g11:binom:pascal}) et
$\binom n0 = \binom nn = 1$, prouver que les entrées de chaque ligne du
\omterm{prop:g11:binom:pascal}{triangle de Pascal} somment à $2^n$~: interpréter les deux côtés comme le
comptage de tous les chemins de l'\omterm{met:g10:proba:tree}{arbre}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:binom:11}
Un politicien prétend avoir $60\,\%$ d'approbation. Dans un \omterm{def:g10:proba:sample}{échantillon}
aléatoire de $10$ personnes, seulement $3$ approuvent.
\begin{enumerate}
  \item Sous l'affirmation, quelle \omterm{def:g11:prob:rv}{loi} suit le nombre $X$
        d'approbations dans l'\omterm{def:g10:proba:sample}{échantillon}~? Calculer $\P(X \leq 3)$.
  \item En utilisant la règle de décision de la
        \cref{met:g11:binom:decision} avec un seuil de $5\,\%$,
        l'observation est-elle compatible avec l'affirmation~?
\end{enumerate}
\end{exercise}

\section{Problème~: la planche de Galton}

\begin{problem}[{Devoir du week-end --- billes, clous et triangle de
Pascal~: comment naît la courbe en cloche, pourquoi les séries de
matchs favorisent la meilleure équipe, et quand crier à la triche}]
\label{pb:g11:binom:1}
Laissez tomber mille billes à travers un réseau de clous, chaque
rebond étant un pile ou face équilibré entre la gauche et la droite,
et les cases du bas se remplissent en une cloche lisse et symétrique
--- à chaque fois. La machine s'appelle une planche de Galton, et
ses mathématiques sont exactement la \omterm{def:g11:prob:rv}{loi} binomiale de ce chapitre
(\cref{thm:g11:binom:binomial}). Ce problème construit le triangle,
fait tourner la planche, arbitre une série au meilleur des sept
matchs, et s'achève là où la binomiale gagne son salaire~: décider
quand une observation doit nous faire douter d'une affirmation.

\medskip
\noindent\textbf{Partie I --- Le triangle.}
\begin{enumerate}
  \item Construire le \omterm{prop:g11:binom:pascal}{triangle de Pascal} jusqu'à la ligne $6$
        (\cref{prop:g11:binom:pascal}). Énoncer et expliquer en une
        phrase la symétrie $\binom nk = \binom{n}{n-k}$ (choisir $k$
        objets, c'est la même chose que\,\dots).
  \item Vérifier sur les lignes $4$ et $5$ que chaque ligne a pour
        somme $2^n$, et le démontrer~: que dénombrent ensemble tous
        les $\binom nk$~?
  \item Retrouver la règle de Pascal
        $\binom{n+1}{k} = \binom nk + \binom{n}{k-1}$ par
        l'argument du comité~: distinguer une personne et répartir
        les comités selon le sort de cette personne.
  \item Calculer $\binom73$ de deux façons~: sur le triangle, puis
        par la formule avec les factorielles.
  \item Vérifier l'identité de l'escalier
        $\binom22 + \binom32 + \binom42 + \binom52 = \binom63$, et
        l'expliquer en faisant cascader la règle de Pascal à partir
        de $\binom63$ vers le bas.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La planche.}
Une bille tombe à travers $n$ rangées de clous~; à chaque clou elle
rebondit à gauche ou à droite avec la \omterm{def:g10:proba:distribution}{probabilité} $\frac12$, de
façon indépendante. On numérote les cases de $0$ à $n$ par le nombre
de rebonds vers la droite.
\begin{enumerate}[resume]
  \item Expliquer, avec la liste de contrôle de la
        \cref{met:g11:binom:recognize}, pourquoi le numéro de case
        suit la \omterm{def:g11:prob:rv}{loi} binomiale
        $\mathcal B\!\left(n, \frac12\right)$.
  \item Pour une petite planche ($n = 4$)~: donner les cinq
        \omterm{def:g10:proba:distribution}{probabilités} des cases. Quelle case est la plus chargée~?
  \item Prenons maintenant $n = 10$ et $1\,024$ billes~: quels sont
        les effectifs \emph{attendus} dans la case centrale, dans la
        case $7$, et dans chaque case du bord~? Décrire la forme du
        tas.
  \item Pour $X \sim \mathcal B\!\left(10, \frac12\right)$~:
        calculer $\E(X)$, $V(X)$ et $\sigma$
        (\cref{prop:g11:binom:expectation})~; puis calculer la
        proportion de billes attendues à moins de $2\sigma$ du
        centre (cases $2$ à $8$) et comparer avec la garantie de
        Tchebychev du \cref{pb:g11:stat:1}.
  \item Une planche inclinée fait rebondir à droite avec la
        \omterm{def:g10:proba:distribution}{probabilité} $0.6$~: donner $\E$, $V$ et $\sigma$ pour
        $n = 10$, et décrire ce qu'il advient du tas.
  \item En une ou deux phrases~: qu'est-ce qui, dans la conception
        de la planche, fabrique la forme en cloche --- et pourquoi
        tant de grandeurs du monde réel (tailles, erreurs de mesure)
        s'empilent-elles de la même façon~? (Le théorème profond qui
        est derrière les deux est le théorème central limite, sommet
        du cours de \omterm{def:g10:proba:distribution}{probabilités} des volumes universitaires.)
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Au meilleur des sept.}
Deux équipes disputent une série~: la première à $4$ victoires
emporte le titre~; les matchs sont indépendants.
\begin{enumerate}[resume]
  \item Équipes de force égale ($p = \frac12$)~: calculer la
        \omterm{def:g10:proba:distribution}{probabilité} que la série se termine par un balayage
        (exactement $4$ matchs).
  \item Calculer la \omterm{def:g10:proba:distribution}{probabilité} que la série aille jusqu'au
        septième match (quel doit être le score après $6$~?).
  \item Compléter la \omterm{def:g11:prob:rv}{loi} de la longueur de la série ($4$, $5$, $6$
        ou $7$ matchs) pour des équipes de force égale, et calculer
        la longueur espérée. Quelles longueurs sont les plus
        probables~?
  \item Supposons maintenant qu'une équipe gagne chaque match avec
        $p = 0.6$. Calculer sa \omterm{def:g10:proba:distribution}{probabilité} de remporter la série
        (victoire en $4$, $5$, $6$ ou $7$ matchs~: dans chaque cas
        l'équipe gagne le dernier match et $3$ des précédents).
        Qu'a fait la série de l'avantage par match~?
  \item Comparer avec une finale unique ($60\,\%$) et avec une série
        au meilleur des trois (la calculer). Énoncer l'effet général
        de la longueur d'une série sur le rapport entre talent et
        chance --- et pourquoi les ligues préfèrent les finales
        longues.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Quand crier à la triche.}
\begin{enumerate}[resume]
  \item On lance une pièce $100$ fois et l'on obtient $62$ piles.
        Pour une pièce équilibrée, donner $\E$, $\sigma$ et la cote
        $z$ (\cref{pb:g11:stat:1}) de l'observation. Verdict selon
        la convention des $2\sigma$~?
  \item Un fournisseur annonce $2\,\%$ de pièces défectueuses. Dans
        un lot de $50$, vous en trouvez $3$. Calculer
        $\P(X \geq 3)$ sous cette annonce
        ($X \sim \mathcal B(50, 0.02)$~; passer par $\P(X = 0)$,
        $\P(X = 1)$, $\P(X = 2)$). Est-ce alarmant au seuil de
        $5\,\%$ (\cref{met:g11:binom:decision},
        \cref{exo:g11:binom:11})~?
  \item Persévérance au loto~: chaque billet est gagnant (de quelque
        chose) avec la \omterm{def:g10:proba:distribution}{probabilité} $\frac{1}{1000}$. Calculer la
        \omterm{def:g10:proba:distribution}{probabilité} d'au moins un gain avec $1\,000$ billets. La
        réponse ($\approx 63\,\%$, et non $100\,\%$~!) cache une
        constante célèbre~: calculer $0.999^{1000}$ et garder le
        nombre $0.368$ en tête pour l'an prochain.
  \item Pour finir --- le portrait de la binomiale~: la liste de
        reconnaissance ($n$ fixé, indépendance, $p$ constant)~; le
        \omterm{prop:g11:binom:pascal}{triangle de Pascal} comme table~; la cloche comme forme~;
        $np$ et $np(1-p)$ comme boussole~; et ses deux héritiers qui
        attendent l'an prochain --- la courbe en cloche continue et
        la loi des grands nombres. Une phrase pour chacun.
\end{enumerate}
\end{problem}
