\chapter{Statistique descriptive}\label{ch:g11:stat}

La statistique comprime une liste de nombres en quelques résumés
signifiants~: où se situent les données (\omterm{def:g11:stat:mean}{moyenne}, \omterm{def:g11:stat:median}{médiane}), et comment
elles sont dispersées (\omterm{def:g11:stat:quartiles}{quartiles}, \omterm{def:g11:stat:variance}{variance}, \omterm{def:g11:stat:variance}{écart type}). Ce chapitre
calcule tout sur un exemple fil rouge. Les mêmes résumés réapparaissent
pour les variables aléatoires dans le \cref{ch:g11:prob}, et leurs lois
probabilistes dans le \cref{ch:g12:sums}.

Tout au long du chapitre, notre jeu de données fil rouge est les notes
(sur $20$) d'une classe de $12$ élèves~:
\[
8,\ 9,\ 10,\ 10,\ 11,\ 12,\ 12,\ 12,\ 13,\ 14,\ 16,\ 17 .
\]

\section{Moyenne et médiane}

\begin{definition}[Moyenne]\label{def:g11:stat:mean}
La \emph{moyenne}\index{moyenne} des valeurs $x_1, x_2, \dots, x_n$ est
\[
\bar x = \frac{x_1 + x_2 + \dots + x_n}{n}.
\]
Si la valeur $x_i$ apparaît avec l'effectif $n_i$ (avec
$n_1 + \dots + n_k = n$), alors
$\bar x = \frac{n_1 x_1 + \dots + n_k x_k}{n}$.
\end{definition}

\begin{definition}[Médiane]\label{def:g11:stat:median}
Une \emph{médiane}\index{médiane} d'un jeu de données trié est une valeur
qui le partage en deux~: au moins la moitié des valeurs sont $\leq$ à
elle, et au moins la moitié sont $\geq$ à elle. En pratique, trier les
$n$ valeurs~; si $n$ est impair, la médiane est la valeur du \omterm{prop:g10:coordgeom:midpoint}{milieu}~; si
$n$ est pair, prendre le \omterm{prop:g10:coordgeom:midpoint}{milieu} des deux valeurs centrales.
\end{definition}

\begin{example}\label{ex:g11:stat:meanmedian}
Pour le jeu fil rouge, la somme est $144$, donc $\bar x = \frac{144}{12}
= 12$. Les deux valeurs centrales ($6$-ième et $7$-ième de la liste
triée) sont toutes deux $12$, donc la \omterm{def:g11:stat:median}{médiane} est $12$ aussi. \omterm{def:g11:stat:mean}{Moyenne} et
\omterm{def:g11:stat:median}{médiane} n'ont pas besoin de coïncider~: la \omterm{def:g11:stat:median}{médiane} ignore \emph{à quel
point} les valeurs extrêmes sont éloignées, la \omterm{def:g11:stat:mean}{moyenne} non. Remplacer la
note haute $17$ par $20$ déplace la \omterm{def:g11:stat:mean}{moyenne} à $12.25$ mais laisse la
\omterm{def:g11:stat:median}{médiane} à $12$.
\end{example}

\section{Quartiles}

\begin{definition}[Quartiles, écart interquartile]\label{def:g11:stat:quartiles}
Le \emph{premier quartile}\index{quartile} $Q_1$ est la plus petite
valeur telle qu'au moins un quart des données est $\leq Q_1$~; le
\emph{troisième quartile} $Q_3$ est la plus petite valeur telle qu'au
moins les trois quarts des données sont $\leq Q_3$. L'\emph{écart
interquartile}\index{écart interquartile} est $Q_3 - Q_1$~: la largeur de la moitié centrale des
données.
\end{definition}

\begin{example}
$n = 12$~: un quart vaut $3$, donc $Q_1$ est la $3$-ième valeur
triée~: $Q_1 = 10$. Trois quarts valent $9$, donc $Q_3$ est la $9$-ième
valeur~: $Q_3 = 13$. L'\omterm{def:g11:stat:quartiles}{écart interquartile} est $3$~: la moitié centrale
de la classe se trouve dans un \omterm{def:g10:numbers:interval}{intervalle} de $3$ points.
\end{example}

\section{Variance et écart type}

\begin{definition}[Variance, écart type]\label{def:g11:stat:variance}
La \emph{variance}\index{variance} du jeu de données est la \omterm{def:g11:stat:mean}{moyenne} des
écarts au carré à la \omterm{def:g11:stat:mean}{moyenne}~:
\[
V = \frac{1}{n}\sum_{i=1}^{n} (x_i - \bar x)^2,
\]
et l'\emph{écart type}\index{écart type} est $\sigma = \sqrt{V}$, exprimé
dans la même unité que les données.
\end{definition}

\begin{proposition}[Formule de König--Huygens]\label{prop:g11:stat:shortcut}
La \omterm{def:g11:stat:variance}{variance} est la \omterm{def:g11:stat:mean}{moyenne} des carrés moins le carré de la \omterm{def:g11:stat:mean}{moyenne}~:
\[
V = \frac{1}{n}\sum_{i=1}^{n} x_i^2 \;-\; \bar x^{\,2}.
\]
\end{proposition}

\begin{proof}
Développer chaque écart au carré~:
$(x_i - \bar x)^2 = x_i^2 - 2\bar x\, x_i + \bar x^2$. En sommant et en
divisant par $n$~:
\[
V = \frac1n \sum x_i^2 - 2\bar x \cdot \underbrace{\frac1n \sum
x_i}_{=\ \bar x} + \frac1n \cdot n\bar x^2
= \frac1n\sum x_i^2 - 2\bar x^2 + \bar x^2
= \frac1n\sum x_i^2 - \bar x^2 .
\]
La même identité, pour les variables aléatoires, est la
\cref{prop:g11:prob:konig}.
\end{proof}

\begin{example}\label{ex:g11:stat:variance}
Pour le jeu fil rouge, les écarts à $\bar x = 12$ sont $-4$, $-3$, $-2$,
$-2$, $-1$, $0$, $0$, $0$, $1$, $2$, $4$, $5$~; leurs carrés somment à
\[
16 + 9 + 4 + 4 + 1 + 0 + 0 + 0 + 1 + 4 + 16 + 25 = 80 .
\]
D'où
\[
V = \frac{80}{12} = \frac{20}{3} \approx 6.67,
\qquad
\sigma = \sqrt{20/3} \approx 2.6 \text{ points}.
\]
En gros, une note typique se situe à environ $2.6$ points de la \omterm{def:g11:stat:mean}{moyenne}.
\end{example}

\begin{proposition}[Changement d'unités]\label{prop:g11:stat:affine}
Si chaque valeur est transformée par $y_i = a x_i + b$, alors
\[
\bar y = a \bar x + b,
\qquad
V_y = a^2\, V_x,
\qquad
\sigma_y = \abs{a}\,\sigma_x .
\]
\end{proposition}

\begin{proof}
$\bar y = \frac1n \sum (a x_i + b) = a\left(\frac1n\sum x_i\right) +
\frac1n \cdot nb = a\bar x + b$. Puis
$y_i - \bar y = a(x_i - \bar x)$~: chaque écart est multiplié par $a$
(le décalage $b$ s'annule), donc chaque écart au carré est multiplié par
$a^2$, et sa \omterm{def:g11:stat:mean}{moyenne} aussi. En prenant les racines carrées,
$\sigma_y = \abs a\,\sigma_x$.
\end{proof}

\begin{example}
Des températures enregistrées en degrés Celsius de \omterm{def:g11:stat:mean}{moyenne} $20$ et
d'\omterm{def:g11:stat:variance}{écart type} $3$ se convertissent en Fahrenheit par
$F = 1.8\,C + 32$~: la \omterm{def:g11:stat:mean}{moyenne} devient $1.8 \times 20 + 32 = 68$,
l'\omterm{def:g11:stat:variance}{écart type} $1.8 \times 3 = 5.4$. Décaler un jeu de données ne change
pas sa dispersion~; le redimensionner redimensionne la dispersion.
\end{example}

\begin{method}[Résumer et comparer des jeux de données]\label{met:g11:stat:compare}
Pour comparer deux jeux de données, mettre côte à côte~: \omterm{def:g11:stat:mean}{moyenne} (ou
\omterm{def:g11:stat:median}{médiane}) pour la \emph{position}, \omterm{def:g11:stat:variance}{écart type} (ou \omterm{def:g11:stat:quartiles}{écart interquartile})
pour la \emph{dispersion}. Le couple (\omterm{def:g11:stat:median}{médiane}, \omterm{def:g11:stat:quartiles}{écart interquartile}) est
robuste aux valeurs extrêmes~; le couple (\omterm{def:g11:stat:mean}{moyenne}, \omterm{def:g11:stat:variance}{écart type}) utilise
chaque valeur et alimente la théorie des \omterm{def:g10:proba:distribution}{probabilités}.
\end{method}

\begin{example}
Deux archers tirent chacun $10$ flèches~; les scores moyennent $8.5$
pour les deux, mais les \omterm{def:g11:stat:variance}{écarts types} sont $0.5$ et $2.1$. Même niveau en
\omterm{def:g11:stat:mean}{moyenne} --- mais le premier archer est bien plus régulier.
\end{example}

\section{Problème~: la cote z, une règle universelle}

\begin{problem}[{Devoir du week-end --- les écarts types comparent
les pommes et les oranges, l'inégalité de Bienaymé--Tchebychev
transforme $\sigma$ en garantie, et la moyenne gagne sa propre
médaille}]
\label{pb:g11:stat:1}
Un $15/20$ à un contrôle difficile vaut-il mieux qu'un $13/20$ à un
contrôle facile~? Un $23$ perdu dans une liste de petits nombres
est-il un accident ou une valeur aberrante~? Les nombres bruts ne
savent pas le dire --- mais divisés par le bon \omterm{def:g11:stat:variance}{écart type}, ils
parlent tous une même langue~: la \emph{\omterm{pb:g11:stat:1}{cote z}}. Ce problème
maîtrise la mécanique de la \omterm{def:g11:stat:variance}{variance}
(\cref{def:g11:stat:variance}, \cref{prop:g11:stat:shortcut}),
construit la règle universelle, et démontre la remarquable inégalité
qui fait de $\sigma$, de simple nombre descriptif, une garantie.

\medskip
\noindent\textbf{Partie I --- Mécanique de la \omterm{def:g11:stat:variance}{variance}.}
\begin{enumerate}
  \item Pour la série $4, 8, 6, 5, 2$~: calculer la \omterm{def:g11:stat:mean}{moyenne}, puis la
        \omterm{def:g11:stat:variance}{variance} directement à partir de la définition, puis
        $\sigma$.
  \item Recalculer la \omterm{def:g11:stat:variance}{variance} avec la formule raccourcie
        (\cref{prop:g11:stat:shortcut}) --- \omterm{def:g11:stat:mean}{moyenne} des carrés moins
        carré de la \omterm{def:g11:stat:mean}{moyenne} --- et vérifier l'accord.
  \item Avec des effectifs~: les notes $8$, $10$, $14$ obtenues par
        $3$, $5$, $2$ élèves. Calculer \omterm{def:g11:stat:mean}{moyenne}, \omterm{def:g11:stat:variance}{variance} et $\sigma$
        (deux décimales).
  \item Des températures ont pour \omterm{def:g11:stat:mean}{moyenne} $20$ et pour \omterm{def:g11:stat:variance}{écart type}
        $\sigma = 3$ (en degrés Celsius). Donner la \omterm{def:g11:stat:mean}{moyenne} et
        $\sigma$ en degrés Fahrenheit ($F = 1.8\,C + 32$~;
        \cref{prop:g11:stat:affine}).
  \item Démontrer les deux règles qui viennent de servir~: ajouter
        une constante $b$ à toutes les valeurs translate la \omterm{def:g11:stat:mean}{moyenne}
        de $b$ et laisse $\sigma$ inchangé~; multiplier par $a$
        multiplie la \omterm{def:g11:stat:mean}{moyenne} par $a$ et $\sigma$ par $\abs a$.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La règle universelle.}
La \emph{cote z}\index{cote z} d'une valeur $x$ dans une série de
\omterm{def:g11:stat:mean}{moyenne} $\bar x$ et d'\omterm{def:g11:stat:variance}{écart type} $\sigma$ est
$z = \dfrac{x - \bar x}{\sigma}$~: le nombre d'\omterm{def:g11:stat:variance}{écarts types} qui
séparent $x$ de la \omterm{def:g11:stat:mean}{moyenne}.
\begin{enumerate}[resume]
  \item Le contrôle A a pour \omterm{def:g11:stat:mean}{moyenne} $12$ et $\sigma = 2$~; le
        contrôle B a pour \omterm{def:g11:stat:mean}{moyenne} $8$ et $\sigma = 4$. Alice a
        obtenu $15$ en A, Bob $13$ en B. Calculer les deux cotes
        $z$~: quelle performance se détache le plus du groupe~?
  \item Calculer les cinq \omterm{pb:g11:stat:1}{cotes z} de la série de la question 1, et
        vérifier qu'elles ont pour \omterm{def:g11:stat:mean}{moyenne} $0$ et pour \omterm{def:g11:stat:variance}{écart type}
        $1$. Expliquer, à partir de la question 5, pourquoi la
        standardisation produit \emph{toujours} une \omterm{def:g11:stat:mean}{moyenne} $0$ et
        un $\sigma = 1$.
  \item Un test d'aptitude a pour \omterm{def:g11:stat:mean}{moyenne} $100$ et $\sigma = 15$.
        Quel score brut correspond à $z = 2$~? Quelle est la cote
        $z$ de $76$~?
  \item Un pédiatre note le poids d'un bébé sous la forme
        «~$z = -2.5$~». Dire ce que cela signifie, et pourquoi la
        croissance se suit en \omterm{pb:g11:stat:1}{cotes z} plutôt qu'en kilogrammes à
        mesure que l'enfant grandit.
  \item La valeur aberrante de l'\cref{exo:g11:stat:10} (série
        $1, 2, 2, 3, 3, 3, 4, 4, 5, 23$)~: calculer la \omterm{def:g11:stat:mean}{moyenne},
        $\sigma$, et la \omterm{pb:g11:stat:1}{cote z} de $23$. Le seuil d'alerte
        classique est $\abs z \geq 3$~: verdict~?
  \item Recalculer $\sigma$ \emph{sans} la valeur $23$, puis la cote
        $z$ qu'aurait $23$ face à la \omterm{def:g11:stat:mean}{moyenne} et à l'\omterm{def:g11:stat:variance}{écart type} des
        données assainies. Qu'a fait la valeur aberrante à la règle
        même censée la mesurer, et quelle parade
        l'\cref{exo:g11:stat:10} suggère-t-elle~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- La garantie de Tchebychev.}
\begin{enumerate}[resume]
  \item Démontrer l'inégalité de Bienaymé--Tchebychev pour une
        \omterm{def:g10:stats:series}{série statistique}~: si une proportion $p$ des valeurs
        vérifie $\abs{x - \bar x} \geq k\sigma$, alors, en ne
        gardant que ces termes dans la somme définissant la
        \omterm{def:g11:stat:variance}{variance},
        \[
        \sigma^2 \geq p \cdot (k\sigma)^2,
        \qquad\text{d'où}\qquad
        p \leq \frac{1}{k^2} .
        \]
  \item Que garantit l'inégalité sur la proportion d'une série
        quelconque située au-delà de $2\sigma$ de sa \omterm{def:g11:stat:mean}{moyenne}~?
        Au-delà de $3\sigma$~? Vérifier l'énoncé à $2\sigma$ sur la
        série de la question 1.
  \item Une usine produit des boulons de longueur \omterm{def:g11:stat:mean}{moyenne} $50$ mm
        avec $\sigma = 0.1$ mm. Que garantit Tchebychev sur la
        proportion de boulons hors de $\intcc{49.7}{50.3}$~? (La
        production réelle fait bien mieux --- la courbe en cloche de
        l'an prochain change ce prudent $\frac19$ en une fraction de
        pourcent.)
  \item Deux fonds rapportent tous deux $5\,\%$ par an en \omterm{def:g11:stat:mean}{moyenne},
        avec $\sigma = 2\,\%$ pour le fonds A et $\sigma = 12\,\%$
        pour le fonds B. Pour chacun, que dit Tchebychev de la
        proportion d'années dont le rendement est inférieur à
        $-19\,\%$~? Quel est le nom financier de $\sigma$~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La médaille de la \omterm{def:g11:stat:mean}{moyenne}, et un
avertissement.}
\begin{enumerate}[resume]
  \item Soit $f(c)$ la \omterm{def:g11:stat:mean}{moyenne} des carrés des écarts $(x_i - c)^2$ à
        un centre $c$ quelconque. Démontrer l'identité
        \[
        f(c) = \sigma^2 + (\bar x - c)^2 ,
        \]
        et conclure~: la \omterm{def:g11:stat:mean}{moyenne} est l'unique centre qui minimise
        l'écart quadratique moyen --- sa médaille, jumelle de la
        minimisation des écarts absolus par la \omterm{def:g11:stat:median}{médiane} au
        \cref{pb:g10:stats:1}.
  \item Vérifier l'identité sur la série de la question 1 avec
        $c = 6$~: calculer $f(6)$ directement, puis par la formule.
  \item Mêmes résumés, données différentes~: vérifier que
        $P = \{3, 5, 7, 9, 11\}$ et $Q = \{1, 7, 7, 7, 13\}$ ont la
        même \omterm{def:g11:stat:mean}{moyenne} mais pas la même \omterm{def:g11:stat:variance}{variance}~; puis construire une
        série de la forme $\{7 - a, 7, 7, 7, 7 + a\}$ ayant
        exactement la \omterm{def:g11:stat:mean}{moyenne} et la \omterm{def:g11:stat:variance}{variance} de $P$ --- et pourtant
        un profil bien à elle.
  \item Démontrer que $\sigma = 0$ force toutes les valeurs à être
        égales. (Que doit valoir chaque terme d'une somme de carrés
        nulle~?)
  \item Pour finir --- l'échelle du statisticien, une ligne par
        barreau~: les centres (\omterm{def:g11:stat:mean}{moyenne}, \omterm{def:g11:stat:median}{médiane}), les dispersions
        ($\sigma$, \omterm{def:g11:stat:quartiles}{écart interquartile}), la règle universelle ($z$),
        la garantie (Tchebychev), et l'avertissement de la question
        18~: les résumés compriment, et comprimer fait perdre ---
        avec la courbe en cloche de l'an prochain qui n'attend que
        de transformer les \omterm{pb:g11:stat:1}{cotes z} en \omterm{def:g10:proba:distribution}{probabilités} exactes.
\end{enumerate}
\end{problem}
