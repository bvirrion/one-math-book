\chapter{Trigonometría: la circunferencia unidad}\label{ch:g11:trigo}

La trigonometría empieza en los triángulos rectángulos, pero su verdadera
casa es la \emph{circunferencia unidad}, donde el \omterm{def:g11:trigo:cossin}{coseno} y el \omterm{def:g11:trigo:cossin}{seno} pasan a
ser funciones de un \omterm{def:g10:numbers:sets}{número real} cualquiera. Este capítulo instala el
\omterm{def:g11:trigo:radian}{radián}, el enrollamiento de la recta real sobre la circunferencia, los
valores exactos que hay que saberse de memoria y las simetrías que generan
todas las identidades clásicas. El estudio de $\cos$ y $\sin$ como
funciones (variación, \omterm{def:g11:deriv:derivative}{derivadas}) se lleva a cabo en el
\cref{ch:g12:trigo}.

\section{Radianes y enrollamiento}

\begin{definition}[Radián]\label{def:g11:trigo:radian}
Sea $\mathcal C$ la circunferencia de radio $1$ centrada en el origen $O$
y sea $I(1, 0)$. La medida en \emph{radianes}\index{radián} de un ángulo
central de $\mathcal C$ es la longitud del arco que recorta sobre
$\mathcal C$. Como la circunferencia completa mide $2\pi$:
\[
360^\circ = 2\pi \text{ rad},
\qquad
180^\circ = \pi \text{ rad},
\qquad
1 \text{ rad} = \frac{180^\circ}{\pi} \approx 57.3^\circ .
\]
\end{definition}

\begin{method}[Convertir]\label{met:g11:trigo:convert}
Los grados y los \omterm{def:g11:trigo:radian}{radianes} son proporcionales: multiplica por
$\frac{\pi}{180}$ para pasar de grados a \omterm{def:g11:trigo:radian}{radianes} y por
$\frac{180}{\pi}$ en sentido contrario. Por ejemplo,
$60^\circ = 60 \times \frac{\pi}{180} = \frac{\pi}{3}$ y
$\frac{3\pi}{4} = \frac{3 \times 180^\circ}{4} = 135^\circ$.
\end{method}

\begin{definition}[Enrollar la recta sobre la circunferencia]\label{def:g11:trigo:winding}
A cada \omterm{def:g10:numbers:sets}{número real} $t$ le asociamos el punto $M(t)$ de $\mathcal C$ que se
obtiene recorriendo una distancia $\abs t$ sobre la circunferencia desde
$I$, en sentido antihorario si $t \geq 0$ y en sentido horario si $t < 0$.
Como la longitud de la circunferencia es $2\pi$, los números $t$ y
$t + 2k\pi$ ($k \in \Z$) llegan al mismo punto:
$M(t + 2k\pi) = M(t)$.
\end{definition}

\begin{example}
$M(0) = I$; $M\!\left(\frac{\pi}{2}\right) = (0, 1)$, lo más alto de la
circunferencia; $M(\pi) = (-1, 0)$; $M\!\left(-\frac{\pi}{2}\right) =
(0, -1)$; y $M\!\left(\frac{9\pi}{4}\right) = M\!\left(\frac{\pi}{4} +
2\pi\right) = M\!\left(\frac{\pi}{4}\right)$.
\end{example}

\section{Coseno y seno}

\begin{definition}[Coseno y seno]\label{def:g11:trigo:cossin}
Para todo real $t$, el \emph{coseno}\index{coseno} y el
\emph{seno}\index{seno} de $t$ son las \omterm{def:g10:coordgeom:system}{coordenadas} del punto $M(t)$ de la
circunferencia unidad:
\[
M(t) = (\cos t,\ \sin t).
\]
\end{definition}

\begin{proposition}[Primeras propiedades]\label{prop:g11:trigo:first}
Para todos $t \in \R$ y $k \in \Z$:
\[
-1 \leq \cos t \leq 1, \qquad
-1 \leq \sin t \leq 1, \qquad
\cos^2 t + \sin^2 t = 1,
\]
\[
\cos(t + 2k\pi) = \cos t, \qquad \sin(t + 2k\pi) = \sin t .
\]
\end{proposition}

\begin{proof}
$M(t)$ está en la circunferencia unidad, así que sus \omterm{def:g10:coordgeom:system}{coordenadas} están
entre $-1$ y $1$ y cumplen la \omterm{def:g10:algebra:equation}{ecuación} de la circunferencia
$x^2 + y^2 = 1$, que es la identidad $\cos^2 t + \sin^2 t = 1$. La
periodicidad no es más que otra forma de escribir
$M(t + 2k\pi) = M(t)$ (\cref{def:g11:trigo:winding}).
\end{proof}

\begin{example}\label{ex:g11:trigo:identity}
Si $\sin t = \frac35$ y $t \in \intcc{\frac\pi2}{\pi}$ (segundo
cuadrante), entonces $\cos^2 t = 1 - \frac{9}{25} = \frac{16}{25}$, luego
$\cos t = \pm\frac45$; en el segundo cuadrante la \omterm{def:g10:coordgeom:system}{abscisa} es negativa, así
que $\cos t = -\frac45$.
\end{example}

Los valores que hay que saberse de memoria:
\begin{center}
\begin{tabular}{c|ccccc}
$t$ & $0$ & $\dfrac{\pi}{6}$ & $\dfrac{\pi}{4}$ & $\dfrac{\pi}{3}$ & $\dfrac{\pi}{2}$\\[6pt]
\hline\rule{0pt}{14pt}%
$\cos t$ & $1$ & $\dfrac{\sqrt3}{2}$ & $\dfrac{\sqrt2}{2}$ & $\dfrac12$ & $0$\\[6pt]
$\sin t$ & $0$ & $\dfrac12$ & $\dfrac{\sqrt2}{2}$ & $\dfrac{\sqrt3}{2}$ & $1$
\end{tabular}
\end{center}

\begin{remark}
Una ayuda para la memoria: los \omterm{def:g11:trigo:cossin}{senos} se leen
$\frac{\sqrt0}{2}, \frac{\sqrt1}{2}, \frac{\sqrt2}{2}, \frac{\sqrt3}{2},
\frac{\sqrt4}{2}$, y los \omterm{def:g11:trigo:cossin}{cosenos} son la misma lista al revés.
\end{remark}

\section{Ángulos asociados}

\begin{proposition}[Ángulos asociados]\label{prop:g11:trigo:associated}
\index{ángulos asociados}
Para todo $t \in \R$:
\[
\begin{aligned}
\cos(-t) &= \cos t, & \sin(-t) &= -\sin t,\\
\cos(\pi - t) &= -\cos t, & \sin(\pi - t) &= \sin t,\\
\cos(\pi + t) &= -\cos t, & \sin(\pi + t) &= -\sin t,\\
\cos\left(\tfrac{\pi}{2} - t\right) &= \sin t, &
\sin\left(\tfrac{\pi}{2} - t\right) &= \cos t .
\end{aligned}
\]
\end{proposition}

\begin{proof}
Cada línea expresa una simetría de la circunferencia.

$M(-t)$ es el simétrico de $M(t)$ respecto del eje $x$: la \omterm{def:g10:coordgeom:system}{abscisa} no
cambia y la ordenada cambia de signo.

$M(\pi - t)$ es el simétrico de $M(t)$ respecto del eje $y$: retroceder
$t$ desde $\pi$ lleva al punto opuesto en horizontal al que se alcanza
avanzando $t$ desde $0$. La \omterm{def:g10:coordgeom:system}{abscisa} cambia de signo y la ordenada no.

$M(\pi + t)$ es diametralmente opuesto a $M(t)$ (medio giro): las dos
\omterm{def:g10:coordgeom:system}{coordenadas} cambian de signo.

$M\!\left(\frac\pi2 - t\right)$ es el simétrico de $M(t)$ respecto de la
diagonal $y = x$, que intercambia las dos \omterm{def:g10:coordgeom:system}{coordenadas}.
\end{proof}

{\small Los cuatro puntos asociados: $M(-t)$ es el reflejo de $M(t)$

\begin{example}
$\cos\frac{2\pi}{3} = \cos\left(\pi - \frac\pi3\right)
= -\cos\frac\pi3 = -\frac12$, y
$\sin\frac{7\pi}{6} = \sin\left(\pi + \frac\pi6\right)
= -\sin\frac\pi6 = -\frac12$. La tabla de cinco valores, ampliada con las
simetrías, cubre toda la circunferencia.
\end{example}

\section{Resolución de ecuaciones trigonométricas}

\begin{theorem}[Las ecuaciones $\cos t = \cos a$ y $\sin t = \sin a$]
\label{thm:g11:trigo:equations}
Para \omterm{def:g10:numbers:sets}{números reales} $t$ y $a$:
\[
\cos t = \cos a \iff t = a + 2k\pi \ \text{o}\ t = -a + 2k\pi
\quad (k \in \Z),
\]
\[
\sin t = \sin a \iff t = a + 2k\pi \ \text{o}\ t = \pi - a + 2k\pi
\quad (k \in \Z).
\]
\end{theorem}

\begin{proof}
Dos puntos de la circunferencia unidad tienen la misma \omterm{def:g10:coordgeom:system}{abscisa} exactamente
cuando son iguales o simétricos respecto del eje $x$; por la
\cref{prop:g11:trigo:associated}, el simétrico de $M(a)$ es $M(-a)$. Así
que $\cos t = \cos a$ significa $M(t) = M(a)$ o $M(t) = M(-a)$, es decir,
$t = \pm a$ salvo un múltiplo de $2\pi$. De igual modo, ordenadas iguales
significan puntos iguales o simétricos respecto del eje $y$, y el
simétrico de $M(a)$ es $M(\pi - a)$.
\end{proof}

\begin{method}[Resolver $\cos t = c$ en un intervalo]\label{met:g11:trigo:solve}
\begin{enumerate}
  \item Halla \emph{un} ángulo $a$ con $\cos a = c$, a partir de la tabla
        de valores conocidos.
  \item Escribe las soluciones generales $t = \pm a + 2k\pi$
        (\cref{thm:g11:trigo:equations}).
  \item Elige los enteros $k$ que caen dentro del \omterm{def:g10:numbers:interval}{intervalo} pedido y
        enumera las soluciones.
\end{enumerate}
Procede igual con $\sin t = c$, con $t = a + 2k\pi$ o
$t = \pi - a + 2k\pi$.
\end{method}

\begin{example}\label{ex:g11:trigo:solve}
Resolvamos $\cos t = \frac12$ en $\intoc{-\pi}{\pi}$. Una solución de
$\cos a = \frac12$ es $a = \frac{\pi}{3}$. Las soluciones generales son
$t = \frac\pi3 + 2k\pi$ y $t = -\frac\pi3 + 2k\pi$. Dentro de
$\intoc{-\pi}{\pi}$ solo aporta $k = 0$:
$t \in \left\{-\frac\pi3,\ \frac\pi3\right\}$.

Resolvamos $\sin t = -\frac{\sqrt2}{2}$ en $\intco{0}{2\pi}$. Un ángulo es
$a = -\frac\pi4$; las soluciones son $t = -\frac\pi4 + 2k\pi$ y
$t = \pi + \frac\pi4 + 2k\pi = \frac{5\pi}{4} + 2k\pi$. En
$\intco{0}{2\pi}$: $t = \frac{7\pi}{4}$ (de $k = 1$) y
$t = \frac{5\pi}{4}$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:trigo:1}
Pasa a \omterm{def:g11:trigo:radian}{radianes}: $30^\circ$, $45^\circ$, $120^\circ$, $270^\circ$. Pasa a
grados: $\frac{\pi}{5}$, $\frac{5\pi}{6}$, $\frac{7\pi}{4}$,
$\frac{2\pi}{9}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:trigo:2}
Sitúa sobre la circunferencia unidad los puntos $M(t)$ para
\[
t = \frac{2\pi}{3},\quad
t = -\frac{\pi}{4},\quad
t = \frac{17\pi}{6},\quad
t = -\frac{7\pi}{2},
\]
después de reducir cada uno, módulo $2\pi$, a un valor de
$\intoc{-\pi}{\pi}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:trigo:3}
Usando la tabla y los \omterm{prop:g11:trigo:associated}{ángulos asociados}, da los valores exactos de
\[
\cos\frac{3\pi}{4}, \quad
\sin\frac{5\pi}{6}, \quad
\cos\left(-\frac{\pi}{3}\right), \quad
\sin\frac{4\pi}{3}, \quad
\cos\frac{11\pi}{6} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:trigo:4}
Sabiendo que $\cos t = \frac{5}{13}$ con
$t \in \intoo{-\frac\pi2}{0}$, calcula $\sin t$ de forma exacta.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:5}
Simplifica, para todo real $x$:
\[
A = \cos(\pi + x) + \cos(-x) + \sin\left(\frac\pi2 - x\right) - \cos x,
\qquad
B = \sin(\pi - x) + \sin(\pi + x) + \sin(-x) .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:6}
Resuelve en $\intoc{-\pi}{\pi}$:
\[
\cos t = \frac{\sqrt3}{2}, \qquad
\sin t = \frac12, \qquad
\cos t = -1 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:7}
Resuelve en $\intco{0}{2\pi}$:
\[
\sin t = -\frac{\sqrt3}{2}, \qquad
\cos t = 0, \qquad
2\sin t + 1 = 0 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:8}
Resuelve en $\R$ la \omterm{def:g10:algebra:equation}{ecuación} $\cos 2t = \cos t$. (Usa el
\cref{thm:g11:trigo:equations} con $a = t$ y discute las dos familias de
soluciones.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:9}
Demuestra que para todo real $x$,
\[
\bigl(\cos x + \sin x\bigr)^2 + \bigl(\cos x - \sin x\bigr)^2 = 2 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:10}
Una rueda de radio $30$ cm rueda sin deslizar. ¿Qué ángulo (en \omterm{def:g11:trigo:radian}{radianes} y
después en grados) gira cuando la bicicleta avanza $1.5$ m? ¿Cuánto avanza
la bicicleta en una vuelta completa de la rueda?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:trigo:11}
Resuelve en $\intoc{-\pi}{\pi}$ la inecuación $\cos t \leq \frac12$.
(Esboza la circunferencia unidad, marca la región donde la \omterm{def:g10:coordgeom:system}{abscisa} es como
mucho $\frac12$ y lee el \omterm{def:g10:numbers:interval}{intervalo} de valores de $t$.)
\end{exercise}

\section{Problema: La noria y la marea}

\begin{problem}[{Problema de fin de semana --- los radianes como distancia
rodada y la circunferencia unidad como reloj maestro de todo fenómeno
periódico}]
\label{pb:g11:trigo:1}
Una noria te lleva alrededor de un círculo; la marea sube y baja el agua
del puerto siguiendo ese mismo círculo matemático. Todo fenómeno periódico
(ruedas, mareas, sonido, estaciones) lee su horario en la circunferencia
unidad de este capítulo (\cref{def:g11:trigo:cossin}), y las ecuaciones
del \cref{met:g11:trigo:solve} dan los instantes exactos. Este problema
convierte, enrolla, modeliza y resuelve, y de paso deja claro por qué los
matemáticos miden los ángulos en \omterm{def:g11:trigo:radian}{radianes}.

\medskip
\noindent\textbf{Parte I --- Los \omterm{def:g11:trigo:radian}{radianes}, la distancia rodada.}
\begin{enumerate}
  \item Pasa a \omterm{def:g11:trigo:radian}{radianes}: $30^\circ$, $135^\circ$, $300^\circ$; y a grados:
        $\frac{3\pi}{4}$, $\frac{5\pi}{6}$
        (\cref{met:g11:trigo:convert}).
  \item La gracia de los \omterm{def:g11:trigo:radian}{radianes}: un arco de ángulo $t$ (en \omterm{def:g11:trigo:radian}{radianes})
        sobre una circunferencia de radio $r$ mide exactamente $r\,t$.
        ¿Qué arco recorta un ángulo de $2$ \omterm{def:g11:trigo:radian}{radianes} en una circunferencia
        de radio $3$ m? (Sin ningún $\pi$: de eso se trata.)
  \item Una rueda de radio $30$ cm rueda $1$ km sin deslizar. ¿Cuántos
        \emph{\omterm{def:g11:trigo:radian}{radianes}} ha girado y cuántas vueltas completas son?
  \item Sitúa sobre la circunferencia unidad y evalúa de forma exacta:
        $\cos\frac{2\pi}{3}$, $\sin\left(-\frac{\pi}{4}\right)$ y
        $\cos\frac{19\pi}{6}$ (reduce antes módulo $2\pi$;
        \cref{prop:g11:trigo:associated}).
  \item Un ángulo $t \in \intoo{\frac\pi2}{\pi}$ cumple
        $\sin t = \frac35$. Usando $\cos^2 t + \sin^2 t = 1$ y el
        cuadrante, halla $\cos t$ y $\tan t$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La noria.}
Una noria gigante tiene radio $60$ m, su eje está a $65$ m del suelo y da
una vuelta cada $30$ minutos. Subes en el punto más bajo en el instante
$t = 0$ (en minutos), así que tu altura cumple
\[
h(t) = 65 - 60 \cos\!\left(\frac{\pi t}{15}\right).
\]
\begin{enumerate}[resume]
  \item Justifica la fórmula: comprueba el ángulo girado al cabo de $t$
        minutos y la altura en $t = 0$, y explica el signo menos.
  \item Calcula tu altura en $t = 7.5$, $t = 15$ y $t = 20$ minutos
        (valores exactos).
  \item ¿En qué instante estás \emph{por primera vez} a $95$ m? (Resuelve
        $h(t) = 95$ con el \cref{met:g11:trigo:solve}.)
  \item ¿Durante qué \omterm{def:g10:numbers:interval}{intervalo} de tiempo del viaje estás al menos a $95$ m
        de altura, y cuánto dura en total? (Compara con el
        \cref{exo:g11:trigo:11}.)
  \item ¿Dónde cambia \emph{más deprisa} la altura? Compara la subida del
        primer minuto ($h(1) - h(0)$) con la subida entre los minutos $7$
        y $8$, y formula la observación (la herramienta que la vuelve
        exacta, la \omterm{def:g11:deriv:derivative}{derivada} de la sinusoide, llega el año que viene).
  \item Turno del ingeniero: diseña la fórmula de la altura de una noria
        de altura máxima $40$ m, altura mínima $4$ m y periodo $12$
        minutos, en la que se sube por abajo en $t = 0$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La marea y las simetrías de la
circunferencia.}
En un puerto, el calado en metros, $t$ horas después de medianoche, es
\[
d(t) = 7 + 3 \sin\!\left(\frac{\pi t}{6}\right).
\]
\begin{enumerate}[resume]
  \item Da el calado \omterm{def:g10:functions:extrema}{máximo} y el \omterm{def:g10:functions:extrema}{mínimo}, los instantes en que se producen
        por primera vez y el periodo de la marea.
  \item Un carguero necesita $8.5$ m de agua. Resuelve $d(t) \geq 8.5$ en
        $\intco{0}{12}$: ¿en qué ventana puede entrar y cuánto dura la
        ventana? ¿Cuándo se abre la siguiente?
  \item Vuelve a deducir dos identidades de \omterm{prop:g11:trigo:associated}{ángulos asociados}
        directamente de las simetrías de la circunferencia:
        $\sin(\pi - t) = \sin t$ (simetría respecto del eje vertical) y
        $\cos(\pi + t) = -\cos t$ (medio giro), y enuncia la identidad de
        los ángulos complementarios
        $\cos\left(\frac\pi2 - t\right) = \sin t$.
  \item Paridad (con el vocabulario del \cref{pb:g11:func:1}): clasifica
        $\sin$, $\cos$ y $t \mapsto \sin t + t^3$ como pares o \omterm{def:g11:func:parity}{impares},
        con demostraciones de una línea a partir de la circunferencia.
  \item Resuelve en $\intco{0}{2\pi}$: $2\sin^2 t - \sin t - 1 = 0$. (Es
        una \omterm{def:g10:algebra:equation}{ecuación} de segundo grado disfrazada: factorízala como en el
        \cref{pb:g11:quad:1} y lee después la circunferencia.)
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El ángulo del matemático.}
\begin{enumerate}[resume]
  \item ¿Cuántos grados es $1$ \omterm{def:g11:trigo:radian}{radián}? Y una trampa de calculadora: ¿qué
        es mayor, $\sin(1)$ (en modo \omterm{def:g11:trigo:radian}{radianes}) o $\sin(1^\circ)$, y por
        qué factor aproximadamente? ¿Moraleja sobre la configuración de la
        calculadora?
  \item Resuelve $\cos t = \sin t$ en $\intco{0}{2\pi}$, usando la
        circunferencia (¿dónde coinciden la \omterm{def:g10:coordgeom:system}{abscisa} y la ordenada?).
  \item Para ángulos pequeños, el arco y su \omterm{def:g11:trigo:cossin}{seno} casi coinciden: compara
        $\sin(0.1)$ con $0.1$ (cinco decimales) y da el error relativo.
        Esta ``\omterm{def:g10:numbers:approx}{aproximación} de ángulos pequeños'' gobierna barcos y
        péndulos; el teorema que hay detrás ($\frac{\sin t}{t} \to 1$) es
        una estrella del capítulo de límites del año que viene.
  \item Final: la circunferencia unidad como reloj maestro, una frase por
        punto: los \omterm{def:g11:trigo:radian}{radianes} convierten los ángulos en números honrados
        (arco $=$ radio $\times$ ángulo); el \omterm{def:g11:trigo:cossin}{coseno} y el \omterm{def:g11:trigo:cossin}{seno} son la
        sombra y la altura de la aguja del reloj; las simetrías de la
        circunferencia son las identidades; resolver $\cos t = c$ lee los
        horarios en el reloj; y todo fenómeno periódico (rueda, marea,
        sonido) es ese mismo reloj disfrazado.
\end{enumerate}
\end{problem}
