\chapter{Estadística descriptiva}\label{ch:g11:stat}

La estadística comprime una lista de números en unos pocos resúmenes con
sentido: dónde se sitúan los datos (\omterm{def:g11:stat:mean}{media}, \omterm{def:g11:stat:median}{mediana}) y hasta qué punto están
dispersos (\omterm{def:g11:stat:quartiles}{cuartiles}, \omterm{def:g11:stat:variance}{varianza}, \omterm{def:g11:stat:variance}{desviación típica}). Este capítulo lo
desarrolla todo sobre un mismo ejemplo. Los mismos resúmenes reaparecen
para las variables aleatorias en el \cref{ch:g11:prob}, y sus leyes
probabilísticas, en el \cref{ch:g12:sums}.

A lo largo del capítulo, nuestro conjunto de datos de referencia son las
notas (sobre $20$) de una clase de $12$ alumnos:
\[
8,\ 9,\ 10,\ 10,\ 11,\ 12,\ 12,\ 12,\ 13,\ 14,\ 16,\ 17 .
\]

\section{Media y mediana}

\begin{definition}[Media]\label{def:g11:stat:mean}
La \emph{media}\index{media} de los valores $x_1, x_2, \dots, x_n$ es
\[
\bar x = \frac{x_1 + x_2 + \dots + x_n}{n}.
\]
Si el valor $x_i$ aparece con frecuencia $n_i$ (con
$n_1 + \dots + n_k = n$), entonces
$\bar x = \frac{n_1 x_1 + \dots + n_k x_k}{n}$.
\end{definition}

\begin{definition}[Mediana]\label{def:g11:stat:median}
Una \emph{mediana}\index{mediana} de un conjunto de datos ordenado es un
valor que lo parte por la mitad: al menos la mitad de los valores son
$\leq$ que él y al menos la mitad son $\geq$ que él. En la práctica se
ordenan los $n$ valores; si $n$ es \omterm{def:g11:func:parity}{impar}, la mediana es el valor central;
si $n$ es par, se toma el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de los dos valores centrales.
\end{definition}

\begin{example}\label{ex:g11:stat:meanmedian}
Para nuestro conjunto de datos, la suma es $144$, luego
$\bar x = \frac{144}{12} = 12$. Los dos valores centrales (el $6$ y el $7$
de la lista ordenada) son los dos $12$, así que la \omterm{def:g11:stat:median}{mediana} también es
$12$. La \omterm{def:g11:stat:mean}{media} y la \omterm{def:g11:stat:median}{mediana} no tienen por qué coincidir: la \omterm{def:g11:stat:median}{mediana}
ignora \emph{cuán lejos} están los valores extremos y la \omterm{def:g11:stat:mean}{media}, no.
Sustituir la nota más alta, $17$, por $20$ mueve la \omterm{def:g11:stat:mean}{media} a $12.25$, pero
deja la \omterm{def:g11:stat:median}{mediana} en $12$.
\end{example}

\section{Cuartiles}

\begin{definition}[Cuartiles, recorrido intercuartílico]\label{def:g11:stat:quartiles}
El \emph{primer cuartil}\index{cuartil} $Q_1$ es el menor valor tal que al
menos una cuarta parte de los datos es $\leq Q_1$; el \emph{tercer
cuartil} $Q_3$ es el menor valor tal que al menos tres cuartas partes de
los datos son $\leq Q_3$. El \emph{recorrido
intercuartílico}\index{recorrido intercuartílico} es $Q_3 - Q_1$: la
anchura de la mitad central de los datos.
\end{definition}

\begin{example}
$n = 12$: la cuarta parte es $3$, así que $Q_1$ es el $3$.\textsuperscript{er}
valor ordenado: $Q_1 = 10$. Tres cuartas partes son $9$, así que $Q_3$ es
el $9$.\textsuperscript{o} valor: $Q_3 = 13$. El \omterm{def:g11:stat:quartiles}{recorrido intercuartílico}
es $3$: la mitad central de la clase cabe en $3$ puntos de nota.
\end{example}

\section{Varianza y desviación típica}

\begin{definition}[Varianza, desviación típica]\label{def:g11:stat:variance}
La \emph{varianza}\index{varianza} del conjunto de datos es la \omterm{def:g11:stat:mean}{media} de
los cuadrados de las desviaciones respecto de la \omterm{def:g11:stat:mean}{media}:
\[
V = \frac{1}{n}\sum_{i=1}^{n} (x_i - \bar x)^2,
\]
y la \emph{desviación típica}\index{desviación típica} es
$\sigma = \sqrt{V}$, expresada en la misma unidad que los datos.
\end{definition}

\begin{proposition}[Fórmula abreviada]\label{prop:g11:stat:shortcut}
La \omterm{def:g11:stat:variance}{varianza} es la \omterm{def:g11:stat:mean}{media} de los cuadrados menos el cuadrado de la \omterm{def:g11:stat:mean}{media}:
\[
V = \frac{1}{n}\sum_{i=1}^{n} x_i^2 \;-\; \bar x^{\,2}.
\]
\end{proposition}

\begin{proof}
Desarrollamos cada desviación al cuadrado:
$(x_i - \bar x)^2 = x_i^2 - 2\bar x\, x_i + \bar x^2$. Sumando y
dividiendo entre $n$:
\[
V = \frac1n \sum x_i^2 - 2\bar x \cdot \underbrace{\frac1n \sum
x_i}_{=\ \bar x} + \frac1n \cdot n\bar x^2
= \frac1n\sum x_i^2 - 2\bar x^2 + \bar x^2
= \frac1n\sum x_i^2 - \bar x^2 .
\]
La misma identidad, para variables aleatorias, es la
\cref{prop:g11:prob:konig}.
\end{proof}

\begin{example}\label{ex:g11:stat:variance}
Para nuestro conjunto de datos, las desviaciones respecto de
$\bar x = 12$ son $-4$, $-3$, $-2$, $-2$, $-1$, $0$, $0$, $0$, $1$, $2$,
$4$, $5$; sus cuadrados suman
\[
16 + 9 + 4 + 4 + 1 + 0 + 0 + 0 + 1 + 4 + 16 + 25 = 80 .
\]
Por tanto,
\[
V = \frac{80}{12} = \frac{20}{3} \approx 6.67,
\qquad
\sigma = \sqrt{20/3} \approx 2.6 \text{ puntos}.
\]
A grandes rasgos, una nota típica se aleja unos $2.6$ puntos de la \omterm{def:g11:stat:mean}{media}.
\end{example}

\begin{proposition}[Cambio de unidades]\label{prop:g11:stat:affine}
Si todos los valores se transforman mediante $y_i = a x_i + b$, entonces
\[
\bar y = a \bar x + b,
\qquad
V_y = a^2\, V_x,
\qquad
\sigma_y = \abs{a}\,\sigma_x .
\]
\end{proposition}

\begin{proof}
$\bar y = \frac1n \sum (a x_i + b) = a\left(\frac1n\sum x_i\right) +
\frac1n \cdot nb = a\bar x + b$. Después,
$y_i - \bar y = a(x_i - \bar x)$: cada desviación queda multiplicada por
$a$ (el desplazamiento $b$ se cancela), así que cada desviación al
cuadrado queda multiplicada por $a^2$, y su \omterm{def:g11:stat:mean}{media} también. Tomando raíces
cuadradas se obtiene $\sigma_y = \abs a\,\sigma_x$.
\end{proof}

\begin{example}
Unas temperaturas registradas en grados Celsius con \omterm{def:g11:stat:mean}{media} $20$ y
\omterm{def:g11:stat:variance}{desviación típica} $3$ se pasan a Fahrenheit con $F = 1.8\,C + 32$: la
\omterm{def:g11:stat:mean}{media} pasa a ser $1.8 \times 20 + 32 = 68$ y la \omterm{def:g11:stat:variance}{desviación típica},
$1.8 \times 3 = 5.4$. Desplazar un conjunto de datos no cambia su
dispersión; cambiarle la escala se la cambia también a ella.
\end{example}

\begin{method}[Resumir y comparar conjuntos de datos]\label{met:g11:stat:compare}
Para comparar dos conjuntos de datos, pon uno junto a otro: la \omterm{def:g11:stat:mean}{media} (o la
\omterm{def:g11:stat:median}{mediana}) para la \emph{posición} y la \omterm{def:g11:stat:variance}{desviación típica} (o el \omterm{def:g11:stat:quartiles}{recorrido
intercuartílico}) para la \emph{dispersión}. La pareja (\omterm{def:g11:stat:median}{mediana}, \omterm{def:g11:stat:quartiles}{recorrido
intercuartílico}) es robusta frente a los valores extremos; la pareja
(\omterm{def:g11:stat:mean}{media}, \omterm{def:g11:stat:variance}{desviación típica}) usa todos los valores y alimenta la teoría de
la \omterm{def:g10:proba:distribution}{probabilidad}.
\end{method}

\begin{example}
Dos arqueros lanzan $10$ flechas cada uno; las puntuaciones promedian
$8.5$ en los dos casos, pero las desviaciones típicas son $0.5$ y $2.1$.
El mismo nivel de \omterm{def:g11:stat:mean}{media}, pero el primer arquero es mucho más regular.
\end{example}

\section{Problema: La puntuación z, una regla universal}

\begin{problem}[{Problema de fin de semana --- las desviaciones típicas
comparan peras con manzanas, la desigualdad de Chebyshev convierte
$\sigma$ en una garantía y la media se gana su propia medalla}]
\label{pb:g11:stat:1}
¿Es mejor un $15$ sobre $20$ en un examen difícil que un $13$ sobre $20$
en uno fácil? ¿Un $23$ en una lista de números pequeños es un accidente o
un valor atípico? Los números en bruto no lo pueden decir, pero divididos
entre la \omterm{def:g11:stat:variance}{desviación típica} adecuada hablan todos un mismo idioma: la
\emph{\omterm{pb:g11:stat:1}{puntuación z}}. Este problema domina la maquinaria de la \omterm{def:g11:stat:variance}{varianza}
(\cref{def:g11:stat:variance}, \cref{prop:g11:stat:shortcut}), construye
la regla universal y demuestra la notable desigualdad que convierte
$\sigma$ de número descriptivo en garantía.

\medskip
\noindent\textbf{Parte I --- Mecánica de la \omterm{def:g11:stat:variance}{varianza}.}
\begin{enumerate}
  \item Para el conjunto de datos $4, 8, 6, 5, 2$: calcula la \omterm{def:g11:stat:mean}{media},
        después la \omterm{def:g11:stat:variance}{varianza} directamente desde la definición y después
        $\sigma$.
  \item Vuelve a calcular la \omterm{def:g11:stat:variance}{varianza} con la fórmula abreviada
        (\cref{prop:g11:stat:shortcut}), la \omterm{def:g11:stat:mean}{media} de los cuadrados menos
        el cuadrado de la \omterm{def:g11:stat:mean}{media}, y comprueba que coinciden.
  \item Con frecuencias: notas $8$, $10$, $14$ obtenidas por $3$, $5$ y
        $2$ alumnos. Calcula la \omterm{def:g11:stat:mean}{media}, la \omterm{def:g11:stat:variance}{varianza} y $\sigma$ (con dos
        decimales).
  \item Unas temperaturas tienen \omterm{def:g11:stat:mean}{media} $20$ y $\sigma = 3$ (en grados
        Celsius). Da la \omterm{def:g11:stat:mean}{media} y $\sigma$ en Fahrenheit
        ($F = 1.8\,C + 32$; \cref{prop:g11:stat:affine}).
  \item Demuestra las dos reglas que acabas de usar: sumar una constante
        $b$ a todos los valores desplaza la \omterm{def:g11:stat:mean}{media} en $b$ y deja $\sigma$
        igual; multiplicar por $a$ multiplica la \omterm{def:g11:stat:mean}{media} por $a$ y $\sigma$
        por $\abs a$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La regla universal.}
La \emph{puntuación z}\index{puntuación z} de un valor $x$ en un conjunto
de datos de \omterm{def:g11:stat:mean}{media} $\bar x$ y \omterm{def:g11:stat:variance}{desviación típica} $\sigma$ es
$z = \dfrac{x - \bar x}{\sigma}$: el número de desviaciones típicas que
separan $x$ de la \omterm{def:g11:stat:mean}{media}.
\begin{enumerate}[resume]
  \item El examen A tiene \omterm{def:g11:stat:mean}{media} $12$ y $\sigma = 2$; el examen B tiene
        \omterm{def:g11:stat:mean}{media} $8$ y $\sigma = 4$. Alicia sacó un $15$ en A y Bruno un $13$
        en B. Calcula las dos puntuaciones z: ¿cuál de las dos actuaciones
        destaca más sobre el grupo?
  \item Calcula las cinco puntuaciones z del conjunto de datos de la
        pregunta 1 y comprueba que tienen \omterm{def:g11:stat:mean}{media} $0$ y \omterm{def:g11:stat:variance}{desviación típica}
        $1$. Explica, a partir de la pregunta 5, por qué la
        estandarización produce \emph{siempre} \omterm{def:g11:stat:mean}{media} $0$ y $\sigma = 1$.
  \item Un test de aptitud tiene \omterm{def:g11:stat:mean}{media} $100$ y $\sigma = 15$. ¿Qué
        puntuación bruta tiene $z = 2$? ¿Cuál es la \omterm{pb:g11:stat:1}{puntuación z} de $76$?
  \item Un pediatra anota el peso de un bebé como ``$z = -2.5$''. Explica
        qué significa eso y por qué el crecimiento se sigue en
        puntuaciones z y no en kilogramos a medida que el niño crece.
  \item El valor atípico del \cref{exo:g11:stat:10} (conjunto
        $1, 2, 2, 3, 3, 3, 4, 4, 5, 23$): calcula la \omterm{def:g11:stat:mean}{media}, $\sigma$ y la
        \omterm{pb:g11:stat:1}{puntuación z} de $23$. El umbral de alarma clásico es
        $\abs z \geq 3$: ¿veredicto?
  \item Vuelve a calcular $\sigma$ \emph{sin} el valor $23$, y la
        \omterm{pb:g11:stat:1}{puntuación z} que tendría $23$ frente a la \omterm{def:g11:stat:mean}{media} y la $\sigma$ de
        los datos limpios. ¿Qué le ha hecho el valor atípico a la propia
        regla que debía medirlo, y qué contramedida sugiere el
        \cref{exo:g11:stat:10}?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La garantía de Chebyshev.}
\begin{enumerate}[resume]
  \item Demuestra la desigualdad de Bienaymé--Chebyshev para un conjunto
        de datos: si una proporción $p$ de los valores cumple
        $\abs{x - \bar x} \geq k\sigma$, entonces, conservando solo esos
        términos en la suma de la \omterm{def:g11:stat:variance}{varianza},
        \[
        \sigma^2 \geq p \cdot (k\sigma)^2,
        \qquad\text{luego}\qquad
        p \leq \frac{1}{k^2} .
        \]
  \item ¿Qué garantiza la desigualdad sobre la proporción de un conjunto
        de datos cualquiera más allá de $2\sigma$ de su \omterm{def:g11:stat:mean}{media}? ¿Y más allá
        de $3\sigma$? Comprueba lo de $2\sigma$ con el conjunto de datos
        de la pregunta 1.
  \item Una fábrica produce tornillos de \omterm{def:g11:stat:mean}{media} $50$ mm y
        $\sigma = 0.1$ mm. ¿Qué garantiza Chebyshev sobre la proporción de
        tornillos fuera de $\intcc{49.7}{50.3}$? (La producción real lo
        hace mucho mejor: la campana del año que viene convierte este
        prudente $\frac19$ en una fracción de punto porcentual.)
  \item Dos fondos rinden de \omterm{def:g11:stat:mean}{media} un $5\,\%$ anual, con $\sigma = 2\,\%$
        para el fondo A y $\sigma = 12\,\%$ para el fondo B. Para cada
        fondo, ¿qué dice Chebyshev sobre la proporción de años con
        rendimientos por debajo del $-19\,\%$? ¿Cuál es el nombre
        financiero de $\sigma$?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La medalla de la \omterm{def:g11:stat:mean}{media}, y un aviso.}
\begin{enumerate}[resume]
  \item Sea $f(c)$ la \omterm{def:g11:stat:mean}{media} de las desviaciones al cuadrado
        $(x_i - c)^2$ respecto de un centro arbitrario $c$. Demuestra la
        identidad
        \[
        f(c) = \sigma^2 + (\bar x - c)^2 ,
        \]
        y concluye: la \omterm{def:g11:stat:mean}{media} es el único punto que minimiza la desviación
        cuadrática \omterm{def:g11:stat:mean}{media}; esa es su medalla, pareja de la minimización de
        las desviaciones en \omterm{def:g10:numbers:abs}{valor absoluto} que logra la \omterm{def:g11:stat:median}{mediana} en el
        \cref{pb:g10:stats:1}.
  \item Comprueba la identidad con el conjunto de datos de la pregunta 1 y
        $c = 6$: calcula $f(6)$ directamente y con la fórmula.
  \item Mismos resúmenes, datos distintos: comprueba que
        $P = \{3, 5, 7, 9, 11\}$ y $Q = \{1, 7, 7, 7, 13\}$ comparten la
        \omterm{def:g11:stat:mean}{media}, pero no la \omterm{def:g11:stat:variance}{varianza}; y construye después un conjunto de la
        forma $\{7 - a, 7, 7, 7, 7 + a\}$ con exactamente la \omterm{def:g11:stat:mean}{media} y la
        \omterm{def:g11:stat:variance}{varianza} de $P$, pero con una forma completamente propia.
  \item Demuestra que $\sigma = 0$ obliga a que todos los valores sean
        iguales. (¿Cómo tiene que ser cada término de una suma de
        cuadrados que vale cero?)
  \item Final: la escalera del estadístico, una línea por peldaño: los
        centros (\omterm{def:g11:stat:mean}{media}, \omterm{def:g11:stat:median}{mediana}), las dispersiones ($\sigma$, \omterm{def:g11:stat:quartiles}{recorrido
        intercuartílico}), la regla universal ($z$), la garantía
        (Chebyshev) y el aviso de la pregunta 18: los resúmenes comprimen,
        y comprimir pierde información; la campana del año que viene
        espera para convertir las puntuaciones z en \omterm{def:g10:proba:distribution}{probabilidades}
        exactas.
\end{enumerate}
\end{problem}
