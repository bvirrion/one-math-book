\chapter{Estadística descriptiva}\label{ch:g11:stat}

Estadísticas comprime una lista de números en algunos resúmenes significativos:
donde se encuentran los datos (\omterm{def:g11:stat:mean}{media}, \omterm{def:g11:stat:median}{mediana}), y qué tan extendido está (cuartiles,
varianza, desviación estándar). Este capítulo resuelve todo en uno.
ejemplo de ejecución. Los mismos resúmenes reaparecen para variables aleatorias en
\cref{ch:g11:prob}, y sus leyes probabilísticas en \cref{ch:g12:sums}.

A lo largo del capítulo, nuestro conjunto de datos en ejecución son las marcas (fuera de $20$) de
una clase de $12$ estudiantes:
\[
8,\ 9,\ 10,\ 10,\ 11,\ 12,\ 12,\ 12,\ 13,\ 14,\ 16,\ 17 .
\]

\section{Media y mediana}

\begin{definition}[Media]\label{def:g11:stat:mean}
La \emph{media}\index{media} de los valores $ x_1, x_2, \dots, x_n $ es
\[
\bar x = \frac{x_1 + x_2 + \dots + x_n}{n}.
\]
si el valor $ x_i $ aparece con \omterm{def:g10:stats:series}{frecuencia} $ n_i $ (con
$ n_1 + \dots + n_k = n $), entonces
$\bar x = \frac{n_1 x_1 + \dots + n_k x_k}{n}$.
\end{definition}

\begin{definition}[Mediana]\label{def:g11:stat:median}
A \emph{mediana}\index{mediana} de un conjunto de datos ordenados es un valor que se divide
a la mitad: al menos la mitad de los valores son $\leq $ y al menos la mitad lo son
$\geq $ él. En la práctica, ordenar los $ n $ valores; si $ n $ es \omterm{def:g11:func:parity}{impar}, la mediana es
el valor medio; si $ n $ es par, toma el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de los dos medios
valores.
\end{definition}

\begin{example}\label{ex:g11:stat:meanmedian}
Para el conjunto de datos en ejecución, la suma es $144$, entonces $\bar x = \frac{144}{12} =
12$. Los dos valores medios ($6$ th y $7$º de la lista ordenada) son ambos
$12$, entonces la \omterm{def:g11:stat:median}{mediana} es $12$ también. \omterm{def:g11:stat:mean}{media} y \omterm{def:g11:stat:median}{mediana} no tiene por qué coincidir: el
\omterm{def:g11:stat:median}{mediana} ignora \emph{Hasta dónde} los valores extremos son, la \omterm{def:g11:stat:mean}{media} no lo hace.
Reemplazo de la marca superior $17$ por $20$ mueve la \omterm{def:g11:stat:mean}{media} a $12.25$ pero se va
la \omterm{def:g11:stat:median}{mediana} en $12$.
\end{example}

\section{Cuartiles}

\begin{definition}[Cuartiles, rango intercuartil]\label{def:g11:stat:quartiles}
El \emph{primer cuartil}\index{primer cuartil} $ Q_1$ es el valor más pequeño tal
que al menos una cuarta parte de los datos son $\leq Q_1$; el \emph{tercero
cuartil} $ Q_3$ es el valor más pequeño tal que al menos tres cuartos de
los datos son $\leq Q_3$. El \emph{rango intercuartil}\index{rango intercuartil} es $ Q_3 - Q_1$: el
ancho de la mitad central de los datos.
\end{definition}

\begin{example}
$ n = 12$: un cuarto es $3$, entonces $ Q_1$ es el $3$ Tercer valor ordenado:
$Q_1 = 10$. Tres cuartos son $9$, entonces $ Q_3$ es el $9$ ésimo valor:
$Q_3 = 13$. El \omterm{def:g10:stats:quartiles}{rango} intercuartílico es $3$: la mitad central de la clase
se sienta dentro $3$ marcas.
\end{example}

\section{Varianza y desviación estándar}

\begin{definition}[Varianza, desviación estándar]\label{def:g11:stat:variance}
El \emph{diferencia}\index{diferencia} del conjunto de datos es la \omterm{def:g11:stat:mean}{media} de la
desviaciones al cuadrado de la \omterm{def:g11:stat:mean}{media}:
\[
V = \frac{1}{n}\sum_{i=1}^{n} (x_i - \bar x)^2,
\]
y el \emph{estándard}\index{estándard} es
$\sigma = \sqrt{V}$, expresado en la misma unidad que los datos.
\end{definition}

\begin{proposition}[Fórmula de atajo]\label{prop:g11:stat:shortcut}
La varianza es la \omterm{def:g11:stat:mean}{media} de los cuadrados menos el cuadrado de la \omterm{def:g11:stat:mean}{media}:
\[
V = \frac{1}{n}\sum_{i=1}^{n} x_i^2 \;-\; \bar x^{\,2}.
\]
\end{proposition}

\begin{proof}
Expande cada desviación al cuadrado:
$(x_i - \bar x)^2 = x_i^2 - 2\bar x\, x_i + \bar x^2$. sumando y
dividiendo por $ n $:
\[
V = \frac1n \sum x_i^2 - 2\bar x \cdot \underbrace{\frac1n \sum
x_i}_{=\ \bar x} + \frac1n \cdot n\bar x^2
= \frac1n\sum x_i^2 - 2\bar x^2 + \bar x^2
= \frac1n\sum x_i^2 - \bar x^2 .
\]
La misma identidad, para variables aleatorias, es
\cref{prop:g11:prob:konig}.
\end{proof}

\begin{example}\label{ex:g11:stat:variance}
Para el conjunto de datos en ejecución, las desviaciones de $\bar x = 12$ son $-4$,
$-3$, $-2$, $-2$, $-1$, $0$, $0$, $0$, $1$, $2$, $4$, $5$; sus cuadrados
suma a
\[
16 + 9 + 4 + 4 + 1 + 0 + 0 + 0 + 1 + 4 + 16 + 25 = 80 .
\]
Por eso
\[
V = \frac{80}{12} = \frac{20}{3} \approx 6.67,
\qquad
\sigma = \sqrt{20/3} \approx 2.6 \text{ marks}.
\]
En términos generales, una marca típica se sitúa aproximadamente $2.6$ marcas alejadas de la
\omterm{def:g11:stat:mean}{media}.
\end{example}

\begin{proposition}[Cambio de unidades]\label{prop:g11:stat:affine}
Si cada valor se transforma por $ y_i = a x_i + b $, entonces
\[
\bar y = a \bar x + b,
\qquad
V_y = a^2\, V_x,
\qquad
\sigma_y = \abs{a}\,\sigma_x .
\]
\end{proposition}

\begin{proof}
$\bar y = \frac1n \sum (a x_i + b) = a\left(\frac1n\sum x_i\right) +
\frac1n \cdot nb = a\bar x + b $. Entonces
$ y_i - \bar y = a(x_i - \bar x)$: cada desviación se escala por $ a $
(el turno $ b $ se cancela), por lo que cada desviación al cuadrado se escala por $ a^2$,
y también lo es su \omterm{def:g11:stat:mean}{media}. Sacar raíces cuadradas da
$\sigma_y = \abs a\,\sigma_x $.
\end{proof}

\begin{example}
Temperaturas registradas en grados Celsius con \omterm{def:g11:stat:mean}{media} $20$ y desviación estándar $3$ convertir a Fahrenheit por $ F = 1.8\,C + 32$: la \omterm{def:g11:stat:mean}{media}
se convierte $1.8 \times 20 + 32 = 68$, el desviación estándar
$1.8 \times 3 = 5.4$. Cambiar un conjunto de datos no cambia su distribución;
reescalarlo reescala la extensión.
\end{example}

\begin{method}[Resumir y comparar conjuntos de datos]\label{met:g11:stat:compare}
Para comparar dos conjuntos de datos, colóquelos uno al lado del otro: \omterm{def:g11:stat:mean}{media} (o \omterm{def:g11:stat:median}{mediana}) para
\emph{posición}, desviación estándar (o \omterm{def:g10:stats:quartiles}{rango} intercuartílico) para
\emph{dispersión}. la pareja (\omterm{def:g11:stat:median}{mediana}, \omterm{def:g10:stats:quartiles}{rango} intercuartílico) es resistente a
valores extremos; la pareja (\omterm{def:g11:stat:mean}{media}, desviación estándar) utiliza todos los valores y
alimenta \omterm{def:g10:proba:distribution}{probabilidad} teoría.
\end{method}

\begin{example}
Dos arqueros cada tiro. $10$ flechas; puntaje promedio $8.5$ para ambos, pero
el desviaciones estándar son $0.5$ y $2.1$. Mismo nivel en promedio --- pero
el primer arquero es mucho más consistente.
\end{example}

\section{Problema: el puntaje z, una regla universal}

\begin{problem}[{Problema de fin de semana --- comparación de desviaciones estándar
Manzanas con naranjas, la desigualdad de Chebyshev cambia $\sigma $ en un
garantía, y la media gana su propia medalla}]
\label{pb:g11:stat:1}
Es $15/20$ en un examen difícil mejor que $13/20$ en uno fácil? es
un $23$ en una lista de números pequeños, ¿un accidente o un valor atípico? crudo
los números no pueden decir --- pero divididos por la derecha desviación estándar todos hablan un idioma, el \emph{\omterm{pb:g11:stat:1}{puntuación z}}. esto
el problema domina la varianza maquinaria
(\cref{def:g11:stat:variance},
\cref{prop:g11:stat:shortcut}), construye el gobernante universal, y
prueba la notable desigualdad que convierte $\sigma $ de un
número descriptivo en una garantía.

\medskip
\noindent\textbf{Part I --- varianza mechanics.}
\begin{enumerate}
  \item Para el conjunto de datos $4, 8, 6, 5, 2$: calcular la \omterm{def:g11:stat:mean}{media}, entonces
        la varianza directamente de la definición, entonces
        $\sigma $.
  \item Vuelva a calcular la varianza con la fórmula abreviada
        (\cref{prop:g11:stat:shortcut}) --- \omterm{def:g11:stat:mean}{media} de cuadrados
        menos cuadrado de \omterm{def:g11:stat:mean}{media} --- y comprobar el acuerdo.
  \item Con \omterm{def:g10:stats:series}{frecuencias}: marcas $8$, $10$, $14$ obtenido por
        $3$, $5$, $2$ estudiantes. Calcular \omterm{def:g11:stat:mean}{media}, varianza y
        $\sigma $ (dos decimales).
  \item Las temperaturas han \omterm{def:g11:stat:mean}{media} $20$ y $\sigma = 3$ (grados
        Celsius). Dale la \omterm{def:g11:stat:mean}{media} y $\sigma $ en grados Fahrenheit
        ($ F = 1.8\,C + 32$; \cref{prop:g11:stat:affine}).
  \item Demuestre las dos reglas que acabamos de usar: sumar una constante $ b $ a
        cada valor cambia la \omterm{def:g11:stat:mean}{media} por $ b $ y hojas $\sigma $
        sin alterar; multiplicando por $ a $ multiplica la \omterm{def:g11:stat:mean}{media} por $ a $
        y $\sigma $ por $\abs a $.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The universal ruler.}
El \emph{puntuación z}\index{puntuación z} de un valor $ x $ en un conjunto de datos de
\omterm{def:g11:stat:mean}{media} $\bar x $ y desviación estándar $\sigma $ es
$ z = \dfrac{x - \bar x}{\sigma}$: el número de estándar
desviaciones separando $ x $ desde \omterm{def:g11:stat:mean}{media}.
\begin{enumerate}[resume]
  \item El examen A tiene \omterm{def:g11:stat:mean}{media} $12$ y $\sigma = 2$; el examen B tiene \omterm{def:g11:stat:mean}{media}
        $8$ y $\sigma = 4$. Alicia anotó $15$ en A, Bob $13$
        en B. Calcular ambos Defectuoso z: cuyo desempeño se destaca
        ¿Más por encima de la multitud?
  \item Calcula los cinco Defectuoso z del conjunto de datos de la pregunta 1,
        y verificar que tengan \omterm{def:g11:stat:mean}{media} $0$ y desviación estándar $1$. Explique a partir de la pregunta 5 por qué
        estandarización \emph{siempre} produce \omterm{def:g11:stat:mean}{media} $0$ y
        $\sigma = 1$.
  \item Una prueba de aptitud tiene \omterm{def:g11:stat:mean}{media} $100$ y $\sigma = 15$. que
        puntuación bruta tiene $ z = 2$? cual es el \omterm{pb:g11:stat:1}{puntuación z} de $76$?
  \item Un pediatra registra el peso de un bebé como
        ``$ z = -2.5$''. Diga qué significa esto y por qué el crecimiento es
        rastreado en Defectuoso z en lugar de kilogramos como el niño
        edades.
  \item El valor atípico de \cref{exo:g11:stat:10} (conjunto de datos
        $1, 2, 2, 3, 3, 3, 4, 4, 5, 23$): calcula la \omterm{def:g11:stat:mean}{media},
        $\sigma $, y el \omterm{pb:g11:stat:1}{puntuación z} de $23$. La alarma clásica
        el umbral es $\abs z \geq 3$: veredicto?
  \item Recalcular $\sigma $ \emph{sin} el valor $23$, y
        el \omterm{pb:g11:stat:1}{puntuación z} eso $23$ tendría contra la limpia
        datos \omterm{def:g11:stat:mean}{media} y $\sigma $. ¿Qué hizo el caso atípico para
        el mismo gobernante pretendía medirlo, y ¿qué significa
        \cref{exo:g11:stat:10} sugerir como contramedida?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Chebyshev's guarantee.}
\begin{enumerate}[resume]
  \item Probar el Bienaymé--desigualdad de Chebyshev para a
        conjunto de datos: si es una proporción $ p $ de los valores satisfacen
        $\abs{x - \bar x} \geq k\sigma $, entonces, manteniendo sólo
        esos términos en la varianza suma,
        \[
        \sigma^2 \geq p \cdot (k\sigma)^2,
        \qquad\text{hence}\qquad
        p \leq \frac{1}{k^2} .
        \]
  \item ¿Qué garantiza la desigualdad sobre la proporción?
        de cualquier conjunto de datos más allá $2\sigma $ de su \omterm{def:g11:stat:mean}{media}? Más allá de
        $3\sigma $? Compruebe el $2\sigma $ declaración sobre el conjunto de datos
        de la pregunta 1.
  \item Una fábrica produce pernos con \omterm{def:g11:stat:mean}{media} $50$ mm y
        $\sigma = 0.1$ mm. ¿Qué garantiza Chebyshev sobre
        la proporción de tornillos fuera
        $\intcc{49.7}{50.3}$? (La producción real funciona mucho mejor
        --- la curva de campana del grado 12 se vuelve así de cautelosa
        $\frac19$ en una fracción de un por ciento.)
  \item Dos fondos regresan $5\,\%$ por año en promedio, con
        $\sigma = 2\,\%$ para el fondo A y $\sigma = 12\,\%$ para
        fondo B. Para cada fondo, ¿qué dice Chebyshev sobre
        la proporción similar a la \omterm{def:g10:proba:distribution}{probabilidad} de años con rendimientos
        abajo $-19\,\%$? Qué es $\sigma $¿El nombre financiero de?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The \omterm{def:g11:stat:mean}{media}'s medal, and a warning.}
\begin{enumerate}[resume]
  \item Sea $ f(c)$ ser la \omterm{def:g11:stat:mean}{media} de las desviaciones al cuadrado
        $(x_i - c)^2$ desde un centro arbitrario $ c $. demostrar el
        identidad
        \[
        f(c) = \sigma^2 + (\bar x - c)^2 ,
        \]
        y concluyo: la \omterm{def:g11:stat:mean}{media} es el único minimizador de \omterm{def:g11:stat:mean}{media}
        desviación al cuadrado --- su medalla, que coincide con la \omterm{def:g11:stat:median}{mediana}'s
        minimización de las desviaciones absolutas en
        \cref{pb:g10:stats:1}.
  \item Verifique la identidad en el conjunto de datos de la pregunta 1 con
        $ c = 6$: calcular $ f(6)$ directamente y a través de la fórmula.
  \item Mismos resúmenes, diferentes datos: comprueba que
        $ P = \{3, 5, 7, 9, 11\}$ y $ Q = \{1, 7, 7, 7, 13\}$
        compartir lo mismo \omterm{def:g11:stat:mean}{media} pero no lo mismo varianza; entonces
        construir un conjunto de datos de la forma
        $\{7 - a, 7, 7, 7, 7 + a\}$ con exactamente la \omterm{def:g11:stat:mean}{media} y
        varianza de $ P $ --- pero con una forma propia.
  \item demostrar que $\sigma = 0$ obliga a todos los valores a ser iguales. (¿Qué
        ¿Debe serlo cada término de una suma cero de cuadrados?)
  \item Final --- la escalera del estadístico, una línea por
        peldaño: centros (\omterm{def:g11:stat:mean}{media}, \omterm{def:g11:stat:median}{mediana}), se extiende ($\sigma $, RIQ),
        el gobernante universal ($ z$), la garantía (Chebyshev),
        y la advertencia de la pregunta 18: los resúmenes se comprimen,
        y la compresión pierde --- con la curva de campana de grado
        12 esperando para girar Defectuoso z en \omterm{def:g10:proba:distribution}{probabilidades} exactas.
\end{enumerate}
\end{problem}
