\chapter{El producto escalar en el plano}\label{ch:g11:scal}

El \omterm{def:g11:scal:dot}{producto escalar} asocia un \emph{número} a una pareja de \omterm{def:g11:vect:vector}{vectores}, y ese
número ve geometría: se anula exactamente para direcciones perpendiculares
y mide ángulos y longitudes. Es el motor del teorema del \omterm{def:g11:trigo:cossin}{coseno}, de las
ecuaciones de circunferencias y de los cálculos de perpendicularidad. La
misma herramienta, en el espacio, es el tema del \cref{ch:g12:space}.

\section{Definición y primeros cálculos}

\begin{definition}[Producto escalar]\label{def:g11:scal:dot}
El \emph{producto escalar}\index{producto escalar} de dos \omterm{def:g11:vect:vector}{vectores}
$\vec u$ y $\vec v$ es el número
\[
\vec u \cdot \vec v
= \tfrac12\left(\norm{\vec u + \vec v}^2 - \norm{\vec u}^2
- \norm{\vec v}^2\right),
\]
donde $\norm{\vec u}$ designa la longitud (o \emph{norma}\index{norma}) de
$\vec u$.
\end{definition}

\begin{theorem}[Las cuatro caras del producto escalar]\label{thm:g11:scal:faces}
Sean $\vec u\,(x, y)$ y $\vec v\,(x', y')$ \omterm{def:g11:vect:vector}{vectores} no nulos y sea
$\theta$ el ángulo que forman. Entonces
\begin{enumerate}
  \item \emph{(\omterm{def:g10:coordgeom:system}{coordenadas})} $\vec u \cdot \vec v = xx' + yy'$;
  \item \emph{(\omterm{def:g11:trigo:cossin}{coseno})} $\vec u \cdot \vec v =
        \norm{\vec u}\,\norm{\vec v}\cos\theta$;
  \item \emph{(proyección)} $\vec u \cdot \vec v = \vec u \cdot \vec v'$,
        donde $\vec v'$ es la proyección ortogonal de $\vec v$ sobre la
        dirección de $\vec u$; si $\vec u = \vect{OA}$ y
        $\vec v = \vect{OB}$, entonces
        $\vec u \cdot \vec v = \overline{OA} \times \overline{OH}$, un
        producto de longitudes con signo, donde $H$ es el pie de la
        perpendicular trazada desde $B$ a la recta $(OA)$;
  \item \emph{(\omterm{def:g11:scal:dot}{normas})} la fórmula de la \cref{def:g11:scal:dot} que lo
        define.
\end{enumerate}
\end{theorem}

\begin{proof}[Demostración de 1 y 2]
\emph{1.} En \omterm{def:g10:coordgeom:system}{coordenadas}, $\norm{\vec u}^2 = x^2 + y^2$ (Pitágoras) y
$\vec u + \vec v$ tiene \omterm{def:g10:coordgeom:system}{coordenadas} $(x + x', y + y')$, luego
\begin{align*}
2\,\vec u \cdot \vec v
&= (x + x')^2 + (y + y')^2 - (x^2 + y^2) - (x'^2 + y'^2)\\
&= 2xx' + 2yy',
\end{align*}
tras \omterm{def:g10:algebra:expand}{desarrollar} los cuadrados y cancelar.

\emph{2.} Elegimos el \omterm{def:g10:coordgeom:system}{sistema de coordenadas} de modo que $\vec u$ apunte
en el sentido positivo del eje $x$: $\vec u\,(\norm{\vec u}, 0)$ y
$\vec v\,(\norm{\vec v}\cos\theta,\ \norm{\vec v}\sin\theta)$
(\cref{ch:g11:trigo}). La fórmula 1 da entonces
$\vec u \cdot \vec v = \norm{\vec u}\,\norm{\vec v}\cos\theta + 0$. La
fórmula 3 se lee en el mismo cálculo: la proyección de $\vec v$ sobre el
eje $x$ tiene longitud con signo
$\norm{\vec v}\cos\theta = \overline{OH}$.
\end{proof}

\begin{example}\label{ex:g11:scal:compute}
$\vec u\,(3, 1)$ y $\vec v\,(2, -4)$: $\vec u \cdot \vec v = 6 - 4 = 2$.
Las \omterm{def:g11:scal:dot}{normas} son $\norm{\vec u} = \sqrt{10}$ y $\norm{\vec v} = \sqrt{20}$,
así que $\cos\theta = \frac{2}{\sqrt{10}\sqrt{20}} = \frac{2}{10\sqrt2}
= \frac{\sqrt2}{10}$: el ángulo es algo menor que $82^\circ$.
\end{example}

\begin{proposition}[Reglas algebraicas]\label{prop:g11:scal:rules}
Para todos los \omterm{def:g11:vect:vector}{vectores} y todo $\lambda \in \R$:
\[
\vec u \cdot \vec v = \vec v \cdot \vec u, \qquad
\vec u \cdot (\vec v + \vec w) = \vec u \cdot \vec v + \vec u \cdot
\vec w, \qquad
(\lambda \vec u)\cdot \vec v = \lambda\,(\vec u \cdot \vec v),
\]
\[
\vec u \cdot \vec u = \norm{\vec u}^2,
\qquad
\norm{\vec u + \vec v}^2 = \norm{\vec u}^2 + 2\,\vec u\cdot\vec v +
\norm{\vec v}^2 .
\]
\end{proposition}

\begin{proof}
Todas se siguen de la fórmula en \omterm{def:g10:coordgeom:system}{coordenadas}: por ejemplo,
$\vec u \cdot (\vec v + \vec w) = x(x' + x'') + y(y' + y'') =
(xx' + yy') + (xx'' + yy'')$. La última identidad desarrolla
$\norm{\vec u + \vec v}^2 = (\vec u + \vec v)\cdot(\vec u + \vec v)$ con
las tres primeras reglas: el \omterm{def:g11:scal:dot}{producto escalar} se comporta exactamente como
un producto ordinario, así que las identidades algebraicas de siempre
valen también para \omterm{def:g11:vect:vector}{vectores}.
\end{proof}

\section{Ortogonalidad}

\begin{definition}[Vectores ortogonales]\label{def:g11:scal:orthogonal}
Dos \omterm{def:g11:vect:vector}{vectores} son \emph{ortogonales}\index{vectores ortogonales}, y se
escribe $\vec u \perp \vec v$, si $\vec u \cdot \vec v = 0$.
\end{definition}

\begin{remark}
Para \omterm{def:g11:vect:vector}{vectores} no nulos, la fórmula 2 del \cref{thm:g11:scal:faces}
muestra que $\vec u \cdot \vec v = 0 \iff \cos\theta = 0 \iff$ las
direcciones son perpendiculares. El \omterm{def:g11:vect:vector}{vector} nulo es ortogonal a todo.
\end{remark}

\begin{example}
$\vec u\,(3, 2)$ y $\vec v\,(-4, 6)$:
$\vec u \cdot \vec v = -12 + 12 = 0$: \omterm{def:g11:scal:orthogonal}{ortogonales}. En general,
$\vec u\,(a, b)$ es siempre ortogonal a $\vec v\,(-b, a)$: un giro de un
cuarto de vuelta.
\end{example}

\section{El teorema del coseno}

\begin{theorem}[Teorema del coseno]\label{thm:g11:scal:cosines}
\index{teorema del coseno}
En un triángulo $ABC$ de lados $a = BC$, $b = CA$, $c = AB$ y ángulo
$\widehat A$ en el vértice $A$:
\[
a^2 = b^2 + c^2 - 2bc\,\cos\widehat A .
\]
\end{theorem}

\begin{proof}
Escribimos $\vect{BC} = \vect{BA} + \vect{AC} = \vect{AC} - \vect{AB}$ y
desarrollamos con la \cref{prop:g11:scal:rules}:
\[
a^2 = \norm{\vect{BC}}^2
= \norm{\vect{AC}}^2 - 2\,\vect{AC}\cdot\vect{AB} + \norm{\vect{AB}}^2
= b^2 + c^2 - 2\,\vect{AB}\cdot\vect{AC}.
\]
Por la fórmula del \omterm{def:g11:trigo:cossin}{coseno},
$\vect{AB}\cdot\vect{AC} = c\,b\,\cos\widehat A$.
\end{proof}

\begin{remark}
Cuando $\widehat A$ es recto, $\cos\widehat A = 0$ y el teorema del \omterm{def:g11:trigo:cossin}{coseno}
se reduce al teorema de Pitágoras $a^2 = b^2 + c^2$: es Pitágoras con un
término corrector para los ángulos no rectos.
\end{remark}

\begin{example}\label{ex:g11:scal:cosines}
Dos lados de un triángulo miden $b = 5$ y $c = 8$ y forman un ángulo
$\widehat A = \frac\pi3$. Entonces
\[
a^2 = 25 + 64 - 2 \times 5 \times 8 \times \frac12 = 89 - 40 = 49,
\]
luego $a = 7$.
\end{example}

\section{Aplicaciones: circunferencias y rectas perpendiculares}

\begin{proposition}[Circunferencia de diámetro dado]\label{prop:g11:scal:diameter}
Un punto $M$ está en la circunferencia de diámetro $[AB]$ si y solo si
\[
\vect{MA} \cdot \vect{MB} = 0 .
\]
\end{proposition}

\begin{proof}
Si $M \neq A, B$, la condición dice que el ángulo en $M$ del triángulo
$AMB$ es recto, y los puntos que ven un segmento bajo ángulo recto son
exactamente los de la circunferencia que lo tiene por diámetro. He aquí
una demostración directa: sea $O$ el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[AB]$ y
$r = \frac{AB}{2}$. Entonces $\vect{MA} = \vect{MO} + \vect{OA}$ y
$\vect{MB} = \vect{MO} + \vect{OB} = \vect{MO} - \vect{OA}$, luego
\[
\vect{MA}\cdot\vect{MB}
= \norm{\vect{MO}}^2 - \norm{\vect{OA}}^2 = MO^2 - r^2,
\]
que se anula exactamente cuando $MO = r$, es decir, cuando $M$ está en la
circunferencia de centro $O$ y radio $r$.
\end{proof}

\begin{proposition}[Vector normal de una recta]\label{prop:g11:scal:normal}
El \omterm{def:g11:vect:vector}{vector} $\vec n\,(a, b)$ es ortogonal a todos los \omterm{def:g11:vect:direction}{vectores directores} de
la recta $d : ax + by + c = 0$; se dice que $\vec n$ es un \emph{vector
normal}\index{vector normal} de $d$. Dos rectas son perpendiculares
exactamente cuando sus \omterm{def:g11:vect:vector}{vectores} normales (o sus \omterm{def:g11:vect:direction}{vectores directores}) son
\omterm{def:g11:scal:orthogonal}{ortogonales}.
\end{proposition}

\begin{proof}
Un \omterm{def:g11:vect:direction}{vector director} de $d$ es $\vec u\,(-b, a)$
(\cref{thm:g11:vect:cartesian}), y
$\vec n \cdot \vec u = a(-b) + b(a) = 0$.
\end{proof}

\begin{method}[Uso de los vectores normales]\label{met:g11:scal:normal}
\begin{itemize}
  \item Recta que pasa por $A$ con normal $\vec n\,(a,b)$: escribe
        $\vect{AM} \cdot \vec n = 0$, que se desarrolla como
        $ax + by + c = 0$ con $c$ determinado por $A$.
  \item Perpendicularidad de $d: ax+by+c = 0$ y $d': a'x+b'y+c' = 0$:
        comprueba que $aa' + bb' = 0$.
\end{itemize}
\end{method}

\begin{example}\label{ex:g11:scal:perpline}
La recta que pasa por $P(2, -1)$ y es perpendicular a
$d : 3x - y + 4 = 0$: un \omterm{def:g11:vect:direction}{vector director} de $d$ es $(1, 3)$, que sirve de
\omterm{def:g11:vect:vector}{vector} \emph{normal} de la perpendicular. \omterm{def:g10:algebra:equation}{Ecuación}: $x + 3y + c = 0$ con
$2 - 3 + c = 0$, luego $c = 1$: la recta es $x + 3y + 1 = 0$.
\end{example}

\begin{example}[Ecuación de una circunferencia]\label{ex:g11:scal:circle}
La circunferencia de centro $\Omega(1, 2)$ y radio $3$ es el conjunto de
los puntos $M(x,y)$ con $\Omega M^2 = 9$:
\[
(x - 1)^2 + (y - 2)^2 = 9 .
\]
Recíprocamente, $x^2 + y^2 - 2x - 4y - 4 = 0$ se reescribe, completando
cuadrados (\cref{ch:g11:quad}), como $(x-1)^2 + (y-2)^2 = 9$: la misma
circunferencia.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:scal:1}
Calcula $\vec u \cdot \vec v$ para
\[
\vec u\,(2, 5),\ \vec v\,(3, -1); \qquad
\vec u\,(1, -4),\ \vec v\,(8, 2); \qquad
\norm{\vec u} = 3,\ \norm{\vec v} = 4,\ \theta = \frac{\pi}{3} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:scal:2}
$ABC$ es un triángulo equilátero de lado $6$. Calcula
$\vect{AB} \cdot \vect{AC}$ y $\vect{AB} \cdot \vect{BC}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:scal:3}
¿Para qué valor de $t$ son \omterm{def:g11:scal:orthogonal}{ortogonales} $\vec u\,(3, t)$ y
$\vec v\,(t - 8, 1)$? ¿Y para qué valor o valores de $t$ es
$\vec u\,(t, 2)$ ortogonal a sí mismo menos $\vec v\,(4, 2)$, es decir,
$\vec u \perp (\vec u - \vec v)$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:scal:4}
Calcula el ángulo $\theta$ que forman $\vec u\,(1, 2)$ y $\vec v\,(3, 1)$
(da $\cos\theta$ de forma exacta y después $\theta$ redondeado al grado).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:5}
Un triángulo tiene lados $b = 4$ y $c = 6$ que forman un ángulo
$\widehat A = \frac{2\pi}{3}$. Calcula el tercer lado $a$. Después, en un
triángulo de lados $a = 7$, $b = 5$, $c = 4$, calcula $\cos\widehat A$ y
decide si $\widehat A$ es agudo u obtuso.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:6}
Sean $A(1, 1)$, $B(5, -1)$ y $C(3, 5)$. Calcula
$\vect{AB} \cdot \vect{AC}$; ¿es recto el ángulo $\widehat A$? Calcula las
tres longitudes de los lados y comprueba tu respuesta con el teorema del
\omterm{def:g11:trigo:cossin}{coseno}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:7}
Da una \omterm{def:g10:algebra:equation}{ecuación} de la circunferencia de centro $\Omega(-2, 3)$ y radio
$5$. Determina después el centro y el radio de la circunferencia
\[
x^2 + y^2 - 6x + 2y - 6 = 0 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:8}
Sean $A(-3, 0)$ y $B(0, 2)$. Usando la \cref{prop:g11:scal:diameter},
halla una \omterm{def:g10:algebra:equation}{ecuación} de la circunferencia de diámetro $[AB]$ y comprueba que
el origen $O(0,0)$ está en ella.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:9}
Halla una \omterm{def:g10:algebra:equation}{ecuación} de la recta que pasa por $P(1, 3)$ y es perpendicular a
$d : 2x + y - 7 = 0$, y las \omterm{def:g10:coordgeom:system}{coordenadas} del pie $H$ de la perpendicular
(el corte de las dos rectas). Deduce la distancia de $P$ a $d$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:10}
Usando $\norm{\vec u + \vec v}^2 = \norm{\vec u}^2 + 2\vec u\cdot\vec v +
\norm{\vec v}^2$ y su análoga para $\vec u - \vec v$, demuestra la
\emph{identidad del paralelogramo}:
\[
\norm{\vec u + \vec v}^2 + \norm{\vec u - \vec v}^2
= 2\norm{\vec u}^2 + 2\norm{\vec v}^2 ,
\]
e interprétala en un paralelogramo (diagonales frente a lados).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:scal:11}
Sean $A$ y $B$ dos puntos con $AB = 4$ y sea $I$ el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de
$[AB]$.
\begin{enumerate}
  \item Demuestra que para todo punto $M$ se cumple
        $\vect{MA}\cdot\vect{MB} = MI^2 - 4$. (Intercala $I$:
        $\vect{MA} = \vect{MI} + \vect{IA}$.)
  \item Deduce el conjunto de los puntos $M$ tales que
        $\vect{MA}\cdot\vect{MB} = 12$.
\end{enumerate}
\end{exercise}

\section{Problema: Dos regalos del producto escalar}

\begin{problem}[{Problema de fin de semana --- las fórmulas de adición
salen de un producto escalar, la distancia a una recta de otro, y de
propina las tres alturas se cortan en un punto}]
\label{pb:g11:scal:1}
El \omterm{def:g11:scal:dot}{producto escalar} parece contabilidad (multiplicar y sumar) y, sin
embargo, entrega casi de balde dos tesoros que resistieron siglos de
persecución de ángulos: las \emph{fórmulas de adición} de la
trigonometría (¡calcular $\cos 15^\circ$ de forma exacta!) y la
\emph{distancia de un punto a una recta} en una fracción limpia. Como
regalo de despedida demuestra que las tres alturas de cualquier triángulo
pasan por un mismo punto. Las cuatro caras del \cref{thm:g11:scal:faces}
pasarán lista.

\medskip
\noindent\textbf{Parte I --- Soltura.}
\begin{enumerate}
  \item Para $\vec u(3, 4)$ y $\vec v(-1, 2)$: calcula
        $\vec u \cdot \vec v$, las dos \omterm{def:g11:scal:dot}{normas} y el ángulo que forman los
        \omterm{def:g11:vect:vector}{vectores} (con una décima de grado).
  \item Halla $k$ para que $(2, k)$ y $(3, -4)$ sean \omterm{def:g11:scal:orthogonal}{ortogonales}
        (\cref{def:g11:scal:orthogonal}).
  \item Cara física: se arrastra un trineo $10$ m con una fuerza de $50$ N
        que forma $60^\circ$ con el suelo. Calcula el trabajo
        $W = \vec F \cdot \vec d$.
  \item Teorema del \omterm{def:g11:trigo:cossin}{coseno} (\cref{thm:g11:scal:cosines}): dos lados de $5$
        y $7$ forman un ángulo de $60^\circ$; halla el tercer lado de
        forma exacta.
  \item Al revés: un triángulo tiene lados $4$, $6$ y $8$. Calcula el
        \omterm{def:g11:trigo:cossin}{coseno} del ángulo opuesto al lado $8$, clasifica el ángulo y dalo
        con una décima de grado.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Primer regalo: las fórmulas de
adición.}
Sean $\vec u = (\cos a, \sin a)$ y $\vec v = (\cos b, \sin b)$: dos
\omterm{def:g11:vect:vector}{vectores} unitarios de ángulos $a$ y $b$.
\begin{enumerate}[resume]
  \item Calcula $\vec u \cdot \vec v$ dos veces, una con \omterm{def:g10:coordgeom:system}{coordenadas} y
        otra con las \omterm{def:g11:scal:dot}{normas} y el ángulo que forman los \omterm{def:g11:vect:vector}{vectores}, y
        concluye la fórmula madre:
        \[
        \cos(a - b) = \cos a \cos b + \sin a \sin b .
        \]
  \item Sustituye $b$ por $-b$ (¡paridad del \omterm{def:g11:trigo:cossin}{seno} y del \omterm{def:g11:trigo:cossin}{coseno}!) para
        obtener $\cos(a + b)$.
  \item Obtén $\sin(a + b)$ a partir de la identidad de los ángulos
        complementarios $\sin x = \cos\left(\frac\pi2 - x\right)$ aplicada
        a $x = a + b$.
  \item Por fin valores exactos: calcula $\cos 15^\circ$ (como
        $\cos(45^\circ - 30^\circ)$) y $\sin 75^\circ$. Comprueba los dos
        numéricamente.
  \item Toma $b = a$ para deducir las fórmulas del ángulo doble:
        $\cos 2a$ (en sus tres formas) y $\sin 2a$.
  \item A partir de $\cos 2a = 1 - 2\sin^2 a$, calcula $\sin^2 15^\circ$ y
        después $\sin 15^\circ$ como un radical anidado, y comprueba que
        su cuadrado coincide con el valor $\frac{\sqrt6 - \sqrt2}{4}$ que
        da el método de la pregunta 9.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Segundo regalo: la distancia a una
recta.}
\begin{enumerate}[resume]
  \item Sea $d$ la recta $ax + by + c = 0$, con \omterm{def:g11:vect:vector}{vector} normal
        $\vec n(a, b)$ (\cref{prop:g11:scal:normal}), y sea $P(x_0, y_0)$
        un punto. Tomando un punto $A$ cualquiera de $d$, explica por qué
        la distancia de $P$ a $d$ es
        $\dfrac{\abs{\vect{AP} \cdot \vec n}}{\norm{\vec n}}$ y deduce la
        fórmula
        \[
        \operatorname{dist}(P, d)
        = \frac{\abs{a x_0 + b y_0 + c}}{\sqrt{a^2 + b^2}} .
        \]
        Calcula la distancia de $P(5, 6)$ a $3x + 4y - 12 = 0$.
  \item Escribe la \omterm{def:g10:algebra:equation}{ecuación} de la circunferencia de centro $(5, 6)$
        \omterm{def:g11:deriv:tangent}{tangente} a esa recta.
  \item Contrasta la fórmula: calcula la distancia del origen a
        $3x + 4y - 12 = 0$ y verifícala calculando explícitamente el pie
        de la perpendicular desde el origen.
  \item Triángulo $A(0,0)$, $B(6,0)$, $C(2,5)$: calcula su área tomando
        como base $[AB]$; calcula después la distancia de $B$ a la recta
        $(AC)$ y comprueba que reproduce la misma área.
  \item La circunferencia de centro $(2, 1)$ y radio $2$, frente a la
        recta $x + y - 6 = 0$: calcula la distancia del centro a la recta
        y clasifica su posición relativa (\omterm{def:g11:deriv:rate}{secante}, \omterm{def:g11:deriv:tangent}{tangente} o exterior).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La propina: las alturas concurren.}
\begin{enumerate}[resume]
  \item Generaliza el \cref{exo:g11:scal:11}: para dos puntos cualesquiera
        $A$ y $B$ de \omterm{prop:g10:coordgeom:midpoint}{punto medio} $I$, demuestra que
        \[
        \vect{MA} \cdot \vect{MB} = MI^2 - \frac{AB^2}{4}
        \quad\text{para todo punto } M .
        \]
  \item Deduce la \cref{prop:g11:scal:diameter} en una línea: el conjunto
        de los puntos $M$ con $\vect{MA} \cdot \vect{MB} = 0$ es la
        circunferencia de diámetro $[AB]$.
  \item Las alturas: sea $H$ el corte de las alturas desde $A$ y desde $B$
        en un triángulo $ABC$ (de modo que $\vect{HA} \cdot \vect{BC} = 0$
        y $\vect{HB} \cdot \vect{CA} = 0$). Demuestra la identidad
        \[
        \vect{HA} \cdot \vect{BC} + \vect{HB} \cdot \vect{CA}
        + \vect{HC} \cdot \vect{AB} = 0
        \]
        desarrollándolo todo desde $H$ con Chasles, y concluye que $H$
        está también sobre la altura desde $C$: las tres alturas son
        concurrentes (en el \emph{ortocentro}\index{ortocentro}).
  \item Final: cuatro caras, tres trofeos. ¿Qué cara del
        \cref{thm:g11:scal:faces} llevó las fórmulas de adición, cuál la
        fórmula de la distancia y cuál el \omterm{pb:g11:scal:1}{ortocentro}? ¿Y por qué
        ``calcular dos veces la misma cantidad'' es el lema de todo el
        capítulo?
\end{enumerate}
\end{problem}
