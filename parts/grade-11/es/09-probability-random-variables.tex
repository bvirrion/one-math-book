\chapter{Probabilidad y variables aleatorias}\label{ch:g11:prob}

La \omterm{def:g10:proba:distribution}{probabilidad} modeliza experimentos cuyo resultado es incierto: lanzar un
dado, sacar una carta, jugar una partida. Este capítulo monta el
vocabulario sobre conjuntos finitos y presenta después al personaje central
de toda la teoría de la \omterm{def:g10:proba:distribution}{probabilidad}: la \emph{\omterm{def:g11:prob:rv}{variable aleatoria}} y su
\emph{\omterm{def:g11:prob:expectation}{esperanza}}. Estas nociones se repasan y se profundizan en el
\cref{ch:g12:randvar}, y las \omterm{def:g10:proba:distribution}{probabilidades} condicionadas llegan en el
\cref{ch:g12:condprob}.

\section{Probabilidad sobre un conjunto finito}

\begin{definition}[Modelo probabilístico]\label{def:g11:prob:model}
Un experimento con un número finito de resultados se modeliza mediante su
\emph{espacio muestral}\index{espacio muestral}
$\Omega = \{\omega_1, \dots, \omega_n\}$ (el conjunto de los resultados)
junto con una \emph{\omterm{def:g10:proba:distribution}{probabilidad}} $\P$ que asigna a cada resultado
$\omega_i$ un número $p_i \geq 0$, con $p_1 + \dots + p_n = 1$. Un
\emph{suceso}\index{suceso} es un subconjunto $A \subseteq \Omega$, y
\[
\P(A) = \sum_{\omega_i \in A} p_i .
\]
Cuando todos los resultados son equiprobables ($p_i = \frac1n$: el
\emph{modelo uniforme}\index{modelo uniforme}),
\[
\P(A) = \frac{\text{número de resultados de } A}
{\text{número de resultados de } \Omega}.
\]
\end{definition}

\begin{example}\label{ex:g11:prob:twodice}
Lanzamos dos dados distinguibles: $\Omega$ es el conjunto de las $36$
parejas ordenadas $(i, j)$ con $1 \leq i, j \leq 6$, todas equiprobables.
El \omterm{def:g11:prob:model}{suceso} $S$ = ``la suma es $7$'' contiene las $6$ parejas
$(1,6)$, $(2,5)$, $(3,4)$, $(4,3)$, $(5,2)$, $(6,1)$, luego
$\P(S) = \frac{6}{36} = \frac16$. El \omterm{def:g11:prob:model}{suceso} ``la suma es $12$'' contiene
solo $(6,6)$: \omterm{def:g10:proba:distribution}{probabilidad} $\frac{1}{36}$. Elegir el \omterm{def:g11:prob:model}{espacio muestral}
\emph{adecuado} (parejas ordenadas, no desordenadas) es lo que hace
aplicable el \omterm{def:g11:prob:model}{modelo uniforme}.
\end{example}

\begin{proposition}[Reglas de la probabilidad]\label{prop:g11:prob:rules}
Para \omterm{def:g11:prob:model}{sucesos} $A, B \subseteq \Omega$:
\[
\P(\emptyset) = 0, \qquad \P(\Omega) = 1, \qquad
\P(\bar A) = 1 - \P(A),
\]
\[
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B),
\]
donde $\bar A$ es el \emph{\omterm{def:g10:proba:operations}{suceso contrario}} de $A$ (todos los resultados
que no están en $A$). En particular, $\P(A \cup B) = \P(A) + \P(B)$ cuando
$A$ y $B$ son \emph{\omterm{def:g10:proba:operations}{incompatibles}} ($A \cap B = \emptyset$).
\end{proposition}

\begin{proof}
El \omterm{def:g11:prob:model}{suceso} vacío no contiene ningún resultado (suma vacía) y $\Omega$ los
contiene todos (suma total $1$). Cada resultado está exactamente en uno de
$A$ o $\bar A$, luego $\P(A) + \P(\bar A) = 1$. Para la \omterm{def:g10:numbers:interunion}{unión}: sumar los
$p_i$ sobre $A$ y después sobre $B$ cuenta dos veces los resultados de
$A \cap B$; restar $\P(A \cap B)$ corrige ese doble recuento.
\end{proof}

\begin{example}
En una clase, el $60\,\%$ estudia música ($M$), el $45\,\%$ practica algún
deporte ($S$) y el $25\,\%$ hace las dos cosas. La proporción que hace al
menos una es $\P(M \cup S) = 0.60 + 0.45 - 0.25 = 0.80$, así que el
$20\,\%$ no hace ninguna (\omterm{def:g10:proba:operations}{suceso contrario}).
\end{example}

\section{Variables aleatorias}

\begin{definition}[Variable aleatoria, distribución]\label{def:g11:prob:rv}
Una \emph{variable aleatoria}\index{variable aleatoria} sobre $\Omega$ es
una \omterm{def:g11:func:function}{función} $X$ que asigna un número a cada resultado. Su
\emph{distribución}\index{distribución} es la tabla de sus valores
posibles $x_1, \dots, x_k$ junto con sus \omterm{def:g10:proba:distribution}{probabilidades}
\[
\P(X = x_i) = \P\bigl(\{\omega : X(\omega) = x_i\}\bigr),
\qquad
\sum_{i=1}^k \P(X = x_i) = 1 .
\]
\end{definition}

\begin{example}\label{ex:g11:prob:game}
Un juego: pagas $2$ euros, lanzas un dado no trucado y recibes el valor
que salga si es al menos $5$, y nada en los demás casos. Sea $G$ la
\emph{ganancia} (lo recibido menos lo pagado). Los resultados $1, 2, 3, 4$
dan $G = -2$; el resultado $5$ da $G = 3$; el resultado $6$ da $G = 4$. La
\omterm{def:g11:prob:rv}{distribución} de $G$:
\begin{center}
\begin{tabular}{c|ccc}
$g$ & $-2$ & $3$ & $4$\\
\hline\rule{0pt}{11pt}%
$\P(G = g)$ & $\dfrac46$ & $\dfrac16$ & $\dfrac16$
\end{tabular}
\end{center}
\end{example}

\begin{method}[Construir una tabla de distribución]\label{met:g11:prob:table}
Enumera los valores posibles de $X$; para cada valor, reúne los resultados
que lo producen y suma sus \omterm{def:g10:proba:distribution}{probabilidades}; y comprueba que la fila de
\omterm{def:g10:proba:distribution}{probabilidades} suma $1$.
\end{method}

\section{Esperanza}

\begin{definition}[Esperanza]\label{def:g11:prob:expectation}
La \emph{esperanza}\index{esperanza} (o \omterm{def:g11:stat:mean}{media}) de $X$ es el promedio de
sus valores ponderado por sus \omterm{def:g10:proba:distribution}{probabilidades}:
\[
\E(X) = \sum_{i=1}^{k} \P(X = x_i)\; x_i .
\]
\end{definition}

\begin{remark}
$\E(X)$ es lo que sería la \omterm{def:g11:stat:mean}{media} de un número enorme de repeticiones
independientes del experimento; la versión precisa de esta afirmación es
la ley de los grandes números, que se demuestra en el \cref{ch:g12:sums}.
Fíjate en la analogía formal con la \omterm{def:g11:stat:mean}{media} de un conjunto de datos con
frecuencias (\cref{def:g11:stat:mean}): las \omterm{def:g10:proba:distribution}{probabilidades} hacen el papel
de las \omterm{def:g10:stats:series}{frecuencias relativas} $\frac{n_i}{n}$.
\end{remark}

\begin{example}\label{ex:g11:prob:gameexp}
Para el juego del \cref{ex:g11:prob:game}:
\[
\E(G) = \frac46 \times (-2) + \frac16 \times 3 + \frac16 \times 4
= \frac{-8 + 3 + 4}{6} = -\frac16 \approx -0.17 .
\]
De \omterm{def:g11:stat:mean}{media}, el jugador pierde unos $17$ céntimos por partida: el juego es
desfavorable. Un juego se llama \emph{equitativo} cuando la ganancia
esperada es $0$.
\end{example}

\begin{proposition}[Linealidad]\label{prop:g11:prob:linearity}
Para todos los reales $a, b$:
\[
\E(aX + b) = a\,\E(X) + b .
\]
\end{proposition}

\begin{proof}
La variable $aX + b$ toma el valor $a x_i + b$ con \omterm{def:g10:proba:distribution}{probabilidad}
$\P(X = x_i)$, luego
\[
\E(aX + b) = \sum_i \P(X = x_i)(a x_i + b)
= a \sum_i \P(X = x_i)\,x_i + b \underbrace{\sum_i \P(X =
x_i)}_{=\,1} = a\,\E(X) + b .
\]
\end{proof}

\section{Varianza y desviación típica}

\begin{definition}[Varianza]\label{def:g11:prob:variance}
La \emph{varianza}\index{varianza} de $X$ mide la dispersión de sus
valores alrededor de la \omterm{def:g11:prob:expectation}{esperanza} $m = \E(X)$:
\[
\V(X) = \E\bigl((X - m)^2\bigr)
= \sum_{i=1}^{k} \P(X = x_i)\,(x_i - m)^2,
\qquad
\sigma(X) = \sqrt{\V(X)} .
\]
\end{definition}

\begin{proposition}[Fórmula abreviada]\label{prop:g11:prob:konig}
\[
\V(X) = \E(X^2) - \E(X)^2 .
\]
\end{proposition}

\begin{proof}
Desarrollamos $(x_i - m)^2 = x_i^2 - 2m\,x_i + m^2$ dentro de la suma:
\[
\V(X) = \sum_i \P(X = x_i)\,x_i^2
- 2m \sum_i \P(X = x_i)\,x_i + m^2\sum_i \P(X = x_i)
= \E(X^2) - 2m^2 + m^2 ,
\]
que es $\E(X^2) - m^2$. Es la versión para \omterm{def:g11:prob:rv}{variables aleatorias} de la
identidad demostrada para conjuntos de datos en la
\cref{prop:g11:stat:shortcut}.
\end{proof}

\begin{proposition}[Transformación afín]\label{prop:g11:prob:affine}
$\V(aX + b) = a^2\,\V(X)$ y $\sigma(aX + b) = \abs a\,\sigma(X)$.
\end{proposition}

\begin{proof}
Por linealidad, $aX + b$ se aparta de su \omterm{def:g11:prob:expectation}{esperanza} $a m + b$ en
$(aX + b) - (am + b) = a(X - m)$: cada desviación queda multiplicada por
$a$, luego cada desviación al cuadrado por $a^2$, y su \omterm{def:g11:stat:mean}{media} ponderada
también. El desplazamiento $b$ ha desaparecido: desplazar una variable no
cambia su dispersión.
\end{proof}

\begin{example}\label{ex:g11:prob:variance}
Para el juego del \cref{ex:g11:prob:game}:
$\E(G^2) = \frac46 \times 4 + \frac16 \times 9 + \frac16 \times 16
= \frac{16 + 9 + 16}{6} = \frac{41}{6}$, luego
\[
\V(G) = \frac{41}{6} - \left(-\frac16\right)^2
= \frac{41}{6} - \frac{1}{36} = \frac{245}{36} \approx 6.8,
\qquad
\sigma(G) \approx 2.6 .
\]
La pérdida \omterm{def:g11:stat:mean}{media} de $0.17$ por partida viene acompañada de vaivenes de
tamaño típico $2.6$ euros: la \omterm{def:g11:prob:expectation}{esperanza} sin \omterm{def:g11:stat:variance}{desviación típica} solo cuenta
la mitad de la historia.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:prob:1}
Se saca una carta de una baraja de $52$ cartas. Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de
sacar: una carta de corazones; un rey; una carta de corazones o un rey; ni
corazones ni rey.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:prob:2}
Se lanzan dos dados no trucados y se anota su suma $S$. Usando el modelo
de $36$ resultados del \cref{ex:g11:prob:twodice}, calcula $\P(S = 6)$,
$\P(S \geq 10)$ y $\P(S \text{ es par})$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:prob:3}
Una \omterm{def:g11:prob:rv}{variable aleatoria} tiene la \omterm{def:g11:prob:rv}{distribución}
\begin{center}
\begin{tabular}{c|cccc}
$x$ & $-1$ & $0$ & $2$ & $5$\\
\hline
$\P(X=x)$ & $0.3$ & $0.2$ & $0.4$ & $p$
\end{tabular}
\end{center}
Halla $p$ y calcula después $\E(X)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:prob:4}
$X$ cumple $\E(X) = 3$ y $\V(X) = 4$. Da la \omterm{def:g11:prob:expectation}{esperanza}, la \omterm{def:g11:prob:variance}{varianza} y la
\omterm{def:g11:stat:variance}{desviación típica} de $Y = 2X - 5$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:5}
Una ruleta de la fortuna tiene $10$ sectores iguales: $5$ marcan $0$
euros, $3$ marcan $2$ euros y $2$ marcan $5$ euros. Jugar cuesta $2$
euros. Da la \omterm{def:g11:prob:rv}{distribución} de la ganancia $G$ (lo ganado menos la apuesta),
calcula $\E(G)$ y decide si el juego es favorable.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:6}
Una urna contiene $3$ bolas rojas y $2$ azules; se sacan dos bolas
\emph{sin} devolución. Sea $X$ el número de bolas rojas extraídas.
Construye la \omterm{def:g11:prob:rv}{distribución} de $X$ (enumerando las $20$ extracciones
ordenadas o con un \omterm{met:g10:proba:tree}{árbol}) y calcula $\E(X)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:7}
Para la ruleta del \cref{exo:g11:prob:5}, calcula $\V(G)$ y $\sigma(G)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:8}
Una compañía de seguros vende una póliza por $80$ euros. Con \omterm{def:g10:proba:distribution}{probabilidad}
$0.05$ el cliente reclama $1000$ euros, con \omterm{def:g10:proba:distribution}{probabilidad} $0.01$ reclama
$5000$ euros y, en los demás casos, nada. Sea $B$ el beneficio de la
compañía en una póliza. Da la \omterm{def:g11:prob:rv}{distribución} de $B$, calcula $\E(B)$ e
interpreta el resultado.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:9}
Una pregunta de tipo test tiene $4$ respuestas, una correcta. Una
respuesta correcta puntúa $+1$; para desincentivar el azar, una respuesta
equivocada puntúa $x < 0$. Un alumno responde al azar. ¿Para qué valor de
$x$ es nula la puntuación esperada del alumno?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:10}
Se lanzan dos monedas no trucadas. Ana propone: ``si las dos monedas
coinciden, te pago $1$ euro; si son distintas, me pagas $1$ euro''.
Calcula la ganancia esperada de cada jugador. ¿Es equitativo el juego? La
misma pregunta con tres monedas, en la que Ana paga $2$ euros si las tres
coinciden.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:prob:11}
Una rifa vende $1000$ papeletas a $5$ euros cada una. Una papeleta gana
$2000$ euros, cinco papeletas ganan $200$ euros y veinte papeletas ganan
$25$ euros.
\begin{enumerate}
  \item Da la \omterm{def:g11:prob:rv}{distribución} de la recaudación total del organizador menos
        los premios, y la de la ganancia $G$ de un jugador cualquiera.
  \item Calcula $\E(G)$ y el beneficio total esperado del organizador.
        Comprueba que las dos respuestas son coherentes.
\end{enumerate}
\end{exercise}

\section{Problema: La partida interrumpida}

\begin{problem}[{Problema de fin de semana --- Pascal, Fermat y el reparto
justo de una apuesta interrumpida: el nacimiento de la esperanza, la magia
de la linealidad y lo que la esperanza no ve}]
\label{pb:g11:prob:1}
En 1654, el caballero de Méré, el mismo jugador cuyas apuestas con dados
abrieron el problema de \omterm{def:g10:proba:distribution}{probabilidad} del volumen anterior, le planteó a
Blaise Pascal una pregunta más profunda: dos jugadores igual de hábiles
tienen que interrumpir su serie con el marcador $5$ a $3$, y seis
victorias se llevan el bote de $64$ pistolas. \emph{¿Cómo hay que repartir
el bote de forma justa?} Las cartas entre Pascal y Fermat que la
respondieron fundaron el concepto que está en el corazón de este capítulo:
la \omterm{def:g11:prob:expectation}{esperanza} (\cref{def:g11:prob:expectation}). Vas a rehacer su respuesta
por los dos caminos, y después conocerás el truco de fiesta de la
linealidad y el único punto ciego de la \omterm{def:g11:prob:expectation}{esperanza}.

\medskip
\noindent\textbf{Parte I --- Soltura con la \omterm{def:g11:prob:expectation}{esperanza}.}
\begin{enumerate}
  \item Lanzas un dado no trucado; tu ganancia es (cara $- 3$) euros.
        Construye la tabla de \omterm{def:g11:prob:rv}{distribución} de la ganancia $G$
        (\cref{met:g11:prob:table}) y calcula $\E(G)$.
  \item Calcula $V(G)$ con la fórmula abreviada
        (\cref{prop:g11:prob:konig}) y $\sigma(G)$ (con dos decimales).
  \item Sin ninguna tabla nueva, da $\E(2G - 1)$ y $V(2G - 1)$
        (\cref{prop:g11:prob:affine}).
  \item Un puesto cobra $m$ euros por lanzar dos dados y te entrega la
        suma en euros. ¿Cuál es el precio \emph{justo} $m$? El puesto
        cobra $8$: ¿cuál es tu pérdida esperada por partida?
  \item Un ordenador portátil que vale $800$ euros tiene un $5\,\%$ de
        \omterm{def:g10:proba:distribution}{probabilidad} de destrucción cada año. Calcula la prima de seguro
        justa y el beneficio esperado de la aseguradora por póliza si
        cobra $60$ euros.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Repartir el bote.}
Marcador $5$--$3$, gana el bote de $64$ pistolas el primero que llegue a
$6$, y cada ronda equivale a lanzar una moneda no trucada.
\begin{enumerate}[resume]
  \item Dos respuestas equivocadas muy tentadoras: repartir $5 : 3$
        (proporcionalmente a las rondas ganadas) o dárselo todo al que va
        por delante. Di en una frase qué tiene de injusta cada una.
  \item El método de Fermat: el jugador A necesita $1$ victoria más y B
        necesita $3$; imagina que se juegan exactamente $3$ rondas más
        pase lo que pase. Enumera los $8$ resultados equiprobables, marca
        quién gana la serie en cada uno y deduce el reparto justo de las
        $64$ pistolas.
  \item El punto delicado que Fermat tuvo que defender: la partida real
        podría acabarse tras una sola ronda; ¿por qué es legítimo, a pesar
        de ello, razonar sobre las $3$ rondas imaginarias?
  \item El método de Pascal, hacia atrás desde el final: calcula la parte
        justa de A en los marcadores (hipotéticos) $5$--$5$, después
        $5$--$4$ y después $5$--$3$, promediando cada vez los dos futuros
        equiprobables. Compara con la pregunta 7.
  \item La misma pregunta con el marcador $4$--$3$: ¿cuántas rondas
        imaginarias hacen falta, qué resultados dan la serie a B y cuál es
        el reparto justo?
  \item Enuncia el principio que comparten los dos métodos (el valor justo
        de un futuro incierto es su \emph{\omterm{def:g11:prob:expectation}{esperanza}}) y nombra dos
        industrias modernas que funcionan enteramente sobre él (la
        pregunta 5 muestra una).
  \item Una comprobación válida para cualquier marcador: explica, con la
        linealidad de la \omterm{def:g11:prob:expectation}{esperanza} (\cref{prop:g11:prob:linearity}), por
        qué las partes justas de los dos jugadores tienen que sumar
        exactamente el bote de $64$ pistolas.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El truco de fiesta de la linealidad.}
\begin{enumerate}[resume]
  \item Recupera $\E(\text{suma de dos dados}) = 7$ en una línea con la
        linealidad (\cref{prop:g11:prob:linearity}), sin ninguna tabla de
        $36$ casillas. Para la \emph{\omterm{def:g11:prob:variance}{varianza}} de la suma, la regla
        $V(X + Y) = V(X) + V(Y)$ exige que los dados sean independientes
        (se admite): calcúlala.
  \item La fiesta de los sombreros: $n$ invitados tiran sus sombreros a un
        montón y cada uno recoge uno al azar. Sea $X$ el número de
        invitados que recuperan su propio sombrero. Para un invitado
        fijado, ¿cuál es la \omterm{def:g10:proba:distribution}{probabilidad} de recuperar el suyo? Sumando
        esas $n$ \omterm{def:g10:proba:distribution}{probabilidades} (¡linealidad!), calcula $\E(X)$. La
        respuesta debería sorprenderte: no depende de $n$.
  \item Comprueba $\E(X) = 1$ por fuerza bruta para $n = 2$ y $n = 3$
        (enumera los $2$ y los $6$ repartos equiprobables de sombreros y
        promedia los recuentos).
  \item Los aciertos de los invitados están lejos de ser independientes
        (si $n - 1$ invitados han recuperado el suyo, el último también lo
        recupera por fuerza). ¿En qué punto exacto del cálculo de la
        pregunta 14 \emph{no} hizo falta la independencia? Enuncia el
        superpoder de la linealidad.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Lo que la \omterm{def:g11:prob:expectation}{esperanza} no ve.}
\begin{enumerate}[resume]
  \item Dos rascas de $5$ euros: el rasca A gana $10$ euros con
        \omterm{def:g10:proba:distribution}{probabilidad} $\frac12$; el rasca B gana $1\,000$ euros con
        \omterm{def:g10:proba:distribution}{probabilidad} $0.005$. Comprueba que las dos ganancias tienen la
        misma \omterm{def:g11:prob:expectation}{esperanza} y calcula después las dos \omterm{def:g11:prob:variance}{varianzas}. ¿Qué rasca
        compra la gente de verdad y qué es lo que está comprando?
  \item El juego de San Petersburgo: lanzas una moneda hasta la primera
        cruz; si la primera cruz sale en el lanzamiento $n$, recibes $2^n$
        euros. Calcula la aportación de cada $n$ a la \omterm{def:g11:prob:expectation}{esperanza} y la
        \omterm{def:g11:prob:expectation}{esperanza} misma. ¿Pagarías $1\,000$ euros por jugar una vez? ¿Qué
        enseña este choque sobre la \omterm{def:g11:prob:expectation}{esperanza}?
  \item El seguro desde el lado del cliente: ¿pagar $20$ euros o afrontar
        un riesgo del $1\,\%$ de perder $1\,000$? Compara las dos
        \omterm{def:g11:prob:expectation}{esperanzas} y explica por qué contratar el seguro puede ser
        racional de todos modos (¿sobre qué no puede ``promediar'' una
        sola familia?).
  \item Final: la biografía de la \omterm{def:g11:prob:expectation}{esperanza}, una frase por capítulo: nació
        en 1654 para repartir un bote; su superpoder es la linealidad, que
        suma \omterm{def:g11:prob:expectation}{esperanzas} sin preguntar por la independencia (los
        sombreros); su punto ciego es el riesgo, que mide la \omterm{def:g11:prob:variance}{varianza} (los
        rascas, San Petersburgo, el seguro); y su coronación llega el año
        que viene, cuando la ley de los grandes números explique
        exactamente por qué la banca siempre gana.
\end{enumerate}
\end{problem}
