\chapter{Derivación: un primer curso}\label{ch:g11:deriv}

¿Qué tan rápido \omterm{def:g11:func:function}{función} ¿cambiar? La respuesta en un solo punto es la
\emph{derivada}, el número más útil de todos los aplicados
matemáticas. Este capítulo lo construye a partir de tasas de cambio promedio,
lo calcula para el \omterm{def:g11:func:reference}{funciones de referencia}, y utiliza su signo para leer el
variaciones de una \omterm{def:g11:func:function}{función}. Todo aquí se amplía y profundiza en
\cref{ch:g12:deriv}.

\section{La tasa promedio de cambio}

\begin{definition}[Tasa promedio de cambio]\label{def:g11:deriv:rate}
Sea $f $ definirse en un \omterm{def:g10:numbers:interval}{intervalo} $ I $, y $ a \neq b $ en $ I $. El
\emph{tasa de cambio promedio}\index{tasa de cambio promedio} de $ f $ entre
$ a $ y $ b $ es
\[
\frac{f(b) - f(a)}{b - a}.
\]
es el \omterm{def:g10:reffunc:affine}{pendiente} del \emph{secante}\index{secante} línea que pasa por los puntos
$\bigl(a, f(a)\bigr)$ y $\bigl(b, f(b)\bigr)$ de la curva.
\end{definition}

\begin{example}
un auto viaja $140$ km entre las 14:00 y las 16:00 horas: su velocidad \omterm{def:g10:stats:mean}{media} es
$\frac{140}{2} = 70$ km/h. El velocímetro, sin embargo, \omterm{def:g10:proba:sample}{muestra} una
\emph{instantáneo} velocidad en cada momento --- esa es exactamente la noción
que la derivada hace preciso.
\end{example}

\section{La derivada en un punto}

\begin{definition}[Derivada en un punto]\label{def:g11:deriv:derivative}
Sea $f$ definida en un \omterm{def:g10:numbers:interval}{intervalo} que contiene $ a $. Para $ h \neq 0$, forma el
tasa de variación \omterm{def:g10:stats:mean}{media} entre $ a $ y $ a + h $:
\[
r(h) = \frac{f(a+h) - f(a)}{h}.
\]
Si $ r(h)$ se aproxima a un único número fijo como $ h $ se acerca arbitrariamente a
$0$, decimos que $ f $ es \emph{diferenciable es $ a $}\index{diferenciable en $ a $}\index{diferenciable};
este numero es el \emph{derivada de $ f $ en $ a $}, escrito $ f'(a)$:
\[
f'(a) = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}.
\]
\end{definition}

\begin{example}\label{ex:g11:deriv:atpoint}
Sea $ f(x) = x^2$ y $ a = 1$. Para $ h \neq 0$:
\[
\frac{f(1+h) - f(1)}{h}
= \frac{(1+h)^2 - 1}{h}
= \frac{1 + 2h + h^2 - 1}{h}
= \frac{h(2 + h)}{h}
= 2 + h .
\]
Como $ h $ aproches $0$, esto se acerca $2$: la \omterm{def:g11:func:function}{función} $ x^2$ es
\omterm{def:g11:deriv:derivative}{diferenciable} en $1$ y $ f'(1) = 2$. Tenga en cuenta la estrategia: \emph{simplificar
el cociente hasta $ h $ ya no aparece en un denominador}, entonces deja
$ h \to 0$.
\end{example}

\begin{example}[Un punto no diferenciable]
Sea $ f(x) = \abs{x}$ y $ a = 0$. Entonces
$\frac{f(h) - f(0)}{h} = \frac{\abs h}{h}$, que es igual $1$ para $ h > 0$
y $-1$ para $ h < 0$: no se aborda ningún número único, y $\abs x $ no es
\omterm{def:g11:deriv:derivative}{diferenciable} en $0$. Su curva tiene una \emph{esquina} allá.
\end{example}

\section{La recta tangente}

\begin{definition}[Tangente]\label{def:g11:deriv:tangent}
Si $ f $ es \omterm{def:g11:deriv:derivative}{diferenciable} en $ a $, el \emph{tangente}\index{tangente} al
curva en el punto $ A = \bigl(a, f(a)\bigr)$ es la línea que pasa $ A $ con
\omterm{def:g10:reffunc:affine}{pendiente} $ f'(a)$. Es \omterm{def:g10:algebra:equation}{ecuación} es
\[
y = f(a) + f'(a)(x - a).
\]
\end{definition}

El \omterm{def:g11:deriv:tangent}{tangente} es la posición límite de la \omterm{def:g11:deriv:rate}{secantes} a través de $ A $: como
$ h \to 0$, el segundo punto $\bigl(a+h, f(a+h)\bigr)$ se desliza a lo largo del
curva hacia $ A $, y el \omterm{def:g11:deriv:rate}{secante}'s \omterm{def:g10:reffunc:affine}{pendiente} $\frac{f(a+h)-f(a)}{h}$
aproches $ f'(a)$.

\begin{example}\label{ex:g11:deriv:tangenteq}
Para $ f(x) = x^2$ en $ a = 1$: $ f(1) = 1$ y $ f'(1) = 2$
(\cref{ex:g11:deriv:atpoint}), por lo que el \omterm{def:g11:deriv:tangent}{tangente} es
\[
y = 1 + 2(x - 1) = 2x - 1 .
\]
esta es la linea $ d_2$ encontrado por álgebra pura en \cref{exo:g11:quad:11}.
\end{example}

\section{Derivadas de las funciones de referencia}

\begin{proposition}\label{prop:g11:deriv:refderivatives}
en su \omterm{def:g11:func:function}{dominios} de diferenciabilidad:
\[
(c)' = 0, \quad
(x)' = 1, \quad
(x^2)' = 2x, \quad
(x^3)' = 3x^2, \quad
\left(\frac1x\right)' = -\frac{1}{x^2}, \quad
(\sqrt x\,)' = \frac{1}{2\sqrt x} \ (x > 0).
\]
De manera más general, $(x^n)' = n\,x^{n-1}$ por cada entero $ n \geq 1$.
\end{proposition}

\begin{proof}
Cada prueba sigue la estrategia de \cref{ex:g11:deriv:atpoint}: simplificar
la tasa, luego deje $ h \to 0$. Arreglar $ a $ en el \omterm{def:g11:func:function}{dominio}.

\emph{Constante e identidad.} $\frac{c - c}{h} = 0$ y
$\frac{(a+h) - a}{h} = 1$ para todo $ h $: los límites son $0$ y $1$.

\emph{Cuadrado.} $\frac{(a+h)^2 - a^2}{h} = \frac{2ah + h^2}{h} = 2a + h
\to 2a $.

\emph{Cubo.} $(a+h)^3 = a^3 + 3a^2h + 3ah^2 + h^3$, entonces la tasa es
$3a^2 + 3ah + h^2 \to 3a^2$.

\emph{Recíproca.} Para $ a \neq 0$ y $ h $ lo suficientemente pequeño como para
$ a + h \neq 0$:
\[
\frac{1}{h}\left(\frac{1}{a+h} - \frac{1}{a}\right)
= \frac{1}{h} \cdot \frac{a - (a+h)}{a(a+h)}
= \frac{-1}{a(a+h)}
\xrightarrow[h \to 0]{} -\frac{1}{a^2}.
\]

\emph{\omterm{def:g11:quad:discriminant}{Raíz} cuadrada.} Para $ a > 0$, multiplica por el conjugado:
\[
\frac{\sqrt{a+h} - \sqrt a}{h}
= \frac{(a + h) - a}{h\,(\sqrt{a+h} + \sqrt a)}
= \frac{1}{\sqrt{a+h} + \sqrt a}
\xrightarrow[h \to 0]{} \frac{1}{2\sqrt a}.
\]

\emph{Poder general.} el patrón $2a $, $3a^2$ continúa: en expansión
$(a+h)^n $, los términos después $ a^n + n\,a^{n-1}h $ todos llevan $ h^2$, entonces el
tasa tiende a $ n\,a^{n-1}$. (Se da una inducción completa en
\cref{ch:g12:deriv}.)
\end{proof}

\section{Reglas de diferenciación}

\begin{theorem}[Operaciones]\label{thm:g11:deriv:rules}
Sea $ u $ y $ v $ ser \omterm{def:g11:deriv:derivative}{diferenciable} en $ I $ y $\lambda \in \R $. Entonces
$ u + v $, $\lambda u $, $ uv $ son \omterm{def:g11:deriv:derivative}{diferenciable} en $ I $, como es
$\frac{u}{v}$ lo que sea $ v \neq 0$, y
\[
(u+v)' = u' + v', \qquad
(\lambda u)' = \lambda u', \qquad
(uv)' = u'v + uv', \qquad
\left(\frac{u}{v}\right)' = \frac{u'v - uv'}{v^2}.
\]
\end{theorem}

\begin{proof}[Prueba de las tres primeras reglas]
Arreglar $ a \in I $ y abreviar $\Delta u = u(a+h) - u(a)$,
$\Delta v = v(a+h) - v(a)$.

\emph{Suma.} la tasa de $ u + v $ es
$\frac{\Delta u + \Delta v}{h} = \frac{\Delta u}{h} + \frac{\Delta v}{h}
\to u'(a) + v'(a)$.

\emph{Múltiple.} la tasa de $\lambda u $ es
$\lambda \frac{\Delta u}{h} \to \lambda u'(a)$.

\emph{Producto.} Suma y resta el término mixto $ u(a+h)\,v(a)$:
\begin{align*}
\frac{u(a+h)v(a+h) - u(a)v(a)}{h}
&= \frac{u(a+h)v(a+h) - u(a+h)v(a) + u(a+h)v(a) - u(a)v(a)}{h}\\
&= u(a+h)\,\frac{\Delta v}{h} + v(a)\,\frac{\Delta u}{h}.
\end{align*}
Como $ h \to 0$: $\frac{\Delta v}{h} \to v'(a)$, $\frac{\Delta u}{h} \to
u'(a)$, y $ u(a+h) \to u(a)$ (su tasa tiene un límite, por lo que
$\Delta u = h \cdot \frac{\Delta u}{h} \to 0$). Por lo tanto, la tasa tiende
a $ u(a)v'(a) + v(a)u'(a)$.

La regla del cociente se demuestra del mismo modo; es \emph{admitido en
este nivel} y demostrado en \cref{ch:g12:deriv}.
\end{proof}

\begin{example}\label{ex:g11:deriv:computations}
\emph{Polinomio.} $ f(x) = 3x^3 - 5x^2 + 7x - 2$:
\[
f'(x) = 3 \times 3x^2 - 5 \times 2x + 7 = 9x^2 - 10x + 7 .
\]

\emph{Cociente.} $ g(x) = \frac{2x+1}{x^2+1}$: con $ u = 2x + 1$, $ v = x^2
+ 1$,
\[
g'(x) = \frac{2(x^2+1) - (2x+1)\,2x}{(x^2+1)^2}
= \frac{-2x^2 - 2x + 2}{(x^2+1)^2}.
\]

\emph{Producto.} $ k(x) = x\sqrt x $ para $ x > 0$:
$ k'(x) = 1 \times \sqrt x + x \times \frac{1}{2\sqrt x}
= \sqrt x + \frac{\sqrt x}{2} = \frac{3\sqrt x}{2}$.
\end{example}

\section{Signo de la derivada y variaciones.}

\begin{theorem}[Derivada y variaciones]\label{thm:g11:deriv:variations}
Sea $ f $ ser \omterm{def:g11:deriv:derivative}{diferenciable} en un \omterm{def:g10:numbers:interval}{intervalo} $ I $.
\begin{enumerate}
  \item Si $ f' \geq 0$ en $ I $, entonces $ f $ es \omterm{def:g11:func:monotone}{creciente} en $ I $.
  \item Si $ f' \leq 0$ en $ I $, entonces $ f $ es \omterm{def:g10:functions:variations}{decreciente} en $ I $.
  \item Si $ f' > 0$ en $ I $ (excepto posiblemente en un número finito de puntos donde
        desaparece), entonces $ f $ es estrictamente \omterm{def:g11:func:monotone}{creciente} en $ I $.
\end{enumerate}
\end{theorem}

\begin{proof}
La intuición es clara: una curva cuyo \omterm{def:g11:deriv:tangent}{tangentes} todo apunta hacia arriba debe
subir --- pero una prueba rigurosa necesita la \omterm{def:g10:stats:mean}{media} teorema del valor. El resultado es
\emph{admitido en este nivel}; ver \cref{thm:g12:deriv:variations}.
\end{proof}

\begin{proposition}[extremo]\label{prop:g11:deriv:extrema}
Si $ f'$ cambia de signo en $ a $ de $+$ a $-$, entonces $ f $ tiene un local \omterm{def:g10:functions:extrema}{máximo}
en $ a $; de $-$ a $+$, un lugareño \omterm{def:g10:functions:extrema}{mínimo}. En tal punto, $ f'(a) = 0$ y
el \omterm{def:g11:deriv:tangent}{tangente} es horizontal.
\end{proposition}

\begin{proof}
Si $ f' \geq 0$ en un \omterm{def:g10:numbers:interval}{intervalo} terminando en $ a $ y $ f' \leq 0$ en un \omterm{def:g10:numbers:interval}{intervalo}
comenzando en $ a $, entonces $ f $ aumenta hasta $ f(a)$ y disminuye después
(\cref{thm:g11:deriv:variations}): $ f(a)$ es el más grande cercano. Desde $ f'$
cambia de signo en $ a $ y se define allí, $ f'(a) = 0$.
\end{proof}

\begin{method}[Estudiar una función]\label{met:g11:deriv:study}
\begin{enumerate}
  \item Calcular $ f'(x)$ y simplificarlo --- idealmente en una forma factorizada.
  \item Determinar el signo de $ f'(x)$ en cada uno \omterm{def:g10:numbers:interval}{intervalo} (para una cuadrática
        $ f'$, usar \cref{thm:g11:quad:sign}).
  \item Registre los resultados en un \emph{\omterm{def:g10:functions:table}{tabla de variaciones}}: una fila para el
        signo de $ f'$, una fila de flechas para $ f $, con los valores de $ f $ en
        los puntos de inflexión.
  \item Concluir: extremos, número de soluciones de $ f(x) = k $, desigualdades.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g11:deriv:study}
Estudiar $ f(x) = x^3 - 3x + 1$ en $\R $. Primero,
$ f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$: una cuadrática con raíces $-1$ y $1$,
positivo fuera de ellos. Entonces $ f $ es \omterm{def:g11:func:monotone}{creciente} en $\intoc{-\infty}{-1}$,
\omterm{def:g10:functions:variations}{decreciente} en $\intcc{-1}{1}$, \omterm{def:g11:func:monotone}{creciente} en $\intco{1}{+\infty}$, con un
locales \omterm{def:g10:functions:extrema}{máximo} $ f(-1) = -1 + 3 + 1 = 3$ y un local \omterm{def:g10:functions:extrema}{mínimo}
$ f(1) = 1 - 3 + 1 = -1$.
\end{example}

\begin{example}[Mejoramiento]\label{ex:g11:deriv:box}
Una caja abierta está hecha de $12 \times 12$ hoja de cm cortando un cuadrado de
lado $ x $ de cada esquina y plegado. Su volumen es
$ V(x) = x(12 - 2x)^2$ para $0 < x < 6$. \omterm{def:g10:algebra:expand}{En expansión},
$ V(x) = 4x^3 - 48x^2 + 144x $, entonces
\[
V'(x) = 12x^2 - 96x + 144 = 12(x^2 - 8x + 12) = 12(x - 2)(x - 6).
\]
En $\intoo{0}{6}$, el factor $ x - 6$ es negativo, entonces $ V'(x)$ tiene el signo
de $-(x-2)$: positivo antes $2$, negativo después. El volumen es \omterm{def:g10:functions:extrema}{máximo}.
para $ x = 2$ cm, dando $ V(2) = 2 \times 8^2 = 128$ centímetro $^3$.
\end{example}

\section{Problema: los tres trabajos de la tangente}

\begin{problem}[{Problema del fin de semana --- el espejo parabólico demostró,
La máquina de búsqueda de raíces de Newton y la viga más fuerte del tronco}]
\label{pb:g11:deriv:1}
A derivada dibuja \omterm{def:g11:deriv:tangent}{tangentes}; Eso suena como un talento modesto.
Este problema \omterm{def:g10:proba:sample}{muestra} la \omterm{def:g11:deriv:tangent}{tangente} tener tres trabajos a la vez:
demuestra el teorema de los faros que quedó \omterm{def:g10:reffunc:affine}{pendiente} en
\cref{pb:g11:quad:1} (cada rayo paralelo a un parábolael eje
se refleja a través del \omterm{pb:g11:quad:1}{enfocar}), impulsa el más rápido
algoritmo de resolución de ecuaciones de uso común --- dentro de cada
calculadora y red neuronal --- y encuentra el más fuerte
viga rectangular que puede producir un tronco redondo.

\medskip
\noindent\textbf{Part I --- \omterm{def:g11:deriv:tangent}{tangente} fluency.}
\begin{enumerate}
  \item Diferenciar $ f(x) = 3x^2 - 5x + 1$ y
        $ g(x) = x^3 - 12x $
        (\cref{thm:g11:deriv:rules}); y recuperarse $\left(x^2
