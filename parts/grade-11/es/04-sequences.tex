\chapter{Sucesiones: un primer curso}\label{ch:g11:seq}

A \omterm{def:g11:seq:sequence}{sucesión} es una lista de números producida por una regla: los sucesivos saldos
de una cuenta de ahorros, el tamaño de una población año tras año. esto
El capítulo estudia las dos familias que dominan las aplicaciones.
\emph{\omterm{def:g11:seq:arithmetic}{aritmética}} sucesiones, que crecen en pasos iguales, y
\emph{\omterm{def:g11:seq:geometric}{geométrico}} sucesiones, que crecen en proporciones iguales. La teoría rigurosa
de límites se desarrolla en \cref{ch:g12:seq}.

\section{Definiendo una sucesión}

\begin{definition}[Sucesión]\label{def:g11:seq:sequence}
A \emph{sucesión}\index{sucesión} $(u_n)$ asigna a cada entero
$ n \geq 0$ (o $ n \geq 1$) a número real $ u_n $, es \emph{término de índice
$ n $}. Se puede dar una sucesión
\begin{itemize}
  \item \emph{explícitamente}, mediante una fórmula para $ u_n $ en términos de $ n $: p.ej.\
        $ u_n = n^2 + 1$;
  \item \emph{recursivamente}, por su primer término y una regla a pasar de cada uno
        término al siguiente: por ejemplo\ $ u_0 = 3$ y $ u_{n+1} = 2u_n - 1$.
\end{itemize}
\end{definition}

\begin{example}
Para $ u_n = n^2 + 1$: $ u_0 = 1$, $ u_1 = 2$, $ u_2 = 5$, y $ u_{10} = 101$
directamente. Para $ u_0 = 3$, $ u_{n+1} = 2u_n - 1$: $ u_1 = 5$, $ u_2 = 9$,
$ u_3 = 17$ --- cada término necesita del anterior; alcanzando $ u_{10}$ toma
diez pasos (o una fórmula general, ver \cref{exo:g11:seq:11}).
\end{example}

\section{sucesiones aritméticas}

\begin{definition}[sucesión aritmética]\label{def:g11:seq:arithmetic}
A \omterm{def:g11:seq:sequence}{sucesión} es \emph{aritmética}\index{aritmética} con
\emph{diferencia común}\index{diferencia común} $ d $ si cada término se obtiene del anterior
uno agregando $ d $:
\[
u_{n+1} = u_n + d \quad \text{para todo } n.
\]
Equivalentemente: la diferencia $ u_{n+1} - u_n $ es constante, igual a $ d $.
\end{definition}

\begin{theorem}[Término general]\label{thm:g11:seq:arithgeneral}
Si $(u_n)$ es \omterm{def:g11:seq:arithmetic}{aritmética} con primer término $ u_0$ y \omterm{def:g11:seq:arithmetic}{diferencia común} $ d $,
entonces
\[
u_n = u_0 + n\,d \quad \text{para todo } n \geq 0,
\qquad\text{and more generally } u_n = u_p + (n - p)\,d .
\]
\end{theorem}

\begin{proof}
para ir de $ u_0$ a $ u_n $, la regla ``añadir $ d $'' se aplica $ n $ veces: uno
paso da $ u_1 = u_0 + d $, dos pasos dan $ u_2 = u_0 + 2d $, y después $ n $
pasos cada aplicación ha aportado uno $ d $, entonces $ u_n = u_0 + nd $.
(Este "y así sucesivamente" se vuelve riguroso mediante la inducción en \cref{ch:g12:seq}.)
La fórmula general sigue contando el $ n - p $ pasos de $ u_p $ a
$ u_n $.
\end{proof}

\begin{theorem}[Suma de enteros consecutivos]\label{thm:g11:seq:intsum}
por cada entero $ n \geq 1$:
\[
1 + 2 + \dots + n = \frac{n(n+1)}{2}.
\]
De manera más general, una suma de términos consecutivos de un \omterm{def:g11:seq:sequence}{sucesión} \omterm{def:g11:seq:arithmetic}{aritmética}
es igual
\[
(\text{number of terms}) \times
\frac{\text{first term} + \text{last term}}{2}.
\]
\end{theorem}

\begin{proof}
escribe la suma $ S $ dos veces, la segunda vez en orden inverso, y sumar
columna por columna:
\[
\begin{array}{ccccccccc}
S & = & 1 & + & 2 & + & \dots & + & n\\
S & = & n & + & (n-1) & + & \dots & + & 1\\
\hline
2S & = & (n+1) & + & (n+1) & + & \dots & + & (n+1)
\end{array}
\]
Hay $ n $ columnas, cada una sumando $ n + 1$, entonces $2S = n(n+1)$. por un
generales \omterm{def:g11:seq:sequence}{sucesión} \omterm{def:g11:seq:arithmetic}{aritmética} el mismo emparejamiento funciona: primero $+$ último $=$
segundo $+$ penúltimo $= \dots $, porque dar un paso adelante en
el extremo izquierdo ($+d $) se compensa con un paso hacia atrás en el extremo derecho
($-d $).
\end{proof}

\begin{example}
$1 + 2 + \dots + 100 = \frac{100 \times 101}{2} = 5050$. la suma de la
números \omterm{def:g11:func:parity}{impares} $1 + 3 + \dots + 99$ ($50$ términos) es
$50 \times \frac{1 + 99}{2} = 2500$.
\end{example}

\section{Sucesiones geométricas}

\begin{definition}[sucesión geométrica]\label{def:g11:seq:geometric}
A \omterm{def:g11:seq:sequence}{sucesión} es \emph{geométrico}\index{geométrico} con \emph{común
relación}\index{común
relación} $ q \neq 0$ si cada término se obtiene del anterior por
multiplicando por $ q $:
\[
u_{n+1} = q\,u_n \quad \text{para todo } n.
\]
De manera equivalente, cuando ningún término desaparece: la proporción $\frac{u_{n+1}}{u_n}$ es
constante, igual a $ q $.
\end{definition}

\begin{theorem}[Término general]\label{thm:g11:seq:geomgeneral}
Si $(u_n)$ es \omterm{def:g11:seq:geometric}{geométrico} con primer término $ u_0$ y proporción $ q $, entonces
\[
u_n = u_0\, q^n \quad \text{para todo } n \geq 0,
\qquad\text{and more generally } u_n = u_p\, q^{\,n-p} .
\]
\end{theorem}

\begin{proof}
El mismo conteo de pasos que en \cref{thm:g11:seq:arithgeneral}: de $ u_0$ a
$ u_n $, la regla "multiplicar por $ q $'' se aplica $ n $ veces, aportando un
factor $ q^n $.
\end{proof}

\begin{theorem}[Suma geométrica]\label{thm:g11:seq:geomsum}
por cada real $ q \neq 1$ y entero $ n \geq 0$:
\[
1 + q + q^2 + \dots + q^n = \frac{1 - q^{\,n+1}}{1 - q}.
\]
\end{theorem}

\begin{proof}
Sea $ S = 1 + q + \dots + q^n $. multiplicar por $ q $:
$ qS = q + q^2 + \dots + q^{n+1}$. Sustraer:
\[
S - qS = \bigl(1 + q + \dots + q^n\bigr)
- \bigl(q + q^2 + \dots + q^{n+1}\bigr) = 1 - q^{\,n+1},
\]
porque cada término intermedio aparece una vez en cada suma y se cancela.
Por eso $(1 - q)S = 1 - q^{\,n+1}$ y dividiendo por $1 - q \neq 0$ da el
fórmula.
\end{proof}

\begin{example}\label{ex:g11:seq:chessboard}
$1 + 2 + 4 + \dots + 2^{10} = \frac{1 - 2^{11}}{1 - 2} = 2^{11} - 1 =
2047$: doblar granos de arroz en las casillas de un tablero de ajedrez abruma
cualquier granero mucho antes del $64$ ésimo cuadrado, donde el total es
$2^{64} - 1 \approx 1.8 \times 10^{19}$.
\end{example}

\begin{method}[Reconocer el tipo de sucesión]\label{met:g11:seq:recognize}
Calcular $ u_{n+1} - u_n $ y simplificar. Si el resultado es una constante $ d $, el
\omterm{def:g11:seq:sequence}{sucesión} es \omterm{def:g11:seq:arithmetic}{aritmética}. De lo contrario calcular $\frac{u_{n+1}}{u_n}$ (términos
distinto de cero) y simplificar: una constante $ q $ medio \omterm{def:g11:seq:geometric}{geométrico}. Si ninguno lo es
constante, la \omterm{def:g11:seq:sequence}{sucesión} no es de ningún tipo --- nunca concluya del
primeros términos solos.
\end{method}

\begin{example}
Para $ u_n = 3 \times 5^n $:
$\frac{u_{n+1}}{u_n} = \frac{3 \times 5^{n+1}}{3 \times 5^n} = 5$ para todo
$ n $: \omterm{def:g11:seq:geometric}{geométrico} con relación $5$. Para $ u_n = n^2$:
$ u_1 - u_0 = 1$ pero $ u_2 - u_1 = 3$, y
$\frac{u_1}{u_0}$ ni siquiera está definido --- tampoco \omterm{def:g11:seq:arithmetic}{aritmética} ni
\omterm{def:g11:seq:geometric}{geométrico}.
\end{example}

\section{Monotonía}

\begin{definition}[sucesión monótona]\label{def:g11:seq:monotone}
A \omterm{def:g11:seq:sequence}{sucesión} $(u_n)$ es \emph{\omterm{def:g11:func:monotone}{creciente}} si $ u_{n+1} \geq u_n $ para todo $ n $,
y \emph{\omterm{def:g10:functions:variations}{decreciente}} si $ u_{n+1} \leq u_n $ para todo $ n $.
\end{definition}

\begin{method}[Estudiar la monotonicidad]\label{met:g11:seq:monotonicity}
Estudia el signo de $ u_{n+1} - u_n $. Para sucesiones con términos positivos, uno
en su lugar puede comparar $\frac{u_{n+1}}{u_n}$ a $1$.
\end{method}

\begin{example}
Un \omterm{def:g11:seq:sequence}{sucesión} \omterm{def:g11:seq:arithmetic}{aritmética} es \omterm{def:g11:func:monotone}{creciente} cuando $ d \geq 0$
($ u_{n+1} - u_n = d $), \omterm{def:g10:functions:variations}{decreciente} cuando $ d \leq 0$. A \omterm{def:g11:seq:sequence}{sucesión} geométrica
con $ u_0 > 0$ y $ q > 1$ es \omterm{def:g11:func:monotone}{creciente}:
$ u_{n+1} - u_n = u_0 q^n (q - 1) > 0$; con $ u_0 > 0$ y $0 < q < 1$ es \omterm{def:g10:functions:variations}{decreciente}.
\end{example}

\section{Comportamiento a largo plazo, informalmente}

¿Qué le pasa a $ u_n $ cuando $ n $ se vuelve muy grande? para un \omterm{def:g11:seq:arithmetic}{aritmética}
\omterm{def:g11:seq:sequence}{sucesión} con $ d > 0$, los términos $ u_0 + nd $ exceder cualquier número fijo
eventualmente. por un \omterm{def:g11:seq:sequence}{sucesión} geométrica con $0 < q < 1$, los términos
$ u_0 q^n $ encogerse hacia $0$: multiplicar repetidamente por $0.9$, digamos, erosiona
cualquier valor inicial. y para $ q > 1$ los términos explotan, como en
\cref{ex:g11:seq:chessboard}.

\begin{remark}
Estas afirmaciones pueden hacerse perfectamente precisas: "los términos eventualmente
permanecer dentro de cualquier distancia dada de $0$'' --- y demostrado. esa es la teoria
de \emph{límites}, el tema de apertura de \cref{ch:g12:seq}.
\end{remark}

\section{Problema: La torre de Brahma y los conejos de Fibonacci}

\begin{problem}[{Problema del fin de semana --- dos recurrencias legendarias:
la torre que acaba con el mundo, la sucesión que crece como el oro,
y el truco auxiliar que doma los préstamos}]
\label{pb:g11:seq:1}
Dos sucesiones gobiernan el folklore de las matemáticas. Uno cuenta el
movimientos de la torre de Brahma --- sesenta y cuatro discos dorados cuyos
La transferencia, dice la leyenda, acabará con el mundo. el otro cuenta
Los conejos de Fibonacci y esconde la proporción áurea. Tampoco lo es
\omterm{def:g11:seq:arithmetic}{aritmética}, tampoco lo es \omterm{def:g11:seq:geometric}{geométrico} --- y ambos se entregan a esto
Armas del capítulo: recurrencias, \omterm{def:g11:seq:geometric}{geométrico} sumas
(\cref{thm:g11:seq:geomsum}), y el \omterm{def:g11:seq:sequence}{sucesión} auxiliar truco de
\cref{exo:g11:seq:11}, que también calcula su hipoteca.

\medskip
\noindent\textbf{Part I --- The tower of Brahma.}
El rompecabezas: $ n $ discos de \omterm{def:g10:functions:variations}{decreciente} los tamaños se apilan en la clavija A;
mueva toda la pila a la clavija C, un disco a la vez, sin colocar nunca
un disco más grande en uno más pequeño (la clavija B puede ayudar). Sea $ h_n $ ser el
número \omterm{def:g10:functions:extrema}{mínimo} de movimientos.
\begin{enumerate}
  \item Juega (con monedas) y graba. $ h_1$, $ h_2$, $ h_3$.
  \item Explica la estrategia detrás de la recurrencia.
        $ h_{n+1} = 2h_n + 1$: lo que debe pasar antes y después
        ¿Se mueve el disco más grande?
  \item Resuelve la recurrencia con el truco de
        \cref{exo:g11:seq:11}: colocar $ v_n = h_n + 1$, espectáculo
        $(v_n)$ es \omterm{def:g11:seq:geometric}{geométrico}, y concluir $ h_n = 2^n - 1$.
  \item La torre de la leyenda tiene $64$ discos, y los monjes se mueven
        un disco por segundo. Usando $2^{10} = 1024 \approx 10^3$,
        estimar el tiempo de transferencia en años (un año es aproximadamente
        $3 \times 10^7$ artículos de segunda clase; comparar
        \cref{ex:g11:seq:chessboard}, el mismo gigante en otro
        historia). ¿Deberíamos preocuparnos?
  \item ¿Por qué ninguna estrategia puede vencer? $2^n - 1$ se mueve? argumentar que
        \emph{cualquier} la solución obedece
        $ h_{n+1} \geq 2 h_n + 1$: ¿Qué debe ser cierto para la cima?
        $ n $ discos justo antes y justo después del disco inferior.
        moverse?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Fibonacci.}
Definir $ F_1 = F_2 = 1$ y $ F_{n+2} = F_{n+1} + F_n $ (cada término
la suma de los dos anteriores --- la regla de conteo de ritmo del
Volumen Middle School, ahora con nombre europeo).
\begin{enumerate}[resume]
  \item Lista $ F_1$ a través de $ F_{12}$.
  \item mostrar que $(F_n)$ tampoco lo es \omterm{def:g11:seq:arithmetic}{aritmética} ni \omterm{def:g11:seq:geometric}{geométrico},
        pero que es estrictamente \omterm{def:g11:func:monotone}{creciente} de $ n = 2$ en
        (\cref{met:g11:seq:monotonicity} y la recurrencia).
  \item Demostrar la identidad de la suma
        \[
        F_1 + F_2 + \dots + F_n = F_{n+2} - 1
        \]
        telescópico: escribe cada uno $ F_k $ como
        $ F_{k+2} - F_{k+1}$ y ver cómo se derrumba la suma. verificar
        eso para $ n = 6$.
  \item Demostrar la identidad de los cuadrados.
        $ F_1^2 + F_2^2 + \dots + F_n^2 = F_n F_{n+1}$,
        telescópico con
        $ F_k F_{k+1} - F_{k-1} F_k = F_k^2$. Verificar para
        $ n = 4$. (\omterm{def:g10:functions:function}{Imagen}: cuadrados de lados
        $1, 1, 2, 3, 5, \dots $ mosaico de un rectángulo --- el famoso
        El esqueleto de la espiral de Fibonacci.)
  \item Estados de identidad de Cassini
        $ F_{n+1} F_{n-1} - F_n^2 = (-1)^n $. Verifíquelo para
        $ n = 4, 5, 6$ --- y reconocer el motor del
        Truco del cuadrado fugaz jugado en la escuela secundaria
        Problema de áreas de volumen.
  \item Demuestre por la recurrencia que
        $ F_{n+2} \geq 2 F_n $: Fibonacci al menos se duplica cada
        dos pasos --- crece al menos tan rápido como un \omterm{def:g11:seq:geometric}{geométrico}
        \omterm{def:g11:seq:sequence}{sucesión} de proporción $\sqrt2$.
  \item Calcular las razones $ r_n = \frac{F_{n+1}}{F_n}$ para
        $ n = 3$ a $10$ (tres decimales). Admitiendo que ellos
        establecer un límite $ L $, pasa la relación
        $ r_{n+1} = 1 + \frac{1}{r_n}$ al límite y resolver:
        cual numero de \cref{pb:g10:algebra:1} hacer los conejos
        adoración?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The auxiliary trick, at the
bank.}
\begin{enumerate}[resume]
  \item Generalizar \cref{exo:g11:seq:11}: para
        $ u_{n+1} = a\,u_n + b $ con $ a \neq 1$, dejar
        $\ell = \frac{b}{1 - a}$ (el \omterm{pb:g10:functions:1}{punto fijo}). Espectáculo
        $ v_n = u_n - \ell $ es \omterm{def:g11:seq:geometric}{geométrico} de proporción $ a $, y
        concluir
        $ u_n = a^n (u_0 - \ell) + \ell $.
  \item Un préstamo: $10\,000$ euros en $1\,\%$ interés por mes,
        reembolsado $300$ euros al mes, por lo que la deuda obedece
        $ d_{n+1} = 1.01\,d_n - 300$. Aplicar la pregunta 13 (fijo
        punto ¡primero!) para obtener una fórmula explícita para $ d_n $.
  \item Con una calculadora, encuentre el primer mes en el que
        La deuda se liquida y se reembolsa el importe total. cuanto
        ¿Cuanto costó el endeudamiento?
  \item un pueblo de $50\,000$ habitantes crece en $2\,\%$ un año
