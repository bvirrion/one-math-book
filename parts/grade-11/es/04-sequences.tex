\chapter{Sucesiones: un primer curso}\label{ch:g11:seq}

Una \omterm{def:g11:seq:sequence}{sucesión} es una lista de números producida por una regla: los saldos
sucesivos de una cuenta de ahorro, el tamaño de una población año tras
año. Este capítulo estudia las dos familias que dominan las aplicaciones:
las progresiones \emph{aritméticas}, que crecen a pasos iguales, y las
\emph{geométricas}, que crecen en razones iguales. La teoría rigurosa de
los límites se desarrolla en el \cref{ch:g12:seq}.

\section{Definir una sucesión}

\begin{definition}[Sucesión]\label{def:g11:seq:sequence}
Una \emph{sucesión}\index{sucesión} $(u_n)$ asigna a cada entero
$n \geq 0$ (o $n \geq 1$) un \omterm{def:g10:numbers:sets}{número real} $u_n$, su \emph{término de índice
$n$}. Una sucesión se puede dar
\begin{itemize}
  \item de forma \emph{explícita}, mediante una fórmula para $u_n$ en
        \omterm{def:g11:func:function}{función} de $n$; por ejemplo, $u_n = n^2 + 1$;
  \item por \emph{recurrencia}, mediante su primer término y una regla
        para pasar de cada término al siguiente; por ejemplo, $u_0 = 3$ y
        $u_{n+1} = 2u_n - 1$.
\end{itemize}
\end{definition}

\begin{example}
Para $u_n = n^2 + 1$: $u_0 = 1$, $u_1 = 2$, $u_2 = 5$, y $u_{10} = 101$
directamente. Para $u_0 = 3$, $u_{n+1} = 2u_n - 1$: $u_1 = 5$, $u_2 = 9$,
$u_3 = 17$; cada término necesita el anterior, y llegar a $u_{10}$ cuesta
diez pasos (o una fórmula general, véase el \cref{exo:g11:seq:11}).
\end{example}

\section{Progresiones aritméticas}

\begin{definition}[Progresión aritmética]\label{def:g11:seq:arithmetic}
Una \omterm{def:g11:seq:sequence}{sucesión} es una \emph{progresión aritmética}\index{progresión
aritmética} de \emph{diferencia}\index{diferencia} $d$ si cada término se
obtiene del anterior sumándole $d$:
\[
u_{n+1} = u_n + d \quad \text{para todo } n.
\]
Equivalentemente: la diferencia $u_{n+1} - u_n$ es constante e igual a
$d$.
\end{definition}

\begin{theorem}[Término general]\label{thm:g11:seq:arithgeneral}
Si $(u_n)$ es una \omterm{def:g11:seq:arithmetic}{progresión aritmética} de primer término $u_0$ y
\omterm{def:g11:seq:arithmetic}{diferencia} $d$, entonces
\[
u_n = u_0 + n\,d \quad \text{para todo } n \geq 0,
\qquad\text{y, más en general, } u_n = u_p + (n - p)\,d .
\]
\end{theorem}

\begin{proof}
Para ir de $u_0$ a $u_n$, la regla ``sumar $d$'' se aplica $n$ veces: un
paso da $u_1 = u_0 + d$, dos pasos dan $u_2 = u_0 + 2d$ y, tras $n$ pasos,
cada aplicación ha aportado un $d$, luego $u_n = u_0 + nd$. (Este
``y así sucesivamente'' se hace riguroso por inducción en el
\cref{ch:g12:seq}.) La fórmula general se obtiene contando los $n - p$
pasos que van de $u_p$ a $u_n$.
\end{proof}

\begin{theorem}[Suma de enteros consecutivos]\label{thm:g11:seq:intsum}
Para todo entero $n \geq 1$:
\[
1 + 2 + \dots + n = \frac{n(n+1)}{2}.
\]
Más en general, una suma de términos consecutivos de una \omterm{def:g11:seq:arithmetic}{progresión
aritmética} vale
\[
(\text{número de términos}) \times
\frac{\text{primer término} + \text{último término}}{2}.
\]
\end{theorem}

\begin{proof}
Escribimos dos veces la suma $S$, la segunda en orden inverso, y sumamos
columna a columna:
\[
\begin{array}{ccccccccc}
S & = & 1 & + & 2 & + & \dots & + & n\\
S & = & n & + & (n-1) & + & \dots & + & 1\\
\hline
2S & = & (n+1) & + & (n+1) & + & \dots & + & (n+1)
\end{array}
\]
Hay $n$ columnas y cada una suma $n + 1$, luego $2S = n(n+1)$. Para una
\omterm{def:g11:seq:arithmetic}{progresión aritmética} cualquiera funciona el mismo emparejamiento:
primero $+$ último $=$ segundo $+$ penúltimo $= \dots$, porque avanzar un
paso por la izquierda ($+d$) se compensa con retroceder un paso por la
derecha ($-d$).
\end{proof}

\begin{example}
$1 + 2 + \dots + 100 = \frac{100 \times 101}{2} = 5050$. La suma de los
números \omterm{def:g11:func:parity}{impares} $1 + 3 + \dots + 99$ ($50$ términos) es
$50 \times \frac{1 + 99}{2} = 2500$.
\end{example}

\section{Progresiones geométricas}

\begin{definition}[Progresión geométrica]\label{def:g11:seq:geometric}
Una \omterm{def:g11:seq:sequence}{sucesión} es una \emph{progresión geométrica}\index{progresión
geométrica} de \emph{razón}\index{razón} $q \neq 0$ si cada término se
obtiene del anterior multiplicándolo por $q$:
\[
u_{n+1} = q\,u_n \quad \text{para todo } n.
\]
Equivalentemente, cuando ningún término se anula: el cociente
$\frac{u_{n+1}}{u_n}$ es constante e igual a $q$.
\end{definition}

\begin{theorem}[Término general]\label{thm:g11:seq:geomgeneral}
Si $(u_n)$ es una \omterm{def:g11:seq:geometric}{progresión geométrica} de primer término $u_0$ y \omterm{def:g11:seq:geometric}{razón}
$q$, entonces
\[
u_n = u_0\, q^n \quad \text{para todo } n \geq 0,
\qquad\text{y, más en general, } u_n = u_p\, q^{\,n-p} .
\]
\end{theorem}

\begin{proof}
El mismo recuento de pasos que en el \cref{thm:g11:seq:arithgeneral}: de
$u_0$ a $u_n$, la regla ``multiplicar por $q$'' se aplica $n$ veces, y
aporta un factor $q^n$.
\end{proof}

\begin{theorem}[Suma geométrica]\label{thm:g11:seq:geomsum}
Para todo real $q \neq 1$ y todo entero $n \geq 0$:
\[
1 + q + q^2 + \dots + q^n = \frac{1 - q^{\,n+1}}{1 - q}.
\]
\end{theorem}

\begin{proof}
Sea $S = 1 + q + \dots + q^n$. Multiplicamos por $q$:
$qS = q + q^2 + \dots + q^{n+1}$. Restamos:
\[
S - qS = \bigl(1 + q + \dots + q^n\bigr)
- \bigl(q + q^2 + \dots + q^{n+1}\bigr) = 1 - q^{\,n+1},
\]
porque cada término intermedio aparece una vez en cada suma y se cancela.
Por tanto, $(1 - q)S = 1 - q^{\,n+1}$, y dividiendo entre $1 - q \neq 0$
se obtiene la fórmula.
\end{proof}

\begin{example}\label{ex:g11:seq:chessboard}
$1 + 2 + 4 + \dots + 2^{10} = \frac{1 - 2^{11}}{1 - 2} = 2^{11} - 1 =
2047$: duplicar granos de arroz en las casillas de un tablero de ajedrez
desborda cualquier granero mucho antes de la casilla $64$, donde el total
es $2^{64} - 1 \approx 1.8 \times 10^{19}$.
\end{example}

\begin{method}[Reconocer el tipo de una sucesión]\label{met:g11:seq:recognize}
Calcula $u_{n+1} - u_n$ y simplifica. Si el resultado es una constante
$d$, la \omterm{def:g11:seq:sequence}{sucesión} es una \omterm{def:g11:seq:arithmetic}{progresión aritmética}. Si no, calcula
$\frac{u_{n+1}}{u_n}$ (con los términos no nulos) y simplifica: una
constante $q$ significa \omterm{def:g11:seq:geometric}{progresión geométrica}. Si ninguna de las dos
cosas es constante, la \omterm{def:g11:seq:sequence}{sucesión} no es de ninguno de los dos tipos; y no
concluyas nunca a partir de los primeros términos solamente.
\end{method}

\begin{example}
Para $u_n = 3 \times 5^n$:
$\frac{u_{n+1}}{u_n} = \frac{3 \times 5^{n+1}}{3 \times 5^n} = 5$ para
todo $n$: \omterm{def:g11:seq:geometric}{progresión geométrica} de \omterm{def:g11:seq:geometric}{razón} $5$. Para $u_n = n^2$:
$u_1 - u_0 = 1$, pero $u_2 - u_1 = 3$, y $\frac{u_1}{u_0}$ ni siquiera
está definido: no es ni aritmética ni geométrica.
\end{example}

\section{Monotonía}

\begin{definition}[Sucesión monótona]\label{def:g11:seq:monotone}
Una \omterm{def:g11:seq:sequence}{sucesión} $(u_n)$ es \emph{\omterm{def:g11:func:monotone}{creciente}} si $u_{n+1} \geq u_n$ para todo
$n$, y \emph{\omterm{def:g10:functions:variations}{decreciente}} si $u_{n+1} \leq u_n$ para todo $n$.
\end{definition}

\begin{method}[Estudiar la monotonía]\label{met:g11:seq:monotonicity}
Estudia el signo de $u_{n+1} - u_n$. Para sucesiones de términos
positivos, se puede comparar en su lugar $\frac{u_{n+1}}{u_n}$ con $1$.
\end{method}

\begin{example}
Una \omterm{def:g11:seq:arithmetic}{progresión aritmética} es \omterm{def:g11:func:monotone}{creciente} cuando $d \geq 0$
($u_{n+1} - u_n = d$) y \omterm{def:g10:functions:variations}{decreciente} cuando $d \leq 0$. Una \omterm{def:g11:seq:geometric}{progresión
geométrica} con $u_0 > 0$ y $q > 1$ es \omterm{def:g11:func:monotone}{creciente}:
$u_{n+1} - u_n = u_0 q^n (q - 1) > 0$; con $u_0 > 0$ y $0 < q < 1$ es
\omterm{def:g10:functions:variations}{decreciente}.
\end{example}

\section{Comportamiento a largo plazo, de manera informal}

¿Qué le ocurre a $u_n$ cuando $n$ se hace muy grande? En una \omterm{def:g11:seq:arithmetic}{progresión
aritmética} con $d > 0$, los términos $u_0 + nd$ acaban superando cualquier
número fijo. En una \omterm{def:g11:seq:geometric}{progresión geométrica} con $0 < q < 1$, los términos
$u_0 q^n$ se encogen hacia $0$: multiplicar repetidamente por $0.9$, por
ejemplo, erosiona cualquier valor de partida. Y para $q > 1$ los términos
explotan, como en el \cref{ex:g11:seq:chessboard}.

\begin{remark}
Estas afirmaciones se pueden hacer perfectamente precisas (``los términos
acaban quedándose a menos de cualquier distancia dada de $0$'') y
demostrar. Esa es la teoría de los \emph{límites}, con la que se abre el
\cref{ch:g12:seq}.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:seq:1}
Para cada \omterm{def:g11:seq:sequence}{sucesión}, calcula $u_1$, $u_2$ y $u_3$:
\[
u_n = \frac{n}{n+1}; \qquad
u_0 = 5,\ u_{n+1} = 3u_n - 2; \qquad
u_n = (-1)^n\,n .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:2}
$(u_n)$ es una \omterm{def:g11:seq:arithmetic}{progresión aritmética} con $u_0 = 7$ y $d = -3$. Calcula
$u_{10}$ y $u_{25}$. $(v_n)$ es una \omterm{def:g11:seq:arithmetic}{progresión aritmética} con $v_3 = 11$ y
$v_8 = 26$. Halla la \omterm{def:g11:seq:arithmetic}{diferencia} y $v_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:3}
$(u_n)$ es una \omterm{def:g11:seq:geometric}{progresión geométrica} con $u_0 = 5$ y $q = 2$. Calcula
$u_8$. $(v_n)$ es una \omterm{def:g11:seq:geometric}{progresión geométrica} de términos positivos con
$v_2 = 12$ y $v_4 = 48$. Halla la \omterm{def:g11:seq:geometric}{razón} y $v_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:4}
Calcula
\[
1 + 2 + 3 + \dots + 500, \qquad
4 + 7 + 10 + \dots + 61, \qquad
1 + \frac12 + \frac14 + \dots + \frac{1}{2^{10}} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:5}
Determina si cada \omterm{def:g11:seq:sequence}{sucesión} es una \omterm{def:g11:seq:arithmetic}{progresión aritmética}, una geométrica o
ninguna de las dos:
\[
u_n = 4n - 1; \qquad
v_n = \frac{2^n}{3^{n+1}}; \qquad
w_n = n^2 + n .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:6}
Un teatro tiene $20$ filas: $16$ butacas en la primera fila, y cada fila
tiene $2$ butacas más que la anterior. ¿Cuántas butacas hay en la última
fila? ¿Y en todo el teatro?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:7}
Una población de bacterias se duplica cada hora; al mediodía hay $500$
bacterias. ¿Cuántas hay a las 8 de la tarde? ¿Al cabo de cuántas horas
completas supera por primera vez el millón? (Resuélvelo probando potencias
sucesivas de $2$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:8}
Cada mes, un ahorrador ingresa $100$ euros en una cuenta que paga un
$0.2\,\%$ de interés mensual sobre el saldo existente (los intereses se
abonan justo antes del ingreso). Sea $c_n$ el saldo inmediatamente después
del ingreso $n$-ésimo, de modo que $c_1 = 100$ y
$c_{n+1} = 1.002\,c_n + 100$. Calcula $c_2$ y $c_3$ y explica por qué
$(c_n)$ no es ni una \omterm{def:g11:seq:arithmetic}{progresión aritmética} ni una geométrica.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:9}
Estudia la monotonía de las sucesiones
\[
u_n = n^2 - 8n \ (n \geq 0), \qquad
v_n = \frac{3^n}{n!}\ (n \geq 1),
\]
donde $n! = 1 \times 2 \times \dots \times n$. (Para $(v_n)$, compara
$\frac{v_{n+1}}{v_n}$ con $1$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:10}
La suma de los $n$ primeros términos de una \omterm{def:g11:seq:arithmetic}{progresión aritmética} con
$u_0 = 3$ y $d = 4$ vale $903$. Halla $n$. (Plantea una \omterm{def:g10:algebra:equation}{ecuación} de
segundo grado en $n$ y usa el \cref{ch:g11:quad}.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:seq:11}
Sean $u_0 = 3$ y $u_{n+1} = 2u_n - 1$.
\begin{enumerate}
  \item Calcula $u_1, u_2, u_3$ y conjetura una fórmula para $u_n$.
  \item Sea $v_n = u_n - 1$. Demuestra que $(v_n)$ es una \omterm{def:g11:seq:geometric}{progresión
        geométrica} y da su \omterm{def:g11:seq:geometric}{razón} y su primer término.
  \item Deduce una fórmula explícita para $u_n$ y comprueba tu conjetura.
\end{enumerate}
\end{exercise}

\section{Problema: La torre de Brahma y los conejos de Fibonacci}

\begin{problem}[{Problema de fin de semana --- dos recurrencias
legendarias: la torre que acaba con el mundo, la sucesión que crece como
el oro y el truco auxiliar que domestica los préstamos}]
\label{pb:g11:seq:1}
Dos sucesiones gobiernan el folclore de las matemáticas. Una cuenta los
movimientos de la torre de Brahma, sesenta y cuatro discos de oro cuyo
traslado, según la leyenda, acabará con el mundo. La otra cuenta los
conejos de Fibonacci y esconde el número áureo. Ninguna de las dos es
aritmética ni geométrica, y las dos se rinden ante las armas de este
capítulo: las recurrencias, las sumas geométricas
(\cref{thm:g11:seq:geomsum}) y el truco de la \omterm{def:g11:seq:sequence}{sucesión} auxiliar del
\cref{exo:g11:seq:11}, que además calcula tu hipoteca.

\medskip
\noindent\textbf{Parte I --- La torre de Brahma.}
El rompecabezas: hay $n$ discos de tamaño \omterm{def:g10:functions:variations}{decreciente} apilados en la
varilla A; hay que trasladar la pila entera a la varilla C, de disco en
disco y sin colocar nunca un disco grande sobre uno pequeño (la varilla B
puede ayudar). Sea $h_n$ el número \omterm{def:g10:functions:extrema}{mínimo} de movimientos.
\begin{enumerate}
  \item Juega (con monedas) y anota $h_1$, $h_2$ y $h_3$.
  \item Explica la estrategia que hay detrás de la recurrencia
        $h_{n+1} = 2h_n + 1$: ¿qué tiene que ocurrir antes y después de
        que se mueva el disco mayor?
  \item Resuelve la recurrencia con el truco del \cref{exo:g11:seq:11}:
        toma $v_n = h_n + 1$, demuestra que $(v_n)$ es una \omterm{def:g11:seq:geometric}{progresión
        geométrica} y concluye que $h_n = 2^n - 1$.
  \item La torre de la leyenda tiene $64$ discos y los monjes mueven un
        disco por segundo. Usando $2^{10} = 1024 \approx 10^3$, estima el
        tiempo de traslado en años (un año son unos $3 \times 10^7$
        segundos; compara con el \cref{ex:g11:seq:chessboard}, el mismo
        gigante en otra historia). ¿Hay que preocuparse?
  \item ¿Por qué ninguna estrategia puede bajar de $2^n - 1$ movimientos?
        Argumenta que \emph{cualquier} solución cumple
        $h_{n+1} \geq 2 h_n + 1$: ¿qué tiene que ser cierto sobre los $n$
        discos superiores justo antes y justo después del movimiento del
        disco de abajo?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Fibonacci.}
Definimos $F_1 = F_2 = 1$ y $F_{n+2} = F_{n+1} + F_n$ (cada término es la
suma de los dos anteriores: la regla del recuento de ritmos del volumen
anterior, ahora con su nombre europeo).
\begin{enumerate}[resume]
  \item Escribe $F_1$ hasta $F_{12}$.
  \item Demuestra que $(F_n)$ no es ni aritmética ni geométrica, pero que
        es estrictamente \omterm{def:g11:func:monotone}{creciente} a partir de $n = 2$
        (\cref{met:g11:seq:monotonicity} y la recurrencia).
  \item Demuestra la identidad de la suma
        \[
        F_1 + F_2 + \dots + F_n = F_{n+2} - 1
        \]
        por telescopaje: escribe cada $F_k$ como $F_{k+2} - F_{k+1}$ y
        observa cómo se derrumba la suma. Compruébala para $n = 6$.
  \item Demuestra la identidad de los cuadrados
        $F_1^2 + F_2^2 + \dots + F_n^2 = F_n F_{n+1}$, telescopando con
        $F_k F_{k+1} - F_{k-1} F_k = F_k^2$. Compruébala para $n = 4$.
        (\omterm{def:g10:functions:function}{Imagen}: cuadrados de lados $1, 1, 2, 3, 5, \dots$ embaldosan un
        rectángulo, el esqueleto de la famosa espiral de Fibonacci.)
  \item La identidad de Cassini afirma que
        $F_{n+1} F_{n-1} - F_n^2 = (-1)^n$. Compruébala para
        $n = 4, 5, 6$ y reconoce en ella el motor del truco del cuadrado
        que desaparece, jugado en el problema de las áreas del volumen
        anterior.
  \item Demuestra a partir de la recurrencia que $F_{n+2} \geq 2 F_n$:
        Fibonacci se duplica al menos cada dos pasos, así que crece al
        menos tan deprisa como una \omterm{def:g11:seq:geometric}{progresión geométrica} de \omterm{def:g11:seq:geometric}{razón}
        $\sqrt2$.
  \item Calcula los cocientes $r_n = \frac{F_{n+1}}{F_n}$ para $n = 3$
        hasta $10$ (con tres decimales). Admitiendo que se estabilizan en
        un límite $L$, pasa al límite en la relación
        $r_{n+1} = 1 + \frac{1}{r_n}$ y resuelve: ¿a qué número del
        \cref{pb:g10:algebra:1} veneran los conejos?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El truco auxiliar, en el banco.}
\begin{enumerate}[resume]
  \item Generaliza el \cref{exo:g11:seq:11}: para $u_{n+1} = a\,u_n + b$
        con $a \neq 1$, toma $\ell = \frac{b}{1 - a}$ (el \omterm{pb:g10:functions:1}{punto fijo}).
        Demuestra que $v_n = u_n - \ell$ es una \omterm{def:g11:seq:geometric}{progresión geométrica} de
        \omterm{def:g11:seq:geometric}{razón} $a$ y concluye que $u_n = a^n (u_0 - \ell) + \ell$.
  \item Un préstamo: $10\,000$ euros al $1\,\%$ de interés mensual, con
        una devolución de $300$ euros al mes, de modo que la deuda cumple
        $d_{n+1} = 1.01\,d_n - 300$. Aplica la pregunta 13 (¡primero el
        \omterm{pb:g10:functions:1}{punto fijo}!) para obtener una fórmula explícita de $d_n$.
  \item Con una calculadora, halla el primer mes en que queda saldada la
        deuda y el importe total devuelto. ¿Cuánto ha costado el préstamo
        en sí?
  \item Una ciudad de $50\,000$ habitantes crece un $2\,\%$ al año y
        recibe además $1\,000$ recién llegados:
        $p_{n+1} = 1.02\,p_n + 1000$. Da la fórmula explícita y la
        población al cabo de $10$ años.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Las dos familias reales.}
\begin{enumerate}[resume]
  \item Calcula $1 + 2 + 3 + \dots + 1000$ (\cref{thm:g11:seq:intsum}: la
        suma del pequeño Gauss del volumen anterior, ahora oficial) y
        $1 + 2 + 4 + \dots + 2^{19}$ (\cref{thm:g11:seq:geomsum}).
  \item Calcula la suma de la \omterm{def:g11:seq:arithmetic}{progresión aritmética}
        $7, 12, 17, \dots, 502$ (¿cuántos términos hay?).
  \item Plan de ahorro: $100$ euros ingresados cada mes, con un
        rendimiento del $0.5\,\%$ mensual; después del ingreso $n$-ésimo,
        el saldo es $100\left(1.005^{n-1} + \dots + 1.005 + 1\right)$.
        Calcula el saldo al cabo de $5$ años ($n = 60$).
  \item Final, el equipo del domador de sucesiones: descripciones
        explícitas frente a recurrentes; las dos familias reales y sus
        fórmulas de suma; la \omterm{def:g11:seq:sequence}{sucesión} auxiliar que convierte las
        recurrencias afines en geométricas; y Fibonacci, primer ciudadano
        ajeno a las dos familias, domesticado hoy con identidades y a la
        espera de las matrices (el año que viene) y de los límites para su
        captura completa. Una frase para cada punto.
\end{enumerate}
\end{problem}
