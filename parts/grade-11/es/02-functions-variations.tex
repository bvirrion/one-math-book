\chapter{Funciones y variación}\label{ch:g11:func}

Este capítulo organiza lo que hay que saber sobre funciones antes de
entrar en el cálculo: la pequeña familia de \emph{\omterm{def:g11:func:reference}{funciones de referencia}}
cuyas \omterm{def:g10:functions:graph}{gráficas} hay que saberse de memoria, el significado preciso de
\emph{\omterm{def:g11:func:monotone}{creciente}} y \emph{\omterm{def:g10:functions:variations}{decreciente}}, y cómo se comporta la variación al
combinar funciones.

\section{Funciones y curvas}

\begin{definition}[Función, dominio, curva]\label{def:g11:func:function}
Una \emph{función}\index{función} $f$ asigna a cada número $x$ de un
conjunto $D \subseteq \R$ (su \emph{dominio}\index{dominio}) exactamente
un número $f(x)$. Su \emph{curva} (o \omterm{def:g10:functions:graph}{gráfica}) es el conjunto de los puntos
$\bigl(x, f(x)\bigr)$ con $x \in D$.
\end{definition}

\begin{example}
La fórmula $f(x) = \frac{\sqrt{x+2}}{x-1}$ exige $x + 2 \geq 0$ (por la
\omterm{def:g11:quad:discriminant}{raíz} cuadrada) y $x \neq 1$ (denominador no nulo): el \omterm{def:g11:func:function}{dominio} es
$\intco{-2}{1} \cup \intoo{1}{+\infty}$.
\end{example}

\section{Las funciones de referencia}

Las cinco funciones siguientes y sus curvas tienen que convertirse en un
reflejo.

\begin{definition}[Funciones de referencia]\label{def:g11:func:reference}
\index{funciones de referencia}
\[
x \mapsto x^2 \text{ (cuadrado)}, \quad
x \mapsto x^3 \text{ (cubo)}, \quad
x \mapsto \sqrt{x}, \quad
x \mapsto \frac{1}{x}, \quad
x \mapsto \abs{x}.
\]
Sus \omterm{def:g11:func:function}{dominios} son, respectivamente, $\R$, $\R$, $\intco{0}{+\infty}$,
$\R \setminus \{0\}$ y $\R$. La curva de $x \mapsto \frac1x$ es una
\emph{hipérbola}\index{hipérbola}.
\end{definition}

\section{Variación en un intervalo}

\begin{definition}[Creciente, decreciente]\label{def:g11:func:monotone}
Sea $f$ definida en un \omterm{def:g10:numbers:interval}{intervalo} $I$. La \omterm{def:g11:func:function}{función} $f$ es
\emph{creciente}\index{función creciente} en $I$ si conserva el orden:
\[
\text{para todos } u, v \in I: \quad u < v \implies f(u) \leq f(v),
\]
y \emph{\omterm{def:g10:functions:variations}{decreciente}} si lo invierte ($u < v \implies f(u) \geq f(v)$). Con
desigualdades estrictas a la derecha, $f$ es \emph{estrictamente}
creciente (resp.\ \omterm{def:g10:functions:variations}{decreciente}). Una \omterm{def:g11:func:function}{función} creciente o \omterm{def:g10:functions:variations}{decreciente} en $I$
se dice \emph{monótona} en $I$.
\end{definition}

\begin{method}[Demostrar una variación a partir de la definición]\label{met:g11:func:variation}
Toma dos números cualesquiera $u < v$ del \omterm{def:g10:numbers:interval}{intervalo} y determina el signo
de la diferencia $f(v) - f(u)$, normalmente \emph{factorizándola}. Si el
signo es constante en el \omterm{def:g10:numbers:interval}{intervalo}, concluye.
\end{method}

\begin{proposition}[Variación de las funciones de referencia]\label{prop:g11:func:refvariations}
\begin{enumerate}
  \item $x^2$ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\intoc{-\infty}{0}$ y
        estrictamente \omterm{def:g11:func:monotone}{creciente} en $\intco{0}{+\infty}$.
  \item $x^3$ es estrictamente \omterm{def:g11:func:monotone}{creciente} en $\R$.
  \item $\sqrt x$ es estrictamente \omterm{def:g11:func:monotone}{creciente} en $\intco{0}{+\infty}$.
  \item $\frac1x$ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\intoo{-\infty}{0}$ y
        en $\intoo{0}{+\infty}$ (pero \emph{no} en su \omterm{def:g10:numbers:interunion}{unión}).
  \item $\abs x$ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\intoc{-\infty}{0}$ y
        estrictamente \omterm{def:g11:func:monotone}{creciente} en $\intco{0}{+\infty}$.
\end{enumerate}
\end{proposition}

\begin{proof}
Sea $u < v$. Factorizamos $f(v) - f(u)$ en cada caso.

\emph{1.} $v^2 - u^2 = (v-u)(v+u)$. En $\intco{0}{+\infty}$: $v - u > 0$ y
$v + u > 0$ (porque $v > u \geq 0$), luego $v^2 > u^2$. En
$\intoc{-\infty}{0}$: $v + u < 0$, luego $v^2 < u^2$.

\emph{2.} $v^3 - u^3 = (v - u)(v^2 + uv + u^2)$, y el segundo factor vale
$\left(v + \frac u2\right)^2 + \frac{3u^2}{4} > 0$ salvo si
$u = v = 0$. Por tanto, $v^3 > u^3$ siempre que $v > u$.

\emph{3.} Para $0 \leq u < v$:
$\sqrt v - \sqrt u = \frac{(\sqrt v - \sqrt u)(\sqrt v + \sqrt u)}
{\sqrt v + \sqrt u} = \frac{v - u}{\sqrt v + \sqrt u} > 0$.

\emph{4.} Para $0 < u < v$: $\frac1v - \frac1u = \frac{u - v}{uv} < 0$,
porque $u - v < 0$ y $uv > 0$; el mismo cálculo vale para $u < v < 0$. En
la \omterm{def:g10:numbers:interunion}{unión} la afirmación falla: $-1 < 1$, pero $\frac{1}{-1} = -1 < 1 =
\frac11$.

\emph{5.} En $\intco{0}{+\infty}$, $\abs x = x$ es estrictamente
\omterm{def:g11:func:monotone}{creciente}; en $\intoc{-\infty}{0}$, $\abs x = -x$ es estrictamente
\omterm{def:g10:functions:variations}{decreciente}.
\end{proof}

\begin{remark}
El punto 4 es una trampa clásica: ``\omterm{def:g10:functions:variations}{decreciente} en $\intoo{-\infty}{0}$ y
en $\intoo{0}{+\infty}$'' \emph{no} es lo mismo que ``\omterm{def:g10:functions:variations}{decreciente} en
$\R \setminus \{0\}$''. La variación solo tiene sentido en \omterm{def:g10:numbers:interval}{intervalos}.
\end{remark}

\begin{example}[Comparar sin calcular]
Como $\sqrt x$ es estrictamente \omterm{def:g11:func:monotone}{creciente}, $\sqrt{7} < \sqrt{8}$; como
$x^2$ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\intoc{-\infty}{0}$,
$(-3)^2 > (-2)^2$. La monotonía convierte las comparaciones de imágenes en
comparaciones de argumentos.
\end{example}

\section{Funciones nuevas a partir de otras}

\begin{proposition}[Transformaciones]\label{prop:g11:func:transforms}
Sea $u$ una \omterm{def:g11:func:function}{función} definida en un \omterm{def:g10:numbers:interval}{intervalo} $I$ y sea $k \in \R$.
\begin{enumerate}
  \item $u + k$ tiene la \emph{misma} variación que $u$ en $I$; su curva
        es la de $u$ desplazada verticalmente en $k$.
  \item Para $\lambda > 0$, $\lambda u$ tiene la misma variación que $u$;
        para $\lambda < 0$, la variación se invierte.
  \item Si $u \geq 0$ en $I$, entonces $\sqrt u$ tiene la misma variación
        que $u$.
  \item Si $u \neq 0$ y $u$ tiene signo constante en $I$, entonces
        $\frac1u$ tiene la variación \emph{contraria} a la de $u$.
\end{enumerate}
\end{proposition}

\begin{proof}
Sean $x_1 < x_2$ en $I$ y escribamos $s = u(x_1)$, $t = u(x_2)$ para sus
imágenes.

\emph{1.} $(u+k)(x_2) - (u+k)(x_1) = t - s$: la diferencia de imágenes no
cambia, luego tampoco su signo.

\emph{2.} $\lambda t - \lambda s = \lambda(t - s)$: el signo se conserva
si $\lambda > 0$ y se invierte si $\lambda < 0$.

\emph{3.} Si $u$ es \omterm{def:g11:func:monotone}{creciente} y $x_1 < x_2$, entonces $0 \leq s \leq t$, y
como la \omterm{def:g11:quad:discriminant}{raíz} cuadrada es \omterm{def:g11:func:monotone}{creciente}
(\cref{prop:g11:func:refvariations}) se obtiene $\sqrt s \leq \sqrt t$. El
mismo argumento vale en el caso \omterm{def:g10:functions:variations}{decreciente}.

\emph{4.} Si $u$ es \omterm{def:g11:func:monotone}{creciente} y $0 < s \leq t$ (o $s \leq t < 0$),
entonces $\frac1s \geq \frac1t$, porque $x \mapsto \frac1x$ es \omterm{def:g10:functions:variations}{decreciente}
a cada lado de $0$: el orden se invierte.
\end{proof}

\begin{example}\label{ex:g11:func:composite}
Estudiemos $f(x) = \sqrt{x^2 + 1}$ en $\R$. La \omterm{def:g11:func:function}{función} interior
$u(x) = x^2 + 1$ es positiva, \omterm{def:g10:functions:variations}{decreciente} en $\intoc{-\infty}{0}$ y
\omterm{def:g11:func:monotone}{creciente} en $\intco{0}{+\infty}$ (un cuadrado desplazado). Por el punto
3, $f$ tiene la misma variación: \omterm{def:g10:functions:variations}{decreciente} y después \omterm{def:g11:func:monotone}{creciente}, con
\omterm{def:g10:functions:extrema}{mínimo} $f(0) = 1$.
\end{example}

\section{Funciones pares e impares}

\begin{definition}[Paridad]\label{def:g11:func:parity}
Sea $f$ definida en un \omterm{def:g11:func:function}{dominio} $D$ simétrico respecto de $0$ (es decir,
$-x \in D$ siempre que $x \in D$). La \omterm{def:g11:func:function}{función} $f$ es
\emph{par}\index{función par} si $f(-x) = f(x)$ para todo $x \in D$, e
\emph{impar}\index{función impar} si $f(-x) = -f(x)$ para todo $x \in D$.
\end{definition}

\begin{proposition}[Significado geométrico]\label{prop:g11:func:paritysym}
La curva de una \omterm{def:g11:func:parity}{función par} es simétrica respecto del eje $y$; la curva de
una \omterm{def:g11:func:parity}{función impar} es simétrica respecto del origen.
\end{proposition}

\begin{proof}
Si $f$ es \omterm{def:g11:func:parity}{par} y $(x, y)$ está en la curva (o sea, $y = f(x)$), entonces
$f(-x) = f(x) = y$: el punto simétrico $(-x, y)$ también está en la curva.
Si $f$ es \omterm{def:g11:func:parity}{impar}, $f(-x) = -y$: el punto $(-x, -y)$, simétrico de $(x,y)$
respecto del origen, está en la curva.
\end{proof}

\begin{example}
$x^2$ y $\abs x$ son \omterm{def:g11:func:parity}{pares}; $x^3$ y $\frac1x$ son \omterm{def:g11:func:parity}{impares};
$f(x) = x^2 + x$ no es ni lo uno ni lo otro: $f(-1) = 0$ mientras que
$f(1) = 2$, luego $f(-1) \neq \pm f(1)$. La paridad reduce el trabajo a la
mitad: una \omterm{def:g11:func:parity}{función par} o \omterm{def:g11:func:parity}{impar} solo hay que estudiarla en
$\intco{0}{+\infty}$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:func:1}
Da el \omterm{def:g11:func:function}{dominio} de cada \omterm{def:g11:func:function}{función}:
\[
f(x) = \sqrt{3 - x}, \qquad
g(x) = \frac{1}{x^2 - 4}, \qquad
h(x) = \frac{\sqrt{x}}{x - 5} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:func:2}
Usando la variación de las \omterm{def:g11:func:reference}{funciones de referencia}, compara sin
calculadora:
\[
\sqrt{11} \text{ y } \sqrt{13}; \qquad
(-2.1)^2 \text{ y } (-2.05)^2; \qquad
\frac{1}{0.9} \text{ y } \frac{1}{1.1}; \qquad
(-1.01)^3 \text{ y } (-0.99)^3 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:func:3}
Determina si cada \omterm{def:g11:func:function}{función} es \omterm{def:g11:func:parity}{par}, \omterm{def:g11:func:parity}{impar} o ninguna de las dos cosas:
\[
f(x) = 3x^4 - x^2, \qquad
g(x) = x^3 + x, \qquad
h(x) = x^2 + x^3, \qquad
k(x) = \frac{x}{x^2+1} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:func:4}
Una \omterm{def:g11:func:function}{función} $f$ tiene el siguiente comportamiento: crece en
$\intcc{-3}{1}$ desde $f(-3) = -2$ hasta $f(1) = 4$ y decrece después en
$\intcc{1}{5}$ hasta $f(5) = 0$. ¿Cuántas soluciones tiene la \omterm{def:g10:algebra:equation}{ecuación}
$f(x) = 1$ en $\intcc{-3}{5}$? ¿Y $f(x) = 5$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:5}
Usando el \cref{met:g11:func:variation} (el signo de $f(v) - f(u)$),
demuestra que $f(x) = x^2 - 4x$ es estrictamente \omterm{def:g11:func:monotone}{creciente} en
$\intco{2}{+\infty}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:6}
Sin calcular ninguna derivada, da la variación de
\[
f(x) = \sqrt{5 - x} \ \text{ en } \intoc{-\infty}{5},
\quad
g(x) = \frac{1}{x^2 + 1} \ \text{ en } \intco{0}{+\infty},
\quad
h(x) = 3 - 2\sqrt{x} \ \text{ en } \intco{0}{+\infty}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:7}
Sea $f(x) = (x-2)^2 - 3$ en $\R$.
\begin{enumerate}
  \item Da la variación de $f$ y su \omterm{def:g10:functions:extrema}{mínimo}.
  \item Deduce la variación de $g(x) = \frac{1}{(x-2)^2+1}$ y su \omterm{def:g10:functions:extrema}{máximo}.
        (Observa que $g = \frac{1}{f + 4}$.)
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:8}
La curva de una \omterm{def:g11:func:function}{función} $f$ pasa por los puntos $(0, 1)$, $(2, 3)$ y
$(4, 1)$, y $f$ es \omterm{def:g11:func:monotone}{creciente} en $\intcc{0}{2}$ y \omterm{def:g10:functions:variations}{decreciente} en
$\intcc{2}{4}$. Esboza una curva posible y esboza después las curvas de
$f + 2$, de $-f$ y de $2f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:9}
Demuestra que la \omterm{def:g11:func:function}{función} $f(x) = \frac{2x+1}{x+1}$ se puede escribir
$f(x) = 2 - \frac{1}{x+1}$ y deduce su variación en
$\intoo{-1}{+\infty}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:10}
¿Verdadero o falso? (Justifícalo o da un contraejemplo.)
\begin{enumerate}
  \item La suma de dos funciones \omterm{def:g11:func:monotone}{crecientes} en $I$ es \omterm{def:g11:func:monotone}{creciente} en $I$.
  \item El producto de dos funciones \omterm{def:g11:func:monotone}{crecientes} en $I$ es \omterm{def:g11:func:monotone}{creciente} en
        $I$.
  \item Si $f$ es \omterm{def:g11:func:monotone}{creciente} en $I$, entonces $f^2$ ($= f \times f$) es
        \omterm{def:g11:func:monotone}{creciente} en $I$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:func:11}
Sea $f(x) = x + \frac{1}{x}$ para $x > 0$.
\begin{enumerate}
  \item Demuestra que para $0 < u < v$ se cumple
        $f(v) - f(u) = (v - u)\,\frac{uv - 1}{uv}$.
  \item Deduce que $f$ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\intoc{0}{1}$ y
        estrictamente \omterm{def:g11:func:monotone}{creciente} en $\intco{1}{+\infty}$.
  \item Concluye que $x + \frac1x \geq 2$ para todo $x > 0$, con igualdad
        solo en $x = 1$.
\end{enumerate}
\end{exercise}

\section{Problema: La fábrica de simetrías}

\begin{problem}[{Problema de fin de semana --- toda función es una parte
par más una parte impar: reglas de paridad, el juego de las
transformaciones y mínimos ganados sin cálculo diferencial}]
\label{pb:g11:func:1}
Algunas funciones son simétricas como en un espejo, otras lo son respecto
de un medio giro y la mayoría no son ni una cosa ni otra; y sin embargo un
pequeño milagro algebraico afirma que \emph{toda} \omterm{def:g11:func:function}{función} es, de una única
manera, la suma de una pieza simétrica de espejo y una pieza simétrica de
medio giro. Este problema demuestra el milagro, juega al juego de las
transformaciones sobre las curvas de referencia
(\cref{prop:g11:func:transforms}) y cierra con la técnica de \omterm{def:g10:algebra:expand}{factorizar}
diferencias del \cref{exo:g11:func:11}, que gana problemas de minimización
un capítulo entero antes de que llegue la derivada.

\medskip
\noindent\textbf{Parte I --- Soltura con la paridad.}
\begin{enumerate}
  \item ¿\omterm{def:g11:func:parity}{Par}, \omterm{def:g11:func:parity}{impar} o ninguna de las dos cosas
        (\cref{def:g11:func:parity})? Justifica cada caso: $x^4 - 3x^2$;
        \ $x^3 - x$; \ $x^2 + x$; \ $\abs x$; \ $\frac1x$.
  \item Enuncia el significado geométrico de cada veredicto
        (\cref{prop:g11:func:paritysym}): ¿qué simetría tiene la curva de
        una \omterm{def:g11:func:parity}{función par}? ¿Y la de una \omterm{def:g11:func:parity}{impar}?
  \item La paridad se multiplica como los signos: demuestra que el
        producto de dos funciones \omterm{def:g11:func:parity}{impares} es \omterm{def:g11:func:parity}{par} y completa la tabla de
        multiplicar entera (\omterm{def:g11:func:parity}{par} $\times$ \omterm{def:g11:func:parity}{par}, \omterm{def:g11:func:parity}{par} $\times$ \omterm{def:g11:func:parity}{impar}, \omterm{def:g11:func:parity}{impar}
        $\times$ \omterm{def:g11:func:parity}{impar}).
  \item Consecuencias rápidas: si $f$ es \omterm{def:g11:func:parity}{impar} y está definida en $0$,
        ¿cuánto vale $f(0)$? Si $f$ es \omterm{def:g11:func:parity}{par} y $f(3) = 7$, ¿cuánto vale
        $f(-3)$? Y demuestra que la única \omterm{def:g11:func:function}{función} a la vez \omterm{def:g11:func:parity}{par} e \omterm{def:g11:func:parity}{impar} es
        la \omterm{def:g11:func:function}{función} nula.
  \item ¿Qué parte de la curva de una \omterm{def:g11:func:parity}{función par} determina toda la
        curva? Da una \omterm{def:g11:func:function}{función} que no sea ni \omterm{def:g11:func:parity}{par} ni \omterm{def:g11:func:parity}{impar} y explica qué
        falla.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El teorema de descomposición.}
Para una \omterm{def:g11:func:function}{función} $f$ definida en un \omterm{def:g11:func:function}{dominio} simétrico respecto de $0$,
tomemos
\[
E(x) = \frac{f(x) + f(-x)}{2},
\qquad
O(x) = \frac{f(x) - f(-x)}{2} .
\]
\begin{enumerate}[resume]
  \item Calcula $E$ y $O$ para $f(x) = x^3 + x^2 + 1$ y comprueba que $E$
        es \omterm{def:g11:func:parity}{par}, $O$ es \omterm{def:g11:func:parity}{impar} y $E + O = f$.
  \item Demuestra el enunciado general: para cualquier $f$, la \omterm{def:g11:func:function}{función} $E$
        es \omterm{def:g11:func:parity}{par}, $O$ es \omterm{def:g11:func:parity}{impar} y $f = E + O$; \emph{toda \omterm{def:g11:func:function}{función} se
        descompone en una parte \omterm{def:g11:func:parity}{par} y una parte \omterm{def:g11:func:parity}{impar}}.
  \item Demuestra que la descomposición es única: si
        $f = E_1 + O_1 = E_2 + O_2$ con $E_i$ \omterm{def:g11:func:parity}{par} y $O_i$ \omterm{def:g11:func:parity}{impar}, usa la
        pregunta 4 sobre $E_1 - E_2$.
  \item Descompón $f(x) = \dfrac{1}{1 - x}$ (para $x \neq \pm 1$):
        reduce $E$ y $O$ al denominador común $1 - x^2$ y simplifica.
  \item Un vistazo al futuro, una frase para cada cosa: las partes \omterm{def:g11:func:parity}{par} e
        \omterm{def:g11:func:parity}{impar} de la \omterm{def:g11:func:function}{función} exponencial serán las funciones gemelas del
        año que viene (coseno y seno hiperbólicos); y ese mismo instinto,
        partir un objeto en piezas simétricas y tratar cada una con su
        propia simetría, mueve el análisis del sonido y de las señales de
        los volúmenes universitarios. ¿Por qué es \emph{útil}, en general,
        una descomposición así?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El juego de las transformaciones.}
\begin{enumerate}[resume]
  \item Partiendo de la curva de $x \mapsto x^2$, describe con precisión
        las curvas de $(x - 3)^2$, \ $x^2 + 2$, \ $(x - 3)^2 + 2$, \
        $-x^2$, \ $2x^2$ (\cref{prop:g11:func:transforms}).
  \item Reduce $f(x) = \abs{x - 2} + \abs{x + 2}$ a una fórmula a trozos
        en los tres \omterm{def:g10:numbers:interval}{intervalos} que determinan $-2$ y $2$, y describe su
        curva (tiene un famoso fondo plano).
  \item ¿Qué transformaciones respetan la paridad? Decide, con
        demostración o contraejemplo: el desplazamiento horizontal
        $f(x - 3)$; el desplazamiento vertical $f(x) + 2$; la dilatación
        vertical $2 f(x)$ (todos aplicados a una $f$ \omterm{def:g11:func:parity}{par}).
  \item Resuelve $\abs{x - 2} + \abs{x + 2} = 6$ usando la fórmula a
        trozos de la pregunta 12.
  \item Ingeniería inversa: una \omterm{def:g11:quad:parabola}{parábola} tiene vértice $(2, -3)$ y pasa
        por $(0, 1)$. Reconstruye su \omterm{def:g10:algebra:equation}{ecuación} en \omterm{thm:g11:quad:canonical}{forma canónica}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- \omterm{def:g10:functions:extrema}{Mínimos} sin cálculo diferencial.}
\begin{enumerate}[resume]
  \item Repite el \cref{exo:g11:func:11} para $g(x) = x + \dfrac4x$ en
        $\intoo{0}{+\infty}$: demuestra que
        $g(v) - g(u) = (v - u)\,\dfrac{uv - 4}{uv}$, localiza el \omterm{def:g10:functions:extrema}{mínimo} y
        da su valor.
  \item Aplícalo: de entre todos los rectángulos de área $16$, ¿cuál tiene
        el menor perímetro? (Escribe
        $P = 2\left(x + \frac{16}{x}\right)$ y usa la máquina de la
        pregunta 16.) Concluye un viejo conocido.
  \item Determina la variación de $h(x) = \sqrt{x^2 + 1}$ en todo $\R$,
        combinando un argumento de composición en $\intco{0}{+\infty}$ con
        un argumento de paridad para el resto.
  \item El punto más cercano: halla el punto de la curva $y = \sqrt x$ más
        próximo a $(3, 0)$. (Minimiza la distancia \emph{al cuadrado}, que
        es un polinomio de segundo grado en $x$, y explica por qué es
        legítimo minimizar el cuadrado.)
  \item Final, la visita a la fábrica: las curvas de referencia son la
        materia prima; las transformaciones, las máquinas; la paridad y la
        descomposición, el control de calidad; la factorización de
        diferencias, el banco de medida. Una frase para cada una, y nombra
        la herramienta eléctrica que entrega el capítulo siguiente.
\end{enumerate}
\end{problem}
