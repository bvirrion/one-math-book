\chapter{Funciones y variaciones}\label{ch:g11:func}

Este capítulo organiza lo que uno necesita saber sobre funciones antes
cálculo: la pequeña familia de \emph{\omterm{def:g11:func:reference}{funciones de referencia}} cuyo \omterm{def:g10:functions:graph}{gráficos} debe
saber de memoria el significado preciso de \emph{\omterm{def:g11:func:monotone}{creciente}} y
\emph{\omterm{def:g10:functions:variations}{decreciente}}y cómo se comportan las variaciones cuando funciones están combinados.

\section{Funciones y curvas}

\begin{definition}[Función, dominio, curva.]\label{def:g11:func:function}
A \emph{función}\index{función} $f $ asigna a cada número $ x $ de un conjunto
$ D \subseteq \R $ (es \emph{dominio}\index{dominio}) exactamente un número
$ f(x)$. Es \emph{curva} (o grafo) es el conjunto de puntos
$\bigl(x, f(x)\bigr)$ para $ x \in D $.
\end{definition}

\begin{example}
la formula $ f(x) = \frac{\sqrt{x+2}}{x-1}$ requiere $ x + 2 \geq 0$ (para
la \omterm{def:g11:quad:discriminant}{raíz} cuadrada) y $ x \neq 1$ (denominador distinto de cero): el \omterm{def:g11:func:function}{dominio} es
$\intco{-2}{1} \cup \intoo{1}{+\infty}$.
\end{example}

x \mapsto x^2 \text{ (square)}, \quad
x \mapsto x^3 \text{ (cube)}, \quad
x \mapsto \sqrt{x}, \quad

\section{Variaciones en un intervalo}

\begin{definition}[Creciendo, disminuyendo]\label{def:g11:func:monotone}
Sea $f$ definida en un \omterm{def:g10:numbers:interval}{intervalo} $ I $. La \omterm{def:g11:func:function}{función} $ f $ es
\emph{creciente}\index{creciente} en $ I $ si conserva el orden:
\[
\text{para todo } u, v \in I: \quad u < v \implies f(u) \leq f(v),
\]
y \emph{\omterm{def:g10:functions:variations}{decreciente}} si invierte el orden ($ u < v \implies f(u) \geq
f(v)$). Con estrictas desigualdades en la derecha, $ f $ es \emph{estrictamente}
aumentando (resp.\ \omterm{def:g10:functions:variations}{decreciente}). A \omterm{def:g11:func:function}{función} que va en aumento o
\omterm{def:g10:functions:variations}{decreciente} en $ I $ es \emph{monótono} en $ I $.
\end{definition}

\begin{method}[Demostrando una variación de la definición.]\label{met:g11:func:variation}
Toma dos números arbitrarios. $ u < v $ en el intervaloy determinar el signo
de la diferencia $ f(v) - f(u)$, generalmente por \emph{\omterm{def:g10:algebra:expand}{factorización}} él. si el
el signo es constante en el \omterm{def:g10:numbers:interval}{intervalo}, concluye.
\end{method}

\begin{proposition}[Variaciones de las funciones de referencia.]\label{prop:g11:func:refvariations}
\begin{enumerate}
  \item $ x^2$ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\intoc{-\infty}{0}$ y estrictamente
        \omterm{def:g11:func:monotone}{creciente} en $\intco{0}{+\infty}$.
  \item $ x^3$ es estrictamente \omterm{def:g11:func:monotone}{creciente} en $\R $.
  \item $\sqrt x $ es estrictamente \omterm{def:g11:func:monotone}{creciente} en $\intco{0}{+\infty}$.
  \item $\frac1x $ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\intoo{-\infty}{0}$ y sigue
        $\intoo{0}{+\infty}$ (pero \emph{no} en su \omterm{def:g10:numbers:interunion}{unión}).
  \item $\abs x $ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\intoc{-\infty}{0}$ y
        estrictamente \omterm{def:g11:func:monotone}{creciente} en $\intco{0}{+\infty}$.
\end{enumerate}
\end{proposition}

\begin{proof}
Sea $ u < v $. factorizamos $ f(v) - f(u)$ en cada caso.

\emph{1.} $ v^2 - u^2 = (v-u)(v+u)$. En $\intco{0}{+\infty}$: $ v - u > 0$
y $ v + u > 0$ (como $ v > u \geq 0$), entonces $ v^2 > u^2$. En
$\intoc{-\infty}{0}$: $ v + u < 0$, entonces $ v^2 < u^2$.

\emph{2.} $ v^3 - u^3 = (v - u)(v^2 + uv + u^2)$, y el segundo factor
es igual $\left(v + \frac u2\right)^2 + \frac{3u^2}{4} > 0$ a menos que
$ u = v = 0$. Por eso $ v^3 > u^3$ cuando sea $ v > u $.

\emph{3.} Para $0 \leq u < v $:
$\sqrt v - \sqrt u = \frac{(\sqrt v - \sqrt u)(\sqrt v + \sqrt u)}
{\sqrt v + \sqrt u} = \frac{v - u}{\sqrt v + \sqrt u} > 0$.

\emph{4.} Para $0 < u < v $:
$\frac1v - \frac1u = \frac{u - v}{uv} < 0$ desde $ u - v < 0$ y $ uv > 0$;
mismo cálculo para $ u < v < 0$. en el \omterm{def:g10:numbers:interunion}{unión} el reclamo falla:
$-1 < 1$ pero $\frac{1}{-1} = -1 < 1 = \frac11$.

\emph{5.} En $\intco{0}{+\infty}$, $\abs x = x $ es estrictamente \omterm{def:g11:func:monotone}{creciente}; en
$\intoc{-\infty}{0}$, $\abs x = -x $ es estrictamente \omterm{def:g10:functions:variations}{decreciente}.
\end{proof}

\begin{remark}
El punto 4 es una trampa clásica: ``\omterm{def:g10:functions:variations}{decreciente} en $\intoo{-\infty}{0}$ y sigue
$\intoo{0}{+\infty}$'' es \emph{no} lo mismo que ``\omterm{def:g10:functions:variations}{decreciente} en
$\R \setminus \{0\}$''. Las variaciones sólo tienen sentido en \omterm{def:g10:numbers:interval}{intervalos}.
\end{remark}

\begin{example}[Comparar sin computar]
Desde $\sqrt x $ es estrictamente \omterm{def:g11:func:monotone}{creciente}, $\sqrt{7} < \sqrt{8}$; desde
$ x^2$ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\intoc{-\infty}{0}$,
$(-3)^2 > (-2)^2$. La monotonicidad convierte las comparaciones de imágenes en
comparaciones de argumentos.
\end{example}

\section{Nuevas funciones de las antiguas}

\begin{proposition}[Transformaciones]\label{prop:g11:func:transforms}
Sea $ u $ ser una \omterm{def:g11:func:function}{función} definido en un \omterm{def:g10:numbers:interval}{intervalo} $ I $ y $ k \in \R $.
\begin{enumerate}
  \item $ u + k $ tiene el \emph{mismo} variaciones como $ u $ en $ I $; su curva es
        el de $ u $ desplazado verticalmente por $ k $.
  \item Para $\lambda > 0$, $\lambda u $ tiene las mismas variaciones que $ u $; para
        $\lambda < 0$, las variaciones se invierten.
  \item Si $ u \geq 0$ en $ I $, entonces $\sqrt u $ tiene las mismas variaciones que
        $ u $.
  \item Si $ u \neq 0$ y $ u $ tiene inicio de sesión constante $ I $, entonces $\frac1u $ tiene
        las variaciones \emph{opuesto} a los de $ u $.
\end{enumerate}
\end{proposition}

\begin{proof}
Sea $ u < v $ en $ I $ (abusando de la notación, comparamos $ x $-valores $ u < v $; escribir
$ s = u(x_1)$, $ t = u(x_2)$ para el imágenes de $ x_1 < x_2$).

\emph{1.} $(u+k)(x_2) - (u+k)(x_1) = t - s $: la diferencia de imágenes es
no cambia, por lo que su signo no cambia.

\emph{2.} $\lambda t - \lambda s = \lambda(t - s)$: el cartel se conserva si
$\lambda > 0$, volteado si $\lambda < 0$.

\emph{3.} Si $ u $ es \omterm{def:g11:func:monotone}{creciente} y $ x_1 < x_2$, entonces $0 \leq s \leq t $, y
$\sqrt{}$ ser \omterm{def:g11:func:monotone}{creciente} (\cref{prop:g11:func:refvariations}) da
$\sqrt s \leq \sqrt t $. Mismo argumento en el \omterm{def:g10:functions:variations}{decreciente} caso.

\emph{4.} Si $ u $ es \omterm{def:g11:func:monotone}{creciente}, $0 < s \leq t $ (o $ s \leq t < 0$), entonces
$\frac1s \geq \frac1t $ porque $ x \mapsto \frac1x $ es \omterm{def:g10:functions:variations}{decreciente} en cada uno
lado de $0$: el orden se invierte.
\end{proof}

\begin{example}\label{ex:g11:func:composite}
Estudiar $ f(x) = \sqrt{x^2 + 1}$ en $\R $. el interior \omterm{def:g11:func:function}{función}
$ u(x) = x^2 + 1$ es positivo, \omterm{def:g10:functions:variations}{decreciente} en $\intoc{-\infty}{0}$ y
\omterm{def:g11:func:monotone}{creciente} en $\intco{0}{+\infty}$ (un cuadrado desplazado). Por el punto 3, $ f $
tiene las mismas variaciones: \omterm{def:g10:functions:variations}{decreciente} entonces \omterm{def:g11:func:monotone}{creciente}, con \omterm{def:g10:functions:extrema}{mínimo}
$ f(0) = 1$.
\end{example}

\section{Funciones pares e impares}

\begin{definition}[Paridad]\label{def:g11:func:parity}
Sea $f$ definida en un \omterm{def:g11:func:function}{dominio} $ D $ simétrico sobre $0$ (es decir.\ $-x \in D $
cuando sea $ x \in D $). La \omterm{def:g11:func:function}{función} $ f $ es \emph{par}\index{función par}
si $ f(-x) = f(x)$ para todo $ x \in D $, y \emph{impar}\index{función impar} si
$ f(-x) = -f(x)$ para todo $ x \in D $.
\end{definition}

\begin{proposition}[Significado geométrico]\label{prop:g11:func:paritysym}
La curva de una \omterm{def:g11:func:parity}{función par} es simétrica con respecto a $ y $-eje; la curva
de una \omterm{def:g11:func:parity}{función impar} es simétrica con respecto al origen.
\end{proposition}

\begin{proof}
Si $ f $ es \omterm{def:g11:func:parity}{par} y $(x, y)$ está en la curva (por lo que $ y = f(x)$), entonces
$ f(-x) = f(x) = y $: el punto espejo $(-x, y)$ también está en la curva. Si
$ f $ es \omterm{def:g11:func:parity}{impar}, $ f(-x) = -y $: el punto $(-x, -y)$, simétrico de $(x,y)$ acerca de
el origen, está en la curva.
\end{proof}

\begin{example}
$ x^2$ y $\abs x $ son \omterm{def:g11:func:parity}{pares}; $ x^3$ y $\frac1x $ son \omterm{def:g11:func:parity}{impares};
$ f(x) = x^2 + x $ tampoco es: $ f(-1) = 0$ mientras $ f(1) = 2$, entonces
$ f(-1) \neq \pm f(1)$. La paridad reduce el trabajo a la mitad: una \omterm{def:g11:func:parity}{par} o \omterm{def:g11:func:parity}{función impar}
sólo hay que estudiarlo $\intco{0}{+\infty}$.
\end{example}

f(x) = 3x^4 - x^2, \qquad

\begin{exercise}[$\star $]\label{exo:g11:func:4}
A \omterm{def:g11:func:function}{función} $ f $ tiene el siguiente comportamiento de variación: \omterm{def:g11:func:monotone}{creciente} en
$\intcc{-3}{1}$ de $ f(-3) = -2$ a $ f(1) = 4$, entonces \omterm{def:g10:functions:variations}{decreciente} en
$\intcc{1}{5}$ hasta $ f(5) = 0$. ¿Cuántas soluciones tiene el \omterm{def:g10:algebra:equation}{ecuación}
$ f(x) = 1$ llevar $\intcc{-3}{5}$? Y $ f(x) = 5$?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:func:5}
Usando \cref{met:g11:func:variation} (signo de $ f(v) - f(u)$), demostrar que

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:func:6}
Sin calcular ninguna derivada, dé las variaciones de

f(x) = \sqrt{5 - x} \ \text{ on } \intoc{-\infty}{5},
\quad
g(x) = \frac{1}{x^2 + 1} \ \text{ on } \intco{0}{+\infty},
\quad
h(x) = 3 - 2\sqrt{x} \ \text{ on } \intco{0}{+\infty}.

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:func:7}
Sea $ f(x) = (x-2)^2 - 3$ en $\R $.
\begin{enumerate}
  \item Dar las variaciones de $ f $ y su \omterm{def:g10:functions:extrema}{mínimo}.
  \item Deducir las variaciones de $ g(x) = \frac{1}{(x-2)^2+1}$ y su
        \omterm{def:g10:functions:extrema}{máximo}. (Nota $ g = \frac{1}{f + 4}$.)

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:func:8}
la curva de una \omterm{def:g11:func:function}{función} $ f $ pasa por los puntos $(0, 1)$, $(2, 3)$
y $(4, 1)$, y $ f $ es \omterm{def:g11:func:monotone}{creciente} en $\intcc{0}{2}$ y \omterm{def:g10:functions:variations}{decreciente} en

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:func:9}
Demuestre que la \omterm{def:g11:func:function}{función} $ f(x) = \frac{2x+1}{x+1}$ se puede escribir

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:func:10}
Verdadero o falso (justifique o dé un contraejemplo):
\begin{enumerate}
  \item la suma de dos funciones \omterm{def:g11:func:monotone}{crecientes} en $ I $ es \omterm{def:g11:func:monotone}{creciente} en $ I $.
  \item El producto de dos funciones \omterm{def:g11:func:monotone}{crecientes} en $ I $ es \omterm{def:g11:func:monotone}{creciente} en
        $ I $.
  \item Si $ f $ es \omterm{def:g11:func:monotone}{creciente} en $ I $, entonces $ f^2$ ($= f \times f $) es
        \omterm{def:g11:func:monotone}{creciente} en $ I $.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g11:func:11}
Sea $ f(x) = x + \frac{1}{x}$ para $ x > 0$.
\begin{enumerate}
  \item Muéstralo por $0 < u < v $,
        $ f(v) - f(u) = (v - u)\,\frac{uv - 1}{uv}$.
  \item deducir eso $ f $ es estrictamente \omterm{def:g10:functions:variations}{decreciente} en $\intoc{0}{1}$ y
        estrictamente \omterm{def:g11:func:monotone}{creciente} en $\intco{1}{+\infty}$.
  \item Concluir que $ x + \frac1x \geq 2$ para todo $ x > 0$, con igualdad
        sólo en $ x = 1$.
\end{enumerate}
\end{exercise}

\section{Problema: la fábrica de simetría}

\begin{problem}[{Problema del fin de semana --- cada función es par
parte más una parte impar: reglas de paridad, el juego de transformación y
mínimos ganados sin cálculo}]
\label{pb:g11:func:1}
Alguno funciones son simétricos en espejo, algunos son de \omterm{def:g10:stats:mean}{media} vuelta
simétricos, la mayoría no lo son, pero es un pequeño milagro algebraico
dice que \emph{cada} \omterm{def:g11:func:function}{función} es, exactamente de una manera, la suma
de una pieza simétrica espejo y una pieza simétrica de \omterm{def:g10:stats:mean}{media} vuelta. esto
problema demuestra el milagro, juega el juego de la transformación en el
curvas de referencia (\cref{prop:g11:func:transforms}), y cierra
con el \omterm{def:g10:algebra:expand}{factorización} de diferencias técnica de
\cref{exo:g11:func:11}, que gana los problemas de minimización por completo
capítulo antes de que llegue la derivada.

\medskip
\noindent\textbf{Part I --- Parity fluency.}
\begin{enumerate}
  \item \omterm{def:g11:func:parity}{Par}, \omterm{def:g11:func:parity}{impar}, o ninguno (\cref{def:g11:func:parity})?
        Justifique cada uno: $ x^4 - 3x^2$; \ $ x^3 - x $; \ $ x^2 + x $; \
        $\abs x $; \ $\frac1x $.
  \item Indique el significado geométrico de cada veredicto.
        (\cref{prop:g11:func:paritysym}): ¿qué simetría hace
        la curva de una \omterm{def:g11:func:parity}{función par} ¿tener? de un \omterm{def:g11:func:parity}{impar} ¿uno?
  \item La paridad se multiplica como signos: demuestra que el producto de
        dos funciones \omterm{def:g11:func:parity}{impares} es pary completar todo
        tabla de multiplicar (\omterm{def:g11:func:parity}{par} $\times $ \omterm{def:g11:func:parity}{par}, \omterm{def:g11:func:parity}{par} $\times $
        \omterm{def:g11:func:parity}{impar}, \omterm{def:g11:func:parity}{impar} $\times $ \omterm{def:g11:func:parity}{impar}).
  \item Consecuencias rápidas: si $ f $ es \omterm{def:g11:func:parity}{impar} y definido en $0$,
        qué es $ f(0)$? Si $ f $ es \omterm{def:g11:func:parity}{par} con $ f(3) = 7$, qué es
        $ f(-3)$? Y demostrar que el único \omterm{def:g11:func:function}{función} ambos \omterm{def:g11:func:parity}{par} y
        \omterm{def:g11:func:parity}{impar} es el cero \omterm{def:g11:func:function}{función}.
  \item ¿Cuánto de un \omterm{def:g11:func:function}{función} parLa curva de determina todos
        eso? dar una \omterm{def:g11:func:function}{función} eso no es ninguno \omterm{def:g11:func:parity}{par} ni \omterm{def:g11:func:parity}{impar}, y
        explica lo que falla.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The decomposition theorem.}
por una \omterm{def:g11:func:function}{función} $ f $ definido en un \omterm{def:g11:func:function}{dominio} simétrico sobre $0$, colocar
\[
E(x) = \frac{f(x) + f(-x)}{2},
\qquad
O(x) = \frac{f(x) - f(-x)}{2} .
\]
\begin{enumerate}[resume]
  \item Calcular $ E $ y $ O $ para $ f(x) = x^3 + x^2 + 1$, y
        comprobar: $ E $ \omterm{def:g11:func:parity}{par}, $ O $ \omterm{def:g11:func:parity}{impar}, $ E + O = f $.
  \item Demuestre la afirmación general: para cualquier $ f $, la \omterm{def:g11:func:function}{función}
        $ E $ es \omterm{def:g11:func:parity}{par}, $ O $ es \omterm{def:g11:func:parity}{impar}, y $ f = E + O $ --- \emph{cada
        \omterm{def:g11:func:function}{función} se divide en un \omterm{def:g11:func:parity}{par} parte y una \omterm{def:g11:func:parity}{impar} parte}.
  \item Demuestre que la división es única: si
        $ f = E_1 + O_1 = E_2 + O_2$ con $ E_i $ \omterm{def:g11:func:parity}{par} y $ O_i $
        \omterm{def:g11:func:parity}{impar}, utilice la pregunta 4 en $ E_1 - E_2$.
  \item Descomponer $ f(x) = \dfrac{1}{1 - x}$ (para
        $ x \neq \pm 1$): poner $ E $ y $ O $ sobre lo común
        denominador $1 - x^2$ y simplificar.
  \item Una mirada al futuro, en una frase cada una: el \omterm{def:g11:func:parity}{par} y \omterm{def:g11:func:parity}{impar}
        partes de la exponencial \omterm{def:g11:func:function}{función} sera el gemelo
        funciones de grado 12 (coseno y seno hiperbólicos); y
        el mismo instinto --- dividir un objeto en simétrico
        piezas y tratar cada una con su propia simetría --- poderes
        Análisis de sonidos y señales de los volúmenes universitarios.
        ¿Por qué hay tal división? \emph{útil}, ¿en general?
\end{enumerate}

\noindent\textbf{Part III --- The transformation game.}
\begin{enumerate}[resume]
  \item Partiendo de la curva de $ x \mapsto x^2$, describir
        precisamente las curvas de $(x - 3)^2$, \ $ x^2 + 2$, \
        $(x - 3)^2 + 2$, \ $-x^2$, \ $2x^2$
        (\cref{prop:g11:func:transforms}).
  \item Reducir $ f(x) = \abs{x - 2} + \abs{x + 2}$ a un
        fórmula por partes en los tres \omterm{def:g10:numbers:interval}{intervalos} cortado por $-2$
        y $2$, y describe su curva (tiene un famoso plano
        abajo).
  \item ¿Qué transformaciones respetan la uniformidad? decidir, con
        prueba o contraejemplo: desplazamiento horizontal
        $ f(x - 3)$; desplazamiento vertical $ f(x) + 2$; estiramiento vertical
        $2 f(x)$ (todo aplicado a un \omterm{def:g11:func:parity}{par} $ f $).
  \item Resolver $\abs{x - 2} + \abs{x + 2} = 6$ usando el

  \item Ingeniería inversa: una \omterm{def:g11:quad:parabola}{parábola} tiene vértice $(2, -3)$

        \omterm{thm:g11:quad:canonical}{forma canónica}.
\end{enumerate}
