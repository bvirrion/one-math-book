\chapter{Vectores y rectas en el plano}\label{ch:g11:vect}

\omterm{def:g11:vect:vector}{vectores} codificar desplazamientos; las líneas son sus trayectorias. La novedad clave
herramienta de este capítulo es la \emph{\omterm{def:g11:vect:det}{determinante}}, un solo número que
decide si dos \omterm{def:g11:vect:vector}{vectores} son \omterm{def:g11:vect:collinear}{colineales} --- y, a partir de ahí, produce
ecuaciones de líneas y resuelve cuestiones de alineación y paralelismo por pura
cálculo. Las mismas ideas se extienden al espacio en \cref{ch:g12:space}.

\section{Vectores: un resumen}

\begin{definition}[Vector, coordenadas]\label{def:g11:vect:vector}
A \emph{vector}\index{vector} $\vec u $ representa un desplazamiento: todos
Las flechas con la misma dirección, orientación y longitud representan lo mismo.
vector. en un \omterm{def:g10:coordgeom:system}{sistema de coordenadas}, $\vec u\,(x, y)$ registra el desplazamiento
($ x $ horizontalmente, $ y $ verticalmente); por puntos $ A(x_A, y_A)$ y
$ B(x_B, y_B)$,
\[
\vect{AB}\,\bigl(x_B - x_A,\ y_B - y_A\bigr).
\]
Los vectores suman \omterm{def:g10:coordgeom:system}{coordenadas} y $\lambda\vec u $ tiene \omterm{def:g10:coordgeom:system}{coordenadas}
$(\lambda x, \lambda y)$.
\end{definition}

\begin{proposition}[Punto medio]\label{prop:g11:vect:midpoint}
El \omterm{prop:g10:coordgeom:midpoint}{punto medio} $ M $ del segmento $[AB]$ tiene \omterm{def:g10:coordgeom:system}{coordenadas}
$\left(\frac{x_A + x_B}{2},\ \frac{y_A + y_B}{2}\right)$.
\end{proposition}

\begin{proof}
$ M $ se caracteriza por $\vect{AM} = \frac12\vect{AB}$: en cuanto a \omterm{def:g10:coordgeom:system}{coordenadas},
$ x_M - x_A = \frac12(x_B - x_A)$, entonces $ x_M = \frac{x_A + x_B}{2}$, y
de manera similar para $ y_M $.
\end{proof}

\section{Colinealidad y el determinante.}

\begin{definition}[Vectores colineales]\label{def:g11:vect:collinear}
Dos \omterm{def:g11:vect:vector}{vectores} son \emph{colineal}\index{colineal} si uno es un
múltiplo escalar del otro: $\vec v = \lambda \vec u $ para algunos
$\lambda \in \R $ (o $\vec u = \vec 0$). Geométricamente: tienen lo mismo
dirección, o uno de ellos es cero.
\end{definition}

\begin{definition}[Determinante]\label{def:g11:vect:det}
El \emph{determinante}\index{determinante} de $\vec u\,(x, y)$ y
$\vec v\,(x', y')$ es
\[
\det(\vec u, \vec v) =
\begin{vmatrix} x & x' \\ y & y' \end{vmatrix}
= x y' - x' y .
\]
\end{definition}

\begin{theorem}[Criterio de colinealidad]\label{thm:g11:vect:collinearity}
Dos \omterm{def:g11:vect:vector}{vectores} $\vec u $ y $\vec v $ son \omterm{def:g11:vect:collinear}{colineales} si y solo si
$\det(\vec u, \vec v) = 0$.
\end{theorem}

\begin{proof}
($\Rightarrow $) Si $\vec v = \lambda\vec u $, entonces $ x' = \lambda x $ y
$ y' = \lambda y $, entonces
$\det(\vec u, \vec v) = x(\lambda y) - (\lambda x)y = 0$; si
$\vec u = \vec 0$, el \omterm{def:g11:vect:det}{determinante} es trivial $0$.

($\Leftarrow $) Suponer $ xy' - x'y = 0$ y $\vec u \neq \vec 0$, decir
$ x \neq 0$ (el caso $ y \neq 0$ es simétrico). Colocar
$\lambda = \frac{x'}{x}$. Entonces $ x' = \lambda x $ por construcción, y el
relación $ xy' = x'y = \lambda x y $ da, después de dividir por $ x \neq 0$,
$ y' = \lambda y $. Por eso $\vec v = \lambda \vec u $.
\end{proof}

\begin{example}\label{ex:g11:vect:collinear}
$\vec u\,(3, -2)$ y $\vec v\,(-6, 4)$:
$\det = 3 \times 4 - (-6) \times (-2) = 12 - 12 = 0$, \omterm{def:g11:vect:collinear}{colineales} (en efecto
$\vec v = -2\vec u $). $\vec u\,(3, -2)$ y $\vec w\,(2, 1)$:
$\det = 3 + 4 = 7 \neq 0$, no \omterm{def:g11:vect:collinear}{colineales}.
\end{example}

\begin{method}[Alineación de tres puntos.]\label{met:g11:vect:alignment}
$ A $, $ B $, $ C $ están alineados si y sólo si $\vect{AB}$ y $\vect{AC}$ son
\omterm{def:g11:vect:collinear}{colineales}: calcular ambos \omterm{def:g11:vect:vector}{vectores} y comprobar
$\det\bigl(\vect{AB}, \vect{AC}\bigr) = 0$.
\end{method}

\section{Ecuaciones cartesianas de rectas.}

\begin{definition}[Vector de dirección]\label{def:g11:vect:direction}
A \emph{vector de direccion}\index{vector de direccion} de una linea $ d $ es cualquiera
distinto de cero \omterm{def:g11:vect:vector}{vector} $\vec u $ tal que $\vect{AB}$ es \omterm{def:g11:vect:collinear}{colineales} con $\vec u $
para todos los puntos $ A, B $ de $ d $.
\end{definition}

\begin{theorem}[ecuación cartesiana]\label{thm:g11:vect:cartesian}
una línea con \omterm{def:g11:vect:direction}{vector de direccion} $\vec u\,(-b, a)$ admite un \omterm{def:g10:algebra:equation}{ecuación} de la
forma
\[
ax + by + c = 0
\qquad (a, b) \neq (0, 0).
\]
Por el contrario, cada uno de esos \omterm{def:g10:algebra:equation}{ecuación} describe una línea con \omterm{def:g11:vect:direction}{vector de direccion}
$\vec u\,(-b, a)$.
\end{theorem}

\begin{proof}
Sea $ d $ pasar por $ A(x_A, y_A)$ con dirección $\vec u\,(-b, a)$. un
punto $ M(x, y)$ se encuentra en $ d $ exactamente cuando $\vect{AM}$ es \omterm{def:g11:vect:collinear}{colineales} con
$\vec u $, es decir.\ (\cref{thm:g11:vect:collinearity})
\[
0 = \det\bigl(\vect{AM}, \vec u\bigr)
= (x - x_A)\,a - (-b)(y - y_A)
= ax + by - (a x_A + b y_A),
\]
un \omterm{def:g10:algebra:equation}{ecuación} de la forma anunciada con $ c = -(a x_A + b y_A)$.
Por el contrario, las soluciones de $ ax + by + c = 0$ con, digamos, $ b \neq 0$ son
los puntos $\left(x, -\frac{a x + c}{b}\right)$: restando dos de ellos
\omterm{def:g10:proba:sample}{muestra} que cada acorde es \omterm{def:g11:vect:collinear}{colineales} con $(-b, a)$, y una solución siempre
existe --- es la línea que lo atraviesa dirigida por $(-b, a)$.
\end{proof}

\begin{method}[Línea que pasa por dos puntos]\label{met:g11:vect:twopoints}
para encontrar un \omterm{def:g10:algebra:equation}{ecuación} de la linea $(AB)$: calcular el \omterm{def:g11:vect:direction}{vector de direccion}
$\vect{AB}\,(x_B - x_A, y_B - y_A)$; combinarlo con $(-b, a)$, es decir,\ tomar
$ a = y_B - y_A $ y $ b = -(x_B - x_A)$; determinar $ c $ conectando el
\omterm{def:g10:coordgeom:system}{coordenadas} de $ A $.
\end{method}

\begin{example}\label{ex:g11:vect:twopoints}
Línea a través $ A(1, 2)$ y $ B(4, 3)$: $\vect{AB}\,(3, 1)$, entonces
$ a = 1$, $ b = -3$, y el \omterm{def:g10:algebra:equation}{ecuación} es $ x - 3y + c = 0$. Conectando $ A $:
$1 - 6 + c = 0$, entonces $ c = 5$. \omterm{def:g10:algebra:equation}{ecuación}: $ x - 3y + 5 = 0$. Consulte con $ B $:
$4 - 9 + 5 = 0$.
\end{example}

\begin{remark}
Cuando $ b \neq 0$, el \omterm{def:g10:algebra:equation}{ecuación} se puede resolver para $ y $: $ y = mx + p $ con
\omterm{def:g10:reffunc:affine}{pendiente} $ m = -\frac ab $. la forma cartesiana $ ax + by + c = 0$ es mas
general: también cubre líneas verticales ($ b = 0$), que no tienen \omterm{def:g10:reffunc:affine}{pendiente}.
\end{remark}

\section{Representación paramétrica}

\begin{definition}[Representación paramétrica]\label{def:g11:vect:parametric}
La línea a través $ A(x_A, y_A)$ con dirección $\vec u\,(\alpha, \beta)$ es
el conjunto de puntos
\[
M\bigl(x_A + t\alpha,\ y_A + t\beta\bigr), \qquad t \in \R .
\]
el numero $ t $ es el \emph{parámetro}: cada valor de $ t $ produce uno
punto de la línea, como si viajara a lo largo de ella con velocidad constante $\vec u $.
\end{definition}

\begin{example}
la linea de \cref{ex:g11:vect:twopoints} también es
$\{(1 + 3t,\ 2 + t) : t \in \R\}$: $ t = 0$ da $ A $, $ t = 1$ da $ B $.
Eliminando $ t $ ($ t = y - 2$, entonces $ x = 1 + 3(y-2)$) se recupera
$ x - 3y + 5 = 0$.
\end{example}

\section{Posiciones relativas de dos líneas.}

\begin{proposition}[Paralelo o intersectándose]\label{prop:g11:vect:positions}
dos líneas con \omterm{def:g11:vect:vector}{vectores} de direccion $\vec u $ y $\vec v $ son paralelos si y
solo si $\det(\vec u, \vec v) = 0$. En \omterm{def:g10:algebra:equation}{ecuación} forma: $ d : ax + by + c =
0$ y $ d' : a'x + b'y + c' = 0$ son paralelos si y sólo si
$ ab' - a'b = 0$. Las rectas no paralelas se encuentran exactamente en un punto.
\end{proposition}

\begin{proof}
Paralelismo significa \omterm{def:g11:vect:collinear}{colineales} direcciones, que es
\cref{thm:g11:vect:collinearity}; con $\vec u\,(-b, a)$ y
$\vec v\,(-b', a')$, el \omterm{def:g11:vect:det}{determinante} es $(-b)a' - (-b')a = ab' - a'b $.
Si las rectas no son paralelas, resolver las dos ecuaciones simultáneamente
(por sustitución o combinación) conduce a una solución única: el sistema
se comporta como un \omterm{def:g11:vect:det}{determinante} distinto de cero $2 \times 2$ sistema.
\end{proof}

\begin{method}[Intersección de dos líneas]\label{met:g11:vect:intersection}
Resuelve el sistema de los dos. ecuaciones: aislar una incógnita en una \omterm{def:g10:algebra:equation}{ecuación}
y sustituir por el otro, o combinar el ecuaciones eliminar uno
desconocido. Compruebe siempre el punto resultante en \emph{ambos} ecuaciones.
\end{method}

\begin{example}\label{ex:g11:vect:intersection}
Intersecarse $ d: x - 3y + 5 = 0$ y $ d': 2x + y - 4 = 0$. De $ d'$:
$ y = 4 - 2x $. Sustituyendo en $ d $:
\[
x - 3(4 - 2x) + 5 = 0
\iff x - 12 + 6x + 5 = 0
\iff 7x = 7
\iff x = 1,
\]
entonces $ y = 2$. Las líneas se encuentran en $(1, 2)$. Registrarse $ d $: $1 - 6 + 5 = 0$.
\end{example}

\section{Problema: rumbos de colisión y coordenadas inteligentes}

\begin{problem}[{Problema del fin de semana --- cruzar caminos no es
colisión: máximo acercamiento en el mar y teoremas demostrados por
elegir el marco correcto}]
\label{pb:g11:vect:1}
Los rumbos rectos de dos barcos se cruzan. ¿Deben chocar los barcos? no
a menos que lleguen al cruce \emph{al mismo tiempo}: los caminos son
conjuntos de puntos, los movimientos son paramétricos y confundirlos ha
barcos reales hundidos. Este problema gestiona un servicio de tráfico de embarcaciones pequeñas.
en líneas paramétricas (\cref{def:g11:vect:parametric}), luego gira
a la geometría pura con el otro superpoder de \omterm{def:g10:coordgeom:system}{coordenadas}:
\emph{eligiendo} ellos inteligentemente, como en \cref{exo:g11:vect:11}, a
hacer que los teoremas clásicos se calculen solos.

\medskip
\noindent\textbf{Part I --- Lines, three ways.}
\begin{enumerate}
  \item Dar una representación paramétrica de la recta que pasa por
        $ A(1, 2)$ con \omterm{def:g11:vect:direction}{vector de direccion} $\vec u(3, -1)$, entonces
        eliminar el parámetro para obtener un cartesiano \omterm{def:g10:algebra:equation}{ecuación}.
  \item para la linea $2x - 5y + 3 = 0$: leer un \omterm{prop:g10:lines:parallel}{dirección
        vector} (\cref{thm:g11:vect:cartesian}), encuentra un punto,
        y escribe una representación paramétrica.
  \item Calcular el \omterm{def:g10:numbers:interunion}{intersección} punto de las dos líneas de
        preguntas 1 y 2 (\cref{met:g11:vect:intersection}).
  \item Confirmar con el \omterm{def:g11:vect:det}{determinante}
        (\cref{thm:g11:vect:collinearity}) que esas dos líneas
        tuvo que cruzarse; entonces \omterm{def:g10:proba:sample}{muestra} que $2x - 5y + 3 = 0$ y
        $-4x + 10y + 1 = 0$ son estrictamente paralelos.
  \item Son $ A(2, 1)$, $ B(5, 3)$, $ C(11, 7)$ alineado
        (\cref{met:g11:vect:alignment})?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The vessel-traffic service.}
en el momento $ t $ (horas), barco $ P $ esta en $(2t,\ 3 + t)$ y barco $ Q $
en $(10 - t,\ 2t - 1)$ (distancias en millas náuticas).
\begin{enumerate}[resume]
  \item Encuentra el cartesiano ecuaciones de los dos \emph{caminos},
        y calcular dónde se cruzan los caminos.
  \item ¿A qué hora pasa cada barco por el punto de cruce? hacer
        ¿Los barcos chocan? Exponga la advertencia general en una
        frase: cruzarse en contra de mociones de reunión.
  \item Calcular la distancia al cuadrado $ D^2(t)$ entre los barcos
        en el momento $ t $, y reducirlo a una cuadrática en $ t $. en
        ¿A qué hora están los barcos más cerca y qué tan cerca están?
        ven (\cref{pb:g11:quad:1} tecnología de vértice)?
  \item La normativa exige una separación de al menos $1$ milla.
        ¿Se viola? Si es así, resuelve $ D^2(t) < 1$ y darle el
        duración de la infracción.
  \item Explique por qué, por \emph{cualquier} dos barcos en línea recta
        cursos a velocidades constantes, $ D^2(t)$ es siempre un
        \omterm{def:g11:quad:quadratic}{función cuadrática} de $ t $ --- entonces ``punto más cercano de
        "Enfoque'' es siempre un cálculo de vértice.
  \item Interceptación: una patrullera parte del origen en
        $ t = 0$ con velocidad constante $(a, b)$ y debe cumplir
        barco $ P $ exactamente en $ t = 3$. Calcular el requerido
        $(a, b)$ y la velocidad de la patrullera.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Clever \omterm{def:g10:coordgeom:system}{coordenadas}.}
Trabajar en el marco de \cref{exo:g11:vect:11}: el paralelogramo
$ ABCD $ se convierte $ A(0,0)$, $ B(1,0)$, $ C(1,1)$, $ D(0,1)$; dejar $ I $
ser el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[AB]$ y $ K $ el \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[CD]$.
\begin{enumerate}[resume]
  \item Calcular un cartesiano \omterm{def:g10:algebra:equation}{ecuación} de la linea $(DI)$.
  \item Intersecarse $(DI)$ con la diagonal $(AC)$: recuperar
        \cref{exo:g11:vect:11}El hecho es que el corte queda exactamente
        un tercio de la diagonal.
  \item ahora la linea $(BK)$: encontrar su \omterm{def:g10:algebra:equation}{ecuación}, se cruza con
        $(AC)$, y concluir el teorema clásico completo:
        \emph{los cevianos $(DI)$ y $(BK)$ trisecar la diagonal
        $(AC)$}.
  \item ¿Por qué era legítimo demostrar el teorema en este?
        marco especial? (¿Qué significa la elección de \omterm{def:g10:coordgeom:system}{coordenadas}
        preservar, y ¿qué implica la afirmación?) Respuesta
        en dos o tres frases.
  \item Una trisección más para practicar: dejemos $ E $ ser el
        \omterm{prop:g10:coordgeom:midpoint}{punto medio} de $[BC]$. ¿De dónde viene la línea? $(AE)$ cortar el
        \emph{otro} diagonal $(DB)$?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Position puzzles.}
\begin{enumerate}[resume]
  \item son las tres lineas $ x + y = 4$, \ $2x - y = 2$, \
        $ x - 2y = -2$ ¿concurrente? (Cruce dos, pruebe el
        tercero.)
  \item por cada real $ m $, dejar $ d_m $ ser la linea
        $ mx - y + 2 - 3m = 0$. mostrar que \emph{cada} $ d_m $
        pasa por uno \omterm{pb:g10:functions:1}{punto fijo}. (Agrupa los términos en
        $ m $.)
  \item en la familia $(d_m)$: qué miembro es paralelo a
        $ y = 2x + 7$? ¿Cuál pasa por el origen?
  \item Final --- los tres trajes de una línea: cartesiana
        (pruebas de membresía en una sustitución), reducido (\omterm{def:g10:reffunc:affine}{pendiente}
        y grafo a la vista), paramétricos (movimiento, sincronización,
        interceptación); el \omterm{def:g11:vect:det}{determinante} como oráculo del paralelismo;
        y el marco bien elegido como fábrica de teoremas. uno
        frase cada uno.
\end{enumerate}
\end{problem}
