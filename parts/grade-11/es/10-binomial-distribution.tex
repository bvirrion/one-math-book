\chapter{La distribución binomial}\label{ch:g11:binom}

Repite varias veces, de forma independiente, un mismo experimento de sí o
no y cuenta los éxitos: la \omterm{def:g11:prob:rv}{distribución} que resulta, la \emph{binomial}, es
la \omterm{def:g11:prob:rv}{distribución} discreta más importante de todas. Este capítulo la
construye con \omterm{met:g10:proba:tree}{árboles} y recuentos de caminos; la fórmula cerrada de esos
recuentos (con factoriales) llega con las herramientas de conteo del
\cref{ch:g12:comb}, y la \omterm{def:g11:prob:rv}{distribución} se retoma en el
\cref{ch:g12:randvar}.

\section{Pruebas de Bernoulli}

\begin{definition}[Prueba de Bernoulli]\label{def:g11:binom:bernoulli}
Una \emph{prueba de Bernoulli}\index{prueba de Bernoulli} es un
experimento con exactamente dos resultados: \emph{éxito}, con \omterm{def:g10:proba:distribution}{probabilidad}
$p$, y \emph{fracaso}, con \omterm{def:g10:proba:distribution}{probabilidad} $1 - p$. Se dice que la \omterm{def:g11:prob:rv}{variable
aleatoria} $X$ que vale $1$ en caso de éxito y $0$ en caso de fracaso sigue
la \emph{\omterm{def:g11:prob:rv}{distribución} de Bernoulli} $\mathcal B(p)$; entonces
\[
\E(X) = p,
\qquad
\V(X) = p(1 - p).
\]
\end{definition}

\begin{proof}[Demostración de las dos fórmulas]
$\E(X) = p \times 1 + (1-p) \times 0 = p$; y como $X^2 = X$ (tanto $0$
como $1$ son su propio cuadrado), $\E(X^2) = p$, luego por la
\cref{prop:g11:prob:konig}, $\V(X) = p - p^2 = p(1-p)$.
\end{proof}

\begin{definition}[Pruebas repetidas e independientes]\label{def:g11:binom:repeated}
Repetir $n$ veces una \omterm{def:g11:binom:bernoulli}{prueba de Bernoulli} de forma
\emph{independiente} significa que el resultado de cada prueba no influye
en los demás y que la \omterm{def:g10:proba:distribution}{probabilidad} de cualquier secuencia completa de
resultados es el \emph{producto} de las \omterm{def:g10:proba:distribution}{probabilidades} a lo largo del
camino correspondiente del \omterm{met:g10:proba:tree}{árbol}: $p$ por cada éxito y $1 - p$ por cada
fracaso.
\end{definition}

\begin{example}\label{ex:g11:binom:threetrials}
Tres pruebas independientes con \omterm{def:g10:proba:distribution}{probabilidad} de éxito $p$. La secuencia
EFE (éxito, fracaso, éxito) tiene \omterm{def:g10:proba:distribution}{probabilidad} $p(1-p)p = p^2(1-p)$, y lo
mismo le ocurre a \emph{todas} las secuencias con exactamente dos éxitos,
estén donde estén: solo importa el \emph{número} de E y de F.
\end{example}

\section{Recuento de caminos y números combinatorios}

\begin{definition}[Número combinatorio]\label{def:g11:binom:coefficient}
En el \omterm{met:g10:proba:tree}{árbol} de $n$ pruebas independientes, el \emph{número
combinatorio}\index{número combinatorio} $\binom{n}{k}$ (que se lee ``$n$
sobre $k$'') es el número de caminos que contienen exactamente $k$
éxitos.
\end{definition}

\begin{example}
$\binom{3}{2} = 3$: los caminos EEF, EFE y FEE. Del mismo modo,
$\binom{3}{0} = 1$ (el camino FFF), $\binom{3}{1} = 3$ y
$\binom{3}{3} = 1$. Por convenio, y también por el \omterm{met:g10:proba:tree}{árbol},
$\binom n0 = \binom nn = 1$ para todo $n$.
\end{example}

\begin{proposition}[Regla de Pascal]\label{prop:g11:binom:pascal}
\index{triángulo de Pascal}
Para $1 \leq k \leq n - 1$:
\[
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}.
\]
\end{proposition}

\begin{proof}
Clasificamos los caminos con $k$ éxitos del \omterm{met:g10:proba:tree}{árbol} de $n$ pruebas según su
\emph{última} prueba. Los que terminan en éxito se obtienen de un camino
de las primeras $n-1$ pruebas con $k - 1$ éxitos: hay $\binom{n-1}{k-1}$.
Los que terminan en fracaso prolongan un camino con $k$ éxitos entre las
primeras $n-1$ pruebas: hay $\binom{n-1}{k}$. Cada camino es exactamente
de uno de los dos tipos.
\end{proof}

La regla de Pascal genera los coeficientes fila a fila: cada entrada es la
suma de las dos que tiene encima.
\[
\begin{array}{ccccccccccc}
&&&&&1&&&&&\\
&&&&1&&1&&&&\\
&&&1&&2&&1&&&\\
&&1&&3&&3&&1&&\\
&1&&4&&6&&4&&1&\\
1&&5&&10&&10&&5&&1
\end{array}
\]

\begin{remark}
Una fórmula cerrada, $\binom nk = \frac{n!}{k!(n-k)!}$, junto con una
teoría sistemática del conteo, se establece en el \cref{ch:g12:comb}. A
este nivel, el \omterm{prop:g11:binom:pascal}{triángulo de Pascal} calcula todos los coeficientes que
necesitamos.
\end{remark}

\section{La distribución binomial}

\begin{theorem}[Distribución binomial]\label{thm:g11:binom:binomial}
Sea $X$ el número de éxitos en $n$ \omterm{def:g11:binom:bernoulli}{pruebas de Bernoulli} independientes de
parámetro $p$. Entonces $X$ sigue la \emph{distribución
binomial}\index{distribución binomial} $\mathcal B(n, p)$:
\[
\P(X = k) = \binom{n}{k}\, p^k (1-p)^{n-k},
\qquad k = 0, 1, \dots, n .
\]
\end{theorem}

\begin{proof}
El \omterm{def:g11:prob:model}{suceso} $X = k$ es el conjunto de todos los caminos con exactamente $k$
éxitos. Cada uno de esos caminos tiene \omterm{def:g10:proba:distribution}{probabilidad} $p^k(1-p)^{n-k}$: el
producto a lo largo del camino contiene $k$ factores $p$ y $n - k$
factores $1-p$, en algún orden (\cref{def:g11:binom:repeated}). Hay
$\binom nk$ caminos así (\cref{def:g11:binom:coefficient}) y sus
\omterm{def:g10:proba:distribution}{probabilidades} se suman.
\end{proof}

\begin{example}\label{ex:g11:binom:quiz}
Una prueba tiene $5$ preguntas independientes, cada una con $4$ opciones;
un alumno responde al azar, así que cada pregunta es un éxito con
$p = \frac14$. El número $X$ de respuestas correctas sigue
$\mathcal B\left(5, \frac14\right)$ y, usando la fila $5$ del \omterm{prop:g11:binom:pascal}{triángulo de
Pascal}:
\[
\P(X = 2) = \binom52 \left(\frac14\right)^2\left(\frac34\right)^3
= 10 \times \frac{1}{16} \times \frac{27}{64}
= \frac{270}{1024} \approx 0.26 .
\]
La \omterm{def:g10:proba:distribution}{probabilidad} de al menos una respuesta correcta se calcula con el
\omterm{def:g10:proba:operations}{suceso contrario}:
$\P(X \geq 1) = 1 - \P(X = 0) = 1 - \left(\frac34\right)^5 \approx
0.76$.
\end{example}

\begin{proposition}[Esperanza y varianza]\label{prop:g11:binom:expectation}
Si $X \sim \mathcal B(n, p)$:
\[
\E(X) = np,
\qquad
\V(X) = np(1-p).
\]
\end{proposition}

\begin{proof}[Justificación]
Escribimos $X = X_1 + X_2 + \dots + X_n$, donde $X_i$ vale $1$ si la
prueba $i$-ésima tiene éxito: cada $X_i$ es una variable de Bernoulli de
\omterm{def:g11:prob:expectation}{esperanza} $p$ (\cref{def:g11:binom:bernoulli}). Las \omterm{def:g11:stat:mean}{medias} se suman, y
sumando las $n$ aportaciones queda $\E(X) = np$. Que las
\emph{\omterm{def:g11:prob:variance}{varianzas}} también se sumen para variables independientes es cierto,
pero más delicado: la fórmula de la \omterm{def:g11:prob:variance}{varianza} se \emph{admite a este
nivel} y se demuestra en el \cref{ch:g12:sums}.
\end{proof}

\begin{method}[Reconocer una situación binomial]\label{met:g11:binom:recognize}
Antes de escribir $X \sim \mathcal B(n, p)$, comprueba tres ingredientes:
un \emph{número fijo} $n$ de pruebas, decidido de antemano; que cada
prueba tenga \emph{dos resultados} con la \emph{misma} \omterm{def:g10:proba:distribution}{probabilidad} de
éxito $p$; y que las pruebas sean \emph{independientes} (con devolución, o
con dispositivos separados). Extraer sin devolución de una población
pequeña \emph{no} es binomial: la \omterm{def:g10:proba:distribution}{probabilidad} cambia en cada extracción
(\cref{exo:g11:prob:6}).
\end{method}

\section{Muestreo: ¿es sorprendente lo observado?}

La \omterm{def:g11:prob:rv}{distribución} binomial responde a una pregunta muy práctica: \emph{si la
\omterm{def:g10:proba:distribution}{probabilidad} de éxito es realmente $p$, qué números de éxitos son
verosímiles?}

\begin{example}\label{ex:g11:binom:decision}
Se supone que una máquina produce como mucho un $10\,\%$ de piezas
defectuosas. En un lote de $10$ piezas, $4$ son defectuosas. ¿Mala suerte
o máquina averiada? Si la máquina está bien, el número de defectuosas
sigue $\mathcal B(10, 0.1)$, y
\[
\P(X \geq 4) = 1 - \P(X \leq 3) \approx 1 - 0.987 = 0.013 :
\]
alrededor de una posibilidad entre $80$. Observar un \omterm{def:g11:prob:model}{suceso} tan
improbable es una señal fuerte: se \emph{rechaza} la hipótesis de que la
máquina siga en el $10\,\%$, sin olvidar que la decisión podría ser
equivocada con \omterm{def:g10:proba:distribution}{probabilidad} de en torno a $0.013$.
\end{example}

\begin{method}[Regla de decisión a partir de un modelo binomial]\label{met:g11:binom:decision}
Para juzgar un número observado $k$ de éxitos frente a una hipótesis
$X \sim \mathcal B(n, p)$: calcula la \omterm{def:g10:proba:distribution}{probabilidad}, bajo la hipótesis, de
un resultado \emph{al menos tan extremo} como $k$. Si esa \omterm{def:g10:proba:distribution}{probabilidad} es
muy pequeña (un convenio habitual: por debajo del $5\,\%$), rechaza la
hipótesis; en caso contrario, la observación es compatible con ella. El
umbral es una elección, no un teorema: la estadística cuantifica el
riesgo y quien la usa lo asume.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g11:binom:1}
Amplía el \omterm{prop:g11:binom:pascal}{triángulo de Pascal} hasta la fila $7$ y da los valores de
$\binom62$, $\binom{7}{3}$ y $\binom74$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:binom:2}
Se lanza un dado no trucado $4$ veces; $X$ cuenta los seises. Justifica
que $X \sim \mathcal B\left(4, \frac16\right)$ y calcula $\P(X = 0)$,
$\P(X = 1)$ y $\P(X \geq 2)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:binom:3}
¿Cuál de las siguientes situaciones es binomial? Justifícalo.
\begin{enumerate}
  \item Número de caras en $20$ lanzamientos de una moneda no trucada.
  \item Número de ases en $5$ cartas repartidas de una baraja.
  \item Número de días de lluvia la semana que viene, si cada día llueve
        con \omterm{def:g10:proba:distribution}{probabilidad} $0.3$ de forma independiente.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:binom:4}
$X \sim \mathcal B(50, 0.2)$. Da $\E(X)$, $\V(X)$ y $\sigma(X)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:5}
Un arquero da en la diana con \omterm{def:g10:proba:distribution}{probabilidad} $0.7$ en cada tiro, de forma
independiente. En $6$ tiros, calcula la \omterm{def:g10:proba:distribution}{probabilidad} de exactamente $4$

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:6}
Una prueba de verdadero o falso tiene $8$ preguntas y un alumno responde

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:7}
Cada caja de cereales que se compra contiene la figurita A o la figurita
B, con \omterm{def:g10:proba:distribution}{probabilidad} $\frac12$ cada una y de forma independiente. Un

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:8}
Un jugador de baloncesto encesta los tiros libres con \omterm{def:g10:proba:distribution}{probabilidad} $p$, de
forma independiente. Sea $X \sim \mathcal B(3, p)$ el número de canastas
en tres tiros. Expresa $\P(X = 3)$ y $\P(X \geq 1)$ en \omterm{def:g11:func:function}{función} de $p$ y

$\frac{27}{64}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:9}
¿Cuántas veces hay que lanzar una moneda no trucada para que la

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:10}
Usando la regla de Pascal (\cref{prop:g11:binom:pascal}) y

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:binom:11}
Un político afirma tener un $60\,\%$ de aprobación. En una \omterm{def:g10:proba:sample}{muestra}

\begin{enumerate}
  \item Bajo esa afirmación, ¿qué \omterm{def:g11:prob:rv}{distribución} sigue el número $X$ de

  \item Usando la regla de decisión del \cref{met:g11:binom:decision} con
        un umbral del $5\,\%$, ¿es la observación compatible con la
        afirmación?
\end{enumerate}
\end{exercise}

\section{Problema: El tablero de Galton}

\begin{problem}[{Problema de fin de semana --- bolas, clavos y el triángulo
de Pascal: cómo nace la campana, por qué las eliminatorias favorecen al
equipo más fuerte y cuándo hay que protestar}]
\label{pb:g11:binom:1}
Deja caer mil bolas por una retícula de clavos, con cada rebote a
izquierda o derecha equivalente a lanzar una moneda no trucada, y las
casillas de abajo se llenan formando una campana suave y simétrica, siempre.
La máquina se llama tablero de Galton y sus matemáticas son exactamente la
\omterm{def:g11:prob:rv}{distribución} binomial de este capítulo
(\cref{thm:g11:binom:binomial}). Este problema construye el triángulo, pone
en marcha el tablero, arbitra una eliminatoria al mejor de siete y termina
donde la binomial se gana el sueldo: decidiendo cuándo una observación
debe hacernos dudar de una afirmación.

\medskip
\noindent\textbf{Parte I --- El triángulo.}
\begin{enumerate}
  \item Construye el \omterm{prop:g11:binom:pascal}{triángulo de Pascal} hasta la fila $6$
        (\cref{prop:g11:binom:pascal}). Enuncia y explica en una frase la
        simetría $\binom nk = \binom{n}{n-k}$ (elegir $k$ objetos es lo
        mismo que\,\dots).
  \item Comprueba en las filas $4$ y $5$ que cada fila suma $2^n$ y
        demuéstralo: ¿qué cuentan todos los $\binom nk$ juntos?
  \item Vuelve a deducir la regla de Pascal
        $\binom{n+1}{k} = \binom nk + \binom{n}{k-1}$ con el argumento del
        comité: fija a una persona destacada y separa los comités según lo
        que le ocurra a esa persona.
  \item Calcula $\binom73$ de dos maneras: con el triángulo y con la
        fórmula de los factoriales.
  \item Comprueba la identidad de la escalera
        $\binom22 + \binom32 + \binom42 + \binom52 = \binom63$ y
        explícala aplicando en cascada la regla de Pascal a $\binom63$
        hacia abajo.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El tablero.}
Una bola cae a través de $n$ filas de clavos; en cada clavo rebota a la
izquierda o a la derecha con \omterm{def:g10:proba:distribution}{probabilidad} $\frac12$, de forma
independiente. Numeramos las casillas de $0$ a $n$ según el número de
rebotes a la derecha.
\begin{enumerate}[resume]
  \item Explica, con la lista de comprobación del
        \cref{met:g11:binom:recognize}, por qué el número de casilla sigue
        la \omterm{def:g11:prob:rv}{distribución} binomial $\mathcal B\!\left(n, \frac12\right)$.
  \item Para un tablero pequeño ($n = 4$): da las cinco \omterm{def:g10:proba:distribution}{probabilidades} de
        casilla. ¿Cuál es la casilla más concurrida?
  \item Ahora $n = 10$ y $1\,024$ bolas: ¿cuántas bolas cabe
        \emph{esperar} en la casilla central, en la casilla $7$ y en cada
        casilla del borde? Describe la forma del montón.
  \item Para $X \sim \mathcal B\!\left(10, \frac12\right)$: calcula
        $\E(X)$, $V(X)$ y $\sigma$ (\cref{prop:g11:binom:expectation});
        calcula después la proporción de bolas que cabe esperar a menos de
        $2\sigma$ del centro (casillas $2$ a $8$) y compárala con la
        garantía de Chebyshev del \cref{pb:g11:stat:1}.
  \item Un tablero inclinado rebota a la derecha con \omterm{def:g10:proba:distribution}{probabilidad} $0.6$:
        da $\E$, $V$ y $\sigma$ para $n = 10$ y describe qué le pasa al
        montón.
  \item En una o dos frases: ¿qué es lo que, en el diseño del tablero,
        fabrica la forma de campana? ¿Y por qué tantas magnitudes del
        mundo real (estaturas, errores de medida) se amontonan igual? (El
        teorema profundo que hay detrás de las dos cosas es el teorema
        central del límite, la cima del curso de \omterm{def:g10:proba:distribution}{probabilidad} de los
        volúmenes universitarios.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Al mejor de siete.}
Dos equipos juegan una eliminatoria: el primero que gane $4$ partidos se
lleva el título, y los partidos son independientes.
\begin{enumerate}[resume]
  \item Equipos igualados ($p = \frac12$): calcula la \omterm{def:g10:proba:distribution}{probabilidad} de que
        la eliminatoria acabe en exactamente $4$ partidos.
  \item Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de que la eliminatoria llegue a los $7$
        partidos (¿cómo tiene que estar el marcador tras $6$?).
  \item Completa la \omterm{def:g11:prob:rv}{distribución} de la duración de la eliminatoria ($4$,
        $5$, $6$ o $7$ partidos) para equipos igualados y calcula la
        duración esperada. ¿Qué duraciones son las más probables?
  \item Ahora uno de los equipos gana cada partido con $p = 0.6$. Calcula
        su \omterm{def:g10:proba:distribution}{probabilidad} de llevarse la eliminatoria (ganando en $4$, $5$,
        $6$ o $7$ partidos: en cada caso gana el último y $3$ de los
        anteriores). ¿Qué le ha hecho la eliminatoria a la ventaja por
        partido?
  \item Compara con una final a partido único ($60\,\%$) y con una
        eliminatoria al mejor de $3$ (calcúlala). Enuncia el efecto
        general de la duración de la eliminatoria sobre el pulso entre
        talento y suerte, y por qué las ligas prefieren finales largas.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Cuándo protestar.}
\begin{enumerate}[resume]
  \item Se lanza una moneda $100$ veces y salen $62$ caras. Para una
        moneda no trucada, da $\E$, $\sigma$ y la \omterm{pb:g11:stat:1}{puntuación z}
        (\cref{pb:g11:stat:1}) de la observación. ¿Veredicto con el
        convenio de $2\sigma$?
  \item Un proveedor afirma tener un $2\,\%$ de piezas defectuosas. En un
        lote de $50$ encuentras $3$ defectuosas. Calcula $\P(X \geq 3)$
        bajo esa afirmación ($X \sim \mathcal B(50, 0.02)$; pasa por
        $\P(X = 0), \P(X = 1), \P(X = 2)$). ¿Es alarmante con el umbral
        del $5\,\%$ (\cref{met:g11:binom:decision},
        \cref{exo:g11:binom:11})?
  \item Insistir en la lotería: cada boleto gana algo con \omterm{def:g10:proba:distribution}{probabilidad}
        $\frac{1}{1000}$. Calcula la \omterm{def:g10:proba:distribution}{probabilidad} de al menos un premio
        con $1\,000$ boletos. La respuesta ($\approx 63\,\%$, ¡no
        $100\,\%$!) esconde una constante famosa: calcula $0.999^{1000}$ y
        guarda el número $0.368$ para el año que viene.
  \item Final: el retrato de la binomial: la lista de reconocimiento ($n$
        fijo, independencia, $p$ constante); el \omterm{prop:g11:binom:pascal}{triángulo de Pascal} como
        su tabla; la campana como su forma; $np$ y $np(1-p)$ como su
        brújula; y sus dos herederas, que esperan el año que viene: la
        campana continua y la ley de los grandes números. Una frase para
        cada cosa.
\end{enumerate}
\end{problem}
