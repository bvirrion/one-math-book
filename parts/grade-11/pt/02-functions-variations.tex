\chapter{Funções e variação}\label{ch:g11:func}

Este capítulo organiza o que é preciso saber sobre funções antes do
cálculo: a pequena família das \emph{\omterm{def:g11:func:reference}{funções de referência}}, cujos
\omterm{def:g10:functions:graph}{gráficos} devem ser conhecidos de cor, o sentido preciso de
\emph{\omterm{def:g11:func:monotone}{crescente}} e \emph{\omterm{def:g10:functions:variations}{decrescente}}, e como a variação se comporta
quando funções são combinadas.

\section{Funções e curvas}

\begin{definition}[Função, domínio, curva]\label{def:g11:func:function}
Uma \emph{função}\index{função} $f$ associa a cada número $x$ de um
conjunto $D \subseteq \R$ (seu \emph{domínio}\index{domínio}) exatamente
um número $f(x)$. Sua \emph{curva} (ou \omterm{def:g10:functions:graph}{gráfico}) é o conjunto dos pontos
$\bigl(x, f(x)\bigr)$ com $x \in D$.
\end{definition}

\begin{example}
A fórmula $f(x) = \frac{\sqrt{x+2}}{x-1}$ exige $x + 2 \geq 0$ (por causa
da raiz quadrada) e $x \neq 1$ (denominador não nulo): o \omterm{def:g11:func:function}{domínio} é
$\intco{-2}{1} \cup \intoo{1}{+\infty}$.
\end{example}

\section{As funções de referência}

As cinco funções a seguir e suas curvas devem virar reflexo.

\begin{definition}[Funções de referência]\label{def:g11:func:reference}
\index{funções de referência}
\[
x \mapsto x^2 \text{ (quadrado)}, \quad
x \mapsto x^3 \text{ (cubo)}, \quad
x \mapsto \sqrt{x}, \quad
x \mapsto \frac{1}{x}, \quad
x \mapsto \abs{x}.
\]
Seus \omterm{def:g11:func:function}{domínios} são, respectivamente, $\R$, $\R$, $\intco{0}{+\infty}$,
$\R \setminus \{0\}$ e $\R$. A curva de $x \mapsto \frac1x$ é uma
\emph{hipérbole}\index{hipérbole}.
\end{definition}

\section{Variação em um intervalo}

\begin{definition}[Crescente, decrescente]\label{def:g11:func:monotone}
Seja $f$ definida em um \omterm{def:g10:numbers:interval}{intervalo} $I$. A \omterm{def:g11:func:function}{função} $f$ é
\emph{crescente}\index{função crescente} em $I$ se preserva a ordem:
\[
\text{para todos } u, v \in I: \quad u < v \implies f(u) \leq f(v),
\]
e \emph{\omterm{def:g10:functions:variations}{decrescente}} se inverte a ordem ($u < v \implies f(u) \geq
f(v)$). Com desigualdades estritas à direita, $f$ é
\emph{estritamente} crescente (resp.\ \omterm{def:g10:functions:variations}{decrescente}). Uma \omterm{def:g11:func:function}{função} crescente
ou \omterm{def:g10:functions:variations}{decrescente} em $I$ é dita \emph{monótona} em $I$.
\end{definition}

\begin{method}[Demonstrar uma variação pela definição]\label{met:g11:func:variation}
Tome dois números arbitrários $u < v$ do \omterm{def:g10:numbers:interval}{intervalo} e determine o sinal da
diferença $f(v) - f(u)$, em geral \emph{fatorando-a}. Se o sinal for
constante no \omterm{def:g10:numbers:interval}{intervalo}, conclua.
\end{method}

\begin{proposition}[Variação das funções de referência]\label{prop:g11:func:refvariations}
\begin{enumerate}
  \item $x^2$ é estritamente \omterm{def:g10:functions:variations}{decrescente} em $\intoc{-\infty}{0}$ e
        estritamente \omterm{def:g11:func:monotone}{crescente} em $\intco{0}{+\infty}$.
  \item $x^3$ é estritamente \omterm{def:g11:func:monotone}{crescente} em $\R$.
  \item $\sqrt x$ é estritamente \omterm{def:g11:func:monotone}{crescente} em $\intco{0}{+\infty}$.
  \item $\frac1x$ é estritamente \omterm{def:g10:functions:variations}{decrescente} em $\intoo{-\infty}{0}$ e em
        $\intoo{0}{+\infty}$ (mas \emph{não} na \omterm{def:g10:numbers:interunion}{união} dos dois).
  \item $\abs x$ é estritamente \omterm{def:g10:functions:variations}{decrescente} em $\intoc{-\infty}{0}$ e
        estritamente \omterm{def:g11:func:monotone}{crescente} em $\intco{0}{+\infty}$.
\end{enumerate}
\end{proposition}

\begin{proof}
Sejam $u < v$. Fatoramos $f(v) - f(u)$ em cada caso.

\emph{1.} $v^2 - u^2 = (v-u)(v+u)$. Em $\intco{0}{+\infty}$: $v - u > 0$
e $v + u > 0$ (pois $v > u \geq 0$), logo $v^2 > u^2$. Em
$\intoc{-\infty}{0}$: $v + u < 0$, logo $v^2 < u^2$.

\emph{2.} $v^3 - u^3 = (v - u)(v^2 + uv + u^2)$, e o segundo fator vale
$\left(v + \frac u2\right)^2 + \frac{3u^2}{4} > 0$, a menos que
$u = v = 0$. Logo $v^3 > u^3$ sempre que $v > u$.

\emph{3.} Para $0 \leq u < v$:
$\sqrt v - \sqrt u = \frac{(\sqrt v - \sqrt u)(\sqrt v + \sqrt u)}
{\sqrt v + \sqrt u} = \frac{v - u}{\sqrt v + \sqrt u} > 0$.

\emph{4.} Para $0 < u < v$: $\frac1v - \frac1u = \frac{u - v}{uv} < 0$,
pois $u - v < 0$ e $uv > 0$; mesmo cálculo para $u < v < 0$. Na \omterm{def:g10:numbers:interunion}{união} a
afirmação falha: $-1 < 1$, mas $\frac{1}{-1} = -1 < 1 = \frac11$.

\emph{5.} Em $\intco{0}{+\infty}$, $\abs x = x$ é estritamente
\omterm{def:g11:func:monotone}{crescente}; em $\intoc{-\infty}{0}$, $\abs x = -x$ é estritamente
\omterm{def:g10:functions:variations}{decrescente}.
\end{proof}

\begin{remark}
O item 4 é uma armadilha clássica: ``\omterm{def:g10:functions:variations}{decrescente} em $\intoo{-\infty}{0}$
e em $\intoo{0}{+\infty}$'' \emph{não} é o mesmo que ``\omterm{def:g10:functions:variations}{decrescente} em
$\R \setminus \{0\}$''. A variação só faz sentido em \omterm{def:g10:numbers:interval}{intervalos}.
\end{remark}

\begin{example}[Comparar sem calcular]
Como $\sqrt x$ é estritamente \omterm{def:g11:func:monotone}{crescente}, $\sqrt{7} < \sqrt{8}$; como
$x^2$ é estritamente \omterm{def:g10:functions:variations}{decrescente} em $\intoc{-\infty}{0}$,
$(-3)^2 > (-2)^2$. A monotonicidade transforma comparações de \omterm{def:g10:functions:function}{imagens} em
comparações de argumentos.
\end{example}

\section{Novas funções a partir de antigas}

\begin{proposition}[Transformações]\label{prop:g11:func:transforms}
Seja $u$ uma \omterm{def:g11:func:function}{função} definida em um \omterm{def:g10:numbers:interval}{intervalo} $I$ e seja $k \in \R$.
\begin{enumerate}
  \item $u + k$ tem a \emph{mesma} variação de $u$ em $I$; sua curva é a
        de $u$ deslocada verticalmente de $k$.
  \item Para $\lambda > 0$, $\lambda u$ tem a mesma variação de $u$; para
        $\lambda < 0$, a variação se inverte.
  \item Se $u \geq 0$ em $I$, então $\sqrt u$ tem a mesma variação de
        $u$.
  \item Se $u \neq 0$ e $u$ tem sinal constante em $I$, então $\frac1u$
        tem variação \emph{oposta} à de $u$.
\end{enumerate}
\end{proposition}

\begin{proof}
Sejam $u < v$ em $I$ (com um abuso de notação, comparamos os valores de
$x$, $u < v$; escreva $s = u(x_1)$, $t = u(x_2)$ para as \omterm{def:g10:functions:function}{imagens} de
$x_1 < x_2$).

\emph{1.} $(u+k)(x_2) - (u+k)(x_1) = t - s$: a diferença das \omterm{def:g10:functions:function}{imagens} não
muda, logo seu sinal não muda.

\emph{2.} $\lambda t - \lambda s = \lambda(t - s)$: o sinal se mantém se
$\lambda > 0$ e se inverte se $\lambda < 0$.

\emph{3.} Se $u$ é \omterm{def:g11:func:monotone}{crescente} e $x_1 < x_2$, então $0 \leq s \leq t$, e
como $\sqrt{}$ é \omterm{def:g11:func:monotone}{crescente} (\cref{prop:g11:func:refvariations}), vem
$\sqrt s \leq \sqrt t$. Mesmo argumento no caso \omterm{def:g10:functions:variations}{decrescente}.

\emph{4.} Se $u$ é \omterm{def:g11:func:monotone}{crescente}, $0 < s \leq t$ (ou $s \leq t < 0$), então
$\frac1s \geq \frac1t$, porque $x \mapsto \frac1x$ é \omterm{def:g10:functions:variations}{decrescente} de cada
lado de $0$: a ordem se inverte.
\end{proof}

\begin{example}\label{ex:g11:func:composite}
Estude $f(x) = \sqrt{x^2 + 1}$ em $\R$. A \omterm{def:g11:func:function}{função} interna
$u(x) = x^2 + 1$ é positiva, \omterm{def:g10:functions:variations}{decrescente} em $\intoc{-\infty}{0}$ e
\omterm{def:g11:func:monotone}{crescente} em $\intco{0}{+\infty}$ (um quadrado deslocado). Pelo item 3,
$f$ tem a mesma variação: \omterm{def:g10:functions:variations}{decrescente} e depois \omterm{def:g11:func:monotone}{crescente}, com \omterm{def:g10:functions:extrema}{mínimo}
$f(0) = 1$.
\end{example}

\section{Funções pares e ímpares}

\begin{definition}[Paridade]\label{def:g11:func:parity}
Seja $f$ definida em um \omterm{def:g11:func:function}{domínio} $D$ simétrico em relação a $0$ (isto é,
$-x \in D$ sempre que $x \in D$). A \omterm{def:g11:func:function}{função} $f$ é
\emph{par}\index{função par} se $f(-x) = f(x)$ para todo $x \in D$, e
\emph{ímpar}\index{função ímpar} se $f(-x) = -f(x)$ para todo $x \in D$.
\end{definition}

\begin{proposition}[Significado geométrico]\label{prop:g11:func:paritysym}
A curva de uma \omterm{def:g11:func:parity}{função par} é simétrica em relação ao eixo $y$; a curva de
uma \omterm{def:g11:func:parity}{função ímpar} é simétrica em relação à origem.
\end{proposition}

\begin{proof}
Se $f$ é \omterm{def:g11:func:parity}{par} e $(x, y)$ está na curva (ou seja, $y = f(x)$), então
$f(-x) = f(x) = y$: o ponto espelhado $(-x, y)$ também está na curva. Se
$f$ é \omterm{def:g11:func:parity}{ímpar}, $f(-x) = -y$: o ponto $(-x, -y)$, simétrico de $(x,y)$ em
relação à origem, está na curva.
\end{proof}

\begin{example}
$x^2$ e $\abs x$ são \omterm{def:g11:func:parity}{pares}; $x^3$ e $\frac1x$ são \omterm{def:g11:func:parity}{ímpares};
$f(x) = x^2 + x$ não é nem \omterm{def:g11:func:parity}{par} nem \omterm{def:g11:func:parity}{ímpar}: $f(-1) = 0$ enquanto
$f(1) = 2$, de modo que $f(-1) \neq \pm f(1)$. A paridade corta o
trabalho pela metade: uma \omterm{def:g11:func:parity}{função par} ou \omterm{def:g11:func:parity}{ímpar} só precisa ser estudada em
$\intco{0}{+\infty}$.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:func:1}
Dê o \omterm{def:g11:func:function}{domínio} de cada \omterm{def:g11:func:function}{função}:

g(x) = \frac{1}{x^2 - 4}, \qquad

\begin{exercise}[$\star$]\label{exo:g11:func:2}

\[
\sqrt{11} \text{ e } \sqrt{13}; \qquad
(-2.1)^2 \text{ e } (-2.05)^2; \qquad
\frac{1}{0.9} \text{ e } \frac{1}{1.1}; \qquad
(-1.01)^3 \text{ e } (-0.99)^3 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:func:3}
Determine se cada \omterm{def:g11:func:function}{função} é \omterm{def:g11:func:parity}{par}, \omterm{def:g11:func:parity}{ímpar} ou nenhuma das duas:
\[
f(x) = 3x^4 - x^2, \qquad
g(x) = x^3 + x, \qquad
h(x) = x^2 + x^3, \qquad
k(x) = \frac{x}{x^2+1} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:func:4}
Uma \omterm{def:g11:func:function}{função} $f$ tem o seguinte comportamento: \omterm{def:g11:func:monotone}{crescente} em
$\intcc{-3}{1}$, de $f(-3) = -2$ até $f(1) = 4$, e depois \omterm{def:g10:functions:variations}{decrescente} em
$\intcc{1}{5}$ até $f(5) = 0$. Quantas soluções a \omterm{def:g10:algebra:equation}{equação} $f(x) = 1$ tem
em $\intcc{-3}{5}$? E $f(x) = 5$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:5}
Usando o \cref{met:g11:func:variation} (sinal de $f(v) - f(u)$), prove
que $f(x) = x^2 - 4x$ é estritamente \omterm{def:g11:func:monotone}{crescente} em $\intco{2}{+\infty}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:6}
Sem calcular nenhuma derivada, dê a variação de
\[
f(x) = \sqrt{5 - x} \ \text{ em } \intoc{-\infty}{5},
\quad
g(x) = \frac{1}{x^2 + 1} \ \text{ em } \intco{0}{+\infty},
\quad
h(x) = 3 - 2\sqrt{x} \ \text{ em } \intco{0}{+\infty}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:7}
Seja $f(x) = (x-2)^2 - 3$ em $\R$.
\begin{enumerate}
  \item Dê a variação de $f$ e seu \omterm{def:g10:functions:extrema}{mínimo}.
  \item Deduza a variação de $g(x) = \frac{1}{(x-2)^2+1}$ e seu \omterm{def:g10:functions:extrema}{máximo}.
        (Note que $g = \frac{1}{f + 4}$.)
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:8}
A curva de uma \omterm{def:g11:func:function}{função} $f$ passa pelos pontos $(0, 1)$, $(2, 3)$ e
$(4, 1)$, e $f$ é \omterm{def:g11:func:monotone}{crescente} em $\intcc{0}{2}$ e \omterm{def:g10:functions:variations}{decrescente} em
$\intcc{2}{4}$. Esboce uma curva possível e depois esboce as curvas de
$f + 2$, de $-f$ e de $2f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:9}
Mostre que a \omterm{def:g11:func:function}{função} $f(x) = \frac{2x+1}{x+1}$ pode ser escrita como
$f(x) = 2 - \frac{1}{x+1}$ e deduza sua variação em
$\intoo{-1}{+\infty}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:10}
Verdadeiro ou falso (justifique ou dê um contraexemplo):
\begin{enumerate}
  \item A soma de duas funções \omterm{def:g11:func:monotone}{crescentes} em $I$ é \omterm{def:g11:func:monotone}{crescente} em $I$.
  \item O produto de duas funções \omterm{def:g11:func:monotone}{crescentes} em $I$ é \omterm{def:g11:func:monotone}{crescente} em $I$.
  \item Se $f$ é \omterm{def:g11:func:monotone}{crescente} em $I$, então $f^2$ ($= f \times f$) é
        \omterm{def:g11:func:monotone}{crescente} em $I$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:func:11}
Seja $f(x) = x + \frac{1}{x}$ para $x > 0$.
\begin{enumerate}
  \item Mostre que, para $0 < u < v$,
        $f(v) - f(u) = (v - u)\,\frac{uv - 1}{uv}$.
  \item Deduza que $f$ é estritamente \omterm{def:g10:functions:variations}{decrescente} em $\intoc{0}{1}$ e
        estritamente \omterm{def:g11:func:monotone}{crescente} em $\intco{1}{+\infty}$.
  \item Conclua que $x + \frac1x \geq 2$ para todo $x > 0$, com igualdade
        somente em $x = 1$.
\end{enumerate}
\end{exercise}

\section{Problema: A fábrica de simetrias}

\begin{problem}[{Problema de fim de semana --- toda função é uma
parte par mais uma parte ímpar: as regras da paridade, o jogo das
transformações e mínimos conquistados sem cálculo}]
\label{pb:g11:func:1}
Algumas funções são simétricas por espelho, outras por meia volta,
a maioria não é nem uma coisa nem outra --- e, no entanto, um
pequeno milagre algébrico afirma que \emph{toda} \omterm{def:g11:func:function}{função} é, de
exatamente uma maneira, a soma de uma parte simétrica por espelho
com uma parte simétrica por meia volta. Este problema demonstra o
milagre, joga o jogo das transformações sobre as curvas de
referência (\cref{prop:g11:func:transforms}) e encerra com a
técnica de \omterm{def:g10:algebra:expand}{fatorar} diferenças do \cref{exo:g11:func:11}, que ganha
problemas de minimização um capítulo inteiro antes de a derivada
chegar.

\medskip
\noindent\textbf{Parte I --- Fluência com paridade.}
\begin{enumerate}
  \item \omterm{def:g11:func:parity}{Par}, \omterm{def:g11:func:parity}{ímpar} ou nenhuma das duas
        (\cref{def:g11:func:parity})? Justifique cada uma:
        $x^4 - 3x^2$; \ $x^3 - x$; \ $x^2 + x$; \ $\abs x$; \
        $\frac1x$.
  \item Enuncie o significado geométrico de cada veredicto
        (\cref{prop:g11:func:paritysym}): que simetria tem a
        curva de uma \omterm{def:g11:func:parity}{função par}? E a de uma \omterm{def:g11:func:parity}{função ímpar}?
  \item A paridade se multiplica como os sinais: prove que o
        produto de duas \omterm{def:g11:func:parity}{funções ímpares} é \omterm{def:g11:func:parity}{par} e complete toda a
        tabela de multiplicação (\omterm{def:g11:func:parity}{par} $\times$ \omterm{def:g11:func:parity}{par}, \omterm{def:g11:func:parity}{par} $\times$
        \omterm{def:g11:func:parity}{ímpar}, \omterm{def:g11:func:parity}{ímpar} $\times$ \omterm{def:g11:func:parity}{ímpar}).
  \item Consequências rápidas: se $f$ é \omterm{def:g11:func:parity}{ímpar} e está definida em
        $0$, quanto vale $f(0)$? Se $f$ é \omterm{def:g11:func:parity}{par} com $f(3) = 7$,
        quanto vale $f(-3)$? E mostre que a única \omterm{def:g11:func:function}{função} ao mesmo
        tempo \omterm{def:g11:func:parity}{par} e \omterm{def:g11:func:parity}{ímpar} é a \omterm{def:g11:func:function}{função} nula.
  \item Quanto da curva de uma \omterm{def:g11:func:parity}{função par} determina toda ela? Dê
        uma \omterm{def:g11:func:function}{função} que não seja nem \omterm{def:g11:func:parity}{par} nem \omterm{def:g11:func:parity}{ímpar} e explique o
        que falha.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O teorema da decomposição.}
Para uma \omterm{def:g11:func:function}{função} $f$ definida em um \omterm{def:g11:func:function}{domínio} simétrico em relação a
$0$, ponha
\[
E(x) = \frac{f(x) + f(-x)}{2},
\qquad
O(x) = \frac{f(x) - f(-x)}{2} .
\]
\begin{enumerate}[resume]
  \item Calcule $E$ e $O$ para $f(x) = x^3 + x^2 + 1$ e
        verifique: $E$ \omterm{def:g11:func:parity}{par}, $O$ \omterm{def:g11:func:parity}{ímpar}, $E + O = f$.
  \item Demonstre o enunciado geral: para qualquer $f$, a \omterm{def:g11:func:function}{função}
        $E$ é \omterm{def:g11:func:parity}{par}, $O$ é \omterm{def:g11:func:parity}{ímpar} e $f = E + O$ --- \emph{toda
        \omterm{def:g11:func:function}{função} se decompõe em uma parte \omterm{def:g11:func:parity}{par} e uma parte \omterm{def:g11:func:parity}{ímpar}}.
  \item Demonstre que a decomposição é única: se
        $f = E_1 + O_1 = E_2 + O_2$ com $E_i$ \omterm{def:g11:func:parity}{par} e $O_i$ \omterm{def:g11:func:parity}{ímpar},
        use a questão 4 sobre $E_1 - E_2$.
  \item Decomponha $f(x) = \dfrac{1}{1 - x}$ (para
        $x \neq \pm 1$): reduza $E$ e $O$ ao denominador comum
        $1 - x^2$ e simplifique.
  \item Um olhar adiante, uma frase para cada: as partes \omterm{def:g11:func:parity}{par} e
        \omterm{def:g11:func:parity}{ímpar} da \omterm{def:g11:func:function}{função} exponencial serão as funções gêmeas do
        ano 12 (cosseno e seno hiperbólicos); e o mesmo instinto
        --- dividir um objeto em pedaços simétricos e tratar cada
        um com a sua própria simetria --- move a análise de som e
        de sinais dos volumes de graduação. Por que uma divisão
        dessas é \emph{útil}, em geral?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O jogo das transformações.}
\begin{enumerate}[resume]
  \item Partindo da curva de $x \mapsto x^2$, descreva com
        precisão as curvas de $(x - 3)^2$, \ $x^2 + 2$, \
        $(x - 3)^2 + 2$, \ $-x^2$, \ $2x^2$
        (\cref{prop:g11:func:transforms}).
  \item Reduza $f(x) = \abs{x - 2} + \abs{x + 2}$ a uma fórmula
        por partes nos três \omterm{def:g10:numbers:interval}{intervalos} delimitados por $-2$ e $2$
        e descreva sua curva (ela tem um fundo plano famoso).
  \item Quais transformações respeitam a paridade \omterm{def:g11:func:parity}{par}? Decida,
        com demonstração ou contraexemplo: deslocamento
        horizontal $f(x - 3)$; deslocamento vertical
        $f(x) + 2$; dilatação vertical $2 f(x)$ (todos aplicados
        a uma $f$ \omterm{def:g11:func:parity}{par}).
  \item Resolva $\abs{x - 2} + \abs{x + 2} = 6$ usando a fórmula
        por partes da questão 12.
  \item Engenharia reversa: uma \omterm{def:g11:quad:parabola}{parábola} tem vértice $(2, -3)$ e
        passa por $(0, 1)$. Reconstrua sua \omterm{def:g10:algebra:equation}{equação} na \omterm{thm:g11:quad:canonical}{forma
        canônica}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- \omterm{def:g10:functions:extrema}{Mínimos} sem cálculo.}
\begin{enumerate}[resume]
  \item Espelhe o \cref{exo:g11:func:11} para
        $g(x) = x + \dfrac4x$ em $\intoo{0}{+\infty}$: mostre que
        $g(v) - g(u) = (v - u)\,\dfrac{uv - 4}{uv}$, localize o
        \omterm{def:g10:functions:extrema}{mínimo} e dê seu valor.
  \item Aplique: entre todos os retângulos de área $16$, qual tem
        o menor perímetro? (Escreva $P = 2\left(x +
        \frac{16}{x}\right)$ e use a máquina da questão 16.) Um
        velho conhecido conclui.
  \item Determine a variação de $h(x) = \sqrt{x^2 + 1}$ em todo o
        $\R$, combinando um argumento de composição em
        $\intco{0}{+\infty}$ com um argumento de paridade para o
        resto.
  \item O ponto mais próximo: encontre o ponto da curva
        $y = \sqrt x$ mais próximo de $(3, 0)$. (Minimize o
        \emph{quadrado} da distância --- um \omterm{def:g11:quad:quadratic}{trinômio do segundo
        grau} em $x$ --- e explique por que minimizar o quadrado é
        legítimo.)
  \item Final, a visita à fábrica: as curvas de referência são a
        matéria-prima; as transformações, as máquinas; a paridade
        e a decomposição, o controle de qualidade; a fatoração da
        diferença, a bancada de medição. Uma frase sobre cada ---
        e nomeie a ferramenta elétrica que o próximo capítulo
        entrega.
\end{enumerate}
\end{problem}
