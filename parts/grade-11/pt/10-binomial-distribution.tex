\chapter{A distribuição binomial}\label{ch:g11:binom}

Repita várias vezes, de forma independente, o mesmo experimento de
sim/não e conte os sucessos: a \omterm{def:g11:prob:rv}{distribuição} resultante --- a
\emph{binomial} --- é a mais importante de todas as distribuições
discretas. Este capítulo a constrói com \omterm{met:g10:proba:tree}{árvores} e contagem de caminhos; a
fórmula fechada para o número de caminhos (com fatoriais) chega com as
ferramentas de contagem do \cref{ch:g12:comb}, e a \omterm{def:g11:prob:rv}{distribuição} é
retomada no \cref{ch:g12:randvar}.

\section{Ensaios de Bernoulli}

\begin{definition}[Ensaio de Bernoulli]\label{def:g11:binom:bernoulli}
Um \emph{ensaio de Bernoulli}\index{ensaio de Bernoulli} é um experimento
com exatamente dois resultados: \emph{sucesso}, com \omterm{def:g10:proba:distribution}{probabilidade} $p$, e
\emph{fracasso}, com \omterm{def:g10:proba:distribution}{probabilidade} $1 - p$. Diz-se que a \omterm{def:g11:prob:rv}{variável
aleatória} $X$ igual a $1$ no sucesso e a $0$ no fracasso segue a
\emph{\omterm{def:g11:prob:rv}{distribuição} de Bernoulli} $\mathcal B(p)$; então
\[
\E(X) = p,
\qquad
\V(X) = p(1 - p).
\]
\end{definition}

\begin{proof}[Demonstração das duas fórmulas]
$\E(X) = p \times 1 + (1-p) \times 0 = p$; e, como $X^2 = X$ (tanto $0$
quanto $1$ são iguais aos próprios quadrados), $\E(X^2) = p$, logo, pela
\cref{prop:g11:prob:konig}, $\V(X) = p - p^2 = p(1-p)$.
\end{proof}

\begin{definition}[Ensaios independentes repetidos]\label{def:g11:binom:repeated}
Repetir um \omterm{def:g11:binom:bernoulli}{ensaio de Bernoulli} $n$ vezes de forma \emph{independente}
significa: o resultado de cada ensaio não influencia os demais, e a
\omterm{def:g10:proba:distribution}{probabilidade} de uma \omterm{def:g11:seq:sequence}{sequência} completa de resultados é o \emph{produto}
das \omterm{def:g10:proba:distribution}{probabilidades} ao longo do caminho correspondente da \omterm{met:g10:proba:tree}{árvore} --- $p$
para cada sucesso, $1 - p$ para cada fracasso.
\end{definition}

\begin{example}\label{ex:g11:binom:threetrials}
Três ensaios independentes com \omterm{def:g10:proba:distribution}{probabilidade} de sucesso $p$. A \omterm{def:g11:seq:sequence}{sequência}
SFS (sucesso, fracasso, sucesso) tem \omterm{def:g10:proba:distribution}{probabilidade} $p(1-p)p = p^2(1-p)$
--- e o mesmo vale para \emph{toda} \omterm{def:g11:seq:sequence}{sequência} com exatamente dois
sucessos, quaisquer que sejam as posições: só o \emph{número} de S e de F
importa.
\end{example}

\section{Contagem de caminhos e coeficientes binomiais}

\begin{definition}[Coeficiente binomial]\label{def:g11:binom:coefficient}
Na \omterm{met:g10:proba:tree}{árvore} de $n$ ensaios independentes, o \emph{coeficiente
binomial}\index{coeficiente binomial} $\binom{n}{k}$ (lê-se ``$n$ escolhe
$k$'') é o número de caminhos que contêm exatamente $k$ sucessos.
\end{definition}

\begin{example}
$\binom{3}{2} = 3$: os caminhos SSF, SFS, FSS. Do mesmo modo,
$\binom{3}{0} = 1$ (o caminho FFF), $\binom{3}{1} = 3$ e
$\binom{3}{3} = 1$. Por convenção e pela \omterm{met:g10:proba:tree}{árvore},
$\binom n0 = \binom nn = 1$ para todo $n$.
\end{example}

\begin{proposition}[Relação de Pascal]\label{prop:g11:binom:pascal}
\index{triângulo de Pascal}
Para $1 \leq k \leq n - 1$:
\[
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}.
\]
\end{proposition}

\begin{proof}
Separe os caminhos da \omterm{met:g10:proba:tree}{árvore} de $n$ ensaios com $k$ sucessos conforme o
seu \emph{último} ensaio. Os que terminam em sucesso vêm de um caminho
dos primeiros $n-1$ ensaios com $k - 1$ sucessos: há $\binom{n-1}{k-1}$
deles. Os que terminam em fracasso prolongam um caminho com $k$ sucessos
entre os primeiros $n-1$ ensaios: $\binom{n-1}{k}$ deles. Todo caminho é
de exatamente um dos dois tipos.
\end{proof}

A relação de Pascal gera os coeficientes linha a linha --- cada entrada é
a soma das duas que estão acima dela:
\[
\begin{array}{ccccccccccc}
&&&&&1&&&&&\\
&&&&1&&1&&&&\\
&&&1&&2&&1&&&\\
&&1&&3&&3&&1&&\\
&1&&4&&6&&4&&1&\\
1&&5&&10&&10&&5&&1
\end{array}
\]

\begin{remark}
Uma fórmula fechada, $\binom nk = \frac{n!}{k!(n-k)!}$, junto com uma
teoria sistemática de contagem, é estabelecida no \cref{ch:g12:comb}.
Neste nível, o \omterm{prop:g11:binom:pascal}{triângulo de Pascal} calcula todos os coeficientes de que
precisamos.
\end{remark}

\section{A distribuição binomial}

\begin{theorem}[Distribuição binomial]\label{thm:g11:binom:binomial}
Seja $X$ o número de sucessos em $n$ \omterm{def:g11:binom:bernoulli}{ensaios de Bernoulli} independentes
de parâmetro $p$. Então $X$ segue a \emph{distribuição
binomial}\index{distribuição binomial} $\mathcal B(n, p)$:
\[
\P(X = k) = \binom{n}{k}\, p^k (1-p)^{n-k},
\qquad k = 0, 1, \dots, n .
\]
\end{theorem}

\begin{proof}
O \omterm{def:g11:prob:model}{evento} $X = k$ é o conjunto de todos os caminhos com exatamente $k$
sucessos. Cada caminho desses tem \omterm{def:g10:proba:distribution}{probabilidade} $p^k(1-p)^{n-k}$: o
produto ao longo do caminho contém $k$ fatores $p$ e $n - k$ fatores
$1-p$, em alguma ordem (\cref{def:g11:binom:repeated}). Há $\binom nk$
caminhos assim (\cref{def:g11:binom:coefficient}), e suas \omterm{def:g10:proba:distribution}{probabilidades}
se somam.
\end{proof}

\begin{example}\label{ex:g11:binom:quiz}
Um teste tem $5$ questões independentes, cada uma com $4$ alternativas;
um aluno responde ao acaso, de modo que cada questão é um sucesso com
$p = \frac14$. O número $X$ de respostas corretas segue
$\mathcal B\left(5, \frac14\right)$ e, usando a linha $5$ do \omterm{prop:g11:binom:pascal}{triângulo de
Pascal}:
\[
\P(X = 2) = \binom52 \left(\frac14\right)^2\left(\frac34\right)^3
= 10 \times \frac{1}{16} \times \frac{27}{64}
= \frac{270}{1024} \approx 0.26 .
\]
A \omterm{def:g10:proba:distribution}{probabilidade} de pelo menos uma resposta correta usa o \omterm{def:g10:proba:operations}{complementar}:
$\P(X \geq 1) = 1 - \P(X = 0) = 1 - \left(\frac34\right)^5 \approx
0.76$.
\end{example}

\begin{proposition}[Esperança e variância]\label{prop:g11:binom:expectation}
Se $X \sim \mathcal B(n, p)$:
\[
\E(X) = np,
\qquad
\V(X) = np(1-p).
\]
\end{proposition}

\begin{proof}[Justificativa]
Escreva $X = X_1 + X_2 + \dots + X_n$, em que $X_i$ vale $1$ se o
$i$-ésimo ensaio é bem-sucedido: cada $X_i$ é uma variável de Bernoulli
de \omterm{def:g11:prob:expectation}{esperança} $p$ (\cref{def:g11:binom:bernoulli}). As \omterm{def:g11:stat:mean}{médias} se somam ---
somar as $n$ contribuições dá $\E(X) = np$. Que as \emph{\omterm{def:g11:prob:variance}{variâncias}}
também se somam para variáveis independentes é verdade, mas mais
delicado: a fórmula da \omterm{def:g11:prob:variance}{variância} é \emph{admitida neste nível} e
demonstrada no \cref{ch:g12:sums}.
\end{proof}

\begin{method}[Reconhecer uma situação binomial]\label{met:g11:binom:recognize}
Antes de escrever $X \sim \mathcal B(n, p)$, confira três ingredientes:
um \emph{número fixo} $n$ de ensaios, decidido de antemão; cada ensaio
com \emph{dois resultados} e a \emph{mesma} \omterm{def:g10:proba:distribution}{probabilidade} de sucesso $p$;
e ensaios \emph{independentes} (com reposição, ou dispositivos
separados). Retirar sem reposição de uma população pequena \emph{não} é
binomial --- a \omterm{def:g10:proba:distribution}{probabilidade} muda a cada retirada
(\cref{exo:g11:prob:6}).
\end{method}

\section{Amostragem: a observação é surpreendente?}

A \omterm{def:g11:prob:rv}{distribuição} binomial responde a uma pergunta muito prática:
\emph{se a \omterm{def:g10:proba:distribution}{probabilidade} de sucesso é mesmo $p$, que números de sucessos
são plausíveis?}

\begin{example}\label{ex:g11:binom:decision}
Uma máquina deveria produzir no máximo $10\%$ de peças defeituosas. Em um
lote de $10$ peças, $4$ estão defeituosas. Azar ou máquina quebrada? Se a
máquina estiver boa, o número de defeituosas segue
$\mathcal B(10, 0.1)$, e
\[
\P(X \geq 4) = 1 - \P(X \leq 3) \approx 1 - 0.987 = 0.013 :
\]
cerca de uma chance em $80$. Observar um \omterm{def:g11:prob:model}{evento} tão improvável é um sinal
forte --- \emph{rejeita-se} a hipótese de que a máquina ainda trabalha a
$10\%$, tendo em mente que a decisão pode estar errada com \omterm{def:g10:proba:distribution}{probabilidade}
de cerca de $0.013$.
\end{example}

\begin{method}[Regra de decisão a partir de um modelo binomial]\label{met:g11:binom:decision}
Para julgar um número observado $k$ de sucessos contra uma hipótese
$X \sim \mathcal B(n, p)$: calcule, sob a hipótese, a \omterm{def:g10:proba:distribution}{probabilidade} de um
resultado \emph{pelo menos tão extremo} quanto $k$. Se essa \omterm{def:g10:proba:distribution}{probabilidade}
for muito pequena (convenção comum: abaixo de $5\%$), rejeite a hipótese;
caso contrário, a observação é compatível com ela. O limiar é uma
escolha, não um teorema --- a estatística quantifica o risco, e o usuário
o aceita.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:binom:1}
Estenda o \omterm{prop:g11:binom:pascal}{triângulo de Pascal} até a linha $7$ e dê os valores de
$\binom62$, $\binom{7}{3}$ e $\binom74$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:binom:2}
Um dado honesto é lançado $4$ vezes; $X$ conta os seis. Justifique que
$X \sim \mathcal B\left(4, \frac16\right)$ e calcule $\P(X = 0)$,
$\P(X = 1)$ e $\P(X \geq 2)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:binom:3}
Qual das situações a seguir é binomial? Justifique.
\begin{enumerate}
  \item Número de caras em $20$ lançamentos de uma moeda honesta.
  \item Número de ases em $5$ cartas distribuídas de um mesmo baralho.
  \item Número de dias de chuva na próxima semana, se cada dia é chuvoso
        com \omterm{def:g10:proba:distribution}{probabilidade} $0.3$, de forma independente.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:binom:4}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:5}
Um arqueiro acerta o alvo com \omterm{def:g10:proba:distribution}{probabilidade} $0.7$ em cada disparo, de
forma independente. Em $6$ disparos, calcule a \omterm{def:g10:proba:distribution}{probabilidade} de

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:6}
Uma prova de verdadeiro ou falso tem $8$ questões; um aluno chuta cada

corretas)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:7}
Cada caixa de cereal comprada contém a figurinha A ou a figurinha B, cada

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:8}
Um jogador de basquete converte lances livres com \omterm{def:g10:proba:distribution}{probabilidade} $p$, de

em três lances. Exprima $\P(X = 3)$ e $\P(X \geq 1)$ em \omterm{def:g11:func:function}{função} de $p$ e

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:9}
Quantas vezes uma moeda honesta precisa ser lançada para que a

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:binom:10}
Usando a relação de Pascal (\cref{prop:g11:binom:pascal}) e

\begin{exercise}[$\star\star\star$]\label{exo:g11:binom:11}

  \item Usando a regra de decisão do \cref{met:g11:binom:decision} com um

\end{enumerate}
\end{exercise}

\section{Problema: O tabuleiro de Galton}

\begin{problem}[{Problema de fim de semana --- bolinhas, pinos e o
triângulo de Pascal: como nasce a forma de sino, por que as séries de
playoff favorecem o time mais forte e quando gritar que houve
trapaça}]
\label{pb:g11:binom:1}
Deixe cair mil bolinhas por uma grade de pinos, cada quique um
lançamento honesto de moeda entre esquerda e direita, e as
canaletas de baixo se preenchem formando um sino liso e simétrico
--- todas as vezes. A máquina chama-se tabuleiro de Galton, e sua
matemática é exatamente a \omterm{def:g11:prob:rv}{distribuição} binomial deste capítulo
(\cref{thm:g11:binom:binomial}). Este problema constrói o
triângulo, opera o tabuleiro, arbitra uma série melhor de sete e
termina onde a binomial ganha o seu salário: decidindo quando uma
observação deve nos fazer duvidar de uma afirmação.

\medskip
\noindent\textbf{Parte I --- O triângulo.}
\begin{enumerate}
  \item Construa o \omterm{prop:g11:binom:pascal}{triângulo de Pascal} até a linha $6$
        (\cref{prop:g11:binom:pascal}). Enuncie e explique em uma
        frase a simetria $\binom nk = \binom{n}{n-k}$ (escolher
        $k$ objetos é o mesmo que\,\dots).
  \item Verifique nas linhas $4$ e $5$ que cada linha soma $2^n$
        e demonstre isso: o que todos os $\binom nk$ juntos
        contam?
  \item Rededuza a relação de Pascal
        $\binom{n+1}{k} = \binom nk + \binom{n}{k-1}$ pelo
        argumento da comissão: fixe uma pessoa distinguida e
        separe as comissões conforme o destino dessa pessoa.
  \item Calcule $\binom73$ de duas maneiras: pelo triângulo e
        pela fórmula com fatoriais.
  \item Verifique a identidade da escada
        $\binom22 + \binom32 + \binom42 + \binom52 = \binom63$ e
        explique-a aplicando a relação de Pascal em cascata a
        partir de $\binom63$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O tabuleiro.}
Uma bolinha cai por $n$ fileiras de pinos; em cada pino ela quica
para a esquerda ou para a direita com \omterm{def:g10:proba:distribution}{probabilidade} $\frac12$, de
forma independente. Numere as canaletas de $0$ a $n$ pelo número
de quiques para a direita.
\begin{enumerate}[resume]
  \item Explique, com a lista de verificação do
        \cref{met:g11:binom:recognize}, por que o número da
        canaleta segue a \omterm{def:g11:prob:rv}{distribuição} binomial
        $\mathcal B\!\left(n, \frac12\right)$.
  \item Para um tabuleiro pequeno ($n = 4$): dê as cinco
        \omterm{def:g10:proba:distribution}{probabilidades} de canaleta. Qual canaleta fica mais
        cheia?
  \item Agora $n = 10$ e $1\,024$ bolinhas: quais são as
        quantidades \emph{esperadas} de bolinhas na canaleta
        central, na canaleta $7$ e em cada canaleta da ponta?
        Descreva a forma da pilha.
  \item Para $X \sim \mathcal B\!\left(10, \frac12\right)$:
        calcule $\E(X)$, $V(X)$ e $\sigma$
        (\cref{prop:g11:binom:expectation}); calcule em seguida
        a proporção de bolinhas esperada a menos de $2\sigma$ do
        centro (canaletas $2$ a $8$) e compare com a garantia de
        Chebyshev do \cref{pb:g11:stat:1}.
  \item Um tabuleiro inclinado quica para a direita com
        \omterm{def:g10:proba:distribution}{probabilidade} $0.6$: dê $\E$, $V$ e $\sigma$ para
        $n = 10$ e descreva o que acontece com a pilha.
  \item Em uma ou duas frases: o que, no projeto do tabuleiro,
        fabrica a forma de sino --- e por que tantas grandezas do
        mundo real (alturas, erros de medição) se empilham do
        mesmo jeito? (O teorema profundo por trás das duas coisas
        é o teorema central do limite, o cume do curso de
        \omterm{def:g10:proba:distribution}{probabilidade} dos volumes de graduação.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Melhor de sete.}
Dois times disputam uma série: o primeiro a $4$ vitórias leva o
título; as partidas são independentes.
\begin{enumerate}[resume]
  \item Times equivalentes ($p = \frac12$): calcule a
        \omterm{def:g10:proba:distribution}{probabilidade} de a série terminar em varrida (exatamente
        $4$ jogos).
  \item Calcule a \omterm{def:g10:proba:distribution}{probabilidade} de a série ir aos $7$ jogos
        completos (qual tem de ser o placar depois de $6$?).
  \item Complete a \omterm{def:g11:prob:rv}{distribuição} da duração da série ($4$, $5$,
        $6$ ou $7$ jogos) para times equivalentes e calcule a
        duração esperada. Quais durações são mais prováveis?
  \item Agora um dos times vence cada jogo com $p = 0.6$.
        Calcule sua \omterm{def:g10:proba:distribution}{probabilidade} de levar a série (vencer em
        $4$, $5$, $6$ ou $7$ jogos: em cada caso o time vence o
        último jogo e $3$ dos anteriores). O que a série fez com
        a vantagem por jogo?
  \item Compare com uma final única ($60\,\%$) e com uma melhor
        de 3 (calcule-a). Enuncie o efeito geral da duração da
        série sobre a disputa entre habilidade e sorte --- e por
        que as ligas preferem finais longas.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Quando gritar que houve trapaça.}
\begin{enumerate}[resume]
  \item Uma moeda é lançada $100$ vezes e mostra $62$ caras. Para
        uma moeda honesta, dê $\E$, $\sigma$ e o \omterm{pb:g11:stat:1}{escore z}
        (\cref{pb:g11:stat:1}) da observação. Qual o veredicto
        sob a convenção de $2\sigma$?
  \item Um fornecedor afirma ter $2\,\%$ de peças defeituosas. Em
        um lote de $50$ você encontra $3$ defeituosas. Calcule
        $\P(X \geq 3)$ sob a afirmação
        ($X \sim \mathcal B(50, 0.02)$; passe por
        $\P(X = 0), \P(X = 1), \P(X = 2)$). Alarmante ao limiar
        de $5\,\%$ (\cref{met:g11:binom:decision},
        \cref{exo:g11:binom:11})?
  \item Persistência na loteria: cada bilhete ganha (alguma
        coisa) com \omterm{def:g10:proba:distribution}{probabilidade} $\frac{1}{1000}$. Calcule a
        \omterm{def:g10:proba:distribution}{probabilidade} de pelo menos uma vitória com $1\,000$
        bilhetes. A resposta ($\approx 63\,\%$, e não
        $100\,\%$!) esconde uma constante famosa: calcule
        $0.999^{1000}$ e guarde o número $0.368$ para o ano 12.
  \item Final --- o retrato da binomial: a lista de
        reconhecimento ($n$ fixo, independência, $p$ constante);
        o \omterm{prop:g11:binom:pascal}{triângulo de Pascal} como sua tabela; o sino como sua
        forma; $np$ e $np(1-p)$ como sua bússola; e seus dois
        herdeiros à espera no ano 12 --- a curva em sino contínua
        e a lei dos grandes números. Uma frase para cada.
\end{enumerate}
\end{problem}
