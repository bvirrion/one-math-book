\chapter{Probabilidade e variáveis aleatórias}\label{ch:g11:prob}

A \omterm{def:g10:proba:distribution}{probabilidade} modela experimentos de resultado incerto: um lançamento
de dado, uma carta retirada, uma partida jogada. Este capítulo monta o
vocabulário em conjuntos finitos e apresenta em seguida a personagem
central de toda a teoria das \omterm{def:g10:proba:distribution}{probabilidades}: a \emph{\omterm{def:g11:prob:rv}{variável aleatória}}
e sua \emph{\omterm{def:g11:prob:expectation}{esperança}}. Essas noções são revistas e aprofundadas no
\cref{ch:g12:randvar}, e as \omterm{def:g10:proba:distribution}{probabilidades} condicionais vêm no
\cref{ch:g12:condprob}.

\section{Probabilidade em um conjunto finito}

\begin{definition}[Modelo probabilístico]\label{def:g11:prob:model}
Um experimento com um número finito de resultados é modelado pelo seu
\emph{espaço amostral}\index{espaço amostral}
$\Omega = \{\omega_1, \dots, \omega_n\}$ (o conjunto dos resultados)
junto com uma \emph{\omterm{def:g10:proba:distribution}{probabilidade}} $\P$ que atribui a cada resultado
$\omega_i$ um número $p_i \geq 0$, com $p_1 + \dots + p_n = 1$. Um
\emph{evento}\index{evento} é um subconjunto $A \subseteq \Omega$, e
\[
\P(A) = \sum_{\omega_i \in A} p_i .
\]
Quando todos os resultados são igualmente prováveis ($p_i = \frac1n$: o
\emph{modelo uniforme}\index{modelo uniforme}),
\[
\P(A) = \frac{\text{número de resultados em } A}
{\text{número de resultados em } \Omega}.
\]
\end{definition}

\begin{example}\label{ex:g11:prob:twodice}
Lance dois dados distinguíveis: $\Omega$ é o conjunto dos $36$ pares
ordenados $(i, j)$ com $1 \leq i, j \leq 6$, todos igualmente prováveis.
O \omterm{def:g11:prob:model}{evento} $S$ = ``a soma é $7$'' contém os $6$ pares $(1,6)$, $(2,5)$,
$(3,4)$, $(4,3)$, $(5,2)$, $(6,1)$, logo
$\P(S) = \frac{6}{36} = \frac16$. O \omterm{def:g11:prob:model}{evento} ``a soma é $12$'' contém
apenas $(6,6)$: \omterm{def:g10:proba:distribution}{probabilidade} $\frac{1}{36}$. Escolher o \omterm{def:g11:prob:model}{espaço amostral}
\emph{certo} --- pares ordenados, não desordenados --- é o que torna o
\omterm{def:g11:prob:model}{modelo uniforme} aplicável.
\end{example}

\begin{proposition}[Regras da probabilidade]\label{prop:g11:prob:rules}
Para \omterm{def:g11:prob:model}{eventos} $A, B \subseteq \Omega$:
\[
\P(\emptyset) = 0, \qquad \P(\Omega) = 1, \qquad
\P(\bar A) = 1 - \P(A),
\]
\[
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B),
\]
em que $\bar A$ é o \emph{\omterm{def:g10:proba:operations}{complementar}} de $A$ (todos os resultados que
não estão em $A$). Em particular, $\P(A \cup B) = \P(A) + \P(B)$ quando
$A$ e $B$ são \emph{\omterm{def:g10:proba:operations}{incompatíveis}} ($A \cap B = \emptyset$).
\end{proposition}

\begin{proof}
O \omterm{def:g11:prob:model}{evento} vazio não contém resultado algum (soma vazia), e $\Omega$
contém todos eles (soma total $1$). Cada resultado está em exatamente um
entre $A$ e $\bar A$, logo $\P(A) + \P(\bar A) = 1$. Para a \omterm{def:g10:numbers:interunion}{união}: somar
os $p_i$ sobre $A$ e depois sobre $B$ conta duas vezes os resultados de
$A \cap B$; subtrair $\P(A \cap B)$ corrige a contagem dupla.
\end{proof}

\begin{example}
Em uma turma, $60\%$ estudam música ($M$), $45\%$ praticam um esporte
($S$) e $25\%$ fazem as duas coisas. A proporção que faz pelo menos uma é
$\P(M \cup S) = 0.60 + 0.45 - 0.25 = 0.80$, de modo que $20\%$ não fazem
nenhuma (\omterm{def:g10:proba:operations}{complementar}).
\end{example}

\section{Variáveis aleatórias}

\begin{definition}[Variável aleatória, distribuição]\label{def:g11:prob:rv}
Uma \emph{variável aleatória}\index{variável aleatória} em $\Omega$ é uma
\omterm{def:g11:func:function}{função} $X$ que atribui um número a cada resultado. Sua
\emph{distribuição}\index{distribuição} é a tabela de seus valores
possíveis $x_1, \dots, x_k$ junto com suas \omterm{def:g10:proba:distribution}{probabilidades}
\[
\P(X = x_i) = \P\bigl(\{\omega : X(\omega) = x_i\}\bigr),
\qquad
\sum_{i=1}^k \P(X = x_i) = 1 .
\]
\end{definition}

\begin{example}\label{ex:g11:prob:game}
Um jogo: pague $2$ euros, lance um dado honesto e receba o valor da face
se ele for pelo menos $5$, e nada caso contrário. Seja $G$ o
\emph{ganho} (o recebido menos o pago). Os resultados $1, 2, 3, 4$ dão
$G = -2$; o resultado $5$ dá $G = 3$; o resultado $6$ dá $G = 4$. A
\omterm{def:g11:prob:rv}{distribuição} de $G$:
\begin{center}
\begin{tabular}{c|ccc}
$g$ & $-2$ & $3$ & $4$\\
\hline\rule{0pt}{11pt}%
$\P(G = g)$ & $\dfrac46$ & $\dfrac16$ & $\dfrac16$
\end{tabular}
\end{center}
\end{example}

\begin{method}[Montar uma tabela de distribuição]\label{met:g11:prob:table}
Liste os valores possíveis de $X$; para cada valor, reúna os resultados
que o produzem e some suas \omterm{def:g10:proba:distribution}{probabilidades}; confira que a linha das
\omterm{def:g10:proba:distribution}{probabilidades} soma $1$.
\end{method}

\section{Esperança}

\begin{definition}[Esperança]\label{def:g11:prob:expectation}
A \emph{esperança}\index{esperança} (ou \omterm{def:g11:stat:mean}{média}) de $X$ é a \omterm{def:g11:stat:mean}{média} dos seus
valores ponderada pelas \omterm{def:g10:proba:distribution}{probabilidades}:
\[
\E(X) = \sum_{i=1}^{k} \P(X = x_i)\; x_i .
\]
\end{definition}

\begin{remark}
$\E(X)$ é o que seria a \omterm{def:g11:stat:mean}{média} de um número enorme de repetições
independentes do experimento --- a versão precisa dessa afirmação é a lei
dos grandes números, demonstrada no \cref{ch:g12:sums}. Note a analogia
formal com a \omterm{def:g11:stat:mean}{média} de um conjunto de dados com \omterm{def:g10:stats:series}{frequências}
(\cref{def:g11:stat:mean}): as \omterm{def:g10:proba:distribution}{probabilidades} fazem o papel das
\omterm{def:g10:stats:series}{frequências relativas} $\frac{n_i}{n}$.
\end{remark}

\begin{example}\label{ex:g11:prob:gameexp}
Para o jogo do \cref{ex:g11:prob:game}:
\[
\E(G) = \frac46 \times (-2) + \frac16 \times 3 + \frac16 \times 4
= \frac{-8 + 3 + 4}{6} = -\frac16 \approx -0.17 .
\]
Em \omterm{def:g11:stat:mean}{média}, o jogador perde cerca de $17$ centavos por partida: o jogo é
desfavorável. Um jogo é dito \emph{justo} quando o ganho esperado é $0$.
\end{example}

\begin{proposition}[Linearidade]\label{prop:g11:prob:linearity}
Para todos os reais $a, b$:
\[
\E(aX + b) = a\,\E(X) + b .
\]
\end{proposition}

\begin{proof}
A variável $aX + b$ assume o valor $a x_i + b$ com \omterm{def:g10:proba:distribution}{probabilidade}
$\P(X = x_i)$, logo
\[
\E(aX + b) = \sum_i \P(X = x_i)(a x_i + b)
= a \sum_i \P(X = x_i)\,x_i + b \underbrace{\sum_i \P(X =
x_i)}_{=\,1} = a\,\E(X) + b .
\]
\end{proof}

\section{Variância e desvio padrão}

\begin{definition}[Variância]\label{def:g11:prob:variance}
A \emph{variância}\index{variância} de $X$ mede a dispersão de seus
valores em torno da \omterm{def:g11:prob:expectation}{esperança} $m = \E(X)$:
\[
\V(X) = \E\bigl((X - m)^2\bigr)
= \sum_{i=1}^{k} \P(X = x_i)\,(x_i - m)^2,
\qquad
\sigma(X) = \sqrt{\V(X)} .
\]
\end{definition}

\begin{proposition}[Fórmula abreviada]\label{prop:g11:prob:konig}
\[
\V(X) = \E(X^2) - \E(X)^2 .
\]
\end{proposition}

\begin{proof}
Expanda $(x_i - m)^2 = x_i^2 - 2m\,x_i + m^2$ dentro da soma:
\[
\V(X) = \sum_i \P(X = x_i)\,x_i^2
- 2m \sum_i \P(X = x_i)\,x_i + m^2\sum_i \P(X = x_i)
= \E(X^2) - 2m^2 + m^2 ,
\]
que é $\E(X^2) - m^2$. Essa é a versão, para variáveis aleatórias, da
identidade demonstrada para conjuntos de dados na
\cref{prop:g11:stat:shortcut}.
\end{proof}

\begin{proposition}[Transformação afim]\label{prop:g11:prob:affine}
$\V(aX + b) = a^2\,\V(X)$ e $\sigma(aX + b) = \abs a\,\sigma(X)$.
\end{proposition}

\begin{proof}
Por linearidade, $aX + b$ desvia-se de sua \omterm{def:g11:prob:expectation}{esperança} $a m + b$ de
$(aX + b) - (am + b) = a(X - m)$: cada desvio fica multiplicado por $a$,
logo cada quadrado de desvio por $a^2$, e a \omterm{def:g11:stat:mean}{média} ponderada deles também.
O deslocamento $b$ desapareceu --- deslocar uma variável não muda sua
dispersão.
\end{proof}

\begin{example}\label{ex:g11:prob:variance}
Para o jogo do \cref{ex:g11:prob:game}:
$\E(G^2) = \frac46 \times 4 + \frac16 \times 9 + \frac16 \times 16
= \frac{16 + 9 + 16}{6} = \frac{41}{6}$, de modo que
\[
\V(G) = \frac{41}{6} - \left(-\frac16\right)^2
= \frac{41}{6} - \frac{1}{36} = \frac{245}{36} \approx 6.8,
\qquad
\sigma(G) \approx 2.6 .
\]
A perda \omterm{def:g11:stat:mean}{média} de $0.17$ por partida vem acompanhada de oscilações de
tamanho típico $2.6$ euros --- a \omterm{def:g11:prob:expectation}{esperança} sem o \omterm{def:g11:stat:variance}{desvio padrão} conta
apenas metade da história.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:prob:1}
Uma carta é retirada de um baralho comum de $52$ cartas. Calcule a
\omterm{def:g10:proba:distribution}{probabilidade} de retirar: uma carta de copas; um rei; uma carta de copas
ou um rei; nem carta de copas nem rei.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:prob:2}
Dois dados honestos são lançados e sua soma $S$ é registrada. Usando o
modelo de $36$ resultados do \cref{ex:g11:prob:twodice}, calcule
$\P(S = 6)$, $\P(S \geq 10)$ e $\P(S \text{ é par})$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:prob:3}
Uma \omterm{def:g11:prob:rv}{variável aleatória} tem \omterm{def:g11:prob:rv}{distribuição}
\begin{center}
\begin{tabular}{c|cccc}
$x$ & $-1$ & $0$ & $2$ & $5$\\
\hline
$\P(X=x)$ & $0.3$ & $0.2$ & $0.4$ & $p$
\end{tabular}
\end{center}
Encontre $p$ e calcule em seguida $\E(X)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:prob:4}
$X$ satisfaz $\E(X) = 3$ e $\V(X) = 4$. Dê a \omterm{def:g11:prob:expectation}{esperança}, a \omterm{def:g11:prob:variance}{variância} e o
\omterm{def:g11:stat:variance}{desvio padrão} de $Y = 2X - 5$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:5}
Uma roleta de prêmios tem $10$ setores iguais: $5$ mostram $0$ euros, $3$
mostram $2$ euros e $2$ mostram $5$ euros. Jogar custa $2$ euros. Dê a
\omterm{def:g11:prob:rv}{distribuição} do ganho $G$ (prêmio menos aposta), calcule $\E(G)$ e decida
se o jogo é favorável.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:6}
Uma urna contém $3$ bolas vermelhas e $2$ azuis; duas bolas são retiradas
\emph{sem} reposição. Seja $X$ o número de bolas vermelhas retiradas.
Monte a \omterm{def:g11:prob:rv}{distribuição} de $X$ (enumerando as $20$ retiradas ordenadas, ou
usando uma \omterm{met:g10:proba:tree}{árvore}) e calcule $\E(X)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:7}
Para a roleta do \cref{exo:g11:prob:5}, calcule $\V(G)$ e $\sigma(G)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:8}
Uma seguradora vende uma apólice por $80$ euros. Com \omterm{def:g10:proba:distribution}{probabilidade}
$0.05$ o cliente reclama $1000$ euros, com \omterm{def:g10:proba:distribution}{probabilidade} $0.01$ reclama
$5000$ euros, e não reclama nada nos demais casos. Seja $B$ o resultado
da empresa em uma apólice. Dê a \omterm{def:g11:prob:rv}{distribuição} de $B$, calcule $\E(B)$ e
interprete.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:9}
Uma questão de múltipla escolha tem $4$ respostas, uma correta. Uma
resposta certa vale $+1$; para desestimular o chute, uma resposta errada
vale $x < 0$. Um aluno responde ao acaso. Para que valor de $x$ a
pontuação esperada do aluno é nula?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:10}
Duas moedas honestas são lançadas. Ana propõe: ``se as duas moedas
coincidirem, eu lhe pago $1$ euro; se forem diferentes, você me paga $1$
euro''. Calcule o ganho esperado de cada jogador. O jogo é justo? Mesma
pergunta com três moedas, em que Ana paga $2$ euros se as três
coincidirem.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:prob:11}
Uma rifa vende $1000$ bilhetes a $5$ euros cada. Um bilhete ganha
$2000$ euros, cinco bilhetes ganham $200$ euros e vinte bilhetes ganham
$25$ euros.
\begin{enumerate}
  \item Dê a \omterm{def:g11:prob:rv}{distribuição} da arrecadação total menos os prêmios, do ponto
        de vista do organizador, e a do ganho $G$ de um único jogador.
  \item Calcule $\E(G)$ e o lucro total esperado do organizador.
        Verifique que as duas respostas são coerentes.
\end{enumerate}
\end{exercise}

\section{Problema: A partida interrompida}

\begin{problem}[{Problema de fim de semana --- Pascal, Fermat e a
divisão justa de uma aposta interrompida: o nascimento da esperança,
a mágica da linearidade e o que a esperança não enxerga}]
\label{pb:g11:prob:1}
Em 1654, o cavaleiro de Méré --- o mesmo jogador cujas apostas em
dados abriram o problema de \omterm{def:g10:proba:distribution}{probabilidade} do volume do ensino
fundamental --- fez a Blaise Pascal uma pergunta mais profunda:
dois jogadores igualmente hábeis têm de interromper sua série no
placar de $5$ a $3$, sendo que seis vitórias levam o bolo de $64$
pistolas. \emph{Como dividir o bolo de forma justa?} As cartas
trocadas entre Pascal e Fermat para responder a isso fundaram o
conceito que está no coração deste capítulo: a \omterm{def:g11:prob:expectation}{esperança}
(\cref{def:g11:prob:expectation}). Você vai rededuzir a resposta
deles pelos dois caminhos e depois conhecer o truque de festa da
linearidade e o único ponto cego da \omterm{def:g11:prob:expectation}{esperança}.

\medskip
\noindent\textbf{Parte I --- Fluência com \omterm{def:g11:prob:expectation}{esperança}.}
\begin{enumerate}
  \item Lance um dado honesto; seu ganho é (face $- 3$) euros.
        Monte a tabela de \omterm{def:g11:prob:rv}{distribuição} do ganho $G$
        (\cref{met:g11:prob:table}) e calcule $\E(G)$.
  \item Calcule $V(G)$ com a fórmula abreviada
        (\cref{prop:g11:prob:konig}) e $\sigma(G)$ (duas casas
        decimais).
  \item Sem nenhuma tabela nova, dê $\E(2G - 1)$ e $V(2G - 1)$
        (\cref{prop:g11:prob:affine}).
  \item Uma barraca cobra $m$ euros para você lançar dois dados e
        lhe paga a soma em euros. Qual é o preço \emph{justo}
        $m$? A barraca cobra $8$: qual é sua perda esperada por
        partida?
  \item Um notebook de $800$ euros tem $5\,\%$ de chance de ser
        destruído a cada ano. Calcule o prêmio de seguro justo e
        o lucro esperado da seguradora por apólice se ela cobrar
        $60$ euros.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Dividir o bolo.}
Placar $5$--$3$, o primeiro a $6$ leva as $64$ pistolas, e as
rodadas são lançamentos de uma moeda honesta.
\begin{enumerate}[resume]
  \item Duas respostas erradas tentadoras: dividir $5 : 3$
        (proporcionalmente às rodadas ganhas) ou dar tudo a quem
        lidera. Diga em uma frase cada o que há de injusto
        nelas.
  \item O método de Fermat: o jogador A precisa de $1$ vitória a
        mais e B precisa de $3$; imagine jogar exatamente mais
        $3$ rodadas, aconteça o que acontecer. Liste os $8$
        resultados igualmente prováveis, marque quem ganha a
        série em cada um e deduza a divisão justa das $64$
        pistolas.
  \item O ponto sutil que Fermat teve de defender: a partida real
        poderia parar depois de uma rodada --- por que é, ainda
        assim, legítimo raciocinar sobre as $3$ rodadas
        imaginárias?
  \item O método de Pascal, de trás para a frente: calcule a
        parte justa de A nos placares (hipotéticos) $5$--$5$,
        depois $5$--$4$ e depois $5$--$3$, a cada vez tirando a
        \omterm{def:g11:stat:mean}{média} dos dois futuros igualmente prováveis. Compare com
        a questão 7.
  \item Mesma pergunta no placar $4$--$3$: quantas rodadas
        imaginárias, quais resultados dão a série a B e qual é a
        divisão justa?
  \item Enuncie o princípio que os dois métodos partilham --- o
        valor justo de um futuro incerto é a sua
        \emph{\omterm{def:g11:prob:expectation}{esperança}} --- e nomeie duas indústrias modernas
        que funcionam inteiramente sobre ele (a questão 5 mostra
        uma).
  \item Uma verificação de sanidade para qualquer placar:
        explique, com a linearidade da \omterm{def:g11:prob:expectation}{esperança}
        (\cref{prop:g11:prob:linearity}), por que as partes
        justas dos dois jogadores têm sempre de somar exatamente
        o bolo de $64$ pistolas.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O truque de festa da
linearidade.}
\begin{enumerate}[resume]
  \item Reobtenha $\E(\text{soma de dois dados}) = 7$ em uma
        linha com a linearidade
        (\cref{prop:g11:prob:linearity}), sem tabela de $36$
        casas. Para a \emph{\omterm{def:g11:prob:variance}{variância}} da soma, a regra
        $V(X + Y) = V(X) + V(Y)$ exige que os dados sejam
        independentes (admitido): calcule-a.
  \item A festa dos chapéus: $n$ convidados jogam seus chapéus em
        uma pilha e cada um pega um de volta ao acaso. Seja $X$ o
        número de convidados que recuperam o próprio chapéu. Para
        um convidado fixado, qual é a \omterm{def:g10:proba:distribution}{probabilidade} de recuperar
        o próprio chapéu? Somando essas $n$ chances
        (linearidade!), calcule $\E(X)$. A resposta deve
        surpreender: ela não depende de $n$.
  \item Verifique $\E(X) = 1$ na força bruta para $n = 2$ e
        $n = 3$ (liste as $2$ e as $6$ distribuições de chapéus
        igualmente prováveis e tire a \omterm{def:g11:stat:mean}{média} das contagens).
  \item As recuperações dos convidados estão longe de ser
        independentes (se $n - 1$ convidados recuperaram o
        próprio chapéu, o último também tem de recuperar). Em que
        ponto exatamente o cálculo da questão 14 \emph{não}
        exigiu independência? Enuncie o superpoder da
        linearidade.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O que a \omterm{def:g11:prob:expectation}{esperança} não enxerga.}
\begin{enumerate}[resume]
  \item Duas raspadinhas de $5$ euros: a raspadinha A paga $10$
        euros com \omterm{def:g10:proba:distribution}{probabilidade} $\frac12$; a B paga $1\,000$
        euros com \omterm{def:g10:proba:distribution}{probabilidade} $0.005$. Verifique que os dois
        ganhos têm a mesma \omterm{def:g11:prob:expectation}{esperança} e calcule as duas
        \omterm{def:g11:prob:variance}{variâncias}. Que raspadinha as pessoas de fato compram, e
        o que elas estão comprando?
  \item O jogo de São Petersburgo: lance uma moeda até sair a
        primeira coroa; se a primeira coroa sair no lançamento
        $n$, você recebe $2^n$ euros. Calcule a contribuição de
        cada $n$ à \omterm{def:g11:prob:expectation}{esperança} e a própria \omterm{def:g11:prob:expectation}{esperança}. Você pagaria
        $1\,000$ euros para jogar uma única vez? O que esse
        conflito ensina sobre a \omterm{def:g11:prob:expectation}{esperança}?
  \item O seguro visto pelo cliente: pagar $20$ euros ou correr
        um risco de $1\,\%$ de perder $1\,000$? Compare as duas
        \omterm{def:g11:prob:expectation}{esperanças} e explique por que comprar o seguro pode ainda
        assim ser racional (sobre o que uma única família não
        consegue ``tirar \omterm{def:g11:stat:mean}{média}''?).
  \item Final --- a biografia da \omterm{def:g11:prob:expectation}{esperança}, uma frase por
        capítulo: nascida em 1654 para dividir um bolo; seu
        superpoder, a linearidade, que soma \omterm{def:g11:prob:expectation}{esperanças} sem pedir
        licença à independência (os chapéus); seu ponto cego, o
        risco, que a \omterm{def:g11:prob:variance}{variância} mede (raspadinhas, São
        Petersburgo, seguro); e sua coroação no ano que vem,
        quando a lei dos grandes números explicar exatamente por
        que a banca sempre ganha.
\end{enumerate}
\end{problem}
