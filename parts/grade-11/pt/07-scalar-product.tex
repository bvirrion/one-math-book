\chapter{O produto escalar no plano}\label{ch:g11:scal}

O \omterm{def:g11:scal:dot}{produto escalar} associa um \emph{número} a um par de \omterm{def:g11:vect:vector}{vetores}, e esse
número enxerga geometria: ele se anula exatamente em direções
perpendiculares e mede ângulos e comprimentos. É o motor por trás da lei
dos \omterm{def:g11:trigo:cossin}{cossenos}, das \omterm{def:g10:algebra:equation}{equações} de círculos e dos cálculos de
perpendicularidade. A mesma ferramenta, no espaço, é assunto do
\cref{ch:g12:space}.

\section{Definição e primeiros cálculos}

\begin{definition}[Produto escalar]\label{def:g11:scal:dot}
O \emph{produto escalar}\index{produto escalar} de dois \omterm{def:g11:vect:vector}{vetores} $\vec u$
e $\vec v$ é o número
\[
\vec u \cdot \vec v
= \tfrac12\left(\norm{\vec u + \vec v}^2 - \norm{\vec u}^2
- \norm{\vec v}^2\right),
\]
onde $\norm{\vec u}$ denota o comprimento (\emph{norma}\index{norma}) de
$\vec u$.
\end{definition}

\begin{theorem}[As quatro faces do produto escalar]\label{thm:g11:scal:faces}
Sejam $\vec u\,(x, y)$ e $\vec v\,(x', y')$ \omterm{def:g11:vect:vector}{vetores} não nulos e seja
$\theta$ o ângulo entre eles. Então
\begin{enumerate}
  \item \emph{(\omterm{def:g10:coordgeom:system}{coordenadas})} $\vec u \cdot \vec v = xx' + yy'$;
  \item \emph{(\omterm{def:g11:trigo:cossin}{cosseno})} $\vec u \cdot \vec v =
        \norm{\vec u}\,\norm{\vec v}\cos\theta$;
  \item \emph{(projeção)} $\vec u \cdot \vec v = \vec u \cdot \vec v'$,
        onde $\vec v'$ é a projeção \omterm{def:g11:scal:orthogonal}{ortogonal} de $\vec v$ sobre a direção
        de $\vec u$; se $\vec u = \vect{OA}$ e $\vec v = \vect{OB}$,
        então $\vec u \cdot \vec v = \overline{OA} \times \overline{OH}$,
        um produto de comprimentos com sinal, em que $H$ é o pé da
        perpendicular baixada de $B$ à reta $(OA)$;
  \item \emph{(\omterm{def:g11:scal:dot}{normas})} a fórmula que define o produto na
        \cref{def:g11:scal:dot}.
\end{enumerate}
\end{theorem}

\begin{proof}[Demonstração de 1 e 2]
\emph{1.} Em \omterm{def:g10:coordgeom:system}{coordenadas}, $\norm{\vec u}^2 = x^2 + y^2$ (Pitágoras) e
$\vec u + \vec v$ tem \omterm{def:g10:coordgeom:system}{coordenadas} $(x + x', y + y')$, logo
\begin{align*}
2\,\vec u \cdot \vec v
&= (x + x')^2 + (y + y')^2 - (x^2 + y^2) - (x'^2 + y'^2)\\
&= 2xx' + 2yy',
\end{align*}
depois de \omterm{def:g10:algebra:expand}{expandir} os quadrados e cancelar.

\emph{2.} Escolha o \omterm{def:g10:coordgeom:system}{sistema de coordenadas} de modo que $\vec u$ aponte ao
longo do semieixo $x$ positivo: $\vec u\,(\norm{\vec u}, 0)$ e
$\vec v\,(\norm{\vec v}\cos\theta,\ \norm{\vec v}\sin\theta)$
(\cref{ch:g11:trigo}). A fórmula 1 dá então
$\vec u \cdot \vec v = \norm{\vec u}\,\norm{\vec v}\cos\theta + 0$. A
fórmula 3 lê-se no mesmo cálculo: a projeção de $\vec v$ sobre o eixo $x$
tem comprimento com sinal
$\norm{\vec v}\cos\theta = \overline{OH}$.
\end{proof}

\begin{example}\label{ex:g11:scal:compute}
$\vec u\,(3, 1)$ e $\vec v\,(2, -4)$: $\vec u \cdot \vec v = 6 - 4 = 2$.
As \omterm{def:g11:scal:dot}{normas} são $\norm{\vec u} = \sqrt{10}$ e $\norm{\vec v} = \sqrt{20}$,
de modo que
$\cos\theta = \frac{2}{\sqrt{10}\sqrt{20}} = \frac{2}{10\sqrt2}
= \frac{\sqrt2}{10}$: o ângulo é um pouco menor que $82^\circ$.
\end{example}

\begin{proposition}[Regras algébricas]\label{prop:g11:scal:rules}
Para todos os \omterm{def:g11:vect:vector}{vetores} e todo $\lambda \in \R$:
\[
\vec u \cdot \vec v = \vec v \cdot \vec u, \qquad
\vec u \cdot (\vec v + \vec w) = \vec u \cdot \vec v + \vec u \cdot
\vec w, \qquad
(\lambda \vec u)\cdot \vec v = \lambda\,(\vec u \cdot \vec v),
\]
\[
\vec u \cdot \vec u = \norm{\vec u}^2,
\qquad
\norm{\vec u + \vec v}^2 = \norm{\vec u}^2 + 2\,\vec u\cdot\vec v +
\norm{\vec v}^2 .
\]
\end{proposition}

\begin{proof}
Todas seguem da fórmula em \omterm{def:g10:coordgeom:system}{coordenadas}: por exemplo,
$\vec u \cdot (\vec v + \vec w) = x(x' + x'') + y(y' + y'') =
(xx' + yy') + (xx'' + yy'')$. A última identidade expande
$\norm{\vec u + \vec v}^2 = (\vec u + \vec v)\cdot(\vec u + \vec v)$
usando as três primeiras regras --- o \omterm{def:g11:scal:dot}{produto escalar} comporta-se
exatamente como um produto comum, de modo que as identidades algébricas
de sempre valem para \omterm{def:g11:vect:vector}{vetores}.
\end{proof}

\section{Ortogonalidade}

\begin{definition}[Vetores ortogonais]\label{def:g11:scal:orthogonal}
Dois \omterm{def:g11:vect:vector}{vetores} são \emph{ortogonais}\index{vetores ortogonais}, o que se
escreve $\vec u \perp \vec v$, se $\vec u \cdot \vec v = 0$.
\end{definition}

\begin{remark}
Para \omterm{def:g11:vect:vector}{vetores} não nulos, a fórmula 2 do \cref{thm:g11:scal:faces} mostra
que $\vec u \cdot \vec v = 0 \iff \cos\theta = 0 \iff$ as direções são
perpendiculares. O \omterm{def:g11:vect:vector}{vetor} nulo é \omterm{def:g11:scal:orthogonal}{ortogonal} a tudo.
\end{remark}

\begin{example}
$\vec u\,(3, 2)$ e $\vec v\,(-4, 6)$:
$\vec u \cdot \vec v = -12 + 12 = 0$: \omterm{def:g11:scal:orthogonal}{ortogonais}. Em geral,
$\vec u\,(a, b)$ é sempre \omterm{def:g11:scal:orthogonal}{ortogonal} a $\vec v\,(-b, a)$ --- uma rotação
de um quarto de volta.
\end{example}

\section{A lei dos cossenos}

\begin{theorem}[Lei dos cossenos]\label{thm:g11:scal:cosines}
\index{lei dos cossenos}
Em um triângulo $ABC$ de lados $a = BC$, $b = CA$, $c = AB$ e ângulo
$\widehat A$ no vértice $A$:
\[
a^2 = b^2 + c^2 - 2bc\,\cos\widehat A .
\]
\end{theorem}

\begin{proof}
Escreva $\vect{BC} = \vect{BA} + \vect{AC} = \vect{AC} - \vect{AB}$ e
expanda com a \cref{prop:g11:scal:rules}:
\[
a^2 = \norm{\vect{BC}}^2
= \norm{\vect{AC}}^2 - 2\,\vect{AC}\cdot\vect{AB} + \norm{\vect{AB}}^2
= b^2 + c^2 - 2\,\vect{AB}\cdot\vect{AC}.
\]
Pela fórmula do \omterm{def:g11:trigo:cossin}{cosseno},
$\vect{AB}\cdot\vect{AC} = c\,b\,\cos\widehat A$.
\end{proof}

\begin{remark}
Quando $\widehat A$ é reto, $\cos\widehat A = 0$ e a lei dos \omterm{def:g11:trigo:cossin}{cossenos} se
reduz ao teorema de Pitágoras $a^2 = b^2 + c^2$: é Pitágoras com um termo
de correção para ângulos não retos.
\end{remark}

\begin{example}\label{ex:g11:scal:cosines}
Dois lados de um triângulo medem $b = 5$ e $c = 8$ e formam um ângulo
$\widehat A = \frac\pi3$. Então
\[
a^2 = 25 + 64 - 2 \times 5 \times 8 \times \frac12 = 89 - 40 = 49,
\]
logo $a = 7$.
\end{example}

\section{Aplicações: círculos e retas perpendiculares}

\begin{proposition}[Círculo de diâmetro dado]\label{prop:g11:scal:diameter}
Um ponto $M$ está no círculo de diâmetro $[AB]$ se, e somente se,
\[
\vect{MA} \cdot \vect{MB} = 0 .
\]
\end{proposition}

\begin{proof}
Se $M \neq A, B$, a condição diz que o ângulo em $M$ no triângulo $AMB$ é
reto, e os pontos que veem um segmento sob ângulo reto são exatamente os
do círculo que tem esse segmento como diâmetro. Eis uma demonstração
direta: seja $O$ o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$ e $r = \frac{AB}{2}$. Então
$\vect{MA} = \vect{MO} + \vect{OA}$ e
$\vect{MB} = \vect{MO} + \vect{OB} = \vect{MO} - \vect{OA}$, logo
\[
\vect{MA}\cdot\vect{MB}
= \norm{\vect{MO}}^2 - \norm{\vect{OA}}^2 = MO^2 - r^2,
\]
que se anula exatamente quando $MO = r$, isto é, quando $M$ está no
círculo de centro $O$ e raio $r$.
\end{proof}

\begin{proposition}[Vetor normal de uma reta]\label{prop:g11:scal:normal}
O \omterm{def:g11:vect:vector}{vetor} $\vec n\,(a, b)$ é \omterm{def:g11:scal:orthogonal}{ortogonal} a todo \omterm{def:g11:vect:direction}{vetor diretor} da reta
$d : ax + by + c = 0$; diz-se que $\vec n$ é um \emph{vetor
normal}\index{vetor normal} de $d$. Duas retas são perpendiculares
exatamente quando seus \omterm{def:g11:vect:vector}{vetores} normais (ou diretores) são \omterm{def:g11:scal:orthogonal}{ortogonais}.
\end{proposition}

\begin{proof}
Um \omterm{def:g11:vect:direction}{vetor diretor} de $d$ é $\vec u\,(-b, a)$
(\cref{thm:g11:vect:cartesian}), e
$\vec n \cdot \vec u = a(-b) + b(a) = 0$.
\end{proof}

\begin{method}[Usar vetores normais]\label{met:g11:scal:normal}
\begin{itemize}
  \item Reta por $A$ com normal $\vec n\,(a,b)$: escreva
        $\vect{AM} \cdot \vec n = 0$, o que se expande em
        $ax + by + c = 0$, com $c$ determinado por $A$.
  \item Perpendicularidade de $d: ax+by+c = 0$ e $d': a'x+b'y+c' = 0$:
        verifique $aa' + bb' = 0$.
\end{itemize}
\end{method}

\begin{example}\label{ex:g11:scal:perpline}
A reta por $P(2, -1)$ perpendicular a $d : 3x - y + 4 = 0$: um \omterm{def:g11:vect:direction}{vetor
diretor} de $d$ é $(1, 3)$, que serve como \omterm{def:g11:vect:vector}{vetor} \emph{normal} da
perpendicular. \omterm{def:g10:algebra:equation}{Equação}: $x + 3y + c = 0$ com $2 - 3 + c = 0$, logo
$c = 1$: a reta é $x + 3y + 1 = 0$.
\end{example}

\begin{example}[Equação de um círculo]\label{ex:g11:scal:circle}
O círculo de centro $\Omega(1, 2)$ e raio $3$ é o conjunto dos pontos
$M(x,y)$ com $\Omega M^2 = 9$:
\[
(x - 1)^2 + (y - 2)^2 = 9 .
\]
Reciprocamente, $x^2 + y^2 - 2x - 4y - 4 = 0$ reescreve-se, completando
os quadrados (\cref{ch:g11:quad}), como $(x-1)^2 + (y-2)^2 = 9$: o mesmo
círculo.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:scal:1}
Calcule $\vec u \cdot \vec v$ para
\[
\vec u\,(2, 5),\ \vec v\,(3, -1); \qquad
\vec u\,(1, -4),\ \vec v\,(8, 2); \qquad
\norm{\vec u} = 3,\ \norm{\vec v} = 4,\ \theta = \frac{\pi}{3} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:scal:2}
$ABC$ é um triângulo equilátero de lado $6$. Calcule
$\vect{AB} \cdot \vect{AC}$ e $\vect{AB} \cdot \vect{BC}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:scal:3}
Para que valor de $t$ os \omterm{def:g11:vect:vector}{vetores} $\vec u\,(3, t)$ e $\vec v\,(t - 8, 1)$
são \omterm{def:g11:scal:orthogonal}{ortogonais}? E para que valor (ou valores) de $t$ o \omterm{def:g11:vect:vector}{vetor}
$\vec u\,(t, 2)$ é \omterm{def:g11:scal:orthogonal}{ortogonal} a ele mesmo menos $\vec v\,(4, 2)$, isto é,
$\vec u \perp (\vec u - \vec v)$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:scal:4}
Calcule o ângulo $\theta$ entre $\vec u\,(1, 2)$ e $\vec v\,(3, 1)$ (dê
$\cos\theta$ exatamente e depois $\theta$ com precisão de um grau).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:5}
Um triângulo tem lados $b = 4$ e $c = 6$ formando um ângulo
$\widehat A = \frac{2\pi}{3}$. Calcule o terceiro lado $a$. Em seguida,
em um triângulo de lados $a = 7$, $b = 5$, $c = 4$, calcule
$\cos\widehat A$ e decida se $\widehat A$ é agudo ou obtuso.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:6}
Sejam $A(1, 1)$, $B(5, -1)$ e $C(3, 5)$. Calcule
$\vect{AB} \cdot \vect{AC}$; o ângulo $\widehat A$ é reto? Calcule os
comprimentos dos três lados e confira sua resposta com a lei dos
\omterm{def:g11:trigo:cossin}{cossenos}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:7}
Dê uma \omterm{def:g10:algebra:equation}{equação} do círculo de centro $\Omega(-2, 3)$ e raio $5$. Determine
em seguida o centro e o raio do círculo
\[
x^2 + y^2 - 6x + 2y - 6 = 0 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:8}
Sejam $A(-3, 0)$ e $B(0, 2)$. Usando a \cref{prop:g11:scal:diameter},
encontre uma \omterm{def:g10:algebra:equation}{equação} do círculo de diâmetro $[AB]$ e verifique que a
origem $O(0,0)$ está nele.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:9}
Encontre uma \omterm{def:g10:algebra:equation}{equação} da reta por $P(1, 3)$ perpendicular a
$d : 2x + y - 7 = 0$ e as \omterm{def:g10:coordgeom:system}{coordenadas} do pé $H$ da perpendicular (a
\omterm{def:g10:numbers:interunion}{interseção} das duas retas). Deduza a distância de $P$ a $d$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:10}
Usando $\norm{\vec u + \vec v}^2 = \norm{\vec u}^2 + 2\vec u\cdot\vec v +
\norm{\vec v}^2$ e seu análogo para $\vec u - \vec v$, demonstre a
\emph{identidade do paralelogramo}:
\[
\norm{\vec u + \vec v}^2 + \norm{\vec u - \vec v}^2
= 2\norm{\vec u}^2 + 2\norm{\vec v}^2 ,
\]
e interprete-a em um paralelogramo (diagonais contra lados).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:scal:11}
Sejam $A$ e $B$ dois pontos com $AB = 4$, e $I$ o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$.
\begin{enumerate}
  \item Mostre que, para todo ponto $M$,
        $\vect{MA}\cdot\vect{MB} = MI^2 - 4$. (Insira $I$:
        $\vect{MA} = \vect{MI} + \vect{IA}$.)
  \item Deduza o conjunto dos pontos $M$ tais que
        $\vect{MA}\cdot\vect{MB} = 12$.
\end{enumerate}
\end{exercise}

\section{Problema: Dois presentes do produto escalar}

\begin{problem}[{Problema de fim de semana --- as fórmulas de adição
caem de um produto escalar, a distância a uma reta de outro, e as
três alturas se encontram de brinde}]
\label{pb:g11:scal:1}
O \omterm{def:g11:scal:dot}{produto escalar} parece contabilidade --- multiplique, some ---
e, no entanto, entrega quase de graça dois tesouros que
resistiram a séculos de caça a ângulos: as \emph{fórmulas de
adição} da trigonometria (calcule $\cos 15^\circ$ exatamente!) e
a \emph{distância de um ponto a uma reta} em uma fração limpa. De
presente de despedida, ele demonstra que as três alturas de todo
triângulo passam por um mesmo ponto. As quatro faces do
\cref{thm:g11:scal:faces} vão se apresentar ao serviço.

\medskip
\noindent\textbf{Parte I --- Fluência.}
\begin{enumerate}
  \item Para $\vec u(3, 4)$ e $\vec v(-1, 2)$: calcule
        $\vec u \cdot \vec v$, as duas \omterm{def:g11:scal:dot}{normas} e o ângulo entre os
        \omterm{def:g11:vect:vector}{vetores} (com precisão de um décimo de grau).
  \item Encontre $k$ para que $(2, k)$ e $(3, -4)$ sejam
        \omterm{def:g11:scal:orthogonal}{ortogonais} (\cref{def:g11:scal:orthogonal}).
  \item Face física: um trenó é puxado por $10$ m por uma força
        de $50$ N a $60^\circ$ do solo. Calcule o trabalho
        $W = \vec F \cdot \vec d$.
  \item Lei dos \omterm{def:g11:trigo:cossin}{cossenos} (\cref{thm:g11:scal:cosines}): dois
        lados de $5$ e $7$ formam um ângulo de $60^\circ$;
        encontre o terceiro lado exatamente.
  \item Ao contrário: um triângulo tem lados $4$, $6$, $8$.
        Calcule o \omterm{def:g11:trigo:cossin}{cosseno} do ângulo oposto ao lado $8$,
        classifique o ângulo e dê-o com precisão de um décimo de
        grau.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Primeiro presente: as fórmulas de
adição.}
Sejam $\vec u = (\cos a, \sin a)$ e $\vec v = (\cos b, \sin b)$:
dois \omterm{def:g11:vect:vector}{vetores} unitários nos ângulos $a$ e $b$.
\begin{enumerate}[resume]
  \item Calcule $\vec u \cdot \vec v$ duas vezes --- uma com
        \omterm{def:g10:coordgeom:system}{coordenadas}, outra com \omterm{def:g11:scal:dot}{normas} e o ângulo entre os \omterm{def:g11:vect:vector}{vetores}
        --- e conclua a fórmula-mãe:
        \[
        \cos(a - b) = \cos a \cos b + \sin a \sin b .
        \]
  \item Troque $b$ por $-b$ (paridades do \omterm{def:g11:trigo:cossin}{seno} e do \omterm{def:g11:trigo:cossin}{cosseno}!)
        para obter $\cos(a + b)$.
  \item Obtenha $\sin(a + b)$ pela identidade complementar
        $\sin x = \cos\left(\frac\pi2 - x\right)$ aplicada a
        $x = a + b$.
  \item Valores exatos, enfim: calcule $\cos 15^\circ$ (como
        $\cos(45^\circ - 30^\circ)$) e $\sin 75^\circ$. Confira
        os dois numericamente.
  \item Faça $b = a$ para deduzir as fórmulas do arco duplo:
        $\cos 2a$ (nas suas três formas) e $\sin 2a$.
  \item De $\cos 2a = 1 - 2\sin^2 a$, calcule $\sin^2 15^\circ$
        e depois $\sin 15^\circ$ como um radical encaixado --- e
        verifique que seu quadrado coincide com o valor
        $\frac{\sqrt6 - \sqrt2}{4}$ que o método da questão 9
        fornece.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Segundo presente: a distância a
uma reta.}
\begin{enumerate}[resume]
  \item Seja $d$ a reta $ax + by + c = 0$, com \omterm{def:g11:vect:vector}{vetor} normal
        $\vec n(a, b)$ (\cref{prop:g11:scal:normal}), e seja
        $P(x_0, y_0)$ um ponto. Tomando um ponto $A$ qualquer de
        $d$, explique por que a distância de $P$ a $d$ é
        $\dfrac{\abs{\vect{AP} \cdot \vec n}}{\norm{\vec n}}$ e
        deduza a fórmula
        \[
        \operatorname{dist}(P, d)
        = \frac{\abs{a x_0 + b y_0 + c}}{\sqrt{a^2 + b^2}} .
        \]
        Calcule a distância de $P(5, 6)$ a $3x + 4y - 12 = 0$.
  \item Escreva a \omterm{def:g10:algebra:equation}{equação} do círculo de centro $(5, 6)$ \omterm{def:g11:deriv:tangent}{tangente}
        a essa reta.
  \item Contraprova da fórmula: calcule a distância da origem a
        $3x + 4y - 12 = 0$ e verifique-a calculando
        explicitamente o pé da perpendicular baixada da origem.
  \item Triângulo $A(0,0)$, $B(6,0)$, $C(2,5)$: calcule sua área
        usando a base $[AB]$; depois calcule a distância de $B$ à
        reta $(AC)$ e verifique que ela reproduz a mesma área.
  \item O círculo de centro $(2, 1)$ e raio $2$, contra a reta
        $x + y - 6 = 0$: calcule a distância do centro à reta e
        classifique a posição relativa (\omterm{def:g11:deriv:rate}{secante}, \omterm{def:g11:deriv:tangent}{tangente},
        exterior).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O brinde: as alturas concorrem.}
\begin{enumerate}[resume]
  \item Generalize o \cref{exo:g11:scal:11}: para dois pontos
        quaisquer $A$, $B$ de \omterm{prop:g10:coordgeom:midpoint}{ponto médio} $I$, demonstre que
        \[
        \vect{MA} \cdot \vect{MB} = MI^2 - \frac{AB^2}{4}
        \quad\text{para todo ponto } M .
        \]
  \item Deduza a \cref{prop:g11:scal:diameter} em uma linha: o
        conjunto dos pontos $M$ com
        $\vect{MA} \cdot \vect{MB} = 0$ é o círculo de diâmetro
        $[AB]$.
  \item As alturas: seja $H$ a \omterm{def:g10:numbers:interunion}{interseção} das alturas relativas a
        $A$ e a $B$ em um triângulo $ABC$ (de modo que
        $\vect{HA} \cdot \vect{BC} = 0$ e
        $\vect{HB} \cdot \vect{CA} = 0$). Demonstre a identidade
        \[
        \vect{HA} \cdot \vect{BC} + \vect{HB} \cdot \vect{CA}
        + \vect{HC} \cdot \vect{AB} = 0
        \]
        --- expanda tudo a partir de $H$ com Chasles --- e
        conclua que $H$ também está na altura relativa a $C$: as
        três alturas são concorrentes (no
        \emph{ortocentro}\index{ortocentro}).
  \item Final: quatro faces, três troféus. Qual face do
        \cref{thm:g11:scal:faces} carregou as fórmulas de
        adição, qual a fórmula da distância, qual o \omterm{pb:g11:scal:1}{ortocentro}
        --- e por que ``calcular a mesma quantidade duas vezes''
        é o lema do capítulo inteiro?
\end{enumerate}
\end{problem}
