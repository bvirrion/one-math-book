\chapter{Estatística descritiva}\label{ch:g11:stat}

A estatística comprime uma lista de números em alguns resumos
significativos: onde os dados se situam (\omterm{def:g11:stat:mean}{média}, \omterm{def:g11:stat:median}{mediana}) e o quanto eles
estão dispersos (quartis, \omterm{def:g11:stat:variance}{variância}, \omterm{def:g11:stat:variance}{desvio padrão}). Este capítulo faz
tudo sobre um mesmo exemplo condutor. Os mesmos resumos reaparecem para
variáveis aleatórias no \cref{ch:g11:prob}, e suas leis probabilísticas
no \cref{ch:g12:sums}.

Ao longo do capítulo, nosso conjunto de dados condutor são as notas (de
$0$ a $20$) de uma turma de $12$ alunos:
\[
8,\ 9,\ 10,\ 10,\ 11,\ 12,\ 12,\ 12,\ 13,\ 14,\ 16,\ 17 .
\]

\section{Média e mediana}

\begin{definition}[Média]\label{def:g11:stat:mean}
A \emph{média}\index{média} dos valores $x_1, x_2, \dots, x_n$ é
\[
\bar x = \frac{x_1 + x_2 + \dots + x_n}{n}.
\]
Se o valor $x_i$ aparece com \omterm{def:g10:stats:series}{frequência} $n_i$ (com
$n_1 + \dots + n_k = n$), então
$\bar x = \frac{n_1 x_1 + \dots + n_k x_k}{n}$.
\end{definition}

\begin{definition}[Mediana]\label{def:g11:stat:median}
Uma \emph{mediana}\index{mediana} de um conjunto de dados ordenado é um
valor que o divide ao meio: pelo menos metade dos valores é $\leq$ a ela
e pelo menos metade é $\geq$ a ela. Na prática, ordene os $n$ valores; se
$n$ é ímpar, a mediana é o valor central; se $n$ é par, tome o \omterm{prop:g10:coordgeom:midpoint}{ponto
médio} dos dois valores centrais.
\end{definition}

\begin{example}\label{ex:g11:stat:meanmedian}
Para o conjunto condutor, a soma é $144$, logo
$\bar x = \frac{144}{12} = 12$. Os dois valores centrais ($6^{\text{o}}$
e $7^{\text{o}}$ da lista ordenada) são ambos $12$, de modo que a \omterm{def:g11:stat:median}{mediana}
também é $12$. \omterm{def:g11:stat:mean}{Média} e \omterm{def:g11:stat:median}{mediana} não precisam coincidir: a \omterm{def:g11:stat:median}{mediana} ignora
\emph{a que distância} estão os valores extremos, e a \omterm{def:g11:stat:mean}{média} não.
Substituir a nota máxima $17$ por $20$ leva a \omterm{def:g11:stat:mean}{média} a $12.25$, mas deixa
a \omterm{def:g11:stat:median}{mediana} em $12$.
\end{example}

\section{Quartis}

\begin{definition}[Quartis, amplitude interquartil]\label{def:g11:stat:quartiles}
O \emph{primeiro quartil}\index{quartil} $Q_1$ é o menor valor tal que
pelo menos um quarto dos dados seja $\leq Q_1$; o \emph{terceiro
quartil} $Q_3$ é o menor valor tal que pelo menos três quartos dos dados
sejam $\leq Q_3$. A \emph{amplitude interquartil}\index{amplitude
interquartil} é $Q_3 - Q_1$: a largura da metade central dos dados.
\end{definition}

\begin{example}
$n = 12$: um quarto é $3$, de modo que $Q_1$ é o $3^{\text{o}}$ valor
ordenado: $Q_1 = 10$. Três quartos é $9$, de modo que $Q_3$ é o
$9^{\text{o}}$ valor: $Q_3 = 13$. A \omterm{def:g11:stat:quartiles}{amplitude interquartil} é $3$: a
metade central da turma cabe em $3$ pontos de nota.
\end{example}

\section{Variância e desvio padrão}

\begin{definition}[Variância, desvio padrão]\label{def:g11:stat:variance}
A \emph{variância}\index{variância} do conjunto de dados é a \omterm{def:g11:stat:mean}{média} dos
quadrados dos desvios em relação à \omterm{def:g11:stat:mean}{média}:
\[
V = \frac{1}{n}\sum_{i=1}^{n} (x_i - \bar x)^2,
\]
e o \emph{desvio padrão}\index{desvio padrão} é $\sigma = \sqrt{V}$,
expresso na mesma unidade dos dados.
\end{definition}

\begin{proposition}[Fórmula abreviada]\label{prop:g11:stat:shortcut}
A \omterm{def:g11:stat:variance}{variância} é a \omterm{def:g11:stat:mean}{média} dos quadrados menos o quadrado da \omterm{def:g11:stat:mean}{média}:
\[
V = \frac{1}{n}\sum_{i=1}^{n} x_i^2 \;-\; \bar x^{\,2}.
\]
\end{proposition}

\begin{proof}
Expanda cada quadrado de desvio:
$(x_i - \bar x)^2 = x_i^2 - 2\bar x\, x_i + \bar x^2$. Somando e
dividindo por $n$:
\[
V = \frac1n \sum x_i^2 - 2\bar x \cdot \underbrace{\frac1n \sum
x_i}_{=\ \bar x} + \frac1n \cdot n\bar x^2
= \frac1n\sum x_i^2 - 2\bar x^2 + \bar x^2
= \frac1n\sum x_i^2 - \bar x^2 .
\]
A mesma identidade, para variáveis aleatórias, é a
\cref{prop:g11:prob:konig}.
\end{proof}

\begin{example}\label{ex:g11:stat:variance}
Para o conjunto condutor, os desvios em relação a $\bar x = 12$ são $-4$,
$-3$, $-2$, $-2$, $-1$, $0$, $0$, $0$, $1$, $2$, $4$, $5$; seus quadrados
somam
\[
16 + 9 + 4 + 4 + 1 + 0 + 0 + 0 + 1 + 4 + 16 + 25 = 80 .
\]
Logo
\[
V = \frac{80}{12} = \frac{20}{3} \approx 6.67,
\qquad
\sigma = \sqrt{20/3} \approx 2.6 \text{ pontos}.
\]
Grosso modo, uma nota típica fica a cerca de $2.6$ pontos da \omterm{def:g11:stat:mean}{média}.
\end{example}

\begin{proposition}[Mudança de unidades]\label{prop:g11:stat:affine}
Se cada valor é transformado por $y_i = a x_i + b$, então
\[
\bar y = a \bar x + b,
\qquad
V_y = a^2\, V_x,
\qquad
\sigma_y = \abs{a}\,\sigma_x .
\]
\end{proposition}

\begin{proof}
$\bar y = \frac1n \sum (a x_i + b) = a\left(\frac1n\sum x_i\right) +
\frac1n \cdot nb = a\bar x + b$. Em seguida,
$y_i - \bar y = a(x_i - \bar x)$: cada desvio fica multiplicado por $a$
(o deslocamento $b$ se cancela), de modo que cada quadrado de desvio fica
multiplicado por $a^2$, e a \omterm{def:g11:stat:mean}{média} deles também. Extraindo a raiz
quadrada, $\sigma_y = \abs a\,\sigma_x$.
\end{proof}

\begin{example}
Temperaturas registradas em graus Celsius com \omterm{def:g11:stat:mean}{média} $20$ e \omterm{def:g11:stat:variance}{desvio padrão}
$3$ convertem-se para Fahrenheit por $F = 1.8\,C + 32$: a \omterm{def:g11:stat:mean}{média} passa a
$1.8 \times 20 + 32 = 68$ e o \omterm{def:g11:stat:variance}{desvio padrão} a $1.8 \times 3 = 5.4$.
Deslocar um conjunto de dados não muda sua dispersão; reescalá-lo
reescala a dispersão.
\end{example}

\begin{method}[Resumir e comparar conjuntos de dados]\label{met:g11:stat:compare}
Para comparar dois conjuntos de dados, coloque lado a lado: \omterm{def:g11:stat:mean}{média} (ou
\omterm{def:g11:stat:median}{mediana}) para a \emph{posição}, \omterm{def:g11:stat:variance}{desvio padrão} (ou \omterm{def:g11:stat:quartiles}{amplitude interquartil})
para a \emph{dispersão}. O par (\omterm{def:g11:stat:median}{mediana}, \omterm{def:g11:stat:quartiles}{amplitude interquartil}) é
robusto a valores extremos; o par (\omterm{def:g11:stat:mean}{média}, \omterm{def:g11:stat:variance}{desvio padrão}) usa todos os
valores e alimenta a teoria das \omterm{def:g10:proba:distribution}{probabilidades}.
\end{method}

\begin{example}
Dois arqueiros disparam $10$ flechas cada; as pontuações têm \omterm{def:g11:stat:mean}{média} $8.5$
para os dois, mas os \omterm{def:g11:stat:variance}{desvios padrão} são $0.5$ e $2.1$. Mesmo nível em
\omterm{def:g11:stat:mean}{média} --- mas o primeiro arqueiro é muito mais consistente.
\end{example}

\section{Problema: O escore z, uma régua universal}

\begin{problem}[{Problema de fim de semana --- os desvios padrão
comparam laranjas com maçãs, a desigualdade de Chebyshev transforma
$\sigma$ em garantia, e a média ganha sua própria medalha}]
\label{pb:g11:stat:1}
Um $15/20$ numa prova difícil é melhor que um $13/20$ numa prova
fácil? Um $23$ em uma lista de números pequenos é acidente ou
valor atípico? Os números brutos não sabem dizer --- mas,
divididos pelo \omterm{def:g11:stat:variance}{desvio padrão} certo, todos passam a falar uma
única língua: a do \emph{\omterm{pb:g11:stat:1}{escore z}}. Este problema domina a
maquinaria da \omterm{def:g11:stat:variance}{variância} (\cref{def:g11:stat:variance},
\cref{prop:g11:stat:shortcut}), constrói a régua universal e
demonstra a notável desigualdade que transforma $\sigma$ de um
número descritivo em uma garantia.

\medskip
\noindent\textbf{Parte I --- Mecânica da \omterm{def:g11:stat:variance}{variância}.}
\begin{enumerate}
  \item Para o conjunto $4, 8, 6, 5, 2$: calcule a \omterm{def:g11:stat:mean}{média}, depois
        a \omterm{def:g11:stat:variance}{variância} diretamente da definição e depois $\sigma$.
  \item Recalcule a \omterm{def:g11:stat:variance}{variância} com a fórmula abreviada
        (\cref{prop:g11:stat:shortcut}) --- \omterm{def:g11:stat:mean}{média} dos quadrados
        menos quadrado da \omterm{def:g11:stat:mean}{média} --- e confira a concordância.
  \item Com \omterm{def:g10:stats:series}{frequências}: notas $8$, $10$, $14$ obtidas por $3$,
        $5$, $2$ alunos. Calcule a \omterm{def:g11:stat:mean}{média}, a \omterm{def:g11:stat:variance}{variância} e $\sigma$
        (duas casas decimais).
  \item Temperaturas têm \omterm{def:g11:stat:mean}{média} $20$ e $\sigma = 3$ (em graus
        Celsius). Dê a \omterm{def:g11:stat:mean}{média} e $\sigma$ em Fahrenheit
        ($F = 1.8\,C + 32$; \cref{prop:g11:stat:affine}).
  \item Demonstre as duas regras que acabou de usar: somar uma
        constante $b$ a todos os valores desloca a \omterm{def:g11:stat:mean}{média} de $b$ e
        deixa $\sigma$ inalterado; multiplicar por $a$ multiplica
        a \omterm{def:g11:stat:mean}{média} por $a$ e $\sigma$ por $\abs a$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A régua universal.}
O \emph{escore z}\index{escore z} de um valor $x$ em
um conjunto de \omterm{def:g11:stat:mean}{média} $\bar x$ e \omterm{def:g11:stat:variance}{desvio padrão} $\sigma$ é
$z = \dfrac{x - \bar x}{\sigma}$: o número de \omterm{def:g11:stat:variance}{desvios padrão} que
separam $x$ da \omterm{def:g11:stat:mean}{média}.
\begin{enumerate}[resume]
  \item A prova A tem \omterm{def:g11:stat:mean}{média} $12$ e $\sigma = 2$; a prova B tem
        \omterm{def:g11:stat:mean}{média} $8$ e $\sigma = 4$. Alice tirou $15$ em A e Bruno
        $13$ em B. Calcule os dois \omterm{pb:g11:stat:1}{escores z}: qual desempenho
        se destaca mais em relação ao grupo?
  \item Calcule os cinco \omterm{pb:g11:stat:1}{escores z} do conjunto da questão 1 e
        verifique que eles têm \omterm{def:g11:stat:mean}{média} $0$ e \omterm{def:g11:stat:variance}{desvio padrão} $1$.
        Explique, a partir da questão 5, por que a padronização
        \emph{sempre} produz \omterm{def:g11:stat:mean}{média} $0$ e $\sigma = 1$.
  \item Um teste de aptidão tem \omterm{def:g11:stat:mean}{média} $100$ e $\sigma = 15$. Que
        pontuação bruta corresponde a $z = 2$? Qual é o \omterm{pb:g11:stat:1}{escore
        z} de $76$?
  \item Um pediatra registra o peso de um bebê como
        ``$z = -2.5$''. Diga o que isso significa e por que o
        crescimento é acompanhado em \omterm{pb:g11:stat:1}{escores z}, e não em
        quilogramas, à medida que a criança envelhece.
  \item O valor atípico do \cref{exo:g11:stat:10} (conjunto
        $1, 2, 2, 3, 3, 3, 4, 4, 5, 23$): calcule a \omterm{def:g11:stat:mean}{média},
        $\sigma$ e o \omterm{pb:g11:stat:1}{escore z} de $23$. O limiar clássico de
        alarme é $\abs z \geq 3$: qual o veredicto?
  \item Recalcule $\sigma$ \emph{sem} o valor $23$ e o \omterm{pb:g11:stat:1}{escore z}
        que $23$ teria contra a \omterm{def:g11:stat:mean}{média} e o $\sigma$ dos dados
        limpos. O que o valor atípico fez com a própria régua
        destinada a medi-lo, e o que o \cref{exo:g11:stat:10}
        sugere como contramedida?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A garantia de Chebyshev.}
\begin{enumerate}[resume]
  \item Demonstre a desigualdade de Bienaymé--Chebyshev para um
        conjunto de dados: se uma proporção $p$ dos valores
        satisfaz $\abs{x - \bar x} \geq k\sigma$, então,
        mantendo apenas esses termos na soma da \omterm{def:g11:stat:variance}{variância},
        \[
        \sigma^2 \geq p \cdot (k\sigma)^2,
        \qquad\text{logo}\qquad
        p \leq \frac{1}{k^2} .
        \]
  \item O que a desigualdade garante sobre a proporção de
        qualquer conjunto de dados além de $2\sigma$ da sua
        \omterm{def:g11:stat:mean}{média}? E além de $3\sigma$? Verifique a afirmação sobre
        $2\sigma$ no conjunto da questão 1.
  \item Uma fábrica produz parafusos de \omterm{def:g11:stat:mean}{média} $50$ mm e
        $\sigma = 0.1$ mm. O que Chebyshev garante sobre a
        proporção de parafusos fora de $\intcc{49.7}{50.3}$? (A
        produção real se sai muito melhor --- a curva em sino do
        ano 12 transforma esse cauteloso $\frac19$ em uma fração
        de por cento.)
  \item Dois fundos rendem em \omterm{def:g11:stat:mean}{média} $5\,\%$ ao ano, com
        $\sigma = 2\,\%$ para o fundo A e $\sigma = 12\,\%$ para
        o fundo B. Para cada fundo, o que Chebyshev diz sobre a
        proporção de anos com rendimento abaixo de $-19\,\%$?
        Qual é o nome financeiro de $\sigma$?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A medalha da \omterm{def:g11:stat:mean}{média}, e um aviso.}
\begin{enumerate}[resume]
  \item Seja $f(c)$ a \omterm{def:g11:stat:mean}{média} dos quadrados dos desvios
        $(x_i - c)^2$ em relação a um centro arbitrário $c$.
        Demonstre a identidade
        \[
        f(c) = \sigma^2 + (\bar x - c)^2 ,
        \]
        e conclua: a \omterm{def:g11:stat:mean}{média} é o único ponto que minimiza o
        desvio quadrático médio --- sua medalha, correspondente à
        minimização dos desvios absolutos pela \omterm{def:g11:stat:median}{mediana} no
        \cref{pb:g10:stats:1}.
  \item Verifique a identidade no conjunto da questão 1 com
        $c = 6$: calcule $f(6)$ diretamente e pela fórmula.
  \item Mesmos resumos, dados diferentes: verifique que
        $P = \{3, 5, 7, 9, 11\}$ e $Q = \{1, 7, 7, 7, 13\}$ têm a
        mesma \omterm{def:g11:stat:mean}{média}, mas não a mesma \omterm{def:g11:stat:variance}{variância}; construa em
        seguida um conjunto da forma $\{7 - a, 7, 7, 7, 7 + a\}$
        com exatamente a \omterm{def:g11:stat:mean}{média} e a \omterm{def:g11:stat:variance}{variância} de $P$ --- e ainda
        assim com uma forma toda sua.
  \item Demonstre que $\sigma = 0$ obriga todos os valores a
        serem iguais. (O que tem de ser cada termo de uma soma
        nula de quadrados?)
  \item Final --- a escada do estatístico, uma linha por degrau:
        centros (\omterm{def:g11:stat:mean}{média}, \omterm{def:g11:stat:median}{mediana}), dispersões ($\sigma$,
        \omterm{def:g11:stat:quartiles}{amplitude interquartil}), a régua universal ($z$), a
        garantia (Chebyshev) e o aviso da questão 18: os resumos
        comprimem, e comprimir perde --- com a curva em sino do
        ano 12 à espera para transformar \omterm{pb:g11:stat:1}{escores z} em
        \omterm{def:g10:proba:distribution}{probabilidades} exatas.
\end{enumerate}
\end{problem}
