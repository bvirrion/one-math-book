\chapter{Funções e equações do segundo grau}\label{ch:g11:quad}

As \omterm{def:g11:quad:quadratic}{funções quadráticas} são as mais simples depois das lineares, e as
primeiras cujos \omterm{def:g10:functions:graph}{gráficos} são de fato curvos. Este capítulo desenvolve a
caixa de ferramentas completa para elas: a \omterm{thm:g11:quad:canonical}{forma canônica}, o
\omterm{def:g11:quad:discriminant}{discriminante}, o sinal de uma expressão do segundo grau e a geometria da
\omterm{def:g11:quad:parabola}{parábola}.

\section{A forma canônica}

\begin{definition}[Função quadrática]\label{def:g11:quad:quadratic}
Uma \emph{função quadrática}\index{função quadrática} é uma \omterm{def:g10:functions:function}{função} da
forma
\[
f(x) = ax^2 + bx + c, \qquad a \neq 0,
\]
onde $a$, $b$, $c$ são \omterm{def:g10:numbers:sets}{números reais}. A expressão $ax^2 + bx + c$ também
é chamada de \emph{trinômio do segundo grau}.
\end{definition}

\begin{theorem}[Forma canônica]\label{thm:g11:quad:canonical}
Toda \omterm{def:g11:quad:quadratic}{função quadrática} pode ser escrita de exatamente uma maneira como
\[
f(x) = a(x - \alpha)^2 + \beta,
\qquad\text{onde } \alpha = -\frac{b}{2a} \text{ e } \beta = f(\alpha).
\]
Essa é a \emph{forma canônica}\index{forma canônica} de $f$.
\end{theorem}

\begin{proof}
Vamos \emph{completar o quadrado}. Como $a \neq 0$, coloque-o em
evidência nos dois primeiros termos:
\[
ax^2 + bx + c = a\left(x^2 + \frac{b}{a}x\right) + c .
\]
Ora, $x^2 + \frac{b}{a}x$ é o começo do quadrado
$\left(x + \frac{b}{2a}\right)^2 = x^2 + \frac ba x + \frac{b^2}{4a^2}$,
de modo que $x^2 + \frac{b}{a}x = \left(x + \frac{b}{2a}\right)^2 -
\frac{b^2}{4a^2}$. Substituindo:
\[
f(x) = a\left(x + \frac{b}{2a}\right)^2 - \frac{b^2}{4a} + c
= a\bigl(x - \alpha\bigr)^2 + \beta,
\]
com $\alpha = -\frac{b}{2a}$ e $\beta = c - \frac{b^2}{4a}$. Calcular em
$x = \alpha$ anula o termo ao quadrado, logo $\beta = f(\alpha)$.
Unicidade: \omterm{def:g10:algebra:expand}{expandir} $a(x-\alpha)^2 + \beta$ e identificar os coeficientes
de $x$ obriga $\alpha = -\frac{b}{2a}$, e então $\beta$ fica determinado.
\end{proof}

\begin{example}\label{ex:g11:quad:canonical}
Seja $f(x) = 2x^2 - 8x + 3$. Então $\alpha = -\frac{-8}{2 \times 2} = 2$
e $\beta = f(2) = 8 - 16 + 3 = -5$, logo
\[
f(x) = 2(x - 2)^2 - 5 .
\]
Confira expandindo: $2(x^2 - 4x + 4) - 5 = 2x^2 - 8x + 8 - 5 =
2x^2 - 8x + 3$.
\end{example}

\begin{proposition}[Extremo e variação]\label{prop:g11:quad:variations}
Seja $f(x) = a(x-\alpha)^2 + \beta$.
\begin{enumerate}
  \item Se $a > 0$, então $f$ é \omterm{def:g10:functions:variations}{decrescente} em $\intoc{-\infty}{\alpha}$ e
        \omterm{def:g10:functions:variations}{crescente} em $\intco{\alpha}{+\infty}$: $f$ tem um
        \emph{\omterm{def:g10:functions:extrema}{mínimo}} $\beta$, atingido em $x = \alpha$.
  \item Se $a < 0$, a variação se inverte: $f$ tem um \emph{\omterm{def:g10:functions:extrema}{máximo}}
        $\beta$ em $x = \alpha$.
\end{enumerate}
\end{proposition}

\begin{proof}
Suponha $a > 0$ e sejam $\alpha \leq u < v$. Então
\[
f(v) - f(u) = a\bigl((v-\alpha)^2 - (u-\alpha)^2\bigr)
= a\,(v - u)\,\bigl((v-\alpha) + (u-\alpha)\bigr).
\]
Cada fator é positivo: $a > 0$, $v - u > 0$ e
$(v-\alpha) + (u-\alpha) > 0$, pois $v > \alpha$ e $u \geq \alpha$. Logo
$f(v) > f(u)$: $f$ é \omterm{def:g10:functions:variations}{crescente} em $\intco{\alpha}{+\infty}$. Para
$u < v \leq \alpha$ o último fator é negativo, de modo que
$f(v) < f(u)$: $f$ é \omterm{def:g10:functions:variations}{decrescente} ali. Por fim,
$f(x) - f(\alpha) = a(x-\alpha)^2 \geq 0$ para todo $x$, logo
$\beta = f(\alpha)$ é o \omterm{def:g10:functions:extrema}{mínimo}. O caso $a < 0$ segue aplicando o que
precede a $-f$.
\end{proof}

\begin{example}
Um fazendeiro tem $100$ metros de cerca para delimitar um terreno
retangular encostado em um muro reto (não é preciso cercar ao longo do
muro). Se $x$ é a largura, a área cercada é
$A(x) = x(100 - 2x) = -2x^2 + 100x$. Aqui
$\alpha = -\frac{100}{2\times(-2)} = 25$ e
$A(25) = 25 \times 50 = 1250$: a área é máxima para uma largura de
$25$ m, dando $1250$ m$^2$.
\end{example}

\section{Raízes e fatoração}

\begin{definition}[Raiz, discriminante]\label{def:g11:quad:discriminant}
Uma \emph{raiz}\index{raiz} do trinômio $ax^2+bx+c$ é uma solução da
\omterm{def:g10:algebra:equation}{equação} $ax^2 + bx + c = 0$. O
\emph{discriminante}\index{discriminante} do trinômio é o número
\[
\Delta = b^2 - 4ac .
\]
\end{definition}

\begin{theorem}[Resolução de $ax^2+bx+c=0$]\label{thm:g11:quad:roots}
\begin{enumerate}
  \item Se $\Delta > 0$, a \omterm{def:g10:algebra:equation}{equação} tem duas raízes distintas
        \[
        x_1 = \frac{-b - \sqrt{\Delta}}{2a},
        \qquad
        x_2 = \frac{-b + \sqrt{\Delta}}{2a},
        \]
        e $ax^2+bx+c = a(x - x_1)(x - x_2)$.
  \item Se $\Delta = 0$, a \omterm{def:g10:algebra:equation}{equação} tem a única \omterm{def:g11:quad:discriminant}{raiz}
        $x_0 = -\frac{b}{2a}$, e $ax^2+bx+c = a(x - x_0)^2$.
  \item Se $\Delta < 0$, a \omterm{def:g10:algebra:equation}{equação} não tem solução real, e o trinômio não
        pode ser fatorado com coeficientes reais.
\end{enumerate}
\end{theorem}

\begin{proof}
Parta da \omterm{thm:g11:quad:canonical}{forma canônica} obtida no \cref{thm:g11:quad:canonical}:
\[
ax^2+bx+c
= a\left(x + \frac{b}{2a}\right)^2 + c - \frac{b^2}{4a}
= a\left[\left(x + \frac{b}{2a}\right)^2 - \frac{\Delta}{4a^2}\right],
\]
pois $c - \frac{b^2}{4a} = \frac{4ac - b^2}{4a} = -\frac{\Delta}{4a} =
a \times \left(-\frac{\Delta}{4a^2}\right)$.

\emph{1.} Se $\Delta > 0$, então $\frac{\Delta}{4a^2} =
\left(\frac{\sqrt\Delta}{2a}\right)^2$ e o colchete é uma diferença de
dois quadrados:
\[
ax^2+bx+c = a\left(x + \frac{b}{2a} - \frac{\sqrt\Delta}{2a}\right)
\left(x + \frac{b}{2a} + \frac{\sqrt\Delta}{2a}\right)
= a(x - x_2)(x - x_1).
\]
Um produto é nulo exatamente quando um de seus fatores é nulo, o que dá
as duas raízes, distintas porque $\sqrt\Delta \neq 0$.

\emph{2.} Se $\Delta = 0$, o colchete é
$\left(x + \frac{b}{2a}\right)^2$, que se anula apenas em
$x_0 = -\frac{b}{2a}$.

\emph{3.} Se $\Delta < 0$, então $-\frac{\Delta}{4a^2} > 0$, de modo que
o colchete é um quadrado somado a um número positivo: ele nunca se anula,
e a \omterm{def:g10:algebra:equation}{equação} não tem solução real. Uma fatoração $a(x-x_1)(x-x_2)$
produziria raízes, logo não existe nenhuma.
\end{proof}

\begin{example}\label{ex:g11:quad:solving}
Resolva $2x^2 - 8x + 3 = 0$. Aqui
$\Delta = (-8)^2 - 4 \times 2 \times 3 = 64 - 24 = 40 > 0$, de modo que
há duas raízes:
\[
x_{1,2} = \frac{8 \pm \sqrt{40}}{4} = \frac{8 \pm 2\sqrt{10}}{4}
= 2 \pm \frac{\sqrt{10}}{2}.
\]
Para $x^2 + x + 1 = 0$: $\Delta = 1 - 4 = -3 < 0$, nenhuma solução real.
\end{example}

\begin{proposition}[Soma e produto das raízes]\label{prop:g11:quad:vieta}
Se $\Delta \geq 0$ e $x_1, x_2$ são as raízes (iguais quando
$\Delta = 0$), então
\[
x_1 + x_2 = -\frac{b}{a},
\qquad
x_1 x_2 = \frac{c}{a} .
\]
Reciprocamente, dois números de soma $s$ e produto $p$ são as soluções de
$x^2 - sx + p = 0$.
\end{proposition}

\begin{proof}
Expanda a fatoração do \cref{thm:g11:quad:roots}:
\[
a(x - x_1)(x - x_2) = a x^2 - a(x_1 + x_2)x + a\,x_1 x_2 .
\]
Identificar os coeficientes com $ax^2 + bx + c$ dá $-a(x_1+x_2) = b$ e
$a\,x_1x_2 = c$. Para a recíproca, se $u + v = s$ e $uv = p$, então
$(x-u)(x-v) = x^2 - sx + p$, de modo que $u$ e $v$ são suas raízes.
\end{proof}

\begin{example}
Encontre dois números de soma $10$ e produto $21$. Eles resolvem
$x^2 - 10x + 21 = 0$; $\Delta = 100 - 84 = 16$, raízes
$\frac{10 \pm 4}{2} = 7$ e $3$. \emph{\omterm{def:g11:quad:discriminant}{Raiz} à vista:} como os
coeficientes de $x^2 - 5x + 4$ satisfazem $1 - 5 + 4 = 0$, o número $1$ é
\omterm{def:g11:quad:discriminant}{raiz}, e a outra é $\frac{c}{a} = 4$ (o produto delas tem de ser
$\frac ca$).
\end{example}

\section{Sinal de um trinômio}

\begin{theorem}[Sinal de um trinômio]\label{thm:g11:quad:sign}
Seja $f(x) = ax^2+bx+c$ com $a \neq 0$.
\begin{enumerate}
  \item Se $\Delta > 0$, com raízes $x_1 < x_2$: $f(x)$ tem o sinal de $a$
        fora de $\intcc{x_1}{x_2}$ e o sinal de $-a$ em
        $\intoo{x_1}{x_2}$.
  \item Se $\Delta = 0$: $f(x)$ tem o sinal de $a$ para todo
        $x \neq x_0$, e anula-se em $x_0$.
  \item Se $\Delta < 0$: $f(x)$ tem o sinal de $a$ para todo $x$.
\end{enumerate}
Em resumo: \emph{um trinômio tem o sinal de $a$, exceto entre suas
raízes}.
\end{theorem}

\begin{proof}
\emph{1.} Escreva $f(x) = a(x-x_1)(x-x_2)$ e acompanhe os sinais dos
fatores. Se $x < x_1$: tanto $x - x_1$ quanto $x - x_2$ são negativos,
seu produto é positivo, logo $f(x)$ tem o sinal de $a$. Se
$x_1 < x < x_2$: os fatores têm sinais contrários, o produto é negativo e
$f(x)$ tem o sinal de $-a$. Se $x > x_2$: os dois fatores são positivos e
$f(x)$ tem novamente o sinal de $a$.

\emph{2.} $f(x) = a(x - x_0)^2$ e $(x-x_0)^2 > 0$ para $x \neq x_0$.

\emph{3.} Pela demonstração do \cref{thm:g11:quad:roots},
$f(x) = a\left[\left(x + \frac{b}{2a}\right)^2 +
\left(-\frac{\Delta}{4a^2}\right)\right]$, e o colchete é sempre
positivo.
\end{proof}

\begin{method}[Resolver inequações do segundo grau]\label{met:g11:quad:inequalities}
Para resolver $ax^2 + bx + c \leq 0$ (ou $\geq$, $<$, $>$):
\begin{enumerate}
  \item calcule $\Delta$ e as raízes, se houver;
  \item monte o \omterm{def:g10:algebra:signtable}{quadro de sinais} usando o \cref{thm:g11:quad:sign} (sinal
        de $a$ fora das raízes);
  \item leia o \omterm{def:g10:numbers:interval}{intervalo} (ou os \omterm{def:g10:numbers:interval}{intervalos}) solução, atento a se as
        próprias raízes estão incluídas (desigualdade ampla) ou não
        (estrita).
\end{enumerate}
\end{method}

\begin{example}\label{ex:g11:quad:inequality}
Resolva $-x^2 + 3x + 4 \geq 0$. Aqui $\Delta = 9 + 16 = 25$, e as raízes
são $\frac{-3 \pm 5}{-2}$, isto é, $x_1 = -1$ e $x_2 = 4$. Como
$a = -1 < 0$, o trinômio é negativo fora das raízes e positivo entre
elas. Com a desigualdade ampla, o conjunto solução é $\intcc{-1}{4}$.
\end{example}

\section{A parábola}

\begin{definition}[Parábola]\label{def:g11:quad:parabola}
O \omterm{def:g10:functions:graph}{gráfico} de uma \omterm{def:g11:quad:quadratic}{função quadrática} $f(x) = a(x-\alpha)^2+\beta$ é uma
\emph{parábola}\index{parábola} de \emph{vértice} $S(\alpha, \beta)$. Ela
tem a concavidade voltada para cima se $a > 0$ e para baixo se $a < 0$, e
é simétrica em relação à reta vertical $x = \alpha$.
\end{definition}

\begin{remark}
A simetria é fácil de verificar: para qualquer $h$,
$f(\alpha + h) = ah^2 + \beta = f(\alpha - h)$, de modo que dois pontos a
igual distância horizontal da reta $x = \alpha$ têm a mesma altura.
\end{remark}

\begin{method}[Ler uma parábola]\label{met:g11:quad:reading}
Para esboçar $f(x) = ax^2+bx+c$: determine a concavidade (sinal de $a$),
o vértice $\left(-\frac{b}{2a},\, f\!\left(-\frac{b}{2a}\right)\right)$,
a \omterm{def:g10:numbers:interunion}{interseção} $(0, c)$ com o eixo $y$ e as raízes, se houver. O eixo de
simetria $x = -\frac{b}{2a}$ coloca então cada ponto calculado duas
vezes.
\end{method}

\begin{example}
Para $f(x) = x^2 - 4x + 3$: concavidade para cima, vértice $(2, -1)$,
corta o eixo $y$ em $(0,3)$, raízes $1$ e $3$ (vistas de imediato:
$1 - 4 + 3 = 0$). Por simetria, o ponto $(4, 3)$ também está na curva,
espelhando $(0, 3)$.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:quad:1}
Resolva as \omterm{def:g10:algebra:equation}{equações}
\[
x^2 - 5x + 6 = 0, \qquad
2x^2 + 3x - 2 = 0, \qquad
x^2 + 2x + 5 = 0, \qquad
9x^2 - 6x + 1 = 0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:quad:2}
Escreva na \omterm{thm:g11:quad:canonical}{forma canônica} e dê o vértice da \omterm{def:g11:quad:parabola}{parábola}:

f(x) = x^2 - 6x + 11, \qquad
g(x) = -2x^2 + 4x + 1, \qquad

\begin{exercise}[$\star$]\label{exo:g11:quad:3}
Resolva as inequações

x^2 - 3x - 10 < 0, \qquad
2x^2 + x + 3 > 0, \qquad
-x^2 + 6x - 9 \geq 0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:quad:4}
Sem calcular $\Delta$, encontre uma \omterm{def:g11:quad:discriminant}{raiz} evidente e depois a outra:
\[
x^2 - 7x + 6 = 0, \qquad
2x^2 + 5x - 7 = 0, \qquad
x^2 + 4x - 5 = 0 .
\]
(Sugestão: teste $x = 1$ e $x = -1$ e use em seguida o produto das
raízes.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:5}
Encontre dois números cuja soma seja $14$ e cujo produto seja $45$.
Encontre em seguida as dimensões de um retângulo de perímetro $34$ e área
$60$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:6}
Seja $f(x) = x^2 + (m-2)x + 1$, onde $m$ é um parâmetro real. Para quais
valores de $m$ a \omterm{def:g10:algebra:equation}{equação} $f(x) = 0$ tem exatamente uma solução? Nenhuma
solução?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:7}
Uma bola é lançada para cima; sua altura após $t$ segundos é
$h(t) = -5t^2 + 20t + 1$ (em metros). Qual é a altura máxima atingida, e
em que instante? Durante qual \omterm{def:g10:numbers:interval}{intervalo} de tempo a bola fica a pelo menos
$16$ m de altura?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:8}
Resolva a \omterm{def:g10:algebra:equation}{equação} $x^4 - 13x^2 + 36 = 0$ fazendo $X = x^2$ (uma \omterm{def:g10:algebra:equation}{equação}
\emph{biquadrada}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:9}
Para quais valores de $x$ o ponto $M(x, x^2)$ da \omterm{def:g11:quad:parabola}{parábola} $y = x^2$ está
estritamente abaixo da reta $y = x + 2$? Interprete em um esboço.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:10}
A soma de um número positivo com o seu inverso é $\frac{13}{6}$. Encontre
o número. (Reduza a uma \omterm{def:g10:algebra:equation}{equação} do segundo grau.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:quad:11}
Seja $\mathcal P$ a \omterm{def:g11:quad:parabola}{parábola} $y = x^2$ e seja $d_m$ a reta $y = mx - 1$,
onde $m$ é um parâmetro real.
\begin{enumerate}
  \item Para quais valores de $m$ as curvas $\mathcal P$ e $d_m$ se

  \item Quando se tocam em exatamente um ponto, calcule o ponto de

        tangente a $\mathcal P$ nesse ponto.)
\end{enumerate}
\end{exercise}

\section{Problema: O ponto secreto da parábola}

\begin{problem}[{Problema de fim de semana --- foco e diretriz: por
que antenas parabólicas, faróis e pontes pênseis desenham todos a
mesma curva, da qual existe essencialmente só uma}]
\label{pb:g11:quad:1}
Toda antena parabólica de toda varanda aponta seu receptor para um
ponto interior preciso; todo farol esconde sua lâmpada no mesmo
lugar especial. Esse ponto --- o \emph{\omterm{pb:g11:quad:1}{foco}} --- é um segredo que
a \omterm{def:g11:quad:parabola}{parábola} guarda desde a geometria grega, e a fórmula da
distância basta para extraí-lo. Pelo caminho, este problema treina
o \omterm{def:g11:quad:discriminant}{discriminante}, lança projéteis por cima de muros, suspende uma
ponte e demonstra um fato final espantoso: a menos de ampliação,
\emph{só existe uma \omterm{def:g11:quad:parabola}{parábola} no mundo}.

\medskip
\noindent\textbf{Parte I --- Fluência com trinômios.}
\begin{enumerate}
  \item Resolva $x^2 - 5x + 6 = 0$, depois $2x^2 + x - 6 = 0$, e
        depois $x^2 + x + 1 = 0$ (\cref{thm:g11:quad:roots}).
  \item Escreva $2x^2 - 4x - 6$ na forma fatorada e na \omterm{thm:g11:quad:canonical}{forma
        canônica} (\cref{thm:g11:quad:canonical}). Junto com a
        forma expandida, você tem agora três trajes de uma mesma
        \omterm{def:g10:functions:function}{função}: diga qual característica cada um exibe à primeira
        vista (raízes, vértice, \omterm{def:g10:numbers:interunion}{interseção} com o eixo $y$).
  \item Resolva $-x^2 + 4x - 3 \geq 0$ com a regra dos sinais
        (\cref{thm:g11:quad:sign}).
  \item Viète como detetive (\cref{prop:g11:quad:vieta},
        continuando a Babilônia do \cref{pb:g10:algebra:1}):
        encontre dois números de soma $7$ e produto $12$
        escrevendo a \omterm{def:g10:algebra:equation}{equação} deles. Depois: uma \omterm{def:g11:quad:discriminant}{raiz} de
        $x^2 - 5x + c = 0$ vale $2$; encontre $c$ e a outra \omterm{def:g11:quad:discriminant}{raiz}
        sem resolver nada.
  \item Para quais valores de $m$ a \omterm{def:g10:algebra:equation}{equação} $x^2 + mx + 9 = 0$
        tem \omterm{def:g11:quad:discriminant}{raiz} dupla? Nenhuma \omterm{def:g11:quad:discriminant}{raiz}? Duas raízes?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O \omterm{pb:g11:quad:1}{foco} revelado.}
Trabalhe com a \omterm{def:g11:quad:parabola}{parábola} $y = x^2$, o ponto
$F\left(0, \frac14\right)$ e a reta horizontal $\delta$ de \omterm{def:g10:algebra:equation}{equação}
$y = -\frac14$.
\begin{enumerate}[resume]
  \item Para um ponto $P(x, x^2)$ da \omterm{def:g11:quad:parabola}{parábola}, expanda
        $PF^2 = x^2 + \left(x^2 - \frac14\right)^2$ e mostre que
        ele é o quadrado perfeito
        $\left(x^2 + \frac14\right)^2$. Deduza $PF$.
  \item Calcule a distância de $P$ à reta $\delta$ e conclua:
        \emph{todo ponto da \omterm{def:g11:quad:parabola}{parábola} está exatamente tão longe de
        $F$ quanto de $\delta$}. ($F$ é o
        \emph{foco}\index{foco}, e $\delta$ a
        \emph{diretriz}\index{diretriz}.)
  \item Demonstre a recíproca: um ponto $P(x, y)$ com
        $PF = \operatorname{dist}(P, \delta)$ tem
        necessariamente $y = x^2$. (Eleve as duas distâncias ao
        quadrado e simplifique.) A \omterm{def:g11:quad:parabola}{parábola} é, assim, um
        \emph{lugar geométrico}, exatamente como o círculo foi no
        \cref{pb:g10:coordgeom:1}: equidistância de um ponto e de
        uma reta, em vez de distância a um único ponto.
  \item Para a \omterm{def:g11:quad:parabola}{parábola} mais aberta $y = \frac{x^2}{4f}$ (com
        $f > 0$), adapte a questão 6 para mostrar que o \omterm{pb:g11:quad:1}{foco} é
        $(0, f)$ e a \omterm{pb:g11:quad:1}{diretriz} é $y = -f$. Aplicação: uma antena
        parabólica tem $60$ cm de largura e $9$ cm de
        profundidade --- sua borda passa por $(30, 9)$. Encontre
        $f$: a que altura acima do vértice o receptor deve ser
        colocado?
  \item A razão pela qual as antenas funcionam: os raios que
        chegam paralelos ao eixo refletem todos \emph{pelo \omterm{pb:g11:quad:1}{foco}}
        (e uma lâmpada no \omterm{pb:g11:quad:1}{foco} lança um feixe paralelo --- faróis
        são antenas ao contrário). Explique por que a
        equidistância da questão 7 torna isso plausível (pense
        nos raios como frentes de onda que avançam) e diga qual
        ferramenta, a chegar no \cref{ch:g11:deriv} (veja o
        \cref{exo:g11:quad:11}), transforma plausibilidade em
        demonstração.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- \omterm{def:g11:quad:parabola}{Parábolas} em serviço.}
\begin{enumerate}[resume]
  \item Uma bola segue a trajetória $y = x - \frac{x^2}{20}$
        ($x$ e $y$ em metros). Encontre onde ela cai, a posição
        horizontal do seu ponto mais alto e a altura máxima
        (\cref{prop:g11:quad:variations}).
  \item Na mesma trajetória, há um muro em $x = 4$. Calcule a
        altura da bola ali: ela passa por cima de um muro de
        $3$ m? E de um muro de $4$ m?
  \item Um arco parabólico vence $40$ m de vão com altura máxima
        de $10$ m: em \omterm{def:g10:coordgeom:system}{coordenadas} centradas,
        $y = 10 - \frac{x^2}{40}$. Um caminhão de $7$ m de altura
        pode passar a $10$ m do centro? E a $15$ m?
  \item A receita de um teatro com ingresso a $p$ euros é
        $R(p) = p\,(120 - 2p)$ (a demanda cai quando o preço
        sobe). Encontre o preço que maximiza a receita e o valor
        \omterm{def:g10:functions:extrema}{máximo}.
  \item Os cabos principais da ponte Golden Gate são parabólicos
        com excelente precisão (uma carga uniforme no tabuleiro
        produz uma \omterm{def:g11:quad:parabola}{parábola} --- uma corrente pendurada livremente
        \emph{não}: sua curva, a catenária, precisa da
        exponencial do ano 12). Com vão de $1\,280$ m e flecha de
        $143$ m, o cabo segue
        $y = 143\left(\frac{x}{640}\right)^2$ a partir do
        centro. A que altura acima do seu ponto mais baixo está o
        cabo em $x = 320$ m?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Retas, tangentes e a última
surpresa.}
\begin{enumerate}[resume]
  \item Complete o \cref{exo:g11:quad:11}: as retas
        $y = mx - 1$ que passam pelo ponto exterior $(0, -1)$
        encontram $y = x^2$ conforme o sinal de um
        \omterm{def:g11:quad:discriminant}{discriminante}. Encontre as duas retas tangentes tiradas
        de $(0, -1)$ e seus pontos de contato.
  \item Agora as retas $y = mx + 1$ que passam pelo ponto
        \emph{interior} $(0, 1)$: calcule o \omterm{def:g11:quad:discriminant}{discriminante} da
        \omterm{def:g10:algebra:equation}{equação} de \omterm{def:g10:numbers:interunion}{interseção} e mostre que toda reta desse tipo
        corta a \omterm{def:g11:quad:parabola}{parábola} duas vezes. Moral: de dentro, não há
        tangentes --- exatamente como num círculo.
  \item A corda focal: intersecte $y = x^2$ com a reta
        horizontal que passa pelo \omterm{pb:g11:quad:1}{foco}, $y = \frac14$. Qual é o
        comprimento da corda? Verifique a regra geral (que vale a
        pena guardar para antenas): para $y = \frac{x^2}{4f}$, a
        corda que passa pelo \omterm{pb:g11:quad:1}{foco} e é paralela à \omterm{pb:g11:quad:1}{diretriz} tem
        comprimento $4f$.
  \item A última surpresa: aplique a ampliação
        $(x, y) \mapsto (kx, ky)$ à \omterm{def:g11:quad:parabola}{parábola} $y = x^2$ e mostre
        que a \omterm{def:g10:functions:function}{imagem} é exatamente $y = \frac{x^2}{k}$. Conclua:
        toda \omterm{def:g11:quad:parabola}{parábola} $y = \frac{x^2}{4f}$ é uma ampliação da
        \omterm{def:g11:quad:parabola}{parábola} unitária --- ao contrário de círculos e elipses
        de proporções diferentes, \emph{todas as \omterm{def:g11:quad:parabola}{parábolas} são
        semelhantes}.
  \item Final: o retrato da \omterm{def:g11:quad:parabola}{parábola} em cinco linhas --- uma
        \omterm{def:g10:algebra:equation}{equação} com três trajes (questão 2), um lugar geométrico
        (questões 7--8), o refletor do engenheiro (questões 9 e
        10), a trajetória de um projétil e o cabo de uma ponte
        (questões 11--15) e uma única curva a menos de ampliação
        (questão 19). Uma frase para cada.
\end{enumerate}
\end{problem}
