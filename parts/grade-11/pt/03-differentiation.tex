\chapter{Derivadas: um primeiro curso}\label{ch:g11:deriv}

Com que rapidez uma \omterm{def:g11:func:function}{função} varia? A resposta em um único ponto é a
\emph{\omterm{def:g11:deriv:derivative}{derivada}}, o número mais útil de toda a matemática aplicada. Este
capítulo a constrói a partir das \omterm{def:g11:deriv:rate}{taxas de variação médias}, calcula-a para
as \omterm{def:g11:func:reference}{funções de referência} e usa seu sinal para ler a variação de uma
\omterm{def:g11:func:function}{função}. Tudo o que está aqui é estendido e aprofundado no
\cref{ch:g12:deriv}.

\section{A taxa de variação média}

\begin{definition}[Taxa de variação média]\label{def:g11:deriv:rate}
Seja $f$ definida em um \omterm{def:g10:numbers:interval}{intervalo} $I$ e sejam $a \neq b$ em $I$. A
\emph{taxa de variação média}\index{taxa de variação média} de $f$ entre
$a$ e $b$ é
\[
\frac{f(b) - f(a)}{b - a}.
\]
Ela é o \omterm{def:g10:reffunc:affine}{coeficiente angular} da reta \emph{secante}\index{secante} que
passa pelos pontos $\bigl(a, f(a)\bigr)$ e $\bigl(b, f(b)\bigr)$ da
curva.
\end{definition}

\begin{example}
Um carro percorre $140$ km entre 14h e 16h: sua velocidade \omterm{def:g10:stats:mean}{média} é
$\frac{140}{2} = 70$ km/h. O velocímetro, porém, mostra a cada instante
uma velocidade \emph{instantânea} --- e é exatamente essa noção que a
\omterm{def:g11:deriv:derivative}{derivada} torna precisa.
\end{example}

\section{A derivada em um ponto}

\begin{definition}[Derivada em um ponto]\label{def:g11:deriv:derivative}
Seja $f$ definida em um \omterm{def:g10:numbers:interval}{intervalo} que contém $a$. Para $h \neq 0$, forme
a \omterm{def:g11:deriv:rate}{taxa de variação média} entre $a$ e $a + h$:
\[
r(h) = \frac{f(a+h) - f(a)}{h}.
\]
Se $r(h)$ se aproxima de um único número fixo quando $h$ fica
arbitrariamente próximo de $0$, dizemos que $f$ é \emph{derivável em
$a$}\index{derivada}\index{derivável}; esse número é a \emph{derivada de
$f$ em $a$}, escrita $f'(a)$:
\[
f'(a) = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}.
\]
\end{definition}

\begin{example}\label{ex:g11:deriv:atpoint}
Seja $f(x) = x^2$ e $a = 1$. Para $h \neq 0$:
\[
\frac{f(1+h) - f(1)}{h}
= \frac{(1+h)^2 - 1}{h}
= \frac{1 + 2h + h^2 - 1}{h}
= \frac{h(2 + h)}{h}
= 2 + h .
\]
Quando $h$ se aproxima de $0$, isso se aproxima de $2$: a \omterm{def:g11:func:function}{função} $x^2$ é
\omterm{def:g11:deriv:derivative}{derivável} em $1$ e $f'(1) = 2$. Note a estratégia: \emph{simplifique o
quociente até $h$ não aparecer mais em nenhum denominador} e só então
faça $h \to 0$.
\end{example}

\begin{example}[Um ponto sem derivada]
Seja $f(x) = \abs{x}$ e $a = 0$. Então
$\frac{f(h) - f(0)}{h} = \frac{\abs h}{h}$, que vale $1$ para $h > 0$ e
$-1$ para $h < 0$: nenhum número único é atingido, e $\abs x$ não é
\omterm{def:g11:deriv:derivative}{derivável} em $0$. Sua curva tem ali um \emph{bico}.
\end{example}

\section{A reta tangente}

\begin{definition}[Tangente]\label{def:g11:deriv:tangent}
Se $f$ é \omterm{def:g11:deriv:derivative}{derivável} em $a$, a \emph{tangente}\index{tangente} à curva no
ponto $A = \bigl(a, f(a)\bigr)$ é a reta que passa por $A$ com
\omterm{def:g10:reffunc:affine}{coeficiente angular} $f'(a)$. Sua \omterm{def:g10:algebra:equation}{equação} é
\[
y = f(a) + f'(a)(x - a).
\]
\end{definition}

A \omterm{def:g11:deriv:tangent}{tangente} é a posição-limite das \omterm{def:g11:deriv:rate}{secantes} que passam por $A$: quando
$h \to 0$, o segundo ponto $\bigl(a+h, f(a+h)\bigr)$ desliza pela curva
rumo a $A$, e o \omterm{def:g10:reffunc:affine}{coeficiente angular} da \omterm{def:g11:deriv:rate}{secante},
$\frac{f(a+h)-f(a)}{h}$, se aproxima de $f'(a)$.

\begin{example}\label{ex:g11:deriv:tangenteq}
Para $f(x) = x^2$ em $a = 1$: $f(1) = 1$ e $f'(1) = 2$
(\cref{ex:g11:deriv:atpoint}), de modo que a \omterm{def:g11:deriv:tangent}{tangente} é
\[
y = 1 + 2(x - 1) = 2x - 1 .
\]
É a reta $d_2$ encontrada por álgebra pura no \cref{exo:g11:quad:11}.
\end{example}

\section{Derivadas das funções de referência}

\begin{proposition}\label{prop:g11:deriv:refderivatives}
Nos seus \omterm{def:g11:func:function}{domínios} de derivabilidade:
\[
(c)' = 0, \quad
(x)' = 1, \quad
(x^2)' = 2x, \quad
(x^3)' = 3x^2, \quad
\left(\frac1x\right)' = -\frac{1}{x^2}, \quad
(\sqrt x\,)' = \frac{1}{2\sqrt x} \ (x > 0).
\]
Mais geralmente, $(x^n)' = n\,x^{n-1}$ para todo inteiro $n \geq 1$.
\end{proposition}

\begin{proof}
Cada demonstração segue a estratégia do \cref{ex:g11:deriv:atpoint}:
simplificar a taxa e depois fazer $h \to 0$. Fixe $a$ no \omterm{def:g11:func:function}{domínio}.

\emph{Constante e identidade.} $\frac{c - c}{h} = 0$ e
$\frac{(a+h) - a}{h} = 1$ para todo $h$: os limites são $0$ e $1$.

\emph{Quadrado.} $\frac{(a+h)^2 - a^2}{h} = \frac{2ah + h^2}{h} = 2a + h
\to 2a$.

\emph{Cubo.} $(a+h)^3 = a^3 + 3a^2h + 3ah^2 + h^3$, de modo que a taxa é
$3a^2 + 3ah + h^2 \to 3a^2$.

\emph{Inverso.} Para $a \neq 0$ e $h$ pequeno o bastante para que
$a + h \neq 0$:
\[
\frac{1}{h}\left(\frac{1}{a+h} - \frac{1}{a}\right)
= \frac{1}{h} \cdot \frac{a - (a+h)}{a(a+h)}
= \frac{-1}{a(a+h)}
\xrightarrow[h \to 0]{} -\frac{1}{a^2}.
\]

\emph{Raiz quadrada.} Para $a > 0$, multiplique pelo conjugado:
\[
\frac{\sqrt{a+h} - \sqrt a}{h}
= \frac{(a + h) - a}{h\,(\sqrt{a+h} + \sqrt a)}
= \frac{1}{\sqrt{a+h} + \sqrt a}
\xrightarrow[h \to 0]{} \frac{1}{2\sqrt a}.
\]

\emph{Potência geral.} O padrão $2a$, $3a^2$ prossegue: ao \omterm{def:g10:algebra:expand}{expandir}
$(a+h)^n$, todos os termos depois de $a^n + n\,a^{n-1}h$ carregam $h^2$,
de modo que a taxa tende a $n\,a^{n-1}$. (Uma indução completa é dada no
\cref{ch:g12:deriv}.)
\end{proof}

\section{Regras de derivação}

\begin{theorem}[Operações]\label{thm:g11:deriv:rules}
Sejam $u$ e $v$ deriváveis em $I$ e seja $\lambda \in \R$. Então
$u + v$, $\lambda u$ e $uv$ são deriváveis em $I$, assim como
$\frac{u}{v}$ onde $v \neq 0$, e
\[
(u+v)' = u' + v', \qquad
(\lambda u)' = \lambda u', \qquad
(uv)' = u'v + uv', \qquad
\left(\frac{u}{v}\right)' = \frac{u'v - uv'}{v^2}.
\]
\end{theorem}

\begin{proof}[Demonstração das três primeiras regras]
Fixe $a \in I$ e abrevie $\Delta u = u(a+h) - u(a)$,
$\Delta v = v(a+h) - v(a)$.

\emph{Soma.} A taxa de $u + v$ é
$\frac{\Delta u + \Delta v}{h} = \frac{\Delta u}{h} + \frac{\Delta v}{h}
\to u'(a) + v'(a)$.

\emph{Múltiplo.} A taxa de $\lambda u$ é
$\lambda \frac{\Delta u}{h} \to \lambda u'(a)$.

\emph{Produto.} Some e subtraia o termo misto $u(a+h)\,v(a)$:
\begin{align*}
\frac{u(a+h)v(a+h) - u(a)v(a)}{h}
&= \frac{u(a+h)v(a+h) - u(a+h)v(a) + u(a+h)v(a) - u(a)v(a)}{h}\\
&= u(a+h)\,\frac{\Delta v}{h} + v(a)\,\frac{\Delta u}{h}.
\end{align*}
Quando $h \to 0$: $\frac{\Delta v}{h} \to v'(a)$,
$\frac{\Delta u}{h} \to u'(a)$ e $u(a+h) \to u(a)$ (sua taxa tem limite,
logo $\Delta u = h \cdot \frac{\Delta u}{h} \to 0$). A taxa, portanto,
tende a $u(a)v'(a) + v(a)u'(a)$.

A regra do quociente demonstra-se nas mesmas linhas; ela é
\emph{admitida neste nível} e demonstrada no \cref{ch:g12:deriv}.
\end{proof}

\begin{example}\label{ex:g11:deriv:computations}
\emph{Polinômio.} $f(x) = 3x^3 - 5x^2 + 7x - 2$:
\[
f'(x) = 3 \times 3x^2 - 5 \times 2x + 7 = 9x^2 - 10x + 7 .
\]

\emph{Quociente.} $g(x) = \frac{2x+1}{x^2+1}$: com $u = 2x + 1$,
$v = x^2 + 1$,
\[
g'(x) = \frac{2(x^2+1) - (2x+1)\,2x}{(x^2+1)^2}
= \frac{-2x^2 - 2x + 2}{(x^2+1)^2}.
\]

\emph{Produto.} $k(x) = x\sqrt x$ para $x > 0$:
$k'(x) = 1 \times \sqrt x + x \times \frac{1}{2\sqrt x}
= \sqrt x + \frac{\sqrt x}{2} = \frac{3\sqrt x}{2}$.
\end{example}

\section{Sinal da derivada e variação}

\begin{theorem}[Derivada e variação]\label{thm:g11:deriv:variations}
Seja $f$ \omterm{def:g11:deriv:derivative}{derivável} em um \omterm{def:g10:numbers:interval}{intervalo} $I$.
\begin{enumerate}
  \item Se $f' \geq 0$ em $I$, então $f$ é \omterm{def:g11:func:monotone}{crescente} em $I$.
  \item Se $f' \leq 0$ em $I$, então $f$ é \omterm{def:g10:functions:variations}{decrescente} em $I$.
  \item Se $f' > 0$ em $I$ (a não ser, talvez, em um número finito de
        pontos onde se anula), então $f$ é estritamente \omterm{def:g11:func:monotone}{crescente} em $I$.
\end{enumerate}
\end{theorem}

\begin{proof}
A intuição é clara --- uma curva cujas \omterm{def:g11:deriv:tangent}{tangentes} apontam todas para cima
tem de subir ---, mas uma demonstração rigorosa exige o teorema do valor
médio. O resultado é \emph{admitido neste nível}; veja o
\cref{thm:g12:deriv:variations}.
\end{proof}

\begin{proposition}[Extremos]\label{prop:g11:deriv:extrema}
Se $f'$ muda de sinal em $a$, de $+$ para $-$, então $f$ tem um \omterm{def:g10:functions:extrema}{máximo}
local em $a$; de $-$ para $+$, um \omterm{def:g10:functions:extrema}{mínimo} local. Em um ponto assim,
$f'(a) = 0$ e a \omterm{def:g11:deriv:tangent}{tangente} é horizontal.
\end{proposition}

\begin{proof}
Se $f' \geq 0$ em um \omterm{def:g10:numbers:interval}{intervalo} que termina em $a$ e $f' \leq 0$ em um
\omterm{def:g10:numbers:interval}{intervalo} que começa em $a$, então $f$ cresce até $f(a)$ e decresce em
seguida (\cref{thm:g11:deriv:variations}): $f(a)$ é o maior valor nas
proximidades. Como $f'$ muda de sinal em $a$ e está definida ali,
$f'(a) = 0$.
\end{proof}

\begin{method}[Estudar uma função]\label{met:g11:deriv:study}
\begin{enumerate}
  \item Calcule $f'(x)$ e simplifique-a --- de preferência até a forma
        fatorada.
  \item Determine o sinal de $f'(x)$ em cada \omterm{def:g10:numbers:interval}{intervalo} (se $f'$ for do
        segundo grau, use o \cref{thm:g11:quad:sign}).
  \item Registre os resultados em um \emph{\omterm{def:g10:functions:table}{quadro de variação}}: uma linha
        para o sinal de $f'$, uma linha de setas para $f$ e os valores de
        $f$ nos pontos de virada.
  \item Conclua: extremos, número de soluções de $f(x) = k$,
        desigualdades.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g11:deriv:study}
Estude $f(x) = x^3 - 3x + 1$ em $\R$. Primeiro,
$f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$: um \omterm{def:g11:quad:quadratic}{trinômio do segundo grau} com raízes
$-1$ e $1$, positivo fora delas. Assim, $f$ é \omterm{def:g11:func:monotone}{crescente} em
$\intoc{-\infty}{-1}$, \omterm{def:g10:functions:variations}{decrescente} em $\intcc{-1}{1}$ e \omterm{def:g11:func:monotone}{crescente} em
$\intco{1}{+\infty}$, com \omterm{def:g10:functions:extrema}{máximo} local $f(-1) = -1 + 3 + 1 = 3$ e \omterm{def:g10:functions:extrema}{mínimo}
local $f(1) = 1 - 3 + 1 = -1$.
\end{example}

\begin{example}[Otimização]\label{ex:g11:deriv:box}
Uma caixa aberta é feita a partir de uma folha $12 \times 12$ cm,
cortando-se um quadrado de lado $x$ em cada canto e dobrando. Seu volume
é $V(x) = x(12 - 2x)^2$ para $0 < x < 6$. Expandindo,
$V(x) = 4x^3 - 48x^2 + 144x$, logo
\[
V'(x) = 12x^2 - 96x + 144 = 12(x^2 - 8x + 12) = 12(x - 2)(x - 6).
\]
Em $\intoo{0}{6}$, o fator $x - 6$ é negativo, de modo que $V'(x)$ tem o
sinal de $-(x-2)$: positivo antes de $2$ e negativo depois. O volume é
\omterm{def:g10:functions:extrema}{máximo} para $x = 2$ cm, dando $V(2) = 2 \times 8^2 = 128$ cm$^3$.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g11:deriv:1}
Usando a definição (taxa de variação e depois $h \to 0$), calcule $f'(2)$
para $f(x) = x^2$ e depois $g'(1)$ para $g(x) = x^2 + 3x$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:deriv:2}
Derive:
\[
f(x) = 4x^3 - 2x^2 + x - 7, \qquad

h(x) = (2x+1)(x^2 - 3) .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:deriv:3}
Dê a \omterm{def:g10:algebra:equation}{equação} da \omterm{def:g11:deriv:tangent}{tangente} à curva de $f$ no ponto de \omterm{def:g10:coordgeom:system}{abscissa} $a$:
\[
f(x) = x^2 - 3x, \ a = 2; \qquad
f(x) = \frac1x, \ a = 1; \qquad
f(x) = \sqrt x, \ a = 4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:deriv:4}
Derive os quocientes:
\[
f(x) = \frac{x+2}{x-1}, \qquad
g(x) = \frac{x^2}{x^2+1}, \qquad
h(x) = \frac{1}{x^2 + x + 1} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:deriv:5}
Estude a variação de $f(x) = x^3 - 6x^2 + 9x - 2$ em $\R$ (\omterm{def:g10:functions:table}{quadro de

\begin{exercise}[$\star\star$]\label{exo:g11:deriv:6}
Estude a variação de $f(x) = \frac{x^2 + 3}{x + 1}$ em

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:deriv:7}
Seja $f(x) = x^2 - 4x + 5$.
\begin{enumerate}
  \item Encontre o ponto da curva onde a \omterm{def:g11:deriv:tangent}{tangente} é horizontal.
  \item Encontre o ponto (ou os pontos) onde a \omterm{def:g11:deriv:tangent}{tangente} é paralela à reta

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:deriv:8}
Um cercado retangular é construído encostado em um muro com $60$ m de

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:deriv:9}
Mostre que, para todo $x > 0$, $\sqrt{x} \leq \frac{x + 1}{2}$,
estudando a \omterm{def:g11:func:function}{função} $f(x) = \frac{x+1}{2} - \sqrt x$ em
$\intoo{0}{+\infty}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:deriv:10}
Uma lata cilíndrica deve conter $500$ cm$^3$. Sua área total (dois discos
mais a lateral) é $S = 2\pi r^2 + 2\pi r h$, com $\pi r^2 h = 500$.
Exprima $S$ em \omterm{def:g11:func:function}{função} apenas de $r$ e mostre que

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:deriv:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:deriv:12}

\section{Problema: Os três ofícios da tangente}

\begin{problem}[{Problema de fim de semana --- o espelho parabólico
demonstrado, a máquina de achar raízes de Newton e a viga mais forte
que se tira de uma tora}]
\label{pb:g11:deriv:1}
Uma \omterm{def:g11:deriv:derivative}{derivada} traça \omterm{def:g11:deriv:tangent}{tangentes}; isso parece um talento modesto. Este
problema mostra a \omterm{def:g11:deriv:tangent}{tangente} exercendo três ofícios ao mesmo tempo:
ela demonstra o teorema do farol deixado pendente no
\cref{pb:g11:quad:1} (todo raio paralelo ao eixo de uma \omterm{def:g11:quad:parabola}{parábola}
reflete pelo \omterm{pb:g11:quad:1}{foco}), move o algoritmo de resolução de \omterm{def:g10:algebra:equation}{equações} mais
rápido em uso corrente --- dentro de toda calculadora e de toda
rede neural --- e encontra a viga retangular mais forte que uma
tora redonda pode render.

\medskip
\noindent\textbf{Parte I --- Fluência com \omterm{def:g11:deriv:tangent}{tangentes}.}
\begin{enumerate}
  \item Derive $f(x) = 3x^2 - 5x + 1$ e $g(x) = x^3 - 12x$
        (\cref{thm:g11:deriv:rules}); e reobtenha
        $\left(x^2\right)'$ em $x = 1$ pela definição (taxa de
        variação e depois $h \to 0$).
  \item Encontre a \omterm{def:g11:deriv:tangent}{tangente} a $y = x^3 - 12x$ em $a = 2$. O que
        ela tem de especial e o que isso sinaliza
        (\cref{prop:g11:deriv:extrema})?
  \item Estabeleça a \omterm{def:g11:deriv:tangent}{tangente} geral à \omterm{def:g11:quad:parabola}{parábola} $y = x^2$ no ponto
        $(a, a^2)$:
        \[
        y = 2ax - a^2 .
        \]
  \item Onde essa \omterm{def:g11:deriv:tangent}{tangente} cruza o eixo $y$? Deduza a receita do
        desenhista: para traçar a \omterm{def:g11:deriv:tangent}{tangente} em um ponto $P$ de
        altura $h$ da \omterm{def:g11:quad:parabola}{parábola}, ligue $P$ ao ponto do eixo
        situado à profundidade\,\dots\ abaixo da origem.
        Enuncie-a.
  \item Monte o \omterm{def:g10:functions:table}{quadro de variação} completo de
        $g(x) = x^3 - 12x$ (sinal de $g'$, extremos locais e seus
        valores, \cref{met:g11:deriv:study}).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O teorema do espelho.}
Seja $P(a, a^2)$ um ponto da \omterm{def:g11:quad:parabola}{parábola} $y = x^2$ (tome
$a \neq 0$), $F\left(0, \frac14\right)$ o \omterm{pb:g11:quad:1}{foco} e $T$ o ponto onde
a \omterm{def:g11:deriv:tangent}{tangente} em $P$ cruza o eixo $y$ (questão 4). Lembre-se, do
\cref{pb:g11:quad:1}, que $PF = a^2 + \frac14$.
\begin{enumerate}[resume]
  \item Calcule a distância $FT$.
  \item Conclua que o triângulo $FPT$ é isósceles em $F$. Que
        dois ângulos, portanto, são iguais?
  \item O raio incidente que passa por $P$, paralelo ao eixo, é
        paralelo à reta $(TF)$ (as duas são verticais). Usando
        ângulos alternos, mostre que a \omterm{def:g11:deriv:tangent}{tangente} em $P$ faz
        \emph{ângulos iguais} com o raio vertical incidente e com
        o segmento $[PF]$.
  \item Física: a luz reflete numa curva como refletiria na sua
        \omterm{def:g11:deriv:tangent}{tangente}, saindo com o mesmo ângulo com que chegou (a lei
        da reflexão). Conclua o \emph{teorema do espelho}: todo
        raio paralelo ao eixo reflete pelo \omterm{pb:g11:quad:1}{foco} --- e, invertendo
        o tempo, uma lâmpada no \omterm{pb:g11:quad:1}{foco} emite um feixe paralelo. A
        antena e o farol do \cref{pb:g11:quad:1} são agora
        teoremas.
  \item Verifique numericamente o triângulo isósceles em $a = 1$:
        calcule $T$, $FP$ e $FT$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A máquina de Newton.}
Para resolver $f(x) = 0$, Newton propôs: parta de um palpite
$x_0$, siga a \omterm{def:g11:deriv:tangent}{tangente} em $x_0$ até o eixo e tome o ponto onde
ela o corta como palpite seguinte.
\begin{enumerate}[resume]
  \item Deduza a fórmula do método:
        \[
        x_{n + 1} = x_n - \frac{f(x_n)}{f'(x_n)} .
        \]
  \item Rode-a em $f(x) = x^2 - 2$ a partir de $x_0 = 1$:
        calcule $x_1$, $x_2$, $x_3$ como frações exatas. Que
        sequência de dois mil anos --- e que máquina do
        \cref{pb:g10:functions:1} --- você acabou de
        reconstruir? Demonstre a identidade por trás da
        coincidência:
        \[
        x - \frac{x^2 - 2}{2x} = \frac12\left(x +
        \frac2x\right).
        \]
  \item Resolva $x^3 = 100$ (a ultrapassagem dos planos de
        poupança do \cref{pb:g10:reffunc:1}): aplique dois passos
        de Newton a $f(x) = x^3 - 100$ a partir de $x_0 = 5$
        (quatro casas decimais) e compare com o
        $\sqrt[3]{100}$ da calculadora.
  \item Sabotagem: tente lançar o método em $f(x) = x^3 - 12x$
        com $x_0 = 2$. O que dá errado e por quê (questão 2)?
        Enuncie a ressalva prática.
  \item Na questão 12 as casas decimais corretas foram mais ou
        menos $1 \to 3 \to 6$. Compare com a bisseção (dividir um
        \omterm{def:g10:numbers:interval}{intervalo} ao meio a cada passo) e diga em uma frase por
        que calculadoras, receptores de GPS e o treinamento de
        redes neurais rodam todos sobre a ideia de Newton.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A viga mais forte.}
Uma viga retangular de largura $w$ e altura $d$ é serrada de uma
tora redonda de diâmetro $30$ cm, de modo que $w^2 + d^2 = 900$. A
rigidez de uma viga é proporcional a $w d^2$ (a altura conta duas
vezes: as vigas ficam de pé!).
\begin{enumerate}[resume]
  \item Exprima a rigidez em \omterm{def:g11:func:function}{função} apenas de $w$:
        $S(w) = 900w - w^3$, em que \omterm{def:g10:numbers:interval}{intervalo}?
  \item Derive, monte o \omterm{def:g10:functions:table}{quadro de variação} e encontre os valores
        ótimos de $w$ e $d$ (valores exatos e depois em
        milímetros).
  \item Mostre que as proporções ótimas satisfazem
        $d = w\sqrt2$ --- a velha razão dos carpinteiros. (Bônus
        de tradição: dividir o diâmetro da tora em três partes
        iguais e levantar perpendiculares até o círculo constrói
        exatamente esse retângulo.)
  \item A escolha ingênua é a viga \emph{quadrada} ($w = d$).
        Calcule sua rigidez e a da viga ótima, e dê o ganho do
        carpinteiro em porcentagem.
  \item Final --- os três ofícios da \omterm{def:g11:deriv:tangent}{tangente}, uma frase para
        cada: geometria (o teorema do espelho), cálculo numérico
        (a máquina de Newton), otimização (a viga), todos movidos
        pelo mesmo objeto. E o aviso adiante: o ano 12 acrescenta
        a convexidade --- de que lado da curva fica a \omterm{def:g11:deriv:tangent}{tangente}
        --- para aguçar os três.
\end{enumerate}
\end{problem}
