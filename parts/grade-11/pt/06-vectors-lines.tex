\chapter{Vetores e retas no plano}\label{ch:g11:vect}

\omterm{def:g11:vect:vector}{vetores} codificar deslocamentos; linhas são suas trajetórias. A chave nova
ferramenta deste capítulo é o \emph{\omterm{def:g11:vect:det}{determinante}}, um único número que
decide se dois \omterm{def:g11:vect:vector}{vetores} são \omterm{def:g11:vect:collinear}{colineares} --- e, a partir daí, produz
equações de linhas e resolve questões de alinhamento e paralelismo por puro
computação. As mesmas ideias estendem-se ao espaço em \cref{ch:g12:space}.

\section{Vetores: uma recapitulação}

\begin{definition}[Vetor, coordenadas]\label{def:g11:vect:vector}
UM \emph{vetores}\index{vetores} $\vec u $ representa um deslocamento: todos
setas com a mesma direção, orientação e comprimento representam o mesmo
vetor. Em um \omterm{def:g10:coordgeom:system}{sistema de coordenação}, $\vec u\,(x, y)$ registra o deslocamento
($x$ horizontalmente, $ y $ verticalmente); por pontos $ A(x_A, y_A)$ e
$ B(x_B, y_B)$,
\[
\vect{AB}\,\bigl(x_B - x_A,\ y_B - y_A\bigr).
\]
Vetores somam \omterm{def:g10:coordgeom:system}{coordenadas} e $\lambda\vec u $ tem \omterm{def:g10:coordgeom:system}{coordenadas}
$(\lambda x, \lambda y)$.
\end{definition}

\begin{proposition}[Ponto médio]\label{prop:g11:vect:midpoint}
O \omterm{prop:g10:coordgeom:midpoint}{ponto médio} $ M $ do segmento $[AB]$ tem \omterm{def:g10:coordgeom:system}{coordenadas}
$\left(\frac{x_A + x_B}{2},\ \frac{y_A + y_B}{2}\right)$.
\end{proposition}

\begin{proof}
$ M $ é caracterizado por $\vect{AM} = \frac12\vect{AB}$: em termos de \omterm{def:g10:coordgeom:system}{coordenadas},
$ x_M - x_A = \frac12(x_B - x_A)$, então $ x_M = \frac{x_A + x_B}{2}$e
da mesma forma para $ y_M $.
\end{proof}

\section{Colinearidade e o determinante}

\begin{definition}[Vetores colineares]\label{def:g11:vect:collinear}
Dois \omterm{def:g11:vect:vector}{vetores} são \emph{colinear}\index{colinear} se alguém é um
múltiplo escalar do outro: $\vec v = \lambda \vec u $ para alguns
$\lambda \in \R $ (ou $\vec u = \vec 0$). Geometricamente: eles têm o mesmo
direção, ou um deles é zero.
\end{definition}

\begin{definition}[Determinante]\label{def:g11:vect:det}
O \emph{determinante}\index{determinante} de $\vec u\,(x, y)$ e
$\vec v\,(x', y')$ é
\[
\det(\vec u, \vec v) =
\begin{vmatrix} x & x' \\ y & y' \end{vmatrix}
= x y' - x' y .
\]
\end{definition}

\begin{theorem}[Critério de colinearidade]\label{thm:g11:vect:collinearity}
Dois \omterm{def:g11:vect:vector}{vetores} $\vec u $ e $\vec v $ são \omterm{def:g11:vect:collinear}{colineares} se e somente se
$\det(\vec u, \vec v) = 0$.
\end{theorem}

\begin{proof}
($\Rightarrow $) Se $\vec v = \lambda\vec u $, então $ x' = \lambda x $ e
$ y' = \lambda y $, então
$\det(\vec u, \vec v) = x(\lambda y) - (\lambda x)y = 0$; se
$\vec u = \vec 0$, o \omterm{def:g11:vect:det}{determinante} é trivialmente $0$.

($\Leftarrow $) Suponha $ xy' - x'y = 0$ e $\vec u \neq \vec 0$, dizer
$ x \neq 0$ (o caso $ y \neq 0$ é simétrico). Definir
$\lambda = \frac{x'}{x}$. Então $ x' = \lambda x $ por construção, e o
relação $ xy' = x'y = \lambda x y $ dá, depois de dividir por $ x \neq 0$,
$ y' = \lambda y $. Por isso $\vec v = \lambda \vec u $.
\end{proof}

\begin{example}\label{ex:g11:vect:collinear}
$\vec u\,(3, -2)$ e $\vec v\,(-6, 4)$:
$\det = 3 \times 4 - (-6) \times (-2) = 12 - 12 = 0$, \omterm{def:g11:vect:collinear}{colineares} (de fato
$\vec v = -2\vec u $). $\vec u\,(3, -2)$ e $\vec w\,(2, 1)$:
$\det = 3 + 4 = 7 \neq 0$, não \omterm{def:g11:vect:collinear}{colineares}.
\end{example}

\begin{method}[Alinhamento de três pontos]\label{met:g11:vect:alignment}
$ A $, $ B $, $ C $ estão alinhados se e somente se $\vect{AB}$ e $\vect{AC}$ são
\omterm{def:g11:vect:collinear}{colineares}: calcular ambos \omterm{def:g11:vect:vector}{vetores} e verifique
$\det\bigl(\vect{AB}, \vect{AC}\bigr) = 0$.
\end{method}

\section{Equações cartesianas de retas}

\begin{definition}[Vetor de direção]\label{def:g11:vect:direction}
UM \emph{vetor de direção}\index{vetor de direção} de uma linha $ d $ é qualquer
diferente de zero vetor $\vec u $ tal que $\vect{AB}$ é \omterm{def:g11:vect:collinear}{colineares} com $\vec u $
para todos os pontos $ A, B $ de $ d $.
\end{definition}

\begin{theorem}[Equação cartesiana]\label{thm:g11:vect:cartesian}
Uma linha com \omterm{def:g11:vect:direction}{vetor de direção} $\vec u\,(-b, a)$ admite um equação do
formulário
\[
ax + by + c = 0
\qquad (a, b) \neq (0, 0).
\]
Por outro lado, cada um desses equação descreve uma linha com \omterm{def:g11:vect:direction}{vetor de direção}
$\vec u\,(-b, a)$.
\end{theorem}

\begin{proof}
Seja $ d $ passar $ A(x_A, y_A)$ com direção $\vec u\,(-b, a)$. Um
ponto $ M(x, y)$ mente $ d $ exatamente quando $\vect{AM}$ é \omterm{def:g11:vect:collinear}{colineares} com
$\vec u $, ou seja,\ (\cref{thm:g11:vect:collinearity})
\[
0 = \det\bigl(\vect{AM}, \vec u\bigr)
= (x - x_A)\,a - (-b)(y - y_A)
= ax + by - (a x_A + b y_A),
\]
um equação do formulário anunciado com $ c = -(a x_A + b y_A)$.
Por outro lado, as soluções de $ ax + by + c = 0$ com, digamos, $ b \neq 0$ são
os pontos $\left(x, -\frac{a x + c}{b}\right)$: subtraindo dois deles
mostra que cada acorde é \omterm{def:g11:vect:collinear}{colineares} com $(-b, a)$, e uma solução sempre
existe --- é a linha através dele dirigida por $(-b, a)$.
\end{proof}

\begin{method}[Linha que passa por dois pontos]\label{met:g11:vect:twopoints}
Para encontrar um equação da linha $(AB)$: calcular o \omterm{def:g11:vect:direction}{vetor de direção}
$\vect{AB}\,(x_B - x_A, y_B - y_A)$; combine com $(-b, a)$, ou seja,\ pegue
$ a = y_B - y_A $ e $ b = -(x_B - x_A)$; determinar $ c $ conectando o
\omterm{def:g10:coordgeom:system}{coordenadas} de $ A $.
\end{method}

\begin{example}\label{ex:g11:vect:twopoints}
Linha através $ A(1, 2)$ e $ B(4, 3)$: $\vect{AB}\,(3, 1)$, então
$ a = 1$, $ b = -3$, e o equação é $ x - 3y + c = 0$. Conectando $ A $:
$1 - 6 + c = 0$, então $ c = 5$. equação: $ x - 3y + 5 = 0$. Verifique com $ B $:
$4 - 9 + 5 = 0$.
\end{example}

\begin{remark}
Quando $ b \neq 0$, o equação pode ser resolvido para $ y $: $ y = mx + p $ com
coeficiente angular $ m = -\frac ab $. A forma cartesiana $ ax + by + c = 0$ é mais
geral: também cobre linhas verticais ($ b = 0$), que não possuem coeficiente angular.
\end{remark}

\section{Representação paramétrica}

\begin{definition}[Representação paramétrica]\label{def:g11:vect:parametric}
A linha através $ A(x_A, y_A)$ com direção $\vec u\,(\alpha, \beta)$ é
o conjunto de pontos
\[
M\bigl(x_A + t\alpha,\ y_A + t\beta\bigr), \qquad t \in \R .
\]
O número $ t $ é o \emph{parâmetro}: cada valor de $ t $ produz um
ponto da linha, como se viajasse ao longo dela com velocidade constante $\vec u $.
\end{definition}

\begin{example}
A linha de \cref{ex:g11:vect:twopoints} também é
$\{(1 + 3t,\ 2 + t) : t \in \R\}$: $ t = 0$ dá $ A $, $ t = 1$ dá $ B $.
Eliminando $ t $ ($ t = y - 2$, então $ x = 1 + 3(y-2)$) recupera
$ x - 3y + 5 = 0$.
\end{example}

\section{Posições relativas de duas linhas}

\begin{proposition}[Paralelo ou cruzando]\label{prop:g11:vect:positions}
Duas linhas com \omterm{def:g11:vect:vector}{vetores} de direção $\vec u $ e $\vec v $ são paralelos se e
somente se $\det(\vec u, \vec v) = 0$. Em equação forma: $ d : ax + by + c =
0$ e $ d' : a'x + b'y + c' = 0$ são paralelos se e somente se
$ ab' - a'b = 0$. Retas não paralelas se encontram exatamente em um ponto.
\end{proposition}

\begin{proof}
Paralelismo significa \omterm{def:g11:vect:collinear}{colineares} direções, o que é
\cref{thm:g11:vect:collinearity}; com $\vec u\,(-b, a)$ e
$\vec v\,(-b', a')$, o \omterm{def:g11:vect:det}{determinante} é $(-b)a' - (-b')a = ab' - a'b $.
Se as linhas não forem paralelas, resolvendo os dois equações simultaneamente
(por substituição ou combinação) leva a uma solução única: o sistema
se comporta como um \omterm{def:g11:vect:det}{determinante} diferente de zero $2 \times 2$ sistema.
\end{proof}

\begin{method}[Interseção de duas linhas]\label{met:g11:vect:intersection}
Resolva o sistema dos dois equações: isolar um desconhecido em um equação
e substituir no outro, ou combinar o equações para eliminar um
desconhecido. Sempre verifique o ponto resultante em \emph{ambos} equações.
\end{method}

\begin{example}\label{ex:g11:vect:intersection}
Cruzar $ d: x - 3y + 5 = 0$ e $ d': 2x + y - 4 = 0$. De $ d'$:
$ y = 4 - 2x $. Substituindo em $ d $:
\[
x - 3(4 - 2x) + 5 = 0
\iff x - 12 + 6x + 5 = 0
\iff 7x = 7
\iff x = 1,
\]
então $ y = 2$. As linhas se encontram em $(1, 2)$. Check-in $ d $: $1 - 6 + 5 = 0$.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star $]\label{exo:g11:vect:1}
Dado $ A(2, -1)$, $ B(5, 3)$ e $ C(-1, 1)$, calcule o \omterm{def:g10:coordgeom:system}{coordenadas} de
$\vect{AB}$, $\vect{AC}$, $\vect{AB} + \vect{AC}$ e $2\vect{AB} -
3\vect{AC}$, e o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[BC]$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g11:vect:2}
São os \omterm{def:g11:vect:vector}{vetores} \omterm{def:g11:vect:collinear}{colineares}?
\[
\vec u\,(4, -6) \text{ and } \vec v\,(-2, 3); \qquad
\vec u\,(2, 5) \text{ and } \vec v\,(3, 7); \qquad
\vec u\,(0, 3) \text{ and } \vec v\,(0, -8).
\]
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g11:vect:3}
Encontre um cartesiano equação da linha através $ A(2, 1)$ com direção
vetor $\vec u\,(3, -1)$, então da linha através $ B(0, 4)$ e
$ C(2, 0)$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g11:vect:4}
Para a linha $ d: 3x - 2y + 6 = 0$, dê um \omterm{def:g11:vect:direction}{vetor de direção}, o

$ P(2, 6)$ e $ Q(1, 4)$ pertencer a $ d $.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:vect:5}
São os pontos $ A(1, 2)$, $ B(4, 8)$, $ C(-2, -4)$ alinhado? Mesma pergunta

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:vect:6}
Determine o \omterm{def:g10:numbers:interunion}{interseção} ponto, se houver:
\[
d: 2x + y - 7 = 0 \ \text{ and }\ d': x - y + 1 = 0;
\qquad
e: x - 2y + 3 = 0 \ \text{ and }\ e': -2x + 4y + 1 = 0 .

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:vect:7}
Dê uma representação paramétrica da linha $ d: 2x - y + 3 = 0$, e um

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:vect:8}
Encontre o equação da linha através $ P(1, -2)$ paralelo a
$ d: 4x - 3y + 2 = 0$, e o valor de $ m $ para o qual a linha
$ mx + 2y - 5 = 0$ é paralelo a $ d $.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:vect:9}
Seja $ A(-1, 0)$, $ B(3, 2)$ e $ C(1, -4)$. Encontre um cartesiano equação do
\emph{\omterm{def:g10:stats:median}{mediana}} do triângulo $ ABC $ emitido de $ C $ (a linha que une $ C $
para o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$).
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g11:vect:10}
Para quais valores do parâmetro $ m $ são os \omterm{def:g11:vect:vector}{vetores}
$\vec u\,(m, 2)$ e $\vec v\,(3, m - 1)$ \omterm{def:g11:vect:collinear}{colineares}?
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g11:vect:11}
Seja $ ABCD $ seja um paralelogramo, $ I $ o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$, e $ J $ o
ponto definido por $\vect{DJ} = \frac23 \vect{DI}$. Trabalhando no
\omterm{def:g10:coordgeom:system}{sistema de coordenação} onde $ A(0,0)$, $ B(1,0)$, $ D(0,1)$ (então $ C(1,1)$):
\begin{enumerate}
  \item Calcule o \omterm{def:g10:coordgeom:system}{coordenadas} de $ I $ e $ J $.
  \item Mostre isso $ A $, $ J $ e $ C $ estão alinhados. (Um fato interessante: a linha
        $(DI)$ corta a diagonal $(AC)$ um terço do caminho para cima.)
\end{enumerate}
\end{exercise}

\section{Problema: Rotas de colisão e coordenadas inteligentes}

\begin{problem}[{Problema de fim de semana --- cruzar caminhos não é
colidindo: maior aproximação no mar e teoremas provados por
escolhendo o quadro certo}]
\label{pb:g11:vect:1}
Os cursos retos de dois navios se cruzam --- os navios devem colidir? Não
a menos que cheguem ao cruzamento \emph{ao mesmo tempo}: caminhos são
conjuntos de pontos, os movimentos são paramétricos e confundir os dois tem
afundaram navios reais. Este problema executa um serviço de tráfego de pequenas embarcações
em linhas paramétricas (\cref{def:g11:vect:parametric}), então vira
à geometria pura com o outro superpoder de \omterm{def:g10:coordgeom:system}{coordenadas}:
\emph{escolhendo} habilmente, como em \cref{exo:g11:vect:11}, para
fazer com que os teoremas clássicos se calculem.

\medskip
\noindent\textbf{Parte I --- Linhas, três maneiras.}
\begin{enumerate}
  \item Dê uma representação paramétrica da reta que passa
        $ A(1, 2)$ com \omterm{def:g11:vect:direction}{vetor de direção} $\vec u(3, -1)$, então
        elimine o parâmetro para obter um cartesiano equação.
  \item Para a linha $2x - 5y + 3 = 0$: leia um direção
        vetor (\cref{thm:g11:vect:cartesian}), encontre um ponto,
        e escreva uma representação paramétrica.
  \item Calcule o \omterm{def:g10:numbers:interunion}{interseção} ponto das duas linhas de
        questões 1 e 2 (\cref{met:g11:vect:intersection}).
  \item Confirme com o \omterm{def:g11:vect:det}{determinante}
        (\cref{thm:g11:vect:collinearity}) que essas duas linhas
        teve que cruzar; então mostre isso $2x - 5y + 3 = 0$ e
        $-4x + 10y + 1 = 0$ são estritamente paralelos.
  \item São $ A(2, 1)$, $ B(5, 3)$, $ C(11, 7)$ alinhado
        (\cref{met:g11:vect:alignment})?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O serviço de tráfego de navios.}
Na hora $ t $ (horas), navio $ P $ está em $(2t,\ 3 + t)$ e navio $ Q $
no $(10 - t,\ 2t - 1)$ (distâncias em milhas náuticas).
\begin{enumerate}[resume]
  \item Encontre o cartesiano equações dos dois \emph{caminhos},
        e calcule onde os caminhos se cruzam.
  \item A que horas cada navio passa pelo ponto de passagem? Faça
        os navios colidem? Declare a advertência geral em um
        frase: cruzando caminhos contra moções de reunião.
  \item Calcule a distância quadrada $ D^2(t)$ entre os navios
        de vez em quando $ t $, e reduzi-lo a um quadrático em $ t $. Em
        a que horas os navios estão mais próximos e a que distância eles
        venha (\cref{pb:g11:quad:1} tecnologia de vértice)?
  \item Os regulamentos exigem uma separação de pelo menos $1$ milha.
        Está violado? Se sim, resolva $ D^2(t) < 1$ e dê o
        duração da violação.
  \item Explique por que, para \emph{qualquer} dois navios em linha reta
        cursos em velocidades constantes, $ D^2(t)$ é sempre um
        \omterm{def:g11:quad:quadratic}{função quadrática} de $ t $ --- então ``ponto mais próximo de
        abordagem'' é sempre um cálculo de vértice.
  \item Interceptação: um barco patrulha parte da origem em
        $ t = 0$ com velocidade constante $(a, b)$ e deve se encontrar
        navio $ P $ exatamente em $ t = 3$. Calcule o necessário
        $(a, b)$ e a velocidade do barco patrulha.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Inteligente \omterm{def:g10:coordgeom:system}{coordenadas}.}
Trabalhar no quadro de \cref{exo:g11:vect:11}: o paralelogramo
$ ABCD $ torna-se $ A(0,0)$, $ B(1,0)$, $ C(1,1)$, $ D(0,1)$; seja $ I $
seja o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[AB]$ e $ K $ o \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[CD]$.
\begin{enumerate}[resume]
  \item Calcule um cartesiano equação da linha $(DI)$.
  \item Cruzar $(DI)$ com a diagonal $(AC)$: recuperar
        \cref{exo:g11:vect:11}é fato --- o corte está exatamente
        um terço da diagonal.
  \item Agora a linha $(BK)$: encontre seu equação, cruza com
        $(AC)$, e conclua o teorema clássico completo:
        \emph{os cevianos $(DI)$ e $(BK)$ trissectar a diagonal
        $(AC)$}.
  \item Por que foi legítimo provar o teorema neste
        quadro especial? (Qual é a escolha de \omterm{def:g10:coordgeom:system}{coordenadas}
        preservar, e o que a afirmação envolve?) Resposta
        em duas ou três frases.
  \item Mais uma trissecção para praticar: vamos $ E $ seja o
        \omterm{prop:g10:coordgeom:midpoint}{ponto médio} de $[BC]$. Onde fica a linha $(AE)$ corte o
        \emph{outro} diagonal $(DB)$?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Quebra-cabeças de posição.}
\begin{enumerate}[resume]
  \item São as três linhas $ x + y = 4$, \ $2x - y = 2$, \
        $ x - 2y = -2$ simultâneo? (Cruze dois, teste o
        terceiro.)
  \item Para cada real $ m $, seja $ d_m $ seja a linha
        $ mx - y + 2 - 3m = 0$. Mostre isso \emph{todo} $ d_m $
        passa por um \omterm{pb:g10:functions:1}{ponto fixo}. (Agrupe os termos em
        $ m $.)
  \item Na família $(d_m)$: qual membro é paralelo a
        $ y = 2x + 7$? Qual passa pela origem?
  \item Finale --- os três figurinos de uma linha: Cartesiana
        (testes de adesão em uma substituição), reduzido (coeficiente angular
        e grafo à vista), paramétrico (movimento, tempo,
        interceptação); o \omterm{def:g11:vect:det}{determinante} como oráculo de paralelismo;
        e o quadro bem escolhido como fábrica de teoremas. Um
        sentença cada.
\end{enumerate}
\end{problem}
