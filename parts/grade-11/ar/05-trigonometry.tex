\chapter{حساب المثلثات: دائرة الوحدة}\label{ch:g11:trigo}

يبدأ حساب المثلثات في المثلثات القائمة، لكن موطنه الحقيقي هو
\emph{دائرة الوحدة}، حيث يصير \omterm{def:g11:trigo:cossin}{جيب التمام} \omterm{def:g11:trigo:cossin}{والجيب} دالتين \omterm{def:g10:numbers:sets}{لعدد
حقيقي} كيفي. يركّب هذا الفصل \omterm{def:g11:trigo:radian}{الراديان}، ولفّ المستقيم
الحقيقي حول الدائرة، والقيم المضبوطة الواجب حفظها، و
التناظرات التي تولّد كل المتطابقات الكلاسيكية. أما دراسة
$\cos$ و $\sin$ دالتين --- تغيرات ومشتقات --- فتجري
في \cref{ch:g12:trigo}.

\section{الراديان واللفّ}

\begin{definition}[الراديان]\label{def:g11:trigo:radian}
لتكن $\mathcal C$ الدائرة التي نصف قطرها $1$ ومركزها المبدأ $O$،
ولتكن $I(1, 0)$. قياس زاوية مركزية في $\mathcal C$ بوحدة \emph{الراديان}\index{راديان}
هو طول القوس الذي تقطعه على $\mathcal C$.
وبما أن محيط الدائرة الكاملة $2\pi$:
\[
360^\circ = 2\pi \text{ راد},
\qquad
180^\circ = \pi \text{ راد},
\qquad
1 \text{ راد} = \frac{180^\circ}{\pi} \approx 57.3^\circ .
\]
\end{definition}

\begin{method}[التحويل]\label{met:g11:trigo:convert}
الدرجات والراديانات متناسبة: اضرب في $\frac{\pi}{180}$ للانتقال
من الدرجات إلى الراديانات، وفي $\frac{180}{\pi}$ في الاتجاه الآخر. فمثلًا
$60^\circ = 60 \times \frac{\pi}{180} = \frac{\pi}{3}$، و
$\frac{3\pi}{4} = \frac{3 \times 180^\circ}{4} = 135^\circ$.
\end{method}

\begin{definition}[لفّ المستقيم على الدائرة]\label{def:g11:trigo:winding}
نرفق بكل \omterm{def:g10:numbers:sets}{عدد حقيقي} $t$ النقطة $M(t)$ من $\mathcal C$
التي نبلغها بالسير مسافة $\abs t$ على الدائرة انطلاقًا من $I$،
في الاتجاه المباشر إذا كان $t \geq 0$، وفي الاتجاه المعاكس إذا كان $t < 0$. وبما أن
المحيط $2\pi$، يبلغ العددان $t$ و $t + 2k\pi$ (حيث $k \in \Z$)
النقطة نفسها: $M(t + 2k\pi) = M(t)$.
\end{definition}

\begin{example}
$M(0) = I$؛ و $M\!\left(\frac{\pi}{2}\right) = (0, 1)$، وهي أعلى نقطة في
الدائرة؛ و $M(\pi) = (-1, 0)$؛ و $M\!\left(-\frac{\pi}{2}\right) = (0, -1)$؛
و $M\!\left(\frac{9\pi}{4}\right) = M\!\left(\frac{\pi}{4} +
2\pi\right) = M\!\left(\frac{\pi}{4}\right)$.
\end{example}

\section{جيب التمام والجيب}

\begin{definition}[جيب التمام والجيب]\label{def:g11:trigo:cossin}
من أجل كل \omterm{def:g10:numbers:sets}{عدد حقيقي} $t$، يكون \emph{جيب التمام}\index{جيب التمام} و
\emph{الجيب}\index{جيب} للعدد $t$ هما إحداثيَي النقطة $M(t)$ من
دائرة الوحدة:
\[
M(t) = (\cos t,\ \sin t).
\]
\end{definition}

$I$، يحط عند النقطة $M(t)$ التي \emph{تعرّف} إحداثياها $\cos t$

\begin{proposition}[الخواص الأولى]\label{prop:g11:trigo:first}
من أجل كل $t \in \R$ و $k \in \Z$:
\[
-1 \leq \cos t \leq 1, \qquad
-1 \leq \sin t \leq 1, \qquad
\cos^2 t + \sin^2 t = 1,
\]
\[
\cos(t + 2k\pi) = \cos t, \qquad \sin(t + 2k\pi) = \sin t .
\]
\end{proposition}

\begin{proof}
تقع $M(t)$ على دائرة الوحدة، إذن يقع إحداثياها بين $-1$ و
$1$، وهما يحققان \omterm{def:g10:algebra:equation}{معادلة} الدائرة $x^2 + y^2 = 1$، وهي
المتطابقة $\cos^2 t + \sin^2 t = 1$. أما الدورية فتعيد صياغة
$M(t + 2k\pi) = M(t)$ (\cref{def:g11:trigo:winding}).
\end{proof}

\begin{example}\label{ex:g11:trigo:identity}
إذا كان $\sin t = \frac35$ و $t \in \intcc{\frac\pi2}{\pi}$ (الربع
الثاني)، فإن $\cos^2 t = 1 - \frac{9}{25} = \frac{16}{25}$، إذن
$\cos t = \pm\frac45$؛ وفي الربع الثاني تكون \omterm{def:g10:coordgeom:system}{الفاصلة} سالبة،
ومنه $\cos t = -\frac45$.
\end{example}

والقيم الواجب حفظها عن ظهر قلب:
\begin{center}
\begin{tabular}{c|ccccc}
$t$ & $0$ & $\dfrac{\pi}{6}$ & $\dfrac{\pi}{4}$ & $\dfrac{\pi}{3}$ & $\dfrac{\pi}{2}$\\[6pt]
\hline\rule{0pt}{14pt}%
$\cos t$ & $1$ & $\dfrac{\sqrt3}{2}$ & $\dfrac{\sqrt2}{2}$ & $\dfrac12$ & $0$\\[6pt]
$\sin t$ & $0$ & $\dfrac12$ & $\dfrac{\sqrt2}{2}$ & $\dfrac{\sqrt3}{2}$ & $1$
\end{tabular}
\end{center}

\begin{remark}
وسيلة للحفظ: تُقرأ الجيوب
$\frac{\sqrt0}{2}, \frac{\sqrt1}{2}, \frac{\sqrt2}{2}, \frac{\sqrt3}{2},
\frac{\sqrt4}{2}$، وجيوب التمام هي القائمة نفسها معكوسة.
\end{remark}

\section{الزوايا المرتبطة}

\begin{proposition}[الزوايا المرتبطة]\label{prop:g11:trigo:associated}
\index{زوايا مرتبطة}
من أجل كل $t \in \R$:
\[
\begin{aligned}
\cos(-t) &= \cos t, & \sin(-t) &= -\sin t,\\
\cos(\pi - t) &= -\cos t, & \sin(\pi - t) &= \sin t,\\
\cos(\pi + t) &= -\cos t, & \sin(\pi + t) &= -\sin t,\\
\cos\left(\tfrac{\pi}{2} - t\right) &= \sin t, &
\sin\left(\tfrac{\pi}{2} - t\right) &= \cos t .
\end{aligned}
\]
\end{proposition}

\begin{proof}
يعبّر كل سطر عن تناظر من تناظرات الدائرة.

فالنقطة $M(-t)$ هي نظيرة $M(t)$ بالنسبة إلى المحور $x$: \omterm{def:g10:coordgeom:system}{فالفاصلة}
لا تتغير، \omterm{def:g10:coordgeom:system}{والترتيب} يغيّر إشارته.

والنقطة $M(\pi - t)$ هي نظيرة $M(t)$ بالنسبة إلى المحور $y$: فالسير
إلى الوراء من $\pi$ بمقدار $t$ يحط في مقابل (أفقيًا) السير
إلى الأمام من $0$ بمقدار $t$. \omterm{def:g10:coordgeom:system}{فالفاصلة} تغيّر إشارتها، \omterm{def:g10:coordgeom:system}{والترتيب}
لا يتغير.

والنقطة $M(\pi + t)$ تقابل $M(t)$ قطريًا (نصف دورة): فيغيّر
الإحداثيان إشارتيهما.

والنقطة $M\!\left(\frac\pi2 - t\right)$ هي نظيرة $M(t)$ بالنسبة إلى
المستقيم القطري $y = x$، الذي يبدّل الإحداثيين.
\end{proof}

\begin{example}
$\cos\frac{2\pi}{3} = \cos\left(\pi - \frac\pi3\right)
= -\cos\frac\pi3 = -\frac12$، و
$\sin\frac{7\pi}{6} = \sin\left(\pi + \frac\pi6\right)
= -\sin\frac\pi6 = -\frac12$. وجدول القيم الخمس، موسَّعًا
بالتناظرات، يغطي الدائرة كلها.
\end{example}

\section{حل المعادلات المثلثية}

\begin{theorem}[المعادلتان $\cos t = \cos a$ و $\sin t = \sin a$]
\label{thm:g11:trigo:equations}
من أجل عددين حقيقيين $t$ و $a$:
\[
\cos t = \cos a \iff t = a + 2k\pi \ \text{أو}\ t = -a + 2k\pi
\quad (k \in \Z),
\]
\[
\sin t = \sin a \iff t = a + 2k\pi \ \text{أو}\ t = \pi - a + 2k\pi
\quad (k \in \Z).
\]
\end{theorem}

\begin{proof}
تتساوى فاصلتا نقطتين من دائرة الوحدة بالضبط عندما تكونان
متساويتين أو نظيرتين إحداهما للأخرى بالنسبة إلى المحور $x$؛ وحسب
\cref{prop:g11:trigo:associated}، نظيرة $M(a)$ هي $M(-a)$. إذن
$\cos t = \cos a$ يعني $M(t) = M(a)$ أو $M(t) = M(-a)$، أي
$t = \pm a$ إلى حد مضاعف للعدد $2\pi$. وبالمثل، تساوي الترتيبين يعني
تساوي النقطتين أو تناظرهما بالنسبة إلى المحور $y$، ونظيرة $M(a)$
هي $M(\pi - a)$.
\end{proof}

\begin{method}[حل $\cos t = c$ على فترة]\label{met:g11:trigo:solve}
\begin{enumerate}
  \item أوجد زاوية \emph{واحدة} $a$ تحقق $\cos a = c$، من جدول
        القيم المعروفة.
  \item اكتب الحلول العامة $t = \pm a + 2k\pi$
        (\cref{thm:g11:trigo:equations}).
  \item اختر \omterm{def:g10:numbers:sets}{الأعداد الصحيحة} $k$ التي تحط داخل \omterm{def:g10:numbers:interval}{الفترة} المطلوبة،
        واذكر الحلول.
\end{enumerate}
وافعل الشيء نفسه من أجل $\sin t = c$ مع $t = a + 2k\pi$ أو
$t = \pi - a + 2k\pi$.
\end{method}

\begin{example}\label{ex:g11:trigo:solve}
حل $\cos t = \frac12$ على $\intoc{-\pi}{\pi}$. أحد حلول
$\cos a = \frac12$ هو $a = \frac{\pi}{3}$. والحلول العامة هي
$t = \frac\pi3 + 2k\pi$ و $t = -\frac\pi3 + 2k\pi$. وداخل
$\intoc{-\pi}{\pi}$، لا يسهم سوى $k = 0$:
$t \in \left\{-\frac\pi3,\ \frac\pi3\right\}$.

وحل $\sin t = -\frac{\sqrt2}{2}$ على $\intco{0}{2\pi}$. إحدى الزوايا هي
$a = -\frac\pi4$؛ والحلول هي $t = -\frac\pi4 + 2k\pi$ و
$t = \pi + \frac\pi4 + 2k\pi = \frac{5\pi}{4} + 2k\pi$. وفي
$\intco{0}{2\pi}$: $t = \frac{7\pi}{4}$ (من $k = 1$) و
$t = \frac{5\pi}{4}$.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g11:trigo:1}
حوّل إلى \omterm{def:g11:trigo:radian}{الراديان}: $30^\circ$ و $45^\circ$ و $120^\circ$ و $270^\circ$.
وحوّل إلى الدرجات: $\frac{\pi}{5}$ و $\frac{5\pi}{6}$ و $\frac{7\pi}{4}$
و $\frac{2\pi}{9}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:trigo:2}
علّم على دائرة الوحدة النقط $M(t)$ من أجل
\[
t = \frac{2\pi}{3},\quad
t = -\frac{\pi}{4},\quad
t = \frac{17\pi}{6},\quad
t = -\frac{7\pi}{2},
\]
بعد ردّ كل قيمة إلى قيمة في $\intoc{-\pi}{\pi}$ بترديد $2\pi$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:trigo:3}
مستعملًا الجدول \omterm{prop:g11:trigo:associated}{والزوايا المرتبطة}، أعطِ القيم المضبوطة للمقادير
\[
\cos\frac{3\pi}{4}, \quad
\sin\frac{5\pi}{6}, \quad
\cos\left(-\frac{\pi}{3}\right), \quad
\sin\frac{4\pi}{3}, \quad
\cos\frac{11\pi}{6} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:trigo:4}
علمًا أن $\cos t = \frac{5}{13}$ حيث $t \in \intoo{-\frac\pi2}{0}$، احسب
$\sin t$ بالضبط.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:5}
بسّط، من أجل أي \omterm{def:g10:numbers:sets}{عدد حقيقي} $x$:
\[
A = \cos(\pi + x) + \cos(-x) + \sin\left(\frac\pi2 - x\right) - \cos x,
\qquad
B = \sin(\pi - x) + \sin(\pi + x) + \sin(-x) .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:6}
حل على $\intoc{-\pi}{\pi}$:
\[
\cos t = \frac{\sqrt3}{2}, \qquad
\sin t = \frac12, \qquad
\cos t = -1 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:7}
حل على $\intco{0}{2\pi}$:
\[
\sin t = -\frac{\sqrt3}{2}, \qquad
\cos t = 0, \qquad
2\sin t + 1 = 0 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:8}
حل على $\R$ \omterm{def:g10:algebra:equation}{المعادلة} $\cos 2t = \cos t$. (استعمل
\cref{thm:g11:trigo:equations} مع $a = t$، وناقش عائلتَي
الحلول.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:9}
بيّن أنه من أجل كل \omterm{def:g10:numbers:sets}{عدد حقيقي} $x$،
\[
\bigl(\cos x + \sin x\bigr)^2 + \bigl(\cos x - \sin x\bigr)^2 = 2 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:10}
تتدحرج عجلة نصف قطرها $30$ سم دون انزلاق. بأي زاوية (\omterm{def:g11:trigo:radian}{بالراديان}
ثم بالدرجات) تدور عندما تتقدم الدراجة $1.5$ م؟
وكم تتقدم الدراجة خلال دورة كاملة للعجلة؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:trigo:11}
حل على $\intoc{-\pi}{\pi}$ المتراجحة
$\cos t \leq \frac12$. (ارسم دائرة الوحدة، وعلّم المنطقة التي تكون فيها
\omterm{def:g10:coordgeom:system}{الفاصلة} لا تتجاوز $\frac12$، واقرأ \omterm{def:g10:numbers:interval}{فترة} قيم
$t$.)
\end{exercise}

\section{مسألة: العجلة الدوارة والمدّ}

\begin{problem}[{مسألة نهاية الأسبوع --- الراديان مسافةً مدحرجة،
ودائرة الوحدة ساعةً رئيسية لكل ظاهرة
دورية}]
\label{pb:g11:trigo:1}
تحملك عجلة دوارة حول دائرة؛ ويحمل المدّ ماء
الميناء صعودًا وهبوطًا على الدائرة الرياضية نفسها. فكل
ظاهرة دورية --- العجلات، والمدّ، والصوت، والفصول --- تقرأ
جدولها على دائرة وحدة هذا الفصل
(\cref{def:g11:trigo:cossin})، وتخبر معادلات
\cref{met:g11:trigo:solve} بالأوقات بالضبط. تحوّل هذه المسألة، وتلفّ، وتنمذج، وتحل ---
وتحسم في الطريق لماذا
يقيس الرياضيون الزوايا \omterm{def:g11:trigo:radian}{بالراديان}.

\medskip
\noindent\textbf{الجزء الأول --- \omterm{def:g11:trigo:radian}{الراديان}، المسافة المدحرجة.}
\begin{enumerate}
  \item حوّل إلى \omterm{def:g11:trigo:radian}{الراديان}: $30^\circ$ و $135^\circ$ و
        $300^\circ$؛ وإلى الدرجات: $\frac{3\pi}{4}$ و
        $\frac{5\pi}{6}$ (\cref{met:g11:trigo:convert}).
  \item جوهر \omterm{def:g11:trigo:radian}{الراديان}: القوس ذو الزاوية $t$ (\omterm{def:g11:trigo:radian}{بالراديان})
        على دائرة نصف قطرها $r$ طوله بالضبط
        $r\,t$. فأي قوس تقطعه زاوية قدرها $2$ \omterm{def:g11:trigo:radian}{راديان} على
        دائرة نصف قطرها $3$ م؟ (فلا $\pi$ في أي مكان --- وهذا هو
        المقصود.)
  \item تتدحرج عجلة نصف قطرها $30$ سم مسافة $1$ كم دون
        انزلاق. فبكم \emph{راديانًا} دارت،
        وكم دورة كاملة يمثّل ذلك؟
  \item علّم على دائرة الوحدة واحسب بالضبط:
        $\cos\frac{2\pi}{3}$ و $\sin\left(-\frac{\pi}{4}\right)$
        و $\cos\frac{19\pi}{6}$ (ورُدّ بترديد $2\pi$ أولًا؛
        \cref{prop:g11:trigo:associated}).
  \item زاوية $t \in \intoo{\frac\pi2}{\pi}$ تحقق
        $\sin t = \frac35$. مستعملًا
        $\cos^2 t + \sin^2 t = 1$ والربع، أوجد
        $\cos t$ و $\tan t$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- العجلة الدوارة.}
لعجلة عملاقة نصف قطر $60$ م، ومحورها على ارتفاع $65$ م عن
الأرض، وتدور دورة كاملة كل $30$ دقيقة. تصعد عند
أخفض نقطة في اللحظة $t = 0$ (بالدقائق)، فيخضع ارتفاعك للعلاقة
\[
h(t) = 65 - 60 \cos\!\left(\frac{\pi t}{15}\right).
\]
\begin{enumerate}[resume]
  \item برّر العبارة: تحقق من الزاوية المقطوعة بعد $t$
        دقيقة، ومن الارتفاع عند $t = 0$، وفسّر إشارة الطرح.
  \item احسب ارتفاعك عند $t = 7.5$ و $t = 15$ و
        $t = 20$ دقيقة (بقيم مضبوطة).
  \item في أي لحظة تكون \emph{لأول مرة} على ارتفاع $95$ م؟ (حل
        $h(t) = 95$ مستعملًا \cref{met:g11:trigo:solve}.)
  \item خلال أي \omterm{def:g10:numbers:interval}{فترة} زمنية من الجولة تكون على ارتفاع $95$ م
        على الأقل --- وكم يدوم ذلك إجمالًا؟ (قارن
        \cref{exo:g11:trigo:11}.)
  \item أين يتغير الارتفاع \emph{أسرع} ما يكون؟ قارن
        الصعود خلال الدقيقة الأولى
        ($h(1) - h(0)$) بالصعود بين الدقيقتين $7$ و
        $8$، وصُغ الملاحظة (أما الأداة التي تجعلها
        مضبوطة --- \omterm{def:g11:deriv:derivative}{مشتقة} الجيبية --- فتصل
        في السنة الأخيرة).
  \item دور المهندس: صمّم عبارة ارتفاع عجلة
        ارتفاعها الأعظمي $40$ م، وارتفاعها الأصغري $4$ م، ودورها
        $12$ دقيقة، والصعود من الأسفل عند
        $t = 0$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- المدّ وتناظرات
الدائرة.}
في ميناء، يكون عمق الماء بالمتر، بعد $t$ ساعة من
\omterm{prop:g10:coordgeom:midpoint}{منتصف} الليل،
\[
d(t) = 7 + 3 \sin\!\left(\frac{\pi t}{6}\right).
\]
\begin{enumerate}[resume]
  \item أعطِ العمقين الأعظمي والأصغري، واللحظتين اللتين يحدثان
        فيهما أول مرة، ودور المدّ.
  \item تحتاج سفينة شحن إلى $8.5$ م من الماء. حل
        $d(t) \geq 8.5$ على $\intco{0}{12}$: في أي نافذة
        يمكنها الدخول، وكم تدوم النافذة؟ ومتى تُفتح
        النافذة التالية؟
  \item استنتج من جديد متطابقتين من متطابقات \omterm{prop:g11:trigo:associated}{الزوايا المرتبطة} انطلاقًا
        من تناظرات الدائرة مباشرة --- $\sin(\pi - t) = \sin t$ (المقابلة
        بالنسبة إلى المحور الشاقولي) و $\cos(\pi + t) = -\cos t$
        (نصف دورة) --- وبيّن المتطابقة التكميلية
        $\cos\left(\frac\pi2 - t\right) = \sin t$.
  \item \omterm{def:g11:func:parity}{الزوجية} (بمفردات \cref{pb:g11:func:1}): صنّف
        $\sin$ و $\cos$ و $t \mapsto \sin t + t^3$ \omterm{def:g11:func:parity}{زوجية} أو
        \omterm{def:g11:func:parity}{فردية}، مع براهين من سطر واحد انطلاقًا من الدائرة.
  \item حل على $\intco{0}{2\pi}$:
        $2\sin^2 t - \sin t - 1 = 0$. (وهي \omterm{def:g10:algebra:equation}{معادلة} تربيعية
        متنكرة: حلّلها إلى عوامل كما في \cref{pb:g11:quad:1}، ثم
        اقرأ الدائرة.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- زاوية الرياضي.}
\begin{enumerate}[resume]
  \item كم درجة يساوي \omterm{def:g11:trigo:radian}{راديان} واحد؟ ثم فخّ الحاسبة:
        أيهما أكبر، $\sin(1)$ (في نمط \omterm{def:g11:trigo:radian}{الراديان}) أم
        $\sin(1^\circ)$ --- وبأي معامل \omterm{def:g10:numbers:approx}{تقريبًا}؟ وما المغزى
        بشأن ضبط الحاسبة؟
  \item حل $\cos t = \sin t$ على $\intco{0}{2\pi}$، مستعملًا
        الدائرة (فأين تتساوى \omterm{def:g10:coordgeom:system}{الفاصلة} \omterm{def:g10:coordgeom:system}{والترتيب}؟).
  \item من أجل الزوايا الصغيرة، يكاد القوس وجيبه يتطابقان:
        قارن $\sin(0.1)$ بالعدد $0.1$ (بخمسة أرقام عشرية)، و
        أعطِ الخطأ النسبي. وهذا ``\omterm{def:g10:numbers:approx}{تقريب} الزاوية
        الصغيرة'' يوجّه السفن والنواسات؛ أما المبرهنة
        وراءه ($\frac{\sin t}{t} \to 1$) فنجمة في فصل
        النهايات في السنة الأخيرة.
  \item الخاتمة --- دائرة الوحدة ساعةً رئيسية، جملة واحدة
        لكل بند: \omterm{def:g11:trigo:radian}{الراديان} يحوّل الزوايا إلى أعداد أمينة (فالقوس
        $=$ نصف القطر $\times$ الزاوية)؛ \omterm{def:g11:trigo:cossin}{وجيب التمام} \omterm{def:g11:trigo:cossin}{والجيب} هما ظل
        عقرب الساعة وارتفاعه؛ وتناظرات الدائرة
        هي المتطابقات؛ وحل $\cos t = c$ يقرأ الأوقات
        على الساعة؛ وكل ظاهرة دورية --- عجلة،
        ومدّ، وصوت --- هي هذه الساعة في ثوب.
\end{enumerate}
\end{problem}
