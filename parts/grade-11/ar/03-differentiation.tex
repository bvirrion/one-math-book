\chapter{الاشتقاق: مدخل أول}\label{ch:g11:deriv}

بأي سرعة تتغير \omterm{def:g11:func:function}{دالة}؟ الجواب عند نقطة واحدة هو
\emph{\omterm{def:g11:deriv:derivative}{المشتقة}}، وهي أنفع عدد مفرد في الرياضيات
التطبيقية كلها. يبني هذا الفصل \omterm{def:g11:deriv:derivative}{المشتقة} انطلاقًا من معدلات التغير المتوسطة،
ويحسبها من أجل \omterm{def:g11:func:reference}{الدوال المرجعية}، ويستعمل إشارتها لقراءة
تغيرات \omterm{def:g11:func:function}{الدالة}. وكل ما هنا يُوسَّع ويُعمَّق في
\cref{ch:g12:deriv}.

\section{معدل التغير المتوسط}

\begin{definition}[معدل التغير المتوسط]\label{def:g11:deriv:rate}
لتكن $f$ معرَّفة على \omterm{def:g10:numbers:interval}{فترة} $I$، وليكن $a \neq b$ فيها. \emph{معدل
التغير المتوسط}\index{معدل التغير المتوسط} \omterm{def:g11:func:function}{للدالة} $f$ بين
$a$ و $b$ هو
\[
\frac{f(b) - f(a)}{b - a}.
\]
وهو \omterm{def:g10:reffunc:affine}{ميل} \emph{القاطع}\index{قاطع} المار بالنقطتين
$\bigl(a, f(a)\bigr)$ و $\bigl(b, f(b)\bigr)$ من \omterm{def:g10:functions:graph}{المنحنى}.
\end{definition}

\begin{example}
تقطع سيارة $140$ كم بين الساعة 2 والساعة 4 بعد الظهر: فسرعتها المتوسطة
$\frac{140}{2} = 70$ كم/سا. أما عداد السرعة فيعرض
سرعة \emph{لحظية} في كل لحظة --- وذلك بالضبط هو المفهوم
الذي تضبطه \omterm{def:g11:deriv:derivative}{المشتقة}.
\end{example}

\section{المشتقة عند نقطة}

\begin{definition}[المشتقة عند نقطة]\label{def:g11:deriv:derivative}
لتكن $f$ معرَّفة على \omterm{def:g10:numbers:interval}{فترة} تحوي $a$. ومن أجل $h \neq 0$، كوّن
\omterm{def:g11:deriv:rate}{معدل التغير المتوسط} بين $a$ و $a + h$:
\[
r(h) = \frac{f(a+h) - f(a)}{h}.
\]
فإذا اقترب $r(h)$ من عدد ثابت وحيد كلما اقترب $h$ من
$0$ بأي قدر، قلنا إن $f$ \emph{قابلة للاشتقاق عند $a$}\index{مشتقة}\index{قابلية الاشتقاق}؛
وهذا العدد هو \emph{مشتقة $f$ عند $a$}، ويُكتب $f'(a)$:
\[
f'(a) = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}.
\]
\end{definition}

\begin{example}\label{ex:g11:deriv:atpoint}
لتكن $f(x) = x^2$ و $a = 1$. من أجل $h \neq 0$:
\[
\frac{f(1+h) - f(1)}{h}
= \frac{(1+h)^2 - 1}{h}
= \frac{1 + 2h + h^2 - 1}{h}
= \frac{h(2 + h)}{h}
= 2 + h .
\]
وكلما اقترب $h$ من $0$ اقترب هذا من $2$: إذن \omterm{def:g11:func:function}{الدالة} $x^2$
قابلة للاشتقاق عند $1$ و $f'(1) = 2$. ولاحظ الاستراتيجية: \emph{بسّط
الخارج حتى لا يعود $h$ يظهر في مقام}، ثم اجعل
$h \to 0$.
\end{example}

\begin{example}[نقطة غير قابلة للاشتقاق]
لتكن $f(x) = \abs{x}$ و $a = 0$. عندئذٍ
$\frac{f(h) - f(0)}{h} = \frac{\abs h}{h}$، وهو يساوي $1$ من أجل $h > 0$
و $-1$ من أجل $h < 0$: فلا يُقترب من عدد وحيد، و $\abs x$ ليست
قابلة للاشتقاق عند $0$. ولمنحنيها \emph{زاوية} هناك.
\end{example}

\section{المماس}

\begin{definition}[المماس]\label{def:g11:deriv:tangent}
إذا كانت $f$ قابلة للاشتقاق عند $a$، فإن \emph{المماس}\index{مماس}
\omterm{def:g10:functions:graph}{للمنحنى} عند النقطة $A = \bigl(a, f(a)\bigr)$ هو المستقيم المار بالنقطة $A$ الذي
ميله $f'(a)$. ومعادلته
\[
y = f(a) + f'(a)(x - a).
\]
\end{definition}

\omterm{def:g11:deriv:tangent}{والمماس} هو الوضع الحدي للقواطع المارة بالنقطة $A$: فكلما
$h \to 0$، انزلقت النقطة الثانية $\bigl(a+h, f(a+h)\bigr)$ على
\omterm{def:g10:functions:graph}{المنحنى} نحو $A$، واقترب \omterm{def:g10:reffunc:affine}{ميل} \omterm{def:g11:deriv:rate}{القاطع} $\frac{f(a+h)-f(a)}{h}$
من $f'(a)$.

من أجل $f(x) = x^2$ عند $a = 1$: لدينا $f(1) = 1$ و $f'(1) = 2$
(\cref{ex:g11:deriv:atpoint})، إذن \omterm{def:g11:deriv:tangent}{المماس} هو
\[
y = 1 + 2(x - 1) = 2x - 1 .
\]
وهو المستقيم $d_2$ الذي وُجد بالجبر الخالص في \cref{exo:g11:quad:11}.
\end{example}

\section{مشتقات الدوال المرجعية}

\begin{proposition}\label{prop:g11:deriv:refderivatives}
على مجموعات قابلية اشتقاقها:
\[
(c)' = 0, \quad
(x)' = 1, \quad
(x^2)' = 2x, \quad
(x^3)' = 3x^2, \quad
\left(\frac1x\right)' = -\frac{1}{x^2}, \quad
(\sqrt x\,)' = \frac{1}{2\sqrt x} \ (x > 0).
\]
وبوجه أعم، $(x^n)' = n\,x^{n-1}$ من أجل كل \omterm{def:g10:numbers:sets}{عدد صحيح} $n \geq 1$.
\end{proposition}

\begin{proof}
يتبع كل برهان استراتيجية \cref{ex:g11:deriv:atpoint}: بسّط
المعدل، ثم اجعل $h \to 0$. ثبّت $a$ في \omterm{def:g11:func:function}{مجموعة التعريف}.

\emph{الثابتة والمطابقة.} $\frac{c - c}{h} = 0$ و
$\frac{(a+h) - a}{h} = 1$ من أجل كل $h$: فالنهايتان $0$ و $1$.

\emph{المربع.} $\frac{(a+h)^2 - a^2}{h} = \frac{2ah + h^2}{h} = 2a + h
\to 2a$.

\emph{التكعيب.} $(a+h)^3 = a^3 + 3a^2h + 3ah^2 + h^3$، إذن المعدل هو
$3a^2 + 3ah + h^2 \to 3a^2$.

\emph{المقلوب.} من أجل $a \neq 0$ و $h$ صغير بما يكفي ليكون
$a + h \neq 0$:
\[
\frac{1}{h}\left(\frac{1}{a+h} - \frac{1}{a}\right)
= \frac{1}{h} \cdot \frac{a - (a+h)}{a(a+h)}
= \frac{-1}{a(a+h)}
\xrightarrow[h \to 0]{} -\frac{1}{a^2}.
\]

\emph{\omterm{def:g11:quad:discriminant}{الجذر} التربيعي.} من أجل $a > 0$، اضرب في المرافق:
\[
\frac{\sqrt{a+h} - \sqrt a}{h}
= \frac{(a + h) - a}{h\,(\sqrt{a+h} + \sqrt a)}
= \frac{1}{\sqrt{a+h} + \sqrt a}
\xrightarrow[h \to 0]{} \frac{1}{2\sqrt a}.
\]

\emph{القوة العامة.} يستمر النمط $2a$، $3a^2$: فبنشر
$(a+h)^n$، تحمل كل الحدود بعد $a^n + n\,a^{n-1}h$ العامل $h^2$، إذن يؤول
المعدل إلى $n\,a^{n-1}$. (ويُعطى برهان كامل بالتراجع في
\cref{ch:g12:deriv}.)
\end{proof}

\section{قواعد الاشتقاق}

\begin{theorem}[العمليات]\label{thm:g11:deriv:rules}
لتكن $u$ و $v$ قابلتين للاشتقاق على $I$ وليكن $\lambda \in \R$. عندئذٍ تكون
$u + v$ و $\lambda u$ و $uv$ قابلة للاشتقاق على $I$، وكذلك
$\frac{u}{v}$ حيثما $v \neq 0$، ولدينا
\[
(u+v)' = u' + v', \qquad
(\lambda u)' = \lambda u', \qquad
(uv)' = u'v + uv', \qquad
\left(\frac{u}{v}\right)' = \frac{u'v - uv'}{v^2}.
\]
\end{theorem}

\begin{proof}[برهان القواعد الثلاث الأولى]
ثبّت $a \in I$ واختصر $\Delta u = u(a+h) - u(a)$ و
$\Delta v = v(a+h) - v(a)$.

\emph{المجموع.} معدل $u + v$ هو
$\frac{\Delta u + \Delta v}{h} = \frac{\Delta u}{h} + \frac{\Delta v}{h}
\to u'(a) + v'(a)$.

\emph{الضرب في عدد.} معدل $\lambda u$ هو
$\lambda \frac{\Delta u}{h} \to \lambda u'(a)$.

\emph{الجداء.} أضف واطرح الحد المختلط $u(a+h)\,v(a)$:
\begin{align*}
\frac{u(a+h)v(a+h) - u(a)v(a)}{h}
&= \frac{u(a+h)v(a+h) - u(a+h)v(a) + u(a+h)v(a) - u(a)v(a)}{h}\\
&= u(a+h)\,\frac{\Delta v}{h} + v(a)\,\frac{\Delta u}{h}.
\end{align*}
وكلما $h \to 0$: $\frac{\Delta v}{h} \to v'(a)$، و $\frac{\Delta u}{h} \to
u'(a)$، و $u(a+h) \to u(a)$ (فمعدلها له نهاية، إذن
$\Delta u = h \cdot \frac{\Delta u}{h} \to 0$). ومنه يؤول المعدل
إلى $u(a)v'(a) + v(a)u'(a)$.

أما قاعدة الخارج فتُبرهن على المنوال نفسه؛ و\emph{نقبلها في هذا
المستوى}، ويُبرهن عليها في \cref{ch:g12:deriv}.
\end{proof}

\begin{example}\label{ex:g11:deriv:computations}
\emph{كثير حدود.} $f(x) = 3x^3 - 5x^2 + 7x - 2$:
\[
f'(x) = 3 \times 3x^2 - 5 \times 2x + 7 = 9x^2 - 10x + 7 .
\]

\emph{خارج قسمة.} $g(x) = \frac{2x+1}{x^2+1}$: بوضع $u = 2x + 1$ و $v = x^2
+ 1$،
\[
g'(x) = \frac{2(x^2+1) - (2x+1)\,2x}{(x^2+1)^2}
= \frac{-2x^2 - 2x + 2}{(x^2+1)^2}.
\]

\emph{جداء.} $k(x) = x\sqrt x$ من أجل $x > 0$:
$k'(x) = 1 \times \sqrt x + x \times \frac{1}{2\sqrt x}
= \sqrt x + \frac{\sqrt x}{2} = \frac{3\sqrt x}{2}$.
\end{example}

\section{إشارة المشتقة والتغيرات}

\begin{theorem}[المشتقة والتغيرات]\label{thm:g11:deriv:variations}
لتكن $f$ قابلة للاشتقاق على \omterm{def:g10:numbers:interval}{فترة} $I$.
\begin{enumerate}
  \item إذا كان $f' \geq 0$ على $I$، فإن $f$ \omterm{def:g11:func:monotone}{متزايدة} على $I$.
  \item وإذا كان $f' \leq 0$ على $I$، فإن $f$ \omterm{def:g10:functions:variations}{متناقصة} على $I$.
  \item وإذا كان $f' > 0$ على $I$ (عدا، ربما، عند عدد منته من النقط التي
        تنعدم عندها)، فإن $f$ \omterm{def:g11:func:monotone}{متزايدة} تمامًا على $I$.
\end{enumerate}
\end{theorem}

\begin{proof}
الحدس واضح --- \omterm{def:g10:functions:graph}{فالمنحنى} الذي تشير كل مماساته إلى أعلى لا بد أن
يتسلق --- لكن البرهان الصارم يحتاج إلى مبرهنة القيمة المتوسطة. والنتيجة
\emph{نقبلها في هذا المستوى}؛ انظر \cref{thm:g12:deriv:variations}.
\end{proof}

\begin{proposition}[القيم الحدية]\label{prop:g11:deriv:extrema}
إذا تغيّرت إشارة $f'$ عند $a$ من $+$ إلى $-$، كان \omterm{def:g11:func:function}{للدالة} $f$ \omterm{def:g10:functions:extrema}{قيمة عظمى} محلية
عند $a$؛ وإذا تغيّرت من $-$ إلى $+$، كانت لها \omterm{def:g10:functions:extrema}{قيمة صغرى} محلية. وعند نقطة كهذه يكون $f'(a) = 0$ و
\omterm{def:g11:deriv:tangent}{المماس} أفقيًا.
\end{proposition}

\begin{proof}
إذا كان $f' \geq 0$ على \omterm{def:g10:numbers:interval}{فترة} تنتهي عند $a$ و $f' \leq 0$ على \omterm{def:g10:numbers:interval}{فترة}
تبدأ عند $a$، فإن $f$ تتزايد حتى $f(a)$ وتتناقص بعدها
(\cref{thm:g11:deriv:variations}): أي أن $f(a)$ أكبر ما يكون في الجوار. وبما أن إشارة $f'$
تتغير عند $a$ وهي معرَّفة هناك، فإن $f'(a) = 0$.
\end{proof}

\begin{method}[دراسة دالة]\label{met:g11:deriv:study}
\begin{enumerate}
  \item احسب $f'(x)$ وبسّطه --- ويُفضَّل أن يكون على صورة محلَّلة إلى عوامل.
  \item حدّد إشارة $f'(x)$ على كل \omterm{def:g10:numbers:interval}{فترة} (وإذا كان $f'$
        تربيعيًا، استعمل \cref{thm:g11:quad:sign}).
  \item سجّل النتائج في \emph{جدول تغيرات}: سطر
        لإشارة $f'$، وسطر من الأسهم من أجل $f$، مع قيم $f$ عند
        نقط الانعطاف.
  \item استنتج: القيم الحدية، وعدد حلول $f(x) = k$، والمتراجحات.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g11:deriv:study}
ادرس $f(x) = x^3 - 3x + 1$ على $\R$. أولًا،
$f'(x) = 3x^2 - 3 = 3(x-1)(x+1)$: وهو مقدار تربيعي جذراه $-1$ و $1$،
وهو موجب خارجهما. إذن $f$ \omterm{def:g11:func:monotone}{متزايدة} على $\intoc{-\infty}{-1}$،
\omterm{def:g10:functions:variations}{ومتناقصة} على $\intcc{-1}{1}$، \omterm{def:g11:func:monotone}{ومتزايدة} على $\intco{1}{+\infty}$، ولها
\omterm{def:g10:functions:extrema}{قيمة عظمى} محلية $f(-1) = -1 + 3 + 1 = 3$ \omterm{def:g10:functions:extrema}{وقيمة صغرى} محلية
$f(1) = 1 - 3 + 1 = -1$.
\end{example}

\begin{example}[الاستمثال]\label{ex:g11:deriv:box}
يُصنع صندوق مفتوح من صفيحة $12 \times 12$ سم بقطع مربع
ضلعه $x$ من كل زاوية ثم الطي. وحجمه
$V(x) = x(12 - 2x)^2$ من أجل $0 < x < 6$. وبالنشر،
$V(x) = 4x^3 - 48x^2 + 144x$، إذن
\[
V'(x) = 12x^2 - 96x + 144 = 12(x^2 - 8x + 12) = 12(x - 2)(x - 6).
\]
وعلى $\intoo{0}{6}$ يكون العامل $x - 6$ سالبًا، إذن يحمل $V'(x)$ إشارة
$-(x-2)$: فهو موجب قبل $2$ وسالب بعدها. إذن يكون الحجم أعظميًا
من أجل $x = 2$ سم، فيعطي $V(2) = 2 \times 8^2 = 128$ سم$^3$.
\end{example}

تُبنى حظيرة مستطيلة مسنودة إلى جدار باستعمال $60$ م من السياج
(ثلاثة أضلاع). عبّر عن المساحة بدلالة العرض $x$، واستعمل
\omterm{def:g11:deriv:derivative}{المشتقة} لإيجاد بعدَي المساحة العظمى. (قارن بطريقة
الرأس في \cref{ch:g11:quad}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:deriv:9}
بيّن أنه من أجل كل $x > 0$، $\sqrt{x} \leq \frac{x + 1}{2}$، بدراسة
\omterm{def:g11:func:function}{الدالة} $f(x) = \frac{x+1}{2} - \sqrt x$ على $\intoo{0}{+\infty}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:deriv:10}
يجب أن تسع علبة أسطوانية $500$ سم$^3$. ومساحة سطحها (قرصان زائد
الجانب) هي $S = 2\pi r^2 + 2\pi r h$ حيث $\pi r^2 h = 500$. عبّر عن $S$
بدلالة $r$ وحده، وبيّن أن $S'(r) = 4\pi r -
\frac{1000}{r^2}$. واستنتج نصف القطر الذي يصغّر المساحة، على صورة
عبارة مضبوطة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:deriv:11}
كم مماسًا \omterm{def:g11:quad:parabola}{للقطع المكافئ} $y = x^2$ يمر بالنقطة الخارجية
$P(1, -3)$؟ حدّد معادلاتها. (ليكن $a$ \omterm{def:g10:coordgeom:system}{فاصلة}
نقطة التماس وعبّر عن كون \omterm{def:g11:deriv:tangent}{المماس} عند $a$ يمر بالنقطة $P$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:deriv:12}
لتكن $f(x) = x^3 - 3x + 1$ (انظر \cref{ex:g11:deriv:study}). مستعملًا
\omterm{def:g10:functions:table}{جدول التغيرات}، حدّد عدد حلول \omterm{def:g10:algebra:equation}{المعادلة}
$f(x) = k$ بحسب قيمة \omterm{def:g10:numbers:sets}{العدد الحقيقي} $k$.
\end{exercise}

\section{مسألة: وظائف المماس الثلاث}

\begin{problem}[{مسألة نهاية الأسبوع --- برهان المرآة المكافئة،
وآلة نيوتن لإيجاد الجذور، وأقوى عارضة في الجذع}]
\label{pb:g11:deriv:1}
ترسم \omterm{def:g11:deriv:derivative}{المشتقة} مماسات؛ ويبدو ذلك موهبة متواضعة.
تبيّن هذه المسألة أن \omterm{def:g11:deriv:tangent}{المماس} يشغل ثلاث وظائف دفعة واحدة: فهو
يبرهن على مبرهنة المصباح الأمامي المعلَّقة في
\cref{pb:g11:quad:1} (كل شعاع مواز لمحور \omterm{def:g11:quad:parabola}{قطع مكافئ}
ينعكس عبر \omterm{pb:g11:quad:1}{البؤرة})، وهو يشغّل أسرع خوارزمية
لحل المعادلات في الاستعمال الشائع --- داخل كل
حاسبة وكل شبكة عصبونية --- وهو يجد أقوى عارضة
مستطيلة يمكن أن يعطيها جذع مستدير.

\medskip
\noindent\textbf{الجزء الأول --- الطلاقة في المماسات.}
\begin{enumerate}
  \item اشتق $f(x) = 3x^2 - 5x + 1$ و
        $g(x) = x^3 - 12x$
        (\cref{thm:g11:deriv:rules})؛ واستعد $\left(x^2
        \right)'$ عند $x = 1$ انطلاقًا من التعريف (معدل
        التغير، ثم $h \to 0$).
  \item أوجد \omterm{def:g11:deriv:tangent}{المماس} \omterm{def:g10:functions:graph}{للمنحنى} $y = x^3 - 12x$ عند $a = 2$. ما
        الخاص فيه، وعلام يدل ذلك
        (\cref{prop:g11:deriv:extrema})؟
  \item أرسِ \omterm{def:g11:deriv:tangent}{المماس} العام \omterm{def:g11:quad:parabola}{للقطع المكافئ} $y = x^2$
        عند النقطة $(a, a^2)$:
        \[
        y = 2ax - a^2 .
        \]
  \item أين يقطع ذلك \omterm{def:g11:deriv:tangent}{المماس} المحور $y$؟ استنتج
        وصفة الرسّام: لرسم \omterm{def:g11:deriv:tangent}{المماس} عند نقطة $P$
        ارتفاعها $h$ على \omterm{def:g11:quad:parabola}{القطع المكافئ}، صِل $P$ بنقطة
        المحور الواقعة على عمق\,\dots\ تحت المبدأ. صُغها.
  \item ابنِ \omterm{def:g10:functions:table}{جدول التغيرات} الكامل \omterm{def:g11:func:function}{للدالة}
        $g(x) = x^3 - 12x$ (إشارة $g'$، والقيم الحدية المحلية و
        قيمها، \cref{met:g11:deriv:study}).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- مبرهنة المرآة.}
لتكن $P(a, a^2)$ نقطة من \omterm{def:g11:quad:parabola}{القطع المكافئ} $y = x^2$ (وليكن
$a \neq 0$)، ولتكن $F\left(0, \frac14\right)$ \omterm{pb:g11:quad:1}{البؤرة}، ولتكن $T$
النقطة التي يقطع فيها \omterm{def:g11:deriv:tangent}{المماس} عند $P$ المحور $y$ (السؤال
4). وتذكّر من \cref{pb:g11:quad:1} أن
$PF = a^2 + \frac14$.
\begin{enumerate}[resume]
  \item احسب المسافة $FT$.
  \item استنتج أن المثلث $FPT$ متساوي الساقين في $F$.
        فأي زاويتين متساويتان إذن؟
  \item الشعاع الوارد عبر $P$، الموازي للمحور،
        يوازي المستقيم $(TF)$ (فكلاهما شاقولي). ومستعملًا
        الزوايا المتبادلة، بيّن أن \omterm{def:g11:deriv:tangent}{المماس} عند $P$ يصنع
        \emph{زاويتين متساويتين} مع الشعاع الشاقولي الوارد و
        مع القطعة $[PF]$.
  \item الفيزياء: ينعكس الضوء عن \omterm{def:g10:functions:graph}{منحنى} كما ينعكس عن مماسه،
        فيغادر بالزاوية نفسها التي وصل بها (قانون
        الانعكاس). استنتج \emph{مبرهنة المرآة}: كل
        شعاع مواز للمحور ينعكس عبر \omterm{pb:g11:quad:1}{البؤرة} ---
        وبعكس الزمن، تبعث لمبة عند \omterm{pb:g11:quad:1}{البؤرة} ضوءًا
        متوازيًا. فالطبق والمصباح الأمامي في
        \cref{pb:g11:quad:1} صارا الآن مبرهنتين.
  \item تحقق عدديًا من تساوي الساقين عند $a = 1$:
        احسب $T$ و $FP$ و $FT$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- آلة نيوتن.}
لحل $f(x) = 0$، اقترح نيوتن: انطلق من تخمين
$x_0$، واتبع \omterm{def:g11:deriv:tangent}{المماس} عند $x_0$ نزولًا إلى المحور، واجعل
\omterm{def:g10:coordgeom:system}{فاصلة} تقاطعه هي التخمين التالي.
\begin{enumerate}[resume]
  \item استنتج صيغة الطريقة:
        \[
        x_{n + 1} = x_n - \frac{f(x_n)}{f'(x_n)} .
        \]
  \item شغّلها على $f(x) = x^2 - 2$ انطلاقًا من $x_0 = 1$: احسب
        $x_1$ و $x_2$ و $x_3$ على صورة كسور مضبوطة. أي
        متتالية عمرها ألفا سنة --- وأي آلة
        من \cref{pb:g10:functions:1} --- أعدت بناءها للتوّ؟
        برهن على المتطابقة وراء هذا التوافق:
        \[
        x - \frac{x^2 - 2}{2x} = \frac12\left(x +
        \frac2x\right).
        \]
  \item حل $x^3 = 100$ (تجاوز خطتَي الادخار في
        \cref{pb:g10:reffunc:1}): طبّق خطوتَي نيوتن على
        $f(x) = x^3 - 100$ انطلاقًا من $x_0 = 5$ (بأربعة أرقام عشرية)، و
        قارن بقيمة $\sqrt[3]{100}$ على الحاسبة.
  \item التخريب: حاول إطلاق الطريقة على
        $f(x) = x^3 - 12x$ عند $x_0 = 2$. ما الذي يسوء، و
        لماذا (السؤال 2)؟ صُغ التحذير العملي.
  \item في السؤال 12 سارت الأرقام العشرية الصحيحة \omterm{def:g10:numbers:approx}{تقريبًا}
        $1 \to 3 \to 6$. قارن بالتنصيف (أي تنصيف
        \omterm{def:g10:numbers:interval}{فترة} في كل خطوة) وقل في جملة واحدة لماذا
        تعمل الحاسبات ومستقبلات التموضع العالمي وتدريب الشبكات
        العصبونية كلها على فكرة نيوتن.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- أقوى عارضة.}
تُنشر عارضة مستطيلة عرضها $w$ وعمقها $d$ من
جذع مستدير قطره $30$ سم، إذن $w^2 + d^2 = 900$. وصلابة
العارضة تتناسب مع $w d^2$ (فالعمق يُحسب مرتين:
لأن الروافد تقف على حرفها!).
\begin{enumerate}[resume]
  \item عبّر عن الصلابة بدلالة $w$ وحده:
        $S(w) = 900w - w^3$، وعلى أي \omterm{def:g10:numbers:interval}{فترة}؟
  \item اشتق، وابنِ \omterm{def:g10:functions:table}{جدول التغيرات}، وأوجد
        $w$ و $d$ الأمثلين (بقيم مضبوطة، ثم إلى
        المليمتر).
  \item بيّن أن النسب المثلى تحقق
        $d = w\sqrt2$ --- وهي نسبة النجارين القديمة. (وطُرفة
        إضافية: تقسيم قطر الجذع إلى ثلاثة أجزاء متساوية
        ورفع أعمدة إلى الدائرة ينشئ
        هذا المستطيل بالضبط.)
  \item الاختيار الساذج هو العارضة \emph{المربعة}
        ($w = d$). احسب صلابتها وصلابة العارضة
        المثلى، وأعطِ ربح النجار بالنسبة المئوية.
  \item الخاتمة --- وظائف \omterm{def:g11:deriv:tangent}{المماس} الثلاث، جملة لكل وظيفة:
        الهندسة (مبرهنة المرآة)، والحساب العددي (آلة
        نيوتن)، والاستمثال (العارضة)، وكلها يشغّلها
        الكائن نفسه. وإشارة إلى الأمام: تضيف السنة الأخيرة التحدب
        --- أي جهة \omterm{def:g11:deriv:tangent}{المماس} من \omterm{def:g10:functions:graph}{المنحنى} --- لتصقل الثلاثة جميعًا.
\end{enumerate}
\end{problem}
