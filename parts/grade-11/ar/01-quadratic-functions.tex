\chapter{الدوال والمعادلات التربيعية}\label{ch:g11:quad}

الدوال التربيعية أبسط الدوال بعد الخطية، وأول
دوال منحنياتها منحنية حقًّا. يطوّر هذا الفصل
عدتها الكاملة: \omterm{thm:g11:quad:canonical}{الصورة النموذجية}، \omterm{def:g11:quad:discriminant}{والمميز}، وإشارة
العبارة التربيعية، وهندسة \omterm{def:g11:quad:parabola}{القطع المكافئ}.

\section{الصورة النموذجية}

\begin{definition}[الدالة التربيعية]\label{def:g11:quad:quadratic}
\emph{الدالة التربيعية}\index{دالة تربيعية} \omterm{def:g10:functions:function}{دالة} على
الصورة
\[
f(x) = ax^2 + bx + c, \qquad a \neq 0,
\]
حيث $a$ و $b$ و $c$ أعداد حقيقية. وتُسمّى العبارة $ax^2 + bx + c$
أيضًا \emph{ثلاثي حدود من الدرجة الثانية}.
\end{definition}

\begin{theorem}[الصورة النموذجية]\label{thm:g11:quad:canonical}
يمكن كتابة كل \omterm{def:g11:quad:quadratic}{دالة تربيعية} بطريقة وحيدة على الصورة
\[
f(x) = a(x - \alpha)^2 + \beta,
\qquad\text{حيث } \alpha = -\frac{b}{2a} \text{ و } \beta = f(\alpha).
\]
وهذه هي \emph{الصورة النموذجية}\index{صورة نموذجية} \omterm{def:g10:functions:function}{للدالة} $f$.
\end{theorem}

\begin{proof}
\emph{نتمّ المربع}. وبما أن $a \neq 0$، نخرجه عاملًا مشتركًا من الحدين
الأولين:
\[
ax^2 + bx + c = a\left(x^2 + \frac{b}{a}x\right) + c .
\]
والآن يكون $x^2 + \frac{b}{a}x$ بداية المربع
$\left(x + \frac{b}{2a}\right)^2 = x^2 + \frac ba x + \frac{b^2}{4a^2}$،
إذن $x^2 + \frac{b}{a}x = \left(x + \frac{b}{2a}\right)^2 -
\frac{b^2}{4a^2}$. وبالتعويض:
\[
f(x) = a\left(x + \frac{b}{2a}\right)^2 - \frac{b^2}{4a} + c
= a\bigl(x - \alpha\bigr)^2 + \beta,
\]
حيث $\alpha = -\frac{b}{2a}$ و $\beta = c - \frac{b^2}{4a}$. وبالحساب
عند $x = \alpha$ ينعدم الحد المربع، إذن $\beta = f(\alpha)$. أما الوحدانية:
فبنشر $a(x-\alpha)^2 + \beta$ ومطابقة معاملات $x$
يُفرض $\alpha = -\frac{b}{2a}$، ثم يتحدد $\beta$.
\end{proof}

\begin{example}\label{ex:g11:quad:canonical}
لتكن $f(x) = 2x^2 - 8x + 3$. عندئذٍ $\alpha = -\frac{-8}{2 \times 2} = 2$ و
$\beta = f(2) = 8 - 16 + 3 = -5$، إذن
\[
f(x) = 2(x - 2)^2 - 5 .
\]
وتحقق \omterm{def:g10:algebra:expand}{بالنشر}: $2(x^2 - 4x + 4) - 5 = 2x^2 - 8x + 8 - 5 =
2x^2 - 8x + 3$.
\end{example}

\begin{proposition}[القيمة الحدية والتغيرات]\label{prop:g11:quad:variations}
لتكن $f(x) = a(x-\alpha)^2 + \beta$.
\begin{enumerate}
  \item إذا كان $a > 0$، فإن $f$ \omterm{def:g10:functions:variations}{متناقصة} على $\intoc{-\infty}{\alpha}$ و
        \omterm{def:g10:functions:variations}{متزايدة} على $\intco{\alpha}{+\infty}$: أي أن \omterm{def:g10:functions:function}{للدالة} $f$ \emph{\omterm{def:g10:functions:extrema}{قيمة صغرى}}
        هي $\beta$، تُبلغ عند $x = \alpha$.
  \item وإذا كان $a < 0$، انقلبت التغيرات: فيكون \omterm{def:g10:functions:function}{للدالة} $f$ \emph{\omterm{def:g10:functions:extrema}{قيمة عظمى}}
        هي $\beta$ عند $x = \alpha$.
\end{enumerate}
\end{proposition}

\begin{proof}
لنفترض أن $a > 0$ وليكن $\alpha \leq u < v$. عندئذٍ
\[
f(v) - f(u) = a\bigl((v-\alpha)^2 - (u-\alpha)^2\bigr)
= a\,(v - u)\,\bigl((v-\alpha) + (u-\alpha)\bigr).
\]
وكل عامل موجب: إذ $a > 0$، و $v - u > 0$، و
$(v-\alpha) + (u-\alpha) > 0$ لأن $v > \alpha$ و $u \geq \alpha$. إذن
$f(v) > f(u)$: أي أن $f$ \omterm{def:g10:functions:variations}{متزايدة} على $\intco{\alpha}{+\infty}$. ومن أجل
$u < v \leq \alpha$ يكون العامل الأخير سالبًا، إذن $f(v) < f(u)$: أي أن $f$
\omterm{def:g10:functions:variations}{متناقصة} هناك. وأخيرًا $f(x) - f(\alpha) = a(x-\alpha)^2 \geq 0$ من أجل كل
$x$، إذن $\beta = f(\alpha)$ هي \omterm{def:g10:functions:extrema}{القيمة الصغرى}. وتنتج الحالة $a < 0$
بتطبيق ما سبق على $-f$.
\end{proof}

\begin{example}
لدى فلّاح $100$ متر من السياج ليحيط حقلًا مستطيلًا مسنودًا إلى
جدار مستقيم (فلا حاجة إلى سياج على طول الجدار). فإذا كان $x$ العرض،
كانت المساحة المحاطة $A(x) = x(100 - 2x) = -2x^2 + 100x$. وهنا $\alpha =
-\frac{100}{2\times(-2)} = 25$ و $A(25) = 25 \times 50 = 1250$: أي أن المساحة
أكبر ما تكون من أجل عرض قدره $25$ م، فتعطي $1250$ م$^2$.
\end{example}

\section{الجذور والتحليل إلى عوامل}

\begin{definition}[الجذر والمميز]\label{def:g11:quad:discriminant}
\emph{جذر}\index{جذر} ثلاثي الحدود $ax^2+bx+c$ هو حل
\omterm{def:g10:algebra:equation}{المعادلة} $ax^2 + bx + c = 0$. و\emph{المميز}\index{مميز}
لثلاثي الحدود هو العدد
\[
\Delta = b^2 - 4ac .
\]
\end{definition}

\begin{theorem}[حل المعادلة $ax^2+bx+c=0$]\label{thm:g11:quad:roots}
\begin{enumerate}
  \item إذا كان $\Delta > 0$، فللمعادلة جذران متمايزان
        \[
        x_1 = \frac{-b - \sqrt{\Delta}}{2a},
        \qquad
        x_2 = \frac{-b + \sqrt{\Delta}}{2a},
        \]
        و $ax^2+bx+c = a(x - x_1)(x - x_2)$.
  \item وإذا كان $\Delta = 0$، فللمعادلة \omterm{def:g11:quad:discriminant}{جذر} وحيد
        $x_0 = -\frac{b}{2a}$، و $ax^2+bx+c = a(x - x_0)^2$.
  \item وإذا كان $\Delta < 0$، فليس \omterm{def:g10:algebra:equation}{للمعادلة} حل حقيقي، ولا
        يمكن تحليل ثلاثي الحدود إلى عوامل ذات معاملات حقيقية.
\end{enumerate}
\end{theorem}

\begin{proof}
لننطلق من \omterm{thm:g11:quad:canonical}{الصورة النموذجية} المحسوبة في \cref{thm:g11:quad:canonical}:
\[
ax^2+bx+c
= a\left(x + \frac{b}{2a}\right)^2 + c - \frac{b^2}{4a}
= a\left[\left(x + \frac{b}{2a}\right)^2 - \frac{\Delta}{4a^2}\right],
\]
لأن $c - \frac{b^2}{4a} = \frac{4ac - b^2}{4a} = -\frac{\Delta}{4a} =
a \times \left(-\frac{\Delta}{4a^2}\right)$.

\emph{1.} إذا كان $\Delta > 0$، فإن $\frac{\Delta}{4a^2} =
\left(\frac{\sqrt\Delta}{2a}\right)^2$ ويكون القوس فرق
مربعين:
\[
ax^2+bx+c = a\left(x + \frac{b}{2a} - \frac{\sqrt\Delta}{2a}\right)
\left(x + \frac{b}{2a} + \frac{\sqrt\Delta}{2a}\right)
= a(x - x_2)(x - x_1).
\]
والجداء معدوم بالضبط عندما ينعدم أحد عامليه، فنحصل على
الجذرين، وهما متمايزان لأن $\sqrt\Delta \neq 0$.

\emph{2.} وإذا كان $\Delta = 0$، صار القوس
$\left(x + \frac{b}{2a}\right)^2$، وهو لا ينعدم إلا عند
$x_0 = -\frac{b}{2a}$.

\emph{3.} وإذا كان $\Delta < 0$، فإن $-\frac{\Delta}{4a^2} > 0$، إذن يكون القوس
مربعًا زائد عدد موجب: فلا ينعدم أبدًا، ولا يكون \omterm{def:g10:algebra:equation}{للمعادلة}
حل حقيقي. والتحليل $a(x-x_1)(x-x_2)$ سينتج جذورًا، إذن
لا وجود له.
\end{proof}

\begin{example}\label{ex:g11:quad:solving}
حل $2x^2 - 8x + 3 = 0$. وهنا $\Delta = (-8)^2 - 4 \times 2 \times 3 =
64 - 24 = 40 > 0$، إذن يوجد جذران:
\[
x_{1,2} = \frac{8 \pm \sqrt{40}}{4} = \frac{8 \pm 2\sqrt{10}}{4}
= 2 \pm \frac{\sqrt{10}}{2}.
\]
ومن أجل $x^2 + x + 1 = 0$: $\Delta = 1 - 4 = -3 < 0$، فلا حل حقيقي.
\end{example}

\begin{proposition}[مجموع الجذرين وجداؤهما]\label{prop:g11:quad:vieta}
إذا كان $\Delta \geq 0$ وكان $x_1, x_2$ الجذرين (وهما متساويان عندما $\Delta = 0$)،
فإن
\[
x_1 + x_2 = -\frac{b}{a},
\qquad
x_1 x_2 = \frac{c}{a} .
\]
وبالعكس، يكون عددان مجموعهما $s$ وجداؤهما $p$ حلَّي
$x^2 - sx + p = 0$.
\end{proposition}

\begin{proof}
انشر التحليل في \cref{thm:g11:quad:roots}:
\[
a(x - x_1)(x - x_2) = a x^2 - a(x_1 + x_2)x + a\,x_1 x_2 .
\]
ومطابقة المعاملات مع $ax^2 + bx + c$ تعطي
$-a(x_1+x_2) = b$ و $a\,x_1x_2 = c$. أما العكس: فإذا كان $u + v = s$ و
$uv = p$، فإن $(x-u)(x-v) = x^2 - sx + p$، إذن $u$ و $v$ جذراها.
\end{proof}

\begin{example}
أوجد عددين مجموعهما $10$ وجداؤهما $21$. إنهما يحلّان
$x^2 - 10x + 21 = 0$؛ و $\Delta = 100 - 84 = 16$، فالجذران
$\frac{10 \pm 4}{2} = 7$ و $3$. \emph{اقتناص الجذور بسرعة:} بما أن
معاملات $x^2 - 5x + 4$ تحقق $1 - 5 + 4 = 0$، فإن العدد $1$
\omterm{def:g11:quad:discriminant}{جذر}، والآخر هو $\frac{c}{a} = 4$ (فجداؤهما يجب أن يساوي
$\frac ca$).
\end{example}

\section{إشارة ثلاثي حدود من الدرجة الثانية}

\begin{theorem}[إشارة ثلاثي الحدود]\label{thm:g11:quad:sign}
لتكن $f(x) = ax^2+bx+c$ مع $a \neq 0$.
\begin{enumerate}
  \item إذا كان $\Delta > 0$، وكان الجذران $x_1 < x_2$: فإن $f(x)$ يحمل إشارة $a$
        خارج $\intcc{x_1}{x_2}$، وإشارة $-a$ على
        $\intoo{x_1}{x_2}$.
  \item وإذا كان $\Delta = 0$: فإن $f(x)$ يحمل إشارة $a$ من أجل كل $x \neq x_0$،
        وينعدم عند $x_0$.
  \item وإذا كان $\Delta < 0$: فإن $f(x)$ يحمل إشارة $a$ من أجل كل $x$.
\end{enumerate}
وباختصار: \emph{يحمل ثلاثي الحدود إشارة $a$، إلا بين جذريه}.
\end{theorem}

\begin{proof}
\emph{1.} اكتب $f(x) = a(x-x_1)(x-x_2)$ وتابع إشارتَي
العاملين. فإذا كان $x < x_1$: كان $x - x_1$ و $x - x_2$ سالبين، وجداؤهما
موجبًا، فيحمل $f(x)$ إشارة $a$. وإذا كان $x_1 < x < x_2$: كانت
إشارتا العاملين مختلفتين، والجداء سالبًا، فيحمل $f(x)$
إشارة $-a$. وإذا كان $x > x_2$: كان العاملان موجبين ويحمل $f(x)$
إشارة $a$ من جديد.

\emph{2.} لدينا $f(x) = a(x - x_0)^2$ و $(x-x_0)^2 > 0$ من أجل $x \neq x_0$.

\emph{3.} من برهان \cref{thm:g11:quad:roots}، لدينا
$f(x) = a\left[\left(x + \frac{b}{2a}\right)^2 +
\left(-\frac{\Delta}{4a^2}\right)\right]$، والقوس دائمًا
موجب.
\end{proof}

\begin{method}[حل المتراجحات التربيعية]\label{met:g11:quad:inequalities}
لحل $ax^2 + bx + c \leq 0$ (أو $\geq$ أو $<$ أو $>$):
\begin{enumerate}
  \item احسب $\Delta$ والجذرين إن وُجدا؛
  \item ارسم \omterm{def:g10:algebra:signtable}{جدول الإشارة} مستعملًا \cref{thm:g11:quad:sign} (إشارة $a$
        خارج الجذرين)؛
  \item اقرأ \omterm{def:g10:numbers:interval}{فترة} الحلول أو فتراتها، منتبهًا إلى ما إذا كان
        الجذران نفسهما مُدرَجين (في المتراجحة الواسعة) أو لا (في التامة).
\end{enumerate}
\end{method}

\begin{example}\label{ex:g11:quad:inequality}
حل $-x^2 + 3x + 4 \geq 0$. وهنا $\Delta = 9 + 16 = 25$، والجذران
هما $\frac{-3 \pm 5}{-2}$، أي $x_1 = -1$ و $x_2 = 4$. وبما أن
$a = -1 < 0$، يكون ثلاثي الحدود سالبًا خارج الجذرين وموجبًا
بينهما. ومع المتراجحة الواسعة، تكون مجموعة الحلول
$\intcc{-1}{4}$.
\end{example}

\section{القطع المكافئ}

\begin{definition}[القطع المكافئ]\label{def:g11:quad:parabola}
\omterm{def:g10:functions:graph}{منحنى} \omterm{def:g11:quad:quadratic}{الدالة التربيعية} $f(x) = a(x-\alpha)^2+\beta$ هو
\emph{قطع مكافئ}\index{قطع مكافئ} \emph{رأسه} $S(\alpha, \beta)$. وهو
مفتوح نحو الأعلى إذا كان $a > 0$، ونحو الأسفل إذا كان $a < 0$، ومتناظر بالنسبة إلى
المستقيم الشاقولي $x = \alpha$.
\end{definition}

\begin{remark}
التحقق من التناظر سهل: فمن أجل أي $h$،
$f(\alpha + h) = ah^2 + \beta = f(\alpha - h)$، إذن تكون لنقطتين تبعدان
البعد الأفقي نفسه عن المستقيم $x = \alpha$ الارتفاع نفسه.
\end{remark}

\begin{method}[قراءة قطع مكافئ]\label{met:g11:quad:reading}
لرسم $f(x) = ax^2+bx+c$: أوجد الاتجاه (إشارة $a$)، و
الرأس $\left(-\frac{b}{2a},\, f\!\left(-\frac{b}{2a}\right)\right)$، و
نقطة \omterm{def:g10:numbers:interunion}{التقاطع} $(0, c)$ مع المحور $y$، والجذرين إن وُجدا. ثم يضع محور
التناظر $x = -\frac{b}{2a}$ كل نقطة محسوبة مرتين.
\end{method}

\begin{example}
من أجل $f(x) = x^2 - 4x + 3$: مفتوح نحو الأعلى، ورأسه $(2, -1)$، ويقطع المحور $y$
في $(0,3)$، وجذراه $1$ و $3$ (باقتناصهما: $1 - 4 + 3 = 0$). وبالتناظر، تقع
النقطة $(4, 3)$ أيضًا على \omterm{def:g10:functions:graph}{المنحنى}، مقابلةً للنقطة $(0, 3)$.
\end{example}

\[
x^2 - 5x + 6 = 0, \qquad
2x^2 + 3x - 2 = 0, \qquad
x^2 + 2x + 5 = 0, \qquad
9x^2 - 6x + 1 = 0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:quad:2}

\[
f(x) = x^2 - 6x + 11, \qquad
g(x) = -2x^2 + 4x + 1, \qquad
h(x) = 3x^2 + 6x .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:quad:3}
حل المتراجحات
\[
x^2 - 3x - 10 < 0, \qquad
2x^2 + x + 3 > 0, \qquad
-x^2 + 6x - 9 \geq 0 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:quad:4}
دون حساب $\Delta$، أوجد جذرًا بديهيًا، ثم \omterm{def:g11:quad:discriminant}{الجذر} الآخر:
\[
x^2 - 7x + 6 = 0, \qquad
2x^2 + 5x - 7 = 0, \qquad
x^2 + 4x - 5 = 0 .
\]
(إرشاد: اختبر $x = 1$ و $x = -1$، ثم استعمل جداء الجذرين.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:5}
أوجد عددين مجموعهما $14$ وجداؤهما $45$. ثم أوجد
بعدَي مستطيل محيطه $34$ ومساحته $60$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:6}
لتكن $f(x) = x^2 + (m-2)x + 1$، حيث $m$ \omterm{def:g10:stats:median}{وسيط} حقيقي. من أجل أي
قيم $m$ يكون \omterm{def:g10:algebra:equation}{للمعادلة} $f(x) = 0$ حل وحيد؟ ومن أجل أيها لا يكون
لها حل؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:7}
تُقذف كرة إلى أعلى؛ وارتفاعها بعد $t$ ثانية هو
$h(t) = -5t^2 + 20t + 1$ (بالمتر). ما الارتفاع الأعظمي المبلوغ،
وفي أي لحظة؟ وخلال أي \omterm{def:g10:numbers:interval}{فترة} زمنية تكون الكرة على ارتفاع $16$ م
على الأقل؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:8}
حل \omterm{def:g10:algebra:equation}{المعادلة} $x^4 - 13x^2 + 36 = 0$ بوضع $X = x^2$
(وهي \omterm{def:g10:algebra:equation}{معادلة} \emph{ثنائية التربيع}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:9}
من أجل أي قيم $x$ تقع النقطة $M(x, x^2)$ من \omterm{def:g11:quad:parabola}{القطع المكافئ} $y = x^2$
تحت المستقيم $y = x + 2$ تمامًا؟ فسّر على رسم.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:quad:10}
مجموع عدد موجب ومقلوبه هو $\frac{13}{6}$. أوجد
العدد. (ردّ المسألة إلى \omterm{def:g10:algebra:equation}{معادلة} تربيعية.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:quad:11}
ليكن $\mathcal P$ \omterm{def:g11:quad:parabola}{القطع المكافئ} $y = x^2$ وليكن $d_m$ المستقيم
$y = mx - 1$، حيث $m$ \omterm{def:g10:stats:median}{وسيط} حقيقي.
\begin{enumerate}
  \item من أجل أي قيم $m$ يتقاطع $\mathcal P$ و $d_m$ في نقطتين؟
        وفي نقطة واحدة؟ ولا يتقاطعان؟
  \item وعندما يتلامسان في نقطة واحدة بالضبط، احسب نقطة التماس.
        (وفي \cref{ch:g11:deriv} ستتعرّف على $d_m$ مماسًا للقطع
        $\mathcal P$ عند تلك النقطة.)
\end{enumerate}
\end{exercise}

\section{مسألة: نقطة القطع المكافئ السرية}

\begin{problem}[{مسألة نهاية الأسبوع --- البؤرة والدليل: لماذا ترسم
أطباق الأقمار الصناعية والمصابيح الأمامية والجسور المعلقة
المنحنى نفسه، الذي لا يوجد منه في الجوهر سوى واحد}]
\label{pb:g11:quad:1}
يصوّب كل طبق قمر صناعي على كل شرفة مستقبِله نحو نقطة
داخلية واحدة بالضبط؛ ويخبئ كل مصباح أمامي لمبته في
الموضع الخاص نفسه. تلك النقطة --- \emph{\omterm{pb:g11:quad:1}{البؤرة}} --- سر
احتفظ به \omterm{def:g11:quad:parabola}{القطع المكافئ} منذ الهندسة اليونانية، وتكفي
صيغة المسافة لاستخراجه. وفي الطريق، تمرّن هذه
المسألة على \omterm{def:g11:quad:discriminant}{المميز}، وتقذف مقذوفات فوق جدران،
وتعلّق جسرًا، وتبرهن على واقعة ختامية مذهلة: إلى
حد التكبير، \emph{لا يوجد في العالم سوى \omterm{def:g11:quad:parabola}{قطع مكافئ} واحد}.

\medskip
\noindent\textbf{الجزء الأول --- الطلاقة في ثلاثيات الحدود.}
\begin{enumerate}
  \item حل $x^2 - 5x + 6 = 0$، ثم $2x^2 + x - 6 = 0$، ثم
        $x^2 + x + 1 = 0$ (\cref{thm:g11:quad:roots}).
  \item اكتب $2x^2 - 4x - 6$ على صورة محلَّلة إلى عوامل وعلى \omterm{thm:g11:quad:canonical}{الصورة
        النموذجية} (\cref{thm:g11:quad:canonical}). ومع الصورة
        المنشورة، صار بين يديك ثلاثة أثواب \omterm{def:g10:functions:function}{لدالة}
        واحدة: قل أي ملمح يعرضه كل ثوب بنظرة
        (الجذران، والرأس، \omterm{def:g10:reffunc:affine}{والترتيب عند المبدأ}).
  \item حل $-x^2 + 4x - 3 \geq 0$ بقاعدة الإشارة
        (\cref{thm:g11:quad:sign}).
  \item فييت محققًا (\cref{prop:g11:quad:vieta}،
        متابعًا لبابل في \cref{pb:g10:algebra:1}): أوجد عددين
        مجموعهما $7$ وجداؤهما $12$ بكتابة
        معادلتهما. ثم: أحد جذري $x^2 - 5x + c = 0$ يساوي
        $2$؛ أوجد $c$ \omterm{def:g11:quad:discriminant}{والجذر} الآخر دون حل أي
        شيء.
  \item من أجل أي قيم $m$ يكون للعبارة $x^2 + mx + 9 = 0$ \omterm{def:g11:quad:discriminant}{جذر}
        مضاعف؟ ولا \omterm{def:g11:quad:discriminant}{جذر}؟ وجذران؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- كشف \omterm{pb:g11:quad:1}{البؤرة}.}
اعمل على \omterm{def:g11:quad:parabola}{القطع المكافئ} $y = x^2$، والنقطة
$F\left(0, \frac14\right)$ والمستقيم الأفقي $\delta$ ذي
\omterm{def:g10:algebra:equation}{المعادلة} $y = -\frac14$.
\begin{enumerate}[resume]
  \item من أجل نقطة $P(x, x^2)$ من \omterm{def:g11:quad:parabola}{القطع المكافئ}، انشر
        $PF^2 = x^2 + \left(x^2 - \frac14\right)^2$ وبيّن أنه
        مربع كامل $\left(x^2 + \frac14\right)^2$.
        واستنتج $PF$.
  \item احسب المسافة من $P$ إلى المستقيم $\delta$، ثم
        استنتج: \emph{كل نقطة من \omterm{def:g11:quad:parabola}{القطع المكافئ} تبعد عن
        $F$ بقدر بعدها عن $\delta$ بالضبط}. (فالنقطة $F$ هي
        \emph{البؤرة}\index{بؤرة}، و $\delta$ هو
        \emph{الدليل}\index{دليل}.)
  \item برهن على العكس: كل نقطة $P(x, y)$ تحقق
        $PF = \operatorname{dist}(P, \delta)$ يجب أن تحقق
        $y = x^2$. (ربّع المسافتين وبسّط.) إذن
        \omterm{def:g11:quad:parabola}{القطع المكافئ} \emph{محل هندسي}، تمامًا كما كانت الدائرة
        في \cref{pb:g10:coordgeom:1}: تساوي البعد عن نقطة
        وعن مستقيم، بدل تساوي البعد عن نقطة واحدة.
  \item من أجل \omterm{def:g11:quad:parabola}{القطع المكافئ} الأوسع $y = \frac{x^2}{4f}$ (حيث
        $f > 0$)، كيّف السؤال 6 لتبيّن أن \omterm{pb:g11:quad:1}{البؤرة} هي
        $(0, f)$ وأن \omterm{pb:g11:quad:1}{الدليل} هو $y = -f$. وتطبيق: عرض
        طبق قمر صناعي $60$ سم وعمقه $9$ سم --- فحافته
        تمر بالنقطة $(30, 9)$. أوجد $f$: كم يجب أن يعلو
        المستقبِل فوق الرأس؟
  \item سبب عمل الأطباق: تنعكس الأشعة الواردة موازية
        للمحور كلها \emph{عبر \omterm{pb:g11:quad:1}{البؤرة}} (ولمبة عند
        \omterm{pb:g11:quad:1}{البؤرة} تلقي حزمة متوازية --- فالمصابيح الأمامية
        أطباق تعمل بالمقلوب). فسّر لماذا يجعل تساوي البعد
        في السؤال 7 ذلك معقولًا (تخيّل الأشعة جبهات موجية
        تسير)، وقل أي أداة، تأتي في
        \cref{ch:g11:deriv} (انظر \cref{exo:g11:quad:11})، تحوّل
        المعقولية إلى برهان.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- قطوع مكافئة في العمل.}
\begin{enumerate}[resume]
  \item تتبع كرة المسار
        $y = x - \frac{x^2}{20}$ (و $x$ و $y$ بالمتر). أوجد
        أين تحط، والموضع الأفقي لأعلى
        نقطة فيها، والارتفاع الأعظمي
        (\cref{prop:g11:quad:variations}).
  \item على المسار نفسه، يقوم جدار عند $x = 4$.
        احسب ارتفاع الكرة هناك: فهل تتجاوز جدارًا ارتفاعه $3$ م؟
        وجدارًا ارتفاعه $4$ م؟
  \item يمتد قوس مكافئ على $40$ م بارتفاع أعظمي
        $10$ م: وبإحداثيات مركزية،
        $y = 10 - \frac{x^2}{40}$. فهل تمر شاحنة ارتفاعها $7$ م
        على بعد $10$ م من المركز؟ وعلى بعد $15$ م؟
  \item إيراد مسرح عند ثمن بطاقة قدره $p$ يورو هو
        $R(p) = p\,(120 - 2p)$ (فالطلب يتناقص مع ارتفاع الثمن).
        أوجد الثمن الذي يعظّم الإيراد، وأوجد \omterm{def:g10:functions:extrema}{القيمة العظمى}.
  \item كوابل جسر البوابة الذهبية الرئيسية مكافئة إلى
        دقة ممتازة (فحمولة سطح منتظمة تنتج
        قطعًا مكافئًا --- أما السلسلة المعلقة بحرية \emph{فلا} تنتجه:
        فمنحنيها، وهو \omterm{def:g10:functions:graph}{منحنى} التعليق، يحتاج إلى \omterm{def:g10:functions:function}{الدالة} الأسية في السنة الأخيرة).
        وبفتحة $1\,280$ م وتدلٍّ قدره $143$ م، يتبع الكابل
        $y = 143\left(\frac{x}{640}\right)^2$ انطلاقًا من المركز.
        فكم يعلو الكابل فوق أخفض نقطة فيه عند
        $x = 320$ م؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- المستقيمات والمماسات والمفاجأة
الأخيرة.}
\begin{enumerate}[resume]
  \item أكمل \cref{exo:g11:quad:11}: المستقيمات
        $y = mx - 1$ المارة بالنقطة الخارجية $(0, -1)$ تلتقي
        $y = x^2$ بحسب إشارة \omterm{def:g11:quad:discriminant}{مميز}.
        أوجد المماسين الصادرين من $(0, -1)$ ونقطتي
        تماسهما.
  \item والآن المستقيمات $y = mx + 1$ المارة بالنقطة \emph{الداخلية}
        $(0, 1)$: احسب \omterm{def:g11:quad:discriminant}{مميز} \omterm{def:g10:algebra:equation}{معادلة}
        \omterm{def:g10:numbers:interunion}{التقاطع} وبيّن أن كل مستقيم كهذا يقطع
        \omterm{def:g11:quad:parabola}{القطع المكافئ} مرتين. والمغزى: لا مماسات من الداخل ---
        تمامًا كما في الدائرة.
  \item الوتر البؤري: قاطع $y = x^2$ مع
        المستقيم الأفقي المار \omterm{pb:g11:quad:1}{بالبؤرة}، $y = \frac14$. ما
        طول الوتر؟ تحقق من القاعدة العامة (وهي جديرة
        بالحفظ من أجل الأطباق): من أجل $y = \frac{x^2}{4f}$، يكون
        الوتر المار \omterm{pb:g11:quad:1}{بالبؤرة} الموازي للدليل
        طوله $4f$.
  \item المفاجأة الأخيرة: طبّق التكبير
        $(x, y) \mapsto (kx, ky)$ على \omterm{def:g11:quad:parabola}{القطع المكافئ} $y = x^2$ و
        بيّن أن الصورة هي بالضبط $y = \frac{x^2}{k}$. واستنتج:
        كل \omterm{def:g11:quad:parabola}{قطع مكافئ} $y = \frac{x^2}{4f}$ هو \omterm{def:g11:quad:parabola}{قطع مكافئ} الوحدة
        مكبَّرًا --- وخلافًا للدوائر والقطوع الناقصة ذات النسب
        المختلفة، \emph{كل القطوع المكافئة متشابهة}.
  \item الخاتمة: صورة \omterm{def:g11:quad:parabola}{القطع المكافئ} في خمسة أسطر --- \omterm{def:g10:algebra:equation}{معادلة}
        بثلاثة أثواب (السؤال 2)، ومحل هندسي
        (السؤالان 7--8)، وعاكس للمهندس (السؤالان 9
        و 10)، ومسار مقذوف وكابل جسر
        (الأسئلة 11--15)، \omterm{def:g10:functions:graph}{ومنحنى} واحد إلى حد التكبير
        (السؤال 19). جملة واحدة لكل بند.
\end{enumerate}
\end{problem}
