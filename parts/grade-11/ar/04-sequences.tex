\chapter{المتتاليات: مدخل أول}\label{ch:g11:seq}

\omterm{def:g11:seq:sequence}{المتتالية} قائمة أعداد تنتجها قاعدة: الأرصدة المتعاقبة
لحساب توفير، أو أحجام مجتمع سنة بعد سنة. يدرس هذا
الفصل العائلتين المهيمنتين على التطبيقات ---
المتتاليات \emph{\omterm{def:g11:seq:arithmetic}{الحسابية}} التي تنمو بخطوات متساوية، و
المتتاليات \emph{\omterm{def:g11:seq:geometric}{الهندسية}} التي تنمو بنسب متساوية. أما النظرية الصارمة
للنهايات فتُطوَّر في \cref{ch:g12:seq}.

\section{تعريف متتالية}

\begin{definition}[المتتالية]\label{def:g11:seq:sequence}
\emph{المتتالية}\index{متتالية} $(u_n)$ ترفق بكل \omterm{def:g10:numbers:sets}{عدد صحيح}
$n \geq 0$ (أو $n \geq 1$) عددًا حقيقيًا $u_n$، هو \emph{حدها ذو \omterm{pb:g11:quad:1}{الدليل}
$n$}. ويمكن إعطاء متتالية
\begin{itemize}
  \item \emph{صراحةً}، بعبارة تعطي $u_n$ بدلالة $n$: مثلًا
        $u_n = n^2 + 1$؛
  \item أو \emph{بالتراجع}، بحدها الأول وقاعدة للانتقال من كل
        حد إلى الذي يليه: مثلًا $u_0 = 3$ و $u_{n+1} = 2u_n - 1$.
\end{itemize}
\end{definition}

\begin{example}
من أجل $u_n = n^2 + 1$: نجد $u_0 = 1$ و $u_1 = 2$ و $u_2 = 5$ و $u_{10} = 101$
مباشرةً. ومن أجل $u_0 = 3$ و $u_{n+1} = 2u_n - 1$: نجد $u_1 = 5$ و $u_2 = 9$
و $u_3 = 17$ --- فكل حد يحتاج إلى سابقه؛ وبلوغ $u_{10}$ يستلزم
عشر خطوات (أو عبارة عامة، انظر \cref{exo:g11:seq:11}).
\end{example}

\section{المتتاليات الحسابية}

\begin{definition}[المتتالية الحسابية]\label{def:g11:seq:arithmetic}
تكون \omterm{def:g11:seq:sequence}{متتالية} \emph{حسابية}\index{متتالية حسابية} ذات
\emph{أساس}\index{أساس} $d$ إذا كان كل حد يُنتَج من الحد السابق
بإضافة $d$:
\[
u_{n+1} = u_n + d \quad \text{من أجل كل } n.
\]
وبصورة مكافئة: الفرق $u_{n+1} - u_n$ ثابت ويساوي $d$.
\end{definition}

\begin{theorem}[الحد العام]\label{thm:g11:seq:arithgeneral}
إذا كانت $(u_n)$ \omterm{def:g11:seq:arithmetic}{حسابية} حدها الأول $u_0$ وأساسها $d$،
فإن
\[
u_n = u_0 + n\,d \quad \text{من أجل كل } n \geq 0,
\qquad\text{وبوجه أعم } u_n = u_p + (n - p)\,d .
\]
\end{theorem}

\begin{proof}
للانتقال من $u_0$ إلى $u_n$، تُطبَّق القاعدة ``أضف $d$'' $n$ مرة: فخطوة
واحدة تعطي $u_1 = u_0 + d$، وخطوتان تعطيان $u_2 = u_0 + 2d$، وبعد $n$
خطوة يكون كل تطبيق قد أسهم بمقدار $d$ واحد، إذن $u_n = u_0 + nd$.
(وتُجعل عبارة ``وهكذا'' صارمة بالتراجع في \cref{ch:g12:seq}.)
وتنتج الصيغة العامة بعدّ الخطوات $n - p$ من $u_p$ إلى
$u_n$.
\end{proof}

\begin{theorem}[مجموع الأعداد الصحيحة المتتالية]\label{thm:g11:seq:intsum}
من أجل كل \omterm{def:g10:numbers:sets}{عدد صحيح} $n \geq 1$:
\[
1 + 2 + \dots + n = \frac{n(n+1)}{2}.
\]
وبوجه أعم، يساوي مجموع حدود \omterm{def:g11:seq:arithmetic}{متتالية حسابية} متعاقبة
\[
(\text{عدد الحدود}) \times
\frac{\text{الحد الأول} + \text{الحد الأخير}}{2}.
\]
\end{theorem}

\begin{proof}
اكتب المجموع $S$ مرتين، والثانية \omterm{def:g10:coordgeom:system}{بترتيب} معكوس، ثم اجمع
عمودًا عمودًا:
\[
\begin{array}{ccccccccc}
S & = & 1 & + & 2 & + & \dots & + & n\\
S & = & n & + & (n-1) & + & \dots & + & 1\\
\hline
2S & = & (n+1) & + & (n+1) & + & \dots & + & (n+1)
\end{array}
\]
فهناك $n$ عمودًا، ومجموع كل عمود $n + 1$، إذن $2S = n(n+1)$. ومن أجل
\omterm{def:g11:seq:arithmetic}{متتالية حسابية} عامة ينجح الاقتران نفسه: الأول $+$ الأخير $=$
الثاني $+$ ما قبل الأخير $= \dots$، لأن التقدم خطوة إلى الأمام
عند الطرف الأيسر ($+d$) يعوّضه التراجع خطوة عند الطرف الأيمن
($-d$).
\end{proof}

\begin{example}
$1 + 2 + \dots + 100 = \frac{100 \times 101}{2} = 5050$. ومجموع
الأعداد \omterm{def:g11:func:parity}{الفردية} $1 + 3 + \dots + 99$ ($50$ حدًّا) هو
$50 \times \frac{1 + 99}{2} = 2500$.
\end{example}

\section{المتتاليات الهندسية}

\begin{definition}[المتتالية الهندسية]\label{def:g11:seq:geometric}
تكون \omterm{def:g11:seq:sequence}{متتالية} \emph{هندسية}\index{متتالية هندسية} ذات \emph{أساس
هندسي}\index{أساس هندسي} $q \neq 0$ إذا كان كل حد يُنتَج من الحد السابق
بالضرب في $q$:
\[
u_{n+1} = q\,u_n \quad \text{من أجل كل } n.
\]
وبصورة مكافئة، عندما لا ينعدم أي حد: تكون النسبة $\frac{u_{n+1}}{u_n}$
ثابتة وتساوي $q$.
\end{definition}

\begin{theorem}[الحد العام]\label{thm:g11:seq:geomgeneral}
إذا كانت $(u_n)$ \omterm{def:g11:seq:geometric}{هندسية} حدها الأول $u_0$ وأساسها $q$، فإن
\[
u_n = u_0\, q^n \quad \text{من أجل كل } n \geq 0,
\qquad\text{وبوجه أعم } u_n = u_p\, q^{\,n-p} .
\]
\end{theorem}

\begin{proof}
عدّ الخطوات نفسه المستعمل في \cref{thm:g11:seq:arithgeneral}: فمن $u_0$ إلى
$u_n$، تُطبَّق القاعدة ``اضرب في $q$'' $n$ مرة، فتسهم
بالعامل $q^n$.
\end{proof}

\begin{theorem}[المجموع الهندسي]\label{thm:g11:seq:geomsum}
من أجل كل \omterm{def:g10:numbers:sets}{عدد حقيقي} $q \neq 1$ وكل \omterm{def:g10:numbers:sets}{عدد صحيح} $n \geq 0$:
\[
1 + q + q^2 + \dots + q^n = \frac{1 - q^{\,n+1}}{1 - q}.
\]
\end{theorem}

\begin{proof}
ليكن $S = 1 + q + \dots + q^n$. اضرب في $q$:
$qS = q + q^2 + \dots + q^{n+1}$. ثم اطرح:
\[
S - qS = \bigl(1 + q + \dots + q^n\bigr)
- \bigl(q + q^2 + \dots + q^{n+1}\bigr) = 1 - q^{\,n+1},
\]
لأن كل حد \omterm{def:g10:stats:median}{وسيط} يظهر مرة في كل مجموع فيتلاشى.
ومنه $(1 - q)S = 1 - q^{\,n+1}$، وبالقسمة على $1 - q \neq 0$ نحصل على
الصيغة.
\end{proof}

\begin{example}\label{ex:g11:seq:chessboard}
$1 + 2 + 4 + \dots + 2^{10} = \frac{1 - 2^{11}}{1 - 2} = 2^{11} - 1 =
2047$: فمضاعفة حبات الأرز على مربعات رقعة شطرنج تغمر
أي مخزن قبل المربع $64$ بكثير، حيث يكون المجموع
$2^{64} - 1 \approx 1.8 \times 10^{19}$.
\end{example}

\begin{method}[التعرّف على نوع متتالية]\label{met:g11:seq:recognize}
احسب $u_{n+1} - u_n$ وبسّطه. فإذا كانت النتيجة ثابتًا $d$، كانت
\omterm{def:g11:seq:arithmetic}{المتتالية حسابية}. وإلا فاحسب $\frac{u_{n+1}}{u_n}$ (والحدود
غير معدومة) وبسّط: فثابت $q$ يعني أنها \omterm{def:g11:seq:geometric}{هندسية}. وإذا لم يكن أي منهما
ثابتًا، \omterm{def:g11:seq:sequence}{فالمتتالية} ليست من النوعين --- ولا تستنتج أبدًا من
الحدود الأولى وحدها.
\end{method}

\begin{example}
من أجل $u_n = 3 \times 5^n$:
$\frac{u_{n+1}}{u_n} = \frac{3 \times 5^{n+1}}{3 \times 5^n} = 5$ من أجل كل
$n$: فهي \omterm{def:g11:seq:geometric}{هندسية} أساسها $5$. ومن أجل $u_n = n^2$:
لدينا $u_1 - u_0 = 1$ لكن $u_2 - u_1 = 3$، و
$\frac{u_1}{u_0}$ ليست معرَّفة أصلًا --- فليست \omterm{def:g11:seq:arithmetic}{حسابية} ولا
\omterm{def:g11:seq:geometric}{هندسية}.
\end{example}

\section{الرتابة}

\begin{definition}[المتتالية الرتيبة]\label{def:g11:seq:monotone}
تكون \omterm{def:g11:seq:sequence}{المتتالية} $(u_n)$ \emph{\omterm{def:g11:func:monotone}{متزايدة}} إذا كان $u_{n+1} \geq u_n$ من أجل كل $n$،
و\emph{\omterm{def:g10:functions:variations}{متناقصة}} إذا كان $u_{n+1} \leq u_n$ من أجل كل $n$.
\end{definition}

\begin{method}[دراسة الرتابة]\label{met:g11:seq:monotonicity}
ادرس إشارة $u_{n+1} - u_n$. ومن أجل المتتاليات ذات الحدود الموجبة، يمكن
بدلًا من ذلك مقارنة $\frac{u_{n+1}}{u_n}$ بالعدد $1$.
\end{method}

\begin{example}
تكون \omterm{def:g11:seq:arithmetic}{المتتالية الحسابية} \omterm{def:g11:func:monotone}{متزايدة} عندما $d \geq 0$
($u_{n+1} - u_n = d$)، \omterm{def:g10:functions:variations}{ومتناقصة} عندما $d \leq 0$. وتكون \omterm{def:g11:seq:geometric}{المتتالية الهندسية}
ذات $u_0 > 0$ و $q > 1$ \omterm{def:g11:func:monotone}{متزايدة}:
$u_{n+1} - u_n = u_0 q^n (q - 1) > 0$؛ وتكون \omterm{def:g10:functions:variations}{متناقصة} عندما $u_0 > 0$ و $0 < q < 1$.
\end{example}

\section{السلوك على المدى الطويل، بصورة غير رسمية}

ماذا يحدث للحد $u_n$ عندما يصير $n$ كبيرًا جدًّا؟ من أجل \omterm{def:g11:seq:arithmetic}{متتالية حسابية}
أساسها $d > 0$، تتجاوز الحدود $u_0 + nd$ أي عدد ثابت
في النهاية. ومن أجل \omterm{def:g11:seq:geometric}{متتالية هندسية} أساسها $0 < q < 1$، تنكمش الحدود
$u_0 q^n$ نحو $0$: فالضرب المتكرر في $0.9$ مثلًا يفتّ
أي قيمة انطلاق. ومن أجل $q > 1$ تنفجر الحدود، كما في
\cref{ex:g11:seq:chessboard}.

\begin{remark}
يمكن جعل هذه العبارات دقيقة تمامًا --- ``تبقى الحدود في النهاية
ضمن أي مسافة معطاة من $0$'' --- والبرهان عليها. وتلك هي نظرية
\emph{النهايات}، وهي الموضوع الافتتاحي في \cref{ch:g12:seq}.
\end{remark}

\section{مسألة: برج براهما وأرانب فيبوناتشي}

\begin{problem}[{مسألة نهاية الأسبوع --- علاقتا تراجع أسطوريتان:
البرج الذي ينهي العالم، والمتتالية التي تنمو مثل الذهب،
والحيلة المساعدة التي تروّض القروض}]
\label{pb:g11:seq:1}
تحكم متتاليتان فولكلور الرياضيات. إحداهما تعدّ
حركات برج براهما --- أربعة وستون قرصًا ذهبيًا يقول
الأسطورة إن نقلها سينهي العالم. والأخرى تعدّ
أرانب فيبوناتشي وتخبئ النسبة الذهبية. وليست أي منهما
\omterm{def:g11:seq:arithmetic}{حسابية} ولا \omterm{def:g11:seq:geometric}{هندسية} --- وكلتاهما تستسلم لأسلحة هذا الفصل:
علاقات التراجع، والمجاميع \omterm{def:g11:seq:geometric}{الهندسية}
(\cref{thm:g11:seq:geomsum})، وحيلة \omterm{def:g11:seq:sequence}{المتتالية} المساعدة في
\cref{exo:g11:seq:11}، التي تحسب قرضك العقاري أيضًا.

\medskip
\noindent\textbf{الجزء الأول --- برج براهما.}
الأحجية: $n$ قرصًا \omterm{def:g10:functions:variations}{متناقصة} الحجم مكدسة على الوتد أ؛
انقل الكومة كلها إلى الوتد ج، قرصًا واحدًا في كل مرة، دون وضع
قرص أكبر فوق قرص أصغر أبدًا (ويمكن للوتد ب أن يساعد). وليكن $h_n$
عدد الحركات الأصغري.
\begin{enumerate}
  \item العب (بقطع نقدية) وسجّل $h_1$ و $h_2$ و $h_3$.
  \item فسّر الاستراتيجية وراء علاقة التراجع
        $h_{n+1} = 2h_n + 1$: ماذا يجب أن يحدث قبل تحرك
        القرص الأكبر وبعده؟
  \item حل علاقة التراجع بحيلة
        \cref{exo:g11:seq:11}: ضع $v_n = h_n + 1$، وبيّن أن
        $(v_n)$ \omterm{def:g11:seq:geometric}{هندسية}، واستنتج $h_n = 2^n - 1$.
  \item لبرج الأسطورة $64$ قرصًا، وينقل الرهبان
        قرصًا واحدًا في الثانية. مستعملًا $2^{10} = 1024 \approx 10^3$،
        قدّر زمن النقل بالسنوات (والسنة نحو
        $3 \times 10^7$ ثانية؛ قارن
        \cref{ex:g11:seq:chessboard}، وهو المارد نفسه في قصة
        أخرى). فهل ينبغي أن نقلق؟
  \item لماذا لا تستطيع أي استراتيجية أن تتفوق على $2^n - 1$ حركة؟ برهن
        على أن \emph{أي} حل يخضع للمتراجحة
        $h_{n+1} \geq 2 h_n + 1$: ماذا يجب أن يصح بشأن الأقراص
        $n$ العليا قبل تحرك القرص السفلي وبعده مباشرة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- فيبوناتشي.}
نعرّف $F_1 = F_2 = 1$ و $F_{n+2} = F_{n+1} + F_n$ (فكل حد
مجموع الحدين اللذين قبله --- وهي قاعدة عدّ الأوزان في
الكتاب السابق، مع اسمها الأوروبي الآن).
\begin{enumerate}[resume]
  \item اذكر $F_1$ حتى $F_{12}$.
  \item بيّن أن $(F_n)$ ليست \omterm{def:g11:seq:arithmetic}{حسابية} ولا \omterm{def:g11:seq:geometric}{هندسية}،
        لكنها \omterm{def:g11:func:monotone}{متزايدة} تمامًا ابتداءً من $n = 2$
        (\cref{met:g11:seq:monotonicity} وعلاقة التراجع).
  \item برهن على متطابقة المجموع
        \[
        F_1 + F_2 + \dots + F_n = F_{n+2} - 1
        \]
        بالتلسكوب: اكتب كل $F_k$ على صورة
        $F_{k+2} - F_{k+1}$ وراقب المجموع ينهار. وتحقق
        منها من أجل $n = 6$.
  \item برهن على متطابقة المربعات
        $F_1^2 + F_2^2 + \dots + F_n^2 = F_n F_{n+1}$،
        بالتلسكوب مع
        $F_k F_{k+1} - F_{k-1} F_k = F_k^2$. وتحقق من أجل
        $n = 4$. (وصورةً: مربعات أضلاعها
        $1, 1, 2, 3, 5, \dots$ تبلّط مستطيلًا --- وهو هيكل
        حلزون فيبوناتشي الشهير.)
  \item تقول متطابقة كاسيني إن
        $F_{n+1} F_{n-1} - F_n^2 = (-1)^n$. تحقق منها من أجل
        $n = 4, 5, 6$ --- وتعرّف على محرك حيلة
        المربع المتلاشي في مسألة المساحات في الكتاب السابق.
  \item بيّن انطلاقًا من علاقة التراجع أن
        $F_{n+2} \geq 2 F_n$: \omterm{def:g11:seq:sequence}{فمتتالية} فيبوناتشي تتضاعف على الأقل كل
        خطوتين --- أي أنها تنمو بسرعة لا تقل عن سرعة \omterm{def:g11:seq:geometric}{متتالية
        هندسية} أساسها $\sqrt2$.
  \item احسب النسب $r_n = \frac{F_{n+1}}{F_n}$ من أجل
        $n = 3$ إلى $10$ (بثلاثة أرقام عشرية). وبقبول أنها
        تستقر على نهاية $L$، مرّر العلاقة
        $r_{n+1} = 1 + \frac{1}{r_n}$ إلى النهاية وحل:
        أي عدد من \cref{pb:g10:algebra:1} تعبده الأرانب؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الحيلة المساعدة، في
المصرف.}
\begin{enumerate}[resume]
  \item عمّم \cref{exo:g11:seq:11}: من أجل
        $u_{n+1} = a\,u_n + b$ حيث $a \neq 1$، ضع
        $\ell = \frac{b}{1 - a}$ (وهي \omterm{pb:g10:functions:1}{النقطة الثابتة}). بيّن أن
        $v_n = u_n - \ell$ \omterm{def:g11:seq:geometric}{هندسية} أساسها $a$، واستنتج
        $u_n = a^n (u_0 - \ell) + \ell$.
  \item قرض: $10\,000$ يورو بفائدة $1\,\%$ شهريًا،
        يُسدَّد بمبلغ $300$ يورو شهريًا، فيخضع الدين للعلاقة
        $d_{n+1} = 1.01\,d_n - 300$. طبّق السؤال 13 (\omterm{pb:g10:functions:1}{والنقطة
        الثابتة} أولًا!) للحصول على عبارة صريحة للحد $d_n$.
  \item بالحاسبة، أوجد أول شهر يُسدَّد فيه الدين
        كاملًا، والمبلغ الكلي المسدَّد. فكم كلّف
        الاقتراض نفسه؟
  \item مدينة عدد سكانها $50\,000$ تنمو بنسبة $2\,\%$ سنويًا
        وتستقبل $1\,000$ وافد فوق ذلك:
        $p_{n+1} = 1.02\,p_n + 1000$. أعطِ العبارة الصريحة
        وعدد السكان بعد $10$ سنوات.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- العائلتان الملكيتان.}
\begin{enumerate}[resume]
  \item احسب $1 + 2 + 3 + \dots + 1000$
        (\cref{thm:g11:seq:intsum} --- وهو مجموع غاوس الصغير في
        الكتاب السابق، رسميًا الآن)، و
        $1 + 2 + 4 + \dots + 2^{19}$
        (\cref{thm:g11:seq:geomsum}).
  \item احسب مجموع \omterm{def:g11:seq:arithmetic}{المتتالية الحسابية}
        $7, 12, 17, \dots, 502$ (وكم عدد حدودها؟).
  \item خطة ادخار: $100$ يورو تُودَع كل شهر، وتربح
        $0.5\,\%$ شهريًا؛ وبعد الإيداع $n$ يكون الرصيد
        $100\left(1.005^{n-1} + \dots + 1.005 + 1\right)$.
        احسب الرصيد بعد $5$ سنوات ($n = 60$).
  \item الخاتمة --- عدة مروّض المتتاليات: الوصف الصريح في مواجهة
        الوصف بالتراجع؛ والعائلتان الملكيتان و
        صيغتا مجموعهما؛ \omterm{def:g11:seq:sequence}{والمتتالية} المساعدة التي تحوّل
        علاقات التراجع التآلفية إلى \omterm{def:g11:seq:geometric}{هندسية}؛ وفيبوناتشي،
        وهو أول المواطنين خارج العائلتين، مروَّضًا اليوم
        بالمتطابقات ومنتظرًا المصفوفات (في السنة الأخيرة) والنهايات
        لأسره الكامل. جملة واحدة لكل بند.
\end{enumerate}
\end{problem}
