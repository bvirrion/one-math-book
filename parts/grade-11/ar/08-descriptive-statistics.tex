\chapter{الإحصاء الوصفي}\label{ch:g11:stat}

يضغط الإحصاء قائمة من الأعداد في بضعة ملخصات ذات معنى:
أين تستقر المعطيات (\omterm{def:g10:stats:mean}{المتوسط}، \omterm{def:g11:stat:median}{والوسيط})، وكم هي متبددة (الربيعيات،
\omterm{def:g11:stat:variance}{والتباين}، \omterm{def:g11:stat:variance}{والانحراف المعياري}). ويشتغل هذا الفصل كله على مثال
واحد متواصل. وتظهر الملخصات نفسها من جديد مع المتغيرات العشوائية في
\cref{ch:g11:prob}، وقوانينها الاحتمالية في \cref{ch:g12:sums}.

وفي الفصل كله، معطياتنا المتواصلة هي علامات (من $20$)
قسم فيه $12$ تلميذًا:
\[
8,\ 9,\ 10,\ 10,\ 11,\ 12,\ 12,\ 12,\ 13,\ 14,\ 16,\ 17 .
\]

\section{المتوسط والوسيط}

\begin{definition}[المتوسط]\label{def:g11:stat:mean}
\emph{المتوسط الحسابي}\index{متوسط حسابي} للقيم $x_1, x_2, \dots, x_n$ هو
\[
\bar x = \frac{x_1 + x_2 + \dots + x_n}{n}.
\]
وإذا ظهرت القيمة $x_i$ \omterm{def:g10:stats:series}{بتواتر} $n_i$ (حيث
$n_1 + \dots + n_k = n$)، فإن
$\bar x = \frac{n_1 x_1 + \dots + n_k x_k}{n}$.
\end{definition}

\begin{definition}[الوسيط]\label{def:g11:stat:median}
\emph{الوسيط}\index{وسيط} لسلسلة مرتَّبة هو قيمة تقسمها
نصفين: فنصف القيم على الأقل $\leq$ منها، ونصفها على الأقل
$\geq$ منها. وعمليًا، رتّب القيم $n$؛ فإذا كان $n$ فرديًا، فالوسيط هو
القيمة الوسطى؛ وإذا كان $n$ زوجيًا، فخذ \omterm{prop:g10:coordgeom:midpoint}{منتصف} القيمتين
الوسطيين.
\end{definition}

\begin{example}\label{ex:g11:stat:meanmedian}
في معطياتنا المتواصلة، المجموع هو $144$، إذن $\bar x = \frac{144}{12} =
12$. والقيمتان الوسطيان (رقم $6$ ورقم $7$ في القائمة المرتَّبة) كلتاهما
$12$، \omterm{def:g11:stat:median}{فالوسيط} هو $12$ أيضًا. وليس من الضروري أن يتطابق \omterm{def:g10:stats:mean}{المتوسط} \omterm{def:g11:stat:median}{والوسيط}:
\omterm{def:g11:stat:median}{فالوسيط} يتجاهل \emph{كم تبعد} القيم المتطرفة، أما \omterm{def:g10:stats:mean}{المتوسط} فلا.
واستبدال $20$ بالعلامة العليا $17$ ينقل \omterm{def:g10:stats:mean}{المتوسط} إلى $12.25$ لكنه يترك
\omterm{def:g11:stat:median}{الوسيط} عند $12$.
\end{example}

\section{الربيعيات}

\begin{definition}[الربيعيات والمدى الربيعي]\label{def:g11:stat:quartiles}
\emph{الربيعي الأول}\index{ربيعي} $Q_1$ هو أصغر قيمة
يكون عندها ربع المعطيات على الأقل $\leq Q_1$؛ و\emph{الربيعي
الثالث} $Q_3$ هو أصغر قيمة يكون عندها ثلاثة أرباع
المعطيات على الأقل $\leq Q_3$. و\emph{المدى الربيعي}\index{مدى ربيعي} هو $Q_3 - Q_1$: أي
عرض النصف المركزي من المعطيات.
\end{definition}

\begin{example}
$n = 12$: الربع هو $3$، إذن $Q_1$ هي القيمة المرتَّبة رقم $3$:
$Q_1 = 10$. وثلاثة الأرباع هي $9$، إذن $Q_3$ هي القيمة رقم $9$:
$Q_3 = 13$. \omterm{def:g11:stat:quartiles}{والمدى الربيعي} هو $3$: فالنصف المركزي من القسم
يقع ضمن $3$ علامات.
\end{example}

\section{التباين والانحراف المعياري}

\begin{definition}[التباين والانحراف المعياري]\label{def:g11:stat:variance}
\emph{تباين}\index{تباين} السلسلة هو \omterm{def:g10:stats:mean}{متوسط}
مربعات الانحرافات عن \omterm{def:g10:stats:mean}{المتوسط}:
\[
V = \frac{1}{n}\sum_{i=1}^{n} (x_i - \bar x)^2,
\]
و\emph{الانحراف المعياري}\index{انحراف معياري} هو
$\sigma = \sqrt{V}$، ويُعبَّر عنه بوحدة المعطيات نفسها.
\end{definition}

\begin{proposition}[الصيغة المختصرة]\label{prop:g11:stat:shortcut}
\omterm{def:g11:stat:variance}{التباين} هو \omterm{def:g10:stats:mean}{متوسط} المربعات ناقص مربع \omterm{def:g10:stats:mean}{المتوسط}:
\[
V = \frac{1}{n}\sum_{i=1}^{n} x_i^2 \;-\; \bar x^{\,2}.
\]
\end{proposition}

\begin{proof}
انشر كل انحراف مربع:
$(x_i - \bar x)^2 = x_i^2 - 2\bar x\, x_i + \bar x^2$. وبالجمع
والقسمة على $n$:
\[
V = \frac1n \sum x_i^2 - 2\bar x \cdot \underbrace{\frac1n \sum
x_i}_{=\ \bar x} + \frac1n \cdot n\bar x^2
= \frac1n\sum x_i^2 - 2\bar x^2 + \bar x^2
= \frac1n\sum x_i^2 - \bar x^2 .
\]
والمتطابقة نفسها، من أجل المتغيرات العشوائية، هي
\cref{prop:g11:prob:konig}.
\end{proof}

\begin{example}\label{ex:g11:stat:variance}
في معطياتنا المتواصلة، الانحرافات عن $\bar x = 12$ هي $-4$
و $-3$ و $-2$ و $-2$ و $-1$ و $0$ و $0$ و $0$ و $1$ و $2$ و $4$ و $5$؛ ومجموع
مربعاتها
\[
16 + 9 + 4 + 4 + 1 + 0 + 0 + 0 + 1 + 4 + 16 + 25 = 80 .
\]
ومنه
\[
V = \frac{80}{12} = \frac{20}{3} \approx 6.67,
\qquad
\sigma = \sqrt{20/3} \approx 2.6 \text{ علامة}.
\]
وبعبارة تقريبية، تبتعد العلامة النموذجية نحو $2.6$ علامة عن
\omterm{def:g10:stats:mean}{المتوسط}.
\end{example}

\begin{proposition}[تغيير الوحدات]\label{prop:g11:stat:affine}
إذا حُوّلت كل قيمة بالعلاقة $y_i = a x_i + b$، فإن
\[
\bar y = a \bar x + b,
\qquad
V_y = a^2\, V_x,
\qquad
\sigma_y = \abs{a}\,\sigma_x .
\]
\end{proposition}

\begin{proof}
$\bar y = \frac1n \sum (a x_i + b) = a\left(\frac1n\sum x_i\right) +
\frac1n \cdot nb = a\bar x + b$. ثم
$y_i - \bar y = a(x_i - \bar x)$: فكل انحراف يُضرب في $a$
(وينمحي الانسحاب $b$)، إذن كل انحراف مربع يُضرب في $a^2$،
وكذلك متوسطها. وبأخذ الجذور التربيعية نحصل على
$\sigma_y = \abs a\,\sigma_x$.
\end{proof}

\begin{example}
درجات حرارة مسجَّلة بالدرجة السيلزية متوسطها $20$ وانحرافها
المعياري $3$ تُحوَّل إلى الفهرنهايتية بالعلاقة $F = 1.8\,C + 32$: فيصير \omterm{def:g10:stats:mean}{المتوسط}
$1.8 \times 20 + 32 = 68$، ويصير \omterm{def:g11:stat:variance}{الانحراف المعياري}
$1.8 \times 3 = 5.4$. فانسحاب سلسلة لا يغيّر تبددها؛
أما ضربها في معامل فيضرب التبدد في المعامل نفسه.
\end{example}

\begin{method}[تلخيص السلاسل ومقارنتها]\label{met:g11:stat:compare}
لمقارنة سلسلتين، ضع جنبًا إلى جنب: \omterm{def:g10:stats:mean}{المتوسط} (أو \omterm{def:g11:stat:median}{الوسيط}) من أجل
\emph{الموضع}، \omterm{def:g11:stat:variance}{والانحراف المعياري} (أو \omterm{def:g11:stat:quartiles}{المدى الربيعي}) من أجل
\emph{التبدد}. والزوج (\omterm{def:g11:stat:median}{الوسيط}، \omterm{def:g11:stat:quartiles}{المدى الربيعي}) صامد أمام
القيم المتطرفة؛ أما الزوج (\omterm{def:g10:stats:mean}{المتوسط}، \omterm{def:g11:stat:variance}{الانحراف المعياري}) فيستعمل كل قيمة
ويغذّي نظرية الاحتمالات.
\end{method}

\begin{example}
راميان يرمي كل منهما $10$ سهام؛ \omterm{def:g10:stats:mean}{ومتوسط} نقاطهما $8.5$ لكليهما، لكن
الانحرافين المعياريين $0.5$ و $2.1$. المستوى نفسه في \omterm{def:g10:stats:mean}{المتوسط} --- لكن
الرامي الأول أكثر انتظامًا بكثير.
\end{example}

\section{مسألة: الدرجة المعيارية، مسطرة كونية}

\begin{problem}[{مسألة نهاية الأسبوع --- الانحرافات المعيارية تقارن
التفاح بالبرتقال، ومتراجحة تشيبيشيف تحوّل $\sigma$ إلى
ضمانة، والمتوسط يفوز بميداليته}]
\label{pb:g11:stat:1}
هل $15/20$ في امتحان صعب أفضل من $13/20$ في امتحان سهل؟ وهل
$23$ في قائمة من الأعداد الصغيرة صدفة أم قيمة شاذة؟ لا تقدر
الأعداد الخام على الجواب --- لكنها إذا قُسمت على \omterm{def:g11:stat:variance}{الانحراف المعياري}
المناسب نطقت كلها بلغة واحدة، هي \emph{\omterm{pb:g11:stat:1}{الدرجة المعيارية}}. وتتقن هذه
المسألة آلة \omterm{def:g11:stat:variance}{التباين}
(\cref{def:g11:stat:variance} و
\cref{prop:g11:stat:shortcut})، وتبني المسطرة الكونية، ثم
تبرهن على المتراجحة اللافتة التي تحوّل $\sigma$ من عدد
وصفي إلى ضمانة.

\medskip
\noindent\textbf{الجزء الأول --- ميكانيكا \omterm{def:g11:stat:variance}{التباين}.}
\begin{enumerate}
  \item من أجل السلسلة $4, 8, 6, 5, 2$: احسب \omterm{def:g10:stats:mean}{المتوسط}، ثم
        \omterm{def:g11:stat:variance}{التباين} مباشرة من التعريف، ثم
        $\sigma$.
  \item أعد حساب \omterm{def:g11:stat:variance}{التباين} بالصيغة المختصرة
        (\cref{prop:g11:stat:shortcut}) --- \omterm{def:g10:stats:mean}{متوسط} المربعات
        ناقص مربع \omterm{def:g10:stats:mean}{المتوسط} --- وتحقق من التوافق.
  \item مع التواترات: علامات $8$ و $10$ و $14$ نالها
        $3$ و $5$ و $2$ تلاميذ. احسب \omterm{def:g10:stats:mean}{المتوسط} \omterm{def:g11:stat:variance}{والتباين} و
        $\sigma$ (برقمين عشريين).
  \item درجات حرارة متوسطها $20$ و $\sigma = 3$ (بالدرجة
        السيلزية). أعطِ \omterm{def:g10:stats:mean}{المتوسط} و $\sigma$ بالفهرنهايتية
        ($F = 1.8\,C + 32$؛ \cref{prop:g11:stat:affine}).
  \item برهن على القاعدتين المستعملتين للتو: إضافة ثابت $b$ إلى
        كل قيمة تنقل \omterm{def:g10:stats:mean}{المتوسط} بمقدار $b$ وتترك $\sigma$
        دون تغيير؛ والضرب في $a$ يضرب \omterm{def:g10:stats:mean}{المتوسط} في $a$
        و $\sigma$ في $\abs a$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- المسطرة الكونية.}
\emph{الدرجة المعيارية}\index{درجة معيارية} لقيمة $x$ في سلسلة
متوسطها $\bar x$ وانحرافها المعياري $\sigma$ هي
$z = \dfrac{x - \bar x}{\sigma}$: أي عدد الانحرافات
المعيارية التي تفصل $x$ عن \omterm{def:g10:stats:mean}{المتوسط}.
\begin{enumerate}[resume]
  \item الامتحان A متوسطه $12$ و $\sigma = 2$؛ والامتحان B متوسطه
        $8$ و $\sigma = 4$. نالت أليس $15$ في A، ونال بوب $13$
        في B. احسب الدرجتين المعياريتين: فأداء من يقف
        أعلى فوق الجمهور؟
  \item احسب الدرجات المعيارية الخمس لسلسلة السؤال 1،
        وتحقق من أن متوسطها $0$ وانحرافها المعياري
        $1$. وفسّر انطلاقًا من السؤال 5 لماذا تنتج
        المعيرة \emph{دائمًا} متوسطًا $0$ و
        $\sigma = 1$.
  \item اختبار استعداد متوسطه $100$ و $\sigma = 15$. فما
        النتيجة الخام التي درجتها $z = 2$؟ وما \omterm{pb:g11:stat:1}{الدرجة المعيارية} للقيمة $76$؟
  \item يسجّل طبيب أطفال وزن رضيع بالصيغة
        ``$z = -2.5$''. قل ما معنى ذلك، ولماذا يُتابَع النمو
        بالدرجات المعيارية لا بالكيلوغرامات كلما كبر الطفل.
  \item القيمة الشاذة في \cref{exo:g11:stat:10} (السلسلة
        $1, 2, 2, 3, 3, 3, 4, 4, 5, 23$): احسب \omterm{def:g10:stats:mean}{المتوسط} و
        $\sigma$ \omterm{pb:g11:stat:1}{والدرجة المعيارية} للقيمة $23$. وعتبة الإنذار
        الكلاسيكية هي $\abs z \geq 3$: فما الحكم؟
  \item أعد حساب $\sigma$ \emph{بدون} القيمة $23$، ثم
        \omterm{pb:g11:stat:1}{الدرجة المعيارية} التي كانت ستنالها $23$ في مواجهة \omterm{def:g10:stats:mean}{متوسط}
        المعطيات النظيفة و $\sigma$ الخاص بها. فماذا فعلت القيمة الشاذة
        بالمسطرة نفسها التي أُريد بها قياسها، وماذا
        يقترح \cref{exo:g11:stat:10} مضادًا لذلك؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- ضمانة تشيبيشيف.}
\begin{enumerate}[resume]
  \item برهن على متراجحة بيينيمي--تشيبيشيف من أجل
        سلسلة: إذا حققت نسبة $p$ من القيم
        $\abs{x - \bar x} \geq k\sigma$، فإنه، بالاحتفاظ فقط
        بتلك الحدود في مجموع \omterm{def:g11:stat:variance}{التباين}،
        \[
        \sigma^2 \geq p \cdot (k\sigma)^2,
        \qquad\text{ومنه}\qquad
        p \leq \frac{1}{k^2} .
        \]
  \item ماذا تضمن المتراجحة بخصوص نسبة
        أي سلسلة تتجاوز $2\sigma$ عن متوسطها؟ وما وراء
        $3\sigma$؟ تحقق من عبارة $2\sigma$ على سلسلة
        السؤال 1.
  \item مصنع ينتج براغي متوسطها $50$ mm و
        $\sigma = 0.1$ mm. فماذا تضمن متراجحة تشيبيشيف بخصوص
        نسبة البراغي خارج
        $\intcc{49.7}{50.3}$؟ (والإنتاج الحقيقي أفضل من ذلك بكثير
        --- إذ يحوّل \omterm{def:g10:functions:graph}{منحنى} الجرس في السنة المقبلة هذا
        $\frac19$ الحذر إلى جزء من المئة.)
  \item صندوقان يعطي كل منهما مردودًا $5\,\%$ في السنة وسطيًا، مع
        $\sigma = 2\,\%$ للصندوق A و $\sigma = 12\,\%$ للصندوق
        B. فماذا تقول متراجحة تشيبيشيف، من أجل كل صندوق، عن
        النسبة الشبيهة \omterm{def:g10:proba:distribution}{بالاحتمال} للسنوات ذات المردود
        دون $-19\,\%$؟ وما اسم $\sigma$ في المالية؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- ميدالية \omterm{def:g10:stats:mean}{المتوسط}، وتحذير.}
\begin{enumerate}[resume]
  \item ليكن $f(c)$ \omterm{def:g10:stats:mean}{متوسط} الانحرافات المربعة
        $(x_i - c)^2$ عن مركز كيفي $c$. برهن على
        المتطابقة
        \[
        f(c) = \sigma^2 + (\bar x - c)^2 ,
        \]
        واستنتج: \omterm{def:g10:stats:mean}{المتوسط} هو المصغِّر الوحيد \omterm{def:g10:stats:mean}{لمتوسط}
        الانحرافات المربعة --- تلك ميداليته، وهي تقابل تصغير \omterm{def:g11:stat:median}{الوسيط}
        للانحرافات المطلقة في
        \cref{pb:g10:stats:1}.
  \item تحقق من المتطابقة على سلسلة السؤال 1 مع
        $c = 6$: احسب $f(6)$ مباشرة ثم بالصيغة.
  \item ملخصات واحدة ومعطيات مختلفة: تحقق من أن
        $P = \{3, 5, 7, 9, 11\}$ و $Q = \{1, 7, 7, 7, 13\}$
        تشتركان في \omterm{def:g10:stats:mean}{المتوسط} لا في \omterm{def:g11:stat:variance}{التباين}؛ ثم
        أنشئ سلسلة على الصورة
        $\{7 - a, 7, 7, 7, 7 + a\}$ لها بالضبط \omterm{def:g10:stats:mean}{متوسط}
        $P$ وتباينها --- ومع ذلك بشكل خاص بها.
  \item برهن على أن $\sigma = 0$ يفرض تساوي كل القيم. (ماذا
        يجب أن يكون كل حد من مجموع مربعات معدوم؟)
  \item الخاتمة --- سلّم الإحصائي، سطر واحد لكل
        درجة: المراكز (\omterm{def:g10:stats:mean}{المتوسط}، \omterm{def:g11:stat:median}{الوسيط})، والتبددات ($\sigma$، \omterm{def:g11:stat:quartiles}{المدى الربيعي})،
        والمسطرة الكونية ($z$)، والضمانة (تشيبيشيف)،
        وتحذير السؤال 18: الملخصات تضغط،
        والضغط يُفقِد --- \omterm{def:g10:functions:graph}{ومنحنى} الجرس ينتظر في السنة المقبلة
        ليحوّل الدرجات المعيارية إلى احتمالات مضبوطة.
\end{enumerate}
\end{problem}
