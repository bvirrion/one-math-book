\chapter{الجداء السلّمي في المستوي}\label{ch:g11:scal}

يرفق \omterm{def:g11:scal:dot}{الجداء السلّمي} \emph{عددًا} بكل زوج من المتجهات، وهذا العدد
يرى الهندسة: فهو ينعدم بالضبط في حالة الاتجاهين المتعامدين،
وهو يقيس الزوايا والأطوال. وهو المحرّك وراء قانون \omterm{def:g11:trigo:cossin}{جيب التمام}،
ومعادلات الدوائر، وحسابات التعامد. أما الأداة نفسها في الفضاء
فهي موضوع \cref{ch:g12:space}.

\section{التعريف والحسابات الأولى}

\begin{definition}[الجداء السلّمي]\label{def:g11:scal:dot}
\emph{الجداء السلّمي}\index{جداء سلّمي} للمتجهتين $\vec u$
و $\vec v$ هو العدد
\[
\vec u \cdot \vec v
= \tfrac12\left(\norm{\vec u + \vec v}^2 - \norm{\vec u}^2
- \norm{\vec v}^2\right),
\]
حيث يرمز $\norm{\vec u}$ إلى طول (\emph{المعيار}\index{معيار}) \omterm{def:g11:vect:vector}{المتجهة} $\vec u$.
\end{definition}

\begin{theorem}[أوجه الجداء السلّمي الأربعة]\label{thm:g11:scal:faces}
لتكن $\vec u\,(x, y)$ و $\vec v\,(x', y')$ متجهتين غير معدومتين، ولتكن
$\theta$ الزاوية بينهما. عندئذٍ
\begin{enumerate}
  \item \emph{(\omterm{def:g10:coordgeom:system}{بالإحداثيات})} $\vec u \cdot \vec v = xx' + yy'$؛
  \item \emph{(\omterm{def:g11:trigo:cossin}{بجيب التمام})} $\vec u \cdot \vec v =
        \norm{\vec u}\,\norm{\vec v}\cos\theta$؛
  \item \emph{(بالإسقاط)} $\vec u \cdot \vec v = \vec u \cdot \vec v'$،
        حيث $\vec v'$ هي المسقط العمودي \omterm{def:g11:vect:vector}{للمتجهة} $\vec v$ على
        اتجاه $\vec u$؛ فإذا كانت $\vec u = \vect{OA}$ و
        $\vec v = \vect{OB}$، فإن
        $\vec u \cdot \vec v = \overline{OA} \times \overline{OH}$، وهو
        جداء طولين جبريين، حيث $H$ هو المسقط العمودي للنقطة
        $B$ على المستقيم $(OA)$؛
  \item \emph{(بالمعايير)} الصيغة المعرِّفة في \cref{def:g11:scal:dot}.
\end{enumerate}
\end{theorem}

\begin{proof}[برهان 1 و 2]
\emph{1.} \omterm{def:g10:coordgeom:system}{بالإحداثيات}، $\norm{\vec u}^2 = x^2 + y^2$ (فيثاغورس)، و
$\vec u + \vec v$ إحداثياتها $(x + x', y + y')$، إذن
\begin{align*}
2\,\vec u \cdot \vec v
&= (x + x')^2 + (y + y')^2 - (x^2 + y^2) - (x'^2 + y'^2)\\
&= 2xx' + 2yy',
\end{align*}
بعد \omterm{def:g10:algebra:expand}{نشر} المربعات والاختصار.

\emph{2.} اختر \omterm{def:g10:coordgeom:system}{نظام الإحداثيات} بحيث تتجه $\vec u$ في الجهة
الموجبة من المحور $x$: $\vec u\,(\norm{\vec u}, 0)$ و
$\vec v\,(\norm{\vec v}\cos\theta,\ \norm{\vec v}\sin\theta)$
(\cref{ch:g11:trigo}). عندئذٍ تعطي الصيغة 1
$\vec u \cdot \vec v = \norm{\vec u}\,\norm{\vec v}\cos\theta + 0$.
وتقرأ الصيغة 3 الحساب نفسه: فالمسقط العمودي \omterm{def:g11:vect:vector}{للمتجهة} $\vec v$ على
المحور $x$ طوله الجبري $\norm{\vec v}\cos\theta =
\overline{OH}$.
\end{proof}

\begin{example}\label{ex:g11:scal:compute}
$\vec u\,(3, 1)$ و $\vec v\,(2, -4)$:
$\vec u \cdot \vec v = 6 - 4 = 2$. والمعياران هما
$\norm{\vec u} = \sqrt{10}$ و $\norm{\vec v} = \sqrt{20}$، إذن
$\cos\theta = \frac{2}{\sqrt{10}\sqrt{20}} = \frac{2}{10\sqrt2}
= \frac{\sqrt2}{10}$: فالزاوية أصغر قليلًا من $82^\circ$.
\end{example}

\begin{proposition}[القواعد الجبرية]\label{prop:g11:scal:rules}
من أجل كل المتجهات وكل $\lambda \in \R$:
\[
\vec u \cdot \vec v = \vec v \cdot \vec u, \qquad
\vec u \cdot (\vec v + \vec w) = \vec u \cdot \vec v + \vec u \cdot
\vec w, \qquad
(\lambda \vec u)\cdot \vec v = \lambda\,(\vec u \cdot \vec v),
\]
\[
\vec u \cdot \vec u = \norm{\vec u}^2,
\qquad
\norm{\vec u + \vec v}^2 = \norm{\vec u}^2 + 2\,\vec u\cdot\vec v +
\norm{\vec v}^2 .
\]
\end{proposition}

\begin{proof}
كلها تنتج من الصيغة \omterm{def:g10:coordgeom:system}{بالإحداثيات}: فمثلًا
$\vec u \cdot (\vec v + \vec w) = x(x' + x'') + y(y' + y'') =
(xx' + yy') + (xx'' + yy'')$. أما المتطابقة الأخيرة فتنشر
$\norm{\vec u + \vec v}^2 = (\vec u + \vec v)\cdot(\vec u + \vec v)$
باستعمال القواعد الثلاث الأولى --- \omterm{def:g11:scal:dot}{فالجداء السلّمي} يتصرف تمامًا
كجداء عادي، فتنطبق المتطابقات الجبرية المعتادة على المتجهات.
\end{proof}

\section{التعامد}

\begin{definition}[المتجهتان المتعامدتان]\label{def:g11:scal:orthogonal}
تكون متجهتان \emph{متعامدتين}\index{متجهتان متعامدتان}، ويُكتب
$\vec u \perp \vec v$، إذا كان $\vec u \cdot \vec v = 0$.
\end{definition}

\begin{remark}
من أجل متجهتين غير معدومتين، تبيّن الصيغة 2 من \cref{thm:g11:scal:faces} أن
$\vec u \cdot \vec v = 0 \iff \cos\theta = 0 \iff$ الاتجاهين
متعامدان. أما \omterm{def:g11:vect:vector}{المتجهة} المعدومة فمتعامدة مع كل شيء.
\end{remark}

\begin{example}
$\vec u\,(3, 2)$ و $\vec v\,(-4, 6)$: $\vec u \cdot \vec v = -12 + 12 =
0$: متعامدتان. وعمومًا، تكون $\vec u\,(a, b)$ متعامدة دائمًا مع
$\vec v\,(-b, a)$ --- وهو دوران بربع دورة.
\end{example}

\section{قانون جيب التمام}

\begin{theorem}[قانون جيب التمام]\label{thm:g11:scal:cosines}
\index{قانون جيب التمام}
في مثلث $ABC$ أضلاعه $a = BC$ و $b = CA$ و $c = AB$ وزاويته
$\widehat A$ عند الرأس $A$:
\[
a^2 = b^2 + c^2 - 2bc\,\cos\widehat A .
\]
\end{theorem}

\begin{proof}
اكتب $\vect{BC} = \vect{BA} + \vect{AC} = \vect{AC} - \vect{AB}$ ثم انشر
بواسطة \cref{prop:g11:scal:rules}:
\[
a^2 = \norm{\vect{BC}}^2
= \norm{\vect{AC}}^2 - 2\,\vect{AC}\cdot\vect{AB} + \norm{\vect{AB}}^2
= b^2 + c^2 - 2\,\vect{AB}\cdot\vect{AC}.
\]
وحسب صيغة \omterm{def:g11:trigo:cossin}{جيب التمام}،
$\vect{AB}\cdot\vect{AC} = c\,b\,\cos\widehat A$.
\end{proof}

\begin{remark}
عندما تكون $\widehat A$ قائمة، يكون $\cos\widehat A = 0$، فيؤول قانون
\omterm{def:g11:trigo:cossin}{جيب التمام} إلى مبرهنة فيثاغورس $a^2 = b^2 + c^2$: فهو
فيثاغورس مع حد تصحيحي من أجل الزوايا غير القائمة.
\end{remark}

\begin{example}\label{ex:g11:scal:cosines}
ضلعان في مثلث قياسهما $b = 5$ و $c = 8$، ويحصران زاوية
$\widehat A = \frac\pi3$. عندئذٍ
\[
a^2 = 25 + 64 - 2 \times 5 \times 8 \times \frac12 = 89 - 40 = 49,
\]
إذن $a = 7$.
\end{example}

\section{تطبيقات: الدوائر والمستقيمات المتعامدة}

\begin{proposition}[الدائرة ذات قطر معلوم]\label{prop:g11:scal:diameter}
تقع النقطة $M$ على الدائرة ذات القطر $[AB]$ إذا وفقط إذا كان
\[
\vect{MA} \cdot \vect{MB} = 0 .
\]
\end{proposition}

\begin{proof}
إذا كانت $M \neq A, B$، فالشرط يقول إن الزاوية عند $M$ في المثلث
$AMB$ قائمة، والنقط التي ترى قطعة تحت زاوية قائمة هي
بالضبط نقط الدائرة ذات ذلك القطر. وإليك برهانًا مباشرًا:
لتكن $O$ \omterm{prop:g10:coordgeom:midpoint}{منتصف} $[AB]$ وليكن $r = \frac{AB}{2}$. عندئذٍ
$\vect{MA} = \vect{MO} + \vect{OA}$ و
$\vect{MB} = \vect{MO} + \vect{OB} = \vect{MO} - \vect{OA}$، إذن
\[
\vect{MA}\cdot\vect{MB}
= \norm{\vect{MO}}^2 - \norm{\vect{OA}}^2 = MO^2 - r^2,
\]
وهذا ينعدم بالضبط عندما يكون $MO = r$، أي عندما تقع $M$ على الدائرة
التي مركزها $O$ ونصف قطرها $r$.
\end{proof}

\begin{proposition}[المتجهة الناظمة لمستقيم]\label{prop:g11:scal:normal}
\omterm{def:g11:vect:vector}{المتجهة} $\vec n\,(a, b)$ متعامدة مع كل \omterm{def:g11:vect:direction}{متجهة توجيه} للمستقيم
$d : ax + by + c = 0$؛ ويقال إن $\vec n$ \emph{متجهة
ناظمة}\index{متجهة ناظمة} للمستقيم $d$. ويكون مستقيمان متعامدين بالضبط
عندما تكون متجهتاهما الناظمتان (أو متجهتا توجيههما) \omterm{def:g11:scal:orthogonal}{متعامدتين}.
\end{proposition}

\begin{proof}
\omterm{def:g11:vect:direction}{متجهة توجيه} المستقيم $d$ هي $\vec u\,(-b, a)$
(\cref{thm:g11:vect:cartesian})، و
$\vec n \cdot \vec u = a(-b) + b(a) = 0$.
\end{proof}

\begin{method}[استعمال المتجهات الناظمة]\label{met:g11:scal:normal}
\begin{itemize}
  \item المستقيم المار بالنقطة $A$ وناظمته $\vec n\,(a,b)$: اكتب
        $\vect{AM} \cdot \vec n = 0$، وهذا ينشر إلى $ax + by + c = 0$
        مع $c$ تحدّده النقطة $A$.
  \item تعامد $d: ax+by+c = 0$ و $d': a'x+b'y+c' = 0$:
        تحقق من $aa' + bb' = 0$.
\end{itemize}
\end{method}

\begin{example}\label{ex:g11:scal:perpline}
المستقيم المار بالنقطة $P(2, -1)$ والعمودي على $d : 3x - y + 4 = 0$: \omterm{def:g11:vect:direction}{متجهة
توجيه} المستقيم $d$ هي $(1, 3)$، وهي تصلح \omterm{def:g11:vect:vector}{متجهة} \emph{ناظمة}
للعمود. \omterm{def:g10:algebra:equation}{والمعادلة}: $x + 3y + c = 0$ مع
$2 - 3 + c = 0$، إذن $c = 1$: فالمستقيم هو $x + 3y + 1 = 0$.
\end{example}

\begin{example}[معادلة دائرة]\label{ex:g11:scal:circle}
الدائرة التي مركزها $\Omega(1, 2)$ ونصف قطرها $3$ هي مجموعة النقط
$M(x,y)$ التي تحقق $\Omega M^2 = 9$:
\[
(x - 1)^2 + (y - 2)^2 = 9 .
\]
وبالعكس، تُكتب $x^2 + y^2 - 2x - 4y - 4 = 0$، بإكمال
المربعات (\cref{ch:g11:quad})، على الصورة $(x-1)^2 + (y-2)^2 = 9$: وهي الدائرة
نفسها.
\end{example}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:10}
باستعمال $\norm{\vec u + \vec v}^2 = \norm{\vec u}^2 + 2\vec u\cdot\vec v +

\section{مسألة: هديتان من الجداء السلّمي}

\begin{problem}[{مسألة نهاية الأسبوع --- صيغ الجمع تسقط
من جداء سلّمي واحد، والمسافة إلى مستقيم من آخر،
وتلاقي الارتفاعات الثلاثة هدية إضافية}]
\label{pb:g11:scal:1}
يبدو \omterm{def:g11:scal:dot}{الجداء السلّمي} مجرد مسك دفاتر --- اضرب، ثم اجمع ---
ومع ذلك فهو يسلّم، بلا مقابل \omterm{def:g10:numbers:approx}{تقريبًا}، كنزين
قاوما قرونًا من مطاردة الزوايا: \emph{صيغ الجمع}
في المثلثات (احسب $\cos 15^\circ$ بالضبط!) و
\emph{المسافة من نقطة إلى مستقيم} في كسر واحد نظيف. وهو
يبرهن عند الوداع على أن ارتفاعات كل مثلث الثلاثة
تمر بنقطة واحدة. وستُستدعى أوجه
\cref{thm:g11:scal:faces} الأربعة كلها إلى الخدمة.

\medskip
\noindent\textbf{الجزء الأول --- التمكّن.}
\begin{enumerate}
  \item من أجل $\vec u(3, 4)$ و $\vec v(-1, 2)$: احسب
        $\vec u \cdot \vec v$، والمعيارين، والزاوية
        بين المتجهتين (إلى عُشر الدرجة).
  \item أوجد $k$ حتى تكون $(2, k)$ و $(3, -4)$ \omterm{def:g11:scal:orthogonal}{متعامدتين}
        (\cref{def:g11:scal:orthogonal}).
  \item الوجه الفيزيائي: يُجرّ زحّافة $10$ m بقوة $50$ N
        تصنع $60^\circ$ مع الأرض. احسب الشغل
        $W = \vec F \cdot \vec d$.
  \item قانون \omterm{def:g11:trigo:cossin}{جيب التمام} (\cref{thm:g11:scal:cosines}): ضلعان
        طولهما $5$ و $7$ يحصران زاوية $60^\circ$؛ أوجد
        الضلع الثالث بالضبط.
  \item بالعكس: مثلث أضلاعه $4$ و $6$ و $8$. احسب
        \omterm{def:g11:trigo:cossin}{جيب} تمام الزاوية المقابلة للضلع $8$، وصنّف
        الزاوية، وأعطها إلى عُشر الدرجة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الهدية الأولى: صيغ
الجمع.}
لتكن $\vec u = (\cos a, \sin a)$ و $\vec v = (\cos b, \sin b)$:
متجهتان واحديتان بزاويتين $a$ و $b$.
\begin{enumerate}[resume]
  \item احسب $\vec u \cdot \vec v$ مرتين --- مرة \omterm{def:g10:coordgeom:system}{بالإحداثيات}،
        ومرة بالمعيارين والزاوية بين
        المتجهتين --- واستنتج الصيغة الأم:
        \[
        \cos(a - b) = \cos a \cos b + \sin a \sin b .
        \]
  \item عوّض $-b$ محل $b$ (وانتبه إلى \omterm{def:g11:func:parity}{زوجية} \omterm{def:g11:trigo:cossin}{جيب التمام} \omterm{def:g11:func:parity}{وفردية} \omterm{def:g11:trigo:cossin}{الجيب}!)
        لتحصل على $\cos(a + b)$.
  \item احصل على $\sin(a + b)$ من متطابقة التتام
        $\sin x = \cos\left(\frac\pi2 - x\right)$ مطبَّقة على
        $x = a + b$.
  \item القيم المضبوطة أخيرًا: احسب $\cos 15^\circ$ (على الصورة
        $\cos(45^\circ - 30^\circ)$) و $\sin 75^\circ$. وتحقق
        من كليهما عدديًا.
  \item ضع $b = a$ لاستخراج صيغتي الزاوية المضاعفة:
        $\cos 2a$ (بصورها الثلاث) و $\sin 2a$.
  \item انطلاقًا من $\cos 2a = 1 - 2\sin^2 a$، احسب
        $\sin^2 15^\circ$، ثم $\sin 15^\circ$ على صورة \omterm{def:g11:quad:discriminant}{جذر}
        متداخل --- وتحقق من أن مربعه يوافق
        القيمة $\frac{\sqrt6 - \sqrt2}{4}$ التي يعطيها
        منهج السؤال 9.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الهدية الثانية: المسافة إلى
مستقيم.}
\begin{enumerate}[resume]
  \item ليكن $d$ المستقيم $ax + by + c = 0$، ومتجهته الناظمة
        $\vec n(a, b)$ (\cref{prop:g11:scal:normal})،
        ولتكن $P(x_0, y_0)$ نقطة. وبأخذ أي نقطة $A$ من $d$،
        فسّر لماذا تكون المسافة من $P$ إلى $d$ هي
        $\dfrac{\abs{\vect{AP} \cdot \vec n}}{\norm{\vec n}}$،
        واستنتج الصيغة
        \[
        \operatorname{dist}(P, d)
        = \frac{\abs{a x_0 + b y_0 + c}}{\sqrt{a^2 + b^2}} .
        \]
        ثم احسب المسافة من $P(5, 6)$ إلى
        $3x + 4y - 12 = 0$.
  \item اكتب \omterm{def:g10:algebra:equation}{معادلة} الدائرة التي مركزها $(5, 6)$
        وتمسّ ذلك المستقيم.
  \item تحقق من الصيغة تحققًا مزدوجًا: احسب المسافة من
        المبدأ إلى $3x + 4y - 12 = 0$، ثم تأكد منها
        بحساب المسقط العمودي للمبدأ
        صراحةً.
  \item المثلث $A(0,0)$ و $B(6,0)$ و $C(2,5)$: احسب مساحته
        باستعمال القاعدة $[AB]$؛ ثم احسب المسافة من
        $B$ إلى المستقيم $(AC)$ وتحقق من أنها تعطي المساحة
        نفسها.
  \item الدائرة التي مركزها $(2, 1)$ ونصف قطرها $2$، في مواجهة
        المستقيم $x + y - 6 = 0$: احسب مسافة المركز
        إلى المستقيم وصنّف وضعهما النسبي (\omterm{def:g11:deriv:rate}{قاطع}،
        \omterm{def:g11:deriv:tangent}{مماس}، منفصلان).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الهدية الإضافية: تلاقي الارتفاعات.}
\begin{enumerate}[resume]
  \item عمّم \cref{exo:g11:scal:11}: من أجل أي نقطتين
        $A$ و $B$ منتصفهما $I$، برهن على
        \[
        \vect{MA} \cdot \vect{MB} = MI^2 - \frac{AB^2}{4}
        \quad\text{من أجل كل نقطة } M .
        \]
  \item استنتج \cref{prop:g11:scal:diameter} في سطر واحد:
        مجموعة النقط $M$ التي تحقق
        $\vect{MA} \cdot \vect{MB} = 0$ هي الدائرة ذات
        القطر $[AB]$.
  \item الارتفاعات: ليكن $H$ \omterm{def:g10:numbers:interunion}{تقاطع}
        الارتفاعين الصادرين من $A$ ومن $B$ في مثلث $ABC$
        (أي $\vect{HA} \cdot \vect{BC} = 0$ و
        $\vect{HB} \cdot \vect{CA} = 0$). برهن على المتطابقة
        \[
        \vect{HA} \cdot \vect{BC} + \vect{HB} \cdot \vect{CA}
        + \vect{HC} \cdot \vect{AB} = 0
        \]
        --- \omterm{def:g10:algebra:expand}{بنشر} كل شيء انطلاقًا من $H$ بعلاقة شال --- ثم
        استنتج أن $H$ تقع أيضًا على الارتفاع الصادر من $C$:
        فالارتفاعات الثلاثة متلاقية (في
        \emph{مركز التعامد}\index{مركز التعامد}).
  \item الخاتمة --- أربعة أوجه، وثلاث جوائز. أي وجه من
        \cref{thm:g11:scal:faces} حمل صيغ الجمع،
        وأيها حمل صيغة المسافة، وأيها حمل \omterm{pb:g11:scal:1}{مركز التعامد} ---
        ولماذا يكون ``احسب المقدار نفسه مرتين'' شعار
        الفصل كله؟
\end{enumerate}
\end{problem}
