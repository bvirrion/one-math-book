\chapter{समतल में सदिश और रेखाएँ}\label{ch:g11:vect}

\omterm{def:g11:vect:vector}{सदिश} विस्थापन समेटते हैं; रेखाएँ उनके पथ हैं। इस अध्याय का प्रमुख नया औज़ार
\emph{\omterm{def:g11:vect:det}{सारणिक}} है, एक अकेली संख्या जो तय कर देती है कि दो \omterm{def:g11:vect:vector}{सदिश} \omterm{def:g11:vect:collinear}{संरेखी} हैं या
नहीं --- और वहीं से रेखाओं के \omterm{def:g10:algebra:equation}{समीकरण} बनाती है तथा संरेखण और समांतरता के
प्रश्न शुद्ध गणना से सुलझा देती है। वही विचार \cref{ch:g12:space} में अंतरिक्ष तक बढ़ते हैं।

\section{सदिश: एक पुनरावलोकन}

\begin{definition}[सदिश, निर्देशांक]\label{def:g11:vect:vector}
\emph{सदिश}\index{सदिश} $\vec u$ किसी विस्थापन को दर्शाता है: एक ही दिशा, ढंग और
लंबाई वाले सभी तीर एक ही सदिश दर्शाते हैं। किसी \omterm{def:g10:coordgeom:system}{निर्देशांक-पद्धति} में $\vec u\,(x, y)$
विस्थापन दर्ज करता है (क्षैतिज दिशा में $x$, ऊर्ध्वाधर दिशा में $y$); और
बिंदुओं $A(x_A, y_A)$ तथा $B(x_B, y_B)$ के लिए
\[
\vect{AB}\,\bigl(x_B - x_A,\ y_B - y_A\bigr).
\]
सदिश निर्देशांक-दर-निर्देशांक जुड़ते हैं, और $\lambda\vec u$ के \omterm{def:g10:coordgeom:system}{निर्देशांक} $(\lambda x, \lambda y)$ हैं।
\end{definition}

\begin{proposition}[मध्यबिंदु]\label{prop:g11:vect:midpoint}
रेखाखंड $[AB]$ के \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} $M$ के \omterm{def:g10:coordgeom:system}{निर्देशांक} $\left(\frac{x_A + x_B}{2},\ \frac{y_A + y_B}{2}\right)$ हैं।
\end{proposition}

\begin{proof}
$M$ का अभिलक्षण $\vect{AM} = \frac12\vect{AB}$ है: निर्देशांक-दर-निर्देशांक $x_M - x_A = \frac12(x_B - x_A)$, इसलिए $x_M = \frac{x_A + x_B}{2}$, और
$y_M$ के लिए भी वैसे ही।
\end{proof}

\section{संरेखता और सारणिक}

\begin{definition}[संरेखी सदिश]\label{def:g11:vect:collinear}
दो \omterm{def:g11:vect:vector}{सदिश} \emph{संरेखी}\index{संरेखी सदिश} हैं यदि उनमें से एक दूसरे का अदिश
गुणज हो: किसी $\lambda \in \R$ के लिए $\vec v = \lambda \vec u$ (या $\vec u = \vec 0$)। ज्यामितीय रूप से: उनकी दिशा एक ही
है, या उनमें से एक शून्य है।
\end{definition}

\begin{definition}[सारणिक]\label{def:g11:vect:det}
$\vec u\,(x, y)$ और $\vec v\,(x', y')$ का \emph{सारणिक}\index{सारणिक} है
\[
\det(\vec u, \vec v) =
\begin{vmatrix} x & x' \\ y & y' \end{vmatrix}
= x y' - x' y .
\]
\end{definition}

\begin{theorem}[संरेखता की कसौटी]\label{thm:g11:vect:collinearity}
दो \omterm{def:g11:vect:vector}{सदिश} $\vec u$ और $\vec v$ \omterm{def:g11:vect:collinear}{संरेखी} हैं यदि और केवल यदि $\det(\vec u, \vec v) = 0$ हो।
\end{theorem}

\begin{proof}
($\Rightarrow$) यदि $\vec v = \lambda\vec u$ हो, तो $x' = \lambda x$ और $y' = \lambda y$, इसलिए $\det(\vec u, \vec v) = x(\lambda y) - (\lambda x)y = 0$; और यदि $\vec u = \vec 0$ हो, तो \omterm{def:g11:vect:det}{सारणिक}
स्पष्टतः $0$ है।

($\Leftarrow$) मान लीजिए $xy' - x'y = 0$ और $\vec u \neq \vec 0$ है, मान लीजिए $x \neq 0$ (स्थिति $y \neq 0$ सममित है)।
$\lambda = \frac{x'}{x}$ रखिए। तब रचना से $x' = \lambda x$, और संबंध $xy' = x'y = \lambda x y$ से, $x \neq 0$ से भाग देने पर, $y' = \lambda y$
मिलता है। अतः $\vec v = \lambda \vec u$।
\end{proof}

\begin{example}\label{ex:g11:vect:collinear}
$\vec u\,(3, -2)$ और $\vec v\,(-6, 4)$: $\det = 3 \times 4 - (-6) \times (-2) = 12 - 12 = 0$, \omterm{def:g11:vect:collinear}{संरेखी} (सचमुच $\vec v = -2\vec u$)। $\vec u\,(3, -2)$ और $\vec w\,(2, 1)$: $\det = 3 + 4 = 7 \neq 0$, \omterm{def:g11:vect:collinear}{संरेखी} नहीं।
\end{example}

\begin{method}[तीन बिंदुओं का संरेखण]\label{met:g11:vect:alignment}
$A$, $B$, $C$ संरेख हैं यदि और केवल यदि $\vect{AB}$ और $\vect{AC}$ \omterm{def:g11:vect:collinear}{संरेखी} हों: दोनों
\omterm{def:g11:vect:vector}{सदिश} निकालिए और $\det\bigl(\vect{AB}, \vect{AC}\bigr) = 0$ जाँचिए।
\end{method}

\section{रेखाओं के कार्तीय समीकरण}

\begin{definition}[दिक् सदिश]\label{def:g11:vect:direction}
रेखा $d$ का \emph{दिक् सदिश}\index{दिक् सदिश} कोई भी शून्येतर \omterm{def:g11:vect:vector}{सदिश} $\vec u$ है
जिसके लिए $d$ के सभी बिंदुओं $A, B$ हेतु $\vect{AB}$ $\vec u$ के \omterm{def:g11:vect:collinear}{संरेखी} हो।
\end{definition}

\begin{theorem}[कार्तीय समीकरण]\label{thm:g11:vect:cartesian}
\omterm{def:g11:vect:direction}{दिक् सदिश} $\vec u\,(-b, a)$ वाली रेखा का एक \omterm{def:g10:algebra:equation}{समीकरण} इस रूप का होता है
\[
ax + by + c = 0
\qquad (a, b) \neq (0, 0).
\]
इसके उलट, ऐसा हर \omterm{def:g10:algebra:equation}{समीकरण} \omterm{def:g11:vect:direction}{दिक् सदिश} $\vec u\,(-b, a)$ वाली एक रेखा बताता है।
\end{theorem}

\begin{proof}
मान लीजिए $d$ $A(x_A, y_A)$ से होकर दिशा $\vec u\,(-b, a)$ में जाती है। बिंदु $M(x, y)$ $d$ पर ठीक तब
है जब $\vect{AM}$ $\vec u$ का \omterm{def:g11:vect:collinear}{संरेखी} हो, अर्थात् (\cref{thm:g11:vect:collinearity})
\[
0 = \det\bigl(\vect{AM}, \vec u\bigr)
= (x - x_A)\,a - (-b)(y - y_A)
= ax + by - (a x_A + b y_A),
\]
जो घोषित रूप का \omterm{def:g10:algebra:equation}{समीकरण} है, जहाँ $c = -(a x_A + b y_A)$ है। इसके उलट, मान लीजिए $b \neq 0$ के साथ
$ax + by + c = 0$ के हल बिंदु $\left(x, -\frac{a x + c}{b}\right)$ हैं: उनमें से दो को घटाने पर पता चलता है कि हर जीवा
$(-b, a)$ की \omterm{def:g11:vect:collinear}{संरेखी} है, और कोई एक हल सदा मौजूद है --- यह वही रेखा है जो उससे होकर
जाती है और $(-b, a)$ द्वारा दिशित है।
\end{proof}

\begin{method}[दो बिंदुओं से जाने वाली रेखा]\label{met:g11:vect:twopoints}
रेखा $(AB)$ का \omterm{def:g10:algebra:equation}{समीकरण} ज्ञात करने के लिए: \omterm{def:g11:vect:direction}{दिक् सदिश} $\vect{AB}\,(x_B - x_A, y_B - y_A)$ निकालिए; उसे $(-b, a)$ से
मिलाइए, अर्थात् $a = y_B - y_A$ और $b = -(x_B - x_A)$ लीजिए; और $A$ के \omterm{def:g10:coordgeom:system}{निर्देशांक} रखकर $c$ ज्ञात
कीजिए।
\end{method}

\begin{example}\label{ex:g11:vect:twopoints}
$A(1, 2)$ और $B(4, 3)$ से होकर जाने वाली रेखा: $\vect{AB}\,(3, 1)$, इसलिए $a = 1$, $b = -3$, और \omterm{def:g10:algebra:equation}{समीकरण}
$x - 3y + c = 0$ है। $A$ रखने पर: $1 - 6 + c = 0$, इसलिए $c = 5$। \omterm{def:g10:algebra:equation}{समीकरण}: $x - 3y + 5 = 0$। $B$ से जाँच:
$4 - 9 + 5 = 0$।
\end{example}

\begin{remark}
जब $b \neq 0$ हो, तब \omterm{def:g10:algebra:equation}{समीकरण} को $y$ के लिए हल किया जा सकता है: \omterm{def:g10:reffunc:affine}{प्रवणता} $m = -\frac ab$
वाला $y = mx + p$। कार्तीय रूप $ax + by + c = 0$ अधिक व्यापक है: वह ऊर्ध्वाधर रेखाओं ($b = 0$)
को भी समेट लेता है, जिनकी कोई \omterm{def:g10:reffunc:affine}{प्रवणता} नहीं होती।
\end{remark}

\section{प्राचलिक निरूपण}

\begin{definition}[प्राचलिक निरूपण]\label{def:g11:vect:parametric}
$A(x_A, y_A)$ से होकर दिशा $\vec u\,(\alpha, \beta)$ में जाने वाली रेखा बिंदुओं का यह समुच्चय है
\[
M\bigl(x_A + t\alpha,\ y_A + t\beta\bigr), \qquad t \in \R .
\]
संख्या $t$ \emph{प्राचल} है: $t$ का हर मान रेखा का एक बिंदु देता है,
मानो उस पर अचर चाल $\vec u$ से चला जा रहा हो।
\end{definition}

\begin{example}
\cref{ex:g11:vect:twopoints} की रेखा $\{(1 + 3t,\ 2 + t) : t \in \R\}$ भी है: $t = 0$ से $A$ मिलता है, $t = 1$ से $B$। $t$
का विलोपन करने पर ($t = y - 2$, इसलिए $x = 1 + 3(y-2)$) $x - 3y + 5 = 0$ फिर मिल जाता है।
\end{example}

\section{दो रेखाओं की सापेक्ष स्थिति}

\begin{proposition}[समांतर या प्रतिच्छेदी]\label{prop:g11:vect:positions}
\omterm{def:g11:vect:direction}{दिक् सदिश} $\vec u$ और $\vec v$ वाली दो रेखाएँ \omterm{def:g11:seq:arithmetic}{समांतर} हैं यदि और केवल यदि $\det(\vec u, \vec v) = 0$
हो। \omterm{def:g10:algebra:equation}{समीकरण} के रूप में: $d : ax + by + c =
0$ और $d' : a'x + b'y + c' = 0$ \omterm{def:g11:seq:arithmetic}{समांतर} हैं यदि और केवल यदि $ab' - a'b = 0$ हो।
असमांतर रेखाएँ ठीक एक बिंदु पर मिलती हैं।
\end{proposition}

\begin{proof}
समांतरता का अर्थ है दिशाओं का \omterm{def:g11:vect:collinear}{संरेखी} होना, अर्थात् \cref{thm:g11:vect:collinearity}; और $\vec u\,(-b, a)$ तथा $\vec v\,(-b', a')$
के साथ \omterm{def:g11:vect:det}{सारणिक} $(-b)a' - (-b')a = ab' - a'b$ है। यदि रेखाएँ \omterm{def:g11:seq:arithmetic}{समांतर} न हों, तो दोनों \omterm{def:g10:algebra:equation}{समीकरणों} को एक साथ
हल करने पर (प्रतिस्थापन या संयोजन से) अद्वितीय हल मिलता है: निकाय शून्येतर
\omterm{def:g11:vect:det}{सारणिक} वाले $2 \times 2$ निकाय की तरह व्यवहार करता है।
\end{proof}

\begin{method}[दो रेखाओं का प्रतिच्छेद]\label{met:g11:vect:intersection}
दोनों \omterm{def:g10:algebra:equation}{समीकरणों} का निकाय हल कीजिए: एक \omterm{def:g10:algebra:equation}{समीकरण} में एक अज्ञात अलग कीजिए और दूसरे
में रखिए, या \omterm{def:g10:algebra:equation}{समीकरणों} को मिलाकर एक अज्ञात का विलोप कीजिए। मिले हुए बिंदु की
जाँच सदा \emph{दोनों} \omterm{def:g10:algebra:equation}{समीकरणों} में कीजिए।
\end{method}

\begin{example}\label{ex:g11:vect:intersection}
$d: x - 3y + 5 = 0$ और $d': 2x + y - 4 = 0$ का प्रतिच्छेद निकालिए। $d'$ से: $y = 4 - 2x$। $d$ में रखने पर:
\[
x - 3(4 - 2x) + 5 = 0
\iff x - 12 + 6x + 5 = 0
\iff 7x = 7
\iff x = 1,
\]
फिर $y = 2$। रेखाएँ $(1, 2)$ पर मिलती हैं। $d$ में जाँच: $1 - 6 + 5 = 0$।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g11:vect:1}
$A(2, -1)$, $B(5, 3)$ और $C(-1, 1)$ दिए हों, तो $\vect{AB}$, $\vect{AC}$, $\vect{AB} + \vect{AC}$ और $2\vect{AB} -
3\vect{AC}$ के \omterm{def:g10:coordgeom:system}{निर्देशांक}
तथा $[BC]$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:vect:2}
क्या \omterm{def:g11:vect:vector}{सदिश} \omterm{def:g11:vect:collinear}{संरेखी} हैं?
\[
\vec u\,(4, -6) \text{ और } \vec v\,(-2, 3); \qquad
\vec u\,(2, 5) \text{ और } \vec v\,(3, 7); \qquad
\vec u\,(0, 3) \text{ और } \vec v\,(0, -8).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:vect:3}
$A(2, 1)$ से होकर \omterm{def:g11:vect:direction}{दिक् सदिश} $\vec u\,(3, -1)$ वाली रेखा का, और फिर $B(0, 4)$ तथा $C(2, 0)$ से होकर
जाने वाली रेखा का कार्तीय \omterm{def:g10:algebra:equation}{समीकरण} ज्ञात कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:vect:4}
रेखा $d: 3x - 2y + 6 = 0$ के लिए एक \omterm{def:g11:vect:direction}{दिक् सदिश} और दोनों निर्देशांक-अक्षों के साथ प्रतिच्छेद
बताइए, और तय कीजिए कि बिंदु $P(2, 6)$ तथा $Q(1, 4)$ $d$ के हैं या नहीं।
\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:vect:6}

d: 2x + y - 7 = 0 \ \text{ और }\ d': x - y + 1 = 0;
\qquad
e: x - 2y + 3 = 0 \ \text{ और }\ e': -2x + 4y + 1 = 0 .

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:vect:7}

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:vect:10}
प्राचल $m$ के किन मानों के लिए \omterm{def:g11:vect:vector}{सदिश} $\vec u\,(m, 2)$ और $\vec v\,(3, m - 1)$ \omterm{def:g11:vect:collinear}{संरेखी} हैं?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:vect:11}
मान लीजिए $ABCD$ एक \omterm{def:g11:seq:arithmetic}{समांतर} चतुर्भुज है, $I$ $[AB]$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} है, और $J$
वह बिंदु है जो $\vect{DJ} = \frac23 \vect{DI}$ से परिभाषित होता है। उस \omterm{def:g10:coordgeom:system}{निर्देशांक-पद्धति} में काम कीजिए
जिसमें $A(0,0)$, $B(1,0)$, $D(0,1)$ हों (जिससे $C(1,1)$):
\begin{enumerate}
  \item $I$ और $J$ के \omterm{def:g10:coordgeom:system}{निर्देशांक} निकालिए।
  \item दिखाइए कि $A$, $J$ और $C$ संरेख हैं। (एक सुंदर तथ्य: रेखा
        $(DI)$ विकर्ण $(AC)$ को नीचे से एक-तिहाई पर काटती है।)
\end{enumerate}
\end{exercise}

\section{समस्या: टक्कर के रास्ते और चतुर निर्देशांक}

\begin{problem}[{सप्ताहांत समस्या --- रास्तों का कटना टकराना नहीं है: समुद्र
में निकटतम पहुँच, और सही ढाँचा चुनकर सिद्ध की गई प्रमेय}]
\label{pb:g11:vect:1}
दो जहाज़ों के सीधे रास्ते एक-दूसरे को काटते हैं --- तो क्या जहाज़ों का टकराना
अनिवार्य है? तब तक नहीं, जब तक वे कटान-बिंदु पर \emph{एक ही समय} न पहुँचें:
रास्ते बिंदुओं के समुच्चय हैं, गतियाँ प्राचलिक हैं, और इन दोनों को गड्डमड्ड
कर देना सचमुच जहाज़ डुबो चुका है। यह समस्या प्राचलिक रेखाओं (\cref{def:g11:vect:parametric}) पर एक
छोटी-सी पोत-यातायात सेवा चलाती है, और फिर \omterm{def:g10:coordgeom:system}{निर्देशांकों} की दूसरी महाशक्ति के
साथ शुद्ध ज्यामिति की ओर मुड़ती है: उन्हें चतुराई से \emph{चुनना}, जैसा \cref{exo:g11:vect:11}
में हुआ, ताकि चिरपरिचित प्रमेय स्वयं अपनी गणना कर लें।

\medskip
\noindent\textbf{भाग I --- रेखाएँ, तीन तरह से।}
\begin{enumerate}
  \item $A(1, 2)$ से होकर \omterm{def:g11:vect:direction}{दिक् सदिश} $\vec u(3, -1)$ वाली रेखा का प्राचलिक निरूपण दीजिए,
        फिर प्राचल का विलोप करके कार्तीय \omterm{def:g10:algebra:equation}{समीकरण} प्राप्त कीजिए।
  \item रेखा $2x - 5y + 3 = 0$ के लिए: एक \omterm{def:g11:vect:direction}{दिक् सदिश} पढ़िए (\cref{thm:g11:vect:cartesian}), एक बिंदु ज्ञात
        कीजिए, और प्राचलिक निरूपण लिखिए।
  \item प्रश्न 1 और 2 की दोनों रेखाओं का प्रतिच्छेद बिंदु निकालिए (\cref{met:g11:vect:intersection})।
  \item \omterm{def:g11:vect:det}{सारणिक} (\cref{thm:g11:vect:collinearity}) से पुष्टि कीजिए कि उन दोनों रेखाओं का मिलना अनिवार्य
        था; फिर दिखाइए कि $2x - 5y + 3 = 0$ और $-4x + 10y + 1 = 0$ कड़ाई से \omterm{def:g11:seq:arithmetic}{समांतर} हैं।
  \item क्या $A(2, 1)$, $B(5, 3)$, $C(11, 7)$ संरेख हैं (\cref{met:g11:vect:alignment})?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- पोत-यातायात सेवा।}
समय $t$ (घंटों में) पर जहाज़ $P$ $(2t,\ 3 + t)$ पर है और जहाज़ $Q$ $(10 - t,\ 2t - 1)$ पर
(दूरियाँ नाविक मील में)।
\begin{enumerate}[resume]
  \item दोनों \emph{रास्तों} के कार्तीय \omterm{def:g10:algebra:equation}{समीकरण} ज्ञात कीजिए, और निकालिए कि
        रास्ते कहाँ कटते हैं।
  \item हर जहाज़ उस कटान-बिंदु से किस समय गुज़रता है? क्या जहाज़ टकराते हैं?
        एक वाक्य में सामान्य चेतावनी बताइए: रास्तों का कटना बनाम गतियों का
        मिलना।
  \item समय $t$ पर दोनों जहाज़ों के बीच वर्ग-दूरी $D^2(t)$ निकालिए और उसे
        $t$ में द्विघात के रूप में लाइए। जहाज़ किस समय सबसे पास होते हैं,
        और कितने पास आते हैं (\cref{pb:g11:quad:1} की शीर्ष-तकनीक)?
  \item नियम कम से कम $1$ मील की दूरी माँगते हैं। क्या उनका उल्लंघन होता
        है? यदि हाँ, तो $D^2(t) < 1$ हल कीजिए और उल्लंघन की अवधि बताइए।
  \item समझाइए कि अचर चालों से सीधे रास्तों पर चलने वाले \emph{किन्हीं भी}
        दो जहाज़ों के लिए $D^2(t)$ $t$ का \omterm{def:g11:quad:quadratic}{द्विघात फलन} ही क्यों होता है ---
        इसलिए ``निकटतम पहुँच बिंदु'' सदा एक शीर्ष-गणना ही है।
  \item अवरोधन: एक गश्ती नौका $t = 0$ पर मूल बिंदु से अचर वेग $(a, b)$ के साथ
        चलती है और उसे जहाज़ $P$ से ठीक $t = 3$ पर मिलना है। आवश्यक $(a, b)$
        और गश्ती नौका की चाल निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- चतुर \omterm{def:g10:coordgeom:system}{निर्देशांक}।}
\cref{exo:g11:vect:11} के ढाँचे में काम कीजिए: \omterm{def:g11:seq:arithmetic}{समांतर} चतुर्भुज $ABCD$ $A(0,0)$, $B(1,0)$, $C(1,1)$,
$D(0,1)$ बन जाता है; मान लीजिए $I$ $[AB]$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} है और $K$ $[CD]$ का।
\begin{enumerate}[resume]
  \item रेखा $(DI)$ का कार्तीय \omterm{def:g10:algebra:equation}{समीकरण} निकालिए।
  \item $(DI)$ को विकर्ण $(AC)$ से काटिए: \cref{exo:g11:vect:11} वाला तथ्य फिर से मिल जाता है
        --- कटान ठीक विकर्ण के नीचे से एक-तिहाई पर है।
  \item अब रेखा $(BK)$: उसका \omterm{def:g10:algebra:equation}{समीकरण} ज्ञात कीजिए, $(AC)$ से काटिए, और पूरी
        चिरपरिचित प्रमेय का निष्कर्ष निकालिए: \emph{चेवियन $(DI)$ और $(BK)$
        विकर्ण $(AC)$ को त्रिभाजित कर देते हैं}।
  \item इस एक विशेष ढाँचे में प्रमेय सिद्ध करना उचित क्यों था? (\omterm{def:g10:coordgeom:system}{निर्देशांकों}
        का चुनाव किसे बनाए रखता है, और कथन में क्या-क्या आता है?) दो-तीन
        वाक्यों में उत्तर दीजिए।
  \item अभ्यास के लिए एक और त्रिभाजन: मान लीजिए $E$ $[BC]$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु}
        है। रेखा $(AE)$ \emph{दूसरे} विकर्ण $(DB)$ को कहाँ काटती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- स्थिति की पहेलियाँ।}
\begin{enumerate}[resume]
  \item क्या तीनों रेखाएँ $x + y = 4$, \ $2x - y = 2$, \ $x - 2y = -2$ संगामी हैं? (दो को काटिए
        और तीसरी पर जाँचिए।)
  \item हर वास्तविक $m$ के लिए मान लीजिए $d_m$ रेखा $mx - y + 2 - 3m = 0$ है। दिखाइए कि
        \emph{हर} $d_m$ एक ही नियत बिंदु से होकर जाती है। ($m$ में पद
        समूहित कीजिए।)
  \item परिवार $(d_m)$ में: कौन-सी सदस्य रेखा $y = 2x + 7$ के \omterm{def:g11:seq:arithmetic}{समांतर} है? कौन-सी मूल
        बिंदु से होकर जाती है?
  \item समापन --- रेखा के तीन वेश: कार्तीय (एक प्रतिस्थापन में सदस्यता की
        जाँच), संक्षिप्त (\omterm{def:g10:reffunc:affine}{प्रवणता} और \omterm{def:g10:functions:graph}{आलेख} एक नज़र में), प्राचलिक (गति, समय,
        अवरोधन); समांतरता के भविष्यवक्ता के रूप में \omterm{def:g11:vect:det}{सारणिक}; और प्रमेय बनाने
        के कारख़ाने के रूप में सोच-समझकर चुना गया ढाँचा। हर एक पर एक वाक्य।
\end{enumerate}
\end{problem}
