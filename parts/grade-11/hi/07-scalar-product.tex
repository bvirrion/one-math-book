\chapter{समतल में अदिश गुणनफल}\label{ch:g11:scal}

\omterm{def:g11:scal:dot}{अदिश गुणनफल} \omterm{def:g10:vectors:vector}{सदिशों} के किसी युग्म के साथ एक \emph{संख्या} जोड़ देता है, और वह
संख्या ज्यामिति देख लेती है: वह ठीक लंब दिशाओं के लिए लुप्त होती है, और कोण
तथा लंबाइयाँ नाप लेती है। \omterm{def:g11:trigo:cossin}{कोज्या} नियम, वृत्तों के \omterm{def:g10:algebra:equation}{समीकरण} और लंबता की गणनाओं
के पीछे यही इंजन है। अंतरिक्ष में यही औज़ार \cref{ch:g12:space} का विषय है।

\section{परिभाषा और पहली गणनाएँ}

\begin{definition}[अदिश गुणनफल]\label{def:g11:scal:dot}
दो \omterm{def:g10:vectors:vector}{सदिशों} $\vec u$ और $\vec v$ का \emph{अदिश गुणनफल}\index{अदिश गुणनफल} यह संख्या है
\[
\vec u \cdot \vec v
= \tfrac12\left(\norm{\vec u + \vec v}^2 - \norm{\vec u}^2
- \norm{\vec v}^2\right),
\]
जहाँ $\norm{\vec u}$ $\vec u$ की लंबाई (\emph{परिमाण}\index{परिमाण}) दर्शाता है।
\end{definition}

\begin{theorem}[अदिश गुणनफल के चार चेहरे]\label{thm:g11:scal:faces}
मान लीजिए $\vec u\,(x, y)$ और $\vec v\,(x', y')$ शून्येतर \omterm{def:g11:vect:vector}{सदिश} हैं और $\theta$ उनके बीच का कोण है। तब
\begin{enumerate}
  \item \emph{(\omterm{def:g10:coordgeom:system}{निर्देशांक})} $\vec u \cdot \vec v = xx' + yy'$;
  \item \emph{(\omterm{def:g11:trigo:cossin}{कोज्या})} $\vec u \cdot \vec v =
        \norm{\vec u}\,\norm{\vec v}\cos\theta$;
  \item \emph{(प्रक्षेप)} $\vec u \cdot \vec v = \vec u \cdot \vec v'$, जहाँ $\vec v'$ $\vec u$ की दिशा पर $\vec v$ का \omterm{def:g11:scal:orthogonal}{लांबिक}
        प्रक्षेप है; और यदि $\vec u = \vect{OA}$ तथा $\vec v = \vect{OB}$ हों, तो $\vec u \cdot \vec v = \overline{OA} \times \overline{OH}$, जो चिह्नित
        लंबाइयों का गुणनफल है, जहाँ $H$ $B$ से रेखा $(OA)$ पर डाले गए लंब
        का पाद है;
  \item \emph{(\omterm{def:g11:scal:dot}{परिमाण})} \cref{def:g11:scal:dot} का परिभाषा-सूत्र।
\end{enumerate}
\end{theorem}

\begin{proof}[1 और 2 की उपपत्ति]
\emph{1.} \omterm{def:g10:coordgeom:system}{निर्देशांकों} में $\norm{\vec u}^2 = x^2 + y^2$ (पाइथागोरस) और $\vec u + \vec v$ के \omterm{def:g10:coordgeom:system}{निर्देशांक} $(x + x', y + y')$ हैं,
इसलिए वर्गों का प्रसार करके काटने पर
\begin{align*}
2\,\vec u \cdot \vec v
&= (x + x')^2 + (y + y')^2 - (x^2 + y^2) - (x'^2 + y'^2)\\
&= 2xx' + 2yy',
\end{align*}

\emph{2.} \omterm{def:g10:coordgeom:system}{निर्देशांक-पद्धति} ऐसी चुनिए कि $\vec u$ धनात्मक $x$-अक्ष की दिशा में
हो: $\vec u\,(\norm{\vec u}, 0)$ और $\vec v\,(\norm{\vec v}\cos\theta,\ \norm{\vec v}\sin\theta)$ (\cref{ch:g11:trigo})। तब सूत्र 1 से $\vec u \cdot \vec v = \norm{\vec u}\,\norm{\vec v}\cos\theta + 0$ मिलता है। सूत्र 3 उसी गणना को
पढ़ लेता है: $x$-अक्ष पर $\vec v$ के प्रक्षेप की चिह्नित लंबाई $\norm{\vec v}\cos\theta =
\overline{OH}$ है।
\end{proof}

\begin{example}\label{ex:g11:scal:compute}
$\vec u\,(3, 1)$ और $\vec v\,(2, -4)$: $\vec u \cdot \vec v = 6 - 4 = 2$। \omterm{def:g11:scal:dot}{परिमाण} $\norm{\vec u} = \sqrt{10}$ और $\norm{\vec v} = \sqrt{20}$ हैं, इसलिए $\cos\theta = \frac{2}{\sqrt{10}\sqrt{20}} = \frac{2}{10\sqrt2}
= \frac{\sqrt2}{10}$: कोण $82^\circ$ से कुछ
कम है।
\end{example}

\begin{proposition}[बीजगणितीय नियम]\label{prop:g11:scal:rules}
सभी \omterm{def:g10:vectors:vector}{सदिशों} और सभी $\lambda \in \R$ के लिए:
\[
\vec u \cdot \vec v = \vec v \cdot \vec u, \qquad
\vec u \cdot (\vec v + \vec w) = \vec u \cdot \vec v + \vec u \cdot
\vec w, \qquad
(\lambda \vec u)\cdot \vec v = \lambda\,(\vec u \cdot \vec v),
\]
\[
\vec u \cdot \vec u = \norm{\vec u}^2,
\qquad
\norm{\vec u + \vec v}^2 = \norm{\vec u}^2 + 2\,\vec u\cdot\vec v +
\norm{\vec v}^2 .
\]
\end{proposition}

\begin{proof}
ये सब निर्देशांक-सूत्र से निकलते हैं: जैसे $\vec u \cdot (\vec v + \vec w) = x(x' + x'') + y(y' + y'') =
(xx' + yy') + (xx'' + yy'')$। अंतिम सर्वसमिका पहले तीन
नियमों का उपयोग करके $\norm{\vec u + \vec v}^2 = (\vec u + \vec v)\cdot(\vec u + \vec v)$ का प्रसार करती है --- \omterm{def:g11:scal:dot}{अदिश गुणनफल} ठीक साधारण
गुणनफल की तरह व्यवहार करता है, इसलिए सामान्य बीजगणितीय सर्वसमिकाएँ \omterm{def:g10:vectors:vector}{सदिशों} पर
भी लागू होती हैं।
\end{proof}

\section{लंबता}

\begin{definition}[लांबिक सदिश]\label{def:g11:scal:orthogonal}
दो \omterm{def:g11:vect:vector}{सदिश} \emph{लांबिक}\index{लांबिक सदिश} कहलाते हैं, और इसे $\vec u \perp \vec v$ लिखते हैं,
यदि $\vec u \cdot \vec v = 0$ हो।
\end{definition}

\begin{remark}
शून्येतर \omterm{def:g10:vectors:vector}{सदिशों} के लिए \cref{thm:g11:scal:faces} का सूत्र 2 दिखाता है कि $\vec u \cdot \vec v = 0 \iff \cos\theta = 0 \iff$ का अर्थ है कि
दिशाएँ परस्पर लंब हैं। शून्य \omterm{def:g11:vect:vector}{सदिश} हर \omterm{def:g11:vect:vector}{सदिश} के \omterm{def:g11:scal:orthogonal}{लांबिक} है।
\end{remark}

\begin{example}
$\vec u\,(3, 2)$ और $\vec v\,(-4, 6)$: $\vec u \cdot \vec v = -12 + 12 =
0$: \omterm{def:g11:scal:orthogonal}{लांबिक}। सामान्य रूप से $\vec u\,(a, b)$ सदा $\vec v\,(-b, a)$ के \omterm{def:g11:scal:orthogonal}{लांबिक} होता है
--- यह एक चौथाई चक्कर का घूर्णन है।
\end{example}

\section{कोज्या नियम}

\begin{theorem}[कोज्या नियम]\label{thm:g11:scal:cosines}
\index{कोज्या नियम}
भुजाओं $a = BC$, $b = CA$, $c = AB$ वाले त्रिभुज $ABC$ में, जिसमें शीर्ष $A$ पर कोण $\widehat A$
है:
\[
a^2 = b^2 + c^2 - 2bc\,\cos\widehat A .
\]
\end{theorem}

\begin{proof}
$\vect{BC} = \vect{BA} + \vect{AC} = \vect{AC} - \vect{AB}$ लिखिए और \cref{prop:g11:scal:rules} से प्रसार कीजिए:
\[
a^2 = \norm{\vect{BC}}^2
= \norm{\vect{AC}}^2 - 2\,\vect{AC}\cdot\vect{AB} + \norm{\vect{AB}}^2
= b^2 + c^2 - 2\,\vect{AB}\cdot\vect{AC}.
\]
कोज्या-सूत्र से $\vect{AB}\cdot\vect{AC} = c\,b\,\cos\widehat A$।
\end{proof}

\begin{remark}
जब $\widehat A$ समकोण हो, तब $\cos\widehat A = 0$ और \omterm{def:g11:trigo:cossin}{कोज्या} नियम पाइथागोरस प्रमेय $a^2 = b^2 + c^2$ बन जाता है:
यह असमकोणों के लिए सुधार-पद के साथ पाइथागोरस ही है।
\end{remark}

तीसरी भुजा $a$ तय हो जाती है।}
\end{omfigure}

\begin{example}\label{ex:g11:scal:cosines}
किसी त्रिभुज की दो भुजाएँ $b = 5$ और $c = 8$ हैं, और उनके बीच का कोण $\widehat A = \frac\pi3$ है। तब
\[
a^2 = 25 + 64 - 2 \times 5 \times 8 \times \frac12 = 89 - 40 = 49,
\]
इसलिए $a = 7$।
\end{example}

\section{अनुप्रयोग: वृत्त और लंब रेखाएँ}

\begin{proposition}[दिए हुए व्यास वाला वृत्त]\label{prop:g11:scal:diameter}
बिंदु $M$ व्यास $[AB]$ वाले वृत्त पर है यदि और केवल यदि
\[
\vect{MA} \cdot \vect{MB} = 0 .
\]
\end{proposition}

\begin{proof}
यदि $M \neq A, B$ हो, तो शर्त कहती है कि त्रिभुज $AMB$ में $M$ पर का कोण समकोण है,
और किसी रेखाखंड को समकोण पर देखने वाले बिंदु ठीक उसी व्यास वाले वृत्त के
बिंदु हैं। यहाँ सीधी उपपत्ति है: मान लीजिए $O$ $[AB]$ का \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} है और
$r = \frac{AB}{2}$ है। तब $\vect{MA} = \vect{MO} + \vect{OA}$ और $\vect{MB} = \vect{MO} + \vect{OB} = \vect{MO} - \vect{OA}$, इसलिए
\[
\vect{MA}\cdot\vect{MB}
= \norm{\vect{MO}}^2 - \norm{\vect{OA}}^2 = MO^2 - r^2,
\]
जो ठीक तब लुप्त होता है जब $MO = r$ हो, अर्थात् जब $M$ केंद्र $O$ और
त्रिज्या $r$ वाले वृत्त पर हो।
\end{proof}

\begin{proposition}[रेखा का अभिलंब सदिश]\label{prop:g11:scal:normal}
\omterm{def:g11:vect:vector}{सदिश} $\vec n\,(a, b)$ रेखा $d : ax + by + c = 0$ के हर \omterm{def:g11:vect:direction}{दिक् सदिश} के \omterm{def:g11:scal:orthogonal}{लांबिक} है; कहा जाता है कि $\vec n$
$d$ का \emph{अभिलंब सदिश}\index{अभिलंब सदिश} है। दो रेखाएँ ठीक तब परस्पर
लंब हैं जब उनके अभिलंब (या दिक्) \omterm{def:g11:vect:vector}{सदिश} \omterm{def:g11:scal:orthogonal}{लांबिक} हों।
\end{proposition}

\begin{proof}
$d$ का एक \omterm{def:g11:vect:direction}{दिक् सदिश} $\vec u\,(-b, a)$ है (\cref{thm:g11:vect:cartesian}), और $\vec n \cdot \vec u = a(-b) + b(a) = 0$।
\end{proof}

\begin{method}[अभिलंब सदिशों का उपयोग]\label{met:g11:scal:normal}
\begin{itemize}
  \item $A$ से होकर अभिलंब $\vec n\,(a,b)$ वाली रेखा: $\vect{AM} \cdot \vec n = 0$ लिखिए, जिसका प्रसार
        $ax + by + c = 0$ है, और $c$ $A$ से तय हो जाता है।
  \item $d: ax+by+c = 0$ और $d': a'x+b'y+c' = 0$ की लंबता: $aa' + bb' = 0$ जाँचिए।
\end{itemize}
\end{method}

\begin{example}\label{ex:g11:scal:perpline}
$P(2, -1)$ से होकर $d : 3x - y + 4 = 0$ पर लंब रेखा: $d$ का एक \omterm{def:g11:vect:direction}{दिक् सदिश} $(1, 3)$ है, जो लंब रेखा
के \emph{अभिलंब} \omterm{def:g11:vect:vector}{सदिश} का काम करता है। \omterm{def:g10:algebra:equation}{समीकरण}: $x + 3y + c = 0$, जहाँ $2 - 3 + c = 0$, इसलिए
$c = 1$: रेखा $x + 3y + 1 = 0$ है।
\end{example}

\begin{example}[वृत्त का समीकरण]\label{ex:g11:scal:circle}
केंद्र $\Omega(1, 2)$ और त्रिज्या $3$ वाला वृत्त उन बिंदुओं $M(x,y)$ का समुच्चय है
जिनके लिए $\Omega M^2 = 9$ हो:
\[
(x - 1)^2 + (y - 2)^2 = 9 .
\]
इसके उलट, वर्ग पूरे करने पर (\cref{ch:g11:quad}) $x^2 + y^2 - 2x - 4y - 4 = 0$ $(x-1)^2 + (y-2)^2 = 9$ बन जाता है: वही वृत्त।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g11:scal:1}
इनके लिए $\vec u \cdot \vec v$ निकालिए:
\[
\vec u\,(2, 5),\ \vec v\,(3, -1); \qquad
\vec u\,(1, -4),\ \vec v\,(8, 2); \qquad
\norm{\vec u} = 3,\ \norm{\vec v} = 4,\ \theta = \frac{\pi}{3} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:scal:2}
$ABC$ भुजा $6$ वाला समबाहु त्रिभुज है। $\vect{AB} \cdot \vect{AC}$ और $\vect{AB} \cdot \vect{BC}$ निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:scal:3}
$t$ के किस मान के लिए $\vec u\,(3, t)$ और $\vec v\,(t - 8, 1)$ \omterm{def:g11:scal:orthogonal}{लांबिक} हैं? और $t$ के किन मानों के
लिए $\vec u\,(t, 2)$ स्वयं में से $\vec v\,(4, 2)$ घटाने पर मिले \omterm{def:g11:vect:vector}{सदिश} के \omterm{def:g11:scal:orthogonal}{लांबिक} है, अर्थात् $\vec u \perp (\vec u - \vec v)$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:scal:4}
$\vec u\,(1, 2)$ और $\vec v\,(3, 1)$ के बीच का कोण $\theta$ निकालिए ($\cos\theta$ यथार्थ रूप में, फिर $\theta$
निकटतम अंश तक)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:5}
किसी त्रिभुज की भुजाएँ $b = 4$, $c = 6$ हैं और उनके बीच का कोण $\widehat A = \frac{2\pi}{3}$ है। तीसरी
भुजा $a$ निकालिए। फिर भुजाओं $a = 7$, $b = 5$, $c = 4$ वाले त्रिभुज में $\cos\widehat A$
निकालिए और तय कीजिए कि $\widehat A$ न्यून कोण है या अधिक कोण।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:6}
मान लीजिए $A(1, 1)$, $B(5, -1)$ और $C(3, 5)$ हैं। $\vect{AB} \cdot \vect{AC}$ निकालिए; क्या कोण $\widehat A$ समकोण है?
तीनों भुजाओं की लंबाइयाँ निकालिए और अपने उत्तर की जाँच \omterm{def:g11:trigo:cossin}{कोज्या} नियम से कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:7}
केंद्र $\Omega(-2, 3)$ और त्रिज्या $5$ वाले वृत्त का \omterm{def:g10:algebra:equation}{समीकरण} बताइए। फिर वृत्त
\[
x^2 + y^2 - 6x + 2y - 6 = 0 .
\]
का केंद्र और त्रिज्या ज्ञात कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:8}
मान लीजिए $A(-3, 0)$ और $B(0, 2)$ हैं। \cref{prop:g11:scal:diameter} का उपयोग करके व्यास $[AB]$ वाले वृत्त का
\omterm{def:g10:algebra:equation}{समीकरण} ज्ञात कीजिए, और जाँचिए कि मूल बिंदु $O(0,0)$ उस पर है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:9}
$P(1, 3)$ से होकर $d : 2x + y - 7 = 0$ पर लंब रेखा का \omterm{def:g10:algebra:equation}{समीकरण}, और लंब के पाद $H$ के \omterm{def:g10:coordgeom:system}{निर्देशांक}
(दोनों रेखाओं का प्रतिच्छेद) ज्ञात कीजिए। इससे $P$ की $d$ से दूरी
निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:scal:10}
$\norm{\vec u + \vec v}^2 = \norm{\vec u}^2 + 2\vec u\cdot\vec v +
\norm{\vec v}^2$ और $\vec u - \vec v$ के लिए उसके समरूप कथन का उपयोग करके \emph{\omterm{def:g11:seq:arithmetic}{समांतर} चतुर्भुज
सर्वसमिका} सिद्ध कीजिए:
\[
\norm{\vec u + \vec v}^2 + \norm{\vec u - \vec v}^2
= 2\norm{\vec u}^2 + 2\norm{\vec v}^2 ,
\]
और \omterm{def:g11:seq:arithmetic}{समांतर} चतुर्भुज में उसकी व्याख्या कीजिए (विकर्ण बनाम भुजाएँ)।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:scal:11}
मान लीजिए $A$ और $B$ ऐसे दो बिंदु हैं कि $AB = 4$, और $I$ $[AB]$ का
\omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} है।
\begin{enumerate}
  \item दिखाइए कि हर बिंदु $M$ के लिए $\vect{MA}\cdot\vect{MB} = MI^2 - 4$। ($I$ डालिए: $\vect{MA} = \vect{MI} + \vect{IA}$।)
  \item इससे उन बिंदुओं $M$ का समुच्चय निकालिए जिनके लिए $\vect{MA}\cdot\vect{MB} = 12$ हो।
\end{enumerate}
\end{exercise}

\section{समस्या: अदिश गुणनफल के दो उपहार}

\begin{problem}[{सप्ताहांत समस्या --- एक ही बिंदु-गुणनफल से योग सूत्र निकल
आते हैं, दूसरे से रेखा तक की दूरी, और तीनों शीर्षलंबों का एक बिंदु पर मिलना
मुफ़्त में}]
\label{pb:g11:scal:1}
\omterm{def:g11:scal:dot}{अदिश गुणनफल} देखने में बही-खाता लगता है --- गुणा कीजिए, जोड़िए --- फिर भी वह
लगभग मुफ़्त में दो ऐसे ख़ज़ाने सौंप देता है जिन्होंने सदियों तक कोण-दर-कोण
पीछा करने वालों को टिकने नहीं दिया: त्रिकोणमिति के \emph{योग सूत्र} ($\cos 15^\circ$
का यथार्थ मान निकालिए!) और एक स्वच्छ भिन्न में \emph{बिंदु से रेखा तक की
दूरी}। विदाई-उपहार के रूप में वह यह भी सिद्ध कर देता है कि हर त्रिभुज के
तीनों शीर्षलंब एक ही बिंदु से गुज़रते हैं। \cref{thm:g11:scal:faces} के चारों चेहरे ड्यूटी पर
हाज़िर होंगे।

\medskip
\noindent\textbf{भाग I --- प्रवाह।}
\begin{enumerate}
  \item $\vec u(3, 4)$ और $\vec v(-1, 2)$ के लिए: $\vec u \cdot \vec v$, दोनों \omterm{def:g11:scal:dot}{परिमाण}, और \omterm{def:g10:vectors:vector}{सदिशों} के बीच का
        कोण निकालिए (अंश के दसवें भाग तक)।
  \item ऐसा $k$ ज्ञात कीजिए कि $(2, k)$ और $(3, -4)$ \omterm{def:g11:scal:orthogonal}{लांबिक} हों (\cref{def:g11:scal:orthogonal})।
  \item भौतिकी वाला चेहरा: एक स्लेज को ज़मीन से $60^\circ$ के कोण पर लगे $50$
        N बल से $10$ m खींचा जाता है। कार्य $W = \vec F \cdot \vec d$ निकालिए।
  \item \omterm{def:g11:trigo:cossin}{कोज्या} नियम (\cref{thm:g11:scal:cosines}): $5$ और $7$ की दो भुजाएँ $60^\circ$ का कोण घेरे
        हुए हैं; तीसरी भुजा यथार्थ रूप में ज्ञात कीजिए।
  \item उलटा: किसी त्रिभुज की भुजाएँ $4$, $6$, $8$ हैं। भुजा $8$ के
        सामने वाले कोण की \omterm{def:g11:trigo:cossin}{कोज्या} निकालिए, कोण का वर्गीकरण कीजिए, और उसे अंश
        के दसवें भाग तक बताइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- पहला उपहार: योग सूत्र।}
मान लीजिए $\vec u = (\cos a, \sin a)$ और $\vec v = (\cos b, \sin b)$ हैं: कोणों $a$ और $b$ पर दो इकाई \omterm{def:g11:vect:vector}{सदिश}।
\begin{enumerate}[resume]
  \item $\vec u \cdot \vec v$ दो बार निकालिए --- एक बार \omterm{def:g10:coordgeom:system}{निर्देशांकों} से, एक बार \omterm{def:g11:scal:dot}{परिमाणों}
        और \omterm{def:g10:vectors:vector}{सदिशों} के बीच के कोण से --- और मूल सूत्र निष्कर्ष निकालिए:
        \[
        \cos(a - b) = \cos a \cos b + \sin a \sin b .
        \]
  \item $b$ के स्थान पर $-b$ रखकर (\omterm{def:g11:trigo:cossin}{ज्या} और \omterm{def:g11:trigo:cossin}{कोज्या} की सम-विषमता!) $\cos(a + b)$
        प्राप्त कीजिए।
  \item $x = a + b$ पर लगाई गई \omterm{def:g10:proba:operations}{पूरक} सर्वसमिका $\sin x = \cos\left(\frac\pi2 - x\right)$ से $\sin(a + b)$ प्राप्त कीजिए।
  \item अंततः यथार्थ मान: $\cos 15^\circ$ ($\cos(45^\circ - 30^\circ)$ के रूप में) और $\sin 75^\circ$ निकालिए। दोनों
        की संख्यात्मक जाँच कीजिए।
  \item $b = a$ रखकर द्विकोण सूत्र निकालिए: $\cos 2a$ (उसके तीनों रूपों में) और
        $\sin 2a$।
  \item $\cos 2a = 1 - 2\sin^2 a$ से $\sin^2 15^\circ$ निकालिए, फिर $\sin 15^\circ$ को नीड़ित करणी के रूप में ---
        और सत्यापित कीजिए कि उसका वर्ग उस मान $\frac{\sqrt6 - \sqrt2}{4}$ से मेल खाता है जो
        प्रश्न 9 की विधि देती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- दूसरा उपहार: रेखा तक की दूरी।}
\begin{enumerate}[resume]
  \item मान लीजिए $d$ रेखा $ax + by + c = 0$ है, जिसका अभिलंब \omterm{def:g11:vect:vector}{सदिश} $\vec n(a, b)$ है (\cref{prop:g11:scal:normal}),
        और $P(x_0, y_0)$ कोई बिंदु है। $d$ पर कोई भी बिंदु $A$ लेकर समझाइए कि
        $P$ की $d$ से दूरी $\dfrac{\abs{\vect{AP} \cdot \vec n}}{\norm{\vec n}}$ क्यों है, और सूत्र निष्कर्ष निकालिए:
        \[
        \operatorname{dist}(P, d)
        = \frac{\abs{a x_0 + b y_0 + c}}{\sqrt{a^2 + b^2}} .
        \]
        $P(5, 6)$ की $3x + 4y - 12 = 0$ से दूरी निकालिए।
  \item $(5, 6)$ पर केंद्रित और उस रेखा को स्पर्श करने वाले वृत्त का \omterm{def:g10:algebra:equation}{समीकरण}
        लिखिए।
  \item सूत्र की जाँच: मूल बिंदु की $3x + 4y - 12 = 0$ से दूरी निकालिए, फिर मूल बिंदु से
        डाले गए लंब का पाद स्पष्ट रूप से निकालकर उसकी पुष्टि कीजिए।
  \item त्रिभुज $A(0,0)$, $B(6,0)$, $C(2,5)$: आधार $[AB]$ लेकर उसका क्षेत्रफल
        निकालिए; फिर $B$ की रेखा $(AC)$ से दूरी निकालिए और जाँचिए कि उससे
        वही क्षेत्रफल मिलता है।
  \item केंद्र $(2, 1)$ और त्रिज्या $2$ वाला वृत्त, रेखा $x + y - 6 = 0$ के सामने:
        केंद्र की रेखा से दूरी निकालिए और उनकी स्थिति का वर्गीकरण कीजिए
        (छेदी, स्पर्शी, असंयुक्त)।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- बोनस: शीर्षलंब संगामी हैं।}
\begin{enumerate}[resume]
  \item \cref{exo:g11:scal:11} को सामान्य कीजिए: \omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} $I$ वाले किन्हीं दो बिंदुओं
        $A$, $B$ के लिए सिद्ध कीजिए
        \[
        \vect{MA} \cdot \vect{MB} = MI^2 - \frac{AB^2}{4}
        \quad\text{प्रत्येक बिंदु } M \text{ के लिए} .
        \]
  \item \cref{prop:g11:scal:diameter} एक पंक्ति में निकालिए: जिन बिंदुओं $M$ के लिए $\vect{MA} \cdot \vect{MB} = 0$ हो
        उनका समुच्चय व्यास $[AB]$ वाला वृत्त है।
  \item शीर्षलंब: मान लीजिए त्रिभुज $ABC$ में $A$ से और $B$ से खींचे
        गए शीर्षलंबों का प्रतिच्छेद $H$ है (इसलिए $\vect{HA} \cdot \vect{BC} = 0$ और $\vect{HB} \cdot \vect{CA} = 0$)।
        सर्वसमिका
        \[
        \vect{HA} \cdot \vect{BC} + \vect{HB} \cdot \vect{CA}
        + \vect{HC} \cdot \vect{AB} = 0
        \]
        सिद्ध कीजिए --- शाल की सहायता से सब कुछ $H$ से खोलिए --- और
        निष्कर्ष निकालिए कि $H$ $C$ से खींचे गए शीर्षलंब पर भी है:
        तीनों शीर्षलंब संगामी हैं (\emph{लंबकेंद्र}\index{लंबकेंद्र} पर)।
  \item समापन: चार चेहरे, तीन ट्रॉफ़ियाँ। \cref{thm:g11:scal:faces} का कौन-सा चेहरा योग सूत्र
        उठाए हुए था, कौन-सा दूरी-सूत्र, कौन-सा \omterm{pb:g11:scal:1}{लंबकेंद्र} --- और ``एक ही
        राशि दो बार निकालिए'' पूरे अध्याय का आदर्श वाक्य क्यों है?
\end{enumerate}
\end{problem}
