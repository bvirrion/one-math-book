\chapter{फलन और उनके परिवर्तन}\label{ch:g11:func}

यह अध्याय उन बातों को व्यवस्थित करता है जो कलन से पहले \omterm{def:g10:functions:function}{फलनों} के बारे में जान
लेनी चाहिए: \emph{\omterm{def:g11:func:reference}{आधार फलनों}} का वह छोटा परिवार जिनके \omterm{def:g10:functions:graph}{आलेख} कंठस्थ होने
चाहिए, \emph{\omterm{def:g11:func:monotone}{वर्धमान}} और \emph{\omterm{def:g10:functions:variations}{ह्रासमान}} का ठीक-ठीक अर्थ, और यह कि \omterm{def:g10:functions:function}{फलनों} को
आपस में जोड़ने पर परिवर्तन कैसा व्यवहार करते हैं।

\section{फलन और वक्र}

\begin{definition}[फलन, प्रांत, वक्र]\label{def:g11:func:function}
\emph{फलन}\index{फलन} $f$ किसी समुच्चय $D \subseteq \R$ (उसका
\emph{प्रांत}\index{प्रांत}) की हर संख्या $x$ के साथ ठीक एक संख्या $f(x)$ जोड़
देता है। उसका \emph{वक्र} (या \omterm{def:g10:functions:graph}{आलेख}) $x \in D$ के लिए बिंदुओं $\bigl(x, f(x)\bigr)$ का समुच्चय है।
\end{definition}

\begin{example}
सूत्र $f(x) = \frac{\sqrt{x+2}}{x-1}$ के लिए $x + 2 \geq 0$ (वर्गमूल के लिए) और $x \neq 1$ (हर शून्येतर) आवश्यक हैं:
\omterm{def:g11:func:function}{प्रांत} $\intco{-2}{1} \cup \intoo{1}{+\infty}$ है।
\end{example}

\section{आधार फलन}

नीचे दिए पाँच \omterm{def:g11:func:function}{फलन} और उनके वक्र सहज-स्वाभाविक हो जाने चाहिए।

\begin{definition}[आधार फलन]\label{def:g11:func:reference}
\index{आधार फलन}
\[
x \mapsto x^2 \text{ (वर्ग)}, \quad
x \mapsto x^3 \text{ (घन)}, \quad
x \mapsto \sqrt{x}, \quad
x \mapsto \frac{1}{x}, \quad
x \mapsto \abs{x}.
\]
उनके \omterm{def:g11:func:function}{प्रांत} क्रमशः $\R$, $\R$, $\intco{0}{+\infty}$, $\R \setminus \{0\}$ और $\R$ हैं। $x \mapsto \frac1x$ का वक्र एक
\emph{अतिपरवलय}\index{अतिपरवलय} है।
\end{definition}

\section{किसी अंतराल पर परिवर्तन}

\begin{definition}[वर्धमान, ह्रासमान]\label{def:g11:func:monotone}
मान लीजिए $f$ किसी \omterm{def:g10:numbers:interval}{अंतराल} $I$ पर परिभाषित है। \omterm{def:g11:func:function}{फलन} $f$ $I$ पर
\emph{वर्धमान}\index{वर्धमान फलन} है यदि वह क्रम बनाए रखता है:
\[
u, v \in I \text{ के लिए}: \quad u < v \implies f(u) \leq f(v),
\]
और \emph{\omterm{def:g10:functions:variations}{ह्रासमान}} है यदि वह क्रम उलट देता है ($u < v \implies f(u) \geq
f(v)$)। दाहिनी ओर कड़ी
असमिकाएँ हों तो $f$ \emph{निरंतर} वर्धमान (क्रमशः \omterm{def:g10:functions:variations}{ह्रासमान}) है। जो \omterm{def:g11:func:function}{फलन}
$I$ पर वर्धमान या \omterm{def:g10:functions:variations}{ह्रासमान} हो वह $I$ पर \emph{एकदिष्ट} कहलाता है।
\end{definition}

\begin{method}[परिभाषा से परिवर्तन सिद्ध करना]\label{met:g11:func:variation}
\omterm{def:g10:numbers:interval}{अंतराल} में कोई भी दो संख्याएँ $u < v$ लीजिए और अंतर $f(v) - f(u)$ का चिह्न तय कीजिए,
प्रायः उसका \emph{\omterm{def:g10:algebra:expand}{गुणनखंडन}} करके। यदि \omterm{def:g10:numbers:interval}{अंतराल} पर चिह्न अचर हो, तो निष्कर्ष
निकाल लीजिए।
\end{method}

\begin{proposition}[आधार फलनों के परिवर्तन]\label{prop:g11:func:refvariations}
\begin{enumerate}
  \item $x^2$ $\intoc{-\infty}{0}$ पर निरंतर \omterm{def:g10:functions:variations}{ह्रासमान} और $\intco{0}{+\infty}$ पर निरंतर \omterm{def:g11:func:monotone}{वर्धमान} है।
  \item $x^3$ $\R$ पर निरंतर \omterm{def:g11:func:monotone}{वर्धमान} है।
  \item $\sqrt x$ $\intco{0}{+\infty}$ पर निरंतर \omterm{def:g11:func:monotone}{वर्धमान} है।
  \item $\frac1x$ $\intoo{-\infty}{0}$ पर और $\intoo{0}{+\infty}$ पर निरंतर \omterm{def:g10:functions:variations}{ह्रासमान} है (पर उनके \omterm{def:g10:numbers:interunion}{सम्मिलन} पर
        \emph{नहीं})।
  \item $\abs x$ $\intoc{-\infty}{0}$ पर निरंतर \omterm{def:g10:functions:variations}{ह्रासमान} और $\intco{0}{+\infty}$ पर निरंतर \omterm{def:g11:func:monotone}{वर्धमान} है।
\end{enumerate}
\end{proposition}

\begin{proof}
मान लीजिए $u < v$ है। हर स्थिति में हम $f(v) - f(u)$ का \omterm{def:g10:algebra:expand}{गुणनखंडन} करते हैं।

\emph{1.} $v^2 - u^2 = (v-u)(v+u)$। $\intco{0}{+\infty}$ पर: $v - u > 0$ और $v + u > 0$ (क्योंकि $v > u \geq 0$), इसलिए $v^2 > u^2$। $\intoc{-\infty}{0}$ पर:
$v + u < 0$, इसलिए $v^2 < u^2$।

\emph{2.} $v^3 - u^3 = (v - u)(v^2 + uv + u^2)$, और दूसरा गुणनखंड $\left(v + \frac u2\right)^2 + \frac{3u^2}{4} > 0$ के बराबर है, जब तक $u = v = 0$ न हो। अतः
जब भी $v > u$ हो, $v^3 > u^3$।

\emph{3.} $0 \leq u < v$ के लिए: $\sqrt v - \sqrt u = \frac{(\sqrt v - \sqrt u)(\sqrt v + \sqrt u)}
{\sqrt v + \sqrt u} = \frac{v - u}{\sqrt v + \sqrt u} > 0$।

\emph{4.} $0 < u < v$ के लिए: $\frac1v - \frac1u = \frac{u - v}{uv} < 0$, क्योंकि $u - v < 0$ और $uv > 0$; $u < v < 0$ के लिए वही गणना।
\omterm{def:g10:numbers:interunion}{सम्मिलन} पर यह दावा टूट जाता है: $-1 < 1$, पर $\frac{1}{-1} = -1 < 1 = \frac11$।

\emph{5.} $\intco{0}{+\infty}$ पर $\abs x = x$ निरंतर \omterm{def:g11:func:monotone}{वर्धमान} है; $\intoc{-\infty}{0}$ पर $\abs x = -x$ निरंतर \omterm{def:g10:functions:variations}{ह्रासमान} है।
\end{proof}

\begin{remark}
बिंदु 4 एक चिरपरिचित जाल है: ``$\intoo{-\infty}{0}$ पर और $\intoo{0}{+\infty}$ पर \omterm{def:g10:functions:variations}{ह्रासमान}'' का अर्थ
``$\R \setminus \{0\}$ पर \omterm{def:g10:functions:variations}{ह्रासमान}'' \emph{नहीं} है। परिवर्तन केवल \omterm{def:g10:numbers:interval}{अंतरालों} पर ही अर्थ
रखते हैं।
\end{remark}

\begin{example}[बिना गणना किए तुलना]
चूँकि $\sqrt x$ निरंतर \omterm{def:g11:func:monotone}{वर्धमान} है, $\sqrt{7} < \sqrt{8}$; और चूँकि $x^2$ $\intoc{-\infty}{0}$ पर निरंतर \omterm{def:g10:functions:variations}{ह्रासमान}
है, $(-3)^2 > (-2)^2$। एकदिष्टता \omterm{def:g10:functions:function}{प्रतिबिंबों} की तुलना को चरों की तुलना में बदल देती है।
\end{example}

\section{पुराने फलनों से नए फलन}

\begin{proposition}[रूपांतरण]\label{prop:g11:func:transforms}
मान लीजिए $u$ किसी \omterm{def:g10:numbers:interval}{अंतराल} $I$ पर परिभाषित \omterm{def:g11:func:function}{फलन} है और $k \in \R$ है।
\begin{enumerate}
  \item $u + k$ के परिवर्तन $I$ पर $u$ जैसे \emph{ही} हैं; उसका वक्र $u$
        का वक्र ही है, जो ऊर्ध्वाधर दिशा में $k$ खिसका हुआ है।
  \item $\lambda > 0$ के लिए $\lambda u$ के परिवर्तन $u$ जैसे हैं; $\lambda < 0$ के लिए
        परिवर्तन उलट जाते हैं।
  \item यदि $I$ पर $u \geq 0$ हो, तो $\sqrt u$ के परिवर्तन $u$ जैसे हैं।
  \item यदि $u \neq 0$ हो और $I$ पर $u$ का चिह्न अचर हो, तो $\frac1u$ के
        परिवर्तन $u$ के \emph{विपरीत} हैं।
\end{enumerate}
\end{proposition}

\begin{proof}
मान लीजिए $I$ में $u < v$ है (संकेतन का थोड़ा दुरुपयोग करते हुए हम $x$
के मान $u < v$ की तुलना करते हैं; $x_1 < x_2$ के \omterm{def:g10:functions:function}{प्रतिबिंबों} को $s = u(x_1)$, $t = u(x_2)$ लिखिए)।

\emph{1.} $(u+k)(x_2) - (u+k)(x_1) = t - s$: \omterm{def:g10:functions:function}{प्रतिबिंबों} का अंतर अपरिवर्तित रहता है, इसलिए उसका चिह्न
भी।

\emph{2.} $\lambda t - \lambda s = \lambda(t - s)$: $\lambda > 0$ होने पर चिह्न बना रहता है, $\lambda < 0$ होने पर पलट जाता
है।

\emph{3.} यदि $u$ \omterm{def:g11:func:monotone}{वर्धमान} हो और $x_1 < x_2$ हो, तो $0 \leq s \leq t$, और $\sqrt{}$ के \omterm{def:g11:func:monotone}{वर्धमान}
होने से (\cref{prop:g11:func:refvariations}) $\sqrt s \leq \sqrt t$ मिलता है। \omterm{def:g10:functions:variations}{ह्रासमान} स्थिति में भी वही तर्क।

\emph{4.} यदि $u$ \omterm{def:g11:func:monotone}{वर्धमान} हो और $0 < s \leq t$ (या $s \leq t < 0$) हो, तो $\frac1s \geq \frac1t$, क्योंकि
$x \mapsto \frac1x$ $0$ के दोनों ओर \omterm{def:g10:functions:variations}{ह्रासमान} है: क्रम उलट जाता है।
\end{proof}

\begin{example}\label{ex:g11:func:composite}
$\R$ पर $f(x) = \sqrt{x^2 + 1}$ का अध्ययन कीजिए। भीतरी \omterm{def:g11:func:function}{फलन} $u(x) = x^2 + 1$ धनात्मक है, $\intoc{-\infty}{0}$ पर
\omterm{def:g10:functions:variations}{ह्रासमान} और $\intco{0}{+\infty}$ पर \omterm{def:g11:func:monotone}{वर्धमान} (खिसका हुआ वर्ग)। बिंदु 3 से $f$ के भी वही
परिवर्तन हैं: पहले \omterm{def:g10:functions:variations}{ह्रासमान} फिर \omterm{def:g11:func:monotone}{वर्धमान}, और \omterm{def:g10:functions:extrema}{न्यूनतम} $f(0) = 1$।
\end{example}

\section{सम और विषम फलन}

\begin{definition}[सम-विषमता]\label{def:g11:func:parity}
मान लीजिए $f$ ऐसे \omterm{def:g11:func:function}{प्रांत} $D$ पर परिभाषित है जो $0$ के परितः सममित है
(अर्थात् जब भी $x \in D$ हो तब $-x \in D$ भी)। \omterm{def:g11:func:function}{फलन} $f$ \emph{सम}\index{सम फलन}
है यदि सभी $x \in D$ के लिए $f(-x) = f(x)$ हो, और \emph{विषम}\index{विषम फलन} है यदि
सभी $x \in D$ के लिए $f(-x) = -f(x)$ हो।
\end{definition}

\begin{proposition}[ज्यामितीय अर्थ]\label{prop:g11:func:paritysym}
\omterm{def:g11:func:parity}{सम फलन} का वक्र $y$-अक्ष के परितः सममित होता है; \omterm{def:g11:func:parity}{विषम फलन} का वक्र मूल
बिंदु के परितः सममित होता है।
\end{proposition}

\begin{proof}
यदि $f$ \omterm{def:g11:func:parity}{सम} हो और $(x, y)$ वक्र पर हो (अर्थात् $y = f(x)$), तो $f(-x) = f(x) = y$: दर्पण-बिंदु
$(-x, y)$ भी वक्र पर है। यदि $f$ \omterm{def:g11:func:parity}{विषम} हो, तो $f(-x) = -y$: बिंदु $(-x, -y)$, जो मूल बिंदु
के परितः $(x,y)$ का सममित है, वक्र पर है।
\end{proof}

\begin{example}
$x^2$ और $\abs x$ \omterm{def:g11:func:parity}{सम} हैं; $x^3$ और $\frac1x$ \omterm{def:g11:func:parity}{विषम} हैं; $f(x) = x^2 + x$ इनमें से कोई नहीं:
$f(-1) = 0$ जबकि $f(1) = 2$, इसलिए $f(-1) \neq \pm f(1)$। सम-विषमता आधा काम बचा देती है: \omterm{def:g11:func:parity}{सम} या \omterm{def:g11:func:parity}{विषम
फलन} का अध्ययन केवल $\intco{0}{+\infty}$ पर करना पर्याप्त है।
\end{example}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:5}
\cref{met:g11:func:variation} ($f(v) - f(u)$ का चिह्न) का उपयोग करके सिद्ध कीजिए कि $f(x) = x^2 - 4x$ $\intco{2}{+\infty}$ पर निरंतर
\omterm{def:g11:func:monotone}{वर्धमान} है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:6}
कोई अवकलज निकाले बिना इनके परिवर्तन बताइए:
\[
f(x) = \sqrt{5 - x} \ \intoc{-\infty}{5} \text{ पर},
\quad
g(x) = \frac{1}{x^2 + 1} \ \intco{0}{+\infty} \text{ पर},
\quad
h(x) = 3 - 2\sqrt{x} \ \intco{0}{+\infty} \text{ पर}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:7}
मान लीजिए $\R$ पर $f(x) = (x-2)^2 - 3$ है।
\begin{enumerate}
  \item $f$ के परिवर्तन और उसका \omterm{def:g10:functions:extrema}{न्यूनतम} बताइए।
  \item $g(x) = \frac{1}{(x-2)^2+1}$ के परिवर्तन और उसका \omterm{def:g10:functions:extrema}{अधिकतम} निकालिए। (ध्यान दीजिए कि $g = \frac{1}{f + 4}$।)
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:8}
\omterm{def:g11:func:function}{फलन} $f$ का वक्र बिंदुओं $(0, 1)$, $(2, 3)$ और $(4, 1)$ से होकर जाता है, और $f$
$\intcc{0}{2}$ पर \omterm{def:g11:func:monotone}{वर्धमान} तथा $\intcc{2}{4}$ पर \omterm{def:g10:functions:variations}{ह्रासमान} है। एक संभव वक्र का रेखाचित्र बनाइए,
फिर $f + 2$, $-f$ और $2f$ के वक्रों के रेखाचित्र बनाइए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:9}
दिखाइए कि \omterm{def:g11:func:function}{फलन} $f(x) = \frac{2x+1}{x+1}$ को $f(x) = 2 - \frac{1}{x+1}$ लिखा जा सकता है, और इससे $\intoo{-1}{+\infty}$ पर उसके
परिवर्तन निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:func:10}
सत्य या असत्य (पुष्ट कीजिए या प्रतिउदाहरण दीजिए):
\begin{enumerate}
  \item $I$ पर दो \omterm{def:g11:func:monotone}{वर्धमान फलनों} का योग $I$ पर \omterm{def:g11:func:monotone}{वर्धमान} होता है।
  \item $I$ पर दो \omterm{def:g11:func:monotone}{वर्धमान फलनों} का गुणनफल $I$ पर \omterm{def:g11:func:monotone}{वर्धमान} होता है।
  \item यदि $f$ $I$ पर \omterm{def:g11:func:monotone}{वर्धमान} हो, तो $f^2$ ($= f \times f$) $I$ पर \omterm{def:g11:func:monotone}{वर्धमान}
        होता है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:func:11}
मान लीजिए $x > 0$ के लिए $f(x) = x + \frac{1}{x}$ है।
\begin{enumerate}
  \item दिखाइए कि $0 < u < v$ के लिए $f(v) - f(u) = (v - u)\,\frac{uv - 1}{uv}$।
  \item इससे निकालिए कि $f$ $\intoc{0}{1}$ पर निरंतर \omterm{def:g10:functions:variations}{ह्रासमान} और $\intco{1}{+\infty}$ पर निरंतर
        \omterm{def:g11:func:monotone}{वर्धमान} है।
  \item निष्कर्ष निकालिए कि सभी $x > 0$ के लिए $x + \frac1x \geq 2$ है, और समता केवल $x = 1$
        पर है।
\end{enumerate}
\end{exercise}

\section{समस्या: सममिति का कारख़ाना}

\begin{problem}[{सप्ताहांत समस्या --- हर फलन एक सम भाग और एक विषम भाग का
योग है: सम-विषमता के नियम, रूपांतरण का खेल, और कलन के बिना जीते गए न्यूनतम}]
\label{pb:g11:func:1}
कुछ \omterm{def:g11:func:function}{फलन} दर्पण-सममित होते हैं, कुछ अर्ध-घूर्णन-सममित, और अधिकांश दोनों में से
कोई नहीं --- फिर भी बीजगणित का एक छोटा-सा चमत्कार कहता है कि \emph{हर} \omterm{def:g11:func:function}{फलन},
ठीक एक ही तरह से, एक दर्पण-सममित टुकड़े और एक अर्ध-घूर्णन-सममित टुकड़े का योग
है। यह समस्या उस चमत्कार को सिद्ध करती है, आधार वक्रों (\cref{prop:g11:func:transforms}) पर रूपांतरण
का खेल खेलती है, और \cref{exo:g11:func:11} की अंतर-गुणनखंडन वाली तकनीक पर समाप्त होती है, जो
अवकलज के आने से पूरा एक अध्याय पहले ही न्यूनतमकरण की समस्याएँ जीत लेती है।

\medskip
\noindent\textbf{भाग I --- सम-विषमता में प्रवाह।}
\begin{enumerate}
  \item \omterm{def:g11:func:parity}{सम}, \omterm{def:g11:func:parity}{विषम}, या कोई नहीं (\cref{def:g11:func:parity})? हर एक को पुष्ट कीजिए: $x^4 - 3x^2$; \
        $x^3 - x$; \ $x^2 + x$; \ $\abs x$; \ $\frac1x$।
  \item हर निर्णय का ज्यामितीय अर्थ बताइए (\cref{prop:g11:func:paritysym}): \omterm{def:g11:func:parity}{सम फलन} के वक्र में
        कौन-सी सममिति होती है? \omterm{def:g11:func:parity}{विषम} के वक्र में?
  \item सम-विषमता चिह्नों की तरह गुणा होती है: सिद्ध कीजिए कि दो \omterm{def:g11:func:parity}{विषम
        फलनों} का गुणनफल \omterm{def:g11:func:parity}{सम} होता है, और पूरी गुणन-सारणी पूरी कीजिए (\omterm{def:g11:func:parity}{सम} $\times$
        \omterm{def:g11:func:parity}{सम}, \omterm{def:g11:func:parity}{सम} $\times$ \omterm{def:g11:func:parity}{विषम}, \omterm{def:g11:func:parity}{विषम} $\times$ \omterm{def:g11:func:parity}{विषम})।
  \item तुरंत निकलने वाले परिणाम: यदि $f$ \omterm{def:g11:func:parity}{विषम} हो और $0$ पर परिभाषित
        हो, तो $f(0)$ क्या है? यदि $f$ \omterm{def:g11:func:parity}{सम} हो और $f(3) = 7$ हो, तो $f(-3)$ क्या
        है? और दिखाइए कि \omterm{def:g11:func:parity}{सम} और \omterm{def:g11:func:parity}{विषम} दोनों होने वाला अकेला \omterm{def:g11:func:function}{फलन} शून्य \omterm{def:g11:func:function}{फलन} है।
  \item \omterm{def:g11:func:parity}{सम फलन} के वक्र का कितना हिस्सा पूरे वक्र को तय कर देता है? ऐसा एक
        \omterm{def:g11:func:function}{फलन} दीजिए जो न \omterm{def:g11:func:parity}{सम} हो न \omterm{def:g11:func:parity}{विषम}, और समझाइए कि वहाँ क्या टूट जाता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- विघटन प्रमेय।}
$0$ के परितः सममित किसी \omterm{def:g11:func:function}{प्रांत} पर परिभाषित \omterm{def:g11:func:function}{फलन} $f$ के लिए रखिए
\[
E(x) = \frac{f(x) + f(-x)}{2},
\qquad
O(x) = \frac{f(x) - f(-x)}{2} .
\]
\begin{enumerate}[resume]
  \item $f(x) = x^3 + x^2 + 1$ के लिए $E$ और $O$ निकालिए, और जाँचिए: $E$ \omterm{def:g11:func:parity}{सम}, $O$
        \omterm{def:g11:func:parity}{विषम}, $E + O = f$।
  \item सामान्य कथन सिद्ध कीजिए: किसी भी $f$ के लिए \omterm{def:g11:func:function}{फलन} $E$ \omterm{def:g11:func:parity}{सम} है,
        $O$ \omterm{def:g11:func:parity}{विषम} है, और $f = E + O$ --- \emph{हर \omterm{def:g11:func:function}{फलन} एक \omterm{def:g11:func:parity}{सम} भाग और एक \omterm{def:g11:func:parity}{विषम} भाग
        में बँट जाता है}।
  \item सिद्ध कीजिए कि यह विभाजन अद्वितीय है: यदि $f = E_1 + O_1 = E_2 + O_2$ हो, जहाँ $E_i$ \omterm{def:g11:func:parity}{सम}
        और $O_i$ \omterm{def:g11:func:parity}{विषम} हैं, तो $E_1 - E_2$ पर प्रश्न 4 का उपयोग कीजिए।
  \item $f(x) = \dfrac{1}{1 - x}$ का विघटन कीजिए ($x \neq \pm 1$ के लिए): $E$ और $O$ को उभयनिष्ठ हर
        $1 - x^2$ पर लाइए और सरल कीजिए।
  \item आगे की एक झलक, एक-एक वाक्य में: चरघातांकी \omterm{def:g11:func:function}{फलन} के \omterm{def:g11:func:parity}{सम} और \omterm{def:g11:func:parity}{विषम} भाग
        कक्षा 12 के जुड़वाँ \omterm{def:g11:func:function}{फलन} होंगे (अतिपरवलयिक कोज्या और ज्या); और वही
        प्रवृत्ति --- किसी वस्तु को सममित टुकड़ों में बाँटकर हर टुकड़े को
        उसकी अपनी सममिति से सँभालना --- स्नातक खंडों में ध्वनि और संकेतों
        के विश्लेषण को चलाती है। सामान्यतः ऐसा विभाजन \emph{उपयोगी} क्यों
        है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- रूपांतरण का खेल।}
\begin{enumerate}[resume]
  \item $x \mapsto x^2$ के वक्र से शुरू करके $(x - 3)^2$, \ $x^2 + 2$, \ $(x - 3)^2 + 2$, \ $-x^2$, \
        $2x^2$ के वक्रों का ठीक-ठीक वर्णन कीजिए (\cref{prop:g11:func:transforms})।
  \item $f(x) = \abs{x - 2} + \abs{x + 2}$ को $-2$ और $2$ से कटे तीनों \omterm{def:g10:numbers:interval}{अंतरालों} पर खंडशः सूत्र में
        बदलिए, और उसके वक्र का वर्णन कीजिए (उसका एक प्रसिद्ध चपटा तल है)।
  \item कौन-से रूपांतरण समता का आदर करते हैं? उपपत्ति या प्रतिउदाहरण के साथ
        तय कीजिए: क्षैतिज खिसकाव $f(x - 3)$; ऊर्ध्वाधर खिसकाव $f(x) + 2$; ऊर्ध्वाधर
        खिंचाव $2 f(x)$ (सब किसी \omterm{def:g11:func:parity}{सम} $f$ पर लगाए गए)।
  \item प्रश्न 12 के खंडशः सूत्र का उपयोग करके $\abs{x - 2} + \abs{x + 2} = 6$ हल कीजिए।
  \item उलटी अभियांत्रिकी: किसी \omterm{def:g11:quad:parabola}{परवलय} का शीर्ष $(2, -3)$ है और वह $(0, 1)$ से होकर
        जाता है। उसका \omterm{def:g10:algebra:equation}{समीकरण} \omterm{thm:g11:quad:canonical}{मानक रूप} में फिर से बनाइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- कलन के बिना \omterm{def:g10:functions:extrema}{न्यूनतम}।}
\begin{enumerate}[resume]
  \item $\intoo{0}{+\infty}$ पर $g(x) = x + \dfrac4x$ के लिए दर्पण \cref{exo:g11:func:11}: दिखाइए कि $g(v) - g(u) = (v - u)\,\dfrac{uv - 4}{uv}$, \omterm{def:g10:functions:extrema}{न्यूनतम} का
        स्थान बताइए, और उसका मान दीजिए।
  \item अनुप्रयोग: क्षेत्रफल $16$ वाले सभी आयतों में से किसका परिमाप सबसे
        कम है? ($P = 2\left(x +
        \frac{16}{x}\right)$ लिखिए और प्रश्न 16 की मशीन काम में लाइए।) एक पुराना
        मित्र निष्कर्ष निकाल देता है।
  \item $h(x) = \sqrt{x^2 + 1}$ के परिवर्तन पूरे $\R$ पर ज्ञात कीजिए, और इसके लिए $\intco{0}{+\infty}$ पर
        संयोजन वाले तर्क को शेष के लिए सम-विषमता वाले तर्क के साथ जोड़िए।
  \item निकटतम बिंदु: वक्र $y = \sqrt x$ का वह बिंदु ज्ञात कीजिए जो $(3, 0)$ के सबसे
        निकट है। (दूरी के \emph{वर्ग} को \omterm{def:g10:functions:extrema}{न्यूनतम} कीजिए --- वह $x$ में
        द्विघात है --- और समझाइए कि वर्ग को \omterm{def:g10:functions:extrema}{न्यूनतम} करना उचित क्यों है।)
  \item समापन, कारख़ाने का चक्कर: आधार वक्र कच्चा माल हैं; रूपांतरण मशीनें;
        सम-विषमता और विघटन गुणवत्ता-नियंत्रण; और अंतर-गुणनखंडन माप-मेज़।
        हर एक पर एक वाक्य --- और उस बड़े औज़ार का नाम बताइए जो अगला अध्याय
        सौंपता है।
\end{enumerate}
\end{problem}
