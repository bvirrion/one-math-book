\chapter{त्रिकोणमिति: इकाई वृत्त}\label{ch:g11:trigo}

त्रिकोणमिति समकोण त्रिभुजों में जन्म लेती है, पर उसका असली घर \emph{इकाई
वृत्त} है, जहाँ \omterm{def:g11:trigo:cossin}{कोज्या} और \omterm{def:g11:trigo:cossin}{ज्या} किसी भी \omterm{def:g10:numbers:sets}{वास्तविक संख्या} के \omterm{def:g11:func:function}{फलन} बन जाते हैं।
यह अध्याय \omterm{def:g11:trigo:radian}{रेडियन}, वास्तविक रेखा को वृत्त पर लपेटना, कंठस्थ रखने योग्य यथार्थ
मान, और वे सममितियाँ स्थापित करता है जिनसे सभी चिरपरिचित सर्वसमिकाएँ जन्म
लेती हैं। $\cos$ और $\sin$ का \omterm{def:g10:functions:function}{फलनों} के रूप में अध्ययन --- परिवर्तन, \omterm{def:g11:deriv:derivative}{अवकलज} ---
\cref{ch:g12:trigo} में किया जाता है।

\section{रेडियन और लपेटना}

\begin{definition}[रेडियन]\label{def:g11:trigo:radian}
मान लीजिए $\mathcal C$ मूल बिंदु $O$ पर केंद्रित त्रिज्या $1$ का वृत्त है, और $I(1, 0)$ है।
$\mathcal C$ के केंद्र पर बने किसी कोण का \emph{रेडियन}\index{रेडियन} माप वह चाप-लंबाई
है जो वह $\mathcal C$ पर काटता है। चूँकि पूरे वृत्त की परिधि $2\pi$ है:
\[
360^\circ = 2\pi \text{ rad},
\qquad
180^\circ = \pi \text{ rad},
\qquad
1 \text{ rad} = \frac{180^\circ}{\pi} \approx 57.3^\circ .
\]
\end{definition}

\begin{method}[रूपांतरण]\label{met:g11:trigo:convert}
अंश और \omterm{def:g11:trigo:radian}{रेडियन} समानुपाती हैं: अंश से \omterm{def:g11:trigo:radian}{रेडियन} में जाने के लिए $\frac{\pi}{180}$ से गुणा
कीजिए, और उलटी दिशा में $\frac{180}{\pi}$ से। जैसे $60^\circ = 60 \times \frac{\pi}{180} = \frac{\pi}{3}$, और $\frac{3\pi}{4} = \frac{3 \times 180^\circ}{4} = 135^\circ$।
\end{method}

\begin{definition}[रेखा को वृत्त पर लपेटना]\label{def:g11:trigo:winding}
हर \omterm{def:g10:numbers:sets}{वास्तविक संख्या} $t$ के साथ $\mathcal C$ का वह बिंदु $M(t)$ जोड़िए जो $I$ से वृत्त
पर $\abs t$ दूरी चलकर मिलता है, $t \geq 0$ होने पर वामावर्त और $t < 0$ होने पर दक्षिणावर्त।
चूँकि परिधि $2\pi$ है, संख्याएँ $t$ और $t + 2k\pi$ ($k \in \Z$) एक ही बिंदु पर पहुँचती
हैं: $M(t + 2k\pi) = M(t)$।
\end{definition}

\begin{example}
$M(0) = I$; $M\!\left(\frac{\pi}{2}\right) = (0, 1)$, जो वृत्त का शीर्ष है; $M(\pi) = (-1, 0)$; $M\!\left(-\frac{\pi}{2}\right) = (0, -1)$; और $M\!\left(\frac{9\pi}{4}\right) = M\!\left(\frac{\pi}{4} +
2\pi\right) = M\!\left(\frac{\pi}{4}\right)$।
\end{example}

\section{कोज्या और ज्या}

\begin{definition}[कोज्या और ज्या]\label{def:g11:trigo:cossin}
हर वास्तविक $t$ के लिए $t$ की \emph{कोज्या}\index{कोज्या} और
\emph{ज्या}\index{ज्या} इकाई वृत्त के बिंदु $M(t)$ के \omterm{def:g10:coordgeom:system}{निर्देशांक} हैं:
\[
M(t) = (\cos t,\ \sin t).
\]
\end{definition}

बिंदु $M(t)$ पर पहुँचती है, और उसी के \omterm{def:g10:coordgeom:system}{निर्देशांक} $\cos t$ तथा $\sin t$ को
\emph{परिभाषित} करते हैं।}

\begin{proposition}[पहले गुण]\label{prop:g11:trigo:first}
सभी $t \in \R$ और $k \in \Z$ के लिए:
\[
-1 \leq \cos t \leq 1, \qquad
-1 \leq \sin t \leq 1, \qquad
\cos^2 t + \sin^2 t = 1,
\]
\[
\cos(t + 2k\pi) = \cos t, \qquad \sin(t + 2k\pi) = \sin t .
\]
\end{proposition}

\begin{proof}
$M(t)$ इकाई वृत्त पर है, इसलिए उसके \omterm{def:g10:coordgeom:system}{निर्देशांक} $-1$ और $1$ के बीच हैं, और वे
वृत्त के \omterm{def:g10:algebra:equation}{समीकरण} $x^2 + y^2 = 1$ को संतुष्ट करते हैं, जो सर्वसमिका $\cos^2 t + \sin^2 t = 1$ ही है। आवर्तिता
$M(t + 2k\pi) = M(t)$ (\cref{def:g11:trigo:winding}) को ही दोहराती है।
\end{proof}

\begin{example}\label{ex:g11:trigo:identity}
यदि $\sin t = \frac35$ और $t \in \intcc{\frac\pi2}{\pi}$ हो (दूसरा चतुर्थांश), तो $\cos^2 t = 1 - \frac{9}{25} = \frac{16}{25}$, इसलिए $\cos t = \pm\frac45$; दूसरे चतुर्थांश
में \omterm{def:g10:coordgeom:system}{भुज} ऋणात्मक होता है, अतः $\cos t = -\frac45$।
\end{example}

कंठस्थ रखने योग्य मान:
\begin{center}
\begin{tabular}{c|ccccc}
$t$ & $0$ & $\dfrac{\pi}{6}$ & $\dfrac{\pi}{4}$ & $\dfrac{\pi}{3}$ & $\dfrac{\pi}{2}$\\[6pt]
\hline\rule{0pt}{14pt}%
$\cos t$ & $1$ & $\dfrac{\sqrt3}{2}$ & $\dfrac{\sqrt2}{2}$ & $\dfrac12$ & $0$\\[6pt]
$\sin t$ & $0$ & $\dfrac12$ & $\dfrac{\sqrt2}{2}$ & $\dfrac{\sqrt3}{2}$ & $1$
\end{tabular}
\end{center}

\begin{remark}
याद रखने की एक तरकीब: ज्याएँ $\frac{\sqrt0}{2}, \frac{\sqrt1}{2}, \frac{\sqrt2}{2}, \frac{\sqrt3}{2},
\frac{\sqrt4}{2}$ की तरह पढ़ी जाती हैं, और कोज्याएँ वही सूची
उल्टे क्रम में हैं।
\end{remark}

\section{संबद्ध कोण}

\begin{proposition}[संबद्ध कोण]\label{prop:g11:trigo:associated}
\index{संबद्ध कोण}
सभी $t \in \R$ के लिए:
\[
\begin{aligned}
\cos(-t) &= \cos t, & \sin(-t) &= -\sin t,\\
\cos(\pi - t) &= -\cos t, & \sin(\pi - t) &= \sin t,\\
\cos(\pi + t) &= -\cos t, & \sin(\pi + t) &= -\sin t,\\
\cos\left(\tfrac{\pi}{2} - t\right) &= \sin t, &
\sin\left(\tfrac{\pi}{2} - t\right) &= \cos t .
\end{aligned}
\]
\end{proposition}

\begin{proof}
हर पंक्ति वृत्त की एक सममिति व्यक्त करती है।

$M(-t)$ $x$-अक्ष में $M(t)$ का परावर्तन है: \omterm{def:g10:coordgeom:system}{भुज} अपरिवर्तित रहता है, कोटि का
चिह्न बदल जाता है।

$M(\pi - t)$ $y$-अक्ष में $M(t)$ का परावर्तन है: $\pi$ से पीछे की ओर $t$ चलना $0$
से आगे की ओर $t$ चलने के (क्षैतिज रूप से) सामने पहुँचता है। \omterm{def:g10:coordgeom:system}{भुज} का चिह्न
बदलता है, कोटि अपरिवर्तित रहती है।

$M(\pi + t)$ $M(t)$ के व्यास के दूसरे सिरे पर है (आधा चक्कर): दोनों \omterm{def:g10:coordgeom:system}{निर्देशांकों} के
चिह्न बदल जाते हैं।

$M\!\left(\frac\pi2 - t\right)$ विकर्ण रेखा $y = x$ में $M(t)$ का परावर्तन है, जो दोनों \omterm{def:g10:coordgeom:system}{निर्देशांक} आपस
में बदल देती है।
\end{proof}

$\cos\frac{2\pi}{3} = \cos\left(\pi - \frac\pi3\right)

= -\sin\frac\pi6 = -\frac12$। पाँच मानों की सारणी, सममितियों से बढ़ाकर, पूरे वृत्त को ढक
लेती है।
\end{example}

\section{त्रिकोणमितीय समीकरण हल करना}

\begin{theorem}[समीकरण $\cos t = \cos a$ और $\sin t = \sin a$]
\label{thm:g11:trigo:equations}
\omterm{def:g10:numbers:sets}{वास्तविक संख्याओं} $t$ और $a$ के लिए:
\[
\cos t = \cos a \iff t = a + 2k\pi \ \text{या}\ t = -a + 2k\pi
\quad (k \in \Z),
\]
\[
\sin t = \sin a \iff t = a + 2k\pi \ \text{या}\ t = \pi - a + 2k\pi
\quad (k \in \Z).
\]
\end{theorem}

\begin{proof}
इकाई वृत्त के दो बिंदुओं के \omterm{def:g10:coordgeom:system}{भुज} ठीक तब बराबर होते हैं जब वे या तो बराबर हों
या $x$-अक्ष में एक-दूसरे के परावर्तन हों; \cref{prop:g11:trigo:associated} से $M(a)$ का परावर्तन $M(-a)$
है। इसलिए $\cos t = \cos a$ का अर्थ है $M(t) = M(a)$ या $M(t) = M(-a)$, अर्थात् $2\pi$ के किसी गुणज तक
$t = \pm a$। इसी तरह कोटियों का बराबर होना बिंदुओं के बराबर होने या $y$-अक्ष में
परावर्तन होने का अर्थ रखता है, और $M(a)$ का परावर्तन $M(\pi - a)$ है।
\end{proof}

\begin{method}[किसी अंतराल पर $\cos t = c$ हल करना]\label{met:g11:trigo:solve}
\begin{enumerate}
  \item ज्ञात मानों की सारणी से \emph{एक} ऐसा कोण $a$ ज्ञात कीजिए जिसके
        लिए $\cos a = c$ हो;
  \item व्यापक हल $t = \pm a + 2k\pi$ लिखिए (\cref{thm:g11:trigo:equations});
  \item वे \omterm{def:g10:numbers:sets}{पूर्णांक} $k$ चुनिए जो माँगे गए \omterm{def:g10:numbers:interval}{अंतराल} में ले जाएँ, और हल
        गिनाइए।
\end{enumerate}
$\sin t = c$ के लिए $t = a + 2k\pi$ या $t = \pi - a + 2k\pi$ के साथ इसी तरह आगे बढ़िए।
\end{method}

\begin{example}\label{ex:g11:trigo:solve}
$\intoc{-\pi}{\pi}$ पर $\cos t = \frac12$ हल कीजिए। $\cos a = \frac12$ का एक हल $a = \frac{\pi}{3}$ है। व्यापक हल $t = \frac\pi3 + 2k\pi$ और $t = -\frac\pi3 + 2k\pi$
हैं। $\intoc{-\pi}{\pi}$ के भीतर केवल $k = 0$ ही योगदान देता है: $t \in \left\{-\frac\pi3,\ \frac\pi3\right\}$।

$\intco{0}{2\pi}$ पर $\sin t = -\frac{\sqrt2}{2}$ हल कीजिए। एक कोण $a = -\frac\pi4$ है; हल $t = -\frac\pi4 + 2k\pi$ और $t = \pi + \frac\pi4 + 2k\pi = \frac{5\pi}{4} + 2k\pi$ हैं। $\intco{0}{2\pi}$ में:
$t = \frac{7\pi}{4}$ ($k = 1$ से) और $t = \frac{5\pi}{4}$।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g11:trigo:1}
\omterm{def:g11:trigo:radian}{रेडियन} में बदलिए: $30^\circ$, $45^\circ$, $120^\circ$, $270^\circ$। अंशों में बदलिए: $\frac{\pi}{5}$, $\frac{5\pi}{6}$,
$\frac{7\pi}{4}$, $\frac{2\pi}{9}$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:trigo:2}
हर एक को $2\pi$ के सापेक्ष $\intoc{-\pi}{\pi}$ के किसी मान पर लाकर बिंदु $M(t)$ इकाई वृत्त
पर अंकित कीजिए:
\[
t = \frac{2\pi}{3},\quad
t = -\frac{\pi}{4},\quad
t = \frac{17\pi}{6},\quad
t = -\frac{7\pi}{2},
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:trigo:3}
सारणी और \omterm{prop:g11:trigo:associated}{संबद्ध कोणों} का उपयोग करके यथार्थ मान बताइए:
\[
\cos\frac{3\pi}{4}, \quad
\sin\frac{5\pi}{6}, \quad
\cos\left(-\frac{\pi}{3}\right), \quad
\sin\frac{4\pi}{3}, \quad
\cos\frac{11\pi}{6} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:trigo:4}
$t \in \intoo{-\frac\pi2}{0}$ के साथ $\cos t = \frac{5}{13}$ दिया हो, तो $\sin t$ का यथार्थ मान निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:5}
किसी भी वास्तविक $x$ के लिए सरल कीजिए:
\[
A = \cos(\pi + x) + \cos(-x) + \sin\left(\frac\pi2 - x\right) - \cos x,
\qquad
B = \sin(\pi - x) + \sin(\pi + x) + \sin(-x) .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:6}
$\intoc{-\pi}{\pi}$ पर हल कीजिए:
\[
\cos t = \frac{\sqrt3}{2}, \qquad
\sin t = \frac12, \qquad
\cos t = -1 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:7}
$\intco{0}{2\pi}$ पर हल कीजिए:
\[
\sin t = -\frac{\sqrt3}{2}, \qquad
\cos t = 0, \qquad
2\sin t + 1 = 0 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:8}
$\R$ पर \omterm{def:g10:algebra:equation}{समीकरण} $\cos 2t = \cos t$ हल कीजिए। ($a = t$ के साथ \cref{thm:g11:trigo:equations} का उपयोग कीजिए, और हलों
के दोनों परिवारों की चर्चा कीजिए।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:9}
दिखाइए कि सभी वास्तविक $x$ के लिए

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:trigo:10}
त्रिज्या $30$ cm का एक पहिया बिना फिसले लुढ़कता है। जब साइकिल $1.5$ m आगे
बढ़ती है तब वह कितने कोण से (पहले \omterm{def:g11:trigo:radian}{रेडियन} में, फिर अंशों में) घूमता है? पहिये
के एक पूरे चक्कर में साइकिल कितनी दूर बढ़ती है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:trigo:11}
$\intoc{-\pi}{\pi}$ पर असमिका $\cos t \leq \frac12$ हल कीजिए। (इकाई वृत्त का रेखाचित्र बनाइए, वह भाग
अंकित कीजिए जहाँ \omterm{def:g10:coordgeom:system}{भुज} अधिक से अधिक $\frac12$ है, और $t$ के मानों का \omterm{def:g10:numbers:interval}{अंतराल} पढ़
लीजिए।)
\end{exercise}

\section{समस्या: विशाल चक्र और ज्वार}

\begin{problem}[{सप्ताहांत समस्या --- रेडियन लुढ़की हुई दूरी के रूप में, और
इकाई वृत्त हर आवर्ती परिघटना की मुख्य घड़ी के रूप में}]
\label{pb:g11:trigo:1}
विशाल चक्र आपको एक वृत्त पर घुमाता है; ज्वार बंदरगाह के पानी को उसी गणितीय
वृत्त पर ऊपर-नीचे ले जाता है। हर आवर्ती परिघटना --- पहिये, ज्वार, ध्वनि,
ऋतुएँ --- अपनी समय-सारणी इसी अध्याय के इकाई वृत्त (\cref{def:g11:trigo:cossin}) से पढ़ती है, और
\cref{met:g11:trigo:solve} के \omterm{def:g10:algebra:equation}{समीकरण} ठीक-ठीक समय बता देते हैं। यह समस्या रूपांतरण करती है, लपेटती
है, प्रतिरूप बनाती है और हल करती है --- और रास्ते में यह भी तय कर देती है कि
गणितज्ञ कोण \omterm{def:g11:trigo:radian}{रेडियन} में क्यों नापते हैं।

\medskip
\noindent\textbf{भाग I --- \omterm{def:g11:trigo:radian}{रेडियन}, लुढ़की हुई दूरी।}
\begin{enumerate}
  \item \omterm{def:g11:trigo:radian}{रेडियन} में बदलिए: $30^\circ$, $135^\circ$, $300^\circ$; और अंशों में: $\frac{3\pi}{4}$, $\frac{5\pi}{6}$
        (\cref{met:g11:trigo:convert})।
  \item \omterm{def:g11:trigo:radian}{रेडियन} का सारा मर्म: त्रिज्या $r$ वाले वृत्त पर कोण $t$ (\omterm{def:g11:trigo:radian}{रेडियन}
        में) का चाप ठीक $r\,t$ लंबा होता है। त्रिज्या $3$ m वाले वृत्त पर
        $2$ \omterm{def:g11:trigo:radian}{रेडियन} का कोण कितना चाप काटता है? (कहीं कोई $\pi$ नहीं ---
        यही तो बात है।)
  \item त्रिज्या $30$ cm का एक पहिया बिना फिसले $1$ km लुढ़कता है। वह
        कितने \emph{\omterm{def:g11:trigo:radian}{रेडियन}} घूमा है, और वे कितने पूरे चक्कर हुए?
  \item इकाई वृत्त पर रखिए और यथार्थ मान निकालिए: $\cos\frac{2\pi}{3}$, $\sin\left(-\frac{\pi}{4}\right)$, और $\cos\frac{19\pi}{6}$
        (पहले $2\pi$ के सापेक्ष घटाइए; \cref{prop:g11:trigo:associated})।
  \item कोई कोण $t \in \intoo{\frac\pi2}{\pi}$ $\sin t = \frac35$ को संतुष्ट करता है। $\cos^2 t + \sin^2 t = 1$ और चतुर्थांश का
        उपयोग करके $\cos t$ तथा $\tan t$ ज्ञात कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- विशाल चक्र।}
एक विशाल चक्र की त्रिज्या $60$ m है, उसका केंद्र ज़मीन से $65$ m ऊपर है,
और वह हर $30$ मिनट में एक चक्कर पूरा करता है। आप समय $t = 0$ (मिनट में) पर
सबसे नीचे वाले बिंदु से सवार होते हैं, इसलिए आपकी ऊँचाई यह मानती है:
\[
h(t) = 65 - 60 \cos\!\left(\frac{\pi t}{15}\right).
\]
\begin{enumerate}[resume]
  \item सूत्र को पुष्ट कीजिए: $t$ मिनट बाद घूमा हुआ कोण जाँचिए, $t = 0$ पर
        ऊँचाई जाँचिए, और ऋण चिह्न की व्याख्या कीजिए।
  \item $t = 7.5$, $t = 15$ और $t = 20$ मिनट पर अपनी ऊँचाई निकालिए (यथार्थ मान)।
  \item आप \emph{पहली बार} $95$ m पर कब होते हैं? (\cref{met:g11:trigo:solve} के साथ $h(t) = 95$ हल
        कीजिए।)
  \item सवारी के किस समय-अंतराल में आप कम से कम $95$ m ऊँचाई पर होते हैं
        --- और कुल कितनी देर? (\cref{exo:g11:trigo:11} से तुलना कीजिए।)
  \item ऊँचाई सबसे \emph{तेज़ी} से कहाँ बदलती है? पहले मिनट की चढ़ाई ($h(1) - h(0)$)
        की तुलना मिनट $7$ और $8$ के बीच की चढ़ाई से कीजिए, और अपना
        प्रेक्षण शब्दों में रखिए (इसे यथार्थ बनाने वाला औज़ार ---
        ज्यावक्रीय \omterm{def:g11:func:function}{फलन} का \omterm{def:g11:deriv:derivative}{अवकलज} --- कक्षा 12 में आता है)।
  \item अभियंता की बारी: ऐसे चक्र का ऊँचाई-सूत्र रचिए जिसकी \omterm{def:g10:functions:extrema}{अधिकतम} ऊँचाई
        $40$ m, \omterm{def:g10:functions:extrema}{न्यूनतम} ऊँचाई $4$ m और आवर्तकाल $12$ मिनट हो, और सवारी
        $t = 0$ पर सबसे नीचे से शुरू हो।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- ज्वार और वृत्त की सममितियाँ।}
किसी बंदरगाह में आधी रात के $t$ घंटे बाद पानी की गहराई, मीटर में, यह है:
\[
d(t) = 7 + 3 \sin\!\left(\frac{\pi t}{6}\right).
\]
\begin{enumerate}[resume]
  \item \omterm{def:g10:functions:extrema}{अधिकतम} और \omterm{def:g10:functions:extrema}{न्यूनतम} गहराई, वे समय जब वे पहली बार आती हैं, और ज्वार का
        आवर्तकाल बताइए।
  \item एक मालवाहक जहाज़ को $8.5$ m पानी चाहिए। $\intco{0}{12}$ पर $d(t) \geq 8.5$ हल कीजिए: वह
        किस खिड़की में भीतर आ सकता है, और वह खिड़की कितनी लंबी है? अगली
        खिड़की कब खुलती है?
  \item \omterm{prop:g11:trigo:associated}{संबद्ध कोणों} की दो सर्वसमिकाएँ सीधे वृत्त की सममितियों से फिर
        निकालिए --- $\sin(\pi - t) = \sin t$ (ऊर्ध्वाधर अक्ष में दर्पण) और $\cos(\pi + t) = -\cos t$ (आधा चक्कर)
        --- और \omterm{def:g10:proba:operations}{पूरक} सर्वसमिका $\cos\left(\frac\pi2 - t\right) = \sin t$ बताइए।
  \item सम-विषमता (\cref{pb:g11:func:1} की शब्दावली): $\sin$, $\cos$ और $t \mapsto \sin t + t^3$ को सम या विषम
        बताइए, और वृत्त से एक-एक पंक्ति की उपपत्ति दीजिए।
  \item $\intco{0}{2\pi}$ पर हल कीजिए: $2\sin^2 t - \sin t - 1 = 0$। (भेस बदला हुआ द्विघात: \cref{pb:g11:quad:1} की तरह
        \omterm{def:g10:algebra:expand}{गुणनखंडन} कीजिए, फिर वृत्त पढ़ लीजिए।)
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- गणितज्ञ का कोण।}
\begin{enumerate}[resume]
  \item $1$ \omterm{def:g11:trigo:radian}{रेडियन} कितने अंश का होता है? और फिर कैलकुलेटर का एक जाल:
        $\sin(1)$ (\omterm{def:g11:trigo:radian}{रेडियन} विधा) और $\sin(1^\circ)$ में कौन बड़ा है --- और मोटे तौर पर
        कितने गुना? कैलकुलेटर की सेटिंग के बारे में क्या सीख मिली?
  \item वृत्त का उपयोग करके $\intco{0}{2\pi}$ पर $\cos t = \sin t$ हल कीजिए (\omterm{def:g10:coordgeom:system}{भुज} और कोटि कहाँ बराबर
        होते हैं?)।
  \item छोटे कोणों के लिए चाप और उसकी \omterm{def:g11:trigo:cossin}{ज्या} लगभग एक ही होते हैं: $\sin(0.1)$ की
        तुलना $0.1$ से कीजिए (पाँच दशमलव स्थान) और आपेक्षिक त्रुटि बताइए।
        यही ``लघु-कोण \omterm{def:g10:numbers:approx}{सन्निकटन}'' जहाज़ और लोलक चलाता है; इसके पीछे की प्रमेय
        ($\frac{\sin t}{t} \to 1$) कक्षा 12 के सीमाओं वाले अध्याय का सितारा है।
  \item समापन --- इकाई वृत्त मुख्य घड़ी के रूप में, हर मद पर एक वाक्य:
        \omterm{def:g11:trigo:radian}{रेडियन} कोणों को ईमानदार संख्याओं में बदल देते हैं (चाप $=$
        त्रिज्या $\times$ कोण); \omterm{def:g11:trigo:cossin}{कोज्या} और \omterm{def:g11:trigo:cossin}{ज्या} घड़ी की सुई की छाया और ऊँचाई
        हैं; वृत्त की सममितियाँ ही सर्वसमिकाएँ हैं; $\cos t = c$ हल करना घड़ी से
        समय पढ़ना है; और हर आवर्ती परिघटना --- पहिया, ज्वार, ध्वनि ---
        यही घड़ी है, बस वेश बदले हुए।
\end{enumerate}
\end{problem}
