\chapter{प्रायिकता और यादृच्छिक चर}\label{ch:g11:prob}

\omterm{def:g10:proba:distribution}{प्रायिकता} ऐसे प्रयोगों का प्रतिरूप बनाती है जिनका परिणाम अनिश्चित हो: पासे
का एक फेंक, निकाला गया एक पत्ता, खेला गया एक खेल। यह अध्याय पहले परिमित
समुच्चयों पर शब्दावली खड़ी करता है, फिर पूरे प्रायिकता-सिद्धांत का केंद्रीय
पात्र प्रस्तुत करता है: \emph{\omterm{def:g11:prob:rv}{यादृच्छिक चर}} और उसकी \emph{\omterm{def:g11:prob:expectation}{प्रत्याशा}}। इन
धारणाओं का पुनरावलोकन और गहराई \cref{ch:g12:randvar} में है, और सप्रतिबंध \omterm{def:g10:proba:distribution}{प्रायिकताएँ} \cref{ch:g12:condprob} में
आती हैं।

\section{परिमित समुच्चय पर प्रायिकता}

\begin{definition}[प्रायिकता प्रतिरूप]\label{def:g11:prob:model}
परिमित परिणामों वाले प्रयोग का प्रतिरूप उसकी \emph{प्रतिदर्श
समष्टि}\index{प्रतिदर्श समष्टि} $\Omega = \{\omega_1, \dots, \omega_n\}$ (परिणामों का समुच्चय) और उसके साथ एक
\emph{\omterm{def:g10:proba:distribution}{प्रायिकता}} $\P$ से बनता है, जो हर परिणाम $\omega_i$ को एक संख्या $p_i \geq 0$ देती है,
जहाँ $p_1 + \dots + p_n = 1$ हो। \emph{घटना}\index{घटना} कोई उपसमुच्चय $A \subseteq \Omega$ है, और
\[
\P(A) = \sum_{\omega_i \in A} p_i .
\]
जब सभी परिणाम समान रूप से संभावित हों ($p_i = \frac1n$: \emph{एकसमान
प्रतिरूप}\index{एकसमान प्रतिरूप}), तब
\[
\P(A) = \frac{A \text{ में परिणामों की संख्या}}
{\Omega \text{ में परिणामों की संख्या}}.
\]
\end{definition}

\begin{example}\label{ex:g11:prob:twodice}
दो अलग-अलग पहचाने जा सकने वाले पासे फेंकिए: $\Omega$ $1 \leq i, j \leq 6$ वाले $36$ क्रमित
युग्मों $(i, j)$ का समुच्चय है, और सभी समान रूप से संभावित हैं। \omterm{def:g11:prob:model}{घटना} $S$ =
``योग $7$ है'' में $6$ युग्म $(1,6)$, $(2,5)$, $(3,4)$, $(4,3)$, $(5,2)$, $(6,1)$ आते हैं,
इसलिए $\P(S) = \frac{6}{36} = \frac16$। \omterm{def:g11:prob:model}{घटना} ``योग $12$ है'' में केवल $(6,6)$ है: \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac{1}{36}$।
\omterm{def:g11:prob:model}{एकसमान प्रतिरूप} तभी लागू होता है जब \emph{सही} \omterm{def:g11:prob:model}{प्रतिदर्श समष्टि} चुनी जाए ---
क्रमित युग्म, अक्रमित नहीं।
\end{example}

\begin{proposition}[प्रायिकता के नियम]\label{prop:g11:prob:rules}
\omterm{def:g10:proba:events}{घटनाओं} $A, B \subseteq \Omega$ के लिए:
\[
\P(\emptyset) = 0, \qquad \P(\Omega) = 1, \qquad
\P(\bar A) = 1 - \P(A),
\]
\[
\P(A \cup B) = \P(A) + \P(B) - \P(A \cap B),
\]
जहाँ $\bar A$ $A$ का \emph{\omterm{def:g10:proba:operations}{पूरक}} है (वे सभी परिणाम जो $A$ में नहीं हैं)।
विशेष रूप से, जब $A$ और $B$ \emph{\omterm{def:g10:proba:operations}{असंगत}} हों ($A \cap B = \emptyset$), तब $\P(A \cup B) = \P(A) + \P(B)$।
\end{proposition}

\begin{proof}
रिक्त \omterm{def:g11:prob:model}{घटना} में कोई परिणाम नहीं है (रिक्त योग), और $\Omega$ में सब हैं (कुल योग
$1$)। हर परिणाम $A$, $\bar A$ में से ठीक एक में है, इसलिए $\P(A) + \P(\bar A) = 1$। \omterm{def:g10:numbers:interunion}{सम्मिलन} के
लिए: $p_i$ को $A$ पर और फिर $B$ पर जोड़ने से $A \cap B$ के परिणाम दो बार गिने
जाते हैं; $\P(A \cap B)$ घटाने से यह दोहरी गिनती सुधर जाती है।
\end{proof}

\begin{example}
किसी कक्षा में $60\%$ संगीत पढ़ते हैं ($M$), $45\%$ कोई खेल खेलते हैं ($S$),
और $25\%$ दोनों करते हैं। कम से कम एक करने वालों का अनुपात $\P(M \cup S) = 0.60 + 0.45 - 0.25 = 0.80$ है, इसलिए $20\%$
कुछ भी नहीं करते (\omterm{def:g10:proba:operations}{पूरक})।
\end{example}

\section{यादृच्छिक चर}

\begin{definition}[यादृच्छिक चर, बंटन]\label{def:g11:prob:rv}
$\Omega$ पर \emph{यादृच्छिक चर}\index{यादृच्छिक चर} वह \omterm{def:g11:func:function}{फलन} $X$ है जो हर परिणाम
को एक संख्या दे देता है। उसका \emph{बंटन}\index{बंटन} उसके संभव मानों $x_1, \dots, x_k$
और उनकी \omterm{def:g10:proba:distribution}{प्रायिकताओं} की सारणी है:
\[
\P(X = x_i) = \P\bigl(\{\omega : X(\omega) = x_i\}\bigr),
\qquad
\sum_{i=1}^k \P(X = x_i) = 1 .
\]
\end{definition}

\begin{example}\label{ex:g11:prob:game}
एक खेल: $2$ यूरो दीजिए, एक न्यायसंगत पासा फेंकिए, और यदि आया हुआ मान कम से
कम $5$ हो तो वही मान पाइए, अन्यथा कुछ नहीं। मान लीजिए $G$ \emph{लाभ} है
(मिला हुआ घटा दिया हुआ)। परिणाम $1, 2, 3, 4$ से $G = -2$ मिलता है; परिणाम $5$ से $G = 3$;
और परिणाम $6$ से $G = 4$। $G$ का \omterm{def:g11:prob:rv}{बंटन}:
\begin{center}
\begin{tabular}{c|ccc}
$g$ & $-2$ & $3$ & $4$\\
\hline\rule{0pt}{11pt}%
$\P(G = g)$ & $\dfrac46$ & $\dfrac16$ & $\dfrac16$
\end{tabular}
\end{center}
\end{example}

\begin{method}[बंटन-सारणी बनाना]\label{met:g11:prob:table}
$X$ के संभव मान गिनाइए; हर मान के लिए उसे देने वाले परिणाम इकट्ठे कीजिए और
उनकी \omterm{def:g10:proba:distribution}{प्रायिकताएँ} जोड़िए; और जाँचिए कि \omterm{def:g10:proba:distribution}{प्रायिकताओं} की पंक्ति का योग $1$ है।
\end{method}

\section{प्रत्याशा}

\begin{definition}[प्रत्याशा]\label{def:g11:prob:expectation}
$X$ की \emph{प्रत्याशा}\index{प्रत्याशा} (या \omterm{def:g11:stat:mean}{माध्य}) उसके मानों का वह औसत है
जो उनकी \omterm{def:g10:proba:distribution}{प्रायिकताओं} से भारित हो:
\[
\E(X) = \sum_{i=1}^{k} \P(X = x_i)\; x_i .
\]
\end{definition}

\begin{remark}
$\E(X)$ वही है जो प्रयोग की बहुत बड़ी संख्या में स्वतंत्र पुनरावृत्तियों का
\omterm{def:g11:stat:mean}{माध्य} होगा --- इस कथन का सुनिश्चित रूप बृहत् संख्याओं का नियम है, जो \cref{ch:g12:sums} में
सिद्ध होता है। बारंबारताओं वाले आँकड़ा-समुच्चय के \omterm{def:g11:stat:mean}{माध्य} (\cref{def:g11:stat:mean}) से औपचारिक
समानता पर ध्यान दीजिए: \omterm{def:g10:proba:distribution}{प्रायिकताएँ} आपेक्षिक बारंबारताओं $\frac{n_i}{n}$ की भूमिका
निभाती हैं।
\end{remark}

\begin{example}\label{ex:g11:prob:gameexp}
\cref{ex:g11:prob:game} के खेल के लिए:
\[
\E(G) = \frac46 \times (-2) + \frac16 \times 3 + \frac16 \times 4
= \frac{-8 + 3 + 4}{6} = -\frac16 \approx -0.17 .
\]
खिलाड़ी औसतन प्रति खेल लगभग $17$ सेंट गँवाता है: खेल प्रतिकूल है। जिस खेल
में प्रत्याशित लाभ $0$ हो उसे \emph{न्यायसंगत} कहते हैं।
\end{example}

\begin{proposition}[रैखिकता]\label{prop:g11:prob:linearity}
सभी वास्तविक $a, b$ के लिए:
\[
\E(aX + b) = a\,\E(X) + b .
\]
\end{proposition}

\begin{proof}
चर $aX + b$ \omterm{def:g10:proba:distribution}{प्रायिकता} $\P(X = x_i)$ के साथ मान $a x_i + b$ लेता है, इसलिए
\[
\E(aX + b) = \sum_i \P(X = x_i)(a x_i + b)
= a \sum_i \P(X = x_i)\,x_i + b \underbrace{\sum_i \P(X =
x_i)}_{=\,1} = a\,\E(X) + b .
\]
\end{proof}

\section{प्रसरण और मानक विचलन}

\begin{definition}[प्रसरण]\label{def:g11:prob:variance}
$X$ का \emph{प्रसरण}\index{प्रसरण} \omterm{def:g11:prob:expectation}{प्रत्याशा} $m = \E(X)$ के आसपास उसके मानों का
प्रसार नापता है:
\[
\V(X) = \E\bigl((X - m)^2\bigr)
= \sum_{i=1}^{k} \P(X = x_i)\,(x_i - m)^2,
\qquad
\sigma(X) = \sqrt{\V(X)} .
\]
\end{definition}

\begin{proposition}[संक्षिप्त सूत्र]\label{prop:g11:prob:konig}
\[
\V(X) = \E(X^2) - \E(X)^2 .
\]
\end{proposition}

\begin{proof}
योग के भीतर $(x_i - m)^2 = x_i^2 - 2m\,x_i + m^2$ का प्रसार कीजिए:
\[
\V(X) = \sum_i \P(X = x_i)\,x_i^2
- 2m \sum_i \P(X = x_i)\,x_i + m^2\sum_i \P(X = x_i)
= \E(X^2) - 2m^2 + m^2 ,
\]
जो $\E(X^2) - m^2$ है। यही उस सर्वसमिका का यादृच्छिक-चर वाला रूप है जो आँकड़ों के लिए
\cref{prop:g11:stat:shortcut} में सिद्ध हुई थी।
\end{proof}

\begin{proposition}[एकघात रूपांतरण]\label{prop:g11:prob:affine}
$\V(aX + b) = a^2\,\V(X)$ और $\sigma(aX + b) = \abs a\,\sigma(X)$।
\end{proposition}

\begin{proof}
रैखिकता से $aX + b$ अपनी \omterm{def:g11:prob:expectation}{प्रत्याशा} $a m + b$ से $(aX + b) - (am + b) = a(X - m)$ जितना हटता है: हर विचलन $a$
गुना हो जाता है, अतः हर वर्ग-विचलन $a^2$ गुना, और उनका भारित \omterm{def:g11:stat:mean}{माध्य} भी।
खिसकाव $b$ ग़ायब हो गया --- किसी चर को खिसकाने से उसका प्रसार नहीं बदलता।
\end{proof}

\begin{example}\label{ex:g11:prob:variance}
\cref{ex:g11:prob:game} के खेल के लिए: $\E(G^2) = \frac46 \times 4 + \frac16 \times 9 + \frac16 \times 16
= \frac{16 + 9 + 16}{6} = \frac{41}{6}$, इसलिए
\[
\V(G) = \frac{41}{6} - \left(-\frac16\right)^2
= \frac{41}{6} - \frac{1}{36} = \frac{245}{36} \approx 6.8,
\qquad
\sigma(G) \approx 2.6 .
\]
प्रति खेल औसतन $0.17$ की हानि के साथ लगभग $2.6$ यूरो के प्रारूपिक झूले भी
आते हैं --- \omterm{def:g11:stat:variance}{मानक विचलन} के बिना अकेली \omterm{def:g11:prob:expectation}{प्रत्याशा} आधी ही कहानी कहती है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g11:prob:1}
$52$ पत्तों की मानक गड्डी से एक पत्ता निकाला जाता है। ये \omterm{def:g10:proba:distribution}{प्रायिकताएँ}
निकालिए: पान का पत्ता निकलना; बादशाह निकलना; पान या बादशाह निकलना; न पान न
बादशाह निकलना।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:prob:2}
दो न्यायसंगत पासे फेंके जाते हैं और उनका योग $S$ दर्ज किया जाता है। \cref{ex:g11:prob:twodice}
के $36$-परिणाम वाले प्रतिरूप का उपयोग करके $\P(S = 6)$, $\P(S \geq 10)$ और $\P(S \text{ सम है})$ निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:prob:3}
किसी \omterm{def:g11:prob:rv}{यादृच्छिक चर} का \omterm{def:g11:prob:rv}{बंटन} यह है:
\begin{center}
\begin{tabular}{c|cccc}
$x$ & $-1$ & $0$ & $2$ & $5$\\
\hline
$\P(X=x)$ & $0.3$ & $0.2$ & $0.4$ & $p$
\end{tabular}
\end{center}
$p$ ज्ञात कीजिए, फिर $\E(X)$ निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:prob:4}
$X$ $\E(X) = 3$ और $\V(X) = 4$ को संतुष्ट करता है। $Y = 2X - 5$ की \omterm{def:g11:prob:expectation}{प्रत्याशा}, \omterm{def:g11:prob:variance}{प्रसरण} और \omterm{def:g11:stat:variance}{मानक
विचलन} बताइए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:5}
एक भाग्य-चक्र में $10$ बराबर खंड हैं: $5$ पर $0$ यूरो, $3$ पर $2$
यूरो, और $2$ पर $5$ यूरो लिखा है। खेलने का शुल्क $2$ यूरो है। लाभ $G$
(जीत घटा दाँव) का \omterm{def:g11:prob:rv}{बंटन} बताइए, $\E(G)$ निकालिए, और तय कीजिए कि खेल अनुकूल है या
नहीं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:6}
एक कलश में $3$ लाल और $2$ नीली गेंदें हैं; दो गेंदें \emph{बिना} वापसी
के निकाली जाती हैं। मान लीजिए $X$ निकली हुई लाल गेंदों की संख्या है। $X$
का \omterm{def:g11:prob:rv}{बंटन} बनाइए ($20$ क्रमित निकालों को गिनाकर, या \omterm{met:g10:proba:tree}{वृक्ष} से), और $\E(X)$
निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:7}
\cref{exo:g11:prob:5} के भाग्य-चक्र के लिए $\V(G)$ और $\sigma(G)$ निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:8}
एक बीमा कंपनी $80$ यूरो में एक बीमा-पत्र बेचती है। \omterm{def:g10:proba:distribution}{प्रायिकता} $0.05$ से
ग्राहक $1000$ यूरो का दावा करता है, \omterm{def:g10:proba:distribution}{प्रायिकता} $0.01$ से $5000$ यूरो का, और
अन्यथा कुछ नहीं। मान लीजिए $B$ एक बीमा-पत्र पर कंपनी का लाभ है। $B$ का
\omterm{def:g11:prob:rv}{बंटन} बताइए, $\E(B)$ निकालिए, और उसकी व्याख्या कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:9}
एक बहुविकल्पीय प्रश्न में $4$ उत्तर हैं, जिनमें एक सही है। सही उत्तर पर
$+1$ अंक मिलते हैं; अनुमान लगाने से रोकने के लिए ग़लत उत्तर पर $x < 0$ अंक
मिलते हैं। एक विद्यार्थी यादृच्छिक रूप से उत्तर देता है। $x$ के किस मान के
लिए विद्यार्थी के अंकों की \omterm{def:g11:prob:expectation}{प्रत्याशा} शून्य है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:prob:10}
दो न्यायसंगत सिक्के उछाले जाते हैं। ऐन प्रस्ताव रखती है: ``यदि दोनों सिक्के
मेल खाएँ तो मैं आपको $1$ यूरो दूँगी; यदि अलग-अलग हों तो आप मुझे $1$ यूरो
देंगे।'' दोनों खिलाड़ियों का प्रत्याशित लाभ निकालिए। क्या खेल न्यायसंगत है?
यही प्रश्न तीन सिक्कों के साथ, जहाँ तीनों मेल खाने पर ऐन $2$ यूरो देती है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:prob:11}
एक लॉटरी $5$ यूरो प्रति टिकट के हिसाब से $1000$ टिकट बेचती है। एक टिकट पर
$2000$ यूरो, पाँच टिकटों पर $200$ यूरो और बीस टिकटों पर $25$ यूरो जीते जाते
हैं।
\begin{enumerate}
  \item आयोजक की कुल आय घटा पुरस्कार का, और किसी एक खिलाड़ी के लाभ $G$ का
        \omterm{def:g11:prob:rv}{बंटन} बताइए।
  \item $\E(G)$ और आयोजक का कुल प्रत्याशित लाभ निकालिए। जाँचिए कि दोनों उत्तर
        आपस में मेल खाते हैं।
\end{enumerate}
\end{exercise}

\section{समस्या: अधूरा खेल}

\begin{problem}[{सप्ताहांत समस्या --- पास्कल, फ़र्मा और बीच में रुके दाँव का
न्यायसंगत बँटवारा: प्रत्याशा का जन्म, रैखिकता का जादू, और वह जो प्रत्याशा
नहीं देख पाती}]
\label{pb:g11:prob:1}
1654 में शेवालिये दे मेरे --- वही जुआरी जिसके पासों के दाँवों से माध्यमिक
विद्यालय खंड की \omterm{def:g10:proba:distribution}{प्रायिकता} वाली समस्या शुरू हुई थी --- ने ब्लेज़ पास्कल से एक
गहरा प्रश्न पूछा: बराबर कौशल वाले दो खिलाड़ियों को अपनी शृंखला $5$ और $3$
के स्कोर पर रोक देनी पड़ती है, जबकि छह जीतें $64$-पिस्टोल का पूरा दाँव ले
जातीं। \emph{दाँव का न्यायसंगत बँटवारा कैसे हो?} इसका उत्तर देने वाले
पास्कल और फ़र्मा के पत्रों ने इस अध्याय के हृदय में बसी धारणा को जन्म दिया:
\omterm{def:g11:prob:expectation}{प्रत्याशा} (\cref{def:g11:prob:expectation})। आप उनका उत्तर दोनों तरीक़ों से फिर से निकालेंगे, फिर
रैखिकता का तमाशा देखेंगे और \omterm{def:g11:prob:expectation}{प्रत्याशा} का इकलौता अंधा कोना भी।

\medskip
\noindent\textbf{भाग I --- \omterm{def:g11:prob:expectation}{प्रत्याशा} में प्रवाह।}
\begin{enumerate}
  \item एक न्यायसंगत पासा फेंकिए; आपका लाभ (फलक $- 3$) यूरो है। लाभ $G$
        की बंटन-सारणी बनाइए (\cref{met:g11:prob:table}) और $\E(G)$ निकालिए।
  \item संक्षिप्त सूत्र (\cref{prop:g11:prob:konig}) से $V(G)$ और $\sigma(G)$ निकालिए (दो दशमलव स्थान)।
  \item बिना कोई नई सारणी बनाए $\E(2G - 1)$ और $V(2G - 1)$ बताइए (\cref{prop:g11:prob:affine})।
  \item एक दुकान दो पासे फेंकने के लिए $m$ यूरो लेती है और आपको योग के
        बराबर यूरो दे देती है। \emph{न्यायसंगत} दाम $m$ क्या है? दुकान
        $8$ लेती है: प्रति खेल आपकी प्रत्याशित हानि कितनी है?
  \item $800$ यूरो का एक लैपटॉप हर वर्ष $5\,\%$ संभावना से नष्ट हो सकता है।
        न्यायसंगत बीमा प्रीमियम निकालिए, और यदि बीमाकर्ता $60$ यूरो ले तो
        प्रति बीमा-पत्र उसका प्रत्याशित लाभ भी।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दाँव का बँटवारा।}
स्कोर $5$--$3$, और $64$ पिस्टोल पहले $6$ जीतने वाला ले जाएगा; हर दौर
न्यायसंगत सिक्के का उछाल है।
\begin{enumerate}[resume]
  \item दो लुभावने ग़लत उत्तर: $5 : 3$ के अनुपात में बाँट देना (जीते गए दौरों
        के अनुपात में), या सब कुछ आगे चल रहे खिलाड़ी को दे देना। एक-एक वाक्य
        में बताइए कि हर एक में अन्याय क्या है।
  \item फ़र्मा की विधि: खिलाड़ी A को $1$ और जीत चाहिए, B को $3$; कल्पना
        कीजिए कि चाहे जो हो, ठीक $3$ दौर और खेले जाते हैं। $8$ समसंभावी
        परिणाम गिनाइए, हर एक में शृंखला का विजेता अंकित कीजिए, और $64$
        पिस्टोल का न्यायसंगत बँटवारा निकालिए।
  \item वह सूक्ष्म बात जिसका बचाव फ़र्मा को करना पड़ा: असली खेल एक ही दौर के
        बाद रुक सकता है --- फिर भी सभी $3$ काल्पनिक दौरों पर तर्क करना
        उचित क्यों है?
  \item पास्कल की विधि, अंत से पीछे की ओर: (काल्पनिक) स्कोरों $5$--$5$,
        फिर $5$--$4$, फिर $5$--$3$ पर A का न्यायसंगत हिस्सा
        निकालिए, और हर बार दोनों समसंभावी भविष्यों का औसत लीजिए। प्रश्न 7
        से तुलना कीजिए।
  \item यही प्रश्न स्कोर $4$--$3$ पर: कितने काल्पनिक दौर, कौन-से परिणाम
        B को शृंखला दिलाते हैं, और न्यायसंगत बँटवारा क्या है?
  \item दोनों विधियों में साझा सिद्धांत बताइए --- किसी अनिश्चित भविष्य का
        न्यायसंगत मूल्य उसकी \emph{\omterm{def:g11:prob:expectation}{प्रत्याशा}} है --- और आज के दो ऐसे उद्योग
        गिनाइए जो पूरी तरह इसी पर चलते हैं (प्रश्न 5 एक दिखा देता है)।
  \item हर स्कोर के लिए एक जाँच: \omterm{def:g11:prob:expectation}{प्रत्याशा} की रैखिकता (\cref{prop:g11:prob:linearity}) से समझाइए कि
        दोनों खिलाड़ियों के न्यायसंगत हिस्सों का योग सदा ठीक $64$-पिस्टोल
        के दाँव के बराबर क्यों होना चाहिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- रैखिकता का तमाशा।}
\begin{enumerate}[resume]
  \item $36$-ख़ानों वाली सारणी बनाए बिना, रैखिकता (\cref{prop:g11:prob:linearity}) से $\E(\text{दो पासों का योग}) = 7$ एक
        पंक्ति में फिर निकालिए। योग के \emph{\omterm{def:g11:prob:variance}{प्रसरण}} के लिए नियम $V(X + Y) = V(X) + V(Y)$ को
        पासों का स्वतंत्र होना चाहिए (स्वीकृत): उसे निकालिए।
  \item टोपी वाली दावत: $n$ अतिथि अपनी टोपियाँ एक ढेर में फेंक देते हैं
        और हर एक यादृच्छिक रूप से एक टोपी उठा लेता है। मान लीजिए $X$ उन
        अतिथियों की गिनती करता है जिन्हें अपनी ही टोपी वापस मिलती है। किसी
        एक नियत अतिथि के लिए अपनी टोपी वापस पाने की \omterm{def:g10:proba:distribution}{प्रायिकता} क्या है? इन
        $n$ संभावनाओं को जोड़कर (रैखिकता!) $\E(X)$ निकालिए। उत्तर आपको चौंका
        देगा: वह $n$ पर निर्भर ही नहीं करता।
  \item $n = 2$ और $n = 3$ के लिए $\E(X) = 1$ की जाँच सीधे गिनकर कीजिए ($2$ और
        $6$ समसंभावी टोपी-बँटवारे गिनाइए और गिनतियों का औसत लीजिए)।
  \item अतिथियों की वापसियाँ स्वतंत्र होने से कोसों दूर हैं (यदि $n - 1$
        अतिथियों को अपनी टोपी मिल गई, तो अंतिम को भी मिलनी ही है)। प्रश्न
        14 की गणना में स्वतंत्रता की आवश्यकता ठीक-ठीक कहाँ \emph{नहीं}
        पड़ी? रैखिकता की महाशक्ति बताइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- वह जो \omterm{def:g11:prob:expectation}{प्रत्याशा} नहीं देख पाती।}
\begin{enumerate}[resume]
  \item $5$ यूरो के दो खुरचन-टिकट: टिकट A \omterm{def:g10:proba:distribution}{प्रायिकता} $\frac12$ से $10$ यूरो
        जिताता है; टिकट B \omterm{def:g10:proba:distribution}{प्रायिकता} $0.005$ से $1\,000$ यूरो। जाँचिए कि दोनों
        लाभों की \omterm{def:g11:prob:expectation}{प्रत्याशा} एक ही है, फिर दोनों \omterm{def:g11:prob:variance}{प्रसरण} निकालिए। असली लोग
        कौन-सा टिकट ख़रीदते हैं, और वे असल में ख़रीद क्या रहे हैं?
  \item सेंट पीटर्सबर्ग खेल: पहली पट आने तक उछालते जाइए; यदि पहली पट उछाल
        $n$ पर आए तो आपको $2^n$ यूरो मिलते हैं। \omterm{def:g11:prob:expectation}{प्रत्याशा} में हर $n$ का
        योगदान और फिर \omterm{def:g11:prob:expectation}{प्रत्याशा} स्वयं निकालिए। क्या आप एक बार खेलने के लिए
        $1\,000$ यूरो देंगे? यह टकराव \omterm{def:g11:prob:expectation}{प्रत्याशा} के बारे में क्या सिखाता है?
  \item ग्राहक की ओर से बीमा: $20$ यूरो दीजिए, या $1\,000$ गँवाने का $1\,\%$
        जोखिम उठाइए? दोनों प्रत्याशाओं की तुलना कीजिए, और समझाइए कि बीमा
        ख़रीदना फिर भी विवेकपूर्ण क्यों हो सकता है (एक अकेला परिवार किस
        चीज़ पर ``औसत'' नहीं ले सकता?)।
  \item समापन --- \omterm{def:g11:prob:expectation}{प्रत्याशा} की जीवनी, हर अध्याय पर एक वाक्य: 1654 में एक
        दाँव बाँटने के लिए जन्म; उसकी महाशक्ति, रैखिकता, जो स्वतंत्रता पूछे
        बिना प्रत्याशाएँ जोड़ लेती है (टोपियाँ); उसका अंधा कोना, जोखिम,
        जिसे \omterm{def:g11:prob:variance}{प्रसरण} नापता है (टिकट, सेंट पीटर्सबर्ग, बीमा); और अगले वर्ष
        उसका राज्याभिषेक, जब बृहत् संख्याओं का नियम ठीक-ठीक समझा देगा कि
        जुआघर हमेशा क्यों जीतता है।
\end{enumerate}
\end{problem}
