\chapter{अनुक्रम: पहला पाठ्यक्रम}\label{ch:g11:seq}

\omterm{def:g11:seq:sequence}{अनुक्रम} किसी नियम से बनी संख्याओं की सूची है: बचत खाते के क्रमिक शेष, या
वर्ष-दर-वर्ष किसी समष्टि का आकार। यह अध्याय उन दो परिवारों का अध्ययन करता है
जो अनुप्रयोगों पर छाए हुए हैं --- \emph{\omterm{def:g11:seq:arithmetic}{समांतर}} \omterm{def:g11:seq:sequence}{अनुक्रम}, जो बराबर डग भरकर
बढ़ते हैं, और \emph{\omterm{def:g11:seq:geometric}{गुणोत्तर}} \omterm{def:g11:seq:sequence}{अनुक्रम}, जो बराबर अनुपात से बढ़ते हैं। सीमाओं
का कठोर सिद्धांत \cref{ch:g12:seq} में विकसित होता है।

\section{अनुक्रम को परिभाषित करना}

\begin{definition}[अनुक्रम]\label{def:g11:seq:sequence}
\emph{अनुक्रम}\index{अनुक्रम} $(u_n)$ हर \omterm{def:g10:numbers:sets}{पूर्णांक} $n \geq 0$ (या $n \geq 1$) के साथ एक
\omterm{def:g10:numbers:sets}{वास्तविक संख्या} $u_n$ जोड़ देता है, जो उसका \emph{सूचक $n$ वाला पद} है।
अनुक्रम इस तरह दिया जा सकता है:
\begin{itemize}
  \item \emph{स्पष्ट रूप से}, $n$ के पदों में $u_n$ का सूत्र देकर: जैसे
        $u_n = n^2 + 1$;
  \item \emph{पुनरावर्ती रूप से}, उसका पहला पद और हर पद से अगले पद तक जाने
        का नियम देकर: जैसे $u_0 = 3$ और $u_{n+1} = 2u_n - 1$।
\end{itemize}
\end{definition}

\begin{example}
$u_n = n^2 + 1$ के लिए: $u_0 = 1$, $u_1 = 2$, $u_2 = 5$, और सीधे $u_{10} = 101$। $u_0 = 3$, $u_{n+1} = 2u_n - 1$ के लिए: $u_1 = 5$, $u_2 = 9$,
$u_3 = 17$ --- हर पद के लिए पिछला पद चाहिए; $u_{10}$ तक पहुँचने में दस क़दम लगते हैं
(या कोई सामान्य सूत्र, \cref{exo:g11:seq:11} देखिए)।
\end{example}

\section{समांतर अनुक्रम}

\begin{definition}[समांतर अनुक्रम]\label{def:g11:seq:arithmetic}
कोई \omterm{def:g11:seq:sequence}{अनुक्रम} \emph{सार्व अंतर}\index{सार्व अंतर} $d$ वाला
\emph{समांतर}\index{समांतर अनुक्रम} \omterm{def:g11:seq:sequence}{अनुक्रम} है यदि हर पद पिछले पद में $d$
जोड़कर मिलता हो:
\[
u_{n+1} = u_n + d \quad n \text{ के लिए}.
\]
समतुल्य रूप से: अंतर $u_{n+1} - u_n$ अचर है और $d$ के बराबर।
\end{definition}

\begin{theorem}[व्यापक पद]\label{thm:g11:seq:arithgeneral}
यदि $(u_n)$ पहले पद $u_0$ और \omterm{def:g11:seq:arithmetic}{सार्व अंतर} $d$ वाला \omterm{def:g11:seq:arithmetic}{समांतर अनुक्रम} हो, तो
\[
u_n = u_0 + n\,d \quad n \geq 0 \text{ के लिए},
\qquad\text{और अधिक सामान्यतः } u_n = u_p + (n - p)\,d .
\]
\end{theorem}

\begin{proof}
$u_0$ से $u_n$ तक जाने में नियम ``$d$ जोड़िए'' $n$ बार लगाया जाता है: एक
क़दम से $u_1 = u_0 + d$ मिलता है, दो क़दम से $u_2 = u_0 + 2d$, और $n$ क़दमों के बाद हर प्रयोग ने
एक-एक $d$ दिया है, इसलिए $u_n = u_0 + nd$। (इस ``और इसी तरह आगे'' को \cref{ch:g12:seq} में आगमन
द्वारा कठोर बनाया गया है।) व्यापक सूत्र $u_p$ से $u_n$ तक के $n - p$ क़दम गिनने
से निकल आता है।
\end{proof}

\begin{theorem}[क्रमागत पूर्णांकों का योग]\label{thm:g11:seq:intsum}
हर \omterm{def:g10:numbers:sets}{पूर्णांक} $n \geq 1$ के लिए:
\[
1 + 2 + \dots + n = \frac{n(n+1)}{2}.
\]
और अधिक सामान्य रूप से, किसी \omterm{def:g11:seq:arithmetic}{समांतर अनुक्रम} के क्रमागत पदों का योग है
\[
(\text{पदों की संख्या}) \times
\frac{\text{प्रथम पद} + \text{अंतिम पद}}{2}.
\]
\end{theorem}

\begin{proof}
योग $S$ को दो बार लिखिए, दूसरी बार उल्टे क्रम में, और स्तंभ-दर-स्तंभ
जोड़िए:
\[
\begin{array}{ccccccccc}
S & = & 1 & + & 2 & + & \dots & + & n\\
S & = & n & + & (n-1) & + & \dots & + & 1\\
\hline
2S & = & (n+1) & + & (n+1) & + & \dots & + & (n+1)
\end{array}
\]
$n$ स्तंभ हैं और हर स्तंभ का योग $n + 1$ है, इसलिए $2S = n(n+1)$। किसी भी \omterm{def:g11:seq:arithmetic}{समांतर
अनुक्रम} के लिए यही जोड़ी बनाना काम करता है: पहला $+$ अंतिम $=$, दूसरा
$+$ अंत से दूसरा $= \dots$, क्योंकि बाएँ सिरे पर एक क़दम आगे बढ़ने ($+d$) की
भरपाई दाएँ सिरे पर एक क़दम पीछे हटने ($-d$) से हो जाती है।
\end{proof}

\begin{example}
$1 + 2 + \dots + 100 = \frac{100 \times 101}{2} = 5050$। विषम संख्याओं $1 + 3 + \dots + 99$ का योग ($50$ पद) $50 \times \frac{1 + 99}{2} = 2500$ है।
\end{example}

\section{गुणोत्तर अनुक्रम}

\begin{definition}[गुणोत्तर अनुक्रम]\label{def:g11:seq:geometric}
कोई \omterm{def:g11:seq:sequence}{अनुक्रम} \emph{सार्व अनुपात}\index{सार्व अनुपात} $q \neq 0$ वाला
\emph{गुणोत्तर}\index{गुणोत्तर अनुक्रम} \omterm{def:g11:seq:sequence}{अनुक्रम} है यदि हर पद पिछले पद को
$q$ से गुणा करके मिलता हो:
\[
u_{n+1} = q\,u_n \quad n \text{ के लिए}.
\]
समतुल्य रूप से, जब कोई पद लुप्त न हो: अनुपात $\frac{u_{n+1}}{u_n}$ अचर है और $q$ के बराबर।
\end{definition}

\begin{theorem}[व्यापक पद]\label{thm:g11:seq:geomgeneral}
यदि $(u_n)$ पहले पद $u_0$ और अनुपात $q$ वाला \omterm{def:g11:seq:geometric}{गुणोत्तर अनुक्रम} हो, तो
\[
u_n = u_0\, q^n \quad n \geq 0 \text{ के लिए},
\qquad\text{और अधिक सामान्यतः } u_n = u_p\, q^{\,n-p} .
\]
\end{theorem}

\begin{proof}
वही क़दम-गिनती जो \cref{thm:g11:seq:arithgeneral} में थी: $u_0$ से $u_n$ तक नियम ``$q$ से गुणा कीजिए''
$n$ बार लगता है, जिससे गुणनखंड $q^n$ जुड़ जाता है।
\end{proof}

\begin{theorem}[गुणोत्तर योग]\label{thm:g11:seq:geomsum}
हर वास्तविक $q \neq 1$ और \omterm{def:g10:numbers:sets}{पूर्णांक} $n \geq 0$ के लिए:
\[
1 + q + q^2 + \dots + q^n = \frac{1 - q^{\,n+1}}{1 - q}.
\]
\end{theorem}

\begin{proof}
मान लीजिए $S = 1 + q + \dots + q^n$ है। $q$ से गुणा कीजिए: $qS = q + q^2 + \dots + q^{n+1}$। घटाइए:
\[
S - qS = \bigl(1 + q + \dots + q^n\bigr)
- \bigl(q + q^2 + \dots + q^{n+1}\bigr) = 1 - q^{\,n+1},
\]
क्योंकि हर बीच का पद दोनों योगों में एक-एक बार आता है और कट जाता है। अतः
$(1 - q)S = 1 - q^{\,n+1}$, और $1 - q \neq 0$ से भाग देने पर सूत्र मिल जाता है।
\end{proof}

\begin{example}\label{ex:g11:seq:chessboard}
$1 + 2 + 4 + \dots + 2^{10} = \frac{1 - 2^{11}}{1 - 2} = 2^{11} - 1 =
2047$: शतरंज की बिसात के खानों पर चावल के दाने दुगुने करते जाने पर $64$-वें
खाने से बहुत पहले ही कोई भी अनाज-भंडार पिट जाता है, जहाँ कुल $2^{64} - 1 \approx 1.8 \times 10^{19}$ हो जाता
है।
\end{example}

\begin{method}[अनुक्रम का प्रकार पहचानना]\label{met:g11:seq:recognize}
$u_{n+1} - u_n$ निकालकर सरल कीजिए। यदि परिणाम कोई अचर $d$ हो, तो \omterm{def:g11:seq:sequence}{अनुक्रम} \omterm{def:g11:seq:arithmetic}{समांतर} है।
अन्यथा $\frac{u_{n+1}}{u_n}$ निकालकर सरल कीजिए (पद शून्येतर हों): कोई अचर $q$ मिलने का
अर्थ है \omterm{def:g11:seq:geometric}{गुणोत्तर}। यदि दोनों में से कोई अचर न हो, तो \omterm{def:g11:seq:sequence}{अनुक्रम} इनमें से किसी
प्रकार का नहीं है --- केवल पहले कुछ पदों से निष्कर्ष कभी मत निकालिए।
\end{method}

\begin{example}
$u_n = 3 \times 5^n$ के लिए: सभी $n$ के लिए $\frac{u_{n+1}}{u_n} = \frac{3 \times 5^{n+1}}{3 \times 5^n} = 5$: अनुपात $5$ वाला \omterm{def:g11:seq:geometric}{गुणोत्तर}। $u_n = n^2$ के
लिए: $u_1 - u_0 = 1$, पर $u_2 - u_1 = 3$, और $\frac{u_1}{u_0}$ तो परिभाषित ही नहीं है --- न \omterm{def:g11:seq:arithmetic}{समांतर}, न
\omterm{def:g11:seq:geometric}{गुणोत्तर}।
\end{example}

\section{एकदिष्टता}

\begin{definition}[एकदिष्ट अनुक्रम]\label{def:g11:seq:monotone}
\omterm{def:g11:seq:sequence}{अनुक्रम} $(u_n)$ \emph{\omterm{def:g11:func:monotone}{वर्धमान}} है यदि सभी $n$ के लिए $u_{n+1} \geq u_n$ हो, और
\emph{\omterm{def:g10:functions:variations}{ह्रासमान}} है यदि सभी $n$ के लिए $u_{n+1} \leq u_n$ हो।
\end{definition}

\begin{method}[एकदिष्टता का अध्ययन]\label{met:g11:seq:monotonicity}
$u_{n+1} - u_n$ के चिह्न का अध्ययन कीजिए। धनात्मक पदों वाले \omterm{def:g11:seq:sequence}{अनुक्रमों} के लिए इसके बदले
$\frac{u_{n+1}}{u_n}$ की $1$ से तुलना की जा सकती है।
\end{method}

\begin{example}
\omterm{def:g11:seq:arithmetic}{समांतर अनुक्रम} $d \geq 0$ होने पर \omterm{def:g11:func:monotone}{वर्धमान} ($u_{n+1} - u_n = d$) और $d \leq 0$ होने पर \omterm{def:g10:functions:variations}{ह्रासमान} होता
है। $u_0 > 0$ और $q > 1$ वाला \omterm{def:g11:seq:geometric}{गुणोत्तर अनुक्रम} \omterm{def:g11:func:monotone}{वर्धमान} है: $u_{n+1} - u_n = u_0 q^n (q - 1) > 0$; और $u_0 > 0$ तथा
$0 < q < 1$ वाला \omterm{def:g10:functions:variations}{ह्रासमान}।
\end{example}

\section{दीर्घ काल का व्यवहार, अनौपचारिक रूप से}

जब $n$ बहुत बड़ा हो जाता है तब $u_n$ का क्या होता है? $d > 0$ वाले \omterm{def:g11:seq:arithmetic}{समांतर
अनुक्रम} के पद $u_0 + nd$ अंततः किसी भी नियत संख्या से आगे निकल जाते हैं। $0 < q < 1$
वाले \omterm{def:g11:seq:geometric}{गुणोत्तर अनुक्रम} के पद $u_0 q^n$ सिकुड़कर $0$ की ओर जाते हैं: मान लीजिए
$0.9$ से बार-बार गुणा करना किसी भी आरंभिक मान को घिस डालता है। और $q > 1$ के
लिए पद फट पड़ते हैं, जैसा \cref{ex:g11:seq:chessboard} में हुआ।

\begin{remark}
इन कथनों को पूरी तरह सुनिश्चित किया जा सकता है --- ``पद अंततः $0$ की किसी
भी दी हुई दूरी के भीतर टिके रहते हैं'' --- और सिद्ध भी। यही \emph{सीमाओं}
का सिद्धांत है, जो \cref{ch:g12:seq} का आरंभिक विषय है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g11:seq:1}
हर \omterm{def:g11:seq:sequence}{अनुक्रम} के लिए $u_1$, $u_2$, $u_3$ निकालिए:
\[
u_n = \frac{n}{n+1}; \qquad
u_0 = 5,\ u_{n+1} = 3u_n - 2; \qquad
u_n = (-1)^n\,n .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:2}
$(u_n)$ $u_0 = 7$ और $d = -3$ वाला \omterm{def:g11:seq:arithmetic}{समांतर अनुक्रम} है। $u_{10}$ और $u_{25}$ निकालिए। $(v_n)$
$v_3 = 11$ और $v_8 = 26$ वाला \omterm{def:g11:seq:arithmetic}{समांतर अनुक्रम} है। \omterm{def:g11:seq:arithmetic}{सार्व अंतर} और $v_0$ ज्ञात कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:3}
$(u_n)$ $u_0 = 5$ और $q = 2$ वाला \omterm{def:g11:seq:geometric}{गुणोत्तर अनुक्रम} है। $u_8$ निकालिए। $(v_n)$ धनात्मक
पदों वाला \omterm{def:g11:seq:geometric}{गुणोत्तर अनुक्रम} है जिसमें $v_2 = 12$ और $v_4 = 48$ हैं। अनुपात और $v_0$
ज्ञात कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:4}
निकालिए:
\[
1 + 2 + 3 + \dots + 500, \qquad
4 + 7 + 10 + \dots + 61, \qquad
1 + \frac12 + \frac14 + \dots + \frac{1}{2^{10}} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:seq:5}
बताइए कि हर \omterm{def:g11:seq:sequence}{अनुक्रम} \omterm{def:g11:seq:arithmetic}{समांतर} है, \omterm{def:g11:seq:geometric}{गुणोत्तर} है, या इनमें से कोई नहीं:
\[
u_n = 4n - 1; \qquad
v_n = \frac{2^n}{3^{n+1}}; \qquad
w_n = n^2 + n .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:6}
एक रंगमंच में $20$ पंक्तियाँ हैं: पहली पंक्ति में $16$ सीटें हैं, और हर
पंक्ति में अपनी पिछली पंक्ति से $2$ सीटें अधिक हैं। अंतिम पंक्ति में कितनी
सीटें हैं? पूरे रंगमंच में कितनी?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:7}
जीवाणुओं की एक समष्टि हर घंटे दुगुनी हो जाती है; दोपहर को $500$ जीवाणु हैं।
रात 8 बजे कितने होंगे? कितने पूरे घंटों के बाद समष्टि पहली बार दस लाख से आगे
निकल जाती है? ($2$ की क्रमिक घातें आज़माकर हल कीजिए।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:8}
एक बचतकर्ता हर महीने $100$ यूरो ऐसे खाते में जमा करता है जो मौजूदा शेष पर
प्रति माह $0.2\%$ ब्याज देता है (ब्याज जमा से ठीक पहले चढ़ाया जाता है)। मान
लीजिए $c_n$ $n$-वीं जमा के ठीक बाद का शेष है, इसलिए $c_1 = 100$ और $c_{n+1} = 1.002\,c_n + 100$। $c_2$
और $c_3$ निकालिए, और समझाइए कि $(c_n)$ न \omterm{def:g11:seq:arithmetic}{समांतर} है न \omterm{def:g11:seq:geometric}{गुणोत्तर}।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:9}
\omterm{def:g11:seq:sequence}{अनुक्रमों}
\[
u_n = n^2 - 8n \ (n \geq 0), \qquad
v_n = \frac{3^n}{n!}\ (n \geq 1),
\]
की एकदिष्टता का अध्ययन कीजिए, जहाँ $n! = 1 \times 2 \times \dots \times n$ है। ($(v_n)$ के लिए $\frac{v_{n+1}}{v_n}$ की $1$
से तुलना कीजिए।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:seq:10}
$u_0 = 3$ और $d = 4$ वाले किसी \omterm{def:g11:seq:arithmetic}{समांतर अनुक्रम} के पहले $n$ पदों का योग $903$ है।
$n$ ज्ञात कीजिए। ($n$ में द्विघात \omterm{def:g10:algebra:equation}{समीकरण} बनाइए और \cref{ch:g11:quad} का उपयोग कीजिए।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g11:seq:11}
मान लीजिए $u_0 = 3$ और $u_{n+1} = 2u_n - 1$ हैं।
\begin{enumerate}
  \item $u_1, u_2, u_3$ निकालिए और $u_n$ के लिए कोई सूत्र अनुमानित कीजिए।
  \item मान लीजिए $v_n = u_n - 1$ है। दिखाइए कि $(v_n)$ \omterm{def:g11:seq:geometric}{गुणोत्तर} है, और उसका अनुपात
        तथा पहला पद बताइए।
  \item $u_n$ का स्पष्ट सूत्र निकालिए और अपने अनुमान की पुष्टि कीजिए।
\end{enumerate}
\end{exercise}

\section{समस्या: ब्रह्मा की मीनार और फिबोनाच्ची के ख़रगोश}

\begin{problem}[{सप्ताहांत समस्या --- दो पौराणिक पुनरावृत्तियाँ: वह मीनार जो
संसार का अंत कर देगी, वह अनुक्रम जो सोने की तरह बढ़ता है, और वह सहायक तरकीब
जो ऋणों को साध लेती है}]
\label{pb:g11:seq:1}
गणित की लोक-कथाओं पर दो \omterm{def:g11:seq:sequence}{अनुक्रमों} का राज है। एक ब्रह्मा की मीनार की चालें
गिनता है --- सोने की चौंसठ तश्तरियाँ, जिनके स्थानांतरण से, कथा कहती है,
संसार का अंत हो जाएगा। दूसरा फिबोनाच्ची के ख़रगोश गिनता है और स्वर्ण अनुपात
छिपाए रखता है। न कोई \omterm{def:g11:seq:arithmetic}{समांतर} है, न \omterm{def:g11:seq:geometric}{गुणोत्तर} --- और दोनों इसी अध्याय के
हथियारों के आगे हथियार डाल देते हैं: पुनरावृत्तियाँ, \omterm{def:g11:seq:geometric}{गुणोत्तर} योग (\cref{thm:g11:seq:geomsum}),
और \cref{exo:g11:seq:11} की सहायक-अनुक्रम वाली तरकीब, जो आपके गृह-ऋण का हिसाब भी लगा देती
है।

\medskip
\noindent\textbf{भाग I --- ब्रह्मा की मीनार।}
पहेली: घटते आकार की $n$ तश्तरियाँ खूँटी A पर लगी हैं; पूरे ढेर को खूँटी C
पर ले जाइए, एक बार में एक तश्तरी, और किसी छोटी तश्तरी पर बड़ी कभी न रखिए
(खूँटी B सहायता कर सकती है)। मान लीजिए $h_n$ \omterm{def:g10:functions:extrema}{न्यूनतम} चालों की संख्या है।
\begin{enumerate}
  \item (सिक्कों से) खेलिए और $h_1$, $h_2$, $h_3$ दर्ज कीजिए।
  \item पुनरावृत्ति $h_{n+1} = 2h_n + 1$ के पीछे की रणनीति समझाइए: सबसे बड़ी तश्तरी के
        हिलने से पहले और बाद में क्या होना ही चाहिए?
  \item \cref{exo:g11:seq:11} की तरकीब से पुनरावृत्ति हल कीजिए: $v_n = h_n + 1$ रखिए, दिखाइए कि $(v_n)$
        \omterm{def:g11:seq:geometric}{गुणोत्तर} है, और $h_n = 2^n - 1$ निष्कर्ष निकालिए।
  \item कथा की मीनार में $64$ तश्तरियाँ हैं, और भिक्षु प्रति सेकंड एक
        तश्तरी हटाते हैं। $2^{10} = 1024 \approx 10^3$ का उपयोग करके स्थानांतरण का समय वर्षों में
        आँकिए (एक वर्ष लगभग $3 \times 10^7$ सेकंड का होता है; \cref{ex:g11:seq:chessboard} से तुलना कीजिए,
        जो एक और कहानी में वही दैत्य है)। क्या हमें चिंता करनी चाहिए?
  \item कोई भी रणनीति $2^n - 1$ चालों से कम में क्यों नहीं कर सकती? तर्क दीजिए
        कि \emph{कोई भी} हल $h_{n+1} \geq 2 h_n + 1$ मानता है: सबसे नीचे वाली तश्तरी के हिलने
        से ठीक पहले और ठीक बाद ऊपर की $n$ तश्तरियों के बारे में क्या सच
        होना ही चाहिए?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- फिबोनाच्ची।}
$F_1 = F_2 = 1$ और $F_{n+2} = F_{n+1} + F_n$ परिभाषित कीजिए (हर पद अपने से पहले के दो पदों का योग ---
माध्यमिक विद्यालय खंड का वही लय-गणना वाला नियम, अब अपने यूरोपीय नाम के साथ)।
\begin{enumerate}[resume]
  \item $F_1$ से $F_{12}$ तक गिनाइए।
  \item दिखाइए कि $(F_n)$ न \omterm{def:g11:seq:arithmetic}{समांतर} है न \omterm{def:g11:seq:geometric}{गुणोत्तर}, पर वह $n = 2$ से आगे निरंतर
        \omterm{def:g11:func:monotone}{वर्धमान} है (\cref{met:g11:seq:monotonicity} और पुनरावृत्ति से)।
  \item योग-सर्वसमिका
        \[
        F_1 + F_2 + \dots + F_n = F_{n+2} - 1
        \]
        क्रमिक निरसन से सिद्ध कीजिए: हर $F_k$ को $F_{k+2} - F_{k+1}$ लिखिए और देखिए कि योग
        कैसे ढह जाता है। $n = 6$ के लिए इसकी पुष्टि कीजिए।
  \item वर्गों की सर्वसमिका $F_1^2 + F_2^2 + \dots + F_n^2 = F_n F_{n+1}$ सिद्ध कीजिए, और इसके लिए $F_k F_{k+1} - F_{k-1} F_k = F_k^2$ के साथ
        क्रमिक निरसन कीजिए। $n = 4$ के लिए पुष्टि कीजिए। (चित्र: भुजाओं $1, 1, 2, 3, 5, \dots$
        वाले वर्ग एक आयत को भर देते हैं --- यही प्रसिद्ध फिबोनाच्ची सर्पिल
        का ढाँचा है।)
  \item कासीनी सर्वसमिका कहती है $F_{n+1} F_{n-1} - F_n^2 = (-1)^n$। $n = 4, 5, 6$ के लिए उसकी पुष्टि कीजिए ---
        और माध्यमिक विद्यालय खंड की क्षेत्रफल वाली समस्या में खेली गई
        ``ग़ायब वर्ग'' की तरकीब का इंजन पहचानिए।
  \item पुनरावृत्ति से दिखाइए कि $F_{n+2} \geq 2 F_n$: फिबोनाच्ची हर दो क़दम में कम से कम
        दुगुना हो जाता है --- वह अनुपात $\sqrt2$ वाले \omterm{def:g11:seq:geometric}{गुणोत्तर अनुक्रम} से कम
        तेज़ नहीं बढ़ता।
  \item $n = 3$ से $10$ तक अनुपात $r_n = \frac{F_{n+1}}{F_n}$ निकालिए (तीन दशमलव स्थान)। यह मानते
        हुए कि वे किसी सीमा $L$ पर आ बैठते हैं, संबंध $r_{n+1} = 1 + \frac{1}{r_n}$ को सीमा तक ले
        जाइए और हल कीजिए: ख़रगोश \cref{pb:g10:algebra:1} की किस संख्या की पूजा करते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- सहायक तरकीब, बैंक में।}
\begin{enumerate}[resume]
  \item \cref{exo:g11:seq:11} को सामान्य कीजिए: $a \neq 1$ वाले $u_{n+1} = a\,u_n + b$ के लिए $\ell = \frac{b}{1 - a}$ रखिए (\omterm{pb:g10:functions:1}{स्थिर
        बिंदु})। दिखाइए कि $v_n = u_n - \ell$ अनुपात $a$ वाला \omterm{def:g11:seq:geometric}{गुणोत्तर} है, और निष्कर्ष
        निकालिए
        $u_n = a^n (u_0 - \ell) + \ell$।
  \item एक ऋण: $10\,000$ यूरो, प्रति माह $1\,\%$ ब्याज पर, और प्रति माह $300$
        यूरो की अदायगी, इसलिए ऋण $d_{n+1} = 1.01\,d_n - 300$ मानता है। प्रश्न 13 लगाइए (पहले
        \omterm{pb:g10:functions:1}{स्थिर बिंदु}!) और $d_n$ का स्पष्ट सूत्र निकालिए।
  \item कैलकुलेटर से वह पहला महीना ज्ञात कीजिए जिसमें ऋण चुक जाता है, और
        कुल चुकाई गई राशि भी। उधार लेने का अपना ख़र्च कितना पड़ा?
  \item $50\,000$ निवासियों का एक नगर प्रति वर्ष $2\,\%$ बढ़ता है और उसके ऊपर
        $1\,000$ नए लोग आ बसते हैं: $p_{n+1} = 1.02\,p_n + 1000$। स्पष्ट सूत्र और $10$ वर्षों बाद की
        जनसंख्या बताइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- दो राजपरिवार।}
\begin{enumerate}[resume]
  \item $1 + 2 + 3 + \dots + 1000$ निकालिए (\cref{thm:g11:seq:intsum} --- माध्यमिक विद्यालय खंड का नन्हे गाउस वाला
        योग, अब औपचारिक रूप में), और $1 + 2 + 4 + \dots + 2^{19}$ (\cref{thm:g11:seq:geomsum})।
  \item \omterm{def:g11:seq:arithmetic}{समांतर अनुक्रम} $7, 12, 17, \dots, 502$ का योग निकालिए (कितने पद?)।
  \item बचत योजना: हर महीने $100$ यूरो जमा, प्रति माह $0.5\,\%$ की कमाई; $n$-वीं
        जमा के बाद शेष $100\left(1.005^{n-1} + \dots + 1.005 + 1\right)$ है। $5$ वर्षों बाद का शेष निकालिए ($n = 60$)।
  \item समापन --- \omterm{def:g11:seq:sequence}{अनुक्रम} साधने का बक्सा: स्पष्ट बनाम पुनरावर्ती वर्णन;
        दोनों राजपरिवार और उनके योग-सूत्र; वह सहायक \omterm{def:g11:seq:sequence}{अनुक्रम} जो एकघात
        पुनरावृत्तियों को \omterm{def:g11:seq:geometric}{गुणोत्तर} में बदल देता है; और फिबोनाच्ची, दोनों
        परिवारों से बाहर का पहला नागरिक, जो आज सर्वसमिकाओं से सधा है और
        पूरी तरह पकड़े जाने के लिए आव्यूहों (कक्षा 12) तथा सीमाओं की
        प्रतीक्षा में है। हर एक पर एक वाक्य।
\end{enumerate}
\end{problem}
