\chapter{वर्णनात्मक सांख्यिकी}\label{ch:g11:stat}

सांख्यिकी संख्याओं की किसी सूची को चंद सार्थक सारांशों में दबा देती है:
आँकड़े कहाँ बैठे हैं (\omterm{def:g11:stat:mean}{माध्य}, \omterm{def:g11:stat:median}{माध्यिका}), और वे कितने फैले हुए हैं (\omterm{def:g11:stat:quartiles}{चतुर्थक},
\omterm{def:g11:stat:variance}{प्रसरण}, \omterm{def:g11:stat:variance}{मानक विचलन})। यह अध्याय सब कुछ एक ही चलते उदाहरण पर करके दिखाता है।
यही सारांश \cref{ch:g11:prob} में यादृच्छिक चरों के लिए फिर आते हैं, और उनके प्रायिकता-नियम
\cref{ch:g12:sums} में।

पूरे अध्याय में हमारा चलता आँकड़ा-समुच्चय $12$ विद्यार्थियों की एक कक्षा के
अंक हैं ($20$ में से):
\[
8,\ 9,\ 10,\ 10,\ 11,\ 12,\ 12,\ 12,\ 13,\ 14,\ 16,\ 17 .
\]

\section{माध्य और माध्यिका}

\begin{definition}[माध्य]\label{def:g11:stat:mean}
मानों $x_1, x_2, \dots, x_n$ का \emph{माध्य}\index{माध्य} है
\[
\bar x = \frac{x_1 + x_2 + \dots + x_n}{n}.
\]
यदि मान $x_i$ बारंबारता $n_i$ के साथ आता हो (जहाँ $n_1 + \dots + n_k = n$), तो $\bar x = \frac{n_1 x_1 + \dots + n_k x_k}{n}$।
\end{definition}

\begin{definition}[माध्यिका]\label{def:g11:stat:median}
किसी क्रमबद्ध आँकड़ा-समुच्चय की \emph{माध्यिका}\index{माध्यिका} वह मान है जो
उसे दो बराबर हिस्सों में बाँट दे: कम से कम आधे मान उससे $\leq$ हों और कम से कम
आधे $\geq$। व्यवहार में $n$ मानों को क्रम में लगाइए; यदि $n$ विषम हो तो
माध्यिका बीच वाला मान है, और यदि $n$ सम हो तो बीच वाले दोनों मानों का
\omterm{prop:g10:coordgeom:midpoint}{मध्यबिंदु} लीजिए।
\end{definition}

\begin{example}\label{ex:g11:stat:meanmedian}
चलते आँकड़ा-समुच्चय के लिए योग $144$ है, इसलिए $\bar x = \frac{144}{12} =
12$। बीच वाले दोनों मान
(क्रमबद्ध सूची के $6$-वाँ और $7$-वाँ) दोनों $12$ हैं, इसलिए \omterm{def:g11:stat:median}{माध्यिका} भी
$12$ है। \omterm{def:g11:stat:mean}{माध्य} और \omterm{def:g11:stat:median}{माध्यिका} का बराबर होना आवश्यक नहीं: \omterm{def:g11:stat:median}{माध्यिका} इस बात की
परवाह नहीं करती कि चरम मान \emph{कितनी दूर} हैं, \omterm{def:g11:stat:mean}{माध्य} करता है। सबसे ऊपर के
अंक $17$ के स्थान पर $20$ रख दीजिए तो \omterm{def:g11:stat:mean}{माध्य} $12.25$ हो जाता है, पर \omterm{def:g11:stat:median}{माध्यिका}
$12$ ही रहती है।
\end{example}

\section{चतुर्थक}

\begin{definition}[चतुर्थक, अंतर-चतुर्थक परास]\label{def:g11:stat:quartiles}
\emph{प्रथम चतुर्थक}\index{चतुर्थक} $Q_1$ वह सबसे छोटा मान है जिसके लिए कम से
कम एक चौथाई आँकड़े $\leq Q_1$ हों; \emph{तृतीय चतुर्थक} $Q_3$ वह सबसे छोटा मान है
जिसके लिए कम से कम तीन चौथाई आँकड़े $\leq Q_3$ हों। \emph{अंतर-चतुर्थक
परास}\index{अंतर-चतुर्थक परास} $Q_3 - Q_1$ है: आँकड़ों के मध्य आधे भाग की चौड़ाई।
\end{definition}

\begin{example}
$n = 12$: एक चौथाई $3$ है, इसलिए $Q_1$ $3$-वाँ क्रमबद्ध मान है: $Q_1 = 10$। तीन
चौथाई $9$ है, इसलिए $Q_3$ $9$-वाँ मान है: $Q_3 = 13$। अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} $3$ है:
कक्षा का मध्य आधा भाग $3$ अंकों के भीतर बैठा है।
\end{example}

\section{प्रसरण और मानक विचलन}

\begin{definition}[प्रसरण, मानक विचलन]\label{def:g11:stat:variance}
आँकड़ा-समुच्चय का \emph{प्रसरण}\index{प्रसरण} \omterm{def:g11:stat:mean}{माध्य} से विचलनों के वर्गों का
\omterm{def:g11:stat:mean}{माध्य} है:
\[
V = \frac{1}{n}\sum_{i=1}^{n} (x_i - \bar x)^2,
\]
और \emph{मानक विचलन}\index{मानक विचलन} $\sigma = \sqrt{V}$ है, जो आँकड़ों के उसी मात्रक में
व्यक्त होता है।
\end{definition}

\begin{proposition}[संक्षिप्त सूत्र]\label{prop:g11:stat:shortcut}
\omterm{def:g11:stat:variance}{प्रसरण} वर्गों के \omterm{def:g11:stat:mean}{माध्य} में से \omterm{def:g11:stat:mean}{माध्य} का वर्ग घटाने पर मिलता है:
\[
V = \frac{1}{n}\sum_{i=1}^{n} x_i^2 \;-\; \bar x^{\,2}.
\]
\end{proposition}

\begin{proof}
हर वर्ग-विचलन का प्रसार कीजिए: $(x_i - \bar x)^2 = x_i^2 - 2\bar x\, x_i + \bar x^2$। जोड़कर $n$ से भाग देने पर:
\[
V = \frac1n \sum x_i^2 - 2\bar x \cdot \underbrace{\frac1n \sum
x_i}_{=\ \bar x} + \frac1n \cdot n\bar x^2
= \frac1n\sum x_i^2 - 2\bar x^2 + \bar x^2
= \frac1n\sum x_i^2 - \bar x^2 .
\]
यही सर्वसमिका यादृच्छिक चरों के लिए \cref{prop:g11:prob:konig} है।
\end{proof}

\begin{example}\label{ex:g11:stat:variance}
चलते आँकड़ा-समुच्चय के लिए $\bar x = 12$ से विचलन $-4$, $-3$, $-2$, $-2$, $-1$, $0$,
$0$, $0$, $1$, $2$, $4$, $5$ हैं; उनके वर्गों का योग है
\[
16 + 9 + 4 + 4 + 1 + 0 + 0 + 0 + 1 + 4 + 16 + 25 = 80 .
\]
अतः
\[
V = \frac{80}{12} = \frac{20}{3} \approx 6.67,
\qquad
\sigma = \sqrt{20/3} \approx 2.6 \text{ अंक}.
\]
मोटे तौर पर कहें तो कोई प्रारूपिक अंक \omterm{def:g11:stat:mean}{माध्य} से लगभग $2.6$ अंक दूर बैठा है।
\end{example}

\begin{proposition}[मात्रक बदलना]\label{prop:g11:stat:affine}
यदि हर मान को $y_i = a x_i + b$ से रूपांतरित किया जाए, तो
\[
\bar y = a \bar x + b,
\qquad
V_y = a^2\, V_x,
\qquad
\sigma_y = \abs{a}\,\sigma_x .
\]
\end{proposition}

\begin{proof}
$\bar y = \frac1n \sum (a x_i + b) = a\left(\frac1n\sum x_i\right) +
\frac1n \cdot nb = a\bar x + b$। तब $y_i - \bar y = a(x_i - \bar x)$: हर विचलन $a$ गुना हो जाता है (खिसकाव $b$ कट जाता है),
इसलिए हर वर्ग-विचलन $a^2$ गुना, और उनका \omterm{def:g11:stat:mean}{माध्य} भी। वर्गमूल लेने पर $\sigma_y = \abs a\,\sigma_x$ मिलता
है।
\end{proof}

\begin{example}
सेल्सियस में दर्ज किए गए तापमान, जिनका \omterm{def:g11:stat:mean}{माध्य} $20$ और \omterm{def:g11:stat:variance}{मानक विचलन} $3$ है,
$F = 1.8\,C + 32$ से फ़ारेनहाइट में बदल जाते हैं: \omterm{def:g11:stat:mean}{माध्य} $1.8 \times 20 + 32 = 68$ हो जाता है और \omterm{def:g11:stat:variance}{मानक विचलन}
$1.8 \times 3 = 5.4$। आँकड़ों को खिसकाने से उनका प्रसार नहीं बदलता; उन्हें गुणा करने से
प्रसार भी उतना ही गुना हो जाता है।
\end{example}

\begin{method}[आँकड़ों का सारांश और तुलना]\label{met:g11:stat:compare}
दो आँकड़ा-समुच्चयों की तुलना के लिए साथ-साथ रखिए: \emph{स्थिति} के लिए \omterm{def:g11:stat:mean}{माध्य}
(या \omterm{def:g11:stat:median}{माध्यिका}), और \emph{प्रसार} के लिए \omterm{def:g11:stat:variance}{मानक विचलन} (या अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास})।
युग्म (\omterm{def:g11:stat:median}{माध्यिका}, अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास}) चरम मानों के आगे टिका रहता है; युग्म
(\omterm{def:g11:stat:mean}{माध्य}, \omterm{def:g11:stat:variance}{मानक विचलन}) हर मान का उपयोग करता है और प्रायिकता-सिद्धांत को खुराक
देता है।
\end{method}

\begin{example}
हर धनुर्धर $10$ तीर चलाता है; दोनों के अंकों का औसत $8.5$ है, पर \omterm{def:g11:stat:variance}{मानक
विचलन} $0.5$ और $2.1$ हैं। औसत स्तर एक ही --- पर पहला धनुर्धर कहीं अधिक
एकरूप है।
\end{example}

\]
का \omterm{def:g11:stat:mean}{माध्य}, \omterm{def:g11:stat:median}{माध्यिका}, \omterm{def:g11:stat:quartiles}{चतुर्थक} और अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास} निकालिए।

\begin{exercise}[$\star$]\label{exo:g11:stat:2}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:stat:3}
एक पासा $50$ बार फेंका जाता है; परिणामों का सारांश बारंबारताओं से है:

\omterm{def:g11:stat:mean}{माध्य} परिणाम और \omterm{def:g11:stat:median}{माध्यिका} निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:stat:4}
$15$ विद्यार्थियों की कक्षा का \omterm{def:g11:stat:mean}{माध्य} अंक $11.2$ है। $16$-वाँ विद्यार्थी
आकर $18$ अंक पाता है। कक्षा का नया \omterm{def:g11:stat:mean}{माध्य} क्या है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:stat:5}
आँकड़ा-समुच्चय $6, 8, x, 12, 14$ का \omterm{def:g11:stat:mean}{माध्य} $10$ है। $x$ ज्ञात कीजिए, फिर पूरे हुए
आँकड़ा-समुच्चय का \omterm{def:g11:stat:variance}{मानक विचलन} निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g11:stat:6}
\omterm{def:g11:stat:mean}{माध्य} $12.4$ और \omterm{def:g11:stat:variance}{मानक विचलन} $2.5$ वाले अंकों को $y = \frac{x}{2} + 4$ से बदला जाता है। नया
\omterm{def:g11:stat:mean}{माध्य} और नया \omterm{def:g11:stat:variance}{मानक विचलन} बताइए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:stat:7}
समूह A ($20$ मान) का \omterm{def:g11:stat:mean}{माध्य} $10$ है; समूह B ($30$ मान) का \omterm{def:g11:stat:mean}{माध्य} $15$ है।

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g11:stat:9}

\section{समस्या: z-मान, एक सार्वभौमिक पैमाना}

\begin{problem}[{सप्ताहांत समस्या --- मानक विचलन सेब की तुलना संतरे से करा
देते हैं, चेबिशेव असमिका $\sigma$ को गारंटी में बदल देती है, और माध्य अपना
अपना तमग़ा जीत लेता है}]
\label{pb:g11:stat:1}
क्या किसी कठिन परीक्षा में $15/20$ किसी आसान परीक्षा के $13/20$ से बेहतर है?
छोटी संख्याओं की सूची में $23$ कोई संयोग है या बहिर्मान? कच्ची संख्याएँ यह
नहीं बता सकतीं --- पर सही \omterm{def:g11:stat:variance}{मानक विचलन} से भाग दे दीजिए तो वे सब एक ही भाषा
बोलने लगती हैं, \emph{\omterm{pb:g11:stat:1}{z-मान}} की भाषा। यह समस्या \omterm{def:g11:stat:variance}{प्रसरण} की मशीनरी (\cref{def:g11:stat:variance},
\cref{prop:g11:stat:shortcut}) पर पकड़ बनाती है, वह सार्वभौमिक पैमाना बनाती है, और वह उल्लेखनीय
असमिका सिद्ध करती है जो $\sigma$ को केवल वर्णनात्मक संख्या से गारंटी में बदल
देती है।

\medskip
\noindent\textbf{भाग I --- \omterm{def:g11:stat:variance}{प्रसरण} की मशीनरी।}
\begin{enumerate}
  \item आँकड़ा-समुच्चय $4, 8, 6, 5, 2$ के लिए: \omterm{def:g11:stat:mean}{माध्य} निकालिए, फिर सीधे परिभाषा से
        \omterm{def:g11:stat:variance}{प्रसरण}, फिर $\sigma$।
  \item संक्षिप्त सूत्र (\cref{prop:g11:stat:shortcut}) से \omterm{def:g11:stat:variance}{प्रसरण} फिर निकालिए --- वर्गों का \omterm{def:g11:stat:mean}{माध्य}
        घटा \omterm{def:g11:stat:mean}{माध्य} का वर्ग --- और मेल जाँचिए।
  \item बारंबारताओं के साथ: अंक $8$, $10$, $14$ क्रमशः $3$, $5$,
        $2$ विद्यार्थियों को मिले। \omterm{def:g11:stat:mean}{माध्य}, \omterm{def:g11:stat:variance}{प्रसरण} और $\sigma$ निकालिए (दो
        दशमलव स्थान)।
  \item तापमानों का \omterm{def:g11:stat:mean}{माध्य} $20$ और $\sigma = 3$ है (सेल्सियस)। फ़ारेनहाइट में
        \omterm{def:g11:stat:mean}{माध्य} और $\sigma$ बताइए ($F = 1.8\,C + 32$; \cref{prop:g11:stat:affine})।
  \item अभी काम में लाए गए दोनों नियम सिद्ध कीजिए: हर मान में अचर $b$
        जोड़ने से \omterm{def:g11:stat:mean}{माध्य} $b$ खिसक जाता है और $\sigma$ अपरिवर्तित रहता है;
        $a$ से गुणा करने पर \omterm{def:g11:stat:mean}{माध्य} $a$ गुना और $\sigma$ $\abs a$ गुना हो
        जाता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- सार्वभौमिक पैमाना।}
\omterm{def:g11:stat:mean}{माध्य} $\bar x$ और \omterm{def:g11:stat:variance}{मानक विचलन} $\sigma$ वाले किसी आँकड़ा-समुच्चय में मान $x$ का
\emph{z-मान}\index{z-मान} $z = \dfrac{x - \bar x}{\sigma}$ है: उतने \omterm{def:g11:stat:variance}{मानक विचलन} जितने $x$ को \omterm{def:g11:stat:mean}{माध्य} से
अलग करते हैं।
\begin{enumerate}[resume]
  \item परीक्षा A का \omterm{def:g11:stat:mean}{माध्य} $12$ और $\sigma = 2$ है; परीक्षा B का \omterm{def:g11:stat:mean}{माध्य} $8$ और
        $\sigma = 4$। ऐलिस ने A में $15$ पाए, बॉब ने B में $13$। दोनों के \omterm{pb:g11:stat:1}{z-मान}
        निकालिए: किसका प्रदर्शन भीड़ से अधिक ऊपर खड़ा है?
  \item प्रश्न 1 के आँकड़ा-समुच्चय के पाँचों \omterm{pb:g11:stat:1}{z-मान} निकालिए, और जाँचिए कि
        उनका \omterm{def:g11:stat:mean}{माध्य} $0$ और \omterm{def:g11:stat:variance}{मानक विचलन} $1$ है। प्रश्न 5 से समझाइए कि
        मानकीकरण से \emph{सदा} \omterm{def:g11:stat:mean}{माध्य} $0$ और $\sigma = 1$ क्यों मिलता है।
  \item किसी अभिरुचि परीक्षा का \omterm{def:g11:stat:mean}{माध्य} $100$ और $\sigma = 15$ है। कौन-से कच्चे अंक
        का $z = 2$ है? $76$ का \omterm{pb:g11:stat:1}{z-मान} क्या है?
  \item एक शिशु-चिकित्सक किसी शिशु का भार ``$z = -2.5$'' दर्ज करता है। बताइए कि
        इसका अर्थ क्या है, और बच्चे की आयु बढ़ने पर वृद्धि किलोग्राम के बजाय
        \omterm{pb:g11:stat:1}{z-मानों} में क्यों आँकी जाती है।
  \item \cref{exo:g11:stat:10} का बहिर्मान (आँकड़ा-समुच्चय $1, 2, 2, 3, 3, 3, 4, 4, 5, 23$): \omterm{def:g11:stat:mean}{माध्य}, $\sigma$ और $23$ का
        \omterm{pb:g11:stat:1}{z-मान} निकालिए। चिरपरिचित ख़तरे की सीमा $\abs z \geq 3$ है: क्या निर्णय है?
  \item मान $23$ के \emph{बिना} $\sigma$ फिर निकालिए, और वह \omterm{pb:g11:stat:1}{z-मान} भी जो $23$
        को स्वच्छ आँकड़ों के \omterm{def:g11:stat:mean}{माध्य} तथा $\sigma$ के सामने मिलता। बहिर्मान ने उसी
        पैमाने के साथ क्या कर डाला जो उसे नापने के लिए था, और \cref{exo:g11:stat:10} इसके
        प्रतिकार के रूप में क्या सुझाता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- चेबिशेव की गारंटी।}
\begin{enumerate}[resume]
  \item किसी आँकड़ा-समुच्चय के लिए ब्येनेमे--चेबिशेव असमिका सिद्ध कीजिए: यदि
        मानों का अनुपात $p$ $\abs{x - \bar x} \geq k\sigma$ को संतुष्ट करता हो, तो \omterm{def:g11:stat:variance}{प्रसरण} के योग
        में केवल वही पद रखने पर
        \[
        \sigma^2 \geq p \cdot (k\sigma)^2,
        \qquad\text{अतः}\qquad
        p \leq \frac{1}{k^2} .
        \]
  \item यह असमिका किसी भी आँकड़ा-समुच्चय के उस अनुपात के बारे में क्या
        गारंटी देती है जो \omterm{def:g11:stat:mean}{माध्य} से $2\sigma$ से आगे है? $3\sigma$ से आगे? प्रश्न 1
        के आँकड़ा-समुच्चय पर $2\sigma$ वाला कथन जाँचिए।
  \item एक कारख़ाना ऐसे बोल्ट बनाता है जिनका \omterm{def:g11:stat:mean}{माध्य} $50$ mm और $\sigma = 0.1$ mm है।
        $\intcc{49.7}{50.3}$ से बाहर पड़ने वाले बोल्टों के अनुपात के बारे में चेबिशेव क्या
        गारंटी देता है? (वास्तविक उत्पादन इससे कहीं अच्छा करता है --- कक्षा
        12 का घंटी-वक्र इस सतर्क $\frac19$ को प्रतिशत के अंश में बदल देता है।)
  \item दो निधियाँ औसतन प्रति वर्ष $5\,\%$ देती हैं, जिनमें निधि A के लिए
        $\sigma = 2\,\%$ और निधि B के लिए $\sigma = 12\,\%$ है। हर निधि के लिए चेबिशेव उन वर्षों के
        प्रायिकता-जैसे अनुपात के बारे में क्या कहता है जिनमें प्रतिफल $-19\,\%$
        से नीचे रहा? $\sigma$ का वित्तीय नाम क्या है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- \omterm{def:g11:stat:mean}{माध्य} का तमग़ा, और एक चेतावनी।}
\begin{enumerate}[resume]
  \item मान लीजिए $f(c)$ किसी मनमाने केंद्र $c$ से वर्ग-विचलनों $(x_i - c)^2$ का
        \omterm{def:g11:stat:mean}{माध्य} है। सर्वसमिका
        \[
        f(c) = \sigma^2 + (\bar x - c)^2 ,
        \]
        सिद्ध कीजिए और निष्कर्ष निकालिए: \omterm{def:g11:stat:mean}{माध्य} ही वह अकेला बिंदु है जो
        \omterm{def:g11:stat:mean}{माध्य} वर्ग-विचलन को \omterm{def:g10:functions:extrema}{न्यूनतम} करता है --- यही उसका तमग़ा है, जो \cref{pb:g10:stats:1}
        में \omterm{def:g11:stat:median}{माध्यिका} द्वारा निरपेक्ष विचलनों के न्यूनतमकरण के समकक्ष है।
  \item $c = 6$ के साथ प्रश्न 1 के आँकड़ा-समुच्चय पर उस सर्वसमिका की जाँच
        कीजिए: $f(6)$ सीधे और सूत्र से, दोनों तरह निकालिए।
  \item वही सारांश, अलग आँकड़े: जाँचिए कि $P = \{3, 5, 7, 9, 11\}$ और $Q = \{1, 7, 7, 7, 13\}$ का \omterm{def:g11:stat:mean}{माध्य} एक ही है
        पर \omterm{def:g11:stat:variance}{प्रसरण} नहीं; फिर $\{7 - a, 7, 7, 7, 7 + a\}$ रूप का ऐसा आँकड़ा-समुच्चय बनाइए जिसका
        \omterm{def:g11:stat:mean}{माध्य} और \omterm{def:g11:stat:variance}{प्रसरण} ठीक $P$ वाले हों --- और फिर भी आकृति अपनी ही हो।
  \item सिद्ध कीजिए कि $\sigma = 0$ होने पर सभी मान बराबर हो जाते हैं। (वर्गों के
        शून्य योग का हर पद क्या होना ही चाहिए?)
  \item समापन --- सांख्यिकीविद् की सीढ़ी, हर डंडे पर एक पंक्ति: केंद्र
        (\omterm{def:g11:stat:mean}{माध्य}, \omterm{def:g11:stat:median}{माध्यिका}), प्रसार ($\sigma$, अंतर-चतुर्थक \omterm{def:g10:stats:quartiles}{परास}), सार्वभौमिक
        पैमाना ($z$), गारंटी (चेबिशेव), और प्रश्न 18 की चेतावनी: सारांश
        दबाते हैं, और दबाव में कुछ खोता है --- और कक्षा 12 का घंटी-वक्र
        \omterm{pb:g11:stat:1}{z-मानों} को यथार्थ \omterm{def:g10:proba:distribution}{प्रायिकताओं} में बदलने के लिए प्रतीक्षा कर रहा है।
\end{enumerate}
\end{problem}
