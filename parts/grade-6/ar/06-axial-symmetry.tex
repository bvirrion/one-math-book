\chapter{التماثل المحوري}\label{ch:g6:symmetry}

اِطوِ ورقة على طول \omterm{def:g6:lines:objects}{مستقيم}: فكل نقطة تسقط على نقطة أخرى،
هي صورتها في المرآة. وهذا الطيّ هو \emph{\omterm{def:g6:symmetry:def}{التماثل المحوري}} (أو
الانعكاس)، وهو أول تحويل هندسي في الكتاب --- أما
الثاني، وهو التماثل المركزي، فيأتي في \cref{ch:g7:central}.

\section{الانعكاسات}

\begin{definition}[النقطة المماثلة]\label{def:g6:symmetry:def}
ليكن $d$ مستقيمًا. \emph{مماثلة}\index{تماثل محوري}
النقطة $M$ بالنسبة إلى $d$ هي النقطة $M'$ التي يكون معها $d$
\omterm{def:g6:symmetry:bisector}{الواسط العمودي} \omterm{def:g6:lines:objects}{للقطعة} $[MM']$: أي إن \omterm{def:g6:lines:objects}{القطعة} $[MM']$ عمودية
على $d$، و $d$ يقطعها في منتصفها. والنقطة الواقعة \emph{على} $d$ هي
مماثلة نفسها.
\end{definition}

\begin{method}[إنشاء نقطة مماثلة]\label{met:g6:symmetry:construct}
لإنشاء \omterm{def:g6:symmetry:def}{مماثلة} $M'$ للنقطة $M$ بالنسبة إلى \omterm{def:g6:lines:objects}{المستقيم} $d$:
\begin{enumerate}
  \item اُرسم العمودي على $d$ المارّ بالنقطة $M$ (بالكوس)؛ وسمِّ
        $H$ النقطة التي يقطع فيها $d$؛
  \item قِس $MH$ بالفرجار أو بالمسطرة؛
  \item وضع $M'$ على العمودي نفسه، في الجهة الأخرى من $d$،
        بحيث $HM' = MH$.
\end{enumerate}
وعلى الورق المربّعات يكون أسرع: عُدّ المربعات من $M$ إلى المحور،
وعُدّ العدد نفسه في الجهة الأخرى.
\end{method}

\begin{example}[على الشبكة]\label{ex:g6:symmetry:grid}
إذا كان المحور خطًّا عموديًّا من الشبكة وكانت $M$ على بُعد $3$ مربعات إلى يساره،
فإن $M'$ على بُعد $3$ مربعات إلى يمينه، في السطر نفسه. وأما المحور
المائل، فالعدّ فيه يسير على الأقطار --- عموديًّا دائمًا على
المحور.
\end{example}

\begin{proposition}[ما يحفظه الانعكاس]\label{prop:g6:symmetry:preserved}
\omterm{def:g6:symmetry:def}{مماثلة} شكل بالنسبة إلى \omterm{def:g6:lines:objects}{مستقيم} شكل \emph{بالشكل نفسه
\omterm{def:g5:solids:volume}{والحجم} نفسه}: فالأطوال والزوايا والمحيطات والمساحات كلها
محفوظة. \omterm{def:g6:symmetry:def}{ومماثلة} \omterm{def:g6:lines:objects}{القطعة} \omterm{def:g6:lines:objects}{قطعة}، \omterm{def:g6:symmetry:def}{ومماثلة} الدائرة دائرة
\omterm{def:g3:shapes:shapes}{بنصف القطر} نفسه، \omterm{def:g6:symmetry:def}{ومماثلة} \omterm{def:g6:lines:objects}{المستقيم} \omterm{def:g6:lines:objects}{مستقيم}.
\end{proposition}
\admitted

\begin{remark}
وشيء واحد \emph{لا} يُحفظ: هو الاتجاه. فحرف «b» المنعكس
يصير «d» --- أي الكتابة المرآتية. وإن كان نسخ شكلك على ورق الشفّ
وقلبه يعيد الصورة، فالانعكاس صحيح.
\end{remark}

\section{محاور تماثل شكل}

\begin{definition}[محور التماثل]\label{def:g6:symmetry:axis}
يكون \omterm{def:g6:lines:objects}{المستقيم} $d$ \emph{محور تماثل}\index{محور تماثل}
لشكل حين يرسل الانعكاس بالنسبة إلى $d$ الشكلَ على
نفسه بالضبط --- فالطيّ على طول $d$ يجعل النصفين ينطبقان.
\end{definition}

\begin{example}\label{ex:g6:symmetry:shapes}
عدّ محاور التماثل:
\begin{itemize}
  \item \omterm{def:g6:shapes:triangles}{المثلث المتساوي الساقين}: $1$ (يمرّ بالرأس الأعلى ووسط
        القاعدة)؛ \omterm{def:g6:shapes:triangles}{والمثلث المتساوي الأضلاع}: $3$؛
  \item والمستطيل: $2$ («الخطان الوسطيان»، \emph{لا}
        القطران)؛ \omterm{def:g6:shapes:quadrilaterals}{والمعيّن}: $2$ (قطراه!)؛ والمربع: $4$؛
  \item والدائرة: كل \omterm{def:g6:lines:objects}{مستقيم} يمرّ بالمركز --- وهي بلا نهاية.
\end{itemize}
\end{example}

\begin{example}[التماثل يبرهن على التساوي]\label{ex:g6:symmetry:proves}
لماذا تتساوى زاويتا القاعدة في \omterm{def:g6:shapes:triangles}{المثلث المتساوي الساقين}؟ اِطوِ
المثلث على طول محور تماثله: فالنصف الأيسر يسقط بالضبط على
النصف الأيمن، فتسقط زاوية القاعدة اليسرى على زاوية القاعدة اليمنى ---
فلا بدّ أن يكون لهما القياس نفسه
(\cref{prop:g6:symmetry:preserved}: فالانعكاسات تحفظ الزوايا).
\end{example}

\section{الواسط العمودي}

\begin{definition}[الواسط العمودي]\label{def:g6:symmetry:bisector}
\emph{الواسط العمودي}\index{واسط عمودي}
\omterm{def:g6:lines:objects}{للقطعة} $[AB]$ هو \omterm{def:g6:lines:objects}{المستقيم} العمودي على $[AB]$ المارّ
بمنتصفها --- وهو \omterm{def:g6:symmetry:axis}{محور التماثل} الذي يبادل $A$ و $B$.
\end{definition}

\begin{proposition}[تساوي البُعد]\label{prop:g6:symmetry:bisector}
تكون النقطة $M$ على \omterm{def:g6:symmetry:bisector}{الواسط العمودي} \omterm{def:g6:lines:objects}{للقطعة} $[AB]$ بالضبط حين
تكون على المسافة نفسها من $A$ ومن $B$: أي $MA = MB$.
\end{proposition}
\admitted

\begin{method}[الإنشاء بالفرجار]\label{met:g6:symmetry:bisector}
لرسم \omterm{def:g6:symmetry:bisector}{الواسط العمودي} \omterm{def:g6:lines:objects}{للقطعة} $[AB]$ دون قياس:
\begin{enumerate}
  \item اِفتح الفرجار على أكثر من نصف $AB$؛
  \item اُرسم قوسين مركزهما $A$، فوق \omterm{def:g6:lines:objects}{القطعة} وتحتها؛
  \item وبالفتحة \emph{نفسها}، اُرسم قوسين مركزهما $B$؛
  \item صِل نقطتَي التقاطع: فهذا \omterm{def:g6:lines:objects}{المستقيم} هو \omterm{def:g6:symmetry:bisector}{الواسط
        العمودي} (فكل نقطة تقاطع على بُعد متساوٍ من $A$ و $B$).
\end{enumerate}
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:1}
اِنقل على الورق المربّعات: محورًا عموديًّا $d$، ونقطة $A$ على بُعد مربعين إلى
يساره، ونقطة $B$ على بُعد خمسة مربعات إلى يساره ومربع واحد أعلى.
واِبنِ مماثلتيهما $A'$ و $B'$ بالنسبة إلى $d$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:2}
اُرسم مستقيمًا $d$ ونقطة $M$ على بُعد $3$ سم من $d$ (لا على ورق
مربّعات). واِبنِ \omterm{def:g6:symmetry:def}{مماثلة} $M'$ للنقطة $M$ بالنسبة إلى $d$ بالكوس
والمسطرة، متّبعًا \cref{met:g6:symmetry:construct}. فكم تبعد
$M$ عن $M'$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:3}
أيّ الحروف اللاتينية الكبيرة، مكتوبة في أبسط صورها،
له \omterm{def:g6:symmetry:axis}{محور تماثل} عمودي (مثل A)؟ وأيّها له محور أفقي (مثل B)؟
وأيّها له الاثنان معًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:4}
كم \omterm{def:g6:symmetry:axis}{محور تماثل} لكل من: \omterm{def:g6:shapes:triangles}{المثلث المتساوي الأضلاع}؟ والمستطيل
(غير المربع)؟ \omterm{def:g6:shapes:quadrilaterals}{والمعيّن} (غير المربع)؟ والمربع؟ والدائرة؟
اُرسم كل شكل مع محاوره.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:5}
صواب أم خطأ؟ «قطرا المستطيل محورا تماثل
للمستطيل.» اِشرح بحجّة الطيّ أو برسم.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:6}
اِنعكس المثلث $ABC$ بالنسبة إلى \omterm{def:g6:lines:objects}{مستقيم} $d$ فصار $A'B'C'$. وإذا كان
$AB = 4$ سم، و $\widehat{BAC} = 70^\circ$، \omterm{def:g4:measure:perimeter}{ومحيط} $ABC$ يساوي
$12$ سم: فأعطِ، دون أيّ إنشاء، $A'B'$
و $\widehat{B'A'C'}$ \omterm{def:g4:measure:perimeter}{ومحيط} $A'B'C'$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:7}
اُرسم \omterm{def:g6:lines:objects}{قطعة} $[AB]$ طولها $7$ سم واِبنِ واسطها
العمودي بالفرجار (\cref{met:g6:symmetry:bisector}). واِختر نقطة
$M$ عليه كيفما كانت وتحقّق بالمسطرة من أن $MA = MB$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:8}
شجرتان تقومان عند النقطتين $A$ و $B$ من حديقة (مستوية). فأين يمكن
غرس نافورة تكون على المسافة نفسها من الشجرتين؟
صِف \emph{كل} المواضع الممكنة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:9}
على الورق المربّعات، اُرسم المحور $d$ (قطرًا من الشبكة في
$45^\circ$) وشكلًا صغيرًا على هيئة حرف L في جهة منه. واِبنِ
شكله المماثل بالنسبة إلى $d$. (اِعكس كل رأس: فمن أجل محور في $45^\circ$،
النقطة التي تبعد $2$ مربعين إلى يمين المحور تسقط $2$ مربعين فوقه.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:10}
اُرسم دائرة مركزها $O$، ومستقيمًا $d$ يمرّ بالنقطة $O$، ونقطة $M$ على
الدائرة. فأين تقع \omterm{def:g6:symmetry:def}{مماثلة} $M$ بالنسبة إلى $d$؟ واِشرح لماذا
يرسل الانعكاس الدائرة على نفسها.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:11}
باستعمال خاصّية تساوي البُعد (\cref{prop:g6:symmetry:bisector})،
اِشرح لماذا تقع نقطتا تقاطع القوسين في
\cref{met:g6:symmetry:bisector} فعلًا على \omterm{def:g6:symmetry:bisector}{الواسط
العمودي} \omterm{def:g6:lines:objects}{للقطعة} $[AB]$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:symmetry:12}
كرة بلياردو عند النقطة $A$ يجب أن ترتدّ عن جدار \omterm{def:g6:lines:objects}{مستقيم} $d$
لتبلغ النقطة $B$ (وكلتاهما في جهة واحدة من الجدار). اِعكس $B$
بالنسبة إلى $d$ فتحصل على $B'$، واُرسم \omterm{def:g6:lines:objects}{القطعة} $[AB']$. اِشرح لماذا يكون
أفضل موضع للارتداد هو حيث تقطع $[AB']$ الجدار. (الفكرة: فللمسار
$A \to P \to B$ الطول نفسه الذي للمسار $A \to P \to B'$.)
\end{exercise}

\section{مسألة: النافورة، والصحن المكسور، والمرآة
الماشية}

\begin{problem}[{مسألة نهاية الأسبوع --- الوسائط العمودية الثلاثة
لمثلث تلتقي في نقطة واحدة: الدائرة المارّة بثلاث
نقط، وما تفعله مرآتان معًا}]
\label{pb:g6:symmetry:1}
في \cref{exo:g6:symmetry:8} كان على النافورة أن تقف على بُعد
متساوٍ من شجرتين \emph{اثنتين}، وكان الجواب مستقيمًا كاملًا من
المواضع. وتغرس هذه المسألة شجرة \emph{ثالثة} --- فينهار \omterm{def:g6:lines:objects}{المستقيم}
كله إلى نقطة واحدة معيّنة تمامًا. وتخبئ تلك
النقطة كنزًا هندسيًّا: فهي تتيح لك رسم الدائرة الوحيدة
المارّة بثلاث نقط معطاة، وإعادة بناء صحن كامل
من شظية مكسورة، وهي تنتمي إلى كل مثلث. ويجري الجزء
الأخير تجربة بمرآتين نتيجتها
تُعلن هندسة السنة المقبلة.

\medskip
\noindent\textbf{الجزء الأول --- شجرتان، ثم خريطة
مناطق.}
\begin{enumerate}
  \item اُرسم نقطتين $A$ و $B$ حيث $AB = 6$ سم واِبنِ
        \omterm{def:g6:symmetry:bisector}{الواسط العمودي} \omterm{def:g6:lines:objects}{للقطعة} $[AB]$ بالفرجار
        (\cref{met:g6:symmetry:bisector}). واِختر نقطة
        $M$ عليه وتحقّق بالمسطرة من أن $MA = MB$.
  \item يقول \cref{prop:g6:symmetry:bisector} أمرين في
        آن واحد: كل نقطة \emph{على} الواسط على بُعد متساوٍ
        من $A$ و $B$، وكل نقطة على بُعد متساوٍ \emph{تقع
        على} الواسط. فأيّهما يجيب عن سؤال
        النافورة في \cref{exo:g6:symmetry:8}؟ وماذا تقول
        القضية عن نقطة \emph{ليست} على
        الواسط؟
  \item خذ نقطة $M$ في جهة الواسط التي فيها
        $A$. \omterm{def:g6:lines:objects}{فالقطعة} $[MB]$ تقطع الواسط في نقطة
        $P$. قارن المسار $M \to P \to B$ بالمسار
        $M \to P \to A$، واِشرح لماذا تكون $M$ أقرب إلى $A$
        منها إلى $B$: فالواسط يشقّ الورقة إلى
        «جهة $A$» و«جهة $B$».
  \item مدرستان $A$ و $B$ تخدمان مدينة. وكل طفل يمشي
        إلى المدرسة الأقرب. اُرسم خريطة صغيرة فيها $A$ و $B$،
        وظلّل المنطقة المضبوطة من المدينة التي تخدمها المدرسة
        $A$. فأيّ \omterm{def:g6:lines:objects}{مستقيم} يكوّن الحدّ؟
  \item تُفتح مدرسة ثالثة $C$ (ضعها بحيث تكوّن المدارس الثلاث
        مثلثًا). اِبنِ \omterm{def:g6:symmetry:bisector}{الواسط العمودي}
        \omterm{def:g6:lines:objects}{للقطعة} $[AB]$ \omterm{def:g6:symmetry:bisector}{والواسط العمودي} \omterm{def:g6:lines:objects}{للقطعة} $[BC]$، وسمِّ
        $O$ نقطة تقاطعهما. فأيّ تساويين بين
        المسافات يحقّقهما $O$، ولماذا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- ثلاث أشجار، ونقطة واحدة،
ودائرة واحدة.}
\begin{enumerate}[resume]
  \item اِستنتج من السؤال 5 أن $OA = OC$ --- وأن
        $O$ يقع إذن أيضًا على \omterm{def:g6:symmetry:bisector}{الواسط العمودي} \omterm{def:g6:lines:objects}{للقطعة}
        $[AC]$، التي لم ترسمها قطّ. واِستنتج: \emph{الوسائط
        العمودية الثلاثة لأضلاع مثلث كلها
        تمرّ بنقطة واحدة.}
  \item اِشرح لماذا تمرّ الدائرة التي مركزها $O$ ونصف قطرها $OA$
        بالنقطة $B$ وبالنقطة $C$ أيضًا: فهي \emph{الدائرة}
        المارّة بالنقط الثلاث --- أي \emph{الدائرة
        المحيطة}\index{دائرة محيطة}
        بالمثلث $ABC$.
  \item اِبنِ مثلثًا أضلاعه $6$ سم و $5$ سم و
        $4$ سم (\cref{met:g6:shapes:construct})، ثم
        دائرته المحيطة. ولماذا يكفي رسم
        واسطين \emph{اثنين} لإيجاد المركز؟
  \item الصحن المكسور: يستخرج عالِم آثار شظية لم يبقَ
        سليمًا منها إلا \omterm{def:g6:lines:objects}{قطعة} من حافة الصحن الدائرية.
        صِف وصفة تستعيد المركز \omterm{def:g3:shapes:shapes}{ونصف
        قطر} الصحن الأصلي --- واِختبرها: اُرسم
        قوس دائرة بفرجارك، و«اِنسَ» المركز،
        وعلّم ثلاث نقط على القوس، ثم أعِد بناءه.
  \item أيمكن أن تكون ثلاث نقط \emph{بلا} دائرة تمرّ
        بها؟ اِفترض أن $A$ و $B$ و $C$ مستقيمية. فماذا يمكن
        أن يقال عن الواسطين العموديين \omterm{def:g6:lines:objects}{للقطعتين} $[AB]$ و
        $[BC]$ (\cref{prop:g6:lines:facts})؟ واِستنتج.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- محاور في كل مكان، والمرآة
الماشية.}
\begin{enumerate}[resume]
  \item عدّ \cref{ex:g6:symmetry:shapes} $3$ محاور
        \omterm{def:g6:shapes:triangles}{للمثلث المتساوي الأضلاع} و $4$ للمربع. فخمّن
        العدد \omterm{def:g4:geometry:polygon}{لخماسي} منتظم ولسداسي منتظم،
        وصِف أين تمرّ المحاور في كل حالة (\omterm{def:g3:numbers:evenodd}{العدد
        الفردي} للأضلاع مقابل الزوجي: فالحالتان
        تختلفان).
  \item اِطوِ ورقة مرة واحدة وقُصّ شكلًا عبر الطبقتين
        معًا؛ ثم اُنشرها. ولماذا يكون للثقب \emph{دائمًا}
        \omterm{def:g6:symmetry:axis}{محور تماثل}، مهما قصصت، وأيّ \omterm{def:g6:lines:objects}{مستقيم} هو
        المحور؟
  \item اِطوِ الورقة مرتين --- والطيّة الثانية عمودية
        على الأولى --- ثم قُصّ واُنشر. فكم \omterm{def:g6:symmetry:axis}{محور
        تماثل} للثقب على الأقلّ، وكم
        نسخة من قصّتك ترى؟ (وقصاصات الثلج الورقية هي
        هذه الفكرة مطويّة أكثر.)
  \item المرآة الماشية: على الورق المربّعات، اُرسم محورًا عموديًّا
        $d_1$، وعَلَمًا صغيرًا $F$ في يساره، ومحورًا عموديًّا
        ثانيًا $d_2$ على بُعد $4$ مربعات إلى يمين $d_1$.
        واِعكس $F$ بالنسبة إلى $d_1$ (فتحصل على الصورة $F'$)، ثم اِعكس
        $F'$ بالنسبة إلى $d_2$ (فتحصل على $F''$). قارن $F$ و $F''$:
        الاتجاه نفسه أم مرآتيّ؟ وبكم مربعًا
        اِنتقل الشكل؟ قارن ذلك بالمسافة بين
        المحورين.
  \item التجربة نفسها مع $d_2$ \emph{عموديًّا} على
        $d_1$: اِعكس العَلَم بالنسبة إلى المحور العمودي، ثم
        الصورة بالنسبة إلى الأفقي. صِف كيف يقع $F''$
        بالنسبة إلى $F$ وإلى نقطة تقاطع
        المحورين. (فقد اِكتشفت للتوّ تحويل السنة
        المقبلة: \cref{ch:g7:central}.)
\end{enumerate}
\end{problem}
