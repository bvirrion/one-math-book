\chapter{الأعداد الصحيحة}\label{ch:g6:wholes}

كل شيء في الرياضيات يبدأ بالعدّ. يراجع هذا الفصل كيف تُكتب
الأعداد الصحيحة وتُقارن وتُجمع --- ويتأمل بعناية في
القسمة، وهي العملية التي تعطي \omterm{def:g3:division:remainder}{خارج قسمة} \emph{و}
باقيًا.

\section{كتابة الأعداد الصحيحة ومقارنتها}

\begin{definition}[القيمة المكانية]\label{def:g6:wholes:place}
طريقتنا في كتابة الأعداد تستعمل عشرة أرقام (من $0$ إلى $9$)، و
\emph{قيمة الرقم تتوقف على مرتبته}\index{قيمة مكانية}: ففي
$5\,208$، يعدّ الرقم $5$ الآلاف، ويعدّ $2$ المئات، ويعدّ
$0$ العشرات (ولا عشرات هنا)، ويعدّ $8$ الآحاد:
\[
5\,208 = 5 \times 1000 + 2 \times 100 + 0 \times 10 + 8 .
\]
\end{definition}

\begin{method}[مقارنة عددين صحيحين]\label{met:g6:wholes:compare}
\begin{enumerate}
  \item العدد ذو الأرقام الأكثر هو الأكبر
        ($1\,203 > 989$).
  \item وإن تساوى عدد الأرقام، فقارن رقمًا رقمًا
        من \emph{اليسار}؛ وأول اختلاف يفصل:
        $6\,742 > 6\,715$ لأن في مرتبة العشرات $4 > 1$.
\end{enumerate}
والرمزان هما $<$ («أصغر من») و $>$ («أكبر من»)؛ والطرف
الضيق من الرمز يشير إلى العدد الصغير.
\end{method}

\begin{example}\label{ex:g6:wholes:order}
رتّب من الأصغر إلى الأكبر: $908$؛ $89$؛ $1\,001$؛ $980$.
أولًا بعدد الأرقام: $89$ (رقمان) يتقدم، و $1\,001$ (أربعة
أرقام) يتأخر. وبين $908$ و $980$: رقم المئات نفسه $9$، ثم
العشرات $0 < 8$، إذن $908 < 980$. والجواب النهائي:
\[
89 < 908 < 980 < 1\,001 .
\]
\end{example}

\section{الجمع والطرح والضرب}

\begin{example}[الحساب بالأعمدة]\label{ex:g6:wholes:columns}
للجمع أو الطرح، حاذِ الآحاد تحت الآحاد، والعشرات تحت
العشرات، واعمل من اليمين إلى اليسار، محتفظًا عند اللزوم:
\[
\begin{array}{r}
4\,5\,7 \\
+\ 2\,8\,6 \\
\hline
7\,4\,3
\end{array}
\qquad\qquad
\begin{array}{r}
6\,0\,3 \\
-\ 1\,4\,7 \\
\hline
4\,5\,6
\end{array}
\]
في الجمع: $7 + 6 = 13$، اُكتب $3$ واحتفظ برقم $1$؛ ثم
$5 + 8 + 1 = 14$، اُكتب $4$ واحتفظ برقم $1$؛ ثم $4 + 2 + 1 = 7$.
وتحقّق من الطرح بالجمع العكسي: $456 + 147 = 603$.
\end{example}

\begin{remark}[المفردات]
ناتج الجمع \emph{مجموع}\index{مجموع}؛ وناتج الطرح
\emph{فرق}\index{فرق}؛ وناتج الضرب
\emph{جداء}\index{جداء}؛ وناتج القسمة
\emph{خارج قسمة}\index{خارج القسمة}. والأعداد الداخلة في العملية هي
\emph{الحدود} (في $+$ و $-$) أو \emph{العوامل} (في $\times$).
\end{remark}

\begin{example}[الحساب الذكي]\label{ex:g6:wholes:smart}
إعادة ترتيب الحدود أو العوامل توفّر جهدًا كثيرًا في الغالب:
\[
17 + 58 + 3 = (17 + 3) + 58 = 20 + 58 = 78 ,
\]
\[
25 \times 17 \times 4 = (25 \times 4) \times 17 = 100 \times 17 = 1700 .
\]
\end{example}

\section{القسمة}

\begin{theorem}[القسمة الإقليدية]\label{thm:g6:wholes:division}
إذا أُعطي عددان صحيحان $a$ (وهو \emph{المقسوم}) و $b \neq 0$ (وهو
\emph{القاسم})، فهناك طريقة واحدة بالضبط لكتابة
\[
a = b \times q + r
\qquad\text{بحيث}\qquad
0 \leq r < b .
\]
والعدد $q$ هو \emph{خارج القسمة}\index{خارج القسمة} و $r$ هو
\emph{الباقي}\index{باقي القسمة} في قسمة $a$ على $b$.
\end{theorem}
\admitted

\begin{example}\label{ex:g6:wholes:division}
اِقسم $137$ على $4$، خطوة خطوة.
\begin{enumerate}
  \item كم مرة يدخل $4$ في $13$؟ ثلاث مرات
        ($4 \times 3 = 12$)، \omterm{def:g3:division:remainder}{والباقي} $13 - 12 = 1$.
  \item وأنزِل $7$: كم مرة يدخل $4$ في $17$؟
        أربع مرات ($4 \times 4 = 16$)، \omterm{def:g3:division:remainder}{والباقي} $1$.
\end{enumerate}
إذن $q = 34$ و $r = 1$:
\[
137 = 4 \times 34 + 1, \qquad\text{وفعلًا } 0 \leq 1 < 4 .
\]
والتحقّق: $4 \times 34 = 136$، و $136 + 1 = 137$.
\end{example}

\begin{definition}[القابل للقسمة]\label{def:g6:wholes:divisible}
حين يكون \omterm{def:g3:division:remainder}{الباقي} $0$، نقول إن $a$
\emph{قابل للقسمة}\index{قابل للقسمة} على $b$: فمثلًا $135$
قابل للقسمة على $5$ لأن $135 = 5 \times 27 + 0$.
\end{definition}

\begin{method}[المسائل اللفظية مع القسمة]\label{met:g6:wholes:problems}
بعد حساب $a = bq + r$، عُد دائمًا إلى السؤال:
\begin{enumerate}
  \item «كم علبة أو فريقًا أو صفحة كاملة؟» --- الجواب هو
        \omterm{def:g3:division:remainder}{خارج القسمة} $q$؛
  \item «وكم بقي؟» --- الجواب هو \omterm{def:g3:division:remainder}{الباقي}
        $r$؛
  \item «وكم علبة تلزم للجميع؟» --- إن كان $r > 0$، فواحدة
        \emph{أكثر} من \omterm{def:g3:division:remainder}{خارج القسمة}: $q + 1$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:wholes:bus}
يذهب $137$ تلميذًا في رحلة بحافلات فيها $40$ مقعدًا. والقسمة:
$137 = 40 \times 3 + 17$. فثلاث حافلات تحمل $120$ تلميذًا، ويبقى $17$
تلميذًا --- وهم يحتاجون إلى حافلة أيضًا! إذن تلزم $4$ حافلات، مع أن
\omterm{def:g3:division:remainder}{خارج القسمة} $3$.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g6:wholes:1}
اُكتب بالأرقام: «ثلاثة آلاف وسبعة وأربعون»؛ «عشرون ألفًا
وخمسمئة»؛ «مئة وأربعة». ثم اُكتب $60\,013$ بالحروف.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:2}
في العدد $47\,209$: ما رقم المئات؟ وما رقم
الآحاد؟ وماذا يعدّ الرقم $4$؟ واُكتب العدد مجموعَ
\omterm{def:g5:division:multiple}{مضاعفات} $1$ و $10$ و $100$، \dots\ كما في
\cref{def:g6:wholes:place}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:3}
اُنقل وأكمل بالإشارة $<$ أو $>$:
\[
507 \;?\; 570, \qquad
1\,099 \;?\; 989, \qquad
23\,456 \;?\; 23\,465, \qquad
9\,999 \;?\; 10\,000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:4}
رتّب من الأصغر إلى الأكبر: $2\,043$؛ $978$؛ $2\,304$؛ $2\,034$؛
$1\,001$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:5}
احسب بالأعمدة: $348 + 576$؛ \quad $805 - 268$؛ \quad
$307 \times 26$. وتحقّق من الطرح بعملية جمع.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:6}
احسب بذكاء، بتجميع الحدود أو العوامل:
\[
38 + 45 + 12 + 5, \qquad
2 \times 83 \times 50, \qquad
4 \times 78 \times 25 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:7}
لكل قسمة، أعطِ \omterm{def:g3:division:remainder}{خارج القسمة} \omterm{def:g3:division:remainder}{والباقي}، واكتب
المساواة $a = b \times q + r$:
\[
95 \div 7, \qquad
230 \div 6, \qquad
504 \div 8 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:8}
دون أن تقسم، اِشرح لماذا لا تكون المساواة $173 = 8 \times 20 + 13$
هي القسمة الإقليدية للعدد $173$ على $8$، وجِد \omterm{def:g3:division:remainder}{خارج القسمة}
\omterm{def:g3:division:remainder}{والباقي} الصحيحين.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:9}
يُعبَّأ البيض في علب من $12$.
\begin{enumerate}
  \item في مزرعة $2\,000$ بيضة. فكم علبة كاملة تستطيع أن تملأ،
        وكم بيضة تبقى؟
  \item ويحتاج مخبز إلى $150$ بيضة. فكم علبة يجب أن يشتري؟
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:10}
اليوم هو الثلاثاء. فأي يوم من الأسبوع يكون بعد $100$ يوم؟
(في الأسبوع $7$ أيام: اِقسم وفكّر في \omterm{def:g3:division:remainder}{الباقي}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:11}
في قسمة على $9$، \omterm{def:g3:division:remainder}{خارج القسمة} $56$ \omterm{def:g3:division:remainder}{والباقي} أكبر ما
يمكن. فما المقسوم؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:wholes:12}
أنا عدد من ثلاثة أرقام. رقم مئاتي ضعف رقم آحادي،
ورقم عشراتي $0$، ومجموع أرقامي $9$. فمن أنا؟
(اِستدلّ خطوة خطوة، ثم تحقّق.)
\end{exercise}

\section{مسألة: غاوس الصغير ومجموع المئة عدد
الأولى}

\begin{problem}[{مسألة نهاية الأسبوع --- جمع $1 + 2 + 3 + \dots +
100$ في عشر ثوانٍ، والأعداد المثلثية}]
\label{pb:g6:wholes:1}
قصة مشهورة: أراد معلّم أن يشغل صفّه، فطلب من
تلاميذه أن يجمعوا كل الأعداد من $1$ إلى $100$. فكتب صبي
$5\,050$ على لوحه في الحال تقريبًا --- وهو كارل فريدريش غاوس، في
التاسعة من عمره، وسيصير من أعظم الرياضيين على
الإطلاق. ولم يكن سرّه إلا \emph{الحساب الذكي}
(\cref{ex:g6:wholes:smart}): تجميع الحدود بحيث تكون كل مجموعة
سهلة. وتعيد هذه المسألة اكتشاف حيلته، وتجعلها تنجح في كل
حال، وتُظهر كم تتكرر هذه المجاميع في الحياة اليومية.

\medskip
\noindent\textbf{الجزء الأول --- أزواج غاوس.}
\begin{enumerate}
  \item احسب $1 + 2 + 3 + 4 + 5$ بتجميع الحدود بذكاء.
  \item وفي $1 + 2 + 3 + \dots + 10$: زوِّج الحدّ الأول مع
        الأخير ($1 + 10$)، والثاني مع ما قبل الأخير
        ($2 + 9$)، وهكذا. فكم يساوي كل زوج؟ وكم عدد
        الأزواج؟ ثم أتمّ الحساب.
  \item والطريقة نفسها في $1 + 2 + \dots + 20$: كم يساوي كل زوج،
        وكم عدد الأزواج، وما \omterm{def:g1:addition:def}{المجموع}؟
  \item والآن \omterm{def:g1:addition:def}{مجموع} المعلّم، $1 + 2 + \dots + 100$: اِشرح
        لماذا تتساوى الأزواج $1 + 100$ و $2 + 99$ و $3 + 98, \dots$
        كلها، ولماذا عددها $50$ بالضبط،
        وآخرها $50 + 51$.
  \item اُكتب جواب غاوس ضربًا واحدًا، ثم
        احسبه.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- حيلة لا تخيب أبدًا.}
\begin{enumerate}[resume]
  \item جرّب التزويج على $1 + 2 + \dots + 9$: هذه المرة يبقى عدد
        واحد بلا شريك. فأيّه؟ زوِّج حوله ثم
        احسب \omterm{def:g1:addition:def}{المجموع}.
  \item وإليك صورة من الحيلة تنجح سواء أكان عدد الحدود
        زوجيًّا أم فرديًّا: اُكتب \omterm{def:g1:addition:def}{المجموع} مرتين، مرة
        بالترتيب ومرة بالعكس، سطرًا تحت سطر،
        \[
        \begin{array}{ccccccc}
        1 & + & 2 & + & \dots & + & 9 \\
        9 & + & 8 & + & \dots & + & 1
        \end{array}
        \]
        ثم اجمع \emph{عمودًا عمودًا}. فكم يساوي كل عمود؟
        وكم عدد الأعمدة؟ واِشرح لماذا يكون \omterm{def:g1:addition:def}{مجموع}
        السطرين $9 \times 10 = 90$، واستنتج أن
        $1 + 2 + \dots + 9 = 45$ --- وهو الجواب نفسه في
        السؤال 6.
  \item اِستعمل حيلة السطرين لحساب $1 + 2 + \dots + 50$،
        ثم $1 + 2 + \dots + 200$.
  \item احسب \omterm{def:g1:addition:def}{مجموع} المئة عدد \emph{الزوجي} الأولى،
        $2 + 4 + 6 + \dots + 200$. (قارن كل حدّ بحدّ
        من \omterm{def:g1:addition:def}{مجموع} غاوس: فلا حاجة إلى تزويج جديد.)
  \item احسب $101 + 102 + \dots + 200$، مستعملًا مجموعين
        تعرفهما وعملية طرح.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- \omterm{pb:g6:wholes:1}{الأعداد المثلثية} في كل مكان.}
والمجاميع $1$ و $3$ و $6$ و $10$ و $15, \dots$ الناتجة عن
$1 + 2 + \dots + n$ تسمّى \emph{أعدادًا
مثلثية}\index{أعداد مثلثية}، لأنها تعدّ أشياء
مرصوصة في مثلث.
\begin{enumerate}[resume]
  \item يلتقي ثمانية أصدقاء، ويصافح كل واحد كل واحد من الآخرين
        مرة واحدة بالضبط. اِشرح لماذا يكون عدد
        المصافحات $7 + 6 + 5 + 4 + 3 + 2 + 1$، ثم احسبه.
  \item وتبني بقّالة هرمًا مثلثيًّا من العلب: $12$ علبة في
        الصف السفلي، و $11$ في الذي فوقه، وهكذا حتى علبة
        واحدة في القمة. فكم علبة تحتاج؟ (اِستعمل حيلة
        السؤال 7.)
  \item اُرسم \omterm{def:g1:addition:def}{المجموع} $1 + 2 + 3 + 4$ سلّمًا من المربعات:
        مربع واحد في العمود الأول، ومربعان في الثاني، وثلاثة
        في الثالث، وأربعة في الرابع. وبيّن على رسمك
        أن سلّمين \emph{اثنين} كهذا، أحدهما مقلوب رأسًا
        على عقب، ينطبقان تمامًا في مستطيل من $4$ صفوف
        فيها $5$ مربعات --- واِشرح لماذا يكون هذا هو حيلة
        السطرين في السؤال 7، مرسومةً بدل أن تكون مكتوبة.
  \item وتدقّ ساعة $1$ مرة عند الواحدة، و $2$ مرتين عند
        الثانية، \dots، و $12$ مرة عند الثانية عشرة. فكم
        دقّة في الساعات الاثنتي عشرة؟ وكم في اليوم
        كله؟
  \item وأخيرة لغاوس: $5\,050$ ثانية --- أهي أكثر أم
        أقلّ من ساعة ونصف؟ حوّلها إلى ساعات ودقائق
        وثوانٍ، مستعملًا القسمة الإقليدية
        (\cref{thm:g6:wholes:division}) مرتين.
\end{enumerate}
\end{problem}
