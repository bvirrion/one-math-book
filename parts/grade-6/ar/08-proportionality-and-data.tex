\chapter{التناسب والمعطيات}\label{ch:g6:propdata}

إذا كانت ثلاثة دفاتر ثمنها $6$ يورو، فستة دفاتر ثمنها $12$: ضاعف
الدفاتر يتضاعف الثمن. والمقادير التي تسلك هذا المسلك
\emph{متناسبة} --- وهي من أنفع الأفكار في الرياضيات
كلها، وتُطوَّر أكثر في \cref{ch:g7:prop}. وينتهي الفصل
بقراءة مخططات المعطيات البسيطة ورسمها.

\section{المقادير المتناسبة}

\begin{definition}[التناسب]\label{def:g6:propdata:def}
يكون مقداران \emph{متناسبين}\index{تناسب} حين
تُحصَل قيم أحدهما من قيم الآخر بالضرب
دائمًا في العدد \emph{نفسه}، ويُسمّى \emph{معامل
التناسب}.
\end{definition}

\begin{example}\label{ex:g6:propdata:table}
دفاتر ثمن الواحد $2$ يورو:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cccc}
الدفاتر & $3$ & $5$ & $8$ & $12$ \\
\hline
الثمن (باليورو) & $6$ & $10$ & $16$ & $24$ \\
\end{tabular}
\end{center}
كل ثمن هو عدد الدفاتر $\times 2$: فالمعامل هو
$2$ (وهو ثمن الوحدة). وللتحقّق السريع من كون جدول متناسبًا:
يجب أن تتساوى كل «خوارج قسمة الأعمدة» $\frac{6}{3}, \frac{10}{5}, \frac{16}{8},
\frac{24}{12}$ --- وهي هنا كلها تساوي $2$.
\end{example}

\begin{example}[مثال مضادّ]\label{ex:g6:propdata:nonexample}
السنّ والطول غير \omterm{def:g6:propdata:def}{متناسبين}: فابن $12$ سنة ليس ضعف طول
ابن $6$ سنوات. وتحقّق بالأعداد: $1.50$ م عند $12$ سنة و
$1.15$ م عند $6$ سنوات تعطي خارجَي القسمة $\frac{1.50}{12} = 0.125$ و
$\frac{1.15}{6} \approx 0.19$ --- وهما غير متساويين.
\end{example}

\begin{method}[إكمال جدول تناسب]\label{met:g6:propdata:complete}
ثلاث طرق، تُختار بحرّية:
\begin{enumerate}
  \item \emph{المعامل}: جِد المضروب فيه من عمود كامل
        واحد، وطبّقه على البقية؛
  \item \emph{الأعمدة}: يمكن ضرب عمود في عدد، أو جمع
        عمودين، لإنتاج عمود جديد؛
  \item \emph{العودة إلى الوحدة}: جِد قيمة عنصر \emph{واحد}
        أولًا، ثم اِضرب.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:propdata:methods}
خمسة أقداح متطابقة ثمنها $15$ يورو؛ فكم ثمن ثمانية أقداح؟

\emph{العودة إلى الوحدة}: ثمن القدح الواحد $15 \div 5 = 3$ يورو، إذن ثمانية
أقداح ثمنها $8 \times 3 = 24$ يورو.

\emph{الأعمدة}: $8 = 5 + 3$؛ وثلاثة أقداح ثمنها $\frac{3}{5}$ من $15$،
أي\ $9$؛ إذن ثمانية أقداح ثمنها $15 + 9 = 24$ يورو. الجواب نفسه، كما
يجب أن يكون.
\end{example}

\section{النسب المئوية}

\begin{definition}[النسبة المئوية]\label{def:g6:propdata:percent}
عبارة «$t\,\%$ من مقدار» تعني \omterm{def:g6:fractions:def}{الكسر} $\frac{t}{100}$ منه:
و $25\,\%$ من $60$ هي $\frac{25}{100} \times 60 = 15$. والنسبة المئوية
\omterm{def:g6:propdata:def}{تناسب} معامله $\frac{t}{100}$.
\end{definition}

\begin{example}\label{ex:g6:propdata:percent}
قسم فيه $30$ تلميذًا نسبة البنات فيه $40\,\%$. وعدد البنات، خطوة
خطوة: $40\,\%$ تعني $\frac{40}{100} = 0.4$، ثم
$0.4 \times 30 = 12$ بنتًا. ومعالم مفيدة:
$50\,\%$ هي النصف، و $25\,\%$ الرُّبع، و $10\,\%$ العُشر ---
إذن $10\,\%$ من $30$ هو $3$، و $40\,\% = 4 \times 10\,\%$ يساوي
$4 \times 3 = 12$: الجواب نفسه، محسوبًا ذهنيًّا.
\end{example}

\section{قراءة المخططات ورسمها}

\begin{method}[قراءة جدول أو مخطط]\label{met:g6:propdata:read}
مهما كانت الصورة --- جدولًا أو مخطط أعمدة أو منحنى:
\begin{enumerate}
  \item اِقرأ العنوان والوحدات على المحورين أولًا؛
  \item وللإجابة عن سؤال، حدّد العمود الصحيح، ثم اِقرأ
        القيمة بعناية على التدريج؛
  \item وللمقارنة، قارن الأعمدة بالارتفاع --- لكن تحقّق من أن المحور
        يبدأ من $0$، وإلا ضلّلت الصورة.
\end{enumerate}
\end{method}

يبيّن مخطط الأعمدة أدناه عدد الكتب المستعارة من مكتبة

\section{مسألة: الخرائط والتصاميم، وكيف تكذب بمخطط}

\begin{problem}[{مسألة نهاية الأسبوع --- سلّم الخريطة تناسب:
فالأطوال تتبع المعامل، والمساحات تتبع مربّعه،
والمخططات السيّئة الرسم تخدع العين}]
\label{pb:g6:propdata:1}
كل خريطة وتصميم \omterm{def:g5:solids:def}{ومجسّم} مصغّر يخضع لقاعدة واحدة: الأطوال الحقيقية
والأطوال المرسومة \emph{متناسبة}
(\cref{def:g6:propdata:def}). وللمعامل اسم مشهور
--- وهو \emph{السلّم}\index{سلم} --- وهو يخفي فخّين
تنصبهما هذه المسألة عمدًا: فالمساحات \emph{لا} تتبع
السلّم، والنسب المئوية، كالمخططات، تتوقف كليًّا على ما
تُقاس بالنسبة إليه.

\medskip
\noindent\textbf{الجزء الأول --- قراءة خريطة.}
خريطة للمشي سلّمها $1:100\,000$: فكل سنتيمتر على الخريطة
يمثّل $100\,000$ سنتيمتر على الأرض.
\begin{enumerate}
  \item حوّل $100\,000$ سم إلى كيلومترات. فماذا يمثّل $1$ سم
        من هذه الخريطة، بالكيلومتر؟
  \item قريتان تبعد إحداهما عن الأخرى $7.5$ سم على الخريطة؛ وشاطئ
        بحيرة منحنى طوله $12$ سم. أعطِ المسافتين الحقيقيتين.
  \item طريق رومانية مستقيمة تمامًا طولها $20$ كم. فكم يبلغ طولها
        على الخريطة؟
  \item خريطة ثانية تعلن «$1$ سم من أجل $5$ كم». اُكتب
        سلّمها على الصورة $1: \,?$. وأيّ الخريطتين تبيّن
        تفاصيل أكثر للمنطقة نفسها؟
  \item اِصنع جدولًا صغيرًا (مسافة على الخريطة $1$ و $2$ و $7.5$ سم
        مقابل المسافة الحقيقية) لخريطة المشي، واِشرح
        لماذا يكون السلّم تناسبًا بالضبط بمعنى
        \cref{def:g6:propdata:def}. وما المعامل الذي
        يحوّل السنتيمترات على الخريطة إلى كيلومترات؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- تصميم غرفة زينة.}
ترسم زينة تصميم غرفتها --- وهي مستطيل $4$ م في
$3$ م --- بسلّم $1:50$.
\begin{enumerate}[resume]
  \item فما أبعاد الغرفة على التصميم؟
  \item سريرها قياسه $2$ م في $0.9$ م، وعرض باب الغرفة
        $80$ سم. أعطِ القياسات الثلاثة على التصميم.
  \item والآن الفخّ. اِحسب \omterm{def:g6:measure:area}{المساحة} الحقيقية للغرفة
        بوحدة م$^2$، ثم \omterm{def:g6:measure:area}{مساحة} تصميمها بوحدة سم$^2$. وحوّل
        \omterm{def:g6:measure:area}{المساحة} الحقيقية إلى سم$^2$
        (\cref{def:g6:measure:area}) واقسمها على \omterm{def:g6:measure:area}{مساحة}
        التصميم: فهل \omterm{def:g3:division:remainder}{خارج القسمة} $50$؟
  \item فسّر العدد الذي وجدته: ففي تصميم بسلّم $1:50$، كل
        سنتيمتر مربع من الورق يمثّل مربعًا حقيقيًّا
        $50$ سم في $50$ سم. فكم سم$^2$ حقيقيًّا يساوي ذلك؟ واُذكر
        القاعدة: حين تُقسم الأطوال على $50$، تُقسم المساحات
        على\,\dots
  \item سجّادة مدوّرة تغطّي $6$ سم$^2$ على التصميم. فأيّ \omterm{def:g6:measure:area}{مساحة}
        حقيقية تغطّي، بوحدة سم$^2$ ثم بوحدة م$^2$؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- كيف تكذب بمخطط
وبنسبة مئوية.}
\begin{enumerate}[resume]
  \item باع متجر بما قيمته $105$ يورو يوم السبت و $110$ يوم
        الأحد. فرسم المدير مخطط أعمدة يبدأ محوره
        العمودي من $100$: فيرتفع عمود السبت $5$ وحدات
        صغيرة، وعمود الأحد $10$. فماذا توحي الصورة
        عن الأحد، وما الحقيقة؟ (اِحسب
        زيادة الأحد نسبةً مئوية من السبت، مقرّبة إلى أقرب
        نسبة صحيحة.) وأيّ تحذير من
        \cref{met:g6:propdata:read} أهمله المدير؟
  \item أعِد رسم المخطط الأمين (أو صِفه)، بمحور
        يبدأ من $0$: فكيف يتقارن العمودان الآن؟
  \item تنمو مجموعة من $40$ إلى $60$ طابعًا: اِحسب
        الزيادة نسبةً مئوية من القيمة الابتدائية. ثم
        تتقلّص عائدة من $60$ إلى $40$: اِحسب النقصان
        نسبةً مئوية من قيمته \emph{هو} الابتدائية. فلماذا يختلف
        الجوابان، من أجل الطوابع $20$ نفسها؟
  \item موسم التخفيضات: معطف ثمنه $100$ يورو يُخفَّض «$20\,\%$»،
        وعند الصندوق «$10\,\%$ إضافية على
        الثمن المخفَّض». اِحسب الثمن النهائي خطوة خطوة.
        فهل \omterm{def:g1:addition:def}{مجموع} التخفيض $30\,\%$؟ اِشرح للمشتري
        ما هو حقًّا.
  \item الخاتمة، عودًا إلى خريطة السؤال 4 ($1$ سم من أجل
        $5$ كم): غابة تشغل $3$ سم$^2$ من تلك الخريطة. فما
        مساحتها الحقيقية، بوحدة كم$^2$؟ واختم بقاعدة
        هذه المسألة كلها: الأطوال تتبع السلّم، والمساحات
        تتبع\,\dots
\end{enumerate}
\end{problem}
