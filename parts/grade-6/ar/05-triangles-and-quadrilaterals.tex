\chapter{المثلثات والرباعيات}\label{ch:g6:shapes}

المثلثات والرباعيات هي الأشكال الأساسية في الهندسة. وهدف
هذا الفصل أن تعرف عائلاتها الخاصّة --- \omterm{def:g6:shapes:triangles}{متساوي الساقين}،
\omterm{def:g6:shapes:triangles}{ومتساوي الأضلاع}، \omterm{def:g6:shapes:triangles}{والقائم الزاوية}؛ والمستطيل \omterm{def:g6:shapes:quadrilaterals}{والمعيّن} والمربع --- من
\emph{خاصّياتها المميِّزة}، وأن تبنيها بدقّة
بالمسطرة والفرجار.

\section{المثلثات}

\begin{definition}[المثلثات الخاصّة]\label{def:g6:shapes:triangles}
للمثلث $ABC$ ثلاثة \omterm{def:g5:solids:def}{رؤوس} وثلاثة أضلاع. وهو:
\begin{itemize}
  \item \emph{متساوي الساقين}\index{مثلث متساوي الساقين} في $A$ حين
        $AB = AC$ (ضلعان متساويان)؛
  \item و\emph{متساوي الأضلاع}\index{مثلث متساوي الأضلاع} حين
        $AB = BC = CA$ (ثلاثة أضلاع متساوية)؛
  \item و\emph{قائم الزاوية}\index{مثلث قائم الزاوية} في $A$ حين تكون الزاوية
        $\widehat{BAC}$ قائمة. والضلع $[BC]$ المقابل \omterm{def:g3:shapes:rightangle}{للزاوية
        القائمة} يُسمّى \emph{الوتر}.
\end{itemize}
\end{definition}

\begin{method}[إنشاء مثلث من أضلاعه الثلاثة]\label{met:g6:shapes:construct}
لإنشاء المثلث $ABC$ حيث $AB = 5$ سم، و $AC = 4$ سم، و $BC = 3$ سم:
\begin{enumerate}
  \item اُرسم \omterm{def:g6:lines:objects}{القطعة} $[AB]$ طولها $5$ سم بالمسطرة؛
  \item اُرسم قوسًا من الدائرة التي مركزها $A$ ونصف قطرها $4$ سم
        (كل النقط التي تبعد $4$ سم عن $A$)؛
  \item اُرسم قوسًا من الدائرة التي مركزها $B$ ونصف قطرها $3$ سم؛
  \item والنقطة $C$ هي تقاطع القوسين؛ صِلها بالنقطتين $A$
        و $B$.
\end{enumerate}
\end{method}

\section{الرباعيات}

\begin{definition}[الرباعيات الخاصّة]\label{def:g6:shapes:quadrilaterals}
\omterm{def:g4:geometry:polygon}{للرباعي} أربعة \omterm{def:g5:solids:def}{رؤوس} وأربعة أضلاع. وهو:
\begin{itemize}
  \item \emph{مستطيل}\index{مستطيل} حين تكون زواياه الأربع
        قائمة؛
  \item و\emph{معيّن}\index{معين} حين تكون أضلاعه الأربعة متساوية
        الطول؛
  \item و\emph{مربع}\index{مربع} حين يكون الأمرين معًا: أربع \omterm{def:g3:shapes:rightangle}{زوايا قائمة}
        \emph{و} أربعة أضلاع متساوية.
\end{itemize}
\end{definition}

\begin{proposition}[خاصّيات القطرين]\label{prop:g6:shapes:diagonals}
في هذه الرباعيات، يتقاطع القطران في منتصفهما
المشترك، وزيادة على ذلك:
\begin{itemize}
  \item في المستطيل، للقطرين \emph{الطول نفسه}؛
  \item وفي \omterm{def:g6:shapes:quadrilaterals}{المعيّن}، القطران \emph{متعامدان}؛
  \item وفي المربع، الأمران معًا.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}
المربع مستطيل بالضرورة (فزواياه الأربع قائمة)، وهو
\omterm{def:g6:shapes:quadrilaterals}{معيّن} بالضرورة (أضلاعه الأربعة متساوية). فكلمة «مستطيل» لا
تعني «ليس مربعًا»! ففي الرياضيات، ينتمي الشكل الأخصّ
إلى العائلة الأعمّ.
\end{remark}

\begin{example}[التعرّف على شكل من خاصّياته]\label{ex:g6:shapes:recognize}
\omterm{def:g4:geometry:polygon}{رباعي} أضلاعه الأربعة طولها $4$ سم وفيه \omterm{def:g3:shapes:rightangle}{زاوية قائمة}.
فما هو؟ تساوي الأضلاع الأربعة يجعله معيّنًا. وفي \omterm{def:g6:shapes:quadrilaterals}{المعيّن}، الزاويتان
المتقابلتان متساويتان \omterm{def:g1:addition:def}{ومجموع} الزوايا الأربع $360^\circ$؛ \omterm{def:g3:shapes:rightangle}{فزاوية قائمة}
واحدة تفرض إذن أن تكون الأربع قائمة (وسيُبرهن على ذلك مع
متوازيات الأضلاع في \cref{ch:g7:central}). إذن الشكل مربع.
\end{example}

\begin{method}[برامج الإنشاء]\label{met:g6:shapes:program}
\emph{برنامج الإنشاء} قائمة تعليمات يستطيع أيّ أحد أن
يتبعها ليعيد رسم شكل بالضبط. اُكتب تعليمة واحدة في كل سطر،
وسمِّ كل نقطة حين تُنشأ، ولا تستعمل إلا: «اُرسم \omterm{def:g6:lines:objects}{القطعة}
/ \omterm{def:g6:lines:objects}{المستقيم} / الدائرة\dots»، و«ضع النقطة حيث يتقاطع \dots». واِختبر
برنامجك باتّباعه حرفيًّا، كلمة كلمة.
\end{method}

\begin{example}\label{ex:g6:shapes:program}
برنامج لمربع $ABCD$ طول ضلعه $4$ سم:
\begin{enumerate}
  \item اُرسم \omterm{def:g6:lines:objects}{القطعة} $[AB]$ حيث $AB = 4$ سم.
  \item اُرسم العمودي على $(AB)$ في $A$؛ وضع عليه $D$ على بُعد
        $4$ سم من $A$ (اِختر جهة من \omterm{def:g6:lines:objects}{المستقيم} والزمها).
  \item اُرسم العمودي على $(AB)$ في $B$؛ وضع عليه $C$ على بُعد
        $4$ سم من $B$، في الجهة نفسها التي فيها $D$.
  \item صِل $C$ و $D$.
\end{enumerate}
وتحقّق بالمسطرة: فالمسافة $DC$ يجب أن تساوي $4$ سم أيضًا.
\end{example}

اِبنِ مثلثًا $ABC$ حيث $AB = 6$ سم، و $AC = 5$ سم، و

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:2}
اِبنِ مثلثًا \omterm{def:g6:shapes:triangles}{متساوي الساقين} $DEF$ حيث $DE = DF = 5$ سم و
$EF = 3$ سم. فأيّ زاويتين من هذا المثلث تبدوان متساويتين؟ قِسهما
للتحقّق.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:3}
اِبنِ مثلثًا \omterm{def:g6:shapes:triangles}{متساوي الأضلاع} طول ضلعه $4$ سم بالمسطرة والفرجار
وحدهما. وقِس إحدى زواياه.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:4}
صنّف كل مثلث انطلاقًا من معطياته (\omterm{def:g6:shapes:triangles}{متساوي الساقين}، \omterm{def:g6:shapes:triangles}{متساوي الأضلاع}،
\omterm{def:g6:shapes:triangles}{قائم الزاوية} --- وقد تصحّ أجوبة عدّة):
(أ) أضلاعه $7$ و $7$ و $7$؛ \
(ب) أضلاعه $5$ و $5$ و $8$؛ \
(ج) زواياه $90^\circ$ و $45^\circ$ و $45^\circ$؛ \
(د) زواياه $60^\circ$ و $60^\circ$ و $60^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:5}
في \omterm{def:g6:shapes:triangles}{المثلث القائم الزاوية}، أيّ ضلع هو الوتر؟ اُرسم مثلثًا
$MNP$ \omterm{def:g6:shapes:triangles}{قائم الزاوية} في $N$، وسمِّ وتره. وأيّ رأس لا
يلمسه الوتر أبدًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:6}
صواب أم خطأ؟ بَرِّر كل جواب بجملة واحدة.
\begin{enumerate}
  \item كل مربع مستطيل.
  \item كل مستطيل مربع.
  \item كل مربع \omterm{def:g6:shapes:quadrilaterals}{معيّن}.
  \item كل \omterm{def:g6:shapes:quadrilaterals}{معيّن} فيه \omterm{def:g3:shapes:rightangle}{زاوية قائمة} مربع.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:7}
اِبنِ مستطيلًا $ABCD$ حيث $AB = 5$ سم و $BC = 3$ سم. وقِس
قطريه: فماذا تلاحظ؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:8}
اِبنِ معيّنًا قطراه $6$ سم و $4$ سم. (اِستعمل
\cref{prop:g6:shapes:diagonals}: اُرسم القطرين أولًا ---
\omterm{def:g6:lines:perp}{متعامدين}، متقاطعين في منتصفيهما --- ثم صِل الأطراف
الأربعة.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:9}
اُكتب برنامج إنشاء \omterm{def:g6:shapes:triangles}{لمثلث متساوي الساقين} قاعدته
$4$ سم وساقاه $6$ سم، على نمط
\cref{ex:g6:shapes:program}. وتبادله مع زميل: أيستطيع أن يتبعه
دون أن يرى شكلك؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:10}
اُرسم دائرة مركزها $O$ وأحد أقطارها $[AB]$. وضع نقطة
$M$ كيفما كانت على الدائرة واُرسم المثلث $AMB$. وقِس الزاوية
$\widehat{AMB}$. ثم حرّك $M$ وقِس من جديد. فماذا تخمّن؟
(وهذه الواقعة المدهشة يُبرهن عليها في \cref{ch:g8:cosine}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:11}
\omterm{def:g4:geometry:polygon}{رباعي} قطراه متساويا الطول ويتقاطعان في
منتصفهما المشترك. فحدّد أيّ شكل يجب أن يكون، مستعملًا
\cref{prop:g6:shapes:diagonals}: مستطيل أم \omterm{def:g6:shapes:quadrilaterals}{معيّن} أم مربع؟
(اِنتبه: خاصّية واحدة فقط أُعطيت.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:shapes:12}
على شبكة منقّطة $4 \times 4$، كم مربعًا مختلفًا تستطيع أن ترسم

اُرسم مربعًا مائلًا واحدًا على الأقل واِشرح لماذا تتساوى أضلاعه.
\end{exercise}

\section{مسألة: ما يعرفه القطران}

\begin{problem}[{مسألة نهاية الأسبوع --- إعادة بناء كل رباعي
خاصّ من قطريه وحدهما، وعدّ أقطار
أيّ مضلع}]
\label{pb:g6:shapes:1}
يقرأ \cref{prop:g6:shapes:diagonals} شكلًا ويخبرك عن
قطريه. وتلعب هذه المسألة اللعبة \emph{بالمقلوب}: إذا أُعطيت
\omterm{def:g6:lines:objects}{قطعتان} متقاطعتان لا غير --- وهما القطران المرتقبان --- فأيّ شكل
يظهر حين تصل أطرافهما الأربعة؟ ستبني «قاموس
أقطار» كاملًا، وتفهم \emph{لماذا} ينجح إنشاء \omterm{def:g6:shapes:quadrilaterals}{المعيّن}
في \cref{exo:g6:shapes:8}، وتلتقي شكلًا ليس له اسم
بعد، وتختم بعدّ أقطار المضلعات
ذات أضلاع أكثر بكثير من أربعة.

\medskip
\noindent\textbf{الجزء الأول --- قاموس الأقطار.}
في كل سؤال، اُرسم \omterm{def:g6:lines:objects}{القطعتين} كما هو موصوف، وصِل أطرافهما
الأربعة بالترتيب، وعيّن \omterm{def:g4:geometry:polygon}{الرباعي} الذي تحصل عليه.
\begin{enumerate}
  \item \omterm{def:g6:lines:objects}{قطعتان} طول كل منهما $6$ سم، متعامدتان، تتقاطعان في
        منتصفهما المشترك. قِس الأضلاع الأربعة والزوايا
        الأربع للشكل الناتج: فما هو؟
  \item \omterm{def:g6:lines:objects}{قطعتان} طولهما $8$ سم و $5$ سم، متعامدتان،
        تتقاطعان في منتصفهما المشترك. فأيّ شكل يظهر
        (\cref{exo:g6:shapes:8})؟
  \item \omterm{def:g6:lines:objects}{قطعتان} طول كل منهما $7$ سم --- متساويتان لكن \emph{غير}
        متعامدتين --- تتقاطعان في منتصفهما المشترك.
        قِس الأضلاع والزوايا: فماذا تخمّن أن يكون
        الشكل؟
  \item \omterm{def:g6:lines:objects}{قطعتان} طولهما $8$ سم و $5$ سم، غير متعامدتين،
        تتقاطعان في منتصفهما المشترك. والشكل الذي تحصل عليه
        ليس أيًّا من الثلاثة في
        \cref{def:g6:shapes:quadrilaterals} --- فصِف ما
        تراه (أيّ الأضلاع تبدو متساوية؟ ومتوازية؟). وهذا
        \omterm{def:g4:geometry:polygon}{الرباعي} يأخذ اسمه في \cref{ch:g7:central}.
  \item لخّص الأسئلة 1--4 في قاموس ذي أربعة
        مداخل: القطران \emph{متساويان / متعامدان / الأمران معًا /
        لا هذا ولا ذاك} (مع التقاطع دائمًا في المنتصفين)
        $\rightarrow$ اسم الشكل (أو وصفه).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- لماذا تنجح الإنشاءات.}
\begin{enumerate}[resume]
  \item في السؤال 2، يقسم القطران الشكل إلى أربعة
        مثلثات قائمة الزاوية حول نقطة التقاطع. فما طولا
        ضلعَي القائمة في كل من هذه المثلثات؟ اِشرح لماذا
        تكون المثلثات الأربعة نسخًا مطابقة بعضها لبعض،
        واستنتج لماذا تتساوى أضلاع الشكل الأربعة --- وهو
        سرّ \cref{exo:g6:shapes:8}.
  \item اِستنتج اِختبارًا بالفرجار وحده: بعد بناء أيّ شكل
        من قطريه، كيف تتحقّق بالفرجار
        وحده (دون قياس بالمسطرة) من أن الأضلاع الأربعة
        متساوية؟
  \item اُكتب برنامج إنشاء
        (\cref{met:g6:shapes:program}) لمربع طول
        \emph{قطريه} $6$ سم --- لا طول أضلاعه.
  \item في مربع السؤال 8، هل الضلع أطول أم
        أقصر من $3$ سم --- أي \omterm{def:g3:shapes:shapes}{نصف القطر}؟ قِس
        لتعرف. ويمكن \emph{البرهنة} على نصف الجواب
        من الآن: فالانتقال من ركن إلى الذي يليه عبر
        المركز التفاف طوله $3 + 3$ سم، والطريق \omterm{def:g6:lines:objects}{المستقيم}
        أقصر دائمًا من الالتفاف --- إذن الضلع أقصر
        من $6$ سم. (وأما البرهنة على أنه يفوق $3$ سم أيضًا،
        وحسابه بالضبط، فشغل
        \cref{ch:g8:pythagoras}.)
  \item مدخل جديد للقاموس: اُرسم \omterm{def:g6:lines:objects}{قطعة} $[AC]$ طولها
        $8$ سم؛ وعلّم عليها النقطة $H$ على بُعد $3$ سم من $A$؛
        واُرسم مارًّا بالنقطة $H$ \omterm{def:g6:lines:objects}{القطعة} العمودية $[BD]$ طولها
        $5$ سم بحيث تكون $H$ منتصف \emph{هذه} \omterm{def:g6:lines:objects}{القطعة}. وصِل $ABCD$. وهذا
        الشكل \emph{طائرة ورقية}\index{طائرة ورقية} (أحد القطرين ينصّف
        الآخر، في \omterm{def:g3:shapes:rightangle}{زاوية قائمة}). وباستعمال مثلثات الأركان كما
        في السؤال 6، جِد أيّ أضلاع \omterm{pb:g6:shapes:1}{الطائرة الورقية} تساوي
        أيًّا.
\end{enumerate}

\noindent\textbf{الجزء الثالث --- عدّ الأقطار.}
\emph{\omterm{def:g4:geometry:circle}{قطر}} \omterm{def:g4:geometry:polygon}{المضلع} يصل بين رأسين غير
متجاورين.
\begin{enumerate}[resume]
  \item اُرسم رباعيًا، وخماسيًا (خمسة \omterm{def:g5:solids:def}{رؤوس})،
        وسداسيًا (ستة \omterm{def:g5:solids:def}{رؤوس})، واُرسم كل أقطارها.
        وعُدَّها: فكم في كل حالة؟
  \item العدّ دون رسم، من أجل \omterm{def:g4:geometry:polygon}{السداسي}: من رأس
        واحد، كم قطرًا يخرج؟ (بكم رأس من \omterm{def:g5:solids:def}{الرؤوس}
        الستة \emph{لا} يمكن وصله \omterm{def:g4:geometry:circle}{بقطر}؟)
        وضربُ ذلك في \omterm{def:g5:solids:def}{الرؤوس} الستة يعدّ كل \omterm{def:g4:geometry:circle}{قطر}
        مرتين بالضبط --- لماذا؟ اِستعد عدّك في السؤال
        11. (وقد ظهر هذا العدّ المزدوج نفسه في مصافحات
        \cref{pb:g6:wholes:1}.)
  \item اِستعمل طريقة السؤال 12 لعدّ أقطار
        \omterm{def:g4:geometry:polygon}{مضلع} ذي $10$ \omterm{def:g5:solids:def}{رؤوس}، ثم ذي $20$ رأسًا.
  \item \omterm{def:g4:geometry:polygon}{لمضلع} $44$ قطرًا بالضبط. فكم رأسًا
        له؟ (جرّب طريقة السؤال 13 على بضعة
        مرشّحين.)
  \item تحقّق من طريقتك على أصغر المضلعات جميعًا: كم
        قطرًا للمثلث، وماذا تجيب
        طريقة العدّ في السؤال 12؟ واختم بجملة
        واحدة عمّا تشترك فيه المصافحات والأقطار.
\end{enumerate}
\end{problem}
