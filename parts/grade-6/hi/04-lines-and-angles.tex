\chapter{रेखाएँ, वृत्त और कोण}\label{ch:g6:lines}

ज्यामिति मापपट्टी, \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका और परकार से शुरू होती है। यह
अध्याय शब्दावली पक्की करता है --- बिंदु, \omterm{def:g6:lines:objects}{रेखाखंड}, किरणें,
रेखाएँ, वृत्त --- रेखाओं की दो ख़ास स्थितियाँ (\omterm{def:g6:lines:perp}{समांतर} और \omterm{def:g6:lines:perp}{लंब})
सामने लाता है, और कोण नापना तथा बनाना सिखाता है।

\section{बिंदु, रेखाखंड, रेखाएँ}

\begin{definition}[बुनियादी वस्तुएँ]\label{def:g6:lines:objects}
दो अलग-अलग बिंदुओं $A$ और $B$ से होकर जाते हैं:
\begin{itemize}
  \item \emph{रेखाखंड}\index{रेखाखंड} $[AB]$: रेखा का वह भाग जो
        $A$ और $B$ के बीच है (उसकी एक लंबाई होती है, जिसे $AB$ लिखते हैं);
  \item \emph{किरण} $[AB)$: $A$ से शुरू होती है, $B$ से होकर जाती
        है और हमेशा आगे बढ़ती रहती है;
  \item \emph{रेखा}\index{रेखा} $(AB)$: दोनों ओर अनंत तक फैली
        रहती है। दो बिंदु ठीक एक ही रेखा तय करते हैं।
\end{itemize}
एक ही रेखा पर पड़े बिंदु \emph{संरेख} कहलाते हैं।
\end{definition}

\section{समांतर और लंब रेखाएँ}

\begin{definition}[लंब, समांतर]\label{def:g6:lines:perp}
दो रेखाएँ \emph{लंब}\index{लंब रेखाएँ} तब होती हैं जब वे \omterm{def:g3:shapes:rightangle}{समकोण}
पर काटती हैं; हम $d_1 \perp d_2$ लिखते हैं और \omterm{def:g3:shapes:rightangle}{समकोण} को एक छोटे
वर्ग से अंकित करते हैं। दो रेखाएँ
\emph{समांतर}\index{समांतर रेखाएँ} तब होती हैं जब वे कितनी भी
आगे बढ़ाई जाएँ, कभी नहीं मिलतीं; हम $d_1 \parallel d_2$ लिखते हैं।
\end{definition}

\omterm{def:g6:lines:perp}{समांतर रेखाएँ}: वही दिशा, कोई कटान-बिंदु नहीं।}
\end{omfigure}

\begin{proposition}[दो काम के तथ्य]\label{prop:g6:lines:facts}
\begin{enumerate}
  \item दो रेखाएँ किसी तीसरी रेखा पर \omterm{def:g6:lines:perp}{लंब} हों, तो वे आपस में
        \omterm{def:g6:lines:perp}{समांतर} होती हैं।
  \item दो रेखाएँ \omterm{def:g6:lines:perp}{समांतर} हों, तो उनमें से एक पर \omterm{def:g6:lines:perp}{लंब} हर रेखा
        दूसरी पर भी \omterm{def:g6:lines:perp}{लंब} होती है।
\end{enumerate}
\end{proposition}
\admitted

\begin{method}[समकोण पट्टिका से बनाना]\label{met:g6:lines:setsquare}
किसी रेखा $d$ पर बिंदु $P$ से होकर \omterm{def:g6:lines:perp}{लंब} खींचने के लिए:
\begin{enumerate}
  \item \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका के \omterm{def:g3:shapes:rightangle}{समकोण} का एक \omterm{def:g5:solids:def}{किनारा} $d$ पर रखो;
  \item पट्टिका को $d$ के साथ तब तक सरकाओ जब तक उसका दूसरा
        \omterm{def:g5:solids:def}{किनारा} $P$ तक न पहुँचे;
  \item उस \omterm{def:g5:solids:def}{किनारे} के साथ रेखा खींचो, और \omterm{def:g3:shapes:rightangle}{समकोण} अंकित करो।
\end{enumerate}
$P$ से होकर \omterm{def:g6:lines:perp}{समांतर} रेखा के लिए: पहले $d$ पर \omterm{def:g6:lines:perp}{लंब} खींचो, फिर
\emph{उस} रेखा पर $P$ से होकर \omterm{def:g6:lines:perp}{लंब}
(\cref{prop:g6:lines:facts})।
\end{method}

\section{वृत्त}

\begin{definition}[वृत्त]\label{def:g6:lines:circle}
केंद्र $O$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $r$ वाला \emph{वृत्त}\index{वृत्त} उन
सारे बिंदुओं से बनता है जो $O$ से ठीक $r$ दूरी पर हैं। केंद्र से
वृत्त तक का \omterm{def:g6:lines:objects}{रेखाखंड} \emph{\omterm{def:g3:shapes:shapes}{त्रिज्या}} है; वृत्त के दो बिंदुओं को
केंद्र से होकर जोड़ने वाला \omterm{def:g6:lines:objects}{रेखाखंड} \emph{\omterm{def:g4:geometry:circle}{व्यास}} है --- उसकी लंबाई
$2r$ है; वृत्त के दो बिंदुओं को जोड़ने वाला \omterm{def:g6:lines:objects}{रेखाखंड} \emph{जीवा} है।
\end{definition}

\begin{example}\label{ex:g6:lines:circle}
``केंद्र $O$ वाला वह वृत्त बनाओ जो $A$ से होकर जाए'': परकार को
$O$ से $A$ तक खोलो --- \omterm{def:g3:shapes:shapes}{त्रिज्या} दूरी $OA$ है --- और घुमा दो। इस
वृत्त का हर बिंदु $O$ से उतनी ही दूरी पर है जितनी $A$।
\end{example}

\section{कोण}

\begin{definition}[कोण]\label{def:g6:lines:angle}
एक ही शुरुआती बिंदु $A$ वाली दो किरणें $[AB)$ और $[AC)$ मिलकर
\emph{कोण}\index{कोण} $\widehat{BAC}$ बनाती हैं; बिंदु $A$ उसका
\emph{\omterm{def:g5:solids:def}{शीर्ष}} है (हमेशा बीच वाला अक्षर!)। कोण
\emph{डिग्री} ($^\circ$) में नापे जाते हैं, पूरे घुमाव के लिए
$0^\circ$ से $360^\circ$ तक। कोई कोण होता है:
\begin{itemize}
  \item \emph{\omterm{def:g3:shapes:rightangle}{समकोण}}, अगर वह $90^\circ$ का हो;
  \item \emph{न्यून}, अगर वह $90^\circ$ से कम का हो;
  \item \emph{अधिक}, अगर वह $90^\circ$ और $180^\circ$ के बीच
        का हो;
  \item \emph{सरल}, अगर वह $180^\circ$ का हो (दोनों किरणें
        मिलकर एक रेखा बना देती हैं)।
\end{itemize}
\end{definition}

\begin{method}[चाँदे से कोण नापना]\label{met:g6:lines:protractor}
\begin{enumerate}
  \item चाँदे का केंद्र ठीक कोण के \omterm{def:g5:solids:def}{शीर्ष} पर रखो;
  \item उसकी $0^\circ$ वाली रेखा कोण की एक भुजा पर मिलाओ;
  \item दूसरी भुजा जिस अंकन को काटती है वही पढ़ो --- उसी पैमाने
        से जो मिलाई गई भुजा पर $0$ से शुरू होता है;
  \item आँख से मोटी जाँच करो: न्यून कोण $90^\circ$ से कम पढ़ा
        जाना चाहिए, अधिक कोण उससे ज़्यादा।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:lines:estimate}
नापने से पहले अंदाज़ा लगाओ! काग़ज़ की शीट के कोने से ज़रा ज़्यादा
खुला कोण $90^\circ$ से थोड़ा ऊपर होता है; \omterm{def:g3:shapes:rightangle}{समकोण} का \omterm{def:g3:division:half}{आधा}
$45^\circ$ है; \omterm{def:g3:shapes:rightangle}{समकोण} की तिहाई $30^\circ$। किसी न्यून दिखने वाले
कोण के लिए तुम्हारा चाँदा $150^\circ$ बताए, तो तुमने ग़लत पैमाना
पढ़ा: सही नाप $180^\circ - 150^\circ = 30^\circ$ है।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g6:lines:1}
तीन ऐसे बिंदु $A$, $B$, $C$ बनाओ जो संरेख न हों। अलग-अलग रंगों
से बनाओ: \omterm{def:g6:lines:objects}{रेखाखंड} $[AB]$, किरण $[CA)$, रेखा $(BC)$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:2}
सही या ग़लत? ``$[AB]$ और $[BA]$ एक ही \omterm{def:g6:lines:objects}{रेखाखंड} हैं।''
``$[AB)$ और $[BA)$ एक ही किरण हैं।'' ``$(AB)$ और $(BA)$ एक ही
रेखा हैं।'' हर उत्तर समझाओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:3}
एक रेखा $d$ और $d$ से बाहर एक बिंदु $P$ बनाओ। \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका से
बनाओ: $P$ से होकर $d$ पर \omterm{def:g6:lines:perp}{लंब}, फिर $P$ से होकर $d$ के \omterm{def:g6:lines:perp}{समांतर}
रेखा। अपने कदम बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:4}
रेखाएँ $a$ और $b$ दोनों किसी रेखा $c$ पर \omterm{def:g6:lines:perp}{लंब} हैं, और एक चौथी
रेखा $e$, $a$ पर \omterm{def:g6:lines:perp}{लंब} है। $a$ और $b$ के बारे में क्या कह सकते हो?
$e$ और $c$ के बारे में? \cref{prop:g6:lines:facts} से पक्का करो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:5}
केंद्र $O$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $3$ cm का वृत्त बनाओ। वृत्त पर एक बिंदु
$M$, भीतर एक बिंदु $N$, बाहर एक बिंदु $P$ रखो। दूरियों $OM$,
$ON$, $OP$ के बारे में $3$ cm की तुलना में क्या कह सकते हो?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:6}
एक वृत्त का \omterm{def:g4:geometry:circle}{व्यास} $9$ cm है। उसकी \omterm{def:g3:shapes:shapes}{त्रिज्या} क्या है? दूसरे की
\omterm{def:g3:shapes:shapes}{त्रिज्या} $2.6$ cm है: उसका \omterm{def:g4:geometry:circle}{व्यास} क्या है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:7}
अंकित कोण को तीन अक्षरों में नाम दो, फिर उसे छाँटो (न्यून,
\omterm{def:g3:shapes:rightangle}{समकोण}, अधिक, सरल): \omterm{def:g5:solids:def}{शीर्ष} $R$ पर $S$ और $T$ की ओर जाती किरणों के
बीच $72^\circ$ का कोण; \omterm{def:g5:solids:def}{शीर्ष} $B$ पर $A$ और $C$ की ओर जाती किरणों
के बीच $148^\circ$ का कोण; \omterm{def:g5:solids:def}{शीर्ष} $O$ पर $M$ और $N$ की ओर जाती
किरणों के बीच $90^\circ$ का कोण।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:8}
अपने बनाए किसी त्रिभुज के तीनों कोणों का पहले अंदाज़ा लगाओ, फिर
चाँदे से नापो। तीनों नाप जोड़ो: क्या मिलता है? (अपना उत्तर
\cref{ch:g7:triangles} के लिए सँभालकर रखो।)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:9}
चाँदे से $65^\circ$ का कोण $\widehat{xOy}$ बनाओ, फिर $130^\circ$
का कोण। बिना चाँदे के दूसरा कोण पहले से कैसे पा सकते हो?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:lines:10}
नक़्शे पर दो गाँव $A$ और $B$ बने हैं। वे बिंदु कहाँ हैं जो $A$ से
$3$ cm \emph{और} $B$ से $2$ cm दूर हैं? एक चित्र बनाकर दिखाओ कि
ऐसे बिंदु कितने हो सकते हैं (दूरी $AB$ के अनुसार $0$, $1$ या $2$)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:lines:11}
एक घड़ी में 3 बजे हैं: दोनों सुइयों के बीच कोण क्या है? यही सवाल
5 बजे के लिए, और 6 बजे के लिए। (पूरा घुमाव $12$ घंटों के लिए
$360^\circ$ है।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:lines:12}
$6$ cm का \omterm{def:g6:lines:objects}{रेखाखंड} $[AB]$ बनाओ। केवल परकार से वह बिंदु $C$ बनाओ
जिसके लिए $AC = BC = 6$ cm हो, और कोण $\widehat{CAB}$ नापो।
तुमने कौन-सा त्रिभुज बनाया, और उसके कोणों के बारे में तुम्हारा
अनुमान क्या है?
\end{exercise}

\section{समस्या: घड़ी के चेहरे की ज्यामिति}

\begin{problem}[{सप्ताहांत समस्या --- कोण घुमाव की भिन्नों के
रूप में: उन्हें घड़ी पर पढ़ना, वे क्षण खोजना जब सुइयाँ मिलती
हैं, और एक दिन को वृत्त-आलेख में काटना}]
\label{pb:g6:lines:1}
घड़ी एक ऐसा चाँदा है जो समय बताता है: उसका चेहरा $360^\circ$ का
पूरा घुमाव है, जिसे बारह घंटा-निशान बारह बराबर कोणों में काट
देते हैं। \Cref{exo:g6:lines:11} में 3 बजे और 6 बजे सुइयाँ नापी
गई थीं; यह समस्या पूरा सिद्धांत बनाती है --- उन समयों समेत जब
\emph{कोई भी} सुई किसी निशान पर नहीं होती --- एक ऐसे सवाल का
उत्तर देती है जो कम ही बड़े ठीक बता पाते हैं (``दोनों सुइयाँ
ठीक एक के ऊपर एक कितनी बार आती हैं?''), और अंत में पूरे दिन को
वृत्त-आलेख में काट देती है।

\medskip
\noindent\textbf{भाग I --- घुमाव की भिन्नें।}
\begin{enumerate}
  \item पूरा घुमाव $360^\circ$ का होता है। \omterm{def:g3:division:half}{आधा} घुमाव, \omterm{def:g3:division:half}{चौथाई}
        घुमाव और बारहवाँ घुमाव कितनी डिग्री का है?
  \item बारह घंटा-निशान घड़ी के चेहरे को केंद्र पर बारह बराबर
        कोणों में काटते हैं। दो पड़ोसी निशानों के बीच कितनी
        डिग्री? निशान गिनकर \cref{exo:g6:lines:11} का 3 बजे वाला
        उत्तर फिर से पाओ।
  \item 1 बजे, 4 बजे और 7 बजे सुइयों के बीच कोण बताओ। (7 बजे
        सुइयाँ चेहरे को दो कोणों में बाँटती हैं; ``सुइयों के बीच
        का कोण'' हमेशा \emph{छोटे} वाले को कहते हैं।)
  \item मिनट की सुई $60$ मिनट में पूरा घुमाव लेती है। वह हर मिनट
        कितनी डिग्री बुहारती है?
  \item घंटे की सुई एक निशान से अगले तक --- $30^\circ$ ---
        $60$ मिनट में जाती है। \emph{वह} हर मिनट कितनी डिग्री
        बुहारती है? (एक \omterm{def:g6:decimals:places}{दशमलव संख्या},
        \cref{ch:g6:decimals}।)
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- वे समय जब कुछ भी किसी निशान पर नहीं
होता।}
\begin{enumerate}[resume]
  \item 3:30 पर मिनट की सुई $6$ पर है। घंटे की सुई ठीक कहाँ है?
        $12$ से (घड़ी की दिशा में नापते हुए) हर सुई का कोण
        निकालो, और उससे सुइयों के बीच का कोण निकालो।
  \item 6:30 पर कई लोग सोचते हैं कि सुइयाँ एक के ऊपर एक हैं।
        पहले अंदाज़ा लगाओ, फिर प्रश्न 6 की तरह कोण निकालो। कौन
        सही था?
  \item 9:15 पर सुइयों के बीच का कोण निकालो।
  \item समझाओ कि 1:00 और 1:10 के बीच कहीं दोनों सुइयाँ ठीक एक के
        ऊपर एक क्यों होनी ही चाहिए: 1:00 पर मिनट की सुई घंटे की
        सुई के मुक़ाबले कहाँ है, और 1:10 पर कहाँ? कौन-सी सुई तेज़
        चलती है, और इससे बात कैसे तय हो जाती है?
  \item बारह घंटों में दोनों सुइयाँ ठीक एक के ऊपर एक कितनी बार
        आती हैं? (लगातार घंटों के बीच हर टुकड़े में एक मुलाक़ात
        होती है --- एक अपवाद के साथ: 11 और 12 के बीच क्या होता
        है? मुलाक़ातों के लगभग समय गिनाओ और गिनती करो।) पूरे दिन
        में कितनी बार?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- किसी बिंदु के चारों ओर कोण:
वृत्त-आलेख।}
\begin{enumerate}[resume]
  \item घड़ी के चेहरे के बारहों खंड मिलकर चेहरा भर देते हैं:
        उनके कोणों का योग $360^\circ$ है। समझाओ कि यह
        \emph{किसी भी} ऐसे कोण-संग्रह के लिए क्यों सही है जिनका
        \omterm{def:g5:solids:def}{शीर्ष} साझा हो और जो उसके चारों ओर पूरा घुमाव भर दें, न
        कहीं ओवरलैप हो, न कहीं ख़ाली जगह।
  \item ज़ोई अपने $24$ घंटों का दिन लिखती है: नींद $9$ घंटे,
        विद्यालय $6$ घंटे, खेल $3$ घंटे, भोजन $2$ घंटे, बाक़ी सब
        $4$ घंटे। उसे एक \emph{वृत्त-आलेख} चाहिए: एक चक्र जिसमें
        हर काम को एक खंड मिले, और कोण समयों के समानुपाती हों।
        हर काम दिन की कौन-सी \omterm{def:g6:fractions:def}{भिन्न} है, और उसके खंड को कितनी
        डिग्री मिलती हैं? जाँचो कि पाँचों कोण मिलकर $360^\circ$
        बनते हैं।
  \item कदम-दर-कदम बताओ कि ज़ोई का वृत्त-आलेख परकार और चाँदे से
        कैसे बनाया जाए (\cref{met:g6:lines:protractor}): पहली
        \omterm{def:g3:shapes:shapes}{त्रिज्या} कहाँ जाती है, और हर नया खंड वहीं से कैसे शुरू
        होता है जहाँ पिछला ख़त्म हुआ।
  \item किसी दूसरे दिन के वृत्त-आलेख में एक खंड ठीक $90^\circ$
        का है। वह चक्र की कौन-सी \omterm{def:g6:fractions:def}{भिन्न} है, और कितने घंटे दिखाता
        है?
  \item आख़िरी सवाल, फिर घड़ी पर: $24$ घंटों में घंटे की सुई
        कितने पूरे घुमाव लेती है? मिनट की सुई? सेकंड की सुई?
        (इनमें से एक उत्तर हज़ार से ऊपर है।)
\end{enumerate}
\end{problem}
