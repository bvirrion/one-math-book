\chapter{त्रिभुज और चतुर्भुज}\label{ch:g6:shapes}

त्रिभुज और \omterm{def:g4:geometry:polygon}{चतुर्भुज} ज्यामिति के बुनियादी आकार हैं। इस अध्याय का
लक्ष्य है उनके ख़ास परिवारों को --- \omterm{def:g6:shapes:triangles}{समद्विबाहु}, \omterm{def:g6:shapes:triangles}{समबाहु} और \omterm{def:g6:shapes:triangles}{समकोण
त्रिभुज}; आयत, \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज}, वर्ग --- उनके \emph{परिभाषक गुणों} से
पहचानना, और उन्हें मापपट्टी तथा परकार से ठीक-ठीक बनाना।

\section{त्रिभुज}

\begin{definition}[ख़ास त्रिभुज]\label{def:g6:shapes:triangles}
त्रिभुज $ABC$ के तीन \omterm{def:g5:solids:def}{शीर्ष} और तीन भुजाएँ होती हैं। वह होता है:
\begin{itemize}
  \item $A$ पर \emph{समद्विबाहु}\index{समद्विबाहु त्रिभुज}, जब
        $AB = AC$ हो (दो बराबर भुजाएँ);
  \item \emph{समबाहु}\index{समबाहु त्रिभुज}, जब
        $AB = BC = CA$ हो (तीन बराबर भुजाएँ);
  \item $A$ पर \emph{समकोण}\index{समकोण त्रिभुज}, जब कोण
        $\widehat{BAC}$ समकोण हो। समकोण के सामने वाली भुजा
        $[BC]$ \emph{कर्ण} कहलाती है।
\end{itemize}
\end{definition}

\begin{method}[तीन भुजाओं से त्रिभुज बनाना]\label{met:g6:shapes:construct}
$AB = 5$ cm, $AC = 4$ cm, $BC = 3$ cm वाला $ABC$ बनाने के लिए:
\begin{enumerate}
  \item मापपट्टी से $5$ cm का \omterm{def:g6:lines:objects}{रेखाखंड} $[AB]$ खींचो;
  \item केंद्र $A$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $4$ cm वाले वृत्त का चाप खींचो
        ($A$ से $4$ cm दूर के सारे बिंदु);
  \item केंद्र $B$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $3$ cm वाले वृत्त का चाप खींचो;
  \item बिंदु $C$ वहीं है जहाँ दोनों चाप काटते हैं; उसे $A$ और
        $B$ से जोड़ दो।
\end{enumerate}
\end{method}

\section{चतुर्भुज}

\begin{definition}[ख़ास चतुर्भुज]\label{def:g6:shapes:quadrilaterals}
\omterm{def:g4:geometry:polygon}{चतुर्भुज} के चार \omterm{def:g5:solids:def}{शीर्ष} और चार भुजाएँ होती हैं। वह होता है:
\begin{itemize}
  \item \emph{आयत}\index{आयत}, जब उसके चारों कोण
        \omterm{def:g6:shapes:triangles}{समकोण} हों;
  \item \emph{समचतुर्भुज}\index{समचतुर्भुज}, जब उसकी चारों भुजाओं की
        लंबाई एक हो;
  \item \emph{वर्ग}\index{वर्ग}, जब दोनों हों: चार \omterm{def:g6:shapes:triangles}{समकोण}
        \emph{और} चार बराबर भुजाएँ।
\end{itemize}
\end{definition}

\begin{proposition}[विकर्णों के गुण]\label{prop:g6:shapes:diagonals}
इन \omterm{def:g4:geometry:polygon}{चतुर्भुजों} में दोनों विकर्ण एक-दूसरे को उनके साझा मध्यबिंदु पर
काटते हैं, और इसके अलावा:
\begin{itemize}
  \item आयत में विकर्णों की \emph{लंबाई बराबर} होती है;
  \item \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज} में विकर्ण आपस में \emph{\omterm{def:g6:lines:perp}{लंब}} होते हैं;
  \item वर्ग में दोनों बातें एक साथ।
\end{itemize}
\end{proposition}
\admitted

\begin{remark}
वर्ग अपने-आप आयत है (उसके चार \omterm{def:g6:shapes:triangles}{समकोण} हैं) और अपने-आप
\omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज} (चार बराबर भुजाएँ)। इसलिए ``आयत'' का मतलब ``वर्ग
नहीं'' नहीं होता! गणित में ज़्यादा ख़ास आकार, ज़्यादा आम परिवार का
सदस्य होता है।
\end{remark}

\begin{example}[गुणों से आकार पहचानना]\label{ex:g6:shapes:recognize}
किसी \omterm{def:g4:geometry:polygon}{चतुर्भुज} की चारों भुजाएँ $4$ cm की हैं और उसमें एक \omterm{def:g6:shapes:triangles}{समकोण}
है। वह क्या है? चार बराबर भुजाएँ उसे \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज} बनाती हैं।
\omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज} में आमने-सामने के कोण बराबर होते हैं और चारों कोणों
का योग $360^\circ$ है; तब एक \omterm{def:g6:shapes:triangles}{समकोण} चारों को \omterm{def:g6:shapes:triangles}{समकोण} बना देता है
(यह \cref{ch:g7:central} में \omterm{def:g6:lines:perp}{समांतर} \omterm{def:g4:geometry:polygon}{चतुर्भुजों} के साथ सिद्ध
होगा)। तो आकार एक वर्ग है।
\end{example}

\begin{method}[रचना के कार्यक्रम]\label{met:g6:shapes:program}
\emph{रचना का कार्यक्रम} निर्देशों की ऐसी सूची है जिसका पालन करके
कोई भी आकृति हूबहू बना सके। हर पंक्ति में एक निर्देश लिखो, हर
बिंदु बनते ही उसका नाम रखो, और केवल इतना कहो: ``\omterm{def:g6:lines:objects}{रेखाखंड} / रेखा /
वृत्त\dots\ खींचो'', ``वह बिंदु रखो जहाँ \dots\ काटते हैं''।
अपना कार्यक्रम शब्द-दर-शब्द पालन करके जाँचो।
\end{method}

\begin{example}\label{ex:g6:shapes:program}
$4$ cm भुजा वाले वर्ग $ABCD$ का कार्यक्रम:
\begin{enumerate}
  \item $AB = 4$ cm वाला \omterm{def:g6:lines:objects}{रेखाखंड} $[AB]$ खींचो।
  \item $A$ पर $(AB)$ पर \omterm{def:g6:lines:perp}{लंब} खींचो; उस पर $A$ से $4$ cm दूर $D$
        रखो (रेखा का एक ओर चुनो और उसी पर टिके रहो)।
  \item $B$ पर $(AB)$ पर \omterm{def:g6:lines:perp}{लंब} खींचो; उस पर $B$ से $4$ cm दूर $C$
        रखो, $D$ वाली ही ओर।
  \item $C$ और $D$ को जोड़ दो।
\end{enumerate}
मापपट्टी से जाँचो: $DC$ भी $4$ cm ही निकलना चाहिए।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g6:shapes:1}
$AB = 6$ cm, $AC = 5$ cm और $BC = 4$ cm वाला त्रिभुज $ABC$
बनाओ, \cref{met:g6:shapes:construct} के अनुसार।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:2}
$DE = DF = 5$ cm और $EF = 3$ cm वाला \omterm{def:g6:shapes:triangles}{समद्विबाहु त्रिभुज} $DEF$
बनाओ। इस त्रिभुज के कौन-से दो कोण बराबर दिखते हैं? जाँचने के लिए
उन्हें नापो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:3}
केवल मापपट्टी और परकार से $4$ cm भुजा वाला \omterm{def:g6:shapes:triangles}{समबाहु त्रिभुज} बनाओ।
उसका कोई एक कोण नापो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:4}
हर त्रिभुज को उसके आँकड़ों से छाँटो (\omterm{def:g6:shapes:triangles}{समद्विबाहु}, \omterm{def:g6:shapes:triangles}{समबाहु},
\omterm{def:g6:shapes:triangles}{समकोण} --- एक से ज़्यादा उत्तर भी हो सकते हैं):
(a) भुजाएँ $7$, $7$, $7$; \
(b) भुजाएँ $5$, $5$, $8$; \
(c) कोण $90^\circ$, $45^\circ$, $45^\circ$; \
(d) कोण $60^\circ$, $60^\circ$, $60^\circ$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:5}
\omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} में कर्ण कौन-सी भुजा है? $N$ पर \omterm{def:g6:shapes:triangles}{समकोण} वाला
त्रिभुज $MNP$ बनाओ, और उसका कर्ण बताओ। कर्ण किस \omterm{def:g5:solids:def}{शीर्ष} को कभी
नहीं छूता?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:6}
सही या ग़लत? हर उत्तर एक वाक्य में पक्का करो।
\begin{enumerate}
  \item हर वर्ग एक आयत है।
  \item हर आयत एक वर्ग है।
  \item हर वर्ग एक \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज} है।
  \item \omterm{def:g6:shapes:triangles}{समकोण} वाला \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज} एक वर्ग है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:7}
$AB = 5$ cm और $BC = 3$ cm वाला आयत $ABCD$ बनाओ। उसके दोनों
विकर्ण नापो: क्या दिखता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:8}
$6$ cm और $4$ cm के विकर्णों वाला \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज} बनाओ।
(\cref{prop:g6:shapes:diagonals} लगाओ: पहले दोनों विकर्ण खींचो
--- \omterm{def:g6:lines:perp}{लंब}, अपने मध्यबिंदुओं पर कटते हुए --- फिर चारों सिरे जोड़
दो।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:9}
\cref{ex:g6:shapes:program} की शैली में $4$ cm आधार और $6$-$6$ cm
बराबर भुजाओं वाले \omterm{def:g6:shapes:triangles}{समद्विबाहु त्रिभुज} की रचना का कार्यक्रम लिखो।
किसी सहपाठी से बदलो: क्या वह तुम्हारी आकृति देखे बिना उसका पालन कर पाता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:10}
केंद्र $O$ वाला एक वृत्त और उसका एक \omterm{def:g4:geometry:circle}{व्यास} $[AB]$ बनाओ। वृत्त पर
कोई भी बिंदु $M$ रखो और त्रिभुज $AMB$ बनाओ। कोण $\widehat{AMB}$
नापो। $M$ को हिलाकर फिर नापो। तुम्हारा अनुमान क्या है?
(यह चौंकाने वाला तथ्य \cref{ch:g8:cosine} में सिद्ध होता है।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:11}
किसी \omterm{def:g4:geometry:polygon}{चतुर्भुज} के विकर्णों की लंबाई बराबर है और वे एक-दूसरे को
उनके साझा मध्यबिंदु पर काटते हैं। \cref{prop:g6:shapes:diagonals}
से तय करो कि वह कौन-सा आकार होगा: आयत, \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज} या वर्ग?
(सावधान: केवल एक ही गुण दिया गया है।)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:shapes:12}
$4 \times 4$ बिंदुओं के जाल पर ऐसे कितने अलग-अलग वर्ग बनाए जा
सकते हैं जिनके कोने जाल के बिंदु हों --- ``तिरछे'' वर्ग भी गिनकर?
कम से कम एक तिरछा वर्ग बनाओ और समझाओ कि उसकी भुजाएँ बराबर क्यों हैं।
\end{exercise}

\section{समस्या: विकर्ण क्या जानते हैं}

\begin{problem}[{सप्ताहांत समस्या --- हर ख़ास चतुर्भुज को केवल
उसके विकर्णों से दोबारा खड़ा करना, और किसी भी बहुभुज के विकर्ण
गिनना}]
\label{pb:g6:shapes:1}
\Cref{prop:g6:shapes:diagonals} किसी आकार को पढ़कर उसके विकर्णों
के बारे में बताती है। यह समस्या वही खेल \emph{उलटा} खेलती है:
केवल दो कटते हुए \omterm{def:g6:lines:objects}{रेखाखंड} दिए हों --- होने वाले विकर्ण --- तो उनके
चारों सिरे जोड़ने पर कौन-सा आकार बनता है? तुम एक पूरी
``विकर्ण-कोशिका'' बनाओगे, समझोगे कि \cref{exo:g6:shapes:8} की
समचतुर्भुज-रचना \emph{क्यों} चलती है, एक ऐसे आकार से मिलोगे
जिसका अभी नाम नहीं है, और अंत में चार से कहीं ज़्यादा भुजाओं वाले
बहुभुजों के विकर्ण गिनोगे।

\medskip
\noindent\textbf{भाग I --- विकर्ण-कोशिका।}
हर प्रश्न में बताए अनुसार दोनों \omterm{def:g6:lines:objects}{रेखाखंड} बनाओ, उनके चारों सिरे
क्रम से जोड़ो, और जो \omterm{def:g4:geometry:polygon}{चतुर्भुज} मिले उसे पहचानो।
\begin{enumerate}
  \item $6$-$6$ cm के दो \omterm{def:g6:lines:objects}{रेखाखंड}, आपस में \omterm{def:g6:lines:perp}{लंब}, अपने साझा
        मध्यबिंदु पर कटते हुए। बने आकार की चारों भुजाएँ और चारों
        कोण नापो: वह क्या है?
  \item $8$ cm और $5$ cm के दो \omterm{def:g6:lines:objects}{रेखाखंड}, आपस में \omterm{def:g6:lines:perp}{लंब}, अपने साझा
        मध्यबिंदु पर कटते हुए। कौन-सा आकार बनता है
        (\cref{exo:g6:shapes:8})?
  \item $7$-$7$ cm के दो \omterm{def:g6:lines:objects}{रेखाखंड} --- बराबर, पर आपस में \omterm{def:g6:lines:perp}{लंब}
        \emph{नहीं} --- अपने साझा मध्यबिंदु पर कटते हुए।
        भुजाएँ और कोण नापो: तुम्हारा अनुमान क्या है कि यह
        कौन-सा आकार है?
  \item $8$ cm और $5$ cm के दो \omterm{def:g6:lines:objects}{रेखाखंड}, आपस में \omterm{def:g6:lines:perp}{लंब} नहीं, अपने
        साझा मध्यबिंदु पर कटते हुए। जो आकार मिलता है वह
        \cref{def:g6:shapes:quadrilaterals} के तीनों में से कोई
        नहीं --- जो दिखे उसका वर्णन करो (कौन-सी भुजाएँ बराबर
        दिखती हैं? \omterm{def:g6:lines:perp}{समांतर}?)। इस \omterm{def:g4:geometry:polygon}{चतुर्भुज} को नाम
        \cref{ch:g7:central} में मिलता है।
  \item प्रश्न 1--4 को चार प्रविष्टियों वाली कोशिका में समेटो:
        विकर्ण \emph{बराबर / \omterm{def:g6:lines:perp}{लंब} / दोनों / कोई नहीं} (हमेशा अपने
        मध्यबिंदुओं पर कटते हुए)
        $\rightarrow$ आकार का नाम (या वर्णन)।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- रचनाएँ क्यों चलती हैं।}
\begin{enumerate}[resume]
  \item प्रश्न 2 में दोनों विकर्ण आकार को कटान-बिंदु के चारों ओर
        चार \omterm{def:g6:shapes:triangles}{समकोण} त्रिभुजों में काट देते हैं। इनमें से हर
        त्रिभुज की दोनों भुजाओं की लंबाई क्या है? समझाओ कि चारों
        त्रिभुज एक-दूसरे की हूबहू नक़ल क्यों हैं, और उससे निकालो
        कि आकार की चारों भुजाएँ बराबर क्यों हैं ---
        \cref{exo:g6:shapes:8} का यही राज़ है।
  \item इससे केवल परकार वाली एक जाँच निकालो: विकर्णों से कोई भी
        आकार बनाने के बाद, बिना मापपट्टी से नापे, केवल परकार से
        कैसे जाँचोगे कि चारों भुजाएँ बराबर हैं?
  \item ऐसे वर्ग की रचना का कार्यक्रम
        (\cref{met:g6:shapes:program}) लिखो जिसके \emph{विकर्ण}
        $6$ cm के हों --- उसकी भुजाएँ नहीं।
  \item प्रश्न 8 के वर्ग में भुजा $3$ cm --- यानी विकर्ण के आधे
        --- से लंबी है या छोटी? नापकर पता करो। उत्तर का एक हिस्सा
        अभी \emph{सिद्ध} भी किया जा सकता है: एक कोने से अगले
        कोने तक केंद्र होकर जाना $3 + 3$ cm का चक्कर है, और सीधा
        रास्ता चक्कर से हमेशा छोटा होता है --- इसलिए भुजा $6$ cm
        से छोटी है। (वह $3$ cm से भी बड़ी है, यह सिद्ध करना और
        उसका ठीक मान निकालना \cref{ch:g8:pythagoras} का काम है।)
  \item कोशिका के लिए एक नई प्रविष्टि: $8$ cm का \omterm{def:g6:lines:objects}{रेखाखंड} $[AC]$
        खींचो; उस पर $A$ से $3$ cm दूर बिंदु $H$ अंकित करो; $H$
        से होकर $5$ cm का \omterm{def:g6:lines:perp}{लंब} \omterm{def:g6:lines:objects}{रेखाखंड} $[BD]$ खींचो, जिसका
        मध्यबिंदु $H$ ही हो। $ABCD$ जोड़ो। यह आकार एक
        \emph{पतंग}\index{पतंग} है (एक विकर्ण दूसरे को \omterm{def:g6:shapes:triangles}{समकोण} पर
        समद्विभाजित करता है)। प्रश्न 6 की तरह कोनों के त्रिभुजों
        से पता करो कि \omterm{pb:g6:shapes:1}{पतंग} की कौन-सी भुजाएँ किनके बराबर हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- विकर्ण गिनना।}
किसी \omterm{def:g4:geometry:polygon}{बहुभुज} का \emph{विकर्ण} ऐसे दो \omterm{def:g5:solids:def}{शीर्षों} को जोड़ता है जो
पड़ोसी न हों।
\begin{enumerate}[resume]
  \item एक \omterm{def:g4:geometry:polygon}{चतुर्भुज}, एक \omterm{def:g4:geometry:polygon}{पंचभुज} (पाँच \omterm{def:g5:solids:def}{शीर्ष}) और एक \omterm{def:g4:geometry:polygon}{षट्भुज} (छह
        \omterm{def:g5:solids:def}{शीर्ष}) बनाओ, और उनके सारे विकर्ण खींचो। उन्हें गिनो: हर
        हाल में कितने हैं?
  \item बिना बनाए गिनो, \omterm{def:g4:geometry:polygon}{षट्भुज} के लिए: एक \omterm{def:g5:solids:def}{शीर्ष} से कितने विकर्ण
        निकलते हैं? (छहों \omterm{def:g5:solids:def}{शीर्षों} में से कितनों से उसे विकर्ण
        \emph{नहीं} जोड़ सकता?) छहों \omterm{def:g5:solids:def}{शीर्षों} से गुणा करने पर हर
        विकर्ण ठीक दो बार गिना जाता है --- क्यों? प्रश्न 11 की
        अपनी गिनती फिर से पाओ। (यही दोहरी गिनती
        \cref{pb:g6:wholes:1} के हाथ मिलाने में आई थी।)
  \item प्रश्न 12 के तरीक़े से $10$ \omterm{def:g5:solids:def}{शीर्षों} वाले \omterm{def:g4:geometry:polygon}{बहुभुज} के, फिर
        $20$ \omterm{def:g5:solids:def}{शीर्षों} वाले \omterm{def:g4:geometry:polygon}{बहुभुज} के विकर्ण गिनो।
  \item किसी \omterm{def:g4:geometry:polygon}{बहुभुज} के ठीक $44$ विकर्ण हैं। उसके कितने \omterm{def:g5:solids:def}{शीर्ष}
        हैं? (प्रश्न 13 का तरीक़ा कुछ उम्मीदवारों पर
        आज़माओ।)
  \item अपना तरीक़ा सबसे छोटे \omterm{def:g4:geometry:polygon}{बहुभुज} पर जाँचो: त्रिभुज के कितने
        विकर्ण होते हैं, और प्रश्न 12 का गिनती-तरीक़ा क्या उत्तर
        देता है? एक वाक्य में बताओ कि हाथ मिलाने और विकर्णों में
        क्या समान है।
\end{enumerate}
\end{problem}
