\chapter{परिमाप, क्षेत्रफल, आयतन}\label{ch:g6:measure}

बाड़ कितनी लंबी है, खेत कितना बड़ा है, टंकी में कितना पानी समाता
है? तीन अलग सवाल, तीन अलग राशियाँ --- \omterm{def:g6:measure:perimeter}{परिमाप}, \omterm{def:g6:measure:area}{क्षेत्रफल}, \omterm{def:g6:measure:volume}{आयतन} ---
और हर एक के अपने मात्रक। इन्हें आपस में गड्डमड्ड कर देना ज्यामिति
की सबसे आम ग़लती है; यह अध्याय उन्हें हमेशा के लिए अलग-अलग कर देता है।

\section{लंबाइयाँ और परिमाप}

\begin{definition}[परिमाप]\label{def:g6:measure:perimeter}
किसी आकृति का \emph{परिमाप}\index{परिमाप} उसकी सीमा की कुल लंबाई
है। वह लंबाई के मात्रकों में नापा जाता है: मिलीमीटर (mm),
सेंटीमीटर (cm), मीटर (m), किलोमीटर (km), जहाँ
\[
1 \text{ km} = 1000 \text{ m}, \qquad
1 \text{ m} = 100 \text{ cm}, \qquad
1 \text{ cm} = 10 \text{ mm}.
\]
\end{definition}

\begin{example}\label{ex:g6:measure:perimeter}
लंबाई $L$ और चौड़ाई $w$ वाले आयत का \omterm{def:g6:measure:perimeter}{परिमाप} है
\[
P = L + w + L + w = 2 \times (L + w).
\]
$7$ cm गुणा $4$ cm के आयत के लिए: $P = 2 \times (7 + 4) = 2 \times 11 =
22$ cm। भुजा $c$ वाले वर्ग का \omterm{def:g6:measure:perimeter}{परिमाप} $4c$ है।
\end{example}

\begin{proposition}[वृत्त की परिधि]\label{prop:g6:measure:circle}
\omterm{def:g3:shapes:shapes}{त्रिज्या} $r$ वाले वृत्त का \omterm{def:g6:measure:perimeter}{परिमाप} (या \emph{परिधि}) है
\[
P = 2 \pi r ,
\]
जहाँ $\pi \approx 3.14$ हर वृत्त के लिए वही संख्या है।
\end{proposition}
\admitted

\begin{example}
एक गोल तालाब की \omterm{def:g3:shapes:shapes}{त्रिज्या} $5$ m है। उसकी सीमा
$2 \times \pi \times 5 = 10\pi \approx 31.4$ m नापती है। ठीक मान
$10\pi$ जितनी देर हो सके बनाए रखो; गोल केवल अंत में करो।
\end{example}

\section{क्षेत्रफल}

\begin{definition}[क्षेत्रफल]\label{def:g6:measure:area}
किसी आकृति का \emph{क्षेत्रफल}\index{क्षेत्रफल} बताता है कि वह
कितनी सतह ढकती है: उसके भीतर कितने इकाई वर्ग समाते हैं। मात्रक:
cm$^2$ ($1$ cm भुजा वाला वर्ग), m$^2$, km$^2$, \dots\ सावधान:
\[
1 \text{ m}^2 = 100 \times 100 \text{ cm}^2 = 10\,000 \text{ cm}^2
\]
--- एक वर्ग मीटर $100$ cm गुणा $100$ cm का वर्ग है, इसलिए मात्रक
की सीढ़ी का हर कदम $100$ का है, $10$ का नहीं।
\end{definition}

यानी कुल $7 \times 4 = 28$ वर्ग। इसीलिए \omterm{def:g6:measure:area}{क्षेत्रफल} $=$ लंबाई
$\times$ चौड़ाई।}
\end{omfigure}

\begin{proposition}[क्षेत्रफल के बुनियादी सूत्र]\label{prop:g6:measure:formulas}
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{l|l}
आकृति & \omterm{def:g6:measure:area}{क्षेत्रफल} \\
\hline
आयत ($L \times w$) & $A = L \times w$ \\
वर्ग (भुजा $c$) & $A = c^2$ \\[4pt]
\omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} (भुजाएँ $a$, $b$) & $A = \dfrac{a \times b}{2}$ \\[4pt]
चक्र (\omterm{def:g3:shapes:shapes}{त्रिज्या} $r$) & $A = \pi r^2$ \\
\end{tabular}
\end{center}
\end{proposition}

\begin{proof}[समकोण त्रिभुज की उपपत्ति]
$a$ और $b$ भुजाओं वाले \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} की दो प्रतियाँ कर्ण के साथ
जोड़ने पर $a \times b$ का आयत बन जाता है। इसलिए त्रिभुज का
\omterm{def:g6:measure:area}{क्षेत्रफल} आयत का \omterm{def:g3:division:half}{आधा} है: $\frac{ab}{2}$। (आयत का सूत्र ऊपर की
इकाई-वर्ग गिनती है; चक्र का सूत्र स्वीकृत है, देखो
\cref{ch:g7:areas}।)
\end{proof}

\begin{example}[जोड़-तोड़ वाली आकृतियाँ]\label{ex:g6:measure:composite}
L-आकार का कमरा $6$ m $\times$ $4$ m का आयत है जिसमें से $2$ m
$\times$ $2$ m का वर्गाकार कोना हटा दिया गया है। उसका \omterm{def:g6:measure:area}{क्षेत्रफल}, कदम-दर-कदम:
\begin{enumerate}
  \item पूरा आयत: $6 \times 4 = 24$ m$^2$;
  \item हटाया गया वर्ग: $2 \times 2 = 4$ m$^2$;
  \item बचा \omterm{def:g6:measure:area}{क्षेत्रफल}: $24 - 4 = 20$ m$^2$।
\end{enumerate}
उसका \emph{\omterm{def:g6:measure:perimeter}{परिमाप}} $20$ किसी चीज़ का नहीं है: L के चारों ओर चलने
पर सीमा फिर भी $6 + 4 + 6 + 4 = 20$ m नापती है --- कोने की दोनों
कटाइयाँ दीवार के दो बराबर टुकड़ों की जगह ले लेती हैं। संख्या
संयोग से वही है, पर एक के लिए वर्ग मीटर और दूसरे के लिए मीटर!
\end{example}

\begin{remark}[वही परिमाप, अलग क्षेत्रफल]
दो आकृतियों का \omterm{def:g6:measure:perimeter}{परिमाप} एक हो सकता है और \omterm{def:g6:measure:area}{क्षेत्रफल} बहुत अलग:
$5 \times 5$ के वर्ग और $9 \times 1$ के आयत, दोनों का \omterm{def:g6:measure:perimeter}{परिमाप}
$20$ है, पर \omterm{def:g6:measure:area}{क्षेत्रफल} $25$ और $9$। \omterm{def:g6:measure:perimeter}{परिमाप} और \omterm{def:g6:measure:area}{क्षेत्रफल} सचमुच
स्वतंत्र राशियाँ हैं।
\end{remark}

\section{आयतन}

\begin{definition}[आयतन]\label{def:g6:measure:volume}
किसी \omterm{def:g5:solids:def}{ठोस} का \emph{आयतन}\index{आयतन} बताता है कि वह कितनी जगह
भरता है: उसके भीतर कितने इकाई \omterm{def:g5:solids:def}{घन} समाते हैं। मात्रक: cm$^3$,
m$^3$, \dots\ जहाँ $1$ m$^3 = 1\,000\,000$ cm$^3$ (सीढ़ी का हर
कदम $1000$ का है)। तरल पदार्थों के लिए \emph{लीटर} भी चलता है:
\[
1 \text{ L} = 1 \text{ dm}^3 = 1000 \text{ cm}^3,
\qquad
1 \text{ m}^3 = 1000 \text{ L}.
\]
\end{definition}

\begin{proposition}[घनाभ का आयतन]\label{prop:g6:measure:box}
लंबाई $L$, चौड़ाई $w$ और ऊँचाई $h$ वाले आयताकार \omterm{def:g5:solids:def}{घनाभ}
(\emph{आयताकार \omterm{def:g5:solids:def}{ठोस}}) का \omterm{def:g6:measure:volume}{आयतन} है
\[
V = L \times w \times h .
\]
\end{proposition}

\begin{proof}
सबसे नीचे की परत में $L \times w$ इकाई \omterm{def:g5:solids:def}{घन} आते हैं
(\cref{def:g6:measure:area} वाला चित्र, वर्गों की जगह घनों के
साथ), और ऐसी $h$ परतें हैं।
\end{proof}

परतें।}
\end{omfigure}

\begin{example}\label{ex:g6:measure:aquarium}
एक जलजीवशाला $60$ cm गुणा $30$ cm गुणा $40$ cm (ऊँचाई) की है। \omterm{def:g6:measure:volume}{आयतन}:
\[
V = 60 \times 30 \times 40 = 72\,000 \text{ cm}^3 = 72 \text{ dm}^3
= 72 \text{ L}.
\]
बदलने का कदम: $1000$ cm$^3$ मिलकर एक लीटर बनते हैं, और
$72\,000 \div 1000 = 72$।
\end{example}

\section{समस्या: रानी दीदो की बाड़}

\begin{problem}[{सप्ताहांत समस्या --- बाड़ की तय लंबाई के साथ
कौन-सा आकार सबसे ज़्यादा ज़मीन घेरता है? वर्ग हर आयत को हराता है,
वृत्त वर्ग को हराता है, और खलिहान की दीवार सब कुछ बदल देती
है}]
\label{pb:g6:measure:1}
कथा कहती है कि रानी दीदो को समुद्र-तट पर उतरने पर ``उतनी ज़मीन''
दी गई ``जितनी एक बैल की खाल घेर सके'' --- तो उसने खाल को एक
बेहद लंबी पतली पट्टी में काटा और इतनी ज़मीन घेर ली कि कार्थेज
नगर बस गया। अब उसकी समस्या तुम्हारी है: एक किसान के पास ठीक
$20$ m बाड़ है। \Cref{exo:g6:measure:8} ने दिखाया था कि एक ही
\omterm{def:g6:measure:perimeter}{परिमाप} वाले दो बाड़ों के \omterm{def:g6:measure:area}{क्षेत्रफल} अलग हो सकते हैं; यह समस्या
\emph{सबसे अच्छा} बाड़ा खोजती है --- और पता चलता है कि खलिहान की
दीवार एक भुजा मुफ़्त दे दे, तो उत्तर पूरी तरह बदल जाता है।

\medskip
\noindent\textbf{भाग I --- बीस मीटर बाड़।}
\begin{enumerate}
  \item बाड़ा आयत होना चाहिए और पूरी $20$ m बाड़ इस्तेमाल होनी
        चाहिए। जाँचो कि $1 \times 9$ और $2 \times 8$ के बाड़े
        दोनों शर्त पूरी करते हैं, और उनके \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो।
  \item समझाओ कि हर योग्य बाड़े की लंबाई और चौड़ाई मिलकर $10$ m
        क्यों बनती हैं। फिर पूर्ण संख्याओं वाले बाड़ों
        ($1 \times 9$ से $5 \times 5$ तक) की उनके क्षेत्रफलों
        समेत पूरी सारणी बनाओ। सबसे अच्छा कौन-सा है?
  \item क्या दशमलव भुजाएँ आज़माने लायक़ हैं? $4.5 \times 5.5$ और
        $4.9 \times 5.1$ के बाड़ों के \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो, और
        $5 \times 5$ के वर्ग से तुलना करो। तुम्हारा अनुमान
        क्या है?
  \item प्रश्न 2 ने बाड़ की समस्या को शुद्ध संख्या के सवाल में
        बदल दिया: \emph{$10$ योग वाली संख्याओं की सारी जोड़ियों
        में किस जोड़ी का \omterm{def:g2:mult:def}{गुणनफल} सबसे बड़ा है?} अपनी सारणी से उत्तर
        दो, और जो सामान्य नियम दिखे उसे लिखो।
  \item यहाँ बीजगणित के बजाय क़ैंची से समझो कि वर्ग से हटना
        हमेशा घाटे का सौदा क्यों है: $5 \times 5$ के बाड़े से
        शुरू करो और एक पट्टी हटाकर दूसरी चिपकाकर उसे
        $6 \times 4$ का बाड़ा बना दो। कौन-सी पट्टी हटती है,
        कौन-सी जुड़ती है, और अदला-बदली में ठीक एक वर्ग मीटर का
        घाटा क्यों होता है? समझाओ कि अगला कदम ($7 \times 3$ तक)
        और ज़्यादा घाटा क्यों देता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- खलिहान, और वृत्त।}
\begin{enumerate}[resume]
  \item अब बाड़ा खलिहान की एक लंबी दीवार से सटाकर बनता है: दीवार
        बाड़े की एक लंबाई की जगह ले लेती है, और $20$ m बाड़ केवल
        बाक़ी तीन भुजाओं पर लगती है। पूर्ण संख्याओं वाले बाड़ों
        (चौड़ाई $w$, दीवार के साथ लंबाई $L$, $w + L + w = 20$)
        और उनके क्षेत्रफलों की सारणी बनाओ। अब कौन-सा बाड़ा जीतता
        है --- और क्या वह वर्ग है?
  \item जीतने वाले को दर्पण से समझाओ
        (\cref{ch:g6:symmetry}): बाड़े को खलिहान की दीवार पर
        परावर्तित करो और दुगुने बाड़े को देखो। दुगुना बाड़ा कितनी
        बाड़ इस्तेमाल करता है, भाग I के अनुसार कौन-सा दुगुना
        बाड़ा सबसे अच्छा है, और उससे असली बाड़ा क्या बनता है?
  \item दीदो ने आयत नहीं बनाया था। $20$ m बाड़ को मोड़कर वृत्त
        बनाओ: $\pi \approx 3.14$ लेकर उसकी \omterm{def:g3:shapes:shapes}{त्रिज्या} (cm तक),
        फिर उसका \omterm{def:g6:measure:area}{क्षेत्रफल} (m$^2$ तक) निकालो
        (\cref{prop:g6:measure:circle},
        \cref{prop:g6:measure:formulas})। $5 \times 5$ के वर्ग
        से तुलना करो।
  \item उलटी समस्या: ठीक $36$ m$^2$ \omterm{def:g6:measure:area}{क्षेत्रफल} वाला बाड़ा चाहिए,
        और बाड़ जितनी कम लगे उतना अच्छा। आयतों $1 \times 36$,
        $2 \times 18$, $3 \times 12$,
        $4 \times 9$ और $6 \times 6$ की तुलना करो: \omterm{def:g6:measure:perimeter}{परिमाप}? कौन-सा
        आकार सबसे सस्ता है, और यह सवाल भाग I का दर्पण-प्रतिबिंब
        कैसे है?
  \item एक वाक्य में: टिन, पाइप और पानी की टंकियाँ इतनी बार
        \emph{गोल} क्यों होती हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{def:g6:measure:perimeter}{परिमाप} और \omterm{def:g6:measure:area}{क्षेत्रफल} अजनबी हैं।}
\begin{enumerate}[resume]
  \item ऐसा आयत ढूँढ़ो जिसका \omterm{def:g6:measure:perimeter}{परिमाप} $100$ m से ज़्यादा हो पर
        \omterm{def:g6:measure:area}{क्षेत्रफल} $1$ m$^2$ से कम। (बहुत लंबा और बहुत पतला।
        दशमलव चलेंगे।)
  \item सही या ग़लत: ``दो आकृतियों में जिसका \omterm{def:g6:measure:perimeter}{परिमाप} बड़ा हो
        उसका \omterm{def:g6:measure:area}{क्षेत्रफल} भी बड़ा होता है।'' इसी समस्या से एक
        प्रति-उदाहरण दो।
  \item $6 \times 4$ का आयत लो और उसकी किसी लंबी भुजा के बीच
        में $1 \times 1$ का वर्गाकार खाँचा काट दो (खाँचा बाहर की
        ओर खुलता है, जैसे कोई दाँत ग़ायब हो)। खाँचे वाली आकृति का
        \omterm{def:g6:measure:area}{क्षेत्रफल} और \omterm{def:g6:measure:perimeter}{परिमाप} निकालो, और दोनों की तुलना मूल आयत से
        करो। हर एक का क्या हुआ?
  \item नक़्शा बनाने वाले एक अजीब बात जानते हैं: नक़्शा जितना
        बारीक़ हो, समुद्र-तट उतना ही \emph{लंबा} नापता है ---
        हर ज़ूम नए छोटे-छोटे खाँचे दिखा देता है, और प्रश्न 13
        बताता है कि हर खाँचा क्या करता है। एक-दो वाक्यों में
        समझाओ कि ``किसी तट की लंबाई'' फिसलनी संख्या क्यों है,
        जबकि ``उसका \omterm{def:g6:measure:area}{क्षेत्रफल}'' नहीं।
  \item किसान की परीक्षा। $36$ m बाड़ और खलिहान की दीवार उपलब्ध
        हो, तो सबसे अच्छा आयताकार बाड़ा ढूँढ़ो (प्रश्न 7 वाला
        दर्पण लगाओ), उसका \omterm{def:g6:measure:area}{क्षेत्रफल} बताओ, और खुले मैदान में
        $36$ m से बने सबसे अच्छे बाड़े से तुलना करो। खलिहान की
        दीवार किसान को कितना कमा कर देती है?
\end{enumerate}
\end{problem}
