\chapter{दशमलव संख्याएँ}\label{ch:g6:decimals}

$2$ और $3$ के बीच ढेरों संख्याएँ हैं: $2.5$, $2.71$,
$2.999$\,\dots\ \omterm{def:g6:decimals:places}{दशमलव संख्याएँ} स्थानीय-मान की प्रणाली को इकाई के
दाईं ओर बढ़ा देती हैं, और वे पूर्ण संख्याओं जैसे ही नियमों पर
चलती हैं --- कुछ जालों के साथ, जिनसे बचना यह अध्याय सिखाएगा।

\section{दशमलव लेखन}

\begin{definition}[दशमलव स्थान]\label{def:g6:decimals:places}
किसी \emph{दशमलव संख्या}\index{दशमलव संख्या} का एक पूर्ण भाग
होता है और एक दशमलव भाग, जिन्हें \omterm{def:g4:decimals:point}{दशमलव बिंदु} अलग करता है। बिंदु
के दाईं ओर हर स्थान अपने से पहले वाले से दस गुना कम का होता है:
दसवाँ, सौवाँ, हज़ारवाँ भाग। जैसे
\[
13.407 = 13 + \frac{4}{10} + \frac{0}{100} + \frac{7}{1000}
= 1 \times 10 + 3 + 4 \times \frac{1}{10} + 7 \times \frac{1}{1000}.
\]
\end{definition}

\begin{remark}[कुछ शून्य मायने रखते हैं, कुछ नहीं]
दशमलव भाग के \emph{अंत} में \omterm{def:g1:counting:zero}{शून्य} जोड़ने से कुछ नहीं बदलता:
$2.5 = 2.50 = 2.500$। पर अंकों के \emph{बीच} के \omterm{def:g1:counting:zero}{शून्य} ज़रूरी हैं:
$13.407 \neq 13.47$। और $0.5$, $0.05$ से बहुत अलग है!
\end{remark}

\begin{method}[दशमलव संख्याओं की तुलना]\label{met:g6:decimals:compare}
\begin{enumerate}
  \item पहले पूर्ण भागों की तुलना करो: $7.2 > 5.99$, क्योंकि $7 > 5$।
  \item पूर्ण भाग बराबर हों, तो दशमलव भागों की तुलना बाईं ओर से
        अंक-दर-अंक करो --- अंत में \omterm{def:g1:counting:zero}{शून्य} भरने से मदद मिलती है:
        $3.4$ और $3.15$ की तुलना के लिए $3.40$ और $3.15$ लिखो;
        चूँकि $40 > 15$ सौवें भाग, इसलिए $3.4 > 3.15$।
\end{enumerate}
सावधान: \emph{लंबा होने का मतलब बड़ा होना नहीं}। $3.15$ में
$3.4$ से ज़्यादा अंक हैं, फिर भी वह छोटी है।
\end{method}

\section{जोड़, घटाव, गुणा}

\begin{method}[दशमलव संख्याओं की खानों में गणना]\label{met:g6:decimals:columns}
\omterm{def:g6:decimals:places}{दशमलव संख्याएँ} जोड़ने या घटाने के लिए \emph{\omterm{def:g4:decimals:point}{दशमलव बिंदु}} एक सीध
में रखो (ताकि इकाई के नीचे इकाई और दसवें भाग के नीचे दसवाँ भाग
आए), ज़रूरत हो तो अंत में \omterm{def:g1:counting:zero}{शून्य} भर दो; फिर पूर्ण संख्याओं की तरह
गणना करो, और नतीजे का बिंदु बाक़ी बिंदुओं के नीचे रखो।
\end{method}

\begin{example}\label{ex:g6:decimals:addsub}
$13.7 + 2.85$ निकालो, बिंदु एक सीध में रखकर ($13.7 = 13.70$):
\[
\begin{array}{r}
1\,3.7\,0 \\
+\ \ \,2.8\,5 \\
\hline
1\,6.5\,5
\end{array}
\]
$6 - 2.35$ निकालो ($6 = 6.00$ लिखो):
$6.00 - 2.35 = 3.65$। जाँच: $3.65 + 2.35 = 6$।
\end{example}

\begin{example}[दशमलव संख्याओं का गुणा]\label{ex:g6:decimals:mult}
$2.3 \times 1.4$ निकालने के लिए: पूर्ण संख्याओं की तरह गुणा करो,
$23 \times 14 = 322$; फिर गुणनखंडों के दशमलव अंक गिनो
($1 + 1 = 2$), और बिंदु ऐसे रखो कि नतीजे में उतने ही अंक हों:
$2.3 \times 1.4 = 3.22$। आकार की मोटी जाँच: $2.3 \times 1.4$,
$2.3 \times 1 = 2.3$ से थोड़ा ज़्यादा होना चाहिए --- और $3.22$ है।
\end{example}

\section{10, 100, 1000 से गुणा और भाग}

\begin{proposition}[बिंदु का खिसकना]\label{prop:g6:decimals:shift}
किसी \omterm{def:g6:decimals:places}{दशमलव संख्या} को $10$, $100$, $1000$ से गुणा करने पर उसका
\omterm{def:g4:decimals:point}{दशमलव बिंदु} $1$, $2$, $3$ स्थान \emph{दाईं} ओर खिसकता है; भाग
देने पर \emph{बाईं} ओर (ज़रूरत पड़ने पर \omterm{def:g1:counting:zero}{शून्य} भरते हुए):
\[
3.75 \times 100 = 375, \qquad
42.1 \div 1000 = 0.0421 .
\]
\end{proposition}

\begin{proof}
$10$ से गुणा करने पर हर अंक दस गुना क़ीमती हो जाता है: दसवें भाग
इकाई बन जाते हैं, इकाई दहाई, और ऐसे ही आगे --- स्थानीय-मान सारणी
में हर अंक एक खाना बाईं ओर सरक जाता है, जो बिंदु को एक स्थान दाईं
ओर सरकाने के बराबर है। भाग देने पर यह उलट जाता है।
\end{proof}

\begin{example}[माप के मात्रक]\label{ex:g6:decimals:units}
मात्रक बदलना ठीक यही खेल है: चूँकि $1$ m $= 100$ cm,
\[
3.75 \text{ m} = 3.75 \times 100 \text{ cm} = 375 \text{ cm},
\qquad
42 \text{ mm} = 42 \div 10 \text{ cm} = 4.2 \text{ cm}.
\]
\end{example}

\section{गोल करना}

\begin{definition}[गोलन]\label{def:g6:decimals:rounding}
किसी संख्या का इकाई तक (या दसवें, सौवें भाग, \dots\ तक)
\emph{गोलन}\index{गोलन} उस परिशुद्धता वाली सबसे नज़दीकी संख्या
है। अगला अंक देखो: वह $0,1,2,3,4$ हो तो नीचे गोल करो (जैसा है
वैसा रखो); $5,6,7,8,9$ हो तो ऊपर।
\end{definition}

\begin{example}
$7.38$ इकाई तक गोल करने पर $7$ है (अगला अंक $3$: वैसा ही रखो);
दसवें भाग तक $7.4$ (अगला अंक $8$: ऊपर गोल)। क़ीमत
$4.996$ सौवें भाग तक गोल करने पर $5.00$ है --- \omterm{def:g6:decimals:rounding}{गोलन} हर अंक बदल
सकता है!
\end{example}

\section{समस्या: 3 के ठीक बाद वाली संख्या होती ही नहीं}

\begin{problem}[{सप्ताहांत समस्या --- किन्हीं दो दशमलव संख्याओं
के बीच हमेशा एक और होती है: संख्या रेखा पर ज़ूम}]
\label{pb:g6:decimals:1}
पूर्ण संख्याओं की सीढ़ी पर हर संख्या का एक पड़ोसी होता है: $7$ के
ठीक बाद $8$ आता है, और बीच में कोई नहीं रहता। \omterm{def:g6:decimals:places}{दशमलव संख्याएँ}
बिलकुल दूसरी दुनिया हैं। $3$ के ठीक बाद वाली संख्या कौन-सी है?
$3.1$? $3.01$? $3.001$? यह समस्या संख्या रेखा पर \emph{ज़ूम}
करने की तकनीक बनाती है (\cref{exo:g6:decimals:11} को आगे
बढ़ाते हुए) और एक मशहूर निष्कर्ष तक पहुँचती है: $3$ के ठीक बाद
वाली संख्या \emph{होती ही नहीं} --- किन्हीं दो \omterm{def:g6:decimals:places}{दशमलव संख्याओं} के
बीच, वे कितनी भी पास क्यों न हों, और के लिए जगह हमेशा बची रहती है।

\medskip
\noindent\textbf{भाग I --- ज़ूम करना।}
\begin{enumerate}
  \item कौन बड़ी है, $2.999$ या $3$? दोनों का \omterm{ex:g1:subtraction:difference}{अंतर} निकालो।
  \item $7.4$ से $7.5$ तक की संख्या रेखा सौवें भागों में अंकित
        करके बनाओ, और उस पर $7.42$, $7.45$ तथा $7.48$ रखो।
        $7.4$ और $7.5$ के ठीक बीच पड़ने वाली दो दशमलव अंकों
        वाली \emph{सारी} संख्याएँ गिनाओ। वे कितनी हैं?
  \item फिर ज़ूम करो: $7.44$ और $7.45$ के बीच तीन दशमलव अंकों
        वाली सारी संख्याएँ गिनाओ। इस बार वे कितनी
        हैं?
  \item इसी विचार से समझाओ कि $7.4$ और $7.5$ के ठीक बीच पड़ने
        वाली, ठीक तीन दशमलव अंकों वाली संख्याएँ कैसे गिनी जाएँ,
        और वह गिनती बताओ।
  \item प्रश्न 2--4 के पीछे छिपी विधि (\emph{ज़ूम}) बताओ: पड़ोसी
        जैसी दिखने वाली दो संख्याएँ, जैसे $5.67$ और $5.68$, दी
        हों, तो एक और दशमलव स्थान लिखने पर उनके ठीक बीच की कोई
        संख्या हमेशा कैसे मिल जाती है? अपनी विधि $5.67$ और $5.68$
        पर लगाओ, फिर $0.1999$ और $0.2$ पर।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- ग़ायब पड़ोसी।}
\begin{enumerate}[resume]
  \item टॉम का दावा है: ``$3$ के ठीक बाद वाली संख्या $3.1$ है।''
        $3$ और $3.1$ के ठीक बीच की कोई संख्या बताकर उसे ग़लत
        साबित करो। टॉम $3.01$ पर, फिर $3.001$ पर हटता है। उसके
        हर उम्मीदवार को हराओ।
  \item समझाओ कि इस खेल में तुमसे \emph{कोई} क्यों नहीं जीत
        सकता: टॉम $3$ से बड़ी जो भी संख्या सुझाए, प्रश्न 5 की
        ज़ूम विधि $3$ और उसके सुझाव के ठीक बीच की एक संख्या बना
        देती है। यह ``$3$ के ठीक बाद वाली संख्या'' के बारे में
        क्या सिद्ध करता है?
  \item अब दूसरा पहलू: टॉम $3$ से ठीक \emph{पहले} वाली संख्या
        खोजता है और $2.9$, फिर $2.99$, फिर $2.999$ आज़माता है।
        उसके तीनों उम्मीदवारों को हराओ। फिर $3 - 2.999$ निकालो,
        और (बिना खानों में गणना किए) बताओ कि $3$ और उस संख्या का
        \omterm{ex:g1:subtraction:difference}{अंतर} क्या होगा जो एक $2$, एक बिंदु और बीस अंक $9$ से
        लिखी गई हो।
  \item \emph{पूर्ण} संख्याओं के लिए यह सब क्यों नहीं चलता?
        एक-दो वाक्यों में समझाओ कि $7$ और $8$ के ठीक बीच कोई
        पूर्ण संख्या क्यों नहीं है, जबकि वहाँ ढेरों \omterm{def:g6:decimals:places}{दशमलव
        संख्याएँ} हैं।
  \item कोई लंबाई ``$3.7$ cm, दसवें भाग तक गोल की हुई''
        बताई जाती है (\cref{def:g6:decimals:rounding})। वह सबसे
        छोटी लंबाई बताओ जो गोल होकर $3.7$ cm बनती है, और समझाओ
        कि $3.75$ cm की लंबाई गोल होकर $3.7$ \emph{नहीं} बनती
        --- यानी असली लंबाई कम से कम $3.65$ cm है और $3.75$ cm
        से ठीक कम।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- अंकों और बिंदुओं के खेल।}
\begin{enumerate}[resume]
  \item सबसे छोटी से सबसे बड़ी के क्रम में लगाओ: $0.6$; \ $0.58$; \
        $0.123$; \ $0.0999$। फिर किसी सहपाठी को एक वाक्य में
        समझाओ कि $0.0999 < 0.6$ क्यों है, जबकि वह ज़्यादा अंकों
        से लिखी गई है (\cref{met:g6:decimals:compare})।
  \item किसी दुकान में क़ीमतों के हमेशा ठीक दो दशमलव अंक होते
        हैं। क्या $4.99$ और $5$ के ठीक बीच कोई \emph{क़ीमत} है?
        प्रश्न 9 से तुलना करो: क़ीमतों और पूर्ण संख्याओं में
        क्या समान है?
  \item अंकों $2$, $5$, $8$ में से हर एक को ठीक एक बार और एक
        \omterm{def:g4:decimals:point}{दशमलव बिंदु} लेकर (पूर्ण भाग ख़ाली नहीं, जैसे
        \cref{exo:g6:decimals:12} में) $6$ के सबसे नज़दीक वाली
        संख्या ढूँढ़ो। अपने सबसे अच्छे उम्मीदवारों की $6$ से दूरी
        निकालकर उत्तर पक्का करो।
  \item गणित की सबसे मशहूर संख्या $\pi$ के लिए
        $3.141 < \pi < 3.142$ है। $3.141$ और $3.142$ के ठीक बीच
        की कोई \omterm{def:g6:decimals:places}{दशमलव संख्या} बताओ; फिर $3.1415$ और $3.1416$ के
        ठीक बीच की एक। (गणितज्ञ $\pi$ को ठीक इसी तरह कसते हैं
        --- हर नया दशमलव अंक एक और ज़ूम है। $\pi$ को काम करते हुए
        तुम \cref{ch:g6:measure} में देखोगे।)
  \item आख़िरी सवाल: समझाओ कि $0$ और $1$ के बीच \emph{तुम्हारी
        बताई किसी भी संख्या से ज़्यादा \omterm{def:g6:decimals:places}{दशमलव संख्याएँ}} क्यों
        हैं। (एक ज़ूम कितनी बनाता है? क्या ज़ूम कभी रुक सकता है?)
\end{enumerate}
\end{problem}
