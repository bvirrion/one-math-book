\chapter{भिन्नें: पहले कदम}\label{ch:g6:fractions}

तीन पिज़्ज़े चार लोगों में बराबर बाँटो तो हर एक को\,\dots\ पूरी
संख्या में पिज़्ज़े नहीं मिलते। भिन्नें बाँटने के लिए ही गढ़ी गई
संख्याएँ हैं। यह अध्याय पूरा चित्र बनाता है: \omterm{def:g6:fractions:def}{भिन्न} किसी पूरे के
हिस्से के रूप में, संख्या रेखा पर एक बिंदु के रूप में, और किसी
भाग के ठीक-ठीक नतीजे के रूप में।

\section{भिन्न का मतलब}

\begin{definition}[भिन्न]\label{def:g6:fractions:def}
किसी इकाई को $b$ बराबर हिस्सों में काटो और उनमें से $a$ हिस्से
लो: जो राशि मिलती है वही \emph{भिन्न}\index{भिन्न} है
\[
\frac{a}{b}
\qquad
\begin{array}{l}
a \text{: \emph{अंश} --- हम कितने हिस्से लेते हैं;}\\
b \text{: \emph{हर} --- इकाई कितने हिस्सों में काटी गई है।}
\end{array}
\]
\end{definition}

\begin{example}\label{ex:g6:fractions:morethanone}
\omterm{def:g6:fractions:def}{भिन्न} $1$ से बड़ी भी हो सकती है: $\frac74$ का मतलब है सात \omterm{def:g3:division:half}{चौथाई}
--- एक पूरी इकाई (चार \omterm{def:g3:division:half}{चौथाई}) और तीन \omterm{def:g3:division:half}{चौथाई} और:
\[
\frac74 = 1 + \frac34 .
\]
\end{example}

\begin{proposition}[संख्या रेखा पर भिन्नें]\label{prop:g6:fractions:line}
संख्या रेखा पर $\frac ab$ रखने के लिए हर इकाई अंतराल को $b$
बराबर हिस्सों में काटो और $0$ से $a$ हिस्से गिनो।
\end{proposition}

\begin{theorem}[भिन्न एक भाग है]\label{thm:g6:fractions:quotient}
\omterm{def:g6:fractions:def}{भिन्न} $\frac ab$ वह संख्या है जिसे $b$ से गुणा करने पर $a$
मिलता है:
\[
\frac ab \times b = a .
\]
दूसरे शब्दों में, $\frac ab$, $a$ में $b$ का ठीक-ठीक \omterm{def:g3:division:remainder}{भागफल}
है --- तब भी, जब भाग ``पूरा नहीं बँटता''।
\end{theorem}
\admitted

\begin{example}\label{ex:g6:fractions:pizzas}
तीन पिज़्ज़े चार लोगों के लिए: हर एक को पिज़्ज़े का $\frac34$
मिलता है। प्रमेय से जाँचो: $\frac34$ के $4$ हिस्से मिलकर
$\frac34 \times 4 = 3$ पिज़्ज़े बनाते हैं। कुछ भिन्नें \omterm{def:g6:decimals:places}{दशमलव
संख्याएँ} होती हैं ($\frac34 = 0.75$), कुछ नहीं: $\frac13 = 0.333\dots$
कभी नहीं रुकता। \omterm{def:g6:fractions:def}{भिन्न} ही \emph{ठीक} मान है; $0.33$ केवल एक सन्निकटन है।
\end{example}

\section{बराबर भिन्नें}

\begin{proposition}[तुल्य भिन्नें]\label{prop:g6:fractions:equal}
किसी \omterm{def:g6:fractions:def}{भिन्न} के \omterm{def:g4:fractions:def}{अंश} और हर, दोनों को एक ही शून्येतर संख्या से गुणा
(या भाग) करने पर \omterm{def:g6:fractions:def}{भिन्न} नहीं बदलती:
\[
\frac{a}{b} = \frac{a \times k}{b \times k} .
\]
\end{proposition}

\begin{proof}[उपपत्ति का विचार]
बँटवारे के हर हिस्से को $k$ छोटे टुकड़ों में काटने से लिए गए
टुकड़ों की संख्या और कुल टुकड़ों की संख्या, दोनों $k$ गुनी हो
जाती हैं, पर ली गई राशि वही रहती है। चित्र सबसे अच्छा कहता है:
\end{proof}

\end{omfigure}

\begin{method}[भिन्न को सरल बनाना]\label{met:g6:fractions:simplify}
कोई ऐसी संख्या ढूँढ़ो जो \omterm{def:g4:fractions:def}{अंश} \emph{और} हर, दोनों में भाग दे, और
दोनों में भाग दे दो। तब तक दोहराओ जब तक कोई साझा \omterm{def:g5:division:multiple}{भाजक} न बचे:
\[
\frac{12}{18} = \frac{12 \div 2}{18 \div 2} = \frac{6}{9}
= \frac{6 \div 3}{9 \div 3} = \frac{2}{3}.
\]
\end{method}

\section{किसी राशि की भिन्न लेना}

\begin{method}[राशि की भिन्न]\label{met:g6:fractions:ofquantity}
किसी राशि का $\frac ab$ निकालने के लिए राशि में $b$ का भाग दो
(एक हिस्सा), फिर $a$ से गुणा करो (हिस्सों की संख्या):
\[
\frac ab \text{ का } Q = (Q \div b) \times a .
\]
(पहले $a$ से गुणा करके फिर $b$ का भाग देने पर भी वही नतीजा मिलता
है --- वही क्रम चुनो जिसमें संख्याएँ आसान पड़ें।)
\end{method}

\begin{example}\label{ex:g6:fractions:ofquantity}
$40$ यूरो का $\frac35$ निकालो, कदम-दर-कदम:
\begin{enumerate}
  \item $40$ का एक पाँचवाँ भाग: $40 \div 5 = 8$;
  \item तीन पाँचवें भाग: $8 \times 3 = 24$।
\end{enumerate}
तो $40$ यूरो का $\frac35$, $24$ यूरो है। जाँच: बचा हुआ
$\frac25$, $16$ यूरो है, और $24 + 16 = 40$।
\end{example}

\section{समस्या: वह चॉकलेट जो कभी ख़त्म ही नहीं होती}

\begin{problem}[{सप्ताहांत समस्या --- आधों के आधे: योगफल
$\frac12 + \frac14 + \frac18 + \dots$ बिना छुए $1$ की ओर सरकता
रहता है}]
\label{pb:g6:fractions:1}
एक चॉकलेट लो। उसका \omterm{def:g3:division:half}{आधा} खा जाओ। फिर \emph{जो बचा है} उसका \omterm{def:g3:division:half}{आधा}
खाओ। फिर जो बचा उसका \omterm{def:g3:division:half}{आधा}, और ऐसे ही आगे। दो बातें एक साथ सच
लगती हैं: चॉकलेट कभी ख़त्म नहीं होती (कुछ न कुछ हमेशा बचा रहता
है), और फिर भी लगभग सारी खा ली जाती है। यह समस्या दोनों अहसासों
को भिन्नों के ठीक-ठीक कथनों में बदल देती है --- और रास्ते में
प्राचीन यूनान के एक धावक तथा ऐसी बाँट-पहेली से मिलाती है जो केवल
आधे-आधे काटने वाले चाक़ू से हल हो जाती है।

\medskip
\noindent\textbf{भाग I --- आधों के आधे।}
चॉकलेट $16$ बराबर ख़ानों का एक आयत है।
\begin{enumerate}
  \item पहला कौर: तुम चॉकलेट का \omterm{def:g3:division:half}{आधा} खाते हो। वह कितने ख़ाने
        हुए? इस कौर को चॉकलेट की \omterm{def:g6:fractions:def}{भिन्न} के रूप में दो तरह से
        लिखो, हर $2$ के साथ और हर $16$ के साथ
        (\cref{prop:g6:fractions:equal})।
  \item दूसरा कौर: जो बचा उसका \omterm{def:g3:division:half}{आधा}। कितने ख़ाने? चित्र से या
        तुल्य भिन्नों से दिखाओ कि यह कौर पूरी चॉकलेट का
        $\frac14$ है: \emph{आधे का \omterm{def:g3:division:half}{आधा} एक \omterm{def:g3:division:half}{चौथाई} होता है}।
  \item तीसरा और चौथा कौर, वही नियम: हर एक को ख़ानों में और
        चॉकलेट की \omterm{def:g6:fractions:def}{भिन्न} के रूप में बताओ। एक कौर से अगले कौर तक
        हर का क्या होता है?
  \item चारों कौरों के बाद कुल कितने ख़ाने खाए गए? वह चॉकलेट की
        कौन-सी \omterm{def:g6:fractions:def}{भिन्न} है, और कौन-सी \omterm{def:g6:fractions:def}{भिन्न} बची है?
  \item प्रश्न 4 सोलहवें भागों में लिखने पर कहता है:
        \[
        \frac{8}{16} + \frac{4}{16} + \frac{2}{16} + \frac{1}{16}
        = \frac{15}{16},
        \qquad\text{यानी}\qquad
        \frac12 + \frac14 + \frac18 + \frac{1}{16}
        = \frac{15}{16} .
        \]
        चारों रूपांतरणों की जाँच करो
        ($\frac12 = \frac{8}{16}$, वग़ैरह)।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- $1$ की ओर जाती सीढ़ी।}
कौर लेते रहने के लिए $64$ ख़ानों वाली महीन चॉकलेट सोचो।
\begin{enumerate}[resume]
  \item पहले छह कौरों के नाप ख़ानों में गिनाओ ($32$,
        फिर\,\dots), और जाँचो कि छठे कौर के बाद ठीक एक ख़ाना
        बचता है। चॉकलेट की कौन-सी \omterm{def:g6:fractions:def}{भिन्न} बची रहती है?
  \item हर कौर के बाद चॉकलेट की कौन-सी \omterm{def:g6:fractions:def}{भिन्न} बची रहती है? छठे
        कौर तक की सूची लिखो ($\frac12$, $\frac14$, \dots), और
        प्रतिरूप बताओ। कोई काल्पनिक चाक़ू आख़िरी ख़ाने को भी \omterm{def:g3:division:half}{आधा}
        कर सके, तो सातवें कौर के बाद क्या बचेगा?
  \item समझाओ कि खाया हुआ कुल $1$ तक \emph{कभी} क्यों नहीं
        पहुँच सकता, चाहे कितने भी कौर लिए जाएँ।
        (\cref{pb:g6:decimals:1} के बीस नौ से तुलना करो: वही
        कहानी?)
  \item फिर भी कुल $1$ से नीचे के हर लक्ष्य को पार कर जाता है।
        एक दोस्त तुम्हें चुनौती देता है कि चॉकलेट का
        $\frac{99}{100}$ से ज़्यादा खा जाओ। कौन छोटी है,
        $\frac{1}{128}$ या $\frac{1}{100}$ --- और किस कौर के बाद
        चुनौती जीत ली जाती है?
  \item $0$ से $1$ तक सोलहवें भागों में अंकित संख्या रेखा पर
        पहले चार कौरों के बाद के कुल रखो: $\frac12$, $\frac34$,
        $\frac78$, $\frac{15}{16}$। हर नया कुल पिछले कुल और $1$
        की तुलना में कहाँ गिरता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- ज़ेनो का धावक, और केवल \omterm{def:g3:division:half}{आधा} करने
वाला चाक़ू।}
\begin{enumerate}[resume]
  \item चौबीस सदी पहले यूनानी दार्शनिक ज़ेनो ने यह कहानी सुनाई:
        दीवार तक पहुँचने के लिए धावक को पहले आधी दूरी तय करनी
        होगी, फिर बची हुई दूरी की आधी, फिर उसकी भी
        आधी\,\dots\ ``तो धावक दीवार तक कभी पहुँचेगा ही नहीं!''
        प्रश्न 8 और 9 की मदद से बताओ कि ज़ेनो की कहानी में
        \emph{सही} क्या है, और जाल कहाँ है। (क्या धावक को वे सब
        कदम उठाने में सचमुच अनंत समय चाहिए?)
  \item बाँट-पहेली: $15$ एक जैसी चॉकलेटें $16$ बच्चों में बराबर
        बाँटो, केवल आधे-आधे काटकर (पूरी चॉकलेट के, आधी चॉकलेट
        के, \omterm{def:g3:division:half}{चौथाई} चॉकलेट के\,\dots)। बताओ कि कैसे काटना है:
        कितनी चॉकलेटें आधों में कटेंगी, कितनी चौथाइयों में,
        कितनी आठवें भागों में, कितनी सोलहवें भागों में, ताकि हर
        बच्चे को हर नाप का एक-एक टुकड़ा मिले।
  \item अपनी बाँट जाँचो: हर बच्चे का हिस्सा भिन्नों के \omterm{def:g1:addition:def}{योगफल} के
        रूप में लिखो, उसे सोलहवें भागों में निकालो, और पक्का करो
        कि $16$ हिस्से मिलकर ठीक $15$ चॉकलेटें ख़र्च करते हैं।
  \item बग़ल की कक्षा में $7$ चॉकलेटें $8$ बच्चों में बराबर बाँटी
        जाती हैं। किस बच्चे को ज़्यादा चॉकलेट मिलती है --- $16$
        में से किसी को, या $8$ में से किसी को? (दोनों हिस्से हर
        $16$ के साथ लिखो, \cref{exo:g6:fractions:8}।)
  \item आख़िरी सवाल: एक दोस्त का दावा है कि \emph{अनगिनत} कौरों
        से पूरी चॉकलेट \emph{ठीक-ठीक} खा ली जाएगी। एक वर्ग बनाओ
        और उसके भीतर कौर रँगो: वर्ग का \omterm{def:g3:division:half}{आधा}, फिर एक \omterm{def:g3:division:half}{चौथाई}, फिर एक
        आठवाँ भाग\,\dots\ बिना रँगे कोने के बारे में क्या दिखता
        है? इस पूरी समस्या के दोनों सच्चे तथ्य ध्यान से लिखो:
        सीमित संख्या में कौरों के बाद क्या बचता है, और वह \omterm{def:g3:division:remainder}{शेषफल}
        कितना छोटा हो जाता है। (``अनंत \omterm{def:g1:addition:def}{योगफल}'' का ठीक-ठीक मतलब
        क्या है, यह अगले खंड की कहानी है।)
\end{enumerate}
\end{problem}
