\chapter{पूर्ण संख्याएँ}\label{ch:g6:wholes}

गणित में सब कुछ गिनने से शुरू होता है। यह अध्याय दोहराता है कि
पूर्ण संख्याएँ कैसे लिखी, तुलना की और जोड़ी-गुणा की जाती हैं ---
और भाग पर ध्यान से नज़र डालता है, वह संक्रिया जो \omterm{def:g3:division:remainder}{भागफल}
\emph{और} \omterm{def:g3:division:remainder}{शेषफल}, दोनों देती है।

\section{पूर्ण संख्याएँ लिखना और उनकी तुलना}

\begin{definition}[स्थानीय मान]\label{def:g6:wholes:place}
संख्याएँ लिखने का हमारा ढंग दस अंकों ($0$ से $9$) का उपयोग करता
है, और \emph{किसी अंक का मान उसके स्थान पर निर्भर करता
है}\index{स्थानीय मान}: $5\,208$ में अंक $5$ हज़ार गिनता है, $2$
सैकड़ा, $0$ दहाई (जो है ही नहीं) और $8$ इकाई:
\[
5\,208 = 5 \times 1000 + 2 \times 100 + 0 \times 10 + 8 .
\]
\end{definition}

\begin{method}[दो पूर्ण संख्याओं की तुलना]\label{met:g6:wholes:compare}
\begin{enumerate}
  \item जिसमें ज़्यादा अंक हों, वही बड़ी है
        ($1\,203 > 989$)।
  \item अंकों की संख्या बराबर हो, तो \emph{बाईं} ओर से
        अंक-दर-अंक तुलना करो; पहला \omterm{ex:g1:subtraction:difference}{अंतर} ही फ़ैसला करता है:
        $6\,742 > 6\,715$, क्योंकि दहाई के स्थान पर $4 > 1$।
\end{enumerate}
चिह्न हैं $<$ (``से छोटी'') और $>$ (``से बड़ी''); चिह्न का छोटा
सिरा छोटी संख्या की ओर रहता है।
\end{method}

\begin{example}\label{ex:g6:wholes:order}
सबसे छोटी से सबसे बड़ी के क्रम में लगाओ: $908$; $89$; $1\,001$; $980$।
पहले अंकों की संख्या से: $89$ (दो अंक) सबसे पहले, $1\,001$ (चार
अंक) सबसे बाद। $908$ और $980$ में: सैकड़े का अंक वही $9$, फिर
दहाई $0 < 8$, इसलिए $908 < 980$। अंतिम उत्तर:
\[
89 < 908 < 980 < 1\,001 .
\]
\end{example}

\section{जोड़, घटाव, गुणा}

\begin{example}[खानों में गणना]\label{ex:g6:wholes:columns}
जोड़ने या घटाने के लिए इकाई के नीचे इकाई और दहाई के नीचे दहाई
मिलाओ, और दाएँ से बाएँ चलो, ज़रूरत पड़ने पर हासिल ले जाते हुए:
\[
\begin{array}{r}
4\,5\,7 \\
+\ 2\,8\,6 \\
\hline
7\,4\,3
\end{array}
\qquad\qquad
\begin{array}{r}
6\,0\,3 \\
-\ 1\,4\,7 \\
\hline
4\,5\,6
\end{array}
\]
जोड़ में: $7 + 6 = 13$, $3$ लिखो, $1$ हासिल; फिर
$5 + 8 + 1 = 14$, $4$ लिखो, $1$ हासिल; फिर $4 + 2 + 1 = 7$।
घटाव की जाँच वापस जोड़कर करो: $456 + 147 = 603$।
\end{example}

\begin{remark}[शब्दावली]
जोड़ का नतीजा \emph{योगफल}\index{योगफल} है; घटाव का
\emph{अंतर}\index{अंतर}; गुणा का
\emph{गुणनफल}\index{गुणनफल}; भाग का
\emph{भागफल}\index{भागफल}। जिन संख्याओं को मिलाया जाता है वे
\emph{पद} ($+$, $-$ के लिए) या \emph{गुणनखंड} ($\times$ के लिए) कहलाती हैं।
\end{remark}

\begin{example}[चतुराई से गणना]\label{ex:g6:wholes:smart}
पदों या गुणनखंडों का क्रम बदल देने से अक्सर मेहनत बच जाती है:
\[
17 + 58 + 3 = (17 + 3) + 58 = 20 + 58 = 78 ,
\]
\[
25 \times 17 \times 4 = (25 \times 4) \times 17 = 100 \times 17 = 1700 .
\]
\end{example}

\section{भाग}

\begin{theorem}[यूक्लिडीय भाग]\label{thm:g6:wholes:division}
दो पूर्ण संख्याएँ $a$ (\emph{भाज्य}) और $b \neq 0$
(\emph{\omterm{def:g5:division:multiple}{भाजक}}) दी हों, तो इन्हें ठीक एक ही तरह लिखा जा सकता है
\[
a = b \times q + r
\qquad\text{जहाँ}\qquad
0 \leq r < b .
\]
संख्या $q$ भाग का \emph{भागफल}\index{भागफल} है और $r$ उसका
\emph{शेषफल}\index{शेषफल}, जब $a$ में $b$ का भाग दिया जाए।
\end{theorem}
\admitted

\begin{example}\label{ex:g6:wholes:division}
$137$ में $4$ का भाग दो, कदम-दर-कदम।
\begin{enumerate}
  \item $13$ में $4$ कितनी बार जाता है? तीन बार
        ($4 \times 3 = 12$), \omterm{def:g3:division:remainder}{शेषफल} $13 - 12 = 1$।
  \item $7$ नीचे उतारो: $17$ में $4$ कितनी बार जाता है?
        चार बार ($4 \times 4 = 16$), \omterm{def:g3:division:remainder}{शेषफल} $1$।
\end{enumerate}
तो $q = 34$ और $r = 1$:
\[
137 = 4 \times 34 + 1, \qquad\text{और सचमुच } 0 \leq 1 < 4 .
\]
जाँच: $4 \times 34 = 136$, और $136 + 1 = 137$।
\end{example}

से छोटा होता है।}
\end{omfigure}

\begin{definition}[विभाज्य]\label{def:g6:wholes:divisible}
\omterm{def:g3:division:remainder}{शेषफल} $0$ हो, तो हम कहते हैं कि $a$, $b$ से
\emph{विभाज्य}\index{विभाज्य} है: जैसे $135$, $5$ से विभाज्य
है, क्योंकि $135 = 5 \times 27 + 0$।
\end{definition}

\begin{method}[भाग वाली शब्द-समस्याएँ]\label{met:g6:wholes:problems}
$a = bq + r$ निकालने के बाद हमेशा सवाल पर लौटो:
\begin{enumerate}
  \item ``कितने पूरे डिब्बे/दल/पन्ने?'' --- उत्तर है \omterm{def:g3:division:remainder}{भागफल}
        $q$;
  \item ``कितने बच गए?'' --- उत्तर है \omterm{def:g3:division:remainder}{शेषफल}
        $r$;
  \item ``सबके लिए कितने डिब्बे चाहिए?'' --- $r > 0$ हो, तो
        \omterm{def:g3:division:remainder}{भागफल} से \emph{एक ज़्यादा}: $q + 1$।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:wholes:bus}
$137$ विद्यार्थी $40$-$40$ सीटों वाली बसों से सैर पर जाते हैं।
भाग: $137 = 40 \times 3 + 17$। तीन बसें $120$ विद्यार्थियों को
ले जाती हैं, और $17$ विद्यार्थी बचते हैं --- उन्हें भी बस
चाहिए! तो $4$ बसें चाहिए, चाहे \omterm{def:g3:division:remainder}{भागफल} $3$ ही हो।
\end{example}

\section{समस्या: नन्हा गाउस और पहली सौ संख्याओं का
योगफल}

\begin{problem}[{सप्ताहांत समस्या --- दस सेकंड में $1 + 2 + 3 + \dots +
100$ जोड़ना, और त्रिभुजाकार संख्याएँ}]
\label{pb:g6:wholes:1}
एक मशहूर क़िस्सा: अपनी कक्षा को व्यस्त रखने के लिए एक मास्टर ने
विद्यार्थियों से $1$ से $100$ तक की सारी संख्याएँ जोड़ने को कहा।
एक लड़के ने लगभग तुरंत अपनी पट्टी पर $5\,050$ लिख दिया --- कार्ल
फ़्रीड्रिख़ गाउस, नौ साल का, जो आगे चलकर अब तक के सबसे बड़े
गणितज्ञों में गिना गया। उसका राज़ \emph{चतुराई से गणना} के सिवा
कुछ न था (\cref{ex:g6:wholes:smart}): पदों को ऐसे समूहों में
बाँटना कि हर समूह आसान हो जाए। यह समस्या उसकी तरकीब फिर से खोजती
है, उसे हर हाल में चलाने लायक़ बनाती है, और दिखाती है कि ऐसे \omterm{def:g1:addition:def}{योगफल}
रोज़मर्रा में कितनी बार आते हैं।

\medskip
\noindent\textbf{भाग I --- गाउस के जोड़े।}
\begin{enumerate}
  \item $1 + 2 + 3 + 4 + 5$ को पदों के चतुर समूह बनाकर निकालो।
  \item $1 + 2 + 3 + \dots + 10$ के लिए: पहले पद को आख़िरी से
        जोड़ो ($1 + 10$), दूसरे को अंत से दूसरे के साथ
        ($2 + 9$), और ऐसे ही आगे। हर जोड़ा कितने का है? जोड़े
        कितने हैं? गणना पूरी करो।
  \item $1 + 2 + \dots + 20$ के लिए वही तरीक़ा: हर जोड़ा कितने
        का, जोड़े कितने, कुल कितना?
  \item अब मास्टर वाला \omterm{def:g1:addition:def}{योगफल}, $1 + 2 + \dots + 100$: समझाओ कि
        जोड़े $1 + 100$, $2 + 99$, $3 + 98, \dots$ सब एक ही मान
        के क्यों हैं, और वे ठीक $50$ ही क्यों हैं, जिनमें
        आख़िरी $50 + 51$ है।
  \item गाउस का उत्तर एक ही गुणा के रूप में लिखो, और उसे
        निकालो।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- वह तरकीब जो कभी नहीं चूकती।}
\begin{enumerate}[resume]
  \item $1 + 2 + \dots + 9$ पर जोड़े बनाकर देखो: इस बार एक
        संख्या बिना जोड़ीदार रह जाती है। कौन-सी? उसके इर्द-गिर्द
        जोड़े बनाकर \omterm{def:g1:addition:def}{योगफल} निकालो।
  \item यहाँ उस तरकीब का एक रूप है जो पदों की संख्या सम हो या
        विषम, दोनों में चलता है: \omterm{def:g1:addition:def}{योगफल} को दो बार लिखो, एक बार
        आगे से और एक बार पीछे से, एक पंक्ति के नीचे दूसरी,
        \[
        \begin{array}{ccccccc}
        1 & + & 2 & + & \dots & + & 9 \\
        9 & + & 8 & + & \dots & + & 1
        \end{array}
        \]
        और \emph{स्तंभ-दर-स्तंभ} जोड़ो। हर स्तंभ कितने का है?
        स्तंभ कितने हैं? समझाओ कि दोनों पंक्तियों का \omterm{def:g1:addition:def}{योगफल}
        $9 \times 10 = 90$ क्यों है, और निष्कर्ष निकालो कि
        $1 + 2 + \dots + 9 = 45$ --- वही उत्तर जो प्रश्न 6 में
        था।
  \item दो पंक्तियों वाली तरकीब से $1 + 2 + \dots + 50$ निकालो,
        फिर $1 + 2 + \dots + 200$।
  \item पहली सौ \emph{सम} संख्याओं का \omterm{def:g1:addition:def}{योगफल} निकालो,
        $2 + 4 + 6 + \dots + 200$। (हर पद की तुलना गाउस के \omterm{def:g1:addition:def}{योगफल}
        के पद से करो: नए जोड़े बनाने की ज़रूरत नहीं।)
  \item $101 + 102 + \dots + 200$ निकालो, दो जाने-पहचाने योगफलों
        और एक घटाव की मदद से।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{pb:g6:wholes:1}{त्रिभुजाकार संख्याएँ} हर जगह।}
योगफलों $1 + 2 + \dots + n$ के कुल $1$, $3$, $6$, $10$, $15, \dots$
\emph{त्रिभुजाकार संख्याएँ}\index{त्रिभुजाकार संख्याएँ} कहलाते
हैं, क्योंकि वे त्रिभुज में जमी हुई वस्तुएँ गिनते हैं।
\begin{enumerate}[resume]
  \item आठ दोस्त मिलते हैं, और हर एक बाक़ी हर एक से ठीक एक बार
        हाथ मिलाता है। समझाओ कि हाथ मिलाने की संख्या
        $7 + 6 + 5 + 4 + 3 + 2 + 1$ क्यों है, और उसे निकालो।
  \item एक दुकानदार टिनों का त्रिभुजाकार \omterm{def:g5:solids:def}{पिरामिड} बनाती है: सबसे
        नीचे की पंक्ति में $12$ टिन, अगली में $11$, और ऐसे ही
        ऊपर एक अकेले टिन तक। उसे कितने टिन चाहिए? (प्रश्न 7 की
        तरकीब लगाओ।)
  \item \omterm{def:g1:addition:def}{योगफल} $1 + 2 + 3 + 4$ को वर्गों की सीढ़ी की तरह बनाओ:
        पहले स्तंभ में एक वर्ग, दूसरे में दो, तीसरे में तीन,
        चौथे में चार। अपने चित्र पर दिखाओ कि ऐसी \emph{दो}
        सीढ़ियाँ, जिनमें एक उलटी रखी हो, ठीक-ठीक $5$-$5$ वर्गों
        की $4$ पंक्तियों वाले आयत में बैठ जाती हैं --- और समझाओ
        कि यह प्रश्न 7 वाली दो पंक्तियों की तरकीब ही है, लिखी
        हुई नहीं, बनी हुई।
  \item एक घड़ी एक बजे $1$ बार, दो बजे $2$ बार, \dots, बारह बजे
        $12$ बार टनटनाती है। बारह घंटों में कुल कितनी बार? और
        पूरे दिन में?
  \item गाउस के लिए एक आख़िरी सवाल: $5\,050$ सेकंड --- यह डेढ़
        घंटे से ज़्यादा है या कम? उसे घंटों, मिनटों और सेकंडों
        में बदलो, यूक्लिडीय भाग (\cref{thm:g6:wholes:division})
        दो बार लगाकर।
\end{enumerate}
\end{problem}
