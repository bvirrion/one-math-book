\chapter{Frações: primeiros passos}\label{ch:g6:fractions}

Repartir três pizzas igualmente entre quatro pessoas dá a cada
uma\,\dots\ um número não inteiro de pizzas. As frações são os números
inventados para repartir. Este capítulo monta o quadro: a \omterm{def:g6:fractions:def}{fração} como
parte de um inteiro, como ponto da reta numérica e como resultado exato
de uma divisão.

\section{O que significa uma fração}

\begin{definition}[Fração]\label{def:g6:fractions:def}
Corte uma unidade em $b$ partes iguais e tome $a$ delas: a quantidade
obtida é a \emph{fração}\index{fração}
\[
\frac{a}{b}
\qquad
\begin{array}{l}
a \text{: o \emph{numerador} --- quantas partes tomamos;}\\
b \text{: o \emph{denominador} --- em quantas partes a unidade é cortada.}
\end{array}
\]
\end{definition}

\begin{example}\label{ex:g6:fractions:morethanone}
Uma \omterm{def:g6:fractions:def}{fração} pode ser maior que $1$: $\frac74$ quer dizer sete quartos
--- uma unidade inteira (quatro quartos) e mais três quartos:
\[
\frac74 = 1 + \frac34 .
\]
\end{example}

\begin{proposition}[Frações na reta numérica]\label{prop:g6:fractions:line}
Para marcar $\frac ab$ na reta numérica, corte cada intervalo unitário
em $b$ partes iguais e conte $a$ partes a partir do $0$.
\end{proposition}

\begin{theorem}[Uma fração é uma divisão]\label{thm:g6:fractions:quotient}
A \omterm{def:g6:fractions:def}{fração} $\frac ab$ é o número que, multiplicado por $b$, dá $a$:
\[
\frac ab \times b = a .
\]
Em outras palavras, $\frac ab$ é o \omterm{def:g3:division:remainder}{quociente} exato de $a$ por $b$ ---
mesmo quando a divisão ``não é exata''.
\end{theorem}
\admitted

\begin{example}\label{ex:g6:fractions:pizzas}
Três pizzas para quatro pessoas: cada pessoa recebe $\frac34$ de pizza.
Confira com o teorema: $4$ partes de $\frac34$ dão
$\frac34 \times 4 = 3$ pizzas. Algumas frações são \omterm{def:g6:decimals:places}{números decimais}
($\frac34 = 0.75$), outras não: $\frac13 = 0.333\dots$ nunca para. A
\omterm{def:g6:fractions:def}{fração} é o valor \emph{exato}; $0.33$ é apenas uma aproximação.
\end{example}

\section{Frações iguais}

\begin{proposition}[Frações equivalentes]\label{prop:g6:fractions:equal}
Uma \omterm{def:g6:fractions:def}{fração} não muda quando o \omterm{def:g4:fractions:def}{numerador} e o \omterm{def:g4:fractions:def}{denominador} dela são ambos
multiplicados (ou divididos) pelo mesmo número não nulo:
\[
\frac{a}{b} = \frac{a \times k}{b \times k} .
\]
\end{proposition}

\begin{proof}[Ideia da demonstração]
Cortar cada parte da partilha em $k$ pedacinhos multiplica por $k$ tanto
o número de pedaços tomados quanto o número total de pedaços, mas a
quantidade tomada é a mesma. Um desenho diz isso melhor:
\end{proof}

\begin{method}[Simplificar uma fração]\label{met:g6:fractions:simplify}
Procure um número que divida o \omterm{def:g4:fractions:def}{numerador} \emph{e} o \omterm{def:g4:fractions:def}{denominador}, e
divida os dois. Repita até não sobrar nenhum \omterm{def:g5:division:multiple}{divisor} comum:
\[
\frac{12}{18} = \frac{12 \div 2}{18 \div 2} = \frac{6}{9}
= \frac{6 \div 3}{9 \div 3} = \frac{2}{3}.
\]
\end{method}

\section{Tomar uma fração de uma quantidade}

\begin{method}[Fração de uma quantidade]\label{met:g6:fractions:ofquantity}
Para calcular $\frac ab$ de uma quantidade, divida a quantidade por $b$
(uma parte) e depois multiplique por $a$ (o número de partes):
\[
\frac ab \text{ de } Q = (Q \div b) \times a .
\]
(Multiplicar primeiro por $a$ e depois dividir por $b$ dá o mesmo
resultado --- escolha a ordem que deixa os números mais simpáticos.)
\end{method}

\begin{example}\label{ex:g6:fractions:ofquantity}
Vamos calcular $\frac35$ de $40$ reais, passo a passo:
\begin{enumerate}
  \item um quinto de $40$: $40 \div 5 = 8$;
  \item três quintos: $8 \times 3 = 24$.
\end{enumerate}
Logo $\frac35$ de $40$ reais são $24$ reais. Confere: os $\frac25$ que
sobram são $16$ reais, e $24 + 16 = 40$.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g6:fractions:1}
Para cada descrição, escreva a \omterm{def:g6:fractions:def}{fração}: um bolo cortado em $8$ fatias,
$3$ comidas; uma barra de chocolate de $12$ quadradinhos, $7$ comidos;
uma torta cortada em $6$, todas as $6$ fatias comidas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:2}
Desenhe uma barra cortada em $5$ partes iguais e pinte $\frac45$ dela.
$\frac45$ é menor ou maior que $1$? E $\frac65$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:3}
Marque numa reta numérica graduada em terços: $\frac13$; \ $\frac53$;
\ $2$; \ $\frac73$. Quais dois desses números estão à mesma distância
do $2$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:4}
Escreva cada \omterm{def:g6:fractions:def}{fração} como um número inteiro mais uma \omterm{def:g6:fractions:def}{fração} menor que
$1$ (como em $\frac74 = 1 + \frac34$):
\[
\frac{9}{4}, \qquad \frac{13}{5}, \qquad \frac{12}{6}, \qquad
\frac{25}{8} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:5}
Complete de modo que as frações fiquem iguais:
\[
\frac{2}{3} = \frac{?}{12}, \qquad
\frac{15}{20} = \frac{3}{?}, \qquad
\frac{7}{10} = \frac{21}{?}, \qquad
\frac{?}{6} = \frac{10}{12}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:6}
Simplifique o máximo possível: $\dfrac{6}{8}$; \quad $\dfrac{15}{25}$;
\quad $\dfrac{18}{24}$; \quad $\dfrac{35}{7}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:7}
Calcule: $\frac12$ de $86$; \quad $\frac34$ de $60$; \quad $\frac25$ de
$35$; \quad $\frac{7}{10}$ de $250$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:8}
Qual é maior: $\frac12$ ou $\frac13$ (do mesmo bolo)? $\frac25$ ou
$\frac35$? $\frac34$ ou $\frac78$ (escreva as duas com \omterm{def:g4:fractions:def}{denominador}
$8$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:fractions:9}
Dê o valor decimal das frações que têm um, e diga qual \omterm{def:g6:fractions:def}{fração} não tem:
$\frac12$; \ $\frac34$; \ $\frac13$; \ $\frac{7}{10}$; \ $\frac{9}{4}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:fractions:10}
Uma turma tem $28$ alunos. Três quartos deles vêm para a escola de
ônibus e o restante vem a pé. Quantos alunos vêm a pé? Resolva passo a
passo e confira que os tamanhos dos dois grupos somam $28$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:fractions:11}
Léo gastou $\frac23$ das economias dele num jogo que custou $18$ reais.
Quanto dinheiro Léo tinha antes? (Um terço das economias é a \omterm{def:g3:division:half}{metade} de
$18$\,\dots\ pense por quê, ou descubra primeiro o tamanho de uma
parte.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:fractions:12}
Um tanque está cheio até $\frac58$ da \omterm{def:g2:measure:units}{capacidade} dele. Depois de usar
$10$ litros, ele fica pela \omterm{def:g3:division:half}{metade}. Qual é a \omterm{def:g2:measure:units}{capacidade} do tanque? (Que
\omterm{def:g6:fractions:def}{fração} do tanque os $10$ litros representam?)
\end{exercise}

\section{Problema: a barra de chocolate que nunca acaba}

\begin{problem}[{Problema de fim de semana --- metades de metades: a soma
$\frac12 + \frac14 + \frac18 + \dots$ se aproxima de $1$ sem nunca
tocá-lo}]
\label{pb:g6:fractions:1}
Pegue uma barra de chocolate. Coma \omterm{def:g3:division:half}{metade} dela. Depois coma \omterm{def:g3:division:half}{metade}
\emph{do que sobrou}. Depois de novo \omterm{def:g3:division:half}{metade} do que sobrou, e assim por
diante. Duas coisas parecem ser verdade ao mesmo tempo: você nunca
termina a barra (sempre sobra algo) e, ainda assim, come quase tudo.
Este problema transforma as duas sensações em afirmações exatas sobre
frações --- e encontra, pelo caminho, um corredor da Grécia antiga e um
quebra-cabeça de partilha resolvido só com uma faca que corta ao meio.

\medskip
\noindent\textbf{Parte I --- \omterm{def:g3:division:half}{Metades} de \omterm{def:g3:division:half}{metades}.}
A barra é um retângulo de $16$ quadradinhos iguais.
\begin{enumerate}
  \item Primeira mordida: você come \omterm{def:g3:division:half}{metade} da barra. Quantos
        quadradinhos são? Escreva a mordida como \omterm{def:g6:fractions:def}{fração} da barra de
        duas maneiras, com \omterm{def:g4:fractions:def}{denominador} $2$ e com \omterm{def:g4:fractions:def}{denominador} $16$
        (\cref{prop:g6:fractions:equal}).
  \item Segunda mordida: \omterm{def:g3:division:half}{metade} do que sobrou. Quantos quadradinhos?
        Mostre, com o desenho ou com frações equivalentes, que essa
        mordida é $\frac14$ da barra inteira: \emph{\omterm{def:g3:division:half}{metade} de uma
        \omterm{def:g3:division:half}{metade} é um quarto}.
  \item Terceira e quarta mordidas, com a mesma regra: dê cada uma em
        quadradinhos e como \omterm{def:g6:fractions:def}{fração} da barra. O que acontece com o
        \omterm{def:g4:fractions:def}{denominador} de uma mordida para a seguinte?
  \item Depois das quatro mordidas, quantos quadradinhos foram comidos
        ao todo? Que \omterm{def:g6:fractions:def}{fração} da barra é isso, e que \omterm{def:g6:fractions:def}{fração} sobrou?
  \item A questão 4 diz, escrita em dezesseis avos:
        \[
        \frac{8}{16} + \frac{4}{16} + \frac{2}{16} + \frac{1}{16}
        = \frac{15}{16},
        \qquad\text{ou seja}\qquad
        \frac12 + \frac14 + \frac18 + \frac{1}{16}
        = \frac{15}{16} .
        \]
        Confira cada uma das quatro reescritas
        ($\frac12 = \frac{8}{16}$, e assim por diante).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A escada rumo a $1$.}
Para continuar mordendo, imagine uma barra mais fina, de $64$
quadradinhos.
\begin{enumerate}[resume]
  \item Liste os tamanhos das seis primeiras mordidas, em quadradinhos
        ($32$, depois\,\dots), e confira que, depois da sexta mordida,
        sobra exatamente um quadradinho. Que \omterm{def:g6:fractions:def}{fração} da barra sobra?
  \item Depois de cada mordida, que \omterm{def:g6:fractions:def}{fração} da barra sobra? Escreva a
        lista ($\frac12$, $\frac14$, \dots) até a sexta mordida e
        descreva o padrão. Se uma faca imaginária pudesse cortar ao
        meio até o último quadradinho, o que sobraria depois de uma
        sétima mordida?
  \item Explique por que o total comido \emph{nunca} pode chegar a $1$,
        por mais mordidas que se dê. (Compare com os vinte noves do
        \cref{pb:g6:decimals:1}: é a mesma história?)
  \item Ainda assim, o total ultrapassa qualquer alvo abaixo de $1$. Um
        amigo desafia você a comer mais que $\frac{99}{100}$ da barra.
        Qual é menor, $\frac{1}{128}$ ou $\frac{1}{100}$ --- e depois
        de que mordida o desafio está ganho?
  \item Numa reta numérica de $0$ a $1$ graduada em dezesseis avos,
        marque os totais comidos depois de cada uma das quatro
        primeiras mordidas: $\frac12$, $\frac34$, $\frac78$,
        $\frac{15}{16}$. Onde cai cada novo total, em relação ao
        anterior e ao $1$?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O corredor de Zenão e uma faca que só
corta ao meio.}
\begin{enumerate}[resume]
  \item Vinte e quatro séculos atrás, o filósofo grego Zenão contou
        esta história: para chegar a um muro, um corredor precisa antes
        percorrer \omterm{def:g3:division:half}{metade} da distância, depois \omterm{def:g3:division:half}{metade} do que resta,
        depois de novo \omterm{def:g3:division:half}{metade} do que resta\,\dots\ ``e assim o corredor
        nunca chega ao muro!'' Usando as questões 8 e 9, diga o que há
        de \emph{certo} na história de Zenão e onde está a armadilha. (O
        corredor precisa mesmo de uma eternidade para dar todos esses
        passos?)
  \item O quebra-cabeça da partilha: reparta igualmente $15$ barras de
        chocolate idênticas entre $16$ crianças, usando apenas cortes ao
        meio (de uma barra, de meia barra, de um quarto de
        barra\,\dots). Descreva como cortar: quantas barras são
        cortadas ao meio, quantas em quartos, em oitavos, em dezesseis
        avos, de modo que cada criança receba um pedaço de cada
        tamanho.
  \item Confira a sua partilha: escreva a parte de cada criança como
        \omterm{def:g1:addition:def}{soma} de frações, calcule-a em dezesseis avos e verifique que as
        $16$ partes juntas usam exatamente as $15$ barras.
  \item Na turma ao lado, $7$ barras são repartidas igualmente entre
        $8$ crianças. Que criança recebe mais chocolate --- uma das
        $16$ ou uma das $8$? (Escreva as duas partes com \omterm{def:g4:fractions:def}{denominador}
        $16$, \cref{exo:g6:fractions:8}.)
  \item O grande final: um amigo afirma que, com \emph{infinitas}
        mordidas, comeria-se \emph{exatamente} a barra inteira. Desenhe
        um quadrado e pinte as mordidas dentro dele: \omterm{def:g3:division:half}{metade} do
        quadrado, depois um quarto, depois um oitavo\,\dots\ O que você
        observa no canto sem pintar? Enuncie com cuidado os dois fatos
        verdadeiros deste problema inteiro: o que sobra depois de
        qualquer quantidade finita de mordidas e quão pequeno esse
        \omterm{def:g3:division:remainder}{resto} fica. (O que uma ``\omterm{def:g1:addition:def}{soma} infinita'' quer dizer exatamente
        é assunto do volume do ensino médio.)
\end{enumerate}
\end{problem}
