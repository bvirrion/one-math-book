\chapter{Simetria axial}\label{ch:g6:symmetry}

Dobre uma folha de papel ao longo de uma reta: cada ponto cai sobre
outro ponto, a imagem espelhada dele. Essa dobra é a \emph{\omterm{def:g6:symmetry:def}{simetria
axial}} (ou reflexão), a primeira transformação geométrica do livro ---
a segunda, a simetria central, vem no \cref{ch:g7:central}.

\section{Reflexões}

\begin{definition}[Ponto simétrico]\label{def:g6:symmetry:def}
Seja $d$ uma reta. O \emph{simétrico}\index{simetria axial} de um ponto
$M$ em relação a $d$ é o ponto $M'$ tal que $d$ é a \omterm{def:g6:symmetry:bisector}{mediatriz} de
$[MM']$: o \omterm{def:g6:lines:objects}{segmento} $[MM']$ é perpendicular a $d$, e $d$ o corta no
ponto médio. Um ponto \emph{sobre} $d$ é o simétrico de si mesmo.
\end{definition}

\begin{method}[Construir um ponto simétrico]\label{met:g6:symmetry:construct}
Para construir o \omterm{def:g6:symmetry:def}{simétrico} $M'$ de $M$ em relação à reta $d$:
\begin{enumerate}
  \item trace a perpendicular a $d$ que passa por $M$ (esquadro); chame
        de $H$ o ponto em que ela cruza $d$;
  \item meça $MH$ com o compasso ou com a régua;
  \item marque $M'$ sobre a mesma perpendicular, do outro lado de $d$,
        com $HM' = MH$.
\end{enumerate}
No papel quadriculado é mais rápido: conte os quadradinhos de $M$ até o
eixo e conte a mesma quantidade do outro lado.
\end{method}

\begin{example}[Na malha quadriculada]\label{ex:g6:symmetry:grid}
Se o eixo for uma linha vertical da malha e $M$ estiver $3$
quadradinhos à esquerda dela, então $M'$ fica $3$ quadradinhos à
direita, na mesma linha. Para um eixo diagonal, a contagem funciona ao
longo das diagonais --- sempre perpendicularmente ao eixo.
\end{example}

\begin{proposition}[O que uma reflexão preserva]\label{prop:g6:symmetry:preserved}
O \omterm{def:g6:symmetry:def}{simétrico} de uma figura em relação a uma reta é uma figura da
\emph{mesma forma e do mesmo tamanho}: comprimentos, ângulos,
\omterm{def:g4:measure:perimeter}{perímetros} e \omterm{def:g4:measure:area}{áreas} são todos preservados. O \omterm{def:g6:symmetry:def}{simétrico} de um \omterm{def:g6:lines:objects}{segmento} é
um \omterm{def:g6:lines:objects}{segmento}; o de um círculo, um círculo de mesmo \omterm{def:g3:shapes:shapes}{raio}; o de uma reta,
uma reta.
\end{proposition}
\admitted

\begin{remark}
Uma coisa \emph{não} é preservada: a orientação. Um ``b'' refletido
vira um ``d'' --- escrita espelhada. Se decalcar a sua figura e virar o
papel vegetal reproduz a imagem, a reflexão está correta.
\end{remark}

\section{Eixos de simetria de uma figura}

\begin{definition}[Eixo de simetria]\label{def:g6:symmetry:axis}
Uma reta $d$ é um \emph{eixo de simetria}\index{eixo de simetria} de
uma figura quando a reflexão em relação a $d$ leva a figura exatamente
sobre ela mesma --- dobrar ao longo de $d$ faz as duas \omterm{def:g3:division:half}{metades}
coincidirem.
\end{definition}

quadrado).}

\begin{example}\label{ex:g6:symmetry:shapes}
Contando \omterm{def:g6:symmetry:axis}{eixos de simetria}:
\begin{itemize}
  \item \omterm{def:g6:shapes:triangles}{triângulo isósceles}: $1$ (pelo \omterm{def:g5:solids:def}{vértice} do topo e pelo meio da
        base); \omterm{def:g6:shapes:triangles}{triângulo equilátero}: $3$;
  \item retângulo: $2$ (as duas ``linhas médias'', \emph{não} as
        diagonais); \omterm{def:g6:shapes:quadrilaterals}{losango}: $2$ (as diagonais dele!); quadrado: $4$;
  \item círculo: toda reta que passa pelo centro --- infinitos.
\end{itemize}
\end{example}

\begin{example}[A simetria demonstra igualdades]\label{ex:g6:symmetry:proves}
Por que os dois ângulos da base de um \omterm{def:g6:shapes:triangles}{triângulo isósceles} são iguais?
Dobre o triângulo ao longo do \omterm{def:g6:symmetry:axis}{eixo de simetria} dele: a \omterm{def:g3:division:half}{metade} esquerda
cai exatamente sobre a direita, então o ângulo da base à esquerda cai
sobre o ângulo da base à direita --- eles têm de ter a mesma medida
(\cref{prop:g6:symmetry:preserved}: as reflexões preservam ângulos).
\end{example}

\section{Mediatriz}

\begin{definition}[Mediatriz]\label{def:g6:symmetry:bisector}
A \emph{mediatriz}\index{mediatriz} de um \omterm{def:g6:lines:objects}{segmento} $[AB]$ é a reta
perpendicular a $[AB]$ que passa pelo ponto médio dele --- é o \omterm{def:g6:symmetry:axis}{eixo de
simetria} que troca $A$ e $B$.
\end{definition}

\begin{proposition}[Equidistância]\label{prop:g6:symmetry:bisector}
Um ponto $M$ está sobre a \omterm{def:g6:symmetry:bisector}{mediatriz} de $[AB]$ exatamente quando está à
mesma distância de $A$ e de $B$: $MA = MB$.
\end{proposition}
\admitted

\begin{method}[Construção com o compasso]\label{met:g6:symmetry:bisector}
Para traçar a \omterm{def:g6:symmetry:bisector}{mediatriz} de $[AB]$ sem medir:
\begin{enumerate}
  \item abra o compasso mais que a \omterm{def:g3:division:half}{metade} de $AB$;
  \item trace dois arcos com centro em $A$, acima e abaixo do
        \omterm{def:g6:lines:objects}{segmento};
  \item com a \emph{mesma} abertura, trace dois arcos com centro em
        $B$;
  \item ligue os dois pontos de cruzamento: essa reta é a \omterm{def:g6:symmetry:bisector}{mediatriz}
        (cada ponto de cruzamento é equidistante de $A$ e de $B$).
\end{enumerate}
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:1}
Copie no papel quadriculado: um eixo vertical $d$, um ponto $A$ dois
quadradinhos à esquerda dele e um ponto $B$ cinco quadradinhos à
esquerda e um quadradinho acima. Construa os \omterm{def:g6:symmetry:def}{simétricos} $A'$ e $B'$
deles em relação a $d$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:2}
Trace uma reta $d$ e um ponto $M$ a $3$ cm de $d$ (fora do papel
quadriculado). Construa o \omterm{def:g6:symmetry:def}{simétrico} $M'$ de $M$ em relação a $d$ com
esquadro e régua, seguindo o \cref{met:g6:symmetry:construct}. Que
distância separa $M$ e $M'$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:3}
Que letras maiúsculas do alfabeto, escritas na forma mais simples, têm
\omterm{def:g6:symmetry:axis}{eixo de simetria} vertical (como o A)? E horizontal (como o B)? E os
dois?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:4}
Quantos \omterm{def:g6:symmetry:axis}{eixos de simetria} tem: um \omterm{def:g6:shapes:triangles}{triângulo equilátero}? um retângulo
(que não seja quadrado)? um \omterm{def:g6:shapes:quadrilaterals}{losango} (não quadrado)? um quadrado? um
círculo? Trace cada figura com os eixos dela.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:5}
Verdadeiro ou falso? ``As diagonais de um retângulo são \omterm{def:g6:symmetry:axis}{eixos de
simetria} do retângulo.'' Explique com um argumento de dobra ou com um
desenho.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:6}
Um triângulo $ABC$ é refletido em relação a uma reta $d$, dando
$A'B'C'$. Sabendo que $AB = 4$ cm, $\widehat{BAC} = 70^\circ$ e que o
\omterm{def:g4:measure:perimeter}{perímetro} de $ABC$ é $12$ cm, dê, sem construção nenhuma, $A'B'$,
$\widehat{B'A'C'}$ e o \omterm{def:g4:measure:perimeter}{perímetro} de $A'B'C'$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:7}
Trace um \omterm{def:g6:lines:objects}{segmento} $[AB]$ de $7$ cm e construa a \omterm{def:g6:symmetry:bisector}{mediatriz} dele com o
compasso (\cref{met:g6:symmetry:bisector}). Escolha um ponto $M$
qualquer sobre ela e confira com a régua que $MA = MB$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:8}
Duas árvores ficam nos pontos $A$ e $B$ de um jardim (plano). Onde se
pode plantar uma fonte de modo que ela fique à mesma distância das duas
árvores? Descreva \emph{todos} os lugares possíveis.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:9}
No papel quadriculado, trace o eixo $d$ (uma diagonal da malha, a
$45^\circ$) e uma pequena figura em forma de L de um dos lados.
Construa a figura simétrica em relação a $d$. (Reflita cada \omterm{def:g5:solids:def}{vértice}:
para um eixo a $45^\circ$, um ponto $2$ quadradinhos à direita do eixo
cai $2$ quadradinhos acima dele.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:10}
Trace um círculo de centro $O$, uma reta $d$ passando por $O$ e um
ponto $M$ sobre o círculo. Onde fica o \omterm{def:g6:symmetry:def}{simétrico} de $M$ em relação a
$d$? Explique por que a reflexão leva o círculo sobre ele mesmo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:11}
Usando a propriedade de equidistância
(\cref{prop:g6:symmetry:bisector}), explique por que os dois pontos de
cruzamento dos arcos do \cref{met:g6:symmetry:bisector} realmente
ficam sobre a \omterm{def:g6:symmetry:bisector}{mediatriz} de $[AB]$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:symmetry:12}
Uma bola de bilhar no ponto $A$ tem de bater numa parede reta $d$ e
chegar ao ponto $B$ (os dois do mesmo lado da parede). Reflita $B$ em
relação a $d$, obtendo $B'$, e trace o \omterm{def:g6:lines:objects}{segmento} $[AB']$. Explique por
que o melhor ponto de rebatida é onde $[AB']$ cruza a parede. (Ideia: o
caminho $A \to P \to B$ tem o mesmo comprimento que
$A \to P \to B'$.)
\end{exercise}

\section{Problema: a fonte, o prato quebrado e o espelho que caminha}

\begin{problem}[{Problema de fim de semana --- as três mediatrizes de
um triângulo se encontram num ponto: o círculo que passa por três
pontos e o que dois espelhos fazem juntos}]
\label{pb:g6:symmetry:1}
No \cref{exo:g6:symmetry:8}, uma fonte tinha de ficar à mesma distância
de \emph{duas} árvores, e a resposta foi uma reta inteira de lugares.
Este problema planta uma \emph{terceira} árvore --- e a reta inteira
desaba num único ponto perfeitamente determinado. Esse ponto esconde um
tesouro da geometria: ele permite traçar o único círculo que passa por
três pontos dados, reconstruir um prato inteiro a partir de um caco, e
pertence a todo triângulo. A última parte faz uma experiência com dois
espelhos cujo resultado anuncia a geometria do ano que vem.

\medskip
\noindent\textbf{Parte I --- Duas árvores e depois um mapa de
territórios.}
\begin{enumerate}
  \item Marque dois pontos $A$ e $B$ com $AB = 6$ cm e construa a
        \omterm{def:g6:symmetry:bisector}{mediatriz} de $[AB]$ com o compasso
        (\cref{met:g6:symmetry:bisector}). Escolha um ponto $M$ sobre
        ela e confira com a régua que $MA = MB$.
  \item A \cref{prop:g6:symmetry:bisector} diz duas coisas ao mesmo
        tempo: todo ponto \emph{sobre} a \omterm{def:g6:symmetry:bisector}{mediatriz} é equidistante de
        $A$ e de $B$, e todo ponto equidistante \emph{está sobre} a
        \omterm{def:g6:symmetry:bisector}{mediatriz}. Qual das duas responde à pergunta da fonte do
        \cref{exo:g6:symmetry:8}? O que a proposição diz de um ponto
        que \emph{não} está sobre a \omterm{def:g6:symmetry:bisector}{mediatriz}?
  \item Tome um ponto $M$ do mesmo lado da \omterm{def:g6:symmetry:bisector}{mediatriz} que $A$. O
        \omterm{def:g6:lines:objects}{segmento} $[MB]$ cruza a \omterm{def:g6:symmetry:bisector}{mediatriz} num ponto $P$. Compare o
        caminho $M \to P \to B$ com o caminho $M \to P \to A$ e
        explique por que $M$ está mais perto de $A$ do que de $B$: a
        \omterm{def:g6:symmetry:bisector}{mediatriz} divide a folha num ``lado do $A$'' e num ``lado do
        $B$''.
  \item Duas escolas $A$ e $B$ atendem uma cidade. Toda criança vai a
        pé até a escola mais próxima. Faça um mapinha com $A$ e $B$ e
        pinte a região exata da cidade atendida pela escola $A$. Que
        reta forma a fronteira?
  \item Abre uma terceira escola $C$ (ponha-a de modo que as três
        escolas formem um triângulo). Construa a \omterm{def:g6:symmetry:bisector}{mediatriz} de $[AB]$ e
        a \omterm{def:g6:symmetry:bisector}{mediatriz} de $[BC]$, e chame de $O$ o ponto de cruzamento
        delas. Que duas igualdades de distâncias $O$ satisfaz, e por
        quê?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Três árvores, um ponto, um círculo.}
\begin{enumerate}[resume]
  \item Deduza da questão 5 que $OA = OC$ --- e portanto que $O$
        também está sobre a \omterm{def:g6:symmetry:bisector}{mediatriz} de $[AC]$, que você nunca
        traçou. Conclua: \emph{as três \omterm{def:g6:symmetry:bisector}{mediatrizes} dos lados de um
        triângulo passam todas por um mesmo ponto.}
  \item Explique por que o círculo de centro $O$ e \omterm{def:g3:shapes:shapes}{raio} $OA$ também
        passa por $B$ e por $C$: ele é \emph{o} círculo que passa pelos
        três pontos --- o \emph{círculo
        circunscrito}\index{círculo circunscrito} do triângulo $ABC$.
  \item Construa um triângulo de lados $6$ cm, $5$ cm e $4$ cm
        (\cref{met:g6:shapes:construct}) e depois o \omterm{pb:g6:symmetry:1}{círculo
        circunscrito} dele. Por que basta traçar apenas \emph{duas}
        \omterm{def:g6:symmetry:bisector}{mediatrizes} para encontrar o centro?
  \item O prato quebrado: um arqueólogo desenterra um caco cuja única
        parte intacta é um pedaço da borda circular do prato. Descreva
        uma receita que recupera o centro e o \omterm{def:g3:shapes:shapes}{raio} do prato original
        --- e teste-a: trace um arco de círculo com o seu compasso,
        ``esqueça'' o centro, marque três pontos sobre o arco e
        reconstrua-o.
  \item Três pontos poderiam \emph{não} ter círculo nenhum passando
        por eles? Suponha que $A$, $B$, $C$ sejam colineares. O que se
        pode dizer das \omterm{def:g6:symmetry:bisector}{mediatrizes} de $[AB]$ e de $[BC]$
        (\cref{prop:g6:lines:facts})? Conclua.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Eixos por toda parte e o espelho que
caminha.}
\begin{enumerate}[resume]
  \item O \cref{ex:g6:symmetry:shapes} contou $3$ eixos para o
        \omterm{def:g6:shapes:triangles}{triângulo equilátero} e $4$ para o quadrado. Conjecture a
        contagem para um \omterm{def:g4:geometry:polygon}{pentágono} regular e para um \omterm{def:g4:geometry:polygon}{hexágono} regular,
        e descreva por onde passam os eixos em cada caso (\omterm{def:g3:numbers:evenodd}{número ímpar}
        de lados contra \omterm{def:g3:numbers:evenodd}{número par}: as duas situações são diferentes).
  \item Dobre uma folha de papel uma vez e recorte uma forma
        atravessando as duas camadas; desdobre. Por que o buraco
        \emph{sempre} tem \omterm{def:g6:symmetry:axis}{eixo de simetria}, qualquer que seja o
        recorte, e que reta é esse eixo?
  \item Dobre a folha duas vezes --- a segunda dobra perpendicular à
        primeira ---, recorte e desdobre. Quantos \omterm{def:g6:symmetry:axis}{eixos de simetria} o
        buraco tem, no mínimo, e quantas cópias do seu recorte você vê?
        (Os floquinhos de neve de papel são essa ideia dobrada mais
        vezes.)
  \item O espelho que caminha: no papel quadriculado, trace um eixo
        vertical $d_1$, uma bandeirinha $F$ à esquerda dele e um
        segundo eixo vertical $d_2$, $4$ quadradinhos à direita de
        $d_1$. Reflita $F$ em relação a $d_1$ (imagem $F'$) e depois
        reflita $F'$ em relação a $d_2$ (imagem $F''$). Compare $F$ e
        $F''$: mesma orientação ou espelhada? De quantos quadradinhos a
        figura andou? Compare com a distância entre os eixos.
  \item A mesma experiência com $d_2$ \emph{perpendicular} a $d_1$:
        reflita a bandeirinha em relação ao eixo vertical e depois a
        imagem em relação ao horizontal. Descreva como $F''$ fica em
        relação a $F$ e ao ponto de cruzamento dos dois eixos. (Você
        acabou de descobrir a transformação do ano que vem:
        \cref{ch:g7:central}.)
\end{enumerate}
\end{problem}
