\chapter{Proporcionalidade e dados}\label{ch:g6:propdata}

Se três cadernos custam $6$ reais, seis cadernos custam $12$: o dobro
de cadernos, o dobro do preço. Grandezas que se comportam assim são
\emph{\omterm{def:g6:propdata:def}{proporcionais}} --- uma das ideias mais úteis de toda a
matemática, desenvolvida mais a fundo no \cref{ch:g7:prop}. O capítulo
termina lendo e traçando gráficos de dados simples.

\section{Grandezas proporcionais}

\begin{definition}[Proporcionalidade]\label{def:g6:propdata:def}
Duas grandezas são \emph{proporcionais}\index{proporcionalidade} quando
os valores de uma são obtidos a partir dos valores da outra
multiplicando sempre pelo \emph{mesmo} número, chamado de
\emph{coeficiente de proporcionalidade}.
\end{definition}

\begin{example}\label{ex:g6:propdata:table}
Cadernos a $2$ reais cada um:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cccc}
cadernos & $3$ & $5$ & $8$ & $12$ \\
\hline
preço (reais) & $6$ & $10$ & $16$ & $24$ \\
\end{tabular}
\end{center}
Cada preço é o número de cadernos $\times 2$: o coeficiente é $2$ (o
preço unitário). Uma conferência rápida de que uma tabela é
proporcional: todos os ``\omterm{def:g3:division:remainder}{quocientes} de coluna'' $\frac{6}{3},
\frac{10}{5}, \frac{16}{8}, \frac{24}{12}$ têm de ser iguais --- aqui
todos valem $2$.
\end{example}

\begin{example}[Um contraexemplo]\label{ex:g6:propdata:nonexample}
Idade e altura não são \omterm{def:g6:propdata:def}{proporcionais}: quem tem $12$ anos não é duas
vezes mais alto que quem tem $6$. Confira nos números: $1.50$ m aos
$12$ anos e $1.15$ m aos $6$ anos dão \omterm{def:g3:division:remainder}{quocientes}
$\frac{1.50}{12} = 0.125$ e $\frac{1.15}{6} \approx 0.19$ --- não são
iguais.
\end{example}

\begin{method}[Completar uma tabela de proporcionalidade]\label{met:g6:propdata:complete}
Três caminhos, à escolha:
\begin{enumerate}
  \item \emph{coeficiente}: descubra o multiplicador a partir de uma
        coluna completa e aplique-o às outras;
  \item \emph{colunas}: uma coluna pode ser multiplicada por um número,
        ou duas colunas podem ser somadas, para produzir uma coluna
        nova;
  \item \emph{voltar à unidade}: descubra primeiro o valor de
        \emph{um} item e depois multiplique.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:propdata:methods}
Cinco canecas iguais custam $15$ reais; quanto custam oito canecas?

\emph{Voltando à unidade}: uma caneca custa $15 \div 5 = 3$ reais,
então oito canecas custam $8 \times 3 = 24$ reais.

\emph{Pelas colunas}: $8 = 5 + 3$; três canecas custam $\frac{3}{5}$ de
$15$, ou seja, $9$; então oito canecas custam $15 + 9 = 24$ reais. A
mesma resposta, como tinha de ser.
\end{example}

\section{Porcentagens}

\begin{definition}[Porcentagem]\label{def:g6:propdata:percent}
``$t\,\%$ de uma quantidade'' quer dizer a \omterm{def:g6:fractions:def}{fração} $\frac{t}{100}$ dela:
$25\,\%$ de $60$ é $\frac{25}{100} \times 60 = 15$. Uma porcentagem é
uma \omterm{def:g6:propdata:def}{proporcionalidade} de coeficiente $\frac{t}{100}$.
\end{definition}

\begin{example}\label{ex:g6:propdata:percent}
Uma turma de $30$ alunos tem $40\,\%$ de meninas. Número de meninas,
passo a passo: $40\,\%$ quer dizer $\frac{40}{100} = 0.4$, e
$0.4 \times 30 = 12$ meninas. Referências úteis: $50\,\%$ é a \omterm{def:g3:division:half}{metade},
$25\,\%$ é um quarto, $10\,\%$ é um décimo --- então $10\,\%$ de $30$ é
$3$, e $40\,\% = 4 \times 10\,\%$ é $4 \times 3 = 12$: a mesma
resposta, calculada de cabeça.
\end{example}

\section{Ler e traçar gráficos}

\begin{method}[Ler uma tabela ou um gráfico]\label{met:g6:propdata:read}
Seja qual for a imagem --- tabela, gráfico de barras, gráfico de
linhas:
\begin{enumerate}
  \item leia primeiro o título e as unidades dos dois eixos;
  \item para responder a uma pergunta, localize a barra ou a coluna
        certa e depois leia o valor com cuidado na graduação;
  \item para comparar, compare as barras pela altura --- mas confira se
        o eixo começa em $0$, senão a imagem pode enganar.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:propdata:chart}
O gráfico de barras abaixo mostra o número de livros emprestados na
biblioteca da escola durante uma semana.

\begin{omfigure}
\begin{tikzpicture}
\begin{axis}[
  width=8.6cm, height=5.2cm,
  ybar, bar width=16pt,
  symbolic x coords={Seg,Ter,Qua,Qui,Sex},
  xtick=data,
  ymin=0, ymax=22,
  ytick={5,10,15,20},
  ylabel={livros emprestados},
  tick label style={font=\small},
  label style={font=\small}]
  \addplot[fill=omDef!30, draw=omDef] coordinates
    {(Seg,12) (Ter,8) (Qua,17) (Qui,6) (Sex,15)};
\end{axis}
\end{tikzpicture}

{\small Um gráfico de barras: uma barra por dia, com altura igual à
contagem.}
\end{omfigure}

Leituras: o dia mais movimentado é quarta-feira ($17$ livros). Terça e
quinta juntas somam $8 + 6 = 14$ livros --- menos que sexta sozinha
($15$). Total da semana: $12 + 8 + 17 + 6 + 15 = 58$ livros.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g6:propdata:1}
A tabela é proporcional? Justifique calculando os \omterm{def:g3:division:remainder}{quocientes}.
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccc}
kg de maçãs & $2$ & $3$ & $5$ \\
\hline
preço (reais) & $5$ & $7.5$ & $12.5$ \\
\end{tabular}
\qquad
\begin{tabular}{l|ccc}
idade (anos) & $2$ & $4$ & $8$ \\
\hline
altura (cm) & $85$ & $103$ & $130$ \\
\end{tabular}
\end{center}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:2}
Quatro croissants custam $4.80$ reais. Descubra o preço de um croissant
e depois o de sete croissants.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:3}
Complete a tabela de \omterm{def:g6:propdata:def}{proporcionalidade} (comece pelo coeficiente!):
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|cccc}
litros de combustível & $10$ & $25$ & $?$ & $60$ \\
\hline
preço (reais) & $18$ & $?$ & $72$ & $?$ \\
\end{tabular}
\end{center}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:4}
Um carro usa $6$ L de combustível a cada $100$ km. Quanto combustível
para $250$ km? E para $350$ km? Que distância ele percorre com $27$ L?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:5}
Calcule de cabeça, usando as referências do
\cref{ex:g6:propdata:percent}: $50\,\%$ de $84$; \quad $25\,\%$ de
$200$; \quad $10\,\%$ de $63$; \quad $30\,\%$ de $70$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:6}
Numa escola de $450$ alunos, $60\,\%$ almoçam no refeitório. Quantos
alunos são? E quantos não almoçam?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:7}
Usando o gráfico de barras do \cref{ex:g6:propdata:chart}: em que dias
foram emprestados menos de $10$ livros? Quantos livros a mais foram
emprestados na quarta do que na quinta?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:8}
As temperaturas do meio-dia de segunda a sexta foram $14$, $16$, $13$,
$17$, $20$ graus. Trace um gráfico de barras desses dados (escolha uma
graduação sensata) e diga qual foi o dia mais quente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:propdata:9}
Uma receita para $6$ pessoas pede $450$ g de farinha e $3$ ovos.
Adapte-a para $10$ pessoas. (Para os ovos, pense antes de escrever um
\omterm{def:g6:decimals:places}{número decimal} de ovos!)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:propdata:10}
A velocidade constante, uma ciclista percorre $24$ km em $1$ hora.
\begin{enumerate}
  \item Que distância ela percorre em $2$ h? E em meia hora? E em
        $1$ h $30$ min?
  \item De quanto tempo ela precisa para $60$ km?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:propdata:11}
Uma loja anuncia ``$20\,\%$ de desconto em tudo''. Calcule o desconto e
o preço novo para: uma bola de $15$ reais; uma raquete de $40$ reais. O
preço novo é proporcional ao preço antigo? Qual é o coeficiente?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:propdata:12}
Duas velas da mesma altura são acesas ao mesmo tempo. A grossa se
consome inteira em $6$ horas e a fina em $4$ horas, cada uma a um ritmo
constante próprio. Depois de quanto tempo a vela fina tem exatamente a
\omterm{def:g3:division:half}{metade} da altura da grossa? (Escreva as alturas restantes depois de $t$
horas como frações da altura inicial.)
\end{exercise}

\section{Problema: mapas, plantas e como mentir com um gráfico}

\begin{problem}[{Problema de fim de semana --- a escala de um mapa é
uma proporcionalidade: os comprimentos seguem o coeficiente, as áreas
seguem o quadrado dele, e gráficos mal traçados enganam o olho}]
\label{pb:g6:propdata:1}
Todo mapa, planta e maquete obedece a uma regra: os comprimentos reais
e os comprimentos desenhados são \emph{\omterm{def:g6:propdata:def}{proporcionais}}
(\cref{def:g6:propdata:def}). O coeficiente tem um nome famoso --- a
\emph{escala}\index{escala} --- e esconde duas armadilhas que este
problema arma de propósito: as \omterm{def:g6:measure:area}{áreas} \emph{não} seguem a escala, e as
porcentagens, como os gráficos, dependem inteiramente daquilo em
relação a que são medidas.

\medskip
\noindent\textbf{Parte I --- Ler um mapa.}
Um mapa de caminhada está na escala $1:100\,000$: cada centímetro no
mapa representa $100\,000$ centímetros no terreno.
\begin{enumerate}
  \item Converta $100\,000$ cm em quilômetros. O que $1$ cm desse mapa
        representa, em km?
  \item Dois vilarejos estão a $7.5$ cm um do outro no mapa; a margem
        de um lago é uma curva de $12$ cm. Dê as distâncias reais.
  \item Uma estrada romana perfeitamente reta tem $20$ km. Que
        comprimento ela tem no mapa?
  \item Um segundo mapa anuncia ``$1$ cm para $5$ km''. Escreva a
        escala dele na forma $1: \,?$. Qual dos dois mapas mostra mais
        detalhes para a mesma região?
  \item Monte uma tabelinha (distância no mapa $1$, $2$, $7.5$ cm
        contra distância real) para o mapa de caminhada e explique por
        que uma escala é exatamente uma \omterm{def:g6:propdata:def}{proporcionalidade} no sentido da
        \cref{def:g6:propdata:def}. Qual é o coeficiente que transforma
        centímetros-no-mapa em quilômetros?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A planta do quarto de Zoe.}
Zoe desenha a planta do quarto dela --- um retângulo de $4$ m por
$3$ m --- na escala $1:50$.
\begin{enumerate}[resume]
  \item Que dimensões o quarto tem na planta?
  \item A cama dela mede $2$ m por $0.9$ m, e o vão da porta tem
        $80$ cm de largura. Dê as três medidas na planta.
  \item Agora a armadilha. Calcule a \omterm{def:g6:measure:area}{área} real do quarto em m$^2$ e
        depois a \omterm{def:g6:measure:area}{área} da planta dele em cm$^2$. Converta a \omterm{def:g6:measure:area}{área} real em
        cm$^2$ (\cref{def:g6:measure:area}) e divida pela \omterm{def:g6:measure:area}{área} da
        planta: o \omterm{def:g3:division:remainder}{quociente} é $50$?
  \item Explique o número que você encontrou: numa planta $1:50$, cada
        centímetro quadrado de papel representa um quadrado real de
        $50$ cm por $50$ cm. Quantos cm$^2$ reais são? Enuncie a regra:
        quando os comprimentos são divididos por $50$, as \omterm{def:g6:measure:area}{áreas} são
        divididas por\,\dots
  \item Um tapete redondo cobre $6$ cm$^2$ na planta. Que \omterm{def:g6:measure:area}{área} real ele
        cobre, em cm$^2$ e depois em m$^2$?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Como mentir com um gráfico (e com uma
porcentagem).}
\begin{enumerate}[resume]
  \item Uma loja vendeu $105$ reais no sábado e $110$ no domingo. O
        gerente traça um gráfico de barras cujo eixo vertical começa em
        $100$: a barra de sábado sobe $5$ unidadezinhas e a de domingo,
        $10$. O que a imagem sugere sobre o domingo, e qual é a
        verdade? (Calcule o aumento de domingo como porcentagem do
        sábado, arredondado ao por cento.) Que alerta do
        \cref{met:g6:propdata:read} o gerente ignorou?
  \item Refaça (ou descreva) o gráfico honesto, com o eixo começando em
        $0$: como as duas barras se comparam agora?
  \item Uma coleção cresce de $40$ para $60$ selos: calcule o aumento
        como porcentagem do valor inicial. Depois ela encolhe de volta
        de $60$ para $40$: calcule a queda como porcentagem do valor
        inicial \emph{dela}. Por que as duas respostas são diferentes,
        para os mesmos $20$ selos?
  \item Época de liquidação: um casaco de $100$ reais recebe ``$20\,\%$
        de desconto'' e, no caixa, mais ``$10\,\%$ sobre o preço já
        reduzido''. Calcule o preço final passo a passo. O desconto
        total é de $30\,\%$? Explique ao cliente quanto ele é de
        verdade.
  \item Final, de volta ao mapa da questão 4 ($1$ cm para $5$ km): uma
        floresta ocupa $3$ cm$^2$ nesse mapa. Qual é a \omterm{def:g6:measure:area}{área} real dela,
        em km$^2$? Conclua com a regra deste problema inteiro: os
        comprimentos seguem a escala, as \omterm{def:g6:measure:area}{áreas} seguem o\,\dots
\end{enumerate}
\end{problem}
