\chapter{Triangles et quadrilatères}\label{ch:g6:shapes}

Les triangles et les \omterm{def:g4:geometry:polygon}{quadrilatères} sont les formes de base de la
géométrie. Le but de ce chapitre est de connaître leurs familles
spéciales --- \omterm{def:g6:shapes:triangles}{isocèles}, équilatéraux, rectangles~; rectangles,
\omterm{def:g6:shapes:quadrilaterals}{losanges}, carrés --- par leurs \emph{propriétés définissantes}, et de
les construire précisément à la règle et au compas.

\section{Triangles}

\begin{definition}[Triangles spéciaux]\label{def:g6:shapes:triangles}
Un triangle $ABC$ a trois \omterm{def:g5:solids:def}{sommets} et trois côtés. Il est~:
\begin{itemize}
  \item \emph{isocèle}\index{triangle isocèle} en $A$ lorsque
        $AB = AC$ (deux côtés égaux)~;
  \item \emph{équilatéral}\index{triangle équilatéral} lorsque
        $AB = BC = CA$ (trois côtés égaux)~;
  \item \emph{rectangle}\index{triangle rectangle} en $A$ lorsque
        l'angle $\widehat{BAC}$ est un \omterm{def:g3:shapes:rightangle}{angle droit}. Le côté $[BC]$
        opposé à l'\omterm{def:g3:shapes:rightangle}{angle droit} s'appelle l'\emph{hypoténuse}.
\end{itemize}
\end{definition}

\begin{method}[Construire un triangle à partir de ses trois côtés]\label{met:g6:shapes:construct}
Pour construire $ABC$ avec $AB = 5$~cm, $AC = 4$~cm, $BC = 3$~cm~:
\begin{enumerate}
  \item tracer le \omterm{def:g6:lines:objects}{segment} $[AB]$ de $5$~cm à la règle~;
  \item tracer un arc du cercle de centre $A$ et de \omterm{def:g3:shapes:shapes}{rayon} $4$~cm
        (tous les points à $4$~cm de $A$)~;
  \item tracer un arc du cercle de centre $B$ et de \omterm{def:g3:shapes:shapes}{rayon} $3$~cm~;
  \item le point $C$ est où les deux arcs se croisent~; le joindre à
        $A$ et $B$.
\end{enumerate}
\end{method}

\section{Quadrilatères}

\begin{definition}[Quadrilatères spéciaux]\label{def:g6:shapes:quadrilaterals}
Un \omterm{def:g4:geometry:polygon}{quadrilatère} a quatre \omterm{def:g5:solids:def}{sommets} et quatre côtés. C'est un~:
\begin{itemize}
  \item \emph{rectangle}\index{rectangle} lorsque ses quatre angles
        sont des \omterm{def:g3:shapes:rightangle}{angles droits}~;
  \item \emph{losange}\index{losange} lorsque ses quatre côtés ont la
        même longueur~;
  \item \emph{carré}\index{carré} lorsqu'il est les deux~: quatre
        \omterm{def:g3:shapes:rightangle}{angles droits} \emph{et} quatre côtés égaux.
\end{itemize}
\end{definition}

\begin{proposition}[Propriétés des diagonales]\label{prop:g6:shapes:diagonals}
Dans ces \omterm{def:g4:geometry:polygon}{quadrilatères}, les deux diagonales se coupent en leur
milieu commun, et de plus~:
\begin{itemize}
  \item dans un rectangle, les diagonales ont la \emph{même
        longueur}~;
  \item dans un \omterm{def:g6:shapes:quadrilaterals}{losange}, les diagonales sont \emph{\omterm{def:g6:lines:perp}{perpendiculaires}}~;
  \item dans un carré, les deux à la fois.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}
Un carré est automatiquement un rectangle (il a quatre \omterm{def:g3:shapes:rightangle}{angles droits})
et automatiquement un \omterm{def:g6:shapes:quadrilaterals}{losange} (quatre côtés égaux). Donc «~rectangle~»
ne signifie pas «~pas un carré~»~! En mathématiques, la forme plus
spéciale appartient à la famille plus générale.
\end{remark}

\begin{example}[Reconnaître une forme d'après ses propriétés]\label{ex:g6:shapes:recognize}
Un \omterm{def:g4:geometry:polygon}{quadrilatère} a quatre côtés de longueur $4$~cm et un \omterm{def:g3:shapes:rightangle}{angle droit}.
Qu'est-ce~? Quatre côtés égaux en font un \omterm{def:g6:shapes:quadrilaterals}{losange}. Dans un \omterm{def:g6:shapes:quadrilaterals}{losange},
les angles opposés sont égaux et les quatre angles additionnent
$360^\circ$~; un \omterm{def:g3:shapes:rightangle}{angle droit} force alors les quatre à être droits
(ce sera prouvé avec les parallélogrammes dans
\cref{ch:g7:central}). Donc la forme est un carré.
\end{example}

\begin{method}[Programmes de construction]\label{met:g6:shapes:program}
Un \emph{programme de construction} est une liste d'instructions que
n'importe qui peut suivre pour reproduire une figure exactement.
Écrire une instruction par ligne, nommer chaque point quand il est
créé, et n'utiliser que~: «~tracer le \omterm{def:g6:lines:objects}{segment} / la droite / le
cercle\,\dots~», «~placer le point où \dots\ se croisent~». Tester
son programme en le suivant littéralement, mot pour mot.
\end{method}

\begin{example}\label{ex:g6:shapes:program}
Programme pour un carré $ABCD$ de côté $4$~cm~:
\begin{enumerate}
  \item Tracer un \omterm{def:g6:lines:objects}{segment} $[AB]$ avec $AB = 4$~cm.
  \item Tracer la \omterm{def:g6:lines:perp}{perpendiculaire} à $(AB)$ en $A$~; dessus, placer
        $D$ à $4$~cm de $A$ (choisir un côté de la droite et s'y
        tenir).
  \item Tracer la \omterm{def:g6:lines:perp}{perpendiculaire} à $(AB)$ en $B$~; dessus, placer
        $C$ à $4$~cm de $B$, du même côté que $D$.
  \item Joindre $C$ et $D$.
\end{enumerate}
Vérifier à la règle~: $DC$ doit aussi mesurer $4$~cm.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g6:shapes:1}
Construire un triangle $ABC$ avec $AB = 6$~cm, $AC = 5$~cm et
$BC = 4$~cm, en suivant \cref{met:g6:shapes:construct}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:2}
Construire un \omterm{def:g6:shapes:triangles}{triangle isocèle} $DEF$ avec $DE = DF = 5$~cm et
$EF = 3$~cm. Quels deux angles de ce triangle semblent égaux~? Les
mesurer pour vérifier.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:3}
Construire un \omterm{def:g6:shapes:triangles}{triangle équilatéral} de côté $4$~cm en n'utilisant que
la règle et le compas. Mesurer un de ses angles.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:4}
Classer chaque triangle d'après ses données (\omterm{def:g6:shapes:triangles}{isocèle}, \omterm{def:g6:shapes:triangles}{équilatéral},
rectangle --- plusieurs réponses peuvent s'appliquer)~:
(a) côtés $7$, $7$, $7$~; \
(b) côtés $5$, $5$, $8$~; \
(c) angles $90^\circ$, $45^\circ$, $45^\circ$~; \
(d) angles $60^\circ$, $60^\circ$, $60^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:5}
Dans un \omterm{def:g6:shapes:triangles}{triangle rectangle}, quel côté est l'hypoténuse~? Dessiner un
triangle $MNP$ rectangle en $N$, et nommer son hypoténuse. Quel \omterm{def:g5:solids:def}{sommet}
l'hypoténuse ne touche-t-elle jamais~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:6}
Vrai ou faux~? Justifier chaque réponse en une phrase.
\begin{enumerate}
  \item Tout carré est un rectangle.
  \item Tout rectangle est un carré.
  \item Tout carré est un \omterm{def:g6:shapes:quadrilaterals}{losange}.
  \item Un \omterm{def:g6:shapes:quadrilaterals}{losange} avec un \omterm{def:g3:shapes:rightangle}{angle droit} est un carré.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:7}
Construire un rectangle $ABCD$ avec $AB = 5$~cm et $BC = 3$~cm.
Mesurer ses deux diagonales~: que remarque-t-on~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:8}
Construire un \omterm{def:g6:shapes:quadrilaterals}{losange} de diagonales $6$~cm et $4$~cm. (Utiliser
\cref{prop:g6:shapes:diagonals}~: tracer d'abord les deux diagonales ---
\omterm{def:g6:lines:perp}{perpendiculaires}, se croisant en leurs milieux --- puis joindre les
quatre extrémités.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:9}
Écrire un programme de construction pour un \omterm{def:g6:shapes:triangles}{triangle isocèle} de base
$4$~cm et de côtés égaux $6$~cm, dans le style de
\cref{ex:g6:shapes:program}. Échanger avec un camarade~: peut-il le
suivre sans voir la figure~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:10}
Tracer un cercle de centre $O$ et un de ses \omterm{def:g4:geometry:circle}{diamètres} $[AB]$. Placer
un point $M$ quelconque sur le cercle et tracer le triangle $AMB$.
Mesurer l'angle $\widehat{AMB}$. Déplacer $M$ et mesurer encore. Que
conjecture-t-on~? (Ce fait surprenant est prouvé dans
\cref{ch:g8:cosine}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:11}
Un \omterm{def:g4:geometry:polygon}{quadrilatère} a des diagonales de même longueur qui se coupent en
leur milieu commun. Déterminer quelle forme ce doit être, en utilisant
\cref{prop:g6:shapes:diagonals}~: rectangle, \omterm{def:g6:shapes:quadrilaterals}{losange} ou carré~?
(Attention~: une seule propriété est donnée.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:shapes:12}
Sur un quadrillage pointé $4 \times 4$, combien de carrés différents
peut-on dessiner dont les coins sont des points du quadrillage --- en
comptant aussi les carrés «~penchés~»~? Dessiner au moins un carré
penché et expliquer pourquoi ses côtés sont égaux.
\end{exercise}

\section{Problème~: ce que savent les diagonales}

\begin{problem}[{Devoir maison --- reconstruire chaque quadrilatère
particulier à partir de ses seules diagonales, et compter les
diagonales d'un polygone quelconque}]
\label{pb:g6:shapes:1}
\Cref{prop:g6:shapes:diagonals} lit une figure et renseigne sur ses
diagonales. Ce problème joue le jeu \emph{à l'envers}~: à partir de
deux \omterm{def:g6:lines:objects}{segments} sécants seulement --- les futures diagonales --- quelle
figure apparaît lorsqu'on joint leurs quatre extrémités~? On y
construira un «~dictionnaire des diagonales~» complet, on comprendra
\emph{pourquoi} la construction du \omterm{def:g6:shapes:quadrilaterals}{losange} de
l'\cref{exo:g6:shapes:8} fonctionne, on rencontrera une figure encore
sans nom, et on terminera en comptant les diagonales de \omterm{def:g4:geometry:polygon}{polygones} à
bien plus de quatre côtés.

\medskip
\noindent\textbf{Partie I --- Le dictionnaire des diagonales.}
Pour chaque question, tracer les deux \omterm{def:g6:lines:objects}{segments} décrits, joindre leurs
quatre extrémités dans l'ordre, et reconnaître le \omterm{def:g4:geometry:polygon}{quadrilatère}
obtenu.
\begin{enumerate}
  \item Deux \omterm{def:g6:lines:objects}{segments} de $6$ cm chacun, \omterm{def:g6:lines:perp}{perpendiculaires}, se coupant
        en leur milieu commun. Mesurer les quatre côtés et les
        quatre angles de la figure obtenue~: qu'est-ce que c'est~?
  \item Deux \omterm{def:g6:lines:objects}{segments} de $8$ cm et $5$ cm, \omterm{def:g6:lines:perp}{perpendiculaires}, se
        coupant en leur milieu commun. Quelle figure apparaît
        (\cref{exo:g6:shapes:8})~?
  \item Deux \omterm{def:g6:lines:objects}{segments} de $7$ cm chacun --- égaux mais \emph{non}
        \omterm{def:g6:lines:perp}{perpendiculaires} --- se coupant en leur milieu commun.
        Mesurer les côtés et les angles~: quelle figure
        conjecture-t-on~?
  \item Deux \omterm{def:g6:lines:objects}{segments} de $8$ cm et $5$ cm, non \omterm{def:g6:lines:perp}{perpendiculaires}, se
        coupant en leur milieu commun. La figure obtenue n'est
        aucune des trois de la~\cref{def:g6:shapes:quadrilaterals} ---
        décrire ce que l'on voit (quels côtés semblent égaux~?
        \omterm{def:g6:lines:perp}{parallèles}~?). Ce \omterm{def:g4:geometry:polygon}{quadrilatère} reçoit son nom au
        \cref{ch:g7:central}.
  \item Résumer les questions 1 à 4 dans un dictionnaire à quatre
        entrées~: diagonales \emph{égales / \omterm{def:g6:lines:perp}{perpendiculaires} / les
        deux / ni l'un ni l'autre} (se coupant toujours en leur
        milieu) $\rightarrow$ nom (ou description) de la figure.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Pourquoi les constructions
fonctionnent.}
\begin{enumerate}[resume]
  \item À la question 2, les deux diagonales découpent la figure en
        quatre \omterm{def:g6:shapes:triangles}{triangles rectangles} autour du point d'intersection.
        Quelles sont les longueurs des deux côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit}
        de chacun de ces triangles~? Expliquer pourquoi les quatre
        triangles sont des copies exactes les uns des autres, et en
        déduire pourquoi les quatre côtés de la figure sont égaux
        --- le secret de l'\cref{exo:g6:shapes:8}.
  \item En déduire un test au compas seul~: après avoir construit
        une figure à partir de ses diagonales, comment vérifier au
        compas seul (sans mesure à la règle) que les quatre côtés
        sont égaux~?
  \item Rédiger un programme de construction
        (\cref{met:g6:shapes:program}) pour un carré dont les
        \emph{diagonales} mesurent $6$ cm --- et non les côtés.
  \item Dans le carré de la question 8, le côté est-il plus long ou
        plus court que $3$ cm --- la \omterm{def:g3:division:half}{moitié} de la diagonale~?
        Mesurer pour le savoir. La \omterm{def:g3:division:half}{moitié} de la réponse peut déjà se
        \emph{démontrer}~: aller d'un \omterm{def:g5:solids:def}{sommet} au suivant en passant
        par le centre est un détour de $3 + 3$ cm, et un chemin
        droit est toujours plus court qu'un détour --- le côté
        mesure donc moins de $6$ cm. (Prouver qu'il dépasse aussi
        $3$ cm, et le calculer exactement, sera l'affaire du
        \cref{ch:g8:pythagoras}.)
  \item Une nouvelle entrée pour le dictionnaire~: tracer un \omterm{def:g6:lines:objects}{segment}
        $[AC]$ de $8$ cm~; y marquer le point $H$ à $3$ cm de $A$~;
        tracer par $H$ le \omterm{def:g6:lines:objects}{segment} \omterm{def:g6:lines:perp}{perpendiculaire} $[BD]$ de $5$ cm
        dont $H$ est \emph{le} milieu. Joindre $ABCD$. Cette figure
        est un \emph{cerf-volant}\index{cerf-volant} (une diagonale
        coupe l'autre en son milieu, à \omterm{def:g3:shapes:rightangle}{angle droit}). En utilisant
        les triangles des coins comme à la question 6, déterminer
        quels côtés d'un \omterm{pb:g6:shapes:1}{cerf-volant} sont égaux entre eux.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Compter les diagonales.}
Une \emph{diagonale} d'un \omterm{def:g4:geometry:polygon}{polygone} joint deux \omterm{def:g5:solids:def}{sommets} qui ne sont pas
voisins.
\begin{enumerate}[resume]
  \item Tracer un \omterm{def:g4:geometry:polygon}{quadrilatère}, un \omterm{def:g4:geometry:polygon}{pentagone} (cinq \omterm{def:g5:solids:def}{sommets}) et un
        \omterm{def:g4:geometry:polygon}{hexagone} (six \omterm{def:g5:solids:def}{sommets}), et tracer toutes leurs diagonales.
        Les compter~: combien y en a-t-il dans chaque cas~?
  \item Compter sans dessiner, pour l'\omterm{def:g4:geometry:polygon}{hexagone}~: d'un \omterm{def:g5:solids:def}{sommet},
        combien de diagonales partent-elles~? (À combien des six
        \omterm{def:g5:solids:def}{sommets} ce \omterm{def:g5:solids:def}{sommet} ne peut-il \emph{pas} être joint par une
        diagonale~?) En multipliant par les six \omterm{def:g5:solids:def}{sommets}, on compte
        chaque diagonale exactement deux fois --- pourquoi~?
        Retrouver le compte de la question 11. (Le même double
        comptage est apparu avec les poignées de main du
        \cref{pb:g6:wholes:1}.)
  \item Utiliser la méthode de la question 12 pour compter les
        diagonales d'un \omterm{def:g4:geometry:polygon}{polygone} à $10$ \omterm{def:g5:solids:def}{sommets}, puis à $20$
        \omterm{def:g5:solids:def}{sommets}.
  \item Un \omterm{def:g4:geometry:polygon}{polygone} a exactement $44$ diagonales. Combien a-t-il de
        \omterm{def:g5:solids:def}{sommets}~? (Essayer la méthode de la question 13 sur
        quelques candidats.)
  \item Vérifier la méthode sur le plus petit \omterm{def:g4:geometry:polygon}{polygone} de tous~:
        combien de diagonales a un triangle, et que répond la
        méthode de comptage de la question 12~? Conclure par une
        phrase sur ce que les poignées de main et les diagonales ont
