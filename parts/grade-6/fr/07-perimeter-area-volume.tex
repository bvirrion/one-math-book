\chapter{Périmètre, aire, volume}\label{ch:g6:measure}

Quelle longueur a la clôture, quelle taille a le champ, combien d'eau
tient dans le réservoir~? Trois questions différentes, trois quantités
différentes --- \omterm{def:g6:measure:perimeter}{périmètre}, \omterm{def:g6:measure:area}{aire}, \omterm{def:g6:measure:volume}{volume} --- chacune avec ses propres
unités. Les confondre est l'erreur la plus courante en géométrie~; ce
chapitre les démêle pour de bon.

\section{Longueurs et périmètre}

\begin{definition}[Périmètre]\label{def:g6:measure:perimeter}
Le \emph{périmètre}\index{périmètre} d'une figure est la longueur
totale de son bord. Il se mesure en unités de longueur~: millimètres
(mm), centimètres (cm), mètres (m), kilomètres (km), avec
\[
1 \text{ km} = 1000 \text{ m}, \qquad
1 \text{ m} = 100 \text{ cm}, \qquad
1 \text{ cm} = 10 \text{ mm}.
\]
\end{definition}

\begin{example}\label{ex:g6:measure:perimeter}
Un rectangle de longueur $L$ et de largeur $\ell$ a pour \omterm{def:g6:measure:perimeter}{périmètre}
\[
P = L + \ell + L + \ell = 2 \times (L + \ell).
\]
Pour un rectangle de $7$~cm sur $4$~cm~: $P = 2 \times (7 + 4) = 2 \times 11 =
22$~cm. Un carré de côté $c$ a pour \omterm{def:g6:measure:perimeter}{périmètre} $4c$.
\end{example}

\begin{proposition}[Circonférence d'un cercle]\label{prop:g6:measure:circle}
Le \omterm{def:g6:measure:perimeter}{périmètre} (ou \emph{circonférence}) d'un cercle de \omterm{def:g3:shapes:shapes}{rayon} $r$ est
\[
P = 2 \pi r ,
\]
où $\pi \approx 3.14$ est le même nombre pour tout cercle.
\end{proposition}
\admitted

\begin{example}
Un étang circulaire a un \omterm{def:g3:shapes:shapes}{rayon} de $5$~m. Son bord mesure
$2 \times \pi \times 5 = 10\pi \approx 31.4$~m. Garder la valeur
exacte $10\pi$ aussi longtemps que possible~; n'\omterm{def:g4:numbers:round}{arrondir} qu'à la fin.
\end{example}

\section{Aires}

\begin{definition}[Aire]\label{def:g6:measure:area}
L'\emph{aire}\index{aire} d'une figure mesure la surface qu'elle
couvre~: combien de carreaux unitaires tiennent à l'intérieur.
Unités~: cm$^2$ (un carré de côté $1$~cm), m$^2$, km$^2$, \dots\
Attention~:
\[
1 \text{ m}^2 = 100 \times 100 \text{ cm}^2 = 10\,000 \text{ cm}^2
\]
--- un mètre carré est un carré de $100$~cm sur $100$~cm, donc chaque
pas de l'échelle des unités vaut $100$, et non $10$.
\end{definition}

\begin{tikzpicture}[scale=0.68]
  \draw[gray!50, thin] (0,0) grid (7,4);
  \draw[very thick, omDef] (0,0) rectangle (7,4);
  \fill[omThm!25] (0,3) rectangle (1,4);
  \draw[omThm, thick] (0,3) rectangle (1,4);
  \node[font=\small, omThm] at (2.6,3.5) {$1$ carreau unitaire};
  \node[font=\small] at (3.5,-0.5) {$7 \times 4 = 28$ carreaux unitaires};
\end{tikzpicture}

{\small L'\omterm{def:g6:measure:area}{aire} d'un rectangle~: $7$ colonnes de $4$ carreaux unitaires
chacune, donc $7 \times 4 = 28$ carreaux en tout. C'est pourquoi
\omterm{def:g6:measure:area}{aire} $=$ longueur $\times$ largeur.}
\end{omfigure}

\begin{proposition}[Formules d'aire de base]\label{prop:g6:measure:formulas}
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{l|l}
figure & \omterm{def:g6:measure:area}{aire} \\
\hline
rectangle ($L \times \ell$) & $A = L \times \ell$ \\
carré (côté $c$) & $A = c^2$ \\[4pt]
\omterm{def:g6:shapes:triangles}{triangle rectangle} (côtés $a$, $b$) & $A = \dfrac{a \times b}{2}$ \\[4pt]
disque (\omterm{def:g3:shapes:shapes}{rayon} $r$) & $A = \pi r^2$ \\
\end{tabular}
\end{center}
\end{proposition}

\begin{proof}[Démonstration pour le triangle rectangle]
Deux copies d'un \omterm{def:g6:shapes:triangles}{triangle rectangle} de côtés $a$ et $b$, collées le
long de l'hypoténuse, forment un rectangle $a \times b$. Donc l'\omterm{def:g6:measure:area}{aire}
du triangle est la \omterm{def:g3:division:half}{moitié} de celle du rectangle~: $\frac{ab}{2}$. (La
formule du rectangle est le comptage de carreaux unitaires ci-dessus~;
la formule du disque est admise, voir \cref{ch:g7:areas}.)
\end{proof}

\begin{tikzpicture}[scale=0.75]
  \draw[thick, fill=omDef!15] (0,0) -- (4,0) -- (0,2.5) -- cycle;
  \draw[thick, omExo, dashed] (4,0) -- (4,2.5) -- (0,2.5);
  \draw[thin] (0.3,0) -- (0.3,0.3) -- (0,0.3);
  \node[below, font=\small] at (2,-0.1) {$a$};
  \node[left, font=\small] at (0,1.25) {$b$};
  \node[font=\small, omExo] at (2.8,1.7) {seconde copie};
\end{tikzpicture}

{\small Un \omterm{def:g6:shapes:triangles}{triangle rectangle} est la \omterm{def:g3:division:half}{moitié} d'un rectangle~: d'où la
formule $\frac{a\times b}{2}$.}
\end{omfigure}

\begin{example}[Figures composées]\label{ex:g6:measure:composite}
Une pièce en L est un rectangle $6$~m $\times$ $4$~m avec un coin
carré $2$~m $\times$ $2$~m enlevé. Son \omterm{def:g6:measure:area}{aire}, étape par étape~:
\begin{enumerate}
  \item rectangle complet~: $6 \times 4 = 24$~m$^2$~;
  \item carré enlevé~: $2 \times 2 = 4$~m$^2$~;
  \item \omterm{def:g6:measure:area}{aire} restante~: $24 - 4 = 20$~m$^2$.
\end{enumerate}
Son \emph{\omterm{def:g6:measure:perimeter}{périmètre}} n'est pas $20$ de quoi que ce soit~: en
parcourant le L, le bord mesure encore $6 + 4 + 6 + 4 = 20$~m ---
les deux coupes du coin remplacent deux morceaux de mur égaux. Même
nombre par coïncidence, mais mètres carrés pour l'un, mètres pour
l'autre~!
\end{example}

\begin{remark}[Même périmètre, aires différentes]
Deux figures peuvent avoir le même \omterm{def:g6:measure:perimeter}{périmètre} et des \omterm{def:g6:measure:area}{aires} très
différentes~: un carré $5 \times 5$ et un rectangle $9 \times 1$ ont
tous deux un \omterm{def:g6:measure:perimeter}{périmètre} de $20$, mais des \omterm{def:g6:measure:area}{aires} de $25$ et $9$.
\omterm{def:g6:measure:perimeter}{Périmètre} et \omterm{def:g6:measure:area}{aire} sont vraiment des quantités indépendantes.
\end{remark}

\section{Volumes}

\begin{definition}[Volume]\label{def:g6:measure:volume}
Le \emph{volume}\index{volume} d'un \omterm{def:g5:solids:def}{solide} mesure l'espace qu'il
remplit~: combien de \omterm{def:g5:solids:def}{cubes} unitaires tiennent à l'intérieur.
Unités~: cm$^3$, m$^3$, \dots\ avec
$1$~m$^3 = 1\,000\,000$~cm$^3$ (chaque pas de l'échelle vaut
$1000$). Pour les liquides, on utilise aussi le \emph{litre}~:
\[
1 \text{ L} = 1 \text{ dm}^3 = 1000 \text{ cm}^3,
\qquad
1 \text{ m}^3 = 1000 \text{ L}.
\]
\end{definition}

\begin{proposition}[Volume d'un pavé]\label{prop:g6:measure:box}
Un \omterm{def:g5:solids:def}{pavé} (un \emph{prisme droit à base rectangulaire}) de longueur $L$,
de largeur $\ell$ et de hauteur $h$ a pour \omterm{def:g6:measure:volume}{volume}
\[
V = L \times \ell \times h .
\]
\end{proposition}

\begin{proof}
La couche du bas contient $L \times \ell$ \omterm{def:g5:solids:def}{cubes} unitaires
(image de \cref{def:g6:measure:area}, avec des \omterm{def:g5:solids:def}{cubes}), et il y a $h$
telles couches.
\end{proof}

\begin{tikzpicture}[scale=0.6]
  \foreach \y in {0,1} {
    \foreach \x in {0,...,3} {
      \foreach \z in {0,1} {
        \begin{scope}[shift={({\x+0.45*\z},{\y+0.3*\z})}]
          \draw[fill=omDef!12] (0,0) rectangle (1,1);
          \draw[fill=omDef!20] (0,1) -- (0.45,1.3) -- (1.45,1.3) -- (1,1)
            -- cycle;
          \draw[fill=omDef!28] (1,0) -- (1.45,0.3) -- (1.45,1.3) -- (1,1)
            -- cycle;
        \end{scope}
      }
    }
  }
  \node[below, font=\small] at (2.5,-0.3)
    {$4 \times 2 \times 2 = 16$ cubes unitaires};
\end{tikzpicture}

{\small Un \omterm{def:g5:solids:def}{pavé} rempli de \omterm{def:g5:solids:def}{cubes} unitaires~: $4 \times 2$ \omterm{def:g5:solids:def}{cubes} par
couche, deux couches.}
\end{omfigure}

\begin{example}\label{ex:g6:measure:aquarium}
Un aquarium mesure $60$~cm sur $30$~cm sur $40$~cm (hauteur).
\omterm{def:g6:measure:volume}{Volume}~:
\[
V = 60 \times 30 \times 40 = 72\,000 \text{ cm}^3 = 72 \text{ dm}^3
= 72 \text{ L}.
\]
Étape par étape pour la conversion~: $1000$~cm$^3$ font un litre, et
$72\,000 \div 1000 = 72$.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g6:measure:1}
Convertir~: $3.5$~m en cm~; $420$~mm en cm~; $0.8$~km en m~;
$25\,000$~m en km.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:2}
Calculer le \omterm{def:g6:measure:perimeter}{périmètre} de~: un rectangle $8$~cm $\times$ $3.5$~cm~; un
carré de côté $6.2$~cm~; un triangle de côtés $5$~cm, $7$~cm et
$9$~cm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:3}
Une piste de course circulaire a un \omterm{def:g3:shapes:shapes}{rayon} de $50$~m. Quelle longueur
a un tour (valeur exacte avec $\pi$, puis arrondie au mètre)~? Combien
de tours font au moins $2$~km~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:4}
Calculer l'\omterm{def:g6:measure:area}{aire} de~: un rectangle $9$~cm $\times$ $4$~cm~; un carré
de côté $7$~m~; un \omterm{def:g6:shapes:triangles}{triangle rectangle} de côtés $6$~cm et $10$~cm~; un
disque de \omterm{def:g3:shapes:shapes}{rayon} $3$~cm (valeur exacte, puis arrondie au cm$^2$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:5}
Convertir~: $3$~m$^2$ en cm$^2$~; $45\,000$~cm$^2$ en m$^2$~;
$2.5$~L en cm$^3$~; $4\,500$~L en m$^3$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:6}
Un champ rectangulaire mesure $120$~m de long et $85$~m de large.
Combien de mètres de clôture faut-il pour l'entourer~? Quelle est son
\omterm{def:g6:measure:area}{aire}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:7}
Calculer le \omterm{def:g6:measure:volume}{volume} d'un \omterm{def:g5:solids:def}{pavé} $5$~cm $\times$ $4$~cm $\times$ $10$~cm,
et d'un \omterm{def:g5:solids:def}{cube} d'\omterm{def:g5:solids:def}{arête} $3$~cm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:8}
Dessiner deux rectangles différents de \omterm{def:g6:measure:perimeter}{périmètre} $16$~cm, et calculer
leurs \omterm{def:g6:measure:area}{aires}. Lequel de vos rectangles a la plus grande \omterm{def:g6:measure:area}{aire}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measure:9}
Une figure en T est faite d'un rectangle horizontal $10 \times 2$ sur
un vertical $2 \times 6$ (mesures en cm). Calculer son \omterm{def:g6:measure:area}{aire}, puis son
\omterm{def:g6:measure:perimeter}{périmètre} (parcourir le bord soigneusement, en additionnant chaque
\omterm{def:g5:solids:def}{arête}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measure:10}
Une piscine est un \omterm{def:g5:solids:def}{pavé} de $25$~m de long, $10$~m de large et $2$~m de
profond.
\begin{enumerate}
  \item Combien de mètres cubes d'eau tient-elle quand elle est
        pleine~?
  \item Combien de litres cela fait-il~?
  \item La piscine est remplie à $50\,000$~L par heure. Combien de
        temps dure le remplissage~?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measure:11}
Un jardin est un carré de côté $20$~m contenant un étang circulaire
de \omterm{def:g3:shapes:shapes}{rayon} $3$~m. Quelle \omterm{def:g6:measure:area}{aire} de gazon y a-t-il (valeur exacte, puis
arrondie au m$^2$)~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:measure:12}
Une tablette de chocolat mesure $15$~cm $\times$ $6$~cm $\times$
$1$~cm. Le fabricant double \emph{les trois} dimensions.
\begin{enumerate}
  \item Par combien le \omterm{def:g6:measure:volume}{volume} est-il multiplié~? Vérifier en
        calculant les deux \omterm{def:g6:measure:volume}{volumes}.
  \item Le prix est multiplié par $4$. La grande tablette est-elle une
        meilleure affaire que la petite~?
\end{enumerate}
\end{exercise}

\section{Problème~: la clôture de la reine Didon}

\begin{problem}[{Devoir maison --- avec une longueur de clôture
fixée, quelle forme enferme le plus de terrain~? Le carré bat tous
les rectangles, le cercle bat le carré, et un mur de grange change
tout}]
\label{pb:g6:measure:1}
La légende raconte que la reine Didon, débarquant sur la côte
d'Afrique, se vit accorder «~autant de terre qu'une peau de bœuf peut
en entourer~» --- elle découpa donc la peau en une seule lanière
immensément longue et fine, et enferma assez de terrain pour fonder
la ville de Carthage. Son problème devient le nôtre~: un fermier
possède exactement $20$ m de clôture. \Cref{exo:g6:measure:8} a
montré que deux enclos de même \omterm{def:g6:measure:perimeter}{périmètre} peuvent avoir des \omterm{def:g6:measure:area}{aires}
différentes~; ce problème cherche le \emph{meilleur} enclos --- et
découvre que la réponse change du tout au tout lorsqu'un mur de
grange offre un côté gratuit.

\medskip
\noindent\textbf{Partie I --- Vingt mètres de clôture.}
\begin{enumerate}
  \item L'enclos doit être un rectangle utilisant les $20$ m de
        clôture. Vérifier qu'un enclos $1 \times 9$ et un enclos
        $2 \times 8$ conviennent tous deux, et calculer leurs \omterm{def:g6:measure:area}{aires}.
  \item Expliquer pourquoi la longueur et la largeur de tout enclos
        convenable ont pour \omterm{def:g1:addition:def}{somme} $10$ m. Dresser ensuite le tableau
        complet des enclos à côtés entiers (de $1 \times 9$ à
        $5 \times 5$) avec leurs \omterm{def:g6:measure:area}{aires}. Lequel est le meilleur~?
  \item Vaut-il la peine d'essayer des côtés décimaux~? Calculer les
        \omterm{def:g6:measure:area}{aires} de l'enclos $4.5 \times 5.5$ et de l'enclos
        $4.9 \times 5.1$, et les comparer au carré $5 \times 5$. Que
        conjecture-t-on~?
  \item La question 2 a transformé le problème de clôture en une
        pure question sur les nombres~: \emph{parmi tous les couples
        de nombres de \omterm{def:g1:addition:def}{somme} $10$, lequel a le plus grand \omterm{def:g2:mult:def}{produit}~?}
        Y répondre à l'aide du tableau, et énoncer la règle générale
        que l'on observe.
  \item Voici pourquoi s'écarter du carré fait toujours perdre, avec
        des ciseaux plutôt qu'avec de l'algèbre~: partir de l'enclos
        $5 \times 5$ et le transformer en l'enclos $6 \times 4$ en
        retirant une bande et en en recollant une autre. Quelle
        bande est retirée, laquelle est ajoutée, et pourquoi
        l'échange fait-il perdre exactement un mètre carré~?
        Expliquer pourquoi une étape de plus (vers $7 \times 3$)
        fait perdre davantage encore.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La grange, et le cercle.}
\begin{enumerate}[resume]
  \item L'enclos est maintenant adossé au long mur d'une grange~: le
        mur remplace une longueur de l'enclos, et les $20$ m de
        clôture ne couvrent que les trois autres côtés. Dresser le
        tableau des enclos à côtés entiers (largeur $w$, longueur
        $L$ le long du mur, $w + L + w = 20$) et de leurs \omterm{def:g6:measure:area}{aires}.
        Quel enclos l'emporte à présent --- et est-ce un carré~?
  \item Expliquer le gagnant à l'aide d'un miroir
        (\cref{ch:g6:symmetry})~: construire le \omterm{def:g6:symmetry:def}{symétrique} de
        l'enclos par rapport au mur de la grange et considérer
        l'enclos doublé. Quelle longueur de clôture l'enclos doublé
        utilise-t-il, quel enclos doublé est le meilleur d'après la
        partie I, et qu'est-ce que cela fait de l'enclos initial~?
  \item Didon n'a pas construit un rectangle. Courber les $20$ m de
        clôture en un cercle~: avec $\pi \approx 3.14$, calculer son
        \omterm{def:g3:shapes:shapes}{rayon} (au cm près), puis son \omterm{def:g6:measure:area}{aire} (au m$^2$ près)
        (\cref{prop:g6:measure:circle},
        \cref{prop:g6:measure:formulas}). Comparer avec le carré
        $5 \times 5$.
  \item Le problème inverse~: on veut un enclos d'\omterm{def:g6:measure:area}{aire} exactement
        $36$ m$^2$, avec le moins de clôture possible. Comparer les
        rectangles $1 \times 36$, $2 \times 18$, $3 \times 12$,
        $4 \times 9$ et $6 \times 6$~: quels \omterm{def:g6:measure:perimeter}{périmètres}~? Quelle
        forme est la moins coûteuse, et en quoi cette question
        est-elle l'image dans le miroir de la partie I~?
  \item En une phrase~: pourquoi les boîtes de conserve, les tuyaux
        et les citernes sont-ils si souvent \emph{ronds}~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- \omterm{def:g6:measure:perimeter}{Périmètre} et \omterm{def:g6:measure:area}{aire} sont deux
étrangers.}
\begin{enumerate}[resume]
  \item Trouver un rectangle dont le \omterm{def:g6:measure:perimeter}{périmètre} dépasse $100$ m mais
        dont l'\omterm{def:g6:measure:area}{aire} est inférieure à $1$ m$^2$. (Très long et très
        fin. Les décimaux sont autorisés.)
  \item Vrai ou faux~: «~de deux figures, celle qui a le plus grand
        \omterm{def:g6:measure:perimeter}{périmètre} a la plus grande \omterm{def:g6:measure:area}{aire}~». Donner un contre-exemple
        tiré de ce problème.
  \item Prendre le rectangle $6 \times 4$ et découper une encoche
        carrée $1 \times 1$ au milieu d'un grand côté (l'encoche
        s'ouvre vers l'extérieur, comme une dent manquante).
        Calculer l'\omterm{def:g6:measure:area}{aire} et le \omterm{def:g6:measure:perimeter}{périmètre} de la figure encochée, et
        comparer les deux à la figure de départ. Qu'est-il arrivé à
        chacun~?
  \item Les cartographes connaissent un fait étrange~: plus la carte
        est détaillée, plus le littoral est \emph{long} --- chaque
        zoom révèle de nouvelles petites encoches, et la question 13
        montre ce que fait chaque encoche. Expliquer en une ou deux
        phrases pourquoi «~la longueur de la côte de Bretagne~» est
        un nombre glissant, alors que «~l'\omterm{def:g6:measure:area}{aire} de la Bretagne~» ne
        l'est pas.
  \item L'examen du fermier. Avec $36$ m de clôture et le mur de la
        grange à disposition, trouver le meilleur enclos
        rectangulaire (utiliser le miroir de la question 7), donner
        son \omterm{def:g6:measure:area}{aire}, et la comparer à celle du meilleur enclos
        construit avec $36$ m en plein champ. Combien le mur de la
        grange rapporte-t-il au fermier~?
\end{enumerate}
\end{problem}
