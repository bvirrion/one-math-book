\chapter{Les nombres entiers}\label{ch:g6:wholes}

Tout en mathématiques commence par le comptage. Ce chapitre revoit
comment les nombres entiers s'écrivent, se comparent et se combinent ---
et regarde de près la division, l'opération qui donne à la fois un
\omterm{def:g3:division:remainder}{quotient} \emph{et} un \omterm{def:g3:division:remainder}{reste}.

\section{Écrire et comparer des entiers}

\begin{definition}[Valeur de position]\label{def:g6:wholes:place}
Notre façon d'écrire les nombres utilise dix chiffres ($0$ à $9$), et
la \emph{valeur d'un chiffre dépend de sa place}\index{valeur de
position}~: dans $5\,208$, le chiffre $5$ compte les milliers, le $2$
les centaines, le $0$ les dizaines (il n'y en a pas) et le $8$ les
unités~:
\[
5\,208 = 5 \times 1000 + 2 \times 100 + 0 \times 10 + 8 .
\]
\end{definition}

\begin{method}[Comparer deux entiers]\label{met:g6:wholes:compare}
\begin{enumerate}
  \item Le nombre avec le plus de chiffres est le plus grand
        ($1\,203 > 989$).
  \item S'ils ont le même nombre de chiffres, on compare chiffre par
        chiffre depuis la \emph{gauche}~; la première \omterm{ex:g1:subtraction:difference}{différence}
        décide~: $6\,742 > 6\,715$ car à la place des dizaines,
        $4 > 1$.
\end{enumerate}
Les symboles sont $<$ («~plus petit que~») et $>$ («~plus grand
que~»)~; le petit côté du symbole pointe vers le petit nombre.
\end{method}

\begin{example}\label{ex:g6:wholes:order}
Ranger du plus petit au plus grand~: $908$~; $89$~; $1\,001$~; $980$.
D'abord par le nombre de chiffres~: $89$ (deux chiffres) vient en
premier, $1\,001$ (quatre chiffres) en dernier. Entre $908$ et
$980$~: même chiffre des centaines $9$, puis dizaines $0 < 8$, donc
$908 < 980$. Réponse finale~:
\[
89 < 908 < 980 < 1\,001 .
\]
\end{example}

\section{Addition, soustraction, multiplication}

\begin{example}[Calculs en colonnes]\label{ex:g6:wholes:columns}
Pour additionner ou soustraire, on aligne les unités sous les unités,
les dizaines sous les dizaines, et on travaille de droite à gauche, en
retenant au besoin~:
\[
\begin{array}{r}
4\,5\,7 \\
+\ 2\,8\,6 \\
\hline
7\,4\,3
\end{array}
\qquad\qquad
\begin{array}{r}
6\,0\,3 \\
-\ 1\,4\,7 \\
\hline
4\,5\,6
\end{array}
\]
Dans l'addition~: $7 + 6 = 13$, on écrit $3$, on retient $1$~; puis
$5 + 8 + 1 = 14$, on écrit $4$, on retient $1$~; puis $4 + 2 + 1 = 7$.
Vérifier la soustraction en additionnant~: $456 + 147 = 603$.
\end{example}

\begin{remark}[Vocabulaire]
Le résultat d'une addition est une \emph{somme}\index{somme}~; d'une
soustraction, une \emph{différence}\index{différence}~; d'une
multiplication, un \emph{produit}\index{produit}~; d'une division, un
\emph{quotient}\index{quotient}. Les nombres combinés sont les
\emph{termes} (pour $+$, $-$) ou \emph{facteurs} (pour $\times$).
\end{remark}

\begin{example}[Calcul malin]\label{ex:g6:wholes:smart}
Réordonner les termes ou facteurs économise souvent du travail~:
\[
17 + 58 + 3 = (17 + 3) + 58 = 20 + 58 = 78 ,
\]
\[
25 \times 17 \times 4 = (25 \times 4) \times 17 = 100 \times 17 = 1700 .
\]
\end{example}

\section{Division}

\begin{theorem}[Division euclidienne]\label{thm:g6:wholes:division}
Étant donnés deux entiers $a$ (le \emph{dividende}) et $b \neq 0$ (le
\emph{\omterm{def:g5:division:multiple}{diviseur}}), il y a exactement une façon d'écrire
\[
a = b \times q + r
\qquad\text{avec}\qquad
0 \leq r < b .
\]
Le nombre $q$ est le \emph{quotient}\index{quotient} et $r$ le
\emph{reste}\index{reste} de la division de $a$ par $b$.
\end{theorem}
\admitted

\begin{example}\label{ex:g6:wholes:division}
Diviser $137$ par $4$, étape par étape.
\begin{enumerate}
  \item Combien de fois $4$ rentre-t-il dans $13$~? Trois fois
        ($4 \times 3 = 12$), \omterm{def:g3:division:remainder}{reste} $13 - 12 = 1$.
  \item On abaisse le $7$~: combien de fois $4$ rentre-t-il dans
        $17$~? Quatre fois ($4 \times 4 = 16$), \omterm{def:g3:division:remainder}{reste} $1$.
\end{enumerate}
Donc $q = 34$ et $r = 1$~:
\[
137 = 4 \times 34 + 1, \qquad\text{et bien } 0 \leq 1 < 4 .
\]
Vérification~: $4 \times 34 = 136$, et $136 + 1 = 137$.
\end{example}

\end{omfigure}

\begin{definition}[Divisible]\label{def:g6:wholes:divisible}
Quand le \omterm{def:g3:division:remainder}{reste} est $0$, on dit que $a$ est
\emph{divisible}\index{divisible} par $b$~: par exemple $135$ est
divisible par $5$ car $135 = 5 \times 27 + 0$.
\end{definition}

\begin{method}[Problèmes de division en mots]\label{met:g6:wholes:problems}
Après avoir calculé $a = bq + r$, toujours revenir à la question~:
\begin{enumerate}
  \item «~Combien de boîtes/équipes/pages complètes~?~» --- la
        réponse est le \omterm{def:g3:division:remainder}{quotient} $q$~;
  \item «~Combien reste-t-il~?~» --- la réponse est le \omterm{def:g3:division:remainder}{reste} $r$~;
  \item «~Combien de boîtes faut-il pour tout le monde~?~» --- si
        $r > 0$, une de \emph{plus} que le \omterm{def:g3:division:remainder}{quotient}~: $q + 1$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:wholes:bus}
$137$ élèves partent en voyage en bus de $40$ places. Division~:
$137 = 40 \times 3 + 17$. Trois bus transportent $120$ élèves, et
$17$ élèves restent --- il leur faut un bus aussi~! Donc $4$ bus sont
nécessaires, même si le \omterm{def:g3:division:remainder}{quotient} est $3$.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g6:wholes:1}
Écrire en chiffres~: «~trois mille quarante-sept~»~; «~vingt mille
cinq cents~»~; «~cent quatre~». Puis écrire $60\,013$ en lettres.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:2}
Dans le nombre $47\,209$~: quel est le chiffre des centaines~? Celui
des unités~? Que compte le chiffre $4$~? Écrire le nombre comme une
\omterm{def:g1:addition:def}{somme} de \omterm{def:g5:division:multiple}{multiples} de $1$, $10$, $100$, \dots\ comme dans
\cref{def:g6:wholes:place}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:3}
Recopier et compléter avec $<$ ou $>$~:
\[
507 \;?\; 570, \qquad
1\,099 \;?\; 989, \qquad
23\,456 \;?\; 23\,465, \qquad
9\,999 \;?\; 10\,000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:4}
Ranger du plus petit au plus grand~: $2\,043$~; $978$~; $2\,304$~;
$2\,034$~; $1\,001$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:5}
Calculer en colonnes~: $348 + 576$~; \quad $805 - 268$~; \quad
$307 \times 26$. Vérifier la soustraction avec une addition.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:6}
Calculer malinement, en regroupant termes ou facteurs~:
\[
38 + 45 + 12 + 5, \qquad
2 \times 83 \times 50, \qquad
4 \times 78 \times 25 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:7}
Pour chaque division, donner le \omterm{def:g3:division:remainder}{quotient} et le \omterm{def:g3:division:remainder}{reste}, et écrire
l'égalité $a = b \times q + r$~:
\[
95 \div 7, \qquad
230 \div 6, \qquad
504 \div 8 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:8}
Sans diviser, expliquer pourquoi l'égalité $173 = 8 \times 20 + 13$
\emph{n'est pas} la division euclidienne de $173$ par $8$, et trouver
le \omterm{def:g3:division:remainder}{quotient} et le \omterm{def:g3:division:remainder}{reste} corrects.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:9}
Les œufs sont emballés par boîtes de $12$.
\begin{enumerate}
  \item Une ferme a $2\,000$ œufs. Combien de boîtes pleines
        peut-elle remplir, et combien d'œufs restent~?
  \item Une boulangerie a besoin de $150$ œufs. Combien de boîtes
        doit-elle acheter~?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:10}
Aujourd'hui, c'est mardi. Quel jour de la semaine sera-t-ce dans
$100$ jours~? (Les semaines ont $7$ jours~: diviser et penser au
\omterm{def:g3:division:remainder}{reste}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:11}
Dans une division par $9$, le \omterm{def:g3:division:remainder}{quotient} est $56$ et le \omterm{def:g3:division:remainder}{reste} est le
plus grand possible. Quel est le dividende~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:wholes:12}
Je suis un nombre à trois chiffres. Mon chiffre des centaines est le
double de mon chiffre des unités, mon chiffre des dizaines est $0$, et
la \omterm{def:g1:addition:def}{somme} de mes chiffres est $9$. Qui suis-je~? (Raisonner étape par
étape, puis vérifier.)
\end{exercise}

\section{Problème~: le petit Gauss et la somme des cent premiers
nombres}

\begin{problem}[{Devoir maison --- additionner $1 + 2 + 3 + \dots +
100$ en dix secondes, et les nombres triangulaires}]
\label{pb:g6:wholes:1}
Une histoire célèbre~: pour occuper sa classe, un maître d'école
demanda à ses élèves d'additionner tous les nombres de $1$ à $100$.
Un garçon écrivit $5\,050$ sur son ardoise presque aussitôt --- Carl
Friedrich Gauss, neuf ans, qui allait devenir l'un des plus grands
mathématiciens de tous les temps. Son secret n'était rien d'autre
qu'un \emph{calcul malin} (\cref{ex:g6:wholes:smart})~: regrouper les
termes pour que chaque groupe soit facile. Ce problème redécouvre son
astuce, la rend valable dans tous les cas, et montre à quel point ces
\omterm{def:g1:addition:def}{sommes} sont fréquentes dans la vie courante.

\medskip
\noindent\textbf{Partie I --- Les paires de Gauss.}
\begin{enumerate}
  \item Calculer $1 + 2 + 3 + 4 + 5$ en regroupant habilement les
        termes.
  \item Pour $1 + 2 + 3 + \dots + 10$~: associer le premier terme au
        dernier ($1 + 10$), le deuxième à l'avant-dernier ($2 + 9$),
        et ainsi de suite. Combien vaut chaque paire~? Combien y
        a-t-il de paires~? Terminer le calcul.
  \item Même méthode pour $1 + 2 + \dots + 20$~: combien vaut chaque
        paire, combien y a-t-il de paires, quel est le total~?
  \item Passer maintenant à la \omterm{def:g1:addition:def}{somme} du maître d'école,
        $1 + 2 + \dots + 100$~: expliquer pourquoi les paires
        $1 + 100$, $2 + 99$, $3 + 98, \dots$ valent toutes la même
        chose, et pourquoi il y en a exactement $50$, la dernière
        étant $50 + 51$.
  \item Écrire la réponse de Gauss sous la forme d'une seule
        multiplication, et la calculer.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Une astuce qui ne rate jamais.}
\begin{enumerate}[resume]
  \item Essayer l'appariement sur $1 + 2 + \dots + 9$~: cette fois,
        un nombre \omterm{def:g3:division:remainder}{reste} sans partenaire. Lequel~? Former les paires
        autour de lui et calculer la \omterm{def:g1:addition:def}{somme}.
  \item Voici une version de l'astuce qui marche que le nombre de
        termes soit \omterm{def:g3:numbers:evenodd}{pair} ou \omterm{def:g3:numbers:evenodd}{impair}~: écrire la \omterm{def:g1:addition:def}{somme} deux fois, une
        fois à l'endroit et une fois à l'envers, l'une sous l'autre,
        \[
        \begin{array}{ccccccc}
        1 & + & 2 & + & \dots & + & 9 \\
        9 & + & 8 & + & \dots & + & 1
        \end{array}
        \]
        puis additionner \emph{colonne par colonne}. Combien vaut
        chaque colonne~? Combien y a-t-il de colonnes~? Expliquer
        pourquoi la \omterm{def:g1:addition:def}{somme} des deux lignes vaut $9 \times 10 = 90$, et
        en déduire que $1 + 2 + \dots + 9 = 45$ --- la même réponse
        qu'à la question 6.
  \item Utiliser l'astuce des deux lignes pour calculer
        $1 + 2 + \dots + 50$, puis $1 + 2 + \dots + 200$.
  \item Calculer la \omterm{def:g1:addition:def}{somme} des cent premiers nombres \emph{\omterm{def:g3:numbers:evenodd}{pairs}},
        $2 + 4 + 6 + \dots + 200$. (Comparer chaque terme à un terme
        de la \omterm{def:g1:addition:def}{somme} de Gauss~: aucun nouvel appariement n'est
        nécessaire.)
  \item Calculer $101 + 102 + \dots + 200$, à l'aide de deux \omterm{def:g1:addition:def}{sommes}
        déjà connues et d'une soustraction.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Les \omterm{pb:g6:wholes:1}{nombres triangulaires} partout.}
Les totaux $1$, $3$, $6$, $10$, $15, \dots$ des \omterm{def:g1:addition:def}{sommes}
$1 + 2 + \dots + n$ s'appellent les \emph{nombres
triangulaires}\index{nombres triangulaires}, parce qu'ils comptent
des objets empilés en triangle.
\begin{enumerate}[resume]
  \item Huit amis se retrouvent, et chacun serre la main de chacun
        des autres exactement une fois. Expliquer pourquoi le nombre
        de poignées de main est $7 + 6 + 5 + 4 + 3 + 2 + 1$, et le
        calculer.
  \item Une épicière construit une \omterm{def:g5:solids:def}{pyramide} triangulaire de boîtes~:
        $12$ boîtes sur la rangée du bas, $11$ sur la suivante, et
        ainsi de suite jusqu'à une seule boîte au \omterm{def:g5:solids:def}{sommet}. Combien lui
        faut-il de boîtes~? (Utiliser l'astuce de la question 7.)
  \item Dessiner la \omterm{def:g1:addition:def}{somme} $1 + 2 + 3 + 4$ comme un escalier de
        carreaux~: un carreau dans la première colonne, deux dans la
        deuxième, trois dans la troisième, quatre dans la quatrième.
        Montrer sur le dessin que \emph{deux} escaliers de ce type,
        dont l'un est retourné, s'emboîtent exactement dans un
        rectangle de $4$ rangées de $5$ carreaux --- et expliquer
        pourquoi c'est l'astuce des deux lignes de la question 7,
        dessinée au lieu d'être écrite.
  \item Une horloge sonne $1$ coup à une heure, $2$ coups à deux
        heures, \dots, $12$ coups à midi. Combien cela fait-il de
        coups sur les douze heures~? Et sur une journée entière~?
  \item Une dernière pour Gauss~: $5\,050$ secondes, est-ce plus ou
        moins qu'une heure et demie~? Convertir cette durée en
        heures, minutes et secondes, en utilisant deux fois la
        division euclidienne (\cref{thm:g6:wholes:division}).
\end{enumerate}
\end{problem}
