\chapter{Los números enteros}\label{ch:g6:wholes}

Todo en matemáticas empieza por contar. Este capítulo repasa cómo
se escriben, se comparan y se combinan los números enteros --- y mira con
atención la división, la operación que da a la vez un \omterm{def:g3:division:remainder}{cociente} \emph{y} un
\omterm{def:g3:division:remainder}{resto}.

\section{Escribir y comparar números enteros}

\begin{definition}[Valor posicional]\label{def:g6:wholes:place}
Nuestra manera de escribir los números usa diez cifras (del $0$ al $9$), y el
\emph{valor de una cifra depende de su lugar}\index{valor posicional}: en
$5\,208$, la cifra $5$ cuenta millares, el $2$ cuenta centenas, el
$0$ cuenta decenas (no hay ninguna) y el $8$ cuenta unidades:
\[
5\,208 = 5 \times 1000 + 2 \times 100 + 0 \times 10 + 8 .
\]
\end{definition}

\begin{method}[Comparar dos números enteros]\label{met:g6:wholes:compare}
\begin{enumerate}
  \item el número con más cifras es el mayor
        ($1\,203 > 989$).
  \item si tienen el mismo número de cifras, compara cifra a cifra
        desde la \emph{izquierda}; la primera \omterm{ex:g1:subtraction:difference}{diferencia} decide:
        $6\,742 > 6\,715$ porque, en las decenas, $4 > 1$.
\end{enumerate}
Los símbolos son $<$ («menor que») y $>$ («mayor que»); el
extremo estrecho del símbolo apunta al número pequeño.
\end{method}

\begin{example}\label{ex:g6:wholes:order}
Ordena de menor a mayor: $908$; $89$; $1\,001$; $980$.
Primero por el número de cifras: $89$ (dos cifras) va delante y $1\,001$ (cuatro
cifras) al final. Entre $908$ y $980$: misma cifra de las centenas, $9$, y luego
decenas $0 < 8$, así que $908 < 980$. Respuesta final:
\[
89 < 908 < 980 < 1\,001 .
\]
\end{example}

\section{Suma, resta y multiplicación}

\begin{example}[Cálculos en columna]\label{ex:g6:wholes:columns}
Para sumar o restar, alinea las unidades bajo las unidades y las decenas bajo las
decenas, y trabaja de derecha a izquierda llevando cuando haga falta:
\[
\begin{array}{r}
4\,5\,7 \\
+\ 2\,8\,6 \\
\hline
7\,4\,3
\end{array}
\qquad\qquad
\begin{array}{r}
6\,0\,3 \\
-\ 1\,4\,7 \\
\hline
4\,5\,6
\end{array}
\]
En la \omterm{def:g1:addition:def}{suma}: $7 + 6 = 13$, escribe $3$ y lleva $1$; luego
$5 + 8 + 1 = 14$, escribe $4$ y lleva $1$; luego $4 + 2 + 1 = 7$.
Comprueba la resta sumando de vuelta: $456 + 147 = 603$.
\end{example}

\begin{remark}[Vocabulario]
El resultado de una \omterm{def:g1:addition:def}{suma} es una \emph{suma}\index{suma}; el de una resta, una
\emph{diferencia}\index{diferencia}; el de una multiplicación, un
\emph{producto}\index{producto}; el de una división, un
\emph{cociente}\index{cociente}. Los números que se combinan son los
\emph{términos} (para $+$ y $-$) o los \emph{factores} (para $\times$).
\end{remark}

\begin{example}[Calcular con astucia]\label{ex:g6:wholes:smart}
Reordenar los términos o los factores ahorra trabajo a menudo:
\[
17 + 58 + 3 = (17 + 3) + 58 = 20 + 58 = 78 ,
\]
\[
25 \times 17 \times 4 = (25 \times 4) \times 17 = 100 \times 17 = 1700 .
\]
\end{example}

\section{La división}

\begin{theorem}[División euclídea]\label{thm:g6:wholes:division}
Dados dos números enteros $a$ (el \emph{dividendo}) y $b \neq 0$ (el
\emph{\omterm{def:g5:division:multiple}{divisor}}), hay exactamente una manera de escribir
\[
a = b \times q + r
\qquad\text{con}\qquad
0 \leq r < b .
\]
El número $q$ es el \emph{cociente}\index{cociente} y $r$ el
\emph{resto}\index{resto} de la división de $a$ entre $b$.
\end{theorem}
\admitted

\begin{example}\label{ex:g6:wholes:division}
Dividimos $137$ entre $4$, paso a paso.
\begin{enumerate}
  \item ¿cuántas veces cabe $4$ en $13$? Tres veces
        ($4 \times 3 = 12$), con \omterm{def:g3:division:remainder}{resto} $13 - 12 = 1$;
  \item baja el $7$: ¿cuántas veces cabe $4$ en $17$?
        Cuatro veces ($4 \times 4 = 16$), con \omterm{def:g3:division:remainder}{resto} $1$.
\end{enumerate}
Así, $q = 34$ y $r = 1$:
\[
137 = 4 \times 34 + 1, \qquad\text{y, en efecto, } 0 \leq 1 < 4 .
\]
Comprobación: $4 \times 34 = 136$ y $136 + 1 = 137$.
\end{example}

\end{omfigure}

\begin{definition}[Divisible]\label{def:g6:wholes:divisible}
Cuando el \omterm{def:g3:division:remainder}{resto} es $0$, decimos que $a$ es
\emph{divisible}\index{divisible} entre $b$: por ejemplo, $135$ es
divisible entre $5$, ya que $135 = 5 \times 27 + 0$.
\end{definition}

\begin{method}[Problemas con enunciado y división]\label{met:g6:wholes:problems}
Después de calcular $a = bq + r$, vuelve siempre a la pregunta:
\begin{enumerate}
  \item «¿cuántas cajas / equipos / páginas completas?» --- la respuesta es el
        \omterm{def:g3:division:remainder}{cociente} $q$;
  \item «¿cuántos sobran?» --- la respuesta es el \omterm{def:g3:division:remainder}{resto}
        $r$;
  \item «¿cuántas cajas hacen falta para todos?» --- si $r > 0$, una
        \emph{más} que el \omterm{def:g3:division:remainder}{cociente}: $q + 1$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:wholes:bus}
$137$ alumnos van de excursión en autobuses de $40$ plazas. División:
$137 = 40 \times 3 + 17$. Tres autobuses llevan a $120$ alumnos y quedan $17$
alumnos --- ¡que también necesitan autobús! Así que hacen falta $4$ autobuses, aunque
el \omterm{def:g3:division:remainder}{cociente} sea $3$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g6:wholes:1}
Escribe con cifras: «tres mil cuarenta y siete»; «veinte mil
quinientos»; «ciento cuatro». Después escribe $60\,013$ con letras.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:2}
En el número $47\,209$: ¿cuál es la cifra de las centenas? ¿Y la de
las unidades? ¿Qué cuenta la cifra $4$? Escribe el número como \omterm{def:g1:addition:def}{suma} de
\omterm{def:g5:division:multiple}{múltiplos} de $1$, $10$, $100$, \dots\ como en la
\cref{def:g6:wholes:place}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:3}
Copia y completa con $<$ o $>$:
\[
507 \;?\; 570, \qquad
1\,099 \;?\; 989, \qquad
23\,456 \;?\; 23\,465, \qquad
9\,999 \;?\; 10\,000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:4}
Ordena de menor a mayor: $2\,043$; $978$; $2\,304$; $2\,034$;
$1\,001$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:5}
Calcula en columna: $348 + 576$; \quad $805 - 268$; \quad
$307 \times 26$. Comprueba la resta con una \omterm{def:g1:addition:def}{suma}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:6}
Calcula con astucia, agrupando términos o factores:
\[
38 + 45 + 12 + 5, \qquad
2 \times 83 \times 50, \qquad
4 \times 78 \times 25 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:7}
Para cada división, da el \omterm{def:g3:division:remainder}{cociente} y el \omterm{def:g3:division:remainder}{resto}, y escribe la
igualdad $a = b \times q + r$:
\[
95 \div 7, \qquad
230 \div 6, \qquad
504 \div 8 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:8}
Sin dividir, explica por qué la igualdad $173 = 8 \times 20 + 13$ \emph{no} es
la división euclídea de $173$ entre $8$, y halla el
\omterm{def:g3:division:remainder}{cociente} y el \omterm{def:g3:division:remainder}{resto} correctos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:9}
Los huevos se envasan en cajas de $12$.
\begin{enumerate}
  \item una granja tiene $2\,000$ huevos. ¿Cuántas cajas completas puede llenar y
        cuántos huevos le sobran?
  \item una panadería necesita $150$ huevos. ¿Cuántas cajas tiene que comprar?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:10}
Hoy es martes. ¿Qué día de la semana será dentro de $100$ días?
(La semana tiene $7$ días: divide y piensa en el \omterm{def:g3:division:remainder}{resto}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:11}
En una división entre $9$, el \omterm{def:g3:division:remainder}{cociente} es $56$ y el \omterm{def:g3:division:remainder}{resto} es el
mayor posible. ¿Cuál es el dividendo?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:wholes:12}
Soy un número de tres cifras. Mi cifra de las centenas es el doble de la de las unidades,
mi cifra de las decenas es $0$ y la \omterm{def:g1:addition:def}{suma} de mis cifras es $9$. ¿Quién soy?
(Razona paso a paso y comprueba después.)
\end{exercise}

\section{Problema: El pequeño Gauss y la suma de los cien primeros
números}

\begin{problem}[{Problema de fin de semana --- sumar $1 + 2 + 3 + \dots +
100$ en diez segundos y los números triangulares}]
\label{pb:g6:wholes:1}
Una historia famosa: para tener entretenida a su clase, un maestro pidió a sus
alumnos que sumaran todos los números del $1$ al $100$. Un niño escribió
$5\,050$ en su pizarra casi al instante --- Carl Friedrich Gauss, de nueve
años, que llegaría a ser uno de los mayores matemáticos
de todos los tiempos. Su secreto no era otro que \emph{calcular con astucia}
(\cref{ex:g6:wholes:smart}): agrupar los términos de modo que cada grupo
salga fácil. Este problema redescubre su truco, lo hace funcionar en cualquier
situación y muestra lo a menudo que estas \omterm{def:g1:addition:def}{sumas} aparecen en la vida diaria.

\medskip
\noindent\textbf{Parte I --- Las parejas de Gauss.}
\begin{enumerate}
  \item calcula $1 + 2 + 3 + 4 + 5$ agrupando los términos con astucia;
  \item para $1 + 2 + 3 + \dots + 10$: empareja el primer término con el
        último ($1 + 10$), el segundo con el penúltimo
        ($2 + 9$), y así sucesivamente. ¿Cuánto vale cada pareja? ¿Cuántas
        parejas hay? Termina el cálculo;
  \item el mismo método para $1 + 2 + \dots + 20$: ¿cuánto vale cada pareja,
        cuántas parejas hay y cuál es el total?
  \item ahora la \omterm{def:g1:addition:def}{suma} del maestro, $1 + 2 + \dots + 100$: explica
        por qué las parejas $1 + 100$, $2 + 99$, $3 + 98, \dots$ valen
        todas lo mismo y por qué hay exactamente $50$,
        siendo la última $50 + 51$;
  \item escribe la respuesta de Gauss como una sola multiplicación y
        calcúlala.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Un truco que nunca falla.}
\begin{enumerate}[resume]
  \item prueba el emparejamiento con $1 + 2 + \dots + 9$: esta vez un
        número se queda sin pareja. ¿Cuál? Empareja a su alrededor
        y calcula la \omterm{def:g1:addition:def}{suma};
  \item aquí tienes una versión del truco que funciona tanto si el número de
        términos es \omterm{def:g3:numbers:evenodd}{par} como si es \omterm{def:g3:numbers:evenodd}{impar}: escribe la \omterm{def:g1:addition:def}{suma} dos veces, una
        hacia delante y otra hacia atrás, una línea debajo de la otra,
        \[
        \begin{array}{ccccccc}
        1 & + & 2 & + & \dots & + & 9 \\
        9 & + & 8 & + & \dots & + & 1
        \end{array}
        \]
        y \omterm{def:g1:addition:def}{suma} \emph{columna a columna}. ¿Cuánto vale cada columna?
        ¿Cuántas columnas hay? Explica por qué la \omterm{def:g1:addition:def}{suma} de
        las dos líneas es $9 \times 10 = 90$ y deduce que
        $1 + 2 + \dots + 9 = 45$ --- la misma respuesta que en la
        pregunta 6;
  \item usa el truco de las dos líneas para calcular $1 + 2 + \dots + 50$
        y después $1 + 2 + \dots + 200$;
  \item calcula la \omterm{def:g1:addition:def}{suma} de los cien primeros números \emph{\omterm{def:g3:numbers:evenodd}{pares}},
        $2 + 4 + 6 + \dots + 200$. (Compara cada término con un término
        de la \omterm{def:g1:addition:def}{suma} de Gauss: no hace falta un emparejamiento nuevo.)
  \item calcula $101 + 102 + \dots + 200$ usando dos \omterm{def:g1:addition:def}{sumas} que
        ya conoces y una resta.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- \omterm{pb:g6:wholes:1}{Números triangulares} por todas partes.}
Los totales $1$, $3$, $6$, $10$, $15, \dots$ de las \omterm{def:g1:addition:def}{sumas}
$1 + 2 + \dots + n$ se llaman \emph{números
triangulares}\index{números triangulares}, porque cuentan objetos
apilados en triángulo.
\begin{enumerate}[resume]
  \item ocho amigos se reúnen y cada uno da la mano exactamente una vez
        a cada uno de los demás. Explica por qué el número de
        apretones de manos es $7 + 6 + 5 + 4 + 3 + 2 + 1$ y calcúlalo;
  \item una tendera construye una \omterm{def:g5:solids:def}{pirámide} triangular de latas: $12$ latas en
        la fila de abajo, $11$ en la siguiente y así hasta una sola
        lata arriba. ¿Cuántas latas necesita? (Usa el truco de la
        pregunta 7.)
  \item dibuja la \omterm{def:g1:addition:def}{suma} $1 + 2 + 3 + 4$ como una escalera de cuadrados:
        un cuadrado en la primera columna, dos en la segunda, tres
        en la tercera y cuatro en la cuarta. Muestra en tu dibujo
        que \emph{dos} escaleras así, una de ellas puesta boca
        abajo, encajan exactamente en un rectángulo de $4$ filas de
        $5$ cuadrados --- y explica por qué esto es el truco de las dos líneas
        de la pregunta 7, dibujado en vez de escrito;
  \item un reloj da $1$ campanada a la una, $2$ campanadas a las dos,
        \dots, $12$ campanadas a las doce. ¿Cuántas
        campanadas son en esas doce horas? ¿Y en un día entero?
  \item una última para Gauss: $5\,050$ segundos, ¿son más o
        menos que una hora y media? Conviértelos en horas,
        minutos y segundos usando dos veces la división euclídea
        (\cref{thm:g6:wholes:division}).
\end{enumerate}
\end{problem}
