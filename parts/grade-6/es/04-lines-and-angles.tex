\chapter{Rectas, circunferencias y ángulos}\label{ch:g6:lines}

La geometría empieza con una regla, una escuadra y un compás. Este capítulo
fija el vocabulario --- puntos, \omterm{def:g6:lines:objects}{segmentos}, semirrectas, rectas, circunferencias ---,
presenta las dos posiciones especiales de las rectas (\omterm{def:g6:lines:perp}{paralelas} y
\omterm{def:g6:lines:perp}{perpendiculares}) y enseña a medir y a trazar ángulos.

\section{Puntos, segmentos y rectas}

\begin{definition}[Objetos básicos]\label{def:g6:lines:objects}
Por dos puntos distintos $A$ y $B$ pasan:
\begin{itemize}
  \item el \emph{segmento}\index{segmento} $[AB]$: la parte de la recta
        comprendida entre $A$ y $B$ (tiene longitud, que se escribe $AB$);
  \item la \emph{semirrecta} $[AB)$: empieza en $A$, pasa por $B$ y
        continúa indefinidamente;
  \item la \emph{recta}\index{recta} $(AB)$: se prolonga indefinidamente por los dos
        lados. Dos puntos determinan exactamente una recta.
\end{itemize}
Los puntos que están en una misma recta se llaman \emph{alineados}.
\end{definition}

\section{Rectas paralelas y perpendiculares}

\begin{definition}[Perpendiculares, paralelas]\label{def:g6:lines:perp}
Dos rectas son \emph{perpendiculares}\index{rectas perpendiculares} cuando se
cortan formando un \omterm{def:g3:shapes:rightangle}{ángulo recto}; escribimos $d_1 \perp d_2$ y marcamos el \omterm{def:g3:shapes:rightangle}{ángulo
recto} con un cuadradito. Dos rectas son
\emph{paralelas}\index{rectas paralelas} cuando no se encuentran nunca, por mucho que
se prolonguen; escribimos $d_1 \parallel d_2$.
\end{definition}

  \draw[thick, omThm] (-1.8,-1.3) -- (1.8,-0.5)

  \end{scope}

{\small Un cruce en \omterm{def:g3:shapes:rightangle}{ángulo recto} (marcado con el cuadradito) y dos
\omterm{def:g6:lines:perp}{paralelas}: la misma dirección, sin punto de corte.}
\end{omfigure}

\begin{proposition}[Dos hechos útiles]\label{prop:g6:lines:facts}
\begin{enumerate}
  \item si dos rectas son \omterm{def:g6:lines:perp}{perpendiculares} a una tercera recta, son
        \omterm{def:g6:lines:perp}{paralelas} entre sí;
  \item si dos rectas son \omterm{def:g6:lines:perp}{paralelas}, toda recta perpendicular a una es
        perpendicular a la otra.
\end{enumerate}
\end{proposition}
\admitted

\begin{method}[Dibujar con la escuadra]\label{met:g6:lines:setsquare}
Para trazar la perpendicular a una recta $d$ por un punto $P$:
\begin{enumerate}
  \item coloca un borde del \omterm{def:g3:shapes:rightangle}{ángulo recto} de la escuadra sobre $d$;
  \item desliza la escuadra a lo largo de $d$ hasta que el otro borde alcance
        $P$;
  \item traza la recta siguiendo ese borde y marca el \omterm{def:g3:shapes:rightangle}{ángulo recto}.
\end{enumerate}
Para una paralela por $P$: traza una perpendicular a $d$ y después la
perpendicular a \emph{esa} recta por $P$
(\cref{prop:g6:lines:facts}).
\end{method}

\section{Circunferencias}

\begin{definition}[Circunferencia]\label{def:g6:lines:circle}
La \emph{circunferencia}\index{circunferencia} de centro $O$ y \omterm{def:g3:shapes:shapes}{radio} $r$ es el conjunto
de todos los puntos que están exactamente a distancia $r$ de $O$. Un \omterm{def:g6:lines:objects}{segmento} que va del
centro a la circunferencia es un \emph{\omterm{def:g3:shapes:shapes}{radio}}; un \omterm{def:g6:lines:objects}{segmento} que une dos puntos
de la circunferencia pasando por el centro es un \emph{\omterm{def:g4:geometry:circle}{diámetro}} --- su longitud es
$2r$; un \omterm{def:g6:lines:objects}{segmento} que une dos puntos de la circunferencia es una \emph{cuerda}.
\end{definition}

\begin{example}\label{ex:g6:lines:circle}
«Traza la circunferencia de centro $O$ que pasa por $A$»: abre el compás
de $O$ a $A$ --- el \omterm{def:g3:shapes:shapes}{radio} es la distancia $OA$ --- y gira. Todos los
puntos de esta circunferencia están a la misma distancia de $O$ que $A$.
\end{example}

\section{Ángulos}

\begin{definition}[Ángulo]\label{def:g6:lines:angle}
Dos semirrectas $[AB)$ y $[AC)$ con el mismo origen $A$ forman el
\emph{ángulo}\index{ángulo} $\widehat{BAC}$; el punto $A$ es su
\emph{\omterm{def:g5:solids:def}{vértice}} (¡siempre la letra del medio!). Los ángulos se miden en
\emph{grados} ($^\circ$), de $0^\circ$ a $360^\circ$ para una vuelta
completa. Un ángulo es:
\begin{itemize}
  \item \emph{recto} si mide $90^\circ$;
  \item \emph{agudo} si mide menos de $90^\circ$;
  \item \emph{obtuso} si mide entre $90^\circ$ y
        $180^\circ$;
  \item \emph{llano} si mide $180^\circ$ (las dos semirrectas forman
        una recta).
\end{itemize}
\end{definition}

\begin{method}[Medir un ángulo con el transportador]\label{met:g6:lines:protractor}
\begin{enumerate}
  \item coloca el centro del transportador exactamente sobre el \omterm{def:g5:solids:def}{vértice} del
        ángulo;
  \item alinea su línea de $0^\circ$ con un lado del ángulo;
  \item lee la graduación por la que pasa el otro lado --- usando la
        escala que empieza en $0$ del lado alineado;
  \item comprueba a ojo: un ángulo agudo tiene que marcar menos de
        $90^\circ$ y uno obtuso, más.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:lines:estimate}
¡Antes de medir, estima! Un ángulo un poco más abierto que la esquina
de una hoja de papel pasa un poco de $90^\circ$; medio \omterm{def:g3:shapes:rightangle}{ángulo recto} son
$45^\circ$; un tercio de \omterm{def:g3:shapes:rightangle}{ángulo recto} son $30^\circ$. Si tu transportador
marca $150^\circ$ para un ángulo que parece agudo, has leído la escala
equivocada: la medida correcta es $180^\circ - 150^\circ = 30^\circ$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g6:lines:1}
Traza tres puntos $A$, $B$ y $C$ no alineados. Traza en colores distintos:
el \omterm{def:g6:lines:objects}{segmento} $[AB]$, la semirrecta $[CA)$ y la recta $(BC)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:2}
¿Verdadero o falso? «$[AB]$ y $[BA]$ son el mismo \omterm{def:g6:lines:objects}{segmento}.»
«$[AB)$ y $[BA)$ son la misma semirrecta.» «$(AB)$ y $(BA)$ son la
misma recta.» Explica cada respuesta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:3}
Traza una recta $d$ y un punto $P$ que no esté en $d$. Construye con la
escuadra: la perpendicular a $d$ por $P$ y después la paralela a $d$
por $P$. Describe tus pasos.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:4}
Las rectas $a$ y $b$ son las dos \omterm{def:g6:lines:perp}{perpendiculares} a una recta $c$, y una cuarta
recta $e$ es perpendicular a $a$. ¿Qué puedes decir de $a$ y $b$?
¿Y de $e$ y $c$? Justifícalo con la \cref{prop:g6:lines:facts}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:5}
Traza una circunferencia de centro $O$ y \omterm{def:g3:shapes:shapes}{radio} $3$ cm. Sitúa un punto $M$ sobre
la circunferencia, un punto $N$ dentro y un punto $P$ fuera. ¿Qué puedes decir
de las distancias $OM$, $ON$ y $OP$ comparadas con $3$ cm?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:6}
Una circunferencia tiene \omterm{def:g4:geometry:circle}{diámetro} $9$ cm. ¿Cuál es su \omterm{def:g3:shapes:shapes}{radio}? Otra tiene \omterm{def:g3:shapes:shapes}{radio}
$2.6$ cm: ¿cuál es su \omterm{def:g4:geometry:circle}{diámetro}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:7}
Nombra con tres letras el ángulo marcado y clasifícalo (agudo, recto,
obtuso o llano): un ángulo de $72^\circ$ con \omterm{def:g5:solids:def}{vértice} $R$ entre las semirrectas
hacia $S$ y $T$; un ángulo de $148^\circ$ con \omterm{def:g5:solids:def}{vértice} $B$ entre las semirrectas
hacia $A$ y $C$; un ángulo de $90^\circ$ con \omterm{def:g5:solids:def}{vértice} $O$ entre las semirrectas
hacia $M$ y $N$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:8}
Estima y después mide con el transportador los tres ángulos de un
triángulo que dibujes tú mismo. \omterm{def:g1:addition:def}{Suma} las tres medidas: ¿qué obtienes?
(Guarda tu respuesta para el \cref{ch:g7:triangles}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:9}
Traza un ángulo $\widehat{xOy}$ de $65^\circ$ con el transportador y después un
ángulo de $130^\circ$. ¿Cómo podrías obtener el segundo a partir del primero
sin transportador?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:lines:10}
En un mapa aparecen dos pueblos $A$ y $B$. ¿Dónde están los puntos que
están a $3$ cm de $A$ \emph{y} a $2$ cm de $B$? Haz un dibujo
que muestre cuántos puntos así puede haber ($0$, $1$ o $2$, según
la distancia $AB$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:lines:11}
Un reloj marca las 3 en punto: ¿qué ángulo forman las dos agujas? La misma
pregunta a las 5 en punto y a las 6 en punto. (Una vuelta completa son $360^\circ$
para $12$ horas.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:lines:12}
Traza un \omterm{def:g6:lines:objects}{segmento} $[AB]$ de $6$ cm. Construye el punto $C$ tal que
$AC = BC = 6$ cm usando solo el compás, y mide el ángulo
$\widehat{CAB}$. ¿Qué triángulo has construido y qué
conjeturas sobre sus ángulos?
\end{exercise}

\section{Problema: La geometría de la esfera del reloj}

\begin{problem}[{Problema de fin de semana --- los ángulos como fracciones de
vuelta: leerlos en un reloj, cazar los momentos en que las agujas
coinciden y repartir un día en un diagrama de sectores}]
\label{pb:g6:lines:1}
Un reloj es un transportador que da la hora: su esfera es una vuelta
completa de $360^\circ$, cortada por las doce marcas horarias en doce
ángulos iguales. \Cref{exo:g6:lines:11} midió las agujas a las
3 y a las 6 en punto; este problema construye la teoría completa
--- incluidos los momentos en que \emph{ninguna} aguja apunta a una marca ---,
responde a una pregunta que pocos adultos aciertan («¿cuántas veces se ponen
las dos agujas exactamente una encima de la otra?») y termina
repartiendo un día entero en un diagrama de sectores.

\medskip
\noindent\textbf{Parte I --- Fracciones de vuelta.}
\begin{enumerate}
  \item una vuelta completa mide $360^\circ$. ¿Cuántos grados son
        media vuelta, un \omterm{def:g3:division:half}{cuarto} de vuelta y un doceavo de vuelta?
  \item las doce marcas horarias cortan la esfera del reloj en doce
        ángulos iguales en el centro. ¿Cuántos grados hay entre dos
        marcas vecinas? Recupera la respuesta del
        \cref{exo:g6:lines:11} para las 3 en punto contando marcas;
  \item di el ángulo entre las agujas a la 1, a las
        4 y a las 7 en punto. (A las 7 las agujas
        separan la esfera en dos ángulos; «el ángulo entre
        las agujas» significa siempre el \emph{menor}.)
  \item el minutero da una vuelta completa en $60$ minutos. ¿Cuántos
        grados barre por minuto?
  \item la aguja horaria va de una marca a la siguiente ---
        $30^\circ$ --- en $60$ minutos. ¿Cuántos grados barre
        \emph{ella} por minuto? (Un \omterm{def:g6:decimals:places}{número decimal},
        \cref{ch:g6:decimals}.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Momentos en que nada apunta a una
marca.}
\begin{enumerate}[resume]
  \item a las 3:30, el minutero apunta al $6$. ¿Dónde está exactamente
        la aguja horaria? Calcula el ángulo de cada aguja desde el
        $12$ (midiendo en el sentido de las agujas del reloj) y deduce el ángulo entre
        las agujas;
  \item a las 6:30 mucha gente cree que las agujas están una encima de
        la otra. Adivina primero y calcula después el ángulo como en la pregunta
        6. ¿Quién tenía razón?
  \item calcula el ángulo entre las agujas a las 9:15;
  \item explica por qué, en algún momento entre la 1:00 y la 1:10, las dos
        agujas tienen que estar exactamente una encima de la otra: ¿dónde está el
        minutero respecto de la aguja horaria a la 1:00 y dónde
        a la 1:10? ¿Qué aguja va más deprisa y por qué eso lo resuelve?
  \item en doce horas, ¿cuántas veces se ponen las dos agujas
        exactamente una encima de la otra? (Hay un encuentro en
        cada tramo entre dos horas sucesivas --- con una
        excepción: ¿qué pasa entre las 11 y las 12? Enumera las horas
        aproximadas de los encuentros y cuéntalas.) ¿Y en un
        día entero?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Ángulos alrededor de un punto: diagramas
de sectores.}
\begin{enumerate}[resume]
  \item los doce sectores de la esfera del reloj llenan juntos la
        esfera: sus ángulos suman $360^\circ$. Explica por qué
        esto vale para \emph{cualquier} colección de ángulos que
        comparten \omterm{def:g5:solids:def}{vértice} y llenan una vuelta completa a su alrededor, sin
        solaparse y sin dejar huecos;
  \item Zoe anota su día de $24$ horas: dormir $9$ h, colegio
        $6$ h, juego $3$ h, comidas $2$ h y todo lo demás $4$ h.
        Quiere un \emph{diagrama de sectores}: un disco en el que cada actividad
        recibe un sector, con ángulos proporcionales a los tiempos.
        ¿Qué \omterm{def:g6:fractions:def}{fracción} del día es cada actividad y cuántos
        grados recibe su sector? Comprueba que los cinco ángulos suman
        $360^\circ$;
  \item describe, paso a paso, cómo dibujar el diagrama de Zoe con
        compás y transportador
        (\cref{met:g6:lines:protractor}): dónde va el primer \omterm{def:g3:shapes:shapes}{radio}
        y cómo cada sector nuevo empieza donde termina el
        anterior;
  \item en otro diagrama de sectores de un día, un sector mide
        exactamente $90^\circ$. ¿Qué \omterm{def:g6:fractions:def}{fracción} del disco es
        y cuántas horas representa?
  \item el final, de vuelta al reloj: en $24$ horas, ¿cuántas
        vueltas completas da la aguja horaria? ¿Y el minutero? ¿Y el
        segundero? (Una de estas respuestas pasa de mil.)
\end{enumerate}
\end{problem}
