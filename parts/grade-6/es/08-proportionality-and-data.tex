\chapter{Proporcionalidad y datos}\label{ch:g6:propdata}

Si tres cuadernos cuestan $6$ euros, seis cuadernos cuestan $12$: el doble de
cuadernos, el doble de precio. Las magnitudes que se comportan así son
\emph{\omterm{def:g6:propdata:def}{proporcionales}} --- una de las ideas más útiles de todas las
matemáticas, que se desarrolla más en el \cref{ch:g7:prop}. El capítulo termina
con la lectura y el trazado de gráficos de datos sencillos.

\section{Magnitudes proporcionales}

\begin{definition}[Proporcionalidad]\label{def:g6:propdata:def}
Dos magnitudes son \emph{proporcionales}\index{proporcionalidad} cuando los
valores de una se obtienen de los valores de la otra multiplicando
siempre por el \emph{mismo} número, llamado \emph{coeficiente de
proporcionalidad}.
\end{definition}

\begin{example}\label{ex:g6:propdata:table}
Cuadernos a $2$ euros cada uno:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cccc}
cuadernos & $3$ & $5$ & $8$ & $12$ \\
\hline
precio (euros) & $6$ & $10$ & $16$ & $24$ \\
\end{tabular}
\end{center}
Cada precio es el número de cuadernos $\times 2$: el coeficiente es
$2$ (el precio unitario). Una comprobación rápida de que una tabla es proporcional:
todos los «\omterm{def:g3:division:remainder}{cocientes} por columnas» $\frac{6}{3}, \frac{10}{5}, \frac{16}{8},
\frac{24}{12}$ tienen que ser iguales --- aquí todos valen $2$.
\end{example}

\begin{example}[Un contraejemplo]\label{ex:g6:propdata:nonexample}
La edad y la estatura no son \omterm{def:g6:propdata:def}{proporcionales}: un niño de $12$ años no mide el doble
que uno de $6$. Compruébalo con números: $1.50$ m a los $12$ años y
$1.15$ m a los $6$ dan \omterm{def:g3:division:remainder}{cocientes} $\frac{1.50}{12} = 0.125$ y
$\frac{1.15}{6} \approx 0.19$ --- no son iguales.
\end{example}

\begin{method}[Completar una tabla de proporcionalidad]\label{met:g6:propdata:complete}
Tres caminos, a elegir libremente:
\begin{enumerate}
  \item \emph{coeficiente}: halla el multiplicador a partir de una columna
        completa y aplícalo a las demás;
  \item \emph{columnas}: una columna se puede multiplicar por un número, o se pueden sumar
        dos columnas, para producir una columna nueva;
  \item \emph{reducción a la unidad}: halla primero el valor de \emph{un}
        artículo y multiplica después.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:propdata:methods}
Cinco tazas iguales cuestan $15$ euros; ¿cuánto cuestan ocho tazas?

\emph{Reducción a la unidad}: una taza cuesta $15 \div 5 = 3$ euros, así que ocho
tazas cuestan $8 \times 3 = 24$ euros.

\emph{Columnas}: $8 = 5 + 3$; tres tazas cuestan $\frac{3}{5}$ de $15$,
o sea, $9$; así que ocho tazas cuestan $15 + 9 = 24$ euros. La misma respuesta, como
tenía que ser.
\end{example}

\section{Porcentajes}

\begin{definition}[Porcentaje]\label{def:g6:propdata:percent}
«El $t\,\%$ de una cantidad» significa la \omterm{def:g6:fractions:def}{fracción} $\frac{t}{100}$ de ella:
el $25\,\%$ de $60$ es $\frac{25}{100} \times 60 = 15$. Un porcentaje es una
\omterm{def:g6:propdata:def}{proporcionalidad} de coeficiente $\frac{t}{100}$.
\end{definition}

\begin{example}\label{ex:g6:propdata:percent}
Una clase de $30$ alumnos tiene un $40\,\%$ de chicas. Número de chicas, paso
a paso: $40\,\%$ significa $\frac{40}{100} = 0.4$, y
$0.4 \times 30 = 12$ chicas. Referencias útiles:
$50\,\%$ es la \omterm{def:g3:division:half}{mitad}, $25\,\%$ un \omterm{def:g3:division:half}{cuarto} y $10\,\%$ una décima parte ---
así que el $10\,\%$ de $30$ es $3$ y el $40\,\% = 4 \times 10\,\%$ es
$4 \times 3 = 12$: la misma respuesta, calculada de cabeza.
\end{example}

\section{Leer y dibujar gráficos}

\begin{method}[Leer una tabla o un gráfico]\label{met:g6:propdata:read}
Sea cual sea la imagen --- tabla, diagrama de barras o gráfico de líneas:
\begin{enumerate}
  \item lee primero el título y las unidades de los dos ejes;
  \item para responder a una pregunta, localiza la barra o la columna correcta y lee
        el valor con cuidado sobre la graduación;
  \item para comparar, compara las barras por su altura --- pero comprueba que el eje
        empieza en $0$, o la imagen puede engañar.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:propdata:chart}
El diagrama de barras siguiente muestra el número de libros prestados en la biblioteca
del colegio durante una semana.

  xtick=data,
  ymin=0, ymax=22,

{\small Un diagrama de barras: una barra por día, con la altura igual al recuento.}

Lecturas: el día de más movimiento es el miércoles ($17$ libros). El martes y el
jueves juntos suman $8 + 6 = 14$ libros --- menos que el viernes
solo ($15$). Total de la semana:
$12 + 8 + 17 + 6 + 15 = 58$ libros.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g6:propdata:1}
¿Es proporcional la tabla? Justifícalo calculando \omterm{def:g3:division:remainder}{cocientes}.
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccc}
kg de manzanas & $2$ & $3$ & $5$ \\
\hline
precio (euros) & $5$ & $7.5$ & $12.5$ \\
\end{tabular}
\qquad
\begin{tabular}{l|ccc}
edad (años) & $2$ & $4$ & $8$ \\
\hline
estatura (cm) & $85$ & $103$ & $130$ \\
\end{tabular}
\end{center}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:2}
Cuatro cruasanes cuestan $4.80$ euros. Halla el precio de un cruasán
y después el de siete cruasanes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:3}
Completa la tabla de \omterm{def:g6:propdata:def}{proporcionalidad} (¡el coeficiente primero!):
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|cccc}
litros de combustible & $10$ & $25$ & $?$ & $60$ \\
\hline
precio (euros) & $18$ & $?$ & $72$ & $?$ \\
\end{tabular}
\end{center}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:4}
Un coche gasta $6$ L de combustible cada $100$ km. ¿Cuánto combustible para $250$ km? ¿Y para
$350$ km? ¿Hasta dónde puede llegar con $27$ L?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:5}
Calcula de cabeza usando las referencias del \cref{ex:g6:propdata:percent}:
el $50\,\%$ de $84$; \quad el $25\,\%$ de $200$; \quad el $10\,\%$ de $63$;
\quad el $30\,\%$ de $70$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:6}
En un colegio de $450$ alumnos, el $60\,\%$ come en el comedor. ¿Cuántos
alumnos son? ¿Cuántos no comen allí?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:7}
Con el diagrama de barras del \cref{ex:g6:propdata:chart}: ¿qué días se prestaron
menos de $10$ libros? ¿Cuántos libros más se prestaron el
miércoles que el jueves?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:propdata:8}
Las temperaturas a mediodía de lunes a viernes fueron $14$, $16$, $13$,
$17$ y $20$ grados. Dibuja un diagrama de barras con estos datos (elige una graduación
razonable) y di cuál fue el día más caluroso.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:propdata:9}
Una receta para $6$ personas necesita $450$ g de harina y $3$ huevos. Adáptala
para $10$ personas. (Para los huevos, ¡piensa antes de escribir un \omterm{def:g6:decimals:places}{número decimal}
de huevos!)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:propdata:10}
A velocidad constante, una ciclista recorre $24$ km en $1$ hora.
\begin{enumerate}
  \item ¿cuánto recorre en $2$ h? ¿Y en media hora? ¿Y en
        $1$ h $30$ min?
  \item ¿cuánto tarda en hacer $60$ km?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:propdata:11}
Una tienda anuncia «$20\,\%$ de descuento en todo». Calcula el descuento y
el precio nuevo de: un balón de $15$ euros; una raqueta de $40$ euros. ¿Es el
precio nuevo proporcional al precio antiguo? ¿Cuál es el coeficiente?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:propdata:12}
Dos velas de la misma altura se encienden a la vez. La gruesa
se consume entera en $6$ horas y la fina en $4$ horas, cada una a
ritmo constante. ¿Al cabo de cuánto tiempo mide la vela fina exactamente la
\omterm{def:g3:division:half}{mitad} que la gruesa? (Expresa las alturas restantes al cabo de
$t$ horas como fracciones de la altura inicial.)
\end{exercise}

\section{Problema: Mapas, planos y cómo mentir con un gráfico}

\begin{problem}[{Problema de fin de semana --- la escala de un mapa es una
proporcionalidad: las longitudes siguen al coeficiente, las áreas siguen a su
cuadrado y los gráficos mal dibujados engañan a la vista}]
\label{pb:g6:propdata:1}
Todo mapa, plano y maqueta obedece a una regla: las longitudes reales y las
longitudes dibujadas son \emph{\omterm{def:g6:propdata:def}{proporcionales}}
(\cref{def:g6:propdata:def}). El coeficiente tiene un nombre famoso
--- la \emph{escala}\index{escala} --- y esconde dos trampas que
este problema hace saltar a propósito: las \omterm{def:g6:measure:area}{áreas} \emph{no} siguen a la
escala, y los porcentajes, como los gráficos, dependen por completo de aquello
con lo que se miden.

\medskip
\noindent\textbf{Parte I --- Leer un mapa.}
Un mapa de senderismo está a escala $1:100\,000$: cada centímetro del mapa
representa $100\,000$ centímetros sobre el terreno.
\begin{enumerate}
  \item convierte $100\,000$ cm en kilómetros. ¿Qué representa $1$ cm
        de este mapa, en km?
  \item dos pueblos distan $7.5$ cm en el mapa; la orilla de
        un lago es una curva de $12$ cm. Da las distancias reales;
  \item una calzada romana perfectamente recta mide $20$ km. ¿Cuánto mide
        en el mapa?
  \item otro mapa anuncia «$1$ cm por $5$ km». Escribe su
        escala en la forma $1: \,?$. ¿Cuál de los dos mapas muestra
        más detalle de una misma región?
  \item haz una tabla pequeña (distancia en el mapa $1$, $2$, $7.5$ cm
        frente a distancia real) para el mapa de senderismo y explica
        por qué una escala es exactamente una \omterm{def:g6:propdata:def}{proporcionalidad} en el sentido de la
        \cref{def:g6:propdata:def}. ¿Cuál es el coeficiente que
        convierte los centímetros del mapa en kilómetros?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El plano de la habitación de Zoe.}
Zoe dibuja el plano de su habitación --- un rectángulo de $4$ m por
$3$ m --- a escala $1:50$.
\begin{enumerate}[resume]
  \item ¿qué dimensiones tiene la habitación en el plano?
  \item su cama mide $2$ m por $0.9$ m y el hueco de la puerta tiene
        $80$ cm de ancho. Da las tres medidas en el plano;
  \item ahora, la trampa. Calcula el \omterm{def:g6:measure:area}{área} real de la habitación en
        m$^2$ y después el \omterm{def:g6:measure:area}{área} de su plano en cm$^2$. Convierte el
        \omterm{def:g6:measure:area}{área} real a cm$^2$
        (\cref{def:g6:measure:area}) y divide entre el \omterm{def:g6:measure:area}{área}
        del plano: ¿es el \omterm{def:g3:division:remainder}{cociente} $50$?
  \item explica el número que has obtenido: en un plano $1:50$, cada
        centímetro cuadrado de papel representa un cuadrado real de
        $50$ cm por $50$ cm. ¿Cuántos cm$^2$ reales son? Enuncia
        la regla: cuando las longitudes se dividen entre $50$, las \omterm{def:g6:measure:area}{áreas} se
        dividen entre\,\dots
  \item una alfombra redonda cubre $6$ cm$^2$ en el plano. ¿Qué \omterm{def:g6:measure:area}{área}
        real cubre, en cm$^2$ y después en m$^2$?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Cómo mentir con un gráfico (y con
un porcentaje).}
\begin{enumerate}[resume]
  \item una tienda vendió por valor de $105$ euros el sábado y $110$ el
        domingo. El gerente dibuja un diagrama de barras cuyo eje
        vertical empieza en $100$: la barra del sábado sube $5$ unidades
        pequeñas y la del domingo, $10$. ¿Qué sugiere la imagen
        sobre el domingo y cuál es la verdad? (Calcula
        el aumento del domingo como porcentaje del sábado, al
        entero más cercano.) ¿Qué advertencia del
        \cref{met:g6:propdata:read} ignoró el gerente?
  \item vuelve a dibujar (o describe) el gráfico honesto, con el eje
        empezando en $0$: ¿cómo se comparan ahora las dos barras?
  \item una colección pasa de $40$ a $60$ sellos: calcula el
        aumento como porcentaje del valor de partida. Después vuelve a
        encoger de $60$ a $40$: calcula la disminución como
        porcentaje de \emph{su} valor de partida. ¿Por qué son distintas las
        dos respuestas para los mismos $20$ sellos?
  \item época de rebajas: un abrigo de $100$ euros lleva un «$20\,\%$
        de descuento» y, en caja, un «$10\,\%$ más sobre el
        precio rebajado». Calcula el precio final paso a paso.
        ¿Es el descuento total del $30\,\%$? Explícale al comprador
        cuál es de verdad;
  \item el final, de vuelta al mapa de la pregunta 4 ($1$ cm por
        $5$ km): un bosque ocupa $3$ cm$^2$ de ese mapa. ¿Cuál
        es su \omterm{def:g6:measure:area}{área} real, en km$^2$? Termina con la regla de
        todo este problema: las longitudes siguen a la escala y las \omterm{def:g6:measure:area}{áreas}
        siguen a su\,\dots
\end{enumerate}
\end{problem}
