\chapter{Perímetro, área y volumen}\label{ch:g6:measure}

¿Cuánto mide la valla, cuán grande es el campo, cuánta agua cabe en el
depósito? Tres preguntas distintas, tres magnitudes distintas ---
\omterm{def:g6:measure:perimeter}{perímetro}, \omterm{def:g6:measure:area}{área} y \omterm{def:g6:measure:volume}{volumen} ---, cada una con sus unidades. Confundirlas es
el error más frecuente en geometría; este capítulo las separa de una vez por
todas.

\section{Longitudes y perímetro}

\begin{definition}[Perímetro]\label{def:g6:measure:perimeter}
El \emph{perímetro}\index{perímetro} de una figura es la longitud total
de su borde. Se mide en unidades de longitud: milímetros (mm),
centímetros (cm), metros (m) y kilómetros (km), con
\[
1 \text{ km} = 1000 \text{ m}, \qquad
1 \text{ m} = 100 \text{ cm}, \qquad
1 \text{ cm} = 10 \text{ mm}.
\]
\end{definition}

\begin{example}\label{ex:g6:measure:perimeter}
Un rectángulo de largo $L$ y ancho $a$ tiene \omterm{def:g6:measure:perimeter}{perímetro}
\[
P = L + w + L + w = 2 \times (L + w).
\]
Para un rectángulo de $7$ cm por $4$ cm: $P = 2 \times (7 + 4) = 2 \times 11 =
22$ cm. Un cuadrado de lado $c$ tiene \omterm{def:g6:measure:perimeter}{perímetro} $4c$.
\end{example}

\begin{proposition}[Longitud de una circunferencia]\label{prop:g6:measure:circle}
El \omterm{def:g6:measure:perimeter}{perímetro} (o \emph{longitud}) de una circunferencia de \omterm{def:g3:shapes:shapes}{radio} $r$ es
\[
P = 2 \pi r ,
\]
donde $\pi \approx 3.14$ es el mismo número para todas las circunferencias.
\end{proposition}
\admitted

\begin{example}
Un estanque circular tiene \omterm{def:g3:shapes:shapes}{radio} $5$ m. Su borde mide
$2 \times \pi \times 5 = 10\pi \approx 31.4$ m. Conserva el valor exacto
$10\pi$ todo lo que puedas; redondea solo al final.
\end{example}

\section{Áreas}

\begin{definition}[Área]\label{def:g6:measure:area}
El \emph{área}\index{área} de una figura mide la superficie que cubre:
cuántos cuadrados unidad caben dentro. Unidades: cm$^2$ (un cuadrado de lado
$1$ cm), m$^2$, km$^2$, \dots\ Cuidado:
\[
1 \text{ m}^2 = 100 \times 100 \text{ cm}^2 = 10\,000 \text{ cm}^2
\]
--- un metro cuadrado es un cuadrado de $100$ cm por $100$ cm, así que cada escalón de
la escalera de unidades vale $100$, no $10$.
\end{definition}

\end{omfigure}

\begin{proposition}[Fórmulas básicas de área]\label{prop:g6:measure:formulas}
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{l|l}
figura & \omterm{def:g6:measure:area}{área} \\
\hline
rectángulo ($L \times a$) & $A = L \times a$ \\
cuadrado (lado $c$) & $A = c^2$ \\[4pt]
\omterm{def:g6:shapes:triangles}{triángulo rectángulo} (catetos $a$, $b$) & $A = \dfrac{a \times b}{2}$ \\[4pt]
disco (\omterm{def:g3:shapes:shapes}{radio} $r$) & $A = \pi r^2$ \\
\end{tabular}
\end{center}
\end{proposition}

\begin{proof}[Demostración para el triángulo rectángulo]
Dos copias de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} de catetos $a$ y $b$, pegadas por la
hipotenusa, forman un rectángulo de $a \times b$. Así que el \omterm{def:g6:measure:area}{área} del triángulo es la
\omterm{def:g3:division:half}{mitad} de la del rectángulo: $\frac{ab}{2}$. (La fórmula del rectángulo es el
recuento de cuadrados unidad de arriba; la del disco se admite, véase el
\cref{ch:g7:areas}.)
\end{proof}

  \draw[thick, omExo, dashed] (4,0) -- (4,2.5) -- (0,2.5);
  \draw[thin] (0.3,0) -- (0.3,0.3) -- (0,0.3);

  \node[left, font=\small] at (0,1.25) {$b$};
  \node[font=\small, omExo] at (2.8,1.7) {segunda copia};
\end{tikzpicture}

{\small Un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} es la \omterm{def:g3:division:half}{mitad} de un rectángulo: de ahí la fórmula
$\frac{a\times b}{2}$.}
\end{omfigure}

\begin{example}[Figuras compuestas]\label{ex:g6:measure:composite}
Una habitación en forma de L es un rectángulo de $6$ m $\times$ $4$ m al que se le ha quitado una
esquina cuadrada de $2$ m $\times$ $2$ m. Su \omterm{def:g6:measure:area}{área}, paso a paso:
\begin{enumerate}
  \item rectángulo completo: $6 \times 4 = 24$ m$^2$;
  \item cuadrado quitado: $2 \times 2 = 4$ m$^2$;
  \item \omterm{def:g6:measure:area}{área} restante: $24 - 4 = 20$ m$^2$.
\end{enumerate}
Su \emph{\omterm{def:g6:measure:perimeter}{perímetro}} no es $20$ de nada: recorriendo la L, el
borde sigue midiendo $6 + 4 + 6 + 4 = 20$ m --- los dos cortes de la
esquina sustituyen a dos trozos de pared iguales. ¡El mismo número por casualidad,
pero metros cuadrados en un caso y metros en el otro!
\end{example}

\begin{remark}[Mismo perímetro, áreas distintas]
Dos figuras pueden tener el mismo \omterm{def:g6:measure:perimeter}{perímetro} y \omterm{def:g6:measure:area}{áreas} muy distintas: un
cuadrado de $5 \times 5$ y un rectángulo de $9 \times 1$ tienen los dos \omterm{def:g6:measure:perimeter}{perímetro}
$20$, pero \omterm{def:g6:measure:area}{áreas} $25$ y $9$. El \omterm{def:g6:measure:perimeter}{perímetro} y el \omterm{def:g6:measure:area}{área} son magnitudes realmente
independientes.
\end{remark}

\section{Volúmenes}

\begin{definition}[Volumen]\label{def:g6:measure:volume}
El \emph{volumen}\index{volumen} de un cuerpo mide el espacio que ocupa:
cuántos \omterm{def:g5:solids:def}{cubos} unidad caben dentro. Unidades: cm$^3$, m$^3$, \dots\ con
$1$ m$^3 = 1\,000\,000$ cm$^3$ (cada escalón de la escalera vale
$1000$). Para los líquidos se usa además el \emph{litro}:
\[
1 \text{ L} = 1 \text{ dm}^3 = 1000 \text{ cm}^3,
\qquad
1 \text{ m}^3 = 1000 \text{ L}.
\]
\end{definition}

\begin{proposition}[Volumen de un ortoedro]\label{prop:g6:measure:box}
Un \omterm{def:g5:solids:def}{ortoedro} (un \emph{prisma rectangular}) de largo $L$, ancho $a$
y altura $h$ tiene \omterm{def:g6:measure:volume}{volumen}
\[
V = L \times w \times h .
\]
\end{proposition}

\begin{proof}
La capa de abajo contiene $L \times a$ \omterm{def:g5:solids:def}{cubos} unidad
(el dibujo de la \cref{def:g6:measure:area}, pero con \omterm{def:g5:solids:def}{cubos}), y hay $h$
capas así.
\end{proof}

{\small Un \omterm{def:g5:solids:def}{ortoedro} lleno de \omterm{def:g5:solids:def}{cubos} unidad: $4 \times 2$ \omterm{def:g5:solids:def}{cubos} por capa, dos
capas.}
\end{omfigure}

\begin{example}\label{ex:g6:measure:aquarium}
Un acuario mide $60$ cm por $30$ cm por $40$ cm (de alto). \omterm{def:g6:measure:volume}{Volumen}:
\[
V = 60 \times 30 \times 40 = 72\,000 \text{ cm}^3 = 72 \text{ dm}^3
= 72 \text{ L}.
\]
Paso a paso para la conversión: $1000$ cm$^3$ hacen un litro, y
$72\,000 \div 1000 = 72$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g6:measure:1}
Convierte: $3.5$ m en cm; $420$ mm en cm; $0.8$ km en m;
$25\,000$ m en km.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:2}
Calcula el \omterm{def:g6:measure:perimeter}{perímetro} de: un rectángulo de $8$ cm $\times$ $3.5$ cm; un
cuadrado de lado $6.2$ cm; un triángulo de lados $5$ cm, $7$ cm y
$9$ cm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:3}
Una pista de atletismo circular tiene \omterm{def:g3:shapes:shapes}{radio} $50$ m. ¿Cuánto mide una vuelta
(valor exacto con $\pi$ y después redondeado al metro)? ¿Cuántas vueltas hacen
al menos $2$ km?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:4}
Calcula el \omterm{def:g6:measure:area}{área} de: un rectángulo de $9$ cm $\times$ $4$ cm; un cuadrado de
lado $7$ m; un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} de catetos $6$ cm y $10$ cm; un disco de
\omterm{def:g3:shapes:shapes}{radio} $3$ cm (valor exacto y después redondeado al cm$^2$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:5}
Convierte: $3$ m$^2$ en cm$^2$; $45\,000$ cm$^2$ en m$^2$; $2.5$ L
en cm$^3$; $4\,500$ L en m$^3$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:6}
Un campo rectangular mide $120$ m de largo y $85$ m de ancho. ¿Cuántos metros de
valla hacen falta para cercarlo? ¿Cuál es su \omterm{def:g6:measure:area}{área}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:7}
Calcula el \omterm{def:g6:measure:volume}{volumen} de un \omterm{def:g5:solids:def}{ortoedro} de $5$ cm $\times$ $4$ cm $\times$ $10$ cm
y el de un \omterm{def:g5:solids:def}{cubo} de \omterm{def:g5:solids:def}{arista} $3$ cm.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:measure:8}
Dibuja dos rectángulos distintos de \omterm{def:g6:measure:perimeter}{perímetro} $16$ cm y calcula sus
\omterm{def:g6:measure:area}{áreas}. ¿Cuál de tus rectángulos tiene mayor \omterm{def:g6:measure:area}{área}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measure:9}
Una figura en forma de T está hecha de un rectángulo horizontal de $10 \times 2$
encima de otro vertical de $2 \times 6$ (medidas en cm). Calcula su
\omterm{def:g6:measure:area}{área} y después su \omterm{def:g6:measure:perimeter}{perímetro} (recorre el borde con cuidado, sumando
todos los tramos).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measure:10}
Una piscina es un \omterm{def:g5:solids:def}{ortoedro} de $25$ m de largo, $10$ m de ancho y
$2$ m de profundidad.
\begin{enumerate}
  \item ¿cuántos metros cúbicos de agua contiene llena?
  \item ¿cuántos litros son?
  \item la piscina se llena a $50\,000$ L por hora. ¿Cuánto tarda el
        llenado?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:measure:11}
Un jardín es un cuadrado de lado $20$ m que contiene un estanque circular de
\omterm{def:g3:shapes:shapes}{radio} $3$ m. ¿Qué \omterm{def:g6:measure:area}{área} de césped hay (valor exacto y después redondeado
al m$^2$)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:measure:12}
Una tableta de chocolate mide $15$ cm $\times$ $6$ cm $\times$ $1$ cm.
El fabricante duplica \emph{las tres} dimensiones.
\begin{enumerate}
  \item ¿por cuánto queda multiplicado el \omterm{def:g6:measure:volume}{volumen}? Compruébalo calculando los dos
        volúmenes;
  \item el precio se multiplica por $4$. ¿Sale más a cuenta la tableta grande
        que la pequeña?
\end{enumerate}
\end{exercise}

\section{Problema: La valla de la reina Dido}

\begin{problem}[{Problema de fin de semana --- con una longitud fija de valla,
¿qué forma encierra más terreno? El cuadrado gana a todos los
rectángulos, el círculo gana al cuadrado y la pared de un granero lo cambia
todo}]
\label{pb:g6:measure:1}
Cuenta la leyenda que a la reina Dido, al desembarcar en la costa de África, le
concedieron «tanta tierra como pueda encerrar una piel de buey» --- así que cortó
la piel en una tira larguísima y estrechísima y encerró terreno
suficiente para fundar la ciudad de Cartago. Su problema es ahora el tuyo: un
granjero tiene exactamente $20$ m de valla. El \Cref{exo:g6:measure:8}
mostró que dos cercados con el mismo \omterm{def:g6:measure:perimeter}{perímetro} pueden tener \omterm{def:g6:measure:area}{áreas}
distintas; este problema busca el \emph{mejor} cercado --- y descubre
que la respuesta cambia por completo cuando la pared de un granero regala
un lado.

\medskip
\noindent\textbf{Parte I --- Veinte metros de valla.}
\begin{enumerate}
  \item el cercado tiene que ser un rectángulo que use los $20$ m de valla.
        Comprueba que un cercado de $1 \times 9$ y otro de $2 \times 8$
        valen, y calcula sus \omterm{def:g6:measure:area}{áreas};
  \item explica por qué el largo y el ancho de todo cercado válido
        suman $10$ m. Después haz la tabla completa de los
        cercados de lados enteros (de $1 \times 9$ a $5 \times 5$) con
        sus \omterm{def:g6:measure:area}{áreas}. ¿Cuál es el mejor?
  \item ¿merece la pena probar con lados decimales? Calcula las \omterm{def:g6:measure:area}{áreas} del
        cercado de $4.5 \times 5.5$ y del de $4.9 \times 5.1$, y
        compáralas con el cuadrado de $5 \times 5$. ¿Qué
        conjeturas?
  \item la pregunta 2 convirtió el problema de la valla en una pregunta
        puramente numérica: \emph{entre todas las parejas de números que suman
        $10$, ¿cuál tiene el \omterm{def:g2:mult:def}{producto} mayor?} Respóndela a partir de
        tu tabla y enuncia la regla general que observas;
  \item he aquí por qué alejarse del cuadrado siempre pierde,
        con tijeras en vez de con álgebra: parte del
        cercado de $5 \times 5$ y conviértelo en el de $6 \times 4$
        quitando una tira y pegando otra. ¿Qué
        tira se quita, cuál se añade y por qué el
        cambio pierde exactamente un metro cuadrado? Explica por qué un
        paso más (hasta $7 \times 3$) pierde aún más.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El granero y el círculo.}
\begin{enumerate}[resume]
  \item ahora el cercado se construye contra la pared larga de un granero: la pared
        sustituye a uno de los largos del cercado y los $20$ m de valla
        cubren solo los otros tres lados. Haz una tabla de los
        cercados de lados enteros (ancho $a$, largo $L$ pegado a la pared,
        $a + L + a = 20$) y sus \omterm{def:g6:measure:area}{áreas}. ¿Qué cercado gana ahora
        y es un cuadrado?
  \item explica el ganador con un espejo
        (\cref{ch:g6:symmetry}): refleja el cercado respecto de la pared
        del granero y considera el cercado doble. ¿Cuánta valla usa
        el cercado doble, cuál es el mejor cercado doble según la Parte I
        y en qué convierte eso al cercado original?
  \item Dido no construyó un rectángulo. Dobla los $20$ m de valla
        formando un círculo: con $\pi \approx 3.14$, calcula su
        \omterm{def:g3:shapes:shapes}{radio} (al cm) y después su \omterm{def:g6:measure:area}{área} (al m$^2$)
        (\cref{prop:g6:measure:circle},
        \cref{prop:g6:measure:formulas}). Compárala con el
        cuadrado de $5 \times 5$;
  \item el problema inverso: se quiere un cercado de \omterm{def:g6:measure:area}{área} exactamente $36$ m$^2$
        con la menor valla posible. Compara los
        rectángulos $1 \times 36$, $2 \times 18$, $3 \times 12$,
        $4 \times 9$ y $6 \times 6$: ¿\omterm{def:g6:measure:perimeter}{perímetros}? ¿Qué forma sale
        más barata y por qué es esta pregunta la imagen especular de la
        Parte I?
  \item en una frase: ¿por qué son tan a menudo \emph{redondos} las latas, las tuberías
        y los depósitos de agua?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El \omterm{def:g6:measure:perimeter}{perímetro} y el \omterm{def:g6:measure:area}{área} son desconocidos.}
\begin{enumerate}[resume]
  \item halla un rectángulo cuyo \omterm{def:g6:measure:perimeter}{perímetro} supere los $100$ m pero
        cuya \omterm{def:g6:measure:area}{área} sea menor que $1$ m$^2$. (Muy largo y muy
        estrecho. Se permiten decimales.)
  \item ¿verdadero o falso?: «de dos figuras, la que tiene mayor
        \omterm{def:g6:measure:perimeter}{perímetro} tiene mayor \omterm{def:g6:measure:area}{área}». Da un contraejemplo
        sacado de este problema;
  \item toma el rectángulo de $6 \times 4$ y recorta una muesca cuadrada de
        $1 \times 1$ en el medio de uno de los lados largos (la muesca
        se abre hacia fuera, como un diente que falta). Calcula el \omterm{def:g6:measure:area}{área}
        y el \omterm{def:g6:measure:perimeter}{perímetro} de la figura con muesca y compáralos
        con los del original. ¿Qué le ha pasado a cada uno?
  \item los cartógrafos conocen un hecho extraño: cuanto más detallado es el mapa,
        \emph{más larga} sale la costa --- cada ampliación
        revela muescas nuevas, y la pregunta 13 muestra lo que hace
        cada muesca. Explica en una o dos frases por qué
        «la longitud de la costa de Bretaña» es un número
        escurridizo, mientras que «el \omterm{def:g6:measure:area}{área} de Bretaña» no lo es;
  \item el examen del granjero. Con $36$ m de valla y la pared del
        granero disponible, halla el mejor cercado rectangular (usa el
        espejo de la pregunta 7), da su \omterm{def:g6:measure:area}{área} y compárala con
        el mejor cercado construido con $36$ m en campo abierto. ¿Cuánto
        le gana al granjero la pared del granero?
\end{enumerate}
\end{problem}
