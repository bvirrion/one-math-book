\chapter{Fracciones: primeros pasos}\label{ch:g6:fractions}

Repartir tres pizzas a partes iguales entre cuatro personas da a cada una\,\dots\
un número no entero de pizzas. Las fracciones son los números inventados para
repartir. Este capítulo construye la imagen: una \omterm{def:g6:fractions:def}{fracción} como parte de un
todo, como punto de la recta numérica y como resultado exacto de una
división.

\section{Qué significa una fracción}

\begin{definition}[Fracción]\label{def:g6:fractions:def}
Corta una unidad en $b$ partes iguales y toma $a$ de ellas: la cantidad
obtenida es la \emph{fracción}\index{fracción}
\[
\frac{a}{b}
\qquad
\begin{array}{l}
a \text{: el \emph{numerador} --- cuántas partes tomamos;}\\
b \text{: el \emph{denominador} --- en cuántas partes se corta la unidad.}
\end{array}
\]
\end{definition}

\begin{example}\label{ex:g6:fractions:morethanone}
Una \omterm{def:g6:fractions:def}{fracción} puede ser mayor que $1$: $\frac74$ significa siete \omterm{def:g3:division:half}{cuartos} ---
una unidad entera (cuatro \omterm{def:g3:division:half}{cuartos}) y tres \omterm{def:g3:division:half}{cuartos} más:
\[
\frac74 = 1 + \frac34 .
\]
\end{example}

\begin{proposition}[Las fracciones en la recta numérica]\label{prop:g6:fractions:line}
Para situar $\frac ab$ en la recta numérica, corta cada intervalo unidad en $b$
partes iguales y cuenta $a$ partes desde $0$.
\end{proposition}

\begin{theorem}[Una fracción es una división]\label{thm:g6:fractions:quotient}
La \omterm{def:g6:fractions:def}{fracción} $\frac ab$ es el número que, multiplicado por $b$, da
$a$:
\[
\frac ab \times b = a .
\]
Dicho de otro modo, $\frac ab$ es el \omterm{def:g3:division:remainder}{cociente} exacto de $a$ entre $b$ --- incluso
cuando la división «no es exacta».
\end{theorem}
\admitted

\begin{example}\label{ex:g6:fractions:pizzas}
Tres pizzas para cuatro personas: cada una recibe $\frac34$ de pizza.
Comprobación con el teorema: $4$ partes de $\frac34$ hacen
$\frac34 \times 4 = 3$ pizzas. Algunas fracciones son \omterm{def:g6:decimals:places}{números decimales}
($\frac34 = 0.75$) y otras no: $\frac13 = 0.333\dots$ no se acaba nunca.
La \omterm{def:g6:fractions:def}{fracción} es el valor \emph{exacto}; $0.33$ es solo una aproximación.
\end{example}

\section{Fracciones iguales}

\begin{proposition}[Fracciones equivalentes]\label{prop:g6:fractions:equal}
Una \omterm{def:g6:fractions:def}{fracción} no cambia cuando su \omterm{def:g4:fractions:def}{numerador} y su \omterm{def:g4:fractions:def}{denominador} se
multiplican (o se dividen) por un mismo número distinto de \omterm{def:g1:counting:zero}{cero}:
\[
\frac{a}{b} = \frac{a \times k}{b \times k} .
\]
\end{proposition}

\begin{proof}[Idea de la demostración]
Cortar cada parte del reparto en $k$ trozos más pequeños multiplica por $k$ tanto
el número de trozos tomados como el número total de trozos, pero
la cantidad tomada es la misma. Un dibujo lo dice mejor:
\end{proof}

    rectangle (\i*2.4+2.35,1.15);

\begin{method}[Simplificar una fracción]\label{met:g6:fractions:simplify}
Busca un número que divida al \omterm{def:g4:fractions:def}{numerador} \emph{y} al \omterm{def:g4:fractions:def}{denominador},
y divide los dos. Repite hasta que no quede ningún \omterm{def:g5:division:multiple}{divisor} común:
\[
\frac{12}{18} = \frac{12 \div 2}{18 \div 2} = \frac{6}{9}
= \frac{6 \div 3}{9 \div 3} = \frac{2}{3}.
\]
\end{method}

\section{Tomar una fracción de una cantidad}

\begin{method}[Fracción de una cantidad]\label{met:g6:fractions:ofquantity}
Para calcular $\frac ab$ de una cantidad, divide la cantidad entre $b$ (una
parte) y después multiplica por $a$ (el número de partes):
\[
\frac ab \text{ de } Q = (Q \div b) \times a .
\]
(Multiplicar primero por $a$ y dividir después entre $b$ da el mismo
resultado --- elige el orden que deje los números más cómodos.)
\end{method}

\begin{example}\label{ex:g6:fractions:ofquantity}
Calcula $\frac35$ de $40$ euros, paso a paso:
\begin{enumerate}
  \item un quinto de $40$: $40 \div 5 = 8$;
  \item tres quintos: $8 \times 3 = 24$.
\end{enumerate}
Así, $\frac35$ de $40$ euros son $24$ euros. Comprobación: los $\frac25$
restantes son $16$ euros, y $24 + 16 = 40$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g6:fractions:1}
Para cada descripción, escribe la \omterm{def:g6:fractions:def}{fracción}: una tarta cortada en $8$
porciones, $3$ comidas; una tableta de chocolate de $12$ onzas, $7$ comidas; un pastel
cortado en $6$, las $6$ porciones comidas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:2}
Dibuja una barra cortada en $5$ partes iguales y colorea $\frac45$. ¿Es
$\frac45$ menor o mayor que $1$? ¿Y $\frac65$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:3}
Sitúa en una recta numérica graduada en tercios: $\frac13$; \ $\frac53$;
\ $2$; \ $\frac73$. ¿Cuáles dos de estos números están a la misma distancia de
$2$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:4}
Escribe cada \omterm{def:g6:fractions:def}{fracción} como un número entero más una \omterm{def:g6:fractions:def}{fracción} menor que $1$
(como en $\frac74 = 1 + \frac34$):
\[
\frac{9}{4}, \qquad \frac{13}{5}, \qquad \frac{12}{6}, \qquad
\frac{25}{8} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:5}
Completa para que las fracciones sean iguales:
\[
\frac{2}{3} = \frac{?}{12}, \qquad
\frac{15}{20} = \frac{3}{?}, \qquad
\frac{7}{10} = \frac{21}{?}, \qquad
\frac{?}{6} = \frac{10}{12}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:6}
Simplifica todo lo posible: $\dfrac{6}{8}$; \quad $\dfrac{15}{25}$;
\quad $\dfrac{18}{24}$; \quad $\dfrac{35}{7}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:7}
Calcula: $\frac12$ de $86$; \quad $\frac34$ de $60$; \quad $\frac25$ de
$35$; \quad $\frac{7}{10}$ de $250$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:8}
¿Cuál es mayor: $\frac12$ o $\frac13$ (de la misma tarta)? ¿$\frac25$
o $\frac35$? ¿$\frac34$ o $\frac78$ (escribe las dos con \omterm{def:g4:fractions:def}{denominador}
$8$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:fractions:9}
Da el valor decimal de las fracciones que lo tengan y di cuál
\omterm{def:g6:fractions:def}{fracción} no lo tiene: $\frac12$; \ $\frac34$; \ $\frac13$; \ $\frac{7}{10}$;
\ $\frac{9}{4}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:fractions:10}
Una clase tiene $28$ alumnos. Tres \omterm{def:g3:division:half}{cuartos} de ellos vienen al colegio en
autobús y el resto vienen andando. ¿Cuántos alumnos vienen andando? Resuelve paso a paso y
comprueba que los dos grupos suman $28$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:fractions:11}
Leo se gastó $\frac23$ de sus ahorros en un juego que costaba $18$ euros. ¿Cuánto
dinero tenía Leo antes? (Un tercio de sus ahorros es la \omterm{def:g3:division:half}{mitad} de
$18$\,\dots\ piensa por qué, o halla primero el tamaño de una parte.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:fractions:12}
Un depósito está lleno hasta $\frac58$ de su \omterm{def:g2:measure:units}{capacidad}. Después de usar $10$ litros,
está lleno hasta la \omterm{def:g3:division:half}{mitad}. ¿Cuál es la \omterm{def:g2:measure:units}{capacidad} del depósito? (¿Qué \omterm{def:g6:fractions:def}{fracción} del
depósito representan los $10$ litros?)
\end{exercise}

\section{Problema: La tableta de chocolate que nunca se acaba}

\begin{problem}[{Problema de fin de semana --- mitades de mitades: la suma
$\frac12 + \frac14 + \frac18 + \dots$ se acerca a $1$ sin llegar nunca
a tocarlo}]
\label{pb:g6:fractions:1}
Coge una tableta de chocolate. Cómete la \omterm{def:g3:division:half}{mitad}. Después cómete la \omterm{def:g3:division:half}{mitad} de \emph{lo
que queda}. Después otra vez la \omterm{def:g3:division:half}{mitad} de lo que queda, y así sucesivamente. Dos cosas
parecen ciertas a la vez: nunca terminas la tableta (siempre queda
algo) y, sin embargo, te la comes casi entera. Este problema convierte las dos
sensaciones en afirmaciones exactas sobre fracciones --- y se cruza, por
el camino, con un corredor de la antigua Grecia y con un reparto resuelto
enteramente con un cuchillo que solo corta por la \omterm{def:g3:division:half}{mitad}.

\medskip
\noindent\textbf{Parte I --- \omterm{def:g3:division:half}{Mitades} de \omterm{def:g3:division:half}{mitades}.}
La tableta es un rectángulo de $16$ onzas iguales.
\begin{enumerate}
  \item primer bocado: te comes la \omterm{def:g3:division:half}{mitad} de la tableta. ¿Cuántas onzas
        son? Escribe el bocado como \omterm{def:g6:fractions:def}{fracción} de la tableta de dos maneras,
        con \omterm{def:g4:fractions:def}{denominador} $2$ y con \omterm{def:g4:fractions:def}{denominador} $16$
        (\cref{prop:g6:fractions:equal}).
  \item segundo bocado: la \omterm{def:g3:division:half}{mitad} de lo que queda. ¿Cuántas onzas? Muestra,
        con el dibujo o con fracciones equivalentes, que este
        bocado es $\frac14$ de la tableta entera: \emph{la \omterm{def:g3:division:half}{mitad} de una \omterm{def:g3:division:half}{mitad}
        es un \omterm{def:g3:division:half}{cuarto}};
  \item tercer y \omterm{def:g3:division:half}{cuarto} bocados, con la misma regla: da cada uno en
        onzas y como \omterm{def:g6:fractions:def}{fracción} de la tableta. ¿Qué le pasa al
        \omterm{def:g4:fractions:def}{denominador} de un bocado al siguiente?
  \item después de los cuatro bocados, ¿cuántas onzas te has comido en
        total? ¿Qué \omterm{def:g6:fractions:def}{fracción} de la tableta es y qué \omterm{def:g6:fractions:def}{fracción}
        queda?
  \item la pregunta 4 dice, escrito en dieciseisavos:
        \[
        \frac{8}{16} + \frac{4}{16} + \frac{2}{16} + \frac{1}{16}
        = \frac{15}{16},
        \qquad\text{es decir}\qquad
        \frac12 + \frac14 + \frac18 + \frac{1}{16}
        = \frac{15}{16} .
        \]
        Comprueba cada una de las cuatro reescrituras
        ($\frac12 = \frac{8}{16}$, y así con las demás).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La escalera hacia $1$.}
Para seguir mordiendo, imagina una tableta más fina de $64$ onzas.
\begin{enumerate}[resume]
  \item enumera los tamaños de los seis primeros bocados, en onzas
        ($32$, después\,\dots), y comprueba que después del sexto bocado
        queda exactamente una onza. ¿Qué \omterm{def:g6:fractions:def}{fracción} de la tableta
        queda?
  \item después de cada bocado, ¿qué \omterm{def:g6:fractions:def}{fracción} de la tableta queda? Escribe
        la lista ($\frac12$, $\frac14$, \dots) hasta el sexto
        bocado y describe el patrón. Si un cuchillo imaginario
        pudiera partir por la \omterm{def:g3:division:half}{mitad} incluso la última onza, ¿qué quedaría después de
        un séptimo bocado?
  \item explica por qué el total comido \emph{nunca} puede llegar a $1$,
        por muchos bocados que se den. (Compara con los
        veinte nueves del \cref{pb:g6:decimals:1}: ¿es la misma historia?)
  \item y, sin embargo, el total supera cualquier objetivo por debajo de $1$. Un amigo
        te reta a comerte más de $\frac{99}{100}$ de la
        tableta. ¿Cuál es menor, $\frac{1}{128}$ o
        $\frac{1}{100}$, y después de qué bocado queda ganado el reto?
  \item en una recta numérica de $0$ a $1$ graduada en dieciseisavos,
        sitúa los totales comidos después de cada uno de los cuatro primeros
        bocados: $\frac12$, $\frac34$, $\frac78$, $\frac{15}{16}$.
        ¿Dónde cae cada total nuevo respecto del
        anterior y respecto de $1$?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El corredor de Zenón y un cuchillo que
solo parte por la \omterm{def:g3:division:half}{mitad}.}
\begin{enumerate}[resume]
  \item hace veinticuatro siglos, el filósofo griego Zenón
        contó esta historia: para llegar a un muro, un corredor tiene que
        recorrer primero la \omterm{def:g3:division:half}{mitad} de la distancia, luego la \omterm{def:g3:division:half}{mitad} de lo que queda, luego
        otra vez la \omterm{def:g3:division:half}{mitad} de lo que queda\,\dots\ «¡así que el corredor no llega
        nunca al muro!». Usando las preguntas 8 y 9, di qué hay de
        \emph{cierto} en la historia de Zenón y dónde está la trampa.
        (¿De verdad necesita el corredor una eternidad para dar todos esos
        pasos?)
  \item el problema del reparto: reparte $15$ tabletas de chocolate iguales
        entre $16$ niños a partes iguales, usando solo cortes por la \omterm{def:g3:division:half}{mitad}
        (de una tableta, de media tableta, de un \omterm{def:g3:division:half}{cuarto} de tableta\,\dots).
        Describe cómo cortar: cuántas tabletas se cortan por la \omterm{def:g3:division:half}{mitad},
        cuántas en \omterm{def:g3:division:half}{cuartos}, en octavos y en dieciseisavos,
        de modo que cada niño reciba un trozo de cada tamaño;
  \item comprueba tu reparto: escribe la parte de cada niño como \omterm{def:g1:addition:def}{suma} de
        fracciones, calcúlala en dieciseisavos y verifica que las
        $16$ partes juntas gastan exactamente las $15$ tabletas;
  \item en la clase de al lado, $7$ tabletas se reparten a partes iguales entre
        $8$ niños. ¿Qué niño recibe más chocolate: uno
        de los $16$ o uno de los $8$? (Escribe las dos partes con
        \omterm{def:g4:fractions:def}{denominador} $16$, \cref{exo:g6:fractions:8}.)
  \item el final: un amigo afirma que con \emph{infinitos}
        bocados uno se comería \emph{exactamente} la tableta entera.
        Dibuja un cuadrado y sombrea dentro los bocados: la \omterm{def:g3:division:half}{mitad} del
        cuadrado, después un \omterm{def:g3:division:half}{cuarto}, después un octavo\,\dots\ ¿Qué
        observas en la esquina sin sombrear? Enuncia con cuidado
        los dos hechos ciertos de todo este problema: qué queda
        después de cualquier número finito de bocados y cuán pequeño se vuelve ese
        \omterm{def:g3:division:remainder}{resto}. (Qué significa exactamente una «\omterm{def:g1:addition:def}{suma} infinita»
        es una historia para el \omterm{def:g5:solids:volume}{volumen} de secundaria superior.)
\end{enumerate}
\end{problem}
