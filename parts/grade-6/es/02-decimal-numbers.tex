\chapter{Números decimales}\label{ch:g6:decimals}

Entre $2$ y $3$ hay muchos números: $2.5$, $2.71$,
$2.999$\,\dots\ \omterm{def:g6:decimals:places}{Números decimales} extender el sistema de valor posicional al
derecha de las unidades, y siguen las mismas reglas que los números enteros ---
Con algunas trampas, este capítulo le enseñará a evitar.

\section{escritura decimal}

\begin{definition}[Lugares decimales]\label{def:g6:decimals:places}
A \emph{número decimal}\index{número decimal} tiene una parte entera y una
parte decimal, separada por \omterm{def:g4:decimals:point}{coma decimal}. Cada lugar a la derecha
del punto vale diez veces menos que el anterior: décimas,
centésimas, milésimas. Por ejemplo
\[
13.407 = 13 + \frac{4}{10} + \frac{0}{100} + \frac{7}{1000}
= 1 \times 10 + 3 + 4 \times \frac{1}{10} + 7 \times \frac{1}{1000}.
\]
\end{definition}

\begin{remark}[Ceros que importan, ceros que no importan]
Añadiendo \omterm{def:g1:counting:zero}{ceros} al \emph{fin} de la parte decimal no cambia nada:
$2.5 = 2.50 = 2.500$. Pero \omterm{def:g1:counting:zero}{ceros} \emph{entre} Los dígitos son esenciales:
$13.407 \neq 13.47$. Y $0.5$ es muy diferente de $0.05$!
\end{remark}

\begin{method}[Comparar números decimales]\label{met:g6:decimals:compare}
\begin{enumerate}
  \item Primero compare las partes enteras: $7.2 > 5.99$ porque $7 > 5$.
  \item Si las partes enteras son iguales, compare el dígito de las partes decimales por
        dígito desde la izquierda --- relleno con final \omterm{def:g1:counting:zero}{ceros} ayuda:
        comparar $3.4$ y $3.15$, escribir $3.40$ y $3.15$; desde $40 > 15$ centésimas, $3.4 > 3.15$.
\end{enumerate}
Tener cuidado: \emph{Más largo no significa más grande}. $3.15$ tiene más dígitos que
$3.4$ pero es más pequeño.
\end{method}

\section{Sumar, restar, multiplicar}

\begin{method}[Cálculos de columnas con decimales]\label{met:g6:decimals:columns}
Para sumar o restar \omterm{def:g6:decimals:places}{números decimales}, alinear el \emph{\omterm{def:g4:decimals:point}{puntos decimales}}
(de modo que las unidades se ubican debajo de las unidades, décimas debajo de décimas), rellenando con final
\omterm{def:g1:counting:zero}{ceros} si es necesario; luego calcula como con números enteros y coloca el punto
del resultado bajo los demás.
\end{method}

\begin{example}\label{ex:g6:decimals:addsub}
Calcular $13.7 + 2.85$, alineando los puntos ($13.7 = 13.70$):
\[
\begin{array}{r}
1\,3.7\,0 \\
+\ \ \,2.8\,5 \\
\hline
1\,6.5\,5
\end{array}
\]
Calcular $6 - 2.35$ (escribir $6 = 6.00$):
$6.00 - 2.35 = 3.65$. Controlar: $3.65 + 2.35 = 6$.
\end{example}

\begin{example}[multiplicar decimales]\label{ex:g6:decimals:mult}
para calcular $2.3 \times 1.4$: multiplicar como números enteros,
$23 \times 14 = 322$; luego cuenta los dígitos decimales de los factores
($1 + 1 = 2$), y coloque el punto de modo que el resultado tenga esa cantidad:
$2.3 \times 1.4 = 3.22$. Control de cordura en tallas: $2.3 \times 1.4$
debería ser un poco más que $2.3 \times 1 = 2.3$ --- y $3.22$ es.
\end{example}

\section{Multiplicar y dividir por 10, 100, 1000}

\begin{proposition}[Cambiando el punto]\label{prop:g6:decimals:shift}
multiplicando un \omterm{def:g6:decimals:places}{número decimal} por $10$, $100$, $1000$ mueve su \omterm{def:g4:decimals:point}{coma decimal} $1$, $2$, $3$ lugares al \emph{bien}; dividirlo lo mueve a la
\emph{izquierda} (relleno con \omterm{def:g1:counting:zero}{ceros} cuando sea necesario):
\[
3.75 \times 100 = 375, \qquad
42.1 \div 1000 = 0.0421 .
\]
\end{proposition}

\begin{proof}
Multiplicando por $10$ hace que cada dígito valga diez veces más: décimos
se convierten en unidades, las unidades se convierten en decenas, y así sucesivamente: cada dígito se mueve uno
columna a la izquierda en la tabla de valor posicional, que es lo mismo que mover
el punto un lugar a la derecha. Dividir invierte esto.
\end{proof}

\begin{example}[Unidades de medida]\label{ex:g6:decimals:units}
Convertir unidades es exactamente este juego: desde $1$ metro $= 100$ centímetro,
\[
3.75 \text{ m} = 3.75 \times 100 \text{ cm} = 375 \text{ cm},
\qquad
42 \text{ mm} = 42 \div 10 \text{ cm} = 4.2 \text{ cm}.
\]
\end{example}

\section{redondeo}

\begin{definition}[redondeo]\label{def:g6:decimals:rounding}
El \emph{redondeo}\index{redondeo} de un número a la unidad (o décimo,
centésimo, \dots) es el número más cercano con esa precisión. Mira el
siguiente dígito: si es $0,1,2,3,4$, \omterm{def:g4:numbers:round}{redondo} abajo (mantener); si es
$5,6,7,8,9$, \omterm{def:g4:numbers:round}{redondo} arriba.
\end{definition}

\begin{example}
$7.38$ redondeado a la unidad es $7$ (siguiente dígito $3$: mantener); redondeado a
el décimo es $7.4$ (siguiente dígito $8$: \omterm{def:g4:numbers:round}{redondo} arriba). el precio
$4.996$ redondeado a la centésima es $5.00$ --- \omterm{def:g6:decimals:rounding}{redondeo} puede cambiar
cada dígito!
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g6:decimals:1}
Escribe como un \omterm{def:g6:decimals:places}{número decimal}: $5 + \dfrac{3}{10} + \dfrac{7}{100}$;
\quad $\dfrac{9}{10}$; \quad $12 + \dfrac{4}{1000}$;
\quad ``ocho unidades y cinco centésimas''.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:decimals:2}
En $86.354$: ¿cuál es el dígito de las décimas? ¿De las centésimas? que
hace el dígito $8$ ¿contar? Escribe este número como en
\cref{def:g6:decimals:places}.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:decimals:3}
Copiar y completar con $<$, $>$ o $=$:
\[
5.3 \;?\; 5.29, \qquad
0.7 \;?\; 0.70, \qquad
2.09 \;?\; 2.9, \qquad
14.5 \;?\; 14.49 .
\]
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:decimals:4}
Ordene de menor a mayor:
$4.2$; \ $4.05$; \ $4.51$; \ $4.15$; \ $4.5$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:decimals:5}
Cual \omterm{def:g6:decimals:places}{números decimales} corresponden a los puntos $ A $, $ B $, $ C $ en un
recta numérica graduada en décimas, si $ A $ es $3$ garrapatas después $6$, $ B $ es
$7$ garrapatas después $6$, y $ C $ es $2$ garrapatas después $7$?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:decimals:6}
Calcular en columnas: $45.8 + 7.65$; \quad $23.4 - 8.72$; \quad
$5.6 \times 2.4$ (¡cuente los dígitos decimales!).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:decimals:7}
Calcular sin ningún trabajo escrito:
\[
6.42 \times 10, \qquad
0.35 \times 1000, \qquad

5 \div 1000 .
\]
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:decimals:8}
Convertir: $2.4$ m en cm; \quad $370$ g en kg; \quad $0.85$ km en

\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:decimals:9}
Redondo $23.867$: a la unidad; al décimo; a la centésima. Redondo

\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g6:decimals:10}
Una baguette cuesta $1.15$. Lena compra tres baguettes y paga con un
$5$ factura. Escribe los dos cálculos necesarios y dale el cambio.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g6:decimals:11}
Encuentra un \omterm{def:g6:decimals:places}{número decimal} estrictamente entre $7.4$ y $7.5$; luego entre
$3.99$ y $4$. ¿Cuántos números de este tipo hay en cada caso?
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g6:decimals:12}
Usando cada uno de los dígitos $2$, $5$, $8$ exactamente una vez y una \omterm{def:g4:decimals:point}{coma decimal}, escribe el mayor número posible, luego el menor posible
uno. (Números como $.58$ no están permitidos: la parte entera debe contener
al menos un dígito.)
\end{exercise}

\section{Problema: el número justo después del 3 no existe}

\begin{problem}[{Problema de fin de semana --- entre dos números decimales siempre hay otro: hacer zoom en la recta numérica}]
\label{pb:g6:decimals:1}
En la escala de números enteros, cada número tiene un al lado
vecino: justo después $7$ llega $8$ y nada vive en
entre. \omterm{def:g6:decimals:places}{Números decimales} son un mundo completamente diferente. que es
el número justo después $3$? ¿Es $3.1$? $3.01$? $3.001$? esto
El problema desarrolla un \emph{hacer zoom} técnica en la recta numérica
(continuando \cref{exo:g6:decimals:11}) y llega a un famoso
conclusión: el número justo después $3$ \emph{no existe} ---
entre dos cualesquiera \omterm{def:g6:decimals:places}{números decimales}, por muy cerca que esté, siempre hay
espacio para más.

\medskip
\noindent\textbf{Parte I --- Zooming in.}
\begin{enumerate}
  \item que es más grande, $2.999$ o $3$? Calcular el \omterm{ex:g1:subtraction:difference}{diferencia}
        entre ellos.
  \item Dibuja una recta numérica desde $7.4$ a $7.5$, graduado en
        centésimas y lugar $7.42$, $7.45$ y $7.48$ en ello.
        Lista \emph{todo} los números con dos dígitos decimales que
        mentir estrictamente entre $7.4$ y $7.5$. cuantos son
ahí?
  \item Zoom de nuevo: entre $7.44$ y $7.45$, enumera todos los
        números con tres dígitos decimales. cuantos hay
        esta vez?
  \item Usando la misma idea, explica cómo contar los números.
        con exactamente tres dígitos decimales estrictamente entre
        $7.4$ y $7.5$, y dale ese recuento.
  \item Describe la receta que se esconde detrás de las preguntas 2 a 4 (la
        \emph{zoom}): dados dos números que se parecen
        vecinos, como $5.67$ y $5.68$, ¿cómo se escribe?
        un decimal más siempre revela un número estrictamente
        entre ellos? Aplica tu receta a $5.67$ y $5.68$,
        luego a $0.1999$ y $0.2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- The missing neighbour.}
\begin{enumerate}[resume]
  \item Tom afirma: "el número justo después $3$ es $3.1$." Demostrar
        se equivocó al nombrar un número estrictamente entre $3$ y
        $3.1$. Tom se retira a $3.01$,
        luego a $3.001$. vencer a cada uno
        de sus candidatos.
  \item Explica por qué \emph{nadie} puede ganar este juego contra ti:
        cualquier número estrictamente mayor que $3$ Tom propone,
        la receta de zoom de la pregunta 5 produce un número estrictamente
        entre $3$ y su propuesta. ¿Qué prueba esto acerca de
        ``el número justo después $3$''?
  \item Ahora el otro lado: Tom busca el número que acaba de
        \emph{antes} $3$ y lo intenta $2.9$, entonces $2.99$, entonces
        $2.999$. Venció a sus tres candidatos. Luego calcula
        $3 - 2.999$ y describir (sin calcular en columnas)
        el \omterm{ex:g1:subtraction:difference}{diferencia} entre $3$ y el número escrito con una
        $2$, un punto y veinte dígitos $9$.
  \item ¿Por qué nada de esto funciona para \emph{entero} ¿números?
        Explica en una o dos frases por qué no hay un todo.
        número estrictamente entre $7$ y $8$, aunque hay
        son muchos \omterm{def:g6:decimals:places}{números decimales} allá.
  \item Una longitud se anuncia como ``$3.7$ cm, redondeado a la
        décimo'' (\cref{def:g6:decimals:rounding}). Dale el
        longitud más pequeña que \omterm{def:g4:numbers:round}{rondas} a $3.7$ cm y explica por qué
        una longitud de $3.75$ cm hace \emph{no} \omterm{def:g4:numbers:round}{redondo} a $3.7$ ---
        entonces la longitud real es al menos $3.65$ cm y estrictamente
        abajo $3.75$ centímetro.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Games with digits and points.}
\begin{enumerate}[resume]
  \item Ordene de menor a mayor: $0.6$; \ $0.58$; \
        $0.123$; \ $0.0999$. Luego explícale a un compañero, en una
        frase, por qué $0.0999 < 0.6$ aunque esté escrito
        con más dígitos (\cref{met:g6:decimals:compare}).
  \item Los precios en una tienda siempre tienen exactamente dos dígitos decimales.
        Hay una \emph{precio} estrictamente entre $4.99$ y $5$?
        Compárese con la pregunta 9: ¿qué significan los precios y los números enteros?
        have in common?
  \item Usando cada uno de los dígitos $2$, $5$, $8$ exactamente una vez y una \omterm{def:g4:decimals:point}{coma decimal} (la parte entera no está vacía, como en
        \cref{exo:g6:decimals:12}), encuentre el número más cercano a
        $6$. Justifica calculando la distancia de tu mejor
        candidatos a $6$.
  \item El número más famoso de las matemáticas, $\pi $, satisface
        $3.141 < \pi < 3.142$. Nombra un \omterm{def:g6:decimals:places}{número decimal} estrictamente entre $3.141$ y $3.142$; entonces uno
estrictamente entre
        $3.1415$ y $3.1416$. (Los matemáticos aprietan $\pi $
        exactamente de esta manera --- cada nuevo dígito decimal es uno más
        zoom. te encontraras $\pi $ en el trabajo en
        \cref{ch:g6:measure}.)
  \item El final: explica por qué hay \emph{más \omterm{def:g6:decimals:places}{números decimales} entre $0$ y $1$ que cualquier número que puedas
        nombre}. (¿Cuántos produce un zoom? ¿Puede el zoom
        parar alguna vez?)
\end{enumerate}
\end{problem}
