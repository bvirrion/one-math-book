\chapter{Los números decimales}\label{ch:g6:decimals}

Entre $2$ y $3$ hay muchísimos números: $2.5$, $2.71$,
$2.999$\,\dots\ Los \omterm{def:g6:decimals:places}{números decimales} extienden el sistema posicional a la
derecha de las unidades y siguen las mismas reglas que los enteros ---
con unas cuantas trampas que este capítulo te enseñará a esquivar.

\section{La escritura decimal}

\begin{definition}[Cifras decimales]\label{def:g6:decimals:places}
Un \emph{número decimal}\index{número decimal} tiene una parte entera y una
parte decimal, separadas por el \omterm{def:g4:decimals:point}{punto decimal}. Cada lugar a la derecha
del punto vale diez veces menos que el anterior: décimas,
centésimas, milésimas. Por ejemplo,
\[
13.407 = 13 + \frac{4}{10} + \frac{0}{100} + \frac{7}{1000}
= 1 \times 10 + 3 + 4 \times \frac{1}{10} + 7 \times \frac{1}{1000}.
\]
\end{definition}

\begin{remark}[Ceros que importan y ceros que no]
Añadir \omterm{def:g1:counting:zero}{ceros} al \emph{final} de la parte decimal no cambia nada:
$2.5 = 2.50 = 2.500$. Pero los \omterm{def:g1:counting:zero}{ceros} \emph{intercalados} son imprescindibles:
$13.407 \neq 13.47$. ¡Y $0.5$ es muy distinto de $0.05$!
\end{remark}

\begin{method}[Comparar números decimales]\label{met:g6:decimals:compare}
\begin{enumerate}
  \item compara primero las partes enteras: $7.2 > 5.99$ porque $7 > 5$;
  \item si las partes enteras son iguales, compara las partes decimales cifra a
        cifra desde la izquierda --- completar con \omterm{def:g1:counting:zero}{ceros} finales ayuda:
        para comparar $3.4$ y $3.15$, escribe $3.40$ y $3.15$;
        como $40 > 15$ centésimas, $3.4 > 3.15$.
\end{enumerate}
Cuidado: \emph{más largo no significa mayor}. $3.15$ tiene más cifras que
$3.4$ pero es menor.
\end{method}

\section{Sumar, restar y multiplicar}

\begin{method}[Cálculos en columna con decimales]\label{met:g6:decimals:columns}
Para sumar o restar \omterm{def:g6:decimals:places}{números decimales}, alinea los \emph{\omterm{def:g4:decimals:point}{puntos decimales}}
(así las unidades quedan bajo las unidades y las décimas bajo las décimas), completando con
\omterm{def:g1:counting:zero}{ceros} finales si hace falta; después calcula como con los enteros y coloca el punto
del resultado bajo los demás.
\end{method}

\begin{example}\label{ex:g6:decimals:addsub}
Calcula $13.7 + 2.85$ alineando los puntos ($13.7 = 13.70$):
\[
\begin{array}{r}
1\,3.7\,0 \\
+\ \ \,2.8\,5 \\
\hline
1\,6.5\,5
\end{array}
\]
Calcula $6 - 2.35$ (escribe $6 = 6.00$):
$6.00 - 2.35 = 3.65$. Comprobación: $3.65 + 2.35 = 6$.
\end{example}

\begin{example}[Multiplicar decimales]\label{ex:g6:decimals:mult}
Para calcular $2.3 \times 1.4$: multiplica como si fueran enteros,
$23 \times 14 = 322$; después cuenta las cifras decimales de los factores
($1 + 1 = 2$) y coloca el punto para que el resultado tenga esas mismas:
$2.3 \times 1.4 = 3.22$. Control de tamaños: $2.3 \times 1.4$
debería ser algo más que $2.3 \times 1 = 2.3$ --- y $3.22$ lo es.
\end{example}

\section{Multiplicar y dividir por 10, 100 y 1000}

\begin{proposition}[Desplazar el punto]\label{prop:g6:decimals:shift}
Multiplicar un \omterm{def:g6:decimals:places}{número decimal} por $10$, $100$ o $1000$ mueve su \omterm{def:g4:decimals:point}{punto
decimal} $1$, $2$ o $3$ lugares a la \emph{derecha}; dividir lo mueve a la
\emph{izquierda} (completando con \omterm{def:g1:counting:zero}{ceros} cuando haga falta):
\[
3.75 \times 100 = 375, \qquad
42.1 \div 1000 = 0.0421 .
\]
\end{proposition}

\begin{proof}
Multiplicar por $10$ hace que cada cifra valga diez veces más: las décimas
pasan a unidades, las unidades a decenas, y así sucesivamente --- cada cifra se mueve una
columna a la izquierda en la tabla posicional, que es lo mismo que mover
el punto un lugar a la derecha. Dividir lo invierte.
\end{proof}

\begin{example}[Unidades de medida]\label{ex:g6:decimals:units}
Convertir unidades es exactamente este juego: como $1$ m $= 100$ cm,
\[
3.75 \text{ m} = 3.75 \times 100 \text{ cm} = 375 \text{ cm},
\qquad
42 \text{ mm} = 42 \div 10 \text{ cm} = 4.2 \text{ cm}.
\]
\end{example}

\section{Redondeo}

\begin{definition}[Redondeo]\label{def:g6:decimals:rounding}
El \emph{redondeo}\index{redondeo} de un número a la unidad (o a la décima,
a la centésima, \dots) es el número más próximo con esa precisión. Mira la
cifra siguiente: si es $0,1,2,3,4$, redondea hacia abajo (deja igual); si es
$5,6,7,8,9$, redondea hacia arriba.
\end{definition}

\begin{example}
$7.38$ redondeado a la unidad es $7$ (cifra siguiente $3$: se deja); redondeado a
la décima es $7.4$ (cifra siguiente $8$: sube). El precio
$4.996$ redondeado a la centésima es $5.00$ --- ¡el \omterm{def:g6:decimals:rounding}{redondeo} puede cambiar
todas las cifras!
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g6:decimals:1}
Escribe como \omterm{def:g6:decimals:places}{número decimal}: $5 + \dfrac{3}{10} + \dfrac{7}{100}$;
\quad $\dfrac{9}{10}$; \quad $12 + \dfrac{4}{1000}$;
\quad «ocho unidades y cinco centésimas».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:2}
En $86.354$: ¿cuál es la cifra de las décimas? ¿Y la de las centésimas? ¿Qué
cuenta la cifra $8$? Escribe este número como en la
\cref{def:g6:decimals:places}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:3}
Copia y completa con $<$, $>$ o $=$:
\[
5.3 \;?\; 5.29, \qquad
0.7 \;?\; 0.70, \qquad
2.09 \;?\; 2.9, \qquad
14.5 \;?\; 14.49 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:4}
Ordena de menor a mayor:
$4.2$; \ $4.05$; \ $4.51$; \ $4.15$; \ $4.5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:5}
¿Qué \omterm{def:g6:decimals:places}{números decimales} corresponden a los puntos $A$, $B$ y $C$ de una
recta numérica graduada en décimas, si $A$ está $3$ marcas después del $6$, $B$ está
$7$ marcas después del $6$ y $C$ está $2$ marcas después del $7$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:6}
Calcula en columna: $45.8 + 7.65$; \quad $23.4 - 8.72$; \quad
$5.6 \times 2.4$ (¡cuenta las cifras decimales!).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:7}
Calcula sin escribir nada:
\[
6.42 \times 10, \qquad
0.35 \times 1000, \qquad
78.1 \div 100, \qquad
5 \div 1000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:8}
Convierte: $2.4$ m en cm; \quad $370$ g en kg; \quad $0.85$ km en
m; \quad $56$ mm en m.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:9}
Redondea $23.867$: a la unidad; a la décima; a la centésima. Redondea
$9.97$ a la décima.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:decimals:10}
Una barra de pan cuesta $1.15$. Lena compra tres barras y paga con un billete de
$5$. Escribe los dos cálculos necesarios y da su cambio.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:decimals:11}
Halla un \omterm{def:g6:decimals:places}{número decimal} estrictamente comprendido entre $7.4$ y $7.5$; después entre
$3.99$ y $4$. ¿Cuántos números así hay en cada caso?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:decimals:12}
Usando cada una de las cifras $2$, $5$ y $8$ exactamente una vez y un \omterm{def:g4:decimals:point}{punto
decimal}, escribe el número más grande posible y después el más pequeño
posible. (No se admiten números como $.58$: la parte entera debe tener
al menos una cifra.)
\end{exercise}

\section{Problema: El número justo después del 3 no existe}

\begin{problem}[{Problema de fin de semana --- entre dos números decimales
cualesquiera siempre hay otro: ampliando la recta numérica}]
\label{pb:g6:decimals:1}
En la escalera de los números enteros, cada número tiene un vecino de al
lado: justo después del $7$ viene el $8$ y no vive nada en
medio. Los \omterm{def:g6:decimals:places}{números decimales} son un mundo completamente distinto. ¿Cuál es
el número justo después del $3$? ¿Es $3.1$? ¿$3.01$? ¿$3.001$? Este
problema desarrolla una técnica de \emph{ampliación} de la recta numérica
(que continúa el \cref{exo:g6:decimals:11}) y llega a una conclusión
famosa: el número justo después del $3$ \emph{no existe} ---
entre dos \omterm{def:g6:decimals:places}{números decimales} cualesquiera, por próximos que estén, siempre hay
sitio para más.

\medskip
\noindent\textbf{Parte I --- Ampliar la escala.}
\begin{enumerate}
  \item ¿cuál es mayor, $2.999$ o $3$? Calcula la \omterm{ex:g1:subtraction:difference}{diferencia}
        entre ellos;
  \item dibuja una recta numérica de $7.4$ a $7.5$, graduada en
        centésimas, y sitúa en ella $7.42$, $7.45$ y $7.48$.
        Enumera \emph{todos} los números de dos cifras decimales que
        están estrictamente entre $7.4$ y $7.5$. ¿Cuántos son?
  \item amplía otra vez: entre $7.44$ y $7.45$, enumera todos los
        números de tres cifras decimales. ¿Cuántos son
        esta vez?
  \item con la misma idea, explica cómo contar los números
        de exactamente tres cifras decimales que están estrictamente entre
        $7.4$ y $7.5$, y da ese recuento;
  \item describe la receta que se esconde detrás de las preguntas 2--4 (la
        \emph{ampliación}): dados dos números que parecen
        vecinos, como $5.67$ y $5.68$, ¿por qué escribir
        una cifra decimal más revela siempre un número estrictamente
        entre ellos? Aplica tu receta a $5.67$ y $5.68$,
        y después a $0.1999$ y $0.2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El vecino que falta.}
\begin{enumerate}[resume]
  \item Tomás afirma: «el número justo después del $3$ es $3.1$». Demuestra
        que se equivoca nombrando un número estrictamente entre $3$ y
        $3.1$. Tomás se repliega a $3.01$ y luego a $3.001$. Vence a cada
        uno de sus candidatos;
  \item explica por qué \emph{nadie} puede ganarte a este juego:
        sea cual sea el número estrictamente mayor que $3$ que proponga Tomás,
        la receta de ampliación de la pregunta 5 produce un número estrictamente
        entre $3$ y su propuesta. ¿Qué demuestra esto sobre
        «el número justo después del $3$»?
  \item ahora el otro lado: Tomás busca el número justo
        \emph{antes} del $3$ y prueba $2.9$, luego $2.99$ y luego
        $2.999$. Vence a sus tres candidatos. Después calcula
        $3 - 2.999$ y describe (sin calcular en columna)
        la \omterm{ex:g1:subtraction:difference}{diferencia} entre $3$ y el número escrito con un
        $2$, un punto y veinte cifras $9$;
  \item ¿por qué nada de esto funciona con los números \emph{enteros}?
        Explica en una o dos frases por qué no hay ningún número
        entero estrictamente entre $7$ y $8$, aunque
        haya muchísimos \omterm{def:g6:decimals:places}{números decimales} ahí;
  \item una longitud se anuncia como «$3.7$ cm, redondeada a la
        décima» (\cref{def:g6:decimals:rounding}). Da la
        longitud más pequeña que redondea a $3.7$ cm y explica por qué
        una longitud de $3.75$ cm \emph{no} redondea a $3.7$ ---
        de modo que la longitud verdadera es al menos $3.65$ cm y estrictamente
        menor que $3.75$ cm.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Juegos con cifras y puntos.}
\begin{enumerate}[resume]
  \item ordena de menor a mayor: $0.6$; \ $0.58$; \
        $0.123$; \ $0.0999$. Después explica a un compañero, en una
        frase, por qué $0.0999 < 0.6$ aunque esté escrito
        con más cifras (\cref{met:g6:decimals:compare});
  \item los precios de una tienda tienen siempre exactamente dos cifras decimales.
        ¿Hay algún \emph{precio} estrictamente entre $4.99$ y $5$?
        Compara con la pregunta 9: ¿qué tienen en común los precios y los números
        enteros?
  \item usando cada una de las cifras $2$, $5$ y $8$ exactamente una vez y
        un \omterm{def:g4:decimals:point}{punto decimal} (con la parte entera no vacía, como en el
        \cref{exo:g6:decimals:12}), halla el número más cercano a
        $6$. Justifícalo calculando la distancia de tus mejores
        candidatos a $6$;
  \item el número más famoso de las matemáticas, $\pi$, cumple
        $3.141 < \pi < 3.142$. Nombra un \omterm{def:g6:decimals:places}{número decimal} estrictamente
        comprendido entre $3.141$ y $3.142$; después uno estrictamente entre
        $3.1415$ y $3.1416$. (Los matemáticos acorralan $\pi$
        exactamente así --- cada cifra decimal nueva es una ampliación
        más. Verás $\pi$ en acción en el
        \cref{ch:g6:measure}.)
  \item el final: explica por qué hay \emph{más \omterm{def:g6:decimals:places}{números
        decimales} entre $0$ y $1$ que cualquier número que puedas
        nombrar}. (¿Cuántos produce una ampliación? ¿Puede la ampliación
        detenerse alguna vez?)
\end{enumerate}
\end{problem}
