\chapter{Triángulos y cuadriláteros}\label{ch:g6:shapes}

Triángulos y \omterm{def:g4:geometry:polygon}{cuadriláteros} son las formas básicas de la geometría. el objetivo
de este capítulo es conocer a sus familias especiales --- \omterm{def:g6:shapes:triangles}{isósceles},
\omterm{def:g6:shapes:triangles}{equilátero}, \omterm{def:g6:shapes:triangles}{triangulos rectángulos}; rectángulos, rombos, cuadrados --- por
su \emph{definir propiedades}, y construirlos precisamente con
regla y compás.

\section{Triangulos}

\begin{definition}[Triángulos especiales]\label{def:g6:shapes:triangles}
un triángulo $ABC $ tiene tres \omterm{def:g5:solids:def}{vértices} y tres lados. Es:
\begin{itemize}
  \item \emph{isósceles}\index{isósceles} en $ A $ cuando
        $ AB = AC $ (dos lados iguales);
  \item \emph{equilátero}\index{equilátero} cuando
        $ AB = a.\,C. = CA $ (tres lados iguales);
  \item \emph{en ángulo recto}\index{en ángulo recto} en $ A $ cuando el ángulo
        $\widehat{BAC}$ es un \omterm{def:g3:shapes:rightangle}{ángulo recto}. el lado $[a.\,C.]$ frente al
        \omterm{def:g3:shapes:rightangle}{ángulo recto} se llama el \emph{hipotenusa}.
\end{itemize}
\end{definition}

\begin{method}[Construir un triángulo a partir de sus tres lados.]\label{met:g6:shapes:construct}
para construir $ ABC $ con $ AB = 5$ centímetro, $ AC = 4$ centímetro, $ a.\,C. = 3$ centímetro:
\begin{enumerate}
  \item dibujar el \omterm{def:g6:lines:objects}{segmento} $[AB]$ de $5$ cm con la regla;
  \item trazar un arco del círculo de centro $ A $ y \omterm{def:g3:shapes:shapes}{radio} $4$ centímetros
        (todos los puntos en $4$ centímetros de $ A $);
  \item trazar un arco del círculo de centro $ B $ y \omterm{def:g3:shapes:shapes}{radio} $3$ centímetro;
  \item el punto $ C $ es donde se cruzan los dos arcos; únete a $ A $ y
        $ B $.
\end{enumerate}
\end{method}

\section{Cuadriláteros}

\begin{definition}[Cuadriláteros especiales]\label{def:g6:shapes:quadrilaterals}
A \omterm{def:g4:geometry:polygon}{cuadrilátero} tiene cuatro \omterm{def:g5:solids:def}{vértices} y cuatro lados. Es un:
\begin{itemize}
  \item \emph{rectángulo}\index{rectángulo} cuando sus cuatro ángulos son
        \omterm{def:g3:shapes:rightangle}{correcto
        ángulos};
  \item \emph{rombo}\index{rombo} cuando sus cuatro lados son iguales
        longitud;
  \item \emph{cuadrados}\index{cuadrados} cuando son ambos: cuatro \omterm{def:g3:shapes:rightangle}{correcto
        ángulos}
        \emph{y} cuatro lados iguales.
\end{itemize}
\end{definition}

\begin{proposition}[Propiedades de las diagonales]\label{prop:g6:shapes:diagonals}
En estos \omterm{def:g4:geometry:polygon}{cuadriláteros}, las dos diagonales se cortan entre sí en sus
punto medio común, y además:
\begin{itemize}
  \item En un rectángulo las diagonales tienen \emph{misma longitud};
  \item en un \omterm{def:g6:shapes:quadrilaterals}{rombo}, las diagonales son \emph{\omterm{def:g6:lines:perp}{perpendicular}};
  \item en un cuadrado, ambos a la vez.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}
Un cuadrado es automáticamente un rectángulo (tiene cuatro \omterm{def:g3:shapes:rightangle}{correcto
        ángulos}) y
automáticamente un \omterm{def:g6:shapes:quadrilaterals}{rombo} (cuatro lados iguales). Entonces "rectángulo" no
significa "no un cuadrado"! En matemáticas, la forma más especial pertenece
a la familia más general.
\end{remark}

\begin{example}[Reconocer una forma a partir de sus propiedades.]\label{ex:g6:shapes:recognize}
A \omterm{def:g4:geometry:polygon}{cuadrilátero} tiene cuatro lados de longitud $4$ cm y uno \omterm{def:g3:shapes:rightangle}{ángulo recto}.
¿Qué es? Cuatro lados iguales lo convierten en un \omterm{def:g6:shapes:quadrilaterals}{rombo}. en un \omterm{def:g6:shapes:quadrilaterals}{rombo}, opuesto
Los ángulos son iguales y los cuatro ángulos suman. $360^\circ $; uno \omterm{def:g3:shapes:rightangle}{ángulo recto} luego obliga a los cuatro a tener razón (esto se demostrará con
paralelogramos en \cref{ch:g7:central}). Entonces la forma es un cuadrado.
\end{example}

\begin{method}[Programas de construcción]\label{met:g6:shapes:program}
A \emph{programa de construcción} es una lista de instrucciones que cualquiera puede
siga para reproducir una figura exactamente. Escriba una instrucción por línea,
nombrar cada punto cuando se crea, y sólo utilizar: ``dibujar el \omterm{def:g6:lines:objects}{segmento}
/ línea / círculo\dots'', ``coloque el punto donde \dots\ cruz''. prueba
su programa siguiéndolo literalmente, palabra por palabra.
\end{method}

\begin{example}\label{ex:g6:shapes:program}
Programa para una plaza $ ABCD $ de lado $4$ centímetro:
\begin{enumerate}
  \item dibuja un \omterm{def:g6:lines:objects}{segmento} $[AB]$ con $ AB = 4$ centímetro.
  \item Dibuja el \omterm{def:g6:lines:perp}{perpendicular} a $(AB)$ en $ A $; en él, coloque $ D $ en
        $4$ centímetros de $ A $ (elige un lado de la línea y cíñete a él).
  \item Dibuja el \omterm{def:g6:lines:perp}{perpendicular} a $(AB)$ en $ B $; en él, coloque $ C $ en
        $4$ centímetros de $ B $, del mismo lado que $ D $.
  \item Unirse $ C $ y $ D $.
\end{enumerate}
Consulta con la regla: $ DC $ debe medir $4$ cm también.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g6:shapes:1}
construir un triángulo $ ABC $ con $ AB = 6$ centímetro, $ AC = 5$ centímetros y
$ a.\,C. = 4$ centímetros,
siguiente \cref{met:g6:shapes:construct}.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:shapes:2}
construir un \omterm{def:g6:shapes:triangles}{triángulo isósceles} $ DEF $ con $ DE = DF = 5$ centímetros y
$ EF = 3$ centímetro. ¿Cuáles dos ángulos de este triángulo parecen iguales? Mídelos
para comprobarlo.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:shapes:3}
construir un \omterm{def:g6:shapes:triangles}{triángulo equilatero} de lado $4$ cm usando solo regla y
brújula. Mide uno de sus ángulos.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:shapes:4}
Clasifica cada triángulo a partir de sus datos (\omterm{def:g6:shapes:triangles}{isósceles}, \omterm{def:g6:shapes:triangles}{equilátero},
\omterm{def:g6:shapes:triangles}{en ángulo recto} --- pueden aplicarse varias respuestas):
(a) lados $7$, $7$, $7$; \
(b) lados $5$, $5$, $8$; \
(c) ángulos $90^\circ $, $45^\circ $, $45^\circ $; \
(d) ángulos $60^\circ $, $60^\circ $, $60^\circ $.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:shapes:5}
en un \omterm{def:g6:shapes:triangles}{triángulo rectángulo}, ¿de qué lado está la hipotenusa? dibuja un triángulo
$ MNP $ \omterm{def:g6:shapes:triangles}{en ángulo recto} en $ N $y nombra su hipotenusa. Cual \omterm{def:g5:solids:def}{vértice} hace
¿La hipotenusa nunca se toca?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:shapes:6}
¿Verdadero o falso? Justifica cada respuesta en una frase.
\begin{enumerate}
  \item Cada cuadrado es un rectángulo.
  \item Todo rectángulo es un cuadrado.
  \item Cada cuadrado es un \omterm{def:g6:shapes:quadrilaterals}{rombo}.
  \item A \omterm{def:g6:shapes:quadrilaterals}{rombo} con un \omterm{def:g3:shapes:rightangle}{ángulo recto} es un cuadrado.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:shapes:7}
construir un rectángulo $ ABCD $ con $ AB = 5$ centímetros y $ a.\,C. = 3$ centímetro. Medida
sus dos diagonales: ¿qué observas?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g6:shapes:8}
Construir un \omterm{def:g6:shapes:quadrilaterals}{rombo} con diagonales de $6$ centímetros y $4$ centímetro. (Usar
\cref{prop:g6:shapes:diagonals}: dibuja las dos diagonales primero ---
\omterm{def:g6:lines:perp}{perpendicular}, cruzando en sus puntos medios --- luego unir los cuatro
termina.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g6:shapes:9}
Escribir un programa de construcción para un \omterm{def:g6:shapes:triangles}{triángulo isósceles} con base
$4$ cm y lados iguales de $6$ cm, al estilo de
\cref{ex:g6:shapes:program}. Intercambio con un compañero: ¿pueden seguir?
¿Sin ver tu figura?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g6:shapes:10}
Dibuja un círculo de centro. $ O $ y uno de sus \omterm{def:g4:geometry:circle}{diámetros} $[AB]$. Coloque cualquier
punto $ M $ en el círculo y dibuja el triángulo $ AMB $. medir el angulo
$\widehat{AMB}$. Mover $ M $ y medir nuevamente. ¿Qué conjeturas?
(Este hecho sorprendente se demuestra en \cref{ch:g8:cosine}.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g6:shapes:11}
A \omterm{def:g4:geometry:polygon}{cuadrilátero} tiene diagonales de la misma longitud que se cortan entre sí en
su punto medio común. Determina qué forma debe tener, usando
\cref{prop:g6:shapes:diagonals}: rectángulo, \omterm{def:g6:shapes:quadrilaterals}{rombo} o cuadrado?
(Cuidado: sólo se proporciona una propiedad).
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g6:shapes:12}
en un $4 \times 4$ cuadrícula de puntos, ¿cuántos cuadrados diferentes puedes dibujar?
¿Cuyas esquinas son puntos de la cuadrícula, contando también los cuadrados "inclinados"?
Dibuja al menos un cuadrado inclinado y explica por qué sus lados son iguales.
\end{exercise}

\section{Problema: lo que saben las diagonales}

\begin{problem}[{Problema de fin de semana --- reconstruir cada cuadrilátero especial solo a partir de sus diagonales y contar las diagonales de cualquier polígono}]
\label{pb:g6:shapes:1}
\Cref{prop:g6:shapes:diagonals} lee una forma y te informa sobre
sus diagonales. Este problema juega el juego. \emph{hacia atrás}: dado
solo dos cruces \omterm{def:g6:lines:objects}{segmentos} --- las futuras diagonales --- qué forma
aparece cuando unes sus cuatro extremos? Construirás un completo
``diccionario diagonal'', entender \emph{por qué} el \omterm{def:g6:shapes:quadrilaterals}{rombo}
construcción de \cref{exo:g6:shapes:8} funciona, encuentra una forma que
aún no tiene nombre, y termina contando las diagonales de \omterm{def:g4:geometry:polygon}{polígonos}
con muchos más de cuatro lados.

\medskip
\noindent\textbf{Parte I --- The diagonal dictionary.}
Para cada pregunta, dibuja los dos \omterm{def:g6:lines:objects}{segmentos} como se describe, únete a sus
cuatro extremos en orden e identifique los \omterm{def:g4:geometry:polygon}{cuadrilátero} lo obtienes.
\begin{enumerate}
  \item Dos \omterm{def:g6:lines:objects}{segmentos} de $6$ cm cada uno, \omterm{def:g6:lines:perp}{perpendicular}, cruzando en
        su punto medio común. Mida los cuatro lados y el
        cuatro ángulos de la forma resultante: ¿qué es?
  \item Dos \omterm{def:g6:lines:objects}{segmentos} de $8$ centímetros y $5$ centímetro, \omterm{def:g6:lines:perp}{perpendicular},
        cruce en su punto medio común. ¿Qué forma aparece?
        (\cref{exo:g6:shapes:8})?
  \item Dos \omterm{def:g6:lines:objects}{segmentos} de $7$ cm cada uno --- igual pero \emph{no}
        \omterm{def:g6:lines:perp}{perpendicular} --- cruce en su punto medio común.
        Mide los lados y los ángulos: ¿cuál conjeturas
        forma de ser?
  \item Dos \omterm{def:g6:lines:objects}{segmentos} de $8$ centímetros y $5$ centímetros, no \omterm{def:g6:lines:perp}{perpendicular},
        cruce en su punto medio común. La forma que obtienes es
        ninguno de los tres
        \cref{def:g6:shapes:quadrilaterals} --- describe lo que
        ves (¿qué lados se ven iguales? \omterm{def:g6:lines:perp}{paralelo}?). Este
        \omterm{def:g4:geometry:polygon}{cuadrilátero} recibe su nombre en \cref{ch:g7:central}.
  \item Resume las preguntas 1 a 4 en un diccionario con cuatro
        entradas: diagonales \emph{igual / \omterm{def:g6:lines:perp}{perpendicular} / ambos /
        tampoco} (siempre cruzando en sus puntos medios)
        $\rightarrow $ nombre (o descripción) de la forma.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Why the constructions work.}
\begin{enumerate}[resume]
  \item En la pregunta 2, las dos diagonales cortan la forma en cuatro
        \omterm{def:g6:shapes:triangles}{triangulos rectángulos} alrededor del punto de cruce. ¿Cuáles son los
        ¿La longitud de dos catetos de cada uno de estos triángulos? Explica por qué
        los cuatro triángulos son copias exactas uno del otro, y
        deduce por qué los cuatro lados de la figura son iguales --- el
        secreto de \cref{exo:g6:shapes:8}.
  \item Deducir una prueba de sólo brújula: después de construir cualquier forma
        desde sus diagonales, ¿cómo se puede comprobar con la brújula?
        solo (sin medidas de regla) que los cuatro lados son
        igual?
  \item Escribir un programa de construcción.
        (\cref{met:g6:shapes:program}) para un cuadrado cuyo
        \emph{diagonales} medida $6$ cm --- no sus lados.
  \item En el cuadrado de la pregunta 8, ¿el lado es más largo o
        más corto que $3$ centímetros--- \omterm{def:g3:division:half}{medio} la diagonal? Medir hasta
        descúbrelo. Uno \omterm{def:g3:division:half}{medio} de la respuesta ya puede ser
        \emph{probado}: ir de una esquina a la siguiente a través del
        centro es un $3 + 3$ cm de desvío, y un camino recto es
        siempre más corto que un desvío --- entonces el lado es más corto
        que $6$ centímetro. (Demostrando que también le gana $3$ centímetros, y
        calcularlo exactamente, es un trabajo para
        \cref{ch:g8:pythagoras}.)
  \item Una nueva entrada para el diccionario: dibujar un \omterm{def:g6:lines:objects}{segmento} $[AC]$ de
        $8$ centímetro; en él, marque el punto $ H $ en $3$ centímetros de $ A $;
        dibujar a través $ H $ el \omterm{def:g6:lines:perp}{perpendicular} \omterm{def:g6:lines:objects}{segmento} $[BD]$ de
        $5$ centímetros con $ H $ como \emph{es} punto medio. Unirse $ ABCD $. esto
        la forma es una \emph{cometa}\index{cometa} (una diagonal biseca
        el otro, en un \omterm{def:g3:shapes:rightangle}{ángulo recto}). Usando triángulos de esquina como
        en la pregunta 6, encuentra qué lados de un \omterm{pb:g6:shapes:1}{cometa} son iguales a
        cual.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Counting diagonals.}
A \emph{diagonal} de un \omterm{def:g4:geometry:polygon}{polígono} une dos \omterm{def:g5:solids:def}{vértices} que no son
vecinos.
\begin{enumerate}[resume]
  \item dibuja un \omterm{def:g4:geometry:polygon}{cuadrilátero}, a \omterm{def:g4:geometry:polygon}{pentágono} (cinco \omterm{def:g5:solids:def}{vértices}) y un
        \omterm{def:g4:geometry:polygon}{hexágono} (seis \omterm{def:g5:solids:def}{vértices}), y dibuja todas sus diagonales.
        Cuéntalos: ¿cuántos en cada caso?
  \item Contando sin dibujar, para el \omterm{def:g4:geometry:polygon}{hexágono}: de uno
        \omterm{def:g5:solids:def}{vértice}, ¿cuántas diagonales quedan? (¿A cuántos de los
        seis \omterm{def:g5:solids:def}{vértices} puede \emph{no} estar unidos por una diagonal?)
        Multiplicando por seis \omterm{def:g5:solids:def}{vértices} cuenta cada diagonal
        exactamente dos veces --- ¿por qué? Recupera tu conteo de pregunta
        11. (La misma doble contabilización apareció en los apretones de manos
        de \cref{pb:g6:wholes:1}.)
  \item Utilice el método de la pregunta 12 para contar las diagonales de
        un \omterm{def:g4:geometry:polygon}{polígono} con $10$ \omterm{def:g5:solids:def}{vértices}, entonces con $20$ \omterm{def:g5:solids:def}{vértices}.
  \item A \omterm{def:g4:geometry:polygon}{polígono} tiene exactamente $44$ diagonales. Cuántos \omterm{def:g5:solids:def}{vértices}
