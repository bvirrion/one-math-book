\section{Reflexiones}

\begin{definition}[Punto simétrico]\label{def:g6:symmetry:def}
Sea $d $ una línea. El \emph{simetrico}\index{simetrico} de un
punto $ M $ al otro lado de $ d $ es el punto $ M'$ tal que $ d $ es el
\omterm{def:g6:symmetry:bisector}{bisectriz perpendicular} de $[MM']$: el \omterm{def:g6:lines:objects}{segmento} $[MM']$ es \omterm{def:g6:lines:perp}{perpendicular}
a $ d $, y $ d $ lo corta por su punto medio. un punto \emph{en} $ d $ es su
propia simétrica.
\end{definition}

\begin{method}[Construyendo un punto simétrico]\label{met:g6:symmetry:construct}
Para construir el \omterm{def:g6:symmetry:def}{simétrico} $ M'$ de $ M $ a través de la línea $ d $:
\begin{enumerate}
  \item dibujar el \omterm{def:g6:lines:perp}{perpendicular} a $ d $ a través de $ M $ (escuadra); llamar
        $ H $ el punto donde se cruza $ d $;
  \item medida $ MH $ con el compás o regla;
  \item lugar $ M'$ en el mismo \omterm{def:g6:lines:perp}{perpendicular}, al otro lado de $ d $, con $ HM' = MH $.
\end{enumerate}
En papel cuadriculado es más rápido: cuenta los cuadrados desde $ M $ al eje, y
cuenta el mismo número en el otro lado.
\end{method}

\begin{example}[en una grilla]\label{ex:g6:symmetry:grid}
Si el eje es una línea de cuadrícula vertical y $ M $ es $3$ cuadrados a su izquierda,
entonces $ M'$ es $3$ cuadrados a su derecha, en la misma fila. para una diagonal
eje, el conteo funciona a lo largo de diagonales --- siempre \omterm{def:g6:lines:perp}{perpendicular} a
el eje.
\end{example}

\begin{proposition}[Que reflejo conserva]\label{prop:g6:symmetry:preserved}
El \omterm{def:g6:symmetry:def}{simétrico} de una figura a través de una línea es una figura del \emph{lo mismo
forma y mismo tamaño}: longitudes, ángulos, \omterm{def:g4:measure:perimeter}{perímetros} y \omterm{def:g4:measure:area}{áreas} son todos
conservado. El \omterm{def:g6:symmetry:def}{simétrico} de un \omterm{def:g6:lines:objects}{segmento} es un \omterm{def:g6:lines:objects}{segmento}, de un círculo a
circulo del mismo \omterm{def:g3:shapes:shapes}{radio}, de una línea una línea.
\end{proposition}
\admitted

\begin{remark}
una cosa es \emph{no} conservado: orientación. A reflejada ``b''
se convierte en una ``d'' --- escritura en espejo. Si trazas tu figura y volteas
el papel de calco reproduce la imagen, el reflejo es correcto.
\end{remark}

\section{Ejes de simetría de una figura.}

\begin{definition}[Eje de simetría]\label{def:g6:symmetry:axis}
una linea $ d $ es un \emph{eje de simetria}\index{eje de simetria} de un
figura cuando la reflexión a través $ d $ envía la figura exactamente a
mismo --- plegándose $ d $ hace que las dos mitades coincidan.
\end{definition}

\begin{example}\label{ex:g6:symmetry:shapes}
Contando ejes de simetría:
\begin{itemize}
  \item \omterm{def:g6:shapes:triangles}{triángulo isósceles}: $1$ (a través de la parte superior \omterm{def:g5:solids:def}{vértice} y el medio
        de la base); \omterm{def:g6:shapes:triangles}{triángulo equilatero}: $3$;
  \item rectángulo: $2$ (las dos ``líneas medias'', \emph{no} el
        diagonales); \omterm{def:g6:shapes:quadrilaterals}{rombo}: $2$ (¡sus diagonales!); cuadrado: $4$;
  \item círculo: cada línea que pasa por el centro --- infinitas.
\end{itemize}
\end{example}

\begin{example}[La simetría demuestra igualdades.]\label{ex:g6:symmetry:proves}
¿Por qué los dos ángulos base de un \omterm{def:g6:shapes:triangles}{triángulo isósceles} ¿igual? Doblar el
triángulo a lo largo de su \omterm{def:g6:symmetry:axis}{eje de simetría}: la izquierda \omterm{def:g3:division:half}{medio} aterriza exactamente en el
correcto \omterm{def:g3:division:half}{medio}, por lo que el ángulo de la base izquierda aterriza en el ángulo de la base derecha ---
deben tener la misma medida
(\cref{prop:g6:symmetry:preserved}: los reflejos preservan los ángulos).
\end{example}

\section{bisectriz perpendicular}

\begin{definition}[bisectriz perpendicular]\label{def:g6:symmetry:bisector}
El \emph{bisectriz perpendicular}\index{bisectriz perpendicular} de un
\omterm{def:g6:lines:objects}{segmento} $[AB]$ es la linea \omterm{def:g6:lines:perp}{perpendicular} a $[AB]$ a través de su
punto medio --- es el \omterm{def:g6:symmetry:axis}{eje de simetría} que intercambia $ A $ y $ B $.
\end{definition}

\begin{proposition}[Equidistancia]\label{prop:g6:symmetry:bisector}
un punto $ M $ esta en el \omterm{def:g6:symmetry:bisector}{bisectriz perpendicular} de $[AB]$ exactamente cuando
está a la misma distancia de $ A $ y de $ B $: $ MA = MB $.
\end{proposition}
\admitted

\begin{method}[Construcción de brújula]\label{met:g6:symmetry:bisector}
para dibujar el \omterm{def:g6:symmetry:bisector}{bisectriz perpendicular} de $[AB]$ sin medir:
\begin{enumerate}
  \item abre la brújula a más de \omterm{def:g3:division:half}{medio} de $ AB $;
  \item dibuja dos arcos centrados en $ A $, por encima y por debajo del \omterm{def:g6:lines:objects}{segmento};
  \item con el \emph{mismo} apertura, dibuje dos arcos centrados en $ B $;
  \item unir los dos puntos de cruce: esta línea es la \omterm{def:g6:symmetry:bisector}{bisectriz perpendicular} (cada punto de cruce está equidistante de $ A $ y $ B $).
\end{enumerate}
\end{method}

\section{Problema: La fuente, el plato roto y el caminar

\begin{problem}[{Problema de fin de semana --- las tres bisectrices perpendiculares de un triángulo se encuentran en un punto: el círculo que pasa por tres puntos, y lo que hacen dos espejos juntos}]
\label{pb:g6:symmetry:1}
En \cref{exo:g6:symmetry:8} una fuente tenía que estar a la misma altura
distancia de \emph{dos} árboles, y la respuesta fue toda una línea de
manchas. Este problema plantea un \emph{tercero} árbol --- y el conjunto
La línea colapsa en un único punto perfectamente determinado. eso
El punto esconde un tesoro de geometría: te permite dibujar el que
círculo que pasa por tres puntos dados, reconstruir un plato completo
de un fragmento roto y pertenece a cada triángulo. el ultimo
parte realiza un experimento con dos espejos cuyo resultado
anuncia la geometría del próximo año.

\medskip
\noindent\textbf{Parte I --- Two trees, then a map of
territories.}
\begin{enumerate}
  \item dibujar dos puntos $ A $ y $ B $ con $ AB = 6$ centímetros y
        construir el \omterm{def:g6:symmetry:bisector}{bisectriz perpendicular} de $[AB]$ con la brújula (\cref{met:g6:symmetry:bisector}). Elige un punto
        $ M $ sobre él y verifique con la regla que $ MA = MB $.
  \item \Cref{prop:g6:symmetry:bisector} dice dos cosas en
        una vez: cada punto \emph{en} la bisectriz es equidistante
        de $ A $ y $ B $, y cada punto equidistante \emph{es
        en} la bisectriz. ¿Cuál de los dos responde a la fuente?
        pregunta de \cref{exo:g6:symmetry:8}? ¿Qué hace el
        proposición dice sobre un punto que es \emph{no} en el
        bisectriz?
  \item toma un punto $ M $ en el mismo lado de la bisectriz que
        $ A $. El \omterm{def:g6:lines:objects}{segmento} $[MB]$ cruza la bisectriz en un punto
        $ P $. Compara el camino $ M \to P \to B $ con el camino

  \item dos escuelas $ A $ y $ B $ servir a un pueblo. Cada niño camina
