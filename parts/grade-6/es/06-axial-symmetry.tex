\chapter{La simetría axial}\label{ch:g6:symmetry}

Dobla una hoja de papel por una recta: cada punto cae sobre otro punto,
su imagen reflejada. Ese doblez es la \emph{\omterm{def:g6:symmetry:def}{simetría axial}} (o
reflexión), la primera transformación geométrica del libro --- la
segunda, la simetría central, llega en el \cref{ch:g7:central}.

\section{Reflexiones}

\begin{definition}[Punto simétrico]\label{def:g6:symmetry:def}
Sea $d$ una recta. El \emph{simétrico}\index{simetría axial} de un
punto $M$ respecto de $d$ es el punto $M'$ tal que $d$ es la
\omterm{def:g6:symmetry:bisector}{mediatriz} de $[MM']$: el \omterm{def:g6:lines:objects}{segmento} $[MM']$ es perpendicular
a $d$ y $d$ lo corta en su punto medio. Un punto que está \emph{sobre} $d$ es su
propio simétrico.
\end{definition}

\begin{method}[Construir el punto simétrico]\label{met:g6:symmetry:construct}
Para construir el \omterm{def:g6:symmetry:def}{simétrico} $M'$ de $M$ respecto de la recta $d$:
\begin{enumerate}
  \item traza la perpendicular a $d$ por $M$ (con la escuadra); llama
        $H$ al punto donde corta a $d$;
  \item mide $MH$ con el compás o con la regla;
  \item coloca $M'$ sobre esa misma perpendicular, al otro lado de $d$,
        con $HM' = MH$.
\end{enumerate}
En papel cuadriculado es más rápido: cuenta las casillas de $M$ al eje y
cuenta las mismas al otro lado.
\end{method}

\begin{example}[En la cuadrícula]\label{ex:g6:symmetry:grid}
Si el eje es una línea vertical de la cuadrícula y $M$ está $3$ casillas a su izquierda,
entonces $M'$ está $3$ casillas a su derecha, en la misma fila. Con un eje
diagonal, el recuento se hace por diagonales --- siempre \omterm{def:g6:lines:perp}{perpendiculares} al
eje.
\end{example}

\begin{proposition}[Qué conserva una reflexión]\label{prop:g6:symmetry:preserved}
El \omterm{def:g6:symmetry:def}{simétrico} de una figura respecto de una recta es una figura de la \emph{misma
forma y del mismo tamaño}: se conservan longitudes, ángulos, \omterm{def:g4:measure:perimeter}{perímetros} y \omterm{def:g4:measure:area}{áreas}.
El \omterm{def:g6:symmetry:def}{simétrico} de un \omterm{def:g6:lines:objects}{segmento} es un \omterm{def:g6:lines:objects}{segmento}; el de una circunferencia, una
circunferencia del mismo \omterm{def:g3:shapes:shapes}{radio}; el de una recta, una recta.
\end{proposition}
\admitted

\begin{remark}
Una cosa \emph{no} se conserva: la orientación. Una «b» reflejada
se convierte en una «d» --- escritura en espejo. Si al calcar tu figura y darle la vuelta
al papel de calco se reproduce la imagen, la reflexión es correcta.
\end{remark}

\section{Ejes de simetría de una figura}

\begin{definition}[Eje de simetría]\label{def:g6:symmetry:axis}
Una recta $d$ es un \emph{eje de simetría}\index{eje de simetría} de una
figura cuando la reflexión respecto de $d$ envía la figura exactamente sobre
sí misma --- doblar por $d$ hace que las dos \omterm{def:g3:division:half}{mitades} encajen.
\end{definition}

\end{omfigure}

\begin{example}\label{ex:g6:symmetry:shapes}
Recuento de \omterm{def:g6:symmetry:axis}{ejes de simetría}:
\begin{itemize}
  \item \omterm{def:g6:shapes:triangles}{triángulo isósceles}: $1$ (por el \omterm{def:g5:solids:def}{vértice} superior y el punto medio
        de la base); \omterm{def:g6:shapes:triangles}{triángulo equilátero}: $3$;
  \item rectángulo: $2$ (las dos «rectas medias», \emph{no} las
        diagonales); \omterm{def:g6:shapes:quadrilaterals}{rombo}: $2$ (¡sus diagonales!); cuadrado: $4$;
  \item circunferencia: toda recta que pase por el centro --- infinitos.
\end{itemize}
\end{example}

\begin{example}[La simetría demuestra igualdades]\label{ex:g6:symmetry:proves}
¿Por qué son iguales los dos ángulos de la base de un \omterm{def:g6:shapes:triangles}{triángulo isósceles}? Dobla el
triángulo por su \omterm{def:g6:symmetry:axis}{eje de simetría}: la \omterm{def:g3:division:half}{mitad} izquierda cae exactamente sobre la
\omterm{def:g3:division:half}{mitad} derecha, así que el ángulo izquierdo de la base cae sobre el derecho ---
tienen que medir lo mismo
(\cref{prop:g6:symmetry:preserved}: las reflexiones conservan los ángulos).
\end{example}

\section{La mediatriz}

\begin{definition}[Mediatriz]\label{def:g6:symmetry:bisector}
La \emph{mediatriz}\index{mediatriz} de un
\omterm{def:g6:lines:objects}{segmento} $[AB]$ es la recta perpendicular a $[AB]$ que pasa por su
punto medio --- es el \omterm{def:g6:symmetry:axis}{eje de simetría} que intercambia $A$ y $B$.
\end{definition}

\begin{proposition}[Equidistancia]\label{prop:g6:symmetry:bisector}
Un punto $M$ está en la \omterm{def:g6:symmetry:bisector}{mediatriz} de $[AB]$ exactamente cuando está
a la misma distancia de $A$ y de $B$: $MA = MB$.
\end{proposition}
\admitted

\begin{method}[Construcción con compás]\label{met:g6:symmetry:bisector}
Para trazar la \omterm{def:g6:symmetry:bisector}{mediatriz} de $[AB]$ sin medir:
\begin{enumerate}
  \item abre el compás más de la \omterm{def:g3:division:half}{mitad} de $AB$;
  \item traza dos arcos con centro en $A$, por encima y por debajo del \omterm{def:g6:lines:objects}{segmento};
  \item con la \emph{misma} abertura, traza dos arcos con centro en $B$;
  \item une los dos puntos de corte: esa recta es la \omterm{def:g6:symmetry:bisector}{mediatriz}
        (cada punto de corte equidista de $A$ y de $B$).
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:1}
Copia en papel cuadriculado: un eje vertical $d$, un punto $A$ dos casillas a su
izquierda y un punto $B$ cinco casillas a su izquierda y una casilla más arriba.
Construye sus puntos \omterm{def:g6:symmetry:def}{simétricos} $A'$ y $B'$ respecto de $d$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:2}
Traza una recta $d$ y un punto $M$ a $3$ cm de $d$ (no en papel
cuadriculado). Construye el \omterm{def:g6:symmetry:def}{simétrico} $M'$ de $M$ respecto de $d$ con escuadra
y regla, siguiendo el \cref{met:g6:symmetry:construct}. ¿A qué distancia
están $M$ y $M'$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:3}
¿Qué letras mayúsculas del alfabeto, escritas en su forma más sencilla,
tienen un \omterm{def:g6:symmetry:axis}{eje de simetría} vertical (como la A)? ¿Y uno horizontal (como la B)?
¿Y los dos?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:4}
¿Cuántos \omterm{def:g6:symmetry:axis}{ejes de simetría} tiene: un \omterm{def:g6:shapes:triangles}{triángulo equilátero}? ¿Un rectángulo
que no sea cuadrado? ¿Un \omterm{def:g6:shapes:quadrilaterals}{rombo} que no sea cuadrado? ¿Un cuadrado? ¿Una circunferencia?
Dibuja cada figura con sus ejes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:5}
¿Verdadero o falso? «Las diagonales de un rectángulo son \omterm{def:g6:symmetry:axis}{ejes de simetría} del
rectángulo.» Explícalo con un argumento de doblez o con un dibujo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:6}
Un triángulo $ABC$ se refleja respecto de una recta $d$ y da $A'B'C'$. Sabiendo que
$AB = 4$ cm, $\widehat{BAC} = 70^\circ$ y que el \omterm{def:g4:measure:perimeter}{perímetro} de $ABC$ es
$12$ cm: di, sin hacer ninguna construcción, $A'B'$,
$\widehat{B'A'C'}$ y el \omterm{def:g4:measure:perimeter}{perímetro} de $A'B'C'$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:7}
Traza un \omterm{def:g6:lines:objects}{segmento} $[AB]$ de $7$ cm y construye su \omterm{def:g6:symmetry:bisector}{mediatriz}
con el compás (\cref{met:g6:symmetry:bisector}). Elige un
punto cualquiera $M$ sobre ella y comprueba con la regla que $MA = MB$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:8}
Dos árboles están en los puntos $A$ y $B$ de un jardín (llano). ¿Dónde se puede
plantar una fuente para que esté a la misma distancia de los dos árboles?
Describe \emph{todos} los sitios posibles.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:9}
En papel cuadriculado, traza el eje $d$ (una diagonal de la cuadrícula a
$45^\circ$) y una figurita en forma de L a un lado. Construye su
figura simétrica respecto de $d$. (Refleja cada \omterm{def:g5:solids:def}{vértice}: con un eje a
$45^\circ$, un punto que está $2$ casillas a la derecha del eje cae $2$ casillas por encima de
él.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:10}
Traza una circunferencia de centro $O$, una recta $d$ que pase por $O$ y un punto $M$ sobre
la circunferencia. ¿Dónde está el \omterm{def:g6:symmetry:def}{simétrico} de $M$ respecto de $d$? Explica por qué la
reflexión envía la circunferencia sobre sí misma.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:11}
Usando la propiedad de equidistancia (\cref{prop:g6:symmetry:bisector}),
explica por qué los dos puntos de corte de los arcos del
\cref{met:g6:symmetry:bisector} están realmente en la \omterm{def:g6:symmetry:bisector}{mediatriz}
de $[AB]$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:symmetry:12}
Una bola de billar en el punto $A$ tiene que rebotar en una banda recta $d$ y
llegar al punto $B$ (los dos del mismo lado de la banda). Refleja $B$
respecto de $d$ y obtén $B'$, y traza el \omterm{def:g6:lines:objects}{segmento} $[AB']$. Explica por qué el
mejor punto de rebote es donde $[AB']$ corta a la banda. (Idea: el camino
$A \to P \to B$ tiene la misma longitud que $A \to P \to B'$.)
\end{exercise}

\section{Problema: La fuente, el plato roto y el espejo que
camina}

\begin{problem}[{Problema de fin de semana --- las tres mediatrices
de un triángulo se cortan en un punto: la circunferencia que pasa por tres
puntos y lo que hacen dos espejos juntos}]
\label{pb:g6:symmetry:1}
En el \cref{exo:g6:symmetry:8}, una fuente tenía que estar a la misma
distancia de \emph{dos} árboles, y la respuesta era toda una recta de
sitios. Este problema planta un \emph{tercer} árbol --- y toda la
recta se reduce a un único punto perfectamente determinado. Ese
punto esconde un tesoro de la geometría: permite trazar la única
circunferencia que pasa por tres puntos dados, reconstruir un plato entero
a partir de un fragmento, y pertenece a todo triángulo. La última
parte hace un experimento con dos espejos cuyo resultado
anuncia la geometría del año que viene.

\medskip
\noindent\textbf{Parte I --- Dos árboles y después un mapa de
territorios.}
\begin{enumerate}
  \item traza dos puntos $A$ y $B$ con $AB = 6$ cm y
        construye la \omterm{def:g6:symmetry:bisector}{mediatriz} de $[AB]$ con el
        compás (\cref{met:g6:symmetry:bisector}). Elige un punto
        $M$ sobre ella y comprueba con la regla que $MA = MB$;
  \item la \Cref{prop:g6:symmetry:bisector} dice dos cosas a la
        vez: todo punto \emph{de} la \omterm{def:g6:symmetry:bisector}{mediatriz} equidista
        de $A$ y de $B$, y todo punto equidistante \emph{está
        en} la \omterm{def:g6:symmetry:bisector}{mediatriz}. ¿Cuál de las dos responde a la pregunta de la
        fuente del \cref{exo:g6:symmetry:8}? ¿Qué dice la
        proposición sobre un punto que \emph{no} está en la
        \omterm{def:g6:symmetry:bisector}{mediatriz}?
  \item toma un punto $M$ del mismo lado de la \omterm{def:g6:symmetry:bisector}{mediatriz} que
        $A$. El \omterm{def:g6:lines:objects}{segmento} $[MB]$ corta a la \omterm{def:g6:symmetry:bisector}{mediatriz} en un punto
        $P$. Compara el camino $M \to P \to B$ con el camino
        $M \to P \to A$ y explica por qué $M$ está más cerca de $A$
        que de $B$: la \omterm{def:g6:symmetry:bisector}{mediatriz} parte la hoja en un
        «lado de $A$» y un «lado de $B$»;
  \item dos colegios $A$ y $B$ sirven a un pueblo. Cada niño va
        andando al colegio más cercano. Dibuja un mapa pequeño con $A$ y $B$
        y sombrea la región exacta del pueblo atendida por el colegio
        $A$. ¿Qué recta forma la frontera?
  \item abre un tercer colegio $C$ (colócalo de modo que los tres colegios
        formen un triángulo). Construye la \omterm{def:g6:symmetry:bisector}{mediatriz}
        de $[AB]$ y la \omterm{def:g6:symmetry:bisector}{mediatriz} de $[BC]$, y
        llama $O$ a su punto de corte. ¿Qué dos igualdades de
        distancias cumple $O$ y por qué?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Tres árboles, un punto, una
circunferencia.}
\begin{enumerate}[resume]
  \item deduce de la pregunta 5 que $OA = OC$ --- y, por tanto,
        que $O$ está también en la \omterm{def:g6:symmetry:bisector}{mediatriz} de
        $[AC]$, que nunca trazaste. Concluye: \emph{las tres
        mediatrices de los lados de un triángulo pasan todas
        por un mismo punto};
  \item explica por qué la circunferencia de centro $O$ y \omterm{def:g3:shapes:shapes}{radio} $OA$
        pasa también por $B$ y por $C$: es
        \emph{la} circunferencia que pasa por los tres puntos --- la
        \emph{circunferencia circunscrita}\index{circunferencia circunscrita}
        del triángulo $ABC$;
  \item construye un triángulo de lados $6$ cm, $5$ cm y
        $4$ cm (\cref{met:g6:shapes:construct}) y después su
        \omterm{pb:g6:symmetry:1}{circunferencia circunscrita}. ¿Por qué basta con trazar solo
        \emph{dos} mediatrices para hallar el centro?
  \item el plato roto: un arqueólogo desentierra un fragmento cuya
        única parte intacta es un trozo del borde circular del plato.
        Describe una receta que recupere el centro y el
        \omterm{def:g3:shapes:shapes}{radio} del plato original --- y pruébala: traza un
        arco de circunferencia con tu compás, «olvida» el centro,
        marca tres puntos del arco y reconstrúyelo;
  \item ¿podrían tres puntos no tener \emph{ninguna} circunferencia que pase por
        ellos? Supón que $A$, $B$ y $C$ están alineados. ¿Qué se puede
        decir de las mediatrices de $[AB]$ y de
        $[BC]$ (\cref{prop:g6:lines:facts})? Concluye.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Ejes por todas partes y el espejo que
camina.}
\begin{enumerate}[resume]
  \item el \Cref{ex:g6:symmetry:shapes} contó $3$ ejes para el
        \omterm{def:g6:shapes:triangles}{triángulo equilátero} y $4$ para el cuadrado. Conjetura
        el recuento para un \omterm{def:g4:geometry:polygon}{pentágono} regular y un \omterm{def:g4:geometry:polygon}{hexágono} regular,
        y describe por dónde pasan los ejes en cada caso (\omterm{def:g3:numbers:evenodd}{número
        impar} de lados frente a \omterm{def:g3:numbers:evenodd}{par}: las dos situaciones
        son distintas);
  \item dobla una hoja de papel una vez y recorta una figura atravesando las dos
        capas; desdobla. ¿Por qué tiene el agujero \emph{siempre} un
        \omterm{def:g6:symmetry:axis}{eje de simetría}, recortes lo que recortes, y qué recta es
        el eje?
  \item dobla la hoja dos veces --- el segundo doblez perpendicular
        al primero ---, recorta y desdobla. ¿Cuántos \omterm{def:g6:symmetry:axis}{ejes de
        simetría} tiene el agujero como mínimo y cuántas
        copias de tu recorte ves? (Los copos de nieve de papel son
        esta idea doblada más veces.)
  \item el espejo que camina: en papel cuadriculado, traza un eje vertical
        $d_1$, una banderita $F$ a su izquierda y un segundo
        eje vertical $d_2$, $4$ casillas a la derecha de $d_1$.
        Refleja $F$ respecto de $d_1$ (imagen $F'$) y después refleja
        $F'$ respecto de $d_2$ (imagen $F''$). Compara $F$ y $F''$:
        ¿misma orientación o reflejada? ¿Cuántas casillas se ha
        desplazado la figura? Compara con la distancia entre los
        ejes;
  \item el mismo experimento con $d_2$ \emph{perpendicular} a
        $d_1$: refleja la banderita respecto del eje vertical y después
        la imagen respecto del horizontal. Describe cómo queda $F''$
        respecto de $F$ y del punto de corte de los
        dos ejes. (Acabas de descubrir la transformación del año
        que viene: \cref{ch:g7:central}.)
\end{enumerate}
\end{problem}
