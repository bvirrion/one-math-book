\section*{अध्याय \ref{ch:g6:propdata} --- समानुपात और आँकड़े}

\begin{solution}{exo:g6:propdata:1}
सेब: \omterm{def:g3:division:remainder}{भागफल} $\frac52 = 2.5$, $\frac{7.5}{3} = 2.5$,
$\frac{12.5}{5} = 2.5$ --- सब बराबर: \omterm{def:g6:propdata:def}{समानुपाती} (दाम
$2.50$ प्रति kg)।

उम्र/ऊँचाई: $\frac{85}{2} = 42.5$ पर $\frac{103}{4} = 25.75$ ---
बराबर नहीं: \omterm{def:g6:propdata:def}{समानुपाती} नहीं।
\end{solution}

\begin{solution}{exo:g6:propdata:2}
एक क्रोसाँ: $4.80 \div 4 = 1.20$ यूरो। सात:
$7 \times 1.20 = 8.40$ यूरो।
\end{solution}

\begin{solution}{exo:g6:propdata:3}
गुणांक: $18 \div 10 = 1.8$ यूरो प्रति लीटर। फिर
$25$ L का दाम $25 \times 1.8 = 45$ यूरो; $72$ यूरो में
$72 \div 1.8 = 40$ L आते हैं; $60$ L का दाम $60 \times 1.8 = 108$ यूरो।
\end{solution}

\begin{solution}{exo:g6:propdata:4}
$250$ km के लिए: $100$ km वाले ईंधन का $2.5$ गुना, यानी
$2.5 \times 6 = 15$ L। $350$ km के लिए: $3.5 \times 6 = 21$ L।
$27$ L में: $27 \div 6 = 4.5$ सौ km, यानी $450$ km।
\end{solution}

\begin{solution}{exo:g6:propdata:5}
$84$ का $50\,\%$: \omterm{def:g3:division:half}{आधा}, $42$। \quad
$200$ का $25\,\%$: \omterm{def:g3:division:half}{चौथाई}, $50$। \quad
$63$ का $10\,\%$: दसवाँ भाग, $6.3$। \quad
$70$ का $30\,\%$: $3 \times 7 = 21$।
\end{solution}

\begin{solution}{exo:g6:propdata:6}
$450$ का $60\,\%$: $\frac{60}{100} \times 450 = 270$ विद्यार्थी
भोजनालय में खाते हैं; $450 - 270 = 180$ नहीं खाते।
\end{solution}

\begin{solution}{exo:g6:propdata:7}
$10$ से कम किताबें: मंगलवार ($8$) और गुरुवार ($6$)। बुधवार में
से गुरुवार: $17 - 6 = 11$ किताबें ज़्यादा।
\end{solution}

\begin{solution}{exo:g6:propdata:8}
$14$, $16$, $13$, $17$, $20$ ऊँचाइयों की पाँच पट्टियों वाला
आलेख (हर $2$ या $5$ डिग्री पर अंकन ठीक रहता है)। सबसे गर्म दिन:
शुक्रवार ($20$ डिग्री)।
\end{solution}

\begin{solution}{exo:g6:propdata:9}
$10$ लोगों के लिए $\frac{10}{6} = \frac53$ से गुणा करो: आटा
$450 \times \frac{10}{6} = 750$ g। अंडे:
$3 \times \frac{10}{6} = 5$ --- सौभाग्य से पूर्ण संख्या। (न होती,
तो पूरा पड़ने के लिए \emph{ऊपर} \omterm{def:g4:numbers:round}{गोल करना} पड़ता।)
\end{solution}

\begin{solution}{exo:g6:propdata:10}
\emph{1.} $2$ घंटे में: $48$ km। आधे घंटे में: $12$ km।
$1$ घंटा $30$ मिनट में: $24 + 12 = 36$ km।

\emph{2.} $60 \div 24 = 2.5$ घंटे, यानी $2$ घंटे $30$ मिनट।
\end{solution}

\begin{solution}{exo:g6:propdata:11}
गेंद: छूट $15$ का $20\,\% = 3$ यूरो, नया दाम $12$ यूरो।
रैकेट: छूट $8$ यूरो, नया दाम $32$ यूरो। हर हाल में नया दाम पुराने
का $80\,\%$ है: \omterm{def:g6:propdata:def}{समानुपाती}, गुणांक
$0.8$।
\end{solution}

\begin{solution}{exo:g6:propdata:12}
$t$ घंटे बाद मोटी मोमबत्ती अपनी ऊँचाई का $\frac t6$ जल चुकी है:
वह शुरुआती ऊँचाई के $1 - \frac t6$ पर खड़ी है; पतली वाली
$1 - \frac t4$ पर। हमें चाहिए
\[
1 - \frac t4 = \frac12 \left(1 - \frac t6\right)
\quad\text{अर्थात्}\quad
1 - \frac t4 = \frac12 - \frac t{12}.
\]
तब $\frac12 = \frac t4 - \frac t{12} = \frac{3t - t}{12} =
\frac{t}{6}$, इसलिए $t = 3$ घंटे। जाँच: $3$ घंटे बाद मोटी
मोमबत्ती $1 - \frac36 = \frac12$ पर है और पतली
$1 - \frac34 = \frac14$ पर --- सचमुच उसकी आधी।
\end{solution}

\begin{solution}{pb:g6:propdata:1}
\textbf{1.} $100\,000$ cm $= 1\,000$ m $= 1$ km: नक़्शे पर एक
सेंटीमीटर ज़मीन पर एक किलोमीटर है।

\textbf{2.} $7.5$ cm $\rightarrow$ गाँवों के बीच $7.5$ km;
$12$ cm $\rightarrow$ झील का $12$ km \omterm{def:g5:solids:def}{किनारा}।

\textbf{3.} $20$ km $\rightarrow$ नक़्शे पर $20$ cm।

\textbf{4.} $5$ km $= 500\,000$ cm, इसलिए मापनी
$1:500\,000$ है। पैदल यात्रा वाला नक़्शा ($1:100\,000$) ज़्यादा
बारीक़ है: वह वहाँ $5$ cm काग़ज़ लेता है जहाँ दूसरा केवल
$1$ cm ख़र्च करता है।

\textbf{5.}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccc}
नक़्शा (cm) & $1$ & $2$ & $7.5$ \\
\hline
असली (km) & $1$ & $2$ & $7.5$ \\
\end{tabular}
\end{center}
असली दूरी $=$ नक़्शे की दूरी $\times 1$ (इन्हीं मात्रकों में):
हमेशा \emph{वही} गुणक, और यही \omterm{def:g6:propdata:def}{समानुपात} की ठीक-ठीक परिभाषा है
(\cref{def:g6:propdata:def})। यहाँ गुणांक $1$
किलोमीटर प्रति सेंटीमीटर है।

\textbf{6.} $4$ m $= 400$ cm और $400 \div 50 = 8$;
$3$ m $= 300$ cm और $300 \div 50 = 6$: योजना पर
$8$ cm $\times$ $6$ cm का आयत दिखता है।

\textbf{7.} पलंग: $200 \div 50 = 4$ cm गुणा $90 \div 50 = 1.8$ cm।
दरवाज़ा: $80 \div 50 = 1.6$ cm।

\textbf{8.} असली \omterm{def:g6:measure:area}{क्षेत्रफल}: $4 \times 3 = 12$ m$^2$। योजना का
\omterm{def:g6:measure:area}{क्षेत्रफल}: $8 \times 6 = 48$ cm$^2$। बदलने पर:
$12$ m$^2 = 12 \times 10\,000 = 120\,000$ cm$^2$, और
\[
120\,000 \div 48 = 2\,500 .
\]
\omterm{def:g3:division:remainder}{भागफल} $50$ नहीं, $2\,500 = 50 \times 50$ है।

\textbf{9.} काग़ज़ का एक cm$^2$, $50$ cm गुणा $50$ cm के असली
वर्ग के लिए खड़ा है, जिसमें
$50 \times 50 = 2\,500$ cm$^2$ आते हैं। इसलिए लंबाइयाँ $50$ से
भाग खाएँ, तो \omterm{def:g6:measure:area}{क्षेत्रफल} $50 \times 50 = 2\,500$ से: \omterm{def:g6:measure:area}{क्षेत्रफल}
मापनी के \emph{वर्ग} पर चलते हैं।

\textbf{10.} $6 \times 2\,500 = 15\,000$ cm$^2$, और
$15\,000 \div 10\,000 = 1.5$ m$^2$।

\textbf{11.} चित्र सुझाता है कि रविवार को \emph{दुगुना} सामान
बिका (पट्टी दुगुनी ऊँची)। सच: बढ़त $105$ में से $5$ यूरो की है,
और $5 \div 105 \approx 0.048$: लगभग $5\,\%$ ज़्यादा,
$100\,\%$ ज़्यादा नहीं। प्रबंधक ने अक्ष के $0$ से शुरू होने की
जाँच वाली चेतावनी अनसुनी कर दी
(\cref{met:g6:propdata:read})।

\textbf{12.} अक्ष $0$ से हो, तो पट्टियाँ $105$ और
$110$ छोटी इकाइयों तक उठती हैं: लगभग एक ही ऊँचाई की दो
पट्टियाँ, $5\,\%$ के \omterm{ex:g1:subtraction:difference}{अंतर} का ईमानदार चित्र।

\textbf{13.} बढ़त: $40$ से शुरू होकर $20$ टिकट की बढ़त:
$\frac{20}{40} = 50\,\%$। घटत: $60$ से शुरू होकर $20$
टिकट की घटत: $\frac{20}{60} = \frac13 \approx
33\,\%$। वही $20$ टिकट, पर शुरुआती मान अलग --- प्रतिशत हमेशा
\emph{किसी चीज़ की} \omterm{def:g6:fractions:def}{भिन्न} होता है, और वह ``चीज़'' बदल गई।

\textbf{14.} $20\,\%$ छूट के बाद: $100 - 20 = 80$ यूरो।
अगली $10\,\%$ छूट $80$ पर लगती है: छूट $8$ यूरो, आख़िरी दाम
$72$ यूरो। कुल छूट: $100$ में से $28$ यूरो, यानी
$28\,\%$ --- $30\,\%$ नहीं। दूसरी छूट पहले से ही घटे हुए दाम पर
लगी, इसलिए उसके यूरो छोटे हैं: अलग-अलग राशियों के प्रतिशत जुड़ते
नहीं।

\textbf{15.} उस नक़्शे पर $1$ cm, $5$ km के लिए खड़ा है, इसलिए
$1$ cm$^2$, $5 \times 5 = 25$ km$^2$ के लिए (प्रश्न 9 का नियम)।
जंगल: $3 \times 25 = 75$ km$^2$। लंबाइयाँ मापनी पर चलती हैं;
\omterm{def:g6:measure:area}{क्षेत्रफल} उसके \emph{वर्ग} पर।

\end{solution}
