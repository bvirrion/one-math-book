\begin{solution}{exo:g6:symmetry:1}
$A'$, $d$ की \emph{दाईं} ओर दो खाने पर है, $A$ की ही पंक्ति में;
$B'$ पाँच खाने दाईं ओर, एक खाना ऊपर --- दर्पण वाली
जगहें।
\end{solution}

\begin{solution}{exo:g6:symmetry:2}
$M$ से होकर $d$ पर \omterm{def:g6:lines:perp}{लंब}, जिसका पाद $H$ है; फिर दूसरी ओर $M'$,
जहाँ $HM' = MH = 3$ cm। दूरी $MM'$,
$3 + 3 = 6$ cm है।
\end{solution}

\begin{solution}{exo:g6:symmetry:3}
खड़ी अक्ष (जैसे A): A, H, I, M, O, T, U, V, W, X, Y।
आड़ी अक्ष (जैसे B): B, C, D, E, H, I, K, O, X।
दोनों: H, I, O, X।
(ठीक उत्तर इस पर थोड़ा निर्भर करता है कि अक्षर कैसे लिखे गए हैं।)
\end{solution}

\begin{solution}{exo:g6:symmetry:4}
\omterm{def:g6:shapes:triangles}{समबाहु त्रिभुज}: $3$ अक्ष। आयत: $2$ (उसकी भुजाओं के \omterm{def:g6:symmetry:bisector}{लंब
समद्विभाजक})। \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज}: $2$ (उसके विकर्ण)। वर्ग: $4$।
वृत्त: अनगिनत (केंद्र से होकर हर रेखा)।
\end{solution}

\begin{solution}{exo:g6:symmetry:5}
\emph{ग़लत।} (वर्ग न होने वाले) आयत को उसके विकर्ण पर मोड़ने से
दोनों आधे नहीं मिलते: दोनों त्रिभुजों का आकार एक है पर वे अलग-अलग
बैठते हैं --- काग़ज़ की शीट पर आज़माओ। केवल वर्ग में ही विकर्ण
अक्ष का काम करते हैं।
\end{solution}

\begin{solution}{exo:g6:symmetry:6}
परावर्तन लंबाइयाँ, कोण और \omterm{def:g4:measure:perimeter}{परिमाप} बचाए रखते हैं
(\cref{prop:g6:symmetry:preserved}): $A'B' = 4$ cm,
$\widehat{B'A'C'} = 70^\circ$, $A'B'C'$ का \omterm{def:g4:measure:perimeter}{परिमाप} $= 12$ cm।
\end{solution}

\begin{solution}{exo:g6:symmetry:7}
\cref{met:g6:symmetry:bisector} की तरह परकार से रचना। रेखा पर
किसी भी $M$ के लिए मापपट्टी $MA = MB$ की पुष्टि कर देती है
(\cref{prop:g6:symmetry:bisector})।
\end{solution}

\begin{solution}{exo:g6:symmetry:8}
$A$ और $B$ से समान दूरी वाले सारे बिंदु $[AB]$ का \omterm{def:g6:symmetry:bisector}{लंब
समद्विभाजक} बनाते हैं (\cref{prop:g6:symmetry:bisector})।
फ़व्वारा उस रेखा पर कहीं भी (बग़ीचे के भीतर) लगाया जा सकता है।
\end{solution}

\begin{solution}{exo:g6:symmetry:9}
हर \omterm{def:g5:solids:def}{शीर्ष} अक्ष पर \omterm{def:g6:lines:perp}{लंब} दिशा में परावर्तित होता है: $45^\circ$
वाली अक्ष के लिए, अक्ष से $k$ खाने दाईं ओर का बिंदु उससे
$k$ खाने ऊपर आ जाता है (और उलटे भी) --- आड़े और खड़े खिसकाव आपस
में बदल जाते हैं। \omterm{def:g5:solids:def}{शीर्षों} को एक-एक करके परावर्तित करो, फिर
उन्हें क्रम से जोड़ दो।
\end{solution}

\begin{solution}{exo:g6:symmetry:10}
$M$ का \omterm{def:g6:symmetry:def}{सममित} बिंदु वह दूसरा बिंदु है जहाँ $M$ से होकर $d$ पर
खींचा \omterm{def:g6:lines:perp}{लंब} वृत्त को काटता है --- वह वृत्त पर ही पड़ता है। कारण:
परावर्तन दूरियाँ बचाए रखता है और $O$ को (जो $d$ पर है) वहीं
छोड़ देता है, इसलिए $O$ से $r$ दूरी वाले हर बिंदु का प्रतिबिंब
फिर $O$ से $r$ दूरी पर होता है: वृत्त ख़ुद पर आ जाता है।
\end{solution}

\begin{solution}{exo:g6:symmetry:11}
हर कटान-बिंदु $P$, $A$ और $B$ दोनों से परकार के एक ही खुलाव
$\rho$ से खींचा गया था: इसलिए $PA = \rho = PB$।
\cref{prop:g6:symmetry:bisector} के अनुसार $A$ और $B$ से समान
दूरी वाला बिंदु $[AB]$ के \omterm{def:g6:symmetry:bisector}{लंब समद्विभाजक} पर पड़ता है; ऐसे दो
बिंदु उस रेखा को तय कर देते हैं।
\end{solution}

\begin{solution}{exo:g6:symmetry:12}
दीवार पर किसी भी टकराव-बिंदु $P$ के लिए $PB = PB'$ ($d$ पर
परावर्तन दूरियाँ बचाए रखता है), इसलिए रास्ते की लंबाई
$AP + PB = AP + PB'$ है। टूटी रेखा $A \to P \to B'$ तब सबसे छोटी
होती है जब वह सीधी हो, यानी जब $P$ \omterm{def:g6:lines:objects}{रेखाखंड} $[AB']$ पर हो। तो
सबसे अच्छा टकराव-बिंदु $[AB']$ और दीवार का कटान है।
\end{solution}

\begin{solution}{pb:g6:symmetry:1}
\textbf{1.} \cref{met:g6:symmetry:bisector} की तरह रचना;
रेखा पर किसी भी $M$ के लिए मापपट्टी $MA = MB$ की पुष्टि कर देती है
(\cref{prop:g6:symmetry:bisector})।

\textbf{2.} फ़व्वारे वाले सवाल को दूसरी दिशा चाहिए:
\emph{सारे} समान-दूरी वाले बिंदु समद्विभाजक पर हैं, इसलिए संभव
जगहें ठीक वही रेखा हैं और कुछ नहीं। और समद्विभाजक पर न पड़ा
बिंदु इसलिए समान दूरी पर \emph{नहीं} है: वह दोनों पेड़ों में से
किसी एक के ठीक ज़्यादा पास है।

\textbf{3.} दोनों रास्तों में $[MP]$ साझा है, और $P$ के
समद्विभाजक पर होने से $PB = PA$। इसलिए रास्ता $M \to P \to B$
उतना ही लंबा है जितना रास्ता $M \to P \to A$, यानी $MP + PA$।
अब $MB$, $B$ तक का सीधा रास्ता है --- यहाँ वह $P$ से होकर जाने
वाला रास्ता \emph{ही} है, इसलिए $MB = MP + PA$; और $A$ तक का
सीधा रास्ता $P$ से होकर जाने वाले चक्कर से छोटा है:
$MA < MP + PA = MB$। समद्विभाजक की $A$ वाली ओर का हर बिंदु $A$ के ज़्यादा पास है।

\textbf{4.} सीमा $[AB]$ का \omterm{def:g6:symmetry:bisector}{लंब समद्विभाजक} है: विद्यालय $A$ ठीक
उस रेखा की अपनी ओर वाला \omterm{def:g3:division:half}{आधा} क़स्बा सँभालता है (प्रश्न 3),
विद्यालय $B$ दूसरा \omterm{def:g3:division:half}{आधा}, और रेखा पर ही रहने वाले बच्चे कोई भी चुन
सकते हैं।

\textbf{5.} $O$, $[AB]$ के समद्विभाजक पर है, इसलिए $OA = OB$;
और $[BC]$ के समद्विभाजक पर, इसलिए $OB = OC$
(\cref{prop:g6:symmetry:bisector})।

\textbf{6.} $OA = OB$ और $OB = OC$ से: $OA = OC$।
\cref{prop:g6:symmetry:bisector} की दूसरी दिशा से, $A$ और $C$ से
समान दूरी वाला बिंदु $[AC]$ के \omterm{def:g6:symmetry:bisector}{लंब समद्विभाजक} पर पड़ता है:
तीसरा समद्विभाजक बिना कहे ही $O$ से होकर जाता है। किसी भी
त्रिभुज के तीनों \omterm{def:g6:symmetry:bisector}{लंब समद्विभाजक} \emph{संगामी} होते हैं।

\textbf{7.} केंद्र $O$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $OA$ वाला वृत्त उन सारे
बिंदुओं से बनता है जो $O$ से $OA$ दूरी पर हैं
(\cref{def:g6:lines:circle}); चूँकि $OB = OA$ और $OC = OA$,
इसलिए $B$ और $C$ उस पर हैं। एक केंद्र, एक \omterm{def:g3:shapes:shapes}{त्रिज्या}, तीन बिंदु
सधे: यही $A$, $B$ और $C$ से होकर जाने वाला वृत्त है।

\textbf{8.} दो समद्विभाजक इसलिए काफ़ी हैं कि तीसरा अपने-आप आ
जाता है (प्रश्न 6): उनका कटान-बिंदु दूरियों की तीनों बराबरियाँ
पहले से पूरी करता है। वृत्त केंद्र $O$ से खींचा जाता है, और
परकार $O$ से तीनों में से किसी भी \omterm{def:g5:solids:def}{शीर्ष} तक खोला जाता है।

\textbf{9.} विधि: बची हुई कोर पर तीन बिंदु $A$, $B$, $C$ अंकित
करो; जीवाओं $[AB]$ और $[BC]$ के \omterm{def:g6:symmetry:bisector}{लंब समद्विभाजक} खींचो; उनका
कटान-बिंदु थाली का केंद्र है, और $A$ तक उसकी दूरी \omterm{def:g3:shapes:shapes}{त्रिज्या}।
यह इसलिए चलता है कि थाली की कोर तीनों अंकित बिंदुओं से होकर
जाने वाला वृत्त है --- और प्रश्न 8 उनसे होकर जाने वाला
\emph{इकलौता} वृत्त खोज देता है। आज़माया गया चाप हूबहू बन जाता है।

\textbf{10.} $A$, $B$, $C$ संरेख हों, तो $[AB]$ का समद्विभाजक
और $[BC]$ का समद्विभाजक दोनों एक ही रेखा $(AB) = (BC)$ पर \omterm{def:g6:lines:perp}{लंब}
हैं --- इसलिए वे आपस में \omterm{def:g6:lines:perp}{समांतर} हैं
(\cref{prop:g6:lines:facts}) और, अलग-अलग होने के कारण, कभी
नहीं कटते। कोई बिंदु तीनों से समान दूरी पर नहीं है, और तीन संरेख
बिंदुओं से होकर कोई वृत्त नहीं जाता।

\textbf{11.} सम \omterm{def:g4:geometry:polygon}{पंचभुज}: $5$ अक्ष, हर एक किसी \omterm{def:g5:solids:def}{शीर्ष} और सामने वाली
भुजा के मध्यबिंदु से होकर (विषम गिनती: हर अक्ष इसी एक तरह की
होती है)। सम \omterm{def:g4:geometry:polygon}{षट्भुज}: $6$ अक्ष --- तीन आमने-सामने के \omterm{def:g5:solids:def}{शीर्षों} से
होकर और तीन आमने-सामने की भुजाओं के मध्यबिंदुओं से होकर
(सम गिनती: दो तरह की, आधी-आधी)। प्रतिरूप: $n$ भुजाओं वाले सम
\omterm{def:g4:geometry:polygon}{बहुभुज} की $n$ अक्ष होती हैं।

\textbf{12.} कटाई एक साथ दोनों परतों में से होकर जाती है, इसलिए
दोनों परतें ठीक मिलते-जुलते टुकड़े खोती हैं; खोलना मोड़-रेखा पर
परावर्तन ही है, जो हर परत के \omterm{def:g5:solids:def}{किनारे} को दूसरी परत के \omterm{def:g5:solids:def}{किनारे} पर
भेज देता है। छेद अपना ही दर्पण-प्रतिबिंब है: मोड़-रेखा उसकी
\omterm{def:g6:symmetry:axis}{सममिति अक्ष} है --- चाहे जो भी काटा गया हो।

\textbf{13.} खुले हुए छेद की कम से कम $2$ \omterm{def:g6:symmetry:axis}{सममिति अक्ष} होती हैं,
दोनों मोड़-रेखाएँ, और कटाई $4$ प्रतियों में दिखती है (हर खुलाई
चित्र दुगुना कर देती है: $1 \to 2 \to 4$)। बर्फ़-फाहे की शैली
में और बार मोड़ने से प्रतियाँ और अक्ष और बढ़ती जाती हैं।

\textbf{14.} $F''$ का दिशा-क्रम $F$ जैसा \emph{ही} है (दो
दर्पण-पलटाव एक-दूसरे को काट देते हैं), नाप वही और ऊँचाई भी
वही --- वह बस $F$ को दाईं ओर खिसका देने जैसा है। खिसकाव
$8$ खाने का है: ठीक दोनों अक्षों के बीच की दूरी का \emph{दुगुना}।
दो \omterm{def:g6:lines:perp}{समांतर} दर्पण मिलकर दर्पण का काम करते ही नहीं: वे
\emph{स्थानांतरण} करते हैं।

\textbf{15.} $F''$ फिर उसी नाप का है, पर अब वह \emph{उलटा} दिखता
है, दोनों अक्षों के कटान-बिंदु के इर्द-गिर्द आधे घुमाव से घूमा
हुआ: $F''$ का हर बिंदु उस केंद्र से होकर $F$ के संगत बिंदु के
ठीक सामने पड़ता है। दो \omterm{def:g6:lines:perp}{लंब} दर्पण मिलकर \emph{\omterm{def:g3:division:half}{आधा} घुमाव} बनाते
हैं --- \cref{ch:g7:central} की केंद्रीय सममिति।

\end{solution}
