\begin{solution}{exo:g6:wholes:2}
सैकड़े का अंक: $2$; इकाई का अंक: $9$; $4$ दस-हज़ार गिनता है।
हिस्सों में बाँटना:
\[
47\,209 = 4 \times 10\,000 + 7 \times 1000 + 2 \times 100 + 0 \times 10
+ 9 .
\]
\end{solution}

\begin{solution}{exo:g6:wholes:4}
अंकों की संख्या से, $978$ और $1\,001$, $2$ से शुरू होने वाली
तीनों चार-अंकी संख्याओं से पहले आते हैं; उनमें दहाई और इकाई की
तुलना करो: $2\,034 < 2\,043 < 2\,304$। अंतिम क्रम:
\[
978 < 1\,001 < 2\,034 < 2\,043 < 2\,304 .
\]
\end{solution}

\begin{solution}{exo:g6:wholes:6}

8\,300$।

$4 \times 78 \times 25 = (4 \times 25) \times 78 = 100 \times 78 =

\end{solution}

\begin{solution}{exo:g6:wholes:7}
$95 = 7 \times 13 + 4$ (\omterm{def:g3:division:remainder}{भागफल} $13$, \omterm{def:g3:division:remainder}{शेषफल} $4$)।

$230 = 6 \times 38 + 2$ (\omterm{def:g3:division:remainder}{भागफल} $38$, \omterm{def:g3:division:remainder}{शेषफल} $2$)।

$504 = 8 \times 63 + 0$ (\omterm{def:g3:division:remainder}{भागफल} $63$, \omterm{def:g3:division:remainder}{शेषफल} $0$: $504$, $8$ से
\omterm{def:g6:wholes:divisible}{विभाज्य} है)।
\end{solution}

\begin{solution}{exo:g6:wholes:8}
संबंध तो सही है, पर \omterm{def:g3:division:remainder}{शेषफल} $13$ \omterm{def:g5:division:multiple}{भाजक} $8$ से छोटा \emph{नहीं} है,
इसलिए यह यूक्लिडीय भाग नहीं है। \omterm{def:g3:division:remainder}{शेषफल} में से एक $8$ और निकालो:
$173 = 8 \times 21 + 5$, जिसमें $0 \leq 5 < 8$:
\omterm{def:g3:division:remainder}{भागफल} $21$, \omterm{def:g3:division:remainder}{शेषफल} $5$।
\end{solution}

\begin{solution}{exo:g6:wholes:9}
\emph{1.} $2\,000 = 12 \times 166 + 8$: खेत $166$ डिब्बे पूरे
भरता है और $8$ अंडे बच जाते हैं।

\emph{2.} $150 = 12 \times 12 + 6$: बारह डिब्बों में केवल $144$
अंडे आते हैं, जो कम पड़ेंगे। नानबाई को $13$ डिब्बे ख़रीदने होंगे।
\end{solution}

\begin{solution}{exo:g6:wholes:10}
$100 = 7 \times 14 + 2$: सौ दिन यानी $14$ पूरे हफ़्ते और $2$ दिन
और। चौदह हफ़्ते बाद फिर मंगलवार होगा; मंगलवार के दो दिन बाद
\emph{गुरुवार} है।
\end{solution}

\begin{solution}{exo:g6:wholes:11}
$9$ से भाग देने पर सबसे बड़ा संभव \omterm{def:g3:division:remainder}{शेषफल} $8$ है (\omterm{def:g3:division:remainder}{शेषफल} \omterm{def:g5:division:multiple}{भाजक} से
छोटा होना चाहिए)। इसलिए भाज्य है
\[
9 \times 56 + 8 = 504 + 8 = 512 .
\]
\end{solution}

\begin{solution}{exo:g6:wholes:12}
अंक मान लो $h$ (सैकड़ा), $0$ (दहाई), $u$ (इकाई)। हमें पता है
$h = 2 \times u$ और $h + 0 + u = 9$, इसलिए $2u + u = 9$, यानी
$3u = 9$, $u = 3$, और $h = 6$। संख्या $603$ है। जाँच: $6$, $3$
का दुगुना है, दहाई का अंक $0$ है, और $6 + 0 + 3 = 9$।
\end{solution}

\begin{solution}{pb:g6:wholes:1}
\textbf{1.} जैसे $(1 + 4) + (2 + 3) + 5 = 5 + 5 + 5 =
15$।

\textbf{2.} हर जोड़ा $11$ का है: $1 + 10$, $2 + 9$, $3 + 8$,
$4 + 7$, $5 + 6$। यानी $5$ जोड़े (दस संख्याएँ, हर जोड़े में दो),
इसलिए \omterm{def:g1:addition:def}{योगफल} $5 \times 11 = 55$ है।

\textbf{3.} जोड़े $21$ के ($1 + 20$, $2 + 19, \dots$), और वे
$10$ हैं: $10 \times 21 = 210$।

\textbf{4.} हर जोड़े में पहली संख्या $1$ बढ़ती है तो दूसरी $1$
घटती है, इसलिए कुल कभी नहीं बदलता:
$1 + 100 = 2 + 99 = 3 + 98 = \dots = 101$। सौ संख्याएँ
$50$ जोड़े बनाती हैं (हर जोड़े में दो), जिनमें सबसे भीतरी
$50 + 51 = 101$ है।

\textbf{5.} $101$ के $50$ जोड़े:
\[
1 + 2 + \dots + 100 = 50 \times 101 = 5\,050 .
\]

\textbf{6.} नौ संख्याओं में बीच वाली, $5$, बिना जोड़ीदार रह
जाती है। उसके इर्द-गिर्द: $1 + 9 = 2 + 8 = 3 + 7 = 4 + 6 = 10$,
चार जोड़े। कुल: $4 \times 10 + 5 = 45$।

\textbf{7.} हर स्तंभ $1 + 9 = 2 + 8 = \dots =
9 + 1 = 10$ का है, और स्तंभ $9$ हैं: दोनों पंक्तियाँ मिलकर
$9 \times 10 = 90$ की हैं। पर दोनों पंक्तियाँ \emph{वही} \omterm{def:g1:addition:def}{योगफल}
दो बार लिखा हुआ हैं, इसलिए एक प्रति उसकी आधी है:
$1 + 2 + \dots + 9 = 90 \div 2 = 45$। कोई संख्या कभी बिना
जोड़ीदार नहीं रहती --- उसका जोड़ीदार ठीक उसके नीचे होता है।

\textbf{8.} $1$ से $50$ तक: $51$ के $50$ स्तंभ, दुगुना \omterm{def:g1:addition:def}{योगफल}
$50 \times 51 = 2\,550$, इसलिए \omterm{def:g1:addition:def}{योगफल} $2\,550 \div 2 = 1\,275$।
$1$ से $200$ तक: $201$ के $200$ स्तंभ, दुगुना \omterm{def:g1:addition:def}{योगफल}
$40\,200$, \omterm{def:g1:addition:def}{योगफल} $20\,100$।

\textbf{9.} हर \omterm{def:g3:numbers:evenodd}{सम संख्या} गाउस के \omterm{def:g1:addition:def}{योगफल} की किसी संख्या की दुगुनी
है: $2 = 2 \times 1$, $4 = 2 \times 2$, \dots,
$200 = 2 \times 100$। इसलिए पूरा \omterm{def:g1:addition:def}{योगफल} गाउस के \omterm{def:g1:addition:def}{योगफल} का दुगुना है:
\[
2 + 4 + \dots + 200 = 2 \times 5\,050 = 10\,100 .
\]

\textbf{10.} $101$ से $200$ तक की संख्याएँ, $1$ से $200$ तक की
संख्याओं में से $1$ से $100$ तक की संख्याएँ हटाने पर मिलती हैं:
\[
101 + 102 + \dots + 200 = 20\,100 - 5\,050 = 15\,050 .
\]

\textbf{11.} दोस्तों को पंक्ति में खड़ा करो। पहला $7$ हाथ
मिलाता है (बाक़ी हर एक से एक)। दूसरा पहले से पहले ही मिल चुका
है, इसलिए वह $6$ \emph{नए} हाथ मिलाता है; तीसरा $5$; और ऐसे ही,
सातवाँ $1$ और आठवाँ कोई नया नहीं।
हर मिलाना ठीक एक बार गिना जाता है:
\[
7 + 6 + 5 + 4 + 3 + 2 + 1 = 28 \text{ बार हाथ मिलाना}
\]
(बीच के इर्द-गिर्द जोड़े बनाओ: $(7+1) + (6+2) + (5+3) + 4 =
8 + 8 + 8 + 4 = 28$)।

\textbf{12.} $1 + 2 + \dots + 12$: $13$ के बारह स्तंभ,
दुगुना \omterm{def:g1:addition:def}{योगफल} $12 \times 13 = 156$, इसलिए
$156 \div 2 = 78$ टिन।

\textbf{13.} $1 + 2 + 3 + 4$ वर्गों की सीढ़ी, और उसके साथ वही
सीढ़ी उलटी रखी हुई, ठीक-ठीक $4$ पंक्तियों और $5$ स्तंभों का आयत
भर देती हैं: आयत की हर पंक्ति दो पंक्तियों वाली तरकीब का एक
स्तंभ है (चढ़ती सीढ़ी का एक कदम, उतरती सीढ़ी के एक कदम से पूरा
हुआ, $1 + 4$, $2 + 3$, $3 + 2$, $4 + 1$ --- हमेशा $5$)। तो
\omterm{def:g1:addition:def}{योगफल} का दुगुना $4 \times 5 = 20$ वर्ग है, और
$1 + 2 + 3 + 4 = 10$: चित्र \emph{ही} प्रश्न 7 है।

\textbf{14.} बारह घंटों में $1 + 2 + \dots + 12 = 78$ बार
टनटनाहट (प्रश्न 12), इसलिए पूरे दिन में $78 \times 2 = 156$ बार।

\textbf{15.} $60$ से भाग दो: $5\,050 = 60 \times 84 + 10$, यानी
$5\,050$ सेकंड $84$ मिनट और $10$ सेकंड हैं। मिनटों में $60$ का
भाग दो: $84 = 60 \times 1 + 24$, यानी $84$ मिनट $1$ घंटा
$24$ मिनट हैं। कुल मिलाकर: $1$ घंटा $24$ मिनट $10$ सेकंड ---
डेढ़ घंटे ($1$ घंटा $30$ मिनट) से कुछ \emph{कम}।

\end{solution}
