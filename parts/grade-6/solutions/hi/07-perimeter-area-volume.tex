\section*{अध्याय \ref{ch:g6:measure} --- परिमाप, क्षेत्रफल, आयतन}

\begin{solution}{exo:g6:measure:1}
$3.5$ m $= 350$ cm; \quad $420$ mm $= 42$ cm; \quad
$0.8$ km $= 800$ m; \quad $25\,000$ m $= 25$ km।
\end{solution}

\begin{solution}{exo:g6:measure:2}
आयत: $2 \times (8 + 3.5) = 2 \times 11.5 = 23$ cm।
वर्ग: $4 \times 6.2 = 24.8$ cm।
त्रिभुज: $5 + 7 + 9 = 21$ cm।
\end{solution}

\begin{solution}{exo:g6:measure:3}
एक चक्कर: $2\pi \times 50 = 100\pi \approx 314$ m। $2$ km
$= 2000$ m के लिए: $2000 \div 314 \approx 6.4$, इसलिए $7$ पूरे
चक्कर चाहिए ($6$ चक्कर केवल लगभग $1\,885$ m बनाते हैं)।
\end{solution}

\begin{solution}{exo:g6:measure:4}
आयत: $9 \times 4 = 36$ cm$^2$। वर्ग: $7^2 = 49$ m$^2$।
\omterm{def:g6:shapes:triangles}{समकोण त्रिभुज}: $\frac{6 \times 10}{2} = 30$ cm$^2$।
चक्र: $\pi \times 3^2 = 9\pi \approx 28$ cm$^2$।
\end{solution}

\begin{solution}{exo:g6:measure:5}
$3$ m$^2 = 30\,000$ cm$^2$; \quad
$45\,000$ cm$^2 = 4.5$ m$^2$; \quad
$2.5$ L $= 2\,500$ cm$^3$; \quad
$4\,500$ L $= 4.5$ m$^3$।
\end{solution}

\begin{solution}{exo:g6:measure:6}
बाड़ (\omterm{def:g6:measure:perimeter}{परिमाप}): $2 \times (120 + 85) = 2 \times 205 = 410$ m।
\omterm{def:g6:measure:area}{क्षेत्रफल}: $120 \times 85 = 10\,200$ m$^2$।
\end{solution}

\begin{solution}{exo:g6:measure:7}
\omterm{def:g5:solids:def}{घनाभ}: $5 \times 4 \times 10 = 200$ cm$^3$। \omterm{def:g5:solids:def}{घन}:
$3 \times 3 \times 3 = 27$ cm$^3$।
\end{solution}

\begin{solution}{exo:g6:measure:8}
\omterm{def:g6:measure:perimeter}{परिमाप} $16$ cm का मतलब लंबाई $+$ चौड़ाई $= 8$ cm। जैसे
$6 \times 2$ (\omterm{def:g6:measure:area}{क्षेत्रफल} $12$ cm$^2$) और $5 \times 3$ (\omterm{def:g6:measure:area}{क्षेत्रफल} $15$ cm$^2$)
--- या वर्ग $4 \times 4$ (\omterm{def:g6:measure:area}{क्षेत्रफल} $16$ cm$^2$)। आकृति वर्ग के
जितनी पास, \omterm{def:g6:measure:area}{क्षेत्रफल} उतना बड़ा।
\end{solution}

\begin{solution}{exo:g6:measure:9}
\omterm{def:g6:measure:area}{क्षेत्रफल}: $10 \times 2 + 2 \times 6 = 20 + 12 = 32$ cm$^2$।

\omterm{def:g6:measure:perimeter}{परिमाप} (डंडी पट्टी के ठीक बीच में): सीमा पर चलते हुए,
\[
10 + 2 + 4 + 6 + 2 + 6 + 4 + 2 = 36 \text{ cm}
\]
(ऊपर; पट्टी का दायाँ सिरा; नीचे दाईं ओर; डंडी की दाईं भुजा;
डंडी का निचला सिरा; डंडी की बाईं भुजा; नीचे बाईं ओर; पट्टी का बायाँ सिरा)।
\end{solution}

\begin{solution}{exo:g6:measure:10}
\emph{1.} $V = 25 \times 10 \times 2 = 500$ m$^3$।

\emph{2.} $500$ m$^3 = 500 \times 1000 = 500\,000$ L।

\emph{3.} $500\,000 \div 50\,000 = 10$ घंटे।
\end{solution}

\begin{solution}{exo:g6:measure:11}
बग़ीचा: $20^2 = 400$ m$^2$। तालाब: $\pi \times 3^2 = 9\pi$ m$^2$।
घास: $400 - 9\pi \approx 400 - 28.3 \approx 372$ m$^2$।
\end{solution}

\begin{solution}{exo:g6:measure:12}
\emph{1.} छोटी चॉकलेट: $15 \times 6 \times 1 = 90$ cm$^3$। बड़ी:
$30 \times 12 \times 2 = 720$ cm$^3$। \omterm{def:g6:measure:volume}{आयतन}
$720 \div 90 = 8$ गुना हो जाता है --- तीनों नाप दुगुने करने पर
\omterm{def:g6:measure:volume}{आयतन} $2 \times 2 \times 2 = 8$ गुना हो जाता है।

\emph{2.} चार गुने दाम में आठ गुनी चॉकलेट: हाँ, प्रति ग्राम बड़ी
चॉकलेट दुगुने फ़ायदे की है।
\end{solution}

\begin{solution}{pb:g6:measure:1}
\textbf{1.} $1 \times 9$: \omterm{def:g6:measure:perimeter}{परिमाप} $2 \times (1 + 9) = 20$ m,
\omterm{def:g6:measure:area}{क्षेत्रफल} $9$ m$^2$। \ $2 \times 8$: \omterm{def:g6:measure:perimeter}{परिमाप} $2 \times 10 = 20$ m,
\omterm{def:g6:measure:area}{क्षेत्रफल} $16$ m$^2$।

\textbf{2.} \omterm{def:g6:measure:perimeter}{परिमाप} (लंबाई $+$ चौड़ाई) का दुगुना है
(\cref{ex:g6:measure:perimeter}), इसलिए लंबाई $+$ चौड़ाई
$= 20 \div 2 = 10$ m। सारणी:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccc}
बाड़ा & $1 \times 9$ & $2 \times 8$ & $3 \times 7$ & $4 \times 6$
& $5 \times 5$ \\
\hline
\omterm{def:g6:measure:area}{क्षेत्रफल} (m$^2$) & $9$ & $16$ & $21$ & $24$ & $25$ \\
\end{tabular}
\end{center}
वर्ग $5 \times 5$ सबसे अच्छा है।

\textbf{3.} $4.5 \times 5.5 = 24.75$ और $4.9 \times 5.1 =
24.99$: $25$ के और-और पास, पर फिर भी उससे नीचे। अनुमान:
वर्ग \omterm{def:g6:measure:perimeter}{परिमाप} $20$ वाले \emph{हर} आयत को हरा देता है, दशमलव
भुजाओं वाले भी।

\textbf{4.} सारणी से: योग $10$ वाली दो संख्याओं का \omterm{def:g2:mult:def}{गुणनफल} तब
सबसे बड़ा होता है जब दोनों संख्याएँ \emph{बराबर} हों ($5$ और
$5$)। सामान्य नियम: तय योग के लिए \omterm{def:g2:mult:def}{गुणनफल} बराबर हिस्सों पर सबसे
बड़ा होता है --- जोड़ी जितनी असमान, \omterm{def:g2:mult:def}{गुणनफल} उतना ही छोटा।

\textbf{5.} $5 \times 5$ के वर्ग से ऊपर वाली पंक्ति हटाओ ---
$5$ इकाई वर्गों की पट्टी --- तो $5 \times 4$ का बाड़ा बचता है;
फिर एक सिरे पर $4$ इकाई वर्गों का स्तंभ चिपका दो, जिससे बाड़ा
$6 \times 4$ बन जाता है। पाँच वर्ग निकले और केवल चार लौटे: जुड़ी
हुई पट्टी $4$ वाली भुजा पर लगी है, जो उस $5$ वाली भुजा से छोटी
है जिस पर हटाई गई पट्टी थी। शुद्ध घाटा:
एक वर्ग मीटर ($25 \to 24$)। अगला कदम, $6 \times 4$ से
$7 \times 3$, $6$ की पंक्ति देकर $3$ का स्तंभ लेता है और तीन
और गँवाता है ($24 \to 21$): बाड़ा वर्ग से जितना दूर, उतनी ही
लंबी पट्टी वह देता है और उतनी ही छोटी वापस पाता है।

\textbf{6.} $w + L + w = 20$ के साथ, यानी $L = 20 - 2w$:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccccc}
$w$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ & $7$ \\
\hline
$L$ & $18$ & $16$ & $14$ & $12$ & $10$ & $8$ & $6$ \\
\omterm{def:g6:measure:area}{क्षेत्रफल} & $18$ & $32$ & $42$ & $48$ & $50$ & $48$ & $42$ \\
\end{tabular}
\end{center}
जीतने वाला $5 \times 10$ है, \omterm{def:g6:measure:area}{क्षेत्रफल} $50$ m$^2$ --- दीवार के
साथ की लंबाई चौड़ाई की दुगुनी: वर्ग \emph{नहीं}।

\textbf{7.} बाड़े को दीवार पर परावर्तित करो: बाड़ा और उसका
दर्पण-प्रतिबिंब मिलकर ऐसा दुगुना बाड़ा बनाते हैं जो बाड़ दो बार
इस्तेमाल करता है, चारों ओर $40$ m बाड़ (दीवार वाली भुजा अब भीतर
है)। भाग I के अनुसार \omterm{def:g6:measure:perimeter}{परिमाप} $40$ वाला सबसे अच्छा आयत वर्ग
$10 \times 10$ है, \omterm{def:g6:measure:area}{क्षेत्रफल} $100$ m$^2$। सबसे अच्छा असली बाड़ा
उसी सबसे अच्छे दुगुने बाड़े का \omterm{def:g3:division:half}{आधा} है: दीवार के साथ $10$ m,
गहराई $5$ m, \omterm{def:g6:measure:area}{क्षेत्रफल} $50$ m$^2$ --- ठीक सारणी वाला विजेता, अब समझाया हुआ।

\textbf{8.} परिधि $20 = 2 \times \pi \times r$, इसलिए
$r = 20 \div 6.28 \approx 3.18$ m। \omterm{def:g6:measure:area}{क्षेत्रफल}:
$\pi r^2 \approx 3.14 \times 3.18 \times 3.18 \approx
31.8$ m$^2$ --- वर्ग के $25$ m$^2$ से आराम से ज़्यादा।
दीदो जानती थी कि वह क्या कर रही है: सीमा की दी हुई लंबाई के लिए
वृत्त सबसे ज़्यादा घेरता है।

\textbf{9.} \omterm{def:g6:measure:perimeter}{परिमाप}: $1 \times 36$: $74$ m; \ $2 \times 18$:
$40$ m; \ $3 \times 12$: $30$ m; \ $4 \times 9$: $26$ m; \
$6 \times 6$: $24$ m। फिर वर्ग --- दिए हुए \omterm{def:g6:measure:area}{क्षेत्रफल} के लिए सबसे
कम बाड़। यह भाग I उलटा है: तय \omterm{def:g6:measure:perimeter}{परिमाप}, सबसे बड़ा \omterm{def:g6:measure:area}{क्षेत्रफल}, या तय
\omterm{def:g6:measure:area}{क्षेत्रफल}, सबसे छोटा \omterm{def:g6:measure:perimeter}{परिमाप}: दोनों वर्ग को ही ताज पहनाते हैं
(और सारे आयतों से आगे, वृत्त को)।

\textbf{10.} गोल दीवार अपने घेरे हुए स्थान के लिए सबसे छोटी सीमा
है (प्रश्न 8), इसलिए गोल टिन, पाइप या टंकी सबसे कम धातु में सबसे
ज़्यादा सामान रखती है --- सामग्री ही पैसा है।

\textbf{11.} जैसे $50$ m $\times$ $0.01$ m: \omterm{def:g6:measure:perimeter}{परिमाप}
$2 \times (50 + 0.01) = 100.02$ m, \omterm{def:g6:measure:area}{क्षेत्रफल}
$50 \times 0.01 = 0.5$ m$^2$। लंबा और पतला: किलोमीटरों की
सीमा, ज़मीन नाम भर की।

\textbf{12.} ग़लत। प्रश्न 11 वाले बाड़े का \omterm{def:g6:measure:perimeter}{परिमाप}
$100.02$ m और \omterm{def:g6:measure:area}{क्षेत्रफल} $0.5$ m$^2$ है, जबकि $5 \times 5$ के
बाड़े का \omterm{def:g6:measure:perimeter}{परिमाप} $20$ m और \omterm{def:g6:measure:area}{क्षेत्रफल} $25$ m$^2$: बड़ा \omterm{def:g6:measure:perimeter}{परिमाप}, कहीं
छोटा \omterm{def:g6:measure:area}{क्षेत्रफल}।

\textbf{13.} \omterm{def:g6:measure:area}{क्षेत्रफल}: एक इकाई वर्ग कम हो गया,
$24 - 1 = 23$ cm$^2$। \omterm{def:g6:measure:perimeter}{परिमाप}: सीमा पर चलते हुए, खाँचा
$1$ cm सीधी दीवार की जगह छोटे वर्ग की तीन भुजाएँ रख देता है,
$1 + 1 + 1 = 3$ cm: \omterm{def:g6:measure:perimeter}{परिमाप} $20$ cm से बढ़कर
$20 - 1 + 3 = 22$ cm हो जाता है। सामान हटाने से सीमा
\emph{लंबी} हो गई।

\textbf{14.} असली तट पर हर ज़ूम नई खाड़ियाँ, चट्टानें और नाले
दिखा देता है --- खाँचों पर खाँचे, और प्रश्न 13 बताता है कि हर
खाँचा सीमा की लंबाई बढ़ाता है पर \omterm{def:g6:measure:area}{क्षेत्रफल} में मुश्किल से कुछ
बदलता है। इसलिए नापी गई लंबाई बारीक़ी के साथ बढ़ती जाती है
(``तटरेखा का विरोधाभास''), जबकि नापा गया \omterm{def:g6:measure:area}{क्षेत्रफल} ठहर जाता है:
\omterm{def:g6:measure:perimeter}{परिमाप} और \omterm{def:g6:measure:area}{क्षेत्रफल} सचमुच स्वतंत्र राशियाँ हैं।

\textbf{15.} दर्पण वाली दलील: दुगुना बाड़ा
$2 \times 36 = 72$ m बाड़ इस्तेमाल करता, और \omterm{def:g6:measure:perimeter}{परिमाप} $72$ वाला
सबसे अच्छा आयत $18 \times 18$ का वर्ग है। तो सबसे अच्छा बाड़ा
दीवार के साथ $18$ m और गहराई $9$ m का है: बाड़ की जाँच
$9 + 18 + 9 = 36$ m, \omterm{def:g6:measure:area}{क्षेत्रफल} $18 \times 9 = 162$ m$^2$। खुले
मैदान में सबसे अच्छा $9 \times 9$ का वर्ग है, \omterm{def:g6:measure:area}{क्षेत्रफल} $81$ m$^2$।
खलिहान की दीवार किसान की ज़मीन ठीक \emph{दुगुनी} कर देती है।

\end{solution}
