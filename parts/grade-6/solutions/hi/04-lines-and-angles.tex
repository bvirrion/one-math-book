\section*{अध्याय \ref{ch:g6:lines} --- रेखाएँ, वृत्त और कोण}

\begin{solution}{exo:g6:lines:1}
मुक्त रचना। \omterm{def:g6:lines:objects}{रेखाखंड} $A$ और $B$ पर रुक जाता है; किरण $C$ से शुरू
होकर $A$ के आगे चलती रहती है; रेखा $B$ और $C$ दोनों ओर चलती
रहती है।
\end{solution}

\begin{solution}{exo:g6:lines:2}
``$[AB] = [BA]$'': \emph{सही} --- दोनों बिंदुओं के बीच का भाग
क्रम पर निर्भर नहीं करता।

``$[AB) = [BA)$'': \emph{ग़लत} --- $[AB)$, $A$ से शुरू होती है,
$[BA)$, $B$ से; वे उलटी दिशाओं में जाती हैं।

``$(AB) = (BA)$'': \emph{सही} --- दोनों नाम एक ही असीम रेखा
बताते हैं।
\end{solution}

\begin{solution}{exo:g6:lines:3}
कदम: \omterm{def:g3:shapes:rightangle}{समकोण} पट्टिका को $d$ के साथ तब तक सरकाओ जब तक उसका \omterm{def:g6:lines:perp}{लंब} वाला
\omterm{def:g5:solids:def}{किनारा} $P$ से होकर न जाए; वह \omterm{def:g6:lines:perp}{लंब} खींचो, उसे $p$ कहो। फिर उसी तरह
$P$ से होकर $p$ पर \omterm{def:g6:lines:perp}{लंब} खींचो: \cref{prop:g6:lines:facts} से वह
$d$ के \omterm{def:g6:lines:perp}{समांतर} है।
\end{solution}

\begin{solution}{exo:g6:lines:4}
$a$ और $b$ दोनों एक ही रेखा $c$ पर \omterm{def:g6:lines:perp}{लंब} हैं, इसलिए
$a \parallel b$ (तथ्य 1)। $e$ और $c$ के लिए: दोनों एक ही रेखा
$a$ पर \omterm{def:g6:lines:perp}{लंब} हैं (हमें बताया गया है कि $e \perp a$, और $a \perp c$
का मतलब $c \perp a$ भी है), इसलिए तथ्य 1 फिर लागू होता है: $e \parallel c$।
\end{solution}

$OP > 3$ cm ($P$ बाहर)।
\end{solution}

\end{solution}

\begin{solution}{exo:g6:lines:7}

$\widehat{MON} = 90^\circ$: \omterm{def:g3:shapes:rightangle}{समकोण}। (\omterm{def:g5:solids:def}{शीर्ष} हमेशा बीच वाला

\begin{solution}{exo:g6:lines:8}
नाप बनाए गए त्रिभुज पर निर्भर करते हैं, पर तीनों कोणों का \omterm{def:g1:addition:def}{योगफल}
हमेशा (बहुत क़रीब से) $180^\circ$ होता है --- छोटे \omterm{ex:g1:subtraction:difference}{अंतर} नापने की
कमी से आते हैं। \Cref{ch:g7:triangles} सिद्ध करता है कि \omterm{def:g1:addition:def}{योगफल}
ठीक-ठीक $180^\circ$ है।
\end{solution}

\begin{solution}{exo:g6:lines:9}
मुक्त रचना। बिना चाँदे के $65^\circ$ से $130^\circ$ पाने के लिए:
$65^\circ$ के कोण की दो बार अगल-बगल नक़ल करो ($65 + 65 =
130$), जैसे अनुरेखण काग़ज़ से या परकार-मापपट्टी वाली कोण-नक़ल से।
\end{solution}

\begin{solution}{exo:g6:lines:10}
$A$ से $3$ cm दूर के बिंदु केंद्र $A$ और \omterm{def:g3:shapes:shapes}{त्रिज्या} $3$ cm वाला
वृत्त बनाते हैं; $B$ से $2$ cm दूर के बिंदु केंद्र $B$ और
\omterm{def:g3:shapes:shapes}{त्रिज्या} $2$ cm वाला वृत्त। माँगे गए बिंदु दोनों वृत्तों के
कटान हैं: दो बिंदु अगर वृत्त काटते हों ($AB$ ठीक $1$ और
$5$ cm के बीच), एक अगर वे छूते हों ($AB = 5$ cm या $AB = 1$ cm),
कोई नहीं अगर वे बहुत दूर हों या एक दूसरे के भीतर हो।
\end{solution}

\begin{solution}{exo:g6:lines:11}
$12$ घंटा-निशान पूरे घुमाव को $360 \div 12 = 30^\circ$ के
$12$ कोणों में बाँटते हैं। 3 बजे सुइयाँ $3$ निशान घेरती हैं:
$90^\circ$ (एक \omterm{def:g3:shapes:rightangle}{समकोण})। 5 बजे: $5 \times 30 = 150^\circ$।
6 बजे: $180^\circ$ (एक सरल कोण)।
\end{solution}

\begin{solution}{exo:g6:lines:12}
$A$ और $B$ को केंद्र मानकर $6$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} के दो चाप खींचो; उनका
कटान-बिंदु $C$ है। तीनों भुजाएँ $6$ cm की हैं: त्रिभुज समबाहु
है, और हर कोण $60^\circ$ का है (नाप इसकी पुष्टि करता है)।
अनुमान: समबाहु त्रिभुज के तीनों कोण बराबर, $60^\circ$ के होते
हैं।
\end{solution}

\begin{solution}{pb:g6:lines:1}
\textbf{1.} \omterm{def:g3:division:half}{आधा} घुमाव: $360 \div 2 = 180^\circ$। \omterm{def:g3:division:half}{चौथाई}:
$360 \div 4 = 90^\circ$। बारहवाँ: $360 \div 12 = 30^\circ$।

\textbf{2.} $360^\circ$ भरते बारह बराबर कोण:
पड़ोसी निशानों के बीच $30^\circ$। 3 बजे सुइयाँ $3$ निशान
घेरती हैं: $3 \times 30 = 90^\circ$ --- एक \omterm{def:g3:shapes:rightangle}{समकोण}, जैसा
\cref{exo:g6:lines:11} में मिला था।

\textbf{3.} 1 बजे: $1 \times 30 = 30^\circ$। 4 बजे:
$4 \times 30 = 120^\circ$। 7 बजे सुइयाँ एक ओर $7$ निशान
घेरती हैं, $7 \times 30 = 210^\circ$, इसलिए सुइयों के बीच का
कोण दूसरी ओर वाला है:
$360 - 210 = 150^\circ$।

\textbf{4.} $60$ मिनट में $360^\circ$: प्रति मिनट $360 \div 60 =
6^\circ$।

\textbf{5.} $60$ मिनट में $30^\circ$: प्रति मिनट $30 \div 60 = 0.5^\circ$
--- आधी डिग्री। घंटे की सुई रेंगती है, मिनट की सुई
लंबे डग भरती है।

\textbf{6.} 3:00 के बाद के $30$ मिनटों में घंटे की सुई $3$ से
$30 \times 0.5 = 15^\circ$ आगे बढ़ चुकी है: वह ठीक $3$ और $4$ के
बीचों-बीच है, $12$ से $90 + 15 = 105^\circ$ पर। मिनट की सुई $6$
पर है: $180^\circ$। सुइयों के बीच का कोण:
$180 - 105 = 75^\circ$।

\textbf{7.} गणना अंदाज़े को हरा देती है: मिनट की सुई
$180^\circ$ पर है, पर घंटे की सुई $6$ छोड़ चुकी है --- वह
$6 \times 30 + 30 \times 0.5 = 180 + 15 = 195^\circ$ पर है।
सुइयों के बीच का कोण $195 - 180 = 15^\circ$ है: पास, पर एक के
ऊपर एक नहीं।

\textbf{8.} मिनट की सुई $3$ पर: $90^\circ$। घंटे की सुई:
$9 \times 30 + 15 \times 0.5 = 270 + 7.5 = 277.5^\circ$।
\omterm{ex:g1:subtraction:difference}{अंतर}: $277.5 - 90 = 187.5^\circ$, इसलिए सुइयों के बीच का
कोण $360 - 187.5 = 172.5^\circ$ है --- लगभग एक सरल
कोण।

\textbf{9.} 1:00 पर मिनट की सुई ($0^\circ$) घंटे की सुई
($30^\circ$) से \emph{पीछे} है। 1:10 पर मिनट की सुई
($60^\circ$) उससे \emph{आगे} है
($30 + 10 \times 0.5 = 35^\circ$)। मिनट की सुई तेज़ चलती है
($6^\circ$ प्रति मिनट बनाम $0.5^\circ$), इसलिए 1:00 और 1:10 के
बीच वह घंटे की सुई को पकड़कर पार कर जाती है --- पार करने के उसी
क्षण दोनों सुइयाँ ठीक एक के ऊपर एक होती हैं।

\textbf{10.} यही पकड़ना बार-बार होता है: सुइयाँ ठीक 12:00 पर
मिलती हैं, फिर लगभग 1:05, 2:11, 3:16, 4:22, 5:27, 6:33, 7:38,
8:44, 9:49 और 10:55 पर। 11 और 12 के बीच जिस मुलाक़ात की उम्मीद
हो, वह ठीक 12:00 \emph{पर} पड़ती है, जो अगला चक्र शुरू कर देती
है --- इसलिए बारह घंटों में सुइयाँ $11$ बार मिलती हैं, $12$
बार नहीं। पूरे दिन में: $22$ बार।

\textbf{11.} एक ही \omterm{def:g5:solids:def}{शीर्ष} वाले कोण, जिनमें न ओवरलैप हो न ख़ाली
जगह, उस \omterm{def:g5:solids:def}{शीर्ष} के चारों ओर पूरा घुमाव भर देते हैं: बिंदु के चारों
ओर एक बार घूमने पर हर कोण ठीक एक बार पार होता है, और पूरा घुमाव
$360^\circ$ का है। इसलिए नाप मिलकर $360^\circ$ बनते हैं ---
चाहे कोण कितने भी हों और कितने भी बड़े-छोटे।

\textbf{12.} दिन का हर घंटा
$360 \div 24 = 15^\circ$ का है। तो:
\[
\text{नींद } 9 \times 15 = 135^\circ, \quad
\text{विद्यालय } 90^\circ, \quad
\text{खेल } 45^\circ, \quad
\text{भोजन } 30^\circ, \quad
\text{बाक़ी } 60^\circ,
\]
और $135 + 90 + 45 + 30 + 60 = 360^\circ$: वृत्त पूरा भर जाता
है, न ख़ाली जगह न ओवरलैप (प्रश्न 11)।

\textbf{13.} परकार से एक वृत्त और एक \omterm{def:g3:shapes:shapes}{त्रिज्या} (शुरू की रेखा)
खींचो। चाँदे का केंद्र वृत्त के केंद्र पर रखो, उसकी $0^\circ$
वाली रेखा उस \omterm{def:g3:shapes:shapes}{त्रिज्या} पर, और $135^\circ$ अंकित करो; नई \omterm{def:g3:shapes:shapes}{त्रिज्या}
खींचो: नींद वाला खंड बन गया। फिर $0^\circ$ वाली रेखा \emph{उसी}
नई \omterm{def:g3:shapes:shapes}{त्रिज्या} पर रखकर विद्यालय के लिए $90^\circ$ अंकित करो, और ऐसे
ही आगे --- हर खंड वहीं से शुरू होता है जहाँ पिछला ख़त्म हुआ।
आख़िरी खंड के बाद चित्र ठीक शुरू वाली \omterm{def:g3:shapes:shapes}{त्रिज्या} पर बंद हो जाता है।

\textbf{14.} $90^\circ$, चक्र का $\frac{90}{360} = \frac14$ है
(\cref{met:g6:fractions:simplify}), इसलिए वह दिन की एक \omterm{def:g3:division:half}{चौथाई}
दिखाता है: $24 \div 4 = 6$ घंटे।

\textbf{15.} घंटे की सुई $12$ घंटों में एक घुमाव लेती है: दिन
में $2$ घुमाव। मिनट की सुई हर घंटे एक घुमाव: $24$ घुमाव।
सेकंड की सुई हर मिनट एक घुमाव: दिन में $24 \times 60 = 1\,440$
घुमाव।

\end{solution}
