\section*{Capítulo \ref{ch:g6:measure} --- Perímetro, área e volume}

\begin{solution}{exo:g6:measure:1}
$3.5$ eu $= 350$ cm; \quad $420$ milímetros $= 42$ cm; \quad
$0.8$ quilômetros $= 800$ m; \quad $25\,000$ eu $= 25$ km.
\end{solution}

\begin{solution}{exo:g6:measure:2}
Retângulo: $2 \times (8 + 3.5) = 2 \times 11.5 = 23$ cm.
Quadrado: $4 \times 6.2 = 24.8$ cm.
Triângulo: $5 + 7 + 9 = 21$ cm.
\end{solution}

\begin{solution}{exo:g6:measure:3}
Uma volta: $2\pi \times 50 = 100\pi \approx 314$ m. Para $2$ quilômetros
$= 2000$ eu: $2000 \div 314 \approx 6.4$, então $7$ são necessárias voltas completas
($6$ voltas só fazem cerca de $1\,885$ m).
\end{solution}

\begin{solution}{exo:g6:measure:4}
Retângulo: $9 \times 4 = 36$ cm $^2$. Quadrado: $7^2 = 49$ eu $^2$.
Triângulo retângulo: $\frac{6 \times 10}{2} = 30$ cm $^2$.
Disco: $\pi \times 3^2 = 9\pi \approx 28$ cm $^2$.
\end{solution}

\begin{solution}{exo:g6:measure:5}
$3$ eu $^2 = 30\,000$ cm $^2$; \quad
$45\,000$ cm $^2 = 4.5$ eu $^2$; \quad
$2.5$ eu $= 2\,500$ cm $^3$; \quad
$4\,500$ eu $= 4.5$ eu $^3$.
\end{solution}

\begin{solution}{exo:g6:measure:6}
Cerca (\omterm{def:g6:measure:perimeter}{perímetro}): $2 \times (120 + 85) = 2 \times 205 = 410$ m.
\omterm{def:g6:measure:area}{área}: $120 \times 85 = 10\,200$ eu $^2$.
\end{solution}

\begin{solution}{exo:g6:measure:7}
\omterm{def:g5:solids:def}{Caixa}: $5 \times 4 \times 10 = 200$ cm $^3$. \omterm{def:g5:solids:def}{cubo}:
$3 \times 3 \times 3 = 27$ cm $^3$.
\end{solution}

\begin{solution}{exo:g6:measure:8}
\omterm{def:g6:measure:perimeter}{perímetro} $16$ cm significa comprimento $+$ largura $= 8$ cm. Por exemplo
$6 \times 2$ (\omterm{def:g6:measure:area}{área} $12$ cm $^2$) e $5 \times 3$ (\omterm{def:g6:measure:area}{área} $15$ cm $^2$)
--- ou o quadrado $4 \times 4$ (\omterm{def:g6:measure:area}{área} $16$ cm $^2$). Quanto mais próximo de um
quadrado, maior será o \omterm{def:g6:measure:area}{área}.
\end{solution}

\begin{solution}{exo:g6:measure:9}
\omterm{def:g6:measure:area}{área}: $10 \times 2 + 2 \times 6 = 20 + 12 = 32$ cm $^2$.

\omterm{def:g6:measure:perimeter}{perímetro} (haste centrada sob a barra): andando,
\[
10 + 2 + 4 + 6 + 2 + 6 + 4 + 2 = 36 \text{ cm}
\]
(parte superior; extremidade direita da barra; parte inferior direita; direita da haste; parte inferior da
caule; esquerda do caule; parte inferior esquerda; extremidade esquerda da barra).
\end{solution}

\begin{solution}{exo:g6:measure:10}
\emph{1.} $ V = 25 \times 10 \times 2 = 500$ eu $^3$.

\emph{2.} $500$ eu $^3 = 500 \times 1000 = 500\,000$ eu.

\emph{3.} $500\,000 \div 50\,000 = 10$ horas.
\end{solution}

\begin{solution}{exo:g6:measure:11}
Jardim: $20^2 = 400$ eu $^2$. Lago: $\pi \times 3^2 = 9\pi $ eu$^2$.
Grama: $400 - 9\pi \approx 400 - 28.3 \approx 372$ eu $^2$.
\end{solution}

\begin{solution}{exo:g6:measure:12}
\emph{1.} Barra pequena: $15 \times 6 \times 1 = 90$ cm $^3$. Barra grande:
$30 \times 12 \times 2 = 720$ cm $^3$. O \omterm{def:g6:measure:volume}{volume} é multiplicado por
$720 \div 90 = 8$ --- duplicar cada uma das três dimensões multiplica
o \omterm{def:g6:measure:volume}{volume} por $2 \times 2 \times 2 = 8$.

\emph{2.} Oito vezes o chocolate por quatro vezes o preço: sim, o
uma barra grande é duas vezes melhor por grama.
\end{solution}

\begin{solution}{pb:g6:measure:1}
\textbf{1.} $1 \times 9$: \omterm{def:g6:measure:perimeter}{perímetro} $2 \times (1 + 9) = 20$ m,
\omterm{def:g6:measure:area}{área} $9$ eu $^2$. \ $2 \times 8$: \omterm{def:g6:measure:perimeter}{perímetro} $2 \times 10 = 20$ m,
\omterm{def:g6:measure:area}{área} $16$ eu $^2$.

\textbf{2.} O \omterm{def:g6:measure:perimeter}{perímetro} é duas vezes (comprimento $+$ largura)
(\cref{ex:g6:measure:perimeter}), então comprimento $+$ largura
$= 20 \div 2 = 10$ m. A mesa:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccc}
pen & $1 \times 9$ & $2 \times 8$ & $3 \times 7$ & $4 \times 6$
& $5 \times 5$ \\
\hline
\omterm{def:g6:measure:area}{área} (m $^2$) & $9$ & $16$ & $21$ & $24$ & $25$ \\
\end{tabular}
\end{center}
A praça $5 \times 5$ é o melhor.

\textbf{3.} $4.5 \times 5.5 = 24.75$ e $4.9 \times 5.1 =
24.99$: cada vez mais perto de $25$, mas ainda abaixo. Conjectura:
o quadrado bate \emph{todo} retângulo de \omterm{def:g6:measure:perimeter}{perímetro} $20$,
lados decimais incluídos.

\textbf{4.} Da mesa: o \omterm{def:g2:mult:def}{produto} de dois números com \omterm{def:g1:addition:def}{soma}
$10$ é maior quando os dois números são \emph{igual} ($5$ e
$5$). Regra geral: para um valor fixo \omterm{def:g1:addition:def}{soma}, o \omterm{def:g2:mult:def}{produto} é maior para
partes iguais --- quanto mais desigual for o par, menor será o
\omterm{def:g2:mult:def}{produto}.

\textbf{5.} Do $5 \times 5$ quadrado, remova a \omterm{def:g6:lines:objects}{linha} superior
--- uma tira de $5$ quadrados unitários --- deixando um $5 \times 4$ caneta;
depois cole uma coluna de $4$ quadrados unitários em uma extremidade, fazendo com que
caneta $6 \times 4$. Cinco quadrados foram retirados e apenas quatro
voltou: a faixa adicionada fica ao longo do $4$-lado, que é
mais curto que o $5$-lado da tira removida coberta. \omterm{def:g5:solids:net}{Líquido} perda:
um metro quadrado ($25 \to 24$). O próximo passo, a partir
$6 \times 4$ para $7 \times 3$, negocia uma \omterm{def:g6:lines:objects}{linha} de $6$ a uma coluna
de $3$ e perde mais três ($24 \to 21$): quanto mais longe a caneta
é do quadrado, quanto mais longa for a faixa que ele desiste e mais
mais curto aquele que ele recebe de volta.

\textbf{6.} Com $ w + L + w = 20$, ou seja\ $ L = 20 - 2w $:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccccc}
$ w $ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ & $7$ \\
\hline
$ L $ & $18$ & $16$ & $14$ & $12$ & $10$ & $8$ & $6$ \\
\omterm{def:g6:measure:area}{área} & $18$ & $32$ & $42$ & $48$ & $50$ & $48$ & $42$ \\
\end{tabular}
\end{center}
O vencedor é $5 \times 10$, de \omterm{def:g6:measure:area}{área} $50$ eu $^2$ --- duas vezes mais
longo (ao longo da parede) quanto largo: \emph{não} um quadrado.

\textbf{7.} Reflita a caneta na parede: caneta mais espelho
a imagem forma uma caneta dupla usando a cerca duas vezes, $40$ m de
cerca ao redor (o lado da parede agora é interno). Pela Parte I, o
melhor retângulo com \omterm{def:g6:measure:perimeter}{perímetro} $40$ é o quadrado, $10 \times
10$, de \omterm{def:g6:measure:area}{área} $100$ eu $^2$. A melhor caneta de verdade é \omterm{def:g3:division:half}{metade} disso
melhor caneta dupla: $10$ m ao longo da parede, $5$ estou profundamente, \omterm{def:g6:measure:area}{área}
$50$ eu $^2$ --- exatamente o vencedor da mesa, agora explicado.

\textbf{8.} Circunferência $20 = 2 \times \pi \times r $, então
$ r = 20 \div 6.28 \approx 3.18$ m. \omterm{def:g6:measure:area}{área}:
$\pi r^2 \approx 3.14 \times 3.18 \times 3.18 \approx
31.8$ eu $^2$ --- confortavelmente mais que o da praça $25$ eu $^2$.
Dido sabia o que estava fazendo: para um determinado trecho de fronteira, o
círculo envolve mais.

\textbf{9.} \omterm{def:g6:measure:perimeter}{Perímetros}: $1 \times 36$: $74$ m; \ $2 \times 18$:
$40$ m; \ $3 \times 12$: $30$ m; \ $4 \times 9$: $26$ m; \
$6 \times 6$: $24$ m. A praça novamente --- menos cerca para o
dado \omterm{def:g6:measure:area}{área}. É a Parte I ao contrário: corrigido \omterm{def:g6:measure:perimeter}{perímetro}, o maior
\omterm{def:g6:measure:area}{área}, ou fixo \omterm{def:g6:measure:area}{área}, menor \omterm{def:g6:measure:perimeter}{perímetro}: ambos coroam o quadrado
(e, além de todos os retângulos, o círculo).

\textbf{10.} UM \omterm{def:g4:numbers:round}{redondo} parede é a borda mais curta para o espaço
inclui (questão 8), então um \omterm{def:g4:numbers:round}{redondo} lata, tubo ou tanque contém o
maior conteúdo com menos metal --- material é dinheiro.

\textbf{11.} Por exemplo $50$ eu $\times $ $0.01$ eu: \omterm{def:g6:measure:perimeter}{perímetro}
$2 \times (50 + 0.01) = 100.02$ m, \omterm{def:g6:measure:area}{área}
$50 \times 0.01 = 0.5$ eu $^2$. Longo e fino: quilômetros de
fronteira, quase nenhuma terra.

\textbf{12.} Falso. A caneta da questão 11 tem \omterm{def:g6:measure:perimeter}{perímetro}
$100.02$ eu e \omterm{def:g6:measure:area}{área} $0.5$ eu $^2$, enquanto o $5 \times 5$ caneta tem
\omterm{def:g6:measure:perimeter}{perímetro} $20$ eu e \omterm{def:g6:measure:area}{área} $25$ eu $^2$: maior \omterm{def:g6:measure:perimeter}{perímetro}, longe
menor \omterm{def:g6:measure:area}{área}.

\textbf{13.} \omterm{def:g6:measure:area}{área}: falta um quadrado unitário,
$24 - 1 = 23$ cm $^2$. \omterm{def:g6:measure:perimeter}{perímetro}: andando por aí, o entalhe
substitui $1$ cm de parede reta por três lados do pequeno
quadrado, $1 + 1 + 1 = 3$ centímetros: o \omterm{def:g6:measure:perimeter}{perímetro} cresce de $20$ centímetros para
$20 - 1 + 3 = 22$ cm. Removendo material \emph{alongado} o
fronteira.

\textbf{14.} Cada zoom numa costa real revela novas baías, rochas
e riachos --- entalhes após entalhes, e a pergunta 13 mostra que
cada entalhe adiciona comprimento de borda, alterando apenas o \omterm{def:g6:measure:area}{área}.
Portanto, o comprimento medido continua crescendo com o nível de detalhe
(“o paradoxo do litoral”), enquanto a medida \omterm{def:g6:measure:area}{área} resolve
para baixo: \omterm{def:g6:measure:perimeter}{perímetro} e \omterm{def:g6:measure:area}{área} são quantidades verdadeiramente independentes.

\textbf{15.} Argumento do espelho: a caneta dupla usaria
$2 \times 36 = 72$ m de cerca, e o melhor retângulo de
\omterm{def:g6:measure:perimeter}{perímetro} $72$ é o $18 \times 18$ quadrado. Então a melhor caneta é
$18$ m ao longo da parede e $9$ m de profundidade: verificação da cerca
$9 + 18 + 9 = 36$ m, \omterm{def:g6:measure:area}{área} $18 \times 9 = 162$ eu $^2$. Em aberto
campo, o melhor é o $9 \times 9$ quadrado, \omterm{def:g6:measure:area}{área} $81$ eu $^2$.
A parede do celeiro exatamente \emph{dobra} a terra do agricultor.
\end{solution}
