\section*{Capítulo \ref{ch:g6:fractions} --- Frações: primeiros passos}

\begin{solution}{exo:g6:fractions:1}
Bolo: $\frac38$ comido. \quad Chocolate: $\frac{7}{12}$. \quad
Torta: $\frac66 = 1$ (a torta inteira).
\end{solution}

\begin{solution}{exo:g6:fractions:2}
Quatro partes em cinco ficam pintadas. $\frac45 < 1$ (quatro partes de
cinco é menos que a barra inteira); $\frac65 > 1$ (seis quintos são uma
barra inteira e mais um quinto).
\end{solution}

\begin{solution}{exo:g6:fractions:3}
Na reta graduada em terços: $\frac13$ fica um traço depois do $0$;
$\frac53$ fica um traço antes do $2$ (pois $2 = \frac63$); $\frac73$
fica um traço depois do $2$. Logo $\frac53$ e $\frac73$ estão os dois à
distância $\frac13$ do $2$.
\end{solution}

\begin{solution}{exo:g6:fractions:4}
$\frac94 = 2 + \frac14$; \quad
$\frac{13}{5} = 2 + \frac35$; \quad
$\frac{12}{6} = 2$ (exatamente); \quad
$\frac{25}{8} = 3 + \frac18$.
\end{solution}

\begin{solution}{exo:g6:fractions:5}
$\frac23 = \frac{8}{12}$ (multiplique por $4$); \quad
$\frac{15}{20} = \frac34$ (divida por $5$); \quad
$\frac{7}{10} = \frac{21}{30}$ (multiplique por $3$); \quad
$\frac56 = \frac{10}{12}$ (divida por $2$).
\end{solution}

\begin{solution}{exo:g6:fractions:6}
$\frac68 = \frac34$; \quad
$\frac{15}{25} = \frac35$; \quad
$\frac{18}{24} = \frac{9}{12} = \frac34$; \quad
$\frac{35}{7} = 5$ (um número inteiro).
\end{solution}

\begin{solution}{exo:g6:fractions:7}
$\frac12$ de $86$: $86 \div 2 = 43$.

$\frac34$ de $60$: $60 \div 4 = 15$, depois $15 \times 3 = 45$.

$\frac25$ de $35$: $35 \div 5 = 7$, depois $7 \times 2 = 14$.

$\frac{7}{10}$ de $250$: $250 \div 10 = 25$, depois
$25 \times 7 = 175$.
\end{solution}

\begin{solution}{exo:g6:fractions:8}
$\frac12 > \frac13$: as \omterm{def:g3:division:half}{metades} são partes maiores que os terços.
$\frac35 > \frac25$: as mesmas partes, em maior número.
$\frac34 = \frac68 < \frac78$: sete oitavos vencem seis oitavos.
\end{solution}

\begin{solution}{exo:g6:fractions:9}
$\frac12 = 0.5$; \quad $\frac34 = 0.75$; \quad
$\frac{7}{10} = 0.7$; \quad $\frac94 = 2.25$. A exceção é
$\frac13 = 0.333\dots$, cuja escrita decimal nunca acaba: ela não é um
\omterm{def:g6:decimals:places}{número decimal}.
\end{solution}

\begin{solution}{exo:g6:fractions:10}
Quem usa ônibus: $\frac34$ de $28$: $28 \div 4 = 7$, depois
$7 \times 3 = 21$ alunos. A pé: $28 - 21 = 7$ alunos (é o quarto que
sobra: $28 \div 4 = 7$). Confere: $21 + 7 = 28$.
\end{solution}

\begin{solution}{exo:g6:fractions:11}
Dois terços das economias são $18$ reais, então \emph{um} terço é
$18 \div 2 = 9$ reais, e as economias inteiras (três terços) eram
$9 \times 3 = 27$ reais. Confere: $\frac23$ de $27$ é $18$.
\end{solution}

\begin{solution}{exo:g6:fractions:12}
A água usada levou o tanque de $\frac58$ para $\frac48$ (a \omterm{def:g3:division:half}{metade} são
quatro oitavos): os $10$ litros representam
$\frac58 - \frac48 = \frac18$ da \omterm{def:g2:measure:units}{capacidade}. Logo o tanque cheio leva
$10 \times 8 = 80$ litros. Confere: $\frac58$ de $80$ é $50$; menos
$10$ ficam $40$, que é de fato a \omterm{def:g3:division:half}{metade} de $80$.
\end{solution}

\begin{solution}{pb:g6:fractions:1}
\textbf{1.} A \omterm{def:g3:division:half}{metade} de $16$ quadradinhos são $8$ quadradinhos. Como
\omterm{def:g6:fractions:def}{fração}: $\frac12$ da barra e, como cada \omterm{def:g3:division:half}{metade} são $8$ quadradinhos em
$16$, também $\frac{8}{16}$ --- a mesma \omterm{def:g6:fractions:def}{fração} escrita de duas maneiras
(\cref{prop:g6:fractions:equal}).

\textbf{2.} A \omterm{def:g3:division:half}{metade} dos $8$ quadradinhos restantes são $4$
quadradinhos, ou seja, $\frac{4}{16}$ da barra. Simplificando por $4$:
$\frac{4}{16} = \frac14$. No desenho: cortar ao meio a \omterm{def:g3:division:half}{metade} restante
produz dois quartos da barra original --- \omterm{def:g3:division:half}{metade} de uma \omterm{def:g3:division:half}{metade} é um
quarto.

\textbf{3.} Terceira mordida: \omterm{def:g3:division:half}{metade} de $4$ quadradinhos são $2$
quadradinhos, isto é, $\frac{2}{16} = \frac18$ da barra. Quarta
mordida: $1$ quadradinho, isto é, $\frac{1}{16}$. De uma mordida para
a seguinte, o \omterm{def:g4:fractions:def}{denominador} \emph{dobra}: $2$, $4$, $8$, $16$.

\textbf{4.} Comidos: $8 + 4 + 2 + 1 = 15$ quadradinhos em $16$, ou
seja, $\frac{15}{16}$ da barra. Sobra: $1$ quadradinho,
$\frac{1}{16}$.

\textbf{5.} $\frac12 = \frac{8}{16}$ (multiplique em cima e embaixo por
$8$), $\frac14 = \frac{4}{16}$ (por $4$), $\frac18 = \frac{2}{16}$ (por
$2$), e $\frac{1}{16}$ fica como está. Contando dezesseis avos:
$8 + 4 + 2 + 1 = 15$ deles, então o total é $\frac{15}{16}$.

\textbf{6.} Mordidas: $32$, $16$, $8$, $4$, $2$, $1$ quadradinhos.
Total comido: $32 + 16 + 8 + 4 + 2 + 1 = 63$ quadradinhos, então sobra
$1$ quadradinho: $\frac{1}{64}$ da barra.

\textbf{7.} Depois de cada mordida sobra
\[
\frac12,\quad \frac14,\quad \frac18,\quad \frac{1}{16},\quad
\frac{1}{32},\quad \frac{1}{64} :
\]
o que sobra cai pela \omterm{def:g3:division:half}{metade} a cada vez --- o \omterm{def:g4:fractions:def}{denominador} dele dobra.
Uma sétima mordida imaginária deixaria $\frac{1}{128}$.

\textbf{8.} Cada mordida come apenas \emph{\omterm{def:g3:division:half}{metade}} do que sobrou, então
a outra \omterm{def:g3:division:half}{metade} do \omterm{def:g3:division:remainder}{resto} continua ali: depois de qualquer quantidade de
mordidas, sempre sobra algo. O total comido é, portanto, sempre menor
que $1$ --- exatamente como os vinte noves do
\cref{pb:g6:decimals:1}, que chegam cada vez mais perto de $3$ sem
alcançá-lo.

\textbf{9.} As duas frações têm \omterm{def:g4:fractions:def}{numerador} $1$, e cortar em $128$ partes
produz partes menores que cortar em $100$: logo
$\frac{1}{128} < \frac{1}{100}$. Depois da sétima mordida sobra
$\frac{1}{128}$, então a parte comida ultrapassa
$1 - \frac{1}{100} = \frac{99}{100}$: o desafio é ganho na sétima
mordida.

\textbf{10.} Em dezesseis avos, os totais são $\frac{8}{16}$,
$\frac{12}{16}$, $\frac{14}{16}$, $\frac{15}{16}$. Cada novo total cai
exatamente \emph{no meio} entre o total anterior e o $1$: a escada
continua reduzindo à \omterm{def:g3:division:half}{metade} a distância que falta até $1$, sem nunca
pisar nele.

\textbf{11.} O que está certo: em toda etapa da descrição de Zenão, de
fato ainda falta alguma distância (questão 8) --- a lista de etapas
nunca acaba. A armadilha: as etapas ficam extremamente curtas, em
distância \emph{e} em tempo de corrida; o corredor não gasta o mesmo
tempo em cada etapa. Descrever a corrida em infinitos pedaços cada vez
menores não torna a corrida em si interminável: o corredor chega ao
muro, e a questão 9 mostra a descrição ultrapassando qualquer ponto
aquém dele.

\textbf{12.} Corte $8$ barras ao meio: $16$ meias barras. Corte $4$
barras em quartos: $16$ pedaços de um quarto. Corte $2$ barras em
oitavos: $16$ pedaços de um oitavo. Corte a última barra em dezesseis
avos: $16$ pedacinhos. Isso usa $8 + 4 + 2 + 1 = 15$ barras, e cada
criança recebe um pedaço de cada tamanho.

\textbf{13.} A parte de cada criança é
\[
\frac12 + \frac14 + \frac18 + \frac{1}{16} = \frac{15}{16}
\]
de barra (questão 5). Dezesseis partes dessas dão
$16 \times \frac{15}{16} = 15$ barras
(\cref{thm:g6:fractions:quotient}): exatamente o que foi cortado, sem
sobra --- uma partilha justa de $15$ barras entre $16$ crianças, só com
cortes ao meio.

\textbf{14.} Os vizinhos recebem $\frac78$ de barra cada um. Com
\omterm{def:g4:fractions:def}{denominador} $16$: $\frac78 = \frac{14}{16}$, enquanto as nossas
crianças recebem $\frac{15}{16}$. Logo uma criança das $16$ recebe
mais: $\frac{15}{16} > \frac{14}{16}$, por um dezesseis avos de barra.

\textbf{15.} No desenho do quadrado, cada mordida pintada preenche
\omterm{def:g3:division:half}{metade} do canto que ainda está sem pintar, e esse canto continua
encolhendo: depois de cada mordida ele fica com \omterm{def:g3:division:half}{metade} do tamanho
anterior, e nunca desaparece. Os dois fatos verdadeiros: (i) depois de
qualquer quantidade finita de mordidas sobra um \omterm{def:g3:division:remainder}{resto} --- o total
comido é sempre estritamente menor que $1$; (ii) esse \omterm{def:g3:division:remainder}{resto} fica menor
que qualquer \omterm{def:g6:fractions:def}{fração} que se queira dizer ($\frac{1}{100}$,
$\frac{1}{1000}$, \dots), bastando morder por tempo suficiente. Dar às
palavras ``a \omterm{def:g1:addition:def}{soma} infinita vale $1$'' o seu sentido exato é tarefa dos
\emph{limites}, no volume do ensino médio.
\end{solution}
