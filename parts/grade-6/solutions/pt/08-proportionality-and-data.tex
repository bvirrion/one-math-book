\section*{Capítulo \ref{ch:g6:propdata} --- Proporcionalidade e dados}

\begin{solution}{exo:g6:propdata:1}
Maçãs: \omterm{def:g3:division:remainder}{quocientes} $\frac52 = 2.5$, $\frac{7.5}{3} = 2.5$,
$\frac{12.5}{5} = 2.5$ --- todos iguais: proporcional (preço de
$2.50$ por quilo).

Idade/altura: $\frac{85}{2} = 42.5$, mas $\frac{103}{4} = 25.75$ ---
não são iguais: não é proporcional.
\end{solution}

\begin{solution}{exo:g6:propdata:2}
Um croissant: $4.80 \div 4 = 1.20$ reais. Sete:
$7 \times 1.20 = 8.40$ reais.
\end{solution}

\begin{solution}{exo:g6:propdata:3}
Coeficiente: $18 \div 10 = 1.8$ reais por litro. Então $25$ L custam
$25 \times 1.8 = 45$ reais; $72$ reais compram $72 \div 1.8 = 40$ L;
$60$ L custam $60 \times 1.8 = 108$ reais.
\end{solution}

\begin{solution}{exo:g6:propdata:4}
Para $250$ km: $2.5$ vezes o combustível de $100$ km, ou seja,
$2.5 \times 6 = 15$ L. Para $350$ km: $3.5 \times 6 = 21$ L. Com
$27$ L: $27 \div 6 = 4.5$ centenas de km, isto é, $450$ km.
\end{solution}

\begin{solution}{exo:g6:propdata:5}
$50\,\%$ de $84$: a \omterm{def:g3:division:half}{metade}, $42$. \quad
$25\,\%$ de $200$: um quarto, $50$. \quad
$10\,\%$ de $63$: um décimo, $6.3$. \quad
$30\,\%$ de $70$: $3 \times 7 = 21$.
\end{solution}

\begin{solution}{exo:g6:propdata:6}
$60\,\%$ de $450$: $\frac{60}{100} \times 450 = 270$ alunos almoçam no
refeitório; $450 - 270 = 180$ não almoçam.
\end{solution}

\begin{solution}{exo:g6:propdata:7}
Menos de $10$ livros: terça ($8$) e quinta ($6$). Quarta menos quinta:
$17 - 6 = 11$ livros a mais.
\end{solution}

\begin{solution}{exo:g6:propdata:8}
Gráfico de barras com cinco barras de alturas $14$, $16$, $13$, $17$,
$20$ (graduação de $2$ em $2$ ou de $5$ em $5$ graus funciona bem). Dia
mais quente: sexta ($20$ graus).
\end{solution}

\begin{solution}{exo:g6:propdata:9}
Para $10$ pessoas, multiplique por $\frac{10}{6} = \frac53$: farinha,
$450 \times \frac{10}{6} = 750$ g. Ovos:
$3 \times \frac{10}{6} = 5$ --- felizmente um número inteiro. (Se não
fosse, arredondaríamos \emph{para cima}, para ter o suficiente.)
\end{solution}

\begin{solution}{exo:g6:propdata:10}
\emph{1.} Em $2$ h: $48$ km. Em meia hora: $12$ km. Em $1$ h $30$:
$24 + 12 = 36$ km.

\emph{2.} $60 \div 24 = 2.5$ horas, isto é, $2$ h $30$ min.
\end{solution}

\begin{solution}{exo:g6:propdata:11}
Bola: desconto de $20\,\%$ de $15 = 3$ reais, preço novo $12$ reais.
Raquete: desconto de $8$ reais, preço novo $32$ reais. O preço novo é
$80\,\%$ do antigo em todos os casos: proporcional, com coeficiente
$0.8$.
\end{solution}

\begin{solution}{exo:g6:propdata:12}
Depois de $t$ horas, a vela grossa queimou $\frac t6$ da altura dela:
ela está em $1 - \frac t6$ da altura inicial; a fina está em
$1 - \frac t4$. Queremos
\[
1 - \frac t4 = \frac12 \left(1 - \frac t6\right)
\quad\text{isto é}\quad
1 - \frac t4 = \frac12 - \frac t{12}.
\]
Então $\frac12 = \frac t4 - \frac t{12} = \frac{3t - t}{12} =
\frac{t}{6}$, logo $t = 3$ horas. Confere: depois de $3$ h a vela
grossa está em $1 - \frac36 = \frac12$ e a fina em
$1 - \frac34 = \frac14$ --- de fato a \omterm{def:g3:division:half}{metade}.
\end{solution}

\begin{solution}{pb:g6:propdata:1}
\textbf{1.} $100\,000$ cm $= 1\,000$ m $= 1$ km: um centímetro no mapa
é um quilômetro no terreno.

\textbf{2.} $7.5$ cm $\rightarrow 7.5$ km entre os vilarejos;
$12$ cm $\rightarrow 12$ km de margem do lago.

\textbf{3.} $20$ km $\rightarrow 20$ cm no mapa.

\textbf{4.} $5$ km $= 500\,000$ cm, então a escala é $1:500\,000$. O
mapa de caminhada ($1:100\,000$) é o mais detalhado: ele usa $5$ cm de
papel onde o outro gasta apenas $1$ cm.

\textbf{5.}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccc}
mapa (cm) & $1$ & $2$ & $7.5$ \\
\hline
real (km) & $1$ & $2$ & $7.5$ \\
\end{tabular}
\end{center}
Distância real $=$ distância no mapa $\times 1$ (nessas unidades):
sempre o \emph{mesmo} multiplicador, que é exatamente a definição de
\omterm{def:g6:propdata:def}{proporcionalidade} (\cref{def:g6:propdata:def}). O coeficiente aqui é
$1$ quilômetro por centímetro.

\textbf{6.} $4$ m $= 400$ cm e $400 \div 50 = 8$; $3$ m $= 300$ cm e
$300 \div 50 = 6$: a planta mostra um retângulo de
$8$ cm $\times$ $6$ cm.

\textbf{7.} Cama: $200 \div 50 = 4$ cm por $90 \div 50 = 1.8$ cm. Vão
da porta: $80 \div 50 = 1.6$ cm.

\textbf{8.} \omterm{def:g6:measure:area}{Área} real: $4 \times 3 = 12$ m$^2$. \omterm{def:g6:measure:area}{Área} da planta:
$8 \times 6 = 48$ cm$^2$. Convertendo:
$12$ m$^2 = 12 \times 10\,000 = 120\,000$ cm$^2$, e
\[
120\,000 \div 48 = 2\,500 .
\]
O \omterm{def:g3:division:remainder}{quociente} não é $50$, e sim $2\,500 = 50 \times 50$.

\textbf{9.} Um cm$^2$ de papel representa um quadrado real de $50$ cm
por $50$ cm, que contém $50 \times 50 = 2\,500$ cm$^2$. Então, quando
os comprimentos são divididos por $50$, as \omterm{def:g6:measure:area}{áreas} são divididas por
$50 \times 50 = 2\,500$: as \omterm{def:g6:measure:area}{áreas} seguem o \emph{quadrado} da escala.

\textbf{10.} $6 \times 2\,500 = 15\,000$ cm$^2$, e
$15\,000 \div 10\,000 = 1.5$ m$^2$.

\textbf{11.} A imagem sugere que o domingo vendeu o \emph{dobro} (uma
barra duas vezes mais alta). Verdade: o aumento é de $5$ reais sobre
$105$, e $5 \div 105 \approx 0.048$: cerca de $5\,\%$ a mais, e não
$100\,\%$ a mais. O gerente ignorou o alerta de conferir se o eixo
começa em $0$ (\cref{met:g6:propdata:read}).

\textbf{12.} Com o eixo a partir de $0$, as barras sobem até $105$ e
$110$ unidadezinhas: duas barras de altura quase igual, a imagem
honesta de uma \omterm{ex:g1:subtraction:difference}{diferença} de $5\,\%$.

\textbf{13.} Para cima: o aumento é de $20$ selos a partir de $40$:
$\frac{20}{40} = 50\,\%$. Para baixo: a queda é de $20$ selos a partir
de $60$: $\frac{20}{60} = \frac13 \approx 33\,\%$. Os mesmos $20$
selos, valores iniciais diferentes --- uma porcentagem é sempre uma
\omterm{def:g6:fractions:def}{fração} \emph{de alguma coisa}, e essa ``alguma coisa'' mudou.

\textbf{14.} Depois dos $20\,\%$ de desconto: $100 - 20 = 80$ reais. Os
$10\,\%$ extras incidem sobre $80$: desconto de $8$ reais, preço final
$72$ reais. Desconto total: $28$ reais em $100$, ou seja, $28\,\%$ ---
e não $30\,\%$. O segundo desconto agiu sobre o preço já reduzido,
então os reais dele são menores: porcentagens de quantidades diferentes
não se somam.

\textbf{15.} Nesse mapa, $1$ cm representa $5$ km, então $1$ cm$^2$
representa $5 \times 5 = 25$ km$^2$ (a regra da questão 9). A floresta:
$3 \times 25 = 75$ km$^2$. Os comprimentos seguem a escala; as \omterm{def:g6:measure:area}{áreas}
seguem o \emph{quadrado} dela.
\end{solution}
