\begin{solution}{exo:g6:wholes:2}

\[
47\,209 = 4 \times 10\,000 + 7 \times 1000 + 2 \times 100 + 0 \times 10

\]
\end{solution}

\begin{solution}{exo:g6:wholes:4}
Por conteo de dígitos, $978$ y $1\,001$ ven antes de los tres
números de cuatro dígitos que comienzan con $2$; entre ellos, compare las decenas y
unidades: $2\,034 < 2\,043 < 2\,304$. Orden final:
\[
978 < 1\,001 < 2\,034 < 2\,043 < 2\,304 .
\]
\end{solution}

\begin{solution}{exo:g6:wholes:6}
$38 + 45 + 12 + 5 = (38 + 12) + (45 + 5) = 50 + 50 = 100$.

$2 \times 83 \times 50 = (2 \times 50) \times 83 = 100 \times 83 =
8\,300$.

$4 \times 78 \times 25 = (4 \times 25) \times 78 = 100 \times 78 =

\end{solution}

\begin{solution}{exo:g6:wholes:7}

\begin{solution}{exo:g6:wholes:8}
La igualdad es cierta, pero la \omterm{def:g3:division:remainder}{resto} $13$ es \emph{no} más pequeño que
el \omterm{def:g5:division:multiple}{divisor} $8$, por lo que no es la división euclidiana. toma uno mas
$8$ fuera del \omterm{def:g3:division:remainder}{resto}: $173 = 8 \times 21 + 5$, con $0 \leq 5 < 8$:
\omterm{def:g3:division:remainder}{cociente} $21$, \omterm{def:g3:division:remainder}{resto} $5$.
\end{solution}

\begin{solution}{exo:g6:wholes:10}

\begin{solution}{exo:g6:wholes:11}
el mas grande posible \omterm{def:g3:division:remainder}{resto} en una división por $9$ es $8$ (el
\omterm{def:g3:division:remainder}{resto} debe ser menor que el \omterm{def:g5:division:multiple}{divisor}). Entonces el dividendo es
\[
9 \times 56 + 8 = 504 + 8 = 512 .
\]
\end{solution}

\begin{solution}{exo:g6:wholes:12}
Sean los dígitos $ h $ (cientos), $0$ (decenas), $ u $ (unidades). sabemos
$ h = 2 \times u $ y $ h + 0 + u = 9$, entonces $2u + u = 9$, donación
$3u = 9$, $ u = 3$, y $ h = 6$. el número es $603$. Controlar: $6$ es dos veces $3$, el dígito de las decenas es $0$, y $6 + 0 + 3 = 9$.
\end{solution}

\begin{solution}{pb:g6:wholes:1}
\textbf{1.} Por ejemplo $(1 + 4) + (2 + 3) + 5 = 5 + 5 + 5 =
15$.

\textbf{2.} Cada par vale $11$: $1 + 10$, $2 + 9$, $3 + 8$,
$4 + 7$, $5 + 6$. Eso es $5$ pares (diez números, dos por par),
entonces el \omterm{def:g1:addition:def}{suma} es $5 \times 11 = 55$.

\textbf{3.} Los pares valen $21$ ($1 + 20$, $2 + 19, \dots $), y hay $10$ de ellos: $10 \times 21 = 210$.

\textbf{4.} En cada par, cuando el primer número crece en $1$ el
segundo se reduce por $1$, por lo que el total nunca cambia:
$1 + 100 = 2 + 99 = 3 + 98 = \dots = 101$. los cien números
hacer $50$ pares (dos números cada uno), siendo el más interno
$50 + 51 = 101$.

\textbf{5.} $50$ pares de $101$:
\[
1 + 2 + \dots + 100 = 50 \times 101 = 5\,050 .
\]

\textbf{6.} Con nueve números, el del medio, $5$, no tiene
compañero. A su alrededor: $1 + 9 = 2 + 8 = 3 + 7 = 4 + 6 = 10$, cuatro
pares. Total: $4 \times 10 + 5 = 45$.

\textbf{7.} Cada columna vale $1 + 9 = 2 + 8 = \dots =
9 + 1 = 10$, y hay $9$ columnas: las dos líneas juntas
valen la pena $9 \times 10 = 90$. Pero las dos líneas son las
\emph{mismo} \omterm{def:g1:addition:def}{suma} escrito dos veces, por lo que una copia vale \omterm{def:g3:division:half}{medio}:
$1 + 2 + \dots + 9 = 90 \div 2 = 45$. Nunca queda ningún número
sin pareja --- su pareja está justo debajo de él.

\textbf{8.} Para $1$ a $50$: $50$ columnas que valen la pena $51$, doble \omterm{def:g1:addition:def}{suma}
$50 \times 51 = 2\,550$, entonces el \omterm{def:g1:addition:def}{suma} es $2\,550 \div 2 = 1\,275$.
Para $1$ a $200$: $200$ columnas que valen la pena $201$, doble \omterm{def:g1:addition:def}{suma}
$40\,200$, \omterm{def:g1:addition:def}{suma} $20\,100$.

\textbf{9.} Cada \omterm{def:g3:numbers:evenodd}{número par} es el doble de un número de
de gauss \omterm{def:g1:addition:def}{suma}: $2 = 2 \times 1$, $4 = 2 \times 2$, \dots,
$200 = 2 \times 100$. Así que todo \omterm{def:g1:addition:def}{suma} es el doble de Gauss:
\[
2 + 4 + \dots + 200 = 2 \times 5\,050 = 10\,100 .
\]

\textbf{10.} Los números de $101$ a $200$ son los números de
$1$ a $200$ con los números de $1$ a $100$ remoto:
\[
101 + 102 + \dots + 200 = 20\,100 - 5\,050 = 15\,050 .
\]

\textbf{11.} Alinea a los amigos. Los primeros temblores $7$ manos (una
con cada uno de los demás). El segundo ya ha saludado al
primero, entonces él tiembla $6$ \emph{nuevo} manos; el tercero, $5$; y así
adelante, el séptimo temblor $1$ y el octavo ninguno que sea nuevo.
Cada apretón de manos se cuenta exactamente una vez:
\[
7 + 6 + 5 + 4 + 3 + 2 + 1 = 28 \text{ handshakes}
\]
(par alrededor del medio: $(7+1) + (6+2) + (5+3) + 4 =
8 + 8 + 8 + 4 = 28$).

\textbf{12.} $1 + 2 + \dots + 12$: valor de doce columnas $13$, doble \omterm{def:g1:addition:def}{suma} $12 \times 13 = 156$, entonces
$156 \div 2 = 78$ latas.

\textbf{13.} la escalera de $1 + 2 + 3 + 4$ cuadrados, más el
misma escalera al revés, llena un rectángulo de $4$ filas y $5$
columnas exactamente: cada fila del rectángulo es una columna del
truco de dos líneas (un escalón de la escalera ascendente completado por un
paso del que cae, $1 + 4$, $2 + 3$, $3 + 2$, $4 + 1$ ---
siempre $5$). Entonces el doble de \omterm{def:g1:addition:def}{suma} es $4 \times 5 = 20$ cuadrados, y
$1 + 2 + 3 + 4 = 10$: la imagen \emph{es} pregunta 7.

\textbf{14.} $1 + 2 + \dots + 12 = 78$ huelgas en doce horas
(pregunta 12), por lo tanto $78 \times 2 = 156$ huelgas en un día entero.

\textbf{15.} dividir por $60$: $5\,050 = 60 \times 84 + 10$, entonces
$5\,050$ segundos es $84$ minutos y $10$ artículos de segunda clase. dividir el
minutos por $60$: $84 = 60 \times 1 + 24$, entonces $84$ minutos es
$1$ hora $24$ minutos. En total: $1$ h $24$ mín. $10$ s --- un poco
\emph{menos} más de una hora y un \omterm{def:g3:division:half}{medio} ($1$ h $30$ mín.).
\end{solution}
