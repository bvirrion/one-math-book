\section*{Capítulo \ref{ch:g6:propdata} --- Proporcionalidad y datos}

\begin{solution}{exo:g6:propdata:1}
Manzanas: \omterm{def:g3:division:remainder}{cocientes} $\frac52 = 2.5$, $\frac{7.5}{3} = 2.5$,
$\frac{12.5}{5} = 2.5$ --- todos iguales: \omterm{def:g6:propdata:def}{proporcional} (precio
$2.50$ por kilogramo).

Edad/altura: $\frac{85}{2} = 42.5$ pero $\frac{103}{4} = 25.75$ --- no
igual: no \omterm{def:g6:propdata:def}{proporcional}.
\end{solution}

\begin{solution}{exo:g6:propdata:2}
Un croissant: $4.80 \div 4 = 1.20$ euros. Siete:
$7 \times 1.20 = 8.40$ euros.
\end{solution}

\begin{solution}{exo:g6:propdata:3}
Coeficiente: $18 \div 10 = 1.8$ euros por litro. Entonces
$25$ costo l $25 \times 1.8 = 45$ euros; $72$ comprar euros
$72 \div 1.8 = 40$ l; $60$ costo l $60 \times 1.8 = 108$ euros.
\end{solution}

\begin{solution}{exo:g6:propdata:4}
Para $250$ kilómetros: $2.5$ veces el combustible de $100$ kilómetros, entonces
$2.5 \times 6 = 15$ L para $350$ kilómetros: $3.5 \times 6 = 21$ L con
$27$ L: $27 \div 6 = 4.5$ cientos de kilómetros, es decir\ $450$ km.
\end{solution}

\begin{solution}{exo:g6:propdata:5}
$50\,\%$ de $84$: \omterm{def:g3:division:half}{medio}, $42$. \quad
$25\,\%$ de $200$: a \omterm{def:g3:division:half}{cuarto}, $50$. \quad
$10\,\%$ de $63$: una décima parte, $6.3$. \quad
$30\,\%$ de $70$: $3 \times 7 = 21$.
\end{solution}

\begin{solution}{exo:g6:propdata:6}
$60\,\%$ de $450$: $\frac{60}{100} \times 450 = 270$ los estudiantes comen en
la cafetería; $450 - 270 = 180$ no.
\end{solution}

\begin{solution}{exo:g6:propdata:7}

\end{solution}

\begin{solution}{exo:g6:propdata:8}
Gráfico de barras con cinco barras de alturas. $14$, $16$, $13$, $17$, $20$

\end{solution}

\begin{solution}{exo:g6:propdata:9}
Para $10$ personas, multiplicar por $\frac{10}{6} = \frac53$: harina
$450 \times \frac{10}{6} = 750$ gramo. Huevos:
$3 \times \frac{10}{6} = 5$ --- por suerte un número entero. (Si fuera
no, uno lo haría \omterm{def:g4:numbers:round}{redondo} \emph{arriba} tener suficiente.)
\end{solution}

\begin{solution}{exo:g6:propdata:10}
\emph{1.} En $2$ h: $48$ km. En \omterm{def:g3:division:half}{medio} una hora: $12$ km. En
$1$ h $30$: $24 + 12 = 36$ km.

\emph{2.} $60 \div 24 = 2.5$ horas, es decir\ $2$ h $30$ mín.
\end{solution}

\begin{solution}{exo:g6:propdata:11}
Bola: descuento $20\,\%$ de $15 = 3$ euros, nuevo precio $12$ euros.
Raqueta: descuento $8$ euros, nuevo precio $32$ euros. El nuevo precio es
$80\,\%$ del antiguo en cada caso: \omterm{def:g6:propdata:def}{proporcional}, coeficiente
$0.8$.
\end{solution}

\begin{solution}{exo:g6:propdata:12}
Después $ t $ horas, la vela espesa ha ardido $\frac t6$ de su altura:
se encuentra en $1 - \frac t6$ de la altura inicial; el delgado en
$1 - \frac t4$. queremos
\[
1 - \frac t4 = \frac12 \left(1 - \frac t6\right)
\quad\text{i.e.}\quad
1 - \frac t4 = \frac12 - \frac t{12}.
\]
Entonces $\frac12 = \frac t4 - \frac t{12} = \frac{3t - t}{12} =
\frac{t}{6}$, entonces $ t = 3$ horas. comprobar: después $3$ h la vela gruesa
se encuentra en $1 - \frac36 = \frac12$ y el delgado en
$1 - \frac34 = \frac14$ --- en efecto \omterm{def:g3:division:half}{medio} de ello.
\end{solution}

\begin{solution}{pb:g6:propdata:1}
\textbf{1.} $100\,000$ centímetro $= 1\,000$ metro $= 1$ km: un centímetro
en el mapa hay un kilómetro sobre el terreno.

\textbf{2.} $7.5$ centímetro $\rightarrow 7.5$ km entre los pueblos;
$12$ centímetro $\rightarrow 12$ km de orilla del lago.

\textbf{3.} $20$ kilómetros $\rightarrow 20$ cm en el mapa.

\textbf{4.} $5$ kilómetros $= 500\,000$ cm, entonces la escala es
$1:500\,000$. El mapa de senderismo ($1:100\,000$) es más
detallado: utiliza $5$ cm de papel donde el otro gasta solo
$1$ centímetro.

\textbf{5.}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccc}
map (cm) & $1$ & $2$ & $7.5$ \\
\hline
real (km) & $1$ & $2$ & $7.5$ \\
\end{tabular}
\end{center}
distancia real $=$ distancia del mapa $\times 1$ (en estas unidades):
siempre el \emph{mismo} multiplicador, que es exactamente el
definición de \omterm{def:g6:propdata:def}{proporcionalidad}
(\cref{def:g6:propdata:def}). El coeficiente aquí es $1$
kilómetro por centímetro.

\textbf{6.} $4$ metro $= 400$ centímetros y $400 \div 50 = 8$;
$3$ metro $= 300$ centímetros y $300 \div 50 = 6$: el plano muestra una
$8$ centímetro $\times $ $6$ cm rectángulo.

\textbf{7.} Cama: $200 \div 50 = 4$ centímetros por $90 \div 50 = 1.8$ centímetro.
Puerta: $80 \div 50 = 1.6$ centímetro.

\textbf{8.} Real \omterm{def:g6:measure:area}{área}: $4 \times 3 = 12$ metro $^2$. Plan \omterm{def:g6:measure:area}{área}:
$8 \times 6 = 48$ centímetro $^2$. Mudado:
$12$ metro $^2 = 12 \times 10\,000 = 120\,000$ centímetro $^2$, y
\[
120\,000 \div 48 = 2\,500 .
\]
El \omterm{def:g3:division:remainder}{cociente} no es $50$ pero $2\,500 = 50 \times 50$.

\textbf{9.} Un centímetro $^2$ de papel representa un cuadrado real de
$50$ centímetros por $50$ cm, que contiene
$50 \times 50 = 2\,500$ centímetro $^2$. Entonces, cuando las longitudes se dividen por
$50$, \omterm{def:g6:measure:area}{áreas} son
        dividido por $50 \times 50 = 2\,500$: \omterm{def:g6:measure:area}{áreas} seguir
el \emph{cuadrado} de la escala.

\textbf{10.} $6 \times 2\,500 = 15\,000$ centímetro $^2$, y
$15\,000 \div 10\,000 = 1.5$ metro $^2$.

\textbf{11.} La imagen sugiere que se vendió el domingo. \emph{dos veces} como
mucho (una barra dos veces más alta). Verdad: el aumento es $5$ salida de euros
de $105$, y $5 \div 105 \approx 0.048$: acerca de $5\,\%$ más,
no $100\,\%$ más. El gerente ignoró la advertencia para verificar
que el eje comienza en $0$
(\cref{met:g6:propdata:read}).

\textbf{12.} Con el eje de $0$, las barras suben a $105$ y
$110$ unidades pequeñas: dos barras de casi la misma altura, la
imagen honesta de un $5\,\%$ \omterm{ex:g1:subtraction:difference}{diferencia}.

\textbf{13.} Arriba: el aumento es $20$ sellos desde un inicio de
$40$: $\frac{20}{40} = 50\,\%$. Abajo: la disminución es $20$
sellos desde un inicio de $60$: $\frac{20}{60} = \frac13 \approx
33\,\%$. Mismo $20$ sellos, diferentes valores iniciales --- a
El porcentaje es siempre un \omterm{def:g6:fractions:def}{fracción} \emph{de algo}, y el
``algo'' cambió.

\textbf{14.} Después $20\,\%$ apagado: $100 - 20 = 80$ euros. el
adicional $10\,\%$ se aplica a $80$: descuento $8$ euros, precio final
$72$ euros. Descuento total: $28$ salida de euros
de $100$, entonces
$28\,\%$ --- no $30\,\%$. El segundo descuento actuó sobre el
precio ya reducido, por lo que sus euros son menores: porcentajes de
diferentes cantidades no suman.

\textbf{15.} En ese mapa, $1$ cm significa $5$ kilómetros, entonces $1$
centímetro $^2$ representa $5 \times 5 = 25$ kilómetros $^2$ (regla de la pregunta 9).
El bosque: $3 \times 25 = 75$ kilómetros $^2$. Las longitudes siguen el
escala; \omterm{def:g6:measure:area}{áreas} sigue su \emph{cuadrado}.
\end{solution}
