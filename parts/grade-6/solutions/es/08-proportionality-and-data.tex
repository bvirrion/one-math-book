\section*{Capítulo \ref{ch:g6:propdata} --- Proporcionalidad y datos}

\begin{solution}{exo:g6:propdata:1}
Manzanas: \omterm{def:g3:division:remainder}{cocientes} $\frac52 = 2.5$, $\frac{7.5}{3} = 2.5$,
$\frac{12.5}{5} = 2.5$ --- todos iguales: proporcional (precio
$2.50$ por kg).

Edad/estatura: $\frac{85}{2} = 42.5$ pero $\frac{103}{4} = 25.75$ --- no son
iguales: no es proporcional.
\end{solution}

\begin{solution}{exo:g6:propdata:2}
Un cruasán: $4.80 \div 4 = 1.20$ euros. Siete:
$7 \times 1.20 = 8.40$ euros.
\end{solution}

\begin{solution}{exo:g6:propdata:3}
Coeficiente: $18 \div 10 = 1.8$ euros por litro. Entonces
$25$ L cuestan $25 \times 1.8 = 45$ euros; con $72$ euros se compran
$72 \div 1.8 = 40$ L; $60$ L cuestan $60 \times 1.8 = 108$ euros.
\end{solution}

\begin{solution}{exo:g6:propdata:4}
Para $250$ km: $2.5$ veces el combustible de $100$ km, o sea,
$2.5 \times 6 = 15$ L. Para $350$ km: $3.5 \times 6 = 21$ L. Con
$27$ L: $27 \div 6 = 4.5$ centenares de km, es decir, $450$ km.
\end{solution}

\begin{solution}{exo:g6:propdata:5}
El $50\,\%$ de $84$: la \omterm{def:g3:division:half}{mitad}, $42$. \quad
El $25\,\%$ de $200$: un \omterm{def:g3:division:half}{cuarto}, $50$. \quad
El $10\,\%$ de $63$: una décima parte, $6.3$. \quad
El $30\,\%$ de $70$: $3 \times 7 = 21$.
\end{solution}

\begin{solution}{exo:g6:propdata:6}
El $60\,\%$ de $450$: $\frac{60}{100} \times 450 = 270$ alumnos comen en el
comedor; $450 - 270 = 180$ no comen allí.
\end{solution}

\begin{solution}{exo:g6:propdata:7}
Menos de $10$ libros: el martes ($8$) y el jueves ($6$). Miércoles
menos jueves: $17 - 6 = 11$ libros más.
\end{solution}

\begin{solution}{exo:g6:propdata:8}
Diagrama de barras con cinco barras de alturas $14$, $16$, $13$, $17$ y $20$
(va bien una graduación cada $2$ o cada $5$ grados). Día más caluroso: el viernes
($20$ grados).
\end{solution}

\begin{solution}{exo:g6:propdata:9}
Para $10$ personas hay que multiplicar por $\frac{10}{6} = \frac53$: harina,
$450 \times \frac{10}{6} = 750$ g. Huevos:
$3 \times \frac{10}{6} = 5$ --- por suerte, un número entero. (Si no lo
fuera, habría que redondear \emph{hacia arriba} para tener suficiente.)
\end{solution}

\begin{solution}{exo:g6:propdata:10}
\emph{1.} En $2$ h: $48$ km. En media hora: $12$ km. En
$1$ h $30$: $24 + 12 = 36$ km.

\emph{2.} $60 \div 24 = 2.5$ horas, es decir, $2$ h $30$ min.
\end{solution}

\begin{solution}{exo:g6:propdata:11}
Balón: descuento del $20\,\%$ de $15 = 3$ euros, precio nuevo $12$ euros.
Raqueta: descuento de $8$ euros, precio nuevo $32$ euros. El precio nuevo es
el $80\,\%$ del antiguo en todos los casos: proporcional, de coeficiente
$0.8$.
\end{solution}

\begin{solution}{exo:g6:propdata:12}
Al cabo de $t$ horas, la vela gruesa ha consumido $\frac t6$ de su altura:
se queda en $1 - \frac t6$ de la altura inicial; la fina, en
$1 - \frac t4$. Queremos
\[
1 - \frac t4 = \frac12 \left(1 - \frac t6\right)
\quad\text{i.e.}\quad
1 - \frac t4 = \frac12 - \frac t{12}.
\]
Entonces $\frac12 = \frac t4 - \frac t{12} = \frac{3t - t}{12} =
\frac{t}{6}$, así que $t = 3$ horas. Comprobación: al cabo de $3$ h la vela gruesa
mide $1 - \frac36 = \frac12$ y la fina
$1 - \frac34 = \frac14$ --- en efecto, la \omterm{def:g3:division:half}{mitad} que ella.
\end{solution}

\begin{solution}{pb:g6:propdata:1}
\textbf{1.} $100\,000$ cm $= 1\,000$ m $= 1$ km: un centímetro
del mapa es un kilómetro sobre el terreno.

\textbf{2.} $7.5$ cm $\rightarrow 7.5$ km entre los pueblos;
$12$ cm $\rightarrow 12$ km de orilla del lago.

\textbf{3.} $20$ km $\rightarrow 20$ cm en el mapa.

\textbf{4.} $5$ km $= 500\,000$ cm, así que la escala es
$1:500\,000$. El mapa de senderismo ($1:100\,000$) es el más
detallado: usa $5$ cm de papel donde el otro gasta solo
$1$ cm.

\textbf{5.}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccc}
mapa (cm) & $1$ & $2$ & $7.5$ \\
\hline
real (km) & $1$ & $2$ & $7.5$ \\
\end{tabular}
\end{center}
Distancia real $=$ distancia en el mapa $\times 1$ (en estas unidades):
siempre el \emph{mismo} multiplicador, que es exactamente la
definición de \omterm{def:g6:propdata:def}{proporcionalidad}
(\cref{def:g6:propdata:def}). El coeficiente es aquí $1$
kilómetro por centímetro.

\textbf{6.} $4$ m $= 400$ cm y $400 \div 50 = 8$;
$3$ m $= 300$ cm y $300 \div 50 = 6$: el plano muestra un
rectángulo de $8$ cm $\times$ $6$ cm.

\textbf{7.} Cama: $200 \div 50 = 4$ cm por $90 \div 50 = 1.8$ cm.
Hueco de la puerta: $80 \div 50 = 1.6$ cm.

\textbf{8.} \omterm{def:g6:measure:area}{Área} real: $4 \times 3 = 12$ m$^2$. \omterm{def:g6:measure:area}{Área} del plano:
$8 \times 6 = 48$ cm$^2$. Convirtiendo:
$12$ m$^2 = 12 \times 10\,000 = 120\,000$ cm$^2$, y
\[
120\,000 \div 48 = 2\,500 .
\]
El \omterm{def:g3:division:remainder}{cociente} no es $50$, sino $2\,500 = 50 \times 50$.

\textbf{9.} Un cm$^2$ de papel representa un cuadrado real de
$50$ cm por $50$ cm, que contiene
$50 \times 50 = 2\,500$ cm$^2$. Así que, cuando las longitudes se dividen entre
$50$, las \omterm{def:g6:measure:area}{áreas} se dividen entre $50 \times 50 = 2\,500$: las \omterm{def:g6:measure:area}{áreas} siguen
el \emph{cuadrado} de la escala.

\textbf{10.} $6 \times 2\,500 = 15\,000$ cm$^2$, y
$15\,000 \div 10\,000 = 1.5$ m$^2$.

\textbf{11.} La imagen sugiere que el domingo se vendió el \emph{doble}
(una barra el doble de alta). La verdad: el aumento es de $5$ euros sobre
$105$, y $5 \div 105 \approx 0.048$: un $5\,\%$ más,
no un $100\,\%$ más. El gerente ignoró la advertencia de comprobar
que el eje empieza en $0$
(\cref{met:g6:propdata:read}).

\textbf{12.} Con el eje desde $0$, las barras llegan a $105$ y
$110$ unidades pequeñas: dos barras de altura casi igual, la
imagen honesta de una \omterm{ex:g1:subtraction:difference}{diferencia} del $5\,\%$.

\textbf{13.} Subida: el aumento es de $20$ sellos desde un punto de partida de
$40$: $\frac{20}{40} = 50\,\%$. Bajada: la disminución es de $20$
sellos desde un punto de partida de $60$: $\frac{20}{60} = \frac13 \approx
33\,\%$. Los mismos $20$ sellos con puntos de partida distintos --- un
porcentaje es siempre una \omterm{def:g6:fractions:def}{fracción} \emph{de algo}, y ese
«algo» ha cambiado.

\textbf{14.} Después del $20\,\%$ de descuento: $100 - 20 = 80$ euros. El
$10\,\%$ adicional se aplica a $80$: descuento de $8$ euros, precio final
$72$ euros. Descuento total: $28$ euros sobre $100$, o sea,
$28\,\%$ --- no $30\,\%$. El segundo descuento actuó sobre el
precio ya rebajado, así que sus euros son menos: los porcentajes de
cantidades distintas no se suman.

\textbf{15.} En ese mapa, $1$ cm representa $5$ km, así que $1$
cm$^2$ representa $5 \times 5 = 25$ km$^2$ (la regla de la pregunta 9).
El bosque: $3 \times 25 = 75$ km$^2$. Las longitudes siguen a la
escala; las \omterm{def:g6:measure:area}{áreas} siguen a su \emph{cuadrado}.
\end{solution}
