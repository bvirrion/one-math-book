\section*{Capítulo \ref{ch:g6:fractions} --- Fracciones: primeros pasos}

\begin{solution}{exo:g6:fractions:1}
Tarta: se han comido $\frac38$. \quad Chocolate: $\frac{7}{12}$. \quad
Pastel: $\frac66 = 1$ (el pastel entero).
\end{solution}

\begin{solution}{exo:g6:fractions:2}
Se colorean cuatro partes de cinco. $\frac45 < 1$ (cuatro partes de
cinco es menos que la barra entera); $\frac65 > 1$ (seis quintos son una
barra entera y un quinto más).
\end{solution}

\begin{solution}{exo:g6:fractions:3}
En la recta graduada en tercios: $\frac13$ está una marca después del $0$;
$\frac53$ está una marca antes del $2$ (ya que $2 = \frac63$); $\frac73$ está
una marca después del $2$. Así que $\frac53$ y $\frac73$ están los dos a distancia
$\frac13$ del $2$.
\end{solution}

\begin{solution}{exo:g6:fractions:4}
$\frac94 = 2 + \frac14$; \quad
$\frac{13}{5} = 2 + \frac35$; \quad
$\frac{12}{6} = 2$ (exactamente); \quad
$\frac{25}{8} = 3 + \frac18$.
\end{solution}

\begin{solution}{exo:g6:fractions:5}
$\frac23 = \frac{8}{12}$ (multiplica por $4$); \quad
$\frac{15}{20} = \frac34$ (divide entre $5$); \quad
$\frac{7}{10} = \frac{21}{30}$ (multiplica por $3$); \quad
$\frac56 = \frac{10}{12}$ (divide entre $2$).
\end{solution}

\begin{solution}{exo:g6:fractions:6}
$\frac68 = \frac34$; \quad
$\frac{15}{25} = \frac35$; \quad
$\frac{18}{24} = \frac{9}{12} = \frac34$; \quad
$\frac{35}{7} = 5$ (un número entero).
\end{solution}

\begin{solution}{exo:g6:fractions:7}
$\frac12$ de $86$: $86 \div 2 = 43$.

$\frac34$ de $60$: $60 \div 4 = 15$, después $15 \times 3 = 45$.

$\frac25$ de $35$: $35 \div 5 = 7$, después $7 \times 2 = 14$.

$\frac{7}{10}$ de $250$: $250 \div 10 = 25$, después $25 \times 7 = 175$.
\end{solution}

\begin{solution}{exo:g6:fractions:8}
$\frac12 > \frac13$: los medios son partes mayores que los tercios.
$\frac35 > \frac25$: las mismas partes, pero más.
$\frac34 = \frac68 < \frac78$: siete octavos ganan a seis octavos.
\end{solution}

\begin{solution}{exo:g6:fractions:9}
$\frac12 = 0.5$; \quad $\frac34 = 0.75$; \quad
$\frac{7}{10} = 0.7$; \quad $\frac94 = 2.25$. La que no encaja es
$\frac13 = 0.333\dots$, cuya escritura decimal no termina nunca: no es un
\omterm{def:g6:decimals:places}{número decimal}.
\end{solution}

\begin{solution}{exo:g6:fractions:10}
En autobús: $\frac34$ de $28$: $28 \div 4 = 7$, después
$7 \times 3 = 21$ alumnos. Andando: $28 - 21 = 7$ alumnos (que son
el \omterm{def:g3:division:half}{cuarto} restante: $28 \div 4 = 7$). Comprobación: $21 + 7 = 28$.
\end{solution}

\begin{solution}{exo:g6:fractions:11}
Dos tercios de los ahorros son $18$ euros, así que \emph{un} tercio son
$18 \div 2 = 9$ euros, y los ahorros enteros (tres tercios) eran
$9 \times 3 = 27$ euros. Comprobación: $\frac23$ de $27$ es $18$.
\end{solution}

\begin{solution}{exo:g6:fractions:12}
El agua usada pasó de $\frac58$ a $\frac48$ (la \omterm{def:g3:division:half}{mitad} son cuatro
octavos) del depósito: los $10$ litros representan
$\frac58 - \frac48 = \frac18$ de la \omterm{def:g2:measure:units}{capacidad}. Así que el depósito lleno contiene
$10 \times 8 = 80$ litros. Comprobación: $\frac58$ de $80$ es $50$, menos
$10$ quedan $40$, que es en efecto la \omterm{def:g3:division:half}{mitad} de $80$.
\end{solution}

\begin{solution}{pb:g6:fractions:1}
\textbf{1.} La \omterm{def:g3:division:half}{mitad} de $16$ onzas son $8$ onzas. Como \omterm{def:g6:fractions:def}{fracción}:
$\frac12$ de la tableta y, como cada \omterm{def:g3:division:half}{mitad} son $8$ onzas de
$16$, también $\frac{8}{16}$ --- la misma \omterm{def:g6:fractions:def}{fracción} escrita de dos maneras
(\cref{prop:g6:fractions:equal}).

\textbf{2.} La \omterm{def:g3:division:half}{mitad} de las $8$ onzas restantes son $4$ onzas,
que son $\frac{4}{16}$ de la tableta. Simplificando entre $4$:
$\frac{4}{16} = \frac14$. En el dibujo: cortar en dos la \omterm{def:g3:division:half}{mitad}
restante produce dos \omterm{def:g3:division:half}{cuartos} de la tableta original --- la \omterm{def:g3:division:half}{mitad} de
una \omterm{def:g3:division:half}{mitad} es un \omterm{def:g3:division:half}{cuarto}.

\textbf{3.} Tercer bocado: la \omterm{def:g3:division:half}{mitad} de $4$ onzas son $2$ onzas, es decir,
$\frac{2}{16} = \frac18$ de la tableta. \omterm{def:g3:division:half}{Cuarto} bocado: $1$ onza,
es decir, $\frac{1}{16}$. De un bocado al siguiente, el \omterm{def:g4:fractions:def}{denominador}
se \emph{duplica}: $2$, $4$, $8$, $16$.

\textbf{4.} Comido: $8 + 4 + 2 + 1 = 15$ onzas de $16$, es
decir, $\frac{15}{16}$ de la tableta. Queda: $1$ onza, $\frac{1}{16}$.

\textbf{5.} $\frac12 = \frac{8}{16}$ (multiplica arriba y abajo por
$8$), $\frac14 = \frac{4}{16}$ (por $4$), $\frac18 = \frac{2}{16}$
(por $2$) y $\frac{1}{16}$ se queda igual. Contando dieciseisavos:
$8 + 4 + 2 + 1 = 15$, así que el total es $\frac{15}{16}$.

\textbf{6.} Bocados: $32$, $16$, $8$, $4$, $2$, $1$ onzas. Total
comido: $32 + 16 + 8 + 4 + 2 + 1 = 63$ onzas, así que queda $1$ onza:
$\frac{1}{64}$ de la tableta.

\textbf{7.} Después de cada bocado queda
\[
\frac12,\quad \frac14,\quad \frac18,\quad \frac{1}{16},\quad
\frac{1}{32},\quad \frac{1}{64} :
\]
el \omterm{def:g3:division:remainder}{resto} se parte por la \omterm{def:g3:division:half}{mitad} cada vez --- su \omterm{def:g4:fractions:def}{denominador} se duplica. Un
séptimo bocado imaginario dejaría $\frac{1}{128}$.

\textbf{8.} Cada bocado se come solo la \emph{\omterm{def:g3:division:half}{mitad}} de lo que queda, así que
la otra \omterm{def:g3:division:half}{mitad} del \omterm{def:g3:division:remainder}{resto} sigue ahí: después de cualquier número
de bocados, siempre queda algo. Por eso el total comido está
siempre por debajo de $1$ --- exactamente igual que los veinte nueves del
\cref{pb:g6:decimals:1}, que se acercan cada vez más a $3$ sin
llegar a él.

\textbf{9.} Las dos fracciones tienen \omterm{def:g4:fractions:def}{numerador} $1$, y cortar en
$128$ partes da partes más pequeñas que cortar en $100$: así que
$\frac{1}{128} < \frac{1}{100}$. Después del séptimo bocado el
\omterm{def:g3:division:remainder}{resto} es $\frac{1}{128}$, así que la parte comida supera
$1 - \frac{1}{100} = \frac{99}{100}$: el reto queda ganado en el
bocado siete.

\textbf{10.} En dieciseisavos, los totales son $\frac{8}{16}$,
$\frac{12}{16}$, $\frac{14}{16}$, $\frac{15}{16}$. Cada total nuevo
cae exactamente \emph{a \omterm{def:g3:division:half}{mitad} de camino} entre el total anterior y $1$:
la escalera va partiendo por la \omterm{def:g3:division:half}{mitad} la distancia que le queda hasta $1$
sin llegar a pisarlo nunca.

\textbf{11.} Lo cierto: en cada etapa de la descripción de Zenón,
queda en efecto algo de distancia (pregunta 8) --- la lista de etapas
no termina nunca. La trampa: las etapas se vuelven extremadamente cortas, en
distancia \emph{y} en tiempo de carrera; el corredor no dedica
el mismo tiempo a cada etapa. Describir la carrera en infinitos
trozos menguantes no hace interminable la carrera en sí: el
corredor llega al muro, y la pregunta 9 muestra que la descripción
rebasa cualquier punto anterior a él.

\textbf{12.} Corta $8$ tabletas por la \omterm{def:g3:division:half}{mitad}: $16$ medias tabletas. Corta $4$
tabletas en \omterm{def:g3:division:half}{cuartos}: $16$ \omterm{def:g3:division:half}{cuartos}. Corta $2$ tabletas en
octavos: $16$ octavos. Corta la última tableta en dieciseisavos:
$16$ trozos pequeños. Eso gasta $8 + 4 + 2 + 1 = 15$ tabletas, y cada
niño recibe un trozo de cada tamaño.

\textbf{13.} La parte de cada niño es
\[
\frac12 + \frac14 + \frac18 + \frac{1}{16} = \frac{15}{16}
\]
de tableta (pregunta 5). Dieciséis partes así hacen
$16 \times \frac{15}{16} = 15$ tabletas
(\cref{thm:g6:fractions:quotient}): exactamente lo que se cortó, sin
que sobre nada --- un reparto justo de $15$ tabletas entre $16$
niños, solo partiendo por la \omterm{def:g3:division:half}{mitad}.

\textbf{14.} Los vecinos reciben $\frac78$ de tableta cada uno. Con
\omterm{def:g4:fractions:def}{denominador} $16$: $\frac78 = \frac{14}{16}$, mientras que nuestros niños
reciben $\frac{15}{16}$. Así que recibe más un niño de los $16$:
$\frac{15}{16} > \frac{14}{16}$, un dieciseisavo de tableta más.

\textbf{15.} En el dibujo del cuadrado, cada bocado sombreado llena la \omterm{def:g3:division:half}{mitad}
de la esquina que sigue sin sombrear, y esa esquina va
encogiendo: después de cada bocado mide la \omterm{def:g3:division:half}{mitad} que antes, y
no desaparece nunca. Los dos hechos ciertos: (i) después de cualquier número
finito de bocados queda un \omterm{def:g3:division:remainder}{resto} --- el total comido está
siempre estrictamente por debajo de $1$; (ii) ese \omterm{def:g3:division:remainder}{resto} se vuelve más pequeño
que cualquier \omterm{def:g6:fractions:def}{fracción} que se quiera nombrar ($\frac{1}{100}$,
$\frac{1}{1000}$, \dots), si se muerde el tiempo suficiente. Dar a las
palabras «la \omterm{def:g1:addition:def}{suma} infinita vale $1$» su sentido exacto es
tarea de los \emph{límites}, en el \omterm{def:g5:solids:volume}{volumen} de secundaria superior.
\end{solution}
