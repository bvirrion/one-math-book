\section*{Hoofdstuk \ref{ch:g6:measure} --- Omtrek, oppervlakte, volume}

\begin{solution}{exo:g6:measure:1}
$3.5$ m $= 350$ cm; \quad $420$ mm $= 42$ cm; \quad
$0.8$ km $= 800$ m; \quad $25\,000$ m $= 25$ km.
\end{solution}

\begin{solution}{exo:g6:measure:2}
Rechthoek: $2 \times (8 + 3.5) = 2 \times 11.5 = 23$ cm.
Vierkant: $4 \times 6.2 = 24.8$ cm.
Driehoek: $5 + 7 + 9 = 21$ cm.
\end{solution}

\begin{solution}{exo:g6:measure:3}
Eén rondje: $2\pi \times 50 = 100\pi \approx 314$ m. Voor $2$ km $= 2000$ m:
$2000 \div 314 \approx 6.4$, dus zijn $7$ volle rondjes nodig ($6$ rondjes maken
maar ongeveer $1\,885$ m).
\end{solution}

\begin{solution}{exo:g6:measure:4}
Rechthoek: $9 \times 4 = 36$ cm$^2$. Vierkant: $7^2 = 49$ m$^2$. \omterm{def:g6:shapes:triangles}{Rechthoekige
driehoek}: $\frac{6 \times 10}{2} = 30$ cm$^2$. Schijf:
$\pi \times 3^2 = 9\pi \approx 28$ cm$^2$.
\end{solution}

\begin{solution}{exo:g6:measure:5}
$3$ m$^2 = 30\,000$ cm$^2$; \quad
$45\,000$ cm$^2 = 4.5$ m$^2$; \quad
$2.5$ L $= 2\,500$ cm$^3$; \quad
$4\,500$ L $= 4.5$ m$^3$.
\end{solution}

\begin{solution}{exo:g6:measure:6}
Hek (de \omterm{def:g6:measure:perimeter}{omtrek}): $2 \times (120 + 85) = 2 \times 205 = 410$ m. \omterm{def:g6:measure:area}{Oppervlakte}:
$120 \times 85 = 10\,200$ m$^2$.
\end{solution}

\begin{solution}{exo:g6:measure:7}
\omterm{def:g5:solids:def}{Balk}: $5 \times 4 \times 10 = 200$ cm$^3$. \omterm{def:g5:solids:def}{Kubus}:
$3 \times 3 \times 3 = 27$ cm$^3$.
\end{solution}

\begin{solution}{exo:g6:measure:8}
\omterm{def:g6:measure:perimeter}{Omtrek} $16$ cm betekent lengte $+$ breedte $= 8$ cm. Bijvoorbeeld $6 \times 2$
(\omterm{def:g6:measure:area}{oppervlakte} $12$ cm$^2$) en $5 \times 3$ (\omterm{def:g6:measure:area}{oppervlakte} $15$ cm$^2$) --- of het
vierkant $4 \times 4$ (\omterm{def:g6:measure:area}{oppervlakte} $16$ cm$^2$). Hoe dichter bij een vierkant,
hoe groter de \omterm{def:g6:measure:area}{oppervlakte}.
\end{solution}

\begin{solution}{exo:g6:measure:9}
\omterm{def:g6:measure:area}{Oppervlakte}: $10 \times 2 + 2 \times 6 = 20 + 12 = 32$ cm$^2$.

\omterm{def:g6:measure:perimeter}{Omtrek} (met de poot midden onder de balk): rondlopend krijg je
\[
10 + 2 + 4 + 6 + 2 + 6 + 4 + 2 = 36 \text{ cm}
\]
(bovenzijde; rechteruiteinde van de balk; onderzijde rechts; rechterkant van de
poot; onderkant van de poot; linkerkant van de poot; onderzijde links;
linkeruiteinde van de balk).
\end{solution}

\begin{solution}{exo:g6:measure:10}
\emph{1.} $V = 25 \times 10 \times 2 = 500$ m$^3$.

\emph{2.} $500$ m$^3 = 500 \times 1000 = 500\,000$ L.

\emph{3.} $500\,000 \div 50\,000 = 10$ uur.
\end{solution}

\begin{solution}{exo:g6:measure:11}
Tuin: $20^2 = 400$ m$^2$. Vijver: $\pi \times 3^2 = 9\pi$ m$^2$. Gras:
$400 - 9\pi \approx 400 - 28.3 \approx 372$ m$^2$.
\end{solution}

\begin{solution}{exo:g6:measure:12}
\emph{1.} Kleine reep: $15 \times 6 \times 1 = 90$ cm$^3$. Grote reep:
$30 \times 12 \times 2 = 720$ cm$^3$. Het \omterm{def:g6:measure:volume}{volume} wordt met
$720 \div 90 = 8$ vermenigvuldigd --- elk van de drie afmetingen verdubbelen
vermenigvuldigt het \omterm{def:g6:measure:volume}{volume} met $2 \times 2 \times 2 = 8$.

\emph{2.} Acht keer zoveel chocolade voor vier keer de prijs: ja, de grote reep
is per gram twee keer zo voordelig.
\end{solution}

\begin{solution}{pb:g6:measure:1}
\textbf{1.} $1 \times 9$: \omterm{def:g6:measure:perimeter}{omtrek} $2 \times (1 + 9) = 20$ m, \omterm{def:g6:measure:area}{oppervlakte} $9$
m$^2$. \ $2 \times 8$: \omterm{def:g6:measure:perimeter}{omtrek} $2 \times 10 = 20$ m, \omterm{def:g6:measure:area}{oppervlakte} $16$ m$^2$.

\textbf{2.} De \omterm{def:g6:measure:perimeter}{omtrek} is twee keer (lengte $+$ breedte)
(\cref{ex:g6:measure:perimeter}), dus is lengte $+$ breedte
$= 20 \div 2 = 10$ m. De tabel:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccc}
weide & $1 \times 9$ & $2 \times 8$ & $3 \times 7$ & $4 \times 6$
& $5 \times 5$ \\
\hline
\omterm{def:g6:measure:area}{oppervlakte} (m$^2$) & $9$ & $16$ & $21$ & $24$ & $25$ \\
\end{tabular}
\end{center}
Het vierkant $5 \times 5$ is de beste.

\textbf{3.} $4.5 \times 5.5 = 24.75$ en $4.9 \times 5.1 = 24.99$: steeds dichter
bij $25$, maar nog altijd eronder. Vermoeden: het vierkant verslaat \emph{elke}
rechthoek met \omterm{def:g6:measure:perimeter}{omtrek} $20$, decimale zijden inbegrepen.

\textbf{4.} Uit de tabel: het \omterm{def:g2:mult:def}{product} van twee getallen met \omterm{def:g1:addition:def}{som} $10$ is het
grootst als de twee getallen \emph{gelijk} zijn ($5$ en $5$). Algemene regel:
bij een vaste \omterm{def:g1:addition:def}{som} is het \omterm{def:g2:mult:def}{product} het grootst bij gelijke delen --- hoe ongelijker
het paar, hoe kleiner het \omterm{def:g2:mult:def}{product}.

\textbf{5.} Haal van het vierkant $5 \times 5$ de bovenste rij weg --- een strook
van $5$ vierkantjes --- zodat een weide $5 \times 4$ overblijft; plak daarna een
kolom van $4$ vierkantjes tegen één uiteinde, wat de weide $6 \times 4$ maakt. Er
gingen vijf vierkantjes af en er kwamen er maar vier terug: de toegevoegde strook
ligt langs de zijde van $4$, en die is korter dan de zijde van $5$ waarlangs de
weggehaalde strook lag. Netto verlies: één vierkante meter ($25 \to 24$). De
volgende stap, van $6 \times 4$ naar $7 \times 3$, ruilt een rij van $6$ voor een
kolom van $3$ en verliest er nog drie ($24 \to 21$): hoe verder de weide van het
vierkant af ligt, hoe langer de strook die ze weggeeft en hoe korter die ze
terugkrijgt.

\textbf{6.} Met $b + l + b = 20$, dus $l = 20 - 2b$:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccccc}
$b$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ & $7$ \\
\hline
$l$ & $18$ & $16$ & $14$ & $12$ & $10$ & $8$ & $6$ \\
\omterm{def:g6:measure:area}{oppervlakte} & $18$ & $32$ & $42$ & $48$ & $50$ & $48$ & $42$ \\
\end{tabular}
\end{center}
De winnaar is $5 \times 10$, met \omterm{def:g6:measure:area}{oppervlakte} $50$ m$^2$ --- twee keer zo lang
(langs de muur) als breed: \emph{geen} vierkant.

\textbf{7.} Spiegel de weide in de muur: de weide met haar \omterm{def:g6:symmetry:def}{spiegelbeeld} vormt een
verdubbelde weide die het hek twee keer gebruikt, $40$ m hek rondom (de muurzijde
ligt nu binnenin). Volgens deel I is de beste rechthoek met \omterm{def:g6:measure:perimeter}{omtrek} $40$ het
vierkant $10 \times 10$, met \omterm{def:g6:measure:area}{oppervlakte} $100$ m$^2$. De beste echte weide is de
\omterm{def:g3:division:half}{helft} van die beste verdubbelde weide: $10$ m langs de muur, $5$ m diep,
\omterm{def:g6:measure:area}{oppervlakte} $50$ m$^2$ --- precies de winnaar uit de tabel, nu verklaard.

\textbf{8.} \omterm{def:g6:measure:perimeter}{Omtrek} $20 = 2 \times \pi \times r$, dus
$r = 20 \div 6.28 \approx 3.18$ m. \omterm{def:g6:measure:area}{Oppervlakte}:
$\pi r^2 \approx 3.14 \times 3.18 \times 3.18 \approx 31.8$ m$^2$ --- ruim meer
dan de $25$ m$^2$ van het vierkant. Dido wist wat ze deed: bij een gegeven lengte
rand omsluit de cirkel het meest.

\textbf{9.} Omtrekken: $1 \times 36$: $74$ m; \ $2 \times 18$: $40$ m; \
$3 \times 12$: $30$ m; \ $4 \times 9$: $26$ m; \ $6 \times 6$: $24$ m. Opnieuw
het vierkant --- het minste hek bij de gegeven \omterm{def:g6:measure:area}{oppervlakte}. Het is deel I
achterstevoren: vaste \omterm{def:g6:measure:perimeter}{omtrek} en grootste \omterm{def:g6:measure:area}{oppervlakte}, of vaste \omterm{def:g6:measure:area}{oppervlakte} en
kleinste \omterm{def:g6:measure:perimeter}{omtrek}: beide kronen het vierkant (en, voorbij alle rechthoeken, de
cirkel).

\textbf{10.} Een ronde wand is de kortste rand voor de ruimte die hij omsluit
(vraag 8), dus houdt een rond blikje, een ronde buis of een ronde tank de meeste
\omterm{def:g2:measure:units}{inhoud} bij het minste metaal --- en materiaal is geld.

\textbf{11.} Bijvoorbeeld $50$ m $\times$ $0.01$ m: \omterm{def:g6:measure:perimeter}{omtrek}
$2 \times (50 + 0.01) = 100.02$ m, \omterm{def:g6:measure:area}{oppervlakte} $50 \times 0.01 = 0.5$ m$^2$. Lang
en dun: kilometers rand, bijna geen land.

\textbf{12.} Niet waar. De weide van vraag 11 heeft \omterm{def:g6:measure:perimeter}{omtrek} $100.02$ m en
\omterm{def:g6:measure:area}{oppervlakte} $0.5$ m$^2$, terwijl de weide $5 \times 5$ \omterm{def:g6:measure:perimeter}{omtrek} $20$ m en
\omterm{def:g6:measure:area}{oppervlakte} $25$ m$^2$ heeft: grotere \omterm{def:g6:measure:perimeter}{omtrek}, veel kleinere \omterm{def:g6:measure:area}{oppervlakte}.

\textbf{13.} \omterm{def:g6:measure:area}{Oppervlakte}: er mist één eenheidsvierkantje, dus $24 - 1 = 23$
cm$^2$. \omterm{def:g6:measure:perimeter}{Omtrek}: rondlopend vervangt de hap $1$ cm rechte wand door drie zijden van
het kleine vierkantje, $1 + 1 + 1 = 3$ cm: de \omterm{def:g6:measure:perimeter}{omtrek} groeit van $20$ cm naar
$20 - 1 + 3 = 22$ cm. Materiaal weghalen heeft de rand \emph{verlengd}.

\textbf{14.} Elke zoom op een echte kust onthult nieuwe baaien, rotsen en
kreken --- happen op happen, en vraag 13 laat zien dat elke hap randlengte
toevoegt zonder de \omterm{def:g6:measure:area}{oppervlakte} noemenswaardig te veranderen. De gemeten lengte
blijft dus groeien met het detailniveau (``de kustlijnparadox''), terwijl de
gemeten \omterm{def:g6:measure:area}{oppervlakte} zich stabiliseert: \omterm{def:g6:measure:perimeter}{omtrek} en \omterm{def:g6:measure:area}{oppervlakte} zijn echt
onafhankelijke grootheden.

\textbf{15.} Het spiegelargument: de verdubbelde weide zou $2 \times 36 = 72$ m
hek gebruiken, en de beste rechthoek met \omterm{def:g6:measure:perimeter}{omtrek} $72$ is het vierkant
$18 \times 18$. De beste weide is dus $18$ m langs de muur en $9$ m diep:
hekcontrole $9 + 18 + 9 = 36$ m, \omterm{def:g6:measure:area}{oppervlakte} $18 \times 9 = 162$ m$^2$. In het
open veld is de beste het vierkant $9 \times 9$, met \omterm{def:g6:measure:area}{oppervlakte} $81$ m$^2$. De
schuurmuur \emph{verdubbelt} precies het land van de boer.
\end{solution}
