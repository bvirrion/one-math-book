\section*{Chapitre \ref{ch:g6:fractions} --- Fractions~: premiers pas}

\begin{solution}{exo:g6:fractions:1}
Gâteau~: $\frac38$ mangés. \quad Chocolat~: $\frac{7}{12}$. \quad
Tarte~: $\frac66 = 1$ (toute la tarte).
\end{solution}

\begin{solution}{exo:g6:fractions:2}
Quatre parts sur cinq sont coloriées. $\frac45 < 1$ (quatre parts sur
cinq, c'est moins que la barre entière)~; $\frac65 > 1$ (six cinquièmes,
c'est une barre entière et un cinquième de plus).
\end{solution}

\begin{solution}{exo:g6:fractions:3}
Sur la droite graduée en tiers~: $\frac13$ est une graduation après
$0$~; $\frac53$ est une graduation avant $2$ (car $2 = \frac63$)~;
$\frac73$ est une graduation après $2$. Donc $\frac53$ et $\frac73$
sont tous deux à distance $\frac13$ de $2$.
\end{solution}

\begin{solution}{exo:g6:fractions:4}
$\frac94 = 2 + \frac14$~; \quad
$\frac{13}{5} = 2 + \frac35$~; \quad
$\frac{12}{6} = 2$ (exactement)~; \quad
$\frac{25}{8} = 3 + \frac18$.
\end{solution}

\begin{solution}{exo:g6:fractions:5}
$\frac23 = \frac{8}{12}$ (multiplier par $4$)~; \quad
$\frac{15}{20} = \frac34$ (diviser par $5$)~; \quad
$\frac{7}{10} = \frac{21}{30}$ (multiplier par $3$)~; \quad
$\frac56 = \frac{10}{12}$ (diviser par $2$).
\end{solution}

\begin{solution}{exo:g6:fractions:6}
$\frac68 = \frac34$~; \quad
$\frac{15}{25} = \frac35$~; \quad
$\frac{18}{24} = \frac{9}{12} = \frac34$~; \quad
$\frac{35}{7} = 5$ (un entier).
\end{solution}

\begin{solution}{exo:g6:fractions:7}
$\frac12$ de $86$~: $86 \div 2 = 43$.

$\frac34$ de $60$~: $60 \div 4 = 15$, puis $15 \times 3 = 45$.

$\frac25$ de $35$~: $35 \div 5 = 7$, puis $7 \times 2 = 14$.

$\frac{7}{10}$ de $250$~: $250 \div 10 = 25$, puis $25 \times 7 = 175$.
\end{solution}

\begin{solution}{exo:g6:fractions:8}
$\frac12 > \frac13$~: les \omterm{def:g3:division:half}{moitiés} sont des parts plus grandes que les
tiers. $\frac35 > \frac25$~: mêmes parts, plus d'entre elles.
$\frac34 = \frac68 < \frac78$~: sept huitièmes battent six huitièmes.
\end{solution}

\begin{solution}{exo:g6:fractions:9}
$\frac12 = 0.5$~; \quad $\frac34 = 0.75$~; \quad
$\frac{7}{10} = 0.7$~; \quad $\frac94 = 2.25$. L'intrus est
$\frac13 = 0.333\dots$, dont l'écriture décimale ne s'arrête jamais~:
ce n'est pas un \omterm{def:g6:decimals:places}{nombre décimal}.
\end{solution}

\begin{solution}{exo:g6:fractions:10}
Usagers du bus~: $\frac34$ de $28$~: $28 \div 4 = 7$, puis
$7 \times 3 = 21$ élèves. À pied~: $28 - 21 = 7$ élèves (c'est le
\omterm{def:g3:division:half}{quart} restant~: $28 \div 4 = 7$). Vérif.~: $21 + 7 = 28$.
\end{solution}

\begin{solution}{exo:g6:fractions:11}
Les deux tiers des économies valent $18$ euros, donc \emph{un} tiers
vaut $18 \div 2 = 9$ euros, et les économies entières (trois tiers)
étaient $9 \times 3 = 27$ euros. Vérif.~: $\frac23$ de $27$ est $18$.
\end{solution}

\begin{solution}{exo:g6:fractions:12}
L'eau utilisée est passée de $\frac58$ à $\frac48$ (une \omterm{def:g3:division:half}{moitié}, c'est
quatre huitièmes) du réservoir~: les $10$ litres représentent
$\frac58 - \frac48 = \frac18$ de la capacité. Donc le réservoir plein
tient $10 \times 8 = 80$ litres. Vérif.~: $\frac58$ de $80$ est $50$,
moins $10$ laisse $40$, qui est bien la \omterm{def:g3:division:half}{moitié} de $80$.
\end{solution}

\begin{solution}{pb:g6:fractions:1}
\textbf{1.} La \omterm{def:g3:division:half}{moitié} de $16$ carrés, c'est $8$ carrés. En
\omterm{def:g6:fractions:def}{fraction}~: $\frac12$ de la tablette, et comme chaque \omterm{def:g3:division:half}{moitié} fait $8$
carrés sur $16$, aussi $\frac{8}{16}$ --- la même \omterm{def:g6:fractions:def}{fraction} écrite de
deux façons (\cref{prop:g6:fractions:equal}).

\textbf{2.} La \omterm{def:g3:division:half}{moitié} des $8$ carrés restants, c'est $4$ carrés,
soit $\frac{4}{16}$ de la tablette. En simplifiant par $4$~:
$\frac{4}{16} = \frac14$. Sur le dessin~: couper en deux la \omterm{def:g3:division:half}{moitié}
restante donne deux \omterm{def:g3:division:half}{quarts} de la tablette de départ --- la \omterm{def:g3:division:half}{moitié}
d'une \omterm{def:g3:division:half}{moitié} est un \omterm{def:g3:division:half}{quart}.

\textbf{3.} Troisième bouchée~: la \omterm{def:g3:division:half}{moitié} de $4$ carrés, soit $2$
carrés, c'est-à-dire $\frac{2}{16} = \frac18$ de la tablette.
Quatrième bouchée~: $1$ carré, c'est-à-dire $\frac{1}{16}$. D'une
bouchée à la suivante, le \omterm{def:g4:fractions:def}{dénominateur} \emph{double}~: $2$, $4$, $8$,
$16$.

\textbf{4.} Mangés~: $8 + 4 + 2 + 1 = 15$ carrés sur $16$, soit
$\frac{15}{16}$ de la tablette. \omterm{def:g3:division:remainder}{Reste}~: $1$ carré, $\frac{1}{16}$.

\textbf{5.} $\frac12 = \frac{8}{16}$ (multiplier en haut et en bas
par $8$), $\frac14 = \frac{4}{16}$ (par $4$),
$\frac18 = \frac{2}{16}$ (par $2$), et $\frac{1}{16}$ \omterm{def:g3:division:remainder}{reste} tel quel.
En comptant les seizièmes~: $8 + 4 + 2 + 1 = 15$, donc le total vaut
$\frac{15}{16}$.

\textbf{6.} Bouchées~: $32$, $16$, $8$, $4$, $2$, $1$ carrés. Total
mangé~: $32 + 16 + 8 + 4 + 2 + 1 = 63$ carrés, il reste donc $1$
carré~: $\frac{1}{64}$ de la tablette.

\textbf{7.} Après chaque bouchée il reste
\[
\frac12,\quad \frac14,\quad \frac18,\quad \frac{1}{16},\quad
\frac{1}{32},\quad \frac{1}{64} :
\]
le \omterm{def:g3:division:remainder}{reste} est divisé par deux à chaque fois --- son \omterm{def:g4:fractions:def}{dénominateur}
double. Une septième bouchée, imaginaire, laisserait $\frac{1}{128}$.

\textbf{8.} Chaque bouchée ne mange que la \emph{\omterm{def:g3:division:half}{moitié}} de ce qui
reste~: l'autre \omterm{def:g3:division:half}{moitié} du \omterm{def:g3:division:remainder}{reste} est donc toujours là. Après un
nombre quelconque de bouchées, il reste toujours quelque chose. Le
total mangé est donc toujours inférieur à $1$ --- exactement comme
les vingt neufs du~\cref{pb:g6:decimals:1}, qui s'approchent
indéfiniment de $3$ sans l'atteindre.

\textbf{9.} Les deux \omterm{def:g6:fractions:def}{fractions} ont le \omterm{def:g4:fractions:def}{numérateur} $1$, et couper en
$128$ parts donne des parts plus petites que couper en $100$~: donc
$\frac{1}{128} < \frac{1}{100}$. Après la septième bouchée, le \omterm{def:g3:division:remainder}{reste}
vaut $\frac{1}{128}$, donc la part mangée dépasse
$1 - \frac{1}{100} = \frac{99}{100}$~: le défi est gagné à la
septième bouchée.

\textbf{10.} En seizièmes, les totaux sont $\frac{8}{16}$,
$\frac{12}{16}$, $\frac{14}{16}$, $\frac{15}{16}$. Chaque nouveau
total tombe exactement \emph{au milieu} entre le total précédent et
$1$~: l'escalier réduit de \omterm{def:g3:division:half}{moitié}, à chaque marche, la distance qui
lui \omterm{def:g3:division:remainder}{reste} jusqu'à $1$, sans jamais poser le pied dessus.

\textbf{11.} Ce qui est juste~: à chaque étape de la description de
Zénon, il reste bel et bien une distance (question 8) --- la liste
des étapes ne s'arrête jamais. Le piège~: les étapes deviennent
extrêmement courtes, en distance \emph{et} en durée~; le coureur ne
passe pas le même temps sur chacune. Décrire la course en une
infinité de morceaux de plus en plus petits ne rend pas la course
elle-même sans fin~: le coureur atteint le mur, et la question 9
montre que la description dépasse tout point situé avant lui.

\textbf{12.} Couper $8$ tablettes en deux~: $16$ demi-morceaux.
Couper $4$ tablettes en quatre~: $16$ \omterm{def:g3:division:half}{quarts}. Couper $2$ tablettes en
huit~: $16$ huitièmes. Couper la dernière tablette en seize~: $16$
petits morceaux. Cela utilise $8 + 4 + 2 + 1 = 15$ tablettes, et
chaque enfant reçoit un morceau de chaque taille.

\textbf{13.} La part de chaque enfant vaut
\[
\frac12 + \frac14 + \frac18 + \frac{1}{16} = \frac{15}{16}
\]
de tablette (question 5). Seize parts pareilles font
$16 \times \frac{15}{16} = 15$ tablettes
(\cref{thm:g6:fractions:quotient})~: exactement ce qui a été coupé,
sans rien en trop --- un partage équitable de $15$ tablettes entre
$16$ enfants, rien qu'en coupant en deux.

\textbf{14.} Les voisins reçoivent $\frac78$ de tablette chacun. Avec
le \omterm{def:g4:fractions:def}{dénominateur} $16$~: $\frac78 = \frac{14}{16}$, alors que nos
enfants reçoivent $\frac{15}{16}$. Un enfant des $16$ reçoit donc
davantage~: $\frac{15}{16} > \frac{14}{16}$, d'un seizième de
tablette.

\textbf{15.} Sur le dessin du carré, chaque bouchée coloriée remplit
la \omterm{def:g3:division:half}{moitié} du coin encore blanc, et ce coin blanc ne cesse de
rétrécir~: après chaque bouchée il est deux fois plus petit
qu'avant, et il ne disparaît jamais. Les deux faits vrais~: (i)
après tout nombre fini de bouchées, il reste quelque chose --- le
total mangé est toujours strictement inférieur à $1$~; (ii) ce \omterm{def:g3:division:remainder}{reste}
devient plus petit que n'importe quelle \omterm{def:g6:fractions:def}{fraction} que l'on voudra
nommer ($\frac{1}{100}$, $\frac{1}{1000}$, \dots), pourvu qu'on morde
assez longtemps. Donner leur sens exact aux mots «~la \omterm{def:g1:addition:def}{somme} infinie
vaut $1$~» est le travail des \emph{limites}, dans le \omterm{def:g5:solids:volume}{volume} suivant.
\end{solution}
