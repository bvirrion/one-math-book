\section*{Chapitre \ref{ch:g6:symmetry} --- La symétrie axiale}

\begin{solution}{exo:g6:symmetry:1}
$A'$ est à deux carreaux à \emph{droite} de $d$, sur la même ligne
que $A$~; $B'$ est à cinq carreaux à droite, un carreau plus haut ---
positions miroir.
\end{solution}

\begin{solution}{exo:g6:symmetry:2}
\omterm{def:g6:lines:perp}{Perpendiculaire} à $d$ par $M$, pied $H$~; puis $M'$ de l'autre côté
avec $HM' = MH = 3$~cm. La distance $MM'$ est $3 + 3 = 6$~cm.
\end{solution}

\begin{solution}{exo:g6:symmetry:3}
Axe vertical (comme A)~: A, H, I, M, O, T, U, V, W, X, Y.
Axe horizontal (comme B)~: B, C, D, E, H, I, K, O, X.
Les deux~: H, I, O, X.
(Les réponses exactes dépendent un peu de la façon dont les lettres
sont dessinées.)
\end{solution}

\begin{solution}{exo:g6:symmetry:4}
\omterm{def:g6:shapes:triangles}{Triangle équilatéral}~: $3$ axes. Rectangle~: $2$ (les \omterm{def:g6:symmetry:bisector}{médiatrices} de
ses côtés). \omterm{def:g6:shapes:quadrilaterals}{Losange}~: $2$ (ses diagonales). Carré~: $4$. Cercle~:
une infinité (toute droite passant par le centre).
\end{solution}

\begin{solution}{exo:g6:symmetry:5}
\emph{Faux.} Plier un rectangle (non carré) le long d'une diagonale
ne fait pas coïncider les deux \omterm{def:g3:division:half}{moitiés}~: les deux triangles ont la
même forme mais se placent différemment --- essayer avec une feuille
de papier. Seul pour un carré les diagonales fonctionnent comme axes.
\end{solution}

\begin{solution}{exo:g6:symmetry:6}
Les réflexions conservent les longueurs, les angles et les \omterm{def:g4:measure:perimeter}{périmètres}
(\cref{prop:g6:symmetry:preserved})~: $A'B' = 4$~cm,
$\widehat{B'A'C'} = 70^\circ$, \omterm{def:g4:measure:perimeter}{périmètre} de $A'B'C' = 12$~cm.
\end{solution}

\begin{solution}{exo:g6:symmetry:7}
Construction au compas comme dans \cref{met:g6:symmetry:bisector}.
Pour tout $M$ sur la droite, la règle confirme $MA = MB$
(\cref{prop:g6:symmetry:bisector}).
\end{solution}

\begin{solution}{exo:g6:symmetry:8}
Tous les points à égale distance de $A$ et de $B$ forment la
\omterm{def:g6:symmetry:bisector}{médiatrice} de $[AB]$ (\cref{prop:g6:symmetry:bisector}). La fontaine
peut se placer n'importe où sur cette droite (dans le jardin).
\end{solution}

\begin{solution}{exo:g6:symmetry:9}
Chaque \omterm{def:g5:solids:def}{sommet} est réfléchi perpendiculairement à l'axe~: pour l'axe à
$45^\circ$, un point à $k$ carreaux à droite de l'axe se pose à $k$
carreaux au-dessus (et inversement) --- les déplacements horizontaux
et verticaux sont échangés. Réfléchir les \omterm{def:g5:solids:def}{sommets} un par un, puis les
joindre dans l'ordre.
\end{solution}

\begin{solution}{exo:g6:symmetry:10}
Le \omterm{def:g6:symmetry:def}{symétrique} de $M$ est la seconde intersection du cercle avec la
\omterm{def:g6:lines:perp}{perpendiculaire} à $d$ par $M$ --- il est sur le cercle. Raison~: la
réflexion conserve les distances et fixe $O$ (qui est sur $d$), donc
l'image de tout point à distance $r$ de $O$ est encore à distance $r$
de $O$~: le cercle se transforme en lui-même.
\end{solution}

\begin{solution}{exo:g6:symmetry:11}
Chaque point de croisement $P$ a été tracé avec la même ouverture de
compas $\rho$ depuis $A$ et depuis $B$~: donc $PA = \rho = PB$. Par
\cref{prop:g6:symmetry:bisector}, un point équidistant de $A$ et de
$B$ se trouve sur la \omterm{def:g6:symmetry:bisector}{médiatrice} de $[AB]$~; deux tels points
déterminent la droite.
\end{solution}

\begin{solution}{exo:g6:symmetry:12}
Pour tout point de rebond $P$ sur le mur, $PB = PB'$ (la réflexion
par rapport à $d$ conserve les distances), donc la longueur du chemin
$AP + PB = AP + PB'$. La ligne brisée $A \to P \to B'$ est la plus
courte quand elle est droite, c'est-à-dire quand $P$ est sur le
\omterm{def:g6:lines:objects}{segment} $[AB']$. Donc le meilleur point de rebond est l'intersection
de $[AB']$ avec le mur.
\end{solution}

\begin{solution}{pb:g6:symmetry:1}
\textbf{1.} Construction comme dans
\cref{met:g6:symmetry:bisector}~; pour tout $M$ de la droite, la
règle confirme $MA = MB$ (\cref{prop:g6:symmetry:bisector}).

\textbf{2.} La question de la fontaine a besoin du second sens~:
\emph{tous} les points équidistants sont sur la \omterm{def:g6:symmetry:bisector}{médiatrice}, donc les
emplacements possibles sont exactement cette droite et rien d'autre.
Et un point qui n'est pas sur la \omterm{def:g6:symmetry:bisector}{médiatrice} n'est donc \emph{pas}
équidistant~: il est strictement plus proche de l'un des deux arbres.

\textbf{3.} Les deux trajets ont en commun le \omterm{def:g6:lines:objects}{segment} $[MP]$, et
$PB = PA$ puisque $P$ est sur la \omterm{def:g6:symmetry:bisector}{médiatrice}. Le trajet
$M \to P \to B$ a donc la même longueur que le trajet
$M \to P \to A$, à savoir $MP + PA$. Or $MB$ est le chemin droit
vers $B$ --- ici c'est justement le trajet par $P$, donc
$MB = MP + PA$~; et le chemin droit vers $A$ est plus court que le
détour par $P$~: $MA < MP + PA = MB$. Tout point du côté de $A$ est
plus proche de $A$.

\textbf{4.} La frontière est la \omterm{def:g6:symmetry:bisector}{médiatrice} de $[AB]$~: l'école $A$
dessert exactement la \omterm{def:g3:division:half}{moitié} de la ville située de son côté de cette
droite (question 3), l'école $B$ l'autre \omterm{def:g3:division:half}{moitié}, et les enfants
habitant sur la droite elle-même peuvent choisir.

\textbf{5.} $O$ est sur la \omterm{def:g6:symmetry:bisector}{médiatrice} de $[AB]$, donc $OA = OB$~; et
sur la \omterm{def:g6:symmetry:bisector}{médiatrice} de $[BC]$, donc $OB = OC$
(\cref{prop:g6:symmetry:bisector}).

\textbf{6.} De $OA = OB$ et $OB = OC$ on tire $OA = OC$. D'après le
second sens de la~\cref{prop:g6:symmetry:bisector}, un point équidistant
de $A$ et de $C$ est sur la \omterm{def:g6:symmetry:bisector}{médiatrice} de $[AC]$~: la troisième
\omterm{def:g6:symmetry:bisector}{médiatrice} passe par $O$ sans qu'on le lui ait demandé. Les trois
\omterm{def:g6:symmetry:bisector}{médiatrices} d'un triangle quelconque sont \emph{concourantes}.

\textbf{7.} Le cercle de centre $O$ et de \omterm{def:g3:shapes:shapes}{rayon} $OA$ est formé de
tous les points situés à la distance $OA$ de $O$
(\cref{def:g6:lines:circle})~; comme $OB = OA$ et $OC = OA$, les
points $B$ et $C$ sont dessus. Un centre, un \omterm{def:g3:shapes:shapes}{rayon}, trois points
desservis~: c'est le cercle passant par $A$, $B$ et $C$.

\textbf{8.} Deux \omterm{def:g6:symmetry:bisector}{médiatrices} suffisent parce que la troisième est
automatique (question 6)~: leur point d'intersection vérifie déjà
les trois égalités de distances. Le cercle se trace de centre $O$,
le compas ouvert de $O$ à l'un quelconque des trois \omterm{def:g5:solids:def}{sommets}.

\textbf{9.} Recette~: marquer trois points $A$, $B$, $C$ sur l'arc
subsistant~; tracer les \omterm{def:g6:symmetry:bisector}{médiatrices} des cordes $[AB]$ et $[BC]$~;
leur point d'intersection est le centre de l'assiette, et sa
distance à $A$ en est le \omterm{def:g3:shapes:shapes}{rayon}. Cela marche parce que le bord de
l'assiette est un cercle passant par les trois points marqués --- et
la question 8 trouve \emph{le} cercle qui passe par eux. L'arc
d'essai se reconstitue exactement.

\textbf{10.} Si $A$, $B$, $C$ sont alignés, la \omterm{def:g6:symmetry:bisector}{médiatrice} de $[AB]$
et celle de $[BC]$ sont toutes deux \omterm{def:g6:lines:perp}{perpendiculaires} à la même
droite $(AB) = (BC)$ --- elles sont donc \omterm{def:g6:lines:perp}{parallèles} entre elles
(\cref{prop:g6:lines:facts}) et, étant distinctes, ne se coupent
jamais. Aucun point n'est équidistant des trois, et aucun cercle ne
passe par trois points alignés.

\textbf{11.} \omterm{def:g4:geometry:polygon}{Pentagone} régulier~: $5$ axes, chacun passant par un
\omterm{def:g5:solids:def}{sommet} et le milieu du côté opposé (\omterm{def:g3:numbers:evenodd}{nombre impair}~: tous les axes
sont de cette seule espèce). \omterm{def:g4:geometry:polygon}{Hexagone} régulier~: $6$ axes --- trois
par des \omterm{def:g5:solids:def}{sommets} opposés et trois par les milieux de côtés opposés
(\omterm{def:g3:numbers:evenodd}{nombre pair}~: deux espèces, \omterm{def:g3:division:half}{moitié-moitié}). La régularité~: un
\omterm{def:g4:geometry:polygon}{polygone} régulier à $n$ côtés a $n$ axes.

\textbf{12.} La découpe traverse les deux épaisseurs à la fois~: les
deux épaisseurs perdent donc des morceaux exactement identiques~; et
déplier, c'est effectuer la symétrie par rapport au pli, qui envoie
le bord de chaque épaisseur sur celui de l'autre. Le trou est son
propre \omterm{def:g6:symmetry:def}{symétrique}~: le pli en est un \omterm{def:g6:symmetry:axis}{axe de symétrie} --- quelle que
soit la découpe.

\textbf{13.} Le trou déplié a au moins $2$ \omterm{def:g6:symmetry:axis}{axes de symétrie}, les
deux plis, et la découpe apparaît en $4$ exemplaires (chaque
dépliage double l'image~: $1 \to 2 \to 4$). Plier davantage, à la
manière des flocons, multiplie encore les copies et les axes.

\textbf{14.} $F''$ a la \emph{même} orientation que $F$ (les deux
retournements se compensent), la même taille et la même hauteur ---
c'est simplement $F$ glissé vers la droite. Le glissement mesure $8$
carreaux~: exactement le \emph{double} de la distance entre les deux
axes. Deux miroirs \omterm{def:g6:lines:perp}{parallèles} ne renversent pas du tout~: ils
\emph{translatent}.

\textbf{15.} $F''$ a de nouveau la même taille, mais il apparaît
maintenant \emph{à l'envers}, tourné d'un demi-tour autour du point
d'intersection des deux axes~: chaque point de $F''$ est
diamétralement opposé au point correspondant de $F$ par rapport à ce
centre. Deux miroirs \omterm{def:g6:lines:perp}{perpendiculaires} se composent en un
\emph{demi-tour} --- la symétrie centrale du \cref{ch:g7:central}.
\end{solution}
