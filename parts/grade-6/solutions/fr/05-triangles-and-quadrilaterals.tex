\section*{Chapitre \ref{ch:g6:shapes} --- Triangles et quadrilatères}

\begin{solution}{exo:g6:shapes:1}
Suivre la méthode~: \omterm{def:g6:lines:objects}{segment} $[AB]$ de $6$~cm, arc de \omterm{def:g3:shapes:shapes}{rayon} $5$~cm
centré en $A$, arc de \omterm{def:g3:shapes:shapes}{rayon} $4$~cm centré en $B$~; le point de
croisement est $C$.
\end{solution}

\begin{solution}{exo:g6:shapes:2}
Base $[EF]$ de $3$~cm, puis deux arcs de \omterm{def:g3:shapes:shapes}{rayon} $5$~cm centrés en $E$
et $F$~; leur croisement est $D$. Les deux angles à la base
$\widehat{DEF}$ et $\widehat{DFE}$ sont égaux (environ $72.5^\circ$
chacun) --- symétrie du \omterm{def:g6:shapes:triangles}{triangle isocèle}
(\cref{ex:g6:symmetry:proves}).
\end{solution}

\begin{solution}{exo:g6:shapes:3}
\omterm{def:g6:lines:objects}{Segment} de $4$~cm, puis deux arcs de \omterm{def:g3:shapes:shapes}{rayon} $4$~cm centrés en ses
extrémités. Chaque angle mesure $60^\circ$.
\end{solution}

\begin{solution}{exo:g6:shapes:4}
(a) \omterm{def:g6:shapes:triangles}{équilatéral} (donc aussi \omterm{def:g6:shapes:triangles}{isocèle}). \quad
(b) \omterm{def:g6:shapes:triangles}{isocèle} ($5 = 5$). \quad
(c) rectangle et \omterm{def:g6:shapes:triangles}{isocèle} (deux angles égaux de $45^\circ$ font face à
deux côtés égaux). \quad
(d) \omterm{def:g6:shapes:triangles}{équilatéral} (trois angles égaux).
\end{solution}

\begin{solution}{exo:g6:shapes:5}
L'hypoténuse est le côté opposé à l'\omterm{def:g3:shapes:rightangle}{angle droit}, toujours le plus long
côté. Dans $MNP$ rectangle en $N$, l'hypoténuse est $[MP]$~: elle ne
touche jamais le \omterm{def:g5:solids:def}{sommet} de l'\omterm{def:g3:shapes:rightangle}{angle droit}, $N$.
\end{solution}

\begin{solution}{exo:g6:shapes:6}
\emph{1. Vrai}~: un carré a quatre \omterm{def:g3:shapes:rightangle}{angles droits}, ce qui est la
définition d'un rectangle.

\emph{2. Faux}~: un rectangle $5 \times 3$ a des côtés inégaux.

\emph{3. Vrai}~: un carré a quatre côtés égaux, définition d'un
\omterm{def:g6:shapes:quadrilaterals}{losange}.

\emph{4. Vrai}~: voir \cref{ex:g6:shapes:recognize}.
\end{solution}

\begin{solution}{exo:g6:shapes:7}
Construction avec des \omterm{def:g6:lines:perp}{perpendiculaires}, comme dans
\cref{ex:g6:shapes:program}. Les deux diagonales mesurent environ
$5.8$~cm~: les diagonales d'un rectangle ont la même longueur
(\cref{prop:g6:shapes:diagonals}).
\end{solution}

\begin{solution}{exo:g6:shapes:8}
Tracer un \omterm{def:g6:lines:objects}{segment} de $6$~cm~; son milieu $O$~; la \omterm{def:g6:lines:perp}{perpendiculaire} en
$O$~; dessus, deux points à $2$~cm de $O$ de chaque côté. Joindre
les quatre extrémités donne le \omterm{def:g6:shapes:quadrilaterals}{losange} (les quatre \omterm{def:g5:solids:def}{sommets} sont aux
distances $3$~cm et $2$~cm de $O$ le long de \omterm{def:g6:lines:perp}{droites perpendiculaires},
donc les quatre côtés sont égaux par symétrie).
\end{solution}

\begin{solution}{exo:g6:shapes:9}
Un programme correct~:
1. tracer un \omterm{def:g6:lines:objects}{segment} $[BC]$ avec $BC = 4$~cm~;
2. tracer l'arc de centre $B$ et de \omterm{def:g3:shapes:shapes}{rayon} $6$~cm au-dessus de
$[BC]$~;
3. tracer l'arc de centre $C$ et de \omterm{def:g3:shapes:shapes}{rayon} $6$~cm au-dessus de
$[BC]$~;
4. nommer $A$ le point de croisement des deux arcs~;
5. tracer les \omterm{def:g6:lines:objects}{segments} $[AB]$ et $[AC]$.
\end{solution}

\begin{solution}{exo:g6:shapes:10}
Quelle que soit la position de $M$ sur le cercle, l'angle
$\widehat{AMB}$ mesure $90^\circ$~: le triangle $AMB$ est rectangle
en $M$ dès que $[AB]$ est un \omterm{def:g4:geometry:circle}{diamètre}. La démonstration attend
\cref{ch:g8:cosine}.
\end{solution}

\begin{solution}{exo:g6:shapes:11}
Des diagonales qui se coupent en leur milieu et ont des longueurs
égales~: c'est la liste des propriétés du \emph{rectangle}
(\cref{prop:g6:shapes:diagonals}). Rien n'oblige les diagonales à
être \omterm{def:g6:lines:perp}{perpendiculaires}, donc ce n'est pas forcément un \omterm{def:g6:shapes:quadrilaterals}{losange} ou un
carré --- un rectangle est la bonne réponse (un carré la
satisferait aussi, mais n'est pas forcé).
\end{solution}

\begin{solution}{exo:g6:shapes:12}
Avec $4 \times 4$ points, il y a $3 \times 3$ cellules de
quadrillage. Carrés alignés~: côtés $1$, $2$ ou $3$ donnent
$9 + 4 + 1 = 14$ carrés. Des carrés penchés existent aussi~: les
petits dont les côtés sont des diagonales de cellules $1 \times 1$
(comme $(1,0)$, $(2,1)$, $(1,2)$, $(0,1)$) --- quatre d'entre eux~;
et les plus grands dont les côtés sont des diagonales de rectangles
$2 \times 1$, comme $(1,0)$, $(3,1)$, $(2,3)$, $(0,2)$ --- deux
d'entre eux. Les quatre côtés d'un tel carré sont égaux parce qu'ils
sont des diagonales de rectangles identiques. Total~:
$14 + 4 + 2 = 20$ carrés. (Tout compte avec un exemple penché correct
et les $14$ alignés est une réussite à ce niveau.)
\end{solution}

\begin{solution}{pb:g6:shapes:1}
\textbf{1.} Les quatre côtés mesurent tous la même chose (environ
$4.2$ cm) et les quatre angles sont droits~: la figure est un
\emph{carré}. (Des diagonales égales donnent les \omterm{def:g3:shapes:rightangle}{angles droits},
comme dans le rectangle~; \omterm{def:g6:lines:perp}{perpendiculaires}, elles donnent les côtés
égaux, comme dans le \omterm{def:g6:shapes:quadrilaterals}{losange}~; les deux à la fois donnent le carré
--- c'est \cref{prop:g6:shapes:diagonals} lue à l'envers.)

\textbf{2.} Un \emph{\omterm{def:g6:shapes:quadrilaterals}{losange}}~: quatre côtés égaux (environ
$4.7$ cm), mais aucun \omterm{def:g3:shapes:rightangle}{angle droit} --- exactement la construction de
l'\cref{exo:g6:shapes:8}.

\textbf{3.} Les quatre angles sont droits, mais les côtés viennent
en deux longueurs différentes~: un \emph{rectangle} --- c'est la
situation de l'\cref{exo:g6:shapes:11}.

\textbf{4.} Aucune égalité de longueur ne résiste à la mesure et
aucun \omterm{def:g3:shapes:rightangle}{angle droit} n'apparaît~; en revanche les côtés opposés sont
égaux deux à deux et semblent \omterm{def:g6:lines:perp}{parallèles}. C'est le «~rectangle
penché~» général, dont les diagonales se coupent simplement en leur
milieu --- le \emph{parallélogramme}, nommé et étudié au
\cref{ch:g7:central}.

\textbf{5.} Diagonales se coupant en leur milieu commun~:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|l}
diagonales\dots & figure \\
\hline
égales et \omterm{def:g6:lines:perp}{perpendiculaires} & carré \\
\omterm{def:g6:lines:perp}{perpendiculaires} seulement & \omterm{def:g6:shapes:quadrilaterals}{losange} \\
égales seulement & rectangle \\
ni l'un ni l'autre & la figure sans nom (voir \cref{ch:g7:central}) \\
\end{tabular}
\end{center}

\textbf{6.} Chaque triangle de coin a un côté de l'\omterm{def:g3:shapes:rightangle}{angle droit} sur
chaque diagonale~: la \omterm{def:g3:division:half}{moitié} de $8$ cm et la \omterm{def:g3:division:half}{moitié} de $5$ cm, soit
des côtés de $4$ cm et $2.5$ cm, avec un \omterm{def:g3:shapes:rightangle}{angle droit} entre eux ---
les mêmes données pour les quatre. Des triangles construits sur les
deux mêmes côtés et l'\omterm{def:g3:shapes:rightangle}{angle droit} qu'ils forment sont des copies
exactes les uns des autres (les construire~: les écartements de
compas sont identiques). Leurs quatre hypoténuses --- les côtés de
la figure --- sont donc égales~: la figure est un \omterm{def:g6:shapes:quadrilaterals}{losange}.

\textbf{7.} Ouvrir le compas à la longueur d'un côté, puis le
promener autour de la figure~: si la pointe et la mine tombent
exactement sur chaque paire de \omterm{def:g5:solids:def}{sommets} voisins, les quatre côtés
valent cet écartement. Aucune règle n'est nécessaire.

\textbf{8.} Programme~:
\begin{enumerate}
  \item Tracer un \omterm{def:g6:lines:objects}{segment} $[AC]$ avec $AC = 6$ cm et placer son
        milieu $O$ (mesurer $3$ cm depuis $A$).
  \item Tracer la \omterm{def:g6:lines:perp}{perpendiculaire} à $(AC)$ passant par $O$.
  \item Y placer $B$ et $D$ à $3$ cm de $O$, un de chaque côté.
  \item Joindre $A$, $B$, $C$, $D$.
\end{enumerate}
Diagonales égales ($6$ cm), \omterm{def:g6:lines:perp}{perpendiculaires}, se coupant en leur
milieu~: d'après le dictionnaire, $ABCD$ est un carré.

\textbf{9.} La mesure donne environ $4.2$ cm~: \emph{plus} que
$3$ cm --- l'intuition courante «~la \omterm{def:g3:division:half}{moitié} de la diagonale~» est
fausse. La \omterm{def:g3:division:half}{moitié} démontrable~: pour aller du \omterm{def:g5:solids:def}{sommet} $A$ au \omterm{def:g5:solids:def}{sommet}
voisin $B$, on peut suivre le côté $[AB]$ en ligne droite, ou faire
le détour $A \to O \to B$ par le centre, qui coûte
$3 + 3 = 6$ cm~; le côté droit est plus court, donc il mesure moins
de $6$ cm. Qu'il dépasse aussi $3$ cm, c'est ce que montre la
règle~; le \cref{ch:g8:pythagoras} le prouvera et en donnera la
valeur exacte ($3$ cm multipliés par la racine carrée de $2$, soit
environ $4.24$ cm).

\textbf{10.} Les quatre triangles de coin en $H$ vont maintenant par
\emph{deux} paires~: deux de côtés $3$ cm et $2.5$ cm (hypoténuses
$[AB]$ et $[AD]$), deux de côtés $5$ cm et $2.5$ cm (hypoténuses
$[CB]$ et $[CD]$). Donc $AB = AD$ et $CB = CD$~: un \omterm{pb:g6:shapes:1}{cerf-volant} a
deux paires de côtés \emph{voisins} égaux (le \omterm{def:g6:shapes:quadrilaterals}{losange} est le
\omterm{pb:g6:shapes:1}{cerf-volant} particulier où les quatre coïncident).

\textbf{11.} \omterm{def:g4:geometry:polygon}{Quadrilatère}~: $2$ diagonales. \omterm{def:g4:geometry:polygon}{Pentagone}~: $5$.
\omterm{def:g4:geometry:polygon}{Hexagone}~: $9$.

\textbf{12.} D'un \omterm{def:g5:solids:def}{sommet} de l'\omterm{def:g4:geometry:polygon}{hexagone}, les diagonales vont vers
tous les \omterm{def:g5:solids:def}{sommets} sauf lui-même et ses deux voisins~: $6 - 3 = 3$
diagonales. Les six \omterm{def:g5:solids:def}{sommets} lancent $6 \times 3 = 18$ extrémités de
diagonales --- mais chaque diagonale a été comptée à ses \emph{deux}
extrémités, une fois depuis chaque bout, exactement comme une
poignée de main comptée par les deux personnes
(\cref{pb:g6:wholes:1}). Compte véritable~: $18 \div 2 = 9$. Cela
correspond au dessin.

\textbf{13.} Dix \omterm{def:g5:solids:def}{sommets}~: $10 - 3 = 7$ depuis chacun, donc
$10 \times 7 \div 2 = 35$ diagonales. Vingt \omterm{def:g5:solids:def}{sommets}~:
$20 \times 17 \div 2 = 170$ diagonales.

\textbf{14.} Essayons des candidats~: $10$ \omterm{def:g5:solids:def}{sommets} donnent $35$
(question 13)~; $11$ \omterm{def:g5:solids:def}{sommets} donnent $11 \times 8 \div 2 = 44$.
Trouvé~: le \omterm{def:g4:geometry:polygon}{polygone} a $11$ \omterm{def:g5:solids:def}{sommets}.

\textbf{15.} Un triangle n'a aucune diagonale~: les «~non-voisins~»
de chaque \omterm{def:g5:solids:def}{sommet} sont\,\dots\ personne, puisque les deux autres
\omterm{def:g5:solids:def}{sommets} sont tous deux ses voisins. La méthode est d'accord~:
$3 - 3 = 0$ depuis chaque \omterm{def:g5:solids:def}{sommet}, total $0$. Poignées de main et
diagonales sont le même comptage sous des habits différents~: des
paires choisies dans un groupe, chaque paire comptée une fois --- les
mathématiques recyclent leurs bonnes idées.
\end{solution}
