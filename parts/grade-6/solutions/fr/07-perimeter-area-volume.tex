\section*{Chapitre \ref{ch:g6:measure} --- Périmètre, aire, volume}

\begin{solution}{exo:g6:measure:1}
$3.5$~m $= 350$~cm~; \quad $420$~mm $= 42$~cm~; \quad
$0.8$~km $= 800$~m~; \quad $25\,000$~m $= 25$~km.
\end{solution}

\begin{solution}{exo:g6:measure:2}
Rectangle~: $2 \times (8 + 3.5) = 2 \times 11.5 = 23$~cm.
Carré~: $4 \times 6.2 = 24.8$~cm.
Triangle~: $5 + 7 + 9 = 21$~cm.
\end{solution}

\begin{solution}{exo:g6:measure:3}
Un tour~: $2\pi \times 50 = 100\pi \approx 314$~m. Pour $2$~km
$= 2000$~m~: $2000 \div 314 \approx 6.4$, donc $7$ tours complets
sont nécessaires ($6$ tours ne font qu'environ $1\,885$~m).
\end{solution}

\begin{solution}{exo:g6:measure:4}
Rectangle~: $9 \times 4 = 36$~cm$^2$. Carré~: $7^2 = 49$~m$^2$.
\omterm{def:g6:shapes:triangles}{Triangle rectangle}~: $\frac{6 \times 10}{2} = 30$~cm$^2$.
Disque~: $\pi \times 3^2 = 9\pi \approx 28$~cm$^2$.
\end{solution}

\begin{solution}{exo:g6:measure:5}
$3$~m$^2 = 30\,000$~cm$^2$~; \quad
$45\,000$~cm$^2 = 4.5$~m$^2$~; \quad
$2.5$~L $= 2\,500$~cm$^3$~; \quad
$4\,500$~L $= 4.5$~m$^3$.
\end{solution}

\begin{solution}{exo:g6:measure:6}
Clôture (\omterm{def:g6:measure:perimeter}{périmètre})~: $2 \times (120 + 85) = 2 \times 205 = 410$~m.
\omterm{def:g6:measure:area}{Aire}~: $120 \times 85 = 10\,200$~m$^2$.
\end{solution}

\begin{solution}{exo:g6:measure:7}
\omterm{def:g5:solids:def}{Pavé}~: $5 \times 4 \times 10 = 200$~cm$^3$. \omterm{def:g5:solids:def}{Cube}~:
$3 \times 3 \times 3 = 27$~cm$^3$.
\end{solution}

\begin{solution}{exo:g6:measure:8}
\omterm{def:g6:measure:perimeter}{Périmètre} $16$~cm signifie longueur $+$ largeur $= 8$~cm. Par
exemple $6 \times 2$ (\omterm{def:g6:measure:area}{aire} $12$~cm$^2$) et $5 \times 3$ (\omterm{def:g6:measure:area}{aire}
$15$~cm$^2$) --- ou le carré $4 \times 4$ (\omterm{def:g6:measure:area}{aire} $16$~cm$^2$). Plus
on se rapproche d'un carré, plus l'\omterm{def:g6:measure:area}{aire} est grande.
\end{solution}

\begin{solution}{exo:g6:measure:9}
\omterm{def:g6:measure:area}{Aire}~: $10 \times 2 + 2 \times 6 = 20 + 12 = 32$~cm$^2$.

\omterm{def:g6:measure:perimeter}{Périmètre} (tige centrée sous la barre)~: en parcourant le bord,
\[
10 + 2 + 4 + 6 + 2 + 6 + 4 + 2 = 36 \text{ cm}
\]
(haut~; extrémité droite de la barre~; dessous droit~; droite de la
tige~; bas de la tige~; gauche de la tige~; dessous gauche~;
extrémité gauche de la barre).
\end{solution}

\begin{solution}{exo:g6:measure:10}
\emph{1.} $V = 25 \times 10 \times 2 = 500$~m$^3$.

\emph{2.} $500$~m$^3 = 500 \times 1000 = 500\,000$~L.

\emph{3.} $500\,000 \div 50\,000 = 10$ heures.
\end{solution}

\begin{solution}{exo:g6:measure:11}
Jardin~: $20^2 = 400$~m$^2$. Étang~: $\pi \times 3^2 = 9\pi$~m$^2$.
Gazon~: $400 - 9\pi \approx 400 - 28.3 \approx 372$~m$^2$.
\end{solution}

\begin{solution}{exo:g6:measure:12}
\emph{1.} Petite tablette~: $15 \times 6 \times 1 = 90$~cm$^3$.
Grande~: $30 \times 12 \times 2 = 720$~cm$^3$. Le \omterm{def:g6:measure:volume}{volume} est
multiplié par $720 \div 90 = 8$ --- doubler chacune des trois
dimensions multiplie le \omterm{def:g6:measure:volume}{volume} par $2 \times 2 \times 2 = 8$.

\emph{2.} Huit fois le chocolat pour quatre fois le prix~: oui, la
grande tablette est deux fois plus avantageuse au gramme.
\end{solution}

\begin{solution}{pb:g6:measure:1}
\textbf{1.} $1 \times 9$~: \omterm{def:g6:measure:perimeter}{périmètre} $2 \times (1 + 9) = 20$ m, \omterm{def:g6:measure:area}{aire}
$9$ m$^2$. \ $2 \times 8$~: \omterm{def:g6:measure:perimeter}{périmètre} $2 \times 10 = 20$ m, \omterm{def:g6:measure:area}{aire}
$16$ m$^2$.

\textbf{2.} Le \omterm{def:g6:measure:perimeter}{périmètre} vaut deux fois (longueur $+$ largeur)
(\cref{ex:g6:measure:perimeter}), donc longueur $+$ largeur
$= 20 \div 2 = 10$ m. Le tableau~:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccc}
enclos & $1 \times 9$ & $2 \times 8$ & $3 \times 7$ & $4 \times 6$
& $5 \times 5$ \\
\hline
\omterm{def:g6:measure:area}{aire} (m$^2$) & $9$ & $16$ & $21$ & $24$ & $25$ \\
\end{tabular}
\end{center}
Le carré $5 \times 5$ est le meilleur.

\textbf{3.} $4.5 \times 5.5 = 24.75$ et $4.9 \times 5.1 = 24.99$~:
de plus en plus proche de $25$, mais toujours en dessous.
Conjecture~: le carré bat \emph{tous} les rectangles de \omterm{def:g6:measure:perimeter}{périmètre}
$20$, côtés décimaux compris.

\textbf{4.} D'après le tableau~: le \omterm{def:g2:mult:def}{produit} de deux nombres de \omterm{def:g1:addition:def}{somme}
$10$ est le plus grand quand les deux nombres sont \emph{égaux} ($5$
et $5$). Règle générale~: à \omterm{def:g1:addition:def}{somme} fixée, le \omterm{def:g2:mult:def}{produit} est le plus grand
pour des parts égales --- plus le couple est déséquilibré, plus le
\omterm{def:g2:mult:def}{produit} est petit.

\textbf{5.} Partir du carré $5 \times 5$ et retirer la rangée du
haut --- une bande de $5$ carreaux --- ce qui laisse un enclos
$5 \times 4$~; puis recoller une colonne de $4$ carreaux à une
extrémité, ce qui donne l'enclos $6 \times 4$. Cinq carreaux ont été
enlevés et quatre seulement sont revenus~: la bande ajoutée se pose
le long du côté $4$, plus court que le côté $5$ que couvrait la bande
retirée. Perte nette~: un mètre carré ($25 \to 24$). L'étape
suivante, de $6 \times 4$ à $7 \times 3$, échange une rangée de $6$
contre une colonne de $3$ et perd trois de plus ($24 \to 21$)~: plus
l'enclos s'éloigne du carré, plus longue est la bande qu'il cède et
plus courte celle qu'il récupère.

\textbf{6.} Avec $w + L + w = 20$, c'est-à-dire $L = 20 - 2w$~:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccccc}
$w$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ & $7$ \\
\hline
$L$ & $18$ & $16$ & $14$ & $12$ & $10$ & $8$ & $6$ \\
\omterm{def:g6:measure:area}{aire} & $18$ & $32$ & $42$ & $48$ & $50$ & $48$ & $42$ \\
\end{tabular}
\end{center}
Le gagnant est $5 \times 10$, d'\omterm{def:g6:measure:area}{aire} $50$ m$^2$ --- deux fois plus
long (le long du mur) que large~: ce n'est \emph{pas} un carré.

\textbf{7.} Construire le \omterm{def:g6:symmetry:def}{symétrique} de l'enclos par rapport au
mur~: l'enclos et son image forment un enclos doublé qui utilise la
clôture deux fois, soit $40$ m de clôture tout autour (le côté du mur
est devenu intérieur). D'après la partie I, le meilleur rectangle de
\omterm{def:g6:measure:perimeter}{périmètre} $40$ est le carré $10 \times 10$, d'\omterm{def:g6:measure:area}{aire} $100$ m$^2$. Le
meilleur enclos réel est la \omterm{def:g3:division:half}{moitié} de ce meilleur enclos doublé~:
$10$ m le long du mur, $5$ m de profondeur, \omterm{def:g6:measure:area}{aire} $50$ m$^2$ ---
exactement le gagnant du tableau, désormais expliqué.

\textbf{8.} Circonférence $20 = 2 \times \pi \times r$, donc
$r = 20 \div 6.28 \approx 3.18$ m. \omterm{def:g6:measure:area}{Aire}~:
$\pi r^2 \approx 3.14 \times 3.18 \times 3.18 \approx 31.8$ m$^2$
--- confortablement plus que les $25$ m$^2$ du carré. Didon savait ce
qu'elle faisait~: à longueur de bordure donnée, c'est le cercle qui
enferme le plus.

\textbf{9.} \omterm{def:g6:measure:perimeter}{Périmètres}~: $1 \times 36$~: $74$ m~; \ $2 \times 18$~:
$40$ m~; \ $3 \times 12$~: $30$ m~; \ $4 \times 9$~: $26$ m~; \
$6 \times 6$~: $24$ m. Le carré à nouveau --- le moins de clôture
pour l'\omterm{def:g6:measure:area}{aire} demandée. C'est la partie I à l'envers~: \omterm{def:g6:measure:perimeter}{périmètre} fixé
et \omterm{def:g6:measure:area}{aire} maximale, ou \omterm{def:g6:measure:area}{aire} fixée et \omterm{def:g6:measure:perimeter}{périmètre} minimal, les deux
couronnent le carré (et, au-delà de tous les rectangles, le cercle).

\textbf{10.} Une paroi ronde est la bordure la plus courte pour
l'espace qu'elle enferme (question 8)~: une boîte, un tuyau ou une
citerne ronds contiennent donc le plus pour le moins de métal --- et
la matière, c'est de l'argent.

\textbf{11.} Par exemple $50$ m $\times$ $0.01$ m~: \omterm{def:g6:measure:perimeter}{périmètre}
$2 \times (50 + 0.01) = 100.02$ m, \omterm{def:g6:measure:area}{aire} $50 \times 0.01 = 0.5$
m$^2$. Long et fin~: des kilomètres de bordure, presque pas de
terrain.

\textbf{12.} Faux. L'enclos de la question 11 a un \omterm{def:g6:measure:perimeter}{périmètre} de
$100.02$ m et une \omterm{def:g6:measure:area}{aire} de $0.5$ m$^2$, tandis que l'enclos
$5 \times 5$ a un \omterm{def:g6:measure:perimeter}{périmètre} de $20$ m et une \omterm{def:g6:measure:area}{aire} de $25$ m$^2$~:
\omterm{def:g6:measure:perimeter}{périmètre} plus grand, \omterm{def:g6:measure:area}{aire} bien plus petite.

\textbf{13.} \omterm{def:g6:measure:area}{Aire}~: il manque un carreau unité, $24 - 1 = 23$
cm$^2$. \omterm{def:g6:measure:perimeter}{Périmètre}~: en faisant le tour, l'encoche remplace $1$ cm de
paroi droite par trois côtés du petit carré, $1 + 1 + 1 = 3$ cm~: le
\omterm{def:g6:measure:perimeter}{périmètre} passe de $20$ cm à $20 - 1 + 3 = 22$ cm. Enlever de la
matière a \emph{allongé} la bordure.

\textbf{14.} Chaque zoom sur une côte réelle révèle de nouvelles
baies, de nouveaux rochers et de nouvelles criques --- des encoches
sur des encoches, et la question 13 montre que chaque encoche
allonge la bordure sans presque rien changer à l'\omterm{def:g6:measure:area}{aire}. La longueur
mesurée ne cesse donc de croître avec le niveau de détail (le
«~paradoxe du littoral~»), alors que l'\omterm{def:g6:measure:area}{aire} mesurée se stabilise~:
\omterm{def:g6:measure:perimeter}{périmètre} et \omterm{def:g6:measure:area}{aire} sont vraiment deux grandeurs indépendantes.

\textbf{15.} Argument du miroir~: l'enclos doublé utiliserait
$2 \times 36 = 72$ m de clôture, et le meilleur rectangle de
\omterm{def:g6:measure:perimeter}{périmètre} $72$ est le carré $18 \times 18$. Le meilleur enclos est
donc $18$ m le long du mur et $9$ m de profondeur~: vérification de
la clôture $9 + 18 + 9 = 36$ m, \omterm{def:g6:measure:area}{aire} $18 \times 9 = 162$ m$^2$. En
plein champ, le meilleur est le carré $9 \times 9$, d'\omterm{def:g6:measure:area}{aire}
$81$ m$^2$. Le mur de la grange \emph{double} exactement le terrain
du fermier.
\end{solution}
