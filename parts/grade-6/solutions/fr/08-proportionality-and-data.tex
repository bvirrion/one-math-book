\section*{Chapitre \ref{ch:g6:propdata} --- Proportionnalité et données}

\begin{solution}{exo:g6:propdata:1}
Pommes~: \omterm{def:g3:division:remainder}{quotients} $\frac52 = 2.5$, $\frac{7.5}{3} = 2.5$,
$\frac{12.5}{5} = 2.5$ --- tous égaux~: proportionnel (prix
$2.50$ par kg).

Âge/taille~: $\frac{85}{2} = 42.5$ mais $\frac{103}{4} = 25.75$ ---
pas égaux~: non proportionnel.
\end{solution}

\begin{solution}{exo:g6:propdata:2}
Un croissant~: $4.80 \div 4 = 1.20$ euros. Sept~:
$7 \times 1.20 = 8.40$ euros.
\end{solution}

\begin{solution}{exo:g6:propdata:3}
Coefficient~: $18 \div 10 = 1.8$ euros par litre. Puis
$25$~L coûtent $25 \times 1.8 = 45$ euros~; $72$ euros achètent
$72 \div 1.8 = 40$~L~; $60$~L coûtent $60 \times 1.8 = 108$ euros.
\end{solution}

\begin{solution}{exo:g6:propdata:4}
Pour $250$~km~: $2.5$ fois le carburant de $100$~km, donc
$2.5 \times 6 = 15$~L. Pour $350$~km~: $3.5 \times 6 = 21$~L. Avec
$27$~L~: $27 \div 6 = 4.5$ centaines de km, c'est-à-dire $450$~km.
\end{solution}

\begin{solution}{exo:g6:propdata:5}
$50\,\%$ de $84$~: \omterm{def:g3:division:half}{moitié}, $42$. \quad
$25\,\%$ de $200$~: un \omterm{def:g3:division:half}{quart}, $50$. \quad
$10\,\%$ de $63$~: un dixième, $6.3$. \quad
$30\,\%$ de $70$~: $3 \times 7 = 21$.
\end{solution}

\begin{solution}{exo:g6:propdata:6}
$60\,\%$ de $450$~: $\frac{60}{100} \times 450 = 270$ élèves mangent
à la cantine~; $450 - 270 = 180$ n'y mangent pas.
\end{solution}

\begin{solution}{exo:g6:propdata:7}
Moins de $10$ livres~: mardi ($8$) et jeudi ($6$). Mercredi moins
jeudi~: $17 - 6 = 11$ livres de plus.
\end{solution}

\begin{solution}{exo:g6:propdata:8}
Diagramme en barres avec cinq barres de hauteurs $14$, $16$, $13$,
$17$, $20$ (graduation tous les $2$ ou $5$ degrés convient). Jour le
plus chaud~: vendredi ($20$ degrés).
\end{solution}

\begin{solution}{exo:g6:propdata:9}
Pour $10$ personnes, multiplier par $\frac{10}{6} = \frac53$~: farine
$450 \times \frac{10}{6} = 750$~g. Œufs~:
$3 \times \frac{10}{6} = 5$ --- heureusement un entier. (Si ce
n'était pas le cas, on arrondirait \emph{vers le haut} pour en avoir
assez.)
\end{solution}

\begin{solution}{exo:g6:propdata:10}
\emph{1.} En $2$~h~: $48$~km. En une demi-heure~: $12$~km. En
$1$~h $30$~: $24 + 12 = 36$~km.

\emph{2.} $60 \div 24 = 2.5$ heures, c'est-à-dire $2$~h $30$~min.
\end{solution}

\begin{solution}{exo:g6:propdata:11}
Ballon~: réduction $20\,\%$ de $15 = 3$ euros, nouveau prix $12$
euros. Raquette~: réduction $8$ euros, nouveau prix $32$ euros. Le
nouveau prix est $80\,\%$ de l'ancien dans tous les cas~:
proportionnel, coefficient $0.8$.
\end{solution}

\begin{solution}{exo:g6:propdata:12}
Après $t$ heures, la grosse bougie a brûlé $\frac t6$ de sa hauteur~:
elle se dresse à $1 - \frac t6$ de la hauteur initiale~; la fine à
$1 - \frac t4$. On veut
\[
1 - \frac t4 = \frac12 \left(1 - \frac t6\right)
\quad\text{c'est-à-dire}\quad
1 - \frac t4 = \frac12 - \frac t{12}.
\]
Alors $\frac12 = \frac t4 - \frac t{12} = \frac{3t - t}{12} =
\frac{t}{6}$, donc $t = 3$ heures. Vérif.~: après $3$~h la grosse
bougie se dresse à $1 - \frac36 = \frac12$ et la fine à
$1 - \frac34 = \frac14$ --- bien la \omterm{def:g3:division:half}{moitié}.
\end{solution}

\begin{solution}{pb:g6:propdata:1}
\textbf{1.} $100\,000$ cm $= 1\,000$ m $= 1$ km~: un centimètre sur
la carte, c'est un kilomètre sur le terrain.

\textbf{2.} $7.5$ cm $\rightarrow 7.5$ km entre les villages~;
$12$ cm $\rightarrow 12$ km de rive.

\textbf{3.} $20$ km $\rightarrow 20$ cm sur la carte.

\textbf{4.} $5$ km $= 500\,000$ cm, donc l'échelle est
$1:500\,000$. La carte de randonnée ($1:100\,000$) est la plus
détaillée~: elle emploie $5$ cm de papier là où l'autre n'en dépense
qu'$1$.

\textbf{5.}
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccc}
carte (cm) & $1$ & $2$ & $7.5$ \\
\hline
réel (km) & $1$ & $2$ & $7.5$ \\
\end{tabular}
\end{center}
Distance réelle $=$ distance sur la carte $\times 1$ (dans ces
unités)~: toujours le \emph{même} multiplicateur, ce qui est
exactement la définition de la \omterm{def:g6:propdata:def}{proportionnalité}
(\cref{def:g6:propdata:def}). Le coefficient vaut ici $1$ kilomètre
par centimètre.

\textbf{6.} $4$ m $= 400$ cm et $400 \div 50 = 8$~;
$3$ m $= 300$ cm et $300 \div 50 = 6$~: le plan montre un rectangle
de $8$ cm $\times$ $6$ cm.

\textbf{7.} Lit~: $200 \div 50 = 4$ cm sur $90 \div 50 = 1.8$ cm.
Ouverture de la porte~: $80 \div 50 = 1.6$ cm.

\textbf{8.} \omterm{def:g6:measure:area}{Aire} réelle~: $4 \times 3 = 12$ m$^2$. \omterm{def:g6:measure:area}{Aire} du plan~:
$8 \times 6 = 48$ cm$^2$. En convertissant~:
$12$ m$^2 = 12 \times 10\,000 = 120\,000$ cm$^2$, et
\[
120\,000 \div 48 = 2\,500 .
\]
Le \omterm{def:g3:division:remainder}{quotient} ne vaut pas $50$ mais $2\,500 = 50 \times 50$.

\textbf{9.} Un cm$^2$ de papier représente un carré réel de $50$ cm
sur $50$ cm, qui contient $50 \times 50 = 2\,500$ cm$^2$. Ainsi,
quand les longueurs sont divisées par $50$, les \omterm{def:g6:measure:area}{aires} sont divisées
par $50 \times 50 = 2\,500$~: les \omterm{def:g6:measure:area}{aires} suivent le \emph{carré} de
l'échelle.

\textbf{10.} $6 \times 2\,500 = 15\,000$ cm$^2$, et
$15\,000 \div 10\,000 = 1.5$ m$^2$.

\textbf{11.} L'image suggère que le dimanche a vendu \emph{deux
fois} plus (une barre deux fois plus haute). La vérité~: la hausse
est de $5$ euros sur $105$, et $5 \div 105 \approx 0.048$~: environ
$5\,\%$ de plus, et non $100\,\%$ de plus. Le gérant a ignoré la
mise en garde qui demande de vérifier que l'axe part de $0$
(\cref{met:g6:propdata:read}).

\textbf{12.} Avec un axe partant de $0$, les barres montent à $105$
et $110$ petites unités~: deux barres de hauteurs presque égales,
l'image honnête d'un écart de $5\,\%$.

\textbf{13.} À la hausse~: l'augmentation est de $20$ timbres à
partir de $40$~: $\frac{20}{40} = 50\,\%$. À la baisse~: la
diminution est de $20$ timbres à partir de $60$~:
$\frac{20}{60} = \frac13 \approx 33\,\%$. Les mêmes $20$ timbres,
mais des valeurs de départ différentes --- un pourcentage est
toujours une \omterm{def:g6:fractions:def}{fraction} \emph{de quelque chose}, et ce «~quelque
chose~» a changé.

\textbf{14.} Après la remise de $20\,\%$~: $100 - 20 = 80$ euros.
Les $10\,\%$ supplémentaires s'appliquent à $80$~: remise de $8$
euros, prix final $72$ euros. Remise totale~: $28$ euros sur $100$,
soit $28\,\%$ --- et non $30\,\%$. La seconde remise a agi sur le
prix déjà soldé, donc ses euros sont plus petits~: des pourcentages
de quantités différentes ne s'additionnent pas.

\textbf{15.} Sur cette carte, $1$ cm représente $5$ km, donc $1$
cm$^2$ représente $5 \times 5 = 25$ km$^2$ (règle de la question 9).
La forêt~: $3 \times 25 = 75$ km$^2$. Les longueurs suivent
l'échelle~; les \omterm{def:g6:measure:area}{aires} suivent son \emph{carré}.
\end{solution}
