\section*{الفصل \ref{ch:g6:measure} --- المحيط والمساحة والحجم}

\begin{solution}{exo:g6:measure:1}
$3.5$ م $= 350$ سم؛ \quad $420$ مم $= 42$ سم؛ \quad
$0.8$ كم $= 800$ م؛ \quad $25\,000$ م $= 25$ كم.
\end{solution}

\begin{solution}{exo:g6:measure:2}
المستطيل: $2 \times (8 + 3.5) = 2 \times 11.5 = 23$ سم.
والمربع: $4 \times 6.2 = 24.8$ سم.
والمثلث: $5 + 7 + 9 = 21$ سم.
\end{solution}

\begin{solution}{exo:g6:measure:3}
اللفّة الواحدة: $2\pi \times 50 = 100\pi \approx 314$ م. ومن أجل $2$ كم
$= 2000$ م: $2000 \div 314 \approx 6.4$، إذن تلزم $7$ لفّات كاملة
(و $6$ لفّات لا تصنع إلا نحو $1\,885$ م).
\end{solution}

\begin{solution}{exo:g6:measure:4}
المستطيل: $9 \times 4 = 36$ سم$^2$. والمربع: $7^2 = 49$ م$^2$.
\omterm{def:g6:shapes:triangles}{والمثلث القائم الزاوية}: $\frac{6 \times 10}{2} = 30$ سم$^2$.
والقرص: $\pi \times 3^2 = 9\pi \approx 28$ سم$^2$.
\end{solution}

\begin{solution}{exo:g6:measure:5}
$3$ م$^2 = 30\,000$ سم$^2$؛ \quad
$45\,000$ سم$^2 = 4.5$ م$^2$؛ \quad
$2.5$ لتر $= 2\,500$ سم$^3$؛ \quad
$4\,500$ لتر $= 4.5$ م$^3$.
\end{solution}

\begin{solution}{exo:g6:measure:6}
السياج (\omterm{def:g6:measure:perimeter}{المحيط}): $2 \times (120 + 85) = 2 \times 205 = 410$ م.
\omterm{def:g6:measure:area}{والمساحة}: $120 \times 85 = 10\,200$ م$^2$.
\end{solution}

\begin{solution}{exo:g6:measure:7}
الصندوق: $5 \times 4 \times 10 = 200$ سم$^3$. \omterm{def:g5:solids:def}{والمكعب}:
$3 \times 3 \times 3 = 27$ سم$^3$.
\end{solution}

\begin{solution}{exo:g6:measure:8}
\omterm{def:g6:measure:perimeter}{المحيط} $16$ سم يعني أن الطول $+$ العرض $= 8$ سم. فمثلًا
$6 \times 2$ (مساحته $12$ سم$^2$) و $5 \times 3$ (مساحته $15$ سم$^2$)
--- أو المربع $4 \times 4$ (مساحته $16$ سم$^2$). وكلما اِقترب من
المربع كبرت \omterm{def:g6:measure:area}{المساحة}.
\end{solution}

\begin{solution}{exo:g6:measure:9}
\omterm{def:g6:measure:area}{المساحة}: $10 \times 2 + 2 \times 6 = 20 + 12 = 32$ سم$^2$.

\omterm{def:g6:measure:perimeter}{والمحيط} (والساق في وسط العارضة): بالدوران حوله،
\[
10 + 2 + 4 + 6 + 2 + 6 + 4 + 2 = 36 \text{ سم}
\]
(الأعلى؛ وطرف العارضة الأيمن؛ والوجه السفلي الأيمن؛ ويمين الساق؛ وأسفل
الساق؛ ويسار الساق؛ والوجه السفلي الأيسر؛ وطرف العارضة الأيسر).
\end{solution}

\begin{solution}{exo:g6:measure:10}
\emph{1.} $V = 25 \times 10 \times 2 = 500$ م$^3$.

\emph{2.} $500$ م$^3 = 500 \times 1000 = 500\,000$ لتر.

\emph{3.} $500\,000 \div 50\,000 = 10$ ساعات.
\end{solution}

\begin{solution}{exo:g6:measure:11}
الحديقة: $20^2 = 400$ م$^2$. والبركة: $\pi \times 3^2 = 9\pi$ م$^2$.
والعشب: $400 - 9\pi \approx 400 - 28.3 \approx 372$ م$^2$.
\end{solution}

\begin{solution}{exo:g6:measure:12}
\emph{1.} اللوح الصغير: $15 \times 6 \times 1 = 90$ سم$^3$. واللوح الكبير:
$30 \times 12 \times 2 = 720$ سم$^3$. \omterm{def:g6:measure:volume}{فالحجم} مضروب في
$720 \div 90 = 8$ --- ومضاعفة كل بُعد من الأبعاد الثلاثة تضرب
\omterm{def:g6:measure:volume}{الحجم} في $2 \times 2 \times 2 = 8$.

\emph{2.} ثماني مرات الشوكولاتة بأربع مرات الثمن: نعم،
اللوح الكبير صفقة أفضل بمرتين للغرام الواحد.
\end{solution}

\begin{solution}{pb:g6:measure:1}
\textbf{1.} الحظيرة $1 \times 9$: محيطها $2 \times (1 + 9) = 20$ م،
ومساحتها $9$ م$^2$. \ والحظيرة $2 \times 8$: محيطها $2 \times 10 = 20$ م،
ومساحتها $16$ م$^2$.

\textbf{2.} \omterm{def:g6:measure:perimeter}{المحيط} ضعف (الطول $+$ العرض)
(\cref{ex:g6:measure:perimeter})، إذن الطول $+$ العرض
$= 20 \div 2 = 10$ م. والجدول:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccc}
الحظيرة & $1 \times 9$ & $2 \times 8$ & $3 \times 7$ & $4 \times 6$
& $5 \times 5$ \\
\hline
\omterm{def:g6:measure:area}{المساحة} (م$^2$) & $9$ & $16$ & $21$ & $24$ & $25$ \\
\end{tabular}
\end{center}
والمربع $5 \times 5$ هو الأفضل.

\textbf{3.} $4.5 \times 5.5 = 24.75$ و $4.9 \times 5.1 =
24.99$: أقرب فأقرب إلى $25$، لكن دونه دائمًا. والتخمين:
المربع يغلب \emph{كل} مستطيل محيطه $20$،
بما في ذلك الأضلاع العشرية.

\textbf{4.} من الجدول: \omterm{def:g2:mult:def}{جداء} عددين مجموعهما
$10$ يكون أكبر ما يكون حين يتساوى العددان ($5$ و
$5$). والقاعدة العامّة: من أجل \omterm{def:g1:addition:def}{مجموع} ثابت، يكون \omterm{def:g2:mult:def}{الجداء} أكبر ما يكون
عند تساوي الجزأين --- وكلما تفاوت الزوجان صغُر
\omterm{def:g2:mult:def}{الجداء}.

\textbf{5.} من المربع $5 \times 5$، اِنزع السطر الأعلى
--- وهو شريط من $5$ مربعات وحدة --- فتبقى حظيرة $5 \times 4$؛
ثم اِلصق عمودًا من $4$ مربعات وحدة في أحد الطرفين، فتصير
الحظيرة $6 \times 4$. فقد أُخذت خمسة مربعات ولم يعُد
إلا أربعة: فالشريط المضاف يقع على طول الضلع $4$، وهو
أقصر من الضلع $5$ الذي كان عليه الشريط المنزوع. والخسارة الصافية:
متر مربع واحد ($25 \to 24$). والخطوة التالية، من
$6 \times 4$ إلى $7 \times 3$، تبادل سطرًا من $6$ بعمود
من $3$ وتخسر ثلاثة أخرى ($24 \to 21$): فكلما اِبتعدت الحظيرة
عن المربع طال الشريط الذي تتخلّى عنه وقصُر
الذي تستعيده.

\textbf{6.} مع $w + L + w = 20$، أي\ $L = 20 - 2w$:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c|ccccccc}
$w$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ & $7$ \\
\hline
$L$ & $18$ & $16$ & $14$ & $12$ & $10$ & $8$ & $6$ \\
\omterm{def:g6:measure:area}{المساحة} & $18$ & $32$ & $42$ & $48$ & $50$ & $48$ & $42$ \\
\end{tabular}
\end{center}
والفائزة هي $5 \times 10$، ومساحتها $50$ م$^2$ --- وطولها
(على طول الجدار) ضعف عرضها: \emph{وليست} مربعًا.

\textbf{7.} اِعكس الحظيرة بالنسبة إلى الجدار: فالحظيرة مع صورتها
المرآتية تكوّنان حظيرة مضاعفة تستعمل السياج مرتين، أي $40$ م من
السياج حول كل شيء (فضلع الجدار صار داخليًّا). وحسب الجزء الأول،
أفضل مستطيل محيطه $40$ هو المربع $10 \times
10$، ومساحته $100$ م$^2$. وأفضل حظيرة حقيقية هي نصف تلك
الحظيرة المضاعفة الأفضل: أي $10$ م على طول الجدار، و $5$ م عمقًا، ومساحتها
$50$ م$^2$ --- وهي بالضبط فائزة الجدول، وقد فُسّرت الآن.

\textbf{8.} طول الدائرة $20 = 2 \times \pi \times r$، إذن
$r = 20 \div 6.28 \approx 3.18$ م. \omterm{def:g6:measure:area}{والمساحة}:
$\pi r^2 \approx 3.14 \times 3.18 \times 3.18 \approx
31.8$ م$^2$ --- أي أكثر بكثير من \omterm{def:g6:measure:area}{مساحة} المربع $25$ م$^2$.
وكانت ديدو تعرف ما تفعل: فمن أجل طول حدّ معطى،
تحيط الدائرة بالأكبر.

\textbf{9.} المحيطات: $1 \times 36$: $74$ م؛ \ و $2 \times 18$:
$40$ م؛ \ و $3 \times 12$: $30$ م؛ \ و $4 \times 9$: $26$ م؛ \
و $6 \times 6$: $24$ م. والمربع من جديد --- أقلّ سياج من أجل
\omterm{def:g6:measure:area}{المساحة} المعطاة. وهو الجزء الأول بالمقلوب: \omterm{def:g6:measure:perimeter}{محيط} ثابت وأكبر
\omterm{def:g6:measure:area}{مساحة}، أو \omterm{def:g6:measure:area}{مساحة} ثابتة وأصغر \omterm{def:g6:measure:perimeter}{محيط}: كلاهما يتوّج المربع
(وفوق كل المستطيلات، الدائرة).

\textbf{10.} الجدار المدوّر هو أقصر حدّ للحيز
الذي يحيط به (السؤال 8)، إذن العلبة أو الأنبوب أو الخزان المدوّر يسع
أكبر محتوى بأقلّ معدن --- والمادة مال.

\textbf{11.} فمثلًا $50$ م $\times$ $0.01$ م: \omterm{def:g6:measure:perimeter}{المحيط}
$2 \times (50 + 0.01) = 100.02$ م، \omterm{def:g6:measure:area}{والمساحة}
$50 \times 0.01 = 0.5$ م$^2$. طويل ورفيع: كيلومترات من
الحدّ، ولا تكاد توجد أرض.

\textbf{12.} خطأ. فحظيرة السؤال 11 محيطها
$100.02$ م ومساحتها $0.5$ م$^2$، بينما الحظيرة $5 \times 5$
محيطها $20$ م ومساحتها $25$ م$^2$: \omterm{def:g6:measure:perimeter}{محيط} أكبر، \omterm{def:g6:measure:area}{ومساحة} أصغر
بكثير.

\textbf{13.} \omterm{def:g6:measure:area}{المساحة}: ينقص مربع وحدة واحد،
أي $24 - 1 = 23$ سم$^2$. \omterm{def:g6:measure:perimeter}{والمحيط}: بالدوران حوله، تعوّض الثلمة
$1$ سم من الجدار \omterm{def:g6:lines:objects}{المستقيم} بثلاثة أضلاع للمربع
الصغير، أي $1 + 1 + 1 = 3$ سم: فينمو \omterm{def:g6:measure:perimeter}{المحيط} من $20$ سم إلى
$20 - 1 + 3 = 22$ سم. فنزع المادة \emph{أطال}
الحدّ.

\textbf{14.} كل تكبير على ساحل حقيقي يكشف خلجانًا وصخورًا
وجداول جديدة --- ثلمات فوق ثلمات، والسؤال 13 يبيّن أن
كل ثلمة تزيد طول الحدّ دون أن تغيّر \omterm{def:g6:measure:area}{المساحة} تقريبًا.
فيظلّ الطول المقيس ينمو مع درجة التفصيل
(«مفارقة الساحل»)، بينما تستقرّ \omterm{def:g6:measure:area}{المساحة}
المقيسة: \omterm{def:g6:measure:perimeter}{فالمحيط} \omterm{def:g6:measure:area}{والمساحة} مقداران مستقلان حقًّا.

\textbf{15.} حجّة المرآة: الحظيرة المضاعفة تستعمل
$2 \times 36 = 72$ م من السياج، وأفضل مستطيل
محيطه $72$ هو المربع $18 \times 18$. إذن أفضل حظيرة
$18$ م على طول الجدار و $9$ م عمقًا: والتحقّق من السياج
$9 + 18 + 9 = 36$ م، \omterm{def:g6:measure:area}{والمساحة} $18 \times 9 = 162$ م$^2$. وفي أرض
مكشوفة، الأفضل هو المربع $9 \times 9$، ومساحته $81$ م$^2$.
فجدار المخزن \emph{يضاعف} أرض الفلاح بالضبط.
\end{solution}
