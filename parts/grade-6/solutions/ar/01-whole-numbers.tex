\begin{solution}{exo:g6:wholes:7}
$95 = 7 \times 13 + 4$ (\omterm{def:g3:division:remainder}{خارج القسمة} $13$، \omterm{def:g3:division:remainder}{والباقي} $4$).

$230 = 6 \times 38 + 2$ (\omterm{def:g3:division:remainder}{خارج القسمة} $38$، \omterm{def:g3:division:remainder}{والباقي} $2$).

$504 = 8 \times 63 + 0$ (\omterm{def:g3:division:remainder}{خارج القسمة} $63$، \omterm{def:g3:division:remainder}{والباقي} $0$: فالعدد $504$

\begin{solution}{exo:g6:wholes:8}
المساواة صحيحة، لكن \omterm{def:g3:division:remainder}{الباقي} $13$ \emph{ليس} أصغر من
القاسم $8$، فليست هي القسمة الإقليدية. أخرِج من \omterm{def:g3:division:remainder}{الباقي}
$8$ أخرى: $173 = 8 \times 21 + 5$، مع $0 \leq 5 < 8$:
\omterm{def:g3:division:remainder}{فخارج القسمة} $21$، \omterm{def:g3:division:remainder}{والباقي} $5$.
\end{solution}

\begin{solution}{exo:g6:wholes:9}
\emph{1.} $2\,000 = 12 \times 166 + 8$: تملأ المزرعة $166$ علبة
كاملة وتبقى $8$ بيضات.

\emph{2.} $150 = 12 \times 12 + 6$: اثنتا عشرة علبة لا تسع إلا $144$

\end{solution}

\begin{solution}{exo:g6:wholes:10}

\end{solution}

\begin{solution}{exo:g6:wholes:11}
أكبر باقٍ ممكن في القسمة على $9$ هو $8$ (\omterm{def:g3:division:remainder}{فالباقي}
يجب أن يكون أصغر من القاسم). إذن المقسوم هو
\[
9 \times 56 + 8 = 504 + 8 = 512 .
\]
\end{solution}

\begin{solution}{exo:g6:wholes:12}
لتكن الأرقام $h$ (المئات) و $0$ (العشرات) و $u$ (الآحاد). نعلم أن
$h = 2 \times u$ و $h + 0 + u = 9$، إذن $2u + u = 9$، فيكون
$3u = 9$ و $u = 3$ و $h = 6$. فالعدد هو $603$. والتحقّق: $6$ ضعف
$3$، ورقم العشرات $0$، و $6 + 0 + 3 = 9$.
\end{solution}

\begin{solution}{pb:g6:wholes:1}
\textbf{1.} مثلًا $(1 + 4) + (2 + 3) + 5 = 5 + 5 + 5 =
15$.

\textbf{2.} كل زوج يساوي $11$: $1 + 10$ و $2 + 9$ و $3 + 8$
و $4 + 7$ و $5 + 6$. وذلك $5$ أزواج (عشرة أعداد، اثنان لكل زوج)،
إذن \omterm{def:g1:addition:def}{المجموع} $5 \times 11 = 55$.

\textbf{3.} أزواج تساوي $21$ ($1 + 20$ و $2 + 19, \dots$)، و
عددها $10$: $10 \times 21 = 210$.

\textbf{4.} في كل زوج، حين يكبر العدد الأول بمقدار $1$ يصغر
الثاني بمقدار $1$، فلا يتغير \omterm{def:g1:addition:def}{المجموع} أبدًا:
$1 + 100 = 2 + 99 = 3 + 98 = \dots = 101$. والأعداد المئة
تصنع $50$ زوجًا (عددان لكل زوج)، وأوسطها
$50 + 51 = 101$.

\textbf{5.} $50$ زوجًا قيمة كل منها $101$:
\[
1 + 2 + \dots + 100 = 50 \times 101 = 5\,050 .
\]

\textbf{6.} مع تسعة أعداد، يبقى الأوسط، $5$، بلا
شريك. وحوله: $1 + 9 = 2 + 8 = 3 + 7 = 4 + 6 = 10$، أي أربعة
أزواج. \omterm{def:g1:addition:def}{والمجموع}: $4 \times 10 + 5 = 45$.

\textbf{7.} كل عمود يساوي $1 + 9 = 2 + 8 = \dots =
9 + 1 = 10$، والأعمدة $9$: فالسطران معًا
يساويان $9 \times 10 = 90$. لكن السطرين هما \omterm{def:g1:addition:def}{المجموع}
\emph{نفسه} مكتوبًا مرتين، فالنسخة الواحدة تساوي النصف:
$1 + 2 + \dots + 9 = 90 \div 2 = 45$. ولا يبقى عدد قطّ
بلا شريك --- فشريكه تحته مباشرة.

\textbf{8.} من $1$ إلى $50$: $50$ عمودًا قيمة كل منها $51$، \omterm{def:g1:addition:def}{والمجموع}
\omterm{def:g5:division:multiple}{المضاعف} $50 \times 51 = 2\,550$، إذن \omterm{def:g1:addition:def}{المجموع} $2\,550 \div 2 = 1\,275$.
ومن $1$ إلى $200$: $200$ عمود قيمة كل منها $201$، \omterm{def:g1:addition:def}{والمجموع} \omterm{def:g5:division:multiple}{المضاعف}
$40\,200$، \omterm{def:g1:addition:def}{والمجموع} $20\,100$.

\textbf{9.} كل \omterm{def:g3:numbers:evenodd}{عدد زوجي} ضعف عدد من
\omterm{def:g1:addition:def}{مجموع} غاوس: $2 = 2 \times 1$ و $4 = 2 \times 2$، \dots،
و $200 = 2 \times 100$. إذن \omterm{def:g1:addition:def}{المجموع} كله ضعف \omterm{def:g1:addition:def}{مجموع} غاوس:
\[
2 + 4 + \dots + 200 = 2 \times 5\,050 = 10\,100 .
\]

\textbf{10.} الأعداد من $101$ إلى $200$ هي الأعداد من
$1$ إلى $200$ بعد حذف الأعداد من $1$ إلى $100$:
\[
101 + 102 + \dots + 200 = 20\,100 - 5\,050 = 15\,050 .
\]

\textbf{11.} صُفّ الأصدقاء. الأول يصافح $7$ أيدٍ (يدًا
لكل واحد من الآخرين). والثاني قد سلّم على الأول من
قبل، فيصافح $6$ أيدٍ \emph{جديدة}؛ والثالث $5$؛ وهكذا،
فالسابع يصافح $1$ والثامن لا يصافح يدًا جديدة.
فكل مصافحة تُعدّ مرة واحدة بالضبط:
\[
7 + 6 + 5 + 4 + 3 + 2 + 1 = 28 \text{ مصافحة}
\]
(زوِّج حول الوسط: $(7+1) + (6+2) + (5+3) + 4 =
8 + 8 + 8 + 4 = 28$).

\textbf{12.} $1 + 2 + \dots + 12$: اثنا عشر عمودًا قيمة كل منها $13$،
\omterm{def:g1:addition:def}{والمجموع} \omterm{def:g5:division:multiple}{المضاعف} $12 \times 13 = 156$، إذن
$156 \div 2 = 78$ علبة.

\textbf{13.} سلّم \omterm{def:g1:addition:def}{المجموع} $1 + 2 + 3 + 4$ من المربعات، مع
السلّم نفسه مقلوبًا، يملآن مستطيلًا من $4$ صفوف و $5$
أعمدة تمامًا: فكل صفّ من المستطيل هو عمود من
حيلة السطرين (درجة من السلّم الصاعد تكمّلها
درجة من السلّم الهابط، $1 + 4$ و $2 + 3$ و $3 + 2$ و $4 + 1$ ---
وكلها $5$). إذن ضعف \omterm{def:g1:addition:def}{المجموع} $4 \times 5 = 20$ مربعًا، و
$1 + 2 + 3 + 4 = 10$: فالصورة \emph{هي} السؤال 7.

\textbf{14.} $1 + 2 + \dots + 12 = 78$ دقّة في اثنتي عشرة ساعة
(السؤال 12)، إذن $78 \times 2 = 156$ دقّة في اليوم كله.

\textbf{15.} اِقسم على $60$: $5\,050 = 60 \times 84 + 10$، إذن
$5\,050$ ثانية هي $84$ دقيقة و $10$ ثوانٍ. ثم اِقسم
الدقائق على $60$: $84 = 60 \times 1 + 24$، إذن $84$ دقيقة هي
$1$ ساعة و $24$ دقيقة. وفي المجموع: $1$ سا و $24$ د و $10$ ثا --- أي
\emph{أقلّ} قليلًا من ساعة ونصف ($1$ سا و $30$ د).
\end{solution}
