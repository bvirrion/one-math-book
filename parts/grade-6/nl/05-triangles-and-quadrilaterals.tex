\chapter{Driehoeken en vierhoeken}\label{ch:g6:shapes}

Driehoeken en \omterm{def:g4:geometry:polygon}{vierhoeken} zijn de basisfiguren van de meetkunde. Het doel van
dit hoofdstuk is hun bijzondere families --- gelijkbenige, gelijkzijdige en
\omterm{def:g6:shapes:triangles}{rechthoekige driehoeken}; rechthoeken, \omterm{def:g6:shapes:quadrilaterals}{ruiten}, vierkanten --- te kennen aan hun
\emph{definiërende eigenschappen}, en ze precies te construeren met lat en
passer.

\section{Driehoeken}

\begin{definition}[Bijzondere driehoeken]\label{def:g6:shapes:triangles}
Een driehoek $ABC$ heeft drie \omterm{def:g5:solids:def}{hoekpunten} en drie zijden. Hij is:
\begin{itemize}
  \item \emph{gelijkbenig}\index{gelijkbenige driehoek} in $A$ als
        $AB = AC$ (twee gelijke zijden);
  \item \emph{gelijkzijdig}\index{gelijkzijdige driehoek} als
        $AB = BC = CA$ (drie gelijke zijden);
  \item \emph{rechthoekig}\index{rechthoekige driehoek} in $A$ als de hoek
        $\widehat{BAC}$ een \omterm{def:g3:shapes:rightangle}{rechte hoek} is. De zijde $[BC]$ tegenover de
        \omterm{def:g3:shapes:rightangle}{rechte hoek} heet de \emph{schuine zijde}.
\end{itemize}
\end{definition}

\begin{method}[Een driehoek uit zijn drie zijden construeren]\label{met:g6:shapes:construct}
Om $ABC$ te construeren met $AB = 5$ cm, $AC = 4$ cm en $BC = 3$ cm:
\begin{enumerate}
  \item teken met de lat het \omterm{def:g6:lines:objects}{lijnstuk} $[AB]$ van $5$ cm;
  \item teken een boog van de cirkel met middelpunt $A$ en \omterm{def:g3:shapes:shapes}{radius} $4$ cm
        (alle punten op $4$ cm van $A$);
  \item teken een boog van de cirkel met middelpunt $B$ en \omterm{def:g3:shapes:shapes}{radius} $3$ cm;
  \item het punt $C$ is waar de twee bogen elkaar snijden; verbind het met
        $A$ en $B$.
\end{enumerate}
\end{method}

\section{Vierhoeken}

\begin{definition}[Bijzondere vierhoeken]\label{def:g6:shapes:quadrilaterals}
Een \omterm{def:g4:geometry:polygon}{vierhoek} heeft vier \omterm{def:g5:solids:def}{hoekpunten} en vier zijden. Hij is een:
\begin{itemize}
  \item \emph{rechthoek}\index{rechthoek} als zijn vier hoeken \omterm{def:g3:shapes:rightangle}{rechte hoeken}
        zijn;
  \item \emph{ruit}\index{ruit} als zijn vier zijden dezelfde lengte hebben;
  \item \emph{vierkant}\index{vierkant} als hij beide is: vier \omterm{def:g3:shapes:rightangle}{rechte hoeken}
        \emph{en} vier gelijke zijden.
\end{itemize}
\end{definition}

\begin{proposition}[Eigenschappen van de diagonalen]\label{prop:g6:shapes:diagonals}
In deze \omterm{def:g4:geometry:polygon}{vierhoeken} snijden de twee diagonalen elkaar in hun gemeenschappelijke
midden, en bovendien:
\begin{itemize}
  \item in een rechthoek zijn de diagonalen \emph{even lang};
  \item in een \omterm{def:g6:shapes:quadrilaterals}{ruit} staan de diagonalen \emph{\omterm{def:g6:lines:perp}{loodrecht}} op elkaar;
  \item in een vierkant beide tegelijk.
\end{itemize}
\end{proposition}
\admitted

\begin{remark}
Een vierkant is automatisch een rechthoek (het heeft vier \omterm{def:g3:shapes:rightangle}{rechte hoeken}) en
automatisch een \omterm{def:g6:shapes:quadrilaterals}{ruit} (vier gelijke zijden). ``Rechthoek'' betekent dus niet
``geen vierkant''! In de wiskunde hoort de bijzonderdere figuur bij de meer
algemene familie.
\end{remark}

\begin{example}[Een figuur herkennen aan haar eigenschappen]\label{ex:g6:shapes:recognize}
Een \omterm{def:g4:geometry:polygon}{vierhoek} heeft vier zijden van $4$ cm en één \omterm{def:g3:shapes:rightangle}{rechte hoek}. Wat is het? Vier
gelijke zijden maken er een \omterm{def:g6:shapes:quadrilaterals}{ruit} van. In een \omterm{def:g6:shapes:quadrilaterals}{ruit} zijn de hoeken die tegenover
elkaar liggen gelijk, en samen zijn de vier hoeken $360^\circ$; één \omterm{def:g3:shapes:rightangle}{rechte hoek}
dwingt dan alle vier recht te zijn (dat wordt met parallellogrammen bewezen in
\cref{ch:g7:central}). De figuur is dus een vierkant.
\end{example}

\begin{method}[Tekenprogramma's]\label{met:g6:shapes:program}
Een \emph{tekenprogramma} is een lijst instructies die iedereen kan volgen om
een figuur precies na te maken. Schrijf één instructie per regel, geef elk punt
een naam op het moment dat het ontstaat, en gebruik alleen: ``teken het
\omterm{def:g6:lines:objects}{lijnstuk} / de \omterm{def:g6:lines:objects}{lijn} / de cirkel\dots'', ``zet het punt waar \dots\ elkaar
snijden''. Test je programma door het letterlijk, woord voor woord, te volgen.
\end{method}

\begin{example}\label{ex:g6:shapes:program}
Een programma voor een vierkant $ABCD$ met zijde $4$ cm:
\begin{enumerate}
  \item Teken een \omterm{def:g6:lines:objects}{lijnstuk} $[AB]$ met $AB = 4$ cm.
  \item Teken de loodlijn op $(AB)$ in $A$; zet daarop $D$ op $4$ cm van $A$
        (kies één kant van de \omterm{def:g6:lines:objects}{lijn} en houd die aan).
  \item Teken de loodlijn op $(AB)$ in $B$; zet daarop $C$ op $4$ cm van $B$,
        aan dezelfde kant als $D$.
  \item Verbind $C$ en $D$.
\end{enumerate}
Controleer met de lat: $DC$ moet ook $4$ cm meten.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g6:shapes:1}
Construeer een driehoek $ABC$ met $AB = 6$ cm, $AC = 5$ cm en $BC = 4$ cm,

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:2}
Construeer een \omterm{def:g6:shapes:triangles}{gelijkbenige driehoek} $DEF$ met $DE = DF = 5$ cm en
$EF = 3$ cm. Welke twee hoeken van deze driehoek lijken gelijk? Meet ze om het

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:3}
Construeer een \omterm{def:g6:shapes:triangles}{gelijkzijdige driehoek} met zijde $4$ cm met alleen lat en

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:4}
Bepaal van elke driehoek het soort uit de gegevens (\omterm{def:g6:shapes:triangles}{gelijkbenig},
\omterm{def:g6:shapes:triangles}{gelijkzijdig}, \omterm{def:g6:shapes:triangles}{rechthoekig} --- er kunnen meerdere antwoorden gelden):
(a) zijden $7$, $7$, $7$; \
(b) zijden $5$, $5$, $8$; \
(c) hoeken $90^\circ$, $45^\circ$, $45^\circ$; \
(d) hoeken $60^\circ$, $60^\circ$, $60^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:5}
Welke zijde van een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} is de schuine zijde? Teken een
driehoek $MNP$ met de \omterm{def:g3:shapes:rightangle}{rechte hoek} in $N$ en noem zijn schuine zijde. Welk
\omterm{def:g5:solids:def}{hoekpunt} raakt de schuine zijde nooit?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:6}
Waar of niet waar? Verantwoord elk antwoord in één zin.
\begin{enumerate}
  \item Elk vierkant is een rechthoek.
  \item Elke rechthoek is een vierkant.
  \item Elk vierkant is een \omterm{def:g6:shapes:quadrilaterals}{ruit}.
  \item Een \omterm{def:g6:shapes:quadrilaterals}{ruit} met een \omterm{def:g3:shapes:rightangle}{rechte hoek} is een vierkant.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:7}
Construeer een rechthoek $ABCD$ met $AB = 5$ cm en $BC = 3$ cm. Meet zijn twee
diagonalen: wat valt je op?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:shapes:8}
Construeer een \omterm{def:g6:shapes:quadrilaterals}{ruit} met diagonalen van $6$ cm en $4$ cm. (Gebruik
\cref{prop:g6:shapes:diagonals}: teken eerst de twee diagonalen ---
\omterm{def:g6:lines:perp}{loodrecht} op elkaar en elkaar snijdend in hun middens --- en verbind daarna de
vier uiteinden.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:9}
Schrijf een tekenprogramma voor een \omterm{def:g6:shapes:triangles}{gelijkbenige driehoek} met basis $4$ cm en
gelijke zijden van $6$ cm, in de stijl van \cref{ex:g6:shapes:program}. Ruil met
een klasgenoot: kan die het volgen zonder jouw figuur te zien?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:10}
Teken een cirkel met middelpunt $O$ en een van zijn \omterm{def:g4:geometry:circle}{diameters} $[AB]$. Zet een
willekeurig punt $M$ op de cirkel en teken de driehoek $AMB$. Meet de hoek
$\widehat{AMB}$. Verschuif $M$ en meet opnieuw. Wat vermoed je? (Dit
verrassende feit wordt bewezen in \cref{ch:g8:cosine}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:shapes:11}
Een \omterm{def:g4:geometry:polygon}{vierhoek} heeft diagonalen van dezelfde lengte die elkaar in hun
gemeenschappelijke midden snijden. Zoek met
\cref{prop:g6:shapes:diagonals} uit welke figuur het moet zijn: rechthoek,
\omterm{def:g6:shapes:quadrilaterals}{ruit} of vierkant? (Pas op: er is maar één eigenschap gegeven.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:shapes:12}
Hoeveel verschillende vierkanten kun je op een puntenraster van $4 \times 4$
tekenen met rasterpunten als \omterm{def:g5:solids:def}{hoekpunten} --- de ``scheve'' vierkanten
meegerekend? Teken minstens één scheef vierkant en leg uit waarom zijn zijden
gelijk zijn.
\end{exercise}

\section{Opgave: wat de diagonalen weten}

\begin{problem}[{Weekendopgave --- elke bijzondere vierhoek uit alleen zijn
diagonalen herbouwen, en de diagonalen van elke veelhoek tellen}]
\label{pb:g6:shapes:1}
\Cref{prop:g6:shapes:diagonals} leest een figuur en vertelt je over haar
diagonalen. In deze opgave speel je het spel \emph{achterstevoren}: gegeven
alleen twee lijnstukken die elkaar snijden --- de toekomstige diagonalen ---
welke figuur verschijnt er als je hun vier uiteinden verbindt? Je bouwt een
compleet ``diagonalenwoordenboek'', begrijpt \emph{waarom} de ruitconstructie
van \cref{exo:g6:shapes:8} werkt, ontmoet een figuur die nog geen naam heeft,
en eindigt met het tellen van de diagonalen van \omterm{def:g4:geometry:polygon}{veelhoeken} met veel meer dan
vier zijden.

\medskip
\noindent\textbf{Deel I --- Het diagonalenwoordenboek.}
Teken bij elke vraag de twee lijnstukken zoals beschreven, verbind hun vier
uiteinden op volgorde, en zeg welke \omterm{def:g4:geometry:polygon}{vierhoek} je krijgt.
\begin{enumerate}
  \item Twee lijnstukken van elk $6$ cm, \omterm{def:g6:lines:perp}{loodrecht} op elkaar, die elkaar in hun
        gemeenschappelijke midden snijden. Meet de vier zijden en de vier
        hoeken van de figuur die je krijgt: wat is het?
  \item Twee lijnstukken van $8$ cm en $5$ cm, \omterm{def:g6:lines:perp}{loodrecht} op elkaar, die elkaar
        in hun gemeenschappelijke midden snijden. Welke figuur verschijnt er
        (\cref{exo:g6:shapes:8})?
  \item Twee lijnstukken van elk $7$ cm --- gelijk, maar \emph{niet} \omterm{def:g6:lines:perp}{loodrecht}
        --- die elkaar in hun gemeenschappelijke midden snijden. Meet de zijden
        en de hoeken: welke figuur vermoed je?
  \item Twee lijnstukken van $8$ cm en $5$ cm, niet \omterm{def:g6:lines:perp}{loodrecht}, die elkaar in
        hun gemeenschappelijke midden snijden. De figuur die je krijgt, is geen
        van de drie van \cref{def:g6:shapes:quadrilaterals} --- beschrijf wat
        je ziet (welke zijden lijken gelijk? \omterm{def:g6:lines:perp}{parallel}?). Deze \omterm{def:g4:geometry:polygon}{vierhoek} krijgt
        zijn naam in \cref{ch:g7:central}.
  \item Vat de vragen 1--4 samen in een woordenboek met vier ingangen:
        diagonalen \emph{gelijk / \omterm{def:g6:lines:perp}{loodrecht} / beide / geen van beide} (steeds
        elkaar snijdend in hun middens) $\rightarrow$ naam (of beschrijving)
        van de figuur.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Waarom de constructies werken.}
\begin{enumerate}[resume]
  \item In vraag 2 verdelen de twee diagonalen de figuur in vier \omterm{def:g6:shapes:triangles}{rechthoekige
        driehoeken} om het snijpunt. Wat zijn de twee rechthoekszijden van elk
        van die driehoeken? Leg uit waarom alle vier de driehoeken precies
        elkaars kopie zijn, en leid daaruit af waarom de vier zijden van de
        figuur gelijk zijn --- het geheim van \cref{exo:g6:shapes:8}.
  \item Leid daaruit een test met alleen de passer af: hoe kun je na het bouwen
        van een figuur uit haar diagonalen met de passer alleen (zonder met de
        lat te meten) nagaan dat de vier zijden gelijk zijn?
  \item Schrijf een tekenprogramma (\cref{met:g6:shapes:program}) voor een
        vierkant waarvan de \emph{diagonalen} $6$ cm meten --- niet de zijden.
  \item Is in het vierkant van vraag 8 de zijde langer of korter dan $3$ cm ---
        de \omterm{def:g3:division:half}{helft} van de diagonaal? Meet het na. De ene \omterm{def:g3:division:half}{helft} van het antwoord
        kun je al \emph{bewijzen}: van de ene hoek naar de volgende via het
        middelpunt is een omweg van $3 + 3$ cm, en een rechte weg is altijd
        korter dan een omweg --- de zijde is dus korter dan $6$ cm. (Bewijzen
        dat hij ook groter is dan $3$ cm, en hem precies uitrekenen, is werk
        voor \cref{ch:g8:pythagoras}.)
  \item Een nieuwe ingang voor het woordenboek: teken een \omterm{def:g6:lines:objects}{lijnstuk} $[AC]$ van
        $8$ cm; zet daarop het punt $H$ op $3$ cm van $A$; teken door $H$ het
        loodrechte \omterm{def:g6:lines:objects}{lijnstuk} $[BD]$ van $5$ cm, met $H$ als \emph{zijn} midden.
        Verbind $ABCD$. Deze figuur is een \emph{vlieger}\index{vlieger} (de
        ene diagonaal deelt de andere middendoor, onder een \omterm{def:g3:shapes:rightangle}{rechte hoek}). Zoek
        met de hoekdriehoeken als in vraag 6 welke zijden van een \omterm{pb:g6:shapes:1}{vlieger} aan
        elkaar gelijk zijn.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Diagonalen tellen.}
Een \emph{diagonaal} van een \omterm{def:g4:geometry:polygon}{veelhoek} verbindt twee \omterm{def:g5:solids:def}{hoekpunten} die geen buren
zijn.
\begin{enumerate}[resume]
  \item Teken een \omterm{def:g4:geometry:polygon}{vierhoek}, een \omterm{def:g4:geometry:polygon}{vijfhoek} (vijf \omterm{def:g5:solids:def}{hoekpunten}) en een \omterm{def:g4:geometry:polygon}{zeshoek} (zes
        \omterm{def:g5:solids:def}{hoekpunten}), en teken al hun diagonalen. Tel ze: hoeveel zijn het per
        geval?
  \item Tellen zonder te tekenen, voor de \omterm{def:g4:geometry:polygon}{zeshoek}: hoeveel diagonalen vertrekken
        uit één \omterm{def:g5:solids:def}{hoekpunt}? (Met hoeveel van de zes \omterm{def:g5:solids:def}{hoekpunten} kan het \emph{niet}
        door een diagonaal verbonden worden?) Met zes vermenigvuldigen telt elke
        diagonaal precies twee keer --- waarom? Vind je telling van vraag 11
        terug. (Datzelfde dubbeltellen kwam voor bij de handdrukken van
        \cref{pb:g6:wholes:1}.)
  \item Tel met de methode van vraag 12 de diagonalen van een \omterm{def:g4:geometry:polygon}{veelhoek} met $10$
        \omterm{def:g5:solids:def}{hoekpunten}, en daarna met $20$ \omterm{def:g5:solids:def}{hoekpunten}.
  \item Een \omterm{def:g4:geometry:polygon}{veelhoek} heeft precies $44$ diagonalen. Hoeveel \omterm{def:g5:solids:def}{hoekpunten} heeft
        hij? (Probeer de methode van vraag 13 op een paar kandidaten.)
