\chapter{Lijnen, cirkels en hoeken}\label{ch:g6:lines}

Meetkunde begint met een lat, een geodriehoek en een passer. Dit hoofdstuk
legt de woorden vast --- punten, lijnstukken, halfrechten, \omterm{def:g6:lines:objects}{lijnen}, cirkels ---
introduceert de twee bijzondere standen van \omterm{def:g6:lines:objects}{lijnen} (\omterm{def:g6:lines:perp}{parallel} en \omterm{def:g6:lines:perp}{loodrecht}), en
leert je hoeken meten en tekenen.

\section{Punten, lijnstukken, lijnen}

\begin{definition}[De basisobjecten]\label{def:g6:lines:objects}
Door twee verschillende punten $A$ en $B$ gaan:
\begin{itemize}
  \item het \emph{lijnstuk}\index{lijnstuk} $[AB]$: het deel van de lijn
        tussen $A$ en $B$ (het heeft een lengte, geschreven $AB$);
  \item de \emph{halfrechte} $[AB)$: begint in $A$, gaat door $B$ en loopt
        eindeloos verder;
  \item de \emph{lijn}\index{lijn} $(AB)$: loopt aan beide kanten eindeloos
        door. Twee punten bepalen precies één lijn.
\end{itemize}
Punten op dezelfde lijn heten \emph{collineair}.
\end{definition}

\section{Parallelle en loodrechte lijnen}

\begin{definition}[Loodrecht, parallel]\label{def:g6:lines:perp}
Twee \omterm{def:g6:lines:objects}{lijnen} staan \emph{loodrecht}\index{loodrechte lijnen} op elkaar als ze
elkaar onder een \omterm{def:g3:shapes:rightangle}{rechte hoek} snijden; we schrijven $d_1 \perp d_2$ en zetten
een vierkantje bij de \omterm{def:g3:shapes:rightangle}{rechte hoek}. Twee \omterm{def:g6:lines:objects}{lijnen} zijn
\emph{parallel}\index{parallelle lijnen} als ze elkaar nooit ontmoeten, hoe
ver je ze ook doortrekt; we schrijven $d_1 \parallel d_2$.
\end{definition}

\begin{proposition}[Twee nuttige feiten]\label{prop:g6:lines:facts}
\begin{enumerate}
  \item Staan twee \omterm{def:g6:lines:objects}{lijnen} beide \omterm{def:g6:lines:perp}{loodrecht} op een derde \omterm{def:g6:lines:objects}{lijn}, dan zijn ze
        \omterm{def:g6:lines:perp}{parallel} aan elkaar.
  \item Zijn twee \omterm{def:g6:lines:objects}{lijnen} \omterm{def:g6:lines:perp}{parallel}, dan staat elke \omterm{def:g6:lines:objects}{lijn} die \omterm{def:g6:lines:perp}{loodrecht} op de
        ene staat, ook \omterm{def:g6:lines:perp}{loodrecht} op de andere.
\end{enumerate}
\end{proposition}
\admitted

\begin{method}[Tekenen met de geodriehoek]\label{met:g6:lines:setsquare}
Om de loodlijn op een \omterm{def:g6:lines:objects}{lijn} $d$ door een punt $P$ te tekenen:
\begin{enumerate}
  \item leg één rand van de \omterm{def:g3:shapes:rightangle}{rechte hoek} van de geodriehoek langs $d$;
  \item schuif de geodriehoek langs $d$ tot de andere rand bij $P$ komt;
  \item teken de \omterm{def:g6:lines:objects}{lijn} langs die rand, en zet het teken van de \omterm{def:g3:shapes:rightangle}{rechte hoek}
        erbij.
\end{enumerate}
Voor een \omterm{def:g6:lines:perp}{parallel} door $P$: teken een loodlijn op $d$, en daarna de loodlijn
op \emph{die} \omterm{def:g6:lines:objects}{lijn} door $P$ (\cref{prop:g6:lines:facts}).
\end{method}

\section{Cirkels}

\begin{definition}[Cirkel]\label{def:g6:lines:circle}
De \emph{cirkel}\index{cirkel} met middelpunt $O$ en \omterm{def:g3:shapes:shapes}{radius} $r$ bestaat uit
alle punten die precies op afstand $r$ van $O$ liggen. Een \omterm{def:g6:lines:objects}{lijnstuk} van het
middelpunt naar de cirkel is een \emph{\omterm{def:g3:shapes:shapes}{radius}}; een \omterm{def:g6:lines:objects}{lijnstuk} dat twee punten
van de cirkel door het middelpunt verbindt, is een \emph{\omterm{def:g4:geometry:circle}{diameter}} --- zijn
lengte is $2r$; een \omterm{def:g6:lines:objects}{lijnstuk} dat twee punten van de cirkel verbindt, is een
\emph{koorde}.
\end{definition}

\begin{example}\label{ex:g6:lines:circle}
``Teken de cirkel met middelpunt $O$ door $A$'': zet de passer open van $O$
tot $A$ --- de \omterm{def:g3:shapes:shapes}{radius} is de afstand $OA$ --- en draai rond. Elk punt van deze
cirkel ligt op dezelfde afstand van $O$ als $A$.
\end{example}

\section{Hoeken}

\begin{definition}[Hoek]\label{def:g6:lines:angle}
Twee halfrechten $[AB)$ en $[AC)$ met hetzelfde beginpunt $A$ vormen de
\emph{hoek}\index{hoek} $\widehat{BAC}$; het punt $A$ is zijn
\emph{\omterm{def:g5:solids:def}{hoekpunt}} (altijd de middelste letter!). Hoeken meet je in
\emph{graden} ($^\circ$), van $0^\circ$ tot $360^\circ$ voor een volle
draai. Een hoek is:
\begin{itemize}
  \item \emph{recht} als hij $90^\circ$ meet;
  \item \emph{scherp} als hij minder dan $90^\circ$ meet;
  \item \emph{stomp} als hij tussen $90^\circ$ en $180^\circ$ meet;
  \item \emph{gestrekt} als hij $180^\circ$ meet (de twee halfrechten vormen
        dan een \omterm{def:g6:lines:objects}{lijn}).
\end{itemize}
\end{definition}

\begin{method}[Een hoek meten met een gradenboog]\label{met:g6:lines:protractor}
\begin{enumerate}
  \item Leg het midden van de gradenboog precies op het \omterm{def:g5:solids:def}{hoekpunt} van de hoek;
  \item leg zijn $0^\circ$-lijn langs één zijde van de hoek;
  \item lees de streepjes af waar de andere zijde langs gaat --- op de schaal
        die bij $0$ begint aan de kant die je hebt uitgelijnd;
  \item controleer met je ogen: een scherpe hoek moet minder dan $90^\circ$
        geven, een stompe meer.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:lines:estimate}
Schat voordat je meet! Een hoek die iets wijder openstaat dan de hoek van een
blad papier, is iets meer dan $90^\circ$; de \omterm{def:g3:division:half}{helft} van een \omterm{def:g3:shapes:rightangle}{rechte hoek} is
$45^\circ$; een derde van een \omterm{def:g3:shapes:rightangle}{rechte hoek} is $30^\circ$. Zegt je gradenboog
$150^\circ$ bij een hoek die scherp lijkt, dan heb je de verkeerde schaal
gelezen: de juiste maat is $180^\circ - 150^\circ = 30^\circ$.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g6:lines:1}
Zet drie punten $A$, $B$ en $C$ die niet op één \omterm{def:g6:lines:objects}{lijn} liggen. Teken in
verschillende kleuren: het \omterm{def:g6:lines:objects}{lijnstuk} $[AB]$, de halfrechte $[CA)$ en de \omterm{def:g6:lines:objects}{lijn}
$(BC)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:2}
Waar of niet waar? ``$[AB]$ en $[BA]$ zijn hetzelfde \omterm{def:g6:lines:objects}{lijnstuk}.'' ``$[AB)$ en
$[BA)$ zijn dezelfde halfrechte.'' ``$(AB)$ en $(BA)$ zijn dezelfde \omterm{def:g6:lines:objects}{lijn}.''
Leg elk antwoord uit.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:3}
Teken een \omterm{def:g6:lines:objects}{lijn} $d$ en een punt $P$ dat niet op $d$ ligt. Construeer met de
geodriehoek: de loodlijn op $d$ door $P$, en daarna de \omterm{def:g6:lines:perp}{parallel} aan $d$ door
$P$. Beschrijf je stappen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:4}
De \omterm{def:g6:lines:objects}{lijnen} $a$ en $b$ staan beide \omterm{def:g6:lines:perp}{loodrecht} op een \omterm{def:g6:lines:objects}{lijn} $c$, en een vierde \omterm{def:g6:lines:objects}{lijn}
$e$ staat \omterm{def:g6:lines:perp}{loodrecht} op $a$. Wat kun je zeggen over $a$ en $b$? En over $e$ en
$c$? Verantwoord het met \cref{prop:g6:lines:facts}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:5}
Teken een cirkel met middelpunt $O$ en \omterm{def:g3:shapes:shapes}{radius} $3$ cm. Zet een punt $M$ op de
cirkel, een punt $N$ erbinnen en een punt $P$ erbuiten. Wat kun je zeggen over
de afstanden $OM$, $ON$ en $OP$ vergeleken met $3$ cm?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:6}
Een cirkel heeft \omterm{def:g4:geometry:circle}{diameter} $9$ cm. Wat is zijn \omterm{def:g3:shapes:shapes}{radius}? Een andere cirkel heeft
\omterm{def:g3:shapes:shapes}{radius} $2.6$ cm: wat is zijn \omterm{def:g4:geometry:circle}{diameter}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:7}
Noem de gemarkeerde hoek met drie letters en zeg daarna van welk soort hij is
(scherp, recht, stomp, gestrekt): een hoek van $72^\circ$ met \omterm{def:g5:solids:def}{hoekpunt} $R$
tussen de halfrechten naar $S$ en $T$; een hoek van $148^\circ$ met \omterm{def:g5:solids:def}{hoekpunt}
$B$ tussen de halfrechten naar $A$ en $C$; een hoek van $90^\circ$ met
\omterm{def:g5:solids:def}{hoekpunt} $O$ tussen de halfrechten naar $M$ en $N$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:8}
Schat en meet daarna met een gradenboog de drie hoeken van een driehoek die je
zelf tekent. Tel de drie maten op: wat vind je? (Houd je antwoord bij voor
\cref{ch:g7:triangles}.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:lines:9}
Teken met een gradenboog een hoek $\widehat{xOy}$ van $65^\circ$, en daarna een
hoek van $130^\circ$. Hoe zou je de tweede uit de eerste kunnen krijgen zonder
gradenboog?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:lines:10}
Twee dorpen $A$ en $B$ staan op een kaart. Waar liggen de punten die op $3$ cm
van $A$ \emph{en} op $2$ cm van $B$ liggen? Maak een tekening die laat zien
hoeveel zulke punten er kunnen zijn ($0$, $1$ of $2$, afhankelijk van de
afstand $AB$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:lines:11}
Een klok wijst 3 uur aan: wat is de hoek tussen de twee wijzers? Dezelfde
vraag om 5 uur, en om 6 uur. (Een volle draai is $360^\circ$ voor $12$ uur.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:lines:12}
Teken een \omterm{def:g6:lines:objects}{lijnstuk} $[AB]$ van $6$ cm. Construeer met alleen de passer het punt
$C$ waarvoor $AC = BC = 6$ cm, en meet de hoek $\widehat{CAB}$. Welke driehoek
heb je gebouwd, en wat vermoed je over zijn hoeken?
\end{exercise}

\section{Opgave: de meetkunde van de klok}

\begin{problem}[{Weekendopgave --- hoeken als breuken van een draai: ze op
een klok lezen, de momenten jagen waarop de wijzers samenvallen, en een dag in
een cirkeldiagram verdelen}]
\label{pb:g6:lines:1}
Een klok is een gradenboog die de tijd vertelt: haar wijzerplaat is een volle
draai van $360^\circ$, door de twaalf uurstreepjes in twaalf gelijke hoeken
verdeeld. In \cref{exo:g6:lines:11} maten we de wijzers om 3 uur en om 6 uur;
in deze opgave bouw je de volledige theorie --- ook voor de tijden waarop
\emph{geen} van de wijzers naar een streepje wijst --- geef je antwoord op een
vraag die weinig volwassenen goed krijgen (``hoe vaak liggen de twee wijzers
precies op elkaar?''), en verdeel je tot slot een hele dag in een
cirkeldiagram.

\medskip
\noindent\textbf{Deel I --- \omterm{def:g6:fractions:def}{Breuken} van een draai.}
\begin{enumerate}
  \item Een volle draai meet $360^\circ$. Hoeveel graden zijn een halve draai,
        een \omterm{def:g3:division:half}{kwart} draai en een twaalfde van een draai?
  \item De twaalf uurstreepjes verdelen de wijzerplaat in twaalf gelijke
        hoeken bij het middelpunt. Hoeveel graden zitten er tussen twee
        buurstreepjes? Vind het antwoord van \cref{exo:g6:lines:11} voor 3 uur
        terug door streepjes te tellen.
  \item Geef de hoek tussen de wijzers om 1 uur, om 4 uur en om 7 uur. (Om 7
        uur verdelen de wijzers de wijzerplaat in twee hoeken; ``de hoek
        tussen de wijzers'' betekent altijd de \emph{kleinste}.)
  \item De minutenwijzer maakt een volle draai in $60$ minuten. Hoeveel graden
        legt hij per minuut af?
  \item De uurwijzer gaat van het ene streepje naar het volgende ---
        $30^\circ$ --- in $60$ minuten. Hoeveel graden legt \emph{hij} per
        minuut af? (Een \omterm{def:g6:decimals:places}{decimaal getal}, \cref{ch:g6:decimals}.)
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Tijden waarop niets naar een streepje wijst.}
\begin{enumerate}[resume]
  \item Om 3:30 wijst de minutenwijzer naar de $6$. Waar staat de uurwijzer
        precies? Reken de hoek van elke wijzer vanaf de $12$ uit (met de klok
        mee gemeten) en leid daaruit de hoek tussen de wijzers af.
  \item Om 6:30 denken veel mensen dat de wijzers op elkaar liggen. Gok eerst,
        en reken de hoek daarna uit zoals in vraag 6. Wie had gelijk?
  \item Reken de hoek tussen de wijzers om 9:15 uit.
  \item Leg uit waarom de twee wijzers ergens tussen 1:00 en 1:10 precies op
        elkaar moeten liggen: waar staat de minutenwijzer om 1:00 ten opzichte
        van de uurwijzer, en waar om 1:10? Welke wijzer loopt sneller, en
        waarom beslist dat de zaak?
  \item Hoe vaak liggen de twee wijzers in twaalf uur precies op elkaar? (In
        elk stuk tussen twee opeenvolgende uren gebeurt één ontmoeting ---
        met één uitzondering: wat gebeurt er tussen 11 en 12? Noem de
        ontmoetingstijden ongeveer en tel ze.) En hoe vaak op een hele dag?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Hoeken om een punt: cirkeldiagrammen.}
\begin{enumerate}[resume]
  \item De twaalf sectoren van de wijzerplaat vullen samen de hele plaat: hun
        hoeken zijn samen $360^\circ$. Leg uit waarom dat geldt voor
        \emph{elke} verzameling hoeken die één \omterm{def:g5:solids:def}{hoekpunt} delen en samen een
        volle draai vullen, zonder overlap en zonder gat.
  \item Zoë houdt haar dag van $24$ uur bij: slapen $9$ u, school $6$ u,
        spelen $3$ u, eten $2$ u, al het overige $4$ u. Ze wil een
        \emph{cirkeldiagram}: een schijf waarin elke bezigheid een sector
        krijgt, met hoeken die evenredig zijn met de tijden. Welke \omterm{def:g6:fractions:def}{breuk} van de
        dag is elke bezigheid, en hoeveel graden krijgt haar sector?
        Controleer dat de vijf hoeken samen $360^\circ$ zijn.
  \item Beschrijf stap voor stap hoe je het cirkeldiagram van Zoë tekent met
        passer en gradenboog (\cref{met:g6:lines:protractor}): waar de eerste
        \omterm{def:g3:shapes:shapes}{radius} komt, en hoe elke nieuwe sector begint waar de vorige eindigt.
  \item In een ander cirkeldiagram van een dag meet één sector precies
        $90^\circ$. Welke \omterm{def:g6:fractions:def}{breuk} van de schijf is dat, en hoeveel uur stelt het
        voor?
  \item De finale, terug bij de klok: hoeveel volle draaien maakt de uurwijzer
        in $24$ uur? En de minutenwijzer? En de secondewijzer? (Een van deze
        antwoorden ligt boven de duizend.)
\end{enumerate}
\end{problem}
