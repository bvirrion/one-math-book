\chapter{Decimale getallen}\label{ch:g6:decimals}

Tussen $2$ en $3$ liggen heel veel getallen: $2.5$, $2.71$,
$2.999$\,\dots\ Decimale getallen breiden het plaatswaardestelsel uit naar
rechts van de eenheden, en ze volgen dezelfde regels als hele getallen ---
met een paar valstrikken die dit hoofdstuk je leert vermijden.

\section{Decimale schrijfwijze}

\begin{definition}[Decimale plaatsen]\label{def:g6:decimals:places}
Een \emph{decimaal getal}\index{decimaal getal} heeft een heel deel en een
decimaal deel, gescheiden door het \omterm{def:g4:decimals:point}{decimaalteken}. Elke plaats rechts van het
\omterm{def:g4:decimals:point}{decimaalteken} is tien keer zo klein als de vorige: tienden, honderdsten,
duizendsten. Zo is
\[
13.407 = 13 + \frac{4}{10} + \frac{0}{100} + \frac{7}{1000}
= 1 \times 10 + 3 + 4 \times \frac{1}{10} + 7 \times \frac{1}{1000}.
\]
\end{definition}

\begin{remark}[Nullen die meetellen en nullen die niets doen]
Nullen \emph{achteraan} het decimale deel bijzetten verandert niets:
$2.5 = 2.50 = 2.500$. Maar nullen \emph{tussen} de cijfers zijn onmisbaar:
$13.407 \neq 13.47$. En $0.5$ is heel iets anders dan $0.05$!
\end{remark}

\begin{method}[Decimale getallen vergelijken]\label{met:g6:decimals:compare}
\begin{enumerate}
  \item Vergelijk eerst de hele delen: $7.2 > 5.99$, want $7 > 5$.
  \item Zijn de hele delen gelijk, vergelijk dan de decimale delen cijfer
        voor cijfer vanaf links --- aanvullen met nullen achteraan helpt: om
        $3.4$ en $3.15$ te vergelijken, schrijf je $3.40$ en $3.15$; omdat
        $40 > 15$ honderdsten, is $3.4 > 3.15$.
\end{enumerate}
Let op: \emph{langer betekent niet groter}. $3.15$ heeft meer cijfers dan
$3.4$, maar is kleiner.
\end{method}

\section{Optellen, aftrekken, vermenigvuldigen}

\begin{method}[Cijferen met decimale getallen]\label{met:g6:decimals:columns}
Om decimale getallen op te tellen of af te trekken, zet je de
\emph{\omterm{def:g4:decimals:point}{decimaaltekens}} onder elkaar (zodat de eenheden onder de eenheden staan
en de tienden onder de tienden), waar nodig aangevuld met nullen achteraan;
reken daarna zoals bij hele getallen, en zet het \omterm{def:g4:decimals:point}{decimaalteken} van de uitkomst
onder de andere.
\end{method}

\begin{example}\label{ex:g6:decimals:addsub}
Reken $13.7 + 2.85$ uit, met de \omterm{def:g4:decimals:point}{decimaaltekens} onder elkaar
($13.7 = 13.70$):
\[
\begin{array}{r}
1\,3.7\,0 \\
+\ \ \,2.8\,5 \\
\hline
1\,6.5\,5
\end{array}
\]
Reken $6 - 2.35$ uit (schrijf $6 = 6.00$): $6.00 - 2.35 = 3.65$. Controle:
$3.65 + 2.35 = 6$.
\end{example}

\begin{example}[Decimale getallen vermenigvuldigen]\label{ex:g6:decimals:mult}
Om $2.3 \times 1.4$ uit te rekenen: vermenigvuldig als hele getallen,
$23 \times 14 = 322$; tel daarna de decimalen van de factoren
($1 + 1 = 2$) en zet het \omterm{def:g4:decimals:point}{decimaalteken} zo dat de uitkomst er even veel heeft:
$2.3 \times 1.4 = 3.22$. Controle op de grootte: $2.3 \times 1.4$ moet iets
meer zijn dan $2.3 \times 1 = 2.3$ --- en $3.22$ is dat.
\end{example}

\section{Vermenigvuldigen en delen door 10, 100, 1000}

\begin{proposition}[Het decimaalteken verschuiven]\label{prop:g6:decimals:shift}
Een \omterm{def:g6:decimals:places}{decimaal getal} met $10$, $100$ of $1000$ vermenigvuldigen schuift zijn
\omterm{def:g4:decimals:point}{decimaalteken} $1$, $2$ of $3$ plaatsen naar \emph{rechts}; delen schuift het
naar \emph{links} (met nullen erbij waar dat nodig is):
\[
3.75 \times 100 = 375, \qquad
42.1 \div 1000 = 0.0421 .
\]
\end{proposition}

\begin{proof}
Met $10$ vermenigvuldigen maakt elk cijfer tien keer zo veel waard: tienden
worden eenheden, eenheden worden tientallen, en zo verder --- elk cijfer
schuift één kolom naar links in de plaatswaardetabel, en dat is hetzelfde als
het \omterm{def:g4:decimals:point}{decimaalteken} één plaats naar rechts schuiven. Delen draait dit om.
\end{proof}

\begin{example}[Maateenheden]\label{ex:g6:decimals:units}
Eenheden omzetten is precies dit spel: omdat $1$ m $= 100$ cm, geldt
\[
3.75 \text{ m} = 3.75 \times 100 \text{ cm} = 375 \text{ cm},
\qquad
42 \text{ mm} = 42 \div 10 \text{ cm} = 4.2 \text{ cm}.
\]
\end{example}

\section{Afronden}

\begin{definition}[Afronden]\label{def:g6:decimals:rounding}
De \emph{afronding}\index{afronden} van een getal op de eenheid (of op het
tiende, honderdste, \dots) is het dichtstbijzijnde getal met die precisie.
Kijk naar het volgende cijfer: is dat $0,1,2,3,4$, dan rond je naar beneden
af (je laat staan); is het $5,6,7,8,9$, dan rond je naar boven af.
\end{definition}

\begin{example}
$7.38$ afgerond op de eenheid is $7$ (volgend cijfer $3$: laten staan);
afgerond op het tiende is het $7.4$ (volgend cijfer $8$: naar boven). De prijs
$4.996$ afgerond op het honderdste is $5.00$ --- \omterm{def:g6:decimals:rounding}{afronden} kan elk cijfer
veranderen!
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g6:decimals:1}
Schrijf als \omterm{def:g6:decimals:places}{decimaal getal}: $5 + \dfrac{3}{10} + \dfrac{7}{100}$;
\quad $\dfrac{9}{10}$; \quad $12 + \dfrac{4}{1000}$;
\quad ``acht eenheden en vijf honderdsten''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:2}
In $86.354$: wat is het cijfer van de tienden? En dat van de honderdsten? Wat
telt het cijfer $8$? Schrijf dit getal op zoals in
\cref{def:g6:decimals:places}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:3}
Neem over en vul in met $<$, $>$ of $=$:
\[
5.3 \;?\; 5.29, \qquad
0.7 \;?\; 0.70, \qquad
2.09 \;?\; 2.9, \qquad
14.5 \;?\; 14.49 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:4}
Zet op volgorde, van klein naar groot:
$4.2$; \ $4.05$; \ $4.51$; \ $4.15$; \ $4.5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:5}
Welke decimale getallen horen bij de punten $A$, $B$ en $C$ op een
getallenlijn met streepjes per tiende, als $A$ $3$ streepjes na $6$ ligt, $B$
$7$ streepjes na $6$, en $C$ $2$ streepjes na $7$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:6}
Reken cijferend uit: $45.8 + 7.65$; \quad $23.4 - 8.72$; \quad

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:7}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:decimals:8}

\begin{exercise}[$\star$]\label{exo:g6:decimals:9}
Rond $23.867$ af: op de eenheid; op het tiende; op het honderdste. Rond

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:decimals:10}
Een stokbrood kost $1.15$. Lena koopt drie stokbroden en betaalt met een

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:decimals:11}
Zoek een \omterm{def:g6:decimals:places}{decimaal getal} dat strikt tussen $7.4$ en $7.5$ ligt; en daarna een

geval?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:decimals:12}
Schrijf met de cijfers $2$, $5$ en $8$, elk precies één keer, en één

bevatten.)
\end{exercise}

\section{Opgave: het getal net na 3 bestaat niet}

\begin{problem}[{Weekendopgave --- tussen twee decimale getallen ligt er
altijd nog een: inzoomen op de getallenlijn}]
\label{pb:g6:decimals:1}
Op de ladder van de hele getallen heeft elk getal een naaste buur: direct na
$7$ komt $8$, en daartussen woont niets. Decimale getallen zijn een heel
andere wereld. Wat is het getal net na $3$? Is dat $3.1$? $3.01$? $3.001$? In
deze opgave ontwikkel je een \emph{zoomtechniek} op de getallenlijn (in het
spoor van \cref{exo:g6:decimals:11}) en kom je bij een beroemde conclusie: het
getal net na $3$ \emph{bestaat niet} --- tussen twee decimale getallen, hoe
dicht ze ook bij elkaar liggen, is er altijd ruimte voor meer.

\medskip
\noindent\textbf{Deel I --- Inzoomen.}
\begin{enumerate}
  \item Wat is groter, $2.999$ of $3$? Reken het \omterm{ex:g1:subtraction:difference}{verschil} tussen die twee
        uit.
  \item Teken een getallenlijn van $7.4$ tot $7.5$ met streepjes per
        honderdste, en zet $7.42$, $7.45$ en $7.48$ erop. Noem \emph{alle}
        getallen met twee decimalen die strikt tussen $7.4$ en $7.5$ liggen.
        Hoeveel zijn het?
  \item Zoom nog eens in: noem alle getallen met drie decimalen tussen $7.44$
        en $7.45$. Hoeveel zijn het nu?
  \item Leg met hetzelfde idee uit hoe je de getallen met precies drie
        decimalen tussen $7.4$ en $7.5$ kunt tellen, en geef dat aantal.
  \item Beschrijf het recept dat achter de vragen 2--4 zit (de \emph{zoom}):
        als twee getallen buren lijken, zoals $5.67$ en $5.68$, hoe levert
        één decimaal extra dan altijd een getal op dat er strikt tussen
        ligt? Pas je recept toe op $5.67$ en $5.68$, en daarna op $0.1999$ en
        $0.2$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De buur die er niet is.}
\begin{enumerate}[resume]
  \item Tom beweert: ``het getal net na $3$ is $3.1$.'' Weerleg hem door een
        getal te noemen dat strikt tussen $3$ en $3.1$ ligt. Tom valt terug
        op $3.01$, en dan op $3.001$. Versla elk van zijn kandidaten.
  \item Leg uit waarom \emph{niemand} dit spel van jou kan winnen: welk getal
        strikt groter dan $3$ Tom ook voorstelt, het zoomrecept van vraag 5
        levert een getal dat strikt tussen $3$ en zijn voorstel ligt. Wat
        bewijst dit over ``het getal net na $3$''?
  \item Nu de andere kant: Tom jaagt op het getal net \emph{voor} $3$ en
        probeert $2.9$, dan $2.99$, dan $2.999$. Versla zijn drie kandidaten.
        Reken daarna $3 - 2.999$ uit, en beschrijf (zonder te cijferen) het
        \omterm{ex:g1:subtraction:difference}{verschil} tussen $3$ en het getal dat je schrijft met een $2$, een
        \omterm{def:g4:decimals:point}{decimaalteken} en twintig cijfers $9$.
  \item Waarom werkt dit alles niet bij \emph{hele} getallen? Leg in één of
        twee zinnen uit waarom er geen heel getal strikt tussen $7$ en $8$
        ligt, terwijl er daar heel veel decimale getallen liggen.
  \item Een lengte wordt aangekondigd als ``$3.7$ cm, afgerond op het
        tiende'' (\cref{def:g6:decimals:rounding}). Geef de kleinste lengte
        die op $3.7$ cm afrondt, en leg uit waarom een lengte van $3.75$ cm
        \emph{niet} op $3.7$ afrondt --- de echte lengte is dus minstens
        $3.65$ cm en strikt kleiner dan $3.75$ cm.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Spelen met cijfers en \omterm{def:g4:decimals:point}{decimaaltekens}.}
\begin{enumerate}[resume]
  \item Zet op volgorde, van klein naar groot: $0.6$; \ $0.58$; \ $0.123$; \
        $0.0999$. Leg daarna in één zin aan een klasgenoot uit waarom
        $0.0999 < 0.6$, ook al is het met meer cijfers geschreven
        (\cref{met:g6:decimals:compare}).
  \item Prijzen in een winkel hebben altijd precies twee decimalen. Bestaat er
        een \emph{prijs} die strikt tussen $4.99$ en $5$ ligt? Vergelijk met
        vraag 9: wat hebben prijzen en hele getallen met elkaar gemeen?
  \item Zoek met de cijfers $2$, $5$ en $8$, elk precies één keer, en één
        \omterm{def:g4:decimals:point}{decimaalteken} (het hele deel niet leeg, zoals in
        \cref{exo:g6:decimals:12}) het getal dat het dichtst bij $6$ ligt.
        Verantwoord je antwoord door de afstand van je beste kandidaten tot
        $6$ uit te rekenen.
  \item Het beroemdste getal van de wiskunde, $\pi$, voldoet aan
        $3.141 < \pi < 3.142$. Noem een \omterm{def:g6:decimals:places}{decimaal getal} dat strikt tussen
        $3.141$ en $3.142$ ligt; en daarna een dat strikt tussen $3.1415$ en
        $3.1416$ ligt. (Wiskundigen klemmen $\pi$ precies zo in --- elke
        nieuwe decimaal is één zoom erbij. Je ziet $\pi$ aan het werk in
        \cref{ch:g6:measure}.)
  \item De finale: leg uit waarom er \emph{meer decimale getallen tussen $0$
        en $1$ liggen dan elk getal dat je kunt noemen}. (Hoeveel levert één
        zoom op? Kan het zoomen ooit ophouden?)
\end{enumerate}
\end{problem}
