\chapter{Gehele getallen}\label{ch:g6:wholes}

Alles in de wiskunde begint met tellen. Dit hoofdstuk herhaalt hoe gehele
getallen geschreven, vergeleken en gecombineerd worden --- en kijkt zorgvuldig
naar het delen, de bewerking die zowel een \omterm{def:g3:division:remainder}{quotiënt} \emph{als} een \omterm{def:g3:division:remainder}{rest}
oplevert.

\section{Gehele getallen schrijven en vergelijken}

\begin{definition}[Plaatswaarde]\label{def:g6:wholes:place}
Onze manier van getallen schrijven gebruikt tien cijfers ($0$ tot $9$), en de
\emph{waarde van een cijfer hangt af van zijn plaats}\index{plaatswaarde}: in
$5\,208$ telt het cijfer $5$ duizendtallen, de $2$ honderdtallen, de $0$
tientallen (die zijn er niet) en de $8$ eenheden:
\[
5\,208 = 5 \times 1000 + 2 \times 100 + 0 \times 10 + 8 .
\]
\end{definition}

\begin{method}[Twee gehele getallen vergelijken]\label{met:g6:wholes:compare}
\begin{enumerate}
  \item Het getal met meer cijfers is het grootste ($1\,203 > 989$).
  \item Hebben ze even veel cijfers, vergelijk dan cijfer voor cijfer vanaf
        \emph{links}; het eerste \omterm{ex:g1:subtraction:difference}{verschil} beslist: $6\,742 > 6\,715$, want
        bij de tientallen is $4 > 1$.
\end{enumerate}
De tekens zijn $<$ (``kleiner dan'') en $>$ (``groter dan''); de smalle kant
van het teken wijst naar het kleine getal.
\end{method}

\begin{example}\label{ex:g6:wholes:order}
Zet op volgorde, van klein naar groot: $908$; $89$; $1\,001$; $980$. Eerst op
het aantal cijfers: $89$ (twee cijfers) komt vooraan, $1\,001$ (vier cijfers)
achteraan. Tussen $908$ en $980$: hetzelfde cijfer $9$ bij de honderdtallen,
dan bij de tientallen $0 < 8$, dus $908 < 980$. Het antwoord:
\[
89 < 908 < 980 < 1\,001 .
\]
\end{example}

\section{Optellen, aftrekken, vermenigvuldigen}

\begin{example}[Cijferen]\label{ex:g6:wholes:columns}
Om op te tellen of af te trekken, zet je de eenheden onder de eenheden en de
tientallen onder de tientallen, en werk je van rechts naar links, met
onthouden waar dat nodig is:
\[
\begin{array}{r}
4\,5\,7 \\
+\ 2\,8\,6 \\
\hline
7\,4\,3
\end{array}
\qquad\qquad
\begin{array}{r}
6\,0\,3 \\
-\ 1\,4\,7 \\
\hline
4\,5\,6
\end{array}
\]
Bij de optelling: $7 + 6 = 13$, schrijf $3$, onthoud $1$; dan
$5 + 8 + 1 = 14$, schrijf $4$, onthoud $1$; dan $4 + 2 + 1 = 7$. Controleer de
aftrekking door terug op te tellen: $456 + 147 = 603$.
\end{example}

\begin{remark}[Woorden]
De uitkomst van een optelling is een \emph{som}\index{som}; van een
aftrekking een \emph{verschil}\index{verschil}; van een vermenigvuldiging een
\emph{product}\index{product}; van een deling een
\emph{quotiënt}\index{quotiënt}. De getallen die je combineert, zijn de
\emph{termen} (bij $+$ en $-$) of de \emph{factoren} (bij $\times$).
\end{remark}

\begin{example}[Slim rekenen]\label{ex:g6:wholes:smart}
Termen of factoren anders groeperen scheelt vaak veel werk:
\[
17 + 58 + 3 = (17 + 3) + 58 = 20 + 58 = 78 ,
\]
\[
25 \times 17 \times 4 = (25 \times 4) \times 17 = 100 \times 17 = 1700 .
\]
\end{example}

\section{Delen}

\begin{theorem}[Deling met rest]\label{thm:g6:wholes:division}
Gegeven twee gehele getallen $a$ (het \emph{deeltal}) en $b \neq 0$ (de
\emph{\omterm{def:g5:division:multiple}{deler}}), is er precies één manier om
\[
a = b \times q + r
\qquad\text{met}\qquad
0 \leq r < b
\]
te schrijven. Het getal $q$ is het \emph{quotiënt}\index{quotiënt} en $r$ de
\emph{rest}\index{rest} van de deling van $a$ door $b$.
\end{theorem}
\admitted

\begin{example}\label{ex:g6:wholes:division}
Deel $137$ door $4$, stap voor stap.
\begin{enumerate}
  \item Hoeveel keer gaat $4$ in $13$? Drie keer ($4 \times 3 = 12$), \omterm{def:g3:division:remainder}{rest}
        $13 - 12 = 1$.
  \item Haal de $7$ naar beneden: hoeveel keer gaat $4$ in $17$? Vier keer
        ($4 \times 4 = 16$), \omterm{def:g3:division:remainder}{rest} $1$.
\end{enumerate}
Dus $q = 34$ en $r = 1$:
\[
137 = 4 \times 34 + 1, \qquad\text{en inderdaad } 0 \leq 1 < 4 .
\]
Controle: $4 \times 34 = 136$, en $136 + 1 = 137$.
\end{example}

\begin{definition}[Deelbaar]\label{def:g6:wholes:divisible}
Als de \omterm{def:g3:division:remainder}{rest} $0$ is, zeggen we dat $a$ \emph{deelbaar}\index{deelbaar} is door
$b$: zo is $135$ deelbaar door $5$, want $135 = 5 \times 27 + 0$.
\end{definition}

\begin{method}[Vraagstukken met een deling]\label{met:g6:wholes:problems}
Ga na het uitrekenen van $a = bq + r$ altijd terug naar de vraag:
\begin{enumerate}
  \item ``Hoeveel volle dozen/teams/bladzijden?'' --- het antwoord is het
        \omterm{def:g3:division:remainder}{quotiënt} $q$;
  \item ``Hoeveel blijven er over?'' --- het antwoord is de \omterm{def:g3:division:remainder}{rest} $r$;
  \item ``Hoeveel dozen zijn er nodig voor alles?'' --- als $r > 0$, één
        \emph{meer} dan het \omterm{def:g3:division:remainder}{quotiënt}: $q + 1$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g6:wholes:bus}
$137$ leerlingen gaan op excursie in bussen met $40$ plaatsen. De deling:
$137 = 40 \times 3 + 17$. Drie bussen vervoeren $120$ leerlingen, en $17$
leerlingen blijven over --- die hebben ook een bus nodig! Er zijn dus $4$
bussen nodig, ook al is het \omterm{def:g3:division:remainder}{quotiënt} $3$.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g6:wholes:1}
Schrijf in cijfers: ``drieduizend zevenenveertig''; ``twintigduizend
vijfhonderd''; ``honderdvier''. Schrijf daarna $60\,013$ in woorden.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:2}
In het getal $47\,209$: wat is het cijfer van de honderdtallen? En dat van de
eenheden? Wat telt het cijfer $4$? Schrijf het getal als een \omterm{def:g1:addition:def}{som} van
\omterm{def:g5:division:multiple}{veelvouden} van $1$, $10$, $100$, \dots\ zoals in
\cref{def:g6:wholes:place}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:3}
Neem over en vul in met $<$ of $>$:
\[
507 \;?\; 570, \qquad
1\,099 \;?\; 989, \qquad
23\,456 \;?\; 23\,465, \qquad
9\,999 \;?\; 10\,000 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:4}
Zet op volgorde, van klein naar groot: $2\,043$; $978$; $2\,304$; $2\,034$;
$1\,001$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:5}
Reken cijferend uit: $348 + 576$; \quad $805 - 268$; \quad
$307 \times 26$. Controleer de aftrekking met een optelling.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:6}
Reken slim, door termen of factoren te groeperen:
\[
38 + 45 + 12 + 5, \qquad
2 \times 83 \times 50, \qquad
4 \times 78 \times 25 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:7}
Geef bij elke deling het \omterm{def:g3:division:remainder}{quotiënt} en de \omterm{def:g3:division:remainder}{rest}, en schrijf de gelijkheid
$a = b \times q + r$ op:
\[
95 \div 7, \qquad
230 \div 6, \qquad
504 \div 8 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:wholes:8}
Leg zonder te delen uit waarom de gelijkheid $173 = 8 \times 20 + 13$
\emph{niet} de deling met \omterm{def:g3:division:remainder}{rest} van $173$ door $8$ is, en zoek het juiste
\omterm{def:g3:division:remainder}{quotiënt} en de juiste \omterm{def:g3:division:remainder}{rest}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:9}
Eieren worden in dozen van $12$ verpakt.
\begin{enumerate}
  \item Een boerderij heeft $2\,000$ eieren. Hoeveel volle dozen kan ze
        vullen, en hoeveel eieren blijven over?
  \item Een bakkerij heeft $150$ eieren nodig. Hoeveel dozen moet ze kopen?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:10}
Vandaag is het dinsdag. Welke dag van de week is het over $100$ dagen? (Een
week heeft $7$ dagen: deel, en denk aan de \omterm{def:g3:division:remainder}{rest}.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:wholes:11}
Bij een deling door $9$ is het \omterm{def:g3:division:remainder}{quotiënt} $56$ en is de \omterm{def:g3:division:remainder}{rest} zo groot als hij
maar kan zijn. Wat is het deeltal?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:wholes:12}
Ik ben een getal van drie cijfers. Mijn cijfer van de honderdtallen is twee
keer mijn cijfer van de eenheden, mijn cijfer van de tientallen is $0$, en de
\omterm{def:g1:addition:def}{som} van mijn cijfers is $9$. Wie ben ik? (Denk stap voor stap, en controleer
daarna.)
\end{exercise}

\section{Opgave: de kleine Gauss en de som van de eerste honderd
getallen}

\begin{problem}[{Weekendopgave --- $1 + 2 + 3 + \dots + 100$ in tien
seconden, en de driehoeksgetallen}]
\label{pb:g6:wholes:1}
Een beroemd verhaal: om zijn klas bezig te houden, vroeg een meester zijn
leerlingen alle getallen van $1$ tot $100$ op te tellen. Eén jongen schreef
bijna onmiddellijk $5\,050$ op zijn schrijfplankje --- Carl Friedrich Gauss,
negen jaar oud, die een van de grootste wiskundigen van alle tijden zou
worden. Zijn geheim was niets anders dan \emph{slim rekenen}
(\cref{ex:g6:wholes:smart}): de termen zo groeperen dat elk groepje makkelijk
is. In deze opgave ontdek je zijn truc opnieuw, laat je hem in elke situatie
werken, en zie je hoe vaak zulke sommen in het dagelijks leven opduiken.

\medskip
\noindent\textbf{Deel I --- De paren van Gauss.}
\begin{enumerate}
  \item Reken $1 + 2 + 3 + 4 + 5$ uit door de termen slim te groeperen.
  \item Voor $1 + 2 + 3 + \dots + 10$: koppel de eerste term aan de laatste
        ($1 + 10$), de tweede aan de voorlaatste ($2 + 9$), en zo verder.
        Hoeveel is elk paar waard? Hoeveel paren zijn er? Maak de berekening
        af.
  \item Dezelfde manier voor $1 + 2 + \dots + 20$: hoeveel is elk paar
        waard, hoeveel paren zijn er, en wat is het totaal?
  \item Nu de \omterm{def:g1:addition:def}{som} van de meester, $1 + 2 + \dots + 100$: leg uit waarom de
        paren $1 + 100$, $2 + 99$, $3 + 98, \dots$ allemaal even veel waard
        zijn, en waarom het er precies $50$ zijn, met $50 + 51$ als laatste.
  \item Schrijf het antwoord van Gauss als één vermenigvuldiging, en reken
        het uit.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Een truc die nooit faalt.}
\begin{enumerate}[resume]
  \item Probeer het koppelen op $1 + 2 + \dots + 9$: nu blijft er één getal
        zonder partner over. Welk? Koppel eromheen en reken de \omterm{def:g1:addition:def}{som} uit.
  \item Hier is een versie van de truc die werkt of het aantal termen \omterm{def:g3:numbers:evenodd}{even} of
        \omterm{def:g3:numbers:evenodd}{oneven} is: schrijf de \omterm{def:g1:addition:def}{som} twee keer, één keer vooruit en één keer
        achteruit, onder elkaar,
        \[
        \begin{array}{ccccccc}
        1 & + & 2 & + & \dots & + & 9 \\
        9 & + & 8 & + & \dots & + & 1
        \end{array}
        \]
        en tel \emph{kolom voor kolom} op. Hoeveel is elke kolom waard?
        Hoeveel kolommen zijn er? Leg uit waarom de \omterm{def:g1:addition:def}{som} van de twee regels
        $9 \times 10 = 90$ is, en besluit dat $1 + 2 + \dots + 9 = 45$ ---
        hetzelfde antwoord als in vraag 6.
  \item Reken met de truc van de twee regels $1 + 2 + \dots + 50$ uit, en
        daarna $1 + 2 + \dots + 200$.
  \item Reken de \omterm{def:g1:addition:def}{som} van de eerste honderd \emph{\omterm{def:g3:numbers:evenodd}{even}} getallen uit,
        $2 + 4 + 6 + \dots + 200$. (Vergelijk elke term met een term van de
        \omterm{def:g1:addition:def}{som} van Gauss: nieuw koppelen is niet nodig.)
  \item Reken $101 + 102 + \dots + 200$ uit met twee sommen die je al kent en
        één aftrekking.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- \omterm{pb:g6:wholes:1}{Driehoeksgetallen}, overal.}
De totalen $1$, $3$, $6$, $10$, $15, \dots$ van de sommen
$1 + 2 + \dots + n$ heten \emph{driehoeksgetallen}\index{driehoeksgetallen},
omdat ze voorwerpen tellen die in een driehoek gestapeld liggen.
\begin{enumerate}[resume]
  \item Acht vrienden ontmoeten elkaar, en ieder geeft precies één keer elk
        van de anderen een hand. Leg uit waarom het aantal handdrukken
        $7 + 6 + 5 + 4 + 3 + 2 + 1$ is, en reken het uit.
  \item Een winkelierster bouwt een driehoekige \omterm{def:g5:solids:def}{piramide} van blikjes: $12$
        blikjes in de onderste rij, $11$ in de volgende, en zo verder tot één
        blikje bovenaan. Hoeveel blikjes heeft ze nodig? (Gebruik de truc van
        vraag 7.)
  \item Teken de \omterm{def:g1:addition:def}{som} $1 + 2 + 3 + 4$ als een trap van vakjes: één vakje in de
        eerste kolom, twee in de tweede, drie in de derde, vier in de vierde.
        Laat op je tekening zien dat \emph{twee} zulke trappen, waarvan er
        één op zijn kop staat, precies in een rechthoek van $4$ rijen van $5$
        vakjes passen --- en leg uit waarom dit de truc van de twee regels
        uit vraag 7 is, getekend in plaats van opgeschreven.
  \item Een klok slaat $1$ keer om één uur, $2$ keer om twee uur, \dots, $12$
        keer om twaalf uur. Hoeveel slagen zijn dat over de twaalf uren? En
        over een hele dag?
  \item Nog één keer Gauss: $5\,050$ seconden --- is dat meer of minder dan
        anderhalf uur? Zet het om in uren, minuten en seconden, met twee keer
        een deling met \omterm{def:g3:division:remainder}{rest} (\cref{thm:g6:wholes:division}).
\end{enumerate}
\end{problem}
