\chapter{Spiegelsymmetrie}\label{ch:g6:symmetry}

Vouw een blad papier langs een \omterm{def:g6:lines:objects}{lijn}: elk punt landt op een ander punt, zijn
\omterm{def:g6:symmetry:def}{spiegelbeeld}. Dat vouwen is \emph{\omterm{def:g6:symmetry:def}{spiegelsymmetrie}} (of spiegeling), de eerste
meetkundige transformatie van dit boek --- de tweede, de puntspiegeling, komt in
\cref{ch:g7:central}.

\section{Spiegelingen}

\begin{definition}[Spiegelbeeld van een punt]\label{def:g6:symmetry:def}
Zij $d$ een \omterm{def:g6:lines:objects}{lijn}. Het \emph{spiegelbeeld}\index{spiegelsymmetrie} van een punt
$M$ in $d$ is het punt $M'$ waarvoor $d$ de \omterm{def:g6:symmetry:bisector}{middelloodlijn} van $[MM']$ is: het
\omterm{def:g6:lines:objects}{lijnstuk} $[MM']$ staat \omterm{def:g6:lines:perp}{loodrecht} op $d$, en $d$ snijdt het in zijn midden. Een
punt \emph{op} $d$ is zijn eigen spiegelbeeld.
\end{definition}

\begin{method}[Een spiegelbeeld construeren]\label{met:g6:symmetry:construct}
Om het \omterm{def:g6:symmetry:def}{spiegelbeeld} $M'$ van $M$ in de \omterm{def:g6:lines:objects}{lijn} $d$ te construeren:
\begin{enumerate}
  \item teken de loodlijn op $d$ door $M$ (met de geodriehoek); noem $H$ het
        punt waar hij $d$ snijdt;
  \item meet $MH$ met de passer of de lat;
  \item zet $M'$ op dezelfde loodlijn, aan de andere kant van $d$, met
        $HM' = MH$.
\end{enumerate}
Op ruitjespapier gaat het sneller: tel de vakjes van $M$ tot de as, en tel er
even veel aan de andere kant.
\end{method}

\begin{example}[Op een raster]\label{ex:g6:symmetry:grid}
Is de as een verticale rasterlijn en ligt $M$ $3$ vakjes links ervan, dan ligt
$M'$ $3$ vakjes rechts ervan, op dezelfde rij. Bij een diagonale as werkt het
tellen langs diagonalen --- altijd \omterm{def:g6:lines:perp}{loodrecht} op de as.
\end{example}

\begin{proposition}[Wat een spiegeling bewaart]\label{prop:g6:symmetry:preserved}
Het \omterm{def:g6:symmetry:def}{spiegelbeeld} van een figuur in een \omterm{def:g6:lines:objects}{lijn} is een figuur met \emph{dezelfde
vorm en dezelfde maat}: lengtes, hoeken, omtrekken en \omterm{def:g4:measure:area}{oppervlaktes} blijven
allemaal bewaard. Het \omterm{def:g6:symmetry:def}{spiegelbeeld} van een \omterm{def:g6:lines:objects}{lijnstuk} is een \omterm{def:g6:lines:objects}{lijnstuk}, van een
cirkel een cirkel met dezelfde \omterm{def:g3:shapes:shapes}{radius}, van een \omterm{def:g6:lines:objects}{lijn} een \omterm{def:g6:lines:objects}{lijn}.
\end{proposition}
\admitted

\begin{remark}
Eén ding blijft \emph{niet} bewaard: de omloopsrichting. Een gespiegelde ``b''
wordt een ``d'' --- spiegelschrift. Als je je figuur overtrekt, het
overtrekpapier omklapt en het beeld precies terugkrijgt, dan is de spiegeling
juist.
\end{remark}

\section{Symmetrieassen van een figuur}

\begin{definition}[Symmetrieas]\label{def:g6:symmetry:axis}
Een \omterm{def:g6:lines:objects}{lijn} $d$ is een \emph{symmetrieas}\index{symmetrieas} van een figuur als de
spiegeling in $d$ de figuur precies op zichzelf afbeeldt --- langs $d$ vouwen
laat de twee \omterm{def:g3:division:half}{helften} op elkaar passen.
\end{definition}

{\small Symmetrieassen van bekende figuren. Pas op bij de rechthoek: zijn
diagonalen zijn \emph{geen} symmetrieassen (vouw langs een diagonaal --- de

\begin{example}\label{ex:g6:symmetry:shapes}
Symmetrieassen tellen:
\begin{itemize}
  \item \omterm{def:g6:shapes:triangles}{gelijkbenige driehoek}: $1$ (door de top en het midden van de basis);
        \omterm{def:g6:shapes:triangles}{gelijkzijdige driehoek}: $3$;
  \item rechthoek: $2$ (de twee ``middellijnen'', \emph{niet} de diagonalen);
        \omterm{def:g6:shapes:quadrilaterals}{ruit}: $2$ (haar diagonalen!); vierkant: $4$;
  \item cirkel: elke \omterm{def:g6:lines:objects}{lijn} door het middelpunt --- oneindig veel.
\end{itemize}
\end{example}

\begin{example}[Symmetrie bewijst gelijkheden]\label{ex:g6:symmetry:proves}
Waarom zijn de twee basishoeken van een \omterm{def:g6:shapes:triangles}{gelijkbenige driehoek} gelijk? Vouw de
driehoek langs zijn \omterm{def:g6:symmetry:axis}{symmetrieas}: de linkerhelft landt precies op de
rechterhelft, dus landt de linkerbasishoek op de rechterbasishoek --- ze moeten
dezelfde maat hebben (\cref{prop:g6:symmetry:preserved}: spiegelingen bewaren
hoeken).
\end{example}

\section{Middelloodlijn}

\begin{definition}[Middelloodlijn]\label{def:g6:symmetry:bisector}
De \emph{middelloodlijn}\index{middelloodlijn} van een \omterm{def:g6:lines:objects}{lijnstuk} $[AB]$ is de
\omterm{def:g6:lines:objects}{lijn} die \omterm{def:g6:lines:perp}{loodrecht} op $[AB]$ staat en door zijn midden gaat --- het is de
\omterm{def:g6:symmetry:axis}{symmetrieas} die $A$ en $B$ verwisselt.
\end{definition}

\begin{proposition}[Gelijke afstanden]\label{prop:g6:symmetry:bisector}
Een punt $M$ ligt op de \omterm{def:g6:symmetry:bisector}{middelloodlijn} van $[AB]$ precies dan als het op
dezelfde afstand van $A$ en van $B$ ligt: $MA = MB$.
\end{proposition}
\admitted

\begin{method}[Constructie met de passer]\label{met:g6:symmetry:bisector}
Om de \omterm{def:g6:symmetry:bisector}{middelloodlijn} van $[AB]$ te tekenen zonder te meten:
\begin{enumerate}
  \item zet de passer open op meer dan de \omterm{def:g3:division:half}{helft} van $AB$;
  \item teken twee bogen met $A$ als middelpunt, boven en onder het \omterm{def:g6:lines:objects}{lijnstuk};
  \item teken met \emph{dezelfde} opening twee bogen met $B$ als middelpunt;
  \item verbind de twee snijpunten: die \omterm{def:g6:lines:objects}{lijn} is de \omterm{def:g6:symmetry:bisector}{middelloodlijn} (elk snijpunt
        ligt op gelijke afstand van $A$ en $B$).
\end{enumerate}
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:1}
Neem over op ruitjespapier: een verticale as $d$, een punt $A$ twee vakjes links
ervan, en een punt $B$ vijf vakjes links ervan en één vakje hoger. Construeer
hun \omterm{def:g6:symmetry:def}{spiegelbeelden} $A'$ en $B'$ in $d$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:2}
Teken een \omterm{def:g6:lines:objects}{lijn} $d$ en een punt $M$ op $3$ cm van $d$ (niet op ruitjespapier).
Construeer met geodriehoek en lat het \omterm{def:g6:symmetry:def}{spiegelbeeld} $M'$ van $M$ in $d$, volgens
\cref{met:g6:symmetry:construct}. Hoe ver liggen $M$ en $M'$ van elkaar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:3}
Welke hoofdletters van het alfabet hebben, in hun eenvoudigste vorm geschreven,
een verticale \omterm{def:g6:symmetry:axis}{symmetrieas} (zoals A)? En een horizontale (zoals B)? En beide?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:4}
Hoeveel symmetrieassen heeft: een \omterm{def:g6:shapes:triangles}{gelijkzijdige driehoek}? een rechthoek die geen
vierkant is? een \omterm{def:g6:shapes:quadrilaterals}{ruit} die geen vierkant is? een vierkant? een cirkel? Teken elke
figuur met haar assen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:5}
Waar of niet waar? ``De diagonalen van een rechthoek zijn symmetrieassen van de
rechthoek.'' Leg het uit met een vouwargument of een tekening.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:6}
Een driehoek $ABC$ wordt in een \omterm{def:g6:lines:objects}{lijn} $d$ gespiegeld op $A'B'C'$. Gegeven
$AB = 4$ cm, $\widehat{BAC} = 70^\circ$ en een \omterm{def:g4:measure:perimeter}{omtrek} van $ABC$ van $12$ cm:
geef zonder enige constructie $A'B'$, $\widehat{B'A'C'}$ en de \omterm{def:g4:measure:perimeter}{omtrek} van
$A'B'C'$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:symmetry:7}
Teken een \omterm{def:g6:lines:objects}{lijnstuk} $[AB]$ van $7$ cm en construeer zijn \omterm{def:g6:symmetry:bisector}{middelloodlijn} met de
passer (\cref{met:g6:symmetry:bisector}). Kies een willekeurig punt $M$ erop en
controleer met de lat dat $MA = MB$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:8}
Twee bomen staan in de punten $A$ en $B$ van een (vlakke) tuin. Waar kun je een
fontein plaatsen zodat die op dezelfde afstand van beide bomen staat? Beschrijf
\emph{alle} mogelijke plekken.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:9}
Teken op ruitjespapier de as $d$ (een diagonaal van het raster onder
$45^\circ$) en een klein L-vormig figuurtje aan één kant. Construeer haar
\omterm{def:g6:symmetry:def}{spiegelbeeld} in $d$. (Spiegel elk \omterm{def:g5:solids:def}{hoekpunt}: bij een as van $45^\circ$ landt een
punt $2$ vakjes rechts van de as $2$ vakjes erboven.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:10}
Teken een cirkel met middelpunt $O$, een \omterm{def:g6:lines:objects}{lijn} $d$ door $O$, en een punt $M$ op
de cirkel. Waar ligt het \omterm{def:g6:symmetry:def}{spiegelbeeld} van $M$ in $d$? Leg uit waarom de
spiegeling de cirkel op zichzelf afbeeldt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:symmetry:11}
Leg met de eigenschap van de gelijke afstanden
(\cref{prop:g6:symmetry:bisector}) uit waarom de twee snijpunten van de bogen in
\cref{met:g6:symmetry:bisector} echt op de \omterm{def:g6:symmetry:bisector}{middelloodlijn} van $[AB]$ liggen.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:symmetry:12}
Een biljartbal in punt $A$ moet via een rechte wand $d$ het punt $B$ bereiken
(beide aan dezelfde kant van de wand). Spiegel $B$ in $d$ op $B'$ en teken het
\omterm{def:g6:lines:objects}{lijnstuk} $[AB']$. Leg uit waarom het beste stuitpunt daar ligt waar $[AB']$ de
wand snijdt. (Idee: het pad $A \to P \to B$ is even lang als
$A \to P \to B'$.)
\end{exercise}

\section{Opgave: de fontein, het gebroken bord en de wandelende spiegel}

\begin{problem}[{Weekendopgave --- de drie middelloodlijnen van een driehoek
komen in één punt samen: de cirkel door drie punten, en wat twee spiegels samen
doen}]
\label{pb:g6:symmetry:1}
In \cref{exo:g6:symmetry:8} moest een fontein op gelijke afstand van
\emph{twee} bomen staan, en het antwoord was een hele \omterm{def:g6:lines:objects}{lijn} plekken. In deze
opgave plant je een \emph{derde} boom --- en stort die hele \omterm{def:g6:lines:objects}{lijn} in tot één
perfect bepaald punt. In dat punt zit een schat van meetkunde verstopt: het
levert je de enige cirkel door drie gegeven punten, laat je een heel bord uit één
scherf terugbouwen, en het hoort bij elke driehoek. In het laatste deel doe je
een proef met twee spiegels waarvan de uitkomst de meetkunde van volgend jaar
aankondigt.

\medskip
\noindent\textbf{Deel I --- Twee bomen, en dan een kaart van gebieden.}
\begin{enumerate}
  \item Teken twee punten $A$ en $B$ met $AB = 6$ cm en construeer de
        \omterm{def:g6:symmetry:bisector}{middelloodlijn} van $[AB]$ met de passer
        (\cref{met:g6:symmetry:bisector}). Kies een punt $M$ erop en controleer
        met de lat dat $MA = MB$.
  \item \Cref{prop:g6:symmetry:bisector} zegt twee dingen tegelijk: elk punt
        \emph{op} de \omterm{def:g6:symmetry:bisector}{middelloodlijn} ligt op gelijke afstand van $A$ en $B$, en
        elk punt op gelijke afstand \emph{ligt op} de \omterm{def:g6:symmetry:bisector}{middelloodlijn}. Welke van
        de twee antwoordt op de fonteinvraag van \cref{exo:g6:symmetry:8}? Wat
        zegt de propositie over een punt dat \emph{niet} op de \omterm{def:g6:symmetry:bisector}{middelloodlijn}
        ligt?
  \item Neem een punt $M$ aan dezelfde kant van de \omterm{def:g6:symmetry:bisector}{middelloodlijn} als $A$. Het
        \omterm{def:g6:lines:objects}{lijnstuk} $[MB]$ snijdt de \omterm{def:g6:symmetry:bisector}{middelloodlijn} in een punt $P$. Vergelijk het
        pad $M \to P \to B$ met het pad $M \to P \to A$, en leg uit waarom $M$
        dichter bij $A$ dan bij $B$ ligt: de \omterm{def:g6:symmetry:bisector}{middelloodlijn} verdeelt het blad in
        een ``$A$-kant'' en een ``$B$-kant''.
  \item Twee scholen $A$ en $B$ bedienen een dorp. Elk kind loopt naar de
        dichtstbijzijnde school. Teken een kaartje met $A$ en $B$, en kleur het
        precieze gebied van het dorp dat door school $A$ bediend wordt. Welke
        \omterm{def:g6:lines:objects}{lijn} vormt de grens?
  \item Er opent een derde school $C$ (zet die zo dat de drie scholen een
        driehoek vormen). Construeer de \omterm{def:g6:symmetry:bisector}{middelloodlijn} van $[AB]$ en die van
        $[BC]$, en noem hun snijpunt $O$. Aan welke twee gelijkheden van
        afstanden voldoet $O$, en waarom?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Drie bomen, één punt, één cirkel.}
\begin{enumerate}[resume]
  \item Leid uit vraag 5 af dat $OA = OC$ --- en dus dat $O$ ook op de
        \omterm{def:g6:symmetry:bisector}{middelloodlijn} van $[AC]$ ligt, die je nooit hebt getekend. Besluit:
        \emph{de drie \omterm{def:g6:symmetry:bisector}{middelloodlijnen} van de zijden van een driehoek gaan alle
        drie door één punt.}
  \item Leg uit waarom de cirkel met middelpunt $O$ en \omterm{def:g3:shapes:shapes}{radius} $OA$ ook door $B$
        en door $C$ gaat: het is \emph{de} cirkel door de drie punten --- de
        \emph{omgeschreven cirkel}\index{omgeschreven cirkel} van de driehoek
        $ABC$.
  \item Construeer een driehoek met zijden $6$ cm, $5$ cm en $4$ cm
        (\cref{met:g6:shapes:construct}), en daarna zijn \omterm{pb:g6:symmetry:1}{omgeschreven cirkel}.
        Waarom is het genoeg om maar \emph{twee} \omterm{def:g6:symmetry:bisector}{middelloodlijnen} te tekenen om
        het middelpunt te vinden?
  \item Het gebroken bord: een archeoloog vindt een scherf waarvan alleen een
        stukje van de ronde rand van het bord nog intact is. Beschrijf een recept
        dat het middelpunt en de \omterm{def:g3:shapes:shapes}{radius} van het oorspronkelijke bord terugvindt
        --- en test het: teken met je passer een cirkelboog, ``vergeet'' het
        middelpunt, zet drie punten op de boog en bouw hem terug.
  \item Kunnen er ooit drie punten zijn waar \emph{geen} cirkel door gaat? Stel
        dat $A$, $B$ en $C$ op één \omterm{def:g6:lines:objects}{lijn} liggen. Wat kun je zeggen over de
        \omterm{def:g6:symmetry:bisector}{middelloodlijnen} van $[AB]$ en van $[BC]$
        (\cref{prop:g6:lines:facts})? Besluit.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Assen overal, en de wandelende spiegel.}
\begin{enumerate}[resume]
  \item \Cref{ex:g6:symmetry:shapes} telde $3$ assen voor de \omterm{def:g6:shapes:triangles}{gelijkzijdige
        driehoek} en $4$ voor het vierkant. Vermoed het aantal voor een
        regelmatige \omterm{def:g4:geometry:polygon}{vijfhoek} en een regelmatige \omterm{def:g4:geometry:polygon}{zeshoek}, en beschrijf waar de
        assen in elk geval lopen (een \omterm{def:g3:numbers:evenodd}{oneven} aantal zijden tegen een \omterm{def:g3:numbers:evenodd}{even}
        aantal: de twee situaties verschillen).
  \item Vouw een blad papier één keer en knip er door beide lagen een vorm uit;
        vouw het weer open. Waarom heeft het gat \emph{altijd} een \omterm{def:g6:symmetry:axis}{symmetrieas},
        wat je ook knipt, en welke \omterm{def:g6:lines:objects}{lijn} is die as?
  \item Vouw het blad twee keer --- de tweede vouw \omterm{def:g6:lines:perp}{loodrecht} op de eerste ---
        knip, en vouw het weer open. Hoeveel symmetrieassen heeft het gat
        minstens, en hoeveel kopieën van je knip zie je? (Papieren sneeuwvlokken
        zijn dit idee, nog verder doorgevouwen.)
  \item De wandelende spiegel: teken op ruitjespapier een verticale as $d_1$,
        een vlaggetje $F$ links ervan, en een tweede verticale as $d_2$, $4$
        vakjes rechts van $d_1$. Spiegel $F$ in $d_1$ (beeld $F'$), en spiegel
        $F'$ daarna in $d_2$ (beeld $F''$). Vergelijk $F$ en $F''$: dezelfde
        omloopsrichting of gespiegeld? Hoeveel vakjes is de figuur opgeschoven?
        Vergelijk dat met de afstand tussen de assen.
  \item Dezelfde proef met $d_2$ \emph{\omterm{def:g6:lines:perp}{loodrecht}} op $d_1$: spiegel het
        vlaggetje in de verticale as, en dat beeld daarna in de horizontale.
        Beschrijf hoe $F''$ ligt ten opzichte van $F$ en van het snijpunt van de
        twee assen. (Je hebt net de transformatie van volgend jaar ontdekt:
        \cref{ch:g7:central}.)
\end{enumerate}
\end{problem}
