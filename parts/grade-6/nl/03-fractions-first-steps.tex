\chapter{Breuken: eerste stappen}\label{ch:g6:fractions}

Drie pizza's eerlijk verdelen over vier mensen geeft elke persoon\,\dots\ geen
heel getal pizza's. \omterm{def:g6:fractions:def}{Breuken} zijn de getallen die voor het verdelen zijn
uitgevonden. Dit hoofdstuk bouwt het beeld op: een \omterm{def:g6:fractions:def}{breuk} als deel van een
geheel, als punt op de getallenlijn, en als de precieze uitkomst van een
deling.

\section{Wat een breuk betekent}

\begin{definition}[Breuk]\label{def:g6:fractions:def}
Verdeel een eenheid in $b$ gelijke delen en neem er $a$ van: de hoeveelheid
die je krijgt, is de \emph{breuk}\index{breuk}
\[
\frac{a}{b}
\qquad
\begin{array}{l}
a \text{: de \emph{teller} --- hoeveel delen we nemen;}\\
b \text{: de \emph{noemer} --- in hoeveel delen de eenheid verdeeld is.}
\end{array}
\]
\end{definition}

\begin{example}\label{ex:g6:fractions:morethanone}
Een \omterm{def:g6:fractions:def}{breuk} kan groter zijn dan $1$: $\frac74$ betekent zeven \omterm{def:g3:division:half}{kwarten} --- één
hele eenheid (vier \omterm{def:g3:division:half}{kwarten}) en drie \omterm{def:g3:division:half}{kwarten} erbij:
\[
\frac74 = 1 + \frac34 .
\]
\end{example}

\begin{proposition}[Breuken op de getallenlijn]\label{prop:g6:fractions:line}
Om $\frac ab$ op de getallenlijn te plaatsen, verdeel je elk stuk van $0$ tot
$1$ in $b$ gelijke delen en tel je $a$ delen af vanaf $0$.
\end{proposition}

\begin{theorem}[Een breuk is een deling]\label{thm:g6:fractions:quotient}
De \omterm{def:g6:fractions:def}{breuk} $\frac ab$ is het getal dat, vermenigvuldigd met $b$, $a$ geeft:
\[
\frac ab \times b = a .
\]
Met andere woorden: $\frac ab$ is het precieze \omterm{def:g3:division:remainder}{quotiënt} van $a$ door $b$ ---
ook als de deling ``niet uitkomt''.
\end{theorem}
\admitted

\begin{example}\label{ex:g6:fractions:pizzas}
Drie pizza's voor vier mensen: ieder krijgt $\frac34$ pizza. Controleer met de
stelling: $4$ porties van $\frac34$ maken $\frac34 \times 4 = 3$ pizza's.
Sommige \omterm{def:g6:fractions:def}{breuken} zijn decimale getallen ($\frac34 = 0.75$), andere niet:
$\frac13 = 0.333\dots$ houdt nooit op. De \omterm{def:g6:fractions:def}{breuk} is de \emph{precieze} waarde;
$0.33$ is slechts een benadering.
\end{example}

\section{Gelijke breuken}

\begin{proposition}[Gelijkwaardige breuken]\label{prop:g6:fractions:equal}
Een \omterm{def:g6:fractions:def}{breuk} verandert niet als je haar \omterm{def:g4:fractions:def}{teller} en \omterm{def:g4:fractions:def}{noemer} beide met hetzelfde
getal (niet \omterm{def:g1:counting:zero}{nul}) vermenigvuldigt of erdoor deelt:
\[
\frac{a}{b} = \frac{a \times k}{b \times k} .
\]
\end{proposition}

\begin{proof}[Idee van het bewijs]
Elk deel van de verdeling in $k$ kleinere stukjes knippen vermenigvuldigt
zowel het aantal genomen stukjes als het totale aantal stukjes met $k$, maar
de genomen hoeveelheid blijft dezelfde. Een plaatje zegt het het best:
\end{proof}

\begin{method}[Een breuk vereenvoudigen]\label{met:g6:fractions:simplify}
Zoek een getal dat de \omterm{def:g4:fractions:def}{teller} \emph{en} de \omterm{def:g4:fractions:def}{noemer} deelt, en deel beide erdoor.
Herhaal dat tot er geen gemeenschappelijke \omterm{def:g5:division:multiple}{deler} meer over is:
\[
\frac{12}{18} = \frac{12 \div 2}{18 \div 2} = \frac{6}{9}
= \frac{6 \div 3}{9 \div 3} = \frac{2}{3}.
\]
\end{method}

\section{Een breuk van een hoeveelheid nemen}

\begin{method}[Breuk van een hoeveelheid]\label{met:g6:fractions:ofquantity}
Om $\frac ab$ van een hoeveelheid te berekenen, deel je de hoeveelheid door
$b$ (één deel) en vermenigvuldig je met $a$ (het aantal delen):
\[
\frac ab \text{ van } Q = (Q \div b) \times a .
\]
(Eerst met $a$ vermenigvuldigen en dan door $b$ delen geeft dezelfde uitkomst
--- kies de volgorde waarbij de getallen het mooist blijven.)
\end{method}

\begin{example}\label{ex:g6:fractions:ofquantity}
Reken $\frac35$ van $40$ euro uit, stap voor stap:
\begin{enumerate}
  \item een vijfde van $40$: $40 \div 5 = 8$;
  \item drie vijfden: $8 \times 3 = 24$.
\end{enumerate}
Dus $\frac35$ van $40$ euro is $24$ euro. Controle: de overige $\frac25$ is
$16$ euro, en $24 + 16 = 40$.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g6:fractions:1}
Schrijf bij elke beschrijving de \omterm{def:g6:fractions:def}{breuk} op: een cake in $8$ stukken gesneden
waarvan $3$ opgegeten; een chocoladereep van $12$ blokjes waarvan $7$
opgegeten; een taart in $6$ stukken waarvan alle $6$ opgegeten.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:2}
Teken een strook in $5$ gelijke delen en kleur er $\frac45$ van in. Is
$\frac45$ kleiner of groter dan $1$? En $\frac65$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:3}
Zet op een getallenlijn met streepjes per derde: $\frac13$; \ $\frac53$;
\ $2$; \ $\frac73$. Welke twee van deze getallen liggen even ver van $2$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:4}
Schrijf elke \omterm{def:g6:fractions:def}{breuk} als een heel getal plus een \omterm{def:g6:fractions:def}{breuk} kleiner dan $1$ (zoals
bij $\frac74 = 1 + \frac34$):
\[
\frac{9}{4}, \qquad \frac{13}{5}, \qquad \frac{12}{6}, \qquad
\frac{25}{8} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:5}
Vul aan zodat de \omterm{def:g6:fractions:def}{breuken} gelijk zijn:
\[
\frac{2}{3} = \frac{?}{12}, \qquad
\frac{15}{20} = \frac{3}{?}, \qquad
\frac{7}{10} = \frac{21}{?}, \qquad
\frac{?}{6} = \frac{10}{12}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:6}
Vereenvoudig zo veel mogelijk: $\dfrac{6}{8}$; \quad $\dfrac{15}{25}$;
\quad $\dfrac{18}{24}$; \quad $\dfrac{35}{7}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:7}
Reken uit: $\frac12$ van $86$; \quad $\frac34$ van $60$; \quad $\frac25$ van
$35$; \quad $\frac{7}{10}$ van $250$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g6:fractions:8}
Wat is groter: $\frac12$ of $\frac13$ (van dezelfde cake)? $\frac25$ of
$\frac35$? $\frac34$ of $\frac78$ (schrijf beide met \omterm{def:g4:fractions:def}{noemer} $8$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:fractions:9}
Geef de decimale waarde van de \omterm{def:g6:fractions:def}{breuken} die er een hebben, en zeg welke \omterm{def:g6:fractions:def}{breuk}
er geen heeft: $\frac12$; \ $\frac34$; \ $\frac13$; \ $\frac{7}{10}$;
\ $\frac{9}{4}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:fractions:10}
Een klas heeft $28$ leerlingen. Driekwart van hen komt met de bus naar school,
de \omterm{def:g3:division:remainder}{rest} komt lopend. Hoeveel leerlingen komen lopend? Los stap voor stap op, en
controleer dat je twee groepen samen $28$ zijn.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g6:fractions:11}
Leo gaf $\frac23$ van zijn spaargeld aan een spel van $18$ euro. Hoeveel geld
had Leo daarvoor? (Een derde van zijn spaargeld is de \omterm{def:g3:division:half}{helft} van $18$\,\dots\
denk na over waarom, of zoek eerst de grootte van één deel.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g6:fractions:12}
Een tank is tot $\frac58$ van zijn \omterm{def:g2:measure:units}{inhoud} gevuld. Na het gebruik van $10$
liter is hij half vol. Wat is de \omterm{def:g2:measure:units}{inhoud} van de tank? (Welke \omterm{def:g6:fractions:def}{breuk} van de tank
zijn die $10$ liter?)
\end{exercise}

\section{Opgave: de chocoladereep die nooit opgaat}

\begin{problem}[{Weekendopgave --- helften van helften: de som
$\frac12 + \frac14 + \frac18 + \dots$ kruipt naar $1$ zonder hem ooit te
raken}]
\label{pb:g6:fractions:1}
Neem een chocoladereep. Eet er de \omterm{def:g3:division:half}{helft} van op. Eet dan de \omterm{def:g3:division:half}{helft} van
\emph{wat er over is}. En dan weer de \omterm{def:g3:division:half}{helft} van wat er over is, en zo verder.
Twee dingen lijken tegelijk waar: je krijgt de reep nooit op (er blijft altijd
iets over), en toch eet je bijna alles op. In deze opgave maak je van beide
gevoelens precieze uitspraken over \omterm{def:g6:fractions:def}{breuken} --- en ontmoet je onderweg een
hardloper uit het oude Griekenland en een verdeelpuzzel die je oplost met een
mes dat alleen maar kan halveren.

\medskip
\noindent\textbf{Deel I --- \omterm{def:g3:division:half}{Helften} van \omterm{def:g3:division:half}{helften}.}
De reep is een rechthoek van $16$ gelijke blokjes.
\begin{enumerate}
  \item Eerste hap: je eet de \omterm{def:g3:division:half}{helft} van de reep. Hoeveel blokjes zijn dat?
        Schrijf de hap op twee manieren als \omterm{def:g6:fractions:def}{breuk} van de reep, met \omterm{def:g4:fractions:def}{noemer} $2$
        en met \omterm{def:g4:fractions:def}{noemer} $16$ (\cref{prop:g6:fractions:equal}).
  \item Tweede hap: de \omterm{def:g3:division:half}{helft} van wat er over is. Hoeveel blokjes? Laat met het
        plaatje of met gelijkwaardige \omterm{def:g6:fractions:def}{breuken} zien dat deze hap $\frac14$ van
        de hele reep is: \emph{de \omterm{def:g3:division:half}{helft} van een \omterm{def:g3:division:half}{helft} is een \omterm{def:g3:division:half}{kwart}}.
  \item Derde en vierde hap, met dezelfde regel: geef elke hap in blokjes en
        als \omterm{def:g6:fractions:def}{breuk} van de reep. Wat gebeurt er van de ene hap naar de volgende
        met de \omterm{def:g4:fractions:def}{noemer}?
  \item Hoeveel blokjes zijn er na de vier happen in totaal opgegeten? Welke
        \omterm{def:g6:fractions:def}{breuk} van de reep is dat, en welke \omterm{def:g6:fractions:def}{breuk} blijft over?
\item Vraag 4 zegt, in zestienden geschreven:
        \[
        \frac{8}{16} + \frac{4}{16} + \frac{2}{16} + \frac{1}{16}
        = \frac{15}{16},
        \qquad\text{dus}\qquad
        \frac12 + \frac14 + \frac18 + \frac{1}{16}
        = \frac{15}{16} .
        \]
        Controleer elk van de vier herschrijvingen
        ($\frac12 = \frac{8}{16}$, en zo verder).
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De trap naar $1$.}
Stel je, om verder te kunnen happen, een fijnere reep van $64$ blokjes voor.
\begin{enumerate}[resume]
  \item Noem de grootte van de eerste zes happen in blokjes ($32$, dan\,\dots)
        en controleer dat er na de zesde hap precies één blokje over is. Welke
        \omterm{def:g6:fractions:def}{breuk} van de reep blijft er over?
  \item Welke \omterm{def:g6:fractions:def}{breuk} van de reep blijft er na elke hap over? Schrijf het rijtje
        op ($\frac12$, $\frac14$, \dots) tot de zesde hap, en beschrijf het
        patroon. Als een verbeeld mes zelfs het laatste blokje kon halveren,
        wat zou er dan na een zevende hap overblijven?
  \item Leg uit waarom het opgegeten totaal \emph{nooit} $1$ kan bereiken, hoe
        vaak je ook hapt. (Vergelijk met de twintig negens van
        \cref{pb:g6:decimals:1}: hetzelfde verhaal?)
  \item En toch komt het totaal voorbij elk doel onder $1$. Een vriend daagt je
        uit om meer dan $\frac{99}{100}$ van de reep op te eten. Wat is
        kleiner, $\frac{1}{128}$ of $\frac{1}{100}$ --- en na welke hap is de
        uitdaging gewonnen?
  \item Zet op een getallenlijn van $0$ tot $1$ met streepjes per zestiende de
        opgegeten totalen na elk van de eerste vier happen: $\frac12$,
        $\frac34$, $\frac78$, $\frac{15}{16}$. Waar landt elk nieuw totaal,
        vergeleken met het vorige en met $1$?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De hardloper van Zeno, en een mes dat alleen
halveert.}
\begin{enumerate}[resume]
  \item Vierentwintig eeuwen geleden vertelde de Griekse filosoof Zeno dit
        verhaal: om een muur te bereiken, moet een hardloper eerst de \omterm{def:g3:division:half}{helft} van
        de afstand afleggen, dan de \omterm{def:g3:division:half}{helft} van wat overblijft, dan weer de \omterm{def:g3:division:half}{helft}
        van wat overblijft\,\dots\ ``dus bereikt de hardloper de muur nooit!''
        Zeg met de vragen 8 en 9 wat er \emph{juist} is in het verhaal van
        Zeno, en waar de valstrik zit. (Heeft de hardloper echt eindeloos veel
        tijd nodig voor al die stappen?)
  \item De verdeelpuzzel: verdeel $15$ identieke chocoladereepjes eerlijk over
        $16$ kinderen, met alleen halveringen (van een reep, van een halve
        reep, van een \omterm{def:g3:division:half}{kwart} reep\,\dots). Beschrijf hoe je snijdt: hoeveel
        reepjes worden in \omterm{def:g3:division:half}{helften} gesneden, hoeveel in \omterm{def:g3:division:half}{kwarten}, in achtsten, in
        zestienden, zodat elk kind één stuk van elke maat krijgt.
  \item Controleer je verdeling: schrijf het deel van elk kind als een \omterm{def:g1:addition:def}{som} van
        \omterm{def:g6:fractions:def}{breuken}, reken het in zestienden uit, en ga na dat de $16$ delen samen
        precies de $15$ reepjes opgebruiken.
  \item In de klas ernaast worden $7$ reepjes eerlijk over $8$ kinderen
        verdeeld. Welk kind krijgt meer chocolade --- een van de $16$ of een van
        de $8$? (Schrijf beide delen met \omterm{def:g4:fractions:def}{noemer} $16$,
        \cref{exo:g6:fractions:8}.)
  \item De finale: een vriendin beweert dat je met \emph{oneindig} veel happen
        \emph{precies} de hele reep zou opeten. Teken een vierkant en kleur de
        happen erin: de \omterm{def:g3:division:half}{helft} van het vierkant, dan een \omterm{def:g3:division:half}{kwart}, dan een
        achtste\,\dots\ Wat valt je op aan het niet-gekleurde hoekje? Formuleer
        zorgvuldig de twee ware feiten van deze hele opgave: wat er na elk
        eindig aantal happen overblijft, en hoe klein die \omterm{def:g3:division:remainder}{rest} wordt. (Wat een
        ``oneindige \omterm{def:g1:addition:def}{som}'' precies betekent, is een verhaal voor het
        bovenbouwvolume.)
\end{enumerate}
\end{problem}
