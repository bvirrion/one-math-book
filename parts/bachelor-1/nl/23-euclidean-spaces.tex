\chapter{Euclidische ruimten}\label{ch:b1:euclid}

Een \omterm{def:b1:euclid:def}{inwendig product} aan een reële \omterm{def:b1:vspaces:def}{vectorruimte} toevoegen koopt de
meetkundige begrippen --- lengten, hoeken, orthogonaliteit,
afstanden --- en één stelling die boven het hoofdstuk uittorent:
elke \omterm{def:b1:vspaces:subspace}{deelruimte} laat een \omterm{thm:b1:euclid:projection}{orthogonale projectie} toe, berekenbaar met
Gram--Schmidt, die de kortste afstand realiseert. De \omterm{def:b1:euclid:isometry}{isometrieën} van
het vlak sluiten het hoofdstuk en de meetkunde van het jaar af.

Overal is $E$ een \emph{reële} \omterm{def:b1:vspaces:def}{vectorruimte}.

\section{Inwendige producten}

\begin{definition}\label{def:b1:euclid:def}
Een \emph{inwendig product}\index{inwendig product} op $E$ is een
\omterm{def:b1:logic:map}{afbeelding} $\langle\cdot,\cdot\rangle \colon E \times E \to \R$ die
bilineair, symmetrisch en positief definiet is ($\langle x,
x\rangle > 0$ voor $x \neq 0$). Een \omterm{def:b1:findim:def}{eindigdimensionale} ruimte die zo
is uitgerust, heet een \emph{euclidische
ruimte}\index{euclidische ruimte}. De \emph{norm} van $x$ is
$\norm{x} = \sqrt{\langle x, x\rangle}$, en $d(x, y) = \norm{x -
y}$.
\end{definition}

\begin{example}\label{ex:b1:euclid:examples}
Op $\R^n$: het canonieke product $\langle x, y\rangle = \sum x_i
y_i$. Op $C(\intcc{a}{b})$: $\langle f, g \rangle = \int_a^b fg$ (de
positieve definietheid is \cref{thm:b1:integration:props}~(4)). Op
$\R_n[X]$: $\langle P, Q\rangle = \int_0^1 PQ$, of $\sum_{i}
P(x_i)Q(x_i)$ over $n+1$ verschillende punten.
\end{example}

\begin{example}[De hoek tussen twee veeltermen]
\label{ex:b1:euclid:polyangle}
Zodra een \omterm{def:b1:euclid:def}{inwendig product} is gekozen, hebben \emph{alle} twee
vectoren ongelijk aan nul een hoek, via $\cos\theta =
\frac{\langle x, y\rangle}{\norm x\,\norm y}$ (een geoorloofde
cosinus dankzij Cauchy--Schwarz). Voor $X$ en $X^2$ met
$\int_0^1$:
\[
\langle X, X^2\rangle = \frac14, \qquad
\norm X = \frac1{\sqrt3}, \qquad \norm{X^2} = \frac1{\sqrt5},
\qquad
\cos\theta = \frac{1/4}{1/\sqrt{15}} = \frac{\sqrt{15}}{4}
\approx 0.968 :
\]
een hoek van ongeveer $14.5$ graden --- op $\intcc{0}{1}$ zijn de
grafieken van $x$ en $x^2$ ``bijna evenwijdig'' in
kwadratisch-gemiddelde zin, en daarom laat het verwijderen van die
gedeelde richting (Gram--Schmidt, hieronder) alleen de kleine
correctie $X^2 - X + \frac16$ over.
\end{example}

\begin{theorem}[Cauchy--Schwarz; eigenschappen van de norm]\label{thm:b1:euclid:cs}
Voor alle $x, y \in E$
geldt\index{Cauchy--Schwarz-ongelijkheid}
\[
\abs{\langle x, y\rangle} \leq \norm x\, \norm y ,
\]
met gelijkheid dan en slechts dan als $x, y$ evenredig zijn.
Bijgevolg voldoet $\norm\cdot$ aan de driehoeksongelijkheid
$\norm{x + y} \leq \norm x + \norm y$ (en $\norm{\lambda x} =
\abs\lambda \norm x$, $\norm x = 0 \iff x = 0$). Bovendien:
\[
\norm{x+y}^2 = \norm x^2 + 2\langle x, y\rangle + \norm y^2,
\qquad
\langle x, y \rangle = \tfrac14\bigl(\norm{x+y}^2 -
\norm{x-y}^2\bigr) .
\]
\end{theorem}

\begin{proof}
Is $y = 0$, dan is alles triviaal. Anders is de kwadratische
uitdrukking $t \mapsto \norm{x + ty}^2 = \norm x^2 + 2t\langle x,
y\rangle + t^2 \norm y^2$ voor alle $t$ $\geq 0$: haar discriminant
is $\leq 0$, en dat is Cauchy--Schwarz; gelijkheid betekent een
dubbele wortel $t_0$, dus $x + t_0 y = 0$ (definietheid):
evenredigheid. Driehoeksongelijkheid: werk uit,
\[
\norm{x + y}^2 = \norm x^2 + 2\langle x, y\rangle + \norm y^2
\leq \norm x^2 + 2\norm x\,\norm y + \norm y^2
= \bigl(\norm x + \norm y\bigr)^2 ,
\]
waarbij de middelste stap Cauchy--Schwarz is; gelijkheid dwingt
$\langle x, y\rangle = \norm x\norm y$ af, het \emph{positieve}
gelijkheidsgeval, dus evenredigheid met een niet-negatieve
verhouding --- meetkundig ontaardt de driehoek alleen wanneer de
twee vectoren dezelfde kant op wijzen. De laatste twee
identiteiten zijn rechtstreekse uitwerkingen (de tweede, de
\emph{polarisatie-identiteit}, wint het product terug uit de
norm).
\end{proof}

\section{Orthogonaliteit}

\begin{definition}\label{def:b1:euclid:orthogonal}
$x \perp y$ wanneer $\langle x, y \rangle = 0$. Een familie heet
\emph{orthogonaal} wanneer haar vectoren paarsgewijs orthogonaal
zijn, en \emph{orthonormaal}\index{orthonormale familie} wanneer
bovendien elk van hen norm $1$ heeft. Het \emph{orthogonaal
complement} van een \omterm{def:b1:vspaces:subspace}{deelruimte} $F$ is
\[
F^{\perp} = \{x \in E : \forall y \in F,\ \langle x, y\rangle =
0\},
\]
een \omterm{def:b1:vspaces:subspace}{deelruimte} van $E$.\index{orthogonaal complement}
\end{definition}

\begin{proposition}\label{prop:b1:euclid:orthfree}
(Pythagoras\index{stelling van Pythagoras}) Is $x \perp y$, dan is
$\norm{x+y}^2 = \norm x^2 + \norm y^2$. Een \omterm{def:b1:euclid:orthogonal}{orthogonale} familie van
vectoren ongelijk aan nul is \omterm{def:b1:vspaces:free}{vrij}. In een \omterm{def:b1:euclid:orthogonal}{orthonormale} \omterm{def:b1:vspaces:free}{basis} $(e_1,
\dots, e_n)$ luiden de \omterm{prop:b1:vspaces:coordinates}{coördinaten} en de producten
\[
x = \sum_{i} \langle x, e_i\rangle\, e_i,
\qquad
\langle x, y \rangle = \sum_i \langle x, e_i\rangle \langle y,
e_i\rangle,
\qquad
\norm x^2 = \sum_i \langle x, e_i\rangle^2 .
\]
\end{proposition}

\begin{proof}
Pythagoras: werk uit. Vrijheid: neem $\langle\,\cdot\,, x_j\rangle$
van een nulcombinatie: $\lambda_j \norm{x_j}^2 = 0$. \omterm{prop:b1:vspaces:coordinates}{Coördinaten}:
schrijf $x = \sum \lambda_i e_i$ en neem het product met $e_j$:
$\lambda_j = \langle x, e_j\rangle$; de twee formules volgen uit de
bilineariteit.
\end{proof}

\begin{example}[Orthonormale coördinaten, met een
Parseval-controle]
\label{ex:b1:euclid:parsevalcheck}
Ontwikkel $x = (1, 2, 3)$ in de \omterm{def:b1:euclid:orthogonal}{orthonormale} \omterm{def:b1:vspaces:free}{basis} van
\cref{exo:b1:euclid:3},
\[
e_1 = \tfrac{1}{\sqrt2}(1,1,0), \quad
e_2 = \tfrac{1}{\sqrt6}(1,-1,2), \quad
e_3 = \tfrac{1}{\sqrt3}(-1,1,1).
\]
Er valt geen stelsel op te lossen --- drie \omterm{def:b1:euclid:def}{inwendige producten}:
\[
\langle x, e_1\rangle = \frac{3}{\sqrt2},
\qquad
\langle x, e_2\rangle = \frac{1 - 2 + 6}{\sqrt6} =
\frac{5}{\sqrt6},
\qquad
\langle x, e_3\rangle = \frac{-1 + 2 + 3}{\sqrt3} =
\frac{4}{\sqrt3}.
\]
Certificering met de normformule van de propositie:
\[
\frac{9}{2} + \frac{25}{6} + \frac{16}{3}
= \frac{27 + 25 + 32}{6} = 14 = \norm{x}^2 = 1 + 4 + 9 .
\]
Deze controle met de som van de kwadraten van de \omterm{prop:b1:vspaces:coordinates}{coördinaten} (een
eindige identiteit van Parseval) kost seconden en vangt teken- en
normalisatiefouten vrijwel zeker op --- maak er een gewoonte van
telkens wanneer een \omterm{def:b1:euclid:orthogonal}{orthonormale} ontwikkeling wordt berekend; haar
oneindigdimensionale versie, voor de fouriercoëfficiënten van
\cref{ex:b1:euclid:trigortho}, is een stelling van het volume van
bachelorjaar 3.
\end{example}

\begin{theorem}[Gram--Schmidt]\label{thm:b1:euclid:gramschmidt}
Elke \omterm{def:b1:euclid:def}{euclidische ruimte} heeft \omterm{def:b1:euclid:orthogonal}{orthonormale} \omterm{def:b1:vspaces:free}{bases}. Expliciet: uit
elke \omterm{def:b1:vspaces:free}{basis} $(v_1, \dots, v_n)$ brengt het
recept\index{Gram--Schmidt-procedure}
\[
w_k = v_k - \sum_{i=1}^{k-1} \langle v_k, e_i\rangle\, e_i ,
\qquad
e_k = \frac{w_k}{\norm{w_k}}
\]
een \omterm{def:b1:euclid:orthogonal}{orthonormale} \omterm{def:b1:vspaces:free}{basis} $(e_1, \dots, e_n)$ voort met
$\operatorname{Vect}(e_1, \dots, e_k) = \operatorname{Vect}(v_1,
\dots, v_k)$ voor elke $k$.
\end{theorem}

\begin{proof}
Inductie op $k$. Neem aan dat $(e_1, \dots, e_{k-1})$ \omterm{def:b1:euclid:orthogonal}{orthonormaal}
is en $\operatorname{Vect}(v_1, \dots, v_{k-1})$ opspant: de vector
$w_k$ is per constructie \omterm{def:b1:euclid:orthogonal}{orthogonaal} met elke $e_j$ ($j < k$)
($\langle w_k, e_j \rangle = \langle v_k, e_j\rangle - \langle v_k,
e_j\rangle$), en $w_k \neq 0$ omdat $v_k \notin
\operatorname{Vect}(v_1, \dots, v_{k-1})$. Normaliseren behoudt de
orthogonaliteit; de \omterm{def:b1:logic:statement}{uitspraak} over het \omterm{def:b1:vspaces:span}{opspansel} geldt omdat $e_k$
een combinatie is van $v_k$ en de eerdere $e_i$, op omkeerbare
wijze.
\end{proof}

\begin{example}[Gram--Schmidt op veeltermen, volledig]
\label{ex:b1:euclid:legendre}
Orthonormaliseer $(1, X, X^2)$ in $\R_2[X]$ met $\langle P,
Q\rangle = \int_0^1 PQ$. \emph{Stap 1}: $\norm{1}^2 = 1$, dus $e_1
= 1$. \emph{Stap 2}: $w_2 = X - \langle X, 1\rangle\,1 = X -
\frac12$, en $\norm{w_2}^2 = \int_0^1\bigl(x - \frac12\bigr)^2 \dd
x = \frac1{12}$: $e_2 = \sqrt{12}\,\bigl(X - \frac12\bigr)$.
\emph{Stap 3}: $\langle X^2, e_1\rangle = \frac13$ en
\[
\langle X^2, e_2\rangle
= \sqrt{12}\int_0^1 x^2\Bigl(x - \frac12\Bigr)\dd x
= \frac{\sqrt{12}}{12},
\qquad\text{dus}\qquad
w_3 = X^2 - \frac13 - \Bigl(X - \frac12\Bigr)
= X^2 - X + \frac16 .
\]
Haar norm werd berekend in \cref{exo:b1:euclid:9}: $\norm{w_3}^2 =
\frac1{180}$, waaruit $e_3 = \sqrt{180}\,\bigl(X^2 - X +
\frac16\bigr)$. De \omterm{def:b1:poly:def}{veeltermen} $1$, $X - \frac12$, $X^2 - X +
\frac16$ zijn, op een schaalfactor na, de eerste
\emph{\omterm{def:b1:poly:def}{Legendre-veeltermen}} van het \omterm{prop:b1:reals:intervals}{interval} $\intcc{0}{1}$; de
constructie gaat één graad tegelijk verder, met elke nieuwe
\omterm{def:b1:poly:def}{veelterm} \omterm{def:b1:euclid:orthogonal}{orthogonaal} met al haar voorgangers. Merk op hoe het
algoritme eerder werk hergebruikt: de \omterm{def:b1:linmaps:projection}{projectie} die in stap 3 wordt
afgetrokken, is precies de beste affiene benadering van $X^2$ die
in \cref{ex:b1:euclid:bestapprox} is gevonden --- Gram--Schmidt
\emph{is} herhaalde \omterm{thm:b1:euclid:projection}{orthogonale projectie}.
\end{example}

\begin{theorem}[Orthogonale projectie]\label{thm:b1:euclid:projection}
Zij $F$ een \omterm{def:b1:vspaces:subspace}{deelruimte} van de \omterm{def:b1:euclid:def}{euclidische ruimte} $E$. Dan is
\[
E = F \oplus F^{\perp},
\]
en wordt de bijbehorende \omterm{def:b1:linmaps:projection}{projectie} $p_F$ op $F$ (de
\emph{orthogonale projectie}\index{orthogonale projectie}) in elke
\omterm{def:b1:euclid:orthogonal}{orthonormale} \omterm{def:b1:vspaces:free}{basis} $(e_1, \dots, e_k)$ van $F$ gegeven door $p_F(x)
= \sum_i \langle x, e_i\rangle e_i$. Zij realiseert de afstand tot
$F$: voor alle $y \in F$ is
\[
\norm{x - p_F(x)} \leq \norm{x - y},
\]
met gelijkheid alleen voor $y = p_F(x)$; men schrijft $d(x, F) =
\norm{x - p_F(x)}$.
\end{theorem}

\begin{proof}
Neem een \omterm{def:b1:euclid:orthogonal}{orthonormale} \omterm{def:b1:vspaces:free}{basis} $(e_i)_{i \leq k}$ van $F$
(\cref{thm:b1:euclid:gramschmidt} binnen $F$) en stel $\pi(x) =
\sum \langle x, e_i\rangle e_i \in F$. Dan is $x - \pi(x) \perp
e_j$ voor elke $j$ (dezelfde opheffing als hierboven), dus $x -
\pi(x) \in F^\perp$: $E = F + F^\perp$. En $F \cap F^\perp =
\{0\}$: zo'n vector voldoet aan $\langle x, x\rangle = 0$. De som
is dus direct en $\pi = p_F$.

Afstand: ontbind voor $y \in F$ het verschil $x - y = (x - p_F(x)) +
(p_F(x) - y)$ in \omterm{def:b1:euclid:orthogonal}{orthogonale} stukken ($F^\perp$ en $F$); Pythagoras
geeft
\[
\norm{x - y}^2 = \norm{x - p_F(x)}^2 + \norm{p_F(x) - y}^2
\geq \norm{x - p_F(x)}^2,
\]
met gelijkheid dan en slechts dan als $y = p_F(x)$.
\end{proof}

\begin{example}[Projecties verlengen nooit]
\label{ex:b1:euclid:bessel}
Toepassing van Pythagoras op de splitsing $x = p_F(x) + (x -
p_F(x))$ geeft
\[
\norm{p_F(x)}^2 = \norm x^2 - \norm{x - p_F(x)}^2 \leq
\norm x^2 ,
\]
met gelijkheid dan en slechts dan als $x \in F$. In een
\omterm{def:b1:euclid:orthogonal}{orthonormale} \omterm{def:b1:vspaces:free}{basis} $(e_1, \dots, e_k)$ van $F$ luidt dit $\sum_{i
\leq k}\langle x, e_i\rangle^2 \leq \norm x^2$ (een
\emph{ongelijkheid van Bessel}): hoeveel \omterm{def:b1:euclid:orthogonal}{orthonormale} richtingen
men ook meet, de kwadraten van de \omterm{prop:b1:vspaces:coordinates}{coördinaten} overtreffen nooit het
kwadraat van de lengte --- vergelijk de exacte gelijkheid van
\cref{ex:b1:euclid:parsevalcheck} wanneer de familie een volledige
\omterm{def:b1:vspaces:free}{basis} is. Deze ongelijkheid van één regel maakt de
fouriercoëfficiënten sommeerbaar in het volume van bachelorjaar 3;
hier verklaart zij al waarom het toevoegen van meer basisfuncties
aan een aanpassing met kleinste kwadraten de rest alleen kan
verkleinen.
\end{example}

\begin{example}[Beste kwadratische benadering]\label{ex:b1:euclid:bestapprox}
In $C(\intcc{0}{1})$ met $\langle f, g\rangle = \int_0^1 fg$ is de
\omterm{def:b1:poly:def}{veelterm} van graad $\leq 1$ die het dichtst bij $f(x) = x^2$ ligt
in de bijbehorende (kwadratisch-gemiddelde) afstand gelijk aan
$p_F(f)$ met $F = \R_1[X]$. Gram--Schmidt op $(1, X)$: $e_1 = 1$,
$w_2 = X - \frac12$, $\norm{w_2}^2 = \int_0^1 (x - \frac12)^2 =
\frac{1}{12}$, $e_2 = \sqrt{12}\,(X - \tfrac12)$. Dan is
\[
p_F(f) = \langle f, e_1\rangle e_1 + \langle f, e_2\rangle e_2
= \frac13 + 12\Bigl(\int_0^1 x^2\bigl(x - \tfrac12\bigr)\dd
x\Bigr)\bigl(X - \tfrac12\bigr)
= X - \frac{1}{6},
\]
met $\int_0^1 x^2(x - \frac12)\dd x = \frac14 - \frac16 =
\frac{1}{12}$. Het idee van de ``kleinste kwadraten'' in één regel
\omterm{def:b1:linmaps:def}{lineaire} algebra.
\end{example}

\begin{method}[Drie wegen naar een afstand $d(x, F)$]
\label{met:b1:euclid:distance}
\begin{enumerate}
  \item \emph{\omterm{def:b1:euclid:orthogonal}{Orthonormale} \omterm{def:b1:vspaces:free}{basis} van $F$}: dan is $p_F(x) = \sum_i
        \langle x, e_i\rangle e_i$ en, volgens Pythagoras,
        \[
        d(x, F)^2 = \norm{x}^2 - \norm{p_F(x)}^2
        = \norm x^2 - \sum_i \langle x, e_i\rangle^2 ,
        \]
        vaak goedkoper dan $x - p_F(x)$ zelf berekenen.
  \item \emph{Normaalvergelijkingen}: los met een willekeurige
        voortbrengende familie van $F$ de vergelijkingen $\langle x
        - p, v_j\rangle = 0$ op voor de coëfficiënten van $p$
        (\cref{exo:b1:euclid:5}) --- geen orthonormalisatie nodig.
  \item \emph{Via het complement}: is $F^\perp$ kleiner dan $F$
        (bijvoorbeeld $F$ een \omterm{def:b1:linmaps:forms}{hypervlak} en $F^\perp$ een rechte
        $\operatorname{Vect}(n)$), projecteer dan op $F^\perp$ in
        plaats daarvan:
        \[
        d(x, F) = \norm{p_{F^\perp}(x)} =
        \frac{\abs{\langle x, n\rangle}}{\norm n} ,
        \]
        wat de klassieke formule voor de afstand tot een vlak is
        (\cref{exo:b1:multivar:8} gebruikt haar).
\end{enumerate}
Weg 3 is een bijzonder geval van een algemene reflex: projecteer
altijd op die van $F$, $F^\perp$ met de kleinste dimensie.
\end{method}

\begin{example}[Goniometrische orthogonaliteit: een voorproef van
Fourier]
\label{ex:b1:euclid:trigortho}
Op $C(\intcc{0}{2\pi})$ met $\langle f, g\rangle =
\frac1\pi\int_0^{2\pi} fg$ is de familie
\[
\Bigl(\frac{1}{\sqrt2},\ \cos x,\ \sin x,\ \cos 2x,\ \sin 2x,\
\dots\Bigr)
\]
\omterm{def:b1:euclid:orthogonal}{orthonormaal}: zo is bijvoorbeeld $\langle \cos px, \cos qx\rangle =
\frac1\pi\int_0^{2\pi}\cos px\cos qx\,\dd x = 0$ voor $p \neq q$
(lineariseer het product tot $\frac12[\cos(p{-}q)x +
\cos(p{+}q)x]$ en integreer over volledige perioden), terwijl
$\frac1\pi\int_0^{2\pi}\cos^2 px\,\dd x = 1$. De \omterm{thm:b1:euclid:projection}{orthogonale
projectie} op het \omterm{def:b1:vspaces:span}{opspansel} van de eerste $2N + 1$ van deze functies
heeft dus als \omterm{prop:b1:vspaces:coordinates}{coördinaten} $\langle f, e_i\rangle$ --- \omterm{thm:b1:integration:def}{integralen}
tegen cosinussen en sinussen. Dat zijn de
\emph{fouriercoëfficiënten} van $f$, en de \omterm{def:b1:linmaps:projection}{projectie} is haar beste
goniometrische benadering in kwadratisch gemiddelde; het volume van
bachelorjaar 3 bestudeert hun convergentie. De orthogonaliteit doet
al het werk: de formules voor de coëfficiënten zijn woordelijk
\cref{thm:b1:euclid:projection}.
\end{example}

\section{Isometrieën van het vlak}

\begin{definition}\label{def:b1:euclid:isometry}
Een endomorfisme $u$ van een \omterm{def:b1:euclid:def}{euclidische ruimte} is een
\emph{isometrie}\index{isometrie} (of \omterm{def:b1:euclid:orthogonal}{orthogonale} \omterm{def:b1:logic:map}{afbeelding})
wanneer het de norm bewaart: $\norm{u(x)} = \norm x$ voor alle $x$
--- equivalent (polarisatie) bewaart het het \omterm{def:b1:euclid:def}{inwendige product};
equivalent voldoet haar matrix $A$ in een \omterm{def:b1:euclid:orthogonal}{orthonormale} \omterm{def:b1:vspaces:free}{basis} aan
$A^{\mathsf T} A = I$. De isometrieën vormen een \omterm{def:b1:structures:group}{groep}, de
\omterm{def:b1:euclid:orthogonal}{orthogonale} \omterm{def:b1:structures:group}{groep} $O(E)$.
\end{definition}

\begin{example}[Een isometrie op het zicht herkennen]
\label{ex:b1:euclid:recognize}
Is $A = \dfrac15\begin{pmatrix} 3 & -4\\ 4 & 3\end{pmatrix}$
\omterm{def:b1:euclid:orthogonal}{orthogonaal}? Kolommen: normen $\frac15\sqrt{9 + 16} = 1$ en
$\frac15\sqrt{16 + 9} = 1$; product $\frac1{25}(3\cdot(-4) +
4\cdot3) = 0$. Ja --- en $\det A = \frac{9 + 16}{25} = 1$, dus is
zij de rotatie $R_\theta$ met $\cos\theta = \frac35$ en
$\sin\theta = \frac45$ (de ``$3$-$4$-$5$-rotatie'', waarvan de
hoek geen opmerkelijke breuk van $\pi$ is). Daarentegen heeft $B =
\frac{1}{\sqrt2}\begin{pmatrix} 1 & 1\\ 0 & 1\end{pmatrix}$ wel
het uiterlijk van een eenheidsdeterminant maar een eerste kolom
zonder norm $1$ ($\frac1{\sqrt2}$): niet \omterm{def:b1:euclid:orthogonal}{orthogonaal} --- een
determinant $\pm1$ alleen certificeert niets, de kolommen moeten
worden gecontroleerd.
\end{example}

\begin{theorem}[Isometrieën van het vlak]\label{thm:b1:euclid:plane}
In een \omterm{def:b1:euclid:orthogonal}{orthonormale} \omterm{def:b1:vspaces:free}{basis} van een euclidisch vlak zijn de matrices
van de \omterm{def:b1:euclid:isometry}{isometrieën} precies
\[
R_\theta = \begin{pmatrix}
\cos\theta & -\sin\theta\\ \sin\theta & \cos\theta
\end{pmatrix}
\quad (\text{rotatie over hoek } \theta,\ \det = 1),
\]
\[
S_\theta = \begin{pmatrix}
\cos\theta & \sin\theta\\ \sin\theta & -\cos\theta
\end{pmatrix}
\quad (\det = -1),
\]
waarbij de laatste de spiegeling is in de rechte die een hoek
$\frac\theta2$ maakt met de eerste basisvector.
\end{theorem}

\begin{proof}
Zij $A = \begin{pmatrix} a & c\\ b & d\end{pmatrix}$ met $A^{\mathsf
T}A = I$: de kolommen hebben norm $1$ en zijn \omterm{def:b1:euclid:orthogonal}{orthogonaal}. De eerste
kolom is $(\cos\theta, \sin\theta)$ voor zekere $\theta$; de tweede,
met norm $1$ en \omterm{def:b1:euclid:orthogonal}{orthogonaal} daarmee, is $\pm(-\sin\theta,
\cos\theta)$. Het teken $+$ geeft $R_\theta$, het teken $-$ geeft
$S_\theta$. Men gaat na dat $S_\theta^2 = I$ en dat de vector
$(\cos\frac\theta2, \sin\frac\theta2)$ wordt vastgehouden terwijl
zijn \omterm{def:b1:euclid:orthogonal}{orthogonale} wordt omgekeerd: een spiegeling. (En $R_\alpha
R_\beta = R_{\alpha+\beta}$: de rotatiegroep is de hoekengroep ---
vergelijk \cref{thm:b1:complex:funceq}.)
\end{proof}

\begin{remark}[Veelgemaakte fouten]
\label{rem:b1:euclid:pitfalls}
\emph{De projectieformule vereist een \omterm{def:b1:euclid:orthogonal}{orthonormale} \omterm{def:b1:vspaces:free}{basis}}: voor een
louter voortbrengende familie $(v_i)$ van $F$ is de som
$\sum_i\langle x, v_i\rangle v_i$ \emph{niet} $p_F(x)$ (toets $F =
\R^2$, $v_1 = e_1$, $v_2 = e_1 + e_2$); los bij een
\omterm{def:b1:euclid:orthogonal}{niet-orthonormale} familie in plaats daarvan de
normaalvergelijkingen op (\cref{met:b1:euclid:distance}~(2)).
\emph{\omterm{def:b1:euclid:orthogonal}{Orthogonale} families moeten $0$ vermijden om \omterm{def:b1:vspaces:free}{vrij} te zijn}:
de nulvector is \omterm{def:b1:euclid:orthogonal}{orthogonaal} met alles, ook met zichzelf --- de
vrijheid in \cref{prop:b1:euclid:orthfree} vereist vectoren
ongelijk aan nul.
\emph{$F^\perp$ hangt af van het \omterm{def:b1:euclid:def}{inwendige product}}: in $\R_1[X]$
is het complement van $\operatorname{Vect}(X)$ voor $\int_0^1 PQ$
niet de constanten maar $\operatorname{Vect}(1 - \frac32 X)$ ---
bereken $\int_0^1 x(1 - \frac32 x) = \frac12 - \frac12 = 0$;
``loodrecht'' is betekenisloos zolang het product niet is
benoemd.
\emph{Werk $\norm{x + y}$ niet \omterm{def:b1:linmaps:def}{lineair} uit}: de juiste identiteit
is $\norm{x+y}^2 = \norm x^2 + 2\langle x, y\rangle + \norm y^2$;
de kruisterm verdwijnt alleen bij orthogonaliteit (Pythagoras), en
de driehoeksongelijkheid is een \emph{ongelijkheid}.
\emph{Eenheidsvectoren naar eenheidsvectoren sturen is niet
genoeg}: $u(x, y) = (x + y,\ 0)$ beeldt beide canonieke
basisvectoren af op de eenheidsvector $(1, 0)$, en toch is
$\norm{u(1,1)} = 2 \neq \sqrt2$: geen \omterm{def:b1:euclid:isometry}{isometrie}. De definitie eist
$\norm{u(x)} = \norm x$ voor \emph{alle} $x$; in matrixtaal
$A^{\mathsf T}A = I$, dus kolommen die norm $1$ hebben \emph{en
paarsgewijs \omterm{def:b1:euclid:orthogonal}{orthogonaal}} zijn --- beide voorwaarden, samen
gecontroleerd.
\end{remark}

\begin{remark}[Waar het inwendige product heen gaat]
\label{rem:b1:euclid:whereused}
De \omterm{thm:b1:euclid:projection}{orthogonale projectie} is de meest toegepaste stelling van het
hoofdstuk: zij ligt ten grondslag aan de kleinste kwadraten (de
weekendopgave van \cref{ch:b1:multivar} bouwt er regressierechten
op), aan de fouriercoëfficiënten
(\cref{ex:b1:euclid:trigortho}) en aan de normaalvergelijkingen
van \cref{exo:b1:euclid:5}, die de numerieke analyse op grote
schaal oplost. De classificatie $R_\theta / S_\theta$ wordt
hieronder voltooid: de weekendopgave classificeert \emph{alle}
afstandsbewarende transformaties van het vlak, \omterm{def:b1:linmaps:def}{lineair} of niet, en
hun eindige \omterm{def:b1:structures:group}{groepen} --- de wiskunde van rozetten en regelmatige
veelhoeken. In het volume van bachelorjaar 2 ontmoet het \omterm{def:b1:euclid:def}{inwendige
product} de theorie van de eigenwaarden (symmetrische matrices,
kwadratische vormen); in bachelorjaar 3 wordt de
oneindigdimensionale euclidische meetkunde de theorie van de
hilbertruimten.
\end{remark}

\begin{remark}[Vooruitzichten binnen boek 3]
\label{rem:b1:euclid:perspectives}
Twee bruggen vertrekken uit dit hoofdstuk. \emph{Achterwaarts},
naar de \omterm{def:b1:linmaps:def}{lineaire} algebra: de grammatrix van
\cref{exo:b1:euclid:11} verpakt \omterm{def:b1:euclid:def}{inwendige producten} in de
machinerie van de determinanten van \cref{ch:b1:det}, en de
\omterm{thm:b1:euclid:projection}{orthogonale projectie} is de bijzondere projector van
\cref{ch:b1:linmaps} waarvan de kern $F^\perp$ is --- al haar
algebra ($p^2 = p$, $s = 2p - \mathrm{id}$) geldt woordelijk, nu
met de bonus dat $\norm{x - p(x)}$ een \emph{afstand} is.
\emph{Voorwaarts}, naar de analyse: \cref{ch:b1:curves} meet de
booglengte met de norm van dit hoofdstuk en classificeert niets
zonder zijn \omterm{def:b1:euclid:isometry}{isometrieën}; \cref{ch:b1:multivar} leest de gradiënt
via Cauchy--Schwarz (de steilste klim) en sluit het volume af met
de kleinste kwadraten, die \cref{thm:b1:euclid:projection}
toegepast op een gegevensvector in $\R^n$ zijn. Het \omterm{def:b1:euclid:def}{inwendige
product} is het punt waar de algebra en de analyse van het boek
elkaar eindelijk ontmoeten.
\end{remark}

\section{Opgave: isometrieën van het vlak en de stelling van
Leonardo}

\begin{problem}\label{pb:b1:euclid:1}
Een \emph{\omterm{def:b1:euclid:isometry}{isometrie} van het vlak} is elke \omterm{def:b1:logic:map}{afbeelding} $f \colon
\R^2 \to \R^2$ die afstanden bewaart: $\norm{f(x) - f(y)} =
\norm{x - y}$ voor alle $x, y$ --- zonder enige aanname van
lineariteit. Deze opgave bewijst dat zulke \omterm{def:b1:logic:map}{afbeeldingen} precies de
translaties, rotaties, spiegelingen en \omterm{pb:b1:euclid:1}{glijspiegelingen} zijn (de
classificatie van de \omterm{def:b1:euclid:isometry}{isometrieën} van het vlak), berekent hun
samenstellingen, en bepaalt al hun \emph{eindige} \omterm{def:b1:structures:group}{groepen}: de
stelling van Leonardo, de wiskunde achter rozetpatronen. Vanaf deel
II vereenzelvigen wij $\R^2$ met $\C$ (\cref{ch:b1:complex}): het
canonieke \omterm{def:b1:euclid:def}{inwendige product} is $\langle z, w\rangle =
\operatorname{Re}(z\conj w)$ en de norm is de
\omterm{def:b1:complex:field}{modulus}.\index{isometrie}\index{glijspiegeling}
\index{stelling van Leonardo}\index{diëdergroep}

\medskip
\noindent\textbf{Deel I --- Elke \omterm{def:b1:euclid:isometry}{isometrie} is affien.}
\begin{enumerate}
  \item Ga na dat de translaties $t_a(x) = x + a$, de \omterm{def:b1:linmaps:def}{lineaire}
        \omterm{def:b1:euclid:isometry}{isometrieën} en al hun samenstellingen \omterm{def:b1:euclid:isometry}{isometrieën} zijn, en
        dat de \omterm{def:b1:euclid:isometry}{isometrieën} onder samenstelling een \omterm{def:b1:structures:group}{groep} vormen.
  \item Zij $f$ een \omterm{def:b1:euclid:isometry}{isometrie} met $f(0) = 0$. Toon aan dat $f$ de
        normen bewaart, en daarna --- met de polarisatie,
        \cref{thm:b1:euclid:cs} --- dat $\langle f(x), f(y)\rangle
        = \langle x, y\rangle$ voor alle $x, y$.
  \item Nog steeds met $f(0) = 0$: werk $\norm{f(x + y) - f(x) -
        f(y)}^2$ en $\norm{f(\lambda x) - \lambda f(x)}^2$ uit met
        vraag 2, en besluit dat $f$ \emph{\omterm{def:b1:linmaps:def}{lineair}} is: $f \in
        O(\R^2)$.
  \item Leid af dat elke \omterm{def:b1:euclid:isometry}{isometrie} $f$ zich \emph{eenduidig}
        schrijft als $f = t_a \circ g$ met $a = f(0)$ en $g$ een
        \omterm{def:b1:linmaps:def}{lineaire} \omterm{def:b1:euclid:isometry}{isometrie} (het \emph{\omterm{def:b1:linmaps:def}{lineaire} deel} van $f$).
  \item Noem $f$ \emph{direct} als $\det g = 1$, en \emph{indirect}
        als $\det g = -1$. Toon aan dat het \omterm{def:b1:linmaps:def}{lineaire} deel van een
        samenstelling de samenstelling van de \omterm{def:b1:linmaps:def}{lineaire} delen is, en
        formuleer de resulterende tekenregel (direct en indirect
        stellen samen als $+1$ en $-1$).
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De vier types.} Via
\cref{thm:b1:euclid:plane} zijn de \omterm{def:b1:linmaps:def}{lineaire} \omterm{def:b1:euclid:isometry}{isometrieën} van $\C$
de \omterm{def:b1:logic:map}{afbeeldingen} $z \mapsto az$ en $z \mapsto a\conj z$ met $\abs a
= 1$; dus is elke \omterm{def:b1:euclid:isometry}{isometrie} van het vlak
\[
f(z) = a z + b \quad (\text{direct}) \qquad\text{of}\qquad
f(z) = a\conj z + b \quad (\text{indirect}), \qquad \abs a = 1 .
\]
\begin{enumerate}[resume]
  \item Ga het woordenboek na: $R_\theta$ is $z \mapsto
        \eu^{\iu\theta}z$ en $S_\theta$ is $z \mapsto
        \eu^{\iu\theta}\conj z$ (toets beide op $1$ en $\iu$).
  \item (Directe geval) Zij $f(z) = az + b$ met $\abs a = 1$. Toon
        aan: is $a = 1$, dan is $f$ een translatie; is $a \neq 1$,
        dan heeft $f$ het eenduidige vaste punt $z_0 = b/(1 - a)$
        en geldt $f(z) - z_0 = a(z - z_0)$: een \emph{rotatie} met
        centrum $z_0$ en hoek $\arg a$.
  \item (Indirecte geval) Zij $f(z) = a\conj z + b$, en $v =
        a\conj b + b$. Toon aan dat $f \circ f = t_v$ en $f \circ
        t_v = t_v \circ f$. Is $v = 0$: toon aan dat het midden
        van $z$ en $f(z)$ een vast punt is, en dat $f$ een
        \emph{spiegeling} in een rechte is. Is $v \neq 0$: toon aan
        dat $r = t_{-v/2}\circ f$ een spiegeling is waarvan de as
        evenwijdig is met $v$, zodat $f = t_{v/2} \circ r$ een
        \emph{\omterm{pb:b1:euclid:1}{glijspiegeling}} is. Besluit: elke \omterm{def:b1:euclid:isometry}{isometrie} van het
        vlak is een translatie, een rotatie, een spiegeling of een
        \omterm{pb:b1:euclid:1}{glijspiegeling} (de classificatie van de \omterm{def:b1:euclid:isometry}{isometrieën} van het
        vlak).
  \item (Samenstellingen) Toon aan: de samenstelling van rotaties
        over de hoeken $\alpha$ en $\beta$ is een rotatie over de
        hoek $\alpha + \beta$ (een translatie als $\alpha + \beta
        \in 2\pi\Z$); de samenstelling van twee spiegelingen is een
        rotatie over tweemaal de hoek tussen de assen (een
        translatie als de assen evenwijdig zijn).
  \item Leid af dat elke \omterm{def:b1:euclid:isometry}{isometrie} van het vlak een samenstelling
        van hoogstens drie spiegelingen is.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Drie identificaties.}
\begin{enumerate}[resume]
  \item Classificeer $f(z) = \iu\conj z + 1 + \iu$ volledig: type,
        as, glijvector.
  \item Zij $f$ de rotatie over de hoek $\frac\pi2$ rond $0$ en $g$
        de rotatie over de hoek $\frac\pi2$ rond $1$. Bereken $g
        \circ f$ in de vorm $z \mapsto az + b$ en identificeer haar
        (type, centrum, hoek).
  \item Zij $r_1(z) = \conj z$ (de spiegeling in de reële as) en
        $r_2(z) = \iu\conj z$ (de spiegeling in de rechte $y = x$).
        Bereken $r_2 \circ r_1$ en toets vraag 9 aan dit voorbeeld.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Eindige \omterm{def:b1:structures:group}{groepen}: de stelling van
Leonardo.} Zij $G$ een \emph{eindige} \omterm{def:b1:structures:group}{groep} \omterm{def:b1:euclid:isometry}{isometrieën} van het
vlak.
\begin{enumerate}[resume]
  \item Toon aan dat \omterm{def:b1:euclid:isometry}{isometrieën} zwaartepunten bewaren: is
        $\lambda_1 + \dots + \lambda_m = 1$ en $f = t_a \circ g$
        ($g$ \omterm{def:b1:linmaps:def}{lineair}), dan is $f\bigl(\sum_i \lambda_i x_i\bigr) =
        \sum_i \lambda_i f(x_i)$.
  \item Stel $c = \frac{1}{\abs G}\sum_{g \in G} g(x_0)$ voor een
        willekeurig gekozen $x_0$. Toon aan dat elke $h \in G$ het
        punt $c$ vasthoudt: een eindige \omterm{def:b1:structures:group}{groep} \omterm{def:b1:euclid:isometry}{isometrieën} heeft een
        gemeenschappelijk vast punt.
  \item Leid af dat men, na conjugatie met $t_{-c}$, mag aannemen
        dat $G \subseteq O(\R^2)$. Zij $G^{+} = \{g \in G : \det g
        = 1\}$; toon aan dat ofwel $G = G^{+}$, ofwel $G^{+}$
        precies index $2$ heeft in $G$ (geef een bijectie $G^+ \to
        G \setminus G^+$).
  \item Toon aan dat een eindige \omterm{def:b1:structures:group}{groep} \emph{rotaties} rond $c$
        cyclisch is: kies onder haar elementen de rotatie
        $R_{\theta_0}$ met de kleinste hoek $\theta_0 \in
        \intoo{0}{2\pi}$, en bewijs met euclidische deling van
        hoeken dat zij voortbrengt; besluit dat $\theta_0 =
        \frac{2\pi}n$ en $G^{+} = \{R_{\theta_0}^{\,k}\} \cong
        C_n$.
  \item Stel $G \neq G^{+}$ en kies een spiegeling $s \in G$. Toon
        aan dat $G = G^{+} \cup sG^{+}$, dat $s r s = r^{-1}$ voor
        elke rotatie $r \in G^{+}$, en dat alle $n$ elementen van
        $sG^{+}$ spiegelingen zijn: $G$ is de \emph{\omterm{pb:b1:euclid:1}{diëdergroep}}
        $D_n$ van orde $2n$.
  \item Besluit (\emph{stelling van Leonardo}): elke eindige \omterm{def:b1:structures:group}{groep}
        \omterm{def:b1:euclid:isometry}{isometrieën} van het vlak is cyclisch, $C_n$, of diëdrisch,
        $D_n$.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Dividenden, en synthese.}
\begin{enumerate}[resume]
  \item Toon rechtstreeks aan dat een eindige \omterm{def:b1:structures:group}{groep} \omterm{def:b1:euclid:isometry}{isometrieën}
        geen enkele translatie en geen enkele \omterm{pb:b1:euclid:1}{glijspiegeling}
        behalve de identiteit kan bevatten (beschouw de machten van
        zo'n element).
  \item Zij $P_n$ de regelmatige $n$-hoek met als hoekpunten de
        $n$-de \omterm{def:b1:complex:unity}{eenheidswortels} ($n \geq 3$). Toon aan dat haar
        symmetriegroep precies $D_n$ is: de $n$ rotaties $z \mapsto
        \omega^k z$ en de $n$ spiegelingen $z \mapsto \omega^k
        \conj z$ met $\omega = \eu^{2\iu\pi/n}$, en geen andere.
  \item Geef vlakke figuren waarvan de symmetriegroepen
        respectievelijk $C_1$, $D_1$, $D_2$ en $C_3$ zijn.
  \item Som de acht elementen van de symmetriegroep van het
        vierkant met hoekpunten $\pm1, \pm\iu$ op als \omterm{def:b1:logic:map}{afbeeldingen}
        $z \mapsto \omega^k z$ of $z \mapsto \omega^k\conj z$, en
        geef de as van elk van de vier spiegelingen.
  \item Zij $f, g$ rotaties over dezelfde hoek $\theta \notin
        2\pi\Z$ rond \emph{verschillende} centra $c_1 \neq c_2$.
        Bereken $f\circ g - g\circ f$ puntsgewijs en toon aan dat
        $f\circ g \neq g\circ f$; toon bovendien aan dat $(f \circ
        g)\circ(g\circ f)^{-1}$ een niet-triviale translatie is,
        zodat elke \omterm{def:b1:structures:group}{groep} die $f$ en $g$ bevat oneindig is --- een
        tweede verklaring voor het enkele centrum in de stelling
        van Leonardo.
  \item Synthese, in vier zinnen: welke twee structurele
        resultaten willekeurige \omterm{def:b1:euclid:isometry}{isometrieën} tot \omterm{def:b1:linmaps:def}{lineaire} algebra
        herleiden (de vragen 3--4) en willekeurige eindige \omterm{def:b1:structures:group}{groepen}
        tot \omterm{def:b1:structures:subgroup}{deelgroepen} van $O(2)$ (vraag 15); wat de volledige
        lijst van \omterm{def:b1:euclid:isometry}{isometrieën} van het vlak is en welke invarianten
        (direct/indirect, vaste punten) de vier types scheiden;
        waarom de samenstellingsregels van vraag 9 de spiegelingen
        tot de voortbrengers van alles maken; en wat de stelling
        van Leonardo op eindige schaal toevoegt. Benoem de twee
        stellingen die in de delen II en IV zijn bewezen.
\end{enumerate}
\end{problem}
