\chapter{Telkunde}\label{ch:b1:counting}

\omterm{def:b1:counting:card}{Eindige verzamelingen} tellen klinkt elementair --- en wordt snel subtiel.
Dit hoofdstuk definieert de \omterm{def:b1:counting:card}{kardinaliteit} zoals het hoort (via bijecties,
in de geest van \cref{ch:b1:logic}), stelt de handvol telprincipes vast
waaruit al het overige volgt, en leidt daaruit de klassieke tellingen af:
lijsten, \omterm{def:b1:counting:objects}{permutaties}, deelverzamelingen, \omterm{def:b1:counting:objects}{binomiaalcoëfficiënten}.

\section{Kardinaliteit van eindige verzamelingen}

\begin{definition}[Eindige verzameling, kardinaliteit]\label{def:b1:counting:card}
Schrijf voor $n \in \N^*$ kortweg $\intint{1}{n} = \{1, 2, \dots, n\}$.
Een \omterm{def:b1:logic:sets}{verzameling} $E$ heet \emph{eindig}\index{eindige verzameling} wanneer
$E = \emptyset$ of wanneer er voor zekere $n \in \N^*$ een bijectie van
$\intint{1}{n}$ op $E$ bestaat; die $n$ is uniek
(\cref{thm:b1:counting:welldef}) en heet de
\emph{kardinaliteit}\index{kardinaliteit} van $E$, genoteerd $\abs{E}$
(met $\abs{\emptyset} = 0$).
\end{definition}

\begin{theorem}[De kardinaliteit is welgedefinieerd]\label{thm:b1:counting:welldef}
Is $m \neq n$, dan bestaat er geen bijectie van $\intint{1}{m}$ op
$\intint{1}{n}$. Preciezer: is $m > n$, dan bestaat er geen injectie van
$\intint{1}{m}$ in $\intint{1}{n}$.
\end{theorem}

\begin{proof}
We bewijzen met inductie naar $n$ de \omterm{def:b1:logic:statement}{uitspraak}: \emph{voor alle $m > n$
bestaat er geen injectie $\intint{1}{m} \to \intint{1}{n}$}. Voor $n = 0$
is het doel leeg en $m \geq 1$: er bestaat helemaal geen \omterm{def:b1:logic:map}{afbeelding}. Neem
de \omterm{def:b1:logic:statement}{uitspraak} aan voor $n$ en stel dat $f \colon \intint{1}{m} \to
\intint{1}{n+1}$ een injectie is met $m > n + 1$. Wordt de waarde $n + 1$
niet bereikt, dan is $f$ een injectie in $\intint{1}{n}$, in tegenspraak
met de inductiehypothese. Anders is $f(a) = n + 1$ voor precies één $a$;
verwissel $f(a)$ en $f(m)$ (formeel: stel samen met de transpositie van
die twee waarden), zodat de nieuwe injectie $g$ voldoet aan $g(m) = n +
1$. De beperking van $g$ tot $\intint{1}{m-1}$ is dan een injectie in
$\intint{1}{n}$ met $m - 1 > n$ --- opnieuw een tegenspraak.
\end{proof}

\begin{corollary}[Duivenhokprincipe]\label{cor:b1:counting:pigeonhole}
Is $\abs{E} > \abs{F}$, dan is geen enkele \omterm{def:b1:logic:map}{afbeelding} $f \colon E \to F$
\omterm{def:b1:logic:inj}{injectief}: twee elementen van $E$ delen hun
beeld\index{duivenhokprincipe}.
\end{corollary}

\begin{proof}
Schrijf $\abs E = m$, $\abs F = n$ met $m > n$, en kies bijecties $u
\colon \intint1m \to E$ en $v \colon F \to \intint1n$. Was $f$ \omterm{def:b1:logic:inj}{injectief},
dan was $v \circ f \circ u$ een injectie van $\intint1m$ in $\intint1n$
(een samenstelling van injecties, \cref{prop:b1:logic:comp}), in
tegenspraak met \cref{thm:b1:counting:welldef}.
\end{proof}

\begin{remark}[Tussenspel: waarom die verwisseling in het bewijs?]\label{rem:b1:counting:swapinterlude}
Het bewijs van \cref{thm:b1:counting:welldef} bevat de eerste
werkelijk vernuftige zet van dit hoofdstuk, die het waard is om
langzaam te herhalen. Het obstakel: om de inductiehypothese te
gebruiken wil je het laatste punt $m$ van het domein \emph{en} het
laatste punt $n+1$ van het doel weglaten, maar $f$ kan een ander
punt $a$ naar $n + 1$ sturen, en dan beschadigt het weglaten van
dat doelpunt de \omterm{def:b1:logic:map}{afbeelding} elders. De remedie: stel $f$ samen met
de transpositie van de twee \emph{waarden} $f(a)$ en $f(m)$ --- een
bijectie van het doel, zodat de injectiviteit behouden blijft ---
waarna de lastige waarde $n + 1$ op de onschadelijke plaats $m$
zit en beide weglatingen schoon verlopen. Dit patroon
``eerst normaliseren, dan snijden'' keert terug: zo herleidt de
recursie voor de derangementen $\sigma^{-1}(n+1)$ in de
weekendopgave van dit hoofdstuk, en zo worden \omterm{def:b1:counting:objects}{permutaties} overal
opgelapt in de opgave over de symmetrische groep in
\cref{ch:b1:structures}.
\end{remark}

\begin{proposition}[Injecties, surjecties en kardinaliteit]\label{prop:b1:counting:injsur}
Zij $E, F$ \omterm{def:b1:counting:card}{eindige verzamelingen} met $\abs{E} = \abs{F}$, en zij $f
\colon E \to F$. Dan geldt
\[
f \text{ injectief} \iff f \text{ surjectief} \iff f \text{ bijectief}.
\]
\end{proposition}

\begin{proof}
Stel $f$ \omterm{def:b1:logic:inj}{injectief}. Dan is $f$ een bijectie van $E$ op $f(E)$, dus
$\abs{f(E)} = \abs{E} = \abs{F}$. Miste $f(E)$ een punt $y_0$ van $F$,
dan was $f$ een injectie van $E$ in $F \setminus \{y_0\}$, een
\omterm{def:b1:logic:sets}{verzameling} met \omterm{def:b1:counting:card}{kardinaliteit} $\abs{F} - 1 < \abs{E}$ --- onmogelijk
wegens het duivenhokprincipe. Dus $f(E) = F$: $f$ is \omterm{def:b1:logic:inj}{surjectief} en
daarmee \omterm{def:b1:logic:inj}{bijectief}.

Stel $f$ \omterm{def:b1:logic:inj}{surjectief}. Kies bij elke $y \in F$ één \omterm{def:b1:logic:map}{origineel} $s(y) \in E$;
dan is $f \circ s = \mathrm{id}_F$, zodat $s$ \omterm{def:b1:logic:inj}{injectief} is
(\cref{prop:b1:logic:comp}). Volgens de vorige alinea, toegepast op $s$
(de \omterm{def:b1:counting:card}{kardinaliteiten} zijn gelijk), is $s$ \omterm{def:b1:logic:inj}{bijectief}. Uit $f \circ s =
\mathrm{id}_F$ volgt $f = \mathrm{id}_F \circ s^{-1} = s^{-1}$, dus is
$f$ \omterm{def:b1:logic:inj}{bijectief}. Ten slotte is een \omterm{def:b1:logic:inj}{bijectieve} \omterm{def:b1:logic:map}{afbeelding} per definitie
zowel \omterm{def:b1:logic:inj}{injectief} als \omterm{def:b1:logic:inj}{surjectief}, waarmee de kring van implicaties gesloten
is.
\end{proof}

\begin{example}[Eindigheid is essentieel]\label{ex:b1:counting:finiteonly}
Op een \emph{eindige} \omterm{def:b1:logic:sets}{verzameling} is \cref{prop:b1:counting:injsur} een
krachtige kortere weg: elke \omterm{def:b1:logic:inj}{injectieve} \omterm{def:b1:logic:map}{afbeelding} van $E$ naar zichzelf
is automatisch een \omterm{def:b1:counting:objects}{permutatie} van $E$ --- de helft van de bijectiviteit
krijg je gratis. Beide implicaties bezwijken op oneindige \omterm{def:b1:logic:sets}{verzamelingen}:
$n \mapsto n + 1$ is \omterm{def:b1:logic:inj}{injectief} van $\N$ naar $\N$ maar mist $0$, en de
\omterm{def:b1:logic:map}{afbeelding} $\N \to \N$ die $0 \mapsto 0$ en $n \mapsto n - 1$ voor $n
\geq 1$ stuurt, is \omterm{def:b1:logic:inj}{surjectief} maar niet \omterm{def:b1:logic:inj}{injectief}. Telkens als deze
propositie wordt ingeroepen, doet de eindigheidshypothese echt werk ---
een thema dat de weekendopgave van \cref{ch:b1:logic} van de andere kant
bekijkt, waar oneindige \omterm{def:b1:logic:sets}{verzamelingen} juist die \omterm{def:b1:logic:sets}{verzamelingen} zijn die
zulke \omterm{def:b1:logic:map}{afbeeldingen} op zichzelf toelaten.
\end{example}

\begin{example}[De helft van het werk, gratis]\label{ex:b1:counting:mod7}
Beschouw de \omterm{def:b1:logic:map}{afbeelding} $f$ op $\{0, 1, \dots, 6\}$ die $k$ naar de
rest van $3k$ bij deling door $7$ stuurt; haar waardentabel luidt
\[
0,\ 3,\ 6,\ 2,\ 5,\ 1,\ 4 .
\]
Is $f$ een bijectie? Alleen de injectiviteit volstaat
(\cref{prop:b1:counting:injsur}): hebben $3k$ en $3k'$ dezelfde
rest, dan deelt $7$ het getal $3(k - k')$, en omdat $7$ priem is en
$3$ niet deelt, deelt het $k - k'$ (lemma van Euclides, hier
gebruikt zoals bekend uit het bovenbouwvolume en bewezen in
\cref{ch:b1:arith}); met $\abs{k - k'} \leq 6$ dwingt dat $k = k'$
af. De surjectiviteit komt er gratis bij --- je hoeft $3k \equiv c$
niet voor elke $c$ op te lossen, al bevestigt de tabel dat elke
waarde precies één keer voorkomt. Deze kortere weg is een werkpaard:
ze bewijst de inverteerbaarheid van de vermenigvuldiging modulo $n$
(\cref{ch:b1:arith}), drijft het koppelen in de stelling van Wilson
aan, en keert in de lineaire algebra terug als ``een endomorfisme
van een eindigdimensionale ruimte is \omterm{def:b1:logic:inj}{injectief} precies wanneer het
\omterm{def:b1:logic:inj}{surjectief} is'' (\cref{ch:b1:findim}).
\end{example}

\section{De telprincipes}

\begin{proposition}[Som- en productregel]\label{prop:b1:counting:rules}
Zij $E, F$ \omterm{def:b1:counting:card}{eindige verzamelingen}.
\begin{enumerate}
  \item Is $E \cap F = \emptyset$, dan is $\abs{E \cup F} = \abs{E} +
        \abs{F}$; algemener geldt voor een \omterm{thm:b1:logic:partition}{partitie} van $E$ in stukken
        $E_1, \dots, E_k$ dat $\abs{E} = \sum_i \abs{E_i}$.
  \item In het algemeen is $\abs{E \cup F} = \abs{E} + \abs{F} - \abs{E
        \cap F}$.
  \item $\abs{E \times F} = \abs{E} \times \abs{F}$.
  \item De \omterm{def:b1:logic:sets}{verzameling} $F^E$ van alle \omterm{def:b1:logic:map}{afbeeldingen} van $E$ naar $F$
        voldoet aan $\abs{F^E} = \abs{F}^{\abs{E}}$.
  \item $\abs{\mathcal{P}(E)} = 2^{\abs{E}}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Plak de opsommingen aan elkaar: is $E = \{x_1, \dots, x_m\}$ en $F =
\{y_1, \dots, y_n\}$ zonder herhaling, dan somt $x_1, \dots, x_m, y_1,
\dots, y_n$ de \omterm{def:b1:logic:sets}{verzameling} $E \cup F$ zonder herhaling op (wegens de
disjunctheid). Inductie breidt dit uit tot $k$ stukken.

(2) $E \cup F$ is de disjuncte vereniging van $E$ en $F \setminus E$, en
$F$ is de disjuncte vereniging van $F \cap E$ en $F \setminus E$; dus is
$\abs{E \cup F} = \abs{E} + \abs{F \setminus E} = \abs{E} + \abs{F} -
\abs{E \cap F}$.

(3) $E \times F$ is de disjuncte vereniging, over $x \in E$, van de
\omterm{def:b1:logic:sets}{verzamelingen} $\{x\} \times F$, elk met \omterm{def:b1:counting:card}{kardinaliteit} $\abs{F}$; pas (1)
toe.

(4) Een \omterm{def:b1:logic:map}{afbeelding} van $E = \{x_1, \dots, x_m\}$ naar $F$ is precies de
keuze van het $m$-tal $(f(x_1), \dots, f(x_m)) \in F^m$; die
overeenkomst is een bijectie, en $\abs{F^m} = \abs{F}^m$ volgens (3) en
inductie.

(5) Deelverzamelingen van $E$ komen \omterm{def:b1:logic:inj}{bijectief} overeen met \omterm{def:b1:logic:map}{afbeeldingen} $E
\to \{0, 1\}$ (stuur $A$ naar haar indicatorfunctie); pas (4) toe.
\end{proof}

\begin{example}[Tellen via het complement]\label{ex:b1:counting:complement}
Hoeveel viercijferige pincodes (cijfers $0$--$9$, volgorde telt,
herhaling toegestaan) bevatten \emph{minstens één} herhaald cijfer?
Rechtstreeks tellen betekent jongleren met de gevallen ``precies
één paar, twee paren, een drietal, een viertal'' --- vijf
overlappende configuraties. Tel liever het complement: alle codes
samen zijn $10^4 = 10\,000$ (productregel), en de codes met vier
verschillende cijfers zijn $10 \times 9 \times 8 \times 7 =
5\,040$ ($4$-variaties), zodat het antwoord
\[
10^4 - 10 \cdot 9 \cdot 8 \cdot 7 = 10\,000 - 5\,040 = 4\,960
\]
luidt. Bijna de helft van alle pincodes herhaalt een cijfer. Het
inzicht: zodra een telling geformuleerd is met ``minstens'' of
``niet alle'', probeer je eerst het complement --- de somregel
garandeert $\abs{A} = \abs{E} - \abs{\overline A}$, en het
complement is vaak één enkele overzichtelijke configuratie.
\end{example}

\begin{example}[Roosterpaden]\label{ex:b1:counting:paths}
Tel de kortste paden van de hoek $(0,0)$ naar de hoek $(4, 3)$ van
een rooster, waarbij je telkens één stap naar rechts (R) of één
stap omhoog (U) zet. Elk zo'n pad zet precies $7$ stappen, waarvan
$4$ een R en $3$ een U; omgekeerd beschrijft elk woord van lengte
$7$ in de letters R en U met vier R's precies één pad. De paden
komen dus \omterm{def:b1:logic:inj}{bijectief} overeen met de keuzes van de posities van de
R's:
\[
\binom{7}{4} = 35 .
\]
Het inzicht zit in de \emph{codering}: de telling werd triviaal op
het moment dat elk pad in een woord vertaald werd, dat wil zeggen
in een \omterm{def:b1:logic:sets}{verzameling} posities --- opnieuw een illustratie van de
leuze dat een correcte telling een vermomde bijectie is
(\cref{met:b1:counting:which}).
\end{example}

\section{Lijsten, permutaties, deelverzamelingen}

\begin{definition}[Variaties, permutaties, combinaties]\label{def:b1:counting:objects}
Zij $E$ een \omterm{def:b1:logic:sets}{verzameling} met $\abs{E} = n$ en zij $0 \leq k \leq n$.
\begin{itemize}
  \item Een \emph{$k$-variatie} van $E$ is een \omterm{def:b1:logic:inj}{injectief} $k$-tal
        elementen van $E$ (een geordende keuze zonder herhaling);
  \item een \emph{permutatie}\index{permutatie} van $E$ is een bijectie
        van $E$ naar zichzelf --- equivalent: een $n$-variatie;
  \item een \emph{$k$-combinatie} is een deelverzameling van $E$ met $k$
        elementen (een ongeordende keuze zonder herhaling). Hun aantal
        wordt genoteerd $\binom{n}{k}$, uitgesproken als ``$n$ boven
        $k$''\index{binomiaalcoëfficiënt}.
\end{itemize}
\end{definition}

\begin{theorem}[De drie tellingen]\label{thm:b1:counting:counts}
Met $n = \abs{E}$ en $0 \leq k \leq n$:
\begin{enumerate}
  \item het aantal $k$-variaties van $E$ is $n (n-1) \cdots (n-k+1) =
        \dfrac{n!}{(n-k)!}$;
  \item het aantal \omterm{def:b1:counting:objects}{permutaties} van $E$ is $n!$;
  \item $\dbinom{n}{k} = \dfrac{n!}{k!\,(n-k)!}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Kies de eerste coördinaat ($n$ mogelijkheden), dan de tweede ($n -
1$ resterende), \dots, dan de $k$-de ($n - k + 1$ mogelijkheden).
Formeel gaat de inductie naar $k$. Voor $k = 1$ zijn er $n$ \omterm{def:b1:logic:inj}{injectieve}
tupels met één term. Neem de telling aan voor $k - 1$. Elke
$k$-variatie $(x_1, \dots, x_k)$ ontstaat uit precies één
$(k-1)$-variatie --- haar afknotting $(x_1, \dots, x_{k-1})$ --- door
er een laatste coördinaat buiten $\{x_1, \dots, x_{k-1}\}$ aan te
plakken, waarvoor precies $n - (k - 1)$ waarden beschikbaar zijn. Het
afknotten deelt de $k$-variaties dus op in klassen van gemeenschappelijke
grootte $n - k + 1$, geïndexeerd door de $(k-1)$-variaties, en de
somregel geeft
\[
\frac{n!}{(n-k+1)!}\;(n - k + 1) = \frac{n!}{(n-k)!} .
\]

(2) is (1) met $k = n$.

(3) Elke $k$-deelverzameling ordent tot $k!$ verschillende
$k$-variaties, en elke $k$-variatie komt van precies één
deelverzameling: dus $\frac{n!}{(n-k)!} = \binom nk \cdot k!$.
\end{proof}

\begin{example}[Ronde tafels: delen door een symmetrie]\label{ex:b1:counting:roundtable}
Op hoeveel manieren kunnen $n$ gasten rond een ronde tafel plaatsnemen,
waarbij twee opstellingen dezelfde zijn zodra elke gast dezelfde linker-
en rechterbuur heeft --- dat wil zeggen op rotatie na? Elke ronde
opstelling komt overeen met precies $n$ lineaire opstellingen (knip de
kring op elk van de $n$ plaatsen door), zodat de $n!$ lineaire
volgordes in groepjes van $n$ samenvallen:
\[
\frac{n!}{n} = (n-1)! \quad\text{ronde opstellingen.}
\]
Equivalent: zet één uitgekozen gast willekeurig neer (waarmee de
draaivrijheid verdwijnt) en orden de overige $n - 1$ gasten met de
klok mee. Voor $n = 6$: $120$ tafels. Beide oplossingen tonen de twee
standaardremedies tegen dubbeltellen: deel door het exacte aantal
herhalingen, of \emph{breek de symmetrie} door één object vast te
pinnen. Beide vergen dat de groep herhalingen voor elke configuratie
even groot is --- wat het bewijs van de formule $\binom nk =
\frac{n!}{k!\,(n-k)!}$ hierboven ook al gebruikte, met $k!$ in de rol
van $n$.
\end{example}

\begin{example}[Een voorwaarde erbij]\label{ex:b1:counting:constraint}
We blijven bij de ronde tafel: hoeveel van de $(n-1)!$ tafels met $n
\geq 3$ gasten zetten twee gegeven gasten $A$ en $B$ \emph{uit
elkaar} (niet naast elkaar)? Tel het complement. Tafels waar $A$ en
$B$ naast elkaar zitten: lijm ze tot één blok --- $n - 1$ objecten
rond de tafel, dus $(n-2)!$ ronde opstellingen --- en orden dan het
paar binnen zijn blok ($2$ manieren): $2\,(n-2)!$ tafels met de twee
naast elkaar. Bijgevolg houden
\[
(n-1)! - 2\,(n-2)! = (n-2)!\,\bigl((n - 1) - 2\bigr)
= (n-3)\,(n-2)!
\]
tafels hen uit elkaar. Controles: $n = 3$ geeft $0$ (rond een
driehoek raakt iedereen aan iedereen) en $n = 4$ geeft $2$, met de
hand na te gaan. De lijmtruc --- behandel een afgedwongen blok als
één object en tel daarna zijn interne opstellingen --- is de
standaardremedie voor buurvoorwaarden, lineair zowel als rond.
\end{example}

\begin{proposition}[Basisidentiteiten]\label{prop:b1:counting:identities}
Voor $0 \leq k \leq n$:
\[
\binom{n}{k} = \binom{n}{n-k},
\qquad
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}
\quad (1 \leq k \leq n-1),
\qquad
\sum_{k=0}^{n} \binom{n}{k} = 2^n .
\]
\end{proposition}

\begin{proof}
Eerste identiteit: $A \mapsto E \setminus A$ is een bijectie tussen de
$k$-deelverzamelingen en de $(n-k)$-deelverzamelingen. Regel van
Pascal\index{regel van Pascal}: leg een element $a \in E$ vast; de
$k$-deelverzamelingen splitsen in die welke $a$ bevatten (kies de $k -
1$ overige: $\binom{n-1}{k-1}$) en die welke $a$ mijden
($\binom{n-1}{k}$). Derde identiteit: beide leden tellen alle
deelverzamelingen van $E$, links opgesplitst naar grootte
(\cref{prop:b1:counting:rules}~(1) en (5)).
\end{proof}

\begin{theorem}[Binomium van Newton]\label{thm:b1:counting:binomial}
Voor alle $a, b$ in een commutatieve ring (bijvoorbeeld $\R$ of $\C$) en
$n \in \N$:
\[
(a + b)^n = \sum_{k=0}^{n} \binom{n}{k} a^k b^{\,n-k} .
\]
\end{theorem}

\begin{proof}
Distributief uitwerken van $(a+b)(a+b)\cdots(a+b)$ levert één term per
keuze, in elke factor, van $a$ of $b$: de term $a^k b^{n-k}$ verschijnt
eenmaal voor elke manier om te kiezen welke $k$ van de $n$ factoren een
$a$ bijdragen --- dat is $\binom nk$ maal. (Alternatief: inductie naar
$n$ met de regel van Pascal.)
\end{proof}

\begin{example}\label{ex:b1:counting:binomial}
Twee klassieke bijzondere gevallen: $a = b = 1$ geeft $\sum_k \binom nk
= 2^n$ terug; $a = -1$, $b = 1$ geeft $\sum_{k} (-1)^k \binom nk = 0$
voor $n \geq 1$: van de deelverzamelingen van een niet-lege \omterm{def:b1:logic:sets}{verzameling}
heeft precies de helft een even \omterm{def:b1:counting:card}{kardinaliteit}.
\end{example}

\begin{example}[Eén identiteit, twee bewijzen]\label{ex:b1:counting:threen}
Het bijzondere geval $a = 2$, $b = 1$ van het binomium luidt
\[
\sum_{k=0}^{n} \binom nk\,2^k = 3^n .
\]
Hier is dezelfde identiteit zonder ook maar één algebraïsche stap.
Het rechterlid telt de woorden van lengte $n$ over het alfabet $\{0,
1, 2\}$ (productregel). Deel elk woord in naar de \omterm{def:b1:logic:sets}{verzameling} $K$
van posities met een letter ongelijk aan nul: een $K$ met $\abs K =
k$ kiezen kost $\binom nk$, waarna elke positie van $K$ onafhankelijk
een $1$ of een $2$ draagt: $2^k$ manieren. De somregel over $k$ geeft
het linkerlid. Behalve het genoegen dat beide overeenstemmen, hebben
de twee bewijzen verschillende verdiensten: het algebraïsche
veralgemeent naar elke waarde van $a$, het combinatorische
\emph{verklaart} de formule en past zich aan voorwaarden aan
(verbied bijvoorbeeld de letter $2$ op de laatste positie) die geen
enkele substitutie vat. Beide technieken paraat houden is de
praktische vaardigheid die dit hoofdstuk traint.
\end{example}

\begin{method}[Welke telling past?]\label{met:b1:counting:which}
Beantwoord vóór het rekenen twee vragen over de keuze: doet de
\emph{volgorde} ertoe, en zijn \emph{herhalingen} toegestaan?
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{l|c|c}
 & volgorde telt & volgorde telt niet \\
\hline
zonder herhaling & $\dfrac{n!}{(n-k)!}$ & $\dbinom{n}{k}$ \\[6pt]
met herhaling & $n^k$ & (\cref{exo:b1:counting:10}) \\
\end{tabular}
\end{center}
Ga vervolgens op zoek naar een bijectie of een \omterm{thm:b1:logic:partition}{partitie} die het probleem
tot deze modeltellingen herleidt; een correcte telling is een vermomde
bijectie.
\end{method}

\begin{remark}[Veelgemaakte fouten bij het tellen]\label{rem:b1:counting:pitfalls}
\begin{enumerate}
  \item \emph{Niet-disjuncte gevallen optellen.} De somregel vergt een
        \omterm{thm:b1:logic:partition}{partitie}; kunnen configuraties aan twee gevallen tegelijk
        voldoen, dan worden ze dubbel geteld --- de remedie is
        inclusie-exclusie (\cref{thm:b1:counting:inclexcl}) of een
        fijnere gevalsindeling.
  \item \emph{Geordend tegenover ongeordend.} ``Een commissie van
        twee'' kiezen is $\binom n2$ en niet $n(n-1)$: beslis
        \emph{vóór het rekenen} of de keuze een volgorde draagt, en
        als een geordende telling makkelijker is, deel dan aan het
        eind door het aantal volgordes --- maar alleen wanneer elk
        ongeordend object uit \emph{even veel} geordende ontstaat.
  \item \emph{Meertrapskeuzes die niet onafhankelijk zijn.} De
        productregel vergt dat het aantal mogelijkheden in elke stap
        onafhankelijk is van de voorgaande keuzes. ``Kies een
        aanvoerder, dan een andere vice-aanvoerder'' is prima
        ($n(n-1)$); ``kies twee spelers die met elkaar overweg
        kunnen'' is helemaal geen product van twee stappen.
  \item \emph{Dubbeltellen door de constructie.} Elk object tweemaal
        bouwen --- bijvoorbeeld handen met \emph{minstens} één aas
        tellen als (kies een aas) $\times$ (kies $4$ andere kaarten)
        --- telt handen met twee azen dubbel. ``Minstens'' vraagt
        vrijwel altijd om het complement
        (\cref{ex:b1:counting:complement}).
\end{enumerate}
\end{remark}

\begin{example}[Een telling in pokerstijl]\label{ex:b1:counting:poker}
Uit een spel van $52$ kaarten is het aantal handen van $5$ kaarten
$\binom{52}{5} = 2\,598\,960$. Handen met precies één aas: kies de aas
($4$ manieren) en dan $4$ kaarten uit de $48$ niet-azen: $4
\binom{48}{4} = 778\,320$. De productregel is van toepassing omdat de
keuze in onafhankelijke stappen uiteenvalt.
\end{example}

\begin{method}[Dubbeltellen]\label{met:b1:counting:doublecount}
Om een identiteit tussen twee teluitdrukkingen te bewijzen, zoek je één
\omterm{def:b1:counting:card}{eindige verzameling} die beide leden tellen --- doorgaans een \omterm{def:b1:logic:sets}{verzameling}
\emph{paren} --- en bereken je haar \omterm{def:b1:counting:card}{kardinaliteit} in twee verschillende
volgordes. Het prototype is het \emph{handdruklemma}: tel op een feest de
paren (persoon, gegeven handdruk). Sommeren over personen geeft $\sum_p
d_p$ (het aantal handdrukken van persoon $p$); sommeren over handdrukken
geeft tweemaal het aantal handdrukken (elke handdruk betreft twee
personen). Bijgevolg is $\sum_p d_p$ even --- zodat het aantal mensen dat
een oneven aantal handen schudde altijd even is, een niet-triviale
conclusie zonder ook maar één formule. Dezelfde motor drijft
\cref{exo:b1:counting:12} en verscheidene vragen van de weekendopgave
hieronder aan.
\end{method}

\begin{example}[De gemiddelde deelverzameling]\label{ex:b1:counting:averagesubset}
Wat is de gemiddelde \omterm{def:b1:counting:card}{kardinaliteit} van een deelverzameling van een
\omterm{def:b1:logic:sets}{verzameling} $E$ met $n$ elementen, als alle $2^n$ deelverzamelingen
even waarschijnlijk zijn? Dubbeltel de paren $(A, a)$ met $a \in A$:
sommeren over deelverzamelingen geeft $\sum_A \abs A$, het gezochte
totaal; sommeren over elementen geeft $n \cdot 2^{n-1}$ (elk van de
$n$ elementen ligt in precies de helft van de deelverzamelingen ---
koppel elke $A$ die $a$ bevat aan $A \setminus \{a\}$). Bijgevolg is
\[
\frac{1}{2^n}\sum_{A \subseteq E} \abs A
= \frac{n\,2^{n-1}}{2^n} = \frac n2 :
\]
deelverzamelingen zijn gemiddeld halfvol --- zoals ook de symmetrie
$A \leftrightarrow \overline A$ voorspelt, die de groottes $k$ en
$n - k$ koppelt. Twee bewijzen, één antwoord, en beide vermijden de
rechtstreekse berekening $\sum_k k\binom nk$ van
\cref{exo:b1:counting:5}: een goedgekozen koppeling vervangt vaak
een identiteit.
\end{example}

\section{Inclusie-exclusie}

\begin{theorem}[Inclusie-exclusie]\label{thm:b1:counting:inclexcl}
Voor \omterm{def:b1:counting:card}{eindige verzamelingen} $A_1, \dots, A_p$ geldt
\[
\Bigl|\, \bigcup_{i=1}^{p} A_i \,\Bigr|
= \sum_{\emptyset \neq I \subseteq \intint{1}{p}}
(-1)^{\abs{I}+1} \Bigl|\, \bigcap_{i \in I} A_i \,\Bigr| .
\]
Voor $p = 3$:
$\abs{A \cup B \cup C} = \abs A + \abs B + \abs C - \abs{A \cap B} -
\abs{A \cap C} - \abs{B \cap C} + \abs{A \cap B \cap C}$.
\end{theorem}

\begin{proof}
Leg een element $x$ van de vereniging vast en tel zijn bijdrage aan het
rechterlid. Zij $J = \{i : x \in A_i\}$, met \omterm{def:b1:counting:card}{kardinaliteit} $m \geq 1$.
Het element $x$ wordt in $\abs{\bigcap_{i \in I} A_i}$ precies één keer
meegeteld wanneer $\emptyset \neq I \subseteq J$, met teken $(-1)^{\abs I
+ 1}$; zijn totale bijdrage is dus
\[
\sum_{k=1}^{m} \binom{m}{k} (-1)^{k+1}
= 1 - \sum_{k=0}^{m} \binom mk (-1)^k = 1 - 0 = 1
\]
volgens \cref{ex:b1:counting:binomial}. Elk element van de vereniging
wordt dus precies één keer geteld.
\end{proof}

\begin{example}[Relatief priem tellen]\label{ex:b1:counting:coprime}
Hoeveel getallen uit $\intint1{120}$ zijn relatief priem met $120 =
2^3 \times 3 \times 5$? Een geheel getal heeft precies dan een factor
gemeen met $120$ wanneer het deelbaar is door $2$, $3$ of $5$; tel
dus het complement van $A_2 \cup A_3 \cup A_5$, waarbij $A_d$ de
veelvouden van $d$ verzamelt. Binnen $\intint1{120}$ zijn er $120/d$
veelvouden van $d$ zodra $d$ het getal $120$ deelt --- geheel zonder
afrondingsfuncties --- en $A_2 \cap A_3 = A_6$, enzovoort.
Inclusie-exclusie geeft
\[
\abs{A_2 \cup A_3 \cup A_5}
= 60 + 40 + 24 - 20 - 12 - 8 + 4 = 88 ,
\]
zodat $120 - 88 = 32$ getallen relatief priem met $120$ zijn. Het is
leerzaam de berekening als een product te hergroeperen:
\[
120 - 88 = 120\Bigl(1 - \frac12\Bigr)\Bigl(1 -
\frac13\Bigr)\Bigl(1 - \frac15\Bigr) = 120 \cdot \frac12 \cdot
\frac23 \cdot \frac45 = 32 :
\]
de drie haakjes uitwerken levert precies de acht getekende termen van
de inclusie-exclusie op, één per deelverzameling van $\{2, 3, 5\}$.
Deze productvorm definieert de indicator van Euler, waarvan de
rekenkundige rol opduikt bij de congruenties van \cref{ch:b1:arith}
en verder wordt uitgewerkt in het volume van bachelorjaar 2.
\end{example}

\begin{example}[Derangementen]\label{ex:b1:counting:derangement}
Een \emph{derangement} is een \omterm{def:b1:counting:objects}{permutatie} zonder vaste punten. Zij $A_i$
de \omterm{def:b1:logic:sets}{verzameling} van de \omterm{def:b1:counting:objects}{permutaties} van $\intint{1}{n}$ die $i$ vasthouden;
dan is $\abs{\bigcap_{i \in I} A_i} = (n - \abs I)!$, en de
inclusie-exclusie telt de \omterm{def:b1:counting:objects}{permutaties} met minstens één vast punt; het
aantal derangementen is
\[
D_n = n! \sum_{k=0}^{n} \frac{(-1)^k}{k!} .
\]
Omdat $\sum (-1)^k / k! \to \eu^{-1}$ (zie \cref{ch:b1:series}), is
ongeveer $37\%$ van alle \omterm{def:b1:counting:objects}{permutaties} een derangement, wat $n$ ook is.
\end{example}

\begin{remark}[Waar dit hoofdstuk gebruikt wordt]\label{rem:b1:counting:whereused}
\omterm{def:b1:counting:objects}{Binomiaalcoëfficiënten} zijn de meest hergebruikte objecten van dit
hoofdstuk: ze dragen het binomium in \cref{ch:b1:poly} (ontwikkeling van
$(X + a)^n$), de formule van Leibniz voor de $n$-de afgeleide van een
product in \cref{ch:b1:derivative}, en de coëfficiënten van
Taylorontwikkelingen in \cref{ch:b1:taylor}. \omterm{def:b1:counting:objects}{Permutaties} keren terug als
groep --- met het teken, gebouwd op het tellen van inversies --- in
\cref{ch:b1:structures}, en dat teken definieert op zijn beurt de
determinanten in \cref{ch:b1:det}. Inclusie-exclusie en de telprincipes
vormen de eindige ruggengraat van de discrete kansrekening, die
uitgewerkt wordt in het volume van bachelorjaar 2; de
derangementengetallen van \cref{ex:b1:counting:derangement} worden
grondig bestudeerd in de weekendopgave hieronder.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:counting:1}
Een nummerplaat bestaat uit twee letters (A--Z), dan drie cijfers, dan
twee letters. Hoeveel platen zijn er mogelijk? En hoeveel zonder herhaalde
letter onder de vier?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:2}
Hoeveel anagrammen (herschikkingen van de letters, zinvol of niet) heeft
het woord \textsc{orange}? En \textsc{banana}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:3}
Uit $7$ vrouwen en $5$ mannen wordt een commissie van $4$ personen
gekozen. Hoeveel commissies zijn er in totaal? Hoeveel met precies $2$
vrouwen? Hoeveel met minstens één man?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:4}
Bewijs dat in elke groep van $13$ personen twee in dezelfde maand jarig
zijn, en dat onder $n + 1$ getallen gekozen uit $\intint{1}{2n}$ er twee
opeenvolgend zijn. \emph{(Beide keren het duivenhokprincipe: benoem de
hokken.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:5}
Bereken $\sum_{k=0}^{n} k \binom{n}{k}$. \emph{Aanwijzing: differentieer
$(1 + x)^n$, of gebruik $k \binom nk = n \binom{n-1}{k-1}$ (bewijs dat).}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:6}
Hoeveel strikt stijgende \omterm{def:b1:logic:map}{afbeeldingen} zijn er van $\intint{1}{k}$ naar
$\intint{1}{n}$? Leid daaruit het aantal stijgende (niet noodzakelijk
strikt stijgende) \omterm{def:b1:logic:map}{afbeeldingen} af. \emph{Aanwijzing voor de tweede
telling: $f$ stijgend $\mapsto$ $g(i) = f(i) + i - 1$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:7}
(Vandermonde) Bewijs, door de $k$-deelverzamelingen te tellen van een
\omterm{def:b1:logic:sets}{verzameling} die in twee blokken van grootte $m$ en $n$ gesplitst is:
\[
\binom{m+n}{k} = \sum_{j=0}^{k} \binom{m}{j} \binom{n}{k-j} .
\]
Leid daaruit $\sum_{j=0}^{n} \binom nj^2 = \binom{2n}{n}$ af.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:8}
Hoeveel getallen uit $\intint{1}{1000}$ zijn deelbaar door $2$ of $3$ of
$5$? (Inclusie-exclusie; $\lfloor 1000/6 \rfloor$ telt de veelvouden van
$6$, enzovoort.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:9}
Tel de surjecties van een \omterm{def:b1:logic:sets}{verzameling} met $4$ elementen op een
\omterm{def:b1:logic:sets}{verzameling} met $2$ elementen, en vervolgens op een \omterm{def:b1:logic:sets}{verzameling} met $3$
elementen. \emph{Aanwijzing: tel de \omterm{def:b1:logic:inj}{niet-surjectieve} \omterm{def:b1:logic:map}{afbeeldingen} met
inclusie-exclusie op de gemiste waarden.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:10}
(Sterren en staven) Bewijs dat het aantal $k$-keuzes uit $n$ objecten
\emph{met} herhaling en zonder volgorde --- equivalent: het aantal
$(x_1, \dots, x_n) \in \N^n$ met $x_1 + \dots + x_n = k$ --- gelijk is
aan $\binom{n + k - 1}{k}$. \emph{Aanwijzing: codeer een oplossing als
een rij van $k$ sterren en $n - 1$ staven.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:counting:11}
Bewijs de formule van \cref{ex:b1:counting:derangement} voor $D_n$ in
detail, en leid daaruit $n! = \sum_{k=0}^{n} \binom{n}{k} D_{n-k}$ af
(bewijs die identiteit ook rechtstreeks door de \omterm{def:b1:counting:objects}{permutaties} in te delen
naar hun \omterm{def:b1:logic:sets}{verzameling} vaste punten).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:counting:12}
Bewijs voor $n \in \N^*$, met een dubbeltelling van de paren
(deelverzameling, gemarkeerd element):
\[
\sum_{k=1}^{n} k \binom{n}{k} = n\, 2^{n-1},
\qquad\text{en vervolgens}\qquad
\sum_{k=1}^{n} k^2 \binom{n}{k} = n(n+1)\, 2^{n-2} .
\]
\emph{Voor de tweede: tel paren gemarkeerde elementen, gelijk of niet.}
\end{exercise}

\section{Opgave: derangementen, of de verkeerd geadresseerde brieven}

\begin{problem}\label{pb:b1:counting:1}
Een secretaresse stopt $n$ brieven willekeurig in $n$ geadresseerde
enveloppen: hoe groot is de kans dat \emph{niemand} de juiste brief
krijgt? Deze klassieke vraag (Montmort, 1708) voert naar de
derangementengetallen $D_n$ van \cref{ex:b1:counting:derangement}. De
formule van de inclusie-exclusie is slechts de openingszet: deze opgave
ontwikkelt de recursies die $D_n$ berekenen, twee verdere onafhankelijke
bewijzen van de formule, de opvallende stelling dat $D_n$ het gehele
getal is dat het dichtst bij $n!/\eu$ ligt, de volledige verdeling van
de vaste punten van een willekeurige \omterm{def:b1:counting:objects}{permutatie}, en de merkwaardige
rekenkunde van de rij $(D_n)$.\index{derangement}
Overal noteert $D_n$ het aantal derangementen (\omterm{def:b1:counting:objects}{permutaties} zonder vaste
punten) van $\intint1n$, met de afspraak $D_0 = 1$ (de lege \omterm{def:b1:counting:objects}{permutatie}
heeft geen vast punt).

\medskip
\noindent\textbf{Deel I --- Kleine gevallen en de telling van de vaste
punten.}
\begin{enumerate}
  \item Bereken $D_1, D_2, D_3$ rechtstreeks, en $D_4$ door de
        derangementen van $\{1, 2, 3, 4\}$ op te sommen, gegroepeerd
        naar de waarde van $\sigma(1)$. (Je vindt $D_4 = 9$.)
  \item Toon voor $0 \leq k \leq n$ aan dat het aantal $P_k(n)$ van de
        \omterm{def:b1:counting:objects}{permutaties} van $\intint1n$ met \emph{precies} $k$ vaste punten
        gelijk is aan $\binom nk D_{n-k}$.
  \item Ga de telling na voor $n = 4$: bereken $P_0(4), \dots, P_4(4)$
        en controleer dat hun som $4! = 24$ is. Wat is bij vier brieven
        waarschijnlijker: geen enkele match, of precies één?
  \item Toon met een dubbeltelling (\cref{met:b1:counting:doublecount})
        van de paren $(\sigma, i)$ met $\sigma(i) = i$ aan dat
        \[
        \sum_{\sigma} \abs{\mathrm{Fix}(\sigma)} = n! :
        \]
        gemiddeld heeft een willekeurige \omterm{def:b1:counting:objects}{permutatie} \emph{precies één}
        vast punt, wat $n \geq 1$ ook is.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Twee recursies en twee nieuwe bewijzen van
de formule.}
\begin{enumerate}[resume]
  \item Bewijs combinatorisch, voor $n \geq 1$:
        \[
        D_{n+1} = n\,(D_n + D_{n-1}) .
        \]
        (Deel de derangementen $\sigma$ van $\intint1{n+1}$ in naar $j =
        \sigma(n+1)$, en vervolgens naar de vraag of $\sigma(j) = n +
        1$; bouw in het geval $\sigma(j) \neq n+1$ een bijectie met de
        derangementen van $\intint1n$ door het \omterm{def:b1:logic:map}{origineel} van $n + 1$
        naar $j$ om te leiden.) Controleer de recursie numeriek tot en
        met $D_6$.
  \item Zet $u_n = D_n - n D_{n-1}$ en leid uit vraag 5 af dat $u_{n+1}
        = -u_n$; besluit met de tweede recursie:
        \[
        D_n = n D_{n-1} + (-1)^n \qquad (n \geq 1).
        \]
  \item Bewijs met inductie uit vraag 6 de formule van
        \cref{ex:b1:counting:derangement},
        \[
        D_n = n! \sum_{k=0}^{n} \frac{(-1)^k}{k!},
        \]
        --- een bewijs dat volledig losstaat van de inclusie-exclusie.
  \item (Binomiale inversie) Zij $(a_n)$ en $(b_n)$ twee rijen met $a_n
        = \sum_{k=0}^n \binom nk b_k$ voor alle $n$. Bewijs dat
        \[
        b_n = \sum_{k=0}^{n} (-1)^{n-k} \binom nk a_k
        \qquad (n \in \N).
        \]
        (Stel eerst de \emph{trinomiale herschikking} $\binom nk \binom
        kj = \binom nj \binom{n-j}{k-j}$ vast en gebruik dan de
        alternerende rijsom uit \cref{ex:b1:counting:binomial}.)
  \item Pas vraag 8 toe op de identiteit $n! = \sum_k \binom nk D_{n-k}$
        uit \cref{exo:b1:counting:11} om een \emph{derde} bewijs van de
        formule voor $D_n$ te krijgen.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Het gehele getal het dichtst bij
$n!/\eu$.} Neem voor dit deel aan --- de theorie wordt opgebouwd in
\cref{ch:b1:series} --- dat $\eu^{-1} = \lim_{n \to \infty} s_n$ met
$s_n = \sum_{k=0}^{n} \frac{(-1)^k}{k!}$, met de strikte afschatting
$\abs{\eu^{-1} - s_n} < \frac1{(n+1)!}$ voor alternerende reeksen, geldig
voor elke $n$.
\begin{enumerate}[resume]
  \item Toon aan dat $\bigl| D_n - n!/\eu \bigr| < \frac1{n+1}$ voor
        alle $n \in \N$.
  \item Leid de hoofdstelling af: \emph{voor elke $n \geq 1$ is $D_n$
        het gehele getal dat het dichtst bij $n!/\eu$ ligt}. Waarom
        heeft het argument $n \geq 1$ nodig?
  \item Bepaal het teken van de fout: toon aan dat $D_n > n!/\eu$
        precies wanneer $n$ even is. (Zoek de eerste verwaarloosde term
        van de alternerende reeks.)
  \item Bereken $D_7$ tot en met $D_{10}$ met de recursie van vraag 5
        en toets $D_{10}$ vervolgens aan $10!/\eu$ ($10! =
        3\,628\,800$, $\eu \approx 2.718281828$).
  \item (De garderobekans) Zij $p_n = D_n/n!$ de kans dat een uniform
        willekeurige \omterm{def:b1:counting:objects}{permutatie} een derangement is. Toon aan dat
        $\abs{p_n - \eu^{-1}} < \frac1{(n+1)!}$ en bereken $p_6$ tot op
        vijf decimalen. Verklaar: waarom is het antwoord op Montmorts
        vraag in wezen onafhankelijk van $n$ --- al bij een dozijn
        brieven?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De verdeling van de vaste punten.}
\begin{enumerate}[resume]
  \item Leg $k \in \N$ vast. Toon aan dat het aandeel van de \omterm{def:b1:counting:objects}{permutaties}
        van $\intint1n$ met precies $k$ vaste punten voldoet aan
        \[
        \frac{P_k(n)}{n!} = \frac{s_{n-k}}{k!}
        \;\xrightarrow[n \to \infty]{}\; \frac{\eu^{-1}}{k!} .
        \]
        (Deze limietwaarden, die samen $1$ opleveren, vormen de
        \emph{poissonverdeling} met parameter $1$, een centraal object
        van de cursus kansrekening in het volume van bachelorjaar 2.)
  \item Toon met een dubbeltelling van de drietallen $(\sigma, i, j)$
        waarin $i \neq j$ beide door $\sigma$ vastgehouden worden aan
        dat $\sum_\sigma \abs{\mathrm{Fix}(\sigma)}\,
        (\abs{\mathrm{Fix}(\sigma)} - 1) = n!$ voor $n \geq 2$. Samen
        met vraag 4: het gemiddelde van $\abs{\mathrm{Fix}}^2$ is $2$,
        zodat de ``spreiding'' (variantie) van het aantal vaste punten
        gelijk is aan $1$ --- opnieuw onafhankelijk van $n$, opnieuw in
        overeenstemming met de poissonverdeling.
  \item Bereken het aandeel van de \omterm{def:b1:counting:objects}{permutaties} met minstens één vast
        punt voor $n = 4, 5, 6$ (als breuk en tot op vier decimalen) en
        vergelijk met $1 - \eu^{-1} \approx 0.6321$.
  \item Toon rechtstreeks --- zonder limieten --- aan dat $s_{n+2} - s_n
        = (-1)^{n+1}\bigl(\frac1{(n+1)!} - \frac1{(n+2)!}\bigr)$, en
        leid daaruit af dat de kansen $p_n = s_n$ van vraag 14
        oscilleren: $p_0 > p_2 > p_4 > \dots$ en $p_1 < p_3 < p_5 <
        \dots$, waarbij de even (respectievelijk oneven) waarden dalen
        (respectievelijk stijgen) naar de gemeenschappelijke limiet
        $\eu^{-1}$.
  \item (Lootjes trekken) $n$ personen trekken elk één naam uit een
        hoed; trekt iemand zijn eigen naam, dan wordt de \emph{hele}
        trekking overgedaan. Schat, met het standaardfeit dat een
        gebeurtenis met kans $p$ gemiddeld $1/p$ pogingen kost, het
        gemiddelde aantal volledige trekkingen, en besluit dat de
        procedure gemiddeld ongeveer $\eu \approx 2.72$ trekkingen kost,
        in wezen onafhankelijk van $n$.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- De rekenkunde van $D_n$, en een synthese.}
\begin{enumerate}[resume]
  \item Verfijn vraag 5: toon aan dat er voor vaste $j \in \intint2n$
        precies $D_{n-1} + D_{n-2}$ derangementen van $\intint1n$ met
        $\sigma(1) = j$ zijn, onafhankelijk van $j$. Leid af dat $n - 1$
        het getal $D_n$ deelt voor elke $n \geq 2$.
  \item Bewijs dat $D_n$ oneven is dan en slechts dan als $n$ even is.
        (Werk modulo $2$ in de recursie van vraag 6.)
  \item Bewijs dat $D_n \equiv (-1)^n \pmod n$ voor $n \geq 1$, en
        controleer de congruentie op het laatste cijfer van $D_{10}$.
  \item Toon met vraag 6 aan dat $\dfrac{D_n}{D_{n-1}} = n +
        \dfrac{(-1)^n}{D_{n-1}}$ voor $n \geq 3$, zodat de verhouding
        van opeenvolgende derangementengetallen \emph{vrijwel exact} $n$
        is; leg in één zin uit waarom dat strookt met $D_n \approx
        n!/\eu$.
  \item Waar precies gebruikte deze opgave: (i) de product- en de
        somregel; (ii) het dubbeltellen; (iii) het binomium; (iv) de
        aangenomen afschatting voor alternerende reeksen? Eén zin per
        onderdeel.
  \item Synthese. De formule voor $D_n$ heeft nu drie bewijzen
        (inclusie-exclusie, recursie met inductie, binomiale inversie).
        Vergelijk in een korte alinea wat elk bewijs \emph{verklaart}:
        welk bewijs rekent het snelst, welk veralgemeent naar andere
        tellingen van vaste punten, en welk onthult waarom $\eu$
        opduikt in een probleem over enveloppen.
\end{enumerate}
\end{problem}
