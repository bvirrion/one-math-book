\chapter{Rationale breuken}\label{ch:b1:fractions}

Een \omterm{def:b1:fractions:field}{rationale breuk} is een quotiënt van \omterm{def:b1:poly:def}{veeltermen}. De centrale stelling
van dit korte hoofdstuk --- de
partieelbreuksplitsing\index{partieelbreuksplitsing} --- breekt elk zulk
quotiënt op in een som van elementaire bouwstenen $\frac{c}{(X - a)^k}$.
Naast haar algebraïsche belang is ze de standaardmachine om rationale
functies te integreren (\cref{ch:b1:integration}) en om bepaalde reeksen
te sommeren (\cref{ch:b1:series}).

\section{Het lichaam $K(X)$}

\begin{definition}\label{def:b1:fractions:field}
Een \emph{rationale breuk}\index{rationale breuk} over $K$ ($= \R$ of
$\C$) is een quotiënt $F = \frac{A}{B}$ met $A, B \in K[X]$ en $B \neq
0$; twee quotiënten $\frac AB$ en $\frac{A'}{B'}$ worden vereenzelvigd
wanneer $AB' = A'B$. Elke breuk heeft een \emph{onvereenvoudigbare vorm}
met $\gcd(A, B) = 1$, uniek op constanten na. Met de natuurlijke
bewerkingen is de \omterm{def:b1:logic:sets}{verzameling} $K(X)$ van de rationale breuken een
\omterm{def:b1:structures:field}{lichaam}.

De \emph{polen}\index{pool} van $F$ (in onvereenvoudigbare vorm) zijn de
wortels van $B$; de \emph{orde} van een pool is haar \omterm{def:b1:poly:derivative}{multipliciteit} als
wortel van $B$. De \emph{graad} van $F$ is $\deg F = \deg A - \deg B \in
\Z \cup \{-\infty\}$.
\end{definition}

\begin{example}[Polen, ordes en graad aflezen]\label{ex:b1:fractions:vocab}
Zij $F = \dfrac{X^3 - X}{X^4 - 2X^3 + X^2}$. Ontbind beide lagen: teller
$X(X-1)(X+1)$, noemer $X^2(X-1)^2$; schrap de gemeenschappelijke factor
$X(X-1)$:
\[
F = \frac{X + 1}{X(X - 1)} \quad\text{(onvereenvoudigbare vorm).}
\]
\omterm{def:b1:fractions:field}{Polen}: $0$ en $1$, beide \emph{enkelvoudig} --- de ordes lees je op de
onvereenvoudigde noemer af, zodat de schijnbare dubbele wortels van de
oorspronkelijke noemer er niet toe doen. Graad: $\deg F = 1 - 2 = -1$,
asymptotisch zichtbaar ($xF(x) \to 1$ als $x \to \infty$). De graad
gedraagt zich als de graad van een \omterm{def:b1:poly:def}{veelterm} ($\deg FG = \deg F + \deg G$,
$\deg(F + G) \leq \max$), een boekhoudregel die hieronder voortdurend bij
de jacht op coëfficiënten gebruikt wordt: elk argument met ``de limiet
van $xF(x)$'' is een vermomde graadtelling.
\end{example}

\begin{proposition}[Geheel deel]\label{prop:b1:fractions:integerpart}
Elke $F = \frac AB \in K(X)$ is op één manier $F = E + \frac RB$ met $E
\in K[X]$ (het \emph{gehele deel}, of veeltermdeel, van $F$) en $\deg R <
\deg B$. Er geldt $E \neq 0$ precies wanneer $\deg F \geq 0$.
\end{proposition}

\begin{proof}
Euclidische deling $A = BE + R$ (\cref{thm:b1:poly:division}), gedeeld
door $B$. Uniciteit: is $E + \frac RB = E' + \frac{R'}{B}$, dan is $(E -
E')B = R' - R$ met $\deg(R' - R) < \deg B$, wat $E = E'$ afdwingt en
daarna $R = R'$.
\end{proof}

\begin{example}[Eerst vereenvoudigen, dan delen]\label{ex:b1:fractions:integerpart}
Bepaal het gehele deel van $F = \dfrac{X^3 + 1}{X^2 - 1}$. Blind delen:
$X^3 + 1 = (X^2 - 1)X + (X + 1)$, dus $F = X + \frac{X + 1}{X^2 - 1}$.
Maar de breuk was niet onvereenvoudigbaar: $X^3 + 1 = (X + 1)(X^2 - X +
1)$ en $X^2 - 1 = (X+1)(X-1)$ delen de factor $X + 1$, en
\[
F = \frac{X^2 - X + 1}{X - 1} = X + \frac{1}{X - 1} :
\]
hetzelfde gehele deel $X$, maar het breukdeel stort in tot één enkele
bouwsteen, en de ``\omterm{def:b1:fractions:field}{pool}'' in $-1$ was nooit een \omterm{def:b1:fractions:field}{pool}. Vereenvoudig altijd
eerst tot de onvereenvoudigbare vorm vóór je op \omterm{def:b1:fractions:field}{polen} jaagt: de \omterm{def:b1:fractions:field}{polen} van
$F$ zijn de wortels van de \emph{onvereenvoudigde} noemer. (Het gehele
deel is ongevoelig voor die vereenvoudiging, zoals de uniciteit in
\cref{prop:b1:fractions:integerpart} waarborgt.)
\end{example}

\section{Partieelbreuksplitsing over $\C$}

\begin{theorem}[Splitsing over $\C$]\label{thm:b1:fractions:complex}
Zij $F = \frac AB \in \C(X)$ onvereenvoudigbaar, met $B = c\,(X -
a_1)^{m_1} \cdots (X - a_r)^{m_r}$. Dan is $F$, op precies één manier,
\[
F = E + \sum_{i=1}^{r} \sum_{k=1}^{m_i}
\frac{c_{i,k}}{(X - a_i)^k},
\qquad E \in \C[X],\ c_{i,k} \in \C ,
\]
waarbij $E$ het gehele deel van $F$ is.
\end{theorem}

\begin{proof}
Wegens \cref{prop:b1:fractions:integerpart} mogen we $\deg A < \deg B$
aannemen en de somsplitsing met $E = 0$ bewijzen.

\emph{De \omterm{def:b1:fractions:field}{polen} splitsen.} Schrijf $B = (X - a_1)^{m_1} B_1$ met $B_1(a_1)
\neq 0$. De \omterm{def:b1:poly:def}{veeltermen} $(X-a_1)^{m_1}$ en $B_1$ zijn \omterm{cor:b1:arith:bezout}{relatief priem} (geen
gemeenschappelijke wortel), dus levert Bézout in $\C[X]$ (zie de opmerking
in \cref{ch:b1:poly}) $U_0, V_0$ met $U_0 (X-a_1)^{m_1} + V_0 B_1 = 1$;
vermenigvuldigen met $A$, zetten van $U = AU_0$ en $V = AV_0$, en delen
door $B$ geeft
\[
\frac AB = \frac{V}{(X - a_1)^{m_1}} + \frac{U}{B_1} .
\]
De graadvoorwaarden kun je afdwingen: deel $V$ door $(X-a_1)^{m_1}$, zeg
$V = (X-a_1)^{m_1}Q + V_1$ met $\deg V_1 < m_1$, en neem het quotiënt op
in de tweede term ($U_1 = U + QB_1$):
\[
\frac AB = \frac{V_1}{(X - a_1)^{m_1}} + \frac{U_1}{B_1},
\qquad \deg V_1 < m_1 ;
\]
graden vergelijken ($\deg A < \deg B$ en $\deg V_1 < m_1$) dwingt ook
$\deg U_1 < \deg B_1$ af. Iteratie op $\frac{U_1}{B_1}$, \omterm{def:b1:fractions:field}{pool} na \omterm{def:b1:fractions:field}{pool},
herleidt alles tot het geval met één \omterm{def:b1:fractions:field}{pool} hieronder.

\emph{Eén \omterm{def:b1:fractions:field}{pool}.} Voor $\frac{A}{(X-a)^m}$ met $\deg A < m$: ontwikkel $A$
naar machten van $(X - a)$, $A = \sum_{j=0}^{m-1} \alpha_j (X - a)^j$ (de
Taylorontwikkeling van een \omterm{def:b1:poly:def}{veelterm}, als in het bewijs van
\cref{prop:b1:poly:multiplicity}); delen geeft precies de bouwstenen
$\frac{\alpha_j}{(X-a)^{m-j}}$. Concreet, voor $\frac{X^2 +
1}{(X+2)^3}$: substitutie van $X = Y - 2$ geeft
\[
X^2 + 1 = (Y - 2)^2 + 1 = Y^2 - 4Y + 5 ,
\qquad\text{dus}\qquad
\frac{X^2+1}{(X+2)^3} = \frac1{Y} - \frac4{Y^2} + \frac5{Y^3}
\]
met $Y = X + 2$: de drie bouwstenen verschijnen door de verschoven
ontwikkeling gewoon te delen --- de snelste weg zodra er één \omterm{def:b1:fractions:field}{pool} van hoge
orde in het spel is, en de aanbevolen weg voor
\cref{exo:b1:fractions:3}.

\emph{Uniciteit.} Stel dat twee splitsingen samenvallen; hun verschil is
een identiteit $0 = \sum_{i,k} \frac{d_{i,k}}{(X - a_i)^k}$.
Vermenigvuldig alles met $(X - a_1)^{m_1}$: elke term krijgt een factor
die in $a_1$ nul wordt, \emph{behalve} die met $i = 1$ en $k = m_1$,
waarvan de coëfficiënt $d_{1,m_1}$ wordt plus termen die minstens één
factor $(X - a_1)$ dragen. Evalueren in $a_1$ (geoorloofd: na de
vermenigvuldiging is er geen \omterm{def:b1:fractions:field}{pool} in $a_1$ meer) geeft $d_{1,m_1} = 0$.
Nu de hoogste coëfficiënt weg is, herhaal je met $(X - a_1)^{m_1 - 1}$,
en zo verder omlaag tot $k = 1$; daarna ga je naar de volgende \omterm{def:b1:fractions:field}{pool}. Alle
$d_{i,k}$ verdwijnen: de splitsing is uniek.
\end{proof}

\begin{method}[De coëfficiënten berekenen]\label{met:b1:fractions:compute}
In de praktijk vermijd je Bézout en combineer je:
\begin{enumerate}
  \item \emph{afdekken voor de hoogste macht}: de coëfficiënt van
        $\frac{1}{(X-a)^m}$ (met $m$ de orde van de \omterm{def:b1:fractions:field}{pool} $a$) is
        \[
        c_{a,m} = \Bigl[\,(X - a)^m F\,\Bigr]_{X = a};
        \]
        voor een \emph{enkelvoudige} \omterm{def:b1:fractions:field}{pool} van $F = \frac AB$ is dat
        $\frac{A(a)}{B'(a)}$;
  \item \emph{evaluaties} in handige punten en \emph{limieten} van
        $xF(x)$ voor $x \to \infty$, om lineaire betrekkingen voor de
        overige coëfficiënten te verzamelen;
  \item \emph{pariteits-} of toevoegingssymmetrieën, wanneer aanwezig, om
        het werk te halveren.
\end{enumerate}
\end{method}

\begin{remark}[Veelgemaakte fouten bij partieelbreuken]\label{rem:b1:fractions:pitfalls}
\begin{enumerate}
  \item \emph{Het gehele deel overslaan.} De splitsing in bouwstenen
        geldt voor breuken van graad $< 0$; is $\deg A \geq \deg B$,
        deel dan eerst (\cref{prop:b1:fractions:integerpart}), of de
        jacht op coëfficiënten levert tegenstrijdigheden op.
  \item \emph{Afdekken buiten haar bereik.} Vermenigvuldigen met $(X -
        a)^k$ en evalueren in $a$ levert de coëfficiënt alleen voor $k =
        m$, de \emph{volledige} orde van de \omterm{def:b1:fractions:field}{pool}; de coëfficiënten van
        lagere orde vergen andere betrekkingen (limieten, evaluaties)
        --- zie \cref{ex:b1:fractions:multiple}.
  \item \emph{Vergeten te vereenvoudigen.} \omterm{def:b1:fractions:field}{Polen} lees je af op de
        \emph{onvereenvoudigde} vorm; een gemeenschappelijke factor
        tussen teller en noemer schept spookpolen
        (\cref{ex:b1:fractions:integerpart}).
  \item \emph{Verkeerde bouwstenen over $\R$.} Boven een irreducibele
        kwadratische factor zijn de tellers \emph{affien} ($\alpha X +
        \beta$) en niet constant; alleen $\frac{c}{X^2 + 1}$ schrijven
        verliest oplossingen --- de juiste vormen worden door
        \cref{thm:b1:fractions:real} voorgeschreven en nooit
        geïmproviseerd.
\end{enumerate}
\end{remark}

\begin{proof}[Bewijs van de formule voor een enkelvoudige pool]
Bij een enkelvoudige \omterm{def:b1:fractions:field}{pool} $a$ is $B = (X - a) Q$ met $Q(a) \neq 0$, en
$B' = Q + (X - a) Q'$, dus $B'(a) = Q(a)$. De afdekwaarde is
$\frac{A(a)}{Q(a)} = \frac{A(a)}{B'(a)}$.
\end{proof}

\begin{example}\label{ex:b1:fractions:simple}
Splits $F = \dfrac{1}{X(X-1)(X-2)}$. Drie enkelvoudige \omterm{def:b1:fractions:field}{polen}; afdekken in
elk ervan:
\[
c_0 = \frac{1}{(0-1)(0-2)} = \frac12,
\quad
c_1 = \frac{1}{1 \times (1 - 2)} = -1,
\quad
c_2 = \frac{1}{2 \times 1} = \frac12,
\]
zodat $F = \dfrac{1/2}{X} - \dfrac{1}{X-1} + \dfrac{1/2}{X-2}$.
\emph{Controle in $X = 3$:} rechtstreeks is $F(3) = \frac{1}{3\cdot2\cdot1}
= \frac16$; uit de splitsing $\frac{1/2}{3} - \frac{1}{2} +
\frac{1/2}{1} = \frac16 - \frac12 + \frac12 = \frac16$.
\end{example}

\begin{example}[Meervoudige pool]\label{ex:b1:fractions:multiple}
Splits $F = \dfrac{X}{(X-1)^2 (X+1)}$. Vorm: $\frac{a}{(X-1)^2} +
\frac{b}{X-1} + \frac{c}{X+1}$.
\begin{itemize}
  \item Afdekken in de dubbele \omterm{def:b1:fractions:field}{pool}: $a = \bigl[\frac{X}{X+1}\bigr]_{X=1}
        = \frac12$.
  \item Afdekken in $-1$: $c = \bigl[\frac{X}{(X-1)^2}\bigr]_{X=-1} =
        -\frac14$.
  \item Limiet van $xF(x)$ in $\infty$: $0 = b + c$, dus $b = \frac14$.
\end{itemize}
\[
F = \frac{1/2}{(X-1)^2} + \frac{1/4}{X-1} - \frac{1/4}{X+1} .
\]
\emph{Controle in $X = 0$:} $F(0) = 0$ en $\frac12 - \frac14 - \frac14 =
0$.
\end{example}

\section{Splitsing over $\R$}

\begin{theorem}[Splitsing over $\R$]\label{thm:b1:fractions:real}
Zij $F \in \R(X)$ onvereenvoudigbaar met noemer
\[
B = c \prod_i (X - a_i)^{m_i} \prod_j (X^2 + p_j X + q_j)^{n_j}
\qquad (p_j^2 - 4q_j < 0).
\]
Dan splitst $F$ op precies één manier als haar gehele deel plus termen
\[
\frac{c_{i,k}}{(X - a_i)^k} \quad (1 \leq k \leq m_i),
\qquad
\frac{\alpha_{j,l}\, X + \beta_{j,l}}{(X^2 + p_j X + q_j)^{l}}
\quad (1 \leq l \leq n_j),
\]
met reële coëfficiënten.
\end{theorem}

\begin{proof}
Splits over $\C$ (\cref{thm:b1:fractions:complex}). Omdat $F$ reëel is,
is de coëfficiënt boven de \omterm{def:b1:fractions:field}{pool} $\conj a$ (op elke orde) de toegevoegde
van die boven $a$ (pas toevoeging toe op de splitsing en beroep je op de
uniciteit). Groepeer elk toegevoegd paar:
\[
\frac{c}{(X - z)^l} + \frac{\conj c}{(X - \conj z)^l}
= \frac{c\,(X - \conj z)^l + \conj c\,(X - z)^l}{\bigl(X^2 - 2\Re(z)X
+ \abs z^2\bigr)^{l}},
\]
waarvan de teller zijn eigen toegevoegde is en dus reëel, van graad $\leq
l$; veelvouden van de reële kwadratische factor afsplitsen verlaagt hem op
elk niveau $l$ tot graad $\leq 1$ (een korte inductie naar beneden). Reële
\omterm{def:b1:fractions:field}{polen} behouden hun reële coëfficiënten (toevoeging houdt ze vast). De
uniciteit volgt uit die over $\C$.
\end{proof}

\begin{example}[Het koppelen van toegevoegde polen bekeken]\label{ex:b1:fractions:pairing}
Het mechanisme van het bewijs, op het kleinste geval: over $\C$ zijn de
\omterm{def:b1:fractions:field}{polen} van $\frac1{X^2+1}$ gelijk aan $\pm\iu$, met afdekcoëfficiënten
$\frac1{2\iu}$ in $\iu$ en $\frac1{-2\iu}$ in $-\iu$ --- elkaars
toegevoegde, zoals de stelling voorspelt:
\[
\frac{1}{X^2 + 1}
= \frac{1/(2\iu)}{X - \iu} - \frac{1/(2\iu)}{X + \iu} .
\]
Herschrijven op de gemeenschappelijke noemer:
\[
\frac{1}{2\iu}\cdot\frac{(X + \iu) - (X - \iu)}{X^2 + 1}
= \frac{1}{2\iu}\cdot\frac{2\iu}{X^2 + 1} = \frac{1}{X^2+1} :
\]
de imaginaire delen heffen elkaar op en de reële bouwsteen verschijnt
ongeschonden. Bij reële integranden verlaat je $\R$ doorgaans nooit ---
maar wanneer je \emph{sommen} in complexe punten evalueert (zoals de
weekendopgave doet met \omterm{def:b1:complex:unity}{eenheidswortels}), zijn de complexe bouwstenen de
natuurlijke munt, en dit koppelen is de wisselkoers tussen beide
splitsingen.
\end{example}

\begin{example}\label{ex:b1:fractions:real}
Splits $F = \dfrac{4}{(X^2+1)(X-1)^2}$ over $\R$. Vorm: $\frac{aX +
b}{X^2 + 1} + \frac{c}{(X-1)^2} + \frac{d}{X - 1}$. Afdekken in de
dubbele \omterm{def:b1:fractions:field}{pool}: $c = \bigl[\frac{4}{X^2+1}\bigr]_{X=1} = 2$. Afdekken in de
complexe \omterm{def:b1:fractions:field}{pool} $\iu$ (de teller boven $X^2 + 1$ via de complexe splitsing
geëvalueerd, of rechtstreeks): vermenigvuldig met $X^2 + 1$ en stel $X =
\iu$:
\[
a\iu + b = \frac{4}{(\iu - 1)^2} = \frac{4}{-2\iu} = 2\iu ,
\]
dus $a = 2$ en $b = 0$. Limiet van $xF(x)$ in oneindig: $0 = a + d$, dus
$d = -2$. Bijgevolg is
\[
F = \frac{2X}{X^2+1} + \frac{2}{(X-1)^2} - \frac{2}{X-1} .
\]
\emph{Controle in $X = 0$:} $F(0) = 4$ en $0 + 2 + 2 = 4$.
\end{example}

\begin{example}[Een derdegraadsnoemer, van begin tot eind]\label{ex:b1:fractions:cubic}
Splits $F = \dfrac{1}{X^3 + 1}$ over $\R$. Ontbind eerst: $X^3 + 1 = (X +
1)(X^2 - X + 1)$, waarbij de kwadratische factor discriminant $-3 < 0$
heeft. Vorm: $\frac{a}{X+1} + \frac{bX + c}{X^2 - X + 1}$. Afdekken in de
enkelvoudige \omterm{def:b1:fractions:field}{pool} $-1$: $a = \bigl[\frac1{X^2 - X + 1}\bigr]_{X=-1} =
\frac13$. Limiet van $xF(x)$ in oneindig: $0 = a + b$, dus $b =
-\frac13$. Evaluatie in $X = 0$: $1 = a + c$, dus $c = \frac23$.
Bijgevolg is
\[
\frac{1}{X^3+1} = \frac13\Bigl(\frac{1}{X+1}
+ \frac{-X + 2}{X^2 - X + 1}\Bigr) ,
\]
bevestigd in $X = 1$: linkerlid $\frac12$, rechterlid
$\frac13\bigl(\frac12 + 1\bigr) = \frac12$. Let op de zuinigheid: drie
onbekenden, drie goedkope lineaire feiten (één afdekking, één limiet, één
evaluatie), zonder ook maar iets uit te werken --- de werkwijze van
\cref{met:b1:fractions:compute} in zuivere vorm.
\end{example}

\begin{remark}[Waar het voor dient]
Eenmaal gesplitst laat een rationale functie zich term voor term
integreren: de bouwstenen $\frac{1}{(x-a)^k}$ hebben elementaire
primitieve functies, en de bouwstenen $\frac{\alpha x + \beta}{(x^2 + px
+ q)^l}$ herleiden tot $\ln$ en $\arctan$
(\cref{ch:b1:integration}). Telescoperende sommen zijn de andere
standaardtoepassing (\cref{exo:b1:fractions:8}).
\end{remark}

\begin{remark}[Waar dit hoofdstuk gebruikt wordt]\label{rem:b1:fractions:whereused}
Partieelbreuken zijn vooral een \emph{voorbewerking}: het hoofdstuk over
integratie (\cref{ch:b1:integration}) haalt elke rationale integrand door
\cref{thm:b1:fractions:real} vóór het integreert, en het hoofdstuk over
reeksen (\cref{ch:b1:series}) telescopeert rationale termen precies zoals
in \cref{exo:b1:fractions:8} en in de weekendopgave hieronder --- die de
techniek helemaal tot $\sum 1/k^2 = \pi^2/6$ doorduwt. De logaritmische
afgeleide $P'/P = \sum m_i/(X - a_i)$ (\cref{exo:b1:poly:11}) duikt op
telkens als de ligging van wortels bestudeerd wordt. Buiten dit volume
ligt de splitsing van $1/\chi(X)$ voor een \omterm{def:b1:diffeq:linear2}{karakteristieke veelterm}
$\chi$ onder de berekening van matrixmachten en van laplacetransformaties
in het volume van bachelorjaar 2: de bouwstenen $\frac{c}{(X - a)^k}$
zijn de algebraïsche schaduw van de oplossingen $t^{k-1}\eu^{at}$ uit
\cref{ch:b1:diffeq}.
\end{remark}

\end{omfigure}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:fractions:1}
Splits over $\R$: $\dfrac{1}{X^2 - 1}$; $\;\dfrac{X}{X^2 - 3X + 2}$;
$\;\dfrac{X^2 + 1}{X(X-1)}$ \emph{(let op het gehele deel)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:fractions:2}
Splits over $\R$: $\dfrac{1}{X(X^2 + 1)}$ en $\dfrac{X^3}{X^2 + X + 1}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:fractions:3}
Splits $\dfrac{1}{X^2(X - 1)}$ en $\dfrac{X + 1}{(X - 1)^3}$
\emph{(substitueer voor de tweede $Y = X - 1$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:4}
Splits $\dfrac{1}{X^4 - 1}$ over $\C$ en vervolgens over $\R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:5}
Splits over $\R$: $\dfrac{X^2}{(X^2 + 1)^2}$, en leid daaruit een
primitieve functie van $x \mapsto \dfrac{x^2}{(x^2+1)^2}$ af, gegeven dat
$\int \frac{\dd x}{(x^2+1)^2} = \frac12\bigl(\arctan x +
\frac{x}{x^2+1}\bigr) + C$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:6}
Splits voor $n \in \N^*$ de breuk $F_n = \dfrac{n!}{X(X+1)\cdots(X+n)}$
\emph{(enkelvoudige \omterm{def:b1:fractions:field}{polen} in $0, -1, \dots, -n$; gebruik de afdekformule
en herken \omterm{def:b1:counting:objects}{binomiaalcoëfficiënten})}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:7}
Bewijs met de identiteit van \cref{exo:b1:poly:11} voor $P = X^n - 1$ dat
\[
\sum_{k=0}^{n-1} \frac{1}{X - \omega^k} = \frac{n X^{n-1}}{X^n - 1},
\qquad \omega = \eu^{2\iu\pi/n},
\]
en evalueer ter controle beide leden in $X = 2$ voor $n = 4$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:8}
Splits $\dfrac{1}{k(k+1)(k+2)}$ en bereken
\[
S_n = \sum_{k=1}^{n} \frac{1}{k(k+1)(k+2)},
\qquad\text{en vervolgens}\qquad
\lim_{n \to \infty} S_n .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fractions:9}
Zij $P \in \R[X]$ \omterm{def:b1:poly:def}{monisch} van graad $n$ met $n$ verschillende reële
wortels $x_1 < \dots < x_n$. Bewijs dat
\[
\sum_{i=1}^{n} \frac{1}{P'(x_i)} = 0 \quad (n \geq 2),
\qquad
\sum_{i=1}^{n} \frac{x_i^{\,n-1}}{P'(x_i)} = 1 .
\]
\emph{Aanwijzing: splits $\frac{X^m}{P}$ voor $m \leq n - 1$ en kijk naar
het afnemen van de coëfficiënten in oneindig --- of gebruik de
\omterm{thm:b1:poly:lagrange}{Lagrange-interpolatie} (\cref{thm:b1:poly:lagrange}) van het monoom $X^m$
in de knooppunten $x_i$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:10}
Splits $\dfrac{1}{X(X+1)^2}$ over $\R$. Neem de waarde $\sum_{k \geq 1}
\frac1{k^2} = \frac{\pi^2}6$ aan (bewezen in de weekendopgave van dit
hoofdstuk) en leid af dat
\[
\sum_{n=1}^{\infty} \frac{1}{n(n+1)^2} = 2 - \frac{\pi^2}{6} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:11}
Splits over $\R$: $\dfrac{1}{(X^2+1)(X^2+4)}$, en vervolgens
$\dfrac{X^2}{(X^2+1)(X^2+4)}$. \emph{Aanwijzing: beide noemers zijn
\omterm{def:b1:poly:def}{veeltermen} in $X^2$: splits eerst $\frac1{(Y+1)(Y+4)}$.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fractions:12}
(Seculaire vergelijkingen) Zij $F = \sum_{i=1}^{r} \frac{c_i}{X - p_i}$
met $p_1 < p_2 < \dots < p_r$ reëel en alle $c_i > 0$.
\begin{enumerate}
  \item Toon aan dat $F$ strikt dalend is op elk interval van haar
        domein, en geef haar limieten in $\pm\infty$ en aan beide zijden
        van elke \omterm{def:b1:fractions:field}{pool}.
  \item Leid af dat voor elke $\lambda > 0$ de vergelijking $F(x) =
        \lambda$ precies $r$ reële oplossingen heeft, één in elk
        interval $\intoo{p_i}{p_{i+1}}$ en één voorbij $p_r$. (Zulke
        vergelijkingen beheersen verstoringen van eigenwaarden; de
        tussenwaardestelling wordt hier gebruikt zoals bekend uit het
        bovenbouwvolume en bewezen in \cref{ch:b1:continuity}.)
\end{enumerate}
\end{exercise}

\section{Opgave: partieelbreuken als motor}

\begin{problem}\label{pb:b1:fractions:1}
De partieelbreuksplitsing oogt als boekhouding; deze opgave laat zien dat
ze een motor is. Voed haar met de breuk $\frac1{X(X+1)\cdots(X+k)}$ en ze
telescopeert hele families sommen in gesloten vorm; voed haar met
$\frac1{X^n - 1}$ en ze brengt goniometrische identiteiten voort als
\[
\sum_{k=1}^{n-1}\frac{1}{\sin^2\frac{k\pi}{n}} = \frac{n^2-1}{3} ;
\]
en één stap verder geperst wringt die identiteit een van de beroemdste
formules van de wiskunde naar buiten, die van Euler:
\[
\sum_{k=1}^{\infty}\frac1{k^2} = \frac{\pi^2}{6}
\]
--- hier verkregen met niets meer dan de algebra van dit hoofdstuk en de
goniometrie van de middelbare
school.\index{telescoperen}\index{Baselprobleem}
Overal is $\omega = \eu^{2\iu\pi/n}$; limieten van rijen worden gebruikt
zoals bekend uit het bovenbouwvolume (\cref{ch:b1:seq} formaliseert ze).

\medskip
\noindent\textbf{Deel I --- De telescoop.}
\begin{enumerate}
  \item Splits $\frac1{X(X+1)}$ en bereken $\sum_{n=1}^{N}
        \frac1{n(n+1)}$ exact; besluit dat de som naar $1$ gaat.
  \item Idem voor $\frac1{X(X+2)}$: toon aan dat $\sum_{n=1}^{N}
        \frac1{n(n+2)} = \frac12\bigl(\frac32 - \frac1{N+1} -
        \frac1{N+2}\bigr) \to \frac34$. (Met een gat blijven er aan elk
        uiteinde \emph{twee} randtermen over.)
  \item Formaliseer het mechanisme: is $F(X) = G(X) - G(X+1)$ voor een
        rationale $G$ zonder \omterm{def:b1:fractions:field}{polen} in $\intco1{+\infty}$, dan is
        $\sum_{n=1}^N F(n) = G(1) - G(N+1)$. Haal de waarde $\frac14$
        van \cref{exo:b1:fractions:8} terug door de getuige $G$ voor $F
        = \frac1{X(X+1)(X+2)}$ aan te wijzen.
  \item Bewijs de algemene faculteitstelescoop: voor $k \geq 1$ is
        \[
        \frac{1}{X(X+1)\cdots(X+k)}
        = \frac1k\biggl(\frac{1}{X(X+1)\cdots(X+k-1)}
        - \frac{1}{(X+1)\cdots(X+k)}\biggr),
        \]
        en leid af dat
        \[
        \sum_{n=1}^{\infty}\frac{1}{n(n+1)\cdots(n+k)}
        = \frac{1}{k \cdot k!} .
        \]
        Toets het geval $k = 2$ aan vraag 3.
  \item Evalueer de splitsing van \cref{exo:b1:fractions:6} in
        goedgekozen punten om te bewijzen dat
        \[
        \sum_{j=0}^{n}(-1)^j\binom nj\,\frac1{j+1}
        = \frac1{n+1},
        \qquad
        \sum_{j=0}^{n}(-1)^j\binom nj\,\frac1{j+2}
        = \frac1{(n+1)(n+2)} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De breuk $1/(X^n - 1)$.}
\begin{enumerate}[resume]
  \item Toon met de afdekformule aan dat
        \[
        \frac{1}{X^n - 1}
        = \frac1n\sum_{k=0}^{n-1}\frac{\omega^k}{X - \omega^k} .
        \]
  \item Twee controles: ga de formule rechtstreeks na voor $n = 2$, en
        toon aan dat de som van de $n$ coëfficiënten nul is voor $n \geq
        2$ --- leg uit waarom dat zo moet zijn (beschouw $xF(x)$ voor $x
        \to \infty$).
  \item Leid met afdekken opnieuw de identiteit van
        \cref{exo:b1:fractions:7} af: $\frac{nX^{n-1}}{X^n-1} = \sum_k
        \frac1{X - \omega^k}$.
  \item Groepeer toegevoegde \omterm{def:b1:fractions:field}{polen} om de reële splitsing te bewijzen:
        met $\theta_k = \frac{2k\pi}n$ is
        \[
        \frac{\omega^k}{X - \omega^k} +
        \frac{\omega^{n-k}}{X - \omega^{n-k}}
        = \frac{2\cos\theta_k\,X - 2}
        {X^2 - 2\cos\theta_k\,X + 1} ,
        \]
        en schrijf de volledige reële splitsing van $\frac1{X^n-1}$ op
        (onderscheid $n$ oneven en $n$ even).
  \item Werk het geval $n = 4$ uit en toets het aan
        \cref{exo:b1:fractions:4}.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Goniometrische sommen, en Eulers
$\pi^2/6$.} Zij $P = 1 + X + \dots + X^{n-1}$, met wortels $\omega,
\omega^2, \dots, \omega^{n-1}$ (alle enkelvoudig).
\begin{enumerate}[resume]
  \item Bewijs met $\frac{P'}P = \sum_{k=1}^{n-1}\frac1{X - \omega^k}$
        (\cref{exo:b1:poly:11}) dat
        \[
        \sum_{k=1}^{n-1}\frac{1}{1 - \omega^k} = \frac{n-1}2 .
        \]
  \item Bewijs $\dfrac1{1 - \eu^{\iu\theta}} = \dfrac12 +
        \dfrac\iu2\,\frac{\cos(\theta/2)}{\sin(\theta/2)}$ voor $\theta
        \notin 2\pi\Z$ (halvehoekontbinding,
        \cref{met:b1:complex:trig}), en leid uit vraag 11 af dat
        $\sum_{k=1}^{n-1} \frac{\cos(k\pi/n)}{\sin(k\pi/n)} = 0$ ---
        ook zichtbaar via de symmetrie $k \leftrightarrow n - k$.
  \item Bewijs, door de identiteit van vraag 11 te differentiëren (dus
        met $\bigl(\frac{P'}P\bigr)' = \frac{P''}P -
        \bigl(\frac{P'}P\bigr)^2$ geëvalueerd in $X = 1$), dat
        \[
        \sum_{k=1}^{n-1}\frac{1}{(1 - \omega^k)^2}
        = \frac{(n-1)(5-n)}{12} .
        \]
  \item Schrijf $\cot t = \frac{\cos t}{\sin t}$ en leid uit de vragen
        12--13 de twee gesloten vormen af:
        \[
        \sum_{k=1}^{n-1}\cot^2\frac{k\pi}n = \frac{(n-1)(n-2)}3,
        \qquad
        \sum_{k=1}^{n-1}\frac1{\sin^2\frac{k\pi}n}
        = \frac{n^2-1}3 .
        \]
  \item Ga beide formules met de hand na voor $n = 3$ en $n = 4$.
  \item Bewijs de ongelijkheden $\cot t < \frac1t < \frac1{\sin t}$ voor
        $t \in \intoo0{\frac\pi2}$ (uit $\sin t < t < \tan t$), en leid
        af dat voor $n = 2m + 1$ en $1 \leq k \leq m$ geldt
        \[
        \cot^2\frac{k\pi}n \;<\; \frac{n^2}{k^2\pi^2}
        \;<\; \frac1{\sin^2\frac{k\pi}n} .
        \]
  \item Sommeer die ongelijkheden voor $k = 1, \dots, m$ (met de
        symmetrie $k \leftrightarrow n - k$ om de formules van vraag 14
        te halveren) en knel in:
        \[
        \sum_{k=1}^{\infty}\frac1{k^2} = \frac{\pi^2}6 .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Hogere bouwstenen.}
\begin{enumerate}[resume]
  \item Bewijs, door de splitsing van $\frac1{X^2-1}$ te kwadrateren en
        de kruisterm opnieuw te splitsen, dat
        \[
        \frac1{(X^2-1)^2}
        = \frac14\Bigl(\frac1{(X-1)^2} + \frac1{(X+1)^2}\Bigr)
        - \frac14\Bigl(\frac1{X-1} - \frac1{X+1}\Bigr),
        \]
        en ga dat na in $X = 0$.
  \item Combineer vraag 18, de telescoop en Eulers waarde (vraag 17) om
        te bewijzen dat
        \[
        \sum_{n=2}^{\infty}\frac1{(n^2-1)^2}
        = \frac{\pi^2}{12} - \frac{11}{16} ,
        \]
        en bevestig de waarde numeriek tot op drie decimalen.
  \item Differentieer de identiteit van vraag 8 om een gesloten vorm
        voor $\sum_{k=0}^{n-1}\frac1{(X - \omega^k)^2}$ te krijgen, en
        ga die na in $X = 2$ met $n = 2$.
  \item Bewijs dat voor $k \geq 2$ geldt
        \[
        \sum_{n=k}^{\infty}\frac{1}{\binom nk} = \frac{k}{k-1} .
        \]
        \emph{(Herleid tot vraag 4 door $1/\binom nk$ met faculteiten te
        schrijven.)}
  \item Geef voor $k = 3$ de exacte partiële som
        $\sum_{n=3}^{N}\frac1{\binom n3}$ en haar limiet.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Synthese.}
\begin{enumerate}[resume]
  \item Schrijf als sluitstuk de volledige reële splitsing van
        $\dfrac1{X^6 - 1}$ uit en ga haar na in $X = 0$.
  \item Waar precies gebruikte de opgave: (i) de uniciteit van de
        splitsing; (ii) de \omterm{def:b1:complex:unity}{eenheidswortels} uit \cref{ch:b1:complex};
        (iii) de logaritmische afgeleide uit \cref{exo:b1:poly:11}? Eén
        zin per onderdeel.
  \item Synthese, in een korte alinea: één algebraïsche identiteit ---
        een breuk in bouwstenen breken --- bracht exacte sommen,
        goniometrische identiteiten en $\pi^2/6$ voort. Becommentarieer
        de taakverdeling tussen algebra (exacte splitsingen, overal
        geldig) en analyse (limieten, insluiten), en wijs aan waar elke
        draad geïndustrialiseerd wordt: telescoperen en vergelijken in
        \cref{ch:b1:series}, en het integreren van de bouwstenen in
        \cref{ch:b1:integration}.
\end{enumerate}
\end{problem}
