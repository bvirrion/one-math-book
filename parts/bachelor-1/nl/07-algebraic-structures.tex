\chapter{Algebraïsche structuren}\label{ch:b1:structures}

Steeds weer duiken dezelfde rekenregels op: bij de gehele getallen, de
reële getallen, de complexe getallen, de congruentieklassen, en straks bij
veeltermen (\cref{ch:b1:poly}), vectoren en matrices
(\cref{ch:b1:vspaces,ch:b1:matrices}). De algebra haalt de gemeenschappelijke
patronen eruit en geeft ze een naam: \emph{\omterm{def:b1:structures:group}{groep}}, \emph{\omterm{def:b1:structures:ring}{ring}},
\emph{\omterm{def:b1:structures:field}{lichaam}}. Een feit één keer bewijzen, op het niveau van de
structuur, bewijst het in één klap voor elk voorbeeld.

\section{Bewerkingen}

\begin{definition}\label{def:b1:structures:law}
Een \emph{bewerking}\index{bewerking} op een \omterm{def:b1:logic:sets}{verzameling} $E$ is een
\omterm{def:b1:logic:map}{afbeelding} $E \times E \to E$, genoteerd $(x, y) \mapsto x * y$. Ze heet
\emph{associatief} wanneer steeds $(x*y)*z = x*(y*z)$, en
\emph{commutatief} wanneer steeds $x * y = y * x$. Een element $e$ is een
\emph{neutraal element} wanneer $e * x = x * e = x$ voor alle $x$; dan is
$x'$ een \emph{inverse} van $x$ wanneer $x * x' = x' * x = e$.
\end{definition}

\begin{proposition}[Uniciteit]\label{prop:b1:structures:unique}
Een bewerking heeft hoogstens één neutraal element; bij een associatieve
bewerking met neutraal element heeft elk element hoogstens één inverse.
\end{proposition}

\begin{proof}
Zijn $e$ en $e'$ neutrale elementen, dan is $e = e * e' = e'$. Inverteren
$x'$ en $x''$ allebei $x$, dan is $x' = x' * e = x' * (x * x'') = (x' *
x) * x'' = e * x'' = x''$.
\end{proof}

\section{Groepen}

\begin{definition}[Groep]\label{def:b1:structures:group}
Een \emph{groep}\index{groep} $(G, *)$ is een \omterm{def:b1:logic:sets}{verzameling} met een
associatieve bewerking die een neutraal element bezit en waarin elk
element een inverse heeft. De groep heet
\emph{abels}\index{abelse groep} wanneer de bewerking commutatief is.
\end{definition}

\begin{example}\label{ex:b1:structures:groups}
$(\Z, +)$, $(\Q, +)$, $(\R, +)$, $(\C, +)$; $(\Q^*, \times)$, $(\R^*,
\times)$, $(\C^*, \times)$, $(\mathbb{U}_n, \times)$ (de \omterm{def:b1:complex:unity}{eenheidswortels},
\cref{def:b1:complex:unity}); de \omterm{def:b1:logic:sets}{verzameling} $\mathfrak{S}(E)$ van de
bijecties van een \omterm{def:b1:logic:sets}{verzameling} $E$ op zichzelf, onder samenstelling --- de
\emph{symmetrische groep}\index{symmetrische groep} van $E$, niet \omterm{def:b1:structures:group}{abels}
zodra $\abs E \geq 3$. Geen \omterm{def:b1:structures:group}{groepen}: $(\N, +)$ (geen inversen), $(\Z,
\times)$ (alleen $\pm 1$ is inverteerbaar).
\end{example}

\begin{proposition}[Rekenregels]\label{prop:b1:structures:rules}
In een \omterm{def:b1:structures:group}{groep} $G$ (multiplicatief genoteerd, met neutraal element $e$)
geldt:
\begin{enumerate}
  \item schrappen: $ax = ay \implies x = y$ en $xa = ya \implies x = y$;
  \item $(ab)^{-1} = b^{-1} a^{-1}$ en $(a^{-1})^{-1} = a$;
  \item voor $a, b \in G$ heeft elk van de vergelijkingen $ax = b$ en $xa
        = b$ precies één oplossing ($x = a^{-1}b$, respectievelijk $x = b
        a^{-1}$).
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Vermenigvuldig aan de juiste kant met $a^{-1}$ en gebruik de
associativiteit. (2) $(b^{-1}a^{-1})(ab) = b^{-1}(a^{-1}a)b = b^{-1}b =
e$, en symmetrisch; de uniciteit van de inverse maakt het af; het tweede
punt is \cref{prop:b1:structures:unique}, toegepast op $a^{-1}$. (3)
Substitueer, en gebruik (1) voor de uniciteit.
\end{proof}

\begin{example}[De symmetrieën van een rechthoek]\label{ex:b1:structures:klein}
Een rechthoek die geen vierkant is, laat precies vier isometrieën op
zichzelf toe: de identiteit $e$, de spiegeling $h$ in de horizontale as,
de spiegeling $v$ in de verticale as, en de halve draai $r$ om het
middelpunt. Samenstelling maakt van die \omterm{def:b1:logic:sets}{verzameling} met vier elementen een
\omterm{def:b1:structures:group}{groep}: elk element is zijn eigen inverse ($h^2 = v^2 = r^2 = e$), en het
product van twee verschillende elementen ongelijk aan $e$ is het derde
($hv = vh = r$: in beide assen spiegelen is de halve draai). De volledige
tabel is symmetrisch, dus de \omterm{def:b1:structures:group}{groep} is \omterm{def:b1:structures:group}{abels} --- en toch is het
\emph{niet} dezelfde \omterm{def:b1:structures:group}{groep} als de draaiingen $\mathbb U_4$ van
\cref{ex:b1:structures:order}: daar heeft $\iu$ \omterm{def:b1:structures:order}{orde} $4$, terwijl hier elk
element \omterm{def:b1:structures:order}{orde} $\leq 2$ heeft. Twee \omterm{def:b1:structures:group}{groepen} van dezelfde grootte kunnen dus
werkelijk verschillende vermenigvuldigingsstructuren hebben --- de figuur
hieronder zet beide tabellen naast elkaar. Deze \omterm{def:b1:structures:group}{groep} met vier elementen
keert terug als $\{\pm1\} \times \{\pm1\}$, en
\cref{exo:b1:structures:7} legt uit waarom elke \omterm{def:b1:structures:group}{groep} waarin alle
kwadraten triviaal zijn, net als deze, \omterm{def:b1:structures:group}{abels} moet zijn.
\end{example}

\begin{definition}[Deelgroep]\label{def:b1:structures:subgroup}
Een deelverzameling $H$ van een \omterm{def:b1:structures:group}{groep} $G$ heet een
\emph{deelgroep}\index{deelgroep} (genoteerd $H \leq G$) wanneer ze $e$
bevat en stabiel is onder de bewerking en onder inverteren. Dan is $H$
zelf een \omterm{def:b1:structures:group}{groep}.

\emph{Criterium:} een niet-lege $H \subseteq G$ is een deelgroep dan en
slechts dan als
\[
\forall x, y \in H, \quad x y^{-1} \in H .
\]
\end{definition}

\begin{proof}[Bewijs van het criterium]
Een \omterm{def:b1:structures:subgroup}{deelgroep} voldoet er duidelijk aan. Omgekeerd, zij $H \neq \emptyset$
die eraan voldoet, en kies $x_0 \in H$. Dan is $e = x_0 x_0^{-1} \in H$;
voor $y \in H$ is $y^{-1} = e\,y^{-1} \in H$; en voor $x, y \in H$ is $xy
= x (y^{-1})^{-1} \in H$.
\end{proof}

\begin{example}\label{ex:b1:structures:subgroups}
$\mathbb{U}_n \leq (\C^*, \times)$: niet-leeg, en voor $z, w \in
\mathbb{U}_n$ is $(zw^{-1})^n = z^n (w^n)^{-1} = 1$. De \omterm{def:b1:structures:subgroup}{deelgroepen} van
$(\Z, +)$ zijn precies de $n\Z$ (bewezen in \cref{thm:b1:arith:gcd}). Een
doorsnede van \omterm{def:b1:structures:subgroup}{deelgroepen} is altijd een \omterm{def:b1:structures:subgroup}{deelgroep}, maar een vereniging
vrijwel nooit (\cref{exo:b1:structures:6}).
\end{example}

\begin{remark}[Veelgemaakte fouten met structuren]\label{rem:b1:structures:pitfalls}
\begin{enumerate}
  \item \emph{Stabiliteit onder de bewerking volstaat niet.} $\N$ is
        binnen $\Z$ stabiel onder optellen en bevat $0$, en is toch geen
        \omterm{def:b1:structures:subgroup}{deelgroep}: de inversen ontbreken. Het criterium $xy^{-1} \in H$
        toetst alles in één keer --- maar pas nadat je $H \neq
        \emptyset$ hebt nagegaan.
  \item \emph{Niet-abelse reflexen.} In een algemene \omterm{def:b1:structures:group}{groep} is $(ab)^2 =
        abab$, wat alleen $a^2b^2$ is wanneer $a$ en $b$ commuteren;
        evenzo is $(ab)^{-1} = b^{-1}a^{-1}$, met omgekeerde volgorde.
        Elke identiteit die je van de schoolalgebra meebrengt moet je
        opnieuw uit de axioma's afleiden of als commutatief markeren.
  \item \emph{\omterm{def:b1:structures:morphism}{Kern} tegenover beeld.} $\ker f$ leeft in het
        \emph{domein}, $\operatorname{im} f$ in het doel; ``$f$ is
        \omterm{def:b1:logic:inj}{injectief} precies wanneer $\ker f$ triviaal is''
        (\cref{prop:b1:structures:kernel}) heeft geen tegenhanger met
        het beeld ($\operatorname{im} f = G'$ is de surjectiviteit).
  \item \emph{\omterm{def:b1:structures:ring}{Ringen} zijn geen \omterm{def:b1:structures:group}{groepen} voor $\times$.} In een \omterm{def:b1:structures:ring}{ring}
        hoeven de meeste elementen niet inverteerbaar te zijn, en
        schrappen met $a$ vergt dat $a$ een eenheid is of dat de \omterm{def:b1:structures:ring}{ring} een
        \omterm{def:b1:structures:field}{integriteitsdomein} is: in $\Z/12\Z$ is $\overline3\,\overline2 =
        \overline3\,\overline6$ terwijl $\overline2 \neq \overline6$
        (\cref{ex:b1:structures:zncomputation}).
\end{enumerate}
\end{remark}

\begin{definition}[Groepsmorfisme]\label{def:b1:structures:morphism}
Zij $(G, *)$ en $(G', \star)$ \omterm{def:b1:structures:group}{groepen}. Een \omterm{def:b1:logic:map}{afbeelding} $f \colon G \to G'$
heet een \emph{morfisme}\index{morfisme!van groepen} wanneer
\[
\forall x, y \in G, \qquad f(x * y) = f(x) \star f(y).
\]
Dan is $f(e_G) = e_{G'}$ en $f(x^{-1}) = f(x)^{-1}$. De
\emph{kern}\index{kern} en het \emph{beeld} van $f$ zijn
\[
\ker f = f^{-1}(\{e_{G'}\}) \leq G,
\qquad
\operatorname{im} f = f(G) \leq G' .
\]
Een \omterm{def:b1:logic:inj}{bijectief} morfisme heet een \emph{isomorfisme}; zijn inverse
\omterm{def:b1:logic:map}{afbeelding} is dan automatisch een morfisme.
\end{definition}

\begin{proof}[Bewijs van de beweringen]
$f(e) = f(e * e) = f(e)\star f(e)$, en $f(e)$ schrappen geeft $e_{G'} =
f(e)$. Vervolgens wijst $f(x)\star f(x^{-1}) = f(x x^{-1}) = e_{G'}$ het
element $f(x^{-1})$ als de inverse aan. \omterm{def:b1:structures:morphism}{Kern}: $e \in \ker f$; zijn $x, y
\in \ker f$, dan is $f(xy^{-1}) = f(x)f(y)^{-1} = e$; het criterium is
van toepassing. Beeld: hetzelfde criterium met $f(x)f(y)^{-1} =
f(xy^{-1})$. Inverse van een isomorfisme: schrijf voor $u, v \in G'$ de
elementen $u = f(x)$ en $v = f(y)$; dan is $f^{-1}(u \star v) =
f^{-1}(f(xy)) = xy = f^{-1}(u) f^{-1}(v)$.
\end{proof}

\begin{proposition}[Injectiviteit via de kern]\label{prop:b1:structures:kernel}
Een groepsmorfisme $f$ is \omterm{def:b1:logic:inj}{injectief} dan en slechts dan als $\ker f =
\{e\}$.
\end{proposition}

\begin{proof}
Is $f$ \omterm{def:b1:logic:inj}{injectief}, dan kan $\ker f$ alleen het ene \omterm{def:b1:logic:map}{origineel} van $e_{G'}$
bevatten, en dat is $e$. Omgekeerd, is $\ker f = \{e\}$ en $f(x) = f(y)$,
dan is $f(xy^{-1}) = f(x) f(y)^{-1} = e_{G'}$, dus $xy^{-1} = e$, oftewel
$x = y$.
\end{proof}

\begin{example}\label{ex:b1:structures:expmorphism}
$\exp \colon (\R, +) \to (\R_+^*, \times)$ is een \omterm{def:b1:structures:morphism}{morfisme} ($\eu^{x+y} =
\eu^x \eu^y$) en \omterm{def:b1:logic:inj}{bijectief} (\cref{prop:b1:functions:expln}): de additieve
en de multiplicatieve structuur zijn isomorf --- de historische
bestaansreden van de logaritmen. Nog een \omterm{def:b1:structures:morphism}{morfisme}: $\theta \mapsto
\eu^{\iu\theta}$ van $(\R, +)$ op de eenheidscirkel $(\mathbb{U},
\times)$, met \omterm{def:b1:structures:morphism}{kern} $2\pi\Z$.
\end{example}

\begin{example}[Het tekenmorfisme]\label{ex:b1:structures:signmorphism}
De \omterm{def:b1:logic:map}{afbeelding} $s \colon (\R^*, \times) \to (\{\pm1\}, \times)$ die $x$
naar zijn teken stuurt is een \omterm{def:b1:structures:morphism}{morfisme}: het teken van een product is het
product van de tekens. Haar \omterm{def:b1:structures:morphism}{kern} is $\intoo0{+\infty}$ (een \omterm{def:b1:structures:subgroup}{deelgroep},
zoals \cref{def:b1:structures:morphism} belooft), haar beeld heel
$\{\pm1\}$: \omterm{def:b1:logic:inj}{surjectief}, en massaal \omterm{def:b1:logic:inj}{niet-injectief}. Twee algemene lessen in
het klein. Ten eerste kan een \omterm{def:b1:structures:morphism}{morfisme} informatie vermorzelen: $s$
onthoudt van $x$ niets dan één bit, en dat is juist haar deugd ---
argumenten met tekens zijn precies de berekeningen die door $s$ heen
factoriseren. Ten tweede zijn \omterm{def:b1:structures:morphism}{morfismen} naar $\{\pm1\}$ de eenvoudigste
``invarianten'': de signatuur van \omterm{def:b1:counting:objects}{permutaties}, opgebouwd in de
weekendopgave van dit hoofdstuk, is hetzelfde verschijnsel op de \omterm{def:b1:structures:group}{groep}
$\mathfrak S_n$, en alle pariteitsargumenten die zij aandrijft dalen door
zo'n tweewaardig \omterm{def:b1:structures:morphism}{morfisme} af.
\end{example}

\begin{definition}[Machten, orde van een element]\label{def:b1:structures:order}
Zet in een \omterm{def:b1:structures:group}{groep} $G$ (multiplicatieve notatie) $x^0 = e$, $x^{k+1} = x^k
x$ en $x^{-k} = (x^k)^{-1}$ voor $k \in \N$; dan is $x^{k+l} = x^k x^l$
voor alle $k, l \in \Z$, zodat $k \mapsto x^k$ een \omterm{def:b1:structures:morphism}{morfisme} $(\Z, +) \to
G$ is waarvan het beeld $\langle x \rangle = \{x^k : k \in \Z\}$ een
\omterm{def:b1:structures:subgroup}{deelgroep} is, de \omterm{def:b1:structures:subgroup}{deelgroep} \emph{voortgebracht} door $x$. De
\emph{orde}\index{orde van een element} van $x$ is de kleinste $m \geq 1$
met $x^m = e$ als die bestaat (dan heeft $\langle x\rangle = \{e, x,
\dots, x^{m-1}\}$ precies $m$ elementen, en is $x^k = e \iff m \mid k$),
en anders $\infty$.
\end{definition}

\begin{example}\label{ex:b1:structures:order}
In $(\C^*, \times)$ heeft $\iu$ \omterm{def:b1:structures:order}{orde} $4$, met $\langle \iu \rangle = \{1,
\iu, -1, -\iu\} = \mathbb{U}_4$; algemener heeft $\omega =
\eu^{2\iu\pi/n}$ \omterm{def:b1:structures:order}{orde} $n$ met $\langle\omega\rangle = \mathbb{U}_n$. In
$(\Z, +)$ heeft elke $x \neq 0$ oneindige \omterm{def:b1:structures:order}{orde}. Waarom de beweringen in de
definitie kloppen: heeft $x$ \omterm{def:b1:structures:order}{orde} $m$, deel dan een willekeurige $k$ door
$m$ ($k = mq + r$ met $0 \leq r < m$, \cref{thm:b1:arith:division}): dan
is $x^k = (x^m)^q x^r = x^r$, zodat de machten met periode $m$ rondgaan;
de opgesomde elementen zijn twee aan twee verschillend wegens de
minimaliteit van $m$, en $x^k = e$ dwingt $r = 0$ af. De \omterm{def:b1:structures:order}{ordes} van
\omterm{def:b1:counting:objects}{permutaties} worden in de weekendopgave hieronder berekend.
\end{example}

\begin{example}[Ordes binnen $\mathbb U_{12}$]\label{ex:b1:structures:orderU12}
Wat is de \omterm{def:b1:structures:order}{orde} van $\omega^k$ in $\mathbb U_n$, met $\omega =
\eu^{2\iu\pi/n}$? Er geldt $(\omega^k)^m = 1$ precies wanneer $n \mid
km$, en met $d = \gcd(n, k)$, $n = dn'$, $k = dk'$ en $\gcd(n', k') = 1$
is $n \mid km \iff n' \mid k'm \iff n' \mid m$ (lemma van Gauss,
\cref{thm:b1:arith:gauss}). De kleinste zulke $m \geq 1$ is $n' =
\frac{n}{\gcd(n,k)}$. In $\mathbb U_{12}$ bijvoorbeeld heeft $\omega^8$
\omterm{def:b1:structures:order}{orde} $\frac{12}{\gcd(12,8)} = 3$ (inderdaad is $\omega^8 =
\eu^{4\iu\pi/3} \in \mathbb U_3$), terwijl $\omega^5$ \omterm{def:b1:structures:order}{orde} $12$ heeft:
die brengt de hele \omterm{def:b1:structures:group}{groep} voort, hoewel het niet de ``standaard''
voortbrenger is. De voortbrengers tellen --- de $k$ met $\gcd(k, n) = 1$
--- geeft de tellingen van \omterm{cor:b1:arith:bezout}{relatief prieme} getallen uit
\cref{ex:b1:counting:coprime} terug: groepentheorie en telkunde ontmoeten
elkaar.
\end{example}

\section{Ringen en lichamen}

\begin{definition}[Ring]\label{def:b1:structures:ring}
Een \emph{ring}\index{ring} $(A, +, \times)$ is een \omterm{def:b1:logic:sets}{verzameling} met twee
bewerkingen zó dat $(A, +)$ een \omterm{def:b1:structures:group}{abelse groep} is (met neutraal element
$0$), $\times$ associatief is met een neutraal element $1$, en $\times$
aan beide zijden distribueert over $+$. De ring heet \emph{commutatief}
wanneer $\times$ dat is. Een element $a$ heet \emph{inverteerbaar} (een
\emph{eenheid}) wanneer $ab = ba = 1$ voor een zekere $b$; de eenheden
vormen een \omterm{def:b1:structures:group}{groep} $(A^\times, \times)$.
\end{definition}

\begin{proof}[Bewijs dat de eenheden een groep vormen]
Stabiliteit: zijn $a, a'$ eenheden met inversen $b, b'$, dan is
\[
(aa')(b'b) = a(a'b')b = a\,1\,b = ab = 1,
\qquad (b'b)(aa') = 1
\]
symmetrisch, dus is $aa'$ een eenheid. Het element $1$ is een eenheid
(zijn eigen inverse), de associativiteit wordt van $A$ overgeërfd, en de
inverse $b$ van een eenheid $a$ is zelf een eenheid (met inverse $a$).
Dus voldoet $(A^\times, \times)$ aan alle groepsaxioma's. Elke \omterm{def:b1:structures:group}{groep} in
dit boek die niet uit \omterm{def:b1:counting:objects}{permutaties} is opgebouwd, ontstaat op deze manier:
$\Q^* = \Q^\times$, $\R^*$, $\C^*$, de eenheden van $\Z/n\Z$ hieronder,
en later de inverteerbare matrices (\cref{ch:b1:matrices}).
\end{proof}

\begin{example}\label{ex:b1:structures:rings}
$\Z, \Q, \R, \C$ zijn commutatieve \omterm{def:b1:structures:ring}{ringen}; $\Z^\times = \{1, -1\}$ en
$\Q^\times = \Q^*$. Later: veeltermringen $K[X]$ (\cref{ch:b1:poly}),
matrixringen (niet-commutatief, \cref{ch:b1:matrices}), en $\Z/n\Z$
hieronder. In elke \omterm{def:b1:structures:ring}{ring} is $0 \times a = 0$ (uit de distributiviteit: $0a
= (0+0)a = 0a + 0a$) en $(-1)a = -a$.
\end{example}

\begin{example}[Idempotenten: nieuwe verschijnselen in nieuwe ringen]\label{ex:b1:structures:idempotents}
In $\Z$ heeft de vergelijking $x^2 = x$, oftewel $x(x - 1) = 0$, alleen
de oplossingen $0$ en $1$. In $\Z/6\Z$ geeft het aflopen van alle
klassen: $\overline0^2 = \overline0$, $\overline1^2 = \overline1$,
$\overline3^2 = \overline9 = \overline3$ en $\overline4^2 = \overline{16}
= \overline4$ --- \emph{vier} idempotenten. De twee exotische komen van
nuldelers: $\overline3\,(\overline3 - \overline1) = \overline3 \times
\overline2 = \overline6 = \overline0$, terwijl geen van beide factoren nul
is. Zulke berekeningen ijken je instinct: vertrouwde feiten over
vergelijkingen overleven in \omterm{def:b1:structures:field}{integriteitsdomeinen} en \omterm{def:b1:structures:field}{lichamen}, maar een
algemene \omterm{def:b1:structures:ring}{ring} kan zich anders gedragen, en doet dat ook --- zie ook de
booleaanse \omterm{def:b1:structures:ring}{ringen} van \cref{exo:b1:structures:10}, waar \emph{elk} element
idempotent is.
\end{example}

\begin{proposition}[Binomium in een commutatieve ring]\label{prop:b1:structures:binomial}
Zijn $a, b$ elementen van een commutatieve \omterm{def:b1:structures:ring}{ring} (algemener: geldt $ab =
ba$), dan is voor $n \in \N$
\[
(a+b)^n = \sum_{k=0}^n \binom nk a^k b^{n-k},
\qquad
a^n - b^n = (a - b) \sum_{k=0}^{n-1} a^k b^{\,n-1-k} .
\]
\end{proposition}

\begin{proof}
De bewijzen van \cref{thm:b1:counting:binomial} en van de meetkundige
identiteit gebruiken alleen de associativiteit, de commutativiteit van de
twee elementen en de distributiviteit --- ze gelden letterlijk.
\end{proof}

\begin{example}[Het binomium in een ongewone ring]\label{ex:b1:structures:binomialring}
Twee snelle opbrengsten van de algemeenheid. In $\Z/p\Z$ ($p$ \omterm{def:b1:arith:prime}{priem})
verdwijnen de middelste \omterm{def:b1:counting:objects}{binomiaalcoëfficiënten} (de eerste stap van
\cref{thm:b1:arith:fermat}), zodat de stelling instort tot de
\emph{droom van de eerstejaars}
\[
(a + b)^p = a^p + b^p \qquad \text{in } \Z/p\Z ,
\]
daar een echte identiteit, hoe misdadig ze over $\R$ ook oogt. En in elke
commutatieve \omterm{def:b1:structures:ring}{ring} met een element $\varepsilon$ waarvoor $\varepsilon^2 =
0$, knot de stelling af: $(a + \varepsilon)^n = a^n +
n\,a^{n-1}\varepsilon$, want alle hogere termen dragen een factor
$\varepsilon^2 = 0$. De coëfficiënt $n\,a^{n-1}$ van $\varepsilon$ is de
afgeleide van $x^n$ --- geen toeval, en een eerste aanwijzing dat
afgeleiden evenzeer algebra als analyse zijn (vergelijk de formele
afgeleide van \cref{ch:b1:poly}).
\end{example}

\begin{definition}[Integriteitsdomein, lichaam]\label{def:b1:structures:field}
Een commutatieve \omterm{def:b1:structures:ring}{ring} $A \neq \{0\}$ heet een
\emph{integriteitsdomein}\index{integriteitsdomein} wanneer ze geen
nuldelers heeft: $ab = 0 \implies a = 0$ of $b = 0$. Ze heet een
\emph{lichaam}\index{lichaam} wanneer elk element ongelijk aan nul
inverteerbaar is. Elk lichaam is een integriteitsdomein (uit $ab = 0$ en
$a \neq 0$ volgt $b = a^{-1}ab = 0$).
\end{definition}

\begin{example}
$\Q$, $\R$ en $\C$ zijn \omterm{def:b1:structures:field}{lichamen}; $\Z$ is een \omterm{def:b1:structures:field}{integriteitsdomein} maar
geen \omterm{def:b1:structures:field}{lichaam}. In een \omterm{def:b1:structures:field}{integriteitsdomein} mag je schrappen bij $\times$: uit
$ab = ac$ en $a \neq 0$ volgt $b = c$.
\end{example}

\section{De ring $\Z/n\Z$}

\begin{definition}\label{def:b1:structures:zn}
Leg $n \in \N^*$ vast. De congruentieklassen modulo $n$
(\cref{ex:b1:logic:congruence}) vormen een \omterm{def:b1:logic:sets}{verzameling}
$\Z/n\Z$\index{Z/nZ@$\Z/n\Z$} met $n$ elementen, genoteerd $\overline 0,
\overline 1, \dots, \overline{n-1}$. De bewerkingen
\[
\overline a + \overline b = \overline{a + b},
\qquad
\overline a \times \overline b = \overline{ab}
\]
zijn welgedefinieerd --- de klassen van de resultaten hangen niet van de
vertegenwoordigers af, juist omdat de \omterm{def:b1:arith:congruence}{congruentie} verenigbaar is met $+$
en $\times$ (\cref{def:b1:arith:congruence}) --- en maken van $\Z/n\Z$
een commutatieve \omterm{def:b1:structures:ring}{ring}.
\end{definition}

\begin{theorem}[Eenheden van $\Z/n\Z$; de lichamen $\Z/p\Z$]\label{thm:b1:structures:zp}
\begin{enumerate}
  \item $\overline a$ is inverteerbaar in $\Z/n\Z$ dan en slechts dan
        als $\gcd(a, n) = 1$.
  \item $\Z/n\Z$ is een \omterm{def:b1:structures:field}{lichaam} dan en slechts dan als $n$ \omterm{def:b1:arith:prime}{priem} is.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) is \cref{prop:b1:arith:invmod}, herschreven met klassen.

(2) Is $n = p$ \omterm{def:b1:arith:prime}{priem}, dan geldt voor elke $\overline a \neq \overline 0$
dat $p \nmid a$, dus $\gcd(a, p) = 1$: inverteerbaar volgens (1) --- een
\omterm{def:b1:structures:field}{lichaam}. Is $n = ab$ met $1 < a, b < n$, dan is $\overline a\, \overline b
= \overline n = \overline 0$ met $\overline a, \overline b \neq \overline
0$: nuldelers, dus zelfs geen \omterm{def:b1:structures:field}{integriteitsdomein}; en $n = 1$ geeft de
nulring, die uitgesloten is.
\end{proof}

\begin{example}[Hoeveel vierkantswortels van $1$?]\label{ex:b1:structures:sqrtone}
Los $x^2 = \overline 1$ op in $\Z/8\Z$ en in $\Z/7\Z$. De acht klassen
modulo $8$ aflopen: $1^2 = 1$, $3^2 = 9 \equiv 1$, $5^2 = 25 \equiv 1$,
$7^2 = 49 \equiv 1$ --- \emph{vier} oplossingen $\{\overline1,
\overline3, \overline5, \overline7\}$, hoewel de veelterm $X^2 - 1$ graad
$2$ heeft. In het \omterm{def:b1:structures:field}{lichaam} $\Z/7\Z$ daarentegen betekent $x^2 =
\overline1$ dat $(x - \overline1)(x + \overline1) = \overline0$, en een
\omterm{def:b1:structures:field}{lichaam} heeft geen nuldelers: $x = \pm\overline1$, slechts twee
oplossingen. Het falen modulo $8$ is na te speuren: $(3-1)(3+1) = 2 \times
4 = 8 \equiv 0$ zonder dat een van beide factoren nul wordt. Moraal: de
vertrouwde regel ``een vergelijking van graad $d$ heeft hoogstens $d$
wortels'' is een stelling over \emph{\omterm{def:b1:structures:field}{integriteitsdomeinen}}
(\cref{cor:b1:poly:nroots} bewijst haar over \omterm{def:b1:structures:field}{lichamen}); in \omterm{def:b1:structures:ring}{ringen} met
nuldelers faalt ze stilzwijgend --- en precies daarom had het koppelbewijs
van de stelling van Wilson (\cref{exo:b1:arith:11}) een \omterm{def:b1:arith:prime}{priem} $p$ nodig.
\end{example}

\begin{example}[Rekenen in $\Z/n\Z$]\label{ex:b1:structures:zncomputation}
In $\Z/12\Z$ zijn de eenheden $\overline 1, \overline 5, \overline 7,
\overline{11}$ (de klassen die \omterm{cor:b1:arith:bezout}{relatief priem} zijn met $12$), en elk is
zijn eigen inverse ($5^2 = 25 \equiv 1$, $7^2 = 49 \equiv 1$, $11^2 = 121
\equiv 1$). De vergelijking $\overline 3\, x = \overline 6$ heeft
\emph{drie} oplossingen ($x \in \{\overline 2, \overline 6,
\overline{10}\}$): zonder inverteerbaarheid mag je niet schrappen. In
$\Z/11\Z$ daarentegen heeft elke vergelijking $\overline a x = \overline
b$ met $\overline a \neq \overline 0$ precies één oplossing.
\end{example}

\begin{example}[De groepsaxioma's als oplosvergunning]\label{ex:b1:structures:solving}
Los in de \omterm{def:b1:structures:group}{groep} $\bigl((\Z/7\Z)^*, \times\bigr)$ de vergelijking
$\overline 3\,x = \overline 5$ op. Volgens
\cref{prop:b1:structures:rules}~(3) bestaat de oplossing, is ze uniek en
is ze gelijk aan $\overline3^{-1}\, \overline5$; omdat $\overline3 \times
\overline5 = \overline{15} = \overline1$, is de inverse van $\overline 3$
gelijk aan $\overline 5$, zodat
\[
x = \overline5 \times \overline5 = \overline{25} = \overline4,
\qquad\text{controle: } \overline3 \times \overline4 =
\overline{12} = \overline5 .
\]
Het gaat minder om het antwoord dan om de garantie: in een \omterm{def:b1:structures:group}{groep} is elke
zulke vergelijking al vóór enige berekening op precies één manier
oplosbaar, zodat een oplosprocedure nooit op ``geen oplossing'' of
``meerdere'' kan stuiten. Vergelijk $\overline3\,x = \overline6$ in
$\Z/12\Z$ hierboven, waar die garantie faalt --- weten in welke structuur
je zit, is weten wat je als vanzelfsprekend mag aannemen.
\end{example}

\begin{example}[Directe producten]\label{ex:b1:structures:product}
Zijn $G$ en $H$ \omterm{def:b1:structures:group}{groepen}, dan is de productverzameling $G \times H$ met de
componentsgewijze bewerking $(g, h)(g', h') = (gg', hh')$ een \omterm{def:b1:structures:group}{groep}: de
axioma's ga je coördinaat voor coördinaat na, met neutraal element $(e_G,
e_H)$ en inversen $(g^{-1}, h^{-1})$. \omterm{def:b1:structures:order}{Ordes} combineren via het kgv: $(g,
h)^m = (g^m, h^m)$ is het neutrale element precies wanneer de \omterm{def:b1:structures:order}{orde} van $g$
en die van $h$ beide $m$ delen. Zo heeft in $\Z/2\Z \times \Z/2\Z$
(additief) elk element ongelijk aan nul \omterm{def:b1:structures:order}{orde} $2$ --- dat is precies de
rechthoeksgroep van \cref{ex:b1:structures:klein} in coördinaten ---
terwijl $\Z/4\Z$ een element van \omterm{def:b1:structures:order}{orde} $4$ bezit: een tweede,
rekenvrij bewijs dat de twee \omterm{def:b1:structures:group}{groepen} van grootte $4$ niet isomorf zijn
(een isomorfisme bewaart \omterm{def:b1:structures:order}{ordes}). Producten zijn de makkelijkste manier om
uit oude \omterm{def:b1:structures:group}{groepen} nieuwe te maken, en het vlak $\R^2 = \R \times \R$ van
\cref{ch:b1:vspaces} is het belangrijkste geval van die constructie.
\end{example}

\begin{remark}[Fermat, structureel bekeken]
In het \omterm{def:b1:structures:field}{lichaam} $\Z/p\Z$ vormen de klassen ongelijk aan nul een
multiplicatieve \omterm{def:b1:structures:group}{groep} met $p - 1$ elementen, en de kleine stelling van
Fermat (\cref{thm:b1:arith:fermat}) zegt: elk element $x$ van die \omterm{def:b1:structures:group}{groep}
voldoet aan $x^{p-1} = \overline 1$. Dat is een geval van een algemeen
feit over eindige \omterm{def:b1:structures:group}{groepen} (de stelling van Lagrange), bewezen in het
tweede jaar; het koppelbewijs van de stelling van Wilson
(\cref{exo:b1:arith:11}) had die groepentheoretische smaak al.
\end{remark}

\begin{remark}[Tussenspel: wat abstractie oplevert]\label{rem:b1:structures:interlude}
Je mag je afvragen wat er gewonnen werd door bijvoorbeeld
\cref{prop:b1:structures:unique} voor een abstracte bewerking te bewijzen
in plaats van voor getallen. Het antwoord luidt: hefboomwerking. Dat
argument van twee regels dekt nu in één klap: inversen van functies onder
samenstelling (\cref{thm:b1:logic:inverse}, waarvan het uniciteitsbewijs
het woord voor woord herhaalt), inversen modulo $n$
(\cref{prop:b1:arith:invmod}), inversen van reële getallen ongelijk aan
nul, eenheden in elke \omterm{def:b1:structures:ring}{ring}, en --- ongezien --- de inverteerbare matrices
van \cref{ch:b1:matrices}, waar de uniciteit van $A^{-1}$ geen enkele regel
bewijs meer zal vergen. Dezelfde zuinigheid geldt voor
\cref{prop:b1:structures:kernel} (één injectiviteitscriterium, hergebruikt
voor lineaire \omterm{def:b1:logic:map}{afbeeldingen} in \cref{ch:b1:linmaps}) en voor het
deelgroepcriterium. Abstractie is hier geen algemeenheid om de
algemeenheid: het is de weigering hetzelfde lemma vijfmaal onder vijf
namen te bewijzen. De prijs --- bijhouden welke axioma's elke \omterm{def:b1:logic:statement}{uitspraak}
werkelijk gebruikte --- is precies wat de oefeningen van dit hoofdstuk
trainen.
\end{remark}

\begin{remark}[Waar dit hoofdstuk gebruikt wordt]\label{rem:b1:structures:whereused}
Het vocabulaire van dit hoofdstuk is de grammatica van de rest van het
volume. \omterm{def:b1:structures:ring}{Ringen} en \omterm{def:b1:structures:field}{lichamen} ordenen \cref{ch:b1:poly} ($K[X]$ is een \omterm{def:b1:structures:ring}{ring}
die $\Z$ nabootst) en \cref{ch:b1:fractions} ($K(X)$ is haar
breukenlichaam); vectorruimten (\cref{ch:b1:vspaces}) zijn \omterm{def:b1:structures:group}{abelse groepen}
waarop een \omterm{def:b1:structures:field}{lichaam} werkt; matrices (\cref{ch:b1:matrices}) vormen de eerste
serieus niet-commutatieve \omterm{def:b1:structures:ring}{ring} van het volume, en hun inverteerbare
elementen een \omterm{def:b1:structures:group}{groep} waarvan de studie de lineaire algebra zelf is.
\omterm{def:b1:structures:morphism}{Morfismen} en \omterm{def:b1:structures:morphism}{kernen} keren terug als lineaire \omterm{def:b1:logic:map}{afbeeldingen} en \omterm{def:b1:structures:morphism}{kernen} in
\cref{ch:b1:linmaps} --- \cref{prop:b1:structures:kernel} \emph{is} het
injectiviteitscriterium van dat hoofdstuk, hier eens en voor altijd
bewezen. De \omterm{ex:b1:structures:groups}{symmetrische groep}, de ster van de weekendopgave hieronder,
levert de signatuur waarop de determinanten in \cref{ch:b1:det} gebouwd
worden.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:structures:1}
Definieer op $E = \R \setminus \{1\}$ de bewerking $x * y = x + y - xy$.
Bewijs dat $(E, *)$ een \omterm{def:b1:structures:group}{abelse groep} is. \emph{(Wijs het neutrale element
en de inverse van $x$ aan; ga de stabiliteit na: waarom is $x * y \neq
1$?)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:2}
Welke van de volgende zijn \omterm{def:b1:structures:group}{groepen}?
\begin{enumerate}
  \item $(\intoo{0}{+\infty}, \times)$;
  \item $(\{-1, 0, 1\}, +)$;
  \item $(\Q^*, \times)$;
  \item de \omterm{def:b1:logic:sets}{verzameling} van de oneven gehele getallen onder optelling.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:3}
Schrijf de samenstellingstabel van de \omterm{ex:b1:structures:groups}{symmetrische groep}
$\mathfrak{S}_3$ van $\{1,2,3\}$ op (zes bijecties: de identiteit, drie
transposities en twee $3$-cykels), en wijs twee elementen aan die niet
commuteren.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:4}
Bewijs dat $H = \{z \in \C^* : \abs z = 1\}$ een \omterm{def:b1:structures:subgroup}{deelgroep} van $(\C^*,
\times)$ is, en dat $\R_+^*$ er nog een is; is $H \cup \R_+^*$ een
\omterm{def:b1:structures:subgroup}{deelgroep}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:5}
Zij $f \colon (\R, +) \to (\C^*, \times)$, $\theta \mapsto
\eu^{\iu\theta}$. Bewijs dat $f$ een \omterm{def:b1:structures:morphism}{morfisme} is, bereken $\ker f$ en
$\operatorname{im} f$, en leid uit
\cref{prop:b1:structures:kernel} af dat $f$ niet \omterm{def:b1:logic:inj}{injectief} is. Beperk het
domein zó dat ze \omterm{def:b1:logic:inj}{injectief} wordt op een zo groot mogelijk interval.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:6}
Zij $H, K$ \omterm{def:b1:structures:subgroup}{deelgroepen} van $G$. Bewijs dat $H \cap K$ een \omterm{def:b1:structures:subgroup}{deelgroep} is,
en dat $H \cup K$ \emph{alleen} een \omterm{def:b1:structures:subgroup}{deelgroep} is wanneer $H \subseteq K$
of $K \subseteq H$. \emph{(Als $h \in H \setminus K$ en $k \in K \setminus
H$, waar kan $hk$ dan leven?)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:7}
Een \omterm{def:b1:structures:group}{groep} $G$ voldoet aan $x^2 = e$ voor alle $x \in G$. Bewijs dat $G$
\omterm{def:b1:structures:group}{abels} is. \emph{(Werk $(xy)^2$ uit.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:8}
In $\Z/18\Z$: som de eenheden op en bepaal de inverse van $\overline 5$;
los $\overline 5\, x = \overline 7$ op; los $\overline 6\, x = \overline
3$ en $\overline 6\, x = \overline{12}$ op.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:9}
Bewijs dat de \omterm{def:b1:logic:sets}{verzameling} $\Z[\sqrt 2] = \{a + b\sqrt 2 : a, b \in \Z\}$
een \omterm{def:b1:structures:ring}{ring} is (een deelring van $\R$), en dat $1 + \sqrt 2$ er een eenheid
van is met oneindig veel verschillende machten --- zodat $\Z[\sqrt
2]^\times$ oneindig is, anders dan $\Z^\times$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:structures:10}
(Booleaanse \omterm{def:b1:structures:ring}{ringen}) Zij $A$ een \omterm{def:b1:structures:ring}{ring} waarin $x^2 = x$ voor elke $x$.
Bewijs dat $x + x = 0$ voor alle $x$, en dat $A$ commutatief is.
\emph{(Werk $(x+x)^2$ en $(x+y)^2$ uit.)} Geef een voorbeeld van zo'n \omterm{def:b1:structures:ring}{ring}
met $\mathcal{P}(E)$, waarbij je het symmetrisch verschil als optelling en
de doorsnede als vermenigvuldiging neemt.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:structures:11}
Zij $G$ een \omterm{def:b1:structures:group}{groep} waarin, voor een zekere vaste $n \geq 1$, geldt $(xy)^n
= x^n y^n$, $(xy)^{n+1} = x^{n+1}y^{n+1}$ en $(xy)^{n+2} =
x^{n+2}y^{n+2}$ voor alle $x, y$. Bewijs dat $G$ \omterm{def:b1:structures:group}{abels} is. \emph{(Leid uit
de drie identiteiten eerst $y^n x = x y^n$ af, dan $y^{n+1} x = x
y^{n+1}$, en besluit.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:12}
\begin{enumerate}
  \item Bepaal alle groepsmorfismen van $(\Z, +)$ naar $(\Z, +)$.
  \item Bewijs dat het enige groepsmorfisme van $(\Q, +)$ naar $(\Z, +)$
        het nulmorfisme is. \emph{(Vergelijk voor $x \in \Q$ en $n \in
        \N^*$ de waarden $f(x)$ en $n\,f(x/n)$.)}
\end{enumerate}
\end{exercise}

\section{Opgave: de symmetrische groep en de schuifpuzzel}

\begin{problem}\label{pb:b1:structures:1}
De \omterm{def:b1:structures:group}{groep} $\mathfrak S_n$ van de \omterm{def:b1:counting:objects}{permutaties} van $\intint1n$ is de oudste
\omterm{def:b1:structures:group}{groep} in de wiskunde en nog altijd de leerzaamste. Deze opgave bouwt haar
structuurtheorie vanaf nul op --- cykels, voortbrenging door
transposities, het \omterm{def:b1:structures:morphism}{morfisme} \emph{signatuur} $\varepsilon \colon
\mathfrak S_n \to \{\pm1\}$ (waarvan het bestaan werkelijk niet-triviaal
is), en de \omterm{pb:b1:structures:1}{alternerende groep} $\mathfrak A_n$ voortgebracht door
$3$-cykels --- en verzilvert die dan op een klassieke puzzel: in het
schuifspel van $3 \times 3$ kan geen enkele reeks zetten twee steentjes
verwisselen en al het overige op zijn plaats laten.
\index{symmetrische groep}\index{signatuur}\index{alternerende groep}
\omterm{def:b1:counting:objects}{Permutaties} werken op $\intint1n$; het product $\sigma\tau$ betekent
``pas eerst $\tau$ toe''; en $[\,v_1, \dots, v_n]$ noteert de \omterm{def:b1:counting:objects}{permutatie}
die $i$ naar $v_i$ stuurt.

\medskip
\noindent\textbf{Deel I --- Cykels en transposities.}
\begin{enumerate}
  \item Verantwoord $\abs{\mathfrak S_n} = n!$
        (\cref{thm:b1:counting:counts}). Bereken in $\mathfrak S_3$
        beide producten van $\sigma = [2, 3, 1]$ en $\tau = [1, 3, 2]$,
        en besluit dat $\mathfrak S_3$ niet \omterm{def:b1:structures:group}{abels} is.
  \item Een \emph{$k$-cykel} $(a_1\ a_2\ \dots\ a_k)$ ($k \geq 2$, de
        $a_i$ twee aan twee verschillend) stuurt $a_1 \mapsto a_2
        \mapsto \dots \mapsto a_k \mapsto a_1$ en houdt al het overige
        vast; zijn \emph{drager} is $\{a_1, \dots, a_k\}$. Bewijs dat
        twee cykels met disjuncte dragers commuteren.
  \item Bewijs dat elke $\sigma \in \mathfrak S_n$ een product is van
        cykels met twee aan twee disjuncte dragers, en dat die
        ontbinding uniek is op de volgorde van de factoren na.
        \emph{(Beschouw voor elke $i$ de rij $i, \sigma(i),
        \sigma^2(i), \dots$: ze moet naar $i$ terugkeren; de zo
        verkregen \emph{banen} partitioneren $\intint1n$, en $\sigma$
        werkt op elk ervan als een cykel.)}
  \item Ontbind $\sigma = [4, 1, 5, 2, 3, 7, 8, 6] \in \mathfrak S_8$ in
        disjuncte cykels. Definieer de \emph{\omterm{def:b1:structures:order}{orde}} van $\sigma$ als in
        \cref{def:b1:structures:order}, bewijs dat de \omterm{def:b1:structures:order}{orde} van een
        product van disjuncte cykels het kgv van hun lengtes is, en
        bereken de \omterm{def:b1:structures:order}{orde} van deze $\sigma$.
  \item Bewijs de telescoperende identiteit
        \[
        (a_1\ a_2\ \dots\ a_k)
        = (a_1\ a_k)(a_1\ a_{k-1})\cdots(a_1\ a_2) ,
        \]
        en besluit dat elke \omterm{def:b1:counting:objects}{permutatie} een product van transposities is.
        Schrijf de $\sigma$ van vraag 4 als zo'n product.
  \item Toon bovendien aan dat de \emph{naburige} transposities $(i\ \
        i{+}1)$ volstaan: voor $a < b$ is
        \[
        (a\ b) = (a\ \ a{+}1)(a{+}1\ \ a{+}2)\cdots(b{-}1\ \ b)
        \cdots(a{+}1\ \ a{+}2)(a\ \ a{+}1),
        \]
        een product van $2(b - a) - 1$ naburige transposities --- een
        \emph{oneven} aantal (die pariteit doet er hieronder tweemaal
        toe).
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De signatuur bestaat.} Zij voor $\sigma \in
\mathfrak S_n$
\[
N(\sigma) = \#\bigl\{(i, j) : i < j,\ \sigma(i) >
\sigma(j)\bigr\}
\]
haar aantal \emph{inversies}, en zet $\varepsilon(\sigma) =
(-1)^{N(\sigma)}$.
\begin{enumerate}[resume]
  \item Bereken $N$ en $\varepsilon$ voor de identiteit, voor een
        transpositie $(i\ \ i{+}1)$, en voor $[2, 3, 1]$.
  \item Bewijs dat voor elke $\sigma$ en elke naburige transpositie
        $\tau = (i\ \ i{+}1)$ geldt: $N(\sigma\tau) = N(\sigma) \pm 1$.
        \emph{(Rechts samenstellen met $\tau$ verwisselt de waarden op
        de plaatsen $i$ en $i + 1$; precies één paar verandert van
        inversiestatus.)}
  \item Leid met vraag 6 af dat voor \emph{elke} transpositie $\tau$
        geldt $\varepsilon(\sigma\tau) = -\varepsilon(\sigma)$; besluit
        dat wanneer $\sigma$ een product van $p$ transposities is,
        $\varepsilon(\sigma) = (-1)^p$ --- in het bijzonder hangt de
        pariteit van $p$ alleen van $\sigma$ af en niet van de gekozen
        ontbinding --- en dat $\varepsilon \colon \mathfrak S_n \to
        \{\pm 1\}$ een groepsmorfisme is.
  \item Toon aan dat een $k$-cykel signatuur $(-1)^{k-1}$ heeft, en dat
        in het algemeen $\varepsilon(\sigma) = (-1)^{n - c(\sigma)}$,
        waarbij $c(\sigma)$ het aantal banen van $\sigma$ is (vaste
        punten meegerekend).
  \item De \emph{\omterm{pb:b1:structures:1}{alternerende groep}} is $\mathfrak A_n =
        \ker\varepsilon$. Verantwoord dat het een \omterm{def:b1:structures:subgroup}{deelgroep} is en bewijs
        $\abs{\mathfrak A_n} = \frac{n!}2$ voor $n \geq 2$. \emph{(Leg
        een transpositie $\tau_0$ vast en beschouw $\sigma \mapsto
        \sigma\tau_0$.)}
  \item Controle op $\sigma = [4, 1, 5, 2, 3, 7, 8, 6]$: bereken
        $\varepsilon(\sigma)$ op drie manieren --- door de inversies te
        tellen, uit het cykeltype via vraag 10, en uit je aantal
        transposities in vraag 5.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- $\mathfrak A_n$ wordt voortgebracht door
$3$-cykels.}
\begin{enumerate}[resume]
  \item Zij $a, b, c, d$ twee aan twee verschillend. Ga de twee
        identiteiten
        \[
        (a\ b)(a\ c) = (a\ c\ b),
        \qquad
        (a\ b)(c\ d) = (a\ c\ b)(a\ c\ d)
        \]
        na.
  \item Bewijs dat voor $n \geq 3$ elk element van $\mathfrak A_n$ een
        product van $3$-cykels is. \emph{(Een even \omterm{def:b1:counting:objects}{permutatie} is een
        product van een even aantal transposities; slik ze twee aan twee
        in.)}
  \item Schrijf $(1\ 2)(3\ 4)$ en de $5$-cykel $(1\ 2\ 3\ 4\ 5)$
        expliciet als producten van $3$-cykels.
  \item Bewijs de conjugatieformule: voor elke $\sigma \in \mathfrak
        S_n$ geldt
        \[
        \sigma\,(a_1\ \dots\ a_k)\,\sigma^{-1}
        = \bigl(\sigma(a_1)\ \dots\ \sigma(a_k)\bigr) .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De schuifpuzzel.} De steentjes $1, \dots, 8$
schuiven in een raamwerk van $3 \times 3$ met één leeg vakje; een
\emph{zet} schuift een steentje dat aan het lege vakje grenst daarin. Nummer
de vakjes $1, \dots, 9$ (rij voor rij; in de opgeloste stand ligt steentje
$i$ in vakje $i$ en is vakje $9$ leeg). Behandel het lege vakje als een
negende steentje, zodat een stand een \omterm{def:b1:counting:objects}{permutatie} $\sigma \in \mathfrak
S_9$ is (steentje $\sigma(i)$ ligt in vakje $i$).
\begin{enumerate}[resume]
  \item Toon aan dat een zet $\sigma$ vervangt door $\sigma \circ \tau$,
        waarbij $\tau$ de transpositie van de twee betrokken vakjes is;
        leid af dat elke zet $\varepsilon(\sigma)$ omklapt.
  \item Zij $d(\sigma)$ de manhattanafstand (rijen plus kolommen) tussen
        het huidige vakje van het lege vakje en zijn thuisvakje $9$.
        Toon aan dat elke zet $d$ met $\pm1$ verandert, zodat elke zet
        ook $(-1)^{d(\sigma)}$ omklapt. Besluit dat
        \[
        I(\sigma) = \varepsilon(\sigma)\cdot(-1)^{d(\sigma)}
        \]
        \emph{invariant} is onder elke zet.
  \item Bewijs de klassieke onmogelijkheid van de puzzel: de stand
        waarin de steentjes $7$ en $8$ verwisseld zijn en al het overige
        (het lege vakje inbegrepen) op zijn plaats ligt, is vanuit de
        opgeloste stand onbereikbaar.
  \item We nemen de omkering aan (het bewijs is een leerzame maar lange
        inductie): elke stand met $I = +1$ is bereikbaar. Leid af dat
        precies de helft van de $8!$ standen met het lege vakje thuis
        oplosbaar is, dat wil zeggen $\frac{8!}2 = 20\,160$.
  \item Leid uit vraag 20 af dat de bereikbare schikkingen van de
        steentjes met het lege vakje thuis precies de \omterm{def:b1:structures:subgroup}{deelgroep}
        $\mathfrak A_8 \leq \mathfrak S_8$ vormen.
  \item Toepassingen van de invariant: kun je (a) de stand bereiken
        waarin de steentjes $1, 2, 3$ cyclisch verwisseld zijn en al het
        overige, het lege vakje inbegrepen, thuis ligt? (b) de stand
        waarin steentje $5$ en het lege vakje van plaats gewisseld zijn
        en alle andere steentjes thuis liggen? Verantwoord beide
        antwoorden met $I$.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Synthese.}
\begin{enumerate}[resume]
  \item Bewijs dat voor $n \geq 3$ de enige groepsmorfismen $f \colon
        \mathfrak S_n \to \{\pm 1\}$ het constante \omterm{def:b1:structures:morphism}{morfisme} en
        $\varepsilon$ zijn. \emph{(Toon met vraag 16 en de
        commutativiteit van $\{\pm1\}$ aan dat $f$ op alle transposities
        dezelfde waarde aanneemt.)}
  \item Waar precies gebruikte de opgave: (i) het begrip \omterm{def:b1:structures:morphism}{morfisme} en
        \cref{prop:b1:structures:kernel}; (ii) de telprincipes van
        \cref{ch:b1:counting}; (iii) de kwestie van de
        welgedefinieerdheid die de vragen 8--9 beslechten? Eén zin per
        onderdeel.
  \item Synthese, in een korte alinea: één pariteitsfunctie, één keer
        welgedefinieerd bewezen, ordent tegelijk de inwendige structuur
        van $\mathfrak S_n$ (de \omterm{def:b1:structures:subgroup}{deelgroep} $\mathfrak A_n$), beslist een
        fysieke puzzel, en zal --- via de formule $\det = \sum_\sigma
        \varepsilon(\sigma)\cdots$ --- de determinanten in
        \cref{ch:b1:det} definiëren. Becommentarieer het terugkerende
        patroon: invarianten zetten ``probeer alle reeksen zetten'' om in
        één berekening.
\end{enumerate}
\end{problem}
