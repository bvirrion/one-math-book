\chapter{Limieten en continuïteit}\label{ch:b1:continuity}

De tussenwaardestelling en de extremumstelling werden in het
bovenbouwvolume op visueel vertrouwen gebruikt. Met de rijen
(\cref{ch:b1:seq}) en de topologie van $\R$ (\cref{ch:b1:topology}) in
handen bewijst dit hoofdstuk ze --- en vervolledigt het de theorie met de
stelling over de monotone bijectie (die $\arcsin$,
$\operatorname{arcosh}$ en hun verwanten uit \cref{ch:b1:functions}
wettigt) en met de stelling van Heine over de \omterm{def:b1:continuity:uniform}{uniforme continuïteit}.

Overal is $I$ een \omterm{prop:b1:reals:intervals}{interval} en $f \colon I \to \R$; ``$x_0 \in \overline
I$'' laat limieten in de randpunten toe.

\section{Limieten van functies}

\begin{definition}[Limiet in een punt]\label{def:b1:continuity:limit}
Zij $x_0 \in \overline{I}$ en $\ell \in \R$. Dan gaat $f(x) \to \ell$
voor $x \to x_0$ wanneer\index{limiet!van een functie}
\[
\forall \varepsilon > 0,\ \exists \delta > 0,\ \forall x \in I,
\qquad \abs{x - x_0} \leq \delta \implies \abs{f(x) - \ell} \leq
\varepsilon .
\]
Limieten in $\pm\infty$ en oneindige limieten worden volgens hetzelfde
patroon gedefinieerd ($\abs{x - x_0} \leq \delta$ wordt $x \geq M$;
$\abs{f(x) - \ell} \leq \varepsilon$ wordt $f(x) \geq M'$). Eenzijdige
limieten beperken $x$ tot $x > x_0$ (genoteerd $x \to x_0^+$) of tot $x <
x_0$. De limiet is uniek zodra ze bestaat (hetzelfde bewijs als
\cref{prop:b1:seq:first}).
\end{definition}

\begin{example}[Een limiet door insluiting]\label{ex:b1:continuity:floorlimit}
Bereken $\lim_{x \to 0} x\,\bigl\lfloor \frac1x \bigr\rfloor$. De
insluiting van de \omterm{thm:b1:reals:floor}{vloer} $\frac1x - 1 < \lfloor \frac1x \rfloor \leq
\frac1x$ geeft, na vermenigvuldigen met $x$ (let op het teken!):
\[
1 - x < x\Bigl\lfloor \frac1x \Bigr\rfloor \leq 1 \quad (x > 0),
\qquad
1 \leq x\Bigl\lfloor \frac1x \Bigr\rfloor < 1 - x \quad (x < 0),
\]
en beide eenzijdige insluitingen sluiten zich op $1$: de limiet is $1$.
Merk op wat er gebeurde: $\lfloor \frac1x\rfloor$ alleen maakt vlak bij
$0$ wilde sprongen, maar de factor $x$ temt elke sprong ($x$ maal een
sprong van $1$ is klein), en alleen de insluiting overleeft. Het inzicht
tot slot: limieten van producten van een kleine factor met een factor van
begrensde oscillatie zijn insluitproblemen en nooit problemen voor de
stelling over de bewerkingen --- die vergt dat \emph{beide} factoren
convergeren.
\end{example}

\begin{theorem}[Karakterisering met rijen]\label{thm:b1:continuity:seqchar}
$f(x) \to \ell$ voor $x \to x_0$ dan en slechts dan als voor
\emph{elke} rij $(u_n)$ punten van $I$ met $u_n \to x_0$ geldt $f(u_n)
\to \ell$.
\end{theorem}

\begin{proof}
($\Rightarrow$) Zij $u_n \to x_0$ en $\varepsilon > 0$. Neem $\delta$ uit
de definitie, en daarna $N$ met $\abs{u_n - x_0} \leq \delta$ voor $n
\geq N$: voorbij $N$ is $\abs{f(u_n) - \ell} \leq \varepsilon$.

($\Leftarrow$) Via de contrapositie. Gaat $f \not\to \ell$, dan verslaat
een zekere $\varepsilon_0 > 0$ elke $\delta$; de keuze $\delta =
\frac{1}{n+1}$ levert $u_n \in I$ met $\abs{u_n - x_0} \leq
\frac{1}{n+1}$ en $\abs{f(u_n) - \ell} > \varepsilon_0$. Dan is $u_n \to
x_0$ terwijl $f(u_n) \not\to \ell$.
\end{proof}

\begin{corollary}[Bewerkingen, samenstelling, ordening]\label{cor:b1:continuity:operations}
Sommen, producten en quotiënten (met een limiet ongelijk aan nul in de
noemer) van limieten gedragen zich als bij rijen; gaat $f \to \ell$ in
$x_0$ en $g \to m$ in $\ell$, en is $g$ rond $\ell$ gedefinieerd met
$g(\ell) = m$ of $f \neq \ell$ nabij $x_0$, dan gaat $g \circ f \to m$ in
$x_0$; limieten bewaren niet-strikte ongelijkheden, en de insluitstelling
geldt.
\end{corollary}

\begin{proof}
Elke \omterm{def:b1:logic:statement}{uitspraak} draagt via \cref{thm:b1:continuity:seqchar} over naar haar
tegenhanger voor rijen
(\cref{thm:b1:seq:operations,thm:b1:seq:order}). Voor de samenstelling
loont het de rechtstreekse keten één keer op te schrijven: bij gegeven
$\varepsilon > 0$ levert de limiet van $g$ in $\ell$ een $\eta > 0$ met
\[
\abs{y - \ell} \leq \eta \implies \abs{g(y) - m} \leq
\varepsilon \quad (y \text{ in het domein van } g),
\]
waarbij het geval $y = \ell$ gedekt is omdat $g(\ell) = m$; vervolgens
levert de limiet van $f$ in $x_0$ een $\delta > 0$ met $\abs{x - x_0}
\leq \delta \implies \abs{f(x) - \ell} \leq \eta$; aaneenrijgen geeft dat
$\abs{x - x_0} \leq \delta$ leidt tot $\abs{g(f(x)) - m} \leq
\varepsilon$. Onder de alternatieve hypothese ($f \neq \ell$ nabij $x_0$)
wordt de waarde $y = \ell$ nooit aan $g$ gevoed en doet haar waarde daar
niet ter zake.
\end{proof}

\begin{example}[Waarom het voorbehoud bij de samenstelling bestaat]\label{ex:b1:continuity:compositiontrap}
Zij $g(y) = 0$ voor $y \neq 0$ en $g(0) = 1$, en zij $f$ identiek $0$.
Dan gaat $f(x) \to 0$ voor $x \to 0$ en $g(y) \to 0$ voor $y \to 0$; en
toch is $g(f(x)) = g(0) = 1$ voor elke $x$, zodat $g \circ f \to 1 \neq
0$. De binnenste functie zit voor altijd \emph{precies op} de verboden
waarde $\ell = 0$, en de limiet van $g$ in $0$ negeert wat $g$ in $0$
doet. Het voorbehoud van \cref{cor:b1:continuity:operations} --- ofwel
$g(\ell) = m$ (dus $g$ \omterm{def:b1:continuity:continuous}{continu} in $\ell$), ofwel $f \neq \ell$ nabij
$x_0$ --- sluit precies dit uit. Het inzicht tot slot: in de praktijk
stel je \emph{\omterm{def:b1:continuity:continuous}{continue}} functies samen en is het voorbehoud gratis; het
bijt alleen wanneer limieten langs geperforeerde \omterm{def:b1:topology:open}{omgevingen} genomen
worden, en daarom sluit de definitie van $\lim_{x \to x_0}$ in dit boek
het punt zelf in wanneer het in het domein ligt.
\end{example}

\section{Continuïteit}

\begin{definition}\label{def:b1:continuity:continuous}
$f$ is \emph{continu in $x_0 \in I$}\index{continuïteit} wanneer $f(x)
\to f(x_0)$ voor $x \to x_0$; en \emph{continu op $I$} wanneer ze in elk
punt continu is. Volgens \cref{thm:b1:continuity:seqchar} is $f$ continu
in $x_0$ precies wanneer $f(u_n) \to f(x_0)$ voor elke rij $u_n \to x_0$
in $I$.
\end{definition}

\begin{example}[Een taxonomie van discontinuïteiten]\label{ex:b1:continuity:taxonomy}
Drie manieren om in een punt te falen, in toenemende ernst.
\emph{Opheffbaar}: $f(x) = \frac{\sin x}{x}$ op $\R^*$ heeft limiet $1$
in $0$; $f(0) = 1$ definiëren herstelt de zaak --- de discontinuïteit was
een gat, geen eigenschap. \emph{Sprong}: $\lfloor x \rfloor$ heeft in een
geheel getal verschillende eenzijdige limieten ($n - 1$ en $n$); geen
enkele keuze van waarde verzoent ze, maar beide halve limieten bestaan.
\emph{Essentieel}: $\sin\frac1x$ heeft in $0$ helemaal geen eenzijdige
limiet (\cref{exo:b1:continuity:1}) --- oscillatie zonder tot rust te
komen. Monotone functies kunnen alleen het middelste soort voortbrengen
(hun eenzijdige limieten bestaan altijd, als \omterm{def:b1:reals:bounds}{suprema} en \omterm{def:b1:reals:bounds}{infima}), en
daarom zijn hun \omterm{def:b1:logic:sets}{verzamelingen} discontinuïteiten hoogstens aftelbaar ---
één rationaal getal per sprong. Afgeleiden kunnen, wegens de stelling van
Darboux (\cref{exo:b1:derivative:10}), alleen het laatste soort
voortbrengen: een functie met een sprongdiscontinuïteit is nooit de
afgeleide van iets.
\end{example}

\begin{proposition}\label{prop:b1:continuity:algebra}
Sommen, producten, quotiënten (waar gedefinieerd) en samenstellingen van
\omterm{def:b1:continuity:continuous}{continue} functies zijn \omterm{def:b1:continuity:continuous}{continu}. \omterm{def:b1:poly:def}{Veeltermen}, \omterm{def:b1:fractions:field}{rationale breuken} (buiten hun
\omterm{def:b1:fractions:field}{polen}), $\abs{\,\cdot\,}$, $\exp$, $\ln$, de goniometrische en de
\omterm{def:b1:functions:hyperbolic}{hyperbolische functies} en hun inversen (\cref{ch:b1:functions}) zijn
\omterm{def:b1:continuity:continuous}{continu} op hun domein.
\end{proposition}

\begin{proof}
Bewerkingen: \cref{cor:b1:continuity:operations}. Constanten en de
identiteit zijn rechtstreeks uit de definitie \omterm{def:b1:continuity:continuous}{continu} ($\delta =
\varepsilon$ bedient de identiteit, elke $\delta$ de constanten); omdat
producten van \omterm{def:b1:continuity:continuous}{continue} functies \omterm{def:b1:continuity:continuous}{continu} zijn, volgt elk monoom $a_k x^k$
met inductie naar $k$, en sommen maken de \omterm{def:b1:poly:def}{veeltermen} af; een \omterm{def:b1:fractions:field}{rationale
breuk} is een quotiënt van twee \omterm{def:b1:poly:def}{veeltermen}, \omterm{def:b1:continuity:continuous}{continu} overal waar de noemer
niet nul wordt. $\abs{\,\cdot\,}$: de omgekeerde driehoeksongelijkheid
geeft $\bigl|\abs{f(x)} - \abs{f(x_0)}\bigr| \leq \abs{f(x) - f(x_0)}$,
zodat dezelfde $\delta$ voldoet. Voor de klassieke functies nemen we de
\omterm{def:b1:continuity:continuous}{continuïteit} hier aan; de afleidbaarheid (bewezen in
\cref{ch:b1:derivative}) is sterker.
\end{proof}

\begin{example}[Maximum en minimum van continue functies]\label{ex:b1:continuity:maxmin}
Zijn $f$ en $g$ \omterm{def:b1:continuity:continuous}{continu}, dan ook $\max(f, g)$ en $\min(f, g)$: er is geen
gevalsonderscheid nodig dankzij de identiteiten
\[
\max(f, g) = \frac{f + g + \abs{f - g}}{2},
\qquad
\min(f, g) = \frac{f + g - \abs{f - g}}{2},
\]
en de \omterm{def:b1:continuity:continuous}{continuïteit} van sommen en van $\abs{\,\cdot\,}$
(\cref{prop:b1:continuity:algebra}). In het bijzonder zijn $f^+ = \max(f,
0)$ en $f^- = \max(-f, 0)$ \omterm{def:b1:continuity:continuous}{continu} met $f = f^+ - f^-$: het splitsen naar
teken dat bij reeksen gebruikt wordt (\cref{ch:b1:series}) en, op volle
schaal, in de integratietheorie van het volume van bachelorjaar 3, kost
niets aan regelmaat.
\end{example}

\begin{theorem}[Tussenwaardestelling]\label{thm:b1:continuity:ivt}
Zij $f$ \omterm{def:b1:continuity:continuous}{continu} op $\intcc{a}{b}$ met $f(a) \leq 0 \leq f(b)$. Dan is
$f(c) = 0$ voor een zekere $c \in \intcc{a}{b}$. Bijgevolg neemt een
\omterm{def:b1:continuity:continuous}{continue} functie op een \omterm{prop:b1:reals:intervals}{interval} elke waarde tussen twee van haar waarden
aan: $f(I)$ is een \omterm{prop:b1:reals:intervals}{interval}.\index{tussenwaardestelling}
\end{theorem}

\begin{proof}
Dichotomie. Zet $a_0 = a$ en $b_0 = b$. Is $\intcc{a_n}{b_n}$ gegeven met
$f(a_n) \leq 0 \leq f(b_n)$, zij $m$ dan het midden: is $f(m) \leq 0$,
houd dan $\intcc{m}{b_n}$, en anders $\intcc{a_n}{m}$; de
tekenvoorwaarden blijven gelden. De rijen $(a_n)$ en $(b_n)$ zijn
ingesloten ($b_n - a_n = \frac{b-a}{2^n}$), met gemeenschappelijke limiet
$c$ (\cref{thm:b1:seq:adjacent}). Wegens de \omterm{def:b1:continuity:continuous}{continuïteit} en
\cref{thm:b1:seq:order} is $f(c) = \lim f(a_n) \leq 0$ en $f(c) = \lim
f(b_n) \geq 0$, dus $f(c) = 0$.

Voor het gevolg: gegeven waarden $f(u) < v < f(w)$, pas het bovenstaande
toe op $x \mapsto f(x) - v$ op het segment met eindpunten $u$ en $w$;
$f(I)$ is dus convex, oftewel een \omterm{prop:b1:reals:intervals}{interval}
(\cref{prop:b1:reals:intervals}).
\end{proof}

\begin{example}[Eén vergelijking, het volledige protocol]\label{ex:b1:continuity:protocol}
Los $\eu^x = 3 - x$ op over $\R$: bestaan, uniciteit, lokalisering. Zet
$g(x) = \eu^x + x - 3$, \omterm{def:b1:continuity:continuous}{continu}. Lokalisering en bestaan: $g(0) = -2 < 0$
en $g(1) = \eu - 2 > 0$, dus plant de tussenwaardestelling een oplossing
in $\intoo{0}{1}$. Uniciteit: $g$ is een som van de strikt stijgende
$\eu^x$ en $x - 3$, en dus strikt stijgend op $\R$; een strikt monotone
functie neemt elke waarde hoogstens één keer aan, zodat de oplossing op
heel $\R$ uniek is (niet alleen in het onderzochte \omterm{prop:b1:reals:intervals}{interval}). Het
protocol --- herschrijf tot $g = 0$, een tekenwisseling voor het bestaan,
de monotonie voor de uniciteit --- beslecht de meeste vragen van het type
``hoeveel oplossingen'' in drie regels, en de verlooptabellen van
\cref{ch:b1:derivative} breiden het uit tot niet-monotone $g$ door $\R$
in monotone takken te knippen.
\end{example}

\begin{example}[Dichotomie als algoritme]\label{ex:b1:continuity:dichotomy}
Het bewijs van \cref{thm:b1:continuity:ivt} rekent. Neem $f(x) = x^3 + x
- 1$: $f(0) = -1 < 0 < 1 = f(1)$, dus ligt er een wortel in
$\intoo{0}{1}$. Halveren:
\[
f(0.5) = -0.375 < 0, \qquad
f(0.75) = 0.171875 > 0, \qquad
f(0.625) = -0.130859375 < 0 ,
\]
zodat de wortel achtereenvolgens in $\intoo{0.5}{1}$, dan
$\intoo{0.5}{0.75}$, dan $\intoo{0.625}{0.75}$ opgesloten raakt (de ware
waarde is $c \approx 0.6823$). Na $n$ stappen is de fout hoogstens
$\frac{b - a}{2^n}$: tien stappen geven drie decimalen, twintig geven er
zes. Het inzicht tot slot: de tussenwaardestelling is niet alleen een
bestaansuitspraak --- haar dichotomiebewijs is een gewaarborgd, zij het
traag, algoritme om wortels te zoeken, waartegen de snelle maar lokale
methode van Newton uit \cref{ch:b1:derivative} afgewogen moet worden.
\end{example}

\begin{theorem}[Extremumstelling]\label{thm:b1:continuity:evt}
Een \omterm{def:b1:continuity:continuous}{continue} functie op een segment $\intcc{a}{b}$ is begrensd en bereikt
haar grenzen: er zijn $c, d \in \intcc{a}{b}$ met
\[
f(c) = \inf_{\intcc{a}{b}} f,
\qquad
f(d) = \sup_{\intcc{a}{b}} f .
\]
Samen met \cref{thm:b1:continuity:ivt}: \emph{het \omterm{def:b1:continuity:continuous}{continue} beeld van een
segment is een segment} $\intcc{f(c)}{f(d)}$.
\index{extremumstelling}
\end{theorem}

\begin{proof}
\emph{Naar boven begrensd:} anders kies je $u_n$ met $f(u_n) \geq n$.
Wegens de compactheid van het segment (\cref{thm:b1:topology:compact})
convergeert een \omterm{def:b1:seq:subsequence}{deelrij} $u_{\varphi(n)} \to x \in \intcc{a}{b}$; de
\omterm{def:b1:continuity:continuous}{continuïteit} geeft $f(u_{\varphi(n)}) \to f(x)$, terwijl
$f(u_{\varphi(n)}) \geq \varphi(n) \to +\infty$: tegenspraak.

\emph{\omterm{def:b1:reals:bounds}{Supremum} bereikt:} zij $M = \sup f$ en kies $v_n$ met $f(v_n) > M -
\frac{1}{n+1}$ (\cref{prop:b1:reals:epsilon}). Extraheer $v_{\varphi(n)}
\to d \in \intcc{a}{b}$: dan is $f(d) = \lim f(v_{\varphi(n)}) = M$
wegens de insluiting. Het \omterm{def:b1:reals:bounds}{infimum} behandel je met $-f$.
\end{proof}

\begin{example}[Positief minimum op een segment]\label{ex:b1:continuity:posmin}
Zij $f$ \omterm{def:b1:continuity:continuous}{continu} op $\intcc{0}{1}$ met $f(x) > 0$ voor elke $x$. Dan is
$\inf f > 0$: volgens de extremumstelling is het \omterm{def:b1:reals:bounds}{infimum} een
\emph{waarde} $f(c)$, en $f(c) > 0$ per hypothese. Een \omterm{def:b1:continuity:continuous}{continue} positieve
functie op een segment blijft dus van $0$ weg --- een argument van twee
regels dat in de komende hoofdstukken tientallen keren gebruikt wordt
(noemers onder controle, insluitingen met trapfuncties, foutgrenzen). Op
een niet-compact \omterm{prop:b1:reals:intervals}{interval} faalt dat spectaculair: $f(x) = x$ op
$\intoc{0}{1}$ is \omterm{def:b1:continuity:continuous}{continu} en positief met $\inf f = 0$, niet bereikt. Het
inzicht tot slot: ``positief'' wordt precies dan ``uniform positief''
wanneer het domein compact is; elke hypothese van de extremumstelling
draagt gewicht.
\end{example}

\begin{remark}[Veelgemaakte fouten rond de drie stellingen]
(i) \emph{\omterm{def:b1:continuity:continuous}{Continue} beelden}: alleen \emph{segmenten} zijn robuust. Het
\omterm{def:b1:continuity:continuous}{continue} beeld van een \omterm{def:b1:topology:open}{open} \omterm{prop:b1:reals:intervals}{interval} hoeft niet \omterm{def:b1:topology:open}{open} te zijn ($x \mapsto
x^2$ beeldt $\intoo{-1}{1}$ af op $\intco{0}{1}$), het beeld van een
\omterm{def:b1:topology:closed}{gesloten verzameling} hoeft niet \omterm{def:b1:topology:closed}{gesloten} te zijn ($\arctan$ beeldt het
\omterm{def:b1:topology:closed}{gesloten} $\R$ af op het \omterm{def:b1:topology:open}{open} $\intoo{-\frac\pi2}{\frac\pi2}$); maar het
beeld van een segment is een segment
(\cref{thm:b1:continuity:evt}). (ii) \emph{De tussenwaardestelling heeft
een \omterm{prop:b1:reals:intervals}{interval} nodig}: de functie $x \mapsto \frac1x$, \omterm{def:b1:continuity:continuous}{continu} op
$\intco{-1}{0} \cup \intoc{0}{1}$, neemt de waarden $-1$ en $1$ aan en
wordt toch nooit nul --- haar domein bestaat uit twee losse stukken, en
de waarde $0$ valt in het gat; benoem altijd het \omterm{prop:b1:reals:intervals}{interval} waarop je de
stelling toepast. (iii) \emph{\omterm{def:b1:continuity:uniform}{Uniforme continuïteit} is een eigenschap van
het paar (functie, \omterm{def:b1:logic:sets}{verzameling})}: $x^2$ is op elk segment \omterm{def:b1:continuity:uniform}{uniform continu}
en op $\R$ niet (\cref{ex:b1:continuity:notuniform}) --- de woorden
``\omterm{def:b1:continuity:uniform}{uniform continu}'' zonder domein zijn betekenisloos. (iv) \emph{De
\omterm{def:b1:continuity:continuous}{continuïteit} van de inverse is niet formeel}: ze geldt op \omterm{prop:b1:reals:intervals}{intervallen} via
de monotonie (\cref{thm:b1:continuity:bijection}), maar een \omterm{def:b1:continuity:continuous}{continue}
bijectie tussen verenigingen van \omterm{prop:b1:reals:intervals}{intervallen} kan een discontinue inverse
hebben --- de opmerking na die stelling staat er omdat studenten haar
zonder de intervalhypothese citeren.
\end{remark}

\section{Monotone functies en inverse functies}

\begin{theorem}[Stelling over de monotone bijectie]\label{thm:b1:continuity:bijection}
Zij $f$ \omterm{def:b1:continuity:continuous}{continu} en strikt monotoon op een \omterm{prop:b1:reals:intervals}{interval} $I$. Dan geldt:
\begin{enumerate}
  \item $f$ is een bijectie van $I$ op het \omterm{prop:b1:reals:intervals}{interval} $J = f(I)$;
  \item de inverse $f^{-1} \colon J \to I$ is strikt monotoon (in
        dezelfde richting) en \emph{\omterm{def:b1:continuity:continuous}{continu}}.
\end{enumerate}
\end{theorem}

\begin{proof}
Zeg dat $f$ strikt stijgend is. (1) De injectiviteit volgt onmiddellijk
uit de strikte monotonie; de surjectiviteit op $f(I)$ is triviaal, en
$f(I)$ is een \omterm{prop:b1:reals:intervals}{interval} wegens \cref{thm:b1:continuity:ivt}.

(2) $f^{-1}$ is strikt stijgend: was $y < y'$ in $J$ maar $f^{-1}(y) \geq
f^{-1}(y')$, dan geeft de stijgende $f$ toepassen dat $y \geq y'$,
ongerijmd. \omterm{def:b1:continuity:continuous}{Continuïteit} van $f^{-1}$ in $y_0 = f(x_0) \in J$: zij
$\varepsilon > 0$. Neem eerst aan dat $x_0$ \omterm{def:b1:topology:closure}{inwendig} aan $I$ is, en maak
$\varepsilon$ zo klein dat $x_0 \pm \varepsilon \in I$: hun beelden
voldoen aan $f(x_0 - \varepsilon) < y_0 < f(x_0 + \varepsilon)$. Neem
$\delta = \min\bigl(y_0 - f(x_0 - \varepsilon),\, f(x_0 + \varepsilon) -
y_0\bigr) > 0$: voor $\abs{y - y_0} \leq \delta$ klemt de monotonie van
$f^{-1}$ het getal $f^{-1}(y)$ tussen $x_0 - \varepsilon$ en $x_0 +
\varepsilon$. Is $x_0$ bijvoorbeeld het linkereindpunt van $I$, dan is
alleen $x_0 + \varepsilon$ beschikbaar: dan is $y_0 = \min J$
(monotonie), voldoet elke $y \in J$ met $y - y_0 \leq f(x_0 +
\varepsilon) - y_0$ aan $x_0 \leq f^{-1}(y) \leq x_0 + \varepsilon$, en
is de eenzijdige afschatting precies de \omterm{def:b1:continuity:continuous}{continuïteit} in een randpunt; het
rechtereindpunt verloopt symmetrisch. (Merk op: de \omterm{def:b1:continuity:continuous}{continuïteit} van
$f^{-1}$ werd \emph{niet} uit die van $f$ afgeleid door symmetrie --- het
is de monotonie op een \omterm{prop:b1:reals:intervals}{interval} die het werk doet.)
\end{proof}

\begin{remark}
Deze stelling is wat \cref{def:b1:functions:arc} en
\cref{prop:b1:functions:invhyp} stilzwijgend gebruikten: $\arcsin$,
$\arccos$, $\arctan$, $\operatorname{arsinh}$, \dots\ zijn \omterm{def:b1:continuity:continuous}{continu}. Een
aanvulling (\cref{exo:b1:continuity:10}): een \omterm{def:b1:continuity:continuous}{continue}
\emph{\omterm{def:b1:logic:inj}{injectieve}} functie op een \omterm{prop:b1:reals:intervals}{interval} is automatisch strikt monotoon,
zodat de hypothese van de monotonie niets kost.
\end{remark}

\begin{example}[Wortels van elke orde]\label{ex:b1:continuity:nthroot}
Voor $n \in \N^*$ is de functie $f(x) = x^n$ \omterm{def:b1:continuity:continuous}{continu} en strikt stijgend
op $\intco{0}{+\infty}$, met $f(0) = 0$ en $f(x) \to +\infty$: haar beeld
is $\intco{0}{+\infty}$ (\cref{thm:b1:continuity:ivt} voor de
intervalstructuur). De stelling over de monotone bijectie levert dan in
één beweging een strikt stijgende \emph{\omterm{def:b1:continuity:continuous}{continue}} inverse
\[
x \mapsto x^{1/n} \colon \intco{0}{+\infty} \to
\intco{0}{+\infty} :
\]
bestaan, uniciteit en \omterm{def:b1:continuity:continuous}{continuïteit} van $n$-de wortels, zonder enige
berekening. Vergelijk met \cref{exo:b1:reals:12}, waar $\sqrt y$ met de
hand uit het \omterm{def:b1:reals:bounds}{supremum} gebouwd werd: één hoofdstuk theorie heeft die
bladzijde werk tot twee regels samengeperst, en dezelfde twee regels
wettigden daarvóór al $\arcsin$, $\arctan$ en
$\operatorname{arsinh}$. Het inzicht tot slot: een goede stelling is
opgeslagen arbeid.
\end{example}

\section{Uniforme continuïteit}

\begin{definition}\label{def:b1:continuity:uniform}
$f \colon I \to \R$ heet \emph{uniform continu}\index{uniforme
continuïteit} wanneer
\[
\forall \varepsilon > 0,\ \exists \delta > 0,\ \forall x, y \in I,
\qquad \abs{x - y} \leq \delta \implies \abs{f(x) - f(y)} \leq
\varepsilon .
\]
Het punt is dat $\delta$ alleen van $\varepsilon$ afhangt en niet van de
plaats in $I$. Uniforme \omterm{def:b1:continuity:continuous}{continuïteit} impliceert \omterm{def:b1:continuity:continuous}{continuïteit}; een
\emph{lipschitzfunctie} ($\abs{f(x) - f(y)} \leq k \abs{x - y}$) is
uniform \omterm{def:b1:continuity:continuous}{continu} ($\delta = \varepsilon/k$).
\end{definition}

\begin{example}\label{ex:b1:continuity:notuniform}
$x \mapsto x^2$ is \omterm{def:b1:continuity:continuous}{continu} op $\R$ maar \emph{niet} \omterm{def:b1:continuity:uniform}{uniform continu}:
$\abs{(n + \frac 1n)^2 - n^2} = 2 + \frac{1}{n^2} \geq 2$ terwijl de
argumenten op afstand $\frac 1n \to 0$ liggen. Op elk begrensd \omterm{prop:b1:reals:intervals}{interval}
is ze lipschitz en dus \omterm{def:b1:continuity:uniform}{uniform continu} --- in overeenstemming met de
stelling van Heine hieronder.
\end{example}

\begin{example}[Uniforme moduli, expliciet]\label{ex:b1:continuity:moduli}
Op een segment waarborgt Heine een uniforme $\delta$; vaak kun je die ook
\emph{berekenen}. Voor $f(x) = x^2$ op $\intcc{0}{10}$:
\[
\abs{x^2 - y^2} = \abs{x + y}\,\abs{x - y} \leq 20\,\abs{x - y},
\]
dus voldoet $\delta = \frac{\varepsilon}{20}$ uniform (een
lipschitzmodulus, lineair in $\varepsilon$). Voor $\sqrt x$ op
$\intcc{0}{1}$ is $\abs{\sqrt x - \sqrt y} \leq \sqrt{\abs{x - y}}$
(\cref{exo:b1:continuity:9}), zodat $\delta = \varepsilon^2$ voldoet ---
uniform maar \emph{niet} lineair: vlak bij $0$ is de vierkantswortel
steil, en die prijs komt tot uiting in de exponent van $\varepsilon$ en
niet in een falen van de uniformiteit. Het inzicht tot slot: \omterm{def:b1:continuity:uniform}{uniforme
continuïteit} is een spectrum en geen ja-of-nee --- de functie
$\delta(\varepsilon)$, de \omterm{def:b1:complex:field}{modulus} genoemd, meet hoe duur de uniformiteit
is, en lipschitz is eenvoudigweg haar beste cijfer.
\end{example}

\begin{theorem}[Heine]\label{thm:b1:continuity:heine}
Een \omterm{def:b1:continuity:continuous}{continue} functie op een \emph{segment} $\intcc{a}{b}$ is \omterm{def:b1:continuity:uniform}{uniform
continu}.\index{stelling van Heine}
\end{theorem}

\begin{proof}
Uit het ongerijmde: stel dat een zekere $\varepsilon_0 > 0$ elke $\delta$
verslaat. Kies met $\delta = \frac{1}{n+1}$ punten $x_n, y_n \in
\intcc{a}{b}$ met $\abs{x_n - y_n} \leq \frac{1}{n+1}$ en $\abs{f(x_n) -
f(y_n)} > \varepsilon_0$. De compactheid
(\cref{thm:b1:topology:compact}) extraheert $x_{\varphi(n)} \to c \in
\intcc{a}{b}$; dan gaat ook $y_{\varphi(n)} \to c$ (insluiting op $\abs{x
- y}$). De \omterm{def:b1:continuity:continuous}{continuïteit} in $c$ geeft $f(x_{\varphi(n)}) \to f(c)$ en
$f(y_{\varphi(n)}) \to f(c)$, dus $\abs{f(x_{\varphi(n)}) -
f(y_{\varphi(n)})} \to 0 < \varepsilon_0$: tegenspraak.
\end{proof}

\begin{example}[Begrensd, continu, en toch niet uniform]\label{ex:b1:continuity:sinx2}
De functie $f(x) = \sin(x^2)$ is \omterm{def:b1:continuity:continuous}{continu} en begrensd op $\R$, maar niet
\omterm{def:b1:continuity:uniform}{uniform continu}. Neem
\[
x_n = \sqrt{2\pi n}, \qquad y_n = \sqrt{2\pi n + \tfrac\pi2}:
\qquad
y_n - x_n = \frac{\pi/2}{\sqrt{2\pi n + \frac\pi2} +
\sqrt{2\pi n}} \longrightarrow 0 ,
\]
en toch is $f(y_n) - f(x_n) = \sin\bigl(2\pi n + \frac\pi2\bigr) -
\sin(2\pi n) = 1 - 0 = 1$ voor elke $n$: geen enkele $\delta$ kan
$\varepsilon = \frac12$ overal bedienen. Meetkundig \emph{versnellen} de
oscillaties van $\sin(x^2)$: de grafiek voltooit een volledige golf over
steeds kortere vensters, zodat de horizontale schaal die een gegeven
$\varepsilon$ vergt naar nul krimpt naarmate $x$ groeit. Het inzicht tot
slot: begrensdheid koopt geen uniformiteit (dit voorbeeld), en
onbegrensdheid sluit haar niet uit ($\sqrt x$,
\cref{exo:b1:continuity:9}); beslissend is de \emph{oscillatiemodulus},
en de stelling van Heine zegt dat compacte domeinen die automatisch
disciplineren.
\end{example}

\begin{example}[Het rekenmachine-experiment, verklaard]\label{ex:b1:continuity:cositeration}
Tik een willekeurig getal in een rekenmachine en druk herhaaldelijk op
$\cos$: het scherm klemt vast op $0.7390851\dots$ Waarom? Na één druk
ligt de waarde in $\intcc{-1}{1}$, na twee in $\intcc{\cos 1}{1}
\subseteq \intcc{0.54}{1}$, een stabiel \omterm{prop:b1:reals:intervals}{interval} voor $\cos$. Daarop is
$\abs{\cos a - \cos b} \leq \sin(1)\,\abs{a - b}$ met $\sin 1 =
0.841\dots < 1$ (de afschatting van product naar som uit
\cref{pb:b1:seq:1}, of de middelwaardeongelijkheid van
\cref{ch:b1:derivative}): de iteratie is contraherend, zodat de stap voor
foutbeheersing uit \cref{met:b1:seq:recurrent} geeft
\[
\abs{u_n - c} \leq (0.842)^{\,n-2}\,\abs{u_2 - c}
\longrightarrow 0 ,
\]
waarbij $c$ het unieke vaste punt met $\cos c = c$ is
(\cref{exo:b1:continuity:6}). Ongeveer $40$ drukken kopen drie decimalen
($0.842^{40} \approx 10^{-3}$) --- een meetkundig tempo, milder dan de
$2^{-n}$ per stap van de dichotomie, maar elke druk kost één
toetsaanslag terwijl elke dichotomiestap een volledige tekenevaluatie
kost. Het inzicht tot slot: het beeld van het vaste punt uit
\cref{ch:b1:seq} en de bestaansstellingen van dit hoofdstuk zijn twee
helften van één verhaal --- de tussenwaardestelling vindt $c$, de
contractie bereikt hem.
\end{example}

\begin{remark}[Waar deze stellingen hierna werken]
Elk van de drie pijlers van dit hoofdstuk drijft een later hoofdstuk aan.
De tussenwaardestelling voedt elk argument voor het bestaan van
oplossingen en de stelling over de monotone bijectie; de extremumstelling
maakt van optimalisatieproblemen stellingen (Rolle en de
middelwaardestelling in \cref{ch:b1:derivative} beginnen precies daar);
en de stelling van Heine is de reden waarom \omterm{def:b1:continuity:continuous}{continue} functies op
segmenten in \cref{ch:b1:integration} geïntegreerd kunnen worden --- de
uniforme $\delta$ is wat de Riemannsommen laat convergeren. In het volume
van bachelorjaar 2 keert hetzelfde drietal terug in genormeerde
vectorruimten, waar de compactheid het werk doet dat segmenten hier doen.
\end{remark}

\begin{remark}[Perspectieven binnen dit volume]
De \omterm{def:b1:continuity:continuous}{continuïteit} wordt weldra overtroffen maar nooit met pensioen
gestuurd. \Cref{ch:b1:derivative} versterkt haar tot afleidbaarheid en
bewijst de wederdienst (afleidbaar impliceert \omterm{def:b1:continuity:continuous}{continu});
\cref{ch:b1:integration} rust er tweemaal op, via Heine voor de
constructie en via de hoofdstelling, waarvan het centrale object $x
\mapsto \int_a^x f$ een louter \omterm{def:b1:continuity:continuous}{continue} $f$ tot een $C^1$-primitieve
opwaardeert. In \cref{ch:b1:multivar} houdt de \omterm{def:b1:continuity:continuous}{continuïteit} in twee
variabelen een valstrik in petto die het voorproeven waard is: de functie
$\frac{xy}{x^2 + y^2}$ (uitgebreid met $0$) is \omterm{def:b1:continuity:continuous}{continu} in $x$ voor elke
vaste $y$ en in $y$ voor elke vaste $x$, en toch niet \omterm{def:b1:continuity:continuous}{continu} in de
oorsprong --- langs de diagonaal $x = y$ is ze constant $\frac12$.
Afzonderlijke \omterm{def:b1:continuity:continuous}{continuïteit} is strikt zwakker dan \omterm{def:b1:continuity:continuous}{continuïteit}: de
karakterisering met rijen overleeft de overgang naar $\R^2$, maar de
rijen moeten uit \emph{elke} richting mogen naderen, niet alleen langs de
assen.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:continuity:1}
Bewijs met de karakterisering met rijen dat $x \mapsto \sin\frac 1x$ geen
limiet heeft in $0^+$ \emph{(wijs twee rijen aan)}. Heeft $x \mapsto x
\sin\frac 1x$ er wel een?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:continuity:2}
Bestudeer de \omterm{def:b1:continuity:continuous}{continuïteit} op $\R$ van $f(x) = \lfloor x \rfloor$, van
$g(x) = x - \lfloor x \rfloor$ en van $h(x) = \lfloor x \rfloor + (x -
\lfloor x\rfloor)^2$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:continuity:3}
Bewijs dat de vergelijking $x^5 - 3x + 1 = 0$ minstens drie reële
oplossingen heeft \emph{(evalueer in goedgekozen punten en pas
\cref{thm:b1:continuity:ivt} toe op drie disjuncte segmenten)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:continuity:4}
Bewijs dat elke \omterm{def:b1:poly:def}{veelterm} van oneven graad een reële wortel heeft.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:5}
(Vast punt) Zij $f \colon \intcc{0}{1} \to \intcc{0}{1}$ \omterm{def:b1:continuity:continuous}{continu}. Bewijs
dat $f$ een vast punt heeft: $f(c) = c$ voor een zekere $c$. Illustreer
dat noch de \omterm{def:b1:continuity:continuous}{continuïteit} noch het segment weggelaten mag worden.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:6}
Bewijs dat de vergelijking $\cos x = x$ precies één reële oplossing
heeft, en dat die in $\intoo{0}{1}$ ligt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:7}
Zij $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{continu} met $f(x) \to +\infty$ voor $x \to
\pm\infty$. Bewijs dat $f$ een globaal minimum op $\R$ bereikt.
\emph{(Herleid tot een segment dat een niveauverzameling bevat.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:8}
Zij $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{continu} en periodiek (met periode $T > 0$).
Bewijs dat $f$ begrensd is en haar grenzen bereikt, en dat er een $c$
bestaat met $f(c + \frac T2) = f(c)$. \emph{(Bestudeer voor het tweede
punt $g(x) = f(x + \frac T2) - f(x)$ over één periode.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:9}
Bewijs dat $x \mapsto \sqrt x$ \omterm{def:b1:continuity:uniform}{uniform continu} is op $\intco{0}{+\infty}$,
hoewel ze vlak bij $0$ niet lipschitz is. \emph{(Bewijs en gebruik
$\abs{\sqrt x - \sqrt y} \leq \sqrt{\abs{x - y}}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:continuity:10}
Zij $f$ \omterm{def:b1:continuity:continuous}{continu} en \omterm{def:b1:logic:inj}{injectief} op een \omterm{prop:b1:reals:intervals}{interval} $I$. Bewijs dat $f$ strikt
monotoon is. \emph{Aanwijzing: zo niet, dan zijn er $a < b < c$ met
bijvoorbeeld $f(b) > f(a)$ en $f(b) > f(c)$; pas de tussenwaardestelling
aan weerszijden van $b$ toe op een waarde tussen $\max(f(a), f(c))$ en
$f(b)$.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:continuity:11}
(Functievergelijking van Cauchy, \omterm{def:b1:continuity:continuous}{continue} geval) Zij $f \colon \R \to \R$
\omterm{def:b1:continuity:continuous}{continu} met $f(x + y) = f(x) + f(y)$ voor alle $x, y$. Bewijs dat $f(x) =
f(1)\,x$ voor alle $x$: eerst op $\N$, $\Z$ en $\Q$ (met de additiviteit
alleen), en daarna op $\R$ met de \omterm{def:b1:continuity:continuous}{continuïteit} en de \omterm{def:b1:topology:dense}{dichtheid}
(\cref{thm:b1:reals:density}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:continuity:12}
Zij $f \colon \intco{0}{+\infty} \to \R$ \omterm{def:b1:continuity:continuous}{continu} met $f(x) \to \ell \in
\R$ voor $x \to +\infty$. Bewijs dat $f$ \emph{uniform} \omterm{def:b1:continuity:continuous}{continu} is op
$\intco{0}{+\infty}$. \emph{(Knip bij een grote $A$: Heine op
$\intcc{0}{A+1}$, de limiet voorbij $A$; laat de twee regimes
overlappen.)}
\end{exercise}

\section{Opgave: de functievergelijking van Cauchy en haar zusters}

\begin{problem}[{Weekendopgave --- $f(x + y) = f(x) + f(y)$: regelmaat
dwingt lineariteit af, en het portret van de monsters}]
\label{pb:b1:continuity:1}
Welke functies voldoen aan $f(x + y) = f(x) + f(y)$ voor alle reële $x,
y$? Cauchy stelde de vraag in 1821; het antwoord is een paradigma.
\Cref{exo:b1:continuity:11} toont aan dat zo'n \emph{additieve} functie
lineair is op $\Q$ en dat volledige \omterm{def:b1:continuity:continuous}{continuïteit} $f(x) = cx$ afdwingt.
Deze opgave verzwakt de hypothese drastisch --- \omterm{def:b1:continuity:continuous}{continuïteit} in één
\emph{enkel} punt, of monotonie, of louter begrensdheid op één klein
\omterm{prop:b1:reals:intervals}{interval} volstaat telkens --- schildert daarna het portret van een
hypothetische niet-lineaire oplossing (haar grafiek vult het vlak), lost
de zustervergelijkingen op die $\eu^{cx}$, $c\ln x$, $x^c$ en $cx^2$
karakteriseren, en sluit af met de vergelijking van Jensen en de stelling
\emph{middelpuntconvex $+$ \omterm{def:b1:continuity:continuous}{continu} $\implies$
convex}.\index{functievergelijking van Cauchy}\index{functievergelijking}
Overal betekent \emph{additief}: $f(x + y) = f(x) + f(y)$ voor alle $x, y
\in \R$.

\medskip
\noindent\textbf{Deel I --- $\Q$-lineariteit, en één punt van
\omterm{def:b1:continuity:continuous}{continuïteit}.}
\begin{enumerate}
  \item Zij $f$ additief. Volgens \cref{exo:b1:continuity:11} is $f(r) =
        f(1)\,r$ voor rationale $r$. Bewijs de fijnere \omterm{def:b1:logic:statement}{uitspraak} die
        hieronder gebruikt wordt: voor \emph{elke} $x \in \R$ en $r \in
        \Q$ is $f(rx) = r\,f(x)$ ($f$ is $\Q$-\emph{lineair}).
  \item Stel dat de additieve $f$ \omterm{def:b1:continuity:continuous}{continu} is in één enkel punt $x_0$.
        Toon aan dat $f$ overal \omterm{def:b1:continuity:continuous}{continu} is \emph{(bereken $f(x + h) -
        f(x)$ met behulp van $f(x_0 + h) - f(x_0)$)}, en dus $f(x) =
        f(1)\,x$.
  \item Toon aan dat een additieve functie vastligt door haar beperking
        tot elke \omterm{def:b1:topology:dense}{dichte} \omterm{def:b1:structures:subgroup}{deelgroep}: stemmen twee additieve functies
        overeen op $\Z + \sqrt2\,\Z$ en zijn beide \omterm{def:b1:continuity:continuous}{continu}, dan zijn ze
        gelijk --- terwijl het zonder \omterm{def:b1:continuity:continuous}{continuïteit} met de
        $\Q$-lineariteit op de \omterm{def:b1:structures:subgroup}{deelgroep} verenigbaar is om $f(1) = 0$ en
        $f(\sqrt 2) = 1$ voor te schrijven. Bereken $f(m + n\sqrt2)$
        voor die voorschrift.
  \item Zij $f$ additief en naar boven begrensd door $M$ op een zeker
        \omterm{prop:b1:reals:intervals}{interval} $\intcc{a}{b}$ met $a < b$. Toon aan dat $f$ naar boven
        begrensd is op $\intcc{0}{\ell}$ met $\ell = b - a$
        \emph{(verschuif over $a$)}.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De ladder van de regelmaat.}
\begin{enumerate}[resume]
  \item In het verlengde van vraag 4: toon met $f(\ell - t) + f(t) =
        f(\ell)$ aan dat $f$ daar ook naar \emph{onderen} begrensd is:
        $\abs f \leq C$ op $\intcc{0}{\ell}$.
  \item Toon aan dat $\abs{f(t)} \leq \frac{C}{n}$ voor $t \in
        \intcc{0}{\ell/n}$, en leid af dat $f$ \omterm{def:b1:continuity:continuous}{continu} is in $0$ \emph{(de
        oneven pariteit handelt de linkerkant af)}, en dus overal (vraag
        2): een additieve functie die op één \omterm{prop:b1:reals:intervals}{interval} begrensd is, is
        lineair.
  \item Leid het monotone geval af: een additieve functie die op een
        zeker $\intcc{a}{b}$ ($a < b$) niet-dalend is, is $f(x) = cx$
        met $c \geq 0$.
  \item Stel de \emph{ladder van de regelmaat} samen: voor een additieve
        $f$ zijn de volgende \omterm{def:b1:logic:statement}{uitspraken} equivalent --- (a) $f(x) = cx$;
        (b) $f$ is \omterm{def:b1:continuity:continuous}{continu}; (c) $f$ is \omterm{def:b1:continuity:continuous}{continu} in één punt; (d) $f$ is
        monotoon op een zeker niet-ontaard \omterm{prop:b1:reals:intervals}{interval}; (e) $f$ is begrensd
        op een zeker niet-ontaard \omterm{prop:b1:reals:intervals}{interval}. Rangschik de implicaties zó
        dat elke ofwel triviaal ofwel al bewezen is.
  \item Ga na dat de vragen 4--6 alleen een grens \emph{naar boven}
        verbruikten: een additieve functie die op één niet-ontaard
        \omterm{prop:b1:reals:intervals}{interval} naar boven begrensd is, is al lineair. Leid de
        spiegeluitspraak voor een grens naar onderen af, en noteer de
        sterkste vorm van sport (e) van de ladder die je zo krijgt.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Portret van een monster.} Neem nu aan dat
$f$ additief maar \emph{niet} lineair is.
\begin{enumerate}[resume]
  \item Toon aan dat er reële getallen $u, v$ ongelijk aan nul zijn met
        $\dfrac{f(u)}{u} \neq \dfrac{f(v)}{v}$, en dat de vectoren $(u,
        f(u))$ en $(v, f(v))$ het vlak opspannen (hun determinant $u
        f(v) - v f(u)$ is niet nul).
  \item Toon aan dat de grafiek van $f$ alle punten
        \[
        r\,(u, f(u)) + s\,(v, f(v)), \qquad r, s \in \Q
        \]
        bevat, en leid af dat de grafiek \emph{\omterm{def:b1:topology:dense}{dicht} ligt in $\R^2$}:
        voor elk punt $(x_0, y_0)$ van het vlak en elke $\varepsilon >
        0$ ligt een zeker $(x, f(x))$ er binnen $\varepsilon$ van
        \emph{(los het reële $2\times2$-stelsel op en benader daarna de
        reële coëfficiënten door rationale)}.
  \item Leid uit vraag 11 het volledige portret af: een niet-lineaire
        additieve functie is op elk niet-ontaard \omterm{prop:b1:reals:intervals}{interval} onbegrensd, in
        elk punt discontinu, op geen enkel \omterm{prop:b1:reals:intervals}{interval} monotoon, en haar
        beeld van elk \omterm{prop:b1:reals:intervals}{interval} ligt \omterm{def:b1:topology:dense}{dicht} in $\R$. Verzoen dat met vraag
        8.
  \item Monsters bestaan --- op een \omterm{def:b1:topology:dense}{dichte} \omterm{def:b1:structures:subgroup}{deelgroep}, constructief:
        definieer op $G = \Z + \sqrt2\,\Z$ de functie $f(m + n\sqrt2) =
        n$. Toon aan dat $f$ welgedefinieerd en additief is op $G$, en
        dat $f$ onbegrensd is op $G \cap \intoo{0}{\varepsilon}$ voor
        elke $\varepsilon > 0$ \emph{(bij vaste $N$ hebben maar eindig
        veel $g = m + n\sqrt2 \in \intoo{0}{1}$ een $\abs n \leq N$;
        terwijl $G \cap \intoo{0}{\varepsilon}$ oneindig is)}. Leg in
        één alinea uit waarom zo'n $f$ tot heel $\R$ uitbreiden een
        basis van $\R$ als $\Q$-vectorruimte vergt (een
        \emph{hamelbasis}), waarvan het bestaan een kwestie van het
        keuzeaxioma is, buiten dit volume.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De zustervergelijkingen.} Alle functies hier
zijn \omterm{def:b1:continuity:continuous}{continu}.
\begin{enumerate}[resume]
  \item Zij $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{continu} en niet identiek $0$, met $f(x
        + y) = f(x)f(y)$. Toon aan dat $f(x) = f\bigl(\frac
        x2\bigr)^2 \geq 0$, dan dat $f > 0$ overal, en dan dat $f(x) =
        \eu^{cx}$ voor een zekere $c$: de exponentiëlen zijn precies de
        \omterm{def:b1:continuity:continuous}{continue} \omterm{def:b1:structures:morphism}{morfismen} van $(\R, +)$ naar $(\R^*, \times)$.
  \item Zij $f \colon \intoo{0}{+\infty} \to \R$ \omterm{def:b1:continuity:continuous}{continu} met $f(xy) =
        f(x) + f(y)$. Toon aan dat $f(x) = c\ln x$ \emph{(vervoer via
        $\exp$)}.
  \item Zij $f \colon \intoo{0}{+\infty} \to \intoo{0}{+\infty}$ \omterm{def:b1:continuity:continuous}{continu}
        met $f(xy) = f(x)f(y)$. Toon aan dat $f(x) = x^c$.
  \item Bepaal alle \omterm{def:b1:continuity:continuous}{continue} $f \colon \R \to \R$ met
        \[
        f(x + y) = f(x) + f(y) + f(x)f(y) .
        \]
        \emph{(Bestudeer $h = 1 + f$; behandel het ontaarde geval
        apart.)}
  \item (Parallellogram) Bepaal alle \omterm{def:b1:continuity:continuous}{continue} $f \colon \R \to \R$ met
        $f(x + y) + f(x - y) = 2f(x) + 2f(y)$: toon aan dat $f$ even is,
        dat $f(0) = 0$, dat $f(nx) = n^2 f(x)$ met inductie, en dan dat
        $f(x) = f(1)\,x^2$. (Deze vergelijking is de vingerafdruk van
        kwadratische vormen --- de parallellogramwet die in het volume
        van bachelorjaar 2 herkent welke normen uit een \omterm{def:b1:topology:closure}{inwendig} product
        komen.)
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Jensen en middelpuntconvexiteit.}
\begin{enumerate}[resume]
  \item (Vergelijking van Jensen) Zij $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{continu} met
        $f\bigl(\frac{x+y}{2}\bigr) = \frac{f(x) + f(y)}{2}$. Toon aan
        dat $g = f - f(0)$ voldoet aan $g\bigl(\frac x2\bigr) =
        \frac{g(x)}{2}$, leid af dat $g$ additief is, en besluit $f(x) =
        cx + d$.
  \item Neem nu alleen de \emph{ongelijkheid} aan: $f$ \omterm{def:b1:continuity:continuous}{continu} met
        \[
        f\Bigl(\frac{x + y}{2}\Bigr) \leq \frac{f(x) +
        f(y)}{2} \qquad (x, y \in \R).
        \]
        Bewijs met inductie naar $n$ dat voor alle dyadische gewichten
        $\lambda = \frac{k}{2^n} \in \intcc{0}{1}$ geldt
        \[
        f\bigl(\lambda x + (1 - \lambda)y\bigr) \leq \lambda
        f(x) + (1 - \lambda) f(y) .
        \]
  \item Breid met de \omterm{def:b1:continuity:continuous}{continuïteit} en de \omterm{def:b1:topology:dense}{dichtheid} van de dyadische
        getallen (\cref{exo:b1:reals:8}) uit tot elke $\lambda \in
        \intcc{0}{1}$: een \omterm{def:b1:continuity:continuous}{continue} middelpuntconvexe functie voldoet
        aan de volledige convexiteitsongelijkheid (het begrip dat in
        \cref{ch:b1:derivative} systematisch bestudeerd wordt).
  \item Toon aan dat de \omterm{def:b1:continuity:continuous}{continuïteit} niet weggelaten mag worden: een
        niet-lineaire additieve $f$ voldoet aan de
        \emph{middelpuntgelijkheid} van vraag 19 en toch aan geen enkele
        convexiteitsongelijkheid op enig \omterm{prop:b1:reals:intervals}{interval} (vraag 12). Moraal:
        middelpuntconvexiteit is een eigenschap in aftelbaar veel stadia
        (de dyadische getallen), convexiteit een eigenschap van het
        continuüm; de \omterm{def:b1:continuity:continuous}{continuïteit} is de brug --- precies zoals in de
        Delen I--II.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Laatste variaties en synthese.}
\begin{enumerate}[resume]
  \item Bepaal alle \omterm{def:b1:continuity:continuous}{continue} $f \colon \R \to \R$ met $f(x + y) = f(x) +
        f(y) + xy$ \emph{(trek de particuliere oplossing
        $\frac{x^2}{2}$ af)}.
  \item Bewijs: is $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{continu} en slechts additief op
        een \emph{\omterm{def:b1:topology:dense}{dichte} \omterm{def:b1:structures:subgroup}{deelgroep}} $G$ (dat wil zeggen $f(g + g') =
        f(g) + f(g')$ voor $g, g' \in G$), dan is $f$ additief op $\R$.
        Algemener: twee \omterm{def:b1:continuity:continuous}{continue} functies die op een \omterm{def:b1:topology:dense}{dichte}
        deelverzameling van $\R$ overeenstemmen zijn gelijk.
  \item Synthese, één zin per onderdeel: (i) formuleer de ladder van de
        regelmaat uit vraag 8 uit het hoofd; (ii) leg uit waarom
        ``grafiek \omterm{def:b1:topology:dense}{dicht} in het vlak'' het juiste mentale beeld is voor
        het falen van de regelmaat; (iii) som de vijf klassieke functies
        op die in de Delen IV--V gekarakteriseerd worden, en de ene
        methode die ze alle ving; (iv) noem de twee plaatsen waar de
        \omterm{def:b1:topology:dense}{dichtheid} van $\Q$ (of van de dyadische getallen) in $\R$ het
        argument droeg, en de plaats waar ze dat niet kon (vraag 13).
\end{enumerate}
\end{problem}
