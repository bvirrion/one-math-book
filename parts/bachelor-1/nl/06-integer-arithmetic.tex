\chapter{Getaltheorie in $\Z$}\label{ch:b1:arith}

De getaltheorie --- de studie van de \omterm{def:b1:arith:divides}{deelbaarheid} in $\Z$ --- werd in het
bovenbouwvolume aangevat. Dit hoofdstuk bouwt haar volledig opnieuw op
vanaf de euclidische deling, met volledige bewijzen: \omterm{thm:b1:arith:gcd}{grootste gemene
deler} en het \omterm{met:b1:arith:euclid}{algoritme van Euclides}, de identiteit van Bézout en het lemma
van Gauss, de \omterm{thm:b1:arith:fta}{priemontbinding}, en het rekenen met \omterm{def:b1:arith:congruence}{congruenties} tot en met
de kleine stelling van Fermat. Naast haar eigen bekoring is deze stof het
model dat \cref{ch:b1:poly} voor veeltermen nabootst.

\section{Deelbaarheid en euclidische deling}

\begin{definition}[Deelbaarheid]\label{def:b1:arith:divides}
Voor $a, b \in \Z$ zeggen we dat $b$ het getal $a$ \emph{deelt}
(genoteerd $b \mid a$)\index{deelbaarheid} wanneer $a = bq$ voor een
zekere $q \in \Z$. Eenvoudige gevolgen: als $b \mid a$ en $b \mid a'$,
dan $b \mid (ua + va')$ voor alle $u, v \in \Z$; als $b \mid a$ en $a
\neq 0$, dan $\abs b \leq \abs a$; en uit $a \mid b$ samen met $b \mid a$
volgt $b = \pm a$.
\end{definition}

\begin{theorem}[Euclidische deling]\label{thm:b1:arith:division}
Voor alle $a \in \Z$ en $b \in \N^*$ bestaat er precies één paar $(q, r)
\in \Z \times \N$ met
\[
a = bq + r, \qquad 0 \leq r < b .
\]
\end{theorem}

\begin{proof}
\emph{Bestaan.} De \omterm{def:b1:logic:sets}{verzameling} $A = \{a - bk : k \in \Z\} \cap \N$ is een
niet-lege deelverzameling van $\N$ (neem $k = -\abs a$: $a + b\abs a \geq
a + \abs a \geq 0$). Zij $r = a - bq$ haar kleinste element. Was $r \geq
b$, dan zou $r - b = a - b(q+1)$ een kleiner element van $A$ zijn:
tegenspraak. Dus $0 \leq r < b$.

\emph{Uniciteit.} Is $bq + r = bq' + r'$ met $0 \leq r, r' < b$, dan is
$b(q - q') = r' - r$ met $\abs{r' - r} < b$: het veelvoud van $b$ in het
linkerlid moet $0$ zijn, dus $q = q'$ en $r = r'$.
\end{proof}

\begin{example}[Positiestelsels door herhaald delen]\label{ex:b1:arith:base7}
Schrijf $2026$ in grondtal $7$. Deel herhaaldelijk door $7$ en houd de
resten bij:
\[
2026 = 7 \times 289 + 3, \quad
289 = 7 \times 41 + 2, \quad
41 = 7 \times 5 + 6, \quad
5 = 7 \times 0 + 5 .
\]
Lees je de resten van laatst naar eerst, dan is $2026 = (5\,6\,2\,3)_7$.
Controle: $5 \times 343 + 6 \times 49 + 2 \times 7 + 3 = 1715 + 294 + 14
+ 3 = 2026$. De uniciteit van de euclidische deling is precies wat elk
cijfer \emph{afdwingt}: bij elke stap is de rest het enige gehele getal
in $\intint06$ dat congruent is met de huidige waarde modulo $7$, zodat
de schrijfwijze in grondtal $7$ uniek is --- het feit dat stilzwijgend
gebruikt wordt telkens als de weekendopgave met ``de cijfers van $n$ in
grondtal $p$'' werkt.
\end{example}

\section{Grootste gemene deler}

\begin{theorem}[Deelgroepen van $\Z$; bestaan van de ggd]\label{thm:b1:arith:gcd}
\begin{enumerate}
  \item Elke deelgroep van $(\Z, +)$ is van de vorm $n\Z = \{nk : k \in
        \Z\}$ voor een unieke $n \in \N$.
  \item Voor $a, b \in \Z$ die niet beide nul zijn, is de \omterm{def:b1:logic:sets}{verzameling}
        $a\Z + b\Z = \{au + bv : u, v \in \Z\}$ een deelgroep van $\Z$
        en dus gelijk aan $d\,\Z$ voor een unieke $d \in \N^*$. Die $d$
        is de \emph{grootste gemene deler}\index{grootste gemene deler}
        $\gcd(a, b)$: hij \omterm{def:b1:arith:divides}{deelt} $a$ en $b$, en elke gemene deler van $a$
        en $b$ \omterm{def:b1:arith:divides}{deelt} $d$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Zij $H \subseteq \Z$ een deelgroep (niet-leeg, stabiel onder
aftrekken; de formele definitie staat in \cref{ch:b1:structures}, en
alleen die twee eigenschappen worden gebruikt). Is $H = \{0\}$, neem dan
$n = 0$. Anders bevat $H$ een element ongelijk aan nul en zijn
tegengestelde, en dus een kleinste strikt positief element $n$. Dan is
$n\Z \subseteq H$. Voor $x \in H$ schrijf je $x = nq + r$ met $0 \leq r <
n$ (\cref{thm:b1:arith:division}); dan is $r = x - nq \in H$, en de
minimaliteit van $n$ dwingt $r = 0$ af: $x \in n\Z$. Uniciteit: $n$ is
het kleinste positieve element van $n\Z$.

(2) $a\Z + b\Z$ bevat $0$ en is stabiel onder aftrekken, dus is het $d\Z$
met $d \geq 1$ (het bevat $a$ of $b$, en een van beide is niet nul). Omdat
$a, b \in d\Z$, \omterm{def:b1:arith:divides}{deelt} $d$ ze beide. En \omterm{def:b1:arith:divides}{deelt} $c$ zowel $a$ als $b$, dan
\omterm{def:b1:arith:divides}{deelt} $c$ elke $au + bv$ --- in het bijzonder $c \mid d$, want $d \in a\Z
+ b\Z$. Dat is de aangekondigde eigenschap (en ze impliceert $\abs c \leq
d$, zodat $d$ de naam \emph{grootste} gemene deler verdient).
\end{proof}

\begin{corollary}[Identiteit van Bézout]\label{cor:b1:arith:bezout}
Voor $a, b$ die niet beide nul zijn bestaan er $u, v \in \Z$ met
\[
au + bv = \gcd(a, b) .
\]
In het bijzonder ($\gcd(a,b) = 1$, het \emph{relatief
prieme}\index{relatief priem} geval): $a$ en $b$ zijn \omterm{cor:b1:arith:bezout}{relatief priem} dan
en slechts dan als $au + bv = 1$ een oplossing heeft.
\end{corollary}

\begin{proof}
Er is $\gcd(a,b) = d \in d\Z = a\Z + b\Z$. Voor de equivalentie: is
$\gcd(a,b) = 1$, dan levert Bézout de oplossing; omgekeerd dwingt $au +
bv = 1$ af dat elke gemene deler van $a$ en $b$ het getal $1$ \omterm{def:b1:arith:divides}{deelt}.
\end{proof}

\begin{method}[Algoritme van Euclides, uitgebreid]\label{met:b1:arith:euclid}
Om $\gcd(a, b)$ te berekenen ($a > b > 0$): deel $a = bq + r$; dan is
$\gcd(a, b) = \gcd(b, r)$ (de gemene delers van $(a,b)$ en van $(b,r)$
vallen samen, want $r = a - bq$); herhaal tot de rest $0$ is; de laatste
rest ongelijk aan nul is de ggd\index{algoritme van Euclides}. De
delingen achterstevoren doorlopen (of de coëfficiënten onderweg
bijhouden) levert een paar van Bézout $(u, v)$.
\end{method}

\begin{example}\label{ex:b1:arith:euclid}
$\gcd(120, 23)$: $120 = 5 \times 23 + 5$; $23 = 4 \times 5 + 3$; $5 =
1\times 3 + 2$; $3 = 1 \times 2 + 1$; $2 = 2 \times 1 + 0$. Dus $\gcd =
1$. Achterstevoren:
\begin{align*}
1 &= 3 - 2 = 3 - (5 - 3) = 2\times 3 - 5 = 2(23 - 4\times 5) - 5 \\
&= 2 \times 23 - 9 \times 5 = 2\times 23 - 9(120 - 5\times 23)
= 47 \times 23 - 9 \times 120 .
\end{align*}
Controle: $47 \times 23 = 1081$ en $9 \times 120 = 1080$.
\end{example}

\begin{theorem}[Lemma van Gauss en gevolgen]\label{thm:b1:arith:gauss}
Zij $a, b, c \in \Z$.
\begin{enumerate}
  \item (Lemma van Gauss\index{lemma van Gauss}) Is $a \mid bc$ en
        $\gcd(a, b) = 1$, dan $a \mid c$.
  \item Is $a \mid c$, $b \mid c$ en $\gcd(a,b) = 1$, dan $ab \mid c$.
  \item Is $\gcd(a, b) = \gcd(a, c) = 1$, dan $\gcd(a, bc) = 1$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Bézout: $au + bv = 1$. Vermenigvuldig met $c$: $acu + bcv = c$. Beide
termen zijn \omterm{def:b1:arith:divides}{deelbaar} door $a$ (de tweede omdat $a \mid bc$), dus $a \mid
c$.

(2) Schrijf $c = aq$; uit $b \mid aq$ en $\gcd(a, b) = 1$ geeft punt (1)
dat $b \mid q$, dus $ab \mid aq = c$.

(3) Er is $au + bv = 1$ en $au' + cv' = 1$. Vermenigvuldig beide
betrekkingen:
\[
1 = (au + bv)(au' + cv')
= a\,\bigl(auu' + ucv' + u'bv\bigr) + bc\,(vv') ,
\]
een betrekking van Bézout tussen $a$ en $bc$: volgens
\cref{cor:b1:arith:bezout} is $\gcd(a, bc) = 1$.
\end{proof}

\begin{example}[Een lineaire diofantische vergelijking oplossen]\label{ex:b1:arith:diophantine}
Bepaal alle $(x, y) \in \Z^2$ met $6x + 10y = 4$. Eerst de
\emph{bestaanstoets}: $\gcd(6, 10) = 2$ \omterm{def:b1:arith:divides}{deelt} $4$, dus zijn er
oplossingen (deelde de ggd het rechterlid niet, dan zou het linkerlid
altijd een veelvoud ervan zijn en zou er geen enkele zijn). Deel door:
$3x + 5y = 2$. Een particuliere oplossing springt in het oog: $(x_0, y_0)
= (-1, 1)$. Voor de algemene trek je af: $3(x + 1) = -5(y - 1)$, dus $3
\mid 5(y-1)$, en het lemma van Gauss ($\gcd(3,5) = 1$) geeft $3 \mid y -
1$: $y = 1 - 3k$, waarna $x = -1 + 5k$. Omgekeerd voldoet elk zo'n paar:
\[
(x, y) = (-1 + 5k,\ 1 - 3k), \qquad k \in \Z .
\]
Het patroon is algemeen: één particuliere oplossing plus de geheeltallige
veelvouden van $\bigl(\frac b{\gcd}, -\frac a{\gcd}\bigr)$ --- dezelfde
structuur ``particulier plus homogeen'' als in \cref{ch:b1:diffeq}, met
het lemma van Gauss in de rol van de uniciteit.
\end{example}

\begin{definition}[Kleinste gemene veelvoud]\label{def:b1:arith:lcm}
$\operatorname{lcm}(a, b)$\index{kleinste gemene veelvoud} is de
voortbrenger in $\N$ van de deelgroep $a\Z \cap b\Z$: het is een gemeen
veelvoud van $a$ en $b$ dat elk gemeen veelvoud \omterm{def:b1:arith:divides}{deelt}, en voor $a, b \in
\N^*$ geldt
\[
\gcd(a,b) \times \operatorname{lcm}(a,b) = ab
\qquad (\text{bewijs in } \cref{exo:b1:arith:5}).
\]
\end{definition}

\begin{example}[Samenloopvragen zijn kgv-vragen]\label{ex:b1:arith:gears}
Twee in elkaar grijpende tandwielen hebben $84$ en $36$ tanden. Na
hoeveel tanden gemeenschappelijke beweging keren ze samen naar hun
beginstand terug? De configuratie herhaalt zich wanneer het aantal
verstreken tanden een gemeen veelvoud van $84$ en $36$ is; de eerste keer
is dat na
\[
\operatorname{lcm}(84, 36) = \frac{84 \times 36}{\gcd(84, 36)}
= \frac{3024}{12} = 252
\]
tanden --- dat wil zeggen $3$ omwentelingen van het grote en $7$ van het
kleine tandwiel ($252/84$ en $252/36$). Let op de praktische weg:
\emph{bereken eerst de ggd} (Euclides: $84 = 2\times36 + 12$, $36 =
3\times12$) en deel dan --- bouw het kgv nooit door veelvouden op te
sommen. Elke vraag over periodieke samenloop (tandwielen, standen van
planeten, samenvallende repeterende breuken) herleidt zich tot deze ene
berekening.
\end{example}

\section{Priemgetallen}

\begin{definition}\label{def:b1:arith:prime}
Een geheel getal $p \geq 2$ heet \emph{priem}\index{priemgetal} wanneer
zijn enige positieve delers $1$ en $p$ zijn. Voor $p$ priem en $a \in \Z$
geldt: ofwel $p \mid a$, ofwel $\gcd(p, a) = 1$. Bijgevolg
(\cref{thm:b1:arith:gauss}) geldt het \emph{lemma van Euclides}: is $p
\mid ab$, dan $p \mid a$ of $p \mid b$.
\end{definition}

\begin{remark}[Primaliteit toetsen met proefdelingen]\label{rem:b1:arith:trialdivision}
Is $n = ab$ met $2 \leq a \leq b$, dan is $a^2 \leq ab = n$, dus $a
\leq \sqrt n$: een samengesteld getal $n$ heeft altijd een \omterm{def:b1:arith:prime}{priemdeler}
$\leq \sqrt n$. Om na te gaan of $n$ \omterm{def:b1:arith:prime}{priem} is volstaat het dus de
\omterm{def:b1:arith:prime}{priemgetallen} tot $\sqrt n$ te proberen. Voor $n = 271$: $\sqrt{271} <
17$, en $271$ is door geen van $2, 3, 5, 7, 11, 13$ \omterm{def:b1:arith:divides}{deelbaar} (oneven,
cijfersom $10$, eindigt niet op $0$ of $5$, en $271 = 7\cdot38 + 5 =
11\cdot24 + 7 = 13\cdot20 + 11$): \omterm{def:b1:arith:prime}{priem}, na zes delingen in plaats van
tweehonderd. De grens $\sqrt n$ is een echte drempel: haar efficiënt
overschrijden voor getallen van honderd cijfers vergt de moderne
primaliteitstests die uit \cref{thm:b1:arith:fermat} gegroeid zijn.
\end{remark}

\begin{theorem}[Euclides]\label{thm:b1:arith:euclidprimes}
Er zijn oneindig veel \omterm{def:b1:arith:prime}{priemgetallen}.
\end{theorem}

\begin{proof}
Elk geheel getal $n \geq 2$ heeft een \omterm{def:b1:arith:prime}{priemdeler}: zijn kleinste deler
$\geq 2$ is \omterm{def:b1:arith:prime}{priem} (een echte ontbinding daarvan zou een kleinere deler
van $n$ opleveren). Stel nu dat $p_1, \dots, p_k$ alle \omterm{def:b1:arith:prime}{priemgetallen}
waren, en zet $N = p_1 p_2 \cdots p_k + 1 \geq 2$. Een zeker \omterm{def:b1:arith:prime}{priemgetal}
$p_i$ \omterm{def:b1:arith:divides}{deelt} $N$; maar $p_i$ \omterm{def:b1:arith:divides}{deelt} ook $N - 1 = p_1\cdots p_k$, dus $p_i
\mid 1$ --- ongerijmd.
\end{proof}

\begin{theorem}[Hoofdstelling van de getaltheorie]\label{thm:b1:arith:fta}
Elk geheel getal $n \geq 2$ is een product van \omterm{def:b1:arith:prime}{priemgetallen}, en de
ontbinding
\[
n = p_1^{\alpha_1} p_2^{\alpha_2} \cdots p_k^{\alpha_k}
\qquad (p_1 < p_2 < \dots < p_k \text{ priem},\ \alpha_i \in \N^*)
\]
is uniek.\index{priemontbinding}
\end{theorem}

\begin{proof}
\emph{Bestaan} met sterke inductie (\cref{thm:b1:logic:induction}): $n =
2$ is \omterm{def:b1:arith:prime}{priem}; voor $n > 2$ is $n$ ofwel \omterm{def:b1:arith:prime}{priem}, ofwel $n = ab$ met $2 \leq
a, b < n$, en de inductiehypothese ontbindt $a$ en $b$.

\emph{Uniciteit.} Stel $p_1 \cdots p_r = q_1 \cdots q_s$ (\omterm{def:b1:arith:prime}{priemgetallen}
met herhaling opgesomd, zeg $r \leq s$) en voer inductie naar $r$. Is $r
= 0$, dan is het linkerlid $1$, wat $s = 0$ afdwingt (een niet-leeg
product van \omterm{def:b1:arith:prime}{priemgetallen} is groter dan $1$). Voor $r \geq 1$: het
\omterm{def:b1:arith:prime}{priemgetal} $p_1$ \omterm{def:b1:arith:divides}{deelt} $q_1(q_2\cdots q_s)$, dus volgens het lemma van
Euclides geldt $p_1 \mid q_1$ of $p_1 \mid q_2\cdots q_s$; herhalen geeft
dat $p_1$ een zekere $q_j$ \omterm{def:b1:arith:divides}{deelt}. Maar $q_j$ is \omterm{def:b1:arith:prime}{priem} en $p_1 \geq 2$:
noodzakelijk is $p_1 = q_j$. Streep die gemeenschappelijke factor weg
(geoorloofd: $\Z$ is een integriteitsdomein) en verkrijg
\[
p_2 \cdots p_r = q_1 \cdots \widehat{q_j} \cdots q_s
\]
(het dakje duidt de weglating aan), een gelijkheid van kortere producten;
volgens de inductiehypothese vallen de twee lijsten $p_2, \dots, p_r$ en
$q_1, \dots, \widehat{q_j}, \dots, q_s$ op de volgorde na samen, en dus
ook de oorspronkelijke. De exponentvorm groepeert gelijke \omterm{def:b1:arith:prime}{priemgetallen}.
\end{proof}

\begin{proposition}[Waarderingen]\label{prop:b1:arith:valuation}
Schrijf voor $p$ \omterm{def:b1:arith:prime}{priem} en $n \in \N^*$ het symbool
$v_p(n)$\index{p-adische waardering@$p$-adische waardering} voor de
exponent van $p$ in de ontbinding van $n$ (met $v_p(n) = 0$ als $p \nmid
n$). Dan is
\[
v_p(mn) = v_p(m) + v_p(n),
\qquad
m \mid n \iff \forall p,\ v_p(m) \leq v_p(n),
\]
\[
v_p\bigl(\gcd(m,n)\bigr) = \min\bigl(v_p(m), v_p(n)\bigr),
\qquad
v_p\bigl(\operatorname{lcm}(m,n)\bigr) = \max\bigl(v_p(m),
v_p(n)\bigr).
\]
\end{proposition}

\begin{proof}
De eerste identiteit geldt omdat de ontbindingen vermenigvuldigen en de
ontbinding van $mn$ uniek is. Is $m \mid n$, schrijf dan $n = mq$ en pas
haar toe. Omgekeerd, geldt $v_p(m) \leq v_p(n)$ voor alle $p$, dan
voldoet het gehele getal $q = \prod_p p^{\,v_p(n) - v_p(m)}$ aan $mq =
n$. De formule voor de ggd: het getal $d = \prod p^{\min}$ \omterm{def:b1:arith:divides}{deelt} beide
volgens het criterium, en elke gemene deler $c$ heeft $v_p(c) \leq \min$
voor alle $p$, dus $c \mid d$; dezelfde redenering voor het kgv met
$\max$.
\end{proof}

\begin{example}[Kwadraten en derde machten via waarderingen]\label{ex:b1:arith:squarecube}
Een geheel getal $n \geq 1$ is een volkomen kwadraat dan en slechts dan
als elke $v_p(n)$ even is (is $n = m^2$, dan is $v_p(n) = 2v_p(m)$;
omgekeerd halveer je elke exponent). Analoog voor derde machten met
veelvouden van $3$. Zo is $21168 = 2^4 \times 3^3 \times 7^2$ geen
kwadraat ($v_3 = 3$ is oneven) en geen derde macht ($v_2 = 4$); het
kleinste positieve gehele getal $m$ waarvoor $21168\,m$ \emph{wel} een
derde macht is, vind je door elke exponent tot het volgende veelvoud van
$3$ aan te vullen:
\[
m = 2^{6-4} \times 3^{3-3} \times 7^{3-2} = 2^2 \times 7 = 28,
\qquad
21168 \times 28 = 2^6\,3^3\,7^3 = (2^2 \times 3 \times 7)^3
= 84^3 .
\]
Het inzicht: multiplicatieve vragen (kwadraten, derde machten, delers,
ggd, kgv) worden \emph{coördinaatsgewijze} vragen over de
exponentvectoren $(v_2, v_3, v_5, \dots)$ --- en de uniciteit van de
\omterm{thm:b1:arith:fta}{priemontbinding} is precies de \omterm{def:b1:logic:statement}{uitspraak} dat die coördinaten bestaan en
welbepaald zijn.
\end{example}

\section{Congruenties}

\begin{definition}\label{def:b1:arith:congruence}
Voor $n \in \N^*$ schrijven we $a \equiv b \pmod n$\index{congruentie}
wanneer $n \mid a - b$. Dat is een \omterm{def:b1:logic:equiv}{equivalentierelatie} die verenigbaar is
met optellen en vermenigvuldigen: is $a \equiv b$ en $a' \equiv b'$
(mod $n$), dan is $a + a' \equiv b + b'$, $aa' \equiv bb'$ en $a^k \equiv
b^k$ voor $k \in \N$.
\end{definition}

\begin{example}[De negenproef]\label{ex:b1:arith:castingnines}
De verenigbaarheid met $+$ en $\times$ is een controlemiddel dat zo oud
is als de handel. Omdat $10 \equiv 1 \pmod 9$, is elk geheel getal modulo
$9$ congruent met zijn cijfersom (bewezen als \cref{exo:b1:arith:2}). Om
de bewering $1234 \times 567 = 699\,678$ te toetsen: de cijfersommen
geven $1234 \equiv 1$ en $567 \equiv 18 \equiv 0 \pmod 9$, dus moet het
product $\equiv 1 \times 0 = 0$ zijn; en inderdaad is $6 + 9 + 9 + 6 + 7
+ 8 = 45 \equiv 0$. De proef slaagt (en het product klopt ook echt). Had
iemand $699\,478$ gemeld, dan zou de cijfersom $43 \equiv 7 \not\equiv 0$
hem meteen ontmaskeren. De toets is eenzijdig --- ze betrapt een fout
tenzij de fout zelf een veelvoud van $9$ is --- en dat is in het klein
precies de les over pseudopriemgetallen uit
\cref{ex:b1:arith:pseudoprime}: controles met \omterm{def:b1:arith:congruence}{congruenties} weerleggen,
ze bewijzen niet.
\end{example}

\begin{proposition}[Inverteerbaarheid modulo $n$]\label{prop:b1:arith:invmod}
$a$ is \emph{inverteerbaar modulo $n$} (dat wil zeggen $ab \equiv 1 \pmod
n$ voor een zekere $b$) dan en slechts dan als $\gcd(a, n) = 1$. De
inverse is dan uniek modulo $n$ en wordt berekend met het uitgebreide
\omterm{met:b1:arith:euclid}{algoritme van Euclides}.
\end{proposition}

\begin{proof}
$ab \equiv 1 \pmod n$ betekent $ab + nk = 1$ voor een zekere $k$: een
betrekking van Bézout, die bestaat precies wanneer $\gcd(a,n) = 1$
(\cref{cor:b1:arith:bezout}). Uniciteit: is $ab \equiv ab' \equiv 1$, dan
is $b \equiv b(ab') = (ab)b' \equiv b' \pmod n$.
\end{proof}

\begin{example}[$7$ inverteren modulo $26$]\label{ex:b1:arith:inv7mod26}
Omdat $\gcd(7, 26) = 1$, is de klasse van $7$ inverteerbaar modulo $26$.
Uitgebreid Euclides:
\[
26 = 3 \times 7 + 5, \qquad
7 = 1 \times 5 + 2, \qquad
5 = 2 \times 2 + 1 ,
\]
en dan achterstevoren:
\[
1 = 5 - 2 \times 2 = 5 - 2(7 - 5) = 3 \times 5 - 2 \times 7
= 3(26 - 3 \times 7) - 2 \times 7 = 3 \times 26 - 11 \times 7 .
\]
Bijgevolg is $7 \times (-11) \equiv 1 \pmod{26}$, oftewel $7^{-1} \equiv
-11 \equiv 15 \pmod{26}$; controle: $7 \times 15 = 105 = 4 \times 26 +
1$. Met de inverse in handen los je elke \omterm{def:b1:arith:congruence}{congruentie} $7x \equiv c
\pmod{26}$ in één vermenigvuldiging op: $x \equiv 15c$. Dit mechanische
inverteren is het werkpaard van het modulaire rekenen --- en van de
protocollen met publieke sleutel die in
\cref{rem:b1:arith:whereused} genoemd worden, waar de moduli honderden
cijfers tellen maar het algoritme exact dit is.
\end{example}

\begin{example}[Wanneer de coëfficiënt niet inverteerbaar is]\label{ex:b1:arith:noninvertible}
Los $12x \equiv 8 \pmod{20}$ op. Hier is $\gcd(12, 20) = 4$, dus $12$ is
niet inverteerbaar modulo $20$ --- maar de vergelijking blijft
hanteerbaar. De \omterm{def:b1:arith:congruence}{congruentie} zegt $20 \mid 12x - 8$; deling van de hele
betrekking door $4$ (een deler van alle drie de ingrediënten) maakt haar
equivalent met $5 \mid 3x - 2$, oftewel
\[
3x \equiv 2 \pmod 5 .
\]
Nu is $\gcd(3, 5) = 1$ en $3^{-1} \equiv 2 \pmod 5$ (want $3 \times 2 = 6
\equiv 1$), dus $x \equiv 4 \pmod 5$: de oplossingen zijn $x \equiv 4, 9,
14, 19 \pmod{20}$ --- \emph{vier} klassen modulo $20$, overeenkomend met
de ggd. (Was het rechterlid niet \omterm{def:b1:arith:divides}{deelbaar} door $4$ geweest, zeg $12x
\equiv 6 \pmod{20}$, dan was er helemaal geen oplossing: het linkerlid is
altijd $\equiv 0 \pmod 4$.) Algemene vorm: $ax \equiv b \pmod n$ is
oplosbaar precies wanneer $\gcd(a, n) \mid b$, en heeft dan precies
$\gcd(a, n)$ oplossingsklassen --- deel alles door de ggd en inverteer.
\end{example}

\begin{theorem}[Kleine stelling van Fermat]\label{thm:b1:arith:fermat}
Zij $p$ \omterm{def:b1:arith:prime}{priem}. Voor elke $a \in \Z$ geldt
\[
a^p \equiv a \pmod p,
\]
en is $p \nmid a$, dan $a^{p-1} \equiv 1 \pmod p$.\index{kleine stelling
van Fermat}
\end{theorem}

\begin{proof}
Merk eerst op dat voor $1 \leq k \leq p - 1$ de \omterm{def:b1:counting:objects}{binomiaalcoëfficiënt}
$\binom pk = \frac{p!}{k!(p-k)!}$ \omterm{def:b1:arith:divides}{deelbaar} is door $p$: immers
$k!\,(p-k)!\, \binom pk = p!$, en $p$ \omterm{def:b1:arith:divides}{deelt} $p!$ maar is \omterm{cor:b1:arith:bezout}{relatief priem}
met $k!(p-k)!$ (alle factoren zijn $< p$), zodat het lemma van Gauss $p
\mid \binom pk$ geeft.

Bewijs nu $a^p \equiv a$ voor $a \in \N$ met inductie. Waar voor $a = 0$.
Is $a^p \equiv a$, dan geeft het binomium
\[
(a+1)^p = \sum_{k=0}^{p} \binom pk a^k
\equiv a^p + 1 \equiv a + 1 \pmod p,
\]
waarbij alle middelste termen modulo $p$ verdwijnen. Voor $a < 0$ pas je
het resultaat toe op $-a$ en onderscheid je $p = 2$ (waar $x \equiv -x$)
van oneven $p$ (waar $(-a)^p = -a^p$). Is ten slotte $p \nmid a$, dan
vermenigvuldig je $a^p \equiv a$ met een inverse van $a$ modulo $p$
(\cref{prop:b1:arith:invmod}).
\end{proof}

\begin{example}[De omkering van Fermat faalt: $341$]\label{ex:b1:arith:pseudoprime}
De kleine stelling van Fermat levert een goedkope toets op
\emph{samengesteldheid}: is $a^{n-1} \not\equiv 1 \pmod n$ voor een
zekere $a$ die \omterm{cor:b1:arith:bezout}{relatief priem} is met $n$, dan is $n$ niet \omterm{def:b1:arith:prime}{priem}. Zou de
toets ook primaliteit kunnen waarborgen? Nee: neem $n = 341 = 11 \times
31$, samengesteld, en $a = 2$. Omdat $2^{10} = 1024 = 3 \times 341 + 1$,
is
\[
2^{10} \equiv 1 \pmod{341}
\qquad\Longrightarrow\qquad
2^{340} = \bigl(2^{10}\bigr)^{34} \equiv 1 \pmod{341} :
\]
het samengestelde getal $341$ doorstaat de toets van Fermat voor het
grondtal $2$ (het is het kleinste zulke \emph{pseudopriemgetal}). Het
grondtal $3$ ontmaskert het ($3^{340} \not\equiv 1$), en in de praktijk
draait een primaliteitstoets daarom op verscheidene grondtallen, plus
verfijningen --- de industriële versies van dit idee zijn wat de grote
\omterm{def:b1:arith:prime}{priemgetallen} uit \cref{rem:b1:arith:whereused} waarborgt. Moraal: een
implicatie en haar omkering leiden gescheiden levens
(\cref{rem:b1:logic:pitfalls}), zelfs bij stellingen.
\end{example}

\begin{example}[Praktisch rekenen met congruenties]\label{ex:b1:arith:congruences}
Wat is de rest van $7^{2026}$ modulo $11$? Volgens Fermat is $7^{10}
\equiv 1 \pmod{11}$. Omdat $2026 = 10 \times 202 + 6$, is
\[
7^{2026} \equiv 7^6 = (7^2)^3 = 49^3 \equiv 5^3 = 125 \equiv 4
\pmod{11}.
\]
De rest is $4$. De strategie: reduceer de exponent modulo de orde die
Fermat levert, en reduceer daarna bij elke stap de tussenmachten.
\end{example}

\begin{remark}[Veelgemaakte fouten in de getaltheorie]\label{rem:b1:arith:pitfalls}
\begin{enumerate}
  \item \emph{Een \omterm{def:b1:arith:congruence}{congruentie} delen.} Uit $ac \equiv bc \pmod n$ mag je
        \emph{niet} $a \equiv b$ besluiten tenzij $\gcd(c, n) = 1$: er
        is $6 \equiv 2 \pmod 4$ maar $3 \not\equiv 1 \pmod 4$. De juiste
        algemene regel \omterm{def:b1:arith:divides}{deelt} ook de \omterm{def:b1:complex:field}{modulus}: $ac \equiv bc \pmod n \iff
        a \equiv b \pmod{n/\gcd(c,n)}$.
  \item \emph{Het lemma van Euclides misbruiken.} Uit $a \mid bc$ volgt
        $a \mid b$ of $a \mid c$ alleen voor \emph{prieme} $a$ (of voor
        $a$ \omterm{cor:b1:arith:bezout}{relatief priem} met een van beide factoren): $6 \mid 4 \times
        9$, en toch \omterm{def:b1:arith:divides}{deelt} $6$ geen van beide factoren.
  \item \emph{\omterm{cor:b1:arith:bezout}{Relatief priem} is een relatie, geen eigenschap.} ``$8$ en
        $9$ zijn \omterm{cor:b1:arith:bezout}{relatief priem}'' is waar hoewel geen van beide \omterm{def:b1:arith:prime}{priem}
        is; ``twee aan twee \omterm{cor:b1:arith:bezout}{relatief priem}'' is sterker dan ``in hun
        geheel \omterm{cor:b1:arith:bezout}{relatief priem}'' ($\gcd(6, 10, 15) = 1$, terwijl geen
        enkel paar \omterm{cor:b1:arith:bezout}{relatief priem} is).
  \item \emph{Exponenten leven niet modulo $n$.} In $a^k \bmod n$ mag de
        exponent alleen modulo de \emph{orde} van $a$ gereduceerd worden
        (bijvoorbeeld $p - 1$ wanneer Fermat van toepassing is), nooit
        modulo $n$: $2^{10} \bmod 11$ is $1$, niet $2^{10 \bmod 11} =
        2^{10}$ --- de reductie die wel werkt is die van
        \cref{ex:b1:arith:congruences}.
\end{enumerate}
\end{remark}

\begin{remark}[Waar dit hoofdstuk gebruikt wordt]\label{rem:b1:arith:whereused}
Dit hoofdstuk is evenzeer een sjabloon als een gereedschapskist. De hele
keten --- euclidische deling, ggd, Bézout, Gauss, unieke ontbinding ---
wordt letterlijk overgespeeld voor veeltermen in \cref{ch:b1:poly}, waar
``graad'' de rol van de absolute waarde speelt; beide hoofdstukken naast
elkaar leggen is de beste manier om ze allebei te begrijpen. Het rekenen
met \omterm{def:b1:arith:congruence}{congruenties} wordt de ring $\Z/n\Z$ in \cref{ch:b1:structures},
waarvan de inverteerbare elementen (\cref{prop:b1:arith:invmod}) het
eerste niet-triviale voorbeeld van een eenhedengroep vormen.
Waarderingen keren terug in de weekendopgave hieronder (de formule van
Legendre) en dragen de irrationaliteitsbewijzen van \cref{ch:b1:reals}.
Buiten dit volume is inverteren met Bézout modulo $n$ de motor van de
cryptografie met publieke sleutel, en is de kleine stelling van Fermat de
grootvader van de primaliteitstests die de grote \omterm{def:b1:arith:prime}{priemgetallen} daar
waarborgen.
\end{remark}

\begin{remark}[Tussenspel: $\Z$ als sjabloon]\label{rem:b1:arith:interlude}
Doe een stap terug van de afzonderlijke stellingen en bekijk de
architectuur van het hoofdstuk: één gereedschap (de euclidische deling)
bracht een classificatie voort (de deelgroepen $n\Z$), die een
bestaansstelling voortbracht (ggd, Bézout), die een rekenkunde van de
\omterm{def:b1:arith:divides}{deelbaarheid} voortbracht (Gauss), die de unieke ontbinding voortbracht
--- en elke verdieping rust alleen op de verdieping eronder. Datzelfde
gebouw wordt in dit volume nog tweemaal opgetrokken met een andere
benedenverdieping: in \cref{ch:b1:poly}, waar delen naar graad het delen
naar grootte vervangt en al het bovenstaande zich \emph{letterlijk}
herhaalt; en, in het klein, binnen elke $\Z/n\Z$ van
\cref{ch:b1:structures}, waar vragen over inverteerbaarheid (de
\cref{prop:b1:arith:invmod} van dit hoofdstuk) structurele \omterm{def:b1:logic:statement}{uitspraken} over
ringen en lichamen worden. Een argument herkennen als ``het
$\Z$-argument, overgeplant'' is de snelste manier om die hoofdstukken te
leren --- en de eerste smaak van de kerngewoonte van de algebra: stellingen
bewijzen over \emph{axioma's} in plaats van over objecten.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:arith:1}
Bereken $\gcd(1\,001, 777)$ met het \omterm{met:b1:arith:euclid}{algoritme van Euclides}, en geef er een
paar van Bézout bij.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:arith:2}
Bewijs de deelbaarheidsregels in grondtal $10$: een geheel getal is
modulo $9$ congruent met de som van zijn cijfers, en modulo $11$ met de
alternerende som van zijn cijfers. Wat is $123\,456\,789$ modulo $9$ en
modulo $11$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:arith:3}
Los op in $\Z$: $91x \equiv 1 \pmod{237}$ \emph{(uitgebreid Euclides)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:arith:4}
Bepaal alle paren $(x, y) \in \Z^2$ met $17x + 39y = 1$; en vervolgens
alle paren met $17 x + 39 y = 5$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:5}
Bewijs dat voor $a, b \in \N^*$ geldt: $\gcd(a,b) \times
\operatorname{lcm}(a,b) = ab$. \emph{(Gebruik de formules voor
waarderingen uit \cref{prop:b1:arith:valuation} en $\min(\alpha,\beta) +
\max(\alpha,\beta) = \alpha + \beta$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:6}
Zij $a = 2^{10} \times 3^4 \times 5^2$ en $b = 2^6 \times 3^7 \times 7$.
Bereken $\gcd(a, b)$, $\operatorname{lcm}(a,b)$ en het aantal positieve
delers van $a$. \emph{(Bewijs de formule $\prod_i (\alpha_i + 1)$ voor
het aantal delers.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:7}
Bewijs dat $\sqrt p$ irrationaal is voor elk \omterm{def:b1:arith:prime}{priemgetal} $p$, met
waarderingen: vergelijk $v_p$ van beide leden van $p q^2 = r^2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:8}
(Chinees restprobleem) Bepaal alle gehele getallen $x$ met
\[
x \equiv 2 \pmod 7, \qquad x \equiv 5 \pmod{11}.
\]
Bewijs onderweg dat voor \omterm{cor:b1:arith:bezout}{relatief prieme} $m, n$ het paar \omterm{def:b1:arith:congruence}{congruenties} $x
\equiv a \ (m)$, $x \equiv b\ (n)$ altijd een oplossing heeft, uniek
modulo $mn$.\index{Chinese reststelling}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:9}
Bereken $3^{1000}$ modulo $7$, en de laatste twee decimale cijfers van
$7^{100}$ \emph{(modulo $100 = 4 \times 25$: gebruik
\cref{exo:b1:arith:8})}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:arith:10}
Bewijs voor $m, n \in \N^*$ dat $\gcd(2^m - 1,\, 2^n - 1) =
2^{\gcd(m,n)} - 1$. \emph{Aanwijzing: toon eerst aan dat de rest van $2^m
- 1$ modulo $2^n - 1$ gelijk is aan $2^r - 1$, waarbij $r$ de rest van
$m$ modulo $n$ is; volg dan het \omterm{met:b1:arith:euclid}{algoritme van Euclides}.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:arith:11}
(Stelling van Wilson) Zij $p$ \omterm{def:b1:arith:prime}{priem}. Bewijs dat
\[
(p-1)! \equiv -1 \pmod p ,
\]
door elke factor van $(p-1)!$ met haar inverse modulo $p$ te koppelen en
de factoren aan te wijzen die met zichzelf gekoppeld worden (los eerst
$x^2 \equiv 1 \pmod p$ op). Ga de omkering na: is $n \geq 2$ niet \omterm{def:b1:arith:prime}{priem},
dan is $(n-1)! \not\equiv -1 \pmod n$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:arith:12}
(Getallen van Fermat) Zet voor $n \in \N$ het getal $F_n = 2^{2^n} + 1$.
\begin{enumerate}
  \item Bewijs dat $F_0 F_1 \cdots F_{n-1} = F_n - 2$ voor $n \geq 1$
        (inductie).
  \item Leid af dat de getallen van Fermat twee aan twee \omterm{cor:b1:arith:bezout}{relatief priem}
        zijn.
  \item Leid daaruit een tweede bewijs af, onafhankelijk van
        \cref{thm:b1:arith:euclidprimes}, dat er oneindig veel
        \omterm{def:b1:arith:prime}{priemgetallen} zijn.
\end{enumerate}
\end{exercise}

\section{Opgave: de formule van Legendre en de overdrachten van Kummer}

\begin{problem}\label{pb:b1:arith:1}
Op hoeveel nullen eindigt de decimale schrijfwijze van $1000!$ --- en,
dieper, wat is de exacte macht van een \omterm{def:b1:arith:prime}{priemgetal} $p$ die $1000!$ \omterm{def:b1:arith:divides}{deelt},
of die een \omterm{def:b1:counting:objects}{binomiaalcoëfficiënt} \omterm{def:b1:arith:divides}{deelt}? De volledige antwoorden zijn twee
juwelen van de elementaire getaltheorie: de \emph{formule van
Legendre}\index{formule van Legendre} $v_p(n!) = \sum_{k\geq1} \lfloor
n/p^k \rfloor$, met haar digitale tegenhanger $v_p(n!) = \frac{n -
s_p(n)}{p-1}$, en de \emph{stelling van Kummer}\index{stelling van
Kummer}: $v_p\binom{m+n}m$ telt de \emph{overdrachten} bij het optellen
van $m$ en $n$ in grondtal $p$. Deze opgave bewijst beide, toetst ze
numeriek aan elkaar, en oogst de klassieke gevolgen --- nullen aan het
einde, de pariteit van de driehoek van Pascal, en een eerste afschatting
in de richting van de priemgetalstelling. Overal is $p$ \omterm{def:b1:arith:prime}{priem}, is
$\floor{x}$ het geheel deel, en noteert $s_p(n)$ de som van de cijfers
van $n$ geschreven in grondtal $p$.

\medskip
\noindent\textbf{Deel I --- Gehele delen, waarderingen en de formule van
Legendre.}
\begin{enumerate}
  \item Opwarmer: bereken $10!$ en lees af op hoeveel nullen het
        eindigt; bereken $v_2(10!)$ en $v_5(10!)$ rechtstreeks uit de
        ontbinding van elke factor $1, 2, \dots, 10$.
  \item Bewijs dat voor $x \in \R$ en $n \in \N^*$ geldt $\bigl\lfloor
        \lfloor x \rfloor / n \bigr\rfloor = \lfloor x/n \rfloor$.
  \item Bewijs dat $v_p(a + b) \geq \min\bigl(v_p(a), v_p(b)\bigr)$ voor
        alle $a, b \in \N^*$, met gelijkheid zodra $v_p(a) \neq
        v_p(b)$.
  \item Toon aan dat het aantal veelvouden van $m$ in $\intint1n$ gelijk
        is aan $\lfloor n/m \rfloor$.
  \item Bewijs de \emph{formule van Legendre}: voor elke $n \in \N^*$
        geldt
        \[
        v_p(n!) = \sum_{k=1}^{\infty}
        \Bigl\lfloor \frac{n}{p^k} \Bigr\rfloor
        \]
        (een eindige som: de termen verdwijnen zodra $p^k > n$).
        \emph{Tel voor elke $k$ de factoren van $\intint1n$ die door
        $p^k$ \omterm{def:b1:arith:divides}{deelbaar} zijn: elke draagt precies één eenheid bij per
        niveau dat ze haalt.}
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De digitale vorm en de nullen aan het
einde.}
\begin{enumerate}[resume]
  \item Bereken $v_5(1000!)$ en $v_2(1000!)$ en besluit: op hoeveel
        nullen eindigt $1000!$?
  \item Bewijs de digitale vorm van de formule van Legendre: schrijf je
        $n = \sum_i a_i p^i$ in grondtal $p$, dan is
        \[
        v_p(n!) = \frac{n - s_p(n)}{p - 1} .
        \]
  \item Twee gevolgen voor $p = 2$: toon aan dat $2^n$ het getal $n!$
        nooit \omterm{def:b1:arith:divides}{deelt}, en dat $2^{n-1}$ het getal $n!$ precies dan \omterm{def:b1:arith:divides}{deelt}
        wanneer $n$ een macht van $2$ is.
  \item Schat het tekort af: toon aan dat $\frac n{p-1} - \log_p(n) - 1
        \leq v_p(n!) < \frac n{p-1}$, zodat $\frac{v_p(n!)}{n} \to
        \frac1{p-1}$: op de lange duur groeit er per eenheid een aandeel
        $\frac1{p-1}$ van een factor $p$ aan.
  \item Zij $Z(n) = v_5(n!)$ het aantal nullen waarop $n!$ eindigt. Toon
        aan dat $Z(n) - Z(n-1) = v_5(n)$, leid af dat $Z$ de waarde $5$
        volledig overslaat (bereken $Z(24)$ en $Z(25)$), en bewijs dat
        geen enkele faculteit op precies vijf nullen eindigt.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De stelling van Kummer.}
\begin{enumerate}[resume]
  \item Bewijs dat $\lfloor x + y \rfloor - \lfloor x \rfloor - \lfloor y
        \rfloor \in \{0, 1\}$ voor alle $x, y \in \R$, en leid uit de
        formule van Legendre af dat
        \[
        v_p\binom{m+n}m
        = \sum_{k\geq1}\Bigl(
        \Bigl\lfloor\frac{m+n}{p^k}\Bigr\rfloor
        - \Bigl\lfloor\frac{m}{p^k}\Bigr\rfloor
        - \Bigl\lfloor\frac{n}{p^k}\Bigr\rfloor\Bigr),
        \]
        een som waarvan elke term $0$ of $1$ is.
  \item Bewijs de \emph{stelling van Kummer}: de $k$-de term van die som
        is gelijk aan $1$ precies wanneer het optellen van $m$ en $n$ in
        grondtal $p$ een overdracht naar positie $k$ voortbrengt; dus
        telt $v_p\binom{m+n}m$ het totale aantal overdrachten.
        \emph{(Schrijf $m = p^km_1 + m_0$ en $n = p^kn_1 + n_0$ met $0
        \leq m_0, n_0 < p^k$ en bekijk $\lfloor (m_0 + n_0)/p^k
        \rfloor$.)}
  \item Leid af dat voor $0 < j < p^k$ geldt
        \[
        v_p\binom{p^k}{j} = k - v_p(j) ,
        \]
        door de overdrachten in de optelling $j + (p^k - j)$ te tellen.
        (In het bijzonder $p \mid \binom p j$ voor $0 < j < p$: de
        sleutelstap van \cref{thm:b1:arith:fermat}, teruggevonden.)
  \item Bewijs dat $v_2\binom{2n}n = s_2(n)$. Leid af dat de centrale
        \omterm{def:b1:counting:objects}{binomiaalcoëfficiënt} altijd even is, en dat $\binom{2n}n \equiv 2
        \pmod 4$ precies wanneer $n$ een macht van $2$ is.
  \item Toon met de identiteit van Vandermonde
        (\cref{exo:b1:counting:7}) en vraag 13 aan dat $\binom{2p}p
        \equiv 2 \pmod p$ voor elk \omterm{def:b1:arith:prime}{priemgetal} $p$.
  \item Bereken $v_3\binom{1000}{500}$ tweemaal: één keer met Kummer
        (schrijf $500$ in grondtal $3$ en tel de overdrachten in $500 +
        500$), één keer met de digitale vorm van Legendre (bereken
        $s_3(500)$ en $s_3(1000)$); ga na dat beide dezelfde waarde
        geven.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De pariteit van de driehoek van Pascal, en
een afschatting van de priemdichtheid.}
\begin{enumerate}[resume]
  \item Bewijs het cijfercriterium: $\binom nk$ is \emph{oneven} dan en
        slechts dan als elk binair cijfer van $k$ hoogstens gelijk is
        aan het overeenkomstige cijfer van $n$. Formuleer en bewijs het
        analoge criterium voor $p \nmid \binom nk$ in grondtal $p$.
  \item Leid af dat rij $n$ van de driehoek van Pascal precies
        $2^{s_2(n)}$ oneven getallen bevat; ga dat na op de rijen $4$ en
        $5$.
  \item Leid af dat alle inwendige getallen $\binom nk$ ($0 < k < n$)
        even zijn dan en slechts dan als $n$ een macht van $2$ is.
  \item Bewijs dat elke priemmacht die $\binom{m+n}m$ \omterm{def:b1:arith:divides}{deelt} hoogstens $m
        + n$ is: uit $p^a \mid \binom{m+n}m$ volgt $p^a \leq m + n$.
        \emph{(Hoeveel termen van de som uit vraag 11 kunnen ongelijk
        aan nul zijn?)}
  \item Leid af dat $\binom{2n}n$ het getal $\operatorname{lcm}(1, 2,
        \dots, 2n)$ \omterm{def:b1:arith:divides}{deelt}, en combineer dat met de ondergrens
        $\binom{2n}n \geq \frac{4^n}{2n+1}$ (die je bewijst: het
        centrale getal is het grootste van de $2n + 1$ getallen van rij
        $2n$) om
        \[
        \operatorname{lcm}(1, \dots, 2n) \geq \frac{4^n}{2n+1}
        \]
        te verkrijgen: de gemene veelvouden van de eerste gehele getallen
        groeien \emph{exponentieel} --- een eerste kwantitatieve blik op
        de overvloed aan \omterm{def:b1:arith:prime}{priemgetallen}.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Synthese.}
\begin{enumerate}[resume]
  \item Bepaal de kleinste $n$ waarvoor $n!$ op minstens $2026$ nullen
        eindigt. \emph{(Schat $Z(n) \approx n/4$ en corrigeer met de
        exacte formule.)}
  \item Nog een laatste kruiscontrole: toon aan dat $7$ het getal
        $\binom{100}{50}$ \emph{niet} \omterm{def:b1:arith:divides}{deelt}, eerst door $50$ in grondtal
        $7$ te schrijven en na te gaan dat de optelling $50 + 50$ geen
        overdrachten kent, en daarna door $v_7(100!)$ en $v_7(50!)$ met
        de formule van Legendre te berekenen.
  \item Waar precies gebruikte de opgave: (i) de unieke \omterm{thm:b1:arith:fta}{priemontbinding};
        (ii) de ontbinding $n = p^k n_1 + n_0$ uit de euclidische
        deling; (iii) een telargument uit \cref{ch:b1:counting}? Eén zin
        per onderdeel.
  \item Synthese, in een korte alinea: de formule van Legendre zet een
        vraag over \omterm{def:b1:arith:divides}{deelbaarheid} om in rekenen met cijfers, en de stelling
        van Kummer leest het antwoord af aan de overdrachten van één
        optelling --- becommentarieer die vertaling, de controles van
        vraag 16, en wat de afschatting van vraag 21 over \omterm{def:b1:arith:prime}{priemgetallen}
        suggereert (de volledige \omterm{def:b1:logic:statement}{uitspraak}, de priemgetalstelling, ligt
        ver buiten dit volume; de tegenhanger van de gereedschapskist
        van dit hoofdstuk voor veeltermen is \cref{ch:b1:poly}).
\end{enumerate}
\end{problem}
