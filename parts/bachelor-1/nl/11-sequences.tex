\chapter{Rijen}\label{ch:b1:seq}

In het bovenbouwvolume werd met rijen gerekend terwijl het limietbegrip
half op goed vertrouwen werd aangenomen. Hier wordt de theorie opnieuw
opgebouwd op de volledigheid van $\R$ (\cref{ch:b1:reals}): elke
klassieke stelling --- de monotone limiet, \omterm{thm:b1:seq:adjacent}{ingesloten rijen},
Bolzano--Weierstrass, het criterium van Cauchy --- is een gezicht van dat
ene axioma. Het hoofdstuk eindigt met de praktische studie van rijen
gegeven door $u_{n+1} = f(u_n)$.

\section{Convergentie}

\begin{definition}[Limiet van een rij]\label{def:b1:seq:limit}
Een rij $(u_n)$ reële getallen \emph{convergeert} naar $\ell \in \R$
wanneer\index{limiet!van een rij}
\[
\forall \varepsilon > 0,\ \exists N \in \N,\ \forall n \geq N,
\qquad \abs{u_n - \ell} \leq \varepsilon .
\]
Men schrijft $u_n \to \ell$ of $\lim u_n = \ell$. Een rij die niet (naar
enig reëel getal) convergeert \emph{divergeert}. Divergentie \emph{naar
$+\infty$}: $\forall M,\ \exists N,\ \forall n \geq N,\ u_n \geq M$
(analoog naar $-\infty$).
\end{definition}

\begin{example}[Een $\varepsilon$--$N$-bewijs, één keer uitgeschreven]\label{ex:b1:seq:epsilonN}
Bewering: $u_n = \dfrac{n^2 + 1}{2n^2 - 3} \to \dfrac12$. Isoleer eerst
de fout:
\[
\Bigl| u_n - \frac12 \Bigr|
= \Bigl| \frac{2(n^2 + 1) - (2n^2 - 3)}{2(2n^2 - 3)} \Bigr|
= \frac{5}{2\,\abs{2n^2 - 3}}
= \frac{5}{2\,(2n^2 - 3)} \quad (n \geq 2).
\]
Overheers haar dan met iets eenvoudigs: voor $n \geq 2$ is $2n^2 - 3 \geq
n^2$, dus is de fout $\leq \frac{5}{2n^2} \leq \frac 5{2n}$. Bij gegeven
$\varepsilon > 0$ levert de \omterm{thm:b1:reals:archimedes}{archimedische eigenschap} een $N \geq
\max\bigl(2, \frac{5}{2\varepsilon}\bigr)$; voor $n \geq N$ is de fout
$\leq \varepsilon$. Klaar. Het inzicht tot slot: een
$\varepsilon$--$N$-bewijs bestaat uit precies drie zetten --- bereken de
fout, schat haar af met een dalende elementaire uitdrukking, en los de
drempel op --- en na de stellingen van dit hoofdstuk (bewerkingen,
insluiting) schrijf je zo'n bewijs vrijwel nooit meer: de stellingen
verpakken die drie zetten eens en voor altijd.
\end{example}

\begin{example}[Divergentie naar oneindig, gewaarborgd]\label{ex:b1:seq:toinfinity}
Bewering: $u_n = n^2 - 100n \to +\infty$. Zonder de dominante term af:
$u_n = n^2\bigl(1 - \frac{100}{n}\bigr) \geq \frac{n^2}{2}$ voor $n \geq
200$. Bij gegeven $M$ neem je $N = \max\bigl(200,
\lceil\sqrt{2M}\rceil\bigr)$: voor $n \geq N$ is $u_n \geq \frac{n^2}{2}
\geq M$. Twee gewoontes zijn hier zichtbaar: de dominante term afzonderen
zet een wedstrijd ($n^2$ tegen $-100n$) om in één schaal maal een factor
die naar $1$ gaat; en de drempel mag reusachtig zijn ($u_{100} = 0$, de
rij is vóór $n = 100$ zelfs negatief) --- divergentie naar $+\infty$ is
een \omterm{def:b1:logic:statement}{uitspraak} over de staart, onverschillig voor eender welke eindige
hoeveelheid wangedrag.
\end{example}

\begin{proposition}[Eerste eigenschappen]\label{prop:b1:seq:first}
\begin{enumerate}
  \item De limiet is uniek, zodra ze bestaat.
  \item Een convergente rij is begrensd.
  \item Convergeert $u_n \to \ell$, dan laat elke wijziging van eindig
        veel termen de convergentie en de limiet ongemoeid.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Zijn $u_n \to \ell$ en $u_n \to \ell'$ met $\ell \neq \ell'$, neem
dan $\varepsilon = \frac{\abs{\ell - \ell'}}{3}$: voorbij beide drempels
is $\abs{\ell - \ell'} \leq \abs{\ell - u_n} + \abs{u_n - \ell'} \leq
2\varepsilon = \frac23 \abs{\ell - \ell'}$, ongerijmd.

(2) Met $\varepsilon = 1$: voorbij $N$ is $\abs{u_n} \leq \abs\ell + 1$;
de eindig vele eerdere termen zijn ook begrensd, dus $\abs{u_n} \leq
\max(\abs{u_0}, \dots, \abs{u_{N-1}}, \abs\ell + 1)$.

(3) In detail: stel $v_n = u_n$ voor $n \geq n_0$ en $u_n \to \ell$. Bij
gegeven $\varepsilon > 0$ neem je de drempel $N$ voor $(u_n)$: voor $n
\geq \max(N, n_0)$ is $\abs{v_n - \ell} = \abs{u_n - \ell} \leq
\varepsilon$. Dus $v_n \to \ell$: de definitie kwantificeert alleen over
$n \geq N$, en elk eindig beginstuk mag overschreven worden ten koste van
een grotere drempel. (Daarom volstaan hypothesen ``voor alle grote $n$''
in dit hele hoofdstuk.)
\end{proof}

\begin{theorem}[Bewerkingen met limieten]\label{thm:b1:seq:operations}
Zijn $u_n \to \ell$ en $v_n \to m$, dan is
\[
u_n + v_n \to \ell + m, \qquad u_n v_n \to \ell m, \qquad
\frac{u_n}{v_n} \to \frac{\ell}{m} \ (\text{als } m \neq 0),
\qquad \abs{u_n} \to \abs\ell .
\]
\end{theorem}

\begin{proof}
\emph{Som:} $\abs{(u_n + v_n) - (\ell + m)} \leq \abs{u_n - \ell} +
\abs{v_n - m} \leq 2\varepsilon$ voorbij de grootste drempel.
\emph{Product:} schrijf
\[
u_n v_n - \ell m = (u_n - \ell)\,v_n + \ell\,(v_n - m);
\]
$(v_n)$ is begrensd door een zekere $B$ (\cref{prop:b1:seq:first}), zodat
het rechterlid $\leq B\abs{u_n - \ell} + \abs{\ell}\,\abs{v_n - m}$ is,
willekeurig klein. \emph{Quotiënt:} het volstaat $\frac 1{v_n}$ te
behandelen. Met $\varepsilon = \frac{\abs m}{2}$ is voorbij een zekere
$N_0$ steeds $\abs{v_n} \geq \frac{\abs m}{2}$, dus
\[
\Bigl| \frac{1}{v_n} - \frac 1m \Bigr|
= \frac{\abs{m - v_n}}{\abs{v_n m}}
\leq \frac{2}{m^2}\,\abs{v_n - m} \longrightarrow 0 .
\]
\emph{Absolute waarde:} $\bigl|\abs{u_n} - \abs\ell\bigr| \leq \abs{u_n -
\ell}$ (de omgekeerde driehoeksongelijkheid,
\cref{prop:b1:complex:rules}).
\end{proof}

\begin{example}[Bewerkingen plus één algebraïsche truc]\label{ex:b1:seq:conjugatetrick}
Bereken $\lim\,\bigl(\sqrt{n^2 + n} - n\bigr)$. De twee stukken gaan
afzonderlijk naar $+\infty$: de stelling over de bewerkingen zegt niets
over hun verschil (een \emph{onbepaalde vorm}). Vermenigvuldig met de
toegevoegde uitdrukking:
\[
\sqrt{n^2 + n} - n
= \frac{(n^2 + n) - n^2}{\sqrt{n^2 + n} + n}
= \frac{n}{\sqrt{n^2+n} + n}
= \frac{1}{\sqrt{1 + \frac1n} + 1} .
\]
Nu convergeert alles: $\sqrt{1 + \frac1n} \to 1$, want $0 \leq \sqrt{1 +
h} - 1 = \frac{h}{\sqrt{1+h} + 1} \leq h$ (opnieuw de toegevoegde
uitdrukking, dan insluiten met $h = \frac1n$); de stelling over de
bewerkingen geeft dan de limiet $\frac{1}{1 + 1} = \frac12$. Het inzicht
tot slot: de stelling over de bewerkingen is geen rekenmachine voor alle
limieten --- onbepaalde vormen ($\infty - \infty$, $\frac00$, $0 \times
\infty$, $1^\infty$) moeten eerst met algebra \emph{omgevormd} worden
(toegevoegde uitdrukkingen, de dominante term afzonderen) tot elk stuk
convergeert; de systematische machine voor hardnekkige gevallen is de
asymptotische ontwikkeling van \cref{ch:b1:taylor}.
\end{example}

\begin{theorem}[Limieten en ordening]\label{thm:b1:seq:order}
\begin{enumerate}
  \item Is $u_n \leq v_n$ voor alle grote $n$ en convergeren beide, dan
        is $\lim u_n \leq \lim v_n$. (Strikte ongelijkheden gaan
        \emph{niet} mee naar de limiet: $\frac 1n > 0$ maar $\lim = 0$.)
  \item (Insluitstelling\index{insluitstelling}) Is $u_n \leq w_n \leq
        v_n$ voor alle grote $n$ en $u_n, v_n \to \ell$, dan is $w_n \to
        \ell$.
  \item Is $u_n \to \ell > 0$, dan is $u_n > \frac\ell2 > 0$ voor alle
        grote $n$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Stel $\ell = \lim u_n > m = \lim v_n$; met $\varepsilon = \frac{\ell
- m}{3}$ voldoen de grote termen aan $v_n \leq m + \varepsilon < \ell -
\varepsilon \leq u_n$, in tegenspraak met $u_n \leq v_n$.

(2) Voorbij de drempels: $\ell - \varepsilon \leq u_n \leq w_n \leq v_n
\leq \ell + \varepsilon$.

(3) is \cref{def:b1:seq:limit} met $\varepsilon = \frac\ell2$.
\end{proof}

\begin{example}[Twee insluitingen]\label{ex:b1:seq:squeezes}
(i) $\dfrac{\sin n}{n} \to 0$: uit $-\frac1n \leq \frac{\sin n}{n} \leq
\frac1n$, waarbij beide muren op $0$ instorten --- de grillige teller
hoef je helemaal niet te begrijpen. (ii) $(2^n + 3^n)^{1/n} \to 3$: klem
de binnenkant in,
\[
3^n \leq 2^n + 3^n \leq 2\cdot3^n
\quad\Longrightarrow\quad
3 \leq (2^n + 3^n)^{1/n} \leq 3\cdot 2^{1/n} ,
\]
en $2^{1/n} = \eu^{\frac{\ln 2}{n}} \to 1$ (net als $5^{1/n}$ in
\cref{exo:b1:seq:2}): de insluiting levert $3$. Het inzicht tot slot: een
som van concurrerende exponentiëlen gedraagt zich als haar \emph{grootste}
term --- de kleinere worden opgeslokt door een onschadelijke constante
factor, die de $n$-de wortel vervolgens uitwist.
\end{example}

\section{Monotone rijen}

\begin{theorem}[Stelling van de monotone limiet]\label{thm:b1:seq:monotone}
Een stijgende rij die naar boven begrensd is convergeert, naar $\sup\{u_n
: n \in \N\}$; een stijgende rij die niet naar boven begrensd is
divergeert naar $+\infty$. (Spiegelbeeldige \omterm{def:b1:logic:statement}{uitspraken} voor dalende
rijen.)\index{stelling van de monotone limiet}
\end{theorem}

\begin{proof}
Zij $s = \sup\{u_n\}$ (\cref{thm:b1:reals:sup}). Bij gegeven $\varepsilon
> 0$ levert de $\varepsilon$-karakterisering
(\cref{prop:b1:reals:epsilon}) een $N$ met $u_N > s - \varepsilon$;
wegens de monotonie is $s - \varepsilon < u_N \leq u_n \leq s$ voor alle
$n \geq N$: convergentie naar $s$. Is de rij onbegrensd, dan is er voor
elke $M$ een $u_N > M$, en de monotonie houdt alle latere termen boven
$M$.
\end{proof}

\begin{example}[De monotone stelling als bestaansmachine]\label{ex:b1:seq:existencemachine}
Zij $u_n = \prod_{k=1}^{n} \bigl(1 + \frac{1}{2^k}\bigr)$. Elke factor
overtreft $1$, dus is $(u_n)$ stijgend. Naar boven begrensd? Neem
logaritmen en gebruik $\ln(1 + x) \leq x$
(\cref{ex:b1:derivative:convexineq} loopt erop vooruit; of de grovere
ongelijkheid $1 + x \leq \eu^x$ uit het bovenbouwvolume):
\[
\ln u_n = \sum_{k=1}^{n} \ln\Bigl(1 + \frac{1}{2^k}\Bigr)
\leq \sum_{k=1}^{n} \frac{1}{2^k} < 1 ,
\]
dus $u_n < \eu$. Stijgend en begrensd: $(u_n)$ convergeert naar een
zekere $\ell \in \intoc{u_1}{\eu}$ --- een volstrekt welbepaald reëel
getal zonder gesloten vorm in zicht ($\ell = 2.384\dots$). Het inzicht
tot slot: de stelling van de monotone limiet is de goedkoopste
bestaansmachine van de analyse; ze doopte $\eu$ zelf
(\cref{ex:b1:seq:e} hieronder), en in \cref{ch:b1:series} zal ze de
convergentie van elke positieve reeks met louter begrensdheid
beslechten.
\end{example}

\begin{theorem}[Ingesloten rijen]\label{thm:b1:seq:adjacent}
Zij $(a_n)$ stijgend en $(b_n)$ dalend, met $b_n - a_n \to 0$. Dan
convergeren beide naar een \emph{gemeenschappelijke} limiet $\ell$, en is
$a_n \leq \ell \leq b_n$ voor elke $n$.\index{ingesloten rijen}
\end{theorem}

\begin{proof}
Ten eerste is $a_n \leq b_n$ voor elke $n$: de rij $(b_n - a_n)$ is
dalend en gaat naar $0$, dus is ze $\geq 0$ (een negatieve term zou haar
onder $0$ vastzetten). Vervolgens is $(a_n)$ stijgend en naar boven
begrensd door $b_0$: ze convergeert naar een zekere $\ell$
(\cref{thm:b1:seq:monotone}); evenzo $(b_n) \to \ell'$; en $\ell' - \ell
= \lim (b_n - a_n) = 0$. De ongelijkheden $a_n \leq \ell \leq b_n$ volgen
uit de monotonie ($\ell = \sup a_k \geq a_n$, enzovoort).
\end{proof}

\begin{example}[Het getal $\eu$]\label{ex:b1:seq:e}
Zet $a_n = \sum_{k=0}^{n} \frac{1}{k!}$ en $b_n = a_n + \frac{1}{n \cdot
n!}$ ($n \geq 1$). Dan stijgt $(a_n)$; en
\[
b_{n+1} - b_n = \frac{1}{(n+1)!} + \frac{1}{(n+1)(n+1)!} -
\frac{1}{n\,n!}
= \frac{n(n+1) + n - (n+1)^2}{n(n+1)(n+1)!}
= \frac{-1}{n(n+1)(n+1)!} < 0 ,
\]
dus daalt $(b_n)$, en $b_n - a_n \to 0$: ingesloten. Hun
gemeenschappelijke limiet is (hier per definitie) het getal $\eu \approx
2.71828$; de ongelijkheden $a_n < \eu < b_n$ zijn scherp genoeg om $\eu
\notin \Q$ te bewijzen (\cref{exo:b1:seq:9}).
\end{example}

\section{Deelrijen en Bolzano--Weierstrass}

\begin{definition}[Deelrij]\label{def:b1:seq:subsequence}
Een \emph{deelrij}\index{deelrij} van $(u_n)$ is een rij
$(u_{\varphi(n)})$ waarbij $\varphi \colon \N \to \N$ strikt stijgend is
(merk op dat $\varphi(n) \geq n$, met inductie).
\end{definition}

\begin{proposition}\label{prop:b1:seq:subsequences}
Is $u_n \to \ell$ ($\ell \in \R$ of $\pm\infty$), dan gaat elke \omterm{def:b1:seq:subsequence}{deelrij}
naar $\ell$. Bijgevolg divergeert een rij met twee \omterm{def:b1:seq:subsequence}{deelrijen} met
verschillende limieten. Omgekeerd, convergeren $(u_{2n})$ en $(u_{2n+1})$
allebei naar \emph{dezelfde} $\ell$, dan is $u_n \to \ell$.
\end{proposition}

\begin{proof}
Voorbij de drempel $N$ voor $(u_n)$ voldoen alle indices $\varphi(n) \geq
n \geq N$ (de ongelijkheid $\varphi(n) \geq n$ is de inductie uit
\cref{def:b1:seq:subsequence}: $\varphi(0) \geq 0$, en $\varphi(n+1) >
\varphi(n) \geq n$ dwingt $\varphi(n+1) \geq n + 1$ af). Voor de
omkering: neem bij gegeven $\varepsilon$ de twee drempels $N_0$ (even) en
$N_1$ (oneven); een willekeurige index $n \geq \max(2N_0, 2N_1 + 1)$ is
ofwel even, $n = 2k$ met $k \geq N_0$, ofwel oneven, $n = 2k+1$ met $k
\geq N_1$ --- in beide gevallen is $\abs{u_n - \ell} \leq \varepsilon$:
elke index wordt door een van de twee \omterm{def:b1:seq:subsequence}{deelrijen} gedekt, en dat is het hele
punt.
\end{proof}

\begin{example}[Limieten van deelrijen]\label{ex:b1:seq:sublimits}
Voor $u_n = (-1)^n \frac{n}{n+1}$ gaat de even \omterm{def:b1:seq:subsequence}{deelrij} naar $1$ en de
oneven naar $-1$, zodat de rij divergeert --- maar wel op ordelijke
wijze, geklonterd rond de twee waarden $\pm 1$. Voor $u_n =
\cos\frac{2\pi n}{3}$ zijn de drie \omterm{def:b1:seq:subsequence}{deelrijen} met indices $3k$, $3k + 1$
en $3k + 2$ constant, gelijk aan $1$, $-\frac12$ en $-\frac12$; de
\omterm{def:b1:logic:sets}{verzameling} limieten van \omterm{def:b1:seq:subsequence}{deelrijen} is $\{1, -\frac12\}$. Het inzicht tot
slot: een \emph{begrensde} rij convergeert precies wanneer ze één enkele
deelrijlimiet heeft (\cref{exo:b1:seq:8}); de divergentie van een
begrensde rij betekent altijd minstens twee klonters, en
Bolzano--Weierstrass hieronder waarborgt dat er minstens één is.
\end{example}

\begin{theorem}[Bolzano--Weierstrass]\label{thm:b1:seq:bw}
Elke begrensde rij reële getallen heeft een convergente
\omterm{def:b1:seq:subsequence}{deelrij}.\index{stelling van Bolzano--Weierstrass}
\end{theorem}

\begin{proof}
Zij $u_n \in \intcc{a}{b}$ voor elke $n$. Bouw geneste segmenten met
dichotomie: zet $\intcc{a_0}{b_0} = \intcc{a}{b}$; bevat
$\intcc{a_k}{b_k}$ de term $u_n$ voor oneindig veel $n$, dan doet een van
zijn twee helften dat nog steeds --- noem die $\intcc{a_{k+1}}{b_{k+1}}$.
De rijen $(a_k)$ en $(b_k)$ zijn ingesloten ($b_k - a_k =
\frac{b-a}{2^k} \to 0$), met gemeenschappelijke limiet $\ell$
(\cref{thm:b1:seq:adjacent}).

Extraheren: kies $\varphi(0)$ met $u_{\varphi(0)} \in \intcc{a_0}{b_0}$,
en daarna, inductief, $\varphi(k+1) > \varphi(k)$ met $u_{\varphi(k+1)}
\in \intcc{a_{k+1}}{b_{k+1}}$ --- mogelijk omdat dat segment oneindig
veel termen bevat. Dan is $a_k \leq u_{\varphi(k)} \leq b_k$, en de
insluitstelling geeft $u_{\varphi(k)} \to \ell$.
\end{proof}

\begin{remark}[Wat Bolzano--Weierstrass wel en niet zegt]
Ze zegt \emph{wel}: uit louter begrensdheid convergeert een zekere
\omterm{def:b1:seq:subsequence}{deelrij} --- bestaan zonder formule, zoals het dichotomiebewijs duidelijk
maakt (niets vertelt ons \emph{welke} indices overleven). Ze zegt
\emph{niet} dat de limiet uniek is: $((-1)^n)$ heeft \omterm{def:b1:seq:subsequence}{deelrijen} die naar
$1$ en naar $-1$ convergeren, en de \omterm{def:b1:logic:sets}{verzameling} deelrijlimieten kan zelfs
oneindig zijn (\cref{ex:b1:seq:sublimits}, en de hele Cantorverzameling
in \cref{pb:b1:topology:1}). Ze overleeft de onbegrensdheid niet: $(n)$
heeft in het geheel geen convergente \omterm{def:b1:seq:subsequence}{deelrij} --- al kun je uit elke
onbegrensde rij altijd een \omterm{def:b1:seq:subsequence}{deelrij} extraheren die naar $+\infty$ of
$-\infty$ gaat (kies bijvoorbeeld $\varphi(k)$ met $u_{\varphi(k)} \geq
k$). Correct gebruikt is de stelling een \emph{bestaanspomp}: ze duikt op
in het hart van het criterium van Cauchy hieronder, van de stelling van
Heine en van de extremumstelling --- telkens om een punt te leveren dat
geen enkele expliciete constructie aanreikt.
\end{remark}

\section{Cauchyrijen en volledigheid}

\begin{definition}\label{def:b1:seq:cauchy}
Een rij $(u_n)$ is een \emph{Cauchyrij}\index{Cauchyrij} wanneer haar
termen willekeurig dicht \emph{bij elkaar} komen:
\[
\forall \varepsilon > 0,\ \exists N,\ \forall p, q \geq N,
\qquad \abs{u_p - u_q} \leq \varepsilon .
\]
\end{definition}

\begin{example}[De cauchy-eigenschap met de hand nagaan]\label{ex:b1:seq:cauchyhand}
Zij $u_n = \sum_{k=0}^{n} \frac{\cos k}{2^k}$ --- geen monotonie, geen
raadbare limiet. Voor $p > q$ is
\[
\abs{u_p - u_q}
= \Bigl| \sum_{k=q+1}^{p} \frac{\cos k}{2^k} \Bigr|
\leq \sum_{k=q+1}^{p} \frac{1}{2^k}
< \frac{1}{2^{q}} ,
\]
wegens de driehoeksongelijkheid, $\abs{\cos k} \leq 1$ en een eindige
meetkundige som. Kies bij gegeven $\varepsilon > 0$ een $N$ met $2^{-N}
\leq \varepsilon$: alle gaten voorbij $N$ zijn $\leq \varepsilon$, de rij
is een \omterm{def:b1:seq:cauchy}{Cauchyrij} en convergeert dus --- naar een limiet die niemand in
gesloten vorm kan benoemen, en dat is nu juist het punt. Het inzicht tot
slot: de aangroeiingen meetkundig overheersen is de standaardmanier om de
cauchy-eigenschap te verdienen, en \cref{ch:b1:series} zal dat argument
bottelen als ``absolute convergentie impliceert convergentie''.
\end{example}

\begin{theorem}[Volledigheid van $\R$]\label{thm:b1:seq:complete}
Een rij reële getallen convergeert dan en slechts dan als ze een
\omterm{def:b1:seq:cauchy}{Cauchyrij} is.\index{volledigheid van R@volledigheid van $\R$}
\end{theorem}

\begin{proof}
($\Rightarrow$) Is $u_n \to \ell$, dan is voorbij de drempel voor
$\frac\varepsilon2$ steeds $\abs{u_p - u_q} \leq \abs{u_p - \ell} +
\abs{\ell - u_q} \leq \varepsilon$.

($\Leftarrow$) Zij $(u_n)$ een \omterm{def:b1:seq:cauchy}{Cauchyrij}. \emph{Ze is begrensd}: met
$\varepsilon = 1$ liggen voorbij $N$ alle termen binnen $1$ van $u_N$, en
de kop is eindig. \emph{Extraheren}: volgens
\cref{thm:b1:seq:bw} convergeert een zekere \omterm{def:b1:seq:subsequence}{deelrij} $u_{\varphi(n)} \to
\ell$. \emph{Besluiten}: neem bij gegeven $\varepsilon > 0$ de drempel
$N$ (cauchy, voor $\frac\varepsilon2$) en een $n \geq N$ met
$\abs{u_{\varphi(n)} - \ell} \leq \frac\varepsilon2$ en $\varphi(n) \geq
N$; dan is voor elke $p \geq N$
\[
\abs{u_p - \ell} \leq \abs{u_p - u_{\varphi(n)}} +
\abs{u_{\varphi(n)} - \ell} \leq \varepsilon . \qedhere
\]
\end{proof}

\begin{remark}
De waarde van het criterium: het waarborgt convergentie \emph{zonder de
limiet te benoemen}. Het faalt over $\Q$ (de decimale afknottingen van
$\sqrt 2$ vormen een \omterm{def:b1:seq:cauchy}{Cauchyrij} rationale getallen zonder rationale
limiet): de volledigheid is een eigenschap van $\R$, equivalent met het
axioma van de \omterm{def:b1:reals:bounds}{bovengrens}. Het is bovendien het werkpaard achter de
convergentie van reeksen (\cref{ch:b1:series}).
\end{remark}

\begin{example}[Een Cauchyrij met een onzichtbare limiet]\label{ex:b1:seq:basel}
Zij $S_n = \sum_{k=1}^{n} \frac{1}{k^2}$. Voor $p > q \geq 1$ is
\[
S_p - S_q = \sum_{k=q+1}^{p} \frac{1}{k^2}
\leq \sum_{k=q+1}^{p} \frac{1}{k(k-1)}
= \sum_{k=q+1}^{p} \Bigl(\frac{1}{k-1} - \frac 1k\Bigr)
= \frac 1q - \frac 1p < \frac 1q ,
\]
dus zijn voorbij $N > \frac1\varepsilon$ alle gaten $\leq \varepsilon$:
$(S_n)$ is een \omterm{def:b1:seq:cauchy}{Cauchyrij} en convergeert dus. Merk op wat er zojuist
gebeurde: we bewezen dat een bepaald reëel getal bestaat zonder er ook
maar een naam voor te hebben. (Het is $\frac{\pi^2}{6}$ --- een beroemde
identiteit van Euler, bewezen in het volume van bachelorjaar 2; niets in
dit hoofdstuk kon ons dat vertellen.) Die taakverdeling --- eerst het
bestaan, later de identificatie, als die er al komt --- is de hele
bestaansreden van het criterium van Cauchy, en de motor van de theorie
van de reeksen in \cref{ch:b1:series}.
\end{example}

\section{Recurrente rijen}

\begin{method}[$u_{n+1} = f(u_n)$ bestuderen]\label{met:b1:seq:recurrent}
Gegeven $f$ en een beginpunt $u_0$:\index{recurrente rij}
\begin{enumerate}
  \item \emph{Stabiel \omterm{prop:b1:reals:intervals}{interval}:} zoek een \omterm{prop:b1:reals:intervals}{interval} $I$ met $f(I)
        \subseteq I$ dat $u_0$ bevat: dan liggen alle $u_n$ in $I$ (met
        inductie).
  \item \emph{Kandidaat-limieten:} is $u_n \to \ell \in I$ en is $f$
        continu in $\ell$ (\cref{ch:b1:continuity}), dan is $\ell$ een
        \emph{vast punt}: $f(\ell) = \ell$. Los $f(x) = x$ op.
  \item \emph{Monotonie:} is $f$ stijgend op $I$, dan is $(u_n)$
        monotoon (stijgend als $u_1 \geq u_0$, anders dalend); samen met
        de begrensdheid maakt \cref{thm:b1:seq:monotone} het af. Is $f$
        dalend, bestudeer dan de twee \omterm{def:b1:seq:subsequence}{deelrijen} $(u_{2n})$ en
        $(u_{2n+1})$, die monotoon zijn voor $f \circ f$.
  \item \emph{Foutbeheersing:} een ongelijkheid $\abs{f(x) - \ell} \leq
        k\abs{x - \ell}$ met $k < 1$ geeft rechtstreeks $\abs{u_n -
        \ell} \leq k^n \abs{u_0 - \ell} \to 0$.
\end{enumerate}
\end{method}

\begin{example}[De methode van Heron]\label{ex:b1:seq:heron}
Zij $u_0 = 2$ en $u_{n+1} = \dfrac12\Bigl(u_n + \dfrac{2}{u_n}\Bigr)$:
het aloude algoritme voor $\sqrt 2$.\index{methode van Heron}
\begin{itemize}
  \item \emph{Stabiliteit:} voor $x > 0$ geeft de ongelijkheid tussen
        het rekenkundig en het meetkundig gemiddelde $\frac12(x +
        \frac2x) \geq \sqrt{x \cdot \frac 2x} = \sqrt 2$; dus is $I =
        \intco{\sqrt 2}{+\infty}$ stabiel en bevat het $u_1$ (inderdaad
        is $u_1 = \frac32 \geq \sqrt 2$).
  \item \emph{Monotonie:} voor $x \geq \sqrt 2$ is $\;x - f(x) =
        \frac{x^2 - 2}{2x} \geq 0$: de rij daalt vanaf $u_1$ en is naar
        onderen begrensd door $\sqrt 2$: ze convergeert.
  \item \emph{Limiet:} de vaste punten voldoen aan $x = \frac12(x +
        \frac2x)$, dus $x^2 = 2$: op $I$ is $\ell = \sqrt 2$.
  \item \emph{Snelheid:} $u_{n+1} - \sqrt 2 = \frac{(u_n -
        \sqrt2)^2}{2u_n} \leq \frac{(u_n - \sqrt 2)^2}{2\sqrt 2}$: het
        aantal juiste cijfers \emph{verdubbelt} ruwweg bij elke stap
        (kwadratische convergentie).
\end{itemize}
\end{example}

\begin{remark}[Veelgemaakte fouten met limieten]
Vier klassiekers. (i) \emph{Kleine stappen impliceren geen convergentie}:
$u_{n+1} - u_n \to 0$ is veel zwakker dan de cauchy-eigenschap --- de
harmonische sommen $H_n$ hebben stappen $\frac{1}{n+1} \to 0$ en
divergeren toch naar $+\infty$ (\cref{exo:b1:seq:5}); de
cauchy-voorwaarde beheerst $\abs{u_p - u_q}$ voor \emph{alle} grote
paren, niet alleen voor opeenvolgende. (ii) \emph{Strikte ongelijkheden
sterven in de limiet}: uit $u_n < v_n$ voor alle $n$ volgt alleen $\lim
u_n \leq \lim v_n$ (\cref{thm:b1:seq:order}); $\frac1n > 0$ en toch is
$\lim = 0$. (iii) \emph{Begrensd is niet convergent}: $((-1)^n)$ is
begrensd en divergeert; begrensdheid plus \emph{monotonie} convergeert,
begrensdheid alleen waarborgt slechts een convergente \omterm{def:b1:seq:subsequence}{deelrij}
(\cref{thm:b1:seq:bw}). (iv) \emph{De vergelijking voor het vaste punt
komt tweede, niet eerste}: voor $u_{n+1} = f(u_n)$ identificeert het
oplossen van $f(\ell) = \ell$ de limiet \emph{pas nadat} de convergentie
bewezen is. De recursie $u_{n+1} = 2u_n$ heeft het unieke vaste punt
$\ell = 0$, en toch loopt de rij vanaf $u_0 = 1$ naar $+\infty$: de
vergelijking $\ell = 2\ell$ had nooit recht op een limiet. Altijd in deze
volgorde: eerst het bestaan (\cref{met:b1:seq:recurrent}, stappen 1--3),
dan de identificatie.
\end{remark}

\begin{example}[Een dalende $f$: de gulden recursie]\label{ex:b1:seq:golden}
Zij $u_0 = 1$ en $u_{n+1} = \dfrac{1}{1 + u_n}$. Hier is $f(x) =
\frac{1}{1+x}$ \emph{dalend}, zodat de rij niet monotoon is (ze wisselt
rond haar limiet); de contractiestap van \cref{met:b1:seq:recurrent} is
het juiste gereedschap. Stabiliteit: is $x \in \intcc{\frac12}{1}$, dan
is $1 + x \in \intcc{\frac32}{2}$, dus $f(x) \in
\intcc{\frac12}{\frac23} \subseteq \intcc{\frac12}{1}$, en $u_1 =
\frac12$ legt de hele rij daar. Vast punt: $\ell = \frac{1}{1+\ell}$ met
$\ell > 0$ geeft $\ell^2 + \ell - 1 = 0$, dus
\[
\ell = \frac{\sqrt5 - 1}{2} = 0.6180\dots
\]
(de omgekeerde van de gulden snede). Contractie: voor $x, y \in
\intcc{\frac12}{1}$ is
\[
\abs{f(x) - f(y)} = \frac{\abs{x - y}}{(1+x)(1+y)}
\leq \frac{\abs{x-y}}{(3/2)^2} = \frac49\,\abs{x - y} ,
\]
dus $\abs{u_n - \ell} \leq \bigl(\frac49\bigr)^{n-1}\abs{u_1 - \ell} \to
0$: convergentie, met meetkundige snelheid en zonder enige monotonie. Het
inzicht tot slot: de monotone methoden en de contractiemethoden verdelen
de wereld van de recursies onder elkaar --- een stijgende $f$ geeft
monotone banen, een dalende $f$ geeft alternerende banen die door een
lipschitzconstante $< 1$ getemd worden (de systematische theorie staat in
\cref{exo:b1:derivative:11}).
\end{example}

\begin{remark}[Perspectieven binnen dit volume]
Rijen zijn het meetinstrument dat de rest van het volume tegen elk object
aan houdt. In \cref{ch:b1:topology} \emph{karakteriseren} ze
geslotenheid en compactheid; in \cref{ch:b1:continuity} vervoeren ze
limieten van functies; in \cref{ch:b1:integration} zijn de Riemannsommen
rijen die naar de integraal convergeren; en \cref{ch:b1:series} \emph{is}
de theorie van één bijzondere klasse rijen, de partiële sommen. Zelfs de
algebrahoofdstukken verbruiken ze: de iteraties van een matrix in
\cref{ch:b1:matrices} vormen rijen waarvan het gedrag (de convergentie
van $A^n$) een vraag uit de lineaire algebra is met het vocabulaire van
dit hoofdstuk. De twee stellingen om overal mee te dragen: de monotone
limiet (bestaan uit ordening) en Bolzano--Weierstrass (bestaan uit
begrensdheid) --- samen brengen zij vrijwel elke limiet in dit boek
voort.
\end{remark}

\begin{remark}[Complexe rijen]
Een rij $(z_n)$ complexe getallen convergeert naar $\ell$ wanneer
$\abs{z_n - \ell} \to 0$; equivalent: wanneer $\Re(z_n) \to \Re(\ell)$ en
$\Im(z_n) \to \Im(\ell)$ (vergelijk $\abs{z}$ met $\abs{\Re z} + \abs{\Im
z}$). De stellingen waarin geen ordening voorkomt --- bewerkingen,
Bolzano--Weierstrass (tweemaal extraheren), het criterium van Cauchy ---
gaan letterlijk over.
\end{remark}

\section{Opgave: Cesàro, Stolz, en de trage val van de sinus}

\begin{problem}[{Weekendopgave --- de stelling van Cesàro--Stolz en de
asymptotiek $u_n \sim \sqrt{3/n}$ voor $u_{n+1} = \sin u_n$}]
\label{pb:b1:seq:1}
De stelling van Cesàro--Stolz is de discrete regel van l'Hospital: om de
limiet van een quotiënt $a_n/b_n$ te vinden volstaat het de limiet van het
quotiënt van de \emph{aangroeiingen} $(a_{n+1} - a_n)/(b_{n+1} - b_n)$ te
vinden. Deze opgave bewijst de stelling, oogst er klassieke limieten mee,
en richt haar dan op een beroemd doelwit: de rij $u_{n+1} = \sin u_n$,
die naar $0$ kruipt met de exact berekenbare snelheid $u_n \sim
\sqrt{3/n}$.\index{stelling van Cesàro--Stolz} Twee feiten uit het
bovenbouwvolume worden hier toegestaan en later in dit volume eerlijk
opnieuw bewezen: de raaklijnongelijkheid
\[
\tag{G1} \eu^{u} \geq 1 + u \quad (u \in \R),
\]
opnieuw bewezen met convexiteit in \cref{ch:b1:derivative}, en de
insluiting van de sinus
\[
\tag{G2} x - \frac{x^3}{6} \;\leq\; \sin x \;\leq\; x -
\frac{x^3}{6} + \frac{x^5}{120} \quad (0 \leq x \leq 1), \qquad
\abs{\sin x} \leq \abs{x} \quad (x \in \R),
\]
opnieuw bewezen met de formule van Taylor in \cref{ch:b1:taylor}.

\medskip
\noindent\textbf{Deel I --- Sommen zonder gesloten formule.}
\begin{enumerate}
  \item Bereken met $1 + 2 + \dots + n = \frac{n(n+1)}{2}$ en $1^2 +
        \dots + n^2 = \frac{n(n+1)(2n+1)}{6}$ de limieten $\lim \frac{1
        + 2 + \dots + n}{n^2}$ en $\lim \frac{1^2 + \dots + n^2}{n^3}$.
  \item Zij $T_n = \sum_{k=1}^n \sqrt k$, waarvoor geen gesloten formule
        bestaat. Bewijs de insluiting
        \[
        \frac{1}{2\sqrt 2}\,n^{3/2} \;\leq\; T_n \;\leq\;
        n^{3/2}
        \]
        \emph{(houd voor de ondergrens alleen de termen $k > \frac n2$
        over)}. Dus heeft $T_n$ de orde $n^{3/2}$ --- maar met welke
        constante? Houd de vraag vast tot vraag 8.
  \item (Telescoperingslemma) Zij $(b_n)$ strikt stijgend en stel dat
        voor alle $k \geq N$ geldt
        \[
        m \;\leq\; \frac{a_{k+1} - a_k}{b_{k+1} - b_k} \;\leq\;
        M .
        \]
        Bewijs dat $m \leq \dfrac{a_n - a_N}{b_n - b_N} \leq M$ voor
        elke $n > N$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De stelling van Cesàro--Stolz.} Zij $(b_n)$
strikt stijgend met $b_n \to +\infty$, en stel dat $\dfrac{a_{n+1} -
a_n}{b_{n+1} - b_n} \to \ell \in \R$.
\begin{enumerate}[resume]
  \item Leg $\varepsilon > 0$ vast. Toon aan dat er een $N$ is met $\ell
        - \varepsilon \leq \dfrac{a_n - a_N}{b_n - b_N} \leq \ell +
        \varepsilon$ voor alle $n > N$.
  \item Stel voor $n > N$ de identiteit
        \[
        \frac{a_n}{b_n} - \ell
        = \frac{a_N - \ell\,b_N}{b_n}
        + \Bigl(1 - \frac{b_N}{b_n}\Bigr)
        \Bigl(\frac{a_n - a_N}{b_n - b_N} - \ell\Bigr)
        \]
        vast, en besluit de stelling: $\dfrac{a_n}{b_n} \to \ell$.
  \item Bewijs de variant met $+\infty$: gaat $\dfrac{a_{n+1} -
        a_n}{b_{n+1} - b_n} \to +\infty$ (met dezelfde hypothesen over
        $(b_n)$), dan gaat $\dfrac{a_n}{b_n} \to +\infty$.
  \item Neem $b_n = n$: haal de stelling over de gemiddelden van Cesàro
        uit \cref{exo:b1:seq:10} terug. Toon daarna aan dat de omkering
        van Cesàro--Stolz faalt: voor $a_n = (-1)^n$ en $b_n = n$
        convergeert het quotiënt $a_n/b_n$ terwijl het quotiënt van de
        aangroeiingen dat niet doet. Stolz is eenrichtingsverkeer.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Eerste dividenden.}
\begin{enumerate}[resume]
  \item Bewijs $(1+h)^{3/2} - 1 = \dfrac{3h + 3h^2 + h^3}{(1+h)^{3/2} +
        1}$ met de toegevoegde uitdrukking, leid af dat $n\bigl((1 +
        \tfrac1n)^{3/2} - 1\bigr) \to \tfrac32$, en besluit met
        Cesàro--Stolz:
        \[
        T_n = \sum_{k=1}^{n} \sqrt k \;\sim\; \tfrac23\,
        n^{3/2} ,
        \]
        waarmee de open vraag van vraag 2 beslecht is.
  \item Leid uit (G1) alleen de insluiting van de logaritme af:
        \[
        \frac{t}{1 + t} \;\leq\; \ln(1 + t) \;\leq\; t
        \qquad (t > -1)
        \]
        \emph{(pas (G1) toe in $u = \ln(1+t)$ en in $u = -t/(1+t)$)}.
  \item Toon aan dat $b_n = \ln n$ strikt stijgend is met $\ln n \to
        +\infty$, en bewijs met Cesàro--Stolz en vraag 9 dat
        \[
        H_n = \sum_{k=1}^{n} \frac 1k \;\sim\; \ln n .
        \]
        (De fijnere structuur $H_n = \ln n + \gamma + o(1)$ is de
        weekendopgave van \cref{ch:b1:series}.)
  \item (Van verhoudingen naar wortels) Zij $u_n > 0$ met
        $\frac{u_{n+1}}{u_n} \to L > 0$. Toon met vraag 9 aan dat
        $\ln\frac{u_{n+1}}{u_n} \to \ln L$; pas Cesàro toe om te
        besluiten dat $\frac{\ln u_n}{n} \to \ln L$, en daarna, met
        (G1), dat $u_n^{1/n} \to L$. Toepassing: bereken
        $\lim\,\binom{2n}{n}^{1/n}$.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De trage val van de sinus.} Zij $u_0 \in \R$
en $u_{n+1} = \sin u_n$.
\begin{enumerate}[resume]
  \item Toon met (G2) aan dat $0 < \sin x < x$ voor $0 < x \leq 1$. Leid
        af: $u_1 \in \intcc{-1}{1}$; is $u_1 = 0$, dan is de rij nul
        vanaf rang $1$; en is $u_1 > 0$ (het geval $u_1 < 0$ verloopt
        symmetrisch, want $\sin$ is oneven), dan is $(u_n)_{n \geq 1}$
        strikt dalend en positief en convergeert ze naar $0$
        \emph{(identificeer de limiet met $\ell = \sin \ell$, via
        $\abs{\sin a - \sin b} \leq \abs{a - b}$, zelf een gevolg van
        (G2) en de formule van product naar som)}.
  \item Neem vanaf nu $u_1 \in \intoc{0}{1}$ aan. Toon met (G2) door
        insluiting aan:
        \[
        \frac{\sin u_n}{u_n} \to 1
        \qquad\text{en}\qquad
        \frac{u_n - \sin u_n}{u_n^{3}} \to \frac16 .
        \]
  \item Bewijs de ontbinding
        \[
        w_n := \frac{1}{u_{n+1}^{2}} - \frac{1}{u_n^{2}}
        = \frac{u_n - \sin u_n}{u_n^{3}} \cdot
        \frac{u_n + \sin u_n}{u_n} \cdot
        \Bigl(\frac{u_n}{\sin u_n}\Bigr)^{2},
        \]
        en leid af dat $w_n \to \frac13$.
  \item Besluit met \cref{exo:b1:seq:10} (de versie met aangroeiingen)
        dat $\frac{1}{n\,u_n^{2}} \to \frac13$, en daarna, met een
        argument met de toegevoegde uitdrukking voor de vierkantswortel,
        het hoofdresultaat:
        \[
        \sqrt n\;u_n \longrightarrow \sqrt 3 ,
        \qquad\text{dus}\qquad
        u_n \sim \sqrt{\frac 3n} .
        \]
  \item Kwantificeer de traagheid: toon aan dat uiteindelijk
        $\sqrt{2/n} \leq u_n \leq 2/\sqrt n$, zodat $u_n \leq 10^{-2}$
        bereiken meer dan $20\,000$ iteraties vergt (ongeveer
        $30\,000$, volgens de asymptotiek). Zet dat af tegen de \omterm{ex:b1:seq:heron}{methode
        van Heron} (\cref{ex:b1:seq:heron}) en leg de structurele reden
        uit: in het vaste punt $0$ is de richtingscoëfficiënt van $\sin$
        gelijk aan $1$ (een \emph{neutraal} vast punt), terwijl
        iteraties die de fout halveren een richtingscoëfficiënt met
        \omterm{def:b1:complex:field}{modulus} $< 1$ nodig hebben.
  \item Toon aan dat voor \emph{elk} beginpunt $u_0 \in \R$ ofwel $u_n =
        0$ vanaf rang $1$, ofwel $\abs{u_n} \sim \sqrt{3/n}$ --- de val
        is universeel, alleen het teken herinnert zich $u_0$.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Het algemene beginsel.} De sinus is één geval
van een machine.
\begin{enumerate}[resume]
  \item Zij $u_n > 0$ met $u_n \to 0$ en $\dfrac{u_n - u_{n+1}}{u_n^{2}}
        \to a > 0$. Bewijs achtereenvolgens: $\frac{u_{n+1}}{u_n} \to
        1$; dan $\frac{1}{u_{n+1}} - \frac{1}{u_n} \to a$; dan $n\,u_n
        \to \frac1a$.
  \item (Exact model) Toon voor $u_{n+1} = \dfrac{u_n}{1 + u_n}$ met
        $u_0 > 0$ aan dat $\frac{1}{u_n}$ rekenkundig is, los exact op,
        en toets de conclusie van vraag 18 aan de exacte formule.
  \item Toon voor $u_{n+1} = u_n \eu^{-u_n}$ met $u_0 > 0$ aan dat $u_n
        \to 0$, gebruik (G1) om $\frac{1 - \eu^{-t}}{t}$ tussen
        $\frac{1}{1+t}$ en $1$ in te sluiten voor $t > 0$, en besluit
        $u_n \sim \frac 1n$.
  \item (Derdegraadscontact, gekwadrateerde telescoop) Zij $u_n > 0$ met
        $u_n \to 0$ en $\dfrac{u_n - u_{n+1}}{u_n^{3}} \to a > 0$. Pas
        de ontbinding van vraag 14 aan om aan te tonen dat
        $\frac{1}{u_{n+1}^2} - \frac{1}{u_n^2} \to 2a$, en besluit
        $n\,u_n^{2} \to \frac{1}{2a}$. Ga na dat $a = \frac16$ Deel IV
        teruggeeft.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Grenzen van de methode, en morelen.}
\begin{enumerate}[resume]
  \item Toon aan dat de hypothese $b_n \to +\infty$ niet weggelaten mag
        worden: voor $a_n = 2 - 2^{-n}$ en $b_n = 1 - 2^{-n}$ gaat het
        quotiënt van de aangroeiingen naar $1$ terwijl $\frac{a_n}{b_n}
        \to 2$. Wijs de precieze regel in het bewijs van vraag 5 aan die
        het begeeft.
  \item (Tweemaal Stolz) Bewijs $\sum_{k=1}^{n} H_k \sim n \ln n$
        \emph{(één toepassing van Cesàro--Stolz, dan vraag 10; schat
        $(n+1)\ln(n+1) - n\ln n$ af met vraag 9)}.
  \item (Meetkundige gemiddelden) Toon aan dat uit $u_n > 0$ en $u_n \to
        \ell > 0$ volgt $(u_1 u_2 \cdots u_n)^{1/n} \to \ell$; en dat
        uit $u_n \to +\infty$ volgt $(u_1 \cdots u_n)^{1/n} \to
        +\infty$. Leid af dat $(n!)^{1/n} \to +\infty$.
  \item Synthese, één zin per onderdeel: (i) waar kwam de volledigheid
        precies in deze opgave binnen; (ii) in welke zin is
        Cesàro--Stolz een discrete regel van l'Hospital (haar
        differentiële tweelingzus rust op de middelwaardestelling van
        \cref{ch:b1:derivative}); (iii) formuleer de vuistregel die de
        contactorde van $f$ in een neutraal vast punt verbindt met de
        afname-exponent van $u_{n+1} = f(u_n)$; (iv) volg de constante
        $3$ uit $\sqrt{3/n}$ terug door de keten $\frac16 \to \frac13
        \to 3$.
\end{enumerate}
\end{problem}
