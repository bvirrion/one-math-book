\chapter{Veeltermen}\label{ch:b1:poly}

\omterm{def:b1:poly:def}{Veeltermen} zijn de lievelingsfuncties van de algebraïcus --- behalve dat
ze hier niet als functies behandeld worden, maar als formele
uitdrukkingen in een onbepaalde $X$, opgeteld en vermenigvuldigd volgens
de regels van een commutatieve \omterm{def:b1:structures:ring}{ring}. De theorie loopt opvallend parallel
met \cref{ch:b1:arith}: een euclidische deling, een ggd met betrekkingen
van Bézout, irreducibele elementen en een unieke ontbinding. Overal
noteert $K$ hetzij $\Q$, hetzij $\R$, hetzij $\C$.

\section{De ring $K[X]$}

\begin{definition}[Veelterm, graad]\label{def:b1:poly:def}
Een \emph{veelterm}\index{veelterm} met coëfficiënten in $K$ is een
formele som
\[
P = a_0 + a_1 X + a_2 X^2 + \dots + a_n X^n
= \sum_{k} a_k X^k,
\]
met $a_k \in K$ die vanaf een zekere index alle nul zijn. Met de
natuurlijke optelling en het product
\[
\Bigl(\sum_i a_i X^i\Bigr)\Bigl(\sum_j b_j X^j\Bigr)
= \sum_k \Bigl(\sum_{i+j=k} a_i b_j\Bigr) X^k
\]
is de \omterm{def:b1:logic:sets}{verzameling} $K[X]$ een commutatieve \omterm{def:b1:structures:ring}{ring}. De \emph{graad} $\deg P$
van $P \neq 0$ is de grootste $n$ met $a_n \neq 0$; $a_n$ is de
\emph{kopcoëfficiënt} ($P$ heet \emph{monisch}\index{monische veelterm}
wanneer $a_n = 1$), en per afspraak is $\deg 0 = -\infty$. Elke veelterm
definieert door substitutie een functie $x \mapsto P(x)$ op $K$.
\end{definition}

\begin{proposition}[Graadregels; integriteitsdomein]\label{prop:b1:poly:degree}
Voor $P, Q \in K[X]$ geldt
\[
\deg(P + Q) \leq \max(\deg P, \deg Q),
\qquad
\deg(PQ) = \deg P + \deg Q .
\]
Bijgevolg is $K[X]$ een \omterm{def:b1:structures:field}{integriteitsdomein}, en zijn de eenheden ervan de
constanten ongelijk aan nul.
\end{proposition}

\begin{proof}
De somregel is duidelijk (voorbij het maximum verdwijnen de
coëfficiënten). Voor het product: zij $a_m$ en $b_n$ de
kopcoëfficiënten, dan is de coëfficiënt van $X^{m+n}$ in $PQ$ gelijk aan
$a_m b_n \neq 0$ ($K$ is een \omterm{def:b1:structures:field}{lichaam} en dus een \omterm{def:b1:structures:field}{integriteitsdomein}), en
alle hogere coëfficiënten verdwijnen. Zijn $P, Q \neq 0$, dan is $\deg PQ
= \deg P + \deg Q \geq 0$, dus $PQ \neq 0$: een \omterm{def:b1:structures:field}{integriteitsdomein}. Is
$PQ = 1$, dan dwingt $\deg P + \deg Q = 0$ af dat $\deg P = \deg Q = 0$:
de inverteerbare elementen zijn de inverteerbare constanten, dus heel
$K^*$.
\end{proof}

\begin{theorem}[Euclidische deling]\label{thm:b1:poly:division}
Zij $A, B \in K[X]$ met $B \neq 0$. Er is precies één paar $(Q, R)$
\omterm{def:b1:poly:def}{veeltermen} met
\[
A = BQ + R, \qquad \deg R < \deg B .
\]
\end{theorem}

\begin{proof}
\emph{Bestaan}, met sterke inductie naar $\deg A$. Is $\deg A < \deg B$,
neem dan $(Q, R) = (0, A)$. Anders schrijf je $A = a X^m + \dots$ en $B =
b X^n + \dots$ met $m \geq n$; de \omterm{def:b1:poly:def}{veelterm} $A_1 = A - \frac ab X^{m-n} B$
heeft graad $< m$ (de kopterme heffen elkaar op), dus is volgens de
inductie $A_1 = BQ_1 + R$ met $\deg R < \deg B$, en $A = B(Q_1 + \frac ab
X^{m-n}) + R$.

\emph{Uniciteit:} is $BQ + R = BQ' + R'$, dan is $B(Q - Q') = R' - R$ met
$\deg(R' - R) < \deg B$; de graadregel dwingt dan $Q - Q' = 0$ af, waarna
$R = R'$.
\end{proof}

\begin{example}\label{ex:b1:poly:division}
Deel $A = X^4 + X^3 - 2X + 1$ door $B = X^2 + 1$:
\[
X^4 + X^3 - 2X + 1 = (X^2 + 1)(X^2 + X - 1) + (-3X + 2).
\]
(Reken na: trek $X^2 B$ af, dan $X B$, dan $-B$; de rest $-3X + 2$ heeft
graad $1 < 2$.)
\end{example}

\begin{method}[Het schema van Horner]\label{met:b1:poly:horner}
Om $P = a_nX^n + \dots + a_0$ in $x$ te evalueren, of om $P$ door $X - x$
te delen, vermijd je het berekenen van machten: lees de coëfficiënten van
links naar rechts en herhaal \emph{vermenigvuldig met $x$, tel de
volgende coëfficiënt op}:
\[
b_n = a_n, \qquad b_{k} = a_{k} + x\,b_{k+1}
\quad (k = n-1, \dots, 0) .
\]
Dan is $b_0 = P(x)$, en de eerdere $b_k$ zijn de coëfficiënten van het
quotiënt: $P = (X - x)(b_nX^{n-1} + \dots + b_1) + b_0$ (werk uit en
vergelijk). Voorbeeld: $P = X^4 - 5X^3 + 6X^2 + 4X - 8$ in $x = 2$: de
$b$'s zijn $1, -3, 0, 4, 0$, dus $P(2) = 0$ en $P = (X-2)(X^3 - 3X^2 +
4)$ --- één regel in plaats van een staartdeling, en $n$
vermenigvuldigingen in plaats van de $\approx n^2/2$ van de naïeve
evaluatie. Het schema in hetzelfde punt herhalen haalt de
\omterm{def:b1:poly:derivative}{multipliciteiten} naar boven (vergelijk \cref{ex:b1:poly:multexample}).
\end{method}

\begin{remark}[{De rekenkunde van $K[X]$}]
Met de euclidische deling in handen draagt de hele rekenkunde van
\cref{ch:b1:arith} over naar $K[X]$, met dezelfde bewijzen en met de
graad in de rol van de absolute waarde: ggd (genormaliseerd tot een
\omterm{def:b1:poly:def}{monische veelterm}), uitgebreid \omterm{met:b1:arith:euclid}{algoritme van Euclides}, identiteit van
Bézout, lemma van Gauss, irreducibele \omterm{def:b1:poly:def}{veeltermen} en unieke ontbinding. We
gebruiken die overgedragen resultaten vrijelijk, en
\cref{exo:b1:poly:6} oefent er één van in.
\end{remark}

\section{Wortels}

\begin{theorem}[Factorstelling]\label{thm:b1:poly:factor}
Zij $P \in K[X]$ en $a \in K$. De rest van $P$ bij deling door $X - a$ is
de constante $P(a)$. In het bijzonder geldt
\[
P(a) = 0 \iff (X - a) \mid P .
\]
Algemener leveren verschillende wortels $a_1, \dots, a_r$ van $P$ de
ontbinding $P = (X - a_1)\cdots(X - a_r)\, Q$ op.
\end{theorem}

\begin{proof}
Deel: $P = (X - a) Q + R$ met $\deg R < 1$, zodat $R$ een constante $c$
is; substitutie van $X = a$ (substitueren respecteert sommen en producten)
geeft $P(a) = c$. De equivalentie volgt. Voor meerdere wortels voer je
inductie naar $r$: het geval $r = 1$ is de zojuist bewezen equivalentie.
Neem de \omterm{def:b1:logic:statement}{uitspraak} aan voor $r - 1$ wortels en zij $a_1, \dots, a_r$
verschillende wortels van $P$. Schrijf $P = (X - a_1)Q_1$; substitutie van
$a_i$ voor elke $i \geq 2$ geeft
\[
0 = P(a_i) = (a_i - a_1)\,Q_1(a_i),
\qquad a_i - a_1 \neq 0 ,
\]
en omdat $K$ geen nuldelers heeft, is $Q_1(a_i) = 0$: de $r - 1$
verschillende punten $a_2, \dots, a_r$ zijn wortels van $Q_1$. De
inductiehypothese ontbindt $Q_1 = (X - a_2)\cdots(X - a_r)\,Q$, en
terugsubstitueren geeft de bewering.
\end{proof}

\begin{corollary}[Een veelterm van graad $n$ heeft hoogstens $n$ wortels]\label{cor:b1:poly:nroots}
Een $P \in K[X]$ ongelijk aan nul en van graad $n$ heeft hoogstens $n$
verschillende wortels in $K$. Bijgevolg is een \omterm{def:b1:poly:def}{veelterm} (van graad $\leq
n$) die in $n + 1$ verschillende punten nul is, de nulveelterm, en zijn
twee \omterm{def:b1:poly:def}{veeltermen} van graad $\leq n$ die in $n+1$ punten overeenstemmen
gelijk.
\end{corollary}

\begin{proof}
Zijn $a_1, \dots, a_r$ verschillende wortels, dan geeft
\cref{thm:b1:poly:factor} dat $P = (X-a_1)\cdots(X-a_r) Q$, zodat $n =
\deg P \geq r$. De twee gevolgen volgen uit het ongerijmde,
respectievelijk door het verschil te beschouwen.
\end{proof}

\begin{example}[De truc met de hulpveelterm]\label{ex:b1:poly:auxiliary}
Zij $P$ de \omterm{def:b1:poly:def}{veelterm} van graad $\leq n$ met
\[
P(k) = \frac{k}{k+1} \qquad (k = 0, 1, \dots, n) ;
\]
hij bestaat en is uniek wegens de \omterm{thm:b1:poly:lagrange}{Lagrange-interpolatie} hieronder. Wat is
$P(n+1)$? Werk de noemers weg: de \omterm{def:b1:poly:def}{veelterm} $Q = (X+1)P - X$ heeft graad
$\leq n + 1$ en is nul in de $n + 1$ punten $0, 1, \dots, n$, dus volgens
\cref{thm:b1:poly:factor} is
\[
Q = c\,X(X-1)(X-2)\cdots(X-n)
\]
voor een zekere constante $c$. Evalueer waar $Q$ onafhankelijk bekend is:
in $X = -1$ is $Q(-1) = 0 \cdot P(-1) + 1 = 1$, terwijl het product
$(-1)(-2)\cdots(-1-n) = (-1)^{n+1}(n+1)!$ oplevert; dus $c =
\frac{(-1)^{n+1}}{(n+1)!}$. Evalueer nu in $X = n + 1$:
\[
(n+2)\,P(n+1) - (n+1) = Q(n+1) = c\,(n+1)! = (-1)^{n+1} ,
\]
zodat $P(n+1) = \dfrac{(n+1) + (-1)^{n+1}}{n+2}$: gelijk aan $1$ voor
oneven $n$ en aan $\frac{n}{n+2}$ voor even $n$ --- de interpolerende
\omterm{def:b1:poly:def}{veelterm} zet het patroon $\frac{n+1}{n+2}$ dus \emph{niet} voort. De truc
om te onthouden: codeer de gegevens als wortels van een hulpveelterm,
bepaal de onbekende constante in een punt buiten de gegevens, en oogst.
\end{example}

\begin{definition}[Afgeleide, multipliciteit]\label{def:b1:poly:derivative}
De \emph{formele afgeleide} van $P = \sum a_k X^k$ is $P' = \sum_{k \geq
1} k\,a_k X^{k-1}$; ze voldoet aan de gewone regels $(P+Q)' = P' + Q'$ en
$(PQ)' = P'Q + PQ'$ (na te gaan op monomen en uit te breiden met de
lineariteit). Een wortel $a$ van $P$ heeft
\emph{multipliciteit}\index{multipliciteit van een wortel} $m \geq 1$
wanneer $(X-a)^m \mid P$ maar $(X-a)^{m+1} \nmid P$; de wortel heet
\emph{enkelvoudig} als $m = 1$ en \emph{meervoudig} als $m \geq 2$.
\end{definition}

\begin{proposition}[Multipliciteit via afgeleiden]\label{prop:b1:poly:multiplicity}
$a$ is een wortel van $P$ van \omterm{def:b1:poly:derivative}{multipliciteit} $\geq m$ dan en slechts dan
als
\[
P(a) = P'(a) = \dots = P^{(m-1)}(a) = 0 .
\]
In het bijzonder is $a$ een meervoudige wortel van $P$ dan en slechts dan
als $P(a) = P'(a) = 0$.
\end{proposition}

\begin{proof}
Schrijf $P = (X - a)^m Q + R$, waarbij $R$ de rest van de deling door
$(X-a)^m$ is, met $\deg R < m$. Differentiëren we $k \leq m - 1$ maal en
evalueren we in $a$, dan draagt de eerste term $0$ bij (elke afgeleide
houdt een factor $(X-a)$ over), zodat $P^{(k)}(a) = R^{(k)}(a)$.

Nu ligt een \omterm{def:b1:poly:def}{veelterm} $R$ van graad $< m$ vast door $R(a), R'(a), \dots,
R^{(m-1)}(a)$: schrijf $R = \sum_{k < m} c_k (X - a)^k$ (dat kan: werk de
machten van $X = (X - a) + a$ uit), dan vind je $R^{(k)}(a) = k!\, c_k$.
Bijgevolg: alle $P^{(k)}(a) = 0$ voor $k < m$ $\iff$ alle $c_k = 0$
$\iff$ $R = 0$ $\iff$ $(X-a)^m \mid P$.
\end{proof}

\begin{example}[Een multipliciteit berekenen]\label{ex:b1:poly:multexample}
Wat is de \omterm{def:b1:poly:derivative}{multipliciteit} van de wortel $2$ in $P = X^4 - 5X^3 + 6X^2 + 4X
- 8$? Evalueer de opeenvolgende afgeleiden in $2$:
\[
P(2) = 16 - 40 + 24 + 8 - 8 = 0, \qquad
P'(2) = 32 - 60 + 24 + 4 = 0,
\]
\[
P''(2) = 48 - 60 + 12 = 0, \qquad
P'''(2) = 48 - 30 = 18 \neq 0
\]
(met $P' = 4X^3 - 15X^2 + 12X + 4$, $P'' = 12X^2 - 30X + 12$ en $P''' =
24X - 30$). Drie waarden nul en dan een waarde ongelijk aan nul:
\omterm{def:b1:poly:derivative}{multipliciteit} precies $3$. Delen geeft $P = (X - 2)^3(X + 1)$ --- wat je
nagaat door $(X-2)^3 = X^3 - 6X^2 + 12X - 8$ uit te werken en met $X + 1$
te vermenigvuldigen. Het inzicht: \omterm{def:b1:poly:derivative}{multipliciteiten} lees je af aan
\emph{evaluaties}, zonder ontbinding --- en precies zo spoor je ze op
wanneer de ontbinding buiten bereik ligt.
\end{example}

\begin{example}[Meervoudige wortels opsporen met een ggd]\label{ex:b1:poly:gcdmultiple}
Ook zonder één bekende wortel levert
\cref{prop:b1:poly:multiplicity} nog een \emph{globale} detector van
meervoudige wortels: $a$ is een meervoudige wortel van $P$ precies wanneer
het een gemeenschappelijke wortel van $P$ en $P'$ is, zodat $P$ een
meervoudige wortel (in $\C$) heeft precies wanneer $\gcd(P, P') \neq 1$
--- berekenbaar met het \omterm{met:b1:arith:euclid}{algoritme van Euclides} zonder ook maar iets op te
lossen. Voorbeeld: $P = X^3 - 3X + 2$ met $P' = 3X^2 - 3 = 3(X - 1)(X +
1)$. De wortels $\pm1$ van $P'$ in $P$ toetsen: $P(1) = 0$ maar $P(-1) =
4$, dus
\[
\gcd(P, P') = X - 1 :
\]
de wortel $1$ is meervoudig; tweemaal delen geeft $P = (X - 1)^2(X + 2)$.
De ggd meldt zelfs de volledige \omterm{def:b1:logic:sets}{verzameling} meervoudige wortels, elk met
een \omterm{def:b1:poly:derivative}{multipliciteit} die één lager ligt --- het feit dat elk
computeralgebrasysteem uitbuit om ``kwadraatvrij te ontbinden'' vóór er
naar wortels gejaagd wordt, en de veeltermtegenhanger van de argumenten
zonder meervoudige wortels in \cref{exo:b1:poly:9}.
\end{example}

\begin{theorem}[Hoofdstelling van de algebra]\label{thm:b1:poly:fta}
Elke niet-constante \omterm{def:b1:poly:def}{veelterm} van $\C[X]$ heeft een wortel in
$\C$.\index{hoofdstelling van de algebra}
\end{theorem}
\admitted

\begin{remark}
Ondanks haar naam is de stelling een \omterm{def:b1:logic:statement}{uitspraak} uit de \emph{analyse}:
elk bekend bewijs gebruikt de volledigheid van $\R$ in een of andere
vorm, en geen enkel is zuiver algebraïsch --- het eerlijke bewijs wordt
gegeven in het volume van bachelorjaar 3, zodra ofwel complexe integratie
ofwel compactheidsargumenten beschikbaar zijn. Wat dit hoofdstuk
werkelijk bewijst, is de \emph{herleiding}: gegeven één wortel voor elke
niet-constante \omterm{def:b1:poly:def}{veelterm}, volgen de volledige ontbindingen over $\C$ en $\R$
hieronder met zuivere algebra.
\end{remark}

\begin{corollary}[Ontbinding over $\C$ en over $\R$]\label{cor:b1:poly:factorization}
\begin{enumerate}
  \item Elke $P \in \C[X]$ ongelijk aan nul ontbindt als
        \[
        P = c\, (X - a_1)^{m_1} \cdots (X - a_r)^{m_r},
        \]
        met $c$ de kopcoëfficiënt, $a_i$ de verschillende complexe
        wortels en $\sum m_i = \deg P$: \emph{met \omterm{def:b1:poly:derivative}{multipliciteit} geteld
        heeft een \omterm{def:b1:poly:def}{veelterm} van graad $n$ precies $n$ complexe wortels}.
  \item Elke $P \in \R[X]$ ongelijk aan nul ontbindt over $\R$ als
        \[
        P = c \prod_i (X - a_i)^{m_i} \prod_j (X^2 + p_j X +
        q_j)^{n_j},
        \]
        waarbij de kwadratische factoren verschillend zijn met $p_j^2 -
        4q_j < 0$ (geen reële wortels).
\end{enumerate}
\end{corollary}

\begin{proof}
(1) Inductie naar de graad, waarbij je telkens één wortel afsplitst met
\cref{thm:b1:poly:factor}; de graden kloppen bij elke stap.

(2) Zij $P$ met reële coëfficiënten. Is $z$ een complexe wortel van
\omterm{def:b1:poly:derivative}{multipliciteit} $m$, dan is $\conj z$ dat ook: toevoegen in $P(z) = 0$
geeft $P(\conj z) = \conj{P(z)} = 0$ (de coëfficiënten zijn hun eigen
toegevoegde), en hetzelfde geldt voor de afgeleiden
(\cref{prop:b1:poly:multiplicity}). Groepeer de niet-reële wortels in
toegevoegde paren: elk paar draagt
\[
(X - z)(X - \conj z) = X^2 - 2\Re(z)\, X + \abs z^2
\]
bij, een reële tweedegraadsveelterm met negatieve discriminant. De reële
wortels leveren de lineaire factoren.
\end{proof}

\begin{example}\label{ex:b1:poly:factorexample}
$X^4 + 4$ werd in \cref{exo:b1:complex:5} over $\R$ ontbonden door de
vier complexe wortels $\pm 1 \pm \iu$ te koppelen: $X^4 + 4 = (X^2 - 2X +
2)(X^2 + 2X + 2)$. Geen van beide kwadratische factoren splitst over $\R$
(discriminanten $-4$). Merk op: een \emph{irreducibele} reële \omterm{def:b1:poly:def}{veelterm}
heeft graad $1$ of $2$ --- en dat is precies wat de ontbindingsstelling
zegt. Hetzelfde koppelen van toegevoegde wortels toegepast op $X^4 + 1$,
waarvan de wortels $\eu^{\pm\iu\pi/4}$ en $\eu^{\pm3\iu\pi/4}$ zijn: elk
paar draagt $X^2 - 2\cos\theta\,X + 1$ bij, dus
\[
X^4 + 1 = \bigl(X^2 - \sqrt2\,X + 1\bigr)
\bigl(X^2 + \sqrt2\,X + 1\bigr) ,
\]
een identiteit die onzichtbaar blijft voor naïeve ontbindingspogingen
over $\Q$ --- de prijs van het aandringen op reële (hier zelfs
irrationale) coëfficiënten, en een standaardingrediënt voor het
integreren van $\frac1{x^4 + 1}$ in \cref{ch:b1:integration}.
\end{example}

\begin{remark}[Veelgemaakte fouten met veeltermen]\label{rem:b1:poly:pitfalls}
\begin{enumerate}
  \item \emph{\omterm{def:b1:poly:def}{Veelterm} tegenover functie.} Over $K = \Q, \R, \C$ vallen
        beide begrippen samen (gelijke functies hebben gelijke
        coëfficiënten, wegens \cref{cor:b1:poly:nroots} en de
        oneindigheid van $K$), maar begripsmatig \emph{is} een \omterm{def:b1:poly:def}{veelterm}
        zijn lijst coëfficiënten: over het \omterm{def:b1:structures:field}{lichaam} $\Z/2\Z$ met twee
        elementen uit \cref{ch:b1:structures} is $X^2 + X$ nul in beide
        punten, en toch niet de nulveelterm.
  \item \emph{Graden bij optelling.} $\deg(P + Q)$ kan onder $\max(\deg
        P, \deg Q)$ zakken wanneer de koptermen elkaar opheffen;
        ``$\deg(P + Q) = \max(\dots)$'' schrijven is alleen veilig bij
        verschillende graden.
  \item \emph{Wortels correct tellen.} ``$n$ wortels'' in
        \cref{cor:b1:poly:factorization} betekent \emph{met
        \omterm{def:b1:poly:derivative}{multipliciteit}, in $\C$}: $X^2 + 1$ heeft geen reële wortels, en
        $(X-1)^2$ heeft één verschillende wortel maar twee met
        \omterm{def:b1:poly:derivative}{multipliciteit}. \omterm{def:b1:logic:statement}{Uitspraken} die de drie tellingen door elkaar
        halen zijn de meest voorkomende bron van valse bewijzen.
  \item \emph{Irreducibiliteit hangt van het \omterm{def:b1:structures:field}{lichaam} af.} $X^2 - 2$ is
        irreducibel over $\Q$ en splitst over $\R$; $X^2 + 1$ is
        irreducibel over $\R$ en splitst over $\C$. Het kale woord
        ``irreducibel'' betekent niets zolang het \omterm{def:b1:structures:field}{lichaam} van de
        coëfficiënten niet genoemd is.
\end{enumerate}
\end{remark}

\section{Coëfficiënten en wortels}

\begin{theorem}[Formules van Vieta]\label{thm:b1:poly:vieta}
Zij $P = X^n + c_{n-1} X^{n-1} + \dots + c_0$ \omterm{def:b1:poly:def}{monisch} met wortels $a_1,
\dots, a_n \in \C$ (met \omterm{def:b1:poly:derivative}{multipliciteit}). Dan is\index{formules van Vieta}
\[
\sum_i a_i = -c_{n-1},
\qquad
\sum_{i < j} a_i a_j = c_{n-2},
\qquad \dots, \qquad
a_1 a_2 \cdots a_n = (-1)^n c_0 ,
\]
waarbij de $k$-de symmetrische functie van de wortels gelijk is aan
$(-1)^k c_{n-k}$.
\end{theorem}

\begin{proof}
Volgens \cref{cor:b1:poly:factorization} is $P = (X - a_1)\cdots(X -
a_n)$ (\omterm{def:b1:poly:def}{monisch}, alle wortels opgesomd). Het product distributief
uitwerken levert één term per manier om in elke factor hetzij $X$, hetzij
de wortelterm $-a_i$ te kiezen: de wortels kiezen in de factoren met
index $i_1 < \dots < i_k$ en $X$ in de $n - k$ andere draagt
$(-a_{i_1})\cdots(-a_{i_k})\,X^{n-k}$ bij. Groeperen naar de macht van
$X$:
\[
P = \sum_{k=0}^{n} (-1)^k
\Bigl(\sum_{i_1 < \dots < i_k} a_{i_1}\cdots a_{i_k}\Bigr)
X^{n-k} ,
\]
en gelijkstellen met $P = \sum_k c_{n-k}X^{n-k}$ (de coëfficiënten zijn
uniek, \cref{def:b1:poly:def}) geeft $c_{n-k} = (-1)^k \sigma_k$, oftewel
$\sigma_k = (-1)^kc_{n-k}$, waarbij $\sigma_k$ de $k$-de symmetrische
functie hierboven noteert. De drie getoonde gevallen zijn $k = 1$, $k =
2$ en $k = n$.
\end{proof}

\begin{example}\label{ex:b1:poly:vieta}
Voor de tweedegraadsveelterm $X^2 - sX + p$: de som van de wortels is $s$
en het product $p$ --- al herhaaldelijk gebruikt
(\cref{exo:b1:complex:8}). Voor een \omterm{def:b1:poly:def}{monische} derdegraadsveelterm $X^3 +
aX^2 + bX + c$ met wortels $\alpha, \beta, \gamma$:
\[
\alpha + \beta + \gamma = -a,
\quad
\alpha\beta + \beta\gamma + \gamma\alpha = b,
\quad
\alpha\beta\gamma = -c ,
\]
waarmee je symmetrische grootheden als $\alpha^2 + \beta^2 + \gamma^2 =
a^2 - 2b$ berekent zonder op te lossen.
\end{example}

\begin{example}[De wortels transformeren zonder ze te vinden]\label{ex:b1:poly:transformroots}
Zij $\alpha, \beta$ de wortels van $X^2 - 3X + 1$. Welke \omterm{def:b1:poly:def}{monische}
tweedegraadsveelterm heeft $\alpha^2, \beta^2$ als wortels? Volgens Vieta
is $\alpha + \beta = 3$ en $\alpha\beta = 1$, dus
\[
\alpha^2 + \beta^2 = (\alpha+\beta)^2 - 2\alpha\beta = 7,
\qquad
\alpha^2\beta^2 = (\alpha\beta)^2 = 1 :
\]
het antwoord is $X^2 - 7X + 1$ --- verkregen zonder $\alpha = \frac{3 +
\sqrt5}2$ te berekenen. (Controle: $\alpha^2 = \frac{7 + 3\sqrt5}2$, en
inderdaad is $\alpha^2 + \beta^2 = 7$.) Dezelfde strategie behandelt
omgekeerden (transformaties van het type $X^2 - \frac ba X + \frac ca$),
verschuivingen en alle symmetrische gegevens: Vieta zet vragen over
\emph{onbekende} wortels om in algebra op \emph{bekende} coëfficiënten.
Ze zal voortdurend van pas komen wanneer de wortels eigenwaarden zijn
(\cref{ch:b1:det}).
\end{example}

\begin{example}[Palindromische vergelijkingen]\label{ex:b1:poly:palindrome}
Los $X^4 + X^3 - 4X^2 + X + 1 = 0$ op. De coëfficiënten lezen in beide
richtingen hetzelfde, dus $0$ is geen wortel en delen door $X^2$ verliest
geen oplossingen:
\[
X^2 + X - 4 + \frac1X + \frac1{X^2} = 0 .
\]
Zet $y = X + \frac1X$: dan is $X^2 + \frac1{X^2} = y^2 - 2$, en de
vergelijking stort in tot
\[
y^2 + y - 6 = 0 \iff (y + 3)(y - 2) = 0 .
\]
Ontrol elke waarde via $X^2 - yX + 1 = 0$: voor $y = 2$ geeft $X^2 - 2X +
1 = (X - 1)^2$ de dubbele wortel $1$; voor $y = -3$ geeft $X^2 + 3X + 1 =
0$ de wortels $X = \frac{-3 \pm \sqrt5}2$. Vier wortels met
\omterm{def:b1:poly:derivative}{multipliciteit} voor een vierdegraadsveelterm, zoals
\cref{cor:b1:poly:factorization} eist --- verkregen door twee
tweedegraadsvergelijkingen op te lossen. De truc werkt bij elke
\emph{palindromische} \omterm{def:b1:poly:def}{veelterm}: hun wortels komen in omgekeerde paren
$\{x, 1/x\}$ (vervang $X$ door $1/X$ en werk de noemers weg), en $y = X +
\frac1X$ is juist de grootheid die op zulke paren constant is, wat de
graad halveert.
\end{example}

\begin{theorem}[Lagrange-interpolatie]\label{thm:b1:poly:lagrange}
Zij $x_0, \dots, x_n$ verschillende punten van $K$ en $y_0, \dots, y_n
\in K$. Er is precies één $P \in K[X]$ van graad $\leq n$ met $P(x_i) =
y_i$ voor alle $i$, namelijk\index{Lagrange-interpolatie}
\[
P = \sum_{i=0}^{n} y_i\, L_i,
\qquad
L_i = \prod_{j \neq i} \frac{X - x_j}{x_i - x_j} .
\]
\end{theorem}

\begin{proof}
Elke $L_i$ heeft graad $n$ en voldoet aan $L_i(x_i) = 1$ en $L_i(x_j) =
0$ voor $j \neq i$ (telkens wordt één factor nul in het overeenkomstige
$x_j$). De getoonde $P$ heeft dus graad $\leq n$ en interpoleert.
Uniciteit: twee interpolerende \omterm{def:b1:poly:def}{veeltermen} van graad $\leq n$ stemmen
overeen in de $n+1$ punten $x_i$ en zijn dus gelijk
(\cref{cor:b1:poly:nroots}).
\end{proof}

\begin{remark}[Tussenspel: veeltermen zijn ook vectoren]\label{rem:b1:poly:interlude}
Een verandering van standpunt die \cref{ch:b1:vspaces} officieel zal
maken: de \omterm{def:b1:poly:def}{veeltermen} van graad $\leq n$ vormen een ruimte waarin optellen
en schalen zich precies als bij coördinaten gedragen --- een \omterm{def:b1:poly:def}{veelterm}
\emph{is} zijn lijst van $n + 1$ coëfficiënten. Drie \omterm{def:b1:logic:statement}{uitspraken} van dit
hoofdstuk zijn stiekem lineaire algebra. De \omterm{thm:b1:poly:lagrange}{Lagrange-interpolatie}
(\cref{thm:b1:poly:lagrange}) zegt dat de evaluatiegegevens $(P(x_0),
\dots, P(x_n))$ de \omterm{def:b1:poly:def}{veelterm} $P$ ondubbelzinnig vastleggen: evalueren in $n
+ 1$ punten is een lineaire bijectie, en de $L_i$ vormen de basis die
daarbij past. De ontwikkeling $R = \sum c_k(X - a)^k$ in het bewijs van
\cref{prop:b1:poly:multiplicity} zegt dat de machten van $(X - a)$ een
ander coördinatenstelsel vormen, met $c_k = R^{(k)}(a)/k!$ als
coördinaten. En \cref{cor:b1:poly:nroots} --- meer wortels dan de graad
dwingt de nulveelterm af --- is de motor van elke uniciteit: ze zal in
\cref{ch:b1:findim} ``een lineaire \omterm{def:b1:logic:map}{afbeelding} die \omterm{def:b1:logic:inj}{injectief} is op een
ruimte van dimensie $n + 1$'' worden. Wanneer die hoofdstukken komen, zal
de ruimte $K_n[X]$ hun lievelingsvoorbeeld zijn; het loont er al vloeiend
in te zijn wanneer je er aankomt.
\end{remark}

\begin{remark}[Waar dit hoofdstuk gebruikt wordt]\label{rem:b1:poly:whereused}
De ontbinding over $\R$ en $\C$ (\cref{cor:b1:poly:factorization}) is de
motor van de partieelbreuksplitsing in \cref{ch:b1:fractions}, en dus van
een grote klasse integralen in \cref{ch:b1:integration}. De ontwikkeling
van een \omterm{def:b1:poly:def}{veelterm} naar machten van $(X - a)$, tegengekomen in het bewijs
van \cref{prop:b1:poly:multiplicity}, is de algebraïsche schaduw van de
Taylorformules van \cref{ch:b1:taylor}. \omterm{def:b1:diffeq:linear2}{Karakteristieke veeltermen}
verschenen al bij de differentiaalvergelijkingen (\cref{ch:b1:diffeq}) en
keren terug bij matrices in \cref{ch:b1:det}; de \omterm{thm:b1:poly:lagrange}{Lagrange-interpolatie} is
de eerste bestaans- en uniciteitsstelling van de numerieke wiskunde, en de
\omterm{pb:b1:poly:1}{Tsjebysjev-veeltermen} van \cref{exo:b1:poly:10} --- waarvan de
weekendopgave hieronder de optimaliteit vaststelt --- vertellen die
discipline \emph{waar} je moet interpoleren. Ten slotte voedt de hele
rekenkunde van $K[X]$, gekopieerd van \cref{ch:b1:arith}, de studie van
$K[X]$-idealen en quotiëntringen in het volume van bachelorjaar 2.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:poly:1}
Voer de euclidische delingen uit: $X^5 - 1$ door $X^2 + X + 1$; en
vervolgens $2X^4 + X^3 - X + 3$ door $X^2 - 2$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:2}
Voor welke $n \in \N$ \omterm{def:b1:arith:divides}{deelt} $X^2 + X + 1$ de \omterm{def:b1:poly:def}{veelterm} $X^{2n} + X^n + 1$?
\emph{Aanwijzing: de wortels van $X^2 + X + 1$ zijn $j$ en $j^2$ met $j =
\eu^{2\iu\pi/3}$; bespreek $n$ modulo $3$.}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:3}
Bepaal de reële $a, b$ zó dat $(X-1)^2$ de \omterm{def:b1:poly:def}{veelterm} $P = X^4 + aX^3 +
bX^2 + 1$ \omterm{def:b1:arith:divides}{deelt}, en ontbind $P$ voor die waarden over $\R$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:4}
Ontbind over $\C$ en over $\R$: $X^3 - 1$; $\;X^4 + X^2 + 1$; $\;X^6 -
1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:5}
Zij $P = X^3 - 6X^2 + 11X - 6$.
\begin{enumerate}
  \item Bepaal de rationale wortels \emph{(een onvereenvoudigbare
        rationale wortel $p/q$ van een \omterm{def:b1:poly:def}{monische veelterm} met gehele
        coëfficiënten is een geheel getal dat de constante term \omterm{def:b1:arith:divides}{deelt}
        --- bewijs dat)}, en ontbind $P$.
  \item Bereken zonder op te lossen, via Vieta, de som van de kwadraten
        en de som van de omgekeerden van de wortels, en toets ze aan de
        ontbinding.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:6}
Bereken $\gcd(X^4 - 1,\; X^3 - X^2 + X - 1)$ met het \omterm{met:b1:arith:euclid}{algoritme van
Euclides}, en schrijf hem als een combinatie $AU + BV$ van de twee
\omterm{def:b1:poly:def}{veeltermen}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:7}
Zij $P \in \R[X]$ met $P(x) \geq 0$ voor alle $x \in \R$. Bewijs dat $P$
een som van twee kwadraten van reële \omterm{def:b1:poly:def}{veeltermen} is: $P = A^2 + B^2$.
\emph{Aanwijzing: in de reële ontbinding hebben reële wortels een even
\omterm{def:b1:poly:derivative}{multipliciteit}; schrijf de kwadratische factoren als $(X - z)(X - \conj
z)$ en gebruik $\abs{\,\cdot\,}^2 = (\Re)^2 + (\Im)^2$ op het product van
de $(X - z)$'s.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:8}
Bepaal de \omterm{def:b1:poly:def}{veelterm} $P$ van graad $\leq 2$ met $P(0) = 1$, $P(1) = 3$ en
$P(2) = 2$, eerst met de formule van Lagrange en daarna door het lineaire
stelsel voor de coëfficiënten op te lossen. Ga na dat beide antwoorden
overeenstemmen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:9}
Bewijs dat $P = X^{2n+1} - 1$ precies één reële wortel heeft, en dat voor
elke $n \geq 1$ de \omterm{def:b1:poly:def}{veelterm} $1 + X + \frac{X^2}{2!} + \dots +
\frac{X^n}{n!}$ geen meervoudige wortel heeft \emph{(vergelijk $P$ en
$P'$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:poly:10}
(\omterm{pb:b1:poly:1}{Tsjebysjev-veeltermen}) Definieer $T_0 = 1$, $T_1 = X$ en $T_{n+1} =
2X\,T_n - T_{n-1}$.
\begin{enumerate}
  \item Bewijs met inductie dat $T_n(\cos\theta) = \cos n\theta$ voor
        alle $\theta$.
  \item Leid de $n$ wortels van $T_n$ en zijn kopcoëfficiënt af.
  \item Bewijs dat $\sup_{x \in \intcc{-1}{1}} \abs{T_n(x)} = 1$, en dat
        die waarde in $n + 1$ punten van $\intcc{-1}{1}$ bereikt wordt.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:poly:11}
Zij $P \in \C[X]$ niet-constant met verschillende wortels $a_1, \dots,
a_r$ (\omterm{def:b1:poly:derivative}{multipliciteiten} $m_1, \dots, m_r$). Bewijs de identiteit van
rationale functies
\[
\frac{P'(X)}{P(X)} = \sum_{i=1}^{r} \frac{m_i}{X - a_i},
\]
en leid daaruit de stelling van Gauss--Lucas af: elke wortel van $P'$
ligt in het convexe omhulsel van de wortels van $P$ \emph{(evalueer de
identiteit in een wortel $w$ van $P'$ die geen wortel van $P$ is, neem
toegevoegden, en lees het resultaat als: $w$ is een gewogen gemiddelde van
de $a_i$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:12}
(Filter met \omterm{def:b1:complex:unity}{eenheidswortels}) Zij $n \in \N^*$ en $j = \eu^{2\iu\pi/3}$.
Bewijs, door $(1 + X)^n$ te evalueren in $1$, $j$ en $j^2$, dat
\[
\sum_{k \geq 0} \binom{n}{3k}
= \frac{2^n + 2\cos\frac{n\pi}{3}}{3} ,
\]
en toets de formule voor $n = 3$ en $n = 6$. \emph{Aanwijzing: $1 + j^m +
j^{2m}$ is gelijk aan $3$ als $3 \mid m$ en anders aan $0$; en $1 + j =
\eu^{\iu\pi/3}$.}
\end{exercise}

\section{Opgave: Tsjebysjev-veeltermen en de vlakste veelterm}

\begin{problem}\label{pb:b1:poly:1}
Welke van alle \emph{\omterm{def:b1:poly:def}{monische}} \omterm{def:b1:poly:def}{veeltermen} van graad $n$ blijft het
dichtst bij nul op $\intcc{-1}1$? Het antwoord --- de stelling van
Tsjebysjev, de geboorteakte van de approximatietheorie --- is
$2^{1-n}T_n$, met $T_n$ de \omterm{pb:b1:poly:1}{Tsjebysjev-veelterm} van
\cref{exo:b1:poly:10}, en geen enkele \omterm{def:b1:poly:def}{monische} mededinger komt onder haar
afwijking $2^{1-n}$.\index{Tsjebysjev-veeltermen} Deze opgave ontwikkelt
de algebra van de familie $(T_n)$ (samenstellingswet, expliciete
coëfficiënten, de familie $U_n$ van de tweede soort, een
differentiaalvergelijking), bewijst de extremaliteitsstelling met haar
gelijkheidsgeval, en verzamelt toepassingen: optimale interpolatiepunten,
de exacte waarde van $\cos 36^\circ$, en een \omterm{def:b1:arith:congruence}{congruentie} $T_p \equiv X^p
\pmod p$. Overal is $T_0 = 1$, $T_1 = X$ en $T_{n+1} = 2X\,T_n -
T_{n-1}$, en gebruiken we vrijelijk $T_n(\cos\theta) = \cos n\theta$ uit
\cref{exo:b1:poly:10}.

\medskip
\noindent\textbf{Deel I --- De familie $(T_n)$.}
\begin{enumerate}
  \item Bereken $T_2, T_3, T_4, T_5$ met de recursie. (Vergelijk $T_3$
        met de identiteit $\cos3\theta = 4\cos^3\theta - 3\cos\theta$
        van \cref{ex:b1:complex:cos3}.)
  \item Bewijs met inductie: $\deg T_n = n$ met kopcoëfficiënt $2^{n-1}$
        voor $n \geq 1$, en $T_n$ heeft de pariteit van $n$ (er komen
        alleen even of alleen oneven machten in voor).
  \item Bewijs het uniciteitsbeginsel: $T_n$ is de \emph{enige} \omterm{def:b1:poly:def}{veelterm}
        die aan $P(\cos\theta) = \cos n\theta$ voldoet voor alle
        $\theta$. (Twee \omterm{def:b1:poly:def}{veeltermen} die op $\intcc{-1}1$ overeenstemmen,
        stemmen overal overeen: \cref{cor:b1:poly:nroots}.)
  \item Leid de samenstellings- en de productwet af:
        \[
        T_m \circ T_n = T_{mn},
        \qquad
        2\,T_m T_n = T_{m+n} + T_{\abs{m-n}} .
        \]
  \item Herinner uit \cref{exo:b1:poly:10} de wortels $x_k =
        \cos\frac{(2k+1)\pi}{2n}$ en de equi-oscillatiepunten $y_k =
        \cos\frac{k\pi}n$ met $T_n(y_k) = (-1)^k$. Schrijf de volledige
        ontbinding van $T_n$ over $\R$ op, en verantwoord dat de $y_k$
        ertussen liggen: $y_n < x_{n-1} < y_{n-1} < \dots < x_0 < y_0$.
  \item Bewijs dat $T_n(\cosh t) = \cosh(nt)$ voor alle $t \in \R$
        (dezelfde inductie, met
        \cref{prop:b1:functions:hyprules}), en leid voor $x \geq 1$ de
        gesloten vorm
        \[
        T_n(x) = \frac{\bigl(x + \sqrt{x^2 - 1}\bigr)^n +
        \bigl(x - \sqrt{x^2 - 1}\bigr)^n}{2}
        \]
        af, zodat $T_n(x) > 1$ voor $x > 1$: buiten $\intcc{-1}1$
        ontsnapt de \omterm{def:b1:poly:def}{veelterm} onmiddellijk.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Coëfficiënten, de familie $U_n$, een
differentiaalvergelijking.}
\begin{enumerate}[resume]
  \item Bewijs met de formule van de Moivre
        (\cref{cor:b1:complex:demoivre}) de expliciete uitdrukking
        \[
        T_n(x) = \sum_{0 \leq 2j \leq n} \binom{n}{2j}\,
        x^{\,n-2j}\,(x^2 - 1)^j ,
        \]
        en ga haar na voor $n = 3$.
  \item Bereken $T_n(1)$, $T_n(-1)$ en $T_n(0)$ voor alle $n$.
  \item Definieer $U_n$ (\emph{van de tweede soort}) door $U_0 = 1$,
        $U_1 = 2X$ en $U_{n+1} = 2X\,U_n - U_{n-1}$. Bewijs dat
        $U_n(\cos\theta) = \frac{\sin(n+1)\theta}{\sin\theta}$ voor
        $\theta \notin \pi\Z$, en dat $T_n' = n\,U_{n-1}$ voor $n \geq
        1$.
  \item Bewijs dat $\abs{\sin n\theta} \leq n\,\abs{\sin\theta}$ voor
        alle $\theta$ (inductie), en leid de afschatting van
        markovtype af:
        \[
        \abs{T_n'(x)} \leq n^2
        \quad\text{op } \intcc{-1}1,
        \qquad\text{met } T_n'(\pm1) = (\pm1)^{n-1}\,n^2 .
        \]
  \item Toon aan dat $y = T_n$ voldoet aan de differentiaalvergelijking
        \[
        (1 - x^2)\,y'' - x\,y' + n^2\,y = 0 ,
        \]
        door de identiteit $\sin\theta\, T_n'(\cos\theta) = n\sin
        n\theta$ naar $\theta$ te differentiëren; ga dat rechtstreeks na
        voor $T_2$.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De extremaliteitsstelling van Tsjebysjev.}
Zij $\widetilde T_n = 2^{1-n}\,T_n$ (\omterm{def:b1:poly:def}{monisch} wegens vraag 2) en schrijf
$\norm{P}_\infty = \sup_{x \in \intcc{-1}1}\abs{P(x)}$.
\begin{enumerate}[resume]
  \item Verantwoord $\norm{\widetilde T_n}_\infty = 2^{1-n}$, bereikt
        met afwisselende tekens in de $n + 1$ punten $y_n < \dots <
        y_0$.
  \item Stel dat een \omterm{def:b1:poly:def}{monische} $P$ van graad $n$ zou voldoen aan $\norm
        P_\infty < 2^{1-n}$, en zet $D = \widetilde T_n - P$. Toon aan
        dat $\deg D \leq n - 1$, en dat $D(y_k)$ voor elke $k = 0,
        \dots, n$ strikt het teken van $(-1)^k$ heeft.
  \item Leid af dat $D$ minstens $n$ verschillende reële wortels heeft
        (één in elk tussenstuk, wegens de tussenwaarde-eigenschap, hier
        gebruikt zoals bekend uit het bovenbouwvolume en bewezen in
        \cref{ch:b1:continuity}), en besluit met de \emph{stelling van
        Tsjebysjev}: elke \omterm{def:b1:poly:def}{monische} $P$ van graad $n$ voldoet aan
        \[
        \norm{P}_\infty \geq 2^{1-n} .
        \]
  \item (Gelijkheidsgeval, eerste stap) Neem nu aan dat $\norm P_\infty
        = 2^{1-n}$ precies, met $P$ \omterm{def:b1:poly:def}{monisch} van graad $n$, en zet $D =
        \widetilde T_n - P$. Toon aan dat $(-1)^kD(y_k) \geq 0$ voor
        alle $k$, en dat uit $D(y_k) = 0$ voor een \emph{inwendig} punt
        $y_k$ ($0 < k < n$) ook $D'(y_k) = 0$ volgt. \emph{(In een
        inwendig punt $y_k$ bereiken zowel $\widetilde T_n$ als $P$ een
        extremum met absolute waarde $\norm{\cdot}_\infty$; een
        afleidbare functie heeft in een inwendig extremum afgeleide nul
        --- hier gebruikt zoals bekend uit het bovenbouwvolume, bewezen
        in \cref{ch:b1:derivative}.)}
  \item (Gelijkheidsgeval, besluit) Tel de wortels van $D$ met
        \omterm{def:b1:poly:derivative}{multipliciteit} om aan te tonen dat $D = 0$: de minimaliserende
        \omterm{def:b1:poly:def}{veelterm} is \emph{uniek}, $P = \widetilde T_n$.
  \item Vervoer alles naar een willekeurig segment $\intcc ab$: toon aan
        dat de minimale supremumnorm van een \omterm{def:b1:poly:def}{monische veelterm} van graad
        $n$ op $\intcc ab$ gelijk is aan $2\bigl(\frac{b-a}4\bigr)^n$,
        bereikt door een herschaalde \omterm{pb:b1:poly:1}{Tsjebysjev-veelterm}.
        \emph{(Substitueer $x = \frac{a+b}2 + \frac{b-a}2\,t$ en volg de
        kopcoëfficiënt.)}
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Toepassingen.}
\begin{enumerate}[resume]
  \item Werk het geval $n = 3$ met de hand uit: lokaliseer de extrema
        van $\widetilde T_3 = X^3 - \frac34X$ op $\intcc{-1}1$, ga de
        viervoudige equi-oscillatie met waarde $\frac14$ na, en besluit
        dat geen enkele \omterm{def:b1:poly:def}{monische} derdegraadsveelterm het beter doet.
  \item (Optimale interpolatiepunten) Voor $n + 1$ punten $x_0, \dots,
        x_n \in \intcc{-1}1$ wordt de interpolatiefout beheerst door
        $\omega(X) = \prod_i (X - x_i)$ (zoals \cref{ch:b1:taylor} zal
        kwantificeren). Bewijs dat de keuze die $\norm\omega_\infty$
        minimaliseert bestaat uit de $n + 1$ wortels van $T_{n+1}$, met
        $\norm\omega_\infty = 2^{-n}$: de punten van Tsjebysjev zijn de
        juiste plaatsen om te interpoleren.
  \item Bewijs met $T_5$ dat $c = \cos 36^\circ$ voldoet aan $16c^5 -
        20c^3 + 5c + 1 = 0$, ontbind die \omterm{def:b1:poly:def}{veelterm} als $(x + 1)(4x^2 - 2x
        - 1)^2$, en besluit
        \[
        \cos 36^\circ = \frac{1 + \sqrt5}4 .
        \]
        Ga de overeenstemming na met $\cos 72^\circ = \frac{\sqrt5 -
        1}4$ uit \cref{exo:b1:complex:8}.
  \item Schat $T_{10}(1.1)$ met de gesloten vorm van vraag 6 (twee
        beduidende cijfers volstaan) en duid het resultaat: een \omterm{def:b1:poly:def}{veelterm}
        die op $\intcc{-1}1$ door $1$ begrensd is, kan in $x = 1.1$ al
        boven $40$ uitkomen. (Dat $T_n$ onder zulke \omterm{def:b1:poly:def}{veeltermen} het
        \emph{snelst} groeit, is nog een extremale eigenschap van de
        familie, buiten deze opgave.)
  \item Bewijs de \omterm{def:b1:arith:congruence}{congruentie}: voor elk oneven \omterm{def:b1:arith:prime}{priemgetal} $p$ zijn alle
        coëfficiënten van $T_p - X^p$ \omterm{def:b1:arith:divides}{deelbaar} door $p$. \emph{(Gebruik
        vraag 7 en $p \mid \binom p{2j}$ voor $0 < 2j < p$, uit het
        bewijs van \cref{thm:b1:arith:fermat}.)} Ga dat na op $T_3$ en
        $T_5$.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Synthese.}
\begin{enumerate}[resume]
  \item Bereken expliciet de \omterm{def:b1:poly:def}{monische} tweedegraadsveelterm met de
        kleinste supremumnorm op $\intcc01$, en haar afwijking. (Vraag
        17 met $n = 2$.)
  \item Waar precies gebruikte de opgave: (i) de starheid van \omterm{def:b1:poly:def}{veeltermen}
        (\cref{cor:b1:poly:nroots}); (ii) de goniometrie van
        \cref{ch:b1:complex} en \cref{ch:b1:functions}; (iii) de
        rekenkunde van de \omterm{def:b1:counting:objects}{binomiaalcoëfficiënten} uit
        \cref{ch:b1:arith}? Eén zin per onderdeel.
  \item Synthese, in een korte alinea: de stelling zegt dat de vlakste
        \omterm{def:b1:poly:def}{monische veelterm} die is welke \emph{equi-oscilleert}, en het
        bewijs zet optimaliteit om in het tellen van wortels.
        Becommentarieer dat mechanisme, de rol van de substitutie $x =
        \cos\theta$ als brug tussen algebra en goniometrie, en noem de
        twee plaatsen waar de opgave feiten uit de analyse nodig had
        (tussenwaardestelling, inwendig extremum) die latere
        hoofdstukken bewijzen.
\end{enumerate}
\end{problem}
