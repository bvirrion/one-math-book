\chapter{Vlakke krommen}\label{ch:b1:curves}

De analyse van \cref{ch:b1:derivative,ch:b1:taylor} was gebouwd voor
functies $y = f(x)$; de meeste krommen uit de meetkunde en de
mechanica --- banen, cirkels die over cirkels rollen, omloopbanen
--- weigeren die vorm en komen in plaats daarvan als
\emph{geparametriseerde} krommen $t \mapsto (x(t), y(t))$ of als
\emph{\omterm{def:b1:curves:polar}{poolkrommen}} $r = r(\theta)$. Dit klassieke hoofdstuk
bestudeert beide, en sluit af met de \omterm{def:b1:curves:conics}{kegelsneden}.

\section{Geparametriseerde krommen}

\begin{definition}\label{def:b1:curves:def}
Een \emph{geparametriseerde kromme}\index{geparametriseerde kromme}
is een \omterm{def:b1:logic:map}{afbeelding} $f \colon I \to \R^2$, $t \mapsto M(t) = (x(t),
y(t))$, met $x, y$ (minstens) van klasse $C^1$ op het \omterm{prop:b1:reals:intervals}{interval} $I$.
De \emph{snelheidsvector} is $f'(t) = (x'(t), y'(t))$; het punt
$M(t_0)$ heet \emph{regulier} wanneer $f'(t_0) \neq (0,0)$, en de
\emph{raaklijn} daar is de rechte door $M(t_0)$ met richting
$f'(t_0)$.
\end{definition}

\begin{proof}[Waarom $f'$ de raaklijn richt]
Volgens Taylor--Young, component per component, is $M(t) = M(t_0) +
(t - t_0) f'(t_0) + o(t - t_0)$: de richting van de koorde
$\frac{M(t) - M(t_0)}{t - t_0}$ nadert tot $f'(t_0)$. De hypothese
van regulariteit is wat van de limiet een \emph{richting} maakt: is
$f'(t_0) = 0$, dan stort de formule in tot $M(t) = M(t_0) + o(t -
t_0)$ en zegt zij niets over hoe de kromme het punt verlaat ---
vandaar de afzonderlijke behandeling van singuliere punten in de
methode hieronder, waar de eerste \omterm{def:b1:derivative:def}{afgeleide} die \emph{niet} nul is
de rol van $f'$ overneemt. Merk ook op dat de raaklijn een
meetkundig object is: elke herparametrisering verandert $f'$ met
een scalaire factor ongelijk aan nul en laat de rechte
onveranderd.
\end{proof}

\begin{method}[Een geparametriseerde kromme onderzoeken]\label{met:b1:curves:study}
\begin{enumerate}
  \item \emph{Verklein het domein} met symmetrieën: betrekkingen
        tussen $M(-t)$, $M(t + T)$, \dots\ en $M(t)$ (spiegelingen,
        translaties, periodiciteit), en teken daarna alleen het
        verkleinde deel.
  \item \emph{Verloop}: tabelleer de tekens van $x'$ en $y'$ samen;
        de kromme beweegt naar rechts of links volgens $x'$, omhoog
        of omlaag volgens $y'$.
  \item \emph{Opmerkelijke punten}: horizontale raaklijn ($y' = 0
        \neq x'$), verticale raaklijn ($x' = 0 \neq y'$); in een
        \emph{singulier} punt ($f' = 0$) onderzoek je hogere
        \omterm{def:b1:derivative:def}{afgeleiden} voor de richting van de raaklijn.
  \item \emph{Asymptotisch gedrag} aan de uiteinden van $I$, en
        daarna schetsen.
\end{enumerate}
\end{method}

\begin{example}[Asymptotische takken, uitgewerkt]
\label{ex:b1:curves:asymptotes}
$x(t) = t$, $y(t) = t + \dfrac1t$ op $\intoo{0}{+\infty}$. Als $t
\to 0^{+}$: $x \to 0$ terwijl $y \to +\infty$ --- de kromme klimt
langs de \emph{verticale asymptoot} $x = 0$ (een eindige limiet
voor de ene coördinaat, oneindig voor de andere). Als $t \to
+\infty$ blazen beide \omterm{prop:b1:vspaces:coordinates}{coördinaten} op, dus toets een rechte: het
verschil
\[
y(t) - x(t) = \frac1t \longrightarrow 0^{+}
\]
toont de \emph{schuine asymptoot} $y = x$, van boven benaderd.
Daartussenin wordt $y' = 1 - \frac1{t^2}$ nul bij $t = 1$: het
punt $(1, 2)$ is het laagste punt van de tak, en $x' = 1 > 0$
overal, dus schuift de kromme altijd naar rechts op. Eén boog,
twee asymptoten, één minimum: een volledig \omterm{def:b1:linmaps:kerim}{beeld} uit drie
berekeningen --- met het algemene recept eronder zichtbaar: gaan
$x$ en $y$ samen naar oneindig, onderzoek dan $y/x$ voor een
kandidaat-helling (hier $\to 1$), en daarna $y -
(\text{helling})\,x$ voor het snijpunt en de kant van de
benadering.
\end{example}

\begin{example}[De astroïde]\label{ex:b1:curves:astroid}
$x(t) = \cos^3 t$, $y(t) = \sin^3 t$.\index{astroïde} Symmetrieën:
$M(t + 2\pi) = M(t)$ (bestudeer één periode); $M(-t)$ is de
spiegeling van $M(t)$ in de $x$-as; $M(\pi - t)$ in de $y$-as;
$M(\frac\pi2 - t)$ in de diagonaal $y = x$: het volstaat $t \in
\intcc{0}{\frac\pi4}$ te bestuderen en uit te vouwen.

Snelheid: $f'(t) = 3\sin t\cos t\,(-\cos t, \sin t)$. Op
$\intoo{0}{\frac\pi2}$ zijn alle punten regulier met een raaklijn
gericht door $(-\cos t, \sin t)$; bij $t = 0$ (het punt $(1,0)$)
wordt de snelheid nul: een \emph{keerpunt}\index{keerpunt}, waar de
kromme omkeert langs de raakrichting $(-1, 0)\cdot(\pm)$ --- uit
symmetrie liggen de vier \omterm{ex:b1:curves:astroid}{keerpunten} in $(\pm1, 0), (0, \pm1)$.
\end{example}

\begin{example}[Een liggende acht, volledig onderzocht]
\label{ex:b1:curves:lissajous}
$x(t) = \sin t$, $y(t) = \sin 2t$ (een \emph{lissajouskromme}).
\emph{Symmetrieën}: $M(t + \pi) = (-x(t), y(t))$ (spiegeling in de
$y$-as), $M(-t) = (-x(t), -y(t))$ (centrale symmetrie), $M(\pi -
t) = (x(t), -y(t))$ (spiegeling in de $x$-as): het volstaat $t \in
\intcc{0}{\frac\pi2}$ te bestuderen en uit te vouwen.
\emph{Verloop}: $x' = \cos t \geq 0$ overal, terwijl $y' = 2\cos
2t$ positief is vóór $t = \frac\pi4$ en negatief erna: de boog
klimt naar rechts tot de top $\bigl(\frac{\sqrt2}2,\ 1\bigr)$ bij
$t = \frac\pi4$ (horizontale raaklijn), en daalt daarna naar
rechts tot $(1, 0)$ bij $t = \frac\pi2$, waar $x' = 0 \neq y'$:
verticale raaklijn. \emph{Dubbelpunt}: $M(0) = M(\pi) = (0,0)$,
met twee verschillende snelheden
\[
f'(0) = (1,\ 2), \qquad f'(\pi) = (-1,\ 2) :
\]
twee \emph{reguliere} takken die elkaar in de oorsprong onder
verschillende hoeken kruisen --- een dubbelpunt, geen singulier
punt: elke doorgang is volmaakt glad, de twee doorgangen delen
alleen hun plaats. De hele kromme is de liggende acht hieronder.
\end{example}

\section{Poolkrommen}

\begin{definition}\label{def:b1:curves:polar}
Een \emph{poolkromme}\index{poolkromme} wordt gegeven door $r =
r(\theta)$: het punt met parameter $\theta$ is
\[
M(\theta) = r(\theta)\,\vec u(\theta),
\qquad \vec u(\theta) = (\cos\theta, \sin\theta).
\]
Met $\vec v(\theta) = (-\sin\theta, \cos\theta) = \vec
u\,'(\theta)$ is de snelheid
\[
M'(\theta) = r'(\theta)\, \vec u(\theta) + r(\theta)\, \vec
v(\theta) .
\]
Gevolgen: waar $r \neq 0$ is het punt regulier, en maakt de raaklijn
met de voerstraal de hoek $V$ gegeven door $\tan V = \frac{r}{r'}$
(de hoek tussen $M'$ en $\vec u$); waar $r(\theta_0) = 0$ gaat de
kromme door de oorsprong \emph{met als raaklijn de voerstraal
$\theta = \theta_0$} (richting $\vec u(\theta_0)$, af te lezen uit
$M'(\theta_0) = r'(\theta_0)\vec u(\theta_0)$, of uit de
limietkoorde: de koorde van $O$ naar $M(\theta)$ wordt gedragen
door $\vec u(\theta)$ zelf, dat naar $\vec u(\theta_0)$ nadert ---
zodat de regel zelfs geldt wanneer $r'(\theta_0) = 0$ en het punt
singulier is, en daarom krijgen \omterm{ex:b1:curves:astroid}{pool-keerpunten} in de oorsprong,
zoals dat van de cardioïde, hun raaklijn gratis).
\end{definition}

\begin{example}[De cardioïde]\label{ex:b1:curves:cardioid}
$r(\theta) = 1 + \cos\theta$.\index{cardioïde} Symmetrie:
$r(-\theta) = r(\theta)$: spiegeling in de $x$-as; bestudeer
$\theta \in \intcc{0}{\pi}$. $r$ daalt van $2$ tot $0$; bij $\theta
= \pi$ is $r = 0$: de kromme bereikt de oorsprong rakend aan de
voerstraal $\theta = \pi$ (de $x$-as), en vormt daar een \omterm{ex:b1:curves:astroid}{keerpunt}
--- de punt van het hart. Raaklijn bij $\theta = 0$: $r' = 0$, dus
$\tan V = \infty$: loodrecht op de as.

De hoek $V$ verdient nog één lezing. Bij $\theta = \frac\pi2$: $r
= 1$ en $r' = -1$, dus $\tan V = \frac{r}{r'} = -1$: de raaklijn
maakt drie kwart van een rechte hoek met de uitgaande
voerstraal --- de kromme buigt al terug naar haar \omterm{ex:b1:curves:astroid}{keerpunt}. De
formule $\tan V = r/r'$ levert langs de hele kromme de
raakrichtingen zonder ook maar één berekening van $M'(\theta)$:
zij is het poolanalogon van het aflezen van een helling.
\end{example}

\begin{example}[Van pool naar cartesisch: een verborgen cirkel]
\label{ex:b1:curves:polarcircle}
Wat is de \omterm{def:b1:curves:polar}{poolkromme} $r = 2\cos\theta$? Vermenigvuldig met $r$:
$r^2 = 2r\cos\theta$, dus $x^2 + y^2 = 2x$, dus
\[
(x - 1)^2 + y^2 = 1 :
\]
de cirkel met middelpunt $(1, 0)$ en straal $1$, die door de
oorsprong gaat. De boekhouding doet ertoe: terwijl $\theta$
$\intoc{-\frac\pi2}{\frac\pi2}$ doorloopt, wordt de hele cirkel
precies één keer bestreken ($r$ wordt nul in beide uiteinden), en
bij $\theta = \pm\frac\pi2$ geeft de regel ``raaklijn in de
oorsprong langs de voerstraal $\theta = \theta_0$'' daar een
\emph{verticale} raaklijn --- in overeenstemming met de meetkunde,
want de verticale as raakt deze cirkel inderdaad in $O$. Voor
$\theta$ buiten dat bereik doorloopt $r < 0$ dezelfde cirkel
opnieuw: een herinnering dat een \omterm{def:b1:curves:polar}{poolkromme} een
\emph{geparametriseerd} object is, dat over zichzelf heen mag
gaan.
\end{example}

\section{Booglengte}

\begin{definition}[Lengte van een boog]\label{def:b1:curves:arclength}
De \emph{lengte}\index{booglengte} van een $C^1$-boog $f \colon
\intcc{a}{b} \to \R^2$ is de \omterm{thm:b1:integration:def}{integraal} van de snelheid:
\[
L = \int_a^b \norm{f'(t)}\,\dd t
= \int_a^b \sqrt{x'(t)^2 + y'(t)^2}\;\dd t .
\]
(Motivering: op een klein \omterm{prop:b1:reals:intervals}{interval} is $M(t + h) \approx M(t) + h
f'(t)$, zodat de boog dicht bij een veelhoek ligt waarvan de
lengten van de segmenten een \omterm{thm:b1:integration:riemann}{Riemannsom} van $\norm{f'}$ vormen,
\cref{thm:b1:integration:riemann}.) Voor een \omterm{def:b1:curves:polar}{poolkromme} $r =
r(\theta)$ heeft de snelheid $r'\vec u + r\vec v$ \omterm{def:b1:euclid:orthogonal}{orthogonale}
componenten, dus is
\[
L = \int_{\theta_1}^{\theta_2}
\sqrt{r'(\theta)^2 + r(\theta)^2}\;\dd\theta .
\]
De lengte hangt niet af van de gekozen (monotone,
$C^1$-)parametrisering: de substitutie $t = \varphi(s)$ in de
\omterm{thm:b1:integration:def}{integraal} (\cref{thm:b1:integration:parts}) vermenigvuldigt $f'$
met $\varphi'$ en $\dd t$ met $\varphi'^{-1}$.
\end{definition}

\begin{example}[Controle: de cirkel]\label{ex:b1:curves:circlelength}
Voor $f(t) = (R\cos t, R\sin t)$ op $\intcc{0}{2\pi}$ is
$\norm{f'} = R$, dus $L = 2\pi R$ --- de definitie geeft de omtrek
terug. Toets op herparametrisering: de \omterm{def:b1:logic:map}{afbeelding} $g(t) = (R\cos
2t, R\sin 2t)$ op $\intcc{0}{\pi}$ tekent \emph{dezelfde} cirkel
met dubbele snelheid $\norm{g'} = 2R$, en
\[
\int_0^{\pi} 2R\,\dd t = 2\pi R
\]
opnieuw: de halve tijd, de dubbele snelheid, dezelfde \omterm{def:b1:curves:arclength}{lengte} ---
de invariantie die in de definitie werd beloofd, één keer op
getallen bekeken. ($g$ over heel $\intcc{0}{2\pi}$ laten lopen zou
$4\pi R$ geven: een kromme die twee keer wordt doorlopen is
tweemaal zo lang \emph{als reis}; de \omterm{def:b1:curves:arclength}{lengte} meet het
geparametriseerde pad, en eerlijke boekhouding van het \omterm{prop:b1:reals:intervals}{interval}
hoort bij de berekening.) Voor de astroïde $(\cos^3t, \sin^3t)$ is
$\norm{f'} = 3\abs{\sin t\cos t} = \tfrac32\abs{\sin 2t}$, en uit
symmetrie $L = 4\int_0^{\pi/2} \tfrac32\sin 2t\,\dd t = 6$: een
kromme getekend binnen de eenheidscirkel, van \omterm{def:b1:curves:arclength}{lengte} $6 < 2\pi$.
De weekendopgave meet de beroemdste boog van allemaal.
\end{example}

\section{Kegelsneden}

\begin{definition}[Definitie met brandpunt en richtlijn]\label{def:b1:curves:conics}
Leg een punt $F$ (\emph{brandpunt}), een rechte $D$ die niet door
$F$ gaat (\emph{richtlijn}) en $e > 0$ (\emph{excentriciteit}) vast.
De \emph{kegelsnede}\index{kegelsnede} van deze gegevens is
\[
\mathcal{C} = \{M : d(M, F) = e\; d(M, D)\}:
\]
een \emph{ellips} voor $e < 1$, een \emph{parabool} voor $e = 1$,
een \emph{hyperbool} voor $e > 1$. (De cirkel verschijnt als
ontaarde limiet $e \to 0$.)
\end{definition}

\begin{example}[De definitie, getoetst op een parabool]
\label{ex:b1:curves:parabolacheck}
Neem de parabool $y^2 = 4x$, dus $2p = 4$: brandpunt $F = (1, 0)$
en richtlijn $D : x = -1$ (de gereduceerde vorm
\cref{thm:b1:curves:reduced} plaatst ze in $\pm\frac p2$). In het
punt $M = (1, 2)$ van de kromme:
\[
MF = \sqrt{(1-1)^2 + 2^2} = 2,
\qquad
d(M, D) = 1 - (-1) = 2 :
\]
gelijk, zoals $e = 1$ eist. In $M' = (4, 4)$: $MF' = \sqrt{9 +
16} = 5$ en opnieuw $d(M', D) = 5$. De definitie met brandpunt en
richtlijn is geen abstractie: het is een paar afstanden dat men in
elk punt kan meten, en de algebra van de gereduceerde
vergelijkingen is niets anders dan die meting, eens en voor altijd
uitgevoerd.
\end{example}

\begin{theorem}[Gereduceerde vergelijkingen]\label{thm:b1:curves:reduced}
In een goedgekozen \omterm{def:b1:euclid:orthogonal}{orthonormaal} assenstelsel:
\[
\text{ellips: } \frac{x^2}{a^2} + \frac{y^2}{b^2} = 1
\quad (a \geq b > 0),
\qquad
\text{hyperbool: } \frac{x^2}{a^2} - \frac{y^2}{b^2} = 1,
\qquad
\text{parabool: } y^2 = 2px,
\]
met, voor de ellips: brandpunten in $(\pm c, 0)$, $c = \sqrt{a^2 -
b^2}$, $e = \frac ca$, en de bifocale karakterisering $MF + MF' =
2a$; voor de hyperbool: $c = \sqrt{a^2 + b^2}$, $e = \frac ca$,
$\abs{MF - MF'} = 2a$, asymptoten $y = \pm\frac ba x$.
\end{theorem}

\begin{proof}
Neem het brandpunt in de oorsprong en de richtlijn verticaal, $x =
-h$ ($h > 0$): de voorwaarde $MF^2 = e^2 d(M, D)^2$ luidt $x^2 + y^2
= e^2 (x + h)^2$. Voor $e = 1$: $y^2 = 2hx + h^2$, een parabool na
de verschuiving $x \mapsto x - \frac h2$ (dus $p = h$). Voor $e \neq
1$ geeft het kwadraat afsplitsen in $x$
\[
(1 - e^2)\Bigl(x - \frac{e^2 h}{1 - e^2}\Bigr)^{\!2} + y^2
= e^2h^2 + \frac{e^4 h^2}{1 - e^2}
= \frac{e^2 h^2}{1 - e^2} .
\]
Stel $X = x - \frac{e^2h}{1-e^2}$ (een verschuiving van het
assenstelsel). Is $e < 1$, deel dan door het positieve rechterlid:
$\frac{X^2}{a^2} + \frac{y^2}{b^2} = 1$ met $a = \frac{eh}{1 -
e^2}$ en $b = \frac{eh}{\sqrt{1-e^2}}$; is $e > 1$, dan wisselen
beide leden van de formule op passende wijze van teken en geeft
dezelfde deling $\frac{X^2}{a^2} - \frac{y^2}{b^2} = 1$ met $a =
\frac{eh}{e^2 - 1}$ en $b = \frac{eh}{\sqrt{e^2 - 1}}$. De vermelde
waarden van $c$ volgen ($c^2 = a^2 - b^2$ of $a^2 + b^2$ geeft in
beide gevallen $c = ea$, waarmee het brandpunt juist wordt
geplaatst), en de bifocale eigenschappen zijn rechtstreekse
verificaties op de gereduceerde vergelijkingen. In detail voor de
ellips, met $F' = (c, 0)$ en $M = (X, y)$ op de kromme:
\[
MF'^2 = (X - c)^2 + y^2
= X^2 - 2cX + c^2 + b^2\Bigl(1 - \frac{X^2}{a^2}\Bigr)
= \frac{c^2}{a^2}X^2 - 2cX + a^2
= (a - eX)^2 ,
\]
met $c^2 + b^2 = a^2$ en $e = \frac ca$; omdat $\abs X \leq a$ en
$e < 1$, is $a - eX > 0$, dus $MF' = a - eX$ zonder dat er ooit een
vierkantswortel wordt getrokken. De spiegelberekening geeft $MF = a
+ eX$, waaruit $MF + MF' = 2a$, constant. Voor de hyperbool levert
dezelfde algebra $MF = \abs{a + eX}$ en $MF' = \abs{eX - a}$, met
verschil $\pm2a$ naargelang de tak.
\end{proof}

\begin{example}\label{ex:b1:curves:conicexample}
$\frac{x^2}{25} + \frac{y^2}{9} = 1$: een ellips met $a = 5$, $b =
3$, $c = 4$: brandpunten $(\pm4, 0)$, excentriciteit $\frac45$. Haar
parametrisering: $(5\cos t, 3\sin t)$ --- een cirkel die
anisotroop is uitgerekt; de som van de afstanden tot de
brandpunten is voor elk van haar punten $10$.
\end{example}

\begin{example}[Een omloopbaan lezen uit haar poolvergelijking]
\label{ex:b1:curves:orbit}
De poolkegelsnede $r = \dfrac{1}{1 + \frac12\cos\theta}$
(\cref{exo:b1:curves:7} met $p = 1$, $e = \frac12$) is een ellips
met een \emph{brandpunt in de oorsprong} --- de meetkunde van een
planeetbaan met de zon in $O$. Haal alles uit $p$ en $e$: uit de
reductie van \cref{thm:b1:curves:reduced} volgt $a = \dfrac{p}{1 -
e^2} = \dfrac{1}{3/4} = \dfrac43$ en $c = ea = \dfrac23$. De twee
apsiden bevestigen het zonder enige theorie:
\[
r(0) = \frac{1}{3/2} = \frac23 = a - c
\quad (\text{perihelium}),
\qquad
r(\pi) = \frac{1}{1/2} = 2 = a + c
\quad (\text{aphelium}),
\]
en hun som $\frac23 + 2 = \frac83 = 2a$ vindt de lange as terug.
De poolvorm is de natuurlijke telkens wanneer een brandpunt fysisch
is onderscheiden; de gereduceerde cartesische vorm telkens wanneer
de symmetrieassen dat zijn. Tussen beide omzetten is precies wat de
berekening met het afgesplitste kwadraat in de stelling doet.

De excentriciteit als draaiknop, ten slotte: houd $p = 1$ vast en
draai aan $e$ in $r = \frac{1}{1 + e\cos\theta}$. Bij $e = 0$: de
cirkel $r = 1$. Bij $e = \frac12$: de zojuist bestudeerde ellips,
met $r$ schommelend tussen $\frac23$ en $2$. Bij $e = 1$: $r(\theta)
\to \infty$ als $\theta \to \pi$ --- de kromme sluit zich niet
meer: een parabool, met haar verste punt tot in het oneindige
geduwd. Bij $e = 2$: de noemer wordt nul bij $\cos\theta =
-\frac12$, en alleen $\theta \in
\intoo{-\frac{2\pi}3}{\frac{2\pi}3}$ overleeft: één tak van een
hyperbool, die langs twee asymptotische richtingen ontsnapt. Eén
formule, de hele familie van de \omterm{def:b1:curves:conics}{kegelsneden}, en het overgangspunt
$e = 1$ zichtbaar als het moment waarop de noemer voor het eerst
nul bereikt.
\end{example}

\begin{remark}[Veelgemaakte fouten]
\label{rem:b1:curves:pitfalls}
\emph{Dubbelpunten zijn geen singuliere punten}: bij een
zelfkruising is elke tak regulier; ``singulier'' slaat op $f'(t_0)
= 0$ voor één waarde van de parameter
(\cref{ex:b1:curves:lissajous} tegenover de \omterm{ex:b1:curves:astroid}{keerpunten} van de
astroïde).
\emph{Verticale raaklijn tegenover \omterm{ex:b1:curves:astroid}{keerpunt}}: $x'(t_0) = 0 \neq
y'(t_0)$ is een regulier punt met verticale raaklijn; alleen $x' =
y' = 0$ vraagt om de ontwikkeling van hogere orde.
\emph{Negatieve stralen worden op de tegenoverliggende voerstraal
afgezet}: voor $r(\theta) < 0$ is het punt $-\abs{r}\,\vec
u(\theta)$, onder hoek $\theta + \pi$ (\cref{exo:b1:curves:5}); dit
vergeten doet binnenlussen verdwijnen of krommen verdubbelen.
\emph{\omterm{def:b1:curves:arclength}{Booglengte} integreert $\norm{f'}$, niet $f'$}: splits de
\omterm{thm:b1:integration:def}{integraal} bij de nulpunten van de snelheid --- voor de astroïde
geeft $\frac32\sin 2t$ over een volledige periode integreren zonder
absolute waarden $0$ in plaats van $6$.
\emph{\omterm{def:b1:curves:conics}{Kegelsneden} moeten worden gereduceerd voordat men ze leest}:
in $9x^2 + 25y^2 - 36x - 50y - 164 = 0$ zijn noch de assen, noch
het middelpunt, noch de excentriciteit zichtbaar voordat de
kwadraten zijn afgesplitst (\cref{exo:b1:curves:6}); en een
kwadratisch deel dat op één veranderlijke verdwijnt betekent
parabool, geen ``ontaarde ellips''.
\end{remark}

\begin{remark}[Waar deze krommen heen gaan]
\label{rem:b1:curves:whereused}
\omterm{def:b1:curves:def}{Geparametriseerde krommen} zijn de taal van de mechanica: banen zijn
krommen, snelheidsvectoren zijn raakvectoren, en de \omterm{def:b1:curves:arclength}{booglengte} van
\cref{def:b1:curves:arclength} is de afgelegde afstand. De
\omterm{def:b1:curves:conics}{kegelsneden} duiken op overal waar een omgekeerd-kwadratische wet
werkt --- planeetbanen zijn ellipsen met de zon in een brandpunt.
In \cref{ch:b1:multivar} worden krommen de niveaukrommen van
functies van twee veranderlijken, en ontmoet de raaklijn van dit
hoofdstuk de gradiënt van het volgende. Het volume van
bachelorjaar 2 voegt de kromming en de lokale canonieke vorm van
een kromme toe; de weekendopgave hieronder haalt, met uitsluitend
het gereedschap van dit jaar, alles boven wat de zeventiende eeuw
over haar beroemdste kromme wist.
\end{remark}

\begin{remark}[Vooruitzichten binnen boek 3]
\label{rem:b1:curves:perspectives}
Dit hoofdstuk verbruikt de hele eerste helft van het volume en
voedt het laatste hoofdstuk. \emph{Verbruikt}: raakvectoren zijn
\omterm{def:b1:derivative:def}{afgeleiden} (\cref{ch:b1:derivative}), \omterm{def:b1:curves:arclength}{booglengte} en oppervlakten
zijn \omterm{thm:b1:integration:def}{integralen} (\cref{ch:b1:integration}), \omterm{ex:b1:curves:astroid}{keerpunten} worden met
taylorontwikkelingen beslecht (\cref{ch:b1:taylor}), de tautochroon
is een \omterm{def:b1:linmaps:def}{lineaire} differentiaalvergelijking (\cref{ch:b1:diffeq}), en
elke afstand en hoek is euclidisch (\cref{ch:b1:euclid}).
\emph{Gevoed}: in \cref{ch:b1:multivar} wordt een kromme $t \mapsto
(x(t), y(t))$ die binnen een niveaukromme $\{f = c\}$ is getekend
met de kettingregel afgeleid, en huwt de resulterende identiteit
$\langle \nabla f,\ f'(t)\rangle = 0$ de raakvectoren van dit
hoofdstuk aan de gradiënten van het volgende --- de raaklijn van
de kromme en de normaalrichting van het oppervlak zijn dezelfde
berekening, vanaf beide oevers bekeken.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:curves:2}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:curves:3}
Geef de raaklijn aan de kromme $(\cos^3 t, \sin^3 t)$ bij $t =

assen wordt afgesneden \omterm{def:b1:curves:arclength}{lengte} $1$ heeft --- een beroemde

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:curves:4}
Schets de \omterm{def:b1:curves:polar}{poolkrommen} $r = \cos 2\theta$ (roos met vier bladen) en
$r = \frac{1}{\cos\theta}$ op $\intoo{-\frac\pi2}{\frac\pi2}$
(herken een rechte).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:5}
Onderzoek de \omterm{def:b1:curves:polar}{poolkromme} $r = 1 + 2\cos\theta$ (een limaçon): het
gebied waar $r \geq 0$ tegenover $r < 0$ (punten met een negatieve
straal liggen op de tegenoverliggende voerstraal), de doorgang door
de oorsprong, de binnenlus, en een schets.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:6}
Identificeer de \omterm{def:b1:curves:conics}{kegelsnede} $9x^2 + 25y^2 - 36x - 50y - 164 = 0$:
reduceer haar door kwadraten af te splitsen, en geef het
middelpunt, de halve assen, de brandpunten en de excentriciteit.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:7}
Bewijs dat de poolvergelijking van een \omterm{def:b1:curves:conics}{kegelsnede} met brandpunt in
de oorsprong
\[
r = \frac{p}{1 + e\cos\theta}
\]
is (met de richtlijn verticaal op afstand $\frac pe$ rechts van het
brandpunt). Welke waarden van $\theta$ zijn toegelaten wanneer $e >
1$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:8}
Leid uit de bifocale eigenschap $MF + MF' = 2a$ de
``tuinmansconstructie'' van de ellips af, en bewijs dat de raaklijn
in $M$ gelijke hoeken maakt met $MF$ en $MF'$
\emph{(spiegelingseigenschap; leid $\norm{M(t) - F} + \norm{M(t) -
F'} = 2a$ af en interpreteer de som van de producten met
eenheidsvectoren die nul wordt)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:curves:9}
De \emph{lemniscaat van Bernoulli} is de \omterm{def:b1:curves:polar}{poolkromme} $r^2 = \cos
2\theta$ (neem $r = \sqrt{\cos2\theta}$ waar dat gedefinieerd is).
\begin{enumerate}
  \item Geef haar domein, haar symmetrieën, haar raaklijnen in de
        oorsprong, en schets haar.
  \item Bewijs dat zij de meetkundige plaats is van de punten $M$
        met $MF \cdot MF' = \frac12$, waarbij $F, F' =
        \bigl(\pm\frac{1}{\sqrt2}, 0\bigr)$. \emph{(Bereken
        $MF^2\,MF'^2$ in poolcoördinaten.)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:10}
Bereken de totale \omterm{def:b1:curves:arclength}{lengte} van de cardioïde $r = 1 + \cos\theta$ met
de poolformule van \cref{def:b1:curves:arclength} \emph{(de
identiteit $1 + \cos\theta = 2\cos^2\frac\theta2$ maakt van de
vierkantswortel $2\abs{\cos\frac\theta2}$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:11}
Toon aan dat de raaklijn aan de ellips $(a\cos t, b\sin t)$ in het
punt met parameter $t$ de vergelijking
\[
\frac{x\cos t}{a} + \frac{y\sin t}{b} = 1
\]
heeft, en leid daaruit de raaklijn aan $\frac{x^2}{a^2} +
\frac{y^2}{b^2} = 1$ in een punt $(x_0, y_0)$ van de ellips af:
$\frac{x\,x_0}{a^2} + \frac{y\,y_0}{b^2} = 1$ (de regel van het
``splitsen van de kwadraten'').
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:curves:12}
De \emph{logaritmische spiraal} is de \omterm{def:b1:curves:polar}{poolkromme} $r =
\eu^{k\theta}$ ($k > 0$ vast, $\theta \in \R$).
\begin{enumerate}
  \item Toon aan dat de hoek $V$ tussen de voerstraal en de
        raaklijn \emph{constant} is ($\tan V = \frac1k$) --- de
        spiraal kruist elke voerstraal onder dezelfde hoek.
  \item Toon aan dat het roteren van de spiraal over een hoek $c$
        haar afbeeldt op haar schaling met de factor $\eu^{-kc}$:
        elke rotatie van de spiraal is een vergroting ervan
        (zelfgelijkvormigheid).
  \item Bereken de \omterm{def:b1:curves:arclength}{lengte} van de boog $\theta \in
        \intoc{-\infty}{\theta_0}$ (als limiet van de \omterm{def:b1:curves:arclength}{lengten} op
        $\intcc{A}{\theta_0}$ met $A \to -\infty$) en merk op dat
        zij eindig is: een kromme die oneindig vaak rond de
        oorsprong spiraalt en toch een eindige \omterm{def:b1:curves:arclength}{lengte} heeft.
\end{enumerate}
\end{exercise}

\section{Opgave: de cycloïde, koningin der krommen}

\begin{problem}\label{pb:b1:curves:1}
Een wiel met straal $R$ rolt zonder glijden langs de $x$-as; het
punt van de velg dat aanvankelijk in de oorsprong ligt tekent de
\emph{cycloïde}. De zeventiende eeuw vocht om deze kromme ---
Galilei woog uitgeknipte papieren exemplaren, Wren mat haar,
Roberval berekende haar oppervlakte, Huygens bouwde er klokken op
--- en elk van hun resultaten ligt binnen het bereik van dit
hoofdstuk. Wij bewijzen de vier klassiekers: de constructie van de
raaklijn, de \omterm{def:b1:curves:arclength}{lengte} $8R$ van Wren, de oppervlakte $3\pi R^2$, en de
tautochrone eigenschap van
Huygens.\index{cycloïde}\index{tautochroon}

\medskip
\noindent\textbf{Deel I --- Rollen en de raaklijn.}
\begin{enumerate}
  \item Nadat het wiel over een hoek $t$ is gedraaid, ligt zijn
        middelpunt in $\Omega(t) = (Rt, R)$ (rollen zonder glijden:
        de contactafstand is gelijk aan de afgerolde boog). Toon
        aan dat het gemarkeerde punt zich bevindt in
        \[
        M(t) = \bigl(R(t - \sin t),\; R(1 - \cos t)\bigr).
        \]
  \item Bereken $f'(t)$ en toon aan dat $\norm{f'(t)} =
        2R\,\abs{\sin\frac t2}$; lokaliseer de singuliere punten
        (\omterm{ex:b1:curves:astroid}{keerpunten} --- vergelijk \cref{exo:b1:curves:2}) en de top
        van elke boog.
  \item Zij $C(t) = (Rt, 0)$ het contactpunt en $T(t) = (Rt, 2R)$
        de top van het wiel. Bewijs dat voor $0 < t < 2\pi$ de
        vector $M - C$ \emph{loodrecht} staat op de kromme in
        $M(t)$ en de vector $T - M$ \emph{rakend} is: om de
        raaklijn aan een cycloïde te tekenen, verbind je het punt
        met de top van zijn rollende cirkel. \emph{(Splits alles af
        via $\sin\frac t2$ en $\cos\frac t2$.)}
  \item Interpreteer vraag 3 kinematisch: het contactpunt is het
        \emph{ogenblikkelijke rotatiecentrum}, en de snelheid van
        $M$ is gelijk aan zijn afstand tot $C$ (bij hoeksnelheid
        $1$). Ga na dat $\norm{M - C} = 2R\abs{\sin\frac t2}$.
  \item Bewijs de betrekking tussen hoogte en snelheid
        \[
        \norm{f'(t)}^2 = 2R\,y(t) :
        \]
        op een cycloïde die met hoeksnelheid $1$ wordt doorlopen is
        de snelheid in elk punt precies de valsnelheid bij een val
        gelijk aan de huidige hoogte. (Bewaar dit voor deel IV.)
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De stelling van Wren: de boog heeft
\omterm{def:b1:curves:arclength}{lengte} $8R$.}
\begin{enumerate}[resume]
  \item Bereken met \cref{def:b1:curves:arclength} de \omterm{def:b1:curves:arclength}{lengte} van
        één boog:
        \[
        L = \int_0^{2\pi} 2R\sin\frac t2\,\dd t = 8R
        \]
        (de stelling van Wren, 1658). Vier wieldiameters, en nergens
        een $\pi$.
  \item Bereken de \omterm{def:b1:curves:arclength}{booglengte} vanaf het \omterm{ex:b1:curves:astroid}{keerpunt}: $s(t) =
        4R\bigl(1 - \cos\frac t2\bigr)$, en ga na dat $s(2\pi) =
        8R$.
  \item Meet nu de boog vanaf de \emph{top} $t = \pi$: $\sigma(t)
        = \abs{4R\cos\frac t2}$. Bewijs de intrinsieke betrekking
        \[
        \sigma^2 = 8R\,\bigl(2R - y\bigr) :
        \]
        het kwadraat van de boogafstand tot de top is evenredig met
        de hoogteval onder de top.
  \item Controles: vind met vraag 8 terug dat de halve boog van de
        top tot het \omterm{ex:b1:curves:astroid}{keerpunt} \omterm{def:b1:curves:arclength}{lengte} $4R$ heeft, en vergelijk met de
        berekening voor de astroïde uit
        \cref{ex:b1:curves:circlelength} --- beide krommen hebben
        algebraïsche \omterm{def:b1:curves:arclength}{lengten} zonder $\pi$; leg uit wat dat mogelijk
        maakt hoewel beide uit cirkels zijn gebouwd. \emph{(Kijk
        naar de gedaante van $\norm{f'}$.)}
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De oppervlakte van Roberval: $3\pi
R^2$.}
\begin{enumerate}[resume]
  \item Verantwoord dat de oppervlakte tussen één boog en de grond
        gelijk is aan $A = \int_0^{2\pi} y(t)\,x'(t)\,\dd t$
        \emph{(de substitutie $x = x(t)$ in $\int y\,\dd x$,
        \cref{thm:b1:integration:parts}; $x$ is stijgend)}.
  \item Bereken
        \[
        A = R^2\int_0^{2\pi}(1 - \cos t)^2\,\dd t = 3\pi R^2 :
        \]
        precies \emph{drie} keer de oppervlakte van het wiel --- de
        verhouding die Galilei door wegen had geraden.
  \item Dezelfde methode voor de astroïde $(\cos^3t, \sin^3t)$:
        toon aan dat de omsloten oppervlakte $\frac{3\pi}8$ is
        \emph{(lineariseer $\sin^4 t\cos^2 t$; alleen de constante
        term overleeft over een volledige periode)}.
  \item Toets de formule van vraag 10 op de bovenste
        eenheidshalfcirkel $(\cos t, \sin t)$, met $t$ van $\pi$
        tot $0$: geeft zij $\frac\pi2$?
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De tautochroon van Huygens.} Keer de
boog om: een wrijvingsloos kraaltje glijdt onder de zwaartekracht
$g$ in de cycloïdale schaal
\[
x(t) = R(t + \sin t), \qquad y(t) = R(1 - \cos t)
\qquad (t \in \intcc{-\pi}{\pi}),
\]
waarvan het laagste punt de oorsprong is ($y$ omhoog gemeten).
\begin{enumerate}[resume]
  \item Bereken de snelheid $\norm{f'(t)} = 2R\cos\frac t2$, de
        \omterm{def:b1:curves:arclength}{booglengte} vanaf de bodem $s(t) = 4R\sin\frac t2$, en
        bewijs de sleutelidentiteit
        \[
        y = \frac{s^2}{8R} .
        \]
  \item Het kraaltje dat in rust wordt losgelaten in het punt met
        parameter $t_0 > 0$ gehoorzaamt aan het behoud van energie:
        noteert $s(\tau)$ zijn boogpositie op tijdstip $\tau$, dan
        is $\frac12\bigl(\frac{\dd s}{\dd\tau}\bigr)^2 + g\,y =
        g\,y_0$. Herschrijf dit met vraag 14 als
        \[
        \Bigl(\frac{\dd s}{\dd\tau}\Bigr)^{\!2}
        = \frac{g}{4R}\,\bigl(s_0^2 - s^2\bigr),
        \qquad s_0 = s(t_0).
        \]
  \item Leid af naar $\tau$ en verkrijg de harmonische oscillator
        \[
        \frac{\dd^2 s}{\dd\tau^2} = -\frac{g}{4R}\,s ;
        \]
        los haar op met \cref{thm:b1:diffeq:homogeneous2} en de
        beginvoorwaarden: $s(\tau) = s_0\cos(\omega\tau)$ met
        $\omega = \sqrt{g/4R}$.
  \item Leid de \emph{tautochrone eigenschap} af (Huygens, 1659):
        de tijd om de bodem te bereiken,
        \[
        T_{\downarrow} = \frac{\pi}{2\omega}
        = \pi\sqrt{\frac Rg}\,,
        \]
        hangt niet af van het punt van loslaten --- kraaltjes die
        tegelijk vanaf willekeurige hoogten van de schaal worden
        losgelaten, komen tegelijk aan.
  \item Ga door invullen na dat $s(\tau) = s_0\cos\omega\tau$
        exact aan de energievergelijking van eerste orde uit vraag
        15 voldoet (en niet alleen aan de \omterm{def:b1:derivative:def}{afgeleide} ervan), en leg
        in één zin uit waarom een cirkelvormige slinger slechts
        \emph{bij benadering} isochroon is terwijl de cycloïde dat
        exact is.
  \item Numeriek: welke straal $R$ maakt de daaltijd precies één
        seconde ($g = 9.81$)? Merk op hoe \omterm{def:b1:topology:dense}{dicht} het antwoord bij
        één meter ligt, en hoe de volledige trillingstijd
        $2\pi\sqrt{4R/g}$ zich verhoudt tot de formule
        $2\pi\sqrt{\ell/g}$ voor een slinger bij kleine hoek met
        $\ell = 4R$.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Dividenden, en synthese.}
\begin{enumerate}[resume]
  \item Pas de raaklijnregel van vraag 3 toe bij $t = \frac\pi2$
        (neem $R = 1$): bereken $M$, $T$ en de richting van $MT$, en
        toets die aan $f'(\frac\pi2)$.
  \item (Trochoïden) Markeer in plaats daarvan een punt op afstand
        $d$ van het middelpunt ($d \neq R$): de kromme is $x = Rt -
        d\sin t$, $y = R - d\cos t$. Toon aan dat de kromme voor $d
        < R$ overal regulier is en geen enkel singulier gedrag
        vertoont ($x' > 0$: zij schuift op), terwijl voor $d > R$
        de abscis $x'$ van teken wisselt en de kromme lussen maakt
        --- het spoorwiel met flens waarvan de punten van de \omterm{def:b1:topology:closure}{rand}
        \emph{achteruit} reizen.
  \item (Voorproefje van de brachistochroon) Vergelijk de daaltijd
        $\pi\sqrt{R/g}$ langs de cycloïde van het \omterm{ex:b1:curves:astroid}{keerpunt} $(\pi R,
        2R)$ van de schaal tot de bodem met de tijd langs de rechte
        glijbaan die dezelfde punten verbindt \emph{(constante
        versnelling $g\sin\alpha$ langs de koorde)}: toon aan dat
        de koorde $\sqrt{\pi^2 + 4}\,\sqrt{R/g} \approx
        3.72\sqrt{R/g}$ kost. De kromme verslaat de rechte --- zij
        is zelfs de snelste van alle krommen, een resultaat van de
        variatierekening.
  \item Toon aan dat op de boog ($0 < t < 2\pi$)
        \[
        \frac{\dd y}{\dd x} = \cot\frac t2,
        \qquad
        \frac{\dd^2 y}{\dd x^2} =
        -\frac{1}{4R\sin^4\frac t2} < 0 :
        \]
        de boog is concaaf, met verticale raaklijnen precies in de
        \omterm{ex:b1:curves:astroid}{keerpunten}.
  \item Vind de verticale raaklijn in het \omterm{ex:b1:curves:astroid}{keerpunt} uit
        \cref{exo:b1:curves:2} meetkundig terug: bereken de
        limietrichting van de koorde $MT$ uit vraag 3 als $t \to
        0^{+}$, zonder ook maar één ontwikkeling.
  \item Synthese, in vier zinnen: welke drie stellingen met namen
        deze opgave heeft bewezen (met hun getallen $8R$, $3\pi
        R^2$, $\pi\sqrt{R/g}$ en hun auteurs); welk enkele
        rekenmiddel (het afsplitsen van $\sin\frac t2$ en
        $\cos\frac t2$) alle delen I, II en IV aandreef; hoe de
        intrinsieke betrekking $y = s^2/8R$ meetkunde omzette in
        een \omterm{def:b1:linmaps:def}{lineaire} differentiaalvergelijking; en op welk
        hoofdstuk van dit boek elk deel steunde.
\end{enumerate}
\end{problem}
