\chapter{Topologie van de reële rechte}\label{ch:b1:topology}

Limieten verwijzen telkens naar hetzelfde meetkundige vocabulaire:
punten ``dicht bij'' een \omterm{def:b1:logic:sets}{verzameling}, \omterm{def:b1:logic:sets}{verzamelingen} ``zonder lekken aan
de \omterm{def:b1:topology:closure}{rand}'', \omterm{prop:b1:reals:intervals}{intervallen} waaruit rijen niet kunnen ontsnappen. Dit
hoofdstuk legt dat vocabulaire vast --- \omterm{def:b1:topology:open}{open} en \omterm{def:b1:topology:closed}{gesloten verzamelingen},
\omterm{def:b1:topology:closure}{inwendige} en \omterm{def:b1:topology:closure}{afsluiting}, \omterm{def:b1:topology:dense}{dichtheid} --- op de reële rechte, en bewijst de
compactheid van segmenten in haar rijenvorm. Dezelfde begrippen, in
genormeerde vectorruimten, zijn stof voor het tweede jaar; op $\R$ zijn
ze binnen bereik en meteen nuttig voor \cref{ch:b1:continuity}.

\section{Open verzamelingen, gesloten verzamelingen}

\begin{definition}[Omgeving, open verzameling]\label{def:b1:topology:open}
Een \omterm{def:b1:logic:sets}{verzameling} $V \subseteq \R$ is een \emph{omgeving}\index{omgeving}
van $x \in \R$ wanneer ze voor een zekere $r > 0$ een \omterm{prop:b1:reals:intervals}{interval} $\intoo{x
- r}{x + r}$ bevat. Een \omterm{def:b1:logic:sets}{verzameling} $U \subseteq \R$ heet
\emph{open}\index{open verzameling} wanneer ze een omgeving van elk van
haar punten is:
\[
\forall x \in U,\ \exists r > 0, \quad \intoo{x - r}{x + r}
\subseteq U .
\]
\end{definition}

\begin{example}\label{ex:b1:topology:open}
\omterm{def:b1:topology:open}{Open} \omterm{prop:b1:reals:intervals}{intervallen} zijn \omterm{def:b1:topology:open}{open}: voor $x \in \intoo{a}{b}$ neem je $r =
\min(x - a,\, b - x) > 0$. Halfrechten $\intoo{a}{+\infty}$ zijn \omterm{def:b1:topology:open}{open};
$\R$ en $\emptyset$ zijn \omterm{def:b1:topology:open}{open} (de laatste bij gebrek aan punten).
$\intcc{0}{1}$ is \emph{niet} \omterm{def:b1:topology:open}{open}: geen enkel \omterm{prop:b1:reals:intervals}{interval} rond $0$ blijft
binnen.
\end{example}

\begin{proposition}[Stabiliteit van open verzamelingen]\label{prop:b1:topology:openstable}
Elke vereniging van \omterm{def:b1:topology:open}{open verzamelingen} is \omterm{def:b1:topology:open}{open}; een \emph{eindige}
doorsnede van \omterm{def:b1:topology:open}{open verzamelingen} is \omterm{def:b1:topology:open}{open}. Oneindige doorsneden kunnen
falen: $\bigcap_{n \geq 1} \intoo{-\frac 1n}{\frac 1n} = \{0\}$, niet
\omterm{def:b1:topology:open}{open}.
\end{proposition}

\begin{proof}
Vereniging: is $x \in \bigcup_i U_i$, dan is $x \in U_{i_0}$ voor een
zekere $i_0$, en het \omterm{prop:b1:reals:intervals}{interval} dat $U_{i_0}$ levert ligt binnen de
vereniging. Eindige doorsnede: is $x \in U_1 \cap \dots \cap U_k$, neem
dan $r = \min(r_1, \dots, r_k) > 0$ van de stralen die elke $U_j$ levert.
Voor het tegenvoorbeeld: elk \omterm{prop:b1:reals:intervals}{interval} rond $0$ bevat een zekere
$\frac{1}{n}$ (Archimedes) en verlaat dus de doorsnede.
\end{proof}

\begin{example}[Openheid waarborgen met expliciete stralen]\label{ex:b1:topology:radii}
Is $U = \{x \in \R : x^2 > 2\}$ \omterm{def:b1:topology:open}{open}? Ja, en het bewijsstuk laat zich
opschrijven: $U = \intoo{-\infty}{-\sqrt2} \cup \intoo{\sqrt2}{+\infty}$,
een vereniging van twee \omterm{def:b1:topology:open}{open} halfrechten, \omterm{def:b1:topology:open}{open} volgens
\cref{prop:b1:topology:openstable}. Je kunt ook punt voor punt
redeneren: voor $x \in U$ met $x > \sqrt 2$ neem je $r = x - \sqrt2 > 0$:
elke $y \in \intoo{x - r}{x + r}$ voldoet aan $y > \sqrt 2$ en dus $y^2 >
2$; symmetrisch aan de linkerkant. Beide stijlen doen ertoe --- de
structurele (bouw uit bekende \omterm{def:b1:topology:open}{open verzamelingen} met verenigingen en
eindige doorsneden) schaalt beter, de $\varepsilon$-stijl werkt wanneer
er geen structuur zichtbaar is; en \cref{ch:b1:continuity} voegt er een
derde en krachtigste aan toe: $U$ is het \omterm{def:b1:logic:map}{origineel} van de \omterm{def:b1:topology:open}{open
verzameling} $\intoo{2}{+\infty}$ onder de continue $x \mapsto x^2$.
\end{example}

\begin{definition}[Gesloten verzameling]\label{def:b1:topology:closed}
Een \omterm{def:b1:logic:sets}{verzameling} $F \subseteq \R$ heet
\emph{gesloten}\index{gesloten verzameling} wanneer haar complement $\R
\setminus F$ \omterm{def:b1:topology:open}{open} is. Wegens De Morgan en
\cref{prop:b1:topology:openstable}: elke doorsnede van gesloten
\omterm{def:b1:logic:sets}{verzamelingen} is gesloten, en eindige verenigingen van gesloten
\omterm{def:b1:logic:sets}{verzamelingen} zijn gesloten.
\end{definition}

\begin{theorem}[Karakterisering van gesloten verzamelingen met rijen]\label{thm:b1:topology:seqclosed}
$F$ is \omterm{def:b1:topology:closed}{gesloten} dan en slechts dan als voor elke rij $(u_n)$ punten van
$F$ die naar een zekere $\ell \in \R$ convergeert, de limiet $\ell$ tot
$F$ behoort. (``\omterm{def:b1:topology:closed}{Gesloten}'' $=$ ``stabiel onder limieten''.)
\end{theorem}

\begin{proof}
($\Rightarrow$) Zij $F$ \omterm{def:b1:topology:closed}{gesloten}, $u_n \in F$, $u_n \to \ell$, en stel
$\ell \notin F$. Het complement is \omterm{def:b1:topology:open}{open}: een zeker $\intoo{\ell - r}{\ell
+ r}$ mijdt $F$. Maar de convergentie legt $u_n$ voor grote $n$ in dat
\omterm{prop:b1:reals:intervals}{interval}: in tegenspraak met $u_n \in F$.

($\Leftarrow$) Stel dat $F$ niet \omterm{def:b1:topology:closed}{gesloten} is: het complement is niet
\omterm{def:b1:topology:open}{open}, dus is er een $x \notin F$ zonder \omterm{prop:b1:reals:intervals}{interval} $\intoo{x - r}{x + r}$
binnen het complement; met $r = \frac{1}{n+1}$ kies je $u_n \in F$ met
$\abs{u_n - x} < \frac{1}{n+1}$. Dan is $u_n \in F$ en $u_n \to x \notin
F$: de eigenschap met rijen faalt.
\end{proof}

\begin{example}[De toets met rijen, beide kanten op]\label{ex:b1:topology:seqtest}
\emph{\omterm{def:b1:topology:closed}{Gesloten}:} $F = \Z \cup \bigl\{n + \frac1n : n \geq 2\bigr\}$. Zij
$u_k \in F$ met $u_k \to \ell$. Het venster $\intcc{\ell - 1}{\ell + 1}$
bevat maar eindig veel punten van $F$ (eindig veel gehele getallen,
eindig veel $n + \frac1n$), en voorbij een zekere rang liggen alle $u_k$
erin: de rij neemt dan eindig veel waarden aan en is, convergerend,
uiteindelijk constant (als in \cref{exo:b1:topology:3}): $\ell \in F$.
\omterm{def:b1:topology:closed}{Gesloten} --- ook al bevat $F$ paren punten op willekeurig kleine afstand
$\frac1n$.

\emph{Niet \omterm{def:b1:topology:closed}{gesloten}:} $G = \bigl\{\frac1m + \frac1n : m, n \in
\N^*\bigr\}$. De rij $\frac1n + \frac1n \in G$ gaat naar $0$, en $0
\notin G$ (een som van twee positieve termen): de toets met rijen faalt,
$G$ is niet \omterm{def:b1:topology:closed}{gesloten}. Interessant genoeg behoort elke $\frac1m$ \emph{wel}
tot $\overline G \cap G$: immers $\frac1m = \frac{1}{m+1} +
\frac{1}{m(m+1)} \in G$. Het inzicht tot slot: om geslotenheid te
bewijzen moet je \emph{alle} convergente rijen tegelijk beheersen
(meestal via een argument van lokale eindigheid of een \omterm{def:b1:topology:closed}{gesloten} formule);
om haar te weerleggen volstaat één goedgekozen ontsnappende rij --- die
asymmetrie maakt de negatieve richting de makkelijke, en alle
tegenvoorbeelden van dit hoofdstuk hebben die vorm van één regel.
\end{example}

\begin{example}\label{ex:b1:topology:closed}
Segmenten $\intcc{a}{b}$, halfrechten $\intco{a}{+\infty}$, \omterm{def:b1:counting:card}{eindige
verzamelingen} en $\Z$ (een convergente rij gehele getallen is
uiteindelijk constant) zijn \omterm{def:b1:topology:closed}{gesloten}. $\intoc{0}{1}$ is \omterm{def:b1:topology:open}{open} noch
\omterm{def:b1:topology:closed}{gesloten} (\omterm{def:b1:topology:open}{open} faalt in $1$; \omterm{def:b1:topology:closed}{gesloten} faalt want $\frac 1n \to 0 \notin$
de \omterm{def:b1:logic:sets}{verzameling}): de meeste \omterm{def:b1:logic:sets}{verzamelingen} zijn geen van beide. $\R$ en
$\emptyset$ zijn zowel \omterm{def:b1:topology:open}{open} als \omterm{def:b1:topology:closed}{gesloten} --- en het zijn de enige zulke
deelverzamelingen van $\R$ (\cref{exo:b1:topology:9}).
\end{example}

\begin{example}[Een open verzameling uit oneindig veel stukken]\label{ex:b1:topology:rminusz}
$\R \setminus \Z = \bigcup_{n \in \Z} \intoo{n}{n+1}$: een oneindige
vereniging van \omterm{def:b1:topology:open}{open} \omterm{prop:b1:reals:intervals}{intervallen}, \omterm{def:b1:topology:open}{open} volgens
\cref{prop:b1:topology:openstable} --- zodat $\Z$ \omterm{def:b1:topology:closed}{gesloten} is zonder enig
argument met rijen. Let op de taakverdeling in de stabiliteitsregels:
\emph{verenigingen} van \omterm{def:b1:topology:open}{open verzamelingen} mogen willekeurig zijn (elk
punt heeft alleen zijn eigen bewijsstuk nodig, geleverd door de ene
\omterm{def:b1:logic:sets}{verzameling} die het bevat), terwijl \emph{doorsneden} eindig moeten
blijven (de bewijsstukken moeten doorsneden worden, en oneindig veel
stralen kunnen tot niets krimpen). \Cref{exo:b1:topology:10} zal laten
zien dat dit voorbeeld de algemene vorm is: elke \omterm{def:b1:topology:open}{open} deelverzameling van
$\R$ is een aftelbare disjuncte vereniging van \omterm{def:b1:topology:open}{open} \omterm{prop:b1:reals:intervals}{intervallen}.
\end{example}

\section{Inwendige, afsluiting, dichtheid}

\begin{definition}[Inwendige, afsluiting, rand]\label{def:b1:topology:closure}
Zij $A \subseteq \R$.
\begin{itemize}
  \item Een punt $x$ is \emph{inwendig} aan $A$ wanneer $A$ een \omterm{def:b1:topology:open}{omgeving}
        van $x$ is; het \emph{inwendige} $\mathring{A}$\index{inwendige}
        is de \omterm{def:b1:logic:sets}{verzameling} van de inwendige punten.
  \item Een punt $x$ is \emph{adherent} aan $A$ wanneer elke \omterm{def:b1:topology:open}{omgeving}
        van $x$ de \omterm{def:b1:logic:sets}{verzameling} $A$ ontmoet; de \emph{afsluiting}
        $\overline{A}$\index{afsluiting} is de \omterm{def:b1:logic:sets}{verzameling} van de
        adherente punten.
  \item De \emph{rand} is $\partial A = \overline A \setminus \mathring
        A$.\index{rand}
\end{itemize}
Er geldt $\mathring A \subseteq A \subseteq \overline A$.
\end{definition}

\begin{proposition}[Hoofdeigenschappen]\label{prop:b1:topology:closureprops}
\begin{enumerate}
  \item $\mathring A$ is de grootste \omterm{def:b1:topology:open}{open verzameling} bevat in $A$; $A$
        is \omterm{def:b1:topology:open}{open} precies wanneer $A = \mathring A$.
  \item $\overline A$ is de kleinste \omterm{def:b1:topology:closed}{gesloten verzameling} die $A$ bevat;
        $A$ is \omterm{def:b1:topology:closed}{gesloten} precies wanneer $A = \overline A$.
  \item (Karakterisering van adherentie met rijen) $x \in \overline A$
        dan en slechts dan als $x$ de limiet is van een rij punten van
        $A$.
  \item Complementeren verwisselt de begrippen: $\R \setminus \overline
        A = \bigl(\R \setminus A\bigr)^{\!\circ}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(4) $x \notin \overline A$ $\iff$ een zekere \omterm{def:b1:topology:open}{omgeving} van $x$ mijdt $A$
$\iff$ een zeker \omterm{prop:b1:reals:intervals}{interval} rond $x$ ligt in $\R \setminus A$ $\iff$ $x$ is
\omterm{def:b1:topology:closure}{inwendig} aan $\R \setminus A$.

(1) $\mathring A$ is \omterm{def:b1:topology:open}{open}: is $x \in \mathring A$, dan is een zeker
$\intoo{x-r}{x+r} \subseteq A$; elk punt $y$ van dat \omterm{prop:b1:reals:intervals}{interval} heeft er een
kleiner \omterm{prop:b1:reals:intervals}{interval} omheen dat erin ligt en dus in $A$: het hele \omterm{prop:b1:reals:intervals}{interval}
ligt in $\mathring A$. Elke \omterm{def:b1:topology:open}{open} $U \subseteq A$ bestaat uit \omterm{def:b1:topology:closure}{inwendige}
punten van $A$, dus $U \subseteq \mathring A$: de grootste. De
karakterisering van openheid volgt.

(2) In detail. Volgens (4) is $\R \setminus \overline A$ het \omterm{def:b1:topology:closure}{inwendige}
van $\R \setminus A$, een \omterm{def:b1:topology:open}{open verzameling} volgens (1): dus is $\overline
A$ \omterm{def:b1:topology:closed}{gesloten}, en ze bevat $A$. Minimaliteit: zij $F \supseteq A$ \omterm{def:b1:topology:closed}{gesloten}.
Dan is $\R \setminus F$ \omterm{def:b1:topology:open}{open} en bevat in $\R \setminus A$, dus geeft de
maximaliteit in (1)
\[
\R \setminus F \subseteq \bigl(\R \setminus A\bigr)^{\!\circ}
= \R \setminus \overline A ,
\]
en opnieuw complementeren geeft $\overline A \subseteq F$. Dus is
$\overline A$ de kleinste \omterm{def:b1:topology:closed}{gesloten} oververzameling. Karakterisering: is
$A = \overline A$, dan is $A$ \omterm{def:b1:topology:closed}{gesloten} (zojuist aangetoond); is $A$
\omterm{def:b1:topology:closed}{gesloten}, dan is ze zelf een \omterm{def:b1:topology:closed}{gesloten} oververzameling van $A$, zodat de
minimaliteit $\overline A \subseteq A$ afdwingt, en dus gelijkheid.

(3) Is $u_n \in A$ met $u_n \to x$, dan bevat elke \omterm{def:b1:topology:open}{omgeving} van $x$ een
zekere $u_n \in A$, dus $x \in \overline A$. Omgekeerd, is $x \in
\overline A$, dan ontmoet elk \omterm{prop:b1:reals:intervals}{interval} $\intoo{x - \frac{1}{n+1}}{x +
\frac{1}{n+1}}$ de \omterm{def:b1:logic:sets}{verzameling} $A$ in een zekere $u_n$, en $u_n \to x$.
\end{proof}

\begin{example}\label{ex:b1:topology:closureexamples}
$\overline{\intoo{0}{1}} = \intcc{0}{1}$; $\mathring{\intcc{0}{1}} =
\intoo{0}{1}$; $\partial\intoo{0}{1} = \{0, 1\}$. Voor $A = \{\frac 1n :
n \in \N^*\}$ is $\overline A = A \cup \{0\}$, $\mathring A = \emptyset$
en $\partial A = A \cup \{0\}$. Voor $\Q$: wegens de \omterm{def:b1:topology:dense}{dichtheid}
(\cref{thm:b1:reals:density}) is elk reëel getal adherent aan $\Q$, dus
$\overline{\Q} = \R$, terwijl $\mathring{\Q} = \emptyset$ (elk \omterm{prop:b1:reals:intervals}{interval}
bevat irrationale getallen): de \omterm{def:b1:topology:closure}{rand} van $\Q$ is heel $\R$.
\end{example}

\begin{example}[Een volledige anatomie]\label{ex:b1:topology:anatomy}
Zij $A = \intoc{0}{1} \,\cup\, \bigl(\Q \cap \intoo{2}{3}\bigr) \,\cup\,
\{4\}$. We berekenen de drie \omterm{def:b1:logic:sets}{verzamelingen} van
\cref{def:b1:topology:closure}, stuk voor stuk.

\emph{\omterm{def:b1:topology:closure}{Inwendige}.} Een punt van $\intoo{0}{1}$ heeft een heel \omterm{prop:b1:reals:intervals}{interval}
binnen $A$: \omterm{def:b1:topology:closure}{inwendig}. Het punt $1$: elk \omterm{prop:b1:reals:intervals}{interval} eromheen lekt rechts van
$1$, waar $A$ niets heeft tot $2$: niet \omterm{def:b1:topology:closure}{inwendig}. Geen enkel punt van $\Q
\cap \intoo{2}{3}$ is \omterm{def:b1:topology:closure}{inwendig} (elk \omterm{prop:b1:reals:intervals}{interval} bevat irrationale getallen,
\cref{thm:b1:reals:density}); en het geïsoleerde punt $4$ evenmin. Dus is
$\mathring A = \intoo{0}{1}$.

\emph{\omterm{def:b1:topology:closure}{Afsluiting}.} Limieten van punten van $A$: heel $\intcc{0}{1}$ ($0 =
\lim \frac1n$ met $\frac 1n \in A$); heel $\intcc{2}{3}$ (elk reëel getal
daar is een limiet van rationale getallen uit het \omterm{prop:b1:reals:intervals}{interval}, opnieuw de
\omterm{def:b1:topology:dense}{dichtheid}); en $4$. Meer niet: een punt buiten $\intcc{0}{1} \cup
\intcc{2}{3} \cup \{4\}$ heeft positieve afstand tot die \omterm{def:b1:topology:closed}{gesloten
verzameling}. Dus is $\overline A = \intcc{0}{1} \cup \intcc{2}{3} \cup
\{4\}$.

\emph{\omterm{def:b1:topology:closure}{Rand}.} $\partial A = \overline A \setminus \mathring A = \{0, 1\}
\cup \intcc{2}{3} \cup \{4\}$.

Het inzicht tot slot: de drie bewerkingen werken \emph{lokaal} --- elk
stuk van $A$ draagt bij naar zijn eigen aard (een massief \omterm{prop:b1:reals:intervals}{interval}
behoudt zijn binnenkant, een dicht-maar-poreus stuk wordt volledig \omterm{def:b1:topology:closure}{rand},
een geïsoleerd punt is zuivere \omterm{def:b1:topology:closure}{rand}) --- en een tekening van $A$ van twee
regels voorspelt elk antwoord voordat er ook maar één bewijs geschreven
is.
\end{example}

\begin{remark}[Veelgemaakte fouten bij puntverzamelingen]
(i) \emph{``Niet \omterm{def:b1:topology:open}{open}'' betekent niet ``\omterm{def:b1:topology:closed}{gesloten}''}: de meeste
\omterm{def:b1:logic:sets}{verzamelingen} zijn geen van beide ($\intoc{0}{1}$), en twee
\omterm{def:b1:logic:sets}{verzamelingen} zijn allebei ($\emptyset$, $\R$) --- \omterm{def:b1:topology:open}{open} en \omterm{def:b1:topology:closed}{gesloten} zijn
geen tegengestelden maar duaal via het complement. (ii) \emph{\omterm{def:b1:topology:closure}{Inwendige}
en \omterm{def:b1:topology:closure}{afsluiting} commuteren niet}: voor $A = \Q$ is
\[
\overline{\mathring A} = \overline\emptyset = \emptyset
\qquad\text{terwijl}\qquad
\bigl(\,\overline A\,\bigr)^{\!\circ} = \mathring \R = \R :
\]
de twee samengestelde operatoren verschillen zoveel als \omterm{def:b1:logic:sets}{verzamelingen}
maar kunnen. (iii) \emph{Oneindige verenigingen van \omterm{def:b1:topology:closed}{gesloten
verzamelingen} kunnen niet \omterm{def:b1:topology:closed}{gesloten} zijn}: $\bigcup_{n\geq1}
\intcc{\frac1n}{1} = \intoc{0}{1}$ --- het spiegelbeeld van het
tegenvoorbeeld met doorsneden uit
\cref{prop:b1:topology:openstable}. (iv) \emph{\omterm{def:b1:topology:dense}{Dicht} betekent niet
groot}: $\Q$ ligt \omterm{def:b1:topology:dense}{dicht}, is aftelbaar en heeft leeg \omterm{def:b1:topology:closure}{inwendige}, en haar
complement ligt eveneens \omterm{def:b1:topology:dense}{dicht}; \omterm{def:b1:topology:dense}{dichtheid} zegt ``willekeurig dicht bij
alles'', niet ``bijna alles'' --- de \omterm{pb:b1:topology:1}{Cantorverzameling} van de
weekendopgave (\cref{pb:b1:topology:1}) maakt het tegenovergestelde punt:
een topologisch kleine \omterm{def:b1:logic:sets}{verzameling} die overaftelbaar groot is.
\end{remark}

\begin{example}[Een exact berekende afsluiting]\label{ex:b1:topology:closurecomputed}
Zij $G = \bigl\{\frac1m + \frac1n : m, n \in \N^*\bigr\}$ (uit
\cref{ex:b1:topology:seqtest}). Bewering:
\[
\overline G = G \,\cup\, \Bigl\{\frac1m : m \in \N^*\Bigr\}
\,\cup\, \{0\} .
\]
($\supseteq$) $\frac1m = \lim_n \bigl(\frac1m + \frac1n\bigr)$ en $0 =
\lim_n \frac2n$: adherent wegens de karakterisering met rijen.
($\subseteq$) Zij $x = \lim_k \bigl( \frac{1}{m_k} + \frac{1}{n_k}\bigr)$;
orden elk paar zó dat $m_k \leq n_k$. Is $(m_k)$ onbegrensd, dan heeft
een \omterm{def:b1:seq:subsequence}{deelrij} $m_k \to \infty$, dus ook $n_k \to \infty$ en $x = 0$.
Anders neemt $(m_k)$ eindig veel waarden aan, waarvan er één, zeg $m$,
oneindig vaak voorkomt; langs die \omterm{def:b1:seq:subsequence}{deelrij} gaat $\frac{1}{n_k} \to x -
\frac1m$: is $(n_k)$ begrensd, dan neemt ze een zekere waarde $n$
oneindig vaak aan en is $x = \frac1m + \frac1n \in G$; zo niet, dan is $x
= \frac1m$. Elk geval landt in de aangekondigde \omterm{def:b1:logic:sets}{verzameling}. Het inzicht
tot slot: een \omterm{def:b1:topology:closure}{afsluiting} berekenen is een gevalsonderzoek op de indices
in compactheidsstijl --- een begrensde index betekent eindig veel waarden
(duivenhok), een onbegrensde index betekent een ontsnappende limiet ---
en het antwoord vertoont de typische structuur van limietpunten in twee
lagen: de \omterm{def:b1:logic:sets}{verzameling}, haar limieten van de eerste generatie, en hun
limiet $0$.
\end{example}

\begin{definition}[Dichtheid, topologische vorm]\label{def:b1:topology:dense}
$A$ heet \emph{dicht}\index{dichtheid} in $\R$ wanneer $\overline A = \R$
--- equivalent: elk niet-leeg \omterm{def:b1:topology:open}{open} \omterm{prop:b1:reals:intervals}{interval} ontmoet $A$; equivalent (via
\cref{prop:b1:topology:closureprops}~(3)): elk reëel getal is een limiet
van elementen van $A$. Voorbeelden: $\Q$, $\R \setminus \Q$, de dyadische
getallen (\cref{exo:b1:reals:8}), dichte \omterm{def:b1:structures:subgroup}{deelgroepen}
(\cref{exo:b1:reals:9}).
\end{definition}

\begin{example}[Dichtheid is relatief]\label{ex:b1:topology:relativedensity}
``\omterm{def:b1:topology:dense}{Dicht}'' zoals hier gedefinieerd betekent \omterm{def:b1:topology:dense}{dicht} \emph{in $\R$}; een
\omterm{def:b1:logic:sets}{verzameling} kan in plaats daarvan slechts in een deel van de rechte \omterm{def:b1:topology:dense}{dicht}
liggen. De dyadische getallen van $\intcc{0}{1}$, dus $D \cap
\intcc{0}{1}$ (\cref{exo:b1:reals:8}), ontmoeten elk \omterm{def:b1:topology:open}{open} \omterm{prop:b1:reals:intervals}{interval} dat in
$\intcc{0}{1}$ ligt, maar missen $\intoo{2}{3}$ natuurlijk volledig: ze
liggen \omterm{def:b1:topology:dense}{dicht} \emph{in} $\intcc{0}{1}$, wat betekent $\overline{D \cap
\intcc{0}{1}} = \intcc{0}{1}$. De algemene uitdrukking ``$A$ ligt \omterm{def:b1:topology:dense}{dicht}
in $B$'' staat kort voor $B \subseteq \overline A$ --- benoem altijd de
omgevende \omterm{def:b1:logic:sets}{verzameling}, want de eindpunten uit de weekendopgave liggen
\omterm{def:b1:topology:dense}{dicht} \emph{in de \omterm{pb:b1:topology:1}{Cantorverzameling}} terwijl ze nergens \omterm{def:b1:topology:dense}{dicht} liggen
\emph{in $\R$}: dezelfde \omterm{def:b1:logic:sets}{verzameling}, twee waarheidsgetrouwe en
tegengesteld klinkende beschrijvingen.
\end{example}

\begin{example}[Met dichtheid omgaan]\label{ex:b1:topology:densitytricks}
Drie snelle zetten die voortdurend terugkeren. \emph{Vergroten}: ligt $A$
\omterm{def:b1:topology:dense}{dicht} en is $A \subseteq B$, dan ligt $B$ \omterm{def:b1:topology:dense}{dicht} (elk \omterm{prop:b1:reals:intervals}{interval} ontmoet $A$
al). \emph{Vervoeren}: ligt $A$ \omterm{def:b1:topology:dense}{dicht}, dan ook $\lambda A + \mu$ voor
$\lambda \neq 0$ --- een \omterm{prop:b1:reals:intervals}{interval} $I$ ontmoet $\lambda A + \mu$ precies
wanneer het \omterm{prop:b1:reals:intervals}{interval} $\frac{I - \mu}{\lambda}$ de \omterm{def:b1:logic:sets}{verzameling} $A$
ontmoet; zo liggen bijvoorbeeld de oneven veelvouden van $10^{-9}$ \omterm{def:b1:topology:dense}{dicht}.
\emph{Doorsnijden faalt}: twee \omterm{def:b1:topology:dense}{dichte} \omterm{def:b1:logic:sets}{verzamelingen} kunnen elkaar
volledig missen ($\Q$ en $\R \setminus \Q$): \omterm{def:b1:topology:dense}{dichtheid} overleeft
verenigingen en affiene \omterm{def:b1:logic:map}{afbeeldingen}, nooit doorsneden.
\end{example}

\section{Compactheid van segmenten}

\begin{theorem}[Segmenten zijn rijencompact]\label{thm:b1:topology:compact}
Zij $a \leq b$. Elke rij punten van $\intcc{a}{b}$ heeft een \omterm{def:b1:seq:subsequence}{deelrij} die
naar een punt \emph{van} $\intcc{a}{b}$
convergeert.\index{compactheid}

Algemener zijn de deelverzamelingen van $\R$ met die eigenschap (elke rij
heeft een \omterm{def:b1:seq:subsequence}{deelrij} die in de \omterm{def:b1:logic:sets}{verzameling} convergeert) precies de
\emph{\omterm{def:b1:topology:closed}{gesloten} en begrensde} \omterm{def:b1:logic:sets}{verzamelingen}.
\end{theorem}

\begin{proof}
Een rij in $\intcc{a}{b}$ is begrensd, dus extraheert
Bolzano--Weierstrass (\cref{thm:b1:seq:bw}) een convergente \omterm{def:b1:seq:subsequence}{deelrij}; haar
limiet blijft in $\intcc{a}{b}$ omdat segmenten \omterm{def:b1:topology:closed}{gesloten} zijn
(\cref{thm:b1:topology:seqclosed}).

Algemeen geval. \emph{(\omterm{def:b1:topology:closed}{Gesloten} en begrensd $\Rightarrow$ compact):} zij
$F$ \omterm{def:b1:topology:closed}{gesloten} en begrensd, en $(u_n)$ een rij in $F$. De begrensdheid van
$F$ begrenst de rij, dus extraheert Bolzano--Weierstrass een
$u_{\varphi(n)} \to \ell$; en $\ell \in F$ omdat $F$ \omterm{def:b1:topology:closed}{gesloten} is en de
\omterm{def:b1:seq:subsequence}{deelrij} een convergente rij punten van $F$ is
(\cref{thm:b1:topology:seqclosed}): de twee hypothesen worden elk één
keer verbruikt, de begrensdheid voor het bestaan van de limiet, de
geslotenheid voor haar lidmaatschap. \emph{(Compact $\Rightarrow$
\omterm{def:b1:topology:closed}{gesloten} en begrensd):} is $F$ onbegrensd, kies dan $u_n \in F$ met
$\abs{u_n} \geq n$; elke \omterm{def:b1:seq:subsequence}{deelrij} is onbegrensd en dus divergent
(\cref{prop:b1:seq:first}): geen enkele convergente \omterm{def:b1:seq:subsequence}{deelrij}. Is $F$ niet
\omterm{def:b1:topology:closed}{gesloten}, neem dan $u_n \in F$ met $u_n \to \ell \notin F$
(\cref{thm:b1:topology:seqclosed}): elke \omterm{def:b1:seq:subsequence}{deelrij} convergeert naar $\ell
\notin F$, zodat geen enkele \omterm{def:b1:seq:subsequence}{deelrij} \emph{in} $F$ convergeert.
\end{proof}

\begin{example}[Geneste compacte verzamelingen]\label{ex:b1:topology:nested}
Een eerste oefening op de stelling. Zij $K_0 \supseteq K_1 \supseteq K_2
\supseteq \dots$ niet-lege compacte (\omterm{def:b1:topology:closed}{gesloten} en begrensde)
deelverzamelingen van $\R$. Dan is $\bigcap_n K_n \neq \emptyset$. Kies
immers voor elke $n$ een $x_n \in K_n$: de rij leeft in de compacte
$K_0$, dus convergeert een \omterm{def:b1:seq:subsequence}{deelrij} $x_{\varphi(n)}$ naar een zekere $x$
(\cref{thm:b1:topology:compact}). Voor elke vaste $m$ liggen de termen
$x_{\varphi(n)}$ met $\varphi(n) \geq m$ alle in de \omterm{def:b1:topology:closed}{gesloten verzameling}
$K_m$, zodat de limiet $x$ in $K_m$ ligt
(\cref{thm:b1:topology:seqclosed}); en omdat $m$ willekeurig was, is $x
\in \bigcap_n K_n$. Een nuttige metgezel: bevat een \omterm{def:b1:topology:open}{open verzameling} $U$
de doorsnede $\bigcap_n K_n$, dan is $U \supseteq K_n$ voor een zekere
$n$ --- pas hetzelfde argument toe op punten $x_n \in K_n \setminus U$;
de limiet $x$ zou in $\bigcap K_n \subseteq U$ liggen, terwijl $U$ \omterm{def:b1:topology:open}{open}
zijn afdwingt dat $x_{\varphi(n)} \in U$ voorbij een zekere rang, een
tegenspraak. Beide \omterm{def:b1:logic:statement}{uitspraken} falen zonder compactheid: $\bigcap_n
\intoo{0}{\frac 1n} = \emptyset$ en $\bigcap_n \intco{n}{+\infty} =
\emptyset$. Het inzicht tot slot: de compactheid zet een oneindige keten
beweringen over niet-leegheid om in één enkel limietpunt --- ze is het
gereedschap dat de overgang naar de oneindige doorsnede overleeft, en de
weekendopgave (\cref{pb:b1:topology:1}) zal er tweemaal op steunen.
\end{example}

\begin{example}[Randen onder verenigingen]\label{ex:b1:topology:boundaryunion}
Steeds geldt $\partial(A \cup B) \subseteq \partial A \cup \partial B$:
een punt van $\partial(A \cup B)$ heeft elke \omterm{def:b1:topology:open}{omgeving} die $A \cup B$
ontmoet (dus $A$ of $B$, oneindig vaak een van beide) en die het
complement van $A \cup B$ ontmoet, dat in beide complementen ligt --- een
korte controle legt het punt dan in $\partial A$ of $\partial B$. De
inclusie kan spectaculair strikt zijn: met $A = \Q$ en $B =
\R\setminus\Q$ is
\[
\partial(A \cup B) = \partial \R = \emptyset ,
\qquad
\partial A \cup \partial B = \R \cup \R = \R :
\]
twee rafelige \omterm{def:b1:logic:sets}{verzamelingen} kunnen tot één naadloze samensmelten, waarbij
hun \omterm{def:b1:topology:closure}{randen} elkaar vernietigen. Het inzicht tot slot: \omterm{def:b1:topology:closure}{inwendigen} en
\omterm{def:b1:topology:closure}{afsluitingen} gedragen zich \emph{monotoon} onder verenigingen en
doorsneden, \omterm{def:b1:topology:closure}{randen} niet --- behandel $\partial$ als een afgeleide
grootheid ($\overline A \setminus \mathring A$) en nooit als een operator
met een eigen algebra.
\end{example}

\begin{remark}[Perspectieven binnen dit volume]
Het hier opgebouwde vocabulaire wordt in dit boek nog tweemaal verbruikt.
In \cref{ch:b1:continuity} is elke stelling een vermomde topologische
\omterm{def:b1:logic:statement}{uitspraak}: de tussenwaardestelling zegt dat continue \omterm{def:b1:logic:map}{afbeeldingen} de
intervaleigenschap bewaren, de extremumstelling dat ze de compactheid
bewaren --- en de bewijzen roepen
\cref{thm:b1:topology:seqclosed,thm:b1:topology:compact} bij naam aan. In
\cref{ch:b1:multivar} worden dezelfde definities in $\R^2$ herlezen met
schijven in plaats van \omterm{prop:b1:reals:intervals}{intervallen}: \omterm{def:b1:topology:open}{open verzamelingen}, \omterm{def:b1:topology:closure}{afsluitingen} en
compactheid gaan woord voor woord over, en de extremumstelling in twee
variabelen rijdt opnieuw op Bolzano--Weierstrass (extraheer op elke
coördinaat). Het ene begrip dat \emph{niet} pijnloos veralgemeent is het
\omterm{prop:b1:reals:intervals}{interval} zelf --- in het vlak vervangt samenhang de convexiteit, een
verhaal dat begint bij het ``alleen $\emptyset$ en $\R$ zijn \omterm{def:b1:topology:open}{open} en
\omterm{def:b1:topology:closed}{gesloten}'' van \cref{exo:b1:topology:9}.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:topology:1}
Zeg voor elke \omterm{def:b1:logic:sets}{verzameling} of ze \omterm{def:b1:topology:open}{open}, \omterm{def:b1:topology:closed}{gesloten}, allebei of geen van beide
is (met verantwoording): $\intoo{0}{1} \cup \intoo{2}{3}$;
$\;\intco{0}{1}$; $\;\{0\} \cup \intcc{1}{2}$; $\;\R \setminus \Z$;
$\;\Q \cap \intoo{0}{1}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:topology:2}
Bepaal $\mathring A$, $\overline A$ en $\partial A$ voor: $A =
\intoc{0}{1} \cup \{2\}$; $\;A = \R \setminus \Q$; $\;A =
\bigl\{\frac{(-1)^n n}{n+1} : n \in \N\bigr\}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:topology:3}
Bewijs dat een \omterm{def:b1:counting:card}{eindige verzameling} \omterm{def:b1:topology:closed}{gesloten} is, eerst via complementen en
daarna via de karakterisering met rijen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:topology:4}
Bewijs dat voor alle $A, B \subseteq \R$ geldt $\overline{A \cup B} =
\overline A \cup \overline B$. Toon met een voorbeeld aan dat
$\overline{A \cap B}$ kan verschillen van $\overline A \cap \overline B$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:5}
Zij $u_n \to \ell$ in $\R$. Bewijs dat de \omterm{def:b1:logic:sets}{verzameling} $\{u_n : n \in \N\}
\cup \{\ell\}$ \omterm{def:b1:topology:closed}{gesloten} is (en dus compact als we erbij nemen dat ze
begrensd is --- wat ze is).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:6}
Zij $U$ \omterm{def:b1:topology:open}{open} en $A$ willekeurig. Bewijs dat $U + A = \{u + a\}$ \omterm{def:b1:topology:open}{open} is.
Leid af dat de som van een \omterm{def:b1:topology:open}{open verzameling} en een willekeurige
\omterm{def:b1:logic:sets}{verzameling} \omterm{def:b1:topology:open}{open} is, en zet daar tegenover: geef twee \emph{\omterm{def:b1:topology:closed}{gesloten}}
\omterm{def:b1:logic:sets}{verzamelingen} waarvan de som niet \omterm{def:b1:topology:closed}{gesloten} is. \emph{(Probeer $\Z$ en
$\sqrt 2\,\Z$, met \cref{exo:b1:reals:9}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:7}
Een punt $x \in A$ heet \emph{geïsoleerd} in $A$ wanneer een zekere
\omterm{def:b1:topology:open}{omgeving} van $x$ de \omterm{def:b1:logic:sets}{verzameling} $A$ alleen in $x$ ontmoet. Bewijs dat elk
punt van $\Z$ geïsoleerd is in $\Z$, dat in $A = \{\frac1n\}$ alle punten
geïsoleerd zijn terwijl $\overline A \neq A$, en dat een \omterm{def:b1:logic:sets}{verzameling}
waarvan alle punten geïsoleerd zijn een leeg \omterm{def:b1:topology:closure}{inwendige} heeft.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:8}
Bewijs dat de \omterm{def:b1:topology:closure}{afsluiting} van een begrensde \omterm{def:b1:logic:sets}{verzameling} begrensd is, en
dat $\sup \overline A = \sup A$ voor niet-lege naar boven begrensde $A$.
Leid af dat $\sup A \in \overline A$: het \omterm{def:b1:reals:bounds}{supremum} is altijd adherent.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:topology:9}
Bewijs dat de enige deelverzamelingen van $\R$ die zowel \omterm{def:b1:topology:open}{open} als
\omterm{def:b1:topology:closed}{gesloten} zijn $\emptyset$ en $\R$ zijn. \emph{Aanwijzing: stel dat $A$
\omterm{def:b1:topology:open}{open} en \omterm{def:b1:topology:closed}{gesloten} is met $A \neq \emptyset$ en $\R \setminus A \neq
\emptyset$; kies $a \in A$ en $b \notin A$, zeg $a < b$, en beschouw $s =
\sup\,(A \cap \intcc{a}{b})$; beslis of $s$ tot $A$ of tot haar
complement kan behoren.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:topology:10}
(Structuur van de \omterm{def:b1:topology:open}{open verzamelingen}) Zij $U \subseteq \R$ \omterm{def:b1:topology:open}{open} en
niet-leeg. Zij voor $x \in U$ de \omterm{def:b1:logic:sets}{verzameling} $I_x$ de vereniging van alle
\omterm{def:b1:topology:open}{open} \omterm{prop:b1:reals:intervals}{intervallen} die $x$ bevatten en in $U$ liggen. Bewijs dat $I_x$ een
\omterm{def:b1:topology:open}{open} \omterm{prop:b1:reals:intervals}{interval} is, dat twee \omterm{def:b1:logic:sets}{verzamelingen} $I_x, I_y$ gelijk of disjunct
zijn, en dat $U$ een vereniging van \emph{aftelbaar veel} twee aan twee
disjuncte \omterm{def:b1:topology:open}{open} \omterm{prop:b1:reals:intervals}{intervallen} is \emph{(kies in elk een rationaal getal)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:11}
Een punt $x \in \R$ heet een \emph{ophopingspunt} van $A$ wanneer elke
\omterm{def:b1:topology:open}{omgeving} van $x$ de \omterm{def:b1:logic:sets}{verzameling} $A \setminus \{x\}$ ontmoet; hun
\omterm{def:b1:logic:sets}{verzameling} is de \emph{afgeleide \omterm{def:b1:logic:sets}{verzameling}} $A'$. Bewijs dat
$\overline A = A \cup A'$, en dat $A$ \omterm{def:b1:topology:closed}{gesloten} is dan en slechts dan als
$A' \subseteq A$. Bepaal $A'$ voor $A = \{\frac 1n : n \in \N^*\}$, voor
$A = \Z$ en voor $A = \Q$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:topology:12}
Definieer voor niet-lege $A \subseteq \R$ de functie $d_A(x) =
\inf\{\abs{x - a} : a \in A\}$. Bewijs:
\begin{enumerate}
  \item $\abs{d_A(x) - d_A(y)} \leq \abs{x - y}$ voor alle $x, y$ ($d_A$
        is $1$-lipschitz);
  \item $d_A(x) = 0$ dan en slechts dan als $x \in \overline A$; in het
        bijzonder is $d_F(x) > 0$ wanneer $F$ \omterm{def:b1:topology:closed}{gesloten} is en $x \notin
        F$;
  \item voor elke $\varepsilon > 0$ is de \omterm{def:b1:logic:sets}{verzameling} $V_\varepsilon =
        \{x : d_F(x) < \varepsilon\}$ \omterm{def:b1:topology:open}{open}, bevat ze $F$, en is
        $\bigcap_{\varepsilon > 0} V_\varepsilon = F$ voor \omterm{def:b1:topology:closed}{gesloten} $F$:
        elke \omterm{def:b1:topology:closed}{gesloten verzameling} is een aftelbare doorsnede van \omterm{def:b1:topology:open}{open
        verzamelingen}.
\end{enumerate}
\end{exercise}

\section{Opgave: de Cantorverzameling, klein en reusachtig tegelijk}

\begin{problem}[{Weekendopgave --- de \omterm{pb:b1:topology:1}{Cantorverzameling} van de middelste
derden: lengte nul, overaftelbaar, perfect, en $C + C = \intcc{0}{2}$}]
\label{pb:b1:topology:1}
Verwijder uit $\intcc{0}{1}$ het \omterm{def:b1:topology:open}{open} middelste derde, dan het middelste
derde van elk overblijvend segment, en herhaal dat eeuwig: wat overleeft
is de \emph{\omterm{pb:b1:topology:1}{Cantorverzameling}} $C$, de fundamentele
tegenvoorbeeldenfabriek van de analyse. Deze opgave bouwt haar op, leest
haar door de machinerie van grondtal $3$ uit \cref{pb:b1:reals:1}, en
stelt haar paradoxale portret vast: totale lengte nul en toch
overaftelbaar; leeg \omterm{def:b1:topology:closure}{inwendige} en toch geen enkel geïsoleerd punt; totaal
onsamenhangend en toch vult $C + C$ het hele segment
$\intcc{0}{2}$.\index{Cantorverzameling} Formeel: $C_0 = \intcc{0}{1}$,
en $C_{n+1}$ ontstaat uit $C_n$ door uit elk segment van $C_n$ het \omterm{def:b1:topology:open}{open}
middelste derde te verwijderen; ten slotte is $C = \bigcap_{n \geq 0}
C_n$. Overal is een \emph{ternaire code} van $x \in \intcc{0}{1}$ elke
cijferrij $(d_k)_{k\geq1}$ met $d_k \in \{0, 1, 2\}$ waarvan de waarde
$\sup_n \sum_{k=1}^n d_k 3^{-k}$ gelijk is aan $x$ --- oneigenlijke codes
(uiteindelijk $2$) zijn toegestaan; volgens \cref{pb:b1:reals:1} (vragen
9--11) heeft elke $x \in \intcc{0}{1}$ één of twee codes, en twee precies
wanneer $x = m/3^N \in \intoo{0}{1}$.

\medskip
\noindent\textbf{Deel I --- De constructie.}
\begin{enumerate}
  \item Beschrijf $C_1$ en $C_2$ expliciet als verenigingen van
        segmenten, en bewijs met inductie: $C_n$ is een disjuncte
        vereniging van $2^n$ \omterm{def:b1:topology:closed}{gesloten} segmenten, elk van lengte $3^{-n}$.
  \item Toon aan dat $C$ \omterm{def:b1:topology:closed}{gesloten} en begrensd --- dus compact
        (\cref{thm:b1:topology:compact}) --- en niet-leeg is, en dat elk
        eindpunt van elk segment van elke $C_n$ tot $C$ behoort.
  \item De totale lengte van $C_n$ is $\bigl(\frac23\bigr)^n$. Leid af
        dat $C$ voor elke $\varepsilon > 0$ door eindig veel segmenten
        met totale lengte $\leq \varepsilon$ overdekt kan worden: de
        \omterm{pb:b1:topology:1}{Cantorverzameling} heeft \emph{lengte nul}.
  \item Toon aan dat een \omterm{prop:b1:reals:intervals}{interval} bevat in $C$ voor elke $n$ lengte
        $\leq 3^{-n}$ heeft en dus een singleton of leeg is: $\mathring
        C = \emptyset$. \omterm{def:b1:topology:closed}{Gesloten} met leeg \omterm{def:b1:topology:closure}{inwendige} betekent dat $C$
        \emph{nergens \omterm{def:b1:topology:dense}{dicht}} is.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De ternaire code.}
\begin{enumerate}[resume]
  \item Bewijs de recursie van de zelfgelijkvormigheid
        \[
        C_{n+1} = \tfrac13 C_n \,\cup\,
        \bigl(\tfrac23 + \tfrac13 C_n\bigr),
        \qquad\text{en dus}\qquad
        C = \tfrac13 C \,\cup\, \bigl(\tfrac23 + \tfrac13
        C\bigr),
        \]
        waarbij de twee stukken disjunct zijn: $C$ bestaat uit twee
        kopieën van zichzelf op schaal $\frac13$.
  \item Bewijs met inductie naar $n$: $x \in C_n$ dan en slechts dan als
        $x$ een ternaire code heeft waarvan de eerste $n$ cijfers in
        $\{0, 2\}$ liggen. Leid daaruit, met het feit dat $x$ hoogstens
        twee codes heeft, af: $x \in C$ dan en slechts dan als $x$ een
        code heeft \emph{zonder enig cijfer $1$} (een \emph{$1$-vrije}
        code).
  \item Codes in actie: geef $1$-vrije codes voor $0$, $1$, $\frac13$ en
        $\frac23$; toon aan dat $\frac14 = (0.\overline{02})_3$ en
        $\frac34 = (0.\overline{20})_3$, zodat beide tot $C$ behoren; en
        ga na dat $\frac14$ \emph{geen} eindpunt van enige $C_n$ is
        (eindpunten hebben de vorm $m/3^n$).
  \item Toon aan dat elke $x \in C$ \emph{precies één} $1$-vrije code
        heeft \emph{(bewijs dat, wanneer $x$ twee codes heeft, precies
        één van beide het cijfer $1$ bevat)}. Besluit: de
        waardeafbeelding is een bijectie van de rijen over $\{0,2\}$ op
        $C$.
  \item (Diagonaal) Zij $k \mapsto x_k$ een willekeurige \omterm{def:b1:logic:map}{afbeelding} $\N^*
        \to C$. Bouw een rij over $\{0, 2\}$ die op index $k$ van de
        code van $x_k$ verschilt, en besluit dat $C$ overaftelbaar is
        --- terwijl vraag 3 daarentegen zegt dat ze metrisch
        verwaarloosbaar is.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Topologisch portret.}
\begin{enumerate}[resume]
  \item Maak de balans op: $C$ is compact, overaftelbaar, van lengte
        nul, nergens \omterm{def:b1:topology:dense}{dicht}. Welke ene insluiting $C \subseteq C_n$
        draagt elk van die eigenschappen?
  \item ($C$ is perfect) Zij $x \in C$ met $1$-vrije code $(d_k)$. Het
        cijfer $d_n$ omklappen ($0 \leftrightarrow 2$) levert een $x_n
        \in C$ met $\abs{x_n - x} = 2 \cdot 3^{-n}$. Besluit dat $C$
        geen enkel geïsoleerd punt heeft: elk punt van $C$ is een limiet
        van \emph{andere} punten van $C$.
  \item Toon aan dat de eindpunten uit vraag 2 een aftelbare \omterm{def:b1:topology:dense}{dichte}
        deelverzameling van $C$ vormen \emph{(knot de code na $n$
        cijfers af en ga verder met nullen; de aftelbaarheid als in
        \cref{exo:b1:topology:10})}. Besluit: het typische punt van $C$
        --- zoals $\frac14$ --- is \emph{geen} eindpunt: de eindpunten
        vormen een aftelbaar skelet binnen een overaftelbaar \omterm{def:b1:structures:field}{lichaam}.
  \item (Totaal onsamenhangend) Zij $x < y$ in $C$. Kies $n$ met $3^{-n}
        < y - x$ en lever een punt $z \in \intoo{x}{y}$ met $z \notin
        C$. Besluit dat de enige niet-lege \omterm{prop:b1:reals:intervals}{intervallen} bevat in $C$ de
        singletons zijn.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Rekenkunde van $C$.}
\begin{enumerate}[resume]
  \item Toon aan dat $1 - C = C$ \emph{(wat doet $x \mapsto 1 - x$ met
        een $1$-vrije code? denk aan $1 = (0.\overline{2})_3$)}.
  \item (Codes optellen) Toon aan dat, hebben $x$ en $x'$ de codes
        $(a_k)$ en $(b_k)$, dan $x + x' = \lim_n\,(t_n + t'_n)$ met
        $t_n, t'_n$ de partiële sommen. Leid af: elke $y \in
        \intcc{0}{1}$ is het \emph{midden} van twee punten van $C$ ---
        kies bij een code $(e_k)$ van $y$ cijfers $a_k, b_k \in \{0,
        2\}$ met $\frac{a_k + b_k}{2} = e_k$.
  \item Besluit $C + C = \intcc{0}{2}$ en, met vraag 14, $C - C =
        \intcc{-1}{1}$. Een concreet geval: schrijf $1$ als een som van
        de twee punten die in vraag 7 gevonden werden en die geen
        eindpunten zijn.
  \item Bezin je: een \omterm{def:b1:logic:sets}{verzameling} van lengte nul waarvan de
        verschilverzameling $\intcc{-1}{1}$ vult. Waarom is er geen
        tegenspraak tussen ``$C$ is metrisch verwaarloosbaar'' en ``$C +
        C$ heeft volle lengte''? (Eén zin; denk na over wat lengte wel
        en niet beheerst.)
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Leden, rationaal en irrationaal.}
\begin{enumerate}[resume]
  \item Combineer vraag 8 met het periodiciteitscriterium uit
        \cref{pb:b1:reals:1}: een punt van $C$ is rationaal dan en
        slechts dan als zijn $1$-vrije code uiteindelijk periodiek is.
        Voer de staartdeling in grondtal $3$ uit om na te gaan dat
        $\frac1{13} = (0.\overline{002})_3 \in C$.
  \item Lever een expliciet \emph{irrationaal} lid van $C$: de waarde
        van de code met $d_k = 2$ op de driehoeksposities $k =
        \frac{j(j+1)}{2}$ en $d_k = 0$ elders. Verantwoord de
        irrationaliteit met het argument van de groeiende gaten uit
        \cref{pb:b1:reals:1} (vraag 20).
  \item (Op een heel segment) Beschouw $h$, die het punt van $C$ met
        $1$-vrije code $(d_k)$ afbeeldt op de waarde van de
        \emph{binaire} rij $\bigl(\frac{d_k}2\bigr)$, dus $h(x) = \sup_n
        \sum_{k=1}^n \frac{d_k}{2}\,2^{-k}$. Toon aan dat $h$ de
        \omterm{def:b1:logic:sets}{verzameling} $C$ \emph{op} $\intcc{0}{1}$ afbeeldt. De
        verwaarloosbare $C$ is dus \omterm{def:b1:logic:inj}{surjectief} op een segment van volle
        lengte --- een tweede bewijs dat $C$ overaftelbaar is.
  \item (Zelfgelijkvormige lengte) Stel dat er een lengtebegrip $L$ voor
        $C$ en haar gekrompen kopieën gedefinieerd zou zijn, verenigbaar
        met schaling ($L(\lambda A) = \lambda L(A)$), invariant onder
        translatie, en additief over de disjuncte ontbinding uit vraag
        5. Toon aan dat dan $L(C) = \frac23\,L(C)$, wat $L(C) = 0$
        afdwingt: de zelfgelijkvormigheid alleen al veroordeelt $C$ tot
        lengte nul.
\end{enumerate}

\medskip
\noindent\textbf{Deel VI --- Een dikke neef, en de moraal.}
\begin{enumerate}[resume]
  \item (Dikke \omterm{pb:b1:topology:1}{Cantorverzameling}) Herhaal de constructie, maar verwijder
        in stadium $n$ ($n = 0, 1, 2, \dots$) uit elk van de $2^n$
        huidige segmenten slechts een centraal \omterm{def:b1:topology:open}{open} \omterm{prop:b1:reals:intervals}{interval} van lengte
        $4^{-(n+1)}$. Toon aan dat de segmentlengtes $l_n$ voldoen aan
        $l_{n+1} = \frac{l_n - 4^{-(n+1)}}{2}$ en $l_n = \frac{2^n +
        1}{2\cdot 4^n} > 0$, dat de resulterende $K = \bigcap K_n$
        compact is met leeg \omterm{def:b1:topology:closure}{inwendige}, en dat de totale verwijderde
        lengte $\sum_{n\geq0} 2^n 4^{-(n+1)} = \frac12$ bedraagt. Neem
        de (intuïtieve, in bachelorjaar 3 bewezen) additiviteit van de
        lengte voor eindige verenigingen van \omterm{prop:b1:reals:intervals}{intervallen} aan, en toon
        met beide \omterm{def:b1:logic:statement}{uitspraken} van \cref{ex:b1:topology:nested} aan dat
        elke eindige familie \omterm{def:b1:topology:open}{open} \omterm{prop:b1:reals:intervals}{intervallen} die $K$ overdekt een
        totale lengte $\geq \frac12$ heeft: $K$ is nergens \omterm{def:b1:topology:dense}{dicht} maar
        \emph{niet} verwaarloosbaar. Kleinheid heeft verscheidene
        niet-equivalente betekenissen.
  \item (Afstanden) Toon aan dat voor een niet-lege \omterm{def:b1:topology:closed}{gesloten} $F
        \subseteq \R$ en $x \in \R$ het \omterm{def:b1:reals:bounds}{infimum} $d(x, F)$
        \emph{bereikt} wordt \emph{(een minimaliserende rij plus
        Bolzano--Weierstrass)}. Bereken vervolgens
        \[
        \max_{y \in \intcc{0}{1}} d(y, C) = \frac16 ,
        \]
        precies bereikt in het middelpunt $y = \frac12$ \emph{(een punt
        van een gat dat in stadium $n$ ontstaat ligt binnen
        $\frac{3^{-n}}{2}$ van de eindpunten van dat gat, die in $C$
        liggen)}.
  \item (Elk punt een deelrijlimiet) Lever met de vragen 12 en 2 één
        enkele rij in $C$ waarvan de \omterm{def:b1:logic:sets}{verzameling} deelrijlimieten
        \emph{heel} $C$ is. (Vergelijk: voor een convergente rij is die
        \omterm{def:b1:logic:sets}{verzameling} één punt --- $C$ verwezenlijkt het andere uiterste
        onder de compacte \omterm{def:b1:logic:sets}{verzamelingen}.)
  \item Synthese, één zin per onderdeel: (i) welke stellingen van dit
        hoofdstuk verbruikte de constructie werkelijk (stabiliteit van
        \omterm{def:b1:topology:closed}{gesloten verzamelingen}, compactheid, karakteriseringen met
        rijen)? (ii) som de vier paradoxale paren van het portret op
        (lengte nul / overaftelbaar, \omterm{def:b1:topology:closed}{gesloten} / leeg \omterm{def:b1:topology:closure}{inwendige}, perfect
        / totaal onsamenhangend, verwaarloosbaar / $C+C$ vol); (iii)
        waar duikt $C$ later weer op (de duivelstrap gebouwd op $h$ in
        de theorie van de continuïteit, en de maattheorie van het volume
        van bachelorjaar 3, waar $C$ ``aftelbaar'' van
        ``verwaarloosbaar'' scheidt)?
\end{enumerate}
\end{problem}
