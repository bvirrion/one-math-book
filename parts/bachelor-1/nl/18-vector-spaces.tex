\chapter{Vectorruimten}\label{ch:b1:vspaces}

Hier begint de lineaire algebra: de axioma's van \omterm{def:b1:vspaces:def}{vectorruimten}
isoleren wat $\R^2$, $\R^3$, veeltermruimten en functieruimten
gemeen hebben --- men kan optellen en schalen. Twee hoofdstukken
bouwen de theorie op (\cref{ch:b1:findim} voegt de dimensie toe); de
taal die zij opzetten --- \omterm{def:b1:vspaces:span}{opspansel}, \omterm{def:b1:vspaces:free}{vrije familie}, \omterm{def:b1:vspaces:free}{basis}, \omterm{def:b1:vspaces:sum}{directe
som} --- is het dagelijks brood van elk hoofdstuk daarna. Overal
staat $K$ voor $\R$ of $\C$ (de \emph{scalairen}).

\section{Definitie en voorbeelden}

\begin{definition}[Vectorruimte]\label{def:b1:vspaces:def}
Een \emph{$K$-vectorruimte}\index{vectorruimte} is een \omterm{def:b1:logic:sets}{verzameling}
$E$ met een optelling die $(E, +)$ tot een \omterm{def:b1:structures:group}{abelse groep} maakt
(nulelement genoteerd $0_E$ of $0$), en een scalaire
vermenigvuldiging $K \times E \to E$ zodat voor alle $\lambda, \mu
\in K$ en $x, y \in E$
\[
\lambda(x + y) = \lambda x + \lambda y,\quad
(\lambda + \mu) x = \lambda x + \mu x,\quad
\lambda(\mu x) = (\lambda\mu) x,\quad
1\,x = x .
\]
Gevolgen: $0\,x = 0_E$, $\lambda\,0_E = 0_E$, $(-1)x = -x$, en
$\lambda x = 0_E \implies \lambda = 0$ of $x = 0_E$ (vermenigvuldig
met $\lambda^{-1}$).
\end{definition}

\begin{proof}[Bewijs van de gevolgen]
Voor $0\,x = 0_E$: uit $(0 + 0)x = 0x + 0x$ en $(0+0)x = 0x$ mag
men $0x$ in de \omterm{def:b1:structures:group}{groep} $(E, +)$ wegstrepen. Voor $\lambda\,0_E$:
dezelfde kunstgreep op $\lambda(0_E + 0_E)$. Voor $(-1)x$: tel $x$
erbij op,
\[
x + (-1)x = 1\,x + (-1)x = \bigl(1 + (-1)\bigr)x = 0\,x = 0_E ,
\]
zodat $(-1)x$ de tegengestelde van $x$ is. Ten slotte, is $\lambda
x = 0_E$ met $\lambda \neq 0$, vermenigvuldig dan met
$\lambda^{-1}$ (de scalairen vormen een \omterm{def:b1:structures:field}{lichaam}) en gebruik de twee
axioma's $\lambda^{-1}(\lambda x) = (\lambda^{-1}\lambda)x = 1x =
x$ samen met $\lambda^{-1}0_E = 0_E$: $x = 0_E$. Hoe klein ook,
deze vier regels worden stilzwijgend gebruikt op elke bladzijde
die volgt --- en de laatste is precies waar \omterm{def:b1:structures:field}{lichamen} nodig zijn:
over de scalairen $\Z$ zou de ``ruimte'' $\Z/2\Z$ haar schenden
met $2\,x = 0$.
\end{proof}

\begin{example}\label{ex:b1:vspaces:examples}
$K^n$ (coördinaatsgewijze bewerkingen); de \omterm{def:b1:poly:def}{veeltermen} $K[X]$; de
functies $\mathcal{F}(A, K)$ van een willekeurige \omterm{def:b1:logic:sets}{verzameling} $A$
naar $K$ (puntsgewijze bewerkingen) --- met daarin de \omterm{def:b1:continuity:continuous}{continue}
functies, de rijen $\mathcal{F}(\N, \R)$ enzovoort; en $\C$ als
$\R$-vectorruimte. In elk geval worden de axioma's geërfd van die
van $K$.
\end{example}

\begin{definition}[Deelruimte]\label{def:b1:vspaces:subspace}
$F \subseteq E$ is een \emph{deelruimte}\index{deelruimte} wanneer
$0_E \in F$ en $F$ stabiel is onder optelling en scalaire
vermenigvuldiging --- equivalent:
\[
F \neq \emptyset
\qquad\text{en}\qquad
\forall x, y \in F,\ \forall \lambda \in K,\quad
x + \lambda y \in F .
\]
Een deelruimte is zelf een \omterm{def:b1:vspaces:def}{vectorruimte}. Elke doorsnede van
deelruimten is een deelruimte; een vereniging bijna nooit (hetzelfde
bewijs als \cref{exo:b1:structures:6}).
\end{definition}

\begin{example}\label{ex:b1:vspaces:subspaces}
In $\mathcal{F}(\R, \R)$: de \omterm{def:b1:continuity:continuous}{continue} functies, de \omterm{def:b1:derivative:def}{afleidbare}, de
\omterm{def:b1:poly:def}{veeltermen} van graad $\leq n$ (genoteerd $K_n[X]$ binnen $K[X]$), de
oplossingen van een homogene lineaire differentiaalvergelijking
(\cref{thm:b1:diffeq:homogeneous2} zei precies dat). Geen
voorbeelden: $\{f : f(0) = 1\}$ (geen nulelement), graad precies $n$
(niet stabiel onder optelling).
\end{example}

\begin{example}[Deelruimte of niet: vier oordelen, beargumenteerd]
\label{ex:b1:vspaces:subspaceornot}
In de ruimte van reële rijen:
\begin{itemize}
  \item $\{u : u \text{ begrensd}\}$ \emph{is} een \omterm{def:b1:vspaces:subspace}{deelruimte}: $0$
        is begrensd, en is $\abs{u_n} \leq M$ en $\abs{v_n} \leq
        M'$, dan is $\abs{u_n + \lambda v_n} \leq M + \abs\lambda
        M'$.
  \item $\{u : u_n \to 1\}$ is \emph{geen} \omterm{def:b1:vspaces:subspace}{deelruimte}: de nulrij
        ontbreekt (en de som van twee leden nadert tot $2$).
  \item $\{u : u \text{ monotoon}\}$ is er \emph{geen}: $u_n = n$
        en $v_n = -n + (-1)^n$ zijn monotoon, hun som $(-1)^n$
        niet; de stabiliteit onder optelling is het axioma dat
        faalt, ook al bevat de \omterm{def:b1:logic:sets}{verzameling} $0$ en alle scalaire
        veelvouden van haar leden.
  \item $\{u : u_{n+1} = u_n^2\}$ is er \emph{geen}: zij bevat $0$,
        maar $2u$ ontsnapt zodra $u$ een lid ongelijk aan nul is
        ($2u_{n+1} \neq (2u_n)^2$ in het algemeen) --- het
        kwadrateren is de niet-lineariteit.
\end{itemize}
De werkvolgorde is altijd dezelfde: toets eerst $0$ (het
goedkoopst), daarna de stabiliteit --- en om te weerleggen verslaat
één expliciet paar tegenvoorbeelden elke hoeveelheid twijfel.
\end{example}

\section{Opspansel, sommen, directe sommen}

\begin{definition}[Lineaire combinaties, opspansel]\label{def:b1:vspaces:span}
Een \emph{lineaire combinatie} van de familie $(x_1, \dots, x_p)$
vectoren van $E$ is elke $\lambda_1 x_1 + \dots + \lambda_p x_p$
($\lambda_i \in K$). De \omterm{def:b1:logic:sets}{verzameling} van al deze combinaties is het
\emph{opspansel}\index{opspansel} $\operatorname{Vect}(x_1, \dots,
x_p)$: het is een \omterm{def:b1:vspaces:subspace}{deelruimte}, en wel de kleinste die de familie
bevat.
\end{definition}

\begin{proof}[Bewijs van de twee beweringen]
Stabiliteit: een som van twee lineaire combinaties $\sum\lambda_i
x_i + \sum\mu_i x_i = \sum(\lambda_i + \mu_i)x_i$ is er weer een,
en een scalair veelvoud $\mu\sum\lambda_i x_i =
\sum(\mu\lambda_i) x_i$ ook; de nulcombinatie toont dat $0$
erbij hoort: het \omterm{def:b1:vspaces:span}{opspansel} is een \omterm{def:b1:vspaces:subspace}{deelruimte}. Minimaliteit: zij
$H$ een willekeurige \omterm{def:b1:vspaces:subspace}{deelruimte} die $x_1, \dots, x_p$ bevat.
Wegens de stabiliteit onder scalaire vermenigvuldiging ligt elke
$\lambda_i x_i$ in $H$, en wegens de stabiliteit onder optelling
hun som: elke lineaire combinatie hoort bij $H$, dat wil zeggen
$\operatorname{Vect}(x_1, \dots, x_p) \subseteq H$. Het \omterm{def:b1:vspaces:span}{opspansel}
is dus bevat in elke \omterm{def:b1:vspaces:subspace}{deelruimte} die de familie bevat: het is de
kleinste.
\end{proof}

\begin{definition}[Som, directe som]\label{def:b1:vspaces:sum}
Voor \omterm{def:b1:vspaces:subspace}{deelruimten} $F, G$ van $E$ is
\[
F + G = \{\,u + v : u \in F,\ v \in G\,\}
\]
een \omterm{def:b1:vspaces:subspace}{deelruimte} (de kleinste die $F \cup G$ bevat). De som is
\emph{direct}\index{directe som}, genoteerd $F \oplus G$, wanneer
elk element van $F + G$ zich \emph{op precies één manier} als $u +
v$ laat schrijven; equivalent (zie hieronder) wanneer $F \cap G =
\{0\}$. Geldt $E = F \oplus G$, dan heten de \omterm{def:b1:vspaces:subspace}{deelruimten}
\emph{complementair}\index{complementaire deelruimten} in $E$.
\end{definition}

\begin{example}[Een som van twee rechten]
\label{ex:b1:vspaces:sumoflines}
Zij in $\R^3$ $F = \operatorname{Vect}\bigl((1,0,1)\bigr)$ en $G
= \operatorname{Vect}\bigl((0,1,1)\bigr)$. Hun som is
\[
F + G = \{\,a(1,0,1) + b(0,1,1)\,\}
= \{(a,\ b,\ a + b)\} = \{(x, y, z) : z = x + y\},
\]
het vlak door de oorsprong dat beide rechten bevat. Het is strikt
groter dan de vereniging $F \cup G$ (het loutere kruis van de twee
rechten): de vector $(1, 1, 2) = (1,0,1) + (0,1,1)$ ligt in de
som maar op geen van beide rechten. En $F \cap G = \{0\}$ (een
gemeenschappelijke vector vereist $a(1,0,1) = b(0,1,1)$, waarvan
de eerste twee \omterm{prop:b1:vspaces:coordinates}{coördinaten} $a = b = 0$ afdwingen): de som is
direct, en $F \oplus G$ is precies dat vlak.
\end{example}

\begin{proposition}\label{prop:b1:vspaces:directsum}
$F + G$ is direct dan en slechts dan als $F \cap G = \{0\}$.
\end{proposition}

\begin{proof}
Ligt een $w \neq 0$ in $F \cap G$, dan zijn $w = w + 0 = 0 + w$ twee
ontbindingen van $w$. Omgekeerd, is $u + v = u' + v'$ met $u, u' \in
F$ en $v, v' \in G$, dan hoort $u - u' = v' - v$ bij $F \cap G =
\{0\}$: de ontbindingen zijn eenduidig.
\end{proof}

\begin{method}[Bewijzen dat $E = F \oplus G$]
\label{met:b1:vspaces:directsum}
Twee dingen na te gaan, elk met zijn standaardopening.
\begin{enumerate}
  \item \emph{Triviale doorsnede.} Neem $x \in F \cap G$, schrijf
        beide voorwaarden voor het lidmaatschap uit, en knijp $x =
        0$ eruit. (Redeneer nooit met een tekening: zie de
        valkuilen hieronder.)
  \item \emph{De som is alles.} Neem een willekeurige $x \in E$ en
        \emph{produceer} de ontbinding $x = f + g$ --- ofwel door
        $f$ te raden vanuit het doel ($f$ moet aan de definiërende
        eigenschap van $F$ voldoen, wat gewoonlijk haar formule
        dicteert), ofwel door het lineaire stelsel op te lossen dat
        $x$ tegen voortbrengers van $F$ en $G$ uitdrukt.
\end{enumerate}
Wordt de formule voor de ontbinding geraden, dan is de
eenduidigheid automatisch dankzij stap 1; is alleen het bestaan
onduidelijk, dan zit het werk in stap 2. De twee voorbeelden
hieronder laten de methode lopen: voor even en oneven functies
wordt de formule voor $f$ afgedwongen door de gewenste identiteit
in $x$ en in $-x$ te evalueren; voor \omterm{def:b1:poly:def}{veeltermen} die in een punt nul
worden, door in $a$ te evalueren.
\end{method}

\begin{example}\label{ex:b1:vspaces:evenodd}
In $\mathcal{F}(\R, \R)$ zijn de even functies $\mathcal{P}$ en de
oneven functies $\mathcal{I}$ \omterm{def:b1:vspaces:sum}{complementair}: elke $f$ schrijft zich
als
\[
f(x) = \underbrace{\frac{f(x) + f(-x)}{2}}_{\text{even}}
+ \underbrace{\frac{f(x) - f(-x)}{2}}_{\text{oneven}},
\]
en een functie die zowel even als oneven is, is nul. (Toegepast op
$\exp$ is dit het paar $(\cosh, \sinh)$ van
\cref{ch:b1:functions}.)
\end{example}

\begin{example}[{Een complementair paar in $K_n[X]$}]
\label{ex:b1:vspaces:hyperplane}
Leg $a \in K$ vast en stel $F = \{P \in K_n[X] : P(a) = 0\}$ en
$G = \operatorname{Vect}(1)$ (de constanten). Dan is $K_n[X] = F
\oplus G$. Inderdaad bestaat $F \cap G$ uit de constanten die in
$a$ nul worden, dus uit $\{0\}$; en elke $P$ ontbindt als
\[
P = \underbrace{\bigl(P - P(a)\bigr)}_{\in F}
+ \underbrace{P(a)}_{\in G} .
\]
De ontbinding is het onthouden waard: de waarde in een punt
aftrekken is de standaardmanier om op ``functies die in $a$ nul
worden'' te projecteren. Merk op dat $F$ een grote \omterm{def:b1:vspaces:subspace}{deelruimte} is en
$G$ een kleine; een \omterm{def:b1:vspaces:sum}{complementair} paar hoeft in geen enkele zin in
evenwicht te zijn.
\end{example}

\begin{example}[Een complementaire deelruimte is nooit eenduidig]
\label{ex:b1:vspaces:manysupplements}
Zij in $\R^2$ $F = \operatorname{Vect}\bigl((1,0)\bigr)$ (de
$x$-as). Zowel $G = \operatorname{Vect}\bigl((0,1)\bigr)$ als $G'
= \operatorname{Vect}\bigl((1,1)\bigr)$ is \omterm{def:b1:vspaces:sum}{complementair} met $F$:
elk snijdt $F$ alleen in $0$, en elk paar sommeert tot $\R^2$. De
ontbindingen van eenzelfde vector verschillen:
\[
(2,\ 1.5) = \underbrace{(2, 0)}_{\in F} +
\underbrace{(0, 1.5)}_{\in G}
= \underbrace{(0.5,\ 0)}_{\in F} +
\underbrace{(1.5,\ 1.5)}_{\in G'} .
\]
In feite is \emph{elke} rechte behalve $F$ zelf een \omterm{def:b1:vspaces:sum}{complementaire}
van $F$ in $\R^2$: \omterm{def:b1:vspaces:sum}{complementairen} zijn er in overvloed, en
spreken over ``de'' \omterm{def:b1:vspaces:sum}{complementaire} is betekenisloos zolang geen
extra structuur (een inproduct, \cref{ch:b1:euclid}) er één
uitkiest.
\end{example}

\section{Vrije families, voortbrengende families, bases}

\begin{definition}\label{def:b1:vspaces:free}
Een familie $(x_1, \dots, x_p)$ vectoren van $E$ heet:
\begin{itemize}
  \item \emph{voortbrengend} (voor $E$) wanneer
        $\operatorname{Vect}(x_1,\dots,x_p) = E$;
  \item \emph{vrij}\index{vrije familie} (haar vectoren
        \emph{lineair onafhankelijk}) wanneer
        \[
        \lambda_1 x_1 + \dots + \lambda_p x_p = 0
        \implies \lambda_1 = \dots = \lambda_p = 0 ;
        \]
        in het andere geval \emph{afhankelijk};
  \item een \emph{basis}\index{basis} wanneer zij vrij en
        voortbrengend is.
\end{itemize}
\end{definition}

\begin{proposition}[Coördinaten]\label{prop:b1:vspaces:coordinates}
$(e_1, \dots, e_n)$ is een \omterm{def:b1:vspaces:free}{basis} van $E$ dan en slechts dan als elke
$x \in E$ zich \emph{op precies één manier} als combinatie $x =
\lambda_1 e_1 + \dots + \lambda_n e_n$ laat schrijven; de scalairen
$\lambda_i$ zijn de \emph{coördinaten}\index{coördinaten} van $x$ in
de \omterm{def:b1:vspaces:free}{basis}.
\end{proposition}

\begin{proof}
Voortbrengend $=$ het bestaan van de ontbinding. Eenduidigheid $=$
vrijheid: twee ontbindingen van dezelfde $x$ verschillen met een
combinatie die gelijk is aan $0$; de vrijheid dwingt al haar
coëfficiënten --- de verschillen van de \omterm{prop:b1:vspaces:coordinates}{coördinaten} --- tot nul.
Omgekeerd geeft een niet-triviale nulcombinatie de twee
ontbindingen $0 = \sum \lambda_i e_i = \sum 0\,e_i$.
\end{proof}

\begin{example}\label{ex:b1:vspaces:bases}
De \emph{canonieke \omterm{def:b1:vspaces:free}{basis}} van $K^n$: $e_i = (0, \dots, 1, \dots,
0)$ (een $1$ op plaats $i$). De monomen $(1, X, X^2, \dots, X^n)$:
een \omterm{def:b1:vspaces:free}{basis} van $K_n[X]$ (vrijheid: een nulcombinatie is de
nulveelterm, dus zijn alle coëfficiënten nul,
\cref{def:b1:poly:def}). In $\C$ over $\R$: de \omterm{def:b1:vspaces:free}{basis} $(1, \iu)$.
\end{example}

\begin{remark}[Coördinaten zijn teamwerk]
\label{rem:b1:vspaces:teameffort}
De eerste coördinaat van $x$ in een \omterm{def:b1:vspaces:free}{basis} $(e_1, \dots, e_n)$ hangt
af van \emph{alle} basisvectoren, niet alleen van $e_1$. In
$\R^2$: de vector $(3, 1)$ heeft eerste coördinaat $3$ in de
canonieke \omterm{def:b1:vspaces:free}{basis}, maar eerste coördinaat $2$ in de \omterm{def:b1:vspaces:free}{basis}
$\bigl((1,0), (1,1)\bigr)$ --- los $(3,1) = a(1,0) + b(1,1)$ op:
$b = 1$, $a = 2$. Eén basisvector veranderen schudt \emph{elke}
coördinaat door elkaar; \cref{ch:b1:matrices} zal dat door elkaar
schudden in de basisovergangsmatrix verpakken.
\end{remark}

\begin{example}[Een kandidaat-basis toetsen, van begin tot eind]
\label{ex:b1:vspaces:testbasis}
Is $\mathcal{F} = (1 + X,\ 1 + X^2,\ X + X^2)$ een \omterm{def:b1:vspaces:free}{basis} van
$\R_2[X]$? Schrijf $u_1, u_2, u_3$ voor de drie \omterm{def:b1:poly:def}{veeltermen}.
\emph{Vrijheid}: een nulcombinatie $a\,u_1 + b\,u_2 + c\,u_3 = 0$
geeft, coëfficiënt na coëfficiënt,
\[
a + b = 0, \qquad a + c = 0, \qquad b + c = 0 ;
\]
door de eerste twee af te trekken volgt $b = c$, en dan geeft de
derde $2b = 0$: $a = b = c = 0$, dus \omterm{def:b1:vspaces:free}{vrij}. \emph{Voortbrengend}:
in plaats van drie stelsels op te lossen, merk je de symmetrische
combinatie op
\[
u_1 + u_2 - u_3 = (1 + X) + (1 + X^2) - (X + X^2) = 2 ,
\]
dus $1 = \frac12(u_1 + u_2 - u_3)$; vervolgens
\[
X = u_1 - 1 = \tfrac12\bigl(u_1 - u_2 + u_3\bigr),
\qquad
X^2 = u_2 - 1 = \tfrac12\bigl(-u_1 + u_2 + u_3\bigr).
\]
De monomen liggen in het \omterm{def:b1:vspaces:span}{opspansel}, dus alles ligt erin:
$\mathcal{F}$ is een \omterm{def:b1:vspaces:free}{basis}. Als toegift geeft het samenvoegen van
de drie formules de \omterm{prop:b1:vspaces:coordinates}{coördinaten} van elke $P = \alpha + \beta X +
\gamma X^2$:
\[
P = \frac{\alpha + \beta - \gamma}{2}\,u_1
+ \frac{\alpha - \beta + \gamma}{2}\,u_2
+ \frac{-\alpha + \beta + \gamma}{2}\,u_3 .
\]
(Controle met $P = X$: \omterm{prop:b1:vspaces:coordinates}{coördinaten} $\bigl(\frac12, -\frac12,
\frac12\bigr)$, zoals hierboven gevonden.) Twee lessen: symmetrie
in de familie verbergt gewoonlijk een kortere weg; en zodra de
dimensie beschikbaar is (\cref{ch:b1:findim}) komt de hele
voortbrengende helft van dit werk gratis --- drie \omterm{def:b1:vspaces:free}{vrije} vectoren
van een $3$-dimensionale ruimte vormen altijd een \omterm{def:b1:vspaces:free}{basis}.
\end{example}

\begin{proposition}[Handige vrijheidscriteria]\label{prop:b1:vspaces:freecriteria}
\begin{enumerate}
  \item Een familie \emph{\omterm{def:b1:poly:def}{veeltermen} ongelijk aan nul met
        paarsgewijs verschillende graden} is \omterm{def:b1:vspaces:free}{vrij}.
  \item Een vector aan een \omterm{def:b1:vspaces:free}{vrije familie} toevoegen houdt haar \omterm{def:b1:vspaces:free}{vrij}
        dan en slechts dan als die vector buiten het \omterm{def:b1:vspaces:span}{opspansel} van
        de familie ligt.
  \item Elke deelfamilie van een \omterm{def:b1:vspaces:free}{vrije familie} is \omterm{def:b1:vspaces:free}{vrij}; elke
        familie die een voortbrengende familie bevat, is
        voortbrengend.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Kijk in een nulcombinatie naar de hoogste aanwezige graad: zijn
coëfficiënt moet nul zijn (niets heft die graad op), en zo cascadeer
je naar beneden.

(2) Is $x \in \operatorname{Vect}(x_1, \dots, x_p)$, dan is de
betrekking $x - \sum\lambda_i x_i = 0$ niet-triviaal. Omgekeerd moet
een niet-triviale nulcombinatie van $(x_1, \dots, x_p, x)$ de $x$
met een coëfficiënt ongelijk aan nul bevatten (anders is zij in
tegenspraak met de vrijheid van de kleine familie), en het oplossen
naar $x$ plaatst die in het \omterm{def:b1:vspaces:span}{opspansel}.

(3) Deelfamilie: een nulcombinatie van de deelfamilie is er een van
de hele familie met de ontbrekende coëfficiënten op $0$ gezet; de
vrijheid van de grote familie doodt ze alle. Overfamilie: elke
vector van $E$ is al een combinatie van het voortbrengende deel;
geef de extra vectoren de coëfficiënt $0$.
\end{proof}

\begin{example}[Het trapprincipe]
\label{ex:b1:vspaces:staircase}
Zij $P_0, P_1, \dots, P_n \in K_n[X]$ met $\deg P_k = k$ voor elke
$k$ (een ``trap'' van graden). Dan is $(P_0, \dots, P_n)$ een \omterm{def:b1:vspaces:free}{basis}
van $K_n[X]$. De vrijheid is
\cref{prop:b1:vspaces:freecriteria}~(1). Voor de voortbrengende
eigenschap redeneren we met eindige afdaling op de graad: zij $Q
\in K_n[X]$, $Q \neq 0$, van graad $d$ met kopcoëfficiënt $a$, en
zij $b \neq 0$ de kopcoëfficiënt van $P_d$. Dan heeft $Q - \frac ab
P_d$ graad $< d$ (de kopterm valt weg); vervang $Q$ door dat
verschil en herhaal: na hoogstens $n + 1$ stappen bereikt men de
nulveelterm, en het terugdraaien van de aftrekkingen drukt $Q$ uit
als combinatie van de $P_k$. Twee trappen kwamen we al tegen: de
verschoven machten $\bigl((X-a)^k\bigr)_{0 \leq k \leq n}$
(\cref{exo:b1:vspaces:4}), en de producten van Newton $\bigl((X -
x_0)(X - x_1)\cdots(X - x_{k-1})\bigr)_{0 \leq k \leq n}$, die in
de weekendopgave aan het werk worden gezet.
\end{example}

\begin{example}[Vrijheid in functieruimten]\label{ex:b1:vspaces:functionfree}
In $\mathcal{F}(\R,\R)$ is de familie $(\eu^{a_1 x}, \dots,
\eu^{a_p x})$ met $a_1 < \dots < a_p$ \omterm{def:b1:vspaces:free}{vrij}: deel een nulcombinatie
door $\eu^{a_p x}$ en laat $x \to +\infty$; de laatste coëfficiënt
sterft, en zo cascadeer je naar beneden (\cref{exo:b1:vspaces:8}
werkt dit en varianten uit). Vrijheid van functies bewijst men door
te \emph{evalueren}: in goedgekozen punten, in het oneindige, of na
afleiden.
\end{example}

\begin{example}[Een verborgen betrekking krimpt een opspansel]
\label{ex:b1:vspaces:hiddenrelation}
Wat is in $\mathcal{F}(\R, \R)$ het \omterm{def:b1:vspaces:span}{opspansel}
$\operatorname{Vect}(1,\ \cos^2,\ \sin^2)$? De identiteit $\cos^2
+ \sin^2 = 1$ is een niet-triviale nulcombinatie
\[
1\cdot\mathbf{1} + (-1)\cos^2 + (-1)\sin^2 = 0 :
\]
de familie is afhankelijk, en het \omterm{def:b1:vspaces:span}{opspansel} wordt al voortgebracht
door $(1, \cos^2)$ alleen ($\sin^2 = 1 - \cos^2$). Die kleinere
familie is \omterm{def:b1:vspaces:free}{vrij}: $a + b\cos^2 x = 0$ voor alle $x$ geeft, in $x =
0$ en $x = \frac\pi2$: $a + b = 0$ en $a = 0$. Het \omterm{def:b1:vspaces:span}{opspansel} is
dus een \emph{vlak} binnen de functieruimte --- en het bevat ook
$\cos 2x = 2\cos^2 x - 1$: families goniometrische functies die
lineair lijken, storten routinematig in onder identiteiten, en
daarom moet vrijheid worden \emph{bewezen}, nooit uit de lengte
van de lijst worden aangenomen.
\end{example}

\begin{remark}[Veelgemaakte fouten]
\label{rem:b1:vspaces:pitfalls}
Vier klassieke vallen.
\emph{Paarsgewijs is niet genoeg}: in $\R^2$ zijn de vectoren
$(1,0)$, $(0,1)$, $(1,1)$ paarsgewijs niet evenredig en toch
afhankelijk --- vrijheid is een eigenschap van de \emph{hele}
familie, getoetst met één globale combinatie, nooit twee aan twee.
\emph{De nulvector vergiftigt alles}: elke familie die $0$ bevat
is afhankelijk ($1\cdot 0 = 0$ is een niet-triviale betrekking),
hoe onschuldig de andere vectoren ook zijn.
\emph{Vereniging is geen som}: $F \cup G$ is vrijwel nooit een
\omterm{def:b1:vspaces:subspace}{deelruimte} (\cref{def:b1:vspaces:subspace}); de kleinste \omterm{def:b1:vspaces:subspace}{deelruimte}
die beide bevat is $F + G$, gewoonlijk veel groter dan de
vereniging --- in $\R^2$ hebben twee verschillende rechten een
kruis als vereniging en het hele vlak als som.
\emph{Direct vereist triviale doorsnede, geen disjunctheid}: twee
\omterm{def:b1:vspaces:subspace}{deelruimten} zijn nooit disjunct (beide bevatten $0$); de juiste
voorwaarde is $F \cap G = \{0\}$, en die moet worden
\emph{bewezen}, niet van een tekening afgelezen --- vergelijk
\cref{ex:b1:vspaces:manysupplements}, waar vele verschillende $G$
voldoen.
\emph{Vrijheid hangt af van de scalairen}: het paar $(1, \iu)$ is
\omterm{def:b1:vspaces:free}{vrij} in $\C$ opgevat als $\R$-vectorruimte, maar afhankelijk in
$\C$ opgevat als $\C$-vectorruimte ($\iu\cdot 1 + (-1)\cdot\iu =
0$). Weet altijd welk \omterm{def:b1:structures:field}{lichaam} werkt voordat je een familie \omterm{def:b1:vspaces:free}{vrij}
noemt --- de weekendopgave van \cref{ch:b1:findim} maakt van
precies deze gevoeligheid irrationaliteitsbewijzen.
\end{remark}

\begin{remark}[Waar deze taal heen gaat]
\label{rem:b1:vspaces:whereused}
Alles na dit hoofdstuk spreekt de taal die hier is opgezet.
\cref{ch:b1:findim} telt basisvectoren en zet ``\omterm{def:b1:vspaces:free}{vrij}'' en
``voortbrengend'' om in ongelijkheden voor één geheel getal, de
dimensie. \cref{ch:b1:linmaps} bestudeert de \omterm{def:b1:logic:map}{afbeeldingen} die met
de twee bewerkingen verenigbaar zijn; directe sommen worden daar
projectoren. \cref{ch:b1:matrices} codeert vectoren door hun
\omterm{prop:b1:vspaces:coordinates}{coördinaten} in een \omterm{def:b1:vspaces:free}{basis} --- \cref{prop:b1:vspaces:coordinates} is
de vergunning voor die codering --- en \cref{ch:b1:euclid} legt
lengten en hoeken bovenop de lineaire structuur. In het volume van
bachelorjaar 2 draaien dezelfde axioma's, woordelijk, over
willekeurige \omterm{def:b1:structures:field}{lichamen} en in oneindige dimensie; nergens in dit
hoofdstuk werd eindigheid gebruikt.
\end{remark}

\begin{remark}[Drie draden om door boek 3 heen te volgen]
\label{rem:b1:vspaces:threads}
Volg drie bepaalde ideeën uit dit hoofdstuk terwijl zij groeien.
\emph{Het trapprincipe} (\cref{ex:b1:vspaces:staircase}) keert

\end{remark}

\section{Opgave: interpolatie, drie bases voor één ruimte}

\begin{problem}\label{pb:b1:vspaces:1}
Leg $n + 1$ \emph{verschillende} punten $x_0, x_1, \dots, x_n$ van
$\R$ vast. Deze opgave herbekijkt de \omterm{thm:b1:poly:lagrange}{Lagrange-interpolatie}
(\cref{thm:b1:poly:lagrange}) met de ogen van dit hoofdstuk: de
ruimte $\R_n[X]$ draagt drie natuurlijke \omterm{def:b1:vspaces:free}{bases} --- die van
Lagrange, die van Newton, en (voor gelijkmatig verdeelde punten) de
binomiale \omterm{def:b1:vspaces:free}{basis} --- en elke \omterm{def:b1:vspaces:free}{basis} maakt één vraag gemakkelijk. De
weg eindigt bij een echte rekenkundige stelling: de
karakterisering van Pólya van de \omterm{def:b1:poly:def}{veeltermen} die $\Z$ in $\Z$
afbeelden.\index{Lagrange-interpolatie}
\index{gedeelde differenties}\index{geheelwaardige veelterm}

\medskip
\noindent\textbf{Deel I --- De \omterm{def:b1:vspaces:free}{Lagrange-basis}.} Stel voor $0 \leq
i \leq n$
\[
L_i \;=\; \prod_{j \neq i} \frac{X - x_j}{x_i - x_j} \;\in\;
\R_n[X].
\]
\begin{enumerate}
  \item Ga na dat $\deg L_i = n$ en dat $L_i(x_j) = 1$ als $j = i$,
        en $0$ als $j \neq i$.
  \item Bewijs dat de familie $(L_0, \dots, L_n)$ \omterm{def:b1:vspaces:free}{vrij} is.
  \item Bewijs dat voor elke $P \in \R_n[X]$
        \[
        P \;=\; \sum_{i=0}^{n} P(x_i)\, L_i ,
        \]
        en besluit dat $(L_0, \dots, L_n)$ een \omterm{def:b1:vspaces:free}{basis} is van
        $\R_n[X]$. \emph{(Beschouw het verschil van de twee leden
        en tel zijn wortels, \cref{cor:b1:poly:nroots}.)}
  \item Leid de interpolatiestelling af: voor willekeurige waarden
        $y_0, \dots, y_n \in \R$ bestaat er precies \emph{één} $P
        \in \R_n[X]$ met $P(x_i) = y_i$ voor alle $i$. Wat zijn de
        \omterm{prop:b1:vspaces:coordinates}{coördinaten} van een \omterm{def:b1:poly:def}{veelterm} $P$ in de \omterm{def:b1:vspaces:free}{Lagrange-basis}?
  \item Bewijs de identiteiten
        \[
        \sum_{i=0}^{n} L_i = 1
        \qquad\text{en, voor } 0 \leq k \leq n,\qquad
        \sum_{i=0}^{n} x_i^{k}\, L_i = X^{k} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De \omterm{def:b1:vspaces:free}{Newton-basis} en de \omterm{pb:b1:vspaces:1}{gedeelde
differenties}.} Stel $N_0 = 1$ en $N_k = (X - x_0)(X - x_1) \cdots
(X - x_{k-1})$ voor $1 \leq k \leq n$. Definieer voor een functie
$f$ die in de knooppunten is gedefinieerd de \emph{\omterm{pb:b1:vspaces:1}{gedeelde
differenties}} door $f[x_i] = f(x_i)$ en
\[
f[x_i, \dots, x_{i+k}] \;=\;
\frac{f[x_{i+1}, \dots, x_{i+k}] - f[x_i, \dots, x_{i+k-1}]}
{x_{i+k} - x_i} .
\]
\begin{enumerate}[resume]
  \item Bewijs dat $(N_0, N_1, \dots, N_n)$ een \omterm{def:b1:vspaces:free}{basis} is van
        $\R_n[X]$.
  \item Bereken $f[x_0, x_1]$ en $f[x_0, x_1, x_2]$ in termen van
        de waarden van $f$, en bereken daarna alle \omterm{pb:b1:vspaces:1}{gedeelde
        differenties} van $f(x) = x^2$ in drie willekeurige
        knooppunten.
  \item (Lemma van Aitken) Zij $R$ de interpolant van $f$ in $x_0,
        \dots, x_{n-1}$ en $Q$ die van $f$ in $x_1, \dots, x_n$,
        beide van graad $\leq n - 1$. Bewijs dat
        \[
        S \;=\; \frac{(X - x_0)\,Q - (X - x_n)\,R}{x_n - x_0}
        \]
        $f$ interpoleert in $x_0, x_1, \dots, x_n$.
  \item Leid met inductie op het aantal knooppunten af dat de
        coëfficiënt van $X^{k}$ in de interpolant van $f$ in $x_0,
        \dots, x_k$ precies $f[x_0, \dots, x_k]$ is.
  \item Bewijs de \emph{interpolatieformule van Newton}: de
        interpolant van $f$ in $x_0, \dots, x_n$ is
        \[
        P \;=\; \sum_{k=0}^{n} f[x_0, \dots, x_k]\, N_k ,
        \]
        en leid de \omterm{def:b1:topology:closed}{gesloten} formule
        \[
        f[x_0, \dots, x_k] \;=\; \sum_{i=0}^{k}
        \frac{f(x_i)}{\prod_{j \neq i,\, j \leq k} (x_i - x_j)}
        \]
        af, waaruit blijkt dat $f[x_0, \dots, x_k]$ niet afhangt
        van de volgorde van de knooppunten.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Gelijkmatig verdeelde knooppunten:
de differentieoperator.} Vanaf nu zijn de knooppunten $0, 1, 2,
\dots$ en stellen we, voor een \omterm{def:b1:poly:def}{veelterm} $P$,
\[
\Delta P(X) = P(X + 1) - P(X),
\qquad
B_k = \frac{X(X-1)\cdots(X-k+1)}{k!} \quad (B_0 = 1).
\]
\begin{enumerate}[resume]
  \item Toon aan dat als $\deg P = m \geq 1$ met kopcoëfficiënt
        $a$, dan $\deg \Delta P = m - 1$ met kopcoëfficiënt
        $m\,a$, en dat $\Delta$ constanten doodt.
  \item Toon aan dat $(B_0, B_1, \dots, B_n)$ een \omterm{def:b1:vspaces:free}{basis} is van
        $\R_n[X]$ en dat $\Delta B_k = B_{k-1}$ voor $k \geq 1$.
  \item (Voorwaartse-differentieformule van Newton) Bewijs dat elke
        $P \in \R_n[X]$ voldoet aan
        \[
        P \;=\; \sum_{k=0}^{n} \bigl(\Delta^{k} P\bigr)(0)\, B_k .
        \]
  \item Bewijs dat voor elke $k \geq 0$
        \[
        \bigl(\Delta^{k} P\bigr)(0) \;=\;
        \sum_{j=0}^{k} (-1)^{k-j} \binom{k}{j} P(j) .
        \]
  \item Toon aan dat als $\deg P = n$ met kopcoëfficiënt $a_n$, dan
        $\Delta^{n} P$ de constante $n!\,a_n$ is en $\Delta^{n+1} P
        = 0$.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- \omterm{pb:b1:vspaces:1}{Geheelwaardige veeltermen}.} Een
\omterm{def:b1:poly:def}{veelterm} $P \in \R[X]$ heet \emph{geheelwaardig} wanneer $P(m) \in
\Z$ voor elke $m \in \Z$.
\begin{enumerate}[resume]
  \item Bewijs dat elke $B_k$ geheelwaardig is. \emph{(Behandel $m
        \geq k$, $0 \leq m < k$ en $m < 0$ afzonderlijk; toon voor
        $m = -q < 0$ aan dat $B_k(-q) = (-1)^k \binom{q + k -
        1}{k}$.)}
  \item Bewijs de \emph{karakterisering van Pólya}: $P \in \R_n[X]$
        is geheelwaardig dan en slechts dan als zijn \omterm{prop:b1:vspaces:coordinates}{coördinaten} in
        de \omterm{def:b1:vspaces:free}{basis} $(B_0, \dots, B_n)$ gehele getallen zijn.
  \item Leid af: neemt $P \in \R_n[X]$ gehele waarden aan in $n +
        1$ \emph{opeenvolgende} gehele getallen $a, a+1, \dots,
        a+n$, dan is $P$ geheelwaardig. \emph{(Verschuif: pas het
        onderzoek toe op $Q(X) = P(X + a)$.)}
  \item Leid uit vraag 16 af dat een product van $k$ opeenvolgende
        gehele getallen altijd \omterm{def:b1:arith:divides}{deelbaar} is door $k!$.
  \item Zij $P = \dfrac{X(X+1)(2X+1)}{6}$. Bereken zijn
        Newton-tabel in $0, 1, 2, 3$, schrijf $P$ in de \omterm{def:b1:vspaces:free}{basis}
        $(B_k)$, en besluit dat $P$ geheelwaardig is hoewel geen
        van zijn monomiale coëfficiënten geheel is. Ga na dat
        $\Delta P = (X+1)^2$ en leid af dat $P(m) = 1^2 + 2^2 +
        \dots + m^2$ voor $m \in \N$.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Dividenden.}
\begin{enumerate}[resume]
  \item Zij $P \in \R_n[X]$ de interpolant van de waarden $2^i$ in
        $i = 0, 1, \dots, n$. Toon aan dat $P = B_0 + B_1 + \dots +
        B_n$ en dat $P(n + 1) = 2^{n+1} - 1$: het
        ``verdubbelingspatroon'' breekt altijd in het
        eerstvolgende punt.
  \item (Discrete primitieve) Bewijs dat voor alle gehele getallen
        $m \geq 1$ en $k \geq 0$
        \[
        \sum_{j=0}^{m-1} B_k(j) \;=\; B_{k+1}(m),
        \]
        dat wil zeggen de hockeystick-identiteit $\sum_{j=k}^{m-1}
        \binom{j}{k} = \binom{m}{k+1}$.
  \item Ontwikkel $X^2$ en $X^3$ in de \omterm{def:b1:vspaces:free}{basis} $(B_k)$ en leid
        \omterm{def:b1:topology:closed}{gesloten} formules af voor $\sum_{j=0}^{m-1} j^2$ en
        $\sum_{j=0}^{m-1} j^3$; vind de identiteit van Nicomachus
        $1^3 + \dots + m^3 = (1 + \dots + m)^2$ terug.
  \item Neem $n = 2$ en knooppunten $0, 1, 2$. Schrijf de
        \omterm{prop:b1:vspaces:coordinates}{coördinaten} van $X^2$ in de drie \omterm{def:b1:vspaces:free}{bases} van deze opgave: de
        monomiale \omterm{def:b1:vspaces:free}{basis}, de \omterm{def:b1:vspaces:free}{Lagrange-basis} en de \omterm{def:b1:vspaces:free}{Newton-basis}.
        Toets de drie antwoorden aan de vragen 4 en 9.
  \item Synthese. In vier zinnen: welk begrip uit de theorie van de
        \omterm{def:b1:vspaces:def}{vectorruimten} vraag 4 automatisch maakt; waarom de
        \omterm{def:b1:vspaces:free}{Newton-basis} \omterm{prop:b1:vspaces:coordinates}{coördinaten} \emph{recursief} berekent terwijl
        de \omterm{def:b1:vspaces:free}{Lagrange-basis} ze \emph{onmiddellijk} aflezen laat; welk
        vrijheidscriterium beide \omterm{def:b1:vspaces:free}{bases} delen; en in welke precieze
        zin de stelling van Pólya zegt dat de geheeltalligheid van
        een \omterm{def:b1:poly:def}{veelterm} een eigenschap is van zijn \omterm{prop:b1:vspaces:coordinates}{coördinaten} \emph{in
        de juiste \omterm{def:b1:vspaces:free}{basis}}.
\end{enumerate}
\end{problem}
