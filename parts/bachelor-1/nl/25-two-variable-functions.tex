\chapter{Functies van twee veranderlijken}\label{ch:b1:multivar}

Het jaar eindigt met een eerste wandeling in hogere dimensie:
functies $f(x, y)$ van twee reële veranderlijken. Alles
veralgemeent --- limieten, \omterm{def:b1:continuity:continuous}{continuïteit}, \omterm{def:b1:derivative:def}{afgeleiden}, extrema ---
maar elk begrip krijgt een draai: limieten kunnen tegelijk langs
elke richting worden benaderd, \omterm{def:b1:derivative:def}{afgeleiden} splitsen zich in
partiële, en de \omterm{def:b1:multivar:partial}{gradiënt} wijst de weg bergop. De volledige theorie
(differentialen, algemene $\R^n$, deelvariëteiten) hoort bij het
tweede jaar; hier zetten wij het vocabulaire en de eerste eerlijke
stellingen neer.

\section{Het vlak $\R^2$; continuïteit}

\begin{definition}\label{def:b1:multivar:topology}
Op $\R^2$ gebruiken wij de euclidische norm $\norm{(x,y)} =
\sqrt{x^2 + y^2}$ (\cref{ch:b1:euclid}). \omterm{def:b1:topology:open}{Open} bollen, \omterm{def:b1:topology:open}{omgevingen} en
\emph{\omterm{def:b1:topology:open}{open} deelverzamelingen} van $\R^2$ worden precies zo
gedefinieerd als in \cref{ch:b1:topology}, met bollen in plaats van
\omterm{prop:b1:reals:intervals}{intervallen}. Een functie $f \colon U \to \R$ ($U \subseteq \R^2$
\omterm{def:b1:topology:open}{open}) heet \emph{\omterm{def:b1:continuity:continuous}{continu} in} $a \in U$ wanneer
\[
\forall\varepsilon > 0,\ \exists\delta > 0, \quad
\norm{X - a} \leq \delta \implies \abs{f(X) - f(a)} \leq
\varepsilon,
\]
met dezelfde karakterisering met rijen als in één veranderlijke.
Sommen, producten, quotiënten en samenstellingen met \omterm{def:b1:continuity:continuous}{continue}
functies van één veranderlijke bewaren de \omterm{def:b1:continuity:continuous}{continuïteit}; de
coördinaatafbeeldingen zijn \omterm{def:b1:continuity:continuous}{continu}, en dus ook de \omterm{def:b1:poly:def}{veeltermen} in
$(x,y)$.
\end{definition}

\begin{example}[De poolgrens, de nette manier om een limiet te
bewijzen]
\label{ex:b1:multivar:polarbound}
Toon aan dat $f(x, y) = \dfrac{x^2y^2}{x^2 + y^2}$ (met $f(0,0) =
0$) \omterm{def:b1:continuity:continuous}{continu} is in de oorsprong. In poolcoördinaten $x =
\rho\cos\theta$, $y = \rho\sin\theta$:
\[
\abs{f} = \frac{\rho^4\cos^2\theta\sin^2\theta}{\rho^2}
= \rho^2\,(\cos\theta\sin\theta)^2 \leq \frac{\rho^2}{4}
\xrightarrow[\rho \to 0]{} 0 ,
\]
een grens \emph{onafhankelijk van $\theta$}: wat de richting van
de benadering ook is, de waarden worden naar $0$ geknepen. Die
uniformiteit in $\theta$ is het hele punt --- een grens als
$\abs g = \abs{\cos\theta\sin\theta}$ (zonder $\rho$) bewijst
niets, en die $g$ is inderdaad de discontinue radiale val van het
volgende voorbeeld.
\end{example}

\begin{example}[De radiale val]\label{ex:b1:multivar:trap}
Zij $f(x, y) = \dfrac{xy}{x^2 + y^2}$ voor $(x,y) \neq (0,0)$ en
$f(0, 0) = 0$. Langs elke as is $f = 0 \to 0$; maar langs de
diagonaal $y = x$ is $f(x, x) = \frac12 \not\to 0$. \emph{Geen
limiet in de oorsprong}: langs elke rechte benaderen, en zelfs
langs elke rechte dezelfde limiet vinden, is niet genoeg (hier
verschillen de limieten langs de rechten al; ergere voorbeelden
stemmen langs alle rechten overeen en falen toch langs een
parabool, \cref{exo:b1:multivar:3}). \omterm{def:b1:continuity:continuous}{Continuïteit} in elke
veranderlijke afzonderlijk impliceert geen \omterm{def:b1:continuity:continuous}{continuïteit}.
\end{example}

\section{Partiële afgeleiden}

\begin{definition}\label{def:b1:multivar:partial}
De \emph{partiële afgeleiden}\index{partiële afgeleide} van $f$ in
$(a, b)$ zijn de \omterm{def:b1:derivative:def}{afgeleiden} in één veranderlijke langs de assen:
\[
\frac{\partial f}{\partial x}(a,b)
= \lim_{h \to 0} \frac{f(a + h, b) - f(a,b)}{h},
\qquad
\frac{\partial f}{\partial y}(a,b)
= \lim_{k \to 0} \frac{f(a, b + k) - f(a,b)}{k}.
\]
$f$ is van klasse $C^1$ op $U$ wanneer beide bestaan en \omterm{def:b1:continuity:continuous}{continu} zijn
op $U$. De \emph{gradiënt}\index{gradiënt} is $\nabla f(a,b) =
\bigl(\frac{\partial f}{\partial x},\, \frac{\partial f}{\partial
y}\bigr)(a,b)$.
\end{definition}

\begin{theorem}[$C^1$ impliceert een raakvlak]\label{thm:b1:multivar:c1}
Zij $f$ van klasse $C^1$ op $U$ en $(a,b) \in U$. Dan geldt, als
$(h, k) \to (0,0)$:
\[
f(a + h, b + k) = f(a, b)
+ h\,\frac{\partial f}{\partial x}(a,b)
+ k\,\frac{\partial f}{\partial y}(a,b)
+ o\bigl(\norm{(h,k)}\bigr) .
\]
In het bijzonder is $f$ \omterm{def:b1:continuity:continuous}{continu}, en heeft de grafiek $z = f(x,y)$ in
elk punt het raakvlak dat uit de formule wordt afgelezen.
\end{theorem}

\begin{example}[Lineaire benadering in actie]
\label{ex:b1:multivar:linapprox}
Schat $f(1.02,\ 0.99)$ voor $f(x, y) = x^3y^2$. In $(1, 1)$: $f =
1$, $\frac{\partial f}{\partial x} = 3x^2y^2 = 3$, $\frac{\partial
f}{\partial y} = 2x^3y = 2$, dus geeft
\cref{thm:b1:multivar:c1}
\[
f(1.02,\ 0.99) \approx 1 + 3\,(0.02) + 2\,(-0.01) = 1.04 ,
\]
tegenover de werkelijke waarde $1.02^3 \times 0.99^2 =
1.04006\dots$ --- de fout is van tweede orde in de aangroeiingen,
zoals de $o(\norm{(h,k)})$ belooft. Het raakvlak aan de grafiek in
$(1, 1, 1)$ is $z = 1 + 3(x - 1) + 2(y - 1)$, de vergelijking die
in de schatting besloten ligt.
\end{example}

\begin{proof}
Verplaats één coördinaat tegelijk:
\[
f(a+h, b+k) - f(a,b)
= \bigl[f(a+h, b+k) - f(a, b+k)\bigr] + \bigl[f(a, b+k) -
f(a,b)\bigr].
\]
Volgens de middelwaardestelling in één veranderlijke
(\cref{thm:b1:derivative:mvt}) is de eerste haak gelijk aan $h\,
\frac{\partial f}{\partial x}(a + \theta h,\, b + k)$ voor zekere
$\theta \in \intoo{0}{1}$, en de tweede aan $k\,\frac{\partial
f}{\partial y}(a,\, b + \theta' k)$. De \omterm{def:b1:continuity:continuous}{continuïteit} van de
\omterm{def:b1:multivar:partial}{partiële afgeleiden} in $(a,b)$ laat toe elk te schrijven als (de
waarde in $(a,b)$) $+$ (een fout $\to 0$); de totale fout is
$h\,\varepsilon_1 + k\,\varepsilon_2 = o(\norm{(h,k)})$, omdat
$\abs h, \abs k \leq \norm{(h,k)}$.
\end{proof}

\begin{theorem}[Kettingregel]\label{thm:b1:multivar:chain}
Zij $f$ van klasse $C^1$ op $U$ en $t \mapsto (x(t), y(t))$ van
klasse $C^1$ van een \omterm{prop:b1:reals:intervals}{interval} naar $U$. Dan is $g(t) =
f\bigl(x(t), y(t)\bigr)$ van klasse $C^1$, met\index{kettingregel!twee
variabelen}
\[
g'(t) = x'(t)\,\frac{\partial f}{\partial x}\bigl(x(t),y(t)\bigr)
+ y'(t)\,\frac{\partial f}{\partial y}\bigl(x(t),y(t)\bigr)
= \bigl\langle \nabla f,\ (x', y')\bigr\rangle .
\]
\end{theorem}

\begin{proof}
Pas \cref{thm:b1:multivar:c1} toe in $(x(t), y(t))$ met $(h, k) =
(x(t+s) - x(t),\, y(t+s) - y(t))$: als $s \to 0$ geeft de
\omterm{def:b1:derivative:def}{afleidbaarheid} in één veranderlijke $h = s\,x'(t) + o(s)$ en $k =
s\,y'(t) + o(s)$, dus $\norm{(h, k)} = O(s)$ en
\[
g(t+s) - g(t)
= h\,\frac{\partial f}{\partial x} + k\,\frac{\partial
f}{\partial y} + o\bigl(\norm{(h,k)}\bigr)
= s\,\Bigl(x'\,\frac{\partial f}{\partial x} +
y'\,\frac{\partial f}{\partial y}\Bigr) + o(s) ,
\]
waarbij de laatste fout zowel de $o(s)$'s van $h$ en $k$ opneemt
(vermenigvuldigd met de vaste waarden van de \omterm{def:b1:multivar:partial}{partiële afgeleiden})
als de $o(O(s))$ van de raakvlakschatting. Deel door $s$ en laat $s
\to 0$. De \omterm{def:b1:continuity:continuous}{continuïteit} van $g'$ volgt uit die van alle
ingrediënten.
\end{proof}

\begin{example}[De kettingregel, op twee manieren nagegaan]
\label{ex:b1:multivar:chainrun}
Zij $f(x, y) = x^2 y$ en $g(t) = f(t, t^2)$. Rechtstreeks: $g(t) =
t^2\cdot t^2 = t^4$, dus $g'(t) = 4t^3$. Met de kettingregel:
$\frac{\partial f}{\partial x} = 2xy$ en $\frac{\partial
f}{\partial y} = x^2$, geëvalueerd langs de kromme $(t, t^2)$:
\[
g'(t) = 1\cdot 2t\cdot t^2 + 2t\cdot t^2 = 2t^3 + 2t^3 = 4t^3 .
\]
De twee berekeningen stemmen overeen, en de splitsing is
betekenisvol: $2t^3$ van de groei komt van het \emph{naar rechts}
bewegen door de $x$-helling, en $2t^3$ van het \emph{omhoog}
bewegen door de $y$-helling. Op krommen waar geen gesloten vorm
voor $g$ bestaat, overleeft alleen de tweede berekening --- en dat
is het punt van de stelling.
\end{example}

\begin{remark}[De gradiënt lezen]
Langs een eenheidsrichting $u$ geeft de kettingregel op $t \mapsto
f(a + tu)$ de \emph{richtingsafgeleide} $\langle \nabla f(a),
u\rangle$: maximaal wanneer $u$ langs $\nabla f(a)$ wijst (volgens
Cauchy--Schwarz, \cref{thm:b1:euclid:cs}). De \omterm{def:b1:multivar:partial}{gradiënt} is de
richting van de steilste klim, en hij staat loodrecht op de
niveaukrommen $\{f = c\}$ (leid $f$ af langs een kromme die in een
niveauverzameling is getekend: de kettingregel geeft
$\langle\nabla f,\, \text{raakvector}\rangle = 0$).
\end{remark}

\begin{example}[Niveaukrommen en gradiënten, op één functie]
\label{ex:b1:multivar:levelcurves}
Neem $f(x, y) = x^2 - y^2$. Haar niveauverzamelingen: $\{f = c\}$
is een hyperbool die links-rechts opent voor $c > 0$ en
boven-onder voor $c < 0$, en het gekruiste paar rechten $y = \pm x$
voor $c = 0$ --- de hoogtekaart van een bergpas, met het \omterm{met:b1:multivar:monge}{zadelpunt}
in de oorsprong waar de twee nulniveaurechten elkaar kruisen.
\omterm{def:b1:multivar:partial}{Gradiënt}: $\nabla f = (2x, -2y)$. In het punt $(2, 1)$ (op het
niveau $c = 3$) is $\nabla f = (4, -2)$, terwijl de raakvector van
de niveaukromme, in de buurt van dat punt geparametriseerd door
$\bigl(t, \sqrt{t^2 - 3}\bigr)$, gelijk is aan $\bigl(1,
\frac{t}{\sqrt{t^2 - 3}}\bigr) = (1, 2)$ bij $t = 2$ --- en
inderdaad is
\[
\langle (4, -2),\ (1, 2)\rangle = 4 - 4 = 0 :
\]
de \omterm{def:b1:multivar:partial}{gradiënt} staat loodrecht op de hoogtelijn en wijst naar hogere
waarden van $f$ (hier: weg van de $y$-as). Nog twee lezingen: de
\omterm{def:b1:multivar:partial}{gradiënt} wordt precies nul in het zadel, waar de hoogtekaart
insnoert; en de raaklijn aan de niveaukromme in $(2,1)$ is $4(x -
2) - 2(y - 1) = 0$, dus $2x - y = 3$ --- de vergelijking
``$\langle \nabla f,\ M - M_0\rangle = 0$'' die de raaklijn aan de
ellips van \cref{exo:b1:curves:11} veralgemeent.
\end{example}

\begin{theorem}[Schwarz]\label{thm:b1:multivar:schwarz}
Is $f$ van klasse $C^2$ (de \omterm{def:b1:multivar:partial}{partiële afgeleiden} van de \omterm{def:b1:multivar:partial}{partiële
afgeleiden} bestaan en zijn \omterm{def:b1:continuity:continuous}{continu}), dan is\index{stelling van
Schwarz}
\[
\frac{\partial^2 f}{\partial x\,\partial y}
= \frac{\partial^2 f}{\partial y\,\partial x} .
\]
\end{theorem}
\admitted

\section{Lokale extrema}

\begin{method}[Extremumonderzoek, georganiseerd]
\label{met:b1:multivar:extremamethod}
\begin{enumerate}
  \item \emph{Los $\nabla f = 0$ volledig op.} Splits elke
        \omterm{def:b1:multivar:partial}{partiële afgeleide} af waar dat kan (producten van \omterm{def:b1:linmaps:def}{lineaire}
        factoren splitsen het stelsel in doorzichtige gevallen,
        zoals in \cref{ex:b1:multivar:fourpoints} hieronder); een
        vergeten geval is een vergeten \omterm{thm:b1:multivar:critical}{kritiek punt}.
  \item \emph{Classificeer elk punt} met de gegevens $r, s, t$ van
        Monge --- telkens \emph{in elk punt} opnieuw berekend,
        nooit één keer voor alles.
  \item \emph{Is $rt - s^2 = 0$}, onderzoek $f$ dan rechtstreeks
        langs goedgekozen krommen door het punt (eerst rechten, dan
        parabolen), op zoek naar ofwel twee tekens (geen extremum)
        ofwel een vast teken met een argument dat alle richtingen
        dekt.
  \item \emph{Stap terug voor het globale \omterm{def:b1:linmaps:kerim}{beeld}}: controleer het
        gedrag in het oneindige (een lokaal minimum hoeft geen
        globaal te zijn), en is het domein niet \omterm{def:b1:topology:open}{open}, behandel dan
        zijn \omterm{def:b1:topology:closure}{rand} apart (\cref{exo:b1:multivar:12}) --- de stelling
        over \omterm{thm:b1:multivar:critical}{kritieke} punten ziet alleen \omterm{def:b1:topology:closure}{inwendige} punten.
\end{enumerate}
\end{method}

\begin{theorem}[Kritieke punten]\label{thm:b1:multivar:critical}
Heeft $f$ (van klasse $C^1$ op de \omterm{def:b1:topology:open}{open verzameling} $U$) een lokaal
extremum in $(a,b) \in U$, dan is $\nabla f(a,b) = (0,0)$: het punt
is \emph{kritiek}\index{kritiek punt}.
\end{theorem}

\begin{proof}
De functies van één veranderlijke $x \mapsto f(x, b)$ en $y \mapsto
f(a, y)$ hebben \omterm{def:b1:topology:closure}{inwendige} lokale extrema in respectievelijk $a$ en
$b$: \cref{prop:b1:derivative:fermat} doodt beide \omterm{def:b1:multivar:partial}{partiële
afgeleiden}.
\end{proof}

\begin{method}[Toets van de tweede orde (notatie van Monge)]\label{met:b1:multivar:monge}
Stel in een \omterm{thm:b1:multivar:critical}{kritiek punt} van een $C^2$-functie
\[
r = \frac{\partial^2 f}{\partial x^2},
\qquad
s = \frac{\partial^2 f}{\partial x \partial y},
\qquad
t = \frac{\partial^2 f}{\partial y^2}
\qquad (\text{de waarden in het punt}).
\]
\begin{itemize}
  \item Is $rt - s^2 > 0$: een lokaal extremum --- een minimum voor
        $r > 0$, een maximum voor $r < 0$;
  \item is $rt - s^2 < 0$: geen extremum (een
        \emph{zadelpunt}\index{zadelpunt});
  \item is $rt - s^2 = 0$: de toets zwijgt; onderzoek
        rechtstreeks.
\end{itemize}
(De rechtvaardiging --- een taylorontwikkeling van Young op orde
$2$ en het tekenonderzoek van de kwadratische vorm $r h^2 + 2shk +
tk^2$ --- wordt in het tweede jaar uitgevoerd; hier wordt de toets
als werkgereedschap gebruikt.)
\end{method}

\begin{example}\label{ex:b1:multivar:extrema}
$f(x,y) = x^3 + y^3 - 3xy$. \omterm{thm:b1:multivar:critical}{Kritieke} punten: $\nabla f = (3x^2 -
3y,\; 3y^2 - 3x) = 0$ geeft $y = x^2$ en $x = y^2$, dus $x = x^4$:
$x \in \{0, 1\}$: de punten $(0,0)$ en $(1,1)$.

Tweede \omterm{def:b1:derivative:def}{afgeleiden}: $r = 6x$, $s = -3$, $t = 6y$. In $(0,0)$: $rt -
s^2 = -9 < 0$: een zadel. In $(1,1)$: $rt - s^2 = 36 - 9 > 0$ en $r
= 6 > 0$: een lokaal minimum, $f(1,1) = -1$. (Niet globaal: $f(x,
0) = x^3 \to -\infty$.)
\end{example}

\begin{example}[Een onderzoek met vier punten, volledig]
\label{ex:b1:multivar:fourpoints}
$f(x, y) = xy\,(3 - x - y) = 3xy - x^2y - xy^2$. \omterm{def:b1:multivar:partial}{Gradiënt}:
\[
\frac{\partial f}{\partial x} = y\,(3 - 2x - y),
\qquad
\frac{\partial f}{\partial y} = x\,(3 - x - 2y).
\]
\omterm{thm:b1:multivar:critical}{Kritieke} punten: is $y = 0$, dan geeft de tweede vergelijking $x
\in \{0, 3\}$; is $x = 0$, dan geeft de eerste $y \in \{0, 3\}$; is
$xy \neq 0$, los dan $2x + y = 3$, $x + 2y = 3$ op: $x = y = 1$.
Vier punten: $(0,0)$, $(3,0)$, $(0,3)$, $(1,1)$. Tweede \omterm{def:b1:derivative:def}{afgeleiden}:
$r = -2y$, $s = 3 - 2x - 2y$, $t = -2x$.
\begin{itemize}
  \item $(1,1)$: $r = -2$, $s = -1$, $t = -2$: $rt - s^2 = 3 > 0$
        en $r < 0$: een lokaal maximum, $f(1,1) = 1$.
  \item $(0,0)$: $r = t = 0$, $s = 3$: $rt - s^2 = -9 < 0$: een
        zadel; net zo $(3, 0)$ ($s = -3$) en $(0, 3)$: drie
        zadels.
\end{itemize}
Het maximum is slechts lokaal: $f(-T, -T) = T^2(3 + 2T) \to
+\infty$. Symmetriecontrole: $f(x, y) = f(y, x)$, en inderdaad zijn
de \omterm{thm:b1:multivar:critical}{kritieke} \omterm{def:b1:logic:sets}{verzameling} en de classificatie symmetrisch in $x
\leftrightarrow y$. Interpretatie: onder de rechthoeken met speling
$x, y \geq 0$, $x + y \leq 3$ is het product $xy(3 - x - y)$ van de
drie ``delen'' van $3$ het grootst wanneer de delen gelijk zijn ---
een schaduw in twee veranderlijken van het
rekenkundig--meetkundige beginsel.
\end{example}

\begin{remark}[Veelgemaakte fouten]
\label{rem:b1:multivar:pitfalls}
\emph{\omterm{def:b1:multivar:partial}{Partiële afgeleiden} kunnen bestaan in een punt van
discontinuïteit}: de radiale val $g(x,y) = \frac{xy}{x^2+y^2}$ uit
\cref{ex:b1:multivar:trap} heeft $\frac{\partial g}{\partial
x}(0,0) = \frac{\partial g}{\partial y}(0,0) = 0$ (beide
beperkingen tot de assen zijn identiek nul), en toch heeft $g$
geen limiet in de oorsprong --- \omterm{def:b1:multivar:partial}{partiële afgeleiden} peilen slechts
twee richtingen, terwijl de \omterm{def:b1:continuity:continuous}{continuïteit} ze alle nodig heeft;
alleen de hypothese $C^1$ herstelt de orde
(\cref{thm:b1:multivar:c1}).
\emph{Limieten langs rechten volstaan nooit}: de functie van
\cref{exo:b1:multivar:3} heeft langs elke rechte limiet $0$ en
toch geen limiet --- probeer altijd parabolen (of poolgrenzen die
uniform in $\theta$ gelden).
\emph{\omterm{thm:b1:multivar:critical}{Kritiek} is noodzakelijk, niet voldoende}: zadels zijn er in
overvloed (drie van de vier punten in
\cref{ex:b1:multivar:fourpoints}); en de stelling geldt alleen op
\emph{\omterm{def:b1:topology:open}{open}} \omterm{def:b1:logic:sets}{verzamelingen} --- op een \omterm{def:b1:topology:closed}{gesloten} schijf kunnen de
extrema op de \omterm{def:b1:topology:closure}{rand} liggen met een \omterm{def:b1:multivar:partial}{gradiënt} ongelijk aan nul
(\cref{exo:b1:multivar:12}).
\emph{Het zwijgende geval $rt - s^2 = 0$ zwijgt werkelijk}: $x^4 +
y^4$ (minimum) en $x^3 + y^3$ (geen van beide) hebben allebei $r =
s = t = 0$ in de oorsprong; alleen een rechtstreeks tekenonderzoek
beslist (\cref{exo:b1:multivar:6}, de functie $h$).
\emph{De \omterm{def:b1:multivar:partial}{gradiënt} staat loodrecht op de niveaukrommen, niet
erlangs}: om een hoogtelijn te volgen beweeg je loodrecht op
$\nabla f$; om het snelst te klimmen beweeg je erlangs --- de twee
verwarren keert de meetkunde van elke hoogtekaart om.
\end{remark}

\begin{remark}[Waarheen twee veranderlijken leiden]
\label{rem:b1:multivar:whereused}
Dit hoofdstuk is een deuropening. De \omterm{def:b1:multivar:partial}{gradiënt} en de kettingregel
breiden zich woordelijk uit tot $n$ veranderlijken in het volume
van bachelorjaar 2, waar de $o(\norm{(h,k)})$ van
\cref{thm:b1:multivar:c1} de \emph{differentiaal} wordt en de
toets van Monge volledig wordt bewezen met de formule van Taylor
van orde twee en met kwadratische vormen. Het bijzondere geval dat
\emph{dit} jaar kan worden beslecht --- kwadratische functies,
waarvoor de ontwikkeling van tweede orde exact is --- is het
onderwerp van de weekendopgave, en het is toevallig het geval dat
de gegevensaanpassing van de wereld aandrijft: regressie met
\omterm{pb:b1:multivar:1}{kleinste kwadraten}. Extrema onder voorwaarden
(\cref{exo:b1:multivar:5} was een voorproefje) worden in
bachelorjaar 2 de multiplicatoren van Lagrange; harmonische
functies (\cref{exo:b1:multivar:7}) keren terug in de complexe
analyse van het volume van bachelorjaar 3.
\end{remark}

\begin{remark}[Vooruitzichten binnen boek 3: het jaar gesloten]
\label{rem:b1:multivar:perspectives}
In dit hoofdstuk schudden de twee helften van het volume elkaar de
hand. De analytische helft leverde haar gereedschap één \omterm{def:b1:derivative:def}{afgeleide}
tegelijk: de middelwaardestelling drijft
\cref{thm:b1:multivar:c1} aan, taylorontwikkelingen drijven de
extremumtoetsen aan, en de $\varepsilon$'s van
\cref{ch:b1:topology} kwamen terug met bollen in plaats van
\omterm{prop:b1:reals:intervals}{intervallen}. De algebraïsche helft leverde de meetkunde: de
\omterm{def:b1:multivar:partial}{gradiënt} wordt gelezen via het \omterm{def:b1:euclid:def}{inwendige product} van
\cref{ch:b1:euclid} (Cauchy--Schwarz maakt hem tot de steilste
richting), de gegevens $(r, s, t)$ van Monge vormen een
symmetrische matrix uit \cref{ch:b1:matrices} met de
determinanttoets van \cref{ch:b1:det}, en de weekendopgave laat de
\omterm{thm:b1:euclid:projection}{orthogonale projectie} op gegevensvectoren lopen. Zelfs de krommen
van \cref{ch:b1:curves} keren terug als niveauverzamelingen. Een
lezer die kan reconstrueren waarom elk van deze vijf overdrachten
werkt, heeft in feite het hele jaar herhaald --- en dat is het
echte doel van dit slothoofdstuk.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b1:multivar:1}
Bereken de \omterm{def:b1:multivar:partial}{partiële afgeleiden}: $f(x,y) = x^2 y + \eu^{xy}$;
$\;g(x,y) = \ln(x^2 + y^2)$ (op $\R^2\setminus\{0\}$); $\;h(x,y) =
\arctan\frac yx$ (op $x > 0$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:multivar:2}
Onderzoek de \omterm{def:b1:continuity:continuous}{continuïteit} in $(0,0)$ (met daar de waarde $0$) van:
\[
f(x,y) = \frac{x^2 y}{x^2 + y^2},
\qquad
g(x,y) = \frac{xy}{x^2 + y^2},
\qquad
h(x,y) = \frac{x^3 + y^3}{x^2 + y^2}.
\]
\emph{(Poolcoördinaten $x = \rho\cos\theta$, $y = \rho\sin\theta$
helpen: begrens waar mogelijk door een functie van $\rho$ alleen.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:3}
Zij $f(x,y) = \dfrac{x^2 y}{x^4 + y^2}$ ($f(0,0) = 0$). Bewijs dat
$f$ in de oorsprong \emph{langs elke rechte} limiet $0$ heeft, maar
dat $f\bigl(x, x^2\bigr) = \frac12$: $f$ is niet \omterm{def:b1:continuity:continuous}{continu} in
$(0,0)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:multivar:4}
Ga de stelling van Schwarz met de hand na op $f(x, y) = x^3 y^2 +
\sin(xy)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:5}
Zij $f$ van klasse $C^1$ op $\R^2$ en $g(t) = f(\cos t, \sin t)$.
Druk $g'(t)$ uit met de kettingregel. Leid af dat $f$, beperkt tot
de eenheidscirkel, extrema aanneemt in punten waar $\nabla f$
evenwijdig is met de voerstraal.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:6}
Bepaal en classificeer de \omterm{thm:b1:multivar:critical}{kritieke} punten van:
\[
f(x, y) = x^2 + xy + y^2 - 3x,
\qquad
g(x, y) = x^2 - y^2 + 4y,
\qquad
h(x, y) = x^4 + y^4 - 2(x - y)^2 .
\]
\emph{(Voor $h$ zwijgt de determinanttoets in de oorsprong:
onderzoek $h(x, x)$ en $h(x, -x)$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:7}
Een functie $f$ heet \emph{harmonisch} wanneer $\frac{\partial^2
f}{\partial x^2} + \frac{\partial^2 f}{\partial y^2} = 0$. Ga na dat
$x^2 - y^2$, $xy$, $\eu^x\cos y$ en $\ln(x^2 + y^2)$ (buiten de
oorsprong) harmonisch zijn.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:8}
Bepaal het punt van het vlak $\{x + 2y - z = 4\} \subseteq \R^3$
dat het dichtst bij de oorsprong ligt, op twee manieren: met de
\omterm{thm:b1:euclid:projection}{orthogonale projectie} (\cref{ch:b1:euclid}), en door de functie van
twee veranderlijken $f(x, y) = x^2 + y^2 + (x + 2y - 4)^2$ te
minimaliseren die door eliminatie van $z$ ontstaat.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:9}
(Laplaciaan in poolcoördinaten, eerste contact) Zij $f$ van klasse
$C^2$ op $\R^2 \setminus \{0\}$ en \emph{radiaal}: $f(x, y) =
\varphi\bigl(\sqrt{x^2+y^2}\bigr)$ met $\varphi$ van klasse $C^2$
op $\intoo{0}{+\infty}$. Bewijs dat
\[
\frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial
y^2} = \varphi''(\rho) + \frac{\varphi'(\rho)}{\rho},
\qquad \rho = \sqrt{x^2 + y^2},
\]
en bepaal alle radiale harmonische functies op het geperforeerde
vlak.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:10}
Geef de vergelijking van het raakvlak aan de grafiek $z = xy$ in
het punt $(1, 1, 1)$. Bepaal daarna alle punten van de grafiek van
$f(x,y) = x^3 + y^3 - 3xy$ waar het raakvlak \emph{horizontaal} is,
en breng het antwoord in verband met
\cref{ex:b1:multivar:extrema}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:11}
Zij $f$ van klasse $C^1$ op $\R^2$.
\begin{enumerate}
  \item Is $\frac{\partial f}{\partial x} = 0$ overal, bewijs dan
        dat $f(x, y)$ alleen van $y$ afhangt.
  \item Bepaal alle $C^1$-oplossingen van de vergelijking
        $\frac{\partial f}{\partial x} = \frac{\partial f}{\partial
        y}$ op $\R^2$. \emph{(Stel $g(u, v) =
        f\bigl(\frac{u+v}2, \frac{u-v}2\bigr)$ en bereken
        $\frac{\partial g}{\partial v}$ met de kettingregel.)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:12}
Bepaal het globale maximum en minimum van $f(x, y) = xy$ op de
\omterm{def:b1:topology:closed}{gesloten} schijf $x^2 + y^2 \leq 1$. \emph{(Neem de extremumstelling
in twee veranderlijken aan: een \omterm{def:b1:continuity:continuous}{continue} functie op de \omterm{def:b1:topology:closed}{gesloten}
schijf bereikt haar grenzen --- bewezen in het volume van
bachelorjaar 2. Behandel de \omterm{def:b1:topology:open}{open} schijf met \omterm{thm:b1:multivar:critical}{kritieke} punten en de
randcirkel met de parametrisering van
\cref{exo:b1:multivar:5}.)}
\end{exercise}

\section{Opgave: kleinste kwadraten en de regressierechte}

\begin{problem}\label{pb:b1:multivar:1}
Welke rechte gaat, gegeven $n$ gegevenspunten, ``het dichtst''
langs alle punten? Legendre en Gauss antwoordden: de rechte die de
som van de \emph{gekwadrateerde} verticale fouten minimaliseert ---
omdat die minimalisering precies door \omterm{def:b1:linmaps:def}{lineaire} algebra exact
oplosbaar is. Deze opgave bewijst eerst de toets van Monge uit
\cref{met:b1:multivar:monge} eerlijk voor kwadratische functies
(het enige geval waarin de ontwikkeling van tweede orde exact is),
en bouwt daarna de \omterm{pb:b1:multivar:1}{normaalvergelijkingen}, de \omterm{pb:b1:multivar:1}{regressierechte} en de
correlatiecoëfficiënt bovenop de euclidische meetkunde van
\cref{ch:b1:euclid}.\index{kleinste kwadraten}\index{regressierechte}
\index{normaalvergelijkingen}\index{correlatie}

\medskip
\noindent\textbf{Deel I --- Kwadratische functies: de toets van
Monge, bewezen.} Leg reële $r, s, t$ vast en zij $q(h, k) = r h^2 +
2s hk + t k^2$.
\begin{enumerate}
  \item Stel dat $r \neq 0$. Stel de vorm met afgesplitst kwadraat
        \[
        q(h, k) = r\Bigl(h + \frac{s}{r}k\Bigr)^{\!2}
        + \frac{rt - s^2}{r}\,k^2
        \]
        vast, en leid af: is $rt - s^2 > 0$, dan heeft $q$ buiten
        de oorsprong strikt het teken van $r$; is $rt - s^2 < 0$,
        dan neemt $q$ beide tekens aan.
  \item Beslecht de overige gevallen: $r = 0$, $t \neq 0$
        (symmetrisch); en $r = t = 0$ ($q = 2shk$). Besluit: $q$
        neemt beide tekens aan dan en slechts dan als $rt - s^2 <
        0$, en $q$ wordt alleen in de oorsprong nul dan en slechts
        dan als $rt - s^2 > 0$.
  \item Zij nu $f(x, y) = \frac12\bigl(r x^2 + 2s xy + t
        y^2\bigr) + \beta x + \gamma y + \delta$ een kwadratische
        functie met een \omterm{thm:b1:multivar:critical}{kritiek punt} $X_0 = (a, b)$. Bewijs de
        \emph{exacte} ontwikkeling
        \[
        f(X_0 + (h,k)) = f(X_0) + \tfrac12\,q(h, k)
        \qquad (\text{zonder restterm}),
        \]
        en leid de classificatie van Monge voor kwadratische
        functies af: een strikt globaal minimum als $rt - s^2 >
        0$, $r > 0$; een strikt globaal maximum als $rt - s^2 >
        0$, $r < 0$; een zadel als $rt - s^2 < 0$.
  \item Stel $rt - s^2 > 0$ en $r > 0$ (dus ook $t > 0$). Bewijs de
        expliciete coërciviteitsgrens
        \[
        q(h, k) \geq c\,(h^2 + k^2),
        \qquad
        c = \frac{rt - s^2}{r + t} > 0 .
        \]
        \emph{(Toon aan dat de kwadratische vorm $q - c(h^2 +
        k^2)$, met coëfficiënten $r - c$, $s$, $t - c$, nog steeds
        aan de gunstige kant van de discriminanttoets ligt.)}
  \item Leid af dat een kwadratische functie met een positief
        definiet kwadratisch deel naar $+\infty$ nadert als
        $\norm{(x,y)} \to \infty$, en dus een \emph{eenduidige
        globale} minimalisator heeft: haar \omterm{thm:b1:multivar:critical}{kritieke} punt.
        Vergelijk met \cref{ex:b1:multivar:extrema}, waar een
        lokaal minimum van een niet-kwadratische functie niet
        globaal was.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De \omterm{pb:b1:multivar:1}{normaalvergelijkingen}.} Zij $C_1,
C_2, b \in \R^n$ (met het canonieke \omterm{def:b1:euclid:def}{inwendige product}), en
\[
f(u, v) = \norm{u\,C_1 + v\,C_2 - b}^2 .
\]
\begin{enumerate}[resume]
  \item Werk $f$ uit en toon aan dat zij een kwadratische functie
        van $(u, v)$ is met als coëfficiënten van het kwadratische
        deel $r = 2\norm{C_1}^2$, $s = 2\langle C_1, C_2\rangle$,
        $t = 2\norm{C_2}^2$.
  \item Toon aan dat $rt - s^2 > 0$ dan en slechts dan als $(C_1,
        C_2)$ \omterm{def:b1:vspaces:free}{vrij} is \emph{(het gelijkheidsgeval van
        Cauchy--Schwarz, \cref{thm:b1:euclid:cs})}; neem dit vanaf
        nu aan.
  \item Toon aan dat de \omterm{thm:b1:multivar:critical}{kritieke} vergelijkingen $\nabla f = 0$ de
        \emph{\omterm{pb:b1:multivar:1}{normaalvergelijkingen}}
        \[
        \begin{pmatrix}
        \norm{C_1}^2 & \langle C_1, C_2\rangle\\
        \langle C_1, C_2\rangle & \norm{C_2}^2
        \end{pmatrix}
        \begin{pmatrix} u\\ v\end{pmatrix}
        =
        \begin{pmatrix} \langle C_1, b\rangle\\ \langle C_2,
        b\rangle\end{pmatrix}
        \]
        zijn --- met links de grammatrix van
        \cref{exo:b1:euclid:11} --- en dat zij precies zeggen: $b -
        (uC_1 + vC_2) \perp C_1, C_2$. Besluit met deel I en
        \cref{thm:b1:euclid:projection}: de eenduidige
        minimalisator geeft $p = uC_1 + vC_2 = $ de \omterm{thm:b1:euclid:projection}{orthogonale
        projectie} van $b$ op $\operatorname{Vect}(C_1, C_2)$.
  \item Toon aan dat de minimale waarde $\norm{b}^2 - \langle p,
        b\rangle = \norm{b - p}^2$ is, en teken het \omterm{def:b1:linmaps:kerim}{beeld} van
        Pythagoras.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De \omterm{pb:b1:multivar:1}{regressierechte}.}
Gegevenspunten $(x_1, y_1), \dots, (x_n, y_n)$, met niet alle $x_i$
gelijk. Minimaliseer
\[
E(\alpha, \beta) = \sum_{i=1}^{n}\bigl(y_i - \alpha - \beta
x_i\bigr)^2 .
\]
Schrijf $\bar x = \frac1n\sum x_i$, $\bar y = \frac1n\sum y_i$, $v_x
= \frac1n\sum x_i^2 - \bar x^2$, $v_y = \frac1n\sum y_i^2 - \bar
y^2$, $c_{xy} = \frac1n\sum x_i y_i - \bar x\bar y$.
\begin{enumerate}[resume]
  \item Herken deel II met $C_1 = (1, \dots, 1)$, $C_2 = (x_1,
        \dots, x_n)$, $b = (y_1, \dots, y_n)$, en schrijf de
        \omterm{pb:b1:multivar:1}{normaalvergelijkingen}
        \[
        n\alpha + \Bigl(\sum x_i\Bigr)\beta = \sum y_i,
        \qquad
        \Bigl(\sum x_i\Bigr)\alpha + \Bigl(\sum
        x_i^2\Bigr)\beta = \sum x_i y_i .
        \]
  \item Los ze op:
        \[
        \beta = \frac{c_{xy}}{v_x},
        \qquad
        \alpha = \bar y - \beta\,\bar x ,
        \]
        en merk op dat de \emph{\omterm{pb:b1:multivar:1}{regressierechte}} $y = \alpha +
        \beta x$ door het middelpunt $(\bar x, \bar y)$ gaat.
  \item Ga na dat $v_x > 0$ precies wanneer de $x_i$ niet alle
        gelijk zijn, en breng dit in overeenstemming met vraag 7.
  \item Bewijs dat de minimale fout
        \[
        E_{\min} = n\Bigl(v_y - \frac{c_{xy}^2}{v_x}\Bigr)
        = n\,v_y\,(1 - \rho^2),
        \qquad
        \rho = \frac{c_{xy}}{\sqrt{v_x v_y}}
        \quad (v_y \neq 0)
        \]
        is. Leid af dat $\abs\rho \leq 1$, met $\abs\rho = 1$ dan
        en slechts dan als de gegevens volmaakt op één lijn
        liggen.
  \item Uitgewerkt voorbeeld: bereken voor de gegevens $(0,0)$,
        $(1,1)$, $(2,3)$, $(3,4)$ de waarden $\bar x, \bar y, v_x,
        c_{xy}$, de \omterm{pb:b1:multivar:1}{regressierechte}, de vier resten en
        $E_{\min}$.
  \item Bewijs dat de resten $\varepsilon_i = y_i - \alpha - \beta
        x_i$ van de optimale rechte voldoen aan $\sum_i
        \varepsilon_i = 0$ en $\sum_i x_i\varepsilon_i = 0$, en
        interpreteer beide via de orthogonaliteit.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Varianten en toepassingen.}
\begin{enumerate}[resume]
  \item (Beste constante) Toon aan dat de constante $\alpha$ die
        $\sum_i (y_i - \alpha)^2$ minimaliseert het gemiddelde
        $\bar y$ is, en dat de minimale waarde $n\,v_y$ bedraagt:
        de variantie meet in welke mate de gegevens niet constant
        zijn.
  \item (Rechte door de oorsprong) Toon aan dat de helling die
        $\sum_i (y_i - \beta x_i)^2$ minimaliseert gelijk is aan
        $\beta_0 = \frac{\sum x_iy_i}{\sum x_i^2}$, en geef een
        voorwaarde op de gegevens waaronder $\beta_0$ samenvalt met
        de helling $c_{xy}/v_x$ uit vraag 11.
  \item (Afstand tot een rechte, opnieuw) Minimaliseer voor een
        punt $M$ en de rechte $D$ door $P_0$ met de
        eenheidsrichtingsvector $w$ de functie $g(\tau) = \norm{P_0
        + \tau w - M}^2$ en leid $d(M, D)^2 = \norm{M - P_0}^2 -
        \langle M - P_0, w\rangle^2$ af; vind de formule $d =
        \frac{\abs{a x_0 + b y_0 + c}}{\sqrt{a^2 + b^2}}$ terug
        voor de rechte $ax + by + c = 0$ in het vlak.
  \item (Variantieanalyse) Bewijs de ontbinding $v_y = \beta^2 v_x
        + \frac{E_{\min}}n$: de variantie van de $y_i$ splitst zich
        in het deel dat door de rechte wordt verklaard plus de
        restvariantie.
  \item (Aanpassing met een parabool) Toon aan dat voor het
        aanpassen van $y = a + bx + cx^2$ de \omterm{pb:b1:multivar:1}{normaalvergelijkingen}
        het $3 \times 3$ stelsel zijn met de momentenmatrix
        \[
        \begin{pmatrix}
        n & \sum x_i & \sum x_i^2\\
        \sum x_i & \sum x_i^2 & \sum x_i^3\\
        \sum x_i^2 & \sum x_i^3 & \sum x_i^4
        \end{pmatrix},
        \]
        een grammatrix die inverteerbaar is zodra drie van de $x_i$
        verschillend zijn \emph{(de vrijheid van $(1, X, X^2)$
        bemonsterd in de gegevens, \cref{exo:b1:euclid:11};
        vergelijk de momentenmatrices van de weekendopgave van
        \cref{ch:b1:det})}.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Robuustheid, en synthese.}
\begin{enumerate}[resume]
  \item Classificeer met deel I de \omterm{thm:b1:multivar:critical}{kritieke} punten van de drie
        kwadratische functies
        \[
        x^2 + xy + y^2, \qquad x^2 + 3xy + y^2, \qquad
        x^2 + 2xy + y^2 ,
        \]
        waarbij je het ontaarde derde geval met de hand behandelt
        (waar wordt het minimum bereikt?).
  \item (Uitschieterexperiment) Voeg het punt $(10, 0)$ toe aan de
        gegevens van vraag 14 en bereken de helling $\beta$
        opnieuw. Wat is er gebeurd, en waarom is de
        \emph{gekwadrateerde} fout zo gevoelig voor één ver
        verwijderd punt?
  \item (Gewogen \omterm{pb:b1:multivar:1}{kleinste kwadraten}) Minimaliseer, gegeven
        gewichten $w_i > 0$, de som $\sum_i w_i(y_i - \alpha -
        \beta x_i)^2$: toon aan dat de oplossing wordt gegeven door
        dezelfde formules als in vraag 11, met gewogen
        gemiddelden, variantie en covariantie (definieer die).
  \item Toon aan dat $E_{\min} = 0$ voor de optimale rechte
        afdwingt dat alle punten er precies op liggen, en verbind
        dit met het gelijkheidsgeval $\abs\rho = 1$ uit vraag 13;
        toets het op de gegevens $(1,1), (2,3), (3,5)$, die op één
        lijn liggen.
