\chapter{गणना}\label{ch:b1:counting}

\omterm{def:b1:counting:card}{परिमित} समुच्चयों की गणना प्रारंभिक लगती है --- और शीघ्र ही सूक्ष्म हो जाती
है। यह अध्याय \omterm{def:b1:counting:card}{गणनांक} की उचित परिभाषा देता है (एकैकी आच्छादनों के द्वारा,
\cref{ch:b1:logic} की भावना में), उन गिनी-चुनी गणना-नियमों को स्थापित करता
है जिनसे सब कुछ निकलता है, और उनसे चिरपरिचित गणनाएँ व्युत्पन्न करता है:
सूचियाँ, \omterm{def:b1:counting:objects}{क्रमचय}, उपसमुच्चय, \omterm{def:b1:counting:objects}{द्विपद गुणांक}।

\section{परिमित समुच्चयों का गणनांक}

\begin{definition}[परिमित समुच्चय, गणनांक]\label{def:b1:counting:card}
$n \in \N^*$ के लिए $\intint{1}{n} = \{1, 2, \dots, n\}$ लिखिए। \omterm{def:b1:logic:sets}{समुच्चय} $E$
\emph{परिमित}\index{परिमित समुच्चय} तब कहलाता है जब $E = \emptyset$ हो, या
किसी $n \in \N^*$ के लिए $\intint{1}{n}$ से $E$ पर कोई एकैकी आच्छादन हो; यह
$n$ अद्वितीय होता है (\cref{thm:b1:counting:welldef}) और $E$ का
\emph{गणनांक}\index{गणनांक} कहलाता है, जिसे $\abs{E}$ लिखा जाता है
($\abs{\emptyset} = 0$ के साथ)।
\end{definition}

\begin{theorem}[गणनांक सुपरिभाषित है]\label{thm:b1:counting:welldef}
यदि $m \neq n$, तो $\intint{1}{m}$ से $\intint{1}{n}$ पर कोई एकैकी आच्छादन
नहीं है। और अधिक ठीक-ठीक: यदि $m > n$, तो $\intint{1}{m}$ से $\intint{1}{n}$
में कोई \omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} नहीं है।
\end{theorem}

\begin{proof}
हम $n$ पर आगमन से यह \omterm{def:b1:logic:statement}{कथन} सिद्ध करते हैं: \emph{सभी $m > n$ के लिए कोई \omterm{def:b1:logic:inj}{एकैकी}
\omterm{def:b1:logic:map}{प्रतिचित्रण} $\intint{1}{m} \to \intint{1}{n}$ नहीं है}। $n = 0$ के लिए
लक्ष्य रिक्त है और $m \geq 1$: कोई \omterm{def:b1:logic:map}{प्रतिचित्रण} है ही नहीं। \omterm{def:b1:logic:statement}{कथन} को $n$ के लिए
मान लीजिए, और मान लीजिए $f \colon \intint{1}{m} \to \intint{1}{n+1}$ एक
\omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} है, जहाँ $m > n + 1$। यदि मान $n + 1$ प्राप्त नहीं होता,
तो $f$ $\intint{1}{n}$ में \omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} है, जो आगमन परिकल्पना का विरोध
करता है। अन्यथा ठीक एक $a$ के लिए $f(a) = n + 1$; $f(a)$ और $f(m)$ की
अदला-बदली कीजिए (औपचारिक रूप से: दोनों मानों के व्यत्यास के साथ संयोजन
कीजिए), जिससे नए \omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} $g$ के लिए $g(m) = n + 1$ हो जाए। तब
$\intint{1}{m-1}$ पर $g$ का प्रतिबंधन $\intint{1}{n}$ में \omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण}
है, जहाँ $m - 1 > n$ --- फिर विरोधाभास।
\end{proof}

\begin{corollary}[कबूतरखाना सिद्धांत]\label{cor:b1:counting:pigeonhole}
यदि $\abs{E} > \abs{F}$, तो कोई भी \omterm{def:b1:logic:map}{प्रतिचित्रण} $f \colon E \to F$ \omterm{def:b1:logic:inj}{एकैकी} नहीं
है: $E$ के कोई दो अवयव अपना प्रतिबिंब साझा करते हैं\index{कबूतरखाना
सिद्धांत}।
\end{corollary}

\begin{proof}
$\abs E = m$, $\abs F = n$ लिखिए, जहाँ $m > n$, और एकैकी आच्छादन $u \colon
\intint1m \to E$ तथा $v \colon F \to \intint1n$ चुनिए। यदि $f$ \omterm{def:b1:logic:inj}{एकैकी} होता,
तो $v \circ f \circ u$ $\intint1m$ से $\intint1n$ में \omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} होता
(\omterm{def:b1:logic:inj}{एकैकी} प्रतिचित्रणों का संयोजन, \cref{prop:b1:logic:comp}), जो
\cref{thm:b1:counting:welldef} का विरोध करता।
\end{proof}

\begin{remark}[मध्यांतर: प्रमेय की उपपत्ति में अदला-बदली क्यों?]\label{rem:b1:counting:swapinterlude}
\cref{thm:b1:counting:welldef} की उपपत्ति में इस अध्याय की पहली सचमुच चतुर
चाल है, जिसे धीरे-धीरे दोहराना उचित है। बाधा यह है: आगमन परिकल्पना लगाने के
लिए हम स्रोत का अंतिम बिंदु $m$ \emph{और} लक्ष्य का अंतिम बिंदु $n+1$ दोनों
हटाना चाहते हैं, पर $f$ किसी दूसरे बिंदु $a$ को $n + 1$ पर भेज सकता है, और
तब लक्ष्य-बिंदु हटाने से \omterm{def:b1:logic:map}{प्रतिचित्रण} अन्यत्र बिगड़ जाता है। उपाय: $f$ को
दोनों \emph{मानों} $f(a)$ और $f(m)$ के व्यत्यास के साथ संयोजित कीजिए --- यह
लक्ष्य का एकैकी आच्छादन है, अतः एकैकीयता बनी रहती है --- इसके बाद कष्टप्रद
मान $n + 1$ निरापद स्थान $m$ पर आ बैठता है और दोनों विलोपन स्वच्छ हो जाते
हैं। ``पहले मानकीकरण, फिर कटाई'' का यह प्रतिरूप बार-बार लौटता है: इसी से इस
अध्याय की सप्ताहांत समस्या में व्यत्यय-पुनरावृत्ति $\sigma^{-1}(n+1)$ की
दिशा बदलती है, और इसी से \cref{ch:b1:structures} की सममित समूह वाली समस्या
में क्रमचयों की मरम्मत होती है।
\end{remark}

\begin{proposition}[एकैकी, आच्छादक प्रतिचित्रण और गणनांक]\label{prop:b1:counting:injsur}
मान लीजिए $E, F$ \omterm{def:b1:counting:card}{परिमित समुच्चय} हैं, जहाँ $\abs{E} = \abs{F}$, और $f \colon
E \to F$। तब
\[
f \text{ एकैकी} \iff f \text{ आच्छादक} \iff f \text{ एकैकी आच्छादक}.
\]
\end{proposition}

\begin{proof}
मान लीजिए $f$ \omterm{def:b1:logic:inj}{एकैकी} है। तब $f$ $E$ से $f(E)$ पर एकैकी आच्छादन है, अतः
$\abs{f(E)} = \abs{E} = \abs{F}$। यदि $f(E)$ $F$ के किसी बिंदु $y_0$ तक न
पहुँचता, तो $f$ $E$ का $F \setminus \{y_0\}$ में \omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} होता,
जिसका \omterm{def:b1:counting:card}{गणनांक} $\abs{F} - 1 < \abs{E}$ है --- कबूतरखाना सिद्धांत से असंभव। अतः
$f(E) = F$: $f$ \omterm{def:b1:logic:inj}{आच्छादक} है, अतः \omterm{def:b1:logic:inj}{एकैकी आच्छादक} भी।

मान लीजिए $f$ \omterm{def:b1:logic:inj}{आच्छादक} है। प्रत्येक $y \in F$ के लिए एक \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} $s(y)
\in E$ चुनिए; तब $f \circ s = \mathrm{id}_F$, अतः $s$ \omterm{def:b1:logic:inj}{एकैकी} है
(\cref{prop:b1:logic:comp})। पिछले अनुच्छेद को $s$ पर लगाने से (\omterm{def:b1:counting:card}{गणनांक} बराबर
हैं) $s$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है। $f \circ s = \mathrm{id}_F$ से $f = \mathrm{id}_F
\circ s^{-1} = s^{-1}$ मिलता है, अतः $f$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है। अंत में, \omterm{def:b1:logic:inj}{एकैकी
आच्छादक} \omterm{def:b1:logic:map}{प्रतिचित्रण} परिभाषा से ही \omterm{def:b1:logic:inj}{एकैकी} और \omterm{def:b1:logic:inj}{आच्छादक} दोनों है, जिससे
निहितार्थों का चक्र बंद हो जाता है।
\end{proof}

\begin{example}[परिमितता अनिवार्य है]\label{ex:b1:counting:finiteonly}
किसी \emph{\omterm{def:b1:counting:card}{परिमित}} \omterm{def:b1:logic:sets}{समुच्चय} पर \cref{prop:b1:counting:injsur} एक सशक्त
संक्षेप है: $E$ से स्वयं उसी में जाने वाला कोई भी \omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} स्वतः
$E$ का \omterm{def:b1:counting:objects}{क्रमचय} होता है --- \omterm{def:b1:logic:inj}{एकैकी} आच्छादकता का आधा भाग मुफ़्त मिल जाता है।
अनंत समुच्चयों पर दोनों निहितार्थ ढह जाते हैं: $n \mapsto n + 1$ $\N$ से
$\N$ में \omterm{def:b1:logic:inj}{एकैकी} है पर $0$ तक नहीं पहुँचता, और $0 \mapsto 0$ तथा $n \geq 1$ के
लिए $n \mapsto n - 1$ भेजने वाला \omterm{def:b1:logic:map}{प्रतिचित्रण} $\N \to \N$ \omterm{def:b1:logic:inj}{आच्छादक} है पर \omterm{def:b1:logic:inj}{एकैकी}
नहीं। जब भी यह प्रतिज्ञप्ति लगाई जाती है, परिमितता की परिकल्पना सचमुच काम कर
रही होती है --- यही विषय \cref{ch:b1:logic} की सप्ताहांत समस्या दूसरी ओर से
देखती है, जहाँ अनंत \omterm{def:b1:logic:sets}{समुच्चय} ठीक वही हैं जो ऐसे स्व-प्रतिचित्रण स्वीकार करते
हैं।
\end{example}

\begin{example}[आधा काम, मुफ़्त में]\label{ex:b1:counting:mod7}
$\{0, 1, \dots, 6\}$ पर उस \omterm{def:b1:logic:map}{प्रतिचित्रण} $f$ पर विचार कीजिए जो $k$ को $7$ से
भाग देने पर $3k$ के शेषफल पर भेजता है; उसकी मान-सारणी यह है
\[
0,\ 3,\ 6,\ 2,\ 5,\ 1,\ 4 .
\]
क्या $f$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है? केवल एकैकीयता पर्याप्त है
(\cref{prop:b1:counting:injsur}): यदि $3k$ और $3k'$ के शेषफल समान हों, तो
$7$ $3(k - k')$ को विभाजित करता है, और चूँकि $7$ अभाज्य है और $3$ को विभाजित
नहीं करता, अतः वह $k - k'$ को विभाजित करता है (यूक्लिड की प्रमेयिका, यहाँ
विद्यालय स्तर पर प्रयुक्त और \cref{ch:b1:arith} में सिद्ध); $\abs{k - k'}
\leq 6$ के साथ इससे $k = k'$ आ जाता है। आच्छादकता मुफ़्त में मिलती है ---
प्रत्येक $c$ के लिए $3k \equiv c$ हल करने की आवश्यकता नहीं, यद्यपि सारणी
पुष्टि करती है कि हर मान ठीक एक बार आता है। यह संक्षेप एक परिश्रमी घोड़ा है:
वह मापांकी गुणन की उत्क्रमणीयता सिद्ध करता है (\cref{ch:b1:arith}), विल्सन
की प्रमेय में युग्मन को शक्ति देता है, और रैखिक बीजगणित में ``\omterm{def:b1:counting:card}{परिमित} विमीय
समष्टि का अंतःरूपांतरण \omterm{def:b1:logic:inj}{एकैकी} है यदि और केवल यदि वह \omterm{def:b1:logic:inj}{आच्छादक} है'' के रूप में
लौटता है (\cref{ch:b1:findim})।
\end{example}

\section{गणना के नियम}

\begin{proposition}[योग और गुणन नियम]\label{prop:b1:counting:rules}
मान लीजिए $E, F$ \omterm{def:b1:counting:card}{परिमित समुच्चय} हैं।
\begin{enumerate}
  \item यदि $E \cap F = \emptyset$, तो $\abs{E \cup F} = \abs{E} + \abs{F}$;
  और अधिक व्यापक रूप से, $E$ के टुकड़ों $E_1, \dots, E_k$ में \omterm{thm:b1:logic:partition}{विभाजन} के लिए
  $\abs{E} = \sum_i \abs{E_i}$।
  \item सामान्य स्थिति में $\abs{E \cup F} = \abs{E} + \abs{F} - \abs{E \cap
  F}$।
  \item $\abs{E \times F} = \abs{E} \times \abs{F}$.
  \item $E$ से $F$ में जाने वाले सभी प्रतिचित्रणों का \omterm{def:b1:logic:sets}{समुच्चय} $F^E$
  $\abs{F^E} = \abs{F}^{\abs{E}}$ को संतुष्ट करता है।
  \item $\abs{\mathcal{P}(E)} = 2^{\abs{E}}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) गणनाओं को जोड़ दीजिए: यदि बिना पुनरावृत्ति के $E = \{x_1, \dots, x_m\}$
और $F = \{y_1, \dots, y_n\}$, तो $x_1, \dots, x_m, y_1, \dots, y_n$ $E \cup
F$ की बिना पुनरावृत्ति गणना कर देता है (असंयुक्तता से)। आगमन इसे $k$ टुकड़ों
तक बढ़ा देता है।

(2) $E \cup F$ $E$ और $F \setminus E$ का असंयुक्त सम्मिलन है, तथा $F$ $F
\cap E$ और $F \setminus E$ का असंयुक्त सम्मिलन है; अतः $\abs{E \cup F} =
\abs{E} + \abs{F \setminus E} = \abs{E} + \abs{F} - \abs{E \cap F}$।

(3) $E \times F$ $x \in E$ पर, समुच्चयों $\{x\} \times F$ का असंयुक्त
सम्मिलन है, जिनमें से प्रत्येक का \omterm{def:b1:counting:card}{गणनांक} $\abs{F}$ है; (1) लगाइए।

(4) $E = \{x_1, \dots, x_m\}$ से $F$ में \omterm{def:b1:logic:map}{प्रतिचित्रण} ठीक $m$-उपक्रम
$(f(x_1), \dots, f(x_m)) \in F^m$ का चयन है; यह अनुरूपता एक एकैकी आच्छादन
है, और (3) तथा आगमन से $\abs{F^m} = \abs{F}^m$।

(5) $E$ के उपसमुच्चय प्रतिचित्रणों $E \to \{0, 1\}$ के साथ एकैकी आच्छादन में
हैं ($A$ को उसके सूचक फलन पर भेजिए); (4) लगाइए।
\end{proof}

\begin{example}[पूरक द्वारा गणना]\label{ex:b1:counting:complement}
$4$ अंकों वाले कितने पिन कोड (अंक $0$--$9$, क्रम महत्त्व रखता है,
पुनरावृत्ति की अनुमति है) में \emph{कम से कम एक} अंक दोहराया जाता है? उन्हें
सीधे गिनने का अर्थ है ``ठीक एक जोड़ा, दो जोड़े, एक तिकड़ी, एक चौकड़ी'' ---
पाँच अतिव्यापी विन्यासों को साधना। इसके बदले पूरक गिनिए: सभी कोड $10^4 =
10\,000$ हैं (गुणन नियम), और चार भिन्न अंकों वाले कोड $10 \times 9 \times 8
\times 7 = 5\,040$ ($4$-विन्यास), अतः उत्तर है
\[
10^4 - 10 \cdot 9 \cdot 8 \cdot 7 = 10\,000 - 5\,040 = 4\,960 .
\]
लगभग आधे पिन कोडों में कोई अंक दोहराया जाता है। सार: जब भी गणना ``कम से कम''
या ``सभी नहीं'' के रूप में कही गई हो, पहले पूरक आज़माइए --- योग नियम की
गारंटी है कि $\abs{A} = \abs{E} - \abs{\overline A}$, और पूरक प्रायः एक ही
स्वच्छ विन्यास होता है।
\end{example}

\begin{example}[जालक पथ]\label{ex:b1:counting:paths}
जालक के कोने $(0,0)$ से कोने $(4, 3)$ तक के लघुतम पथ गिनिए, जहाँ हर बार केवल
एक कदम दाएँ (R) या एक कदम ऊपर (U) चला जा सकता है। ऐसा प्रत्येक पथ ठीक $7$
कदम लेता है, जिनमें $4$ R और $3$ U हैं; विलोमतः, अक्षरों R, U का चार R वाला
$7$ लंबाई का कोई भी शब्द ठीक एक पथ का वर्णन करता है। अतः पथ R के स्थानों के
चयनों के साथ एकैकी आच्छादन में हैं:
\[
\binom{7}{4} = 35 .
\]
सार \emph{कूटन} में है: जैसे ही हर पथ को एक शब्द में, अर्थात् स्थानों के
उपसमुच्चय में, अनूदित किया गया, गणना तुच्छ हो गई --- इस सूत्रवाक्य का एक और
उदाहरण कि सही गणना छिपा हुआ एकैकी आच्छादन है (\cref{met:b1:counting:which})।
\end{example}

\section{सूचियाँ, क्रमचय, उपसमुच्चय}

\begin{definition}[विन्यास, क्रमचय, संचय]\label{def:b1:counting:objects}
मान लीजिए $E$ एक \omterm{def:b1:logic:sets}{समुच्चय} है, जहाँ $\abs{E} = n$, और $0 \leq k \leq n$।
\begin{itemize}
  \item $E$ का \emph{$k$-विन्यास} $E$ के अवयवों का \omterm{def:b1:logic:inj}{एकैकी} $k$-उपक्रम है (बिना
  पुनरावृत्ति का क्रमित चयन);
  \item $E$ का \emph{क्रमचय}\index{क्रमचय} $E$ से स्वयं उसी पर एकैकी आच्छादन
  है --- समतुल्य रूप से, एक $n$-विन्यास;
  \item \emph{$k$-संचय} $E$ का $k$ अवयवों वाला उपसमुच्चय है (बिना
  पुनरावृत्ति का अक्रमित चयन)। उनकी संख्या $\binom{n}{k}$ लिखी जाती है, जिसे
  ``$n$ में से $k$'' पढ़ते हैं\index{द्विपद गुणांक}।
\end{itemize}
\end{definition}

\begin{theorem}[तीनों गणनाएँ]\label{thm:b1:counting:counts}
$n = \abs{E}$ और $0 \leq k \leq n$ के साथ:
\begin{enumerate}
  \item $E$ के $k$-विन्यासों की संख्या $n (n-1) \cdots (n-k+1) =
  \dfrac{n!}{(n-k)!}$ है;
  \item $E$ के क्रमचयों की संख्या $n!$ है;
  \item $\dbinom{n}{k} = \dfrac{n!}{k!\,(n-k)!}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) पहला निर्देशांक चुनिए ($n$ तरीके), फिर दूसरा ($n - 1$ शेष चयन), \dots,
फिर $k$-वाँ ($n - k + 1$ चयन)। औपचारिक रूप से $k$ पर आगमन कीजिए। $k = 1$ के
लिए $n$ एक-पद वाले \omterm{def:b1:logic:inj}{एकैकी} उपक्रम हैं। गणना को $k - 1$ के लिए मान लीजिए।
प्रत्येक $k$-विन्यास $(x_1, \dots, x_k)$ ठीक एक $(k-1)$-विन्यास --- उसके
काटे हुए रूप $(x_1, \dots, x_{k-1})$ --- से, $\{x_1, \dots, x_{k-1}\}$ के
बाहर एक अंतिम निर्देशांक जोड़कर प्राप्त होता है, जिसके लिए ठीक $n - (k - 1)$
मान उपलब्ध हैं। इस प्रकार $k$-विन्यास, कटाई के द्वारा, $(k-1)$-विन्यासों से
सूचीबद्ध, $n - k + 1$ आकार वाले वर्गों में विभाजित हो जाते हैं, और योग नियम
देता है
\[
\frac{n!}{(n-k+1)!}\;(n - k + 1) = \frac{n!}{(n-k)!} .
\]

(2) $k = n$ के साथ (1) ही है।

(3) प्रत्येक $k$-उपसमुच्चय $k!$ भिन्न $k$-विन्यासों में क्रमित होता है, और
प्रत्येक $k$-विन्यास ठीक एक उपसमुच्चय से उत्पन्न होता है: अतः
$\frac{n!}{(n-k)!} = \binom nk \cdot k!$।
\end{proof}

\begin{example}[गोल मेज़: सममिति से भाग देना]\label{ex:b1:counting:roundtable}
$n$ अतिथि गोल मेज़ के चारों ओर कितने प्रकार से बैठ सकते हैं, जबकि दो बैठकें
तब समान मानी जाती हैं जब हर अतिथि के बाएँ और दाएँ वही पड़ोसी हों --- अर्थात्
घूर्णन तक? प्रत्येक वृत्तीय बैठक ठीक $n$ रैखिक बैठकों के अनुरूप है (वृत्त को
$n$ स्थानों में से किसी पर काटिए), अतः $n!$ रैखिक क्रम $n$ के समूहों में
सिमट जाते हैं:
\[
\frac{n!}{n} = (n-1)! \quad\text{वृत्तीय बैठकें।}
\]
समतुल्य रूप से: एक विशिष्ट अतिथि को कहीं भी बिठा दीजिए (जिससे घूर्णन की
स्वतंत्रता समाप्त हो जाए), फिर शेष $n - 1$ अतिथियों को दक्षिणावर्त क्रम में
रखिए। $n = 6$ के लिए: $120$ मेज़ें। दोनों हल अधिगणना के दो मानक उपाय दिखाते
हैं: पुनरावृत्तियों की ठीक संख्या से भाग दीजिए, या एक वस्तु को कील ठोंककर
\emph{सममिति भंग} कर दीजिए। दोनों के लिए आवश्यक है कि पुनरावृत्ति-समूह का
आकार हर विन्यास के लिए एक ही हो --- यही ऊपर $\binom nk =
\frac{n!}{k!\,(n-k)!}$ के सूत्र की उपपत्ति ने भी प्रयोग किया था, जहाँ $n$ के
स्थान पर $k!$ था।
\end{example}

\begin{example}[एक प्रतिबंध जोड़ना]\label{ex:b1:counting:constraint}
गोल मेज़ को आगे बढ़ाते हैं: $n \geq 3$ अतिथियों की $(n-1)!$ मेज़ों में कितनी
दो दिए हुए अतिथियों $A$ और $B$ को \emph{अलग} बिठाती हैं (सटा हुआ नहीं)? पूरक
गिनिए। वे मेज़ें जिनमें $A$ और $B$ साथ बैठते हैं: उन्हें एक ही खंड में चिपका
दीजिए --- मेज़ के चारों ओर $n - 1$ वस्तुएँ, अर्थात् $(n-2)!$ वृत्तीय विन्यास
--- फिर युग्म को उसके खंड के भीतर क्रमित कीजिए ($2$ तरीके): $2\,(n-2)!$ सटी
हुई मेज़ें। अतः
\[
(n-1)! - 2\,(n-2)! = (n-2)!\,\bigl((n - 1) - 2\bigr)
= (n-3)\,(n-2)!
\]
मेज़ें उन्हें अलग रखती हैं। जाँच: $n = 3$ से $0$ मिलता है (त्रिभुज के चारों
ओर हर कोई हर किसी को छूता है) और $n = 4$ से $2$, जिसे हाथ से आसानी से
सूचीबद्ध किया जा सकता है। चिपकाने की युक्ति --- बाध्य खंड को एक वस्तु मानिए,
फिर उसके भीतरी विन्यास गिनिए --- सटान-प्रतिबंधों का, रैखिक हो या वृत्तीय,
मानक उपाय है।
\end{example}

\begin{proposition}[मूल सर्वसमिकाएँ]\label{prop:b1:counting:identities}
$0 \leq k \leq n$ के लिए:
\[
\binom{n}{k} = \binom{n}{n-k},
\qquad
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}
\quad (1 \leq k \leq n-1),
\qquad
\sum_{k=0}^{n} \binom{n}{k} = 2^n .
\]
\end{proposition}

\begin{proof}
पहली सर्वसमिका: $A \mapsto E \setminus A$ $k$-उपसमुच्चयों और
$(n-k)$-उपसमुच्चयों के बीच एकैकी आच्छादन है। पास्कल का नियम\index{पास्कल का
नियम}: एक अवयव $a \in E$ नियत कीजिए; $k$-उपसमुच्चय उनमें बँट जाते हैं जो $a$
को समेटते हैं (शेष $k - 1$ चुनिए: $\binom{n-1}{k-1}$) और जो $a$ से बचते हैं
($\binom{n-1}{k}$)। तीसरी सर्वसमिका: दोनों पक्ष $E$ के सभी उपसमुच्चय गिनते
हैं, बाईं ओर आकार के अनुसार बाँटकर (\cref{prop:b1:counting:rules}~(1) और
(5))।
\end{proof}

\begin{theorem}[द्विपद प्रमेय]\label{thm:b1:counting:binomial}
किसी क्रमविनिमेय वलय (मान लीजिए $\R$ या $\C$) के सभी $a, b$ और $n \in \N$ के
लिए:
\[
(a + b)^n = \sum_{k=0}^{n} \binom{n}{k} a^k b^{\,n-k} .
\]
\end{theorem}

\begin{proof}
$(a+b)(a+b)\cdots(a+b)$ का वितरण नियम से प्रसार करने पर, प्रत्येक गुणनखंड
में $a$ या $b$ के हर चयन के लिए एक पद मिलता है: पद $a^k b^{n-k}$ उतनी बार
आता है जितने प्रकार से $n$ गुणनखंडों में से $k$ $a$ का योगदान करने के लिए
चुने जा सकते हैं --- अर्थात् $\binom nk$ बार। (वैकल्पिक रूप से: पास्कल के
नियम का प्रयोग करते हुए $n$ पर आगमन कीजिए।)
\end{proof}

\begin{example}\label{ex:b1:counting:binomial}
दो चिरपरिचित विशेषीकरण: $a = b = 1$ से $\sum_k \binom nk = 2^n$ मिलता है; $a
= -1$, $b = 1$ से $n \geq 1$ के लिए $\sum_{k} (-1)^k \binom nk = 0$ मिलता
है: किसी अरिक्त \omterm{def:b1:logic:sets}{समुच्चय} के उपसमुच्चयों में ठीक आधे सम \omterm{def:b1:counting:card}{गणनांक} वाले हैं।
\end{example}

\begin{example}[एक सर्वसमिका, दो उपपत्तियाँ]\label{ex:b1:counting:threen}
द्विपद प्रमेय का विशेषीकरण $a = 2$, $b = 1$ यह कहता है
\[
\sum_{k=0}^{n} \binom nk\,2^k = 3^n .
\]
यही सर्वसमिका बिना किसी बीजगणित के भी मिलती है। दायाँ पक्ष वर्णमाला $\{0, 1,
2\}$ पर $n$ लंबाई के शब्द गिनता है (गुणन नियम)। प्रत्येक शब्द को अशून्य
अक्षर वाले स्थानों के \omterm{def:b1:logic:sets}{समुच्चय} $K$ के अनुसार वर्गीकृत कीजिए: $\abs K = k$
वाला $K$ चुनने में $\binom nk$ लगता है, फिर $K$ का प्रत्येक स्थान स्वतंत्र
रूप से $1$ या $2$ धारण करता है: $2^k$ तरीके। $k$ पर योग नियम बायाँ पक्ष दे
देता है। सहमति के आनंद से आगे, दोनों उपपत्तियों के गुण भिन्न हैं: बीजीय
उपपत्ति $a$ के किसी भी मान तक सामान्यीकृत होती है, जबकि संयोजनात्मक उपपत्ति
सूत्र को \emph{समझाती} है और ऐसे प्रतिबंधों के अनुकूल ढल जाती है (मान लीजिए
अंतिम स्थान पर अक्षर $2$ निषिद्ध हो) जिन्हें कोई प्रतिस्थापन नहीं पकड़ पाता।
दोनों तकनीकों को सक्रिय रखना ही वह व्यावहारिक कौशल है जिसका अभ्यास यह अध्याय
कराता है।
\end{example}

\begin{method}[कौन-सी गणना लागू होती है?]\label{met:b1:counting:which}
संगणना से पहले चयन के विषय में दो प्रश्नों के उत्तर दीजिए: क्या \emph{क्रम}
महत्त्व रखता है, और क्या \emph{पुनरावृत्ति} की अनुमति है?
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{l|c|c}
 & क्रम महत्त्व रखता है & क्रम महत्त्व नहीं रखता \\ \hline पुनरावृत्ति नहीं
 & $\dfrac{n!}{(n-k)!}$ & $\dbinom{n}{k}$ \\[6pt] पुनरावृत्ति की अनुमति &
 $n^k$ & (\cref{exo:b1:counting:10}) \\
\end{tabular}
\end{center}
फिर ऐसा एकैकी आच्छादन या \omterm{thm:b1:logic:partition}{विभाजन} खोजिए जो समस्या को इन आदर्श गणनाओं तक ले आए;
सही गणना छिपा हुआ एकैकी आच्छादन है।
\end{method}

\begin{remark}[गणना में सामान्य भूलें]\label{rem:b1:counting:pitfalls}
\begin{enumerate}
  \item \emph{असंयुक्त न होने वाली स्थितियों को जोड़ना।} योग नियम के लिए
  \omterm{thm:b1:logic:partition}{विभाजन} चाहिए; यदि कोई विन्यास एक साथ दो स्थितियों को संतुष्ट कर सकता है,
  तो वह दो बार गिना जाता है --- उपाय है समावेशन--अपवर्जन
  (\cref{thm:b1:counting:inclexcl}) या और सूक्ष्म स्थिति-विभाजन।
  \item \emph{क्रमित बनाम अक्रमित।} ``दो लोगों की समिति'' चुनना $\binom n2$
  है, $n(n-1)$ नहीं: \emph{संगणना से पहले} तय कीजिए कि चयन क्रम धारण करता है
  या नहीं, और यदि क्रमित गणना आसान हो तो अंत में क्रमों की संख्या से भाग
  दीजिए --- पर तभी, जब प्रत्येक अक्रमित वस्तु \emph{समान} संख्या में क्रमित
  वस्तुओं से उत्पन्न होती हो।
  \item \emph{बहु-चरणीय चयन जो स्वतंत्र नहीं हैं।} गुणन नियम के लिए आवश्यक
  है कि हर चरण पर विकल्पों की संख्या पिछले चयनों से स्वतंत्र हो। ``एक कप्तान
  चुनिए, फिर उससे भिन्न एक उपकप्तान'' ठीक है ($n(n-1)$); ``ऐसे दो खिलाड़ी
  चुनिए जिनकी आपस में बनती हो'' दो-चरणीय गुणन है ही नहीं।
  \item \emph{रचना से ही दुहरी गणना।} प्रत्येक वस्तु को दो बार बनाना ---
  उदाहरणार्थ \emph{कम से कम} एक इक्के वाले हाथ (एक इक्का चुनिए) $\times$
  ($4$ और पत्ते चुनिए) के रूप में गिनना --- दो इक्कों वाले हाथों की अधिगणना
  कर देता है। ``कम से कम'' प्रायः सदा पूरक की माँग करता है
  (\cref{ex:b1:counting:complement})।
\end{enumerate}
\end{remark}

\begin{example}[ताश जैसी गणना]\label{ex:b1:counting:poker}
$52$ पत्तों की गड्डी में से $5$ पत्तों के हाथों की संख्या $\binom{52}{5} =
2\,598\,960$ है। ठीक एक इक्के वाले हाथ: इक्का चुनिए ($4$ तरीके), फिर $48$
गैर-इक्कों में से $4$ पत्ते: $4 \binom{48}{4} = 778\,320$। गुणन नियम इसलिए
लागू होता है कि चयन स्वतंत्र चरणों में बँट जाता है।
\end{example}

\begin{method}[दुहरी गणना]\label{met:b1:counting:doublecount}
दो गणना-व्यंजकों के बीच सर्वसमिका सिद्ध करने के लिए ऐसा एक \omterm{def:b1:counting:card}{परिमित समुच्चय}
खोजिए जिसे दोनों पक्ष गिनते हों --- प्रायः \emph{युग्मों} का \omterm{def:b1:logic:sets}{समुच्चय} --- और
उसका \omterm{def:b1:counting:card}{गणनांक} दो भिन्न क्रमों में आँकिए। आदिरूप है \emph{हस्तमिलन प्रमेयिका}:
किसी समारोह में (व्यक्ति, मिलाया गया हाथ) युग्म गिनिए। लोगों पर योग करने से
$\sum_p d_p$ मिलता है (प्रत्येक व्यक्ति $p$ के हस्तमिलनों की संख्या);
हस्तमिलनों पर योग करने से हस्तमिलनों की संख्या का दुगुना (हर हस्तमिलन में दो
व्यक्ति होते हैं)। अतः $\sum_p d_p$ सम है --- अर्थात् विषम बार हाथ मिलाने
वाले लोगों की संख्या सदा सम होती है, जो बिना किसी सूत्र के प्राप्त एक अतुच्छ
निष्कर्ष है। यही इंजन \cref{exo:b1:counting:12} को और नीचे दी गई सप्ताहांत
समस्या के कई प्रश्नों को चलाता है।
\end{method}

\begin{example}[औसत उपसमुच्चय]\label{ex:b1:counting:averagesubset}
$n$ अवयवों वाले \omterm{def:b1:logic:sets}{समुच्चय} $E$ के उपसमुच्चय का औसत \omterm{def:b1:counting:card}{गणनांक} क्या है, जबकि सभी
$2^n$ उपसमुच्चय समसंभाव्य हैं? $a \in A$ वाले युग्मों $(A, a)$ की दुहरी गणना
कीजिए: उपसमुच्चयों पर योग करने से $\sum_A \abs A$ मिलता है, जो अभीष्ट कुल
है; अवयवों पर योग करने से $n \cdot 2^{n-1}$ मिलता है ($n$ अवयवों में से
प्रत्येक ठीक आधे उपसमुच्चयों में है --- $a$ वाले प्रत्येक $A$ को $A
\setminus \{a\}$ के साथ युग्मित कीजिए)। अतः
\[
\frac{1}{2^n}\sum_{A \subseteq E} \abs A
= \frac{n\,2^{n-1}}{2^n} = \frac n2 :
\]
उपसमुच्चय औसतन आधे भरे होते हैं --- जैसा सममिति $A \leftrightarrow \overline
A$ (जो आकारों $k$ और $n - k$ को युग्मित करती है) भी बताती है। दो उपपत्तियाँ,
एक उत्तर, और दोनों \cref{exo:b1:counting:5} की सीधी संगणना $\sum_k k\binom
nk$ से बच जाती हैं: सुचयनित युग्मन प्रायः सर्वसमिका की जगह ले लेता है।
\end{example}

\section{समावेशन--अपवर्जन}

\begin{theorem}[समावेशन--अपवर्जन]\label{thm:b1:counting:inclexcl}
\omterm{def:b1:counting:card}{परिमित} समुच्चयों $A_1, \dots, A_p$ के लिए:
\[
\Bigl|\, \bigcup_{i=1}^{p} A_i \,\Bigr|
= \sum_{\emptyset \neq I \subseteq \intint{1}{p}}
(-1)^{\abs{I}+1} \Bigl|\, \bigcap_{i \in I} A_i \,\Bigr| .
\]
$p = 3$ के लिए: $\abs{A \cup B \cup C} = \abs A + \abs B + \abs C - \abs{A
\cap B} - \abs{A \cap C} - \abs{B \cap C} + \abs{A \cap B \cap C}$।
\end{theorem}

\begin{proof}
सम्मिलन का एक अवयव $x$ नियत कीजिए और दाएँ पक्ष में उसका योगदान गिनिए। मान
लीजिए $J = \{i : x \in A_i\}$, जिसका \omterm{def:b1:counting:card}{गणनांक} $m \geq 1$ है। अवयव $x$
$\abs{\bigcap_{i \in I} A_i}$ में ठीक तब एक बार गिना जाता है जब $\emptyset
\neq I \subseteq J$, और उसका चिह्न $(-1)^{\abs I + 1}$ होता है; उसका कुल
योगदान है
\[
\sum_{k=1}^{m} \binom{m}{k} (-1)^{k+1}
= 1 - \sum_{k=0}^{m} \binom mk (-1)^k = 1 - 0 = 1
\]
\cref{ex:b1:counting:binomial} से। अतः सम्मिलन का प्रत्येक अवयव ठीक एक बार
गिना जाता है।
\end{proof}

\begin{example}[सहअभाज्य पूर्णांकों की गणना]\label{ex:b1:counting:coprime}
$\intint1{120}$ के कितने पूर्णांक $120 = 2^3 \times 3 \times 5$ के सहअभाज्य
हैं? कोई पूर्णांक $120$ के साथ गुणनखंड ठीक तब साझा करता है जब वह $2$, $3$ या
$5$ से विभाज्य हो, अतः $A_2 \cup A_3 \cup A_5$ का पूरक गिनिए, जहाँ $A_d$ $d$
के गुणजों को इकट्ठा करता है। $\intint1{120}$ के भीतर, जब भी $d$ $120$ को
विभाजित करता है, $d$ के गुणज $120/d$ हैं --- किसी फ़र्श फलन की आवश्यकता नहीं
--- और $A_2 \cap A_3 = A_6$, इत्यादि। समावेशन--अपवर्जन:
\[
\abs{A_2 \cup A_3 \cup A_5}
= 60 + 40 + 24 - 20 - 12 - 8 + 4 = 88 ,
\]
अतः $120 - 88 = 32$ पूर्णांक $120$ के सहअभाज्य हैं। संगणना को गुणनफल के रूप
में पुनर्समूहित करना शिक्षाप्रद है:
\[
120 - 88 = 120\Bigl(1 - \frac12\Bigr)\Bigl(1 -
\frac13\Bigr)\Bigl(1 - \frac15\Bigr) = 120 \cdot \frac12 \cdot
\frac23 \cdot \frac45 = 32 :
\]
तीनों कोष्ठकों का प्रसार करने पर ठीक समावेशन--अपवर्जन के आठ चिह्नित पद मिलते
हैं, $\{2, 3, 5\}$ के प्रत्येक उपसमुच्चय के लिए एक। यह गुणनफल-रूप ऑयलर का
टोशंट फलन परिभाषित करता है, जिसकी अंकगणितीय भूमिका \cref{ch:b1:arith} की
सर्वांगसमताओं के साथ प्रकट होती है और स्नातक वर्ष 2 के खंड में विकसित की गई
है।
\end{example}

\begin{example}[व्यत्यय]\label{ex:b1:counting:derangement}
\emph{व्यत्यय} ऐसा \omterm{def:b1:counting:objects}{क्रमचय} है जिसका कोई अचल बिंदु न हो। मान लीजिए $A_i$ $i$
को अचल रखने वाले $\intint{1}{n}$ के क्रमचयों का \omterm{def:b1:logic:sets}{समुच्चय} है; तब
$\abs{\bigcap_{i \in I} A_i} = (n - \abs I)!$, और समावेशन--अपवर्जन कम से कम
एक अचल बिंदु वाले \omterm{def:b1:counting:objects}{क्रमचय} गिन देता है; व्यत्ययों की संख्या है
\[
D_n = n! \sum_{k=0}^{n} \frac{(-1)^k}{k!} .
\]
चूँकि $\sum (-1)^k / k! \to \eu^{-1}$ (\cref{ch:b1:series} देखिए), $n$ जो भी
हो, सभी क्रमचयों में लगभग $37\%$ व्यत्यय होते हैं।
\end{example}

\begin{remark}[यह अध्याय कहाँ काम आता है]\label{rem:b1:counting:whereused}
\omterm{def:b1:counting:objects}{द्विपद गुणांक} इस अध्याय की सबसे अधिक पुनःप्रयुक्त वस्तुएँ हैं: वे
\cref{ch:b1:poly} में द्विपद प्रमेय ($(X + a)^n$ का प्रसार),
\cref{ch:b1:derivative} में गुणनफल के $n$-वें अवकलज का लाइब्निज़ सूत्र, और
\cref{ch:b1:taylor} में टेलर प्रसारों के गुणांक चलाते हैं। \omterm{def:b1:counting:objects}{क्रमचय}
\cref{ch:b1:structures} में एक समूह के रूप में लौटते हैं --- व्युत्क्रमण
गिनकर बनाए गए चिह्न के साथ --- और वही चिह्न \cref{ch:b1:det} में सारणिक
परिभाषित करता है। समावेशन--अपवर्जन और गणना के नियम विविक्त प्रायिकता की
\omterm{def:b1:counting:card}{परिमित} रीढ़ हैं, जिसे स्नातक वर्ष 2 के खंड में विकसित किया गया है;
\cref{ex:b1:counting:derangement} की व्यत्यय संख्याओं का गहन अध्ययन नीचे दी
गई सप्ताहांत समस्या में है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:counting:1}
किसी पंजीकरण पट्टिका पर दो अक्षर (A--Z), फिर तीन अंक, फिर दो अक्षर होते हैं।
कितनी पट्टिकाएँ संभव हैं? और कितनी ऐसी जिनमें चारों अक्षरों में कोई अक्षर
दोहराया न गया हो?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:2}
शब्द $\mathrm{ORANGE}$ के कितने वर्णविपर्यय (अक्षरों के पुनर्विन्यास, सार्थक
हों या न हों) हैं? और $\mathrm{BANANA}$ के?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:3}
$7$ स्त्रियों और $5$ पुरुषों में से $4$ लोगों की एक समिति चुनी जाती है।
कितनी समितियाँ: कुल मिलाकर? ठीक $2$ स्त्रियों वाली? कम से कम एक पुरुष वाली?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:4}
सिद्ध कीजिए कि $13$ लोगों के किसी भी समूह में दो का जन्म-माह एक ही होता है;
और यह कि $\intint{1}{2n}$ में से चुने गए किन्हीं $n + 1$ पूर्णांकों में दो
क्रमागत होते हैं। \emph{(दोनों बार कबूतरखाना: डिब्बों के नाम बताइए।)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:5}
$\sum_{k=0}^{n} k \binom{n}{k}$ की संगणना कीजिए। \emph{संकेत: $(1 + x)^n$ का
अवकलन कीजिए, या $k \binom nk = n \binom{n-1}{k-1}$ का प्रयोग कीजिए (उसे
सिद्ध कीजिए)।}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:6}
$\intint{1}{k}$ से $\intint{1}{n}$ में कितने पूर्णतः वर्धमान \omterm{def:b1:logic:map}{प्रतिचित्रण}
हैं? उससे वर्धमान (जो पूर्णतः वर्धमान होना आवश्यक नहीं) प्रतिचित्रणों की
संख्या निकालिए। \emph{दूसरी गणना के लिए संकेत: $f$ वर्धमान $\mapsto$ $g(i) =
f(i) + i - 1$।}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:7}
(वांडरमोंड) $m$ और $n$ आकार के दो खंडों में बँटे \omterm{def:b1:logic:sets}{समुच्चय} के $k$-उपसमुच्चय
गिनकर सिद्ध कीजिए:
\[
\binom{m+n}{k} = \sum_{j=0}^{k} \binom{m}{j} \binom{n}{k-j} .
\]
$\sum_{j=0}^{n} \binom nj^2 = \binom{2n}{n}$ निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:8}
$\intint{1}{1000}$ के कितने पूर्णांक $2$ या $3$ या $5$ से विभाज्य हैं?
(समावेशन--अपवर्जन; $\lfloor 1000/6 \rfloor$ $6$ के गुणज गिनता है, इत्यादि।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:9}
$4$ अवयवों के \omterm{def:b1:logic:sets}{समुच्चय} से $2$ अवयवों के \omterm{def:b1:logic:sets}{समुच्चय} पर आच्छादनों की गणना कीजिए;
फिर $3$ अवयवों के \omterm{def:b1:logic:sets}{समुच्चय} पर। \emph{संकेत: छूट गए मानों पर समावेशन--अपवर्जन
से अनाच्छादक \omterm{def:b1:logic:map}{प्रतिचित्रण} गिनिए।}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:10}
(तारे और छड़ें) सिद्ध कीजिए कि $n$ वस्तुओं के, पुनरावृत्ति \emph{सहित} और
क्रम की उपेक्षा करते हुए, $k$-चयनों की संख्या --- समतुल्य रूप से, $x_1 +
\dots + x_n = k$ वाले $(x_1, \dots, x_n) \in \N^n$ की संख्या --- $\binom{n +
k - 1}{k}$ है। \emph{संकेत: किसी हल को $k$ तारों और $n - 1$ छड़ों की एक
पंक्ति के रूप में कूटित कीजिए।}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:counting:11}
$D_n$ के लिए \cref{ex:b1:counting:derangement} का सूत्र विस्तार से सिद्ध
कीजिए, और उससे $n! = \sum_{k=0}^{n} \binom{n}{k} D_{n-k}$ निकालिए (इस
सर्वसमिका को क्रमचयों का उनके अचल-बिंदु \omterm{def:b1:logic:sets}{समुच्चय} के अनुसार वर्गीकरण करके सीधे
भी सिद्ध कीजिए)।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:counting:12}
$n \in \N^*$ के लिए (उपसमुच्चय, चिह्नित अवयव) युग्मों की दुहरी गणना से सिद्ध
कीजिए:
\[
\sum_{k=1}^{n} k \binom{n}{k} = n\, 2^{n-1},
\qquad\text{फिर}\qquad
\sum_{k=1}^{n} k^2 \binom{n}{k} = n(n+1)\, 2^{n-2} .
\]
\emph{दूसरी के लिए: चिह्नित अवयवों के युग्म गिनिए, चाहे वे बराबर हों या
नहीं।}
\end{exercise}

\section{समस्या: व्यत्यय, अथवा ग़लत पते वाले पत्र}

\begin{problem}\label{pb:b1:counting:1}
कोई सचिव $n$ पते लिखे लिफ़ाफ़ों में $n$ पत्र यादृच्छिक रूप से रखता है: इसकी
क्या संभावना है कि \emph{किसी को भी} सही पत्र न मिले? यह चिरपरिचित प्रश्न
(मोंमोर, 1708) \cref{ex:b1:counting:derangement} की व्यत्यय संख्याओं $D_n$
तक ले जाता है। समावेशन--अपवर्जन का सूत्र तो केवल पहली चाल है: यह समस्या
$D_n$ की संगणना करने वाली पुनरावृत्तियाँ, सूत्र की दो और स्वतंत्र
उपपत्तियाँ, यह चौंकाने वाली प्रमेय कि $D_n$ $n!/\eu$ के निकटतम पूर्णांक है,
यादृच्छिक \omterm{def:b1:counting:objects}{क्रमचय} के अचल बिंदुओं का पूरा बंटन, और अनुक्रम $(D_n)$ का
कुतूहलजनक अंकगणित विकसित करती है।\index{व्यत्यय} आगे सर्वत्र $D_n$
$\intint1n$ के व्यत्ययों (अचल-बिंदु रहित क्रमचयों) की संख्या है, और परिपाटी
$D_0 = 1$ है (रिक्त \omterm{def:b1:counting:objects}{क्रमचय} का कोई अचल बिंदु नहीं है)।

\medskip \noindent\textbf{भाग I --- छोटी स्थितियाँ और अचल-बिंदु जनगणना।}
\begin{enumerate}
  \item $D_1, D_2, D_3$ की सीधे संगणना कीजिए, और $D_4$ की संगणना $\{1, 2, 3,
  4\}$ के व्यत्ययों को $\sigma(1)$ के मान के अनुसार समूहबद्ध करके सूचीबद्ध
  करते हुए कीजिए। (आपको $D_4 = 9$ मिलना चाहिए।)
  \item $0 \leq k \leq n$ के लिए दिखाइए कि $\intint1n$ के \emph{ठीक} $k$ अचल
  बिंदुओं वाले क्रमचयों की संख्या $P_k(n)$ $\binom nk D_{n-k}$ है।
  \item $n = 4$ के लिए जनगणना सत्यापित कीजिए: $P_0(4), \dots, P_4(4)$ की
  संगणना कीजिए और जाँचिए कि उनका योग $4! = 24$ है। चार पत्रों के लिए क्या
  अधिक संभावित है: कोई मेल नहीं, या ठीक एक मेल?
  \item $\sigma(i) = i$ वाले युग्मों $(\sigma, i)$ की दुहरी गणना
  (\cref{met:b1:counting:doublecount}) से दिखाइए कि
        \[
        \sum_{\sigma} \abs{\mathrm{Fix}(\sigma)} = n! :
        \]
        अर्थात् $n \geq 1$ जो भी हो, यादृच्छिक \omterm{def:b1:counting:objects}{क्रमचय} में औसतन \emph{ठीक एक}
        अचल बिंदु होता है।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- दो पुनरावृत्तियाँ और सूत्र की दो नई
उपपत्तियाँ।}
\begin{enumerate}[resume]
  \item $n \geq 1$ के लिए संयोजनात्मक रूप से सिद्ध कीजिए:
        \[
        D_{n+1} = n\,(D_n + D_{n-1}) .
        \]
        ($\intint1{n+1}$ के व्यत्ययों $\sigma$ को पहले $j = \sigma(n+1)$ के
        अनुसार, फिर इस आधार पर वर्गीकृत कीजिए कि $\sigma(j) = n + 1$ है या
        नहीं; $\sigma(j) \neq n+1$ की स्थिति में $n + 1$ के \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब}
        की दिशा $j$ की ओर मोड़कर $\intint1n$ के व्यत्ययों के साथ एकैकी
        आच्छादन बनाइए।) पुनरावृत्ति को $D_6$ तक संख्यात्मक रूप से जाँचिए।
  \item $u_n = D_n - n D_{n-1}$ रखकर प्रश्न 5 से $u_{n+1} = -u_n$ निकालिए,
  और दूसरी पुनरावृत्ति पर पहुँचिए:
        \[
        D_n = n D_{n-1} + (-1)^n \qquad (n \geq 1).
        \]
  \item प्रश्न 6 से आगमन द्वारा \cref{ex:b1:counting:derangement} का सूत्र
  सिद्ध कीजिए,
        \[
        D_n = n! \sum_{k=0}^{n} \frac{(-1)^k}{k!},
        \]
        --- ऐसी उपपत्ति जो समावेशन--अपवर्जन से पूर्णतः स्वतंत्र है।
  \item (द्विपद प्रतिलोमन) मान लीजिए $(a_n)$ और $(b_n)$ ऐसे दो अनुक्रम हैं
  कि सभी $n$ के लिए $a_n = \sum_{k=0}^n \binom nk b_k$। सिद्ध कीजिए कि
        \[
        b_n = \sum_{k=0}^{n} (-1)^{n-k} \binom nk a_k
        \qquad (n \in \N).
        \]
        (पहले \emph{त्रिपद पुनर्लेखन} $\binom nk \binom kj = \binom nj
        \binom{n-j}{k-j}$ स्थापित कीजिए, फिर \cref{ex:b1:counting:binomial}
        के एकांतरित पंक्ति-योग का प्रयोग कीजिए।)
  \item \cref{exo:b1:counting:11} की सर्वसमिका $n! = \sum_k \binom nk
  D_{n-k}$ पर प्रश्न 8 लगाकर $D_n$ के सूत्र की \emph{तीसरी} उपपत्ति प्राप्त
  कीजिए।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- $n!/\eu$ का निकटतम पूर्णांक।} इस भाग
के लिए यह मान लीजिए --- सिद्धांत \cref{ch:b1:series} में खड़ा किया गया है
--- कि $\eu^{-1} = \lim_{n \to \infty} s_n$, जहाँ $s_n = \sum_{k=0}^{n}
\frac{(-1)^k}{k!}$, और प्रत्येक $n$ के लिए कठोर एकांतरित-श्रेणी परिबंध
$\abs{\eu^{-1} - s_n} < \frac1{(n+1)!}$ है।
\begin{enumerate}[resume]
  \item दिखाइए कि सभी $n \in \N$ के लिए $\bigl| D_n - n!/\eu \bigr| <
  \frac1{n+1}$।
  \item मुख्य प्रमेय निकालिए: \emph{प्रत्येक $n \geq 1$ के लिए $D_n$
  $n!/\eu$ का निकटतम पूर्णांक है}। तर्क को $n \geq 1$ की आवश्यकता क्यों है?
  \item त्रुटि का चिह्न निर्धारित कीजिए: दिखाइए कि $D_n > n!/\eu$ ठीक तब जब
  $n$ सम हो। (एकांतरित श्रेणी का पहला उपेक्षित पद ढूँढ़िए।)
  \item प्रश्न 5 की पुनरावृत्ति से $D_7$ से $D_{10}$ तक की संगणना कीजिए, फिर
  $D_{10}$ को $10!/\eu$ के सामने जाँचिए ($10! = 3\,628\,800$, $\eu \approx
  2.718281828$)।
  \item (टोपी-वापसी प्रायिकता) मान लीजिए $p_n = D_n/n!$ वह प्रायिकता है कि
  समरूप यादृच्छिक \omterm{def:b1:counting:objects}{क्रमचय} व्यत्यय है। दिखाइए $\abs{p_n - \eu^{-1}} <
  \frac1{(n+1)!}$ और $p_6$ की पाँच दशमलव स्थानों तक संगणना कीजिए। टिप्पणी
  कीजिए: मोंमोर के प्रश्न का उत्तर $n$ से मूलतः स्वतंत्र क्यों है --- वह भी
  दर्जन भर पत्रों पर ही?
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- अचल बिंदुओं का बंटन।}
\begin{enumerate}[resume]
  \item $k \in \N$ नियत कीजिए। दिखाइए कि $\intint1n$ के ठीक $k$ अचल बिंदुओं
  वाले क्रमचयों का अनुपात यह संतुष्ट करता है
        \[
        \frac{P_k(n)}{n!} = \frac{s_{n-k}}{k!}
        \;\xrightarrow[n \to \infty]{}\; \frac{\eu^{-1}}{k!} .
        \]
        (ये सीमांत मान, जिनका योग $1$ है, प्राचल $1$ का \emph{प्वासों बंटन}
        बनाते हैं, जो स्नातक वर्ष 2 के खंड के प्रायिकता पाठ्यक्रम की एक
        केंद्रीय वस्तु है।)
  \item ऐसी त्रिकियों $(\sigma, i, j)$ की दुहरी गणना से, जिनमें $i \neq j$
  दोनों $\sigma$ द्वारा अचल रखे जाते हैं, दिखाइए कि $n \geq 2$ के लिए
  $\sum_\sigma \abs{\mathrm{Fix}(\sigma)}\, (\abs{\mathrm{Fix}(\sigma)} - 1)
  = n!$। प्रश्न 4 के साथ मिलाकर: $\abs{\mathrm{Fix}}^2$ का औसत $2$ है, अतः
  अचल बिंदुओं की संख्या का ``फैलाव'' (प्रसरण) $1$ के बराबर है --- फिर $n$ से
  स्वतंत्र, और फिर प्वासों नियम के अनुरूप।
  \item $n = 4, 5, 6$ के लिए कम से कम एक अचल बिंदु वाले क्रमचयों का अनुपात
  संगणित कीजिए (भिन्न के रूप में और चार दशमलव स्थानों तक), और $1 - \eu^{-1}
  \approx 0.6321$ से तुलना कीजिए।
  \item सीधे --- बिना किसी सीमा के --- दिखाइए कि $s_{n+2} - s_n =
  (-1)^{n+1}\bigl(\frac1{(n+1)!} - \frac1{(n+2)!}\bigr)$, और उससे निकालिए कि
  प्रश्न 14 की प्रायिकताएँ $p_n = s_n$ दोलन करती हैं: $p_0 > p_2 > p_4 >
  \dots$ और $p_1 < p_3 < p_5 < \dots$, जहाँ सम (क्रमशः विषम) मान उभयनिष्ठ
  सीमा $\eu^{-1}$ की ओर घटते (क्रमशः बढ़ते) हैं।
  \item (गुप्त उपहार) $n$ लोगों में से हर एक टोपी से एक नाम निकालता है; यदि
  किसी को अपना ही नाम निकल आए, तो \emph{पूरा} निष्कर्षण नए सिरे से दोहराया
  जाता है। इस मानक तथ्य का प्रयोग करते हुए कि प्रायिकता $p$ की घटना में औसतन
  $1/p$ प्रयास लगते हैं, आवश्यक पूर्ण निष्कर्षणों की औसत संख्या का आकलन
  कीजिए, और निष्कर्ष निकालिए कि यह प्रक्रिया औसतन लगभग $\eu \approx 2.72$
  निष्कर्षणों की पड़ती है, और वह भी $n$ से मूलतः स्वतंत्र रूप से।
\end{enumerate}

\medskip \noindent\textbf{भाग V --- $D_n$ का अंकगणित, और एक संश्लेषण।}
\begin{enumerate}[resume]
  \item प्रश्न 5 को परिष्कृत कीजिए: दिखाइए कि नियत $j \in \intint2n$ के लिए
  $\sigma(1) = j$ वाले $\intint1n$ के व्यत्ययों की संख्या ठीक $D_{n-1} +
  D_{n-2}$ है, और वह $j$ से स्वतंत्र है। उससे निकालिए कि प्रत्येक $n \geq 2$
  के लिए $n - 1$ $D_n$ को विभाजित करता है।
  \item सिद्ध कीजिए कि $D_n$ विषम है यदि और केवल यदि $n$ सम हो। (प्रश्न 6 की
  पुनरावृत्ति में $2$ के सापेक्ष काम कीजिए।)
  \item सिद्ध कीजिए कि $n \geq 1$ के लिए $D_n \equiv (-1)^n \pmod n$, और
  सर्वांगसमता को $D_{10}$ के अंतिम अंक पर जाँचिए।
  \item प्रश्न 6 से दिखाइए कि $n \geq 3$ के लिए $\dfrac{D_n}{D_{n-1}} = n +
  \dfrac{(-1)^n}{D_{n-1}}$, अतः क्रमागत व्यत्यय संख्याओं का अनुपात
  \emph{लगभग ठीक} $n$ है; एक वाक्य में समझाइए कि यह $D_n \approx n!/\eu$ के
  अनुरूप क्यों है।
  \item इस समस्या में ठीक कहाँ प्रयोग हुआ: (क) गुणन और योग नियम; (ख) दुहरी
  गणना; (ग) द्विपद प्रमेय; (घ) स्वीकृत एकांतरित-श्रेणी परिबंध? प्रत्येक के
  लिए एक वाक्य।
  \item संश्लेषण। $D_n$ के सूत्र की अब तीन उपपत्तियाँ हैं (समावेशन--अपवर्जन,
  पुनरावृत्ति के साथ आगमन, द्विपद प्रतिलोमन)। एक छोटे अनुच्छेद में तुलना
  कीजिए कि प्रत्येक उपपत्ति क्या \emph{समझाती} है: कौन-सी सबसे तेज़ संगणना
  करती है, कौन-सी अन्य अचल-बिंदु गणनाओं तक सामान्यीकृत होती है, और कौन-सी यह
  प्रकट करती है कि लिफ़ाफ़ों की समस्या में $\eu$ क्यों आ जाता है।
\end{enumerate}
\end{problem}
