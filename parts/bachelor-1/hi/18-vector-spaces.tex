\chapter{सदिश समष्टियाँ}\label{ch:b1:vspaces}

रैखिक बीजगणित यहीं से आरंभ होता है: \omterm{def:b1:vspaces:def}{सदिश समष्टि} के अभिगृहीत ठीक वही बात अलग
करके दिखाते हैं जो $\R^2$, $\R^3$, \omterm{def:b1:poly:def}{बहुपद} समष्टियों और फलन समष्टियों में
उभयनिष्ठ है --- उनमें जोड़ा जा सकता है, और मापित किया जा सकता है। सिद्धांत
दो अध्यायों में खड़ा होता है (\cref{ch:b1:findim} विमा जोड़ता है); और जो
भाषा वे रचते हैं --- \omterm{def:b1:vspaces:span}{जनित समष्टि}, \omterm{def:b1:vspaces:free}{स्वतंत्र कुल}, \omterm{def:b1:vspaces:free}{आधार}, \omterm{def:b1:vspaces:sum}{प्रत्यक्ष योग} --- वही
उनके बाद के हर अध्याय की रोज़ की रोटी है। सर्वत्र $K$ का अर्थ $\R$ या $\C$
है (\emph{अदिश})।

\section{परिभाषा और उदाहरण}

\begin{definition}[सदिश समष्टि]\label{def:b1:vspaces:def}
\emph{$K$-सदिश समष्टि}\index{सदिश समष्टि} एक ऐसा \omterm{def:b1:logic:sets}{समुच्चय} $E$ है जिस पर एक
योग हो जो $(E, +)$ को \omterm{def:b1:structures:group}{आबेली समूह} बनाता हो (शून्य को $0_E$ अथवा $0$ लिखा जाता
है), और एक अदिश गुणन $K \times E \to E$ हो, ऐसा कि सभी $\lambda, \mu \in K$
और $x, y \in E$ के लिए:
\[
\lambda(x + y) = \lambda x + \lambda y,\quad
(\lambda + \mu) x = \lambda x + \mu x,\quad
\lambda(\mu x) = (\lambda\mu) x,\quad
1\,x = x .
\]
परिणाम: $0\,x = 0_E$, $\lambda\,0_E = 0_E$, $(-1)x = -x$, और $\lambda x =
0_E \implies \lambda = 0$ अथवा $x = 0_E$ ($\lambda^{-1}$ से गुणा कीजिए)।
\end{definition}

\begin{proof}[परिणामों की उपपत्ति]
$0\,x = 0_E$ के लिए: $(0 + 0)x = 0x + 0x$ और $(0+0)x = 0x$ से, \omterm{def:b1:structures:group}{समूह} $(E, +)$
में $0x$ को काट दीजिए। $\lambda\,0_E$ के लिए: $\lambda(0_E + 0_E)$ पर वही
चाल। $(-1)x$ के लिए: $x$ जोड़िए,
\[
x + (-1)x = 1\,x + (-1)x = \bigl(1 + (-1)\bigr)x = 0\,x = 0_E ,
\]
अतः $(-1)x$ $x$ का योज्य प्रतिलोम है। अंत में यदि $\lambda \neq 0$ के साथ
$\lambda x = 0_E$: तो $\lambda^{-1}$ से गुणा कीजिए (अदिश एक \omterm{def:b1:structures:field}{क्षेत्र} बनाते
हैं) और दोनों अभिगृहीत $\lambda^{-1}(\lambda x) = (\lambda^{-1}\lambda)x =
1x = x$ का $\lambda^{-1}0_E = 0_E$ के साथ प्रयोग कीजिए: $x = 0_E$। छोटे होते
हुए भी ये चार नियम आगे के हर पृष्ठ पर चुपचाप काम आते हैं --- और अंतिम नियम
ठीक वहीं है जहाँ \omterm{def:b1:structures:field}{क्षेत्र} चाहिए: अदिश $\Z$ पर ``समष्टि'' $\Z/2\Z$ उसका
उल्लंघन कर देती, क्योंकि $2\,x = 0$।
\end{proof}

\begin{example}\label{ex:b1:vspaces:examples}
$K^n$ (निर्देशांकवार संक्रियाएँ); \omterm{def:b1:poly:def}{बहुपद} $K[X]$; किसी भी \omterm{def:b1:logic:sets}{समुच्चय} $A$ से $K$
में जाने वाले फलन $\mathcal{F}(A, K)$ (बिंदुवार संक्रियाएँ) --- जिनमें \omterm{def:b1:continuity:continuous}{संतत}
फलन, अनुक्रम $\mathcal{F}(\N, \R)$ आदि आ जाते हैं; और $\R$-सदिश समष्टि के
रूप में $\C$। हर स्थिति में अभिगृहीत $K$ के अभिगृहीतों से विरासत में मिलते
हैं।
\end{example}

\begin{definition}[उपसमष्टि]\label{def:b1:vspaces:subspace}
$F \subseteq E$ \emph{उपसमष्टि}\index{उपसमष्टि} तब कहलाती है जब $0_E \in F$
और $F$ योग तथा अदिश गुणन के अंतर्गत स्थायी हो --- अथवा तुल्य रूप से:
\[
F \neq \emptyset
\qquad\text{और}\qquad
\forall x, y \in F,\ \forall \lambda \in K,\quad
x + \lambda y \in F .
\]
उपसमष्टि स्वयं एक \omterm{def:b1:vspaces:def}{सदिश समष्टि} होती है। उपसमष्टियों का कोई भी सर्वनिष्ठ
उपसमष्टि होता है; संघ लगभग कभी नहीं होता (\cref{exo:b1:structures:6} जैसी ही
उपपत्ति)।
\end{definition}

\begin{example}\label{ex:b1:vspaces:subspaces}
$\mathcal{F}(\R, \R)$ में: \omterm{def:b1:continuity:continuous}{संतत} फलन, \omterm{def:b1:derivative:def}{अवकलनीय} फलन, $\leq n$ घात वाले \omterm{def:b1:poly:def}{बहुपद}
($K[X]$ के भीतर $K_n[X]$ लिखा जाता है), किसी समघातीय रैखिक अवकल समीकरण के हल
(\cref{thm:b1:diffeq:homogeneous2} ने ठीक यही कहा था)। प्रति-उदाहरण: $\{f :
f(0) = 1\}$ (शून्य नहीं है), ठीक $n$ घात वाले \omterm{def:b1:poly:def}{बहुपद} (योग के अंतर्गत स्थायी
नहीं)।
\end{example}

\begin{example}[उपसमष्टि है या नहीं: चार निर्णय, तर्क सहित]
\label{ex:b1:vspaces:subspaceornot}
वास्तविक अनुक्रमों की समष्टि में:
\begin{itemize}
  \item $\{u : u \text{ परिबद्ध}\}$ \omterm{def:b1:vspaces:subspace}{उपसमष्टि} \emph{है}: $0$ परिबद्ध है, और
  यदि $\abs{u_n} \leq M$, $\abs{v_n} \leq M'$, तो $\abs{u_n + \lambda v_n}
  \leq M + \abs\lambda M'$।
  \item $\{u : u_n \to 1\}$ \emph{नहीं} है: शून्य अनुक्रम अनुपस्थित है (और
  दो सदस्यों का योग $2$ की ओर जाता है)।
  \item $\{u : u \text{ एकदिष्ट}\}$ \emph{नहीं} है: $u_n = n$ और $v_n = -n +
  (-1)^n$ एकदिष्ट हैं, पर उनका योग $(-1)^n$ नहीं; यहाँ योग के अंतर्गत
  स्थायित्व वाला अभिगृहीत विफल होता है, यद्यपि \omterm{def:b1:logic:sets}{समुच्चय} में $0$ है और अपने
  सदस्यों के सभी अदिश गुणज भी।
  \item $\{u : u_{n+1} = u_n^2\}$ \emph{नहीं} है: उसमें $0$ है, पर $u$ के
  अशून्य सदस्य होते ही $2u$ बाहर निकल जाता है (व्यापक रूप से $2u_{n+1} \neq
  (2u_n)^2$) --- अरैखिकता वर्ग करने में है।
\end{itemize}
काम का क्रम सदा एक ही है: पहले $0$ की जाँच (सबसे सस्ती), फिर स्थायित्व ---
और खंडन के लिए एक स्पष्ट प्रति-उदाहरण की जोड़ी किसी भी मात्रा के संदेह से
बेहतर है।
\end{example}

\section{जनित समष्टि, योग, प्रत्यक्ष योग}

\begin{definition}[रैखिक संयोजन, जनित समष्टि]\label{def:b1:vspaces:span}
$E$ के सदिशों के कुल $(x_1, \dots, x_p)$ का \emph{रैखिक संयोजन} कोई भी
$\lambda_1 x_1 + \dots + \lambda_p x_p$ है ($\lambda_i \in K$)। उन सबका
\omterm{def:b1:logic:sets}{समुच्चय} \emph{जनित समष्टि}\index{जनित समष्टि} $\operatorname{Vect}(x_1,
\dots, x_p)$ है: यह एक \omterm{def:b1:vspaces:subspace}{उपसमष्टि} है, और उस कुल को समाहित करने वाली सबसे छोटी।
\end{definition}

\begin{proof}[दोनों कथनों की उपपत्ति]
स्थायित्व: दो रैखिक संयोजनों का योग $\sum\lambda_i x_i + \sum\mu_i x_i =
\sum(\lambda_i + \mu_i)x_i$ फिर से एक रैखिक संयोजन है, और अदिश गुणज
$\mu\sum\lambda_i x_i = \sum(\mu\lambda_i) x_i$ भी; शून्य संयोजन दिखाता है
कि $0$ उसमें है: अतः \omterm{def:b1:vspaces:span}{जनित समष्टि} एक \omterm{def:b1:vspaces:subspace}{उपसमष्टि} है। न्यूनतमता: मान लीजिए $H$
$x_1, \dots, x_p$ को समाहित करने वाली कोई \omterm{def:b1:vspaces:subspace}{उपसमष्टि} है। अदिश गुणन के अंतर्गत
स्थायित्व से प्रत्येक $\lambda_i x_i \in H$, और योग के अंतर्गत स्थायित्व से
उनका योग $H$ में है: अतः हर रैखिक संयोजन $H$ का सदस्य है, अर्थात्
$\operatorname{Vect}(x_1, \dots, x_p) \subseteq H$। इस प्रकार \omterm{def:b1:vspaces:span}{जनित समष्टि} उस
कुल को समाहित करने वाली हर \omterm{def:b1:vspaces:subspace}{उपसमष्टि} के भीतर है: वही सबसे छोटी है।
\end{proof}

\begin{definition}[योग, प्रत्यक्ष योग]\label{def:b1:vspaces:sum}
$E$ की उपसमष्टियों $F, G$ के लिए:
\[
F + G = \{\,u + v : u \in F,\ v \in G\,\}
\]
एक \omterm{def:b1:vspaces:subspace}{उपसमष्टि} है ($F \cup G$ को समाहित करने वाली सबसे छोटी)। यह योग
\emph{प्रत्यक्ष}\index{प्रत्यक्ष योग} तब कहलाता है और $F \oplus G$ लिखा जाता
है, जब $F + G$ का हर अवयव \emph{अद्वितीय} रूप से $u + v$ के रूप में विघटित
होता हो; तुल्य रूप से (नीचे देखिए) जब $F \cap G = \{0\}$ हो। और जब $E = F
\oplus G$ हो, तो वे उपसमष्टियाँ $E$ में \emph{पूरक}\index{पूरक उपसमष्टियाँ}
कहलाती हैं।
\end{definition}

\begin{example}[दो रेखाओं का योग]
\label{ex:b1:vspaces:sumoflines}
$\R^3$ में मान लीजिए $F = \operatorname{Vect}\bigl((1,0,1)\bigr)$ और $G =
\operatorname{Vect}\bigl((0,1,1)\bigr)$। उनका योग है
\[
F + G = \{\,a(1,0,1) + b(0,1,1)\,\}
= \{(a,\ b,\ a + b)\} = \{(x, y, z) : z = x + y\},
\]
अर्थात् मूल बिंदु से होकर जाने वाला वह समतल जिसमें दोनों रेखाएँ हैं। यह संघ
$F \cup G$ (केवल दो कटती हुई रेखाओं का आकार) से सचमुच बड़ा है: सदिश $(1, 1,
2) = (1,0,1) + (0,1,1)$ योग में है पर किसी भी रेखा पर नहीं। और $F \cap G =
\{0\}$ (उभयनिष्ठ सदिश के लिए $a(1,0,1) = b(0,1,1)$ चाहिए, जिसके पहले दो
\omterm{prop:b1:vspaces:coordinates}{निर्देशांक} $a = b = 0$ बाध्य कर देते हैं): अतः योग प्रत्यक्ष है, और $F
\oplus G$ ठीक वही समतल है।
\end{example}

\begin{proposition}\label{prop:b1:vspaces:directsum}
$F + G$ प्रत्यक्ष है तभी जब $F \cap G = \{0\}$।
\end{proposition}

\begin{proof}
यदि कोई $w \neq 0$ $F \cap G$ में हो: तो $w = w + 0 = 0 + w$ $w$ के दो विघटन
हैं। विलोमतः, यदि $u, u' \in F$, $v, v' \in G$ के साथ $u + v = u' + v'$, तो
$u - u' = v' - v$ $F \cap G = \{0\}$ का सदस्य है: अतः विघटन अद्वितीय हैं।
\end{proof}

\begin{method}[$E = F \oplus G$ सिद्ध करना]
\label{met:b1:vspaces:directsum}
दो बातें जाँचनी हैं, और हर एक की अपनी मानक पहली चाल है।
\begin{enumerate}
  \item \emph{तुच्छ सर्वनिष्ठ।} $x \in F \cap G$ लीजिए, दोनों
  सदस्यता-प्रतिबंध खोलकर लिखिए, और $x = 0$ तक दबा दीजिए। (चित्र देखकर कभी
  तर्क मत कीजिए: नीचे दी गई भूलें देखिए।)
  \item \emph{योग ही सब कुछ है।} कोई भी $x \in E$ लीजिए और विघटन $x = f + g$
  \emph{बनाकर दिखाइए} --- या तो लक्ष्य से $f$ का अनुमान लगाकर ($f$ को $F$ का
  परिभाषक गुण संतुष्ट करना ही है, जो प्रायः उसका सूत्र तय कर देता है), या
  फिर $F$ और $G$ के जनकों के सापेक्ष $x$ को व्यक्त करने वाला रैखिक निकाय हल
  करके।
\end{enumerate}
जब विघटन का सूत्र अनुमान से मिल जाए, तो अद्वितीयता चरण 1 से स्वतः आ जाती है;
और जब केवल अस्तित्व अस्पष्ट हो, तो सारा काम चरण 2 में है। नीचे के दोनों
उदाहरण यही विधि चलाते हैं: सम/विषम फलनों के लिए $f$ का सूत्र इच्छित
सर्वसमिका का $x$ और $-x$ पर मूल्यांकन करने से बाध्य हो जाता है; और किसी
बिंदु पर शून्य होने वाले बहुपदों के लिए $a$ पर मूल्यांकन करने से।
\end{method}

\begin{example}\label{ex:b1:vspaces:evenodd}
$\mathcal{F}(\R, \R)$ में सम फलन $\mathcal{P}$ और विषम फलन $\mathcal{I}$
\omterm{def:b1:vspaces:sum}{पूरक} हैं: कोई भी $f$ यों लिखा जाता है
\[
f(x) = \underbrace{\frac{f(x) + f(-x)}{2}}_{\text{सम}}
+ \underbrace{\frac{f(x) - f(-x)}{2}}_{\text{विषम}},
\]
और जो फलन सम भी हो और विषम भी, वह शून्य है। ($\exp$ पर लगाने से यह
\cref{ch:b1:functions} की जोड़ी $(\cosh, \sinh)$ ही है।)
\end{example}

\begin{example}[{$K_n[X]$ में एक पूरक जोड़ी}]
\label{ex:b1:vspaces:hyperplane}
$a \in K$ नियत कीजिए और $F = \{P \in K_n[X] : P(a) = 0\}$, $G =
\operatorname{Vect}(1)$ (अचर फलन) रखिए। तब $K_n[X] = F \oplus G$। वस्तुतः $F
\cap G$ में वे अचर फलन हैं जो $a$ पर शून्य होते हैं, अर्थात् $\{0\}$; और हर
$P$ यों विघटित होता है
\[
P = \underbrace{\bigl(P - P(a)\bigr)}_{\in F}
+ \underbrace{P(a)}_{\in G} .
\]
यह विघटन याद रखने योग्य है: किसी बिंदु पर के मान को घटा देना ही ``$a$ पर
शून्य होने वाले फलनों'' पर प्रक्षेप करने की मानक विधि है। ध्यान दीजिए कि $F$
एक बड़ी \omterm{def:b1:vspaces:subspace}{उपसमष्टि} है और $G$ एक छोटी; \omterm{def:b1:vspaces:sum}{पूरक} जोड़ी का किसी भी अर्थ में संतुलित
होना आवश्यक नहीं।
\end{example}

\begin{example}[पूरक उपसमष्टि कभी अद्वितीय नहीं होती]
\label{ex:b1:vspaces:manysupplements}
$\R^2$ में मान लीजिए $F = \operatorname{Vect}\bigl((1,0)\bigr)$ ($x$-अक्ष)।
$G = \operatorname{Vect}\bigl((0,1)\bigr)$ और $G' =
\operatorname{Vect}\bigl((1,1)\bigr)$ दोनों $F$ के \omterm{def:b1:vspaces:sum}{पूरक} हैं: प्रत्येक $F$ से
केवल $0$ पर मिलता है, और प्रत्येक जोड़ी का योग $\R^2$ है। एक ही सदिश के
विघटन भिन्न होते हैं:
\[
(2,\ 1.5) = \underbrace{(2, 0)}_{\in F} +
\underbrace{(0, 1.5)}_{\in G}
= \underbrace{(0.5,\ 0)}_{\in F} +
\underbrace{(1.5,\ 1.5)}_{\in G'} .
\]
वस्तुतः $F$ के अतिरिक्त \emph{हर} रेखा $\R^2$ में $F$ की \omterm{def:b1:vspaces:sum}{पूरक} है: \omterm{def:b1:vspaces:sum}{पूरक} भरपूर
मात्रा में हैं, और जब तक कोई अतिरिक्त संरचना (अंतर्गुणन,
\cref{ch:b1:euclid}) किसी एक को न चुन ले, तब तक ``वह'' \omterm{def:b1:vspaces:sum}{पूरक} कहना निरर्थक है।
\end{example}

\section{स्वतंत्र कुल, जनक कुल, आधार}

\begin{definition}\label{def:b1:vspaces:free}
$E$ के सदिशों का कुल $(x_1, \dots, x_p)$:
\begin{itemize}
  \item ($E$ का) \emph{जनक} कहलाता है जब $\operatorname{Vect}(x_1,\dots,x_p)
  = E$;
  \item \emph{स्वतंत्र}\index{स्वतंत्र कुल} (उसके सदिश \emph{रैखिकतः
  स्वतंत्र}) कहलाता है जब
        \[
        \lambda_1 x_1 + \dots + \lambda_p x_p = 0
        \implies \lambda_1 = \dots = \lambda_p = 0 ;
        \]
        अन्यथा \emph{परतंत्र};
  \item और \emph{आधार}\index{आधार} कहलाता है जब वह स्वतंत्र भी हो और जनक भी।
\end{itemize}
\end{definition}

\begin{proposition}[निर्देशांक]\label{prop:b1:vspaces:coordinates}
$(e_1, \dots, e_n)$ $E$ का \omterm{def:b1:vspaces:free}{आधार} है तभी जब हर $x \in E$ \emph{अद्वितीय} रूप
से संयोजन $x = \lambda_1 e_1 + \dots + \lambda_n e_n$ हो; और अदिश
$\lambda_i$ उस \omterm{def:b1:vspaces:free}{आधार} में $x$ के \emph{निर्देशांक}\index{निर्देशांक} कहलाते
हैं।
\end{proposition}

\begin{proof}
जनक होना $=$ विघटन का अस्तित्व। अद्वितीयता $=$ स्वतंत्रता: एक ही $x$ के दो
विघटन ऐसे संयोजन जितने भिन्न होते हैं जो $0$ के बराबर है; स्वतंत्रता उसके
सभी गुणांकों को --- अर्थात् निर्देशांकों के अंतरों को --- शून्य होने पर
बाध्य कर देती है। विलोमतः, कोई अतुच्छ शून्य संयोजन दो विघटन $0 = \sum
\lambda_i e_i = \sum 0\,e_i$ दे देता है।
\end{proof}

\begin{example}\label{ex:b1:vspaces:bases}
$K^n$ का \emph{विहित \omterm{def:b1:vspaces:free}{आधार}}: $e_i = (0, \dots, 1, \dots, 0)$ ($i$ स्थान पर
$1$)। एकपदी $(1, X, X^2, \dots, X^n)$: $K_n[X]$ का एक \omterm{def:b1:vspaces:free}{आधार} (स्वतंत्रता:
शून्य संयोजन शून्य \omterm{def:b1:poly:def}{बहुपद} है, अतः सभी गुणांक शून्य हो जाते हैं,
\cref{def:b1:poly:def})। $\R$ पर $\C$ में: \omterm{def:b1:vspaces:free}{आधार} $(1, \iu)$।
\end{example}

\begin{remark}[निर्देशांक सामूहिक परिश्रम हैं]
\label{rem:b1:vspaces:teameffort}
किसी \omterm{def:b1:vspaces:free}{आधार} $(e_1, \dots, e_n)$ में $x$ का पहला \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} \emph{सभी}
आधार-सदिशों पर निर्भर करता है, केवल $e_1$ पर नहीं। $\R^2$ में: सदिश $(3, 1)$
का पहला \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} विहित \omterm{def:b1:vspaces:free}{आधार} में $3$ है, पर \omterm{def:b1:vspaces:free}{आधार} $\bigl((1,0), (1,1)\bigr)$
में $2$ --- $(3,1) = a(1,0) + b(1,1)$ हल कीजिए: $b = 1$, $a = 2$। एक
आधार-सदिश बदलने से \emph{हर} \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} फिर से फेंटा जाता है;
\cref{ch:b1:matrices} इस फेंटने को आधार-परिवर्तन आव्यूह में बाँध देगा।
\end{remark}

\begin{example}[संभावित आधार की जाँच, आदि से अंत तक]
\label{ex:b1:vspaces:testbasis}
क्या $\mathcal{F} = (1 + X,\ 1 + X^2,\ X + X^2)$ $\R_2[X]$ का \omterm{def:b1:vspaces:free}{आधार} है? तीनों
बहुपदों को $u_1, u_2, u_3$ लिखिए। \emph{स्वतंत्रता}: शून्य संयोजन $a\,u_1 +
b\,u_2 + c\,u_3 = 0$ से गुणांक-दर-गुणांक मिलता है
\[
a + b = 0, \qquad a + c = 0, \qquad b + c = 0 ;
\]
पहले दो को घटाने पर $b = c$, फिर तीसरे से $2b = 0$ मिलता है: $a = b = c =
0$, अतः \omterm{def:b1:vspaces:free}{स्वतंत्र}। \emph{जनक होना}: तीन निकाय हल करने के बदले सममित संयोजन पर
ध्यान दीजिए
\[
u_1 + u_2 - u_3 = (1 + X) + (1 + X^2) - (X + X^2) = 2 ,
\]
अतः $1 = \frac12(u_1 + u_2 - u_3)$; फिर
\[
X = u_1 - 1 = \tfrac12\bigl(u_1 - u_2 + u_3\bigr),
\qquad
X^2 = u_2 - 1 = \tfrac12\bigl(-u_1 + u_2 + u_3\bigr).
\]
एकपदी \omterm{def:b1:vspaces:span}{जनित समष्टि} में हैं, अतः सब कुछ उसमें है: $\mathcal{F}$ एक \omterm{def:b1:vspaces:free}{आधार} है।
बोनस के रूप में, तीनों प्रदर्शित पंक्तियों को जोड़ देने पर किसी भी $P =
\alpha + \beta X + \gamma X^2$ के \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} मिल जाते हैं:
\[
P = \frac{\alpha + \beta - \gamma}{2}\,u_1
+ \frac{\alpha - \beta + \gamma}{2}\,u_2
+ \frac{-\alpha + \beta + \gamma}{2}\,u_3 .
\]
($P = X$ के साथ जाँच: \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} $\bigl(\frac12, -\frac12, \frac12\bigr)$,
जैसा ऊपर मिला था।) दो सीखें: कुल में छिपी सममिति प्रायः कोई छोटा रास्ता देने
वाला संयोजन छिपाए रहती है; और विमा उपलब्ध होते ही (\cref{ch:b1:findim}) इस
काम का पूरा जनक-वाला आधा हिस्सा मुफ़्त में मिल जाएगा --- $3$-विमीय समष्टि के
तीन \omterm{def:b1:vspaces:free}{स्वतंत्र} सदिश सदा \omterm{def:b1:vspaces:free}{आधार} बनाते हैं।
\end{example}

\begin{proposition}[स्वतंत्रता की उपयोगी कसौटियाँ]\label{prop:b1:vspaces:freecriteria}
\begin{enumerate}
  \item \emph{युग्मशः भिन्न घातों वाले अशून्य बहुपदों} का कुल \omterm{def:b1:vspaces:free}{स्वतंत्र} होता
  है।
  \item किसी \omterm{def:b1:vspaces:free}{स्वतंत्र कुल} में एक सदिश जोड़ने पर वह \omterm{def:b1:vspaces:free}{स्वतंत्र} बना रहता है तभी
  जब वह सदिश उस कुल की \omterm{def:b1:vspaces:span}{जनित समष्टि} के बाहर हो।
  \item \omterm{def:b1:vspaces:free}{स्वतंत्र कुल} का कोई भी उपकुल \omterm{def:b1:vspaces:free}{स्वतंत्र} होता है; और किसी जनक कुल को
  समाहित करने वाला कोई भी कुल जनक होता है।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) किसी शून्य संयोजन में उपस्थित उच्चतम घात देखिए: उसका गुणांक शून्य होना
ही है (उस घात को कोई नहीं काटता), और फिर सीढ़ी दर सीढ़ी नीचे उतरिए।

(2) यदि $x \in \operatorname{Vect}(x_1, \dots, x_p)$, तो संबंध $x -
\sum\lambda_i x_i = 0$ अतुच्छ है। विलोमतः, $(x_1, \dots, x_p, x)$ का कोई
अतुच्छ शून्य संयोजन $x$ को अशून्य गुणांक के साथ लिए बिना नहीं रह सकता
(अन्यथा वह छोटे कुल की स्वतंत्रता का विरोध करेगा), और $x$ के लिए हल करने पर
वह \omterm{def:b1:vspaces:span}{जनित समष्टि} में आ जाता है।

(3) उपकुल: उपकुल का कोई शून्य संयोजन पूरे कुल का ही शून्य संयोजन है जिसमें
अनुपस्थित गुणांक $0$ रख दिए गए हों; बड़े कुल की स्वतंत्रता उन सबको मार देती
है। अधिकुल: $E$ का हर सदिश पहले से ही जनक भाग का संयोजन है; अतिरिक्त सदिशों
को गुणांक $0$ दे दीजिए।
\end{proof}

\begin{example}[सीढ़ी सिद्धांत]
\label{ex:b1:vspaces:staircase}
मान लीजिए $P_0, P_1, \dots, P_n \in K_n[X]$ ऐसा है कि प्रत्येक $k$ के लिए
$\deg P_k = k$ (घातों की एक ``सीढ़ी'')। तब $(P_0, \dots, P_n)$ $K_n[X]$ का
\omterm{def:b1:vspaces:free}{आधार} है। स्वतंत्रता \cref{prop:b1:vspaces:freecriteria}~(1) है। जनक होने के
लिए घात पर परिमित अवरोहण से तर्क कीजिए: मान लीजिए $Q \in K_n[X]$, $Q \neq
0$, जिसकी घात $d$ और अग्र गुणांक $a$ है, और $b \neq 0$ $P_d$ का अग्र गुणांक
है। तब $Q - \frac ab P_d$ की घात $< d$ है (शीर्ष पद कट जाते हैं); $Q$ को इस
अंतर से बदलकर दोहराने पर अधिक से अधिक $n + 1$ चरणों में शून्य \omterm{def:b1:poly:def}{बहुपद} तक पहुँच
जाते हैं, और घटावों को उलटकर खोलने पर $Q$ $P_k$ के संयोजन के रूप में व्यक्त
हो जाता है। दो सीढ़ियाँ पहले ही मिल चुकी हैं: सरकाई हुई घातें
$\bigl((X-a)^k\bigr)_{0 \leq k \leq n}$ (\cref{exo:b1:vspaces:4}), और न्यूटन
गुणनफल $\bigl((X - x_0)(X - x_1)\cdots(X - x_{k-1})\bigr)_{0 \leq k \leq
n}$, जिन्हें सप्ताहांत समस्या में काम पर लगाया गया है।
\end{example}

\begin{example}[फलन समष्टियों में स्वतंत्रता]\label{ex:b1:vspaces:functionfree}
$\mathcal{F}(\R,\R)$ में $a_1 < \dots < a_p$ के साथ कुल $(\eu^{a_1 x},
\dots, \eu^{a_p x})$ \omterm{def:b1:vspaces:free}{स्वतंत्र} है: किसी शून्य संयोजन को $\eu^{a_p x}$ से भाग
दीजिए और $x \to +\infty$ लीजिए; अंतिम गुणांक मर जाता है, और फिर सीढ़ी दर
सीढ़ी नीचे उतरा जाता है (\cref{exo:b1:vspaces:8} इसे और इसके रूपांतरों को
विस्तार से देता है)। फलनों की स्वतंत्रता \emph{मूल्यांकन} से सिद्ध की जाती
है: सुचुने हुए बिंदुओं पर, अनंत पर, अथवा अवकलन के बाद।
\end{example}

\begin{example}[एक छिपा संबंध जनित समष्टि को सिकोड़ देता है]
\label{ex:b1:vspaces:hiddenrelation}
$\mathcal{F}(\R, \R)$ में $\operatorname{Vect}(1,\ \cos^2,\ \sin^2)$ क्या
है? सर्वसमिका $\cos^2 + \sin^2 = 1$ एक अतुच्छ शून्य संयोजन है
\[
1\cdot\mathbf{1} + (-1)\cos^2 + (-1)\sin^2 = 0 :
\]
अतः कुल परतंत्र है, और \omterm{def:b1:vspaces:span}{जनित समष्टि} पहले से ही अकेले $(1, \cos^2)$ से जनित है
($\sin^2 = 1 - \cos^2$)। वह छोटा कुल \omterm{def:b1:vspaces:free}{स्वतंत्र} है: सभी $x$ के लिए $a +
b\cos^2 x = 0$ से $x = 0$ और $x = \frac\pi2$ पर मिलता है: $a + b = 0$ और $a
= 0$। अतः \omterm{def:b1:vspaces:span}{जनित समष्टि} फलन समष्टि के भीतर एक \emph{समतल} है --- और उसमें
$\cos 2x = 2\cos^2 x - 1$ भी है: त्रिकोणमितीय फलनों के जो कुल देखने में
\omterm{def:b1:vspaces:free}{स्वतंत्र} लगते हैं, वे सर्वसमिकाओं के नीचे आए दिन ढह जाते हैं, और यही कारण है
कि स्वतंत्रता \emph{सिद्ध} करनी पड़ती है, सूची की लंबाई देखकर मान कभी नहीं
लेनी चाहिए।
\end{example}

\begin{remark}[सामान्य भूलें]
\label{rem:b1:vspaces:pitfalls}
चार शास्त्रीय जाल। \emph{युग्मशः देखना पर्याप्त नहीं है}: $\R^2$ में सदिश
$(1,0)$, $(0,1)$, $(1,1)$ युग्मशः अनुपाती नहीं हैं, फिर भी परतंत्र हैं ---
स्वतंत्रता \emph{पूरे} कुल का गुण है, जिसे एक ही वैश्विक संयोजन से जाँचा
जाता है, दो-दो करके कभी नहीं। \emph{शून्य सदिश सब कुछ विषैला कर देता है}:
$0$ को समाहित करने वाला कोई भी कुल परतंत्र है ($1\cdot 0 = 0$ एक अतुच्छ
संबंध है), शेष सदिश चाहे कितने ही निर्दोष क्यों न हों। \emph{संघ योग नहीं
है}: $F \cup G$ लगभग कभी \omterm{def:b1:vspaces:subspace}{उपसमष्टि} नहीं होता
(\cref{def:b1:vspaces:subspace}); दोनों को समाहित करने वाली सबसे छोटी
\omterm{def:b1:vspaces:subspace}{उपसमष्टि} $F + G$ है, जो प्रायः संघ से कहीं बड़ी होती है --- $\R^2$ में दो
भिन्न रेखाओं का संघ केवल दो कटती हुई रेखाएँ हैं, पर योग पूरा समतल।
\emph{प्रत्यक्ष होने के लिए तुच्छ सर्वनिष्ठ चाहिए, असंयुक्तता नहीं}: दो
उपसमष्टियाँ कभी असंयुक्त नहीं होतीं (दोनों में $0$ है); सही प्रतिबंध $F \cap
G = \{0\}$ है, और उसे \emph{सिद्ध} करना पड़ता है, किसी चित्र से पढ़ नहीं
लिया जाता --- \cref{ex:b1:vspaces:manysupplements} देखिए, जहाँ अनेक भिन्न
$G$ चलते हैं। \emph{स्वतंत्रता अदिशों पर निर्भर करती है}: जोड़ी $(1, \iu)$
$\R$-सदिश समष्टि के रूप में देखे गए $\C$ में \omterm{def:b1:vspaces:free}{स्वतंत्र} है, पर $\C$-सदिश
समष्टि के रूप में देखे गए $\C$ में परतंत्र ($\iu\cdot 1 + (-1)\cdot\iu =
0$)। किसी कुल को \omterm{def:b1:vspaces:free}{स्वतंत्र} घोषित करने से पहले यह सदा जान लीजिए कि कौन-सा
\omterm{def:b1:structures:field}{क्षेत्र} काम कर रहा है --- \cref{ch:b1:findim} की सप्ताहांत समस्या ठीक इसी
संवेदनशीलता को अपरिमेयता की उपपत्तियों में बदल देती है।
\end{remark}

\begin{remark}[यह भाषा कहाँ जाती है]
\label{rem:b1:vspaces:whereused}
इस अध्याय के बाद का सब कुछ यहीं रची गई भाषा बोलता है। \cref{ch:b1:findim}
आधार-सदिशों की गिनती करता है और ``\omterm{def:b1:vspaces:free}{स्वतंत्र}'' तथा ``जनक'' को एक ही पूर्णांक
--- विमा --- पर की असमिकाओं में बदल देता है। \cref{ch:b1:linmaps} उन
प्रतिचित्रणों का अध्ययन करता है जो दोनों संक्रियाओं के अनुकूल हों; वहाँ
\omterm{def:b1:vspaces:sum}{प्रत्यक्ष योग} प्रक्षेपक बन जाते हैं। \cref{ch:b1:matrices} सदिशों को किसी
\omterm{def:b1:vspaces:free}{आधार} में उनके निर्देशांकों से कूटबद्ध करता है --- और
\cref{prop:b1:vspaces:coordinates} उस कूटबद्धता का लाइसेंस है --- तथा
\cref{ch:b1:euclid} रैखिक संरचना के ऊपर लंबाइयाँ और कोण जोड़ देता है। स्नातक
वर्ष~2 के खंड में वही अभिगृहीत, अक्षरशः, किसी भी \omterm{def:b1:structures:field}{क्षेत्र} पर और अनंत विमा में
चलते हैं; इस अध्याय में कहीं भी परिमितता का प्रयोग नहीं हुआ।
\end{remark}

\begin{remark}[पुस्तक 3 भर पीछा करने योग्य तीन धागे]
\label{rem:b1:vspaces:threads}
इस अध्याय के तीन विशिष्ट विचारों को बढ़ते हुए देखिए। \emph{सीढ़ी सिद्धांत}
(\cref{ex:b1:vspaces:staircase}) इसी अध्याय की सप्ताहांत समस्या में न्यूटन
\omterm{def:b1:vspaces:free}{आधार} के रूप में लौटता है, वहीं द्विपद \omterm{def:b1:vspaces:free}{आधार} $(B_k)$ के रूप में, और
\cref{ch:b1:det} की सप्ताहांत समस्या में बहुपदीय एकांतरक वाली चाल के रूप
में: एक प्रमेयिका, तीन सारणिक-मुक्त लाभांश। \emph{स्वतंत्रता की कसौटी के रूप
में मूल्यांकन} (\cref{ex:b1:vspaces:functionfree}) \cref{ch:b1:linmaps} की
अंतर्वेशन तुल्याकारिता बन जाता है, फिर \cref{ch:b1:det} की वांडरमोंड कसौटी,
फिर \cref{ch:b1:euclid} की ग्राम कसौटी: वही प्रतिवर्त, तीन बार तीक्ष्ण किया
हुआ। \emph{\omterm{def:b1:vspaces:sum}{प्रत्यक्ष योग}} (\cref{def:b1:vspaces:sum}) \cref{ch:b1:linmaps}
में प्रक्षेपक बन जाते हैं, \cref{ch:b1:euclid} में लांबिक \omterm{thm:b1:logic:partition}{विभाजन} $E = F
\oplus F^\perp$, और \cref{ch:b1:multivar} की सप्ताहांत समस्या में न्यूनतम
वर्गों का व्याख्यात-जमा-अवशिष्ट विघटन। इस पुस्तक का बहुत थोड़ा ही ऐसा है जो
मूलतः इन तीन विचारों में से किसी एक का नया वस्त्र पहने रूप न हो।
\end{remark}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:vspaces:6}
अनुक्रमों की समष्टि में मान लीजिए $F$ अभिसारी अनुक्रमों का \omterm{def:b1:logic:sets}{समुच्चय} है और $G
= \operatorname{Vect}(u)$, जहाँ $u_n = (-1)^n$। सिद्ध कीजिए कि $F \cap G =
\{0\}$। क्या $F + G$ अनुक्रमों की पूरी समष्टि है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:vspaces:7}
मान लीजिए $F, G, H$ $E$ की उपसमष्टियाँ हैं। सिद्ध कीजिए कि
\[
F \cap (G + (F \cap H)) = (F \cap G) + (F \cap H),
\]
और $\R^2$ में एक उदाहरण से दिखाइए कि अप्रतिबंधित वितरण नियम $F \cap (G + H)
= (F\cap G) + (F \cap H)$ विफल हो जाता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:vspaces:8}
सिद्ध कीजिए कि $\mathcal{F}(\R, \R)$ के निम्नलिखित कुल \omterm{def:b1:vspaces:free}{स्वतंत्र} हैं:

  \item $(\cos x, \sin x, \cos 2x, \sin 2x)$;
  \item भिन्न-भिन्न $a_i$ के लिए $(x \mapsto \abs{x - a_1}, \dots, x \mapsto
  \abs{x - a_p})$ \emph{(हर फलन के लिए अवकलनीयता ठीक एक बिंदु पर विफल होती
  है)}।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:vspaces:9}
मान लीजिए $E$ एक $K$-सदिश समष्टि है और $F, G, H$ ऐसी उपसमष्टियाँ हैं कि $F +
G = F + H$, $F \cap G = F \cap H$ और $G \subseteq H$। सिद्ध कीजिए $G = H$।
$G \subseteq H$ की परिकल्पना के बिना एक प्रति-उदाहरण दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:vspaces:10}
$\R[X]$ में मान लीजिए $\mathcal P$ \emph{सम} बहुपदों का \omterm{def:b1:logic:sets}{समुच्चय} है ($P(-X) =

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:vspaces:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:vspaces:12}
मान लीजिए $E$ $\R$ (अथवा $\C$) पर एक \omterm{def:b1:vspaces:def}{सदिश समष्टि} है और $F_1, \dots, F_k$ $E$
की \emph{उचित} उपसमष्टियाँ हैं (प्रत्येक $F_i \neq E$)।
\begin{enumerate}
  \item $k = 2$ की स्थिति सीधे निपटाइए: यदि $F_1 \not\subseteq F_2$ और $F_2
  \not\subseteq F_1$, तो $x \in F_1 \setminus F_2$ और $y \in F_2 \setminus
  F_1$ चुनिए और $x + y$ का स्थान बताइए।
  \item व्यापक रूप से सिद्ध कीजिए कि $E \neq F_1 \cup \dots \cup F_k$: किसी
  अनंत \omterm{def:b1:structures:field}{क्षेत्र} पर की \omterm{def:b1:vspaces:def}{सदिश समष्टि} कभी उचित उपसमष्टियों का परिमित संघ नहीं
  होती। \emph{($k$ को न्यूनतम लीजिए, शेष $F_i$ के बाहर $x \in F_1$ चुनिए, $y
  \notin F_1$ चुनिए, और रेखा $t \mapsto y + tx$ का पीछा कीजिए।)}
\end{enumerate}
\end{exercise}

\section{समस्या: अंतर्वेशन, एक समष्टि के लिए तीन आधार}

\begin{problem}\label{pb:b1:vspaces:1}
$\R$ के $n + 1$ \emph{भिन्न-भिन्न} बिंदु $x_0, x_1, \dots, x_n$ नियत कीजिए।
यह समस्या \omterm{thm:b1:poly:lagrange}{लाग्रांज अंतर्वेशन} (\cref{thm:b1:poly:lagrange}) को इस अध्याय की
आँखों से फिर से देखती है: समष्टि $\R_n[X]$ तीन स्वाभाविक \omterm{def:b1:vspaces:free}{आधार} धारण करती है
--- लाग्रांज का, न्यूटन का, और (समान \omterm{prop:b1:reals:intervals}{अंतराल} वाले बिंदुओं के लिए) द्विपद \omterm{def:b1:vspaces:free}{आधार}
--- और हर \omterm{def:b1:vspaces:free}{आधार} किसी एक प्रश्न को आसान बना देता है। रास्ता एक सच्ची अंकगणितीय
प्रमेय पर समाप्त होता है: $\Z$ को $\Z$ में भेजने वाले बहुपदों का पोल्या
द्वारा दिया गया अभिलक्षण।\index{लाग्रांज अंतर्वेशन} \index{विभाजित
अंतर}\index{पूर्णांक-मान बहुपद}

\medskip \noindent\textbf{भाग I --- लाग्रांज \omterm{def:b1:vspaces:free}{आधार}।} $0 \leq i \leq n$ के लिए
रखिए
\[
L_i \;=\; \prod_{j \neq i} \frac{X - x_j}{x_i - x_j} \;\in\;
\R_n[X].
\]
\begin{enumerate}
  \item जाँचिए कि $\deg L_i = n$, और यह कि $j = i$ होने पर $L_i(x_j) = 1$,
  तथा $j \neq i$ होने पर $0$।
  \item सिद्ध कीजिए कि कुल $(L_0, \dots, L_n)$ \omterm{def:b1:vspaces:free}{स्वतंत्र} है।
  \item सिद्ध कीजिए कि हर $P \in \R_n[X]$ के लिए,
        \[
        P \;=\; \sum_{i=0}^{n} P(x_i)\, L_i ,
        \]
        और निष्कर्ष निकालिए कि $(L_0, \dots, L_n)$ $\R_n[X]$ का \omterm{def:b1:vspaces:free}{आधार} है।
        \emph{(दोनों पक्षों का अंतर लीजिए और उसके मूल गिनिए,
        \cref{cor:b1:poly:nroots}।)}
  \item अंतर्वेशन प्रमेय निकालिए: किन्हीं भी मानों $y_0, \dots, y_n \in \R$
  के लिए एक \emph{अद्वितीय} $P \in \R_n[X]$ ऐसा है कि सभी $i$ के लिए $P(x_i)
  = y_i$। लाग्रांज \omterm{def:b1:vspaces:free}{आधार} में किसी \omterm{def:b1:poly:def}{बहुपद} $P$ के \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} क्या हैं?
  \item सर्वसमिकाएँ सिद्ध कीजिए
        \[
        \sum_{i=0}^{n} L_i = 1
        \qquad\text{और, यदि } 0 \leq k \leq n,\qquad
        \sum_{i=0}^{n} x_i^{k}\, L_i = X^{k} .
        \]
\end{enumerate}

\medskip \noindent\textbf{भाग II --- न्यूटन \omterm{def:b1:vspaces:free}{आधार} और \omterm{pb:b1:vspaces:1}{विभाजित अंतर}।} $N_0 = 1$
रखिए और $1 \leq k \leq n$ के लिए $N_k = (X - x_0)(X - x_1) \cdots (X -
x_{k-1})$। गाँठों पर परिभाषित किसी फलन $f$ के लिए \emph{\omterm{pb:b1:vspaces:1}{विभाजित अंतर}} यों
परिभाषित कीजिए: $f[x_i] = f(x_i)$ और
\[
f[x_i, \dots, x_{i+k}] \;=\;
\frac{f[x_{i+1}, \dots, x_{i+k}] - f[x_i, \dots, x_{i+k-1}]}
{x_{i+k} - x_i} .
\]
\begin{enumerate}[resume]
  \item सिद्ध कीजिए कि $(N_0, N_1, \dots, N_n)$ $\R_n[X]$ का \omterm{def:b1:vspaces:free}{आधार} है।
  \item $f$ के मानों के पदों में $f[x_0, x_1]$ और $f[x_0, x_1, x_2]$ संगणित
  कीजिए, फिर तीन स्वेच्छ गाँठों पर $f(x) = x^2$ के सभी \omterm{pb:b1:vspaces:1}{विभाजित अंतर} संगणित
  कीजिए।
  \item (ऐटकिन की प्रमेयिका) मान लीजिए $R$ $x_0, \dots, x_{n-1}$ पर $f$ का
  अंतर्वेशन करता है और $Q$ $x_1, \dots, x_n$ पर $f$ का, और दोनों की घात
  $\leq n - 1$ है। सिद्ध कीजिए कि
        \[
        S \;=\; \frac{(X - x_0)\,Q - (X - x_n)\,R}{x_n - x_0}
        \]
        $x_0, x_1, \dots, x_n$ पर $f$ का अंतर्वेशन करता है।
  \item गाँठों की संख्या पर आगमन से निष्कर्ष निकालिए कि $x_0, \dots, x_k$ पर
  $f$ के अंतर्वेशक में $X^{k}$ का गुणांक ठीक $f[x_0, \dots, x_k]$ है।
  \item \emph{न्यूटन का अंतर्वेशन सूत्र} सिद्ध कीजिए: $x_0, \dots, x_n$ पर
  $f$ का अंतर्वेशक है
        \[
        P \;=\; \sum_{k=0}^{n} f[x_0, \dots, x_k]\, N_k ,
        \]
        और \omterm{def:b1:topology:closed}{संवृत} सूत्र निकालिए
        \[
        f[x_0, \dots, x_k] \;=\; \sum_{i=0}^{k}
        \frac{f(x_i)}{\prod_{j \neq i,\, j \leq k} (x_i - x_j)} ,
        \]
        जो दिखाता है कि $f[x_0, \dots, x_k]$ गाँठों के क्रम पर निर्भर नहीं
        करता।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- समान \omterm{prop:b1:reals:intervals}{अंतराल} वाली गाँठें: अंतर संकारक।}
अब से गाँठें $0, 1, 2, \dots$ हैं और किसी \omterm{def:b1:poly:def}{बहुपद} $P$ के लिए हम रखते हैं
\[
\Delta P(X) = P(X + 1) - P(X),
\qquad
B_k = \frac{X(X-1)\cdots(X-k+1)}{k!} \quad (B_0 = 1).
\]
\begin{enumerate}[resume]
  \item दिखाइए कि यदि $\deg P = m \geq 1$ और उसका अग्र गुणांक $a$ है, तो
  $\deg \Delta P = m - 1$ और उसका अग्र गुणांक $m\,a$ है, तथा यह कि $\Delta$
  अचरों को मार देता है।
  \item दिखाइए कि $(B_0, B_1, \dots, B_n)$ $\R_n[X]$ का \omterm{def:b1:vspaces:free}{आधार} है और यह कि $k
  \geq 1$ के लिए $\Delta B_k = B_{k-1}$।
  \item (न्यूटन का अग्र-अंतर सूत्र) सिद्ध कीजिए कि हर $P \in \R_n[X]$
  संतुष्ट करता है
        \[
        P \;=\; \sum_{k=0}^{n} \bigl(\Delta^{k} P\bigr)(0)\, B_k .
        \]
  \item सिद्ध कीजिए कि हर $k \geq 0$ के लिए,
        \[
        \bigl(\Delta^{k} P\bigr)(0) \;=\;
        \sum_{j=0}^{k} (-1)^{k-j} \binom{k}{j} P(j) .
        \]
  \item दिखाइए कि यदि $\deg P = n$ और उसका अग्र गुणांक $a_n$ है, तो
  $\Delta^{n} P$ अचर $n!\,a_n$ है और $\Delta^{n+1} P = 0$।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- \omterm{pb:b1:vspaces:1}{पूर्णांक-मान बहुपद}।} \omterm{def:b1:poly:def}{बहुपद} $P \in
\R[X]$ \emph{पूर्णांक-मान} तब कहलाता है जब हर $m \in \Z$ के लिए $P(m) \in
\Z$ हो।
\begin{enumerate}[resume]
  \item सिद्ध कीजिए कि प्रत्येक $B_k$ पूर्णांक-मान है। \emph{($m \geq k$, $0
  \leq m < k$ और $m < 0$ को अलग-अलग निपटाइए; $m = -q < 0$ के लिए $B_k(-q) =
  (-1)^k \binom{q + k - 1}{k}$ दिखाइए।)}
  \item \emph{पोल्या का अभिलक्षण} सिद्ध कीजिए: $P \in \R_n[X]$ पूर्णांक-मान
  है तभी जब \omterm{def:b1:vspaces:free}{आधार} $(B_0, \dots, B_n)$ में उसके \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} पूर्णांक हों।
  \item निष्कर्ष निकालिए: यदि $P \in \R_n[X]$ $n + 1$ \emph{क्रमागत}
  पूर्णांकों $a, a+1, \dots, a+n$ पर पूर्णांक मान लेता है, तो $P$
  पूर्णांक-मान है। \emph{(सरकाइए: अध्ययन को $Q(X) = P(X + a)$ पर लगाइए।)}
  \item प्रश्न 16 से निष्कर्ष निकालिए कि $k$ क्रमागत पूर्णांकों का गुणनफल
  सदा $k!$ से विभाज्य होता है।
  \item मान लीजिए $P = \dfrac{X(X+1)(2X+1)}{6}$। $0, 1, 2, 3$ पर उसकी न्यूटन
  सारणी संगणित कीजिए, $P$ को \omterm{def:b1:vspaces:free}{आधार} $(B_k)$ में लिखिए, और निष्कर्ष निकालिए कि
  $P$ पूर्णांक-मान है, यद्यपि उसका कोई भी एकपदी गुणांक पूर्णांक नहीं है।
  $\Delta P = (X+1)^2$ सत्यापित कीजिए और $m \in \N$ के लिए $P(m) = 1^2 + 2^2
  + \dots + m^2$ निकालिए।
\end{enumerate}

\medskip \noindent\textbf{भाग V --- लाभांश।}
\begin{enumerate}[resume]
  \item मान लीजिए $P \in \R_n[X]$ $i = 0, 1, \dots, n$ पर मानों $2^i$ का
  अंतर्वेशन करता है। दिखाइए कि $P = B_0 + B_1 + \dots + B_n$ और यह कि $P(n +
  1) = 2^{n+1} - 1$: ``दुगुना होने का प्रतिरूप'' सदा ठीक अगले ही बिंदु पर
  टूट जाता है।
  \item (विविक्त प्रतिअवकलज) सिद्ध कीजिए कि सभी पूर्णांकों $m \geq 1$ और $k
  \geq 0$ के लिए,
        \[
        \sum_{j=0}^{m-1} B_k(j) \;=\; B_{k+1}(m),
        \]
        अर्थात् हॉकी-स्टिक सर्वसमिका $\sum_{j=k}^{m-1} \binom{j}{k} =
        \binom{m}{k+1}$।
  \item $X^2$ और $X^3$ को \omterm{def:b1:vspaces:free}{आधार} $(B_k)$ में प्रसारित कीजिए और
  $\sum_{j=0}^{m-1} j^2$ तथा $\sum_{j=0}^{m-1} j^3$ के लिए \omterm{def:b1:topology:closed}{संवृत} सूत्र
  निकालिए; निकोमैकस की सर्वसमिका $1^3 + \dots + m^3 = (1 + \dots + m)^2$ फिर
  से प्राप्त कीजिए।
  \item $n = 2$ और गाँठें $0, 1, 2$ लीजिए। इस समस्या के तीनों आधारों ---
  एकपदी \omterm{def:b1:vspaces:free}{आधार}, लाग्रांज \omterm{def:b1:vspaces:free}{आधार}, न्यूटन \omterm{def:b1:vspaces:free}{आधार} --- में $X^2$ के \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} लिखिए।
  तीनों उत्तर प्रश्न 4 और 9 के सामने जाँचिए।
  \item संश्लेषण। चार वाक्यों में: कौन-सी सदिश-समष्टि अवधारणा प्रश्न 4 को
  स्वतः कर देती है; न्यूटन \omterm{def:b1:vspaces:free}{आधार} \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} \emph{पुनरावर्ती} रूप से क्यों
  संगणित करता है जबकि लाग्रांज \omterm{def:b1:vspaces:free}{आधार} उन्हें \emph{तत्काल} पढ़ लेता है; दोनों
  \omterm{def:b1:vspaces:free}{आधार} कौन-सी स्वतंत्रता-कसौटी साझा करते हैं; और ठीक किस अर्थ में पोल्या की
  प्रमेय कहती है कि किसी \omterm{def:b1:poly:def}{बहुपद} की पूर्णांकता \emph{सही \omterm{def:b1:vspaces:free}{आधार} में} उसके
  निर्देशांकों का गुण है।
\end{enumerate}
\end{problem}
