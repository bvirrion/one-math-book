\chapter{अनुक्रम}\label{ch:b1:seq}

उच्चतर माध्यमिक खंड में अनुक्रमों को साधा तो गया था, पर सीमा की धारणा आधी
विश्वास पर ली गई थी। यहाँ सिद्धांत $\R$ की पूर्णता (\cref{ch:b1:reals}) पर
नए सिरे से खड़ा किया गया है: हर चिरपरिचित प्रमेय --- एकदिष्ट अभिसरण, \omterm{thm:b1:seq:adjacent}{संलग्न
अनुक्रम}, बोल्ज़ानो--वाइरश्ट्रास, कोशी कसौटी --- उसी एक अभिगृहीत का एक मुख
है। अध्याय $u_{n+1} = f(u_n)$ से परिभाषित अनुक्रमों के व्यावहारिक अध्ययन पर
समाप्त होता है।

\section{अभिसरण}

\begin{definition}[अनुक्रम की सीमा]\label{def:b1:seq:limit}
वास्तविक संख्याओं का अनुक्रम $(u_n)$ $\ell \in \R$ की ओर \emph{अभिसरित} तब
होता है जब\index{सीमा!अनुक्रम की}
\[
\forall \varepsilon > 0,\ \exists N \in \N,\ \forall n \geq N,
\qquad \abs{u_n - \ell} \leq \varepsilon .
\]
लिखा जाता है $u_n \to \ell$ या $\lim u_n = \ell$। जो अनुक्रम (किसी भी
वास्तविक संख्या की ओर) अभिसरित नहीं होता वह \emph{अपसरित} होता है।
\emph{$+\infty$ की ओर} अपसरण: $\forall M,\ \exists N,\ \forall n \geq N,\
u_n \geq M$ (इसी प्रकार $-\infty$)।
\end{definition}

\begin{example}[एक $\varepsilon$--$N$ उपपत्ति, एक बार पूरी लिखी हुई]\label{ex:b1:seq:epsilonN}
दावा: $u_n = \dfrac{n^2 + 1}{2n^2 - 3} \to \dfrac12$। पहले त्रुटि अलग कीजिए:
\[
\Bigl| u_n - \frac12 \Bigr|
= \Bigl| \frac{2(n^2 + 1) - (2n^2 - 3)}{2(2n^2 - 3)} \Bigr|
= \frac{5}{2\,\abs{2n^2 - 3}}
= \frac{5}{2\,(2n^2 - 3)} \quad (n \geq 2).
\]
फिर उसे किसी सरल राशि से दबाइए: $n \geq 2$ के लिए $2n^2 - 3 \geq n^2$, अतः
त्रुटि $\leq \frac{5}{2n^2} \leq \frac 5{2n}$ है। $\varepsilon > 0$ दिया हो,
तो \omterm{thm:b1:reals:archimedes}{आर्किमिडीज़ गुणधर्म} $N \geq \max\bigl(2, \frac{5}{2\varepsilon}\bigr)$
देता है; $n \geq N$ के लिए त्रुटि $\leq \varepsilon$ है। काम पूरा। समापन का
सार: $\varepsilon$--$N$ उपपत्ति में ठीक तीन चालें होती हैं --- त्रुटि संगणित
कीजिए, उसे किसी ह्रासमान प्रारंभिक व्यंजक से परिबद्ध कीजिए, और देहली के लिए
हल कीजिए --- और इस अध्याय की प्रमेयों (संक्रियाएँ, दबाव) के बाद ऐसी उपपत्ति
लगभग कभी फिर नहीं लिखनी पड़ती: प्रमेय उन तीनों चालों को एक बार और सदा के लिए
बाँध देती हैं।
\end{example}

\begin{example}[अनंत की ओर अपसरण, प्रमाणित]\label{ex:b1:seq:toinfinity}
दावा: $u_n = n^2 - 100n \to +\infty$। प्रभावी पद का गुणनखंडन कीजिए: $n \geq
200$ के लिए $u_n = n^2\bigl(1 - \frac{100}{n}\bigr) \geq \frac{n^2}{2}$। $M$
दिया हो, तो $N = \max\bigl(200, \lceil\sqrt{2M}\rceil\bigr)$ लीजिए: $n \geq
N$ के लिए $u_n \geq \frac{n^2}{2} \geq M$। दो आदतें यहाँ दिखाई देती हैं:
प्रभावी-पद गुणनखंडन एक प्रतियोगिता ($n^2$ बनाम $-100n$) को एक ही पैमाने और
$1$ की ओर जाते किसी गुणक में बदल देता है; और देहली बहुत बड़ी हो सकती है
($u_{100} = 0$, और अनुक्रम $n = 100$ से पहले तो ऋणात्मक तक है) --- $+\infty$
की ओर अपसरण पूँछ के विषय में \omterm{def:b1:logic:statement}{कथन} है, और किसी भी परिमित मात्रा की बदमाशी से
उदासीन।
\end{example}

\begin{proposition}[पहले गुणधर्म]\label{prop:b1:seq:first}
\begin{enumerate}
  \item सीमा, यदि विद्यमान हो, अद्वितीय होती है।
  \item अभिसारी अनुक्रम परिबद्ध होता है।
  \item यदि $u_n \to \ell$, तो परिमित कितने पदों का कोई भी परिवर्तन अभिसरण
  और सीमा दोनों को अपरिवर्तित छोड़ देता है।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) यदि $\ell \neq \ell'$ के साथ $u_n \to \ell$ और $u_n \to \ell'$, तो
$\varepsilon = \frac{\abs{\ell - \ell'}}{3}$ लीजिए: दोनों देहलियों के आगे
$\abs{\ell - \ell'} \leq \abs{\ell - u_n} + \abs{u_n - \ell'} \leq
2\varepsilon = \frac23 \abs{\ell - \ell'}$, जो असंगत है।

(2) $\varepsilon = 1$ के साथ: $N$ के आगे $\abs{u_n} \leq \abs\ell + 1$; और
उससे पहले के परिमित कितने पद भी परिबद्ध हैं, अतः $\abs{u_n} \leq
\max(\abs{u_0}, \dots, \abs{u_{N-1}}, \abs\ell + 1)$।

(3) विस्तार से: मान लीजिए $n \geq n_0$ के लिए $v_n = u_n$ और $u_n \to \ell$।
$\varepsilon > 0$ दिया हो, तो $(u_n)$ के लिए देहली $N$ लीजिए: $n \geq
\max(N, n_0)$ के लिए $\abs{v_n - \ell} = \abs{u_n - \ell} \leq \varepsilon$।
अतः $v_n \to \ell$: परिभाषा केवल $n \geq N$ पर परिमाणन करती है, और किसी भी
परिमित उपसर्ग को देहली बड़ी करने की क़ीमत पर बदला जा सकता है। (इसीलिए इस
अध्याय में सर्वत्र ``सभी बड़े $n$ के लिए'' वाली परिकल्पनाएँ पर्याप्त हैं।)
\end{proof}

\begin{theorem}[सीमाओं पर संक्रियाएँ]\label{thm:b1:seq:operations}
यदि $u_n \to \ell$ और $v_n \to m$, तो
\[
u_n + v_n \to \ell + m, \qquad u_n v_n \to \ell m, \qquad
\frac{u_n}{v_n} \to \frac{\ell}{m} \ (\text{यदि } m \neq 0),
\qquad \abs{u_n} \to \abs\ell .
\]
\end{theorem}

\begin{proof}
\emph{योग:} बड़ी देहली के आगे $\abs{(u_n + v_n) - (\ell + m)} \leq \abs{u_n
- \ell} + \abs{v_n - m} \leq 2\varepsilon$। \emph{गुणनफल:} लिखिए
\[
u_n v_n - \ell m = (u_n - \ell)\,v_n + \ell\,(v_n - m);
\]
$(v_n)$ किसी $B$ से परिबद्ध है (\cref{prop:b1:seq:first}), अतः दायाँ पक्ष
$\leq B\abs{u_n - \ell} + \abs{\ell}\,\abs{v_n - m}$ है, जो मनचाहा छोटा है।
\emph{भागफल:} $\frac 1{v_n}$ की स्थिति सँभाल लेना पर्याप्त है। $\varepsilon
= \frac{\abs m}{2}$ के साथ: किसी $N_0$ के आगे $\abs{v_n} \geq \frac{\abs
m}{2}$, अतः
\[
\Bigl| \frac{1}{v_n} - \frac 1m \Bigr|
= \frac{\abs{m - v_n}}{\abs{v_n m}}
\leq \frac{2}{m^2}\,\abs{v_n - m} \longrightarrow 0 .
\]
\emph{निरपेक्ष मान:} $\bigl|\abs{u_n} - \abs\ell\bigr| \leq \abs{u_n -
\ell}$ (विलोम त्रिभुज असमिका, \cref{prop:b1:complex:rules})।
\end{proof}

\begin{example}[संक्रियाएँ और एक बीजीय युक्ति]\label{ex:b1:seq:conjugatetrick}
$\lim\,\bigl(\sqrt{n^2 + n} - n\bigr)$ संगणित कीजिए। दोनों टुकड़े अलग-अलग
$+\infty$ की ओर जाते हैं: संक्रिया-प्रमेय उनके अंतर के विषय में कुछ नहीं
कहती (एक \emph{अनिश्चित रूप})। संयुग्मी से गुणा कीजिए:
\[
\sqrt{n^2 + n} - n
= \frac{(n^2 + n) - n^2}{\sqrt{n^2 + n} + n}
= \frac{n}{\sqrt{n^2+n} + n}
= \frac{1}{\sqrt{1 + \frac1n} + 1} .
\]
अब सब कुछ अभिसरित होता है: $\sqrt{1 + \frac1n} \to 1$, क्योंकि $0 \leq
\sqrt{1 + h} - 1 = \frac{h}{\sqrt{1+h} + 1} \leq h$ (फिर संयुग्मी, फिर $h =
\frac1n$ के साथ दबाव); और तब संक्रिया-प्रमेय सीमा $\frac{1}{1 + 1} =
\frac12$ दे देती है। समापन का सार: संक्रिया-प्रमेय सभी सीमाओं का कैलकुलेटर
नहीं है --- अनिश्चित रूपों ($\infty - \infty$, $\frac00$, $0 \times \infty$,
$1^\infty$) को पहले बीजगणित (संयुग्मी, प्रभावी पद का गुणनखंडन) से
\emph{रूपांतरित} करना पड़ता है जब तक हर टुकड़ा अभिसरित न हो जाए; और ज़िद्दी
स्थितियों की व्यवस्थित मशीन \cref{ch:b1:taylor} का अनंतस्पर्शी प्रसार है।
\end{example}

\begin{theorem}[सीमाएँ और क्रम]\label{thm:b1:seq:order}
\begin{enumerate}
  \item यदि सभी बड़े $n$ के लिए $u_n \leq v_n$, और दोनों अभिसरित होते हैं,
  तो $\lim u_n \leq \lim v_n$। (कठोर असमिकाएँ सीमा तक \emph{नहीं} जातीं:
  $\frac 1n > 0$, फिर भी $\lim = 0$।)
  \item (दबाव प्रमेय\index{दबाव प्रमेय}) यदि सभी बड़े $n$ के लिए $u_n \leq
  w_n \leq v_n$ और $u_n, v_n \to \ell$, तो $w_n \to \ell$।
  \item यदि $u_n \to \ell > 0$, तो सभी बड़े $n$ के लिए $u_n > \frac\ell2 >
  0$।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) मान लीजिए $\ell = \lim u_n > m = \lim v_n$; $\varepsilon = \frac{\ell -
m}{3}$ के साथ बड़े पद $v_n \leq m + \varepsilon < \ell - \varepsilon \leq
u_n$ संतुष्ट करते हैं, जो $u_n \leq v_n$ का विरोध करता है।

(2) देहलियों के आगे: $\ell - \varepsilon \leq u_n \leq w_n \leq v_n \leq
\ell + \varepsilon$।

(3) $\varepsilon = \frac\ell2$ के साथ \cref{def:b1:seq:limit} ही है।
\end{proof}

\begin{example}[दो दबाव]\label{ex:b1:seq:squeezes}
(क) $\dfrac{\sin n}{n} \to 0$: $-\frac1n \leq \frac{\sin n}{n} \leq \frac1n$
से, जहाँ दोनों दीवारें $0$ पर ढह जाती हैं --- ऊपर के अनियमित अंश को समझने की
आवश्यकता ही नहीं। (ख) $(2^n + 3^n)^{1/n} \to 3$: भीतर को घेरिए,
\[
3^n \leq 2^n + 3^n \leq 2\cdot3^n
\quad\Longrightarrow\quad
3 \leq (2^n + 3^n)^{1/n} \leq 3\cdot 2^{1/n} ,
\]
और $2^{1/n} = \eu^{\frac{\ln 2}{n}} \to 1$ (\cref{exo:b1:seq:2} में
$5^{1/n}$ की भाँति): दबाव $3$ दे देता है। समापन का सार: प्रतिस्पर्धी
चरघातांकियों का योग अपने \emph{सबसे बड़े} पद जैसा व्यवहार करता है --- छोटे
पद किसी निरापद अचर गुणक में समा जाते हैं, जिसे $n$-वाँ मूल मिटा देता है।
\end{example}

\section{एकदिष्ट अनुक्रम}

\begin{theorem}[एकदिष्ट सीमा प्रमेय]\label{thm:b1:seq:monotone}
ऊपर से परिबद्ध वर्धमान अनुक्रम अभिसरित होता है, $\sup\{u_n : n \in \N\}$ की
ओर; और ऊपर से अपरिबद्ध वर्धमान अनुक्रम $+\infty$ की ओर अपसरित होता है।
(ह्रासमान अनुक्रमों के लिए दर्पण \omterm{def:b1:logic:statement}{कथन}।)\index{एकदिष्ट सीमा प्रमेय}
\end{theorem}

\begin{proof}
मान लीजिए $s = \sup\{u_n\}$ (\cref{thm:b1:reals:sup})। $\varepsilon > 0$
दिया हो, तो $\varepsilon$-अभिलक्षण (\cref{prop:b1:reals:epsilon}) ऐसा $N$
देता है कि $u_N > s - \varepsilon$; एकदिष्टता से सभी $n \geq N$ के लिए $s -
\varepsilon < u_N \leq u_n \leq s$: अर्थात् $s$ की ओर अभिसरण। यदि अपरिबद्ध
हो: प्रत्येक $M$ के लिए कोई $u_N > M$, और एकदिष्टता बाद के सभी पदों को $M$
के ऊपर रखती है।
\end{proof}

\begin{example}[अस्तित्व-मशीन के रूप में एकदिष्ट प्रमेय]\label{ex:b1:seq:existencemachine}
मान लीजिए $u_n = \prod_{k=1}^{n} \bigl(1 + \frac{1}{2^k}\bigr)$। प्रत्येक
गुणनखंड $1$ से बड़ा है, अतः $(u_n)$ वर्धमान है। ऊपर से परिबद्ध? लघुगणक लीजिए
और $\ln(1 + x) \leq x$ का प्रयोग कीजिए (\cref{ex:b1:derivative:convexineq}
इसका पूर्वानुमान करता है; अथवा उच्चतर माध्यमिक खंड का स्थूल $1 + x \leq
\eu^x$):
\[
\ln u_n = \sum_{k=1}^{n} \ln\Bigl(1 + \frac{1}{2^k}\Bigr)
\leq \sum_{k=1}^{n} \frac{1}{2^k} < 1 ,
\]
अतः $u_n < \eu$। वर्धमान और परिबद्ध: अतः $(u_n)$ किसी $\ell \in
\intoc{u_1}{\eu}$ की ओर अभिसरित होता है --- एक पूर्णतः सुपरिभाषित वास्तविक
संख्या, जिसका कोई संवृत रूप दिखाई नहीं देता ($\ell = 2.384\dots$)। समापन का
सार: एकदिष्ट सीमा प्रमेय विश्लेषण की सबसे सस्ती अस्तित्व-मशीन है; उसी ने
$\eu$ को नाम दिया (नीचे \cref{ex:b1:seq:e}), और \cref{ch:b1:series} में वह
केवल परिबद्धता से हर धनात्मक श्रेणी का अभिसरण तय कर देगी।
\end{example}

\begin{theorem}[संलग्न अनुक्रम]\label{thm:b1:seq:adjacent}
मान लीजिए $(a_n)$ वर्धमान है, $(b_n)$ ह्रासमान, और $b_n - a_n \to 0$। तब
दोनों अभिसरित होते हैं, एक \emph{उभयनिष्ठ} सीमा $\ell$ की ओर, और सभी $n$ के
लिए $a_n \leq \ell \leq b_n$।\index{संलग्न अनुक्रम}
\end{theorem}

\begin{proof}
पहले, सभी $n$ के लिए $a_n \leq b_n$: अनुक्रम $(b_n - a_n)$ ह्रासमान है और
$0$ की ओर जाता है, अतः वह $\geq 0$ है (कोई ऋणात्मक पद उसे $0$ से नीचे जमा
देता)। फिर $(a_n)$ वर्धमान है और $b_0$ से ऊपर परिबद्ध: अतः वह किसी $\ell$
(\cref{thm:b1:seq:monotone}) की ओर अभिसरित होता है; इसी प्रकार $(b_n) \to
\ell'$; और $\ell' - \ell = \lim (b_n - a_n) = 0$। असमिकाएँ $a_n \leq \ell
\leq b_n$ एकदिष्टता से निकलती हैं ($\ell = \sup a_k \geq a_n$, इत्यादि)।
\end{proof}

  \node[above, font=\small, text=gray] at (0.46,0.055)
    {$b_n - a_n \to 0$};
\end{tikzpicture}

{\small \omterm{thm:b1:seq:adjacent}{संलग्न अनुक्रम}: $(a_n)$ चढ़ता है, $(b_n)$ उतरता है, और उनके बीच का
\omterm{prop:b1:reals:intervals}{अंतराल} सिकुड़कर $0$ हो जाता है। प्रत्येक \omterm{prop:b1:reals:intervals}{अंतराल} $\intcc{a_n}{b_n}$ अपने बाद
के सभी अंतरालों को समेटता है, और उभयनिष्ठ सीमा $\ell$ ही वह अद्वितीय बिंदु
है जो हर \omterm{prop:b1:reals:intervals}{अंतराल} में बचा रहता है --- यही चित्र नीचे बोल्ज़ानो--वाइरश्ट्रास की
और \cref{ch:b1:continuity} में मध्यवर्ती मान प्रमेय की द्विभाजन उपपत्तियों
के पीछे है।}
\end{omfigure}

\begin{example}[संख्या $\eu$]\label{ex:b1:seq:e}
$a_n = \sum_{k=0}^{n} \frac{1}{k!}$ और $b_n = a_n + \frac{1}{n \cdot n!}$
($n \geq 1$) रखिए। तब $(a_n)$ बढ़ता है; और
\[
b_{n+1} - b_n = \frac{1}{(n+1)!} + \frac{1}{(n+1)(n+1)!} -
\frac{1}{n\,n!}
= \frac{n(n+1) + n - (n+1)^2}{n(n+1)(n+1)!}
= \frac{-1}{n(n+1)(n+1)!} < 0 ,
\]
अतः $(b_n)$ घटता है, और $b_n - a_n \to 0$: अर्थात् संलग्न। उनकी उभयनिष्ठ
सीमा (यहाँ परिभाषा से) संख्या $\eu \approx 2.71828$ है; और असमिकाएँ $a_n <
\eu < b_n$ $\eu \notin \Q$ सिद्ध करने के लिए पर्याप्त तीक्ष्ण हैं
(\cref{exo:b1:seq:9})।
\end{example}

\section{उपानुक्रम और बोल्ज़ानो--वाइरश्ट्रास}

\begin{definition}[उपानुक्रम]\label{def:b1:seq:subsequence}
$(u_n)$ का \emph{उपानुक्रम}\index{उपानुक्रम} अनुक्रम $(u_{\varphi(n)})$ है,
जहाँ $\varphi \colon \N \to \N$ कठोरतः वर्धमान है (ध्यान दीजिए $\varphi(n)
\geq n$, आगमन से)।
\end{definition}

\begin{proposition}\label{prop:b1:seq:subsequences}
यदि $u_n \to \ell$ ($\ell \in \R$ या $\pm\infty$), तो प्रत्येक \omterm{def:b1:seq:subsequence}{उपानुक्रम}
$\ell$ की ओर जाता है। फलस्वरूप, जिस अनुक्रम के दो उपानुक्रमों की सीमाएँ
भिन्न हों वह अपसरित होता है। विलोमतः, यदि $(u_{2n})$ और $(u_{2n+1})$ दोनों
\emph{एक ही} $\ell$ की ओर अभिसरित होते हैं, तो $u_n \to \ell$।
\end{proposition}

\begin{proof}
$(u_n)$ के लिए देहली $N$ के आगे सभी सूचकांक $\varphi(n) \geq n \geq N$ योग्य
हैं (असमिका $\varphi(n) \geq n$ वही आगमन है जो \cref{def:b1:seq:subsequence}
में नोट किया गया: $\varphi(0) \geq 0$, और $\varphi(n+1) > \varphi(n) \geq n$
$\varphi(n+1) \geq n + 1$ पर बाध्य कर देता है)। विलोम के लिए: $\varepsilon$
दिया हो, तो दोनों देहलियाँ $N_0$ (सम) और $N_1$ (विषम) लीजिए; कोई भी सूचकांक
$n \geq \max(2N_0, 2N_1 + 1)$ या तो सम है, $k \geq N_0$ के साथ $n = 2k$, या
विषम, $k \geq N_1$ के साथ $n = 2k+1$ --- और दोनों स्थितियों में $\abs{u_n -
\ell} \leq \varepsilon$: अर्थात् हर सूचकांक दोनों उपानुक्रमों में से किसी एक
में आ जाता है, और यही पूरी बात है।
\end{proof}

\begin{example}[उपानुक्रमीय सीमाएँ]\label{ex:b1:seq:sublimits}
$u_n = (-1)^n \frac{n}{n+1}$ के लिए: सम \omterm{def:b1:seq:subsequence}{उपानुक्रम} $1$ की ओर जाता है और विषम
$-1$ की ओर, अतः अनुक्रम अपसरित होता है --- पर वह यह सुव्यवस्थित ढंग से करता
है, दोनों मानों $\pm 1$ के आसपास गुच्छे बनाकर। $u_n = \cos\frac{2\pi n}{3}$
के लिए: सूचकांकों $3k$, $3k + 1$, $3k + 2$ वाले तीनों \omterm{def:b1:seq:subsequence}{उपानुक्रम} अचर हैं और
क्रमशः $1$, $-\frac12$, $-\frac12$ के बराबर; उपानुक्रमीय सीमाओं का \omterm{def:b1:logic:sets}{समुच्चय}
$\{1, -\frac12\}$ है। समापन का सार: कोई \emph{परिबद्ध} अनुक्रम ठीक तब
अभिसरित होता है जब उसकी एक ही उपानुक्रमीय सीमा हो (\cref{exo:b1:seq:8});
परिबद्ध अनुक्रम का अपसरण सदा कम से कम दो गुच्छों का अर्थ रखता है, और नीचे
बोल्ज़ानो--वाइरश्ट्रास गारंटी देता है कि कम से कम एक गुच्छा है।
\end{example}

\begin{theorem}[बोल्ज़ानो--वाइरश्ट्रास]\label{thm:b1:seq:bw}
वास्तविक संख्याओं के प्रत्येक परिबद्ध अनुक्रम का कोई अभिसारी \omterm{def:b1:seq:subsequence}{उपानुक्रम} होता
है।\index{बोल्ज़ानो--वाइरश्ट्रास प्रमेय}
\end{theorem}

\begin{proof}
मान लीजिए सभी $n$ के लिए $u_n \in \intcc{a}{b}$। द्विभाजन से नेस्टेड खंड
बनाइए: $\intcc{a_0}{b_0} = \intcc{a}{b}$ रखिए; और यदि $\intcc{a_k}{b_k}$ में
अनंत कितने $n$ के लिए $u_n$ हो, तो उसके दोनों अर्धों में से एक में अब भी
अनंत कितने $n$ के लिए $u_n$ होगा --- उसे $\intcc{a_{k+1}}{b_{k+1}}$ कहिए।
अनुक्रम $(a_k)$, $(b_k)$ संलग्न हैं ($b_k - a_k = \frac{b-a}{2^k} \to 0$),
जिनकी उभयनिष्ठ सीमा $\ell$ है (\cref{thm:b1:seq:adjacent})।

निष्कर्षण: ऐसा $\varphi(0)$ चुनिए कि $u_{\varphi(0)} \in \intcc{a_0}{b_0}$,
फिर पुनरावर्ती रूप से ऐसा $\varphi(k+1) > \varphi(k)$ कि $u_{\varphi(k+1)}
\in \intcc{a_{k+1}}{b_{k+1}}$ --- यह संभव है, क्योंकि उस खंड में अनंत कितने
पद हैं। तब $a_k \leq u_{\varphi(k)} \leq b_k$, और दबाव प्रमेय
$u_{\varphi(k)} \to \ell$ दे देती है।
\end{proof}

\begin{remark}[बोल्ज़ानो--वाइरश्ट्रास क्या कहता है और क्या नहीं]
वह यह \emph{कहता है}: केवल परिबद्धता से कोई \omterm{def:b1:seq:subsequence}{उपानुक्रम} अभिसरित होता है ---
अर्थात् अस्तित्व, बिना किसी सूत्र के, जैसा द्विभाजन उपपत्ति स्पष्ट कर देती
है (कुछ भी नहीं बताता कि \emph{कौन-से} सूचकांक बचते हैं)। वह यह \emph{नहीं}
कहता कि सीमा अद्वितीय है: $((-1)^n)$ के ऐसे \omterm{def:b1:seq:subsequence}{उपानुक्रम} हैं जो $1$ की ओर और
$-1$ की ओर अभिसरित होते हैं, और उपानुक्रमीय सीमाओं का \omterm{def:b1:logic:sets}{समुच्चय} अनंत तक हो
सकता है (\cref{ex:b1:seq:sublimits}, और \cref{pb:b1:topology:1} में तो पूरा
कैंटर \omterm{def:b1:logic:sets}{समुच्चय})। अपरिबद्धता में वह जीवित नहीं रहता: $(n)$ का कोई अभिसारी
\omterm{def:b1:seq:subsequence}{उपानुक्रम} है ही नहीं --- यद्यपि किसी भी अपरिबद्ध अनुक्रम से $+\infty$ या
$-\infty$ की ओर जाता \omterm{def:b1:seq:subsequence}{उपानुक्रम} सदा निकाला जा सकता है (मान लीजिए ऐसा
$\varphi(k)$ चुनिए कि $u_{\varphi(k)} \geq k$)। सही ढंग से प्रयुक्त होने पर
यह प्रमेय एक \emph{अस्तित्व-पंप} है: वह नीचे कोशी कसौटी के मर्म पर, हाइने की
प्रमेय में और चरम मान प्रमेय में प्रकट होती है --- और सदा ऐसा बिंदु उत्पन्न
करने के लिए जो कोई स्पष्ट रचना नहीं देती।
\end{remark}

\section{कोशी अनुक्रम और पूर्णता}

\begin{definition}\label{def:b1:seq:cauchy}
अनुक्रम $(u_n)$ \emph{कोशी अनुक्रम}\index{कोशी अनुक्रम} तब है जब उसके पद
\emph{एक-दूसरे के} मनचाहे निकट आ जाएँ:
\[
\forall \varepsilon > 0,\ \exists N,\ \forall p, q \geq N,
\qquad \abs{u_p - u_q} \leq \varepsilon .
\]
\end{definition}

\begin{example}[हाथ से कोशी गुणधर्म सत्यापित करना]\label{ex:b1:seq:cauchyhand}
मान लीजिए $u_n = \sum_{k=0}^{n} \frac{\cos k}{2^k}$ --- न कोई एकदिष्टता, न
कोई अनुमेय सीमा। $p > q$ के लिए:
\[
\abs{u_p - u_q}
= \Bigl| \sum_{k=q+1}^{p} \frac{\cos k}{2^k} \Bigr|
\leq \sum_{k=q+1}^{p} \frac{1}{2^k}
< \frac{1}{2^{q}} ,
\]
जो त्रिभुज असमिका, $\abs{\cos k} \leq 1$ और एक परिमित गुणोत्तर योग से मिलता
है। $\varepsilon > 0$ दिया हो, तो ऐसा $N$ चुनिए कि $2^{-N} \leq
\varepsilon$: $N$ के आगे सारे \omterm{prop:b1:reals:intervals}{अंतराल} $\leq \varepsilon$ हैं, अतः अनुक्रम
कोशी है, इसलिए अभिसरित --- और वह भी ऐसी सीमा की ओर जिसे कोई संवृत रूप में
नाम नहीं दे सकता, और यही तो बात है। समापन का सार: वृद्धियों का गुणोत्तर दमन
कोशी गुणधर्म कमाने का मानक उपाय है, और \cref{ch:b1:series} इस तर्क को
``निरपेक्ष अभिसरण से अभिसरण'' के रूप में बोतल में बंद कर देगा।
\end{example}

\begin{theorem}[$\R$ की पूर्णता]\label{thm:b1:seq:complete}
वास्तविक संख्याओं का अनुक्रम अभिसरित होता है यदि और केवल यदि वह \omterm{def:b1:seq:cauchy}{कोशी अनुक्रम}
हो।\index{पूर्णता@$\R$ की पूर्णता}
\end{theorem}

\begin{proof}
($\Rightarrow$) यदि $u_n \to \ell$: $\frac\varepsilon2$ के लिए देहली के आगे
$\abs{u_p - u_q} \leq \abs{u_p - \ell} + \abs{\ell - u_q} \leq \varepsilon$।

($\Leftarrow$) मान लीजिए $(u_n)$ कोशी है। \emph{वह परिबद्ध है}: $\varepsilon
= 1$ के साथ, $N$ के आगे सभी पद $u_N$ के $1$ के भीतर हैं, और आरंभिक भाग
परिमित है। \emph{निष्कर्षण}: \cref{thm:b1:seq:bw} से कोई \omterm{def:b1:seq:subsequence}{उपानुक्रम}
$u_{\varphi(n)} \to \ell$। \emph{निष्कर्ष}: $\varepsilon > 0$ दिया हो, तो
$N$ (कोशी, $\frac\varepsilon2$ के लिए) और ऐसा $n \geq N$ लीजिए कि
$\abs{u_{\varphi(n)} - \ell} \leq \frac\varepsilon2$ और $\varphi(n) \geq N$;
तब प्रत्येक $p \geq N$ के लिए:
\[
\abs{u_p - \ell} \leq \abs{u_p - u_{\varphi(n)}} +
\abs{u_{\varphi(n)} - \ell} \leq \varepsilon . \qedhere
\]
\end{proof}

\begin{remark}
इस कसौटी का मूल्य: वह अभिसरण को \emph{सीमा का नाम लिए बिना} प्रमाणित कर देती
है। $\Q$ पर वह विफल है ($\sqrt 2$ की दशमलव कटाइयाँ परिमेय संख्याओं का ऐसा
\omterm{def:b1:seq:cauchy}{कोशी अनुक्रम} बनाती हैं जिसकी कोई परिमेय सीमा नहीं): अतः पूर्णता $\R$ का
गुणधर्म है, जो \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} अभिगृहीत के तुल्य है। श्रेणियों के अभिसरण
(\cref{ch:b1:series}) के पीछे का परिश्रमी घोड़ा भी वही है।
\end{remark}

\begin{example}[अदृश्य सीमा वाला एक कोशी अनुक्रम]\label{ex:b1:seq:basel}
मान लीजिए $S_n = \sum_{k=1}^{n} \frac{1}{k^2}$। $p > q \geq 1$ के लिए:
\[
S_p - S_q = \sum_{k=q+1}^{p} \frac{1}{k^2}
\leq \sum_{k=q+1}^{p} \frac{1}{k(k-1)}
= \sum_{k=q+1}^{p} \Bigl(\frac{1}{k-1} - \frac 1k\Bigr)
= \frac 1q - \frac 1p < \frac 1q ,
\]
अतः $N > \frac1\varepsilon$ के आगे सभी \omterm{prop:b1:reals:intervals}{अंतराल} $\leq \varepsilon$ हैं:
$(S_n)$ कोशी है, इसलिए अभिसरित। ध्यान दीजिए कि अभी क्या हुआ: हमने किसी
विशिष्ट वास्तविक संख्या का अस्तित्व सिद्ध कर दिया, यद्यपि उसके लिए हमारे पास
कोई नाम नहीं है। (वह $\frac{\pi^2}{6}$ है --- ऑयलर की एक प्रसिद्ध सर्वसमिका,
जो स्नातक वर्ष 2 के खंड में सिद्ध है; इस अध्याय की कोई चीज़ हमें यह नहीं बता
सकती थी।) यही श्रम-विभाजन --- अस्तित्व अभी, पहचान बाद में, यदि कभी --- कोशी
कसौटी की पूरी बात है, और \cref{ch:b1:series} में श्रेणियों के सिद्धांत का
इंजन।
\end{example}

\section{पुनरावर्ती अनुक्रम}

\begin{method}[$u_{n+1} = f(u_n)$ का अध्ययन]\label{met:b1:seq:recurrent}
$f$ और आरंभिक बिंदु $u_0$ दिए हों:\index{पुनरावर्ती अनुक्रम}
\begin{enumerate}
  \item \emph{स्थायी \omterm{prop:b1:reals:intervals}{अंतराल}:} ऐसा \omterm{prop:b1:reals:intervals}{अंतराल} $I$ खोजिए जिसमें $f(I) \subseteq I$
  हो और जो $u_0$ को समेटे: तब सभी $u_n \in I$ (आगमन से)।
  \item \emph{उम्मीदवार सीमाएँ:} यदि $u_n \to \ell \in I$ और $f$ $\ell$ पर
  संतत है (\cref{ch:b1:continuity}), तो $\ell$ एक \emph{अचल बिंदु} है:
  $f(\ell) = \ell$। $f(x) = x$ हल कीजिए।
  \item \emph{एकदिष्टता:} यदि $f$ $I$ पर वर्धमान है, तो $(u_n)$ एकदिष्ट है
  ($u_1 \geq u_0$ होने पर वर्धमान, अन्यथा ह्रासमान); परिबद्धता के साथ मिलकर
  \cref{thm:b1:seq:monotone} निष्कर्ष दे देती है। यदि $f$ ह्रासमान है, तो
  दोनों \omterm{def:b1:seq:subsequence}{उपानुक्रम} $(u_{2n})$ और $(u_{2n+1})$ देखिए, जो $f \circ f$ के लिए
  एकदिष्ट हैं।
  \item \emph{त्रुटि पर नियंत्रण:} $k < 1$ के साथ असमिका $\abs{f(x) - \ell}
  \leq k\abs{x - \ell}$ सीधे $\abs{u_n - \ell} \leq k^n \abs{u_0 - \ell} \to
  0$ दे देती है।
\end{enumerate}
\end{method}

\begin{example}[हीरो की विधि]\label{ex:b1:seq:heron}
मान लीजिए $u_0 = 2$ और $u_{n+1} = \dfrac12\Bigl(u_n + \dfrac{2}{u_n}\Bigr)$:
यही $\sqrt 2$ के लिए प्राचीन कलनविधि है।\index{हीरो की विधि}
\begin{itemize}
  \item \emph{स्थायित्व:} $x > 0$ के लिए समांतर--गुणोत्तर माध्य असमिका
  $\frac12(x + \frac2x) \geq \sqrt{x \cdot \frac 2x} = \sqrt 2$ देती है; अतः
  $I = \intco{\sqrt 2}{+\infty}$ स्थायी है और उसमें $u_1$ है (वस्तुतः $u_1 =
  \frac32 \geq \sqrt 2$)।
  \item \emph{एकदिष्टता:} $x \geq \sqrt 2$ के लिए $\;x - f(x) = \frac{x^2 -
  2}{2x} \geq 0$: अनुक्रम $u_1$ से आगे घटता है, और नीचे से $\sqrt 2$ द्वारा
  परिबद्ध है: अतः अभिसरित।
  \item \emph{सीमा:} अचल बिंदु $x = \frac12(x + \frac2x)$ हल करते हैं,
  अर्थात् $x^2 = 2$: $I$ पर $\ell = \sqrt 2$।
  \item \emph{गति:} $u_{n+1} - \sqrt 2 = \frac{(u_n - \sqrt2)^2}{2u_n} \leq
  \frac{(u_n - \sqrt 2)^2}{2\sqrt 2}$: सही अंकों की संख्या हर पद पर लगभग
  \emph{दुगुनी} हो जाती है (द्विघातीय अभिसरण)।
\end{itemize}
\end{example}

\begin{remark}[सीमाओं की सामान्य भूलें]
चार चिरपरिचित। (क) \emph{छोटे पद अभिसरण नहीं दे देते}: $u_{n+1} - u_n \to 0$
कोशी गुणधर्म से कहीं दुर्बल है --- हरात्मक योग $H_n$ के पद $\frac{1}{n+1}
\to 0$ हैं, फिर भी वे $+\infty$ की ओर अपसरित होते हैं (\cref{exo:b1:seq:5});
कोशी शर्त \emph{सभी} बड़े युग्मों के लिए $\abs{u_p - u_q}$ को नियंत्रित करती
है, केवल क्रमागत के लिए नहीं। (ख) \emph{कठोर असमिकाएँ सीमा में मर जाती हैं}:
सभी $n$ के लिए $u_n < v_n$ से केवल $\lim u_n \leq \lim v_n$ मिलता है
(\cref{thm:b1:seq:order}); $\frac1n > 0$, फिर भी $\lim = 0$। (ग)
\emph{परिबद्ध होना अभिसारी होना नहीं है}: $((-1)^n)$ परिबद्ध है और अपसरित
होता है; परिबद्धता के साथ \emph{एकदिष्टता} अभिसरण देती है, अकेली परिबद्धता
केवल किसी अभिसारी \omterm{def:b1:seq:subsequence}{उपानुक्रम} की गारंटी देती है (\cref{thm:b1:seq:bw})। (घ)
\emph{अचल-बिंदु समीकरण दूसरे स्थान पर आता है, पहले नहीं}: $u_{n+1} = f(u_n)$
के लिए $f(\ell) = \ell$ हल करने से सीमा की पहचान \emph{तभी} होती है जब
अभिसरण सिद्ध हो चुका हो। पुनरावृत्ति $u_{n+1} = 2u_n$ का अद्वितीय अचल बिंदु
$\ell = 0$ है, फिर भी $u_0 = 1$ से चलने पर अनुक्रम $+\infty$ की ओर भागता है:
समीकरण $\ell = 2\ell$ को किसी सीमा का अधिकार कभी था ही नहीं। क्रम सदा यही
है: पहले अस्तित्व (\cref{met:b1:seq:recurrent}, पद 1--3), फिर पहचान।
\end{remark}

\begin{example}[ह्रासमान $f$: स्वर्ण पुनरावृत्ति]\label{ex:b1:seq:golden}
मान लीजिए $u_0 = 1$ और $u_{n+1} = \dfrac{1}{1 + u_n}$। यहाँ $f(x) =
\frac{1}{1+x}$ \emph{ह्रासमान} है, अतः अनुक्रम एकदिष्ट नहीं है (वह अपनी सीमा
के इर्द-गिर्द एकांतरित होता है); \cref{met:b1:seq:recurrent} का संकुचन-पद ही
सही औज़ार है। स्थायित्व: यदि $x \in \intcc{\frac12}{1}$, तो $1 + x \in
\intcc{\frac32}{2}$, अतः $f(x) \in \intcc{\frac12}{\frac23} \subseteq
\intcc{\frac12}{1}$, और $u_1 = \frac12$ पूरे अनुक्रम को वहीं रख देता है। अचल
बिंदु: $\ell > 0$ के साथ $\ell = \frac{1}{1+\ell}$ $\ell^2 + \ell - 1 = 0$
देता है, अर्थात्
\[
\ell = \frac{\sqrt5 - 1}{2} = 0.6180\dots
\]
(स्वर्ण अनुपात का प्रतिलोम)। संकुचन: $x, y \in \intcc{\frac12}{1}$ के लिए,
\[
\abs{f(x) - f(y)} = \frac{\abs{x - y}}{(1+x)(1+y)}
\leq \frac{\abs{x-y}}{(3/2)^2} = \frac49\,\abs{x - y} ,
\]
अतः $\abs{u_n - \ell} \leq \bigl(\frac49\bigr)^{n-1}\abs{u_1 - \ell} \to 0$:
गुणोत्तर गति के साथ अभिसरण, और किसी एकदिष्टता की आवश्यकता नहीं। समापन का
सार: एकदिष्ट विधियाँ और संकुचन विधियाँ पुनरावर्ती दुनिया को आपस में बाँट
लेती हैं --- वर्धमान $f$ एकदिष्ट कक्षाएँ देता है, ह्रासमान $f$ एकांतरित
कक्षाएँ, जिन्हें लिप्शिट्ज़ अचर $< 1$ वश में कर लेता है (व्यवस्थित सिद्धांत
\cref{exo:b1:derivative:11} है)।
\end{example}

\begin{remark}[इस खंड के भीतर आगे की दृष्टि]
अनुक्रम वह मापक यंत्र हैं जिसे शेष खंड हर वस्तु के सामने रखता है।
\cref{ch:b1:topology} में वे संवृतता और संहतता को \emph{अभिलक्षित} करते हैं;
\cref{ch:b1:continuity} में वे फलनों की सीमाएँ ढोते हैं;
\cref{ch:b1:integration} में रीमान योग ऐसे अनुक्रम हैं जो समाकल की ओर
अभिसरित होते हैं; \cref{ch:b1:series} तो अनुक्रमों के एक विशेष वर्ग, अर्थात्
आंशिक योगों, का सिद्धांत \emph{ही है}। बीजगणित के अध्याय भी उन्हें खाते हैं:
\cref{ch:b1:matrices} में किसी आव्यूह की पुनरावृत्तियाँ ऐसे अनुक्रम बनाती
हैं जिनका व्यवहार ($A^n$ का अभिसरण) इसी अध्याय की शब्दावली वाला
रैखिक-बीजगणित का प्रश्न है। सर्वत्र साथ ले जाने योग्य दो प्रमेय: एकदिष्ट
सीमा (क्रम से अस्तित्व) और बोल्ज़ानो--वाइरश्ट्रास (परिबद्धता से अस्तित्व)
--- इन दोनों के बीच इस पुस्तक की लगभग हर सीमा जन्म लेती है।
\end{remark}

\begin{remark}[सम्मिश्र अनुक्रम]
सम्मिश्र संख्याओं का अनुक्रम $(z_n)$ $\ell$ की ओर तब अभिसरित होता है जब
$\abs{z_n - \ell} \to 0$; समतुल्य रूप से, जब $\Re(z_n) \to \Re(\ell)$ और
$\Im(z_n) \to \Im(\ell)$ ($\abs{z}$ की $\abs{\Re z} + \abs{\Im z}$ से तुलना
कीजिए)। जिन प्रमेयों में क्रम नहीं आता --- संक्रियाएँ,
बोल्ज़ानो--वाइरश्ट्रास (दो बार निष्कर्षण), कोशी कसौटी --- वे अक्षरशः चली
जाती हैं।
\end{remark}

\section{समस्या: चेज़ारो, स्टोल्ज़, और ज्या का धीमा पतन}

\begin{problem}[{सप्ताहांत समस्या --- चेज़ारो--स्टोल्ज़ प्रमेय और $u_{n+1} = \sin
u_n$ के लिए अनंतस्पर्शी
$u_n \sim \sqrt{3/n}$}]
\label{pb:b1:seq:1}
चेज़ारो--स्टोल्ज़ प्रमेय विविक्त लोपिताल नियम है: भागफल $a_n/b_n$ की सीमा
खोजने के लिए \emph{वृद्धियों} के भागफल $(a_{n+1} - a_n)/(b_{n+1} - b_n)$ की
सीमा खोज लेना पर्याप्त है। यह समस्या उस प्रमेय को सिद्ध करती है, उससे
चिरपरिचित सीमाएँ बटोरती है, और फिर उसे एक प्रसिद्ध लक्ष्य पर साधती है:
अनुक्रम $u_{n+1} = \sin u_n$, जो ठीक-ठीक संगणनीय गति $u_n \sim \sqrt{3/n}$
से $0$ की ओर सरकता है।\index{चेज़ारो--स्टोल्ज़ प्रमेय} उच्चतर माध्यमिक खंड
के दो तथ्य यहाँ दिए हुए मान लिए गए हैं और आगे इसी खंड में ईमानदारी से सिद्ध
किए गए हैं: स्पर्शरेखा असमिका
\[
\tag{G1} \eu^{u} \geq 1 + u \quad (u \in \R),
\]
जो \cref{ch:b1:derivative} में उत्तलता से फिर सिद्ध होती है, और ज्या का
घेराव
\[
\tag{G2} x - \frac{x^3}{6} \;\leq\; \sin x \;\leq\; x -
\frac{x^3}{6} + \frac{x^5}{120} \quad (0 \leq x \leq 1), \qquad
\abs{\sin x} \leq \abs{x} \quad (x \in \R),
\]
जो \cref{ch:b1:taylor} में टेलर सूत्र से फिर सिद्ध होता है।

\medskip \noindent\textbf{भाग I --- बिना संवृत सूत्र वाले योग।}
\begin{enumerate}
  \item $1 + 2 + \dots + n = \frac{n(n+1)}{2}$ और $1^2 + \dots + n^2 =
  \frac{n(n+1)(2n+1)}{6}$ का प्रयोग करके $\lim \frac{1 + 2 + \dots +
  n}{n^2}$ और $\lim \frac{1^2 + \dots + n^2}{n^3}$ संगणित कीजिए।
  \item मान लीजिए $T_n = \sum_{k=1}^n \sqrt k$, जिसका कोई संवृत सूत्र नहीं
  है। घेराव सिद्ध कीजिए
        \[
        \frac{1}{2\sqrt 2}\,n^{3/2} \;\leq\; T_n \;\leq\;
        n^{3/2}
        \]
        \emph{(निचले परिबंध के लिए केवल पद $k > \frac n2$ रखिए)}। अतः $T_n$
        की कोटि $n^{3/2}$ है --- पर अचर कौन-सा? यह प्रश्न प्रश्न 8 तक रोक
        रखिए।
  \item (दूरबीनी प्रमेयिका) मान लीजिए $(b_n)$ कठोरतः वर्धमान है और सभी $k
  \geq N$ के लिए
        \[
        m \;\leq\; \frac{a_{k+1} - a_k}{b_{k+1} - b_k} \;\leq\;
        M .
        \]
        सिद्ध कीजिए कि प्रत्येक $n > N$ के लिए $m \leq \dfrac{a_n - a_N}{b_n
        - b_N} \leq M$।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- चेज़ारो--स्टोल्ज़ प्रमेय।} मान लीजिए
$(b_n)$ कठोरतः वर्धमान है और $b_n \to +\infty$, तथा $\dfrac{a_{n+1} -
a_n}{b_{n+1} - b_n} \to \ell \in \R$।
\begin{enumerate}[resume]
  \item $\varepsilon > 0$ नियत कीजिए। दिखाइए कि ऐसा $N$ है कि सभी $n > N$ के
  लिए $\ell - \varepsilon \leq \dfrac{a_n - a_N}{b_n - b_N} \leq \ell +
  \varepsilon$।
  \item $n > N$ के लिए सर्वसमिका स्थापित कीजिए
        \[
        \frac{a_n}{b_n} - \ell
        = \frac{a_N - \ell\,b_N}{b_n}
        + \Bigl(1 - \frac{b_N}{b_n}\Bigr)
        \Bigl(\frac{a_n - a_N}{b_n - b_N} - \ell\Bigr),
        \]
        और प्रमेय पर पहुँचिए: $\dfrac{a_n}{b_n} \to \ell$।
  \item $+\infty$ वाला रूपांतर सिद्ध कीजिए: यदि $\dfrac{a_{n+1} -
  a_n}{b_{n+1} - b_n} \to +\infty$ ($(b_n)$ पर वही परिकल्पनाएँ), तो
  $\dfrac{a_n}{b_n} \to +\infty$।
  \item $b_n = n$ लीजिए: \cref{exo:b1:seq:10} की चेज़ारो माध्य प्रमेय वापस
  मिल जाती है। फिर दिखाइए कि चेज़ारो--स्टोल्ज़ का विलोम विफल है: $a_n =
  (-1)^n$, $b_n = n$ के लिए भागफल $a_n/b_n$ अभिसरित होता है जबकि वृद्धियों
  का भागफल नहीं। स्टोल्ज़ एकतरफ़ा मार्ग है।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- पहले लाभांश।}
\begin{enumerate}[resume]
  \item संयुग्मन से $(1+h)^{3/2} - 1 = \dfrac{3h + 3h^2 + h^3}{(1+h)^{3/2} +
  1}$ सिद्ध कीजिए, उससे $n\bigl((1 + \tfrac1n)^{3/2} - 1\bigr) \to \tfrac32$
  निकालिए, और चेज़ारो--स्टोल्ज़ से निष्कर्ष निकालिए:
        \[
        T_n = \sum_{k=1}^{n} \sqrt k \;\sim\; \tfrac23\,
        n^{3/2} ,
        \]
        जिससे प्रश्न 2 का लटका हुआ प्रश्न सुलझ जाता है।
  \item केवल (G1) से लघुगणक का घेराव निकालिए
        \[
        \frac{t}{1 + t} \;\leq\; \ln(1 + t) \;\leq\; t
        \qquad (t > -1)
        \]
        \emph{((G1) को $u = \ln(1+t)$ पर और $u = -t/(1+t)$ पर लगाइए)}।
  \item दिखाइए कि $b_n = \ln n$ कठोरतः वर्धमान है और $\ln n \to +\infty$, और
  चेज़ारो--स्टोल्ज़ तथा प्रश्न 9 से सिद्ध कीजिए कि
        \[
        H_n = \sum_{k=1}^{n} \frac 1k \;\sim\; \ln n .
        \]
        (सूक्ष्मतर संरचना $H_n = \ln n + \gamma + o(1)$ \cref{ch:b1:series}
        की सप्ताहांत समस्या है।)
  \item (अनुपातों से मूलों तक) मान लीजिए $u_n > 0$ और $\frac{u_{n+1}}{u_n}
  \to L > 0$। प्रश्न 9 का प्रयोग करके दिखाइए $\ln\frac{u_{n+1}}{u_n} \to \ln
  L$; चेज़ारो लगाकर $\frac{\ln u_n}{n} \to \ln L$ निष्कर्ष निकालिए, फिर (G1)
  के साथ $u_n^{1/n} \to L$। अनुप्रयोग: $\lim\,\binom{2n}{n}^{1/n}$ संगणित
  कीजिए।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- ज्या का धीमा पतन।} मान लीजिए $u_0 \in
\R$ और $u_{n+1} = \sin u_n$।
\begin{enumerate}[resume]
  \item (G2) से दिखाइए कि $0 < x \leq 1$ के लिए $0 < \sin x < x$। उससे
  निकालिए: $u_1 \in \intcc{-1}{1}$; यदि $u_1 = 0$, तो अनुक्रम कोटि $1$ से
  शून्य है; और यदि $u_1 > 0$ ($u_1 < 0$ की स्थिति सममित है, $\sin$ विषम), तो
  $(u_n)_{n \geq 1}$ कठोरतः ह्रासमान, धनात्मक है और $0$ की ओर अभिसरित होता
  है \emph{(सीमा की पहचान $\ell = \sin \ell$ से कीजिए, जिसमें $\abs{\sin a -
  \sin b} \leq \abs{a - b}$ का प्रयोग कीजिए, जो स्वयं (G2) और गुणनफल-से-योग
  सूत्र का परिणाम है)}।
  \item अब से $u_1 \in \intoc{0}{1}$ मान लीजिए। (G2) का प्रयोग करते हुए दबाव
  से दिखाइए:
        \[
        \frac{\sin u_n}{u_n} \to 1
        \qquad\text{और}\qquad
        \frac{u_n - \sin u_n}{u_n^{3}} \to \frac16 .
        \]
  \item गुणनखंडन सिद्ध कीजिए
        \[
        w_n := \frac{1}{u_{n+1}^{2}} - \frac{1}{u_n^{2}}
        = \frac{u_n - \sin u_n}{u_n^{3}} \cdot
        \frac{u_n + \sin u_n}{u_n} \cdot
        \Bigl(\frac{u_n}{\sin u_n}\Bigr)^{2},
        \]
        और $w_n \to \frac13$ निकालिए।
  \item \cref{exo:b1:seq:10} (वृद्धि वाला रूप) से निष्कर्ष निकालिए कि
  $\frac{1}{n\,u_n^{2}} \to \frac13$, और फिर वर्गमूल के लिए संयुग्मन के तर्क
  से मुख्य परिणाम:
        \[
        \sqrt n\;u_n \longrightarrow \sqrt 3 ,
        \qquad\text{अर्थात्}\qquad
        u_n \sim \sqrt{\frac 3n} .
        \]
  \item धीमेपन का परिमाणन कीजिए: दिखाइए कि अंततः $\sqrt{2/n} \leq u_n \leq
  2/\sqrt n$, जिससे $u_n \leq 10^{-2}$ तक पहुँचने के लिए $20\,000$ से अधिक
  पुनरावृत्तियाँ चाहिए (अनंतस्पर्शी के अनुसार लगभग $30\,000$)। \omterm{ex:b1:seq:heron}{हीरो की विधि}
  (\cref{ex:b1:seq:heron}) से तुलना कीजिए और संरचनात्मक कारण समझाइए: अचल
  बिंदु $0$ पर $\sin$ की प्रवणता $1$ है (एक \emph{उदासीन} अचल बिंदु), जबकि
  त्रुटि आधी करने वाली पुनरावृत्तियों को $< 1$ \omterm{def:b1:complex:field}{मापांक} वाली प्रवणता चाहिए।
  \item दिखाइए कि \emph{प्रत्येक} आरंभिक बिंदु $u_0 \in \R$ के लिए या तो
  कोटि $1$ से $u_n = 0$, या $\abs{u_n} \sim \sqrt{3/n}$ --- अर्थात् पतन
  सार्वभौमिक है, केवल चिह्न $u_0$ का स्मरण रखता है।
\end{enumerate}

\medskip \noindent\textbf{भाग V --- व्यापक सिद्धांत।} ज्या इस मशीन का एक ही
उदाहरण है।
\begin{enumerate}[resume]
  \item मान लीजिए $u_n > 0$, $u_n \to 0$, और $\dfrac{u_n - u_{n+1}}{u_n^{2}}
  \to a > 0$। क्रमशः सिद्ध कीजिए: $\frac{u_{n+1}}{u_n} \to 1$; फिर
  $\frac{1}{u_{n+1}} - \frac{1}{u_n} \to a$; फिर $n\,u_n \to \frac1a$।
  \item (यथार्थ प्रतिरूप) $u_{n+1} = \dfrac{u_n}{1 + u_n}$, $u_0 > 0$ के
  लिए: दिखाइए कि $\frac{1}{u_n}$ समांतर है, उसे यथार्थ रूप से हल कीजिए, और
  प्रश्न 18 के निष्कर्ष को यथार्थ सूत्र के सामने जाँचिए।
  \item $u_{n+1} = u_n \eu^{-u_n}$, $u_0 > 0$ के लिए: दिखाइए $u_n \to 0$, $t
  > 0$ के लिए (G1) से $\frac{1 - \eu^{-t}}{t}$ को $\frac{1}{1+t}$ और $1$ के
  बीच दबाइए, और निष्कर्ष निकालिए $u_n \sim \frac 1n$।
  \item (त्रिघात संपर्क, वर्गित दूरबीन) मान लीजिए $u_n > 0$, $u_n \to 0$,
  $\dfrac{u_n - u_{n+1}}{u_n^{3}} \to a > 0$। प्रश्न 14 का गुणनखंडन ढालकर
  दिखाइए कि $\frac{1}{u_{n+1}^2} - \frac{1}{u_n^2} \to 2a$, और निष्कर्ष
  निकालिए $n\,u_n^{2} \to \frac{1}{2a}$। जाँचिए कि $a = \frac16$ भाग IV को
  वापस दे देता है।
\end{enumerate}

\medskip \noindent\textbf{भाग VI --- विधि की सीमाएँ, और उपदेश।}
\begin{enumerate}[resume]
  \item दिखाइए कि परिकल्पना $b_n \to +\infty$ छोड़ी नहीं जा सकती: $a_n = 2 -
  2^{-n}$ और $b_n = 1 - 2^{-n}$ के लिए वृद्धियों का भागफल $1$ की ओर जाता है
  जबकि $\frac{a_n} {b_n} \to 2$। प्रश्न 5 की उपपत्ति की ठीक वह पंक्ति बताइए
  जो टूट जाती है।
  \item (स्टोल्ज़ दो बार) सिद्ध कीजिए $\sum_{k=1}^{n} H_k \sim n \ln n$
  \emph{(चेज़ारो--स्टोल्ज़ का एक प्रयोग, फिर प्रश्न 10; प्रश्न 9 का प्रयोग
  करके $(n+1)\ln(n+1) - n\ln n$ को परिबद्ध कीजिए)}।
  \item (गुणोत्तर माध्य) यदि $u_n > 0$ और $u_n \to \ell > 0$, तो दिखाइए
  $(u_1 u_2 \cdots u_n)^{1/n} \to \ell$; यदि $u_n \to +\infty$, तो दिखाइए
  $(u_1 \cdots u_n)^{1/n} \to +\infty$। उससे $(n!)^{1/n} \to +\infty$
  निकालिए।
  \item संश्लेषण, एक-एक वाक्य में: (क) इस समस्या में पूर्णता ठीक कहाँ आई;
  (ख) चेज़ारो--स्टोल्ज़ किस अर्थ में विविक्त लोपिताल नियम है (उसका अवकलीय
  जुड़वाँ \cref{ch:b1:derivative} की माध्य मान प्रमेय पर टिका है); (ग)
  उदासीन अचल बिंदु पर $f$ के संपर्क-क्रम को $u_{n+1} = f(u_n)$ के
  क्षय-घातांक से जोड़ने वाला अनुमान-नियम कहिए; (घ) $\sqrt{3/n}$ के अचर $3$
  को पाइपलाइन $\frac16 \to \frac13 \to 3$ में पीछे तक खोजिए।
\end{enumerate}
\end{problem}
