\chapter{सीमाएँ और सांतत्य}\label{ch:b1:continuity}

मध्यवर्ती मान प्रमेय और चरम मान प्रमेय उच्चतर माध्यमिक स्तर पर दृश्य विश्वास
के भरोसे प्रयुक्त हुई थीं। अनुक्रम (\cref{ch:b1:seq}) और $\R$ की सांस्थिति
(\cref{ch:b1:topology}) हाथ में आ जाने पर यह अध्याय उन्हें सिद्ध करता है ---
और सिद्धांत को एकदिष्ट एकैकी आच्छादन प्रमेय (जो \cref{ch:b1:functions} के
$\arcsin$, $\operatorname{arcosh}$ और उनके सजातियों को वैध ठहराती है) तथा
\omterm{def:b1:continuity:uniform}{एकसमान सांतत्य} पर हाइने की प्रमेय के साथ पूरा करता है।

आगे सर्वत्र $I$ एक \omterm{prop:b1:reals:intervals}{अंतराल} है और $f \colon I \to \R$; ``$x_0 \in \overline
I$'' सिरों पर सीमाओं की अनुमति देता है।

\section{फलनों की सीमाएँ}

\begin{definition}[किसी बिंदु पर सीमा]\label{def:b1:continuity:limit}
मान लीजिए $x_0 \in \overline{I}$ और $\ell \in \R$। तब $x \to x_0$ होने पर
$f(x) \to \ell$ तब कहते हैं जब\index{सीमा!फलन की}
\[
\forall \varepsilon > 0,\ \exists \delta > 0,\ \forall x \in I,
\qquad \abs{x - x_0} \leq \delta \implies \abs{f(x) - \ell} \leq
\varepsilon .
\]
$\pm\infty$ पर सीमाएँ और अनंत सीमाएँ उसी प्रतिरूप से परिभाषित होती हैं
($\abs{x - x_0} \leq \delta$ के स्थान पर $x \geq M$; $\abs{f(x) - \ell} \leq
\varepsilon$ के स्थान पर $f(x) \geq M'$)। एकपक्षीय सीमाएँ $x$ को $x > x_0$
तक (लिखा जाता है $x \to x_0^+$) या $x < x_0$ तक सीमित कर देती हैं। सीमा, जब
विद्यमान हो, अद्वितीय होती है (वही उपपत्ति जो \cref{prop:b1:seq:first} की
है)।
\end{definition}

\begin{example}[दबाव से एक सीमा]\label{ex:b1:continuity:floorlimit}
$\lim_{x \to 0} x\,\bigl\lfloor \frac1x \bigr\rfloor$ संगणित कीजिए। \omterm{thm:b1:reals:floor}{फ़र्श} का
घेराव $\frac1x - 1 < \lfloor \frac1x \rfloor \leq \frac1x$ $x$ से गुणा करने
पर (चिह्न का ध्यान रखिए!) देता है
\[
1 - x < x\Bigl\lfloor \frac1x \Bigr\rfloor \leq 1 \quad (x > 0),
\qquad
1 \leq x\Bigl\lfloor \frac1x \Bigr\rfloor < 1 - x \quad (x < 0),
\]
और दोनों एकपक्षीय दबाव $1$ पर बंद हो जाते हैं: अतः सीमा $1$ है। ध्यान दीजिए
कि अभी क्या हुआ: अकेला $\lfloor \frac1x\rfloor$ $0$ के पास जंगली छलाँगें
लगाता है, पर गुणक $x$ प्रत्येक छलाँग को वश में कर लेता है ($x$ गुना इकाई
छलाँग छोटी होती है), और केवल घेराव बचा रहता है। समापन का सार: किसी छोटे गुणक
और किसी परिबद्ध-दोलन वाले गुणक के गुणनफल की सीमाएँ दबाव की समस्याएँ हैं,
संक्रिया-प्रमेय की कभी नहीं --- संक्रिया-प्रमेय को \emph{दोनों} गुणकों का
अभिसरित होना चाहिए।
\end{example}

\begin{theorem}[अनुक्रमीय अभिलक्षण]\label{thm:b1:continuity:seqchar}
$x \to x_0$ होने पर $f(x) \to \ell$ यदि और केवल यदि: $u_n \to x_0$ वाले $I$
के बिंदुओं के \emph{प्रत्येक} अनुक्रम $(u_n)$ के लिए $f(u_n) \to \ell$ हो।
\end{theorem}

\begin{proof}
($\Rightarrow$) मान लीजिए $u_n \to x_0$ और $\varepsilon > 0$। परिभाषा से
$\delta$ लीजिए, फिर ऐसा $N$ कि $n \geq N$ के लिए $\abs{u_n - x_0} \leq
\delta$: तब $N$ के आगे $\abs{f(u_n) - \ell} \leq \varepsilon$।

($\Leftarrow$) प्रतिधनात्मक रूप। यदि $f \not\to \ell$: तो कोई $\varepsilon_0
> 0$ हर $\delta$ को हरा देता है; $\delta = \frac{1}{n+1}$ चुनने पर ऐसा $u_n
\in I$ मिलता है कि $\abs{u_n - x_0} \leq \frac{1}{n+1}$ और $\abs{f(u_n) -
\ell} > \varepsilon_0$। तब $u_n \to x_0$ पर $f(u_n) \not\to \ell$।
\end{proof}

\begin{corollary}[संक्रियाएँ, संयोजन, क्रम]\label{cor:b1:continuity:operations}
सीमाओं के योग, गुणनफल, भागफल (नीचे अशून्य सीमा के साथ) वैसा ही व्यवहार करते
हैं जैसा अनुक्रमों के लिए; और यदि $x_0$ पर $f \to \ell$ तथा $\ell$ पर $g \to
m$ हो, $g$ $\ell$ के आसपास परिभाषित हो और $x_0$ के पास $g(\ell) = m$ या $f
\neq \ell$ हो, तो $x_0$ पर $g \circ f \to m$; सीमाएँ चौड़ी असमिकाएँ सुरक्षित
रखती हैं, और दबाव प्रमेय सत्य रहती है।
\end{corollary}

\begin{proof}
प्रत्येक \omterm{def:b1:logic:statement}{कथन} \cref{thm:b1:continuity:seqchar} के द्वारा अपने अनुक्रम-अनुरूप
(\cref{thm:b1:seq:operations,thm:b1:seq:order}) तक चला जाता है। संयोजन के
लिए सीधी शृंखला एक बार लिख लेना उचित है: $\varepsilon > 0$ दिया हो, तो
$\ell$ पर $g$ की सीमा ऐसा $\eta > 0$ देती है कि
\[
\abs{y - \ell} \leq \eta \implies \abs{g(y) - m} \leq
\varepsilon \quad (y \text{ के प्रांत में } g),
\]
जहाँ स्थिति $y = \ell$ इसलिए ढँक जाती है कि $g(\ell) = m$; फिर $x_0$ पर $f$
की सीमा ऐसा $\delta > 0$ देती है कि $\abs{x - x_0} \leq \delta \implies
\abs{f(x) - \ell} \leq \eta$; और शृंखला बनाने पर $\abs{x - x_0} \leq \delta$
$\abs{g(f(x)) - m} \leq \varepsilon$ दे देता है। वैकल्पिक परिकल्पना ($x_0$
के पास $f \neq \ell$) में मान $y = \ell$ $g$ को कभी दिया ही नहीं जाता, अतः
वहाँ उसका मान निरर्थक है।
\end{proof}

\begin{example}[संयोजन की शर्त क्यों है]\label{ex:b1:continuity:compositiontrap}
मान लीजिए $y \neq 0$ के लिए $g(y) = 0$ और $g(0) = 1$, और मान लीजिए $f$
तत्समक रूप से $0$ है। तब $x \to 0$ होने पर $f(x) \to 0$, और $y \to 0$ होने
पर $g(y) \to 0$; फिर भी प्रत्येक $x$ के लिए $g(f(x)) = g(0) = 1$, अतः $g
\circ f \to 1 \neq 0$। भीतरी फलन सदा वर्जित मान $\ell = 0$ \emph{पर ही} बैठा
रहता है, और $0$ पर $g$ की सीमा इस बात की उपेक्षा करती है कि $0$ पर $g$ क्या
करता है। \cref{cor:b1:continuity:operations} की शर्त --- या तो $g(\ell) = m$
(अर्थात् $\ell$ पर $g$ \omterm{def:b1:continuity:continuous}{संतत} हो), या $x_0$ के पास $f \neq \ell$ --- ठीक इसी
को रोकने के लिए है। समापन का सार: व्यवहार में हम \emph{\omterm{def:b1:continuity:continuous}{संतत}} फलनों का संयोजन
करते हैं और यह शर्त मुफ़्त मिल जाती है; वह तभी काटती है जब सीमाएँ छिद्रित
प्रतिवेशों के अनुदिश ली जाती हैं, और इसीलिए इस पुस्तक में प्रयुक्त $\lim_{x
\to x_0}$ की परिभाषा उस बिंदु को, जब वह प्रांत में हो, समेट लेती है।
\end{example}

\section{सांतत्य}

\begin{definition}\label{def:b1:continuity:continuous}
$f$ \emph{$x_0 \in I$ पर संतत}\index{सांतत्य} तब है जब $x \to x_0$ होने पर
$f(x) \to f(x_0)$; और \emph{$I$ पर संतत} तब जब वह प्रत्येक बिंदु पर संतत हो।
\cref{thm:b1:continuity:seqchar} से: $f$ $x_0$ पर संतत है तभी जब $I$ के
प्रत्येक अनुक्रम $u_n \to x_0$ के लिए $f(u_n) \to f(x_0)$ हो।
\end{definition}

\begin{example}[असांतत्यों का वर्गीकरण]\label{ex:b1:continuity:taxonomy}
किसी बिंदु पर विफल होने के तीन तरीके, बढ़ती गंभीरता के क्रम में। \emph{हटाने
योग्य}: $\R^*$ पर $f(x) = \frac{\sin x}{x}$ की $0$ पर सीमा $1$ है; $f(0) =
1$ परिभाषित कर देने से वह सुधर जाता है --- वह असांतत्य एक छेद था, कोई
विशेषता नहीं। \emph{कूद}: किसी पूर्णांक पर $\lfloor x \rfloor$ की एकपक्षीय
सीमाएँ भिन्न हैं ($n - 1$ और $n$); कोई भी मान उनमें मेल नहीं करा सकता, पर
दोनों अर्ध-सीमाएँ विद्यमान हैं। \emph{अनिवार्य}: $0$ पर $\sin\frac1x$ की कोई
एकपक्षीय सीमा है ही नहीं (\cref{exo:b1:continuity:1}) --- अर्थात् बिना ठहराव
का दोलन। एकदिष्ट फलन केवल बीच वाला प्रकार उत्पन्न कर सकते हैं (उनकी एकपक्षीय
सीमाएँ सदा विद्यमान होती हैं, क्योंकि वे \omterm{def:b1:reals:bounds}{उच्चतम} और \omterm{def:b1:reals:bounds}{निम्नतम} हैं), और इसीलिए
उनके असांतत्य \omterm{def:b1:logic:sets}{समुच्चय} अधिकतम गणनीय होते हैं --- प्रति कूद एक परिमेय संख्या।
इसके विपरीत, डार्बू की प्रमेय (\cref{exo:b1:derivative:10}) से अवकलज केवल
अंतिम प्रकार उत्पन्न कर सकते हैं: कूद वाले असांतत्य वाला फलन कभी किसी का
अवकलज नहीं होता।
\end{example}

\begin{proposition}\label{prop:b1:continuity:algebra}
\omterm{def:b1:continuity:continuous}{संतत} फलनों के योग, गुणनफल, भागफल (जहाँ परिभाषित हों) और संयोजन \omterm{def:b1:continuity:continuous}{संतत} होते
हैं। \omterm{def:b1:poly:def}{बहुपद}, परिमेय भिन्नें (अपने ध्रुवों से बाहर), $\abs{\,\cdot\,}$,
$\exp$, $\ln$, त्रिकोणमितीय और \omterm{def:b1:functions:hyperbolic}{अतिपरवलयिक फलन} तथा उनके प्रतिलोम
(\cref{ch:b1:functions}) अपने प्रांतों पर \omterm{def:b1:continuity:continuous}{संतत} हैं।
\end{proposition}

\begin{proof}
संक्रियाएँ: \cref{cor:b1:continuity:operations}। अचर और तत्समक फलन सीधे
परिभाषा से \omterm{def:b1:continuity:continuous}{संतत} हैं (तत्समक के लिए $\delta = \varepsilon$ और अचरों के लिए
कोई भी $\delta$ काम करता है); \omterm{def:b1:continuity:continuous}{संतत} फलनों के गुणनफल \omterm{def:b1:continuity:continuous}{संतत} होने से प्रत्येक
एकपद $a_k x^k$ $k$ पर आगमन से निकल आता है, और योग बहुपदों का काम पूरा कर
देते हैं; \omterm{def:b1:fractions:field}{परिमेय भिन्न} दो बहुपदों का भागफल है, जो जहाँ भी हर लुप्त न हो वहाँ
\omterm{def:b1:continuity:continuous}{संतत} है। $\abs{\,\cdot\,}$: विलोम त्रिभुज असमिका, $\bigl|\abs{f(x)} -
\abs{f(x_0)}\bigr| \leq \abs{f(x) - f(x_0)}$, अतः वही $\delta$ काम करता है।
चिरपरिचित फलनों के लिए हम \omterm{def:b1:continuity:continuous}{सांतत्य} यहाँ दिया हुआ मान लेते हैं; अवकलनीयता
(\cref{ch:b1:derivative} में सिद्ध) उससे प्रबल है।
\end{proof}

\begin{example}[संतत फलनों के अधिकतम और न्यूनतम]\label{ex:b1:continuity:maxmin}
यदि $f, g$ \omterm{def:b1:continuity:continuous}{संतत} हैं, तो $\max(f, g)$ और $\min(f, g)$ भी \omterm{def:b1:continuity:continuous}{संतत} हैं: और इसके
लिए किसी स्थिति-विश्लेषण की आवश्यकता नहीं, क्योंकि सर्वसमिकाएँ
\[
\max(f, g) = \frac{f + g + \abs{f - g}}{2},
\qquad
\min(f, g) = \frac{f + g - \abs{f - g}}{2},
\]
तथा योगों और $\abs{\,\cdot\,}$ (\cref{prop:b1:continuity:algebra}) की
\omterm{def:b1:continuity:continuous}{सांतत्य} पर्याप्त हैं। विशेष रूप से $f^+ = \max(f, 0)$ और $f^- = \max(-f, 0)$
\omterm{def:b1:continuity:continuous}{संतत} हैं और $f = f^+ - f^-$: अर्थात् श्रेणियों (\cref{ch:b1:series}) में और,
पूरे पैमाने पर, स्नातक वर्ष 3 के खंड के समाकलन सिद्धांत में प्रयुक्त
चिह्न-विभाजन नियमितता की दृष्टि से मुफ़्त पड़ता है।
\end{example}

\begin{theorem}[मध्यवर्ती मान प्रमेय]\label{thm:b1:continuity:ivt}
मान लीजिए $f$ $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} है और $f(a) \leq 0 \leq f(b)$। तब किसी
$c \in \intcc{a}{b}$ के लिए $f(c) = 0$। फलस्वरूप, किसी \omterm{prop:b1:reals:intervals}{अंतराल} पर \omterm{def:b1:continuity:continuous}{संतत} फलन
अपने किन्हीं दो मानों के बीच का हर मान लेता है: $f(I)$ एक \omterm{prop:b1:reals:intervals}{अंतराल}
है।\index{मध्यवर्ती मान प्रमेय}
\end{theorem}

\begin{proof}
द्विभाजन। $a_0 = a$, $b_0 = b$ रखिए। $f(a_n) \leq 0 \leq f(b_n)$ वाला
$\intcc{a_n}{b_n}$ दिया हो, तो मान लीजिए $m$ मध्यबिंदु है: यदि $f(m) \leq
0$, तो $\intcc{m}{b_n}$ रखिए, अन्यथा $\intcc{a_n}{m}$; चिह्न की शर्तें बनी
रहती हैं। अनुक्रम $(a_n), (b_n)$ संलग्न हैं ($b_n - a_n = \frac{b-a}{2^n}$),
जिनकी उभयनिष्ठ सीमा $c$ है (\cref{thm:b1:seq:adjacent})। \omterm{def:b1:continuity:continuous}{सांतत्य} और
\cref{thm:b1:seq:order} से: $f(c) = \lim f(a_n) \leq 0$ तथा $f(c) = \lim
f(b_n) \geq 0$, अतः $f(c) = 0$।

परिणाम के लिए: मान $f(u) < v < f(w)$ दिए हों, तो ऊपर वाली बात $u$ और $w$
सिरों वाले खंड पर $x \mapsto f(x) - v$ के लिए लगाइए; अतः $f(I)$ उत्तल है,
अर्थात् एक \omterm{prop:b1:reals:intervals}{अंतराल} (\cref{prop:b1:reals:intervals})।
\end{proof}

\begin{example}[एक समीकरण, पूरा प्रोटोकॉल]\label{ex:b1:continuity:protocol}
$\R$ पर $\eu^x = 3 - x$ हल कीजिए: अस्तित्व, अद्वितीयता, स्थान। $g(x) = \eu^x
+ x - 3$ रखिए, जो \omterm{def:b1:continuity:continuous}{संतत} है। स्थान और अस्तित्व: $g(0) = -2 < 0$ और $g(1) = \eu
- 2 > 0$, अतः मध्यवर्ती मान प्रमेय $\intoo{0}{1}$ में एक हल रोप देती है।
अद्वितीयता: $g$ कठोरतः वर्धमान $\eu^x$ और $x - 3$ का योग है, अतः $\R$ पर
कठोरतः वर्धमान; और कठोरतः एकदिष्ट फलन हर मान अधिक से अधिक एक बार लेता है,
इसलिए हल पूरे $\R$ पर अद्वितीय है (केवल जाँचे गए \omterm{prop:b1:reals:intervals}{अंतराल} में नहीं)। यह
प्रोटोकॉल --- $g = 0$ के रूप में पुनर्व्यवस्था, अस्तित्व के लिए
चिह्न-परिवर्तन, अद्वितीयता के लिए एकदिष्टता --- ``कितने हल'' वाले अधिकांश
प्रश्न तीन पंक्तियों में निपटा देता है, और \cref{ch:b1:derivative} की विचरण
सारणियाँ $\R$ को एकदिष्ट शाखाओं में काटकर उसे अ-एकदिष्ट $g$ तक बढ़ा देती
हैं।
\end{example}

\begin{example}[कलनविधि के रूप में द्विभाजन]\label{ex:b1:continuity:dichotomy}
\cref{thm:b1:continuity:ivt} की उपपत्ति संगणना करती है। $f(x) = x^3 + x - 1$
लीजिए: $f(0) = -1 < 0 < 1 = f(1)$, अतः कोई मूल $\intoo{0}{1}$ में है। आधा
करने पर:
\[
f(0.5) = -0.375 < 0, \qquad
f(0.75) = 0.171875 > 0, \qquad
f(0.625) = -0.130859375 < 0 ,
\]
अतः मूल क्रमशः $\intoo{0.5}{1}$ में, फिर $\intoo{0.5}{0.75}$ में, फिर
$\intoo{0.625}{0.75}$ में फँसता जाता है (सच्चा मान: $c \approx 0.6823$)। $n$
पदों के बाद त्रुटि अधिक से अधिक $\frac{b - a}{2^n}$ है: दस पद तीन दशमलव देते
हैं, बीस पद छह। समापन का सार: मध्यवर्ती मान प्रमेय केवल अस्तित्व का \omterm{def:b1:logic:statement}{कथन} नहीं
है --- उसकी द्विभाजन उपपत्ति एक गारंटीशुदा, यद्यपि धीमी, मूल-खोज कलनविधि है,
जिसके सामने \cref{ch:b1:derivative} की तेज़ पर स्थानीय न्यूटन विधि को नापना
चाहिए।
\end{example}

\begin{theorem}[चरम मान प्रमेय]\label{thm:b1:continuity:evt}
खंड $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} फलन परिबद्ध होता है और अपने परिबंध प्राप्त करता
है: ऐसे $c, d \in \intcc{a}{b}$ हैं कि
\[
f(c) = \inf_{\intcc{a}{b}} f,
\qquad
f(d) = \sup_{\intcc{a}{b}} f .
\]
\cref{thm:b1:continuity:ivt} के साथ मिलाकर: \emph{खंड का \omterm{def:b1:continuity:continuous}{संतत} प्रतिबिंब खंड
होता है} $\intcc{f(c)}{f(d)}$। \index{चरम मान प्रमेय}
\end{theorem}

\begin{proof}
\emph{ऊपर से परिबद्ध:} अन्यथा ऐसा $u_n$ चुनिए कि $f(u_n) \geq n$। खंड की
संहतता (\cref{thm:b1:topology:compact}) से कोई \omterm{def:b1:seq:subsequence}{उपानुक्रम} $u_{\varphi(n)} \to
x \in \intcc{a}{b}$; \omterm{def:b1:continuity:continuous}{सांतत्य} $f(u_{\varphi(n)}) \to f(x)$ देती है, पर
$f(u_{\varphi(n)}) \geq \varphi(n) \to +\infty$: विरोधाभास।

\emph{\omterm{def:b1:reals:bounds}{उच्चतम} प्राप्त होता है:} मान लीजिए $M = \sup f$ और ऐसे $v_n$ चुनिए कि
$f(v_n) > M - \frac{1}{n+1}$ (\cref{prop:b1:reals:epsilon})। $v_{\varphi(n)}
\to d \in \intcc{a}{b}$ निकालिए: तब दबाव से $f(d) = \lim f(v_{\varphi(n)}) =
M$। \omterm{def:b1:reals:bounds}{निम्नतम} $-f$ से सँभल जाता है।
\end{proof}

\begin{example}[खंड पर धनात्मक न्यूनतम]\label{ex:b1:continuity:posmin}
मान लीजिए $f$ $\intcc{0}{1}$ पर \omterm{def:b1:continuity:continuous}{संतत} है और प्रत्येक $x$ के लिए $f(x) > 0$।
तब $\inf f > 0$: चरम मान प्रमेय से \omterm{def:b1:reals:bounds}{निम्नतम} एक \emph{मान} $f(c)$ है, और
परिकल्पना से $f(c) > 0$। अतः खंड पर \omterm{def:b1:continuity:continuous}{संतत} धनात्मक फलन $0$ से दूर परिबद्ध रहता
है --- यह दो पंक्ति का तर्क आगे के अध्यायों में दर्जन भर बार प्रयुक्त होता
है (हर नियंत्रण में, सोपान-फलन के घेराव, त्रुटि के परिबंध)। असंहत \omterm{prop:b1:reals:intervals}{अंतराल} पर
यह शानदार ढंग से विफल हो जाता है: $\intoc{0}{1}$ पर $f(x) = x$ \omterm{def:b1:continuity:continuous}{संतत} और
धनात्मक है, $\inf f = 0$ के साथ, जो प्राप्त नहीं होता। समापन का सार:
``धनात्मक'' ठीक तब ``एकसमान रूप से धनात्मक'' बन जाता है जब प्रांत संहत हो;
चरम मान प्रमेय की हर परिकल्पना भार ढो रही है।
\end{example}

\begin{remark}[तीनों प्रमेयों के इर्द-गिर्द सामान्य भूलें]
(क) \emph{\omterm{def:b1:continuity:continuous}{संतत} प्रतिबिंब}: केवल \emph{खंड} ही दृढ़ हैं। \omterm{def:b1:topology:open}{विवृत} \omterm{prop:b1:reals:intervals}{अंतराल} का \omterm{def:b1:continuity:continuous}{संतत}
प्रतिबिंब \omterm{def:b1:topology:open}{विवृत} होना आवश्यक नहीं ($x \mapsto x^2$ $\intoo{-1}{1}$ को
$\intco{0}{1}$ पर भेजता है), और \omterm{def:b1:topology:closed}{संवृत समुच्चय} का प्रतिबिंब \omterm{def:b1:topology:closed}{संवृत} होना आवश्यक
नहीं ($\arctan$ \omterm{def:b1:topology:closed}{संवृत} $\R$ को \omterm{def:b1:topology:open}{विवृत} $\intoo{-\frac\pi2}{\frac\pi2}$ पर भेजता
है); पर खंड का प्रतिबिंब खंड होता है (\cref{thm:b1:continuity:evt})। (ख)
\emph{मध्यवर्ती मान प्रमेय को \omterm{prop:b1:reals:intervals}{अंतराल} चाहिए}: $\intco{-1}{0} \cup
\intoc{0}{1}$ पर \omterm{def:b1:continuity:continuous}{संतत} फलन $x \mapsto \frac1x$ मान $-1$ और $1$ लेता है फिर भी
कभी लुप्त नहीं होता --- उसका प्रांत दो असंयुक्त टुकड़े हैं, और मान $0$ उस
\omterm{prop:b1:reals:intervals}{अंतराल} में गिर जाता है; जिस \omterm{prop:b1:reals:intervals}{अंतराल} पर प्रमेय लगाई जा रही है उसका नाम सदा
लीजिए। (ग) \emph{\omterm{def:b1:continuity:uniform}{एकसमान सांतत्य} (फलन, \omterm{def:b1:logic:sets}{समुच्चय}) युग्म का गुणधर्म है}: $x^2$
हर खंड पर \omterm{def:b1:continuity:uniform}{एकसमान रूप से संतत} है पर $\R$ पर नहीं
(\cref{ex:b1:continuity:notuniform}) --- प्रांत बताए बिना ``\omterm{def:b1:continuity:uniform}{एकसमान रूप से
संतत}'' शब्द निरर्थक हैं। (घ) \emph{प्रतिलोम की \omterm{def:b1:continuity:continuous}{सांतत्य} औपचारिक नहीं है}: वह
एकदिष्टता के द्वारा अंतरालों पर सत्य है
(\cref{thm:b1:continuity:bijection}), पर अंतरालों के सम्मिलनों के बीच कोई
\omterm{def:b1:continuity:continuous}{संतत} एकैकी आच्छादन असंतत प्रतिलोम रख सकता है --- उस प्रमेय के बाद वाली
टिप्पणी इसीलिए है कि विद्यार्थी उसे \omterm{prop:b1:reals:intervals}{अंतराल} की परिकल्पना के बिना उद्धृत करते
हैं।
\end{remark}

\section{एकदिष्ट फलन और प्रतिलोम फलन}

\begin{theorem}[एकदिष्ट एकैकी आच्छादन प्रमेय]\label{thm:b1:continuity:bijection}
मान लीजिए $f$ किसी \omterm{prop:b1:reals:intervals}{अंतराल} $I$ पर \omterm{def:b1:continuity:continuous}{संतत} और कठोरतः एकदिष्ट है। तब
\begin{enumerate}
  \item $f$ $I$ से \omterm{prop:b1:reals:intervals}{अंतराल} $J = f(I)$ पर एकैकी आच्छादन है;
  \item प्रतिलोम $f^{-1} \colon J \to I$ कठोरतः एकदिष्ट (उसी दिशा में) और
  \emph{\omterm{def:b1:continuity:continuous}{संतत}} है।
\end{enumerate}
\end{theorem}

\begin{proof}
मान लीजिए $f$ कठोरतः वर्धमान है। (1) एकैकीयता कठोर एकदिष्टता से तुरंत मिलती
है; $f(I)$ पर आच्छादकता तुच्छ है, और \cref{thm:b1:continuity:ivt} से $f(I)$
एक \omterm{prop:b1:reals:intervals}{अंतराल} है।

(2) $f^{-1}$ कठोरतः वर्धमान है: यदि $J$ में $y < y'$ पर $f^{-1}(y) \geq
f^{-1}(y')$, तो वर्धमान $f$ लगाने पर $y \geq y'$ मिलता, जो असंगत है। $y_0 =
f(x_0) \in J$ पर $f^{-1}$ की \omterm{def:b1:continuity:continuous}{सांतत्य}: मान लीजिए $\varepsilon > 0$। पहले मान
लीजिए $x_0$ $I$ का आंतरिक बिंदु है, और $\varepsilon$ को इतना सिकोड़िए कि
$x_0 \pm \varepsilon \in I$: उनके प्रतिबिंब $f(x_0 - \varepsilon) < y_0 <
f(x_0 + \varepsilon)$ संतुष्ट करते हैं। $\delta = \min\bigl(y_0 - f(x_0 -
\varepsilon),\, f(x_0 + \varepsilon) - y_0\bigr) > 0$ लीजिए: $\abs{y - y_0}
\leq \delta$ के लिए $f^{-1}$ की एकदिष्टता $f^{-1}(y)$ को $x_0 - \varepsilon$
और $x_0 + \varepsilon$ के बीच दबा देती है। यदि $x_0$, मान लीजिए, $I$ का
बायाँ सिरा है, तो केवल $x_0 + \varepsilon$ उपलब्ध है: तब $y_0 = \min J$
(एकदिष्टता), $y - y_0 \leq f(x_0 + \varepsilon) - y_0$ वाला प्रत्येक $y \in
J$ $x_0 \leq f^{-1}(y) \leq x_0 + \varepsilon$ संतुष्ट करता है, और यह
एकपक्षीय आकलन ही सिरे पर \omterm{def:b1:continuity:continuous}{सांतत्य} है; दायाँ सिरा सममित है। (ध्यान दीजिए:
$f^{-1}$ की \omterm{def:b1:continuity:continuous}{सांतत्य} $f$ की \omterm{def:b1:continuity:continuous}{सांतत्य} से सममिति द्वारा \emph{नहीं} निकाली गई
--- यह काम \omterm{prop:b1:reals:intervals}{अंतराल} पर एकदिष्टता करती है।)
\end{proof}

\begin{remark}
यही प्रमेय \cref{def:b1:functions:arc} और \cref{prop:b1:functions:invhyp} ने
चुपचाप प्रयोग की थी: $\arcsin$, $\arccos$, $\arctan$,
$\operatorname{arsinh}$, \dots\ \omterm{def:b1:continuity:continuous}{संतत} हैं। एक पूरक
(\cref{exo:b1:continuity:10}): \omterm{prop:b1:reals:intervals}{अंतराल} पर \omterm{def:b1:continuity:continuous}{संतत} \emph{\omterm{def:b1:logic:inj}{एकैकी}} फलन स्वतः कठोरतः
एकदिष्ट होता है, अतः एकदिष्टता की परिकल्पना कुछ भी महँगी नहीं पड़ती।
\end{remark}

\begin{example}[सभी कोटियों के मूल]\label{ex:b1:continuity:nthroot}
$n \in \N^*$ के लिए फलन $f(x) = x^n$ $\intco{0}{+\infty}$ पर \omterm{def:b1:continuity:continuous}{संतत} और कठोरतः
वर्धमान है, जहाँ $f(0) = 0$ और $f(x) \to +\infty$: उसका प्रतिबिंब
$\intco{0}{+\infty}$ है (अंतराल-संरचना के लिए \cref{thm:b1:continuity:ivt})।
तब एकदिष्ट एकैकी आच्छादन प्रमेय एक ही चाल में कठोरतः वर्धमान \emph{\omterm{def:b1:continuity:continuous}{संतत}}
प्रतिलोम दे देती है
\[
x \mapsto x^{1/n} \colon \intco{0}{+\infty} \to
\intco{0}{+\infty} :
\]
अर्थात् $n$-वें मूलों का अस्तित्व, अद्वितीयता और \omterm{def:b1:continuity:continuous}{सांतत्य}, और वह भी बिना किसी
संगणना के। \cref{exo:b1:reals:12} से तुलना कीजिए, जिसने $\sqrt y$ को \omterm{def:b1:reals:bounds}{उच्चतम}
से हाथ से बनाया था: एक अध्याय के सिद्धांत ने उस पूरे पृष्ठ के श्रम को दो
पंक्तियों में सिकोड़ दिया, और उन्हीं दो पंक्तियों ने उससे पहले $\arcsin$,
$\arctan$ और $\operatorname{arsinh}$ को भी वैध ठहराया। समापन का सार: अच्छी
प्रमेय संचित श्रम है।
\end{example}

\section{एकसमान सांतत्य}

\begin{definition}\label{def:b1:continuity:uniform}
$f \colon I \to \R$ \emph{एकसमान रूप से संतत}\index{एकसमान सांतत्य} तब है जब
\[
\forall \varepsilon > 0,\ \exists \delta > 0,\ \forall x, y \in I,
\qquad \abs{x - y} \leq \delta \implies \abs{f(x) - f(y)} \leq
\varepsilon .
\]
मुख्य बात: $\delta$ केवल $\varepsilon$ पर निर्भर करता है, $I$ में स्थान पर
नहीं। एकसमान \omterm{def:b1:continuity:continuous}{सांतत्य} से \omterm{def:b1:continuity:continuous}{सांतत्य} आती है; और \emph{लिप्शिट्ज़} फलन ($\abs{f(x)
- f(y)} \leq k \abs{x - y}$) एकसमान रूप से \omterm{def:b1:continuity:continuous}{संतत} होता है ($\delta =
\varepsilon/k$)।
\end{definition}

\begin{example}\label{ex:b1:continuity:notuniform}
$x \mapsto x^2$ $\R$ पर \omterm{def:b1:continuity:continuous}{संतत} है पर \omterm{def:b1:continuity:uniform}{एकसमान रूप से संतत} \emph{नहीं}: $\abs{(n
+ \frac 1n)^2 - n^2} = 2 + \frac{1}{n^2} \geq 2$, यद्यपि प्रांतिक $\frac 1n
\to 0$ दूरी पर हैं। किसी भी परिबद्ध \omterm{prop:b1:reals:intervals}{अंतराल} पर वह लिप्शिट्ज़ है, अतः \omterm{def:b1:continuity:uniform}{एकसमान
रूप से संतत} --- जो नीचे हाइने की प्रमेय के अनुरूप है।
\end{example}

\begin{example}[एकसमान मापांक, स्पष्ट रूप से]\label{ex:b1:continuity:moduli}
खंड पर हाइने एकसमान $\delta$ की गारंटी देता है; प्रायः उसे \emph{संगणित} भी
किया जा सकता है। $\intcc{0}{10}$ पर $f(x) = x^2$ के लिए:
\[
\abs{x^2 - y^2} = \abs{x + y}\,\abs{x - y} \leq 20\,\abs{x - y},
\]
अतः $\delta = \frac{\varepsilon}{20}$ एकसमान रूप से काम करता है (एक
लिप्शिट्ज़ \omterm{def:b1:complex:field}{मापांक}, $\varepsilon$ में रैखिक)। $\intcc{0}{1}$ पर $\sqrt x$ के
लिए: $\abs{\sqrt x - \sqrt y} \leq \sqrt{\abs{x - y}}$
(\cref{exo:b1:continuity:9}), अतः $\delta = \varepsilon^2$ काम करता है ---
एकसमान, पर \emph{रैखिक नहीं}: $0$ के पास वर्गमूल खड़ा है, और उसकी क़ीमत
$\varepsilon$ के घातांक में दिखाई देती है, एकसमानता की विफलता में नहीं।
समापन का सार: \omterm{def:b1:continuity:uniform}{एकसमान सांतत्य} कोई हाँ/ना नहीं, एक पूरा वर्णक्रम है --- फलन
$\delta(\varepsilon)$, जिसे \omterm{def:b1:complex:field}{मापांक} कहते हैं, नापता है कि एकसमानता कितनी
महँगी है, और लिप्शिट्ज़ उसका सर्वोत्तम श्रेणी-अंक भर है।
\end{example}

\begin{theorem}[हाइने]\label{thm:b1:continuity:heine}
किसी \emph{खंड} $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} फलन \omterm{def:b1:continuity:uniform}{एकसमान रूप से संतत} होता
है।\index{हाइने की प्रमेय}
\end{theorem}

\begin{proof}
विरोधाभास से: मान लीजिए कोई $\varepsilon_0 > 0$ हर $\delta$ को हरा देता है।
$\delta = \frac{1}{n+1}$ के साथ ऐसे $x_n, y_n \in \intcc{a}{b}$ चुनिए कि
$\abs{x_n - y_n} \leq \frac{1}{n+1}$ और $\abs{f(x_n) - f(y_n)} >
\varepsilon_0$। संहतता (\cref{thm:b1:topology:compact}) $x_{\varphi(n)} \to
c \in \intcc{a}{b}$ निकाल देती है; तब $y_{\varphi(n)} \to c$ भी ($\abs{x -
y}$ पर दबाव)। $c$ पर \omterm{def:b1:continuity:continuous}{सांतत्य} $f(x_{\varphi(n)}) \to f(c)$ और
$f(y_{\varphi(n)}) \to f(c)$ देती है, अतः $\abs{f(x_{\varphi(n)}) -
f(y_{\varphi(n)})} \to 0 < \varepsilon_0$: विरोधाभास।
\end{proof}

\begin{example}[परिबद्ध, संतत, फिर भी एकसमान नहीं]\label{ex:b1:continuity:sinx2}
फलन $f(x) = \sin(x^2)$ $\R$ पर \omterm{def:b1:continuity:continuous}{संतत} और परिबद्ध है, पर \omterm{def:b1:continuity:uniform}{एकसमान रूप से संतत}
नहीं। लीजिए
\[
x_n = \sqrt{2\pi n}, \qquad y_n = \sqrt{2\pi n + \tfrac\pi2}:
\qquad
y_n - x_n = \frac{\pi/2}{\sqrt{2\pi n + \frac\pi2} +
\sqrt{2\pi n}} \longrightarrow 0 ,
\]
फिर भी प्रत्येक $n$ के लिए $f(y_n) - f(x_n) = \sin\bigl(2\pi n +
\frac\pi2\bigr) - \sin(2\pi n) = 1 - 0 = 1$: कोई एक $\delta$ $\varepsilon =
\frac12$ के लिए सर्वत्र काम नहीं कर सकता। ज्यामितीय रूप से, $\sin(x^2)$ के
दोलन \emph{त्वरित} होते जाते हैं: आलेख निरंतर छोटी होती खिड़कियों पर पूरी
तरंग पूरी कर लेता है, अतः दिए हुए $\varepsilon$ के लिए आवश्यक क्षैतिज पैमाना
$x$ के बढ़ने के साथ सिकुड़कर शून्य हो जाता है। समापन का सार: परिबद्धता
एकसमानता नहीं ख़रीदती (यही उदाहरण), और अपरिबद्धता उसे रोकती नहीं ($\sqrt x$,
\cref{exo:b1:continuity:9}); निर्णय \emph{दोलन का \omterm{def:b1:complex:field}{मापांक}} करता है, और हाइने
की प्रमेय कहती है कि संहत प्रांत उसे स्वतः अनुशासित कर देते हैं।
\end{example}

\begin{example}[कैलकुलेटर का प्रयोग, समझाया हुआ]\label{ex:b1:continuity:cositeration}
कैलकुलेटर में कोई भी संख्या टाइप कीजिए और बार-बार $\cos$ दबाइए: प्रदर्शन
$0.7390851\dots$ पर जम जाता है। क्यों? एक बार दबाने के बाद मान
$\intcc{-1}{1}$ में है, दो बार के बाद $\intcc{\cos 1}{1} \subseteq
\intcc{0.54}{1}$ में, जो $\cos$ के लिए स्थायी \omterm{prop:b1:reals:intervals}{अंतराल} है। उस पर $\abs{\cos a
- \cos b} \leq \sin(1)\,\abs{a - b}$ के साथ $\sin 1 = 0.841\dots < 1$
(\cref{pb:b1:seq:1} का गुणनफल-से-योग परिबंध, अथवा \cref{ch:b1:derivative} की
माध्य मान असमिका): अर्थात् पुनरावृत्ति संकुचन करती है, अतः
\cref{met:b1:seq:recurrent} के त्रुटि-नियंत्रण पद से,
\[
\abs{u_n - c} \leq (0.842)^{\,n-2}\,\abs{u_2 - c}
\longrightarrow 0 ,
\]
जहाँ $c$ अद्वितीय अचल बिंदु $\cos c = c$ है (\cref{exo:b1:continuity:6})।
लगभग $40$ बार दबाने से तीन दशमलव मिल जाते हैं ($0.842^{40} \approx 10^{-3}$)
--- अर्थात् गुणोत्तर दर, जो द्विभाजन की प्रति पद $2^{-n}$ से मंद है, पर हर
बार दबाने में एक ही कुंजी लगती है जबकि द्विभाजन के हर पद में पूरा
चिह्न-मूल्यांकन। समापन का सार: \cref{ch:b1:seq} का अचल-बिंदु चित्र और इस
अध्याय की अस्तित्व-प्रमेयें एक ही कथा के दो आधे हैं --- मध्यवर्ती मान प्रमेय
$c$ ढूँढ़ लेती है, और संकुचन उस तक पहुँच जाता है।
\end{example}

\begin{remark}[ये प्रमेयें आगे कहाँ काम करती हैं]
इस अध्याय के तीनों स्तंभ आगे एक-एक अध्याय को शक्ति देते हैं। मध्यवर्ती मान
प्रमेय हलों के अस्तित्व के हर तर्क को और एकदिष्ट एकैकी आच्छादन प्रमेय को
पोषित करती है; चरम मान प्रमेय इष्टतमीकरण की समस्याओं को प्रमेयों में बदल
देती है (\cref{ch:b1:derivative} में रोल और माध्य मान प्रमेय ठीक वहीं से
आरंभ होती हैं); और हाइने की प्रमेय ही वह कारण है कि खंडों पर \omterm{def:b1:continuity:continuous}{संतत} फलन
\cref{ch:b1:integration} में समाकलित किए जा सकते हैं --- एकसमान $\delta$ ही
रीमान योगों को अभिसरित कराता है। स्नातक वर्ष 2 के खंड में यही तिकड़ी मानकित
सदिश समष्टियों में लौटती है, जहाँ वह काम संहतता करती है जो यहाँ खंड करते
हैं।
\end{remark}

\begin{remark}[इस खंड के भीतर आगे की दृष्टि]
\omterm{def:b1:continuity:continuous}{सांतत्य} का दर्जा अब घटने वाला है, पर वह कभी सेवानिवृत्त नहीं होती।
\Cref{ch:b1:derivative} उसे अवकलनीयता तक प्रबल कर देता है और बदले में उपकार
भी करता है (अवकलनीय होने से \omterm{def:b1:continuity:continuous}{संतत} होना निकलता है); \cref{ch:b1:integration}
उस पर दो बार टिकता है, हाइने के द्वारा निर्माण में और मूल प्रमेय के द्वारा,
जिसकी केंद्रीय वस्तु $x \mapsto \int_a^x f$ मात्र \omterm{def:b1:continuity:continuous}{संतत} $f$ को $C^1$
प्रतिअवकलज तक उठा देती है। \cref{ch:b1:multivar} में दो चरों की \omterm{def:b1:continuity:continuous}{सांतत्य} में
एक जाल है जिसकी झलक अभी देख लेनी चाहिए: फलन $\frac{xy}{x^2 + y^2}$ ($0$ से
विस्तारित) प्रत्येक नियत $y$ के लिए $x$ में और प्रत्येक नियत $x$ के लिए $y$
में \omterm{def:b1:continuity:continuous}{संतत} है, फिर भी मूलबिंदु पर \omterm{def:b1:continuity:continuous}{संतत} नहीं है --- विकर्ण $x = y$ के अनुदिश वह
सदा $\frac12$ रहता है। पृथक् \omterm{def:b1:continuity:continuous}{सांतत्य} \omterm{def:b1:continuity:continuous}{सांतत्य} से कठोरतः दुर्बल है: अनुक्रमीय
अभिलक्षण $\R^2$ तक चला तो जाता है, पर अनुक्रमों को \emph{प्रत्येक} दिशा से
पास आने की अनुमति होनी चाहिए, केवल अक्षों के अनुदिश नहीं।
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:8}
मान लीजिए $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{संतत} और आवर्ती है (आवर्तकाल $T > 0$)। सिद्ध

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:9}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:continuity:10}
मान लीजिए $f$ किसी \omterm{prop:b1:reals:intervals}{अंतराल} $I$ पर \omterm{def:b1:continuity:continuous}{संतत} और \omterm{def:b1:logic:inj}{एकैकी} है। सिद्ध कीजिए कि $f$ कठोरतः
एकदिष्ट है। \emph{संकेत: यदि नहीं, तो ऐसे $a < b < c$ हैं कि, मान लीजिए,
$f(b) > f(a)$ और $f(b) > f(c)$; $\max(f(a), f(c))$ और $f(b)$ के बीच के किसी
मान पर $b$ के दोनों ओर मध्यवर्ती मान प्रमेय लगाइए।}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:continuity:11}

\omterm{def:b1:continuity:continuous}{सांतत्य} और \omterm{def:b1:topology:dense}{सघनता} से (\cref{thm:b1:reals:density})।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:continuity:12}

$\intcc{0}{A+1}$ पर हाइने, और $A$ से आगे सीमा; दोनों क्षेत्रों को अतिव्यापी
बनाइए।)}
\end{exercise}

\section{समस्या: कोशी का फलनीय समीकरण और उसकी बहनें}

\begin{problem}[{सप्ताहांत समस्या --- $f(x + y) = f(x) + f(y)$: नियमितता रैखिकता पर बाध्य कर देती है, और राक्षसों
का चित्र}]
\label{pb:b1:continuity:1}
कौन-से फलन सभी वास्तविक $x, y$ के लिए $f(x + y) = f(x) + f(y)$ संतुष्ट करते
हैं? कोशी ने यह प्रश्न 1821 में उठाया था; उत्तर एक आदर्श है।
\Cref{exo:b1:continuity:11} दिखाता है कि ऐसा \emph{योज्य} फलन $\Q$ पर रैखिक
होता है और पूरी \omterm{def:b1:continuity:continuous}{सांतत्य} $f(x) = cx$ पर बाध्य कर देती है। यह समस्या परिकल्पना
को नाटकीय रूप से तीक्ष्ण करती है --- \emph{एक ही} बिंदु पर \omterm{def:b1:continuity:continuous}{सांतत्य}, अथवा
एकदिष्टता, अथवा किसी एक छोटे \omterm{prop:b1:reals:intervals}{अंतराल} पर मात्र परिबद्धता, इनमें से हर एक
पर्याप्त है --- फिर किसी काल्पनिक अरैखिक हल का चित्र खींचती है (उसका आलेख
समतल भर देता है), $\eu^{cx}$, $c\ln x$, $x^c$ और $cx^2$ को अभिलक्षित करने
वाली बहन-समीकरण हल करती है, और येंसन के समीकरण तथा प्रमेय
\emph{मध्यबिंदु-उत्तल $+$ \omterm{def:b1:continuity:continuous}{संतत} $\implies$ उत्तल} पर समाप्त होती
है।\index{कोशी का फलनीय समीकरण}\index{फलनीय समीकरण} आगे सर्वत्र \emph{योज्य}
का अर्थ है: सभी $x, y \in \R$ के लिए $f(x + y) = f(x) + f(y)$।

\medskip \noindent\textbf{भाग I --- $\Q$-रैखिकता, और \omterm{def:b1:continuity:continuous}{सांतत्य} का एक बिंदु।}
\begin{enumerate}
  \item मान लीजिए $f$ योज्य है। \cref{exo:b1:continuity:11} से परिमेय $r$ के
  लिए $f(r) = f(1)\,r$। नीचे प्रयुक्त होने वाला सूक्ष्मतर \omterm{def:b1:logic:statement}{कथन} सिद्ध कीजिए:
  \emph{प्रत्येक} $x \in \R$ और $r \in \Q$ के लिए $f(rx) = r\,f(x)$ ($f$
  $\Q$-\emph{रैखिक} है)।
  \item मान लीजिए योज्य $f$ किसी एक ही बिंदु $x_0$ पर \omterm{def:b1:continuity:continuous}{संतत} है। दिखाइए कि $f$
  सर्वत्र \omterm{def:b1:continuity:continuous}{संतत} है \emph{($f(x + h) - f(x)$ को $f(x_0 + h) - f(x_0)$ के पदों
  में संगणित कीजिए)}, अतः $f(x) = f(1)\,x$।
  \item दिखाइए कि योज्य फलन किसी भी \omterm{def:b1:topology:dense}{सघन} \omterm{def:b1:structures:subgroup}{उपसमूह} पर अपने प्रतिबंधन से
  निर्धारित हो जाता है: यदि दो योज्य फलन $\Z + \sqrt2\,\Z$ पर मेल खाते हों
  और दोनों \omterm{def:b1:continuity:continuous}{संतत} हों, तो वे बराबर हैं --- जबकि \omterm{def:b1:continuity:continuous}{सांतत्य} के बिना $f(1) = 0$ और
  $f(\sqrt 2) = 1$ निर्धारित कर देना उस \omterm{def:b1:structures:subgroup}{उपसमूह} पर $\Q$-रैखिकता के अनुरूप है।
  इस निर्धारण के लिए $f(m + n\sqrt2)$ संगणित कीजिए।
  \item मान लीजिए $f$ योज्य है और $a < b$ वाले किसी \omterm{prop:b1:reals:intervals}{अंतराल} $\intcc{a}{b}$ पर
  ऊपर से $M$ द्वारा परिबद्ध है। दिखाइए कि $f$ $\intcc{0}{\ell}$, $\ell = b -
  a$ पर ऊपर से परिबद्ध है \emph{($a$ से स्थानांतरण कीजिए)}।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- नियमितता की सीढ़ी।}
\begin{enumerate}[resume]
  \item प्रश्न 4 को आगे बढ़ाते हुए: $f(\ell - t) + f(t) = f(\ell)$ का प्रयोग
  करके दिखाइए कि $f$ $\intcc{0}{\ell}$ पर नीचे से भी परिबद्ध है: वहाँ $\abs
  f \leq C$।
  \item दिखाइए कि $t \in \intcc{0}{\ell/n}$ के लिए $\abs{f(t)} \leq
  \frac{C}{n}$, और उससे निकालिए कि $f$ $0$ पर \omterm{def:b1:continuity:continuous}{संतत} है \emph{(विषमता बाईं ओर
  सँभाल लेती है)}, अतः सर्वत्र (प्रश्न 2): अर्थात् किसी एक \omterm{prop:b1:reals:intervals}{अंतराल} पर परिबद्ध
  योज्य फलन रैखिक होता है।
  \item एकदिष्ट स्थिति निकालिए: किसी $\intcc{a}{b}$ ($a < b$) पर अह्रासमान
  योज्य फलन $c \geq 0$ के साथ $f(x) = cx$ है।
  \item \emph{नियमितता की सीढ़ी} जोड़िए: योज्य $f$ के लिए ये तुल्य हैं ---
  (क) $f(x) = cx$; (ख) $f$ \omterm{def:b1:continuity:continuous}{संतत}; (ग) $f$ किसी एक बिंदु पर \omterm{def:b1:continuity:continuous}{संतत}; (घ) $f$ किसी
  अनपकर्षी \omterm{prop:b1:reals:intervals}{अंतराल} पर एकदिष्ट; (ङ) $f$ किसी अनपकर्षी \omterm{prop:b1:reals:intervals}{अंतराल} पर परिबद्ध।
  निहितार्थों को इस प्रकार सजाइए कि हर एक या तो तुच्छ हो या पहले ही सिद्ध।
  \item सत्यापित कीजिए कि प्रश्न 4--6 ने केवल \emph{ऊपर} का परिबंध खपाया:
  अर्थात् किसी एक अनपकर्षी \omterm{prop:b1:reals:intervals}{अंतराल} पर ऊपर से परिबद्ध योज्य फलन पहले ही रैखिक
  है। नीचे के परिबंध के लिए दर्पण \omterm{def:b1:logic:statement}{कथन} निकालिए, और इस प्रकार मिले सीढ़ी के
  डंडे (ङ) का प्रबलतम रूप लिखिए।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- किसी राक्षस का चित्र।} अब मान लीजिए
$f$ योज्य है पर रैखिक \emph{नहीं}।
\begin{enumerate}[resume]
  \item दिखाइए कि ऐसी अशून्य वास्तविक संख्याएँ $u, v$ हैं कि
  $\dfrac{f(u)}{u} \neq \dfrac{f(v)}{v}$, और यह कि सदिश $(u, f(u))$ तथा $(v,
  f(v))$ समतल को फैलाते हैं (उनका सारणिक $u f(v) - v f(u)$ अशून्य है)।
  \item दिखाइए कि $f$ के आलेख में ये सभी बिंदु हैं
        \[
        r\,(u, f(u)) + s\,(v, f(v)), \qquad r, s \in \Q ,
        \]
        और उससे निकालिए कि वह आलेख \emph{$\R^2$ में \omterm{def:b1:topology:dense}{सघन}} है: समतल के
        प्रत्येक बिंदु $(x_0, y_0)$ और प्रत्येक $\varepsilon > 0$ के लिए कोई
        $(x, f(x))$ उससे $\varepsilon$ के भीतर है \emph{($2\times2$ वास्तविक
        निकाय हल कीजिए, फिर वास्तविक गुणांकों का परिमेय संख्याओं से सन्निकटन
        कीजिए)}।
  \item प्रश्न 11 से पूरा चित्र निकालिए: अरैखिक योज्य फलन प्रत्येक अनपकर्षी
  \omterm{prop:b1:reals:intervals}{अंतराल} पर अपरिबद्ध है, प्रत्येक बिंदु पर असंतत है, किसी भी \omterm{prop:b1:reals:intervals}{अंतराल} पर
  एकदिष्ट नहीं है, और किसी भी \omterm{prop:b1:reals:intervals}{अंतराल} का उसका प्रतिबिंब $\R$ में \omterm{def:b1:topology:dense}{सघन} है।
  प्रश्न 8 के साथ मेल बैठाइए।
  \item राक्षस विद्यमान हैं --- किसी \omterm{def:b1:topology:dense}{सघन} \omterm{def:b1:structures:subgroup}{उपसमूह} पर, रचनात्मक रूप से: $G = \Z
  + \sqrt2\,\Z$ पर $f(m + n\sqrt2) = n$ परिभाषित कीजिए। दिखाइए कि $f$ $G$ पर
  सुपरिभाषित और योज्य है, और यह कि प्रत्येक $\varepsilon > 0$ के लिए $f$ $G
  \cap \intoo{0}{\varepsilon}$ पर अपरिबद्ध है \emph{(नियत $N$ के लिए केवल
  परिमित कितने $g = m + n\sqrt2 \in \intoo{0}{1}$ के लिए $\abs n \leq N$;
  फिर भी $G \cap \intoo{0}{\varepsilon}$ अनंत है)}। एक अनुच्छेद में समझाइए
  कि ऐसे $f$ को पूरे $\R$ तक बढ़ाने के लिए $\R$ का $\Q$-सदिश समष्टि के रूप
  में आधार (अर्थात् \emph{हामेल आधार}) क्यों चाहिए, जिसका अस्तित्व
  चयन-अभिगृहीत का विषय है और इस खंड से परे।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- बहन-समीकरण।} यहाँ सभी फलन \omterm{def:b1:continuity:continuous}{संतत} हैं।
\begin{enumerate}[resume]
  \item मान लीजिए $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{संतत} है, तत्समक रूप से $0$ नहीं, और
  $f(x + y) = f(x)f(y)$। दिखाइए $f(x) = f\bigl(\frac x2\bigr)^2 \geq 0$, फिर
  सर्वत्र $f > 0$, फिर किसी $c$ के लिए $f(x) = \eu^{cx}$: अर्थात् चरघातांकी
  ही $(\R, +)$ से $(\R^*, \times)$ तक की \omterm{def:b1:continuity:continuous}{संतत} समाकारिताएँ हैं।
  \item मान लीजिए $f \colon \intoo{0}{+\infty} \to \R$ \omterm{def:b1:continuity:continuous}{संतत} है और $f(xy) =
  f(x) + f(y)$। दिखाइए $f(x) = c\ln x$ \emph{($\exp$ से ले जाइए)}।
  \item मान लीजिए $f \colon \intoo{0}{+\infty} \to \intoo{0}{+\infty}$ \omterm{def:b1:continuity:continuous}{संतत}
  है और $f(xy) = f(x)f(y)$। दिखाइए $f(x) = x^c$।
  \item ऐसे सभी \omterm{def:b1:continuity:continuous}{संतत} $f \colon \R \to \R$ खोजिए जिनके लिए
        \[
        f(x + y) = f(x) + f(y) + f(x)f(y) .
        \]
        \emph{($h = 1 + f$ का अध्ययन कीजिए; अपकर्षी स्थिति अलग से सँभालिए।)}
  \item (समांतर चतुर्भुज) ऐसे सभी \omterm{def:b1:continuity:continuous}{संतत} $f \colon \R \to \R$ खोजिए जिनके लिए
  $f(x + y) + f(x - y) = 2f(x) + 2f(y)$: दिखाइए कि $f$ सम है, $f(0) = 0$,
  आगमन से $f(nx) = n^2 f(x)$, फिर $f(x) = f(1)\,x^2$। (यह समीकरण द्विघात
  रूपों की पहचान है --- वही समांतर चतुर्भुज नियम, जो स्नातक वर्ष 2 के खंड
  में यह पकड़ लेता है कि कौन-से मानक किसी अंतर्गुणन से आते हैं।)
\end{enumerate}

\medskip \noindent\textbf{भाग V --- येंसन और मध्यबिंदु उत्तलता।}
\begin{enumerate}[resume]
  \item (येंसन का समीकरण) मान लीजिए $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{संतत} है और
  $f\bigl(\frac{x+y}{2}\bigr) = \frac{f(x) + f(y)}{2}$। दिखाइए कि $g = f -
  f(0)$ $g\bigl(\frac x2\bigr) = \frac{g(x)}{2}$ संतुष्ट करता है, उससे
  निकालिए कि $g$ योज्य है, और निष्कर्ष निकालिए $f(x) = cx + d$।
  \item अब केवल \emph{असमिका} मान लीजिए: $f$ \omterm{def:b1:continuity:continuous}{संतत} है और
        \[
        f\Bigl(\frac{x + y}{2}\Bigr) \leq \frac{f(x) +
        f(y)}{2} \qquad (x, y \in \R).
        \]
        $n$ पर आगमन से सिद्ध कीजिए कि सभी द्विआधारी भार $\lambda =
        \frac{k}{2^n} \in \intcc{0}{1}$ के लिए:
        \[
        f\bigl(\lambda x + (1 - \lambda)y\bigr) \leq \lambda
        f(x) + (1 - \lambda) f(y) .
        \]
  \item \omterm{def:b1:continuity:continuous}{सांतत्य} और द्विआधारी संख्याओं की \omterm{def:b1:topology:dense}{सघनता} (\cref{exo:b1:reals:8}) से
  इसे प्रत्येक $\lambda \in \intcc{0}{1}$ तक बढ़ाइए: अर्थात् \omterm{def:b1:continuity:continuous}{संतत}
  मध्यबिंदु-उत्तल फलन पूरी उत्तलता असमिका संतुष्ट करता है (यह धारणा
  \cref{ch:b1:derivative} में व्यवस्थित रूप से पढ़ी गई है)।
  \item दिखाइए कि \omterm{def:b1:continuity:continuous}{सांतत्य} छोड़ी नहीं जा सकती: कोई अरैखिक योज्य $f$ प्रश्न 19
  की \emph{मध्यबिंदु समता} तो संतुष्ट करता है पर किसी भी \omterm{prop:b1:reals:intervals}{अंतराल} पर कोई
  उत्तलता असमिका नहीं (प्रश्न 12)। उपदेश: मध्यबिंदु उत्तलता गणनीय-चरण वाला
  गुणधर्म है (द्विआधारी), और उत्तलता सातत्य वाला; \omterm{def:b1:continuity:continuous}{सांतत्य} ही सेतु है --- ठीक
  वैसे ही जैसे भाग I और II में।
\end{enumerate}

\medskip \noindent\textbf{भाग VI --- अंतिम रूपांतर और संश्लेषण।}
\begin{enumerate}[resume]
  \item ऐसे सभी \omterm{def:b1:continuity:continuous}{संतत} $f \colon \R \to \R$ खोजिए जिनके लिए $f(x + y) = f(x) +
  f(y) + xy$ \emph{(विशिष्ट हल $\frac{x^2}{2}$ घटा दीजिए)}।
  \item सिद्ध कीजिए: यदि $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{संतत} है और केवल किसी \emph{\omterm{def:b1:topology:dense}{सघन}
  \omterm{def:b1:structures:subgroup}{उपसमूह}} $G$ पर योज्य है (अर्थात् $g, g' \in G$ के लिए $f(g + g') = f(g) +
  f(g')$), तो $f$ $\R$ पर योज्य है। और अधिक व्यापक रूप से, $\R$ के किसी \omterm{def:b1:topology:dense}{सघन}
  उपसमुच्चय पर मेल खाने वाले दो \omterm{def:b1:continuity:continuous}{संतत} फलन बराबर होते हैं।
  \item संश्लेषण, एक-एक वाक्य में: (क) प्रश्न 8 की नियमितता की सीढ़ी स्मरण
  से कहिए; (ख) समझाइए कि नियमितता की विफलता के लिए ``आलेख समतल में \omterm{def:b1:topology:dense}{सघन}'' सही
  मानसिक चित्र क्यों है; (ग) भाग IV और V में अभिलक्षित पाँचों चिरपरिचित फलन
  और वह एक विधि गिनाइए जिसने उन सबको पकड़ा; (घ) वे दो स्थान बताइए जहाँ $\R$
  में $\Q$ (या द्विआधारी संख्याओं) की \omterm{def:b1:topology:dense}{सघनता} ने तर्क ढोया, और वह स्थान जहाँ
  वह ऐसा नहीं कर सकी (प्रश्न 13)।
\end{enumerate}
\end{problem}
