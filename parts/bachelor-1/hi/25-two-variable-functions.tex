\chapter{दो चरों के फलन}\label{ch:b1:multivar}

वर्ष उच्चतर विमा में पहली चहलकदमी के साथ समाप्त होता है: दो वास्तविक चरों के
फलन $f(x, y)$। सब कुछ सामान्यीकृत हो जाता है --- सीमाएँ, \omterm{def:b1:continuity:continuous}{सांतत्य}, \omterm{def:b1:derivative:def}{अवकलज},
चरम --- पर हर धारणा में एक नया मोड़ आ जाता है: किसी सीमा तक एक साथ हर दिशा
से पहुँचा जा सकता है, \omterm{def:b1:derivative:def}{अवकलज} आंशिक अवकलजों में बँट जाता है, और \omterm{def:b1:multivar:partial}{प्रवणता} चढ़ाई
की दिशा दिखा देती है। पूरा सिद्धांत (अवकल, व्यापक $\R^n$, चिकनी उपसतहें)
दूसरे वर्ष का विषय है; यहाँ हम शब्दावली और पहली ईमानदार प्रमेयें रखते हैं।

\section{समतल $\R^2$; सांतत्य}

\begin{definition}\label{def:b1:multivar:topology}
$\R^2$ पर यूक्लिडीय मानक $\norm{(x,y)} = \sqrt{x^2 + y^2}$ लीजिए
(\cref{ch:b1:euclid})। \omterm{def:b1:topology:open}{विवृत} गोले, \omterm{def:b1:topology:open}{प्रतिवेश} और $\R^2$ के \emph{\omterm{def:b1:topology:open}{विवृत}
उपसमुच्चय} ठीक वैसे ही परिभाषित होते हैं जैसे \cref{ch:b1:topology} में,
अंतरालों के स्थान पर गोलों के साथ। कोई फलन $f \colon U \to \R$ ($U
\subseteq \R^2$ \omterm{def:b1:topology:open}{विवृत}) $a \in U$ पर \emph{\omterm{def:b1:continuity:continuous}{संतत}} कहलाता है जब
\[
\forall\varepsilon > 0,\ \exists\delta > 0, \quad
\norm{X - a} \leq \delta \implies \abs{f(X) - f(a)} \leq
\varepsilon,
\]
और एक चर वाली वही अनुक्रमीय अभिलक्षणता यहाँ भी चलती है। योग, गुणनफल,
भागफल तथा एक चर वाले \omterm{def:b1:continuity:continuous}{संतत} फलनों के साथ संयोजन \omterm{def:b1:continuity:continuous}{सांतत्य} को बनाए रखते हैं;
\omterm{prop:b1:vspaces:coordinates}{निर्देशांक} \omterm{def:b1:logic:map}{प्रतिचित्रण} \omterm{def:b1:continuity:continuous}{संतत} हैं, अतः $(x,y)$ के \omterm{def:b1:poly:def}{बहुपद} भी।
\end{definition}

\begin{example}[ध्रुवीय परिबंध --- किसी सीमा को सिद्ध करने का साफ़ रास्ता]
\label{ex:b1:multivar:polarbound}
दिखाइए कि $f(x, y) = \dfrac{x^2y^2}{x^2 + y^2}$ (जहाँ $f(0,0) = 0$) मूल
बिंदु पर \omterm{def:b1:continuity:continuous}{संतत} है। ध्रुवीय निर्देशांकों $x = \rho\cos\theta$, $y =
\rho\sin\theta$ में:
\[
\abs{f} = \frac{\rho^4\cos^2\theta\sin^2\theta}{\rho^2}
= \rho^2\,(\cos\theta\sin\theta)^2 \leq \frac{\rho^2}{4}
\xrightarrow[\rho \to 0]{} 0 ,
\]
एक ऐसा परिबंध जो \emph{$\theta$ से स्वतंत्र} है: पहुँचने की दिशा चाहे कोई
भी हो, मान $0$ की ओर दबा दिए जाते हैं। $\theta$ में यही एकरूपता सारा दम है
--- $\abs g = \abs{\cos\theta\sin\theta}$ जैसा परिबंध (जिसमें $\rho$ बचता
ही नहीं) कुछ भी सिद्ध नहीं करता, और सचमुच वही $g$ अगले उदाहरण का असंतत
त्रिज्य फंदा है।
\end{example}

\begin{example}[त्रिज्य फंदा]\label{ex:b1:multivar:trap}
मान लीजिए $(x,y) \neq (0,0)$ के लिए $f(x, y) = \dfrac{xy}{x^2 + y^2}$ और
$f(0, 0) = 0$। प्रत्येक अक्ष के सहारे $f = 0 \to 0$; पर विकर्ण $y = x$ के
सहारे $f(x, x) = \frac12 \not\to 0$। \emph{मूल बिंदु पर कोई सीमा नहीं}: हर
रेखा के सहारे पहुँचना, और हर रेखा पर एक ही सीमा पा लेना भी, पर्याप्त नहीं
है (यहाँ रेखीय सीमाएँ आपस में नहीं मिलतीं; इससे बदतर उदाहरण हर रेखा पर
मिलते हैं और फिर किसी परवलय पर विफल हो जाते हैं,
\cref{exo:b1:multivar:3})। हर चर में अलग-अलग \omterm{def:b1:continuity:continuous}{सांतत्य} पूरा \omterm{def:b1:continuity:continuous}{सांतत्य} नहीं
देता।
\end{example}

\section{आंशिक अवकलज}

\begin{definition}\label{def:b1:multivar:partial}
$(a, b)$ पर $f$ के \emph{आंशिक अवकलज}\index{आंशिक अवकलज} अक्षों के सहारे
लिए गए एक चर वाले \omterm{def:b1:derivative:def}{अवकलज} हैं:
\[
\frac{\partial f}{\partial x}(a,b)
= \lim_{h \to 0} \frac{f(a + h, b) - f(a,b)}{h},
\qquad
\frac{\partial f}{\partial y}(a,b)
= \lim_{k \to 0} \frac{f(a, b + k) - f(a,b)}{k}.
\]
$f$ $U$ पर $C^1$ वर्ग का कहलाता है जब दोनों विद्यमान हों और दोनों $U$ पर
\omterm{def:b1:continuity:continuous}{संतत} हों। \emph{प्रवणता}\index{प्रवणता} $\nabla f(a,b) =
\bigl(\frac{\partial f}{\partial x},\, \frac{\partial f}{\partial
y}\bigr)(a,b)$ है।
\end{definition}

\begin{theorem}[$C^1$ से स्पर्श समतल]\label{thm:b1:multivar:c1}
मान लीजिए $f$ $U$ पर $C^1$ है और $(a,b) \in U$। तब, $(h, k) \to (0,0)$
होने पर:
\[
f(a + h, b + k) = f(a, b)
+ h\,\frac{\partial f}{\partial x}(a,b)
+ k\,\frac{\partial f}{\partial y}(a,b)
+ o\bigl(\norm{(h,k)}\bigr) .
\]
विशेष रूप से $f$ \omterm{def:b1:continuity:continuous}{संतत} है, और आलेख $z = f(x,y)$ का प्रत्येक बिंदु पर वही
स्पर्श समतल है जो इसी सूत्र से पढ़ लिया जाता है।
\end{theorem}

\begin{example}[रैखिक सन्निकटन, काम पर]
\label{ex:b1:multivar:linapprox}
$f(x, y) = x^3y^2$ के लिए $f(1.02,\ 0.99)$ का आकलन कीजिए। $(1, 1)$ पर:
$f = 1$, $\frac{\partial f}{\partial x} = 3x^2y^2 = 3$,
$\frac{\partial f}{\partial y} = 2x^3y = 2$, अतः
\cref{thm:b1:multivar:c1} देती है
\[
f(1.02,\ 0.99) \approx 1 + 3\,(0.02) + 2\,(-0.01) = 1.04 ,
\]
जबकि यथार्थ मान $1.02^3 \times 0.99^2 = 1.04006\dots$ है --- त्रुटि
वृद्धियों में द्वितीय कोटि की है, जैसा $o(\norm{(h,k)})$ का वचन है।
$(1, 1, 1)$ पर आलेख का स्पर्श समतल $z = 1 + 3(x - 1) + 2(y - 1)$ है, वही
समीकरण जो इस आकलन में छिपा हुआ था।
\end{example}

\begin{proof}
एक बार में एक ही \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} हिलाइए:
\[
f(a+h, b+k) - f(a,b)
= \bigl[f(a+h, b+k) - f(a, b+k)\bigr] + \bigl[f(a, b+k) -
f(a,b)\bigr].
\]
एक चर वाली माध्य मान प्रमेय (\cref{thm:b1:derivative:mvt}) से पहला कोष्ठक
किसी $\theta \in \intoo{0}{1}$ के लिए $h\,\frac{\partial f}{\partial x}(a +
\theta h,\, b + k)$ है, और दूसरा $k\,\frac{\partial f}{\partial y}(a,\, b +
\theta' k)$। $(a,b)$ पर आंशिक अवकलजों का \omterm{def:b1:continuity:continuous}{सांतत्य} हमें प्रत्येक को ($(a,b)$
पर का मान) $+$ (त्रुटि $\to 0$) के रूप में लिखने देता है; कुल त्रुटि
$h\,\varepsilon_1 + k\,\varepsilon_2 = o(\norm{(h,k)})$ है, क्योंकि $\abs
h, \abs k \leq \norm{(h,k)}$।
\end{proof}

\begin{theorem}[शृंखला नियम]\label{thm:b1:multivar:chain}
मान लीजिए $f$ $U$ पर $C^1$ है और $t \mapsto (x(t), y(t))$ किसी \omterm{prop:b1:reals:intervals}{अंतराल} से
$U$ में $C^1$ है। तब $g(t) = f\bigl(x(t), y(t)\bigr)$ $C^1$ है, और\index{शृंखला नियम!दो चर}
\[
g'(t) = x'(t)\,\frac{\partial f}{\partial x}\bigl(x(t),y(t)\bigr)
+ y'(t)\,\frac{\partial f}{\partial y}\bigl(x(t),y(t)\bigr)
= \bigl\langle \nabla f,\ (x', y')\bigr\rangle .
\]
\end{theorem}

\begin{proof}
$(x(t), y(t))$ पर \cref{thm:b1:multivar:c1} को $(h, k) = (x(t+s) - x(t),\,
y(t+s) - y(t))$ के साथ लगाइए: $s \to 0$ होने पर एक चर वाली अवकलनीयता $h =
s\,x'(t) + o(s)$ तथा $k = s\,y'(t) + o(s)$ देती है, अतः $\norm{(h, k)} =
O(s)$ और
\[
g(t+s) - g(t)
= h\,\frac{\partial f}{\partial x} + k\,\frac{\partial
f}{\partial y} + o\bigl(\norm{(h,k)}\bigr)
= s\,\Bigl(x'\,\frac{\partial f}{\partial x} +
y'\,\frac{\partial f}{\partial y}\Bigr) + o(s) ,
\]
जहाँ अंतिम त्रुटि $h$ और $k$ दोनों के $o(s)$ को (आंशिक अवकलजों के नियत
मानों से गुणित) तथा स्पर्श-समतल आकलन के $o(O(s))$ को --- दोनों को सोख लेती
है। $s$ से भाग दीजिए और $s \to 0$ लीजिए। $g'$ का \omterm{def:b1:continuity:continuous}{सांतत्य} सभी अवयवों के
\omterm{def:b1:continuity:continuous}{सांतत्य} से निकल आता है।
\end{proof}

\begin{example}[शृंखला नियम, दोनों तरफ़ से जाँचा हुआ]
\label{ex:b1:multivar:chainrun}
मान लीजिए $f(x, y) = x^2 y$ और $g(t) = f(t, t^2)$। सीधे: $g(t) = t^2\cdot
t^2 = t^4$, अतः $g'(t) = 4t^3$। शृंखला नियम से: $\frac{\partial f}{\partial
x} = 2xy$ और $\frac{\partial f}{\partial y} = x^2$, जिन्हें वक्र $(t, t^2)$
पर आँका जाए:
\[
g'(t) = 1\cdot 2t\cdot t^2 + 2t\cdot t^2 = 2t^3 + 2t^3 = 4t^3 .
\]
दोनों संगणनाएँ मेल खाती हैं, और यह बँटवारा अर्थपूर्ण है: वृद्धि का $2t^3$
$x$-प्रवणता से होकर \emph{दाईं ओर} चलने से आता है, और $2t^3$ $y$-प्रवणता
से होकर \emph{ऊपर} चलने से। जिन वक्रों पर $g$ का कोई संवृत रूप नहीं होता,
वहाँ केवल दूसरी संगणना बची रहती है --- प्रमेय का सारा प्रयोजन यही है।
\end{example}

\begin{remark}[प्रवणता को पढ़ना]
किसी एकक दिशा $u$ के सहारे $t \mapsto f(a + tu)$ पर शृंखला नियम
\emph{दैशिक \omterm{def:b1:derivative:def}{अवकलज}} $\langle \nabla f(a), u\rangle$ देता है: यह तब \omterm{def:b1:reals:bounds}{उच्चतम}
होता है जब $u$ $\nabla f(a)$ की दिशा में हो (कोशी--श्वार्ट्ज़ से,
\cref{thm:b1:euclid:cs})। \omterm{def:b1:multivar:partial}{प्रवणता} तीव्रतम आरोहण की दिशा है, और वह स्तर
वक्रों $\{f = c\}$ पर \omterm{def:b1:euclid:orthogonal}{लांबिक} है (किसी स्तर \omterm{def:b1:logic:sets}{समुच्चय} में खींचे गए वक्र के
सहारे $f$ को अवकलित कीजिए: शृंखला नियम $\langle\nabla f,\,
\text{स्पर्शरेखा}\rangle = 0$ देता है)।
\end{remark}

\begin{example}[स्तर वक्र और प्रवणताएँ, एक ही फलन पर]
\label{ex:b1:multivar:levelcurves}
$f(x, y) = x^2 - y^2$ लीजिए। उसके स्तर \omterm{def:b1:logic:sets}{समुच्चय}: $c > 0$ के लिए $\{f = c\}$
दाएँ-बाएँ खुलता अतिपरवलय है, $c < 0$ के लिए ऊपर-नीचे खुलता, और $c = 0$ के
लिए एक-दूसरे को काटती रेखाओं का युग्म $y = \pm x$ --- किसी पर्वतीय दर्रे का
समोच्च चित्र, जिसका \omterm{met:b1:multivar:monge}{काठी बिंदु} मूल बिंदु पर है, जहाँ दोनों शून्य-स्तर
रेखाएँ कटती हैं। \omterm{def:b1:multivar:partial}{प्रवणता}: $\nabla f = (2x, -2y)$। बिंदु $(2, 1)$ पर (स्तर
$c = 3$ पर): $\nabla f = (4, -2)$, जबकि स्तर वक्र का स्पर्श सदिश, यदि उसे
उस बिंदु के पास $\bigl(t, \sqrt{t^2 - 3}\bigr)$ से प्राचलित किया जाए,
$\bigl(1, \frac{t}{\sqrt{t^2 - 3}}\bigr)$ है, अर्थात् $t = 2$ पर $(1, 2)$
--- और सचमुच
\[
\langle (4, -2),\ (1, 2)\rangle = 4 - 4 = 0 :
\]
\omterm{def:b1:multivar:partial}{प्रवणता} समोच्च पर लंब है और $f$ के बड़े मानों की ओर संकेत करती है (यहाँ:
$y$-अक्ष से दूर)। दो और पाठ: \omterm{def:b1:multivar:partial}{प्रवणता} ठीक काठी पर लुप्त होती है, जहाँ
समोच्च चित्र सिकुड़ कर चिमटी बन जाता है; और $(2,1)$ पर स्तर वक्र की
\emph{स्पर्शरेखा} $4(x - 2) - 2(y - 1) = 0$, अर्थात् $2x - y = 3$ है ---
वही समीकरण ``$\langle \nabla f,\ M - M_0\rangle = 0$'' जो
\cref{exo:b1:curves:11} की दीर्घवृत्त-स्पर्शरेखा का सामान्यीकरण है।
\end{example}

\begin{theorem}[श्वार्ट्ज़]\label{thm:b1:multivar:schwarz}
यदि $f$ $C^2$ वर्ग का है (आंशिक अवकलजों के \omterm{def:b1:multivar:partial}{आंशिक अवकलज} विद्यमान और \omterm{def:b1:continuity:continuous}{संतत}
हैं), तो\index{श्वार्ट्ज़ की प्रमेय}
\[
\frac{\partial^2 f}{\partial x\,\partial y}
= \frac{\partial^2 f}{\partial y\,\partial x} .
\]
\end{theorem}
\admitted

\section{स्थानीय चरम}

\begin{method}[चरम-अध्ययन, क्रम से]
\label{met:b1:multivar:extremamethod}
\begin{enumerate}
  \item \emph{$\nabla f = 0$ को पूरा-पूरा हल कीजिए।} जहाँ भी संभव हो,
        प्रत्येक \omterm{def:b1:multivar:partial}{आंशिक अवकलज} का गुणनखंडन कीजिए (\omterm{def:b1:linmaps:def}{रैखिक} गुणनखंडों के
        गुणनफल निकाय को पारदर्शी स्थितियों में बाँट देते हैं, जैसे नीचे
        \cref{ex:b1:multivar:fourpoints} में); एक भूली हुई स्थिति एक भूला
        हुआ \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु} है।
  \item \emph{प्रत्येक बिंदु का वर्गीकरण} मोंज राशियों $r, s, t$ से
        कीजिए --- और उन्हें \emph{हर बिंदु पर} फिर से संगणित कीजिए, एक
        ही बार सबके लिए नहीं।
  \item \emph{यदि $rt - s^2 = 0$}, तो उस बिंदु से होकर जाने वाले
        सुचुने वक्रों पर (पहले रेखाएँ, फिर परवलय) $f$ की सीधी जाँच
        कीजिए, और या तो दो चिह्न ढूँढ़िए (कोई चरम नहीं) या एक जकड़ा हुआ
        चिह्न, जिसके साथ सभी दिशाओं को ढकने वाला तर्क हो।
  \item \emph{समग्र चित्र के लिए पीछे हटिए}: अनंत पर व्यवहार जाँचिए (कोई
        स्थानीय न्यूनतम समग्र भी हो, यह ज़रूरी नहीं), और यदि प्रांत
        \omterm{def:b1:topology:open}{विवृत} न हो, तो उसकी \omterm{def:b1:topology:closure}{परिसीमा} का अलग उपचार कीजिए
        (\cref{exo:b1:multivar:12}) --- \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु} वाली प्रमेय केवल
        \omterm{def:b1:topology:closure}{अंतःभाग} के बिंदु देखती है।
\end{enumerate}
\end{method}

\begin{theorem}[क्रांतिक बिंदु]\label{thm:b1:multivar:critical}
यदि \omterm{def:b1:topology:open}{विवृत समुच्चय} $U$ पर $C^1$ फलन $f$ का $(a,b) \in U$ पर स्थानीय चरम है,
तो $\nabla f(a,b) = (0,0)$: वह बिंदु \emph{क्रांतिक}\index{क्रांतिक बिंदु}
कहलाता है।
\end{theorem}

\begin{proof}
एक चर वाले फलनों $x \mapsto f(x, b)$ तथा $y \mapsto f(a, y)$ के क्रमशः $a$
और $b$ पर भीतरी स्थानीय चरम हैं: \cref{prop:b1:derivative:fermat} दोनों
आंशिक अवकलजों को मार देती है।
\end{proof}

\begin{method}[द्वितीय-कोटि कसौटी (मोंज संकेतन)]\label{met:b1:multivar:monge}
किसी $C^2$ फलन के \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु} पर रखिए
\[
r = \frac{\partial^2 f}{\partial x^2},
\qquad
s = \frac{\partial^2 f}{\partial x \partial y},
\qquad
t = \frac{\partial^2 f}{\partial y^2}
\qquad (\text{बिंदु पर के मान}).
\]
\begin{itemize}
  \item यदि $rt - s^2 > 0$: स्थानीय चरम --- $r > 0$ पर न्यूनतम, $r < 0$
        पर अधिकतम;
  \item यदि $rt - s^2 < 0$: कोई चरम नहीं (एक \emph{काठी
        बिंदु}\index{काठी बिंदु});
  \item यदि $rt - s^2 = 0$: कसौटी मौन है; सीधी जाँच कीजिए।
\end{itemize}
(इसकी पुष्टि --- कोटि $2$ तक का टेलर--यंग प्रसार और द्विघात रूप $r h^2 +
2shk + tk^2$ का चिह्न-अध्ययन --- दूसरे वर्ष में की जाती है; यहाँ कसौटी
काम के औज़ार की तरह प्रयोग होती है।)
\end{method}

\begin{example}\label{ex:b1:multivar:extrema}
$f(x,y) = x^3 + y^3 - 3xy$। \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु}: $\nabla f = (3x^2 - 3y,\; 3y^2
- 3x) = 0$ से $y = x^2$ और $x = y^2$, अतः $x = x^4$: $x \in \{0, 1\}$:
बिंदु $(0,0)$ और $(1,1)$।

द्वितीय \omterm{def:b1:derivative:def}{अवकलज}: $r = 6x$, $s = -3$, $t = 6y$। $(0,0)$ पर: $rt - s^2 = -9 <
0$: काठी। $(1,1)$ पर: $rt - s^2 = 36 - 9 > 0$, $r = 6 > 0$: स्थानीय
न्यूनतम, $f(1,1) = -1$। (समग्र नहीं: $f(x, 0) = x^3 \to -\infty$।)
\end{example}

\begin{example}[चार बिंदुओं का पूरा अध्ययन]
\label{ex:b1:multivar:fourpoints}
$f(x, y) = xy\,(3 - x - y) = 3xy - x^2y - xy^2$। \omterm{def:b1:multivar:partial}{प्रवणता}:
\[
\frac{\partial f}{\partial x} = y\,(3 - 2x - y),
\qquad
\frac{\partial f}{\partial y} = x\,(3 - x - 2y).
\]
\omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु}: यदि $y = 0$, तो दूसरा समीकरण $x \in \{0, 3\}$ देता है; यदि
$x = 0$, तो पहला $y \in \{0, 3\}$ देता है; और यदि $xy \neq 0$, तो $2x + y =
3$, $x + 2y = 3$ हल कीजिए: $x = y = 1$। चार बिंदु: $(0,0)$, $(3,0)$,
$(0,3)$, $(1,1)$। द्वितीय \omterm{def:b1:derivative:def}{अवकलज}: $r = -2y$, $s = 3 - 2x - 2y$, $t = -2x$।
\begin{itemize}
  \item $(1,1)$: $r = -2$, $s = -1$, $t = -2$: $rt - s^2 = 3 > 0$, $r <
        0$: स्थानीय अधिकतम, $f(1,1) = 1$।
  \item $(0,0)$: $r = t = 0$, $s = 3$: $rt - s^2 = -9 < 0$: काठी; इसी
        प्रकार $(3, 0)$ ($s = -3$) और $(0, 3)$: तीन काठियाँ।
\end{itemize}
अधिकतम केवल स्थानीय है: $f(-T, -T) = T^2(3 + 2T) \to +\infty$। सममिति की
जाँच: $f(x, y) = f(y, x)$, और सचमुच क्रांतिक \omterm{def:b1:logic:sets}{समुच्चय} तथा वर्गीकरण दोनों $x
\leftrightarrow y$ में सममित हैं। व्याख्या: $x, y \geq 0$, $x + y \leq 3$
वाले ढीले आयतों में $3$ के तीनों ``भागों'' का गुणनफल $xy(3 - x - y)$ तब
सबसे बड़ा होता है जब भाग बराबर हों --- समांतर माध्य--गुणोत्तर माध्य वाले
सिद्धांत की दो चरों वाली छाया।
\end{example}

\begin{remark}[आम फिसलनें]
\label{rem:b1:multivar:pitfalls}
\emph{\omterm{def:b1:multivar:partial}{आंशिक अवकलज} किसी असांतत्य के बिंदु पर भी हो सकते हैं}:
\cref{ex:b1:multivar:trap} के त्रिज्य फंदे $g(x,y) =
\frac{xy}{x^2+y^2}$ के लिए $\frac{\partial g}{\partial x}(0,0) =
\frac{\partial g}{\partial y}(0,0) = 0$ है (दोनों अक्ष-प्रतिबंध सर्वत्र
लुप्त हैं), फिर भी $g$ की मूल बिंदु पर कोई सीमा नहीं --- \omterm{def:b1:multivar:partial}{आंशिक अवकलज} केवल
दो दिशाएँ टटोलते हैं, जबकि \omterm{def:b1:continuity:continuous}{सांतत्य} को सब चाहिए; व्यवस्था केवल $C^1$
परिकल्पना लौटाती है (\cref{thm:b1:multivar:c1})।
\emph{रेखीय सीमाएँ कभी पर्याप्त नहीं}: \cref{exo:b1:multivar:3} के फलन की
हर रेखा के सहारे सीमा $0$ है और फिर भी कोई सीमा नहीं --- परवलय हमेशा
आज़माइए (या $\theta$ में एकरूप ध्रुवीय परिबंध)।
\emph{क्रांतिक होना आवश्यक है, पर्याप्त नहीं}: काठियाँ भरी पड़ी हैं
(\cref{ex:b1:multivar:fourpoints} के चार में से तीन बिंदु); और प्रमेय केवल
\emph{\omterm{def:b1:topology:open}{विवृत}} समुच्चयों पर लागू होती है --- किसी \omterm{def:b1:topology:closed}{संवृत} चक्रिका पर चरम अशून्य
\omterm{def:b1:multivar:partial}{प्रवणता} के साथ \omterm{def:b1:topology:closure}{परिसीमा} पर बैठ सकते हैं (\cref{exo:b1:multivar:12})।
\emph{मौन स्थिति $rt - s^2 = 0$ सचमुच मौन है}: $x^4 + y^4$ (न्यूनतम) और
$x^3 + y^3$ (कुछ भी नहीं) --- दोनों के लिए मूल बिंदु पर $r = s = t = 0$ है;
निर्णय केवल सीधा चिह्न-अध्ययन करता है (\cref{exo:b1:multivar:6}, फलन $h$)।
\emph{\omterm{def:b1:multivar:partial}{प्रवणता} स्तर वक्रों पर \omterm{def:b1:euclid:orthogonal}{लांबिक} है, उनके सहारे नहीं}: किसी समोच्च
रेखा पर चलने के लिए $\nabla f$ के लंब चलिए; सबसे तेज़ चढ़ने के लिए उसी के
सहारे --- दोनों को मिला देना हर समोच्च चित्र की ज्यामिति उलट देता है।
\end{remark}

\begin{remark}[दो चर किधर ले जाते हैं]
\label{rem:b1:multivar:whereused}
यह अध्याय एक द्वार है। \omterm{def:b1:multivar:partial}{प्रवणता} और शृंखला नियम स्नातक वर्ष~2 के खंड में
अक्षरशः $n$ चरों तक फैल जाते हैं, जहाँ \cref{thm:b1:multivar:c1} का
$o(\norm{(h,k)})$ \emph{अवकल} बन जाता है और मोंज कसौटी कोटि-दो टेलर सूत्र
तथा द्विघात रूपों के रास्ते पूरी तरह सिद्ध होती है। जो विशेष स्थिति
\emph{इसी} वर्ष निपटाई जा सकती है --- द्विघात फलन, जिनके लिए द्वितीय-कोटि
प्रसार यथार्थ है --- वही सप्ताहांत समस्या का विषय है, और संयोग से वही
स्थिति दुनिया भर की आँकड़ा-फिटिंग चलाती है: \omterm{pb:b1:multivar:1}{न्यूनतम वर्ग} प्रतिगमन। विवश
चरम (\cref{exo:b1:multivar:5} एक झलक था) वर्ष~2 में लाग्रांज गुणक बन जाते
हैं; और प्रसंवादी फलन (\cref{exo:b1:multivar:7}) स्नातक वर्ष~3 के खंड के
सम्मिश्र विश्लेषण में लौटते हैं।
\end{remark}

\begin{remark}[पुस्तक 3 के भीतर परिप्रेक्ष्य: वर्ष, बंद]
\label{rem:b1:multivar:perspectives}
यही वह अध्याय है जहाँ इस खंड के दोनों हिस्से हाथ मिलाते हैं। विश्लेषण वाले
हिस्से ने अपने औज़ार एक-एक \omterm{def:b1:derivative:def}{अवकलज} करके दिए: माध्य मान प्रमेय
\cref{thm:b1:multivar:c1} को चलाती है, टेलर प्रसार चरम-कसौटियाँ चलाते हैं,
और \cref{ch:b1:topology} के $\varepsilon$ अंतरालों की जगह गोले लेकर लौट
आए। बीजगणित वाले हिस्से ने ज्यामिति दी: \omterm{def:b1:multivar:partial}{प्रवणता} \cref{ch:b1:euclid} के
\omterm{def:b1:euclid:def}{अंतर्गुणन} से पढ़ी जाती है (कोशी--श्वार्ट्ज़ उसे तीव्रतम दिशा बनाता है),
मोंज राशियाँ $(r, s, t)$ \cref{ch:b1:matrices} का एक सममित आव्यूह हैं
जिनकी कसौटी \cref{ch:b1:det} का सारणिक है, और सप्ताहांत समस्या
आँकड़ा-सदिशों पर \omterm{thm:b1:euclid:projection}{लांबिक प्रक्षेप} चलाती है। \cref{ch:b1:curves} के वक्र तक
स्तर समुच्चयों के रूप में लौट आते हैं। जो पाठक इन पाँचों हस्तांतरणों का
कारण फिर से खड़ा कर सके, उसने वस्तुतः पूरे वर्ष की पुनरावृत्ति कर ली ---
और इस अंतिम अध्याय का असली प्रयोजन यही है।
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:multivar:1}
\omterm{def:b1:multivar:partial}{आंशिक अवकलज} संगणित कीजिए:
$f(x,y) = x^2 y + \eu^{xy}$; $\;g(x,y) = \ln(x^2 + y^2)$
($\R^2\setminus\{0\}$ पर); $\;h(x,y) = \arctan\frac yx$ ($x > 0$ पर)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:multivar:2}
$(0,0)$ पर (जहाँ मान $0$ है) \omterm{def:b1:continuity:continuous}{सांतत्य} का अध्ययन कीजिए:
\[
f(x,y) = \frac{x^2 y}{x^2 + y^2},
\qquad
g(x,y) = \frac{xy}{x^2 + y^2},
\qquad
h(x,y) = \frac{x^3 + y^3}{x^2 + y^2}.
\]
\emph{(ध्रुवीय \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} $x = \rho\cos\theta$, $y = \rho\sin\theta$ मदद
करते हैं: जहाँ संभव हो, केवल $\rho$ के किसी फलन से परिबद्ध कीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:3}
मान लीजिए $f(x,y) = \dfrac{x^2 y}{x^4 + y^2}$ ($f(0,0) = 0$)। सिद्ध कीजिए
कि मूल बिंदु पर \emph{हर सरल रेखा के सहारे} $f$ की सीमा $0$ है, पर
$f\bigl(x, x^2\bigr) = \frac12$: अतः $f$ $(0,0)$ पर \omterm{def:b1:continuity:continuous}{संतत} नहीं है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:multivar:4}
$f(x, y) = x^3 y^2 + \sin(xy)$ पर श्वार्ट्ज़ की प्रमेय हाथ से जाँचिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:5}
मान लीजिए $f$ $\R^2$ पर $C^1$ है और $g(t) = f(\cos t, \sin t)$। शृंखला
नियम से $g'(t)$ व्यक्त कीजिए। इससे निष्कर्ष निकालिए कि एकक वृत्त पर
प्रतिबंधित $f$ अपने चरम उन बिंदुओं पर प्राप्त करता है जहाँ $\nabla f$
त्रिज्या सदिश के समांतर है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:6}
इनके \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु} ढूँढ़िए और उनका वर्गीकरण कीजिए:
\[
f(x, y) = x^2 + xy + y^2 - 3x,
\qquad
g(x, y) = x^2 - y^2 + 4y,
\qquad
h(x, y) = x^4 + y^4 - 2(x - y)^2 .
\]
\emph{($h$ के लिए मूल बिंदु पर सारणिक-कसौटी मौन है: $h(x, x)$ और $h(x,
-x)$ जाँचिए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:7}
कोई फलन $f$ \emph{प्रसंवादी} कहलाता है जब $\frac{\partial^2 f}{\partial
x^2} + \frac{\partial^2 f}{\partial y^2} = 0$ हो। जाँचिए कि $x^2 - y^2$,
$xy$, $\eu^x\cos y$ और $\ln(x^2 + y^2)$ (मूल बिंदु के बाहर) प्रसंवादी हैं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:8}
समतल $\{x + 2y - z = 4\} \subseteq \R^3$ का वह बिंदु ढूँढ़िए जो मूल बिंदु
के सबसे निकट है, दो रास्तों से: \omterm{thm:b1:euclid:projection}{लांबिक प्रक्षेप} द्वारा
(\cref{ch:b1:euclid}), और $z$ का विलोपन करके मिले दो चरों वाले फलन $f(x, y)
= x^2 + y^2 + (x + 2y - 4)^2$ को न्यूनतम करके।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:9}
(ध्रुवीय निर्देशांकों में लाप्लासीय, पहला संपर्क) मान लीजिए $f$ $\R^2
\setminus \{0\}$ पर $C^2$ है और \emph{त्रिज्य} है: $f(x, y) =
\varphi\bigl(\sqrt{x^2+y^2}\bigr)$, जहाँ $\varphi$ $\intoo{0}{+\infty}$ पर
$C^2$ वर्ग का है। सिद्ध कीजिए कि
\[
\frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial
y^2} = \varphi''(\rho) + \frac{\varphi'(\rho)}{\rho},
\qquad \rho = \sqrt{x^2 + y^2},
\]
और छिद्रित समतल पर सभी त्रिज्य प्रसंवादी फलन ज्ञात कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:10}
बिंदु $(1, 1, 1)$ पर आलेख $z = xy$ के स्पर्श समतल का समीकरण दीजिए। फिर
$f(x,y) = x^3 + y^3 - 3xy$ के आलेख के वे सभी बिंदु ढूँढ़िए जहाँ स्पर्श
समतल \emph{क्षैतिज} है, और उत्तर को \cref{ex:b1:multivar:extrema} से
जोड़िए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:11}
मान लीजिए $f$ $\R^2$ पर $C^1$ है।
\begin{enumerate}
  \item यदि सर्वत्र $\frac{\partial f}{\partial x} = 0$, तो सिद्ध कीजिए
        कि $f(x, y)$ केवल $y$ पर निर्भर करता है।
  \item $\R^2$ पर समीकरण $\frac{\partial f}{\partial x} = \frac{\partial
        f}{\partial y}$ के सभी $C^1$ हल ज्ञात कीजिए। \emph{($g(u, v) =
        f\bigl(\frac{u+v}2, \frac{u-v}2\bigr)$ रखिए और शृंखला नियम से
        $\frac{\partial g}{\partial v}$ संगणित कीजिए।)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:12}
\omterm{def:b1:topology:closed}{संवृत} चक्रिका $x^2 + y^2 \leq 1$ पर $f(x, y) = xy$ के समग्र अधिकतम और
न्यूनतम ज्ञात कीजिए। \emph{(दो चरों वाली चरम मान प्रमेय मान लीजिए: \omterm{def:b1:topology:closed}{संवृत}
चक्रिका पर कोई \omterm{def:b1:continuity:continuous}{संतत} फलन अपने परिबंध प्राप्त कर लेता है --- उपपत्ति स्नातक
वर्ष~2 के खंड में है। \omterm{def:b1:topology:open}{विवृत} चक्रिका का उपचार क्रांतिक बिंदुओं से कीजिए और
परिसीमा-वृत्त का \cref{exo:b1:multivar:5} के प्राचलन से।)}
\end{exercise}

\section{समस्या: न्यूनतम वर्ग और प्रतिगमन रेखा}

\begin{problem}\label{pb:b1:multivar:1}
$n$ आँकड़ा-बिंदु दिए हों: कौन-सी सरल रेखा उन सबके ``सबसे पास'' से गुज़रती
है? लजांद्र और गाउस का उत्तर था: वह रेखा जो ऊर्ध्वाधर त्रुटियों के
\emph{वर्गों} का योग न्यूनतम करे --- क्योंकि वही न्यूनीकरण रैखिक बीजगणित से
यथार्थतः हल हो जाता है। यह समस्या पहले \cref{met:b1:multivar:monge} की मोंज
कसौटी को द्विघात फलनों के लिए ईमानदारी से सिद्ध करती है (यही वह अकेली
स्थिति है जहाँ द्वितीय-कोटि प्रसार यथार्थ होता है), और फिर
\cref{ch:b1:euclid} की यूक्लिडीय ज्यामिति पर \omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण}, \omterm{pb:b1:multivar:1}{प्रतिगमन
रेखा} और सहसंबंध गुणांक खड़े करती
है।\index{न्यूनतम वर्ग}\index{प्रतिगमन रेखा}
\index{प्रसामान्य समीकरण}\index{सहसंबंध}

\medskip
\noindent\textbf{भाग I --- द्विघात फलन: मोंज कसौटी, सिद्ध।} वास्तविक $r,
s, t$ नियत कीजिए और $q(h, k) = r h^2 + 2s hk + t k^2$ रखिए।
\begin{enumerate}
  \item मान लीजिए $r \neq 0$। पूर्ण-वर्ग रूप
        \[
        q(h, k) = r\Bigl(h + \frac{s}{r}k\Bigr)^{\!2}
        + \frac{rt - s^2}{r}\,k^2
        \]
        स्थापित कीजिए, और निष्कर्ष निकालिए: यदि $rt - s^2 > 0$, तो मूल
        बिंदु के बाहर $q$ का चिह्न कठोरतः $r$ का है; यदि $rt - s^2 < 0$,
        तो $q$ दोनों चिह्न लेता है।
  \item बची हुई स्थितियाँ निपटाइए: $r = 0$, $t \neq 0$ (सममित); और $r =
        t = 0$ ($q = 2shk$)। निष्कर्ष निकालिए: $q$ दोनों चिह्न लेता है
        यदि और केवल यदि $rt - s^2 < 0$, और $q$ केवल मूल बिंदु पर लुप्त
        होता है यदि और केवल यदि $rt - s^2 > 0$।
  \item अब मान लीजिए $f(x, y) = \frac12\bigl(r x^2 + 2s xy + t
        y^2\bigr) + \beta x + \gamma y + \delta$ कोई द्विघात फलन है
        जिसका एक \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु} $X_0 = (a, b)$ है। \emph{यथार्थ} प्रसार
        \[
        f(X_0 + (h,k)) = f(X_0) + \tfrac12\,q(h, k)
        \qquad (\text{कोई शेषफल नहीं})
        \]
        सिद्ध कीजिए, और इससे द्विघात फलनों के लिए मोंज वर्गीकरण
        निकालिए: $rt - s^2 > 0$, $r > 0$ पर कठोर समग्र न्यूनतम; $rt -
        s^2 > 0$, $r < 0$ पर कठोर समग्र अधिकतम; और $rt - s^2 < 0$ पर
        काठी।
  \item मान लीजिए $rt - s^2 > 0$ और $r > 0$ (अतः $t > 0$ भी)। प्रबलता
        का स्पष्ट परिबंध
        \[
        q(h, k) \geq c\,(h^2 + k^2),
        \qquad
        c = \frac{rt - s^2}{r + t} > 0
        \]
        सिद्ध कीजिए। \emph{(दिखाइए कि द्विघात रूप $q - c(h^2 + k^2)$,
        जिसके गुणांक $r - c$, $s$, $t - c$ हैं, अब भी विविक्तकर-कसौटी
        की ऋणेतर राशियाँ रखता है।)}
  \item इससे निष्कर्ष निकालिए कि धनात्मक निश्चित द्विघात भाग वाला कोई
        द्विघात फलन $\norm{(x,y)} \to \infty$ पर $+\infty$ की ओर जाता
        है, अतः उसका एक \emph{अद्वितीय समग्र} न्यूनकारी है: उसका
        \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु}। \cref{ex:b1:multivar:extrema} से तुलना कीजिए,
        जहाँ किसी अद्विघात फलन का स्थानीय न्यूनतम समग्र नहीं था।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- \omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण}।} मान लीजिए $C_1, C_2, b \in
\R^n$ (मानक \omterm{def:b1:euclid:def}{अंतर्गुणन} के साथ) और
\[
f(u, v) = \norm{u\,C_1 + v\,C_2 - b}^2 .
\]
\begin{enumerate}[resume]
  \item $f$ का प्रसार कीजिए और दिखाइए कि वह $(u, v)$ का द्विघात फलन है,
        जिसके द्विघात भाग के गुणांक $r = 2\norm{C_1}^2$, $s = 2\langle
        C_1, C_2\rangle$, $t = 2\norm{C_2}^2$ हैं।
  \item दिखाइए कि $rt - s^2 > 0$ यदि और केवल यदि $(C_1, C_2)$ \omterm{def:b1:vspaces:free}{स्वतंत्र}
        है \emph{(कोशी--श्वार्ट्ज़ की समता-स्थिति,
        \cref{thm:b1:euclid:cs})}; आगे इसे मान लीजिए।
  \item दिखाइए कि क्रांतिक समीकरण $\nabla f = 0$ ही
        \emph{\omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण}}
        \[
        \begin{pmatrix}
        \norm{C_1}^2 & \langle C_1, C_2\rangle\\
        \langle C_1, C_2\rangle & \norm{C_2}^2
        \end{pmatrix}
        \begin{pmatrix} u\\ v\end{pmatrix}
        =
        \begin{pmatrix} \langle C_1, b\rangle\\ \langle C_2,
        b\rangle\end{pmatrix}
        \]
        हैं --- बाईं ओर \cref{exo:b1:euclid:11} का ग्राम आव्यूह --- और
        कि वे ठीक यही कहते हैं: $b - (uC_1 + vC_2) \perp C_1, C_2$। भाग
        I और \cref{thm:b1:euclid:projection} से निष्कर्ष निकालिए:
        अद्वितीय न्यूनकारी $p = uC_1 + vC_2$ देता है, जो $b$ का
        $\operatorname{Vect}(C_1, C_2)$ पर \omterm{thm:b1:euclid:projection}{लांबिक प्रक्षेप} है।
  \item दिखाइए कि न्यूनतम मान $\norm{b}^2 - \langle p, b\rangle =
        \norm{b - p}^2$ है, और पाइथागोरसीय चित्र खींचिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- \omterm{pb:b1:multivar:1}{प्रतिगमन रेखा}।} आँकड़ा-बिंदु $(x_1, y_1),
\dots, (x_n, y_n)$, जिनमें सभी $x_i$ बराबर नहीं हैं। न्यूनतम कीजिए
\[
E(\alpha, \beta) = \sum_{i=1}^{n}\bigl(y_i - \alpha - \beta
x_i\bigr)^2 .
\]
लिखिए $\bar x = \frac1n\sum x_i$, $\bar y = \frac1n\sum y_i$,
$v_x = \frac1n\sum x_i^2 - \bar x^2$, $v_y = \frac1n\sum y_i^2 -
\bar y^2$, $c_{xy} = \frac1n\sum x_i y_i - \bar x\bar y$।
\begin{enumerate}[resume]
  \item $C_1 = (1, \dots, 1)$, $C_2 = (x_1, \dots, x_n)$, $b = (y_1,
        \dots, y_n)$ के साथ भाग II पहचानिए, और \omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण}
        लिखिए:
        \[
        n\alpha + \Bigl(\sum x_i\Bigr)\beta = \sum y_i,
        \qquad
        \Bigl(\sum x_i\Bigr)\alpha + \Bigl(\sum
        x_i^2\Bigr)\beta = \sum x_i y_i .
        \]
  \item उन्हें हल कीजिए:
        \[
        \beta = \frac{c_{xy}}{v_x},
        \qquad
        \alpha = \bar y - \beta\,\bar x ,
        \]
        और देखिए कि \emph{\omterm{pb:b1:multivar:1}{प्रतिगमन रेखा}} $y = \alpha + \beta x$ माध्य
        बिंदु $(\bar x, \bar y)$ से होकर जाती है।
  \item जाँचिए कि $v_x > 0$ ठीक तब है जब सभी $x_i$ बराबर न हों, और इसे
        प्रश्न 7 से मिलाइए।
  \item सिद्ध कीजिए कि न्यूनतम त्रुटि
        \[
        E_{\min} = n\Bigl(v_y - \frac{c_{xy}^2}{v_x}\Bigr)
        = n\,v_y\,(1 - \rho^2),
        \qquad
        \rho = \frac{c_{xy}}{\sqrt{v_x v_y}}
        \quad (v_y \neq 0)
        \]
        है। इससे $\abs\rho \leq 1$ निकालिए, जहाँ $\abs\rho = 1$ यदि और
        केवल यदि आँकड़े पूर्णतः संरेख हों।
  \item हल किया हुआ उदाहरण: आँकड़ों $(0,0)$, $(1,1)$, $(2,3)$, $(3,4)$
        के लिए $\bar x, \bar y, v_x, c_{xy}$, \omterm{pb:b1:multivar:1}{प्रतिगमन रेखा}, चारों
        अवशेष और $E_{\min}$ संगणित कीजिए।
  \item सिद्ध कीजिए कि इष्टतम रेखा के अवशेष $\varepsilon_i = y_i -
        \alpha - \beta x_i$ $\sum_i \varepsilon_i = 0$ तथा $\sum_i
        x_i\varepsilon_i = 0$ को संतुष्ट करते हैं, और दोनों की व्याख्या
        लांबिकता से कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- रूपांतर और अनुप्रयोग।}
\begin{enumerate}[resume]
  \item (सर्वोत्तम अचर) दिखाइए कि $\sum_i (y_i - \alpha)^2$ को न्यूनतम
        करने वाला अचर $\alpha$ माध्य $\bar y$ है, और न्यूनतम मान
        $n\,v_y$: प्रसरण यही नापता है कि आँकड़े अचर होने में कितना
        चूकते हैं।
  \item (मूल बिंदु से होकर जाने वाली रेखा) दिखाइए कि $\sum_i (y_i -
        \beta x_i)^2$ को न्यूनतम करने वाली \omterm{def:b1:multivar:partial}{प्रवणता} $\beta_0 =
        \frac{\sum x_iy_i}{\sum x_i^2}$ है, और आँकड़ों पर वह शर्त दीजिए
        जिसके अंतर्गत $\beta_0$ प्रश्न 11 की \omterm{def:b1:multivar:partial}{प्रवणता} $c_{xy}/v_x$ के
        साथ मेल खाए।
  \item (किसी रेखा तक की दूरी, फिर से) किसी बिंदु $M$ और $P_0$ से होकर
        जाने वाली, एकक सदिश $w$ से दिष्ट रेखा $D$ के लिए $g(\tau) =
        \norm{P_0 + \tau w - M}^2$ न्यूनतम कीजिए और $d(M, D)^2 =
        \norm{M - P_0}^2 - \langle M - P_0, w\rangle^2$ निकालिए; समतल
        की रेखा $ax + by + c = 0$ के लिए सूत्र $d = \frac{\abs{a x_0 +
        b y_0 + c}}{\sqrt{a^2 + b^2}}$ पुनः प्राप्त कीजिए।
  \item (प्रसरण का विश्लेषण) विघटन $v_y = \beta^2 v_x +
        \frac{E_{\min}}n$ सिद्ध कीजिए: $y_i$ का प्रसरण रेखा से
        व्याख्यायित भाग और अवशिष्ट प्रसरण में बँट जाता है।
  \item (परवलयिक फिटिंग) $y = a + bx + cx^2$ फिट करने के लिए दिखाइए कि
        \omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण} वह $3 \times 3$ निकाय हैं जिसका आघूर्ण आव्यूह
        \[
        \begin{pmatrix}
        n & \sum x_i & \sum x_i^2\\
        \sum x_i & \sum x_i^2 & \sum x_i^3\\
        \sum x_i^2 & \sum x_i^3 & \sum x_i^4
        \end{pmatrix}
        \]
        है --- एक ग्राम आव्यूह, जो $x_i$ में से तीन के भिन्न होते ही
        व्युत्क्रमणीय हो जाता है \emph{(आँकड़ों पर नमूने लिए गए $(1, X,
        X^2)$ की स्वतंत्रता, \cref{exo:b1:euclid:11};
        \cref{ch:b1:det} की सप्ताहांत समस्या के आघूर्ण आव्यूहों से
        तुलना कीजिए)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- सुदृढ़ता, और संश्लेषण।}
\begin{enumerate}[resume]
  \item तीनों द्विघात फलनों
        \[
        x^2 + xy + y^2, \qquad x^2 + 3xy + y^2, \qquad
        x^2 + 2xy + y^2
        \]
        के क्रांतिक बिंदुओं का वर्गीकरण भाग I से कीजिए, और तीसरी
        अपह्रासी स्थिति हाथ से निपटाइए (न्यूनतम कहाँ प्राप्त होता है?)।
  \item (बहिर्मान का प्रयोग) प्रश्न 14 के आँकड़ों में बिंदु $(10, 0)$
        जोड़िए और \omterm{def:b1:multivar:partial}{प्रवणता} $\beta$ फिर से संगणित कीजिए। क्या हुआ, और
        \emph{वर्ग} त्रुटि एक ही दूर बैठे बिंदु के प्रति इतनी
        संवेदनशील क्यों है?
  \item (भारित \omterm{pb:b1:multivar:1}{न्यूनतम वर्ग}) भार $w_i > 0$ दिए हों: $\sum_i w_i(y_i -
        \alpha - \beta x_i)^2$ न्यूनतम कीजिए और दिखाइए कि हल वही सूत्र
        देते हैं जो प्रश्न 11 में हैं, बशर्ते माध्य, प्रसरण और सहप्रसरण
        भारित लिए जाएँ (उन्हें परिभाषित कीजिए)।
  \item दिखाइए कि इष्टतम रेखा के लिए $E_{\min} = 0$ सभी बिंदुओं को ठीक
        उसी पर बाध्य कर देता है, और इसे प्रश्न 13 की समता-स्थिति
        $\abs\rho = 1$ से जोड़िए; संरेख आँकड़ों $(1,1), (2,3), (3,5)$ पर
        जाँचिए।
  \item संश्लेषण, चार वाक्यों में: द्विघात फलन ही वह अकेला वर्ग क्यों
        हैं जिसके लिए इस अध्याय की द्वितीय-कोटि कसौटी को किसी स्वीकृत
        प्रमेय की आवश्यकता नहीं; \omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण} विश्लेषणी न्यूनीकरण
        को \cref{ch:b1:euclid} के \omterm{thm:b1:euclid:projection}{लांबिक प्रक्षेप} से कैसे पहचान लेते
        हैं; सहसंबंध गुणांक क्या नापता है और कौन-सी असमिका उसे परिबद्ध
        करती है; और अध्याय 18--23 के कौन-से टुकड़े (\omterm{def:b1:vspaces:free}{आधार}, ग्राम तथा
        आघूर्ण आव्यूह, \omterm{def:b1:linmaps:projection}{प्रक्षेप}) फिर से प्रकट हुए। भाग II और III में
        अध्ययन की गई विधि और समीकरणों के नाम लीजिए।
\end{enumerate}
\end{problem}
