\chapter{टेलर सूत्र और अनंतस्पर्शी प्रसार}\label{ch:b1:taylor}

किसी बिंदु के पास चिकना फलन \omterm{def:b1:poly:def}{बहुपद} जितना ही अच्छा होता है --- और उसकी त्रुटि
नियंत्रित की जा सकती है। टेलर सूत्र इसे तीन स्वादों में यथार्थ कर देते हैं
(\omterm{thm:b1:integration:def}{समाकल} शेषफल, लाग्रांज शेषफल, यंग शेषफल), और उनसे मिले \emph{अनंतस्पर्शी
प्रसार}, बीजगणितीय रूप से साधे जाने पर, प्रारंभिक विश्लेषण का सबसे तीक्ष्ण
औज़ार बन जाते हैं: सीमाएँ, तुल्यताएँ, स्थानीय व्यवहार, अनंतस्पर्शी रेखाएँ।

\section{तुलना का संकेतन}

\begin{definition}[लांडाउ संकेतन]\label{def:b1:taylor:landau}
मान लीजिए $f, g$ $x_0$ ($x_0 \in \R$ या $\pm\infty$) के पास परिभाषित हैं। तब
$x \to x_0$ होने पर लिखा जाता है:\index{लांडाउ संकेतन}
\begin{itemize}
  \item $f = o(g)$ (``लघु-$o$'') तब जब $\varepsilon(x) \to 0$ के साथ $f =
  \varepsilon g$;
  \item $f = O(g)$ (``बृहत्-$O$'') तब जब $u$ $x_0$ के पास परिबद्ध होते हुए
  $f = u g$;
  \item $f \sim g$ (``तुल्य''\index{तुल्य फलन}) तब जब $\varepsilon \to 0$ के
  साथ $f = (1 + \varepsilon) g$ --- समतुल्य रूप से $f - g = o(g)$।
\end{itemize}
यही संकेतन अनुक्रमों पर भी लागू होता है ($n \to \infty$)।
\end{definition}

\begin{proposition}[नियम]\label{prop:b1:taylor:landaurules}
जब $x \to x_0$:
\begin{enumerate}
  \item $\sim$ एक \omterm{def:b1:logic:equiv}{तुल्यता संबंध} है; $f \sim g$ से यह आता है कि $f$ और $g$ की
  सीमाएँ, ($x_0$ के पास) चिह्न, और शून्यों की अनुपस्थिति एक ही हैं;
  \item तुल्यताएँ \emph{गुणा और भाग होती हैं}: $f_1 \sim g_1$, $f_2 \sim
  g_2$ से $f_1 f_2 \sim g_1 g_2$ और $\frac{f_1}{f_2} \sim \frac{g_1}{g_2}$
  आते हैं;
  \item तुल्यताएँ \emph{जुड़ती नहीं}: $+\infty$ पर $x + 1 \sim x$ और $-x
  \sim -x + 2$, फिर भी योग $1$ और $2$ तुल्य नहीं हैं। जोड़ने के लिए स्पष्ट
  $o(\cdot)$ पदों वाले प्रसारों पर लौटिए;
  \item $o(g) + o(g) = o(g)$, $\;u \cdot o(g) = o(ug)$, $\;o(o(g)) = o(g)$,
  और $f \sim g \iff f = g + o(g)$।
\end{enumerate}
\end{proposition}

\begin{proof}
हर एक परिभाषाओं की छोटी-सी हेराफेरी है; उदाहरणार्थ $f_1 f_2 =
(1+\varepsilon_1)(1+\varepsilon_2) g_1 g_2$ और $(1 +
\varepsilon_1)(1+\varepsilon_2) \to 1$। (4) के दो अंश अपनी-अपनी पंक्ति के
अधिकारी हैं। $u\cdot o(g) = o(ug)$: यदि $\varepsilon \to 0$ के साथ $f =
\varepsilon g$, तो उसी $\varepsilon$ के साथ $uf = \varepsilon\,(ug)$।
$o(o(g)) = o(g)$: यदि $f = \varepsilon_1 h$ और $h = \varepsilon_2 g$, जहाँ
दोनों $\varepsilon_i \to 0$, तो $f = (\varepsilon_1\varepsilon_2) g$ और
दोनों अतिसूक्ष्म राशियों का गुणनफल भी एक है। तुल्यता $f \sim g \iff f = g +
o(g)$ परिभाषा को दो बार पढ़ना ही है: $f - g = \varepsilon g$। और (3) का
प्रति-उदाहरण ही (3) की उपपत्ति है।
\end{proof}

\begin{example}[तुलना का पैमाना]\label{ex:b1:taylor:scale}
जब $x \to +\infty$, तब मानक पैमाना बढ़ती प्रबलता के क्रम में यह है:
\[
1 \;=\; o(\ln x), \quad
\ln x = o(x^{0.01}), \quad
x^{0.01} = o(\sqrt x), \quad
\sqrt x = o(x^{10}), \quad
x^{10} = o(\eu^{x}), \quad
\eu^x = o(\eu^{2x}) ,
\]
और हर पद \cref{prop:b1:functions:powerrules} की वृद्धि तुलनाओं का उदाहरण है
(घातें लघुगणकों को हराती हैं, चरघातांकी घातों को, और एक ही कुल के भीतर
घातांक निर्णय करता है)। दो आदतें बनाने योग्य हैं: पहली, किसी छोटे वर्ग में
गिरता $O(\cdot)$ चुपचाप उन्नत हो जाता है ($O(\ln x)$ $o(x^{0.01})$ भी है);
दूसरी, $x \to 0^+$ पर प्रतिस्थापन $x \mapsto \frac1x$ के द्वारा पूरी सीढ़ी
उलट जाती है --- वहाँ $\ln x = o(x^{-0.01})$, अतः ``$x^\alpha \ln x \to 0$''
प्रत्येक $\alpha > 0$ के लिए सत्य है। इस पैमाने को सीधा रखना
\cref{ch:b1:series} के हर अनंतस्पर्शी तर्क का आधा हिस्सा है।
\end{example}

\begin{example}[प्रसारों की अद्वितीयता, और सम-विषमता का लाभांश]\label{ex:b1:taylor:uniqueness}
यदि कोई फलन $0$ पर एक ही कोटि तक दो प्रसार स्वीकार करता है,
\[
a_0 + a_1 x + \dots + a_n x^n + o(x^n)
= b_0 + b_1 x + \dots + b_n x^n + o(x^n),
\]
तो प्रत्येक $k$ के लिए $a_k = b_k$: घटाइए और $c_k = a_k - b_k$ रखिए, फिर
सर्वसमिका $c_0 + c_1 x + \dots + c_n x^n = o(x^n)$ का मान $x \to 0$ पर लेकर
$c_0 = 0$ पाइए; $x$ से भाग दीजिए और दोहराइए --- हर भाग वैध है क्योंकि बचा
हुआ व्यंजक फिर $o(x^{n-k})$ है। अतः गुणांक आंतरिक हैं, और उन्हें \emph{किसी
भी} रास्ते से संगणित किया जा सकता है (टेलर \omterm{def:b1:derivative:def}{अवकलज}, ज्ञात प्रसारों पर बीजगणित,
समाकलन): सभी रास्तों को मेल खाना ही होगा। लाभांश: किसी \emph{सम} फलन के
प्रसार में केवल सम घातें होती हैं --- $x$ के स्थान पर $-x$ रखिए और
अद्वितीयता का आह्वान कीजिए; इसी प्रकार विषम फलनों में विषम घातें। इसीलिए
नीचे की सारणी में $\cos$ $o(x^{2p})$ के बदले $o(x^{2p+1})$ ढोता है:
अनुपस्थित विषम पद मुफ़्त की सूचना है, अर्थात् बिना कुछ दिए एक कोटि की अधिक
यथार्थता।
\end{example}

\section{तीनों टेलर सूत्र}

\begin{theorem}[समाकल शेषफल वाला टेलर]\label{thm:b1:taylor:integral}
मान लीजिए $f$ $a$ और $x$ को समेटने वाले किसी \omterm{prop:b1:reals:intervals}{अंतराल} पर $C^{n+1}$ वर्ग का है।
तब\index{टेलर सूत्र}
\[
f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}\,(x - a)^k
+ \int_a^x \frac{(x - t)^n}{n!}\, f^{(n+1)}(t)\, \dd t .
\]
\end{theorem}

\begin{proof}
$n$ पर आगमन। $n = 0$ के लिए: $f(x) = f(a) + \int_a^x f'(t)\dd t$ मूल प्रमेय
(\cref{thm:b1:integration:ftc}) है। पद: शेषफल का \omterm{thm:b1:integration:parts}{खंडशः समाकलन} कीजिए,
\[
\int_a^x \frac{(x-t)^n}{n!} f^{(n+1)}(t)\,\dd t
= \Bigl[-\frac{(x-t)^{n+1}}{(n+1)!} f^{(n+1)}(t)\Bigr]_a^x
+ \int_a^x \frac{(x-t)^{n+1}}{(n+1)!} f^{(n+2)}(t)\,\dd t ,
\]
जहाँ कोष्ठक पद $\frac{f^{(n+1)}(a)}{(n+1)!}(x - a)^{n+1}$ का योगदान करता है।
\end{proof}

\begin{example}[अपने शेषफल सहित एक यथार्थ प्रसार]\label{ex:b1:taylor:lnexact}
$\ln(1 + x)$ के लिए \omterm{thm:b1:integration:def}{समाकल} शेषफल को बिना कुछ $n$ बार अवकलित किए पूरी तरह
स्पष्ट किया जा सकता है: परिमित गुणोत्तर सर्वसमिका $\frac{1}{1+t} =
\sum_{k=0}^{n-1}(-t)^k + \frac{(-t)^n}{1+t}$ का $0$ से $x$ तक समाकलन कीजिए:
\[
\ln(1 + x) = \sum_{k=1}^{n} \frac{(-1)^{k-1}x^k}{k}
+ (-1)^n \int_0^x \frac{t^n}{1 + t}\,\dd t ,
\]
और $0 \leq x \leq 1$ के लिए शेषफल $\int_0^x t^n\,\dd t =
\frac{x^{n+1}}{n+1}$ से परिबद्ध है। यह टेलर--यंग से दो प्रकार से प्रबल है:
यह किसी नियत $x$ के लिए वैध \emph{सर्वसमिका} है (केवल $x \to 0$ के लिए
नहीं), और त्रुटि का परिबंध संख्यात्मक है। सप्ताहांत समस्या
(\cref{pb:b1:taylor:1}) ऐसे ही यथार्थ रूपों पर जीती है; नीचे का टेलर--यंग
सीमाओं के लिए हल्का औज़ार है, जहाँ केवल त्रुटि का आकार महत्त्व रखता है।
\end{example}

\begin{theorem}[टेलर--लाग्रांज असमिका]\label{thm:b1:taylor:lagrange}
मान लीजिए $f$ $C^{n+1}$ है और $a$ तथा $x$ के बीच $\abs{f^{(n+1)}} \leq M$।
तब
\[
\Bigl| f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k \Bigr|
\leq M\, \frac{\abs{x - a}^{n+1}}{(n+1)!} .
\]
\end{theorem}

\begin{proof}
\omterm{thm:b1:integration:def}{समाकल} शेषफल को परिबद्ध कीजिए: $\bigl|\int_a^x \frac{(x-t)^n}{n!}
f^{(n+1)}(t)\,\dd t\bigr| \leq M \bigl|\int_a^x \frac{\abs{x-t}^n}{n!}\dd
t\bigr| = M\frac{\abs{x-a}^{n+1}}{(n+1)!}$।
\end{proof}

\begin{example}[प्रमाणित संख्याएँ]\label{ex:b1:taylor:certified}
$\sqrt{1.02}$ क्या है? $a = 0$ पर $f(t) = \sqrt{1 + t}$ पर, कोटि $2$, $x =
0.02$ के साथ टेलर--लाग्रांज लगाइए:
\[
\sqrt{1.02} \approx 1 + \frac{0.02}{2} - \frac{0.02^2}{8}
= 1.00995 ,
\qquad
\abs{f'''(t)} = \frac{3}{8}(1+t)^{-5/2} \leq \frac 38 ,
\]
अतः त्रुटि अधिक से अधिक $\frac38 \cdot \frac{0.02^3}{6} = 5\cdot10^{-7}$ है:
$\sqrt{1.02} = 1.00995$, छह प्रमाणित दशमलव के साथ (सच्चा मान
$1.0099504938\dots$ --- परिबंध लगभग तीक्ष्ण है)। समापन का सार: टेलर--यंग
केवल यह कहता है कि त्रुटि \emph{कितनी तेज़ी से} लुप्त होती है;
टेलर--लाग्रांज उसी \omterm{def:b1:poly:def}{बहुपद} को \emph{प्रमाणपत्र} में बदल देता है --- एक संख्या
और एक सिद्ध त्रुटि-दंड। इस पुस्तक में जब भी कोई दशमलव दावा किया गया है, उसके
पीछे लाग्रांज प्रकार का कोई परिबंध खड़ा है; और सप्ताहांत समस्या
(\cref{pb:b1:taylor:1}) इसी विचार का औद्योगीकरण करती है।
\end{example}

\begin{theorem}[टेलर--यंग]\label{thm:b1:taylor:young}
मान लीजिए $f$ $a$ पर $n$ बार \omterm{def:b1:derivative:def}{अवकलनीय} है। तब $x \to a$ होने पर:
\[
f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}\,(x-a)^k
+ o\bigl((x-a)^n\bigr) .
\]
\end{theorem}

\begin{proof}
$n$ पर आगमन। $n = 1$ के लिए यह \omterm{def:b1:derivative:def}{अवकलज} की परिभाषा है
(\cref{def:b1:derivative:def})। \omterm{def:b1:logic:statement}{कथन} को कोटि $n - 1$ पर मान लीजिए, और मान
लीजिए $f$ $a$ पर $n$ बार \omterm{def:b1:derivative:def}{अवकलनीय} है। आगमन परिकल्पना $f'$ पर लगाइए (जो $a$ पर
$n-1$ बार \omterm{def:b1:derivative:def}{अवकलनीय} है):
\[
f'(t) = \sum_{k=0}^{n-1} \frac{f^{(k+1)}(a)}{k!}(t-a)^k + r(t),
\qquad r(t) = o\bigl((t-a)^{n-1}\bigr).
\]
$g(x) = f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k$ रखिए; तब $g' =
r$ और $g(a) = 0$। $\varepsilon > 0$ दिया हो, तो ऐसा $\delta$ चुनिए कि
$\abs{t-a} \leq \delta$ के लिए $\abs{r(t)} \leq \varepsilon\abs{t -
a}^{n-1}$; और $a$ से $x$ तक के खंड पर (जहाँ $\abs{g'} \leq
\varepsilon\abs{x-a}^{n-1}$) माध्य मान असमिका (\cref{thm:b1:derivative:mvt})
$\abs{g(x)} \leq \varepsilon\abs{x - a}^n$ देती है: अर्थात् ठीक $g(x) =
o((x-a)^n)$।
\end{proof}

\begin{remark}[तीन सूत्र, तीन क़ीमतें, तीन उत्पाद]
परिकल्पनाएँ ठीक निष्कर्षों के अनुपात में हैं। टेलर--यंग सबसे कम माँगता है
(केवल \emph{उसी बिंदु पर} $n$ \omterm{def:b1:derivative:def}{अवकलज}) और सबसे कम देता है: एक गुणात्मक
$o((x-a)^n)$, जो सीमाओं के लिए उत्तम और प्रमाणित अंकों के लिए निरर्थक है।
लाग्रांज असमिका \emph{\omterm{prop:b1:reals:intervals}{अंतराल} पर} $C^{n+1}$ और वहाँ परिबंध $M$ माँगती है, और
बदले में संख्यात्मक त्रुटि-दंड देती है। \omterm{thm:b1:integration:def}{समाकल} रूप वही नियमितता माँगकर सबसे
अधिक देता है: त्रुटि एक स्पष्ट वस्तु के रूप में, जिसे रूपांतरित किया जा सकता
है (\omterm{thm:b1:integration:parts}{खंडशः समाकलन}, टुकड़ों में परिबद्धन, चर-परिवर्तन) --- और यही रूप
\cref{pb:b1:integration:1} की अपरिमेयता-मशीन को शक्ति देता है। लक्ष्य को
सँभालने वाला सबसे दुर्बल सूत्र चुनना पांडित्य नहीं है: \cref{pb:b1:taylor:1}
का चपटा फलन हर कोटि पर टेलर--यंग संतुष्ट करता है, जबकि उसके विषय में हर
प्रबलतर निष्कर्ष $0$ से दूर असत्य है।
\end{remark}

\begin{proposition}[$0$ पर मानक प्रसार]
जब $x \to 0$, तब प्रत्येक नियत कोटि $n$ के लिए:\index{टेलर प्रसार!मानक}
\begin{align*}
\eu^x &= 1 + x + \frac{x^2}{2!} + \dots + \frac{x^n}{n!} + o(x^n),\\
\cos x &= 1 - \frac{x^2}{2!} + \frac{x^4}{4!} - \dots
+ \frac{(-1)^p x^{2p}}{(2p)!} + o(x^{2p+1}),\\
\sin x &= x - \frac{x^3}{3!} + \dots + \frac{(-1)^p
x^{2p+1}}{(2p+1)!} + o(x^{2p+2}),\\
\frac{1}{1 - x} &= 1 + x + x^2 + \dots + x^n + o(x^n),\\
\ln(1 + x) &= x - \frac{x^2}{2} + \frac{x^3}{3} - \dots +
\frac{(-1)^{n-1} x^n}{n} + o(x^n),\\
(1 + x)^\alpha &= 1 + \alpha x + \frac{\alpha(\alpha-1)}{2!}x^2 +
\dots + \binom{\alpha}{n} x^n + o(x^n),
\end{align*}
जहाँ वास्तविक $\alpha$ के लिए $\binom{\alpha}{n} = \frac{\alpha(\alpha -
1)\cdots(\alpha - n + 1)}{n!}$। ($\cosh$ और $\sinh$: $\cos$ जैसे ही, पर
$\sin$ में एकांतरित चिह्न नहीं।) \label{prop:b1:taylor:standard}
\end{proposition}

\begin{proof}
प्रत्येक फलन $0$ के पास चिकना है और उसके \omterm{def:b1:derivative:def}{अवकलज} आसानी से निकलते हैं:
$(\eu^x)^{(k)} = \eu^x$; $\sin$ और $\cos$ के \omterm{def:b1:derivative:def}{अवकलज} आवर्तकाल $4$ के साथ चक्कर
लगाते हैं; $\bigl((1+x)^\alpha\bigr)^{(k)} = \alpha(\alpha - 1)\cdots(\alpha
- k + 1)(1+x)^{\alpha - k}$; $\bigl(\ln(1+x)\bigr)^{(k)} =
\frac{(-1)^{k-1}(k-1)!}{(1+x)^k}$। $a = 0$ पर टेलर--यंग लगाइए। (गुणोत्तर
वाला यथार्थ है: $\frac{1}{1-x} - \sum_0^n x^k = \frac{x^{n+1}}{1 - x} =
o(x^n)$।)
\end{proof}

\begin{method}[प्रसारों के साथ संगणना]\label{met:b1:taylor:compute}
\begin{enumerate}
  \item पहले \emph{लक्ष्य कोटि $n$ नियत कीजिए}, और हर मध्यवर्ती परिणाम को
  वहीं काटिए --- ऊपर के पद ढोना व्यर्थ श्रम है, और नीचे के छोड़ देना भूल।
  \item \emph{योग, गुणनफल}: प्रत्येक गुणक को कोटि $n$ तक प्रसारित कीजिए और
  गुणा कीजिए, $x^n$ से आगे छोड़ते हुए।
  \item \emph{संयोजन} $f(u(x))$, जहाँ $u(x) \to 0$: $u$ के प्रसार को $f$ के
  प्रसार में कोटि-दर-कोटि प्रतिस्थापित कीजिए।
  \item \emph{भागफल}: $v \to 0$ के साथ $\frac{1}{1 + v}$ लिखिए और गुणोत्तर
  प्रसार का प्रयोग कीजिए।
  \item प्रसार का पद-दर-पद \emph{समाकलन} कीजिए (अवकलन के लिए अधिक सावधानी
  चाहिए; न्यायसंगति: $0$ से $x$ तक $o(t^n)$ का \omterm{thm:b1:integration:def}{समाकल} $o(x^{n+1})$ है, जो
  सीधे परिबद्धन से मिलता है)।
\end{enumerate}
\end{method}

\begin{example}[संयोजन, पूरे लेखा-जोखा के साथ]\label{ex:b1:taylor:esin}
$\eu^{\sin x}$ को कोटि $3$ तक प्रसारित कीजिए। भीतरी प्रसार: $u = \sin x = x
- \frac{x^3}{6} + o(x^3)$, जो सचमुच $0$ की ओर जाता है। बाहरी: $\eu^u = 1 + u
+ \frac{u^2}{2} + \frac{u^3}{6} + o(u^3)$, और $o(u^3) = o(x^3)$, क्योंकि $u
\sim x$। $u$ की घातें, $x^3$ पर काटी हुई:
\[
u^2 = x^2 + o(x^3), \qquad u^3 = x^3 + o(x^3)
\]
(आड़ा पद $2x\cdot(-\frac{x^3}{6})$ पहले से ही $x^4$ है)। जोड़िए:
\[
\eu^{\sin x} = 1 + \Bigl(x - \frac{x^3}{6}\Bigr) + \frac{x^2}{2}
+ \frac{x^3}{6} + o(x^3)
= 1 + x + \frac{x^2}{2} + o(x^3) :
\]
जहाँ दोनों $x^3$ योगदान ठीक-ठीक कट जाते हैं। समापन का सार: $\eu^{\sin x}$ और
$\eu^x$ कोटि $3$ तक मेल खाते हैं --- इसलिए नहीं कि स्थूल रूप से $\sin x
\approx x$, बल्कि इसलिए कि घातांकों की पहली असहमति ($-\frac{x^3}{6}$) $\eu^0
= 1$ से गुणित होकर आती है और फिर बाहरी चरघातांकी के त्रिघात पद से मिल जाती
है; कोटि-दर-कोटि लेखा-जोखा ऐसी साजिशें पकड़ लेता है, आँख से देखना कभी नहीं।
(अगला पद $-\frac{x^4}{8}$ है: संधि कोटि $4$ पर टूट जाती है।)
\end{example}

\begin{example}\label{ex:b1:taylor:tan}
कोटि $5$ पर $\tan$ का प्रसार। $\tan x = \sin x \cdot \frac{1}{\cos x}$
लिखिए:
\[
\frac{1}{\cos x} = \frac{1}{1 - \bigl(\frac{x^2}{2} - \frac{x^4}{24}
+ o(x^5)\bigr)}
= 1 + \Bigl(\frac{x^2}{2} - \frac{x^4}{24}\Bigr)
+ \Bigl(\frac{x^2}{2}\Bigr)^{\!2} + o(x^5)
= 1 + \frac{x^2}{2} + \frac{5x^4}{24} + o(x^5),
\]
फिर
\[
\tan x = \Bigl(x - \frac{x^3}{6} + \frac{x^5}{120}\Bigr)
\Bigl(1 + \frac{x^2}{2} + \frac{5x^4}{24}\Bigr) + o(x^5)
= x + \frac{x^3}{3} + \frac{2x^5}{15} + o(x^5) .
\]
\end{example}

\section{अनुप्रयोग}

\begin{example}[सीमाएँ]\label{ex:b1:taylor:limits}
\[
\lim_{x \to 0} \frac{x - \sin x}{x^3}:
\qquad
x - \sin x = \frac{x^3}{6} + o(x^3) \sim \frac{x^3}{6},
\qquad\text{अतः सीमा है } \frac16
\]
--- जिससे \cref{exo:b1:functions:9} में उठाया गया प्रश्न सुलझ जाता है। इसी
प्रकार $\displaystyle\lim_{x\to0}\Bigl(\frac{\sin x}{x}\Bigr)^{1/x^2}$:
लघुगणक है
\[
\frac{1}{x^2}\ln\Bigl(1 - \frac{x^2}{6} + o(x^2)\Bigr)
= \frac{1}{x^2}\Bigl(-\frac{x^2}{6} + o(x^2)\Bigr)
\longrightarrow -\frac16,
\qquad\text{सीमा } \eu^{-1/6}.
\]
\end{example}

\begin{remark}[प्रसारों की सामान्य भूलें]
(क) \emph{तुल्यताओं को कभी मत जोड़िए या घटाइए}: $\tan x \sim x$ और $\sin x
\sim x$ से $\tan x - \sin x \sim 0$ (जो निरर्थक है) \emph{नहीं} निकाला जा
सकता --- ईमानदार रास्ता प्रसार हैं:
\[
\tan x - \sin x
= \Bigl(x + \frac{x^3}{3}\Bigr) - \Bigl(x - \frac{x^3}{6}\Bigr)
+ o(x^3) = \frac{x^3}{2} + o(x^3) \sim \frac{x^3}{2} .
\]
(ख) \emph{नरसंहार के आगे तक प्रसारित कीजिए}: उसी संगणना में कोटि $1$ को केवल
$x - x = 0$ दिखाई देता है; जब भी अग्र पद कट जाएँ, तब तक कोटि बढ़ाइए जब तक
कोई अशून्य गुणांक बच न जाए, और तभी तुल्यता में लौटिए। (ग) \emph{तुल्यताएँ
चरघातांकी से होकर नहीं जातीं}: $n^2 + n \sim n^2$, फिर भी $\eu^{n^2+n} =
\eu^{n}\,\eu^{n^2}$ $\eu^{n^2}$ के तुल्य \emph{नहीं} है --- चरघातांकी केवल
\emph{घातांक के उन प्रसारों} पर लगाइए जिनकी त्रुटि $0$ की ओर जाती हो, घातांक
की तुल्यताओं पर कभी नहीं। (लघुगणक अधिक सुरक्षित हैं: यदि $u_n \sim v_n \to
\ell \neq 1$, $\ell > 0$, तो $\ln u_n \sim \ln v_n$।) (घ) \emph{$o(\cdot)$
का कलन एकदिशीय है}: $o(x^2) + o(x^2) = o(x^2)$, $5\,o(x^2) = o(x^2)$,
$x\cdot o(x^2) = o(x^3)$ --- पर $o(x^2)$ कोई विशिष्ट फलन नहीं है, अतः दो ऐसे
पदों को कभी आपस में मत काटिए: $o(x^2) - o(x^2)$ $o(x^2)$ है, $0$ नहीं।
\end{remark}

\begin{proposition}[स्थानीय व्यवहार]\label{prop:b1:taylor:local}
मान लीजिए $c \neq 0$ के साथ $f(x) = f(a) + c\,(x - a)^p +
o\bigl((x-a)^p\bigr)$ (अचर के बाद पहला अशून्य पद; किसी क्रांतिक बिंदु पर $p
\geq 2$)।
\begin{itemize}
  \item यदि $p$ सम है: $c > 0$ होने पर $f$ का $a$ पर स्थानीय न्यूनतम है, और
  $c < 0$ होने पर स्थानीय अधिकतम।
  \item यदि $p$ विषम है: कोई चरम नहीं ($f - f(a)$ चिह्न बदलता है); और यदि
  इसके अतिरिक्त प्रसार किसी रैखिक पद $f'(a)(x - a)$ के बाद आरंभ हो, तो आलेख
  अपनी स्पर्श रेखा को काट देता है: अर्थात् \emph{नति-परिवर्तन}।
\end{itemize}
\end{proposition}

\begin{proof}
$a$ के पास $f(x) - f(a) = (x-a)^p\bigl(c + o(1)\bigr)$ का चिह्न $c\,(x-a)^p$
का चिह्न होता है: $p$ सम होने पर अचर चिह्न, और $p$ विषम होने पर बदलता हुआ।
\end{proof}

\begin{example}[घातांकों को $o(1)$ तक प्रसारित करना पड़ता है]\label{ex:b1:taylor:expdiscipline}
$u_n = \bigl(1 + \frac1n\bigr)^{n^2}$ की तुल्यता खोजिए। \emph{घातांक} को तब
तक प्रसारित कीजिए जब तक उसकी त्रुटि $0$ की ओर न जाने लगे:
\[
n^2 \ln\Bigl(1 + \frac1n\Bigr)
= n^2\Bigl(\frac1n - \frac{1}{2n^2} +
O\Bigl(\frac{1}{n^3}\Bigr)\Bigr)
= n - \frac12 + O\Bigl(\frac1n\Bigr),
\]
अतः $\eu^{O(1/n)} \to 1$ के साथ $u_n = \eu^{\,n - 1/2}\,\eu^{O(1/n)}$:
\[
u_n \;\sim\; \eu^{\,n - \frac12} .
\]
ध्यान दीजिए कि कम सावधानी से क्या ग़लत हो जाता: घातांक को $n^2\cdot\frac1n =
n + O(1)$ पर रोक देने से $\eu^{O(1)}$ गुणक बच जाता है --- जो परिबद्ध तो है
पर $1$ की ओर नहीं जाता --- और तब कोई तुल्यता नहीं कही जा सकती। ऊपर की भूलों
वाला नियम, धनात्मक रूप में: $\eu^{a_n}$ की तुल्यता के लिए $a_n$ का प्रसार
\emph{शून्य की ओर जाते किसी पद तक} चाहिए, और उससे पहले का हर गुणांक यथार्थ
रखना पड़ता है।
\end{example}

\begin{example}[किसी चपटे क्रांतिक बिंदु का वर्गीकरण]\label{ex:b1:taylor:flatmin}
$0$ के पास $f(x) = \cos x + \frac{x^2}{2}$ का अध्ययन कीजिए। $f'(0) = 0$ और
$f''(0) = -\cos 0 + 1 = 0$ दोनों: द्वितीय-अवकलज परीक्षण मौन है। इसके बदले
प्रसार कीजिए:
\[
f(x) = \Bigl(1 - \frac{x^2}{2} + \frac{x^4}{24} +
o(x^4)\Bigr) + \frac{x^2}{2}
= 1 + \frac{x^4}{24} + o(x^4) :
\]
पहला अशून्य पद $c\,x^p$, जहाँ $p = 4$ सम है और $c = \frac{1}{24} > 0$: अतः
स्थानीय न्यूनतम, और वह भी असाधारण चपटेपन का (आलेख अपने न्यूनतम मान को $x^2$
की तरह नहीं, $x^4$ की तरह छोड़ता है)। समापन का सार: प्रसार एक ही पंक्ति में
वह देख लेता है जिसे दोहराया हुआ अवकलन धुँधला कर देता है --- और
\cref{prop:b1:taylor:local} ``पहला बचा हुआ पद'' से ``स्थानीय आकार'' तक का
व्यवस्थित शब्दकोश है।
\end{example}

\begin{example}[अनंत पर प्रसार]\label{ex:b1:taylor:atinfinity}
दो संगणनाएँ जहाँ चर $+\infty$ की ओर भागता है और प्रतिस्थापन $h = \frac1x \to
0^+$ पूरा औज़ार-संदूक आयात कर देता है। पहली,
\[
x - x^2\ln\Bigl(1 + \frac1x\Bigr)
= x - x^2\Bigl(\frac1x - \frac{1}{2x^2} +
O\Bigl(\frac{1}{x^3}\Bigr)\Bigr)
= \frac12 + O\Bigl(\frac1x\Bigr) \longrightarrow \frac12 .
\]
दूसरी, अनंत पर \omterm{def:b1:functions:arc}{प्रतिलोम स्पर्शज्या}: $x > 0$ के लिए $\arctan x +
\arctan\frac1x = \frac\pi2$ (\cref{prop:b1:functions:arcidentities}) से और
$0$ पर $\arctan$ के प्रसार (\cref{exo:b1:taylor:3}) से,
\[
\arctan x = \frac\pi2 - \arctan\frac1x
= \frac\pi2 - \frac1x + \frac{1}{3x^3} +
o\Bigl(\frac{1}{x^3}\Bigr) :
\]
अर्थात् आलेख अपनी अनंतस्पर्शी रेखा $y = \frac\pi2$ के पास नीचे से, चाल
$\frac1x$ से, आता है। समापन का सार: अनंत पर प्रसारों का कोई अलग सिद्धांत है
ही नहीं --- एक व्युत्क्रम प्रतिस्थापन उन्हें $0$ पर के प्रसारों पर ले आता
है, बशर्ते हर मध्यवर्ती $O$ और $o$ ईमानदारी से साथ ले जाया जाए।
\end{example}

\begin{example}[प्रसार से अनंतस्पर्शी रेखा]\label{ex:b1:taylor:asymptote}
जब $x \to +\infty$,
\[
\sqrt{x^2 + x} = x\sqrt{1 + \tfrac1x}
= x\Bigl(1 + \frac{1}{2x} - \frac{1}{8x^2} + o\bigl(\tfrac{1}{x^2}\bigr)\Bigr)
= x + \frac12 - \frac{1}{8x} + o\bigl(\tfrac 1x\bigr):
\]
अतः रेखा $y = x + \frac12$ अनंतस्पर्शी है, और वक्र उसके पास \emph{नीचे से}
आता है (अगला पद $-\frac{1}{8x}$ ऋणात्मक है)।
\end{example}

\begin{remark}[प्रसार आगे कहाँ काम करते हैं]
अनंतस्पर्शी प्रसार शेष पुस्तक की स्थायी भाषा हैं: \cref{ch:b1:series} में वे
अभिसरण तय करते हैं (तुल्यताएँ तुलना-कसौटियों को पोषित करती हैं, और $\sum
\frac{1}{n^\alpha}$ का अध्ययन वेश बदला प्रसार है); स्नातक वर्ष 2 के खंड में
वे \emph{घात श्रेणी} बन जाते हैं, जहाँ टेलर \omterm{def:b1:poly:def}{बहुपद} अनंत पद और एक
अभिसरण-त्रिज्या पा लेता है; और भौतिकी का हर रैखिकीकरण --- लोलक, प्रथम कोटि
का विक्षोभ --- $o(\cdot)$ को चुपचाप गिरा देने वाला टेलर--यंग \omterm{def:b1:logic:statement}{कथन} है। उकेरने
योग्य एकमात्र चेतावनी: प्रसार किसी फलन का वर्णन केवल \emph{एक बिंदु के पास}
करता है --- सप्ताहांत समस्या का चपटा फलन देखिए, जिसका $0$ पर प्रसार तत्समक
रूप से शून्य है, यद्यपि फलन वैसा नहीं।
\end{remark}

\begin{remark}[इस खंड के भीतर आगे की दृष्टि]
प्रसार शेष विश्लेषण की और आने वाली ज्यामिति की काम की भाषा हैं।
\Cref{ch:b1:series} उन्हें अभिसरण के निर्णयों में बदल देता है: व्यापक पद की
तुल्यता अपने पहले पद पर काटा हुआ प्रसार ही है, और सूक्ष्मतर कसौटियाँ
(त्रुटि-नियंत्रण सहित एकांतरित) दूसरा पद भी खा जाती हैं। \Cref{ch:b1:curves}
दोनों निर्देशांक फलनों के प्रसारों से स्थानीय ज्यामिति पढ़ता है: कोई
प्राचलित वक्र किसी बिंदु पर कटता है, चूमता है या उभयाग्र बनाता है, यह इससे
तय होता है कि $x(t)$ और $y(t)$ में $t$ की कौन-सी घातें बच रहती हैं ---
अर्थात् \cref{prop:b1:taylor:local} का समतलीय रूप। और \cref{ch:b1:multivar}
जान-बूझकर कोटि एक पर रुक जाता है: स्पर्श तल दो चरों का टेलर--यंग \omterm{def:b1:logic:statement}{कथन} है, और
पूरा द्वितीय-कोटि सिद्धांत (हेस्सी आव्यूह, काठी बिंदु) स्नातक वर्ष 2 के खंड
के लिए स्थगित है। साझा धागा: इस पुस्तक के हर ``स्थानीय'' प्रश्न का उत्तर
किसी प्रसार का पहला बचा हुआ पद लिख देने से मिलता है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:taylor:1}
$0$ पर प्रसार दीजिए: $\eu^{2x}$ कोटि $3$ तक; $\;\ln(1 - x)$ कोटि $4$ तक;
$\;\sqrt{1 + x}$ कोटि $3$ तक; $\;\dfrac{1}{1 + x^2}$ कोटि $6$ तक।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:taylor:2}
सीमाएँ संगणित कीजिए:
\[
\lim_{x\to 0} \frac{\eu^x - 1 - x}{x^2},
\qquad
\lim_{x\to 0} \frac{\cos x - \sqrt{1 - x^2}}{x^4},
\qquad
\lim_{x\to 0} \frac{\ln(1+x) - \sin x}{x^2}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:taylor:3}
$\frac{1}{1 + x^2}$ के प्रसार का समाकलन करके $0$ पर $\arctan x$ का कोटि $5$
तक प्रसार कीजिए, और $(1 - x^2)^{-1/2}$ के प्रसार का समाकलन करके $\arcsin x$
का कोटि $5$ तक।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:taylor:4}
$\intcc{0}{1}$ पर $\exp$ के लिए टेलर--लाग्रांज का प्रयोग करके सिद्ध कीजिए कि
\[
\Bigl| \eu - \sum_{k=0}^{n} \frac{1}{k!} \Bigr| \leq
\frac{3}{(n+1)!},
\]
और ऐसा $n$ निर्धारित कीजिए जो $\eu$ के $6$ यथार्थ दशमलव की गारंटी दे।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:5}
$\frac1n$ में कोटि $2$ तक प्रसार कीजिए और उससे सीमा तथा अभिसरण की गति
निकालिए:
\[
\Bigl(1 + \frac 1n\Bigr)^{\!n}
= \eu\Bigl(1 - \frac{1}{2n} + \frac{11}{24n^2} +
o\Bigl(\frac{1}{n^2}\Bigr)\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:6}
$0$ पर $f(x) = x^2 - x^4$ के और $g(x) = x^3 + x^5$ के स्थानीय व्यवहार का
अध्ययन कीजिए; और $a = 1$ पर $h(x) = \eu^x$ के आलेख की उसकी स्पर्श रेखा के
सापेक्ष स्थिति पहले स्थानीय रूप से, फिर समग्र रूप से खोजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:7}
$f(x) = \sqrt[3]{x^3 + x^2}$ की $\pm\infty$ पर अनंतस्पर्शी रेखाएँ और उनके
सापेक्ष वक्र की स्थिति निर्धारित कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:8}
जब $n \to \infty$, तब इनकी तुल्यता खोजिए
\[
u_n = \sqrt{n+1} - \sqrt n, \qquad
v_n = \ln(n+1) - \ln n, \qquad
w_n = \sin\frac{1}{n} - \tan\frac{1}{n},
\]
और हर एक को $n$ की किसी घात गुना किसी अचर के रूप में लिखिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:9}
मान लीजिए $f$ $\R$ पर $C^2$ है। सिद्ध कीजिए कि प्रत्येक $x$ और $h > 0$ के
लिए:
\[
\abs{f'(x)} \leq \frac{\abs{f(x+h) - f(x-h)}}{2h} +
\frac{h}{2}\sup_{\intcc{x-h}{x+h}}\abs{f''} ,
\]
और उससे लांडाउ--कोल्मोगोरोव प्रकार की असमिका निकालिए: यदि $\R$ पर $\abs f
\leq M_0$ और $\abs{f''} \leq M_2$, तो सर्वत्र $\abs{f'} \leq \sqrt{2 M_0
M_2}$। \emph{($h$ पर इष्टतमीकरण कीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:10}
अनुक्रम $u_0 \in \intoo{0}{\pi}$, $u_{n+1} = \sin u_n$ घटकर $0$ तक जाता है
(संक्षेप में न्यायसंगत ठहराइए)। उसकी गति खोजने के लिए $v_n =
\frac{1}{u_n^2}$ पर विचार कीजिए:
\begin{enumerate}
  \item $\sin$ के प्रसार का प्रयोग करके सिद्ध कीजिए $v_{n+1} - v_n \to
  \frac13$;
  \item चेज़ारो (\cref{exo:b1:seq:10}) के साथ $\frac{v_n}{n} \to \frac13$
  निकालिए, फिर तुल्यता $u_n \sim \sqrt{\dfrac{3}{n}}$।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:11}
(अनंतताओं के अंतर) संगणना कीजिए
\[
\lim_{x \to 0} \Bigl(\frac{1}{x^2} - \frac{1}{\sin^2 x}\Bigr)
\qquad\text{और}\qquad
\lim_{x \to 0^+} \Bigl(\frac 1x - \frac{1}{\ln(1 + x)}\Bigr)
\]
इसके लिए उभयनिष्ठ हर पर लाइए और अंश तथा हर को अलग-अलग प्रसारित कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:12}
(अंतर्निहित मूलों का अनंतस्पर्शी) दिखाइए कि प्रत्येक $k \in \N^*$ के लिए
समीकरण $\tan x = x$ का $\intoo{k\pi - \frac\pi2}{k\pi + \frac\pi2}$ में ठीक
एक हल $x_k$ है, कि $\varepsilon_k = \arctan\frac{1}{x_k}$ के साथ $x_k = k\pi
+ \frac\pi2 - \varepsilon_k$, और उससे प्रसार निकालिए
\[
x_k = k\pi + \frac\pi2 - \frac{1}{k\pi} + o\Bigl(\frac
1k\Bigr) \qquad (k \to \infty).
\]
\end{exercise}

\section{समस्या: एकांतरित योग, प्रमाणित अंक, और $\cos 1$ की अपरिमेयता}

\begin{problem}[{सप्ताहांत समस्या --- एकांतरित आकलन $\abs{S - S_n} \leq a_{n+1}$: सिद्ध दशमलवों के साथ $\ln 2$ और $\pi$,
मैकिन का सूत्र, और $\cos 1 \notin \Q$}]
\label{pb:b1:taylor:1}
घटते पदों वाला एकांतरित योग संख्यात्मक विश्लेषण की सबसे मैत्रीपूर्ण वस्तु
है: उसकी त्रुटि पहले छोड़े गए पद से परिबद्ध है, और वह भी \emph{ज्ञात चिह्न
के साथ}। यह समस्या उस सिद्धांत को संलग्न-अनुक्रम प्रमेय से सिद्ध करती है,
फिर उसे तीन तरीक़ों से ख़र्च करती है: $\ln 2$ के लिए प्रमाणित दशमलव (तीन
प्रतिस्पर्धी रास्ते) और $\pi$ के लिए (लाइब्निज़, फिर मैकिन का 1706 का सूत्र,
जो सदियों तक कीर्तिमान संगणनाओं के पीछे का विचार बना रहा), $\cos 1$, $\sin
1$ और $\cosh 1$ की अपरिमेयता, और प्रतिभार के रूप में टेलर--लाग्रांज
\emph{समता} तथा वह चपटा फलन जिसका टेलर प्रसार झूठ बोलता है।\index{एकांतरित
श्रेणी का आकलन}\index{मैकिन का सूत्र}\index{अपरिमेयता!$\cos 1$ की} आगे
सर्वत्र ``श्रेणी'' की भाषा अनौपचारिक है: यहाँ हर योग \emph{आंशिक योगों का
अनुक्रम} है, जैसा \cref{ex:b1:seq:e} में है; सिद्धांत स्वयं
\cref{ch:b1:series} में खुलता है।

\medskip \noindent\textbf{भाग I --- एकांतरित आकलन।} मान लीजिए $(a_k)_{k \geq
0}$ घटकर $0$ तक जाता है और $S_n = \sum_{k=0}^{n} (-1)^k a_k$।
\begin{enumerate}
  \item दिखाइए कि $(S_{2n+1})$ अह्रासमान है, $(S_{2n})$ अवर्धमान, और वे
  संलग्न हैं (\cref{thm:b1:seq:adjacent}): दोनों एक उभयनिष्ठ $S$ की ओर
  अभिसरित होते हैं, जहाँ प्रत्येक $n$ के लिए
        \[
        S_{2n+1} \leq S \leq S_{2n},
        \qquad
        \abs{S - S_n} \leq a_{n+1} ,
        \]
        और त्रुटि का चिह्न पहले छोड़े गए पद का चिह्न है। इसके अतिरिक्त
        दिखाइए कि यदि कमी \emph{कठोर} हो, तो ये सभी असमिकाएँ कठोर हैं।
  \item पहला लाभांश: चरघातांकी श्रेणी में $x = 1$ के लिए $a = 0$ पर
  \cref{thm:b1:taylor:lagrange} से तुलना कीजिए: दिखाइए कि $T_n =
  \sum_{k=0}^{n} \frac{(-1)^k}{k!}$ $\eu^{-1}$ की ओर अभिसरित होता है, जहाँ
  $\abs{\eu^{-1} - T_n} < \frac{1}{(n+1)!}$।
  \item (लाइब्निज़, 1674) \emph{यथार्थ} परिमित सर्वसमिका
        \[
        \frac{1}{1 + t^2} = \sum_{k=0}^{n} (-1)^k t^{2k}
        + \frac{(-1)^{n+1} t^{2n+2}}{1 + t^2} ,
        \]
        का $\intcc{0}{1}$ पर समाकलन करके सिद्ध कीजिए
        \[
        \frac\pi4 = \sum_{k=0}^{n} \frac{(-1)^k}{2k+1} + \rho_n,
        \qquad \abs{\rho_n} \leq \frac{1}{2n+3} .
        \]
  \item धीमापन: $\pi$ के छह यथार्थ दशमलव की गारंटी के लिए लाइब्निज़ के कितने
  पद चाहिए? (लगभग बीस लाख।) पीड़ा महसूस करने के लिए $4 S_4 = 4\bigl(1 -
  \frac13 + \frac15 - \frac17 + \frac19\bigr)$ और $\pi$ से उसकी दूरी का मान
  निकालिए।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- $\ln 2$ तीन तरीक़ों से।}
\begin{enumerate}[resume]
  \item (तरीक़ा 1: एकांतरित हरात्मक) $\intcc{0}{1}$ पर समाकलित
  $\frac{1}{1+t} = \sum_{k=0}^{n-1}(-1)^k t^k + \frac{(-1)^n t^n}{1+t}$ से:
        \[
        \ln 2 = \sum_{k=1}^{n} \frac{(-1)^{k-1}}{k} + (-1)^n
        R_n, \qquad \frac{1}{2(n+1)} \leq R_n \leq
        \frac{1}{n+1} :
        \]
        त्रुटि का यथार्थ क्रम $\frac 1n$ है --- अर्थात् छह दशमलव के लिए दस
        लाख पद।
  \item (तरीक़ा 2: तेज़ श्रेणी) $\frac{1}{1 - t^2} = \sum_{k=0}^{n} t^{2k} +
  \frac{t^{2n+2}}{1-t^2}$ का $0$ से $x \in \intoo{0}{1}$ तक समाकलन कीजिए और
  $x = \frac13$ पर मान लीजिए (ध्यान दीजिए $\frac{1 + 1/3}{1 - 1/3} = 2$):
        \[
        \ln 2 = 2\sum_{k=0}^{n} \frac{(1/3)^{2k+1}}{2k+1} +
        \tilde\rho_n,
        \qquad
        0 < \tilde\rho_n \leq \frac{9}{4}\cdot
        \frac{(1/3)^{2n+3}}{2n+3} :
        \]
        अर्थात् गुणोत्तर अभिसरण, लगभग एक अंक प्रति पद।
  \item (तरीक़ा 3: \omterm{thm:b1:integration:riemann}{रीमान योग} और एक छिपी सर्वसमिका) आगमन से सर्वसमिका सिद्ध
  कीजिए
        \[
        \sum_{k=1}^{2n} \frac{(-1)^{k-1}}{k} = H_{2n} - H_n =
        \sum_{k=1}^{n} \frac{1}{n + k} ,
        \]
        और \cref{ex:b1:integration:riemannexample} की रीमान-योग सीमा के रूप
        में $\ln 2$ वापस पाइए: तरीक़ा 1 और 3 गुप्त रूप से वही संख्या दो बार
        देखे जाने पर हैं।
  \item छह पदों पर मुक़ाबला: $\sum_{k=1}^{6} \frac{(-1)^{k-1}}{k} = 0.6167$
  की तुलना $n = 5$ पर तरीक़ा 2 से कीजिए, जो पहले ही $\leq 1.1\cdot10^{-7}$
  त्रुटि के साथ $\ln 2 = 0.693147$ दे देता है। संरचनात्मक कारण समझाइए
  (अभिसरण-अंतराल के भीतर गहरा मूल्यांकन-बिंदु बनाम उसकी सीमा पर)।
  \item $\ln 2$ के \emph{दस} दशमलव प्रमाणित करने के लिए तरीक़ा 2 के कितने पद
  चाहिए? दिखाइए कि $n = 10$ पर्याप्त है।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- मैकिन का सूत्र।}
\begin{enumerate}[resume]
  \item $(5 + \iu)^4$ संगणित कीजिए और सम्मिश्र सर्वसमिका सत्यापित कीजिए
        \[
        (5 + \iu)^4 = 2\,(1 + \iu)\,(239 + \iu) .
        \]
        कोणांक लेकर (\cref{ch:b1:complex} परिपाटियाँ) मैकिन का सूत्र निकालिए
        \[
        \frac\pi4 = 4\arctan\frac15 - \arctan\frac{1}{239} .
        \]
        (जाँचिए कि कोई कोणांक $\intoo{-\frac\pi2}{\frac\pi2}$ से बाहर नहीं
        जाता।)
  \item प्रश्न 3 की भाँति $0 < x < 1$ के लिए स्थापित कीजिए:
        \[
        \arctan x = \sum_{k=0}^{n} \frac{(-1)^k
        x^{2k+1}}{2k+1} + r_n(x), \qquad \abs{r_n(x)} \leq
        \frac{x^{2n+3}}{2n+3} .
        \]
  \item छह पदों से $\pi$ को सात दशमलव तक प्रमाणित कीजिए: की कुल त्रुटि को
        \[
        \pi \approx 16\sum_{k=0}^{4}
        \frac{(-1)^k (1/5)^{2k+1}}{2k+1}
        - 4\Bigl(\frac{1}{239} - \frac{1}{3\cdot239^3}\Bigr)
        \]
        $16\,\frac{(1/5)^{11}}{11} + 4\,\frac{(1/239)^5}{5} < 5\cdot10^{-8}$
        से परिबद्ध कीजिए, और परिणामी मान $3.1415926\dots$ दीजिए।
  \item अब उपलब्ध $\pi$ के तीनों रास्तों की तुलना कीजिए --- लाइब्निज़
  (प्रश्न 4), \cref{pb:b1:integration:1} के डाल्ज़ेल \omterm{thm:b1:integration:def}{समाकल} (त्रुटि
  $4^{1-5m}$), मैकिन (त्रुटि $\approx 16\cdot 5^{-(2n+3)}$) --- प्रति पद
  अंकों में, और समझाइए कि मूल्यांकन-बिंदु को छोटा करना सबसे अच्छा क्यों है।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- पूर्णांक-जाल, एकांतरित संस्करण।}
\begin{enumerate}[resume]
  \item मान लीजिए $\cos 1 = \frac pq$। $\sum_{k} \frac{(-1)^k}{(2k)!}$ के
  कठोर एकांतरित घेराव (प्रश्न 1) को $2n \geq q$ वाले $(2n)!$ से गुणा कीजिए,
  और विरोधाभास निकालिए: $\cos 1$ अपरिमेय है।
  \item $\sin 1 = \sum_k \frac{(-1)^k}{(2k+1)!}$ के लिए ढालिए ($(2n+1)!$ से
  गुणा कीजिए): $\sin 1 \notin \Q$। दोनों अपरिमेय, फिर भी $\cos^2 1 + \sin^2
  1 = 1$: अर्थात् अपरिमेयता बीजगणित के अंतर्गत स्थायी नहीं है।
  \item अनेकांतरित चचेरा भाई: $\cosh 1 = \sum_k \frac{1}{(2k)!}$ (आंशिक-योग
  के अर्थ में, दुतरफ़ा पूँछ-परिबंध $0 < \cosh 1 - \sum_{k \leq n}
  \frac{1}{(2k)!} < \frac{2}{(2n+2)!}$ के साथ, जिसे सिद्ध करना है)। उसी जाल
  से $\cosh 1 \notin \Q$ निष्कर्ष निकालिए।
  \item प्रत्येक पूर्णांक $m \geq 1$ के लिए $\cos\frac 1m$ तक बढ़ाइए:
  $m^{2n}(2n)!$ से गुणा कीजिए और $\cos\frac1m \notin \Q$ निष्कर्ष निकालिए।
  व्यापक $b > 1$ के साथ $\cos\frac ab$ के लिए वही प्रयास कहाँ टूट जाता है?
  (वह हर पहचानिए जो अब नहीं कटता।)
\end{enumerate}

\medskip \noindent\textbf{भाग V --- तीक्ष्णतर और गहरा: समता-रूप, और वह फलन
जो टेलर को धोखा देता है।}
\begin{enumerate}[resume]
  \item (टेलर--लाग्रांज, समता-रूप) मान लीजिए $f$ $a$ और $x$ के बीच $n + 1$
  बार \omterm{def:b1:derivative:def}{अवकलनीय} है। $g(t) = f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(t)}{k!}(x -
  t)^k - A\,\frac{(x-t)^{n+1}}{(n+1)!}$ परिभाषित कीजिए, जहाँ अचर $A$ ऐसा
  चुना गया हो कि $g(a) = 0$। $g'$ संगणित कीजिए (योग दूरबीनी हो जाता है),
  $\intcc{a}{x}$ पर रोल लगाइए, और निष्कर्ष निकालिए कि $a$ तथा $x$ के बीच
  कठोरतः स्थित ऐसा $c$ विद्यमान है कि
        \[
        f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k
        + \frac{f^{(n+1)}(c)}{(n+1)!}\,(x-a)^{n+1} .
        \]
  \item समता का लाभांश: $x > 0$ के लिए दिखाइए
        \[
        \eu^x > 1 + x + \frac{x^2}{2!} + \dots +
        \frac{x^n}{n!}
        \]
        (और वह भी प्रत्येक $n$ के लिए कठोरतः), और बताइए कि $n$ की सम-विषमता
        के अनुसार $x < 0$ के लिए असमिका कहाँ उलट जाती है।
  \item कोटि $3$ पर $\sin(0.5)$ पर शेषफलों का मानदंड लीजिए: यंग केवल
  $o(x^3)$ देता है (कोई संख्या नहीं); लाग्रांज $\abs{\sin 0.5 - (0.5 -
  \frac{0.5^3}{6})} \leq \frac{0.5^5}{120} = 2.61\cdot10^{-4}$ देता है; और
  एकांतरित आकलन वही परिबंध \emph{और साथ में} चिह्न की सूचना $\sin 0.5 > 0.5
  - \frac{0.5^3}{6}$ देता है। सच्ची त्रुटि $2.59\cdot10^{-4}$ से तुलना
  कीजिए: परिबंध लगभग प्राप्त हो जाता है। आप कौन-सा औज़ार, और कब, उठाएँगे?
  \item (चपटा फलन) $x \neq 0$ के लिए $f(x) = \eu^{-1/x^2}$, और $f(0) = 0$
  रखिए। दिखाइए कि $f$ $0$ पर \omterm{def:b1:continuity:continuous}{संतत} है, कि $f'(0) = 0$, और अधिक व्यापक रूप से
  --- यह सिद्ध करके कि प्रत्येक \omterm{def:b1:derivative:def}{अवकलज} किसी \omterm{def:b1:poly:def}{बहुपद} $P_k$ के लिए $f^{(k)}(x) =
  P_k\bigl(\frac1x\bigr)\eu^{-1/x^2}$ रूप का है (आगमन) --- कि सभी $k$ के लिए
  $f^{(k)}(0) = 0$ \emph{(\omterm{prop:b1:functions:powerrules}{वृद्धि तुलना}
  \cref{prop:b1:functions:powerrules})}। निष्कर्ष: $0$ पर $f$ के सभी टेलर
  \omterm{def:b1:poly:def}{बहुपद} लुप्त हो जाते हैं, फिर भी $x \neq 0$ के लिए $f(x) > 0$: अर्थात्
  टेलर--यंग हर कोटि पर सत्य है और $0$ से दूर $f$ के विषय में कुछ नहीं कहता।
  प्रसार \emph{अंकुरों} का वर्णन करते हैं, फलनों का नहीं।
\end{enumerate}

\medskip \noindent\textbf{भाग VI --- संश्लेषण।}
\begin{enumerate}[resume]
  \item जाल को एक बार और, $\eu^{-1} = \sum_k \frac{(-1)^k}{k!}$ पर चलाइए:
  कठोर एकांतरित घेराव को $n!$ से गुणा कीजिए और $\eu^{-1} \notin \Q$, अतः
  $\eu \notin \Q$ निष्कर्ष निकालिए --- इस खंड में इस तथ्य की तीसरी उपपत्ति।
  तीनों गिनाइए (\omterm{thm:b1:seq:adjacent}{संलग्न अनुक्रम}, \cref{exo:b1:seq:9}; \omterm{thm:b1:integration:def}{समाकल},
  \cref{pb:b1:integration:1}; एकांतरित योग, यहाँ) और हर एक को क्या चाहिए था।
  \item महीन शर्त: एकदिष्टता सजावट नहीं है। विषम $k$ के लिए $b_k = \frac1k$
  और सम $k$ के लिए $b_k = \frac{1}{k^2}$ लीजिए: $b_k$ धनात्मक हैं और $0$ की
  ओर जाते हैं, फिर भी $\sum (-1)^k b_k$ के आंशिक योग $-\infty$ की ओर अपसरित
  होते हैं। इसे सिद्ध कीजिए \emph{(आंशिक योग को सम भाग, जो
  \cref{ex:b1:seq:basel} से परिबद्ध है, और विषम भाग, जो हरात्मक श्रेणी के
  आधे पर हावी है \cref{exo:b1:seq:5}, में तोड़िए)}, और ठीक-ठीक बताइए कि
  प्रश्न 1 के किस पद ने एकदिष्टता प्रयुक्त की थी।
  \item $(2 + \iu)(3 + \iu) = 5(1 + \iu)$ के द्वारा ऑयलर की सरल सर्वसमिका
  $\arctan\frac12 + \arctan\frac13 = \frac\pi4$ सत्यापित कीजिए, इस रास्ते से
  $\pi$ के छह दशमलव के लिए आवश्यक पदों का आकलन कीजिए ($n = 10$ पर्याप्त है),
  और उसे प्रश्न 13 की क्रम-सूची में लाइब्निज़ और मैकिन के बीच रखिए।
  \item संश्लेषण, एक-एक वाक्य में: (क) एकांतरित आकलन और उसके दोनों उत्पाद
  (परिबंध और चिह्न) कहिए; (ख) प्रमाणित संख्याओं के लिए स्पष्ट शेषफल वाली
  यथार्थ परिमित सर्वसमिकाएँ सीमा-कथनों से बेहतर क्यों हैं; (ग) समस्या की
  सूची ($10^{-7}$ तक हाथ से $\pi$, दस दशमलव तक $\ln 2$, चार अपरिमेयता
  उपपत्तियाँ, एक समता प्रमेय, एक चेतावनी-उदाहरण); (घ) इनमें से कौन-से धागे
  \cref{ch:b1:series} उठाएगा (एकांतरित श्रेणी कसौटी, निरपेक्ष बनाम सप्रतिबंध
  अभिसरण, और उसकी सप्ताहांत समस्या का पुनर्व्यवस्था-नाटक)।
\end{enumerate}
\end{problem}
