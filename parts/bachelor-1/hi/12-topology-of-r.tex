\chapter{वास्तविक रेखा की सांस्थिति}\label{ch:b1:topology}

सीमाएँ बार-बार उसी ज्यामितीय शब्दावली की ओर लौटती हैं: किसी \omterm{def:b1:logic:sets}{समुच्चय} के
``निकट'' बिंदु, ``बिना सीमा-रिसाव'' वाले \omterm{def:b1:logic:sets}{समुच्चय}, ऐसे \omterm{prop:b1:reals:intervals}{अंतराल} जिनसे अनुक्रम
भाग नहीं सकते। यह अध्याय उस शब्दावली को --- \omterm{def:b1:topology:open}{विवृत} और \omterm{def:b1:topology:closed}{संवृत समुच्चय}, \omterm{def:b1:topology:closure}{अंतःभाग}
और \omterm{def:b1:topology:closure}{संवरक}, \omterm{def:b1:topology:dense}{सघनता} --- वास्तविक रेखा पर तय करता है, और खंडों की संहतता उसके
अनुक्रमीय रूप में सिद्ध करता है। यही धारणाएँ, मानकित सदिश समष्टियों में,
द्वितीय वर्ष की सामग्री हैं; $\R$ पर वे पहुँच के भीतर हैं और
\cref{ch:b1:continuity} के लिए तुरंत उपयोगी।

\section{विवृत समुच्चय, संवृत समुच्चय}

\begin{definition}[प्रतिवेश, विवृत समुच्चय]\label{def:b1:topology:open}
\omterm{def:b1:logic:sets}{समुच्चय} $V \subseteq \R$ $x \in \R$ का \emph{प्रतिवेश}\index{प्रतिवेश} तब है
जब उसमें किसी $r > 0$ के लिए \omterm{prop:b1:reals:intervals}{अंतराल} $\intoo{x - r}{x + r}$ समाया हो। \omterm{def:b1:logic:sets}{समुच्चय}
$U \subseteq \R$ \emph{विवृत}\index{विवृत समुच्चय} तब है जब वह अपने प्रत्येक
बिंदु का प्रतिवेश हो:
\[
\forall x \in U,\ \exists r > 0, \quad \intoo{x - r}{x + r}
\subseteq U .
\]
\end{definition}

\begin{example}\label{ex:b1:topology:open}
\omterm{def:b1:topology:open}{विवृत} \omterm{prop:b1:reals:intervals}{अंतराल} \omterm{def:b1:topology:open}{विवृत} हैं: $x \in \intoo{a}{b}$ के लिए $r = \min(x - a,\, b -
x) > 0$ लीजिए। अर्धरेखाएँ $\intoo{a}{+\infty}$ \omterm{def:b1:topology:open}{विवृत} हैं; $\R$ और
$\emptyset$ \omterm{def:b1:topology:open}{विवृत} हैं (दूसरा रिक्तार्थ रूप से)। $\intcc{0}{1}$ \omterm{def:b1:topology:open}{विवृत}
\emph{नहीं} है: $0$ के चारों ओर का कोई \omterm{prop:b1:reals:intervals}{अंतराल} भीतर नहीं रहता।
\end{example}

\begin{proposition}[विवृत समुच्चयों का स्थायित्व]\label{prop:b1:topology:openstable}
\omterm{def:b1:topology:open}{विवृत} समुच्चयों का कोई भी सम्मिलन \omterm{def:b1:topology:open}{विवृत} होता है; और \omterm{def:b1:topology:open}{विवृत} समुच्चयों का
\emph{परिमित} प्रतिच्छेदन \omterm{def:b1:topology:open}{विवृत} होता है। अनंत प्रतिच्छेदन विफल हो सकते हैं:
$\bigcap_{n \geq 1} \intoo{-\frac 1n}{\frac 1n} = \{0\}$, जो \omterm{def:b1:topology:open}{विवृत} नहीं है।
\end{proposition}

\begin{proof}
सम्मिलन: यदि $x \in \bigcup_i U_i$, तो किसी $i_0$ के लिए $x \in U_{i_0}$, और
$U_{i_0}$ से मिला \omterm{prop:b1:reals:intervals}{अंतराल} सम्मिलन के भीतर बैठ जाता है। परिमित प्रतिच्छेदन:
यदि $x \in U_1 \cap \dots \cap U_k$, तो प्रत्येक $U_j$ से मिली त्रिज्याओं का
$r = \min(r_1, \dots, r_k) > 0$ लीजिए। प्रति-उदाहरण के लिए: $0$ के चारों ओर
के किसी भी \omterm{prop:b1:reals:intervals}{अंतराल} में कोई $\frac{1}{n}$ आ जाता है (आर्किमिडीज़), अतः वह
प्रतिच्छेदन से बाहर निकल जाता है।
\end{proof}

\begin{example}[स्पष्ट त्रिज्याओं से विवृतता प्रमाणित करना]\label{ex:b1:topology:radii}
क्या $U = \{x \in \R : x^2 > 2\}$ \omterm{def:b1:topology:open}{विवृत} है? हाँ, और प्रमाणपत्र लिखा जा सकता
है: $U = \intoo{-\infty}{-\sqrt2} \cup \intoo{\sqrt2}{+\infty}$, अर्थात् दो
\omterm{def:b1:topology:open}{विवृत} अर्धरेखाओं का सम्मिलन, जो \cref{prop:b1:topology:openstable} से \omterm{def:b1:topology:open}{विवृत}
है। वैकल्पिक रूप से, बिंदु-दर-बिंदु तर्क कीजिए: $x > \sqrt 2$ वाले $x \in U$
के लिए $r = x - \sqrt2 > 0$ लीजिए: प्रत्येक $y \in \intoo{x - r}{x + r}$ $y
> \sqrt 2$ संतुष्ट करता है, अतः $y^2 > 2$; और बाईं ओर सममित रूप से। दोनों
शैलियाँ महत्त्व रखती हैं --- संरचनात्मक शैली (ज्ञात \omterm{def:b1:topology:open}{विवृत} समुच्चयों से
सम्मिलन और परिमित प्रतिच्छेदन द्वारा बनाना) बड़े पैमाने पर बेहतर काम करती
है, और $\varepsilon$-शैली तब काम आती है जब कोई संरचना दिखाई न दे; और
\cref{ch:b1:continuity} तीसरी, सबसे शक्तिशाली शैली जोड़ेगा: $U$ संतत $x
\mapsto x^2$ के अंतर्गत \omterm{def:b1:topology:open}{विवृत} $\intoo{2}{+\infty}$ का \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} है।
\end{example}

\begin{definition}[संवृत समुच्चय]\label{def:b1:topology:closed}
\omterm{def:b1:logic:sets}{समुच्चय} $F \subseteq \R$ \emph{संवृत}\index{संवृत समुच्चय} तब है जब उसका
पूरक $\R \setminus F$ \omterm{def:b1:topology:open}{विवृत} हो। डी मॉर्गन और
\cref{prop:b1:topology:openstable} से: संवृत समुच्चयों का कोई भी प्रतिच्छेदन
संवृत होता है, और संवृत समुच्चयों के परिमित सम्मिलन संवृत होते हैं।
\end{definition}

\begin{theorem}[संवृत समुच्चयों का अनुक्रमीय अभिलक्षण]\label{thm:b1:topology:seqclosed}
$F$ \omterm{def:b1:topology:closed}{संवृत} है यदि और केवल यदि: $F$ के बिंदुओं के प्रत्येक ऐसे अनुक्रम $(u_n)$
के लिए जो किसी $\ell \in \R$ की ओर अभिसरित होता हो, सीमा $\ell$ $F$ की सदस्य
हो। (``\omterm{def:b1:topology:closed}{संवृत}'' $=$ ``सीमाओं के अंतर्गत स्थायी''।)
\end{theorem}

\begin{proof}
($\Rightarrow$) मान लीजिए $F$ \omterm{def:b1:topology:closed}{संवृत} है, $u_n \in F$, $u_n \to \ell$, और मान
लीजिए $\ell \notin F$। पूरक \omterm{def:b1:topology:open}{विवृत} है: अतः कोई $\intoo{\ell - r}{\ell + r}$
$F$ से बचता है। पर अभिसरण बड़े $n$ के लिए $u_n$ को उसी \omterm{prop:b1:reals:intervals}{अंतराल} में रख देता
है: $u_n \in F$ से विरोधाभास।

($\Leftarrow$) मान लीजिए $F$ \omterm{def:b1:topology:closed}{संवृत} नहीं है: तो पूरक \omterm{def:b1:topology:open}{विवृत} नहीं है, अतः किसी
$x \notin F$ के लिए कोई \omterm{prop:b1:reals:intervals}{अंतराल} $\intoo{x - r}{x + r}$ पूरक के भीतर नहीं है;
$r = \frac{1}{n+1}$ लेकर ऐसा $u_n \in F$ चुनिए कि $\abs{u_n - x} <
\frac{1}{n+1}$। तब $u_n \in F$, $u_n \to x \notin F$: अर्थात् अनुक्रमीय
गुणधर्म विफल हो जाता है।
\end{proof}

\begin{example}[अनुक्रमीय कसौटी, दोनों ओर]\label{ex:b1:topology:seqtest}
\emph{\omterm{def:b1:topology:closed}{संवृत}:} $F = \Z \cup \bigl\{n + \frac1n : n \geq 2\bigr\}$। मान लीजिए
$u_k \to \ell$ के साथ $u_k \in F$। खिड़की $\intcc{\ell - 1}{\ell + 1}$ में
$F$ के केवल परिमित कितने बिंदु हैं (परिमित कितने पूर्णांक, परिमित कितने $n +
\frac1n$), और किसी कोटि के आगे सभी $u_k$ उसी में हैं: तब अनुक्रम परिमित
कितने मान लेता है, और अभिसारी होने से अंततः अचर हो जाता है (जैसा
\cref{exo:b1:topology:3} में है): $\ell \in F$। अतः \omterm{def:b1:topology:closed}{संवृत} --- यद्यपि $F$ में
$\frac1n$ दूरी पर, अर्थात् मनचाहे निकट, बिंदु-युग्म हैं।

\emph{\omterm{def:b1:topology:closed}{संवृत} नहीं:} $G = \bigl\{\frac1m + \frac1n : m, n \in \N^*\bigr\}$।
अनुक्रम $\frac1n + \frac1n \in G$ $0$ की ओर जाता है, और $0 \notin G$ (दो
धनात्मक पदों का योग): अनुक्रमीय कसौटी विफल हो जाती है, अतः $G$ \omterm{def:b1:topology:closed}{संवृत} नहीं
है। रोचक बात यह कि प्रत्येक $\frac1m$ $\overline G \cap G$ का सदस्य
\emph{है}: वस्तुतः $\frac1m = \frac{1}{m+1} + \frac{1}{m(m+1)} \in G$। समापन
का सार: संवृतता सिद्ध करने के लिए \emph{सभी} अभिसारी अनुक्रमों को एक साथ
नियंत्रित कीजिए (प्रायः किसी स्थानीय परिमितता या संवृत-सूत्र वाले तर्क से);
उसका खंडन करने के लिए एक सुचयनित भागता हुआ अनुक्रम पर्याप्त है --- यह
असममिति ऋणात्मक दिशा को आसान बना देती है, और इस अध्याय के सभी प्रति-उदाहरणों
का यही एक-पंक्ति आकार है।
\end{example}

\begin{example}\label{ex:b1:topology:closed}
खंड $\intcc{a}{b}$, अर्धरेखाएँ $\intco{a}{+\infty}$, \omterm{def:b1:counting:card}{परिमित समुच्चय}, $\Z$
(पूर्णांकों का अभिसारी अनुक्रम अंततः अचर होता है) \omterm{def:b1:topology:closed}{संवृत} हैं। $\intoc{0}{1}$
न \omterm{def:b1:topology:open}{विवृत} है ($1$ पर विफल) और न \omterm{def:b1:topology:closed}{संवृत} ($\frac 1n \to 0 \notin$ \omterm{def:b1:logic:sets}{समुच्चय}):
अधिकांश \omterm{def:b1:logic:sets}{समुच्चय} दोनों में से कुछ भी नहीं हैं। $\R$ और $\emptyset$ दोनों
\omterm{def:b1:topology:open}{विवृत} भी हैं और \omterm{def:b1:topology:closed}{संवृत} भी --- और $\R$ के ऐसे केवल यही उपसमुच्चय हैं
(\cref{exo:b1:topology:9})।
\end{example}

\begin{example}[अनंत टुकड़ों से बना एक विवृत समुच्चय]\label{ex:b1:topology:rminusz}
$\R \setminus \Z = \bigcup_{n \in \Z} \intoo{n}{n+1}$: \omterm{def:b1:topology:open}{विवृत} अंतरालों का
अनंत सम्मिलन, जो \cref{prop:b1:topology:openstable} से \omterm{def:b1:topology:open}{विवृत} है --- अतः $\Z$
\omterm{def:b1:topology:closed}{संवृत} है, और उसके लिए किसी अनुक्रमीय तर्क की आवश्यकता नहीं। स्थायित्व के
नियमों में श्रम-विभाजन पर ध्यान दीजिए: \emph{सम्मिलन} स्वेच्छ हो सकते हैं
(हर बिंदु को केवल अपना प्रमाणपत्र चाहिए, जो उसे समेटने वाला एक ही \omterm{def:b1:logic:sets}{समुच्चय} दे
देता है), जबकि \emph{प्रतिच्छेदन} परिमित रहने चाहिए (प्रमाणपत्रों का
प्रतिच्छेदन लेना पड़ता है, और अनंत कितनी त्रिज्याएँ सिकुड़कर शून्य हो सकती
हैं)। \Cref{exo:b1:topology:10} दिखाएगा कि यह उदाहरण ही सामान्य आकार है:
$\R$ का प्रत्येक \omterm{def:b1:topology:open}{विवृत} उपसमुच्चय \omterm{def:b1:topology:open}{विवृत} अंतरालों का गणनीय असंयुक्त सम्मिलन
होता है।
\end{example}

\section{अंतःभाग, संवरक, सघनता}

\begin{definition}[अंतःभाग, संवरक, परिसीमा]\label{def:b1:topology:closure}
मान लीजिए $A \subseteq \R$।
\begin{itemize}
  \item बिंदु $x$ $A$ का \emph{आंतरिक} बिंदु तब है जब $A$ $x$ का \omterm{def:b1:topology:open}{प्रतिवेश}
  हो; \emph{अंतःभाग} $\mathring{A}$\index{अंतःभाग} आंतरिक बिंदुओं का \omterm{def:b1:logic:sets}{समुच्चय}
  है।
  \item बिंदु $x$ $A$ से \emph{अभिलग्न} तब है जब $x$ का प्रत्येक \omterm{def:b1:topology:open}{प्रतिवेश}
  $A$ से मिलता हो; \emph{संवरक} $\overline{A}$\index{संवरक} अभिलग्न बिंदुओं
  का \omterm{def:b1:logic:sets}{समुच्चय} है।
  \item \emph{परिसीमा} $\partial A = \overline A \setminus \mathring A$
  है।\index{परिसीमा}
\end{itemize}
तब $\mathring A \subseteq A \subseteq \overline A$।
\end{definition}

\begin{proposition}[मुख्य गुणधर्म]\label{prop:b1:topology:closureprops}
\begin{enumerate}
  \item $\mathring A$ $A$ में समाया सबसे बड़ा \omterm{def:b1:topology:open}{विवृत समुच्चय} है; $A$ \omterm{def:b1:topology:open}{विवृत} है
  तभी जब $A = \mathring A$।
  \item $\overline A$ $A$ को समेटने वाला सबसे छोटा \omterm{def:b1:topology:closed}{संवृत समुच्चय} है; $A$
  \omterm{def:b1:topology:closed}{संवृत} है तभी जब $A = \overline A$।
  \item (अभिलग्नता का अनुक्रमीय अभिलक्षण) $x \in \overline A$ यदि और केवल
  यदि $x$ $A$ के बिंदुओं के किसी अनुक्रम की सीमा हो।
  \item पूरक लेने से दोनों धारणाओं की अदला-बदली हो जाती है: $\R \setminus
  \overline A = \bigl(\R \setminus A\bigr)^{\!\circ}$।
\end{enumerate}
\end{proposition}

\begin{proof}
(4) $x \notin \overline A$ $\iff$ $x$ का कोई \omterm{def:b1:topology:open}{प्रतिवेश} $A$ से बचता है $\iff$
$x$ के चारों ओर कोई \omterm{prop:b1:reals:intervals}{अंतराल} $\R \setminus A$ में है $\iff$ $x$ $\R \setminus
A$ का आंतरिक बिंदु है।

(1) $\mathring A$ \omterm{def:b1:topology:open}{विवृत} है: यदि $x \in \mathring A$, तो कोई
$\intoo{x-r}{x+r} \subseteq A$; उस \omterm{prop:b1:reals:intervals}{अंतराल} के प्रत्येक बिंदु $y$ के चारों ओर
उसी के भीतर एक छोटा \omterm{prop:b1:reals:intervals}{अंतराल} है, अतः $A$ के भीतर भी: इसलिए पूरा \omterm{prop:b1:reals:intervals}{अंतराल}
$\mathring A$ में है। कोई भी \omterm{def:b1:topology:open}{विवृत} $U \subseteq A$ $A$ के आंतरिक बिंदुओं से
बना है, अतः $U \subseteq \mathring A$: अर्थात् सबसे बड़ा। विवृतता का
अभिलक्षण इसी से निकलता है।

(2) विस्तार से। (4) से $\R \setminus \overline A$ $\R \setminus A$ का
\omterm{def:b1:topology:closure}{अंतःभाग} है, जो (1) से \omterm{def:b1:topology:open}{विवृत समुच्चय} है: अतः $\overline A$ \omterm{def:b1:topology:closed}{संवृत} है, और उसमें
$A$ समाया है। न्यूनतमता: मान लीजिए $F \supseteq A$ \omterm{def:b1:topology:closed}{संवृत} है। तब $\R
\setminus F$ \omterm{def:b1:topology:open}{विवृत} है और $\R \setminus A$ में समाया है, अतः (1) की अधिकतमता
से,
\[
\R \setminus F \subseteq \bigl(\R \setminus A\bigr)^{\!\circ}
= \R \setminus \overline A ,
\]
और फिर पूरक लेने पर: $\overline A \subseteq F$। इस प्रकार $\overline A$ सबसे
छोटा \omterm{def:b1:topology:closed}{संवृत} अधिसमुच्चय है। अभिलक्षण: यदि $A = \overline A$, तो $A$ \omterm{def:b1:topology:closed}{संवृत} है
(अभी दिखाया गया); और यदि $A$ \omterm{def:b1:topology:closed}{संवृत} है, तो वह स्वयं $A$ का \omterm{def:b1:topology:closed}{संवृत} अधिसमुच्चय
है, अतः न्यूनतमता $\overline A \subseteq A$ पर बाध्य कर देती है, और समता मिल
जाती है।

(3) यदि $u_n \in A$, तो $u_n \to x$: $x$ के प्रत्येक \omterm{def:b1:topology:open}{प्रतिवेश} में कोई $u_n
\in A$ है, अतः $x \in \overline A$। विलोमतः, यदि $x \in \overline A$:
प्रत्येक \omterm{prop:b1:reals:intervals}{अंतराल} $\intoo{x - \frac{1}{n+1}}{x + \frac{1}{n+1}}$ $A$ से किसी
$u_n$ पर मिलता है, और $u_n \to x$।
\end{proof}

\begin{example}\label{ex:b1:topology:closureexamples}
$\overline{\intoo{0}{1}} = \intcc{0}{1}$; $\mathring{\intcc{0}{1}} =
\intoo{0}{1}$; $\partial\intoo{0}{1} = \{0, 1\}$। $A = \{\frac 1n : n \in
\N^*\}$ के लिए: $\overline A = A \cup \{0\}$, $\mathring A = \emptyset$,
$\partial A = A \cup \{0\}$। $\Q$ के लिए: \omterm{def:b1:topology:dense}{सघनता}
(\cref{thm:b1:reals:density}) से प्रत्येक वास्तविक संख्या $\Q$ से अभिलग्न
है, अतः $\overline{\Q} = \R$ जबकि $\mathring{\Q} = \emptyset$ (हर \omterm{prop:b1:reals:intervals}{अंतराल} में
अपरिमेय संख्याएँ हैं): अर्थात् $\Q$ की \omterm{def:b1:topology:closure}{परिसीमा} पूरा $\R$ है।
\end{example}

\begin{example}[पूरी शल्य-रचना]\label{ex:b1:topology:anatomy}
मान लीजिए $A = \intoc{0}{1} \,\cup\, \bigl(\Q \cap \intoo{2}{3}\bigr)
\,\cup\, \{4\}$। हम \cref{def:b1:topology:closure} के तीनों \omterm{def:b1:logic:sets}{समुच्चय}
टुकड़ा-दर-टुकड़ा संगणित करते हैं।

\emph{\omterm{def:b1:topology:closure}{अंतःभाग}।} $\intoo{0}{1}$ के बिंदु के चारों ओर पूरा \omterm{prop:b1:reals:intervals}{अंतराल} $A$ के भीतर
है: अतः आंतरिक। बिंदु $1$: उसके चारों ओर हर \omterm{prop:b1:reals:intervals}{अंतराल} $1$ के दाईं ओर रिस जाता
है, जहाँ $2$ तक $A$ में कुछ नहीं है: अतः आंतरिक नहीं। $\Q \cap \intoo{2}{3}$
का कोई बिंदु आंतरिक नहीं है (हर \omterm{prop:b1:reals:intervals}{अंतराल} में अपरिमेय संख्याएँ हैं,
\cref{thm:b1:reals:density}); और वियुक्त $4$ भी नहीं। अतः $\mathring A =
\intoo{0}{1}$।

\emph{\omterm{def:b1:topology:closure}{संवरक}।} $A$ के बिंदुओं की सीमाएँ: पूरा $\intcc{0}{1}$ ($\frac 1n \in
A$ के साथ $0 = \lim \frac1n$); पूरा $\intcc{2}{3}$ (वहाँ की हर वास्तविक
संख्या उस \omterm{prop:b1:reals:intervals}{अंतराल} की परिमेय संख्याओं की सीमा है, फिर \omterm{def:b1:topology:dense}{सघनता}); और $4$। इसके
अतिरिक्त कुछ नहीं: $\intcc{0}{1} \cup \intcc{2}{3} \cup \{4\}$ के बाहर के
बिंदु की उस \omterm{def:b1:topology:closed}{संवृत समुच्चय} से दूरी धनात्मक है। अतः $\overline A =
\intcc{0}{1} \cup \intcc{2}{3} \cup \{4\}$।

\emph{\omterm{def:b1:topology:closure}{परिसीमा}।} $\partial A = \overline A \setminus \mathring A = \{0, 1\}
\cup \intcc{2}{3} \cup \{4\}$।

समापन का सार: तीनों संक्रियाएँ \emph{स्थानीय} रूप से काम करती हैं --- $A$ का
हर टुकड़ा अपने स्वभाव के अनुसार योगदान करता है (ठोस \omterm{prop:b1:reals:intervals}{अंतराल} अपना भीतर रख लेता
है, सघन-पर-छिद्रिल टुकड़ा पूरा का पूरा \omterm{def:b1:topology:closure}{परिसीमा} बन जाता है, और वियुक्त बिंदु
शुद्ध \omterm{def:b1:topology:closure}{परिसीमा} है) --- और $A$ का दो पंक्ति का रेखाचित्र कोई उपपत्ति लिखे जाने
से पहले ही हर उत्तर बता देता है।
\end{example}

\begin{remark}[बिंदु-समुच्चय तर्क की सामान्य भूलें]
(क) \emph{``\omterm{def:b1:topology:open}{विवृत} नहीं'' का अर्थ ``\omterm{def:b1:topology:closed}{संवृत}'' नहीं है}: अधिकांश \omterm{def:b1:logic:sets}{समुच्चय} दोनों
में से कुछ भी नहीं हैं ($\intoc{0}{1}$), और दो \omterm{def:b1:logic:sets}{समुच्चय} दोनों हैं
($\emptyset$, $\R$) --- \omterm{def:b1:topology:open}{विवृत} और \omterm{def:b1:topology:closed}{संवृत} विपरीत नहीं, बल्कि पूरक लेने से आपस
में द्वैत हैं। (ख) \emph{\omterm{def:b1:topology:closure}{अंतःभाग} और \omterm{def:b1:topology:closure}{संवरक} क्रम-विनिमेय नहीं हैं}: $A = \Q$
के लिए,
\[
\overline{\mathring A} = \overline\emptyset = \emptyset
\qquad\text{जबकि}\qquad
\bigl(\,\overline A\,\bigr)^{\!\circ} = \mathring \R = \R :
\]
दोनों दोहराए हुए संकारक उतने ही भिन्न हैं जितने \omterm{def:b1:logic:sets}{समुच्चय} हो सकते हैं। (ग)
\emph{\omterm{def:b1:topology:closed}{संवृत} समुच्चयों के अनंत सम्मिलन \omterm{def:b1:topology:closed}{संवृत} होने में विफल हो सकते हैं}:
$\bigcup_{n\geq1} \intcc{\frac1n}{1} = \intoc{0}{1}$ --- अर्थात्
\cref{prop:b1:topology:openstable} के प्रतिच्छेदन वाले प्रति-उदाहरण का
दर्पण। (घ) \emph{\omterm{def:b1:topology:dense}{सघन} होने का अर्थ बड़ा होना नहीं है}: $\Q$ \omterm{def:b1:topology:dense}{सघन}, गणनीय और
रिक्त \omterm{def:b1:topology:closure}{अंतःभाग} वाला है, और उसका पूरक भी \omterm{def:b1:topology:dense}{सघन} है; \omterm{def:b1:topology:dense}{सघनता} ``हर चीज़ के मनचाहे
निकट'' कहती है, ``लगभग सब कुछ'' नहीं --- और सप्ताहांत समस्या का \omterm{pb:b1:topology:1}{कैंटर
समुच्चय} (\cref{pb:b1:topology:1}) उलटी बात कहता है, अर्थात् सांस्थितिक
दृष्टि से छोटा पर आकार में अगणनीय रूप से बड़ा \omterm{def:b1:logic:sets}{समुच्चय}।
\end{remark}

\begin{example}[एक संवरक, यथार्थ रूप से संगणित]\label{ex:b1:topology:closurecomputed}
मान लीजिए $G = \bigl\{\frac1m + \frac1n : m, n \in \N^*\bigr\}$
(\cref{ex:b1:topology:seqtest} से)। दावा:
\[
\overline G = G \,\cup\, \Bigl\{\frac1m : m \in \N^*\Bigr\}
\,\cup\, \{0\} .
\]
($\supseteq$) $\frac1m = \lim_n \bigl(\frac1m + \frac1n\bigr)$ और $0 =
\lim_n \frac2n$: अनुक्रमीय अभिलक्षण से अभिलग्न। ($\subseteq$) मान लीजिए $x =
\lim_k \bigl( \frac{1}{m_k} + \frac{1}{n_k}\bigr)$; प्रत्येक युग्म को इस
प्रकार क्रमित कीजिए कि $m_k \leq n_k$। यदि $(m_k)$ अपरिबद्ध है, तो किसी
\omterm{def:b1:seq:subsequence}{उपानुक्रम} पर $m_k \to \infty$, अतः $n_k \to \infty$ भी और $x = 0$। अन्यथा
$(m_k)$ परिमित कितने मान लेता है, और उनमें से कोई एक, मान लीजिए $m$, अनंत
बार; उस \omterm{def:b1:seq:subsequence}{उपानुक्रम} पर $\frac{1}{n_k} \to x - \frac1m$: यदि $(n_k)$ परिबद्ध
है, तो वह कोई मान $n$ अनंत बार लेता है और $x = \frac1m + \frac1n \in G$; और
यदि नहीं, तो $x = \frac1m$। हर स्थिति घोषित \omterm{def:b1:logic:sets}{समुच्चय} में आ गिरती है। समापन का
सार: \omterm{def:b1:topology:closure}{संवरक} की संगणना सूचकांकों पर संहतता-शैली का स्थिति-विश्लेषण है ---
परिबद्ध सूचकांक का अर्थ है परिमित कितने मान (कबूतरखाना), अपरिबद्ध सूचकांक का
अर्थ है कोई सीमा भाग जाना --- और उत्तर सीमा-बिंदुओं की विशिष्ट दो-परत संरचना
दिखा देता है: \omterm{def:b1:logic:sets}{समुच्चय}, उसकी पहली पीढ़ी की सीमाएँ, और उनकी सीमा $0$।
\end{example}

\begin{definition}[सघनता, सांस्थितिक रूप]\label{def:b1:topology:dense}
$A$ $\R$ में \emph{सघन}\index{सघनता} तब है जब $\overline A = \R$ --- समतुल्य
रूप से, हर अरिक्त \omterm{def:b1:topology:open}{विवृत} \omterm{prop:b1:reals:intervals}{अंतराल} $A$ से मिलता है; और समतुल्य रूप से
(\cref{prop:b1:topology:closureprops}~(3) से), हर वास्तविक संख्या $A$ के
अवयवों की सीमा है। उदाहरण: $\Q$, $\R \setminus \Q$, द्विआधारी संख्याएँ
(\cref{exo:b1:reals:8}), सघन \omterm{def:b1:structures:subgroup}{उपसमूह} (\cref{exo:b1:reals:9})।
\end{definition}

\begin{example}[सघनता सापेक्ष है]\label{ex:b1:topology:relativedensity}
यहाँ परिभाषित ``\omterm{def:b1:topology:dense}{सघन}'' का अर्थ है \emph{$\R$ में} \omterm{def:b1:topology:dense}{सघन}; कोई \omterm{def:b1:logic:sets}{समुच्चय} इसके बदले
रेखा के केवल किसी भाग में \omterm{def:b1:topology:dense}{सघन} हो सकता है। $\intcc{0}{1}$ की द्विआधारी
संख्याएँ, अर्थात् $D \cap \intcc{0}{1}$ (\cref{exo:b1:reals:8}),
$\intcc{0}{1}$ में समाए हर \omterm{def:b1:topology:open}{विवृत} \omterm{prop:b1:reals:intervals}{अंतराल} से मिलती हैं पर निस्संदेह
$\intoo{2}{3}$ से पूरी तरह चूक जाती हैं: वे $\intcc{0}{1}$ \emph{में} \omterm{def:b1:topology:dense}{सघन}
हैं, अर्थात् $\overline{D \cap \intcc{0}{1}} = \intcc{0}{1}$। सामान्य वाक्य
``$A$ $B$ में \omterm{def:b1:topology:dense}{सघन} है'' $B \subseteq \overline A$ का संक्षेप है --- परिवेशी
\omterm{def:b1:logic:sets}{समुच्चय} का नाम सदा लीजिए, क्योंकि सप्ताहांत समस्या के सिरे \emph{\omterm{pb:b1:topology:1}{कैंटर
समुच्चय} में} \omterm{def:b1:topology:dense}{सघन} हैं और साथ ही \emph{$\R$ में} कहीं \omterm{def:b1:topology:dense}{सघन} नहीं: एक ही \omterm{def:b1:logic:sets}{समुच्चय},
दो सच्चे और विपरीत सुनाई देने वाले वर्णन।
\end{example}

\begin{example}[सघनता से निपटना]\label{ex:b1:topology:densitytricks}
तीन त्वरित चालें जो लगातार लौटती हैं। \emph{बड़ा करना}: यदि $A$ \omterm{def:b1:topology:dense}{सघन} है और $A
\subseteq B$, तो $B$ \omterm{def:b1:topology:dense}{सघन} है (हर \omterm{prop:b1:reals:intervals}{अंतराल} पहले ही $A$ से मिलता है)। \emph{ले
जाना}: यदि $A$ \omterm{def:b1:topology:dense}{सघन} है, तो $\lambda \neq 0$ के लिए $\lambda A + \mu$ भी \omterm{def:b1:topology:dense}{सघन}
है --- \omterm{prop:b1:reals:intervals}{अंतराल} $I$ $\lambda A + \mu$ से तभी मिलता है जब \omterm{prop:b1:reals:intervals}{अंतराल} $\frac{I -
\mu}{\lambda}$ $A$ से मिले; अतः, मान लीजिए, $10^{-9}$ के विषम गुणज \omterm{def:b1:topology:dense}{सघन} हैं।
\emph{प्रतिच्छेदन विफल है}: दो \omterm{def:b1:topology:dense}{सघन} \omterm{def:b1:logic:sets}{समुच्चय} एक-दूसरे से पूरी तरह चूक सकते हैं
($\Q$ और $\R \setminus \Q$): \omterm{def:b1:topology:dense}{सघनता} सम्मिलन और ऐफ़ीन प्रतिचित्रणों में बची
रहती है, प्रतिच्छेदन में कभी नहीं।
\end{example}

\section{खंडों की संहतता}

\begin{theorem}[खंड अनुक्रमीय रूप से संहत हैं]\label{thm:b1:topology:compact}
मान लीजिए $a \leq b$। $\intcc{a}{b}$ के बिंदुओं के प्रत्येक अनुक्रम का कोई
\omterm{def:b1:seq:subsequence}{उपानुक्रम} $\intcc{a}{b}$ \emph{के} किसी बिंदु की ओर अभिसरित होता
है।\index{संहतता}

और अधिक व्यापक रूप से, $\R$ के जिन उपसमुच्चयों में यह गुणधर्म है (हर अनुक्रम
का कोई \omterm{def:b1:seq:subsequence}{उपानुक्रम} उसी \omterm{def:b1:logic:sets}{समुच्चय} में अभिसरित होता है) वे ठीक \emph{\omterm{def:b1:topology:closed}{संवृत} और
परिबद्ध} \omterm{def:b1:logic:sets}{समुच्चय} हैं।
\end{theorem}

\begin{proof}
$\intcc{a}{b}$ का अनुक्रम परिबद्ध है, अतः बोल्ज़ानो--वाइरश्ट्रास
(\cref{thm:b1:seq:bw}) कोई अभिसारी \omterm{def:b1:seq:subsequence}{उपानुक्रम} निकाल देता है; और उसकी सीमा
$\intcc{a}{b}$ में ही रहती है, क्योंकि खंड \omterm{def:b1:topology:closed}{संवृत} होते हैं
(\cref{thm:b1:topology:seqclosed})।

सामान्य स्थिति। \emph{(\omterm{def:b1:topology:closed}{संवृत} परिबद्ध $\Rightarrow$ संहत)}: मान लीजिए $F$
\omterm{def:b1:topology:closed}{संवृत} और परिबद्ध है और $(u_n)$ $F$ में कोई अनुक्रम है। $F$ की परिबद्धता
अनुक्रम को परिबद्ध कर देती है, अतः बोल्ज़ानो--वाइरश्ट्रास $u_{\varphi(n)}
\to \ell$ निकाल देता है; और $\ell \in F$, क्योंकि $F$ \omterm{def:b1:topology:closed}{संवृत} है और वह
\omterm{def:b1:seq:subsequence}{उपानुक्रम} $F$ के बिंदुओं का अभिसारी अनुक्रम है
(\cref{thm:b1:topology:seqclosed}): दोनों परिकल्पनाएँ एक-एक करके खप जाती
हैं, परिबद्धता सीमा के अस्तित्व के लिए और संवृतता उसकी सदस्यता के लिए।
\emph{(संहत $\Rightarrow$ \omterm{def:b1:topology:closed}{संवृत} और परिबद्ध)}: यदि $F$ अपरिबद्ध है, तो ऐसा
$u_n \in F$ चुनिए कि $\abs{u_n} \geq n$; तब हर \omterm{def:b1:seq:subsequence}{उपानुक्रम} अपरिबद्ध है, अतः
अपसारी (\cref{prop:b1:seq:first}): कोई अभिसारी \omterm{def:b1:seq:subsequence}{उपानुक्रम} है ही नहीं। यदि $F$
\omterm{def:b1:topology:closed}{संवृत} नहीं है, तो $u_n \to \ell \notin F$ वाला $u_n \in F$ लीजिए
(\cref{thm:b1:topology:seqclosed}): हर \omterm{def:b1:seq:subsequence}{उपानुक्रम} $\ell \notin F$ की ओर
अभिसरित होता है, अतः कोई \omterm{def:b1:seq:subsequence}{उपानुक्रम} $F$ \emph{में} अभिसरित नहीं होता।
\end{proof}

\begin{example}[नेस्टेड संहत समुच्चय]\label{ex:b1:topology:nested}
प्रमेय का पहला अभ्यास। मान लीजिए $K_0 \supseteq K_1 \supseteq K_2 \supseteq
\dots$ $\R$ के अरिक्त संहत (\omterm{def:b1:topology:closed}{संवृत} परिबद्ध) उपसमुच्चय हैं। तब $\bigcap_n K_n
\neq \emptyset$। वस्तुतः, प्रत्येक $n$ के लिए $x_n \in K_n$ चुनिए: यह
अनुक्रम संहत $K_0$ में रहता है, अतः कोई \omterm{def:b1:seq:subsequence}{उपानुक्रम} $x_{\varphi(n)}$ किसी $x$
की ओर अभिसरित होता है (\cref{thm:b1:topology:compact})। प्रत्येक नियत $m$ के
लिए $\varphi(n) \geq m$ वाले पद $x_{\varphi(n)}$ सभी \omterm{def:b1:topology:closed}{संवृत समुच्चय} $K_m$ में
हैं, अतः सीमा $x$ $K_m$ में है (\cref{thm:b1:topology:seqclosed}); और चूँकि
$m$ स्वेच्छ था, $x \in \bigcap_n K_n$। एक उपयोगी साथी: यदि कोई \omterm{def:b1:topology:open}{विवृत समुच्चय}
$U$ $\bigcap_n K_n$ को समेटता है, तो किसी $n$ के लिए $U \supseteq K_n$ ---
वही तर्क बिंदुओं $x_n \in K_n \setminus U$ पर लगाइए; सीमा $x$ $\bigcap K_n
\subseteq U$ में होती, फिर भी $U$ के \omterm{def:b1:topology:open}{विवृत} होने से अंततः $x_{\varphi(n)} \in
U$ अनिवार्य हो जाता, जो विरोधाभास है। संहतता के बिना दोनों \omterm{def:b1:logic:statement}{कथन} विफल हैं:
$\bigcap_n \intoo{0}{\frac 1n} = \emptyset$ और $\bigcap_n \intco{n}{+\infty}
= \emptyset$। समापन का सार: संहतता अरिक्तता के अनंत दावों की शृंखला को एक ही
सीमा-बिंदु में बदल देती है --- यही वह औज़ार है जो अनंत प्रतिच्छेदन तक जीवित
बचता है, और सप्ताहांत समस्या (\cref{pb:b1:topology:1}) उस पर दो बार टिकेगी।
\end{example}

\begin{example}[सम्मिलन के अंतर्गत परिसीमाएँ]\label{ex:b1:topology:boundaryunion}
सदा $\partial(A \cup B) \subseteq \partial A \cup \partial B$: $\partial(A
\cup B)$ के बिंदु का हर \omterm{def:b1:topology:open}{प्रतिवेश} $A \cup B$ से मिलता है (अतः $A$ या $B$ से,
अनंत बार किसी एक से) और $A \cup B$ के पूरक से भी, जो दोनों पूरकों में है ---
इसके बाद एक छोटी जाँच उस बिंदु को $\partial A$ या $\partial B$ में रख देती
है। यह अंतर्विष्टि अत्यंत कठोर हो सकती है: $A = \Q$ और $B = \R\setminus\Q$
के साथ,
\[
\partial(A \cup B) = \partial \R = \emptyset ,
\qquad
\partial A \cup \partial B = \R \cup \R = \R :
\]
दो ऊबड़-खाबड़ \omterm{def:b1:logic:sets}{समुच्चय} जुड़कर एक निर्बाध \omterm{def:b1:logic:sets}{समुच्चय} बन सकते हैं, और उनकी
परिसीमाएँ एक-दूसरे को नष्ट कर देती हैं। समापन का सार: \omterm{def:b1:topology:closure}{अंतःभाग} और \omterm{def:b1:topology:closure}{संवरक}
सम्मिलन तथा प्रतिच्छेदन के अंतर्गत \emph{एकदिष्ट} रूप से व्यवहार करते हैं,
पर परिसीमाएँ नहीं --- $\partial$ को व्युत्पन्न राशि मानिए ($\overline A
\setminus \mathring A$), अपने ही बीजगणित वाला संकारक कभी नहीं।
\end{example}

\begin{remark}[इस खंड के भीतर आगे की दृष्टि]
यहाँ बनी शब्दावली इस पुस्तक में दो बार और खपती है। \cref{ch:b1:continuity}
में हर प्रमेय वेश बदला हुआ सांस्थिति का \omterm{def:b1:logic:statement}{कथन} है: मध्यवर्ती मान प्रमेय कहती है
कि संतत \omterm{def:b1:logic:map}{प्रतिचित्रण} \omterm{prop:b1:reals:intervals}{अंतराल} का गुणधर्म सुरक्षित रखते हैं, और चरम मान प्रमेय
कहती है कि वे संहतता सुरक्षित रखते हैं --- और उपपत्तियाँ
\cref{thm:b1:topology:seqclosed,thm:b1:topology:compact} का नाम लेकर बुलाती
हैं। \cref{ch:b1:multivar} में वही परिभाषाएँ $\R^2$ में, अंतरालों के स्थान
पर वृत्तचकतियों के साथ, फिर से पढ़ी जाती हैं: \omterm{def:b1:topology:open}{विवृत समुच्चय}, \omterm{def:b1:topology:closure}{संवरक} और संहतता
शब्दशः चले जाते हैं, और दो-चर वाली चरम मान प्रमेय फिर बोल्ज़ानो--वाइरश्ट्रास
पर सवार होती है (प्रत्येक निर्देशांक पर निष्कर्षण)। एकमात्र धारणा जो बिना
कष्ट के \emph{सामान्यीकृत नहीं} होती वह स्वयं \omterm{prop:b1:reals:intervals}{अंतराल} है --- समतल में उत्तलता
का स्थान संबद्धता ले लेती है, और वह कथा \cref{exo:b1:topology:9} के ``केवल
$\emptyset$ और $\R$ ही \omterm{def:b1:topology:open}{विवृत} और \omterm{def:b1:topology:closed}{संवृत} हैं'' से आरंभ होती है।
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:topology:1}
प्रत्येक \omterm{def:b1:logic:sets}{समुच्चय} के लिए बताइए कि वह \omterm{def:b1:topology:open}{विवृत} है, \omterm{def:b1:topology:closed}{संवृत} है, दोनों है, या कुछ भी
नहीं (न्यायसंगति सहित): $\intoo{0}{1} \cup \intoo{2}{3}$; $\;\intco{0}{1}$;
$\;\{0\} \cup \intcc{1}{2}$; $\;\R \setminus \Z$; $\;\Q \cap \intoo{0}{1}$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:topology:2}
इनके लिए $\mathring A$, $\overline A$ और $\partial A$ निर्धारित कीजिए: $A =
\intoc{0}{1} \cup \{2\}$; $\;A = \R \setminus \Q$; $\;A =
\bigl\{\frac{(-1)^n n}{n+1} : n \in \N\bigr\}$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:topology:3}
सिद्ध कीजिए कि \omterm{def:b1:counting:card}{परिमित समुच्चय} \omterm{def:b1:topology:closed}{संवृत} होता है --- पहले पूरकों से, फिर
अनुक्रमीय अभिलक्षण से।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:topology:4}
सिद्ध कीजिए कि किसी भी $A, B \subseteq \R$ के लिए $\overline{A \cup B} =
\overline A \cup \overline B$। उदाहरण से दिखाइए कि $\overline{A \cap B}$
$\overline A \cap \overline B$ से भिन्न हो सकता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:5}
मान लीजिए $\R$ में $u_n \to \ell$। सिद्ध कीजिए कि \omterm{def:b1:logic:sets}{समुच्चय} $\{u_n : n \in
\N\} \cup \{\ell\}$ \omterm{def:b1:topology:closed}{संवृत} है (और परिबद्ध होने की बात जोड़ दें तो संहत भी ---
जो वह है)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:6}
मान लीजिए $U$ \omterm{def:b1:topology:open}{विवृत} है और $A$ स्वेच्छ। सिद्ध कीजिए कि $U + A = \{u + a\}$
\omterm{def:b1:topology:open}{विवृत} है। उससे निकालिए कि किसी \omterm{def:b1:topology:open}{विवृत समुच्चय} और किसी भी \omterm{def:b1:logic:sets}{समुच्चय} का योग \omterm{def:b1:topology:open}{विवृत}
होता है, और इसके विपरीत ऐसे दो \emph{\omterm{def:b1:topology:closed}{संवृत}} \omterm{def:b1:logic:sets}{समुच्चय} दिखाइए जिनका योग \omterm{def:b1:topology:closed}{संवृत}
नहीं है। \emph{(\cref{exo:b1:reals:9} के साथ $\Z$ और $\sqrt 2\,\Z$
आज़माइए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:7}
बिंदु $x \in A$ $A$ में \emph{वियुक्त} तब है जब $x$ का कोई \omterm{def:b1:topology:open}{प्रतिवेश} $A$ से
केवल $x$ पर मिले। सिद्ध कीजिए कि $\Z$ का प्रत्येक बिंदु $\Z$ में वियुक्त है,
कि $A = \{\frac1n\}$ के सभी बिंदु वियुक्त हैं फिर भी $\overline A \neq A$,
और यह कि जिस \omterm{def:b1:logic:sets}{समुच्चय} के सभी बिंदु वियुक्त हों उसका \omterm{def:b1:topology:closure}{अंतःभाग} रिक्त होता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:8}
सिद्ध कीजिए कि परिबद्ध \omterm{def:b1:logic:sets}{समुच्चय} का \omterm{def:b1:topology:closure}{संवरक} परिबद्ध होता है, और यह कि ऊपर से
परिबद्ध अरिक्त $A$ के लिए $\sup \overline A = \sup A$। उससे निकालिए कि $\sup
A \in \overline A$: अर्थात् \omterm{def:b1:reals:bounds}{उच्चतम} सदा अभिलग्न होता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:topology:9}
सिद्ध कीजिए कि $\R$ के केवल वही उपसमुच्चय \omterm{def:b1:topology:open}{विवृत} और \omterm{def:b1:topology:closed}{संवृत} दोनों हैं जो
$\emptyset$ और $\R$ हैं। \emph{संकेत: मान लीजिए $A$ \omterm{def:b1:topology:open}{विवृत}, \omterm{def:b1:topology:closed}{संवृत} है, जहाँ $A
\neq \emptyset$ और $\R \setminus A \neq \emptyset$; $a \in A$, $b \notin A$
चुनिए, मान लीजिए $a < b$, और $s = \sup\,(A \cap \intcc{a}{b})$ पर विचार
कीजिए; तय कीजिए कि $s$ $A$ का सदस्य हो सकता है या उसके पूरक का।}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:topology:10}
(\omterm{def:b1:topology:open}{विवृत} समुच्चयों की संरचना) मान लीजिए $U \subseteq \R$ \omterm{def:b1:topology:open}{विवृत} और अरिक्त है।
$x \in U$ के लिए मान लीजिए $I_x$ उन सभी \omterm{def:b1:topology:open}{विवृत} अंतरालों का सम्मिलन है जो $x$
को समेटते हैं और $U$ में समाए हैं। सिद्ध कीजिए कि $I_x$ \omterm{def:b1:topology:open}{विवृत} \omterm{prop:b1:reals:intervals}{अंतराल} है, कि
दो \omterm{def:b1:logic:sets}{समुच्चय} $I_x$, $I_y$ या तो बराबर हैं या असंयुक्त, और यह कि $U$
\emph{गणनीय कितने} जोड़ों में असंयुक्त \omterm{def:b1:topology:open}{विवृत} अंतरालों का सम्मिलन है
\emph{(हर एक में एक परिमेय संख्या चुनिए)}।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:11}
बिंदु $x \in \R$ $A$ का \emph{संचय बिंदु} तब है जब $x$ का प्रत्येक \omterm{def:b1:topology:open}{प्रतिवेश}
$A \setminus \{x\}$ से मिले; उनका \omterm{def:b1:logic:sets}{समुच्चय} \emph{व्युत्पन्न \omterm{def:b1:logic:sets}{समुच्चय}} $A'$ है।
सिद्ध कीजिए कि $\overline A = A \cup A'$, और यह कि $A$ \omterm{def:b1:topology:closed}{संवृत} है यदि और केवल
यदि $A' \subseteq A$। $A = \{\frac 1n : n \in \N^*\}$ के लिए, $A = \Z$ के
लिए, और $A = \Q$ के लिए $A'$ निर्धारित कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:topology:12}
अरिक्त $A \subseteq \R$ के लिए $d_A(x) = \inf\{\abs{x - a} : a \in A\}$
परिभाषित कीजिए। सिद्ध कीजिए:
\begin{enumerate}
  \item सभी $x, y$ के लिए $\abs{d_A(x) - d_A(y)} \leq \abs{x - y}$ ($d_A$
  $1$-लिप्शिट्ज़ है);
  \item $d_A(x) = 0$ यदि और केवल यदि $x \in \overline A$; विशेष रूप से, यदि
  $F$ \omterm{def:b1:topology:closed}{संवृत} है और $x \notin F$, तो $d_F(x) > 0$;
  \item प्रत्येक $\varepsilon > 0$ के लिए \omterm{def:b1:logic:sets}{समुच्चय} $V_\varepsilon = \{x :
  d_F(x) < \varepsilon\}$ \omterm{def:b1:topology:open}{विवृत} है, $F$ को समेटता है, और \omterm{def:b1:topology:closed}{संवृत} $F$ के लिए
  $\bigcap_{\varepsilon > 0} V_\varepsilon = F$: अर्थात् प्रत्येक \omterm{def:b1:topology:closed}{संवृत
  समुच्चय} \omterm{def:b1:topology:open}{विवृत} समुच्चयों का गणनीय प्रतिच्छेदन है।
\end{enumerate}
\end{exercise}

\section{समस्या: कैंटर समुच्चय, एक साथ छोटा और विशाल}

\begin{problem}[{सप्ताहांत समस्या --- कैंटर का मध्य-तिहाई \omterm{def:b1:logic:sets}{समुच्चय}: लंबाई शून्य, अगणनीय, पूर्ण,
और $C + C =
\intcc{0}{2}$}]
\label{pb:b1:topology:1}
$\intcc{0}{1}$ में से उसका \omterm{def:b1:topology:open}{विवृत} मध्य-तिहाई हटाइए, फिर बचे हुए हर खंड का
मध्य-तिहाई, और यही सदा दोहराइए: जो बचता है वह \emph{\omterm{pb:b1:topology:1}{कैंटर समुच्चय}} $C$ है,
अर्थात् विश्लेषण की मूल प्रति-उदाहरण-निर्माणशाला। यह समस्या उसका निर्माण
करती है, उसे \cref{pb:b1:reals:1} की आधार-$3$ मशीनरी से पढ़ती है, और उसका
विरोधाभासी चित्र स्थापित करती है: कुल लंबाई शून्य, फिर भी अगणनीय; रिक्त
\omterm{def:b1:topology:closure}{अंतःभाग}, फिर भी कोई वियुक्त बिंदु नहीं; पूर्णतः असंबद्ध, फिर भी $C + C$ पूरा
खंड $\intcc{0}{2}$ भर देता है।\index{कैंटर समुच्चय} औपचारिक रूप से: $C_0 =
\intcc{0}{1}$, और $C_{n+1}$ $C_n$ से $C_n$ के प्रत्येक खंड का \omterm{def:b1:topology:open}{विवृत}
मध्य-तिहाई हटाकर मिलता है; अंत में $C = \bigcap_{n \geq 0} C_n$। आगे
सर्वत्र, $x \in \intcc{0}{1}$ का \emph{त्रिआधारी कूट} $d_k \in \{0, 1, 2\}$
वाली कोई भी अंक-माला $(d_k)_{k\geq1}$ है जिसका मान $\sup_n \sum_{k=1}^n d_k
3^{-k}$ $x$ के बराबर हो --- अनुचित कूट (अंततः $2$) भी स्वीकार्य हैं;
\cref{pb:b1:reals:1} (प्रश्न 9--11) से प्रत्येक $x \in \intcc{0}{1}$ के एक
या दो कूट होते हैं, और दो ठीक तभी जब $x = m/3^N \in \intoo{0}{1}$।

\medskip \noindent\textbf{भाग I --- निर्माण।}
\begin{enumerate}
  \item $C_1$ और $C_2$ को खंडों के सम्मिलन के रूप में स्पष्ट रूप से वर्णित
  कीजिए, और आगमन से सिद्ध कीजिए: $C_n$ $2^n$ \omterm{def:b1:topology:closed}{संवृत} खंडों का असंयुक्त सम्मिलन
  है, जिनमें से प्रत्येक की लंबाई $3^{-n}$ है।
  \item दिखाइए कि $C$ \omterm{def:b1:topology:closed}{संवृत}, परिबद्ध --- अतः संहत
  (\cref{thm:b1:topology:compact}) --- और अरिक्त है, और यह कि प्रत्येक $C_n$
  के प्रत्येक खंड का प्रत्येक सिरा $C$ का सदस्य है।
  \item $C_n$ की कुल लंबाई $\bigl(\frac23\bigr)^n$ है। उससे निकालिए कि
  प्रत्येक $\varepsilon > 0$ के लिए \omterm{def:b1:logic:sets}{समुच्चय} $C$ को कुल लंबाई $\leq
  \varepsilon$ वाले परिमित कितने खंडों से ढका जा सकता है: अर्थात् \omterm{pb:b1:topology:1}{कैंटर
  समुच्चय} की \emph{लंबाई शून्य} है।
  \item दिखाइए कि $C$ में समाए \omterm{prop:b1:reals:intervals}{अंतराल} की लंबाई प्रत्येक $n$ के लिए $\leq
  3^{-n}$ है, अतः वह एकल \omterm{def:b1:logic:sets}{समुच्चय} है या रिक्त: $\mathring C = \emptyset$।
  \omterm{def:b1:topology:closed}{संवृत} होने के साथ रिक्त \omterm{def:b1:topology:closure}{अंतःभाग} वाला होने से $C$ \emph{कहीं \omterm{def:b1:topology:dense}{सघन} नहीं} है।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- त्रिआधारी कूट।}
\begin{enumerate}[resume]
  \item स्वसमरूपता की पुनरावृत्ति सिद्ध कीजिए
        \[
        C_{n+1} = \tfrac13 C_n \,\cup\,
        \bigl(\tfrac23 + \tfrac13 C_n\bigr),
        \qquad\text{अतः}\qquad
        C = \tfrac13 C \,\cup\, \bigl(\tfrac23 + \tfrac13
        C\bigr),
        \]
        जहाँ दोनों टुकड़े असंयुक्त हैं: अर्थात् $C$ मापक $\frac13$ पर अपनी
        ही दो प्रतियाँ है।
  \item $n$ पर आगमन से सिद्ध कीजिए: $x \in C_n$ यदि और केवल यदि $x$ का कोई
  ऐसा त्रिआधारी कूट हो जिसके पहले $n$ अंक $\{0, 2\}$ में हों। इस तथ्य का
  प्रयोग करके कि $x$ के अधिक से अधिक दो कूट हैं, निकालिए: $x \in C$ यदि और
  केवल यदि $x$ का कोई ऐसा कूट हो जिसमें \emph{कोई अंक $1$ के बराबर न हो} (एक
  \emph{$1$-मुक्त} कूट)।
  \item कूट काम पर: $0$, $1$, $\frac13$, $\frac23$ के $1$-मुक्त कूट दीजिए;
  दिखाइए $\frac14 = (0.\overline{02})_3$ और $\frac34 = (0.\overline{20})_3$,
  अतः दोनों $C$ के सदस्य हैं; और जाँचिए कि $\frac14$ किसी भी $C_n$ का सिरा
  \emph{नहीं} है (सिरों का रूप $m/3^n$ होता है)।
  \item दिखाइए कि प्रत्येक $x \in C$ का \emph{ठीक एक} $1$-मुक्त कूट है
  \emph{(जब $x$ के दो कूट हों, तब सिद्ध कीजिए कि उस जोड़े में से ठीक एक में
  अंक $1$ है)}। निष्कर्ष: मान-प्रतिचित्रण $\{0,2\}$-मालाओं से $C$ पर एकैकी
  आच्छादन है।
  \item (विकर्ण) मान लीजिए $k \mapsto x_k$ कोई \omterm{def:b1:logic:map}{प्रतिचित्रण} $\N^* \to C$ है।
  ऐसी $\{0, 2\}$-माला बनाइए जो सूचकांक $k$ पर $x_k$ के कूट से भिन्न हो, और
  निष्कर्ष निकालिए कि $C$ अगणनीय है --- जबकि इसके विपरीत प्रश्न 3 कहता है कि
  वह मापीय दृष्टि से नगण्य है।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- सांस्थितिक चित्र।}
\begin{enumerate}[resume]
  \item अब तक का लेखा जोड़िए: $C$ संहत, अगणनीय, लंबाई-शून्य और कहीं \omterm{def:b1:topology:dense}{सघन} नहीं
  है। कौन-सी एक अंतर्विष्टि $C \subseteq C_n$ किस गुणधर्म को ढोती है?
  \item ($C$ पूर्ण है) मान लीजिए $1$-मुक्त कूट $(d_k)$ के साथ $x \in C$। अंक
  $d_n$ ($0 \leftrightarrow 2$) को पलटने से $x_n \in C$ मिलता है, जहाँ
  $\abs{x_n - x} = 2 \cdot 3^{-n}$। निष्कर्ष निकालिए कि $C$ का कोई वियुक्त
  बिंदु नहीं है: $C$ का प्रत्येक बिंदु $C$ के \emph{दूसरे} बिंदुओं की सीमा
  है।
  \item दिखाइए कि प्रश्न 2 के सिरे $C$ का गणनीय \omterm{def:b1:topology:dense}{सघन} उपसमुच्चय बनाते हैं
  \emph{(कूट को $n$ अंकों के बाद काटिए और आगे $0$ लगाइए; गणनीयता
  \cref{exo:b1:topology:10} की भाँति)}। निष्कर्ष: $C$ का विशिष्ट बिंदु ---
  जैसे $\frac14$ --- कोई सिरा \emph{नहीं} है: सिरे उस अगणनीय काया के भीतर एक
  गणनीय ढाँचा हैं।
  \item (पूर्णतः असंबद्ध) मान लीजिए $C$ में $x < y$। ऐसा $n$ चुनिए कि
  $3^{-n} < y - x$, और ऐसा बिंदु $z \in \intoo{x}{y}$ उत्पन्न कीजिए कि $z
  \notin C$। निष्कर्ष निकालिए कि $C$ में समाए एकमात्र अरिक्त \omterm{prop:b1:reals:intervals}{अंतराल} एकल
  \omterm{def:b1:logic:sets}{समुच्चय} हैं।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- $C$ का अंकगणित।}
\begin{enumerate}[resume]
  \item दिखाइए $1 - C = C$ \emph{($x \mapsto 1 - x$ किसी $1$-मुक्त कूट के
  साथ क्या करता है? $1 = (0.\overline{2})_3$ स्मरण कीजिए।)}
  \item (कूटों का योग) दिखाइए कि यदि $x$, $x'$ के कूट $(a_k)$, $(b_k)$ हैं,
  तो $x + x' = \lim_n\,(t_n + t'_n)$, जहाँ $t_n, t'_n$ आंशिक योग हैं। उससे
  निकालिए: प्रत्येक $y \in \intcc{0}{1}$ $C$ के दो बिंदुओं का
  \emph{मध्यबिंदु} है --- $y$ का कोई कूट $(e_k)$ दिया हो, तो ऐसे अंक $a_k,
  b_k \in \{0, 2\}$ चुनिए कि $\frac{a_k + b_k}{2} = e_k$।
  \item निष्कर्ष निकालिए $C + C = \intcc{0}{2}$, और प्रश्न 14 के साथ $C - C
  = \intcc{-1}{1}$। मूर्त उदाहरण: $1$ को प्रश्न 7 में मिले दोनों गैर-सिरा
  बिंदुओं के योग के रूप में लिखिए।
  \item विचार कीजिए: लंबाई-शून्य वाला ऐसा \omterm{def:b1:logic:sets}{समुच्चय} जिसका अंतर-समुच्चय
  $\intcc{-1}{1}$ भर देता है। ``$C$ मापीय दृष्टि से नगण्य है'' और ``$C + C$
  की लंबाई पूरी है'' में कोई विरोध क्यों नहीं है? (एक वाक्य; सोचिए कि लंबाई
  क्या नियंत्रित करती है और क्या नहीं।)
\end{enumerate}

\medskip \noindent\textbf{भाग V --- सदस्य, परिमेय और अपरिमेय।}
\begin{enumerate}[resume]
  \item प्रश्न 8 को \cref{pb:b1:reals:1} की आवर्तिता कसौटी के साथ मिलाइए:
  $C$ का कोई बिंदु परिमेय है यदि और केवल यदि उसका $1$-मुक्त कूट अंततः आवर्ती
  हो। $\frac1{13} = (0.\overline{002})_3 \in C$ जाँचने के लिए आधार-$3$ का
  लंबा भाग चलाइए।
  \item $C$ का एक स्पष्ट रूप से \emph{अपरिमेय} सदस्य उत्पन्न कीजिए: उस कूट
  का मान जिसमें त्रिभुजीय स्थानों $k = \frac{j(j+1)}{2}$ पर $d_k = 2$ हों और
  अन्यत्र $d_k = 0$। अपरिमेयता को \cref{pb:b1:reals:1} (प्रश्न 20) के
  बढ़ते-अंतराल वाले तर्क से न्यायसंगत ठहराइए।
  \item (पूरे खंड पर आच्छादन) $h$ पर विचार कीजिए, जो $C$ के उस बिंदु को,
  जिसका $1$-मुक्त कूट $(d_k)$ है, \emph{द्विआधारी} माला
  $\bigl(\frac{d_k}2\bigr)$ के मान पर भेजता है, अर्थात् $h(x) = \sup_n
  \sum_{k=1}^n \frac{d_k}{2}\,2^{-k}$। दिखाइए कि $h$ $C$ को $\intcc{0}{1}$
  \emph{पर} आच्छादित करता है। अतः नगण्य $C$ पूरी लंबाई वाले खंड पर आच्छादित
  हो जाता है --- और यह $C$ की अगणनीयता की दूसरी उपपत्ति है।
  \item (स्वसमरूप लंबाई) मान लीजिए $C$ और उसकी सिकुड़ी हुई प्रतियों के लिए
  लंबाई की कोई धारणा $L$ परिभाषित होती जो मापन ($L(\lambda A) = \lambda
  L(A)$), स्थानांतरण-अपरिवर्त्यता और प्रश्न 5 के असंयुक्त वियोजन पर
  योगात्मकता का आदर करती। दिखाइए कि तब $L(C) = \frac23\,L(C)$, जो $L(C) = 0$
  पर बाध्य कर देता है: अर्थात् केवल स्वसमरूपता ही $C$ को लंबाई-शून्य का दंड
  सुना देती है।
\end{enumerate}

\medskip \noindent\textbf{भाग VI --- एक मोटा चचेरा भाई, और उपदेश।}
\begin{enumerate}[resume]
  \item (मोटा \omterm{pb:b1:topology:1}{कैंटर समुच्चय}) वही रचना दोहराइए, पर चरण $n$ ($n = 0, 1, 2,
  \dots$) पर वर्तमान $2^n$ खंडों में से प्रत्येक से केवल $4^{-(n+1)}$ लंबाई
  का एक केंद्रीय \omterm{def:b1:topology:open}{विवृत} \omterm{prop:b1:reals:intervals}{अंतराल} हटाइए। दिखाइए कि खंड-लंबाइयाँ $l_n$ $l_{n+1} =
  \frac{l_n - 4^{-(n+1)}}{2}$, $l_n = \frac{2^n + 1}{2\cdot 4^n} > 0$ का
  पालन करती हैं, कि परिणामी $K = \bigcap K_n$ रिक्त \omterm{def:b1:topology:closure}{अंतःभाग} वाला संहत
  \omterm{def:b1:logic:sets}{समुच्चय} है, और यह कि कुल हटाई गई लंबाई $\sum_{n\geq0} 2^n 4^{-(n+1)} =
  \frac12$ है। अंतरालों के परिमित सम्मिलनों के लिए लंबाई की (सहज, स्नातक
  वर्ष 3 वाली) योगात्मकता मान लेते हुए और \cref{ex:b1:topology:nested} के
  दोनों कथनों का प्रयोग करते हुए दिखाइए कि $K$ को ढकने वाले \omterm{def:b1:topology:open}{विवृत} अंतरालों
  के किसी भी परिमित परिवार की कुल लंबाई $\geq \frac12$ है: अर्थात् $K$ कहीं
  \omterm{def:b1:topology:dense}{सघन} नहीं है पर नगण्य \emph{नहीं} है। छोटेपन के कई असमतुल्य अर्थ हैं।
  \item (दूरियाँ) दिखाइए कि अरिक्त \omterm{def:b1:topology:closed}{संवृत} $F \subseteq \R$ और $x \in \R$ के
  लिए \omterm{def:b1:reals:bounds}{निम्नतम} $d(x, F)$ \emph{प्राप्त होता है} \emph{(न्यूनकारी अनुक्रम और
  बोल्ज़ानो--वाइरश्ट्रास)}। फिर संगणना कीजिए
        \[
        \max_{y \in \intcc{0}{1}} d(y, C) = \frac16 ,
        \]
        जो ठीक केंद्र $y = \frac12$ पर प्राप्त होता है \emph{(चरण $n$ पर बने
        \omterm{prop:b1:reals:intervals}{अंतराल} का कोई बिंदु उस \omterm{prop:b1:reals:intervals}{अंतराल} के सिरों से $\frac{3^{-n}}{2}$ के भीतर
        है, और वे सिरे $C$ में हैं)}।
  \item (हर बिंदु एक उपानुक्रमीय सीमा) प्रश्न 12 और 2 का प्रयोग करके $C$ में
  एक ही ऐसा अनुक्रम उत्पन्न कीजिए जिसकी उपानुक्रमीय सीमाओं का \omterm{def:b1:logic:sets}{समुच्चय}
  \emph{पूरा} $C$ हो। (तुलना: अभिसारी अनुक्रम के लिए वह \omterm{def:b1:logic:sets}{समुच्चय} एक ही बिंदु
  होता है --- $C$ संहत समुच्चयों में विपरीत छोर को साकार करता है।)
  \item संश्लेषण, एक-एक वाक्य में: (क) इस रचना ने इस अध्याय की कौन-सी
  प्रमेयें सचमुच खपाईं (\omterm{def:b1:topology:closed}{संवृत} समुच्चयों का स्थायित्व, संहतता, अनुक्रमीय
  अभिलक्षण)? (ख) इस चित्र के चार विरोधाभासी युग्म गिनाइए (लंबाई
  शून्य/अगणनीय, \omterm{def:b1:topology:closed}{संवृत}/रिक्त \omterm{def:b1:topology:closure}{अंतःभाग}, पूर्ण/पूर्णतः असंबद्ध, नगण्य/$C+C$
  पूर्ण); (ग) $C$ आगे कहाँ फिर उभरता है (सांतत्य के सिद्धांत में $h$ पर बनी
  शैतान की सीढ़ी, और स्नातक वर्ष 3 के खंड का माप-सिद्धांत, जहाँ $C$
  ``गणनीय'' को ``नगण्य'' से अलग कर देता है)?
\end{enumerate}
\end{problem}
