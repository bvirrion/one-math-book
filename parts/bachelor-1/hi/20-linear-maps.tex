\chapter{रैखिक प्रतिचित्रण}\label{ch:b1:linmaps}

सदिश समष्टियों के बीच अध्ययन योग्य \omterm{def:b1:logic:map}{प्रतिचित्रण} वही हैं जो संरचना के अनुकूल
हों: \emph{\omterm{def:b1:linmaps:def}{रैखिक प्रतिचित्रण}}। उनकी दो मूलभूत उपसमष्टियाँ --- \omterm{def:b1:linmaps:kerim}{अष्टि} और
\omterm{def:b1:linmaps:kerim}{प्रतिबिंब} --- एकैकीपन और आच्छादकता नापती हैं, और \omterm{def:b1:findim:def}{परिमित विमा} में
कोटि--शून्यता प्रमेय उनके आकारों को एक ही संरक्षण नियम में बाँध देती है।
\omterm{def:b1:linmaps:projection}{प्रक्षेप} और सममितियाँ, फिर \omterm{def:b1:linmaps:forms}{रैखिक रूप} और \omterm{def:b1:linmaps:forms}{अधिसमतल}, अध्याय को बंद करते हैं।

\section{परिभाषाएँ और प्रथम गुण}

\begin{definition}\label{def:b1:linmaps:def}
मान लीजिए $E, F$ $K$-सदिश समष्टियाँ हैं। \omterm{def:b1:logic:map}{प्रतिचित्रण} $u \colon E \to F$
\emph{रैखिक}\index{रैखिक प्रतिचित्रण} तब कहलाता है जब
\[
\forall x, y \in E,\ \forall \lambda \in K, \qquad
u(x + \lambda y) = u(x) + \lambda u(y).
\]
तब $u(0) = 0$ और $u(\sum \lambda_i x_i) = \sum \lambda_i u(x_i)$। रैखिक
प्रतिचित्रणों का \omterm{def:b1:logic:sets}{समुच्चय} $\mathcal{L}(E, F)$ स्वयं एक \omterm{def:b1:vspaces:def}{सदिश समष्टि} है; रैखिक
प्रतिचित्रणों का संयुक्त \omterm{def:b1:logic:map}{प्रतिचित्रण} रैखिक होता है, और जोड़ी में द्विरैखिक।
\emph{अंतःसमाकारिता} कोई रैखिक $E \to E$ है; \emph{तुल्याकारिता} कोई \omterm{def:b1:logic:inj}{एकैकी
आच्छादक} रैखिक \omterm{def:b1:logic:map}{प्रतिचित्रण} है (तब उसका प्रतिलोम स्वतः रैखिक होता है); किसी
तुल्याकारिता का $u^{-1}$, तथा तुल्याकारिताओं के संयुक्त \omterm{def:b1:logic:map}{प्रतिचित्रण},
तुल्याकारिताएँ होते हैं।
\end{definition}

\begin{proof}[प्रतिलोम के रैखिक होने की उपपत्ति]
मान लीजिए $u$ \omterm{def:b1:linmaps:def}{रैखिक} और \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है, $y, y' \in F$ और $\lambda \in K$।
$x = u^{-1}(y)$ तथा $x' = u^{-1}(y')$ रखिए। तब
\[
u\bigl(x + \lambda x'\bigr) = u(x) + \lambda u(x')
= y + \lambda y' ,
\]
और दोनों सिरों पर $u^{-1}$ लगाने से: $u^{-1}(y + \lambda y') = x + \lambda
x' = u^{-1}(y) + \lambda\,u^{-1}(y')$। $u^{-1}$ के विषय में कुछ भी संगणित
नहीं किया गया: रैखिकता केवल $u$ के परिभाषक गुण के सहारे स्थानांतरित हो जाती
है --- यह प्रतिरूप याद रखने योग्य है, क्योंकि संरचना प्रायः एकैकी आच्छादनों
पर मुफ़्त में सवारी कर लेती है।
\end{proof}

\begin{proposition}[रैखिक प्रतिचित्रण आधार पर जान लिया जाता है]\label{prop:b1:linmaps:basis}
मान लीजिए $(e_1, \dots, e_n)$ $E$ का \omterm{def:b1:vspaces:free}{आधार} है और $(v_1, \dots, v_n)$ $F$ के
स्वेच्छ सदिश। ठीक एक ही \omterm{def:b1:linmaps:def}{रैखिक प्रतिचित्रण} $u \colon E \to F$ ऐसा है कि सभी
$i$ के लिए $u(e_i) = v_i$। इसके अतिरिक्त:
\[
u \text{ एकैकी} \iff (v_i) \text{ स्वतंत्र};
\qquad
u \text{ आच्छादक} \iff (v_i),\ F \text{ का जनक कुल है} .
\]
\end{proposition}

\begin{proof}
अस्तित्व/अद्वितीयता: हर $x$ के \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} $x = \sum \lambda_i e_i$ अद्वितीय
हैं (\cref{prop:b1:vspaces:coordinates}); रैखिकता $u(x) = \sum \lambda_i
v_i$ बाध्य कर देती है, और यह सूत्र सचमुच एक \omterm{def:b1:linmaps:def}{रैखिक प्रतिचित्रण} परिभाषित करता
है।

एकैकीपन: नीचे के \cref{prop:b1:linmaps:kernel} से $u$ \omterm{def:b1:logic:inj}{एकैकी} है तभी जब उसकी
\omterm{def:b1:linmaps:kerim}{अष्टि} तुच्छ हो। अब $u\bigl(\sum\lambda_i e_i\bigr) = 0$ का अर्थ ठीक
$\sum\lambda_i v_i = 0$ है। यदि $(v_i)$ \omterm{def:b1:vspaces:free}{स्वतंत्र} है, तो यह हर $\lambda_i =
0$ बाध्य कर देता है, अर्थात् \omterm{def:b1:linmaps:kerim}{अष्टि} घटकर $0$ रह जाती है: अतः \omterm{def:b1:logic:inj}{एकैकी}। यदि
$(v_i)$ परतंत्र है, तो कोई अतुच्छ संबंध $\sum\lambda_i v_i = 0$ \omterm{def:b1:linmaps:kerim}{अष्टि} में
अशून्य सदिश $\sum\lambda_i e_i$ पैदा कर देता है ($(e_i)$ \omterm{def:b1:vspaces:free}{स्वतंत्र} है): अतः
\omterm{def:b1:logic:inj}{एकैकी} नहीं। दोनों शर्तें पद-दर-पद मेल खाती हैं।

आच्छादकता: $u$ का \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} सभी $\sum\lambda_i v_i$ का \omterm{def:b1:logic:sets}{समुच्चय} है, अर्थात्
ठीक $\operatorname{Vect}(v_1, \dots, v_n)$, जो $F$ के बराबर है तभी जब वह कुल
जनक हो।
\end{proof}

\begin{definition}[अष्टि और प्रतिबिंब]\label{def:b1:linmaps:kerim}
$u \in \mathcal{L}(E, F)$ के लिए:
\[
\ker u = \{x \in E : u(x) = 0\} \subseteq E,
\qquad
\operatorname{im} u = u(E) \subseteq F ,
\]
दोनों उपसमष्टियाँ हैं (कसौटी से सीधा सत्यापन)।\index{अष्टि}\index{प्रतिबिंब}
\end{definition}

\begin{method}[अष्टि और प्रतिबिंब, व्यवहार में]
\label{met:b1:linmaps:kerim}
\emph{\omterm{def:b1:linmaps:kerim}{अष्टि}}: $u(x) = 0$ को $x$ के निर्देशांकों (अथवा गुणांकों) पर एक निकाय
के रूप में लिखिए, हल कीजिए, और प्राचलन कीजिए --- \omterm{def:b1:linmaps:kerim}{अष्टि} \omterm{def:b1:vspaces:free}{आधार} समेत निकल आती है
(\cref{met:b1:findim:computedim})। \emph{\omterm{def:b1:linmaps:kerim}{प्रतिबिंब}}: वह $E$ के \emph{किसी
भी} जनक कुल के प्रतिबिंबों की \omterm{def:b1:vspaces:span}{जनित समष्टि} है --- प्रायः किसी \omterm{def:b1:vspaces:free}{आधार} की, अतः
$\operatorname{im} u = \operatorname{Vect}\bigl(u(e_1), \dots,
u(e_n)\bigr)$; फिर अनावश्यक \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} हटाकर \omterm{def:b1:vspaces:free}{आधार} निकाल लीजिए। \emph{छोटा
रास्ता}: दोनों में से जो सरल हो उसे संगणित कीजिए और दूसरे की विमा
कोटि--शून्यता से मुफ़्त में पाइए (\cref{thm:b1:linmaps:ranknullity}); और जब
\omterm{def:b1:linmaps:kerim}{प्रतिबिंब} के लिए कोई विश्वसनीय संभावित \omterm{def:b1:vspaces:subspace}{उपसमष्टि} ज्ञात हो, तो विमाओं की तुलना
सरल अंतर्विष्टता को समानता में बदल देती है (\cref{thm:b1:findim:subspaces})।
नीचे \cref{ex:b1:linmaps:delta} में दोनों छोटे रास्ते प्रयुक्त हुए हैं।
\end{method}

\begin{proposition}\label{prop:b1:linmaps:kernel}
$u$ \omterm{def:b1:logic:inj}{एकैकी} है $\iff$ $\ker u = \{0\}$; $u$ \omterm{def:b1:logic:inj}{आच्छादक} है $\iff$
$\operatorname{im} u = F$।
\end{proposition}

\begin{proof}
समूहों की तरह ही (\cref{prop:b1:structures:kernel}): $u(x) = u(y) \iff u(x -
y) = 0 \iff x - y \in \ker u$। दूसरा बिंदु परिभाषा ही है।
\end{proof}

\section{कोटि--शून्यता प्रमेय}

\begin{definition}\label{def:b1:linmaps:rank}
$u \in \mathcal{L}(E, F)$ की \emph{कोटि}\index{कोटि!किसी रैखिक प्रतिचित्रण
की} ($E$ के \omterm{def:b1:findim:def}{परिमित-विमीय} होने पर) $\operatorname{rk} u = \dim
\operatorname{im} u$ है --- और वह $E$ के किसी भी \omterm{def:b1:vspaces:free}{आधार} $(e_i)$ के लिए कुल
$\bigl(u(e_1), \dots, u(e_n)\bigr)$ की कोटि भी है।
\end{definition}

\begin{theorem}[कोटि--शून्यता]\label{thm:b1:linmaps:ranknullity}
मान लीजिए $E$ \omterm{def:b1:findim:def}{परिमित-विमीय} है और $u \in \mathcal{L}(E, F)$।
तब\index{कोटि--शून्यता प्रमेय}
\[
\dim E = \dim \ker u + \operatorname{rk} u .
\]
अधिक यथार्थ रूप से, यदि $S$ $E$ में $\ker u$ की कोई \omterm{def:b1:vspaces:sum}{पूरक} \omterm{def:b1:vspaces:subspace}{उपसमष्टि} है, तो $u$
सीमित होकर $S$ से $\operatorname{im} u$ पर एक \emph{तुल्याकारिता} बन जाता
है।
\end{theorem}

\begin{proof}
मान लीजिए $S$ $E = \ker u \oplus S$ संतुष्ट करती है
(\cref{thm:b1:findim:subspaces}), और $v \colon S \to \operatorname{im} u$
$u$ का सीमन है।

$v$ \omterm{def:b1:logic:inj}{एकैकी} है: $\ker v = S \cap \ker u = \{0\}$।

$v$ \omterm{def:b1:logic:inj}{आच्छादक} है: $x = k + s$ वाला कोई भी $u(x)$ ($k \in \ker u$, $s \in S$)
$u(s) = v(s)$ के बराबर है।

अतः $v$ एक तुल्याकारिता है; और तुल्याकारिता \omterm{def:b1:vspaces:free}{आधार} को \omterm{def:b1:vspaces:free}{आधार} पर भेजती है
(\cref{prop:b1:linmaps:basis}), अतः $\dim S = \dim\operatorname{im} u$, और
$\dim E = \dim\ker u + \dim S$ निष्कर्ष दे देता है।
\end{proof}

\begin{example}[माँग के अनुसार प्रतिचित्रण बनाना]
\label{ex:b1:linmaps:tospec}
ऐसा $u \in \mathcal{L}(\R^3)$ बनाइए कि $\ker u = \operatorname{Vect}(1,1,1)$
और $\operatorname{im} u = \{z = 0\}$। पहले जाँच: कोटि--शून्यता $1 + 2 = 3$
माँगती है --- जो संगत है, अतः कोई हल हो सकता है। \omterm{def:b1:linmaps:kerim}{अष्टि} के अनुकूल कोई \omterm{def:b1:vspaces:free}{आधार}
चुनिए, जैसे $\bigl((1,1,1),\ e_1,\ e_2\bigr)$
(\cref{ex:b1:findim:completion}), और \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} निर्धारित कीजिए
(\cref{prop:b1:linmaps:basis}):
\[
u(1,1,1) = 0, \qquad u(e_1) = e_1, \qquad u(e_2) = e_2 .
\]
तब $\ker u \supseteq \operatorname{Vect}(1,1,1)$ और $\operatorname{im} u =
\operatorname{Vect}(e_1, e_2) = \{z = 0\}$; कोटि--शून्यता $\dim\ker u = 1$
बाध्य कर देती है, अतः \omterm{def:b1:linmaps:kerim}{अष्टि} ठीक वही निर्धारित रेखा है। स्पष्ट रूप से, $(x,
y, z) = z(1,1,1) + (x - z)e_1 + (y - z)e_2$ का विघटन करने पर:
\[
u(x, y, z) = (x - z,\ y - z,\ 0).
\]
यह नुस्ख़ा सामान्यीकृत हो जाता है: निर्धारित \omterm{def:b1:linmaps:kerim}{अष्टि} $N$ और निर्धारित
\omterm{def:b1:linmaps:kerim}{प्रतिबिंब} $I$ वाला \omterm{def:b1:linmaps:def}{रैखिक प्रतिचित्रण} ठीक तब होता है जब $\dim N + \dim I =
\dim E$ --- आवश्यकता कोटि--शून्यता है, और पर्याप्तता यही रचना।
\end{example}

\begin{corollary}\label{cor:b1:linmaps:samedim}
यदि $\dim E = \dim F$ (परिमित), तो $u \in \mathcal{L}(E, F)$ के लिए:
\[
u \text{ एकैकी} \iff u \text{ आच्छादक} \iff u \text{एकैकी आच्छादक}.
\]
विशेष रूप से यह \omterm{def:b1:findim:def}{परिमित विमा} में अंतःसमाकारिताओं पर लागू होता है। (अनंत विमा
में यह विफल हो जाता है: $K[X]$ पर \omterm{def:b1:derivative:def}{अवकलज} \omterm{def:b1:logic:inj}{आच्छादक} है पर \omterm{def:b1:logic:inj}{एकैकी} नहीं, और $P
\mapsto XP$ \omterm{def:b1:logic:inj}{एकैकी} है पर \omterm{def:b1:logic:inj}{आच्छादक} नहीं।)
\end{corollary}

\begin{proof}
\omterm{def:b1:logic:inj}{एकैकी} $\iff \dim\ker u = 0 \iff \operatorname{rk} u = \dim E = \dim F \iff
\operatorname{im} u = F$ (पूरी विमा वाली \omterm{def:b1:vspaces:subspace}{उपसमष्टि} सब कुछ होती है,
\cref{thm:b1:findim:subspaces}) $\iff$ \omterm{def:b1:logic:inj}{आच्छादक}।
\end{proof}

\begin{example}[अंतर्वेशन, संरचनात्मक दृष्टि से]\label{ex:b1:linmaps:interpolation}
भिन्न-भिन्न $x_0, \dots, x_n$ नियत कीजिए और $u \colon \R_n[X] \to \R^{n+1}$,
$P \mapsto (P(x_0), \dots, P(x_n))$ लीजिए: यह \omterm{def:b1:linmaps:def}{रैखिक} है। उसकी \omterm{def:b1:linmaps:kerim}{अष्टि} $\{P :
\deg P \leq n,\ n+1 \text{ मूल}\} = \{0\}$ है (\cref{cor:b1:poly:nroots})।
विमाएँ बराबर हैं $n + 1$: अतः $u$ एक तुल्याकारिता है --- अर्थात् लाग्रांज
अंतर्वेशक (\cref{thm:b1:poly:lagrange}) का अस्तित्व \emph{और} अद्वितीयता,
दोनों एक ही पंक्ति में।

यही एक-पंक्ति वाला प्रतिरूप उन आँकड़ों को भी सँभाल लेता है जिनमें मान और
\omterm{def:b1:derivative:def}{अवकलज} मिले-जुले हों: $v \colon \R_3[X] \to \R^4$, $P \mapsto \bigl(P(0),
P'(0), P(1), P'(1)\bigr)$ \omterm{def:b1:linmaps:def}{रैखिक} है, और उसकी \omterm{def:b1:linmaps:kerim}{अष्टि} में $\leq 3$ घात वाले वे
\omterm{def:b1:poly:def}{बहुपद} हैं जिनके $0$ \emph{और} $1$ दोनों पर दुहरे मूल हों, अर्थात् जो $4$ घात
वाले $X^2(X-1)^2$ से विभाज्य हों: केवल $P = 0$। विमाएँ फिर बराबर हैं: अतः
आँकड़ों का हर चतुष्क $(P(0), P'(0), P(1), P'(1))$ ठीक एक ही त्रिघातीय \omterm{def:b1:poly:def}{बहुपद}
से साकार होता है --- यही एरमीट अंतर्वेशन है, जो कोई सूत्र लिखे जाने से पहले
ही एक अष्टि-संगणना से मिल जाता है (\cref{ch:b1:det} की सप्ताहांत समस्या अपने
सारणिक से मिलती है)।
\end{example}

\begin{example}[कोटि--शून्यता काम पर: अंतर संकारक]
\label{ex:b1:linmaps:delta}
मान लीजिए $\Delta \colon \R_n[X] \to \R_n[X]$, $P \mapsto P(X+1) - P(X)$: यह
\omterm{def:b1:linmaps:def}{रैखिक} है। \omterm{def:b1:linmaps:kerim}{अष्टि}: यदि $\Delta P = 0$, तो $P(0) = P(1) = P(2) = \dots$, अतः $P
- P(0)$ के अनंत कितने मूल हैं और वह शून्य हो जाता है
(\cref{cor:b1:poly:nroots}): अर्थात् $\ker\Delta$ अचरों की रेखा है।
कोटि--शून्यता: $\operatorname{rk}\Delta = (n + 1) - 1 = n$। चूँकि अनचर $P$
के लिए $\deg \Delta P < \deg P$ (शीर्ष पद कट जाते हैं), अतः
$\operatorname{im}\Delta \subseteq \R_{n-1}[X]$, जिसकी विमा ठीक $n$ है:
इसलिए यह अंतर्विष्टता समानता है। निष्कर्ष, और वह भी किसी \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} की
संगणना के बिना: $\leq n - 1$ घात वाला \emph{हर} \omterm{def:b1:poly:def}{बहुपद} $Q$ किसी न किसी अंतर
$Q = P(X+1) - P(X)$ के रूप में है --- अर्थात् विविक्त प्रतिअवकलज का अस्तित्व
है। (\cref{ch:b1:vspaces} की सप्ताहांत समस्या से तुलना कीजिए, जहाँ $\Delta$
को द्विपद \omterm{def:b1:vspaces:free}{आधार} में स्पष्ट रूप से उलटा गया था।)
\end{example}

\begin{example}[किसी संयुक्त प्रतिचित्रण के अनुदिश कोटि का लेखा-जोखा]
\label{ex:b1:linmaps:rankbookkeeping}
$\R_2[X]$ पर \omterm{def:b1:derivative:def}{अवकलज} $D(P) = P'$ (\omterm{def:b1:linmaps:rank}{कोटि} $2$: \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} $\R_1[X]$, \omterm{def:b1:linmaps:kerim}{अष्टि} अचर) को
स्वयं से संयुक्त कीजिए। तब $D \circ D = D^2$ $P \mapsto P''$ भेजता है, जिसका
\omterm{def:b1:linmaps:kerim}{प्रतिबिंब} $\R_0[X]$ है: \omterm{def:b1:linmaps:rank}{कोटि} $1$। व्यापक परिबंधों से तुलना कीजिए: स्थूल
परिबंध $\operatorname{rk} D^2 \leq \min(2, 2) = 2$ देता है; और
\cref{exo:b1:linmaps:12} का यथार्थ सूत्र इस हानि का ठीक-ठीक हिसाब देता है,
\[
\operatorname{rk} D^2 = \operatorname{rk} D -
\dim\bigl(\ker D \cap \operatorname{im} D\bigr) = 2 - 1 = 1 ,
\]
क्योंकि अचर (बाहरी $D$ की \omterm{def:b1:linmaps:kerim}{अष्टि}) $\R_1[X]$ (भीतरी $D$ का \omterm{def:b1:linmaps:kerim}{प्रतिबिंब}) के भीतर
बैठे हैं, और उनकी विमा $1$ है। \omterm{def:b1:linmaps:rank}{कोटि} ठीक वहीं खोती है जहाँ बाहरी \omterm{def:b1:linmaps:kerim}{अष्टि} भीतरी
\omterm{def:b1:linmaps:kerim}{प्रतिबिंब} पर घात लगाती है --- संयुक्त प्रतिचित्रणों की कोटियाँ बदमाशी करें
तो यही वाक्य याद रखिए।
\end{example}

\begin{example}[ऑयलर संकारक की अष्टि और प्रतिबिंब]
\label{ex:b1:linmaps:euler}
$\R_n[X]$ पर मान लीजिए $u(P) = X\,P'$ (यह \omterm{def:b1:linmaps:def}{रैखिक} है: अवकलन और $X$ से गुणन,
दोनों \omterm{def:b1:linmaps:def}{रैखिक} हैं)। \emph{\omterm{def:b1:linmaps:kerim}{अष्टि}}: $XP' = 0$ $P' = 0$ बाध्य कर देता है (बहुपदों
का गुणनफल तभी शून्य होता है जब कोई एक गुणनखंड शून्य हो), अतः $\ker u$ अचरों
की रेखा है। \emph{\omterm{def:b1:linmaps:kerim}{प्रतिबिंब}}: एकपदी \omterm{def:b1:vspaces:free}{आधार} पर,
\[
u(X^k) = k\,X^{k} \qquad (k = 0, 1, \dots, n),
\]
अतः $\operatorname{im} u = \operatorname{Vect}(X, 2X^2, \dots, nX^n) =
\operatorname{Vect}(X, X^2, \dots, X^n)$: अर्थात् शून्य अचर पद वाले \omterm{def:b1:poly:def}{बहुपद}।
कोटि--शून्यता से जाँच: $\operatorname{rk} u = (n + 1) - 1 = n$, जो सचमुच वही
विमा है जो मिली। दो बातें रखने योग्य हैं। पहली, यहाँ $\operatorname{im} u
\oplus \ker u = \R_n[X]$ --- पर यह इस संकारक का एक \emph{सुखद संयोग} है, कोई
प्रमेय नहीं: $\R_1[X]$ पर सरकाव जैसे $v(P) = P'$ के लिए $\ker v =
\operatorname{im} v = \R_0[X]$ और योग प्रत्यक्ष नहीं है। दूसरी, संबंध
$u(X^k) = kX^k$ कहता है कि हर एकपदी बस $u$ से मापित हो जाती है --- अर्थात्
\omterm{def:b1:logic:map}{प्रतिचित्रण} के अनुकूल एक \omterm{def:b1:vspaces:free}{आधार}, जो स्नातक वर्ष~2 के खंड में विकसित होने वाले
अभिलक्षणिक-मान के विचार का बीज है।
\end{example}

\section{प्रक्षेप और सममितियाँ}

\begin{definition}\label{def:b1:linmaps:projection}
मान लीजिए $E = F \oplus G$। \emph{$G$ के अनुदिश $F$ पर
प्रक्षेप}\index{प्रक्षेप} $x = f + g$ (अद्वितीय विघटन) को $p(x) = f$ पर
भेजता है; और उससे संबद्ध \emph{सममिति}\index{सममिति} $s(x) = f - g$ है।
दोनों \omterm{def:b1:linmaps:def}{रैखिक} हैं, और $s = 2p - \mathrm{id}$।
\end{definition}

\begin{theorem}[बीजगणितीय अभिलक्षण]\label{thm:b1:linmaps:projchar}
\begin{enumerate}
  \item कोई अंतःसमाकारिता $p$ \omterm{def:b1:linmaps:projection}{प्रक्षेप} है (किसी $G$ के अनुदिश किसी $F$ पर)
  तभी जब $p \circ p = p$; और तब $F = \operatorname{im} p = \ker(p -
  \mathrm{id})$ तथा $G = \ker p$।
  \item कोई अंतःसमाकारिता $s$ \omterm{def:b1:linmaps:projection}{सममिति} है तभी जब $s \circ s = \mathrm{id}$; और
  तब $E = \ker(s - \mathrm{id}) \oplus \ker(s + \mathrm{id})$।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) कोई \omterm{def:b1:linmaps:projection}{प्रक्षेप} $p(f + g) = f$ और $p(f) = f$ संतुष्ट करता है: $p^2 = p$।
विलोमतः मान लीजिए $p^2 = p$; $F = \operatorname{im} p$, $G = \ker p$ रखिए।
हर $x$ $p(x) \in F$ और $p\bigl(x - p(x)\bigr) = p(x) - p^2(x) = 0$ के साथ $x
= p(x) + (x - p(x))$ लिखा जाता है: $E = F + G$। यदि $y \in F \cap G$: तो $y
= p(z)$ और $p(y) = 0$, अतः $y = p(z) = p^2(z) = p(y) = 0$: अर्थात् \omterm{def:b1:vspaces:sum}{प्रत्यक्ष
योग}, और $p$ $G$ के अनुदिश $F$ पर \omterm{def:b1:linmaps:projection}{प्रक्षेप} है। अंत में $F$ पर: $y = p(z)$
$p(y) = y$ देता है, अतः $F \subseteq \ker(p - \mathrm{id})$, और विलोमतः
$p(y) = y$ $y$ को \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} में डाल देता है।

(2) संगति $s = 2p - \mathrm{id}$, $p = \frac{s + \mathrm{id}}2$
अंतःसमाकारिताओं के बीच एकैकी आच्छादन है, और उसके अंतर्गत
\[
s^2 = 4p^2 - 4p + \mathrm{id} = \mathrm{id}
\iff 4p^2 = 4p \iff p^2 = p :
\]
सममितियाँ ठीक प्रक्षेपों के अनुरूप हो जाती हैं। उपसमष्टियों का अनुवाद: $s(x)
= x \iff p(x) = x$, अतः $\ker(s - \mathrm{id}) = \operatorname{im} p = F$;
और $s(x) = -x \iff 2p(x) = 0 \iff x \in \ker p = G$, अतः $\ker(s +
\mathrm{id}) = G$। बिंदु (1) का \omterm{def:b1:vspaces:sum}{प्रत्यक्ष योग} $E = F \oplus G$ वही घोषित
विघटन बन जाता है --- $s$ के अचल सदिशों और उलटे सदिशों में।
\end{proof}

\begin{example}[एक प्रक्षेप और उसकी सममिति, स्पष्ट रूप से]
\label{ex:b1:linmaps:projexplicit}
$\R^2$ में $G = \operatorname{Vect}(0,1)$ के अनुदिश $F =
\operatorname{Vect}(1,1)$ पर \omterm{def:b1:linmaps:projection}{प्रक्षेप} कीजिए। $(x, y) = a(1,1) + b(0,1)$ का
विघटन कीजिए: पहला \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} $a = x$ देता है, दूसरा $b = y - x$। अतः
\[
p(x, y) = (x, x),
\qquad
s(x, y) = 2p(x,y) - (x,y) = (x,\ 2x - y).
\]
बीजगणित जाँचिए: $p(p(x,y)) = p(x,x) = (x,x)$, और $s(s(x,y)) = s(x, 2x - y) =
(x, 2x - (2x - y)) = (x, y)$। ज्यामितीय रूप से $s$ ऊर्ध्वाधर दिशा में रेखा
$y = x$ के पार का ``तिरछा परावर्तन'' है: वह $F$ को बिंदुवार अचल रखता है और
$G$ को उलट देता है। यदि हमने उसी $F$ पर $G' = \operatorname{Vect}(1,-1)$ के
अनुदिश \omterm{def:b1:linmaps:projection}{प्रक्षेप} किया होता, तो सूत्र बदलकर $p'(x,y) = \bigl(\frac{x+y}2,
\frac{x+y}2\bigr)$ हो जाता: कोई \omterm{def:b1:linmaps:projection}{प्रक्षेप} अपने \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} \emph{और} अपनी \omterm{def:b1:linmaps:kerim}{अष्टि}
से तय होता है, अकेले \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} से कभी नहीं।
\end{example}

\section{रैखिक रूप और अधिसमतल}

\begin{definition}\label{def:b1:linmaps:forms}
$E$ पर \emph{रैखिक रूप}\index{रैखिक रूप} कोई \omterm{def:b1:linmaps:def}{रैखिक प्रतिचित्रण} $\varphi
\colon E \to K$ है। $E$ ($\dim E = n$) का \emph{अधिसमतल}\index{अधिसमतल} $n -
1$ विमा वाली \omterm{def:b1:vspaces:subspace}{उपसमष्टि} है।
\end{definition}

\begin{example}[एक मूल्यांकन रूप और उसका अधिसमतल]
\label{ex:b1:linmaps:evaluationform}
$\R_2[X]$ पर मूल्यांकन $\varphi(P) = P(2)$ एक \omterm{def:b1:linmaps:forms}{रैखिक रूप} है, और अशून्य
($\varphi(1) = 1$)। उसकी \omterm{def:b1:linmaps:kerim}{अष्टि} $2$ पर शून्य होने वाले बहुपदों का \omterm{def:b1:linmaps:forms}{अधिसमतल} है,
अर्थात् (गुणनखंड प्रमेय, \cref{thm:b1:poly:factor}) $\R_2[X]$ के भीतर $X -
2$ के गुणज:
\[
\ker\varphi = \operatorname{Vect}\bigl(X - 2,\ X(X - 2)\bigr),
\qquad \dim = 2 .
\]
$(1, X, X^2)$ पर निर्देशांकों में $\varphi(a + bX + cX^2) = a + 2b + 4c$:
\omterm{def:b1:findim:def}{परिमित-विमीय} समष्टि पर हर \omterm{def:b1:linmaps:forms}{रैखिक रूप}, \omterm{def:b1:vspaces:free}{आधार} नियत होते ही, निर्देशांकों में एक
नियत रैखिक व्यंजक होता है --- रूप ``पंक्ति सदिश'' हैं, जैसा
\cref{ch:b1:matrices} अक्षरशः बना देगा, और यहाँ गुणांकों की पंक्ति $(1, 2,
4)$ एक वांडरमोंड पंक्ति है: मूल्यांकन रूप ही वे द्वार हैं जिनसे
\cref{ch:b1:det} की सप्ताहांत समस्या का अंतर्वेशन-सिद्धांत रैखिक बीजगणित में
प्रवेश करता है।
\end{example}

\begin{theorem}\label{thm:b1:linmaps:hyperplanes}
$E$ के \omterm{def:b1:linmaps:forms}{अधिसमतल} ठीक अशून्य \omterm{def:b1:linmaps:def}{रैखिक} रूपों की अष्टियाँ हैं। दो अशून्य रूपों की
\omterm{def:b1:linmaps:kerim}{अष्टि} एक ही होती है तभी जब वे अनुपाती हों।
\end{theorem}

\begin{proof}
यदि $\varphi \neq 0$: तो $\operatorname{rk}\varphi = 1$ ($K$ की कोई अशून्य
\omterm{def:b1:vspaces:subspace}{उपसमष्टि} \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} है), अतः $\dim\ker\varphi = n - 1$: अर्थात् एक \omterm{def:b1:linmaps:forms}{अधिसमतल}।
विलोमतः मान लीजिए $H$ कोई \omterm{def:b1:linmaps:forms}{अधिसमतल} है और $(e_1, \dots, e_{n-1})$ $H$ का ऐसा
\omterm{def:b1:vspaces:free}{आधार} जिसे $e_n$ से पूर्ण किया गया है: तब ``अंतिम \omterm{prop:b1:vspaces:coordinates}{निर्देशांक}'' वाले रूप की
\omterm{def:b1:linmaps:kerim}{अष्टि} $H$ है।

अनुपाती रूपों की \omterm{def:b1:linmaps:kerim}{अष्टि} एक ही होती है। विलोमतः मान लीजिए $\ker\varphi =
\ker\psi = H$ और $a \notin H$ चुनिए: तब हर $x$ $x = h + \lambda a$ लिखा जाता
है (क्योंकि $E = H \oplus Ka$), और
\[
\varphi(x) = \lambda \varphi(a), \qquad \psi(x) = \lambda\psi(a):
\]
अतः $\varphi = \frac{\varphi(a)}{\psi(a)}\,\psi$।
\end{proof}

\begin{example}\label{ex:b1:linmaps:hyperplane}
$K^n$ में \omterm{def:b1:linmaps:forms}{अधिसमतल} कोई \omterm{def:b1:logic:sets}{हल-समुच्चय} $\{a_1 x_1 + \dots + a_n x_n = 0\}$ है,
जहाँ $a_i$ सब शून्य न हों --- $\R^3$ में मूल बिंदु से होकर जाने वाले समतल का
वही परिचित समीकरण। फलन समष्टियों में मूल्यांकन रूप $P \mapsto P(1)$ अथवा $f
\mapsto \int_0^1 f$ $\R_n[X]$ के, $C(\intcc{0}{1})$ के \omterm{def:b1:linmaps:forms}{अधिसमतल} परिभाषित करते
हैं (\cref{exo:b1:findim:6} देखिए)।
\end{example}

\begin{example}[एक अधिसमतल, तीन तरीक़ों से]
\label{ex:b1:linmaps:hyperplanerun}
$\R^3$ पर $\varphi(x, y, z) = x - 2y + 3z$ लीजिए और $H = \ker\varphi$।
\emph{\omterm{def:b1:vspaces:free}{आधार}}: $x = 2y - 3z$ हल कीजिए:
\[
(2y - 3z,\ y,\ z) = y\,(2, 1, 0) + z\,(-3, 0, 1),
\]
दो \omterm{def:b1:vspaces:free}{स्वतंत्र} सदिश: $\dim H = 2$, अर्थात् एक \omterm{def:b1:linmaps:forms}{अधिसमतल}, जैसा
\cref{thm:b1:linmaps:hyperplanes} $\varphi \neq 0$ से बता देता है।
\emph{\omterm{def:b1:vspaces:sum}{पूरक} रेखा}: $H$ के बाहर का कोई भी सदिश एक रेखा जनित करता है, जैसे $a =
(1, 0, 0)$ ($\varphi(a) = 1 \neq 0$); और किसी भी $v$ का विघटन स्पष्ट है:
\[
v = \underbrace{\bigl(v - \varphi(v)\,a\bigr)}_{\in\,H}
+ \underbrace{\varphi(v)\,a}_{\in\,\operatorname{Vect}(a)},
\]
क्योंकि $\varphi\bigl(v - \varphi(v)a\bigr) = \varphi(v) -
\varphi(v)\varphi(a) = 0$। \emph{अनुपातिता}: यदि $\psi(x,y,z) = -2x + 4y -
6z$, तो $\psi = -2\varphi$ और दोनों की \omterm{def:b1:linmaps:kerim}{अष्टि} $H$ है; विलोमतः $H$ पर शून्य
होने वाला कोई भी रूप $\varphi$ का गुणज है (\cref{exo:b1:linmaps:8}) ---
अर्थात् \omterm{def:b1:linmaps:forms}{अधिसमतल} का समीकरण मापन तक अद्वितीय है, और यह तथ्य ज्यामिति में
समतलों के लिए निरंतर काम आता है।
\end{example}

\begin{remark}[सामान्य भूलें]
\label{rem:b1:linmaps:pitfalls}
\emph{\omterm{def:b1:linmaps:kerim}{अष्टि} और \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} अलग-अलग समष्टियों में रहते हैं}: $\ker u \subseteq
E$, $\operatorname{im} u \subseteq F$; योग $\ker u + \operatorname{im} u$ का
अर्थ केवल अंतःसमाकारिताओं के लिए बनता है, और तब भी उसका प्रत्यक्ष होना
आवश्यक नहीं ($u(x, y) = (y, 0)$ में $\ker u = \operatorname{im} u$ है;
\cref{exo:b1:linmaps:7} बताता है कि प्रत्यक्षता कब होती है)। \emph{$u^2 = 0$
का अर्थ $u = 0$ नहीं है}: वही $u(x,y) = (y, 0)$ शून्य हुए बिना वर्ग करने पर
शून्य हो जाता है --- $u^2 = 0$ वास्तव में यह कहता है कि $\operatorname{im} u
\subseteq \ker u$ (\cref{exo:b1:linmaps:5})। \emph{\omterm{def:b1:logic:inj}{एकैकी} $\iff$ \omterm{def:b1:logic:inj}{आच्छादक} के
लिए बराबर परिमित विमाएँ चाहिए}: $K[X]$ पर \omterm{def:b1:derivative:def}{अवकलज} \omterm{def:b1:logic:inj}{आच्छादक} है पर \omterm{def:b1:logic:inj}{एकैकी} नहीं, और
$P \mapsto XP$ \omterm{def:b1:logic:inj}{एकैकी} है पर \omterm{def:b1:logic:inj}{आच्छादक} नहीं (\cref{cor:b1:linmaps:samedim}); और
\emph{भिन्न} विमाओं वाली समष्टियों के बीच तो एक निहितार्थ बस असंभव ही है
($\operatorname{rk} u \leq \min(\dim E, \dim F)$)। \emph{\omterm{def:b1:linmaps:kerim}{प्रतिबिंब} निर्धारित
करना \omterm{def:b1:vspaces:free}{आधार} पर चलता है, किसी भी कुल पर नहीं}: $u(1, 0) = a$, $u(0, 1) = b$,
$u(1, 1) = c$ की माँग $u$ को अति-निर्धारित कर देती है, जब तक $c = a + b$ न
हो; \omterm{def:b1:linmaps:def}{रैखिक प्रतिचित्रण} \omterm{def:b1:vspaces:free}{आधार} पर \omterm{def:b1:vspaces:free}{स्वतंत्र} है, और बाक़ी हर जगह बँधा हुआ।
\emph{संयुक्त \omterm{def:b1:logic:map}{प्रतिचित्रण} \omterm{def:b1:linmaps:rank}{कोटि} को सुरक्षित नहीं रखता}: वह केवल गिर ही सकती
है, $\operatorname{rk}(vu) \leq \min(\operatorname{rk} u, \operatorname{rk}
v)$ (\cref{exo:b1:linmaps:4}), और हानि का यथार्थ माप
\cref{exo:b1:linmaps:12} में है।
\end{remark}

\begin{remark}[ये प्रतिचित्रण कहाँ जाते हैं]
\label{rem:b1:linmaps:whereused}
\omterm{def:b1:linmaps:def}{रैखिक प्रतिचित्रण} अब \emph{आव्यूह} बनने वाले हैं: \omterm{def:b1:vspaces:free}{आधार} नियत होते ही
\cref{ch:b1:matrices} हर $u \in \mathcal{L}(E, F)$ को एक आयताकार सारणी से
कूटबद्ध कर देता है, और संयुक्त \omterm{def:b1:logic:map}{प्रतिचित्रण} आव्यूह-गुणनफल बन जाता है --- फिर
कोटि--शून्यता \cref{ch:b1:det} में \omterm{def:b1:linmaps:def}{रैखिक} निकायों का सिद्धांत चलाती है।
\omterm{def:b1:linmaps:projection}{प्रक्षेप} \cref{ch:b1:euclid} में अपनी सबसे उपयोगी विशेष स्थिति में लौटते
हैं, अर्थात् \emph{लांबिक} \omterm{def:b1:linmaps:projection}{प्रक्षेप}, जहाँ \omterm{def:b1:linmaps:kerim}{अष्टि} को \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} के लंबवत चुना
जाता है। नीचे की सप्ताहांत समस्या प्रक्षेपक-बीजगणित को वहाँ तक ले जाती है
जहाँ तक प्रथम वर्ष के औज़ार पहुँचते हैं, अर्थात् फिटिंग की प्रमेयिका तक; और
स्नातक वर्ष~2 का खंड अनुरेख तथा अभिलक्षणिक-मान सिद्धांत के साथ और आगे जाता
है, जिसके लिए स्थायी उपसमष्टियों पर के प्रक्षेपक ही बुनियादी ईंटें हैं।
\end{remark}

\begin{remark}[पुस्तक 3 के भीतर के परिदृश्य: कोटि--शून्यता तीन बार और]
\label{rem:b1:linmaps:perspectives}
संरक्षण नियम $\dim E = \dim\ker u + \operatorname{rk} u$ इस खंड के समाप्त
होने से पहले तीन बार फिर से पढ़ा जाएगा। \cref{ch:b1:det} में वह हल-समुच्चयों
का आकार बन जाता है: $p$ अज्ञातों और \omterm{def:b1:linmaps:rank}{कोटि} $r$ वाले किसी संगत निकाय का
\omterm{def:b1:logic:sets}{हल-समुच्चय} $p - r$ विमा का होता है --- अर्थात् वेश बदली हुई अष्टि-विमा।
\cref{ch:b1:euclid} में वह लांबिक रूप से विभाजित होता है, $\dim F + \dim
F^\perp = \dim E$, और हर दूरी-संगणना को शक्ति देता है। \cref{ch:b1:multivar}
की सप्ताहांत समस्या में वह न्यूनतम वर्गों का मुनीम है: $n$ प्रेक्षण, $2$
आरोपित प्राचल, $n - 2$ अवशिष्ट विमाएँ, और पाइथागोरसीय सर्वसमिका $\norm b^2 =
\norm p^2 + \norm{b - p}^2$ कोटि--शून्यता की यूक्लिडीय छाया है। एक प्रमेय,
चार वेश।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:1}
कौन-से \omterm{def:b1:logic:map}{प्रतिचित्रण} \omterm{def:b1:linmaps:def}{रैखिक} हैं?
\begin{enumerate}
  \item $\R^2 \to \R^2$, $(x, y) \mapsto (x + y, x - 2y)$;
  \item $\R^2 \to \R$, $(x, y) \mapsto xy$;
  \item $\R[X] \to \R[X]$, $P \mapsto P' + XP$;
  \item $\mathcal{F}(\R,\R) \to \R$, $f \mapsto f(3)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:2}
मान लीजिए $u \colon \R^3 \to \R^3$, $(x,y,z) \mapsto (x + y - z,\; 2x + y +
z,\; 3x + 2y)$। $\ker u$ निर्धारित कीजिए (\omterm{def:b1:vspaces:free}{आधार}, विमा), $\operatorname{rk}
u$, तथा $\operatorname{im} u$ का एक \omterm{def:b1:vspaces:free}{आधार}। क्या $u$ \omterm{def:b1:logic:inj}{एकैकी} है? \omterm{def:b1:logic:inj}{आच्छादक}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:3}
मान लीजिए $u \colon \R_n[X] \to \R_n[X]$, $P \mapsto P - P'$। सिद्ध कीजिए कि
$u$ एक तुल्याकारिता है: एक बार $\ker u$ के द्वारा, और एक बार प्रतिलोम दिखाकर
\emph{($P + P' + P'' + \dots$ पर विचार कीजिए)}।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:4}
मान लीजिए $u \in \mathcal{L}(E, F)$ और $v \in \mathcal{L}(F, G)$, और
समष्टियाँ \omterm{def:b1:findim:def}{परिमित-विमीय} हैं। सिद्ध कीजिए:
\[
\operatorname{rk}(v \circ u) \leq
\min\bigl(\operatorname{rk} u,\ \operatorname{rk} v\bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:5}
मान लीजिए $u$ $E$ (\omterm{def:b1:findim:def}{परिमित-विमीय}) की ऐसी अंतःसमाकारिता है कि $u^2 = 0$। सिद्ध
कीजिए कि $\operatorname{im} u \subseteq \ker u$, अतः $\operatorname{rk} u
\leq \frac{\dim E}{2}$। $E = \R^2$ के लिए समानता वाला एक उदाहरण दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:6}
मान लीजिए $p, q$ $E$ के ऐसे \omterm{def:b1:linmaps:projection}{प्रक्षेप} हैं कि $p \circ q = q \circ p$। सिद्ध
कीजिए कि $p \circ q$ एक \omterm{def:b1:linmaps:projection}{प्रक्षेप} है, और
\[
\operatorname{im}(pq) = \operatorname{im} p \cap \operatorname{im}
q ,
\qquad
\ker (pq) = \ker p + \ker q .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:7}
मान लीजिए $u \in \mathcal{L}(E)$, और $E$ \omterm{def:b1:findim:def}{परिमित-विमीय} हैं। निम्नलिखित की
तुल्यता सिद्ध कीजिए:
\begin{enumerate}
  \item $E = \ker u \oplus \operatorname{im} u$;
  \item $\ker u = \ker u^2$;
  \item $\operatorname{im} u = \operatorname{im} u^2$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:8}
मान लीजिए $\varphi, \psi$ $E$ पर ऐसे \omterm{def:b1:linmaps:forms}{रैखिक रूप} हैं कि $\ker\varphi \subseteq
\ker\psi$। सिद्ध कीजिए कि किसी $\lambda \in K$ के लिए $\psi =
\lambda\varphi$ (अपभ्रष्ट स्थितियों समेत)।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:linmaps:9}
मान लीजिए $\dim E = n$ के साथ $u \in \mathcal{L}(E)$, और मान लीजिए $u^n = 0$
पर $u^{n-1} \neq 0$ (अर्थात् एक \emph{अधिकतम शून्यंभावी} अंतःसमाकारिता)।
$u^{n-1}(x) \neq 0$ वाला $x$ चुनिए; सिद्ध कीजिए कि $\bigl(x, u(x), \dots,
u^{n-1}(x)\bigr)$ $E$ का \omterm{def:b1:vspaces:free}{आधार} है। \emph{(किसी शून्य संयोजन पर $u$ की घातें
लगाइए, $u^{n-1}$ से आरंभ करते हुए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:linmaps:10}
मान लीजिए $f \circ f = -\mathrm{id}$ के साथ $f \in \mathcal{L}(\R^n)$।
\begin{enumerate}
  \item सिद्ध कीजिए कि $f$ एक तुल्याकारिता है और यह कि $\lambda \in \R$ के
  साथ कोई भी $x \neq 0$ $f(x) = \lambda x$ संतुष्ट नहीं करता।
  \item सिद्ध कीजिए कि $n$ सम है। \emph{संकेत: $x_1 \neq 0$ चुनिए; दिखाइए कि
  $\operatorname{Vect}(x_1, f(x_1))$ $f$ के अंतर्गत स्थायी एक समतल है; उसके
  बाहर $x_2$ चुनिए और यही दोहराइए, यह सिद्ध करते हुए कि $\bigl(x_1, f(x_1),
  x_2, f(x_2), \dots\bigr)$ \omterm{def:b1:vspaces:free}{स्वतंत्र} बना रहता है।}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:11}
मान लीजिए $u, v \in \mathcal{L}(E, F)$, और समष्टियाँ \omterm{def:b1:findim:def}{परिमित-विमीय} हैं।
दुतरफ़ा परिबंध सिद्ध कीजिए
\[
\abs{\operatorname{rk} u - \operatorname{rk} v}
\;\leq\; \operatorname{rk}(u + v)
\;\leq\; \operatorname{rk} u + \operatorname{rk} v .
\]
\emph{(\omterm{def:b1:reals:bounds}{ऊपरी परिबंध} के लिए $\operatorname{im}(u+v)$ की $\operatorname{im} u +
\operatorname{im} v$ से तुलना कीजिए; निचले के लिए \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} को चतुराई से
लगाइए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:linmaps:12}
(फ्रोबेनियस की असमिका) मान लीजिए $u \in \mathcal{L}(E, F)$, $w \in
\mathcal{L}(F, G)$ और $v \in \mathcal{L}(G, H)$, और सभी समष्टियाँ
\omterm{def:b1:findim:def}{परिमित-विमीय} हैं। यथार्थ सूत्र सिद्ध कीजिए
\[
\operatorname{rk}(v \circ w) = \operatorname{rk} w -
\dim\bigl(\ker v \cap \operatorname{im} w\bigr),
\]
और फ्रोबेनियस की असमिका निकालिए
\[
\operatorname{rk}(v \circ w) + \operatorname{rk}(w \circ u)
\;\leq\; \operatorname{rk} w + \operatorname{rk}(v \circ w \circ
u) .
\]
जाँचिए कि $w = \mathrm{id}_F$ वाली स्थिति सिल्वेस्टर की असमिका है, जो
\cref{exo:b1:matrices:10} में आव्यूह-रूप में सिद्ध की गई है।
\end{exercise}

\section{समस्या: प्रक्षेपक-कलन और फिटिंग की प्रमेयिका}

\begin{problem}\label{pb:b1:linmaps:1}
\omterm{def:b1:linmaps:projection}{प्रक्षेप} वही अंतःसमाकारिताएँ हैं जो \omterm{def:b1:vspaces:sum}{प्रत्यक्ष योग} पैदा करते हैं, और विलोमतः
भी: हर सर्वसमिका $E = F_1 \oplus \dots \oplus F_k$ चुपके से प्रक्षेपकों का
ऐसा कुल है जिनका योग तत्समक है। यह समस्या उसी शब्दकोश को विकसित करती है ---
एक प्रक्षेपक का बीजगणित, दो का, $k$ का --- फिर वही स्थिरीकरण वाले विचार किसी
भी अंतःसमाकारिता पर लगाकर \emph{फिटिंग की प्रमेयिका} सिद्ध करती है:
\omterm{def:b1:findim:def}{परिमित-विमीय} समष्टि की हर अंतःसमाकारिता एक शून्यंभावी भाग और एक
व्युत्क्रमणीय भाग में बँट जाती है। सर्वत्र $E$ विमा $n$ वाली $K$-सदिश समष्टि
है, और \emph{प्रक्षेपक} का अर्थ है $p^2 = p$ वाला $p \in \mathcal{L}(E)$
(\cref{thm:b1:linmaps:projchar})। \index{प्रक्षेप}\index{फिटिंग की
प्रमेयिका}\index{शून्यंभावी}

\medskip \noindent\textbf{भाग I --- एक प्रक्षेपक के इर्द-गिर्द का बीजगणित।}
मान लीजिए $p$ एक प्रक्षेपक है, $p \neq 0$, $p \neq \mathrm{id}$।
\begin{enumerate}
  \item दिखाइए कि $\mathrm{id} - p$ एक प्रक्षेपक है और
  $\operatorname{im}(\mathrm{id} - p)$ तथा $\ker(\mathrm{id} - p)$ पहचानिए।
  \item $(\lambda\,\mathrm{id} + \mu\,p)^2$ संगणित कीजिए और वे सभी युग्म
  $(\lambda, \mu) \in K^2$ निर्धारित कीजिए जिनके लिए $\lambda\,\mathrm{id} +
  \mu\,p$ एक प्रक्षेपक है।
  \item दिखाइए कि $\mathcal{L}(E)$ का समतल $\operatorname{Vect}(\mathrm{id},
  p)$ संयुक्त \omterm{def:b1:logic:map}{प्रतिचित्रण} के अंतर्गत स्थायी है, और यह कि हर \omterm{def:b1:poly:def}{बहुपद} $Q \in
  K[X]$ के लिए
        \[
        Q(p) = Q(0)\,\mathrm{id} + \bigl(Q(1) -
        Q(0)\bigr)\,p .
        \]
  \item निर्धारित कीजिए कि किन $(\lambda, \mu)$ के लिए \omterm{def:b1:logic:map}{प्रतिचित्रण}
  $\lambda\,\mathrm{id} + \mu\,p$ व्युत्क्रमणीय है, और उसका प्रतिलोम
  $\alpha\,\mathrm{id} + \beta\,p$ के रूप में दीजिए। उत्तर की व्याख्या
  $\operatorname{im} p$ पर और $\ker p$ पर $\lambda\,\mathrm{id} + \mu\,p$ की
  क्रिया के द्वारा कीजिए।
  \item मान लीजिए $p'$ \emph{उसी \omterm{def:b1:linmaps:kerim}{प्रतिबिंब}} $\operatorname{im} p' =
  \operatorname{im} p$ वाला दूसरा प्रक्षेपक है। दिखाइए कि $p\,p' = p'$ और
  $p'\,p = p$। ये सर्वसमिकाएँ उसी \omterm{def:b1:vspaces:subspace}{उपसमष्टि} पर भिन्न-भिन्न अष्टियों के अनुदिश
  किए गए प्रक्षेपों को संयुक्त करने के विषय में क्या कहती हैं?
\end{enumerate}

\medskip \noindent\textbf{भाग II --- दो प्रक्षेपक।} मान लीजिए $p, q$ $E$ के
प्रक्षेपक हैं; और मान लीजिए अभिलक्षण $2$ नहीं है ($K = \R, \C$ के लिए यह
सत्य है)।
\begin{enumerate}[resume]
  \item मान लीजिए $p + q$ एक प्रक्षेपक है। $(p + q)^2$ का प्रसार करके $pq +
  qp = 0$ दिखाइए; बाईं ओर से, फिर दाईं ओर से $p$ के साथ संयुक्त करके $pq =
  qp$ निकालिए, और $pq = qp = 0$ निष्कर्ष निकालिए।
  \item विलोमतः मान लीजिए $pq = qp = 0$। दिखाइए कि $p + q$ एक प्रक्षेपक है,
  और
        \[
        \operatorname{im}(p + q) = \operatorname{im} p \oplus
        \operatorname{im} q,
        \qquad
        \ker(p + q) = \ker p \cap \ker q .
        \]
  \item दिखाइए कि $p - q$ एक प्रक्षेपक है तभी जब $pq = qp = q$।
  \emph{(प्रश्न 6--7 को $\mathrm{id} - p$ और $q$ पर लगाइए।)}
  \item $pq = qp = q$ का ज्यामितीय अर्थ दिखाइए: यह तभी सत्य है जब
  $\operatorname{im} q \subseteq \operatorname{im} p$ और $\ker p \subseteq
  \ker q$। (तब $q \leq p$ लिखा जाता है: ``$q$ कम पर \omterm{def:b1:linmaps:projection}{प्रक्षेप} करता है, और
  अधिक के अनुदिश''।)
  \item अब मान लीजिए $p, q$ क्रमविनिमेय हैं। \cref{exo:b1:linmaps:6} से
  स्मरण कीजिए कि $pq$ $\ker p + \ker q$ के अनुदिश $\operatorname{im} p \cap
  \operatorname{im} q$ पर प्रक्षेपक है। दिखाइए कि $r = p + q - pq$ एक
  प्रक्षेपक है, और
        \[
        \operatorname{im} r = \operatorname{im} p +
        \operatorname{im} q,
        \qquad
        \ker r = \ker p \cap \ker q .
        \]
        \emph{($\mathrm{id} - r = (\mathrm{id} - p)(\mathrm{id} - q)$ पर
        विचार कीजिए।)}
\end{enumerate}

\medskip \noindent\textbf{भाग III --- तत्समक के विघटन।}
\begin{enumerate}[resume]
  \item मान लीजिए $E = F_1 \oplus \dots \oplus F_k$, और $x = x_1 + \dots +
  x_k$ के लिए (अद्वितीय विघटन, $x_i \in F_i$) $p_i(x) = x_i$ रखिए। दिखाइए कि
  प्रत्येक $p_i$ एक प्रक्षेपक है, कि $i \neq j$ के लिए $p_i p_j = 0$, और यह
  कि $p_1 + \dots + p_k = \mathrm{id}$; $\operatorname{im} p_i$ तथा $\ker
  p_i$ पहचानिए।
  \item विलोमतः मान लीजिए $p_1, \dots, p_k \in \mathcal{L}(E)$ $p_1 + \dots
  + p_k = \mathrm{id}$ संतुष्ट करते हैं और सभी $i \neq j$ के लिए $p_i p_j =
  0$। दिखाइए कि प्रत्येक $p_i$ एक प्रक्षेपक है और यह कि $E =
  \operatorname{im} p_1 \oplus \dots \oplus \operatorname{im} p_k$।
  \item $p + q = \mathrm{id}$ वाले दो प्रक्षेपक: दिखाइए कि $pq = qp = 0$
  स्वतः लागू हो जाता है।
  \item $p + q + r = \mathrm{id}$ वाले तीन प्रक्षेपक: दिखाइए कि $p + q$ एक
  प्रक्षेपक है, और प्रश्न 6 से निष्कर्ष निकालिए कि \emph{सभी} युग्मशः गुणनफल
  शून्य हो जाते हैं --- अतः $E = \operatorname{im} p \oplus
  \operatorname{im} q \oplus \operatorname{im} r$, और वह भी गुणनफलों पर बिना
  किसी परिकल्पना के।
  \item $p_1 + \dots + p_k = \mathrm{id}$ वाले $k$ प्रक्षेपकों के लिए: पहले
  दिखाइए कि किन्हीं भी उपसमष्टियों के लिए $\dim (F_1 + \dots + F_k) \leq
  \dim F_1 + \dots + \dim F_k$, और समानता तभी होती है जब योग प्रत्यक्ष हो;
  फिर $E = \operatorname{im} p_1 + \dots + \operatorname{im} p_k$ दिखाइए, और
  सिद्ध कीजिए कि \emph{यदि} इसके अतिरिक्त $\sum_i \operatorname{rk} p_i \leq
  n$, तो योग प्रत्यक्ष है और $i \neq j$ के लिए $p_i p_j = 0$।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- पुनरावृत्त अष्टियाँ: फिटिंग की
प्रमेयिका।} मान लीजिए $u \in \mathcal{L}(E)$, $\dim E = n$।
\begin{enumerate}[resume]
  \item हर $k \geq 0$ के लिए मान्य दोनों शृंखलाएँ दिखाइए:
        \[
        \ker u^k \subseteq \ker u^{k+1},
        \qquad
        \operatorname{im} u^{k+1} \subseteq \operatorname{im}
        u^k .
        \]
  \item दिखाइए कि यदि किसी $r$ के लिए $\ker u^{r} = \ker u^{r+1}$, तो सभी $k
  \geq r$ के लिए $\ker u^{k} = \ker u^{r}$; और प्रतिबिंबों के लिए तदनुरूप
  स्थिरीकरण कहिए तथा सिद्ध कीजिए।
  \item इससे निष्कर्ष निकालिए कि $\ker u^{r} = \ker u^{r+1}$ वाला कोई सबसे
  छोटा पूर्णांक $r$ है, कि $r \leq n$, और यह कि \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} भी उसी $r$ पर
  स्थिर हो जाते हैं।
  \item (फिटिंग की प्रमेयिका) सिद्ध कीजिए कि
        \[
        E \;=\; \ker u^{r} \,\oplus\, \operatorname{im} u^{r} .
        \]
  \item दिखाइए कि दोनों उपसमष्टियाँ $u$ के अंतर्गत स्थायी हैं, कि $\ker
  u^{r}$ पर $u$ का सीमन शून्यंभावी है, और यह कि $\operatorname{im} u^{r}$ पर
  $u$ का सीमन $\operatorname{im} u^{r}$ की तुल्याकारिता है: अर्थात् हर
  अंतःसमाकारिता किसी विहित \omterm{def:b1:vspaces:sum}{प्रत्यक्ष योग} पर ``शून्यंभावी जमा व्युत्क्रमणीय''
  होती है।
  \item मान लीजिए $\pi$ $\operatorname{im} u^{r}$ के अनुदिश $\ker u^{r}$ पर
  प्रक्षेपक है। दिखाइए कि $\pi \circ u = u \circ \pi$।
\end{enumerate}

\medskip \noindent\textbf{भाग V --- एक हल किया हुआ उदाहरण, और संश्लेषण।}
\begin{enumerate}[resume]
  \item $\R^3$ पर मान लीजिए $u(x, y, z) = (y, 0, z)$। $u^2$ और $u^3$ संगणित
  कीजिए, स्थिरीकरण सूचकांक $r$, उपसमष्टियाँ $\ker u^{r}$ तथा
  $\operatorname{im} u^{r}$, और फिटिंग प्रक्षेपक $\pi$ निर्धारित कीजिए, तथा
  सूत्रों पर सत्यापित कीजिए कि $\pi u = u\pi$ और यह कि $u$ एक गुणनखंड पर
  शून्यंभावी है और दूसरे पर \omterm{def:b1:logic:inj}{एकैकी आच्छादक}।
  \item तुल्यताएँ दिखाइए: $u$ शून्यंभावी $\iff$ $\ker u^{r} = E$ $\iff$ $\pi
  = \mathrm{id}$; और इससे निष्कर्ष निकालिए कि $n$-विमीय समष्टि की कोई भी
  शून्यंभावी अंतःसमाकारिता सदा $u^{n} = 0$ संतुष्ट करती है (शून्यंभाविता
  सूचकांक विमा से कभी अधिक नहीं होता)।
  \item (अद्वितीयता) मान लीजिए $E = A \oplus B$, जहाँ $A, B$ $u$ के अंतर्गत
  स्थायी हैं, सीमन $u|_A$ शून्यंभावी है और $u|_B$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक}। सिद्ध कीजिए
  $A = \ker u^{r}$ और $B = \operatorname{im} u^{r}$: अर्थात् फिटिंग विघटन
  अद्वितीय है।
  \item संश्लेषण, चार वाक्यों में: भाग III प्रत्यक्ष योगों और प्रक्षेपकों के
  कुलों के बीच कौन-सा शब्दकोश स्थापित करता है; प्रश्न 14 को गुणनफलों पर कोई
  परिकल्पना क्यों नहीं चाहिए थी जबकि प्रश्न 15 को \omterm{def:b1:linmaps:rank}{कोटि} की परिकल्पना चाहिए थी
  (और स्नातक वर्ष~2 का कौन-सा औज़ार, अर्थात् अनुरेख, उसे हटा देता है);
  फिटिंग की प्रमेयिका किस अर्थ में \cref{exo:b1:linmaps:7} का स्थिरीकृत रूप
  है; और स्नातक वर्ष~2 के खंड के अभिलक्षणिक-मान सिद्धांत में फिटिंग के दोनों
  गुणनखंड क्या बन जाते हैं। भाग IV में सिद्ध प्रमेय का नाम बताइए।
\end{enumerate}
\end{problem}
