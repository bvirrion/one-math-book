\chapter{तर्क, समुच्चय और प्रतिचित्रण}\label{ch:b1:logic}

अब तक उपपत्तियाँ ``सिद्ध करने'' के एक अनौपचारिक किंतु ईमानदार बोध के सहारे
लिखी जाती रही हैं। स्नातक गणित का यह पहला अध्याय खेल के नियमों को स्पष्ट कर
देता है: गणितीय \omterm{def:b1:logic:statement}{कथन} क्या है, संयोजक और परिमाणक कथनों को कैसे जोड़ते हैं,
उपपत्ति में कौन-सी चालें वैध हैं --- और फिर इसी नींव पर गणित की दो
सार्वभौमिक भाषाएँ खड़ी करता है: \omterm{def:b1:logic:sets}{समुच्चय} और \omterm{def:b1:logic:map}{प्रतिचित्रण}।

\section{कथन और संयोजक}

\begin{definition}[कथन, संयोजक]\label{def:b1:logic:statement}
\emph{कथन}\index{कथन} (या \emph{प्रतिज्ञप्ति}) ऐसा वाक्य है जो या तो सत्य
(स) है या असत्य (अ) --- ठीक इनमें से एक। कथनों $P$ और $Q$ से बनते हैं:
\begin{itemize}
  \item \emph{निषेध} $\lnot P$ (``$P$ नहीं''), जो ठीक तब सत्य है जब $P$
  असत्य हो;
  \item \emph{संयोजन} $P \land Q$ (``$P$ और $Q$''), जो ठीक तब सत्य है जब
  दोनों सत्य हों;
  \item \emph{वियोजन} $P \lor Q$ (``$P$ या $Q$''), जो ठीक तब सत्य है जब कम
  से कम एक सत्य हो (यह ``या'' समावेशी है);
  \item \emph{निहितार्थ} $P \implies Q$, जो ठीक तब असत्य है जब $P$ सत्य हो
  और $Q$ असत्य;
  \item \emph{तुल्यता} $P \iff Q$, जो ठीक तब सत्य है जब $P$ और $Q$ का
  सत्य-मान समान हो।
\end{itemize}
\end{definition}

\begin{remark}
$P \implies Q$ की सत्य सारणी पर एक क्षण ठहरना चाहिए: जब $P$ असत्य हो, तब $P
\implies Q$ \emph{सत्य} होता है, चाहे $Q$ कुछ भी हो। ``यदि $2 < 1$ तो $0 =
5$'' एक सत्य निहितार्थ है। निहितार्थ इस विषय में कुछ नहीं कहता कि उसकी
परिकल्पना विफल होने पर क्या होता है।
\end{remark}

\begin{proposition}[कथनों पर संगणना के नियम]\label{prop:b1:logic:rules}
सभी कथनों $P$, $Q$, $R$ के लिए:
\begin{enumerate}
  \item $\lnot(\lnot P) \iff P$;
  \item डी मॉर्गन के नियम\index{डी मॉर्गन के नियम}: $\lnot(P \land Q) \iff
  (\lnot P) \lor (\lnot Q)$ तथा $\lnot(P \lor Q) \iff (\lnot P) \land (\lnot
  Q)$;
  \item $(P \implies Q) \iff \bigl((\lnot P) \lor Q\bigr)$, अतः $\lnot(P
  \implies Q) \iff P \land (\lnot Q)$;
  \item प्रतिधनात्मकता\index{प्रतिधनात्मकता}: $(P \implies Q) \iff
  \bigl((\lnot Q) \implies (\lnot P)\bigr)$;
  \item $(P \iff Q) \iff \bigl((P \implies Q) \land (Q \implies
        P)\bigr)$;
  \item वितरणशीलता: $P \land (Q \lor R) \iff (P \land Q) \lor (P \land R)$
  तथा $P \lor (Q \land R) \iff (P \lor Q) \land (P \lor R)$।
\end{enumerate}
\end{proposition}

\begin{proof}
प्रत्येक तुल्यता सत्य सारणियों की तुलना से जाँची जाती है: $P$, $Q$, $R$ से
बने दो संयुक्त \omterm{def:b1:logic:statement}{कथन} ठीक तब तुल्य हैं जब वे (चार या आठ) प्रत्येक स्थिति में एक
ही सत्य-मान लेते हों। पहले डी मॉर्गन नियम के लिए एक सारणी पूरी लिखते हैं:
\begin{center}
\begin{tabular}{cc|c|c|cc|c}
$P$ & $Q$ & $P \land Q$ & $\lnot(P \land Q)$ & $\lnot P$ & $\lnot Q$ &
$(\lnot P) \lor (\lnot Q)$ \\ \hline स & स & स & अ & अ & अ & अ \\ स & अ & अ
& स & अ & स & स \\ अ & स & अ & स & स & अ & स \\ अ & अ & अ & स & स & स & स \\
\end{tabular}
\end{center}
स्तंभ $4$ और $7$ मेल खाते हैं, जिससे नियम सिद्ध हो जाता है। प्रतिधनात्मकता
के लिए एक शाब्दिक संक्षेप तेज़ पड़ता है: $P \implies Q$ ठीक उसी स्थिति में
असत्य है जिसमें ($P$ सत्य, $Q$ असत्य), और $(\lnot Q) \implies (\lnot P)$ ठीक
उसी स्थिति में असत्य है जिसमें ($\lnot Q$ सत्य, $\lnot P$ असत्य), अर्थात्
($Q$ असत्य, $P$ सत्य) --- वही एकमात्र स्थिति, अतः दोनों निहितार्थों की
सारणियाँ समान हैं। शेष नियम इसी प्रकार जाँचे जाते हैं; ध्यान दीजिए कि (3)
प्रत्येक निहितार्थ को वियोजन में बदल देता है, जिससे (2) यंत्रवत् निषेध का
नियम $\lnot(P \implies Q) \iff P \land (\lnot Q)$ दे देता है: किसी निहितार्थ
का खंडन करने के लिए ऐसी स्थिति दिखानी पड़ती है जिसमें परिकल्पना सत्य हो और
निष्कर्ष असत्य।
\end{proof}

\section{परिमाणक}

\begin{definition}[परिमाणक]\label{def:b1:logic:quantifiers}
मान लीजिए $P(x)$ \omterm{def:b1:logic:sets}{समुच्चय} $E$ के किसी अवयव $x$ का कोई गुणधर्म है।
\begin{itemize}
  \item $\forall x \in E,\ P(x)$ (``$E$ के सभी $x$ के लिए $P(x)$'') तब सत्य
  है जब $E$ का प्रत्येक अवयव $P$ को संतुष्ट करता हो;
  \item $\exists x \in E,\ P(x)$ (``$E$ में ऐसा $x$ है कि $P(x)$'') तब सत्य
  है जब $E$ का कम से कम एक अवयव $P$ को संतुष्ट करता हो।
\end{itemize}
``अद्वितीय रूप से एक ऐसा अवयव है'' के लिए $\exists!$ लिखा जाता है।
\end{definition}

\begin{proposition}[परिमाणकों का निषेध]\label{prop:b1:logic:negquant}
\[
\lnot\bigl(\forall x \in E,\ P(x)\bigr) \iff
\exists x \in E,\ \lnot P(x),
\qquad
\lnot\bigl(\exists x \in E,\ P(x)\bigr) \iff
\forall x \in E,\ \lnot P(x).
\]
\end{proposition}

\begin{proof}
पहली तुल्यता को दोनों दिशाओं में देखते हैं; दूसरी सममित है। यदि $\forall x
\in E,\ P(x)$ असत्य है, तो प्रत्येक अवयव $P$ को संतुष्ट नहीं करता: \omterm{def:b1:logic:sets}{समुच्चय}
$A = \{x \in E : \lnot P(x)\}$ रिक्त नहीं हो सकता, और उसका कोई भी अवयव
$\exists x \in E,\ \lnot P(x)$ का साक्षी है। विलोमतः, यदि कोई $x_0 \in E$
$\lnot P(x_0)$ को संतुष्ट करता है, तो $x_0$ एक प्रतिउदाहरण है और सार्वत्रिक
\omterm{def:b1:logic:statement}{कथन} विफल हो जाता है। दूसरे नियम के लिए: ``कोई $x$ $P$ को संतुष्ट नहीं करता''
का अर्थ है कि \omterm{def:b1:logic:sets}{समुच्चय} $\{x : P(x)\}$ रिक्त है, अर्थात् प्रत्येक $x$ अपने
पूरक $A$ में स्थित है। परिमाणकों की एक नेस्टेड श्रृंखला पर क्रमशः लगाने से
ये दोनों नियम \cref{ex:b1:logic:limit} की यांत्रिक प्रक्रिया देते हैं: निषेध
बाएँ से दाएँ चलता है, प्रत्येक $\forall$ को $\exists$ में और प्रत्येक
$\exists$ को $\forall$ में बदलता जाता है, और अंत में सबसे भीतरी विधेय का
निषेध करता है।
\end{proof}

\begin{example}[रोज़मर्रा के गणितीय वाक्यों का निषेध]\label{ex:b1:logic:negpractice}
मान लीजिए $f \colon \R \to \R$। वाक्य ``$f$ वर्धमान है'' इस प्रकार पढ़ा जाता
है
\[
\forall x \in \R,\ \forall y \in \R,\quad
x \leq y \implies f(x) \leq f(y) ,
\]
और \cref{prop:b1:logic:negquant} तथा नियम $\lnot(P \implies Q) \iff P \land
\lnot Q$ से इसका निषेध:
\[
\exists x \in \R,\ \exists y \in \R,\quad
x \leq y \ \text{ और }\ f(x) > f(y) :
\]
एक ही साक्षी युग्म पर्याप्त है। इसी प्रकार ``$f$ परिबद्ध है'' का अर्थ है
$\exists M \in \R,\ \forall x \in \R,\ \abs{f(x)} \leq M$, जिसका निषेध
\[
\forall M \in \R,\ \exists x \in \R,\quad \abs{f(x)} > M :
\]
\emph{कोई भी} परिबंध प्रस्तावित किया जाए, कोई-न-कोई बिंदु उससे आगे निकल जाता
है। सार: सही निषेध में ``नहीं'' कभी किसी परिमाणित खंड पर नहीं लगता --- वह एक
नया धनात्मक \omterm{def:b1:logic:statement}{कथन} होता है, जिसमें भूमिकाएँ बदल जाती हैं: जो साक्षी पहले मिलते
थे, अब उन्हें स्वयं प्रस्तुत करना पड़ता है।
\end{example}

\begin{example}[परिमाणकों का क्रम]\label{ex:b1:logic:order}
भिन्न परिमाणकों का क्रम महत्त्व रखता है:
\[
\forall x \in \R,\ \exists y \in \R,\ y > x
\quad\text{सत्य है (लीजिए } y = x+1\text{),}
\]
\[
\exists y \in \R,\ \forall x \in \R,\ y > x
\quad\text{असत्य है (कोई वास्तविक संख्या सभी वास्तविक संख्याओं से बड़ी नहीं होती)।}
\]
पहले \omterm{def:b1:logic:statement}{कथन} में $y$ का $x$ पर निर्भर होना संभव है; दूसरे में एक ही $y$ को सभी
$x$ के लिए काम करना होगा। दूसरी ओर, दो समान परिमाणक सदैव आपस में स्थान बदल
सकते हैं।
\end{example}

\begin{example}[तीन परिमाणकों वाली परिभाषा पढ़ना]\label{ex:b1:logic:limit}
वाक्य ``अनुक्रम $(u_n)$ का अभिसरण $\ell$ की ओर होता है'' \cref{ch:b1:seq}
में इस प्रकार लिखा जाएगा
\[
\forall \varepsilon > 0,\ \exists N \in \N,\ \forall n \geq N,\quad
\abs{u_n - \ell} \leq \varepsilon .
\]
\cref{prop:b1:logic:negquant} को तीन बार लगाने पर इसका निषेध है
\[
\exists \varepsilon > 0,\ \forall N \in \N,\ \exists n \geq N,\quad
\abs{u_n - \ell} > \varepsilon .
\]
ऐसे वाक्यों का, उनके अर्थ पर सोचे बिना, यंत्रवत् निषेध कर पाना एक वास्तविक
कौशल है: यह तार्किक काम को गणितीय काम से अलग कर देता है।
\end{example}

\section{उपपत्ति की विधियाँ}

\begin{method}[उपपत्ति के मानक प्रारूप]\label{met:b1:logic:proofs}
सिद्ध करने के लिए\dots
\begin{enumerate}
  \item \emph{निहितार्थ $P \implies Q$ को सीधे}: $P$ मान लीजिए, $Q$ निकालिए;
  \item \emph{प्रतिधनात्मकता द्वारा}: $\lnot Q$ मान लीजिए, $\lnot P$ निकालिए
  --- \cref{prop:b1:logic:rules}~(4) से वैध;
  \item \emph{विरोधाभास द्वारा}: \omterm{def:b1:logic:statement}{कथन} को असत्य मान लीजिए और विरोधाभास तक
  पहुँचिए;
  \item \emph{तुल्यता}: दोनों निहितार्थ अलग-अलग सिद्ध कीजिए (या ज्ञात
  तुल्यताओं की शृंखला बनाइए);
  \item \emph{``सभी के लिए'' वाला \omterm{def:b1:logic:statement}{कथन}}: $E$ में एक \emph{स्वेच्छ} $x$ चुनिए
  (``मान लीजिए $x \in E$'') और $P(x)$ सिद्ध कीजिए;
  \item \emph{``अस्तित्व है'' वाला \omterm{def:b1:logic:statement}{कथन}}: एक साक्षी प्रस्तुत कीजिए, या
  अस्तित्व अप्रत्यक्ष रूप से सिद्ध कीजिए;
  \item \emph{आगमन द्वारा}: \cref{thm:b1:logic:induction} देखिए।
\end{enumerate}
किसी सुचयनित किंतु स्वेच्छ अवयव के विषय में \omterm{def:b1:logic:statement}{कथन} सिद्ध करते समय उस अवयव को
कभी अतिरिक्त गुणधर्म मत दीजिए: ``मान लीजिए $x \in \R$'' के बाद ``चूँकि $x >
0$\dots'' ऋणात्मक $x$ के विषय में कुछ भी सिद्ध नहीं करता।
\end{method}

\begin{remark}[उपपत्तियों में सामान्य भूलें]\label{rem:b1:logic:pitfalls}
चार चिरपरिचित जाल, जिन्हें एक बार नाम देकर पहचान लेना उचित है।
\begin{enumerate}
  \item \emph{प्रतिधनात्मक के स्थान पर विलोम।} $Q \implies P$ $P \implies Q$
  के तुल्य \emph{नहीं} है; केवल $\lnot Q \implies \lnot P$ तुल्य है। ``यदि
  वर्षा होती है तो सड़क गीली होती है'' से गीली सड़क देखकर वर्षा का निष्कर्ष
  निकालने का अधिकार नहीं मिलता।
  \item \emph{तुल्यता को एक ही निहितार्थ से सिद्ध करना।} ``यदि और केवल यदि''
  का दावा दो प्रमेय है; घोषित कीजिए कि कौन-सी दिशा सिद्ध की जा रही है, और
  दोनों सिद्ध कीजिए। $\iff$ की शृंखलाएँ तभी वैध हैं जब \emph{प्रत्येक} कड़ी
  सचमुच उत्क्रमणीय हो --- उदाहरणतः समीकरण का वर्ग करना उत्क्रमणीय नहीं है।
  \item \emph{उलटी दिशा की उपपत्तियाँ।} अभीष्ट निष्कर्ष से आरंभ करके किसी
  सत्य \omterm{def:b1:logic:statement}{कथन} तक पहुँचना कुछ भी सिद्ध नहीं करता ($-1 = 1$ से वर्ग करके सत्य $1
  = 1$ निकल आता है)। कोई संगणना उलटी दिशा में \emph{खोजी} भले जाए, पर उसे
  सीधी दिशा में, या स्पष्ट तुल्यताओं के साथ, \emph{लिखना} पड़ता है।
  \item \emph{नियत साक्षी बनाम स्वेच्छ अवयव।} $\exists x,\ P(x)$ सिद्ध करने
  के लिए एक ही चतुराई से चुना हुआ $x$ प्रस्तुत किया जा सकता है; $\forall x,\
  P(x)$ सिद्ध करने के लिए चुना हुआ $x$ स्वेच्छ बना रहना चाहिए। दोनों को मिला
  देना --- सार्वत्रिक दावे को एक उदाहरण पर जाँच लेना --- आरंभिक
  उत्तर-पुस्तिकाओं की सबसे आम भूल है।
\end{enumerate}
\end{remark}

\begin{example}[प्रतिधनात्मकता और विरोधाभास काम पर]\label{ex:b1:logic:sqrt2}
\emph{$n \in \N$ के लिए: यदि $n^2$ सम है तो $n$ सम है।} प्रतिधनात्मकता
द्वारा: यदि $n$ विषम है, $n = 2k+1$, तो $n^2 = 4k^2 + 4k + 1$ विषम है।

\emph{$\sqrt 2$ अपरिमेय है।} विरोधाभास द्वारा: मान लीजिए $\sqrt 2 = p/q$,
जहाँ $p, q \in \N^*$ और भिन्न न्यूनतम पदों में है। तब $p^2 = 2q^2$ सम है,
अतः $p$ सम है (पिछला बिंदु), $p = 2r$; तब $q^2 = 2r^2$ सम है, अतः $q$ सम है
--- जो न्यूनतम पदों का विरोध करता है।
\end{example}

\begin{theorem}[आगमन]\label{thm:b1:logic:induction}
मान लीजिए $P(n)$ पूर्णांक $n$ का कोई गुणधर्म है। यदि
\begin{enumerate}
  \item $P(0)$ सत्य है, तथा
  \item सभी $n \in \N$ के लिए $P(n) \implies P(n+1)$,
\end{enumerate}
तो $P(n)$ सभी $n \in \N$ के लिए सत्य है।

\emph{प्रबल आगमन:} यदि (2) के स्थान पर यह रखा जाए तो निष्कर्ष अपरिवर्तित
रहता है: सभी $n$ के लिए $\bigl(P(0) \land \dots \land P(n)\bigr) \implies
P(n+1)$।
\end{theorem}

\begin{proof}
यह स्वयं $\N$ का गुणधर्म है, जो इसके तुल्य है: \emph{$\N$ के प्रत्येक अरिक्त
उपसमुच्चय में एक न्यूनतम अवयव होता है} (जिसे हम ज्ञात मान लेते हैं)।
वस्तुतः, मान लीजिए (1) और (2) सत्य हैं तथा $A = \{n \in \N : P(n) \text{
असत्य}\}$ है। यदि $A \neq \emptyset$, तो उसमें एक न्यूनतम अवयव $m$ है; (1)
से $m \neq 0$; तब $m - 1 \notin A$, अतः $P(m-1)$ सत्य है, और (2) से $P(m)$
मिलता है --- विरोधाभास। अतः $A = \emptyset$। प्रबल आगमन के लिए वही तर्क
लगाइए: $P(0), \dots, P(m-1)$ सभी सत्य हैं, क्योंकि $m$ $A$ का न्यूनतम अवयव
है।
\end{proof}

\begin{example}[अद्वितीय अस्तित्व सिद्ध करना]\label{ex:b1:logic:uniqueexistence}
\omterm{def:b1:logic:statement}{कथन} $\exists!\,x,\ P(x)$ वस्तुतः \emph{दो} \omterm{def:b1:logic:statement}{कथन} हैं, जिन्हें अलग-अलग सिद्ध
किया जाता है: अस्तित्व (कोई $x_0$ प्रस्तुत या निर्मित कीजिए जिसके लिए
$P(x_0)$) और अद्वितीयता ($P(x)$ तथा $P(x')$ मानकर $x = x'$ निकालिए)। नमूना:
\emph{ऐसा अद्वितीय वास्तविक $x$ है जिसके लिए $x^3 + x = 2$।} अस्तित्व: $x_0
= 1$ काम करता है, क्योंकि $1 + 1 = 2$। अद्वितीयता: यदि $x^3 + x = x'^3 +
x'$, तो
\[
0 = (x^3 - x'^3) + (x - x')
= (x - x')\,\bigl(x^2 + xx' + x'^2 + 1\bigr),
\]
और दूसरा गुणनखंड धनात्मक है (वह $\bigl(x + \tfrac{x'}2\bigr)^2 + \tfrac34
x'^2 + 1 \geq 1$ के बराबर है), अतः $x = x'$। श्रम-विभाजन पर ध्यान दीजिए:
अस्तित्व के लिए एक सौभाग्यपूर्ण अनुमान काम आया, जबकि अद्वितीयता के लिए
\emph{स्वेच्छ} हलों पर वैध बीजगणित। कोई भी तर्क दूसरे का काम नहीं करता, और
एक हल मिल जाने के बाद दूसरे आधे को भूल जाना सदा बना रहने वाला प्रलोभन है।
\end{example}

\begin{example}\label{ex:b1:logic:induction}
सभी $n \in \N^*$ के लिए: $\;\sum_{k=1}^n k = \frac{n(n+1)}{2}$। आधार स्थिति
$n = 1$: दोनों पक्ष $1$ के बराबर हैं। पद: $n$ के लिए सूत्र मानकर,
\[
\sum_{k=1}^{n+1} k = \frac{n(n+1)}{2} + (n+1)
= (n+1)\Bigl(\frac n2 + 1\Bigr) = \frac{(n+1)(n+2)}{2}. \qedhere
\]
\end{example}

\begin{example}[प्रबल आगमन काम पर]\label{ex:b1:logic:strongind}
\emph{प्रत्येक पूर्णांक $n \geq 2$ अभाज्य संख्याओं का गुणनफल है} (यहाँ
अभाज्य से तात्पर्य ऐसे पूर्णांक $\geq 2$ से है जिसके $\geq 1$ भाजक केवल $1$
और वह स्वयं हों; अभाज्य संख्याओं का अपने आप में अध्ययन \cref{ch:b1:arith}
में है)। साधारण आगमन यहाँ असहाय है: $95 = 5 \times 19$ का गुणनखंडन जानने से
$96$ के विषय में कुछ नहीं मिलता। प्रबल आगमन ठीक बैठता है। आधार स्थिति: $2$
अभाज्य है, अतः (एक गुणनखंड वाला) अभाज्यों का गुणनफल है। पद: मान लीजिए $n
\geq 2$ और मान लीजिए कि $2 \leq m \leq n$ वाला प्रत्येक पूर्णांक $m$
अभाज्यों का गुणनफल है। यदि $n + 1$ अभाज्य है, तो काम पूरा। अन्यथा $n + 1 =
ab$, जहाँ $2 \leq a, b \leq n$; प्रबल परिकल्पना से $a$ और $b$ दोनों अभाज्यों
के गुणनफल हैं, अतः $n + 1$ भी। सार: प्रबल आगमन ठीक तभी सही औज़ार है जब
$P(n+1)$ का ``कारण'' किसी अप्रत्याशित पूर्ववर्ती कोटि पर हो, न कि कोटि $n$
पर।
\end{example}

\section{समुच्चय}

\begin{definition}[समुच्चय पर संक्रियाएँ]\label{def:b1:logic:sets}
हम \emph{समुच्चय}\index{समुच्चय} की धारणा और सदस्यता संबंध $x \in E$ को
मूलभूत मान लेते हैं। किसी परिवेशी समुच्चय $E$ के भीतर समुच्चयों $A, B$ के
लिए:
\begin{itemize}
  \item \emph{अंतर्विष्टि}: $A \subseteq B$ तब जब $\forall x,\ x \in A
  \implies x \in B$; समता $A = B$ तब जब $A \subseteq B$ और $B \subseteq A$;
  \item \emph{सम्मिलन} $A \cup B$, \emph{प्रतिच्छेदन} $A \cap B$,
  \emph{अंतर} $A \setminus B = \{x \in A : x \notin B\}$, \emph{पूरक}
  $\overline{A} = E \setminus A$;
  \item \emph{रिक्त समुच्चय} $\emptyset$, जो प्रत्येक समुच्चय में अंतर्विष्ट
  है;
  \item \emph{घात समुच्चय}\index{घात समुच्चय} $\mathcal{P}(E)$: $E$ के सभी
  उपसमुच्चयों का समुच्चय;
  \item \emph{कार्तीय गुणन} $E \times F$: क्रमित युग्मों $(x, y)$ का
  समुच्चय, जहाँ $x \in E$, $y \in F$।
\end{itemize}
\end{definition}

\begin{example}[घात समुच्चय से परिचय]\label{ex:b1:logic:powerset}
$E = \{a, b\}$ के लिए:
\[
\mathcal P(E) = \bigl\{\, \emptyset,\ \{a\},\ \{b\},\ \{a, b\}
\,\bigr\},
\]
चार अवयव --- और प्रकार-अनुशासन पर ध्यान दीजिए: $a \in E$, किंतु $\{a\} \in
\mathcal P(E)$; \omterm{def:b1:logic:statement}{कथन} $a \in \mathcal P(E)$ और $\{a\} \subseteq \mathcal P(E)$
लिखे हुए रूप में असत्य हैं (दूसरे के लिए $a$ का $E$ का \emph{उपसमुच्चय} होना
आवश्यक होता)। शून्य से आरंभ करके दोहराने पर: $\mathcal P(\emptyset) =
\{\emptyset\}$ में एक अवयव है, $\mathcal P(\mathcal P(\emptyset)) =
\{\emptyset, \{\emptyset\}\}$ में दो, अगले में चार --- समुच्चयों के \omterm{def:b1:logic:sets}{समुच्चय}
भी साधारण \omterm{def:b1:logic:sets}{समुच्चय} ही हैं, और \cref{ch:b1:counting} इस दुगुना होने के
प्रतिरूप की पुष्टि करेगा: $\abs{\mathcal P(E)} = 2^{\abs E}$। स्तरों ($x$,
$\{x\}$, $\{\{x\}\}$) को अलग-अलग बनाए रखना
\cref{exo:b1:logic:11,exo:b1:logic:12} जैसे अभ्यासों की आधी लड़ाई है।
\end{example}

\begin{proposition}[समुच्चय बीजगणित]\label{prop:b1:logic:setalgebra}
$E$ के उपसमुच्चयों $A, B, C$ के लिए:
\begin{enumerate}
  \item $A \cap (B \cup C) = (A \cap B) \cup (A \cap C)$ तथा $A \cup (B \cap
  C) = (A \cup B) \cap (A \cup C)$;
  \item डी मॉर्गन: $\overline{A \cup B} = \overline{A} \cap \overline{B}$
  तथा $\overline{A \cap B} = \overline{A} \cup \overline{B}$;
  \item $A \subseteq B \iff \overline{B} \subseteq \overline{A}$.
\end{enumerate}
\end{proposition}

\begin{proof}
प्रत्येक सर्वसमिका \cref{prop:b1:logic:rules} के किसी नियम का अनुवाद है, इस
शब्दकोश के द्वारा: ($\in A$ या नहीं) $\leftrightarrow$ (\omterm{def:b1:logic:statement}{कथन} सत्य या असत्य);
उदाहरणार्थ $x \in \overline{A \cup B} \iff \lnot(x \in A \lor x \in B) \iff
(x \notin A) \land (x \notin B) \iff x \in \overline{A} \cap \overline{B}$।
बिंदु (3) प्रतिधनात्मकता है। दूसरे नमूने के रूप में, पहला वितरण नियम पूरा:
\[
x \in A \cap (B \cup C)
\iff (x \in A) \land \bigl(x \in B \lor x \in C\bigr)
\iff \bigl(x \in A \land x \in B\bigr) \lor
\bigl(x \in A \land x \in C\bigr),
\]
\cref{prop:b1:logic:rules}~(6) की वितरणशीलता से; और अंतिम \omterm{def:b1:logic:statement}{कथन} $x \in (A \cap
B) \cup (A \cap C)$ पढ़ा जाता है। इस प्रकार की प्रत्येक \omterm{def:b1:logic:sets}{समुच्चय} सर्वसमिका
इसी एक यांत्रिक अनुवाद से सिद्ध हो जाती है --- यही कारण है कि इनमें से किसी
को कंठस्थ करने की आवश्यकता नहीं।
\end{proof}

\begin{method}[समुच्चयों की समता सिद्ध करना]\label{met:b1:logic:doubleinc}
$A = B$ सिद्ध करने के लिए दोनों अंतर्विष्टियाँ सिद्ध कीजिए: मान लीजिए $x \in
A$, दिखाइए $x \in B$; फिर मान लीजिए $x \in B$, दिखाइए $x \in A$। वैकल्पिक
रूप से, जब प्रत्येक पद सचमुच तुल्यता हो, तब तुल्यताओं की शृंखला $x \in A
\iff \dots \iff x \in B$ बनाइए।
\end{method}

\section{प्रतिचित्रण}

\begin{definition}[प्रतिचित्रण, प्रतिबिंब, पूर्वप्रतिबिंब]\label{def:b1:logic:map}
\emph{प्रतिचित्रण}\index{प्रतिचित्रण} (या \emph{फलन}) $f \colon E \to F$
\omterm{def:b1:logic:sets}{समुच्चय} $E$ (\emph{प्रांत}) के प्रत्येक अवयव $x$ को \omterm{def:b1:logic:sets}{समुच्चय} $F$
(\emph{सहप्रांत}) का ठीक एक अवयव $f(x)$ सौंपता है। $A \subseteq E$ और $B
\subseteq F$ के लिए:
\[
f(A) = \{f(x) : x \in A\} \subseteq F,
\qquad
f^{-1}(B) = \{x \in E : f(x) \in B\} \subseteq E
\]
क्रमशः $A$ का \emph{अग्र प्रतिबिंब} और $B$ का \emph{पूर्वप्रतिबिंब}
\index{पूर्वप्रतिबिंब} कहलाते हैं। $f \colon E \to F$ और $g \colon F \to G$
का \emph{संयोजन} $g \circ f \colon E \to G$ है, $x \mapsto g(f(x))$।
\end{definition}

\begin{remark}
संकेतन $f^{-1}(B)$ किसी प्रतिलोम \omterm{def:b1:logic:map}{प्रतिचित्रण} को \emph{पूर्वमान्य नहीं} करता:
$f^{-1}(B)$ प्रत्येक $f$ के लिए परिभाषित है। \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} प्रतिबिंबों की
तुलना में अधिक सुव्यवहृत हैं: $f^{-1}$ सम्मिलन, प्रतिच्छेदन और पूरक को
सुरक्षित रखता है, जबकि $f(A \cap A') \subseteq f(A) \cap f(A')$ कठोर हो सकता
है (\cref{exo:b1:logic:8})।
\end{remark}

\begin{example}[प्रतिबिंब और पूर्वप्रतिबिंब की संगणना]\label{ex:b1:logic:images}
मान लीजिए $f \colon \R \to \R$, $x \mapsto x^2$। तब:
\[
f\bigl(\intcc{-1}{2}\bigr) = \intcc04, \qquad
f^{-1}\bigl(\intcc14\bigr) = \intcc{-2}{-1} \cup \intcc12, \qquad
f^{-1}(\{-1\}) = \emptyset .
\]
पहले के लिए: प्रत्येक $x \in \intcc{-1}2$ के लिए $x^2 \in \intcc04$, और
प्रत्येक $y \in \intcc04$ $y = (\sqrt y)^2$ के रूप में प्राप्त होता है, जहाँ
$\sqrt y \in \intcc02 \subseteq \intcc{-1}2$ --- ध्यान दीजिए कि प्रतिबिंब
$\intcc14 = \{(-1)^2, 2^2\}$ \emph{नहीं} है: अंतरालों के प्रतिबिंब केवल
सिरों से नहीं निकाले जाते। दूसरे के लिए: $1 \leq x^2 \leq 4 \iff 1 \leq \abs
x \leq 2$, जो दो टुकड़ों में बँट जाता है। तीसरा दिखाता है कि \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब}
रिक्त भी हो सकता है --- $f^{-1}(B)$ सदा अर्थपूर्ण है, चाहे $B$ का प्रतिबिंब
से प्रतिच्छेदन कितना ही छोटा हो। अंत में इसी उदाहरण पर ऊपर की टिप्पणी वाली
कठोरता की परिघटना देखिए: $A = \intcc{-1}0$ और $A' = \intcc01$ लेने पर $f(A
\cap A') = f(\{0\}) = \{0\}$ मिलता है, जबकि $f(A) \cap f(A') = \intcc01$।
\end{example}

\begin{definition}[एकैकी, आच्छादक, एकैकी आच्छादक]\label{def:b1:logic:inj}
\omterm{def:b1:logic:map}{प्रतिचित्रण} $f \colon E \to F$ है:
\begin{itemize}
  \item \emph{एकैकी}\index{एकैकी} तब, जब भिन्न अवयवों के प्रतिबिंब भिन्न
  हों: $\forall x, x' \in E,\ f(x) = f(x') \implies x = x'$;
  \item \emph{आच्छादक}\index{आच्छादक} तब, जब $F$ का प्रत्येक अवयव प्राप्त
  होता हो: $\forall y \in F,\ \exists x \in E,\ f(x) = y$;
  \item \emph{एकैकी आच्छादक}\index{एकैकी आच्छादक} तब, जब वह दोनों हो,
  अर्थात् प्रत्येक $y \in F$ का ठीक एक \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} हो।
\end{itemize}
\end{definition}

\begin{theorem}[प्रतिलोम प्रतिचित्रण]\label{thm:b1:logic:inverse}
\omterm{def:b1:logic:map}{प्रतिचित्रण} $f \colon E \to F$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है यदि और केवल यदि ऐसा
\omterm{def:b1:logic:map}{प्रतिचित्रण} $g \colon F \to E$ हो जिसके लिए $g \circ f = \mathrm{id}_E$ और
$f \circ g = \mathrm{id}_F$। उस स्थिति में $g$ अद्वितीय होता है; उसे
$f^{-1}$ लिखा जाता है और $f$ का \emph{प्रतिलोम} कहा जाता है, और $f^{-1}$
स्वयं \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है, जिसके लिए $(f^{-1})^{-1} = f$।
\end{theorem}

\begin{proof}
($\Rightarrow$) यदि $f$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है, तो प्रत्येक $y \in F$ का अद्वितीय
\omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} है; $g(y)$ को वही \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} परिभाषित कीजिए। तब रचना से
$f(g(y)) = y$, और $g(f(x)) = x$, क्योंकि $x$ $f(x)$ का \emph{एकमात्र}
\omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} है।

($\Leftarrow$) मान लीजिए ऐसा $g$ विद्यमान है। यदि $f(x) = f(x')$, तो $g$
लगाने पर $x = x'$: अर्थात् $f$ \omterm{def:b1:logic:inj}{एकैकी} है। $y \in F$ के लिए $x = g(y)$ $f(x) =
y$ को संतुष्ट करता है: अर्थात् $f$ \omterm{def:b1:logic:inj}{आच्छादक} है।

अद्वितीयता: यदि $g$ और $h$ दोनों काम करते हैं, तो $g = g \circ \mathrm{id}_F
= g \circ (f \circ h) = (g \circ f) \circ h = h$। अंत में, सर्वसमिकाओं का यह
युग्म $f$ और $g$ में सममित है, अतः $g = f^{-1}$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है और उसका
प्रतिलोम $f$ है।
\end{proof}

\begin{example}[व्यवहार में प्रतिलोम की संगणना]\label{ex:b1:logic:inversecomp}
मान लीजिए $f \colon \R \to \intoo0{+\infty}$, $f(x) = \eu^{2x+1}$। प्रतिलोम
निकालने के लिए दिए हुए $y > 0$ के लिए $y = f(x)$ हल कीजिए:
\[
y = \eu^{2x+1} \iff \ln y = 2x + 1 \iff x = \frac{\ln y - 1}2 ,
\]
घोषित प्रांतों पर हर पद उत्क्रमणीय है। यह संगणना एक साथ सब कुछ दे देती है:
सहप्रांत के प्रत्येक $y$ के लिए ठीक एक हल $x$ है, अतः $f$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है,
और
\[
f^{-1} \colon \intoo0{+\infty} \to \R,
\qquad
f^{-1}(y) = \frac{\ln y - 1}2 .
\]
दोनों संयोजनों की त्वरित जाँच ($f^{-1}(f(x)) = \frac{(2x+1) - 1}2 = x$ और
$f(f^{-1}(y)) = \eu^{\ln y} = y$) \cref{thm:b1:logic:inverse} की कसौटी की
पुष्टि करती है। सार: ``$x$ के लिए हल कीजिए और तुल्यताओं पर दृष्टि रखिए'' एक
ही साथ अस्तित्व की उपपत्ति, अद्वितीयता की उपपत्ति और सूत्र है --- पर यह तभी
काम करता है जब सहप्रांत सही घोषित किया गया हो ($f$ $\R$ पर \omterm{def:b1:logic:inj}{आच्छादक}
\emph{नहीं} है)।
\end{example}

\begin{proposition}[संयोजन और तीनों गुणधर्म]\label{prop:b1:logic:comp}
मान लीजिए $f \colon E \to F$ और $g \colon F \to G$।
\begin{enumerate}
  \item यदि $f$ और $g$ \omterm{def:b1:logic:inj}{एकैकी} (क्रमशः \omterm{def:b1:logic:inj}{आच्छादक}, \omterm{def:b1:logic:inj}{एकैकी आच्छादक}) हैं, तो $g
  \circ f$ भी वैसा ही है; और \omterm{def:b1:logic:inj}{एकैकी आच्छादक} स्थिति में $(g \circ f)^{-1} =
  f^{-1} \circ g^{-1}$।
  \item यदि $g \circ f$ \omterm{def:b1:logic:inj}{एकैकी} है, तो $f$ \omterm{def:b1:logic:inj}{एकैकी} है। यदि $g \circ f$ \omterm{def:b1:logic:inj}{आच्छादक}
  है, तो $g$ \omterm{def:b1:logic:inj}{आच्छादक} है।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) यदि $g(f(x)) = g(f(x'))$, तो $g$ की एकैकीयता से $f(x) = f(x')$ मिलता है,
फिर $f$ की एकैकीयता से $x = x'$। यदि $z \in G$, तो $g$ की आच्छादकता से ऐसा
$y$ मिलता है कि $g(y) = z$, फिर $f$ की आच्छादकता से ऐसा $x$ कि $f(x) = y$,
अतः $g(f(x)) = z$। \omterm{def:b1:logic:inj}{एकैकी आच्छादक} स्थिति में सीधे जाँच लिया जाता है कि
$f^{-1} \circ g^{-1}$ $g \circ f$ का द्विपक्षीय प्रतिलोम है, और
\cref{thm:b1:logic:inverse} की अद्वितीयता निष्कर्ष दे देती है।

(2) यदि $f(x) = f(x')$, तो $g(f(x)) = g(f(x'))$, और $g \circ f$ की एकैकीयता
से $x = x'$। यदि $z \in G$, तो $g \circ f$ की आच्छादकता से ऐसा $x$ मिलता है
कि $g(f(x)) = z$: तब $y = f(x)$ $g(y) = z$ को संतुष्ट करता है।
\end{proof}

\begin{example}[बिंदु (2) तीक्ष्ण है]\label{ex:b1:logic:compsharp}
\cref{prop:b1:logic:comp}~(2) में निष्कर्षों को और आगे नहीं बढ़ाया जा सकता:
$g \circ f$ का \omterm{def:b1:logic:inj}{एकैकी आच्छादक} होना $f$ को \omterm{def:b1:logic:inj}{आच्छादक} या $g$ को \omterm{def:b1:logic:inj}{एकैकी} होने के लिए
बाध्य \emph{नहीं} करता। लीजिए $E = G = \{1\}$, $F = \{1, 2\}$, जिनमें $f(1)
= 1$ और $g(1) = g(2) = 1$: तब $g \circ f = \mathrm{id}_E$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है,
फिर भी $f$ अवयव $2$ तक नहीं पहुँचता और $g$ दोनों अवयवों को एक में मिला देता
है। इसका उपदेश एक सुनिश्चित लेखा-नियम है: संयोजन की सूचना एकैकीयता के लिए
\emph{भीतरी} \omterm{def:b1:logic:map}{प्रतिचित्रण} की ओर बहती है और आच्छादकता के लिए \emph{बाहरी}
\omterm{def:b1:logic:map}{प्रतिचित्रण} की ओर, कभी उलटी दिशा में नहीं। (\cref{exo:b1:logic:9} इसी
परिघटना को अनंत समुच्चयों के साथ बनाता है, जहाँ यही एकपक्षीय प्रतिलोमों का
इंजन है।)
\end{example}

\begin{example}\label{ex:b1:logic:injexamples}
$f \colon \R \to \R$, $x \mapsto x^2$ न तो \omterm{def:b1:logic:inj}{एकैकी} है ($f(-1) = f(1)$) और न
\omterm{def:b1:logic:inj}{आच्छादक} ($-1$ का कोई \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} नहीं)। प्रांत और सहप्रांत को सीमित करने
पर $f \colon \R_+ \to \R_+$, $x \mapsto x^2$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है, जिसका
प्रतिलोम $y \mapsto \sqrt y$ है। किसी \omterm{def:b1:logic:map}{प्रतिचित्रण} की एकैकीयता या आच्छादकता
घोषित प्रांत और सहप्रांत पर निर्भर करती है, केवल सूत्र पर नहीं।
\end{example}

\section{संबंध}

\begin{definition}[तुल्यता संबंध]\label{def:b1:logic:equiv}
\omterm{def:b1:logic:sets}{समुच्चय} $E$ पर \emph{द्विआधारी संबंध} $\mathcal{R}$ \emph{तुल्यता
संबंध}\index{तुल्यता संबंध} तब कहलाता है जब वह \emph{स्वतुल्य} (सभी $x$ के
लिए $x \mathbin{\mathcal{R}} x$), \emph{सममित} ($x \mathbin{\mathcal{R}} y
\implies y \mathbin{\mathcal{R}} x$) और \emph{संक्रमक} ($x
\mathbin{\mathcal{R}} y$ तथा $y \mathbin{\mathcal{R}} z$ से $x
\mathbin{\mathcal{R}} z$) हो। $x$ का \emph{तुल्यता वर्ग} $\mathrm{cl}(x) =
\{y \in E : x \mathbin{\mathcal{R}} y\}$ है।
\end{definition}

\begin{example}[तीनों अभिगृहीतों की जाँच]\label{ex:b1:logic:fracpart}
$\R$ पर $x \mathbin{\mathcal{R}} y$ तब घोषित कीजिए जब $x - y \in \Z$।
\emph{स्वतुल्य:} $x - x = 0 \in \Z$। \emph{सममित:} यदि $x - y \in \Z$, तो $y
- x = -(x - y) \in \Z$। \emph{संक्रमक:} यदि $x - y \in \Z$ और $y - z \in
\Z$, तो $x - z = (x - y) + (y - z) \in \Z$ (पूर्णांकों का योग)। अतः
$\mathcal R$ एक \omterm{def:b1:logic:equiv}{तुल्यता संबंध} है, और $\mathrm{cl}(x) = x + \Z = \{x + k : k
\in \Z\}$: प्रत्येक वर्ग में $\intco01$ का ठीक एक प्रतिनिधि होता है, अर्थात्
उसका \emph{भिन्नात्मक भाग}। इसके विपरीत, $\R$ पर संबंध ``$\abs{x - y} \leq
1$'' स्वतुल्य और सममित है पर संक्रमक \emph{नहीं} ($0 \mathbin{\mathcal R} 1$
और $1 \mathbin{\mathcal R} 2$, फिर भी $\abs{0 - 2} > 1$): निकटता का संचार
नहीं होता, और वर्गों में कोई \omterm{thm:b1:logic:partition}{विभाजन} नहीं बनता --- अभिगृहीतों की जाँच जब
यंत्रवत् लगने लगे, तब स्मरण रखने योग्य एक उपयोगी प्रतिउदाहरण।
\end{example}

\begin{theorem}[वर्ग विभाजन बनाते हैं]\label{thm:b1:logic:partition}
मान लीजिए $\mathcal{R}$ $E$ पर एक \omterm{def:b1:logic:equiv}{तुल्यता संबंध} है। तब \omterm{def:b1:logic:equiv}{तुल्यता वर्ग} अरिक्त
हैं, जोड़ों में असंयुक्त या समान हैं, और उनका सम्मिलन $E$ है: वे $E$ का एक
\emph{विभाजन}\index{विभाजन} बनाते हैं। विलोमतः, $E$ का प्रत्येक \omterm{thm:b1:logic:partition}{विभाजन} ठीक
एक \omterm{def:b1:logic:equiv}{तुल्यता संबंध} (``एक ही टुकड़े में होना'') से इसी प्रकार उत्पन्न होता है।
\end{theorem}

\begin{proof}
स्वतुल्यता से $x \in \mathrm{cl}(x)$, अतः वर्ग अरिक्त हैं और उनका सम्मिलन
$E$ है। मान लीजिए $\mathrm{cl}(x) \cap \mathrm{cl}(y) \neq \emptyset$, और
$z$ दोनों में स्थित है। तब $x \mathbin{\mathcal{R}} z$ और $y
\mathbin{\mathcal{R}} z$, अतः सममिति और संक्रमकता से $x
\mathbin{\mathcal{R}} y$। अब किसी भी $t \in \mathrm{cl}(y)$ के लिए संक्रमकता
से $t \in \mathrm{cl}(x)$ मिलता है, और सममित रूप से भी: दोनों वर्ग समान हैं।
विलोम के लिए, मान लीजिए $(E_i)_{i \in I}$ $E$ का एक \omterm{thm:b1:logic:partition}{विभाजन} है और $x
\mathbin{\mathcal S} y$ को ``कोई टुकड़ा $x$ और $y$ दोनों को समेटता है'' के
अर्थ में परिभाषित कीजिए। \emph{स्वतुल्य:} $x$ किसी टुकड़े में है, जो तब $x$
को दो बार समेटता है। \emph{सममित:} परिभाषा की शर्त $x$ और $y$ में सममित है।
\emph{संक्रमक:} यदि $x, y \in E_i$ और $y, z \in E_j$, तो $y \in E_i \cap
E_j$, अतः $E_i = E_j$ (भिन्न टुकड़े असंयुक्त होते हैं) और $x, z$ एक ही
टुकड़े में हैं। $x$ का $\mathcal S$-वर्ग ठीक वही टुकड़ा है जिसमें $x$ है,
अतः वर्ग वही दिए हुए टुकड़े हैं। अंत में, संबंध अपने वर्गों से निर्धारित हो
जाता है: समान वर्गों वाले दो \omterm{def:b1:logic:equiv}{तुल्यता संबंध} उन्हीं युग्मों को संबंधित करते
हैं, क्योंकि प्रत्येक $x$ और $y$ को ठीक तब संबंधित करता है जब $y$ $x$ के
वर्ग का सदस्य हो --- यहीं से अद्वितीयता का दावा निकलता है।
\end{proof}

\begin{example}\label{ex:b1:logic:congruence}
$\Z$ पर $n$ के सापेक्ष सर्वांगसमता ($x \equiv y \pmod n$ तब जब $n$ $x - y$
को विभाजित करता है) एक \omterm{def:b1:logic:equiv}{तुल्यता संबंध} है; उसके वर्ग $n$ से भाग देने पर एक दिए
हुए शेषफल वाले पूर्णांकों के $n$ \omterm{def:b1:logic:sets}{समुच्चय} हैं। यही उदाहरण
\cref{ch:b1:structures} में वलय $\Z/n\Z$ बन जाता है।
\end{example}

\begin{definition}[क्रम संबंध]\label{def:b1:logic:order}
$E$ पर संबंध $\preceq$ \emph{क्रम}\index{क्रम संबंध} तब कहलाता है जब वह
स्वतुल्य, \emph{प्रतिसममित} ($x \preceq y$ और $y \preceq x$ से $x = y$) और
संक्रमक हो। क्रम \emph{पूर्ण} तब कहलाता है जब कोई भी दो अवयव तुलनीय हों,
अन्यथा \emph{आंशिक}। अवयव $M \in A \subseteq E$ $A$ का \emph{महत्तम अवयव} तब
कहलाता है जब सभी $a \in A$ के लिए $a \preceq M$; महत्तम (और न्यूनतम) अवयव,
जब विद्यमान हों, अद्वितीय होते हैं।
\end{definition}

\begin{example}\label{ex:b1:logic:orders}
$(\R, \leq)$ पूर्णतः क्रमित है। $(\mathcal{P}(E), \subseteq)$ तभी आंशिक रूप
से क्रमित हो जाता है जब $E$ में दो अवयव हों: $\{a\}$ और $\{b\}$ तुलनीय नहीं
हैं। $\mathcal{P}(\{a,b\})$ के उपसमुच्चय $A = \{\{a\}, \{b\}\}$ का कोई
महत्तम अवयव नहीं है, फिर भी उसका एक ऊपरी परिबंध $\{a, b\}$ है: महत्तम अवयवों
और ऊपरी परिबंधों का यह भेद $\R$ के लिए \cref{ch:b1:reals} में लौटता है।
\end{example}

\begin{example}[जाली $\N^2$ पर दो क्रम]\label{ex:b1:logic:lexorder}
प्राकृत संख्याओं के युग्मों की घटक-दर-घटक तुलना कीजिए: $(a, b) \preceq (a',
b')$ तब जब $a \leq a'$ \emph{और} $b \leq b'$ (\emph{गुणन क्रम})। यह एक क्रम
है --- प्रत्येक अभिगृहीत निर्देशांक-दर-निर्देशांक विरासत में मिलता है --- पर
आंशिक: $(1, 3)$ और $(2, 0)$ अतुलनीय हैं। अब शब्दकोश की तरह तुलना कीजिए: $(a,
b) \preceq_{\mathrm{lex}} (a', b')$ तब जब $a < a'$, या $a = a'$ और $b \leq
b'$ (\emph{कोशीय क्रम})। संक्रमकता के लिए दो स्थितियों की जाँच चाहिए पर वह
सत्य है, और अब कोई भी दो युग्म तुलनीय हैं: यह क्रम पूर्ण है। दोनों क्रम एक
ही \omterm{def:b1:logic:sets}{समुच्चय} को भिन्न प्रकार से क्रमित करते हैं --- $(0, 100)
\preceq_{\mathrm{lex}} (1, 0)$, यद्यपि गुणन क्रम कुछ नहीं कहता --- यह स्मरण
दिलाता हुआ कि क्रम एक ऐसी संरचना है जिसे हम \emph{चुनते} हैं, \omterm{def:b1:logic:sets}{समुच्चय} का
गुणधर्म नहीं। कोशीय तुलना कई छँटाई-कसौटियों को एक में बदलने की मानक युक्ति
भी है।
\end{example}

\begin{remark}[मध्यांतर: आकार माने एकैकी आच्छादन]\label{rem:b1:logic:interlude}
इस अध्याय का एक मौन विषय प्रकाश में लाने योग्य है: \omterm{def:b1:logic:inj}{एकैकी आच्छादक} \omterm{def:b1:logic:map}{प्रतिचित्रण}
ही गणितज्ञ की ``समान आकार'' की धारणा हैं। परिमित समुच्चयों के लिए यह
\cref{ch:b1:counting} की गणना-कलन बन जाती है, जहाँ हर सूत्र गुप्त रूप से एक
एकैकी आच्छादन है; अनंत समुच्चयों के लिए यह नीचे दी गई सप्ताहांत समस्या बन
जाती है, जहाँ $\N$, $\Q$ और $\R$ के आकार सचमुच भिन्न निकलते हैं। यही शब्दकोश
इस खंड में दो बार और, परिष्कृत रूपों में, लौटता है: अनुक्रम
(\cref{ch:b1:seq}) और कुछ नहीं, \omterm{def:b1:logic:map}{प्रतिचित्रण} $\N \to \R$ ही हैं, अतः
अनुक्रमों के विषय में \omterm{def:b1:logic:statement}{कथन} प्रतिचित्रणों के एक \omterm{def:b1:logic:sets}{समुच्चय} के विषय में \omterm{def:b1:logic:statement}{कथन} हैं;
और रैखिक बीजगणित सदिश समष्टियों को एकैकी आच्छादनों से नहीं, बल्कि
\emph{रैखिक} एकैकी आच्छादनों से नापेगा, जिनका अस्तित्व एक ही संख्या --- विमा
(\cref{ch:b1:findim}) --- से नियंत्रित होता है। जब भी कोई नया ``समानपन''
प्रकट होता है --- समशक्तता, समूहों की तुल्याकारिता
(\cref{ch:b1:structures}), रैखिक तुल्याकारिता ---
\cref{thm:b1:logic:inverse} का प्रतिरूप दुहराता है: समानपन एक उत्क्रमणीय,
संरचना-संरक्षी \omterm{def:b1:logic:map}{प्रतिचित्रण} है।
\end{remark}

\begin{remark}[यह अध्याय कहाँ काम आता है]\label{rem:b1:logic:whereused}
सर्वत्र --- पर कुछ स्थान विशेष रूप से चिह्नित करने योग्य हैं।
\cref{ex:b1:logic:limit} का तीन-परिमाणक व्यायाम
\cref{ch:b1:seq,ch:b1:continuity} की रोज़ की रोटी है: हर सीमा की उपपत्ति एक
स्वेच्छ $\varepsilon$ के विरुद्ध खेला जाने वाला खेल है। \omterm{def:b1:logic:equiv}{तुल्यता वर्ग}
\cref{ch:b1:structures} में $\Z/n\Z$ के सर्वांगसमता वर्गों के रूप में लौटते
हैं, जहाँ \cref{thm:b1:logic:partition} का \omterm{thm:b1:logic:partition}{विभाजन} अपनी एक बीजीय संरचना पा
लेता है। \omterm{def:b1:logic:order}{क्रम संबंध}, ऊपरी परिबंध और लघुतम ऊपरी परिबंध \cref{ch:b1:reals} में
$\R$ के अभिगृहीतीय हृदय बन जाते हैं। \omterm{def:b1:logic:inj}{एकैकी}, \omterm{def:b1:logic:inj}{आच्छादक} और \omterm{def:b1:logic:inj}{एकैकी आच्छादक}
\omterm{def:b1:logic:map}{प्रतिचित्रण} \cref{ch:b1:linmaps} के रैखिक प्रतिचित्रणों के रूप में लौटते
हैं, जहाँ एकैकीयता की जाँच एक ही सदिश (अष्टि) पर की जा सकती है; और नीचे दी
गई सप्ताहांत समस्या एकैकी आच्छादन की सादी धारणा को \emph{अनंत समुच्चयों के
आकारों} के सिद्धांत में बदल देती है, जिसके निष्कर्ष ($\Q$ की गणनीयता, $\R$
की अगणनीयता) \cref{ch:b1:reals,ch:b1:topology} में फिर उभरते हैं।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:logic:1}
प्रत्येक \omterm{def:b1:logic:statement}{कथन} का निषेध ``नहीं'' शब्द का प्रयोग किए बिना लिखिए:
\begin{enumerate}
  \item $\forall x \in \R,\ \exists y \in \R,\ x + y > 0$;
  \item $\exists x \in \R,\ \forall y \in \R,\ xy = 0$;
  \item $\forall \varepsilon > 0,\ \exists \delta > 0,\ \forall x \in
        \R,\ \abs{x} \leq \delta \implies \abs{f(x)} \leq \varepsilon$
(किसी नियत \omterm{def:b1:logic:map}{प्रतिचित्रण} $f \colon \R \to \R$ के लिए)।
\end{enumerate}
फिर तय कीजिए कि \omterm{def:b1:logic:statement}{कथन} (1) और (2) सत्य हैं या नहीं।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:logic:2}
मान लीजिए $P, Q$ \omterm{def:b1:logic:statement}{कथन} हैं। सत्य सारणियों का प्रयोग करके सिद्ध कीजिए कि
$\lnot(P \implies Q) \iff P \land (\lnot Q)$, और उससे इसका निषेध निकालिए:
``यदि कोई फलन अवकलनीय है तो वह संतत है''।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:logic:3}
प्रतिधनात्मकता द्वारा सिद्ध कीजिए: $x \in \R$ के लिए, यदि $x^3 + x \geq 2$
तो $x \geq 1$। फिर विरोधाभास द्वारा सिद्ध कीजिए: कोई लघुतम पूर्णतः धनात्मक
वास्तविक संख्या नहीं है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:logic:4}
आगमन द्वारा सिद्ध कीजिए कि सभी $n \in \N$ के लिए:
\begin{enumerate}
  \item $\sum_{k=0}^{n} 2^k = 2^{n+1} - 1$;
  \item $4^n + 5$ $3$ से विभाज्य है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:logic:5}
निम्नलिखित ``उपपत्ति'' में दोष खोजिए, जो सिद्ध करती है कि सभी पेंसिलों का
रंग एक ही है। \emph{मान लीजिए $P(n)$: ``$n$ पेंसिलों के प्रत्येक \omterm{def:b1:logic:sets}{समुच्चय} में
सभी पेंसिलों का रंग एक ही है''। $P(1)$ स्पष्ट है। $P(n)$ मानकर $n+1$
पेंसिलें लीजिए; अंतिम पेंसिल हटाने पर पहली $n$ का रंग एक है; पहली पेंसिल
हटाने पर अंतिम $n$ का रंग एक है; अतः सभी $n+1$ का रंग एक है।}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:logic:6}
मान लीजिए $A, B, C$ $E$ के उपसमुच्चय हैं। सिद्ध कीजिए:
\begin{enumerate}
  \item $A \setminus B = A \cap \overline{B}$;
  \item $(A \cup B) \setminus C = (A \setminus C) \cup (B \setminus
        C)$;
  \item $A \subseteq B \iff A \cup B = B \iff A \cap B = A$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:logic:7}
प्रत्येक \omterm{def:b1:logic:map}{प्रतिचित्रण} के लिए (उपपत्ति सहित) तय कीजिए कि वह \omterm{def:b1:logic:inj}{एकैकी} है, \omterm{def:b1:logic:inj}{आच्छादक}
है या \omterm{def:b1:logic:inj}{एकैकी आच्छादक}:
\begin{enumerate}
  \item $f \colon \N \to \N$, $n \mapsto n + 1$;
  \item $g \colon \Z \to \Z$, $n \mapsto n + 1$;
  \item $h \colon \R \setminus \{1\} \to \R$, $x \mapsto
        \frac{x+1}{x-1}$.
\end{enumerate}
$h$ के लिए सहप्रांत ऐसा समायोजित कीजिए कि वह \omterm{def:b1:logic:inj}{एकैकी आच्छादक} हो जाए, और
प्रतिलोम की संगणना कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:logic:8}
मान लीजिए $f \colon E \to F$, तथा $A, A' \subseteq E$ और $B, B' \subseteq
F$।
\begin{enumerate}
  \item सिद्ध कीजिए $f^{-1}(B \cap B') = f^{-1}(B) \cap f^{-1}(B')$ तथा $f(A
  \cup A') = f(A) \cup f(A')$।
  \item सिद्ध कीजिए $f(A \cap A') \subseteq f(A) \cap f(A')$ और ऐसा उदाहरण
  दीजिए जिसमें अंतर्विष्टि कठोर हो।
  \item सिद्ध कीजिए: $f$ \omterm{def:b1:logic:inj}{एकैकी} है यदि और केवल यदि सभी $A, A'$ के लिए $f(A
  \cap A') = f(A) \cap f(A')$।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:logic:9}
मान लीजिए $f \colon E \to F$ और $g \colon F \to E$ $g \circ f =
\mathrm{id}_E$ को संतुष्ट करते हैं। सिद्ध कीजिए कि $f$ \omterm{def:b1:logic:inj}{एकैकी} है और $g$
\omterm{def:b1:logic:inj}{आच्छादक} है। ऐसा उदाहरण दीजिए जिसमें $f$ और $g$ में से कोई भी \omterm{def:b1:logic:inj}{एकैकी आच्छादक} न
हो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:logic:10}
$\R$ पर $x \mathbin{\mathcal{R}} y \iff x^2 - y^2 = x - y$ परिभाषित कीजिए।
सिद्ध कीजिए कि $\mathcal{R}$ एक \omterm{def:b1:logic:equiv}{तुल्यता संबंध} है और प्रत्येक वास्तविक $x$ के
\omterm{def:b1:logic:equiv}{तुल्यता वर्ग} का वर्णन कीजिए। किन वर्गों में ठीक एक अवयव है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:logic:11}
(कैंटर) मान लीजिए $E$ एक \omterm{def:b1:logic:sets}{समुच्चय} है। सिद्ध कीजिए कि $E$ से $\mathcal{P}(E)$
पर कोई आच्छादन नहीं है। \emph{संकेत: $f \colon E \to \mathcal{P}(E)$ दिया
हो, तो $D = \{x \in E : x \notin f(x)\}$ पर विचार कीजिए।}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:logic:12}
मान लीजिए $f \colon E \to F$ एक \omterm{def:b1:logic:map}{प्रतिचित्रण} है। $\Phi \colon \mathcal{P}(F)
\to \mathcal{P}(E)$ को $\Phi(B) = f^{-1}(B)$ द्वारा परिभाषित कीजिए।
\begin{enumerate}
  \item सिद्ध कीजिए कि $f$ \omterm{def:b1:logic:inj}{आच्छादक} है यदि और केवल यदि $\Phi$ \omterm{def:b1:logic:inj}{एकैकी} है।
  \item सिद्ध कीजिए कि $f$ \omterm{def:b1:logic:inj}{एकैकी} है यदि और केवल यदि $\Phi$ \omterm{def:b1:logic:inj}{आच्छादक} है।
\end{enumerate}
\end{exercise}

\section{समस्या: अनंतताओं की तुलना}

\begin{problem}\label{pb:b1:logic:1}
दो समुच्चयों में ``अवयवों की संख्या समान'' कब होती है? कैंटर का उत्तर --- जब
उनके बीच एकैकी आच्छादन हो --- अनंत समुच्चयों के लिए भी प्रयोग करने योग्य
निकलता है, और वह अनंत को सचमुच भिन्न आकारों में बाँट देता है। यह समस्या इस
अध्याय की सादी परिभाषाओं से पूरा औज़ार-संदूक खड़ा करती है:
कैंटर--श्रोडर--बर्नस्टाइन प्रमेय (दो \omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} मिलकर एक एकैकी
आच्छादन गढ़ देते हैं), $\Q$ की गणनीयता, विकर्ण तर्क द्वारा $\R$ की अगणनीयता,
और कैंटर का 1874 का चकित करने वाला निष्कर्ष: \emph{अबीजीय संख्याएँ विद्यमान
हैं, और वे प्रचुर मात्रा में हैं} --- बिना एक भी संख्या प्रस्तुत किए।
\index{गणनीय समुच्चय}\index{कैंटर--श्रोडर--बर्नस्टाइन प्रमेय} आगे सर्वत्र,
समुच्चयों $E$ और $F$ के लिए $E \preceq F$ तब लिखिए जब $E$ से $F$ में कोई
\omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} हो, और $E \approx F$ (``$E$ और $F$ \emph{समशक्त} हैं'') तब
जब $E$ से $F$ पर कोई एकैकी आच्छादन हो।

\medskip \noindent\textbf{भाग I --- तुलना की शब्दावली।}
\begin{enumerate}
  \item दिखाइए कि $\approx$ \omterm{def:b1:logic:equiv}{तुल्यता संबंध} की भाँति व्यवहार करता है: $E
  \approx E$; यदि $E \approx F$ तो $F \approx E$; यदि $E \approx F$ और $F
  \approx G$ तो $E \approx G$। (\cref{thm:b1:logic:inverse} और
  \cref{prop:b1:logic:comp} को ठीक-ठीक उद्धृत कीजिए।)
  \item दिखाइए कि $\preceq$ संक्रमक है, और यह कि \omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} $f \colon
  E \to F$ सदा $E \approx f(E)$ प्रेरित करता है।
  \item मान लीजिए $E \neq \emptyset$। दिखाइए कि $E \preceq F$ यदि और केवल
  यदि $F$ से $E$ पर कोई आच्छादन विद्यमान हो।
  \item सत्यापित कीजिए कि $n \mapsto n + 1$ $\N$ से $\N^* = \N \setminus
  \{0\}$ पर एकैकी आच्छादन है, और यह कि
        \[
        \sigma(n) = \frac n2 \ \ (n \text{ सम}), \qquad
        \sigma(n) = -\frac{n+1}2 \ \ (n \text{ विषम})
        \]
        $\N$ से $\Z$ पर एकैकी आच्छादन है। अतः एक बिंदु हटा देने से, या
        ऋणात्मक पक्ष में दुगुना कर देने से, $\N$ का आकार नहीं बदलता।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- कैंटर--श्रोडर--बर्नस्टाइन प्रमेय।} मान
लीजिए $f \colon E \to F$ और $g \colon F \to E$ दो \omterm{def:b1:logic:inj}{एकैकी} \omterm{def:b1:logic:map}{प्रतिचित्रण} हैं।
परिभाषित कीजिए
\[
C_0 = E \setminus g(F), \qquad C_{n+1} = g\bigl(f(C_n)\bigr)
\ \ (n \in \N), \qquad C = \bigcup_{n \in \N} C_n,
\]
और मान लीजिए $h \colon E \to F$ $x \in C$ को $f(x)$ पर भेजता है तथा $x
\notin C$ को उस अद्वितीय $y \in F$ पर जिसके लिए $g(y) = x$।
\begin{enumerate}[resume]
  \item जाँचिए कि $h$ सुपरिभाषित है: यदि $x \notin C$, तो $x \in g(F)$, और
  $g(y) = x$ वाला अवयव $y$ अद्वितीय है।
  \item दिखाइए कि $g\bigl(f(C)\bigr) = \bigcup_{n \geq 1} C_n \subseteq C$।
  (अग्र प्रतिबिंब सम्मिलन के साथ क्रम-विनिमेय हैं: \cref{exo:b1:logic:8}।)
  \item दिखाइए कि $h$ \omterm{def:b1:logic:inj}{एकैकी} है। (तीन स्थितियाँ; मिश्रित स्थिति $x \in C$,
  $x' \notin C$ में दिखाइए कि $h(x) = h(x')$ होने पर $x' \in g(f(C))
  \subseteq C$ मानना पड़ेगा।)
  \item दिखाइए कि $h$ \omterm{def:b1:logic:inj}{आच्छादक} है: $y \in F$ दिया हो, तो $g(y) \notin C$ और
  किसी $n \geq 1$ के लिए $g(y) \in C_n$ --- इन दोनों स्थितियों में भेद कीजिए
  ($g(y) \in C_0$ असंभव क्यों है?) --- और हर स्थिति में $y$ का एक
  \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} प्रस्तुत कीजिए।
  \item \emph{कैंटर--श्रोडर--बर्नस्टाइन प्रमेय} के साथ निष्कर्ष निकालिए: यदि
  $E \preceq F$ और $F \preceq E$, तो $E \approx F$। एक वाक्य में टिप्पणी
  कीजिए कि इस \omterm{def:b1:logic:statement}{कथन} को अतुच्छ क्या बनाता है।
  \item दो अनुप्रयोग। (क) दिखाइए $\intcc01 \approx \intoo01$। (ख) दिखाइए कि
  $\varphi(p, q) = 2^p(2q + 1) - 1$ $\N \times \N$ से $\N$ पर एकैकी आच्छादन
  परिभाषित करता है --- एकैकीयता सम-विषम तर्क से, आच्छादकता प्रबल आगमन
  (\cref{thm:b1:logic:induction}) से। अतः $\N \times \N \approx \N$:
  पूर्णांक बिंदुओं का समतल रेखा से बड़ा नहीं है।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- \omterm{pb:b1:logic:1}{गणनीय समुच्चय}।} \omterm{def:b1:logic:sets}{समुच्चय} $E$ को
\emph{अधिकतम गणनीय} तब कहिए जब $E \preceq \N$, और \emph{गणनीय} तब जब $E
\approx \N$।
\begin{enumerate}[resume]
  \item दिखाइए कि $A \subseteq \N$ जैसा प्रत्येक अनंत उपसमुच्चय गणनीय है।
  ($\varphi(n)$ को पुनरावर्ती रूप से $A \setminus \{\varphi(0), \dots,
  \varphi(n-1)\}$ के न्यूनतम अवयव के रूप में परिभाषित कीजिए; दिखाइए कि
  $\varphi$ पूर्णतः वर्धमान है, $\varphi(n) \geq n$ को संतुष्ट करता है, और
  $A$ के प्रत्येक अवयव तक पहुँचता है।)
  \item इससे निकालिए कि कोई \omterm{def:b1:logic:sets}{समुच्चय} अधिकतम गणनीय है यदि और केवल यदि वह
  परिमित या गणनीय हो, और देखिए कि प्रश्न 9 यह संक्षेप देता है: यदि $E
  \preceq \N$ और $\N \preceq E$, तो $E$ गणनीय है।
  \item दिखाइए कि यदि $E$ और $F$ अधिकतम गणनीय हैं, तो $E \times F$ भी है।
  इससे निकालिए कि $\Z \times \N^*$ गणनीय है।
  \item दिखाइए कि $\Q$ गणनीय है। (प्रत्येक परिमेय संख्या को धनात्मक हर के
  साथ न्यूनतम पदों में लिखकर $\Q$ को $\Z \times \N^*$ में \omterm{def:b1:logic:inj}{एकैकी} रूप से भेजिए
  --- उस निरूपण की अद्वितीयता \cref{ch:b1:arith} में सिद्ध है; फिर प्रश्न 12
  लगाइए।)
  \item दिखाइए कि अधिकतम गणनीय समुच्चयों का गणनीय सम्मिलन अधिकतम गणनीय होता
  है: यदि प्रत्येक $E_n$ ($n \in \N$) अधिकतम गणनीय है, तो $\bigcup_{n \in
  \N} E_n$ भी है। ($x$ को युग्म $(n, f_n(x))$ पर भेजिए, जहाँ $n$ $x \in E_n$
  वाला \emph{न्यूनतम} सूचकांक है।)
  \item दिखाइए कि $\N$ के \emph{परिमित} उपसमुच्चयों का \omterm{def:b1:logic:sets}{समुच्चय} गणनीय है।
  (किसी परिमित उपसमुच्चय $F$ को $\sum_{i \in F} 2^i$ पर भेजिए; एकैकीयता के
  लिए उस बृहत्तम अवयव की तुलना कीजिए जहाँ दो परिमित \omterm{def:b1:logic:sets}{समुच्चय} भिन्न होते हैं,
  और \cref{exo:b1:logic:4} का $\sum_{k=0}^{m-1} 2^k = 2^m - 1$ प्रयोग
  कीजिए।)
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- विकर्णीकरण।} मान लीजिए $\{0,1\}^{\N}$
सभी प्रतिचित्रणों $u \colon \N \to \{0, 1\}$ का \omterm{def:b1:logic:sets}{समुच्चय} है, अर्थात्
द्विआधारी अनुक्रमों का \omterm{def:b1:logic:sets}{समुच्चय}।
\begin{enumerate}[resume]
  \item $\mathcal{P}(\N)$ और $\{0,1\}^{\N}$ के बीच एकैकी आच्छादन की रचना
  कीजिए (सूचक फलन)।
  \item (विकर्ण तर्क) मान लीजिए $\Phi \colon \N \to \{0,1\}^{\N}$ कोई
  \omterm{def:b1:logic:map}{प्रतिचित्रण} है। $d(n) = 1 - \Phi(n)(n)$ से परिभाषित अनुक्रम $d$ पर विचार
  कीजिए। दिखाइए कि $d$ $\Phi$ के प्रतिबिंब में नहीं है, और निष्कर्ष निकालिए
  कि $\{0,1\}^{\N}$ अधिकतम गणनीय \emph{नहीं} है। एक वाक्य में समझाइए कि
  प्रश्न 17 के रास्ते यह ठीक $E = \N$ के लिए कैंटर की प्रमेय
  (\cref{exo:b1:logic:11}) क्यों है।
  \item यह मान लीजिए --- जो विद्यालय से परिचित है और \cref{ch:b1:reals} में
  कठोरता से स्थापित है --- कि प्रत्येक $x \in \intco01$ का अद्वितीय
  \emph{उचित} दशमलव प्रसार $x = 0.d_1 d_2 d_3\dots$ होता है (जो $9$ की अनंत
  माला पर समाप्त न होता हो)। $\intco01$ के अवयवों का कोई भी अनुक्रम
  $(x_n)_{n \geq 1}$ दिया हो, तो ऐसा $x \in \intco01$ रचिए कि सभी $n$ के लिए
  $x \neq x_n$: उसका $n$-वाँ अंक $5$ चुनिए यदि $x_n$ का $n$-वाँ अंक $5$ से
  भिन्न हो, अन्यथा $6$। सावधानी से न्यायसंगत कीजिए कि $x$ उचित है और
  प्रत्येक $x_n$ से बचता है, और निष्कर्ष निकालिए कि $\intco01$ अधिकतम गणनीय
  नहीं है।
  \item इससे निकालिए कि $\R$ अगणनीय है, और यह कि अपरिमेय संख्याओं का \omterm{def:b1:logic:sets}{समुच्चय}
  $\R \setminus \Q$ भी अगणनीय है। किस सुनिश्चित अर्थ में ``अधिकांश''
  वास्तविक संख्याएँ अपरिमेय हैं?
\end{enumerate}

\medskip \noindent\textbf{भाग V --- कैंटर की 1874 की प्रमेय: अबीजीय संख्याएँ
विद्यमान हैं।} वास्तविक संख्या $x$ \emph{बीजीय} \index{बीजीय संख्या} तब
कहलाती है जब पूर्णांक गुणांकों वाले किसी अशून्य बहुपद $P$ के लिए $P(x) = 0$
हो, और अन्यथा \emph{अबीजीय} \index{अबीजीय संख्या}। इस भाग के लिए यह मान
लीजिए --- \cref{ch:b1:poly} में यह सिद्ध है --- कि घात $n$ के अशून्य बहुपद
के अधिक से अधिक $n$ वास्तविक मूल होते हैं।
\begin{enumerate}[resume]
  \item दिखाइए कि प्रत्येक परिमेय संख्या बीजीय है, और $\sqrt 2$ तथा $\sqrt 2
  + \sqrt 3$ का विलोपन करने वाले पूर्णांक गुणांकों के स्पष्ट बहुपद खोजिए।
  \item नियत $n \in \N$ के लिए दिखाइए कि पूर्णांक गुणांकों वाले, अधिकतम घात
  $n$ के बहुपदों का \omterm{def:b1:logic:sets}{समुच्चय} गणनीय है। (उसे $\Z^{n+1}$ में \omterm{def:b1:logic:inj}{एकैकी} रूप से भेजिए
  और प्रश्न 13 के साथ $n$ पर आगमन कीजिए।)
  \item इससे निकालिए कि पूर्णांक गुणांकों वाले \emph{सभी} बहुपदों का \omterm{def:b1:logic:sets}{समुच्चय}
  गणनीय है।
  \item \emph{बीजीय संख्याओं पर कैंटर की प्रमेय} सिद्ध कीजिए: बीजीय वास्तविक
  संख्याओं का \omterm{def:b1:logic:sets}{समुच्चय} $\mathcal{A}$ गणनीय है।
  \item निष्कर्ष निकालिए: अबीजीय वास्तविक संख्याएँ विद्यमान हैं, और अबीजीय
  संख्याओं का \omterm{def:b1:logic:sets}{समुच्चय} अगणनीय है। फिर कुछ वाक्यों में पूरी समस्या का
  लेखा-जोखा लीजिए: शृंखला $\N \approx \Z \approx \Q \approx \mathcal{A}$,
  $\R \approx$ तक की कठोर छलाँग, जो (मूलतः) $\mathcal{P}(\N)$ है, कहाँ
  कौन-सा औज़ार (कैंटर--श्रोडर--बर्नस्टाइन, गणनीय सम्मिलन, विकर्ण) निर्णायक
  रहा --- और यह सिद्ध करने का दार्शनिक प्रहार कि अगणनीय रूप से अनेक अबीजीय
  संख्याएँ विद्यमान हैं, बिना एक का भी नाम लिए। (किसी \emph{विशिष्ट} संख्या,
  जैसे $\pi$, का अबीजीय होना सिद्ध करना कहीं कठिन है और इस खंड की सीमा से
  परे।)
\end{enumerate}
\end{problem}
