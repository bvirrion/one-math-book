\chapter{खंड पर समाकलन}\label{ch:b1:integration}

उच्चतर माध्यमिक खंड का \omterm{thm:b1:integration:def}{समाकल} क्षेत्रफलों पर टिका था, जो सहज रूप से लिए गए
थे। यह अध्याय उसका निर्माण करता है: पहले सोपान फलनों के लिए, जहाँ \omterm{thm:b1:integration:def}{समाकल} एक
परिमित योग है, फिर \omterm{def:b1:continuity:continuous}{संतत} (और खंडशः \omterm{def:b1:continuity:continuous}{संतत}) फलनों के लिए एकसमान सन्निकटन द्वारा
--- और यहीं हाइने की प्रमेय (\cref{thm:b1:continuity:heine}) अपनी रोटी कमाती
है। इसके बाद कलन की मूल प्रमेय इस निर्माण को प्रतिअवकलजों से जोड़ देती है,
और \omterm{thm:b1:integration:riemann}{रीमान योग} उसे विविक्त औसतों से।

आगे सर्वत्र $a < b$ वास्तविक हैं।

\section{सोपान फलन}

\begin{definition}\label{def:b1:integration:step}
$\varphi \colon \intcc{a}{b} \to \R$ \emph{सोपान फलन}\index{सोपान फलन} तब है
जब कोई उपविभाजन $a = x_0 < x_1 < \dots < x_n = b$ ऐसा हो कि $\varphi$
प्रत्येक खुले \omterm{prop:b1:reals:intervals}{अंतराल} $\intoo{x_{i-1}}{x_i}$ पर अचर, $c_i$ के बराबर, हो
(गाँठों पर मान अप्रतिबंधित हैं)। उसका \omterm{thm:b1:integration:def}{समाकल} है
\[
\int_a^b \varphi = \sum_{i=1}^{n} c_i\,(x_i - x_{i-1}),
\]
जो चुने हुए उपविभाजन से स्वतंत्र है (दो उपविभाजनों को उनके उभयनिष्ठ उपविभाजन
से सूक्ष्म कीजिए: सूक्ष्मीकरण से दोनों पक्ष अपरिवर्तित रहते हैं)।
\end{definition}

\begin{proposition}\label{prop:b1:integration:stepprops}
सोपान फलनों पर \omterm{thm:b1:integration:def}{समाकल} रैखिक और वर्धमान ($\varphi \leq \psi \implies
\int\varphi \leq \int\psi$) है, और $a < c < b$ के लिए शाल संबंध $\int_a^b =
\int_a^c + \int_c^b$ संतुष्ट करता है।
\end{proposition}

\begin{proof}
इंजन है \emph{सूक्ष्मीकरण-अपरिवर्त्यता}, जो परिभाषा में ही कही गई है: किसी
उपविभाजन में एक अतिरिक्त गाँठ $t \in \intoo{x_{i-1}}{x_i}$ डालने से पद
$c_i(x_i - x_{i-1})$ के स्थान पर $c_i(t - x_{i-1}) + c_i(x_i - t)$ आ जाता है
--- वही संख्या --- अतः किसी भी परिमित सूक्ष्मीकरण से \omterm{thm:b1:integration:def}{समाकल} अपरिवर्तित रहता
है। अब उपविभाजन $\sigma$ वाला $\varphi$ और उपविभाजन $\sigma'$ वाला $\psi$
लीजिए: उभयनिष्ठ सूक्ष्मीकरण $\sigma \cup \sigma'$ पर दोनों \emph{एक ही}
गाँठों वाले \omterm{def:b1:integration:step}{सोपान फलन} हैं, और हर टुकड़े पर $\varphi + \lambda\psi$ $c_i +
\lambda d_i$ के बराबर अचर है: अतः रैखिकता परिमित योगों की रैखिकता पर सिमट
जाती है। वृद्धि: हर टुकड़े पर $c_i \leq d_i$ $\sum c_i \Delta_i \leq \sum
d_i \Delta_i$ देता है (लंबाइयाँ $\Delta_i \geq 0$)। शाल: गाँठ $c$ डालिए और
उस पर योग को तोड़ दीजिए।
\end{proof}

\section{संतत फलन का समाकल}

\begin{theorem}[एकसमान सन्निकटन]\label{thm:b1:integration:approx}
मान लीजिए $f$ $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} है। प्रत्येक $\varepsilon > 0$ के लिए
ऐसे \omterm{def:b1:integration:step}{सोपान फलन} $\varphi, \psi$ हैं कि
\[
\varphi \leq f \leq \psi
\qquad\text{और}\qquad
\psi - \varphi \leq \varepsilon \text{ पर } \intcc{a}{b}.
\]
\end{theorem}

\begin{proof}
हाइने की प्रमेय (\cref{thm:b1:continuity:heine}) से $f$ \omterm{def:b1:continuity:uniform}{एकसमान रूप से संतत}
है: $\varepsilon$ के लिए $\delta$ चुनिए, और जाल $< \delta$ का कोई उपविभाजन
(मान लीजिए $n > \frac{b - a}{\delta}$ टुकड़ों वाला समान \omterm{prop:b1:reals:intervals}{अंतराल} का)। प्रत्येक
\omterm{def:b1:topology:closed}{संवृत} टुकड़े $\intcc{x_{i-1}}{x_i}$ पर $f$ अपना न्यूनतम $m_i$ और अधिकतम
$M_i$ प्राप्त करता है (\cref{thm:b1:continuity:evt}), और $M_i - m_i \leq
\varepsilon$ (दोनों चरम बिंदु $\delta$ के भीतर हैं)। $\intoo{x_{i-1}}{x_i}$
पर $\varphi = m_i$ और $\psi = M_i$ परिभाषित कीजिए (और गाँठों पर $\varphi =
\psi = f$)।
\end{proof}

\begin{theorem}[समाकल की परिभाषा]\label{thm:b1:integration:def}
मान लीजिए $f$ $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} है। दोनों संख्याएँ
\[
I_-(f) = \sup\Bigl\{\int_a^b \varphi : \varphi \text{ सोपान},\
\varphi \leq f\Bigr\},
\qquad
I_+(f) = \inf\Bigl\{\int_a^b \psi : \psi \text{ सोपान},\ \psi \geq
f\Bigr\}
\]
बराबर हैं; उनका उभयनिष्ठ मान $\int_a^b f$ का \emph{समाकल}\index{समाकल} है
(जिसे $\int_a^b f(t)\,\dd t$ भी लिखा जाता है)। वह सोपान फलनों पर पिछली धारणा
से मेल खाता है, और असांतत्यों पर $\intcc{a}{b}$ को तोड़कर खंडशः \omterm{def:b1:continuity:continuous}{संतत} फलनों
तक बढ़ जाता है (वहाँ शाल परिभाषा का काम करता है)।
\end{theorem}

\begin{proof}
दोनों \omterm{def:b1:logic:sets}{समुच्चय} अरिक्त हैं ($f$ परिबद्ध है) और प्रत्येक निचला सोपान \omterm{thm:b1:integration:def}{समाकल} हर
ऊपरी से $\leq$ है (सोपान फलनों पर एकदिष्टता): अतः $I_-(f) \leq I_+(f)$।
\cref{thm:b1:integration:approx} से प्रत्येक $\varepsilon$ के लिए ऐसा युग्म
है कि $\int\psi - \int\varphi \leq \varepsilon(b - a)$: अतः \omterm{def:b1:reals:bounds}{उच्चतम} और
\omterm{def:b1:reals:bounds}{निम्नतम} आपस में दब जाते हैं, $I_- = I_+$।
\end{proof}

\begin{example}[खंडशः संतत, बिना किसी नाटक के]\label{ex:b1:integration:piecewise}
$\intcc{0}{3}$ पर \omterm{thm:b1:reals:floor}{फ़र्श फलन} वेश बदला \omterm{def:b1:integration:step}{सोपान फलन} है: उसकी कूदों पर तोड़ने से,
\[
\int_0^3 \lfloor t \rfloor\,\dd t
= \int_0^1 0 + \int_1^2 1 + \int_2^3 2 = 0 + 1 + 2 = 3 ,
\]
और कूद-बिंदुओं $1, 2$ \emph{पर} के मान निरर्थक हैं: परिमित कितने बिंदुओं पर
फलन बदलने से कोई \omterm{thm:b1:integration:def}{समाकल} नहीं बदलता (घेरने वाले \omterm{def:b1:integration:step}{सोपान फलन} अप्रभावित रहते हैं)।
``खंडशः \omterm{def:b1:continuity:continuous}{संतत}'' वाले विस्तार की पूरी सामग्री यही है: परिमित कितने असांतत्यों
पर काटिए, प्रत्येक \omterm{def:b1:continuity:continuous}{संतत} टुकड़े को समाकलित कीजिए, जोड़ दीजिए --- शाल परिभाषा
के रूप में।
\end{example}

\begin{example}[परिभाषा एक बार संगणना करती है]\label{ex:b1:integration:defcompute}
$\intcc{0}{1}$ पर $f(x) = x$ लीजिए और उसे $n$ समान टुकड़ों में काटिए।
टुकड़ों पर अचर सर्वोत्तम \omterm{def:b1:integration:step}{सोपान फलन} $k$-वें टुकड़े पर $\varphi =
\frac{k-1}{n}$ और $\psi = \frac kn$ हैं, जहाँ
\[
\int_0^1 \varphi = \sum_{k=1}^{n} \frac{k-1}{n}\cdot\frac1n
= \frac{n-1}{2n},
\qquad
\int_0^1 \psi = \sum_{k=1}^{n} \frac{k}{n}\cdot\frac1n
= \frac{n+1}{2n} .
\]
प्रत्येक निचला \omterm{thm:b1:integration:def}{समाकल} हर ऊपरी से $\leq I_-(f) \leq I_+(f) \leq$ है, अतः सभी
$n$ के लिए $\frac{n-1}{2n} \leq I_-(f) \leq I_+(f) \leq \frac{n+1}{2n}$:
दोनों $\frac12$ पर दब जाते हैं, और सीधे परिभाषा से $\int_0^1 x\,\dd x =
\frac12$। समापन का सार: परिभाषा से समाकलन हम पहली और अंतिम बार कर रहे हैं
--- नीचे की मूल प्रमेय ऐसी सभी संगणनाओं के स्थान पर एक प्रतिअवकलज खोज रख
देती है, और इस अध्याय का पूरा आर्थिक प्रयोजन यही है।
\end{example}

\begin{theorem}[गुणधर्म]\label{thm:b1:integration:props}
$\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} (या खंडशः \omterm{def:b1:continuity:continuous}{संतत}) $f, g$ और $\lambda \in \R$ के लिए:
\begin{enumerate}
  \item रैखिकता: $\int (f + \lambda g) = \int f + \lambda \int g$;
  \item एकदिष्टता: $f \leq g \implies \int_a^b f \leq \int_a^b g$; और
  $\bigl|\int_a^b f\bigr| \leq \int_a^b \abs f \leq (b - a)\, \sup\abs f$;
  \item शाल: $\int_a^b = \int_a^c + \int_c^b$ (परिपाटी $\int_b^a =
  -\int_a^b$ के साथ, जो सीमाओं के किसी भी क्रम के लिए वैध है);
  \item कठोर धनात्मकता: यदि $f$ \omterm{def:b1:continuity:continuous}{संतत} है, $f \geq 0$ और $\int_a^b f = 0$, तो
  $\intcc{a}{b}$ पर सर्वत्र $f = 0$।
\end{enumerate}
\end{theorem}

\begin{proof}
(1)--(3) \omterm{def:b1:reals:bounds}{उच्चतम}/\omterm{def:b1:reals:bounds}{निम्नतम} वाली परिभाषा के द्वारा सोपान फलनों से सीमा तक चले
जाते हैं। रैखिकता के विवरण एक बार देने योग्य हैं: $\varepsilon > 0$ दिया हो,
तो $\varphi_f \leq f \leq \psi_f$ और $\varphi_g \leq g \leq \psi_g$ को $\leq
\varepsilon$ (\cref{thm:b1:integration:approx}) अंतरालों के साथ घेरिए।
$\lambda \geq 0$ के लिए $\varphi_f + \lambda\varphi_g \leq f + \lambda g
\leq \psi_f + \lambda\psi_g$ सोपान फलनों से $\leq (1 + \lambda)\varepsilon$
\omterm{prop:b1:reals:intervals}{अंतराल} वाला घेराव है, और उसके सोपान \omterm{thm:b1:integration:def}{समाकल} $\int \varphi_f +
\lambda\int\varphi_g$ इत्यादि के बराबर हैं
(\cref{prop:b1:integration:stepprops}): अतः $\varepsilon \to 0$ लेने पर
$\int(f + \lambda g)$ $\int f + \lambda\int g$ पर दब जाता है। $\lambda < 0$
के लिए $\lambda$ से गुणा करना $g$ के घेराव को \emph{उलट} देता है ---
$\lambda g$ का निचला \omterm{def:b1:integration:step}{सोपान फलन} $\lambda \psi_g$ है --- और वही दबाव भूमिकाएँ
बदलकर चल जाता है। परिबंध $\abs{\int f} \leq \int \abs f$ $-\abs f \leq f
\leq \abs f$ और एकदिष्टता से आता है।

(4) प्रतिधनात्मक रूप से: यदि $f(x_0) = m > 0$, तो \omterm{def:b1:continuity:continuous}{सांतत्य} $\eta > 0$ लंबाई
का ऐसा उपअंतराल देती है जिस पर $f \geq \frac m2$; और वहाँ $\frac m2$ तथा
अन्यत्र $0$ वाला \omterm{def:b1:integration:step}{सोपान फलन} $\leq f$ है, अतः $\int f \geq \frac{m\eta}{2} >
0$।
\end{proof}

\begin{example}[शाल काम पर: निरपेक्ष मान वाले समाकल]\label{ex:b1:integration:chasleswork}
निरपेक्ष मान का समाकलन करने के लिए वहाँ काटिए जहाँ चिह्न बदलता है।
\[
\int_0^2 \abs{x - 1}\,\dd x
= \int_0^1 (1 - x)\,\dd x + \int_1^2 (x - 1)\,\dd x
= \frac12 + \frac12 = 1 ,
\]
और $\pi$ पर $\intcc{0}{2\pi}$ काटने पर:
\[
\int_0^{2\pi} \abs{\sin t}\,\dd t
= \int_0^{\pi} \sin t\,\dd t - \int_{\pi}^{2\pi} \sin t\,\dd t
= 2 + 2 = 4 ,
\]
जबकि $\int_0^{2\pi} \sin t\,\dd t = 0$: निरसन सचमुच होता है, और इसीलिए कठोर
धनात्मकता वाले \omterm{def:b1:logic:statement}{कथन} (\cref{thm:b1:integration:props}~(4)) में परिकल्पना $f
\geq 0$ है --- उसके बिना लुप्त होता \omterm{thm:b1:integration:def}{समाकल} $f$ के विषय में कुछ सिद्ध नहीं
करता। समापन का सार: $\int \abs f$ \emph{क्षेत्रफल} नापता है, $\int f$
\emph{चिह्नित संतुलन}; और असमिका $\abs{\int f} \leq \int\abs f$ ठीक-ठीक यही
अभिलेख है कि निरसन क्या नष्ट कर सकता है।
\end{example}

\section{कलन की मूल प्रमेय}

\begin{theorem}[कलन की मूल प्रमेय]\label{thm:b1:integration:ftc}
मान लीजिए $f$ किसी \omterm{prop:b1:reals:intervals}{अंतराल} $I$ पर \omterm{def:b1:continuity:continuous}{संतत} है और $a \in I$। फलन
\[
F(x) = \int_a^x f(t)\, \dd t
\]
$I$ पर $C^1$ वर्ग का है, जहाँ $F' = f$: अर्थात् किसी \omterm{prop:b1:reals:intervals}{अंतराल} पर प्रत्येक \omterm{def:b1:continuity:continuous}{संतत}
फलन के प्रतिअवकलज होते हैं। फलस्वरूप, $f$ के \emph{किसी भी} प्रतिअवकलज $G$
के लिए:\index{कलन की मूल प्रमेय}
\[
\int_a^b f(t)\,\dd t = G(b) - G(a) .
\]
\end{theorem}

\begin{proof}
$x_0 \in I$ और $\varepsilon > 0$ नियत कीजिए; $x_0$ पर \omterm{def:b1:continuity:continuous}{सांतत्य} ऐसा $\delta$
देती है कि $\abs{t - x_0} \leq \delta$ के लिए $\abs{f(t) - f(x_0)} \leq
\varepsilon$। $0 < \abs{h} \leq \delta$ (और $x_0 + h \in I$) के लिए शाल देता
है
\[
\frac{F(x_0 + h) - F(x_0)}{h} - f(x_0)
= \frac 1h \int_{x_0}^{x_0+h} \bigl(f(t) - f(x_0)\bigr)\dd t ,
\]
जिसका निरपेक्ष मान $\leq \frac{1}{\abs h}\cdot \abs h\, \varepsilon =
\varepsilon$ है (परिबंध (2), जो सीमाओं के किसी भी क्रम के लिए वैध है)। अतः
$F'(x_0) = f(x_0)$; $F' = f$ \omterm{def:b1:continuity:continuous}{संतत} है: $F$ $C^1$ है। यदि $G' = f$ भी हो, तो
\omterm{prop:b1:reals:intervals}{अंतराल} पर $(G - F)' = 0$, अतः $G = F + c$
(\cref{cor:b1:derivative:monotone}), और $G(b) - G(a) = F(b) - F(a) =
\int_a^b f$।
\end{proof}

\begin{example}[संगणना से पहले सममिति]\label{ex:b1:integration:parity}
सममित \omterm{prop:b1:reals:intervals}{अंतराल} पर सम-विषमता ही काम कर देती है: यदि $f$ विषम है, तो प्रतिस्थापन
$t \mapsto -t$ $\int_{-a}^{0} f$ को $-\int_0^a f$ पर भेज देता है, अतः
\[
\int_{-a}^{a} f(t)\,\dd t = 0 ;
\qquad\text{यदि $f$ सम है,}\quad
\int_{-a}^{a} f = 2\int_0^a f .
\]
इस प्रकार कोई प्रतिअवकलज दिखे बिना ही $\int_{-1}^{1} \frac{t^3\cos t}{1 +
t^4}\,\dd t = 0$ (समाकल्य विषम है), और $\int_{-\pi}^{\pi} t^2\cos t\,\dd t =
2\int_0^\pi t^2\cos t\, \dd t$। तकनीकों की ओर हाथ बढ़ाने से पहले सममिति जाँच
लीजिए: सबसे तेज़ \omterm{thm:b1:integration:def}{समाकल} वही है जिसे कभी संगणित ही न करना पड़े।
\end{example}

\begin{example}[अवकलज को देखते ही पहचानना]\label{ex:b1:integration:halfangle}
$\displaystyle\int_0^{\pi/2} \frac{\dd x}{1 + \cos x}$ संगणित कीजिए।
अर्ध-कोण सर्वसमिका $1 + \cos x = 2\cos^2\frac x2$ समाकल्य को
$\frac{1}{2}\bigl(1 + \tan^2\frac x2\bigr)$ में बदल देती है, जो ठीक
$\tan\frac x2$ का \omterm{def:b1:derivative:def}{अवकलज} है:
\[
\int_0^{\pi/2} \frac{\dd x}{1 + \cos x}
= \Bigl[\tan\frac x2\Bigr]_0^{\pi/2} = \tan\frac\pi4 = 1 .
\]
किसी प्रतिस्थापन-मशीनरी की आवश्यकता नहीं पड़ी --- केवल यह प्रतिवर्त कि
समाकल्य को किसी का \omterm{def:b1:derivative:def}{अवकलज} मानकर पढ़ा जाए, और शेष काम मूल प्रमेय कर देती है।
(ऐसे त्रिकोणमितीय समाकलों के पीछे का व्यवस्थित औज़ार, प्रतिस्थापन $t =
\tan\frac x2$, \cref{thm:b1:integration:parts}~(2) से बने मानक औज़ार-संदूक
का हिस्सा है।)
\end{example}

\begin{example}[समाकलों से परिभाषित फलन]\label{ex:b1:integration:definedby}
मूल प्रमेय फलन गढ़ देती है। मान लीजिए
\[
F(x) = \int_0^x \eu^{-t^2}\,\dd t .
\]
चिरपरिचित फलनों का कोई भी संयोजन ऐसा नहीं है जिसका \omterm{def:b1:derivative:def}{अवकलज} $\eu^{-t^2}$ हो
(लिउविल की एक प्रमेय, यहाँ स्वीकृत); फिर भी $F$ विद्यमान है, $F'(x) =
\eu^{-x^2} > 0$ के साथ $C^1$ है, कठोरतः वर्धमान है, विषम है ($t \mapsto -t$
प्रतिस्थापित कीजिए), और परिबद्ध: $x \geq 1$ के लिए,
\[
F(x) - F(1) = \int_1^x \eu^{-t^2}\dd t
\leq \int_1^x \eu^{-t}\dd t \leq \eu^{-1} ,
\]
अतः $F \leq F(1) + \eu^{-1} \leq 1 + \eu^{-1}$। (यथार्थ सीमा
$\frac{\sqrt\pi}{2}$ स्नातक वर्ष 3 के खंड में द्विसमाकलों से संगणित की गई
है।) चलती सीमाओं के लिए शृंखला नियम: $\frac{\dd}{\dd x}\int_x^{x^2}
\eu^{-t^2}\dd t = 2x\,\eu^{-x^4} - \eu^{-x^2}$। समापन का सार: समाकलन पुराने
फलनों से नए फलन \emph{गढ़ता} है, और उनके सभी गुणधर्म समाकल्य से पढ़े जा सकते
हैं --- जो प्रतिअवकलज आप लिख नहीं सकते, वह भी पूरी तरह आपके नियंत्रण में एक
फलन है।
\end{example}

\begin{example}[बिना मान निकाले आकलन]\label{ex:b1:integration:estimate}
समाकलों $R_n = \int_0^1 \frac{t^n}{1 + t}\,\dd t$ का कोई सुखद संवृत रूप नहीं
है, फिर भी एकदिष्टता उन्हें ठीक-ठीक कील देती है: $\intcc{0}{1}$ पर $\frac12
\leq \frac{1}{1+t} \leq 1$, अतः
\[
\frac{1}{2(n+1)} = \frac12\int_0^1 t^n\,\dd t
\;\leq\; R_n \;\leq\; \int_0^1 t^n \,\dd t = \frac{1}{n+1} :
\]
अर्थात् क्षय का यथार्थ क्रम ($R_n \sim$ $\frac1n$ का गुणज, वस्तुतः $R_n \sim
\frac{1}{2n}$) --- दो पंक्तियों में और बिना किसी प्रतिअवकलज के। इस अध्याय और
अगले की सप्ताहांत समस्याएँ ठीक ऐसे ही घेरावों पर चलती हैं --- किसी \omterm{thm:b1:integration:def}{समाकल} के
सामने विश्लेषक की पहली सहज क्रिया उसे \emph{परिबद्ध} करना होनी चाहिए, और
तभी, यदि आवश्यक हो, संगणना।
\end{example}

\begin{example}[औसत मान]\label{ex:b1:integration:average}
$\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} $f$ का \emph{माध्य} $\frac{1}{b-a}\int_a^b f$ है।
ज्या के मेहराब के लिए:
\[
\frac{1}{\pi}\int_0^\pi \sin t\,\dd t
= \frac{1}{\pi}\bigl[-\cos t\bigr]_0^\pi = \frac{2}{\pi}
\approx 0.637 :
\]
अर्थात् पूरे धनात्मक मेहराब का औसत $\frac12$ नहीं, बल्कि $\frac2\pi$ है ---
वक्र त्रिभुज की तुलना में अधिक समय ऊपर बिताता है।
\cref{exo:b1:integration:11} (समाकलों के लिए माध्य मान प्रमेय, $g = 1$) से
यह माध्य एक \emph{मान} है: किसी $c \in \intoo{0}{\pi}$ के लिए $\sin c =
\frac2\pi$। और इस अध्याय के रीमान योगों से यह माध्य $n$ नमूनों के साधारण
औसतों की सीमा है --- अर्थात् आँकड़ों के विविक्त माध्य और किसी संकेत के \omterm{def:b1:continuity:continuous}{संतत}
माध्य के बीच का सेतु, और इसी तरह \omterm{thm:b1:integration:def}{समाकल} भौतिकी में प्रवेश करता है।
\end{example}

\begin{theorem}[खंडशः समाकलन; प्रतिस्थापन]\label{thm:b1:integration:parts}
\begin{enumerate}
  \item यदि $u, v$ $\intcc{a}{b}$ पर $C^1$ हैं:\index{खंडशः समाकलन}
        \[
        \int_a^b u'v = \bigl[uv\bigr]_a^b - \int_a^b uv' .
        \]
  \item यदि $\varphi$ $\intcc{\alpha}{\beta}$ पर $C^1$ है और $f$
  $\varphi(\intcc{\alpha}{\beta})$ पर \omterm{def:b1:continuity:continuous}{संतत} है:\index{प्रतिस्थापन}
        \[
        \int_{\alpha}^{\beta} f\bigl(\varphi(t)\bigr)\,\varphi'(t)\,
        \dd t = \int_{\varphi(\alpha)}^{\varphi(\beta)} f(x)\, \dd x .
        \]
\end{enumerate}
\end{theorem}

\begin{proof}
(1) $(uv)' = u'v + uv'$; $\intcc{a}{b}$ पर समाकलन कीजिए और $C^1$ फलन $uv$ पर
मूल प्रमेय लगाइए।

(2) मान लीजिए $F$ प्रतिबिंब \omterm{prop:b1:reals:intervals}{अंतराल} पर $f$ का प्रतिअवकलज है
(\cref{thm:b1:integration:ftc})। तब $(F \circ \varphi)' = (f \circ
\varphi)\,\varphi'$ (शृंखला नियम), अतः दोनों पक्ष $F(\varphi(\beta)) -
F(\varphi(\alpha))$ के बराबर हैं।
\end{proof}

\begin{example}\label{ex:b1:integration:parts}
$\int_0^1 t\,\eu^t \dd t = \bigl[t\,\eu^t\bigr]_0^1 - \int_0^1 \eu^t\dd t =
\eu - (\eu - 1) = 1$। और प्रतिस्थापन $x = \sin t$ ($t \in
\intcc{0}{\frac\pi2}$) के साथ:
\[
\int_0^1 \sqrt{1 - x^2}\, \dd x
= \int_0^{\pi/2} \cos t \cdot \cos t \, \dd t
= \int_0^{\pi/2} \frac{1 + \cos 2t}{2}\, \dd t = \frac\pi4 ,
\]
--- अर्थात् इकाई वृत्तचकती का एक चौथाई, जैसा ज्यामिति माँगती है।
(\cref{met:b1:complex:trig} का रैखिकीकरण काम पर।)
\end{example}

\begin{remark}[समाकल कलन की सामान्य भूलें]
(क) \emph{प्रतिस्थापन पूरे \omterm{prop:b1:reals:intervals}{अंतराल} पर $C^1$ होने चाहिए}: परिवर्तन $x =
\frac1t$ $0$ को पार करते हुए अवैध है; उसे $\int_{-1}^{1}\frac{\dd x}{1 +
x^2}$ पर आँख मूँदकर लगाने से यह ``सिद्ध'' हो जाता है कि \omterm{thm:b1:integration:def}{समाकल} अपने ही विपरीत
के बराबर है। जब किसी प्रतिस्थापन में कोई विचित्रता हो, तो पहले \omterm{prop:b1:reals:intervals}{अंतराल} काटिए
(शाल), हर टुकड़े पर प्रतिस्थापन कीजिए, और तभी फिर से जोड़िए। (ख)
\emph{लघुगणकीय प्रतिअवकलजों को निरपेक्ष मान चाहिए}: $2$ के दोनों ओर अलग-अलग
$\int \frac{\dd x}{x - 2} = \ln\abs{x - 2} + C$ --- $\intoo{0}{1}$ पर $\ln(x
- 2)$ लिखना किसी ऋणात्मक संख्या का लघुगणक लिखना है; और अचर $C$ विचित्रता के
दोनों ओर भिन्न हो सकता है। (ग) \emph{लुप्त होता \omterm{thm:b1:integration:def}{समाकल} फलन को नहीं मार देता}:
$\int_0^{2\pi}\sin = 0$; $f = 0$ निकालने से पहले समाकल्य की धनात्मकता आवश्यक
है (\cref{ex:b1:integration:chasleswork})। (घ) \emph{\omterm{thm:b1:integration:riemann}{रीमान योग} अंशांकित होने
चाहिए}: $\frac{b-a}{n}\sum f\bigl(a + k\frac{b-a}{n}\bigr)$ में बाहर का पद
और भीतर के बिंदु उसी उपविभाजन से मेल खाने चाहिए --- बार-बार होने वाली भूल यह
है कि योग $\sum_{k=1}^{n} f\bigl(\frac kn\bigr)$ \emph{बिना} गुणक $\frac1n$
के लिख दिया जाता है, जो $\int_0^1 f$ की ओर अभिसरित होने के बदले अपसरित हो
जाता है। \cref{thm:b1:integration:riemann} का आह्वान करने से पहले जाँच-सूची:
$\frac1n$ बाहर निकालिए, योज्य को $\frac kn$ के $f$ के रूप में फिर से लिखिए,
$f$ का नाम दीजिए और उसकी \omterm{def:b1:continuity:continuous}{सांतत्य} जाँचिए।
\end{remark}

\begin{example}[अनुमान लगाइए, अवकलन कीजिए, समायोजन कीजिए]\label{ex:b1:integration:guessdiff}
$\int_1^x (\ln t)^2\,\dd t$ क्या है? $P$ \omterm{def:b1:poly:def}{बहुपद} के साथ $t\,P(\ln t)$ रूप के
प्रतिअवकलज का अनुमान लगाइए और अवकलन कीजिए:
\[
\bigl(t\,P(\ln t)\bigr)' = P(\ln t) + P'(\ln t) .
\]
हमें $P(u) + P'(u) = u^2$ चाहिए: $P(u) = u^2 - 2u + 2$ लीजिए ($u^2$ से नीचे
की ओर गुणांक मिलाते हुए)। अतः
\[
\int_1^x (\ln t)^2\,\dd t
= \bigl[t\bigl((\ln t)^2 - 2\ln t + 2\bigr)\bigr]_1^x
= x(\ln x)^2 - 2x\ln x + 2x - 2 ,
\]
यह परिणाम अन्यथा दो बार \omterm{thm:b1:integration:parts}{खंडशः समाकलन} से मिलता। समापन का सार: ($\ln t$ में
\omterm{def:b1:poly:def}{बहुपद}) या (\omterm{def:b1:poly:def}{बहुपद} गुणा $\eu^{\lambda t}$) रूप के समाकल्यों के लिए प्रतिअवकलज
का आकार वही रहता है --- अतः आकार वाले अनुमान का अवकलन समाकलन को गुणांकों पर
रैखिक बीजगणित में बदल देता है, जो दोहराए हुए \omterm{thm:b1:integration:parts}{खंडशः समाकलन} से तेज़ और कम भूल
वाला है।
\end{example}

\begin{example}[लौटता हुआ समाकल]\label{ex:b1:integration:boomerang}
$I = \int_0^{\pi/2} \eu^x \cos x\,\dd x$ संगणित कीजिए। दो बार \omterm{thm:b1:integration:parts}{खंडशः समाकलन}
कीजिए, और हर बार त्रिकोणमितीय गुणक का अवकलन कीजिए:
\[
I = \bigl[\eu^x \sin x\bigr]_0^{\pi/2}
- \int_0^{\pi/2} \eu^x \sin x\,\dd x
= \eu^{\pi/2} - J,
\qquad
J = \bigl[-\eu^x\cos x\bigr]_0^{\pi/2}
+ \int_0^{\pi/2} \eu^x\cos x\,\dd x = 1 + I .
\]
\omterm{thm:b1:integration:def}{समाकल} स्वयं अपने पास लौट आया है: $I = \eu^{\pi/2} - 1 - I$, जिससे
\[
I = \frac{\eu^{\pi/2} - 1}{2} .
\]
समापन का सार: जब समाकल्य ऐसे दो फलनों का गुणनफल हो जो अवकलन के अंतर्गत स्वयं
को फिर उत्पन्न करते हों ($\eu^{ax}$, $\cos bx$, $\sin bx$), तब दो बार \omterm{thm:b1:integration:parts}{खंडशः
समाकलन} अज्ञात \omterm{thm:b1:integration:def}{समाकल} के लिए एक रैखिक समीकरण दे देता है --- समाकलन के बदले उसे
हल कीजिए; समतुल्य रूप से $\eu^{(a + \iu b)x}$ (\cref{ch:b1:complex}) से होकर
जाइए और वास्तविक भाग लीजिए। दोनों रास्ते वही उत्तर देते हैं, और यह जाँच लेना
कि वे देते हैं, एक मुफ़्त संगति-परीक्षण है।
\end{example}

\section{रीमान योग}

\begin{theorem}[रीमान योग]\label{thm:b1:integration:riemann}
मान लीजिए $f$ $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} है। तब\index{रीमान योग}
\[
S_n = \frac{b - a}{n} \sum_{k=0}^{n-1}
f\Bigl(a + k\,\frac{b-a}{n}\Bigr)
\xrightarrow[n \to \infty]{} \int_a^b f(t)\, \dd t ,
\]
और उपअंतरालों के भीतर किन्हीं भी मूल्यांकन-बिंदुओं के साथ भी वही।
\end{theorem}

\begin{proof}
$S_n$ उस \omterm{def:b1:integration:step}{सोपान फलन} $\varphi_n$ का \omterm{thm:b1:integration:def}{समाकल} है जो $k$-वें उपअंतराल पर $f(a +
k\frac{b-a}{n})$ के बराबर है। $\varepsilon > 0$ दिया हो, तो \omterm{def:b1:continuity:uniform}{एकसमान सांतत्य}
(हाइने) $\delta$ देती है; और $n > \frac{b-a}{\delta}$ के लिए किसी भी
उपअंतराल का हर बिंदु अपने मूल्यांकन-बिंदु के $\delta$ के भीतर है, अतः
$\intcc{a}{b}$ पर $\abs{f - \varphi_n} \leq \varepsilon$, जिससे
\[
\Bigl| \int_a^b f - S_n \Bigr|
= \Bigl| \int_a^b (f - \varphi_n) \Bigr|
\leq (b-a)\,\varepsilon . \qedhere
\]
\end{proof}

\begin{example}\label{ex:b1:integration:riemannexample}
$\displaystyle\sum_{k=1}^{n} \frac{1}{n + k} = \frac 1n \sum_{k=1}^{n}
\frac{1}{1 + k/n} \xrightarrow[n\to\infty]{} \int_0^1 \frac{\dd x}{1 + x} =
\ln 2$: यह ऐसी सीमा है जो प्रारंभिक परिबंधों को दिखाई नहीं देती, पर \omterm{thm:b1:integration:riemann}{रीमान

\end{example}

\begin{example}[अंशांकन के साथ एक दूसरा रीमान योग]\label{ex:b1:integration:riemann2}
$\lim_{n\to\infty} \sum_{k=1}^{n} \dfrac{n}{(n+k)^2}$ खोजिए। अंशांकन कीजिए:
\[
\sum_{k=1}^{n} \frac{n}{(n+k)^2}
= \frac{1}{n}\sum_{k=1}^{n} \frac{n^2}{(n+k)^2}
= \frac1n \sum_{k=1}^{n} \frac{1}{\bigl(1 + \frac kn\bigr)^2} ,
\]
जो $\intcc{0}{1}$ पर \omterm{def:b1:continuity:continuous}{संतत} $f(x) = \frac{1}{(1+x)^2}$ का \omterm{thm:b1:integration:riemann}{रीमान योग} है: अतः
सीमा है
\[
\int_0^1 \frac{\dd x}{(1 + x)^2}
= \Bigl[-\frac{1}{1+x}\Bigr]_0^1 = \frac12 .
\]
समापन का सार: पूरी कला बीच की पंक्ति में है --- योज्य को ठीक एक गुणक
$\frac1n$ निकालने की क़ीमत पर $f(\frac kn)$ के आकार में लाइए; आकार सही हो
जाने पर विश्लेषण प्रमेय करती है और अंकगणित मूल प्रमेय।
\end{example}

\begin{remark}[समाकल आगे कहाँ काम करता है]
इस अध्याय की हर रचना की एक कड़ी आगे है। \omterm{thm:b1:integration:riemann}{रीमान योग} \cref{ch:b1:series} में
श्रेणियों और समाकलों के बीच सेतु के रूप में लौटते हैं ($\sum
\frac{1}{n^\alpha}$ की $\int \frac{\dd t}{t^\alpha}$ से तुलना); \omterm{thm:b1:integration:def}{समाकल} शेषफल
\cref{ch:b1:taylor} में टेलर सूत्र का सबसे तीक्ष्ण रूप है; और $\sup$-आधारित
परिभाषा स्नातक वर्ष 3 के खंड के लेबेग \omterm{thm:b1:integration:def}{समाकल} का आदिरूप है, जहाँ वही तीन
गुणधर्म (रैखिकता, एकदिष्टता, एक अभिसरण प्रमेय) फलनों के कहीं बड़े वर्ग पर
फिर खड़े किए जाते हैं। और नीचे दी गई सप्ताहांत समस्या \omterm{thm:b1:integration:parts}{खंडशः समाकलन} को
अंकगणित में बदल देती है: $\pi^2$ की अपरिमेयता।
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:b1:integration:5}
(कोशी--श्वार्ज़) $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} $f, g$ के लिए सिद्ध कीजिए

\Bigl(\int_a^b fg\Bigr)^{\!2} \leq \int_a^b f^2 \cdot \int_a^b g^2 ,
\]
इसके लिए $\int_a^b (f + \lambda g)^2 \geq 0$ को $\lambda$ में द्विघात के रूप

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:integration:6}
मान लीजिए $f$ $\R$ पर \omterm{def:b1:continuity:continuous}{संतत} और $T$-आवर्ती है। सिद्ध कीजिए कि $\int_a^{a+T} f$
$a$ पर निर्भर नहीं करता, और यह कि $x \to +\infty$ होने पर $\frac1x \int_0^x
f(t)\,\dd t \to \frac 1T \int_0^T f$।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:integration:7}
$\intcc{0}{1}$ पर \omterm{def:b1:continuity:continuous}{संतत} $f$ के लिए, जहाँ $\int_0^1 f = \frac12$, सिद्ध कीजिए
कि $f$ का $\intcc{0}{1}$ में कोई अचल बिंदु है। \emph{($f(x) - x$ का समाकलन
कीजिए और कठोर धनात्मकता (\cref{thm:b1:integration:props}~(4)) का उसके
प्रतिधनात्मक रूप में, मध्यवर्ती मान प्रमेय के साथ मिलाकर, प्रयोग कीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:integration:8}
(वालिस \omterm{thm:b1:integration:def}{समाकल}) मान लीजिए $W_n = \int_0^{\pi/2} \sin^n t\,\dd t$।
\begin{enumerate}
  \item \omterm{thm:b1:integration:parts}{खंडशः समाकलन} से पुनरावृत्ति $n W_n = (n-1) W_{n-2}$ ($n \geq 2$)
  सिद्ध कीजिए, और $W_0, W_1$, फिर $W_{2p}$ तथा $W_{2p+1}$ संवृत रूप में
  संगणित कीजिए।
  \item सिद्ध कीजिए कि $(W_n)$ ह्रासमान है और $\frac{W_{n+1}}{W_n} \to 1$;
  सिद्ध कीजिए कि राशि $(n+1)\,W_{n+1} W_n$ अचर और $\frac\pi2$ के बराबर है;
  और उससे तुल्यता $W_n \sim \sqrt{\dfrac{\pi}{2n}}$ निकालिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:integration:9}
(नाइवन: $\pi$ अपरिमेय है) मान लीजिए $a, b \in \N^*$ के साथ $\pi = \frac ab$,
और चुने जाने वाले $n$ के लिए रखिए
\[
P(x) = \frac{x^n (a - bx)^n}{n!},
\qquad
I_n = \int_0^{\pi} P(x)\sin x\, \dd x .
\]
\begin{enumerate}
  \item सिद्ध कीजिए कि $0 < I_n \leq \pi\,\frac{(\pi a)^n}{n!}$, जो बड़े $n$
  के लिए $< 1$ है।
  \item सिद्ध कीजिए कि $P$ और उसके सभी \omterm{def:b1:derivative:def}{अवकलज} $0$ तथा $\pi = \frac ab$ पर
  \emph{पूर्णांक} मान लेते हैं। \emph{(द्विपद प्रसार: $k!$ गुना $P$ के
  गुणांक $k \geq n$ के लिए पूर्णांक हैं; और $P(\pi - x) = P(x)$।)}
  \item $Q = P - P'' + P^{(4)} - \dots$ (एक परिमित योग) रखिए। जाँचिए कि
  $\bigl(Q'\sin x - Q\cos x\bigr)' = P \sin x$, और उससे निकालिए कि $I_n =
  Q(\pi) + Q(0)$ पूर्णांक है।
  \item निष्कर्ष निकालिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:integration:10}
मान लीजिए $f$ $\intcc{a}{b}$ पर $C^1$ है। \omterm{thm:b1:integration:parts}{खंडशः समाकलन} से रीमान--लेबेग
प्रकार की सीमा सिद्ध कीजिए
\[
\int_a^b f(t)\sin(\lambda t)\,\dd t \xrightarrow[\lambda \to
+\infty]{} 0
\]
फिर उसे केवल \omterm{def:b1:continuity:continuous}{संतत} $f$ के लिए भी सिद्ध कीजिए, सोपान फलनों से एकसमान सन्निकटन
के द्वारा (\cref{thm:b1:integration:approx})।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:integration:11}
(समाकलों के लिए माध्य मान प्रमेय) मान लीजिए $f, g$ $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} है
और $g \geq 0$। सिद्ध कीजिए कि ऐसा $c \in \intcc{a}{b}$ विद्यमान है कि
\[
\int_a^b f(t)\,g(t)\,\dd t = f(c)\int_a^b g(t)\,\dd t ,
\]
और उदाहरण से दिखाइए कि परिकल्पना $g \geq 0$ छोड़ी नहीं जा सकती।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:integration:12}
(आघूर्ण शून्यों पर बाध्य करते हैं) मान लीजिए $f$ $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} है
और
\[
\int_a^b f(t)\,t^k\,\dd t = 0 \qquad \text{के लिए } k = 0, 1,
\dots, n .
\]
सिद्ध कीजिए कि $f$ $\intoo{a}{b}$ के $n + 1$ भिन्न बिंदुओं पर लुप्त होता है।
\emph{(यदि $f$ केवल $z_1 < \dots < z_m$ पर चिह्न बदलता हो, जहाँ $m \leq n$,
तो $f$ का $P(t) = (t - z_1) \cdots (t - z_m)$ के सामने समाकलन कीजिए और कठोर
धनात्मकता का प्रयोग कीजिए।)}
\end{exercise}

\begin{remark}[इस खंड के भीतर आगे की दृष्टि]
आगे के तीन अध्याय सीधे इसी पर टिके हैं। \Cref{ch:b1:taylor} \omterm{thm:b1:integration:def}{समाकल} शेषफल ढोता
है --- तीनों टेलर सूत्रों में सबसे तीक्ष्ण $n$ बार दोहराया गया \omterm{thm:b1:integration:parts}{खंडशः समाकलन}
ही है। \Cref{ch:b1:series} योगों के समाकलों से घेराव को $\sum n^{-\alpha}$
की निर्णायक कसौटी में बदल देता है, और उसकी सप्ताहांत समस्या उसी घेराव को
परिष्कृत करके ऑयलर का अचर निकाल लेती है। \Cref{ch:b1:curves} \omterm{thm:b1:integration:def}{समाकल} को
ज्यामितीय बना देता है: किसी प्राचलित चाप की लंबाई $\int \sqrt{x'(t)^2 +
y'(t)^2}\,\dd t$ है, जो किसी खंड पर \omterm{def:b1:continuity:continuous}{संतत} फलन का \omterm{thm:b1:integration:def}{समाकल} है --- ठीक वही वस्तु
जो यहाँ बनी है, और उसके लिए किसी अनुचित सिद्धांत की आवश्यकता नहीं। सबसे अधिक
पुनःप्रयुक्त तथ्य सबसे विनम्र होगा: $\bigl|\int f\bigr| \leq (b -
a)\sup\abs{f}$, अर्थात् वह असमिका जो हर बिंदुवार आकलन को समाकल-आकलन में बदल
देती है।
\end{remark}

\section{समस्या: समाकल की अपरिमेयता-मशीन}

\begin{problem}[{सप्ताहांत समस्या --- $\eu$ और $\pi^2$ अपरिमेय हैं, $\eu$ छह दशमलव तक, और $\frac{22}{7} > \pi$
उपपत्ति सहित}]
\label{pb:b1:integration:1}
इस पूरी समस्या को एक ही क्रियाविधि चलाती है: \emph{ऐसा व्यंजक जिसे धनात्मक
पूर्णांक होना ही चाहिए, फिर भी जो सिद्ध रूप से $1$ से छोटा है, विद्यमान नहीं
हो सकता}। \Cref{exo:b1:integration:9} (नाइवन) ने उसे एक बार चलाकर $\pi
\notin \Q$ सिद्ध किया था; यहाँ हम उसका औद्योगीकरण करते हैं। मशीन को तीन भाग
चाहिए: एक \emph{पूर्णांकता} निवेश (सुचयनित बहुपदों के सिरों पर मान), एक
\emph{लघुता} निवेश (\omterm{thm:b1:integration:def}{समाकल} को कुचल देने वाला गुणक $\frac{1}{n!}$), और एक
\emph{सेतु} (\omterm{thm:b1:integration:parts}{खंडशः समाकलन}) जो दोनों को जोड़े। हम सिद्ध करते हैं कि $\eu$
अपरिमेय है और उसे प्रमाणित त्रुटि के साथ संगणित करते हैं, लजांद्र की
तीक्ष्णतर प्रमेय सिद्ध करते हैं कि $\pi^2$ अपरिमेय है, और विश्लेषण के सबसे
मनोहर \omterm{thm:b1:integration:def}{समाकल} पर समाप्त होते हैं: $\int_0^1 \frac{x^4(1 - x)^4}{1 + x^2}\dd x
= \frac{22}{7} - \pi$, जो $\pi$ का हाथ से घेराव कर देता
है।\index{अपरिमेयता!$\pi^2$ की}\index{अपरिमेयता!$\eu$ की}

\medskip \noindent\textbf{भाग I --- ईंधन।}
\begin{enumerate}
  \item सिद्ध कीजिए कि प्रत्येक नियत $c > 0$ के लिए $\dfrac{c^{\,n}}{n!} \to
  0$ \emph{($n \geq 2c$ के आगे हर पद कम से कम आधा हो जाता है)}।
  \item (बीटा \omterm{thm:b1:integration:def}{समाकल}) $m$ पर आगमन और \omterm{thm:b1:integration:parts}{खंडशः समाकलन} से सिद्ध कीजिए:
        \[
        \int_0^1 x^{\,k}\,(1 - x)^{\,m}\,\dd x =
        \frac{k!\,m!}{(k + m + 1)!}
        \qquad (k, m \in \N).
        \]
  \item $\displaystyle\int_0^1 \bigl(x(1-x)\bigr)^n \dd x =
  \frac{1}{(2n+1)\binom{2n}{n}}$ निकालिए, और --- परिबंध $x(1 - x) \leq
  \frac14$ से तुलना करके --- आकलन $\binom{2n}{n} \geq \dfrac{4^n}{2n+1}$, जो
  \cref{pb:b1:seq:1} के $\binom{2n}{n}^{1/n} \to 4$ से मेल खाता है।
  \item नीचे दो बार प्रयुक्त होने वाली लघुता-प्रमेयिका सिद्ध कीजिए:
  $\intoo{0}{1}$ पर प्रत्येक \omterm{def:b1:continuity:continuous}{संतत} $g > 0$ के लिए,
        \[
        0 < \int_0^1 \bigl(x(1-x)\bigr)^n g(x)\,\dd x
        \leq \frac{\sup_{\intcc{0}{1}}\abs g}{4^{\,n}} .
        \]
\end{enumerate}

\medskip \noindent\textbf{भाग II --- $\eu$: अपरिमेयता, फिर छह दशमलव।} $A_n =
\displaystyle\int_0^1 x^n \eu^x \dd x$ रखिए।
\begin{enumerate}[resume]
  \item $A_0$ और $A_1$ संगणित कीजिए, पुनरावृत्ति $A_n = \eu - n A_{n-1}$
  सिद्ध कीजिए, और परिबंध $0 < A_n \leq \dfrac{\eu}{n+1}$ भी।
  \item आगमन से दिखाइए कि $\alpha_n, \beta_n \in \Z$ के साथ $A_n = \alpha_n
  + \beta_n \eu$।
  \item उससे निकालिए कि $\eu$ अपरिमेय है \emph{(यदि $\eu = \frac pq$, तो
  बड़े $n$ के लिए $q A_n$ $\intoo{0}{1}$ में फँसा हुआ पूर्णांक है)}।
  \cref{exo:b1:seq:9} की उपपत्ति से तुलना कीजिए: वही चरम बिंदु, भिन्न ईंधन।
  \item आगमन और \omterm{thm:b1:integration:parts}{खंडशः समाकलन} से यथार्थ शेषफल-सूत्र सिद्ध कीजिए
        \[
        \eu = \sum_{k=0}^{n} \frac{1}{k!} + R_n,
        \qquad
        R_n = \frac{1}{n!}\int_0^1 (1 - t)^n\,\eu^{\,t}\,\dd t,
        \qquad
        \frac{1}{(n+1)!} \leq R_n \leq \frac{\eu}{(n+1)!} .
        \]
  \item $n = 9$ लीजिए: \cref{ex:b1:seq:e} में $b_2 = 2.75$ से मिले $\eu <
  2.75$ का प्रयोग करके $R_9$ को परिबद्ध कीजिए, योग का मान निकालिए, और
  प्रमाणित घेराव $2.7182818 \leq \eu \leq 2.7182823$ पर पहुँचिए --- अर्थात्
  छह दशमलव, $\eu \approx 2.718282$, उपपत्ति सहित।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- लजांद्र की प्रमेय: $\pi^2$ अपरिमेय
है।} मान लीजिए $f(x) = \dfrac{x^n (1 - x)^n}{n!}$, और मान लीजिए $a, b \in
\N^*$ के साथ $\pi^2 = \frac ab$।
\begin{enumerate}[resume]
  \item दिखाइए $f(1 - x) = f(x)$ और $\intoo{0}{1}$ पर $0 < f \leq
  \dfrac{1}{4^n\,n!}$।
  \item दिखाइए कि प्रत्येक $k \geq 0$ के लिए $f^{(k)}(0)$ और $f^{(k)}(1)$
  पूर्णांक हैं \emph{($x^n(1-x)^n$ का पूर्णांक गुणांकों के साथ प्रसार कीजिए;
  $k \geq n$ के लिए $\frac{k!}{n!} \in \Z$; फिर सममिति का प्रयोग कीजिए)}।
  \item परिभाषित कीजिए
        \[
        G = b^{\,n} \sum_{k=0}^{n} (-1)^k\,
        \pi^{2n - 2k} f^{(2k)} .
        \]
        दिखाइए कि $G(0)$ और $G(1)$ पूर्णांक हैं \emph{(प्रत्येक $b^n
        \pi^{2n-2k} = a^{\,n-k}\,b^{\,k}$)}।
  \item दूरबीनी $G'' + \pi^2 G = b^n \pi^{2n+2} f = \pi^2 a^n f$ सत्यापित
  कीजिए, फिर
        \[
        \frac{\dd}{\dd x}\Bigl(G'(x)\sin \pi x - \pi\,G(x)\cos
        \pi x\Bigr) = \pi^2 a^n f(x)\sin \pi x .
        \]
  \item $\intcc{0}{1}$ पर समाकलन कीजिए और निष्कर्ष निकालिए
        \[
        \pi\,a^n \int_0^1 f(x)\sin(\pi x)\,\dd x = G(0) + G(1)
        \in \Z ,
        \]
        अर्थात् $\dfrac{\pi a^n}{4^n\,n!}$ से परिबद्ध एक \emph{धनात्मक}
        पूर्णांक।
  \item प्रश्न 1 के साथ निष्कर्ष निकालिए कि $\pi^2$ अपरिमेय है (लजांद्र,
  1794), और यह कि इससे \cref{exo:b1:integration:9} प्रबल हो जाता है: $\pi^2$
  की अपरिमेयता से $\pi$ की अपरिमेयता क्यों आती है, और विलोम क्यों नहीं?
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- मशीन को समझना।}
\begin{enumerate}[resume]
  \item भाग II और III में दोनों विरोधी बल (सिरों के आँकड़ों की पूर्णांकता;
  \omterm{thm:b1:integration:def}{समाकल} की विश्लेषणात्मक लघुता) और सेतु ढूँढ़िए। फिर समझाइए कि $f$ में गुणक
  $\frac{1}{n!}$ ही मर्म क्यों है: यदि उसे हटा दिया जाए तो पूर्णांकता बची
  रहती है, पर कौन-सी असमिका मर जाती है, और किन दावा की गई भिन्नों $\frac ab$
  के लिए तब उपपत्ति विफल हो जाती है?
  \item प्रभाविता: मान लीजिए कोई $a \leq 10$ के साथ $\pi^2 = \frac ab$ का
  दावा करता है। दिखाइए कि विरोधाभास $n = 7$ पर ही आ गिरता है:
  $\pi\,(10/4)^7/7! \approx 0.38 < 1$ संगणित कीजिए। यह मशीन केवल खंडन नहीं
  करती; वह एक नियत, संगणनीय चरण पर खंडन करती है।
  \item सेतु की बिना शर्त संगति-जाँच कीजिए: दो बार \omterm{thm:b1:integration:parts}{खंडशः समाकलन} से सिद्ध
  कीजिए कि
        \[
        \int_0^1 x(1 - x)\sin(\pi x)\,\dd x = \frac{4}{\pi^3},
        \]
        और $n = 1$ पर प्रश्न 14 के साथ मेल बैठाइए ($\pi^2$ को सांकेतिक रखिए:
        दूरबीनी सर्वसमिका $\pi^3 \int_0^1 f_1 \sin \pi x = -(f_1''(0) +
        f_1''(1)) = 4$ कहती है)।
  \item $\eu^x$ और $\sin \pi x$ को इस मशीन के अष्टक बनने योग्य क्या बनाता
  है? वह गुणधर्म पहचानिए (हर एक अचर गुणांकों वाले किसी रैखिक अवकल समीकरण को
  संतुष्ट करता है, अतः बार-बार \omterm{thm:b1:integration:parts}{खंडशः समाकलन} आरंभ पर लौट आता है), और सीमांत
  बताइए: वही मशीन, एरमीट और लिंडेमान द्वारा परिष्कृत होकर, $\eu$ और $\pi$ को
  \emph{अबीजीय} सिद्ध कर देती है --- जो इस खंड से परे है।
\end{enumerate}

\medskip \noindent\textbf{भाग V --- $\frac{22}{7}$ बनाम $\pi$, और उपदेश।}
\begin{enumerate}[resume]
  \item \omterm{def:b1:poly:def}{बहुपद} भाग स्थापित कीजिए
        \[
        \frac{x^4(1-x)^4}{1 + x^2}
        = x^6 - 4x^5 + 5x^4 - 4x^2 + 4 - \frac{4}{1 + x^2},
        \]
        और चिरपरिचित सर्वसमिका निकालिए
        \[
        \int_0^1 \frac{x^4 (1-x)^4}{1 + x^2}\,\dd x
        = \frac{22}{7} - \pi .
        \]
  \item समाकल्य धनात्मक है: निष्कर्ष निकालिए $\pi < \frac{22}{7}$। फिर
  $\frac{1}{1+x^2}$ को $\frac12$ और $1$ के बीच परिबद्ध करके तथा $\int_0^1
  (x(1-x))^4 = \frac{1}{630}$ (प्रश्न 3) का प्रयोग करके सिद्ध कीजिए
        \[
        \frac{22}{7} - \frac{1}{630} \;\leq\; \pi \;\leq\;
        \frac{22}{7} - \frac{1}{1260} ,
        \]
        अर्थात् $3.14126 \leq \pi \leq 3.14207$: हाथ से दो सही दशमलव।
  \item सामान्यीकृत कीजिए: $x^{4m}(1-x)^{4m}$ को $1 + x^2$ से भाग देकर
  दिखाइए कि शेषफल अचर $(-4)^m$ है \emph{($x^2 + 1$ के सापेक्ष काम कीजिए:
  $(1-x)^2 \equiv -2x$)}, और ऐसी परिमेय संख्याएँ $r_m$ निकालिए कि
        \[
        \abs{\pi - r_m} \leq 4^{\,1 - 5m} ,
        \]
        और जाँचिए कि $m = 1$ प्रश्न 20--21 को फिर दे देता है।
  \item इन परिमेय संख्याओं का \cref{pb:b1:derivative:1} के सन्निकटन-सिद्धांत
  से सामना कराइए: डिरिक्ले की गारंटी $\frac{1}{49}$ के सामने $\abs{\pi -
  \frac{22}{7}} \approx 1.26\cdot10^{-3}$ संगणित कीजिए, और $\frac{1}{113^2}
  \approx 7.8\cdot10^{-5}$ के सामने $\abs{\pi - \frac{355}{113}} \approx
  2.7\cdot10^{-7}$ उद्धृत कीजिए: $\pi$ के लिए असाधारण रूप से अच्छे परिमेय
  सन्निकटन विद्यमान हैं --- जो संगत है, क्योंकि $\pi$ के बुरी तरह सन्निकटनीय
  होने का कोई प्रमाण नहीं है।
  \item (पूर्णांक-जाल, अमूर्त रूप में) वह प्रमेयिका सिद्ध कीजिए जो सब कुछ एक
  सूत्र में बाँध देती है: \emph{यदि $x \in \R$ और ऐसे पूर्णांक $a_n, b_n$
  हैं कि $0 < \abs{a_n + b_n x} \to 0$, तो $x$ अपरिमेय है।} इस समस्या में,
  \cref{exo:b1:integration:9} में, \cref{exo:b1:seq:9} में और
  \cref{pb:b1:derivative:1} में उसके उदाहरण गिनाइए।
  \item संश्लेषण, एक-एक वाक्य में: (क) मशीन के तीनों भाग और इस अध्याय के
  औज़ार-संदूक में हर एक कहाँ रहता है; (ख) \omterm{thm:b1:integration:def}{समाकल} क्या देता है जो
  \cref{pb:b1:derivative:1} की माध्य मान प्रमेय नहीं दे सकती थी; (ग) निकाले
  गए परिणामों की सूची (दो अपरिमेयताएँ, एक छह-दशमलव अचर, $\pi$ का एक घेराव,
  एक द्विपद परिबंध); (घ) सीमांत (एरमीट, लिंडेमान; और वही जाल, $\zeta(2)$ तथा
  $\zeta(3)$ पर चलाया हुआ, बीसवीं शताब्दी के अंकगणित में)।
\end{enumerate}
\end{problem}
