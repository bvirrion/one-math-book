\chapter{अवकलन}\label{ch:b1:derivative}

उच्चतर माध्यमिक खंड में \omterm{def:b1:derivative:def}{अवकलज} लगातार संगणित किए गए थे; जो अनुपस्थित था वह
प्रमेयों की वह शृंखला है जो संगणना को फलनों के विषय में सूचना में बदल देती
है: रोल की प्रमेय, माध्य मान प्रमेय, और उनके परिणाम --- एकदिष्टता की
कसौटियाँ, लिप्शिट्ज़ परिबंध, उत्तलता। इस अध्याय की हर बात किसी \omterm{prop:b1:reals:intervals}{अंतराल} $I$ पर
परिभाषित फलनों की है।

\section{अवकलज}

\begin{definition}\label{def:b1:derivative:def}
$f \colon I \to \R$ \emph{$x_0 \in I$ पर अवकलनीय}\index{अवकलज} तब है जब $x
\to x_0$ होने पर अंतर-भागफल $\frac{f(x) - f(x_0)}{x - x_0}$ की कोई (परिमित)
सीमा हो; उस सीमा को $f'(x_0)$ लिखा जाता है। समतुल्य रूप से:
\[
f(x_0 + h) = f(x_0) + f'(x_0)\,h + h\,\varepsilon(h),
\qquad \varepsilon(h) \xrightarrow[h \to 0]{} 0 ,
\]
और तब आलेख स्पर्श रेखा $y = f(x_0) + f'(x_0)(x - x_0)$ स्वीकार करता है।
$x_0$ पर अवकलनीयता से $x_0$ पर \omterm{def:b1:continuity:continuous}{सांतत्य} आती है (ऊपर का प्रदर्शन पढ़िए)। $f$
$I$ पर अवकलनीय तब है जब वह हर बिंदु पर हो; $f$ $C^1$ वर्ग का तब है जब इसके
अतिरिक्त $f'$ \omterm{def:b1:continuity:continuous}{संतत} हो, और $C^k$ वर्ग\index{Ck@$C^k$ वर्ग का फलन} का तब जब
$f$ को $k$ बार अवकलित किया जा सके और $f^{(k)}$ \omterm{def:b1:continuity:continuous}{संतत} हो।
\end{definition}

\begin{example}\label{ex:b1:derivative:notc1}
``\omterm{def:b1:derivative:def}{अवकलनीय} $\Rightarrow$ \omterm{def:b1:continuity:continuous}{संतत}'' का विलोम विफल है: $0$ पर $\abs{\,\cdot\,}$।
और अधिक आश्चर्यजनक बात यह कि \omterm{def:b1:derivative:def}{अवकलनीय} होने से $C^1$ होना नहीं आता: फलन $f(x)
= x^2 \sin\frac 1x$ ($f(0) = 0$) सर्वत्र \omterm{def:b1:derivative:def}{अवकलनीय} है, जहाँ $f'(0) = 0$, पर
$f'(x) = 2x \sin\frac1x - \cos\frac 1x$ की $0$ पर कोई सीमा नहीं
(\cref{exo:b1:derivative:2})।
\end{example}

\begin{example}[ठीक एक ही बिंदु पर अवकलनीय]\label{ex:b1:derivative:onepoint}
मान लीजिए $x \in \Q$ के लिए $f(x) = x^2$ और $x \notin \Q$ के लिए $f(x) = 0$।
$0$ पर: $\bigl|\frac{f(h) - 0}{h}\bigr| \leq \abs h \to 0$, अतः $f$ $0$ पर
\omterm{def:b1:derivative:def}{अवकलनीय} है और $f'(0) = 0$। किसी भी $x_0 \neq 0$ पर $f$ \omterm{def:b1:continuity:continuous}{संतत} तक नहीं है:
$x_0$ की ओर अभिसरित होते परिमेय और अपरिमेय अनुक्रम $f$ को क्रमशः $x_0^2 \neq
0$ और $0$ पर भेजते हैं (\omterm{def:b1:topology:dense}{सघनता}, \cref{thm:b1:reals:density})। अतः अवकलनीयता
सचमुच \emph{बिंदुवार} धारणा है: वह $\R$ के एक बिंदु पर सत्य हो सकती है और
अन्यत्र कहीं नहीं। व्यवहार के लिए उपदेश: एकदिष्टता कसौटी या रोल जैसे कथनों
को \omterm{def:b1:derivative:def}{अवकलज} \emph{किसी \omterm{prop:b1:reals:intervals}{अंतराल} पर} चाहिए --- वियुक्त बिंदुओं पर, चाहे कितने ही
हों, $f'(x_0)$ का होना किसी भी समग्र निष्कर्ष का आधार नहीं बनता।
\end{example}

\begin{theorem}[संक्रियाएँ]\label{thm:b1:derivative:operations}
यदि $f, g$ $x_0$ पर \omterm{def:b1:derivative:def}{अवकलनीय} हैं (और जहाँ सूत्र अर्थपूर्ण हों):
\[
(f + g)' = f' + g', \qquad
(fg)' = f'g + fg', \qquad
\Bigl(\frac fg\Bigr)' = \frac{f'g - fg'}{g^2},
\]
और यदि $g$ $f(x_0)$ पर \omterm{def:b1:derivative:def}{अवकलनीय} है: $\;(g \circ f)'(x_0) =
g'\bigl(f(x_0)\bigr)\, f'(x_0)$ (शृंखला नियम\index{शृंखला नियम})।
\end{theorem}

\begin{proof}
योग: तुरंत। गुणनफल: लिखिए
\[
f(x)g(x) - f(x_0)g(x_0)
= \bigl(f(x) - f(x_0)\bigr) g(x) + f(x_0)\bigl(g(x) - g(x_0)\bigr),
\]
$x - x_0$ से भाग दीजिए और $x \to x_0$ लीजिए ($g$ $x_0$ पर \omterm{def:b1:continuity:continuous}{संतत} है)। भागफल:
$\frac{1/g(x) - 1/g(x_0)}{x - x_0} = \frac{-1}{g(x)g(x_0)}\cdot\frac{g(x) -
g(x_0)}{x - x_0}$ के द्वारा $\frac 1g$ सँभालिए, फिर गुणनफल नियम लगाइए।
शृंखला नियम: $y_0 = f(x_0)$ के साथ $y \neq y_0$ के लिए $\theta(y) =
\frac{g(y) - g(y_0)}{y - y_0}$ और $\theta(y_0) = g'(y_0)$ परिभाषित कीजिए:
$\theta$ $y_0$ पर \omterm{def:b1:continuity:continuous}{संतत} है, और $x \neq x_0$ के लिए
\[
\frac{g(f(x)) - g(f(x_0))}{x - x_0}
= \theta\bigl(f(x)\bigr)\cdot \frac{f(x) - f(x_0)}{x - x_0}
\longrightarrow g'(y_0)\, f'(x_0),
\]
जहाँ पहला गुणक सीमाओं के संयोजन से मिलता है (यह युक्ति $f(x) = f(x_0)$ वाली
स्थिति स्वच्छ रूप से सँभाल लेती है, जहाँ भोला ``$f(x) - f(x_0)$ से गुणा और
भाग'' टूट जाता है)।
\end{proof}

\begin{theorem}[प्रतिलोम फलन का अवकलज]\label{thm:b1:derivative:inverse}
मान लीजिए $f$ $I$ पर \omterm{def:b1:continuity:continuous}{संतत} और कठोरतः एकदिष्ट है, $x_0$ पर \omterm{def:b1:derivative:def}{अवकलनीय} है और
$f'(x_0) \neq 0$। तब $f^{-1}$ (\cref{thm:b1:continuity:bijection}) $y_0 =
f(x_0)$ पर \omterm{def:b1:derivative:def}{अवकलनीय} है, जहाँ
\[
(f^{-1})'(y_0) = \frac{1}{f'(x_0)} = \frac{1}{f'\bigl(f^{-1}(y_0)\bigr)} .
\]
यदि $f'(x_0) = 0$, तो प्रतिलोम की $y_0$ पर ऊर्ध्वाधर स्पर्श रेखा होती है।
\end{theorem}

\begin{proof}
$y \to y_0$ के लिए $x = f^{-1}(y)$ रखिए: $f^{-1}$ की \omterm{def:b1:continuity:continuous}{सांतत्य} $x \to x_0$
देती है, और
\[
\frac{f^{-1}(y) - f^{-1}(y_0)}{y - y_0}
= \frac{x - x_0}{f(x) - f(x_0)}
= \frac{1}{\dfrac{f(x) - f(x_0)}{x - x_0}}
\longrightarrow \frac{1}{f'(x_0)} .
\]
ऊर्ध्वाधर स्पर्श रेखा का दावा: यदि $f'(x_0) = 0$, तो दिखाया गया भागफल ऐसी
राशि का प्रतिलोम है जो $0$ की ओर जाती है और एक ही चिह्न बनाए रखती है (कठोरतः
वर्धमान $f$ के लिए सभी $x \neq x_0$ पर $\frac{f(x) - f(x_0)}{x - x_0} > 0$):
अतः $f^{-1}$ का अंतर-भागफल $+\infty$ की ओर जाता है (ह्रासमान $f$ के लिए
$-\infty$ की ओर)। प्रतिलोम \omterm{def:b1:continuity:continuous}{संतत} तो रहता है पर $y_0$ पर \omterm{def:b1:derivative:def}{अवकलनीय} नहीं --- उसका
आलेख, जो $f$ के आलेख का विकर्ण में परावर्तन है, ठीक वहीं ऊर्ध्वाधर खड़ा हो
जाता है जहाँ $f$ का क्षैतिज चला था, जैसा $x^3$ के सामने $0$ पर $x^{1/3}$
दिखाता है।
\end{proof}

\begin{example}[प्रतिलोम अवकलज, दो बार]\label{ex:b1:derivative:inverseaction}
यह प्रमेय चिरपरिचित अवकलजों को बिना किसी सीमा-कार्य के फिर से संगणित कर देती
है। $\ln = \exp^{-1}$ के लिए: $y = \eu^x$ पर,
\[
(\ln)'(y) = \frac{1}{\exp'(x)} = \frac{1}{\eu^{x}} = \frac1y ,
\]
जो प्रत्येक $y > 0$ के लिए वैध है, क्योंकि $\exp' = \exp$ कभी लुप्त नहीं
होता। $\arctan = \tan^{-1}$ के लिए: $y = \tan x$ पर,
\[
(\arctan)'(y) = \frac{1}{1 + \tan^2 x} = \frac{1}{1 + y^2} ,
\]
जहाँ $\tan' = 1 + \tan^2 > 0$ प्रयुक्त हुआ। समापन का सार: यह सूत्र किसी फलन
के विषय में ज्ञान को उसके प्रतिलोम के विषय में ज्ञान में बदल देता है, और
उसकी क़ीमत एक प्रतिस्थापन है --- और वह प्रतिस्थापन ($x = \ln y$, $x =
\arctan y$) ठीक यही \omterm{def:b1:logic:statement}{कथन} है कि दोनों चर एकैकी आच्छादन के आमने-सामने के पक्षों
पर रहते हैं।
\end{example}

\section{रोल और माध्य मान प्रमेय}

\begin{proposition}[भीतरी चरम]\label{prop:b1:derivative:fermat}
यदि $f$ $I$ के किसी \emph{भीतरी} बिंदु $x_0$ पर \omterm{def:b1:derivative:def}{अवकलनीय} है और वहाँ उसका
स्थानीय चरम है, तो $f'(x_0) = 0$।
\end{proposition}

\begin{proof}
मान लीजिए स्थानीय अधिकतम: ऐसा $r > 0$ है कि $\abs{x - x_0} \leq r$ के लिए
$f(x) \leq f(x_0)$, और भीतरी होने से $I$ के भीतर $x_0$ के दोनों पक्ष उपलब्ध
हैं। $0 < h \leq r$ के लिए भागफल $\frac{f(x_0 + h) - f(x_0)}{h}$ का अंश
$\leq 0$ और हर $> 0$ है: वह $\leq 0$ है, और उसकी सीमा $f'(x_0)$ $\leq 0$
विरासत में पाती है (चौड़ी असमिकाएँ सीमाओं तक जाती हैं,
\cref{thm:b1:seq:order}); और $-r \leq h < 0$ के लिए भागफल $\geq 0$ है, जो
$f'(x_0) \geq 0$ देता है। अतः $f'(x_0) = 0$। (सिरे पर केवल एक ही चिह्न
उपलब्ध होता है: वहाँ निष्कर्ष विफल हो जाता है --- $\intcc{0}{1}$ पर $x$
देखिए, जिसका अधिकतम $1$ पर है और \omterm{def:b1:derivative:def}{अवकलज} $1$।)
\end{proof}

\begin{theorem}[रोल]\label{thm:b1:derivative:rolle}
मान लीजिए $f$ $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत}, $\intoo{a}{b}$ पर \omterm{def:b1:derivative:def}{अवकलनीय} है और $f(a)
= f(b)$। तब किसी $c \in \intoo{a}{b}$ के लिए $f'(c) = 0$।\index{रोल की
प्रमेय}
\end{theorem}

\begin{proof}
चरम मान प्रमेय (\cref{thm:b1:continuity:evt}) से $f$ $\intcc{a}{b}$ पर अपने
अधिकतम और न्यूनतम प्राप्त करता है। यदि दोनों सिरों पर प्राप्त होते हैं, तो
($f(a) = f(b)$ होने से) अधिकतम $=$ न्यूनतम और $f$ अचर है: कोई भी भीतरी $c$
काम करता है। अन्यथा कोई चरम किसी भीतरी बिंदु $c$ पर प्राप्त होता है, और
\cref{prop:b1:derivative:fermat} $f'(c) = 0$ दे देती है।
\end{proof}

\begin{theorem}[माध्य मान प्रमेय]\label{thm:b1:derivative:mvt}
मान लीजिए $f$ $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} और $\intoo{a}{b}$ पर \omterm{def:b1:derivative:def}{अवकलनीय} है। ऐसा $c
\in \intoo{a}{b}$ विद्यमान है कि
\[
f(b) - f(a) = f'(c)\,(b - a) .
\]
\emph{माध्य मान असमिका:} यदि इसके अतिरिक्त $\intoo{a}{b}$ पर $m \leq f' \leq
M$, तो $m(b-a) \leq f(b) - f(a) \leq M(b-a)$; विशेष रूप से $\abs{f'} \leq K$
से $f$ का $K$-लिप्शिट्ज़ होना आ जाता है।\index{माध्य मान प्रमेय}
\end{theorem}

\begin{proof}
$g(x) = f(x) - \frac{f(b) - f(a)}{b - a}(x - a)$ पर रोल लगाइए: $g$
$\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} है, भीतर \omterm{def:b1:derivative:def}{अवकलनीय} है, और $g(a) = f(a) = g(b)$। जिस
बिंदु $c$ पर $g'(c) = 0$, वहाँ: $f'(c) = \frac{f(b)-f(a)}{b-a}$। असमिका
$f'(c)$ को परिबद्ध करने से निकलती है; और लिप्शिट्ज़ वाला \omterm{def:b1:logic:statement}{कथन} उसे हर
बिंदु-युग्म पर लगाने से।
\end{proof}

\begin{example}[न्यूटन की विधि हीरो की ही है]\label{ex:b1:derivative:newton}
$f(x) = 0$ हल करने की न्यूटन विधि वक्र के स्थान पर वर्तमान अनुमान $x_n$ पर
उसकी स्पर्श रेखा रख देती है और उस स्पर्श रेखा का मूल अगला अनुमान मान लेती
है:
\[
0 = f(x_n) + f'(x_n)(x_{n+1} - x_n)
\quad\Longrightarrow\quad
x_{n+1} = x_n - \frac{f(x_n)}{f'(x_n)} .
\]
इसे $f(x) = x^2 - 2$ पर चलाइए:
\[
x_{n+1} = x_n - \frac{x_n^2 - 2}{2x_n}
= \frac{x_n}{2} + \frac{1}{x_n}
= \frac12\Bigl(x_n + \frac{2}{x_n}\Bigr) :
\]
यह ठीक हीरो की पुनरावृत्ति (\cref{ex:b1:seq:heron}) है, दो सहस्राब्दी पहले।
वहाँ देखी गई द्विघातीय गति अब स्पर्श रेखा के चित्र से समझ में आ जाती है:
किसी सरल मूल के पास वक्र और स्पर्श रेखा द्वितीय कोटि की त्रुटि भर भिन्न होते
हैं, अतः हर पद त्रुटि का लगभग वर्ग कर देता है --- व्यापक \omterm{def:b1:logic:statement}{कथन}
\cref{ch:b1:taylor} के टेलर परिबंधों से निकलता है। समापन का सार: जहाँ
द्विभाजन (\cref{ex:b1:continuity:dichotomy}) केवल \omterm{def:b1:continuity:continuous}{सांतत्य} का प्रयोग करता है
और प्रति पद एक बिट कमाता है, वहाँ न्यूटन एक \omterm{def:b1:derivative:def}{अवकलज} ख़र्च करके प्रति पद सही
अंकों की संख्या दुगुनी कर देता है।
\end{example}

\begin{example}[आकलक के रूप में माध्य मान प्रमेय]\label{ex:b1:derivative:mvtestimate}
$\sqrt{101}$ कितना बड़ा है? प्रमेय को $\intcc{100}{101}$ पर $f(t) = \sqrt t$
के लिए लगाइए: किसी $c \in \intoo{100}{101}$ के लिए,
\[
\sqrt{101} - 10 = \frac{1}{2\sqrt c},
\qquad\text{अतः}\qquad
\frac{1}{2\sqrt{101}} < \sqrt{101} - 10 < \frac{1}{20} = 0.05 ,
\]
और चूँकि $\sqrt{101} < 10.05$, बायाँ परिबंध $\frac{1}{20.1} > 0.0497$ से
बड़ा है: अतः $10.0497 < \sqrt{101} < 10.05$ (सच्चा मान $10.049875\dots$) ---
अर्थात् एक ही अवकलज-मूल्यांकन से तीन सही दशमलव। इसी प्रकार $\abs{\sin a -
\sin b} \leq \abs{a - b}$ (परिबंध $\abs{\cos}\leq 1$): \cref{ch:b1:seq} से
प्रयुक्त सभी लिप्शिट्ज़ आकलन यही प्रमेय हैं। समापन का सार: माध्य मान प्रमेय
शून्य कोटि का टेलर सूत्र है --- वह एक अज्ञात बिंदु $c$ के बदले एक कठोर
असमिका दे देती है, और \cref{ch:b1:taylor} ठीक इसी सौदे को बार-बार दोहराएगा।
\end{example}

\begin{corollary}[एकदिष्टता की कसौटी]\label{cor:b1:derivative:monotone}
मान लीजिए $f$ $I$ पर \omterm{def:b1:continuity:continuous}{संतत} है और \omterm{def:b1:topology:closure}{अंतःभाग} पर \omterm{def:b1:derivative:def}{अवकलनीय}।
\begin{enumerate}
  \item \omterm{def:b1:topology:closure}{अंतःभाग} पर $f' \geq 0$ $\iff$ $f$ वर्धमान है; $f' = 0$ $\iff$ $f$
  अचर।
  \item यदि परिमित कितने बिंदुओं को छोड़कर, जहाँ वह लुप्त होता है, $f' > 0$,
  तो $f$ \emph{कठोरतः} वर्धमान है।
\end{enumerate}
\end{corollary}

\begin{proof}
यदि $f' \geq 0$: $I$ में $x < y$ के लिए $\intcc{x}{y}$ पर माध्य मान प्रमेय
$f(y) - f(x) = f'(c)(y - x) \geq 0$ देती है। विलोमतः, वर्धमान फलन के
अंतर-भागफल $\geq 0$ हैं, अतः उनकी सीमाएँ भी। अचर वाली स्थिति: पिछली बात $f'$
और $-f' \geq 0$ पर लगाइए। कठोर रूप: $f$ वर्धमान है; और $x < y$ के लिए समता
$f(x) = f(y)$ $f$ को $\intcc{x}{y}$ पर जमा देती, जिससे वहाँ $f' = 0$
अनिवार्य हो जाता --- अर्थात् अनंत कितने बिंदु।
\end{proof}

\begin{example}[बराबर अवकलज, असमान फलन]\label{ex:b1:derivative:intervalmatters}
$\R^* = \intoo{-\infty}{0} \cup \intoo{0}{+\infty}$ पर $f(x) = \ln\abs x$ और
$g(x) = \ln\abs x + \mathbf{1}_{x>0}$ (दाईं अर्धरेखा पर ही $1$ जोड़िए) दोनों
$f' = g' = \frac1x$ संतुष्ट करते हैं। वे किसी अचर भर भिन्न नहीं हैं: कसौटी
``$f' = 0 \implies f$ अचर'' \emph{\omterm{prop:b1:reals:intervals}{अंतराल}} का \omterm{def:b1:logic:statement}{कथन} है --- उसकी उपपत्ति दो
बिंदुओं के बीच माध्य मान प्रमेय चलाती है, जिसके लिए उन्हें जोड़ने वाले पूरे
खंड का प्रांत में होना आवश्यक है। अलग-अलग हर अर्धरेखा पर $\frac1x$ के
प्रतिअवकलज $\ln\abs x + c$ हैं, जिनमें प्रति अर्धरेखा एक अचर है, अर्थात् कुल
दो स्वतंत्र अचर। \Cref{ch:b1:integration} यही महीन शर्त विरासत में पाता है:
किसी फलन का ``वही'' प्रतिअवकलज \emph{उसके प्रांत के प्रत्येक \omterm{prop:b1:reals:intervals}{अंतराल} पर} अचर
तक सुपरिभाषित होता है, और प्रतिअवकलज सारणियाँ चुपचाप संबद्धता मान लेती हैं।
\end{example}

\begin{example}[कठोरता मुफ़्त में]\label{ex:b1:derivative:strictness}
$x \mapsto x^3$ $\R$ पर \emph{कठोरतः} वर्धमान है, यद्यपि उसका \omterm{def:b1:derivative:def}{अवकलज} $0$ पर
लुप्त हो जाता है: कसौटी का उपवाक्य ``परिमित कितने बिंदुओं को छोड़कर $f' >
0$'' ठीक ऐसे चपटे बिंदुओं के लिए बना है। इसके विपरीत, अकेला $f' \geq 0$ केवल
चौड़े अर्थ में वृद्धि देता है (अचर फलन भी योग्य है), और किसी पूरे उपअंतराल
पर लुप्त होने वाला \omterm{def:b1:derivative:def}{अवकलज} फलन को वहाँ जमा ही देता है। व्यावहारिक नियम: कठोर
एकदिष्टता का दावा करने के लिए $f'$ के शून्य गिनाइए; परिमित कितने (और अधिक
व्यापक रूप से, किसी उपअंतराल पर एक भी नहीं) निरापद हैं, पर उनका पूरा \omterm{prop:b1:reals:intervals}{अंतराल}
घातक है।
\end{example}

\begin{example}[एक पूरा विचरण-अध्ययन]\label{ex:b1:derivative:variations}
$\R$ पर $f(x) = x^3 - 3x + 1$ का अध्ययन कीजिए। \omterm{def:b1:derivative:def}{अवकलज}: $f'(x) = 3(x^2 - 1)$,
जो $\intoo{-\infty}{-1}$ पर धनात्मक, $\intoo{-1}{1}$ पर ऋणात्मक और
$\intoo{1}{+\infty}$ पर धनात्मक है: अतः एकदिष्टता कसौटी से $f$ बढ़ता है, फिर
घटता है, फिर बढ़ता है, जिसमें स्थानीय अधिकतम $f(-1) = 3$ और स्थानीय न्यूनतम
$f(1) = -1$ है। सीमाएँ: $\mp\infty$ पर $\mp\infty$। प्रत्येक एकदिष्ट शाखा पर
मध्यवर्ती मान प्रमेय के साथ विचरण सारणी से पढ़े गए परिणाम: $f$ इनमें से
प्रत्येक में ठीक एक बार लुप्त होता है
\[
\intoo{-\infty}{-1}, \qquad \intoo{-1}{1}, \qquad
\intoo{1}{+\infty}
\]
(संधियों पर मानों के चिह्न विपरीत हैं: $3 > 0 > -1$), अतः समीकरण $x^3 - 3x +
1 = 0$ के ठीक तीन वास्तविक मूल हैं; संख्यात्मक रूप से वे $-1.88$, $0.35$,
$1.53$ के पास बैठते हैं। समापन का सार: विचरण सारणी एक \emph{उपपत्ति-यंत्र}
है, कोई रेखाचित्र नहीं --- एकदिष्ट शाखा और चिह्न-परिवर्तन मिलकर ठीक एक मूल
देते हैं, और सारणी सभी शाखाएँ पूरी तरह गिना देती है।
\end{example}

\begin{theorem}[लाइब्निज़ सूत्र]\label{thm:b1:derivative:leibniz}
यदि $f, g$ $n$ बार \omterm{def:b1:derivative:def}{अवकलनीय} हैं, तो $fg$ भी है, और\index{लाइब्निज़ सूत्र}
\[
(fg)^{(n)} = \sum_{k=0}^{n} \binom nk f^{(k)}\, g^{(n-k)} .
\]
\end{theorem}

\begin{proof}
$n$ पर आगमन, ठीक द्विपद प्रमेय के समांतर। स्थिति $n = 1$ गुणनफल नियम है।
कोटि $n$ पर सूत्र मानकर एक बार और अवकलन कीजिए:
\[
(fg)^{(n+1)} = \sum_{k=0}^{n} \binom nk
\Bigl( f^{(k+1)} g^{(n-k)} + f^{(k)} g^{(n-k+1)} \Bigr),
\]
फिर पहले योग को $j = k + 1$ से पुनःसूचीबद्ध कीजिए और $f^{(j)} g^{(n+1-j)}$
का गुणांक इकट्ठा कीजिए: वह पास्कल के नियम
(\cref{prop:b1:counting:identities}) से $\binom{n}{j-1} + \binom nj =
\binom{n+1}{j}$ है, और सीमांत पद $j = 0$ तथा $j = n + 1$ $\binom{n+1}{0} =
\binom{n+1}{n+1} = 1$ ढो रहे हैं, जैसा उन्हें ढोना चाहिए।
\end{proof}

\begin{example}[लाइब्निज़ काम पर]\label{ex:b1:derivative:leibnizaction}
$n \geq 2$ के लिए $\bigl(x^2 \eu^x\bigr)^{(n)}$ संगणित कीजिए। $f = x^2$
लीजिए, जिसके \omterm{def:b1:derivative:def}{अवकलज} शीघ्र मर जाते हैं ($f' = 2x$, $f'' = 2$, और $k \geq 3$ के
लिए $f^{(k)} = 0$), और $g = \eu^x$: तो लाइब्निज़ योग के केवल तीन पद बचते
हैं,
\[
\bigl(x^2\eu^x\bigr)^{(n)}
= \binom n0 x^2 \eu^x + \binom n1 (2x)\,\eu^x + \binom n2\,
2\,\eu^x
= \eu^x\bigl(x^2 + 2nx + n(n-1)\bigr).
\]
$n = 1$ पर संगति की जाँच: $\eu^x(x^2 + 2x)$, जो सचमुच $(x^2\eu^x)'$ है।
समापन का सार: लाइब्निज़ का प्रयोग तब कीजिए जब एक गुणक \omterm{def:b1:poly:def}{बहुपद} हो --- तब योग
में केवल $\deg + 1$ पद रहते हैं, और सूत्र संवृत रूप है, कोई अमूर्त सर्वसमिका
नहीं। (दो अनंत जीवंत गुणकों, जैसे $\eu^x\sin x$, के लिए \cref{ch:b1:complex}
के सम्मिश्र चरघातांकी बेहतर औज़ार हैं।)
\end{example}

\section{उत्तलता}

\begin{definition}\label{def:b1:derivative:convex}
$f \colon I \to \R$ \emph{उत्तल}\index{उत्तल फलन} तब है जब हर जीवा आलेख के
ऊपर हो:
\[
\forall x, y \in I,\ \forall t \in \intcc{0}{1}, \quad
f\bigl(tx + (1-t)y\bigr) \leq t f(x) + (1-t) f(y).
\]
($f$ \emph{अवतल} तब है जब $-f$ उत्तल हो।)
\end{definition}

\begin{theorem}[अवकलीय अभिलक्षण]\label{thm:b1:derivative:convexchar}
मान लीजिए $f$ $I$ पर \omterm{def:b1:derivative:def}{अवकलनीय} है। ये तुल्य हैं:
\begin{enumerate}
  \item $f$ \omterm{def:b1:derivative:convex}{उत्तल} है;
  \item $f'$ $I$ पर वर्धमान है;
  \item आलेख हर स्पर्श रेखा के ऊपर है: सभी $x, y \in I$ के लिए $f(y) \geq
  f(x) + f'(x)(y - x)$।
\end{enumerate}
यदि $f$ दो बार \omterm{def:b1:derivative:def}{अवकलनीय} है: $f$ \omterm{def:b1:derivative:convex}{उत्तल} $\iff f'' \geq 0$।
\end{theorem}

\begin{proof}
(1 $\Rightarrow$ 3) उत्तलता को $t \in \intoc{0}{1}$ के लिए $\frac{f(x +
t(y-x)) - f(x)}{t} \leq f(y) - f(x)$ के रूप में लिखिए; और $t \to 0^+$ लीजिए:
$f'(x)(y - x) \leq f(y) - f(x)$।

(3 $\Rightarrow$ 2) $x < y$ के लिए $x$ पर और $y$ पर दोनों स्पर्श-असमिकाएँ
$f'(x)(y-x) \leq f(y) - f(x) \leq f'(y)(y - x)$ देती हैं, अतः $f'(x) \leq
f'(y)$।

(2 $\Rightarrow$ 1) $x < y$ और $t \in \intoo{0}{1}$ नियत कीजिए, और $z = tx +
(1-t)y \in \intoo{x}{y}$ रखिए। $\intcc{x}{z}$ तथा $\intcc{z}{y}$ पर माध्य
मान प्रमेय से ऐसे $c_1 < z < c_2$ हैं कि
\[
\frac{f(z) - f(x)}{z - x} = f'(c_1) \leq f'(c_2)
= \frac{f(y) - f(z)}{y - z} ,
\]
और हर हटाने ($z - x = (1-t)(y-x)$, $y - z = t(y-x)$) पर यह ठीक उत्तलता
असमिका में पुनर्व्यवस्थित हो जाता है।

दो बार \omterm{def:b1:derivative:def}{अवकलनीय} स्थिति: $f'' \geq 0 \iff f'$ वर्धमान
(\cref{cor:b1:derivative:monotone})।
\end{proof}

\begin{example}[चिरपरिचित उत्तलता असमिकाएँ]\label{ex:b1:derivative:convexineq}
$\exp$ \omterm{def:b1:derivative:convex}{उत्तल} है ($\exp'' = \exp > 0$): $0$ पर उसकी स्पर्श रेखा सभी $x$ के
लिए $\eu^x \geq 1 + x$ देती है। $\ln$ अवतल है: $1$ पर उसकी स्पर्श रेखा $\ln
x \leq x - 1$ देती है; और $0 < a \leq b$ के लिए उसकी जीवाएँ गुणोत्तर तथा
समांतर माध्यों के बीच की असमिका देती हैं: अवतलता में $t = \frac12$ लेने पर,
\[
\ln\frac{a + b}{2} \geq \frac{\ln a + \ln b}{2} = \ln\sqrt{ab},
\qquad\text{अतः}\qquad
\sqrt{ab} \leq \frac{a+b}{2} .
\]
व्यापक समांतर--गुणोत्तर असमिका \cref{exo:b1:derivative:9} है।
\end{example}

\begin{example}[शून्य से एक उत्तलता असमिका]\label{ex:b1:derivative:xlnx}
फलन $f(t) = t\ln t$ $\intoo{0}{+\infty}$ पर \omterm{def:b1:derivative:convex}{उत्तल} है: $f''(t) = \frac1t >
0$। उसकी मध्यबिंदु असमिका, $2$ से गुणा करने पर, कहती है: सभी $a, b > 0$ के
लिए,
\[
a\ln a + b\ln b \;\geq\; (a + b)\,\ln\frac{a + b}{2} ,
\]
जहाँ समता तभी जब $a = b$ (कठोर उत्तलता)। परीक्षण: $a = 1$, $b = 3$ $4\ln 2 =
2.773$ के सामने $3\ln 3 = 3.296$ देता है। यह निर्दोष दिखने वाली असमिका उस
\emph{एंट्रॉपी} तुलना की दो-बिंदु स्थिति है जो येंसन असमिका
(\cref{exo:b1:derivative:9}) के साथ और स्नातक वर्ष 3 के खंड के
सूचना-सैद्धांतिक अनंतस्पर्शी में फिर प्रकट होती है। समापन का सार: कोई असमिका
गढ़नी हो तो ऐसा फलन खोजिए जिसके द्वितीय \omterm{def:b1:derivative:def}{अवकलज} का कोई चिह्न हो, और लिख डालिए
कि उत्तलता क्या कहती है --- अवकलीय अभिलक्षण एक ही चिह्न-जाँच को अनंत
असमिकाओं में बदल देता है।
\end{example}

\begin{remark}[अवकलजों की सामान्य भूलें]
(क) \emph{एक बिंदु पर धनात्मक \omterm{def:b1:derivative:def}{अवकलज} उसके पास एकदिष्टता नहीं देता}: $f(x) =
\frac x2 + x^2\sin\frac1x$ ($f(0) = 0$ के साथ) में $f'(0) = \frac12 > 0$ है,
फिर भी
\[
f'(x) = \frac12 + 2x\sin\frac1x - \cos\frac1x
\]
प्रत्येक $x_n = \frac{1}{2\pi n}$ पर वह $-\frac12$ के बराबर है: अर्थात् $0$
के हर \omterm{def:b1:topology:open}{प्रतिवेश} में उतार हैं। एकदिष्टता को $f' \geq 0$ \emph{किसी \omterm{prop:b1:reals:intervals}{अंतराल} पर}
चाहिए (\cref{cor:b1:derivative:monotone}); बिंदुवार चिह्न केवल स्पर्श रेखा
के काटे जाने को नियंत्रित करता है। (ख) \emph{रोल की तीनों परिकल्पनाएँ सक्रिय
हैं}: $\intcc{-1}{1}$ पर $\abs x$ (कोई भीतरी अवकलनीयता नहीं), $\intcc{0}{1}$
पर $x$ (सिरे बराबर नहीं), और $\intcc{0}{1}$ पर $x - \lfloor x\rfloor$ ($1$
पर \omterm{def:b1:continuity:continuous}{सांतत्य} विफल) --- हर एक ठीक एक परिकल्पना और निष्कर्ष तोड़ देता है। (ग)
\emph{\omterm{def:b1:derivative:def}{अवकलज} असंतत हो सकते हैं, पर मनमाने ढंग से नहीं}: $f'$ दोलन कर सकता है
(\cref{ex:b1:derivative:notc1}) फिर भी वह सदा मध्यवर्ती मान गुणधर्म संतुष्ट
करता है (डार्बू, \cref{exo:b1:derivative:10}): \omterm{def:b1:derivative:def}{अवकलज} कभी कूदता नहीं --- यदि
आपने कूद वाली कोई एकपक्षीय ``अवकलज-सीमा'' संगणित की है, तो आपने किसी
अनवकलनीय फलन का अवकलन किया है। (घ) \emph{प्रतिलोम वाले सूत्र को $f' \neq 0$
चाहिए}: $x \mapsto x^3$ एक चिकना कठोरतः वर्धमान एकैकी आच्छादन है जिसके
प्रतिलोम $x^{1/3}$ की $0$ पर ऊर्ध्वाधर स्पर्श रेखा है --- प्रतिलोम की
अवकलनीयता ठीक वहीं खोती है जहाँ $f'$ लुप्त होता है
(\cref{thm:b1:derivative:inverse})।
\end{remark}

\begin{remark}[माध्य मान प्रमेय आगे कहाँ काम करती है]
अगले अध्यायों का लगभग हर परिमाणात्मक \omterm{def:b1:logic:statement}{कथन} इसी अध्याय की माध्य मान प्रमेय है,
बस वेश बदले हुए: कलन की मूल प्रमेय (\cref{ch:b1:integration}) क्षेत्रफल फलन
का अवकलन करती है और एकदिष्टता कसौटी पर समाप्त होती है; टेलर--लाग्रांज सूत्र
(\cref{ch:b1:taylor}) $n$ बार दोहराई गई माध्य मान प्रमेय है; न्यूटन विधि और
अचल-बिंदु पुनरावृत्तियों (\cref{exo:b1:derivative:11}) का त्रुटि-विश्लेषण
लिप्शिट्ज़ रूप है; और इस अध्याय की सप्ताहांत समस्या
(\cref{pb:b1:derivative:1}) उसी लिप्शिट्ज़ परिबंध को संख्या-सिद्धांत में बदल
देती है --- बीजीय संख्याओं और परिमेय संख्याओं के बीच एक प्रतिकर्षण असमिका,
जिससे इतिहास की पहली \omterm{pb:b1:logic:1}{अबीजीय संख्या} मिलती है। स्नातक वर्ष 2 के खंड में माध्य
मान \emph{असमिका} कई चरों में बची रहती है, यद्यपि समता नहीं।
\end{remark}

\begin{example}[अवतलता से यंग असमिका]\label{ex:b1:derivative:young}
मान लीजिए $p, q > 1$ और $\frac1p + \frac1q = 1$। सभी $a, b > 0$ के लिए:
\[
ab \;\leq\; \frac{a^p}{p} + \frac{b^q}{q} .
\]
उपपत्ति: भार $\frac1p, \frac1q$ के साथ $\ln$ की अवतलता का एक ही प्रयोग
(दो-बिंदु येंसन असमिका, जैसा \cref{exo:b1:derivative:9} में है):
\[
\ln\Bigl(\frac{a^p}{p} + \frac{b^q}{q}\Bigr)
\;\geq\; \frac1p \ln(a^p) + \frac1q \ln(b^q)
= \ln a + \ln b = \ln(ab),
\]
और $\ln$ के वर्धमान होने से लघुगणकों की असमिका दावे में बदल जाती है; समता
तभी जब $a^p = b^q$ (कठोर अवतलता)। स्थिति $p = q = 2$ वेश बदली हुई
समांतर-गुणोत्तर असमिका $ab \leq \frac{a^2 + b^2}{2}$ है। समापन का सार: यंग
असमिका स्नातक वर्ष 2 के खंड की होल्डर और मिंकोव्स्की असमिकाओं का बीजीय बीज
है --- $\ln$ के विषय में एक ही अवतलता \omterm{def:b1:logic:statement}{कथन}, जिसे मानकों के लिए काट लिया जाता
है।
\end{example}

\begin{remark}[इस खंड के भीतर आगे की दृष्टि]
खंड समाप्त होने से पहले \omterm{def:b1:derivative:def}{अवकलज} तीन नए जीवन पाता है। \cref{ch:b1:taylor} में
वह दोहराता है: किसी बिंदु पर $n$ \omterm{def:b1:derivative:def}{अवकलज} एक \omterm{def:b1:poly:def}{बहुपद} और एक नियंत्रित त्रुटि में
सिमट जाते हैं, और माध्य मान प्रमेय लाग्रांज शेषफल बन जाती है।
\cref{ch:b1:curves} में अवकलन ज्यामितीय हो जाता है: प्राचलित वक्र $t \mapsto
(x(t), y(t))$ के लिए युग्म $(x'(t), y'(t))$ वेग \emph{सदिश} है, स्पर्शिता
संरेखता बन जाती है, और क्रांतिक बिंदु वर्गीकृत करने योग्य उभयाग्र बन जाते
हैं। \cref{ch:b1:multivar} में एक समय में एक चर जमा दिया जाता है: आंशिक
\omterm{def:b1:derivative:def}{अवकलज} इस अध्याय को दो बार दोहरा देते हैं, और स्पर्श रेखा बढ़कर स्पर्श तल बन
जाती है। तीनों अध्याय वही व्याकरण विरासत में पाते हैं --- स्थानीय रैखिक
सन्निकटन और एक त्रुटि पद --- जो पहली बार यहीं बोला गया है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:derivative:1}
इनका अवकलन कीजिए (प्रांत बताते हुए): $x^x$; $\;\ln\bigl(x +
\sqrt{x^2+1}\bigr)$; $\;\arctan\frac{1}{x}$; $\;\sqrt{1 + \eu^{2x}}$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:2}
\cref{ex:b1:derivative:notc1} पूरा कीजिए: सिद्ध कीजिए कि $f(x) = x^2
\sin\frac1x$, $f(0) = 0$, $0$ पर \omterm{def:b1:derivative:def}{अवकलनीय} है और $f'(0) = 0$, तथा यह कि $f'$
की $0$ पर कोई सीमा नहीं है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:3}
माध्य मान प्रमेय अथवा स्पर्श-असमिकाओं का प्रयोग करके सिद्ध कीजिए कि सभी $x >
0$ के लिए:
\[
\frac{x}{1 + x} < \ln(1 + x) < x .
\]
प्रत्येक $x > 0$ के लिए $\lim_{n\to\infty} \bigl(1 + \frac xn\bigr)^n =
\eu^x$ निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:4}
मान लीजिए $P$ ऐसा वास्तविक \omterm{def:b1:poly:def}{बहुपद} है जिसके $k$ भिन्न वास्तविक मूल हैं। सिद्ध
कीजिए कि $P'$ के कम से कम $k - 1$ भिन्न वास्तविक मूल हैं, जो $P$ के मूलों के
बीच-बीच में आते हैं। उससे निकालिए कि यदि $P$ के सभी मूल वास्तविक हैं, तो
$P'$ के भी।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:5}
मान लीजिए $f$ $\R$ पर \omterm{def:b1:derivative:def}{अवकलनीय} है और $x \to +\infty$ होने पर $f' (x)\to
\ell$। सिद्ध कीजिए कि $\frac{f(x)}{x} \to \ell$ \emph{($\intcc{A}{x}$ पर
माध्य मान प्रमेय)}। क्या $f(x+1) - f(x) \to \ell$ भी सत्य है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:6}
(एक विविक्त रोल) मान लीजिए $f$ $I$ पर $n$ बार \omterm{def:b1:derivative:def}{अवकलनीय} है और $n + 1$ भिन्न
बिंदुओं पर लुप्त होता है। सिद्ध कीजिए कि $f^{(n)}$ कम से कम एक बार लुप्त
होता है। अनुप्रयोग: $n+1$ बिंदुओं पर लुप्त होने वाला घात $\leq n$ का \omterm{def:b1:poly:def}{बहुपद}
शून्य है (फिर से)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:7}
मान लीजिए $f$ $\intcc{a}{b}$ पर दो बार \omterm{def:b1:derivative:def}{अवकलनीय} है, $f(a) = f(b) = 0$, और
किसी भीतरी $x_0$ के लिए $f(x_0) > 0$। सिद्ध कीजिए कि किसी $c \in
\intoo{a}{b}$ के लिए $f''(c) < 0$। \emph{(दो माध्य मान प्रमेय और प्रवणताओं
की तुलना।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:8}
$\intoo{0}{+\infty}$ पर फलन $f(x) = \dfrac{\ln x}{x}$ का अध्ययन कीजिए:
विचरण, सीमाएँ, अधिकतम। उससे निकालिए कि सभी वास्तविक $\eu \leq a < b$ के लिए
$a^b > b^a$, और प्रसिद्ध विशेष स्थिति तय कीजिए: $\eu^\pi$, $\pi^\eu$ में कौन
बड़ा है? छोटे पूर्णांक युग्मों $(2,3)$ और $(2,4)$ के सामने जाँचिए: वे भिन्न
व्यवहार क्यों करते हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:9}
(समांतर--गुणोत्तर असमिका) व्यापक भारों के साथ $\ln$ की अवतलता का प्रयोग करके
($n$ बिंदुओं के लिए येंसन असमिका, जिसे $n$ पर आगमन से सिद्ध करना है) दिखाइए
कि धनात्मक वास्तविक संख्याओं $a_1, \dots, a_n$ के लिए:
\[
\sqrt[n]{a_1 a_2 \cdots a_n} \leq \frac{a_1 + \dots + a_n}{n},
\]
जहाँ समता तभी जब सभी $a_i$ बराबर हों।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:10}
(डार्बू: \omterm{def:b1:derivative:def}{अवकलज} मध्यवर्ती मान लेते हैं) मान लीजिए $f$ $I$ पर \omterm{def:b1:derivative:def}{अवकलनीय} है और
$I$ में $a < b$ ऐसे हैं कि $f'(a) < v < f'(b)$। $g(x) = f(x) - vx$ पर और उस
बिंदु पर विचार करके जहाँ $g$ $\intcc{a}{b}$ पर अपना न्यूनतम प्राप्त करता है,
सिद्ध कीजिए कि किसी $c \in \intoo{a}{b}$ के लिए $f'(c) = v$ --- यद्यपि $f'$
का \omterm{def:b1:continuity:continuous}{संतत} होना आवश्यक नहीं (\cref{exo:b1:derivative:2})।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:11}
मान लीजिए $f \colon \R \to \R$ \omterm{def:b1:derivative:def}{अवकलनीय} है और सभी $x$ के लिए $\abs{f'(x)}
\leq k < 1$ (एक \emph{संकुचन})। सिद्ध कीजिए कि $f$ का ठीक एक अचल बिंदु
$\ell$ है, और यह कि प्रत्येक अनुक्रम $u_{n+1} = f(u_n)$ $\ell$ की ओर अभिसरित
होता है, जहाँ $\abs{u_n - \ell} \leq k^n\abs{u_0 - \ell}$। \emph{(अस्तित्व:
लिप्शिट्ज़ परिबंध का प्रयोग करते हुए किसी बड़े खंड पर $f(x) - x$ पर
मध्यवर्ती मान प्रमेय लगाइए; अथवा कोशी कसौटी के साथ पूर्णता का प्रयोग
कीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:12}
(कोशी की माध्य मान प्रमेय और लोपिताल का नियम)
\begin{enumerate}
  \item मान लीजिए $f, g$ $\intcc{a}{b}$ पर \omterm{def:b1:continuity:continuous}{संतत} और $\intoo{a}{b}$ पर \omterm{def:b1:derivative:def}{अवकलनीय}
  हैं, तथा वहाँ $g'$ कभी शून्य नहीं होता। सिद्ध कीजिए कि $g(b) \neq g(a)$ और
  यह कि कोई $c \in \intoo{a}{b}$ यह संतुष्ट करता है
        \[
        \frac{f(b) - f(a)}{g(b) - g(a)} = \frac{f'(c)}{g'(c)}
        \]
        \emph{(उपयुक्त अचर $\lambda$ के लिए $h = f - \lambda g$ पर रोल
        लगाइए)}।
  \item उससे किसी बिंदु पर $\frac00$ रूप में लोपिताल का नियम निकालिए: यदि $x
  \to a^+$ होने पर $f(a) = g(a) = 0$ और $\frac{f'(x)}{g'(x)} \to \ell$, तो
  $\frac{f(x)}{g(x)} \to \ell$।
  \item दिखाइए कि विलोम विफल है: $f(x) = x^2\sin\frac1x$ ($f(0) = 0$) और
  $g(x) = x$ के लिए भागफल $\frac{f}{g}$ की $0$ पर सीमा है पर $\frac{f'}{g'}$
  की कोई नहीं।
\end{enumerate}
\end{exercise}

\section{समस्या: लिउविल असमिका और पहली अबीजीय संख्या}

\begin{problem}[{सप्ताहांत समस्या --- बीजीय संख्याएँ परिमेय संख्याओं को दूर धकेलती हैं: $\abs{x - p/q} \geq C/q^d$,
और $\sum 10^{-n!}$ की अबीजीयता}]
\label{pb:b1:derivative:1}
कोई वास्तविक संख्या \emph{बीजीय} तब है जब वह पूर्णांक गुणांकों वाले किसी
अशून्य \omterm{def:b1:poly:def}{बहुपद} का मूल हो, और अन्यथा \emph{अबीजीय}। 1844 में लिउविल ने वह पहली
संख्या प्रस्तुत की जिसकी अबीजीयता कभी \emph{सिद्ध} हुई थी, और उसकी उपपत्ति
का इंजन इसी अध्याय की माध्य मान प्रमेय है: घात $d$ की \omterm{pb:b1:logic:1}{बीजीय संख्या} का परिमेय
संख्याओं से $C/q^d$ से बेहतर सन्निकटन नहीं हो सकता --- अतः जिस संख्या का
सन्निकटन \emph{हर घात से तेज़} हो सके वह बीजीय नहीं हो सकती। यह समस्या वह
असमिका खड़ी करती है, लिउविल की संख्या $L = 0.110001000\dots$ (क्रमगुणित
स्थानों पर इकाइयाँ, \cref{pb:b1:reals:1} की अंक-मशीनरी से) बनाती है, उसकी
अबीजीयता सिद्ध करती है, और कैंटर की प्रतिद्वंद्वी उपपत्ति तथा $\sqrt2$ और
$2^{1/3}$ के लिए प्रभावी परिबंधों पर समाप्त होती है।\index{अबीजीय
संख्या}\index{लिउविल असमिका}\index{बीजीय संख्या}

\medskip \noindent\textbf{भाग I --- परिमेय संख्याओं का सन्निकटन कितना अच्छा
हो सकता है?}
\begin{enumerate}
  \item दिखाइए कि दो भिन्न परिमेय $\frac ab \neq \frac pq$ ($b, q \geq 1$ के
  साथ लिखे हुए) $\bigl|\frac ab - \frac pq\bigr| \geq \frac{1}{bq}$ संतुष्ट
  करते हैं। उससे निकालिए: यदि $x = \frac ab$ और $0 < \bigl|x - \frac
  pq\bigr| < \frac{1}{bq}$, तो ऐसा कोई $\frac pq$ है ही नहीं --- अर्थात्
  परिमेय संख्या शेष सभी परिमेय संख्याओं को $\frac 1q$ के पैमाने पर दूर धकेल
  देती है।
  \item सिद्ध कीजिए कि \emph{प्रत्येक} परिमेय $\frac pq$ ($q \geq 1$) के लिए
  $\bigl|\sqrt2 - \frac pq\bigr| \geq \frac{1}{4q^2}$ \emph{(यदि दूरी $1$ से
  अधिक हो तो यह स्पष्ट है; अन्यथा $\abs{\sqrt2 + p/q} < 4$ को परिबद्ध कीजिए
  और अशून्य पूर्णांक $\abs{p^2 - 2q^2} \geq 1$ का प्रयोग कीजिए)}।
  \item दूसरी दिशा में: जाँचिए कि $(p, q) \mapsto (p + 2q, p + q)$ $\abs{p^2
  - 2q^2} = 1$ सुरक्षित रखता है, $(1,1)$ से युग्म $(3,2)$, $(7,5)$,
  $(17,12)$, $(41,29)$, $(99,70)$ उत्पन्न कीजिए, और दिखाइए कि हर एक यह
  संतुष्ट करता है
        \[
        \Bigl|\sqrt2 - \frac pq\Bigr| =
        \frac{1}{q^2\,(\sqrt2 + p/q)} < \frac{1}{2q^2} :
        \]
        अर्थात् कोटि $2$ के अनंत कितने सन्निकटन। प्रश्न 2 के साथ: $\sqrt 2$
        का सन्निकटन-घातांक \emph{ठीक} $2$ है।
  \item (डिरिक्ले) मान लीजिए $x$ अपरिमेय है और $N \in \N^*$। $N$ डिब्बों
  $\intco{\frac kN}{\frac{k + 1}{N}}$ में $0, x, 2x, \dots, Nx$ के $N + 1$
  भिन्नात्मक भागों पर विचार कीजिए: कबूतरखाना सिद्धांत
  (\cref{cor:b1:counting:pigeonhole}) से दो एक ही डिब्बे में गिरते हैं। उससे
  $\abs{qx - p} < \frac 1N$ के साथ $q \leq N$ और $p$ निकालिए, अतः $\bigl|x -
  \frac pq\bigr| < \frac{1}{q^2}$ वाली अनंत कितनी परिमेय संख्याएँ: अर्थात्
  \emph{प्रत्येक} अपरिमेय संख्या का कोटि $2$ तक सन्निकटन हो सकता है।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- लिउविल असमिका।} मान लीजिए $x$ अपरिमेय
और बीजीय है।
\begin{enumerate}[resume]
  \item दिखाइए कि $x$ पर लुप्त होने वाले अशून्य पूर्णांक बहुपदों में एक ऐसा
  है, मान लीजिए घात $d$ का $P$, जिसका \emph{कोई परिमेय मूल नहीं}; और $d \geq
  2$ जाँचिए \emph{($\Q$ पर गुणनखंड $X - \frac ab$ भाग दीजिए और हर हटाइए; घात
  $1$ $x$ को परिमेय बना देती)}।
  \item दिखाइए कि प्रत्येक परिमेय $\frac pq$ ($q \geq 1$) के लिए
  $\bigl|P\bigl(\frac pq\bigr)\bigr| \geq \frac{1}{q^d}$ \emph{($q^d P(p/q)$
  अशून्य पूर्णांक है)}।
  \item मान लीजिए $M = \max_{\intcc{x-1}{x+1}} \abs{P'}$
  (\cref{thm:b1:continuity:evt})। $x$ और $\frac pq$ के बीच माध्य मान प्रमेय
  का प्रयोग करके \emph{लिउविल असमिका} सिद्ध कीजिए: $C = \min\bigl(1, \frac
  1M\bigr) > 0$ के साथ,
        \[
        \Bigl| x - \frac pq \Bigr| \geq \frac{C}{q^{\,d}}
        \qquad\text{प्रत्येक परिमेय } \frac pq,\ q \geq 1 .
        \]
  \item $x$ को \emph{लिउविल संख्या} तब कहिए जब प्रत्येक $n \in \N$ के लिए
  ऐसी परिमेय $\frac pq$ हो कि $q \geq 2$ और $0 < \bigl|x - \frac pq\bigr| <
  q^{-n}$। सिद्ध कीजिए कि लिउविल संख्या अपरिमेय होती है \emph{(प्रश्न 1: ऐसा
  $n$ चुनिए कि $2^{\,n-1} > b$)}।
  \item लिउविल की प्रमेय सिद्ध कीजिए: \emph{लिउविल संख्या अबीजीय होती है}
  \emph{(प्रश्न 7 और 8 मिलाइए: बड़े $n$ के लिए असमिका $C < q^{\,d-n}$ विफल
  हो जाती है)}।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- संख्या $L$।}
\begin{enumerate}[resume]
  \item मान लीजिए $L$ उस दशमलव अंक-माला का मान (\cref{pb:b1:reals:1} के अर्थ
  में) है जिसमें स्थानों $n!$ ($n = 1, 2, 3, \dots$) पर अंक $1$ है और
  अन्यत्र $0$, अर्थात् $t_k = \sum_{n=1}^{k} 10^{-n!}$ के साथ $L = \sup_k
  t_k$। $L$ के पहले $25$ अंक लिखिए।
  \item प्रत्येक $k \geq 1$ के लिए पूँछ का घेराव सिद्ध कीजिए:
        \[
        10^{-(k+1)!} \;\leq\; L - t_k \;\leq\;
        \frac{10}{9}\,10^{-(k+1)!} \;<\; 2\cdot 10^{-(k+1)!}
        \]
        \emph{($t_k$ से आगे हर आंशिक योग को किसी परिमित गुणोत्तर योग से
        परिबद्ध कीजिए)}।
  \item $q_k = 10^{k!}$ के साथ $t_k = \frac{p_k}{q_k}$ लिखिए। दिखाइए $0 < L
  - \frac{p_k}{q_k} < \frac{2}{q_k^{\,k+1}}$, और निष्कर्ष निकालिए कि $L$
  प्रश्न 8 के अर्थ में लिउविल संख्या है।
  \item निष्कर्ष: $L$ अबीजीय है --- इतिहास का पहला स्पष्ट उदाहरण (लिउविल,
  1844)। उसकी अपरिमेयता सीधे भी जाँचिए: उसके अंक अंततः आवर्ती नहीं हैं
  (बढ़ते \omterm{prop:b1:reals:intervals}{अंतराल}, जैसा \cref{pb:b1:reals:1}, प्रश्न 20 में है)।
  \item सामान्यीकृत कीजिए: प्रत्येक अंक $1$ के स्थान पर कोई स्वेच्छ अशून्य
  अंक $d_n \in \intint{1}{9}$ रखिए। दिखाइए कि मान अब भी लिउविल संख्या है, और
  उससे --- इन अंक-चयनों पर \cref{pb:b1:reals:1} (प्रश्न 22) का विकर्ण तर्क
  लगाकर --- निकालिए कि इस आकार की अगणनीय कितनी अबीजीय संख्याएँ हैं।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- सन्निकटन-कोटियों की सोपान-रचना।} $x$ को
\emph{कोटि $\mu$ तक सन्निकटनीय} तब कहिए जब किसी अचर $c > 0$ के लिए अनंत
कितनी परिमेय संख्याएँ $\bigl|x - \frac pq\bigr| < \frac{c}{q^{\mu}}$ संतुष्ट
करें।
\begin{enumerate}[resume]
  \item भाग I से III तक की सोपान-रचना जोड़िए: परिमेय संख्याएँ कोटि $1$ तक
  सन्निकटनीय हैं और उससे बेहतर नहीं; $\sqrt 2$ कोटि $2$ तक और उससे बेहतर
  नहीं; प्रत्येक अपरिमेय संख्या कम से कम कोटि $2$ तक; घात $d$ की \omterm{pb:b1:logic:1}{बीजीय
  संख्या} $d$ से आगे किसी कोटि तक नहीं; और लिउविल संख्याएँ हर कोटि तक।
  प्रत्येक दावे को संबंधित प्रश्न का हवाला देकर न्यायसंगत ठहराइए।
  \item दिखाइए कि प्रत्येक परिमेय $r = \frac ab$ के लिए $L + r$ लिउविल
  संख्या है \emph{(सन्निकटकों का स्थानांतरण कीजिए: नए हर $b\,q_k$ हैं)}।
  निष्कर्ष निकालिए कि लिउविल --- और इसलिए अबीजीय --- संख्याएँ $\R$ में \omterm{def:b1:topology:dense}{सघन}
  हैं।
  \item (कैंटर, 1874) सिद्ध कीजिए कि बीजीय संख्याओं का \omterm{def:b1:logic:sets}{समुच्चय} गणनीय है: ऐसे
  पूर्णांक \omterm{def:b1:poly:def}{बहुपद} परिमित कितने हैं जिनकी घात और $\abs{\text{गुणांकों}}$ का
  योग मिलकर $h$ से परिबद्ध हों, और हर एक के अधिक से अधिक $\deg$ मूल हैं; और
  परिमित समुच्चयों का गणनीय सम्मिलन गणनीय होता है। चूँकि कोई अनुक्रम $\R$ को
  नहीं भरता (\cref{pb:b1:reals:1}, प्रश्न 22), अतः अबीजीय संख्याएँ विद्यमान
  हैं --- वस्तुतः वे अगणनीय \omterm{def:b1:logic:sets}{समुच्चय} बनाती हैं। दोनों उपपत्तियों की तुलना
  कीजिए: लिउविल की उपपत्ति क्या देती है जो कैंटर की नहीं दे सकती?
  \item प्रश्न 2 से सीधे सिद्ध कीजिए कि $\sqrt 2$ लिउविल संख्या \emph{नहीं}
  है \emph{($n \geq 3$ के लिए असमिका $q^{-n} > \frac{1}{4q^2}$ $q$ को
  परिबद्ध कर देती है; तब केवल परिमित कितने उम्मीदवार परिमेय बचते हैं, और वे
  सब $\sqrt2$ से धनात्मक दूरी पर हैं)}। सामान्यीकृत कीजिए: कोई भी \omterm{pb:b1:logic:1}{बीजीय
  संख्या} लिउविल नहीं है।
\end{enumerate}

\medskip \noindent\textbf{भाग V --- प्रभावी अचर।}
\begin{enumerate}[resume]
  \item पेल युग्म $(99, 70)$ के लिए: $99^2 - 2\cdot70^2 = 1$ सत्यापित कीजिए
  और यथार्थ त्रुटि निकालिए
        \[
        \sqrt2 - \frac{99}{70}
        = \frac{-1}{70^2\,\bigl(\sqrt2 + \frac{99}{70}\bigr)},
        \qquad
        \Bigl|\sqrt 2 - \frac{99}{70}\Bigr| \approx 7.2\cdot
        10^{-5} :
        \]
        अर्थात् तीन अंकों की भिन्न से पाँच सही अंक।
  \item $x = 2^{1/3}$, $P = X^3 - 2$ पर भाग II चलाइए: जाँचिए कि $P$ का कोई
  परिमेय मूल नहीं है, $M = \max_{\intcc{x-1}{x+1}} 3t^2 \leq 3\,(1 +
  2^{1/3})^2 < 16$ को परिबद्ध कीजिए, और प्रभावी असमिका पर पहुँचिए
        \[
        \Bigl| 2^{1/3} - \frac pq \Bigr| \geq
        \frac{1}{16\,q^3} \qquad \text{सभी } \frac pq .
        \]
  \item लाभ: दिखाइए कि $2^{1/3}$ का $10^{-6}$ के भीतर सन्निकटन करने वाली
  किसी भी परिमेय संख्या का हर $q \geq 40$ होना चाहिए।
  \item दिखाइए कि आधार $10$ निरर्थक है: द्विआधारी अनुरूप $\sum_{n\geq1}
  2^{-n!}$ (क्रमगुणित स्थानों पर इकाइयों वाली द्विआधारी माला का मान) भी
  लिउविल संख्या है, अतः अबीजीय।
\end{enumerate}

\medskip \noindent\textbf{भाग VI --- सीमांत और संश्लेषण।}
\begin{enumerate}[resume]
  \item मान लीजिए $x^\dagger$ उस दशमलव माला का मान है जिसमें ठीक स्थानों
  $3^k$ ($k \geq 0$) पर इकाइयाँ हैं। दिखाइए कि $x^\dagger$ कोटि $3$ तक
  सन्निकटनीय है, और लिउविल असमिका से निकालिए कि $x^\dagger$ न परिमेय है और न
  कोई द्विघात अपरिमेय। समझाइए कि विधि वहीं क्यों अटक जाती है: कोटि $3$ घात
  $\geq 3$ की बीजीयता के अनुरूप है, और उस अंतर को पाटना (प्रत्येक \omterm{pb:b1:logic:1}{बीजीय
  संख्या} के लिए कोई भी घातांक $> 2$ पर्याप्त है) रोथ की प्रमेय है, जो इस खंड
  से बहुत परे है।
  \item कैंटर को परिमाणित कीजिए: दिखाइए कि $\intint{-H}{H}$ में गुणांकों
  वाले बहुपदों से मिलने वाली घात $\leq d$ की बीजीय संख्याएँ अधिक से अधिक
  $d\,(2H + 1)^{d+1}$ हैं। (यही परिमितता प्रश्न 17 को काम करने देती है।)
  \item संश्लेषण, एक-एक वाक्य में: (क) लिउविल की उपपत्ति की एकमात्र
  विश्लेषणात्मक सामग्री बताइए (इस अध्याय की कौन-सी प्रमेय, कहाँ प्रयुक्त);
  (ख) उसे चलाने वाला तनाव कहिए (पूर्णांकता $\abs{P(p/q)} \geq q^{-d}$ पर
  बाध्य करती है, और चिकनाई $\abs{P(p/q)} > M\abs{x - p/q}$ को वर्जित करती
  है); (ग) अबीजीय संख्याओं के अस्तित्व की लिउविल और कैंटर की उपपत्तियों की
  तुलना कीजिए; (घ) बताइए कि इस खंड में यही विषय फिर कहाँ मिलता है ---
  \cref{ch:b1:integration} की सप्ताहांत समस्या $\pi$ की अपरिमेयता उसी
  पूर्णांकता-बनाम-लघुता वाले दबाव से सिद्ध करती है, बस अवकलजों के स्थान पर
  समाकल रखकर।
\end{enumerate}
\end{problem}
