\chapter{मानक फलन}\label{ch:b1:functions}

विश्लेषण उतना ही उपयोगी है जितना फलनों का वह भंडार जिस पर अधिकार हो।
माध्यमिक विद्यालय से मिले संग्रह --- घात, चरघातांकी, लघुगणक, त्रिकोणमितीय
फलन --- में यह अध्याय उनके प्रतिलोम फलन ($\arcsin$, $\arccos$, $\arctan$) और
अतिपरवलयिक कुल जोड़ता है। अवकलजों का प्रयोग उच्चतर माध्यमिक स्तर पर स्वतंत्र
रूप से किया गया है; प्रतिलोम फलनों के पीछे का सिद्धांत
\cref{ch:b1:continuity,ch:b1:derivative} में पूरा होता है।

\section{चरघातांकी, लघुगणक, घातें}

\begin{proposition}[पुनरावलोकन और अभिलक्षण]\label{prop:b1:functions:expln}
$\exp \colon \R \to \intoo{0}{+\infty}$ और $\ln \colon \intoo{0}{+\infty}
\to \R$ परस्पर प्रतिलोम एकैकी आच्छादन हैं, कठोरतः वर्धमान, जहाँ
\[
\exp(x + y) = \exp x \exp y, \qquad \ln(xy) = \ln x + \ln y,
\qquad \exp' = \exp, \qquad \ln'(x) = \frac 1x .
\]
इसके अतिरिक्त $\exp$ \emph{एकमात्र} अवकलनीय फलन $f \colon \R \to \R$ है
जिसके लिए $f' = f$ और $f(0) = 1$।
\end{proposition}

\begin{proof}
पुनरावलोकन उच्चतर माध्यमिक सामग्री है। अद्वितीयता के लिए मान लीजिए $f' = f$,
$f(0) = 1$, और $g(x) = f(x)\,\eu^{-x}$ रखिए। तब $g' = f'\eu^{-x} - f\eu^{-x}
= 0$, अतः $g$ $g(0) = 1$ के बराबर अचर है: $f = \exp$।
\end{proof}

\begin{definition}[व्यापक घातें]\label{def:b1:functions:powers}
$x > 0$ और $\alpha \in \R$ के लिए: $\;x^\alpha = \eu^{\alpha \ln
x}$\index{घात फलन}। $a > 0$, $a \neq 1$ के लिए \emph{आधार $a$ का लघुगणक}
$\log_a x = \frac{\ln x}{\ln a}$ है, जो $x \mapsto a^x$ का प्रतिलोम है।
\end{definition}

\begin{example}[चरघातांकी समीकरण हल करना]\label{ex:b1:functions:expequations}
$\R$ में $2^x = 5^{\,x-1}$ हल कीजिए। दोनों पक्ष धनात्मक हैं, अतः लघुगणक
लीजिए --- यह पद उत्क्रमणीय है:
\[
x\ln2 = (x - 1)\ln5
\iff x(\ln2 - \ln5) = -\ln5
\iff x = \frac{\ln5}{\ln5 - \ln2} = \frac{\ln 5}{\ln\frac52}
\approx 1.756 .
\]
अब $x > 0$ के लिए $x^{\sqrt2} = 3$ हल कीजिए: घात $\frac1{\sqrt2}$ तक उठाइए
(अर्थात् परस्पर प्रतिलोम एकैकी आच्छादन लगाइए): $x = 3^{1/\sqrt2} = \eu^{(\ln
3)/\sqrt2} \approx 2.175$। सार: घातों को मिलाने वाला हर समीकरण $\ln$ और
$\exp$ से होकर खुलता है, क्योंकि परिभाषा $x^\alpha = \eu^{\alpha\ln x}$
घातों की सारी हेराफेरी को घातांकों के अंकगणित तक सीमित कर देती है --- पर
केवल उसी प्रांत $x > 0$ पर जहाँ वह परिभाषा रहती है।
\end{example}

\begin{example}[द्विगुणन काल]\label{ex:b1:functions:doubling}
कोई राशि प्रति पद $3\%$ से बढ़ती है: $n$ पदों के बाद वह $(1.03)^n$ से गुणित
हो जाती है। वह दुगुनी कब होगी? $(1.03)^n \geq 2$ हल कीजिए:
\[
n\ln(1.03) \geq \ln2
\iff n \geq \frac{\ln 2}{\ln 1.03}
= \frac{0.6931}{0.02956} \approx 23.4 ,
\]
अतः पहला द्विगुणन पद $24$ पर होता है। (वित्तज्ञों का ``$72$ का नियम'', जो
द्विगुणन काल का आकलन $72$ को प्रतिशत में दर से भाग देकर करता है, यही संगणना
है, बस सन्निकटन $\ln(1 + x) \approx x$ के साथ, जिसका परिमाणन
\cref{ch:b1:taylor} में है।) चरघातांकी प्रक्रियाओं पर विचार उनके लघुगणकों के
द्वारा करना सबसे अच्छा रहता है: उस पैमाने पर वृद्धि रैखिक हो जाती है और
प्रश्न भाग बन जाते हैं।
\end{example}

\begin{example}[$2^{2026}$ कितना लंबा है?]\label{ex:b1:functions:digits}
पूर्णांक $N \geq 1$ के दशमलव अंकों की संख्या $\floor{\log_{10} N} + 1$ है
(वस्तुतः $N$ के ठीक $d$ अंक तब होते हैं जब $10^{d-1} \leq N < 10^d$, अर्थात्
$d - 1 \leq \log_{10}N < d$)। $N = 2^{2026}$ के लिए:
\[
\log_{10} 2^{2026} = 2026\,\log_{10}2
= 2026 \times 0.301030 = 609.887 ,
\]
अतः $2^{2026}$ के $610$ अंक हैं। भिन्नात्मक भाग एक बोनस भी लाता है:
$10^{0.887} \approx 7.7$, अतः यह संख्या $7$ से \emph{आरंभ} होती है। एक गुणा
ने ऐसी संख्या के विषय में प्रश्न का उत्तर दे दिया जिसे कोई कभी पूरा लिखेगा
भी नहीं --- लघुगणक गुणात्मक आकार को योगात्मक आकार में सिकोड़ देते हैं, जो
उनका पूरा ऐतिहासिक प्रयोजन है (\cref{ex:b1:structures:expmorphism} इस वाक्य
को ठीक-ठीक कह देता है)।
\end{example}

\begin{proposition}[घात के नियम और वृद्धि की तुलना]\label{prop:b1:functions:powerrules}
$x, y > 0$ और $\alpha, \beta \in \R$ के लिए:
\[
x^{\alpha+\beta} = x^\alpha x^\beta, \quad
(x^\alpha)^\beta = x^{\alpha\beta}, \quad
(xy)^\alpha = x^\alpha y^\alpha, \quad
(x^\alpha)' = \alpha\, x^{\alpha - 1}.
\]
वृद्धि का पैमाना\index{वृद्धि तुलना}: जब $x \to +\infty$, तब प्रत्येक
$\alpha > 0$ और $\beta > 0$ के लिए
\[
\frac{(\ln x)^\beta}{x^\alpha} \longrightarrow 0,
\qquad
\frac{x^\alpha}{\eu^{\beta x}} \longrightarrow 0 .
\]
\end{proposition}

\begin{proof}
ये सर्वसमिकाएँ परिभाषा के द्वारा $\exp$ और $\ln$ की सर्वसमिकाओं का लिप्यंतरण
हैं। दूसरी (जो सबसे कम स्पष्ट है) के लिए विस्तार से: $x^\alpha > 0$ और
$\ln(x^\alpha) = \alpha\ln x$ (परिभाषा पर $\ln$ लगाइए), अतः
\[
(x^\alpha)^\beta = \eu^{\beta\ln(x^\alpha)}
= \eu^{\beta\alpha\ln x} = x^{\alpha\beta} ;
\]
पहली और तीसरी $\exp(u + v) = \exp u\exp v$ तथा $\ln(xy) = \ln x + \ln y$ के
द्वारा उसी प्रकार के दो-पंक्ति के उद्घाटन हैं। अवकलज शृंखला नियम है:
$(\eu^{\alpha\ln x})' = \frac{\alpha}{x} \eu^{\alpha\ln x}$।

तुलनाएँ: $\frac{\ln t}{t} \to 0$ से (जो उच्चतर माध्यमिक खंड में सिद्ध है) $t
= x^{\alpha/\beta}$ प्रतिस्थापित कीजिए: $\frac{\ln x}{x^{\alpha/\beta}} \to
0$, फिर घात $\beta$ तक उठाइए। दूसरी के लिए पहली में $x = \ln u$ प्रतिस्थापित
कीजिए।
\end{proof}

\begin{example}[दो सीमाएँ जो हर पाठक की अपनी होनी चाहिए]\label{ex:b1:functions:xx}
$\lim_{x \to 0^+} x^x$ और $\lim_{x \to +\infty} x^{1/x}$ क्या हैं? दोनों
घातें चरघातांकी के द्वारा परिभाषित हैं, अतः पहले घातांक तय कीजिए:
\[
x^x = \eu^{x\ln x}, \qquad
x\ln x = -\frac{\ln(1/x)}{1/x} \longrightarrow 0
\quad (x \to 0^+),
\qquad\text{अतः } x^x \longrightarrow \eu^0 = 1 ;
\]
और जब $x \to +\infty$, तब सीधे वृद्धि की तुलना से $x^{1/x} = \eu^{(\ln x)/x}
\to \eu^0 = 1$। सार: अनिश्चित घात ($0^0$ या $\infty^0$ के आकार की)
\emph{सदा} $u^v = \eu^{v\ln u}$ लिखकर और गुणनफल $v \ln u$ का विश्लेषण करके
सँभाली जाती है --- कभी आधार और घातांक को अलग-अलग देखकर अनुमान लगाकर नहीं।
जिस फलन $x \mapsto \frac{\ln x}{x}$ ने दोनों सीमाएँ तय कीं, उसका संपूर्ण
अध्ययन इस अध्याय की सप्ताहांत समस्या में है।
\end{example}

\begin{remark}[घातों और लघुगणकों की सामान्य भूलें]\label{rem:b1:functions:pitfalls}
\begin{enumerate}
  \item \emph{प्रांत।} अपरिमेय $\alpha$ के लिए $x^\alpha$ को $x > 0$ चाहिए;
  $(-8)^{1/3}$ से बचकर ``$-8$ का वास्तविक घनमूल'' कहना बेहतर है, क्योंकि
  नियम $(x^\alpha)^\beta = x^{\alpha\beta}$ ऋणात्मक संख्याओं पर चुपचाप विफल
  हो जाता है: $\bigl((-8)^2\bigr)^{1/6} = 2 \neq -2$।
  \item \emph{$\sqrt{x^2} = \abs x$}, $x$ नहीं: निरपेक्ष मान भूल जाना
  ऋणात्मक हल खो देने का चिरपरिचित कारण है।
  \item \emph{$\ln(xy) = \ln x + \ln y$ के लिए $x, y > 0$ आवश्यक है।}
  ऋणात्मक प्रांतिकों के लिए $\ln(xy)$ परिभाषित किया जा सकता है, जबकि दायाँ
  पक्ष नहीं।
  \item \emph{कौन-सा फलन बढ़ता है?} $x^\alpha$ में \emph{आधार} बदलता है;
  $a^x$ में \emph{घातांक}। उनकी वृद्धियाँ अत्यंत भिन्न हैं
  (\cref{prop:b1:functions:powerrules}), और $x^{\ln x}$ या $x^{1/x}$ जैसे
  मिश्रित व्यंजकों को किसी भी तर्क से पहले $\eu^{(\cdot)}$ के रूप में फिर से
  लिखना चाहिए --- जैसा \cref{ex:b1:functions:xx} में है।
\end{enumerate}
\end{remark}

\section{प्रतिलोम त्रिकोणमितीय फलन}

\begin{definition}[$\arcsin$, $\arccos$, $\arctan$]\label{def:b1:functions:arc}
प्रतिबंधन
\[
\sin \colon \intcc{-\tfrac\pi2}{\tfrac\pi2} \to \intcc{-1}{1},
\qquad
\cos \colon \intcc{0}{\pi} \to \intcc{-1}{1},
\qquad
\tan \colon \intoo{-\tfrac\pi2}{\tfrac\pi2} \to \R
\]
कठोरतः एकदिष्ट एकैकी आच्छादन हैं। उनके प्रतिलोम \omterm{def:b1:logic:map}{प्रतिचित्रण}
$\arcsin$\index{प्रतिलोम ज्या}, $\arccos$\index{प्रतिलोम कोज्या} और
$\arctan$\index{प्रतिलोम स्पर्शज्या} लिखे जाते हैं। अतः उदाहरणार्थ $y =
\arcsin x$ ($x \in \intcc{-1}{1}$) $\intcc{-\frac\pi2}{\frac\pi2}$ का
\emph{वही} कोण है जिसकी ज्या $x$ है।
\end{definition}

\begin{proposition}[अवकलज]\label{prop:b1:functions:arcderiv}
अपने प्रांतों के अंतःभाग पर:
\[
\arcsin' x = \frac{1}{\sqrt{1 - x^2}},
\qquad
\arccos' x = \frac{-1}{\sqrt{1 - x^2}},
\qquad
\arctan' x = \frac{1}{1 + x^2}.
\]
\end{proposition}

\begin{proof}
\cref{ch:b1:derivative} में सिद्ध होने वाले प्रतिलोम-फलन नियम का पूर्वानुमान
करते हुए: यदि $f$ अवकलनीय एकैकी आच्छादन है और $f' \neq 0$, तो $(f^{-1})'(x)
= \frac{1}{f'(f^{-1}(x))}$। $\arcsin$ के लिए: $y = \arcsin x$ के साथ,
\[
\arcsin' x = \frac{1}{\cos y}
= \frac{1}{\sqrt{1 - \sin^2 y}} = \frac{1}{\sqrt{1 - x^2}},
\]
जहाँ $\cos y = +\sqrt{1 - \sin^2 y}$, क्योंकि $y \in
\intoo{-\frac\pi2}{\frac\pi2}$ से $\cos y > 0$ आ जाता है। इसी प्रकार
$\intoo{0}{\pi}$ पर $\sin y > 0$ के साथ $\arccos' x = \frac{1}{-\sin y}$, और
$\tan' = 1 + \tan^2$ का प्रयोग करते हुए $\arctan' x = \frac{1}{1 + \tan^2 y}
= \frac{1}{1 + x^2}$।
\end{proof}

\begin{example}[एक छिपे हुए अचर को पहचानना]\label{ex:b1:functions:hiddenconst}
$\intoo{-1} {+\infty}$ पर $g(x) = \arctan\dfrac{1 - x}{1 + x}$ का अध्ययन
कीजिए। शृंखला नियम और एक छोटी संगणना:
\[
g'(x) = \frac{1}{1 + \bigl(\frac{1-x}{1+x}\bigr)^2}\cdot
\frac{-(1+x) - (1-x)}{(1+x)^2}
= \frac{-2}{(1+x)^2 + (1-x)^2}
= \frac{-1}{1 + x^2} ,
\]
क्योंकि $(1+x)^2 + (1-x)^2 = 2 + 2x^2$। अतः $g' = -\arctan'$: फलन $g +
\arctan$ का अवकलज \emph{अंतराल} $\intoo{-1}{+\infty}$ पर शून्य है, अतः वह
वहाँ अचर है; $x = 0$ पर उसका मान $\arctan 1 + \arctan 0 = \frac\pi4$ है।
निष्कर्ष:
\[
\arctan\frac{1 - x}{1 + x} = \frac\pi4 - \arctan x
\qquad (x > -1) .
\]
$\intoo{-\infty}{-1}$ पर वही अवकलज संगणना सत्य है, पर अचर भिन्न है
($-\frac{3\pi}4$: $x \to -\infty$ पर सीमा का मान लीजिए)। सार: ``शून्य अवकलज
का अर्थ अचर'' अंतराल-दर-अंतराल \omterm{def:b1:logic:statement}{कथन} है --- ठीक वही सूक्ष्मता जिसका उपयोग
\cref{exo:b1:functions:6} और \cref{exo:b1:functions:10} में होता है।
\end{example}

\begin{proposition}[मानक सर्वसमिकाएँ]\label{prop:b1:functions:arcidentities}
\begin{enumerate}
  \item $x \in \intcc{-1}{1}$ के लिए: $\arcsin x + \arccos x =
  \dfrac{\pi}{2}$।
  \item $x > 0$ के लिए: $\arctan x + \arctan\dfrac 1x = \dfrac{\pi}{2}$ (और
  $x < 0$ के लिए $-\frac\pi2$)।
  \item $\sin(\arccos x) = \cos(\arcsin x) = \sqrt{1 - x^2}$; $\abs x < 1$
  के लिए $\;\tan(\arcsin x) = \dfrac{x}{\sqrt{1 - x^2}}$।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) $x \mapsto \arcsin x + \arccos x$ का अवकलज $\intoo{-1}{1}$ पर शून्य है
(\cref{prop:b1:functions:arcderiv}), अतः फलन वहाँ अचर है और $0$ पर अपने मान
$\frac\pi2$ के बराबर है; सिरों के मान $\pm 1$ सीधे जाँचे जाते हैं
($\frac\pi2 + 0$ और $-\frac\pi2 + \pi$)।

(2) $\intoo{0}{+\infty}$ पर वही विधि: अवकलज $\frac{1}{1+x^2} +
\frac{-1/x^2}{1 + 1/x^2} = \frac{1}{1+x^2} - \frac{1}{x^2 + 1} = 0$ है, और
$x = 1$ पर योग $2\arctan 1 = \frac\pi2$ है। $x < 0$ के लिए $\arctan$ की
विषमता का प्रयोग कीजिए।

(3) $y = \arccos x \in \intcc{0}{\pi}$ के साथ: $\sin y \geq 0$, अतः $\sin y
= \sqrt{1 - \cos^2 y} = \sqrt{1 - x^2}$; इसी प्रकार $\cos(\arcsin x)$ के लिए
भी, और स्पर्शज्या का सूत्र इनका भागफल है।
\end{proof}

\begin{remark}
$\arcsin(\sin\theta) = \theta$ \emph{केवल} $\theta \in
\intcc{-\frac\pi2}{\frac\pi2}$ के लिए सत्य है: उदाहरणार्थ $\arcsin(\sin\pi)
= 0 \neq \pi$। दूसरी दिशा का संयोजन, $\sin(\arcsin x) = x$, पूरे
$\intcc{-1}{1}$ पर वैध है।
\end{remark}

\begin{example}[मुख्य अंतराल पर लौटना]
\label{ex:b1:functions:wrap}
संगणना कीजिए
\[
\arctan\Bigl(\tan\frac{3\pi}4\Bigr)
\qquad\text{और}\qquad
\arctan\Bigl(\tan\frac{17\pi}5\Bigr).
\]
विधि: कोण के स्थान पर $\intoo{-\frac\pi2}{\frac\pi2}$ का वही अद्वितीय कोण
रखिए जिसकी स्पर्शज्या वही हो, अर्थात् $\pi$ ($\tan$ का आवर्तकाल) का उचित
गुणज घटा दीजिए। पहला: $\frac{3\pi}4 - \pi = -\frac\pi4$, अतः उत्तर
$-\frac\pi4$ है। दूसरा: $\frac{17\pi} 5 - 3\pi = \frac{2\pi}5 \in
\intoo{-\frac\pi2}{\frac\pi2}$, अतः उत्तर $\frac{2\pi}5$ है। यह संगणना वेश
बदले हुए $\pi$ से कोण का यूक्लिडीय भाग ही है --- और $\arcsin$
($\intcc{-\frac\pi2}{\frac\pi2}$ में परावर्तित, आवर्तकाल $2\pi$) तथा
$\arccos$ ($\intcc0\pi$ में परावर्तित) के लिए अनुरूप विधियाँ
\cref{exo:b1:functions:1} और \cref{exo:b1:functions:10} के खंडशः उत्तर को
चलाती हैं।
\end{example}

\begin{example}[प्रतिलोम स्पर्शज्याओं का सुरक्षित योग]\label{ex:b1:functions:arctansum}
आइए सिद्ध करें
\[
\arctan\frac12 + \arctan\frac15 + \arctan\frac18 = \frac\pi4 .
\]
दो सामग्रियाँ: स्पर्शज्या का योग-सूत्र, और --- वह पद जो आरंभिक विद्यार्थी
भूल जाते हैं --- योग का \emph{स्थानीकरण}। पहले, $u = \arctan\frac12$, $v =
\arctan\frac15$ के साथ $\tan(u + v) = \frac{\tan u + \tan v}{1 - \tan u\tan
v}$ देता है
\[
\tan(u + v) = \frac{\frac12 + \frac15}{1 - \frac1{10}}
= \frac{7/10}{9/10} = \frac79,
\qquad\text{फिर}\qquad
\tan\Bigl(u + v + \arctan\frac18\Bigr)
= \frac{\frac79 + \frac18}{1 - \frac7{72}}
= \frac{65/72}{65/72} = 1 .
\]
दूसरे, तीनों कोणों में से प्रत्येक $\intoo0{\frac\pi4}$ में है (उनके
प्रांतिक $1$ से कम हैं), अतः योग $\intoo0{\frac{3\pi}4}$ में है; वहाँ
स्पर्शज्या $1$ वाला एकमात्र कोण $\frac\pi4$ है। स्थानीकरण के बिना निष्कर्ष
``$\pi$ के गुणज तक $\frac\pi4$'' होता --- अर्थात् आधी उपपत्ति। यही दो-पदीय
अनुशासन \cref{exo:b1:functions:8} और \cref{exo:b1:functions:12} को चलाता है।
\end{example}

\begin{example}[प्रतिलोम स्पर्शज्या वाला एक समीकरण]\label{ex:b1:functions:arctaneq}
$\arctan x + \arctan 2x = \dfrac\pi4$ हल कीजिए। पहले स्थानीकरण: बाएँ पक्ष का
चिह्न $x$ का चिह्न है (दोनों पदों का यही है), अतः किसी भी हल के लिए $x > 0$,
और तब योग $\intoo0\pi$ में है। स्पर्शज्या लीजिए ($\pi$ लंबाई के किसी भी
अंतराल पर वह \omterm{def:b1:logic:inj}{एकैकी} नहीं है, पर स्थानीकरण के साथ मिलकर यह पर्याप्त होगा):
योग-सूत्र देता है
\[
\tan\bigl(\arctan x + \arctan 2x\bigr)
= \frac{3x}{1 - 2x^2} = 1
\iff 2x^2 + 3x - 1 = 0
\iff x = \frac{-3 \pm \sqrt{17}}4 .
\]
ऋणात्मक मूल स्थानीकरण से हटा दिया जाता है; धनात्मक उम्मीदवार $x_0 =
\frac{\sqrt{17} - 3}4 \approx 0.28$ के लिए योग $\intoo0\pi$ में है और उसकी
स्पर्शज्या $1$ है, और ऐसा एकमात्र कोण $\frac\pi4$ है ($\intoo{\frac\pi2}\pi$
में स्पर्शज्या ऋणात्मक है): $x_0$ अद्वितीय हल है। तर्क के आकार पर ध्यान
दीजिए: स्पर्शज्या लेने से हल बन सकते हैं, खो कभी नहीं सकते, अतः पहले बहुपदीय
समीकरण हल कीजिए \emph{और फिर} स्थानीकरण से छानिए --- वही आगे-जाँच का अनुशासन
जो समीकरण का वर्ग करने में लगता है।
\end{example}

\section{अतिपरवलयिक फलन}

\begin{definition}[$\cosh$, $\sinh$, $\tanh$]\label{def:b1:functions:hyperbolic}
$x \in \R$ के लिए:
\[
\cosh x = \frac{\eu^x + \eu^{-x}}{2},
\qquad
\sinh x = \frac{\eu^x - \eu^{-x}}{2},
\qquad
\tanh x = \frac{\sinh x}{\cosh x}
\]
(\emph{अतिपरवलयिक कोज्या, ज्या, स्पर्शज्या}\index{अतिपरवलयिक फलन})। $\cosh$
सम है, $\sinh$ और $\tanh$ विषम हैं।
\end{definition}

\begin{proposition}[मूल गुणधर्म]\label{prop:b1:functions:hyprules}
सभी $x, y \in \R$ के लिए:
\begin{enumerate}
  \item $\cosh^2 x - \sinh^2 x = 1$;
  \item $\cosh' = \sinh$, $\;\sinh' = \cosh$, $\;\tanh' = 1 - \tanh^2
        = \dfrac{1}{\cosh^2}$;
  \item $\cosh(x+y) = \cosh x\cosh y + \sinh x \sinh y$ और $\sinh(x+y) =
  \sinh x \cosh y + \cosh x \sinh y$;
  \item $\sinh$ $\R$ का $\R$ पर कठोरतः वर्धमान एकैकी आच्छादन है; $\R_+$ तक
  प्रतिबंधित $\cosh$ $\intco{1}{+\infty}$ पर कठोरतः वर्धमान एकैकी आच्छादन
  है; $\tanh$ $\R$ का $\intoo{-1}{1}$ पर कठोरतः वर्धमान एकैकी आच्छादन है।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) $\bigl(\frac{\eu^x + \eu^{-x}}{2}\bigr)^2 - \bigl(\frac{\eu^x -
\eu^{-x}}{2}\bigr)^2 = \frac{(\eu^{2x} + 2 + \eu^{-2x}) - (\eu^{2x} -
2 + \eu^{-2x})}{4} = 1$.

(2) परिभाषा के सूत्रों का अवकलन कीजिए; $\tanh$ के लिए भागफल नियम
$\frac{\cosh^2 - \sinh^2}{\cosh^2}$ देता है, जो (1) से $1 - \tanh^2$ भी है
और $\frac{1}{\cosh^2}$ भी।

(3) परिभाषाओं का प्रयोग करके दाएँ पक्षों का प्रसार कीजिए। पहले सूत्र के लिए
विस्तार से:
\[
\cosh x\cosh y + \sinh x\sinh y
= \frac{(\eu^x + \eu^{-x})(\eu^y + \eu^{-y})
+ (\eu^x - \eu^{-x})(\eu^y - \eu^{-y})}{4} ;
\]
आठ गुणनफलों में से चार ``मिश्रित'' ($\eu^{x-y}$, $\eu^{y-x}$) जोड़ों में कट
जाते हैं, जबकि $\eu^{x+y}$ और $\eu^{-(x+y)}$ प्रत्येक दो बार आते हैं: कुल
$\frac{2\eu^{x+y} + 2\eu^{-(x+y)}}{4} = \cosh(x+y)$ है। ज्या का सूत्र इसके
समान ही है, बस उसमें मिश्रित पद बच जाते हैं।

(4) $\sinh' = \cosh \geq 1 > 0$, अतः $\sinh$ कठोरतः वर्धमान है, जिसकी सीमाएँ
$\pm\infty$ हैं (प्रभावी पद $\pm\frac{\eu^{\abs x}}{2}$); एकैकी आच्छादन वाला
\omterm{def:b1:logic:statement}{कथन} तब मध्यवर्ती मान प्रमेय से निकलता है (जो उच्चतर माध्यमिक स्तर पर
प्रयुक्त है; व्यवस्थित उपचार \cref{ch:b1:continuity} में)। $\R_+$ पर $\cosh'
= \sinh \geq 0$ केवल $0$ पर लुप्त होता है: अतः $\cosh 0 = 1$ से $+\infty$ तक
कठोरतः वर्धमान। और $\tanh' > 0$, जिसकी सीमाएँ $\pm\infty$ पर $\pm 1$ हैं:
विस्तार से,
\[
\tanh x = \frac{\eu^x - \eu^{-x}}{\eu^x + \eu^{-x}}
= \frac{1 - \eu^{-2x}}{1 + \eu^{-2x}}
\xrightarrow[x \to +\infty]{} 1 ,
\]
अंश और हर को $\eu^x$ से भाग देने के बाद, और विषमता $-\infty$ पर सीमा $-1$ दे
देती है; अतः कठोरतः वर्धमान फलन $\tanh$ $\R$ को $\intoo{-1}1$ पर भेजता है।
\end{proof}

\begin{remark}[``अतिपरवलयिक'' क्यों?]
बिंदु $(\cosh t, \sinh t)$ अतिपरवलय $x^2 - y^2 = 1$ की शाखा $x > 0$ पर चलता
है (\cref{prop:b1:functions:hyprules}~(1) से), ठीक वैसे ही जैसे $(\cos t,
\sin t)$ वृत्त $x^2 + y^2 = 1$ पर चलता है। हर वृत्तीय सर्वसमिका का एक
अतिपरवलयिक सहोदर है, जिसमें चिह्नों के परिवर्तन $\sinh^2 \leftrightarrow
-\sin^2$ से नियंत्रित होते हैं।
\end{remark}

\begin{example}[$\tanh$ का योग-सूत्र]\label{ex:b1:functions:tanhadd}
\cref{prop:b1:functions:hyprules}~(3) के दोनों योग-सूत्रों को $\cosh x\cosh
y$ से भाग देने पर:
\[
\tanh(x + y)
= \frac{\sinh x\cosh y + \cosh x\sinh y}
{\cosh x\cosh y + \sinh x\sinh y}
= \frac{\tanh x + \tanh y}{1 + \tanh x\,\tanh y} ,
\]
यह स्पर्शज्या के योग-सूत्र का अतिपरवलयिक सहोदर है --- जहाँ त्रिकोणमिति में
$-$ है, वहाँ $+$। एक लाभांश: चूँकि $\abs{\tanh} < 1$, दायाँ पक्ष ``वेग-योग''
का ऐसा नियम है जो $\intoo{-1}1$ कभी नहीं छोड़ता: यदि $u, v \in \intoo{-1}1$,
तो $\frac{u + v}{1 + uv} \in \intoo{-1}1$ भी (\omterm{def:b1:logic:inj}{एकैकी} आच्छादकता से संभव $u =
\tanh a$, $v = \tanh b$ लिखिए और सूत्र को उलटी दिशा में पढ़िए)। इसकी जाँच
बीजगणितीय रूप से, अतिपरवलयिक फलनों के बिना, कुछ कष्टप्रद अभ्यास है; $\tanh$
से प्राचलन उसे एक पंक्ति का बना देता है --- वही रणनीति जो वृत्तीय फलन इकाई
वृत्त के लिए देते हैं।
\end{example}

\begin{proposition}[प्रतिलोम अतिपरवलयिक फलन]\label{prop:b1:functions:invhyp}
\cref{prop:b1:functions:hyprules}~(4) से प्राप्त प्रतिलोमों के संवृत रूप
हैं:
\[
\operatorname{arsinh} x = \ln\bigl(x + \sqrt{x^2 + 1}\bigr)
\ (x \in \R),
\qquad
\operatorname{arcosh} x = \ln\bigl(x + \sqrt{x^2 - 1}\bigr)
\ (x \geq 1),
\]
\[
\operatorname{artanh} x = \frac 12 \ln\frac{1 + x}{1 - x}
\ (\abs x < 1),
\]
जिनके अवकलज क्रमशः $\frac{1}{\sqrt{x^2+1}}$, $\frac{1}{\sqrt{x^2-1}}$ ($x >
1$) और $\frac{1}{1 - x^2}$ हैं।
\end{proposition}

\begin{proof}
$\operatorname{arsinh}$ के लिए: $x = \sinh y = \frac{\eu^y - \eu^{-y}}{2}$
हल कीजिए। $u = \eu^y > 0$ रखने पर: $u^2 - 2xu - 1 = 0$, अतः $u = x +
\sqrt{x^2 + 1}$ (मूल $x - \sqrt{x^2+1}$ ऋणात्मक है), और $y = \ln(x +
\sqrt{x^2+1})$। शेष दोनों समरूप संगणनाएँ हैं ($\operatorname{arcosh}$ के लिए
$u^2 - 2xu + 1 = 0$, जहाँ मूल $\geq 1$ रखा जाता है; $\operatorname{artanh}$
के लिए दो पंक्ति का पुनर्व्यवस्थापन)। अवकलज: लघुगणकीय व्यंजकों का अवकलन
कीजिए, उदाहरणार्थ
\[
\bigl(\ln(x + \sqrt{x^2+1})\bigr)'
= \frac{1 + \frac{x}{\sqrt{x^2+1}}}{x + \sqrt{x^2+1}}
= \frac{1}{\sqrt{x^2 + 1}} . \qedhere
\]
\end{proof}

\begin{example}[संवृत रूप काम पर]\label{ex:b1:functions:arcoshtwo}
$t \geq 0$ के साथ $\cosh t = 2$ का हल, संवृत रूप से, $t =
\operatorname{arcosh} 2 = \ln(2 + \sqrt3) \approx 1.317$ है; दूसरा हल समता
से $-t$ है --- और सचमुच $\ln(2 - \sqrt3) = \ln\frac1{2 + \sqrt3} = -\ln(2 +
\sqrt3)$: द्विघात $u^2 - 4u + 1 = 0$ के दोनों मूल $u = \eu^t$ परस्पर
प्रतिलोम हैं, जैसा उनका गुणनफल $1$ (व्येता) माँगता है। यह छोटी संगणना
सामान्य प्रतिरूप दिखा देती है: अतिपरवलयिक समीकरण $\eu^t$ में द्विघातों में
बदल जाते हैं, और सममिति $t \mapsto -t$ द्विघात की सममिति $u \mapsto 1/u$ के
रूप में प्रकट होती है --- \cref{exo:b1:functions:7} हल करते समय यह स्मरण
रखने योग्य है।
\end{example}

\begin{method}[सही प्राथमिक रूप चुनना]\label{met:b1:functions:primitive}
समाकलन (\cref{ch:b1:integration}) के लिए कंठस्थ करने योग्य तीन
अवकलज-प्रतिरूप:
\[
\int \frac{\dd x}{1 + x^2} = \arctan x + C,
\qquad
\int \frac{\dd x}{\sqrt{1 - x^2}} = \arcsin x + C,
\qquad
\int \frac{\dd x}{\sqrt{x^2 + 1}} = \operatorname{arsinh} x + C,
\]
और किसी व्यापक द्विघात के लिए, वर्ग पूरा करके और पुनर्मापन करके इन्हीं तक
पहुँचिए।
\end{method}

\begin{remark}[यह अध्याय कहाँ काम आता है]\label{rem:b1:functions:whereused}
यह अध्याय आगे आने वाले पूरे विश्लेषण की काम की शब्दावली है।
\cref{prop:b1:functions:powerrules} का वृद्धि पैमाना
\cref{ch:b1:seq,ch:b1:series} में सर्वत्र अभिसरण के प्रश्न तय करता है;
\cref{prop:b1:functions:arcderiv} और \cref{prop:b1:functions:invhyp} के
अवकलज-सूत्र वही प्रतिअवकलज हैं जिनकी आवश्यकता \cref{ch:b1:integration} में
\cref{met:b1:functions:primitive} के द्वारा सबसे अधिक पड़ती है; \omterm{def:b1:functions:hyperbolic}{अतिपरवलयिक
फलन} \cref{ch:b1:diffeq} में समीकरण $y'' = y$ के हलों का प्राचलन ठीक वैसे
करते हैं जैसे वृत्तीय फलन $y'' = -y$ के हलों का। $f' = f$, $f(0) = 1$
(\cref{prop:b1:functions:expln}) से $\exp$ का अभिलक्षण रैखिक अवकल समीकरणों
के सिद्धांत का एकविमीय बीज है, और शृंखला वक्र $y = \cosh x$
\cref{ch:b1:curves} के समतल वक्रों में लौटता है।
\end{remark}

\begin{remark}[मध्यांतर: प्रतिलोम-फलन कार्यक्रम]\label{rem:b1:functions:interlude}
इस अध्याय ने एक ही कार्यक्रम चार बार चलाया: फलन को तब तक प्रतिबंधित कीजिए जब
तक वह कठोरतः एकदिष्ट एकैकी आच्छादन न बन जाए, प्रतिलोम को नाम दीजिए, और हर
सूत्र उसके पार ले जाइए। हर पद पर जो \emph{प्रयुक्त} हुआ --- कि अंतराल पर
संतत कठोरतः एकदिष्ट फलन किसी अंतराल पर एकैकी आच्छादन है, और उसका प्रतिलोम
संतत है तथा क्रांतिक बिंदुओं से दूर अवकलनीय --- वह उच्चतर माध्यमिक उधार पर
लिया गया था। यह ऋण इसी खंड के भीतर चुकाया जाता है: \cref{ch:b1:continuity}
एकैकी आच्छादन वाला \omterm{def:b1:logic:statement}{कथन} (एकदिष्ट प्रतिलोम प्रमेय, जो मध्यवर्ती मान प्रमेय पर
टिका है) सिद्ध करता है, और \cref{ch:b1:derivative} वह अवकलज नियम $(f^{-1})'
= 1/(f' \circ f^{-1})$ सिद्ध करता है जिसने \cref{prop:b1:functions:arcderiv}
और \cref{prop:b1:functions:invhyp} का हर सूत्र चुपचाप बनाया। उन अध्यायों को
$\arcsin$ और $\operatorname{arsinh}$ को ध्यान में रखकर पढ़ना --- उन हल किए
हुए उदाहरणों के रूप में जिन्हें न्यायसंगत ठहराने के लिए सिद्धांत खड़ा किया
गया --- ही इस प्रत्यक्ष चक्रीयता से निकलने का अभीष्ट मार्ग है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:functions:1}
कैलकुलेटर के बिना संगणना कीजिए: $\arcsin\frac{\sqrt 3}{2}$;
$\;\arccos\bigl(-\frac 12\bigr)$; $\;\arctan(-1)$;
$\;\arcsin\bigl(\sin\frac{5\pi}{6}\bigr)$;
$\;\arccos\bigl(\cos\frac{7\pi}{4}\bigr)$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:functions:2}
$f(x) = \arcsin(2x - 1)$ का परिभाषा-प्रांत दीजिए और जहाँ परिभाषित हो वहाँ
$f'$ संगणित कीजिए। $g(x) = \arctan\sqrt{x}$ के लिए भी यही प्रश्न।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:functions:3}
सिद्ध कीजिए कि सभी $x \in \R$ के लिए: $\cosh x + \sinh x = \eu^x$, और
प्रत्येक $n \in \Z$ के लिए $(\cosh x + \sinh x)^n = \cosh nx + \sinh nx$
(अतिपरवलयिक दे मॉयवर सूत्र)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:functions:4}
$\cos(2\arcsin x)$ और $\sin(2\arctan x)$ को $x$ के बीजीय व्यंजकों में सरल
कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:functions:5}
बड़े $x$ के लिए फलनों $x^{100}$, $\;\eu^{\sqrt x}$, $\;(\ln x)^{1000}$,
$\;\eu^{x/100}$, $\;x^{\ln x}$ को सबसे धीमी से सबसे तेज़ वृद्धि तक क्रम में
रखिए, और \cref{prop:b1:functions:powerrules} पर आधारित न्यायसंगति दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:functions:6}
फलन $f(x) = \arctan\dfrac{2x}{1 - x^2}$ का उसके प्रांत पर अध्ययन कीजिए: $f'$
संगणित कीजिए, $(2\arctan x)'$ से तुलना कीजिए, और प्रांत के तीनों अंतरालों
में से प्रत्येक पर $f(x)$ को $\arctan x$ के पदों में व्यक्त कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:functions:7}
$\R$ में हल कीजिए: $\;\cosh x = 2$; फिर $5\cosh x - 4 \sinh x = 3$ (हलों को
लघुगणकों से व्यक्त कीजिए)। \emph{दूसरे के लिए संकेत: सब कुछ $u = \eu^x$ से
लिखिए।}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:functions:8}
सर्वसमिका $\arctan\frac{1}{2} + \arctan\frac{1}{3} = \frac{\pi}{4}$ सिद्ध
कीजिए, फिर मैकिन का सूत्र\index{मैकिन का सूत्र}
\[
4\arctan\frac 15 - \arctan\frac{1}{239} = \frac{\pi}{4}.
\]
\emph{संकेत: योग-सूत्र से दोनों पक्षों की स्पर्शज्या संगणित कीजिए, और
नियंत्रण रखिए कि प्रत्येक पक्ष किस अंतराल में है।}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:functions:9}
सिद्ध कीजिए कि सभी $x \geq 0$ के लिए: $\;x - \dfrac{x^3}{6} \leq \sin x \leq
x$ \emph{(अंतरों के क्रमिक अवकलजों का अध्ययन कीजिए)}, और उससे निकालिए कि
$\lim_{x \to 0^+} \frac{x - \sin x}{x^3} = \frac 16$ प्रशंसनीय है --- सीमा
स्वयं \cref{ch:b1:taylor} में स्थापित की गई है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:functions:10}
$x \in \intcc{-1}{1}$ के लिए $h(x) = \arcsin\bigl(2x\sqrt{1 - x^2}\,\bigr) -
2\arcsin x$ रखिए। सर्वसमिका $\sin 2y = 2\sin y\cos y$ से $h = 0$ की अपेक्षा
हो सकती है --- पर $h$ तत्समक रूप से शून्य \emph{नहीं} है। जहाँ $h'$ विद्यमान
है उन खुले अंतरालों पर उसकी संगणना कीजिए, सुचयनित बिंदुओं पर $h$ का मान
निकालिए, और $\intcc{-1}{1}$ पर $h$ का पूरा खंडशः-अचर वर्णन दीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:functions:11}
(गुडरमान फलन) $x \in \R$ के लिए $g(x) = \arctan(\sinh x)$ रखिए। सिद्ध कीजिए
कि $g$ विषम, कठोरतः वर्धमान एकैकी आच्छादन है जो $\R$ को
$\intoo{-\frac\pi2}{\frac\pi2}$ पर भेजता है, कि $g'(x) = \frac{1}{\cosh x}$,
और यह कि $\tan g(x) = \sinh x$, $\;\sin g(x) = \tanh x$, $\;\cos g(x) =
\frac{1}{\cosh x}$: फलन $g$ वृत्तीय और अतिपरवलयिक त्रिकोणमिति को सम्मिश्र
संख्याओं के बिना जोड़ देता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:functions:12}
सिद्ध कीजिए कि
\[
\arctan 1 + \arctan 2 + \arctan 3 = \pi .
\]
\emph{संकेत: पहले $\arctan 2 + \arctan 3$ संगणित कीजिए, और
\cref{ex:b1:functions:arctansum} की भाँति योग का स्थानीकरण कीजिए; ध्यान रखिए
कि यहाँ योग-सूत्र का हर ऋणात्मक है।}
\end{exercise}

\section{समस्या: समीकरण $x^y = y^x$}

\begin{problem}\label{pb:b1:functions:1}
धनात्मक संख्याओं के कौन-से युग्म $x^y = y^x$ को संतुष्ट करते हैं? एक संयोगवश
दिखने वाला उदाहरण $2^4 = 4^2 = 16$ सब जानते हैं; यह समस्या दिखाती है कि
उसमें संयोग कुछ भी नहीं है। पूरा समीकरण एक ही फलन
\[
f(t) = \frac{\ln t}{t} \qquad (t > 0),
\]
के विचरण से नियंत्रित होता है, जिसके अध्ययन से मिलते हैं: पूरा \omterm{def:b1:logic:sets}{हल-समुच्चय}
(एक विकर्ण और $(\eu, \eu)$ से होकर जाती एक वक्र शाखा), शाखा का एक परिमेय
प्राचलन, यह तथ्य कि $(2, 4)$ उसका \emph{एकमात्र} पूर्णांक बिंदु है, और उसके
सभी परिमेय बिंदुओं का वर्गीकरण --- और साथ ही लाभांश के रूप में $\eu^\pi$ तथा
$\pi^\eu$ की तुलना और $\bigl(1 + \frac1n\bigr)^n$ का $\eu$ तक एकदिष्ट
अभिसरण।\index{घात फलन}

\medskip \noindent\textbf{भाग I --- फलन $f(t) = \ln t / t$।}
\begin{enumerate}
  \item न्यायसंगत ठहराइए कि $f$ $\intoo0{+\infty}$ पर अवकलनीय है, $f'$
  संगणित कीजिए, और विचरण सारणी बनाइए: $f$ $\intoc0\eu$ पर कठोरतः बढ़ता है,
  $\intco\eu{+\infty}$ पर कठोरतः घटता है, और उसका अधिकतम $f(\eu) =
  \frac1\eu$ है।
  \item $0^+$ और $+\infty$ पर $f$ की सीमाएँ
  (\cref{prop:b1:functions:powerrules}), $f$ का चिह्न ($\intoo01$ पर
  ऋणात्मक, $1$ पर शून्य, उसके आगे धनात्मक) निर्धारित कीजिए, और आलेख का
  रेखाचित्र बनाइए।
  \item सीधी संगणना से जाँचिए कि $f(2) = f(4)$। (इस समता को ध्यान में रखिए:
  पूरी समस्या इसी से उपजती है।)
  \item मान लीजिए $c \in \R$। $c$ के मान के अनुसार $f(t) = c$ के हलों की
  संख्या पर विचार कीजिए: $c \leq 0$ के लिए ठीक एक; $0 < c < \frac1\eu$ के
  लिए ठीक दो (एक $\intoo1\eu$ में, एक $\intoo\eu{+\infty}$ में); $c =
  \frac1\eu$ के लिए ठीक एक; $c > \frac1\eu$ के लिए कोई नहीं।
  \item उससे निकालिए कि $x \neq \eu$ वाले प्रत्येक $x > 0$ के लिए $\eu^x >
  x^\eu$, और विशेष रूप से यह तय कीजिए कि $\eu^\pi$ और $\pi^\eu$ में कौन बड़ा
  है।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- समीकरण और उसका वक्र।} इस भाग में $x, y
> 0$।
\begin{enumerate}[resume]
  \item दिखाइए कि $x^y = y^x \iff f(x) = f(y)$।
  \item उससे \omterm{def:b1:logic:sets}{हल-समुच्चय} की संरचना निकालिए: सभी विकर्ण युग्म $(x, x)$; और
  \emph{अतुच्छ} युग्म ($x \neq y$), जो यह संतुष्ट करते हैं: दोनों निर्देशांक
  $> 1$ हैं, और एक $\intoo1\eu$ में है जबकि दूसरा $\intoo\eu{+\infty}$ में।
  \item अतुच्छ युग्मों का प्राचलन कीजिए: $t > 0$, $t \neq 1$ के साथ $y = tx$
  लिखकर दिखाइए कि $x^y = y^x$ से अनिवार्य हो जाता है
        \[
        x(t) = t^{\frac1{t-1}},
        \qquad
        y(t) = t\,x(t) = t^{\frac{t}{t-1}},
        \]
        और यह कि विलोमतः ऐसा हर युग्म एक हल है।
  \item सत्यापित कीजिए कि $t = 2$ से $(2, 4)$ मिलता है, और सममिति $x(1/t) =
  y(t)$, $y(1/t) = x(t)$ सिद्ध कीजिए: प्राचल का प्रतिलोमन दोनों निर्देशांकों
  की अदला-बदली कर देता है।
  \item $t \to 1$ पर $x(t)$ और $y(t)$ की सीमाएँ निर्धारित कीजिए (दोनों $\eu$
  की ओर जाते हैं), फिर $t \to +\infty$ पर ($x \to 1$, $y \to +\infty$) और $t
  \to 0^+$ पर ($x \to +\infty$, $y \to 1$)। परिणामी शाखा का वर्णन कीजिए:
  रेखाओं $x = 1$ और $y = 1$ के अनंतस्पर्शी एक वक्र, जो विकर्ण को $(\eu,
  \eu)$ पर काटता है।
  \item दिखाइए कि $t \mapsto x(t)$ $\intoo1{+\infty}$ पर कठोरतः ह्रासमान है।
  \emph{($h(t) = 1 - \frac1t - \ln t$ का अध्ययन कीजिए, जिसका चिह्न $\ln x(t)
  = \frac{\ln t}{t - 1}$ के अवकलज को नियंत्रित करता है।)}
  \item निष्कर्ष निकालिए कि अतुच्छ हल $x^{\varphi(x)} = \varphi(x)^x$ वाला
  एक कठोरतः ह्रासमान एकैकी आच्छादन $\varphi \colon \intoo1\eu \to
  \intoo\eu{+\infty}$ परिभाषित करते हैं, और यह कि $\varphi$ शाखा का एक
  स्वप्रतिलोमन है: जहाँ भी दोनों पक्ष परिभाषित हों वहाँ $\varphi(\varphi(x))
  = x$।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- पूर्णांक और परिमेय बिंदु।}
\begin{enumerate}[resume]
  \item सिद्ध कीजिए कि $x^y = y^x$ के अतुच्छ \emph{पूर्णांक} हल केवल $(2,
  4)$ और $(4, 2)$ हैं।
  \item $n \in \N^*$ के लिए $t = 1 + \frac1n$ के साथ प्राचलन लगाइए और दिखाइए
  कि
        \[
        x_n = \Bigl(1 + \frac1n\Bigr)^{n},
        \qquad
        y_n = \Bigl(1 + \frac1n\Bigr)^{n+1}
        \]
        प्रत्येक $n$ के लिए एक अतुच्छ \emph{परिमेय} हल है, जहाँ $(x_1, y_1)
        = (2, 4)$।
  \item विलोमतः, मान लीजिए $(x, y)$ $y > x$ वाला अतुच्छ परिमेय हल है, और $t
  = y/x = 1 + \frac rs$ को न्यूनतम पदों में लिखिए ($r, s \in \N^*$)। $x =
  t^{s/r}$ का प्रयोग करते हुए और अभाज्य गुणनखंडन की अद्वितीयता मानते हुए (जो
  विद्यालय से परिचित है; \cref{ch:b1:arith} में सिद्ध), दिखाइए कि $x$ का
  परिमेय होना $s$ और $s + r$ दोनों को पूर्णांकों की $r$-वीं घातें होने पर
  बाध्य कर देता है।
  \item द्विपद प्रमेय से दिखाइए कि $r \geq 2$ होने पर $a^r - b^r = r$ का कोई
  पूर्णांक हल $a > b \geq 1$ नहीं है, और निष्कर्ष निकालिए: $x^y = y^x$ के
  परिमेय हल ठीक प्रश्न 14 के युग्म $(x_n, y_n)$ (और उनकी अदला-बदलियाँ) हैं।
  \item स्थिति $n = 2$ संख्यात्मक रूप से सत्यापित कीजिए: $f(9/4)$ और
  $f(27/8)$ की चार दशमलव स्थानों तक संगणना कीजिए और जाँचिए कि वे मेल खाते
  हैं।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- लाभांश।}
\begin{enumerate}[resume]
  \item प्रश्न 7 और 11 का प्रयोग करके, बिना किसी और संगणना के, सिद्ध कीजिए
  कि अनुक्रम $x_n = \bigl(1 + \frac1n\bigr)^n$ कठोरतः वर्धमान है और $x_n <
  \eu$, कि $y_n = \bigl(1 + \frac1n\bigr)^{n+1}$ कठोरतः ह्रासमान है और $y_n
  > \eu$, और यह कि दोनों $\eu$ की ओर अभिसरित होते हैं। (प्राचल $t_n = 1 +
  \frac1n$ घटकर $1$ तक जाता है।)
  \item $1 < a < b$ के लिए व्यापक तुलना नियम स्थापित कीजिए: यदि $\eu \leq
  a$, तो $a^b > b^a$; यदि $b \leq \eu$, तो $a^b < b^a$; और दोनों उदाहरणों
  $(2, 3)$ तथा $(2, 5)$ से दिखाइए कि मिश्रित स्थिति $a < \eu < b$ में दोनों
  परिणाम सचमुच होते हैं।
  \item मान लीजिए प्रश्न 12 का स्वप्रतिलोमन $\varphi$ $\eu$ पर अवकलनीय है
  (वह है)। $x = \eu$ पर सर्वसमिका $\varphi(\varphi(x)) = x$ का अवकलन कीजिए
  और $\varphi'(\eu) = -1$ निकालिए: शाखा विकर्ण को समकोण पर काटती है।
  \item $y = 3x$ वाला अद्वितीय हल-युग्म संवृत रूप में खोजिए, और $f$ के
  द्वारा उसे चार दशमलव तक संख्यात्मक रूप से जाँचिए।
  \item संख्याओं $\sqrt2$, $\sqrt[3]3$, $\sqrt[4]4$, $\sqrt[5]5$ में से सबसे
  बड़ी निर्धारित कीजिए और उन दो की पहचान कीजिए जो बराबर हैं। \emph{($n^{1/n}
  = \eu^{f(n)}$ की तुलना कीजिए।)}
\end{enumerate}

\medskip \noindent\textbf{भाग V --- संश्लेषण।}
\begin{enumerate}[resume]
  \item चतुर्थांश-समतल $x, y > 0$ में $x^y = y^x$ के पूरे \omterm{def:b1:logic:sets}{हल-समुच्चय} का
  वर्णन ऐसे रूप में कीजिए जिसे आप स्मरण से खींच सकें --- विकर्ण और शाखा,
  उनका प्रतिच्छेद $(\eu, \eu)$, अनंतस्पर्शी रेखाएँ, पूर्णांक बिंदु $(2, 4)$,
  और $(\eu, \eu)$ पर संचित होते परिमेय बिंदु।
  \item इस समस्या में ठीक कहाँ प्रयोग हुआ: (क)
  \cref{prop:b1:functions:powerrules} की वृद्धि तुलनाएँ; (ख) मध्यवर्ती मान
  प्रमेय (एकैकी आच्छादन वाले कथनों के द्वारा); (ग) स्वीकृत अभाज्य गुणनखंडन
  की अद्वितीयता? प्रत्येक के लिए एक वाक्य।
  \item एक छोटे अनुच्छेद में उपदेश: एक ही विचरण सारणी ने दो अज्ञातों वाला
  समीकरण हल कर दिया, उसके परिमेय बिंदुओं का वर्गीकरण कर दिया, और $\bigl(1 +
  \frac1n\bigr)^n$ का एकदिष्ट अभिसरण सिद्ध कर दिया --- इस मितव्ययिता पर
  टिप्पणी कीजिए, और बताइए कि प्रत्येक धागा आगे इस खंड में कहाँ औद्योगिक रूप
  ले लेता है (अनुक्रम के लिए \cref{ch:b1:seq}, विचरण सारणियों के लिए
  \cref{ch:b1:derivative}, और सारणी में जिस यथार्थता की कमी है उसके लिए
  \cref{ch:b1:taylor})।
\end{enumerate}
\end{problem}
