\chapter{वास्तविक संख्याएँ}\label{ch:b1:reals}

सारा विश्लेषण एक ही गुणधर्म पर टिका है, जो $\R$ को $\Q$ से अलग करता है: ऊपर
से परिबद्ध प्रत्येक अरिक्त \omterm{def:b1:logic:sets}{समुच्चय} का एक \emph{लघुतम} \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} होता है।
यह अध्याय उसे ठीक-ठीक कहता है, उसके पहले परिणाम निकालता है --- \omterm{thm:b1:reals:archimedes}{आर्किमिडीज़
गुणधर्म}, \omterm{thm:b1:reals:floor}{फ़र्श फलन}, परिमेय तथा अपरिमेय संख्याओं की सघनता --- और वह शब्दावली
खड़ी करता है (\omterm{def:b1:reals:bounds}{उच्चतम}, \omterm{def:b1:reals:bounds}{निम्नतम}, अधिकतम, न्यूनतम) जो \cref{ch:b1:seq} से आगे
लगातार प्रयुक्त होती है।

\section{ऊपरी परिबंध का गुणधर्म}

\begin{definition}[परिबंध, उच्चतम और निम्नतम]\label{def:b1:reals:bounds}
मान लीजिए $A \subseteq \R$ अरिक्त है। वास्तविक $M$ $A$ का \emph{ऊपरी
परिबंध}\index{ऊपरी परिबंध} तब है जब सभी $a \in A$ के लिए $a \leq M$; $A$
\emph{ऊपर से परिबद्ध} तब है जब उसका कोई ऊपरी परिबंध हो (इसी प्रकार नीचे से,
निचले परिबंधों के साथ; \emph{परिबद्ध} का अर्थ है दोनों)। $A$ का
\emph{अधिकतम} ऐसा ऊपरी परिबंध है जो स्वयं $A$ का सदस्य हो।

\emph{उच्चतम} $\sup A$\index{उच्चतम} $A$ का लघुतम ऊपरी परिबंध है, जब वह
विद्यमान हो; \emph{निम्नतम} $\inf A$\index{निम्नतम} महत्तम निचला परिबंध है।
\end{definition}

\begin{theorem}[$\R$ का पूर्णता अभिगृहीत]\label{thm:b1:reals:sup}
$\R$ ऐसा क्रमित \omterm{def:b1:structures:field}{क्षेत्र} है जिसमें $\Q$ समाया हुआ है और जिसमें \emph{ऊपर से
परिबद्ध प्रत्येक अरिक्त उपसमुच्चय का \omterm{def:b1:reals:bounds}{उच्चतम} होता है}।\index{पूर्णता@$\R$ की
पूर्णता}
\end{theorem}
\admitted

\begin{remark}
हम इसे $\R$ का परिभाषक अभिगृहीत मानते हैं; कोई प्रतिरूप बनाना (डेडेकिंड
विच्छेदों से या परिमेय संख्याओं के कोशी अनुक्रमों से) और उसकी अद्वितीयता
सिद्ध करना ईमानदार तो है पर लंबा, अतः उसे आगे के अध्ययन के लिए छोड़ा गया है।
ध्यान दीजिए कि $\Q$ यह गुणधर्म पूरा नहीं करता: $\{x \in \Q : x^2 < 2\}$ ऊपर
से परिबद्ध है पर $\Q$ में उसका कोई लघुतम \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} \emph{नहीं} है --- उसका
उम्मीदवार $\sqrt 2$ वहाँ अनुपस्थित है (\cref{ex:b1:logic:sqrt2})। विपरीत
लेने से ($\sup(-A) = -\inf A$) नीचे से परिबद्ध प्रत्येक अरिक्त \omterm{def:b1:logic:sets}{समुच्चय} का
\omterm{def:b1:reals:bounds}{निम्नतम} भी मिल जाता है।
\end{remark}

\begin{proposition}[$\varepsilon$-अभिलक्षण]\label{prop:b1:reals:epsilon}
मान लीजिए $A \neq \emptyset$ ऊपर से परिबद्ध है और $s \in \R$। तब $s = \sup
A$ यदि और केवल यदि
\begin{enumerate}
  \item $s$ \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} है: $\forall a \in A$, $a \leq s$; तथा
  \item उससे छोटा कोई नहीं है: $\forall \varepsilon > 0$, $\exists a \in A$,
  $a > s - \varepsilon$।
\end{enumerate}
\end{proposition}

\begin{proof}
यदि $s = \sup A$: (1) परिभाषा से सत्य है, और (2) के लिए $s - \varepsilon <
s$ \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} नहीं है, जो ठीक $a > s - \varepsilon$ के अस्तित्व के बराबर
है। विलोमतः, (1) कहता है कि $s$ \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} है; (2) कहता है कि कोई $t < s$
\omterm{def:b1:reals:bounds}{ऊपरी परिबंध} नहीं है ($\varepsilon = s - t$ लीजिए): अतः $s$ लघुतम है।
\end{proof}

\begin{example}\label{ex:b1:reals:supexamples}
$\sup \intoo{0}{1} = 1$, जो प्राप्त नहीं होता (कोई अधिकतम नहीं); $\sup
\intcc{0}{1} = 1 = \max$। $A = \{1 - \frac 1n : n \in \N^*\}$ के लिए: $\sup
A = 1$, जो प्राप्त नहीं होता; $\inf A = \min A = 0$। अधिकतम, जब विद्यमान हो,
\omterm{def:b1:reals:bounds}{उच्चतम} ही होता है; $\sup$ का पूरा प्रयोजन यही है कि जब अधिकतम विद्यमान न हो
तब उसका विकल्प उपलब्ध रहे।
\end{example}

\begin{example}[व्यवहार में उच्चतम की संगणना]\label{ex:b1:reals:supcompute}
\cref{prop:b1:reals:epsilon} के दो पूरे अभ्यास।

\emph{\omterm{def:b1:logic:sets}{समुच्चय} $A = \{x + \frac1x : x > 0\}$।} प्रत्येक $x > 0$ के लिए $x +
\frac1x - 2 = \frac{(\,\sqrt x - 1/\sqrt x\,)^2}{1} \geq 0$, अतः $2$ निचला
परिबंध है; और $2 = 1 + \frac11 \in A$: इसलिए $\inf A = \min A = 2$, जो $x =
1$ पर प्राप्त होता है। ऊपर की ओर $A$ अपरिबद्ध है ($x + \frac1x > x$ किसी भी
$M$ को \cref{thm:b1:reals:archimedes} से पार कर सकता है): अतः $\R$ में $\sup
A$ विद्यमान नहीं है ($\overline\R$ में वह $+\infty$ है)।

\emph{\omterm{def:b1:logic:sets}{समुच्चय} $B = \bigl\{\frac{m}{m + n} : m, n \in \N^*\bigr\}$।} प्रत्येक
अवयव $\intoo{0}{1}$ में है, अतः $0$ और $1$ परिबंध हैं। इनमें से कोई प्राप्त
नहीं होता: $\frac{m}{m+n} = 1$ $n = 0$ पर बाध्य कर देता। \omterm{def:b1:reals:bounds}{उच्चतम} के लिए $n =
1$ को नियत कीजिए और $m$ को बढ़ने दीजिए: $m + 1 > \frac1\varepsilon$ होते ही
$\frac{m}{m+1} = 1 - \frac{1}{m+1} > 1 - \varepsilon$ (आर्किमिडीज़): $\sup B
= 1$। सममित रूप से ($m = 1$, $n$ बड़ा), $\inf B = 0$। समापन का सार: \omterm{def:b1:reals:bounds}{उच्चतम}
को कीलने के लिए \omterm{def:b1:logic:sets}{समुच्चय} के भीतर एक सुचयनित \emph{एक-प्राचल पथ} पर्याप्त है
--- यहाँ पथ $n = 1$ --- और $\varepsilon$-अभिलक्षण इससे अधिक कुछ नहीं माँगता।
\end{example}

\begin{example}[निम्नतम का दर्पण]\label{ex:b1:reals:infmirror}
\omterm{def:b1:reals:bounds}{निम्नतम} का अपना $\varepsilon$-अभिलक्षण है, जो $\inf A = -\sup(-A)$ के द्वारा
\cref{prop:b1:reals:epsilon} से मिलता है: $i = \inf A$ तभी जब $i$ $A$ को
नीचे से परिबद्ध करता हो और प्रत्येक $\varepsilon > 0$ के लिए कोई $a \in A$
$a < i + \varepsilon$ संतुष्ट करता हो। दोनों परिबंधों का एक साथ अभ्यास: मान
लीजिए
\[
A = \Bigl\{(-1)^n + \frac1n : n \in \N^*\Bigr\}
= \Bigl\{0,\ \tfrac32,\ -\tfrac23,\ \tfrac54,\ -\tfrac45,\
\dots\Bigr\} .
\]
सम सूचकांक $1 + \frac1n \leq \frac32$ देते हैं, जहाँ $n = 2$ पर समता है: और
चूँकि विषम-सूचकांक वाले मान भी $\leq 0 < \frac32$ हैं, अतः $\sup A = \max A
= \frac32$। विषम सूचकांक $-1 + \frac1n > -1$ देते हैं, जो $-1$ की ओर घटते
जाते हैं: $A$ का प्रत्येक अवयव $> -1$ है, और विषम $n > \frac1\varepsilon$ के
लिए $-1 + \frac1n$ $-1 + \varepsilon$ को हरा देता है: $\inf A = -1$, जो
प्राप्त नहीं होता। एक ही \omterm{def:b1:logic:sets}{समुच्चय}, और चारों व्यवहार प्रदर्शित: ऐसा \omterm{def:b1:reals:bounds}{उच्चतम} जो
अधिकतम है, और ऐसा \omterm{def:b1:reals:bounds}{निम्नतम} जो न्यूनतम नहीं है।
\end{example}

\begin{remark}[उच्चतम और निम्नतम की सामान्य भूलें]
अधिकांश अंक चार भूलों से कटते हैं। (क) \emph{$\sup$ और $\max$ को मिला देना}:
$\sup A$ का $A$ का सदस्य होना आवश्यक नहीं है; $\max$ तभी लिखिए जब $A$ का ऐसा
अवयव प्रस्तुत कर दिया हो जो \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} हो। (ख) \emph{कठोर असमिकाओं को
\omterm{def:b1:reals:bounds}{उच्चतम} तक ले जाना}: यदि सभी $a \in A$ के लिए $a < b$, तो केवल $\sup A \leq
b$ निकाला जा सकता है --- साक्षी $A = \intoo{0}{1}$, $b = 1$। (ग) \emph{वैधता
जाँचे बिना $\sup A$ लिख देना}: यह संकेत माँगता है कि $A$ अरिक्त और ऊपर से
परिबद्ध हो (\cref{met:b1:reals:supproofs}); $\sup \emptyset$ और $\sup \N$
$\R$ में अपरिभाषित हैं ($\overline\R$ की परिपाटियाँ एक अलग, स्पष्ट कार्य
हैं)। (घ) \emph{\omterm{def:b1:logic:sets}{समुच्चय} संक्रियाएँ}: $\sup(A \cup B) = \max(\sup A, \sup B)$
सदा सत्य है, पर $A \cap B$ के लिए कुछ भी व्यापक रूप से सत्य नहीं --- वह
रिक्त हो सकता है, और रिक्त न भी हो तब $\sup(A \cap B)$ $\min(\sup A, \sup
B)$ से बहुत नीचे हो सकता है: $A = \{0, 2\}$ और $B = \{0, 3\}$ लीजिए, जहाँ
$\sup(A \cap B) = 0$।
\end{remark}

\begin{example}[परिमित समुच्चयों के अधिकतम होते हैं --- एक मौन प्रयुक्त प्रमेयिका]\label{ex:b1:reals:finitemax}
प्रत्येक परिमित अरिक्त $F \subseteq \R$ का अधिकतम (और न्यूनतम) होता है।
अवयवों की संख्या पर आगमन: एकल \omterm{def:b1:logic:sets}{समुच्चय} $\{a\}$ के लिए $\max = a$; यदि दावा
$n$-अवयवी समुच्चयों के लिए सत्य है और $F$ में $n + 1$ अवयव हैं, तो कोई भी $a
\in F$ चुनिए: \omterm{def:b1:logic:sets}{समुच्चय} $F \setminus \{a\}$ का अधिकतम $m$ है, और $\max F$ $a
\leq m$ होने पर $m$ है, अन्यथा $a$। इसमें पूर्णता कहीं नहीं है --- यह शुद्ध
क्रम और आगमन है, जो $\Q$ में भी वैध है --- फिर भी इस प्रमेयिका को एक बार
ईमानदारी से कहना उचित है, क्योंकि आगे की उपपत्तियाँ उसे चुपचाप बुलाती हैं:
नीचे की फ़र्श-रचना (``परिमित परास में फँसे पूर्णांकों के \omterm{def:b1:logic:sets}{समुच्चय} का महत्तम
अवयव होता है''), \cref{ch:b1:seq} का हर $\max(\abs{u_0}, \dots,
\abs{u_{N-1}}, \dots)$ परिबंध, और \cref{ch:b1:continuity} का हर ``परिमित
कितने $\delta$ में से सबसे बड़ा लीजिए''। अनंत \omterm{def:b1:logic:sets}{समुच्चय} ही वे हैं जहाँ अधिकतम
मर जाते हैं और \omterm{def:b1:reals:bounds}{उच्चतम} कमान सँभाल लेते हैं: यह अध्याय अनंत स्थिति के लिए ही
है।
\end{example}

\begin{theorem}[आर्किमिडीज़ गुणधर्म]\label{thm:b1:reals:archimedes}
प्रत्येक $x \in \R$ के लिए ऐसा $n \in \N$ है कि $n > x$। समतुल्य रूप से: सभी
$\varepsilon > 0$ और $y > 0$ के लिए कोई गुणज $n\varepsilon$ $y$ को पार कर
जाता है।\index{आर्किमिडीज़ गुणधर्म}
\end{theorem}

\begin{proof}
मान लीजिए ऐसा नहीं है: कोई $x$ $\N$ का \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} है। तब $s = \sup \N$
विद्यमान है (\cref{thm:b1:reals:sup})। $\varepsilon = 1$ के साथ
\cref{prop:b1:reals:epsilon}~(2) से ऐसा $n \in \N$ है कि $n > s - 1$; पर तब
$n + 1 \in \N$ और $n + 1 > s$, जो $s$ के \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} होने का विरोध करता है।
दूसरे रूप के लिए मान लीजिए $\varepsilon > 0$ और $y > 0$: पहला रूप $x =
\frac{y}{\varepsilon}$ पर लगाने से ऐसा $n \in \N$ मिलता है कि $n >
\frac{y}{\varepsilon}$, और $\varepsilon > 0$ से गुणा करने पर (जो कठोर
असमिकाएँ बनाए रखता है) $n\varepsilon > y$ मिलता है। विलोमतः, दूसरा रूप
$\varepsilon = 1$ और $y = x$ के साथ $x > 0$ के लिए पहला रूप लौटा देता है, और
$n = 1$ $x \leq 0$ को सँभाल लेता है: दोनों \omterm{def:b1:logic:statement}{कथन} कठोरतः तुल्य हैं।
\end{proof}

\begin{example}[आर्किमिडीज़ काम पर]\label{ex:b1:reals:archimedeswork}
तीन तात्कालिक उपयोग, जिनकी आगे लगातार आवश्यकता पड़ती है। (क) \emph{कोई
धनात्मक वास्तविक संख्या हर $\frac1n$ से नीचे नहीं है}: यदि $0 <
\varepsilon$, तो $n > \frac1\varepsilon$ चुनिए; तब $\frac1n < \varepsilon$।
दूसरे शब्दों में, $\R$ में कोई अतिसूक्ष्म संख्या नहीं है --- अनौपचारिक
``$\frac1n$ मनचाहा छोटा हो जाता है'' ठीक यही प्रमेय है। (ख) \emph{स्पष्ट
देहलियाँ}: $\frac{1}{n^2} \leq 10^{-6}$ के लिए $n$ कितना बड़ा होना चाहिए?
इतना पर्याप्त है कि $n \geq 10^3$ --- आर्किमिडीज़ गारंटी देता है कि ऐसे $n$
हैं, और बीजगणित उन्हें ढूँढ़ देता है। (ग) \emph{घातें हर परिबंध को हरा देती
हैं}: $2^n \geq n + 1$ (आगमन), अतः प्रत्येक $M$ के लिए $2$ की कोई घात $M$ को
पार कर जाती है: यही गुणोत्तर वृद्धि \cref{exo:b1:reals:8} में द्विआधारी
संख्याओं के लिए प्रयुक्त होती है। समापन का सार: \omterm{thm:b1:reals:archimedes}{आर्किमिडीज़ गुणधर्म} ही ``$n$
पर्याप्त बड़ा लीजिए'' रूप के हर वाक्य के पीछे की अनुमति है --- अब से हम उस
वाक्य का स्वतंत्र प्रयोग करेंगे, और यह उदाहरण उसका एक बार का औचित्य है।
\end{example}

\begin{theorem}[फ़र्श फलन]\label{thm:b1:reals:floor}
प्रत्येक $x \in \R$ के लिए ठीक एक पूर्णांक है, अर्थात्
\emph{फ़र्श}\index{फ़र्श फलन} $\lfloor x \rfloor$, जिसके लिए
\[
\lfloor x \rfloor \leq x < \lfloor x \rfloor + 1 .
\]
\end{theorem}

\begin{proof}
\emph{अस्तित्व।} \omterm{def:b1:logic:sets}{समुच्चय} $E = \{k \in \Z : k \leq x\}$ अरिक्त है:
\cref{thm:b1:reals:archimedes} से ऐसा $m \in \N$ है कि $m > -x$, और तब $-m <
x$, अतः $-m \in E$। वह ऊपर से परिबद्ध भी है (किसी भी पूर्णांक $n > x$ से, जो
उसी कारण से विद्यमान है), अतः परिमित परास $\intint{-m}{n}$ में फँसे
पूर्णांकों का \omterm{def:b1:logic:sets}{समुच्चय} होने के नाते उसका एक महत्तम अवयव $k = \max E$ है। तब
$k \leq x$, और $k + 1 \notin E$ का अर्थ है $x < k + 1$।

\emph{अद्वितीयता।} यदि $k$ और $k'$ दोनों असमिकाएँ संतुष्ट करते हैं, तो $k
\leq x < k' + 1$ $k \leq k'$ देता है, और सममित रूप से $k' \leq k$।
\end{proof}

\begin{example}[व्यवहार में फ़र्श]\label{ex:b1:reals:floorpractice}
$\lfloor 3.7 \rfloor = 3$, $\lfloor 5 \rfloor = 5$, और $\lfloor -3.7 \rfloor
= -4$: \omterm{thm:b1:reals:floor}{फ़र्श} \emph{नीचे} जाता है, $0$ की ओर नहीं। \cref{thm:b1:reals:floor}
की अद्वितीयता के दो परिणाम, जिनका प्रयोग हम चुपचाप करेंगे। पहला, $n \in \Z$
के लिए
\[
\lfloor x + n \rfloor = \lfloor x \rfloor + n ,
\]
क्योंकि $\lfloor x \rfloor + n$ ऐसा पूर्णांक है जो $x + n$ की दोनों परिभाषक
असमिकाएँ संतुष्ट करता है --- और ऐसा केवल एक ही पूर्णांक करता है। दूसरा,
$\lfloor \, \cdot \, \rfloor$ अह्रासमान है: यदि $x \leq y$, तो $\lfloor x
\rfloor \leq x \leq y < \lfloor y \rfloor + 1$, और पूर्णांक $< \lfloor y
\rfloor + 1$ $\leq \lfloor y \rfloor$ है। किंतु ध्यान रखिए कि व्यापक रूप से
$\lfloor 2x \rfloor \neq 2\lfloor x \rfloor$: $x = 0.6$ से $\lfloor 1.2
\rfloor = 1 \neq 0 = 2\lfloor 0.6 \rfloor$ मिलता है।

जो \emph{सचमुच} सत्य है वह एक हल की हुई सर्वसमिका है, जिसे स्मरण रखना उचित
है (एरमीट की, अपने सरलतम रूप में): प्रत्येक वास्तविक $x$ के लिए,
\[
\lfloor x \rfloor + \Bigl\lfloor x + \frac12 \Bigr\rfloor
= \lfloor 2x \rfloor .
\]
$u \in \intco{0}{1}$ के साथ $x = \lfloor x\rfloor + u$ लिखिए और दो स्थितियों
में बाँटिए। यदि $u < \frac12$: बायाँ पक्ष $\lfloor x\rfloor + \lfloor
x\rfloor = 2\lfloor x\rfloor$ है, और $2u \in \intco{0}{1}$ के साथ $2x =
2\lfloor x\rfloor + 2u$, अतः दायाँ पक्ष भी $2\lfloor x\rfloor$ है। यदि $u
\geq \frac12$: बायाँ पक्ष $\lfloor x\rfloor + (\lfloor x\rfloor + 1)$ है, और
$2u \in \intco{1}{2}$ दाएँ पक्ष को $2\lfloor x\rfloor + 1$ बना देता है।
समापन का सार: $\lfloor x + \frac12\rfloor$ $x$ का निकटतम पूर्णांक तक
\emph{पूर्णांकन} है, अतः सर्वसमिका कहती है कि \omterm{thm:b1:reals:floor}{फ़र्श} और पूर्णांकन मिलकर
दुगुने का \omterm{thm:b1:reals:floor}{फ़र्श} देते हैं --- और भिन्नात्मक भाग $u$ पर स्थिति-विभाजन हर
फ़र्श-सर्वसमिका के पीछे की मानक तकनीक है
(\cref{exo:b1:reals:2,exo:b1:reals:3} भी उसी पर चलते हैं)।
\end{example}

\begin{theorem}[$\Q$ और $\R \setminus \Q$ की सघनता]\label{thm:b1:reals:density}
किन्हीं दो वास्तविक संख्याओं $x < y$ के बीच एक परिमेय और एक अपरिमेय संख्या
होती है।\index{सघनता}
\end{theorem}

\begin{proof}
\emph{एक परिमेय।} \cref{thm:b1:reals:archimedes} से ऐसा $n \in \N^*$ चुनिए
कि $n > \frac{1}{y - x}$, अतः $ny - nx > 1$। मान लीजिए $m = \lfloor nx
\rfloor + 1$। एक ओर $nx < \lfloor nx \rfloor + 1 = m$
(\cref{thm:b1:reals:floor}); दूसरी ओर $m = \lfloor nx \rfloor + 1 \leq nx +
1 < ny$। $n$ से भाग देने पर: $x < \frac mn < y$।

\emph{एक अपरिमेय।} पिछला बिंदु युग्म $x - \sqrt 2 < y - \sqrt 2$ पर लगाइए:
कोई परिमेय $q$ उनके बीच है, और तब $q + \sqrt 2 \in \intoo{x}{y}$ अपरिमेय है
(यदि $q + \sqrt 2$ परिमेय होता, तो $\sqrt 2$ भी होता)।
\end{proof}

\begin{example}[सघनता की उपपत्ति को चलाना]\label{ex:b1:reals:densityrun}
यह उपपत्ति एक कलनविधि है; आइए उसे $x = 1.414$ और $y = \sqrt 2$ पर चलाएँ।
चूँकि $1.4142^2 = 1.99996164 < 2$, अतः $\sqrt 2 > 1.4142$, इसलिए $y - x >
0.0002$ और $\frac{1}{y - x} < 5000$: अतः चयन $n = 5000$ वैध है। तब $nx =
7070$, अतः $m = \lfloor 7070 \rfloor + 1 = 7071$, और उत्पन्न परिमेय संख्या
है
\[
\frac{m}{n} = \frac{7071}{5000} = 1.4142,
\qquad
1.414 < 1.4142 < \sqrt 2 .
\]
समापन का सार: उपपत्ति को $n$ $\frac{1}{y-x}$ से केवल थोड़ा बड़ा चाहिए, और वह
$x$ के आगे $\frac 1n$ का पहला गुणज लौटा देती है। सघनता कोई अमूर्त चमत्कार
नहीं है --- वह वेश बदला हुआ लंबा भाग है, और यही विषय सप्ताहांत समस्या
(\cref{pb:b1:reals:1}) में विस्तार से विकसित होता है।
\end{example}

\begin{remark}[पूर्णता का अगला प्रयोग कहाँ]
\Cref{thm:b1:reals:sup} इस पुस्तक का एकमात्र अबीजीय अभिगृहीत है, और विश्लेषण
की हर अस्तित्व-प्रमेय वही अभिगृहीत भिन्न वस्त्रों में है: एकदिष्ट अभिसरण
प्रमेय (\cref{ch:b1:seq}), बोल्ज़ानो--वाइरश्ट्रास प्रमेय
(\cref{ch:b1:topology}), मध्यवर्ती मान और चरम मान प्रमेय
(\cref{ch:b1:continuity}), और निम्न योगों के \omterm{def:b1:reals:bounds}{उच्चतम} के रूप में समाकल की
परिभाषा ही (\cref{ch:b1:integration})। स्नातक वर्ष 3 का खंड माप-सिद्धांत और
हिल्बर्ट समष्टियाँ इसी एक अभिगृहीत पर खड़ी करता है। जब आगे के अध्यायों की
कोई उपपत्ति शून्य से कोई वास्तविक संख्या पैदा कर दे, तब छिपा हुआ \omterm{def:b1:reals:bounds}{उच्चतम}
ढूँढ़िए।
\end{remark}

\begin{remark}[विविक्तता और सघनता के बीच]
$\Z$ और $\Q$ $\R$ के भीतर दो विपरीत छोरों पर बैठते हैं: प्रत्येक पूर्णांक के
चारों ओर लंबाई $1$ का ऐसा \omterm{prop:b1:reals:intervals}{अंतराल} है जिसमें कोई दूसरा पूर्णांक नहीं है
(विविक्तता --- यही \omterm{thm:b1:reals:floor}{फ़र्श} को सुपरिभाषित बनाती है), जबकि किन्हीं दो वास्तविक
संख्याओं के बीच अनंत परिमेय संख्याएँ हैं (सघनता)। उल्लेखनीय है कि $\R$ के
\emph{योगात्मक उपसमूहों} के लिए बीच में कुछ है ही नहीं:
\cref{exo:b1:reals:9} सिद्ध करता है कि ऐसा \omterm{def:b1:structures:subgroup}{उपसमूह} या तो $\alpha\Z$ रूप का
(विविक्त) होता है या सघन --- और यही द्विभाजन \cref{ch:b1:seq} में $\{\sin
n\}$ की सघनता तथा \cref{pb:b1:continuity:1} के रचनात्मक राक्षस को शक्ति देता
है। साधारण \omterm{def:b1:logic:sets}{समुच्चय}, निस्संदेह, दोनों व्यवहार स्वतंत्र रूप से मिलाते हैं: $\Z
\cup \Q\cap\intcc{0}{1}$ दूर-दूर विविक्त है और बीच में सघन।
\end{remark}

\begin{method}[उच्चतम और निम्नतम के साथ समताएँ सिद्ध करना]\label{met:b1:reals:supproofs}
$\sup A = s$ सिद्ध करने के लिए: जाँचिए कि $s$ $A$ को ऊपर से परिबद्ध करता है,
फिर प्रत्येक $\varepsilon > 0$ के लिए (या किसी अनुक्रम $\varepsilon = \frac
1n$ के लिए) $A$ का ऐसा अवयव प्रस्तुत कीजिए जो $s - \varepsilon$ से बड़ा हो।
उच्चतमों की तुलना के लिए प्रयोग कीजिए: $A \subseteq B \implies \sup A \leq
\sup B$; और सभी $a, b$ के लिए $\sup(A + B) = \sup A + \sup B$, जहाँ $A + B =
\{a + b\}$ (\cref{exo:b1:reals:5})। जब तक यह ज्ञात न हो कि $A$ अरिक्त और ऊपर
से परिबद्ध है, तब तक $\sup A$ कभी मत लिखिए।
\end{method}

\section{अंतराल}

\begin{proposition}[अंतरालों का अभिलक्षण]\label{prop:b1:reals:intervals}
उपसमुच्चय $I \subseteq \R$ \emph{अंतराल}\index{अंतराल} (परिचित प्रकारों
$\intoo{a}{b}$, $\intcc{a}{b}$, $\intco{a}{b}$, $\intoc{a}{b}$, अर्धरेखाएँ,
$\R$, $\emptyset$, एकल समुच्चयों में से एक) है यदि और केवल यदि वह
\emph{उत्तल} हो:
\[
\forall x, y \in I,\ \forall z \in \R, \quad
x \leq z \leq y \implies z \in I .
\]
\end{proposition}

\begin{proof}
सूचीबद्ध प्रत्येक प्रकार स्पष्टतः उत्तल है। विलोमतः, मान लीजिए $I$ उत्तल और
अरिक्त है। यदि $I$ नीचे से परिबद्ध है तो $a = \inf I$ रखिए, अन्यथा $a =
-\infty$; इसी प्रकार $b = \sup I$ या $+\infty$। हमारा दावा है $\intoo{a}{b}
\subseteq I \subseteq \intcc{a}{b}$ (जहाँ $\pm\infty$ पर स्पष्ट परिपाटियाँ
हैं)। दूसरी अंतर्विष्टि परिबंधों की परिभाषा ही है। पहली के लिए मान लीजिए $z
\in \intoo{a}{b}$: चूँकि $z > a$, अतः $z$ निचला परिबंध नहीं है (अथवा $a =
-\infty$), इसलिए किसी $x \in I$ के लिए $x < z$; इसी प्रकार किसी $y \in I$ के
लिए $y > z$; और उत्तलता $z \in I$ रख देती है।

अब दुहरी अंतर्विष्टि $\intoo{a}{b} \subseteq I \subseteq \intcc{a}{b}$ से
प्रकार पढ़ लेना शेष है: खुले \omterm{prop:b1:reals:intervals}{अंतराल} और उसके संवृत रूप के बीच दबे हुए \omterm{def:b1:logic:sets}{समुच्चय}
$\intoo{a}{b}$ से केवल (परिमित) सिरों की उपस्थिति या अनुपस्थिति में भिन्न
होते हैं। स्पष्ट रूप से: यदि $a, b \in \R$, तो $(a \in I,\ b \in I)$ की
चारों संभावनाएँ $\intoo{a}{b}$, $\intco{a}{b}$, $\intoc{a}{b}$,
$\intcc{a}{b}$ देती हैं (अपकर्षी स्थितियाँ $a = b$ सहित: $a \in I$ होने पर
एकल \omterm{def:b1:logic:sets}{समुच्चय}); यदि $a = -\infty$ और $b \in \R$, तो $\intoo{-\infty}{b}$ या
$\intoc{-\infty}{b}$ मिलता है; $a \in \R$, $b = +\infty$ के लिए सममित रूप
से; और $a = -\infty$, $b = +\infty$ $I = \R$ देता है। हर स्थिति सूची में है:
काम पूरा।
\end{proof}

\begin{remark}[उत्तलता ही सही कसौटी क्यों है]
यह प्रतिज्ञप्ति एक \emph{ज्यामितीय} परिभाषा (दस आकारों की सूची) को
\emph{एक-पंक्ति की तार्किक} कसौटी में बदल देती है, और व्यवहार में वही कसौटी
काम आती है: यह सिद्ध करने के लिए कि कोई \omterm{def:b1:logic:sets}{समुच्चय} \omterm{prop:b1:reals:intervals}{अंतराल} है, कभी यह मत ढूँढ़िए
कि वह दस आकारों में से कौन-सा है --- उत्तलता सत्यापित कीजिए और प्रकार छाँटने
का काम प्रतिज्ञप्ति पर छोड़ दीजिए। \cref{ch:b1:continuity} की मध्यवर्ती मान
प्रमेय ठीक इसी प्रकार कही जाएगी (``\omterm{prop:b1:reals:intervals}{अंतराल} का संतत प्रतिबिंब \omterm{prop:b1:reals:intervals}{अंतराल} होता
है''), और उसकी उपपत्ति उत्तलता उत्पन्न करती है, आकार नहीं।
\end{remark}

\begin{remark}[विस्तारित वास्तविक रेखा]
दो संकेत जोड़कर $\overline\R = \R \cup \{-\infty, +\infty\}$ में काम करना
सुविधाजनक है, और परिपाटियाँ ये हैं: $A$ के ऊपर से अपरिबद्ध होने पर $\sup A =
+\infty$, तथा $\sup \emptyset = -\infty$। तब $\R$ के \emph{प्रत्येक}
उपसमुच्चय का $\overline\R$ में \omterm{def:b1:reals:bounds}{उच्चतम} होता है --- यह संकेतन-सुविधा
\cref{ch:b1:seq} में सीमाओं के लिए स्वतंत्र रूप से प्रयुक्त होती है।
\end{remark}

\begin{example}[$\overline\R$ में संगणना]\label{ex:b1:reals:extendedcompute}
इन परिपाटियों के लागू रहते: $\sup \Z = +\infty$, $\inf \Z = -\infty$; $A =
\{n + (-1)^n n : n \in \N\} = \{0, 4, 0, 8, \dots\} \cup \{0\}$ के लिए $\sup
A = +\infty$ (सम पद $2n$ अपरिबद्ध हैं) और $\inf A = \min A = 0$; तथा
$\sup\emptyset = -\infty \leq \inf\emptyset = +\infty$ --- अर्थात् वह
एकमात्र \omterm{def:b1:logic:sets}{समुच्चय} जिसका \omterm{def:b1:reals:bounds}{उच्चतम} उसके \omterm{def:b1:reals:bounds}{निम्नतम} से \emph{छोटा} है, जो स्मरण दिलाता
है कि परिपाटियाँ इसलिए चुनी गई हैं कि अंतर्विष्टि के सापेक्ष $\sup$ वर्धमान
और $\inf$ ह्रासमान बना रहे:
\[
A \subseteq B \implies \sup A \leq \sup B
\quad\text{और}\quad \inf A \geq \inf B ,
\]
जो अब अरिक्तता की किसी शर्त के बिना वैध है। परिपाटियाँ जो \emph{नहीं} देतीं
वह है अंकगणित: $+\infty + (-\infty)$ और $0 \times (+\infty)$ अपरिभाषित रहते
हैं, और उच्चतमों की हर बीजीय हेराफेरी को पहले यह जाँचना होगा कि वह कभी
इन्हें न बनाए। विस्तारित रेखा लेखा-जोखा है, संख्या-पद्धति नहीं।
\end{example}

\begin{example}[वह उच्चतम जो $\Q$ से भाग निकला]\label{ex:b1:reals:supescape}
आरंभिक टिप्पणी वाले \omterm{def:b1:logic:sets}{समुच्चय} $A = \{x \in \Q : x^2 < 2\}$ पर लौटिए और उसका
\omterm{def:b1:reals:bounds}{उच्चतम} \emph{$\R$ में} संगणित कीजिए। वह अरिक्त ($1 \in A$) और $1.5$ से ऊपर
परिबद्ध है (यदि $x > 1.5$, तो $x^2 > 2.25 > 2$), अतः $s = \sup A$ विद्यमान
है। हमारा दावा है $s = \sqrt 2$ (\cref{exo:b1:reals:12} में बनी वास्तविक
संख्या)। \omterm{def:b1:reals:bounds}{ऊपरी परिबंध}: प्रत्येक $a \in A$ $a < \sqrt2$ संतुष्ट करता है --- $a
\leq 0$ के लिए यह स्पष्ट है, और $a > 0$ के लिए $a \geq \sqrt2$ $a^2 \geq 2$
दे देता। इससे छोटा कुछ काम नहीं करता: $t < \sqrt2$ दिया हो, तो सघनता
(\cref{thm:b1:reals:density}) ऐसा परिमेय $q$ देती है कि $\max(1, t) < q <
\sqrt 2$, और तब $q^2 < 2$, अतः $q \in A$ $t$ को पार कर जाता है।
\cref{prop:b1:reals:epsilon} से $s = \sqrt2 \notin \Q$। समापन का सार: परिमेय
संख्याओं के \omterm{def:b1:logic:sets}{समुच्चय} का \omterm{def:b1:reals:bounds}{उच्चतम} परिमेय होना आवश्यक नहीं --- पूर्णता ठीक यही
वचन है कि $\R$, $\Q$ के विपरीत, किसी \omterm{def:b1:reals:bounds}{उच्चतम} को कभी भागने नहीं देता; यही
उदाहरण अध्याय की आरंभिक टिप्पणी है, जो अब केवल इंगित नहीं, सिद्ध की जा चुकी
है।
\end{example}

\begin{remark}[इस खंड के भीतर आगे की दृष्टि]
इस अध्याय के तीनों औज़ारों के आगे अलग-अलग जीवन हैं। \omterm{def:b1:reals:bounds}{उच्चतम} विश्लेषण वाला आधा
भाग चलाता है: एकदिष्ट सीमाएँ (\cref{ch:b1:seq}), समाकल की परिभाषा ही
(\cref{ch:b1:integration}), और \cref{ch:b1:euclid} की ज्यामिति में किसी
बिंदु की किसी उपसमष्टि से दूरी --- अर्थात् वह \omterm{def:b1:reals:bounds}{निम्नतम} जिसे लांबिक प्रक्षेप
न्यूनतम में बदल देता है। \omterm{thm:b1:reals:floor}{फ़र्श फलन} वहीं लौटता है जहाँ विविक्त संतत से मिलता
है: अंक-प्रसार (इस अध्याय की सप्ताहांत समस्या), डिरिक्ले का कबूतरखाना
सन्निकटन (\cref{pb:b1:derivative:1}), योगों की समाकल से तुलना
(\cref{ch:b1:series})। सघनता के तर्क \cref{ch:b1:continuity} में एक विधि बन
जाते हैं: संतत फलनों की किसी सर्वसमिका को केवल $\Q$ पर जाँचना पर्याप्त है
--- कोशी के फलनीय समीकरण (\cref{pb:b1:continuity:1}) का आधा भाग ठीक यही चाल
है। जब संदेह हो कि इस खंड की कोई उपपत्ति अपने अस्तित्व-कथन कहाँ से पाती है,
तो उत्तर लगभग सदा यही अध्याय है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:reals:1}
इनके \omterm{def:b1:reals:bounds}{उच्चतम}, \omterm{def:b1:reals:bounds}{निम्नतम}, अधिकतम, न्यूनतम --- जब विद्यमान हों --- उपपत्ति सहित
निर्धारित कीजिए:
\[
A = \Bigl\{\frac{1}{n} : n \in \N^*\Bigr\},
\qquad
B = \Bigl\{\frac{(-1)^n n}{n+1} : n \in \N\Bigr\},
\qquad
C = \{x \in \R : x^2 < 3\}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:reals:2}
सिद्ध कीजिए कि सभी $x, y \in \R$ के लिए $\lfloor x \rfloor + \lfloor y
\rfloor \leq \lfloor x + y \rfloor \leq \lfloor x \rfloor + \lfloor y
\rfloor + 1$, और यह कि दोनों परिबंध प्राप्त होते हैं।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:reals:3}
सिद्ध कीजिए कि प्रत्येक $x \in \R$ और $n \in \N^*$ के लिए $\Bigl\lfloor
\frac{\lfloor nx \rfloor}{n} \Bigr\rfloor = \lfloor x \rfloor$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:reals:4}
मान लीजिए $A \subseteq B$ $\R$ के अरिक्त उपसमुच्चय हैं और $B$ परिबद्ध हैं।
सिद्ध कीजिए $\inf B \leq \inf A \leq \sup A \leq \sup B$।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:5}
अरिक्त परिबद्ध $A, B \subseteq \R$ के लिए $A + B = \{a + b : a \in A,\ b \in
B\}$ और $-A = \{-a : a \in A\}$ परिभाषित कीजिए। सिद्ध कीजिए:
\[
\sup(A + B) = \sup A + \sup B,
\qquad
\sup(-A) = -\inf A .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:6}
मान लीजिए $f, g \colon E \to \R$ परिबद्ध फलन हैं। सिद्ध कीजिए
\[
\sup_{x \in E}\, \bigl(f(x) + g(x)\bigr) \leq \sup_{x \in E} f(x) +
\sup_{x \in E} g(x),
\]
और ऐसा उदाहरण दीजिए जिसमें असमिका कठोर हो। यह \cref{exo:b1:reals:5} का विरोध
क्यों नहीं करता?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:7}
सिद्ध कीजिए कि $\sqrt 2 + \sqrt 3$ अपरिमेय है। \emph{(उसका वर्ग कीजिए और
$\sqrt 6$ की अपरिमेयता का प्रयोग कीजिए, जिसे \cref{exo:b1:arith:7} के द्वारा
सिद्ध किया जाना है।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:8}
सिद्ध कीजिए कि द्विआधारी परिमेय संख्याओं का \omterm{def:b1:logic:sets}{समुच्चय} $D =
\bigl\{\frac{m}{2^n} : m \in \Z,\ n \in \N\bigr\}$ $\R$ में सघन है: किन्हीं
दो वास्तविक संख्याओं के बीच एक द्विआधारी परिमेय संख्या होती है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:reals:9}
मान लीजिए $G$ $(\R, +)$ का ऐसा \omterm{def:b1:structures:subgroup}{उपसमूह} है कि $G \neq \{0\}$। $\alpha =
\inf\,(G \cap \intoo{0}{+\infty})$ रखिए। सिद्ध कीजिए:
\begin{enumerate}
  \item यदि $\alpha > 0$, तो $G = \alpha\Z$;
  \item यदि $\alpha = 0$, तो $G$ $\R$ में सघन है।
\end{enumerate}
उससे निकालिए कि $\Z + \sqrt 2\,\Z$ $\R$ में सघन है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:reals:10}
धनात्मक वास्तविक संख्याओं के अरिक्त समुच्चयों $A, B$ के लिए $AB = \{ab : a
\in A, b \in B\}$ रखिए। सिद्ध कीजिए $\sup(AB) = \sup A \cdot \sup B$
(परिबद्ध स्थिति में), और उदाहरण से दिखाइए कि धनात्मकता अनिवार्य है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:11}
अरिक्त परिबद्ध $A \subseteq \R$ के लिए \emph{व्यास} परिभाषित कीजिए
\[
\operatorname{diam} A = \sup\,\{\abs{a - a'} : a, a' \in A\} .
\]
सिद्ध कीजिए कि $\operatorname{diam} A = \sup A - \inf A$, और यह कि
$\intcc{\inf A}{\sup A}$ $A$ को समेटने वाला लघुतम संवृत \omterm{prop:b1:reals:intervals}{अंतराल} है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:reals:12}
मान लीजिए $y > 0$ और $E = \{x \geq 0 : x^2 \leq y\}$। सिद्ध कीजिए कि $E$
अरिक्त और ऊपर से परिबद्ध है, और यह कि $s = \sup E$ $s^2 = y$ संतुष्ट करता है
\emph{(हर स्थिति में एक छोटा $h > 0$ प्रस्तुत करके, जो \omterm{def:b1:reals:bounds}{उच्चतम} की परिभाषा का
विरोध करे, $s^2 < y$ और $s^2 > y$ को हटा दीजिए)}। उससे निकालिए कि प्रत्येक
$y > 0$ का अद्वितीय वर्गमूल $\sqrt y > 0$ है और $y \mapsto \sqrt y$
$\intoo{0}{+\infty}$ पर वर्धमान है।
\end{exercise}

\section{समस्या: अंक-प्रसार और परिमेय संख्याओं की लय}

\begin{problem}[{सप्ताहांत समस्या --- $b$-आदिक प्रसार: अस्तित्व, अद्वितीयता, और आवर्तिता
$\Q$ को अभिलक्षित करती है}]
\label{pb:b1:reals:1}
$\intco{0}{1}$ की प्रत्येक वास्तविक संख्या का प्रत्येक आधार $b \geq 2$ में
अंक-प्रसार होता है; अंक $b - 1$ की अंतहीन मालाओं पर रोक लगा देने पर वह
प्रसार अद्वितीय हो जाता है; और वह ठीक तब अंततः आवर्ती होता है जब संख्या
परिमेय हो। यह समस्या तीनों तथ्य केवल पूर्णता अभिगृहीत से सिद्ध करती है ---
कोई अनुक्रम नहीं, कोई श्रेणी नहीं: केवल \omterm{def:b1:reals:bounds}{उच्चतम}, \omterm{thm:b1:reals:archimedes}{आर्किमिडीज़ गुणधर्म} और \omterm{thm:b1:reals:floor}{फ़र्श
फलन} --- और अंक-रूप में कैंटर के विकर्ण तर्क पर समाप्त होती है। आगे सर्वत्र
$b \geq 2$ एक नियत पूर्णांक है (\emph{आधार}), \emph{अंक} $\intint{0}{b-1}$
का अवयव है, और अंक-माला $(d_n)_{n \geq 1}$ \emph{उचित} तब है जब वह अंततः $b
- 1$ के बराबर न हो जाए (अर्थात्: प्रत्येक $N$ के लिए ऐसा $n > N$ है कि $d_n
\leq b - 2$)।\index{आदिक प्रसार@$b$-आदिक प्रसार}\index{दशमलव प्रसार}

\medskip \noindent\textbf{भाग I --- हाथ से अंक।} आधार $b$ में $p$ का $q$ से
लंबा भाग: वर्तमान शेषफल को $b$ से गुणा कीजिए, $q$ से भाग दीजिए, भागफल को
अगला अंक लिखिए, और शेषफल रख लीजिए।
\begin{enumerate}
  \item आधार $10$ में कलनविधि को $\frac 18$ पर और $\frac 17$ पर चलाइए, और हर
  पद पर अंक \emph{तथा} शेषफल लिखते जाइए। जाँचिए कि $\frac 17$ के शेषफल $1,
  3, 2, 6, 4, 5$ में चक्कर लगाते हैं और अंक $142857$ फिर सदा के लिए दोहराते
  हैं।
  \item $\frac 13$ और $\frac{5}{16}$ के आधार-$2$ प्रसार, तथा $\frac 12$ का
  आधार-$3$ प्रसार संगणित कीजिए। देखिए: एक संख्या समाप्त हो जाती है, दो
  दोहराती हैं --- और $\frac 12$, जो आधार $10$ में इतना सीधा है, आधार $3$ में
  सदा दोहराता रहता है।
  \item न्यूनतम पदों में $x = \frac pq \in \intco{0}{1}$ के लिए दिखाइए कि
  कलनविधि से बने अंक अंततः सभी $0$ हैं यदि और केवल यदि किसी $N$ के लिए शेषफल
  $b^N p \bmod q$ लुप्त हो जाए, यदि और केवल यदि $q$ किसी घात $b^N$ को
  विभाजित करे, यदि और केवल यदि $q$ का प्रत्येक \omterm{def:b1:arith:prime}{अभाज्य} गुणनखंड $b$ को विभाजित
  करे। जाँचिए: $\frac{1}{20}$ आधार $10$ में समाप्त होता है, $3$ में नहीं।
  \item \emph{कटाई} $s_n = \lfloor b^n x \rfloor / b^n$ परिभाषित कीजिए। $x =
  \sqrt 2$ और $b = 10$ के लिए $s_0, \dots, s_4$ की संगणना कीजिए, और हर पद पर
  सत्यापित कीजिए कि दो क्रमागत वर्ग $2$ को घेर लेते हैं (उदाहरणार्थ
  $1.4142^2 = 1.99996164 < 2 < 2.00024449 = 1.4143^2$), तथा हर बार $s_n \leq
  \sqrt 2 < s_n + 10^{-n}$ जाँचिए।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- \omterm{def:b1:reals:bounds}{उच्चतम} से अस्तित्व।} $x \in
\intco{0}{1}$ नियत कीजिए और $A_n = \lfloor b^n x \rfloor$ तथा $n \geq 1$ के
लिए $d_n = A_n - b\,A_{n-1}$ रखिए।
\begin{enumerate}[resume]
  \item दिखाइए $A_0 = 0$ और $b\,A_{n-1} \leq A_n \leq b\,A_{n-1} + b - 1$;
  निष्कर्ष निकालिए कि प्रत्येक $d_n$ एक अंक है।
  \item दिखाइए कि $s_n := A_n b^{-n}$ यह संतुष्ट करता है
        \[
        s_n = \sum_{k=1}^{n} d_k\,b^{-k}
        \qquad\text{और}\qquad
        s_n \leq x < s_n + b^{-n} .
        \]
  \item आगमन से $b^n \geq n + 1$ सिद्ध कीजिए, फिर दिखाइए कि $(s_n)$
  अह्रासमान है और $x = \sup_n s_n$ \emph{(\cref{prop:b1:reals:epsilon} और
  \cref{thm:b1:reals:archimedes} का प्रयोग कीजिए)}।
  \item दिखाइए कि माला $(d_n)$ उचित है: यदि सभी $k > N$ के लिए $d_k = b -
  1$, तो परिमित गुणोत्तर योग से $n > N$ के लिए $s_n$ संगणित कीजिए और प्रश्न
  6 का विरोध कीजिए।
  \item विलोमतः, मान लीजिए $(e_n)_{n \geq 1}$ कोई उचित अंक-माला है और $t_n =
  \sum_{k=1}^n e_k b^{-k}$। दिखाइए कि $y = \sup_n t_n$ विद्यमान है,
  $\intco{0}{1}$ में स्थित है, और प्रत्येक $n$ के लिए $t_n \leq y < t_n +
  b^{-n}$ संतुष्ट करता है \emph{(कठोर असमिका के लिए $m > n$ वाला अंक $e_m
  \leq b - 2$ प्रयोग कीजिए)}। उससे $\lfloor b^n y \rfloor = b^n t_n$
  निकालिए, और फिर यह कि प्रश्न 5 के अर्थ में $y$ के अंक ठीक $e_n$ हैं।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- अद्वितीयता, क्रम, विस्थापन।}
\begin{enumerate}[resume]
  \item प्रश्न 5--9 को \emph{$b$-आदिक प्रसार प्रमेय} में जोड़िए: \omterm{def:b1:logic:map}{प्रतिचित्रण}
  $x \mapsto (d_n)$ और $(e_n) \mapsto \sup_n t_n$ परस्पर प्रतिलोम एकैकी
  आच्छादन हैं, जो $\intco{0}{1}$ और उचित अंक-मालाओं के \omterm{def:b1:logic:sets}{समुच्चय} के बीच हैं।
  विशेष रूप से किन्हीं दो भिन्न उचित मालाओं का मान एक नहीं होता।
  \item अब अनुचित मालाएँ भी मानिए। दिखाइए कि जिस माला के लिए सभी $n > M$ पर
  $e_n = b - 1$ हो (जहाँ $M \geq 0$ न्यूनतम है), उसका मान $t_M + b^{-M}$ है;
  निष्कर्ष निकालिए कि आधार $10$ में $0.999\dots = 1$, और यह कि दो अंक-निरूपण
  वाली वास्तविक संख्याएँ ठीक $b$-आदिक भिन्नें $m/b^N \in \intoo{0}{1}$ हैं
  --- शेष हर वास्तविक संख्या का, अनुचित मालाओं के बीच भी, केवल एक निरूपण है।
  \item सिद्ध कीजिए कि प्रश्न 10 का एकैकी आच्छादन कोशीय क्रम के लिए
  क्रम-संरक्षी है: यदि $x$ और $y$ की उचित मालाएँ पहली बार सूचकांक $m$ पर
  भिन्न होती हैं, तो $x < y$ यदि और केवल यदि $d_m < e_m$।
  \item (विस्थापन प्रमेयिका) मान लीजिए $x \in \intco{0}{1}$ के अंक $(d_n)$
  हैं। दिखाइए कि $bx$ के भिन्नात्मक भाग के अंक $(d_{n+1})_{n \geq 1}$ हैं
  \emph{(पूर्णांक $K$ के लिए $\lfloor u - K \rfloor = \lfloor u \rfloor - K$
  का प्रयोग करके $\lfloor b^n(bx - A_1)\rfloor$ संगणित कीजिए)}, और आगमन से
  निकालिए कि $b^m x$ के भिन्नात्मक भाग के अंक $(d_{n+m})_{n \geq 1}$ हैं।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- परिमेयता ही आवर्तिता है।} मान लीजिए $x
= \frac pq \in \intco{0}{1}$ न्यूनतम पदों में है और $r_n = b^n p \bmod q$
$b^n p$ को $q$ से भाग देने पर यूक्लिडीय भाग का शेषफल है।
\begin{enumerate}[resume]
  \item दिखाइए $A_n = \dfrac{b^n p - r_n}{q}$ और $r_n = (b\,r_{n-1}) \bmod
  q$।
  \item दिखाइए $d_n = \Bigl\lfloor \dfrac{b\,r_{n-1}}{q} \Bigr\rfloor$:
  अर्थात् हर अंक केवल पिछले शेषफल का फलन है। यही भाग I का लंबा भाग है।
  \item $r_0, \dots, r_q$ पर कबूतरखाना सिद्धांत
  (\cref{cor:b1:counting:pigeonhole}) लगाइए और निष्कर्ष निकालिए: प्रत्येक
  परिमेय संख्या का प्रसार अंततः आवर्ती है, जिसका पूर्वावर्त और आवर्त दोनों
  अधिक से अधिक $q$ हैं।
  \item विलोमतः, मान लीजिए $y \in \intco{0}{1}$ के अंक \emph{शुद्ध रूप से}
  आवर्ती हैं: सभी $n \geq 1$ के लिए $d_{n+T} = d_n$। विस्थापन प्रमेयिका और
  प्रश्न 10 की अद्वितीयता का प्रयोग करके दिखाइए कि $b^T y$ का भिन्नात्मक भाग
  $y$ के बराबर है, और $(b^T - 1)\,y \in \N$ निकालिए: अतः $y$ परिमेय है और
  उसका हर $b^T - 1$ को \omterm{def:b1:arith:divides}{विभाजित करता है}। क्रियाविधि $0.(142857)$ पर सत्यापित
  कीजिए: $142857 \times 7 = 999999$।
  \item विस्थापन करके अंततः आवर्ती स्थिति सँभालिए, और \emph{आवर्तिता कसौटी}
  कहिए: $x \in \intco{0}{1}$ परिमेय है यदि और केवल यदि उसका उचित $b$-आदिक
  प्रसार अंततः आवर्ती हो --- और यह किसी एक आधार में तभी होता है जब सभी
  आधारों में हो।
  \item $\gcd(q, b) = 1$ वाले $x = \frac 1q$ के लिए दिखाइए कि प्रसार शुद्ध
  रूप से आवर्ती है और उसका लघुतम आवर्त वह लघुतम $T \geq 1$ है जिसके लिए $b^T
  \equiv 1 \pmod q$ ($q$ के सापेक्ष $b$ की गुणात्मक कोटि)। जाँचिए कि $q =
  7$, $b = 10$ के लिए $7$ के सापेक्ष $10$ की घातें $3, 2, 6, 4, 5, 1$ में
  चलती हैं: कोटि $6$, जो प्रश्न 1 से मेल खाती है।
\end{enumerate}

\medskip \noindent\textbf{भाग V --- लाभांश और विकर्ण।}
\begin{enumerate}[resume]
  \item मान लीजिए $x^*$ $\intco{0}{1}$ की वह वास्तविक संख्या है जिसके
  आधार-$10$ अंक त्रिभुजीय स्थानों $\frac{j(j + 1)}{2}$ ($j \geq 1$) पर $1$
  हैं और अन्यत्र $0$: $x^* = 0.101001000100001\dots$ दिखाइए कि उसकी अंक-माला
  उचित है पर अंततः आवर्ती नहीं \emph{(आवर्त $T$ होने पर अधिक से अधिक $T$ के
  अंतरालों पर इकाइयाँ आनी पड़तीं, पर \omterm{prop:b1:reals:intervals}{अंतराल} बढ़ते जाते हैं)}, और निष्कर्ष
  निकालिए कि $x^*$ अपरिमेय है: अर्थात् ऐसी संख्या जिसकी अपरिमेयता शुद्ध लय
  से सिद्ध हुई।
  \item दिखाइए कि प्रत्येक आधार $b$ के लिए \omterm{def:b1:logic:sets}{समुच्चय} $\{m/b^n : m \in \Z, n
  \in \N\}$ $\R$ में सघन है (\cref{exo:b1:reals:8} का सामान्यीकरण), और यह कि
  प्रत्येक परिमेय $\frac pq \in \intoo{0}{1}$ का आधार $q$ में \emph{समाप्त
  होने वाला} प्रसार है। उपदेश: समाप्त होना (संख्या, आधार) युग्म का गुणधर्म
  है; आवर्तिता --- अर्थात् परिमेयता --- आंतरिक है।
  \item (कैंटर का विकर्ण) मान लीजिए $k \mapsto x_k$ $\N^*$ से $\intco{0}{1}$
  तक कोई \omterm{def:b1:logic:map}{प्रतिचित्रण} है। अंक-माला इस प्रकार परिभाषित कीजिए: $e_k = 1$ यदि
  $x_k$ का $k$-वाँ अंक $1$ से भिन्न हो, और अन्यथा $e_k = 2$। दिखाइए कि
  $(e_k)$ उचित है, कि उसका मान $y$ $\intco{0}{1}$ में है, और यह कि प्रत्येक
  $k$ के लिए $y \neq x_k$। निष्कर्ष निकालिए: कोई भी \omterm{def:b1:logic:map}{प्रतिचित्रण} $\N^* \to
  \intco{0}{1}$ \omterm{def:b1:logic:inj}{आच्छादक} नहीं है। (गणनीयता की शब्दावली और इस प्रमेय का असली
  घर \cref{ch:b1:topology} है।)
  \item दिखाइए कि यदि $x$ और $y$ के उचित प्रसार सूचकांक $n$ तक मेल खाते हैं,
  तो $\abs{x - y} < b^{-n}$, और आधार $10$ में $x = 0.1$, $y = 0.0999$ से
  विलोम का खंडन कीजिए: संख्याओं की निकटता अंकों के मेल को बाध्य नहीं करती।
  दोष किन वास्तविक संख्याओं का है?
  \item आधार $b = 2$ में $x = \frac{1}{10}$ पर भाग IV चलाइए: शेषफल और अंक तब
  तक संगणित कीजिए जब तक वे चक्कर न लगाने लगें, और निष्कर्ष निकालिए
  $\frac{1}{10} = (0.0\overline{0011})_2$, जिसका पूर्वावर्त $1$ और आवर्त $4$
  है। प्रश्न 3 के द्वारा समझाइए कि कोई भी परिमित द्विआधारी माला
  $\frac{1}{10}$ के बराबर क्यों नहीं होगी --- यही कारण है कि संगणक का
  प्लवन-बिंदु $0.1 + 0.2$ ठीक-ठीक $0.3$ नहीं होता।
  \item संश्लेषण। एक-एक वाक्य में: उपपत्ति ने कहाँ प्रयोग किया (क) पूर्णता,
  (ख) \omterm{thm:b1:reals:archimedes}{आर्किमिडीज़ गुणधर्म}, (ग) \omterm{thm:b1:reals:floor}{फ़र्श} की अद्वितीयता वाला उपवाक्य, (घ)
  कबूतरखाना सिद्धांत? और उपदेश: $\intco{0}{1}$ उचित अंक-मालाओं से पूरी
  निष्ठा के साथ कूटित हो जाता है, और परिमेयता आवर्तिता के रूप में पढ़ ली
  जाती है --- फिर भी विश्लेषण अंकों के बदले \omterm{def:b1:reals:bounds}{उच्चतम} को पसंद करता है। क्यों?
  (दो अंक-मालाओं को जोड़ने पर विचार कीजिए।)
\end{enumerate}
\end{problem}
