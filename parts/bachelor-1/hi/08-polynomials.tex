\chapter{बहुपद}\label{ch:b1:poly}

\omterm{def:b1:poly:def}{बहुपद} बीजगणितज्ञ के प्रिय फलन हैं --- सिवाय इसके कि यहाँ उन्हें फलन नहीं
माना जाता, बल्कि किसी अनिर्धार्य $X$ में औपचारिक व्यंजक, जिन्हें क्रमविनिमेय
\omterm{def:b1:structures:ring}{वलय} के नियमों से जोड़ा और गुणा किया जाता है। यह सिद्धांत \cref{ch:b1:arith}
के साथ आश्चर्यजनक रूप से समांतर चलता है: एक यूक्लिडीय भाग, एक \omterm{thm:b1:arith:gcd}{महत्तम
समापवर्तक} और बेज़ू संबंध, अखंडनीय अवयव और एक अद्वितीय गुणनखंडन। आगे सर्वत्र
$K$ $\Q$, $\R$ या $\C$ दर्शाता है।

\section{वलय $K[X]$}

\begin{definition}[बहुपद, घात]\label{def:b1:poly:def}
$K$ में गुणांकों वाला \emph{बहुपद}\index{बहुपद} एक औपचारिक योग है
\[
P = a_0 + a_1 X + a_2 X^2 + \dots + a_n X^n
= \sum_{k} a_k X^k,
\]
जहाँ किसी सूचकांक से आगे सभी $a_k \in K$ शून्य हैं। स्वाभाविक योग और गुणनफल
\[
\Bigl(\sum_i a_i X^i\Bigr)\Bigl(\sum_j b_j X^j\Bigr)
= \sum_k \Bigl(\sum_{i+j=k} a_i b_j\Bigr) X^k,
\]
के साथ \omterm{def:b1:logic:sets}{समुच्चय} $K[X]$ क्रमविनिमेय \omterm{def:b1:structures:ring}{वलय} है। $P \neq 0$ की \emph{घात} $\deg P$
वह बृहत्तम $n$ है जिसके लिए $a_n \neq 0$; $a_n$ \emph{अग्र गुणांक} है ($a_n
= 1$ होने पर $P$ \emph{इकाई-अग्र}\index{इकाई-अग्र बहुपद} कहलाता है), और
परिपाटी से $\deg 0 = -\infty$। प्रत्येक बहुपद प्रतिस्थापन द्वारा $K$ पर एक
फलन $x \mapsto P(x)$ परिभाषित करता है।
\end{definition}

\begin{proposition}[घात के नियम; पूर्णांकीय प्रांत]\label{prop:b1:poly:degree}
$P, Q \in K[X]$ के लिए:
\[
\deg(P + Q) \leq \max(\deg P, \deg Q),
\qquad
\deg(PQ) = \deg P + \deg Q .
\]
फलस्वरूप $K[X]$ \omterm{def:b1:structures:field}{पूर्णांकीय प्रांत} है, और उसकी इकाइयाँ अशून्य अचर हैं।
\end{proposition}

\begin{proof}
योग का नियम स्पष्ट है (अधिकतम से आगे के गुणांक लुप्त हो जाते हैं)। गुणनफल के
लिए मान लीजिए $a_m$ और $b_n$ अग्र गुणांक हैं: $PQ$ में $X^{m+n}$ का गुणांक
$a_m b_n \neq 0$ है ($K$ \omterm{def:b1:structures:field}{क्षेत्र} है, अतः \omterm{def:b1:structures:field}{पूर्णांकीय प्रांत}), और उससे ऊपर के
सभी गुणांक लुप्त हो जाते हैं। यदि $P, Q \neq 0$, तो $\deg PQ = \deg P + \deg
Q \geq 0$, अतः $PQ \neq 0$: \omterm{def:b1:structures:field}{पूर्णांकीय प्रांत}। यदि $PQ = 1$, तो $\deg P +
\deg Q = 0$ $\deg P = \deg Q = 0$ पर बाध्य कर देता है: व्युत्क्रमणीय अवयव
व्युत्क्रमणीय अचर हैं, अर्थात् पूरा $K^*$।
\end{proof}

\begin{theorem}[यूक्लिडीय भाग]\label{thm:b1:poly:division}
मान लीजिए $B \neq 0$ के साथ $A, B \in K[X]$। बहुपदों का ठीक एक युग्म $(Q,
R)$ ऐसा है कि
\[
A = BQ + R, \qquad \deg R < \deg B .
\]
\end{theorem}

\begin{proof}
\emph{अस्तित्व}, $\deg A$ पर प्रबल आगमन से। यदि $\deg A < \deg B$, तो $(Q,
R) = (0, A)$ लीजिए। अन्यथा $m \geq n$ के साथ $A = a X^m + \dots$, $B = b X^n
+ \dots$ लिखिए; \omterm{def:b1:poly:def}{बहुपद} $A_1 = A - \frac ab X^{m-n} B$ की घात $< m$ है (अग्र
पद कट जाते हैं), अतः आगमन से $\deg R < \deg B$ के साथ $A_1 = BQ_1 + R$, और
$A = B(Q_1 + \frac ab X^{m-n}) + R$।

\emph{अद्वितीयता}: यदि $BQ + R = BQ' + R'$, तो $\deg(R' - R) < \deg B$ के
साथ $B(Q - Q') = R' - R$; घात के नियम से इससे $Q - Q' = 0$ अनिवार्य हो जाता
है, और फिर $R = R'$।
\end{proof}

\begin{example}\label{ex:b1:poly:division}
$A = X^4 + X^3 - 2X + 1$ को $B = X^2 + 1$ से भाग दीजिए:
\[
X^4 + X^3 - 2X + 1 = (X^2 + 1)(X^2 + X - 1) + (-3X + 2).
\]
(संगणना: पहले $X^2 B$ घटाइए, फिर $X B$, फिर $-B$; शेषफल $-3X + 2$ की घात $1
< 2$ है।)
\end{example}

\begin{method}[हॉर्नर विधि]\label{met:b1:poly:horner}
$x$ पर $P = a_nX^n + \dots + a_0$ का मान निकालने के लिए, अथवा $P$ को $X - x$
से भाग देने के लिए, घातों की संगणना से बचिए: गुणांक बाएँ से दाएँ पढ़िए और
\emph{$x$ से गुणा कीजिए, अगला गुणांक जोड़िए} दोहराते जाइए:
\[
b_n = a_n, \qquad b_{k} = a_{k} + x\,b_{k+1}
\quad (k = n-1, \dots, 0) .
\]
तब $b_0 = P(x)$, और पहले वाले $b_k$ भागफल के गुणांक हैं: $P = (X -
x)(b_nX^{n-1} + \dots + b_1) + b_0$ (प्रसार करके तुलना कीजिए)। उदाहरण: $x =
2$ पर $P = X^4 - 5X^3 + 6X^2 + 4X - 8$: $b$ हैं $1, -3, 0, 4, 0$, अतः $P(2)
= 0$ और $P = (X-2)(X^3 - 3X^2 + 4)$ --- अर्थात् लंबे भाग के बदले एक पंक्ति,
और भोले मान-निकालने के $\approx n^2/2$ के बदले $n$ गुणा। इसी विधि को उसी
बिंदु पर दोहराने से बहुलताएँ निकल आती हैं (\cref{ex:b1:poly:multexample} से
तुलना कीजिए)।
\end{method}

\begin{remark}[{$K[X]$ का अंकगणित}]
यूक्लिडीय भाग हाथ में आ जाने पर \cref{ch:b1:arith} का पूरा अंकगणित उन्हीं
उपपत्तियों के साथ $K[X]$ पर चला जाता है, जहाँ निरपेक्ष मान की भूमिका घात
निभाती है: \omterm{thm:b1:arith:gcd}{महत्तम समापवर्तक} (\omterm{def:b1:poly:def}{इकाई-अग्र} होने तक मानकीकृत), विस्तारित
\omterm{met:b1:arith:euclid}{यूक्लिडीय कलनविधि}, बेज़ू सर्वसमिका, गाउस की प्रमेयिका, अखंडनीय \omterm{def:b1:poly:def}{बहुपद} और
अद्वितीय गुणनखंडन। हम इन स्थानांतरित परिणामों का स्वतंत्र रूप से प्रयोग करते
हैं, और \cref{exo:b1:poly:6} उनमें से एक का अभ्यास कराता है।
\end{remark}

\section{मूल}

\begin{theorem}[गुणनखंड प्रमेय]\label{thm:b1:poly:factor}
मान लीजिए $P \in K[X]$ और $a \in K$। $X - a$ से भाग देने पर $P$ का शेषफल अचर
$P(a)$ होता है। विशेष रूप से
\[
P(a) = 0 \iff (X - a) \mid P .
\]
और अधिक व्यापक रूप से, $P$ के भिन्न मूल $a_1, \dots, a_r$ गुणनखंडन $P = (X -
a_1)\cdots(X - a_r)\, Q$ देते हैं।
\end{theorem}

\begin{proof}
भाग दीजिए: $\deg R < 1$ के साथ $P = (X - a) Q + R$, अतः $R$ कोई अचर $c$ है;
$X = a$ प्रतिस्थापित करने पर (प्रतिस्थापन योग और गुणनफल का आदर करता है)
$P(a) = c$ मिलता है। तुल्यता इसी से निकलती है। कई मूलों के लिए $r$ पर आगमन
कीजिए: स्थिति $r = 1$ अभी सिद्ध की गई तुल्यता ही है। \omterm{def:b1:logic:statement}{कथन} को $r - 1$ मूलों के
लिए मान लीजिए और मान लीजिए $a_1, \dots, a_r$ $P$ के भिन्न मूल हैं। $P = (X -
a_1)Q_1$ लिखिए; प्रत्येक $i \geq 2$ के लिए $a_i$ प्रतिस्थापित करने पर:
\[
0 = P(a_i) = (a_i - a_1)\,Q_1(a_i),
\qquad a_i - a_1 \neq 0 ,
\]
और चूँकि $K$ में कोई शून्य भाजक नहीं है, $Q_1(a_i) = 0$: अर्थात् $r - 1$
भिन्न बिंदु $a_2, \dots, a_r$ $Q_1$ के मूल हैं। आगमन परिकल्पना $Q_1 = (X -
a_2)\cdots(X - a_r)\,Q$ का गुणनखंडन कर देती है, और वापस प्रतिस्थापित करने पर
दावा मिल जाता है।
\end{proof}

\begin{corollary}[घात $n$ के बहुपद के अधिक से अधिक $n$ मूल होते हैं]\label{cor:b1:poly:nroots}
घात $n$ वाले अशून्य $P \in K[X]$ के $K$ में अधिक से अधिक $n$ भिन्न मूल होते
हैं। फलस्वरूप, $n + 1$ भिन्न बिंदुओं पर लुप्त होने वाला (घात $\leq n$ का)
\omterm{def:b1:poly:def}{बहुपद} शून्य \omterm{def:b1:poly:def}{बहुपद} है, और $n+1$ बिंदुओं पर मेल खाने वाले घात $\leq n$ के दो
\omterm{def:b1:poly:def}{बहुपद} बराबर होते हैं।
\end{corollary}

\begin{proof}
यदि $a_1, \dots, a_r$ भिन्न मूल हैं, तो \cref{thm:b1:poly:factor} $P =
(X-a_1)\cdots(X-a_r) Q$ देता है, अतः $n = \deg P \geq r$। दोनों परिणाम
विरोधाभास से और अंतर लेकर निकलते हैं।
\end{proof}

\begin{example}[सहायक-बहुपद वाली युक्ति]\label{ex:b1:poly:auxiliary}
मान लीजिए $P$ घात $\leq n$ का वह \omterm{def:b1:poly:def}{बहुपद} है जिसके लिए
\[
P(k) = \frac{k}{k+1} \qquad (k = 0, 1, \dots, n) ;
\]
वह विद्यमान और अद्वितीय है, जैसा नीचे \omterm{thm:b1:poly:lagrange}{लाग्रांज अंतर्वेशन} से मिलता है।
$P(n+1)$ क्या है? हर हटाइए: \omterm{def:b1:poly:def}{बहुपद} $Q = (X+1)P - X$ की घात $\leq n + 1$ है और
वह $n + 1$ बिंदुओं $0, 1, \dots, n$ पर लुप्त होता है, अतः
\cref{thm:b1:poly:factor} से
\[
Q = c\,X(X-1)(X-2)\cdots(X-n)
\]
किसी अचर $c$ के लिए। वहाँ मान लीजिए जहाँ $Q$ स्वतंत्र रूप से ज्ञात हो: $X =
-1$ पर $Q(-1) = 0 \cdot P(-1) + 1 = 1$, जबकि गुणनफल $(-1)(-2)\cdots(-1-n) =
(-1)^{n+1}(n+1)!$ के बराबर है; अतः $c = \frac{(-1)^{n+1}}{(n+1)!}$। अब $X =
n + 1$ पर मान लीजिए:
\[
(n+2)\,P(n+1) - (n+1) = Q(n+1) = c\,(n+1)! = (-1)^{n+1} ,
\]
अतः $P(n+1) = \dfrac{(n+1) + (-1)^{n+1}}{n+2}$: विषम $n$ के लिए वह $1$ के
बराबर है, और सम $n$ के लिए $\frac{n}{n+2}$ के --- अर्थात् अंतर्वेशी \omterm{def:b1:poly:def}{बहुपद}
प्रतिरूप $\frac{n+1}{n+2}$ को आगे \emph{नहीं} बढ़ाता। स्मरण रखने योग्य
युक्ति: आँकड़ों को किसी सहायक \omterm{def:b1:poly:def}{बहुपद} के मूलों के रूप में कूटित कीजिए, अज्ञात
अचर की पहचान आँकड़ों से बाहर किसी बिंदु पर कीजिए, और फल काटिए।
\end{example}

\begin{definition}[अवकलज, बहुलता]\label{def:b1:poly:derivative}
$P = \sum a_k X^k$ का \emph{औपचारिक अवकलज} $P' = \sum_{k \geq 1} k\,a_k
X^{k-1}$ है; वह सामान्य नियम $(P+Q)' = P' + Q'$, $(PQ)' = P'Q + PQ'$ संतुष्ट
करता है (एकपदों पर जाँचा गया और रैखिकता से विस्तारित)। $P$ के मूल $a$ की
\emph{बहुलता}\index{मूल की बहुलता} $m \geq 1$ तब है जब $(X-a)^m \mid P$ हो
पर $(X-a)^{m+1} \nmid P$ न हो; मूल $m = 1$ होने पर \emph{सरल} और $m \geq 2$
होने पर \emph{बहु} कहलाता है।
\end{definition}

\begin{proposition}[अवकलजों से बहुलता]\label{prop:b1:poly:multiplicity}
$a$ $P$ का \omterm{def:b1:poly:derivative}{बहुलता} $\geq m$ वाला मूल है यदि और केवल यदि
\[
P(a) = P'(a) = \dots = P^{(m-1)}(a) = 0 .
\]
विशेष रूप से, $a$ $P$ का बहु मूल है यदि और केवल यदि $P(a) = P'(a) = 0$।
\end{proposition}

\begin{proof}
$P = (X - a)^m Q + R$ लिखिए, जहाँ $R$ $(X-a)^m$ से भाग का शेषफल है, $\deg R
< m$। $k \leq m - 1$ बार अवकलन करके $a$ पर मान लेने पर: पहला पद $0$ का
योगदान करता है (हर अवकलज एक गुणनखंड $(X-a)$ बनाए रखता है), अतः $P^{(k)}(a) =
R^{(k)}(a)$।

अब घात $< m$ का \omterm{def:b1:poly:def}{बहुपद} $R$ $R(a), R'(a), \dots, R^{(m-1)}(a)$ से निर्धारित हो
जाता है: $R = \sum_{k < m} c_k (X - a)^k$ लिखने पर (यह संभव है: $X = (X - a)
+ a$ की घातों का प्रसार कीजिए) $R^{(k)}(a) = k!\, c_k$ मिलता है। अतः: $k <
m$ के लिए सभी $P^{(k)}(a) = 0$ $\iff$ सभी $c_k = 0$ $\iff$ $R = 0$ $\iff$
$(X-a)^m \mid P$।
\end{proof}

\begin{example}[बहुलता की संगणना]\label{ex:b1:poly:multexample}
$P = X^4 - 5X^3 + 6X^2 + 4X - 8$ में मूल $2$ की \omterm{def:b1:poly:derivative}{बहुलता} क्या है? $2$ पर
क्रमिक अवकलजों का मान लीजिए:
\[
P(2) = 16 - 40 + 24 + 8 - 8 = 0, \qquad
P'(2) = 32 - 60 + 24 + 4 = 0,
\]
\[
P''(2) = 48 - 60 + 12 = 0, \qquad
P'''(2) = 48 - 30 = 18 \neq 0
\]
(जहाँ $P' = 4X^3 - 15X^2 + 12X + 4$, $P'' = 12X^2 - 30X + 12$, $P''' = 24X -
30$)। तीन लुप्त मान और फिर एक अशून्य: अतः \omterm{def:b1:poly:derivative}{बहुलता} ठीक $3$। भाग देने पर $P =
(X - 2)^3(X + 1)$ --- जिसकी जाँच $(X-2)^3 = X^3 - 6X^2 + 12X - 8$ का प्रसार
करके और $X + 1$ से गुणा करके की जा सकती है। सार: बहुलताएँ \emph{मानों} से
पढ़ी जाती हैं, किसी गुणनखंडन की आवश्यकता नहीं --- और ठीक इसी तरह उन्हें तब
पकड़ा जाता है जब गुणनखंडन पहुँच से बाहर हो।
\end{example}

\begin{example}[महत्तम समापवर्तक से बहु मूल पकड़ना]\label{ex:b1:poly:gcdmultiple}
जब कोई मूल ज्ञात न हो, तब भी \cref{prop:b1:poly:multiplicity} एक
\emph{समग्र} बहु-मूल संसूचक देता है: $a$ $P$ का बहु मूल है तभी जब वह $P$ और
$P'$ का उभयनिष्ठ मूल हो, अतः $P$ का (किसी $\C$ में) बहु मूल है तभी जब
$\gcd(P, P') \neq 1$ --- और यह \omterm{met:b1:arith:euclid}{यूक्लिडीय कलनविधि} से, बिना कुछ हल किए,
संगणनीय है। नमूना: $P = X^3 - 3X + 2$, $P' = 3X^2 - 3 = 3(X - 1)(X + 1)$।
$P$ के भीतर $P'$ के मूल $\pm1$ जाँचिए: $P(1) = 0$, पर $P(-1) = 4$, अतः
\[
\gcd(P, P') = X - 1 :
\]
मूल $1$ बहु है; दो बार भाग देने पर $P = (X - 1)^2(X + 2)$। \omterm{thm:b1:arith:gcd}{महत्तम समापवर्तक}
तो बहु मूलों का पूरा \omterm{def:b1:logic:sets}{समुच्चय} बता देता है, प्रत्येक की \omterm{def:b1:poly:derivative}{बहुलता} एक कम करके ---
यही तथ्य हर संगणक-बीजगणित प्रणाली किसी भी मूल-खोज से पहले ``वर्ग-मुक्त
गुणनखंडन'' के लिए प्रयोग करती है, और यह \cref{exo:b1:poly:9} के बहु-मूल-रहित
तर्कों का बहुपदीय जुड़वाँ है।
\end{example}

\begin{theorem}[बीजगणित की मूल प्रमेय]\label{thm:b1:poly:fta}
$\C[X]$ के प्रत्येक अनचर \omterm{def:b1:poly:def}{बहुपद} का $\C$ में कोई मूल है।\index{बीजगणित की मूल
प्रमेय}
\end{theorem}
\admitted

\begin{remark}
नाम के बावजूद यह प्रमेय \emph{विश्लेषण} का \omterm{def:b1:logic:statement}{कथन} है: उसकी हर ज्ञात उपपत्ति
$\R$ की पूर्णता को किसी न किसी रूप में प्रयोग करती है, और कोई भी विशुद्ध रूप
से बीजीय नहीं है --- ईमानदार उपपत्ति स्नातक वर्ष 3 के खंड में दी गई है, जब
सम्मिश्र समाकलन या संहतता के तर्क उपलब्ध हो जाते हैं। यह अध्याय जो सचमुच
सिद्ध करता है वह \emph{न्यूनीकरण} है: प्रत्येक अनचर \omterm{def:b1:poly:def}{बहुपद} के लिए एक मूल मान
लेने पर नीचे $\C$ तथा $\R$ पर के पूरे गुणनखंडन शुद्ध बीजगणित से निकल आते
हैं।
\end{remark}

\begin{corollary}[$\C$ पर और $\R$ पर गुणनखंडन]\label{cor:b1:poly:factorization}
\begin{enumerate}
  \item प्रत्येक अशून्य $P \in \C[X]$ इस प्रकार गुणनखंडित होता है
        \[
        P = c\, (X - a_1)^{m_1} \cdots (X - a_r)^{m_r},
        \]
        जहाँ $c$ अग्र गुणांक है, $a_i$ भिन्न सम्मिश्र मूल हैं, $\sum m_i =
        \deg P$: \emph{\omterm{def:b1:poly:derivative}{बहुलता} सहित गिनने पर घात $n$ के \omterm{def:b1:poly:def}{बहुपद} के ठीक $n$
        सम्मिश्र मूल होते हैं}।
  \item प्रत्येक अशून्य $P \in \R[X]$ $\R$ पर इस प्रकार गुणनखंडित होता है
        \[
        P = c \prod_i (X - a_i)^{m_i} \prod_j (X^2 + p_j X +
        q_j)^{n_j},
        \]
        जहाँ द्विघात गुणनखंड भिन्न हैं और $p_j^2 - 4q_j < 0$ (कोई वास्तविक
        मूल नहीं)।
\end{enumerate}
\end{corollary}

\begin{proof}
(1) घात पर आगमन, \cref{thm:b1:poly:factor} से एक-एक करके मूल अलग करते हुए;
हर पद पर घातों की गणना मेल खाती है।

(2) मान लीजिए $P$ के गुणांक वास्तविक हैं। यदि $z$ \omterm{def:b1:poly:derivative}{बहुलता} $m$ वाला सम्मिश्र
मूल है, तो $\conj z$ भी है: $P(z) = 0$ का संयुग्मी लेने पर $P(\conj z) =
\conj{P(z)} = 0$ मिलता है (गुणांक अपने ही संयुग्मी हैं), और यही अवकलजों पर
भी लागू होता है (\cref{prop:b1:poly:multiplicity})। अवास्तविक मूलों को
संयुग्मी जोड़ों में समूहबद्ध कीजिए: प्रत्येक जोड़ा योगदान करता है
\[
(X - z)(X - \conj z) = X^2 - 2\Re(z)\, X + \abs z^2 ,
\]
जो ऋणात्मक विविक्तकर वाला वास्तविक द्विघात है। वास्तविक मूल रैखिक गुणनखंड
देते हैं।
\end{proof}

\begin{example}\label{ex:b1:poly:factorexample}
\cref{exo:b1:complex:5} में $X^4 + 4$ का $\R$ पर गुणनखंडन चारों सम्मिश्र
मूलों $\pm 1 \pm \iu$ को युग्मित करके किया गया था: $X^4 + 4 = (X^2 - 2X +
2)(X^2 + 2X + 2)$। कोई भी द्विघात $\R$ पर नहीं टूटता (विविक्तकर $-4$)। ध्यान
दीजिए: \emph{अखंडनीय} वास्तविक \omterm{def:b1:poly:def}{बहुपद} की घात $1$ या $2$ होती है --- गुणनखंडन
प्रमेय ठीक यही कहती है। वही संयुग्मी-युग्मन $X^4 + 1$ पर चलाइए, जिसके मूल
$\eu^{\pm\iu\pi/4}$ और $\eu^{\pm3\iu\pi/4}$ हैं: प्रत्येक जोड़ा $X^2 -
2\cos\theta\,X + 1$ का योगदान करता है, अतः
\[
X^4 + 1 = \bigl(X^2 - \sqrt2\,X + 1\bigr)
\bigl(X^2 + \sqrt2\,X + 1\bigr) ,
\]
यह ऐसी सर्वसमिका है जो $\Q$ पर भोले गुणनखंडन-प्रयासों को दिखाई नहीं देती ---
यही वास्तविक (यहाँ तो अपरिमेय) गुणांकों पर अड़े रहने की कीमत है, और
\cref{ch:b1:integration} में $\frac1{x^4 + 1}$ के समाकलन के लिए एक मानक
निवेश।
\end{example}

\begin{remark}[बहुपदों की सामान्य भूलें]\label{rem:b1:poly:pitfalls}
\begin{enumerate}
  \item \emph{\omterm{def:b1:poly:def}{बहुपद} बनाम फलन।} $K = \Q, \R, \C$ पर दोनों धारणाएँ मेल खाती
  हैं (बराबर फलनों के गुणांक बराबर होते हैं, \cref{cor:b1:poly:nroots} और
  $K$ के अनंत होने से), पर वैचारिक रूप से \omterm{def:b1:poly:def}{बहुपद} अपनी गुणांक-सूची \emph{ही}
  है: \cref{ch:b1:structures} के दो-अवयवी \omterm{def:b1:structures:field}{क्षेत्र} $\Z/2\Z$ पर $X^2 + X$
  दोनों बिंदुओं पर लुप्त होता है, फिर भी वह शून्य \omterm{def:b1:poly:def}{बहुपद} नहीं है।
  \item \emph{योग में घात।} अग्र पदों के कट जाने पर $\deg(P + Q)$ $\max(\deg
  P, \deg Q)$ से नीचे गिर सकती है; ``$\deg(P + Q) = \max(\dots)$'' लिखना
  केवल भिन्न घातों के लिए सुरक्षित है।
  \item \emph{मूलों की सही गणना।} \cref{cor:b1:poly:factorization} में ``$n$
  मूल'' का अर्थ है \emph{$\C$ में, \omterm{def:b1:poly:derivative}{बहुलता} सहित}: $X^2 + 1$ का कोई वास्तविक
  मूल नहीं है, और $(X-1)^2$ का एक भिन्न मूल है पर \omterm{def:b1:poly:derivative}{बहुलता} सहित दो। तीनों
  गणनाओं को मिला देना मिथ्या उपपत्तियों का सबसे आम स्रोत है।
  \item \emph{अखंडनीयता \omterm{def:b1:structures:field}{क्षेत्र} पर निर्भर करती है।} $X^2 - 2$ $\Q$ पर
  अखंडनीय है और $\R$ पर टूट जाता है; $X^2 + 1$ $\R$ पर अखंडनीय है और $\C$ पर
  टूट जाता है। जब तक गुणांक-क्षेत्र का नाम न लिया जाए, केवल ``अखंडनीय'' शब्द
  का कोई अर्थ नहीं है।
\end{enumerate}
\end{remark}

\section{गुणांक और मूल}

\begin{theorem}[व्येता के सूत्र]\label{thm:b1:poly:vieta}
मान लीजिए $P = X^n + c_{n-1} X^{n-1} + \dots + c_0$ \omterm{def:b1:poly:def}{इकाई-अग्र} है और उसके मूल
$a_1, \dots, a_n \in \C$ हैं (\omterm{def:b1:poly:derivative}{बहुलता} सहित)। तब\index{व्येता के सूत्र}
\[
\sum_i a_i = -c_{n-1},
\qquad
\sum_{i < j} a_i a_j = c_{n-2},
\qquad \dots, \qquad
a_1 a_2 \cdots a_n = (-1)^n c_0 ,
\]
जहाँ मूलों का $k$-वाँ सममित फलन $(-1)^k c_{n-k}$ है।
\end{theorem}

\begin{proof}
\cref{cor:b1:poly:factorization} से $P = (X - a_1)\cdots(X - a_n)$
(\omterm{def:b1:poly:def}{इकाई-अग्र}, सभी मूल सूचीबद्ध)। गुणनफल का वितरण नियम से प्रसार करने पर
प्रत्येक गुणनखंड में $X$ या मूल-पद $-a_i$ चुनने के हर तरीके के लिए एक पद
मिलता है: $i_1 < \dots < i_k$ से सूचीबद्ध गुणनखंडों में मूल और शेष $n - k$
में $X$ चुनने पर योगदान $(-a_{i_1})\cdots(-a_{i_k})\,X^{n-k}$ होता है। $X$
की घात के अनुसार समूहबद्ध करने पर:
\[
P = \sum_{k=0}^{n} (-1)^k
\Bigl(\sum_{i_1 < \dots < i_k} a_{i_1}\cdots a_{i_k}\Bigr)
X^{n-k} ,
\]
और $P = \sum_k c_{n-k}X^{n-k}$ से पहचान करने पर (गुणांक अद्वितीय हैं,
\cref{def:b1:poly:def}) $c_{n-k} = (-1)^k \sigma_k$ मिलता है, अर्थात्
$\sigma_k = (-1)^kc_{n-k}$, जहाँ $\sigma_k$ ऊपर दिखाया गया $k$-वाँ सममित फलन
है। तीनों दिखाई गई स्थितियाँ $k = 1$, $k = 2$ और $k = n$ हैं।
\end{proof}

\begin{example}\label{ex:b1:poly:vieta}
द्विघात $X^2 - sX + p$ के लिए: मूलों का योग $s$, गुणनफल $p$ --- जो पहले ही
बार-बार प्रयुक्त हो चुका है (\cref{exo:b1:complex:8})। मूलों $\alpha, \beta,
\gamma$ वाले \omterm{def:b1:poly:def}{इकाई-अग्र} त्रिघात $X^3 + aX^2 + bX + c$ के लिए:
\[
\alpha + \beta + \gamma = -a,
\quad
\alpha\beta + \beta\gamma + \gamma\alpha = b,
\quad
\alpha\beta\gamma = -c ,
\]
जिससे $\alpha^2 + \beta^2 + \gamma^2 = a^2 - 2b$ जैसी सममित राशियाँ बिना हल
किए संगणित की जा सकती हैं।
\end{example}

\begin{example}[मूलों को खोजे बिना उनका रूपांतरण]\label{ex:b1:poly:transformroots}
मान लीजिए $\alpha, \beta$ $X^2 - 3X + 1$ के मूल हैं। किस \omterm{def:b1:poly:def}{इकाई-अग्र} द्विघात
के मूल $\alpha^2, \beta^2$ हैं? व्येता से $\alpha + \beta = 3$ और
$\alpha\beta = 1$, अतः
\[
\alpha^2 + \beta^2 = (\alpha+\beta)^2 - 2\alpha\beta = 7,
\qquad
\alpha^2\beta^2 = (\alpha\beta)^2 = 1 :
\]
उत्तर है $X^2 - 7X + 1$ --- और वह भी $\alpha = \frac{3 + \sqrt5}2$ संगणित
किए बिना। (जाँच: $\alpha^2 = \frac{7 + 3\sqrt5}2$, और सचमुच $\alpha^2 +
\beta^2 = 7$।) यही रणनीति व्युत्क्रमों ($X^2 - \frac ba X + \frac ca$ प्रकार
के रूपांतरण), स्थानांतरण और किसी भी सममित आँकड़े को सँभाल लेती है: व्येता
\emph{अज्ञात} मूलों के प्रश्नों को \emph{ज्ञात} गुणांकों पर बीजगणित में बदल
देता है। जब मूल अभिलक्षणिक मान होंगे (\cref{ch:b1:det}), तब यह लगातार काम
आएगा।
\end{example}

\begin{example}[पूर्वापर-सम समीकरण]\label{ex:b1:poly:palindrome}
$X^4 + X^3 - 4X^2 + X + 1 = 0$ हल कीजिए। गुणांक दोनों दिशाओं में एक जैसे
पढ़े जाते हैं, अतः $0$ मूल नहीं है और $X^2$ से भाग देने पर कोई हल नहीं खोता:
\[
X^2 + X - 4 + \frac1X + \frac1{X^2} = 0 .
\]
$y = X + \frac1X$ रखिए: तब $X^2 + \frac1{X^2} = y^2 - 2$, और समीकरण सिमटकर
बन जाता है
\[
y^2 + y - 6 = 0 \iff (y + 3)(y - 2) = 0 .
\]
प्रत्येक मान को $X^2 - yX + 1 = 0$ से खोलिए: $y = 2$ के लिए $X^2 - 2X + 1 =
(X - 1)^2$ द्विक मूल $1$ देता है; $y = -3$ के लिए $X^2 + 3X + 1 = 0$ $X =
\frac{-3 \pm \sqrt5}2$ देता है। चतुर्घात के लिए \omterm{def:b1:poly:derivative}{बहुलता} सहित चार मूल, जैसा
\cref{cor:b1:poly:factorization} माँगता है --- और वह भी दो द्विघात हल करके।
यह युक्ति हर \emph{पूर्वापर-सम} \omterm{def:b1:poly:def}{बहुपद} पर लागू होती है: उनके मूल व्युत्क्रम
जोड़ों $\{x, 1/x\}$ में आते हैं ($X$ के स्थान पर $1/X$ रखिए और हर हटाइए), और
$y = X + \frac1X$ ठीक वही राशि है जो ऐसे जोड़ों पर अचर रहती है, जिससे घात
आधी हो जाती है।
\end{example}

\begin{theorem}[लाग्रांज अंतर्वेशन]\label{thm:b1:poly:lagrange}
मान लीजिए $x_0, \dots, x_n$ $K$ के भिन्न बिंदु हैं और $y_0, \dots, y_n \in
K$। घात $\leq n$ का ठीक एक $P \in K[X]$ ऐसा है कि सभी $i$ के लिए $P(x_i) =
y_i$, अर्थात्\index{लाग्रांज अंतर्वेशन}
\[
P = \sum_{i=0}^{n} y_i\, L_i,
\qquad
L_i = \prod_{j \neq i} \frac{X - x_j}{x_i - x_j} .
\]
\end{theorem}

\begin{proof}
प्रत्येक $L_i$ की घात $n$ है और वह $L_i(x_i) = 1$ संतुष्ट करता है, तथा $j
\neq i$ के लिए $L_i(x_j) = 0$ (हर गुणनखंड तदनुरूप $x_j$ पर लुप्त हो जाता
है)। अतः दिखाए गए $P$ की घात $\leq n$ है और वह अंतर्वेशन करता है।
अद्वितीयता: घात $\leq n$ के दो अंतर्वेशी \omterm{def:b1:poly:def}{बहुपद} $n+1$ बिंदुओं $x_i$ पर मेल
खाते हैं, अतः बराबर हैं (\cref{cor:b1:poly:nroots})।
\end{proof}

\begin{remark}[मध्यांतर: बहुपद सदिश भी हैं]\label{rem:b1:poly:interlude}
दृष्टिकोण का एक ऐसा परिवर्तन जिसे \cref{ch:b1:vspaces} औपचारिक बना देगा: घात
$\leq n$ के \omterm{def:b1:poly:def}{बहुपद} ऐसी समष्टि बनाते हैं जिसमें योग और अदिश गुणन ठीक
निर्देशांकों की भाँति व्यवहार करते हैं --- अर्थात् \omterm{def:b1:poly:def}{बहुपद} अपने $n + 1$
गुणांकों की सूची \emph{ही है}। इस अध्याय के तीन \omterm{def:b1:logic:statement}{कथन} गुप्त रूप से रैखिक
बीजगणित हैं। \omterm{thm:b1:poly:lagrange}{लाग्रांज अंतर्वेशन} (\cref{thm:b1:poly:lagrange}) कहता है कि
मान-आँकड़े $(P(x_0), \dots, P(x_n))$ $P$ को अद्वितीय रूप से निर्धारित कर
देते हैं: $n + 1$ बिंदुओं पर मान लेना एक रैखिक एकैकी आच्छादन है, और $L_i$
उसके अनुकूल आधार हैं। \cref{prop:b1:poly:multiplicity} की उपपत्ति में आया
प्रसार $R = \sum c_k(X - a)^k$ कहता है कि $(X - a)$ की घातें एक और
निर्देशांक-निकाय बनाती हैं, जिनमें निर्देशांक $c_k = R^{(k)}(a)/k!$ हैं। और
\cref{cor:b1:poly:nroots} --- घात से अधिक मूल होने पर \omterm{def:b1:poly:def}{बहुपद} शून्य हो जाता है
--- हर अद्वितीयता का इंजन है: वह \cref{ch:b1:findim} में ``विमा $n + 1$ की
समष्टि पर \omterm{def:b1:logic:inj}{एकैकी} रैखिक \omterm{def:b1:logic:map}{प्रतिचित्रण}'' बन जाएगा। जब वे अध्याय आएँगे, तब समष्टि
$K_n[X]$ उनका प्रिय उदाहरण होगी; वहाँ पहले से ही उसमें प्रवीण होकर पहुँचना
लाभदायक है।
\end{remark}

\begin{remark}[यह अध्याय कहाँ काम आता है]\label{rem:b1:poly:whereused}
$\R$ और $\C$ पर गुणनखंडन (\cref{cor:b1:poly:factorization})
\cref{ch:b1:fractions} में आंशिक भिन्नों का इंजन है, और इसीलिए
\cref{ch:b1:integration} में समाकलों के एक बड़े वर्ग का भी। \omterm{def:b1:poly:def}{बहुपद} का $(X -
a)$ की घातों में प्रसार, जो \cref{prop:b1:poly:multiplicity} की उपपत्ति में
मिला, \cref{ch:b1:taylor} के टेलर सूत्रों की बीजीय छाया है। \omterm{def:b1:diffeq:linear2}{अभिलक्षणिक बहुपद}
अवकल समीकरणों के लिए पहले ही आ चुके हैं (\cref{ch:b1:diffeq}) और
\cref{ch:b1:det} में आव्यूहों के लिए लौटते हैं; \omterm{thm:b1:poly:lagrange}{लाग्रांज अंतर्वेशन}
संख्यात्मक विश्लेषण की पहली अस्तित्व-और-अद्वितीयता प्रमेय है, और
\cref{exo:b1:poly:10} के \omterm{pb:b1:poly:1}{चेबिशेव बहुपद} --- जिनकी इष्टतमता नीचे दी गई
सप्ताहांत समस्या स्थापित करती है --- उस विषय को बताते हैं कि अंतर्वेशन
\emph{कहाँ} करना चाहिए। अंत में, \cref{ch:b1:arith} से नकल किया हुआ $K[X]$
का पूरा अंकगणित स्नातक वर्ष 2 के खंड में $K[X]$-गुणजावली और भागफल वलयों के
अध्ययन को पोषित करता है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:poly:1}
यूक्लिडीय भाग कीजिए: $X^5 - 1$ को $X^2 + X + 1$ से; फिर $2X^4 + X^3 - X + 3$
को $X^2 - 2$ से।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:2}
किन $n \in \N$ के लिए $X^2 + X + 1$ $X^{2n} + X^n + 1$ को \omterm{def:b1:arith:divides}{विभाजित करता है}?
\emph{संकेत: $X^2 + X + 1$ के मूल $j = \eu^{2\iu\pi/3}$ के साथ $j$ और $j^2$
हैं; $3$ के सापेक्ष $n$ पर विचार कीजिए।}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:3}
वे वास्तविक $a, b$ निर्धारित कीजिए जिनके लिए $(X-1)^2$ $P = X^4 + aX^3 +
bX^2 + 1$ को \omterm{def:b1:arith:divides}{विभाजित करता है}, फिर इन मानों के लिए $\R$ पर $P$ का गुणनखंडन
कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:4}
$\C$ पर और $\R$ पर गुणनखंडन कीजिए: $X^3 - 1$; $\;X^4 + X^2 + 1$; $\;X^6 -
1$।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:5}
मान लीजिए $P = X^3 - 6X^2 + 11X - 6$।
\begin{enumerate}
  \item परिमेय मूल खोजिए \emph{(\omterm{def:b1:poly:def}{इकाई-अग्र} पूर्णांक \omterm{def:b1:poly:def}{बहुपद} का न्यूनतम पदों में
  परिमेय मूल $p/q$ ऐसा पूर्णांक होता है जो अचर पद को \omterm{def:b1:arith:divides}{विभाजित करता है} --- इसे
  सिद्ध कीजिए)}, और $P$ का गुणनखंडन कीजिए।
  \item बिना हल किए, व्येता के द्वारा मूलों के वर्गों का योग और उनके
  प्रतिलोमों का योग संगणित कीजिए, और गुणनखंडन पर जाँचिए।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:6}
\omterm{met:b1:arith:euclid}{यूक्लिडीय कलनविधि} से $\gcd(X^4 - 1,\; X^3 - X^2 + X - 1)$ संगणित कीजिए, और
उसे दोनों बहुपदों के संयोजन $AU + BV$ के रूप में लिखिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:7}
मान लीजिए $P \in \R[X]$ ऐसा है कि सभी $x \in \R$ के लिए $P(x) \geq 0$। सिद्ध
कीजिए कि $P$ वास्तविक बहुपदों के दो वर्गों का योग है: $P = A^2 + B^2$।
\emph{संकेत: वास्तविक गुणनखंडन में वास्तविक मूलों की \omterm{def:b1:poly:derivative}{बहुलता} सम होती है;
द्विघात गुणनखंडों को $(X - z)(X - \conj z)$ के रूप में लिखिए और $(X - z)$ के
गुणनफल पर $\abs{\,\cdot\,}^2 = (\Re)^2 + (\Im)^2$ का प्रयोग कीजिए।}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:8}
घात $\leq 2$ का वह \omterm{def:b1:poly:def}{बहुपद} $P$ खोजिए जिसके लिए $P(0) = 1$, $P(1) = 3$, $P(2) =
2$ --- पहले लाग्रांज के सूत्र से, फिर गुणांकों पर रैखिक निकाय हल करके।
सत्यापित कीजिए कि दोनों उत्तर मेल खाते हैं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:9}
सिद्ध कीजिए कि $P = X^{2n+1} - 1$ का ठीक एक वास्तविक मूल है, और यह कि
प्रत्येक $n \geq 1$ के लिए \omterm{def:b1:poly:def}{बहुपद} $1 + X + \frac{X^2}{2!} + \dots +
\frac{X^n}{n!}$ का कोई बहु मूल नहीं है \emph{($P$ और $P'$ की तुलना कीजिए)}।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:poly:10}
(\omterm{pb:b1:poly:1}{चेबिशेव बहुपद}) $T_0 = 1$, $T_1 = X$ और $T_{n+1} = 2X\,T_n - T_{n-1}$
परिभाषित कीजिए।
\begin{enumerate}
  \item आगमन से सिद्ध कीजिए कि सभी $\theta$ के लिए $T_n(\cos\theta) = \cos
  n\theta$।
  \item उससे $T_n$ के $n$ मूल और उसका अग्र गुणांक निकालिए।
  \item सिद्ध कीजिए कि $\sup_{x \in \intcc{-1}{1}} \abs{T_n(x)} = 1$, जो
  $\intcc{-1}{1}$ के $n + 1$ बिंदुओं पर प्राप्त होता है।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:poly:11}
मान लीजिए $P \in \C[X]$ अनचर है और उसके भिन्न मूल $a_1, \dots, a_r$ हैं
(बहुलताएँ $m_1, \dots, m_r$)। परिमेय फलनों की सर्वसमिका सिद्ध कीजिए
\[
\frac{P'(X)}{P(X)} = \sum_{i=1}^{r} \frac{m_i}{X - a_i},
\]
और उससे गाउस--लूका प्रमेय निकालिए: $P'$ का प्रत्येक मूल $P$ के मूलों के
उत्तल आवरण में स्थित है \emph{(सर्वसमिका का मान $P'$ के ऐसे मूल $w$ पर लीजिए
जो $P$ का मूल न हो, संयुग्मी लीजिए, और परिणाम को इस रूप में पढ़िए कि $w$
$a_i$ का भारित औसत है)}।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:12}
(इकाई-मूल छननी) मान लीजिए $n \in \N^*$ और $j = \eu^{2\iu\pi/3}$। $1$, $j$
तथा $j^2$ पर $(1 + X)^n$ का मान लेकर सिद्ध कीजिए कि
\[
\sum_{k \geq 0} \binom{n}{3k}
= \frac{2^n + 2\cos\frac{n\pi}{3}}{3} ,
\]
और $n = 3$ तथा $n = 6$ के लिए सूत्र जाँचिए। \emph{संकेत: $1 + j^m + j^{2m}$
$3 \mid m$ होने पर $3$ के बराबर है और अन्यथा $0$; तथा $1 + j =
\eu^{\iu\pi/3}$।}
\end{exercise}

\section{समस्या: चेबिशेव बहुपद और सबसे चपटा बहुपद}

\begin{problem}\label{pb:b1:poly:1}
घात $n$ के सभी \emph{\omterm{def:b1:poly:def}{इकाई-अग्र}} बहुपदों में कौन $\intcc{-1}1$ पर शून्य के
सबसे निकट रहता है? उत्तर --- चेबिशेव की प्रमेय, जो सन्निकटन सिद्धांत का
जन्म-प्रमाणपत्र है --- $2^{1-n}T_n$ है, जहाँ $T_n$ \cref{exo:b1:poly:10} का
\omterm{pb:b1:poly:1}{चेबिशेव बहुपद} है, और कोई \omterm{def:b1:poly:def}{इकाई-अग्र} प्रतिद्वंद्वी उसके विचलन $2^{1-n}$ को
नहीं हरा सकता।\index{चेबिशेव बहुपद} यह समस्या कुल $(T_n)$ का बीजगणित (संयोजन
नियम, स्पष्ट गुणांक, द्वितीय प्रकार का कुल $U_n$, एक अवकल समीकरण) विकसित
करती है, चरम प्रमेय को उसकी समता-स्थिति सहित सिद्ध करती है, और अनुप्रयोग
बटोरती है: इष्टतम अंतर्वेशन बिंदु, $\cos 36^\circ$ का यथार्थ मान, और
\omterm{def:b1:arith:congruence}{सर्वांगसमता} $T_p \equiv X^p \pmod p$। आगे सर्वत्र $T_0 = 1$, $T_1 = X$,
$T_{n+1} = 2X\,T_n - T_{n-1}$, और हम \cref{exo:b1:poly:10} के
$T_n(\cos\theta) = \cos n\theta$ का स्वतंत्र रूप से प्रयोग करते हैं।

\medskip \noindent\textbf{भाग I --- कुल $(T_n)$।}
\begin{enumerate}
  \item पुनरावृत्ति से $T_2, T_3, T_4, T_5$ संगणित कीजिए। ($T_3$ की तुलना
  \cref{ex:b1:complex:cos3} की सर्वसमिका $\cos3\theta = 4\cos^3\theta -
  3\cos\theta$ से कीजिए।)
  \item आगमन से सिद्ध कीजिए: $\deg T_n = n$, जिसका अग्र गुणांक $n \geq 1$ के
  लिए $2^{n-1}$ है, और $T_n$ की सम-विषमता $n$ जैसी है (केवल सम या केवल विषम
  घातें आती हैं)।
  \item अद्वितीयता का सिद्धांत सिद्ध कीजिए: $T_n$ \emph{एकमात्र} ऐसा \omterm{def:b1:poly:def}{बहुपद}
  है जो सभी $\theta$ के लिए $P(\cos\theta) = \cos n\theta$ संतुष्ट करता है।
  ($\intcc{-1}1$ पर मेल खाने वाले दो \omterm{def:b1:poly:def}{बहुपद} सर्वत्र मेल खाते हैं:
  \cref{cor:b1:poly:nroots}।)
  \item संयोजन और गुणनफल के नियम निकालिए:
        \[
        T_m \circ T_n = T_{mn},
        \qquad
        2\,T_m T_n = T_{m+n} + T_{\abs{m-n}} .
        \]
  \item \cref{exo:b1:poly:10} से मूल $x_k = \cos\frac{(2k+1)\pi}{2n}$ और
  $T_n(y_k) = (-1)^k$ वाले समदोलन बिंदु $y_k = \cos\frac{k\pi}n$ स्मरण
  कीजिए। $\R$ पर $T_n$ का पूरा गुणनखंडन लिखिए, और न्यायसंगत ठहराइए कि $y_k$
  एक-दूसरे के बीच आते हैं: $y_n < x_{n-1} < y_{n-1} < \dots < x_0 < y_0$।
  \item सिद्ध कीजिए कि सभी $t \in \R$ के लिए $T_n(\cosh t) = \cosh(nt)$ (वही
  आगमन, \cref{prop:b1:functions:hyprules} का प्रयोग करते हुए), और $x \geq 1$
  के लिए संवृत रूप निकालिए
        \[
        T_n(x) = \frac{\bigl(x + \sqrt{x^2 - 1}\bigr)^n +
        \bigl(x - \sqrt{x^2 - 1}\bigr)^n}{2} ,
        \]
        अतः $x > 1$ के लिए $T_n(x) > 1$: $\intcc{-1}1$ के बाहर \omterm{def:b1:poly:def}{बहुपद} तुरंत
        भाग निकलता है।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- गुणांक, कुल $U_n$, एक अवकल समीकरण।}
\begin{enumerate}[resume]
  \item दे मॉयवर के सूत्र (\cref{cor:b1:complex:demoivre}) से स्पष्ट व्यंजक
  सिद्ध कीजिए
        \[
        T_n(x) = \sum_{0 \leq 2j \leq n} \binom{n}{2j}\,
        x^{\,n-2j}\,(x^2 - 1)^j ,
        \]
        और उसे $n = 3$ के लिए सत्यापित कीजिए।
  \item सभी $n$ के लिए $T_n(1)$, $T_n(-1)$ और $T_n(0)$ संगणित कीजिए।
  \item $U_0 = 1$, $U_1 = 2X$, $U_{n+1} = 2X\,U_n - U_{n-1}$ से $U_n$
  (\emph{द्वितीय प्रकार}) परिभाषित कीजिए। सिद्ध कीजिए कि $\theta \notin
  \pi\Z$ के लिए $U_n(\cos\theta) = \frac{\sin(n+1)\theta}{\sin\theta}$, और
  $n \geq 1$ के लिए $T_n' = n\,U_{n-1}$।
  \item सिद्ध कीजिए कि सभी $\theta$ के लिए $\abs{\sin n\theta} \leq
  n\,\abs{\sin\theta}$ (आगमन), और उससे मार्कोव प्रकार का परिबंध निकालिए
        \[
        \abs{T_n'(x)} \leq n^2
        \quad\text{पर } \intcc{-1}1,
        \qquad\text{के साथ } T_n'(\pm1) = (\pm1)^{n-1}\,n^2 .
        \]
  \item दिखाइए कि $y = T_n$ अवकल समीकरण
        \[
        (1 - x^2)\,y'' - x\,y' + n^2\,y = 0 ,
        \]
        संतुष्ट करता है --- इसके लिए सर्वसमिका $\sin\theta\,
        T_n'(\cos\theta) = n\sin n\theta$ का $\theta$ के सापेक्ष अवकलन
        कीजिए; $T_2$ के लिए सीधे सत्यापित कीजिए।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- चेबिशेव की चरम प्रमेय।} मान लीजिए
$\widetilde T_n = 2^{1-n}\,T_n$ (प्रश्न 2 से \omterm{def:b1:poly:def}{इकाई-अग्र}) और $\norm{P}_\infty
= \sup_{x \in \intcc{-1}1}\abs{P(x)}$ लिखिए।
\begin{enumerate}[resume]
  \item न्यायसंगत ठहराइए कि $\norm{\widetilde T_n}_\infty = 2^{1-n}$, जो $n
  + 1$ बिंदुओं $y_n < \dots < y_0$ पर एकांतरित चिह्नों के साथ प्राप्त होता
  है।
  \item मान लीजिए घात $n$ के किसी \omterm{def:b1:poly:def}{इकाई-अग्र} $P$ के लिए $\norm P_\infty <
  2^{1-n}$ होता, और $D = \widetilde T_n - P$ रखिए। दिखाइए कि $\deg D \leq n
  - 1$, और यह कि प्रत्येक $k = 0, \dots, n$ के लिए $D(y_k)$ का चिह्न कठोरतः
  $(-1)^k$ वाला है।
  \item उससे निकालिए कि $D$ के कम से कम $n$ भिन्न वास्तविक मूल हैं (हर
  अंतराल में एक, मध्यवर्ती मान गुणधर्म से, जो यहाँ उच्चतर माध्यमिक स्तर पर
  प्रयुक्त है और \cref{ch:b1:continuity} में सिद्ध), और \emph{चेबिशेव की
  प्रमेय} पर पहुँचिए: घात $n$ का प्रत्येक \omterm{def:b1:poly:def}{इकाई-अग्र} $P$ यह संतुष्ट करता है
        \[
        \norm{P}_\infty \geq 2^{1-n} .
        \]
  \item (समता-स्थिति, पहला पद) अब मान लीजिए ठीक $\norm P_\infty = 2^{1-n}$,
  जहाँ $P$ घात $n$ का \omterm{def:b1:poly:def}{इकाई-अग्र} है, और मान लीजिए $D = \widetilde T_n - P$।
  दिखाइए कि सभी $k$ के लिए $(-1)^kD(y_k) \geq 0$, और यह कि यदि किसी
  \emph{भीतरी} बिंदु $y_k$ ($0 < k < n$) पर $D(y_k) = 0$ हो, तो $D'(y_k) =
  0$ भी। \emph{(भीतरी $y_k$ पर $\widetilde T_n$ और $P$ दोनों निरपेक्ष मान
  $\norm{\cdot} _\infty$ का चरम प्राप्त करते हैं; भीतरी चरम पर अवकलनीय फलन
  का अवकलज शून्य होता है --- यह उच्चतर माध्यमिक स्तर पर प्रयुक्त है और
  \cref{ch:b1:derivative} में सिद्ध।)}
  \item (समता-स्थिति, निष्कर्ष) \omterm{def:b1:poly:derivative}{बहुलता} सहित $D$ के मूल गिनकर दिखाइए कि $D =
  0$: अर्थात् न्यूनकारी \emph{अद्वितीय} है, $P = \widetilde T_n$।
  \item किसी स्वेच्छ खंड $\intcc ab$ पर ले जाइए: दिखाइए कि $\intcc ab$ पर
  घात-$n$ के \omterm{def:b1:poly:def}{इकाई-अग्र बहुपद} का लघुतम उच्चतम-मानक
  $2\bigl(\frac{b-a}4\bigr)^n$ है, जो पुनर्मापित \omterm{pb:b1:poly:1}{चेबिशेव बहुपद} से प्राप्त
  होता है। \emph{($x = \frac{a+b}2 + \frac{b-a}2\,t$ प्रतिस्थापित कीजिए और
  अग्र गुणांक का हिसाब रखिए।)}
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- अनुप्रयोग।}
\begin{enumerate}[resume]
  \item स्थिति $n = 3$ हाथ से कीजिए: $\intcc{-1}1$ पर $\widetilde T_3 = X^3
  - \frac34X$ के चरम बिंदु ढूँढ़िए, मान $\frac14$ के साथ चार गुना समदोलन
  सत्यापित कीजिए, और निष्कर्ष निकालिए कि कोई \omterm{def:b1:poly:def}{इकाई-अग्र} त्रिघात इससे बेहतर
  नहीं करता।
  \item (इष्टतम अंतर्वेशन बिंदु) $n + 1$ बिंदुओं $x_0, \dots, x_n \in
  \intcc{-1}1$ के लिए अंतर्वेशन-त्रुटि $\omega(X) = \prod_i (X - x_i)$ से
  नियंत्रित होती है (जैसा \cref{ch:b1:taylor} परिमाणित करेगा)। सिद्ध कीजिए
  कि $\norm\omega_\infty$ को न्यूनतम करने वाला चयन $T_{n+1}$ के $n + 1$
  मूलों का \omterm{def:b1:logic:sets}{समुच्चय} है, जहाँ $\norm\omega_\infty = 2^{-n}$: चेबिशेव बिंदु ही
  अंतर्वेशन के सही स्थान हैं।
  \item $T_5$ का प्रयोग करके सिद्ध कीजिए कि $c = \cos 36^\circ$ $16c^5 -
  20c^3 + 5c + 1 = 0$ संतुष्ट करता है, इस \omterm{def:b1:poly:def}{बहुपद} का गुणनखंडन $(x + 1)(4x^2 -
  2x - 1)^2$ के रूप में कीजिए, और निष्कर्ष निकालिए
        \[
        \cos 36^\circ = \frac{1 + \sqrt5}4 .
        \]
        \cref{exo:b1:complex:8} के $\cos 72^\circ = \frac{\sqrt5 - 1}4$ के
        साथ संगति जाँचिए।
  \item प्रश्न 6 के संवृत रूप से $T_{10}(1.1)$ का आकलन कीजिए (दो सार्थक अंक
  पर्याप्त हैं), और व्याख्या कीजिए: जो \omterm{def:b1:poly:def}{बहुपद} $\intcc{-1}1$ पर $1$ से परिबद्ध
  है, वह $x = 1.1$ पर ही $40$ पार कर सकता है। (ऐसे बहुपदों में $T_n$ सबसे
  \emph{तेज़ी} से बढ़ता है --- यह इस कुल का एक और चरम गुणधर्म है, जो इस
  समस्या से परे है।)
  \item \omterm{def:b1:arith:congruence}{सर्वांगसमता} सिद्ध कीजिए: प्रत्येक विषम \omterm{def:b1:arith:prime}{अभाज्य} $p$ के लिए $T_p - X^p$
  के सभी गुणांक $p$ से विभाज्य हैं। \emph{(प्रश्न 7 और
  \cref{thm:b1:arith:fermat} की उपपत्ति से $0 < 2j < p$ के लिए $p \mid
  \binom p{2j}$ का प्रयोग कीजिए।)} $T_3$ और $T_5$ पर सत्यापित कीजिए।
\end{enumerate}

\medskip \noindent\textbf{भाग V --- संश्लेषण।}
\begin{enumerate}[resume]
  \item $\intcc01$ पर लघुतम उच्चतम-मानक वाला \omterm{def:b1:poly:def}{इकाई-अग्र} द्विघात और उसका विचलन
  स्पष्ट रूप से संगणित कीजिए। ($n = 2$ के साथ प्रश्न 17।)
  \item इस समस्या में ठीक कहाँ प्रयोग हुआ: (क) बहुपदों की दृढ़ता
  (\cref{cor:b1:poly:nroots}); (ख) \cref{ch:b1:complex} और
  \cref{ch:b1:functions} की त्रिकोणमिति; (ग) \cref{ch:b1:arith} के द्विपद
  गुणांकों का अंकगणित? प्रत्येक के लिए एक वाक्य।
  \item एक छोटे अनुच्छेद में संश्लेषण: प्रमेय कहती है कि सबसे चपटा \omterm{def:b1:poly:def}{इकाई-अग्र
  बहुपद} वही है जो \emph{समदोलन} करता है, और उपपत्ति इष्टतमता को मूल-गणना में
  बदल देती है। इस क्रियाविधि पर, बीजगणित और त्रिकोणमिति के बीच सेतु के रूप
  में प्रतिस्थापन $x = \cos\theta$ की भूमिका पर टिप्पणी कीजिए, और वे दो
  स्थान बताइए जहाँ समस्या को विश्लेषण के ऐसे तथ्य (मध्यवर्ती मान प्रमेय,
  भीतरी चरम) चाहिए थे जिन्हें आगे के अध्याय सिद्ध करते हैं।
\end{enumerate}
\end{problem}
