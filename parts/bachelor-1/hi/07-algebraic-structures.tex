\chapter{बीजीय संरचनाएँ}\label{ch:b1:structures}

संगणना के वही नियम बार-बार लौटते रहते हैं: पूर्णांक, वास्तविक संख्याएँ,
सम्मिश्र संख्याएँ, \omterm{def:b1:arith:congruence}{सर्वांगसमता} वर्ग, और शीघ्र ही बहुपद (\cref{ch:b1:poly})
तथा सदिश और आव्यूह (\cref{ch:b1:vspaces,ch:b1:matrices})। बीजगणित इन साझे
प्रतिरूपों को निकालकर उन्हें नाम दे देता है: \emph{\omterm{def:b1:structures:group}{समूह}}, \emph{\omterm{def:b1:structures:ring}{वलय}},
\emph{\omterm{def:b1:structures:field}{क्षेत्र}}। किसी तथ्य को एक बार, संरचना के स्तर पर, सिद्ध कर देना उसे एक
साथ हर उदाहरण के लिए सिद्ध कर देता है।

\section{संयोजन-नियम}

\begin{definition}\label{def:b1:structures:law}
\omterm{def:b1:logic:sets}{समुच्चय} $E$ पर \emph{संयोजन-नियम}\index{संयोजन-नियम} एक \omterm{def:b1:logic:map}{प्रतिचित्रण} $E
\times E \to E$ है, जिसे $(x, y) \mapsto x * y$ लिखा जाता है। वह
\emph{साहचर्य} तब है जब सदा $(x*y)*z = x*(y*z)$ हो, और \emph{क्रमविनिमेय} तब
जब सदा $x * y = y * x$ हो। अवयव $e$ \emph{तत्समक} तब है जब सभी $x$ के लिए $e
* x = x * e = x$; और तब $x'$ $x$ का \emph{प्रतिलोम} तब है जब $x * x' = x' *
x = e$।
\end{definition}

\begin{proposition}[अद्वितीयता]\label{prop:b1:structures:unique}
किसी नियम का अधिक से अधिक एक तत्समक होता है; और तत्समक वाले साहचर्य नियम के
लिए प्रत्येक अवयव का अधिक से अधिक एक प्रतिलोम होता है।
\end{proposition}

\begin{proof}
यदि $e$ और $e'$ तत्समक हैं: $e = e * e' = e'$। यदि $x'$ और $x''$ $x$ का
प्रतिलोमन करते हैं: $x' = x' * e = x' * (x * x'') = (x' * x) * x'' = e * x''
= x''$।
\end{proof}

\section{समूह}

\begin{definition}[समूह]\label{def:b1:structures:group}
\emph{समूह}\index{समूह} $(G, *)$ ऐसा \omterm{def:b1:logic:sets}{समुच्चय} है जिस पर साहचर्य नियम हो,
जिसका कोई तत्समक हो और जिसमें प्रत्येक अवयव का प्रतिलोम हो। नियम के
क्रमविनिमेय होने पर समूह \emph{आबेली}\index{आबेली समूह} कहलाता है।
\end{definition}

\begin{example}\label{ex:b1:structures:groups}
$(\Z, +)$, $(\Q, +)$, $(\R, +)$, $(\C, +)$; $(\Q^*, \times)$, $(\R^*,
\times)$, $(\C^*, \times)$, $(\mathbb{U}_n, \times)$ (\omterm{def:b1:complex:unity}{इकाई के मूल},
\cref{def:b1:complex:unity}); \omterm{def:b1:logic:sets}{समुच्चय} $E$ के स्वयं उसी पर एकैकी आच्छादनों का
\omterm{def:b1:logic:sets}{समुच्चय} $\mathfrak{S}(E)$, संयोजन के अंतर्गत --- अर्थात् $E$ का \emph{सममित
समूह}\index{सममित समूह}, जो $\abs E \geq 3$ होते ही अनाबेली हो जाता है। \omterm{def:b1:structures:group}{समूह}
नहीं: $(\N, +)$ (प्रतिलोम नहीं), $(\Z, \times)$ (केवल $\pm 1$
व्युत्क्रमणीय)।
\end{example}

\begin{proposition}[संगणना के नियम]\label{prop:b1:structures:rules}
\omterm{def:b1:structures:group}{समूह} $G$ में (गुणन-संकेतन में, तत्समक $e$):
\begin{enumerate}
  \item काटना: $ax = ay \implies x = y$ और $xa = ya \implies x = y$;
  \item $(ab)^{-1} = b^{-1} a^{-1}$ और $(a^{-1})^{-1} = a$;
  \item $a, b \in G$ के लिए प्रत्येक समीकरण $ax = b$ तथा $xa = b$ का
  अद्वितीय हल है ($x = a^{-1}b$, क्रमशः $x = b a^{-1}$)।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) उपयुक्त ओर से $a^{-1}$ से गुणा कीजिए और साहचर्यता का प्रयोग कीजिए। (2)
$(b^{-1}a^{-1})(ab) = b^{-1}(a^{-1}a)b = b^{-1}b = e$, और सममित रूप से भी;
प्रतिलोम की अद्वितीयता निष्कर्ष दे देती है; दूसरा बिंदु $a^{-1}$ पर
\cref{prop:b1:structures:unique} लगाना ही है। (3) प्रतिस्थापित कीजिए, और
अद्वितीयता के लिए (1) का प्रयोग कीजिए।
\end{proof}

\begin{example}[आयत की सममितियाँ]\label{ex:b1:structures:klein}
(वर्ग न होने वाला) आयत स्वयं अपने पर ठीक चार सममितियाँ स्वीकार करता है:
तत्समक $e$, क्षैतिज अक्ष में परावर्तन $h$, ऊर्ध्वाधर अक्ष में परावर्तन $v$,
और केंद्र के परितः अर्ध-घुमाव $r$। संयोजन इस चार-अवयवी \omterm{def:b1:logic:sets}{समुच्चय} को \omterm{def:b1:structures:group}{समूह} बना
देता है: हर अवयव अपना ही प्रतिलोम है ($h^2 = v^2 = r^2 = e$), और किन्हीं दो
भिन्न अतत्समक अवयवों का गुणनफल तीसरा होता है ($hv = vh = r$: दोनों अक्षों
में परावर्तन ही अर्ध-घुमाव है)। पूरी सारणी सममित है, अतः \omterm{def:b1:structures:group}{समूह} \omterm{def:b1:structures:group}{आबेली} है ---
फिर भी वह \cref{ex:b1:structures:order} के घूर्णनों $\mathbb U_4$ वाला \omterm{def:b1:structures:group}{समूह}
\emph{नहीं} है: वहाँ $\iu$ की \omterm{def:b1:structures:order}{कोटि} $4$ है, जबकि यहाँ हर \omterm{def:b1:structures:order}{अवयव की कोटि} $\leq
2$ है। अतः एक ही आकार के दो समूहों की गुणन-संरचनाएँ सचमुच भिन्न हो सकती हैं
--- नीचे का चित्र दोनों सारणियाँ साथ-साथ दिखाता है। यह चार-अवयवी \omterm{def:b1:structures:group}{समूह}
$\{\pm1\} \times \{\pm1\}$ के रूप में लौटता है, और
\cref{exo:b1:structures:7} समझाता है कि जिस \omterm{def:b1:structures:group}{समूह} में सभी वर्ग तुच्छ हों वह,
इसी की भाँति, \omterm{def:b1:structures:group}{आबेली} क्यों होना चाहिए।
\end{example}

\begin{definition}[उपसमूह]\label{def:b1:structures:subgroup}
\omterm{def:b1:structures:group}{समूह} $G$ का उपसमुच्चय $H$ \emph{उपसमूह}\index{उपसमूह} (लिखा जाता है $H \leq
G$) तब कहलाता है जब उसमें $e$ हो और वह नियम तथा प्रतिलोमन के अंतर्गत स्थायी
हो। तब $H$ स्वयं एक \omterm{def:b1:structures:group}{समूह} है।

\emph{कसौटी:} अरिक्त $H \subseteq G$ उपसमूह है यदि और केवल यदि
\[
\forall x, y \in H, \quad x y^{-1} \in H .
\]
\end{definition}

\begin{proof}[कसौटी की उपपत्ति]
\omterm{def:b1:structures:subgroup}{उपसमूह} स्पष्टतः इसे संतुष्ट करता है। विलोमतः, मान लीजिए $H \neq \emptyset$
इसे संतुष्ट करता है, और $x_0 \in H$ चुनिए। तब $e = x_0 x_0^{-1} \in H$; $y
\in H$ के लिए $y^{-1} = e\,y^{-1} \in H$; और $x, y \in H$ के लिए $xy = x
(y^{-1})^{-1} \in H$।
\end{proof}

\begin{example}\label{ex:b1:structures:subgroups}
$\mathbb{U}_n \leq (\C^*, \times)$: अरिक्त, और $z, w \in \mathbb{U}_n$ के
लिए $(zw^{-1})^n = z^n (w^n)^{-1} = 1$। $(\Z, +)$ के \omterm{def:b1:structures:subgroup}{उपसमूह} ठीक $n\Z$ हैं
(उपपत्ति \cref{thm:b1:arith:gcd} में)। उपसमूहों का प्रतिच्छेदन सदा \omterm{def:b1:structures:subgroup}{उपसमूह}
होता है, पर सम्मिलन लगभग कभी नहीं (\cref{exo:b1:structures:6})।
\end{example}

\begin{remark}[संरचनाओं की सामान्य भूलें]\label{rem:b1:structures:pitfalls}
\begin{enumerate}
  \item \emph{नियम के अंतर्गत स्थायित्व पर्याप्त नहीं है।} $\N$ $\Z$ के भीतर
  योग के अंतर्गत स्थायी है और उसमें $0$ भी है, फिर भी वह \omterm{def:b1:structures:subgroup}{उपसमूह} नहीं है:
  प्रतिलोम अनुपस्थित हैं। कसौटी $xy^{-1} \in H$ सब कुछ एक साथ जाँच लेती है
  --- पर तभी, जब $H \neq \emptyset$ जाँच लिया गया हो।
  \item \emph{अनाबेली प्रतिवर्त।} किसी व्यापक \omterm{def:b1:structures:group}{समूह} में $(ab)^2 = abab$, जो
  $a$ और $b$ के क्रम-विनिमेय होने पर ही $a^2b^2$ है; इसी प्रकार $(ab)^{-1} =
  b^{-1}a^{-1}$, जिसमें क्रम उलट जाता है। विद्यालयी बीजगणित से आयातित हर
  सर्वसमिका को अभिगृहीतों से फिर व्युत्पन्न करना होगा, या उस पर
  क्रमविनिमेयता की शर्त लगानी होगी।
  \item \emph{\omterm{def:b1:structures:morphism}{अष्टि} बनाम प्रतिबिंब।} $\ker f$ \emph{स्रोत} में रहती है,
  $\operatorname{im} f$ लक्ष्य में; ``$f$ \omterm{def:b1:logic:inj}{एकैकी} है यदि और केवल यदि $\ker f$
  तुच्छ है'' (\cref{prop:b1:structures:kernel}) का प्रतिबिंब के साथ कोई
  अनुरूप नहीं है ($\operatorname{im} f = G'$ तो आच्छादकता है)।
  \item \emph{\omterm{def:b1:structures:ring}{वलय} $\times$ के लिए \omterm{def:b1:structures:group}{समूह} नहीं हैं।} किसी \omterm{def:b1:structures:ring}{वलय} में अधिकांश
  अवयवों का व्युत्क्रमणीय होना आवश्यक नहीं है, और $a$ से काटने के लिए $a$ का
  इकाई होना या \omterm{def:b1:structures:ring}{वलय} का \omterm{def:b1:structures:field}{पूर्णांकीय प्रांत} होना आवश्यक है: $\Z/12\Z$ में
  $\overline3\,\overline2 = \overline3\,\overline6$, फिर भी $\overline2 \neq
  \overline6$ (\cref{ex:b1:structures:zncomputation})।
\end{enumerate}
\end{remark}

\begin{definition}[समूह समाकारिता]\label{def:b1:structures:morphism}
मान लीजिए $(G, *)$ और $(G', \star)$ \omterm{def:b1:structures:group}{समूह} हैं। \omterm{def:b1:logic:map}{प्रतिचित्रण} $f \colon G \to
G'$ \emph{समाकारिता}\index{समाकारिता!समूहों की} तब कहलाता है जब
\[
\forall x, y \in G, \qquad f(x * y) = f(x) \star f(y).
\]
तब $f(e_G) = e_{G'}$ और $f(x^{-1}) = f(x)^{-1}$। $f$ की
\emph{अष्टि}\index{अष्टि} और \emph{प्रतिबिंब} हैं
\[
\ker f = f^{-1}(\{e_{G'}\}) \leq G,
\qquad
\operatorname{im} f = f(G) \leq G' .
\]
\omterm{def:b1:logic:inj}{एकैकी आच्छादक} समाकारिता \emph{तुल्याकारिता} कहलाती है; तब उसका प्रतिलोम
\omterm{def:b1:logic:map}{प्रतिचित्रण} स्वतः समाकारिता होता है।
\end{definition}

\begin{proof}[दावों की उपपत्ति]
$f(e) = f(e * e) = f(e)\star f(e)$, और $f(e)$ को काटने पर $e_{G'} = f(e)$
मिलता है। फिर $f(x)\star f(x^{-1}) = f(x x^{-1}) = e_{G'}$ $f(x^{-1})$ को
प्रतिलोम के रूप में पहचान लेता है। \omterm{def:b1:structures:morphism}{अष्टि}: $e \in \ker f$; यदि $x, y \in \ker
f$, तो $f(xy^{-1}) = f(x)f(y)^{-1} = e$; कसौटी लागू हो जाती है। प्रतिबिंब:
$f(x)f(y)^{-1} = f(xy^{-1})$ के साथ वही कसौटी। तुल्याकारिता का प्रतिलोम: $u,
v \in G'$ के लिए $u = f(x)$, $v = f(y)$ लिखिए; तब $f^{-1}(u \star v) =
f^{-1}(f(xy)) = xy = f^{-1}(u) f^{-1}(v)$।
\end{proof}

\begin{proposition}[अष्टि के द्वारा एकैकीयता]\label{prop:b1:structures:kernel}
\omterm{def:b1:structures:group}{समूह} \omterm{def:b1:structures:morphism}{समाकारिता} $f$ \omterm{def:b1:logic:inj}{एकैकी} है यदि और केवल यदि $\ker f = \{e\}$।
\end{proposition}

\begin{proof}
यदि $f$ \omterm{def:b1:logic:inj}{एकैकी} है, तो $\ker f$ में $e_{G'}$ का केवल वही एक \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} हो
सकता है, अर्थात् $e$। विलोमतः, यदि $\ker f = \{e\}$ और $f(x) = f(y)$, तो
$f(xy^{-1}) = f(x) f(y)^{-1} = e_{G'}$, अतः $xy^{-1} = e$, अर्थात् $x = y$।
\end{proof}

\begin{example}\label{ex:b1:structures:expmorphism}
$\exp \colon (\R, +) \to (\R_+^*, \times)$ एक \omterm{def:b1:structures:morphism}{समाकारिता} है ($\eu^{x+y} =
\eu^x \eu^y$), और \omterm{def:b1:logic:inj}{एकैकी आच्छादक} भी (\cref{prop:b1:functions:expln}): अर्थात्
योगात्मक और गुणात्मक संरचनाएँ तुल्याकारी हैं --- यही लघुगणकों के अस्तित्व का
ऐतिहासिक कारण है। एक और \omterm{def:b1:structures:morphism}{समाकारिता}: $(\R, +)$ से इकाई वृत्त $(\mathbb{U},
\times)$ पर $\theta \mapsto \eu^{\iu\theta}$, जिसकी \omterm{def:b1:structures:morphism}{अष्टि} $2\pi\Z$ है।
\end{example}

\begin{example}[चिह्न समाकारिता]\label{ex:b1:structures:signmorphism}
$x$ को उसके चिह्न पर भेजने वाला \omterm{def:b1:logic:map}{प्रतिचित्रण} $s \colon (\R^*, \times) \to
(\{\pm1\}, \times)$ एक \omterm{def:b1:structures:morphism}{समाकारिता} है: गुणनफल का चिह्न चिह्नों का गुणनफल होता
है। उसकी \omterm{def:b1:structures:morphism}{अष्टि} $\intoo0{+\infty}$ है (जो \omterm{def:b1:structures:subgroup}{उपसमूह} है, जैसा
\cref{def:b1:structures:morphism} वादा करता है), और उसका प्रतिबिंब पूरा
$\{\pm1\}$: \omterm{def:b1:logic:inj}{आच्छादक}, पर अत्यधिक अनेकैकी। लघु रूप में दो व्यापक शिक्षाएँ।
पहली, कोई \omterm{def:b1:structures:morphism}{समाकारिता} सूचना को कुचल सकती है: $s$ $x$ का केवल एक बिट स्मरण रखता
है और कुछ नहीं, और यही उसका गुण है --- चिह्न वाले तर्क ठीक वही संगणनाएँ हैं
जो $s$ से होकर गुज़रती हैं। दूसरी, $\{\pm1\}$ तक जाने वाली समाकारिताएँ सरलतम
``अपरिवर्त्य'' हैं: इस अध्याय की सप्ताहांत समस्या में बनने वाला क्रमचयों का
संकेतक \omterm{def:b1:structures:group}{समूह} $\mathfrak S_n$ पर वही परिघटना है, और वह जिन सम-विषम तर्कों को
शक्ति देता है वे सब ऐसी ही द्विमानी \omterm{def:b1:structures:morphism}{समाकारिता} से उतरते हैं।
\end{example}

\begin{definition}[घातें, अवयव की कोटि]\label{def:b1:structures:order}
\omterm{def:b1:structures:group}{समूह} $G$ में (गुणन संकेतन) $x^0 = e$, $x^{k+1} = x^k x$ और $k \in \N$ के लिए
$x^{-k} = (x^k)^{-1}$ रखिए; तब सभी $k, l \in \Z$ के लिए $x^{k+l} = x^k x^l$,
अतः $k \mapsto x^k$ एक \omterm{def:b1:structures:morphism}{समाकारिता} $(\Z, +) \to G$ है जिसका प्रतिबिंब $\langle
x \rangle = \{x^k : k \in \Z\}$ \omterm{def:b1:structures:subgroup}{उपसमूह} है, अर्थात् $x$ से \emph{जनित}
\omterm{def:b1:structures:subgroup}{उपसमूह}। $x$ की \emph{कोटि}\index{अवयव की कोटि} वह लघुतम $m \geq 1$ है जिसके
लिए $x^m = e$ हो, यदि ऐसा कोई हो (तब $\langle x\rangle = \{e, x, \dots,
x^{m-1}\}$ में ठीक $m$ अवयव होते हैं, और $x^k = e \iff m \mid k$), और अन्यथा
$\infty$।
\end{definition}

\begin{example}\label{ex:b1:structures:order}
$(\C^*, \times)$ में: $\iu$ की \omterm{def:b1:structures:order}{कोटि} $4$ है, जहाँ $\langle \iu \rangle = \{1,
\iu, -1, -\iu\} = \mathbb{U}_4$; और अधिक व्यापक रूप से $\omega =
\eu^{2\iu\pi/n}$ की \omterm{def:b1:structures:order}{कोटि} $n$ है और $\langle\omega\rangle = \mathbb{U}_n$।
$(\Z, +)$ में प्रत्येक $x \neq 0$ की \omterm{def:b1:structures:order}{कोटि} अनंत है। परिभाषा के दावे क्यों
सत्य हैं: यदि $x$ की \omterm{def:b1:structures:order}{कोटि} $m$ है, तो किसी भी $k$ को $m$ से भाग दीजिए ($k =
mq + r$, $0 \leq r < m$, \cref{thm:b1:arith:division}): $x^k = (x^m)^q x^r =
x^r$, अतः घातें आवर्तकाल $m$ के साथ चक्कर लगाती हैं, सूचीबद्ध अवयव $m$ की
लघुतमता से जोड़ों में भिन्न हैं, और $x^k = e$ $r = 0$ पर बाध्य कर देता है।
क्रमचयों की कोटियाँ नीचे दी गई सप्ताहांत समस्या में संगणित की गई हैं।
\end{example}

\begin{example}[$\mathbb U_{12}$ के भीतर कोटियाँ]\label{ex:b1:structures:orderU12}
$\omega = \eu^{2\iu\pi/n}$ के लिए $\mathbb U_n$ में $\omega^k$ की \omterm{def:b1:structures:order}{कोटि} क्या
है? $(\omega^k)^m = 1$ तभी जब $n \mid km$, और $\gcd(n', k') = 1$ के साथ $d =
\gcd(n, k)$, $n = dn'$, $k = dk'$ लिखने पर $n \mid km \iff n' \mid k'm \iff
n' \mid m$ (गाउस की प्रमेयिका, \cref{thm:b1:arith:gauss})। ऐसा लघुतम $m \geq
1$ $n' = \frac{n}{\gcd(n,k)}$ है। उदाहरण के लिए $\mathbb U_{12}$ में
$\omega^8$ की \omterm{def:b1:structures:order}{कोटि} $\frac{12}{\gcd(12,8)} = 3$ है (वस्तुतः $\omega^8 =
\eu^{4\iu\pi/3} \in \mathbb U_3$), जबकि $\omega^5$ की \omterm{def:b1:structures:order}{कोटि} $12$ है: वह पूरे
\omterm{def:b1:structures:group}{समूह} को जनता है, यद्यपि वह ``मानक'' जनक नहीं है। जनकों को गिनना --- अर्थात्
$\gcd(k, n) = 1$ वाले $k$ गिनना --- \cref{ex:b1:counting:coprime} की
\omterm{cor:b1:arith:bezout}{सहअभाज्य} गणनाओं को फिर से दे देता है: \omterm{def:b1:structures:group}{समूह} सिद्धांत और गणना यहाँ मिल जाते
हैं।
\end{example}

\section{वलय और क्षेत्र}

\begin{definition}[वलय]\label{def:b1:structures:ring}
\emph{वलय}\index{वलय} $(A, +, \times)$ ऐसा \omterm{def:b1:logic:sets}{समुच्चय} है जिस पर दो नियम इस
प्रकार हों: $(A, +)$ \omterm{def:b1:structures:group}{आबेली समूह} है (तत्समक $0$); $\times$ साहचर्य है और उसका
तत्समक $1$ है; और $\times$ $+$ पर दोनों ओर से वितरित होता है। $\times$ के
क्रमविनिमेय होने पर वलय \emph{क्रमविनिमेय} कहलाता है। अवयव $a$
\emph{व्युत्क्रमणीय} (एक \emph{इकाई}) तब है जब किसी $b$ के लिए $ab = ba = 1$
हो; इकाइयाँ एक \omterm{def:b1:structures:group}{समूह} $(A^\times, \times)$ बनाती हैं।
\end{definition}

\begin{proof}[इकाइयाँ समूह बनाती हैं, इसकी उपपत्ति]
स्थायित्व: यदि $a, a'$ प्रतिलोमों $b, b'$ वाली इकाइयाँ हैं, तो
\[
(aa')(b'b) = a(a'b')b = a\,1\,b = ab = 1,
\qquad (b'b)(aa') = 1
\]
और सममित रूप से भी, अतः $aa'$ इकाई है। अवयव $1$ इकाई है (अपना ही प्रतिलोम),
साहचर्यता $A$ से विरासत में मिलती है, और इकाई $a$ का प्रतिलोम $b$ स्वयं इकाई
है (जिसका प्रतिलोम $a$ है)। अतः $(A^\times, \times)$ \omterm{def:b1:structures:group}{समूह} के सभी अभिगृहीत
संतुष्ट करता है। इस पुस्तक का हर वह \omterm{def:b1:structures:group}{समूह} जो क्रमचयों से नहीं बना, इसी प्रकार
उत्पन्न होता है: $\Q^* = \Q^\times$, $\R^*$, $\C^*$, नीचे $\Z/n\Z$ की
इकाइयाँ, और आगे व्युत्क्रमणीय आव्यूह (\cref{ch:b1:matrices})।
\end{proof}

\begin{example}\label{ex:b1:structures:rings}
$\Z, \Q, \R, \C$ क्रमविनिमेय \omterm{def:b1:structures:ring}{वलय} हैं; $\Z^\times = \{1, -1\}$, $\Q^\times =
\Q^*$। आगे: बहुपद \omterm{def:b1:structures:ring}{वलय} $K[X]$ (\cref{ch:b1:poly}), आव्यूह \omterm{def:b1:structures:ring}{वलय} (अक्रमविनिमेय,
\cref{ch:b1:matrices}), और नीचे $\Z/n\Z$। हर \omterm{def:b1:structures:ring}{वलय} में $0 \times a = 0$
(वितरणशीलता से: $0a = (0+0)a = 0a + 0a$), और $(-1)a = -a$।
\end{example}

\begin{example}[वर्गसम अवयव: नए वलयों में नई परिघटनाएँ]\label{ex:b1:structures:idempotents}
$\Z$ में समीकरण $x^2 = x$, अर्थात् $x(x - 1) = 0$, के केवल हल $0$ और $1$
हैं। $\Z/6\Z$ में सभी वर्ग जाँचिए: $\overline0^2 = \overline0$,
$\overline1^2 = \overline1$, $\overline3^2 = \overline9 = \overline3$ और
$\overline4^2 = \overline{16} = \overline4$ --- अर्थात् \emph{चार} वर्गसम
अवयव। दोनों विलक्षण अवयव शून्य भाजकों से आते हैं: $\overline3\,(\overline3 -
\overline1) = \overline3 \times \overline2 = \overline6 = \overline0$, जहाँ
कोई भी गुणनखंड शून्य नहीं है। ऐसी संगणनाएँ हमारी सहज बुद्धि को अंशांकित करती
हैं: समीकरणों के परिचित तथ्य पूर्णांकीय प्रांतों और क्षेत्रों में बचे रहते
हैं, पर कोई व्यापक \omterm{def:b1:structures:ring}{वलय} भिन्न व्यवहार कर सकता है और करता भी है ---
\cref{exo:b1:structures:10} के बूली \omterm{def:b1:structures:ring}{वलय} भी देखिए, जहाँ \emph{प्रत्येक} अवयव
वर्गसम है।
\end{example}

\begin{proposition}[क्रमविनिमेय वलय में द्विपद प्रमेय]\label{prop:b1:structures:binomial}
यदि $a, b$ किसी क्रमविनिमेय \omterm{def:b1:structures:ring}{वलय} के अवयव हैं (और अधिक व्यापक रूप से, यदि $ab
= ba$), तो $n \in \N$ के लिए:
\[
(a+b)^n = \sum_{k=0}^n \binom nk a^k b^{n-k},
\qquad
a^n - b^n = (a - b) \sum_{k=0}^{n-1} a^k b^{\,n-1-k} .
\]
\end{proposition}

\begin{proof}
\cref{thm:b1:counting:binomial} और गुणोत्तर सर्वसमिका की उपपत्तियाँ केवल
साहचर्यता, दोनों अवयवों की क्रमविनिमेयता और वितरणशीलता का प्रयोग करती हैं
--- वे अक्षरशः लागू हो जाती हैं।
\end{proof}

\begin{example}[अपरिचित वलय में द्विपद प्रमेय]\label{ex:b1:structures:binomialring}
इस व्यापकता के दो तात्कालिक लाभ। $\Z/p\Z$ ($p$ \omterm{def:b1:arith:prime}{अभाज्य}) में बीच के \omterm{def:b1:counting:objects}{द्विपद
गुणांक} लुप्त हो जाते हैं (\cref{thm:b1:arith:fermat} का पहला पद), अतः प्रमेय
सिकुड़कर \emph{नवागंतुक का स्वप्न} बन जाती है
\[
(a + b)^p = a^p + b^p \qquad \text{में } \Z/p\Z ,
\]
जो वहाँ सच्ची सर्वसमिका है, चाहे $\R$ पर वह कितनी ही अपराधी क्यों न दिखे। और
$\varepsilon^2 = 0$ वाला अवयव $\varepsilon$ रखने वाले किसी भी क्रमविनिमेय
\omterm{def:b1:structures:ring}{वलय} में प्रमेय कट जाती है: $(a + \varepsilon)^n = a^n +
n\,a^{n-1}\varepsilon$, जहाँ ऊपर के सभी पद $\varepsilon^2 = 0$ का गुणनखंड
लिए हुए हैं। $\varepsilon$ का गुणांक $n\,a^{n-1}$ $x^n$ का अवकलज है --- यह
संयोग नहीं है, और यह पहला संकेत है कि अवकलज जितने विश्लेषण के हैं उतने ही
बीजगणित के भी (तुलना कीजिए \cref{ch:b1:poly} के औपचारिक अवकलज से)।
\end{example}

\begin{definition}[पूर्णांकीय प्रांत, क्षेत्र]\label{def:b1:structures:field}
क्रमविनिमेय \omterm{def:b1:structures:ring}{वलय} $A \neq \{0\}$ \emph{पूर्णांकीय प्रांत}\index{पूर्णांकीय
प्रांत} तब है जब उसमें कोई शून्य भाजक न हो: $ab = 0 \implies a = 0$ या $b =
0$। वह \emph{क्षेत्र}\index{क्षेत्र} तब है जब उसका प्रत्येक अशून्य अवयव
व्युत्क्रमणीय हो। प्रत्येक क्षेत्र पूर्णांकीय प्रांत है ($ab = 0$ और $a \neq
0$ से $b = a^{-1}ab = 0$ मिलता है)।
\end{definition}

\begin{example}
$\Q$, $\R$, $\C$ \omterm{def:b1:structures:field}{क्षेत्र} हैं; $\Z$ \omterm{def:b1:structures:field}{पूर्णांकीय प्रांत} है पर \omterm{def:b1:structures:field}{क्षेत्र} नहीं।
\omterm{def:b1:structures:field}{पूर्णांकीय प्रांत} में $\times$ के लिए काटना वैध है: $ab = ac$ और $a \neq 0$
से $b = c$ निकलता है।
\end{example}

\section{वलय $\Z/n\Z$}

\begin{definition}\label{def:b1:structures:zn}
$n \in \N^*$ नियत कीजिए। $n$ के सापेक्ष \omterm{def:b1:arith:congruence}{सर्वांगसमता} वर्ग
(\cref{ex:b1:logic:congruence}) $n$ अवयवों का \omterm{def:b1:logic:sets}{समुच्चय}
$\Z/n\Z$\index{Z/nZ@$\Z/n\Z$} बनाते हैं, जिसे $\overline 0, \overline 1,
\dots, \overline{n-1}$ लिखा जाता है। संक्रियाएँ
\[
\overline a + \overline b = \overline{a + b},
\qquad
\overline a \times \overline b = \overline{ab}
\]
सुपरिभाषित हैं --- परिणामों के वर्ग प्रतिनिधियों पर निर्भर नहीं करते, ठीक
इसलिए कि \omterm{def:b1:arith:congruence}{सर्वांगसमता} $+$ और $\times$ के अनुकूल है
(\cref{def:b1:arith:congruence}) --- और वे $\Z/n\Z$ को क्रमविनिमेय \omterm{def:b1:structures:ring}{वलय} बना
देती हैं।
\end{definition}

\begin{theorem}[$\Z/n\Z$ की इकाइयाँ; क्षेत्र $\Z/p\Z$]\label{thm:b1:structures:zp}
\begin{enumerate}
  \item $\overline a$ $\Z/n\Z$ में व्युत्क्रमणीय है यदि और केवल यदि $\gcd(a,
  n) = 1$।
  \item $\Z/n\Z$ \omterm{def:b1:structures:field}{क्षेत्र} है यदि और केवल यदि $n$ \omterm{def:b1:arith:prime}{अभाज्य} हो।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) वर्गों के साथ फिर से लिखा हुआ \cref{prop:b1:arith:invmod} ही है।

(2) यदि $n = p$ \omterm{def:b1:arith:prime}{अभाज्य} है, तो प्रत्येक $\overline a \neq \overline 0$ के लिए
$p \nmid a$, अतः $\gcd(a, p) = 1$: (1) से व्युत्क्रमणीय --- अर्थात् \omterm{def:b1:structures:field}{क्षेत्र}।
यदि $1 < a, b < n$ के साथ $n = ab$, तो $\overline a, \overline b \neq
\overline 0$ के साथ $\overline a\, \overline b = \overline n = \overline 0$:
अर्थात् शून्य भाजक, अतः वह \omterm{def:b1:structures:field}{पूर्णांकीय प्रांत} तक नहीं है; और $n = 1$ शून्य
\omterm{def:b1:structures:ring}{वलय} देता है, जो वर्जित है।
\end{proof}

\begin{example}[$1$ के कितने वर्गमूल?]\label{ex:b1:structures:sqrtone}
$\Z/8\Z$ में और $\Z/7\Z$ में $x^2 = \overline 1$ हल कीजिए। $8$ के सापेक्ष
आठों वर्ग जाँचिए: $1^2 = 1$, $3^2 = 9 \equiv 1$, $5^2 = 25 \equiv 1$, $7^2 =
49 \equiv 1$ --- अर्थात् \emph{चार} हल $\{\overline1, \overline3,
\overline5, \overline7\}$, यद्यपि बहुपद $X^2 - 1$ की घात $2$ है। इसके विपरीत
\omterm{def:b1:structures:field}{क्षेत्र} $\Z/7\Z$ में $x^2 = \overline1$ का अर्थ है $(x - \overline1)(x +
\overline1) = \overline0$, और \omterm{def:b1:structures:field}{क्षेत्र} में कोई शून्य भाजक नहीं होता: $x =
\pm\overline1$, अर्थात् केवल दो हल। $8$ के सापेक्ष विफलता का कारण पहचाना जा
सकता है: $(3-1)(3+1) = 2 \times 4 = 8 \equiv 0$, जहाँ कोई भी गुणनखंड लुप्त
नहीं होता। उपदेश: परिचित नियम ``घात $d$ के समीकरण के अधिक से अधिक $d$ मूल
होते हैं'' \emph{पूर्णांकीय प्रांतों} के विषय में प्रमेय है
(\cref{cor:b1:poly:nroots} उसे क्षेत्रों पर सिद्ध करता है); शून्य भाजकों
वाले वलयों में वह चुपचाप विफल हो जाता है --- और ठीक इसीलिए विल्सन की प्रमेय
की युग्मन उपपत्ति (\cref{exo:b1:arith:11}) को $p$ का \omterm{def:b1:arith:prime}{अभाज्य} होना चाहिए था।
\end{example}

\begin{example}[$\Z/n\Z$ में संगणना]\label{ex:b1:structures:zncomputation}
$\Z/12\Z$ में: इकाइयाँ $\overline 1, \overline 5, \overline 7,
\overline{11}$ हैं ($12$ से \omterm{cor:b1:arith:bezout}{सहअभाज्य} वर्ग), और हर एक अपना ही प्रतिलोम है
($5^2 = 25 \equiv 1$, $7^2 = 49 \equiv 1$, $11^2 = 121 \equiv 1$)। समीकरण
$\overline 3\, x = \overline 6$ के \emph{तीन} हल हैं ($x \in \{\overline 2,
\overline 6, \overline{10}\}$): व्युत्क्रमणीयता के बिना कोई काटना नहीं। इसके
विपरीत $\Z/11\Z$ में $\overline a \neq \overline 0$ वाले हर समीकरण
$\overline a x = \overline b$ का ठीक एक हल है।
\end{example}

\begin{example}[समूह अभिगृहीत हल करने की अनुमति के रूप में]\label{ex:b1:structures:solving}
\omterm{def:b1:structures:group}{समूह} $\bigl((\Z/7\Z)^*, \times\bigr)$ में $\overline 3\,x = \overline 5$ हल
कीजिए। \cref{prop:b1:structures:rules}~(3) से हल विद्यमान और अद्वितीय है, और
वह $\overline3^{-1}\, \overline5$ है; चूँकि $\overline3 \times \overline5 =
\overline{15} = \overline1$, $\overline 3$ का प्रतिलोम $\overline 5$ है, अतः
\[
x = \overline5 \times \overline5 = \overline{25} = \overline4,
\qquad\text{जाँच: } \overline3 \times \overline4 =
\overline{12} = \overline5 .
\]
यहाँ उत्तर से अधिक महत्त्व गारंटी का है: किसी \omterm{def:b1:structures:group}{समूह} में ऐसा हर समीकरण किसी भी
संगणना से \emph{पहले} ही अद्वितीय रूप से हल हो सकने वाला होता है, अतः हल
करने की कोई प्रक्रिया कभी ``कोई हल नहीं'' या ``अनेक हल'' पर नहीं टकरा सकती।
ऊपर $\Z/12\Z$ में $\overline3\,x = \overline6$ से तुलना कीजिए, जहाँ यह
गारंटी विफल है --- यह जानना कि हम किस संरचना में हैं, यह जानना है कि हम किस
बात को दिया हुआ मान सकते हैं।
\end{example}

\begin{example}[प्रत्यक्ष गुणन]\label{ex:b1:structures:product}
यदि $G$ और $H$ \omterm{def:b1:structures:group}{समूह} हैं, तो गुणन \omterm{def:b1:logic:sets}{समुच्चय} $G \times H$ घटक-दर-घटक नियम $(g,
h)(g', h') = (gg', hh')$ के साथ एक \omterm{def:b1:structures:group}{समूह} है: अभिगृहीत
निर्देशांक-दर-निर्देशांक जाँचे जाते हैं, तत्समक $(e_G, e_H)$ और प्रतिलोम
$(g^{-1}, h^{-1})$ हैं। कोटियाँ लघुत्तम समापवर्त्य से जुड़ती हैं: $(g, h)^m
= (g^m, h^m)$ तत्समक तभी है जब $g$ की \omterm{def:b1:structures:order}{कोटि} और $h$ की \omterm{def:b1:structures:order}{कोटि} दोनों $m$ को
विभाजित करें। अतः $\Z/2\Z \times \Z/2\Z$ (योगात्मक) में प्रत्येक अशून्य \omterm{def:b1:structures:order}{अवयव
की कोटि} $2$ है --- और यह ठीक \cref{ex:b1:structures:klein} वाला आयत \omterm{def:b1:structures:group}{समूह} ही
है, निर्देशांकों में --- जबकि $\Z/4\Z$ में \omterm{def:b1:structures:order}{कोटि} $4$ का अवयव है: यह दूसरी,
बिना किसी संगणना की उपपत्ति है कि $4$ आकार के दोनों \omterm{def:b1:structures:group}{समूह} तुल्याकारी नहीं हैं
(तुल्याकारिता कोटियाँ सुरक्षित रखती है)। पुराने समूहों से नए \omterm{def:b1:structures:group}{समूह} गढ़ने का
सबसे आसान उपाय गुणन है, और \cref{ch:b1:vspaces} का समतल $\R^2 = \R \times
\R$ इस रचना का सबसे महत्त्वपूर्ण उदाहरण है।
\end{example}

\begin{remark}[फर्मा, संरचनात्मक रूप में]
\omterm{def:b1:structures:field}{क्षेत्र} $\Z/p\Z$ में अशून्य वर्ग $p - 1$ अवयवों वाला गुणात्मक \omterm{def:b1:structures:group}{समूह} बनाते
हैं, और फर्मा की लघु प्रमेय (\cref{thm:b1:arith:fermat}) कहती है: इस \omterm{def:b1:structures:group}{समूह} का
प्रत्येक अवयव $x$ $x^{p-1} = \overline 1$ को संतुष्ट करता है। यह परिमित
समूहों के विषय में एक व्यापक तथ्य (लाग्रांज की प्रमेय) का उदाहरण है, जो
द्वितीय वर्ष में सिद्ध किया जाता है; विल्सन की प्रमेय की युग्मन उपपत्ति
(\cref{exo:b1:arith:11}) में भी यही समूह-सैद्धांतिक स्वाद पहले से था।
\end{remark}

\begin{remark}[मध्यांतर: अमूर्तता से क्या मिलता है]\label{rem:b1:structures:interlude}
यह पूछना उचित है कि, मान लीजिए, \cref{prop:b1:structures:unique} को संख्याओं
के बदले किसी अमूर्त नियम के लिए सिद्ध करने से क्या मिला। उत्तर है लाभ का
उत्तोलन। वह दो पंक्ति का तर्क अब एक साथ इन सबको समेटता है: संयोजन के अंतर्गत
फलनों के प्रतिलोम (\cref{thm:b1:logic:inverse}, जिसकी अद्वितीयता की उपपत्ति
वह शब्दशः दोहराता है), $n$ के सापेक्ष प्रतिलोम
(\cref{prop:b1:arith:invmod}), अशून्य वास्तविक संख्याओं के प्रतिलोम, किसी भी
\omterm{def:b1:structures:ring}{वलय} की इकाइयों के प्रतिलोम, और --- बिना देखे ही --- \cref{ch:b1:matrices} के
व्युत्क्रमणीय आव्यूहों के प्रतिलोम, जहाँ $A^{-1}$ की अद्वितीयता के लिए
उपपत्ति की एक पंक्ति भी नहीं चाहिए होगी। यही मितव्ययिता
\cref{prop:b1:structures:kernel} पर भी लागू है (एक ही एकैकीयता कसौटी, जो
\cref{ch:b1:linmaps} में रैखिक प्रतिचित्रणों के लिए फिर काम आती है) और
\omterm{def:b1:structures:subgroup}{उपसमूह} कसौटी पर भी। यहाँ अमूर्तता व्यापकता के लिए व्यापकता नहीं है: वह एक ही
प्रमेयिका को पाँच नामों से पाँच बार सिद्ध करने से इनकार है। इसकी कीमत --- यह
हिसाब रखना कि हर \omterm{def:b1:logic:statement}{कथन} ने वास्तव में कौन-से अभिगृहीत प्रयुक्त किए --- ठीक वही
है जिसका अभ्यास इस अध्याय के अभ्यास कराते हैं।
\end{remark}

\begin{remark}[यह अध्याय कहाँ काम आता है]\label{rem:b1:structures:whereused}
इस अध्याय की शब्दावली शेष खंड का व्याकरण है। \omterm{def:b1:structures:ring}{वलय} और \omterm{def:b1:structures:field}{क्षेत्र}
\cref{ch:b1:poly} ($K[X]$ $\Z$ का अनुकरण करने वाला \omterm{def:b1:structures:ring}{वलय} है) और
\cref{ch:b1:fractions} ($K(X)$ उसका भिन्न-क्षेत्र है) को व्यवस्थित करते हैं;
सदिश समष्टियाँ (\cref{ch:b1:vspaces}) ऐसे \omterm{def:b1:structures:group}{आबेली समूह} हैं जिन पर कोई \omterm{def:b1:structures:field}{क्षेत्र}
क्रिया करता है; आव्यूह (\cref{ch:b1:matrices}) इस खंड का पहला गंभीर रूप से
अक्रमविनिमेय \omterm{def:b1:structures:ring}{वलय} बनाते हैं, और उनके व्युत्क्रमणीय अवयव ऐसा \omterm{def:b1:structures:group}{समूह} बनाते हैं
जिसका अध्ययन स्वयं रैखिक बीजगणित है। समाकारिताएँ और अष्टियाँ
\cref{ch:b1:linmaps} में रैखिक प्रतिचित्रणों और शून्य समष्टियों के रूप में
लौटती हैं --- \cref{prop:b1:structures:kernel} उस अध्याय की एकैकीयता कसौटी
\emph{ही है}, जो यहाँ एक बार और सदा के लिए सिद्ध हो चुकी है। \omterm{ex:b1:structures:groups}{सममित समूह}, जो
नीचे दी गई सप्ताहांत समस्या का नायक है, वह संकेतक देता है जिस पर
\cref{ch:b1:det} में सारणिक खड़े किए जाते हैं।
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:5}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:6}
मान लीजिए $H, K$ $G$ के \omterm{def:b1:structures:subgroup}{उपसमूह} हैं। सिद्ध कीजिए कि $H \cap K$ \omterm{def:b1:structures:subgroup}{उपसमूह} है, और

H$ हो। \emph{(यदि $h \in H \setminus K$ और $k \in K \setminus H$, तो $hk$
कहाँ रह सकता है?)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:structures:10}

\begin{exercise}[$\star\star\star$]\label{exo:b1:structures:11}

\end{exercise}

\section{समस्या: सममित समूह और 8-पहेली}

\begin{problem}\label{pb:b1:structures:1}
$\intint1n$ के क्रमचयों का \omterm{def:b1:structures:group}{समूह} $\mathfrak S_n$ गणित का सबसे पुराना \omterm{def:b1:structures:group}{समूह} है
और आज भी सबसे शिक्षाप्रद। यह समस्या उसका संरचना-सिद्धांत शून्य से खड़ा करती
है --- चक्र, व्यत्यासों से जनन, \emph{संकेतक} \omterm{def:b1:structures:morphism}{समाकारिता} $\varepsilon \colon
\mathfrak S_n \to \{\pm1\}$ (जिसका अस्तित्व सचमुच अतुच्छ है), और $3$-चक्रों
से जनित \omterm{pb:b1:structures:1}{प्रत्यावर्ती समूह} $\mathfrak A_n$ --- और फिर उसे एक चिरपरिचित पहेली
पर भुनाती है: $3 \times 3$ सरकती-पट्टिका वाले खेल में चालों का कोई भी
अनुक्रम दो पट्टिकाओं की अदला-बदली करके शेष सब कुछ यथावत् नहीं रख सकता।
\index{सममित समूह}\index{संकेतक}\index{प्रत्यावर्ती समूह} \omterm{def:b1:counting:objects}{क्रमचय} $\intint1n$
पर क्रिया करते हैं; गुणनफल $\sigma\tau$ का अर्थ है ``पहले $\tau$ लगाइए'';
$[\,v_1, \dots, v_n]$ वह \omterm{def:b1:counting:objects}{क्रमचय} दर्शाता है जो $i$ को $v_i$ पर भेजता है।

\medskip \noindent\textbf{भाग I --- चक्र और व्यत्यास।}
\begin{enumerate}
  \item $\abs{\mathfrak S_n} = n!$ को न्यायसंगत ठहराइए
  (\cref{thm:b1:counting:counts})। $\mathfrak S_3$ में $\sigma = [2, 3, 1]$
  और $\tau = [1, 3, 2]$ के दोनों गुणनफल संगणित कीजिए, और निष्कर्ष निकालिए कि
  $\mathfrak S_3$ \omterm{def:b1:structures:group}{आबेली} नहीं है।
  \item \emph{$k$-चक्र} $(a_1\ a_2\ \dots\ a_k)$ ($k \geq 2$, जहाँ $a_i$
  जोड़ों में भिन्न हैं) $a_1 \mapsto a_2 \mapsto \dots \mapsto a_k \mapsto
  a_1$ भेजता है और शेष सब कुछ अचल रखता है; उसका \emph{आश्रय} $\{a_1, \dots,
  a_k\}$ है। सिद्ध कीजिए कि असंयुक्त आश्रयों वाले दो चक्र क्रम-विनिमेय होते
  हैं।
  \item सिद्ध कीजिए कि प्रत्येक $\sigma \in \mathfrak S_n$ जोड़ों में
  असंयुक्त आश्रयों वाले चक्रों का गुणनफल है, और यह वियोजन गुणनखंडों के क्रम
  तक अद्वितीय है। \emph{(प्रत्येक $i$ के लिए अनुक्रम $i, \sigma(i),
  \sigma^2(i), \dots$ पर विचार कीजिए: उसे $i$ पर लौटना ही पड़ता है; इस
  प्रकार बनी \emph{कक्षाएँ} $\intint1n$ का \omterm{thm:b1:logic:partition}{विभाजन} करती हैं, और $\sigma$
  प्रत्येक पर चक्र की भाँति क्रिया करता है।)}
  \item $\sigma = [4, 1, 5, 2, 3, 7, 8, 6] \in \mathfrak S_8$ का असंयुक्त
  चक्रों में वियोजन कीजिए। \cref{def:b1:structures:order} की भाँति $\sigma$
  की \emph{\omterm{def:b1:structures:order}{कोटि}} परिभाषित करते हुए सिद्ध कीजिए कि असंयुक्त चक्रों के गुणनफल
  की \omterm{def:b1:structures:order}{कोटि} उनकी लंबाइयों का लघुत्तम समापवर्त्य है, और इस $\sigma$ की \omterm{def:b1:structures:order}{कोटि}
  संगणित कीजिए।
  \item दूरबीनी सर्वसमिका सिद्ध कीजिए
        \[
        (a_1\ a_2\ \dots\ a_k)
        = (a_1\ a_k)(a_1\ a_{k-1})\cdots(a_1\ a_2) ,
        \]
        और निष्कर्ष निकालिए कि प्रत्येक \omterm{def:b1:counting:objects}{क्रमचय} व्यत्यासों का गुणनफल है।
        प्रश्न 4 के $\sigma$ को ऐसे गुणनफल के रूप में लिखिए।
  \item आगे दिखाइए कि \emph{संलग्न} व्यत्यास $(i\ \ i{+}1)$ ही पर्याप्त हैं:
  $a < b$ के लिए
        \[
        (a\ b) = (a\ \ a{+}1)(a{+}1\ \ a{+}2)\cdots(b{-}1\ \ b)
        \cdots(a{+}1\ \ a{+}2)(a\ \ a{+}1),
        \]
        जो $2(b - a) - 1$ संलग्न व्यत्यासों का गुणनफल है --- अर्थात् एक
        \emph{विषम} संख्या (यह सम-विषमता नीचे दो बार महत्त्व रखेगी)।
\end{enumerate}

\medskip \noindent\textbf{भाग II --- संकेतक का अस्तित्व।} $\sigma \in
\mathfrak S_n$ के लिए मान लीजिए
\[
N(\sigma) = \#\bigl\{(i, j) : i < j,\ \sigma(i) >
\sigma(j)\bigr\}
\]
उसके \emph{व्युत्क्रमणों} की संख्या है, और $\varepsilon(\sigma) =
(-1)^{N(\sigma)}$ रखिए।
\begin{enumerate}[resume]
  \item तत्समक के लिए, व्यत्यास $(i\ \ i{+}1)$ के लिए, और $[2, 3, 1]$ के लिए
  $N$ तथा $\varepsilon$ संगणित कीजिए।
  \item सिद्ध कीजिए कि प्रत्येक $\sigma$ और प्रत्येक संलग्न व्यत्यास $\tau =
  (i\ \ i{+}1)$ के लिए $N(\sigma\tau) = N(\sigma) \pm 1$। \emph{(दाईं ओर से
  $\tau$ के साथ संयोजन स्थानों $i$ और $i + 1$ के मानों की अदला-बदली कर देता
  है; ठीक एक युग्म अपनी व्युत्क्रमण-स्थिति बदलता है।)}
  \item प्रश्न 6 का प्रयोग करके निकालिए कि \emph{किसी भी} व्यत्यास $\tau$ के
  लिए $\varepsilon(\sigma\tau) = -\varepsilon(\sigma)$; निष्कर्ष निकालिए कि
  यदि $\sigma$ $p$ व्यत्यासों का गुणनफल है, तो $\varepsilon(\sigma) =
  (-1)^p$ --- विशेष रूप से $p$ की सम-विषमता केवल $\sigma$ पर निर्भर करती है,
  चुने हुए गुणनखंडन पर नहीं --- और यह कि $\varepsilon \colon \mathfrak S_n
  \to \{\pm 1\}$ एक \omterm{def:b1:structures:group}{समूह} \omterm{def:b1:structures:morphism}{समाकारिता} है।
  \item दिखाइए कि $k$-चक्र का संकेतक $(-1)^{k-1}$ है, और यह कि व्यापक रूप से
  $\varepsilon(\sigma) = (-1)^{n - c(\sigma)}$, जहाँ $c(\sigma)$ $\sigma$ की
  कक्षाओं की संख्या है (अचल बिंदु सम्मिलित)।
  \item \emph{\omterm{pb:b1:structures:1}{प्रत्यावर्ती समूह}} $\mathfrak A_n = \ker\varepsilon$ है।
  न्यायसंगत ठहराइए कि वह \omterm{def:b1:structures:subgroup}{उपसमूह} है, और $n \geq 2$ के लिए $\abs{\mathfrak
  A_n} = \frac{n!}2$ सिद्ध कीजिए। \emph{(एक व्यत्यास $\tau_0$ नियत कीजिए और
  $\sigma \mapsto \sigma\tau_0$ पर विचार कीजिए।)}
  \item $\sigma = [4, 1, 5, 2, 3, 7, 8, 6]$ पर संगति की जाँच:
  $\varepsilon(\sigma)$ तीन प्रकार से संगणित कीजिए --- व्युत्क्रमण गिनकर,
  प्रश्न 10 के द्वारा चक्र-प्रकार से, और प्रश्न 5 में की गई अपनी
  व्यत्यास-गणना से।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- $\mathfrak A_n$ $3$-चक्रों से जनित
है।}
\begin{enumerate}[resume]
  \item मान लीजिए $a, b, c, d$ जोड़ों में भिन्न हैं। दोनों सर्वसमिकाएँ
  सत्यापित कीजिए
        \[
        (a\ b)(a\ c) = (a\ c\ b),
        \qquad
        (a\ b)(c\ d) = (a\ c\ b)(a\ c\ d) .
        \]
  \item सिद्ध कीजिए कि $n \geq 3$ के लिए $\mathfrak A_n$ का प्रत्येक अवयव
  $3$-चक्रों का गुणनफल है। \emph{(सम \omterm{def:b1:counting:objects}{क्रमचय} व्यत्यासों की सम संख्या का
  गुणनफल होता है; उन्हें दो के जोड़ों में समेटिए।)}
  \item $(1\ 2)(3\ 4)$ और $5$-चक्र $(1\ 2\ 3\ 4\ 5)$ को $3$-चक्रों के गुणनफल
  के रूप में स्पष्ट रूप से लिखिए।
  \item संयुग्मन सूत्र सिद्ध कीजिए: प्रत्येक $\sigma \in \mathfrak S_n$ के
  लिए
        \[
        \sigma\,(a_1\ \dots\ a_k)\,\sigma^{-1}
        = \bigl(\sigma(a_1)\ \dots\ \sigma(a_k)\bigr) .
        \]
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- 8-पहेली।} पट्टिकाएँ $1, \dots, 8$ एक $3
\times 3$ चौखट में सरकती हैं, जिसमें एक कोष्ठ रिक्त है; एक \emph{चाल} रिक्त
कोष्ठ से सटी हुई किसी पट्टिका को उसमें सरका देती है। कोष्ठों को $1, \dots,
9$ संख्याएँ दीजिए (पंक्ति-दर-पंक्ति; हल की हुई स्थिति में पट्टिका $i$ कोष्ठ
$i$ में है और रिक्त कोष्ठ $9$ पर है)। रिक्त कोष्ठ को नौवीं पट्टिका मानिए,
जिससे कोई स्थिति एक \omterm{def:b1:counting:objects}{क्रमचय} $\sigma \in \mathfrak S_9$ हो जाती है (पट्टिका
$\sigma(i)$ कोष्ठ $i$ में बैठती है)।
\begin{enumerate}[resume]
  \item दिखाइए कि कोई चाल $\sigma$ के स्थान पर $\sigma \circ \tau$ रख देती
  है, जहाँ $\tau$ दोनों संबंधित कोष्ठों का व्यत्यास है; उससे निकालिए कि हर
  चाल $\varepsilon(\sigma)$ को पलट देती है।
  \item मान लीजिए $d(\sigma)$ रिक्त कोष्ठ के वर्तमान कोष्ठ और उसके घर-कोष्ठ
  $9$ के बीच की पंक्ति-स्तंभ दूरी (पंक्तियाँ और स्तंभ मिलाकर) है। दिखाइए कि
  हर चाल $d$ को $\pm1$ से बदलती है, अतः हर चाल $(-1)^{d(\sigma)}$ को भी पलट
  देती है। निष्कर्ष निकालिए कि
        \[
        I(\sigma) = \varepsilon(\sigma)\cdot(-1)^{d(\sigma)}
        \]
        हर चाल के अंतर्गत \emph{अपरिवर्ती} है।
  \item पहेली की चिरपरिचित असंभवता सिद्ध कीजिए: जो स्थिति पट्टिकाओं $7$ और
  $8$ की अदला-बदली करती है और शेष सब कुछ (रिक्त कोष्ठ सहित) यथावत् रखती है,
  वह हल की हुई स्थिति से नहीं पाई जा सकती।
  \item हम विलोम मान लेते हैं (उसकी उपपत्ति शिक्षाप्रद पर लंबा आगमन है): $I
  = +1$ वाली प्रत्येक स्थिति तक पहुँचा जा सकता है। उससे निकालिए कि रिक्त
  कोष्ठ के घर पर होने वाली $8!$ स्थितियों में से ठीक आधी हल हो सकने वाली
  हैं, अर्थात् $\frac{8!}2 = 20\,160$।
  \item प्रश्न 20 से निकालिए कि रिक्त कोष्ठ के घर पर होने वाले, पहुँचे जा
  सकने वाले पट्टिका-विन्यास ठीक \omterm{def:b1:structures:subgroup}{उपसमूह} $\mathfrak A_8 \leq \mathfrak S_8$
  बनाते हैं।
  \item अपरिवर्त्य के अनुप्रयोग: क्या (क) वह स्थिति पाई जा सकती है जिसमें
  पट्टिकाएँ $1, 2, 3$ चक्रीय रूप से क्रमचयित हों और शेष सब कुछ, रिक्त कोष्ठ
  सहित, घर पर हो? (ख) वह स्थिति जिसमें पट्टिका $5$ और रिक्त कोष्ठ ने स्थान
  बदल लिए हों और शेष सभी पट्टिकाएँ घर पर हों? दोनों उत्तर $I$ से न्यायसंगत
  ठहराइए।
\end{enumerate}

\medskip \noindent\textbf{भाग V --- संश्लेषण।}
\begin{enumerate}[resume]
  \item सिद्ध कीजिए कि $n \geq 3$ के लिए एकमात्र \omterm{def:b1:structures:group}{समूह} समाकारिताएँ $f \colon
  \mathfrak S_n \to \{\pm 1\}$ अचर \omterm{def:b1:structures:morphism}{समाकारिता} और $\varepsilon$ हैं।
  \emph{(प्रश्न 16 और $\{\pm1\}$ की क्रमविनिमेयता का प्रयोग करके दिखाइए कि
  $f$ सभी व्यत्यासों पर एक ही मान लेता है।)}
  \item इस समस्या में ठीक कहाँ प्रयोग हुआ: (क) \omterm{def:b1:structures:morphism}{समाकारिता} की धारणा और
  \cref{prop:b1:structures:kernel}; (ख) \cref{ch:b1:counting} के गणना-नियम;
  (ग) वह सुपरिभाषितता का प्रश्न जिसे प्रश्न 8--9 सुलझाते हैं? प्रत्येक के
  लिए एक वाक्य।
  \item एक छोटे अनुच्छेद में संश्लेषण: एक ही सम-विषमता फलन, जिसकी
  सुपरिभाषितता एक बार सिद्ध हुई, एक साथ $\mathfrak S_n$ की भीतरी संरचना
  (\omterm{def:b1:structures:subgroup}{उपसमूह} $\mathfrak A_n$) को व्यवस्थित करता है, एक भौतिक पहेली का निर्णय
  करता है, और --- सूत्र $\det = \sum_\sigma \varepsilon(\sigma)\cdots$ के
  द्वारा --- \cref{ch:b1:det} में सारणिक परिभाषित करेगा। इस बार-बार लौटने
  वाले प्रतिरूप पर टिप्पणी कीजिए: अपरिवर्त्य ``चालों के सारे अनुक्रम
  आज़माइए'' को एक ही संगणना में बदल देते हैं।
\end{enumerate}
\end{problem}
