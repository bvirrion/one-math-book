\chapter{सम्मिश्र संख्याएँ}\label{ch:b1:complex}

सम्मिश्र संख्याएँ उच्चतर माध्यमिक खंड में द्विघात समीकरणों की संगणना की एक
युक्ति के रूप में मिली थीं। यह अध्याय उन्हें एक केंद्रीय वस्तु मानता है:
चरघातांकी रूप और उसके परिणाम (दे मॉयवर, $n$-वें मूल, \omterm{def:b1:complex:unity}{इकाई के मूल}), $\C$ और
समतल ज्यामिति के बीच व्यवस्थित अनुवाद, तथा त्रिकोणमितीय सर्वसमिकाएँ सिद्ध
करने की मशीन के रूप में $\eu^{\iu\theta}$ का प्रयोग।

\section{क्षेत्र $\C$, मापांक और संयुग्मी}

\begin{definition}[सम्मिश्र क्षेत्र]\label{def:b1:complex:field}
$\C = \{a + \iu b : a, b \in \R\}$, जिसमें सामान्य योग है और गुणन $\iu^2 =
-1$ से निर्धारित होता है। प्रत्येक अशून्य $z = a + \iu b$ का प्रतिलोम है:
$z^{-1} = \frac{a - \iu b}{a^2 + b^2}$ --- \cref{ch:b1:structures} की भाषा
में, $\C$ एक क्षेत्र है। लिखा जाता है $a = \Re(z)$, $b = \Im(z)$, $\conj{z}
= a - \iu b$ (\emph{संयुग्मी}\index{संयुग्मी}) और $\abs{z} = \sqrt{a^2 +
b^2}$ (\emph{मापांक}\index{मापांक})।
\end{definition}

\begin{proposition}[संयुग्मी और मापांक के नियम]\label{prop:b1:complex:rules}
$z, w \in \C$ के लिए:
\begin{enumerate}
  \item $\conj{z + w} = \conj z + \conj w$, $\conj{zw} = \conj z\,
        \conj w$, $\conj{\conj z} = z$;
  \item $z \conj z = \abs z^2$; $\;\Re(z) = \frac{z + \conj z}{2}$,
        $\Im(z) = \frac{z - \conj z}{2\iu}$;
  \item $\abs{zw} = \abs z\, \abs w$, और $z \neq 0$ के लिए $\abs{z^{-1}} =
  \abs{z}^{-1}$;
  \item (त्रिभुज असमिका\index{त्रिभुज असमिका}) $\abs{z + w} \leq \abs z +
  \abs w$, जहाँ समता यदि और केवल यदि $z$ और $w$ $0$ से निकलने वाली एक ही
  किरण पर हों ($w = \lambda z$, या $\lambda \geq 0$ के साथ $z = \lambda w$);
  \item (विलोम त्रिभुज असमिका) $\bigl|\abs z - \abs w\bigr| \leq \abs{z -
  w}$।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) और (2) वास्तविक तथा काल्पनिक भागों पर सीधी संगणनाएँ हैं। (3):
$\abs{zw}^2 = zw\,\conj{zw} = z\conj z\, w \conj w = \abs z^2 \abs w^2$, फिर
वर्गमूल लीजिए; प्रतिलोम के लिए $z \cdot z^{-1} = 1$ पर लगाइए।

(4) दोनों पक्ष ऋणेतर हैं, अतः वर्गों की तुलना कीजिए:
\[
\abs{z+w}^2 = (z+w)(\conj z + \conj w)
= \abs z^2 + \abs w^2 + 2\,\Re(z \conj w),
\]
और $\Re(z\conj w) \leq \abs{z \conj w} = \abs z \abs w$ से $\abs{z+w}^2 \leq
(\abs z + \abs w)^2$ मिलता है। समता के लिए $\Re(z \conj w) = \abs{z \conj
w}$ आवश्यक है, अर्थात् $z \conj w \in \R_+$; यदि $w \neq 0$, तो इससे
$\lambda \geq 0$ के साथ $z = \frac{z\conj w}{\abs w^2}\, w = \lambda w$
मिलता है (और स्थिति $w = 0$ तुच्छ है)।

(5) $\abs z = \abs{(z - w) + w} \leq \abs{z-w} + \abs w$ से $\abs z - \abs w
\leq \abs{z - w}$ मिलता है; दूसरे चिह्न के लिए $z$ और $w$ की अदला-बदली
कीजिए।
\end{proof}

\begin{example}[मापांक और कोणांक का पूरा अभ्यास]\label{ex:b1:complex:workout}
$w = \dfrac{3 + 4\iu}{1 - 2\iu}$ को बीजीय रूप में लिखिए और उसका \omterm{def:b1:complex:field}{मापांक} दो
बार संगणित कीजिए। हर के \omterm{def:b1:complex:field}{संयुग्मी} से गुणा करने पर:
\[
w = \frac{(3 + 4\iu)(1 + 2\iu)}{(1 - 2\iu)(1 + 2\iu)}
= \frac{3 + 6\iu + 4\iu - 8}{1 + 4}
= \frac{-5 + 10\iu}{5} = -1 + 2\iu .
\]
सीधे: $\abs w = \sqrt{1 + 4} = \sqrt5$। भागफल के नियम
(\cref{prop:b1:complex:rules}~(3)) से: $\abs w = \frac{\abs{3 +
4\iu}}{\abs{1 - 2\iu}} = \frac5{\sqrt5} = \sqrt5$ --- वही उत्तर, और बीजीय
रूप की कोई आवश्यकता नहीं। यह शिक्षा सामान्य है: \omterm{def:b1:complex:field}{मापांक} और कोणांक गुणनफल तथा
भागफल से होकर अच्छी तरह जाते हैं, और वास्तविक तथा काल्पनिक भाग योगों से
होकर। जो निरूपण सामने की संक्रियाओं से मेल खाता हो उसे चुनिए, और बदलिए तभी
जब विवश होना पड़े।
\end{example}

\begin{example}[संयुग्मी वाले समीकरण]\label{ex:b1:complex:conjeq}
$\C$ में हल कीजिए: $\;z + 2\conj z = 6 + 2\iu$। $z$ और $\conj z$ को मिलाने
वाला समीकरण $z$ में बहुपद \emph{नहीं} है; भरोसेमंद चाल है वास्तविक
निर्देशांकों में तोड़ देना। $z = x + \iu y$ के साथ:
\[
z + 2\conj z = 3x - \iu y ,
\]
अतः समीकरण $3x = 6$ और $-y = 2$ कहता है: अद्वितीय हल $z = 2 - 2\iu$ है।
(जाँच: $(2 - 2\iu) + 2(2 + 2\iu) = 6 + 2\iu$।) वैकल्पिक रूप से, पूरे समीकरण
का \omterm{def:b1:complex:field}{संयुग्मी} लेकर $\conj z + 2z = 6 - 2\iu$ प्राप्त कीजिए और अज्ञातों $z,
\conj z$ में रैखिक निकाय हल कीजिए --- वही उत्तर, और गुणांकों के सम्मिश्र
होने पर यह उपयोगी युक्ति है। $z$ और $\conj z$ वाले समीकरण वस्तुतः दो
वास्तविक समीकरणों के निकाय हैं; यहाँ ``घात $1$, एक हल'' की अपेक्षा सुरक्षित
है, पर $z\conj z = -1$ (कोई हल नहीं) दिखाता है कि गुणनफल आते ही बहुपदीय
अंतर्ज्ञान विफल हो जाता है।
\end{example}

\section{चरघातांकी रूप}

\begin{definition}[काल्पनिक प्रांतिक का सम्मिश्र चरघातांकी]\label{def:b1:complex:expi}
$\theta \in \R$ के लिए परिभाषित कीजिए
\[
\eu^{\iu\theta} = \cos\theta + \iu \sin\theta .
\]
प्रत्येक $z \neq 0$ को $r = \abs z > 0$ के साथ $z = r\,\eu^{\iu\theta}$ लिखा
जा सकता है; $\theta$ $z$ का एक \emph{कोणांक}\index{कोणांक} है, जो $2\pi$ का
गुणज जोड़ने तक निर्धारित है। $\intoc{-\pi}{\pi}$ में स्थित मान \emph{मुख्य
कोणांक} है, जिसे $\arg z$ लिखते हैं।
\end{definition}

\begin{example}[पहले मान, और एक प्रसिद्ध सर्वसमिका]\label{ex:b1:complex:eulerid}
मुख्य कोणों पर परिभाषा पढ़ने से:
\[
\eu^{\iu\pi/2} = \iu, \qquad
\eu^{\iu\pi} = -1, \qquad
\eu^{2\iu\pi} = 1, \qquad
\eu^{\iu\pi/4} = \frac{\sqrt2}2\,(1 + \iu) .
\]
इनमें दूसरी, $\eu^{\iu\pi} + 1 = 0$ के रूप में पुनर्व्यवस्थित करने पर,
$\eu$, $\iu$, $\pi$, $1$ और $0$ को जोड़ने वाली ऑयलर की प्रसिद्ध सर्वसमिका
है; पुस्तक के इस पड़ाव पर वह प्रमेय से अधिक एक परिभाषा का खुलना है, और उसकी
असली सामग्री --- \cref{ch:b1:functions} का विश्लेषणात्मक चरघातांकी फलन,
काल्पनिक प्रांतिकों तक विस्तारित होकर, उसी नाम का अधिकारी क्यों है ---
\cref{ch:b1:series} की घात श्रेणी से तय होती है। इस बीच ऊपर का प्रदर्शन एक
रूपांतरण-सारणी के रूप में कंठस्थ करने योग्य है: जब भी चित्र से कोई कोणांक
पढ़ा जाता है, वह मौन रूप से प्रयुक्त होता है।
\end{example}

\begin{theorem}[फलनीय समीकरण]\label{thm:b1:complex:funceq}
सभी $\theta, \varphi \in \R$ के लिए:
\[
\eu^{\iu\theta}\, \eu^{\iu\varphi} = \eu^{\iu(\theta + \varphi)},
\qquad
\abs{\eu^{\iu\theta}} = 1,
\qquad
\conj{\eu^{\iu\theta}} = \eu^{-\iu\theta} = (\eu^{\iu\theta})^{-1}.
\]
फलस्वरूप $\abs{zw} = \abs z \abs w$ कोणांकों को भी ले चलता है: $\arg(zw)
\equiv \arg z + \arg w \pmod{2\pi}$।
\end{theorem}

\begin{proof}
गुणनफल का प्रसार कीजिए और योग-सूत्र प्रयोग कीजिए:
\[
(\cos\theta + \iu\sin\theta)(\cos\varphi + \iu\sin\varphi)
= (\cos\theta\cos\varphi - \sin\theta\sin\varphi)
+ \iu\,(\sin\theta\cos\varphi + \cos\theta\sin\varphi),
\]
जो $\cos(\theta+\varphi) + \iu\sin(\theta+\varphi)$ है। \omterm{def:b1:complex:field}{मापांक}
$\sqrt{\cos^2\theta + \sin^2\theta} = 1$ है, और \omterm{def:b1:complex:field}{संयुग्मी} का सूत्र कोज्या तथा
ज्या की सम-विषमता है; वह $\eu^{\iu\theta}$ का प्रतिलोम कर देता है, क्योंकि
$\eu^{\iu\theta}\eu^{-\iu\theta} = \eu^0 = 1$।
\end{proof}

\begin{corollary}[दे मॉयवर का सूत्र]\label{cor:b1:complex:demoivre}
$\theta \in \R$ और $n \in \Z$ के लिए: $\;(\cos\theta + \iu\sin\theta)^n =
\cos(n\theta) + \iu\sin(n\theta)$।\index{दे मॉयवर का सूत्र}
\end{corollary}

\begin{proof}
$n \geq 0$ के लिए आगमन कीजिए। स्थिति $n = 0$ $1 = 1$ कहती है। $n$ के लिए
सूत्र मानने पर फलनीय समीकरण (\cref{thm:b1:complex:funceq}) देता है
\[
\bigl(\eu^{\iu\theta}\bigr)^{n+1}
= \bigl(\eu^{\iu\theta}\bigr)^{n}\,\eu^{\iu\theta}
= \eu^{\iu n\theta}\,\eu^{\iu\theta}
= \eu^{\iu(n+1)\theta} ,
\]
जो कोटि $n + 1$ पर सूत्र है। $n < 0$ के लिए $m > 0$ के साथ $n = -m$ लिखिए:
चूँकि $\eu^{\iu m\theta}$ का प्रतिलोम $\eu^{-\iu m\theta}$ है (फिर
\cref{thm:b1:complex:funceq}),
\[
\bigl(\eu^{\iu\theta}\bigr)^{-m}
= \bigl(\eu^{\iu m\theta}\bigr)^{-1}
= \eu^{\iu(-m)\theta} . \qedhere
\]
\end{proof}

\begin{example}[दे मॉयवर से $\cos 3\theta$ का प्रसार]\label{ex:b1:complex:cos3}
$c = \cos\theta$, $s = \sin\theta$ लिखिए। दे मॉयवर और द्विपद प्रमेय देते हैं
\[
\cos 3\theta + \iu \sin 3\theta = (c + \iu s)^3
= c^3 - 3cs^2 + \iu\,(3c^2 s - s^3),
\]
और वास्तविक भागों की पहचान करके, फिर $s^2 = 1 - c^2$ प्रतिस्थापित करके:
\[
\cos 3\theta = c^3 - 3c(1 - c^2) = 4\cos^3\theta - 3\cos\theta .
\]
काल्पनिक भाग $\sin 3\theta = 3\sin\theta - 4\sin^3\theta$ मुफ़्त में दे देता
है: एक सम्मिश्र सर्वसमिका सदा \emph{दो} वास्तविक सर्वसमिकाएँ साथ लाती है।
उलटी दिशा में पढ़ने पर, बॉक्स वाली सर्वसमिका चिरपरिचित त्रिभाजन समीकरण की
कुंजी है: $\cos\theta$ से $\cos(\theta/3)$ की रचना का अर्थ है त्रिघात $4x^3
- 3x = \cos\theta$ हल करना, और वहीं से \cref{ch:b1:poly} का बीजगणित काम
सँभाल लेता है।
\end{example}

\begin{proposition}[ऑयलर के सूत्र]\label{prop:b1:complex:euler}
\[
\cos\theta = \frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2},
\qquad
\sin\theta = \frac{\eu^{\iu\theta} - \eu^{-\iu\theta}}{2\iu}.
\]
\end{proposition}

\begin{proof}
$\eu^{\iu\theta} = \cos\theta + \iu\sin\theta$ और $\eu^{-\iu\theta} =
\cos\theta - \iu\sin\theta$ को क्रमशः जोड़िए और घटाइए।
\end{proof}

\begin{method}[चरघातांकी के द्वारा त्रिकोणमिति]\label{met:b1:complex:trig}
\begin{enumerate}
  \item $\cos^p\theta \sin^q\theta$ का \emph{रैखिकीकरण कीजिए} (घातों को
  $\cos k\theta$, $\sin k\theta$ के योग में बदलिए): ऑयलर के सूत्र
  प्रतिस्थापित कीजिए, द्विपद प्रमेय (\cref{thm:b1:counting:binomial}) से
  प्रसार कीजिए, और \omterm{def:b1:complex:field}{संयुग्मी} पदों को समूहबद्ध कीजिए।
  \item $\cos n\theta$ का $\cos\theta$ में बहुपद के रूप में \emph{प्रसार
  कीजिए}: $\cos n\theta = \Re\bigl((\cos\theta + \iu\sin\theta)^n\bigr)$
  लिखिए, प्रसार कीजिए, और $\sin$ की सम घातों को $\sin^2 = 1 - \cos^2$ से बदल
  दीजिए।
  \item $\sum_k \cos k\theta$ जैसी \emph{त्रिकोणमितीय श्रेणियों का योग
  कीजिए}: उसमें गुणोत्तर योग $\sum_k (\eu^{\iu\theta})^k$ का वास्तविक भाग
  पहचानिए।
  \item \emph{अर्ध-कोण गुणनखंडन}: सभी $p, q$ के लिए,
        \[
        \eu^{\iu p} + \eu^{\iu q}
        = 2 \cos\tfrac{p - q}{2}\; \eu^{\iu \frac{p+q}{2}},
        \qquad
        \eu^{\iu p} - \eu^{\iu q}
        = 2\iu \sin\tfrac{p - q}{2}\; \eu^{\iu \frac{p+q}{2}} .
        \]
\end{enumerate}
\end{method}

\begin{example}[रैखिकीकरण]\label{ex:b1:complex:linearize}
\[
\cos^3\theta
= \Bigl(\frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}\Bigr)^{\!3}
= \frac{\eu^{3\iu\theta} + 3\eu^{\iu\theta} + 3\eu^{-\iu\theta} +
\eu^{-3\iu\theta}}{8}
= \frac{\cos 3\theta + 3\cos\theta}{4}.
\]
यह रूप तुरंत समाकलित हो जाता है --- यही कारण है कि \cref{ch:b1:integration}
में रैखिकीकरण महत्त्व रखता है। मिश्रित गुणनफल भी उसी प्रकार काम करता है,
केवल दोनों ऑयलर सूत्र एक साथ लगते हैं:
\[
\sin^2\theta\cos^2\theta
= \Bigl(\frac{\sin2\theta}2\Bigr)^{\!2}
= \frac{1}{4}\cdot
\Bigl(\frac{\eu^{2\iu\theta} - \eu^{-2\iu\theta}}{2\iu}
\Bigr)^{\!2}
= \frac{2 - \eu^{4\iu\theta} - \eu^{-4\iu\theta}}{16}
= \frac{1 - \cos4\theta}{8} ,
\]
जहाँ पहले पद में द्विगुण-कोण के संक्षेप ने एक द्विपद प्रसार बचा दिया ---
यंत्रवत् काम करने से पहले ऐसा कुछ ढूँढ़ लेना सदा लाभदायक है।
\end{example}

\begin{example}[एक द्विपद त्रिकोणमितीय योग]\label{ex:b1:complex:binomsum}
$n \in \N$ और $\theta \in \R$ के लिए $S = \sum_{k=0}^{n}\binom nk \cos
k\theta$ का मान निकालिए। उसमें द्विपद प्रसार का वास्तविक भाग पहचानिए:
\[
S = \Re\sum_{k=0}^n \binom nk \bigl(\eu^{\iu\theta}\bigr)^k
= \Re\bigl(1 + \eu^{\iu\theta}\bigr)^n ,
\]
फिर अर्ध-कोण गुणनखंडन (\cref{met:b1:complex:trig}~(4)) कीजिए: $1 +
\eu^{\iu\theta} = 2\cos\frac\theta2\,\eu^{\iu\theta/2}$, अतः
\[
S = \Re\Bigl(2^n\cos^n\frac\theta2\;\eu^{\iu n\theta/2}\Bigr)
= 2^n \cos^n\frac\theta2\,\cos\frac{n\theta}2 .
\]
काल्पनिक भाग $\sum_k\binom nk\sin k\theta =
2^n\cos^n\frac\theta2\sin\frac{n\theta}2$ मुफ़्त में दे देता है। जाँच:
$\theta = 0$ से $\sum\binom nk = 2^n$ वापस मिलता है, और $\theta = \pi$ से $n
\geq 1$ के लिए $S = 0$ मिलता है (प्रत्येक गुणनखंड $\cos\frac\pi2$ लुप्त हो
जाता है), अर्थात् \cref{ex:b1:counting:binomial} का एकांतरित पंक्ति-योग। यह
विधि --- ``कोज्या के योग को किसी सम्मिश्र घात की छाया के रूप में देखिए, फिर
अर्ध-कोण गुणनखंडन कीजिए'' --- ठीक \cref{exo:b1:complex:6} वाली है, बस
गुणोत्तर श्रेणी की जगह द्विपद प्रमेय आ जाता है।
\end{example}

\section{सम्मिश्र संख्याओं के मूल}

\begin{theorem}[$n$-वें मूल]\label{thm:b1:complex:roots}
मान लीजिए $a = r\,\eu^{\iu\alpha} \neq 0$ और $n \in \N^*$। समीकरण $z^n = a$
के ठीक $n$ हल हैं:
\[
z_k = r^{1/n}\, \eu^{\iu\left(\frac{\alpha}{n} +
\frac{2k\pi}{n}\right)},
\qquad k = 0, 1, \dots, n - 1 .
\]
\end{theorem}

\begin{proof}
$z = \rho\,\eu^{\iu\theta}$ ($\rho > 0$) लिखिए। तब $z^n = \rho^n \eu^{\iu
n\theta} = r \eu^{\iu\alpha}$ यदि और केवल यदि $\rho^n = r$ (\omterm{def:b1:complex:field}{मापांक}) और
$n\theta \equiv \alpha \pmod{2\pi}$ (कोणांक), अर्थात् किसी $k \in \Z$ के लिए
$\rho = r^{1/n}$ और $\theta = \frac{\alpha}{n} + \frac{2k\pi}{n}$। अब देखना
यह है कि दो पूर्णांक $k, k'$ एक ही संख्या कब देते हैं: यह ठीक तब होता है जब
कोण $2\pi$ के गुणज से भिन्न हों,
\[
\frac{2k\pi}n - \frac{2k'\pi}n \in 2\pi\Z
\iff \frac{k - k'}n \in \Z
\iff n \mid k - k' .
\]
यूक्लिडीय भाग से प्रत्येक $k \in \Z$ $n$ के सापेक्ष $\{0, 1, \dots, n-1\}$
के ठीक एक अवयव के सर्वांगसम है, अतः यह परास प्रत्येक हल को एक बार सूचीबद्ध
करता है और गणना ठीक $n$ है। (यही तर्क $\mathbb U_n$ के भीतर चलाने पर दिखता
है कि मूल एक सम $n$-भुज बनाते हैं: $k$ के क्रमागत मान नियत कोण
$\frac{2\pi}n$ से घुमा देते हैं।)
\end{proof}

\begin{example}[$-27$ के घनमूल]\label{ex:b1:complex:cuberoots}
$z^3 = -27$ हल कीजिए। दाएँ पक्ष का चरघातांकी रूप: $-27 = 27\,\eu^{\iu\pi}$,
अतः तीनों मूल हैं
\[
z_k = 3\,\eu^{\iu(\frac\pi3 + \frac{2k\pi}3)}, \quad k = 0, 1, 2 :
\qquad
z_0 = 3\eu^{\iu\pi/3} = \frac32 + \frac{3\sqrt3}2\,\iu,
\quad
z_1 = -3,
\quad
z_2 = \conj{z_0} .
\]
दो जाँच। पहली, वास्तविक मूल $-3$ स्पष्ट है, और शेष दो $\pm\frac{2\pi}3$ से
उसके घूर्णन हैं --- समतुल्य रूप से $-3j$ और $-3j^2$। दूसरी, बीजगणित पुष्टि
करता है: $z^3 + 27 = (z + 3)(z^2 - 3z + 9)$, और द्विघात का विविक्तकर $9 - 36
= -27 < 0$ है, जिसके मूल $\frac{3 \pm 3\iu\sqrt3}2 = z_0, \conj{z_0}$ हैं।
सार: वास्तविक दाएँ पक्षों के लिए अवास्तविक मूल सदा \omterm{def:b1:complex:field}{संयुग्मी} जोड़ों में आते
हैं, अतः \omterm{def:b1:logic:sets}{हल-समुच्चय} का चित्र वास्तविक अक्ष के सापेक्ष सममित होता है ---
\cref{ch:b1:poly} की वास्तविक गुणनखंडन प्रमेय की एक झलक।
\end{example}

\begin{example}[अवास्तविक दायाँ पक्ष]\label{ex:b1:complex:fourthroots}
$z^4 = -8 + 8\iu\sqrt3$ हल कीजिए। दाएँ पक्ष का चरघातांकी रूप: \omterm{def:b1:complex:field}{मापांक}
$\sqrt{64 + 192} = 16$, कोणांक $\theta$, जहाँ $\cos\theta = -\frac12$,
$\sin\theta = \frac{\sqrt3}2$, अर्थात् $\theta = \frac{2\pi}3$। चारों मूल
हैं
\[
z_k = 2\,\eu^{\iu(\frac\pi6 + \frac{k\pi}2)}, \quad k = 0, 1, 2,
3 :
\qquad
z_0 = \sqrt3 + \iu,\quad z_1 = \iu z_0 = -1 + \iu\sqrt3,
\]
\[
z_2 = -z_0 = -\sqrt3 - \iu,\qquad z_3 = -\iu z_0 = 1 - \iu\sqrt3 .
\]
(जाँच: $z_0^2 = 2 + 2\iu\sqrt3$, अतः $z_0^4 = (2 + 2\iu\sqrt3)^2 = 4 - 12 +
8\iu\sqrt3 = -8 + 8\iu\sqrt3$।) एक बार \emph{कोई एक} मूल मिल जाए तो शेष तीन
मुफ़्त में मिल जाते हैं: वे $\frac\pi2$ से उसके क्रमिक घूर्णन हैं, अर्थात्
इकाई के चौथे मूलों से उसके गुणनफल --- \cref{thm:b1:complex:roots} के पीछे की
सामान्य संरचना, जिसका उपयोग हर मूल को नए सिरे से संगणित करने से पहले कर लेना
चाहिए। इस बार \omterm{def:b1:complex:field}{संयुग्मी} सममिति नहीं है: दायाँ पक्ष वास्तविक नहीं है।
\end{example}

\begin{definition}[इकाई के मूल]\label{def:b1:complex:unity}
\emph{इकाई के $n$-वें मूल}\index{इकाई के मूल} $z^n = 1$ के हल हैं:
\[
\mathbb{U}_n = \bigl\{\, \omega^k : k = 0, \dots, n-1 \,\bigr\},
\qquad \omega = \eu^{2\iu\pi/n}.
\]
वे गुणन के अंतर्गत एक समूह बनाते हैं (\cref{ch:b1:structures}) और इकाई वृत्त
में अंतर्निहित सम $n$-भुज के शीर्षों पर बैठते हैं।
\end{definition}

\begin{proposition}[इकाई के मूलों का योग]\label{prop:b1:complex:sumroots}
$n \geq 2$ के लिए इकाई के $n$-वें मूलों का योग शून्य है: $\sum_{k=0}^{n-1}
\omega^k = 0$।
\end{proposition}

\begin{proof}
अनुपात $\omega \neq 1$ वाला गुणोत्तर योग: $\sum_{k=0}^{n-1} \omega^k =
\frac{\omega^n - 1}{\omega - 1} = 0$, क्योंकि $\omega^n = 1$।
\end{proof}

\begin{example}[वास्तविक और काल्पनिक भाग पढ़ना]\label{ex:b1:complex:sumrootsparts}
$\sum_{k=0}^{n-1}\omega^k = 0$ को वास्तविक और काल्पनिक भागों में तोड़ने से
दो त्रिकोणमितीय सर्वसमिकाएँ मुफ़्त में मिलती हैं:
\[
\sum_{k=0}^{n-1}\cos\frac{2k\pi}n = 0,
\qquad
\sum_{k=0}^{n-1}\sin\frac{2k\pi}n = 0
\qquad (n \geq 2).
\]
ज्यामितीय रूप से: इकाई वृत्त में अंतर्निहित सम $n$-भुज का केंद्रक उसका
केंद्र है --- शीर्ष ठीक-ठीक संतुलित हो जाते हैं। $n = 5$ के लिए पहली
सर्वसमिका $1 + 2\cos\frac{2\pi}5 + 2\cos\frac{4\pi}5 = 0$ देती है ($k$ को $n
- k$ के साथ युग्मित करके), जो $\cos\frac{2\pi}5$ की \cref{exo:b1:complex:8}
वाली संगणना का आरंभ-बिंदु है।
\end{example}

\begin{example}[बीजीय रूप में वर्गमूल]\label{ex:b1:complex:sqrt}
त्रिकोणमिति के बिना $z^2 = 3 + 4\iu$ हल करने के लिए $z = x + \iu y$ रखिए:
\[
x^2 - y^2 = 3, \qquad 2xy = 4, \qquad x^2 + y^2 = \abs{3 + 4\iu} = 5 .
\]
पहले और अंतिम को जोड़ने पर $x^2 = 4$, अतः $x = \pm 2$, फिर \emph{उसी}
चिह्न-युग्मन ($xy = 2 > 0$) के साथ $y = 2/x = \pm 1$: $z = \pm(2 + \iu)$।
सामान्य सूत्र के साथ मिलाकर यह सम्मिश्र गुणांकों वाला हर द्विघात समीकरण हल
कर देता है (\cref{exo:b1:complex:7})।
\end{example}

\section{सम्मिश्र संख्याएँ और समतल ज्यामिति}

\begin{proposition}[ज्यामितीय शब्दकोश]\label{prop:b1:complex:geometry}
समतल के बिंदु $M(x, y)$ की पहचान उसके \emph{सम्मिश्र निर्देशांक} $z = x +
\iu y$\index{सम्मिश्र निर्देशांक} से कीजिए। सम्मिश्र निर्देशांकों $a, b, c$
वाले भिन्न बिंदुओं $A, B, C$ के लिए:
\begin{enumerate}
  \item $\abs{b - a}$ दूरी $AB$ है;
  \item $\arg\dfrac{c - a}{b - a}$ सदिशों $\vect{AB}$ और $\vect{AC}$ के बीच
  का कोण है ($2\pi$ के सापेक्ष);
  \item $A, B, C$ संरेख हैं यदि और केवल यदि $\dfrac{c - a}{b - a} \in \R$;
  रेखाएँ $AB$ और $AC$ लंब हैं यदि और केवल यदि $\dfrac{c - a}{b - a} \in
  \iu\R$।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) $\vect{AB}$ के सम्मिश्र निर्देशांक $b - a$ पर \omterm{def:b1:complex:field}{मापांक} की परिभाषा ही है।
(2): $b - a = r\eu^{\iu\theta}$, $c - a = s\eu^{\iu\varphi}$ लिखिए; तब
$\frac{c-a}{b-a} = \frac sr \eu^{\iu(\varphi - \theta)}$ का कोणांक $\varphi
- \theta$ है, जो $\vect{AB}$ से $\vect{AC}$ तक का कोण है। (3): संरेखता का
अर्थ है कोण $0$ या $\pi$, अर्थात् कोणांक $\pi\Z$ में, अर्थात् भागफल
वास्तविक; लंबता का अर्थ है कोण $\pm\frac\pi2$, अर्थात् भागफल शुद्ध काल्पनिक।
(भागफल अशून्य है, क्योंकि $C \neq A$।)
\end{proof}

\begin{remark}[मध्यांतर: $\C$ गुणन से युक्त समतल है]\label{rem:b1:complex:interlude}
इस अध्याय को संभव बनाने वाली बात पर ठहरना उचित है: समतल $\R^2$ में हर दिशा
में योग हैं, पर कोई ईश्वरप्रदत्त गुणन नहीं --- और $\C$ वही समतल \emph{है},
एक गुणन से युक्त, जिसमें किसी नियत संख्या से गुणा करना घुमाता और मापता है।
इसी एक संरचना को इस खंड में तीन बार और दुहा जाएगा। \cref{ch:b1:matrices} में
$a + \iu b$ से गुणन पंक्तियों $(a, -b)$ और $(b, a)$ वाले $2 \times 2$ आव्यूह
के रूप में फिर प्रकट होता है: सम्मिश्र अंकगणित आव्यूह गुणनफलों का पहला,
पूर्णतः मूर्त परिवार है। \cref{ch:b1:euclid} में सूत्र $\Re(\conj z\,w)$
अदिश गुणन निकलता है और $\abs z$ यूक्लिडीय मानक: यहाँ सिद्ध त्रिभुज असमिका
वहाँ की कोशी--श्वार्ज़ कथा का आदर्श है। और \cref{ch:b1:curves} में गतिशील
बिंदु को $t \mapsto z(t)$ लिखना सबसे अच्छा रहता है, जिससे वेग और त्वरण
सम्मिश्र-मान वाले अवकलज बन जाते हैं --- उदाहरण के लिए वृत्तीय गति सीधे-सीधे
$z(t) = R\,\eu^{\iu\omega t}$ है। एक अच्छा गुणन, और चार अध्यायों का लाभांश।
\end{remark}

\begin{proposition}[प्रतिचित्रण $z \mapsto az + b$]\label{prop:b1:complex:similitude}
मान लीजिए $a \in \C^*$, $b \in \C$। समतल का रूपांतरण $f(z) = az + b$:
\begin{itemize}
  \item $a = 1$ होने पर स्थानांतरण है;
  \item अन्यथा उसका अद्वितीय अचल बिंदु $\zeta = \frac{b}{1-a}$ है, और $f(z)
  - \zeta = a\,(z - \zeta)$: अर्थात् $f$ केंद्र $\zeta$ और कोण $\arg a$ वाले
  घूर्णन का, केंद्र $\zeta$ और अनुपात $\abs a$ वाले अनुमापन के साथ संयोजन
  है।
\end{itemize}
विशेष रूप से $z \mapsto \eu^{\iu\theta} z$ मूलबिंदु के परितः कोण $\theta$ का
घूर्णन है, और $\conj z$ वास्तविक अक्ष में परावर्तन है।
\end{proposition}

\begin{proof}
यदि $a = 1$, तो $f(z) = z + b$ सम्मिश्र निर्देशांक $b$ वाले सदिश से
स्थानांतरण करता है। यदि $a \neq 1$, तो अचल-बिंदु समीकरण $z = az + b$ का
अद्वितीय हल $\zeta = \frac{b}{1-a}$ है, और तब $f(z) - \zeta = az + b -
(a\zeta + b) = a(z - \zeta)$। $a = \abs a\, \eu^{\iu\arg a}$ लिखने पर, $a$
से गुणा करना $\zeta$ तक की दूरियों को $\abs a$ से मापता है और $\zeta$ पर के
कोणों में $\arg a$ जोड़ देता है, जो घोषित संयोजन ही है।
\end{proof}

\begin{example}[प्रतिचित्रण $z \mapsto az+b$ का वर्गीकरण]\label{ex:b1:complex:classify}
$f(z) = \iu z + 1$ लीजिए। यहाँ $a = \iu \neq 1$: अचल बिंदु है
\[
\zeta = \frac{b}{1 - a} = \frac1{1 - \iu}
= \frac{1 + \iu}{2},
\]
और चूँकि $\abs a = 1$, कोई अनुमापन है ही नहीं: $f$ केंद्र $\frac{1+\iu}2$ और
कोण $\arg \iu = \frac\pi2$ का शुद्ध घूर्णन है। जाँच: $f(\zeta) =
\iu\,\frac{1+\iu}2 + 1 = \frac{\iu - 1}2 + 1 = \frac{1 + \iu}2 = \zeta$, तथा
$f(0) = 1$, $f(1) = 1 + \iu$, $f(1 + \iu) = \iu\,(1 + \iu) + 1 = \iu$: इकाई
वर्ग के चारों बिंदु $0, 1, 1+\iu, \iu$ अपने केंद्र $\zeta$ के चारों ओर, एक
बार में एक चौथाई घुमाव, चक्कर लगाते हैं --- ठीक वही जो वर्ग के केंद्र के
परितः $\frac\pi2$ का घूर्णन करेगा।
\end{example}

के शीर्ष $0 \to 1 \to 1{+}\iu \to \iu \to 0$ चक्कर लगाते हैं; कोई भी बिंदु
सरल रेखा पर नहीं चलता, फिर भी पूरा वर्ग दृढ़ता से घूम जाता है।}
\end{omfigure}

\begin{remark}[मापांक और कोणांक की सामान्य भूलें]\label{rem:b1:complex:pitfalls}
\begin{enumerate}
  \item \emph{$\C$ में कोई असमिका नहीं।} अवास्तविक संख्याओं के लिए $z \leq
  w$ लिखना निरर्थक है; केवल \omterm{def:b1:complex:field}{मापांक}, वास्तविक भाग और काल्पनिक भाग की तुलना की
  जा सकती है।
  \item \emph{$\sqrt{\phantom z}$ केवल $\R_+$ के लिए सुरक्षित है।} प्रत्येक
  अशून्य सम्मिश्र संख्या के \emph{दो} वर्गमूल होते हैं और किसी को वरीयता
  प्राप्त नहीं है: लिखिए ``मान लीजिए $\delta$ $\Delta$ का एक वर्गमूल है''
  (\cref{ex:b1:complex:sqrt} की भाँति संगणित), कभी $\sqrt\Delta$ नहीं ---
  नियम $\sqrt{ab} = \sqrt a\sqrt b$ तो $a = b = -1$ पर ही विफल हो जाता है।
  \item \emph{कोणांक $2\pi$ के सापेक्ष रहते हैं।} $\eu^{\iu\alpha} =
  \eu^{\iu\beta}$ से $\alpha \equiv \beta \pmod{2\pi}$ निकालिए, $\alpha =
  \beta$ नहीं; $2k\pi$ भूल जाना ही वह कारण है जिससे $z^n = a$ के \omterm{def:b1:logic:sets}{हल-समुच्चय}
  अपने $n$ अवयवों में से $n - 1$ खो देते हैं।
  \item \emph{$\abs{z + w}$ $\abs z + \abs w$ नहीं है।} त्रिभुज असमिका में
  समता संरेख होने वाली अपवादिक स्थिति है (\cref{prop:b1:complex:rules}~(4));
  सामान्यतः योग के \omterm{def:b1:complex:field}{मापांक} का आकलन करना पड़ता है, उसे संगणित नहीं किया जा
  सकता।
\end{enumerate}
\end{remark}

\begin{example}\label{ex:b1:complex:equilateral}
$A, B, C$ (सम्मिश्र निर्देशांक $a, b, c$, जोड़ों में भिन्न) उस समबाहु
त्रिभुज को बनाते हैं जिसके शीर्ष प्रत्यक्ष (वामावर्त) क्रम में हैं, यदि और
केवल यदि $\frac{c - a}{b - a} = \eu^{\iu\pi/3}$: केंद्र $A$ और कोण
$\frac\pi3$ का घूर्णन $B$ को $C$ पर भेजता है। दोनों दिशाएँ एक साथ सममित
समीकरण $a^2 + b^2 + c^2 = ab + bc + ca$ (\cref{exo:b1:complex:10}) से पकड़ी
जाती हैं।
\end{example}

\begin{remark}[यह अध्याय कहाँ काम आता है]\label{rem:b1:complex:whereused}
चरघातांकी रूप इस खंड की सबसे अधिक पुनःप्रयुक्त संगणना-युक्ति है। \omterm{def:b1:complex:unity}{इकाई के मूल}
\cref{ch:b1:structures} में चक्रीय समूह का मानक उदाहरण बन जाते हैं और
\cref{ch:b1:poly} में $X^n - 1$ का गुणनखंडन चलाते हैं; वहाँ \omterm{def:b1:complex:field}{संयुग्मी} मूलों
को समूहबद्ध करने से वे वास्तविक गुणनखंडन बनते हैं जिनका प्रयोग
\cref{ch:b1:fractions} में आंशिक भिन्नें करती हैं। रैखिकीकरण
(\cref{met:b1:complex:trig}) \cref{ch:b1:integration} में त्रिकोणमितीय घातों
के समाकलन की मानक तैयारी है, और \cref{ch:b1:diffeq} के अभिलक्षणिक समीकरणों
के सम्मिश्र मूलों के वास्तविक तथा काल्पनिक भाग दोलायमान हल $\eu^{\lambda
t}\cos\omega t$ उत्पन्न करते हैं। ज्यामितीय शब्दकोश आव्यूह के वेश में लौटता
है: घूर्णन और समरूपताएँ \cref{ch:b1:matrices,ch:b1:euclid} के लांबिक आव्यूह
बन जाते हैं, और \cref{ch:b1:curves} के प्राचलित वक्र प्रायः \omterm{def:b1:logic:map}{प्रतिचित्रण} $t
\mapsto z(t) \in \C$ के रूप में लिखना सबसे अच्छा रहता है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:complex:1}
चरघातांकी रूप में लिखिए: $1 + \iu$; $\;\sqrt 3 - \iu$; $\;-5$; $\;\dfrac{1 +
\iu}{\sqrt 3 - \iu}$। उससे $\cos\frac{5\pi}{12}$ और $\sin\frac{5\pi}{12}$
निकालिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:2}
$(1 + \iu)^{20}$ की संगणना कीजिए। किन $n \in \N$ के लिए $(1 + \iu)^n$
वास्तविक संख्या है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:3}
ऐसे $z \in \C$ के \omterm{def:b1:logic:sets}{समुच्चय} का ज्यामितीय वर्णन कीजिए कि: $\abs{z - 2} = \abs{z
+ \iu}$; $\;\abs{z - 1} = 2$; $\;\dfrac{z - 1}{z + 1} \in \iu\R$ ($z \neq
-1$ के लिए)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:4}
$\sin^4\theta$ का रैखिकीकरण कीजिए, और $\cos 4\theta$ का $\cos\theta$ में
बहुपद के रूप में प्रसार कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:5}
$z^3 = 8\iu$ हल कीजिए और हलों को एक चित्र पर रखिए। $z^4 = -4$ हल कीजिए और
$X^4 + 4$ का दो वास्तविक द्विघात बहुपदों में गुणनखंडन कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:6}
$\eu^{\iu\theta} \neq 1$ और $n \in \N$ वाले $\theta \in \R$ के लिए
\[
C_n = \sum_{k=0}^{n} \cos k\theta
\qquad\text{और}\qquad
S_n = \sum_{k=0}^{n} \sin k\theta
\]
की संगणना, गुणोत्तर श्रेणी $\sum_k \eu^{\iu k\theta}$ का योग करके और
अर्ध-कोण गुणनखंडन का प्रयोग करके, कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:7}
$\C$ में हल कीजिए: $z^2 - (3 + 4\iu) z + (-1 + 5\iu) = 0$। \emph{(विविक्तकर
संगणित कीजिए और \cref{ex:b1:complex:sqrt} की भाँति उसके वर्गमूल निकालिए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:8}
मान लीजिए $\omega = \eu^{2\iu\pi/5}$।
\begin{enumerate}
  \item $1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$ को न्यायसंगत
  ठहराइए।
  \item $u = \omega + \omega^4$ और $v = \omega^2 + \omega^3$ रखिए। $u + v$
  और $uv$ संगणित कीजिए, और उससे निकालिए कि $u$ और $v$ $X^2 + X - 1$ के मूल
  हैं।
  \item निष्कर्ष निकालिए कि $\cos\frac{2\pi}{5} = \frac{\sqrt 5 - 1}{4}$।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:9}
सिद्ध कीजिए कि सभी $z, w \in \C$ के लिए (समांतर चतुर्भुज सर्वसमिका):
\[
\abs{z + w}^2 + \abs{z - w}^2 = 2\abs z^2 + 2\abs w^2 ,
\]
और शीर्षों $0, z, w, z + w$ वाले समांतर चतुर्भुज में इसकी ज्यामितीय व्याख्या
कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:complex:10}
सिद्ध कीजिए कि सम्मिश्र निर्देशांकों $a, b, c$ वाले जोड़ों में भिन्न तीन
बिंदु समबाहु त्रिभुज (किसी भी दिशा में) बनाते हैं यदि और केवल यदि
\[
a^2 + b^2 + c^2 = ab + bc + ca .
\]
\emph{संकेत: प्रत्यक्ष स्थिति $\frac{c-a}{b-a} = -j^2$ है और अप्रत्यक्ष
स्थिति $\frac{c-a}{b-a} = -j$, जहाँ $j = \eu^{2\iu\pi/3}$ $j^2 + j + 1 = 0$
को संतुष्ट करता है; अथवा $a + jb + j^2c$ और $a + j^2 b + jc$ का गुणनखंडन
कीजिए।}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:complex:11}
$n \in \N^*$ के लिए $P = \prod_{k=1}^{n-1} \bigl(1 - \omega^k\bigr)$ का मान
निकालिए, जहाँ $\omega = \eu^{2\iu\pi/n}$। \emph{संकेत: $X^n - 1 =
\prod_{k=0}^{n-1} (X - \omega^k)$; $X - 1$ से भाग दीजिए और $X = 1$ पर मान
लीजिए।} $\prod_{k=1}^{n-1} \sin\frac{k\pi}{n} = \dfrac{n}{2^{n-1}}$ निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:12}
मान लीजिए $n \geq 2$। $\C$ में समीकरण $(z + 1)^n = (z - 1)^n$ हल कीजिए:
दिखाइए कि उसके ठीक $n - 1$ हल हैं, जो सभी शुद्ध काल्पनिक हैं, अर्थात्
\[
z_k = -\iu\,\frac{\cos\frac{k\pi}{n}}{\sin\frac{k\pi}{n}},
\qquad k = 1, \dots, n - 1 .
\]
\emph{संकेत: $z = 1$ हल नहीं है, अतः भाग दीजिए और \omterm{def:b1:complex:unity}{इकाई के मूल} प्रयोग कीजिए;
फिर \cref{met:b1:complex:trig} का अर्ध-कोण गुणनखंडन लगाइए।}
\end{exercise}

\section{समस्या: नेपोलियन की प्रमेय}

\begin{problem}\label{pb:b1:complex:1}
किसी \emph{स्वेच्छ} त्रिभुज की प्रत्येक भुजा पर बाहर की ओर एक समबाहु त्रिभुज
खड़ा कीजिए: उन त्रिभुजों के तीनों केंद्र सदा एक समबाहु त्रिभुज बनाते हैं। यह
नेपोलियन की प्रमेय\index{नेपोलियन की प्रमेय} है --- ऐसा \omterm{def:b1:logic:statement}{कथन} जिसके सत्य होने
का कोई दृश्य कारण नहीं है, और जिसे $j = \eu^{2\iu\pi/3}$ का बीजगणित तीन
पंक्तियों की संगणना में सिद्ध कर देता है। यह समस्या पूरा औज़ार-संदूक
(घूर्णन, प्रत्यक्ष समरूपताएँ, \cref{exo:b1:complex:10} की $j$-कसौटी) खड़ा
करती है, नेपोलियन की प्रमेय भीतरी और बाहरी दोनों रूपों में सिद्ध करती है,
अपकर्षी स्थिति ढूँढ़ती है, और उसी घराने के एक दूसरे रत्न पर समाप्त होती है:
समबाहु त्रिभुजों पर वान स्कूटन की प्रमेय और टॉलेमी असमिका। आगे सर्वत्र समतल
के बिंदुओं की पहचान उनके सम्मिश्र निर्देशांकों से की गई है।

\medskip \noindent\textbf{भाग I --- संख्या $j$ और घूर्णन।}
\begin{enumerate}
  \item $j$ का बीजीय रूप संगणित कीजिए, फिर $j^2$, $j^3$, $1 + j + j^2$,
  $\conj j$ और $j^{-1}$। $1$, $j$, $j^2$ को इकाई वृत्त के एक रेखाचित्र पर
  रखिए।
  \item दिखाइए कि केंद्र $a$ और कोण $\theta$ का घूर्णन $r(z) = a +
  \eu^{\iu\theta}(z - a)$ है, और यह कि कोणों $\theta$ तथा $\theta'$ वाले दो
  घूर्णनों का संयोजन $\theta + \theta' \notin 2\pi\Z$ होने पर कोण $\theta +
  \theta'$ का घूर्णन है, और अन्यथा स्थानांतरण।
  (\cref{prop:b1:complex:similitude} का प्रयोग कीजिए।)
  \item मान लीजिए $b \neq c$। दिखाइए कि ठीक दो बिंदु $p$ ऐसे हैं जो $(b, c,
  p)$ को समबाहु बनाते हैं, अर्थात् $p = b + \eu^{\pm\iu\pi/3}(c - b)$। किसी
  \emph{प्रत्यक्ष} त्रिभुज $(a, b, c)$ के लिए उदाहरण $a = 0$, $b = 1$, $c =
  \iu$ पर जाँचिए कि चयन $\eu^{-\iu\pi/3}$ वही है जो रेखा $BC$ के $a$ से परली
  ओर स्थित है --- अर्थात् \emph{बाहरी} शीर्ष।
  \item \cref{exo:b1:complex:10} के हल से $(a, b, c)$ प्रत्यक्ष समबाहु है
  यदि और केवल यदि $a + jb + j^2c = 0$। दोनों घूर्णन सर्वसमिकाएँ सिद्ध कीजिए
        \[
        b + jc + j^2a = j^2\,(a + jb + j^2c),
        \qquad
        c + ja + j^2b = j\,(a + jb + j^2c),
        \]
        और उससे निकालिए कि यह कसौटी $(a, b, c)$ के चक्रीय \omterm{def:b1:counting:objects}{क्रमचय} के अंतर्गत
        अपरिवर्ती है।
  \item दिखाइए कि यदि $a + jb + j^2c = 0$ हो और तीनों बिंदुओं में से दो
  संपाती हों, तो तीनों संपाती हैं। (अतः कसौटी ठीक-ठीक यही अभिलक्षित करती है:
  प्रत्यक्ष समबाहु त्रिभुज, या एक ही बिंदु।)
\end{enumerate}

\medskip \noindent\textbf{भाग II --- प्रत्यक्ष समरूपताएँ।}
\begin{enumerate}[resume]
  \item दिखाइए कि $a \in \C^*$ वाले \omterm{def:b1:logic:map}{प्रतिचित्रण} $f(z) = az + b$
  (\emph{प्रत्यक्ष समरूपताएँ}) संयोजन और प्रतिलोमन के अंतर्गत स्थायी हैं:
  उनमें से दो का संयोजन, और किसी का भी प्रतिलोम, फिर इसी रूप का है।
  (\cref{ch:b1:structures} की भाषा में: वे एक समूह बनाती हैं।)
  \item $z_1 \neq z_2$ और $w_1 \neq w_2$ दिए हों, तो दिखाइए कि $f(z_1) =
  w_1$ और $f(z_2) = w_2$ वाली \emph{अद्वितीय} प्रत्यक्ष समरूपता $f$ है, और
  $a$ तथा $b$ स्पष्ट रूप से दीजिए।
  \item दिखाइए कि प्रत्यक्ष समरूपता $f(z) = az + b$ सभी दूरियों को $\abs a$
  से गुणा करती है और किसी भी त्रिभुज $(a', b', c')$ की \emph{आकृति} $\frac{c
  - a'}{b' - a'}$ सुरक्षित रखती है \emph{(अंश और हर दोनों $a$ से गुणित होते
  हैं)}। उससे निकालिए कि दो त्रिभुज ठीक तब प्रत्यक्ष रूप से समरूप हैं जब
  उनकी आकृतियाँ बराबर हों।
  \item $f(0) = 1$ और $f(1) = \iu$ वाली प्रत्यक्ष समरूपता पूर्णतः निर्धारित
  कीजिए: $a$, $b$, अचल बिंदु, अनुपात और कोण दीजिए।
  \item मान लीजिए $\alpha + \beta = 1$। दिखाइए कि ``भारित बिंदु'' $g(a', b')
  = \alpha a' + \beta b'$ प्रत्येक प्रत्यक्ष समरूपता के साथ क्रम-विनिमेय है:
  $f(\alpha a' + \beta b') = \alpha f(a') + \beta f(b')$। उससे निकालिए कि
  केंद्रक, मध्यबिंदु और नीचे आने वाले नेपोलियन केंद्र सब समरूपताओं द्वारा ले
  जाए जाते हैं --- यही वह बीजीय अनुमति है जो हर ``व्यापकता खोए बिना,
  परिवृत्त को इकाई वृत्त पर रख लीजिए'' वाले तर्क के पीछे है।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- नेपोलियन की प्रमेय।} मान लीजिए $(a, b,
c)$ एक प्रत्यक्ष त्रिभुज है। प्रत्येक भुजा पर बाहर की ओर समबाहु त्रिभुज खड़ा
कीजिए (प्रश्न 3) और मान लीजिए $n_a$, $n_b$, $n_c$ क्रमशः $[b, c]$, $[c, a]$,
$[a, b]$ पर खड़े त्रिभुजों के केंद्र (केंद्रक) हैं।
\begin{enumerate}[resume]
  \item दिखाइए कि
        \[
        n_a = \alpha b + \beta c, \qquad
        n_b = \alpha c + \beta a, \qquad
        n_c = \alpha a + \beta b,
        \qquad\text{जहाँ}\quad
        \alpha = \frac{3 + \iu\sqrt3}{6},\ \beta = \conj\alpha .
        \]
  \item $\alpha + \beta = 1$ जाँचिए, और उससे निकालिए कि त्रिभुज $(n_a, n_b,
  n_c)$ का केंद्रक $(a, b, c)$ के केंद्रक के \emph{समान} है।
  \item प्रश्न 4 की सर्वसमिकाओं का प्रयोग करते हुए दिखाइए कि
        \[
        n_a + j\,n_b + j^2 n_c
        = (\alpha j^2 + \beta j)\,(a + jb + j^2c) .
        \]
  \item $\alpha j + \beta$ संगणित कीजिए और निष्कर्ष निकालिए: $\alpha j^2 +
  \beta j = j(\alpha j + \beta) = 0$, अतः \emph{प्रत्येक} त्रिभुज के लिए
  $n_a + j n_b + j^2 n_c = 0$: बाहरी नेपोलियन त्रिभुज प्रत्यक्ष समबाहु है
  (अथवा एक बिंदु)। नेपोलियन की प्रमेय सिद्ध हो गई।
  \item इसके बदले समबाहु त्रिभुज \emph{भीतर की ओर} खड़े कीजिए (प्रश्न 3 में
  चयन $\eu^{+\iu\pi/3}$) और मान लीजिए $m_a, m_b, m_c$ उनके केंद्र हैं।
  दिखाइए $m_a = \beta b + \alpha c$ (और चक्रीय रूप से), फिर सिद्ध कीजिए $m_a
  + j^2 m_b + j\,m_c = 0$: भीतरी नेपोलियन त्रिभुज भी समबाहु है, पर विपरीत
  दिशा वाला।
  \item अपकर्ष ढूँढ़िए: दिखाइए कि $n_a = n_b = n_c$ ठीक तब होता है जब $a +
  j^2 b + jc = 0$, अर्थात् जब $(a, b, c)$ एक \emph{अप्रत्यक्ष} समबाहु
  त्रिभुज हो। \emph{(प्रश्न 5 से $n_a, n_b, n_c$ में दो बिंदु तभी संपाती हैं
  जब तीनों हों; फिर $n_a + j^2 n_b + j n_c = j(\alpha + \beta j)(a + j^2b +
  jc)$ का प्रयोग कीजिए और $\alpha + \beta j \neq 0$ जाँचिए।)}
  \item पूरी संगणना त्रिभुज $a = 0$, $b = 1$, $c = \iu$ पर कीजिए: $n_a, n_b,
  n_c$ यथार्थ रूप से दीजिए, और तीनों भुजाओं की वर्ग-लंबाइयों की सीधी संगणना
  से सत्यापित कीजिए कि त्रिभुज समबाहु है, जिसकी वर्ग-भुजा $\frac{2 +
  \sqrt3}{3}$ है।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- टॉलेमी और वान स्कूटन।}
\begin{enumerate}[resume]
  \item सभी सम्मिश्र $a, b, c, d$ के लिए वैध यह सर्वसमिका सिद्ध कीजिए:
        \[
        (a - b)(c - d) + (a - d)(b - c) = (a - c)(b - d) .
        \]
  \item उससे \emph{टॉलेमी असमिका}\index{टॉलेमी असमिका} निकालिए: किन्हीं चार
  बिंदुओं $A, B, C, D$ के लिए
        \[
        AC \cdot BD \;\leq\; AB \cdot CD + AD \cdot BC ,
        \]
        जहाँ समता यदि और केवल यदि $(a-b)(c-d)$ और $(a-d)(b-c)$ $0$ से निकलने
        वाली एक ही किरण पर हों।
  \item दिखाइए कि $\theta, \varphi \in \R$ के लिए $\abs{\eu^{\iu\theta} -
  \eu^{\iu\varphi}} = 2\,\abs{\sin\frac{\theta - \varphi}2}$।
  \item (वान स्कूटन की प्रमेय\index{वान स्कूटन की प्रमेय}) मान लीजिए $(a, b,
  c) = (1, j, j^2)$ --- भाग II से प्रत्यक्ष समबाहु त्रिभुजों में यह व्यापकता
  नहीं खोता --- और मान लीजिए $\theta \in \intoo{2\pi/3}{4\pi/3}$ के साथ $p =
  \eu^{\iu\theta}$, जो परिवृत्त के उस चाप $BC$ का बिंदु है जिसमें $A$ नहीं
  है। सिद्ध कीजिए
        \[
        PA = PB + PC .
        \]
        \emph{(तीनों दूरियाँ प्रश्न 20 से व्यक्त कीजिए और योग-से-गुणनफल
        सूत्र का प्रयोग कीजिए।)}
  \item $p = -1$ पर सत्यापित कीजिए कि यह चक्रीय क्रम $A, B, P, C$ के लिए
  टॉलेमी असमिका की समता-स्थिति ही है: $(a - b)(p - c)$ और $(a - c)(b - p)$
  संगणित कीजिए और जाँचिए कि उनका अनुपात धनात्मक वास्तविक है।
\end{enumerate}

\medskip \noindent\textbf{भाग V --- संश्लेषण।}
\begin{enumerate}[resume]
  \item समबाहु त्रिभुज $(1, j, j^2)$ का अपना बाहरी नेपोलियन त्रिभुज संगणित
  कीजिए, और परिणाम का ज्यामितीय वर्णन कीजिए।
  \item इस समस्या में ठीक कहाँ प्रयोग हुआ: (क) गुणन का घूर्णन के रूप में
  होना; (ख) समरूपताओं की समूह संरचना और आकृति-अपरिवर्त्य; (ग)
  \cref{met:b1:complex:trig} का अर्ध-कोण गुणनखंडन? प्रत्येक के लिए एक वाक्य।
  \item एक छोटे अनुच्छेद में इस समस्या का उपदेश स्पष्ट कीजिए: समतल ज्यामिति
  और $\C$ के बीजगणित के बीच का शब्दकोश क्या देता है, क्या लेता है, और आपकी
  दृष्टि में दोनों उपपत्ति-चरणों में से कौन --- सर्वसमिका $\alpha j + \beta
  = 0$ या चिरपरिचित चित्र --- नेपोलियन की प्रमेय को बेहतर \emph{समझाता} है?
  एक ऐसा स्थान बताइए जहाँ यह शब्दकोश इस खंड में आगे आव्यूह के रूप में फिर
  प्रकट होगा।
\end{enumerate}
\end{problem}
