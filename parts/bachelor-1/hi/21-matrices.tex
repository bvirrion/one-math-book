\chapter{आव्यूह}\label{ch:b1:matrices}

आव्यूह निर्देशांकों में लिखा गया \omterm{def:b1:linmaps:def}{रैखिक प्रतिचित्रण} है। यह अध्याय वही शब्दकोश
खड़ा करता है --- संयुक्त \omterm{def:b1:logic:map}{प्रतिचित्रण} आव्यूह-गुणनफल बन जाता है, \omterm{def:b1:logic:inj}{एकैकी}
आच्छादकता व्युत्क्रमणीयता, और \omterm{def:b1:matrices:changeofbasis}{आधार-परिवर्तन} संयुग्मन --- तथा कलनविधि वाला
पक्ष भी: \omterm{met:b1:matrices:gauss}{पंक्ति संक्रियाएँ}, कोटियों और प्रतिलोमों की संगणना। आव्यूह माध्यमिक
खंड में पहली बार मिले थे, और अब वे
\cref{ch:b1:vspaces,ch:b1:findim,ch:b1:linmaps} के सिद्धांत पर टिका दिए गए
हैं।

\section{आव्यूह और रैखिक प्रतिचित्रण}

\begin{definition}\label{def:b1:matrices:def}
$\mathcal{M}_{n,p}(K)$ अदिशों की $n \times p$ सारणियों $A = (a_{ij})$ की
\omterm{def:b1:vspaces:def}{सदिश समष्टि} है ($i$: पंक्ति, $j$: स्तंभ), जिसकी विमा $np$ है (\omterm{def:b1:vspaces:free}{आधार}: केवल एक
$1$ वाले आव्यूह $E_{ij}$)। $E$ का \omterm{def:b1:vspaces:free}{आधार} $\mathcal{B} = (e_1, \dots, e_p)$ और
$F$ का \omterm{def:b1:vspaces:free}{आधार} $\mathcal{C}$ दिए हों ($\dim F = n$), तो \emph{$u \in
\mathcal{L}(E, F)$ का आव्यूह} वह सारणी है जिसका $j$-वाँ स्तंभ $\mathcal{C}$
में $u(e_j)$ के \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} देता है:
\[
\operatorname{Mat}_{\mathcal{B},\mathcal{C}}(u) = (a_{ij}),
\qquad u(e_j) = \sum_{i=1}^{n} a_{ij}\, f_i .
\]
\omterm{def:b1:logic:map}{प्रतिचित्रण} $u \mapsto \operatorname{Mat}_{\mathcal{B},\mathcal{C}}(u)$
$\mathcal{L}(E, F)$ से $\mathcal{M}_{n,p}(K)$ पर एक तुल्याकारिता है
(\cref{prop:b1:linmaps:basis}: \omterm{def:b1:linmaps:def}{रैखिक प्रतिचित्रण} ठीक $e_j$ के प्रतिबिंबों का
चुनाव ही है)।
\end{definition}

\begin{example}[अवकलज, आव्यूह के रूप में]
\label{ex:b1:matrices:derivative}
$\R_3[X]$ पर मान लीजिए $D(P) = P'$। एकपदी \omterm{def:b1:vspaces:free}{आधार} $(1, X, X^2, X^3)$ में: $D(1)
= 0$, $D(X) = 1$, $D(X^2) = 2X$, $D(X^3) = 3X^2$, अतः
\[
\operatorname{Mat}(D) =
\begin{pmatrix}
0 & 1 & 0 & 0\\
0 & 0 & 2 & 0\\
0 & 0 & 0 & 3\\
0 & 0 & 0 & 0
\end{pmatrix}.
\]
\emph{विभाजित} \omterm{def:b1:vspaces:free}{आधार} $\bigl(1,\ X,\ \frac{X^2}2,\ \frac{X^3}6\bigr)$ में हर
आधार-सदिश पिछले पर चला जाता है ($D\bigl(\frac{X^k}{k!}\bigr) =
\frac{X^{k-1}}{(k-1)!}$), और आव्यूह शुद्ध सरकाव बन जाता है: अधिविकर्ण पर
इकाइयाँ, बाक़ी जगह शून्य। दो सीखें: आव्यूह (\omterm{def:b1:logic:map}{प्रतिचित्रण}, \omterm{def:b1:vspaces:free}{आधार}) की
\emph{जोड़ी} का होता है, अकेले \omterm{def:b1:logic:map}{प्रतिचित्रण} का नहीं; और अच्छा \omterm{def:b1:vspaces:free}{आधार} संरचना को
एक नज़र में दिखा देता है --- सरकाव-रूप तुरंत दिखा देता है कि $\R_3[X]$ पर
$D^4 = 0$, क्योंकि आव्यूह की हर घात अपने इकाइयों वाले विकर्ण को एक क़दम और
बाहर धकेल देती है।
\end{example}

\begin{definition}[गुणनफल]\label{def:b1:matrices:product}
$A \in \mathcal{M}_{n,p}$ और $B \in \mathcal{M}_{p,q}$ के लिए:
\[
(AB)_{ik} = \sum_{j=1}^{p} a_{ij}\, b_{jk}
\qquad (1 \leq i \leq n,\ 1 \leq k \leq q).
\]
यह ठीक संयुक्त \omterm{def:b1:logic:map}{प्रतिचित्रण} का आव्यूह है: $\operatorname{Mat}(v \circ u) =
\operatorname{Mat}(v)\, \operatorname{Mat}(u)$ (बीच में \omterm{def:b1:vspaces:free}{आधार} मेल खाते हुए)।
इसी प्रकार, यदि $X$ $x$ के निर्देशांकों का स्तंभ है, तो $u(x)$ का स्तंभ $AX$
है।
\end{definition}

\begin{proof}[संयोजन-सूत्र की उपपत्ति]
\[
v(u(e_k)) = v\Bigl(\sum_j b_{jk} f_j\Bigr) = \sum_j b_{jk}\, v(f_j)
= \sum_j b_{jk} \sum_i a_{ij}\, g_i
= \sum_i \Bigl(\sum_j a_{ij} b_{jk}\Bigr) g_i . \qedhere
\]
\end{proof}

\begin{proposition}[बीजगणित $\mathcal{M}_n(K)$]\label{prop:b1:matrices:ring}
वर्ग आव्यूह $\mathcal{M}_n(K)$ एक \omterm{def:b1:structures:ring}{वलय} बनाते हैं ($n \geq 2$ के लिए
अक्रमविनिमेय), जिसका तत्समक $I_n$ है; उसका इकाइयों का \omterm{def:b1:structures:group}{समूह} \emph{व्यापक
रैखिक समूह}\index{व्यापक रैखिक समूह} $GL_n(K)$ है, जो \omterm{def:b1:logic:inj}{एकैकी आच्छादक}
अंतःसमाकारिताओं के अनुरूप है। $A, B \in \mathcal{M}_n(K)$ के लिए:
\[
AB = I_n \implies A \in GL_n(K) \text{ और } B = A^{-1}
\]
(एकतरफ़ा प्रतिलोम दुतरफ़ा होते हैं, \cref{cor:b1:linmaps:samedim} से)।
\end{proposition}

\begin{proof}
\omterm{def:b1:structures:ring}{वलय} के अभिगृहीत \cref{def:b1:matrices:def} की तुल्याकारिता के द्वारा
$\mathcal{L}(E)$ से स्थानांतरित हो जाते हैं: वह संयुक्त \omterm{def:b1:logic:map}{प्रतिचित्रण} को
गुणनफल में और योग को योग में बदल देती है, अतः साहचर्य, वितरण नियम और $I_n$
की भूमिका प्रतिचित्रणों के तदनुरूप तथ्यों से विरासत में मिल जाते हैं, बिना
किसी प्रविष्टि-दर-प्रविष्टि सत्यापन के। अक्रमविनिमयता: $E_{12}E_{21} =
E_{11} \neq E_{22} = E_{21}E_{12}$। यदि $AB = I_n$: तो $A$ की अंतःसमाकारिता
$a$ $a \circ b = \mathrm{id}$ संतुष्ट करती है, अतः $a$ \omterm{def:b1:logic:inj}{आच्छादक} है ($x =
a(b(x))$ हर $x$ का एक \omterm{def:b1:logic:map}{पूर्वप्रतिबिंब} दिखा देता है), इसलिए \omterm{def:b1:findim:def}{परिमित विमा} में
\omterm{def:b1:logic:inj}{एकैकी आच्छादक} (\cref{cor:b1:linmaps:samedim}); $a \circ b = \mathrm{id}$ को
बाईं ओर से $a^{-1}$ के साथ संयुक्त करने पर $b = a^{-1}$ मिलता है, और फिर
$b\circ a = \mathrm{id}$ भी: अर्थात् एकतरफ़ा प्रतिलोम आरंभ से ही दुतरफ़ा था
--- और यह कृपा कड़ाई से \omterm{def:b1:findim:def}{परिमित-विमीय} है।
\end{proof}

\begin{definition}[परिवर्त; अनुरेख]\label{def:b1:matrices:transpose}
$A = (a_{ij}) \in \mathcal{M}_{n,p}$ का \emph{परिवर्त} $A^{\mathsf T} =
(a_{ji}) \in \mathcal{M}_{p,n}$ है; वह $(AB)^{\mathsf T} = B^{\mathsf T}
A^{\mathsf T}$ और $(A^{\mathsf T})^{\mathsf T} = A$ संतुष्ट करता है। किसी
वर्ग आव्यूह का \emph{अनुरेख}\index{अनुरेख} $\operatorname{tr} A = \sum_i
a_{ii}$ है; वह \omterm{def:b1:linmaps:def}{रैखिक} है, और
\[
\operatorname{tr}(AB) = \operatorname{tr}(BA)
\qquad (A \in \mathcal{M}_{n,p},\ B \in \mathcal{M}_{p,n}).
\]
\end{definition}

\begin{proof}[अनुरेख-सर्वसमिका की उपपत्ति]
$\operatorname{tr}(AB) = \sum_i \sum_j a_{ij} b_{ji}$ और
$\operatorname{tr}(BA) = \sum_j \sum_i b_{ji} a_{ij}$: वही दुहरा योग।
\end{proof}

\begin{example}[अनुरेख काम पर]
\label{ex:b1:matrices:tracework}
$\operatorname{Vect}(0,1)$ के अनुदिश $\operatorname{Vect}(1,1)$ पर
\cref{ch:b1:linmaps} का \omterm{def:b1:linmaps:projection}{प्रक्षेप}, अर्थात् $p(x, y) = (x, x)$, विहित \omterm{def:b1:vspaces:free}{आधार} में
आव्यूह $A = \begin{pmatrix} 1 & 0\\ 1 & 0\end{pmatrix}$ रखता है: वस्तुतः
$A^2 = A$, और
\[
\operatorname{tr} A = 1 = \operatorname{rk} A ,
\]
जो \cref{exo:b1:matrices:8} को दर्शाता है: वर्गसम आव्यूहों के लिए \omterm{def:b1:matrices:transpose}{अनुरेख}
\omterm{def:b1:linmaps:kerim}{प्रतिबिंब} की विमा \emph{गिनता} है, चाहे आव्यूह किसी भी तिरछे \omterm{def:b1:vspaces:free}{आधार} में लिखा
गया हो। अपरिवर्तिता की क्रियाविधि सर्वसमिका $\operatorname{tr}(AB) =
\operatorname{tr}(BA)$ है:
\[
\operatorname{tr}\bigl(P^{-1}(AP)\bigr) =
\operatorname{tr}\bigl((AP)P^{-1}\bigr) = \operatorname{tr} A ,
\]
अतः $A$ के सदृश सभी आव्यूहों का \omterm{def:b1:matrices:transpose}{अनुरेख} वही होता है --- यह किसी अंतःसमाकारिता
का पहला \emph{संख्यात्मक अचर} है, जिसके साथ \cref{ch:b1:det} में सारणिक आ
मिलेगा (नीचे की सप्ताहांत समस्या वाली जोड़ी $(s, p)$)।
\end{example}

\begin{example}[सममित जमा प्रतिसममित]
\label{ex:b1:matrices:symsplit}
$A$ को \emph{सममित} कहिए जब $A^{\mathsf T} = A$, और \emph{प्रतिसममित} जब
$A^{\mathsf T} = -A$। हर वर्ग आव्यूह अद्वितीय रूप से एक जमा दूसरे के रूप में
बँट जाता है:
\[
A = \underbrace{\frac{A + A^{\mathsf T}}{2}}_{\text{सममित}}
+ \underbrace{\frac{A - A^{\mathsf T}}{2}}_{\text{प्रतिसममित}},
\]
और जो आव्यूह दोनों हो वह शून्य है ($A = -A$): अतः दोनों \omterm{def:b1:logic:sets}{समुच्चय}
$\mathcal{M}_n(K)$ की \omterm{def:b1:vspaces:sum}{पूरक उपसमष्टियाँ} हैं --- यह फलनों के सम/विषम \omterm{thm:b1:logic:partition}{विभाजन}
(\cref{ex:b1:vspaces:evenodd}) का ठीक-ठीक तदनुरूप है, जहाँ $x \mapsto -x$ की
भूमिका परिवर्तन निभाता है। विमाएँ: सममित आव्यूह विकर्ण पर और उसके ऊपर
\omterm{def:b1:vspaces:free}{स्वतंत्र} होता है, और प्रतिसममित कड़ाई से ऊपर (विकर्ण शून्य):
\[
\frac{n(n+1)}{2} + \frac{n(n-1)}{2} = n^2 ,
\]
और गिनती का संतुलित हो जाना ग्रासमान द्वारा प्रत्यक्षता की पुष्टि है। $n =
2$ के लिए: $\begin{pmatrix} 1 & 5\\ 1 & 2\end{pmatrix} = \begin{pmatrix} 1 &
3\\ 3 & 2\end{pmatrix} + \begin{pmatrix} 0 & 2\\ -2 & 0\end{pmatrix}$। सममित
आव्यूह \cref{ch:b1:multivar} में द्वितीय \omterm{def:b1:derivative:def}{अवकलज} के आँकड़ों के रूप में लौटते
हैं (मोंज त्रिक $r, s, t$), और सममित-लांबिक आव्यूहों का वर्गीकरण
\cref{exo:b1:euclid:12} में है।
\end{example}

\section{आधार-परिवर्तन}

\begin{definition}\label{def:b1:matrices:changeofbasis}
मान लीजिए $\mathcal{B}, \mathcal{B}'$ $E$ के \omterm{def:b1:vspaces:free}{आधार} हैं। \emph{आधार-परिवर्तन
आव्यूह}\index{आधार-परिवर्तन} $P = P_{\mathcal{B}\to\mathcal{B}'}$ के स्तंभ
\emph{नए} आधार-सदिशों के \emph{पुराने} \omterm{def:b1:vspaces:free}{आधार} में \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} हैं। वह
व्युत्क्रमणीय है, $P^{-1} = P_{\mathcal{B}'\to\mathcal{B}}$, और \omterm{prop:b1:vspaces:coordinates}{निर्देशांक}
$X = PX'$ के अनुसार बदलते हैं (पुराना $=$ $P\,\cdot$ नया)।
\end{definition}

\begin{example}[आधार-परिवर्तन आव्यूह पढ़ना]
\label{ex:b1:matrices:changerun}
$\R^2$ में, विहित $\mathcal B$ से $\mathcal B' = \bigl((1,1), (1,-1)\bigr)$
तक:
\[
P = P_{\mathcal B\to\mathcal B'} =
\begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix}
\]
(नए सदिश पुराने निर्देशांकों में, स्तंभ-दर-स्तंभ लिखे हुए)। पुराने
निर्देशांकों वाले सदिश $X = (3, 1)^{\mathsf T}$ के नए \omterm{prop:b1:vspaces:coordinates}{निर्देशांक} $X' =
P^{-1}X = \frac12(3 + 1,\ 3 - 1)^{\mathsf T} = (2, 1)^{\mathsf T}$ हैं:
वस्तुतः $2(1,1) + 1(1,-1) = (3,1)$। दिशा का ध्यान रखिए --- आव्यूह $P$
\emph{नए} \omterm{def:b1:vspaces:free}{आधार} से बनता है, पर वह निर्देशांकों को \emph{नए से पुराने} में
बदलता है ($X = PX'$); पुराने से नए जाने की क़ीमत प्रतिलोम है। हर रूपांतरण के
बाद जाँच $2(1,1) + (1,-1) = (3,1)$ लिख लेने से उलटे-$P$ वाली त्रुटि पकड़ में
आ जाती है, जो इस अध्याय की सबसे आम भूल है।
\end{example}

\begin{theorem}[किसी प्रतिचित्रण के लिए आधार-परिवर्तन]\label{thm:b1:matrices:conjugation}
मान लीजिए $u \in \mathcal{L}(E)$, जिसका $\mathcal{B}$ में आव्यूह $A$ है और
$\mathcal{B}'$ में $A'$, तथा $P = P_{\mathcal{B}\to\mathcal{B}'}$। तब
\[
A' = P^{-1} A\, P .
\]
इस प्रकार संबंधित दो आव्यूह \emph{सदृश}\index{सदृश आव्यूह} कहलाते हैं। (दो
जोड़ी आधारों वाले $u \colon E \to F$ के लिए सूत्र $A' = Q^{-1} A P$ है ---
\emph{तुल्य} आव्यूह\index{तुल्य आव्यूह}।)
\end{theorem}

\begin{proof}
किसी भी $x$ के लिए: $X = PX'$, और \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} $Y = AX$, $Y = PY'$ संतुष्ट करता
है। अतः $PY' = APX'$, अर्थात् सभी $X'$ के लिए $Y' = (P^{-1}AP)X'$: इसलिए नए
\omterm{def:b1:vspaces:free}{आधार} में $u$ का आव्यूह $P^{-1}AP$ है ($X'$ के लिए विहित स्तंभ लीजिए)।
\end{proof}

\begin{example}[अच्छा आधार प्रतिचित्रण को पारदर्शी बना देता है]
\label{ex:b1:matrices:conjugationrun}
मान लीजिए $u(x, y) = (y, x)$ (अदला-बदली), जिसका विहित \omterm{def:b1:vspaces:free}{आधार} में आव्यूह $A =
\begin{pmatrix} 0 & 1\\ 1 & 0\end{pmatrix}$ है। \omterm{def:b1:vspaces:free}{आधार} $\mathcal B' =
\bigl((1,1), (1,-1)\bigr)$ में:
\[
P = \begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix},
\qquad
P^{-1} = \frac12\begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix},
\qquad
P^{-1} A P = \begin{pmatrix} 1 & 0\\ 0 & -1 \end{pmatrix}.
\]
सचमुच किसी आव्यूह-गुणनफल की आवश्यकता नहीं थी: $u$ $(1,1)$ को अचल रखता है और
$(1,-1)$ को उलट देता है, अतः $\mathcal B'$ में उसका आव्यूह
$\operatorname{diag}(1, -1)$ \emph{होना ही} था --- अदला-बदली रेखा $y = x$ के
पार का परावर्तन है। किसी दिए हुए अंतःसमाकारिता के लिए ऐसा \omterm{def:b1:vspaces:free}{आधार} खोजना जिसमें
उसका आव्यूह विकर्ण हो जाए, स्नातक वर्ष~2 के खंड की केंद्रीय समस्या है
(लघुकरण-सिद्धांत); और नीचे की सप्ताहांत समस्या दिखाती है कि अकेली
बहुपद-सर्वसमिकाएँ ही कितनी दूर तक ले जाती हैं।
\end{example}

\begin{example}[आधार-परिवर्तन, उलटी दिशा में चलाया हुआ]
\label{ex:b1:matrices:reverserun}
$G = \operatorname{Vect}(1,-1)$ के अनुदिश $F = \operatorname{Vect}(1,1)$ पर
\omterm{def:b1:linmaps:projection}{प्रक्षेप} का, अनुकूलित \omterm{def:b1:vspaces:free}{आधार} $\mathcal B' = \bigl((1,1),(1,-1)\bigr)$ में,
पारदर्शी आव्यूह $A' = \operatorname{diag}(1, 0)$ है। उसका विहित-आधार वाला
आव्यूह पाने के लिए \cref{thm:b1:matrices:conjugation} को उलटी दिशा में
चलाइए, $A = P A' P^{-1}$:
\[
P = \begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix},
\quad
P^{-1} = \frac12\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix},
\quad
A = P\begin{pmatrix} 1 & 0\\ 0 & 0\end{pmatrix}P^{-1}
= \frac12\begin{pmatrix} 1 & 1\\ 1 & 1\end{pmatrix}.
\]
जाँच: $A^2 = A$ (वर्गसम), $\operatorname{tr} A = 1 = \operatorname{rk} A$,
और $A\binom{1}{1} = \binom11$, $A\binom{1}{-1} = 0$, जैसा निर्धारित था। यही
उलटी दिशा --- अच्छे \omterm{def:b1:vspaces:free}{आधार} में आव्यूह गढ़िए, फिर वापस संयुग्मित कीजिए --- वह
विधि है जिससे व्यवहार में घूर्णन, परावर्तन और \omterm{def:b1:linmaps:projection}{प्रक्षेप} के आव्यूह सचमुच बनाए
जाते हैं।
\end{example}

\begin{theorem}[कोटि मानक रूप]\label{thm:b1:matrices:rank}
किसी आव्यूह की \emph{कोटि} (उसके स्तंभों की कोटि, अथवा तुल्य रूप से संबद्ध
\omterm{def:b1:linmaps:def}{रैखिक प्रतिचित्रण} की कोटि) तुल्यता का एकमात्र अचर है: $r$ कोटि वाला हर $A
\in \mathcal{M}_{n,p}$ निम्नलिखित के तुल्य है
\[
J_r = \begin{pmatrix} I_r & 0 \\ 0 & 0 \end{pmatrix},
\]
और $\operatorname{rk}(A^{\mathsf T}) = \operatorname{rk}(A)$: अर्थात्
पंक्ति-कोटि स्तंभ-कोटि के बराबर होती है।
\end{theorem}

\begin{proof}
मान लीजिए $u \colon E \to F$ की कोटि $r$ है। $\ker u$ की कोई \omterm{def:b1:vspaces:sum}{पूरक} $S$ चुनिए
($\dim S = r$, \cref{thm:b1:linmaps:ranknullity}) जिसका \omterm{def:b1:vspaces:free}{आधार} $(e_1, \dots,
e_r)$ हो, और उसे $\ker u$ के \omterm{def:b1:vspaces:free}{आधार} से पूर्ण करके $E$ का \omterm{def:b1:vspaces:free}{आधार} बनाइए; \omterm{def:b1:linmaps:kerim}{प्रतिबिंब}
$f_i = u(e_i)$, $i \leq r$, $\operatorname{im} u$ का \omterm{def:b1:vspaces:free}{आधार} बनाते हैं (सीमन एक
तुल्याकारिता है), जिसे $F$ के \omterm{def:b1:vspaces:free}{आधार} तक पूर्ण कर लीजिए। इन आधारों में $u$ का
आव्यूह ठीक $J_r$ है। अतः व्युत्क्रमणीय $P, Q$ के लिए $A = Q J_r P^{-1}$।

\omterm{def:b1:matrices:transpose}{परिवर्त} लेने पर: $A^{\mathsf T} = (P^{-1})^{\mathsf T} J_r^{\mathsf T}
Q^{\mathsf T}$, जहाँ $J_r^{\mathsf T}$ उसी रूप का है (कोटि $r$) और बाहरी
गुणनखंड व्युत्क्रमणीय हैं (व्युत्क्रमणीय का \omterm{def:b1:matrices:transpose}{परिवर्त} व्युत्क्रमणीय होता है,
$AA^{-1} = I$ पर $(AB)^{\mathsf T} = B^{\mathsf T}A^{\mathsf T}$ लगाने से):
$\operatorname{rk} A^{\mathsf T} = r$।
\end{proof}

\section{पंक्ति संक्रियाएँ}

\begin{method}[आव्यूहों पर गाउसीय विलोपन]\label{met:b1:matrices:gauss}
तीन \emph{प्राथमिक पंक्ति संक्रियाएँ}\index{पंक्ति संक्रियाएँ} --- दो
पंक्तियों की अदला-बदली, किसी पंक्ति को $\lambda \neq 0$ से गुणा करना, किसी
पंक्ति का गुणज दूसरी में जोड़ना --- कोटि नहीं बदलतीं (हर एक किसी
व्युत्क्रमणीय आव्यूह से बाईं ओर का गुणन है)। कलनविधि: एक धुरी बनाइए (सबसे
बाईं अशून्य प्रविष्टि), उसके स्तंभ को नीचे से साफ़ कीजिए, अगली पंक्ति और
अगले स्तंभ पर बढ़िए; प्राप्त सोपानिक रूप की धुरियों की संख्या ही कोटि है।

\emph{प्रतिलोम की संगणना:} कलनविधि को खंड $(A \mid I_n)$ पर तब तक चलाइए जब
तक बायाँ खंड $I_n$ न बन जाए (यह संभव है तभी जब $A$ व्युत्क्रमणीय हो); तब
दायाँ खंड $A^{-1}$ है --- वस्तुतः प्रयुक्त प्राथमिक आव्यूहों का गुणनफल
$A^{-1}$ के बराबर है।
\end{method}

\begin{example}\label{ex:b1:matrices:inverse}
$A = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$: $(A \mid I_2)$ का लघुकरण
कीजिए:
\[
\begin{pmatrix} 1 & 2 & 1 & 0\\ 3 & 4 & 0 & 1 \end{pmatrix}
\to
\begin{pmatrix} 1 & 2 & 1 & 0\\ 0 & -2 & -3 & 1 \end{pmatrix}
\to
\begin{pmatrix} 1 & 0 & -2 & 1\\ 0 & 1 & \tfrac32 & -\tfrac12
\end{pmatrix},
\]
(संक्रियाएँ: $L_2 \leftarrow L_2 - 3L_1$; फिर $L_1 \leftarrow L_1 + L_2$,
$L_2 \leftarrow -\frac12 L_2$)। अतः $A^{-1} = \begin{pmatrix} -2 & 1 \\
\tfrac32 & -\tfrac12\end{pmatrix}$। \emph{जाँच:} $AA^{-1} = I_2$।
\end{example}

\begin{example}[प्राचल वाली कोटि, केवल पंक्तियों से]
\label{ex:b1:matrices:rankparameter}
$m \in \R$ के लिए $M_m = \begin{pmatrix} 1 & 1 & m\\ 1 & m & 1\\ m & 1 &
1\end{pmatrix}$ की कोटि। लघुकरण कीजिए: $L_2 \leftarrow L_2 - L_1$ और $L_3
\leftarrow L_3 - mL_1$ ये पंक्तियाँ देते हैं
\[
(1,\ 1,\ m), \qquad (0,\ m - 1,\ 1 - m), \qquad
(0,\ 1 - m,\ 1 - m^2).
\]
\emph{स्थिति $m = 1$}: अंतिम दो पंक्तियाँ शून्य हो जाती हैं --- एक धुरी,
$\operatorname{rk} M_1 = 1$ (मूल तीनों पंक्तियाँ बराबर थीं)। \emph{स्थिति $m
\neq 1$}: $L_2$ को $\frac1{m-1}$ से और $L_3$ को $\frac1{1-m}$ से मापित करके
$(0, 1, -1)$ और $(0, 1, 1 + m)$ पाइए, फिर $L_3 \leftarrow L_3 - L_2 = (0, 0,
m + 2)$। यदि $m = -2$: दो धुरियाँ, कोटि $2$; अन्यथा तीन धुरियाँ, कोटि $3$।
सारांश:
\[
\operatorname{rk} M_m =
\begin{cases}
1 & m = 1,\\
2 & m = -2,\\
3 & \text{अन्यथा}.
\end{cases}
\]
यही देहलियाँ \cref{ch:b1:det} में एक ही सारणिक-संगणना से निकल आएँगी
(\cref{exo:b1:det:7} का \omterm{def:b1:poly:def}{बहुपद} $-(m+2)(m-1)^2$) --- पर ध्यान दीजिए कि विलोपन
वह देता है जो सारणिक नहीं देता: अपभ्रष्ट स्थितियों में कोटि का \emph{मान},
केवल यह तथ्य नहीं कि वह गिर गई।
\end{example}

\begin{example}[घातों की संगणना]\label{ex:b1:matrices:powers}
$A = \begin{pmatrix} 1 & 1 \\ 0 & 1\end{pmatrix} = I + N$, जहाँ $N =
E_{12}$, $N^2 = 0$। चूँकि $I$ और $N$ क्रमविनिमेय हैं, द्विपद प्रमेय
(\cref{prop:b1:structures:binomial}) कट जाती है:
\[
A^k = I + kN = \begin{pmatrix} 1 & k \\ 0 & 1 \end{pmatrix}
\qquad (k \in \N, \text{ तथा } k \in \Z \text{ भी, बशर्ते } A^{-1} = I -
N).
\]
\end{example}

\begin{method}[$A^n$ की संगणना: तीन रास्ते]
\label{met:b1:matrices:powers}
\begin{enumerate}
  \item \emph{द्विपद रास्ता}: यदि $A = \lambda I + N$, जहाँ $N$ शून्यंभावी
  है, तो द्विपद प्रमेय कट जाती है (\cref{ex:b1:matrices:powers},
  \cref{exo:b1:matrices:5}); और वह लागू इसलिए होती है क्योंकि $\lambda I$ हर
  चीज़ के साथ क्रमविनिमेय है।
  \item \emph{\omterm{def:b1:poly:def}{बहुपद} रास्ता}: $A$ द्वारा संतुष्ट कोई बहुपद-सर्वसमिका खोजिए
  (विमा $2$ में सदा $A^2 = sA - pI$) और $X^n$ का उसके सापेक्ष लघुकरण कीजिए;
  नीचे की सप्ताहांत समस्या यह रास्ता पूरा-पूरा बनाती है।
  \item \emph{सादृश्य रास्ता}: ऐसा व्युत्क्रमणीय $P$ खोजिए कि $P^{-1}AP = D$
  सरल हो (विकर्ण, अथवा सरकाव), $D^n$ संगणित कीजिए, और फिर उलट दीजिए: $A^n =
  P D^n P^{-1}$ (\cref{thm:b1:matrices:conjugation},
  \cref{ex:b1:matrices:conjugationrun}); ऐसे $P$ की व्यवस्थित खोज ही स्नातक
  वर्ष~2 का लघुकरण-सिद्धांत है।
\end{enumerate}
रास्ता कोई भी हो, परिणाम को $n = 0, 1, 2$ पर जाँच लीजिए: ये तीन सस्ती
परीक्षाएँ लगभग हर चूक पकड़ लेती हैं।
\end{method}

\begin{remark}[सामान्य भूलें: अक्रमविनिमयता की क़ीमत]
\label{rem:b1:matrices:pitfalls}
अदिश बीजगणित की हर वह सर्वसमिका, जिसकी उपपत्ति गुणनखंडों का क्रम बदलती है,
$\mathcal{M}_n(K)$, $n \geq 2$ में मर जाती है। \emph{वर्ग}: $(A + B)^2 = A^2
+ AB + BA + B^2$, और बीच का भाग $2AB$ तक तभी ढहता है जब $AB = BA$
(\cref{exo:b1:matrices:1})। \emph{गुणनफलों की घातें}: $(AB)^k$ $ABAB\cdots$
है, $A^kB^k$ नहीं। \emph{शून्य भाजक}: $E_{12} \neq 0$ के साथ $E_{12}E_{12} =
0$; और इसीलिए \emph{कोई निरसन नहीं}: $AB = AC$ से $B = C$ तभी निकलता है जब
$A$ व्युत्क्रमणीय हो ($A^{-1}$ से गुणा कीजिए --- सही ओर से)। \emph{\omterm{def:b1:matrices:transpose}{अनुरेख}}:
$\operatorname{tr}(AB) = \operatorname{tr}(BA)$ सदा, पर व्यापक रूप से
$\operatorname{tr}(AB) \neq \operatorname{tr}A\operatorname{tr}B$ ($A = B =
I_2$ लीजिए: $2 \neq 4$), और $\operatorname{tr}(ABC) =
\operatorname{tr}(BCA)$ (चक्रीय), जबकि $\operatorname{tr}(ACB)$ भिन्न हो
सकते हैं। \emph{\omterm{def:b1:matrices:transpose}{परिवर्त} क्रम उलट देते हैं}: $(AB)^{\mathsf T} = B^{\mathsf
T}A^{\mathsf T}$ --- इस उलटाव को भूल जाना लांबिकता की संगणनाओं में सबसे आम
त्रुटि है (\cref{ch:b1:euclid})। संदेह हो तो किसी भी दावा की गई सर्वसमिका को
$E_{12}$ और $E_{21}$ पर परखिए: सबसे छोटी अक्रमविनिमेय जोड़ी अधिकांश ग़लत
सूत्रों को एक ही पंक्ति में खंडित कर देती है।
\end{remark}

\begin{remark}[यह शब्दकोश कहाँ जाता है]
\label{rem:b1:matrices:whereused}
आव्यूह-शब्दकोश इस खंड के शेष हर पृष्ठ पर काम आता है: \cref{ch:b1:det} हर
वर्ग आव्यूह से एक ही संख्या जोड़ देता है जो व्युत्क्रमणीयता तय कर देती है,
और $AX = B$ को व्यवस्थित रूप से हल कर देता है; \cref{ch:b1:euclid} उन
आव्यूहों को छाँट लेता है जो लंबाइयाँ सुरक्षित रखते हैं (लांबिक आव्यूह); और
\cref{ch:b1:multivar} में दो चरों के किसी फलन का द्वितीय-कोटि व्यवहार एक
सममित $2 \times 2$ आव्यूह है। ऊपर लगभग चलते-चलते आया \omterm{def:b1:matrices:transpose}{अनुरेख} एक शक्तिशाली अचर
बन जाता है: \cref{exo:b1:matrices:6,exo:b1:matrices:8} उसका पहला स्वाद देते
हैं, और स्नातक वर्ष~2 का खंड उसी पर अभिलक्षणिक-मान सिद्धांत खड़ा करता है।
सप्ताहांत समस्या दूसरा भारवाहक विकसित करती है: किसी आव्यूह द्वारा संतुष्ट
बहुपद-सर्वसमिकाएँ, जो $A^n$ की संगणना को दो-पदीय \omterm{pb:b1:matrices:1}{रैखिक पुनरावृत्ति} में बदल
देती हैं।
\end{remark}

\begin{remark}[पुस्तक 3 के भीतर के परिदृश्य]
\label{rem:b1:matrices:perspectives}
यहाँ प्रस्तुत तीन आव्यूह-कुलों की इस खंड में आगे भेंट तय है। \emph{सममित
आव्यूह} (\cref{ex:b1:matrices:symsplit}) दो चरों के फलनों के द्वितीय-कोटि
आँकड़े ढोते हैं: \cref{ch:b1:multivar} की मोंज कसौटी किसी सममित $2\times2$
आव्यूह के चिह्न-व्यवहार के विषय में एक \omterm{def:b1:logic:statement}{कथन} है, और उसका सारणिक $rt - s^2$
\cref{ch:b1:det} की मशीनरी से संगणित होता है। \emph{लांबिक आव्यूह}
($A^{\mathsf T}A = I$) \cref{ch:b1:euclid} के समदूरक हैं, जहाँ \omterm{def:b1:matrices:transpose}{परिवर्त} को
अंततः उसका ज्यामितीय अर्थ मिल जाता है: वह अंतर्गुणन की बीजगणितीय छाया है।
\emph{व्युत्क्रमणीय आव्यूहों} की व्यावहारिक परीक्षा \cref{ch:b1:det} में
होती है --- एक ही संख्या, $\det A \neq 0$ --- और वही उस खोज को बंद करती है
जो इस अध्याय ने पंक्ति-लघुकरण से आरंभ की थी। फिर \omterm{def:b1:matrices:transpose}{अनुरेख} और सारणिक सप्ताहांत
समस्या की अचर जोड़ी $(s, p)$ के रूप में यात्रा करते हैं, ठीक स्नातक वर्ष~2
के अभिलक्षणिक-मान सिद्धांत तक।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b1:matrices:1}
मान लीजिए $A = \begin{pmatrix} 1 & 2 \\ 0 & 1 \end{pmatrix}$ और $B =
\begin{pmatrix} 0 & 1 \\ 1 & 0\end{pmatrix}$। $AB$, $BA$, $A^2 - B^2$ और
$(A+B)(A-B)$ संगणित कीजिए; और समझाइए कि अंतिम दो भिन्न क्यों हैं।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:2}
निम्नलिखित की कोटि संगणित कीजिए
\[
M = \begin{pmatrix}
1 & 2 & 3\\
2 & 4 & 6\\
1 & 1 & 1
\end{pmatrix},
\qquad
N = \begin{pmatrix}
1 & 1 & 0 & 2\\
0 & 1 & 1 & 1\\
1 & 2 & 1 & 3
\end{pmatrix}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:3}
पंक्ति-लघुकरण से $A = \begin{pmatrix} 1 & 0 & 1\\ 2 & 1 & 1\\ 1 & 1 & 1
\end{pmatrix}$ का प्रतिलोम निकालिए, और एक गुणनफल पर जाँचिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:4}
$\R_2[X]$ के विहित \omterm{def:b1:vspaces:free}{आधार} में अंतःसमाकारिता $u(P) = P(X + 1)$ का आव्यूह लिखिए।
बिना संगणना के समझाइए कि वह व्युत्क्रमणीय क्यों है, और $u^{-1}$ का आव्यूह
दीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:5}
मान लीजिए $A = \begin{pmatrix} 2 & 1 \\ 0 & 2\end{pmatrix}$। $A = 2I + N$
लिखिए, $N^2$ संगणित कीजिए, और द्विपद प्रमेय से सभी $k \in \N$ के लिए $A^k$
निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:6}
सिद्ध कीजिए कि ऐसे कोई आव्यूह $A, B \in \mathcal{M}_n(K)$ नहीं हैं ($K = \R$
अथवा $\C$ के साथ) जिनके लिए $AB - BA = I_n$। \emph{(\omterm{def:b1:matrices:transpose}{अनुरेख} लीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:7}
आव्यूह $A$ \emph{शून्यंभावी} कहलाता है जब किसी $m$ के लिए $A^m = 0$। सिद्ध
कीजिए कि तब $I - A$ व्युत्क्रमणीय है, और
\[
(I - A)^{-1} = I + A + A^2 + \dots + A^{m-1} .
\]
अनुप्रयोग: $\begin{pmatrix} 1 & 2 & 3\\ 0 & 1 & 2\\ 0 & 0 & 1\end{pmatrix}$
का प्रतिलोम निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:8}
मान लीजिए $A \in \mathcal{M}_n(\R)$ $A^2 = A$ संतुष्ट करता है (वर्गसम)।
सिद्ध कीजिए कि $\operatorname{tr} A = \operatorname{rk} A$। \emph{($A$ की
व्याख्या किसी \omterm{def:b1:linmaps:projection}{प्रक्षेप} के रूप में कीजिए और कोई अनुकूलित \omterm{def:b1:vspaces:free}{आधार} चुनिए;
\cref{thm:b1:matrices:conjugation} कहता है कि \omterm{def:b1:matrices:transpose}{अनुरेख} \omterm{def:b1:vspaces:free}{आधार} पर निर्भर नहीं
करता, क्योंकि $\operatorname{tr}(P^{-1}MP) = \operatorname{tr} M$।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:9}
मान लीजिए $J \in \mathcal{M}_n(\R)$ सब-इकाई आव्यूह है। $J^2$ संगणित कीजिए,
और $a, b \in \R$ के लिए $M = aI + bJ$ की व्युत्क्रमणीयता की शर्त तथा
$M^{-1}$ निकालिए \emph{(उसी रूप $\alpha I + \beta J$ का प्रतिलोम खोजिए)}।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:10}
(कोटि-असमिकाएँ) $A, B \in \mathcal{M}_n(K)$ के लिए सिद्ध कीजिए
\[
\operatorname{rk}(A + B) \leq \operatorname{rk} A +
\operatorname{rk} B,
\qquad
\operatorname{rk}(AB) \geq \operatorname{rk} A + \operatorname{rk}
B - n .
\]
\emph{(दूसरी के लिए --- सिल्वेस्टर की असमिका --- $A$ के \omterm{def:b1:logic:map}{प्रतिचित्रण} के
$\operatorname{im} B$ तक के सीमन पर कोटि--शून्यता लगाइए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:11}
मान लीजिए $D = \operatorname{diag}(d_1, \dots, d_n)$, जहाँ $d_i$
\emph{युग्मशः भिन्न} हैं।
\begin{enumerate}
  \item सिद्ध कीजिए कि कोई आव्यूह $A$ $D$ के साथ क्रमविनिमेय है तभी जब $A$
  विकर्ण हो। \emph{($AD$ और $DA$ की $(i,j)$ प्रविष्टियों की तुलना कीजिए।)}
  \item इससे $\mathcal{M}_n(K)$ का \emph{केंद्र} निकालिए: \emph{हर} आव्यूह
  के साथ क्रमविनिमेय आव्यूह ठीक अदिश आव्यूह $\lambda I_n$ हैं। \emph{(पहले
  $D$ के सामने परखिए, फिर आव्यूहों $E_{ij}$ के सामने।)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:12}
(एक-कोटि आव्यूह) मान लीजिए $A \in \mathcal{M}_n(K)$, $A \neq 0$।
\begin{enumerate}
  \item सिद्ध कीजिए कि $\operatorname{rk} A = 1$ तभी जब किसी अशून्य स्तंभ $C
  \in \mathcal{M}_{n,1}$ और किसी अशून्य पंक्ति $L \in \mathcal{M}_{1,n}$ के
  लिए $A = CL$।
  \item ऐसे $A$ के लिए सिद्ध कीजिए $A^2 = (\operatorname{tr} A)\,A$; और
  निष्कर्ष निकालिए कि एक-कोटि आव्यूह शून्यंभावी है तभी जब उसका \omterm{def:b1:matrices:transpose}{अनुरेख} शून्य
  हो।
  \item यदि $\operatorname{tr} A \neq -1$, तो सिद्ध कीजिए कि $I_n + A$
  व्युत्क्रमणीय है, और
        \[
        (I_n + A)^{-1} = I_n - \frac{1}{1 + \operatorname{tr}
        A}\,A ,
        \]
        और यह कि $\operatorname{tr} A = -1$ होने पर $I_n + A$ व्युत्क्रमणीय
        \emph{नहीं} है। \emph{($I_n + A$ द्वारा मारा जाने वाला कोई सदिश
        खोजिए।)}
\end{enumerate}
\end{exercise}

\section{समस्या: बहुपद-भाग से आव्यूह की घातें}

\begin{problem}\label{pb:b1:matrices:1}
$A^{100}$ को प्रविष्टि-दर-प्रविष्टि संगणित करना निराशाजनक है; उसे $A$ द्वारा
संतुष्ट किसी बहुपद-सर्वसमिका के द्वारा संगणित करना तीन पंक्तियों का काम है।
यह समस्या उस विधि को शून्य से बनाती है: $X^n$ का यूक्लिडीय भाग, वह सर्वसमिका
$A^2 - sA + pI = 0$ जिसे हर $2 \times 2$ आव्यूह सत्यापित करता है (विमा $2$
में केली--हैमिल्टन प्रमेय), और आव्यूह-घातों तथा \omterm{def:b1:linmaps:def}{रैखिक} पुनरावृत्तियों के बीच
का शब्दकोश --- और साथ चलता उदाहरण \omterm{pb:b1:matrices:1}{फ़िबोनाच्ची अंक}। \index{केली--हैमिल्टन
प्रमेय (विमा 2)} \index{फ़िबोनाच्ची अंक}\index{रैखिक पुनरावृत्ति}

\medskip \noindent\textbf{भाग I --- शेषफल-कलन।} $s, p \in K$ और $D = X^2 -
sX + p$ नियत कीजिए।
\begin{enumerate}
  \item औचित्य दीजिए कि हर $n \in \N$ के लिए अद्वितीय $Q_n \in K[X]$ और
  $(a_n, b_n) \in K^2$ ऐसे हैं कि
        \[
        X^n = Q_n\,D + a_n X + b_n ,
        \]
        और $(a_0, b_0)$ तथा $(a_1, b_1)$ संगणित कीजिए।
  \item $X$ से गुणा करके और फिर से भाग देकर पुनरावृत्तियाँ स्थापित कीजिए
        \[
        a_{n+1} = s\,a_n + b_n,
        \qquad
        b_{n+1} = -p\,a_n ,
        \]
        और $a_{n+2} = s\,a_{n+1} - p\,a_n$ निकालिए: अर्थात् गुणांक-अनुक्रम
        $D$ से जुड़ी \omterm{pb:b1:matrices:1}{रैखिक पुनरावृत्ति} का पालन करता है।
  \item मान लीजिए $D$ के दो भिन्न मूल $\lambda \neq \mu$ हैं। भाग-सर्वसमिका
  का मूल्यांकन करके सिद्ध कीजिए
        \[
        a_n = \frac{\lambda^n - \mu^n}{\lambda - \mu},
        \qquad
        b_n = \frac{\lambda\mu^n - \mu\lambda^n}{\lambda - \mu} .
        \]
  \item मान लीजिए $D = (X - \lambda)^2$। भाग-सर्वसमिका के \omterm{def:b1:derivative:def}{अवकलज} का प्रयोग
  करके $a_n = n\lambda^{n-1}$ और $b_n = (1 - n)\lambda^{n}$ सिद्ध कीजिए।
  \item दिखाइए कि बहुपदों में किसी नियत आव्यूह $M \in \mathcal{M}_k(K)$ का
  प्रतिस्थापन योग और गुणनफल का आदर करता है: $(PQ)(M) = P(M)\,Q(M)$। इससे
  निष्कर्ष निकालिए कि यदि $D(M) = 0$, तो
        \[
        M^n = a_n\,M + b_n\,I \qquad (n \in \N).
        \]
\end{enumerate}

\medskip \noindent\textbf{भाग II --- विमा 2: \omterm{def:b1:matrices:transpose}{अनुरेख}, सारणिक संख्या,
केली--हैमिल्टन।} $A = \begin{pmatrix} a & b\\ c & d\end{pmatrix}$ के लिए $s
= a + d = \operatorname{tr} A$ और $p = ad - bc$ रखिए (वही संख्या जिसे
\cref{ch:b1:det} सारणिक नाम देगा)।
\begin{enumerate}[resume]
  \item सीधी संगणना से \emph{विमा $2$ में केली--हैमिल्टन सर्वसमिका} सत्यापित
  कीजिए:
        \[
        A^2 - s\,A + p\,I_2 = 0 .
        \]
  \item सीधे प्रसार से सिद्ध कीजिए कि $p$ गुणात्मक है: स्पष्ट संकेतन में
  $p(AB) = p(A)\,p(B)$। फिर दिखाइए: $A$ व्युत्क्रमणीय है तभी जब $p \neq 0$,
  और तब
        \[
        A^{-1} = \frac1p\,\bigl(s\,I_2 - A\bigr).
        \]
  \item मान लीजिए $A = \begin{pmatrix} 1 & 1\\ 0 & 2\end{pmatrix}$। $s$,
  $p$, तथा $D$ के मूल संगणित कीजिए, और $A^n$ के लिए \omterm{def:b1:topology:closed}{संवृत} सूत्र निकालिए; उसे
  $A^2$ की सीधी संगणना के सामने जाँचिए।
  \item मान लीजिए $A = \begin{pmatrix} 3 & 1\\ -1 & 1\end{pmatrix}$। दिखाइए
  कि $D$ का एक दुहरा मूल है और $A^n$ संगणित कीजिए; $n = 2$ पर जाँचिए।
  \item मान लीजिए $F = \begin{pmatrix} 1 & 1\\ 1 & 0\end{pmatrix}$, और
  फ़िबोनाच्ची अंकों को $F_0 = 0$, $F_1 = 1$, $F_{n+2} = F_{n+1} + F_n$ से
  परिभाषित कीजिए। सिद्ध कीजिए
        \[
        F^n = \begin{pmatrix} F_{n+1} & F_n\\ F_n &
        F_{n-1}\end{pmatrix} \quad (n \geq 1),
        \]
        इससे बिने का सूत्र $F_n = \dfrac{\varphi^n - \psi^n}{\sqrt5}$
        निकालिए, जहाँ $\varphi = \frac{1 + \sqrt5}2$, $\psi = \frac{1 -
        \sqrt5}2$, और प्रश्न 7 का प्रयोग करके कासिनी की सर्वसमिका
        $F_{n+1}F_{n-1} - F_n^2 = (-1)^n$।
\end{enumerate}

\medskip \noindent\textbf{भाग III --- \omterm{def:b1:linmaps:def}{रैखिक} पुनरावृत्तियाँ, संरचनात्मक
दृष्टि से।} $p \neq 0$ के साथ $s, p \in K$ नियत कीजिए, और मान लीजिए $E_D$ उन
अनुक्रमों का \omterm{def:b1:logic:sets}{समुच्चय} है जो सभी $n$ के लिए $u_{n+2} = s\,u_{n+1} - p\,u_n$
संतुष्ट करते हैं।
\begin{enumerate}[resume]
  \item दिखाइए कि $E_D$ विमा $2$ वाली \omterm{def:b1:vspaces:def}{सदिश समष्टि} है
  (\cref{exo:b1:findim:10} को अनुकूलित कीजिए)।
  \item दिखाइए कि भाग I का अनुक्रम $(a_n)$ $E_D$ का वही अवयव है जिसके आरंभिक
  मान $0, 1$ हैं, और यह कि हर $u \in E_D$ संतुष्ट करता है
        \[
        u_n = u_1\,a_n + u_0\,b_n \qquad (n \in \N),
        \]
        जहाँ $(b_n)$ भाग I जैसा ही है: अर्थात् भाग के शेषफल \emph{सभी}
        पुनरावृत्तियाँ एक ही साथ हल कर देते हैं।
  \item यदि $\lambda \neq \mu$ $D$ के मूल हैं, तो दिखाइए कि
  $\bigl((\lambda^n), (\mu^n)\bigr)$ $E_D$ का \omterm{def:b1:vspaces:free}{आधार} है; और यदि $\lambda \neq
  0$ के साथ $D = (X-\lambda)^2$, तो दिखाइए कि $\bigl((\lambda^n),
  (n\lambda^n)\bigr)$ एक \omterm{def:b1:vspaces:free}{आधार} है।
  \item पूरी तरह हल कीजिए: $u_{n+2} = u_{n+1} + 6u_n$, $u_0 = 1$, $u_1 = 8$;
  और उत्तर $u_2$ तथा $u_3$ पर जाँचिए।
  \item मान लीजिए $C = \begin{pmatrix} 0 & 1\\ -p & s\end{pmatrix}$ ($D$ का
  \emph{सहचर आव्यूह})। दिखाइए कि
        \[
        \begin{pmatrix} u_{n}\\ u_{n+1}\end{pmatrix}
        = C^n \begin{pmatrix} u_0\\ u_1\end{pmatrix}
        \quad (u \in E_D),
        \]
        और यह कि $\operatorname{tr} C = s$ तथा $p(C) = p$: अर्थात्
        पुनरावृत्ति और आव्यूह वही एक \omterm{def:b1:poly:def}{बहुपद} $D$ ढोते हैं।
\end{enumerate}

\medskip \noindent\textbf{भाग IV --- घात तीन।} मान लीजिए $D_3 = X^3 - \alpha
X^2 - \beta X - \gamma$ और
\[
C_3 = \begin{pmatrix} 0 & 1 & 0\\ 0 & 0 & 1\\ \gamma & \beta &
\alpha \end{pmatrix}.
\]
\begin{enumerate}[resume]
  \item दिखाइए कि $D_3(C_3) = 0$। \emph{($C_3$ की घातों के अंतर्गत विहित
  आधार-सदिशों के \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} संगणित कीजिए: $C_3$ का \omterm{def:b1:logic:map}{प्रतिचित्रण} $e_1 \mapsto
  \dots \mapsto$ भेजता है, जो अंतिम पंक्ति से बाध्य संयोजन है।)}
  \item दिखाइए कि यदि $D_3$ के तीन भिन्न मूल $\lambda_1, \lambda_2,
  \lambda_3$ हैं, तो $X^n$ को $D_3$ से भाग देने पर मिला शेषफल $R_n$ गाँठों
  $\lambda_i$ पर मानों $\lambda_i^n$ का \emph{लाग्रांज अंतर्वेशक} है
  (\cref{thm:b1:poly:lagrange}); और इससे निष्कर्ष निकालिए कि $C_3^{\,n}$ की
  हर प्रविष्टि $\lambda_1^n, \lambda_2^n, \lambda_3^n$ का कोई नियत \omterm{def:b1:linmaps:def}{रैखिक}
  संयोजन है।
  \item हल कीजिए: $u_0 = 0$, $u_1 = 1$, $u_2 = 1$ के साथ $u_{n+3} = 2u_{n+2}
  + u_{n+1} - 2u_n$। \emph{($D_3 = (X - 1)(X + 1)(X - 2)$ का गुणनखंडन
  कीजिए।)} $u_3$ पर जाँचिए।
  \item $(X - \lambda)^3$ के सापेक्ष $X^n$ का शेषफल संगणित कीजिए
  \emph{($\lambda$ पर $X^n$ का टेलर प्रसार)}, और जब $N^3 = 0$ हो तथा $N$
  आसपास की हर चीज़ के साथ क्रमविनिमेय हो, तब $(\lambda I + N)^n$ के लिए एक
  सूत्र निकालिए; उसे द्विपद प्रमेय के सामने जाँचिए।
  \item दिखाइए कि भिन्न मूलों वाले $D_3$ के लिए कोटि-$3$ की पुनरावृत्ति का
  व्यापक हल $u_n = c_1 \lambda_1^n + c_2\lambda_2^n + c_3\lambda_3^n$ है:
  सिद्ध कीजिए कि तीनों गुणोत्तर अनुक्रम हल-समष्टि का \omterm{def:b1:vspaces:free}{आधार} बनाते हैं।
  \emph{(स्वतंत्रता के लिए किसी शून्य संयोजन का $n = 0, 1, 2$ पर मूल्यांकन
  कीजिए और भिन्न-भिन्न गाँठों $\lambda_i$ पर एक अंतर्वेशन-निकाय पहचानिए।)}
\end{enumerate}

\medskip \noindent\textbf{भाग V --- फ़िबोनाच्ची लाभांश, और संश्लेषण।}
\begin{enumerate}[resume]
  \item $F_1 + F_2 + \dots + F_n = F_{n+2} - 1$ सिद्ध कीजिए।
  \item $F^{m+n} = F^m F^n$ से योग-सूत्र निकालिए
        \[
        F_{m+n} = F_{m+1}F_n + F_m F_{n-1},
        \]
        और $F_{2n} = F_n(F_{n+1} + F_{n-1})$ निकालिए।
  \item सिद्ध कीजिए कि हर $n \geq 0$ के लिए $F_n$ $\varphi^n/\sqrt5$ के सबसे
  निकट का पूर्णांक है।
  \item मान लीजिए $t_n = \operatorname{tr}(F^n) = F_{n+1} + F_{n-1}$
  (\emph{लूका अंक} $L_n$)। $t_{n+2} = t_{n+1} + t_n$, $t_1 = 1$, $t_2 = 3$
  दिखाइए, यह कि $L_n = \varphi^n + \psi^n$, और $F_{2n} = F_n L_n$ फिर से
  प्राप्त कीजिए।
  \item संश्लेषण, चार वाक्यों में: $2 \times 2$ आव्यूह की घातें
  $\mathcal{M}_2(K)$ के समतल $\operatorname{Vect}(I, A)$ में क्यों रहती हैं
  (कौन-सा विमा-तर्क किसी द्विघातीय सर्वसमिका की गारंटी देता है, और भाग II ने
  कौन-सी स्पष्ट सर्वसमिका दी); यूक्लिडीय भाग घातांकन को दो-पदीय पुनरावृत्ति
  में कैसे बदल देता है; इस समस्या का कौन-सा \omterm{def:b1:logic:statement}{कथन} सभी विमाओं में मान्य किसी
  प्रमेय की $n = 2$ वाली स्थिति है (उसका नाम बताइए, और कहिए कि इस शृंखला में
  वह कहाँ सिद्ध की गई है); और सहचर आव्यूह वाली रचना इस चित्र में क्या जोड़ती
  है।
\end{enumerate}
\end{problem}
