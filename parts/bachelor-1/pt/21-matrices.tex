\chapter{Matrizes}\label{ch:b1:matrices}

Uma matriz é uma \omterm{def:b1:linmaps:def}{aplicação linear} escrita em \omterm{prop:b1:vspaces:coordinates}{coordenadas}. Este
capítulo instala o dicionário --- a composição torna-se produto de
matrizes, a bijetividade torna-se invertibilidade, a \omterm{def:b1:matrices:changeofbasis}{mudança de base}
torna-se conjugação --- e o lado algorítmico: \omterm{met:b1:matrices:gauss}{operações sobre linhas},
cálculo de postos e de inversas. Encontradas pela primeira vez no
volume do ensino médio, as matrizes ganham agora o fundamento teórico
do \cref{ch:b1:vspaces,ch:b1:findim,ch:b1:linmaps}.

\section{Matrizes e aplicações lineares}

\begin{definition}\label{def:b1:matrices:def}
$\mathcal{M}_{n,p}(K)$ é o \omterm{def:b1:vspaces:def}{espaço vetorial} dos quadros $n \times p$ de escalares $A
= (a_{ij})$ ($i$:
linha, $j$: coluna), de dimensão $np$ (\omterm{def:b1:vspaces:free}{base}: as matrizes $E_{ij}$
com um único $1$). Dadas \omterm{def:b1:vspaces:free}{bases} $\mathcal{B} = (e_1, \dots, e_p)$ de $E$ e $\mathcal{C}$ de $F$ ($\dim F = n$),
a \emph{matriz de $u \in \mathcal{L}(E, F)$} é o quadro cuja $j$-ésima coluna lista as
\omterm{prop:b1:vspaces:coordinates}{coordenadas} de $u(e_j)$ em $\mathcal{C}$:
\[
\operatorname{Mat}_{\mathcal{B},\mathcal{C}}(u) = (a_{ij}),
\qquad u(e_j) = \sum_{i=1}^{n} a_{ij}\, f_i .
\]
A \omterm{def:b1:logic:map}{aplicação} $u \mapsto \operatorname{Mat}_{\mathcal{B},\mathcal{C}}(u)$ é um isomorfismo de $\mathcal{L}(E, F)$ sobre $\mathcal{M}_{n,p}(K)$ (a \cref{prop:b1:linmaps:basis}:
uma \omterm{def:b1:linmaps:def}{aplicação linear} é exatamente uma escolha de imagens dos $e_j$).
\end{definition}

\begin{example}[A derivada, como matriz]
\label{ex:b1:matrices:derivative}
Seja $D(P) = P'$ em $\R_3[X]$. Na \omterm{def:b1:vspaces:free}{base} monomial $(1, X, X^2,
X^3)$: $D(1) = 0$, $D(X) = 1$,
$D(X^2) = 2X$, $D(X^3) = 3X^2$, logo
\[
\operatorname{Mat}(D) =
\begin{pmatrix}
0 & 1 & 0 & 0\\
0 & 0 & 2 & 0\\
0 & 0 & 0 & 3\\
0 & 0 & 0 & 0
\end{pmatrix}.
\]
Na \omterm{def:b1:vspaces:free}{base} \emph{dividida} $\bigl(1,\ X,\ \frac{X^2}2,\
\frac{X^3}6\bigr)$, cada vetor da \omterm{def:b1:vspaces:free}{base} é levado no
anterior ($D\bigl(\frac{X^k}{k!}\bigr) = \frac{X^{k-1}}{(k-1)!}$), e a matriz torna-se o deslocamento puro: uns na
superdiagonal, zeros em todo o resto. Duas morais: a matriz pertence
ao \emph{par} (\omterm{def:b1:logic:map}{aplicação}, \omterm{def:b1:vspaces:free}{base}), e não à \omterm{def:b1:logic:map}{aplicação} sozinha; e uma
boa \omterm{def:b1:vspaces:free}{base} torna a estrutura visível de relance --- a forma de
deslocamento mostra na hora que $D^4 = 0$ em $\R_3[X]$, cada
potência da matriz empurrando a sua diagonal de uns um passo mais
para fora.
\end{example}

\begin{definition}[Produto]\label{def:b1:matrices:product}
Para $A \in \mathcal{M}_{n,p}$ e $B \in \mathcal{M}_{p,q}$:
\[
(AB)_{ik} = \sum_{j=1}^{p} a_{ij}\, b_{jk}
\qquad (1 \leq i \leq n,\ 1 \leq k \leq q).
\]
Isto é precisamente a matriz da composição: $\operatorname{Mat}(v \circ u) = \operatorname{Mat}(v)\,
\operatorname{Mat}(u)$ (com as \omterm{def:b1:vspaces:free}{bases}
coincidindo no meio). Do mesmo modo, se $X$ é a coluna das
\omterm{prop:b1:vspaces:coordinates}{coordenadas} de $x$, a coluna de $u(x)$ é $AX$.
\end{definition}

\begin{proof}[Demonstração da fórmula da composição]
\[
v(u(e_k)) = v\Bigl(\sum_j b_{jk} f_j\Bigr) = \sum_j b_{jk}\, v(f_j)
= \sum_j b_{jk} \sum_i a_{ij}\, g_i
= \sum_i \Bigl(\sum_j a_{ij} b_{jk}\Bigr) g_i . \qedhere
\]
\end{proof}

\begin{proposition}[A álgebra $\mathcal{M}_n(K)$]\label{prop:b1:matrices:ring}
As matrizes quadradas $\mathcal{M}_n(K)$ formam um \omterm{def:b1:structures:ring}{anel} (não comutativo para
$n
\geq 2$), com identidade $I_n$; o seu \omterm{def:b1:structures:group}{grupo} de unidades é o
\emph{grupo linear geral}\index{grupo linear geral} $GL_n(K)$,
correspondente aos endomorfismos \omterm{def:b1:logic:inj}{bijetivos}. Para $A, B \in
\mathcal{M}_n(K)$:
\[
AB = I_n \implies A \in GL_n(K) \text{ e } B = A^{-1}
\]
(as inversas de um lado são inversas dos dois lados, pelo \cref{cor:b1:linmaps:samedim}).
\end{proposition}

\begin{proof}
Os axiomas de \omterm{def:b1:structures:ring}{anel} transportam-se de $\mathcal{L}(E)$ através do isomorfismo da
\cref{def:b1:matrices:def}: ele converte composição em produto e soma em soma, logo a
associatividade, a distributividade e o papel de $I_n$ são herdados
dos fatos correspondentes sobre aplicações, sem nenhuma verificação
entrada a entrada. Não comutatividade: $E_{12}E_{21} = E_{11} \neq E_{22}
= E_{21}E_{12}$. Se $AB = I_n$: o
endomorfismo $a$ de $A$ satisfaz $a \circ b = \mathrm{id}$, logo $a$ é \omterm{def:b1:logic:inj}{sobrejetivo}
($x =
a(b(x))$ exibe uma \omterm{def:b1:logic:map}{pré-imagem} de todo $x$), e portanto \omterm{def:b1:logic:inj}{bijetivo} em
\omterm{def:b1:findim:def}{dimensão finita} (o \cref{cor:b1:linmaps:samedim}); compondo $a
\circ b = \mathrm{id}$ com $a^{-1}$ à esquerda dá
$b =
a^{-1}$, e então $b\circ a = \mathrm{id}$ também: a inversa de um lado era, o tempo
todo, inversa dos dois lados --- um favor estritamente de \omterm{def:b1:findim:def}{dimensão
finita}.
\end{proof}

\begin{definition}[Transposta; traço]\label{def:b1:matrices:transpose}
A \emph{transposta} de $A = (a_{ij}) \in \mathcal{M}_{n,p}$ é $A^{\mathsf T}
= (a_{ji}) \in \mathcal{M}_{p,n}$; ela satisfaz $(AB)^{\mathsf T} = B^{\mathsf T}
A^{\mathsf T}$ e $(A^{\mathsf T})^{\mathsf T} = A$. O
\emph{traço}\index{traço} de uma matriz quadrada é $\operatorname{tr} A
= \sum_i a_{ii}$; ele é
\omterm{def:b1:linmaps:def}{linear}, e
\[
\operatorname{tr}(AB) = \operatorname{tr}(BA)
\qquad (A \in \mathcal{M}_{n,p},\ B \in \mathcal{M}_{p,n}).
\]
\end{definition}

\begin{proof}[Demonstração da identidade do traço]
$\operatorname{tr}(AB) = \sum_i \sum_j a_{ij} b_{ji}$ e $\operatorname{tr}(BA) = \sum_j \sum_i b_{ji} a_{ij}$: a mesma soma dupla.
\end{proof}

\begin{example}[O traço em ação]
\label{ex:b1:matrices:tracework}
A \omterm{def:b1:linmaps:projection}{projeção} do \cref{ch:b1:linmaps} sobre $\operatorname{Vect}(1,1)$ ao longo de $\operatorname{Vect}(0,1)$, $p(x, y) = (x, x)$, tem matriz
$A = \begin{pmatrix} 1 & 0\\ 1 &
0\end{pmatrix}$ na \omterm{def:b1:vspaces:free}{base} canônica: de fato $A^2 = A$, e
\[
\operatorname{tr} A = 1 = \operatorname{rk} A ,
\]
ilustrando o \cref{exo:b1:matrices:8}: para idempotentes, o \omterm{def:b1:matrices:transpose}{traço} \emph{conta} a
dimensão da \omterm{def:b1:linmaps:kerim}{imagem}, seja qual for a \omterm{def:b1:vspaces:free}{base} enviesada em que a matriz
está escrita. O mecanismo da invariância é a identidade $\operatorname{tr}(AB) = \operatorname{tr}(BA)$:
\[
\operatorname{tr}\bigl(P^{-1}(AP)\bigr) =
\operatorname{tr}\bigl((AP)P^{-1}\bigr) = \operatorname{tr} A ,
\]
logo todas as \omterm{thm:b1:matrices:conjugation}{matrizes semelhantes} a $A$ partilham o seu \omterm{def:b1:matrices:transpose}{traço} --- o
primeiro \emph{invariante numérico} de um endomorfismo, ao qual se
juntará o determinante no \cref{ch:b1:det} (o par $(s, p)$ do problema de fim
de semana abaixo).
\end{example}

\begin{example}[Simétrica mais antissimétrica]
\label{ex:b1:matrices:symsplit}
Chame $A$ de \emph{simétrica} quando $A^{\mathsf T} = A$, e de
\emph{antissimétrica} quando $A^{\mathsf T} = -A$. Toda matriz quadrada se decompõe
de modo único como uma mais a outra:
\[
A = \underbrace{\frac{A + A^{\mathsf T}}{2}}_{\text{simétrica}}
+ \underbrace{\frac{A - A^{\mathsf T}}{2}}_{\text{antissimétrica}},
\]
e uma matriz que é as duas coisas é nula ($A = -A$): os dois
\omterm{def:b1:logic:sets}{conjuntos} são \omterm{def:b1:vspaces:sum}{subespaços suplementares} de $\mathcal{M}_n(K)$ --- o análogo exato
da decomposição par/ímpar de funções (o \cref{ex:b1:vspaces:evenodd}), com a transposição a
fazer o papel de $x \mapsto -x$. Dimensões: uma matriz simétrica é \omterm{def:b1:vspaces:free}{livre} na
diagonal e acima dela, e uma antissimétrica estritamente acima
(diagonal nula):
\[
\frac{n(n+1)}{2} + \frac{n(n-1)}{2} = n^2 ,
\]
e o fechamento da contagem é a confirmação, por Grassmann, do
caráter direto. Para $n = 2$: $\begin{pmatrix} 1 & 5\\ 1 &
2\end{pmatrix} = \begin{pmatrix} 1 & 3\\ 3 & 2\end{pmatrix} +
\begin{pmatrix} 0 & 2\\ -2 & 0\end{pmatrix}$. As matrizes simétricas voltam
como os dados de segunda ordem do \cref{ch:b1:multivar} (a tripla de Monge $r, s,
t$), e as simétricas ortogonais são classificadas no \cref{exo:b1:euclid:12}.
\end{example}

\section{Mudança de base}

\begin{definition}\label{def:b1:matrices:changeofbasis}
Sejam $\mathcal{B}, \mathcal{B}'$ \omterm{def:b1:vspaces:free}{bases} de $E$. A \emph{matriz de mudança de
base}\index{mudança de base} $P =
P_{\mathcal{B}\to\mathcal{B}'}$ tem por colunas as \omterm{prop:b1:vspaces:coordinates}{coordenadas}
dos vetores da \emph{nova} \omterm{def:b1:vspaces:free}{base} na \omterm{def:b1:vspaces:free}{base} \emph{antiga}. Ela é
invertível, $P^{-1} = P_{\mathcal{B}'\to\mathcal{B}}$, e as \omterm{prop:b1:vspaces:coordinates}{coordenadas} transformam-se por $X = PX'$
(antigas $=$ $P\,\cdot$ novas).
\end{definition}

\begin{example}[Ler a matriz de mudança de base]
\label{ex:b1:matrices:changerun}
Em $\R^2$, da \omterm{def:b1:vspaces:free}{base} canônica $\mathcal B$ para $\mathcal B' =
\bigl((1,1), (1,-1)\bigr)$:
\[
P = P_{\mathcal B\to\mathcal B'} =
\begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix}
\]
(os vetores novos escritos em \omterm{prop:b1:vspaces:coordinates}{coordenadas} antigas, coluna a
coluna). O vetor de \omterm{prop:b1:vspaces:coordinates}{coordenadas} antigas $X = (3, 1)^{\mathsf T}$ tem \omterm{prop:b1:vspaces:coordinates}{coordenadas} novas
$X' = P^{-1}X = \frac12(3 + 1,\ 3 - 1)^{\mathsf T} =
(2, 1)^{\mathsf T}$: de fato $2(1,1) + 1(1,-1) = (3,1)$. Atenção ao sentido --- a matriz $P$ é
construída a partir da \omterm{def:b1:vspaces:free}{base} \emph{nova} mas converte \omterm{prop:b1:vspaces:coordinates}{coordenadas}
\emph{novas em antigas} ($X = PX'$); passar de antigas para novas
custa a inversa. Escrever a verificação $2(1,1) + (1,-1) = (3,1)$ depois de cada
conversão pega o erro do $P$ invertido, que é o mais comum do
capítulo.
\end{example}

\begin{theorem}[Mudança de base para uma aplicação]\label{thm:b1:matrices:conjugation}
Seja $u \in \mathcal{L}(E)$ com matriz $A$ em $\mathcal{B}$ e $A'$ em $\mathcal{B}'$, e $P = P_{\mathcal{B}\to\mathcal{B}'}$. Então
\[
A' = P^{-1} A\, P .
\]
Duas matrizes relacionadas desta maneira dizem-se
\emph{semelhantes}\index{matrizes semelhantes}. (Para $u \colon E \to F$ com dois
pares de \omterm{def:b1:vspaces:free}{bases}, a fórmula é $A' = Q^{-1} A P$ --- matrizes
\emph{equivalentes}\index{matrizes equivalentes}.)
\end{theorem}

\begin{proof}
Para todo $x$: $X = PX'$ e a \omterm{def:b1:linmaps:kerim}{imagem} satisfaz $Y = AX$, $Y = PY'$.
Logo $PY' = APX'$, isto é, $Y' = (P^{-1}AP)X'$ para todo $X'$: a matriz de $u$ na nova
\omterm{def:b1:vspaces:free}{base} é $P^{-1}AP$ (tome para $X'$ as colunas canônicas).
\end{proof}

\begin{example}[Uma boa base torna uma aplicação transparente]
\label{ex:b1:matrices:conjugationrun}
Seja $u(x, y) = (y, x)$ (a troca), com matriz $A = \begin{pmatrix} 0
& 1\\ 1 & 0\end{pmatrix}$ na \omterm{def:b1:vspaces:free}{base} canônica. Na \omterm{def:b1:vspaces:free}{base}
$\mathcal B' = \bigl((1,1), (1,-1)\bigr)$:
\[
P = \begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix},
\qquad
P^{-1} = \frac12\begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix},
\qquad
P^{-1} A P = \begin{pmatrix} 1 & 0\\ 0 & -1 \end{pmatrix}.
\]
Nenhum produto de matrizes foi realmente necessário: $u$ fixa
$(1,1)$ e inverte $(1,-1)$, logo em $\mathcal B'$ a sua matriz \emph{tem de}
ser $\operatorname{diag}(1, -1)$ --- a troca é a reflexão em torno da reta $y = x$.
Encontrar, para um dado endomorfismo, uma \omterm{def:b1:vspaces:free}{base} em que a sua matriz
se torne diagonal é o problema central do volume do segundo ano de
graduação (a teoria da redução); o problema de fim de semana abaixo
mostra quão longe as identidades polinomiais sozinhas já levam.
\end{example}

\begin{example}[Mudança de base, executada ao contrário]
\label{ex:b1:matrices:reverserun}
A \omterm{def:b1:linmaps:projection}{projeção} sobre $F = \operatorname{Vect}(1,1)$ ao longo de $G =
\operatorname{Vect}(1,-1)$ tem, na \omterm{def:b1:vspaces:free}{base} adaptada
$\mathcal
B' = \bigl((1,1),(1,-1)\bigr)$, a matriz transparente $A' =
\operatorname{diag}(1, 0)$. Para obter a sua matriz na \omterm{def:b1:vspaces:free}{base}
canônica, execute o \cref{thm:b1:matrices:conjugation} ao contrário, $A = P A'
P^{-1}$:
\[
P = \begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix},
\quad
P^{-1} = \frac12\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix},
\quad
A = P\begin{pmatrix} 1 & 0\\ 0 & 0\end{pmatrix}P^{-1}
= \frac12\begin{pmatrix} 1 & 1\\ 1 & 1\end{pmatrix}.
\]
Verificação: $A^2 = A$ (idempotente), $\operatorname{tr} A = 1 =
\operatorname{rk} A$, e $A\binom{1}{1} = \binom11$, $A\binom{1}{-1} = 0$, como
prescrito. Este sentido inverso --- desenhar a matriz na \omterm{def:b1:vspaces:free}{base} boa e
depois conjugar de volta --- é como as matrizes de rotação, de
reflexão e de \omterm{def:b1:linmaps:projection}{projeção} são de fato produzidas na prática.
\end{example}

\begin{theorem}[Forma normal do posto]\label{thm:b1:matrices:rank}
O \emph{posto} de uma matriz (o posto das suas colunas,
equivalentemente o da \omterm{def:b1:linmaps:def}{aplicação linear} associada) é o único
invariante da equivalência: toda $A \in \mathcal{M}_{n,p}$ de posto $r$ é equivalente a
\[
J_r = \begin{pmatrix} I_r & 0 \\ 0 & 0 \end{pmatrix},
\]
e $\operatorname{rk}(A^{\mathsf T}) = \operatorname{rk}(A)$: o posto por linhas é igual ao posto por colunas.
\end{theorem}

\begin{proof}
Seja $u \colon E \to F$ de posto $r$. Escolha um \omterm{def:b1:vspaces:sum}{suplementar} $S$ de $\ker u$
($\dim S = r$, o \cref{thm:b1:linmaps:ranknullity}) com \omterm{def:b1:vspaces:free}{base} $(e_1, \dots, e_r)$, completada por uma \omterm{def:b1:vspaces:free}{base} de $\ker u$
numa \omterm{def:b1:vspaces:free}{base} de $E$; as imagens $f_i = u(e_i)$, $i \leq r$, formam uma \omterm{def:b1:vspaces:free}{base} de $\operatorname{im} u$
(a restrição é um isomorfismo), completada numa \omterm{def:b1:vspaces:free}{base} de $F$. Nestas
\omterm{def:b1:vspaces:free}{bases} a matriz de $u$ é exatamente $J_r$. Logo $A = Q J_r P^{-1}$ para $P, Q$
invertíveis.

Transpondo: $A^{\mathsf T} = (P^{-1})^{\mathsf T} J_r^{\mathsf T} Q^{\mathsf T}$ com $J_r^{\mathsf T}$ da mesma forma (posto $r$) e os fatores
externos invertíveis (a \omterm{def:b1:matrices:transpose}{transposta} de uma invertível é invertível, a
partir de $(AB)^{\mathsf T} = B^{\mathsf T}A^{\mathsf T}$ aplicado a $AA^{-1} = I$): $\operatorname{rk} A^{\mathsf T} = r$.
\end{proof}

\section{Operações sobre linhas}

\begin{method}[Eliminação de Gauss em matrizes]\label{met:b1:matrices:gauss}
As três \emph{operações elementares sobre linhas}\index{operações
sobre linhas} --- trocar duas linhas, multiplicar uma linha por
$\lambda \neq 0$, somar a uma linha um múltiplo de outra --- não alteram o
posto (cada uma é uma multiplicação à esquerda por uma matriz
invertível). Algoritmo: crie um pivô (a entrada não nula mais à
esquerda), zere a sua coluna abaixo, passe à linha e à coluna
seguintes; o número de pivôs da forma escalonada resultante é o
posto.

\emph{Cálculo da inversa:} execute o algoritmo no bloco $(A \mid
I_n)$ até
que o bloco da esquerda se torne $I_n$ (possível se e somente se $A$
é invertível); o bloco da direita é então $A^{-1}$ --- de fato, o
produto das matrizes elementares usadas é igual a $A^{-1}$.
\end{method}

\begin{example}\label{ex:b1:matrices:inverse}
$A = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$: reduza $(A \mid
I_2)$:
\[
\begin{pmatrix} 1 & 2 & 1 & 0\\ 3 & 4 & 0 & 1 \end{pmatrix}
\to
\begin{pmatrix} 1 & 2 & 1 & 0\\ 0 & -2 & -3 & 1 \end{pmatrix}
\to
\begin{pmatrix} 1 & 0 & -2 & 1\\ 0 & 1 & \tfrac32 & -\tfrac12
\end{pmatrix},
\]
(operações: $L_2 \leftarrow L_2 - 3L_1$; depois $L_1 \leftarrow L_1
+ L_2$, $L_2 \leftarrow -\frac12 L_2$). Logo $A^{-1} =
\begin{pmatrix} -2 & 1 \\ \tfrac32 & -\tfrac12\end{pmatrix}$.
\emph{Verificação:} $AA^{-1} = I_2$.
\end{example}

\begin{example}[Posto com um parâmetro, só por linhas]
\label{ex:b1:matrices:rankparameter}
Para $m \in \R$, o posto de $M_m = \begin{pmatrix} 1 & 1 & m\\ 1 & m & 1\\ m & 1 &
1\end{pmatrix}$. Reduza: $L_2 \leftarrow L_2 - L_1$ e $L_3
\leftarrow L_3 - mL_1$ dão as linhas
\[
(1,\ 1,\ m), \qquad (0,\ m - 1,\ 1 - m), \qquad
(0,\ 1 - m,\ 1 - m^2).
\]
\emph{Caso $m = 1$}: as duas últimas linhas se anulam --- um pivô,
$\operatorname{rk} M_1 = 1$ (as três linhas originais eram iguais). \emph{Caso $m \neq 1$}:
multiplique $L_2$ por $\frac1{m-1}$ e $L_3$ por $\frac1{1-m}$ para obter $(0, 1, -1)$ e
$(0, 1, 1 + m)$, e depois $L_3 \leftarrow L_3 - L_2 = (0, 0, m + 2)$. Se $m = -2$: dois pivôs, posto $2$; caso
contrário três pivôs, posto $3$. Resumo:
\[
\operatorname{rk} M_m =
\begin{cases}
1 & m = 1,\\
2 & m = -2,\\
3 & \text{caso contrário}.
\end{cases}
\]
Os mesmos limiares sairão de um único cálculo de determinante no
\cref{ch:b1:det} (o \omterm{def:b1:poly:def}{polinômio} $-(m+2)(m-1)^2$ do \cref{exo:b1:det:7}) --- mas note o que a eliminação
dá e o determinante não: o \emph{valor} do posto nos casos
degenerados, e não apenas o fato de ele ter caído.
\end{example}

\begin{example}[Calcular potências]\label{ex:b1:matrices:powers}
$A = \begin{pmatrix} 1 & 1 \\ 0 & 1\end{pmatrix} = I + N$ com $N =
E_{12}$, $N^2 = 0$. Como $I$ e $N$ comutam, o teorema
binomial (a \cref{prop:b1:structures:binomial}) trunca:
\[
A^k = I + kN = \begin{pmatrix} 1 & k \\ 0 & 1 \end{pmatrix}
\qquad (k \in \N, \text{ e } k \in \Z \text{ usando } A^{-1} = I -
N).
\]
\end{example}

\begin{method}[Calcular $A^n$: as três vias]
\label{met:b1:matrices:powers}
\begin{enumerate}
  \item \emph{Via binomial}: se $A = \lambda I + N$ com $N$ nilpotente, o teorema
        binomial trunca (o \cref{ex:b1:matrices:powers}, o \cref{exo:b1:matrices:5}); ele se aplica porque
        $\lambda I$ comuta com tudo.
  \item \emph{Via polinomial}: encontre uma identidade polinomial
        satisfeita por $A$ (em dimensão $2$, sempre $A^2 = sA -
        pI$) e reduza
        $X^n$ \omterm{def:b1:complex:field}{módulo} ela; o problema de fim de semana abaixo
        constrói esta via por completo.
  \item \emph{Via da semelhança}: encontre um $P$ invertível com
        $P^{-1}AP = D$ simples (diagonal, ou deslocamento), calcule $D^n$, e
        desfaça: $A^n = P D^n P^{-1}$ (o \cref{thm:b1:matrices:conjugation}, o \cref{ex:b1:matrices:conjugationrun}); a busca sistemática de
        tal $P$ é a teoria da redução do segundo ano.
\end{enumerate}
Qualquer que seja a via, confira o resultado em $n = 0, 1, 2$: três testes
baratos que pegam quase todo deslize.
\end{method}

\begin{remark}[Armadilhas comuns: o preço da não comutatividade]
\label{rem:b1:matrices:pitfalls}
Toda identidade da álgebra escalar cuja demonstração reordena
fatores morre em $\mathcal{M}_n(K)$, $n \geq 2$.
\emph{Quadrados}: $(A + B)^2 = A^2 + AB + BA + B^2$, e o meio colapsa em $2AB$ apenas se
$AB = BA$ (o \cref{exo:b1:matrices:1}).
\emph{Potências de produtos}: $(AB)^k$ é $ABAB\cdots$, e não $A^kB^k$.
\emph{Divisores de zero}: $E_{12}E_{12} = 0$ com $E_{12} \neq 0$; por consequência,
\emph{nada de cancelamento}: $AB = AC$ implica $B = C$ apenas quando
$A$ é invertível (multiplique por $A^{-1}$ --- do lado correto).
\emph{\omterm{def:b1:matrices:transpose}{Traços}}: $\operatorname{tr}(AB) = \operatorname{tr}(BA)$ sempre, mas $\operatorname{tr}(AB) \neq
\operatorname{tr}A\operatorname{tr}B$ em geral (tome $A = B =
I_2$:
$2 \neq 4$), e $\operatorname{tr}(ABC) =
\operatorname{tr}(BCA)$ (cíclico), ao passo que $\operatorname{tr}(ACB)$ pode diferir.
\emph{As \omterm{def:b1:matrices:transpose}{transpostas} invertem}: $(AB)^{\mathsf T} = B^{\mathsf
T}A^{\mathsf T}$ --- esquecer a inversão é o
erro mais comum nos cálculos de ortogonalidade (o \cref{ch:b1:euclid}).
Na dúvida, teste qualquer identidade alegada em $E_{12}$ e
$E_{21}$: o menor par que não comuta refuta a maioria das fórmulas
falsas numa linha.
\end{remark}

\begin{remark}[Para onde vai o dicionário]
\label{rem:b1:matrices:whereused}
O dicionário matricial é usado em cada página restante deste
volume: o \cref{ch:b1:det} associa a cada matriz quadrada um único número que
decide a invertibilidade, e resolve $AX = B$ de modo sistemático;
o \cref{ch:b1:euclid} singulariza as matrizes que preservam comprimentos (as
matrizes ortogonais); e no \cref{ch:b1:multivar}, o comportamento de segunda ordem
de uma função de duas variáveis é uma matriz simétrica $2 \times 2$. O
\omterm{def:b1:matrices:transpose}{traço}, introduzido acima quase de passagem, torna-se um invariante
poderoso: os \cref{exo:b1:matrices:6,exo:b1:matrices:8} dão um primeiro gostinho, e o volume do segundo
ano de graduação constrói sobre ele a teoria dos autovalores. O
problema de fim de semana desenvolve o outro cavalo de batalha: as
identidades polinomiais satisfeitas por uma matriz, que transformam
o cálculo de $A^n$ numa \omterm{pb:b1:matrices:1}{recorrência linear} de dois termos.
\end{remark}

\begin{remark}[Perspectivas dentro do Livro 3]
\label{rem:b1:matrices:perspectives}
Três famílias de matrizes introduzidas aqui têm encontros marcados
mais adiante neste volume. As \emph{matrizes simétricas}
(o \cref{ex:b1:matrices:symsplit}) carregam os dados de segunda ordem das funções de duas
variáveis: o teste de Monge do \cref{ch:b1:multivar} é um enunciado sobre o
comportamento de sinal de uma matriz simétrica $2\times2$, e o seu
determinante $rt - s^2$ é calculado pela maquinaria do \cref{ch:b1:det}. As
\emph{matrizes ortogonais} ($A^{\mathsf T}A = I$) são as isometrias do \cref{ch:b1:euclid}, onde
a \omterm{def:b1:matrices:transpose}{transposta} finalmente adquire o seu significado geométrico: é a
sombra algébrica do produto interno. As \emph{matrizes invertíveis}
encontram o seu teste prático no \cref{ch:b1:det} --- um único número, $\det A \neq 0$
--- fechando a busca que este capítulo abriu com a redução por
linhas. O \omterm{def:b1:matrices:transpose}{traço} e o determinante viajam então como o par invariante
$(s, p)$ do problema de fim de semana, até a teoria dos autovalores
do segundo ano.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:matrices:1}
Sejam $A = \begin{pmatrix} 1 & 2 \\ 0 & 1 \end{pmatrix}$ e $B =
\begin{pmatrix} 0 & 1 \\ 1 & 0\end{pmatrix}$. Calcule $AB$, $BA$, $A^2 - B^2$ e $(A+B)(A-B)$; explique por
que as duas últimas diferem.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:2}
Calcule o posto de
\[
M = \begin{pmatrix}
1 & 2 & 3\\
2 & 4 & 6\\
1 & 1 & 1
\end{pmatrix},
\qquad
N = \begin{pmatrix}
1 & 1 & 0 & 2\\
0 & 1 & 1 & 1\\
1 & 2 & 1 & 3
\end{pmatrix}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:3}
Inverta, por redução de linhas, $A = \begin{pmatrix} 1 & 0 & 1\\ 2 & 1 &
1\\ 1 & 1 & 1 \end{pmatrix}$, e confira num produto.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:4}
Escreva a matriz, na \omterm{def:b1:vspaces:free}{base} canônica de $\R_2[X]$, do endomorfismo
$u(P) = P(X + 1)$. Explique, sem calcular, por que ela é invertível, e dê a
matriz de $u^{-1}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:5}
Seja $A = \begin{pmatrix} 2 & 1 \\ 0 & 2\end{pmatrix}$. Escreva $A =
2I + N$, calcule $N^2$, e deduza $A^k$ para todo
$k \in \N$ pelo teorema binomial.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:6}
Demonstre que não existem matrizes $A, B \in \mathcal{M}_n(K)$ (com $K = \R$ ou $\C$) tais
que $AB - BA = I_n$. \emph{(Tome \omterm{def:b1:matrices:transpose}{traços}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:7}
Uma matriz $A$ é \emph{nilpotente} quando $A^m = 0$ para algum $m$.
Demonstre que $I - A$ é então invertível, com
\[
(I - A)^{-1} = I + A + A^2 + \dots + A^{m-1} .
\]
Aplicação: inverta $\begin{pmatrix} 1 & 2 & 3\\ 0 & 1 & 2\\ 0 & 0 &
1\end{pmatrix}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:8}
Seja $A \in \mathcal{M}_n(\R)$ satisfazendo $A^2 = A$ (idempotente). Demonstre que
$\operatorname{tr} A = \operatorname{rk} A$. \emph{(Interprete $A$ como uma \omterm{def:b1:linmaps:projection}{projeção} e escolha uma \omterm{def:b1:vspaces:free}{base}
adaptada; o \cref{thm:b1:matrices:conjugation} diz que o \omterm{def:b1:matrices:transpose}{traço} não depende da \omterm{def:b1:vspaces:free}{base}, pois $\operatorname{tr}(P^{-1}MP) =
\operatorname{tr} M$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:9}
Seja $J \in \mathcal{M}_n(\R)$ a matriz com todas as entradas iguais a $1$. Calcule
$J^2$ e deduza, para $a, b \in \R$, a condição de invertibilidade de $M
= aI + bJ$
junto com $M^{-1}$ \emph{(procure uma inversa da mesma forma
$\alpha I + \beta J$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:10}
(Desigualdades de posto) Para $A, B \in \mathcal{M}_n(K)$, demonstre
\[
\operatorname{rk}(A + B) \leq \operatorname{rk} A +
\operatorname{rk} B,
\qquad
\operatorname{rk}(AB) \geq \operatorname{rk} A + \operatorname{rk}
B - n .
\]
\emph{(Para a segunda --- a desigualdade de Sylvester --- aplique o
teorema do núcleo e da \omterm{def:b1:linmaps:kerim}{imagem} à restrição da \omterm{def:b1:logic:map}{aplicação} de $A$ a
$\operatorname{im} B$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:11}
Seja $D = \operatorname{diag}(d_1, \dots, d_n)$ com os $d_i$ \emph{dois a dois distintos}.
\begin{enumerate}
  \item Demonstre que uma matriz $A$ comuta com $D$ se e somente se
        $A$ é diagonal. \emph{(Compare as entradas $(i,j)$ de $AD$
        e de $DA$.)}
  \item Deduza o \emph{centro} de $\mathcal{M}_n(K)$: as matrizes que comutam
        com \emph{toda} matriz são exatamente as matrizes escalares
        $\lambda I_n$. \emph{(Teste contra $D$, e depois contra as matrizes
        $E_{ij}$.)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:12}
(Matrizes de posto um) Seja $A \in \mathcal{M}_n(K)$, $A \neq 0$.
\begin{enumerate}
  \item Demonstre que $\operatorname{rk} A = 1$ se e somente se $A =
        CL$ para uma coluna
        não nula $C \in \mathcal{M}_{n,1}$ e uma linha não nula $L \in \mathcal{M}_{1,n}$.
  \item Para tal $A$, demonstre $A^2 = (\operatorname{tr} A)\,A$; deduza que uma matriz de
        posto um é nilpotente se e somente se o seu \omterm{def:b1:matrices:transpose}{traço} é nulo.
  \item Se $\operatorname{tr} A \neq -1$, demonstre que $I_n + A$ é invertível com
        \[
        (I_n + A)^{-1} = I_n - \frac{1}{1 + \operatorname{tr}
        A}\,A ,
        \]
        e que $I_n + A$ \emph{não} é invertível quando $\operatorname{tr} A = -1$.
        \emph{(Encontre um vetor aniquilado por $I_n + A$.)}
\end{enumerate}
\end{exercise}

\section{Problema: potências de uma matriz por divisão polinomial}

\begin{problem}\label{pb:b1:matrices:1}
Calcular $A^{100}$ entrada a entrada é desesperador; calculá-lo
através de uma identidade polinomial satisfeita por $A$ leva três
linhas. Este problema constrói o método do zero: a divisão
euclidiana de $X^n$, a identidade $A^2 - sA + pI = 0$ verificada por toda matriz
$2
\times 2$ (o teorema de Cayley--Hamilton em dimensão $2$), e o
dicionário entre potências de matrizes e \omterm{pb:b1:matrices:1}{recorrências lineares} ---
com os \omterm{pb:b1:matrices:1}{números de Fibonacci} como exemplo condutor.
\index{teorema de Cayley--Hamilton (dimensão 2)}
\index{números de Fibonacci}\index{recorrência linear}

\medskip
\noindent\textbf{Parte I --- O cálculo dos restos.} Fixe $s, p
\in K$ e
$D = X^2 - sX + p$.
\begin{enumerate}
  \item Justifique que para cada $n \in \N$ existem únicos $Q_n
        \in K[X]$ e $(a_n, b_n) \in K^2$
        com
        \[
        X^n = Q_n\,D + a_n X + b_n ,
        \]
        e calcule $(a_0, b_0)$ e $(a_1, b_1)$.
  \item Multiplicando por $X$ e dividindo de novo, estabeleça as
        recorrências
        \[
        a_{n+1} = s\,a_n + b_n,
        \qquad
        b_{n+1} = -p\,a_n ,
        \]
        e deduza $a_{n+2} = s\,a_{n+1} - p\,a_n$: a sequência dos coeficientes obedece à
        \omterm{pb:b1:matrices:1}{recorrência linear} associada a $D$.
  \item Suponha que $D$ tem duas raízes distintas $\lambda \neq \mu$. Avaliando
        a identidade da divisão, demonstre
        \[
        a_n = \frac{\lambda^n - \mu^n}{\lambda - \mu},
        \qquad
        b_n = \frac{\lambda\mu^n - \mu\lambda^n}{\lambda - \mu} .
        \]
  \item Suponha $D = (X - \lambda)^2$. Usando a \omterm{def:b1:derivative:def}{derivada} da identidade da divisão,
        demonstre $a_n = n\lambda^{n-1}$ e $b_n = (1 - n)\lambda^{n}$.
  \item Mostre que substituir uma matriz fixa $M \in
        \mathcal{M}_k(K)$ em \omterm{def:b1:poly:def}{polinômios}
        respeita somas e produtos: $(PQ)(M) = P(M)\,Q(M)$. Deduza que se $D(M) =
        0$, então
        \[
        M^n = a_n\,M + b_n\,I \qquad (n \in \N).
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Dimensão 2: \omterm{def:b1:matrices:transpose}{traço}, número
determinante, Cayley--Hamilton.} Para $A = \begin{pmatrix} a & b\\ c &
d\end{pmatrix}$ ponha $s = a + d = \operatorname{tr} A$ e $p = ad
- bc$ (o
número que o \cref{ch:b1:det} chamará de determinante).
\begin{enumerate}[resume]
  \item Verifique por cálculo direto a \emph{identidade de
        Cayley--Hamilton em dimensão $2$}:
        \[
        A^2 - s\,A + p\,I_2 = 0 .
        \]
  \item Demonstre por desenvolvimento direto que $p$ é
        multiplicativo: com a notação óbvia, $p(AB) = p(A)\,p(B)$. Mostre então:
        $A$ é invertível se e somente se $p \neq 0$, e nesse caso
        \[
        A^{-1} = \frac1p\,\bigl(s\,I_2 - A\bigr).
        \]
  \item Seja $A = \begin{pmatrix} 1 & 1\\ 0 & 2\end{pmatrix}$. Calcule $s$, $p$, as raízes de $D$, e deduza
        uma fórmula \omterm{def:b1:topology:closed}{fechada} para $A^n$; confira-a contra um cálculo
        direto de $A^2$.
  \item Seja $A = \begin{pmatrix} 3 & 1\\ -1 & 1\end{pmatrix}$. Mostre que $D$ tem uma raiz dupla e calcule
        $A^n$; confira em $n = 2$.
  \item Seja $F = \begin{pmatrix} 1 & 1\\ 1 & 0\end{pmatrix}$ e defina os \omterm{pb:b1:matrices:1}{números de Fibonacci} por $F_0 = 0$,
        $F_1 = 1$, $F_{n+2} = F_{n+1} + F_n$. Demonstre
        \[
        F^n = \begin{pmatrix} F_{n+1} & F_n\\ F_n &
        F_{n-1}\end{pmatrix} \quad (n \geq 1),
        \]
        deduza a fórmula de Binet $F_n = \dfrac{\varphi^n -
        \psi^n}{\sqrt5}$, onde $\varphi = \frac{1 + \sqrt5}2$, $\psi = \frac{1 - \sqrt5}2$, e,
        usando a questão 7, a identidade de Cassini $F_{n+1}F_{n-1} - F_n^2 = (-1)^n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- \omterm{pb:b1:matrices:1}{Recorrências lineares},
estruturalmente.} Fixe $s, p \in K$ com $p \neq 0$, e seja $E_D$ o \omterm{def:b1:logic:sets}{conjunto} das
sequências com $u_{n+2} = s\,u_{n+1} - p\,u_n$ para todo $n$.
\begin{enumerate}[resume]
  \item Mostre que $E_D$ é um \omterm{def:b1:vspaces:def}{espaço vetorial} de dimensão $2$
        (adapte o \cref{exo:b1:findim:10}).
  \item Mostre que a sequência $(a_n)$ da Parte I é o elemento de
        $E_D$ com valores iniciais $0, 1$, e que todo $u
        \in E_D$
        satisfaz
        \[
        u_n = u_1\,a_n + u_0\,b_n \qquad (n \in \N),
        \]
        com $(b_n)$ como na Parte I: os restos da divisão resolvem
        \emph{todas} as recorrências de uma só vez.
  \item Se $\lambda \neq \mu$ são as raízes de $D$, mostre que $\bigl((\lambda^n), (\mu^n)\bigr)$ é uma \omterm{def:b1:vspaces:free}{base}
        de $E_D$; se $D = (X-\lambda)^2$ com $\lambda \neq 0$, mostre que $\bigl((\lambda^n), (n\lambda^n)\bigr)$ é uma.
  \item Resolva completamente: $u_{n+2} = u_{n+1} + 6u_n$, $u_0 = 1$, $u_1 = 8$; confira
        a resposta em $u_2$ e $u_3$.
  \item Seja $C = \begin{pmatrix} 0 & 1\\ -p & s\end{pmatrix}$ (a \emph{matriz companheira} de $D$). Mostre que
        \[
        \begin{pmatrix} u_{n}\\ u_{n+1}\end{pmatrix}
        = C^n \begin{pmatrix} u_0\\ u_1\end{pmatrix}
        \quad (u \in E_D),
        \]
        e que $\operatorname{tr} C = s$ e $p(C) = p$: a recorrência e a matriz carregam o
        mesmo \omterm{def:b1:poly:def}{polinômio} $D$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Grau três.} Seja $D_3 = X^3 -
\alpha X^2 - \beta X - \gamma$ e
\[
C_3 = \begin{pmatrix} 0 & 1 & 0\\ 0 & 0 & 1\\ \gamma & \beta &
\alpha \end{pmatrix}.
\]
\begin{enumerate}[resume]
  \item Mostre que $D_3(C_3) = 0$. \emph{(Calcule as imagens dos vetores da
        \omterm{def:b1:vspaces:free}{base} canônica pelas potências de $C_3$: a \omterm{def:b1:logic:map}{aplicação} de
        $C_3$ leva $e_1 \mapsto \dots \mapsto$ uma combinação forçada pela última
        linha.)}
  \item Mostre que se $D_3$ tem três raízes distintas $\lambda_1,
        \lambda_2, \lambda_3$, o
        resto $R_n$ da divisão de $X^n$ por $D_3$ é o
        \emph{interpolante de Lagrange} dos valores $\lambda_i^n$ nos nós
        $\lambda_i$ (o \cref{thm:b1:poly:lagrange}); deduza que toda entrada de $C_3^{\,n}$ é uma
        combinação linear fixa de $\lambda_1^n, \lambda_2^n, \lambda_3^n$.
  \item Resolva: $u_{n+3} = 2u_{n+2} + u_{n+1} - 2u_n$ com $u_0 =
        0$, $u_1 = 1$, $u_2 = 1$.
        \emph{(Fatore $D_3 = (X - 1)(X
        + 1)(X - 2)$.)} Confira em $u_3$.
  \item Calcule o resto de $X^n$ módulo $(X -
        \lambda)^3$ \emph{(expansão de
        Taylor de $X^n$ em $\lambda$)}, e deduza uma fórmula para
        $(\lambda I +
        N)^n$ quando $N^3 = 0$ e $N$ comuta com tudo o que estiver
        à vista; confira-a contra o teorema binomial.
  \item Mostre que, para $D_3$ com raízes distintas, a solução
        geral da recorrência de ordem $3$ é $u_n = c_1
        \lambda_1^n + c_2\lambda_2^n + c_3\lambda_3^n$: demonstre que
        as três sequências geométricas formam uma \omterm{def:b1:vspaces:free}{base} do espaço de
        soluções. \emph{(Para a liberdade, avalie uma combinação
        nula em $n = 0, 1, 2$ e reconheça um sistema de interpolação nos
        nós distintos $\lambda_i$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos de Fibonacci, e síntese.}
\begin{enumerate}[resume]
  \item Demonstre $F_1 + F_2 + \dots + F_n = F_{n+2} - 1$.
  \item De $F^{m+n} = F^m F^n$, obtenha a fórmula de adição
        \[
        F_{m+n} = F_{m+1}F_n + F_m F_{n-1},
        \]
        e deduza $F_{2n} = F_n(F_{n+1} + F_{n-1})$.
  \item Demonstre que $F_n$ é o inteiro mais próximo de $\varphi^n/\sqrt5$ para
        todo $n \geq 0$.
  \item Seja $t_n = \operatorname{tr}(F^n) = F_{n+1} + F_{n-1}$ (os \emph{números de Lucas} $L_n$). Mostre que
        $t_{n+2} =
        t_{n+1} + t_n$, $t_1 = 1$, $t_2 = 3$, que $L_n =
        \varphi^n + \psi^n$, e recupere $F_{2n} = F_n L_n$.
  \item Síntese, em quatro frases: por que as potências de uma
        matriz $2
        \times 2$ vivem no plano $\operatorname{Vect}(I, A)$ de $\mathcal{M}_2(K)$ (qual argumento
        de dimensão garante uma identidade quadrática, e qual
        identidade explícita a Parte II produziu); como a divisão
        euclidiana converte a exponenciação numa recorrência de
        dois termos; qual enunciado deste problema é o caso $n =
        2$
        de um teorema válido em todas as dimensões (nomeie-o, e
        diga onde ele é demonstrado nesta \omterm{def:b1:series:def}{série}); e o que a
        construção da matriz companheira acrescenta ao quadro.
\end{enumerate}
\end{problem}
