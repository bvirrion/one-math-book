\chapter{Números Reais}\label{ch:b1:reals}

Toda a análise repousa sobre uma propriedade que distingue $\R$ de
$\Q$: todo \omterm{def:b1:logic:sets}{conjunto} não vazio limitado superiormente tem uma
\emph{menor} \omterm{def:b1:reals:bounds}{cota superior}. Este capítulo a enuncia com precisão, extrai
as suas primeiras consequências --- a \omterm{thm:b1:reals:archimedes}{propriedade arquimediana}, a
função \omterm{thm:b1:reals:floor}{parte inteira}, a densidade dos racionais e dos irracionais --- e
monta o vocabulário (sup, inf, máx, mín) usado constantemente a partir
do \cref{ch:b1:seq}.

\section{A propriedade do supremo}

\begin{definition}[Cotas, sup e inf]\label{def:b1:reals:bounds}
Seja $A \subseteq \R$ não vazio. Um real $M$ é uma \emph{cota
superior}\index{cota superior} de $A$ quando $a \leq M$ para todo $a \in A$;
$A$ é \emph{limitado superiormente} quando tem uma cota superior
(analogamente por baixo, com cotas inferiores; \emph{limitado} significa
as duas coisas). Um \emph{máximo} de $A$ é uma cota superior que
pertence a $A$.

O \emph{supremo} $\sup A$\index{supremo} é a menor cota superior
de $A$, quando existe; o \emph{ínfimo} $\inf A$\index{ínfimo} é
a maior cota inferior.
\end{definition}

\begin{theorem}[Axioma da completude de $\R$]\label{thm:b1:reals:sup}
$\R$ é um \omterm{def:b1:structures:field}{corpo} ordenado que contém $\Q$ no qual \emph{todo
subconjunto não vazio limitado superiormente tem \omterm{def:b1:reals:bounds}{supremo}}.\index{completude
de R@completude de $\R$}
\end{theorem}
\admitted

\begin{remark}
Tomamos isso como axioma definidor de $\R$; construir um modelo (por
cortes de Dedekind ou por sequências de Cauchy de racionais) e
demonstrar a sua unicidade é honesto, mas longo, e fica para estudo
posterior. Note que $\Q$ falha a propriedade: $\{x \in \Q : x^2 < 2\}$ é limitado
superiormente, mas não tem menor \omterm{def:b1:reals:bounds}{cota superior} \emph{em $\Q$} --- o seu
candidato, $\sqrt 2$, está faltando (\cref{ex:b1:logic:sqrt2}). Passando aos
opostos ($\sup(-A) = -\inf A$), todo \omterm{def:b1:logic:sets}{conjunto} não vazio limitado inferiormente
tem \omterm{def:b1:reals:bounds}{ínfimo}.
\end{remark}

\begin{proposition}[A caracterização por $\varepsilon$]\label{prop:b1:reals:epsilon}
Seja $A \neq \emptyset$ limitado superiormente e $s \in \R$. Então $s =
\sup A$ se, e somente se,
\begin{enumerate}
  \item $s$ é \omterm{def:b1:reals:bounds}{cota superior}: $\forall a \in A$, $a \leq s$; e
  \item nada menor o é: $\forall \varepsilon > 0$, $\exists a \in
        A$, $a > s - \varepsilon$.
\end{enumerate}
\end{proposition}

\begin{proof}
Se $s = \sup A$: (1) vale por definição e, para (2), $s -
\varepsilon < s$ não é \omterm{def:b1:reals:bounds}{cota superior}, o que é exatamente a
existência de $a > s - \varepsilon$. Reciprocamente, (1) diz que $s$ é
\omterm{def:b1:reals:bounds}{cota superior}; (2) diz que nenhum $t < s$ é \omterm{def:b1:reals:bounds}{cota superior} (tome $\varepsilon
= s - t$): $s$ é a menor delas.
\end{proof}

\begin{example}\label{ex:b1:reals:supexamples}
$\sup \intoo{0}{1} = 1$, não atingido (sem máximo);
$\sup \intcc{0}{1} = 1 = \max$. Para $A = \{1 - \frac 1n : n \in
\N^*\}$: $\sup A = 1$, não atingido; $\inf A = \min A = 0$. Um máximo,
quando existe, é o \omterm{def:b1:reals:bounds}{supremo}; toda a razão de ser do $\sup$ é dispor de
um substituto quando o máximo não existe.
\end{example}

\begin{example}[Calculando supremos na prática]\label{ex:b1:reals:supcompute}
Dois exercícios completos da \cref{prop:b1:reals:epsilon}.

\emph{O \omterm{def:b1:logic:sets}{conjunto} $A = \{x + \frac1x : x > 0\}$.} Para todo $x > 0$,
$x + \frac1x - 2 = \frac{(\,\sqrt x - 1/\sqrt x\,)^2}{1} \geq 0$,
de modo que $2$ é cota inferior; e $2 = 1 + \frac11 \in A$: portanto,
$\inf A = \min A = 2$, atingido em $x = 1$. Por cima, $A$ é
ilimitado ($x + \frac1x > x$ pode exceder qualquer $M$ pelo
\cref{thm:b1:reals:archimedes}): $\sup A$ não existe em $\R$
(vale $+\infty$ em $\overline\R$).

\emph{O \omterm{def:b1:logic:sets}{conjunto} $B = \bigl\{\frac{m}{m + n} : m, n \in
\N^*\bigr\}$.} Todo elemento está em $\intoo{0}{1}$, de modo que $0$ e
$1$ são cotas. Nenhuma delas é atingida: $\frac{m}{m+n} = 1$ forçaria
$n = 0$. Para o \omterm{def:b1:reals:bounds}{supremo}, congele $n = 1$ e faça $m$ crescer:
$\frac{m}{m+1} = 1 - \frac{1}{m+1} > 1 - \varepsilon$ assim que
$m + 1 > \frac1\varepsilon$ (Arquimedes): $\sup B = 1$.
Simetricamente ($m = 1$, $n$ grande), $\inf B = 0$. A ideia de
fechamento: para fixar um \omterm{def:b1:reals:bounds}{supremo}, basta um \emph{caminho a um
parâmetro} bem escolhido dentro do \omterm{def:b1:logic:sets}{conjunto} --- aqui o caminho $n = 1$
--- e a caracterização por $\varepsilon$ nada mais exige.
\end{example}

\begin{example}[O espelho do ínfimo]\label{ex:b1:reals:infmirror}
O \omterm{def:b1:reals:bounds}{ínfimo} tem a sua própria caracterização por $\varepsilon$, obtida
da \cref{prop:b1:reals:epsilon} por meio de $\inf A = -\sup(-A)$:
$i = \inf A$ se, e somente se, $i$ limita $A$ inferiormente e, para todo
$\varepsilon > 0$, algum $a \in A$ tem $a < i + \varepsilon$. Um
exercício com as duas cotas ao mesmo tempo: seja
\[
A = \Bigl\{(-1)^n + \frac1n : n \in \N^*\Bigr\}
= \Bigl\{0,\ \tfrac32,\ -\tfrac23,\ \tfrac54,\ -\tfrac45,\
\dots\Bigr\} .
\]
Os índices pares dão $1 + \frac1n \leq \frac32$, com igualdade em
$n = 2$: como também os valores de índice ímpar são $\leq 0 <
\frac32$, obtemos $\sup A = \max A = \frac32$. Os índices ímpares dão
$-1 + \frac1n > -1$, decrescendo rumo a $-1$: todo elemento de
$A$ é $> -1$, e $-1 + \varepsilon$ é superado por $-1 +
\frac1n$
para $n$ ímpar com $n > \frac1\varepsilon$: $\inf A = -1$, não
atingido. Um só \omterm{def:b1:logic:sets}{conjunto}, com os quatro comportamentos à mostra: um
\omterm{def:b1:reals:bounds}{supremo} que é máximo e um \omterm{def:b1:reals:bounds}{ínfimo} que não é mínimo.
\end{example}

\begin{remark}[Armadilhas frequentes com sup e inf]
Quatro erros respondem pela maioria dos pontos perdidos. (i) \emph{Confundir
$\sup$ e $\max$}: $\sup A$ não precisa pertencer a $A$; escreva $\max$
apenas depois de exibir um elemento de $A$ que seja \omterm{def:b1:reals:bounds}{cota superior}.
(ii) \emph{Passar desigualdades estritas ao \omterm{def:b1:reals:bounds}{supremo}}: se $a <
b$ para todo $a \in A$, só se pode concluir $\sup A \leq b$ ---
testemunha: $A = \intoo{0}{1}$, $b = 1$. (iii) \emph{Escrever $\sup A$
antes de verificar a legalidade}: o símbolo exige $A$ não vazio e
limitado superiormente (\cref{met:b1:reals:supproofs}); $\sup \emptyset$
e $\sup \N$ não estão definidos em $\R$ (as convenções de
$\overline\R$ são um ato separado e explícito). (iv) \emph{Operações
com \omterm{def:b1:logic:sets}{conjuntos}}: $\sup(A \cup B) = \max(\sup A, \sup B)$ sempre, mas
nada geral vale para $A \cap B$ --- ele pode ser vazio e, mesmo quando
não é, $\sup(A \cap B)$ pode ficar muito abaixo de
$\min(\sup A, \sup B)$: tome $A = \{0, 2\}$ e $B = \{0, 3\}$,
em que $\sup(A \cap B) = 0$.
\end{remark}

\begin{example}[Conjuntos finitos têm máximo --- um lema usado em silêncio]\label{ex:b1:reals:finitemax}
Todo $F \subseteq \R$ finito e não vazio tem máximo (e
mínimo). Indução no número de elementos: um \omterm{def:b1:logic:sets}{conjunto} unitário
$\{a\}$ tem $\max = a$; se a afirmação vale para \omterm{def:b1:logic:sets}{conjuntos} de $n$
elementos e $F$ tem $n + 1$ elementos, tome $a \in F$ qualquer: o
\omterm{def:b1:logic:sets}{conjunto} $F
\setminus \{a\}$ tem um máximo $m$, e $\max F$ é $m$ se $a
\leq m$, e $a$ caso contrário. Nenhuma completude está envolvida --- é
pura ordem mais indução, válida já em $\Q$ --- e, ainda assim, o lema
merece um enunciado honesto, porque as demonstrações que virão o
invocam silenciosamente: a construção da \omterm{thm:b1:reals:floor}{parte inteira} abaixo (``um
\omterm{def:b1:logic:sets}{conjunto} de inteiros preso a um \omterm{prop:b1:reals:intervals}{intervalo} finito tem maior elemento''),
toda estimativa $\max(\abs{u_0}, \dots, \abs{u_{N-1}}, \dots)$ do
\cref{ch:b1:seq}, todo ``tome o maior dos finitos $\delta$'' do
\cref{ch:b1:continuity}. É nos \omterm{def:b1:logic:sets}{conjuntos} infinitos que os
máximos morrem e os \omterm{def:b1:reals:bounds}{supremos} assumem: este capítulo existe para o caso
infinito.
\end{example}

\begin{theorem}[Propriedade arquimediana]\label{thm:b1:reals:archimedes}
Para todo $x \in \R$ existe $n \in \N$ com $n > x$. Equivalentemente:
para todos $\varepsilon > 0$ e $y > 0$, algum múltiplo $n\varepsilon$
excede $y$.\index{propriedade arquimediana}
\end{theorem}

\begin{proof}
Suponha que não: algum $x$ é \omterm{def:b1:reals:bounds}{cota superior} de $\N$. Então $s = \sup \N$
existe (\cref{thm:b1:reals:sup}). Pela
\cref{prop:b1:reals:epsilon}~(2) com $\varepsilon = 1$, existe $n
\in \N$ com $n > s - 1$; mas então $n + 1 \in \N$ e $n + 1 > s$,
contradizendo que $s$ seja \omterm{def:b1:reals:bounds}{cota superior}. Para a segunda forma,
sejam $\varepsilon > 0$ e $y > 0$: a primeira forma aplicada a $x
= \frac{y}{\varepsilon}$ produz $n \in \N$ com $n >
\frac{y}{\varepsilon}$, e multiplicar por $\varepsilon > 0$
(o que preserva desigualdades estritas) dá $n\varepsilon > y$.
Reciprocamente, a segunda forma com $\varepsilon = 1$ e $y = x$
recupera a primeira para $x > 0$, e $n = 1$ trata $x \leq 0$:
os dois enunciados são estritamente equivalentes.
\end{proof}

\begin{example}[Arquimedes em ação]\label{ex:b1:reals:archimedeswork}
Três usos imediatos, constantemente necessários mais adiante. (i) \emph{Nenhum
real positivo fica abaixo de todos os $\frac1n$}: se $0 < \varepsilon$,
tome $n > \frac1\varepsilon$; então $\frac1n < \varepsilon$. Em
outras palavras, $\R$ não contém infinitésimos --- o informal
``$\frac1n$ fica arbitrariamente pequeno'' é exatamente este teorema.
(ii) \emph{Limiares explícitos}: quão grande deve ser $n$ para que
$\frac{1}{n^2} \leq 10^{-6}$? Basta que $n \geq 10^3$
--- Arquimedes garante que tais $n$ existem, e a álgebra
os localiza. (iii) \emph{As potências vencem qualquer cota}: $2^n \geq n +
1$ (indução), de modo que, para todo $M$, alguma potência de $2$
excede $M$: o crescimento geométrico usado para os diádicos no
\cref{exo:b1:reals:8}. A ideia de fechamento: a \omterm{thm:b1:reals:archimedes}{propriedade
arquimediana} é a licença por trás de toda frase do tipo ``tome
$n$ suficientemente grande'' --- daqui em diante usamos essa frase
livremente, e este exemplo é a sua justificativa de uma vez por todas.
\end{example}

\begin{theorem}[Função parte inteira]\label{thm:b1:reals:floor}
Para todo $x \in \R$ existe exatamente um inteiro, a
\emph{parte inteira}\index{parte inteira} $\lfloor x \rfloor$, com
\[
\lfloor x \rfloor \leq x < \lfloor x \rfloor + 1 .
\]
\end{theorem}

\begin{proof}
\emph{Existência.} O \omterm{def:b1:logic:sets}{conjunto} $E = \{k \in \Z : k \leq x\}$ é não vazio:
pelo \cref{thm:b1:reals:archimedes} existe $m \in \N$ com $m > -x$,
e então $-m < x$, de modo que $-m \in E$. Ele é limitado superiormente (por
qualquer inteiro $n > x$, que existe pela mesma razão), de sorte que,
sendo um \omterm{def:b1:logic:sets}{conjunto} de inteiros preso ao \omterm{prop:b1:reals:intervals}{intervalo} finito $\intint{-m}{n}$, tem
maior elemento $k = \max E$. Então $k \leq x$, e $k + 1 \notin E$
significa $x < k + 1$.

\emph{Unicidade.} Se $k$ e $k'$ satisfazem ambos as desigualdades,
então $k \leq x < k' + 1$ dá $k \leq k'$ e, simetricamente, $k'
\leq k$.
\end{proof}

\begin{example}[Partes inteiras na prática]\label{ex:b1:reals:floorpractice}
$\lfloor 3.7 \rfloor = 3$, $\lfloor 5 \rfloor = 5$ e $\lfloor
-3.7 \rfloor = -4$: a \omterm{thm:b1:reals:floor}{parte inteira} vai para \emph{baixo}, não em direção a $0$.
Duas consequências da unicidade no \cref{thm:b1:reals:floor} que
usaremos silenciosamente. Primeira, para $n \in \Z$,
\[
\lfloor x + n \rfloor = \lfloor x \rfloor + n ,
\]
porque $\lfloor x \rfloor + n$ é um inteiro que satisfaz as duas
desigualdades definidoras para $x + n$ --- e só um inteiro as satisfaz.
Segunda, $\lfloor \, \cdot \, \rfloor$ é não decrescente: se $x \leq
y$, então $\lfloor x \rfloor \leq x \leq y < \lfloor y \rfloor + 1$,
e um inteiro $< \lfloor y \rfloor + 1$ é $\leq \lfloor y
\rfloor$. Cuidado, porém: $\lfloor 2x \rfloor \neq 2\lfloor
x \rfloor$ em geral: $x = 0.6$ dá $\lfloor 1.2 \rfloor = 1
\neq 0 = 2\lfloor 0.6 \rfloor$.

O que \emph{é} verdade é uma identidade trabalhada que vale guardar (a
de Hermite, no seu caso mais simples): para todo real $x$,
\[
\lfloor x \rfloor + \Bigl\lfloor x + \frac12 \Bigr\rfloor
= \lfloor 2x \rfloor .
\]
Escreva $x = \lfloor x\rfloor + u$ com $u \in \intco{0}{1}$ e
separe dois casos. Se $u < \frac12$: o lado esquerdo é $\lfloor
x\rfloor + \lfloor x\rfloor = 2\lfloor x\rfloor$, e $2x =
2\lfloor x\rfloor + 2u$ com $2u \in \intco{0}{1}$, de modo que o lado
direito é $2\lfloor x\rfloor$ também. Se $u \geq \frac12$: o lado
esquerdo é $\lfloor x\rfloor + (\lfloor x\rfloor + 1)$, e $2u \in
\intco{1}{2}$ faz do lado direito $2\lfloor x\rfloor + 1$. A
ideia de fechamento: $\lfloor x + \frac12\rfloor$ é o
\emph{arredondamento} de $x$ ao inteiro mais próximo, de modo que a
identidade diz que \omterm{thm:b1:reals:floor}{parte inteira} mais arredondamento é igual à \omterm{thm:b1:reals:floor}{parte
inteira} do dobro --- e a separação em casos pela parte fracionária $u$ é
a técnica padrão por trás de toda identidade de \omterm{thm:b1:reals:floor}{parte inteira}
(o \cref{exo:b1:reals:2,exo:b1:reals:3} também se apoia nela).
\end{example}

\begin{theorem}[Densidade de $\Q$ e de $\R \setminus \Q$]\label{thm:b1:reals:density}
Entre dois reais quaisquer $x < y$ existem um racional e um
irracional.\index{densidade}
\end{theorem}

\begin{proof}
\emph{Um racional.} Pelo \cref{thm:b1:reals:archimedes}, tome $n \in
\N^*$
com $n > \frac{1}{y - x}$, de modo que $ny - nx > 1$. Seja $m = \lfloor
nx \rfloor + 1$. De um lado, $nx < \lfloor nx \rfloor + 1 = m$
(\cref{thm:b1:reals:floor}); do outro, $m = \lfloor nx \rfloor +
1 \leq nx + 1 < ny$. Dividindo por $n$: $x < \frac mn < y$.

\emph{Um irracional.} Aplique o ponto anterior ao par $x -
\sqrt 2 < y - \sqrt 2$: algum racional $q$ fica entre eles, e então
$q + \sqrt 2 \in \intoo{x}{y}$ é irracional (se $q + \sqrt 2$ fosse
racional, $\sqrt 2$ também seria).
\end{proof}

\begin{example}[Executando a demonstração da densidade]\label{ex:b1:reals:densityrun}
A demonstração é um algoritmo; executemo-lo em $x = 1.414$ e
$y = \sqrt 2$. Como $1.4142^2 = 1.99996164 < 2$, temos $\sqrt 2
> 1.4142$, de modo que $y - x > 0.0002$ e $\frac{1}{y - x} < 5000$: a
escolha $n = 5000$ é legítima. Então $nx = 7070$, de sorte que $m = \lfloor
7070 \rfloor + 1 = 7071$, e o racional produzido é
\[
\frac{m}{n} = \frac{7071}{5000} = 1.4142,
\qquad
1.414 < 1.4142 < \sqrt 2 .
\]
A ideia de fechamento: a demonstração precisa de $n$ apenas ligeiramente
maior que $\frac{1}{y-x}$, e devolve o primeiro múltiplo de $\frac 1n$
além de $x$. A densidade não é um milagre abstrato --- é uma divisão
longa disfarçada, tema desenvolvido a fundo no problema de fim de
semana (\cref{pb:b1:reals:1}).
\end{example}

\begin{remark}[Onde a completude é usada em seguida]
O \Cref{thm:b1:reals:sup} é o único axioma não algébrico deste
livro, e todo teorema de existência da análise é esse axioma
vestindo roupas diferentes: o teorema da convergência monótona
(\cref{ch:b1:seq}), o teorema de Bolzano--Weierstrass
(\cref{ch:b1:topology}), os teoremas do valor intermediário e de Weierstrass
(\cref{ch:b1:continuity}) e a própria definição da
integral como \omterm{def:b1:reals:bounds}{supremo} de somas inferiores (\cref{ch:b1:integration}).
O volume do terceiro ano de graduação constrói a teoria da medida e os
espaços de Hilbert sobre esse mesmo único axioma. Quando uma demonstração
dos capítulos seguintes produzir um número real do nada, procure o
\omterm{def:b1:reals:bounds}{supremo} escondido.
\end{remark}

\begin{remark}[Entre a discretude e a densidade]
$\Z$ e $\Q$ ocupam extremos opostos dentro de $\R$: em torno de cada
inteiro há um vão de comprimento $1$ sem nenhum outro
(discretude --- é isso que torna a \omterm{thm:b1:reals:floor}{parte inteira} bem definida),
ao passo que entre dois reais quaisquer há infinitos racionais
(densidade). Notavelmente, para \emph{\omterm{def:b1:structures:subgroup}{subgrupos} aditivos} de $\R$
não há nada intermediário: o \cref{exo:b1:reals:9} demonstra que
um tal \omterm{def:b1:structures:subgroup}{subgrupo} é ou da forma $\alpha\Z$ (discreto), ou
denso --- dicotomia que alimenta a densidade de $\{\sin n\}$ no
\cref{ch:b1:seq} e o monstro construtivo do
\cref{pb:b1:continuity:1}. \omterm{def:b1:logic:sets}{Conjuntos} gerais, é claro, misturam os
comportamentos livremente: $\Z \cup \Q\cap\intcc{0}{1}$ é discreto ao longe
e denso no meio.
\end{remark}

\begin{method}[Demonstrando igualdades com sup e inf]\label{met:b1:reals:supproofs}
Para demonstrar que $\sup A = s$: verifique que $s$ limita $A$ superiormente
e depois produza, para cada $\varepsilon > 0$ (ou para uma sequência $\varepsilon = \frac
1n$), um elemento de $A$ acima de $s - \varepsilon$. Para comparar
\omterm{def:b1:reals:bounds}{supremos}, use: $A \subseteq B \implies \sup A \leq \sup B$; e, para todos $a, b$:
$\sup(A + B) = \sup A + \sup B$, em que $A + B = \{a + b\}$
(\cref{exo:b1:reals:5}). Nunca escreva $\sup A$ antes de saber que $A$ é
não vazio e limitado superiormente.
\end{method}

\section{Intervalos}

\begin{proposition}[Caracterização dos intervalos]\label{prop:b1:reals:intervals}
Um subconjunto $I \subseteq \R$ é um \emph{intervalo}\index{intervalo} (um
dos tipos familiares $\intoo{a}{b}$, $\intcc{a}{b}$, $\intco{a}{b}$,
$\intoc{a}{b}$, semirretas, $\R$, $\emptyset$, \omterm{def:b1:logic:sets}{conjuntos} unitários) se, e
somente se, é \emph{convexo}:
\[
\forall x, y \in I,\ \forall z \in \R, \quad
x \leq z \leq y \implies z \in I .
\]
\end{proposition}

\begin{proof}
Todo tipo da lista é claramente convexo. Reciprocamente, seja $I$ convexo
e não vazio. Ponha $a = \inf I$ se $I$ é limitado inferiormente, e $a =
-\infty$ caso contrário; do mesmo modo, $b = \sup I$ ou $+\infty$.
Afirmamos que $\intoo{a}{b}
\subseteq I \subseteq \intcc{a}{b}$ (com as convenções óbvias em
$\pm\infty$). A segunda inclusão é a definição de cotas. Para
a primeira, seja $z \in \intoo{a}{b}$: como $z > a$, $z$ não é cota
inferior (ou $a = -\infty$), de modo que algum $x \in I$ tem $x < z$;
analogamente, algum $y \in I$ tem $y > z$; a convexidade põe $z \in I$.

Resta ler o tipo a partir da dupla inclusão
$\intoo{a}{b} \subseteq I \subseteq \intcc{a}{b}$: os \omterm{def:b1:logic:sets}{conjuntos}
espremidos entre um \omterm{prop:b1:reals:intervals}{intervalo} aberto e o seu fecho diferem de
$\intoo{a}{b}$ apenas pela presença ou ausência das extremidades
(finitas). Explicitamente: se $a, b \in \R$, as quatro possibilidades
para $(a \in I,\ b \in I)$ dão $\intoo{a}{b}$, $\intco{a}{b}$,
$\intoc{a}{b}$, $\intcc{a}{b}$ (incluindo os casos degenerados
$a = b$: \omterm{def:b1:logic:sets}{conjunto} unitário se $a \in I$); se $a = -\infty$ e $b \in
\R$, obtêm-se $\intoo{-\infty}{b}$ ou $\intoc{-\infty}{b}$;
simetricamente para $a \in \R$, $b = +\infty$; e $a = -\infty$,
$b = +\infty$ dá $I = \R$. Todo caso está na lista: pronto.
\end{proof}

\begin{remark}[Por que a convexidade é o teste certo]
A \omterm{def:b1:logic:statement}{proposição} converte uma definição \emph{geométrica} (uma lista
de dez formas) num teste \emph{lógico de uma linha}, e o teste
é o que de fato se usa: para demonstrar que um \omterm{def:b1:logic:sets}{conjunto} é um \omterm{prop:b1:reals:intervals}{intervalo},
nunca caçe qual das dez formas ele é --- verifique a convexidade
e deixe a \omterm{def:b1:logic:statement}{proposição} decidir o tipo. O teorema do valor
intermediário do \cref{ch:b1:continuity} será enunciado exatamente
assim (``a imagem contínua de um \omterm{prop:b1:reals:intervals}{intervalo} é um
\omterm{prop:b1:reals:intervals}{intervalo}''), e a sua demonstração produz a convexidade, não a
forma.
\end{remark}

\begin{remark}[A reta real estendida]
É conveniente adjungir dois símbolos e trabalhar em $\overline\R = \R
\cup \{-\infty, +\infty\}$, com as convenções $\sup A = +\infty$
quando $A$ não é limitado superiormente e $\sup \emptyset = -\infty$. Então
\emph{todo} subconjunto de $\R$ tem \omterm{def:b1:reals:bounds}{supremo} em $\overline\R$ --- um
conforto notacional usado livremente para limites no \cref{ch:b1:seq}.
\end{remark}

\begin{example}[Calculando em $\overline\R$]\label{ex:b1:reals:extendedcompute}
Com as convenções em vigor: $\sup \Z = +\infty$, $\inf \Z =
-\infty$; para $A = \{n + (-1)^n n : n \in \N\} = \{0, 4, 0, 8,
\dots\} \cup \{0\}$, $\sup A = +\infty$ (os termos pares $2n$ são
ilimitados) e $\inf A = \min A = 0$; e $\sup\emptyset =
-\infty \leq \inf\emptyset = +\infty$ --- o único \omterm{def:b1:logic:sets}{conjunto} cujo
\omterm{def:b1:reals:bounds}{supremo} é \emph{menor} que o seu \omterm{def:b1:reals:bounds}{ínfimo}, lembrete de que as
convenções são escolhidas para tornar o $\sup$ crescente e o $\inf$
decrescente em relação à inclusão:
\[
A \subseteq B \implies \sup A \leq \sup B
\quad\text{e}\quad \inf A \geq \inf B ,
\]
válido agora sem nenhuma ressalva de não vacuidade. O que as convenções
\emph{não} fornecem é aritmética: $+\infty + (-\infty)$ e $0
\times (+\infty)$ permanecem indefinidos, e toda manipulação algébrica de
\omterm{def:b1:reals:bounds}{supremos} deve antes verificar que nunca os forma.
A reta estendida é escrituração, não um sistema numérico.
\end{example}

\begin{example}[O supremo que escapou de $\Q$]\label{ex:b1:reals:supescape}
Volte ao \omterm{def:b1:logic:sets}{conjunto} da observação de abertura, $A = \{x \in \Q : x^2 <
2\}$, e calcule o seu \omterm{def:b1:reals:bounds}{supremo} \emph{em $\R$}. Ele é não vazio
($1 \in A$) e limitado superiormente por $1.5$ (se $x > 1.5$, então $x^2 >
2.25 > 2$), de modo que $s = \sup A$ existe. Afirmamos que $s = \sqrt 2$ (o
número real construído no \cref{exo:b1:reals:12}). \omterm{def:b1:reals:bounds}{Cota superior}: todo
$a \in A$ satisfaz $a < \sqrt2$ --- para $a \leq 0$ isso é
claro, e, para $a > 0$, $a \geq \sqrt2$ daria $a^2 \geq 2$.
Nada menor funciona: dado $t < \sqrt2$, a densidade
(\cref{thm:b1:reals:density}) fornece um racional $q$ com
$\max(1, t) < q < \sqrt 2$, e então $q^2 < 2$, de modo que $q \in A$
excede $t$. Pela \cref{prop:b1:reals:epsilon}, $s = \sqrt2
\notin \Q$. A ideia de fechamento: o \omterm{def:b1:reals:bounds}{supremo} de um \omterm{def:b1:logic:sets}{conjunto} de
racionais não precisa ser racional --- a completude é precisamente a
promessa de que $\R$, ao contrário de $\Q$, nunca deixa um \omterm{def:b1:reals:bounds}{supremo}
escapar; este exemplo é a observação de abertura do capítulo, agora
demonstrada em vez de apenas apontada.
\end{example}

\begin{remark}[Perspectivas dentro deste volume]
As três ferramentas do capítulo têm carreiras distintas pela frente. O
\omterm{def:b1:reals:bounds}{supremo} conduz a metade analítica: os limites monótonos
(\cref{ch:b1:seq}), a própria definição da integral
(\cref{ch:b1:integration}) e, na geometria do
\cref{ch:b1:euclid}, a distância de um ponto a um subespaço ---
um \omterm{def:b1:reals:bounds}{ínfimo} que a projeção ortogonal transforma em mínimo. A
função \omterm{thm:b1:reals:floor}{parte inteira} volta sempre que o discreto encontra o
contínuo: as expansões em algarismos (o problema de fim de semana deste
capítulo), a aproximação de Dirichlet por casa dos pombos
(\cref{pb:b1:derivative:1}), as comparações integrais de somas
(\cref{ch:b1:series}). Os argumentos de densidade se promovem a método
no \cref{ch:b1:continuity}: uma identidade entre funções contínuas
só precisa ser verificada em $\Q$ --- metade da equação funcional de
Cauchy (\cref{pb:b1:continuity:1}) é exatamente essa jogada. Quando
houver dúvida sobre de onde uma demonstração deste volume tira os seus
enunciados de existência, a resposta é quase sempre: deste capítulo.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:reals:1}
Determine (com demonstrações) sup, inf, máx e mín --- quando existirem --- de:
\[
A = \Bigl\{\frac{1}{n} : n \in \N^*\Bigr\},
\qquad
B = \Bigl\{\frac{(-1)^n n}{n+1} : n \in \N\Bigr\},
\qquad
C = \{x \in \R : x^2 < 3\}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:reals:2}
Demonstre que, para todos $x, y \in \R$: $\lfloor x \rfloor + \lfloor y
\rfloor \leq \lfloor x + y \rfloor \leq \lfloor x \rfloor + \lfloor y
\rfloor + 1$, e que as duas estimativas são atingidas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:reals:3}
Demonstre que, para todo $x \in \R$ e todo $n \in \N^*$:
$\Bigl\lfloor \frac{\lfloor nx \rfloor}{n} \Bigr\rfloor = \lfloor x
\rfloor$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:reals:4}
Sejam $A \subseteq B$ subconjuntos não vazios de $\R$, com $B$ limitado.
Demonstre que $\inf B \leq \inf A \leq \sup A \leq \sup B$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:5}
Para $A, B \subseteq \R$ não vazios e limitados, defina $A + B = \{a + b : a
\in A,\ b \in B\}$ e $-A = \{-a : a \in A\}$. Demonstre:
\[
\sup(A + B) = \sup A + \sup B,
\qquad
\sup(-A) = -\inf A .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:6}
Sejam $f, g \colon E \to \R$ funções limitadas. Demonstre que
\[
\sup_{x \in E}\, \bigl(f(x) + g(x)\bigr) \leq \sup_{x \in E} f(x) +
\sup_{x \in E} g(x),
\]
e dê um exemplo em que a desigualdade é estrita. Por que isso não
contradiz o \cref{exo:b1:reals:5}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:7}
Demonstre que $\sqrt 2 + \sqrt 3$ é irracional. \emph{(Eleve ao quadrado e
use a irracionalidade de $\sqrt 6$, a ser demonstrada via
\cref{exo:b1:arith:7}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:8}
Demonstre que o \omterm{def:b1:logic:sets}{conjunto} $D = \bigl\{\frac{m}{2^n} : m \in \Z,\ n \in
\N\bigr\}$ dos racionais diádicos é denso em $\R$: entre dois reais
quaisquer há um racional diádico.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:reals:9}
Seja $G$ um \omterm{def:b1:structures:subgroup}{subgrupo} de $(\R, +)$ com $G \neq \{0\}$. Ponha $\alpha =
\inf\,(G \cap \intoo{0}{+\infty})$. Demonstre:
\begin{enumerate}
  \item se $\alpha > 0$, então $G = \alpha\Z$;
  \item se $\alpha = 0$, então $G$ é denso em $\R$.
\end{enumerate}
Deduza que $\Z + \sqrt 2\,\Z$ é denso em $\R$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:reals:10}
Para $A, B$ \omterm{def:b1:logic:sets}{conjuntos} não vazios de reais positivos, seja $AB = \{ab : a \in
A, b \in B\}$. Demonstre que $\sup(AB) = \sup A \cdot \sup B$ (caso limitado)
e mostre, com um exemplo, que a positividade é essencial.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:11}
Para $A \subseteq \R$ não vazio e limitado, defina o
\emph{diâmetro}
\[
\operatorname{diam} A = \sup\,\{\abs{a - a'} : a, a' \in A\} .
\]
Demonstre que $\operatorname{diam} A = \sup A - \inf A$ e que
$\intcc{\inf A}{\sup A}$ é o menor \omterm{prop:b1:reals:intervals}{intervalo} fechado
que contém $A$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:reals:12}
Sejam $y > 0$ e $E = \{x \geq 0 : x^2 \leq y\}$. Demonstre que $E$ é
não vazio e limitado superiormente, e que $s = \sup E$ satisfaz $s^2 =
y$ \emph{(descarte $s^2 < y$ e $s^2 > y$ exibindo, em cada
caso, um pequeno $h > 0$ que contradiz a definição do
\omterm{def:b1:reals:bounds}{supremo})}. Deduza que todo $y > 0$ tem uma única raiz quadrada
$\sqrt y > 0$ e que $y \mapsto \sqrt y$ é crescente em
$\intoo{0}{+\infty}$.
\end{exercise}

\section{Problema: Expansões em algarismos e o ritmo dos racionais}

\begin{problem}[{Problema de fim de semana --- expansões $b$-ádicas:
existência, unicidade e a periodicidade que caracteriza $\Q$}]
\label{pb:b1:reals:1}
Todo real de $\intco{0}{1}$ tem uma expansão em algarismos em toda base
$b \geq 2$; a expansão é única desde que se proíbam cadeias finais do
algarismo $b - 1$; e ela é periódica a partir de certo ponto exatamente
quando o número é racional. Este problema demonstra os três fatos
apenas a partir do axioma da completude --- sem sequências, sem séries:
só o \omterm{def:b1:reals:bounds}{supremo}, a \omterm{thm:b1:reals:archimedes}{propriedade arquimediana} e a função \omterm{thm:b1:reals:floor}{parte inteira} ---
e se encerra com o argumento diagonal de Cantor em forma de algarismos.
Ao longo do problema, $b \geq 2$ é um inteiro fixado (a \emph{base}), um
\emph{algarismo} é um elemento de $\intint{0}{b-1}$, e uma cadeia de
algarismos $(d_n)_{n \geq 1}$ é \emph{própria} quando não é
igual a $b - 1$ a partir de certo ponto (isto é: para todo $N$ existe $n >
N$ com $d_n \leq b - 2$).\index{expansao b-adica@expansão $b$-ádica}\index{expansão decimal}

\medskip
\noindent\textbf{Parte I --- Algarismos à mão.} Divisão longa de $p$
por $q$ na base $b$: multiplique o resto atual por $b$, divida
por $q$, registre o quociente como o algarismo seguinte e guarde o resto.
\begin{enumerate}
  \item Na base $10$, execute o algoritmo em $\frac 18$ e em
        $\frac 17$, registrando a cada passo o algarismo \emph{e}
        o resto. Confira que os restos de $\frac 17$
        percorrem o ciclo $1, 3, 2, 6, 4, 5$ e que os algarismos
        $142857$ se repetem para sempre.
  \item Calcule as expansões na base $2$ de $\frac 13$ e de
        $\frac{5}{16}$, e a expansão na base $3$ de $\frac 12$.
        Observe: um número termina, e os outros dois se repetem ---
        e $\frac 12$, tão dócil na base $10$, repete-se para sempre na
        base $3$.
  \item Para $x = \frac pq \in \intco{0}{1}$ na forma irredutível, mostre
        que os algarismos produzidos pelo algoritmo são, a partir de certo
        ponto, todos $0$ se, e somente se, o resto $b^N p \bmod q$
        se anula para algum $N$, se, e somente se, $q$ \omterm{def:b1:arith:divides}{divide} alguma
        potência $b^N$, se, e somente se, todo fator \omterm{def:b1:arith:prime}{primo} de $q$
        \omterm{def:b1:arith:divides}{divide} $b$. Confira: $\frac{1}{20}$ termina na base
        $10$, mas não na base $3$.
  \item Defina o \emph{truncamento} $s_n = \lfloor b^n x \rfloor
        / b^n$. Para $x = \sqrt 2$ e $b = 10$, calcule $s_0,
        \dots, s_4$ verificando a cada passo que dois
        quadrados consecutivos cercam $2$ (por exemplo, $1.4142^2
        = 1.99996164 < 2 < 2.00024449 = 1.4143^2$), e confira
        $s_n \leq \sqrt 2 < s_n + 10^{-n}$ em cada caso.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Existência, a partir do \omterm{def:b1:reals:bounds}{supremo}.} Fixe
$x \in \intco{0}{1}$ e ponha $A_n = \lfloor b^n x \rfloor$ e
$d_n = A_n - b\,A_{n-1}$ para $n \geq 1$.
\begin{enumerate}
  \item Mostre que $A_0 = 0$ e $b\,A_{n-1} \leq A_n \leq b\,A_{n-1} +
        b - 1$; conclua que cada $d_n$ é um algarismo.
  \item Mostre que $s_n := A_n b^{-n}$ satisfaz
        \[
        s_n = \sum_{k=1}^{n} d_k\,b^{-k}
        \qquad\text{e}\qquad
        s_n \leq x < s_n + b^{-n} .
        \]
  \item Demonstre que $b^n \geq n + 1$ por indução e mostre depois que
        $(s_n)$ é não decrescente e que $x = \sup_n s_n$
        \emph{(use a \cref{prop:b1:reals:epsilon} e o
        \cref{thm:b1:reals:archimedes})}.
  \item Mostre que a cadeia $(d_n)$ é própria: se $d_k = b - 1$
        para todo $k > N$, calcule $s_n$ para $n > N$ por uma soma
        geométrica finita e contradiga a questão 6.
  \item Reciprocamente, seja $(e_n)_{n \geq 1}$ uma cadeia própria de
        algarismos qualquer e $t_n = \sum_{k=1}^n e_k b^{-k}$. Mostre que $y
        = \sup_n t_n$ existe, está em $\intco{0}{1}$ e
        satisfaz $t_n \leq y < t_n + b^{-n}$ para todo $n$
        \emph{(para a desigualdade estrita, use um algarismo $e_m \leq b
        - 2$ com $m > n$)}. Deduza que $\lfloor b^n y \rfloor = b^n
        t_n$ e, em seguida, que os algarismos de $y$, no sentido da
        questão 5, são exatamente os $e_n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Unicidade, ordem, deslocamento.}
\begin{enumerate}
  \item Reúna as questões 5--9 no \emph{teorema da expansão
        $b$-ádica}: as aplicações $x \mapsto (d_n)$ e $(e_n) \mapsto
        \sup_n t_n$ são bijeções inversas uma da outra entre
        $\intco{0}{1}$ e o \omterm{def:b1:logic:sets}{conjunto} das cadeias próprias de algarismos. Em
        particular, duas cadeias próprias distintas nunca têm o mesmo
        valor.
  \item Permita agora cadeias impróprias. Mostre que uma cadeia com $e_n
        = b - 1$ para todo $n > M$ (com $M \geq 0$ mínimo) tem
        valor $t_M + b^{-M}$; conclua que $0.999\dots = 1$ na
        base $10$ e que os reais com duas representações em algarismos
        são exatamente as frações $b$-ádicas
        $m/b^N \in \intoo{0}{1}$ --- todo outro real tem apenas
        uma, mesmo entre as cadeias impróprias.
  \item Demonstre que a bijeção da questão 10 preserva a ordem
        lexicográfica: se as cadeias próprias de $x$ e de $y$ diferem
        pela primeira vez no índice $m$, então $x < y$ se, e somente se,
        $d_m < e_m$.
  \item (Lema do deslocamento) Seja $x \in \intco{0}{1}$ com algarismos
        $(d_n)$. Mostre que a parte fracionária de $bx$ tem algarismos
        $(d_{n+1})_{n \geq 1}$ \emph{(calcule $\lfloor b^n(bx -
        A_1)\rfloor$ usando $\lfloor u - K \rfloor = \lfloor u
        \rfloor - K$ para $K$ inteiro)} e deduza por indução
        que a parte fracionária de $b^m x$ tem algarismos
        $(d_{n+m})_{n \geq 1}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Racionalidade é periodicidade.} Seja $x
= \frac pq \in \intco{0}{1}$ na forma irredutível e $r_n = b^n p
\bmod q$ o resto da divisão euclidiana de $b^n p$ por
$q$.
\begin{enumerate}
  \item Mostre que $A_n = \dfrac{b^n p - r_n}{q}$ e $r_n =
        (b\,r_{n-1}) \bmod q$.
  \item Mostre que $d_n = \Bigl\lfloor \dfrac{b\,r_{n-1}}{q}
        \Bigr\rfloor$: cada algarismo é função apenas do resto
        anterior. É exatamente a divisão longa da Parte I.
  \item Aplique o princípio da casa dos pombos
        (\cref{cor:b1:counting:pigeonhole}) a $r_0, \dots, r_q$
        e conclua: a expansão de todo racional é
        periódica a partir de certo ponto, com pré-período e período no
        máximo $q$.
  \item Reciprocamente, suponha que os algarismos de $y \in \intco{0}{1}$
        sejam \emph{puramente} periódicos: $d_{n+T} = d_n$ para todo $n
        \geq 1$. Usando o lema do deslocamento e a unicidade da
        questão 10, mostre que a parte fracionária de $b^T y$
        é igual a $y$ e deduza que $(b^T - 1)\,y \in \N$: assim, $y$ é
        racional, com denominador que \omterm{def:b1:arith:divides}{divide} $b^T - 1$. Verifique o
        mecanismo em $0.(142857)$: $142857 \times 7 = 999999$.
  \item Trate o caso periódico a partir de certo ponto por deslocamento
        e enuncie o \emph{critério de periodicidade}: $x \in \intco{0}{1}$ é
        racional se, e somente se, a sua expansão $b$-ádica própria é
        periódica a partir de certo ponto --- numa base se, e somente se,
        em todas.
  \item Para $x = \frac 1q$ com $\gcd(q, b) = 1$, mostre que a
        expansão é puramente periódica e que o seu menor período é
        o menor $T \geq 1$ com $b^T \equiv 1 \pmod q$ (a
        ordem multiplicativa de $b$ módulo $q$). Confira que, para
        $q = 7$, $b = 10$, as potências de $10$ módulo $7$ percorrem
        $3, 2, 6, 4, 5, 1$: ordem $6$, coincidindo com a questão
        1.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos e a diagonal.}
\begin{enumerate}
  \item Seja $x^*$ o real de $\intco{0}{1}$ cujos algarismos na base $10$
        valem $1$ nas posições triangulares $\frac{j(j +
        1)}{2}$
        ($j \geq 1$) e $0$ nas demais: $x^* =
        0.101001000100001\dots$ Mostre que a sua cadeia de algarismos é
        própria, mas não periódica a partir de certo ponto \emph{(um
        período $T$ obrigaria a haver uns com espaçamento no máximo $T$,
        mas os espaçamentos crescem)}, e conclua que $x^*$ é irracional:
        um número demonstrado irracional por puro ritmo.
  \item Mostre que, para toda base $b$, o \omterm{def:b1:logic:sets}{conjunto} $\{m/b^n : m \in \Z,
        n \in \N\}$ é denso em $\R$ (generalizando o
        \cref{exo:b1:reals:8}) e que todo racional $\frac pq
        \in \intoo{0}{1}$ tem expansão \emph{finita} na
        base $q$. Moral: terminar é uma propriedade do par
        (número, base); a periodicidade --- a racionalidade --- é
        intrínseca.
  \item (Diagonal de Cantor) Seja $k \mapsto x_k$ uma \omterm{def:b1:logic:map}{aplicação} qualquer de
        $\N^*$ em $\intco{0}{1}$. Defina a cadeia de algarismos $e_k =
        1$ se o $k$-ésimo algarismo de $x_k$ for diferente de $1$, e
        $e_k = 2$ caso contrário. Mostre que $(e_k)$ é própria, que
        o seu valor $y$ está em $\intco{0}{1}$ e que $y \neq
        x_k$ para todo $k$. Conclua: nenhuma \omterm{def:b1:logic:map}{aplicação} $\N^* \to
        \intco{0}{1}$ é \omterm{def:b1:logic:inj}{sobrejetiva}. (O vocabulário da
        enumerabilidade, e o lugar próprio deste teorema, está no
        \cref{ch:b1:topology}.)
  \item Mostre que, se as expansões próprias de $x$ e de $y$ coincidem
        até o índice $n$, então $\abs{x - y} < b^{-n}$, e refute a
        recíproca com $x = 0.1$, $y = 0.0999$ na base $10$:
        a proximidade dos números não força a coincidência dos algarismos.
        De quem é a culpa?
  \item Execute a Parte IV em $x = \frac{1}{10}$ na base $b = 2$:
        calcule restos e algarismos até que ciclem e
        conclua que $\frac{1}{10} = (0.0\overline{0011})_2$, com
        pré-período $1$ e período $4$. Explique, via a questão 3, por que
        nenhuma cadeia binária finita jamais será igual a $\frac{1}{10}$
        --- a razão pela qual, em ponto flutuante, $0.1 + 0.2$ não é
        exatamente $0.3$ num computador.
  \item Síntese. Em uma frase cada: onde a demonstração usou
        (i) a completude, (ii) a \omterm{thm:b1:reals:archimedes}{propriedade arquimediana}, (iii)
        a cláusula de unicidade da \omterm{thm:b1:reals:floor}{parte inteira}, (iv) o princípio
        da casa dos pombos? E a moral: $\intco{0}{1}$ é fielmente
        codificado por cadeias próprias de algarismos, e a racionalidade
        se lê como periodicidade --- e, ainda assim, a análise prefere o
        \omterm{def:b1:reals:bounds}{supremo} aos algarismos. Por quê? (Pense em somar duas cadeias de
        algarismos.)
\end{enumerate}
\end{problem}
