\chapter{Fórmulas de Taylor e Expansões Assintóticas}\label{ch:b1:taylor}

Perto de um ponto, uma função suave vale tanto quanto um \omterm{def:b1:poly:def}{polinômio} ---
com um erro controlável. As fórmulas de Taylor tornam isso exato em três
versões (resto integral, resto de Lagrange, resto de Young), e as
\emph{expansões assintóticas} resultantes, manipuladas algebricamente,
tornam-se a ferramenta mais afiada da análise elementar: limites,
equivalentes, comportamento local, assíntotas.

\section{Notação de comparação}

\begin{definition}[Notação de Landau]\label{def:b1:taylor:landau}
Sejam $f, g$ definidas perto de $x_0$ ($x_0 \in \R$ ou $\pm\infty$). Escreve-se, quando
$x \to x_0$:\index{notação de Landau}
\begin{itemize}
  \item $f = o(g)$ (``o pequeno'') quando $f = \varepsilon g$ com
        $\varepsilon(x) \to 0$;
  \item $f = O(g)$ (``O grande'') quando $f = u g$ com $u$ limitada perto de
        $x_0$;
  \item $f \sim g$ (``equivalente''\index{funções equivalentes}) quando
        $f = (1 + \varepsilon) g$ com $\varepsilon \to 0$ ---
        equivalentemente $f - g = o(g)$.
\end{itemize}
A mesma notação aplica-se a sequências ($n \to \infty$).
\end{definition}

\begin{proposition}[Regras]\label{prop:b1:taylor:landaurules}
Quando $x \to x_0$:
\begin{enumerate}
  \item $\sim$ é uma \omterm{def:b1:logic:equiv}{relação de equivalência}; $f \sim g$ implica que
        $f$ e $g$ compartilham limites, sinais (perto de $x_0$) e a
        ausência de zeros;
  \item equivalentes \emph{multiplicam-se e dividem-se}: $f_1 \sim g_1$, $f_2
        \sim g_2$
        implicam $f_1 f_2 \sim g_1 g_2$ e $\frac{f_1}{f_2}
        \sim \frac{g_1}{g_2}$;
  \item equivalentes \emph{não} se somam: $x + 1 \sim x$ e $-x \sim
        -x + 2$ em $+\infty$, e
        no entanto as somas $1$ e $2$ não são
        equivalentes. Para somar, volte às expansões com termos
        $o(\cdot)$ explícitos;
  \item $o(g) + o(g) = o(g)$, $\;u \cdot o(g) = o(ug)$, $\;o(o(g)) =
        o(g)$, e $f \sim g \iff f = g + o(g)$.
\end{enumerate}
\end{proposition}

\begin{proof}
Cada uma é uma curta manipulação das definições; por exemplo
$f_1 f_2 = (1+\varepsilon_1)(1+\varepsilon_2) g_1 g_2$ e $(1 +
\varepsilon_1)(1+\varepsilon_2) \to 1$. Dois itens de (4) merecem
a sua linha. $u\cdot o(g) = o(ug)$: se $f = \varepsilon g$ com
$\varepsilon \to 0$, então $uf = \varepsilon\,(ug)$ com o mesmo
$\varepsilon$. $o(o(g)) = o(g)$: se $f = \varepsilon_1 h$ e
$h = \varepsilon_2 g$ com ambos $\varepsilon_i \to 0$, então $f =
(\varepsilon_1\varepsilon_2) g$ e o produto dos dois
infinitésimos é um infinitésimo. A equivalência $f \sim g \iff f = g +
o(g)$ é a definição lida duas vezes: $f - g = \varepsilon g$.
O contraexemplo em (3) é a demonstração de (3).
\end{proof}

\begin{example}[A escala de comparação]\label{ex:b1:taylor:scale}
Quando $x \to +\infty$, a escala padrão lê-se, em ordem crescente
de força:
\[
1 \;=\; o(\ln x), \quad
\ln x = o(x^{0.01}), \quad
x^{0.01} = o(\sqrt x), \quad
\sqrt x = o(x^{10}), \quad
x^{10} = o(\eu^{x}), \quad
\eu^x = o(\eu^{2x}) ,
\]
cada passo sendo uma instância das comparações de crescimento da
\cref{prop:b1:functions:powerrules} (potências vencem logaritmos,
exponenciais vencem potências, e dentro de uma mesma família o expoente
decide). Dois hábitos que vale a pena formar: primeiro, um $O(\cdot)$ que
aterrissa numa classe menor melhora silenciosamente ($O(\ln x)$ também é
$o(x^{0.01})$); segundo, em $x \to 0^+$ toda a escada
se inverte pela substituição $x \mapsto \frac1x$ ---
$\ln x = o(x^{-0.01})$ ali, de modo que ``$x^\alpha \ln x \to 0$''
vale para todo $\alpha > 0$. Manter a escala em ordem é
metade de todo argumento assintótico do
\cref{ch:b1:series}.
\end{example}

\begin{example}[Unicidade das expansões, e um dividendo de paridade]\label{ex:b1:taylor:uniqueness}
Se uma função admite duas expansões em $0$ até a mesma ordem,
\[
a_0 + a_1 x + \dots + a_n x^n + o(x^n)
= b_0 + b_1 x + \dots + b_n x^n + o(x^n),
\]
então $a_k = b_k$ para todo $k$: subtraindo e pondo $c_k =
a_k - b_k$, avalie a
identidade $c_0 + c_1 x + \dots + c_n x^n
= o(x^n)$ quando $x \to 0$ para obter $c_0 = 0$; divida
por $x$ e repita --- cada divisão é legítima porque a expressão
restante é novamente $o(x^{n-k})$. Os coeficientes são portanto
intrínsecos, e podem ser calculados por \emph{qualquer} caminho
(\omterm{def:b1:derivative:def}{derivadas} de Taylor, álgebra sobre expansões conhecidas, integração):
todos os caminhos devem concordar. Dividendo: uma função \emph{par}
tem apenas potências pares na sua expansão --- substitua $x$ por $-x$ e
invoque a unicidade; do mesmo modo as funções ímpares têm potências
ímpares. É por isso que $\cos$ carrega $o(x^{2p+1})$ em vez de $o(x^{2p})$ na
tabela abaixo: o termo ímpar ausente é informação gratuita, uma ordem
de precisão de graça.
\end{example}

\section{As três fórmulas de Taylor}

\begin{theorem}[Taylor com resto integral]\label{thm:b1:taylor:integral}
Seja $f$ de classe $C^{n+1}$ num \omterm{prop:b1:reals:intervals}{intervalo} contendo $a$ e $x$.
Então\index{fórmula de Taylor}
\[
f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}\,(x - a)^k
+ \int_a^x \frac{(x - t)^n}{n!}\, f^{(n+1)}(t)\, \dd t .
\]
\end{theorem}

\begin{proof}
Indução sobre $n$. Para $n = 0$: $f(x) = f(a) + \int_a^x f'(t)\dd t$ é
o teorema fundamental (\cref{thm:b1:integration:ftc}). Passo:
integre o resto por partes,
\[
\int_a^x \frac{(x-t)^n}{n!} f^{(n+1)}(t)\,\dd t
= \Bigl[-\frac{(x-t)^{n+1}}{(n+1)!} f^{(n+1)}(t)\Bigr]_a^x
+ \int_a^x \frac{(x-t)^{n+1}}{(n+1)!} f^{(n+2)}(t)\,\dd t ,
\]
o colchete contribuindo com o termo $\frac{f^{(n+1)}(a)}{(n+1)!}(x -
a)^{n+1}$.
\end{proof}

\begin{example}[Uma expansão exata com o seu resto]\label{ex:b1:taylor:lnexact}
Para $\ln(1 + x)$ o resto integral pode ser tornado completamente
explícito sem derivar coisa alguma $n$ vezes: integre a identidade
geométrica finita $\frac{1}{1+t} =
\sum_{k=0}^{n-1}(-t)^k + \frac{(-t)^n}{1+t}$ de $0$ a $x$:
\[
\ln(1 + x) = \sum_{k=1}^{n} \frac{(-1)^{k-1}x^k}{k}
+ (-1)^n \int_0^x \frac{t^n}{1 + t}\,\dd t ,
\]
e para $0 \leq x \leq 1$ o resto é majorado por $\int_0^x
t^n\,\dd t = \frac{x^{n+1}}{n+1}$. Isto é mais forte do que
Taylor--Young de duas maneiras: é uma \emph{identidade} válida para um
$x$ fixo (não apenas quando $x \to 0$), e a estimativa do erro é
numérica. O problema de fim de semana (\cref{pb:b1:taylor:1}) vive sobre tais formas
exatas; Taylor--Young abaixo é a ferramenta mais leve para limites, em
que só importa a forma do erro.
\end{example}

\begin{theorem}[Desigualdade de Taylor--Lagrange]\label{thm:b1:taylor:lagrange}
Seja $f$ de classe $C^{n+1}$ com $\abs{f^{(n+1)}} \leq M$ entre $a$ e
$x$. Então
\[
\Bigl| f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k \Bigr|
\leq M\, \frac{\abs{x - a}^{n+1}}{(n+1)!} .
\]
\end{theorem}

\begin{proof}
Majore o resto integral: $\bigl|\int_a^x \frac{(x-t)^n}{n!}
f^{(n+1)}(t)\,\dd t\bigr| \leq M \bigl|\int_a^x
\frac{\abs{x-t}^n}{n!}\dd t\bigr| = M\frac{\abs{x-a}^{n+1}}{(n+1)!}$.
\end{proof}

\begin{example}[Numérica certificada]\label{ex:b1:taylor:certified}
Quanto vale $\sqrt{1.02}$? Aplique Taylor--Lagrange a $f(t) =
\sqrt{1 + t}$ em $a = 0$,
ordem $2$, $x = 0.02$:
\[
\sqrt{1.02} \approx 1 + \frac{0.02}{2} - \frac{0.02^2}{8}
= 1.00995 ,
\qquad
\abs{f'''(t)} = \frac{3}{8}(1+t)^{-5/2} \leq \frac 38 ,
\]
de modo que o erro é no máximo $\frac38 \cdot \frac{0.02^3}{6} =
5\cdot10^{-7}$: $\sqrt{1.02} = 1.00995$ com seis decimais
certificadas (valor verdadeiro $1.0099504938\dots$ --- a estimativa é quase ótima).
A ideia de fechamento: Taylor--Young diz apenas \emph{quão rápido} o
erro se anula; Taylor--Lagrange converte o mesmo \omterm{def:b1:poly:def}{polinômio} num
\emph{certificado}, um número acompanhado de uma barra de erro
demonstrada. Sempre que uma afirmação decimal é feita neste livro, há
uma estimativa do tipo Lagrange por trás dela; o problema de fim de
semana (\cref{pb:b1:taylor:1}) industrializa a ideia.
\end{example}

\begin{theorem}[Taylor--Young]\label{thm:b1:taylor:young}
Seja $f$ $n$ vezes \omterm{def:b1:derivative:def}{derivável} em $a$. Então, quando $x \to a$:
\[
f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}\,(x-a)^k
+ o\bigl((x-a)^n\bigr) .
\]
\end{theorem}

\begin{proof}
Indução sobre $n$. Para $n = 1$ isto é a definição da \omterm{def:b1:derivative:def}{derivada}
(\cref{def:b1:derivative:def}). Suponha o enunciado na ordem $n - 1$, e seja $f$ $n$ vezes
\omterm{def:b1:derivative:def}{derivável} em $a$. Aplique a hipótese de indução a $f'$ (que é $n-1$
vezes \omterm{def:b1:derivative:def}{derivável} em $a$):
\[
f'(t) = \sum_{k=0}^{n-1} \frac{f^{(k+1)}(a)}{k!}(t-a)^k + r(t),
\qquad r(t) = o\bigl((t-a)^{n-1}\bigr).
\]
Seja $g(x) = f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k$; então $g' = r$ e $g(a) = 0$. Dado $\varepsilon > 0$, escolha $\delta$
com $\abs{r(t)} \leq \varepsilon\abs{t - a}^{n-1}$ para $\abs{t-a}
\leq \delta$; a desigualdade do valor médio
(\cref{thm:b1:derivative:mvt}) aplicada no segmento de $a$ a $x$ (onde $\abs{g'} \leq \varepsilon\abs{x-a}^{n-1}$) dá
$\abs{g(x)} \leq \varepsilon\abs{x - a}^n$: exatamente $g(x) =
o((x-a)^n)$.
\end{proof}

\begin{remark}[Três fórmulas, três preços, três produtos]
As hipóteses graduam-se exatamente com as conclusões.
Taylor--Young pede o mínimo ($n$ \omterm{def:b1:derivative:def}{derivadas} \emph{no ponto} apenas) e
devolve o mínimo: um $o((x-a)^n)$ qualitativo, perfeito para limites, inútil
para dígitos certificados. A desigualdade de Lagrange pede $C^{n+1}$
\emph{no \omterm{prop:b1:reals:intervals}{intervalo}} e uma cota $M$ ali, e devolve uma barra de erro
numérica. A forma integral pede a mesma regularidade e devolve o mais
possível: o erro como um objeto explícito que se pode transformar
(integrar por partes, majorar por pedaços, mudar de variável) --- é a
forma que alimentou a máquina de irracionalidade do \cref{pb:b1:integration:1}. Escolher a
fórmula mais fraca que sustenta o objetivo não é pedantismo: a função
plana do \cref{pb:b1:taylor:1} satisfaz Taylor--Young em toda ordem, ao passo que
toda conclusão mais forte a seu respeito é falsa longe de $0$.
\end{remark}

\begin{proposition}[Expansões padrão em $0$]
Quando $x \to 0$, para toda ordem fixa $n$:
\index{expansão de Taylor!padrão}
\begin{align*}
\eu^x &= 1 + x + \frac{x^2}{2!} + \dots + \frac{x^n}{n!} + o(x^n),\\
\cos x &= 1 - \frac{x^2}{2!} + \frac{x^4}{4!} - \dots
+ \frac{(-1)^p x^{2p}}{(2p)!} + o(x^{2p+1}),\\
\sin x &= x - \frac{x^3}{3!} + \dots + \frac{(-1)^p
x^{2p+1}}{(2p+1)!} + o(x^{2p+2}),\\
\frac{1}{1 - x} &= 1 + x + x^2 + \dots + x^n + o(x^n),\\
\ln(1 + x) &= x - \frac{x^2}{2} + \frac{x^3}{3} - \dots +
\frac{(-1)^{n-1} x^n}{n} + o(x^n),\\
(1 + x)^\alpha &= 1 + \alpha x + \frac{\alpha(\alpha-1)}{2!}x^2 +
\dots + \binom{\alpha}{n} x^n + o(x^n),
\end{align*}
onde $\binom{\alpha}{n} = \frac{\alpha(\alpha - 1)\cdots(\alpha - n
+ 1)}{n!}$ para $\alpha$ real. ($\cosh$ e $\sinh$: iguais a $\cos$,
$\sin$ sem os sinais alternados.)
\label{prop:b1:taylor:standard}
\end{proposition}

\begin{proof}
Cada função é suave perto de $0$ com \omterm{def:b1:derivative:def}{derivadas} fáceis de avaliar:
$(\eu^x)^{(k)} = \eu^x$; as \omterm{def:b1:derivative:def}{derivadas} de $\sin$ e $\cos$ ciclam
com período $4$; $\bigl((1+x)^\alpha\bigr)^{(k)} = \alpha(\alpha -
1)\cdots(\alpha - k + 1)(1+x)^{\alpha - k}$;
$\bigl(\ln(1+x)\bigr)^{(k)} = \frac{(-1)^{k-1}(k-1)!}{(1+x)^k}$.
Aplique Taylor--Young em $a = 0$. (A geométrica é exata:
$\frac{1}{1-x} - \sum_0^n x^k = \frac{x^{n+1}}{1 - x} = o(x^n)$.)
\end{proof}

\begin{method}[Calcular com expansões]\label{met:b1:taylor:compute}
\begin{enumerate}
  \item \emph{Fixe primeiro a ordem alvo} $n$, e trunque ali todo
        resultado intermediário --- carregar termos de ordem superior
        é trabalho desperdiçado, descartar os de ordem inferior é um
        erro.
  \item \emph{Somas, produtos}: expanda cada fator até a ordem $n$ e
        multiplique, descartando o que passa de $x^n$.
  \item \emph{Composição} $f(u(x))$ com $u(x) \to 0$: substitua
        a expansão de $u$ na de $f$, ordem a ordem.
  \item \emph{Quocientes}: escreva $\frac{1}{1 + v}$ com $v \to 0$ e
        use a expansão geométrica.
  \item \emph{Integre} uma expansão termo a termo (derivar exige mais
        cuidado --- justificativa: a \omterm{thm:b1:integration:def}{integral} de $o(t^n)$ de $0$ a $x$
        é $o(x^{n+1})$, por majoração direta).
\end{enumerate}
\end{method}

\begin{example}[Composição, com a contabilidade à vista]\label{ex:b1:taylor:esin}
Expanda $\eu^{\sin x}$ até a ordem $3$. Expansão interna: $u = \sin x
= x - \frac{x^3}{6} + o(x^3)$, que de fato
tende a $0$. Externa: $\eu^u = 1 + u + \frac{u^2}{2} + \frac{u^3}{6} + o(u^3)$, e $o(u^3) = o(x^3)$ pois $u \sim x$. Potências de $u$,
truncadas em $x^3$:
\[
u^2 = x^2 + o(x^3), \qquad u^3 = x^3 + o(x^3)
\]
(o termo cruzado $2x\cdot(-\frac{x^3}{6})$ já é $x^4$).
Junte tudo:
\[
\eu^{\sin x} = 1 + \Bigl(x - \frac{x^3}{6}\Bigr) + \frac{x^2}{2}
+ \frac{x^3}{6} + o(x^3)
= 1 + x + \frac{x^2}{2} + o(x^3) :
\]
as duas contribuições em $x^3$ cancelam-se exatamente. A ideia de
fechamento: $\eu^{\sin x}$ e $\eu^x$ coincidem até a ordem $3$ --- não porque
$\sin x \approx x$ grosseiramente, mas porque a primeira discordância dos
expoentes ($-\frac{x^3}{6}$) entra multiplicada por $\eu^0 = 1$ e é então compensada
pelo termo cúbico da exponencial externa; a contabilidade ordem a
ordem detecta tais conspirações, o olho nunca. (O termo seguinte é
$-\frac{x^4}{8}$: a trégua acaba na ordem $4$.)
\end{example}

\begin{example}\label{ex:b1:taylor:tan}
Expansão de $\tan$ na ordem $5$. Escreva $\tan x =
\sin x \cdot \frac{1}{\cos x}$:
\[
\frac{1}{\cos x} = \frac{1}{1 - \bigl(\frac{x^2}{2} - \frac{x^4}{24}
+ o(x^5)\bigr)}
= 1 + \Bigl(\frac{x^2}{2} - \frac{x^4}{24}\Bigr)
+ \Bigl(\frac{x^2}{2}\Bigr)^{\!2} + o(x^5)
= 1 + \frac{x^2}{2} + \frac{5x^4}{24} + o(x^5),
\]
então
\[
\tan x = \Bigl(x - \frac{x^3}{6} + \frac{x^5}{120}\Bigr)
\Bigl(1 + \frac{x^2}{2} + \frac{5x^4}{24}\Bigr) + o(x^5)
= x + \frac{x^3}{3} + \frac{2x^5}{15} + o(x^5) .
\]
\end{example}

\section{Aplicações}

\begin{example}[Limites]\label{ex:b1:taylor:limits}
\[
\lim_{x \to 0} \frac{x - \sin x}{x^3}:
\qquad
x - \sin x = \frac{x^3}{6} + o(x^3) \sim \frac{x^3}{6},
\qquad\text{logo o limite é } \frac16
\]
--- resolvendo a questão levantada no \cref{exo:b1:functions:9}.
Do mesmo modo $\displaystyle\lim_{x\to0}\Bigl(\frac{\sin
x}{x}\Bigr)^{1/x^2}$: o logaritmo é
\[
\frac{1}{x^2}\ln\Bigl(1 - \frac{x^2}{6} + o(x^2)\Bigr)
= \frac{1}{x^2}\Bigl(-\frac{x^2}{6} + o(x^2)\Bigr)
\longrightarrow -\frac16,
\qquad\text{limite } \eu^{-1/6}.
\]
\end{example}

\begin{remark}[Armadilhas comuns com expansões]
(i) \emph{Nunca some nem subtraia equivalentes}: de $\tan x \sim
x$ e $\sin x \sim x$ não
se pode \emph{concluir} $\tan x -
\sin x \sim 0$ (o que não faz sentido) --- a via
honesta são as expansões:
\[
\tan x - \sin x
= \Bigl(x + \frac{x^3}{3}\Bigr) - \Bigl(x - \frac{x^3}{6}\Bigr)
+ o(x^3) = \frac{x^3}{2} + o(x^3) \sim \frac{x^3}{2} .
\]
(ii) \emph{Expanda para além do massacre}: no mesmo cálculo, a ordem
$1$ vê apenas $x - x = 0$; sempre que os termos dominantes se cancelam,
aumente a ordem até que um coeficiente não nulo sobreviva, e só então
converta de volta para um equivalente. (iii) \emph{Equivalentes não
passam através de exponenciais}: $n^2 + n \sim n^2$, e no entanto $\eu^{n^2+n} = \eu^{n}\,\eu^{n^2}$ \emph{não}
é equivalente a $\eu^{n^2}$ --- exponencie apenas \emph{expansões do
expoente} cujo erro tende a $0$, nunca equivalentes do expoente.
(Logaritmos são mais seguros: se $u_n \sim v_n \to \ell \neq
1$, $\ell > 0$, então $\ln u_n \sim \ln v_n$.) (iv)
\emph{O cálculo com $o(\cdot)$ é unidirecional}: $o(x^2) + o(x^2) =
o(x^2)$, $5\,o(x^2) = o(x^2)$, $x\cdot o(x^2) = o(x^3)$ ---
mas um $o(x^2)$ não é uma função específica, logo nunca cancele dois
deles um contra o outro: $o(x^2) - o(x^2)$ é $o(x^2)$, e não $0$.
\end{remark}

\begin{proposition}[Comportamento local]\label{prop:b1:taylor:local}
Suponha $f(x) = f(a) + c\,(x - a)^p + o\bigl((x-a)^p\bigr)$ com $c
\neq 0$ (primeiro termo não nulo depois da constante;
$p \geq 2$ num ponto crítico).
\begin{itemize}
  \item Se $p$ é par: $f$ tem um mínimo local em $a$ se $c > 0$, e um
        máximo local se $c < 0$.
  \item Se $p$ é ímpar: nenhum extremo ($f - f(a)$ muda de sinal); se além
        disso a expansão começa depois de um termo linear $f'(a)(x
        - a)$, o
        gráfico atravessa a sua tangente: uma \emph{inflexão}.
\end{itemize}
\end{proposition}

\begin{proof}
Perto de $a$, $f(x) - f(a) = (x-a)^p\bigl(c + o(1)\bigr)$ tem o sinal de $c\,(x-a)^p$: sinal constante para $p$
par, e sinal variável para $p$ ímpar.
\end{proof}

\begin{example}[Expoentes devem ser expandidos até $o(1)$]\label{ex:b1:taylor:expdiscipline}
Encontre um equivalente de $u_n = \bigl(1 + \frac1n\bigr)^{n^2}$.
Expanda o \emph{expoente} até que o seu erro tenda a $0$:
\[
n^2 \ln\Bigl(1 + \frac1n\Bigr)
= n^2\Bigl(\frac1n - \frac{1}{2n^2} +
O\Bigl(\frac{1}{n^3}\Bigr)\Bigr)
= n - \frac12 + O\Bigl(\frac1n\Bigr),
\]
logo $u_n = \eu^{\,n - 1/2}\,\eu^{O(1/n)}$ com $\eu^{O(1/n)} \to
1$:
\[
u_n \;\sim\; \eu^{\,n - \frac12} .
\]
Note o que teria dado errado com menos cuidado: parar o expoente em
$n^2\cdot\frac1n = n + O(1)$ deixa um fator $\eu^{O(1)}$ --- limitado, mas não tendendo a $1$ --- e
nenhum equivalente pode ser afirmado. A regra das armadilhas acima, em
forma positiva: um equivalente de $\eu^{a_n}$ exige a expansão de $a_n$
\emph{até um termo que tende a zero}, mantendo exatamente todo
coeficiente anterior a esse.
\end{example}

\begin{example}[Classificando um ponto crítico achatado]\label{ex:b1:taylor:flatmin}
Estude $f(x) = \cos x + \frac{x^2}{2}$ perto de $0$. Tanto $f'(0) =
0$ quanto $f''(0) = -\cos 0 + 1 = 0$: o teste da
segunda \omterm{def:b1:derivative:def}{derivada} é mudo. Expanda em vez disso:
\[
f(x) = \Bigl(1 - \frac{x^2}{2} + \frac{x^4}{24} +
o(x^4)\Bigr) + \frac{x^2}{2}
= 1 + \frac{x^4}{24} + o(x^4) :
\]
primeiro termo não nulo $c\,x^p$ com $p = 4$ par e $c =
\frac{1}{24} > 0$: um mínimo
local, de achatamento incomum (o gráfico deixa o seu valor mínimo como
$x^4$, e não como $x^2$). A ideia de fechamento: a expansão vê numa
linha o que a derivação iterada obscurece --- e a \cref{prop:b1:taylor:local} é o dicionário
sistemático que traduz ``primeiro termo sobrevivente'' em ``forma
local''.
\end{example}

\begin{example}[Expansões no infinito]\label{ex:b1:taylor:atinfinity}
Dois cálculos em que a variável tende a $+\infty$ e a substituição
$h = \frac1x \to 0^+$ importa toda a caixa de ferramentas. Primeiro,
\[
x - x^2\ln\Bigl(1 + \frac1x\Bigr)
= x - x^2\Bigl(\frac1x - \frac{1}{2x^2} +
O\Bigl(\frac{1}{x^3}\Bigr)\Bigr)
= \frac12 + O\Bigl(\frac1x\Bigr) \longrightarrow \frac12 .
\]
Segundo, o \omterm{def:b1:functions:arc}{arco-tangente} no infinito: de $\arctan x +
\arctan\frac1x = \frac\pi2$ para $x > 0$ (\cref{prop:b1:functions:arcidentities}) e
da expansão de $\arctan$ em $0$ (\cref{exo:b1:taylor:3}),
\[
\arctan x = \frac\pi2 - \arctan\frac1x
= \frac\pi2 - \frac1x + \frac{1}{3x^3} +
o\Bigl(\frac{1}{x^3}\Bigr) :
\]
o gráfico aproxima-se da sua assíntota $y = \frac\pi2$ por baixo, à velocidade
$\frac1x$. A ideia de fechamento: não há teoria separada de expansões
no infinito --- uma única substituição recíproca reduz todas elas a
expansões em $0$, desde que todo $O$ e todo $o$ intermediário seja
carregado honestamente.
\end{example}

\begin{example}[Assíntota por expansão]\label{ex:b1:taylor:asymptote}
Quando $x \to +\infty$,
\[
\sqrt{x^2 + x} = x\sqrt{1 + \tfrac1x}
= x\Bigl(1 + \frac{1}{2x} - \frac{1}{8x^2} + o\bigl(\tfrac{1}{x^2}\bigr)\Bigr)
= x + \frac12 - \frac{1}{8x} + o\bigl(\tfrac 1x\bigr):
\]
a reta $y = x + \frac12$ é uma assíntota, aproximada \emph{por baixo} (o termo
seguinte $-\frac{1}{8x}$ é negativo).
\end{example}

\begin{remark}[Onde as expansões atuam a seguir]
As expansões assintóticas são a língua corrente do resto do livro: no
\cref{ch:b1:series} elas decidem a convergência (os equivalentes alimentam os testes
de comparação, e o estudo de $\sum
\frac{1}{n^\alpha}$ é uma expansão disfarçada); no
volume do segundo ano de graduação elas tornam-se \emph{séries de
potências}, em que o \omterm{def:b1:poly:def}{polinômio} de Taylor adquire infinitos termos e um
raio de convergência; e toda linearização em física --- o pêndulo, a
perturbação de primeira ordem --- é um enunciado de Taylor--Young com
o $o(\cdot)$ silenciosamente descartado. O único aviso que vale gravar: uma
expansão descreve uma função apenas \emph{perto de um ponto} --- veja
a função plana do problema de fim de semana, cuja expansão em $0$ é
identicamente nula sem que a função o seja.
\end{remark}

\begin{remark}[Perspectivas dentro deste volume]
As expansões são a língua de trabalho do resto da análise e da
geometria que virá. O \Cref{ch:b1:series} converte-as em veredictos de convergência:
um equivalente do termo geral é uma expansão truncada no seu primeiro
termo, e os testes mais finos (séries alternadas com controle do erro)
consomem também o segundo termo. O \Cref{ch:b1:curves} lê a geometria local a partir
das expansões das duas funções coordenadas: se uma curva parametrizada
atravessa, tangencia ou forma uma cúspide num ponto é decidido por
quais potências de $t$ sobrevivem em $x(t)$ e $y(t)$ --- a versão
plana da \cref{prop:b1:taylor:local}. E o \cref{ch:b1:multivar} para de propósito na ordem um: o plano
tangente é um enunciado de Taylor--Young em duas variáveis, com a
teoria completa de segunda ordem (hessianas, pontos de sela) adiada
para o volume do segundo ano de graduação. O fio comum: toda questão
``local'' neste livro é respondida escrevendo o primeiro termo
sobrevivente de uma expansão.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:taylor:1}
Dê as expansões em $0$: $\eu^{2x}$ até a ordem $3$;
$\;\ln(1 - x)$ até a ordem $4$; $\;\sqrt{1 + x}$ até a ordem $3$;
$\;\dfrac{1}{1 + x^2}$ até a ordem $6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:taylor:2}
Calcule os limites:
\[
\lim_{x\to 0} \frac{\eu^x - 1 - x}{x^2},
\qquad
\lim_{x\to 0} \frac{\cos x - \sqrt{1 - x^2}}{x^4},
\qquad
\lim_{x\to 0} \frac{\ln(1+x) - \sin x}{x^2}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:taylor:3}
Expanda $\arctan x$ em $0$ até a ordem $5$ integrando a expansão de $\frac{1}{1 + x^2}$, e
$\arcsin x$ até a ordem $5$ integrando a de $(1 - x^2)^{-1/2}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:taylor:4}
Usando Taylor--Lagrange para $\exp$ em $\intcc{0}{1}$, demonstre que
\[
\Bigl| \eu - \sum_{k=0}^{n} \frac{1}{k!} \Bigr| \leq
\frac{3}{(n+1)!},
\]
e determine um $n$ que garanta $6$ decimais exatas de $\eu$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:5}
Expanda até a ordem $2$ em $\frac1n$ e deduza o limite e a
velocidade de convergência:
\[
\Bigl(1 + \frac 1n\Bigr)^{\!n}
= \eu\Bigl(1 - \frac{1}{2n} + \frac{11}{24n^2} +
o\Bigl(\frac{1}{n^2}\Bigr)\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:6}
Estude o comportamento local em $0$ de $f(x) = x^2 - x^4$ e de $g(x) =
x^3 + x^5$; e determine a
posição do gráfico de $h(x) = \eu^x$ relativamente à sua tangente em $a = 1$,
primeiro localmente e depois globalmente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:7}
Determine as assíntotas em $\pm\infty$ de $f(x) =
\sqrt[3]{x^3 + x^2}$ e a posição da curva
relativamente a elas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:8}
Encontre o equivalente, quando $n \to \infty$, de
\[
u_n = \sqrt{n+1} - \sqrt n, \qquad
v_n = \ln(n+1) - \ln n, \qquad
w_n = \sin\frac{1}{n} - \tan\frac{1}{n},
\]
cada um como uma potência de $n$ vezes uma constante.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:9}
Seja $f$ de classe $C^2$ em $\R$. Demonstre que para todo $x$ e todo $h > 0$:
\[
\abs{f'(x)} \leq \frac{\abs{f(x+h) - f(x-h)}}{2h} +
\frac{h}{2}\sup_{\intcc{x-h}{x+h}}\abs{f''} ,
\]
e deduza a desigualdade do tipo Landau--Kolmogorov: se $\abs f \leq
M_0$ e $\abs{f''} \leq M_2$
em $\R$, então $\abs{f'} \leq
\sqrt{2 M_0 M_2}$ em toda parte. \emph{(Otimize em $h$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:10}
A sequência $u_0 \in \intoo{0}{\pi}$, $u_{n+1} = \sin u_n$
decresce para $0$ (justifique brevemente). Para achar a sua
velocidade, considere $v_n
= \frac{1}{u_n^2}$:
\begin{enumerate}
  \item usando a expansão de $\sin$, demonstre $v_{n+1} - v_n \to
        \frac13$;
  \item com Cesàro (\cref{exo:b1:seq:10}), deduza $\frac{v_n}{n}
        \to \frac13$, e depois o equivalente $u_n \sim
        \sqrt{\dfrac{3}{n}}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:11}
(Diferenças de infinitos) Calcule
\[
\lim_{x \to 0} \Bigl(\frac{1}{x^2} - \frac{1}{\sin^2 x}\Bigr)
\qquad\text{e}\qquad
\lim_{x \to 0^+} \Bigl(\frac 1x - \frac{1}{\ln(1 + x)}\Bigr)
\]
reduzindo ao mesmo denominador e expandindo numerador e denominador
separadamente.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:12}
(Assintótica de raízes implícitas) Mostre que para todo $k \in
\N^*$ a
equação $\tan x = x$ tem exatamente uma solução $x_k$ em $\intoo{k\pi - \frac\pi2}{k\pi + \frac\pi2}$, que $x_k = k\pi
+ \frac\pi2 - \varepsilon_k$ com
$\varepsilon_k =
\arctan\frac{1}{x_k}$, e deduza a expansão
\[
x_k = k\pi + \frac\pi2 - \frac{1}{k\pi} + o\Bigl(\frac
1k\Bigr) \qquad (k \to \infty).
\]
\end{exercise}

\section{Problema: Somas alternadas, dígitos certificados e a
irracionalidade de $\cos 1$}

\begin{problem}[{Problema de fim de semana --- a estimativa alternada
$\abs{S - S_n} \leq a_{n+1}$: $\ln 2$ e $\pi$ com decimais
demonstradas, a fórmula de Machin, e $\cos 1 \notin \Q$}]
\label{pb:b1:taylor:1}
Uma soma alternada com termos decrescentes é o objeto mais amigável da
análise numérica: o seu erro é majorado pelo primeiro termo omitido,
\emph{com sinal conhecido}. Este problema demonstra esse princípio com
o teorema das \omterm{thm:b1:seq:adjacent}{sequências adjacentes}, e depois gasta-o de três
maneiras: decimais certificadas para $\ln 2$ (três vias concorrentes) e
para $\pi$ (Leibniz, e depois a fórmula de Machin de 1706, ainda hoje a
ideia por trás dos cálculos recordistas ao longo de séculos), a
irracionalidade de $\cos 1$, $\sin 1$ e $\cosh 1$ e, como contrapeso, a
\emph{igualdade} de Taylor--Lagrange e a função plana cuja expansão de
Taylor mente.\index{estimativa das séries alternadas}\index{fórmula de
Machin}\index{irracionalidade!de cos 1@de $\cos 1$} Ao longo do texto,
a linguagem de ``série'' é informal: toda soma aqui é uma
\emph{sequência de somas parciais}, como no \cref{ex:b1:seq:e}; a teoria
propriamente dita abre-se no \cref{ch:b1:series}.

\medskip
\noindent\textbf{Parte I --- A estimativa alternada.} Seja
$(a_k)_{k \geq 0}$ decrescente para $0$ e $S_n = \sum_{k=0}^{n}
(-1)^k a_k$.
\begin{enumerate}
  \item Mostre que $(S_{2n+1})$ é não decrescente, $(S_{2n})$ não crescente, e que
        são adjacentes (\cref{thm:b1:seq:adjacent}): ambas convergem para um $S$ comum com,
        para todo $n$,
        \[
        S_{2n+1} \leq S \leq S_{2n},
        \qquad
        \abs{S - S_n} \leq a_{n+1} ,
        \]
        o erro tendo o sinal do primeiro termo omitido.
        Mostre além disso que, se o decrescimento for \emph{estrito},
        todas essas desigualdades são estritas.
  \item Primeiro dividendo: para $x = 1$ na série exponencial,
        compare com o \cref{thm:b1:taylor:lagrange} em $a = 0$:
        mostre que $T_n = \sum_{k=0}^{n} \frac{(-1)^k}{k!}$
        converge para $\eu^{-1}$ com $\abs{\eu^{-1} - T_n} <
        \frac{1}{(n+1)!}$.
  \item (Leibniz, 1674) Da identidade finita \emph{exata}
        \[
        \frac{1}{1 + t^2} = \sum_{k=0}^{n} (-1)^k t^{2k}
        + \frac{(-1)^{n+1} t^{2n+2}}{1 + t^2} ,
        \]
        integrada em $\intcc{0}{1}$, demonstre
        \[
        \frac\pi4 = \sum_{k=0}^{n} \frac{(-1)^k}{2k+1} + \rho_n,
        \qquad \abs{\rho_n} \leq \frac{1}{2n+3} .
        \]
  \item Lentidão: quantos termos de Leibniz garantem seis decimais
        exatas de $\pi$? (Cerca de dois milhões.) Avalie $4 S_4
        = 4\bigl(1 - \frac13 + \frac15 - \frac17 +
        \frac19\bigr)$ e a sua
        distância a $\pi$, para sentir a dor.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- $\ln 2$ de três maneiras.}
\begin{enumerate}[resume]
  \item (Via 1: harmônica alternada) De $\frac{1}{1+t} =
        \sum_{k=0}^{n-1}(-1)^k t^k + \frac{(-1)^n t^n}{1+t}$
        integrada em $\intcc{0}{1}$:
        \[
        \ln 2 = \sum_{k=1}^{n} \frac{(-1)^{k-1}}{k} + (-1)^n
        R_n, \qquad \frac{1}{2(n+1)} \leq R_n \leq
        \frac{1}{n+1} :
        \]
        o erro é da ordem exata de $\frac 1n$ --- um milhão de
        termos para seis decimais.
  \item (Via 2: a série rápida) Integre $\frac{1}{1 - t^2} =
        \sum_{k=0}^{n} t^{2k} + \frac{t^{2n+2}}{1-t^2}$ de $0$
        a $x \in \intoo{0}{1}$ e avalie em $x = \frac13$
        (note $\frac{1 + 1/3}{1 - 1/3} = 2$):
        \[
        \ln 2 = 2\sum_{k=0}^{n} \frac{(1/3)^{2k+1}}{2k+1} +
        \tilde\rho_n,
        \qquad
        0 < \tilde\rho_n \leq \frac{9}{4}\cdot
        \frac{(1/3)^{2n+3}}{2n+3} :
        \]
        convergência geométrica, cerca de um dígito por termo.
  \item (Via 3: \omterm{thm:b1:integration:riemann}{somas de Riemann} e uma identidade escondida) Demonstre
        por indução a identidade
        \[
        \sum_{k=1}^{2n} \frac{(-1)^{k-1}}{k} = H_{2n} - H_n =
        \sum_{k=1}^{n} \frac{1}{n + k} ,
        \]
        e recupere $\ln 2$ como limite de \omterm{thm:b1:integration:riemann}{somas de Riemann} do
        \cref{ex:b1:integration:riemannexample}: as Vias 1 e 3
        são secretamente o mesmo número visto duas vezes.
  \item Duelo com seis termos: compare $\sum_{k=1}^{6}
        \frac{(-1)^{k-1}}{k} = 0.6167$ com a Via 2 em $n = 5$,
        que já dá $\ln 2 = 0.693147$ com erro $\leq
        1.1\cdot10^{-7}$. Explique a razão estrutural
        (ponto de avaliação bem no interior do \omterm{prop:b1:reals:intervals}{intervalo} de
        convergência versus na sua \omterm{def:b1:topology:closure}{fronteira}).
  \item Quantos termos da Via 2 certificam \emph{dez} decimais de
        $\ln 2$? Mostre que $n = 10$ basta.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A fórmula de Machin.}
\begin{enumerate}[resume]
  \item Calcule $(5 + \iu)^4$ e verifique a identidade complexa
        \[
        (5 + \iu)^4 = 2\,(1 + \iu)\,(239 + \iu) .
        \]
        Tomando argumentos (convenções do \cref{ch:b1:complex}),
        deduza a fórmula de Machin
        \[
        \frac\pi4 = 4\arctan\frac15 - \arctan\frac{1}{239} .
        \]
        (Verifique que nenhum argumento sai de
        $\intoo{-\frac\pi2}{\frac\pi2}$.)
  \item Como na questão 3, estabeleça para $0 < x < 1$:
        \[
        \arctan x = \sum_{k=0}^{n} \frac{(-1)^k
        x^{2k+1}}{2k+1} + r_n(x), \qquad \abs{r_n(x)} \leq
        \frac{x^{2n+3}}{2n+3} .
        \]
  \item Certifique $\pi$ com sete decimais usando seis termos: majore
        o erro total de
        \[
        \pi \approx 16\sum_{k=0}^{4}
        \frac{(-1)^k (1/5)^{2k+1}}{2k+1}
        - 4\Bigl(\frac{1}{239} - \frac{1}{3\cdot239^3}\Bigr)
        \]
        por $16\,\frac{(1/5)^{11}}{11} +
        4\,\frac{(1/239)^5}{5} < 5\cdot10^{-8}$, e dê o valor
        resultante $3.1415926\dots$
  \item Compare as três vias para $\pi$ agora disponíveis ---
        Leibniz (questão 4), as integrais de Dalzell do
        \cref{pb:b1:integration:1} (erro $4^{1-5m}$), Machin
        (erro $\approx 16\cdot 5^{-(2n+3)}$) --- em dígitos por
        termo, e explique por que encolher o ponto de avaliação
        vence tudo.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A armadilha dos inteiros, edição
alternada.}
\begin{enumerate}[resume]
  \item Suponha $\cos 1 = \frac pq$. Multiplique o enquadramento alternado estrito
        de $\sum_{k}
        \frac{(-1)^k}{(2k)!}$ (questão 1) por $(2n)!$ com $2n
        \geq q$, e obtenha uma
        contradição: $\cos 1$ é irracional.
  \item Adapte a $\sin 1 = \sum_k \frac{(-1)^k}{(2k+1)!}$
        (multiplicando por $(2n+1)!$): $\sin 1 \notin \Q$. Ambos
        irracionais, e no entanto $\cos^2 1 + \sin^2 1 = 1$: a irracionalidade
        não é estável sob operações algébricas.
  \item O \omterm{def:b1:arith:prime}{primo} não alternado: $\cosh 1 = \sum_k
        \frac{1}{(2k)!}$ (no sentido de somas parciais,
        com a majoração bilateral da cauda $0 < \cosh 1 - \sum_{k \leq n}
        \frac{1}{(2k)!} < \frac{2}{(2n+2)!}$, a ser demonstrada).
        Conclua $\cosh 1 \notin \Q$ pela mesma armadilha.
  \item Vá até $\cos\frac 1m$ para todo inteiro $m \geq 1$:
        multiplique por $m^{2n}(2n)!$ e conclua $\cos\frac1m
        \notin \Q$. Onde é que a mesma
        tentativa falha para $\cos\frac ab$ com $b > 1$ geral? (Identifique o
        denominador que deixa de ser eliminado.)
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Mais afiado e mais sombrio: a forma de
igualdade, e uma função que engana Taylor.}
\begin{enumerate}[resume]
  \item (Taylor--Lagrange, forma de igualdade) Seja $f$ $n + 1$
        vezes \omterm{def:b1:derivative:def}{derivável} entre $a$ e $x$. Defina $g(t) =
        f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(t)}{k!}(x - t)^k -
        A\,\frac{(x-t)^{n+1}}{(n+1)!}$ com a constante
        $A$ escolhida de modo que $g(a) = 0$. Calcule $g'$ (a soma
        telescopa), aplique Rolle em $\intcc{a}{x}$, e conclua
        que existe $c$ estritamente entre $a$ e $x$ com
        \[
        f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k
        + \frac{f^{(n+1)}(c)}{(n+1)!}\,(x-a)^{n+1} .
        \]
  \item Dividendo da igualdade: para $x > 0$ mostre
        \[
        \eu^x > 1 + x + \frac{x^2}{2!} + \dots +
        \frac{x^n}{n!}
        \]
        (estritamente, para todo $n$), e localize onde a
        desigualdade se inverte para $x < 0$ conforme a paridade
        de $n$.
  \item Compare os restos em $\sin(0.5)$ na ordem $3$:
        Young dá apenas $o(x^3)$ (nenhum número); Lagrange dá
        $\abs{\sin 0.5 - (0.5 - \frac{0.5^3}{6})} \leq
        \frac{0.5^5}{120} = 2.61\cdot10^{-4}$; a estimativa alternada dá a mesma cota \emph{mais} a
        informação de sinal $\sin 0.5 > 0.5 - \frac{0.5^3}{6}$. Compare com o erro verdadeiro
        $2.59\cdot10^{-4}$: a cota é quase atingida. Qual ferramenta você
        escolheria, e quando?
  \item (A função plana) Seja $f(x) = \eu^{-1/x^2}$ para $x
        \neq 0$, $f(0) = 0$. Mostre que $f$
        é \omterm{def:b1:continuity:continuous}{contínua} em $0$, que $f'(0) = 0$, e mais geralmente ---
        demonstrando que toda \omterm{def:b1:derivative:def}{derivada} tem a forma $f^{(k)}(x) =
        P_k\bigl(\frac1x\bigr)\eu^{-1/x^2}$ para um
        \omterm{def:b1:poly:def}{polinômio} $P_k$ (indução) --- que $f^{(k)}(0) = 0$ para todo $k$
        \emph{(comparação de crescimento, \cref{prop:b1:functions:powerrules})}. Conclua: todos os
        \omterm{def:b1:poly:def}{polinômios} de Taylor de $f$ em $0$ se anulam, e no entanto
        $f(x) > 0$ para $x \neq 0$: Taylor--Young vale em toda ordem, e nada
        diz sobre $f$ longe de $0$. As expansões descrevem
        \emph{germes}, e não funções.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Síntese.}
\begin{enumerate}[resume]
  \item Aplique a armadilha mais uma vez, a $\eu^{-1} = \sum_k
        \frac{(-1)^k}{k!}$: multiplique o
        enquadramento alternado estrito por $n!$ e conclua $\eu^{-1} \notin \Q$,
        logo $\eu \notin \Q$ --- a terceira demonstração deste fato
        no volume. Liste as três (\omterm{thm:b1:seq:adjacent}{sequências adjacentes},
        \cref{exo:b1:seq:9}; integrais,
        \cref{pb:b1:integration:1}; somas alternadas, aqui) e
        o que cada uma exigiu.
  \item As letras miúdas: a monotonia não é decorativa. Seja
        $b_k = \frac1k$ para $k$ ímpar e $b_k = \frac{1}{k^2}$
        para $k$ par: os $b_k$ são positivos e tendem a $0$,
        e no entanto as somas parciais de $\sum (-1)^k b_k$ divergem para
        $-\infty$. Demonstre-o \emph{(separe a soma parcial na
        parte par, majorada pelo
        \cref{ex:b1:seq:basel}, e na parte ímpar, que
        domina metade da série harmônica,
        \cref{exo:b1:seq:5})}, e diga exatamente qual passo da
        questão 1 usou a monotonia.
  \item Verifique a identidade mais simples de Euler $\arctan\frac12 +
        \arctan\frac13 = \frac\pi4$ via
        $(2 + \iu)(3 + \iu) =
        5(1 + \iu)$, estime os termos necessários para seis decimais
        de $\pi$ por essa via ($n = 10$ basta), e situe-a
        entre Leibniz e Machin na classificação da questão
        13.
  \item Síntese, uma frase para cada: (i) enuncie a estimativa
        alternada e as suas duas saídas (cota e sinal); (ii) por que
        identidades finitas exatas com restos explícitos vencem
        enunciados de limite para a numérica certificada; (iii)
        inventário do problema ($\pi$ com $10^{-7}$ à mão, $\ln 2$
        com dez decimais, quatro demonstrações de irracionalidade,
        um teorema de igualdade, um exemplo de advertência); (iv)
        quais desses fios o \cref{ch:b1:series} retomará (o teste das séries
        alternadas, convergência absoluta versus condicional,
        e o drama do rearranjo do seu problema de fim de semana).
\end{enumerate}
\end{problem}
