\chapter{Contagem}\label{ch:b1:counting}

Contar \omterm{def:b1:counting:card}{conjuntos finitos} parece elementar --- e logo se torna sutil.
Este capítulo define a \omterm{def:b1:counting:card}{cardinalidade} como se deve (por meio de bijeções,
no espírito do \cref{ch:b1:logic}), estabelece o punhado de princípios de
contagem dos quais tudo decorre e deduz as contagens clássicas: listas,
permutações, subconjuntos, \omterm{def:b1:counting:objects}{coeficientes binomiais}.

\section{Cardinalidade dos conjuntos finitos}

\begin{definition}[Conjunto finito, cardinalidade]\label{def:b1:counting:card}
Para $n \in \N^*$, escreva $\intint{1}{n} = \{1, 2, \dots,
n\}$. Um \omterm{def:b1:logic:sets}{conjunto} $E$ é \emph{finito}\index{conjunto finito} quando $E = \emptyset$
ou existe uma bijeção de $\intint{1}{n}$ sobre $E$ para
algum $n \in \N^*$; esse $n$ é único (\cref{thm:b1:counting:welldef})
e é a \emph{cardinalidade}\index{cardinalidade} de $E$, escrita
$\abs{E}$ (com $\abs{\emptyset} = 0$).
\end{definition}

\begin{theorem}[A cardinalidade está bem definida]\label{thm:b1:counting:welldef}
Se $m \neq n$, não existe bijeção de $\intint{1}{m}$
sobre $\intint{1}{n}$. Mais precisamente, se $m > n$ não existe
injeção de $\intint{1}{m}$ em $\intint{1}{n}$.
\end{theorem}

\begin{proof}
Demonstramos por indução em $n$ o enunciado: \emph{para todo $m > n$,
não existe injeção $\intint{1}{m} \to \intint{1}{n}$}. Para $n = 0$ o contradomínio é vazio e $m \geq 1$: não existe
\omterm{def:b1:logic:map}{aplicação} alguma. Suponha o enunciado para $n$ e suponha que $f \colon
\intint{1}{m} \to \intint{1}{n+1}$ seja uma
injeção com $m > n + 1$. Se o valor $n + 1$ não é atingido, $f$
é uma injeção em $\intint{1}{n}$, o que contradiz a hipótese de
indução. Caso contrário, $f(a) = n + 1$ para exatamente um $a$;
troque $f(a)$ e $f(m)$ (formalmente: componha com a transposição dos
dois valores), de modo que a nova injeção $g$ tenha $g(m) = n + 1$.
Então a restrição de $g$ a $\intint{1}{m-1}$ é uma
injeção em $\intint{1}{n}$ com $m - 1 > n$ ---
contradição, novamente.
\end{proof}

\begin{corollary}[Princípio da casa dos pombos]\label{cor:b1:counting:pigeonhole}
Se $\abs{E} > \abs{F}$, nenhuma \omterm{def:b1:logic:map}{aplicação} $f \colon E \to F$ é \omterm{def:b1:logic:inj}{injetiva}: dois
elementos de $E$ partilham a mesma imagem\index{princípio da casa dos pombos}.
\end{corollary}

\begin{proof}
Escreva $\abs E = m$, $\abs F = n$ com $m > n$ e escolha bijeções
$u \colon \intint1m \to E$ e $v \colon F \to \intint1n$. Se $f$
fosse \omterm{def:b1:logic:inj}{injetiva}, $v \circ f \circ u$ seria uma injeção de
$\intint1m$ em $\intint1n$ (composta de injeções,
\cref{prop:b1:logic:comp}), contradizendo o
\cref{thm:b1:counting:welldef}.
\end{proof}

\begin{remark}[Interlúdio: por que a troca na demonstração do teorema?]\label{rem:b1:counting:swapinterlude}
A demonstração do \cref{thm:b1:counting:welldef} contém a primeira
jogada genuinamente engenhosa do capítulo, que vale a pena revisitar
devagar. O obstáculo: para aplicar a hipótese de indução queremos
apagar o último ponto $m$ do domínio \emph{e} o último ponto
$n+1$ do contradomínio, mas $f$ pode enviar algum outro ponto $a$ a $n
+ 1$, e então apagar o ponto do contradomínio estraga a \omterm{def:b1:logic:map}{aplicação} em
outro lugar. O remédio: compor $f$ com a transposição dos
dois \emph{valores} $f(a)$ e $f(m)$ --- uma bijeção do
contradomínio, de modo que a injetividade se preserva --- após o que o
valor incômodo $n + 1$ passa a ocupar a posição inofensiva $m$, e
as duas remoções ficam limpas. Esse padrão ``normalizar primeiro,
cortar depois'' reaparece: é assim que a recorrência dos desarranjos
redireciona $\sigma^{-1}(n+1)$ no problema de fim de semana deste capítulo, e
assim que as permutações são remendadas ao longo de todo o problema do
\cref{ch:b1:structures} sobre o grupo simétrico.
\end{remark}

\begin{proposition}[Injeções, sobrejeções e cardinalidade]\label{prop:b1:counting:injsur}
Sejam $E, F$ \omterm{def:b1:counting:card}{conjuntos finitos} com $\abs{E} = \abs{F}$, e seja $f \colon E
\to F$. Então
\[
f \text{ injetiva} \iff f \text{ sobrejetiva} \iff f \text{ bijetiva}.
\]
\end{proposition}

\begin{proof}
Suponha $f$ \omterm{def:b1:logic:inj}{injetiva}. Então $f$ é uma bijeção de $E$ sobre $f(E)$,
logo $\abs{f(E)} = \abs{E} = \abs{F}$. Se $f(E)$ deixasse de fora um ponto $y_0$ de
$F$, então $f$ seria uma injeção de $E$ em $F \setminus \{y_0\}$,
\omterm{def:b1:logic:sets}{conjunto} de \omterm{def:b1:counting:card}{cardinalidade} $\abs{F} - 1 < \abs{E}$ --- impossível pelo
princípio da casa dos pombos. Logo $f(E) = F$: $f$ é \omterm{def:b1:logic:inj}{sobrejetiva} e,
portanto, \omterm{def:b1:logic:inj}{bijetiva}.

Suponha $f$ \omterm{def:b1:logic:inj}{sobrejetiva}. Escolha para cada $y \in F$ uma \omterm{def:b1:logic:map}{pré-imagem} $s(y)
\in E$; então $f \circ s = \mathrm{id}_F$, de modo que $s$ é \omterm{def:b1:logic:inj}{injetiva}
(\cref{prop:b1:logic:comp}). Pelo parágrafo anterior aplicado a $s$
(as \omterm{def:b1:counting:card}{cardinalidades} são iguais), $s$ é \omterm{def:b1:logic:inj}{bijetiva}. De $f \circ s =
\mathrm{id}_F$ obtemos $f = \mathrm{id}_F \circ s^{-1} = s^{-1}$, logo
$f$ é \omterm{def:b1:logic:inj}{bijetiva}. Por fim, uma \omterm{def:b1:logic:map}{aplicação} \omterm{def:b1:logic:inj}{bijetiva} é, por definição, ao
mesmo tempo \omterm{def:b1:logic:inj}{injetiva} e \omterm{def:b1:logic:inj}{sobrejetiva}, o que fecha o ciclo de implicações.
\end{proof}

\begin{example}[A finitude é essencial]\label{ex:b1:counting:finiteonly}
Num \omterm{def:b1:logic:sets}{conjunto} \emph{\omterm{def:b1:counting:card}{finito}}, a \cref{prop:b1:counting:injsur} é um atalho
poderoso: toda \omterm{def:b1:logic:map}{aplicação} \omterm{def:b1:logic:inj}{injetiva} de $E$ em si mesmo é automaticamente
uma \omterm{def:b1:counting:objects}{permutação} de $E$ --- metade da bijetividade vem de graça. As duas
implicações desmoronam em \omterm{def:b1:logic:sets}{conjuntos} infinitos: $n \mapsto n + 1$ é
\omterm{def:b1:logic:inj}{injetiva} de $\N$ em $\N$, mas não atinge $0$, e a \omterm{def:b1:logic:map}{aplicação} $\N \to \N$
que envia $0 \mapsto 0$ e $n \mapsto n - 1$ para $n \geq 1$ é
\omterm{def:b1:logic:inj}{sobrejetiva}, mas não \omterm{def:b1:logic:inj}{injetiva}. Sempre que esta \omterm{def:b1:logic:statement}{proposição} é invocada,
a hipótese de finitude está fazendo trabalho de verdade --- tema que o
problema de fim de semana do \cref{ch:b1:logic} explora pelo outro lado, em
que os \omterm{def:b1:logic:sets}{conjuntos} infinitos são precisamente os que admitem tais
aplicações de si em si.
\end{example}

\begin{example}[Metade do trabalho, de graça]\label{ex:b1:counting:mod7}
Considere a \omterm{def:b1:logic:map}{aplicação} $f$ em $\{0, 1, \dots, 6\}$ que envia $k$ ao
resto da divisão de $3k$ por $7$; sua tabela de valores é
\[
0,\ 3,\ 6,\ 2,\ 5,\ 1,\ 4 .
\]
$f$ é uma bijeção? A injetividade sozinha basta
(\cref{prop:b1:counting:injsur}): se $3k$ e $3k'$ têm o mesmo
resto, $7$ divide $3(k - k')$ e, como $7$ é primo e
não divide $3$, ele divide $k - k'$ (lema de Euclides, usado aqui no
nível do ensino médio e demonstrado no \cref{ch:b1:arith}); com
$\abs{k - k'} \leq 6$ isso força $k = k'$. A sobrejetividade vem
de graça --- não é preciso resolver $3k \equiv c$ para cada $c$, embora a
tabela confirme que todo valor aparece exatamente uma vez. O atalho é
um cavalo de batalha: demonstra a invertibilidade da multiplicação
modular (\cref{ch:b1:arith}), alimenta o emparelhamento do
teorema de Wilson e reaparece em álgebra linear como ``um
endomorfismo de um espaço de dimensão finita é \omterm{def:b1:logic:inj}{injetivo} se, e somente
se, é \omterm{def:b1:logic:inj}{sobrejetivo}'' (\cref{ch:b1:findim}).
\end{example}

\section{Os princípios de contagem}

\begin{proposition}[Regras da soma e do produto]\label{prop:b1:counting:rules}
Sejam $E, F$ \omterm{def:b1:counting:card}{conjuntos finitos}.
\begin{enumerate}
  \item Se $E \cap F = \emptyset$, então $\abs{E \cup F} = \abs{E} +
        \abs{F}$; mais geralmente, para uma \omterm{thm:b1:logic:partition}{partição} de $E$ em peças
        $E_1, \dots, E_k$, $\abs{E} = \sum_i \abs{E_i}$.
  \item Em geral, $\abs{E \cup F} = \abs{E} + \abs{F} - \abs{E \cap
        F}$.
  \item $\abs{E \times F} = \abs{E} \times \abs{F}$.
  \item O \omterm{def:b1:logic:sets}{conjunto} $F^E$ de todas as aplicações de $E$ em $F$ satisfaz
        $\abs{F^E} = \abs{F}^{\abs{E}}$.
  \item $\abs{\mathcal{P}(E)} = 2^{\abs{E}}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Concatene enumerações: se $E = \{x_1, \dots, x_m\}$ e $F =
\{y_1, \dots, y_n\}$ sem repetição, então $x_1, \dots, x_m, y_1,
\dots, y_n$ enumera $E \cup F$ sem repetição (por serem disjuntos).
A indução estende isso a $k$ peças.

(2) $E \cup F$ é a união disjunta de $E$ e $F \setminus E$, e
$F$ é a união disjunta de $F \cap E$ e $F \setminus E$; logo
$\abs{E \cup F} = \abs{E} + \abs{F \setminus E} = \abs{E} + \abs{F} -
\abs{E \cap F}$.

(3) $E \times F$ é a união disjunta, sobre $x \in E$, dos \omterm{def:b1:logic:sets}{conjuntos}
$\{x\} \times F$, cada um de \omterm{def:b1:counting:card}{cardinalidade} $\abs{F}$; aplique (1).

(4) Uma \omterm{def:b1:logic:map}{aplicação} de $E = \{x_1, \dots, x_m\}$ em $F$ é exatamente a escolha
da $m$-upla $(f(x_1), \dots, f(x_m)) \in F^m$; essa
correspondência é uma bijeção, e $\abs{F^m} = \abs{F}^m$ por (3) e
indução.

(5) Os subconjuntos de $E$ correspondem bijetivamente às aplicações $E \to \{0, 1\}$
(envie $A$ à sua função indicadora); aplique (4).
\end{proof}

\begin{example}[Contagem pelo complementar]\label{ex:b1:counting:complement}
Quantas senhas de $4$ dígitos (algarismos de $0$ a $9$, a ordem importa,
repetições permitidas) contêm \emph{pelo menos um} algarismo repetido?
Contá-las diretamente significa manejar os casos ``exatamente um par,
dois pares, uma trinca, uma quadra'' --- cinco configurações que se
sobrepõem. Conte antes o complementar: as senhas em geral somam
$10^4 = 10\,000$ (regra do produto) e as senhas com quatro algarismos
distintos somam $10 \times 9 \times 8 \times 7 = 5\,040$
($4$-arranjos), de modo que a resposta é
\[
10^4 - 10 \cdot 9 \cdot 8 \cdot 7 = 10\,000 - 5\,040 = 4\,960 .
\]
Quase metade de todas as senhas repete um algarismo. A ideia: sempre que
uma contagem é formulada com ``pelo menos'' ou ``nem todos'', tente
primeiro o complementar --- a regra da soma garante que
$\abs{A} = \abs{E} - \abs{\overline A}$, e o complementar é
muitas vezes uma única configuração limpa.
\end{example}

\begin{example}[Caminhos na grade]\label{ex:b1:counting:paths}
Conte os caminhos mínimos do canto $(0,0)$ ao canto
$(4, 3)$ de uma grade, movendo-se apenas um passo para a direita (D) ou
um passo para cima (C) de cada vez. Todo caminho desses tem exatamente
$7$ passos, dos quais $4$ são D e $3$ são C; reciprocamente, qualquer
palavra de comprimento $7$ nas letras D, C com quatro D descreve
exatamente um caminho. Os caminhos correspondem, portanto, bijetivamente
às escolhas das posições dos D:
\[
\binom{7}{4} = 35 .
\]
A ideia é a \emph{codificação}: a contagem tornou-se trivial no
momento em que cada caminho foi traduzido em uma palavra, isto é, em um
subconjunto de posições --- mais uma instância do lema de que uma
contagem correta é uma bijeção disfarçada (\cref{met:b1:counting:which}).
\end{example}

\section{Listas, permutações, subconjuntos}

\begin{definition}[Arranjos, permutações, combinações]\label{def:b1:counting:objects}
Seja $E$ um \omterm{def:b1:logic:sets}{conjunto} com $\abs{E} = n$ e seja $0 \leq k \leq n$.
\begin{itemize}
  \item Um \emph{$k$-arranjo} de $E$ é uma $k$-upla \omterm{def:b1:logic:inj}{injetiva} de
        elementos de $E$ (uma seleção ordenada sem repetição);
  \item uma \emph{permutação}\index{permutação} de $E$ é uma bijeção
        de $E$ em si mesmo --- equivalentemente, um $n$-arranjo;
  \item uma \emph{$k$-combinação} é um subconjunto de $E$ com $k$
        elementos (uma seleção não ordenada e sem repetição). O seu
        número escreve-se $\binom{n}{k}$, lido ``combinações de $n$, $k$ a $k$''
        \index{coeficiente binomial}.
\end{itemize}
\end{definition}

\begin{theorem}[As três contagens]\label{thm:b1:counting:counts}
Com $n = \abs{E}$ e $0 \leq k \leq n$:
\begin{enumerate}
  \item o número de $k$-arranjos de $E$ é $n (n-1) \cdots
        (n-k+1) = \dfrac{n!}{(n-k)!}$;
  \item o número de permutações de $E$ é $n!$;
  \item $\dbinom{n}{k} = \dfrac{n!}{k!\,(n-k)!}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Escolha a primeira coordenada ($n$ modos), depois a segunda ($n - 1$
escolhas restantes), \dots, depois a $k$-ésima ($n - k + 1$ escolhas).
Formalmente, faça indução em $k$. Para $k = 1$ há $n$ uplas \omterm{def:b1:logic:inj}{injetivas} de
um só termo. Suponha a contagem para $k - 1$. Cada
$k$-arranjo $(x_1, \dots, x_k)$ é obtido de exatamente um
$(k-1)$-arranjo --- o seu truncamento $(x_1, \dots, x_{k-1})$ ---
acrescentando uma última coordenada fora de $\{x_1, \dots,
x_{k-1}\}$, para a qual há exatamente $n - (k - 1)$ valores disponíveis.
Os $k$-arranjos ficam assim repartidos, pelo truncamento, em
classes de tamanho comum $n - k + 1$ indexadas pelos
$(k-1)$-arranjos, e a regra da soma dá
\[
\frac{n!}{(n-k+1)!}\;(n - k + 1) = \frac{n!}{(n-k)!} .
\]

(2) é (1) com $k = n$.

(3) Cada $k$-subconjunto se ordena de $k!$ modos distintos, dando
$k$-arranjos distintos, e todo $k$-arranjo provém de exatamente um
subconjunto: logo $\frac{n!}{(n-k)!} = \binom nk \cdot k!$.
\end{proof}

\begin{example}[Mesas redondas: quocientar pela simetria]\label{ex:b1:counting:roundtable}
De quantos modos $n$ convidados podem se sentar em torno de uma mesa
redonda, sendo duas disposições idênticas quando cada convidado tem os
mesmos vizinhos à esquerda e à direita --- isto é, a menos de rotação?
Cada disposição circular corresponde a exatamente $n$ disposições
lineares (corte o círculo em qualquer dos $n$ lugares), de modo que as
$n!$ ordens lineares se agrupam de $n$ em $n$:
\[
\frac{n!}{n} = (n-1)! \quad\text{disposições circulares.}
\]
Equivalentemente: sente um convidado distinguido em qualquer lugar
(matando a liberdade de rotação) e depois ordene os $n - 1$ convidados
restantes no sentido horário. Para $n = 6$: $120$ mesas. As duas
soluções ilustram os dois remédios usuais para a contagem excessiva:
dividir pelo número exato de repetições ou \emph{quebrar a simetria}
fixando um objeto. Ambos exigem que o grupo de repetições tenha o mesmo
tamanho em toda configuração --- o que a demonstração da fórmula
$\binom nk = \frac{n!}{k!\,(n-k)!}$ acima também usou, com $k!$ no lugar de $n$.
\end{example}

\begin{example}[Acrescentando uma restrição]\label{ex:b1:counting:constraint}
Continuando com a mesa redonda: entre as $(n-1)!$ mesas de $n \geq
3$
convidados, quantas mantêm dois convidados dados $A$ e $B$
\emph{separados} (não adjacentes)? Conte o complementar. Mesas em que
$A$ e $B$ se sentam juntos: cole-os num único bloco --- $n - 1$ objetos
em torno da mesa, isto é, $(n-2)!$ disposições circulares --- e depois
ordene o par dentro do bloco ($2$ modos): $2\,(n-2)!$ mesas com
adjacência. Portanto
\[
(n-1)! - 2\,(n-2)! = (n-2)!\,\bigl((n - 1) - 2\bigr)
= (n-3)\,(n-2)!
\]
mesas os mantêm separados. Verificações de sanidade: $n = 3$ dá $0$ (em
torno de um triângulo, todos se tocam) e $n = 4$ dá $2$, facilmente
listadas à mão. O truque da colagem --- tratar um bloco forçado como um
único objeto e depois contar suas disposições internas --- é o remédio
usual para restrições de adjacência, lineares ou circulares.
\end{example}

\begin{proposition}[Identidades básicas]\label{prop:b1:counting:identities}
Para $0 \leq k \leq n$:
\[
\binom{n}{k} = \binom{n}{n-k},
\qquad
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}
\quad (1 \leq k \leq n-1),
\qquad
\sum_{k=0}^{n} \binom{n}{k} = 2^n .
\]
\end{proposition}

\begin{proof}
Primeira identidade: $A \mapsto E \setminus A$ é uma bijeção entre
os $k$-subconjuntos e os $(n-k)$-subconjuntos. Regra de
Pascal\index{regra de Pascal}: fixe um elemento $a \in E$; os
$k$-subconjuntos se repartem entre os que contêm $a$ (escolha os $k - 1$
restantes: $\binom{n-1}{k-1}$) e os que evitam $a$ ($\binom{n-1}{k}$). Terceira
identidade: os dois lados contam todos os subconjuntos de $E$, à
esquerda separados por tamanho
(\cref{prop:b1:counting:rules}~(1) e (5)).
\end{proof}

\begin{theorem}[Teorema binomial]\label{thm:b1:counting:binomial}
Para todos $a, b$ num anel comutativo (digamos $\R$ ou $\C$) e $n \in \N$:
\[
(a + b)^n = \sum_{k=0}^{n} \binom{n}{k} a^k b^{\,n-k} .
\]
\end{theorem}

\begin{proof}
Expandir $(a+b)(a+b)\cdots(a+b)$ distributivamente produz um termo por
escolha, em cada fator, de $a$ ou de $b$: o termo $a^k b^{n-k}$
aparece uma vez para cada modo de escolher quais $k$ dos $n$ fatores
contribuem com $a$ --- isto é, $\binom nk$ vezes. (Alternativamente: faça
indução em $n$ usando a regra de Pascal.)
\end{proof}

\begin{example}\label{ex:b1:counting:binomial}
Duas especializações clássicas: $a = b = 1$ recupera $\sum_k \binom nk
= 2^n$; $a = -1$, $b = 1$ dá $\sum_{k} (-1)^k \binom nk = 0$ para
$n \geq 1$: entre os subconjuntos de um \omterm{def:b1:logic:sets}{conjunto} não vazio, exatamente
metade tem \omterm{def:b1:counting:card}{cardinalidade} par.
\end{example}

\begin{example}[Uma identidade, duas demonstrações]\label{ex:b1:counting:threen}
A especialização $a = 2$, $b = 1$ do teorema binomial diz que
\[
\sum_{k=0}^{n} \binom nk\,2^k = 3^n .
\]
Eis a mesma identidade sem álgebra nenhuma. O lado direito
conta as palavras de comprimento $n$ sobre o alfabeto $\{0, 1, 2\}$
(regra do produto). Classifique cada palavra pelo \omterm{def:b1:logic:sets}{conjunto} $K$ das
posições que carregam uma letra não nula: escolher $K$ com $\abs K = k$ custa
$\binom nk$, e depois cada posição de $K$ carrega, independentemente, $1$
ou $2$: $2^k$ modos. A regra da soma sobre $k$ dá o lado esquerdo.
Além do prazer da concordância, as duas demonstrações têm virtudes
diferentes: a algébrica se generaliza a qualquer valor de $a$; a
combinatória \emph{explica} a fórmula e se adapta a restrições
(proibir a letra $2$ na última posição, digamos) que substituição
alguma capta. Manter as duas técnicas em atividade é a habilidade
prática que este capítulo treina.
\end{example}

\begin{method}[Que contagem se aplica?]\label{met:b1:counting:which}
Antes de calcular, responda a duas perguntas sobre a seleção: a
\emph{ordem} importa e as \emph{repetições} são permitidas?
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{l|c|c}
 & a ordem importa & a ordem não importa \\
\hline
sem repetição & $\dfrac{n!}{(n-k)!}$ & $\dbinom{n}{k}$ \\[6pt]
com repetição & $n^k$ & (\cref{exo:b1:counting:10}) \\
\end{tabular}
\end{center}
Depois, procure uma bijeção ou uma \omterm{thm:b1:logic:partition}{partição} que reduza o problema a
essas contagens modelo; uma contagem correta é uma bijeção disfarçada.
\end{method}

\begin{remark}[Armadilhas frequentes em contagem]\label{rem:b1:counting:pitfalls}
\begin{enumerate}
  \item \emph{Somar casos não disjuntos.} A regra da soma exige uma
        \omterm{thm:b1:logic:partition}{partição}; se configurações podem satisfazer dois casos ao
        mesmo tempo, elas são contadas duas vezes --- o remédio é a
        inclusão--exclusão (\cref{thm:b1:counting:inclexcl}) ou
        uma separação de casos mais fina.
  \item \emph{Ordenado versus não ordenado.} Escolher ``uma comissão de
        duas pessoas'' é $\binom n2$, e não $n(n-1)$: decida \emph{antes de
        calcular} se a seleção carrega uma ordem e, se a contagem
        ordenada for mais fácil, divida no fim pelo número de
        ordenações --- mas só quando cada objeto não ordenado provier
        do \emph{mesmo} número de objetos ordenados.
  \item \emph{Escolhas em várias etapas que não são independentes.}
        A regra do produto exige que o número de opções em cada etapa
        seja independente das escolhas anteriores. ``Escolha um
        capitão e depois um vice-capitão diferente'' é legítimo
        ($n(n-1)$); ``escolha dois jogadores que se deem bem'' não é
        de modo algum um produto em duas etapas.
  \item \emph{Contagem dupla por construção.} Construir cada objeto
        duas vezes --- por exemplo, contar as mãos com \emph{pelo
        menos} um ás como (escolher um ás) $\times$ (escolher mais $4$
        cartas) --- conta em excesso as mãos com dois ases. ``Pelo
        menos'' quase sempre pede o complementar
        (\cref{ex:b1:counting:complement}).
\end{enumerate}
\end{remark}

\begin{example}[Uma contagem de pôquer]\label{ex:b1:counting:poker}
De um baralho de $52$ cartas, o número de mãos de $5$ cartas é
$\binom{52}{5} = 2\,598\,960$. Mãos contendo exatamente um ás:
escolha o ás ($4$ modos) e depois $4$ cartas entre as $48$ que não são
ases: $4 \binom{48}{4} = 778\,320$. A regra do produto se aplica porque a
escolha se divide em etapas independentes.
\end{example}

\begin{method}[Contagem dupla]\label{met:b1:counting:doublecount}
Para demonstrar uma identidade entre duas expressões de contagem,
encontre um único \omterm{def:b1:counting:card}{conjunto finito} que ambos os lados contam ---
tipicamente um \omterm{def:b1:logic:sets}{conjunto} de \emph{pares} --- e avalie sua \omterm{def:b1:counting:card}{cardinalidade}
em duas ordens diferentes. O protótipo é o \emph{lema do aperto de
mão}: numa festa, conte os pares (pessoa, mão apertada). Somando sobre
as pessoas obtém-se $\sum_p d_p$ (o número de apertos de mão de cada
pessoa $p$); somando sobre os apertos obtém-se o dobro do número de
apertos (cada um envolve duas pessoas). Portanto $\sum_p d_p$ é par
--- de modo que o número de pessoas que apertaram um número ímpar de
mãos é sempre par, conclusão não trivial obtida sem fórmula alguma. O
mesmo motor aciona o \cref{exo:b1:counting:12} e várias questões do problema
de fim de semana abaixo.
\end{method}

\begin{example}[O subconjunto médio]\label{ex:b1:counting:averagesubset}
Qual é a \omterm{def:b1:counting:card}{cardinalidade} média de um subconjunto de um \omterm{def:b1:logic:sets}{conjunto} $E$ de $n$
elementos, sendo todos os $2^n$ subconjuntos igualmente prováveis?
Conte duas vezes os pares $(A, a)$ com $a \in A$: somando sobre os
subconjuntos obtém-se $\sum_A \abs A$, o total que queremos; somando sobre
os elementos obtém-se $n \cdot 2^{n-1}$ (cada um dos $n$ elementos está em
exatamente metade dos subconjuntos --- emparelhe cada $A$ que contém
$a$ com $A
\setminus \{a\}$). Portanto
\[
\frac{1}{2^n}\sum_{A \subseteq E} \abs A
= \frac{n\,2^{n-1}}{2^n} = \frac n2 :
\]
os subconjuntos estão, em média, pela metade --- como a simetria $A
\leftrightarrow \overline A$ (que emparelha os tamanhos $k$ e $n - k$)
também prevê. Duas demonstrações, uma só resposta, e ambas evitam o
cálculo direto $\sum_k k\binom nk$ do \cref{exo:b1:counting:5}: um
emparelhamento bem escolhido substitui muitas vezes uma identidade.
\end{example}

\section{Inclusão--exclusão}

\begin{theorem}[Inclusão--exclusão]\label{thm:b1:counting:inclexcl}
Para \omterm{def:b1:counting:card}{conjuntos finitos} $A_1, \dots, A_p$:
\[
\Bigl|\, \bigcup_{i=1}^{p} A_i \,\Bigr|
= \sum_{\emptyset \neq I \subseteq \intint{1}{p}}
(-1)^{\abs{I}+1} \Bigl|\, \bigcap_{i \in I} A_i \,\Bigr| .
\]
Para $p = 3$:
$\abs{A \cup B \cup C} = \abs A + \abs B + \abs C - \abs{A \cap B} -
\abs{A \cap C} - \abs{B \cap C} + \abs{A \cap B \cap C}$.
\end{theorem}

\begin{proof}
Fixe um elemento $x$ da união e conte a sua contribuição ao
lado direito. Seja $J = \{i : x \in A_i\}$, de \omterm{def:b1:counting:card}{cardinalidade} $m \geq
1$. O elemento $x$ é contado uma vez em $\abs{\bigcap_{i \in I} A_i}$
exatamente quando $\emptyset \neq I \subseteq J$, com sinal
$(-1)^{\abs I + 1}$; sua contribuição total é
\[
\sum_{k=1}^{m} \binom{m}{k} (-1)^{k+1}
= 1 - \sum_{k=0}^{m} \binom mk (-1)^k = 1 - 0 = 1
\]
pelo \cref{ex:b1:counting:binomial}. Assim, cada elemento da união é
contado exatamente uma vez.
\end{proof}

\begin{example}[Contando os inteiros coprimos]\label{ex:b1:counting:coprime}
Quantos inteiros de $\intint1{120}$ são coprimos com $120 = 2^3
\times 3 \times 5$? Um inteiro tem um fator comum com $120$ exatamente
quando é divisível por $2$, $3$ ou $5$; conte então o complementar
de $A_2 \cup A_3 \cup A_5$, em que $A_d$ reúne os múltiplos de
$d$. Dentro de $\intint1{120}$, os múltiplos de $d$ somam $120/d$
sempre que $d$ divide $120$ --- sem necessidade de partes inteiras --- e
$A_2 \cap A_3 = A_6$, etc. Inclusão--exclusão:
\[
\abs{A_2 \cup A_3 \cup A_5}
= 60 + 40 + 24 - 20 - 12 - 8 + 4 = 88 ,
\]
de modo que $120 - 88 = 32$ inteiros são coprimos com $120$. É
instrutivo reagrupar o cálculo como um produto:
\[
120 - 88 = 120\Bigl(1 - \frac12\Bigr)\Bigl(1 -
\frac13\Bigr)\Bigl(1 - \frac15\Bigr) = 120 \cdot \frac12 \cdot
\frac23 \cdot \frac45 = 32 :
\]
expandir os três parênteses reproduz exatamente os oito
termos com sinal da inclusão--exclusão, um por subconjunto de $\{2, 3,
5\}$. Essa forma de produto define a função totiente de Euler, cujo
papel aritmético aparece com as congruências do \cref{ch:b1:arith} e é
desenvolvido no volume do segundo ano de graduação.
\end{example}

\begin{example}[Desarranjos]\label{ex:b1:counting:derangement}
Um \emph{desarranjo} é uma \omterm{def:b1:counting:objects}{permutação} sem ponto fixo. Seja $A_i$
o \omterm{def:b1:logic:sets}{conjunto} das permutações de $\intint{1}{n}$ que fixam $i$;
então $\abs{\bigcap_{i \in I} A_i} = (n - \abs I)!$, e a
inclusão--exclusão conta as permutações com pelo menos um ponto
fixo; os desarranjos somam
\[
D_n = n! \sum_{k=0}^{n} \frac{(-1)^k}{k!} .
\]
Como $\sum (-1)^k / k! \to \eu^{-1}$ (veja o \cref{ch:b1:series}), cerca de
$37\%$ de todas as permutações são desarranjos, seja qual for $n$.
\end{example}

\begin{remark}[Onde este capítulo é usado]\label{rem:b1:counting:whereused}
Os \omterm{def:b1:counting:objects}{coeficientes binomiais} são os objetos deste capítulo mais reutilizados:
eles conduzem o teorema binomial no \cref{ch:b1:poly}
(expansão de $(X + a)^n$), a fórmula de Leibniz para a $n$-ésima
derivada de um produto no \cref{ch:b1:derivative} e os
coeficientes das expansões de Taylor no \cref{ch:b1:taylor}.
As permutações voltam como um grupo --- com o sinal construído a partir
da contagem das inversões --- no \cref{ch:b1:structures}, e o
sinal, por sua vez, define os determinantes no \cref{ch:b1:det}.
A inclusão--exclusão e os princípios de contagem são a espinha dorsal
finita da probabilidade discreta, desenvolvida no volume do segundo ano
de graduação; os números de desarranjos do \cref{ex:b1:counting:derangement} são
estudados a fundo no problema de fim de semana abaixo.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:counting:1}
Uma placa de veículo é formada por duas letras (A--Z), depois três
algarismos e depois duas letras. Quantas placas são possíveis? E quantas
sem letra repetida entre as quatro?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:2}
Quantos anagramas (rearranjos das letras, com ou sem sentido) tem a
palavra \textsc{orange}? E \textsc{banana}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:3}
Uma comissão de $4$ pessoas é escolhida entre $7$ mulheres e $5$ homens.
Quantas comissões há: no total? com exatamente $2$ mulheres? com pelo
menos um homem?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:4}
Demonstre que, em qualquer grupo de $13$ pessoas, duas fazem aniversário
no mesmo mês; e que, entre quaisquer $n + 1$ inteiros escolhidos em
$\intint{1}{2n}$, dois são consecutivos. \emph{(Casa dos pombos nas duas
vezes: nomeie as casas.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:5}
Calcule $\sum_{k=0}^{n} k \binom{n}{k}$. \emph{Sugestão: derive
$(1 + x)^n$, ou use $k \binom nk = n \binom{n-1}{k-1}$ (demonstre esta igualdade).}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:6}
Quantas aplicações estritamente crescentes existem de $\intint{1}{k}$ em $\intint{1}{n}$? Deduza o número de aplicações
crescentes (não necessariamente estritamente). \emph{Sugestão para a
segunda contagem: $f$ crescente $\mapsto$ $g(i) = f(i) + i - 1$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:7}
(Vandermonde) Demonstre, contando os $k$-subconjuntos de um \omterm{def:b1:logic:sets}{conjunto}
repartido em dois blocos de tamanhos $m$ e $n$:
\[
\binom{m+n}{k} = \sum_{j=0}^{k} \binom{m}{j} \binom{n}{k-j} .
\]
Deduza que $\sum_{j=0}^{n} \binom nj^2 = \binom{2n}{n}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:8}
Quantos inteiros de $\intint{1}{1000}$ são divisíveis por
$2$, por $3$ ou por $5$? (Inclusão--exclusão; $\lfloor 1000/6 \rfloor$
conta os múltiplos de $6$, etc.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:9}
Conte as sobrejeções de um \omterm{def:b1:logic:sets}{conjunto} de $4$ elementos sobre um \omterm{def:b1:logic:sets}{conjunto}
de $2$ elementos; depois sobre um \omterm{def:b1:logic:sets}{conjunto} de $3$ elementos.
\emph{Sugestão: conte as aplicações não \omterm{def:b1:logic:inj}{sobrejetivas} com
inclusão--exclusão sobre os valores não atingidos.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:10}
(Estrelas e barras) Demonstre que o número de $k$-seleções de $n$
objetos \emph{com} repetição, ignorando a ordem --- equivalentemente, o
número de $(x_1, \dots, x_n) \in \N^n$ com $x_1 + \dots + x_n = k$ --- é $\binom{n + k - 1}{k}$. \emph{Sugestão: codifique uma solução como uma fila
de $k$ estrelas e $n - 1$ barras.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:counting:11}
Demonstre em detalhe a fórmula do \cref{ex:b1:counting:derangement} para $D_n$ e
deduza que $n! = \sum_{k=0}^{n} \binom{n}{k} D_{n-k}$ (demonstre também
esta identidade diretamente, classificando as permutações pelo seu
\omterm{def:b1:logic:sets}{conjunto} de pontos fixos).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:counting:12}
Para $n \in \N^*$, demonstre por uma contagem dupla de pares
(subconjunto, elemento marcado):
\[
\sum_{k=1}^{n} k \binom{n}{k} = n\, 2^{n-1},
\qquad\text{então}\qquad
\sum_{k=1}^{n} k^2 \binom{n}{k} = n(n+1)\, 2^{n-2} .
\]
\emph{Para a segunda: conte pares de elementos marcados, iguais ou não.}
\end{exercise}

\section{Problema: Desarranjos, ou as cartas trocadas}

\begin{problem}\label{pb:b1:counting:1}
Uma secretária coloca $n$ cartas em $n$ envelopes endereçados ao acaso:
qual é a chance de que \emph{ninguém} receba a carta certa?
Esta questão clássica (Montmort, 1708) leva aos números de desarranjos
$D_n$ do \cref{ex:b1:counting:derangement}. A fórmula de
inclusão--exclusão é apenas a jogada de abertura: este problema
desenvolve as recorrências que calculam $D_n$, duas outras demonstrações
independentes da fórmula, o notável teorema de que $D_n$ é o inteiro
mais próximo de $n!/\eu$, a distribuição completa dos pontos fixos de
uma \omterm{def:b1:counting:objects}{permutação} aleatória e a curiosa aritmética da sequência
$(D_n)$.\index{desarranjo}
Ao longo do problema, $D_n$ denota o número de desarranjos (permutações
sem pontos fixos) de $\intint1n$, com a convenção $D_0 = 1$
(a \omterm{def:b1:counting:objects}{permutação} vazia não tem ponto fixo).

\medskip
\noindent\textbf{Parte I --- Casos pequenos e o censo dos pontos
fixos.}
\begin{enumerate}
  \item Calcule $D_1, D_2, D_3$ diretamente, e $D_4$ listando os
        desarranjos de $\{1, 2, 3, 4\}$ agrupados pelo valor de
        $\sigma(1)$. (Você deve encontrar $D_4 = 9$.)
  \item Para $0 \leq k \leq n$, mostre que o número $P_k(n)$ de
        permutações de $\intint1n$ com \emph{exatamente} $k$ pontos
        fixos é $\binom nk D_{n-k}$.
  \item Verifique o censo para $n = 4$: calcule $P_0(4), \dots,
        P_4(4)$ e confira que a soma é $4! = 24$. O que é mais
        provável com quatro cartas: nenhum acerto ou exatamente um?
  \item Por contagem dupla (\cref{met:b1:counting:doublecount})
        dos pares $(\sigma, i)$ com $\sigma(i) = i$, mostre que
        \[
        \sum_{\sigma} \abs{\mathrm{Fix}(\sigma)} = n! :
        \]
        em média, uma \omterm{def:b1:counting:objects}{permutação} aleatória tem \emph{exatamente um}
        ponto fixo, seja qual for $n \geq 1$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Duas recorrências e duas novas
demonstrações da fórmula.}
\begin{enumerate}
  \item Demonstre combinatoriamente, para $n \geq 1$:
        \[
        D_{n+1} = n\,(D_n + D_{n-1}) .
        \]
        (Classifique os desarranjos $\sigma$ de $\intint1{n+1}$ por
        $j = \sigma(n+1)$ e depois conforme $\sigma(j) = n + 1$ ou não; no
        caso $\sigma(j) \neq n+1$, construa uma bijeção com os
        desarranjos de $\intint1n$ redirecionando a \omterm{def:b1:logic:map}{pré-imagem} de
        $n + 1$ para $j$.) Verifique a recorrência numericamente até
        $D_6$.
  \item Pondo $u_n = D_n - n D_{n-1}$, deduza da questão 5
        que $u_{n+1} = -u_n$, e conclua a segunda recorrência:
        \[
        D_n = n D_{n-1} + (-1)^n \qquad (n \geq 1).
        \]
  \item A partir da questão 6, demonstre por indução a fórmula do
        \cref{ex:b1:counting:derangement},
        \[
        D_n = n! \sum_{k=0}^{n} \frac{(-1)^k}{k!},
        \]
        --- demonstração inteiramente independente da
        inclusão--exclusão.
  \item (Inversão binomial) Sejam $(a_n)$ e $(b_n)$ duas
        sequências tais que $a_n = \sum_{k=0}^n \binom nk b_k$ para
        todo $n$. Demonstre que
        \[
        b_n = \sum_{k=0}^{n} (-1)^{n-k} \binom nk a_k
        \qquad (n \in \N).
        \]
        (Estabeleça primeiro a \emph{revisão trinomial}
        $\binom nk \binom kj = \binom nj \binom{n-j}{k-j}$, e depois
        use a soma alternada de uma linha do
        \cref{ex:b1:counting:binomial}.)
  \item Aplique a questão 8 à identidade $n! = \sum_k \binom nk
        D_{n-k}$ do \cref{exo:b1:counting:11} para obter uma
        \emph{terceira} demonstração da fórmula de $D_n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O inteiro mais próximo de
$n!/\eu$.} Admita nesta parte --- a teoria é construída no
\cref{ch:b1:series} --- que
$\eu^{-1} = \lim_{n \to \infty} s_n$, em que $s_n = \sum_{k=0}^{n}
\frac{(-1)^k}{k!}$, com a estimativa estrita das séries alternadas
$\abs{\eu^{-1} - s_n} < \frac1{(n+1)!}$ para todo $n$.
\begin{enumerate}
  \item Mostre que $\bigl| D_n - n!/\eu \bigr| < \frac1{n+1}$ para
        todo $n \in \N$.
  \item Deduza o teorema central: \emph{para todo $n \geq 1$,
        $D_n$ é o inteiro mais próximo de $n!/\eu$}. Por que o
        argumento precisa de $n \geq 1$?
  \item Determine o sinal do erro: mostre que $D_n > n!/\eu$
        exatamente quando $n$ é par. (Localize o primeiro termo
        desprezado da série alternada.)
  \item Calcule $D_7$ até $D_{10}$ com a recorrência da
        questão 5 e depois confira $D_{10}$ contra $10!/\eu$
        ($10! = 3\,628\,800$, $\eu \approx 2.718281828$).
  \item (A probabilidade do chapeleiro) Seja $p_n = D_n/n!$ a
        probabilidade de que uma \omterm{def:b1:counting:objects}{permutação} uniformemente aleatória seja
        um desarranjo. Mostre que $\abs{p_n - \eu^{-1}} < \frac1{(n+1)!}$
        e calcule $p_6$ com cinco casas decimais. Comente: por que a
        resposta à questão de Montmort é essencialmente independente
        de $n$ --- já para uma dúzia de cartas?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A distribuição dos pontos fixos.}
\begin{enumerate}
  \item Fixe $k \in \N$. Mostre que a proporção das permutações
        de $\intint1n$ com exatamente $k$ pontos fixos satisfaz
        \[
        \frac{P_k(n)}{n!} = \frac{s_{n-k}}{k!}
        \;\xrightarrow[n \to \infty]{}\; \frac{\eu^{-1}}{k!} .
        \]
        (Esses valores-limite, cuja soma é $1$, formam a
        \emph{distribuição de Poisson} de parâmetro $1$, objeto
        central do curso de probabilidade do volume do segundo ano de
        graduação.)
  \item Por contagem dupla das triplas $(\sigma, i, j)$, em que $i
        \neq j$ são ambos fixados por $\sigma$, mostre que
        $\sum_\sigma \abs{\mathrm{Fix}(\sigma)}\,
        (\abs{\mathrm{Fix}(\sigma)} - 1) = n!$ para $n \geq 2$.
        Combinado com a questão 4: a média de
        $\abs{\mathrm{Fix}}^2$ é $2$, de modo que a ``dispersão'' (variância)
        do número de pontos fixos vale $1$ --- de novo
        independentemente de $n$, de novo em acordo com a lei de Poisson.
  \item Calcule a proporção das permutações que têm pelo menos um
        ponto fixo para $n = 4, 5, 6$ (em frações e com quatro
        casas decimais) e compare com $1 - \eu^{-1} \approx
        0.6321$.
  \item Mostre diretamente --- sem precisar de limites --- que $s_{n+2} - s_n
        = (-1)^{n+1}\bigl(\frac1{(n+1)!} - \frac1{(n+2)!}\bigr)$,
        e deduza que as probabilidades $p_n = s_n$ da
        questão 14 oscilam: $p_0 > p_2 > p_4 > \dots$ e
        $p_1 < p_3 < p_5 < \dots$, os valores pares (resp.\ ímpares)
        decrescendo (resp.\ crescendo) rumo ao limite comum
        $\eu^{-1}$.
  \item (Amigo oculto) $n$ pessoas tiram, cada uma, um nome de um
        chapéu; se alguém tira o próprio nome, o sorteio \emph{inteiro}
        é reiniciado do zero. Usando o fato usual de que um evento de
        probabilidade $p$ exige em média $1/p$ tentativas,
        estime o número médio de sorteios completos necessários e
        conclua que o procedimento custa cerca de $\eu \approx
        2.72$ sorteios em média, essencialmente de modo independente
        de $n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- A aritmética de $D_n$, e uma
síntese.}
\begin{enumerate}
  \item Refine a questão 5: mostre que, para $j \in
        \intint2n$ fixado, os desarranjos de $\intint1n$ com
        $\sigma(1) = j$ somam exatamente $D_{n-1} + D_{n-2}$,
        independentemente de $j$. Deduza que $n - 1$ divide $D_n$
        para todo $n \geq 2$.
  \item Demonstre que $D_n$ é ímpar se, e somente se, $n$ é par.
        (Trabalhe módulo $2$ na recorrência da questão 6.)
  \item Demonstre que $D_n \equiv (-1)^n \pmod n$ para $n \geq 1$ e
        confira a congruência no último algarismo de $D_{10}$.
  \item Mostre, a partir da questão 6, que $\dfrac{D_n}{D_{n-1}} = n +
        \dfrac{(-1)^n}{D_{n-1}}$ para $n \geq 3$, de modo que a razão entre
        números de desarranjos consecutivos é \emph{quase exatamente}
        $n$; explique em uma frase por que isso é coerente com
        $D_n \approx n!/\eu$.
  \item Onde exatamente este problema usou: (i) as regras do produto e
        da soma; (ii) a contagem dupla; (iii) o teorema binomial;
        (iv) a estimativa admitida das séries alternadas? Uma frase
        para cada.
  \item Síntese. A fórmula de $D_n$ tem agora três demonstrações
        (inclusão--exclusão, recorrência mais indução, inversão
        binomial). Num parágrafo curto, compare o que cada demonstração
        \emph{explica}: qual delas calcula mais rápido, qual se
        generaliza a outras contagens de pontos fixos e qual revela
        por que $\eu$ aparece num problema sobre envelopes.
\end{enumerate}
\end{problem}
