\chapter{Funções de Duas Variáveis}\label{ch:b1:multivar}

O ano termina com um primeiro passeio pela dimensão superior: funções
$f(x, y)$ de duas variáveis reais. Tudo se generaliza --- limites,
\omterm{def:b1:continuity:continuous}{continuidade}, \omterm{def:b1:derivative:def}{derivadas}, extremos ---, mas cada noção ganha uma
reviravolta: os limites podem ser abordados por todas as direções ao
mesmo tempo, as \omterm{def:b1:derivative:def}{derivadas} separam-se em parciais, e o \omterm{def:b1:multivar:partial}{gradiente} aponta
o caminho para cima. A teoria completa (diferenciais, $\R^n$ geral,
subvariedades) pertence ao segundo ano; aqui instalamos o vocabulário
e os primeiros teoremas honestos.

\section{O plano $\R^2$; continuidade}

\begin{definition}\label{def:b1:multivar:topology}
Em $\R^2$, use a norma euclidiana $\norm{(x,y)} = \sqrt{x^2 + y^2}$ (o \cref{ch:b1:euclid}). As bolas \omterm{def:b1:topology:open}{abertas},
as \omterm{def:b1:topology:open}{vizinhanças} e os \emph{subconjuntos \omterm{def:b1:topology:open}{abertos}} de $\R^2$ definem-se
exatamente como no \cref{ch:b1:topology}, com bolas no lugar de \omterm{prop:b1:reals:intervals}{intervalos}. Uma função
$f \colon U \to \R$ ($U \subseteq \R^2$ \omterm{def:b1:topology:open}{aberto}) é \emph{\omterm{def:b1:continuity:continuous}{contínua} em} $a \in U$ quando
\[
\forall\varepsilon > 0,\ \exists\delta > 0, \quad
\norm{X - a} \leq \delta \implies \abs{f(X) - f(a)} \leq
\varepsilon,
\]
com a mesma caracterização sequencial que em uma variável. Somas,
produtos, quocientes e composições com funções \omterm{def:b1:continuity:continuous}{contínuas} de uma
variável preservam a \omterm{def:b1:continuity:continuous}{continuidade}; as aplicações \omterm{prop:b1:vspaces:coordinates}{coordenadas} são
\omterm{def:b1:continuity:continuous}{contínuas}, logo os \omterm{def:b1:poly:def}{polinômios} em $(x,y)$ também o são.
\end{definition}

\begin{example}[A majoração polar, o modo limpo de demonstrar um limite]
\label{ex:b1:multivar:polarbound}
Mostre que $f(x, y) = \dfrac{x^2y^2}{x^2 + y^2}$ (com $f(0,0) =
0$) é \omterm{def:b1:continuity:continuous}{contínua} na origem. Em \omterm{prop:b1:vspaces:coordinates}{coordenadas}
polares $x =
\rho\cos\theta$, $y = \rho\sin\theta$:
\[
\abs{f} = \frac{\rho^4\cos^2\theta\sin^2\theta}{\rho^2}
= \rho^2\,(\cos\theta\sin\theta)^2 \leq \frac{\rho^2}{4}
\xrightarrow[\rho \to 0]{} 0 ,
\]
uma cota \emph{independente de $\theta$}: qualquer que seja a direção
de aproximação, os valores são espremidos até $0$. Essa uniformidade
em $\theta$ é todo o ponto --- uma cota como $\abs g =
\abs{\cos\theta\sin\theta}$ (sem nenhum $\rho$
restante) nada demonstra, e de fato esse $g$ é a armadilha radial
descontínua do exemplo seguinte.
\end{example}

\begin{example}[A armadilha radial]\label{ex:b1:multivar:trap}
Seja $f(x, y) = \dfrac{xy}{x^2 + y^2}$ para $(x,y) \neq (0,0)$, $f(0,
0) = 0$. Ao longo de cada eixo, $f = 0 \to 0$; mas ao
longo da diagonal $y = x$, $f(x, x) = \frac12 \not\to 0$. \emph{Nenhum limite na origem}:
aproximar-se por todas as retas, e até encontrar o mesmo limite ao
longo de cada uma, não basta (aqui os limites ao longo das retas
discordam; exemplos piores concordam ao longo de todas as retas e
ainda assim falham ao longo de uma parábola, o \cref{exo:b1:multivar:3}). A \omterm{def:b1:continuity:continuous}{continuidade}
em cada variável separadamente não implica a \omterm{def:b1:continuity:continuous}{continuidade}.
\end{example}

\section{Derivadas parciais}

\begin{definition}\label{def:b1:multivar:partial}
As \emph{derivadas parciais}\index{derivada parcial} de $f$ em
$(a, b)$ são as \omterm{def:b1:derivative:def}{derivadas} de uma variável ao longo dos eixos:
\[
\frac{\partial f}{\partial x}(a,b)
= \lim_{h \to 0} \frac{f(a + h, b) - f(a,b)}{h},
\qquad
\frac{\partial f}{\partial y}(a,b)
= \lim_{k \to 0} \frac{f(a, b + k) - f(a,b)}{k}.
\]
$f$ é de classe $C^1$ em $U$ quando ambas existem e são \omterm{def:b1:continuity:continuous}{contínuas} em
$U$. O \emph{gradiente}\index{gradiente} é $\nabla f(a,b) =
\bigl(\frac{\partial f}{\partial x},\, \frac{\partial f}{\partial
y}\bigr)(a,b)$.
\end{definition}

\begin{theorem}[$C^1$ implica um plano tangente]\label{thm:b1:multivar:c1}
Seja $f$ de classe $C^1$ em $U$ e $(a,b) \in U$. Então, quando $(h, k) \to
(0,0)$:
\[
f(a + h, b + k) = f(a, b)
+ h\,\frac{\partial f}{\partial x}(a,b)
+ k\,\frac{\partial f}{\partial y}(a,b)
+ o\bigl(\norm{(h,k)}\bigr) .
\]
Em particular, $f$ é \omterm{def:b1:continuity:continuous}{contínua}, e o gráfico $z = f(x,y)$ tem em cada ponto o
plano tangente que a fórmula fornece.
\end{theorem}

\begin{example}[Aproximação linear em ação]
\label{ex:b1:multivar:linapprox}
Estime $f(1.02,\ 0.99)$ para $f(x, y) = x^3y^2$. Em $(1, 1)$: $f = 1$, $\frac{\partial f}{\partial x} = 3x^2y^2 = 3$, $\frac{\partial f}{\partial y} = 2x^3y = 2$, logo o
\cref{thm:b1:multivar:c1} dá
\[
f(1.02,\ 0.99) \approx 1 + 3\,(0.02) + 2\,(-0.01) = 1.04 ,
\]
contra o valor verdadeiro $1.02^3 \times 0.99^2 = 1.04006\dots$ --- o erro é de segunda ordem nos
incrementos, como o $o(\norm{(h,k)})$ promete. O plano tangente ao gráfico em
$(1, 1, 1)$ é $z = 1 + 3(x - 1) + 2(y - 1)$, a equação implícita na estimativa.
\end{example}

\begin{proof}
Mova uma coordenada de cada vez:
\[
f(a+h, b+k) - f(a,b)
= \bigl[f(a+h, b+k) - f(a, b+k)\bigr] + \bigl[f(a, b+k) -
f(a,b)\bigr].
\]
Pelo teorema do valor médio de uma variável (o \cref{thm:b1:derivative:mvt}), o primeiro
colchete é $h\,
\frac{\partial f}{\partial x}(a + \theta h,\, b + k)$ para algum $\theta \in \intoo{0}{1}$, e o segundo é $k\,\frac{\partial
f}{\partial y}(a,\, b + \theta' k)$. A \omterm{def:b1:continuity:continuous}{continuidade}
das parciais em $(a,b)$ permite escrever cada uma como (valor em
$(a,b)$) $+$ (erro $\to 0$); o erro total é $h\,\varepsilon_1 +
k\,\varepsilon_2 = o(\norm{(h,k)})$, pois $\abs h, \abs k \leq
\norm{(h,k)}$.
\end{proof}

\begin{theorem}[Regra da cadeia]\label{thm:b1:multivar:chain}
Seja $f$ de classe $C^1$ em $U$ e $t \mapsto (x(t), y(t))$ de classe $C^1$ de um
\omterm{prop:b1:reals:intervals}{intervalo} em $U$. Então $g(t) = f\bigl(x(t), y(t)\bigr)$ é de classe $C^1$, com\index{regra da
cadeia!duas variáveis}
\[
g'(t) = x'(t)\,\frac{\partial f}{\partial x}\bigl(x(t),y(t)\bigr)
+ y'(t)\,\frac{\partial f}{\partial y}\bigl(x(t),y(t)\bigr)
= \bigl\langle \nabla f,\ (x', y')\bigr\rangle .
\]
\end{theorem}

\begin{proof}
Aplique o \cref{thm:b1:multivar:c1} em $(x(t), y(t))$ com $(h, k) =
(x(t+s) - x(t),\, y(t+s) - y(t))$: quando $s \to 0$, a \omterm{def:b1:derivative:def}{derivabilidade}
em uma variável dá $h = s\,x'(t) + o(s)$ e $k = s\,y'(t)
+ o(s)$, logo $\norm{(h, k)} = O(s)$ e
\[
g(t+s) - g(t)
= h\,\frac{\partial f}{\partial x} + k\,\frac{\partial
f}{\partial y} + o\bigl(\norm{(h,k)}\bigr)
= s\,\Bigl(x'\,\frac{\partial f}{\partial x} +
y'\,\frac{\partial f}{\partial y}\Bigr) + o(s) ,
\]
absorvendo o erro final tanto os $o(s)$ de $h$ e de $k$
(multiplicados pelos valores fixos das parciais) quanto o $o(O(s))$ da
estimativa do plano tangente. Divida por $s$ e faça $s \to 0$. A
\omterm{def:b1:continuity:continuous}{continuidade} de $g'$ decorre da de todos os ingredientes.
\end{proof}

\begin{example}[A regra da cadeia, verificada dos dois modos]
\label{ex:b1:multivar:chainrun}
Sejam $f(x, y) = x^2 y$ e $g(t) = f(t, t^2)$. Diretamente: $g(t) =
t^2\cdot t^2 = t^4$, logo $g'(t) = 4t^3$. Pela regra da
cadeia: $\frac{\partial f}{\partial x} = 2xy$ e $\frac{\partial
f}{\partial y} = x^2$, avaliadas ao longo da curva $(t, t^2)$:
\[
g'(t) = 1\cdot 2t\cdot t^2 + 2t\cdot t^2 = 2t^3 + 2t^3 = 4t^3 .
\]
Os dois cálculos concordam, e a separação é significativa: $2t^3$ do
crescimento vem de mover-se \emph{para a direita} através da
inclinação em $x$, e $2t^3$ de mover-se \emph{para cima} através da
inclinação em $y$. Em curvas onde não existe forma \omterm{def:b1:topology:closed}{fechada} para $g$,
só o segundo cálculo sobrevive --- é esse o ponto do teorema.
\end{example}

\begin{remark}[Ler o gradiente]
Ao longo de uma direção unitária $u$, a regra da cadeia aplicada a
$t \mapsto f(a + tu)$ dá a \emph{\omterm{def:b1:derivative:def}{derivada} direcional} $\langle \nabla f(a),
u\rangle$: máxima quando $u$ aponta
ao longo de $\nabla f(a)$ (por Cauchy--Schwarz, o \cref{thm:b1:euclid:cs}). O \omterm{def:b1:multivar:partial}{gradiente} é a
direção de subida mais íngreme, e é \omterm{def:b1:euclid:orthogonal}{ortogonal} às curvas de nível
$\{f = c\}$ (derive $f$ ao longo de uma curva traçada num \omterm{def:b1:logic:sets}{conjunto} de
nível: a regra da cadeia dá $\langle\nabla f,\,
\text{tangente}\rangle = 0$).
\end{remark}

\begin{example}[Curvas de nível e gradientes, numa só função]
\label{ex:b1:multivar:levelcurves}
Tome $f(x, y) = x^2 - y^2$. Os seus \omterm{def:b1:logic:sets}{conjuntos} de nível: $\{f = c\}$ é uma hipérbole \omterm{def:b1:topology:open}{aberta}
para a esquerda e para a direita se $c > 0$, e para cima e para baixo
se $c < 0$, e o par de retas cruzadas $y = \pm x$ para $c = 0$ --- o mapa de
contornos de um colo de montanha, com o \omterm{met:b1:multivar:monge}{ponto de sela} na origem, onde
as duas retas de nível zero se cruzam. \omterm{def:b1:multivar:partial}{Gradiente}: $\nabla
f = (2x, -2y)$. No ponto
$(2, 1)$ (no nível $c = 3$): $\nabla f = (4, -2)$, ao passo que o vetor tangente da
curva de nível, parametrizada perto desse ponto por $\bigl(t,
\sqrt{t^2 - 3}\bigr)$, é $\bigl(1, \frac{t}{\sqrt{t^2 -
3}}\bigr) = (1, 2)$ em
$t = 2$ --- e de fato
\[
\langle (4, -2),\ (1, 2)\rangle = 4 - 4 = 0 :
\]
\omterm{def:b1:multivar:partial}{gradiente} perpendicular ao contorno, apontando para valores mais altos
de $f$ (aqui: afastando-se do eixo $y$). Mais duas leituras: o
\omterm{def:b1:multivar:partial}{gradiente} anula-se exatamente na sela, onde o mapa de contornos se
estrangula; e a \emph{reta} tangente à curva de nível em $(2,1)$ é
$4(x - 2) - 2(y - 1) = 0$, isto é, $2x -
y = 3$ --- a equação ``$\langle \nabla f,\ M - M_0\rangle =
0$'' que generaliza a tangente
à elipse do \cref{exo:b1:curves:11}.
\end{example}

\begin{theorem}[Schwarz]\label{thm:b1:multivar:schwarz}
Se $f$ é de classe $C^2$ (as parciais das parciais existem e são
\omterm{def:b1:continuity:continuous}{contínuas}), então\index{teorema de Schwarz}
\[
\frac{\partial^2 f}{\partial x\,\partial y}
= \frac{\partial^2 f}{\partial y\,\partial x} .
\]
\end{theorem}
\admitted

\section{Extremos locais}

\begin{method}[Estudos de extremos, organizados]
\label{met:b1:multivar:extremamethod}
\begin{enumerate}
  \item \emph{Resolva $\nabla f = 0$ por completo.} Fatore cada parcial sempre
        que possível (produtos de fatores \omterm{def:b1:linmaps:def}{lineares} \omterm{def:b1:arith:divides}{dividem} o sistema
        em casos transparentes, como no \cref{ex:b1:multivar:fourpoints} abaixo); um caso
        esquecido é um \omterm{thm:b1:multivar:critical}{ponto crítico} esquecido.
  \item \emph{Classifique cada ponto} com os dados de Monge $r, s,
        t$ ---
        recalculados \emph{em cada ponto}, nunca uma vez por todas.
  \item \emph{Se $rt - s^2 = 0$}, examine $f$ diretamente ao longo de curvas
        bem escolhidas pelo ponto (primeiro retas, depois parábolas),
        caçando ou dois sinais (nenhum extremo) ou um sinal fixo com
        um argumento que cubra todas as direções.
  \item \emph{Recue para o quadro global}: verifique o comportamento
        no infinito (um mínimo local pode não ser global) e, se o
        domínio não for \omterm{def:b1:topology:open}{aberto}, trate a sua \omterm{def:b1:topology:closure}{fronteira} separadamente
        (o \cref{exo:b1:multivar:12}) --- o teorema dos \omterm{thm:b1:multivar:critical}{pontos críticos} só enxerga pontos
        \omterm{def:b1:topology:closure}{interiores}.
\end{enumerate}
\end{method}

\begin{theorem}[Pontos críticos]\label{thm:b1:multivar:critical}
Se $f$ (de classe $C^1$ no \omterm{def:b1:topology:open}{aberto} $U$) tem um extremo local em $(a,b)
\in U$,
então $\nabla f(a,b) = (0,0)$: o ponto é \emph{crítico}\index{ponto crítico}.
\end{theorem}

\begin{proof}
As funções de uma variável $x \mapsto f(x, b)$ e $y \mapsto f(a,
y)$ têm extremos locais
\omterm{def:b1:topology:closure}{interiores} em $a$ e em $b$, respectivamente: a \cref{prop:b1:derivative:fermat} mata as duas
parciais.
\end{proof}

\begin{method}[Teste de segunda ordem (notação de Monge)]\label{met:b1:multivar:monge}
Num \omterm{thm:b1:multivar:critical}{ponto crítico} de uma função de classe $C^2$, ponha
\[
r = \frac{\partial^2 f}{\partial x^2},
\qquad
s = \frac{\partial^2 f}{\partial x \partial y},
\qquad
t = \frac{\partial^2 f}{\partial y^2}
\qquad (\text{valores no ponto}).
\]
\begin{itemize}
  \item Se $rt - s^2 > 0$: extremo local --- mínimo se $r > 0$, máximo se
        $r < 0$;
  \item se $rt - s^2 < 0$: nenhum extremo (um \emph{ponto de
        sela}\index{ponto de sela});
  \item se $rt - s^2 = 0$: o teste é mudo; examine diretamente.
\end{itemize}
(A justificativa --- uma expansão de Taylor--Young na ordem $2$ e o
estudo do sinal da forma quadrática $r h^2 + 2shk + tk^2$ --- é feita no segundo ano;
o teste é usado aqui como ferramenta de trabalho.)
\end{method}

\begin{example}\label{ex:b1:multivar:extrema}
$f(x,y) = x^3 + y^3 - 3xy$. \omterm{thm:b1:multivar:critical}{Pontos críticos}: $\nabla f = (3x^2 -
3y,\; 3y^2 - 3x) = 0$ dá $y = x^2$ e $x = y^2$, logo
$x = x^4$: $x \in \{0, 1\}$: os pontos $(0,0)$ e $(1,1)$.

\omterm{def:b1:derivative:def}{Derivadas} segundas: $r = 6x$, $s = -3$, $t = 6y$. Em $(0,0)$: $rt -
s^2 = -9 < 0$:
sela. Em $(1,1)$: $rt - s^2 = 36 - 9 > 0$, $r = 6
> 0$: mínimo local, $f(1,1) = -1$. (Não global:
$f(x, 0) = x^3 \to
-\infty$.)
\end{example}

\begin{example}[Um estudo de quatro pontos, por extenso]
\label{ex:b1:multivar:fourpoints}
$f(x, y) = xy\,(3 - x - y) = 3xy - x^2y - xy^2$. \omterm{def:b1:multivar:partial}{Gradiente}:
\[
\frac{\partial f}{\partial x} = y\,(3 - 2x - y),
\qquad
\frac{\partial f}{\partial y} = x\,(3 - x - 2y).
\]
\omterm{thm:b1:multivar:critical}{Pontos críticos}: se $y = 0$, a segunda equação dá $x \in
\{0, 3\}$; se $x = 0$, a
primeira dá $y \in \{0, 3\}$; se $xy
\neq 0$, resolva $2x + y = 3$, $x + 2y = 3$: $x = y = 1$. Quatro
pontos: $(0,0)$, $(3,0)$, $(0,3)$, $(1,1)$. \omterm{def:b1:derivative:def}{Derivadas} segundas:
$r = -2y$, $s = 3 - 2x - 2y$, $t = -2x$.
\begin{itemize}
  \item $(1,1)$: $r = -2$, $s = -1$, $t = -2$: $rt - s^2 = 3 >
        0$, $r < 0$: máximo
        local, $f(1,1) = 1$.
  \item $(0,0)$: $r = t = 0$, $s = 3$: $rt - s^2 = -9 < 0$: sela; do mesmo modo
        $(3, 0)$ ($s = -3$) e $(0, 3)$: três selas.
\end{itemize}
O máximo é apenas local: $f(-T, -T) = T^2(3 + 2T) \to
+\infty$. Verificação de simetria: $f(x, y) = f(y, x)$, e de
fato o \omterm{def:b1:logic:sets}{conjunto} crítico e a classificação são simétricos em $x
\leftrightarrow y$.
Interpretação: entre os retângulos com folga $x, y \geq 0$, $x + y \leq 3$, o produto
$xy(3 - x - y)$ das três ``partes'' de $3$ é máximo quando as partes são iguais
--- uma sombra em duas variáveis do princípio das médias aritmética e
geométrica.
\end{example}

\begin{remark}[Armadilhas comuns]
\label{rem:b1:multivar:pitfalls}
\emph{As \omterm{def:b1:multivar:partial}{derivadas parciais} podem existir num ponto de
descontinuidade}: a armadilha radial $g(x,y) = \frac{xy}{x^2+y^2}$ do \cref{ex:b1:multivar:trap} tem $\frac{\partial g}{\partial
x}(0,0) = \frac{\partial g}{\partial y}(0,0) = 0$ (as duas
restrições aos eixos anulam-se identicamente), e no entanto $g$ não
tem limite na origem --- as parciais sondam apenas duas direções, ao
passo que a \omterm{def:b1:continuity:continuous}{continuidade} precisa de todas; só a hipótese $C^1$
restaura a ordem (o \cref{thm:b1:multivar:c1}).
\emph{Limites ao longo de retas nunca bastam}: a função do \cref{exo:b1:multivar:3} tem
limite $0$ ao longo de toda reta e ainda assim não tem limite ---
tente sempre parábolas (ou cotas polares válidas uniformemente em
$\theta$).
\emph{Crítico é necessário, não suficiente}: as selas abundam (três
dos quatro pontos do \cref{ex:b1:multivar:fourpoints}); e o teorema vale apenas em \omterm{def:b1:logic:sets}{conjuntos}
\emph{\omterm{def:b1:topology:open}{abertos}} --- num disco \omterm{def:b1:topology:closed}{fechado}, os extremos podem ficar na
\omterm{def:b1:topology:closure}{fronteira} com \omterm{def:b1:multivar:partial}{gradiente} não nulo (o \cref{exo:b1:multivar:12}).
\emph{O caso mudo $rt - s^2 = 0$ é genuinamente mudo}: $x^4
+ y^4$ (mínimo) e $x^3 + y^3$
(nenhum dos dois) têm ambos $r = s = t
= 0$ na origem; só um estudo direto de
sinal decide (o \cref{exo:b1:multivar:6}, função $h$).
\emph{O \omterm{def:b1:multivar:partial}{gradiente} é \omterm{def:b1:euclid:orthogonal}{ortogonal} às curvas de nível, e não paralelo a
elas}: para seguir uma linha de contorno, mova-se perpendicularmente a
$\nabla
f$; para subir o mais rápido possível, mova-se ao longo dele ---
misturar os dois inverte a geometria de todo mapa de contornos.
\end{remark}

\begin{remark}[Para onde levam as duas variáveis]
\label{rem:b1:multivar:whereused}
Este capítulo é uma porta. O \omterm{def:b1:multivar:partial}{gradiente} e a regra da cadeia
estendem-se palavra por palavra a $n$ variáveis no volume do segundo
ano de graduação, onde o $o(\norm{(h,k)})$ do \cref{thm:b1:multivar:c1} torna-se a \emph{diferencial}
e o teste de Monge é demonstrado por completo via a fórmula de Taylor
de ordem dois e as formas quadráticas. O caso particular que pode ser
resolvido \emph{neste} ano --- as funções quadráticas, para as quais
a expansão de segunda ordem é exata --- é o assunto do problema de fim
de semana, e acontece de ser o caso que move o ajuste de dados do
mundo inteiro: a regressão por \omterm{pb:b1:multivar:1}{mínimos quadrados}. Os extremos com
restrição (o \cref{exo:b1:multivar:5} foi uma prévia) tornam-se os multiplicadores de
Lagrange no segundo ano; as funções harmônicas (o \cref{exo:b1:multivar:7}) voltam na
análise complexa do volume do terceiro ano de graduação.
\end{remark}

\begin{remark}[Perspectivas dentro do Livro 3: o ano, fechado]
\label{rem:b1:multivar:perspectives}
Este capítulo é o lugar em que as duas metades do volume se
cumprimentam. A metade de análise forneceu as suas ferramentas uma
\omterm{def:b1:derivative:def}{derivada} de cada vez: o teorema do valor médio conduz o \cref{thm:b1:multivar:c1}, as
expansões de Taylor conduzem os testes de extremo, e os $\varepsilon$ do \cref{ch:b1:topology}
voltaram com bolas em vez de \omterm{prop:b1:reals:intervals}{intervalos}. A metade de álgebra forneceu
a geometria: o \omterm{def:b1:multivar:partial}{gradiente} é lido através do \omterm{def:b1:euclid:def}{produto interno} do \cref{ch:b1:euclid}
(Cauchy--Schwarz faz dele a direção mais íngreme), os dados de Monge
$(r, s, t)$ formam uma matriz simétrica do \cref{ch:b1:matrices} com o teste de determinante
do \cref{ch:b1:det}, e o problema de fim de semana executa uma \omterm{thm:b1:euclid:projection}{projeção ortogonal}
sobre vetores de dados. Até as curvas do \cref{ch:b1:curves} voltam como \omterm{def:b1:logic:sets}{conjuntos}
de nível. Um leitor capaz de reconstruir por que cada uma dessas cinco
passagens de bastão funciona terá, na prática, revisado o ano inteiro
--- que é o verdadeiro propósito deste capítulo final.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:multivar:1}
Calcule as \omterm{def:b1:multivar:partial}{derivadas parciais}: $f(x,y) = x^2 y + \eu^{xy}$; $\;g(x,y) = \ln(x^2 + y^2)$ (em $\R^2\setminus\{0\}$); $\;h(x,y) = \arctan\frac yx$ (em
$x > 0$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:multivar:2}
Estude a \omterm{def:b1:continuity:continuous}{continuidade} em $(0,0)$ (com valor $0$ ali) de:
\[
f(x,y) = \frac{x^2 y}{x^2 + y^2},
\qquad
g(x,y) = \frac{xy}{x^2 + y^2},
\qquad
h(x,y) = \frac{x^3 + y^3}{x^2 + y^2}.
\]
\emph{(As \omterm{prop:b1:vspaces:coordinates}{coordenadas} polares $x = \rho\cos\theta$, $y = \rho\sin\theta$ ajudam: majore por uma
função só de $\rho$ quando for possível.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:3}
Seja $f(x,y) = \dfrac{x^2 y}{x^4 + y^2}$ ($f(0,0) = 0$). Demonstre que $f$ tem limite $0$ na origem
\emph{ao longo de toda reta}, mas que $f\bigl(x, x^2\bigr) = \frac12$: $f$ não é \omterm{def:b1:continuity:continuous}{contínua} em
$(0,0)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:multivar:4}
Verifique à mão o teorema de Schwarz em $f(x, y) = x^3 y^2 + \sin(xy)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:5}
Seja $f$ de classe $C^1$ em $\R^2$ e $g(t) = f(\cos t, \sin t)$. Exprima $g'(t)$ pela
regra da cadeia. Deduza que $f$ restrita ao círculo unitário atinge
extremos em pontos onde $\nabla f$ é paralelo ao vetor radial.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:6}
Encontre e classifique os \omterm{thm:b1:multivar:critical}{pontos críticos} de:
\[
f(x, y) = x^2 + xy + y^2 - 3x,
\qquad
g(x, y) = x^2 - y^2 + 4y,
\qquad
h(x, y) = x^4 + y^4 - 2(x - y)^2 .
\]
\emph{(Para $h$, o teste do determinante é mudo na origem: examine
$h(x, x)$ e $h(x, -x)$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:7}
Uma função $f$ é \emph{harmônica} quando $\frac{\partial^2 f}{\partial
x^2} + \frac{\partial^2 f}{\partial y^2} = 0$. Verifique que $x^2 -
y^2$,
$xy$, $\eu^x\cos y$ e $\ln(x^2 + y^2)$ (fora da origem) são harmônicas.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:8}
Encontre o ponto do plano $\{x + 2y - z = 4\} \subseteq \R^3$ mais próximo da origem, de duas
maneiras: por \omterm{thm:b1:euclid:projection}{projeção ortogonal} (o \cref{ch:b1:euclid}), e minimizando a função de
duas variáveis $f(x, y) = x^2 + y^2 + (x + 2y - 4)^2$ obtida eliminando $z$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:9}
(O laplaciano em \omterm{prop:b1:vspaces:coordinates}{coordenadas} polares, primeiro contato) Seja $f$ de
classe $C^2$ em $\R^2 \setminus \{0\}$ e \emph{radial}: $f(x, y) =
\varphi\bigl(\sqrt{x^2+y^2}\bigr)$, com $\varphi$ de classe
$C^2$ em $\intoo{0}{+\infty}$. Demonstre que
\[
\frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial
y^2} = \varphi''(\rho) + \frac{\varphi'(\rho)}{\rho},
\qquad \rho = \sqrt{x^2 + y^2},
\]
e encontre todas as funções harmônicas radiais no plano perfurado.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:10}
Dê a equação do plano tangente ao gráfico $z = xy$ no ponto $(1, 1, 1)$.
Encontre em seguida todos os pontos do gráfico de $f(x,y) = x^3 + y^3 - 3xy$ em que o plano
tangente é \emph{horizontal}, e relacione a resposta com o \cref{ex:b1:multivar:extrema}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:11}
Seja $f$ de classe $C^1$ em $\R^2$.
\begin{enumerate}
  \item Se $\frac{\partial f}{\partial x} = 0$ em toda parte, demonstre que $f(x, y)$ depende
        apenas de $y$.
  \item Encontre todas as soluções de classe $C^1$ da equação $\frac{\partial f}{\partial x} = \frac{\partial
        f}{\partial y}$
        em $\R^2$. \emph{(Ponha $g(u, v) =
        f\bigl(\frac{u+v}2, \frac{u-v}2\bigr)$ e calcule $\frac{\partial g}{\partial v}$ pela regra da
        cadeia.)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:12}
Encontre o máximo e o mínimo globais de $f(x, y) = xy$ no disco \omterm{def:b1:topology:closed}{fechado} $x^2 + y^2 \leq 1$.
\emph{(Admita o teorema dos valores extremos em duas variáveis: uma
função \omterm{def:b1:continuity:continuous}{contínua} no disco \omterm{def:b1:topology:closed}{fechado} atinge os seus limites ---
demonstrado no volume do segundo ano de graduação. Trate o disco
\omterm{def:b1:topology:open}{aberto} por \omterm{thm:b1:multivar:critical}{pontos críticos} e o círculo de \omterm{def:b1:topology:closure}{fronteira} pela
parametrização do \cref{exo:b1:multivar:5}.)}
\end{exercise}

\section{Problema: mínimos quadrados e a reta de regressão}

\begin{problem}\label{pb:b1:multivar:1}
Dados $n$ pontos, qual reta passa ``mais perto'' de todos eles?
Legendre e Gauss responderam: a reta que minimiza a soma dos erros
verticais \emph{ao quadrado} --- porque essa minimização é exatamente
resolúvel pela álgebra linear. Este problema primeiro demonstra
honestamente o teste de Monge do \cref{met:b1:multivar:monge} para funções quadráticas (o
único caso em que a expansão de segunda ordem é exata), e depois
constrói as \omterm{pb:b1:multivar:1}{equações normais}, a \omterm{pb:b1:multivar:1}{reta de regressão} e o coeficiente de
correlação sobre a geometria euclidiana do
\cref{ch:b1:euclid}.\index{mínimos quadrados}\index{reta de regressão}
\index{equações normais}\index{correlação}

\medskip
\noindent\textbf{Parte I --- Funções quadráticas: o teste de Monge,
demonstrado.} Fixe reais $r, s, t$ e seja $q(h, k) = r h^2 + 2s hk +
t k^2$.
\begin{enumerate}
  \item Suponha $r \neq 0$. Estabeleça a forma com quadrado completado
        \[
        q(h, k) = r\Bigl(h + \frac{s}{r}k\Bigr)^{\!2}
        + \frac{rt - s^2}{r}\,k^2 ,
        \]
        e deduza: se $rt - s^2 > 0$, $q$ tem o sinal estrito de $r$ fora da
        origem; se $rt - s^2 < 0$, $q$ assume os dois sinais.
  \item Resolva os casos restantes: $r = 0$, $t \neq 0$ (simétrico); e
        $r = t = 0$ ($q = 2shk$). Conclua: $q$ assume os dois sinais se e
        somente se $rt - s^2 < 0$, e $q$ anula-se apenas na origem se e
        somente se $rt - s^2 > 0$.
  \item Seja agora $f(x, y) = \frac12\bigl(r x^2 + 2s xy + t
        y^2\bigr) + \beta x + \gamma y + \delta$ uma função quadrática com um \omterm{thm:b1:multivar:critical}{ponto crítico}
        $X_0 = (a, b)$. Demonstre a expansão \emph{exata}
        \[
        f(X_0 + (h,k)) = f(X_0) + \tfrac12\,q(h, k)
        \qquad (\text{sem resto}),
        \]
        e deduza a classificação de Monge para funções quadráticas:
        mínimo global estrito se $rt - s^2 > 0$, $r >
        0$; máximo global estrito
        se $rt - s^2 > 0$, $r < 0$; sela se $rt - s^2 < 0$.
  \item Suponha $rt - s^2 > 0$ e $r > 0$ (logo também $t > 0$). Demonstre a
        cota explícita de coercividade
        \[
        q(h, k) \geq c\,(h^2 + k^2),
        \qquad
        c = \frac{rt - s^2}{r + t} > 0 .
        \]
        \emph{(Mostre que a forma quadrática $q - c(h^2 + k^2)$, de coeficientes
        $r - c$, $s$, $t - c$, ainda tem dados de teste de
        discriminante não negativos.)}
  \item Deduza que uma função quadrática com parte quadrática
        definida positiva tende a $+\infty$ quando $\norm{(x,y)} \to
        \infty$, e portanto
        tem um \emph{único minimizador global}: o seu \omterm{thm:b1:multivar:critical}{ponto crítico}.
        Contraste com o \cref{ex:b1:multivar:extrema}, em que um mínimo local de uma função
        não quadrática não era global.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- As \omterm{pb:b1:multivar:1}{equações normais}.} Sejam $C_1,
C_2, b \in \R^n$
(\omterm{def:b1:euclid:def}{produto interno} canônico), e
\[
f(u, v) = \norm{u\,C_1 + v\,C_2 - b}^2 .
\]
\begin{enumerate}[resume]
  \item Desenvolva $f$ e mostre que ela é uma função quadrática de
        $(u,
        v)$ com coeficientes de parte quadrática $r = 2\norm{C_1}^2$, $s = 2\langle C_1, C_2\rangle$,
        $t = 2\norm{C_2}^2$.
  \item Mostre que $rt - s^2 > 0$ se e somente se $(C_1, C_2)$ é \omterm{def:b1:vspaces:free}{livre} \emph{(o caso
        de igualdade de Cauchy--Schwarz, o \cref{thm:b1:euclid:cs})}; suponha isso
        daqui em diante.
  \item Mostre que as equações críticas $\nabla f = 0$ são as \emph{\omterm{pb:b1:multivar:1}{equações
        normais}}
        \[
        \begin{pmatrix}
        \norm{C_1}^2 & \langle C_1, C_2\rangle\\
        \langle C_1, C_2\rangle & \norm{C_2}^2
        \end{pmatrix}
        \begin{pmatrix} u\\ v\end{pmatrix}
        =
        \begin{pmatrix} \langle C_1, b\rangle\\ \langle C_2,
        b\rangle\end{pmatrix}
        \]
        --- a matriz de Gram do \cref{exo:b1:euclid:11} à esquerda --- e que elas dizem
        exatamente: $b - (uC_1 + vC_2)
        \perp C_1, C_2$. Conclua, com a Parte I e o \cref{thm:b1:euclid:projection}: o
        minimizador único dá $p = uC_1 + vC_2 = $, a \omterm{thm:b1:euclid:projection}{projeção ortogonal} de $b$ sobre
        $\operatorname{Vect}(C_1, C_2)$.
  \item Mostre que o valor mínimo é $\norm{b}^2 - \langle p,
        b\rangle = \norm{b - p}^2$, e desenhe a figura
        pitagórica.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- A \omterm{pb:b1:multivar:1}{reta de regressão}.} Pontos de dados
$(x_1, y_1), \dots, (x_n, y_n)$, com nem todos os $x_i$ iguais. Minimize
\[
E(\alpha, \beta) = \sum_{i=1}^{n}\bigl(y_i - \alpha - \beta
x_i\bigr)^2 .
\]
Escreva $\bar x = \frac1n\sum x_i$, $\bar y = \frac1n\sum y_i$, $v_x = \frac1n\sum x_i^2 - \bar x^2$, $v_y = \frac1n\sum y_i^2 -
\bar y^2$, $c_{xy} = \frac1n\sum x_i y_i - \bar x\bar y$.
\begin{enumerate}[resume]
  \item Reconheça a Parte II com $C_1 = (1, \dots, 1)$, $C_2 =
        (x_1, \dots, x_n)$, $b = (y_1, \dots, y_n)$, e escreva as
        \omterm{pb:b1:multivar:1}{equações normais}
        \[
        n\alpha + \Bigl(\sum x_i\Bigr)\beta = \sum y_i,
        \qquad
        \Bigl(\sum x_i\Bigr)\alpha + \Bigl(\sum
        x_i^2\Bigr)\beta = \sum x_i y_i .
        \]
  \item Resolva-as:
        \[
        \beta = \frac{c_{xy}}{v_x},
        \qquad
        \alpha = \bar y - \beta\,\bar x ,
        \]
        e observe que a \emph{\omterm{pb:b1:multivar:1}{reta de regressão}} $y = \alpha
        + \beta x$ passa pelo
        ponto médio $(\bar x, \bar
        y)$.
  \item Verifique que $v_x > 0$ exatamente quando os $x_i$ não são
        todos iguais, e case isto com a questão 7.
  \item Demonstre que o erro mínimo é
        \[
        E_{\min} = n\Bigl(v_y - \frac{c_{xy}^2}{v_x}\Bigr)
        = n\,v_y\,(1 - \rho^2),
        \qquad
        \rho = \frac{c_{xy}}{\sqrt{v_x v_y}}
        \quad (v_y \neq 0).
        \]
        Deduza $\abs\rho \leq 1$, com $\abs\rho = 1$ se e somente se os dados estão
        perfeitamente alinhados.
  \item Exemplo trabalhado: para os dados $(0,0)$, $(1,1)$, $(2,3)$,
        $(3,4)$, calcule $\bar x, \bar y, v_x, c_{xy}$, a \omterm{pb:b1:multivar:1}{reta de regressão}, os quatro
        resíduos e $E_{\min}$.
  \item Demonstre que os resíduos $\varepsilon_i = y_i - \alpha -
        \beta x_i$ da reta ótima satisfazem
        $\sum_i
        \varepsilon_i = 0$ e $\sum_i x_i\varepsilon_i = 0$, e interprete ambos via ortogonalidade.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Variações e aplicações.}
\begin{enumerate}[resume]
  \item (Melhor constante) Mostre que a constante $\alpha$ que
        minimiza $\sum_i (y_i - \alpha)^2$ é a média $\bar
        y$, e que o valor mínimo é
        $n\,v_y$: a variância mede o fracasso dos dados em serem
        constantes.
  \item (Reta pela origem) Mostre que a inclinação que minimiza
        $\sum_i (y_i - \beta x_i)^2$ é $\beta_0 = \frac{\sum
        x_iy_i}{\sum x_i^2}$, e dê uma condição sobre os dados para que
        $\beta_0$ coincida com a inclinação $c_{xy}/v_x$ da questão 11.
  \item (Distância a uma reta, de novo) Para um ponto $M$ e a reta
        $D$ por $P_0$ dirigida pelo vetor unitário $w$, minimize
        $g(\tau) = \norm{P_0 + \tau w - M}^2$ e deduza $d(M, D)^2 = \norm{M - P_0}^2 - \langle M - P_0,
        w\rangle^2$; recupere a fórmula $d = \frac{\abs{a x_0 +
        b y_0 + c}}{\sqrt{a^2 + b^2}}$ para a reta
        $ax + by + c
        = 0$ no plano.
  \item (Análise da variância) Demonstre a decomposição $v_y =
        \beta^2 v_x + \frac{E_{\min}}n$: a
        variância dos $y_i$ separa-se na parte explicada pela reta
        mais a variância residual.
  \item (Ajuste parabólico) Para ajustar $y = a + bx + cx^2$, mostre que as
        \omterm{pb:b1:multivar:1}{equações normais} são o sistema $3 \times 3$ com a matriz de
        momentos
        \[
        \begin{pmatrix}
        n & \sum x_i & \sum x_i^2\\
        \sum x_i & \sum x_i^2 & \sum x_i^3\\
        \sum x_i^2 & \sum x_i^3 & \sum x_i^4
        \end{pmatrix},
        \]
        uma matriz de Gram, invertível assim que três dos $x_i$ são
        distintos \emph{(liberdade de $(1, X, X^2)$ amostrada nos dados, o
        \cref{exo:b1:euclid:11}; compare com as matrizes de momentos do problema de fim
        de semana do \cref{ch:b1:det})}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Robustez, e síntese.}
\begin{enumerate}[resume]
  \item Classifique os \omterm{thm:b1:multivar:critical}{pontos críticos} das três funções quadráticas
        \[
        x^2 + xy + y^2, \qquad x^2 + 3xy + y^2, \qquad
        x^2 + 2xy + y^2 ,
        \]
        pela Parte I, tratando à mão o terceiro caso, degenerado
        (onde o mínimo é atingido?).
  \item (Experimento com ponto atípico) Acrescente o ponto $(10, 0)$
        aos dados da questão 14 e recalcule a inclinação $\beta$. O
        que aconteceu, e por que o erro \emph{ao quadrado} é tão
        sensível a um único ponto distante?
  \item (\omterm{pb:b1:multivar:1}{Mínimos quadrados} ponderados) Dados pesos $w_i > 0$,
        minimize $\sum_i w_i(y_i - \alpha - \beta x_i)^2$: mostre que a solução é dada pelas mesmas
        fórmulas da questão 11 com médias, variância e covariância
        ponderadas (defina-as).
  \item Mostre que, para a reta ótima, $E_{\min} = 0$ força todos os pontos a
        estarem exatamente sobre ela, e conecte com o caso de
        igualdade $\abs\rho = 1$ da questão 13; teste nos dados alinhados
        $(1,1), (2,3), (3,5)$.
  \item Síntese, em quatro frases: por que as funções quadráticas são
        a única classe em que o teste de segunda ordem deste capítulo
        não precisa de nenhum teorema admitido; como as \omterm{pb:b1:multivar:1}{equações
        normais} identificam a minimização analítica com a \omterm{thm:b1:euclid:projection}{projeção
        ortogonal} do \cref{ch:b1:euclid}; o que o coeficiente de correlação mede e
        qual desigualdade o limita; e quais peças dos Capítulos
        18--23 (\omterm{def:b1:vspaces:free}{bases}, matrizes de Gram e de momentos, projeções)
        reapareceram. Nomeie o método e as equações estudados nas
        Partes II--III.
\end{enumerate}
\end{problem}
