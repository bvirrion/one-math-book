\chapter{Funções de duas variáveis}\label{ch:b1:multivar}

O ano termina com uma primeira caminhada para uma dimensão superior: funções
$f(x, y)$ de duas variáveis reais. Tudo generaliza --- \omterm{def:b1:topology:closure}{limites},
continuidade, derivadas, extremos --- mas cada noção ganha uma reviravolta:
\omterm{def:b1:topology:closure}{limites} podem ser alcançados em todas as direções ao mesmo tempo, derivadas
dividido em parciais, e o \omterm{def:b1:multivar:partial}{gradiente} aponta o caminho para cima. O
teoria completa (diferenciais, geral $\R^n $, subvariedades) pertence a
o segundo ano; aqui definimos o vocabulário e o primeiro honesto
teoremas.

\section{O plano $\R^2$; continuidade}

\begin{definition}\label{def:b1:multivar:topology}
Sobre $\R^2$, use a norma euclidiana $\norm{(x,y)} = \sqrt{x^2 + y^2}$
(\cref{ch:b1:euclid}). \omterm{def:b1:topology:open}{Abrir} bolas, bairros, \emph{\omterm{def:b1:topology:open}{abrir}
subconjuntos} de $\R^2$ são definidos exatamente como em
\cref{ch:b1:topology}, com bolas no lugar de \omterm{prop:b1:reals:intervals}{intervalos}. Uma função
$ f \colon U \to \R $ ($ U \subseteq \R^2$ \omterm{def:b1:topology:open}{abrir}) é \emph{contínua
no} $ a \in U $ quando
\[
\forall\varepsilon > 0,\ \exists\delta > 0, \quad
\norm{X - a} \leq \delta \implies \abs{f(X) - f(a)} \leq
\varepsilon,
\]
com a mesma caracterização sequencial de uma variável.
Somas, produtos, quocientes e composições com contínua
funções de uma variável preservam continuidade; a coordenada aplicações são
contínua, portanto, também são \omterm{def:b1:poly:def}{polinômios} em $(x,y)$.
\end{definition}

\begin{example}[O limite polar, a maneira limpa de provar um limite]
\label{ex:b1:multivar:polarbound}
Mostre isso $ f(x, y) = \dfrac{x^2y^2}{x^2 + y^2}$ (com $ f(0,0) =
0$) é contínua na origem. Em polar \omterm{prop:b1:vspaces:coordinates}{coordenadas} $ x =
\rho\cos\theta $, $ y = \rho\sin\theta $:
\[
\abs{f} = \frac{\rho^4\cos^2\theta\sin^2\theta}{\rho^2}
= \rho^2\,(\cos\theta\sin\theta)^2 \leq \frac{\rho^2}{4}
\xrightarrow[\rho \to 0]{} 0 ,
\]
um \omterm{def:b1:topology:closure}{limite} \emph{independente de $\theta $}: qualquer que seja a direção
abordagem, os valores são comprimidos para $0$. Essa uniformidade em
$\theta $ é o ponto principal --- um \omterm{def:b1:topology:closure}{limite} como $\abs g =
\abs{\cos\theta\sin\theta}$ (não $\rho $ esquerda) não prova nada,
e de fato isso $ g $ é a armadilha radial descontínua do
próximo exemplo.
\end{example}

\begin{example}[A armadilha radial]\label{ex:b1:multivar:trap}
Deixar $ f(x, y) = \dfrac{xy}{x^2 + y^2}$ para $(x,y) \neq (0,0)$, $ f(0,
0) = 0$. Ao longo de cada eixo, $ f = 0 \to 0$; mas ao longo da diagonal $ y
= x $, $ f(x, x) = \frac12 \not\to 0$. \emph{Sem \omterm{def:b1:topology:closure}{limite} na origem}:
aproximando-se ao longo de cada linha, e até mesmo encontrando o mesmo \omterm{def:b1:topology:closure}{limite} ao longo
cada um, não é suficiente (aqui os \omterm{def:b1:topology:closure}{limites} da linha discordam; exemplos piores
concordam em todas as linhas, mas falham em uma parábola,
\cref{exo:b1:multivar:3}). Continuidade em cada variável separadamente
não implica continuidade.
\end{example}

\section{Derivadas parciais}

\begin{definition}\label{def:b1:multivar:partial}
O \emph{derivadas parciais}\index{derivadas parciais} de $ f $ no
$(a, b)$ são a única variável derivadas ao longo dos eixos:
\[
\frac{\partial f}{\partial x}(a,b)
= \lim_{h \to 0} \frac{f(a + h, b) - f(a,b)}{h},
\qquad
\frac{\partial f}{\partial y}(a,b)
= \lim_{k \to 0} \frac{f(a, b + k) - f(a,b)}{k}.
\]
$ f $ é de classe $ C^1$ sobre $ U $ quando ambos existem e são contínua sobre
$ U $. O \emph{gradiente}\index{gradiente} é $\nabla f(a,b) =
\bigl(\frac{\partial f}{\partial x},\, \frac{\partial f}{\partial
y}\bigr)(a,b)$.
\end{definition}

\begin{theorem}[$ C^1$ implica um plano tangente]\label{thm:b1:multivar:c1}
Deixar $ f $ ser $ C^1$ sobre $ U $ e $(a,b) \in U $. Então, como $(h, k) \to
(0,0)$:
\[
f(a + h, b + k) = f(a, b)
+ h\,\frac{\partial f}{\partial x}(a,b)
+ k\,\frac{\partial f}{\partial y}(a,b)
+ o\bigl(\norm{(h,k)}\bigr) .
\]
Em particular $ f $ é contínua, e o gráfico $ z = f(x,y)$ tem em
cada ponto do plano tangente lê a fórmula.
\end{theorem}

\begin{example}[Aproximação linear em ação]
\label{ex:b1:multivar:linapprox}
Estimativa $ f(1.02,\ 0.99)$ para $ f(x, y) = x^3y^2$. No $(1, 1)$:
$ f = 1$, $\frac{\partial f}{\partial x} = 3x^2y^2 = 3$,
$\frac{\partial f}{\partial y} = 2x^3y = 2$, então
\cref{thm:b1:multivar:c1} dá
\[
f(1.02,\ 0.99) \approx 1 + 3\,(0.02) + 2\,(-0.01) = 1.04 ,
\]
contra o verdadeiro valor $1.02^3 \times 0.99^2 = 1.04006\dots $ ---
o erro é de segunda ordem nos incrementos, pois o
$ o(\norm{(h,k)})$ promessas. O plano tangente ao gráfico em
$(1, 1, 1)$ é $ z = 1 + 3(x - 1) + 2(y - 1)$, a equação
implícita na estimativa.
\end{example}

\begin{proof}
Mova uma coordenada de cada vez:
\[
f(a+h, b+k) - f(a,b)
= \bigl[f(a+h, b+k) - f(a, b+k)\bigr] + \bigl[f(a, b+k) -
f(a,b)\bigr].
\]
Pelo teorema do valor médio de uma variável
(\cref{thm:b1:derivative:mvt}), o primeiro colchete é $ h\,
\frac{\partial f}{\partial x}(a + \theta h,\, b + k)$ para alguns
$\theta \in \intoo{0}{1}$, e o segundo é $ k\,\frac{\partial
f}{\partial y}(a,\, b + \theta' k)$. Continuidade das parciais em
$(a,b)$ vamos escrever cada um como (valor em $(a,b)$) $+$
(erro $\to 0$); o erro total é $ h\,\varepsilon_1 +
k\,\varepsilon_2 = o(\norm{(h,k)})$ desde $\abs h, \abs k \leq
\norm{(h,k)}$.
\end{proof}

\begin{theorem}[Regra da cadeia]\label{thm:b1:multivar:chain}
Deixar $ f $ ser $ C^1$ sobre $ U $ e $ t \mapsto (x(t), y(t))$ ser $ C^1$ de
um \omterm{prop:b1:reals:intervals}{intervalo} em $ U $. Então $ g(t) = f\bigl(x(t), y(t)\bigr)$ é
$ C^1$, com\index{regra da cadeia!duas variáveis}
\[
g'(t) = x'(t)\,\frac{\partial f}{\partial x}\bigl(x(t),y(t)\bigr)
+ y'(t)\,\frac{\partial f}{\partial y}\bigl(x(t),y(t)\bigr)
= \bigl\langle \nabla f,\ (x', y')\bigr\rangle .
\]
\end{theorem}

\begin{proof}
Aplicar \cref{thm:b1:multivar:c1} no $(x(t), y(t))$ com $(h, k) =
(x(t+s) - x(t),\, y(t+s) - y(t))$: como $ s \to 0$, uma variável
diferenciabilidade dá $ h = s\,x'(t) + o(s)$ e $ k = s\,y'(t)
+ o(s)$, então $\norm{(h, k)} = O(s)$ e
\[
g(t+s) - g(t)
= h\,\frac{\partial f}{\partial x} + k\,\frac{\partial
f}{\partial y} + o\bigl(\norm{(h,k)}\bigr)
= s\,\Bigl(x'\,\frac{\partial f}{\partial x} +
y'\,\frac{\partial f}{\partial y}\Bigr) + o(s) ,
\]
o erro final absorvendo tanto o $ o(s)$de $ h $ e $ k $
(multiplicado pelos valores fixos das parciais) e o
$ o(O(s))$ da estimativa do plano tangente. Dividir por $ s $ e deixe
$ s \to 0$. Continuidade de $ g'$ segue de tudo
ingredientes.
\end{proof}

\begin{example}[A regra da cadeia, verificada em ambos os sentidos]
\label{ex:b1:multivar:chainrun}
Deixar $ f(x, y) = x^2 y $ e $ g(t) = f(t, t^2)$. Diretamente: $ g(t) =
t^2\cdot t^2 = t^4$, então $ g'(t) = 4t^3$. Pela regra da cadeia:
$\frac{\partial f}{\partial x} = 2xy $ e $\frac{\partial
f}{\partial y} = x^2$, avaliado ao longo da curva $(t, t^2)$:
\[
g'(t) = 1\cdot 2t\cdot t^2 + 2t\cdot t^2 = 2t^3 + 2t^3 = 4t^3 .
\]
Os dois cálculos concordam e a \omterm{thm:b1:logic:partition}{divisão} é significativa: $2t^3$
do crescimento vem da movimentação \emph{para a direita} através do
$ x $-declive, $2t^3$ de se mover \emph{para cima} através do
$ y $-declive. Em curvas onde não há forma fechada para $ g $ existe, apenas
o segundo cálculo sobrevive – esse é o ponto do teorema.
\end{example}

\begin{remark}[Lendo o gradiente]
Ao longo de uma direção de unidade $ u $, a regra da cadeia em $ t \mapsto f(a + tu)$
dá o \emph{direcional derivada} $\langle \nabla f(a),
u\rangle $: máximo quando $ u $ aponta ao longo $\nabla f(a)$ (por
Cauchy-Schwarz, \cref{thm:b1:euclid:cs}). O \omterm{def:b1:multivar:partial}{gradiente} é o
direção de subida mais íngreme, e é ortogonal para o nível
curvas $\{f = c\}$ (diferenciar $ f $ ao longo de uma curva desenhada em
nível conjunto: a regra da cadeia dá $\langle\nabla f,\,
\text{tangent}\rangle = 0$).
\end{remark}

\begin{example}[Curvas de nível e gradientes, em uma função]
\label{ex:b1:multivar:levelcurves}
Pegar $ f(x, y) = x^2 - y^2$. Seu nível conjuntos: $\{f = c\}$ é um
hipérbole abrindo da esquerda para a direita para $ c > 0$, de cima para baixo para $ c < 0$,
e o par de linhas cruzadas $ y = \pm x $ para $ c = 0$ --- o
contorno \omterm{def:b1:logic:map}{aplicação} de uma passagem de montanha, com ponto de sela no
origem onde as duas linhas de nível zero se cruzam. \omterm{def:b1:multivar:partial}{gradiente}: $\nabla
f = (2x, -2y)$. No ponto $(2, 1)$ (no nível $ c = 3$):
$\nabla f = (4, -2)$, enquanto o vetor tangente do nível
curva, parametrizada perto desse ponto por $\bigl(t,
\sqrt{t^2 - 3}\bigr)$, é $\bigl(1, \frac{t}{\sqrt{t^2 -
3}}\bigr) = (1, 2)$ no $ t = 2$ --- e de fato
\[
\langle (4, -2),\ (1, 2)\rangle = 4 - 4 = 0 :
\]
\omterm{def:b1:multivar:partial}{gradiente} perpendicular ao contorno, apontando para cima
valores de $ f $ (aqui: longe do $ y $-eixo). Mais dois
leituras: o \omterm{def:b1:multivar:partial}{gradiente} desaparece exatamente na sela, onde
o contorno \omterm{def:b1:logic:map}{aplicação} beliscões; e a tangente \emph{linha} para o
curva de nível em $(2,1)$ é $4(x - 2) - 2(y - 1) = 0$, ou seja\ $2x -
y = 3$ --- a equação ``$\langle \nabla f,\ M - M_0\rangle =
0$'' que generaliza a tangente da elipse de
\cref{exo:b1:curves:11}.
\end{example}

\begin{theorem}[Preto]\label{thm:b1:multivar:schwarz}
Se $ f $ é de classe $ C^2$ (parciais de parciais existem e são
contínua), então\index{Teorema de Schwarz}
\[
\frac{\partial^2 f}{\partial x\,\partial y}
= \frac{\partial^2 f}{\partial y\,\partial x} .
\]
\end{theorem}
\admitted

\section{Extremos locais}

\begin{method}[Estudos extremos, organizados]
\label{met:b1:multivar:extremamethod}
\begin{enumerate}
  \item \emph{Resolver $\nabla f = 0$ completamente.} Fatore cada
        parcial sempre que possível (produtos de \omterm{def:b1:linmaps:def}{linear} fatores
        dividir o sistema em casos transparentes, como em
        \cref{ex:b1:multivar:fourpoints} abaixo); um esquecido
        caso é um esquecido ponto \omterm{thm:b1:multivar:critical}{crítico}.
  \item \emph{Classifique cada ponto} com os dados de Monge $ r, s,
        t $ --- recalculado \emph{em cada ponto}, nunca uma vez por
        tudo.
  \item \emph{Se $ rt - s^2 = 0$}, examinar $ f $ diretamente ao longo
        curvas bem escolhidas através do ponto (linhas primeiro, depois
        parábolas), caçando dois sinais (sem extremo)
        ou para uma placa trancada com um argumento cobrindo todos
        direções.
  \item \emph{Dê um passo para trás para ver a \omterm{def:b1:linmaps:kerim}{imagem} global}: verifique o
        comportamento no infinito (um mínimo local pode não ser global
        um), e se o domínio não for \omterm{def:b1:topology:open}{abrir}, trate seu \omterm{def:b1:topology:closure}{limite}
        separadamente (\cref{exo:b1:multivar:12}) --- o ponto \omterm{thm:b1:multivar:critical}{crítico} teorema vê \omterm{def:b1:topology:closure}{interior} apenas pontos.
\end{enumerate}
\end{method}

\begin{theorem}[Pontos críticos]\label{thm:b1:multivar:critical}
Se $ f $ ($ C^1$ no aberto $ U $) tem um extremo local em $(a,b)
\in U $, então $\nabla f(a,b) = (0,0)$: o ponto é
\emph{crítico}\index{crítico}.
\end{theorem}

\begin{proof}
As funções de uma variável $ x \mapsto f(x, b)$ e $ y \mapsto f(a,
y)$ ter \omterm{def:b1:topology:closure}{interior} extremos locais em $ a $, resp.\ $ b $:
\cref{prop:b1:derivative:fermat} mata ambos os parciais.
\end{proof}

\begin{method}[Teste de segunda ordem (notação Monge)]\label{met:b1:multivar:monge}
Em um ponto \omterm{thm:b1:multivar:critical}{crítico} de um $ C^2$ função, conjunto
\[
r = \frac{\partial^2 f}{\partial x^2},
\qquad
s = \frac{\partial^2 f}{\partial x \partial y},
\qquad
t = \frac{\partial^2 f}{\partial y^2}
\qquad (\text{values at the point}).
\]
\begin{itemize}
  \item Se $ rt - s^2 > 0$: extremo local --- mínimo para $ r > 0$,
        máximo para $ r < 0$;
  \item se $ rt - s^2 < 0$: sem extremo (um \emph{sela
        ponto}\index{sela
        ponto});
  \item se $ rt - s^2 = 0$: o teste é silencioso; examinar diretamente.
\end{itemize}
(Justificativa --- uma expansão Taylor--Young no pedido $2$ e o
estudo de sinal da forma quadrática $ r h^2 + 2shk + tk^2$ --- é
realizado no segundo ano; o teste é usado aqui como um trabalho
ferramenta.)
\end{method}

\begin{example}\label{ex:b1:multivar:extrema}
$ f(x,y) = x^3 + y^3 - 3xy $. Pontos \omterm{thm:b1:multivar:critical}{críticos}: $\nabla f = (3x^2 -
3y,\; 3y^2 - 3x) = 0$ dá $ y = x^2$ e $ x = y^2$, então $ x = x^4$:
$ x \in \{0, 1\}$: pontos $(0,0)$ e $(1,1)$.

Segundo derivadas: $ r = 6x $, $ s = -3$, $ t = 6y $. No $(0,0)$: $ rt -
s^2 = -9 < 0$: sela. No $(1,1)$: $ rt - s^2 = 36 - 9 > 0$, $ r = 6
> 0$: mínimo local, $ f(1,1) = -1$. (Não global: $ f(x, 0) = x^3 \to
-\infty $.)
\end{example}

\begin{example}[Um estudo de quatro pontos, na íntegra]
\label{ex:b1:multivar:fourpoints}
$ f(x, y) = xy\,(3 - x - y) = 3xy - x^2y - xy^2$. \omterm{def:b1:multivar:partial}{gradiente}:
\[
\frac{\partial f}{\partial x} = y\,(3 - 2x - y),
\qquad
\frac{\partial f}{\partial y} = x\,(3 - x - 2y).
\]
Pontos \omterm{thm:b1:multivar:critical}{críticos}: se $ y = 0$, a segunda equação dá $ x \in
\{0, 3\}$; se $ x = 0$, o primeiro dá $ y \in \{0, 3\}$; se $ xy
\neq 0$, resolver $2x + y = 3$, $ x + 2y = 3$: $ x = y = 1$. Quatro
pontos: $(0,0)$, $(3,0)$, $(0,3)$, $(1,1)$. Segundo derivadas:
$ r = -2y $, $ s = 3 - 2x - 2y $, $ t = -2x $.
\begin{itemize}
  \item $(1,1)$: $ r = -2$, $ s = -1$, $ t = -2$: $ rt - s^2 = 3 >
        0$, $ r < 0$: máximo local, $ f(1,1) = 1$.
  \item $(0,0)$: $ r = t = 0$, $ s = 3$: $ rt - s^2 = -9 < 0$:
        sela; da mesma maneira $(3, 0)$ ($ s = -3$) e $(0, 3)$: três
        selas.
\end{itemize}
O máximo é apenas local: $ f(-T, -T) = T^2(3 + 2T) \to
+\infty $. Verificação de simetria: $ f(x, y) = f(y, x)$, e de fato o
\omterm{thm:b1:multivar:critical}{crítico} conjunto e a classificação é simétrica em $ x
\leftrightarrow y $. Interpretação: entre retângulos com folga
$ x, y \geq 0$, $ x + y \leq 3$, o produto $ xy(3 - x - y)$ de
as três “partes” de $3$ é maior quando as partes são iguais
--- uma sombra de duas variáveis do princípio AM--GM.
\end{example}

\begin{remark}[Armadilhas comuns]
\label{rem:b1:multivar:pitfalls}
\emph{\omterm{def:b1:multivar:partial}{Derivadas parciais} pode existir em um ponto de
descontinuidade}: a armadilha radial $ g(x,y) = \frac{xy}{x^2+y^2}$ de
\cref{ex:b1:multivar:trap} tem $\frac{\partial g}{\partial
x}(0,0) = \frac{\partial g}{\partial y}(0,0) = 0$ (ambos os eixos
restrições desaparecem de forma idêntica), ainda $ g $ não tem \omterm{def:b1:topology:closure}{limite} no
origem --- parciais sondam apenas duas direções, continuidade necessidades
todos eles; apenas o $ C^1$ hipótese restaura a ordem
(\cref{thm:b1:multivar:c1}).
\emph{\omterm{def:b1:topology:closure}{Limites} de linha nunca são suficientes}: \cref{exo:b1:multivar:3}de
função tem \omterm{def:b1:topology:closure}{limite} $0$ ao longo de todas as linhas e ainda sem \omterm{def:b1:topology:closure}{limite} ---
sempre tente parábolas (ou \omterm{def:b1:topology:closure}{limites} polares válidos uniformemente em
$\theta $).
\emph{\omterm{thm:b1:multivar:critical}{Crítico} é necessário, não suficiente}: selas abundam
(três dos quatro pontos em \cref{ex:b1:multivar:fourpoints}); e
o teorema se mantém \emph{\omterm{def:b1:topology:open}{abrir}} conjuntos apenas --- em um \omterm{def:b1:topology:closed}{fechado} disco,
extremos podem ficar no \omterm{def:b1:topology:closure}{limite} com diferente de zero \omterm{def:b1:multivar:partial}{gradiente}
(\cref{exo:b1:multivar:12}).
\emph{O caso silencioso $ rt - s^2 = 0$ é genuinamente silencioso}: $ x^4
+ y^4$ (mínimo) e $ x^3 + y^3$ (nenhum) ambos têm $ r = s = t
= 0$ na origem; apenas um estudo de sinal \omterm{def:b1:vspaces:sum}{direto} decide
(\cref{exo:b1:multivar:6}, função $ h $).
\emph{O \omterm{def:b1:multivar:partial}{gradiente} é ortogonal para curvas de nível, não ao longo
eles}: para seguir uma linha de contorno, mova perpendicularmente $\nabla
f $; para subir mais rápido, mova-se ao longo dela --- misturando as duas inversões
a geometria de cada contorno \omterm{def:b1:logic:map}{aplicação}.
\end{remark}

\begin{remark}[Onde duas variáveis levam]
\label{rem:b1:multivar:whereused}
Este capítulo é uma porta. O \omterm{def:b1:multivar:partial}{gradiente} e a regra da cadeia se estende
literalmente para $ n $ variáveis no volume Ano~2, onde o
$ o(\norm{(h,k)})$ de \cref{thm:b1:multivar:c1} torna-se o
\emph{diferencial} e o teste de Monge é comprovado integralmente através do
fórmula de Taylor de ordem dois e formas quadráticas. O caso especial
isso pode ser resolvido \emph{esse} ano --- funções quadráticas,
para o qual a expansão de segunda ordem é exata --- é o assunto
do problema do fim de semana, e acontece que é o caso que ocorre
ajuste de dados do mundo: \omterm{pb:b1:multivar:1}{mínimos quadrados} regressão. Restrito
extremos (\cref{exo:b1:multivar:5} foi uma prévia) tornou-se Lagrange
multiplicadores no Ano~2; funções harmônicas
(\cref{exo:b1:multivar:7}) retorna na análise complexa do
Volume do ano ~ 3.
\end{remark}

metade da análise forneceu suas ferramentas um derivada de cada vez: o
teorema do valor médio impulsiona \cref{thm:b1:multivar:c1},Taylor
expansões conduzem os testes extremos, e o $\varepsilon $ de
\cref{ch:b1:topology} voltou com bolas em vez de
\omterm{prop:b1:reals:intervals}{intervalos}. A metade da álgebra forneceu a geometria: o \omterm{def:b1:multivar:partial}{gradiente}
é lido através do \omterm{def:b1:euclid:def}{produto interno} de \cref{ch:b1:euclid}
(Cauchy-Schwarz torna a direção mais íngreme), o Monge
dados $(r, s, t)$ é uma matriz simétrica de
\cref{ch:b1:matrices} com o teste determinante de
\cref{ch:b1:det}, e o problema do fim de semana é executado \omterm{thm:b1:euclid:projection}{ortogonal
projeção} em vetores de dados. Mesmo as curvas de
\cref{ch:b1:curves} retornar como nível conjuntos. Um leitor que pode
reconstruir por que cada um desses cinco trabalhos de transferência tem, em
efeito, revisado o ano inteiro --- que é o verdadeiro propósito
deste capítulo final.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star $]\label{exo:b1:multivar:1}
Calcule o \omterm{def:b1:multivar:partial}{derivadas parciais}:
$ f(x,y) = x^2 y + \eu^{xy}$; $\;g(x,y) = \ln(x^2 + y^2)$ (sobre
$\R^2\setminus\{0\}$); $\;h(x,y) = \arctan\frac yx $ (sobre $ x > 0$).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:b1:multivar:2}
Estude o continuidade no $(0,0)$ (com valor $0$ lá) de:
\[
f(x,y) = \frac{x^2 y}{x^2 + y^2},
\qquad
g(x,y) = \frac{xy}{x^2 + y^2},
\qquad
h(x,y) = \frac{x^3 + y^3}{x^2 + y^2}.
\]
\emph{(Polar \omterm{prop:b1:vspaces:coordinates}{coordenadas} $ x = \rho\cos\theta $, $ y = \rho\sin\theta $
ajuda: vinculado por uma função de $\rho $ sozinho quando possível.)}
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:b1:multivar:3}
Deixar $ f(x,y) = \dfrac{x^2 y}{x^4 + y^2}$ ($ f(0,0) = 0$). Prove isso
$ f $ tem \omterm{def:b1:topology:closure}{limite} $0$ na origem \emph{ao longo de cada linha reta},
mas isso $ f\bigl(x, x^2\bigr) = \frac12$: $ f $ não é contínua no
$(0,0)$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:b1:multivar:4}
Verifique o teorema de Schwarz manualmente em $ f(x, y) = x^3 y^2 + \sin(xy)$.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:b1:multivar:5}
Deixar $ f $ ser $ C^1$ sobre $\R^2$ e $ g(t) = f(\cos t, \sin t)$. Expressar
$ g'(t)$ através da regra da cadeia. Deduza isso $ f $ restrito à unidade
círculo atinge extremos em pontos onde $\nabla f $ é paralelo a
o vetor raio.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:b1:multivar:6}
Encontre e classifique o pontos \omterm{thm:b1:multivar:critical}{críticos} de:
\[
f(x, y) = x^2 + xy + y^2 - 3x,
\qquad
g(x, y) = x^2 - y^2 + 4y,
\qquad
h(x, y) = x^4 + y^4 - 2(x - y)^2 .
\]
\emph{(Para $ h $, o teste do determinante é silencioso na origem:
examinar $ h(x, x)$ e $ h(x, -x)$.)}
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:b1:multivar:7}
Uma função $ f $ é \emph{harmônico} quando $\frac{\partial^2 f}{\partial
x^2} + \frac{\partial^2 f}{\partial y^2} = 0$. Verifique isso $ x^2 -
y^2$, $ xy $, $\eu^x\cos y $ e $\ln(x^2 + y^2)$ (fora da origem) são
harmônico.
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:b1:multivar:8}
Encontre o ponto do plano $\{x + 2y - z = 4\} \subseteq \R^3$
mais próximo da origem, de duas maneiras: por projeção ortogonal
(\cref{ch:b1:euclid}), e minimizando a função de duas variáveis
$ f(x, y) = x^2 + y^2 + (x + 2y - 4)^2$ obtido eliminando $ z $.
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:b1:multivar:9}
(Laplaciano em polar \omterm{prop:b1:vspaces:coordinates}{coordenadas}, primeiro contato) Seja $ f $ ser $ C^2$ sobre
$\R^2 \setminus \{0\}$ e \emph{radial}: $ f(x, y) =
\varphi\bigl(\sqrt{x^2+y^2}\bigr)$ com $\varphi $ de classe $ C^2$ sobre
$\intoo{0}{+\infty}$. Prove isso
\[
\frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial
y^2} = \varphi''(\rho) + \frac{\varphi'(\rho)}{\rho},
\qquad \rho = \sqrt{x^2 + y^2},
\]
e encontre todas as funções harmônicas radiais no plano perfurado.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:b1:multivar:10}
Dê a equação do plano tangente ao gráfico $ z = xy $ em
o ponto $(1, 1, 1)$. Então encontre todos os pontos do gráfico de
$ f(x,y) = x^3 + y^3 - 3xy $ onde está o plano tangente
\emph{horizontal}, e relacione a resposta com
\cref{ex:b1:multivar:extrema}.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:b1:multivar:11}
Deixar $ f $ ser $ C^1$ sobre $\R^2$.
\begin{enumerate}
  \item Se $\frac{\partial f}{\partial x} = 0$ em todos os lugares, prove
        isso $ f(x, y)$ depende apenas de $ y $.
  \item Encontre tudo $ C^1$ soluções da equação
        $\frac{\partial f}{\partial x} = \frac{\partial
        f}{\partial y}$ sobre $\R^2$. \emph{(\omterm{def:b1:logic:sets}{Definir} $ g(u, v) =
        f\bigl(\frac{u+v}2, \frac{u-v}2\bigr)$ e calcular
        $\frac{\partial g}{\partial v}$ pela regra da cadeia.)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:b1:multivar:12}
Encontre o máximo e o mínimo globais de $ f(x, y) = xy $ no
\omterm{def:b1:topology:closed}{fechado} disco $ x^2 + y^2 \leq 1$. \emph{(Admita as duas variáveis
teorema do valor extremo: a contínua função no \omterm{def:b1:topology:closed}{fechado} disco
atinge seus \omterm{def:b1:topology:closure}{limites} --- comprovado no volume do Ano ~ 2. Trate o
\omterm{def:b1:topology:open}{abrir} disco por pontos \omterm{thm:b1:multivar:critical}{críticos} e o \omterm{def:b1:topology:closure}{limite} círculo pelo
parametrização de \cref{exo:b1:multivar:5}.)}
\end{exercise}

\section{Problema: mínimos quadrados e a linha de regressão}

\begin{problem}\label{pb:b1:multivar:1}
Dado $ n $ pontos de dados, cuja linha reta passa “mais próxima” de
todos eles? Legendre e Gauss responderam: a linha que minimiza o
soma do \emph{ao quadrado} erros verticais --- porque isso
a minimização é exatamente solucionável pela álgebra \omterm{def:b1:linmaps:def}{linear}. Este problema
prova pela primeira vez o teste de Monge de \cref{met:b1:multivar:monge}
honestamente para funções quadráticas (o único caso em que o
expansão de segunda ordem é exata), então constrói o normais
equações, o \omterm{pb:b1:multivar:1}{linha de regressão}, e o coeficiente de correlação
no topo da geometria euclidiana de
\cref{ch:b1:euclid}.\index{mínimos quadrados}\index{linha de regressão}
\index{equações normais}\index{correlação}

\medskip
\noindent\textbf{Parte I --- Funções quadráticas: o teste de Monge,
comprovado.} Corrigir reais $ r, s, t $ e deixe $ q(h, k) = r h^2 + 2s hk +
t k^2$.
\begin{enumerate}
  \item Suponha $ r \neq 0$. Estabeleça o formulário quadrado preenchido
        \[
        q(h, k) = r\Bigl(h + \frac{s}{r}k\Bigr)^{\!2}
        + \frac{rt - s^2}{r}\,k^2 ,
        \]
        e deduza: se $ rt - s^2 > 0$, $ q $ tem o sinal estrito
        de $ r $ fora da origem; se $ rt - s^2 < 0$, $ q $ leva os dois
        sinais.
  \item Resolva os casos restantes: $ r = 0$, $ t \neq 0$
        (simétrico); e $ r = t = 0$ ($ q = 2shk $). Concluir:
        $ q $ assume ambos os sinais se $ rt - s^2 < 0$, e $ q $
        desaparece apenas na origem se $ rt - s^2 > 0$.
  \item Seja agora $ f(x, y) = \frac12\bigl(r x^2 + 2s xy + t
        y^2\bigr) + \beta x + \gamma y + \delta $ ser um quadrático
        funcionar com um ponto \omterm{thm:b1:multivar:critical}{crítico} $ X_0 = (a, b)$. Prove o
        \emph{exato} expansão
        \[
        f(X_0 + (h,k)) = f(X_0) + \tfrac12\,q(h, k)
        \qquad (\text{no remainder}),
        \]
        e deduzir a classificação de Monge para quadrática
        funções: mínimo global estrito se $ rt - s^2 > 0$, $ r >
        0$; máximo global estrito se $ rt - s^2 > 0$, $ r < 0$;
        sela se $ rt - s^2 < 0$.
  \item Suponha $ rt - s^2 > 0$ e $ r > 0$ (assim também $ t > 0$).
        Prove o \omterm{def:b1:topology:closure}{limite} de coercividade explícito
        \[
        q(h, k) \geq c\,(h^2 + k^2),
        \qquad
        c = \frac{rt - s^2}{r + t} > 0 .
        \]
        \emph{(Mostre que a forma quadrática $ q - c(h^2 + k^2)$,
        com coeficientes $ r - c $, $ s $, $ t - c $, ainda tem
        dados de teste discriminante não negativo.)}
  \item Deduza que uma função quadrática com definido positivo
        parte quadrática tende a $+\infty $ como $\norm{(x,y)} \to
        \infty $, e portanto tem um \emph{único global}
        minimizador: seu ponto \omterm{thm:b1:multivar:critical}{crítico}. Compare com
        \cref{ex:b1:multivar:extrema}, onde um mínimo local de
        uma função não quadrática não era global.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O \omterm{pb:b1:multivar:1}{equações normais}.} Deixar $ C_1,
C_2, b \in \R^n $ (canônico \omterm{def:b1:euclid:def}{produto interno}), e
\[
f(u, v) = \norm{u\,C_1 + v\,C_2 - b}^2 .
\]
\begin{enumerate}[resume]
  \item Expandir $ f $ e mostre que é uma função quadrática de $(u,
        v)$ com coeficientes de parte quadrática $ r = 2\norm{C_1}^2$,
        $ s = 2\langle C_1, C_2\rangle $, $ t = 2\norm{C_2}^2$.
  \item Mostre isso $ rt - s^2 > 0$ se e somente se $(C_1, C_2)$ é
        \omterm{def:b1:vspaces:free}{livre} \emph{(o caso da igualdade de Cauchy-Schwarz,
        \cref{thm:b1:euclid:cs})}; assuma isso de agora em diante.
  \item Mostre que as equações críticas $\nabla f = 0$ são os
        \emph{\omterm{pb:b1:multivar:1}{equações normais}}
        \[
        \begin{pmatrix}
        \norm{C_1}^2 & \langle C_1, C_2\rangle\\
        \langle C_1, C_2\rangle & \norm{C_2}^2
        \end{pmatrix}
        \begin{pmatrix} u\\ v\end{pmatrix}
        =
        \begin{pmatrix} \langle C_1, b\rangle\\ \langle C_2,
        b\rangle\end{pmatrix}
        \]
        --- a matriz Gram de \cref{exo:b1:euclid:11} no
        esquerda --- e que eles dizem exatamente: $ b - (uC_1 + vC_2)
        \perp C_1, C_2$. Conclua com a Parte I e
        \cref{thm:b1:euclid:projection}: o minimizador exclusivo
        dá $ p = uC_1 + vC_2 = $ o projeção ortogonal de
        $ b $ para $\operatorname{Vect}(C_1, C_2)$.
  \item Mostre que o valor mínimo é $\norm{b}^2 - \langle p,
        b\rangle = \norm{b - p}^2$, e desenhe o pitagórico
        foto.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O \omterm{pb:b1:multivar:1}{linha de regressão}.} Pontos de dados
$(x_1, y_1), \dots, (x_n, y_n)$, nem todos $ x_i $ igual. Minimizar
\[
E(\alpha, \beta) = \sum_{i=1}^{n}\bigl(y_i - \alpha - \beta
x_i\bigr)^2 .
\]
Escrever $\bar x = \frac1n\sum x_i $, $\bar y = \frac1n\sum y_i $,
$ v_x = \frac1n\sum x_i^2 - \bar x^2$, $ v_y = \frac1n\sum y_i^2 -
\bar y^2$, $ c_{xy} = \frac1n\sum x_i y_i - \bar x\bar y $.
\begin{enumerate}[resume]
  \item Reconhecer a Parte II com $ C_1 = (1, \dots, 1)$, $ C_2 =
        (x_1, \dots, x_n)$, $ b = (y_1, \dots, y_n)$e escreva
        o \omterm{pb:b1:multivar:1}{equações normais}
        \[
        n\alpha + \Bigl(\sum x_i\Bigr)\beta = \sum y_i,
        \qquad
        \Bigl(\sum x_i\Bigr)\alpha + \Bigl(\sum
        x_i^2\Bigr)\beta = \sum x_i y_i .
        \]
  \item Resolva-os:
        \[
        \beta = \frac{c_{xy}}{v_x},
        \qquad
        \alpha = \bar y - \beta\,\bar x ,
        \]
        e observe que o \emph{\omterm{pb:b1:multivar:1}{linha de regressão}} $ y = \alpha
        + \beta x $ passa pelo ponto médio $(\bar x, \bar
        y)$.
  \item Verificar $ v_x > 0$ exatamente quando o $ x_i $ não são todos
        igual e combine isso com a pergunta 7.
  \item Prove que o erro mínimo é
        \[
        E_{\min} = n\Bigl(v_y - \frac{c_{xy}^2}{v_x}\Bigr)
        = n\,v_y\,(1 - \rho^2),
        \qquad
        \rho = \frac{c_{xy}}{\sqrt{v_x v_y}}
        \quad (v_y \neq 0).
        \]
        Deduzir $\abs\rho \leq 1$, com $\abs\rho = 1$ se e
        somente se os dados estiverem perfeitamente alinhados.
  \item Exemplo resolvido: para os dados $(0,0)$, $(1,1)$, $(2,3)$,
        $(3,4)$, calcular $\bar x, \bar y, v_x, c_{xy}$, o
        \omterm{pb:b1:multivar:1}{linha de regressão}, os quatro resíduos e $ E_{\min}$.
  \item Prove que os resíduos $\varepsilon_i = y_i - \alpha -
        \beta x_i $ da linha ótima satisfaz $\sum_i
        \varepsilon_i = 0$ e $\sum_i x_i\varepsilon_i = 0$,
        e interpretar ambos via ortogonalidade.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Variações e aplicações.}
\begin{enumerate}[resume]
  \item (\omterm{def:b1:arith:prime}{Melhor} constante) Mostre que a constante $\alpha $
        minimizando $\sum_i (y_i - \alpha)^2$ é a média $\bar
        y $, e que o valor mínimo é $ n\,v_y $: a variação
        mede a falha dos dados em serem constantes.
  \item (Linha que passa pela origem) Mostre que a minimização da inclinação
        $\sum_i (y_i - \beta x_i)^2$ é $\beta_0 = \frac{\sum
        x_iy_i}{\sum x_i^2}$, e forneça uma condição sobre os dados
        para $\beta_0$ coincidir com a inclinação
        $ c_{xy}/v_x $ da questão 11.
  \item (Distância até uma linha, novamente) Para um ponto $ M $ e a linha
        $ D $ através $ P_0$ direcionado pelo vetor unitário $ w $,
        minimizar $ g(\tau) = \norm{P_0 + \tau w - M}^2$ e
        deduzir $ d(M, D)^2 = \norm{M - P_0}^2 - \langle M - P_0,
        w\rangle^2$; recuperar a fórmula $ d = \frac{\abs{a x_0 +
        b y_0 + c}}{\sqrt{a^2 + b^2}}$ para a linha $ ax + by + c
        = 0$ no plano.
  \item (Análise de variância) Prove a decomposição $ v_y =
        \beta^2 v_x + \frac{E_{\min}}n $: a variância do
        $ y_i $ se \omterm{def:b1:arith:divides}{divide} na parte explicada pela linha mais
        a variância residual.
  \item (Ajuste parabólico) Para ajustar $ y = a + bx + cx^2$, mostre que
        o \omterm{pb:b1:multivar:1}{equações normais} são os $3 \times 3$ sistema com
        a matriz de momento
        \[
        \begin{pmatrix}
        n & \sum x_i & \sum x_i^2\\
        \sum x_i & \sum x_i^2 & \sum x_i^3\\
        \sum x_i^2 & \sum x_i^3 & \sum x_i^4
        \end{pmatrix},
        \]
        uma matriz Gram que é invertível assim que três de
        o $ x_i $ são distintos \emph{(liberdade de $(1, X, X^2)$
        amostrado nos dados, \cref{exo:b1:euclid:11}; comparar
        as matrizes de momento do problema de fim de semana de
        \cref{ch:b1:det})}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Robustez e síntese.}
\begin{enumerate}[resume]
  \item Classifique o pontos \omterm{thm:b1:multivar:critical}{críticos} dos três quadráticos
        funções
        \[
        x^2 + xy + y^2, \qquad x^2 + 3xy + y^2, \qquad
        x^2 + 2xy + y^2 ,
        \]
        pela Parte I, tratando o terceiro caso degenerado manualmente
        (onde é alcançado o mínimo?).
  \item (Experiência atípica) Anexe o ponto $(10, 0)$ para o
        dados da questão 14 e recalcular a inclinação $\beta $.
        O que aconteceu e por que \emph{ao quadrado} erro então
        sensível a um ponto distante?
  \item (Ponderado \omterm{pb:b1:multivar:1}{mínimos quadrados}) Dados pesos $ w_i > 0$,
        minimizar $\sum_i w_i(y_i - \alpha - \beta x_i)^2$: mostrar
        a solução é dada pelas mesmas fórmulas da questão
        11 com médias ponderadas, variância e covariância
        (defina-os).
  \item Mostre que para a reta ótima, $ E_{\min} = 0$ forças
        todos os pontos fiquem exatamente sobre ele e conecte-se com o
        caso de igualdade $\abs\rho = 1$ da questão 13; teste no
        dados alinhados $(1,1), (2,3), (3,5)$.
  \item Síntese, em quatro frases: por que funções quadráticas
        são a única classe onde a segunda ordem deste capítulo
        o teste não precisa de teorema admitido; como o \omterm{pb:b1:multivar:1}{equações normais}
        identificar a minimização analítica com o \omterm{thm:b1:euclid:projection}{ortogonal
        projeção} de \cref{ch:b1:euclid}; qual a correlação
        medidas de coeficiente e qual desigualdade o limita;
        e quais partes dos Capítulos 18-23 (\omterm{def:b1:vspaces:free}{bases}, Gram e
        matrizes de momento, projeções) reapareceu. Nomeie o
        método e as equações estudadas nas Partes II-III.
\end{enumerate}
\end{problem}
