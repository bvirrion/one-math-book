\chapter{Sequências}\label{ch:b1:seq}

As sequências foram manipuladas no volume do ensino médio com o conceito
de limite tomado em parte de boa-fé. Aqui a teoria é reconstruída sobre
a completude de $\R$ (\cref{ch:b1:reals}): todo teorema clássico
--- convergência monótona, \omterm{thm:b1:seq:adjacent}{sequências adjacentes}, Bolzano--Weierstrass,
critério de Cauchy --- é uma face desse único axioma. O capítulo
termina com o estudo prático das sequências definidas por $u_{n+1} =
f(u_n)$.

\section{Convergência}

\begin{definition}[Limite de uma sequência]\label{def:b1:seq:limit}
Uma sequência $(u_n)$ de reais \emph{converge} para $\ell \in \R$
quando\index{limite!de uma sequência}
\[
\forall \varepsilon > 0,\ \exists N \in \N,\ \forall n \geq N,
\qquad \abs{u_n - \ell} \leq \varepsilon .
\]
Escreve-se $u_n \to \ell$ ou $\lim u_n = \ell$. Uma sequência que não
converge (para nenhum real) \emph{diverge}. Divergência \emph{para
$+\infty$}: $\forall M,\ \exists N,\ \forall n \geq N,\ u_n \geq M$
(analogamente para $-\infty$).
\end{definition}

\begin{example}[Uma demonstração $\varepsilon$--$N$, escrita por extenso uma vez]\label{ex:b1:seq:epsilonN}
Afirmação: $u_n = \dfrac{n^2 + 1}{2n^2 - 3} \to \dfrac12$. Primeiro
isole o erro:
\[
\Bigl| u_n - \frac12 \Bigr|
= \Bigl| \frac{2(n^2 + 1) - (2n^2 - 3)}{2(2n^2 - 3)} \Bigr|
= \frac{5}{2\,\abs{2n^2 - 3}}
= \frac{5}{2\,(2n^2 - 3)} \quad (n \geq 2).
\]
Depois domine-o por algo simples: para $n \geq 2$, $2n^2 - 3
\geq n^2$, de modo que o erro é $\leq \frac{5}{2n^2} \leq \frac 5{2n}$.
Dado $\varepsilon > 0$, a \omterm{thm:b1:reals:archimedes}{propriedade arquimediana} fornece $N \geq
\max\bigl(2, \frac{5}{2\varepsilon}\bigr)$; para $n \geq N$ o
erro é $\leq \varepsilon$. Pronto. A ideia de fechamento: uma demonstração
$\varepsilon$--$N$ tem exatamente três movimentos --- calcular o
erro, limitá-lo por uma expressão elementar decrescente e resolver para
o limiar --- e, depois dos teoremas deste capítulo (operações,
confronto), quase nunca mais se escreve uma demonstração dessas: os
teoremas empacotam os três movimentos de uma vez por todas.
\end{example}

\begin{example}[Divergência para o infinito, certificada]\label{ex:b1:seq:toinfinity}
Afirmação: $u_n = n^2 - 100n \to +\infty$. Fatore o termo dominante:
$u_n = n^2\bigl(1 - \frac{100}{n}\bigr) \geq \frac{n^2}{2}$ para
$n \geq 200$. Dado $M$, tome $N = \max\bigl(200,
\lceil\sqrt{2M}\rceil\bigr)$: para $n \geq N$, $u_n \geq
\frac{n^2}{2} \geq M$. Dois hábitos estão à mostra: a
fatoração pelo termo dominante converte uma competição ($n^2$ contra
$-100n$) numa única escala vezes um fator que tende a $1$; e o limiar
pode ser enorme ($u_{100} = 0$, e a sequência é até negativa antes de
$n = 100$) --- a divergência para $+\infty$ é uma afirmação sobre a
cauda, indiferente a qualquer quantidade finita de mau comportamento.
\end{example}

\begin{proposition}[Primeiras propriedades]\label{prop:b1:seq:first}
\begin{enumerate}
  \item O limite, se existe, é único.
  \item Uma sequência convergente é limitada.
  \item Se $u_n \to \ell$, modificar finitos termos não altera nem a
        convergência nem o limite.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Se $u_n \to \ell$ e $u_n \to \ell'$ com $\ell \neq \ell'$,
tome $\varepsilon = \frac{\abs{\ell - \ell'}}{3}$: além dos dois
limiares, $\abs{\ell - \ell'} \leq \abs{\ell - u_n} + \abs{u_n -
\ell'} \leq 2\varepsilon = \frac23 \abs{\ell - \ell'}$, absurdo.

(2) Com $\varepsilon = 1$: além de $N$, $\abs{u_n} \leq \abs\ell +
1$; os finitos termos anteriores também são limitados, de modo que $\abs{u_n}
\leq \max(\abs{u_0}, \dots, \abs{u_{N-1}}, \abs\ell + 1)$.

(3) Em detalhe: suponha $v_n = u_n$ para $n \geq n_0$ e $u_n \to
\ell$. Dado $\varepsilon > 0$, tome o limiar $N$ de
$(u_n)$: para $n \geq \max(N, n_0)$, $\abs{v_n - \ell} = \abs{u_n
- \ell} \leq \varepsilon$. Logo, $v_n \to \ell$: a definição
quantifica apenas sobre $n \geq N$, e qualquer prefixo finito pode ser
sobrescrito ao custo de aumentar o limiar. (É por isso que hipóteses do
tipo ``para todo $n$ grande'' bastam em todo este capítulo.)
\end{proof}

\begin{theorem}[Operações com limites]\label{thm:b1:seq:operations}
Se $u_n \to \ell$ e $v_n \to m$, então
\[
u_n + v_n \to \ell + m, \qquad u_n v_n \to \ell m, \qquad
\frac{u_n}{v_n} \to \frac{\ell}{m} \ (\text{se} m \neq 0),
\qquad \abs{u_n} \to \abs\ell .
\]
\end{theorem}

\begin{proof}
\emph{Soma:} $\abs{(u_n + v_n) - (\ell + m)} \leq \abs{u_n - \ell} +
\abs{v_n - m} \leq 2\varepsilon$ além do maior limiar.
\emph{Produto:} escreva
\[
u_n v_n - \ell m = (u_n - \ell)\,v_n + \ell\,(v_n - m);
\]
$(v_n)$ é limitada por algum $B$ (\cref{prop:b1:seq:first}), de modo que o
lado direito é $\leq B\abs{u_n - \ell} + \abs{\ell}\,\abs{v_n - m}$,
arbitrariamente pequeno. \emph{Quociente:} basta tratar $\frac
1{v_n}$. Com $\varepsilon = \frac{\abs m}{2}$: além de certo $N_0$,
$\abs{v_n} \geq \frac{\abs m}{2}$, de modo que
\[
\Bigl| \frac{1}{v_n} - \frac 1m \Bigr|
= \frac{\abs{m - v_n}}{\abs{v_n m}}
\leq \frac{2}{m^2}\,\abs{v_n - m} \longrightarrow 0 .
\]
\emph{Valor absoluto:} $\bigl|\abs{u_n} - \abs\ell\bigr| \leq
\abs{u_n - \ell}$ (desigualdade triangular inversa,
\cref{prop:b1:complex:rules}).
\end{proof}

\begin{example}[Operações mais um truque algébrico]\label{ex:b1:seq:conjugatetrick}
Calcule $\lim\,\bigl(\sqrt{n^2 + n} - n\bigr)$. As duas parcelas
tendem separadamente a $+\infty$: o teorema das operações nada diz
sobre a diferença delas (uma \emph{forma indeterminada}). Multiplique
pelo conjugado:
\[
\sqrt{n^2 + n} - n
= \frac{(n^2 + n) - n^2}{\sqrt{n^2 + n} + n}
= \frac{n}{\sqrt{n^2+n} + n}
= \frac{1}{\sqrt{1 + \frac1n} + 1} .
\]
Agora tudo converge: $\sqrt{1 + \frac1n} \to 1$, pois
$0 \leq \sqrt{1 + h} - 1 = \frac{h}{\sqrt{1+h} + 1} \leq h$
(conjugado de novo, e depois confronto com $h = \frac1n$); então o
teorema das operações dá o limite $\frac{1}{1 + 1} =
\frac12$. A ideia de fechamento: o teorema das operações não é uma
calculadora de todos os limites --- as formas indeterminadas ($\infty -
\infty$, $\frac00$, $0 \times \infty$, $1^\infty$) devem antes ser
\emph{transformadas} pela álgebra (conjugados, fatoração do termo
dominante) até que cada pedaço convirja; a máquina sistemática para os
casos resistentes é a expansão assintótica do
\cref{ch:b1:taylor}.
\end{example}

\begin{theorem}[Limites e ordem]\label{thm:b1:seq:order}
\begin{enumerate}
  \item Se $u_n \leq v_n$ para todo $n$ grande e ambas convergem, então
        $\lim u_n \leq \lim v_n$. (As desigualdades estritas \emph{não}
        passam ao limite: $\frac 1n > 0$, mas $\lim = 0$.)
  \item (Teorema do confronto\index{teorema do confronto}) Se $u_n \leq w_n
        \leq v_n$ para todo $n$ grande e $u_n, v_n \to \ell$, então
        $w_n \to \ell$.
  \item Se $u_n \to \ell > 0$, então $u_n > \frac\ell2 > 0$ para todo
        $n$ grande.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Suponha $\ell = \lim u_n > m = \lim v_n$; com $\varepsilon =
\frac{\ell - m}{3}$, os termos grandes satisfazem $v_n \leq m + \varepsilon <
\ell - \varepsilon \leq u_n$, contradizendo $u_n \leq v_n$.

(2) Além dos limiares: $\ell - \varepsilon \leq u_n \leq w_n \leq
v_n \leq \ell + \varepsilon$.

(3) é a \cref{def:b1:seq:limit} com $\varepsilon = \frac\ell2$.
\end{proof}

\begin{example}[Dois confrontos]\label{ex:b1:seq:squeezes}
(i) $\dfrac{\sin n}{n} \to 0$: de $-\frac1n \leq \frac{\sin
n}{n} \leq \frac1n$, com as duas paredes se fechando sobre $0$ --- sem
nenhuma necessidade de entender o numerador errático. (ii) $(2^n +
3^n)^{1/n} \to 3$: enquadre o interior,
\[
3^n \leq 2^n + 3^n \leq 2\cdot3^n
\quad\Longrightarrow\quad
3 \leq (2^n + 3^n)^{1/n} \leq 3\cdot 2^{1/n} ,
\]
e $2^{1/n} = \eu^{\frac{\ln 2}{n}} \to 1$ (como para $5^{1/n}$ no
\cref{exo:b1:seq:2}): o confronto entrega $3$. A ideia de
fechamento: uma soma de exponenciais em competição se comporta como o
seu \emph{maior} termo --- as menores são absorvidas por um fator
constante inofensivo, que a raiz $n$-ésima depois apaga.
\end{example}

\section{Sequências monótonas}

\begin{theorem}[Teorema do limite monótono]\label{thm:b1:seq:monotone}
Uma sequência crescente e limitada superiormente converge, para
$\sup\{u_n : n \in
\N\}$; uma sequência crescente não limitada superiormente diverge
para $+\infty$. (Enunciados espelhados para sequências
decrescentes.)\index{teorema do limite monótono}
\end{theorem}

\begin{proof}
Seja $s = \sup\{u_n\}$ (\cref{thm:b1:reals:sup}). Dado $\varepsilon >
0$, a caracterização por $\varepsilon$
(\cref{prop:b1:reals:epsilon}) fornece $N$ com $u_N > s -
\varepsilon$; pela monotonicidade, $s - \varepsilon < u_N \leq u_n \leq
s$ para todo $n \geq N$: convergência para $s$. Se ilimitada: para todo
$M$ algum $u_N > M$, e a monotonicidade mantém todos os termos
posteriores acima de $M$.
\end{proof}

\begin{example}[O teorema monótono como máquina de existência]\label{ex:b1:seq:existencemachine}
Seja $u_n = \prod_{k=1}^{n} \bigl(1 + \frac{1}{2^k}\bigr)$. Cada
fator excede $1$, de modo que $(u_n)$ é crescente. Limitada
superiormente? Tome logaritmos e use $\ln(1 + x) \leq x$
(o \cref{ex:b1:derivative:convexineq} a antecipa; ou a desigualdade
grosseira $1 + x \leq \eu^x$, do volume do ensino médio):
\[
\ln u_n = \sum_{k=1}^{n} \ln\Bigl(1 + \frac{1}{2^k}\Bigr)
\leq \sum_{k=1}^{n} \frac{1}{2^k} < 1 ,
\]
de modo que $u_n < \eu$. Crescente e limitada: $(u_n)$ converge para algum
$\ell \in \intoc{u_1}{\eu}$ --- um número real perfeitamente bem definido, sem
forma fechada à vista ($\ell = 2.384\dots$). A ideia de
fechamento: o teorema do limite monótono é a máquina de existência mais
barata da análise; foi ela que nomeou o próprio $\eu$
(\cref{ex:b1:seq:e}, abaixo), e no \cref{ch:b1:series} ela decidirá
a convergência de toda série de termos positivos por mera
limitação.
\end{example}

\begin{theorem}[Sequências adjacentes]\label{thm:b1:seq:adjacent}
Sejam $(a_n)$ crescente e $(b_n)$ decrescente, com $b_n - a_n \to
0$. Então ambas convergem, para um limite \emph{comum} $\ell$, e $a_n
\leq \ell \leq b_n$ para todo $n$.\index{sequências adjacentes}
\end{theorem}

\begin{proof}
Primeiro, $a_n \leq b_n$ para todo $n$: a sequência $(b_n - a_n)$ é
decrescente e tende a $0$, de modo que é $\geq 0$ (um termo negativo a
congelaria abaixo de $0$). Então $(a_n)$ é crescente e limitada
superiormente por $b_0$: ela converge para algum $\ell$
(\cref{thm:b1:seq:monotone}); do mesmo modo, $(b_n) \to \ell'$; e $\ell' - \ell = \lim (b_n - a_n) =
0$. As desigualdades $a_n \leq \ell \leq b_n$ decorrem da
monotonicidade ($\ell = \sup a_k \geq a_n$, etc.).
\end{proof}

\begin{example}[O número $\eu$]\label{ex:b1:seq:e}
Ponha $a_n = \sum_{k=0}^{n} \frac{1}{k!}$ e $b_n = a_n + \frac{1}{n
\cdot n!}$ ($n \geq 1$). Então $(a_n)$ cresce; e
\[
b_{n+1} - b_n = \frac{1}{(n+1)!} + \frac{1}{(n+1)(n+1)!} -
\frac{1}{n\,n!}
= \frac{n(n+1) + n - (n+1)^2}{n(n+1)(n+1)!}
= \frac{-1}{n(n+1)(n+1)!} < 0 ,
\]
de modo que $(b_n)$ decresce, e $b_n - a_n \to 0$: adjacentes. O seu limite
comum é (por definição, aqui) o número $\eu \approx 2.71828$; as
desigualdades $a_n < \eu < b_n$ são suficientemente finas para demonstrar que
$\eu \notin
\Q$ (\cref{exo:b1:seq:9}).
\end{example}

\section{Subsequências e Bolzano--Weierstrass}

\begin{definition}[Subsequência]\label{def:b1:seq:subsequence}
Uma \emph{subsequência}\index{subsequência} de $(u_n)$ é uma sequência
$(u_{\varphi(n)})$, em que $\varphi \colon \N \to \N$ é estritamente
crescente (note que $\varphi(n) \geq n$, por indução).
\end{definition}

\begin{proposition}\label{prop:b1:seq:subsequences}
Se $u_n \to \ell$ ($\ell \in \R$ ou $\pm\infty$), toda \omterm{def:b1:seq:subsequence}{subsequência}
tende a $\ell$. Consequentemente, uma sequência com duas \omterm{def:b1:seq:subsequence}{subsequências}
de limites diferentes diverge. Reciprocamente, se $(u_{2n})$ e
$(u_{2n+1})$ convergem ambas para o \emph{mesmo} $\ell$, então $u_n \to
\ell$.
\end{proposition}

\begin{proof}
Além do limiar $N$ de $(u_n)$, todos os índices $\varphi(n) \geq n
\geq N$ servem (a desigualdade $\varphi(n) \geq n$ é a
indução notada na \cref{def:b1:seq:subsequence}: $\varphi(0)
\geq 0$, e $\varphi(n+1) > \varphi(n) \geq n$ força
$\varphi(n+1) \geq n + 1$). Para a recíproca: dado
$\varepsilon$, tome os dois limiares $N_0$ (pares) e $N_1$
(ímpares); um índice arbitrário $n \geq \max(2N_0, 2N_1 + 1)$ é
par, $n = 2k$ com $k \geq N_0$, ou ímpar, $n = 2k+1$ com
$k \geq N_1$ --- e, nos dois casos, $\abs{u_n - \ell} \leq
\varepsilon$: todo índice é coberto por uma das duas
\omterm{def:b1:seq:subsequence}{subsequências}, e essa é toda a questão.
\end{proof}

\begin{example}[Limites subsequenciais]\label{ex:b1:seq:sublimits}
Para $u_n = (-1)^n \frac{n}{n+1}$: a \omterm{def:b1:seq:subsequence}{subsequência} de índices pares tende a
$1$, e a de índices ímpares a $-1$, de modo que a sequência diverge ---
mas o faz de maneira organizada, agrupando-se em torno dos dois valores
$\pm 1$. Para $u_n = \cos\frac{2\pi n}{3}$: as três \omterm{def:b1:seq:subsequence}{subsequências} de
índices $3k$, $3k + 1$ e $3k + 2$ são constantes, iguais a $1$,
$-\frac12$ e $-\frac12$; o \omterm{def:b1:logic:sets}{conjunto} dos limites subsequenciais é $\{1,
-\frac12\}$. A ideia de fechamento: uma sequência \emph{limitada}
converge exatamente quando tem um único limite subsequencial
(\cref{exo:b1:seq:8}); a divergência de uma sequência limitada sempre
significa ao menos dois agrupamentos, e Bolzano--Weierstrass, abaixo,
garante que existe ao menos um.
\end{example}

\begin{theorem}[Bolzano--Weierstrass]\label{thm:b1:seq:bw}
Toda sequência limitada de reais tem uma \omterm{def:b1:seq:subsequence}{subsequência}
convergente.\index{teorema de Bolzano--Weierstrass}
\end{theorem}

\begin{proof}
Seja $u_n \in \intcc{a}{b}$ para todo $n$. Construa segmentos encaixados por
dicotomia: ponha $\intcc{a_0}{b_0} = \intcc{a}{b}$; dado
$\intcc{a_k}{b_k}$ que contém $u_n$ para infinitos $n$, uma das
suas duas metades ainda contém $u_n$ para infinitos $n$ --- chame-a
$\intcc{a_{k+1}}{b_{k+1}}$. As sequências $(a_k)$ e $(b_k)$ são
adjacentes ($b_k - a_k = \frac{b-a}{2^k} \to 0$), com limite comum
$\ell$ (\cref{thm:b1:seq:adjacent}).

Extração: escolha $\varphi(0)$ com $u_{\varphi(0)} \in
\intcc{a_0}{b_0}$ e, indutivamente, $\varphi(k+1) > \varphi(k)$
com $u_{\varphi(k+1)} \in \intcc{a_{k+1}}{b_{k+1}}$ --- o que é possível,
pois esse segmento contém infinitos termos. Então $a_k \leq
u_{\varphi(k)} \leq b_k$, e o teorema do confronto dá
$u_{\varphi(k)} \to \ell$.
\end{proof}

\begin{remark}[O que Bolzano--Weierstrass diz, e o que não diz]
Ele \emph{diz}: só da limitação, alguma \omterm{def:b1:seq:subsequence}{subsequência}
converge --- existência sem fórmula, como a demonstração por dicotomia
deixa claro (nada nos diz \emph{quais} índices sobrevivem). Ele
\emph{não} diz que o limite é único: $((-1)^n)$ tem
\omterm{def:b1:seq:subsequence}{subsequências} convergindo para $1$ e para $-1$, e o \omterm{def:b1:logic:sets}{conjunto} dos
limites subsequenciais pode até ser infinito
(\cref{ex:b1:seq:sublimits}, e todo o \omterm{def:b1:logic:sets}{conjunto} de Cantor no
\cref{pb:b1:topology:1}). Ele não sobrevive à ilimitação:
$(n)$ não tem \omterm{def:b1:seq:subsequence}{subsequência} convergente alguma --- embora sempre se
possa extrair de qualquer sequência ilimitada uma \omterm{def:b1:seq:subsequence}{subsequência} que
tenda a $+\infty$ ou a $-\infty$ (escolha $\varphi(k)$ com
$u_{\varphi(k)} \geq k$, digamos). Usado corretamente, o teorema é uma
\emph{bomba de existência}: ele aparece no ponto crucial do critério de
Cauchy, abaixo, do teorema de Heine e do teorema de Weierstrass ---
sempre para produzir um ponto que nenhuma construção explícita oferece.
\end{remark}

\section{Sequências de Cauchy e completude}

\begin{definition}\label{def:b1:seq:cauchy}
Uma sequência $(u_n)$ é uma \emph{sequência de
Cauchy}\index{sequência de Cauchy} quando os seus termos se tornam
arbitrariamente próximos \emph{uns dos outros}:
\[
\forall \varepsilon > 0,\ \exists N,\ \forall p, q \geq N,
\qquad \abs{u_p - u_q} \leq \varepsilon .
\]
\end{definition}

\begin{example}[Verificando a propriedade de Cauchy à mão]\label{ex:b1:seq:cauchyhand}
Seja $u_n = \sum_{k=0}^{n} \frac{\cos k}{2^k}$ --- sem monotonicidade,
sem limite adivinhável. Para $p > q$:
\[
\abs{u_p - u_q}
= \Bigl| \sum_{k=q+1}^{p} \frac{\cos k}{2^k} \Bigr|
\leq \sum_{k=q+1}^{p} \frac{1}{2^k}
< \frac{1}{2^{q}} ,
\]
pela desigualdade triangular, por $\abs{\cos k} \leq 1$ e por uma soma
geométrica finita. Dado $\varepsilon > 0$, escolha $N$ com $2^{-N}
\leq \varepsilon$: todas as distâncias além de $N$ são $\leq \varepsilon$,
a sequência é de Cauchy e, portanto, converge --- para um limite que
ninguém sabe nomear em forma fechada, o que é exatamente o ponto. A
ideia de fechamento: a dominação geométrica dos incrementos é a maneira
padrão de conquistar a propriedade de Cauchy, e o \cref{ch:b1:series}
engarrafará o argumento como ``convergência absoluta implica
convergência''.
\end{example}

\begin{theorem}[Completude de $\R$]\label{thm:b1:seq:complete}
Uma sequência de reais converge se, e somente se, é uma \omterm{def:b1:seq:cauchy}{sequência de
Cauchy}.\index{completude de R@completude de $\R$}
\end{theorem}

\begin{proof}
($\Rightarrow$) Se $u_n \to \ell$: além do limiar para
$\frac\varepsilon2$, $\abs{u_p - u_q} \leq \abs{u_p - \ell} +
\abs{\ell - u_q} \leq \varepsilon$.

($\Leftarrow$) Seja $(u_n)$ de Cauchy. \emph{Ela é limitada}: com
$\varepsilon = 1$, além de $N$ todos os termos ficam a distância $1$ de $u_N$,
e a cabeça é finita. \emph{Extração}: pelo
\cref{thm:b1:seq:bw}, alguma \omterm{def:b1:seq:subsequence}{subsequência} $u_{\varphi(n)} \to \ell$.
\emph{Conclusão}: dado $\varepsilon > 0$, tome $N$ (de Cauchy, para
$\frac\varepsilon2$) e $n \geq N$ com $\abs{u_{\varphi(n)} - \ell}
\leq \frac\varepsilon2$ e $\varphi(n) \geq N$; então, para todo $p
\geq N$:
\[
\abs{u_p - \ell} \leq \abs{u_p - u_{\varphi(n)}} +
\abs{u_{\varphi(n)} - \ell} \leq \varepsilon . \qedhere
\]
\end{proof}

\begin{remark}
O valor do critério: ele certifica a convergência \emph{sem
nomear o limite}. Ele falha sobre $\Q$ (as truncagens decimais de
$\sqrt 2$ formam uma \omterm{def:b1:seq:cauchy}{sequência de Cauchy} de racionais sem limite
racional): a completude é uma propriedade de $\R$, equivalente ao
axioma do \omterm{def:b1:reals:bounds}{supremo}. Ele é também o cavalo de batalha por trás da
convergência das séries (\cref{ch:b1:series}).
\end{remark}

\begin{example}[Uma sequência de Cauchy com limite invisível]\label{ex:b1:seq:basel}
Seja $S_n = \sum_{k=1}^{n} \frac{1}{k^2}$. Para $p > q \geq 1$:
\[
S_p - S_q = \sum_{k=q+1}^{p} \frac{1}{k^2}
\leq \sum_{k=q+1}^{p} \frac{1}{k(k-1)}
= \sum_{k=q+1}^{p} \Bigl(\frac{1}{k-1} - \frac 1k\Bigr)
= \frac 1q - \frac 1p < \frac 1q ,
\]
de modo que, além de $N > \frac1\varepsilon$, todas as distâncias são $\leq \varepsilon$:
$(S_n)$ é de Cauchy e, portanto, converge. Note o que acabou de
acontecer: demonstramos que um número real específico existe sem ter
nome algum para ele. (Ele vale $\frac{\pi^2}{6}$ --- célebre identidade de
Euler, demonstrada no volume do segundo ano de graduação; nada neste
capítulo poderia nos dizer isso.) Essa divisão de trabalho --- existência
agora, identificação depois, se algum dia --- é todo o sentido do
critério de Cauchy, e o motor da teoria das séries no
\cref{ch:b1:series}.
\end{example}

\section{Sequências recorrentes}

\begin{method}[Estudando $u_{n+1} = f(u_n)$]\label{met:b1:seq:recurrent}
Dados $f$ e um ponto de partida $u_0$:\index{sequência recorrente}
\begin{enumerate}
  \item \emph{\omterm{prop:b1:reals:intervals}{Intervalo} estável:} encontre um \omterm{prop:b1:reals:intervals}{intervalo} $I$ com $f(I)
        \subseteq I$ contendo $u_0$: então todos os $u_n \in I$ (por
        indução).
  \item \emph{Candidatos a limite:} se $u_n \to \ell \in I$ e $f$ é
        contínua em $\ell$ (\cref{ch:b1:continuity}), então $\ell$
        é um \emph{ponto fixo}: $f(\ell) = \ell$. Resolva $f(x) = x$.
  \item \emph{Monotonicidade:} se $f$ é crescente em $I$, então
        $(u_n)$ é monótona (crescente se $u_1 \geq u_0$, e
        decrescente caso contrário); combinado com a limitação, o
        \cref{thm:b1:seq:monotone} conclui. Se $f$ é decrescente,
        estude as duas \omterm{def:b1:seq:subsequence}{subsequências} $(u_{2n})$ e $(u_{2n+1})$,
        que são monótonas para $f \circ f$.
  \item \emph{Controle do erro:} uma desigualdade $\abs{f(x) - \ell} \leq
        k\abs{x - \ell}$ com $k < 1$ dá $\abs{u_n - \ell} \leq
        k^n \abs{u_0 - \ell} \to 0$ diretamente.
\end{enumerate}
\end{method}

\begin{example}[Método de Heron]\label{ex:b1:seq:heron}
Sejam $u_0 = 2$ e $u_{n+1} = \dfrac12\Bigl(u_n +
\dfrac{2}{u_n}\Bigr)$: o antigo algoritmo para
$\sqrt 2$.\index{método de Heron}
\begin{itemize}
  \item \emph{Estabilidade:} para $x > 0$, a desigualdade entre as médias
        aritmética e geométrica dá $\frac12(x + \frac2x) \geq \sqrt{x \cdot
        \frac 2x} = \sqrt 2$; assim, $I = \intco{\sqrt 2}{+\infty}$ é
        estável e contém $u_1$ (com efeito, $u_1 = \frac32 \geq
        \sqrt 2$).
  \item \emph{Monotonicidade:} para $x \geq \sqrt 2$, $\;x - f(x) =
        \frac{x^2 - 2}{2x} \geq 0$: a sequência decresce a partir de
        $u_1$ e é limitada inferiormente por $\sqrt 2$: ela converge.
  \item \emph{Limite:} os pontos fixos resolvem $x = \frac12(x +
        \frac2x)$, isto é, $x^2 = 2$: em $I$, $\ell = \sqrt 2$.
  \item \emph{Velocidade:} $u_{n+1} - \sqrt 2 = \frac{(u_n -
        \sqrt2)^2}{2u_n} \leq \frac{(u_n - \sqrt 2)^2}{2\sqrt 2}$:
        o número de algarismos corretos aproximadamente \emph{dobra} a
        cada passo (convergência quadrática).
\end{itemize}
\end{example}

\begin{remark}[Armadilhas frequentes com limites]
Quatro clássicas. (i) \emph{Passos pequenos não implicam convergência}:
$u_{n+1} - u_n \to 0$ é muito mais fraco que a propriedade de Cauchy ---
as somas harmônicas $H_n$ têm passos $\frac{1}{n+1} \to 0$ e, ainda assim,
divergem para $+\infty$ (\cref{exo:b1:seq:5}); a condição de Cauchy
controla $\abs{u_p - u_q}$ para \emph{todos} os pares grandes, e não
apenas os consecutivos. (ii) \emph{As desigualdades estritas morrem no
limite}: de $u_n < v_n$ para todo $n$ obtém-se apenas $\lim u_n
\leq \lim v_n$ (\cref{thm:b1:seq:order}); $\frac1n > 0$ e, ainda assim,
$\lim = 0$. (iii) \emph{Limitada não é convergente}: $((-1)^n)$
é limitada e diverge; limitação mais \emph{monotonicidade}
converge, e a limitação sozinha só garante uma \omterm{def:b1:seq:subsequence}{subsequência}
convergente (\cref{thm:b1:seq:bw}). (iv) \emph{A equação de ponto fixo
vem em segundo lugar, não em primeiro}: para $u_{n+1} = f(u_n)$,
resolver $f(\ell) = \ell$ identifica o limite \emph{apenas depois}
de demonstrada a convergência. A recorrência $u_{n+1} = 2u_n$ tem o
único ponto fixo $\ell = 0$ e, no entanto, a partir de $u_0 = 1$ a
sequência corre para $+\infty$: a equação $\ell = 2\ell$ nunca teve
direito a um limite. Ordem das operações, sempre: existência
primeiro (\cref{met:b1:seq:recurrent}, passos 1--3), identificação
depois.
\end{remark}

\begin{example}[Um $f$ decrescente: a recorrência áurea]\label{ex:b1:seq:golden}
Sejam $u_0 = 1$ e $u_{n+1} = \dfrac{1}{1 + u_n}$. Aqui $f(x) =
\frac{1}{1+x}$ é \emph{decrescente}, de modo que a sequência não é
monótona (ela alterna em torno do seu limite); o passo de contração
do \cref{met:b1:seq:recurrent} é a ferramenta certa. Estabilidade: se
$x \in \intcc{\frac12}{1}$, então $1 + x \in
\intcc{\frac32}{2}$, de modo que $f(x) \in \intcc{\frac12}{\frac23}
\subseteq \intcc{\frac12}{1}$, e $u_1 = \frac12$ põe a
sequência inteira ali. Ponto fixo: $\ell = \frac{1}{1+\ell}$
com $\ell > 0$ dá $\ell^2 + \ell - 1 = 0$, isto é,
\[
\ell = \frac{\sqrt5 - 1}{2} = 0.6180\dots
\]
(o inverso da razão áurea). Contração: para $x, y \in
\intcc{\frac12}{1}$,
\[
\abs{f(x) - f(y)} = \frac{\abs{x - y}}{(1+x)(1+y)}
\leq \frac{\abs{x-y}}{(3/2)^2} = \frac49\,\abs{x - y} ,
\]
de modo que $\abs{u_n - \ell} \leq \bigl(\frac49\bigr)^{n-1}\abs{u_1 -
\ell} \to 0$: convergência, com velocidade geométrica, sem
necessidade de monotonicidade. A ideia de fechamento: os métodos
monótonos e os métodos de contração repartem entre si o mundo
recorrente --- $f$ crescente dá órbitas monótonas, e $f$ decrescente dá
órbitas alternadas domadas por uma constante de Lipschitz $< 1$ (a
teoria sistemática é o \cref{exo:b1:derivative:11}).
\end{example}

\begin{remark}[Perspectivas dentro deste volume]
As sequências são o instrumento de medida que o restante do volume
encosta em todo objeto. No \cref{ch:b1:topology} elas
\emph{caracterizam} a fecho e a compacidade; no
\cref{ch:b1:continuity} elas transportam limites de funções; no
\cref{ch:b1:integration} as somas de Riemann são sequências que convergem
para a integral; o \cref{ch:b1:series} \emph{é} a teoria de uma
classe especial de sequências, as somas parciais. Até os capítulos de
álgebra as consomem: os iterados de uma matriz no
\cref{ch:b1:matrices} formam sequências cujo comportamento
(convergência de $A^n$) é uma questão de álgebra linear com o
vocabulário deste capítulo. Os dois teoremas a levar consigo por toda
parte: limite monótono (existência a partir da ordem) e
Bolzano--Weierstrass (existência a partir da limitação) --- entre os
dois nasce quase todo limite deste livro.
\end{remark}

\begin{remark}[Sequências complexas]
Uma sequência $(z_n)$ de números complexos converge para $\ell$ quando
$\abs{z_n - \ell} \to 0$; equivalentemente, quando $\Re(z_n) \to \Re(\ell)$
e $\Im(z_n) \to \Im(\ell)$ (compare $\abs{z}$ com $\abs{\Re z} +
\abs{\Im z}$). Os teoremas que não envolvem ordem --- operações,
Bolzano--Weierstrass (extraia duas vezes), critério de Cauchy ---
transferem-se literalmente.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:seq:1}
Diretamente da \cref{def:b1:seq:limit}, demonstre que
$\dfrac{2n+1}{n+3} \to 2$ e que $(u_n) = ((-1)^n)$ diverge.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:seq:2}
Calcule os limites:
\[
\frac{n^2 - 3n + 1}{2n^2 + 5},
\qquad
\sqrt{n+1} - \sqrt n,
\qquad
\frac{2^n + n^3}{3^n - n^2},
\qquad
\sqrt[n]{5}\ \Bigl(= 5^{1/n}\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:seq:3}
Demonstre a comparação padrão: se $\abs{q} < 1$, então $q^n \to 0$
\emph{(escreva $\frac{1}{\abs q} = 1 + h$, com $h > 0$, e use a
desigualdade de Bernoulli $(1+h)^n \geq 1 + nh$, a ser demonstrada por
indução)}. Quais são os comportamentos para $q = 1$, $q = -1$ e $\abs q >
1$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:seq:4}
Seja $u_{n+1} = \frac{u_n + 3}{2}$, com $u_0 = 0$. Encontre o ponto fixo
$\ell$, demonstre que $v_n = u_n - \ell$ é geométrica e dê uma
fórmula explícita e o limite de $(u_n)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:5}
(Série harmônica) Seja $H_n = \sum_{k=1}^{n} \frac 1k$. Demonstre que
$H_{2n} - H_n \geq \frac12$ para todo $n \geq 1$ e conclua que
$(H_n)$ \emph{não} é uma \omterm{def:b1:seq:cauchy}{sequência de Cauchy} e, portanto, diverge (para
$+\infty$, por ser crescente).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:6}
Suponha que $(u_{2n})$, $(u_{2n+1})$ e $(u_{3n})$ convirjam todas.
Demonstre que $(u_n)$ converge. \emph{(Encontre \omterm{def:b1:seq:subsequence}{subsequências} comuns
para igualar os limites.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:7}
Estude a sequência $u_0 = 0$, $u_{n+1} = \sqrt{2 + u_n}$:
estabilidade, monotonicidade, limite. Depois demonstre a estimativa do
erro $\abs{u_n - 2} \leq \dfrac{2}{3^{\,n}}$ \emph{(mostre que $2 - u_{n+1} =
\dfrac{2 - u_n}{2 + \sqrt{2 + u_n}}$ e limite o denominador inferiormente
por $3$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:8}
Seja $(u_n)$ limitada, tal que toda \omterm{def:b1:seq:subsequence}{subsequência} convergente de
$(u_n)$ tenha o \emph{mesmo} limite $\ell$. Demonstre que $u_n \to \ell$.
\emph{(Absurdo mais Bolzano--Weierstrass.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:9}
Com a notação do \cref{ex:b1:seq:e}, suponha que $\eu = \frac pq$
com $p, q \in \N^*$. Usando $a_q < \eu < b_q = a_q + \frac{1}{q\,
q!}$, multiplique por $q!$ e derive uma contradição entre dois
inteiros. Conclua: $\eu$ é irracional.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:10}
(Médias de Cesàro) Para uma sequência $(u_n)_{n \geq 1}$, ponha $c_n =
\frac{u_1 + \dots + u_n}{n}$.
\begin{enumerate}
  \item Demonstre que $u_n \to \ell$ implica $c_n \to \ell$ \emph{(corte
        a soma num limiar $N$; limite a cabeça por uma quantidade fixa
        dividida por $n$, e a cauda por $\varepsilon$)}.
  \item Mostre com um exemplo que a recíproca falha.
  \item Deduza que, se $u_{n+1} - u_n \to \ell$, então $\frac{u_n}{n}
        \to \ell$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:11}
Seja $(u_n)$ satisfazendo $0 \leq u_{m+n} \leq u_m + u_n$ para todos $m, n$
(subaditividade). Demonstre que $\bigl(\frac{u_n}{n}\bigr)$ converge para
$\inf_{n \geq 1} \frac{u_n}{n}$. \emph{(Para $m$ fixado, escreva $n = qm
+ r$ e limite $\frac{u_n}{n}$ usando $u_n \leq q\,u_m + u_r$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:12}
Usando a densidade do \omterm{def:b1:structures:subgroup}{subgrupo} $\Z + 2\pi\Z$ de $(\R, +)$
(\cref{exo:b1:reals:9}), demonstre que a sequência $(\sin n)_{n \in
\N}$ é densa em $\intcc{-1}{1}$ --- em particular, ela diverge.
\end{exercise}

\section{Problema: Cesàro, Stolz e a queda lenta do seno}

\begin{problem}[{Problema de fim de semana --- o teorema de Cesàro--Stolz
e a assintótica $u_n \sim \sqrt{3/n}$ para $u_{n+1} = \sin
u_n$}]
\label{pb:b1:seq:1}
O teorema de Cesàro--Stolz é a regra de l'Hôpital discreta: para
encontrar o limite de um quociente $a_n/b_n$, basta encontrar o
limite do quociente dos \emph{incrementos} $(a_{n+1} -
a_n)/(b_{n+1} - b_n)$. Este problema demonstra o teorema, colhe com ele
limites clássicos e depois o aponta para um alvo famoso:
a sequência $u_{n+1} = \sin u_n$, que se arrasta até $0$ na
velocidade exatamente calculável $u_n \sim \sqrt{3/n}$.\index{teorema de
Cesaro--Stolz@teorema de Cesàro--Stolz} Dois fatos do volume do ensino
médio são concedidos aqui e redemonstrados honestamente mais adiante
neste volume: a desigualdade da reta tangente
\[
\tag{G1} \eu^{u} \geq 1 + u \quad (u \in \R),
\]
redemonstrada por convexidade no \cref{ch:b1:derivative}, e o enquadramento
do seno
\[
\tag{G2} x - \frac{x^3}{6} \;\leq\; \sin x \;\leq\; x -
\frac{x^3}{6} + \frac{x^5}{120} \quad (0 \leq x \leq 1), \qquad
\abs{\sin x} \leq \abs{x} \quad (x \in \R),
\]
redemonstrado pela fórmula de Taylor no \cref{ch:b1:taylor}.

\medskip
\noindent\textbf{Parte I --- Somas sem fórmulas fechadas.}
\begin{enumerate}
  \item Usando $1 + 2 + \dots + n = \frac{n(n+1)}{2}$ e $1^2 +
        \dots + n^2 = \frac{n(n+1)(2n+1)}{6}$, calcule
        $\lim \frac{1 + 2 + \dots + n}{n^2}$ e $\lim \frac{1^2 +
        \dots + n^2}{n^3}$.
  \item Seja $T_n = \sum_{k=1}^n \sqrt k$, para o qual não existe fórmula
        fechada. Demonstre o enquadramento
        \[
        \frac{1}{2\sqrt 2}\,n^{3/2} \;\leq\; T_n \;\leq\;
        n^{3/2}
        \]
        \emph{(guarde apenas os termos $k > \frac n2$ para a estimativa
        inferior)}. Assim, $T_n$ tem a ordem $n^{3/2}$ --- mas com que
        constante? Segure a pergunta até a questão 8.
  \item (Lema telescópico) Seja $(b_n)$ estritamente crescente e
        suponha que, para todo $k \geq N$,
        \[
        m \;\leq\; \frac{a_{k+1} - a_k}{b_{k+1} - b_k} \;\leq\;
        M .
        \]
        Demonstre que $m \leq \dfrac{a_n - a_N}{b_n - b_N} \leq M$
        para todo $n > N$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O teorema de Cesàro--Stolz.} Seja
$(b_n)$ estritamente crescente com $b_n \to +\infty$, e
suponha que $\dfrac{a_{n+1} - a_n}{b_{n+1} - b_n} \to \ell \in \R$.
\begin{enumerate}
  \item Fixe $\varepsilon > 0$. Mostre que existe $N$ tal que
        $\ell - \varepsilon \leq \dfrac{a_n - a_N}{b_n - b_N}
        \leq \ell + \varepsilon$ para todo $n > N$.
  \item Estabeleça, para $n > N$, a identidade
        \[
        \frac{a_n}{b_n} - \ell
        = \frac{a_N - \ell\,b_N}{b_n}
        + \Bigl(1 - \frac{b_N}{b_n}\Bigr)
        \Bigl(\frac{a_n - a_N}{b_n - b_N} - \ell\Bigr),
        \]
        e conclua o teorema: $\dfrac{a_n}{b_n} \to \ell$.
  \item Demonstre a variante $+\infty$: se $\dfrac{a_{n+1} -
        a_n}{b_{n+1} - b_n} \to +\infty$ (mesmas hipóteses sobre
        $(b_n)$), então $\dfrac{a_n}{b_n} \to +\infty$.
  \item Tome $b_n = n$: recupere o teorema da média de Cesàro do
        \cref{exo:b1:seq:10}. Depois mostre que a recíproca de
        Cesàro--Stolz falha: para $a_n = (-1)^n$, $b_n = n$, o
        quociente $a_n/b_n$ converge, ao passo que o quociente dos
        incrementos não. Stolz é uma via de mão única.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Primeiros dividendos.}
\begin{enumerate}
  \item Demonstre que $(1+h)^{3/2} - 1 = \dfrac{3h + 3h^2 +
        h^3}{(1+h)^{3/2} + 1}$ por conjugação, deduza
        $n\bigl((1 + \tfrac1n)^{3/2} - 1\bigr) \to \tfrac32$ e conclua com Cesàro--Stolz:
        \[
        T_n = \sum_{k=1}^{n} \sqrt k \;\sim\; \tfrac23\,
        n^{3/2} ,
        \]
        resolvendo o suspense da questão 2.
  \item Só a partir de (G1), deduza o enquadramento do logaritmo
        \[
        \frac{t}{1 + t} \;\leq\; \ln(1 + t) \;\leq\; t
        \qquad (t > -1)
        \]
        \emph{(aplique (G1) em $u = \ln(1+t)$ e em $u =
        -t/(1+t)$)}.
  \item Mostre que $b_n = \ln n$ é estritamente crescente com
        $\ln n \to +\infty$, e demonstre, com Cesàro--Stolz e a
        questão 9, que
        \[
        H_n = \sum_{k=1}^{n} \frac 1k \;\sim\; \ln n .
        \]
        (A estrutura mais fina $H_n = \ln n + \gamma + o(1)$ é o
        problema de fim de semana do \cref{ch:b1:series}.)
  \item (De razões a raízes) Seja $u_n > 0 $ com
        $\frac{u_{n+1}}{u_n} \to L > 0$. Usando a questão 9, mostre que
        $\ln\frac{u_{n+1}}{u_n} \to \ln L$; aplique Cesàro para
        concluir que $\frac{\ln u_n}{n} \to \ln L$ e, depois, com (G1),
        que $u_n^{1/n} \to L$. Aplicação: calcule
        $\lim\,\binom{2n}{n}^{1/n}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A queda lenta do seno.} Sejam
$u_0 \in \R$ e $u_{n+1} = \sin u_n$.
\begin{enumerate}
  \item A partir de (G2), mostre que $0 < \sin x < x$ para $0 < x \leq 1$.
        Deduza: $u_1 \in \intcc{-1}{1}$; se $u_1 = 0$, a
        sequência é nula a partir do posto $1$; e, se $u_1 > 0$ (o
        caso $u_1 < 0$ sendo simétrico, pois $\sin$ é ímpar), então
        $(u_n)_{n \geq 1}$ é estritamente decrescente, positiva e
        converge para $0$ \emph{(identifique o limite via $\ell =
        \sin \ell$, usando $\abs{\sin a - \sin b} \leq \abs{a -
        b}$, ela própria consequência de (G2) e da fórmula de
        transformação de produto em soma)}.
  \item Suponha de agora em diante que $u_1 \in \intoc{0}{1}$. Mostre, por
        confronto, usando (G2):
        \[
        \frac{\sin u_n}{u_n} \to 1
        \qquad\text{e}\qquad
        \frac{u_n - \sin u_n}{u_n^{3}} \to \frac16 .
        \]
  \item Demonstre a fatoração
        \[
        w_n := \frac{1}{u_{n+1}^{2}} - \frac{1}{u_n^{2}}
        = \frac{u_n - \sin u_n}{u_n^{3}} \cdot
        \frac{u_n + \sin u_n}{u_n} \cdot
        \Bigl(\frac{u_n}{\sin u_n}\Bigr)^{2},
        \]
        e deduza que $w_n \to \frac13$.
  \item Conclua com o \cref{exo:b1:seq:10} (versão dos incrementos)
        que $\frac{1}{n\,u_n^{2}} \to \frac13$ e, depois, por um
        argumento de conjugação para a raiz quadrada, o resultado
        central:
        \[
        \sqrt n\;u_n \longrightarrow \sqrt 3 ,
        \qquad\text{i.e.}\qquad
        u_n \sim \sqrt{\frac 3n} .
        \]
  \item Quantifique a lentidão: mostre que, a partir de certo ponto, $\sqrt{2/n}
        \leq u_n \leq 2/\sqrt n$, de modo que atingir $u_n \leq
        10^{-2}$ exige mais de $20\,000$ iterações (cerca de
        $30\,000$, pela assintótica). Contraste com o \omterm{ex:b1:seq:heron}{método de Heron}
        (\cref{ex:b1:seq:heron}) e explique a razão
        estrutural: no ponto fixo $0$, a inclinação de
        $\sin$ vale $1$ (um ponto fixo \emph{neutro}), ao passo que
        iterações que \omterm{def:b1:arith:divides}{dividem} o erro ao meio precisam de inclinação de
        módulo $< 1$.
  \item Mostre que, para \emph{todo} ponto de partida $u_0 \in \R$,
        ou $u_n = 0$ a partir do posto $1$, ou $\abs{u_n} \sim
        \sqrt{3/n}$ --- a queda é universal, e só o sinal
        se lembra de $u_0$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- O princípio geral.} O seno é
uma instância de uma máquina.
\begin{enumerate}
  \item Sejam $u_n > 0$, $u_n \to 0$ e $\dfrac{u_n -
        u_{n+1}}{u_n^{2}} \to a > 0$. Demonstre sucessivamente:
        $\frac{u_{n+1}}{u_n} \to 1$; depois $\frac{1}{u_{n+1}} -
        \frac{1}{u_n} \to a$; depois $n\,u_n \to \frac1a$.
  \item (Modelo exato) Para $u_{n+1} = \dfrac{u_n}{1 + u_n}$, com $u_0
        > 0$: mostre que $\frac{1}{u_n}$ é aritmética, resolva exatamente
        e confira a conclusão da questão 18 contra a fórmula exata.
  \item Para $u_{n+1} = u_n \eu^{-u_n}$, com $u_0 > 0$: mostre que $u_n \to
        0$, use (G1) para enquadrar $\frac{1 - \eu^{-t}}{t}$ entre
        $\frac{1}{1+t}$ e $1$ para $t > 0$, e conclua que $u_n
        \sim \frac 1n$.
  \item (Contato cúbico, telescópio ao quadrado) Sejam $u_n > 0$, $u_n
        \to 0$, $\dfrac{u_n - u_{n+1}}{u_n^{3}} \to a > 0$.
        Adapte a fatoração da questão 14 para mostrar que
        $\frac{1}{u_{n+1}^2} - \frac{1}{u_n^2} \to 2a$ e
        conclua que $n\,u_n^{2} \to \frac{1}{2a}$. Verifique que $a =
        \frac16$ recupera a Parte IV.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Limites do método e morais.}
\begin{enumerate}
  \item Mostre que a hipótese $b_n \to +\infty$ não pode ser
        abandonada: para $a_n = 2 - 2^{-n}$ e $b_n = 1 - 2^{-n}$,
        o quociente dos incrementos tende a $1$, ao passo que $\frac{a_n}
        {b_n} \to 2$. Aponte a linha exata da demonstração da questão 5
        que se quebra.
  \item (Stolz duas vezes) Demonstre que $\sum_{k=1}^{n} H_k \sim n \ln n$
        \emph{(uma \omterm{def:b1:logic:map}{aplicação} de Cesàro--Stolz, depois a questão
        10; limite $(n+1)\ln(n+1) - n\ln n$ usando a questão 9)}.
  \item (Médias geométricas) Se $u_n > 0$ e $u_n \to \ell > 0$,
        mostre que $(u_1 u_2 \cdots u_n)^{1/n} \to \ell$; se $u_n \to
        +\infty$,
        mostre que $(u_1 \cdots u_n)^{1/n} \to +\infty$.
        Deduza que $(n!)^{1/n} \to +\infty$.
  \item Síntese, uma frase para cada: (i) onde exatamente entrou a
        completude neste problema; (ii) em que sentido
        Cesàro--Stolz é uma regra de l'Hôpital discreta (a sua gêmea
        diferencial repousa no teorema do valor médio do
        \cref{ch:b1:derivative}); (iii) enuncie a heurística que
        liga a ordem de contato de $f$ num ponto fixo neutro
        ao expoente de decaimento de $u_{n+1} = f(u_n)$;
        (iv) rastreie a constante $3$ de $\sqrt{3/n}$ ao longo do
        encadeamento $\frac16 \to \frac13 \to 3$.
\end{enumerate}
\end{problem}
