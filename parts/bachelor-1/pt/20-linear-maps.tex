\chapter{Aplicações Lineares}\label{ch:b1:linmaps}

As aplicações que vale a pena estudar entre \omterm{def:b1:vspaces:def}{espaços vetoriais} são as
compatíveis com a estrutura: as \emph{\omterm{def:b1:linmaps:def}{aplicações lineares}}. Os seus
dois \omterm{def:b1:vspaces:subspace}{subespaços} fundamentais --- \omterm{def:b1:linmaps:kerim}{núcleo} e \omterm{def:b1:linmaps:kerim}{imagem} --- medem a
injetividade e a sobrejetividade, e em \omterm{def:b1:findim:def}{dimensão finita} o teorema do
\omterm{def:b1:linmaps:kerim}{núcleo} e da \omterm{def:b1:linmaps:kerim}{imagem} amarra os seus tamanhos numa única lei de
conservação. Projeções e simetrias, e depois \omterm{def:b1:linmaps:forms}{formas lineares} e
\omterm{def:b1:linmaps:forms}{hiperplanos}, fecham o capítulo.

\section{Definições e primeiras propriedades}

\begin{definition}\label{def:b1:linmaps:def}
Sejam $E, F$ \omterm{def:b1:vspaces:def}{espaços vetoriais} sobre $K$. Uma \omterm{def:b1:logic:map}{aplicação} $u \colon E \to F$ é
\emph{linear}\index{aplicação linear} quando
\[
\forall x, y \in E,\ \forall \lambda \in K, \qquad
u(x + \lambda y) = u(x) + \lambda u(y).
\]
Então $u(0) = 0$ e $u(\sum \lambda_i x_i) = \sum \lambda_i u(x_i)$.
O \omterm{def:b1:logic:sets}{conjunto} $\mathcal{L}(E, F)$ das aplicações lineares é ele próprio um \omterm{def:b1:vspaces:def}{espaço
vetorial}; a composição de aplicações lineares é linear, e bilinear
no par. Um \emph{endomorfismo} é uma \omterm{def:b1:logic:map}{aplicação} linear $E \to E$; um
\emph{isomorfismo} é uma \omterm{def:b1:logic:map}{aplicação} linear \omterm{def:b1:logic:inj}{bijetiva} (a sua inversa é
então automaticamente linear); a inversa $u^{-1}$ de um isomorfismo
e as compostas de isomorfismos são isomorfismos.
\end{definition}

\begin{proof}[Demonstração de que a inversa é linear]
Seja $u$ \omterm{def:b1:linmaps:def}{linear} \omterm{def:b1:logic:inj}{bijetiva}, $y, y' \in F$ e $\lambda \in K$.
Ponha $x = u^{-1}(y)$ e $x' = u^{-1}(y')$. Então
\[
u\bigl(x + \lambda x'\bigr) = u(x) + \lambda u(x')
= y + \lambda y' ,
\]
e aplicando $u^{-1}$ às duas pontas: $u^{-1}(y + \lambda y') = x
+ \lambda x' = u^{-1}(y) + \lambda\,u^{-1}(y')$. Nada sobre $u^{-1}$ foi
calculado: a linearidade transporta-se apenas através da propriedade
que define $u$ --- um padrão que vale a pena reter, pois a estrutura
muitas vezes viaja de graça ao longo das bijeções.
\end{proof}

\begin{proposition}[Uma aplicação linear é conhecida numa base]\label{prop:b1:linmaps:basis}
Seja $(e_1, \dots, e_n)$ uma \omterm{def:b1:vspaces:free}{base} de $E$ e $(v_1, \dots, v_n)$ vetores arbitrários de $F$.
Existe exatamente uma \omterm{def:b1:linmaps:def}{aplicação linear} $u \colon
E \to F$ com $u(e_i) = v_i$ para todo
$i$. Além disso:
\[
u \text{ injetiva} \iff (v_i) \text{ livre};
\qquad
u \text{ sobrejetiva} \iff (v_i) \text{ gera } F .
\]
\end{proposition}

\begin{proof}
Existência e unicidade: todo $x$ tem \omterm{prop:b1:vspaces:coordinates}{coordenadas} únicas $x = \sum
\lambda_i e_i$
(a \cref{prop:b1:vspaces:coordinates}); a linearidade força $u(x) = \sum \lambda_i v_i$, e essa fórmula define de fato
uma \omterm{def:b1:linmaps:def}{aplicação linear}.

Injetividade: pela \cref{prop:b1:linmaps:kernel} abaixo, $u$ é \omterm{def:b1:logic:inj}{injetiva} se e somente se o
seu \omterm{def:b1:linmaps:kerim}{núcleo} é trivial. Ora, $u\bigl(\sum\lambda_i
e_i\bigr) = 0$ significa exatamente $\sum\lambda_i v_i = 0$. Se
$(v_i)$ é \omterm{def:b1:vspaces:free}{livre}, isto força todo $\lambda_i = 0$, isto é, o \omterm{def:b1:linmaps:kerim}{núcleo} reduz-se a
$0$: \omterm{def:b1:logic:inj}{injetiva}. Se $(v_i)$ é linearmente dependente, uma relação não
trivial $\sum\lambda_i v_i = 0$ produz o vetor não nulo $\sum\lambda_i e_i$ no \omterm{def:b1:linmaps:kerim}{núcleo} ($(e_i)$ é
\omterm{def:b1:vspaces:free}{livre}): não \omterm{def:b1:logic:inj}{injetiva}. As duas condições correspondem termo a termo.

Sobrejetividade: a \omterm{def:b1:linmaps:kerim}{imagem} de $u$ é o \omterm{def:b1:logic:sets}{conjunto} de todos os $\sum\lambda_i
v_i$,
isto é, exatamente $\operatorname{Vect}(v_1, \dots, v_n)$, que é igual a $F$ se e somente se a
família é geradora.
\end{proof}

\begin{definition}[Núcleo e imagem]\label{def:b1:linmaps:kerim}
Para $u \in \mathcal{L}(E, F)$:
\[
\ker u = \{x \in E : u(x) = 0\} \subseteq E,
\qquad
\operatorname{im} u = u(E) \subseteq F ,
\]
ambos \omterm{def:b1:vspaces:subspace}{subespaços} (verificação direta com o
critério).\index{núcleo}\index{imagem}
\end{definition}

\begin{method}[Núcleo e imagem, na prática]
\label{met:b1:linmaps:kerim}
\emph{\omterm{def:b1:linmaps:kerim}{Núcleo}}: escreva $u(x) = 0$ como um sistema nas \omterm{prop:b1:vspaces:coordinates}{coordenadas}
(ou nos coeficientes) de $x$, resolva e parametrize --- o \omterm{def:b1:linmaps:kerim}{núcleo}
sai com uma \omterm{def:b1:vspaces:free}{base} anexada (o \cref{met:b1:findim:computedim}). \emph{\omterm{def:b1:linmaps:kerim}{Imagem}}: é o \omterm{def:b1:vspaces:span}{espaço
gerado} pelas imagens de \emph{qualquer} família geradora de $E$ ---
em geral uma \omterm{def:b1:vspaces:free}{base}, logo $\operatorname{im} u =
\operatorname{Vect}\bigl(u(e_1), \dots, u(e_n)\bigr)$; depois elimine as imagens
redundantes para extrair uma \omterm{def:b1:vspaces:free}{base}. \emph{Atalho}: calcule aquele
dos dois que for mais fácil e obtenha de graça a dimensão do outro
pelo teorema do \omterm{def:b1:linmaps:kerim}{núcleo} e da \omterm{def:b1:linmaps:kerim}{imagem} (\cref{thm:b1:linmaps:ranknullity}); quando se conhece um
candidato plausível a \omterm{def:b1:vspaces:subspace}{subespaço} \omterm{def:b1:linmaps:kerim}{imagem}, comparar dimensões promove
a inclusão fácil a igualdade (o \cref{thm:b1:findim:subspaces}). Os dois atalhos são usados
no \cref{ex:b1:linmaps:delta} abaixo.
\end{method}

\begin{proposition}\label{prop:b1:linmaps:kernel}
$u$ é \omterm{def:b1:logic:inj}{injetiva} $\iff$ $\ker u = \{0\}$; $u$ é \omterm{def:b1:logic:inj}{sobrejetiva} $\iff$ $\operatorname{im} u = F$.
\end{proposition}

\begin{proof}
Como para os \omterm{def:b1:structures:group}{grupos} (a \cref{prop:b1:structures:kernel}): $u(x) = u(y) \iff
u(x - y) = 0 \iff x - y \in \ker u$. O segundo ponto é a
definição.
\end{proof}

\section{O teorema do núcleo e da imagem}

\begin{definition}\label{def:b1:linmaps:rank}
O \emph{posto}\index{posto!de uma aplicação linear} de $u \in \mathcal{L}(E,
F)$ (com
$E$ de \omterm{def:b1:findim:def}{dimensão finita}) é $\operatorname{rk} u = \dim
\operatorname{im} u$ --- que é também o posto da família
$\bigl(u(e_1), \dots, u(e_n)\bigr)$ para qualquer \omterm{def:b1:vspaces:free}{base} $(e_i)$ de $E$.
\end{definition}

\begin{theorem}[Teorema do núcleo e da imagem]\label{thm:b1:linmaps:ranknullity}
Seja $E$ de \omterm{def:b1:findim:def}{dimensão finita} e $u \in \mathcal{L}(E, F)$.
Então\index{teorema do núcleo e da imagem}
\[
\dim E = \dim \ker u + \operatorname{rk} u .
\]
Mais precisamente, se $S$ é um \omterm{def:b1:vspaces:subspace}{subespaço} \omterm{def:b1:vspaces:sum}{suplementar} qualquer de
$\ker u$ em $E$, então $u$ se restringe a um \emph{isomorfismo} de
$S$ sobre $\operatorname{im} u$.
\end{theorem}

\begin{proof}
Seja $S$ tal que $E = \ker u \oplus S$ (o \cref{thm:b1:findim:subspaces}), e seja $v \colon S \to
\operatorname{im} u$ a restrição de $u$.

$v$ é \omterm{def:b1:logic:inj}{injetiva}: $\ker v = S \cap \ker u = \{0\}$.

$v$ é \omterm{def:b1:logic:inj}{sobrejetiva}: todo $u(x)$ com $x = k + s$ ($k \in \ker u$, $s \in S$) é igual
a $u(s) = v(s)$.

Logo $v$ é um isomorfismo; um isomorfismo leva uma \omterm{def:b1:vspaces:free}{base} numa \omterm{def:b1:vspaces:free}{base}
(a \cref{prop:b1:linmaps:basis}), logo $\dim S = \dim\operatorname{im}
u$, e $\dim E = \dim\ker u + \dim S$ conclui.
\end{proof}

\begin{example}[Construir uma aplicação sob encomenda]
\label{ex:b1:linmaps:tospec}
Construa $u \in \mathcal{L}(\R^3)$ com $\ker u =
\operatorname{Vect}(1,1,1)$ e $\operatorname{im} u = \{z =
0\}$. Primeiro a coerência: o teorema
do \omterm{def:b1:linmaps:kerim}{núcleo} e da \omterm{def:b1:linmaps:kerim}{imagem} exige $1 + 2 = 3$ --- coerente, logo pode existir
uma solução. Escolha uma \omterm{def:b1:vspaces:free}{base} adaptada ao \omterm{def:b1:linmaps:kerim}{núcleo}, digamos $\bigl((1,1,1),\ e_1,\ e_2\bigr)$
(o \cref{ex:b1:findim:completion}), e prescreva as imagens (a \cref{prop:b1:linmaps:basis}):
\[
u(1,1,1) = 0, \qquad u(e_1) = e_1, \qquad u(e_2) = e_2 .
\]
Então $\ker u \supseteq \operatorname{Vect}(1,1,1)$ e $\operatorname{im} u = \operatorname{Vect}(e_1, e_2) = \{z =
0\}$; o teorema do \omterm{def:b1:linmaps:kerim}{núcleo} e da \omterm{def:b1:linmaps:kerim}{imagem} força $\dim\ker u = 1$,
logo o \omterm{def:b1:linmaps:kerim}{núcleo} é exatamente a reta prescrita. Explicitamente,
decompondo $(x, y, z)
= z(1,1,1) + (x - z)e_1 + (y - z)e_2$:
\[
u(x, y, z) = (x - z,\ y - z,\ 0).
\]
A receita generaliza-se: uma \omterm{def:b1:linmaps:def}{aplicação linear} com \omterm{def:b1:linmaps:kerim}{núcleo} prescrito
$N$ e \omterm{def:b1:linmaps:kerim}{imagem} prescrita $I$ existe exatamente quando $\dim N + \dim I = \dim E$ --- a
necessidade é o teorema do \omterm{def:b1:linmaps:kerim}{núcleo} e da \omterm{def:b1:linmaps:kerim}{imagem}, e a suficiência é
esta construção.
\end{example}

\begin{corollary}\label{cor:b1:linmaps:samedim}
Se $\dim E = \dim F$ (finitas), então para $u \in \mathcal{L}(E, F)$:
\[
u \text{ injetiva} \iff u \text{ sobrejetiva} \iff u \text{bijetiva}.
\]
Em particular isto vale para os endomorfismos em \omterm{def:b1:findim:def}{dimensão finita}.
(Falha em dimensão infinita: em $K[X]$, a \omterm{def:b1:derivative:def}{derivada} é \omterm{def:b1:logic:inj}{sobrejetiva} mas
não \omterm{def:b1:logic:inj}{injetiva}, e $P \mapsto XP$ é \omterm{def:b1:logic:inj}{injetiva} mas não \omterm{def:b1:logic:inj}{sobrejetiva}.)
\end{corollary}

\begin{proof}
\omterm{def:b1:logic:inj}{Injetiva} $\iff \dim\ker u = 0 \iff \operatorname{rk} u = \dim E =
\dim F \iff \operatorname{im} u = F$ (um \omterm{def:b1:vspaces:subspace}{subespaço} de dimensão máxima é o espaço inteiro,
o \cref{thm:b1:findim:subspaces}) $\iff$ \omterm{def:b1:logic:inj}{sobrejetiva}.
\end{proof}

\begin{example}[Interpolação, estruturalmente]\label{ex:b1:linmaps:interpolation}
Fixe $x_0, \dots, x_n$ distintos e seja $u \colon \R_n[X] \to
\R^{n+1}$, $P \mapsto (P(x_0), \dots, P(x_n))$: \omterm{def:b1:linmaps:def}{linear}. O seu \omterm{def:b1:linmaps:kerim}{núcleo} é
$\{P : \deg P \leq n,\ n+1 \text{ roots}\} = \{0\}$ (o \cref{cor:b1:poly:nroots}). Dimensões iguais $n + 1$: $u$ é um isomorfismo ---
existência \emph{e} unicidade do interpolante de Lagrange (o \cref{thm:b1:poly:lagrange})
numa linha.

O mesmo padrão de uma linha trata dados que misturam valores e
\omterm{def:b1:derivative:def}{derivadas}: $v \colon \R_3[X] \to \R^4$, $P \mapsto \bigl(P(0),
P'(0), P(1), P'(1)\bigr)$ é \omterm{def:b1:linmaps:def}{linear}, e o seu \omterm{def:b1:linmaps:kerim}{núcleo} consiste nos
\omterm{def:b1:poly:def}{polinômios} de grau $\leq 3$ com raízes duplas em $0$ \emph{e} em
$1$, isto é, divisíveis por $X^2(X-1)^2$, de grau $4$: só $P = 0$.
Dimensões iguais de novo: toda quádrupla de dados $(P(0),
P'(0), P(1), P'(1))$ é realizada
por exatamente uma cúbica --- a interpolação de Hermite, garantida
por um cálculo de \omterm{def:b1:linmaps:kerim}{núcleo} antes de escrever qualquer fórmula (o
problema de fim de semana do \cref{ch:b1:det} encontra o seu determinante).
\end{example}

\begin{example}[O teorema do núcleo e da imagem em ação: o operador de
diferenças]
\label{ex:b1:linmaps:delta}
Seja $\Delta \colon \R_n[X] \to \R_n[X]$, $P \mapsto P(X+1) -
P(X)$: \omterm{def:b1:linmaps:def}{linear}. \omterm{def:b1:linmaps:kerim}{Núcleo}: se $\Delta P = 0$, então $P(0) = P(1) =
P(2) = \dots$, logo
$P - P(0)$ tem infinitas raízes e se anula (o \cref{cor:b1:poly:nroots}): $\ker\Delta$ é a reta das
constantes. Pelo teorema do \omterm{def:b1:linmaps:kerim}{núcleo} e da \omterm{def:b1:linmaps:kerim}{imagem}: $\operatorname{rk}\Delta = (n + 1) - 1
= n$. Como $\deg \Delta P < \deg P$
para $P$ não constante (os termos de topo se cancelam), $\operatorname{im}\Delta \subseteq \R_{n-1}[X]$, que
tem dimensão exatamente $n$: a inclusão é uma igualdade. Conclusão,
sem calcular nenhuma \omterm{def:b1:logic:map}{pré-imagem}: \emph{todo} \omterm{def:b1:poly:def}{polinômio} $Q$ de grau
$\leq n - 1$ é uma diferença $Q = P(X+1)
- P(X)$ --- a primitiva discreta existe.
(Compare com o problema de fim de semana do \cref{ch:b1:vspaces}, em que $\Delta$
foi invertido explicitamente na \omterm{def:b1:vspaces:free}{base} binomial.)
\end{example}

\begin{example}[Contabilidade do posto ao longo de uma composição]
\label{ex:b1:linmaps:rankbookkeeping}
Em $\R_2[X]$, componha a \omterm{def:b1:derivative:def}{derivada} $D(P) = P'$ (\omterm{def:b1:linmaps:rank}{posto} $2$: \omterm{def:b1:linmaps:kerim}{imagem}
$\R_1[X]$, \omterm{def:b1:linmaps:kerim}{núcleo} as constantes) consigo mesma. Então $D
\circ D = D^2$ leva
$P \mapsto P''$, com \omterm{def:b1:linmaps:kerim}{imagem} $\R_0[X]$: \omterm{def:b1:linmaps:rank}{posto} $1$. Compare com as cotas
gerais: a grosseira dá $\operatorname{rk} D^2 \leq \min(2, 2) = 2$; a fórmula exata do \cref{exo:b1:linmaps:12} contabiliza
a perda com precisão,
\[
\operatorname{rk} D^2 = \operatorname{rk} D -
\dim\bigl(\ker D \cap \operatorname{im} D\bigr) = 2 - 1 = 1 ,
\]
pois as constantes (\omterm{def:b1:linmaps:kerim}{núcleo} do $D$ externo) ficam dentro de
$\R_1[X]$ (\omterm{def:b1:linmaps:kerim}{imagem} do $D$ interno) com dimensão $1$. Perde-se \omterm{def:b1:linmaps:rank}{posto}
exatamente onde o \omterm{def:b1:linmaps:kerim}{núcleo} externo embosca a \omterm{def:b1:linmaps:kerim}{imagem} interna --- a
frase a reter quando os \omterm{def:b1:linmaps:rank}{postos} de uma composição se comportam mal.
\end{example}

\begin{example}[Núcleo e imagem do operador de Euler]
\label{ex:b1:linmaps:euler}
Em $\R_n[X]$, seja $u(P) = X\,P'$ (\omterm{def:b1:linmaps:def}{linear}: derivar e multiplicar por $X$ o
são). \emph{\omterm{def:b1:linmaps:kerim}{Núcleo}}: $XP' = 0$ força $P' = 0$ (um produto de
\omterm{def:b1:poly:def}{polinômios} só se anula se um dos fatores se anular), logo $\ker u$
é a reta das constantes. \emph{\omterm{def:b1:linmaps:kerim}{Imagem}}: na \omterm{def:b1:vspaces:free}{base} monomial,
\[
u(X^k) = k\,X^{k} \qquad (k = 0, 1, \dots, n),
\]
logo $\operatorname{im} u = \operatorname{Vect}(X, 2X^2, \dots,
nX^n) = \operatorname{Vect}(X, X^2, \dots, X^n)$: os \omterm{def:b1:poly:def}{polinômios} com termo constante nulo. Confira contra
o teorema do \omterm{def:b1:linmaps:kerim}{núcleo} e da \omterm{def:b1:linmaps:kerim}{imagem}: $\operatorname{rk} u = (n + 1) - 1 = n$, que é de fato a dimensão
encontrada. Duas observações a guardar. Primeiro, aqui $\operatorname{im} u \oplus \ker u = \R_n[X]$ ---
mas isso é um \emph{feliz acaso} deste operador, não um teorema:
para o operador do tipo deslocamento $v(P) = P'$ em $\R_1[X]$, $\ker v = \operatorname{im}
v = \R_0[X]$ e a
soma não é direta. Segundo, a relação $u(X^k) = kX^k$ diz que cada monômio é
apenas reescalado por $u$ --- uma \omterm{def:b1:vspaces:free}{base} adaptada à \omterm{def:b1:logic:map}{aplicação}, o
germe da ideia de autovalor desenvolvida no volume do segundo ano
de graduação.
\end{example}

\section{Projeções e simetrias}

\begin{definition}\label{def:b1:linmaps:projection}
Seja $E = F \oplus G$. A \emph{projeção sobre $F$ ao longo de
$G$}\index{projeção} leva $x = f + g$ (decomposição única) em $p(x) = f$; a
\emph{simetria}\index{simetria} associada é $s(x)
= f - g$. Ambas são
\omterm{def:b1:linmaps:def}{lineares}, e $s = 2p - \mathrm{id}$.
\end{definition}

\begin{theorem}[Caracterização algébrica]\label{thm:b1:linmaps:projchar}
\begin{enumerate}
  \item Um endomorfismo $p$ é uma \omterm{def:b1:linmaps:projection}{projeção} (sobre algum $F$ ao longo
        de algum $G$) se e somente se $p \circ p = p$; então $F =
        \operatorname{im} p = \ker(p - \mathrm{id})$ e $G = \ker
        p$.
  \item Um endomorfismo $s$ é uma simetria se e somente se $s \circ s
        = \mathrm{id}$;
        então $E = \ker(s - \mathrm{id}) \oplus
        \ker(s + \mathrm{id})$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Uma \omterm{def:b1:linmaps:projection}{projeção} satisfaz $p(f + g) = f$ e $p(f) = f$: $p^2 = p$. Reciprocamente,
seja $p^2 = p$; ponha $F = \operatorname{im} p$, $G = \ker
p$. Todo $x$ se escreve $x = p(x) + (x - p(x))$ com
$p(x) \in F$ e $p\bigl(x - p(x)\bigr) = p(x) - p^2(x) = 0$: $E = F + G$. Se $y \in
F \cap G$: $y = p(z)$ e $p(y) = 0$, logo $y = p(z) = p^2(z) = p(y) =
0$: \omterm{def:b1:vspaces:sum}{soma
direta}, e $p$ é a \omterm{def:b1:linmaps:projection}{projeção} sobre $F$ ao longo de $G$. Por fim, em
$F$: $y = p(z)$ dá $p(y) = y$, logo $F \subseteq
\ker(p - \mathrm{id})$, e reciprocamente $p(y) = y$ põe $y$
na \omterm{def:b1:linmaps:kerim}{imagem}.

(2) A correspondência $s = 2p - \mathrm{id}$, $p = \frac{s +
\mathrm{id}}2$ é uma bijeção entre
endomorfismos, e sob ela
\[
s^2 = 4p^2 - 4p + \mathrm{id} = \mathrm{id}
\iff 4p^2 = 4p \iff p^2 = p :
\]
as simetrias correspondem exatamente às projeções. Traduzindo os
\omterm{def:b1:vspaces:subspace}{subespaços}: $s(x) = x \iff p(x) = x$, logo $\ker(s - \mathrm{id}) =
\operatorname{im} p = F$; e $s(x) = -x \iff 2p(x) = 0 \iff x
\in \ker p = G$, logo $\ker(s + \mathrm{id}) = G$. A \omterm{def:b1:vspaces:sum}{soma direta}
$E = F \oplus G$ do ponto (1) torna-se a decomposição anunciada nos vetores
fixos e nos vetores invertidos de $s$.
\end{proof}

\begin{example}[Uma projeção e a sua simetria, explicitamente]
\label{ex:b1:linmaps:projexplicit}
Em $\R^2$, projete sobre $F = \operatorname{Vect}(1,1)$ ao longo de $G =
\operatorname{Vect}(0,1)$. Decomponha $(x, y) = a(1,1) + b(0,1)$:
a primeira coordenada dá $a = x$, e a segunda $b = y - x$. Logo
\[
p(x, y) = (x, x),
\qquad
s(x, y) = 2p(x,y) - (x,y) = (x,\ 2x - y).
\]
Confira a álgebra: $p(p(x,y)) = p(x,x) = (x,x)$, e $s(s(x,y)) =
s(x, 2x - y) = (x, 2x - (2x - y)) = (x, y)$. Geometricamente, $s$ é a
``reflexão oblíqua'' em torno da reta $y = x$ na direção vertical:
fixa $F$ ponto a ponto e inverte $G$. Se tivéssemos projetado sobre
o mesmo $F$ ao longo de $G' =
\operatorname{Vect}(1,-1)$, a fórmula mudaria para $p'(x,y) = \bigl(\frac{x+y}2, \frac{x+y}2\bigr)$: uma
\omterm{def:b1:linmaps:projection}{projeção} é determinada pela sua \omterm{def:b1:linmaps:kerim}{imagem} \emph{e} pelo seu \omterm{def:b1:linmaps:kerim}{núcleo},
nunca pela \omterm{def:b1:linmaps:kerim}{imagem} sozinha.
\end{example}

\section{Formas lineares e hiperplanos}

\begin{definition}\label{def:b1:linmaps:forms}
Uma \emph{forma linear}\index{forma linear} em $E$ é uma \omterm{def:b1:linmaps:def}{aplicação
linear} $\varphi \colon E \to K$. Um \emph{hiperplano}\index{hiperplano} de $E$ ($\dim E = n$)
é um \omterm{def:b1:vspaces:subspace}{subespaço} de dimensão $n - 1$.
\end{definition}

\begin{example}[Uma forma de avaliação e o seu hiperplano]
\label{ex:b1:linmaps:evaluationform}
Em $\R_2[X]$, a avaliação $\varphi(P) = P(2)$ é uma \omterm{def:b1:linmaps:forms}{forma linear}, não nula
($\varphi(1) = 1$). O seu \omterm{def:b1:linmaps:kerim}{núcleo} é o \omterm{def:b1:linmaps:forms}{hiperplano} dos \omterm{def:b1:poly:def}{polinômios} que se anulam
em $2$, isto é (teorema do fator, \cref{thm:b1:poly:factor}), os múltiplos de $X - 2$
dentro de $\R_2[X]$:
\[
\ker\varphi = \operatorname{Vect}\bigl(X - 2,\ X(X - 2)\bigr),
\qquad \dim = 2 .
\]
Em \omterm{prop:b1:vspaces:coordinates}{coordenadas} na \omterm{def:b1:vspaces:free}{base} $(1, X, X^2)$, $\varphi(a + bX + cX^2) = a +
2b + 4c$: toda \omterm{def:b1:linmaps:forms}{forma linear} num espaço
de \omterm{def:b1:findim:def}{dimensão finita} é, uma vez fixada uma \omterm{def:b1:vspaces:free}{base}, uma expressão \omterm{def:b1:linmaps:def}{linear}
fixa nas \omterm{prop:b1:vspaces:coordinates}{coordenadas} --- as formas são ``vetores linha'', como o
\cref{ch:b1:matrices} tornará literal, e a linha de coeficientes aqui, $(1, 2, 4)$, é uma
linha de Vandermonde: as formas de avaliação são o modo como a
teoria da interpolação do problema de fim de semana do \cref{ch:b1:det} entra
na álgebra linear.
\end{example}

\begin{theorem}\label{thm:b1:linmaps:hyperplanes}
Os \omterm{def:b1:linmaps:forms}{hiperplanos} de $E$ são exatamente os \omterm{def:b1:linmaps:kerim}{núcleos} das \omterm{def:b1:linmaps:forms}{formas lineares}
não nulas. Duas formas não nulas têm o mesmo \omterm{def:b1:linmaps:kerim}{núcleo} se e somente se
são proporcionais.
\end{theorem}

\begin{proof}
Se $\varphi \neq 0$: $\operatorname{rk}\varphi = 1$ (a \omterm{def:b1:linmaps:kerim}{imagem} é um \omterm{def:b1:vspaces:subspace}{subespaço} não nulo de $K$), logo
$\dim\ker\varphi = n - 1$: um \omterm{def:b1:linmaps:forms}{hiperplano}. Reciprocamente, seja $H$ um \omterm{def:b1:linmaps:forms}{hiperplano}, $(e_1, \dots, e_{n-1})$
uma \omterm{def:b1:vspaces:free}{base} de $H$ completada por $e_n$: a forma ``última coordenada''
tem \omterm{def:b1:linmaps:kerim}{núcleo} $H$.

Formas proporcionais partilham o \omterm{def:b1:linmaps:kerim}{núcleo}. Reciprocamente, suponha
$\ker\varphi = \ker\psi = H$ e escolha $a \notin H$: todo $x$ se escreve $x = h + \lambda a$ (pois $E = H \oplus Ka$), e
\[
\varphi(x) = \lambda \varphi(a), \qquad \psi(x) = \lambda\psi(a):
\]
logo $\varphi = \frac{\varphi(a)}{\psi(a)}\,\psi$.
\end{proof}

\begin{example}\label{ex:b1:linmaps:hyperplane}
Em $K^n$, um \omterm{def:b1:linmaps:forms}{hiperplano} é um \omterm{def:b1:logic:sets}{conjunto} de soluções $\{a_1 x_1 + \dots + a_n
x_n = 0\}$ com nem
todos os $a_i$ nulos --- a familiar equação de um plano pela origem
em $\R^3$. Em espaços de funções, formas de avaliação como $P \mapsto P(1)$ ou
$f \mapsto \int_0^1 f$ definem \omterm{def:b1:linmaps:forms}{hiperplanos} de $\R_n[X]$ e de $C(\intcc{0}{1})$ (cf.\ o \cref{exo:b1:findim:6}).
\end{example}

\begin{example}[Um hiperplano, trabalhado de três maneiras]
\label{ex:b1:linmaps:hyperplanerun}
Tome $\varphi(x, y, z) = x - 2y + 3z$ em $\R^3$ e $H =
\ker\varphi$. \emph{\omterm{def:b1:vspaces:free}{Base}}: resolva $x = 2y - 3z$:
\[
(2y - 3z,\ y,\ z) = y\,(2, 1, 0) + z\,(-3, 0, 1),
\]
dois vetores \omterm{def:b1:vspaces:free}{livres}: $\dim H = 2$, um \omterm{def:b1:linmaps:forms}{hiperplano}, como o \cref{thm:b1:linmaps:hyperplanes} prevê a
partir de $\varphi \neq
0$. \emph{Reta \omterm{def:b1:vspaces:sum}{suplementar}}: qualquer vetor fora de
$H$ gera uma, por exemplo $a = (1, 0, 0)$ ($\varphi(a) = 1 \neq 0$); a decomposição de um $v$
arbitrário é explícita:
\[
v = \underbrace{\bigl(v - \varphi(v)\,a\bigr)}_{\in\,H}
+ \underbrace{\varphi(v)\,a}_{\in\,\operatorname{Vect}(a)},
\]
pois $\varphi\bigl(v - \varphi(v)a\bigr) = \varphi(v) -
\varphi(v)\varphi(a) = 0$. \emph{Proporcionalidade}: se $\psi(x,y,z)
= -2x + 4y - 6z$, então $\psi = -2\varphi$ e ambas
têm \omterm{def:b1:linmaps:kerim}{núcleo} $H$; reciprocamente, toda forma que se anula em $H$ é um
múltiplo de $\varphi$ (o \cref{exo:b1:linmaps:8}) --- a equação de um \omterm{def:b1:linmaps:forms}{hiperplano} é
única a menos de escala, fato usado constantemente para planos em
geometria.
\end{example}

\begin{remark}[Armadilhas comuns]
\label{rem:b1:linmaps:pitfalls}
\emph{\omterm{def:b1:linmaps:kerim}{Núcleo} e \omterm{def:b1:linmaps:kerim}{imagem} vivem em espaços diferentes}: $\ker u
\subseteq E$, $\operatorname{im} u \subseteq F$;
a soma $\ker u
+ \operatorname{im} u$ só faz sentido para endomorfismos, e mesmo então não
precisa ser direta ($u(x, y) = (y, 0)$ tem $\ker u
= \operatorname{im} u$; o \cref{exo:b1:linmaps:7} caracteriza quando a
soma é direta).
\emph{$u^2 = 0$ não significa $u = 0$}: o mesmo $u(x,y) =
(y, 0)$ tem quadrado
nulo sem se anular --- o que $u^2 = 0$ realmente diz é $\operatorname{im} u \subseteq \ker u$
(o \cref{exo:b1:linmaps:5}).
\emph{\omterm{def:b1:logic:inj}{Injetiva} $\iff$ \omterm{def:b1:logic:inj}{sobrejetiva} exige dimensões finitas iguais}:
em $K[X]$, a \omterm{def:b1:derivative:def}{derivada} é \omterm{def:b1:logic:inj}{sobrejetiva} e não \omterm{def:b1:logic:inj}{injetiva}, e $P \mapsto XP$ é
\omterm{def:b1:logic:inj}{injetiva} e não \omterm{def:b1:logic:inj}{sobrejetiva} (o \cref{cor:b1:linmaps:samedim}); e entre espaços de dimensões
\emph{diferentes} uma das implicações é simplesmente impossível
($\operatorname{rk} u \leq \min(\dim E, \dim F)$).
\emph{Prescrever imagens funciona numa \omterm{def:b1:vspaces:free}{base}, não numa família
qualquer}: exigir $u(1, 0) = a$, $u(0, 1) = b$, $u(1, 1) = c$ superdetermina $u$ a não
ser que $c = a + b$; uma \omterm{def:b1:linmaps:def}{aplicação linear} é \omterm{def:b1:vspaces:free}{livre} numa \omterm{def:b1:vspaces:free}{base} e escrava em
todo o resto.
\emph{O \omterm{def:b1:linmaps:rank}{posto} não se preserva por composição}: ele só pode cair,
$\operatorname{rk}(vu) \leq \min(\operatorname{rk} u,
\operatorname{rk} v)$ (o \cref{exo:b1:linmaps:4}), com a perda exata medida no \cref{exo:b1:linmaps:12}.
\end{remark}

\begin{remark}[Para onde vão estas aplicações]
\label{rem:b1:linmaps:whereused}
As \omterm{def:b1:linmaps:def}{aplicações lineares} estão prestes a tornar-se \emph{matrizes}:
uma vez fixadas as \omterm{def:b1:vspaces:free}{bases}, o \cref{ch:b1:matrices} codifica cada $u \in \mathcal{L}(E,
F)$ por um quadro
retangular, e a composição torna-se o produto de matrizes --- o
teorema do \omterm{def:b1:linmaps:kerim}{núcleo} e da \omterm{def:b1:linmaps:kerim}{imagem} conduz então a teoria dos sistemas
\omterm{def:b1:linmaps:def}{lineares} no \cref{ch:b1:det}. As projeções voltam no \cref{ch:b1:euclid} no seu caso
particular mais útil, a \omterm{def:b1:linmaps:projection}{projeção} \emph{ortogonal}, em que o \omterm{def:b1:linmaps:kerim}{núcleo}
é escolhido perpendicular à \omterm{def:b1:linmaps:kerim}{imagem}. O problema de fim de semana
abaixo leva a álgebra dos projetores tão longe quanto as
ferramentas do primeiro ano alcançam, até o lema de Fitting; o
volume do segundo ano de graduação vai além, com o traço e com a
teoria dos autovalores, para a qual os projetores sobre \omterm{def:b1:vspaces:subspace}{subespaços}
estáveis são os blocos de construção básicos.
\end{remark}

\begin{remark}[Perspectivas dentro do Livro 3: núcleo e imagem mais três
vezes]
\label{rem:b1:linmaps:perspectives}
A lei de conservação $\dim E = \dim\ker u + \operatorname{rk} u$ será relida três vezes antes do fim do
volume. No \cref{ch:b1:det} ela torna-se a forma dos \omterm{def:b1:logic:sets}{conjuntos} de soluções: um
sistema compatível com $p$ incógnitas e \omterm{def:b1:linmaps:rank}{posto} $r$ tem um \omterm{def:b1:logic:sets}{conjunto}
de soluções de dimensão $p - r$ --- a dimensão do \omterm{def:b1:linmaps:kerim}{núcleo}
disfarçada. No \cref{ch:b1:euclid} ela se \omterm{def:b1:arith:divides}{divide} ortogonalmente, $\dim F + \dim
F^\perp = \dim E$, e
alimenta todo cálculo de distância. No problema de fim de semana do
\cref{ch:b1:multivar}, ela é a contadora dos mínimos quadrados: $n$ observações,
$2$ parâmetros ajustados, $n - 2$ dimensões de resíduo, e a
identidade pitagórica $\norm b^2 = \norm p^2 + \norm{b - p}^2$ é a sombra euclidiana do teorema do
\omterm{def:b1:linmaps:kerim}{núcleo} e da \omterm{def:b1:linmaps:kerim}{imagem}. Um teorema, quatro fantasias.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:1}
Quais destas aplicações são \omterm{def:b1:linmaps:def}{lineares}?
\begin{enumerate}
  \item $\R^2 \to \R^2$, $(x, y) \mapsto (x + y, x - 2y)$;
  \item $\R^2 \to \R$, $(x, y) \mapsto xy$;
  \item $\R[X] \to \R[X]$, $P \mapsto P' + XP$;
  \item $\mathcal{F}(\R,\R) \to \R$, $f \mapsto f(3)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:2}
Seja $u \colon \R^3 \to \R^3$, $(x,y,z) \mapsto (x + y - z,\; 2x + y
+ z,\; 3x + 2y)$. Determine $\ker u$ (\omterm{def:b1:vspaces:free}{base}, dimensão), $\operatorname{rk} u$, e
uma \omterm{def:b1:vspaces:free}{base} de $\operatorname{im} u$. $u$ é \omterm{def:b1:logic:inj}{injetiva}? \omterm{def:b1:logic:inj}{sobrejetiva}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:3}
Seja $u \colon \R_n[X] \to \R_n[X]$, $P \mapsto P - P'$. Demonstre que $u$ é um isomorfismo: uma vez via
$\ker u$, e outra exibindo a inversa \emph{(considere $P + P' + P'' + \dots$)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:4}
Sejam $u \in \mathcal{L}(E, F)$ e $v \in \mathcal{L}(F, G)$, com os espaços de \omterm{def:b1:findim:def}{dimensão finita}. Demonstre:
\[
\operatorname{rk}(v \circ u) \leq
\min\bigl(\operatorname{rk} u,\ \operatorname{rk} v\bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:5}
Seja $u$ um endomorfismo de $E$ (de \omterm{def:b1:findim:def}{dimensão finita}) com $u^2 = 0$.
Demonstre que $\operatorname{im} u \subseteq \ker u$, logo $\operatorname{rk} u \leq \frac{\dim E}{2}$. Para $E = \R^2$, dê um exemplo com
igualdade.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:6}
Sejam $p, q$ projeções de $E$ com $p \circ q = q \circ p$. Demonstre que $p \circ q$ é uma
\omterm{def:b1:linmaps:projection}{projeção}, com
\[
\operatorname{im}(pq) = \operatorname{im} p \cap \operatorname{im}
q ,
\qquad
\ker (pq) = \ker p + \ker q .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:7}
Seja $u \in \mathcal{L}(E)$, com $E$ de \omterm{def:b1:findim:def}{dimensão finita}. Demonstre a equivalência
de:
\begin{enumerate}
  \item $E = \ker u \oplus \operatorname{im} u$;
  \item $\ker u = \ker u^2$;
  \item $\operatorname{im} u = \operatorname{im} u^2$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:8}
Sejam $\varphi, \psi$ \omterm{def:b1:linmaps:forms}{formas lineares} em $E$ com $\ker\varphi
\subseteq \ker\psi$. Demonstre que $\psi = \lambda\varphi$
para algum $\lambda \in K$ (incluindo os casos degenerados).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:linmaps:9}
Seja $u \in \mathcal{L}(E)$ com $\dim E = n$, e suponha $u^n = 0$ mas $u^{n-1} \neq 0$ (um
endomorfismo \emph{maximamente nilpotente}). Escolha $x$ com $u^{n-1}(x) \neq 0$;
demonstre que $\bigl(x, u(x), \dots,
u^{n-1}(x)\bigr)$ é uma \omterm{def:b1:vspaces:free}{base} de $E$. \emph{(Aplique potências de
$u$ a uma combinação nula, começando por $u^{n-1}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:linmaps:10}
Seja $f \in \mathcal{L}(\R^n)$ com $f \circ f = -\mathrm{id}$.
\begin{enumerate}
  \item Demonstre que $f$ é um isomorfismo e que nenhum $x \neq 0$
        satisfaz $f(x) = \lambda x$ com $\lambda \in \R$.
  \item Demonstre que $n$ é par. \emph{Sugestão: escolha $x_1 \neq 0$;
        mostre que $\operatorname{Vect}(x_1, f(x_1))$ é um plano estável por $f$; escolha $x_2$
        fora dele e itere, demonstrando que $\bigl(x_1, f(x_1), x_2, f(x_2), \dots\bigr)$ continua \omterm{def:b1:vspaces:free}{livre}.}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:11}
Sejam $u, v \in \mathcal{L}(E, F)$, com os espaços de \omterm{def:b1:findim:def}{dimensão finita}. Demonstre a
estimativa bilateral
\[
\abs{\operatorname{rk} u - \operatorname{rk} v}
\;\leq\; \operatorname{rk}(u + v)
\;\leq\; \operatorname{rk} u + \operatorname{rk} v .
\]
\emph{(Para a \omterm{def:b1:reals:bounds}{cota superior}, compare $\operatorname{im}(u+v)$ com $\operatorname{im} u + \operatorname{im} v$; para a
inferior, aplique a \omterm{def:b1:reals:bounds}{cota superior} com esperteza.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:linmaps:12}
(Desigualdade de Frobenius) Sejam $u \in \mathcal{L}(E, F)$, $w \in
\mathcal{L}(F, G)$ e $v \in \mathcal{L}(G, H)$, com todos
os espaços de \omterm{def:b1:findim:def}{dimensão finita}. Demonstre a fórmula exata
\[
\operatorname{rk}(v \circ w) = \operatorname{rk} w -
\dim\bigl(\ker v \cap \operatorname{im} w\bigr),
\]
e deduza a desigualdade de Frobenius
\[
\operatorname{rk}(v \circ w) + \operatorname{rk}(w \circ u)
\;\leq\; \operatorname{rk} w + \operatorname{rk}(v \circ w \circ
u) .
\]
Verifique que o caso $w = \mathrm{id}_F$ é a desigualdade de Sylvester,
demonstrada em forma matricial no \cref{exo:b1:matrices:10}.
\end{exercise}

\section{Problema: cálculo de projetores e o lema de Fitting}

\begin{problem}\label{pb:b1:linmaps:1}
As projeções são os endomorfismos que as \omterm{def:b1:vspaces:sum}{somas diretas} produzem, e
reciprocamente: toda identidade $E = F_1 \oplus \dots \oplus F_k$ é, secretamente, uma família
de projetores cuja soma é a identidade. Este problema desenvolve
esse dicionário --- a álgebra de um projetor, de dois, de $k$ --- e
depois aplica as mesmas ideias de estabilização a um endomorfismo
arbitrário e demonstra o \emph{lema de Fitting}: todo endomorfismo
de um espaço de \omterm{def:b1:findim:def}{dimensão finita} se decompõe numa parte nilpotente e
numa parte invertível. Ao longo do texto, $E$ é um \omterm{def:b1:vspaces:def}{espaço vetorial}
sobre $K$ de dimensão $n$, e \emph{projetor} significa $p \in \mathcal{L}(E)$ com
$p^2 = p$ (o \cref{thm:b1:linmaps:projchar}).
\index{projeção}\index{lema de Fitting}\index{nilpotente}

\medskip
\noindent\textbf{Parte I --- A álgebra em torno de um projetor.}
Seja $p$ um projetor, $p \neq 0$, $p \neq \mathrm{id}$.
\begin{enumerate}
  \item Mostre que $\mathrm{id} - p$ é um projetor e identifique $\operatorname{im}(\mathrm{id} - p)$ e
        $\ker(\mathrm{id} - p)$.
  \item Calcule $(\lambda\,\mathrm{id} + \mu\,p)^2$ e determine todos os pares $(\lambda, \mu) \in K^2$ para os
        quais $\lambda\,\mathrm{id} + \mu\,p$ é um projetor.
  \item Mostre que o plano $\operatorname{Vect}(\mathrm{id}, p)$ de $\mathcal{L}(E)$ é estável por
        composição, e que para todo \omterm{def:b1:poly:def}{polinômio} $Q \in K[X]$,
        \[
        Q(p) = Q(0)\,\mathrm{id} + \bigl(Q(1) -
        Q(0)\bigr)\,p .
        \]
  \item Determine para quais $(\lambda, \mu)$ a \omterm{def:b1:logic:map}{aplicação} $\lambda\,\mathrm{id} + \mu\,p$ é invertível,
        e dê a sua inversa na forma $\alpha\,\mathrm{id} + \beta\,p$. Interprete a resposta
        através da ação de $\lambda\,\mathrm{id} + \mu\,p$ sobre $\operatorname{im} p$ e sobre $\ker p$.
  \item Seja $p'$ outro projetor com a \emph{mesma \omterm{def:b1:linmaps:kerim}{imagem}} $\operatorname{im} p' = \operatorname{im} p$.
        Mostre que $p\,p' = p'$ e $p'\,p = p$. O que dizem estas identidades
        sobre compor projeções sobre o mesmo \omterm{def:b1:vspaces:subspace}{subespaço} ao longo de
        \omterm{def:b1:linmaps:kerim}{núcleos} diferentes?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Dois projetores.} Sejam $p, q$
projetores de $E$; suponha que a característica não é $2$
(verdade para $K = \R, \C$).
\begin{enumerate}[resume]
  \item Suponha que $p + q$ é um projetor. Desenvolvendo $(p + q)^2$,
        mostre que $pq + qp = 0$; compondo com $p$ à esquerda e depois à
        direita, deduza $pq = qp$, e conclua $pq = qp =
        0$.
  \item Reciprocamente, suponha $pq = qp = 0$. Mostre que $p + q$ é um
        projetor, com
        \[
        \operatorname{im}(p + q) = \operatorname{im} p \oplus
        \operatorname{im} q,
        \qquad
        \ker(p + q) = \ker p \cap \ker q .
        \]
  \item Mostre que $p - q$ é um projetor se e somente se $pq = qp
        = q$.
        \emph{(Aplique as questões 6--7 a $\mathrm{id} - p$ e a $q$.)}
  \item Mostre o significado geométrico de $pq = qp = q$: vale se e somente
        se $\operatorname{im} q \subseteq
        \operatorname{im} p$ e $\ker p \subseteq \ker q$. (Escreve-se então $q \leq p$: ``$q$ projeta
        sobre menos, ao longo de mais''.)
  \item Agora suponha que $p$ e $q$ comutam. Lembre do \cref{exo:b1:linmaps:6} que
        $pq$ é o projetor sobre $\operatorname{im} p \cap \operatorname{im} q$ ao longo de $\ker p + \ker q$. Mostre que
        $r = p + q - pq$ é um projetor com
        \[
        \operatorname{im} r = \operatorname{im} p +
        \operatorname{im} q,
        \qquad
        \ker r = \ker p \cap \ker q .
        \]
        \emph{(Considere $\mathrm{id} - r = (\mathrm{id} -
        p)(\mathrm{id} - q)$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Decomposições da identidade.}
\begin{enumerate}[resume]
  \item Seja $E = F_1 \oplus \dots \oplus F_k$ e, para $x = x_1 +
        \dots + x_k$ (decomposição única, $x_i \in F_i$),
        ponha $p_i(x) = x_i$. Mostre que cada $p_i$ é um projetor, que
        $p_i p_j = 0$ para $i \neq j$, e que $p_1 + \dots + p_k
        = \mathrm{id}$; identifique $\operatorname{im} p_i$ e $\ker p_i$.
  \item Reciprocamente, sejam $p_1, \dots, p_k \in \mathcal{L}(E)$ satisfazendo $p_1 + \dots + p_k = \mathrm{id}$ e $p_i p_j =
        0$
        para todos $i \neq j$. Mostre que cada $p_i$ é um projetor e
        que $E = \operatorname{im} p_1 \oplus
        \dots \oplus \operatorname{im} p_k$.
  \item Dois projetores com $p + q = \mathrm{id}$: mostre que $pq
        = qp = 0$ vale
        automaticamente.
  \item Três projetores com $p + q + r = \mathrm{id}$: mostre que $p + q$ é um
        projetor, e deduza da questão 6 que \emph{todos} os
        produtos dois a dois se anulam --- logo $E =
        \operatorname{im} p \oplus \operatorname{im} q \oplus
        \operatorname{im} r$, sem
        hipótese alguma sobre os produtos.
  \item Para $k$ projetores com $p_1 + \dots + p_k =
        \mathrm{id}$: mostre primeiro que, para
        \omterm{def:b1:vspaces:subspace}{subespaços} quaisquer, $\dim
        (F_1 + \dots + F_k) \leq \dim F_1 + \dots + \dim F_k$, com igualdade se e somente se
        a soma é direta; depois mostre que $E = \operatorname{im} p_1 + \dots +
        \operatorname{im} p_k$, e demonstre que
        \emph{se} além disso $\sum_i \operatorname{rk} p_i \leq n$, a soma é direta e $p_i p_j = 0$ para
        $i \neq j$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- \omterm{def:b1:linmaps:kerim}{Núcleos} iterados: o lema de
Fitting.} Seja $u \in \mathcal{L}(E)$, $\dim E = n$.
\begin{enumerate}[resume]
  \item Mostre as duas cadeias, válidas para todo $k \geq 0$:
        \[
        \ker u^k \subseteq \ker u^{k+1},
        \qquad
        \operatorname{im} u^{k+1} \subseteq \operatorname{im}
        u^k .
        \]
  \item Mostre que se $\ker u^{r} = \ker u^{r+1}$ para algum $r$, então $\ker u^{k} = \ker u^{r}$ para todo
        $k \geq r$; enuncie e demonstre a estabilização análoga para as
        imagens.
  \item Deduza que existe um menor inteiro $r$ com $\ker
        u^{r} = \ker u^{r+1}$, que
        $r \leq n$, e que as imagens estabilizam no mesmo $r$.
  \item (Lema de Fitting) Demonstre que
        \[
        E \;=\; \ker u^{r} \,\oplus\, \operatorname{im} u^{r} .
        \]
  \item Mostre que os dois \omterm{def:b1:vspaces:subspace}{subespaços} são estáveis por $u$, que a
        restrição de $u$ a $\ker u^{r}$ é nilpotente, e que a restrição de
        $u$ a $\operatorname{im} u^{r}$ é um isomorfismo de $\operatorname{im} u^{r}$: todo endomorfismo é,
        sobre uma \omterm{def:b1:vspaces:sum}{soma direta} canônica, ``nilpotente mais
        invertível''.
  \item Seja $\pi$ o projetor sobre $\ker u^{r}$ ao longo de $\operatorname{im} u^{r}$. Mostre
        que $\pi \circ u = u
        \circ \pi$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Um caso trabalhado, e síntese.}
\begin{enumerate}[resume]
  \item Seja $u(x, y, z) = (y, 0, z)$ em $\R^3$. Calcule $u^2$ e $u^3$, determine o
        índice de estabilização $r$, os \omterm{def:b1:vspaces:subspace}{subespaços} $\ker u^{r}$ e $\operatorname{im} u^{r}$, o
        projetor de Fitting $\pi$, e verifique nas fórmulas que
        $\pi u = u\pi$ e que $u$ é nilpotente num dos fatores e \omterm{def:b1:logic:inj}{bijetivo} no
        outro.
  \item Mostre as equivalências: $u$ nilpotente $\iff$ $\ker u^{r} =
        E$
        $\iff$ $\pi = \mathrm{id}$; e deduza que um endomorfismo nilpotente de um
        espaço de dimensão $n$ satisfaz sempre $u^{n} = 0$ (o índice de
        nilpotência nunca ultrapassa a dimensão).
  \item (Unicidade) Suponha $E = A \oplus B$ com $A, B$ estáveis por $u$, a
        restrição $u|_A$ nilpotente e $u|_B$ \omterm{def:b1:logic:inj}{bijetiva}. Demonstre
        que $A = \ker u^{r}$ e $B =
        \operatorname{im} u^{r}$: a decomposição de Fitting é única.
  \item Síntese, em quatro frases: que dicionário a Parte III
        estabelece entre \omterm{def:b1:vspaces:sum}{somas diretas} e famílias de projetores;
        por que a questão 14 não precisou de hipóteses sobre
        produtos enquanto a questão 15 precisou de uma hipótese
        sobre \omterm{def:b1:linmaps:rank}{postos} (e qual ferramenta do segundo ano, o traço, a
        remove); em que sentido o lema de Fitting é a versão
        estabilizada do \cref{exo:b1:linmaps:7}; e no que se tornam os dois fatores
        de Fitting na teoria dos autovalores do volume do segundo
        ano de graduação. Nomeie o teorema demonstrado na Parte IV.
\end{enumerate}
\end{problem}
