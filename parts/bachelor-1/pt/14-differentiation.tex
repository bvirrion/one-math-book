\chapter{Derivação}\label{ch:b1:derivative}

As \omterm{def:b1:derivative:def}{derivadas} foram calculadas ao longo de todo o volume do ensino
médio; o que faltava é a cadeia de teoremas que transforma cálculo em
informação sobre as funções: o teorema de Rolle, o teorema do valor
médio e as suas consequências --- critérios de monotonicidade,
estimativas de Lipschitz, \omterm{def:b1:derivative:convex}{convexidade}. Tudo neste capítulo diz respeito
a funções definidas num \omterm{prop:b1:reals:intervals}{intervalo} $I$.

\section{A derivada}

\begin{definition}\label{def:b1:derivative:def}
$f \colon I \to \R$ é \emph{derivável em $x_0 \in
I$}\index{derivada} quando o quociente de diferenças
$\frac{f(x) - f(x_0)}{x - x_0}$ tem limite (finito) quando $x \to x_0$;
o limite é escrito $f'(x_0)$. Equivalentemente:
\[
f(x_0 + h) = f(x_0) + f'(x_0)\,h + h\,\varepsilon(h),
\qquad \varepsilon(h) \xrightarrow[h \to 0]{} 0 ,
\]
admitindo então o gráfico a reta tangente $y = f(x_0) + f'(x_0)(x -
x_0)$. A derivabilidade em $x_0$ implica a \omterm{def:b1:continuity:continuous}{continuidade} em $x_0$ (leia
a expressão). $f$ é derivável em $I$ quando o é em todo ponto;
$f$ é de classe $C^1$ quando, além disso, $f'$ é \omterm{def:b1:continuity:continuous}{contínua}, e de
classe $C^k$\index{funcao Ck@função $C^k$} quando $f$ pode ser
derivada $k$ vezes com $f^{(k)}$ \omterm{def:b1:continuity:continuous}{contínua}.
\end{definition}

\begin{example}\label{ex:b1:derivative:notc1}
A recíproca de ``\omterm{def:b1:derivative:def}{derivável} $\Rightarrow$ \omterm{def:b1:continuity:continuous}{contínua}'' falha:
$\abs{\,\cdot\,}$ em $0$. Mais surpreendentemente, \omterm{def:b1:derivative:def}{derivável} não
implica $C^1$: a função $f(x) = x^2 \sin\frac 1x$ ($f(0) = 0$) é
\omterm{def:b1:derivative:def}{derivável} em toda parte, com $f'(0) = 0$, mas $f'(x) = 2x
\sin\frac1x - \cos\frac 1x$ não tem limite em $0$
(\cref{exo:b1:derivative:2}).
\end{example}

\begin{example}[Derivável em exatamente um ponto]\label{ex:b1:derivative:onepoint}
Sejam $f(x) = x^2$ para $x \in \Q$ e $f(x) = 0$ para $x \notin
\Q$. Em $0$: $\bigl|\frac{f(h) - 0}{h}\bigr| \leq \abs h \to 0$,
de modo que $f$ é \omterm{def:b1:derivative:def}{derivável} em $0$, com $f'(0) = 0$. Em qualquer $x_0
\neq 0$, $f$ nem sequer é \omterm{def:b1:continuity:continuous}{contínua}: sequências racionais e
irracionais que convergem a $x_0$ mandam $f$ a $x_0^2 \neq 0$ e a
$0$, respectivamente (\omterm{def:b1:topology:dense}{densidade}, \cref{thm:b1:reals:density}). Assim, a
\omterm{def:b1:derivative:def}{derivabilidade} é uma noção genuinamente \emph{pontual}: ela pode
valer num ponto de $\R$ e em nenhum outro. A moral para
a prática: enunciados como o critério de monotonicidade ou Rolle
exigem a \omterm{def:b1:derivative:def}{derivada} \emph{num \omterm{prop:b1:reals:intervals}{intervalo}} --- possuir
$f'(x_0)$ em pontos isolados, por mais numerosos que sejam, não sustenta
conclusão global alguma.
\end{example}

\begin{theorem}[Operações]\label{thm:b1:derivative:operations}
Se $f, g$ são deriváveis em $x_0$ (e onde as fórmulas fazem
sentido):
\[
(f + g)' = f' + g', \qquad
(fg)' = f'g + fg', \qquad
\Bigl(\frac fg\Bigr)' = \frac{f'g - fg'}{g^2},
\]
e, se $g$ é \omterm{def:b1:derivative:def}{derivável} em $f(x_0)$:
$\;(g \circ f)'(x_0) = g'\bigl(f(x_0)\bigr)\, f'(x_0)$ (regra da
cadeia\index{regra da cadeia}).
\end{theorem}

\begin{proof}
Soma: imediata. Produto: escreva
\[
f(x)g(x) - f(x_0)g(x_0)
= \bigl(f(x) - f(x_0)\bigr) g(x) + f(x_0)\bigl(g(x) - g(x_0)\bigr),
\]
divida por $x - x_0$ e faça $x \to x_0$ ($g$ é \omterm{def:b1:continuity:continuous}{contínua} em $x_0$).
Quociente: trate $\frac 1g$ via $\frac{1/g(x) - 1/g(x_0)}{x - x_0} =
\frac{-1}{g(x)g(x_0)}\cdot\frac{g(x) - g(x_0)}{x - x_0}$ e aplique depois
a regra do produto. Regra da cadeia: com $y_0 = f(x_0)$, defina
$\theta(y) = \frac{g(y) - g(y_0)}{y - y_0}$ para $y \neq y_0$ e
$\theta(y_0) = g'(y_0)$: $\theta$ é \omterm{def:b1:continuity:continuous}{contínua} em $y_0$ e, para
$x \neq x_0$,
\[
\frac{g(f(x)) - g(f(x_0))}{x - x_0}
= \theta\bigl(f(x)\bigr)\cdot \frac{f(x) - f(x_0)}{x - x_0}
\longrightarrow g'(y_0)\, f'(x_0),
\]
o primeiro fator pela composição de limites (esse artifício trata
limpamente o caso $f(x) = f(x_0)$, em que o ingênuo ``multiplicar e dividir
por $f(x) - f(x_0)$'' se quebra).
\end{proof}

\begin{theorem}[Derivada de uma função inversa]\label{thm:b1:derivative:inverse}
Seja $f$ \omterm{def:b1:continuity:continuous}{contínua} e estritamente monótona em $I$, \omterm{def:b1:derivative:def}{derivável}
em $x_0$ com $f'(x_0) \neq 0$. Então $f^{-1}$
(\cref{thm:b1:continuity:bijection}) é \omterm{def:b1:derivative:def}{derivável} em $y_0 =
f(x_0)$, com
\[
(f^{-1})'(y_0) = \frac{1}{f'(x_0)} = \frac{1}{f'\bigl(f^{-1}(y_0)\bigr)} .
\]
Se $f'(x_0) = 0$, a inversa tem tangente vertical em $y_0$.
\end{theorem}

\begin{proof}
Para $y \to y_0$, ponha $x = f^{-1}(y)$: a \omterm{def:b1:continuity:continuous}{continuidade} de $f^{-1}$ dá $x
\to x_0$, e
\[
\frac{f^{-1}(y) - f^{-1}(y_0)}{y - y_0}
= \frac{x - x_0}{f(x) - f(x_0)}
= \frac{1}{\dfrac{f(x) - f(x_0)}{x - x_0}}
\longrightarrow \frac{1}{f'(x_0)} .
\]
Afirmação da tangente vertical: se $f'(x_0) = 0$, o quociente exibido
é o inverso de uma quantidade que tende a $0$ mantendo
um sinal constante (para $f$ estritamente crescente, $\frac{f(x) -
f(x_0)}{x - x_0} > 0$ para todo $x \neq x_0$): o quociente
de diferenças de $f^{-1}$ tende, portanto, a $+\infty$ (a $-\infty$
para $f$ decrescente). A inversa permanece \omterm{def:b1:continuity:continuous}{contínua}, mas não é
\omterm{def:b1:derivative:def}{derivável} em $y_0$ --- o seu gráfico, reflexão do de $f$
na diagonal, fica vertical exatamente onde o de $f$ corria
horizontal, como $x^{1/3}$ em $0$ ilustra contra $x^3$.
\end{proof}

\begin{example}[Derivadas de inversas, duas vezes]\label{ex:b1:derivative:inverseaction}
O teorema recalcula as \omterm{def:b1:derivative:def}{derivadas} clássicas sem nenhum trabalho
com limites. Para $\ln = \exp^{-1}$: em $y = \eu^x$,
\[
(\ln)'(y) = \frac{1}{\exp'(x)} = \frac{1}{\eu^{x}} = \frac1y ,
\]
válido para todo $y > 0$, pois $\exp' = \exp$ nunca se anula. Para
$\arctan = \tan^{-1}$: em $y = \tan x$,
\[
(\arctan)'(y) = \frac{1}{1 + \tan^2 x} = \frac{1}{1 + y^2} ,
\]
usando $\tan' = 1 + \tan^2 > 0$. A ideia de fechamento: a fórmula
converte conhecimento sobre uma função em conhecimento sobre a sua
inversa ao preço de uma substituição --- e a
substituição ($x = \ln y$, $x = \arctan y$) é exatamente a
afirmação de que as duas variáveis vivem em lados opostos da
bijeção.
\end{example}

\section{Rolle e o teorema do valor médio}

\begin{proposition}[Extremo interior]\label{prop:b1:derivative:fermat}
Se $f$ é \omterm{def:b1:derivative:def}{derivável} num ponto \emph{\omterm{def:b1:topology:closure}{interior}} $x_0$ de $I$ e
tem ali um extremo local, então $f'(x_0) = 0$.
\end{proposition}

\begin{proof}
Digamos que seja um máximo local: existe $r > 0$ com $f(x) \leq f(x_0)$
para $\abs{x - x_0} \leq r$, e a interioridade garante que os dois
lados de $x_0$ estão disponíveis dentro de $I$. Para $0 < h \leq r$, o
quociente $\frac{f(x_0 + h) - f(x_0)}{h}$ tem numerador $\leq 0$
e denominador $> 0$: ele é $\leq 0$, e o seu limite $f'(x_0)$
herda $\leq 0$ (desigualdades largas passam ao limite,
\cref{thm:b1:seq:order}); para $-r \leq h < 0$, o quociente é
$\geq 0$, o que dá $f'(x_0) \geq 0$. Portanto, $f'(x_0) = 0$. (Numa
extremidade, só um sinal está disponível: a conclusão falha ali ---
pense em $x$ em $\intcc{0}{1}$, máximo em $1$ com \omterm{def:b1:derivative:def}{derivada}
$1$.)
\end{proof}

\begin{theorem}[Rolle]\label{thm:b1:derivative:rolle}
Seja $f$ \omterm{def:b1:continuity:continuous}{contínua} em $\intcc{a}{b}$, \omterm{def:b1:derivative:def}{derivável} em
$\intoo{a}{b}$, com $f(a) = f(b)$. Então $f'(c) = 0$ para algum $c \in
\intoo{a}{b}$.\index{teorema de Rolle}
\end{theorem}

\begin{proof}
Pelo teorema de Weierstrass (\cref{thm:b1:continuity:evt}), $f$
atinge o seu máximo e o seu mínimo em $\intcc{a}{b}$. Se ambos são
atingidos nas extremidades, então (como $f(a) = f(b)$) máximo $=$ mínimo e $f$
é constante: qualquer $c$ \omterm{def:b1:topology:closure}{interior} serve. Caso contrário, um extremo é
atingido num ponto \omterm{def:b1:topology:closure}{interior} $c$, e a
\cref{prop:b1:derivative:fermat} dá $f'(c) = 0$.
\end{proof}

\begin{theorem}[Teorema do valor médio]\label{thm:b1:derivative:mvt}
Seja $f$ \omterm{def:b1:continuity:continuous}{contínua} em $\intcc{a}{b}$, \omterm{def:b1:derivative:def}{derivável} em
$\intoo{a}{b}$. Existe $c \in \intoo{a}{b}$ com
\[
f(b) - f(a) = f'(c)\,(b - a) .
\]
\emph{Desigualdade do valor médio:} se, além disso, $m \leq f' \leq M$ em
$\intoo{a}{b}$, então $m(b-a) \leq f(b) - f(a) \leq M(b-a)$; em
particular, $\abs{f'} \leq K$ implica que $f$ é $K$-lipschitziana.
\index{teorema do valor médio}
\end{theorem}

\begin{proof}
Aplique Rolle a $g(x) = f(x) - \frac{f(b) - f(a)}{b - a}(x - a)$: $g$
é \omterm{def:b1:continuity:continuous}{contínua} em $\intcc{a}{b}$, \omterm{def:b1:derivative:def}{derivável} no \omterm{def:b1:topology:closure}{interior}, e $g(a) =
f(a) = g(b)$. No ponto $c$ em que $g'(c) = 0$: $f'(c) =
\frac{f(b)-f(a)}{b-a}$. A desigualdade decorre de estimar $f'(c)$;
o enunciado de Lipschitz a aplica a todo par de pontos.
\end{proof}

\begin{example}[O método de Newton é o de Heron]\label{ex:b1:derivative:newton}
O método de Newton para resolver $f(x) = 0$ substitui a curva pela
sua tangente na estimativa atual $x_n$ e toma a raiz da tangente
como estimativa seguinte:
\[
0 = f(x_n) + f'(x_n)(x_{n+1} - x_n)
\quad\Longrightarrow\quad
x_{n+1} = x_n - \frac{f(x_n)}{f'(x_n)} .
\]
Execute-o em $f(x) = x^2 - 2$:
\[
x_{n+1} = x_n - \frac{x_n^2 - 2}{2x_n}
= \frac{x_n}{2} + \frac{1}{x_n}
= \frac12\Bigl(x_n + \frac{2}{x_n}\Bigr) :
\]
exatamente a iteração de Heron (\cref{ex:b1:seq:heron}), dois
milênios antes. A velocidade quadrática ali observada fica agora
explicada pelo retrato da tangente: perto de uma raiz simples, curva e
tangente diferem por um erro de segunda ordem, de modo que cada passo
aproximadamente eleva o erro ao quadrado --- o enunciado geral decorre
das estimativas de Taylor do \cref{ch:b1:taylor}. A ideia de fechamento:
onde a dicotomia (\cref{ex:b1:continuity:dichotomy}) usa apenas
\omterm{def:b1:continuity:continuous}{continuidade} e ganha um bit por passo, Newton gasta uma
\omterm{def:b1:derivative:def}{derivada} para dobrar o número de algarismos corretos por passo.
\end{example}

\begin{example}[O teorema do valor médio como estimador]\label{ex:b1:derivative:mvtestimate}
Quanto vale $\sqrt{101}$? Aplique o teorema a $f(t) = \sqrt t$
em $\intcc{100}{101}$: para algum $c \in \intoo{100}{101}$,
\[
\sqrt{101} - 10 = \frac{1}{2\sqrt c},
\qquad\text{logo}\qquad
\frac{1}{2\sqrt{101}} < \sqrt{101} - 10 < \frac{1}{20} = 0.05 ,
\]
e, como $\sqrt{101} < 10.05$, a estimativa da esquerda excede
$\frac{1}{20.1} > 0.0497$: assim, $10.0497 < \sqrt{101} < 10.05$
(valor verdadeiro: $10.049875\dots$) --- três casas decimais corretas a
partir de uma avaliação de \omterm{def:b1:derivative:def}{derivada}. Do mesmo modo, $\abs{\sin a - \sin b} \leq
\abs{a - b}$ (estimativa $\abs{\cos}\leq 1$): as estimativas de Lipschitz
usadas desde o \cref{ch:b1:seq} são todas este teorema. A ideia de
fechamento: o teorema do valor médio é uma fórmula de Taylor de ordem
zero --- ele troca um ponto desconhecido $c$ por uma desigualdade firme,
e o \cref{ch:b1:taylor} iterará exatamente essa troca.
\end{example}

\begin{corollary}[Critério de monotonicidade]\label{cor:b1:derivative:monotone}
Seja $f$ \omterm{def:b1:continuity:continuous}{contínua} em $I$ e \omterm{def:b1:derivative:def}{derivável} no \omterm{def:b1:topology:closure}{interior}.
\begin{enumerate}
  \item $f' \geq 0$ no \omterm{def:b1:topology:closure}{interior} $\iff$ $f$ é crescente; $f' = 0$
        $\iff$ $f$ constante.
  \item Se $f' > 0$ salvo em finitos pontos em que ela se anula,
        $f$ é \emph{estritamente} crescente.
\end{enumerate}
\end{corollary}

\begin{proof}
Se $f' \geq 0$: para $x < y$ em $I$, o teorema do valor médio em
$\intcc{x}{y}$ dá $f(y) - f(x) = f'(c)(y - x) \geq 0$. Reciprocamente,
os quocientes de diferenças de uma função crescente são $\geq 0$, de
modo que os seus limites também. O caso constante: aplique o anterior a
$f'$ e a $-f' \geq 0$. Versão estrita: $f$ é crescente; a igualdade $f(x) =
f(y)$ para $x < y$ congelaria $f$ em $\intcc{x}{y}$, forçando $f' =
0$ ali --- infinitos pontos.
\end{proof}

\begin{example}[Derivadas iguais, funções diferentes]\label{ex:b1:derivative:intervalmatters}
Em $\R^* = \intoo{-\infty}{0} \cup \intoo{0}{+\infty}$, tanto
$f(x) = \ln\abs x$ quanto $g(x) = \ln\abs x + \mathbf{1}_{x>0}$
(some $1$ apenas na semirreta direita) satisfazem $f' = g' =
\frac1x$. Elas não diferem por uma constante: o critério
``$f' = 0 \implies f$ constante'' é um enunciado de \emph{\omterm{prop:b1:reals:intervals}{intervalo}}
--- a sua demonstração roda o teorema do valor médio entre dois pontos,
o que exige que todo o segmento que os une esteja no domínio. Em cada
semirreta separadamente, as primitivas de $\frac1x$ são
$\ln\abs x + c$, com uma constante por semirreta, duas
constantes independentes no total. O \Cref{ch:b1:integration}
herda essa letra miúda: ``a'' primitiva de uma função fica bem definida
a menos de uma constante \emph{em cada \omterm{prop:b1:reals:intervals}{intervalo} do seu domínio}, e as
tabelas de primitivas supõem silenciosamente a conexidade.
\end{example}

\begin{example}[Estrita de graça]\label{ex:b1:derivative:strictness}
$x \mapsto x^3$ é \emph{estritamente} crescente em $\R$, ainda que
a sua \omterm{def:b1:derivative:def}{derivada} se anule em $0$: a cláusula do critério
``$f' > 0$ salvo em finitos pontos'' foi desenhada exatamente
para tais pontos planos. Em contrapartida, $f' \geq 0$ sozinho só dá
crescimento no sentido largo (uma função constante se qualifica), e
uma \omterm{def:b1:derivative:def}{derivada} que se anula num subintervalo inteiro de fato congela a
função ali. A regra prática: para afirmar monotonicidade
estrita, liste os zeros de $f'$; finitos (ou, mais
geralmente, nenhum em subintervalo algum) é inofensivo, e um \omterm{prop:b1:reals:intervals}{intervalo}
deles é fatal.
\end{example}

\begin{example}[Um estudo de variação completo]\label{ex:b1:derivative:variations}
Estude $f(x) = x^3 - 3x + 1$ em $\R$. \omterm{def:b1:derivative:def}{Derivada}: $f'(x) =
3(x^2 - 1)$, positiva em $\intoo{-\infty}{-1}$, negativa em
$\intoo{-1}{1}$, positiva em $\intoo{1}{+\infty}$: pelo
critério de monotonicidade, $f$ cresce, depois decresce, depois
cresce, com um máximo local $f(-1) = 3$ e um mínimo local
$f(1) = -1$. Limites: $\mp\infty$ em $\mp\infty$. Consequências,
lidas na tabela de variação com o teorema do valor
intermediário em cada ramo monótono: $f$ se anula exatamente uma vez em
cada um de
\[
\intoo{-\infty}{-1}, \qquad \intoo{-1}{1}, \qquad
\intoo{1}{+\infty}
\]
(os valores nas junções têm sinais opostos: $3 > 0 > -1$),
de modo que a equação $x^3 - 3x + 1 = 0$ tem exatamente três raízes
reais; numericamente, elas ficam perto de $-1.88$, $0.35$ e $1.53$. A
ideia de fechamento: uma tabela de variação é um \emph{instrumento de
demonstração}, não um esboço --- ramo monótono mais mudança de sinal é
igual a exatamente uma raiz, e a tabela enumera os ramos
exaustivamente.
\end{example}

\begin{theorem}[Fórmula de Leibniz]\label{thm:b1:derivative:leibniz}
Se $f, g$ são $n$ vezes deriváveis, então $fg$ também é,
e\index{fórmula de Leibniz}
\[
(fg)^{(n)} = \sum_{k=0}^{n} \binom nk f^{(k)}\, g^{(n-k)} .
\]
\end{theorem}

\begin{proof}
Indução em $n$, exatamente paralela ao teorema binomial. O
caso $n = 1$ é a regra do produto. Supondo a fórmula no posto
$n$, derive mais uma vez:
\[
(fg)^{(n+1)} = \sum_{k=0}^{n} \binom nk
\Bigl( f^{(k+1)} g^{(n-k)} + f^{(k)} g^{(n-k+1)} \Bigr),
\]
depois reindexe a primeira soma com $j = k + 1$ e reúna o
coeficiente de $f^{(j)} g^{(n+1-j)}$: ele é $\binom{n}{j-1} +
\binom nj = \binom{n+1}{j}$, pela regra de Pascal
(\cref{prop:b1:counting:identities}), com os termos de \omterm{def:b1:topology:closure}{fronteira} $j = 0$
e $j = n + 1$ carregando $\binom{n+1}{0} = \binom{n+1}{n+1} =
1$, como devem.
\end{proof}

\begin{example}[Leibniz em ação]\label{ex:b1:derivative:leibnizaction}
Calcule $\bigl(x^2 \eu^x\bigr)^{(n)}$ para $n \geq 2$. Tome $f =
x^2$, cujas \omterm{def:b1:derivative:def}{derivadas} morrem depressa ($f' = 2x$, $f'' = 2$,
$f^{(k)} = 0$ para $k \geq 3$), e $g = \eu^x$: só três termos
da soma de Leibniz sobrevivem,
\[
\bigl(x^2\eu^x\bigr)^{(n)}
= \binom n0 x^2 \eu^x + \binom n1 (2x)\,\eu^x + \binom n2\,
2\,\eu^x
= \eu^x\bigl(x^2 + 2nx + n(n-1)\bigr).
\]
Verificação de sanidade em $n = 1$: $\eu^x(x^2 + 2x)$, que é de fato
$(x^2\eu^x)'$. A ideia de fechamento: use Leibniz quando um dos fatores
for um \omterm{def:b1:poly:def}{polinômio} --- a soma tem então apenas $\deg + 1$ termos, e
a fórmula é uma forma \omterm{def:b1:topology:closed}{fechada}, não uma identidade abstrata. (Para dois
fatores infinitamente vivos, como $\eu^x\sin x$, as exponenciais
complexas do \cref{ch:b1:complex} são a ferramenta melhor.)
\end{example}

\section{Convexidade}

\begin{definition}\label{def:b1:derivative:convex}
$f \colon I \to \R$ é \emph{convexa}\index{função convexa} quando
toda corda fica acima do gráfico:
\[
\forall x, y \in I,\ \forall t \in \intcc{0}{1}, \quad
f\bigl(tx + (1-t)y\bigr) \leq t f(x) + (1-t) f(y).
\]
($f$ é \emph{côncava} quando $-f$ é convexa.)
\end{definition}

\begin{theorem}[Caracterizações diferenciais]\label{thm:b1:derivative:convexchar}
Seja $f$ \omterm{def:b1:derivative:def}{derivável} em $I$. São equivalentes:
\begin{enumerate}
  \item $f$ é \omterm{def:b1:derivative:convex}{convexa};
  \item $f'$ é crescente em $I$;
  \item o gráfico fica acima de toda tangente: $f(y) \geq f(x) +
        f'(x)(y - x)$ para todos $x, y \in I$.
\end{enumerate}
Se $f$ é duas vezes \omterm{def:b1:derivative:def}{derivável}: $f$ \omterm{def:b1:derivative:convex}{convexa} $\iff f'' \geq 0$.
\end{theorem}

\begin{proof}
(1 $\Rightarrow$ 3) A \omterm{def:b1:derivative:convex}{convexidade} escrita como $\frac{f(x + t(y-x)) -
f(x)}{t} \leq f(y) - f(x)$ para $t \in \intoc{0}{1}$; faça $t \to 0^+$:
$f'(x)(y - x) \leq f(y) - f(x)$.

(3 $\Rightarrow$ 2) Para $x < y$, as duas desigualdades de tangente em $x$
e em $y$ dão $f'(x)(y-x) \leq f(y) - f(x) \leq f'(y)(y - x)$,
donde $f'(x) \leq f'(y)$.

(2 $\Rightarrow$ 1) Fixe $x < y$ e $t \in \intoo{0}{1}$, e seja $z
= tx + (1-t)y \in \intoo{x}{y}$. Pelo teorema do valor médio em
$\intcc{x}{z}$ e em $\intcc{z}{y}$, existem $c_1 < z < c_2$ com
\[
\frac{f(z) - f(x)}{z - x} = f'(c_1) \leq f'(c_2)
= \frac{f(y) - f(z)}{y - z} ,
\]
e eliminar denominadores ($z - x = (1-t)(y-x)$, $y - z = t(y-x)$)
rearranja exatamente na desigualdade de \omterm{def:b1:derivative:convex}{convexidade}.

Caso duas vezes \omterm{def:b1:derivative:def}{derivável}: $f'' \geq 0 \iff f'$ crescente
(\cref{cor:b1:derivative:monotone}).
\end{proof}

\begin{example}[Desigualdades clássicas de convexidade]\label{ex:b1:derivative:convexineq}
$\exp$ é \omterm{def:b1:derivative:convex}{convexa} ($\exp'' = \exp > 0$): a sua tangente em $0$ dá
$\eu^x \geq 1 + x$ para todo $x$. $\ln$ é côncava: a sua tangente em $1$
dá $\ln x \leq x - 1$; as suas cordas dão, para $0 < a \leq b$, a
desigualdade entre as médias geométrica e aritmética: tomando $t =
\frac12$ na concavidade,
\[
\ln\frac{a + b}{2} \geq \frac{\ln a + \ln b}{2} = \ln\sqrt{ab},
\qquad\text{logo}\qquad
\sqrt{ab} \leq \frac{a+b}{2} .
\]
A desigualdade aritmético-geométrica geral é o
\cref{exo:b1:derivative:9}.
\end{example}

\begin{example}[Uma desigualdade de convexidade do zero]\label{ex:b1:derivative:xlnx}
A função $f(t) = t\ln t$ é \omterm{def:b1:derivative:convex}{convexa} em $\intoo{0}{+\infty}$:
$f''(t) = \frac1t > 0$. A sua desigualdade do ponto médio, multiplicada por
$2$, diz: para todos $a, b > 0$,
\[
a\ln a + b\ln b \;\geq\; (a + b)\,\ln\frac{a + b}{2} ,
\]
com igualdade se, e somente se, $a = b$ (\omterm{def:b1:derivative:convex}{convexidade} estrita). Teste: $a =
1$, $b = 3$ dá $3\ln 3 = 3.296$ contra $4\ln 2 = 2.773$.
Esta desigualdade inócua é o caso de dois pontos da comparação de
\emph{entropia} que reaparece com a desigualdade de Jensen
(\cref{exo:b1:derivative:9}) e nas assintóticas da teoria da
informação do volume do terceiro ano. A
ideia de fechamento: para fabricar uma desigualdade, encontre uma função
cuja segunda \omterm{def:b1:derivative:def}{derivada} tenha sinal e escreva o que a \omterm{def:b1:derivative:convex}{convexidade}
diz --- a caracterização diferencial converte uma verificação de sinal
em infinitas desigualdades.
\end{example}

\begin{remark}[Armadilhas frequentes com derivadas]
(i) \emph{Uma \omterm{def:b1:derivative:def}{derivada} positiva num ponto não dá
monotonicidade perto dele}: $f(x) = \frac x2 + x^2\sin\frac1x$ (com
$f(0) = 0$) tem $f'(0) = \frac12 > 0$ e, no entanto,
\[
f'(x) = \frac12 + 2x\sin\frac1x - \cos\frac1x
\]
vale $-\frac12$ em cada $x_n = \frac{1}{2\pi n}$: toda
\omterm{def:b1:topology:open}{vizinhança} de $0$ contém descidas. A monotonicidade exige $f'
\geq 0$ \emph{num \omterm{prop:b1:reals:intervals}{intervalo}}
(\cref{cor:b1:derivative:monotone}); o sinal pontual só
controla o cruzamento da reta tangente. (ii) \emph{As três hipóteses de
Rolle estão todas ativas}: $\abs x$ em $\intcc{-1}{1}$
(sem \omterm{def:b1:derivative:def}{derivabilidade} \omterm{def:b1:topology:closure}{interior}), $x$ em $\intcc{0}{1}$ (extremidades não
iguais) e $x - \lfloor x\rfloor$ em $\intcc{0}{1}$ (a \omterm{def:b1:continuity:continuous}{continuidade}
falha em $1$) quebram cada uma exatamente uma hipótese e a
conclusão. (iii) \emph{As \omterm{def:b1:derivative:def}{derivadas} podem ser descontínuas, mas
não arbitrariamente}: $f'$ pode oscilar
(\cref{ex:b1:derivative:notc1}) e, ainda assim, sempre satisfaz a
propriedade do valor intermediário (Darboux,
\cref{exo:b1:derivative:10}): uma \omterm{def:b1:derivative:def}{derivada} nunca salta --- se
você calcular um ``limite de \omterm{def:b1:derivative:def}{derivada}'' lateral com salto, terá
derivado uma função não \omterm{def:b1:derivative:def}{derivável}. (iv) \emph{A fórmula da
inversa precisa de $f' \neq 0$}: $x \mapsto x^3$ é uma bijeção suave
estritamente crescente cuja inversa $x^{1/3}$ tem
tangente vertical em $0$ --- a \omterm{def:b1:derivative:def}{derivabilidade} da inversa se
perde exatamente onde $f'$ se anula
(\cref{thm:b1:derivative:inverse}).
\end{remark}

\begin{remark}[Onde o teorema do valor médio trabalha em seguida]
Quase todo enunciado quantitativo dos próximos capítulos é o
teorema do valor médio deste capítulo fantasiado: o teorema fundamental
do cálculo (\cref{ch:b1:integration}) deriva a função área
e conclui com o critério de monotonicidade; a fórmula de
Taylor--Lagrange (\cref{ch:b1:taylor}) é o teorema do valor
médio iterado $n$ vezes; a análise de erro do método de Newton
e das iterações de ponto fixo (\cref{exo:b1:derivative:11}) é
a forma de Lipschitz; e o problema de fim de semana deste capítulo
(\cref{pb:b1:derivative:1}) converte a mesma estimativa de Lipschitz em
teoria dos números --- uma desigualdade de repulsão entre \omterm{pb:b1:logic:1}{números
algébricos} e racionais, que produz o primeiro \omterm{pb:b1:logic:1}{número transcendente}
da história. No volume do segundo ano, a \emph{desigualdade} do valor
médio sobrevive em várias variáveis, quando a igualdade não
sobrevive.
\end{remark}

\begin{example}[A desigualdade de Young a partir da concavidade]\label{ex:b1:derivative:young}
Sejam $p, q > 1$ com $\frac1p + \frac1q = 1$. Para todos $a, b > 0$:
\[
ab \;\leq\; \frac{a^p}{p} + \frac{b^q}{q} .
\]
Demonstração por uma \omterm{def:b1:logic:map}{aplicação} da concavidade de $\ln$ com pesos
$\frac1p, \frac1q$ (a desigualdade de Jensen em dois pontos, como no
\cref{exo:b1:derivative:9}):
\[
\ln\Bigl(\frac{a^p}{p} + \frac{b^q}{q}\Bigr)
\;\geq\; \frac1p \ln(a^p) + \frac1q \ln(b^q)
= \ln a + \ln b = \ln(ab),
\]
e, sendo $\ln$ crescente, a desigualdade dos logaritmos se converte na
afirmação; a igualdade vale se, e somente se, $a^p = b^q$ (concavidade
estrita). O caso $p = q = 2$ é a desigualdade aritmético-geométrica
$ab \leq
\frac{a^2 + b^2}{2}$ disfarçada. A ideia de fechamento: a desigualdade
de Young é a semente algébrica das desigualdades de Hölder e Minkowski
do volume do segundo ano --- um enunciado de concavidade
sobre $\ln$, colhido para normas.
\end{example}

\begin{remark}[Perspectivas dentro deste volume]
A \omterm{def:b1:derivative:def}{derivada} ganha três vidas novas antes que o volume termine.
No \cref{ch:b1:taylor} ela itera: $n$ \omterm{def:b1:derivative:def}{derivadas} num ponto
se comprimem num \omterm{def:b1:poly:def}{polinômio} mais um erro controlado, e o
teorema do valor médio se torna o resto de Lagrange. No
\cref{ch:b1:curves}, a derivação se torna geométrica: para uma
curva parametrizada $t \mapsto (x(t), y(t))$, o par $(x'(t),
y'(t))$ é um \emph{vetor} velocidade, a tangência se torna
colinearidade, e os pontos críticos se tornam cúspides a classificar. No
\cref{ch:b1:multivar}, uma variável é congelada de cada vez: as \omterm{def:b1:derivative:def}{derivadas}
parciais repetem este capítulo duas vezes, e a reta tangente
cresce num plano tangente. Os três capítulos herdam a mesma
gramática --- aproximação linear local mais um termo de erro ---
falada aqui pela primeira vez.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:derivative:1}
Derive (especificando os domínios): $x^x$;
$\;\ln\bigl(x + \sqrt{x^2+1}\bigr)$; $\;\arctan\frac{1}{x}$;
$\;\sqrt{1 + \eu^{2x}}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:2}
Complete o \cref{ex:b1:derivative:notc1}: demonstre que $f(x) = x^2
\sin\frac1x$, $f(0) = 0$, é \omterm{def:b1:derivative:def}{derivável} em $0$ com $f'(0) = 0$,
e que $f'$ não tem limite em $0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:3}
Usando o teorema do valor médio ou as desigualdades de tangente,
demonstre que, para todo $x > 0$:
\[
\frac{x}{1 + x} < \ln(1 + x) < x .
\]
Deduza que $\lim_{n\to\infty} \bigl(1 + \frac xn\bigr)^n = \eu^x$ para
todo $x > 0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:4}
Seja $P$ um \omterm{def:b1:poly:def}{polinômio} real com $k$ raízes reais distintas. Demonstre que
$P'$ tem pelo menos $k - 1$ raízes reais distintas, intercaladas com as
de $P$. Deduza que, se $P$ tem todas as raízes reais, então $P'$ também
tem.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:5}
Seja $f$ \omterm{def:b1:derivative:def}{derivável} em $\R$ com $f' (x)\to \ell$ quando $x \to
+\infty$. Demonstre que $\frac{f(x)}{x} \to \ell$ \emph{(teorema do valor
médio em $\intcc{A}{x}$)}. E $f(x+1) - f(x) \to \ell$, também vale?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:6}
(Um Rolle discreto) Seja $f$ $n$ vezes \omterm{def:b1:derivative:def}{derivável} em $I$ e que
se anule em $n + 1$ pontos distintos. Demonstre que $f^{(n)}$ se anula
ao menos uma vez. Aplicação: um \omterm{def:b1:poly:def}{polinômio} de grau $\leq n$ que se anula
em $n+1$ pontos é nulo (de novo).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:7}
Seja $f$ duas vezes \omterm{def:b1:derivative:def}{derivável} em $\intcc{a}{b}$ com $f(a) = f(b)
= 0$ e $f(x_0) > 0$ para algum $x_0$ \omterm{def:b1:topology:closure}{interior}. Demonstre que $f''(c) <
0$ para algum $c \in \intoo{a}{b}$. \emph{(Dois teoremas do valor médio e
uma comparação de inclinações.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:8}
Estude a função $f(x) = \dfrac{\ln x}{x}$ em
$\intoo{0}{+\infty}$: variações, limites, máximo. Deduza que $a^b
> b^a$ para todos os reais $\eu \leq a < b$, e resolva o famoso caso
particular: qual é maior, $\eu^\pi$ ou $\pi^\eu$? Confira contra os
pares de inteiros pequenos $(2,3)$ e $(2,4)$: por que eles se comportam
de modo diferente?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:9}
(Desigualdade aritmético-geométrica) Usando a concavidade de $\ln$ com
pesos gerais (a desigualdade de Jensen para $n$ pontos, a ser
demonstrada por indução em $n$), mostre que, para reais positivos
$a_1, \dots, a_n$:
\[
\sqrt[n]{a_1 a_2 \cdots a_n} \leq \frac{a_1 + \dots + a_n}{n},
\]
com igualdade se, e somente se, todos os $a_i$ forem iguais.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:10}
(Darboux: as \omterm{def:b1:derivative:def}{derivadas} assumem valores intermediários) Seja $f$
\omterm{def:b1:derivative:def}{derivável} em $I$ e sejam $a < b$ em $I$ com $f'(a) < v < f'(b)$.
Considerando $g(x) = f(x) - vx$ e o ponto em que $g$ atinge
o seu mínimo em $\intcc{a}{b}$, demonstre que $f'(c) = v$ para algum $c
\in \intoo{a}{b}$ --- mesmo que $f'$ não precise ser \omterm{def:b1:continuity:continuous}{contínua}
(\cref{exo:b1:derivative:2}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:11}
Seja $f \colon \R \to \R$ \omterm{def:b1:derivative:def}{derivável} com $\abs{f'(x)} \leq k
< 1$ para todo $x$ (uma \emph{contração}). Demonstre que $f$ tem exatamente
um ponto fixo $\ell$ e que toda sequência $u_{n+1} = f(u_n)$
converge para $\ell$, com $\abs{u_n - \ell} \leq k^n\abs{u_0 -
\ell}$. \emph{(Existência: aplique o teorema do valor intermediário a
$f(x) - x$ num segmento grande, usando a estimativa de Lipschitz; ou
use a completude com o critério de Cauchy.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:12}
(Teorema do valor médio de Cauchy e regra de l'Hôpital)
\begin{enumerate}
  \item Sejam $f, g$ \omterm{def:b1:continuity:continuous}{contínuas} em $\intcc{a}{b}$,
        deriváveis em $\intoo{a}{b}$, com $g'$ nunca nula
        ali. Demonstre que $g(b) \neq g(a)$ e que algum $c \in
        \intoo{a}{b}$ satisfaz
        \[
        \frac{f(b) - f(a)}{g(b) - g(a)} = \frac{f'(c)}{g'(c)}
        \]
        \emph{(aplique Rolle a $h = f - \lambda g$ para a constante
        $\lambda$ certa)}.
  \item Deduza a regra de l'Hôpital na forma $\frac00$ num
        ponto: se $f(a) = g(a) = 0$ e
        $\frac{f'(x)}{g'(x)} \to \ell$ quando $x \to a^+$, então
        $\frac{f(x)}{g(x)} \to \ell$.
  \item Mostre que a recíproca falha: para $f(x) = x^2\sin\frac1x$
        ($f(0) = 0$) e $g(x) = x$, o quociente
        $\frac{f}{g}$ tem limite em $0$, mas
        $\frac{f'}{g'}$ não tem.
\end{enumerate}
\end{exercise}

\section{Problema: A desigualdade de Liouville e o primeiro
número transcendente}

\begin{problem}[{Problema de fim de semana --- os \omterm{pb:b1:logic:1}{números algébricos}
repelem os racionais: $\abs{x - p/q} \geq C/q^d$, e a transcendência de
$\sum 10^{-n!}$}]
\label{pb:b1:derivative:1}
Um número real é \emph{algébrico} quando é raiz de um \omterm{def:b1:poly:def}{polinômio} não
nulo com coeficientes inteiros, e \emph{transcendente}
caso contrário. Em 1844, Liouville produziu o primeiro número jamais
\emph{demonstrado} transcendente, e o motor da sua demonstração é o
teorema do valor médio deste capítulo: um \omterm{pb:b1:logic:1}{número algébrico} de grau $d$
não pode ser aproximado por racionais melhor do que $C/q^d$ --- de modo
que um número aproximável \emph{mais depressa que qualquer potência}
não pode ser algébrico. Este problema constrói a desigualdade, constrói
o número de Liouville $L = 0.110001000\dots$ (uns nas posições
fatoriais, via a maquinaria de algarismos do \cref{pb:b1:reals:1}),
demonstra a sua transcendência e termina com a demonstração rival de
Cantor e com estimativas efetivas para $\sqrt2$ e
$2^{1/3}$.\index{número transcendente}\index{desigualdade de
Liouville}\index{número algébrico}

\medskip
\noindent\textbf{Parte I --- Quão bem os racionais podem ser
aproximados?}
\begin{enumerate}
  \item Mostre que dois racionais distintos $\frac ab \neq \frac pq$
        (escritos com $b, q \geq 1$) satisfazem $\bigl|\frac ab -
        \frac pq\bigr| \geq \frac{1}{bq}$. Deduza: se $x =
        \frac ab$ e $0 < \bigl|x - \frac pq\bigr| <
        \frac{1}{bq}$, então $\frac pq$ algum assim existe --- um racional
        repele todos os outros racionais na escala $\frac 1q$.
  \item Demonstre que, para \emph{todo} racional $\frac pq$ ($q \geq
        1$): $\bigl|\sqrt2 - \frac pq\bigr| \geq
        \frac{1}{4q^2}$ \emph{(se a distância excede $1$, isso é
        claro; caso contrário, estime $\abs{\sqrt2 + p/q} < 4$ e
        use o inteiro não nulo $\abs{p^2 - 2q^2} \geq 1$)}.
  \item Na outra direção: verifique que $(p, q) \mapsto (p +
        2q, p + q)$ preserva $\abs{p^2 - 2q^2} = 1$, gere
        a partir de $(1,1)$ os pares $(3,2)$, $(7,5)$, $(17,12)$,
        $(41,29)$, $(99,70)$, e mostre que cada um satisfaz
        \[
        \Bigl|\sqrt2 - \frac pq\Bigr| =
        \frac{1}{q^2\,(\sqrt2 + p/q)} < \frac{1}{2q^2} :
        \]
        infinitas aproximações de ordem $2$. Com a
        questão 2: o expoente de aproximação de $\sqrt 2$ é
        \emph{exatamente} $2$.
  \item (Dirichlet) Seja $x$ irracional e $N \in \N^*$.
        Considere as $N + 1$ partes fracionárias de $0, x, 2x,
        \dots, Nx$ nas $N$ casas $\intco{\frac kN}{\frac{k +
        1}{N}}$: pelo princípio da casa dos pombos
        (\cref{cor:b1:counting:pigeonhole}), duas caem numa mesma casa.
        Deduza a existência de $q \leq N$ e de $p$ com $\abs{qx - p} <
        \frac 1N$, e portanto de infinitos racionais com
        $\bigl|x - \frac pq\bigr| < \frac{1}{q^2}$: \emph{todo}
        irracional é aproximável na ordem $2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A desigualdade de Liouville.} Seja $x$
irracional e algébrico.
\begin{enumerate}
  \item Mostre que, entre os \omterm{def:b1:poly:def}{polinômios} inteiros não nulos que se
        anulam em $x$, existe um, digamos $P$ de grau $d$,
        \emph{sem raiz racional}; e verifique que $d \geq 2$
        \emph{(divida por um fator $X - \frac ab$ sobre $\Q$ e
        elimine denominadores; grau $1$ tornaria $x$
        racional)}.
  \item Mostre que, para todo racional $\frac pq$ ($q \geq 1$):
        $\bigl|P\bigl(\frac pq\bigr)\bigr| \geq \frac{1}{q^d}$
        \emph{($q^d P(p/q)$ é um inteiro não nulo)}.
  \item Seja $M = \max_{\intcc{x-1}{x+1}} \abs{P'}$
        (\cref{thm:b1:continuity:evt}). Usando o teorema do valor
        médio entre $x$ e $\frac pq$, demonstre a
        \emph{desigualdade de Liouville}: com $C = \min\bigl(1,
        \frac 1M\bigr) > 0$,
        \[
        \Bigl| x - \frac pq \Bigr| \geq \frac{C}{q^{\,d}}
        \qquad\text{para todo racional} \frac pq,\ q \geq 1 .
        \]
  \item Chame $x$ de \emph{número de Liouville} quando, para todo $n \in
        \N$, existir um racional $\frac pq$ com $q \geq 2$ e
        $0 < \bigl|x - \frac pq\bigr| < q^{-n}$. Demonstre que um
        número de Liouville é irracional \emph{(questão 1: escolha
        $n$ com $2^{\,n-1} > b$)}.
  \item Demonstre o teorema de Liouville: \emph{um número de Liouville é
        transcendente} \emph{(combine as questões 7 e 8: a
        desigualdade $C < q^{\,d-n}$ falha para $n$ grande)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O número $L$.}
\begin{enumerate}
  \item Seja $L$ o valor (no sentido do
        \cref{pb:b1:reals:1}) da cadeia decimal de algarismos com
        algarismo $1$ nas posições $n!$ ($n = 1, 2, 3, \dots$)
        e $0$ nas demais, isto é, $L = \sup_k t_k$ com $t_k =
        \sum_{n=1}^{k} 10^{-n!}$. Escreva os $25$ primeiros
        algarismos de $L$.
  \item Demonstre o enquadramento da cauda, para todo $k \geq 1$:
        \[
        10^{-(k+1)!} \;\leq\; L - t_k \;\leq\;
        \frac{10}{9}\,10^{-(k+1)!} \;<\; 2\cdot 10^{-(k+1)!}
        \]
        \emph{(estime toda soma parcial além de $t_k$ por uma soma
        geométrica finita)}.
  \item Escreva $t_k = \frac{p_k}{q_k}$ com $q_k = 10^{k!}$.
        Mostre que $0 < L - \frac{p_k}{q_k} < \frac{2}{q_k^{\,k+1}}$
        e conclua que $L$ é um número de Liouville no sentido
        da questão 8.
  \item Conclua: $L$ é transcendente --- o primeiro exemplo explícito
        da história (Liouville, 1844). Verifique diretamente a sua
        irracionalidade: os seus algarismos não são periódicos a partir
        de ponto algum (espaçamentos crescentes, como no \cref{pb:b1:reals:1},
        questão 20).
  \item Generalize: substitua cada algarismo $1$ por um algarismo
        não nulo arbitrário $d_n \in \intint{1}{9}$. Mostre que o valor ainda é
        um número de Liouville e deduza --- pelo argumento
        diagonal do \cref{pb:b1:reals:1} (questão 22), aplicado
        a essas escolhas de algarismos --- que existe uma quantidade
        não enumerável de \omterm{pb:b1:logic:1}{números transcendentes} dessa forma.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A hierarquia das ordens de
aproximação.} Diga que $x$ é \emph{aproximável na ordem $\mu$} quando,
para alguma constante $c > 0$, infinitos racionais satisfazem
$\bigl|x - \frac pq\bigr| < \frac{c}{q^{\mu}}$.
\begin{enumerate}
  \item Monte a hierarquia a partir das Partes I--III: os racionais são
        aproximáveis na ordem $1$ e não melhor; $\sqrt 2$ na
        ordem $2$ e não melhor; todo irracional em ordem pelo
        menos $2$; um \omterm{pb:b1:logic:1}{número algébrico} de grau $d$ em ordem alguma
        além de $d$; os números de Liouville em toda ordem. Justifique
        cada afirmação citando a questão pertinente.
  \item Mostre que $L + r$ é um número de Liouville para todo
        racional $r = \frac ab$ \emph{(translade os
        aproximantes: os novos denominadores são $b\,q_k$)}.
        Conclua que os números de Liouville --- logo, os
        transcendentes --- são \omterm{def:b1:topology:dense}{densos} em $\R$.
  \item (Cantor, 1874) Demonstre que o \omterm{def:b1:logic:sets}{conjunto} dos \omterm{pb:b1:logic:1}{números algébricos}
        é enumerável: há finitos \omterm{def:b1:poly:def}{polinômios}
        inteiros com grau mais soma dos $\abs{\text{coeficientes}}$
        limitados por $h$, cada um com no máximo $\deg$ raízes; e uma
        união enumerável de \omterm{def:b1:counting:card}{conjuntos finitos} é enumerável. Como
        sequência alguma esgota $\R$ (\cref{pb:b1:reals:1}, questão
        22), existem \omterm{pb:b1:logic:1}{números transcendentes} --- de fato, eles formam um
        \omterm{def:b1:logic:sets}{conjunto} não enumerável. Compare as duas demonstrações: o que a
        de Liouville dá que a de Cantor não pode dar?
  \item Demonstre diretamente, a partir da questão 2, que $\sqrt 2$
        \emph{não} é um número de Liouville \emph{(para $n \geq 3$,
        a desigualdade $q^{-n} > \frac{1}{4q^2}$ limita $q$;
        restam então apenas finitos racionais candidatos, todos
        a distância positiva de $\sqrt2$)}. Generalize:
        nenhum \omterm{pb:b1:logic:1}{número algébrico} é de Liouville.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Constantes efetivas.}
\begin{enumerate}
  \item Para o par de Pell $(99, 70)$: verifique que $99^2 - 2\cdot70^2
        = 1$ e avalie o erro exato
        \[
        \sqrt2 - \frac{99}{70}
        = \frac{-1}{70^2\,\bigl(\sqrt2 + \frac{99}{70}\bigr)},
        \qquad
        \Bigl|\sqrt 2 - \frac{99}{70}\Bigr| \approx 7.2\cdot
        10^{-5} :
        \]
        cinco algarismos corretos a partir de uma fração de três
        algarismos.
  \item Execute a Parte II em $x = 2^{1/3}$, $P = X^3 - 2$: verifique que $P$
        não tem raiz racional, estime $M = \max_{\intcc{x-1}{x+1}}
        3t^2 \leq 3\,(1 + 2^{1/3})^2 < 16$ e conclua a
        desigualdade efetiva
        \[
        \Bigl| 2^{1/3} - \frac pq \Bigr| \geq
        \frac{1}{16\,q^3} \qquad \text{para todos} \frac pq .
        \]
  \item Recompensa: mostre que todo racional que aproxime $2^{1/3}$
        a menos de $10^{-6}$ deve ter denominador $q \geq 40$.
  \item Mostre que a base $10$ é irrelevante: o análogo binário
        $\sum_{n\geq1} 2^{-n!}$ (valor da cadeia binária
        com uns nas posições fatoriais) também é um
        número de Liouville, logo transcendente.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- \omterm{def:b1:topology:closure}{Fronteiras} e síntese.}
\begin{enumerate}
  \item Seja $x^\dagger$ o valor da cadeia decimal com
        uns exatamente nas posições $3^k$ ($k \geq 0$). Mostre que
        $x^\dagger$ é aproximável na ordem $3$ e deduza,
        da desigualdade de Liouville, que $x^\dagger$ não é
        racional nem um irracional quadrático. Explique por que o
        método empaca aí: ordem $3$ é compatível com
        algebricidade de grau $\geq 3$, e fechar essa lacuna
        (qualquer expoente $> 2$ basta, para todo \omterm{pb:b1:logic:1}{número
        algébrico}) é o teorema de Roth, muito além deste volume.
  \item Quantifique Cantor: mostre que os \omterm{pb:b1:logic:1}{números algébricos} de
        grau $\leq d$ dados por \omterm{def:b1:poly:def}{polinômios} com coeficientes
        em $\intint{-H}{H}$ são no máximo $d\,(2H + 1)^{d+1}$.
        (Essa finitude é o que fez a questão 17 funcionar.)
  \item Síntese, uma frase para cada: (i) localize o único
        ingrediente analítico da demonstração de Liouville (qual teorema
        deste capítulo, usado onde); (ii) enuncie a tensão
        que a alimenta (a integralidade força $\abs{P(p/q)} \geq
        q^{-d}$, e a suavidade proíbe $\abs{P(p/q)} > M\abs{x -
        p/q}$); (iii) contraste as demonstrações de Liouville e de
        Cantor da existência de \omterm{pb:b1:logic:1}{números transcendentes}; (iv) nomeie
        onde este volume reencontra o tema --- o problema de fim de
        semana do \cref{ch:b1:integration} demonstra que $\pi$ é
        irracional pelo mesmo confronto entre integralidade e pequenez,
        com integrais no lugar de \omterm{def:b1:derivative:def}{derivadas}.
\end{enumerate}
\end{problem}
