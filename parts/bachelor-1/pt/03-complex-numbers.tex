\chapter{Números Complexos}\label{ch:b1:complex}

Os números complexos foram encontrados no volume do ensino médio como um
recurso de cálculo para equações do segundo grau. Este capítulo os trata
como objeto central: a forma exponencial e suas consequências (De Moivre,
raízes $n$-ésimas, \omterm{def:b1:complex:unity}{raízes da unidade}), a tradução sistemática entre $\C$ e
a geometria plana, e o uso de $\eu^{\iu\theta}$ como uma máquina para
demonstrar identidades trigonométricas.

\section{O corpo $\C$, módulo e conjugado}

\begin{definition}[O corpo dos complexos]\label{def:b1:complex:field}
$\C = \{a + \iu b : a, b \in \R\}$ com a adição usual e a
multiplicação determinada por $\iu^2 = -1$. Todo $z = a + \iu
b$ não nulo
tem um inverso: $z^{-1} = \frac{a - \iu b}{a^2 + b^2}$ --- na
linguagem do \cref{ch:b1:structures}, $\C$ é um corpo. Escreve-se $a =
\Re(z)$, $b = \Im(z)$, $\conj{z} = a - \iu b$ (o
\emph{conjugado}\index{conjugado}) e $\abs{z} = \sqrt{a^2 + b^2}$
(o \emph{módulo}\index{módulo}).
\end{definition}

\begin{proposition}[Regras do conjugado e do módulo]\label{prop:b1:complex:rules}
Para $z, w \in \C$:
\begin{enumerate}
  \item $\conj{z + w} = \conj z + \conj w$, $\conj{zw} = \conj z\,
        \conj w$, $\conj{\conj z} = z$;
  \item $z \conj z = \abs z^2$; $\;\Re(z) = \frac{z + \conj z}{2}$,
        $\Im(z) = \frac{z - \conj z}{2\iu}$;
  \item $\abs{zw} = \abs z\, \abs w$, e $\abs{z^{-1}} =
        \abs{z}^{-1}$
        para $z \neq 0$;
  \item (desigualdade triangular\index{desigualdade triangular}) $\abs{z + w}
        \leq \abs z + \abs w$, com igualdade se, e somente se, $z$ e
        $w$ estão numa mesma semirreta com origem em $0$ ($w = \lambda z$ ou
        $z = \lambda w$ com $\lambda \geq 0$);
  \item (desigualdade triangular inversa) $\bigl|\abs z - \abs w\bigr|
        \leq \abs{z - w}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) e (2) são cálculos diretos sobre as partes real e imaginária. (3):
$\abs{zw}^2 = zw\,\conj{zw} = z\conj z\, w \conj w = \abs z^2 \abs
w^2$, e depois extraia raízes quadradas; aplique a $z \cdot z^{-1} = 1$ para o
inverso.

(4) Os dois lados são não negativos; compare então os quadrados:
\[
\abs{z+w}^2 = (z+w)(\conj z + \conj w)
= \abs z^2 + \abs w^2 + 2\,\Re(z \conj w),
\]
e $\Re(z\conj w) \leq \abs{z \conj w} = \abs z \abs w$ dá
$\abs{z+w}^2 \leq (\abs z + \abs w)^2$. A igualdade força $\Re(z \conj
w) = \abs{z \conj w}$, isto é, $z \conj w \in \R_+$; se $w \neq 0$, isso
dá $z = \frac{z\conj w}{\abs w^2}\, w = \lambda w$ com $\lambda
\geq 0$ (e o caso $w = 0$ é trivial).

(5) $\abs z = \abs{(z - w) + w} \leq \abs{z-w} + \abs w$ dá $\abs z
- \abs w \leq \abs{z - w}$; troque $z$ e $w$ para o outro sinal.
\end{proof}

\begin{example}[Um exercício completo de módulo e argumento]\label{ex:b1:complex:workout}
Ponha $w = \dfrac{3 + 4\iu}{1 - 2\iu}$ na forma algébrica e calcule
o seu \omterm{def:b1:complex:field}{módulo} de duas maneiras. Multiplicando pelo \omterm{def:b1:complex:field}{conjugado} do
denominador:
\[
w = \frac{(3 + 4\iu)(1 + 2\iu)}{(1 - 2\iu)(1 + 2\iu)}
= \frac{3 + 6\iu + 4\iu - 8}{1 + 4}
= \frac{-5 + 10\iu}{5} = -1 + 2\iu .
\]
Diretamente: $\abs w = \sqrt{1 + 4} = \sqrt5$. Pela regra do quociente
(\cref{prop:b1:complex:rules}~(3)): $\abs w = \frac{\abs{3 +
4\iu}}{\abs{1 - 2\iu}} = \frac5{\sqrt5} = \sqrt5$ --- mesma resposta,
sem precisar da forma algébrica. A lição se generaliza: \omterm{def:b1:complex:field}{módulos} e
argumentos viajam bem através de produtos e quocientes; partes reais e
imaginárias viajam bem através de somas. Escolha a representação que
corresponde às operações em jogo e converta apenas quando for obrigado.
\end{example}

\begin{example}[Equações que envolvem o conjugado]\label{ex:b1:complex:conjeq}
Resolva em $\C$: $\;z + 2\conj z = 6 + 2\iu$. Uma equação que mistura $z$
e $\conj z$ \emph{não} é polinomial em $z$; a jogada segura
é separar em coordenadas reais. Com $z = x + \iu y$:
\[
z + 2\conj z = 3x - \iu y ,
\]
de modo que a equação diz que $3x = 6$ e $-y = 2$: a única solução
é $z = 2 - 2\iu$. (Verificação: $(2 - 2\iu) + 2(2 + 2\iu) = 6 +
2\iu$.) Alternativamente, conjugue a equação inteira para obter
$\conj z + 2z = 6 - 2\iu$ e resolva o sistema linear nas
incógnitas $z, \conj z$ --- mesma resposta, e um truque útil quando
os coeficientes são complexos. Equações em $z$ e $\conj z$ são
realmente sistemas de duas equações reais; esperar ``grau $1$, uma
solução'' é seguro aqui, mas $z\conj z = -1$ (sem solução) mostra
a intuição polinomial falhando assim que aparecem produtos.
\end{example}

\section{Forma exponencial}

\begin{definition}[Exponencial complexa de argumento imaginário]\label{def:b1:complex:expi}
Para $\theta \in \R$ define-se
\[
\eu^{\iu\theta} = \cos\theta + \iu \sin\theta .
\]
Todo $z \neq 0$ pode ser escrito como $z = r\,\eu^{\iu\theta}$ com $r =
\abs z > 0$; $\theta$ é um \emph{argumento}\index{argumento} de $z$,
determinado a menos de um múltiplo de $2\pi$. O valor em
$\intoc{-\pi}{\pi}$ é o \emph{argumento principal}, escrito
$\arg z$.
\end{definition}

\begin{example}[Primeiros valores, e uma identidade famosa]\label{ex:b1:complex:eulerid}
Lendo a definição nos ângulos cardeais:
\[
\eu^{\iu\pi/2} = \iu, \qquad
\eu^{\iu\pi} = -1, \qquad
\eu^{2\iu\pi} = 1, \qquad
\eu^{\iu\pi/4} = \frac{\sqrt2}2\,(1 + \iu) .
\]
A segunda, reorganizada como $\eu^{\iu\pi} + 1 = 0$, é a célebre
identidade de Euler, que liga $\eu$, $\iu$, $\pi$, $1$ e $0$; nesta
altura do livro ela é uma definição sendo desenrolada, e não um
teorema, e o seu verdadeiro conteúdo --- por que a função exponencial
analítica do \cref{ch:b1:functions}, estendida a argumentos
imaginários, merece o mesmo nome --- fica resolvido pelas séries de
potências do \cref{ch:b1:series}. Enquanto isso, vale memorizar o
destaque acima como tabela de conversão: ele é usado silenciosamente
sempre que um argumento é lido numa figura.
\end{example}

\begin{theorem}[A equação funcional]\label{thm:b1:complex:funceq}
Para todos $\theta, \varphi \in \R$:
\[
\eu^{\iu\theta}\, \eu^{\iu\varphi} = \eu^{\iu(\theta + \varphi)},
\qquad
\abs{\eu^{\iu\theta}} = 1,
\qquad
\conj{\eu^{\iu\theta}} = \eu^{-\iu\theta} = (\eu^{\iu\theta})^{-1}.
\]
Consequentemente, $\abs{zw} = \abs z \abs w$ transporta também os argumentos:
$\arg(zw) \equiv \arg z + \arg w \pmod{2\pi}$.
\end{theorem}

\begin{proof}
Expanda o produto e use as fórmulas de adição:
\[
(\cos\theta + \iu\sin\theta)(\cos\varphi + \iu\sin\varphi)
= (\cos\theta\cos\varphi - \sin\theta\sin\varphi)
+ \iu\,(\sin\theta\cos\varphi + \cos\theta\sin\varphi),
\]
que é $\cos(\theta+\varphi) + \iu\sin(\theta+\varphi)$. O \omterm{def:b1:complex:field}{módulo}
é $\sqrt{\cos^2\theta + \sin^2\theta} = 1$, e a fórmula do \omterm{def:b1:complex:field}{conjugado}
é a paridade do cosseno e do seno; ela inverte $\eu^{\iu\theta}$, pois
$\eu^{\iu\theta}\eu^{-\iu\theta} = \eu^0 = 1$.
\end{proof}

\begin{corollary}[Fórmula de De Moivre]\label{cor:b1:complex:demoivre}
Para $\theta \in \R$ e $n \in \Z$:
$\;(\cos\theta + \iu\sin\theta)^n = \cos(n\theta) +
\iu\sin(n\theta)$.\index{fórmula de De Moivre}
\end{corollary}

\begin{proof}
Para $n \geq 0$, faça indução. O caso $n = 0$ diz $1 = 1$. Supondo
a fórmula para $n$, a equação funcional
(\cref{thm:b1:complex:funceq}) dá
\[
\bigl(\eu^{\iu\theta}\bigr)^{n+1}
= \bigl(\eu^{\iu\theta}\bigr)^{n}\,\eu^{\iu\theta}
= \eu^{\iu n\theta}\,\eu^{\iu\theta}
= \eu^{\iu(n+1)\theta} ,
\]
que é a fórmula no posto $n + 1$. Para $n < 0$, escreva $n = -m$
com $m > 0$: como $\eu^{\iu m\theta}$ tem inverso
$\eu^{-\iu m\theta}$ (\cref{thm:b1:complex:funceq}, novamente),
\[
\bigl(\eu^{\iu\theta}\bigr)^{-m}
= \bigl(\eu^{\iu m\theta}\bigr)^{-1}
= \eu^{\iu(-m)\theta} . \qedhere
\]
\end{proof}

\begin{example}[Expandindo $\cos 3\theta$ por De Moivre]\label{ex:b1:complex:cos3}
Escreva $c = \cos\theta$, $s = \sin\theta$. De Moivre e o teorema
binomial dão
\[
\cos 3\theta + \iu \sin 3\theta = (c + \iu s)^3
= c^3 - 3cs^2 + \iu\,(3c^2 s - s^3),
\]
e, identificando as partes reais e substituindo depois $s^2 = 1 - c^2$:
\[
\cos 3\theta = c^3 - 3c(1 - c^2) = 4\cos^3\theta - 3\cos\theta .
\]
A parte imaginária fornece $\sin 3\theta = 3\sin\theta -
4\sin^3\theta$ de graça: uma identidade complexa carrega sempre
\emph{duas} identidades reais. Lida ao contrário, a identidade em
destaque é a chave da clássica equação da trissecção: construir
$\cos(\theta/3)$ a partir de $\cos\theta$ significa resolver a cúbica $4x^3 -
3x = \cos\theta$, e é aí que a álgebra do
\cref{ch:b1:poly} assume o comando.
\end{example}

\begin{proposition}[Fórmulas de Euler]\label{prop:b1:complex:euler}
\[
\cos\theta = \frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2},
\qquad
\sin\theta = \frac{\eu^{\iu\theta} - \eu^{-\iu\theta}}{2\iu}.
\]
\end{proposition}

\begin{proof}
Some e, respectivamente, subtraia $\eu^{\iu\theta} = \cos\theta +
\iu\sin\theta$ e $\eu^{-\iu\theta} = \cos\theta - \iu\sin\theta$.
\end{proof}

\begin{method}[Trigonometria via exponenciais]\label{met:b1:complex:trig}
\begin{enumerate}
  \item \emph{Linearizar} $\cos^p\theta \sin^q\theta$ (converter potências numa
        soma de $\cos k\theta$, $\sin k\theta$): substitua as fórmulas de
        Euler, expanda com o teorema binomial
        (\cref{thm:b1:counting:binomial}) e reagrupe os termos \omterm{def:b1:complex:field}{conjugados}.
  \item \emph{Expandir} $\cos n\theta$ como polinômio em $\cos\theta$:
        escreva $\cos n\theta = \Re\bigl((\cos\theta +
        \iu\sin\theta)^n\bigr)$, expanda e converta as potências pares de
        $\sin$ por $\sin^2 = 1 - \cos^2$.
  \item \emph{Somar séries trigonométricas} como $\sum_k \cos k\theta$:
        reconheça a parte real de uma soma geométrica $\sum_k
        (\eu^{\iu\theta})^k$.
  \item A \emph{fatoração do arco metade}: para todos $p, q$,
        \[
        \eu^{\iu p} + \eu^{\iu q}
        = 2 \cos\tfrac{p - q}{2}\; \eu^{\iu \frac{p+q}{2}},
        \qquad
        \eu^{\iu p} - \eu^{\iu q}
        = 2\iu \sin\tfrac{p - q}{2}\; \eu^{\iu \frac{p+q}{2}} .
        \]
\end{enumerate}
\end{method}

\begin{example}[Linearização]\label{ex:b1:complex:linearize}
\[
\cos^3\theta
= \Bigl(\frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}\Bigr)^{\!3}
= \frac{\eu^{3\iu\theta} + 3\eu^{\iu\theta} + 3\eu^{-\iu\theta} +
\eu^{-3\iu\theta}}{8}
= \frac{\cos 3\theta + 3\cos\theta}{4}.
\]
Essa forma se integra imediatamente --- a razão exata pela qual a
linearização importa no \cref{ch:b1:integration}. Um produto misto funciona
do mesmo modo, apenas com as duas fórmulas de Euler ao mesmo tempo:
\[
\sin^2\theta\cos^2\theta
= \Bigl(\frac{\sin2\theta}2\Bigr)^{\!2}
= \frac{1}{4}\cdot
\Bigl(\frac{\eu^{2\iu\theta} - \eu^{-2\iu\theta}}{2\iu}
\Bigr)^{\!2}
= \frac{2 - \eu^{4\iu\theta} - \eu^{-4\iu\theta}}{16}
= \frac{1 - \cos4\theta}{8} ,
\]
em que o atalho do arco duplo, no primeiro passo, poupou uma
expansão binomial --- sempre vale procurar por ele antes de
mecanizar.
\end{example}

\begin{example}[Uma soma trigonométrica binomial]\label{ex:b1:complex:binomsum}
Para $n \in \N$ e $\theta \in \R$, avalie $S =
\sum_{k=0}^{n}\binom nk \cos k\theta$. Reconheça a parte real de
uma expansão binomial:
\[
S = \Re\sum_{k=0}^n \binom nk \bigl(\eu^{\iu\theta}\bigr)^k
= \Re\bigl(1 + \eu^{\iu\theta}\bigr)^n ,
\]
e depois fatore o arco metade (\cref{met:b1:complex:trig}~(4)): $1 +
\eu^{\iu\theta} = 2\cos\frac\theta2\,\eu^{\iu\theta/2}$, de modo que
\[
S = \Re\Bigl(2^n\cos^n\frac\theta2\;\eu^{\iu n\theta/2}\Bigr)
= 2^n \cos^n\frac\theta2\,\cos\frac{n\theta}2 .
\]
A parte imaginária fornece $\sum_k\binom nk\sin k\theta =
2^n\cos^n\frac\theta2\sin\frac{n\theta}2$ de graça. Verificações de sanidade:
$\theta = 0$ recupera $\sum\binom nk = 2^n$, e $\theta = \pi$
dá $S = 0$ para $n \geq 1$ (cada fator $\cos\frac\pi2$
se anula), isto é, a soma alternada de uma linha do
\cref{ex:b1:counting:binomial}. O método --- ``ver a soma de cossenos
como a sombra de uma potência complexa e depois fatorar arcos metade''
--- é exatamente o do \cref{exo:b1:complex:6}, com o teorema binomial
no lugar da série geométrica.
\end{example}

\section{Raízes de números complexos}

\begin{theorem}[Raízes $n$-ésimas]\label{thm:b1:complex:roots}
Sejam $a = r\,\eu^{\iu\alpha} \neq 0$ e $n \in \N^*$. A equação
$z^n = a$ tem exatamente $n$ soluções:
\[
z_k = r^{1/n}\, \eu^{\iu\left(\frac{\alpha}{n} +
\frac{2k\pi}{n}\right)},
\qquad k = 0, 1, \dots, n - 1 .
\]
\end{theorem}

\begin{proof}
Escreva $z = \rho\,\eu^{\iu\theta}$ ($\rho > 0$). Então $z^n =
\rho^n \eu^{\iu n\theta} = r \eu^{\iu\alpha}$ se, e somente se, $\rho^n
= r$ (\omterm{def:b1:complex:field}{módulos}) e $n\theta \equiv \alpha \pmod{2\pi}$ (argumentos),
isto é, $\rho = r^{1/n}$ e $\theta = \frac{\alpha}{n} +
\frac{2k\pi}{n}$ para algum $k \in \Z$. Resta ver quando dois
inteiros $k, k'$ dão o mesmo número: isso acontece exatamente quando
os ângulos diferem por um múltiplo de $2\pi$,
\[
\frac{2k\pi}n - \frac{2k'\pi}n \in 2\pi\Z
\iff \frac{k - k'}n \in \Z
\iff n \mid k - k' .
\]
Pela divisão euclidiana, todo $k \in \Z$ é congruente módulo $n$ a
exatamente um elemento de $\{0, 1, \dots, n-1\}$, de modo que esse intervalo lista
cada solução uma única vez e a contagem é exatamente $n$. (O mesmo
argumento, feito dentro de $\mathbb U_n$, mostra que as raízes formam um
$n$-ágono regular: valores consecutivos de $k$ rodam pelo ângulo fixo
$\frac{2\pi}n$.)
\end{proof}

\begin{example}[Raízes cúbicas de $-27$]\label{ex:b1:complex:cuberoots}
Resolva $z^3 = -27$. Forma exponencial do lado direito: $-27 =
27\,\eu^{\iu\pi}$, de modo que as três raízes são
\[
z_k = 3\,\eu^{\iu(\frac\pi3 + \frac{2k\pi}3)}, \quad k = 0, 1, 2 :
\qquad
z_0 = 3\eu^{\iu\pi/3} = \frac32 + \frac{3\sqrt3}2\,\iu,
\quad
z_1 = -3,
\quad
z_2 = \conj{z_0} .
\]
Duas verificações. Primeiro, a raiz real $-3$ é a evidente, e as
outras duas são suas rotações de $\pm\frac{2\pi}3$ --- equivalentemente,
$-3j$ e $-3j^2$. Segundo, a álgebra confirma: $z^3 + 27 = (z +
3)(z^2 - 3z + 9)$, e a equação do segundo grau tem discriminante $9 - 36 =
-27 < 0$ com raízes $\frac{3 \pm 3\iu\sqrt3}2 = z_0, \conj{z_0}$.
A ideia: para lados direitos reais, as raízes não reais vêm sempre
em pares \omterm{def:b1:complex:field}{conjugados}, de modo que uma figura do \omterm{def:b1:logic:sets}{conjunto} solução é
simétrica em relação ao eixo real --- uma antecipação do teorema de
fatoração real do \cref{ch:b1:poly}.
\end{example}

\begin{example}[Um lado direito não real]\label{ex:b1:complex:fourthroots}
Resolva $z^4 = -8 + 8\iu\sqrt3$. Forma exponencial do lado direito:
módulo $\sqrt{64 + 192} = 16$, argumento $\theta$ com
$\cos\theta = -\frac12$, $\sin\theta = \frac{\sqrt3}2$, isto é,
$\theta = \frac{2\pi}3$. As quatro raízes são
\[
z_k = 2\,\eu^{\iu(\frac\pi6 + \frac{k\pi}2)}, \quad k = 0, 1, 2,
3 :
\qquad
z_0 = \sqrt3 + \iu,\quad z_1 = \iu z_0 = -1 + \iu\sqrt3,
\]
\[
z_2 = -z_0 = -\sqrt3 - \iu,\qquad z_3 = -\iu z_0 = 1 - \iu\sqrt3 .
\]
(Verificação: $z_0^2 = 2 + 2\iu\sqrt3$, logo $z_0^4 = (2 +
2\iu\sqrt3)^2 = 4 - 12 + 8\iu\sqrt3 = -8 + 8\iu\sqrt3$.) Uma vez encontrada
\emph{uma} raiz, as outras três saem de graça: são as suas
rotações sucessivas de $\frac\pi2$, isto é, os seus produtos pelas
raízes quartas da unidade --- a estrutura geral por trás do
\cref{thm:b1:complex:roots}, que vale explorar antes de recalcular
cada raiz do zero. Desta vez não há simetria conjugada: o
lado direito não é real.
\end{example}

\begin{definition}[Raízes da unidade]\label{def:b1:complex:unity}
As \emph{raízes $n$-ésimas da unidade}\index{raízes da unidade} são as
soluções de $z^n = 1$:
\[
\mathbb{U}_n = \bigl\{\, \omega^k : k = 0, \dots, n-1 \,\bigr\},
\qquad \omega = \eu^{2\iu\pi/n}.
\]
Elas formam um grupo com a multiplicação (\cref{ch:b1:structures}) e
ocupam os vértices de um $n$-ágono regular inscrito no círculo unitário.
\end{definition}

\begin{proposition}[Soma das raízes da unidade]\label{prop:b1:complex:sumroots}
Para $n \geq 2$, as raízes $n$-ésimas da unidade somam zero:
$\sum_{k=0}^{n-1} \omega^k = 0$.
\end{proposition}

\begin{proof}
Soma geométrica de razão $\omega \neq 1$:
$\sum_{k=0}^{n-1} \omega^k = \frac{\omega^n - 1}{\omega - 1} = 0$,
pois $\omega^n = 1$.
\end{proof}

\begin{example}[Lendo as partes real e imaginária]\label{ex:b1:complex:sumrootsparts}
Separar $\sum_{k=0}^{n-1}\omega^k = 0$ em partes real e imaginária
fornece duas identidades trigonométricas de graça:
\[
\sum_{k=0}^{n-1}\cos\frac{2k\pi}n = 0,
\qquad
\sum_{k=0}^{n-1}\sin\frac{2k\pi}n = 0
\qquad (n \geq 2).
\]
Geometricamente: o baricentro de um $n$-ágono regular inscrito no
círculo unitário é o seu centro --- os vértices se equilibram
exatamente. Para $n = 5$, a primeira identidade dá $1 + 2\cos\frac{2\pi}5 +
2\cos\frac{4\pi}5 = 0$ (emparelhando $k$ com $n - k$), ponto de partida do
cálculo de $\cos\frac{2\pi}5$ feito no \cref{exo:b1:complex:8}.
\end{example}

\begin{example}[Raízes quadradas na forma algébrica]\label{ex:b1:complex:sqrt}
Para resolver $z^2 = 3 + 4\iu$ sem trigonometria, ponha $z = x + \iu y$:
\[
x^2 - y^2 = 3, \qquad 2xy = 4, \qquad x^2 + y^2 = \abs{3 + 4\iu} = 5 .
\]
Somando a primeira e a última: $x^2 = 4$, logo $x = \pm 2$, e então $y = 2/x =
\pm 1$ com o \emph{mesmo} pareamento de sinais ($xy = 2 > 0$): $z = \pm(2 +
\iu)$. Combinado com a fórmula usual, isso resolve toda equação do
segundo grau com coeficientes complexos
(\cref{exo:b1:complex:7}).
\end{example}

\section{Números complexos e geometria plana}

\begin{proposition}[Dicionário geométrico]\label{prop:b1:complex:geometry}
Identifique o ponto $M(x, y)$ do plano com o seu \emph{afixo} $z = x
+ \iu y$\index{afixo}. Para pontos distintos $A, B, C$ de afixos $a, b,
c$:
\begin{enumerate}
  \item $\abs{b - a}$ é a distância $AB$;
  \item $\arg\dfrac{c - a}{b - a}$ é o ângulo entre os vetores
        $\vect{AB}$ e $\vect{AC}$ (mód.\ $2\pi$);
  \item $A, B, C$ são colineares se, e somente se, $\dfrac{c - a}{b - a} \in
        \R$; as retas $AB$ e $AC$ são perpendiculares se, e somente se,
        $\dfrac{c - a}{b - a} \in \iu\R$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) é a definição do \omterm{def:b1:complex:field}{módulo} aplicada a $b - a$, o afixo de
$\vect{AB}$. (2): escreva $b - a = r\eu^{\iu\theta}$, $c - a =
s\eu^{\iu\varphi}$; então $\frac{c-a}{b-a} = \frac sr
\eu^{\iu(\varphi - \theta)}$ tem argumento $\varphi - \theta$, o
ângulo de $\vect{AB}$ a $\vect{AC}$. (3): a colinearidade significa ângulo $0$
ou $\pi$, isto é, argumento em $\pi\Z$, isto é, o quociente é real; a
perpendicularidade significa ângulo $\pm\frac\pi2$, isto é, o quociente é
puramente imaginário. (O quociente é não nulo, pois $C \neq A$.)
\end{proof}

\begin{remark}[Interlúdio: $\C$ é o plano com uma multiplicação]\label{rem:b1:complex:interlude}
Vale a pena parar para pensar no que torna este capítulo possível:
o plano $\R^2$ carrega adições em todas as direções, mas nenhuma
multiplicação dada de antemão --- e $\C$ \emph{é} o plano
equipado com uma, na qual multiplicar por um número fixo
gira e dilata. Essa única estrutura será explorada mais três
vezes neste volume. No \cref{ch:b1:matrices}, a multiplicação por
$a + \iu b$ reaparece como a matriz $2 \times 2$ de linhas $(a,
-b)$ e $(b, a)$: a aritmética complexa é uma primeira família,
inteiramente concreta, de produtos de matrizes. No \cref{ch:b1:euclid}, a
fórmula $\Re(\conj z\,w)$ revela-se o produto interno, e
$\abs z$ a norma euclidiana: a desigualdade triangular demonstrada aqui
é o modelo da história de Cauchy--Schwarz de lá. E no
\cref{ch:b1:curves}, um ponto em movimento é mais bem escrito como $t \mapsto
z(t)$, de modo que a velocidade e a aceleração se tornam
derivadas de valores complexos --- o movimento circular, por exemplo,
é simplesmente $z(t) =
R\,\eu^{\iu\omega t}$. Uma boa multiplicação, quatro
capítulos de dividendos.
\end{remark}

\begin{proposition}[Aplicações $z \mapsto az + b$]\label{prop:b1:complex:similitude}
Sejam $a \in \C^*$, $b \in \C$. A transformação $f(z) = az + b$ do
plano:
\begin{itemize}
  \item é uma translação se $a = 1$;
  \item caso contrário, tem um único ponto fixo $\zeta = \frac{b}{1-a}$,
        e $f(z) - \zeta = a\,(z - \zeta)$: $f$ é a rotação de
        centro $\zeta$ e ângulo $\arg a$, composta com a homotetia
        de centro $\zeta$ e razão $\abs a$.
\end{itemize}
Em particular, $z \mapsto \eu^{\iu\theta} z$ é a rotação de ângulo
$\theta$ em torno da origem, e $\conj z$ é a reflexão no eixo real.
\end{proposition}

\begin{proof}
Se $a = 1$, $f(z) = z + b$ translada pelo vetor de afixo $b$. Se
$a \neq 1$, a equação de ponto fixo $z = az + b$ tem a única
solução $\zeta = \frac{b}{1-a}$, e então $f(z) - \zeta = az + b -
(a\zeta + b) = a(z - \zeta)$. Escrevendo $a = \abs a\, \eu^{\iu\arg a}$,
a multiplicação por $a$ dilata as distâncias a $\zeta$ pelo fator
$\abs a$ e soma $\arg a$ aos ângulos em $\zeta$, o que é a composição
anunciada.
\end{proof}

\begin{example}[Classificando uma aplicação $z \mapsto az+b$]\label{ex:b1:complex:classify}
Tome $f(z) = \iu z + 1$. Aqui $a = \iu \neq 1$: o ponto fixo é
\[
\zeta = \frac{b}{1 - a} = \frac1{1 - \iu}
= \frac{1 + \iu}{2},
\]
e, como $\abs a = 1$, não há dilatação alguma: $f$ é a rotação
pura de centro $\frac{1+\iu}2$ e ângulo $\arg \iu = \frac\pi2$.
Verificação: $f(\zeta) = \iu\,\frac{1+\iu}2 + 1 = \frac{\iu - 1}2 + 1 =
\frac{1 + \iu}2 = \zeta$, e $f(0) = 1$, $f(1) = 1 + \iu$, $f(1 +
\iu) = \iu\,(1 + \iu) + 1 = \iu$: os quatro pontos $0, 1, 1+\iu, \iu$
do quadrado unitário giram em torno do seu centro $\zeta$, um quarto
de volta de cada vez --- exatamente o que uma rotação de $\frac\pi2$ em
torno do centro do quadrado deve fazer.
\end{example}

\begin{remark}[Armadilhas frequentes com módulos e argumentos]\label{rem:b1:complex:pitfalls}
\begin{enumerate}
  \item \emph{Não há desigualdades em $\C$.} Escrever $z \leq w$ para
        números não reais não faz sentido; só \omterm{def:b1:complex:field}{módulos}, partes reais
        e partes imaginárias podem ser comparados.
  \item \emph{$\sqrt{\phantom z}$ está reservado a $\R_+$.} Todo
        número complexo não nulo tem \emph{duas} raízes quadradas e
        nenhuma delas é privilegiada: escreva ``seja $\delta$ uma raiz
        quadrada de $\Delta$'' (calculada como no
        \cref{ex:b1:complex:sqrt}), nunca $\sqrt\Delta$ ---
        a regra $\sqrt{ab} = \sqrt a\sqrt b$ já falha em
        $a = b = -1$.
  \item \emph{Os argumentos vivem módulo $2\pi$.} De
        $\eu^{\iu\alpha} = \eu^{\iu\beta}$ conclua $\alpha
        \equiv \beta \pmod{2\pi}$, e não $\alpha = \beta$;
        esquecer o $2k\pi$ é como os \omterm{def:b1:logic:sets}{conjuntos} solução de $z^n = a$
        perdem $n - 1$ dos seus $n$ elementos.
  \item \emph{$\abs{z + w}$ não é $\abs z + \abs w$.} A igualdade
        na desigualdade triangular é o caso excepcional de
        alinhamento (\cref{prop:b1:complex:rules}~(4)); em geral, o
        \omterm{def:b1:complex:field}{módulo} de uma soma deve ser estimado, e não calculado.
\end{enumerate}
\end{remark}

\begin{example}\label{ex:b1:complex:equilateral}
$A, B, C$ (de afixos $a, b, c$, dois a dois distintos) formam um
triângulo equilátero com os vértices em ordem direta (anti-horária) se, e
somente se, $\frac{c - a}{b - a} = \eu^{\iu\pi/3}$: a rotação de
centro $A$ e ângulo $\frac\pi3$ leva $B$ em $C$. As duas orientações
juntas ficam captadas pela equação simétrica $a^2 + b^2 + c^2 = ab
+ bc + ca$ (\cref{exo:b1:complex:10}).
\end{example}

\begin{remark}[Onde este capítulo é usado]\label{rem:b1:complex:whereused}
A forma exponencial é o recurso de cálculo mais reutilizado do
volume. As \omterm{def:b1:complex:unity}{raízes da unidade} tornam-se o exemplo padrão de grupo
cíclico no \cref{ch:b1:structures} e conduzem a fatoração
de $X^n - 1$ no \cref{ch:b1:poly}; agrupar ali as raízes conjugadas
produz as fatorações reais usadas pelas frações parciais no
\cref{ch:b1:fractions}. A linearização (\cref{met:b1:complex:trig})
é a preparação usual para integrar potências trigonométricas no
\cref{ch:b1:integration}, e as equações características do
\cref{ch:b1:diffeq} têm raízes complexas cujas partes real e imaginária
produzem as soluções oscilantes $\eu^{\lambda t}\cos\omega
t$. O dicionário
geométrico volta em roupagem matricial: rotações e semelhanças tornam-se
as matrizes ortogonais do \cref{ch:b1:matrices,ch:b1:euclid}, e as curvas parametrizadas do
\cref{ch:b1:curves} são muitas vezes mais bem escritas como aplicações $t \mapsto z(t)
\in \C$.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:complex:1}
Ponha na forma exponencial: $1 + \iu$; $\;\sqrt 3 - \iu$; $\;-5$;
$\;\dfrac{1 + \iu}{\sqrt 3 - \iu}$. Deduza
$\cos\frac{5\pi}{12}$ e $\sin\frac{5\pi}{12}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:2}
Calcule $(1 + \iu)^{20}$. Para quais $n \in \N$ o número $(1 + \iu)^n$ é
real?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:3}
Descreva geometricamente o \omterm{def:b1:logic:sets}{conjunto} dos $z \in \C$ tais que:
$\abs{z - 2} = \abs{z + \iu}$; $\;\abs{z - 1} = 2$;
$\;\dfrac{z - 1}{z + 1} \in \iu\R$ (para $z \neq -1$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:4}
Linearize $\sin^4\theta$ e expanda $\cos 4\theta$ como polinômio em
$\cos\theta$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:5}
Resolva $z^3 = 8\iu$ e situe as soluções numa figura. Resolva
$z^4 = -4$ e fatore $X^4 + 4$ em dois polinômios reais do segundo grau.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:6}
Para $\theta \in \R$ com $\eu^{\iu\theta} \neq 1$ e $n \in \N$,
calcule
\[
C_n = \sum_{k=0}^{n} \cos k\theta
\qquad\text{e}\qquad
S_n = \sum_{k=0}^{n} \sin k\theta
\]
somando a série geométrica $\sum_k \eu^{\iu k\theta}$ e usando
a fatoração do arco metade.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:7}
Resolva em $\C$: $z^2 - (3 + 4\iu) z + (-1 + 5\iu) = 0$. \emph{(Calcule
o discriminante e extraia suas raízes quadradas como no
\cref{ex:b1:complex:sqrt}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:8}
Seja $\omega = \eu^{2\iu\pi/5}$.
\begin{enumerate}
  \item Justifique que $1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$.
  \item Ponha $u = \omega + \omega^4$ e $v = \omega^2 + \omega^3$.
        Calcule $u + v$ e $uv$, e deduza que $u$ e $v$ são as
        raízes de $X^2 + X - 1$.
  \item Conclua que $\cos\frac{2\pi}{5} = \frac{\sqrt 5 - 1}{4}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:9}
Demonstre que, para todos $z, w \in \C$ (identidade do paralelogramo):
\[
\abs{z + w}^2 + \abs{z - w}^2 = 2\abs z^2 + 2\abs w^2 ,
\]
e interprete-a geometricamente no paralelogramo de vértices $0,
z, w, z + w$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:complex:10}
Demonstre que três pontos dois a dois distintos, de afixos $a, b, c$,
formam um triângulo equilátero (em qualquer orientação) se, e somente se,
\[
a^2 + b^2 + c^2 = ab + bc + ca .
\]
\emph{Sugestão: o caso direto é $\frac{c-a}{b-a} = -j^2$ e o
caso indireto $\frac{c-a}{b-a} = -j$, em que $j = \eu^{2\iu\pi/3}$
satisfaz $j^2 + j + 1 = 0$; ou fatore $a + jb + j^2c$ e $a + j^2 b
+ jc$.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:complex:11}
Para $n \in \N^*$, avalie $P = \prod_{k=1}^{n-1}
\bigl(1 - \omega^k\bigr)$, em que $\omega = \eu^{2\iu\pi/n}$.
\emph{Sugestão: $X^n - 1 = \prod_{k=0}^{n-1} (X - \omega^k)$; divida por
$X - 1$ e avalie em $X = 1$.} Deduza que
$\prod_{k=1}^{n-1} \sin\frac{k\pi}{n} = \dfrac{n}{2^{n-1}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:12}
Seja $n \geq 2$. Resolva a equação $(z + 1)^n = (z - 1)^n$ em $\C$:
mostre que ela tem exatamente $n - 1$ soluções, todas puramente
imaginárias, a saber,
\[
z_k = -\iu\,\frac{\cos\frac{k\pi}{n}}{\sin\frac{k\pi}{n}},
\qquad k = 1, \dots, n - 1 .
\]
\emph{Sugestão: $z = 1$ não é solução; divida e use as \omterm{def:b1:complex:unity}{raízes da
unidade}; depois aplique a fatoração do arco metade do
\cref{met:b1:complex:trig}.}
\end{exercise}

\section{Problema: O teorema de Napoleão}

\begin{problem}\label{pb:b1:complex:1}
Erga um triângulo equilátero para fora sobre cada lado de um triângulo
\emph{arbitrário}: os três centros desses triângulos formam sempre um
triângulo equilátero. Este é o teorema de Napoleão
\index{teorema de Napoleão} --- um enunciado sem razão visível para
ser verdadeiro, que a álgebra de $j = \eu^{2\iu\pi/3}$ demonstra em três
linhas de cálculo. Este problema constrói todo o instrumental
(rotações, semelhanças diretas, o critério de $j$ do
\cref{exo:b1:complex:10}), demonstra o teorema de Napoleão nas versões interna
e externa, localiza o caso degenerado e se encerra com uma segunda joia
da mesma escola: o teorema de van Schooten sobre triângulos equiláteros
e a desigualdade de Ptolomeu. Ao longo do problema, os pontos do plano
são identificados com os seus afixos.

\medskip
\noindent\textbf{Parte I --- O número $j$ e as rotações.}
\begin{enumerate}
  \item Calcule a forma algébrica de $j$, depois $j^2$, $j^3$, $1 + j
        + j^2$, $\conj j$ e $j^{-1}$. Situe $1$, $j$, $j^2$ num
        esboço do círculo unitário.
  \item Mostre que a rotação de centro $a$ e ângulo $\theta$ é
        $r(z) = a + \eu^{\iu\theta}(z - a)$, e que a
        composta de duas rotações, de ângulos $\theta$ e
        $\theta'$, é uma rotação de ângulo $\theta + \theta'$ se
        $\theta + \theta' \notin 2\pi\Z$, e uma translação
        caso contrário. (Use a \cref{prop:b1:complex:similitude}.)
  \item Sejam $b \neq c$. Mostre que existem exatamente dois pontos $p$
        que tornam $(b, c, p)$ equilátero, a saber, $p = b +
        \eu^{\pm\iu\pi/3}(c - b)$. Para um triângulo \emph{direto}
        $(a, b, c)$, verifique no exemplo $a = 0$, $b = 1$, $c =
        \iu$ que a escolha $\eu^{-\iu\pi/3}$ é a que fica do
        lado da reta $BC$ oposto a $a$ --- o ápice
        \emph{externo}.
  \item Pela solução do \cref{exo:b1:complex:10}, $(a, b, c)$
        é equilátero direto se, e somente se, $a + jb + j^2c = 0$. Demonstre as duas
        identidades de rotação
        \[
        b + jc + j^2a = j^2\,(a + jb + j^2c),
        \qquad
        c + ja + j^2b = j\,(a + jb + j^2c),
        \]
        e deduza que o critério é invariante por \omterm{def:b1:counting:objects}{permutação}
        cíclica de $(a, b, c)$.
  \item Mostre que, se $a + jb + j^2c = 0$ e dois dos três pontos
        coincidem, então os três coincidem. (Assim, o critério
        caracteriza exatamente: triângulo equilátero direto, ou um
        único ponto.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Semelhanças diretas.}
\begin{enumerate}
  \item Mostre que as aplicações $f(z) = az + b$ com $a \in \C^*$
        (\emph{semelhanças diretas}) são estáveis por composição
        e por inversão: a composta de duas delas, e a
        inversa de qualquer uma delas, é de novo dessa forma. (Na
        linguagem do \cref{ch:b1:structures}: elas formam um grupo.)
  \item Dados $z_1 \neq z_2$ e $w_1 \neq w_2$, mostre que existe uma
        \emph{única} semelhança direta $f$ com $f(z_1) = w_1$ e
        $f(z_2) = w_2$, e dê $a$ e $b$ explicitamente.
  \item Mostre que uma semelhança direta $f(z) = az + b$ multiplica
        todas as distâncias por $\abs a$ e preserva a \emph{forma}
        $\frac{c - a'}{b' - a'}$ de qualquer triângulo $(a', b', c')$
        \emph{(numerador e denominador são ambos multiplicados por
        $a$)}. Deduza que dois triângulos são diretamente semelhantes
        exatamente quando as suas formas são iguais.
  \item Determine completamente a semelhança direta com $f(0) = 1$
        e $f(1) = \iu$: dê $a$, $b$, o ponto fixo, a razão
        e o ângulo.
  \item Seja $\alpha + \beta = 1$. Mostre que o ``ponto ponderado''
        $g(a', b') = \alpha a' + \beta b'$ comuta com toda
        semelhança direta: $f(\alpha a' + \beta b') = \alpha f(a')
        + \beta f(b')$. Deduza que baricentros, pontos médios e os
        centros de Napoleão abaixo são todos transportados pelas
        semelhanças --- a licença algébrica por trás de todo argumento
        do tipo ``sem perda de generalidade, coloque o círculo
        circunscrito no círculo unitário''.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O teorema de Napoleão.} Seja $(a, b, c)$
um triângulo direto. Sobre cada lado, erga o triângulo equilátero
externo (questão 3) e sejam $n_a$, $n_b$, $n_c$ os centros
(baricentros) dos triângulos erguidos sobre $[b, c]$, $[c, a]$ e
$[a, b]$, respectivamente.
\begin{enumerate}
  \item Mostre que
        \[
        n_a = \alpha b + \beta c, \qquad
        n_b = \alpha c + \beta a, \qquad
        n_c = \alpha a + \beta b,
        \qquad\text{com}\quad
        \alpha = \frac{3 + \iu\sqrt3}{6},\ \beta = \conj\alpha .
        \]
  \item Verifique que $\alpha + \beta = 1$ e deduza que o triângulo
        $(n_a, n_b, n_c)$ tem o \emph{mesmo baricentro} que
        $(a, b, c)$.
  \item Usando as identidades da questão 4, mostre que
        \[
        n_a + j\,n_b + j^2 n_c
        = (\alpha j^2 + \beta j)\,(a + jb + j^2c) .
        \]
  \item Calcule $\alpha j + \beta$ e conclua: $\alpha j^2 +
        \beta j = j(\alpha j + \beta) = 0$, de modo que $n_a + j n_b + j^2
        n_c = 0$ para \emph{todo} triângulo: o triângulo
        externo de Napoleão é equilátero direto (ou um ponto). O
        teorema de Napoleão está demonstrado.
  \item Erga os triângulos equiláteros \emph{para dentro}
        (escolha $\eu^{+\iu\pi/3}$ na questão 3) e sejam $m_a, m_b,
        m_c$ os seus centros. Mostre que $m_a = \beta b + \alpha c$ (e
        ciclicamente), e depois demonstre que $m_a + j^2 m_b + j\,m_c = 0$: o
        triângulo interno de Napoleão também é equilátero, com a
        orientação oposta.
  \item Localize a degenerescência: mostre que $n_a = n_b = n_c$ acontece
        exatamente quando $a + j^2 b + jc = 0$, isto é, quando $(a, b, c)$
        é um triângulo equilátero \emph{indireto}. \emph{(Dois
        pontos entre $n_a, n_b, n_c$ coincidem se, e somente se, os três
        coincidem, pela questão 5; depois use $n_a + j^2 n_b + j n_c = j(\alpha +
        \beta j)(a + j^2b + jc)$ e verifique que $\alpha + \beta j \neq
        0$.)}
  \item Faça o cálculo completo no triângulo $a = 0$,
        $b = 1$, $c = \iu$: dê $n_a, n_b, n_c$ exatamente e
        verifique, pelo cálculo direto dos três quadrados dos
        comprimentos dos lados, que o triângulo é equilátero, de lado
        ao quadrado $\frac{2 + \sqrt3}{3}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Ptolomeu e van Schooten.}
\begin{enumerate}
  \item Demonstre a identidade, válida para todos os complexos $a, b, c, d$:
        \[
        (a - b)(c - d) + (a - d)(b - c) = (a - c)(b - d) .
        \]
  \item Deduza a \emph{desigualdade de Ptolomeu}\index{desigualdade de
        Ptolomeu}: para quaisquer quatro pontos $A, B, C, D$,
        \[
        AC \cdot BD \;\leq\; AB \cdot CD + AD \cdot BC ,
        \]
        com igualdade se, e somente se, $(a-b)(c-d)$ e
        $(a-d)(b-c)$ estão numa mesma semirreta com origem em $0$.
  \item Mostre que, para $\theta, \varphi \in \R$,
        $\abs{\eu^{\iu\theta} - \eu^{\iu\varphi}} =
        2\,\abs{\sin\frac{\theta - \varphi}2}$.
  \item (Teorema de van Schooten\index{teorema de van Schooten}) Seja
        $(a, b, c) = (1, j, j^2)$ --- pela Parte II, isso não implica perda de
        generalidade entre os triângulos equiláteros diretos --- e seja
        $p = \eu^{\iu\theta}$ com $\theta \in
        \intoo{2\pi/3}{4\pi/3}$, um ponto do círculo circunscrito sobre o
        arco $BC$ que não contém $A$. Demonstre que
        \[
        PA = PB + PC .
        \]
        \emph{(Exprima as três distâncias com a questão 20 e use
        uma fórmula de transformação de soma em produto.)}
  \item Verifique em $p = -1$ que isso é exatamente o caso de igualdade
        da desigualdade de Ptolomeu para a ordem cíclica $A, B, P, C$:
        calcule $(a - b)(p - c)$ e $(a - c)(b - p)$ e confira
        que a razão entre eles é um real positivo.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Síntese.}
\begin{enumerate}
  \item Calcule o triângulo externo de Napoleão do próprio triângulo
        equilátero $(1, j, j^2)$ e descreva o resultado
        geometricamente.
  \item Onde exatamente o problema usou: (i) a multiplicação como
        rotação; (ii) a estrutura de grupo das semelhanças e o
        invariante de forma; (iii) a fatoração do arco metade do
        \cref{met:b1:complex:trig}? Uma frase para cada.
  \item Num parágrafo curto, articule a moral do problema:
        o que o dicionário entre a geometria plana e a álgebra de $\C$
        compra, o que ele custa e qual dos dois passos da demonstração
        --- a identidade $\alpha j + \beta = 0$ ou a figura clássica ---
        \emph{explica} melhor, a seu ver, o teorema de Napoleão.
        Mencione um lugar em que o dicionário reaparecerá em forma
        matricial mais adiante neste volume.
\end{enumerate}
\end{problem}
