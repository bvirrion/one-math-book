\chapter{Estruturas Algébricas}\label{ch:b1:structures}

As mesmas regras de cálculo reaparecem sem cessar: inteiros, números
reais, números complexos, classes de \omterm{def:b1:arith:congruence}{congruência} e, em breve, polinômios
(\cref{ch:b1:poly}), vetores e matrizes
(\cref{ch:b1:vspaces,ch:b1:matrices}). A álgebra extrai os padrões comuns
e lhes dá nome: \emph{\omterm{def:b1:structures:group}{grupo}}, \emph{\omterm{def:b1:structures:ring}{anel}}, \emph{\omterm{def:b1:structures:field}{corpo}}.
Demonstrar um fato uma vez, no nível da estrutura, demonstra-o para
todos os exemplos de uma só vez.

\section{Leis de composição}

\begin{definition}\label{def:b1:structures:law}
Uma \emph{lei de composição}\index{lei de composição} num \omterm{def:b1:logic:sets}{conjunto} $E$ é
uma \omterm{def:b1:logic:map}{aplicação} $E \times E \to E$, escrita $(x, y) \mapsto x * y$. Ela é
\emph{associativa} quando $(x*y)*z = x*(y*z)$ sempre, e \emph{comutativa}
quando $x * y = y * x$ sempre. Um elemento $e$ é um \emph{neutro} quando
$e * x = x * e = x$ para todo $x$; então $x'$ é um \emph{inverso} de $x$
quando $x * x' = x' * x = e$.
\end{definition}

\begin{proposition}[Unicidade]\label{prop:b1:structures:unique}
Uma lei tem no máximo um neutro; para uma lei associativa com neutro,
cada elemento tem no máximo um inverso.
\end{proposition}

\begin{proof}
Se $e$ e $e'$ são neutros: $e = e * e' = e'$. Se $x'$ e $x''$
invertem $x$: $x' = x' * e = x' * (x * x'') = (x' * x) * x'' = e * x''
= x''$.
\end{proof}

\section{Grupos}

\begin{definition}[Grupo]\label{def:b1:structures:group}
Um \emph{grupo}\index{grupo} $(G, *)$ é um \omterm{def:b1:logic:sets}{conjunto} munido de uma lei
associativa que admite um neutro e no qual todo elemento tem um inverso.
O grupo é \emph{abeliano}\index{grupo abeliano} quando a lei é
comutativa.
\end{definition}

\begin{example}\label{ex:b1:structures:groups}
$(\Z, +)$, $(\Q, +)$, $(\R, +)$, $(\C, +)$; $(\Q^*, \times)$,
$(\R^*, \times)$, $(\C^*, \times)$, $(\mathbb{U}_n, \times)$ (\omterm{def:b1:complex:unity}{raízes da
unidade}, \cref{def:b1:complex:unity}); o \omterm{def:b1:logic:sets}{conjunto} $\mathfrak{S}(E)$ das
bijeções de um \omterm{def:b1:logic:sets}{conjunto} $E$ sobre si mesmo, com a composição --- o
\emph{grupo simétrico}\index{grupo simétrico} de $E$, não \omterm{def:b1:structures:group}{abeliano} assim
que $\abs E \geq 3$. Não são \omterm{def:b1:structures:group}{grupos}: $(\N, +)$ (sem inversos),
$(\Z, \times)$ (só $\pm 1$ é invertível).
\end{example}

\begin{proposition}[Regras de cálculo]\label{prop:b1:structures:rules}
Num \omterm{def:b1:structures:group}{grupo} $G$ (escrito multiplicativamente, com neutro $e$):
\begin{enumerate}
  \item cancelamento: $ax = ay \implies x = y$ e $xa = ya \implies
        x = y$;
  \item $(ab)^{-1} = b^{-1} a^{-1}$ e $(a^{-1})^{-1} = a$;
  \item para $a, b \in G$, cada equação $ax = b$ e $xa = b$ tem uma
        única solução ($x = a^{-1}b$, resp.\ $x = b a^{-1}$).
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Multiplique por $a^{-1}$ do lado apropriado, usando a
associatividade. (2) $(b^{-1}a^{-1})(ab) = b^{-1}(a^{-1}a)b = b^{-1}b =
e$ e simetricamente; a unicidade do inverso conclui; o segundo
ponto é a \cref{prop:b1:structures:unique} aplicada a $a^{-1}$. (3)
Substitua e use (1) para a unicidade.
\end{proof}

\begin{example}[As simetrias de um retângulo]\label{ex:b1:structures:klein}
Um retângulo (não quadrado) admite exatamente quatro isometrias sobre
si mesmo: a identidade $e$, a reflexão no eixo horizontal $h$, a
reflexão no eixo vertical $v$ e a meia-volta $r$ em torno do
centro. A composição faz desse \omterm{def:b1:logic:sets}{conjunto} de quatro elementos um \omterm{def:b1:structures:group}{grupo}:
cada elemento é o seu próprio inverso ($h^2 = v^2 = r^2 = e$), e o
produto de dois elementos distintos e diferentes do neutro é o terceiro
($hv = vh = r$: refletir nos dois eixos é a meia-volta). A
tabela completa é simétrica, de modo que o \omterm{def:b1:structures:group}{grupo} é \omterm{def:b1:structures:group}{abeliano} --- e no
entanto ele \emph{não} é o mesmo \omterm{def:b1:structures:group}{grupo} que as rotações $\mathbb U_4$ do
\cref{ex:b1:structures:order}: lá, $\iu$ tem \omterm{def:b1:structures:order}{ordem} $4$, ao passo que
aqui todo elemento tem \omterm{def:b1:structures:order}{ordem} $\leq 2$. Dois \omterm{def:b1:structures:group}{grupos} do mesmo
tamanho podem, assim, ter estruturas multiplicativas genuinamente
diferentes --- a figura abaixo exibe as duas tabelas lado a lado. Este
\omterm{def:b1:structures:group}{grupo} de quatro elementos volta como $\{\pm1\} \times \{\pm1\}$, e o
\cref{exo:b1:structures:7} explica por que todo \omterm{def:b1:structures:group}{grupo} em que os
quadrados são triviais deve, como este, ser \omterm{def:b1:structures:group}{abeliano}.
\end{example}

\begin{definition}[Subgrupo]\label{def:b1:structures:subgroup}
Um subconjunto $H$ de um \omterm{def:b1:structures:group}{grupo} $G$ é um \emph{subgrupo}\index{subgrupo}
(escreve-se $H \leq G$) quando contém $e$ e é estável pela lei
e pela inversão. Então $H$ é ele próprio um \omterm{def:b1:structures:group}{grupo}.

\emph{Critério:} um $H \subseteq G$ não vazio é um subgrupo se, e somente
se,
\[
\forall x, y \in H, \quad x y^{-1} \in H .
\]
\end{definition}

\begin{proof}[Demonstração do critério]
Um \omterm{def:b1:structures:subgroup}{subgrupo} obviamente o satisfaz. Reciprocamente, seja $H \neq \emptyset$
satisfazendo-o, e tome $x_0 \in H$. Então $e = x_0 x_0^{-1} \in H$; para
$y \in H$, $y^{-1} = e\,y^{-1} \in H$; e, para $x, y \in H$, $xy =
x (y^{-1})^{-1} \in H$.
\end{proof}

\begin{example}\label{ex:b1:structures:subgroups}
$\mathbb{U}_n \leq (\C^*, \times)$: é não vazio e, para $z, w \in
\mathbb{U}_n$, $(zw^{-1})^n = z^n (w^n)^{-1} = 1$. Os \omterm{def:b1:structures:subgroup}{subgrupos} de
$(\Z, +)$ são exatamente os $n\Z$ (demonstrado no
\cref{thm:b1:arith:gcd}). Uma interseção de \omterm{def:b1:structures:subgroup}{subgrupos} é sempre um
\omterm{def:b1:structures:subgroup}{subgrupo}, mas uma união quase nunca é
(\cref{exo:b1:structures:6}).
\end{example}

\begin{remark}[Armadilhas frequentes com estruturas]\label{rem:b1:structures:pitfalls}
\begin{enumerate}
  \item \emph{A estabilidade pela lei não basta.} $\N$ é
        estável pela adição dentro de $\Z$ e contém $0$, mas não
        é \omterm{def:b1:structures:subgroup}{subgrupo}: faltam os inversos. O critério
        $xy^{-1} \in H$ testa tudo de uma vez --- mas apenas
        depois de verificar que $H \neq \emptyset$.
  \item \emph{Reflexos não \omterm{def:b1:structures:group}{abelianos}.} Num \omterm{def:b1:structures:group}{grupo} geral, $(ab)^2 =
        abab$, o que só é $a^2b^2$ quando $a$ e $b$ comutam;
        do mesmo modo, $(ab)^{-1} = b^{-1}a^{-1}$, com a \omterm{def:b1:structures:order}{ordem} invertida.
        Toda identidade importada da álgebra escolar deve ser
        redemonstrada a partir dos axiomas ou marcada como comutativa.
  \item \emph{\omterm{def:b1:structures:morphism}{Núcleo} versus imagem.} $\ker f$ vive na
        \emph{origem}, $\operatorname{im} f$ no destino;
        ``$f$ é \omterm{def:b1:logic:inj}{injetiva} se, e somente se, $\ker f$ é trivial''
        (\cref{prop:b1:structures:kernel}) não tem análogo com
        a imagem ($\operatorname{im} f = G'$ é a sobrejetividade).
  \item \emph{Anéis não são \omterm{def:b1:structures:group}{grupos} para $\times$.} Num \omterm{def:b1:structures:ring}{anel}, a maioria
        dos elementos não precisa ser invertível, e cancelar por $a$
        exige que $a$ seja uma unidade ou que o \omterm{def:b1:structures:ring}{anel} seja um \omterm{def:b1:structures:field}{domínio
        de integridade}: em $\Z/12\Z$, $\overline3\,\overline2 =
        \overline3\,\overline6$, e no entanto $\overline2 \neq \overline6$
        (\cref{ex:b1:structures:zncomputation}).
\end{enumerate}
\end{remark}

\begin{definition}[Morfismo de grupos]\label{def:b1:structures:morphism}
Sejam $(G, *)$ e $(G', \star)$ \omterm{def:b1:structures:group}{grupos}. Uma \omterm{def:b1:logic:map}{aplicação} $f \colon G \to G'$
é um \emph{morfismo}\index{morfismo!de grupos} quando
\[
\forall x, y \in G, \qquad f(x * y) = f(x) \star f(y).
\]
Então $f(e_G) = e_{G'}$ e $f(x^{-1}) = f(x)^{-1}$. O
\emph{núcleo}\index{núcleo} e a \emph{imagem} de $f$ são
\[
\ker f = f^{-1}(\{e_{G'}\}) \leq G,
\qquad
\operatorname{im} f = f(G) \leq G' .
\]
Um morfismo \omterm{def:b1:logic:inj}{bijetivo} é um \emph{isomorfismo}; a sua \omterm{def:b1:logic:map}{aplicação} inversa é
então automaticamente um morfismo.
\end{definition}

\begin{proof}[Demonstração das afirmações]
$f(e) = f(e * e) = f(e)\star f(e)$, e cancelar $f(e)$ dá
$e_{G'} = f(e)$. Então $f(x)\star f(x^{-1}) = f(x x^{-1}) = e_{G'}$
identifica $f(x^{-1})$ como o inverso. \omterm{def:b1:structures:morphism}{Núcleo}: $e \in \ker f$; se
$x, y \in \ker f$, $f(xy^{-1}) = f(x)f(y)^{-1} = e$; o critério
se aplica. Imagem: mesmo critério com $f(x)f(y)^{-1} = f(xy^{-1})$.
Inversa de um isomorfismo: para $u, v \in G'$, escreva $u = f(x)$, $v =
f(y)$; então $f^{-1}(u \star v) = f^{-1}(f(xy)) = xy =
f^{-1}(u) f^{-1}(v)$.
\end{proof}

\begin{proposition}[Injetividade via o núcleo]\label{prop:b1:structures:kernel}
Um \omterm{def:b1:structures:morphism}{morfismo} de \omterm{def:b1:structures:group}{grupos} $f$ é \omterm{def:b1:logic:inj}{injetivo} se, e somente se, $\ker f =
\{e\}$.
\end{proposition}

\begin{proof}
Se $f$ é \omterm{def:b1:logic:inj}{injetivo}, $\ker f$ só pode conter a única \omterm{def:b1:logic:map}{pré-imagem} de
$e_{G'}$, que é $e$. Reciprocamente, se $\ker f = \{e\}$ e $f(x) =
f(y)$, então $f(xy^{-1}) = f(x) f(y)^{-1} = e_{G'}$, de modo que $xy^{-1} = e$,
isto é, $x = y$.
\end{proof}

\begin{example}\label{ex:b1:structures:expmorphism}
$\exp \colon (\R, +) \to (\R_+^*, \times)$ é um \omterm{def:b1:structures:morphism}{morfismo}
($\eu^{x+y} = \eu^x \eu^y$), \omterm{def:b1:logic:inj}{bijetivo}
(\cref{prop:b1:functions:expln}): as estruturas aditiva e multiplicativa
são isomorfas --- a razão de ser histórica dos
logaritmos. Outro \omterm{def:b1:structures:morphism}{morfismo}: $\theta \mapsto \eu^{\iu\theta}$ de
$(\R, +)$ sobre o círculo unitário $(\mathbb{U}, \times)$, com \omterm{def:b1:structures:morphism}{núcleo}
$2\pi\Z$.
\end{example}

\begin{example}[O morfismo sinal]\label{ex:b1:structures:signmorphism}
A \omterm{def:b1:logic:map}{aplicação} $s \colon (\R^*, \times) \to (\{\pm1\}, \times)$ que envia
$x$ ao seu sinal é um \omterm{def:b1:structures:morphism}{morfismo}: o sinal de um produto é o
produto dos sinais. O seu \omterm{def:b1:structures:morphism}{núcleo} é $\intoo0{+\infty}$ (um
\omterm{def:b1:structures:subgroup}{subgrupo}, como a \cref{def:b1:structures:morphism} promete), e a sua
imagem é todo o $\{\pm1\}$: \omterm{def:b1:logic:inj}{sobrejetivo}, maciçamente não \omterm{def:b1:logic:inj}{injetivo}.
Duas lições gerais em miniatura. Primeira, um \omterm{def:b1:structures:morphism}{morfismo} pode esmagar
informação: $s$ nada guarda de $x$ além de um bit, e isso
é a sua virtude --- argumentos de sinal são exatamente os cálculos
que se fatoram por $s$. Segunda, os \omterm{def:b1:structures:morphism}{morfismos} para $\{\pm1\}$ são os
``invariantes'' mais simples: o sinal das permutações, construído no
problema de fim de semana deste capítulo, é o mesmo fenômeno no
\omterm{def:b1:structures:group}{grupo} $\mathfrak S_n$, e os argumentos de paridade que ele alimenta
descem todos por um \omterm{def:b1:structures:morphism}{morfismo} de dois valores desse tipo.
\end{example}

\begin{definition}[Potências, ordem de um elemento]\label{def:b1:structures:order}
Num \omterm{def:b1:structures:group}{grupo} $G$ (notação multiplicativa), ponha $x^0 = e$, $x^{k+1} =
x^k x$ e $x^{-k} = (x^k)^{-1}$ para $k \in \N$; então $x^{k+l} = x^k
x^l$ para todos $k, l \in \Z$, de modo que $k \mapsto x^k$ é um \omterm{def:b1:structures:morphism}{morfismo} $(\Z,
+) \to G$ cuja imagem $\langle x \rangle = \{x^k : k \in \Z\}$ é um
\omterm{def:b1:structures:subgroup}{subgrupo}, o \omterm{def:b1:structures:subgroup}{subgrupo} \emph{gerado} por $x$. A
\emph{ordem}\index{ordem de um elemento} de $x$ é o menor $m \geq
1$ com $x^m = e$, se existir (e então $\langle x\rangle = \{e, x,
\dots, x^{m-1}\}$ tem exatamente $m$ elementos, e $x^k = e \iff m
\mid k$), e $\infty$ caso contrário.
\end{definition}

\begin{example}\label{ex:b1:structures:order}
Em $(\C^*, \times)$: $\iu$ tem \omterm{def:b1:structures:order}{ordem} $4$, com $\langle \iu \rangle
= \{1, \iu, -1, -\iu\} = \mathbb{U}_4$; mais geralmente, $\omega =
\eu^{2\iu\pi/n}$ tem \omterm{def:b1:structures:order}{ordem} $n$ e $\langle\omega\rangle =
\mathbb{U}_n$. Em $(\Z, +)$, todo $x \neq 0$ tem \omterm{def:b1:structures:order}{ordem} infinita.
Por que valem as afirmações da definição: se $x$ tem \omterm{def:b1:structures:order}{ordem} $m$, divida
qualquer $k$ por $m$ ($k = mq + r$, $0 \leq r < m$,
\cref{thm:b1:arith:division}): $x^k = (x^m)^q x^r = x^r$, de modo que as
potências se repetem com período $m$, os elementos listados são dois a
dois distintos pela minimalidade de $m$, e $x^k = e$ força $r = 0$. As
ordens de permutações são calculadas no problema de fim de semana abaixo.
\end{example}

\begin{example}[Ordens dentro de $\mathbb U_{12}$]\label{ex:b1:structures:orderU12}
Qual é a \omterm{def:b1:structures:order}{ordem} de $\omega^k$ em $\mathbb U_n$, para $\omega =
\eu^{2\iu\pi/n}$? Tem-se $(\omega^k)^m = 1$ se, e somente se, $n \mid km$ e,
escrevendo $d = \gcd(n, k)$, $n = dn'$, $k = dk'$ com $\gcd(n', k')
= 1$: $n \mid km \iff n' \mid k'm \iff n' \mid m$ (lema de Gauss,
\cref{thm:b1:arith:gauss}). O menor $m \geq 1$ desses é
$n' = \frac{n}{\gcd(n,k)}$. Em $\mathbb U_{12}$, por exemplo,
$\omega^8$ tem \omterm{def:b1:structures:order}{ordem} $\frac{12}{\gcd(12,8)} = 3$ (com efeito,
$\omega^8 = \eu^{4\iu\pi/3} \in \mathbb U_3$), ao passo que $\omega^5$
tem \omterm{def:b1:structures:order}{ordem} $12$: ele gera o \omterm{def:b1:structures:group}{grupo} inteiro, embora não seja
o gerador ``padrão''. Contar os geradores --- os $k$
com $\gcd(k, n) = 1$ --- recupera as contagens de números \omterm{cor:b1:arith:bezout}{primos entre
si} do \cref{ex:b1:counting:coprime}: a teoria de \omterm{def:b1:structures:group}{grupos} e a contagem se encontram.
\end{example}

\section{Anéis e corpos}

\begin{definition}[Anel]\label{def:b1:structures:ring}
Um \emph{anel}\index{anel} $(A, +, \times)$ é um \omterm{def:b1:logic:sets}{conjunto} com duas leis tais
que: $(A, +)$ é um \omterm{def:b1:structures:group}{grupo abeliano} (neutro $0$); $\times$ é
associativa com um neutro $1$; e $\times$ é distributiva em relação a $+$
dos dois lados. O anel é \emph{comutativo} quando $\times$ o é. Um
elemento $a$ é \emph{invertível} (uma \emph{unidade}) quando $ab = ba = 1$
para algum $b$; as unidades formam um \omterm{def:b1:structures:group}{grupo} $(A^\times, \times)$.
\end{definition}

\begin{proof}[Demonstração de que as unidades formam um grupo]
Estabilidade: se $a, a'$ são unidades com inversos $b, b'$, então
\[
(aa')(b'b) = a(a'b')b = a\,1\,b = ab = 1,
\qquad (b'b)(aa') = 1
\]
e simetricamente, de modo que $aa'$ é uma unidade. O elemento $1$ é uma
unidade (o seu próprio inverso), a associatividade é herdada de $A$, e o
inverso $b$ de uma unidade $a$ é ele próprio uma unidade (com inverso
$a$). Logo, $(A^\times, \times)$ satisfaz todos os axiomas de \omterm{def:b1:structures:group}{grupo}. Todo \omterm{def:b1:structures:group}{grupo}
deste livro que não seja construído a partir de permutações surge assim:
$\Q^* = \Q^\times$, $\R^*$, $\C^*$, as unidades de $\Z/n\Z$ abaixo
e, mais tarde, as matrizes invertíveis (\cref{ch:b1:matrices}).
\end{proof}

\begin{example}\label{ex:b1:structures:rings}
$\Z, \Q, \R, \C$ são anéis comutativos; $\Z^\times = \{1, -1\}$,
$\Q^\times = \Q^*$. Mais adiante: os anéis de polinômios $K[X]$
(\cref{ch:b1:poly}), os anéis de matrizes (não comutativos,
\cref{ch:b1:matrices}) e $\Z/n\Z$ abaixo. Em todo \omterm{def:b1:structures:ring}{anel}, $0 \times
a = 0$ (por distributividade: $0a = (0+0)a = 0a + 0a$) e
$(-1)a = -a$.
\end{example}

\begin{example}[Idempotentes: fenômenos novos em anéis novos]\label{ex:b1:structures:idempotents}
Em $\Z$, a equação $x^2 = x$, isto é, $x(x - 1) = 0$, só tem
as soluções $0$ e $1$. Em $\Z/6\Z$, testando todas as classes:
$\overline0^2 = \overline0$, $\overline1^2 = \overline1$,
$\overline3^2 = \overline9 = \overline3$ e $\overline4^2 =
\overline{16} = \overline4$ --- \emph{quatro} idempotentes. Os dois
exóticos vêm dos divisores de zero: $\overline3\,(\overline3 -
\overline1) = \overline3 \times \overline2 = \overline6 =
\overline0$, sem que
nenhum dos fatores seja nulo. Cálculos assim calibram
os instintos: fatos familiares sobre equações sobrevivem em
\omterm{def:b1:structures:field}{domínios de integridade} e \omterm{def:b1:structures:field}{corpos}, mas um \omterm{def:b1:structures:ring}{anel} geral pode se comportar
--- e se comporta --- de outro modo; veja também os anéis booleanos do
\cref{exo:b1:structures:10}, em que \emph{todo} elemento é
idempotente.
\end{example}

\begin{proposition}[Teorema binomial num anel comutativo]\label{prop:b1:structures:binomial}
Se $a, b$ são elementos de um \omterm{def:b1:structures:ring}{anel} comutativo (mais geralmente, se
$ab = ba$), então, para $n \in \N$:
\[
(a+b)^n = \sum_{k=0}^n \binom nk a^k b^{n-k},
\qquad
a^n - b^n = (a - b) \sum_{k=0}^{n-1} a^k b^{\,n-1-k} .
\]
\end{proposition}

\begin{proof}
As demonstrações do \cref{thm:b1:counting:binomial} e da identidade
geométrica usam apenas a associatividade, a comutatividade dos dois
elementos e a distributividade --- elas se aplicam literalmente.
\end{proof}

\begin{example}[O teorema binomial num anel pouco familiar]\label{ex:b1:structures:binomialring}
Dois dividendos rápidos da generalidade. Em $\Z/p\Z$ ($p$ \omterm{def:b1:arith:prime}{primo}), os
\omterm{def:b1:counting:objects}{coeficientes binomiais} intermediários se anulam
(primeiro passo do \cref{thm:b1:arith:fermat}), de modo que o teorema
colapsa no \emph{sonho do calouro}
\[
(a + b)^p = a^p + b^p \qquad \text{em} \Z/p\Z ,
\]
uma identidade genuína ali, por mais criminosa que pareça sobre $\R$.
E, em qualquer \omterm{def:b1:structures:ring}{anel} comutativo que contenha um elemento $\varepsilon$
com $\varepsilon^2 = 0$, o teorema se trunca:
$(a + \varepsilon)^n = a^n + n\,a^{n-1}\varepsilon$, com todos os termos
superiores carregando um fator $\varepsilon^2 = 0$. O coeficiente
$n\,a^{n-1}$ de $\varepsilon$ é a derivada de $x^n$ ---
o que não é acidente, e é um primeiro indício de que as derivadas são
tanto álgebra quanto análise (compare com a derivada formal do
\cref{ch:b1:poly}).
\end{example}

\begin{definition}[Domínio de integridade, corpo]\label{def:b1:structures:field}
Um \omterm{def:b1:structures:ring}{anel} comutativo $A \neq \{0\}$ é um \emph{domínio de
integridade}\index{domínio de integridade} quando não tem divisores de
zero: $ab = 0
\implies a = 0$ ou $b = 0$. Ele é um \emph{corpo}\index{corpo} quando
todo elemento não nulo é invertível. Todo corpo é um domínio de
integridade ($ab = 0$ e $a \neq 0$ dão $b = a^{-1}ab = 0$).
\end{definition}

\begin{example}
$\Q$, $\R$, $\C$ são \omterm{def:b1:structures:field}{corpos}; $\Z$ é um \omterm{def:b1:structures:field}{domínio de integridade}, mas não é
\omterm{def:b1:structures:field}{corpo}. Num \omterm{def:b1:structures:field}{domínio de integridade} vale o cancelamento para $\times$: $ab =
ac$ e $a \neq 0$ implicam $b = c$.
\end{example}

\section{O anel $\Z/n\Z$}

\begin{definition}\label{def:b1:structures:zn}
Fixe $n \in \N^*$. As classes de \omterm{def:b1:arith:congruence}{congruência} módulo $n$
(\cref{ex:b1:logic:congruence}) formam um \omterm{def:b1:logic:sets}{conjunto}
$\Z/n\Z$\index{Z/nZ@$\Z/n\Z$} de $n$ elementos, escritas $\overline 0,
\overline 1, \dots, \overline{n-1}$. As operações
\[
\overline a + \overline b = \overline{a + b},
\qquad
\overline a \times \overline b = \overline{ab}
\]
estão bem definidas --- as classes dos resultados não dependem dos
representantes, precisamente porque a \omterm{def:b1:arith:congruence}{congruência} é compatível com $+$
e $\times$ (\cref{def:b1:arith:congruence}) --- e fazem de $\Z/n\Z$ um
\omterm{def:b1:structures:ring}{anel} comutativo.
\end{definition}

\begin{theorem}[Unidades de $\Z/n\Z$; os corpos $\Z/p\Z$]\label{thm:b1:structures:zp}
\begin{enumerate}
  \item $\overline a$ é invertível em $\Z/n\Z$ se, e somente se,
        $\gcd(a, n) = 1$.
  \item $\Z/n\Z$ é um \omterm{def:b1:structures:field}{corpo} se, e somente se, $n$ é \omterm{def:b1:arith:prime}{primo}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) é a \cref{prop:b1:arith:invmod} reescrita com classes.

(2) Se $n = p$ é \omterm{def:b1:arith:prime}{primo}, todo $\overline a \neq \overline 0$ tem $p
\nmid a$, de modo que $\gcd(a, p) = 1$: invertível por (1) --- um \omterm{def:b1:structures:field}{corpo}. Se $n =
ab$ com $1 < a, b < n$, então $\overline a\, \overline b = \overline
n = \overline 0$ com $\overline a, \overline b \neq \overline 0$:
divisores de zero, logo nem sequer um \omterm{def:b1:structures:field}{domínio de integridade}; e $n = 1$
dá o \omterm{def:b1:structures:ring}{anel} nulo, excluído.
\end{proof}

\begin{example}[Quantas raízes quadradas de $1$?]\label{ex:b1:structures:sqrtone}
Resolva $x^2 = \overline 1$ em $\Z/8\Z$ e em $\Z/7\Z$. Testando as
oito classes módulo $8$: $1^2 = 1$, $3^2 = 9 \equiv 1$, $5^2 = 25
\equiv 1$, $7^2 = 49 \equiv 1$ --- \emph{quatro} soluções
$\{\overline1, \overline3, \overline5, \overline7\}$, embora
o polinômio $X^2 - 1$ tenha grau $2$. No \omterm{def:b1:structures:field}{corpo} $\Z/7\Z$,
em contrapartida, $x^2 = \overline1$ significa $(x - \overline1)(x +
\overline1) = \overline0$, e um \omterm{def:b1:structures:field}{corpo} não tem divisores de zero: $x =
\pm\overline1$, apenas duas soluções. A falha módulo $8$ é
rastreável: $(3-1)(3+1) = 2 \times 4 = 8 \equiv 0$, sem que nenhum dos
fatores se anule. Moral: a regra familiar ``uma equação de grau $d$
tem no máximo $d$ raízes'' é um teorema sobre \emph{\omterm{def:b1:structures:field}{domínios de
integridade}} (o \cref{cor:b1:poly:nroots} a demonstra sobre
\omterm{def:b1:structures:field}{corpos}); em anéis com divisores de zero ela falha silenciosamente --- e
é exatamente por isso que a demonstração por emparelhamento do teorema
de Wilson (\cref{exo:b1:arith:11}) precisou de $p$ \omterm{def:b1:arith:prime}{primo}.
\end{example}

\begin{example}[Calculando em $\Z/n\Z$]\label{ex:b1:structures:zncomputation}
Em $\Z/12\Z$: as unidades são $\overline 1, \overline 5, \overline 7,
\overline{11}$ (as classes primas com $12$), e cada uma é a sua própria
inversa ($5^2 = 25 \equiv 1$, $7^2 = 49 \equiv 1$, $11^2 = 121 \equiv
1$). A equação $\overline 3\, x = \overline 6$ tem \emph{três}
soluções ($x \in \{\overline 2, \overline 6, \overline{10}\}$):
sem invertibilidade, não há cancelamento. Em $\Z/11\Z$, em contrapartida,
toda equação $\overline a x = \overline b$ com $\overline a \neq
\overline 0$ tem exatamente uma solução.
\end{example}

\begin{example}[Os axiomas de grupo como licença para resolver]\label{ex:b1:structures:solving}
No \omterm{def:b1:structures:group}{grupo} $\bigl((\Z/7\Z)^*, \times\bigr)$, resolva $\overline
3\,x = \overline 5$. Pela \cref{prop:b1:structures:rules}~(3), a
solução existe, é única e vale $\overline3^{-1}\,
\overline5$; como $\overline3 \times \overline5 = \overline{15}
= \overline1$, o inverso de $\overline 3$ é $\overline 5$, de modo que
\[
x = \overline5 \times \overline5 = \overline{25} = \overline4,
\qquad\text{verificação:} \overline3 \times \overline4 =
\overline{12} = \overline5 .
\]
O ponto não é tanto a resposta quanto a garantia: num \omterm{def:b1:structures:group}{grupo},
toda equação desse tipo é solúvel de modo único \emph{antes} de
qualquer cálculo, de sorte que um procedimento de resolução nunca pode
esbarrar em ``nenhuma solução'' ou ``várias''. Compare com $\overline3\,x = \overline6$
em $\Z/12\Z$ acima, em que a garantia falha --- saber em que
estrutura se está é saber o que se pode dar por certo.
\end{example}

\begin{example}[Produtos diretos]\label{ex:b1:structures:product}
Se $G$ e $H$ são \omterm{def:b1:structures:group}{grupos}, o \omterm{def:b1:logic:sets}{conjunto} produto $G \times H$ com a lei
componente a componente $(g, h)(g', h') = (gg', hh')$ é um \omterm{def:b1:structures:group}{grupo}:
os axiomas se verificam coordenada a coordenada, com neutro
$(e_G, e_H)$ e inversos $(g^{-1}, h^{-1})$. As ordens se combinam pelo
mmc: $(g, h)^m = (g^m, h^m)$ é o neutro se, e somente se, a \omterm{def:b1:structures:order}{ordem}
de $g$ e a \omterm{def:b1:structures:order}{ordem} de $h$ \omterm{def:b1:arith:divides}{dividem} ambas $m$. Assim, em $\Z/2\Z
\times \Z/2\Z$ (aditivo) todo elemento não nulo tem \omterm{def:b1:structures:order}{ordem} $2$
--- este é exatamente o \omterm{def:b1:structures:group}{grupo} do retângulo do
\cref{ex:b1:structures:klein} em coordenadas --- ao passo que $\Z/4\Z$
tem um elemento de \omterm{def:b1:structures:order}{ordem} $4$: uma segunda demonstração, sem cálculo,
de que os dois \omterm{def:b1:structures:group}{grupos} de tamanho $4$ não são isomorfos (um
isomorfismo preserva as ordens). Os produtos são o modo mais fácil de
fabricar \omterm{def:b1:structures:group}{grupos} novos a partir de antigos, e o plano $\R^2 = \R \times
\R$ do \cref{ch:b1:vspaces} é a instância mais importante da construção.
\end{example}

\begin{remark}[Fermat, estruturalmente]
No \omterm{def:b1:structures:field}{corpo} $\Z/p\Z$, as classes não nulas formam um \omterm{def:b1:structures:group}{grupo} multiplicativo
com $p - 1$ elementos, e o pequeno teorema de Fermat
(\cref{thm:b1:arith:fermat}) diz: todo elemento $x$ desse \omterm{def:b1:structures:group}{grupo}
satisfaz $x^{p-1} = \overline 1$. Trata-se de uma instância de um fato geral
sobre \omterm{def:b1:structures:group}{grupos} finitos (teorema de Lagrange), demonstrado no segundo
ano; a demonstração por emparelhamento do teorema de Wilson
(\cref{exo:b1:arith:11}) já tinha esse sabor de teoria de \omterm{def:b1:structures:group}{grupos}.
\end{remark}

\begin{remark}[Interlúdio: o que a abstração compra]\label{rem:b1:structures:interlude}
É justo perguntar o que se ganhou ao demonstrar, digamos, a
\cref{prop:b1:structures:unique} para uma lei abstrata em vez de
para números. A resposta é alavancagem. Aquele argumento de duas linhas
cobre agora, de uma só vez: os inversos de funções pela composição
(\cref{thm:b1:logic:inverse}, cuja demonstração de unicidade ele repete
palavra por palavra), os inversos módulo $n$
(\cref{prop:b1:arith:invmod}), os inversos de reais não nulos, das
unidades de qualquer \omterm{def:b1:structures:ring}{anel} e --- sem sequer serem vistas --- das
matrizes invertíveis do \cref{ch:b1:matrices}, em que a unicidade de $A^{-1}$
não precisará de uma única linha de demonstração. A mesma economia vale
para a \cref{prop:b1:structures:kernel} (um critério de injetividade,
reutilizado para as aplicações lineares no \cref{ch:b1:linmaps}) e para o
critério de \omterm{def:b1:structures:subgroup}{subgrupo}. A abstração aqui não é generalidade por si
mesma: é a recusa a demonstrar o mesmo lema cinco vezes
sob cinco nomes. O preço --- controlar quais axiomas
cada enunciado realmente usou --- é exatamente o que os exercícios
deste capítulo treinam.
\end{remark}

\begin{remark}[Onde este capítulo é usado]\label{rem:b1:structures:whereused}
O vocabulário deste capítulo é a gramática do restante do
volume. Anéis e \omterm{def:b1:structures:field}{corpos} organizam o \cref{ch:b1:poly} ($K[X]$ é um
\omterm{def:b1:structures:ring}{anel} que imita $\Z$) e o \cref{ch:b1:fractions} ($K(X)$ é o seu
\omterm{def:b1:structures:field}{corpo} de frações); os espaços vetoriais (\cref{ch:b1:vspaces}) são
\omterm{def:b1:structures:group}{grupos abelianos} com um \omterm{def:b1:structures:field}{corpo} agindo sobre eles; as matrizes
(\cref{ch:b1:matrices}) formam o primeiro \omterm{def:b1:structures:ring}{anel} seriamente
não comutativo do volume, e os seus elementos invertíveis formam um
\omterm{def:b1:structures:group}{grupo} cujo estudo é a própria álgebra linear. \omterm{def:b1:structures:morphism}{Morfismos} e \omterm{def:b1:structures:morphism}{núcleos} voltam
como aplicações lineares e \omterm{def:b1:structures:morphism}{núcleos} no \cref{ch:b1:linmaps} ---
a \cref{prop:b1:structures:kernel} \emph{é} o critério de injetividade
daquele capítulo, demonstrado aqui de uma vez por todas. O
\omterm{ex:b1:structures:groups}{grupo simétrico}, estrela do problema de fim de semana abaixo, fornece o
sinal sobre o qual os determinantes são construídos no \cref{ch:b1:det}.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:structures:1}
Em $E = \R \setminus \{1\}$, defina $x * y = x + y - xy$. Demonstre que
$(E, *)$ é um \omterm{def:b1:structures:group}{grupo abeliano}. \emph{(Identifique o neutro e o
inverso de $x$; verifique a estabilidade: por que $x * y \neq 1$?)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:2}
Quais dos seguintes são \omterm{def:b1:structures:group}{grupos}?
\begin{enumerate}
  \item $(\intoo{0}{+\infty}, \times)$;
  \item $(\{-1, 0, 1\}, +)$;
  \item $(\Q^*, \times)$;
  \item o \omterm{def:b1:logic:sets}{conjunto} dos inteiros ímpares com a adição.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:3}
Escreva a tabela de composição do \omterm{ex:b1:structures:groups}{grupo simétrico} $\mathfrak{S}_3$
de $\{1,2,3\}$ (seis bijeções: identidade, três transposições, dois
$3$-ciclos) e exiba dois elementos que não comutam.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:4}
Demonstre que $H = \{z \in \C^* : \abs z = 1\}$ é um \omterm{def:b1:structures:subgroup}{subgrupo} de $(\C^*,
\times)$, e que $\R_+^*$ é outro; $H \cup \R_+^*$ é um
\omterm{def:b1:structures:subgroup}{subgrupo}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:5}
Seja $f \colon (\R, +) \to (\C^*, \times)$, $\theta \mapsto
\eu^{\iu\theta}$. Demonstre que $f$ é um \omterm{def:b1:structures:morphism}{morfismo}, calcule $\ker f$ e
$\operatorname{im} f$, e deduza da
\cref{prop:b1:structures:kernel} que $f$ não é \omterm{def:b1:logic:inj}{injetivo}. Restrinja
o domínio de modo a torná-lo \omterm{def:b1:logic:inj}{injetivo} num intervalo o maior possível.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:6}
Sejam $H, K$ \omterm{def:b1:structures:subgroup}{subgrupos} de $G$. Demonstre que $H \cap K$ é um \omterm{def:b1:structures:subgroup}{subgrupo}
e que $H \cup K$ é um \omterm{def:b1:structures:subgroup}{subgrupo} \emph{apenas} quando $H \subseteq K$ ou
$K \subseteq H$. \emph{(Se $h \in H \setminus K$ e $k \in K
\setminus H$, onde pode viver $hk$?)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:7}
Um \omterm{def:b1:structures:group}{grupo} $G$ satisfaz $x^2 = e$ para todo $x \in G$. Demonstre que $G$ é
\omterm{def:b1:structures:group}{abeliano}. \emph{(Expanda $(xy)^2$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:8}
Em $\Z/18\Z$: liste as unidades e encontre a inversa de $\overline 5$;
resolva $\overline 5\, x = \overline 7$; resolva $\overline 6\, x =
\overline 3$ e $\overline 6\, x = \overline{12}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:9}
Demonstre que o \omterm{def:b1:logic:sets}{conjunto} $\Z[\sqrt 2] = \{a + b\sqrt 2 : a, b \in \Z\}$ é
um \omterm{def:b1:structures:ring}{anel} (um subanel de $\R$) e que $1 + \sqrt 2$ é uma unidade dele
com infinitas potências distintas --- de modo que $\Z[\sqrt 2]^\times$ é
infinito, ao contrário de $\Z^\times$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:structures:10}
(Anéis booleanos) Seja $A$ um \omterm{def:b1:structures:ring}{anel} em que $x^2 = x$ para todo $x$.
Demonstre que $x + x = 0$ para todo $x$ e que $A$ é comutativo.
\emph{(Expanda $(x+x)^2$ e $(x+y)^2$.)} Dê um exemplo de um \omterm{def:b1:structures:ring}{anel} desses
com $\mathcal{P}(E)$, tomando a diferença simétrica como
adição e a interseção como multiplicação.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:structures:11}
Seja $G$ um \omterm{def:b1:structures:group}{grupo} em que, para algum $n \geq 1$ fixado, $(xy)^n =
x^n y^n$, $(xy)^{n+1} = x^{n+1}y^{n+1}$ e $(xy)^{n+2} =
x^{n+2}y^{n+2}$ para todos $x, y$. Demonstre que $G$ é \omterm{def:b1:structures:group}{abeliano}.
\emph{(Das três identidades, deduza primeiro $y^n x = x y^n$, depois
$y^{n+1} x = x y^{n+1}$, e conclua.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:12}
\begin{enumerate}
  \item Determine todos os \omterm{def:b1:structures:morphism}{morfismos} de \omterm{def:b1:structures:group}{grupos} de $(\Z, +)$ em $(\Z, +)$.
  \item Demonstre que o único \omterm{def:b1:structures:morphism}{morfismo} de \omterm{def:b1:structures:group}{grupos} de $(\Q, +)$ em
        $(\Z, +)$ é o \omterm{def:b1:structures:morphism}{morfismo} nulo. \emph{(Para $x \in \Q$ e
        $n \in \N^*$, compare $f(x)$ e $n\,f(x/n)$.)}
\end{enumerate}
\end{exercise}

\section{Problema: O grupo simétrico e o jogo dos oito}

\begin{problem}\label{pb:b1:structures:1}
O \omterm{def:b1:structures:group}{grupo} $\mathfrak S_n$ das permutações de $\intint1n$ é o
\omterm{def:b1:structures:group}{grupo} mais antigo da matemática e ainda o mais instrutivo. Este
problema constrói a sua teoria estrutural do zero --- ciclos,
geração por transposições, o \omterm{def:b1:structures:morphism}{morfismo} \emph{sinal}
$\varepsilon \colon \mathfrak S_n \to \{\pm1\}$ (cuja existência é
genuinamente não trivial) e o \omterm{pb:b1:structures:1}{grupo alternado} $\mathfrak A_n$
gerado pelos $3$-ciclos --- e depois a converte num quebra-cabeça
clássico: no jogo de peças deslizantes $3 \times 3$, nenhuma sequência de
movimentos consegue trocar duas peças deixando todo o resto no lugar.
\index{grupo simétrico}\index{sinal}\index{grupo alternado}
As permutações agem sobre $\intint1n$; os produtos $\sigma\tau$ significam
``aplique $\tau$ primeiro''; $[\,v_1, \dots, v_n]$ denota a
\omterm{def:b1:counting:objects}{permutação} que envia $i$ a $v_i$.

\medskip
\noindent\textbf{Parte I --- Ciclos e transposições.}
\begin{enumerate}
  \item Justifique que $\abs{\mathfrak S_n} = n!$
        (\cref{thm:b1:counting:counts}). Em $\mathfrak S_3$,
        calcule os dois produtos de $\sigma = [2, 3, 1]$ e $\tau =
        [1, 3, 2]$, e conclua que $\mathfrak S_3$ não é
        \omterm{def:b1:structures:group}{abeliano}.
  \item Um \emph{$k$-ciclo} $(a_1\ a_2\ \dots\ a_k)$ ($k \geq 2$,
        com os $a_i$ dois a dois distintos) envia $a_1 \mapsto a_2
        \mapsto \dots \mapsto a_k \mapsto a_1$ e fixa
        todo o resto; o seu \emph{suporte} é $\{a_1, \dots,
        a_k\}$. Demonstre que dois ciclos com suportes disjuntos
        comutam.
  \item Demonstre que toda $\sigma \in \mathfrak S_n$ é um produto
        de ciclos com suportes dois a dois disjuntos, e que essa
        decomposição é única a menos da \omterm{def:b1:structures:order}{ordem} dos fatores.
        \emph{(Considere, para cada $i$, a sequência $i, \sigma(i),
        \sigma^2(i), \dots$: ela deve voltar a $i$; as \emph{órbitas}
        resultantes formam uma \omterm{thm:b1:logic:partition}{partição} de $\intint1n$, e $\sigma$ age em
        cada uma delas como um ciclo.)}
  \item Decomponha $\sigma = [4, 1, 5, 2, 3, 7, 8, 6] \in
        \mathfrak S_8$ em ciclos disjuntos. Definindo a
        \emph{\omterm{def:b1:structures:order}{ordem}} de $\sigma$ como na
        \cref{def:b1:structures:order}, demonstre que a \omterm{def:b1:structures:order}{ordem} de um
        produto de ciclos disjuntos é o mmc dos seus comprimentos,
        e calcule a \omterm{def:b1:structures:order}{ordem} desse $\sigma$.
  \item Demonstre a identidade telescópica
        \[
        (a_1\ a_2\ \dots\ a_k)
        = (a_1\ a_k)(a_1\ a_{k-1})\cdots(a_1\ a_2) ,
        \]
        e conclua que toda \omterm{def:b1:counting:objects}{permutação} é um produto de
        transposições. Escreva o $\sigma$ da questão 4 como um tal
        produto.
  \item Mostre ainda que as transposições \emph{adjacentes}
        $(i\ \ i{+}1)$ bastam: para $a < b$,
        \[
        (a\ b) = (a\ \ a{+}1)(a{+}1\ \ a{+}2)\cdots(b{-}1\ \ b)
        \cdots(a{+}1\ \ a{+}2)(a\ \ a{+}1),
        \]
        um produto de $2(b - a) - 1$ transposições adjacentes ---
        um número \emph{ímpar} (essa paridade importará duas vezes
        mais abaixo).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O sinal existe.} Para $\sigma
\in \mathfrak S_n$, seja
\[
N(\sigma) = \#\bigl\{(i, j) : i < j,\ \sigma(i) >
\sigma(j)\bigr\}
\]
o seu número de \emph{inversões}, e ponha $\varepsilon(\sigma) =
(-1)^{N(\sigma)}$.
\begin{enumerate}
  \item Calcule $N$ e $\varepsilon$ para a identidade, para uma
        transposição $(i\ \ i{+}1)$ e para $[2, 3, 1]$.
  \item Demonstre que, para toda $\sigma$ e toda transposição
        adjacente $\tau = (i\ \ i{+}1)$:
        $N(\sigma\tau) = N(\sigma) \pm 1$. \emph{(Compor com
        $\tau$ à direita troca os valores nas posições $i$ e
        $i + 1$; exatamente um par muda o seu estado de
        inversão.)}
  \item Deduza, usando a questão 6, que, para \emph{qualquer}
        transposição $\tau$, $\varepsilon(\sigma\tau) =
        -\varepsilon(\sigma)$; conclua que, se $\sigma$ é um
        produto de $p$ transposições, então $\varepsilon(\sigma)
        = (-1)^p$ --- em particular, a paridade de $p$ depende apenas
        de $\sigma$, e não da fatoração escolhida --- e que
        $\varepsilon \colon \mathfrak S_n \to \{\pm 1\}$ é um
        \omterm{def:b1:structures:morphism}{morfismo} de \omterm{def:b1:structures:group}{grupos}.
  \item Mostre que um $k$-ciclo tem sinal $(-1)^{k-1}$, e que,
        em geral, $\varepsilon(\sigma) = (-1)^{n -
        c(\sigma)}$, em que $c(\sigma)$ é o número de órbitas de
        $\sigma$ (pontos fixos incluídos).
  \item O \emph{\omterm{pb:b1:structures:1}{grupo alternado}} é $\mathfrak A_n =
        \ker\varepsilon$. Justifique que ele é um \omterm{def:b1:structures:subgroup}{subgrupo} e demonstre que
        $\abs{\mathfrak A_n} = \frac{n!}2$ para $n \geq 2$.
        \emph{(Fixe uma transposição $\tau_0$ e considere $\sigma
        \mapsto \sigma\tau_0$.)}
  \item Verificação de coerência em $\sigma = [4, 1, 5, 2, 3, 7, 8, 6]$:
        calcule $\varepsilon(\sigma)$ de três maneiras --- contando
        inversões, pelo tipo de ciclo via a questão 10, e pela sua
        contagem de transposições na questão 5.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- $\mathfrak A_n$ é gerado por
$3$-ciclos.}
\begin{enumerate}
  \item Sejam $a, b, c, d$ dois a dois distintos. Verifique as duas
        identidades
        \[
        (a\ b)(a\ c) = (a\ c\ b),
        \qquad
        (a\ b)(c\ d) = (a\ c\ b)(a\ c\ d) .
        \]
  \item Demonstre que, para $n \geq 3$, todo elemento de $\mathfrak
        A_n$ é um produto de $3$-ciclos. \emph{(Uma
        \omterm{def:b1:counting:objects}{permutação} par é um produto de um número par de
        transposições; absorva-as duas a duas.)}
  \item Escreva $(1\ 2)(3\ 4)$ e o $5$-ciclo $(1\ 2\ 3\ 4\ 5)$
        explicitamente como produtos de $3$-ciclos.
  \item Demonstre a fórmula de conjugação: para toda $\sigma \in
        \mathfrak S_n$,
        \[
        \sigma\,(a_1\ \dots\ a_k)\,\sigma^{-1}
        = \bigl(\sigma(a_1)\ \dots\ \sigma(a_k)\bigr) .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O jogo dos oito.} As peças $1, \dots, 8$
deslizam num quadro $3 \times 3$ com uma casa vazia; um \emph{movimento}
desliza para a casa vazia uma peça adjacente a ela. Numere as casas
$1, \dots, 9$ (linha por linha; a posição resolvida tem a peça $i$ na
casa $i$ e a casa vazia na casa $9$). Trate a casa vazia como
uma nona peça, de modo que uma posição é uma \omterm{def:b1:counting:objects}{permutação} $\sigma \in \mathfrak
S_9$ (a peça $\sigma(i)$ está na casa $i$).
\begin{enumerate}
  \item Mostre que um movimento substitui $\sigma$ por $\sigma \circ \tau$,
        em que $\tau$ é a transposição das duas casas
        envolvidas; deduza que cada movimento inverte
        $\varepsilon(\sigma)$.
  \item Seja $d(\sigma)$ a distância de Manhattan (linhas mais
        colunas) entre a casa atual da casa vazia e a sua
        casa de origem $9$. Mostre que cada movimento altera $d$ de $\pm1$,
        de modo que cada movimento também inverte $(-1)^{d(\sigma)}$. Conclua que
        \[
        I(\sigma) = \varepsilon(\sigma)\cdot(-1)^{d(\sigma)}
        \]
        é \emph{invariante} por todo movimento.
  \item Demonstre a impossibilidade clássica do quebra-cabeça: a posição
        que troca as peças $7$ e $8$ e deixa todo o resto
        (inclusive a casa vazia) no lugar não pode ser alcançada
        a partir da posição resolvida.
  \item Admitimos a recíproca (a sua demonstração é uma indução
        instrutiva, porém longa): toda posição com $I = +1$ é
        alcançável. Deduza que exatamente metade das $8!$ posições
        com a casa vazia em casa é solúvel, isto é,
        $\frac{8!}2 = 20\,160$.
  \item Deduza da questão 20 que os arranjos de peças alcançáveis
        com a casa vazia em casa formam exatamente o
        \omterm{def:b1:structures:subgroup}{subgrupo} $\mathfrak A_8 \leq \mathfrak S_8$.
  \item Aplicações do invariante: é possível alcançar (a) a
        posição em que as peças $1, 2, 3$ estão ciclicamente
        permutadas e todo o resto, casa vazia inclusive, está em casa?
        (b) a posição em que a peça $5$ e a casa vazia
        trocaram de lugar e todas as demais peças estão em casa? Justifique
        as duas respostas com $I$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Síntese.}
\begin{enumerate}
  \item Demonstre que, para $n \geq 3$, os únicos \omterm{def:b1:structures:morphism}{morfismos} de \omterm{def:b1:structures:group}{grupos} $f
        \colon \mathfrak S_n \to \{\pm 1\}$ são o \omterm{def:b1:structures:morphism}{morfismo} constante
        e $\varepsilon$. \emph{(Usando a questão 16 e a
        comutatividade de $\{\pm1\}$, mostre que $f$ assume o mesmo
        valor em todas as transposições.)}
  \item Onde exatamente o problema usou: (i) o conceito de \omterm{def:b1:structures:morphism}{morfismo}
        e a \cref{prop:b1:structures:kernel}; (ii) os princípios de
        contagem do \cref{ch:b1:counting}; (iii) a questão da boa
        definição que as questões 8--9 resolvem? Uma
        frase para cada.
  \item Síntese, num parágrafo curto: uma única função de paridade,
        demonstrada bem definida uma só vez, organiza simultaneamente a
        estrutura interna de $\mathfrak S_n$ (o \omterm{def:b1:structures:subgroup}{subgrupo}
        $\mathfrak A_n$), decide um quebra-cabeça físico e ---
        através da fórmula $\det = \sum_\sigma
        \varepsilon(\sigma)\cdots$ --- definirá os determinantes
        no \cref{ch:b1:det}. Comente o padrão recorrente:
        os invariantes convertem ``experimentar todas as sequências de
        movimentos'' num único cálculo.
\end{enumerate}
\end{problem}
