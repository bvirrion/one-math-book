\chapter{Limites e Continuidade}\label{ch:b1:continuity}

O teorema do valor intermediário e o teorema de Weierstrass foram
usados no nível do ensino médio por fé visual. Com as sequências
(\cref{ch:b1:seq}) e a topologia de $\R$ (\cref{ch:b1:topology})
em mãos, este capítulo os demonstra --- e completa a teoria com
o teorema da bijeção monótona (que legitima $\arcsin$,
$\operatorname{arcosh}$ e os seus semelhantes do \cref{ch:b1:functions})
e com o teorema de Heine sobre \omterm{def:b1:continuity:uniform}{continuidade uniforme}.

Ao longo do capítulo, $I$ é um \omterm{prop:b1:reals:intervals}{intervalo} e $f \colon I \to \R$; ``$x_0 \in
\overline I$'' permite limites nas extremidades.

\section{Limites de funções}

\begin{definition}[Limite num ponto]\label{def:b1:continuity:limit}
Sejam $x_0 \in \overline{I}$ e $\ell \in \R$. Então $f(x) \to \ell$ quando
$x \to x_0$ se\index{limite!de uma função}
\[
\forall \varepsilon > 0,\ \exists \delta > 0,\ \forall x \in I,
\qquad \abs{x - x_0} \leq \delta \implies \abs{f(x) - \ell} \leq
\varepsilon .
\]
Os limites em $\pm\infty$ e os limites infinitos são definidos pelo mesmo
padrão ($\abs{x - x_0} \leq \delta$ passa a $x \geq M$; $\abs{f(x) -
\ell} \leq \varepsilon$
passa a $f(x) \geq M'$). Os limites laterais
restringem $x$ a $x > x_0$ (escrito $x \to x_0^+$) ou a $x < x_0$. O
limite é único quando existe (mesma demonstração da
\cref{prop:b1:seq:first}).
\end{definition}

\begin{example}[Um limite pelo confronto]\label{ex:b1:continuity:floorlimit}
Calcule $\lim_{x \to 0} x\,\bigl\lfloor \frac1x \bigr\rfloor$.
O enquadramento da \omterm{thm:b1:reals:floor}{parte inteira} $\frac1x - 1 < \lfloor \frac1x \rfloor \leq
\frac1x$ dá, após multiplicar por $x$ (atenção ao sinal!):
\[
1 - x < x\Bigl\lfloor \frac1x \Bigr\rfloor \leq 1 \quad (x > 0),
\qquad
1 \leq x\Bigl\lfloor \frac1x \Bigr\rfloor < 1 - x \quad (x < 0),
\]
e os dois confrontos laterais se fecham sobre $1$: o limite é $1$. Note
o que aconteceu: $\lfloor \frac1x\rfloor$ sozinha tem saltos selvagens perto de
$0$, mas o fator $x$ doma cada salto ($x$ vezes um salto unitário é
pequeno), e só o enquadramento sobrevive. A ideia de fechamento:
limites de produtos de um fator pequeno por um fator de oscilação
limitada são problemas de confronto, nunca problemas do teorema das
operações --- o teorema das operações precisa que \emph{ambos} os
fatores convirjam.
\end{example}

\begin{theorem}[Caracterização sequencial]\label{thm:b1:continuity:seqchar}
$f(x) \to \ell$ quando $x \to x_0$ se, e somente se: para \emph{toda}
sequência $(u_n)$ de pontos de $I$ com $u_n \to x_0$, tem-se
$f(u_n) \to \ell$.
\end{theorem}

\begin{proof}
($\Rightarrow$) Sejam $u_n \to x_0$ e $\varepsilon > 0$. Tome
$\delta$ da definição e depois $N$ com $\abs{u_n - x_0} \leq
\delta$ para $n \geq N$: além de $N$, $\abs{f(u_n) - \ell} \leq
\varepsilon$.

($\Leftarrow$) Contrapositiva. Se $f \not\to \ell$: algum
$\varepsilon_0 > 0$ derrota todo $\delta$; escolhendo $\delta =
\frac{1}{n+1}$, produzem-se $u_n \in I$ com $\abs{u_n - x_0} \leq
\frac{1}{n+1}$ e $\abs{f(u_n) - \ell} > \varepsilon_0$. Então $u_n
\to x_0$, mas $f(u_n) \not\to \ell$.
\end{proof}

\begin{corollary}[Operações, composição, ordem]\label{cor:b1:continuity:operations}
Somas, produtos e quocientes (com limite não nulo no denominador) de
limites se comportam como para as sequências; se $f \to \ell$ em $x_0$ e
$g \to m$ em $\ell$, e $g$ está definida em torno de $\ell$ com
$g(\ell) = m$ ou $f \neq \ell$ perto de $x_0$, então $g \circ f \to m$ em $x_0$;
os limites preservam desigualdades largas, e vale o teorema do
confronto.
\end{corollary}

\begin{proof}
Cada enunciado se transfere pelo
\cref{thm:b1:continuity:seqchar} ao seu análogo para sequências
(\cref{thm:b1:seq:operations,thm:b1:seq:order}). Para a composição,
vale a pena escrever o encadeamento direto uma vez: dado $\varepsilon >
0$, o limite de $g$ em $\ell$ fornece $\eta > 0$ com
\[
\abs{y - \ell} \leq \eta \implies \abs{g(y) - m} \leq
\varepsilon \quad (y \text{ no domínio de} g),
\]
em que o caso $y = \ell$ está coberto porque $g(\ell) = m$; então
o limite de $f$ em $x_0$ fornece $\delta > 0$ com $\abs{x -
x_0} \leq \delta \implies \abs{f(x) - \ell} \leq \eta$; encadeando,
$\abs{x - x_0} \leq \delta$ dá $\abs{g(f(x)) - m} \leq
\varepsilon$. Na hipótese alternativa ($f \neq \ell$ perto de
$x_0$), o valor $y = \ell$ nunca é alimentado a $g$, e o valor dela
ali é irrelevante.
\end{proof}

\begin{example}[Por que a ressalva da composição existe]\label{ex:b1:continuity:compositiontrap}
Sejam $g(y) = 0$ para $y \neq 0$ e $g(0) = 1$, e seja $f$
identicamente $0$. Então $f(x) \to 0$ quando $x \to 0$, e $g(y) \to 0$
quando $y \to 0$; e, no entanto, $g(f(x)) = g(0) = 1$ para todo $x$, de modo que $g
\circ f \to 1 \neq 0$. A função interna fica \emph{exatamente sobre}
o valor proibido $\ell = 0$ para sempre, e o limite de $g$ em $0$
ignora o que $g$ faz em $0$. A ressalva do
\cref{cor:b1:continuity:operations} --- ou $g(\ell) = m$
(isto é, $g$ \omterm{def:b1:continuity:continuous}{contínua} em $\ell$), ou $f \neq \ell$ perto de $x_0$ ---
é precisamente o que exclui isso. A ideia de fechamento: na
prática compõem-se funções \emph{\omterm{def:b1:continuity:continuous}{contínuas}} e a ressalva é
gratuita; ela morde apenas quando os limites são tomados ao longo de
\omterm{def:b1:topology:open}{vizinhanças} perfuradas, e é por isso que a definição de
$\lim_{x \to x_0}$ usada neste livro inclui o ponto quando ele está
no domínio.
\end{example}

\section{Continuidade}

\begin{definition}\label{def:b1:continuity:continuous}
$f$ é \emph{contínua em $x_0 \in I$}\index{continuidade} quando $f(x)
\to f(x_0)$ quando $x \to x_0$; \emph{contínua em $I$} quando é
contínua em todo ponto. Pelo
\cref{thm:b1:continuity:seqchar}: $f$ é contínua em $x_0$ se, e somente se,
$f(u_n) \to f(x_0)$ para toda sequência $u_n \to x_0$ em $I$.
\end{definition}

\begin{example}[Uma taxonomia das descontinuidades]\label{ex:b1:continuity:taxonomy}
Três modos de falhar num ponto, em ordem crescente de gravidade.
\emph{Removível}: $f(x) = \frac{\sin x}{x}$ em $\R^*$ tem limite
$1$ em $0$; definir $f(0) = 1$ a conserta --- a descontinuidade
era um buraco, não uma característica. \emph{Salto}: $\lfloor x \rfloor$ num
inteiro tem limites laterais distintos ($n - 1$ e $n$); escolha alguma
de valor pode reconciliá-los, mas os dois meios-limites existem.
\emph{Essencial}: $\sin\frac1x$ em $0$ não tem limite lateral algum
(\cref{exo:b1:continuity:1}) --- oscilação sem
acomodação. As funções monótonas só produzem o tipo do meio
(os seus limites laterais sempre existem, sendo \omterm{def:b1:reals:bounds}{supremos} e \omterm{def:b1:reals:bounds}{ínfimos}),
e é por isso que os seus \omterm{def:b1:logic:sets}{conjuntos} de descontinuidade são no máximo
enumeráveis --- um racional por salto. As derivadas, pelo teorema de
Darboux (\cref{exo:b1:derivative:10}), só produzem o último tipo:
uma função com descontinuidade de salto nunca é derivada de coisa
alguma.
\end{example}

\begin{proposition}\label{prop:b1:continuity:algebra}
Somas, produtos, quocientes (onde definidos) e compostas de
funções \omterm{def:b1:continuity:continuous}{contínuas} são \omterm{def:b1:continuity:continuous}{contínuos}. Os \omterm{def:b1:poly:def}{polinômios}, as \omterm{def:b1:fractions:field}{frações racionais}
(fora dos seus \omterm{def:b1:fractions:field}{polos}), $\abs{\,\cdot\,}$, $\exp$, $\ln$, as
funções trigonométricas e hiperbólicas e as suas inversas
(\cref{ch:b1:functions}) são \omterm{def:b1:continuity:continuous}{contínuos} nos seus domínios.
\end{proposition}

\begin{proof}
Operações: \cref{cor:b1:continuity:operations}. As constantes e
a identidade são \omterm{def:b1:continuity:continuous}{contínuas} diretamente pela definição
($\delta = \varepsilon$ serve à identidade, e qualquer $\delta$ às
constantes); sendo \omterm{def:b1:continuity:continuous}{contínuos} os produtos de funções \omterm{def:b1:continuity:continuous}{contínuas},
cada monômio $a_k x^k$ decorre por indução em $k$, e as somas
encerram os \omterm{def:b1:poly:def}{polinômios}; uma \omterm{def:b1:fractions:field}{fração racional} é um quociente de dois
\omterm{def:b1:poly:def}{polinômios}, \omterm{def:b1:continuity:continuous}{contínua} onde o denominador não se
anula. $\abs{\,\cdot\,}$: desigualdade triangular inversa,
$\bigl|\abs{f(x)} - \abs{f(x_0)}\bigr| \leq \abs{f(x) -
f(x_0)}$, de modo que o mesmo $\delta$ serve. Para as funções
clássicas, concedemos aqui a \omterm{def:b1:continuity:continuous}{continuidade}; a derivabilidade (demonstrada
no \cref{ch:b1:derivative}) é mais forte.
\end{proof}

\begin{example}[Máximo e mínimo de funções contínuas]\label{ex:b1:continuity:maxmin}
Se $f, g$ são \omterm{def:b1:continuity:continuous}{contínuas}, então $\max(f, g)$ e $\min(f, g)$ também são:
sem nenhuma análise de casos, graças às identidades
\[
\max(f, g) = \frac{f + g + \abs{f - g}}{2},
\qquad
\min(f, g) = \frac{f + g - \abs{f - g}}{2},
\]
e à \omterm{def:b1:continuity:continuous}{continuidade} das somas e de $\abs{\,\cdot\,}$
(\cref{prop:b1:continuity:algebra}). Em particular, $f^+ =
\max(f, 0)$ e $f^- = \max(-f, 0)$ são \omterm{def:b1:continuity:continuous}{contínuas}, com $f =
f^+ - f^-$: a separação por sinal usada para as séries
(\cref{ch:b1:series}) e, em escala plena, na teoria da integração do
volume do terceiro ano de graduação, não custa nada em regularidade.
\end{example}

\begin{theorem}[Teorema do valor intermediário]\label{thm:b1:continuity:ivt}
Seja $f$ \omterm{def:b1:continuity:continuous}{contínua} em $\intcc{a}{b}$ com $f(a) \leq 0 \leq
f(b)$. Então $f(c) = 0$ para algum $c \in \intcc{a}{b}$.
Consequentemente, uma função \omterm{def:b1:continuity:continuous}{contínua} num \omterm{prop:b1:reals:intervals}{intervalo} assume todo valor
entre dois quaisquer dos seus valores: $f(I)$ é um
\omterm{prop:b1:reals:intervals}{intervalo}.\index{teorema do valor intermediário}
\end{theorem}

\begin{proof}
Dicotomia. Ponha $a_0 = a$, $b_0 = b$. Dado $\intcc{a_n}{b_n}$ com
$f(a_n) \leq 0 \leq f(b_n)$, seja $m$ o ponto médio: se $f(m) \leq
0$, guarde $\intcc{m}{b_n}$; caso contrário, guarde $\intcc{a_n}{m}$; as
condições de sinal persistem. As sequências $(a_n), (b_n)$ são adjacentes
($b_n -
a_n = \frac{b-a}{2^n}$), com limite comum $c$
(\cref{thm:b1:seq:adjacent}). Pela \omterm{def:b1:continuity:continuous}{continuidade} e pelo
\cref{thm:b1:seq:order}: $f(c) = \lim f(a_n) \leq 0$ e $f(c) =
\lim f(b_n) \geq 0$, de modo que $f(c) = 0$.

Para a consequência: dados valores $f(u) < v < f(w)$, aplique o acima
a $x \mapsto f(x) - v$ no segmento de extremidades $u$ e $w$;
assim, $f(I)$ é convexo, isto é, um \omterm{prop:b1:reals:intervals}{intervalo}
(\cref{prop:b1:reals:intervals}).
\end{proof}

\begin{example}[Uma equação, o protocolo completo]\label{ex:b1:continuity:protocol}
Resolva $\eu^x = 3 - x$ em $\R$: existência, unicidade,
localização. Ponha $g(x) = \eu^x + x - 3$, \omterm{def:b1:continuity:continuous}{contínua}. Localização e
existência: $g(0) = -2 < 0$ e $g(1) = \eu - 2 > 0$, de modo que o
teorema do valor intermediário planta uma solução em
$\intoo{0}{1}$. Unicidade: $g$ é a soma da estritamente
crescente $\eu^x$ com $x - 3$, logo estritamente crescente em
$\R$; uma função estritamente monótona assume cada valor no máximo
uma vez, de sorte que a solução é única em todo o $\R$ (e não apenas
no \omterm{prop:b1:reals:intervals}{intervalo} sondado). O protocolo --- rearranjar para $g = 0$, mudança
de sinal para a existência, monotonicidade para a unicidade --- resolve
a maioria das perguntas do tipo ``quantas soluções'' em três linhas, e
as tabelas de variação do \cref{ch:b1:derivative}
o estendem a $g$ não monótona, cortando $\R$ em ramos monótonos.
\end{example}

\begin{example}[A dicotomia como algoritmo]\label{ex:b1:continuity:dichotomy}
A demonstração do \cref{thm:b1:continuity:ivt} calcula. Tome $f(x) =
x^3 + x - 1$: $f(0) = -1 < 0 < 1 = f(1)$, de modo que uma raiz está em
$\intoo{0}{1}$. Dividindo ao meio:
\[
f(0.5) = -0.375 < 0, \qquad
f(0.75) = 0.171875 > 0, \qquad
f(0.625) = -0.130859375 < 0 ,
\]
de sorte que a raiz fica sucessivamente presa em $\intoo{0.5}{1}$, depois em
$\intoo{0.5}{0.75}$, depois em $\intoo{0.625}{0.75}$ (valor verdadeiro:
$c \approx 0.6823$). Após $n$ passos, o erro é no máximo
$\frac{b - a}{2^n}$: dez passos dão três casas decimais, vinte dão
seis. A ideia de fechamento: o teorema do valor intermediário não é
apenas um enunciado de existência --- a sua demonstração por dicotomia
é um algoritmo garantido, ainda que lento, de busca de raízes, contra o
qual o método de Newton do \cref{ch:b1:derivative}, rápido mas local,
deve ser medido.
\end{example}

\begin{theorem}[Teorema de Weierstrass]\label{thm:b1:continuity:evt}
Uma função \omterm{def:b1:continuity:continuous}{contínua} num segmento $\intcc{a}{b}$ é limitada e
atinge as suas cotas: existem $c, d \in \intcc{a}{b}$ com
\[
f(c) = \inf_{\intcc{a}{b}} f,
\qquad
f(d) = \sup_{\intcc{a}{b}} f .
\]
Combinado com o \cref{thm:b1:continuity:ivt}: \emph{a imagem \omterm{def:b1:continuity:continuous}{contínua}
de um segmento é um segmento} $\intcc{f(c)}{f(d)}$.
\index{teorema de Weierstrass}
\end{theorem}

\begin{proof}
\emph{Limitada superiormente:} caso contrário, tome $u_n$ com $f(u_n) \geq n$. Pela compacidade do segmento (\cref{thm:b1:topology:compact}), uma
\omterm{def:b1:seq:subsequence}{subsequência} $u_{\varphi(n)} \to x \in \intcc{a}{b}$; a \omterm{def:b1:continuity:continuous}{continuidade} dá
$f(u_{\varphi(n)}) \to f(x)$, mas $f(u_{\varphi(n)}) \geq \varphi(n)
\to +\infty$: contradição.

\emph{\omterm{def:b1:reals:bounds}{Supremo} atingido:} sejam $M = \sup f$ e $v_n$ com $f(v_n) >
M - \frac{1}{n+1}$ (\cref{prop:b1:reals:epsilon}). Extraia
$v_{\varphi(n)} \to d \in \intcc{a}{b}$: então $f(d) = \lim
f(v_{\varphi(n)}) = M$, pelo confronto. O \omterm{def:b1:reals:bounds}{ínfimo} se trata com
$-f$.
\end{proof}

\begin{example}[Mínimo positivo num segmento]\label{ex:b1:continuity:posmin}
Seja $f$ \omterm{def:b1:continuity:continuous}{contínua} em $\intcc{0}{1}$ com $f(x) > 0$ para
todo $x$. Então $\inf f > 0$: pelo teorema de Weierstrass, o
\omterm{def:b1:reals:bounds}{ínfimo} é um \emph{valor} $f(c)$, e $f(c) > 0$ por hipótese.
Assim, uma função \omterm{def:b1:continuity:continuous}{contínua} positiva num segmento fica afastada de
$0$ --- argumento de duas linhas usado uma dúzia de vezes nos
capítulos que vêm (denominadores sob controle, enquadramentos por
funções escada, estimativas de erro). Num \omterm{prop:b1:reals:intervals}{intervalo} não compacto isso
falha espetacularmente: $f(x) = x$ em $\intoc{0}{1}$ é \omterm{def:b1:continuity:continuous}{contínua} e
positiva, com $\inf f = 0$ não atingido. A ideia de fechamento:
``positiva'' se promove a ``uniformemente positiva'' exatamente quando
o domínio é compacto; toda hipótese do teorema de Weierstrass sustenta
carga.
\end{example}

\begin{remark}[Armadilhas frequentes em torno dos três teoremas]
(i) \emph{Imagens \omterm{def:b1:continuity:continuous}{contínuas}}: só os \emph{segmentos} são robustos.
A imagem \omterm{def:b1:continuity:continuous}{contínua} de um \omterm{prop:b1:reals:intervals}{intervalo} \omterm{def:b1:topology:open}{aberto} não precisa ser \omterm{def:b1:topology:open}{aberta}
($x \mapsto x^2$ leva $\intoo{-1}{1}$ sobre $\intco{0}{1}$), a
imagem de um \omterm{def:b1:topology:closed}{fechado} não precisa ser \omterm{def:b1:topology:closed}{fechada} ($\arctan$ leva o
\omterm{def:b1:topology:closed}{fechado} $\R$ sobre o \omterm{def:b1:topology:open}{aberto}
$\intoo{-\frac\pi2}{\frac\pi2}$); mas a imagem de um segmento é
um segmento (\cref{thm:b1:continuity:evt}). (ii) \emph{O teorema do valor
intermediário precisa de um \omterm{prop:b1:reals:intervals}{intervalo}}: a função $x
\mapsto \frac1x$, \omterm{def:b1:continuity:continuous}{contínua} em $\intco{-1}{0} \cup
\intoc{0}{1}$, assume os valores $-1$ e $1$ e, no entanto, nunca se
anula --- o seu domínio são duas peças disjuntas, e o valor
$0$ cai no vão; nomeie sempre o \omterm{prop:b1:reals:intervals}{intervalo} em que o
teorema é aplicado. (iii) \emph{A \omterm{def:b1:continuity:uniform}{continuidade uniforme} é propriedade
do par (função, \omterm{def:b1:logic:sets}{conjunto})}: $x^2$ é \omterm{def:b1:continuity:uniform}{uniformemente contínua} em todo
segmento e, ainda assim, não em $\R$
(\cref{ex:b1:continuity:notuniform}) --- as palavras ``\omterm{def:b1:continuity:uniform}{uniformemente
contínua}'' sem um domínio não significam nada. (iv)
\emph{A \omterm{def:b1:continuity:continuous}{continuidade} da inversa não é formal}: ela vale em
\omterm{prop:b1:reals:intervals}{intervalos} por monotonicidade
(\cref{thm:b1:continuity:bijection}), mas uma bijeção \omterm{def:b1:continuity:continuous}{contínua}
entre uniões de \omterm{prop:b1:reals:intervals}{intervalos} pode ter inversa descontínua ---
a observação após aquele teorema existe porque os estudantes o citam
sem a hipótese de \omterm{prop:b1:reals:intervals}{intervalo}.
\end{remark}

\section{Funções monótonas e funções inversas}

\begin{theorem}[Teorema da bijeção monótona]\label{thm:b1:continuity:bijection}
Seja $f$ \omterm{def:b1:continuity:continuous}{contínua} e estritamente monótona num \omterm{prop:b1:reals:intervals}{intervalo} $I$. Então
\begin{enumerate}
  \item $f$ é uma bijeção de $I$ sobre o \omterm{prop:b1:reals:intervals}{intervalo} $J = f(I)$;
  \item a inversa $f^{-1} \colon J \to I$ é estritamente monótona (no
        mesmo sentido) e \emph{\omterm{def:b1:continuity:continuous}{contínua}}.
\end{enumerate}
\end{theorem}

\begin{proof}
Digamos que $f$ seja estritamente crescente. (1) A injetividade é
imediata pela monotonicidade estrita; a sobrejetividade sobre $f(I)$ é
trivial, e $f(I)$ é um \omterm{prop:b1:reals:intervals}{intervalo} pelo \cref{thm:b1:continuity:ivt}.

(2) $f^{-1}$ é estritamente crescente: se $y < y'$ em $J$, mas $f^{-1}(y)
\geq f^{-1}(y')$, aplicar a crescente $f$ dá $y \geq y'$,
absurdo. \omterm{def:b1:continuity:continuous}{Continuidade} de $f^{-1}$ em $y_0 = f(x_0) \in J$: seja
$\varepsilon > 0$. Suponha primeiro $x_0$ \omterm{def:b1:topology:closure}{interior} a $I$ e
encolha $\varepsilon$ de modo que $x_0 \pm \varepsilon \in I$: as suas
imagens satisfazem $f(x_0 - \varepsilon) < y_0 < f(x_0 +
\varepsilon)$. Tome
$\delta = \min\bigl(y_0 - f(x_0 - \varepsilon),\, f(x_0 +
\varepsilon) - y_0\bigr) > 0$: para $\abs{y - y_0} \leq \delta$,
a monotonicidade de $f^{-1}$ comprime $f^{-1}(y)$ entre $x_0 -
\varepsilon$ e $x_0 + \varepsilon$. Se $x_0$ é, digamos, a
extremidade esquerda de $I$, só $x_0 + \varepsilon$ está disponível: então
$y_0 = \min J$ (monotonicidade), todo $y \in J$ com $y - y_0
\leq f(x_0 + \varepsilon) - y_0$ satisfaz $x_0 \leq f^{-1}(y)
\leq x_0 + \varepsilon$, e a estimativa unilateral é exatamente a
\omterm{def:b1:continuity:continuous}{continuidade} numa extremidade; a extremidade direita é simétrica.
(Note: a \omterm{def:b1:continuity:continuous}{continuidade} de $f^{-1}$
\emph{não} foi deduzida da de $f$ por simetria --- é a
monotonicidade num \omterm{prop:b1:reals:intervals}{intervalo} que faz o trabalho.)
\end{proof}

\begin{remark}
Este teorema é o que a \cref{def:b1:functions:arc} e a
\cref{prop:b1:functions:invhyp} usaram silenciosamente: $\arcsin$, $\arccos$,
$\arctan$, $\operatorname{arsinh}$, \dots\ são \omterm{def:b1:continuity:continuous}{contínuas}. Um
complemento (\cref{exo:b1:continuity:10}): uma função \omterm{def:b1:continuity:continuous}{contínua} e
\emph{\omterm{def:b1:logic:inj}{injetiva}} num \omterm{prop:b1:reals:intervals}{intervalo} é automaticamente estritamente
monótona, de modo que a hipótese de monotonicidade nada custa.
\end{remark}

\begin{example}[Raízes de todas as ordens]\label{ex:b1:continuity:nthroot}
Para $n \in \N^*$, a função $f(x) = x^n$ é \omterm{def:b1:continuity:continuous}{contínua} e
estritamente crescente em $\intco{0}{+\infty}$, com $f(0) = 0$ e
$f(x) \to +\infty$: a sua imagem é $\intco{0}{+\infty}$
(\cref{thm:b1:continuity:ivt} para a estrutura de \omterm{prop:b1:reals:intervals}{intervalo}). O
teorema da bijeção monótona entrega então, de uma só vez, uma
inversa \emph{\omterm{def:b1:continuity:continuous}{contínua}} estritamente crescente
\[
x \mapsto x^{1/n} \colon \intco{0}{+\infty} \to
\intco{0}{+\infty} :
\]
existência, unicidade e \omterm{def:b1:continuity:continuous}{continuidade} das raízes $n$-ésimas, sem
cálculo algum. Compare com o \cref{exo:b1:reals:12}, que construiu
$\sqrt y$ à mão a partir do \omterm{def:b1:reals:bounds}{supremo}: um capítulo de teoria
comprimiu aquela página de trabalho em duas linhas, e as mesmas duas
linhas legitimaram $\arcsin$, $\arctan$ e
$\operatorname{arsinh}$ antes dela. A ideia de fechamento: um bom
teorema é trabalho armazenado.
\end{example}

\section{Continuidade uniforme}

\begin{definition}\label{def:b1:continuity:uniform}
$f \colon I \to \R$ é \emph{uniformemente contínua}\index{continuidade
uniforme} quando
\[
\forall \varepsilon > 0,\ \exists \delta > 0,\ \forall x, y \in I,
\qquad \abs{x - y} \leq \delta \implies \abs{f(x) - f(y)} \leq
\varepsilon .
\]
O ponto: $\delta$ depende apenas de $\varepsilon$, e não da
localização em $I$. A \omterm{def:b1:continuity:continuous}{continuidade} uniforme implica a \omterm{def:b1:continuity:continuous}{continuidade}; uma
função \emph{lipschitziana} ($\abs{f(x) - f(y)} \leq k \abs{x - y}$)
é uniformemente \omterm{def:b1:continuity:continuous}{contínua} ($\delta = \varepsilon/k$).
\end{definition}

\begin{example}\label{ex:b1:continuity:notuniform}
$x \mapsto x^2$ é \omterm{def:b1:continuity:continuous}{contínua} em $\R$, mas \emph{não} é \omterm{def:b1:continuity:uniform}{uniformemente
contínua}: $\abs{(n + \frac 1n)^2 - n^2} = 2 + \frac{1}{n^2} \geq 2$,
embora os argumentos estejam à distância $\frac 1n \to 0$. Em qualquer
\omterm{prop:b1:reals:intervals}{intervalo} limitado ela é lipschitziana, logo \omterm{def:b1:continuity:uniform}{uniformemente contínua} ---
coerente com o teorema de Heine, abaixo.
\end{example}

\begin{example}[Módulos uniformes, explicitamente]\label{ex:b1:continuity:moduli}
Num segmento, Heine garante um $\delta$ uniforme; muitas vezes também
se pode \emph{calculá-lo}. Para $f(x) = x^2$ em $\intcc{0}{10}$:
\[
\abs{x^2 - y^2} = \abs{x + y}\,\abs{x - y} \leq 20\,\abs{x - y},
\]
de modo que $\delta = \frac{\varepsilon}{20}$ serve uniformemente (um
\omterm{def:b1:complex:field}{módulo} lipschitziano, linear em $\varepsilon$). Para $\sqrt x$ em
$\intcc{0}{1}$: $\abs{\sqrt x - \sqrt y} \leq \sqrt{\abs{x -
y}}$ (\cref{exo:b1:continuity:9}), de sorte que $\delta = \varepsilon^2$
serve --- uniforme, mas \emph{não} linear: perto de $0$ a raiz
quadrada é íngreme, e o preço aparece no expoente de
$\varepsilon$, não numa falha de uniformidade. A ideia de
fechamento: a \omterm{def:b1:continuity:uniform}{continuidade uniforme} é um espectro, não um sim/não ---
a função $\delta(\varepsilon)$, chamada \omterm{def:b1:complex:field}{módulo}, mede
o quão cara é a uniformidade, e ser lipschitziana é simplesmente a sua
melhor nota.
\end{example}

\begin{theorem}[Heine]\label{thm:b1:continuity:heine}
Uma função \omterm{def:b1:continuity:continuous}{contínua} num \emph{segmento} $\intcc{a}{b}$ é \omterm{def:b1:continuity:uniform}{uniformemente
contínua}.\index{teorema de Heine}
\end{theorem}

\begin{proof}
Por absurdo: suponha que algum $\varepsilon_0 > 0$ derrote todo
$\delta$. Com $\delta = \frac{1}{n+1}$, tome $x_n, y_n \in
\intcc{a}{b}$ com $\abs{x_n - y_n} \leq \frac{1}{n+1}$ e
$\abs{f(x_n) - f(y_n)} > \varepsilon_0$. A compacidade
(\cref{thm:b1:topology:compact}) extrai $x_{\varphi(n)} \to c \in
\intcc{a}{b}$; então $y_{\varphi(n)} \to c$ também (confronto sobre $\abs{x -
y}$). A \omterm{def:b1:continuity:continuous}{continuidade} em $c$ dá $f(x_{\varphi(n)}) \to f(c)$ e
$f(y_{\varphi(n)}) \to f(c)$, de modo que $\abs{f(x_{\varphi(n)}) -
f(y_{\varphi(n)})} \to 0 < \varepsilon_0$: contradição.
\end{proof}

\begin{example}[Limitada, contínua e, ainda assim, não uniformemente]\label{ex:b1:continuity:sinx2}
A função $f(x) = \sin(x^2)$ é \omterm{def:b1:continuity:continuous}{contínua} e limitada em
$\R$, mas não é \omterm{def:b1:continuity:uniform}{uniformemente contínua}. Tome
\[
x_n = \sqrt{2\pi n}, \qquad y_n = \sqrt{2\pi n + \tfrac\pi2}:
\qquad
y_n - x_n = \frac{\pi/2}{\sqrt{2\pi n + \frac\pi2} +
\sqrt{2\pi n}} \longrightarrow 0 ,
\]
e, no entanto, $f(y_n) - f(x_n) = \sin\bigl(2\pi n + \frac\pi2\bigr) -
\sin(2\pi n) = 1 - 0 = 1$ para todo $n$: nenhum $\delta$ único pode
servir a $\varepsilon = \frac12$ em toda parte. Geometricamente, as
oscilações de $\sin(x^2)$ \emph{aceleram}: o gráfico
completa uma onda inteira em janelas cada vez mais curtas, de modo que
a escala horizontal que um dado $\varepsilon$ exige encolhe a zero
quando $x$ cresce. A ideia de fechamento: a limitação não compra
uniformidade (este exemplo), e a ilimitação não a impede
($\sqrt x$, \cref{exo:b1:continuity:9}); o que decide é o
\emph{módulo de oscilação}, e o teorema de Heine diz que domínios
compactos o disciplinam automaticamente.
\end{example}

\begin{example}[A experiência da calculadora, explicada]\label{ex:b1:continuity:cositeration}
Digite um número qualquer numa calculadora e aperte $\cos$
repetidamente: o visor trava em $0.7390851\dots$ Por quê? Após um
toque, o valor está em $\intcc{-1}{1}$; após dois, em $\intcc{\cos 1}{1}
\subseteq \intcc{0.54}{1}$, um \omterm{prop:b1:reals:intervals}{intervalo} estável para $\cos$. Nele,
$\abs{\cos a - \cos b} \leq \sin(1)\,\abs{a - b}$ com $\sin 1 =
0.841\dots < 1$
(a estimativa de produto em soma do
\cref{pb:b1:seq:1}, ou a desigualdade do valor médio do
\cref{ch:b1:derivative}): a iteração contrai, de modo que, pelo
passo de controle do erro do \cref{met:b1:seq:recurrent},
\[
\abs{u_n - c} \leq (0.842)^{\,n-2}\,\abs{u_2 - c}
\longrightarrow 0 ,
\]
em que $c$ é o único ponto fixo $\cos c = c$
(\cref{exo:b1:continuity:6}). Cerca de $40$ toques compram três
casas decimais ($0.842^{40} \approx 10^{-3}$) --- uma taxa geométrica,
mais branda que o $2^{-n}$ por passo da dicotomia, mas cada toque custa
uma tecla, ao passo que cada passo de dicotomia custa uma avaliação
completa de sinal. A ideia de fechamento: o retrato de ponto fixo do
\cref{ch:b1:seq} e os teoremas de existência deste capítulo são
duas metades de uma mesma história --- o TVI encontra $c$, e a contração
o alcança.
\end{example}

\begin{remark}[Onde estes teoremas trabalham em seguida]
Os três pilares deste capítulo alimentam cada um deles um capítulo
posterior. O teorema do valor intermediário alimenta todo argumento de
existência de soluções e o teorema da bijeção monótona; o teorema de
Weierstrass transforma problemas de otimização em teoremas (Rolle e o
teorema do valor médio no \cref{ch:b1:derivative} começam exatamente
ali); o teorema de Heine é a razão pela qual funções \omterm{def:b1:continuity:continuous}{contínuas} em
segmentos podem ser integradas no \cref{ch:b1:integration} --- o
$\delta$ uniforme é o que faz as somas de Riemann convergirem. No
volume do segundo ano de graduação, o mesmo trio reaparece em espaços
vetoriais normados, com a compacidade fazendo o trabalho que os
segmentos fazem aqui.
\end{remark}

\begin{remark}[Perspectivas dentro deste volume]
A \omterm{def:b1:continuity:continuous}{continuidade} está prestes a ser superada, mas nunca aposentada.
O \Cref{ch:b1:derivative} a fortalece em derivabilidade e
retribui o favor (derivável implica \omterm{def:b1:continuity:continuous}{contínua});
o \cref{ch:b1:integration} se apoia nela duas vezes, por Heine para
a construção e pelo teorema fundamental, cujo
objeto central $x \mapsto \int_a^x f$ promove um $f$ meramente
\omterm{def:b1:continuity:continuous}{contínuo} a uma primitiva $C^1$. No
\cref{ch:b1:multivar}, a \omterm{def:b1:continuity:continuous}{continuidade} em duas variáveis guarda uma armadilha
que vale antecipar: a função $\frac{xy}{x^2 + y^2}$ (estendida
por $0$) é \omterm{def:b1:continuity:continuous}{contínua} em $x$ para cada $y$ fixado e em $y$ para
cada $x$ fixado e, ainda assim, não é \omterm{def:b1:continuity:continuous}{contínua} na origem --- ao longo da
diagonal $x = y$ ela vale constantemente $\frac12$. A \omterm{def:b1:continuity:continuous}{continuidade}
separada é estritamente mais fraca que a \omterm{def:b1:continuity:continuous}{continuidade}: a caracterização
sequencial sobrevive à mudança para $\R^2$, mas as sequências devem
poder se aproximar de \emph{todas} as direções, e não apenas ao longo
dos eixos.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:continuity:1}
Usando a caracterização sequencial, demonstre que $x \mapsto
\sin\frac 1x$ não tem limite em $0^+$ \emph{(exiba duas sequências)}.
E $x \mapsto x \sin\frac 1x$, tem?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:continuity:2}
Estude a \omterm{def:b1:continuity:continuous}{continuidade} em $\R$ de $f(x) = \lfloor x \rfloor$ e de
$g(x) = x - \lfloor x \rfloor$, e de $h(x) = \lfloor x \rfloor +
(x - \lfloor x\rfloor)^2$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:continuity:3}
Demonstre que a equação $x^5 - 3x + 1 = 0$ tem pelo menos três soluções
reais \emph{(avalie em pontos bem escolhidos e aplique o
\cref{thm:b1:continuity:ivt} em três segmentos disjuntos)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:continuity:4}
Demonstre que todo \omterm{def:b1:poly:def}{polinômio} de grau ímpar tem uma raiz real.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:5}
(Ponto fixo) Seja $f \colon \intcc{0}{1} \to \intcc{0}{1}$ \omterm{def:b1:continuity:continuous}{contínua}.
Demonstre que $f$ tem um ponto fixo: $f(c) = c$ para algum
$c$. Ilustre que nem a \omterm{def:b1:continuity:continuous}{continuidade} nem o segmento podem ser
abandonados.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:6}
Demonstre que a equação $\cos x = x$ tem exatamente uma solução real,
e que ela está em $\intoo{0}{1}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:7}
Seja $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{contínua} com $f(x) \to +\infty$ quando $x
\to \pm\infty$. Demonstre que $f$ atinge um mínimo global em $\R$.
\emph{(Reduza a um segmento que contenha um \omterm{def:b1:logic:sets}{conjunto} de subnível.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:8}
Seja $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{contínua} e periódica (de período $T >
0$). Demonstre que $f$ é limitada e atinge as suas cotas, e que existe
$c$ com $f(c + \frac T2) = f(c)$. \emph{(Para o segundo
ponto, estude $g(x) = f(x + \frac T2) - f(x)$ ao longo de um período.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:continuity:9}
Demonstre que $x \mapsto \sqrt x$ é \omterm{def:b1:continuity:uniform}{uniformemente contínua} em
$\intco{0}{+\infty}$, embora não seja lipschitziana perto de $0$.
\emph{(Demonstre e use $\abs{\sqrt x - \sqrt y} \leq \sqrt{\abs{x -
y}}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:continuity:10}
Seja $f$ \omterm{def:b1:continuity:continuous}{contínua} e \omterm{def:b1:logic:inj}{injetiva} num \omterm{prop:b1:reals:intervals}{intervalo} $I$. Demonstre que
$f$ é estritamente monótona. \emph{Sugestão: se não, existem $a < b < c$
com, digamos, $f(b) > f(a)$ e $f(b) > f(c)$; aplique o teorema do valor
intermediário a um valor entre $\max(f(a), f(c))$ e $f(b)$ dos
dois lados de $b$.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:continuity:11}
(Equação funcional de Cauchy, caso \omterm{def:b1:continuity:continuous}{contínuo}) Seja $f \colon \R \to
\R$ \omterm{def:b1:continuity:continuous}{contínua} com $f(x + y) = f(x) + f(y)$ para todos $x, y$.
Demonstre que $f(x) = f(1)\,x$ para todo $x$: primeiro em $\N$, $\Z$ e $\Q$
(só pela aditividade) e depois em $\R$, por \omterm{def:b1:continuity:continuous}{continuidade} e \omterm{def:b1:topology:dense}{densidade}
(\cref{thm:b1:reals:density}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:continuity:12}
Seja $f \colon \intco{0}{+\infty} \to \R$ \omterm{def:b1:continuity:continuous}{contínua} com
$f(x) \to \ell \in \R$ quando $x \to +\infty$. Demonstre que $f$ é
\emph{uniformemente} \omterm{def:b1:continuity:continuous}{contínua} em $\intco{0}{+\infty}$. \emph{(Corte num
$A$ grande: Heine em $\intcc{0}{A+1}$, e o limite além de $A$; faça os
dois regimes se sobreporem.)}
\end{exercise}

\section{Problema: A equação funcional de Cauchy e as suas irmãs}

\begin{problem}[{Problema de fim de semana --- $f(x + y) = f(x) + f(y)$:
a regularidade força a linearidade, e o retrato dos monstros}]
\label{pb:b1:continuity:1}
Que funções satisfazem $f(x + y) = f(x) + f(y)$ para todos os reais $x$ e
$y$? Cauchy levantou a questão em 1821; a resposta é um paradigma.
O \Cref{exo:b1:continuity:11} mostra que uma tal função \emph{aditiva}
é linear em $\Q$ e que a \omterm{def:b1:continuity:continuous}{continuidade} plena força $f(x)
= cx$. Este problema afia a hipótese dramaticamente ---
\omterm{def:b1:continuity:continuous}{continuidade} num \emph{único} ponto, ou monotonicidade, ou mera
limitação num pequeno \omterm{prop:b1:reals:intervals}{intervalo}, cada uma basta --- e depois pinta
o retrato de uma hipotética solução não linear (o seu gráfico
preenche o plano), resolve as equações irmãs que caracterizam
$\eu^{cx}$, $c\ln x$, $x^c$ e $cx^2$, e se encerra com a equação de Jensen
e o teorema \emph{convexa no ponto médio $+$ \omterm{def:b1:continuity:continuous}{contínua}
$\implies$ convexa}.\index{equação funcional de
Cauchy}\index{equação funcional} Ao longo do problema, \emph{aditiva}
significa: $f(x + y) = f(x) + f(y)$ para todos $x, y \in \R$.

\medskip
\noindent\textbf{Parte I --- $\Q$-linearidade e um ponto de
\omterm{def:b1:continuity:continuous}{continuidade}.}
\begin{enumerate}
  \item Seja $f$ aditiva. Pelo \cref{exo:b1:continuity:11},
        $f(r) = f(1)\,r$ para $r$ racional. Demonstre o enunciado mais
        fino usado abaixo: para \emph{todo} $x \in \R$ e todo $r
        \in \Q$, $f(rx) = r\,f(x)$ ($f$ é
        $\Q$-\emph{linear}).
  \item Suponha a $f$ aditiva \omterm{def:b1:continuity:continuous}{contínua} num único
        ponto $x_0$. Mostre que $f$ é \omterm{def:b1:continuity:continuous}{contínua} em toda parte
        \emph{(calcule $f(x + h) - f(x)$ em termos de $f(x_0 + h)
        - f(x_0)$)}, e portanto $f(x) = f(1)\,x$.
  \item Mostre que uma função aditiva fica determinada pela sua
        restrição a qualquer \omterm{def:b1:structures:subgroup}{subgrupo} \omterm{def:b1:topology:dense}{denso}: se duas funções aditivas
        coincidem em $\Z + \sqrt2\,\Z$ e ambas são
        \omterm{def:b1:continuity:continuous}{contínuas}, elas são iguais --- ao passo que, sem \omterm{def:b1:continuity:continuous}{continuidade},
        prescrever $f(1) = 0$ e $f(\sqrt 2) = 1$ é compatível
        com a $\Q$-linearidade no \omterm{def:b1:structures:subgroup}{subgrupo}. Calcule $f(m +
        n\sqrt2)$
        para essa prescrição.
  \item Seja $f$ aditiva e limitada superiormente por $M$ em algum
        \omterm{prop:b1:reals:intervals}{intervalo} $\intcc{a}{b}$ com $a < b$. Mostre que $f$ é
        limitada superiormente em $\intcc{0}{\ell}$, com $\ell = b - a$
        \emph{(translade por $a$)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A escada da regularidade.}
\begin{enumerate}
  \item Continuando a questão 4: usando $f(\ell - t) + f(t) =
        f(\ell)$, mostre que $f$ é também limitada \emph{inferiormente} em
        $\intcc{0}{\ell}$: $\abs f \leq C$ ali.
  \item Mostre que $\abs{f(t)} \leq \frac{C}{n}$ para $t \in
        \intcc{0}{\ell/n}$, e deduza que $f$ é \omterm{def:b1:continuity:continuous}{contínua} em
        $0$ \emph{(a imparidade trata o lado esquerdo)}, logo
        em toda parte (questão 2): uma função aditiva limitada em
        um \omterm{prop:b1:reals:intervals}{intervalo} é linear.
  \item Deduza o caso monótono: uma função aditiva
        não decrescente em algum $\intcc{a}{b}$ ($a < b$) é $f(x)
        = cx$ com $c \geq 0$.
  \item Monte a \emph{escada da regularidade}: para uma $f$
        aditiva, são equivalentes --- (a) $f(x) = cx$;
        (b) $f$ \omterm{def:b1:continuity:continuous}{contínua}; (c) $f$ \omterm{def:b1:continuity:continuous}{contínua} num ponto; (d)
        $f$ monótona em algum \omterm{prop:b1:reals:intervals}{intervalo} não degenerado; (e) $f$
        limitada em algum \omterm{prop:b1:reals:intervals}{intervalo} não degenerado. Organize as
        implicações de modo que cada uma seja trivial ou já
        demonstrada.
  \item Verifique que as questões 4--6 consumiram apenas uma
        estimativa \emph{superior}: uma função aditiva limitada
        superiormente num \omterm{prop:b1:reals:intervals}{intervalo} não degenerado já é linear. Deduza
        o enunciado espelhado para uma estimativa inferior e registre
        a forma mais forte assim obtida do degrau (e) da escada.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Retrato de um monstro.} Suponha
agora que $f$ seja aditiva, mas \emph{não} linear.
\begin{enumerate}
  \item Mostre que existem reais não nulos $u, v$ com
        $\dfrac{f(u)}{u} \neq \dfrac{f(v)}{v}$, e que os
        vetores $(u, f(u))$ e $(v, f(v))$ geram o plano (o seu
        determinante $u f(v) - v f(u)$ é não nulo).
  \item Mostre que o gráfico de $f$ contém todos os pontos
        \[
        r\,(u, f(u)) + s\,(v, f(v)), \qquad r, s \in \Q ,
        \]
        e deduza que o gráfico é \emph{\omterm{def:b1:topology:dense}{denso} em $\R^2$}: para
        todo ponto $(x_0, y_0)$ do plano e todo
        $\varepsilon > 0$, algum $(x, f(x))$ está a menos de
        $\varepsilon$ dele \emph{(resolva o sistema real $2\times2$
        e depois aproxime os coeficientes reais por
        racionais)}.
  \item Deduza da questão 11 o retrato completo: uma função aditiva
        não linear é ilimitada em todo \omterm{prop:b1:reals:intervals}{intervalo} não degenerado,
        descontínua em todo ponto, monótona em \omterm{prop:b1:reals:intervals}{intervalo} nenhum,
        e a sua imagem de qualquer \omterm{prop:b1:reals:intervals}{intervalo} é \omterm{def:b1:topology:dense}{densa} em
        $\R$. Reconcilie com a questão 8.
  \item Monstros existem --- num \omterm{def:b1:structures:subgroup}{subgrupo} \omterm{def:b1:topology:dense}{denso}, construtivamente:
        em $G = \Z + \sqrt2\,\Z$, defina $f(m + n\sqrt2) = n$.
        Mostre que $f$ está bem definida e é aditiva em $G$, e que
        $f$ é ilimitada em $G \cap \intoo{0}{\varepsilon}$ para
        todo $\varepsilon > 0$ \emph{(para $N$ fixado, apenas
        finitos $g = m + n\sqrt2 \in \intoo{0}{1}$ têm
        $\abs n \leq N$; e, no entanto, $G \cap \intoo{0}{\varepsilon}$ é
        infinito)}. Explique num parágrafo por que estender uma tal
        $f$ a todo o $\R$ exige uma base de $\R$ como
        $\Q$-espaço vetorial (uma \emph{base de Hamel}), cuja
        existência é uma questão de axioma da escolha, além deste
        volume.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- As equações irmãs.} Todas as
funções aqui são \omterm{def:b1:continuity:continuous}{contínuas}.
\begin{enumerate}
  \item Seja $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{contínua}, não identicamente
        $0$, com $f(x + y) = f(x)f(y)$. Mostre que $f(x) =
        f\bigl(\frac x2\bigr)^2 \geq 0$, depois que $f > 0$ em toda parte,
        e depois que $f(x) = \eu^{cx}$ para algum $c$: as exponenciais
        são exatamente os \omterm{def:b1:structures:morphism}{morfismos} \omterm{def:b1:continuity:continuous}{contínuos} de $(\R, +)$ em
        $(\R^*, \times)$.
  \item Seja $f \colon \intoo{0}{+\infty} \to \R$ \omterm{def:b1:continuity:continuous}{contínua}
        com $f(xy) = f(x) + f(y)$. Mostre que $f(x) = c\ln x$
        \emph{(transporte por $\exp$)}.
  \item Seja $f \colon \intoo{0}{+\infty} \to \intoo{0}{+\infty}$
        \omterm{def:b1:continuity:continuous}{contínua} com $f(xy) = f(x)f(y)$. Mostre que $f(x) = x^c$.
  \item Encontre todas as $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{contínuas} com
        \[
        f(x + y) = f(x) + f(y) + f(x)f(y) .
        \]
        \emph{(Estude $h = 1 + f$; trate o caso degenerado
        separadamente.)}
  \item (Paralelogramo) Encontre todas as $f \colon \R \to \R$
        \omterm{def:b1:continuity:continuous}{contínuas} com $f(x + y) + f(x - y) = 2f(x) + 2f(y)$: mostre que $f$ é
        par, que $f(0) = 0$, que $f(nx) = n^2 f(x)$ por indução, e depois que
        $f(x) = f(1)\,x^2$. (Esta equação é a impressão digital das
        formas quadráticas --- a lei do paralelogramo que detecta,
        no volume do segundo ano, quais normas provêm de um produto
        interno.)
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Jensen e a convexidade no ponto médio.}
\begin{enumerate}
  \item (Equação de Jensen) Seja $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{contínua}
        com $f\bigl(\frac{x+y}{2}\bigr) = \frac{f(x)
        + f(y)}{2}$. Mostre que $g = f - f(0)$ satisfaz
        $g\bigl(\frac x2\bigr) = \frac{g(x)}{2}$, deduza que
        $g$ é aditiva e conclua que $f(x) = cx + d$.
  \item Suponha agora apenas a \emph{desigualdade}: $f$ \omterm{def:b1:continuity:continuous}{contínua}
        com
        \[
        f\Bigl(\frac{x + y}{2}\Bigr) \leq \frac{f(x) +
        f(y)}{2} \qquad (x, y \in \R).
        \]
        Demonstre por indução em $n$ que, para todos os pesos diádicos
        $\lambda = \frac{k}{2^n} \in \intcc{0}{1}$:
        \[
        f\bigl(\lambda x + (1 - \lambda)y\bigr) \leq \lambda
        f(x) + (1 - \lambda) f(y) .
        \]
  \item Estenda por \omterm{def:b1:continuity:continuous}{continuidade} e pela \omterm{def:b1:topology:dense}{densidade} dos diádicos
        (\cref{exo:b1:reals:8}) a todo $\lambda \in
        \intcc{0}{1}$: uma função \omterm{def:b1:continuity:continuous}{contínua} e convexa no ponto médio
        satisfaz a desigualdade de convexidade plena (noção
        estudada sistematicamente no \cref{ch:b1:derivative}).
  \item Mostre que a \omterm{def:b1:continuity:continuous}{continuidade} não pode ser abandonada: uma $f$
        aditiva não linear satisfaz a \emph{igualdade no ponto médio}
        da questão 19 e, ainda assim, nenhuma desigualdade de
        convexidade em \omterm{prop:b1:reals:intervals}{intervalo} algum (questão 12). Moral: a
        convexidade no ponto médio é uma propriedade de estágios
        enumeráveis (os diádicos), e a convexidade, uma propriedade do
        \omterm{def:b1:continuity:continuous}{contínuo}; a \omterm{def:b1:continuity:continuous}{continuidade} é a ponte --- exatamente como nas
        Partes I--II.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Últimas variações e síntese.}
\begin{enumerate}
  \item Encontre todas as $f \colon \R \to \R$ \omterm{def:b1:continuity:continuous}{contínuas} com $f(x + y)
        = f(x) + f(y) + xy$ \emph{(subtraia a solução particular
        $\frac{x^2}{2}$)}.
  \item Demonstre: se $f \colon \R \to \R$ é \omterm{def:b1:continuity:continuous}{contínua} e
        aditiva apenas num \emph{\omterm{def:b1:structures:subgroup}{subgrupo} \omterm{def:b1:topology:dense}{denso}} $G$ (isto é,
        $f(g + g') = f(g) + f(g')$ para $g, g' \in G$), então $f$
        é aditiva em $\R$. Mais geralmente, duas funções \omterm{def:b1:continuity:continuous}{contínuas}
        que coincidem num subconjunto \omterm{def:b1:topology:dense}{denso} de $\R$ são
        iguais.
  \item Síntese, uma frase para cada: (i) enuncie de memória a
        escada da regularidade da questão 8; (ii) explique por que
        ``gráfico \omterm{def:b1:topology:dense}{denso} no plano'' é a imagem mental certa
        para a falha da regularidade; (iii) liste as cinco
        funções clássicas caracterizadas nas Partes IV--V e o
        método único que capturou todas elas; (iv) nomeie os dois
        lugares em que a \omterm{def:b1:topology:dense}{densidade} de $\Q$ (ou dos diádicos) em $\R$
        carregou o argumento, e o lugar em que ela não pôde
        (questão 13).
\end{enumerate}
\end{problem}
