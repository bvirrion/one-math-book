\chapter{Frações Racionais}\label{ch:b1:fractions}

Uma \omterm{def:b1:fractions:field}{fração racional} é um quociente de \omterm{def:b1:poly:def}{polinômios}. O teorema central
deste capítulo curto --- a decomposição em frações
parciais\index{decomposição em frações parciais} --- quebra qualquer
quociente desses numa soma de tijolos elementares $\frac{c}{(X - a)^k}$.
Além do seu interesse algébrico, ele é a máquina padrão para
integrar funções racionais (\cref{ch:b1:integration}) e para
somar certas séries (\cref{ch:b1:series}).

\section{O corpo $K(X)$}

\begin{definition}\label{def:b1:fractions:field}
Uma \emph{fração racional}\index{fração racional} sobre $K$ ($= \R$
ou $\C$) é um quociente $F = \frac{A}{B}$ com $A, B \in K[X]$, $B \neq
0$; dois quocientes $\frac AB$ e $\frac{A'}{B'}$ são identificados quando
$AB' = A'B$. Toda fração tem uma \emph{forma reduzida} com
$\gcd(A, B) = 1$, única a menos de constantes. Com as operações naturais,
o \omterm{def:b1:logic:sets}{conjunto} $K(X)$ é um \omterm{def:b1:structures:field}{corpo}.

Os \emph{polos}\index{polo} de $F$ (na forma reduzida) são as raízes de
$B$; a \emph{ordem} de um polo é a sua \omterm{def:b1:poly:derivative}{multiplicidade} como raiz de $B$.
O \emph{grau} de $F$ é $\deg F = \deg A - \deg B \in \Z \cup
\{-\infty\}$.
\end{definition}

\begin{example}[Lendo polos, ordens e grau]\label{ex:b1:fractions:vocab}
Seja $F = \dfrac{X^3 - X}{X^4 - 2X^3 + X^2}$. Fatore as duas camadas:
o numerador, $X(X-1)(X+1)$, e o denominador, $X^2(X-1)^2$; cancele o
fator comum $X(X-1)$:
\[
F = \frac{X + 1}{X(X - 1)} \quad\text{(forma reduzida).}
\]
\omterm{def:b1:fractions:field}{Polos}: $0$ e $1$, ambos \emph{simples} --- as ordens são lidas no
denominador reduzido, de modo que as aparentes raízes duplas do
original são irrelevantes. Grau: $\deg F = 1 - 2 = -1$,
visível assintoticamente ($xF(x) \to 1$ quando $x \to \infty$). O
grau se comporta como o grau polinomial ($\deg FG = \deg F +
\deg G$, $\deg(F + G) \leq \max$), regra de escrituração usada
constantemente nas caças a coeficientes abaixo: cada argumento do tipo
``limite de $xF(x)$'' é uma contagem de graus disfarçada.
\end{example}

\begin{proposition}[Parte inteira]\label{prop:b1:fractions:integerpart}
Todo $F = \frac AB \in K(X)$ se escreve de modo único como $F = E + \frac RB$ com $E
\in K[X]$ (a \emph{parte inteira}, ou parte polinomial, de $F$) e
$\deg R < \deg B$. Tem-se $E \neq 0$ se, e somente se, $\deg F \geq 0$.
\end{proposition}

\begin{proof}
Divisão euclidiana $A = BE + R$ (\cref{thm:b1:poly:division}), dividida
por $B$. Unicidade: se $E + \frac RB = E' + \frac{R'}{B}$, então $(E -
E')B = R' - R$
com $\deg(R' - R) < \deg B$, o que força $E = E'$ e depois $R
= R'$.
\end{proof}

\begin{example}[Reduza primeiro, divida depois]\label{ex:b1:fractions:integerpart}
Encontre a parte inteira de $F = \dfrac{X^3 + 1}{X^2 - 1}$. Dividindo
às cegas: $X^3 + 1 = (X^2 - 1)X + (X + 1)$, de modo que $F = X +
\frac{X + 1}{X^2 - 1}$. Mas a fração não estava reduzida: $X^3 + 1
= (X + 1)(X^2 - X + 1)$ e $X^2 - 1 = (X+1)(X-1)$ partilham o fator
$X + 1$, e
\[
F = \frac{X^2 - X + 1}{X - 1} = X + \frac{1}{X - 1} :
\]
a mesma parte inteira $X$, mas a parte fracionária colapsa num
único tijolo, e o ``\omterm{def:b1:fractions:field}{polo}'' em $-1$ nunca foi \omterm{def:b1:fractions:field}{polo} algum.
Reduza sempre à forma irredutível antes de caçar \omterm{def:b1:fractions:field}{polos}: os \omterm{def:b1:fractions:field}{polos} de
$F$ são as raízes do denominador \emph{reduzido}. (A parte inteira
é insensível à simplificação, como a unicidade da
\cref{prop:b1:fractions:integerpart} garante.)
\end{example}

\section{Decomposição em frações parciais sobre $\C$}

\begin{theorem}[Decomposição sobre $\C$]\label{thm:b1:fractions:complex}
Seja $F = \frac AB \in \C(X)$ na forma reduzida, com $B = c\,(X -
a_1)^{m_1} \cdots (X - a_r)^{m_r}$. Então $F$ se escreve, de modo único,
\[
F = E + \sum_{i=1}^{r} \sum_{k=1}^{m_i}
\frac{c_{i,k}}{(X - a_i)^k},
\qquad E \in \C[X],\ c_{i,k} \in \C ,
\]
sendo $E$ a parte inteira de $F$.
\end{theorem}

\begin{proof}
Pela \cref{prop:b1:fractions:integerpart}, podemos supor $\deg A < \deg
B$
e demonstrar a decomposição em soma com $E = 0$.

\emph{Separando os \omterm{def:b1:fractions:field}{polos}.} Escreva $B = (X - a_1)^{m_1} B_1$ com
$B_1(a_1) \neq 0$. Os \omterm{def:b1:poly:def}{polinômios} $(X-a_1)^{m_1}$ e $B_1$ são
\omterm{cor:b1:arith:bezout}{primos entre si} (sem raiz comum), de modo que, por Bézout em $\C[X]$
(veja a observação no \cref{ch:b1:poly}), existem $U_0, V_0$ com $U_0 (X-a_1)^{m_1} +
V_0 B_1 = 1$; multiplicando por $A$ e pondo $U = AU_0$, $V =
AV_0$, e dividindo depois por $B$:
\[
\frac AB = \frac{V}{(X - a_1)^{m_1}} + \frac{U}{B_1} .
\]
As condições sobre os graus podem ser impostas: divida $V$ por
$(X-a_1)^{m_1}$, digamos $V = (X-a_1)^{m_1}Q + V_1$ com $\deg V_1 <
m_1$, e absorva o quociente no segundo termo ($U_1 = U +
QB_1$):
\[
\frac AB = \frac{V_1}{(X - a_1)^{m_1}} + \frac{U_1}{B_1},
\qquad \deg V_1 < m_1 ;
\]
comparar os graus ($\deg A < \deg B$ e $\deg V_1 < m_1$) força
também $\deg U_1 < \deg B_1$. Iterar em $\frac{U_1}{B_1}$,
\omterm{def:b1:fractions:field}{polo} após \omterm{def:b1:fractions:field}{polo}, reduz tudo ao caso de um só \omterm{def:b1:fractions:field}{polo}, abaixo.

\emph{Um só \omterm{def:b1:fractions:field}{polo}.} Para $\frac{A}{(X-a)^m}$ com $\deg A < m$: expanda $A$
em potências de $(X - a)$, $A = \sum_{j=0}^{m-1} \alpha_j (X - a)^j$
(expansão de Taylor de um \omterm{def:b1:poly:def}{polinômio}, como na demonstração da
\cref{prop:b1:poly:multiplicity}); dividindo, obtêm-se exatamente os tijolos
$\frac{\alpha_j}{(X-a)^{m-j}}$. Concretamente, para $\frac{X^2 +
1}{(X+2)^3}$: substituindo $X = Y - 2$,
\[
X^2 + 1 = (Y - 2)^2 + 1 = Y^2 - 4Y + 5 ,
\qquad\text{logo}\qquad
\frac{X^2+1}{(X+2)^3} = \frac1{Y} - \frac4{Y^2} + \frac5{Y^3}
\]
com $Y = X + 2$: os três tijolos aparecem por simples divisão da
expansão transladada --- a via mais rápida sempre que um único
\omterm{def:b1:fractions:field}{polo} de ordem alta está em jogo, e a recomendada para o
\cref{exo:b1:fractions:3}.

\emph{Unicidade.} Suponha que duas decomposições coincidam; a sua
diferença é uma identidade $0 = \sum_{i,k} \frac{d_{i,k}}{(X -
a_i)^k}$. Multiplique tudo por $(X - a_1)^{m_1}$: todo termo
adquire um fator que se anula em $a_1$, \emph{exceto} aquele com
$i = 1$, $k = m_1$, cujo coeficiente se torna $d_{1,m_1}$ mais
termos que carregam ao menos um fator $(X - a_1)$. Avaliando em
$a_1$ (legítimo: após a multiplicação, nenhum \omterm{def:b1:fractions:field}{polo} resta em
$a_1$), obtém-se $d_{1,m_1} = 0$. Eliminado o coeficiente do topo,
repita com $(X - a_1)^{m_1 - 1}$, e assim por diante, descendo até $k = 1$;
depois passe ao \omterm{def:b1:fractions:field}{polo} seguinte. Todos os $d_{i,k}$ se anulam: a
decomposição é única.
\end{proof}

\begin{method}[Calculando os coeficientes]\label{met:b1:fractions:compute}
Na prática, evite Bézout; combine:
\begin{enumerate}
  \item \emph{multiplicação-avaliação para a potência mais alta}: o
        coeficiente de $\frac{1}{(X-a)^m}$ (com $m$ a ordem do \omterm{def:b1:fractions:field}{polo} $a$) é
        \[
        c_{a,m} = \Bigl[\,(X - a)^m F\,\Bigr]_{X = a};
        \]
        para um \omterm{def:b1:fractions:field}{polo} \emph{simples} de $F = \frac AB$, isso vale
        $\frac{A(a)}{B'(a)}$;
  \item \emph{avaliações} em pontos convenientes e \emph{limites} de
        $xF(x)$ quando $x \to \infty$, para reunir relações lineares para os
        coeficientes restantes;
  \item simetrias de \emph{paridade} ou de conjugação, quando houver,
        para cortar o trabalho pela metade.
\end{enumerate}
\end{method}

\begin{remark}[Armadilhas frequentes com frações parciais]\label{rem:b1:fractions:pitfalls}
\begin{enumerate}
  \item \emph{Pular a parte inteira.} A decomposição em tijolos
        aplica-se a frações de grau $< 0$; quando $\deg A \geq
        \deg B$, divida primeiro
        (\cref{prop:b1:fractions:integerpart}), ou a
        caça aos coeficientes produzirá contradições.
  \item \emph{Multiplicação-avaliação fora do seu alcance.} Multiplicar
        por $(X - a)^k$ e avaliar em $a$ fornece o coeficiente
        apenas para $k = m$, a ordem \emph{plena} do \omterm{def:b1:fractions:field}{polo}; os
        coeficientes de ordem inferior exigem outras relações (limites,
        avaliações) --- veja o \cref{ex:b1:fractions:multiple}.
  \item \emph{Esquecer de reduzir.} Os \omterm{def:b1:fractions:field}{polos} são lidos na
        forma \emph{reduzida}; um fator comum entre numerador
        e denominador cria \omterm{def:b1:fractions:field}{polos} fantasmas
        (\cref{ex:b1:fractions:integerpart}).
  \item \emph{Formas erradas de tijolo sobre $\R$.} Acima de uma
        quadrática irredutível, os numeradores são \emph{afins} ($\alpha X +
        \beta$), e não constantes; escrever apenas $\frac{c}{X^2 + 1}$
        perde soluções --- as formas corretas são ditadas pelo
        \cref{thm:b1:fractions:real}, nunca improvisadas.
\end{enumerate}
\end{remark}

\begin{proof}[Demonstração da fórmula do polo simples]
Perto de um \omterm{def:b1:fractions:field}{polo} simples $a$: $B = (X - a) Q$ com $Q(a) \neq 0$, e $B' =
Q + (X - a) Q'$, de modo que $B'(a) = Q(a)$. O valor da
multiplicação-avaliação é $\frac{A(a)}{Q(a)} = \frac{A(a)}{B'(a)}$.
\end{proof}

\begin{example}\label{ex:b1:fractions:simple}
Decomponha $F = \dfrac{1}{X(X-1)(X-2)}$. Três \omterm{def:b1:fractions:field}{polos} simples;
multiplicação-avaliação em cada um:
\[
c_0 = \frac{1}{(0-1)(0-2)} = \frac12,
\quad
c_1 = \frac{1}{1 \times (1 - 2)} = -1,
\quad
c_2 = \frac{1}{2 \times 1} = \frac12,
\]
de modo que $F = \dfrac{1/2}{X} - \dfrac{1}{X-1} + \dfrac{1/2}{X-2}$.
\emph{Verificação em $X = 3$:} diretamente, $F(3) = \frac{1}{3\cdot2\cdot1} =
\frac16$; pela decomposição, $\frac{1/2}{3} - \frac{1}{2} +
\frac{1/2}{1} = \frac16 - \frac12 + \frac12 = \frac16$.
\end{example}

\begin{example}[Polo múltiplo]\label{ex:b1:fractions:multiple}
Decomponha $F = \dfrac{X}{(X-1)^2 (X+1)}$. Forma:
$\frac{a}{(X-1)^2} + \frac{b}{X-1} + \frac{c}{X+1}$.
\begin{itemize}
  \item Multiplicação-avaliação no \omterm{def:b1:fractions:field}{polo} duplo: $a = \bigl[\frac{X}{X+1}\bigr]_{X=1}
        = \frac12$.
  \item Multiplicação-avaliação em $-1$: $c = \bigl[\frac{X}{(X-1)^2}\bigr]_{X=-1} =
        -\frac14$.
  \item Limite de $xF(x)$ em $\infty$: $0 = b + c$, logo $b = \frac14$.
\end{itemize}
\[
F = \frac{1/2}{(X-1)^2} + \frac{1/4}{X-1} - \frac{1/4}{X+1} .
\]
\emph{Verificação em $X = 0$:} $F(0) = 0$ e $\frac12 - \frac14 - \frac14
= 0$.
\end{example}

\section{Decomposição sobre $\R$}

\begin{theorem}[Decomposição sobre $\R$]\label{thm:b1:fractions:real}
Seja $F \in \R(X)$ na forma reduzida, com denominador
\[
B = c \prod_i (X - a_i)^{m_i} \prod_j (X^2 + p_j X + q_j)^{n_j}
\qquad (p_j^2 - 4q_j < 0).
\]
Então $F$ se decompõe de modo único como a sua parte inteira mais termos
\[
\frac{c_{i,k}}{(X - a_i)^k} \quad (1 \leq k \leq m_i),
\qquad
\frac{\alpha_{j,l}\, X + \beta_{j,l}}{(X^2 + p_j X + q_j)^{l}}
\quad (1 \leq l \leq n_j),
\]
com coeficientes reais.
\end{theorem}

\begin{proof}
Decomponha sobre $\C$ (\cref{thm:b1:fractions:complex}). Como $F$ é
real, o coeficiente sobre o \omterm{def:b1:fractions:field}{polo} $\conj a$ (em cada ordem) é o
conjugado do coeficiente sobre $a$ (aplique a conjugação à
decomposição e invoque a unicidade). Agrupe cada par conjugado:
\[
\frac{c}{(X - z)^l} + \frac{\conj c}{(X - \conj z)^l}
= \frac{c\,(X - \conj z)^l + \conj c\,(X - z)^l}{\bigl(X^2 - 2\Re(z)X
+ \abs z^2\bigr)^{l}},
\]
cujo numerador é o seu próprio conjugado, logo real, de grau $\leq l$;
destacar múltiplos da quadrática real o abaixa a grau
$\leq 1$ em cada nível $l$ (uma pequena indução descendente). Os \omterm{def:b1:fractions:field}{polos}
reais mantêm os seus coeficientes reais (a conjugação os fixa). A
unicidade decorre da unicidade sobre $\C$.
\end{proof}

\begin{example}[Observando o emparelhamento de conjugados]\label{ex:b1:fractions:pairing}
O mecanismo da demonstração, no menor caso: sobre $\C$, os
\omterm{def:b1:fractions:field}{polos} de $\frac1{X^2+1}$ são $\pm\iu$, com coeficientes de
multiplicação-avaliação $\frac1{2\iu}$ em $\iu$ e $\frac1{-2\iu}$ em $-\iu$ ---
conjugados um do outro, como o teorema prevê:
\[
\frac{1}{X^2 + 1}
= \frac{1/(2\iu)}{X - \iu} - \frac{1/(2\iu)}{X + \iu} .
\]
Recombinando sobre o denominador comum:
\[
\frac{1}{2\iu}\cdot\frac{(X + \iu) - (X - \iu)}{X^2 + 1}
= \frac{1}{2\iu}\cdot\frac{2\iu}{X^2 + 1} = \frac{1}{X^2+1} :
\]
as partes imaginárias se cancelam e o tijolo real reaparece intacto.
Para integrandos reais normalmente nunca se sai de $\R$ --- mas, ao
avaliar \emph{somas} em pontos complexos (como o problema de fim de
semana faz com as \omterm{def:b1:complex:unity}{raízes da unidade}), os tijolos complexos são a moeda
natural, e este emparelhamento é a taxa de câmbio entre as duas
decomposições.
\end{example}

\begin{example}\label{ex:b1:fractions:real}
Decomponha $F = \dfrac{4}{(X^2+1)(X-1)^2}$ sobre $\R$. Forma:
$\frac{aX + b}{X^2 + 1} + \frac{c}{(X-1)^2} + \frac{d}{X - 1}$.
Multiplicação-avaliação no \omterm{def:b1:fractions:field}{polo} duplo: $c = \bigl[\frac{4}{X^2+1}\bigr]_{X=1} =
2$. Multiplicação-avaliação no \omterm{def:b1:fractions:field}{polo} complexo $\iu$ (o numerador sobre
$X^2 + 1$ avaliado pela decomposição complexa, ou diretamente):
multiplique por $X^2 + 1$ e faça $X = \iu$:
\[
a\iu + b = \frac{4}{(\iu - 1)^2} = \frac{4}{-2\iu} = 2\iu ,
\]
de modo que $a = 2$, $b = 0$. Limite de $xF(x)$ no infinito: $0 = a + d$, logo
$d = -2$. Portanto,
\[
F = \frac{2X}{X^2+1} + \frac{2}{(X-1)^2} - \frac{2}{X-1} .
\]
\emph{Verificação em $X = 0$:} $F(0) = 4$ e $0 + 2 + 2 = 4$.
\end{example}

\begin{example}[Um denominador cúbico, do início ao fim]\label{ex:b1:fractions:cubic}
Decomponha $F = \dfrac{1}{X^3 + 1}$ sobre $\R$. Fatore primeiro: $X^3 +
1 = (X + 1)(X^2 - X + 1)$, tendo a quadrática discriminante $-3 <
0$. Forma: $\frac{a}{X+1} + \frac{bX + c}{X^2 - X + 1}$. Multiplicação-avaliação
no \omterm{def:b1:fractions:field}{polo} simples $-1$: $a = \bigl[\frac1{X^2 - X +
1}\bigr]_{X=-1} = \frac13$. Limite de $xF(x)$ no infinito: $0 = a +
b$, logo $b = -\frac13$. Avaliação em $X = 0$: $1 = a + c$, logo $c =
\frac23$. Portanto,
\[
\frac{1}{X^3+1} = \frac13\Bigl(\frac{1}{X+1}
+ \frac{-X + 2}{X^2 - X + 1}\Bigr) ,
\]
confirmado em $X = 1$: lado esquerdo $\frac12$, lado direito
$\frac13\bigl(\frac12 + 1\bigr) = \frac12$. Note a economia:
três incógnitas, três fatos lineares baratos (uma
multiplicação-avaliação, um limite, uma avaliação), sem expandir nada
--- o fluxo de trabalho do \cref{met:b1:fractions:compute} na sua forma pura.
\end{example}

\begin{remark}[Para que serve]
Uma vez decomposta, uma função racional se integra termo a termo: os
tijolos $\frac{1}{(x-a)^k}$ têm primitivas elementares, e os tijolos
$\frac{\alpha x + \beta}{(x^2 + px + q)^l}$ se reduzem a $\ln$ e
$\arctan$ (\cref{ch:b1:integration}). As somas telescópicas são a outra
\omterm{def:b1:logic:map}{aplicação} padrão (\cref{exo:b1:fractions:8}).
\end{remark}

\begin{remark}[Onde este capítulo é usado]\label{rem:b1:fractions:whereused}
As frações parciais são, acima de tudo, uma etapa de
\emph{pré-processamento}: o capítulo de integração (\cref{ch:b1:integration})
passa todo integrando racional pelo \cref{thm:b1:fractions:real} antes de
integrar, e o capítulo de séries (\cref{ch:b1:series}) telescopa termos
racionais exatamente como no \cref{exo:b1:fractions:8} e no problema de fim
de semana abaixo --- que leva a técnica até
$\sum 1/k^2 = \pi^2/6$. A derivada logarítmica $P'/P = \sum
m_i/(X - a_i)$ (\cref{exo:b1:poly:11}) reaparece sempre que se estudam as
localizações das raízes. Além deste volume, a decomposição de
$1/\chi(X)$ para um \omterm{def:b1:poly:def}{polinômio} característico $\chi$ está por trás do
cálculo de potências de matrizes e das transformadas de Laplace no
volume do segundo ano de graduação: os tijolos $\frac{c}{(X - a)^k}$ são a sombra
algébrica das soluções $t^{k-1}\eu^{at}$ encontradas no \cref{ch:b1:diffeq}.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b1:fractions:1}
Decomponha sobre $\R$: $\dfrac{1}{X^2 - 1}$;
$\;\dfrac{X}{X^2 - 3X + 2}$; $\;\dfrac{X^2 + 1}{X(X-1)}$ \emph{(atenção
à parte inteira)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:fractions:2}
Decomponha sobre $\R$: $\dfrac{1}{X(X^2 + 1)}$ e
$\dfrac{X^3}{X^2 + X + 1}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:fractions:3}
Decomponha $\dfrac{1}{X^2(X - 1)}$ e $\dfrac{X + 1}{(X - 1)^3}$
\emph{(para a segunda, substitua $Y = X - 1$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:4}
Decomponha sobre $\C$ e depois sobre $\R$: $\dfrac{1}{X^4 - 1}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:5}
Decomponha sobre $\R$: $\dfrac{X^2}{(X^2 + 1)^2}$, e deduza uma
primitiva de $x \mapsto \dfrac{x^2}{(x^2+1)^2}$, dado que
$\int \frac{\dd x}{(x^2+1)^2} = \frac12\bigl(\arctan x +
\frac{x}{x^2+1}\bigr) + C$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:6}
Para $n \in \N^*$, decomponha $F_n = \dfrac{n!}{X(X+1)\cdots(X+n)}$
\emph{(\omterm{def:b1:fractions:field}{polos} simples em $0, -1, \dots, -n$; use a fórmula da
multiplicação-avaliação e reconheça \omterm{def:b1:counting:objects}{coeficientes binomiais})}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:7}
Usando a identidade do \cref{exo:b1:poly:11} para $P = X^n - 1$,
demonstre que
\[
\sum_{k=0}^{n-1} \frac{1}{X - \omega^k} = \frac{n X^{n-1}}{X^n - 1},
\qquad \omega = \eu^{2\iu\pi/n},
\]
e avalie os dois lados em $X = 2$ para $n = 4$, como verificação.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:8}
Decomponha $\dfrac{1}{k(k+1)(k+2)}$ e calcule
\[
S_n = \sum_{k=1}^{n} \frac{1}{k(k+1)(k+2)},
\qquad\text{e depois}\qquad
\lim_{n \to \infty} S_n .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fractions:9}
Seja $P \in \R[X]$ \omterm{def:b1:poly:def}{mônico} de grau $n$ com $n$ raízes reais distintas
$x_1 < \dots < x_n$. Demonstre que
\[
\sum_{i=1}^{n} \frac{1}{P'(x_i)} = 0 \quad (n \geq 2),
\qquad
\sum_{i=1}^{n} \frac{x_i^{\,n-1}}{P'(x_i)} = 1 .
\]
\emph{Sugestão: decomponha $\frac{X^m}{P}$ para $m \leq n - 1$ e observe
o decaimento dos coeficientes no infinito --- ou use a \omterm{thm:b1:poly:lagrange}{interpolação de
Lagrange} (\cref{thm:b1:poly:lagrange}) do monômio $X^m$ nos nós
$x_i$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:10}
Decomponha $\dfrac{1}{X(X+1)^2}$ sobre $\R$. Admitindo o valor
$\sum_{k \geq 1} \frac1{k^2} = \frac{\pi^2}6$ (demonstrado no problema de fim de semana
deste capítulo), deduza que
\[
\sum_{n=1}^{\infty} \frac{1}{n(n+1)^2} = 2 - \frac{\pi^2}{6} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:11}
Decomponha sobre $\R$: $\dfrac{1}{(X^2+1)(X^2+4)}$, e depois
$\dfrac{X^2}{(X^2+1)(X^2+4)}$. \emph{Sugestão: os dois denominadores são
\omterm{def:b1:poly:def}{polinômios} em $X^2$: decomponha antes $\frac1{(Y+1)(Y+4)}$.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fractions:12}
(Equações seculares) Seja $F = \sum_{i=1}^{r} \frac{c_i}{X - p_i}$
com $p_1 < p_2 < \dots < p_r$ reais e todos os $c_i > 0$.
\begin{enumerate}
  \item Mostre que $F$ é estritamente decrescente em cada intervalo do
        seu domínio e dê os seus limites em $\pm\infty$ e dos dois
        lados de cada \omterm{def:b1:fractions:field}{polo}.
  \item Deduza que, para todo $\lambda > 0$, a equação $F(x) =
        \lambda$ tem exatamente $r$ soluções reais, uma em cada
        intervalo $\intoo{p_i}{p_{i+1}}$ e uma além de $p_r$.
        (Equações desse tipo governam perturbações de autovalores; o
        teorema do valor intermediário é usado aqui no nível do ensino
        médio e demonstrado no \cref{ch:b1:continuity}.)
\end{enumerate}
\end{exercise}

\section{Problema: As frações parciais como motor}

\begin{problem}\label{pb:b1:fractions:1}
A decomposição em frações parciais parece escrituração contábil; este
problema mostra que ela é um motor. Alimentada com a fração
$\frac1{X(X+1)\cdots(X+k)}$, ela telescopa famílias inteiras de somas
em forma fechada; alimentada com $\frac1{X^n - 1}$, ela produz
identidades trigonométricas como
\[
\sum_{k=1}^{n-1}\frac{1}{\sin^2\frac{k\pi}{n}} = \frac{n^2-1}{3} ;
\]
e, empurrada um passo adiante, essa identidade espreme uma das
fórmulas mais célebres da matemática, a de Euler:
\[
\sum_{k=1}^{\infty}\frac1{k^2} = \frac{\pi^2}{6}
\]
--- obtida aqui sem nada além da álgebra deste capítulo e da
trigonometria do ensino médio.\index{telescopagem}\index{problema de
Basileia}
Ao longo do problema, $\omega = \eu^{2\iu\pi/n}$; os limites de sequências são
usados no nível do ensino médio (o \cref{ch:b1:seq} os formaliza).

\medskip
\noindent\textbf{Parte I --- O telescópio.}
\begin{enumerate}
  \item Decomponha $\frac1{X(X+1)}$ e calcule $\sum_{n=1}^{N}
        \frac1{n(n+1)}$ exatamente; conclua que a soma tende a
        $1$.
  \item O mesmo para $\frac1{X(X+2)}$: mostre que $\sum_{n=1}^{N}
        \frac1{n(n+2)} = \frac12\bigl(\frac32 - \frac1{N+1} -
        \frac1{N+2}\bigr) \to \frac34$. (Com um salto, \emph{dois}
        termos de fronteira sobrevivem em cada ponta.)
  \item Formalize o mecanismo: se $F(X) = G(X) - G(X+1)$ para
        alguma racional $G$ sem \omterm{def:b1:fractions:field}{polos} em $\intco1{+\infty}$,
        então $\sum_{n=1}^N F(n) = G(1) - G(N+1)$. Recupere o
        valor $\frac14$ do \cref{exo:b1:fractions:8} exibindo
        a testemunha $G$ para $F = \frac1{X(X+1)(X+2)}$.
  \item Demonstre o telescópio fatorial geral: para $k \geq 1$,
        \[
        \frac{1}{X(X+1)\cdots(X+k)}
        = \frac1k\biggl(\frac{1}{X(X+1)\cdots(X+k-1)}
        - \frac{1}{(X+1)\cdots(X+k)}\biggr),
        \]
        e deduza que
        \[
        \sum_{n=1}^{\infty}\frac{1}{n(n+1)\cdots(n+k)}
        = \frac{1}{k \cdot k!} .
        \]
        Confira o caso $k = 2$ contra a questão 3.
  \item Avalie a decomposição do \cref{exo:b1:fractions:6}
        em pontos bem escolhidos para demonstrar que
        \[
        \sum_{j=0}^{n}(-1)^j\binom nj\,\frac1{j+1}
        = \frac1{n+1},
        \qquad
        \sum_{j=0}^{n}(-1)^j\binom nj\,\frac1{j+2}
        = \frac1{(n+1)(n+2)} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A fração $1/(X^n - 1)$.}
\begin{enumerate}
  \item Mostre, pela fórmula da multiplicação-avaliação, que
        \[
        \frac{1}{X^n - 1}
        = \frac1n\sum_{k=0}^{n-1}\frac{\omega^k}{X - \omega^k} .
        \]
  \item Duas verificações de sanidade: confira a fórmula diretamente para
        $n =
        2$, e mostre que a soma dos $n$ coeficientes
        se anula para $n \geq 2$ --- explique por que ela deve se anular
        (considere $xF(x)$ quando $x \to \infty$).
  \item Reobtenha por multiplicação-avaliação a identidade do
        \cref{exo:b1:fractions:7}:
        $\frac{nX^{n-1}}{X^n-1} = \sum_k \frac1{X - \omega^k}$.
  \item Agrupe os \omterm{def:b1:fractions:field}{polos} conjugados para demonstrar a decomposição real:
        com $\theta_k = \frac{2k\pi}n$,
        \[
        \frac{\omega^k}{X - \omega^k} +
        \frac{\omega^{n-k}}{X - \omega^{n-k}}
        = \frac{2\cos\theta_k\,X - 2}
        {X^2 - 2\cos\theta_k\,X + 1} ,
        \]
        e escreva a decomposição real completa de
        $\frac1{X^n-1}$ (distinga $n$ ímpar e $n$ par).
  \item Especialize para $n = 4$ e confira contra o
        \cref{exo:b1:fractions:4}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Somas trigonométricas e o
$\pi^2/6$ de Euler.} Seja $P = 1 + X + \dots + X^{n-1}$, cujas raízes são
$\omega, \omega^2, \dots, \omega^{n-1}$ (todas simples).
\begin{enumerate}
  \item Usando $\frac{P'}P = \sum_{k=1}^{n-1}\frac1{X - \omega^k}$
        (\cref{exo:b1:poly:11}), demonstre que
        \[
        \sum_{k=1}^{n-1}\frac{1}{1 - \omega^k} = \frac{n-1}2 .
        \]
  \item Demonstre que $\dfrac1{1 - \eu^{\iu\theta}} = \dfrac12 +
        \dfrac\iu2\,\frac{\cos(\theta/2)}{\sin(\theta/2)}$ para
        $\theta \notin 2\pi\Z$ (fatoração do arco metade,
        \cref{met:b1:complex:trig}), e deduza da questão 11
        que $\sum_{k=1}^{n-1}
        \frac{\cos(k\pi/n)}{\sin(k\pi/n)} = 0$ --- também visível
        pela simetria $k \leftrightarrow n - k$.
  \item Derivando a identidade da questão 11 (isto é, usando
        $\bigl(\frac{P'}P\bigr)' = \frac{P''}P -
        \bigl(\frac{P'}P\bigr)^2$ avaliado em $X = 1$), demonstre que
        \[
        \sum_{k=1}^{n-1}\frac{1}{(1 - \omega^k)^2}
        = \frac{(n-1)(5-n)}{12} .
        \]
  \item Escrevendo $\cot t = \frac{\cos t}{\sin t}$, deduza das
        questões 12--13 as duas formas fechadas
        \[
        \sum_{k=1}^{n-1}\cot^2\frac{k\pi}n = \frac{(n-1)(n-2)}3,
        \qquad
        \sum_{k=1}^{n-1}\frac1{\sin^2\frac{k\pi}n}
        = \frac{n^2-1}3 .
        \]
  \item Verifique as duas fórmulas à mão para $n = 3$ e $n = 4$.
  \item Demonstre as desigualdades $\cot t < \frac1t < \frac1{\sin t}$
        para $t \in \intoo0{\frac\pi2}$ (a partir de $\sin t < t < \tan
        t$) e deduza, para $n = 2m + 1$ e $1 \leq k \leq m$:
        \[
        \cot^2\frac{k\pi}n \;<\; \frac{n^2}{k^2\pi^2}
        \;<\; \frac1{\sin^2\frac{k\pi}n} .
        \]
  \item Some essas desigualdades para $k = 1, \dots, m$ (usando a
        simetria $k \leftrightarrow n - k$ para reduzir à metade as fórmulas
        da questão 14) e comprima:
        \[
        \sum_{k=1}^{\infty}\frac1{k^2} = \frac{\pi^2}6 .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Tijolos de ordem superior.}
\begin{enumerate}
  \item Elevando ao quadrado a decomposição de $\frac1{X^2-1}$ e
        redecompondo o termo cruzado, demonstre que
        \[
        \frac1{(X^2-1)^2}
        = \frac14\Bigl(\frac1{(X-1)^2} + \frac1{(X+1)^2}\Bigr)
        - \frac14\Bigl(\frac1{X-1} - \frac1{X+1}\Bigr),
        \]
        e confira em $X = 0$.
  \item Combine a questão 18, o telescópio e o valor de Euler
        (questão 17) para demonstrar que
        \[
        \sum_{n=2}^{\infty}\frac1{(n^2-1)^2}
        = \frac{\pi^2}{12} - \frac{11}{16} ,
        \]
        e confirme o valor numericamente com três casas decimais.
  \item Derive a identidade da questão 8 para obter uma
        forma fechada para $\sum_{k=0}^{n-1}\frac1{(X -
        \omega^k)^2}$, e confira em $X = 2$, $n = 2$.
  \item Demonstre que, para $k \geq 2$,
        \[
        \sum_{n=k}^{\infty}\frac{1}{\binom nk} = \frac{k}{k-1} .
        \]
        \emph{(Reduza à questão 4 escrevendo $1/\binom nk$ com
        fatoriais.)}
  \item Para $k = 3$, dê a soma parcial exata
        $\sum_{n=3}^{N}\frac1{\binom n3}$ e o seu limite.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Síntese.}
\begin{enumerate}
  \item Como cálculo de coroamento, escreva a decomposição real
        completa de $\dfrac1{X^6 - 1}$ e confira em $X =
        0$.
  \item Onde exatamente o problema usou: (i) a unicidade da
        decomposição; (ii) as \omterm{def:b1:complex:unity}{raízes da unidade} do
        \cref{ch:b1:complex}; (iii) a derivada logarítmica
        do \cref{exo:b1:poly:11}? Uma frase para cada.
  \item Síntese, num parágrafo curto: uma única identidade algébrica
        --- quebrar uma fração em tijolos --- gerou somas
        exatas, identidades trigonométricas e $\pi^2/6$. Comente
        a divisão de trabalho entre a álgebra (decomposições exatas,
        válidas em toda parte) e a análise (limites,
        compressão), e aponte onde cada fio é
        industrializado: telescopagem e comparação no
        \cref{ch:b1:series}, integração dos tijolos no
        \cref{ch:b1:integration}.
\end{enumerate}
\end{problem}
