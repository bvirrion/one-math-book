\chapter{Dérivation}\label{ch:b1:derivative}

On a calculé des \omterm{def:b1:derivative:def}{dérivées} tout au long du volume du secondaire~; ce qui
manquait, c'est la chaîne de théorèmes qui transforme le calcul en
information sur les fonctions~: le \omterm{thm:b1:derivative:rolle}{théorème de Rolle}, le \omterm{thm:b1:derivative:mvt}{théorème des
accroissements finis}, et leurs conséquences --- critères de monotonie,
majorations lipschitziennes, \omterm{def:b1:derivative:convex}{convexité}. Tout ce chapitre concerne des
fonctions définies sur un \omterm{prop:b1:reals:intervals}{intervalle} $I$.

\section{La dérivée}

\begin{definition}\label{def:b1:derivative:def}
$f \colon I \to \R$ est \emph{dérivable en $x_0 \in
I$}\index{dérivée} lorsque le taux d'accroissement
$\frac{f(x) - f(x_0)}{x - x_0}$ admet une limite (finie) quand $x \to x_0$~;
cette limite se note $f'(x_0)$. De façon équivalente~:
\[
f(x_0 + h) = f(x_0) + f'(x_0)\,h + h\,\varepsilon(h),
\qquad \varepsilon(h) \xrightarrow[h \to 0]{} 0 ,
\]
le graphe admettant alors la tangente $y = f(x_0) + f'(x_0)(x -
x_0)$. La dérivabilité en $x_0$ entraîne la \omterm{def:b1:continuity:continuous}{continuité} en $x_0$ (relire
la formule affichée). $f$ est dérivable sur $I$ lorsqu'elle l'est en tout point~;
$f$ est de classe $C^1$ lorsque de plus $f'$ est \omterm{def:b1:continuity:continuous}{continue}, et de
classe $C^k$\index{fonction de classe Ck@fonction de classe $C^k$} lorsque $f$ peut être
dérivée $k$ fois avec $f^{(k)}$ \omterm{def:b1:continuity:continuous}{continue}.
\end{definition}

\begin{example}\label{ex:b1:derivative:notc1}
La réciproque de «~\omterm{def:b1:derivative:def}{dérivable} $\Rightarrow$ \omterm{def:b1:continuity:continuous}{continue}~» est fausse~:
$\abs{\,\cdot\,}$ en $0$. Plus surprenant, \omterm{def:b1:derivative:def}{dérivable} n'implique pas
$C^1$~: la fonction $f(x) = x^2 \sin\frac 1x$ ($f(0) = 0$) est
\omterm{def:b1:derivative:def}{dérivable} partout, avec $f'(0) = 0$, mais $f'(x) = 2x
\sin\frac1x - \cos\frac 1x$ n'a pas de limite en $0$
(\cref{exo:b1:derivative:2}).
\end{example}

\begin{example}[Dérivable en exactement un point]\label{ex:b1:derivative:onepoint}
Soit $f(x) = x^2$ pour $x \in \Q$ et $f(x) = 0$ pour $x \notin
\Q$. En $0$~: $\bigl|\frac{f(h) - 0}{h}\bigr| \leq \abs h \to 0$,
donc $f$ est \omterm{def:b1:derivative:def}{dérivable} en $0$ avec $f'(0) = 0$. En tout $x_0
\neq 0$, $f$ n'est même pas \omterm{def:b1:continuity:continuous}{continue}~: des suites de rationnels et
d'irrationnels convergeant vers $x_0$ envoient $f$ sur $x_0^2 \neq 0$ et sur
$0$ respectivement (\omterm{def:b1:topology:dense}{densité}, \cref{thm:b1:reals:density}). La
\omterm{def:b1:derivative:def}{dérivabilité} est donc une notion authentiquement \emph{ponctuelle}~: elle peut
avoir lieu en un point de $\R$ et nulle part ailleurs. La morale pour
la pratique~: des énoncés comme le critère de monotonie ou le \omterm{thm:b1:derivative:rolle}{théorème de Rolle}
exigent la \omterm{def:b1:derivative:def}{dérivée} \emph{sur un \omterm{prop:b1:reals:intervals}{intervalle}} --- posséder
$f'(x_0)$ en des points isolés, si nombreux soient-ils, n'autorise
absolument aucune conclusion globale.
\end{example}

\begin{theorem}[Opérations]\label{thm:b1:derivative:operations}
Si $f, g$ sont \omterm{def:b1:derivative:def}{dérivables} en $x_0$ (et là où les formules ont un
sens)~:
\[
(f + g)' = f' + g', \qquad
(fg)' = f'g + fg', \qquad
\Bigl(\frac fg\Bigr)' = \frac{f'g - fg'}{g^2},
\]
et si $g$ est \omterm{def:b1:derivative:def}{dérivable} en $f(x_0)$~:
$\;(g \circ f)'(x_0) = g'\bigl(f(x_0)\bigr)\, f'(x_0)$ (\omterm{thm:b1:derivative:operations}{règle de la
chaîne}\index{règle de la chaîne}).
\end{theorem}

\begin{proof}
Somme~: immédiat. Produit~: écrivons
\[
f(x)g(x) - f(x_0)g(x_0)
= \bigl(f(x) - f(x_0)\bigr) g(x) + f(x_0)\bigl(g(x) - g(x_0)\bigr),
\]
divisons par $x - x_0$ et faisons tendre $x \to x_0$ ($g$ est \omterm{def:b1:continuity:continuous}{continue} en $x_0$).
Quotient~: traitons $\frac 1g$ via $\frac{1/g(x) - 1/g(x_0)}{x - x_0} =
\frac{-1}{g(x)g(x_0)}\cdot\frac{g(x) - g(x_0)}{x - x_0}$, puis appliquons
la règle du produit. Composée~: avec $y_0 = f(x_0)$, posons
$\theta(y) = \frac{g(y) - g(y_0)}{y - y_0}$ pour $y \neq y_0$ et
$\theta(y_0) = g'(y_0)$~: $\theta$ est \omterm{def:b1:continuity:continuous}{continue} en $y_0$, et pour
$x \neq x_0$,
\[
\frac{g(f(x)) - g(f(x_0))}{x - x_0}
= \theta\bigl(f(x)\bigr)\cdot \frac{f(x) - f(x_0)}{x - x_0}
\longrightarrow g'(y_0)\, f'(x_0),
\]
le premier facteur par composition des limites (ce procédé traite
proprement le cas $f(x) = f(x_0)$, où le naïf «~multiplier et diviser
par $f(x) - f(x_0)$~» s'effondre).
\end{proof}

\begin{theorem}[Dérivée d'une fonction réciproque]\label{thm:b1:derivative:inverse}
Soit $f$ \omterm{def:b1:continuity:continuous}{continue} et strictement monotone sur $I$, \omterm{def:b1:derivative:def}{dérivable}
en $x_0$ avec $f'(x_0) \neq 0$. Alors $f^{-1}$
(\cref{thm:b1:continuity:bijection}) est \omterm{def:b1:derivative:def}{dérivable} en $y_0 =
f(x_0)$, avec
\[
(f^{-1})'(y_0) = \frac{1}{f'(x_0)} = \frac{1}{f'\bigl(f^{-1}(y_0)\bigr)} .
\]
Si $f'(x_0) = 0$, la réciproque admet une tangente verticale en $y_0$.
\end{theorem}

\begin{proof}
Pour $y \to y_0$, posons $x = f^{-1}(y)$~: la \omterm{def:b1:continuity:continuous}{continuité} de $f^{-1}$ donne $x
\to x_0$, et
\[
\frac{f^{-1}(y) - f^{-1}(y_0)}{y - y_0}
= \frac{x - x_0}{f(x) - f(x_0)}
= \frac{1}{\dfrac{f(x) - f(x_0)}{x - x_0}}
\longrightarrow \frac{1}{f'(x_0)} .
\]
Tangente verticale~: si $f'(x_0) = 0$, le quotient affiché
est l'inverse d'une quantité qui tend vers $0$ en gardant
un signe constant (pour $f$ strictement croissante, $\frac{f(x) -
f(x_0)}{x - x_0} > 0$ pour tout $x \neq x_0$)~: le taux
d'accroissement de $f^{-1}$ tend donc vers $+\infty$ (vers $-\infty$
pour $f$ décroissante). La réciproque reste \omterm{def:b1:continuity:continuous}{continue} mais n'est pas
\omterm{def:b1:derivative:def}{dérivable} en $y_0$ --- son graphe, symétrique de celui de $f$ par
rapport à la diagonale, se dresse à la verticale exactement là où celui de $f$
courait à l'horizontale, comme l'illustre $x^{1/3}$ en $0$ face à $x^3$.
\end{proof}

\begin{example}[Dérivées de réciproques, deux fois]\label{ex:b1:derivative:inverseaction}
Le théorème recalcule les \omterm{def:b1:derivative:def}{dérivées} classiques sans aucun travail
sur les limites. Pour $\ln = \exp^{-1}$~: en $y = \eu^x$,
\[
(\ln)'(y) = \frac{1}{\exp'(x)} = \frac{1}{\eu^{x}} = \frac1y ,
\]
valable pour tout $y > 0$ puisque $\exp' = \exp$ ne s'annule jamais. Pour
$\arctan = \tan^{-1}$~: en $y = \tan x$,
\[
(\arctan)'(y) = \frac{1}{1 + \tan^2 x} = \frac{1}{1 + y^2} ,
\]
en utilisant $\tan' = 1 + \tan^2 > 0$. L'idée à retenir~: la formule
convertit la connaissance d'une fonction en connaissance de sa
réciproque au prix d'une seule substitution --- et cette
substitution ($x = \ln y$, $x = \arctan y$) dit exactement que
les deux variables vivent de part et d'autre de la \omterm{def:b1:logic:inj}{bijection}.
\end{example}

\section{Rolle et le théorème des accroissements finis}

\begin{proposition}[Extremum intérieur]\label{prop:b1:derivative:fermat}
Si $f$ est \omterm{def:b1:derivative:def}{dérivable} en un point \emph{\omterm{def:b1:topology:closure}{intérieur}} $x_0$ de $I$ et y
présente un extremum local, alors $f'(x_0) = 0$.
\end{proposition}

\begin{proof}
Disons un maximum local~: il existe $r > 0$ tel que $f(x) \leq f(x_0)$
pour $\abs{x - x_0} \leq r$, et l'intériorité garantit que les deux
côtés de $x_0$ sont disponibles dans $I$. Pour $0 < h \leq r$, le
quotient $\frac{f(x_0 + h) - f(x_0)}{h}$ a un numérateur $\leq 0$
et un dénominateur $> 0$~: il est $\leq 0$, et sa limite $f'(x_0)$
hérite du $\leq 0$ (les inégalités larges passent à la limite,
\cref{thm:b1:seq:order})~; pour $-r \leq h < 0$ le quotient est
$\geq 0$, d'où $f'(x_0) \geq 0$. Donc $f'(x_0) = 0$. (À une
extrémité, un seul signe est disponible~: la conclusion y tombe en défaut ---
songeons à $x$ sur $\intcc{0}{1}$, maximal en $1$ avec une \omterm{def:b1:derivative:def}{dérivée} égale à
$1$.)
\end{proof}

\begin{theorem}[Rolle]\label{thm:b1:derivative:rolle}
Soit $f$ \omterm{def:b1:continuity:continuous}{continue} sur $\intcc{a}{b}$, \omterm{def:b1:derivative:def}{dérivable} sur
$\intoo{a}{b}$, avec $f(a) = f(b)$. Alors $f'(c) = 0$ pour un certain $c \in
\intoo{a}{b}$.\index{théorème de Rolle}
\end{theorem}

\begin{proof}
D'après le \omterm{thm:b1:continuity:evt}{théorème des bornes atteintes} (\cref{thm:b1:continuity:evt}), $f$
atteint son maximum et son minimum sur $\intcc{a}{b}$. Si tous deux sont
atteints aux extrémités, alors (puisque $f(a) = f(b)$) max $=$ min et $f$
est constante~: n'importe quel $c$ \omterm{def:b1:topology:closure}{intérieur} convient. Sinon un extremum est
atteint en un point \omterm{def:b1:topology:closure}{intérieur} $c$, et
la \cref{prop:b1:derivative:fermat} donne $f'(c) = 0$.
\end{proof}

\begin{theorem}[Théorème des accroissements finis]\label{thm:b1:derivative:mvt}
Soit $f$ \omterm{def:b1:continuity:continuous}{continue} sur $\intcc{a}{b}$, \omterm{def:b1:derivative:def}{dérivable} sur
$\intoo{a}{b}$. Il existe $c \in \intoo{a}{b}$ tel que
\[
f(b) - f(a) = f'(c)\,(b - a) .
\]
\emph{Inégalité des accroissements finis~:} si de plus $m \leq f' \leq M$ sur
$\intoo{a}{b}$, alors $m(b-a) \leq f(b) - f(a) \leq M(b-a)$~; en
particulier $\abs{f'} \leq K$ entraîne que $f$ est $K$-lipschitzienne.
\index{théorème des accroissements finis}
\end{theorem}

\begin{proof}
Appliquons Rolle à $g(x) = f(x) - \frac{f(b) - f(a)}{b - a}(x - a)$~: $g$
est \omterm{def:b1:continuity:continuous}{continue} sur $\intcc{a}{b}$, \omterm{def:b1:derivative:def}{dérivable} à l'\omterm{def:b1:topology:closure}{intérieur}, et $g(a) =
f(a) = g(b)$. Au point $c$ où $g'(c) = 0$~: $f'(c) =
\frac{f(b)-f(a)}{b-a}$. L'inégalité s'obtient en encadrant $f'(c)$~;
l'énoncé lipschitzien l'applique à tout couple de points.
\end{proof}

\begin{example}[La méthode de Newton, c'est celle de Héron]\label{ex:b1:derivative:newton}
La méthode de Newton pour résoudre $f(x) = 0$ remplace la courbe par
sa tangente au point courant $x_n$ et prend la racine de cette
tangente comme point suivant~:
\[
0 = f(x_n) + f'(x_n)(x_{n+1} - x_n)
\quad\Longrightarrow\quad
x_{n+1} = x_n - \frac{f(x_n)}{f'(x_n)} .
\]
Faisons-la tourner sur $f(x) = x^2 - 2$~:
\[
x_{n+1} = x_n - \frac{x_n^2 - 2}{2x_n}
= \frac{x_n}{2} + \frac{1}{x_n}
= \frac12\Bigl(x_n + \frac{2}{x_n}\Bigr) :
\]
c'est exactement l'itération de Héron (\cref{ex:b1:seq:heron}), avec deux
millénaires d'avance. La vitesse quadratique qu'on y observait s'explique
maintenant par l'image de la tangente~: près d'une racine simple, la courbe et
la tangente diffèrent d'une erreur du second ordre, de sorte que chaque étape
élève à peu près l'erreur au carré --- l'énoncé général découle des
majorations de Taylor du \cref{ch:b1:taylor}. L'idée à retenir~:
là où la dichotomie (\cref{ex:b1:continuity:dichotomy}) n'utilise que
la \omterm{def:b1:continuity:continuous}{continuité} et gagne un bit par étape, Newton dépense une
\omterm{def:b1:derivative:def}{dérivée} pour doubler le nombre de décimales correctes à chaque étape.
\end{example}

\begin{example}[Le théorème des accroissements finis comme estimateur]\label{ex:b1:derivative:mvtestimate}
Quelle est la taille de $\sqrt{101}$~? Appliquons le théorème à $f(t) = \sqrt t$
sur $\intcc{100}{101}$~: pour un certain $c \in \intoo{100}{101}$,
\[
\sqrt{101} - 10 = \frac{1}{2\sqrt c},
\qquad\text{donc}\qquad
\frac{1}{2\sqrt{101}} < \sqrt{101} - 10 < \frac{1}{20} = 0.05 ,
\]
et puisque $\sqrt{101} < 10.05$, la borne de gauche dépasse
$\frac{1}{20.1} > 0.0497$~: ainsi $10.0497 < \sqrt{101} < 10.05$
(valeur exacte $10.049875\dots$) --- trois décimales correctes pour une seule
évaluation de \omterm{def:b1:derivative:def}{dérivée}. De même $\abs{\sin a - \sin b} \leq
\abs{a - b}$ (majoration $\abs{\cos}\leq 1$)~: les estimations
lipschitziennes utilisées depuis le \cref{ch:b1:seq} sont toutes ce théorème.
L'idée à retenir~: le \omterm{thm:b1:derivative:mvt}{théorème des accroissements finis} est une formule de
Taylor à l'ordre zéro --- il échange un point inconnu $c$ contre une inégalité
solide, et le \cref{ch:b1:taylor} itérera exactement cet échange.
\end{example}

\begin{corollary}[Critère de monotonie]\label{cor:b1:derivative:monotone}
Soit $f$ \omterm{def:b1:continuity:continuous}{continue} sur $I$, \omterm{def:b1:derivative:def}{dérivable} sur l'\omterm{def:b1:topology:closure}{intérieur}.
\begin{enumerate}
  \item $f' \geq 0$ sur l'\omterm{def:b1:topology:closure}{intérieur} $\iff$ $f$ est croissante~; $f' = 0$
        $\iff$ $f$ constante.
  \item Si $f' > 0$ sauf en un nombre fini de points où elle s'annule,
        $f$ est \emph{strictement} croissante.
\end{enumerate}
\end{corollary}

\begin{proof}
Si $f' \geq 0$~: pour $x < y$ dans $I$, le \omterm{thm:b1:derivative:mvt}{théorème des accroissements finis} sur
$\intcc{x}{y}$ donne $f(y) - f(x) = f'(c)(y - x) \geq 0$. Réciproquement,
les taux d'accroissement d'une fonction croissante sont $\geq 0$, donc leurs
limites aussi. Cas constant~: appliquons ce qui précède à $f'$ et
$-f' \geq 0$. Version stricte~: $f$ est croissante~; une égalité $f(x) =
f(y)$ pour $x < y$ figerait $f$ sur $\intcc{x}{y}$, y forçant $f' =
0$ --- une infinité de points.
\end{proof}

\begin{example}[Dérivées égales, fonctions différentes]\label{ex:b1:derivative:intervalmatters}
Sur $\R^* = \intoo{-\infty}{0} \cup \intoo{0}{+\infty}$, les deux fonctions
$f(x) = \ln\abs x$ et $g(x) = \ln\abs x + \mathbf{1}_{x>0}$
(on ajoute $1$ sur la seule demi-droite de droite) vérifient $f' = g' =
\frac1x$. Elles ne diffèrent pas d'une constante~: le critère
«~$f' = 0 \implies f$ constante~» est un énoncé d'\emph{\omterm{prop:b1:reals:intervals}{intervalle}}
--- sa démonstration fait tourner le \omterm{thm:b1:derivative:mvt}{théorème des accroissements finis} entre
deux points, ce qui exige que tout le segment qui les joint soit dans le
domaine. Sur chaque demi-droite séparément, les primitives de
$\frac1x$ sont $\ln\abs x + c$, avec une constante par demi-droite,
soit deux constantes indépendantes en tout. Le \cref{ch:b1:integration}
hérite de cette clause en petits caractères~: «~la~» primitive d'une fonction
est bien définie à une constante près \emph{sur chaque \omterm{prop:b1:reals:intervals}{intervalle} de son
domaine}, et les tables de primitives supposent silencieusement la connexité.
\end{example}

\begin{example}[La stricte monotonie gratuitement]\label{ex:b1:derivative:strictness}
$x \mapsto x^3$ est \emph{strictement} croissante sur $\R$ bien
que sa \omterm{def:b1:derivative:def}{dérivée} s'annule en $0$~: la clause du critère
«~$f' > 0$ sauf en un nombre fini de points~» est précisément conçue
pour de tels points plats. En revanche, $f' \geq 0$ seul ne donne que
la croissance au sens large (une fonction constante convient), et
une \omterm{def:b1:derivative:def}{dérivée} nulle sur tout un \omterm{prop:b1:reals:intervals}{sous-intervalle} y fige bel et bien la
fonction. La règle pratique~: pour affirmer la stricte
monotonie, dresser la liste des zéros de $f'$~; un nombre fini (ou,
plus généralement, aucun sur un \omterm{prop:b1:reals:intervals}{sous-intervalle}) est inoffensif, un
\omterm{prop:b1:reals:intervals}{intervalle} de zéros est fatal.
\end{example}

\begin{example}[Une étude de variations complète]\label{ex:b1:derivative:variations}
Étudions $f(x) = x^3 - 3x + 1$ sur $\R$. \omterm{def:b1:derivative:def}{Dérivée}~: $f'(x) =
3(x^2 - 1)$, positive sur $\intoo{-\infty}{-1}$, négative sur
$\intoo{-1}{1}$, positive sur $\intoo{1}{+\infty}$~: d'après le
critère de monotonie, $f$ croît, puis décroît, puis
croît, avec un maximum local $f(-1) = 3$ et un minimum local
$f(1) = -1$. Limites~: $\mp\infty$ en $\mp\infty$. Conséquences,
lues sur le tableau de variations à l'aide du \omterm{thm:b1:continuity:ivt}{théorème des valeurs
intermédiaires} sur chaque branche monotone~: $f$ s'annule exactement une fois
dans chacun de
\[
\intoo{-\infty}{-1}, \qquad \intoo{-1}{1}, \qquad
\intoo{1}{+\infty}
\]
(les valeurs aux jonctions sont de signes opposés~: $3 > 0 > -1$),
donc l'équation $x^3 - 3x + 1 = 0$ a exactement trois racines
réelles~; numériquement, elles sont voisines de $-1.88$, $0.35$, $1.53$. L'idée
à retenir~: un tableau de variations est un \emph{instrument de démonstration},
non un croquis --- branche monotone plus changement de signe égale
exactement une racine, et le tableau énumère les branches de
façon exhaustive.
\end{example}

\begin{theorem}[Formule de Leibniz]\label{thm:b1:derivative:leibniz}
Si $f$ et $g$ sont $n$ fois \omterm{def:b1:derivative:def}{dérivables}, $fg$ l'est aussi,
et\index{formule de Leibniz}
\[
(fg)^{(n)} = \sum_{k=0}^{n} \binom nk f^{(k)}\, g^{(n-k)} .
\]
\end{theorem}

\begin{proof}
Récurrence sur $n$, exactement parallèle à celle de la formule du
binôme. Le cas $n = 1$ est la règle du produit. En supposant la formule au rang
$n$, dérivons une fois de plus~:
\[
(fg)^{(n+1)} = \sum_{k=0}^{n} \binom nk
\Bigl( f^{(k+1)} g^{(n-k)} + f^{(k)} g^{(n-k+1)} \Bigr),
\]
puis réindexons la première somme avec $j = k + 1$ et regroupons le
coefficient de $f^{(j)} g^{(n+1-j)}$~: c'est $\binom{n}{j-1} +
\binom nj = \binom{n+1}{j}$ par la règle de Pascal
(\cref{prop:b1:counting:identities}), les termes de bord $j = 0$
et $j = n + 1$ portant $\binom{n+1}{0} = \binom{n+1}{n+1} =
1$ comme il se doit.
\end{proof}

\begin{example}[Leibniz à l'œuvre]\label{ex:b1:derivative:leibnizaction}
Calculons $\bigl(x^2 \eu^x\bigr)^{(n)}$ pour $n \geq 2$. Prenons $f =
x^2$, dont les \omterm{def:b1:derivative:def}{dérivées} meurent vite ($f' = 2x$, $f'' = 2$,
$f^{(k)} = 0$ pour $k \geq 3$), et $g = \eu^x$~: seuls trois termes
de la somme de Leibniz survivent,
\[
\bigl(x^2\eu^x\bigr)^{(n)}
= \binom n0 x^2 \eu^x + \binom n1 (2x)\,\eu^x + \binom n2\,
2\,\eu^x
= \eu^x\bigl(x^2 + 2nx + n(n-1)\bigr).
\]
Vérification en $n = 1$~: $\eu^x(x^2 + 2x)$, qui est bien
$(x^2\eu^x)'$. L'idée à retenir~: employer Leibniz quand l'un des facteurs
est un \omterm{def:b1:poly:def}{polynôme} --- la somme n'a alors que $\deg + 1$ termes, et
la formule est une forme close, non une identité abstraite. (Pour deux
facteurs infatigables comme $\eu^x\sin x$, les exponentielles
complexes du \cref{ch:b1:complex} sont le meilleur outil.)
\end{example}

\section{Convexité}

\begin{definition}\label{def:b1:derivative:convex}
$f \colon I \to \R$ est \emph{convexe}\index{fonction convexe} lorsque
toute corde est au-dessus du graphe~:
\[
\forall x, y \in I,\ \forall t \in \intcc{0}{1}, \quad
f\bigl(tx + (1-t)y\bigr) \leq t f(x) + (1-t) f(y).
\]
($f$ est \emph{concave} lorsque $-f$ est convexe.)
\end{definition}

\begin{theorem}[Caractérisations différentielles]\label{thm:b1:derivative:convexchar}
Soit $f$ \omterm{def:b1:derivative:def}{dérivable} sur $I$. Les \omterm{def:b1:logic:statement}{assertions} suivantes sont équivalentes~:
\begin{enumerate}
  \item $f$ est \omterm{def:b1:derivative:convex}{convexe}~;
  \item $f'$ est croissante sur $I$~;
  \item le graphe est au-dessus de toute tangente~: $f(y) \geq f(x) +
        f'(x)(y - x)$ pour tous $x, y \in I$.
\end{enumerate}
Si $f$ est deux fois \omterm{def:b1:derivative:def}{dérivable}~: $f$ \omterm{def:b1:derivative:convex}{convexe} $\iff f'' \geq 0$.
\end{theorem}

\begin{proof}
(1 $\Rightarrow$ 3) La \omterm{def:b1:derivative:convex}{convexité} s'écrit $\frac{f(x + t(y-x)) -
f(x)}{t} \leq f(y) - f(x)$ pour $t \in \intoc{0}{1}$~; faisons $t \to 0^+$~:
$f'(x)(y - x) \leq f(y) - f(x)$.

(3 $\Rightarrow$ 2) Pour $x < y$, les deux inégalités de tangente en $x$
et en $y$ donnent $f'(x)(y-x) \leq f(y) - f(x) \leq f'(y)(y - x)$,
d'où $f'(x) \leq f'(y)$.

(2 $\Rightarrow$ 1) Fixons $x < y$ et $t \in \intoo{0}{1}$, et posons $z
= tx + (1-t)y \in \intoo{x}{y}$. Par le \omterm{thm:b1:derivative:mvt}{théorème des accroissements finis} sur
$\intcc{x}{z}$ et $\intcc{z}{y}$~: il existe $c_1 < z < c_2$ avec
\[
\frac{f(z) - f(x)}{z - x} = f'(c_1) \leq f'(c_2)
= \frac{f(y) - f(z)}{y - z} ,
\]
et, en chassant les dénominateurs ($z - x = (1-t)(y-x)$, $y - z = t(y-x)$),
on retrouve exactement l'inégalité de \omterm{def:b1:derivative:convex}{convexité}.

Cas deux fois \omterm{def:b1:derivative:def}{dérivable}~: $f'' \geq 0 \iff f'$ croissante
(\cref{cor:b1:derivative:monotone}).
\end{proof}

\begin{example}[Inégalités de convexité classiques]\label{ex:b1:derivative:convexineq}
$\exp$ est \omterm{def:b1:derivative:convex}{convexe} ($\exp'' = \exp > 0$)~: sa tangente en $0$ donne
$\eu^x \geq 1 + x$ pour tout $x$. $\ln$ est concave~: sa tangente en $1$
donne $\ln x \leq x - 1$~; ses cordes donnent, pour $0 < a \leq b$,
l'inégalité entre moyenne géométrique et moyenne arithmétique~: en prenant $t =
\frac12$ dans la concavité,
\[
\ln\frac{a + b}{2} \geq \frac{\ln a + \ln b}{2} = \ln\sqrt{ab},
\qquad\text{donc}\qquad
\sqrt{ab} \leq \frac{a+b}{2} .
\]
L'inégalité arithmético-géométrique générale est l'\cref{exo:b1:derivative:9}.
\end{example}

\begin{example}[Une inégalité de convexité fabriquée de toutes pièces]\label{ex:b1:derivative:xlnx}
La fonction $f(t) = t\ln t$ est \omterm{def:b1:derivative:convex}{convexe} sur $\intoo{0}{+\infty}$~:
$f''(t) = \frac1t > 0$. Son inégalité de milieu, multipliée par
$2$, s'écrit~: pour tous $a, b > 0$,
\[
a\ln a + b\ln b \;\geq\; (a + b)\,\ln\frac{a + b}{2} ,
\]
avec égalité si et seulement si $a = b$ (\omterm{def:b1:derivative:convex}{convexité} stricte). Essai~: $a =
1$, $b = 3$ donne $3\ln 3 = 3.296$ contre $4\ln 2 = 2.773$.
Cette inégalité anodine est le cas à deux points de la comparaison
d'\emph{entropie} qui reparaît avec l'inégalité de Jensen
(\cref{exo:b1:derivative:9}) et dans les asymptotiques
de la théorie de l'information du volume de Licence 3. L'idée
à retenir~: pour fabriquer une inégalité, trouver une fonction
dont la \omterm{def:b1:derivative:def}{dérivée} seconde a un signe et écrire ce que dit la \omterm{def:b1:derivative:convex}{convexité}
--- la caractérisation différentielle transforme une seule vérification de
signe en une infinité d'inégalités.
\end{example}

\begin{remark}[Pièges classiques avec les dérivées]
(i) \emph{Une \omterm{def:b1:derivative:def}{dérivée} strictement positive en un point ne donne pas la
monotonie au \omterm{def:b1:topology:open}{voisinage} de ce point}~: $f(x) = \frac x2 + x^2\sin\frac1x$ (avec
$f(0) = 0$) vérifie $f'(0) = \frac12 > 0$, et pourtant
\[
f'(x) = \frac12 + 2x\sin\frac1x - \cos\frac1x
\]
vaut $-\frac12$ en chaque $x_n = \frac{1}{2\pi n}$~: tout
\omterm{def:b1:topology:open}{voisinage} de $0$ contient des descentes. La monotonie exige $f'
\geq 0$ \emph{sur un \omterm{prop:b1:reals:intervals}{intervalle}}
(\cref{cor:b1:derivative:monotone})~; le signe ponctuel ne
contrôle que la traversée de la tangente. (ii) \emph{Les trois
hypothèses de Rolle sont toutes actives}~: $\abs x$ sur $\intcc{-1}{1}$
(pas de \omterm{def:b1:derivative:def}{dérivabilité} à l'\omterm{def:b1:topology:closure}{intérieur}), $x$ sur $\intcc{0}{1}$ (valeurs aux
bornes distinctes) et $x - \lfloor x\rfloor$ sur $\intcc{0}{1}$ (la \omterm{def:b1:continuity:continuous}{continuité}
tombe en défaut en $1$) brisent chacune exactement une hypothèse et la
conclusion. (iii) \emph{Les \omterm{def:b1:derivative:def}{dérivées} peuvent être discontinues, mais
pas n'importe comment}~: $f'$ peut osciller
(\cref{ex:b1:derivative:notc1}) tout en satisfaisant toujours la
propriété des valeurs intermédiaires (Darboux,
\cref{exo:b1:derivative:10})~: une \omterm{def:b1:derivative:def}{dérivée} ne saute jamais --- si
l'on calcule une «~limite de \omterm{def:b1:derivative:def}{dérivée}~» unilatérale présentant un saut,
c'est qu'on a dérivé une fonction non \omterm{def:b1:derivative:def}{dérivable}. (iv) \emph{La
formule de la réciproque exige $f' \neq 0$}~: $x \mapsto x^3$ est une
\omterm{def:b1:logic:inj}{bijection} lisse strictement croissante dont la réciproque $x^{1/3}$ a une
tangente verticale en $0$ --- la \omterm{def:b1:derivative:def}{dérivabilité} de la réciproque se
perd exactement là où $f'$ s'annule
(\cref{thm:b1:derivative:inverse}).
\end{remark}

\begin{remark}[Où le théorème des accroissements finis travaille ensuite]
Presque tout énoncé quantitatif des chapitres à venir est le \omterm{thm:b1:derivative:mvt}{théorème
des accroissements finis} de ce chapitre déguisé~: le théorème fondamental
de l'analyse (\cref{ch:b1:integration}) dérive la fonction aire
et conclut par le critère de monotonie~; l'inégalité de
Taylor--Lagrange (\cref{ch:b1:taylor}) est le \omterm{thm:b1:derivative:mvt}{théorème des accroissements
finis} itéré $n$ fois~; l'analyse d'erreur de la méthode de Newton
et des itérations de point fixe (\cref{exo:b1:derivative:11}) en est
la forme lipschitzienne~; et le devoir maison de ce chapitre
(\cref{pb:b1:derivative:1}) transforme la même majoration lipschitzienne en
théorie des nombres --- une inégalité de répulsion entre \omterm{pb:b1:logic:1}{nombres
algébriques} et rationnels, qui livre le premier \omterm{pb:b1:logic:1}{nombre transcendant}
de l'histoire. Dans le volume de Licence 2, l'\emph{inégalité} des
accroissements finis survit en plusieurs variables là où l'égalité
ne survit pas.
\end{remark}

\begin{example}[L'inégalité de Young par la concavité]\label{ex:b1:derivative:young}
Soient $p, q > 1$ avec $\frac1p + \frac1q = 1$. Pour tous $a, b > 0$~:
\[
ab \;\leq\; \frac{a^p}{p} + \frac{b^q}{q} .
\]
Démonstration par une seule application de la concavité de $\ln$ avec les poids
$\frac1p, \frac1q$ (l'inégalité de Jensen à deux points, comme dans
l'\cref{exo:b1:derivative:9})~:
\[
\ln\Bigl(\frac{a^p}{p} + \frac{b^q}{q}\Bigr)
\;\geq\; \frac1p \ln(a^p) + \frac1q \ln(b^q)
= \ln a + \ln b = \ln(ab),
\]
et la croissance de $\ln$ convertit l'inégalité entre logarithmes en
l'énoncé voulu~; égalité si et seulement si $a^p = b^q$ (concavité stricte). Le
cas $p = q = 2$ est l'inégalité arithmético-géométrique $ab \leq
\frac{a^2 + b^2}{2}$ déguisée. L'idée à retenir~: l'inégalité de Young
est la graine algébrique des inégalités de Hölder et de Minkowski
du volume de Licence 2 --- un seul énoncé de concavité
sur $\ln$, moissonné pour les normes.
\end{example}

\begin{remark}[Perspectives à l'intérieur de ce volume]
La \omterm{def:b1:derivative:def}{dérivée} acquiert trois nouvelles vies avant la fin du volume.
Au \cref{ch:b1:taylor}, elle s'itère~: $n$ \omterm{def:b1:derivative:def}{dérivées} en un point
se compriment en un \omterm{def:b1:poly:def}{polynôme} plus une erreur contrôlée, et le
\omterm{thm:b1:derivative:mvt}{théorème des accroissements finis} devient le reste de Lagrange. Au
\cref{ch:b1:curves}, la dérivation devient géométrique~: pour une
courbe paramétrée $t \mapsto (x(t), y(t))$, le couple $(x'(t),
y'(t))$ est un \emph{vecteur} vitesse, la tangence devient
colinéarité, et les points critiques deviennent des points de rebroussement à
classer. Au \cref{ch:b1:multivar}, on gèle une variable à la fois~: les
\omterm{def:b1:derivative:def}{dérivées} partielles rejouent ce chapitre deux fois, et la tangente
s'épanouit en plan tangent. Ces trois chapitres héritent de la même
grammaire --- approximation linéaire locale plus terme d'erreur ---
parlée ici pour la première fois.
\end{remark}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b1:derivative:1}
Dériver (en précisant les domaines)~: $x^x$~;
$\;\ln\bigl(x + \sqrt{x^2+1}\bigr)$~; $\;\arctan\frac{1}{x}$~;
$\;\sqrt{1 + \eu^{2x}}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:2}
Compléter l'\cref{ex:b1:derivative:notc1}~: démontrer que $f(x) = x^2
\sin\frac1x$, $f(0) = 0$, est \omterm{def:b1:derivative:def}{dérivable} en $0$ avec $f'(0) = 0$,
et que $f'$ n'a pas de limite en $0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:3}
À l'aide du \omterm{thm:b1:derivative:mvt}{théorème des accroissements finis} ou des inégalités de tangente,
démontrer que pour tout $x > 0$~:
\[
\frac{x}{1 + x} < \ln(1 + x) < x .
\]
En déduire $\lim_{n\to\infty} \bigl(1 + \frac xn\bigr)^n = \eu^x$ pour
tout $x > 0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:4}
Soit $P$ un \omterm{def:b1:poly:def}{polynôme} réel ayant $k$ racines réelles distinctes. Démontrer que
$P'$ a au moins $k - 1$ racines réelles distinctes, entrelacées avec celles
de $P$. En déduire que si $P$ a toutes ses racines réelles, $P'$ aussi.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:5}
Soit $f$ \omterm{def:b1:derivative:def}{dérivable} sur $\R$ avec $f' (x)\to \ell$ quand $x \to
+\infty$. Démontrer que $\frac{f(x)}{x} \to \ell$ \emph{(\omterm{thm:b1:derivative:mvt}{théorème des
accroissements finis} sur $\intcc{A}{x}$)}. A-t-on aussi $f(x+1) - f(x) \to \ell$~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:6}
(Un Rolle discret) Soit $f$ $n$ fois \omterm{def:b1:derivative:def}{dérivable} sur $I$ et
s'annulant en $n + 1$ points distincts. Démontrer que $f^{(n)}$ s'annule au
moins une fois. Application~: un \omterm{def:b1:poly:def}{polynôme} de degré $\leq n$ s'annulant en
$n+1$ points est nul (encore une fois).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:7}
Soit $f$ deux fois \omterm{def:b1:derivative:def}{dérivable} sur $\intcc{a}{b}$ avec $f(a) = f(b)
= 0$ et $f(x_0) > 0$ pour un certain $x_0$ \omterm{def:b1:topology:closure}{intérieur}. Démontrer que $f''(c) <
0$ pour un certain $c \in \intoo{a}{b}$. \emph{(Deux \omterm{thm:b1:derivative:mvt}{théorèmes des accroissements
finis} et une comparaison de pentes.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:8}
Étudier la fonction $f(x) = \dfrac{\ln x}{x}$ sur
$\intoo{0}{+\infty}$~: variations, limites, maximum. En déduire que $a^b
> b^a$ pour tous réels $\eu \leq a < b$, et trancher le fameux cas
particulier~: lequel de $\eu^\pi$, $\pi^\eu$ est le plus grand~?
Confronter aux petits couples d'entiers $(2,3)$ et $(2,4)$~: pourquoi se
comportent-ils différemment~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:9}
(Inégalité arithmético-géométrique) À l'aide de la concavité de $\ln$ avec
des poids quelconques (inégalité de Jensen à $n$ points, à démontrer par
récurrence sur $n$), montrer que pour des réels strictement positifs $a_1, \dots, a_n$~:
\[
\sqrt[n]{a_1 a_2 \cdots a_n} \leq \frac{a_1 + \dots + a_n}{n},
\]
avec égalité si et seulement si tous les $a_i$ sont égaux.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:10}
(Darboux~: les \omterm{def:b1:derivative:def}{dérivées} prennent les valeurs intermédiaires) Soit $f$
\omterm{def:b1:derivative:def}{dérivable} sur $I$ et $a < b$ dans $I$ avec $f'(a) < v < f'(b)$.
En considérant $g(x) = f(x) - vx$ et le point où $g$ atteint
son minimum sur $\intcc{a}{b}$, démontrer que $f'(c) = v$ pour un certain $c
\in \intoo{a}{b}$ --- alors même que $f'$ n'est pas nécessairement \omterm{def:b1:continuity:continuous}{continue}
(\cref{exo:b1:derivative:2}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:11}
Soit $f \colon \R \to \R$ \omterm{def:b1:derivative:def}{dérivable} avec $\abs{f'(x)} \leq k
< 1$ pour tout $x$ (une \emph{contraction}). Démontrer que $f$ a exactement
un point fixe $\ell$, et que toute suite $u_{n+1} = f(u_n)$
converge vers $\ell$ avec $\abs{u_n - \ell} \leq k^n\abs{u_0 -
\ell}$. \emph{(Existence~: appliquer le \omterm{thm:b1:continuity:ivt}{théorème des valeurs intermédiaires} à
$f(x) - x$ sur un grand segment, en utilisant la majoration lipschitzienne~; ou
utiliser la complétude avec le critère de Cauchy.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:12}
(\omterm{thm:b1:derivative:mvt}{Théorème des accroissements finis} généralisé de Cauchy et règle de l'Hospital)
\begin{enumerate}
  \item Soient $f, g$ \omterm{def:b1:continuity:continuous}{continues} sur $\intcc{a}{b}$,
        \omterm{def:b1:derivative:def}{dérivables} sur $\intoo{a}{b}$, avec $g'$ ne s'annulant jamais
        sur cet \omterm{prop:b1:reals:intervals}{intervalle}. Démontrer que $g(b) \neq g(a)$ et qu'un certain $c \in
        \intoo{a}{b}$ vérifie
        \[
        \frac{f(b) - f(a)}{g(b) - g(a)} = \frac{f'(c)}{g'(c)}
        \]
        \emph{(appliquer Rolle à $h = f - \lambda g$ pour la bonne
        constante $\lambda$)}.
  \item En déduire la règle de l'Hospital sous la forme $\frac00$ en un
        point~: si $f(a) = g(a) = 0$ et
        $\frac{f'(x)}{g'(x)} \to \ell$ quand $x \to a^+$, alors
        $\frac{f(x)}{g(x)} \to \ell$.
  \item Montrer que la réciproque est fausse~: pour $f(x) = x^2\sin\frac1x$
        ($f(0) = 0$) et $g(x) = x$, le quotient
        $\frac{f}{g}$ a une limite en $0$ mais
        $\frac{f'}{g'}$ n'en a pas.
\end{enumerate}
\end{exercise}

\section{Problème~: l'inégalité de Liouville et le premier
nombre transcendant}

\begin{problem}[{Devoir maison --- les \omterm{pb:b1:logic:1}{nombres algébriques} repoussent les
rationnels~: $\abs{x - p/q} \geq C/q^d$, et la transcendance de
$\sum 10^{-n!}$}]
\label{pb:b1:derivative:1}
Un nombre réel est \emph{algébrique} lorsqu'il est racine d'un \omterm{def:b1:poly:def}{polynôme}
non nul à coefficients entiers, et \emph{\omterm{pb:b1:logic:1}{transcendant}}
sinon. En 1844, Liouville produisit le premier nombre dont la
transcendance ait jamais été \emph{démontrée}, et le moteur de sa
démonstration est le \omterm{thm:b1:derivative:mvt}{théorème des accroissements finis} de ce chapitre~: un
\omterm{pb:b1:logic:1}{nombre algébrique} de degré $d$ ne peut pas être approché par des rationnels
mieux que $C/q^d$ --- de sorte qu'un nombre approchable \emph{plus vite que
toute puissance} ne peut pas être algébrique. Ce problème construit
l'inégalité, construit le nombre de Liouville $L = 0.110001000\dots$ (des uns
aux positions factorielles, via la machinerie des chiffres du
\cref{pb:b1:reals:1}), démontre sa transcendance, et se termine par la
démonstration rivale de Cantor et des bornes effectives pour $\sqrt2$ et
$2^{1/3}$.\index{nombre transcendant}\index{inégalité de
Liouville}\index{nombre algébrique}

\medskip
\noindent\textbf{Partie I --- Avec quelle précision peut-on approcher les
rationnels~?}
\begin{enumerate}
  \item Montrer que deux rationnels distincts $\frac ab \neq \frac pq$
        (écrits avec $b, q \geq 1$) vérifient $\bigl|\frac ab -
        \frac pq\bigr| \geq \frac{1}{bq}$. En déduire~: si $x =
        \frac ab$ et $0 < \bigl|x - \frac pq\bigr| <
        \frac{1}{bq}$, aucun tel $\frac pq$ n'existe --- un rationnel
        repousse tous les autres rationnels à l'échelle $\frac 1q$.
  \item Démontrer que pour \emph{tout} rationnel $\frac pq$ ($q \geq
        1$)~: $\bigl|\sqrt2 - \frac pq\bigr| \geq
        \frac{1}{4q^2}$ \emph{(si la distance dépasse $1$, c'est
        clair~; sinon, majorer $\abs{\sqrt2 + p/q} < 4$ et
        utiliser l'entier non nul $\abs{p^2 - 2q^2} \geq 1$)}.
  \item Dans l'autre sens~: vérifier que $(p, q) \mapsto (p +
        2q, p + q)$ préserve $\abs{p^2 - 2q^2} = 1$, engendrer
        à partir de $(1,1)$ les couples $(3,2)$, $(7,5)$, $(17,12)$,
        $(41,29)$, $(99,70)$, et montrer que chacun vérifie
        \[
        \Bigl|\sqrt2 - \frac pq\Bigr| =
        \frac{1}{q^2\,(\sqrt2 + p/q)} < \frac{1}{2q^2} :
        \]
        soit une infinité d'approximations d'ordre $2$. Avec
        la question 2~: l'exposant d'approximation de $\sqrt 2$ vaut
        \emph{exactement} $2$.
  \item (Dirichlet) Soit $x$ irrationnel et $N \in \N^*$.
        Considérer les $N + 1$ parties fractionnaires de $0, x, 2x,
        \dots, Nx$ dans les $N$ boîtes $\intco{\frac kN}{\frac{k +
        1}{N}}$~: par le \omterm{cor:b1:counting:pigeonhole}{principe des tiroirs}
        (\cref{cor:b1:counting:pigeonhole}), deux d'entre elles tombent
        dans une même boîte. En déduire $q \leq N$ et $p$ avec $\abs{qx - p} <
        \frac 1N$, donc une infinité de rationnels avec
        $\bigl|x - \frac pq\bigr| < \frac{1}{q^2}$~: \emph{tout}
        irrationnel est approchable à l'ordre $2$.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- L'\omterm{pb:b1:derivative:1}{inégalité de Liouville}.} Soit $x$
irrationnel et algébrique.
\begin{enumerate}[resume]
  \item Montrer que parmi les \omterm{def:b1:poly:def}{polynômes} non nuls à coefficients entiers
        s'annulant en $x$ il y en a un, disons $P$ de degré $d$,
        \emph{sans racine rationnelle}~; et vérifier $d \geq 2$
        \emph{(factoriser un facteur $X - \frac ab$ sur $\Q$ et
        chasser les dénominateurs~; un degré $1$ rendrait $x$
        rationnel)}.
  \item Montrer que pour tout rationnel $\frac pq$ ($q \geq 1$)~:
        $\bigl|P\bigl(\frac pq\bigr)\bigr| \geq \frac{1}{q^d}$
        \emph{($q^d P(p/q)$ est un entier non nul)}.
  \item Soit $M = \max_{\intcc{x-1}{x+1}} \abs{P'}$
        (\cref{thm:b1:continuity:evt}). À l'aide du \omterm{thm:b1:derivative:mvt}{théorème des
        accroissements finis} entre $x$ et $\frac pq$, démontrer
        l'\emph{\omterm{pb:b1:derivative:1}{inégalité de Liouville}}~: avec $C = \min\bigl(1,
        \frac 1M\bigr) > 0$,
        \[
        \Bigl| x - \frac pq \Bigr| \geq \frac{C}{q^{\,d}}
        \qquad\text{pour tout rationnel } \frac pq,\ q \geq 1 .
        \]
  \item On dit que $x$ est un \emph{nombre de Liouville} lorsque, pour tout $n \in
        \N$, il existe un rationnel $\frac pq$ avec $q \geq 2$ et
        $0 < \bigl|x - \frac pq\bigr| < q^{-n}$. Démontrer qu'un
        nombre de Liouville est irrationnel \emph{(question 1~: choisir
        $n$ tel que $2^{\,n-1} > b$)}.
  \item Démontrer le théorème de Liouville~: \emph{un nombre de Liouville est
        \omterm{pb:b1:logic:1}{transcendant}} \emph{(combiner les questions 7 et 8~: l'inégalité
        $C < q^{\,d-n}$ tombe en défaut pour $n$ grand)}.
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Le nombre $L$.}
\begin{enumerate}[resume]
  \item Soit $L$ la valeur (au sens du
        \cref{pb:b1:reals:1}) de la suite de chiffres décimaux valant
        $1$ aux positions $n!$ ($n = 1, 2, 3, \dots$)
        et $0$ ailleurs, c'est-à-dire $L = \sup_k t_k$ avec $t_k =
        \sum_{n=1}^{k} 10^{-n!}$. Écrire les $25$ premiers
        chiffres de $L$.
  \item Démontrer l'encadrement du reste, pour tout $k \geq 1$~:
        \[
        10^{-(k+1)!} \;\leq\; L - t_k \;\leq\;
        \frac{10}{9}\,10^{-(k+1)!} \;<\; 2\cdot 10^{-(k+1)!}
        \]
        \emph{(majorer toute somme partielle au-delà de $t_k$ par une somme
        géométrique finie)}.
  \item Écrire $t_k = \frac{p_k}{q_k}$ avec $q_k = 10^{k!}$.
        Montrer $0 < L - \frac{p_k}{q_k} < \frac{2}{q_k^{\,k+1}}$,
        et conclure que $L$ est un nombre de Liouville au sens
        de la question 8.
  \item Conclure~: $L$ est \omterm{pb:b1:logic:1}{transcendant} --- le premier exemple
        explicite de l'histoire (Liouville, 1844). Recouper directement
        son irrationalité~: ses chiffres ne sont pas périodiques à partir
        d'un certain rang (écarts croissants, comme au \cref{pb:b1:reals:1},
        question 20).
  \item Généraliser~: remplacer chaque chiffre $1$ par un chiffre non nul
        quelconque $d_n \in \intint{1}{9}$. Montrer que la valeur est
        encore un nombre de Liouville, et en déduire --- par l'argument
        diagonal du \cref{pb:b1:reals:1} (question 22) appliqué
        à ces choix de chiffres --- qu'il existe une infinité non
        dénombrable de \omterm{pb:b1:logic:1}{nombres transcendants} de cette forme.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La hiérarchie des ordres
d'approximation.} On dit que $x$ est \emph{approchable à l'ordre $\mu$} lorsque,
pour une certaine constante $c > 0$, une infinité de rationnels vérifient
$\bigl|x - \frac pq\bigr| < \frac{c}{q^{\mu}}$.
\begin{enumerate}[resume]
  \item Assembler la hiérarchie à partir des parties I à III~: les rationnels sont
        approchables à l'ordre $1$ et pas mieux~; $\sqrt 2$ à
        l'ordre $2$ et pas mieux~; tout irrationnel à l'ordre
        $2$ au moins~; un \omterm{pb:b1:logic:1}{nombre algébrique} de degré $d$ à aucun ordre
        au-delà de $d$~; les nombres de Liouville à tout ordre. Justifier
        chaque affirmation en citant la question pertinente.
  \item Montrer que $L + r$ est un nombre de Liouville pour tout
        rationnel $r = \frac ab$ \emph{(translater les
        approximants~: les nouveaux dénominateurs sont $b\,q_k$)}.
        En déduire que les nombres de Liouville --- donc les \omterm{pb:b1:logic:1}{nombres
        transcendants} --- sont \omterm{def:b1:topology:dense}{denses} dans $\R$.
  \item (Cantor, 1874) Démontrer que l'\omterm{def:b1:logic:sets}{ensemble} des \omterm{pb:b1:logic:1}{nombres algébriques}
        est dénombrable~: il n'y a qu'un nombre fini de \omterm{def:b1:poly:def}{polynômes} à
        coefficients entiers dont le degré plus la somme des $\abs{\text{coefficients}}$
        est majoré par $h$, chacun ayant au plus $\deg$ racines~; une
        réunion dénombrable d'\omterm{def:b1:counting:card}{ensembles finis} est dénombrable. Puisqu'aucune
        suite n'épuise $\R$ (\cref{pb:b1:reals:1}, question
        22), il existe des \omterm{pb:b1:logic:1}{nombres transcendants} --- ils forment même un
        \omterm{def:b1:logic:sets}{ensemble} non dénombrable. Comparer les deux démonstrations~: que donne
        celle de Liouville que celle de Cantor ne peut pas donner~?
  \item Démontrer directement, à partir de la question 2, que $\sqrt 2$ n'est
        \emph{pas} un nombre de Liouville \emph{(pour $n \geq 3$,
        l'inégalité $q^{-n} > \frac{1}{4q^2}$ borne $q$~;
        il ne reste alors qu'un nombre fini de rationnels candidats, tous
        à distance strictement positive de $\sqrt2$)}. Généraliser~:
        aucun \omterm{pb:b1:logic:1}{nombre algébrique} n'est de Liouville.
\end{enumerate}

\medskip
\noindent\textbf{Partie V --- Constantes effectives.}
\begin{enumerate}[resume]
  \item Pour le couple de Pell $(99, 70)$~: vérifier $99^2 - 2\cdot70^2
        = 1$ et évaluer l'erreur exacte
        \[
        \sqrt2 - \frac{99}{70}
        = \frac{-1}{70^2\,\bigl(\sqrt2 + \frac{99}{70}\bigr)},
        \qquad
        \Bigl|\sqrt 2 - \frac{99}{70}\Bigr| \approx 7.2\cdot
        10^{-5} :
        \]
        cinq chiffres corrects pour une fraction à trois chiffres.
  \item Faire tourner la partie II sur $x = 2^{1/3}$, $P = X^3 - 2$~: vérifier que $P$
        n'a pas de racine rationnelle, majorer $M = \max_{\intcc{x-1}{x+1}}
        3t^2 \leq 3\,(1 + 2^{1/3})^2 < 16$, et conclure l'inégalité
        effective
        \[
        \Bigl| 2^{1/3} - \frac pq \Bigr| \geq
        \frac{1}{16\,q^3} \qquad \text{pour tout } \frac pq .
        \]
  \item Le gain~: montrer que tout rationnel approchant $2^{1/3}$
        à $10^{-6}$ près doit avoir un dénominateur $q \geq 40$.
  \item Montrer que la base $10$ n'y est pour rien~: l'analogue
        binaire $\sum_{n\geq1} 2^{-n!}$ (valeur de la suite binaire
        ayant des uns aux positions factorielles) est aussi un nombre de
        Liouville, donc \omterm{pb:b1:logic:1}{transcendant}.
\end{enumerate}

\medskip
\noindent\textbf{Partie VI --- \omterm{def:b1:topology:closure}{Frontières} et synthèse.}
\begin{enumerate}[resume]
  \item Soit $x^\dagger$ la valeur de la suite décimale ayant des
        uns exactement aux positions $3^k$ ($k \geq 0$). Montrer
        que $x^\dagger$ est approchable à l'ordre $3$, et déduire
        de l'\omterm{pb:b1:derivative:1}{inégalité de Liouville} que $x^\dagger$ n'est ni
        rationnel ni un irrationnel quadratique. Expliquer pourquoi la
        méthode s'arrête là~: l'ordre $3$ est compatible avec
        l'algébricité de degré $\geq 3$, et combler cet écart
        (tout exposant $> 2$ suffit, pour tout \omterm{pb:b1:logic:1}{nombre algébrique})
        est le théorème de Roth, bien au-delà de ce volume.
  \item Quantifier Cantor~: montrer que les \omterm{pb:b1:logic:1}{nombres algébriques} de
        degré $\leq d$ fournis par des \omterm{def:b1:poly:def}{polynômes} à coefficients
        dans $\intint{-H}{H}$ sont au plus $d\,(2H + 1)^{d+1}$.
        (C'est cette finitude qui a fait fonctionner la question 17.)
  \item Synthèse, une phrase pour chaque point~: (i) localiser l'unique
        ingrédient analytique de la démonstration de Liouville (quel théorème
        de ce chapitre, utilisé où)~; (ii) énoncer la tension
        qui l'alimente (le caractère entier force $\abs{P(p/q)} \geq
        q^{-d}$, la régularité interdit $\abs{P(p/q)} > M\abs{x -
        p/q}$)~; (iii) opposer les démonstrations de Liouville et de Cantor
        de l'existence de \omterm{pb:b1:logic:1}{nombres transcendants}~; (iv) nommer
        l'endroit où ce volume retrouve le thème --- le devoir
        maison du \cref{ch:b1:integration} démontre l'irrationalité de $\pi$
        par le même étau «~caractère entier contre petitesse~»,
        avec des intégrales à la place des \omterm{def:b1:derivative:def}{dérivées}.
\end{enumerate}
\end{problem}
