\chapter{التطبيقات الخطية}\label{ch:b1:linmaps}

التطبيقات الجديرة بالدراسة بين الفضاءات المتجهية هي المتوافقة
مع البنية: \emph{التطبيقات الخطية}. وفضاءاها الجزئيان
الأساسيان --- \omterm{def:b1:linmaps:kerim}{النواة} \omterm{def:b1:linmaps:kerim}{والصورة} --- يقيسان التباين و
الشمول، وفي البُعد المنتهي تربط مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة}
حجميهما في قانون حفظ واحد. وتختم الفصل الإسقاطات
والتماثلات، ثم الأشكال الخطية والمستويات الفائقة.

\section{التعاريف والخصائص الأولى}

\begin{definition}\label{def:b1:linmaps:def}
ليكن $E, F$ فضاءين متجهيين على $K$. يكون \omterm{def:b1:logic:map}{التطبيق} $u \colon E \to F$
\emph{خطيًا}\index{تطبيق خطي} إذا كان
\[
\forall x, y \in E,\ \forall \lambda \in K, \qquad
u(x + \lambda y) = u(x) + \lambda u(y).
\]
وعندئذ $u(0) = 0$ و $u(\sum \lambda_i x_i) = \sum \lambda_i u(x_i)$.
\omterm{def:b1:logic:sets}{ومجموعة} التطبيقات الخطية $\mathcal{L}(E, F)$ هي نفسها \omterm{def:b1:vspaces:def}{فضاء متجهي}؛
وتركيب التطبيقات الخطية خطي، وثنائي الخطية في الزوج. و
\emph{التشاكل الذاتي} \omterm{def:b1:logic:map}{تطبيق} خطي $E \to E$؛ و\emph{التشاكل التقابلي}
\omterm{def:b1:logic:map}{تطبيق} خطي تقابلي (ويكون معكوسه عندئذ خطيًا تلقائيًا)؛
ويكون $u^{-1}$ لتشاكل تقابلي، وتراكيب التشاكلات التقابلية،
تشاكلات تقابلية.
\end{definition}

\begin{proof}[برهان أن المعكوس خطي]
ليكن $u$ \omterm{def:b1:linmaps:def}{خطيًا} \omterm{def:b1:logic:inj}{وتقابليًا} وليكن $y, y' \in F$ و $\lambda \in K$.
ضع $x = u^{-1}(y)$ و $x' = u^{-1}(y')$. عندئذ
\[
u\bigl(x + \lambda x'\bigr) = u(x) + \lambda u(x')
= y + \lambda y' ,
\]
وبتطبيق $u^{-1}$ على الطرفين: $u^{-1}(y + \lambda y') = x
+ \lambda x' = u^{-1}(y) + \lambda\,u^{-1}(y')$. ولم يُحسب شيء عن
$u^{-1}$: فالخطية تنتقل عبر الخاصية المعرِّفة
للمقدار $u$ وحدها --- وهو نمط جدير بالحفظ،
إذ كثيرًا ما تركب البنيةُ التقابلاتِ بالمجّان.
\end{proof}

\begin{proposition}[التطبيق الخطي معلوم على أساس]\label{prop:b1:linmaps:basis}
ليكن $(e_1, \dots, e_n)$ \omterm{def:b1:vspaces:free}{أساسًا} للمقدار $E$ ولتكن $(v_1, \dots, v_n)$
متجهات كيفية من $F$. يوجد \omterm{def:b1:linmaps:def}{تطبيق خطي} واحد بالضبط $u \colon
E \to F$ يحقق $u(e_i) = v_i$ من أجل كل $i$. وفوق ذلك:
\[
u \text{ متباين} \iff (v_i) \text{ حرة};
\qquad
u \text{ شامل} \iff (v_i) \text{ تولّد } F .
\]
\end{proposition}

\begin{proof}
الوجود والوحدانية: لكل $x$ \omterm{prop:b1:vspaces:coordinates}{إحداثيات} وحيدة $x = \sum
\lambda_i e_i$ (\cref{prop:b1:vspaces:coordinates})؛ وتفرض الخطية
$u(x) = \sum \lambda_i v_i$، وهذه الصيغة تعرّف فعلًا تطبيقًا
\omterm{def:b1:linmaps:def}{خطيًا}.

التباين: حسب \cref{prop:b1:linmaps:kernel} أدناه، يكون $u$
\omterm{def:b1:logic:inj}{متباينًا} إذا وفقط إذا كانت نواته بديهية. والآن يعني $u\bigl(\sum\lambda_i
e_i\bigr) = 0$ بالضبط $\sum\lambda_i v_i = 0$. فإذا كانت $(v_i)$
\omterm{def:b1:vspaces:free}{حرة}، فُرض كل $\lambda_i = 0$، أي تُردّ \omterm{def:b1:linmaps:kerim}{النواة}
إلى $0$: فهو متباين. وإذا كانت $(v_i)$ مرتبطة، فإن علاقة غير بديهية
$\sum\lambda_i v_i = 0$ تنتج المتجهة غير المعدومة
$\sum\lambda_i e_i$ في \omterm{def:b1:linmaps:kerim}{النواة} (لأن $(e_i)$ \omterm{def:b1:vspaces:free}{حرة}): فليس
\omterm{def:b1:logic:inj}{متباينًا}. والشرطان يتطابقان حدًّا بحدّ.

الشمول: \omterm{def:b1:linmaps:kerim}{صورة} $u$ هي \omterm{def:b1:logic:sets}{مجموعة} كل المقادير $\sum\lambda_i
v_i$، أي بالضبط $\operatorname{Vect}(v_1, \dots, v_n)$، وهي
تساوي $F$ إذا وفقط إذا كانت العائلة مولِّدة.
\end{proof}

\begin{definition}[النواة والصورة]\label{def:b1:linmaps:kerim}
من أجل $u \in \mathcal{L}(E, F)$:
\[
\ker u = \{x \in E : u(x) = 0\} \subseteq E,
\qquad
\operatorname{im} u = u(E) \subseteq F ,
\]
وكلاهما \omterm{def:b1:vspaces:subspace}{فضاء جزئي} (بالتحقق المباشر من
المحك).\index{نواة}\index{صورة}
\end{definition}

\begin{method}[النواة والصورة عمليًا]
\label{met:b1:linmaps:kerim}
\emph{\omterm{def:b1:linmaps:kerim}{النواة}}: اكتب $u(x) = 0$ جملةً على \omterm{prop:b1:vspaces:coordinates}{إحداثيات}
(أو معاملات) $x$، وحُلَّ، وعلّم --- فتخرج \omterm{def:b1:linmaps:kerim}{النواة}
مصحوبةً \omterm{def:b1:vspaces:free}{بأساس}
(\cref{met:b1:findim:computedim}). \emph{\omterm{def:b1:linmaps:kerim}{والصورة}}: هي \omterm{def:b1:vspaces:span}{الفضاء المولَّد}
بصور \emph{أيّ} عائلة مولِّدة للمقدار $E$ ---
وعادةً \omterm{def:b1:vspaces:free}{أساس}، ومنه $\operatorname{im} u =
\operatorname{Vect}\bigl(u(e_1), \dots, u(e_n)\bigr)$؛ ثم
احذف الصور الزائدة لاستخراج \omterm{def:b1:vspaces:free}{أساس}. \emph{والاختصار}:
احسب أيّهما أسهل واحصل على بُعد الآخر
بالمجّان بمبرهنة \omterm{def:b1:linmaps:rank}{الرتبة}
(\cref{thm:b1:linmaps:ranknullity})؛ وحين يكون \omterm{def:b1:vspaces:subspace}{فضاء جزئي} مرشّح معقول
\omterm{def:b1:linmaps:kerim}{للصورة} معلومًا، ترقّي مقارنةُ الأبعاد الاحتواءَ السهل
إلى مساواة
(\cref{thm:b1:findim:subspaces}). والاختصاران كلاهما مستعمل في
\cref{ex:b1:linmaps:delta} أدناه.
\end{method}

\begin{proposition}\label{prop:b1:linmaps:kernel}
يكون $u$ \omterm{def:b1:logic:inj}{متباينًا} $\iff$ $\ker u = \{0\}$؛ ويكون $u$ \omterm{def:b1:logic:inj}{شاملًا} $\iff$
$\operatorname{im} u = F$.
\end{proposition}

\begin{proof}
كما من أجل الزمر (\cref{prop:b1:structures:kernel}): $u(x) = u(y) \iff
u(x - y) = 0 \iff x - y \in \ker u$. والنقطة الثانية هي
التعريف.
\end{proof}

\section{مبرهنة الرتبة}

\begin{definition}\label{def:b1:linmaps:rank}
\emph{رتبة}\index{رتبة!رتبة تطبيق خطي} المقدار $u \in \mathcal{L}(E,
F)$ (مع $E$ ذي \omterm{def:b1:findim:def}{بُعد منته}) هي $\operatorname{rk} u = \dim
\operatorname{im} u$ --- وهي كذلك رتبة العائلة
$\bigl(u(e_1), \dots, u(e_n)\bigr)$ من أجل أيّ \omterm{def:b1:vspaces:free}{أساس} $(e_i)$ للمقدار $E$.
\end{definition}

\begin{theorem}[مبرهنة الرتبة]\label{thm:b1:linmaps:ranknullity}
ليكن $E$ \omterm{def:b1:findim:def}{ذا بُعد منته} ولتكن $u \in \mathcal{L}(E, F)$.
عندئذ\index{مبرهنة الرتبة}
\[
\dim E = \dim \ker u + \operatorname{rk} u .
\]
وبدقة أكبر، إذا كان $S$ أيَّ \omterm{def:b1:vspaces:subspace}{فضاء جزئي} متتامّ للمقدار $\ker u$ في
$E$، فإن $u$ يقتصر ليعطي \emph{\omterm{def:b1:structures:morphism}{تشاكلًا} \omterm{def:b1:logic:inj}{تقابليًا}} من $S$ على
$\operatorname{im} u$.
\end{theorem}

\begin{proof}
ليحقق $S$ الشرط $E = \ker u \oplus S$
(\cref{thm:b1:findim:subspaces})، ولتكن $v \colon S \to
\operatorname{im} u$ قصر $u$.

و $v$ متباين: $\ker v = S \cap \ker u = \{0\}$.

و $v$ شامل: فأيّ $u(x)$ مع $x = k + s$ ($k \in \ker u$ و $s
\in S$) يساوي $u(s) = v(s)$.

ومنه \omterm{def:b1:logic:map}{فالتطبيق} $v$ تشاكل تقابلي؛ والتشاكل التقابلي يرسل \omterm{def:b1:vspaces:free}{أساسًا} إلى \omterm{def:b1:vspaces:free}{أساس}
(\cref{prop:b1:linmaps:basis})، ومنه $\dim S = \dim\operatorname{im}
u$، ويختم $\dim E = \dim\ker u + \dim S$.
\end{proof}

\begin{example}[بناء تطبيق حسب المواصفات]
\label{ex:b1:linmaps:tospec}
ابنِ $u \in \mathcal{L}(\R^3)$ يحقق $\ker u =
\operatorname{Vect}(1,1,1)$ و $\operatorname{im} u = \{z =
0\}$. وتحقق من السلامة أولًا: تقتضي مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة} $1 + 2 = 3$ ---
وهذا متسق، ومنه قد يوجد حلّ. اختر \omterm{def:b1:vspaces:free}{أساسًا} مكيَّفًا مع
\omterm{def:b1:linmaps:kerim}{النواة}، وليكن $\bigl((1,1,1),\ e_1,\ e_2\bigr)$
(\cref{ex:b1:findim:completion})، وافرض الصور
(\cref{prop:b1:linmaps:basis}):
\[
u(1,1,1) = 0, \qquad u(e_1) = e_1, \qquad u(e_2) = e_2 .
\]
عندئذ $\ker u \supseteq \operatorname{Vect}(1,1,1)$ و
$\operatorname{im} u = \operatorname{Vect}(e_1, e_2) = \{z =
0\}$؛ وتفرض مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة} $\dim\ker u = 1$، ومنه \omterm{def:b1:linmaps:kerim}{فالنواة}
هي المستقيم المفروض بالضبط. وصراحةً، بتفكيك $(x, y, z)
= z(1,1,1) + (x - z)e_1 + (y - z)e_2$:
\[
u(x, y, z) = (x - z,\ y - z,\ 0).
\]
وتتعمّم الوصفة: \omterm{def:b1:linmaps:def}{فالتطبيق الخطي} ذو \omterm{def:b1:linmaps:kerim}{النواة} المفروضة $N$
\omterm{def:b1:linmaps:kerim}{والصورة} المفروضة $I$ موجود بالضبط عندما يكون $\dim N + \dim I = \dim E$
--- واللزوم هو مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة}، والكفاية هي هذا
البناء.
\end{example}

\begin{corollary}\label{cor:b1:linmaps:samedim}
إذا كان $\dim E = \dim F$ (منتهيًا)، فإنه من أجل $u \in \mathcal{L}(E, F)$:
\[
u \text{ متباين} \iff u \text{ شامل} \iff u \text{
تقابلي}.
\]
وعلى الخصوص يصحّ هذا من أجل التشاكلات الذاتية في البُعد المنتهي. (وهو
يفشل في البُعد غير المنتهي: فعلى $K[X]$، تكون \omterm{def:b1:derivative:def}{المشتقة} شاملة
لا متباينة، ويكون $P \mapsto XP$ \omterm{def:b1:logic:inj}{متباينًا} لا
\omterm{def:b1:logic:inj}{شاملًا}.)
\end{corollary}

\begin{proof}
متباين $\iff \dim\ker u = 0 \iff \operatorname{rk} u = \dim E =
\dim F \iff \operatorname{im} u = F$ (لأن \omterm{def:b1:vspaces:subspace}{الفضاء الجزئي} ذا البُعد الكامل
هو كل شيء، \cref{thm:b1:findim:subspaces}) $\iff$ شامل.
\end{proof}

\begin{example}[المقايسة، بنيويًا]\label{ex:b1:linmaps:interpolation}
ثبّت $x_0, \dots, x_n$ متمايزة ولتكن $u \colon \R_n[X] \to
\R^{n+1}$ و $P \mapsto (P(x_0), \dots, P(x_n))$: وهو خطي. ونواته
هي $\{P : \deg P \leq n,\ n+1 \text{ جذرًا}\} = \{0\}$
(\cref{cor:b1:poly:nroots}). والبُعدان متساويان $n + 1$: ومنه \omterm{def:b1:logic:map}{فالتطبيق} $u$
تشاكل تقابلي --- أي وجود مقايِس لاغرانج \emph{و}وحدانيته
(\cref{thm:b1:poly:lagrange}) في سطر واحد.

والنمط ذو السطر الواحد نفسه يعالج معطيات تخلط القيم و
المشتقات: \omterm{def:b1:logic:map}{فالتطبيق} $v \colon \R_3[X] \to \R^4$ و $P \mapsto \bigl(P(0),
P'(0), P(1), P'(1)\bigr)$ خطي، وتتألف نواته من
كثيرات حدود من الدرجة $\leq 3$ لها جذور مضاعفة عند $0$ \emph{و}
$1$، أي تقبل القسمة على $X^2(X-1)^2$ من الدرجة $4$: ومنه لا يبقى إلا $P =
0$. والبُعدان متساويان مرة أخرى: ومنه فكل رباعية معطيات $(P(0),
P'(0), P(1), P'(1))$ يحققها كثيرُ حدود تكعيبي واحد بالضبط ---
أي مقايسة إرميت، ممنوحةً بحساب \omterm{def:b1:linmaps:kerim}{نواة} قبل
كتابة أيّ صيغة (وتلقى مسألة نهاية الأسبوع في \cref{ch:b1:det}
محدّدَها).
\end{example}

\begin{example}[مبرهنة الرتبة في العمل: مؤثر الفرق]
\label{ex:b1:linmaps:delta}
لتكن $\Delta \colon \R_n[X] \to \R_n[X]$ و $P \mapsto P(X+1) -
P(X)$: وهو خطي. \omterm{def:b1:linmaps:kerim}{والنواة}: إذا كان $\Delta P = 0$، فإن $P(0) = P(1) =
P(2) = \dots$، ومنه فللمقدار $P - P(0)$ جذور لا تُحصى وهو
ينعدم (\cref{cor:b1:poly:nroots}): ومنه \omterm{def:b1:linmaps:kerim}{فالنواة} $\ker\Delta$ هي مستقيم
الثوابت. ومبرهنة \omterm{def:b1:linmaps:rank}{الرتبة}: $\operatorname{rk}\Delta = (n + 1) - 1
= n$. ولأن $\deg \Delta P < \deg P$ من أجل $P$ غير ثابت (لأن الحدود
العليا تتلاشى)، $\operatorname{im}\Delta \subseteq \R_{n-1}[X]$،
وبُعده $n$ بالضبط: ومنه فالاحتواء مساواة.
والنتيجة، دون أيّ حساب للصور العكسية: \emph{كل}
\omterm{def:b1:poly:def}{كثير حدود} $Q$ من الدرجة $\leq n - 1$ هو فرق $Q = P(X+1)
- P(X)$ --- أي إن الدالة الأصلية المتقطّعة موجودة. (وقارن
مسألة نهاية الأسبوع في \cref{ch:b1:vspaces}، حيث قُلب $\Delta$
صراحةً في \omterm{def:b1:vspaces:free}{الأساس} الثنائي.)
\end{example}

\begin{example}[مسك دفاتر الرتبة على امتداد تركيب]
\label{ex:b1:linmaps:rankbookkeeping}
على $\R_2[X]$، ركّب \omterm{def:b1:derivative:def}{المشتقة} $D(P) = P'$ (رتبتها $2$:
\omterm{def:b1:linmaps:kerim}{والصورة} $\R_1[X]$، \omterm{def:b1:linmaps:kerim}{والنواة} الثوابت) مع نفسها. عندئذ يرسل $D
\circ D = D^2$ المقدارَ $P \mapsto P''$، وصورته $\R_0[X]$: \omterm{def:b1:linmaps:rank}{فالرتبة}
$1$. وقارن بالحواصر العامة: يعطي الفجّ
$\operatorname{rk} D^2 \leq \min(2, 2) = 2$؛ وتحاسب الصيغة المضبوطة
في \cref{exo:b1:linmaps:12} الخسارةَ بدقة،
\[
\operatorname{rk} D^2 = \operatorname{rk} D -
\dim\bigl(\ker D \cap \operatorname{im} D\bigr) = 2 - 1 = 1 ,
\]
لأن الثوابت (\omterm{def:b1:linmaps:kerim}{نواة} $D$ الخارجي) تقع داخل
$\R_1[X]$ (\omterm{def:b1:linmaps:kerim}{صورة} $D$ الداخلي) ببُعد $1$. \omterm{def:b1:linmaps:rank}{فالرتبة}
تُفقد بالضبط حيث تكمن \omterm{def:b1:linmaps:kerim}{النواة} الخارجية \omterm{def:b1:linmaps:kerim}{للصورة} الداخلية
--- وهي الجملة الواجب حفظها حين تسيء رتب التركيب السلوك.
\end{example}

\begin{example}[نواة مؤثر أويلر وصورته]
\label{ex:b1:linmaps:euler}
على $\R_n[X]$، لتكن $u(P) = X\,P'$ (وهو خطي: لأن الاشتقاق و
الضرب في $X$ خطيان). \emph{\omterm{def:b1:linmaps:kerim}{النواة}}: يفرض $XP' = 0$ أن $P' =
0$ (لأن جداء كثيرَي حدود لا ينعدم إلا إذا انعدم أحد العاملين)،
ومنه \omterm{def:b1:linmaps:kerim}{فالنواة} $\ker u$ هي مستقيم الثوابت. \emph{\omterm{def:b1:linmaps:kerim}{والصورة}}: على
\omterm{def:b1:vspaces:free}{الأساس} الوحيد الحدّ،
\[
u(X^k) = k\,X^{k} \qquad (k = 0, 1, \dots, n),
\]
ومنه $\operatorname{im} u = \operatorname{Vect}(X, 2X^2, \dots,
nX^n) = \operatorname{Vect}(X, X^2, \dots, X^n)$: أي
كثيرات الحدود ذات الحدّ الثابت المعدوم. وللتحقق إزاء مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة}:
$\operatorname{rk} u = (n + 1) - 1 = n$، وهو فعلًا البُعد
الموجود. وملاحظتان جديرتان بالحفظ. الأولى، أنه هنا
$\operatorname{im} u \oplus \ker u = \R_n[X]$ --- لكن هذه \emph{مصادفة
سعيدة} لهذا المؤثر، لا مبرهنة: فمن أجل $v(P) = P'$ الشبيه
بالإزاحة على $\R_1[X]$، $\ker v = \operatorname{im}
v = \R_0[X]$ والمجموع ليس مباشرًا. والثانية، أن العلاقة
$u(X^k) = kX^k$ تقول إن كل وحيد حدّ يُضرب في سلّم فحسب بالمقدار $u$ ---
أي أساسٌ مكيَّف مع \omterm{def:b1:logic:map}{التطبيق}، وهو جرثومة فكرة القيمة الذاتية
المطوَّرة في مجلّد السنة~الثانية.
\end{example}

\section{الإسقاطات والتماثلات}

\begin{definition}\label{def:b1:linmaps:projection}
ليكن $E = F \oplus G$. \emph{الإسقاط على $F$ على امتداد
$G$}\index{إسقاط} يرسل $x = f + g$ (بالتفكيك الوحيد) إلى
$p(x) = f$؛ و\emph{التماثل}\index{تماثل} المرتبط به هو $s(x)
= f - g$. وكلاهما خطي، و $s = 2p - \mathrm{id}$.
\end{definition}

\begin{theorem}[التمييز الجبري]\label{thm:b1:linmaps:projchar}
\begin{enumerate}
  \item يكون التشاكل الذاتي $p$ إسقاطًا (على $F$ ما على امتداد
        $G$ ما) إذا وفقط إذا كان $p \circ p = p$؛ وعندئذ $F =
        \operatorname{im} p = \ker(p - \mathrm{id})$ و $G = \ker
        p$.
  \item يكون التشاكل الذاتي $s$ تماثلًا إذا وفقط إذا كان $s \circ s
        = \mathrm{id}$؛ وعندئذ $E = \ker(s - \mathrm{id}) \oplus
        \ker(s + \mathrm{id})$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) يحقق \omterm{def:b1:linmaps:projection}{الإسقاط} $p(f + g) = f$ و $p(f) = f$: ومنه $p^2 = p$.
وبالعكس، ليكن $p^2 = p$؛ وضع $F = \operatorname{im} p$ و $G = \ker
p$. وكل $x$ يُكتب $x = p(x) + (x - p(x))$ مع $p(x) \in F$ و
$p\bigl(x - p(x)\bigr) = p(x) - p^2(x) = 0$: ومنه $E = F + G$. وإذا كان $y \in
F \cap G$: فإن $y = p(z)$ و $p(y) = 0$، ومنه $y = p(z) = p^2(z) = p(y) =
0$: \omterm{def:b1:vspaces:sum}{فالمجموع مباشر}، ويكون $p$ الإسقاطَ على $F$ على امتداد $G$.
وأخيرًا على $F$: يعطي $y = p(z)$ أن $p(y) = y$، ومنه $F \subseteq
\ker(p - \mathrm{id})$، وبالعكس يضع $p(y) = y$ المقدارَ $y$ في
\omterm{def:b1:linmaps:kerim}{الصورة}.

(2) التقابل $s = 2p - \mathrm{id}$ و $p = \frac{s +
\mathrm{id}}2$ تقابلٌ بين التشاكلات الذاتية، وتحته
\[
s^2 = 4p^2 - 4p + \mathrm{id} = \mathrm{id}
\iff 4p^2 = 4p \iff p^2 = p :
\]
فتقابل التماثلاتُ الإسقاطاتِ بالضبط. وبترجمة
الفضاءات الجزئية: $s(x) = x \iff p(x) = x$، ومنه $\ker(s - \mathrm{id}) =
\operatorname{im} p = F$؛ و $s(x) = -x \iff 2p(x) = 0 \iff x
\in \ker p = G$، ومنه $\ker(s + \mathrm{id}) = G$. ويصير \omterm{def:b1:vspaces:sum}{المجموع المباشر}
$E = F \oplus G$ في النقطة (1) التفكيكَ المعلن
إلى المتجهات الصامدة والمتجهات المعكوسة للمقدار $s$.
\end{proof}

\begin{example}[إسقاط وتماثله، صراحةً]
\label{ex:b1:linmaps:projexplicit}
في $\R^2$، أسقط على $F = \operatorname{Vect}(1,1)$ على امتداد $G =
\operatorname{Vect}(0,1)$. فكّك $(x, y) = a(1,1) + b(0,1)$:
تعطي الإحداثية الأولى $a = x$، والثانية $b = y - x$.
ومنه
\[
p(x, y) = (x, x),
\qquad
s(x, y) = 2p(x,y) - (x,y) = (x,\ 2x - y).
\]
وتحقق من الجبر: $p(p(x,y)) = p(x,x) = (x,x)$، و $s(s(x,y)) =
s(x, 2x - y) = (x, 2x - (2x - y)) = (x, y)$. وهندسيًا، يكون $s$
هو «الانعكاس المائل» عبر المستقيم $y = x$ في
الاتجاه العمودي: فهو يثبّت $F$ نقطةً نقطة ويعكس $G$. ولو
أسقطنا على $F$ نفسه على امتداد $G' =
\operatorname{Vect}(1,-1)$ بدل ذلك، لتغيرت الصيغة إلى
$p'(x,y) = \bigl(\frac{x+y}2, \frac{x+y}2\bigr)$: \omterm{def:b1:linmaps:projection}{فالإسقاط}
يتحدد بصورته \emph{و}نواته، لا \omterm{def:b1:linmaps:kerim}{بالصورة}
وحدها أبدًا.
\end{example}

\section{الأشكال الخطية والمستويات الفائقة}

\begin{definition}\label{def:b1:linmaps:forms}
\emph{الشكل الخطي}\index{شكل خطي} على $E$ هو \omterm{def:b1:linmaps:def}{تطبيق خطي}
$\varphi \colon E \to K$. و\emph{المستوي الفائق}\index{مستوٍ فائق} في
$E$ ($\dim E = n$) هو \omterm{def:b1:vspaces:subspace}{فضاء جزئي} بُعده $n - 1$.
\end{definition}

\begin{example}[شكل تقويم ومستويه الفائق]
\label{ex:b1:linmaps:evaluationform}
على $\R_2[X]$، يكون التقويم $\varphi(P) = P(2)$ شكلًا
\omterm{def:b1:linmaps:def}{خطيًا} غير معدوم ($\varphi(1) = 1$). ونواته هي \omterm{def:b1:linmaps:forms}{المستوي الفائق}
لكثيرات الحدود المنعدمة عند $2$، أي (بمبرهنة العامل،
\cref{thm:b1:poly:factor}) مضاعفات $X - 2$ داخل
$\R_2[X]$:
\[
\ker\varphi = \operatorname{Vect}\bigl(X - 2,\ X(X - 2)\bigr),
\qquad \dim = 2 .
\]
وبالإحداثيات على $(1, X, X^2)$، $\varphi(a + bX + cX^2) = a +
2b + 4c$: فكل \omterm{def:b1:linmaps:forms}{شكل خطي} على فضاء ذي \omterm{def:b1:findim:def}{بُعد منته} هو،
ما إن يُثبَّت \omterm{def:b1:vspaces:free}{أساس}، عبارةٌ خطية مثبَّتة في
\omterm{prop:b1:vspaces:coordinates}{الإحداثيات} --- فالأشكال «متجهات سطرية»، كما
سيجعله \cref{ch:b1:matrices} حرفيًا، وسطر المعاملات
هنا، $(1, 2, 4)$، سطرُ فاندرموند: فأشكال التقويم
هي كيف تدخل نظرية المقايسة في مسألة نهاية الأسبوع في \cref{ch:b1:det}
إلى الجبر الخطي.
\end{example}

\begin{theorem}\label{thm:b1:linmaps:hyperplanes}
المستويات الفائقة في $E$ هي بالضبط نوى الأشكال الخطية
غير المعدومة. ولشكلين غير معدومين \omterm{def:b1:linmaps:kerim}{النواة} نفسها إذا وفقط إذا كانا
متناسبين.
\end{theorem}

\begin{proof}
إذا كان $\varphi \neq 0$: فإن $\operatorname{rk}\varphi = 1$ (لأن \omterm{def:b1:linmaps:kerim}{الصورة} \omterm{def:b1:vspaces:subspace}{فضاء
جزئي} غير معدوم من $K$)، ومنه $\dim\ker\varphi = n - 1$: أي \omterm{def:b1:linmaps:forms}{مستوٍ فائق}.
وبالعكس، ليكن $H$ مستويًا فائقًا، ولتكن $(e_1, \dots, e_{n-1})$ \omterm{def:b1:vspaces:free}{أساسًا}
للمقدار $H$ متمَّمًا بالمقدار $e_n$: فللشكل «الإحداثية الأخيرة» نواةٌ
$H$.

والشكلان المتناسبان يتقاسمان نواتهما. وبالعكس، افترض
$\ker\varphi = \ker\psi = H$ واختر $a \notin H$: فكل $x$ يُكتب
$x = h + \lambda a$ (لأن $E = H \oplus Ka$)، و
\[
\varphi(x) = \lambda \varphi(a), \qquad \psi(x) = \lambda\psi(a):
\]
ومنه $\varphi = \frac{\varphi(a)}{\psi(a)}\,\psi$.
\end{proof}

\begin{example}\label{ex:b1:linmaps:hyperplane}
في $K^n$، يكون \omterm{def:b1:linmaps:forms}{المستوي الفائق} \omterm{def:b1:logic:sets}{مجموعة} حلول $\{a_1 x_1 + \dots + a_n
x_n = 0\}$ مع عدم انعدام كل $a_i$ --- أي المعادلة المألوفة
لمستوٍ مارّ بالمبدأ في $\R^3$. وفي فضاءات الدوال، تعرّف أشكال
التقويم $P \mapsto P(1)$ أو $f \mapsto \int_0^1 f$ مستويات فائقة
في $\R_n[X]$ وفي $C(\intcc{0}{1})$ (قارن \cref{exo:b1:findim:6}).
\end{example}

\begin{example}[مستوٍ فائق واحد، معالَجًا بثلاث كيفيات]
\label{ex:b1:linmaps:hyperplanerun}
خذ $\varphi(x, y, z) = x - 2y + 3z$ على $\R^3$ و $H =
\ker\varphi$. \emph{\omterm{def:b1:vspaces:free}{الأساس}}: حُلَّ $x = 2y - 3z$:
\[
(2y - 3z,\ y,\ z) = y\,(2, 1, 0) + z\,(-3, 0, 1),
\]
ومتجهتان حرتان: ومنه $\dim H = 2$، أي \omterm{def:b1:linmaps:forms}{مستوٍ فائق}، كما
يتنبّأ \cref{thm:b1:linmaps:hyperplanes} من $\varphi \neq
0$. \emph{ومستقيم متتامّ}: أيّ متجهة خارج $H$ تولّد واحدًا،
مثلًا $a = (1, 0, 0)$ ($\varphi(a) = 1 \neq 0$)؛ و
تفكيك $v$ كيفية صريح:
\[
v = \underbrace{\bigl(v - \varphi(v)\,a\bigr)}_{\in\,H}
+ \underbrace{\varphi(v)\,a}_{\in\,\operatorname{Vect}(a)},
\]
لأن $\varphi\bigl(v - \varphi(v)a\bigr) = \varphi(v) -
\varphi(v)\varphi(a) = 0$. \emph{والتناسب}: إذا كان $\psi(x,y,z)
= -2x + 4y - 6z$، فإن $\psi = -2\varphi$ ولكليهما \omterm{def:b1:linmaps:kerim}{النواة}
$H$؛ وبالعكس، فأيّ شكل ينعدم على $H$ هو مضاعف للمقدار
$\varphi$ (\cref{exo:b1:linmaps:8}) --- فمعادلة
\omterm{def:b1:linmaps:forms}{مستوٍ فائق} وحيدة بغضّ النظر عن السلّم، وهي واقعة تُستعمل باستمرار من أجل
المستويات في الهندسة.
\end{example}

\begin{remark}[مزالق شائعة]
\label{rem:b1:linmaps:pitfalls}
\emph{\omterm{def:b1:linmaps:kerim}{النواة} \omterm{def:b1:linmaps:kerim}{والصورة} تعيشان في فضاءين مختلفين}: $\ker u
\subseteq E$ و $\operatorname{im} u \subseteq F$؛ والمجموع $\ker u
+ \operatorname{im} u$ لا معنى له إلا من أجل التشاكلات الذاتية، و
حتى عندئذ ليس عليه أن يكون مباشرًا (فللتطبيق $u(x, y) = (y, 0)$ لدينا $\ker u
= \operatorname{im} u$؛ ويميّز \cref{exo:b1:linmaps:7}
متى يقع الانفصال).
\emph{المقدار $u^2 = 0$ لا يعني $u = 0$}: \omterm{def:b1:logic:map}{فالتطبيق} $u(x,y) =
(y, 0)$ نفسه مربعه معدوم دون أن ينعدم --- وما يقوله $u^2 = 0$
حقًا هو $\operatorname{im} u \subseteq \ker u$
(\cref{exo:b1:linmaps:5}).
\emph{المتباين $\iff$ الشامل يحتاج إلى بُعدين منتهيين
متساويين}: فعلى $K[X]$، تكون \omterm{def:b1:derivative:def}{المشتقة} شاملة لا
متباينة، ويكون $P \mapsto XP$ \omterm{def:b1:logic:inj}{متباينًا} لا \omterm{def:b1:logic:inj}{شاملًا}
(\cref{cor:b1:linmaps:samedim})؛ وبين فضاءين ذوَي بُعدين
\emph{مختلفين}، يكون أحد الاستلزامين محالًا ببساطة
($\operatorname{rk} u \leq \min(\dim E, \dim F)$).
\emph{فرض الصور يجري على \omterm{def:b1:vspaces:free}{أساس}، لا على أيّ عائلة}:
فطلب $u(1, 0) = a$ و $u(0, 1) = b$ و $u(1, 1) = c$
يحدّد $u$ تحديدًا زائدًا ما لم يكن $c = a + b$؛ \omterm{def:b1:linmaps:def}{فالتطبيق الخطي} حرّ على
\omterm{def:b1:vspaces:free}{أساس}، ومستعبَد في كل ما عداه.
\emph{\omterm{def:b1:linmaps:rank}{الرتبة} لا تُحفظ بالتركيب}: فهي لا يمكن إلا أن تهبط،
$\operatorname{rk}(vu) \leq \min(\operatorname{rk} u,
\operatorname{rk} v)$ (\cref{exo:b1:linmaps:4})، والخسارة المضبوطة
مقيسةٌ في \cref{exo:b1:linmaps:12}.
\end{remark}

\begin{remark}[إلى أين تذهب هذه التطبيقات]
\label{rem:b1:linmaps:whereused}
التطبيقات الخطية على وشك أن تصير \emph{مصفوفات}: فما إن تُثبَّت الأسس،
يرمّز \cref{ch:b1:matrices} كلَّ $u \in \mathcal{L}(E,
F)$ بجدول مستطيل، ويصير التركيب جداءَ
مصفوفات --- ثم تحرّك مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة} نظرية الجمل
الخطية في \cref{ch:b1:det}. وتعود الإسقاطات في
\cref{ch:b1:euclid} في حالتها الخاصة الأنفع، وهي
\omterm{def:b1:linmaps:projection}{الإسقاط} \emph{المتعامد}، حيث تُختار \omterm{def:b1:linmaps:kerim}{النواة}
عمودية على \omterm{def:b1:linmaps:kerim}{الصورة}. وتدفع مسألة نهاية الأسبوع أدناه
جبرَ المساقط إلى أبعد ما تبلغه أدوات السنة الأولى، حتى مبرهنة
فيتينغ المساعدة؛ ويذهب مجلّد السنة~الثانية أبعد بالأثر
وبنظرية القيم الذاتية، وتكون فيها المساقط على الفضاءات الجزئية
المستقرة لبنات البناء الأساسية.
\end{remark}

\begin{remark}[منظورات داخل الكتاب 3: مبرهنة الرتبة ثلاث مرات أخرى]
\label{rem:b1:linmaps:perspectives}
سوف يُعاد قراءة قانون الحفظ $\dim E = \dim\ker u + \operatorname{rk} u$
ثلاث مرات قبل نهاية المجلّد. ففي
\cref{ch:b1:det} يصير شكلَ مجموعات الحلول: فللجملة
المتوافقة ذات $p$ مجهولًا \omterm{def:b1:linmaps:rank}{ورتبة} $r$ مجموعةُ حلول
بُعدها $p - r$ --- أي بُعد \omterm{def:b1:linmaps:kerim}{النواة} متنكّرًا. وفي
\cref{ch:b1:euclid} ينقسم انقسامًا متعامدًا، $\dim F + \dim
F^\perp = \dim E$، ويحرّك كل حساب مسافة. وفي
مسألة نهاية الأسبوع في \cref{ch:b1:multivar}، يكون محاسبَ
المربعات الصغرى: $n$ رصدًا، و $2$ وسيطًا مقايَسين،
و $n - 2$ بُعدًا للباقي، وتكون متطابقة فيثاغورس $\norm b^2 = \norm p^2 + \norm{b - p}^2$ ظلَّ
مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة} الإقليدي. مبرهنة واحدة وأربعة أزياء.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:1}
أيّ التطبيقات خطي؟
\begin{enumerate}
  \item $\R^2 \to \R^2$، $(x, y) \mapsto (x + y, x - 2y)$؛
  \item $\R^2 \to \R$، $(x, y) \mapsto xy$؛
  \item $\R[X] \to \R[X]$، $P \mapsto P' + XP$؛
  \item $\mathcal{F}(\R,\R) \to \R$، $f \mapsto f(3)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:2}
لتكن $u \colon \R^3 \to \R^3$ و $(x,y,z) \mapsto (x + y - z,\; 2x + y
+ z,\; 3x + 2y)$. حدّد $\ker u$ (\omterm{def:b1:vspaces:free}{بأساس} وبُعد)، و
$\operatorname{rk} u$، \omterm{def:b1:vspaces:free}{وأساسًا} للمقدار $\operatorname{im} u$. وهل $u$
متباين؟ وهل هو شامل؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:3}
لتكن $u \colon \R_n[X] \to \R_n[X]$ مع $P \mapsto P - P'$. برهن على أن
$u$ تشاكل تقابلي: مرةً عبر $\ker u$، ومرةً بإظهار
المعكوس \emph{(انظر في $P + P' + P'' + \dots$)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:linmaps:4}
لتكن $u \in \mathcal{L}(E, F)$ و $v \in \mathcal{L}(F, G)$، والفضاءات
ذات \omterm{def:b1:findim:def}{بُعد منته}. برهن على:
\[
\operatorname{rk}(v \circ u) \leq
\min\bigl(\operatorname{rk} u,\ \operatorname{rk} v\bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:5}
ليكن $u$ \omterm{def:b1:structures:morphism}{تشاكلًا} ذاتيًا للمقدار $E$ (ذي \omterm{def:b1:findim:def}{بُعد منته}) مع $u^2 =
0$. برهن على أن $\operatorname{im} u \subseteq \ker u$، ومنه
$\operatorname{rk} u \leq \frac{\dim E}{2}$. ومن أجل $E = \R^2$، أعطِ
مثالًا بالمساواة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:6}
ليكن $p, q$ إسقاطين للمقدار $E$ مع $p \circ q = q \circ p$.
برهن على أن $p \circ q$ \omterm{def:b1:linmaps:projection}{إسقاط}، مع
\[
\operatorname{im}(pq) = \operatorname{im} p \cap \operatorname{im}
q ,
\qquad
\ker (pq) = \ker p + \ker q .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:7}
ليكن $u \in \mathcal{L}(E)$ مع $E$ ذي \omterm{def:b1:findim:def}{بُعد منته}. برهن على
تكافؤ:
\begin{enumerate}
  \item $E = \ker u \oplus \operatorname{im} u$؛
  \item $\ker u = \ker u^2$؛
  \item $\operatorname{im} u = \operatorname{im} u^2$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:8}
ليكن $\varphi, \psi$ شكلين خطيين على $E$ مع $\ker\varphi
\subseteq \ker\psi$. برهن على $\psi = \lambda\varphi$ من أجل
$\lambda \in K$ ما (بما في ذلك الحالات المنحلّة).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:linmaps:9}
ليكن $u \in \mathcal{L}(E)$ مع $\dim E = n$، ولنفترض $u^n = 0$
لكن $u^{n-1} \neq 0$ (أي تشاكل ذاتي \emph{\omterm{pb:b1:linmaps:1}{معدوم القوى} أكبريًا}).
اختر $x$ يحقق $u^{n-1}(x) \neq 0$؛ وبرهن على أن $\bigl(x, u(x), \dots,
u^{n-1}(x)\bigr)$ \omterm{def:b1:vspaces:free}{أساس} للمقدار $E$. \emph{(طبّق قوى $u$ على
تركيبة معدومة، بادئًا بالمقدار $u^{n-1}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:linmaps:10}
لتكن $f \in \mathcal{L}(\R^n)$ مع $f \circ f = -\mathrm{id}$.
\begin{enumerate}
  \item برهن على أن $f$ تشاكل تقابلي وعلى أنه لا يوجد $x \neq 0$
        يحقق $f(x) = \lambda x$ مع $\lambda \in \R$.
  \item برهن على أن $n$ زوجي. \emph{إرشاد: اختر $x_1 \neq 0$؛ وبيّن
        أن $\operatorname{Vect}(x_1, f(x_1))$ مستوٍ مستقر تحت
        $f$؛ واختر $x_2$ خارجه وكرّر، مبرهنًا على أن
        $\bigl(x_1, f(x_1), x_2, f(x_2), \dots\bigr)$ تبقى \omterm{def:b1:vspaces:free}{حرة}.}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:linmaps:11}
لتكن $u, v \in \mathcal{L}(E, F)$، والفضاءات ذات \omterm{def:b1:findim:def}{بُعد منته}.
برهن على الحصر ذي الطرفين
\[
\abs{\operatorname{rk} u - \operatorname{rk} v}
\;\leq\; \operatorname{rk}(u + v)
\;\leq\; \operatorname{rk} u + \operatorname{rk} v .
\]
\emph{(من أجل \omterm{def:b1:reals:bounds}{الحاصر الأعلى}، قارن $\operatorname{im}(u+v)$ بالمقدار
$\operatorname{im} u + \operatorname{im} v$؛ ومن أجل الأدنى،
طبّق \omterm{def:b1:reals:bounds}{الحاصر الأعلى} بذكاء.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:linmaps:12}
(متراجحة فروبينيوس) لتكن $u \in \mathcal{L}(E, F)$ و $w \in
\mathcal{L}(F, G)$ و $v \in \mathcal{L}(G, H)$، وكل الفضاءات
ذات \omterm{def:b1:findim:def}{بُعد منته}. برهن على الصيغة المضبوطة
\[
\operatorname{rk}(v \circ w) = \operatorname{rk} w -
\dim\bigl(\ker v \cap \operatorname{im} w\bigr),
\]
واستنتج متراجحة فروبينيوس
\[
\operatorname{rk}(v \circ w) + \operatorname{rk}(w \circ u)
\;\leq\; \operatorname{rk} w + \operatorname{rk}(v \circ w \circ
u) .
\]
وتحقق من أن الحالة $w = \mathrm{id}_F$ هي متراجحة
سيلفستر، المبرهن عليها في صورة مصفوفات في \cref{exo:b1:matrices:10}.
\end{exercise}

\section{مسألة: حساب المساقط ومبرهنة فيتينغ المساعدة}

\begin{problem}\label{pb:b1:linmaps:1}
الإسقاطات هي التشاكلات الذاتية التي تنتجها المجاميع المباشرة،
وبالعكس: فكل متطابقة $E = F_1 \oplus \dots \oplus F_k$ هي
سرًّا عائلةُ مساقط مجموعها \omterm{def:b1:logic:map}{التطبيق} المطابق. وتطوّر هذه
المسألة ذلك القاموس --- جبر مسقط واحد،
ومسقطين، و $k$ مسقطًا --- ثم تطبّق أفكار
الاستقرار نفسها على تشاكل ذاتي كيفي وتبرهن على
\emph{مبرهنة فيتينغ المساعدة}: أي إن كل تشاكل ذاتي لفضاء
ذي \omterm{def:b1:findim:def}{بُعد منته} ينقسم إلى جزء \omterm{pb:b1:linmaps:1}{معدوم القوى} وجزء
قابل للقلب. وفي كل ما يأتي، $E$ \omterm{def:b1:vspaces:def}{فضاء متجهي} على $K$ بُعده
$n$، وتعني \emph{المسقط} $p \in \mathcal{L}(E)$
مع $p^2 = p$ (\cref{thm:b1:linmaps:projchar}).
\index{إسقاط}\index{مبرهنة فيتينغ المساعدة}\index{معدوم القوى}

\medskip
\noindent\textbf{الجزء 1 --- الجبر حول مسقط واحد.}
ليكن $p$ مسقطًا، مع $p \neq 0$ و $p \neq \mathrm{id}$.
\begin{enumerate}
  \item بيّن أن $\mathrm{id} - p$ مسقط وعيّن
        $\operatorname{im}(\mathrm{id} - p)$ و
        $\ker(\mathrm{id} - p)$.
  \item احسب $(\lambda\,\mathrm{id} + \mu\,p)^2$ وحدّد
        كل الأزواج $(\lambda, \mu) \in K^2$ التي يكون من أجلها
        $\lambda\,\mathrm{id} + \mu\,p$ مسقطًا.
  \item بيّن أن المستوي $\operatorname{Vect}(\mathrm{id}, p)$
        من $\mathcal{L}(E)$ مستقر بالتركيب، وأنه
        من أجل كل \omterm{def:b1:poly:def}{كثير حدود} $Q \in K[X]$،
        \[
        Q(p) = Q(0)\,\mathrm{id} + \bigl(Q(1) -
        Q(0)\bigr)\,p .
        \]
  \item حدّد من أجل أيّ $(\lambda, \mu)$ يكون \omterm{def:b1:logic:map}{التطبيق}
        $\lambda\,\mathrm{id} + \mu\,p$ قابلًا للقلب، وأعطِ
        معكوسه في \omterm{def:b1:linmaps:kerim}{الصورة} $\alpha\,\mathrm{id} + \beta\,p$.
        وفسّر الجواب عبر فعل
        $\lambda\,\mathrm{id} + \mu\,p$ على $\operatorname{im} p$
        وعلى $\ker p$.
  \item ليكن $p'$ مسقطًا آخر له \emph{\omterm{def:b1:linmaps:kerim}{الصورة} نفسها}
        $\operatorname{im} p' = \operatorname{im} p$. بيّن أن
        $p\,p' = p'$ و $p'\,p = p$. وماذا تقول هاتان المتطابقتان
        عن تركيب الإسقاطات على \omterm{def:b1:vspaces:subspace}{الفضاء الجزئي} نفسه
        على امتداد نوى مختلفة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- مسقطان.} ليكن $p, q$
مسقطين للمقدار $E$؛ ولنفترض أن المميّز ليس $2$ (وهذا صحيح
من أجل $K = \R, \C$).
\begin{enumerate}[resume]
  \item افترض أن $p + q$ مسقط. وبنشر $(p + q)^2$،
        بيّن $pq + qp = 0$؛ وبالتركيب مع $p$ من اليسار، ثم
        من اليمين، استنتج $pq = qp$، واختم بأن $pq = qp =
        0$.
  \item وبالعكس، افترض $pq = qp = 0$. بيّن أن $p + q$
        مسقط، مع
        \[
        \operatorname{im}(p + q) = \operatorname{im} p \oplus
        \operatorname{im} q,
        \qquad
        \ker(p + q) = \ker p \cap \ker q .
        \]
  \item بيّن أن $p - q$ مسقط إذا وفقط إذا كان $pq = qp
        = q$. \emph{(طبّق السؤالين 6--7 على $\mathrm{id} - p$
        و $q$.)}
  \item بيّن المعنى الهندسي للمقدار $pq = qp = q$: فهو يصحّ إذا
        وفقط إذا كان $\operatorname{im} q \subseteq
        \operatorname{im} p$ و $\ker p \subseteq \ker q$. (ويُكتب
        عندئذ $q \leq p$: «$q$ يسقط على أقلّ، على امتداد
        أكثر».)
  \item ولنفترض الآن أن $p, q$ يتبادلان. وتذكّر من
        \cref{exo:b1:linmaps:6} أن $pq$ هو المسقط على
        $\operatorname{im} p \cap \operatorname{im} q$ على امتداد
        $\ker p + \ker q$. بيّن أن $r = p + q - pq$
        مسقط مع
        \[
        \operatorname{im} r = \operatorname{im} p +
        \operatorname{im} q,
        \qquad
        \ker r = \ker p \cap \ker q .
        \]
        \emph{(انظر في $\mathrm{id} - r = (\mathrm{id} -
        p)(\mathrm{id} - q)$.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- تفكيكات \omterm{def:b1:logic:map}{التطبيق} المطابق.}
\begin{enumerate}[resume]
  \item ليكن $E = F_1 \oplus \dots \oplus F_k$ و، من أجل $x = x_1 +
        \dots + x_k$ (بالتفكيك الوحيد، $x_i \in F_i$)، ضع
        $p_i(x) = x_i$. بيّن أن كل $p_i$ مسقط، وأن
        $p_i p_j = 0$ من أجل $i \neq j$، وأن $p_1 + \dots + p_k
        = \mathrm{id}$؛ وعيّن $\operatorname{im} p_i$ و
        $\ker p_i$.
  \item وبالعكس، لتحقق $p_1, \dots, p_k \in \mathcal{L}(E)$
        الشرطين $p_1 + \dots + p_k = \mathrm{id}$ و $p_i p_j =
        0$ من أجل كل $i \neq j$. بيّن أن كل $p_i$
        مسقط وأن $E = \operatorname{im} p_1 \oplus
        \dots \oplus \operatorname{im} p_k$.
  \item مسقطان مع $p + q = \mathrm{id}$: بيّن أن $pq
        = qp = 0$ يتحقق تلقائيًا.
  \item ثلاثة مساقط مع $p + q + r = \mathrm{id}$: بيّن
        أن $p + q$ مسقط، واستنتج من السؤال 6
        أن \emph{كل} الجداءات مثنى مثنى تنعدم --- ومنه $E =
        \operatorname{im} p \oplus \operatorname{im} q \oplus
        \operatorname{im} r$، دون أيّ فرض على الجداءات.
  \item من أجل $k$ مسقطًا مع $p_1 + \dots + p_k =
        \mathrm{id}$: بيّن أولًا أنه من أجل أيّ فضاءات جزئية، $\dim
        (F_1 + \dots + F_k) \leq \dim F_1 + \dots + \dim F_k$،
        مع المساواة إذا وفقط إذا كان المجموع مباشرًا؛ ثم
        بيّن $E = \operatorname{im} p_1 + \dots +
        \operatorname{im} p_k$، وبرهن على أنه \emph{إذا} كان
        فوق ذلك $\sum_i \operatorname{rk} p_i \leq n$، فإن \omterm{def:b1:vspaces:sum}{المجموع
        مباشر} و $p_i p_j = 0$ من أجل $i \neq j$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- النوى المكرَّرة: مبرهنة فيتينغ المساعدة.}
ليكن $u \in \mathcal{L}(E)$ مع $\dim E = n$.
\begin{enumerate}[resume]
  \item بيّن السلسلتين الصحيحتين من أجل كل $k \geq 0$:
        \[
        \ker u^k \subseteq \ker u^{k+1},
        \qquad
        \operatorname{im} u^{k+1} \subseteq \operatorname{im}
        u^k .
        \]
  \item بيّن أنه إذا كان $\ker u^{r} = \ker u^{r+1}$ من أجل $r$ ما،
        فإن $\ker u^{k} = \ker u^{r}$ من أجل كل $k \geq r$؛ واذكر
        الاستقرار النظير من أجل الصور وبرهن عليه.
  \item استنتج أنه يوجد أصغر عدد صحيح $r$ يحقق $\ker
        u^{r} = \ker u^{r+1}$، وأن $r \leq n$، وأن الصور
        تستقر عند $r$ نفسه.
  \item (مبرهنة فيتينغ المساعدة) برهن على أن
        \[
        E \;=\; \ker u^{r} \,\oplus\, \operatorname{im} u^{r} .
        \]
  \item بيّن أن الفضاءين الجزئيين مستقران تحت $u$، وأن
        قصر $u$ على $\ker u^{r}$ \omterm{pb:b1:linmaps:1}{معدوم القوى}، وأن
        قصر $u$ على $\operatorname{im} u^{r}$ تشاكل
        تقابلي للمقدار $\operatorname{im} u^{r}$: أي إن كل
        تشاكل ذاتي هو، على \omterm{def:b1:vspaces:sum}{مجموع مباشر} قانوني، «\omterm{pb:b1:linmaps:1}{معدوم القوى}
        زائد قابل للقلب».
  \item ليكن $\pi$ المسقط على $\ker u^{r}$ على امتداد
        $\operatorname{im} u^{r}$. بيّن أن $\pi \circ u = u
        \circ \pi$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- حالة معالَجة، وتوليفة.}
\begin{enumerate}[resume]
  \item لتكن $u(x, y, z) = (y, 0, z)$ على $\R^3$. احسب $u^2$ و
        $u^3$، وحدّد دليل الاستقرار $r$، والفضاءين
        الجزئيين $\ker u^{r}$ و $\operatorname{im} u^{r}$، ومسقط
        فيتينغ $\pi$، وتحقق على الصيغ من أن
        $\pi u = u\pi$ ومن أن $u$ \omterm{pb:b1:linmaps:1}{معدوم القوى} على عامل،
        وتقابلي على الآخر.
  \item بيّن التكافؤات: يكون $u$ \omterm{pb:b1:linmaps:1}{معدوم القوى} $\iff$ $\ker u^{r} =
        E$ $\iff$ $\pi = \mathrm{id}$؛ واستنتج أن
        التشاكل الذاتي \omterm{pb:b1:linmaps:1}{المعدوم القوى} لفضاء ذي بُعد $n$
        يحقق دائمًا $u^{n} = 0$ (فدليل انعدام القوى لا
        يتجاوز البُعد أبدًا).
  \item (الوحدانية) افترض $E = A \oplus B$ مع $A, B$ مستقرين
        تحت $u$، وقصر $u|_A$ \omterm{pb:b1:linmaps:1}{معدوم القوى} و $u|_B$
        تقابلي. برهن على $A = \ker u^{r}$ و $B =
        \operatorname{im} u^{r}$: أي إن تفكيك فيتينغ
        وحيد.
  \item توليفة، في أربع جمل: أيّ قاموس يؤسّسه الجزء 3
        بين المجاميع المباشرة وعائلات
        المساقط؛ ولماذا لم يحتج السؤال 14 إلى أيّ فروض على
        الجداءات بينما احتاج السؤال 15 إلى فرض على \omterm{def:b1:linmaps:rank}{الرتبة}
        (وأيّ أداة من السنة~الثانية، وهي الأثر، تزيله)؛ وبأيّ
        معنى تكون مبرهنة فيتينغ المساعدة الصيغةَ المستقرة للمقدار
        \cref{exo:b1:linmaps:7}؛ وماذا يصير عاملا
        فيتينغ في نظرية القيم الذاتية في مجلّد السنة~الثانية.
        وسمِّ المبرهنة المبرهن عليها في الجزء 4.
\end{enumerate}
\end{problem}
