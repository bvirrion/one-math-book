\chapter{الاشتقاق}\label{ch:b1:derivative}

حُسبت المشتقات في مجلّد الثانوي كله؛ وما كان
ينقص هو سلسلة المبرهنات التي تحوّل الحساب إلى
معلومات عن الدوال: مبرهنة رول، ومبرهنة التزايدات المنتهية،
ونتائجهما --- محكات الرتابة، وحواصر ليبشيتز،
والتحدّب. وكل ما في هذا الفصل يخصّ دوالّ معرَّفة على
\omterm{prop:b1:reals:intervals}{فترة} $I$.

\section{المشتقة}

\begin{definition}\label{def:b1:derivative:def}
تكون $f \colon I \to \R$ \emph{قابلة للاشتقاق عند $x_0 \in
I$}\index{مشتقة} إذا كان لنسبة الفرق
$\frac{f(x) - f(x_0)}{x - x_0}$ نهايةٌ (منتهية) عندما $x \to x_0$؛
وتُكتب النهاية $f'(x_0)$. وبكيفية مكافئة:
\[
f(x_0 + h) = f(x_0) + f'(x_0)\,h + h\,\varepsilon(h),
\qquad \varepsilon(h) \xrightarrow[h \to 0]{} 0 ,
\]
ويقبل المنحنى عندئذ المماس $y = f(x_0) + f'(x_0)(x -
x_0)$. والقابلية للاشتقاق عند $x_0$ تستلزم \omterm{def:b1:continuity:continuous}{الاتصال} عند $x_0$ (اقرأ
المعروضة). وتكون $f$ قابلة للاشتقاق على $I$ إذا كانت كذلك عند كل نقطة؛
وتكون $f$ من الصنف $C^1$ إذا كانت $f'$ \omterm{def:b1:continuity:continuous}{متصلة} فوق ذلك، ومن
الصنف $C^k$\index{دالة Ck@دالة من الصنف $C^k$} إذا أمكن اشتقاق $f$
عدد $k$ مرة مع $f^{(k)}$ \omterm{def:b1:continuity:continuous}{متصلة}.
\end{definition}

\begin{example}\label{ex:b1:derivative:notc1}
عكس «\omterm{def:b1:derivative:def}{قابلة للاشتقاق} $\Rightarrow$ \omterm{def:b1:continuity:continuous}{متصلة}» يفشل:
فالمقدار $\abs{\,\cdot\,}$ عند $0$. والأدهش أن القابلية للاشتقاق لا
تستلزم الصنف $C^1$: فالدالة $f(x) = x^2 \sin\frac 1x$ ($f(0) = 0$)
\omterm{def:b1:derivative:def}{قابلة للاشتقاق} في كل مكان، مع $f'(0) = 0$، لكن $f'(x) = 2x
\sin\frac1x - \cos\frac 1x$ لا نهاية له عند $0$
(\cref{exo:b1:derivative:2}).
\end{example}

\begin{example}[قابلة للاشتقاق عند نقطة واحدة بالضبط]\label{ex:b1:derivative:onepoint}
لتكن $f(x) = x^2$ من أجل $x \in \Q$ و $f(x) = 0$ من أجل $x \notin
\Q$. عند $0$: $\bigl|\frac{f(h) - 0}{h}\bigr| \leq \abs h \to 0$،
ومنه فالدالة $f$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} عند $0$ مع $f'(0) = 0$. وعند أيّ $x_0
\neq 0$، لا تكون $f$ \omterm{def:b1:continuity:continuous}{متصلة} أصلًا: إذ ترسل المتتاليات الناطقة
والصمّاء المتقاربة إلى $x_0$ الدالةَ $f$ إلى $x_0^2 \neq 0$ وإلى
$0$ على الترتيب (\omterm{def:b1:topology:dense}{بالكثافة}، \cref{thm:b1:reals:density}). ومنه فالقابلية
للاشتقاق مفهوم \emph{نقطي} حقًا: إذ يمكن أن تصحّ
عند نقطة واحدة من $\R$ ولا تصحّ في أيّ موضع آخر. والعبرة
العملية: تقتضي عبارات مثل محك الرتابة أو مبرهنة رول
المشتقةَ \emph{على \omterm{prop:b1:reals:intervals}{فترة}} --- فامتلاك
$f'(x_0)$ عند نقاط معزولة، مهما كثرت، لا يدعم أيّ
نتيجة شاملة أصلًا.
\end{example}

\begin{theorem}[العمليات]\label{thm:b1:derivative:operations}
إذا كانت $f, g$ قابلتين للاشتقاق عند $x_0$ (وحيث تكون الصيغ ذات
معنى):
\[
(f + g)' = f' + g', \qquad
(fg)' = f'g + fg', \qquad
\Bigl(\frac fg\Bigr)' = \frac{f'g - fg'}{g^2},
\]
وإذا كانت $g$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} عند $f(x_0)$:
$\;(g \circ f)'(x_0) = g'\bigl(f(x_0)\bigr)\, f'(x_0)$ (قاعدة
السلسلة\index{قاعدة السلسلة}).
\end{theorem}

\begin{proof}
المجموع: مباشر. والجداء: اكتب
\[
f(x)g(x) - f(x_0)g(x_0)
= \bigl(f(x) - f(x_0)\bigr) g(x) + f(x_0)\bigl(g(x) - g(x_0)\bigr),
\]
واقسم على $x - x_0$ ودع $x \to x_0$ (فالدالة $g$ \omterm{def:b1:continuity:continuous}{متصلة} عند $x_0$).
والقسمة: عالج $\frac 1g$ عبر $\frac{1/g(x) - 1/g(x_0)}{x - x_0} =
\frac{-1}{g(x)g(x_0)}\cdot\frac{g(x) - g(x_0)}{x - x_0}$، ثم طبّق
قاعدة الجداء. وقاعدة السلسلة: مع $y_0 = f(x_0)$، عرّف
$\theta(y) = \frac{g(y) - g(y_0)}{y - y_0}$ من أجل $y \neq y_0$ و
$\theta(y_0) = g'(y_0)$: فتكون $\theta$ \omterm{def:b1:continuity:continuous}{متصلة} عند $y_0$، ومن أجل
$x \neq x_0$،
\[
\frac{g(f(x)) - g(f(x_0))}{x - x_0}
= \theta\bigl(f(x)\bigr)\cdot \frac{f(x) - f(x_0)}{x - x_0}
\longrightarrow g'(y_0)\, f'(x_0),
\]
والعامل الأول بتركيب النهايات (وهذه الحيلة تعالج
حالة $f(x) = f(x_0)$ معالجة نظيفة، حيث تنكسر الطريقة الساذجة
«اضرب واقسم على $f(x) - f(x_0)$»).
\end{proof}

\begin{theorem}[مشتقة دالة معكوسة]\label{thm:b1:derivative:inverse}
لتكن $f$ \omterm{def:b1:continuity:continuous}{متصلة} ورتيبة قطعًا على $I$، \omterm{def:b1:derivative:def}{وقابلة للاشتقاق}
عند $x_0$ مع $f'(x_0) \neq 0$. عندئذ تكون $f^{-1}$
(\cref{thm:b1:continuity:bijection}) \omterm{def:b1:derivative:def}{قابلة للاشتقاق} عند $y_0 =
f(x_0)$، مع
\[
(f^{-1})'(y_0) = \frac{1}{f'(x_0)} = \frac{1}{f'\bigl(f^{-1}(y_0)\bigr)} .
\]
وإذا كان $f'(x_0) = 0$، فللمعكوس مماس عمودي عند $y_0$.
\end{theorem}

\begin{proof}
من أجل $y \to y_0$، ضع $x = f^{-1}(y)$: فيعطي \omterm{def:b1:continuity:continuous}{اتصال} $f^{-1}$ أن $x
\to x_0$، و
\[
\frac{f^{-1}(y) - f^{-1}(y_0)}{y - y_0}
= \frac{x - x_0}{f(x) - f(x_0)}
= \frac{1}{\dfrac{f(x) - f(x_0)}{x - x_0}}
\longrightarrow \frac{1}{f'(x_0)} .
\]
وأمّا ادعاء المماس العمودي: فإذا كان $f'(x_0) = 0$، تكون القسمة المعروضة
مقلوبَ مقدار يؤول إلى $0$ محافظًا على
إشارة ثابتة واحدة (فمن أجل $f$ متزايدة قطعًا، $\frac{f(x) -
f(x_0)}{x - x_0} > 0$ من أجل كل $x \neq x_0$): ومنه تؤول نسبة
الفرق للمقدار $f^{-1}$ إلى $+\infty$ (وإلى $-\infty$
من أجل $f$ متناقصة). ويبقى المعكوس متصلًا لكنه ليس
قابلًا للاشتقاق عند $y_0$ --- فمنحناه، وهو انعكاس منحنى $f$
عبر القطر، ينتصب عموديًا بالضبط حيث كان منحنى $f$
أفقيًا، كما يوضّح $x^{1/3}$ عند $0$ إزاء $x^3$.
\end{proof}

\begin{example}[مشتقات المعكوسات، مرتين]\label{ex:b1:derivative:inverseaction}
تعيد المبرهنة حساب المشتقات الكلاسيكية دون أيّ عمل على
النهايات. فمن أجل $\ln = \exp^{-1}$: عند $y = \eu^x$،
\[
(\ln)'(y) = \frac{1}{\exp'(x)} = \frac{1}{\eu^{x}} = \frac1y ,
\]
وهذا صحيح من أجل كل $y > 0$ لأن $\exp' = \exp$ لا ينعدم أبدًا. ومن أجل
$\arctan = \tan^{-1}$: عند $y = \tan x$،
\[
(\arctan)'(y) = \frac{1}{1 + \tan^2 x} = \frac{1}{1 + y^2} ,
\]
باستعمال $\tan' = 1 + \tan^2 > 0$. والفكرة النافذة: تحوّل الصيغة
معرفةً عن دالة إلى معرفة عن
معكوسها بثمن تعويض واحد --- والتعويض
($x = \ln y$ و $x = \arctan y$) هو بالضبط
\omterm{def:b1:logic:statement}{عبارة} أن المتغيرين يعيشان على جانبَي
التقابل.
\end{example}

\section{رول ومبرهنة التزايدات المنتهية}

\begin{proposition}[القيمة الحدّية الداخلية]\label{prop:b1:derivative:fermat}
إذا كانت $f$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} عند نقطة \emph{داخلية} $x_0$ من $I$ و
كانت لها قيمة حدّية موضعية هناك، فإن $f'(x_0) = 0$.
\end{proposition}

\begin{proof}
لنقل إنها قيمة عظمى موضعية: فيوجد $r > 0$ يحقق $f(x) \leq f(x_0)$
من أجل $\abs{x - x_0} \leq r$، وتضمن الداخلية توفّر
جانبَي $x_0$ داخل $I$. فمن أجل $0 < h \leq r$ تكون
القسمة $\frac{f(x_0 + h) - f(x_0)}{h}$ بسطها $\leq 0$
ومقامها $> 0$: ومنه فهي $\leq 0$، وترث نهايتها $f'(x_0)$
الشرطَ $\leq 0$ (لأن المتراجحات الواسعة تمرّ إلى النهايات،
\cref{thm:b1:seq:order})؛ ومن أجل $-r \leq h < 0$ تكون القسمة
$\geq 0$، فينتج $f'(x_0) \geq 0$. ومنه $f'(x_0) = 0$. (وعند
طرف، لا تتوفر إلا إشارة واحدة: فتفشل النتيجة هناك ---
فكّر في $x$ على $\intcc{0}{1}$، وهي عظمى عند $1$ ومشتقتها
$1$.)
\end{proof}

\begin{theorem}[رول]\label{thm:b1:derivative:rolle}
لتكن $f$ \omterm{def:b1:continuity:continuous}{متصلة} على $\intcc{a}{b}$ \omterm{def:b1:derivative:def}{وقابلة للاشتقاق} على
$\intoo{a}{b}$، مع $f(a) = f(b)$. عندئذ $f'(c) = 0$ من أجل $c \in
\intoo{a}{b}$ ما.\index{مبرهنة رول}
\end{theorem}

\begin{proof}
حسب مبرهنة القيم الحدّية (\cref{thm:b1:continuity:evt})، تبلغ $f$
قيمتيها العظمى والصغرى على $\intcc{a}{b}$. فإذا بُلغت كلتاهما
عند الطرفين، فإن (لأن $f(a) = f(b)$) العظمى $=$ الصغرى وتكون $f$
ثابتة: فيفي أيّ $c$ داخلي بالغرض. وإلا فتُبلغ قيمة حدّية
عند نقطة داخلية $c$، ويعطي
\cref{prop:b1:derivative:fermat} أن $f'(c) = 0$.
\end{proof}

\begin{theorem}[مبرهنة التزايدات المنتهية]\label{thm:b1:derivative:mvt}
لتكن $f$ \omterm{def:b1:continuity:continuous}{متصلة} على $\intcc{a}{b}$ \omterm{def:b1:derivative:def}{وقابلة للاشتقاق} على
$\intoo{a}{b}$. يوجد $c \in \intoo{a}{b}$ يحقق
\[
f(b) - f(a) = f'(c)\,(b - a) .
\]
\emph{متراجحة التزايدات المنتهية:} إذا كان فوق ذلك $m \leq f' \leq M$ على
$\intoo{a}{b}$، فإن $m(b-a) \leq f(b) - f(a) \leq M(b-a)$؛ وعلى
الخصوص يستلزم $\abs{f'} \leq K$ أن تكون $f$ ليبشيتزية بالثابت $K$.
\index{مبرهنة التزايدات المنتهية}
\end{theorem}

\begin{proof}
طبّق رول على $g(x) = f(x) - \frac{f(b) - f(a)}{b - a}(x - a)$: فالدالة $g$
\omterm{def:b1:continuity:continuous}{متصلة} على $\intcc{a}{b}$ \omterm{def:b1:derivative:def}{وقابلة للاشتقاق} بالداخل، و $g(a) =
f(a) = g(b)$. وعند النقطة $c$ حيث $g'(c) = 0$: $f'(c) =
\frac{f(b)-f(a)}{b-a}$. وتتبع المتراجحة بحصر $f'(c)$؛
وتطبّقها \omterm{def:b1:logic:statement}{عبارة} ليبشيتز على كل زوج من النقاط.
\end{proof}

\begin{example}[طريقة نيوتن هي طريقة هيرون]\label{ex:b1:derivative:newton}
تستبدل طريقة نيوتن لحلّ $f(x) = 0$ بالمنحنى
مماسَه عند التخمين الحالي $x_n$ وتأخذ جذر المماس
تخمينًا تاليًا:
\[
0 = f(x_n) + f'(x_n)(x_{n+1} - x_n)
\quad\Longrightarrow\quad
x_{n+1} = x_n - \frac{f(x_n)}{f'(x_n)} .
\]
نفّذها على $f(x) = x^2 - 2$:
\[
x_{n+1} = x_n - \frac{x_n^2 - 2}{2x_n}
= \frac{x_n}{2} + \frac{1}{x_n}
= \frac12\Bigl(x_n + \frac{2}{x_n}\Bigr) :
\]
وهو تكرار هيرون بالضبط (\cref{ex:b1:seq:heron})، سابقًا بألفَي
سنة. والسرعة التربيعية المرصودة هناك مفسَّرة الآن
بصورة المماس: فبجوار جذر بسيط، يختلف المنحنى عن
المماس بخطأ من الرتبة الثانية، ومنه فكل خطوة تربّع الخطأ
تقريبًا --- \omterm{def:b1:logic:statement}{والعبارة} العامة تتبع من حواصر تايلور في
\cref{ch:b1:taylor}. والفكرة النافذة:
حيث تستعمل الثنائية (\cref{ex:b1:continuity:dichotomy}) الاتصالَ
وحده وتربح بتّة واحدة لكل خطوة، تنفق نيوتن
مشتقةً \omterm{def:b1:poly:derivative}{لتضاعف} عدد الأرقام الصحيحة في كل خطوة.
\end{example}

\begin{example}[مبرهنة التزايدات المنتهية مقدِّرًا]\label{ex:b1:derivative:mvtestimate}
كم يبلغ $\sqrt{101}$؟ طبّق المبرهنة على $f(t) = \sqrt t$
على $\intcc{100}{101}$: فمن أجل $c \in \intoo{100}{101}$ ما،
\[
\sqrt{101} - 10 = \frac{1}{2\sqrt c},
\qquad\text{ومنه}\qquad
\frac{1}{2\sqrt{101}} < \sqrt{101} - 10 < \frac{1}{20} = 0.05 ,
\]
ولأن $\sqrt{101} < 10.05$، يتجاوز الحاصر الأيسر
$\frac{1}{20.1} > 0.0497$: ومنه $10.0497 < \sqrt{101} < 10.05$
(والقيمة الحقيقية $10.049875\dots$) --- أي ثلاثة أرقام عشرية صحيحة من تقويم
\omterm{def:b1:derivative:def}{مشتقة} واحد. وكذلك $\abs{\sin a - \sin b} \leq
\abs{a - b}$ (بالحاصر $\abs{\cos}\leq 1$): فتقديرات ليبشيتز
المستعملة منذ \cref{ch:b1:seq} كلها هذه المبرهنة. والفكرة
النافذة: مبرهنة التزايدات المنتهية صيغةُ تايلور من الرتبة صفر
--- فهي تقايض نقطةً مجهولة $c$ بمتراجحة صلبة، و
سوف يكرّر \cref{ch:b1:taylor} هذه المقايضة بالضبط.
\end{example}

\begin{corollary}[محك الرتابة]\label{cor:b1:derivative:monotone}
لتكن $f$ \omterm{def:b1:continuity:continuous}{متصلة} على $I$ \omterm{def:b1:derivative:def}{وقابلة للاشتقاق} على الداخل.
\begin{enumerate}
  \item $f' \geq 0$ على الداخل $\iff$ $f$ متزايدة؛ و $f' = 0$
        $\iff$ $f$ ثابتة.
  \item إذا كان $f' > 0$ إلا عند عدد منته من النقاط ينعدم عندها،
        فإن $f$ متزايدة \emph{قطعًا}.
\end{enumerate}
\end{corollary}

\begin{proof}
إذا كان $f' \geq 0$: فمن أجل $x < y$ في $I$، تعطي مبرهنة التزايدات المنتهية على
$\intcc{x}{y}$ أن $f(y) - f(x) = f'(c)(y - x) \geq 0$. وبالعكس،
تكون نسب الفروق لدالة متزايدة $\geq 0$، ومنه تكون
نهاياتها كذلك. وأمّا حالة الثابت: فطبّق ما سبق على $f'$ و
$-f' \geq 0$. والصيغة القطعية: $f$ متزايدة؛ ومساواة $f(x) =
f(y)$ من أجل $x < y$ ستجمّد $f$ على $\intcc{x}{y}$، فتفرض $f' =
0$ هناك --- أي عند نقاط لا تُحصى.
\end{proof}

\begin{example}[مشتقتان متساويتان ودالتان غير متساويتين]\label{ex:b1:derivative:intervalmatters}
على $\R^* = \intoo{-\infty}{0} \cup \intoo{0}{+\infty}$، تحقق كلٌّ من
$f(x) = \ln\abs x$ و $g(x) = \ln\abs x + \mathbf{1}_{x>0}$
(بإضافة $1$ على نصف المستقيم الأيمن فقط) الشرطَ $f' = g' =
\frac1x$. وهما لا تختلفان بثابت: فالمحك
«$f' = 0 \implies f$ ثابتة» عبارةٌ عن \emph{\omterm{prop:b1:reals:intervals}{فترة}}
--- إذ يجري برهانه مبرهنةَ التزايدات المنتهية بين نقطتين،
وهذا يقتضي وقوع القطعة الواصلة بينهما كلها في
\omterm{def:b1:logic:sets}{مجموعة} التعريف. وعلى كل نصف مستقيم على حدة، تكون الدوال الأصلية
للمقدار $\frac1x$ هي $\ln\abs x + c$ بثابت واحد لكل نصف مستقيم،
أي ثابتين مستقلين إجمالًا. ويرث \Cref{ch:b1:integration}
هذه التفاصيل الدقيقة: فالدالة الأصلية «الوحيدة» لدالة
معرَّفة جيدًا بغضّ النظر عن ثابت \emph{على كل \omterm{prop:b1:reals:intervals}{فترة} من
\omterm{def:b1:logic:sets}{مجموعة} تعريفها}، وتفترض جداول الدوال الأصلية الترابطَ صامتة.
\end{example}

\begin{example}[القطعية بالمجّان]\label{ex:b1:derivative:strictness}
الدالة $x \mapsto x^3$ متزايدة \emph{قطعًا} على $\R$ وإن
انعدمت مشتقتها عند $0$: فبند المحك
«$f' > 0$ إلا عند عدد منته من النقاط» مصمَّم بالضبط
لمثل هذه النقاط المنبسطة. وعلى النقيض، لا يعطي $f' \geq 0$ وحده إلا
التزايد بالمعنى الواسع (فالدالة الثابتة تفي بالغرض)، و
\omterm{def:b1:derivative:def}{المشتقة} المنعدمة على \omterm{prop:b1:reals:intervals}{فترة} جزئية كاملة تجمّد
الدالة هناك. والقاعدة العملية: لادعاء الرتابة القطعية،
اذكر جذور $f'$؛ فعددٌ منته منها (أو، \omterm{def:b1:logic:statement}{بعبارة} أعمّ، لا شيء
منها على أيّ \omterm{prop:b1:reals:intervals}{فترة} جزئية) غير مؤذٍ، وأمّا \omterm{prop:b1:reals:intervals}{فترة}
منها فقاتلة.
\end{example}

\begin{example}[دراسة تغيرات كاملة]\label{ex:b1:derivative:variations}
ادرس $f(x) = x^3 - 3x + 1$ على $\R$. \omterm{def:b1:derivative:def}{المشتقة}: $f'(x) =
3(x^2 - 1)$، وهي موجبة على $\intoo{-\infty}{-1}$، وسالبة على
$\intoo{-1}{1}$، وموجبة على $\intoo{1}{+\infty}$: ومنه، حسب
محك الرتابة، تتزايد $f$ ثم تتناقص ثم
تتزايد، بقيمة عظمى موضعية $f(-1) = 3$ وقيمة صغرى موضعية
$f(1) = -1$. والنهايات: $\mp\infty$ عند $\mp\infty$. والنتائج،
مقروءةً من جدول التغيرات مع مبرهنة القيم
الوسطى على كل فرع رتيب: تنعدم $f$ مرة واحدة بالضبط في
كل من
\[
\intoo{-\infty}{-1}, \qquad \intoo{-1}{1}, \qquad
\intoo{1}{+\infty}
\]
(فالقيم عند الوصلات ذات إشارات متعاكسة: $3 > 0 > -1$)،
ومنه فللمعادلة $x^3 - 3x + 1 = 0$ ثلاثة جذور حقيقية
بالضبط؛ وعدديًا تقع بجوار $-1.88$ و $0.35$ و $1.53$. والفكرة
النافذة: جدول التغيرات \emph{أداة برهان}
لا رسمٌ تخطيطي --- ففرعٌ رتيب مع تغيّر إشارة يساوي
جذرًا واحدًا بالضبط، ويعدّد الجدول الفروعَ
تعدادًا حاصرًا.
\end{example}

\begin{theorem}[صيغة لايبنتز]\label{thm:b1:derivative:leibniz}
إذا كانت $f, g$ قابلتين للاشتقاق $n$ مرة، فكذلك $fg$،
و\index{صيغة لايبنتز}
\[
(fg)^{(n)} = \sum_{k=0}^{n} \binom nk f^{(k)}\, g^{(n-k)} .
\]
\end{theorem}

\begin{proof}
بالاستقراء على $n$، على نحو مواز تمامًا لمبرهنة ثنائي الحدّ. و
حالة $n = 1$ هي قاعدة الجداء. وبفرض الصيغة عند الرتبة
$n$، اشتق مرة أخرى:
\[
(fg)^{(n+1)} = \sum_{k=0}^{n} \binom nk
\Bigl( f^{(k+1)} g^{(n-k)} + f^{(k)} g^{(n-k+1)} \Bigr),
\]
ثم أعد تدليل المجموع الأول بالمقدار $j = k + 1$ واجمع
معامل $f^{(j)} g^{(n+1-j)}$: فهو $\binom{n}{j-1} +
\binom nj = \binom{n+1}{j}$ حسب قاعدة باسكال
(\cref{prop:b1:counting:identities})، ويحمل الحدّان الحدّيان $j = 0$
و $j = n + 1$ المقدارَ $\binom{n+1}{0} = \binom{n+1}{n+1} =
1$ كما ينبغي.
\end{proof}

\begin{example}[لايبنتز في العمل]\label{ex:b1:derivative:leibnizaction}
احسب $\bigl(x^2 \eu^x\bigr)^{(n)}$ من أجل $n \geq 2$. خذ $f =
x^2$، ومشتقاته تموت سريعًا ($f' = 2x$ و $f'' = 2$ و
$f^{(k)} = 0$ من أجل $k \geq 3$)، و $g = \eu^x$: فلا ينجو إلا ثلاثة حدود
من مجموع لايبنتز،
\[
\bigl(x^2\eu^x\bigr)^{(n)}
= \binom n0 x^2 \eu^x + \binom n1 (2x)\,\eu^x + \binom n2\,
2\,\eu^x
= \eu^x\bigl(x^2 + 2nx + n(n-1)\bigr).
\]
وللتحقق عند $n = 1$: $\eu^x(x^2 + 2x)$، وهو فعلًا
$(x^2\eu^x)'$. والفكرة النافذة: استعمل لايبنتز حين يكون أحد العاملين
\omterm{def:b1:poly:def}{كثير حدود} --- إذ لا يكون في المجموع عندئذ إلا $\deg + 1$ حدًا، و
تكون الصيغة صورةً مغلقة لا متطابقة مجردة. (وأمّا من أجل
عاملين لا نهائيَي الحيوية مثل $\eu^x\sin x$، فالأسّيات
العقدية من \cref{ch:b1:complex} أداة أفضل.)
\end{example}

\section{التحدّب}

\begin{definition}\label{def:b1:derivative:convex}
تكون $f \colon I \to \R$ \emph{محدَّبة}\index{دالة محدَّبة} إذا وقع
كل وتر فوق المنحنى:
\[
\forall x, y \in I,\ \forall t \in \intcc{0}{1}, \quad
f\bigl(tx + (1-t)y\bigr) \leq t f(x) + (1-t) f(y).
\]
(وتكون $f$ \emph{مقعّرة} إذا كانت $-f$ محدَّبة.)
\end{definition}

\begin{theorem}[التمييزات التفاضلية]\label{thm:b1:derivative:convexchar}
لتكن $f$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} على $I$. تتكافأ العبارات الآتية:
\begin{enumerate}
  \item $f$ \omterm{def:b1:derivative:convex}{محدَّبة}؛
  \item $f'$ متزايدة على $I$؛
  \item يقع المنحنى فوق كل مماس: $f(y) \geq f(x) +
        f'(x)(y - x)$ من أجل كل $x, y \in I$.
\end{enumerate}
وإذا كانت $f$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} مرتين: تكون $f$ \omterm{def:b1:derivative:convex}{محدَّبة} $\iff f'' \geq 0$.
\end{theorem}

\begin{proof}
(1 $\Rightarrow$ 3) يُكتب التحدّب $\frac{f(x + t(y-x)) -
f(x)}{t} \leq f(y) - f(x)$ من أجل $t \in \intoc{0}{1}$؛ ودع $t \to 0^+$:
$f'(x)(y - x) \leq f(y) - f(x)$.

(3 $\Rightarrow$ 2) من أجل $x < y$، تعطي متراجحتا المماس عند $x$
وعند $y$ أن $f'(x)(y-x) \leq f(y) - f(x) \leq f'(y)(y - x)$،
ومنه $f'(x) \leq f'(y)$.

(2 $\Rightarrow$ 1) ثبّت $x < y$ و $t \in \intoo{0}{1}$، وليكن $z
= tx + (1-t)y \in \intoo{x}{y}$. وحسب مبرهنة التزايدات المنتهية على
$\intcc{x}{z}$ وعلى $\intcc{z}{y}$: يوجد $c_1 < z < c_2$ يحققان
\[
\frac{f(z) - f(x)}{z - x} = f'(c_1) \leq f'(c_2)
= \frac{f(y) - f(z)}{y - z} ,
\]
وبإزالة المقامات ($z - x = (1-t)(y-x)$ و $y - z = t(y-x)$)
يُعاد الترتيب فيعطي متراجحة التحدّب بالضبط.

وحالة القابلية للاشتقاق مرتين: $f'' \geq 0 \iff f'$ متزايدة
(\cref{cor:b1:derivative:monotone}).
\end{proof}

\begin{example}[متراجحات التحدّب الكلاسيكية]\label{ex:b1:derivative:convexineq}
الدالة $\exp$ \omterm{def:b1:derivative:convex}{محدَّبة} ($\exp'' = \exp > 0$): ويعطي مماسها عند $0$ أن
$\eu^x \geq 1 + x$ من أجل كل $x$. والدالة $\ln$ مقعّرة: ويعطي مماسها عند $1$
أن $\ln x \leq x - 1$؛ وتعطي أوتارها، من أجل $0 < a \leq b$،
المتراجحةَ بين المتوسطين الهندسي والحسابي: فبأخذ $t =
\frac12$ في التقعّر،
\[
\ln\frac{a + b}{2} \geq \frac{\ln a + \ln b}{2} = \ln\sqrt{ab},
\qquad\text{ومنه}\qquad
\sqrt{ab} \leq \frac{a+b}{2} .
\]
والمتراجحة العامة بين المتوسطين الحسابي والهندسي هي
\cref{exo:b1:derivative:9}.
\end{example}

\begin{example}[متراجحة تحدّب من الصفر]\label{ex:b1:derivative:xlnx}
الدالة $f(t) = t\ln t$ \omterm{def:b1:derivative:convex}{محدَّبة} على $\intoo{0}{+\infty}$:
إذ $f''(t) = \frac1t > 0$. ومتراجحتها عند المنتصف، مضروبةً في
$2$، تُقرأ: من أجل كل $a, b > 0$،
\[
a\ln a + b\ln b \;\geq\; (a + b)\,\ln\frac{a + b}{2} ,
\]
مع المساواة إذا وفقط إذا كان $a = b$ (بالتحدّب القطعي). ولتجربتها: $a =
1$ و $b = 3$ يعطي $3\ln 3 = 3.296$ إزاء $4\ln 2 = 2.773$.
وهذه المتراجحة البريئة هي حالة النقطتين من مقارنة
\emph{الإنتروبيا} التي تعود مع متراجحة ينسن
(\cref{exo:b1:derivative:9}) وفي المقاربات
المعلوماتية في مجلّد السنة الثالثة. والفكرة
النافذة: لتصنيع متراجحة، جد دالةً
لمشتقتها الثانية إشارة واكتب ما يقوله التحدّب
--- فالتمييز التفاضلي يحوّل تحقّق إشارة واحدًا
إلى متراجحات لا تُحصى.
\end{example}

\begin{remark}[مزالق شائعة مع المشتقات]
(أ) \emph{\omterm{def:b1:derivative:def}{المشتقة} الموجبة عند نقطة واحدة لا تعطي
الرتابة بجوارها}: فالمقدار $f(x) = \frac x2 + x^2\sin\frac1x$ (مع
$f(0) = 0$) له $f'(0) = \frac12 > 0$، ومع ذلك
\[
f'(x) = \frac12 + 2x\sin\frac1x - \cos\frac1x
\]
يساوي $-\frac12$ عند كل $x_n = \frac{1}{2\pi n}$: فكل
\omterm{def:b1:topology:open}{جوار} للنقطة $0$ يحتوي هبوطات. وتحتاج الرتابة إلى $f'
\geq 0$ \emph{على \omterm{prop:b1:reals:intervals}{فترة}}
(\cref{cor:b1:derivative:monotone})؛ وأمّا الإشارة النقطية فلا
تضبط إلا عبور مستقيم المماس. (ب) \emph{فرضيات رول
الثلاث كلها فاعلة}: فالمقدار $\abs x$ على $\intcc{-1}{1}$
(لا قابلية اشتقاق بالداخل)، و $x$ على $\intcc{0}{1}$ (الطرفان غير
متساويين)، و $x - \lfloor x\rfloor$ على $\intcc{0}{1}$ (يفشل \omterm{def:b1:continuity:continuous}{الاتصال}
عند $1$) يكسر كلٌّ منها فرضًا واحدًا بالضبط ويكسر
النتيجة. (ج) \emph{قد تكون المشتقات منقطعة، لكن
لا كيفما اتفق}: فقد تتذبذب $f'$
(\cref{ex:b1:derivative:notc1}) ومع ذلك تحقق دائمًا
خاصية القيم الوسطى (دربو،
\cref{exo:b1:derivative:10}): \omterm{def:b1:derivative:def}{فالمشتقة} لا تقفز أبدًا --- وإذا
حسبتَ «نهاية \omterm{def:b1:derivative:def}{مشتقة}» من جهة واحدة فيها قفزة، فقد
اشتققتَ دالة غير \omterm{def:b1:derivative:def}{قابلة للاشتقاق}. (د) \emph{صيغة
المعكوس تحتاج إلى $f' \neq 0$}: \omterm{def:b1:logic:map}{فالتطبيق} $x \mapsto x^3$ تقابل ملس
متزايد قطعًا معكوسه $x^{1/3}$ له مماس
عمودي عند $0$ --- فقابلية اشتقاق المعكوس
تُفقد بالضبط حيث تنعدم $f'$
(\cref{thm:b1:derivative:inverse}).
\end{remark}

\begin{remark}[أين تعمل مبرهنة التزايدات المنتهية تاليًا]
تكاد كل \omterm{def:b1:logic:statement}{عبارة} كمّية في الفصول الآتية أن تكون مبرهنةَ
التزايدات المنتهية في هذا الفصل متنكّرة: فالمبرهنة الأساسية
في التكامل (\cref{ch:b1:integration}) تشتق دالة
المساحة وتختم بمحك الرتابة؛ وصيغة
تايلور--لاغرانج (\cref{ch:b1:taylor}) هي مبرهنة التزايدات
المنتهية مكرَّرة $n$ مرة؛ وتحليل خطأ طريقة نيوتن
وتكرارات النقطة الصامدة (\cref{exo:b1:derivative:11}) هو
صيغة ليبشيتز؛ وتحوّل مسألة نهاية الأسبوع في هذا الفصل
(\cref{pb:b1:derivative:1}) حاصرَ ليبشيتز نفسه إلى
نظرية أعداد --- أي متراجحة تنافر بين الأعداد
الجبرية والأعداد الناطقة، تعطي أول عدد متسامٍ
في التاريخ. وفي مجلّد السنة الثانية، تنجو \emph{متراجحة}
التزايدات المنتهية في عدة متغيرات حيث لا تنجو المساواة.
\end{remark}

\begin{example}[متراجحة يونغ من التقعّر]\label{ex:b1:derivative:young}
ليكن $p, q > 1$ مع $\frac1p + \frac1q = 1$. من أجل كل $a, b > 0$:
\[
ab \;\leq\; \frac{a^p}{p} + \frac{b^q}{q} .
\]
والبرهان \omterm{def:b1:logic:map}{بتطبيق} واحد لتقعّر $\ln$ بالوزنين
$\frac1p, \frac1q$ (وهي متراجحة ينسن ذات النقطتين، كما في
\cref{exo:b1:derivative:9}):
\[
\ln\Bigl(\frac{a^p}{p} + \frac{b^q}{q}\Bigr)
\;\geq\; \frac1p \ln(a^p) + \frac1q \ln(b^q)
= \ln a + \ln b = \ln(ab),
\]
ويحوّل تزايد $\ln$ متراجحةَ اللوغاريتمات إلى
الادعاء؛ والمساواة إذا وفقط إذا كان $a^p = b^q$ (بالتقعّر القطعي). وحالة
$p = q = 2$ هي المتراجحة بين المتوسطين الحسابي والهندسي $ab \leq
\frac{a^2 + b^2}{2}$ متنكّرة. والفكرة النافذة: متراجحة
يونغ هي البذرة الجبرية لمتراجحتَي هولدر ومنكوفسكي
في مجلّد السنة الثانية --- أي \omterm{def:b1:logic:statement}{عبارة} تقعّر واحدة
عن $\ln$، محصودةً من أجل المعايير.
\end{example}

\begin{remark}[منظورات داخل هذا المجلّد]
تكتسب \omterm{def:b1:derivative:def}{المشتقة} ثلاث حيوات جديدة قبل نهاية المجلّد.
ففي \cref{ch:b1:taylor} تتكرر: إذ تُضغط $n$ \omterm{def:b1:derivative:def}{مشتقة} عند نقطة
في \omterm{def:b1:poly:def}{كثير حدود} واحد مع خطأ مضبوط، وتصير
مبرهنة التزايدات المنتهية باقيَ لاغرانج. وفي
\cref{ch:b1:curves}، يصير الاشتقاق هندسيًا: فمن أجل
منحن معلَّمي $t \mapsto (x(t), y(t))$، يكون الزوج $(x'(t),
y'(t))$ \emph{متجهة} سرعة، ويصير التماسّ
استقامية، وتصير النقاط الحرجة نقاطَ ارتداد تُصنَّف. وفي
\cref{ch:b1:multivar}، يُجمَّد متغير واحد في كل مرة: فتكرّر
المشتقات الجزئية هذا الفصل مرتين، وينمو مستقيم المماس
إلى مستوٍ مماس. وترث الفصول الثلاثة كلها القواعد نفسها ---
تقريب خطي موضعي مع حدّ خطأ ---
المنطوقة أول مرة هنا.
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:6}
(رول متقطّع) لتكن $f$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} $n$ مرة على $I$ و
تنعدم عند $n + 1$ نقطة متمايزة. برهن على أن $f^{(n)}$ ينعدم
مرة واحدة على الأقل. وتطبيقًا: \omterm{def:b1:poly:def}{كثير الحدود} من درجة $\leq n$ المنعدم عند
$n+1$ نقطة معدوم (مرة أخرى).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:7}
لتكن $f$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} مرتين على $\intcc{a}{b}$ مع $f(a) = f(b)
= 0$ و $f(x_0) > 0$ من أجل $x_0$ داخلية ما. برهن على أن $f''(c) <
0$ من أجل $c \in \intoo{a}{b}$ ما. \emph{(بمبرهنتَي تزايدات منتهية و
مقارنة ميول.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:8}
ادرس الدالة $f(x) = \dfrac{\ln x}{x}$ على
$\intoo{0}{+\infty}$: التغيرات والنهايات والقيمة العظمى. واستنتج أن $a^b
> b^a$ من أجل كل الأعداد الحقيقية $\eu \leq a < b$، واحسم الحالة الخاصة
الشهيرة: أيّهما أكبر، $\eu^\pi$ أم $\pi^\eu$؟ وتحقق إزاء
زوجَي الأعداد الصحيحة الصغيرة $(2,3)$ و $(2,4)$: ولماذا يسلكان
سلوكًا مختلفًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:9}
(المتراجحة بين المتوسطين الحسابي والهندسي) باستعمال تقعّر $\ln$ بأوزان
عامة (متراجحة ينسن من أجل $n$ نقطة، ويُبرهن عليها
بالاستقراء على $n$)، بيّن أنه من أجل الأعداد الحقيقية الموجبة $a_1, \dots, a_n$:
\[
\sqrt[n]{a_1 a_2 \cdots a_n} \leq \frac{a_1 + \dots + a_n}{n},
\]
مع المساواة إذا وفقط إذا تساوت كل $a_i$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:10}
(دربو: المشتقات تأخذ القيم الوسطى) لتكن $f$
\omterm{def:b1:derivative:def}{قابلة للاشتقاق} على $I$ وليكن $a < b$ في $I$ مع $f'(a) < v < f'(b)$.
وبالنظر في $g(x) = f(x) - vx$ وفي النقطة التي تبلغ عندها $g$
قيمتها الصغرى على $\intcc{a}{b}$، برهن على أن $f'(c) = v$ من أجل $c
\in \intoo{a}{b}$ ما --- وإن لم يكن على $f'$ أن تكون \omterm{def:b1:continuity:continuous}{متصلة}
(\cref{exo:b1:derivative:2}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:11}
لتكن $f \colon \R \to \R$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} مع $\abs{f'(x)} \leq k
< 1$ من أجل كل $x$ (أي \emph{تقلّص}). برهن على أن للمقدار $f$
نقطة صامدة واحدة بالضبط $\ell$، وعلى أن كل متتالية $u_{n+1} = f(u_n)$
تتقارب إلى $\ell$ مع $\abs{u_n - \ell} \leq k^n\abs{u_0 -
\ell}$. \emph{(الوجود: طبّق مبرهنة القيم الوسطى على
$f(x) - x$ على قطعة كبيرة، مستعملًا حاصر ليبشيتز؛ أو استعمل
التمام مع محك كوشي.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:12}
(مبرهنة التزايدات المنتهية لكوشي وقاعدة لوبيتال)
\begin{enumerate}
  \item لتكن $f, g$ متصلتين على $\intcc{a}{b}$ و
        قابلتين للاشتقاق على $\intoo{a}{b}$، مع $g'$ لا ينعدم
        هناك أبدًا. برهن على أن $g(b) \neq g(a)$ وعلى أن $c \in
        \intoo{a}{b}$ ما يحقق
        \[
        \frac{f(b) - f(a)}{g(b) - g(a)} = \frac{f'(c)}{g'(c)}
        \]
        \emph{(طبّق رول على $h = f - \lambda g$ من أجل الثابت
        $\lambda$ الصحيح)}.
  \item استنتج قاعدة لوبيتال في الصورة $\frac00$ عند
        نقطة: إذا كان $f(a) = g(a) = 0$ و
        $\frac{f'(x)}{g'(x)} \to \ell$ عندما $x \to a^+$، فإن
        $\frac{f(x)}{g(x)} \to \ell$.
  \item بيّن أن العكس يفشل: فمن أجل $f(x) = x^2\sin\frac1x$
        ($f(0) = 0$) و $g(x) = x$، يكون للقسمة
        $\frac{f}{g}$ نهايةٌ عند $0$ لكن
        $\frac{f'}{g'}$ لا نهاية له.
\end{enumerate}
\end{exercise}

\section{مسألة: متراجحة ليوفيل وأول عدد
متسامٍ}

\begin{problem}[{مسألة نهاية الأسبوع --- الأعداد الجبرية تنافر
الأعداد الناطقة: $\abs{x - p/q} \geq C/q^d$، وتسامي
$\sum 10^{-n!}$}]
\label{pb:b1:derivative:1}
يكون العدد الحقيقي \emph{جبريًا} إذا كان جذرًا \omterm{def:b1:poly:def}{لكثير حدود}
غير معدوم بمعاملات صحيحة، و\emph{متساميًا}
فيما عدا ذلك. وفي سنة 1844 أنتج ليوفيل أول عدد \emph{بُرهن}
على تساميه قط، ومحرّك برهانه هو
مبرهنة التزايدات المنتهية في هذا الفصل: \omterm{pb:b1:logic:1}{فالعدد الجبري} من الدرجة $d$
لا يمكن تقريبه بالأعداد الناطقة أفضل من $C/q^d$ --- ومنه فالعدد
القابل للتقريب \emph{أسرع من كل قوة} لا يمكن أن يكون
جبريًا. وتبني هذه المسألة المتراجحة، وتنشئ
عدد ليوفيل $L = 0.110001000\dots$ (بآحاد عند المواضع
العاملية، عبر آلة الأرقام في \cref{pb:b1:reals:1})،
وتبرهن على تساميه، وتُختم ببرهان كانتور المنافس وبحواصر
فعّالة من أجل $\sqrt2$ و
$2^{1/3}$.\index{عدد متسام}\index{متراجحة
ليوفيل}\index{عدد جبري}

\medskip
\noindent\textbf{الجزء 1 --- إلى أيّ حد يمكن تقريب الأعداد
الناطقة؟}
\begin{enumerate}
  \item بيّن أن العددين الناطقين المتمايزين $\frac ab \neq \frac pq$
        (مكتوبين مع $b, q \geq 1$) يحققان $\bigl|\frac ab -
        \frac pq\bigr| \geq \frac{1}{bq}$. واستنتج: إذا كان $x =
        \frac ab$ و $0 < \bigl|x - \frac pq\bigr| <
        \frac{1}{bq}$، فلا وجود لمثل هذا $\frac pq$ --- أي إن العدد الناطق
        ينافر كل الأعداد الناطقة الأخرى على سلّم $\frac 1q$.
  \item برهن على أنه من أجل \emph{كل} عدد ناطق $\frac pq$ ($q \geq
        1$): $\bigl|\sqrt2 - \frac pq\bigr| \geq
        \frac{1}{4q^2}$ \emph{(فإذا تجاوزت المسافة $1$ فهذا
        جليّ؛ وإلا فحُدّ $\abs{\sqrt2 + p/q} < 4$ و
        استعمل العدد الصحيح غير المعدوم $\abs{p^2 - 2q^2} \geq 1$)}.
  \item وفي الاتجاه الآخر: تحقق من أن $(p, q) \mapsto (p +
        2q, p + q)$ يحفظ $\abs{p^2 - 2q^2} = 1$، وولّد
        انطلاقًا من $(1,1)$ الأزواجَ $(3,2)$ و $(7,5)$ و $(17,12)$ و
        $(41,29)$ و $(99,70)$، وبيّن أن كلًّا منها يحقق
        \[
        \Bigl|\sqrt2 - \frac pq\Bigr| =
        \frac{1}{q^2\,(\sqrt2 + p/q)} < \frac{1}{2q^2} :
        \]
        أي تقريبات لا تُحصى من الرتبة $2$. ومع
        السؤال 2: يكون أُسّ التقريب للمقدار $\sqrt 2$
        هو $2$ \emph{بالضبط}.
  \item (ديريكليه) ليكن $x$ أصمّ و $N \in \N^*$.
        انظر في الأجزاء الكسرية $N + 1$ للمقادير $0, x, 2x,
        \dots, Nx$ في الأدراج $N$ $\intco{\frac kN}{\frac{k +
        1}{N}}$: فحسب مبدأ الأدراج
        (\cref{cor:b1:counting:pigeonhole})، يقع اثنان في درج واحد.
        واستنتج $q \leq N$ و $p$ يحققان $\abs{qx - p} <
        \frac 1N$، ومنه أعدادًا ناطقة لا تُحصى تحقق
        $\bigl|x - \frac pq\bigr| < \frac{1}{q^2}$: أي إن \emph{كل}
        عدد أصمّ قابل للتقريب من الرتبة $2$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- متراجحة ليوفيل.} ليكن $x$
أصمّ وجبريًا.
\begin{enumerate}[resume]
  \item بيّن أنه بين كثيرات الحدود الصحيحة غير المعدومة
        المنعدمة عند $x$ يوجد واحد، وليكن $P$ من الدرجة $d$،
        \emph{بلا جذر ناطق}؛ وتحقق من $d \geq 2$
        \emph{(اقسم على عامل $X - \frac ab$ على $\Q$ و
        أزل المقامات؛ فالدرجة $1$ ستجعل $x$
        ناطقًا)}.
  \item بيّن أنه من أجل كل عدد ناطق $\frac pq$ ($q \geq 1$):
        $\bigl|P\bigl(\frac pq\bigr)\bigr| \geq \frac{1}{q^d}$
        \emph{(لأن $q^d P(p/q)$ عدد صحيح غير معدوم)}.
  \item لتكن $M = \max_{\intcc{x-1}{x+1}} \abs{P'}$
        (\cref{thm:b1:continuity:evt}). وباستعمال مبرهنة التزايدات
        المنتهية بين $x$ و $\frac pq$، برهن على
        \emph{متراجحة ليوفيل}: مع $C = \min\bigl(1,
        \frac 1M\bigr) > 0$،
        \[
        \Bigl| x - \frac pq \Bigr| \geq \frac{C}{q^{\,d}}
        \qquad\text{من أجل كل عدد ناطق } \frac pq,\ q \geq 1 .
        \]
  \item سمِّ $x$ \emph{عدد ليوفيل} إذا كان من أجل كل $n \in
        \N$ يوجد عدد ناطق $\frac pq$ مع $q \geq 2$ و
        $0 < \bigl|x - \frac pq\bigr| < q^{-n}$. برهن على أن عدد
        ليوفيل أصمّ \emph{(السؤال 1: اختر
        $n$ يحقق $2^{\,n-1} > b$)}.
  \item برهن على مبرهنة ليوفيل: \emph{عدد ليوفيل
        متسامٍ} \emph{(اجمع بين السؤالين 7 و 8: فالمتراجحة
        $C < q^{\,d-n}$ تفشل من أجل $n$ كبير)}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- العدد $L$.}
\begin{enumerate}[resume]
  \item ليكن $L$ القيمة (بمعنى
        \cref{pb:b1:reals:1}) لسلسلة الأرقام العشرية ذات
        الرقم $1$ في المواضع $n!$ ($n = 1, 2, 3, \dots$)
        و $0$ فيما عداها، أي $L = \sup_k t_k$ مع $t_k =
        \sum_{n=1}^{k} 10^{-n!}$. اكتب الأرقام $25$
        الأولى للمقدار $L$.
  \item برهن على تأطير الذيل، من أجل كل $k \geq 1$:
        \[
        10^{-(k+1)!} \;\leq\; L - t_k \;\leq\;
        \frac{10}{9}\,10^{-(k+1)!} \;<\; 2\cdot 10^{-(k+1)!}
        \]
        \emph{(حُدّ كل مجموع جزئي بعد $t_k$ بمجموع
        هندسي منته)}.
  \item اكتب $t_k = \frac{p_k}{q_k}$ مع $q_k = 10^{k!}$.
        بيّن $0 < L - \frac{p_k}{q_k} < \frac{2}{q_k^{\,k+1}}$،
        واستنتج أن $L$ عدد ليوفيل بمعنى
        السؤال 8.
  \item استنتج: العدد $L$ متسامٍ --- وهو أول مثال صريح
        في التاريخ (ليوفيل، 1844). وتحقق من صممه
        مباشرةً: فأرقامه ليست دورية في النهاية (لتنامي الفجوات، كما في \cref{pb:b1:reals:1}،
        السؤال 20).
  \item عمّم: استبدل بكل رقم $1$ رقمًا كيفيًا
        غير معدوم $d_n \in \intint{1}{9}$. بيّن أن القيمة
        تبقى عدد ليوفيل، واستنتج --- بحجة القطر
        في \cref{pb:b1:reals:1} (السؤال 22) مطبَّقةً
        على اختيارات الأرقام هذه --- أن الأعداد المتسامية
        من هذا الشكل لا تُحصى عددًا غير قابل للعدّ.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- تراتب رتب
التقريب.} قل إن $x$ \emph{قابل للتقريب من الرتبة $\mu$} إذا كان من أجل
ثابت $c > 0$ ما يحقق عدد لا يُحصى من الأعداد الناطقة
$\bigl|x - \frac pq\bigr| < \frac{c}{q^{\mu}}$.
\begin{enumerate}[resume]
  \item ركّب التراتب من الأجزاء 1--3: الأعداد الناطقة
        قابلة للتقريب من الرتبة $1$ ولا أفضل؛ و $\sqrt 2$ من
        الرتبة $2$ ولا أفضل؛ وكل عدد أصمّ من الرتبة
        $2$ على الأقل؛ \omterm{pb:b1:logic:1}{والعدد الجبري} من الدرجة $d$ من لا رتبة
        بعد $d$؛ وأعداد ليوفيل من كل رتبة. وسوّغ
        كل ادعاء بالاستشهاد بالسؤال المعني.
  \item بيّن أن $L + r$ عدد ليوفيل من أجل كل
        عدد ناطق $r = \frac ab$ \emph{(انسحب
        بالمقرّبات: فالمقامات الجديدة هي $b\,q_k$)}.
        واستنتج أن أعداد ليوفيل --- ومنه المتسامية ---
        \omterm{def:b1:topology:dense}{كثيفة} في $\R$.
  \item (كانتور، 1874) برهن على أن \omterm{def:b1:logic:sets}{مجموعة} الأعداد الجبرية
        قابلة للعدّ: فعدد كثيرات الحدود الصحيحة
        التي درجتها مضافًا إليها مجموع $\abs{\text{المعاملات}}$
        محدود بالمقدار $h$ منتهٍ، ولكلٍّ منها $\deg$ جذرًا على الأكثر؛ و
        الاتحاد القابل للعدّ لمجموعات منتهية قابل للعدّ. ولأن لا
        متتالية تستنفد $\R$ (\cref{pb:b1:reals:1}، السؤال
        22)، توجد أعداد متسامية --- بل تكوّن
        \omterm{def:b1:logic:sets}{مجموعة} غير قابلة للعدّ. وقارن البرهانين: فماذا يعطي
        برهان ليوفيل ولا يستطيع برهان كانتور إعطاءه؟
  \item برهن مباشرةً من السؤال 2 على أن $\sqrt 2$
        \emph{ليس} عدد ليوفيل \emph{(فمن أجل $n \geq 3$،
        تحدّ المتراجحة $q^{-n} > \frac{1}{4q^2}$ المقدارَ $q$؛
        ولا يبقى عندئذ إلا عدد منته من الأعداد الناطقة المرشّحة، وكلها
        على مسافة موجبة من $\sqrt2$)}. وعمّم:
        لا \omterm{pb:b1:logic:1}{عدد جبري} هو عدد ليوفيل.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- ثوابت فعّالة.}
\begin{enumerate}[resume]
  \item من أجل زوج بيل $(99, 70)$: تحقق من $99^2 - 2\cdot70^2
        = 1$ وقوّم الخطأ المضبوط
        \[
        \sqrt2 - \frac{99}{70}
        = \frac{-1}{70^2\,\bigl(\sqrt2 + \frac{99}{70}\bigr)},
        \qquad
        \Bigl|\sqrt 2 - \frac{99}{70}\Bigr| \approx 7.2\cdot
        10^{-5} :
        \]
        أي خمسة أرقام صحيحة من كسر ذي ثلاثة أرقام.
  \item نفّذ الجزء 2 على $x = 2^{1/3}$ مع $P = X^3 - 2$: تحقق من أن $P$
        ليس له جذر ناطق، وحُدّ $M = \max_{\intcc{x-1}{x+1}}
        3t^2 \leq 3\,(1 + 2^{1/3})^2 < 16$، واستنتج
        المتراجحة الفعّالة
        \[
        \Bigl| 2^{1/3} - \frac pq \Bigr| \geq
        \frac{1}{16\,q^3} \qquad \text{من أجل كل } \frac pq .
        \]
  \item والربح: بيّن أن أيّ عدد ناطق يقرّب $2^{1/3}$
        بدقة $10^{-6}$ يجب أن يكون مقامه $q \geq 40$.
  \item بيّن أن الأساس $10$ لا شأن له: فالنظير الثنائي
        $\sum_{n\geq1} 2^{-n!}$ (وهو قيمة السلسلة الثنائية
        ذات الآحاد في المواضع العاملية) هو كذلك
        عدد ليوفيل، ومنه متسامٍ.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 6 --- الحدود الأمامية والتوليفة.}
\begin{enumerate}[resume]
  \item ليكن $x^\dagger$ قيمة السلسلة العشرية ذات
        الآحاد بالضبط في المواضع $3^k$ ($k \geq 0$). بيّن
        أن $x^\dagger$ قابل للتقريب من الرتبة $3$، واستنتج
        من متراجحة ليوفيل أن $x^\dagger$ ليس
        ناطقًا ولا عددًا أصمّ تربيعيًا. وفسّر لماذا
        تتعثّر الطريقة هناك: فالرتبة $3$ متوافقة مع
        الجبرية من الدرجة $\geq 3$، وسدّ تلك الفجوة
        (إذ يكفي أيّ أُسّ $> 2$، من أجل كل \omterm{pb:b1:logic:1}{عدد
        جبري}) هو مبرهنة روث، وهي أبعد بكثير من هذا المجلّد.
  \item كمِّم كانتور: بيّن أن عدد الأعداد الجبرية من
        الدرجة $\leq d$ المعطاة بكثيرات حدود بمعاملات
        في $\intint{-H}{H}$ هو $d\,(2H + 1)^{d+1}$ على الأكثر.
        (وهذا التناهي هو ما جعل السؤال 17 يعمل.)
  \item توليفة، جملة واحدة لكلٍّ: (أ) حدّد المكوّن
        التحليلي الوحيد في برهان ليوفيل (أيّ مبرهنة
        من هذا الفصل، ومستعملة أين)؛ (ب) اذكر التوتر
        الذي يحرّكه (فالصحيحية تفرض $\abs{P(p/q)} \geq
        q^{-d}$، والملاسة تمنع $\abs{P(p/q)} > M\abs{x -
        p/q}$)؛ (ج) وقابل برهانَي ليوفيل وكانتور
        على وجود الأعداد المتسامية؛ (د) وسمِّ
        الموضع الذي يلتقي فيه هذا المجلّد بالموضوع مرة أخرى --- إذ تبرهن
        مسألة نهاية الأسبوع في \cref{ch:b1:integration} على صمم $\pi$
        بحصر الصحيحية إزاء الصغر نفسه، بتكاملات بدل
        المشتقات.
\end{enumerate}
\end{problem}
