\chapter{الفضاءات الإقليدية}\label{ch:b1:euclid}

إضافة \omterm{def:b1:euclid:def}{جداء سلّمي} إلى \omterm{def:b1:vspaces:def}{فضاء متجهي} حقيقي تشتري المفاهيم
الهندسية --- الأطوال والزوايا والتعامد والمسافات --- و
مبرهنةً واحدة تعلو على الفصل: أي إن لكل \omterm{def:b1:vspaces:subspace}{فضاء جزئي}
إسقاطًا متعامدًا، قابلًا للحساب بغرام--شميت، ويحقق
أقصر مسافة. وتختم تقايسات المستوي الفصلَ وهندسة
السنة.

وفي كل ما يأتي، $E$ \omterm{def:b1:vspaces:def}{فضاء متجهي} \emph{حقيقي}.

\section{الجداءات السلّمية}

\begin{definition}\label{def:b1:euclid:def}
\emph{الجداء السلّمي}\index{جداء سلّمي} على $E$ \omterm{def:b1:logic:map}{تطبيق}
$\langle\cdot,\cdot\rangle \colon E \times E \to \R$ ثنائي الخطية، ومتماثل، ومعيَّن موجب ($\langle x, x\rangle >
0$ من أجل $x \neq 0$). والفضاء ذو البُعد المنتهي المزوَّد به
\emph{فضاء إقليدي}\index{فضاء إقليدي}. و\emph{معيار}
$x$ هو $\norm{x} = \sqrt{\langle x, x\rangle}$، و $d(x, y) =
\norm{x - y}$.
\end{definition}

\begin{example}\label{ex:b1:euclid:examples}
على $\R^n$: الجداء القانوني $\langle x, y\rangle = \sum x_i
y_i$. وعلى $C(\intcc{a}{b})$: $\langle f, g \rangle = \int_a^b fg$
(والتعيين الموجب هو \cref{thm:b1:integration:props}~(4)). وعلى
$\R_n[X]$: $\langle P, Q\rangle = \int_0^1 PQ$، أو $\sum_{i}
P(x_i)Q(x_i)$ على $n+1$ نقطة متمايزة.
\end{example}

\begin{example}[الزاوية بين كثيرَي حدود]
\label{ex:b1:euclid:polyangle}
ما إن يُختار \omterm{def:b1:euclid:def}{جداء سلّمي}، تكتسب \emph{أيّ} متجهتين غير معدومتين
زاويةً، عبر $\cos\theta = \frac{\langle x,
y\rangle}{\norm x\,\norm y}$ (وهو جيب تمام مشروع
بكوشي--شوارتز). ومن أجل $X$ و $X^2$ في $\int_0^1$:
\[
\langle X, X^2\rangle = \frac14, \qquad
\norm X = \frac1{\sqrt3}, \qquad \norm{X^2} = \frac1{\sqrt5},
\qquad
\cos\theta = \frac{1/4}{1/\sqrt{15}} = \frac{\sqrt{15}}{4}
\approx 0.968 :
\]
أي زاوية نحو $14.5$ درجة --- فعلى $\intcc{0}{1}$، يكون
منحنيا $x$ و $x^2$ «شبه متوازيين» بمعنى
المتوسط التربيعي، ولهذا لا تترك إزالة ذلك الاتجاه
المشترك (غرام--شميت أدناه) إلا التصحيح
الصغير $X^2 - X + \frac16$.
\end{example}

\begin{theorem}[كوشي--شوارتز؛ خصائص المعيار]\label{thm:b1:euclid:cs}
من أجل كل $x, y \in E$:\index{متراجحة كوشي--شوارتز}
\[
\abs{\langle x, y\rangle} \leq \norm x\, \norm y ,
\]
مع المساواة إذا وفقط إذا كان $x, y$ متناسبين. ونتيجةً لذلك يحقق $\norm\cdot$
متراجحةَ المثلث $\norm{x + y} \leq \norm x + \norm
y$ (و $\norm{\lambda x} = \abs\lambda \norm x$ و $\norm x = 0 \iff
x = 0$). وفوق ذلك:
\[
\norm{x+y}^2 = \norm x^2 + 2\langle x, y\rangle + \norm y^2,
\qquad
\langle x, y \rangle = \tfrac14\bigl(\norm{x+y}^2 -
\norm{x-y}^2\bigr) .
\]
\end{theorem}

\begin{proof}
إذا كان $y = 0$، فكل شيء بديهي. وإلا فإن \omterm{def:b1:poly:def}{كثير الحدود} من الدرجة الثانية $t
\mapsto \norm{x + ty}^2 = \norm x^2 + 2t\langle x, y\rangle + t^2
\norm y^2$ هو $\geq 0$ من أجل كل $t$: ومنه فمميّزه $\leq 0$،
وهذا هو كوشي--شوارتز؛ والمساواة تعني جذرًا مضاعفًا $t_0$، أي
$x + t_0 y = 0$ (بالتعيين): ومنه التناسب. ومتراجحة
المثلث: انشر،
\[
\norm{x + y}^2 = \norm x^2 + 2\langle x, y\rangle + \norm y^2
\leq \norm x^2 + 2\norm x\,\norm y + \norm y^2
= \bigl(\norm x + \norm y\bigr)^2 ,
\]
والخطوة الوسطى هي كوشي--شوارتز؛ وتفرض المساواة $\langle
x, y\rangle = \norm x\norm y$، أي حالة المساواة \emph{الموجبة}،
أي التناسب بنسبة غير سالبة ---
وهندسيًا، لا ينحلّ المثلث إلا عندما تشير المتجهتان
في الاتجاه نفسه. والمتطابقتان الأخيرتان
نشران مباشران (والثانية، وهي
\emph{متطابقة الاستقطاب}، تستعيد الجداء من المعيار).
\end{proof}

\section{التعامد}

\begin{definition}\label{def:b1:euclid:orthogonal}
يكون $x \perp y$ إذا كان $\langle x, y \rangle = 0$. وتكون العائلة
\emph{متعامدة} إذا كانت متجهاتها متعامدة مثنى مثنى، و
\emph{متعامدة ممنظمة}\index{عائلة متعامدة ممنظمة} إذا كان معيار
كلٍّ منها فوق ذلك $1$. و\emph{المتمم المتعامد} \omterm{def:b1:vspaces:subspace}{لفضاء جزئي} $F$ هو
\[
F^{\perp} = \{x \in E : \forall y \in F,\ \langle x, y\rangle =
0\},
\]
وهو \omterm{def:b1:vspaces:subspace}{فضاء جزئي} من $E$.\index{متمم متعامد}
\end{definition}

\begin{proposition}\label{prop:b1:euclid:orthfree}
(فيثاغورس\index{مبرهنة فيثاغورس}) إذا كان $x \perp y$ فإن
$\norm{x+y}^2 = \norm x^2 + \norm y^2$. وكل عائلة \omterm{def:b1:euclid:orthogonal}{متعامدة} من
متجهات غير معدومة حرةٌ. وفي \omterm{def:b1:vspaces:free}{أساس} متعامد ممنظم $(e_1, \dots,
e_n)$، تكون \omterm{prop:b1:vspaces:coordinates}{الإحداثيات} والجداءات
\[
x = \sum_{i} \langle x, e_i\rangle\, e_i,
\qquad
\langle x, y \rangle = \sum_i \langle x, e_i\rangle \langle y,
e_i\rangle,
\qquad
\norm x^2 = \sum_i \langle x, e_i\rangle^2 .
\]
\end{proposition}

\begin{proof}
فيثاغورس: انشر. والحرية: خذ $\langle\,\cdot\,, x_j\rangle$
لتركيبة معدومة: $\lambda_j \norm{x_j}^2 = 0$. \omterm{prop:b1:vspaces:coordinates}{والإحداثيات}:
اكتب $x = \sum \lambda_i e_i$ وخذ الجداء مع $e_j$:
$\lambda_j = \langle x, e_j\rangle$؛ وتتبع الصيغتان
بثنائية الخطية.
\end{proof}

\begin{example}[الإحداثيات المتعامدة الممنظمة، مع تحقق بارسيفال]
\label{ex:b1:euclid:parsevalcheck}
انشر $x = (1, 2, 3)$ في \omterm{def:b1:vspaces:free}{الأساس} المتعامد الممنظم للمقدار
\cref{exo:b1:euclid:3}،
\[
e_1 = \tfrac{1}{\sqrt2}(1,1,0), \quad
e_2 = \tfrac{1}{\sqrt6}(1,-1,2), \quad
e_3 = \tfrac{1}{\sqrt3}(-1,1,1).
\]
فلا جملة تُحلّ --- بل ثلاثة جداءات سلّمية:
\[
\langle x, e_1\rangle = \frac{3}{\sqrt2},
\qquad
\langle x, e_2\rangle = \frac{1 - 2 + 6}{\sqrt6} =
\frac{5}{\sqrt6},
\qquad
\langle x, e_3\rangle = \frac{-1 + 2 + 3}{\sqrt3} =
\frac{4}{\sqrt3}.
\]
والمصادقة بصيغة المعيار في القضية:
\[
\frac{9}{2} + \frac{25}{6} + \frac{16}{3}
= \frac{27 + 25 + 32}{6} = 14 = \norm{x}^2 = 1 + 4 + 9 .
\]
وتحقق مجموع مربعات \omterm{prop:b1:vspaces:coordinates}{الإحداثيات} هذا (وهو متطابقة بارسيفال
منتهية) يكلّف ثوانيَ ويمسك أخطاء الإشارة والتنظيم
بيقين شبه تامّ --- فاجعله عادة كلما حُسب
نشر متعامد ممنظم؛ وصيغته غير المنتهية البُعد، من أجل
معاملات فورييه في \cref{ex:b1:euclid:trigortho}، مبرهنةٌ في مجلّد السنة~الثالثة.
\end{example}

\begin{theorem}[غرام--شميت]\label{thm:b1:euclid:gramschmidt}
لكل \omterm{def:b1:euclid:def}{فضاء إقليدي} أسسٌ \omterm{def:b1:euclid:orthogonal}{متعامدة ممنظمة}. وصراحةً، من أيّ
\omterm{def:b1:vspaces:free}{أساس} $(v_1, \dots, v_n)$، تنتج الوصفة\index{طريقة غرام--شميت}
\[
w_k = v_k - \sum_{i=1}^{k-1} \langle v_k, e_i\rangle\, e_i ,
\qquad
e_k = \frac{w_k}{\norm{w_k}}
\]
\omterm{def:b1:vspaces:free}{أساسًا} متعامدًا ممنظمًا $(e_1, \dots, e_n)$ مع
$\operatorname{Vect}(e_1, \dots, e_k) = \operatorname{Vect}(v_1,
\dots, v_k)$ من أجل كل $k$.
\end{theorem}

\begin{proof}
بالاستقراء على $k$. وبفرض $(e_1, \dots, e_{k-1})$ \omterm{def:b1:euclid:orthogonal}{متعامدة ممنظمة}
تولّد $\operatorname{Vect}(v_1, \dots, v_{k-1})$: تكون المتجهة
$w_k$ \omterm{def:b1:euclid:orthogonal}{متعامدة} مع كل $e_j$ ($j < k$) بحكم البناء
($\langle w_k, e_j \rangle = \langle v_k, e_j\rangle - \langle v_k,
e_j\rangle$)، و $w_k \neq 0$ لأن $v_k \notin
\operatorname{Vect}(v_1, \dots, v_{k-1})$. ويبقي التنظيم
التعامدَ؛ وتصحّ \omterm{def:b1:logic:statement}{عبارة} التوليد لأن $e_k$
تركيبةٌ من $v_k$ ومن المقادير $e_i$ السابقة، بكيفية قابلة للقلب.
\end{proof}

\begin{example}[غرام--شميت على كثيرات الحدود، كاملًا]
\label{ex:b1:euclid:legendre}
عمّد $(1, X, X^2)$ في $\R_2[X]$ بالجداء $\langle P,
Q\rangle = \int_0^1 PQ$. \emph{الخطوة 1}: $\norm{1}^2 = 1$، ومنه
$e_1 = 1$. \emph{والخطوة 2}: $w_2 = X - \langle X, 1\rangle\,1 = X -
\frac12$، و $\norm{w_2}^2 = \int_0^1\bigl(x - \frac12\bigr)^2
\dd x = \frac1{12}$: ومنه $e_2 = \sqrt{12}\,\bigl(X - \frac12\bigr)$.
\emph{والخطوة 3}: $\langle X^2, e_1\rangle = \frac13$ و
\[
\langle X^2, e_2\rangle
= \sqrt{12}\int_0^1 x^2\Bigl(x - \frac12\Bigr)\dd x
= \frac{\sqrt{12}}{12},
\qquad\text{ومنه}\qquad
w_3 = X^2 - \frac13 - \Bigl(X - \frac12\Bigr)
= X^2 - X + \frac16 .
\]
ومعياره محسوب في \cref{exo:b1:euclid:9}: $\norm{w_3}^2 =
\frac1{180}$، ومنه $e_3 = \sqrt{180}\,\bigl(X^2 - X +
\frac16\bigr)$. وكثيرات الحدود $1$ و $X - \frac12$ و $X^2 - X +
\frac16$ هي، بغضّ النظر عن السلّم، أوائل \emph{كثيرات حدود لوجاندر}
\omterm{prop:b1:reals:intervals}{للفترة} $\intcc{0}{1}$؛ ويستمر البناء درجةً
درجة، وكل \omterm{def:b1:poly:def}{كثير حدود} جديد متعامد مع كل
سابقيه. ولاحظ كيف تعيد الخوارزمية تدوير العمل السابق: \omterm{def:b1:linmaps:projection}{فالإسقاط}
المطروح في الخطوة 3 هو بالضبط أفضل تقريب أفيني
للمقدار $X^2$ الموجود في \cref{ex:b1:euclid:bestapprox}
--- فغرام--شميت \emph{هو} \omterm{thm:b1:euclid:projection}{إسقاط متعامد} مكرَّر.
\end{example}

\begin{theorem}[الإسقاط المتعامد]\label{thm:b1:euclid:projection}
ليكن $F$ \omterm{def:b1:vspaces:subspace}{فضاءً جزئيًا} من \omterm{def:b1:euclid:def}{الفضاء الإقليدي} $E$. عندئذ
\[
E = F \oplus F^{\perp},
\]
ويُعطى \omterm{def:b1:linmaps:projection}{الإسقاط} المرتبط $p_F$ على $F$ (وهو \emph{الإسقاط
المتعامد}\index{إسقاط متعامد})، في أيّ \omterm{def:b1:vspaces:free}{أساس} متعامد ممنظم
$(e_1, \dots, e_k)$ للمقدار $F$، بالمقدار $p_F(x) = \sum_i
\langle x, e_i\rangle e_i$. وهو يحقق المسافة إلى $F$:
فمن أجل كل $y \in F$،
\[
\norm{x - p_F(x)} \leq \norm{x - y},
\]
مع المساواة من أجل $y = p_F(x)$ فقط؛ ويُكتب $d(x, F) = \norm{x -
p_F(x)}$.
\end{theorem}

\begin{proof}
خذ \omterm{def:b1:vspaces:free}{أساسًا} متعامدًا ممنظمًا $(e_i)_{i \leq k}$ للمقدار $F$
(\cref{thm:b1:euclid:gramschmidt} داخل $F$) وضع $\pi(x) =
\sum \langle x, e_i\rangle e_i \in F$. عندئذ $x - \pi(x) \perp e_j$
من أجل كل $j$ (بالتلاشي نفسه أعلاه)، ومنه $x - \pi(x) \in
F^\perp$: أي $E = F + F^\perp$. و $F \cap F^\perp = \{0\}$: لأن مثل هذه
المتجهة تحقق $\langle x, x\rangle = 0$. ومنه \omterm{def:b1:vspaces:sum}{فالمجموع مباشر} و
$\pi = p_F$.

والمسافة: من أجل $y \in F$، فكّك $x - y = (x - p_F(x)) + (p_F(x) -
y)$، وهما قطعتان متعامدتان ($F^\perp$ و $F$)؛ وبفيثاغورس:
\[
\norm{x - y}^2 = \norm{x - p_F(x)}^2 + \norm{p_F(x) - y}^2
\geq \norm{x - p_F(x)}^2,
\]
والمساواة إذا وفقط إذا كان $y = p_F(x)$.
\end{proof}

\begin{example}[الإسقاطات لا تطيل أبدًا]
\label{ex:b1:euclid:bessel}
\omterm{def:b1:logic:map}{بتطبيق} فيثاغورس على الانقسام $x = p_F(x) + (x - p_F(x))$:
\[
\norm{p_F(x)}^2 = \norm x^2 - \norm{x - p_F(x)}^2 \leq
\norm x^2 ,
\]
مع المساواة إذا وفقط إذا كان $x \in F$. وفي \omterm{def:b1:vspaces:free}{أساس} متعامد ممنظم $(e_1,
\dots, e_k)$ للمقدار $F$ يُقرأ هذا $\sum_{i \leq k}\langle x,
e_i\rangle^2 \leq \norm x^2$ (وهي \emph{متراجحة بسّل}):
فمهما بلغ عدد الاتجاهات \omterm{def:b1:euclid:orthogonal}{المتعامدة الممنظمة} التي يقيسها المرء، لا تتجاوز
مربعات \omterm{prop:b1:vspaces:coordinates}{الإحداثيات} مربعَ الطول --- وقارن
المساواة المضبوطة في \cref{ex:b1:euclid:parsevalcheck} عندما تكون
العائلة \omterm{def:b1:vspaces:free}{أساسًا} كاملًا. وهذه المتراجحة ذات السطر الواحد هي ما يجعل
معاملات فورييه قابلة للجمع في مجلّد السنة~الثالثة؛ وهي هنا
تفسّر أصلًا لماذا لا تستطيع إضافة دوال \omterm{def:b1:vspaces:free}{أساس} أخرى إلى مقايسة
بالمربعات الصغرى إلا أن تنقص الباقي.
\end{example}

\begin{example}[أفضل تقريب من الدرجة الثانية]\label{ex:b1:euclid:bestapprox}
في $C(\intcc{0}{1})$ بالجداء $\langle f, g\rangle = \int_0^1 fg$، يكون
\omterm{def:b1:poly:def}{كثير الحدود} من الدرجة $\leq 1$ الأقرب إلى $f(x) = x^2$ في
المسافة المرتبطة (متوسط تربيعي) هو $p_F(f)$ حيث $F = \R_1[X]$.
وغرام--شميت على $(1, X)$: $e_1 = 1$ و $w_2 = X - \frac12$ و
$\norm{w_2}^2 = \int_0^1 (x - \frac12)^2 = \frac{1}{12}$ و $e_2 =
\sqrt{12}\,(X - \tfrac12)$. عندئذ
\[
p_F(f) = \langle f, e_1\rangle e_1 + \langle f, e_2\rangle e_2
= \frac13 + 12\Bigl(\int_0^1 x^2\bigl(x - \tfrac12\bigr)\dd
x\Bigr)\bigl(X - \tfrac12\bigr)
= X - \frac{1}{6},
\]
باستعمال $\int_0^1 x^2(x - \frac12)\dd x = \frac14 - \frac16 =
\frac{1}{12}$. أي فكرة «المربعات الصغرى» في سطر واحد من الجبر
الخطي.
\end{example}

\begin{method}[ثلاثة طرق إلى مسافة $d(x, F)$]
\label{met:b1:euclid:distance}
\begin{enumerate}
  \item \emph{\omterm{def:b1:vspaces:free}{أساس} متعامد ممنظم للمقدار $F$}: عندئذ $p_F(x) = \sum_i
        \langle x, e_i\rangle e_i$ و، بفيثاغورس،
        \[
        d(x, F)^2 = \norm{x}^2 - \norm{p_F(x)}^2
        = \norm x^2 - \sum_i \langle x, e_i\rangle^2 ,
        \]
        وهو أرخص كثيرًا من حساب $x - p_F(x)$ نفسه.
  \item \emph{المعادلات الناظمية}: مع أيّ عائلة مولِّدة للمقدار
        $F$، حُلَّ $\langle x - p, v_j\rangle = 0$ من أجل
        معاملات $p$ (\cref{exo:b1:euclid:5}) --- دون أيّ
        تعميد.
  \item \emph{عبر المتمم}: إذا كان $F^\perp$ أصغر
        من $F$ (مثلًا $F$ \omterm{def:b1:linmaps:forms}{مستوٍ فائق} و $F^\perp$ مستقيم
        $\operatorname{Vect}(n)$)، فأسقط على $F^\perp$
        بدلًا من ذلك:
        \[
        d(x, F) = \norm{p_{F^\perp}(x)} =
        \frac{\abs{\langle x, n\rangle}}{\norm n} ,
        \]
        وهي صيغة المسافة الكلاسيكية إلى مستوٍ
        (ويستعملها \cref{exo:b1:multivar:8}).
\end{enumerate}
والطريق 3 حالة خاصة من منعكس عام: أسقط دائمًا
على أصغر بُعدًا من $F$ و $F^\perp$.
\end{method}

\begin{example}[التعامد المثلثي: عرض مسبق لفورييه]
\label{ex:b1:euclid:trigortho}
على $C(\intcc{0}{2\pi})$ بالجداء $\langle f, g\rangle =
\frac1\pi\int_0^{2\pi} fg$، تكون العائلة
\[
\Bigl(\frac{1}{\sqrt2},\ \cos x,\ \sin x,\ \cos 2x,\ \sin 2x,\
\dots\Bigr)
\]
\omterm{def:b1:euclid:orthogonal}{متعامدة ممنظمة}: فمثلًا $\langle \cos px, \cos qx\rangle =
\frac1\pi\int_0^{2\pi}\cos px\cos qx\,\dd x = 0$ من أجل $p \neq q$
(بتخطيط الجداء إلى $\frac12[\cos(p{-}q)x +
\cos(p{+}q)x]$ والمكاملة على دورات كاملة)، بينما
$\frac1\pi\int_0^{2\pi}\cos^2 px\,\dd x = 1$. ومنه فللإسقاط
المتعامد على \omterm{def:b1:vspaces:span}{الفضاء المولَّد} بالدوال $2N + 1$ الأولى منها
\omterm{prop:b1:vspaces:coordinates}{إحداثيات} $\langle f, e_i\rangle$ --- أي تكاملات
في مقابل جيوب التمام والجيوب. وهذه هي \emph{معاملات
فورييه} للمقدار $f$، ويكون \omterm{def:b1:linmaps:projection}{الإسقاط} أفضل
تقريب مثلثي بالمتوسط التربيعي؛ ويدرس مجلّد السنة~الثالثة
تقاربها. ويقوم التعامد بكل العمل:
فصيغ المعاملات هي
\cref{thm:b1:euclid:projection} حرفيًا.
\end{example}

\section{تقايسات المستوي}

\begin{definition}\label{def:b1:euclid:isometry}
يكون التشاكل الذاتي $u$ \omterm{def:b1:euclid:def}{لفضاء إقليدي}
\emph{تقايسًا}\index{تقايس} (أو تطبيقًا متعامدًا) إذا حفظ
المعيار: $\norm{u(x)} = \norm x$ من أجل كل $x$ ---
وبكيفية مكافئة (بالاستقطاب) يحفظ \omterm{def:b1:euclid:def}{الجداء السلّمي}؛
وبكيفية مكافئة تحقق مصفوفته $A$ في \omterm{def:b1:vspaces:free}{أساس} متعامد ممنظم
$A^{\mathsf T} A = I$. وتكوّن التقايسات زمرةً، هي \omterm{def:b1:structures:group}{الزمرة} \omterm{def:b1:euclid:orthogonal}{المتعامدة}
$O(E)$.
\end{definition}

\begin{example}[التعرّف على تقايس بالنظر]
\label{ex:b1:euclid:recognize}
هل $A = \dfrac15\begin{pmatrix} 3 & -4\\ 4 & 3\end{pmatrix}$
\omterm{def:b1:euclid:orthogonal}{متعامدة}؟ الأعمدة: المعياران $\frac15\sqrt{9 + 16} = 1$ و
$\frac15\sqrt{16 + 9} = 1$؛ والجداء $\frac1{25}(3\cdot(-4) +
4\cdot3) = 0$. نعم --- و $\det A = \frac{9 + 16}{25} = 1$، ومنه
فهي الدوران $R_\theta$ مع $\cos\theta = \frac35$ و
$\sin\theta = \frac45$ (وهو «دوران $3$-$4$-$5$»، وزاويته
ليست كسرًا لافتًا من $\pi$). وعلى النقيض فإن $B =
\frac{1}{\sqrt2}\begin{pmatrix} 1 & 1\\ 0 & 1\end{pmatrix}$ لها
مظهر مسلَّم بالمحدد الواحدي لكن عمودها الأول ليس \omterm{def:b1:poly:def}{واحديًا}
($\frac1{\sqrt2}$): فهي ليست \omterm{def:b1:euclid:orthogonal}{متعامدة} --- فالمحدد $\pm1$ وحده
لا يصادق على شيء، بل يجب التحقق من الأعمدة.
\end{example}

\begin{theorem}[تقايسات المستوي]\label{thm:b1:euclid:plane}
في \omterm{def:b1:vspaces:free}{أساس} متعامد ممنظم لمستوٍ إقليدي، تكون مصفوفات
التقايسات بالضبط
\[
R_\theta = \begin{pmatrix}
\cos\theta & -\sin\theta\\ \sin\theta & \cos\theta
\end{pmatrix}
\quad (\text{دوران بالزاوية } \theta,\ \det = 1),
\]
\[
S_\theta = \begin{pmatrix}
\cos\theta & \sin\theta\\ \sin\theta & -\cos\theta
\end{pmatrix}
\quad (\det = -1),
\]
والأخيرة هي الانعكاس في المستقيم الذي يصنع الزاوية
$\frac\theta2$ مع متجهة \omterm{def:b1:vspaces:free}{الأساس} الأولى.
\end{theorem}

\begin{proof}
لتكن $A = \begin{pmatrix} a & c\\ b & d\end{pmatrix}$ مع
$A^{\mathsf T}A = I$: فالأعمدة واحدية \omterm{def:b1:euclid:orthogonal}{ومتعامدة}. والعمود
الأول هو $(\cos\theta, \sin\theta)$ من أجل $\theta$ ما؛ والثاني،
الواحدي والمتعامد معه، هو $\pm(-\sin\theta, \cos\theta)$. وتعطي
الإشارة $+$ المقدارَ $R_\theta$؛ وتعطي الإشارة $-$ المقدارَ $S_\theta$. ويُتحقق من
$S_\theta^2 = I$ ومن أن المتجهة $(\cos\frac\theta2,
\sin\frac\theta2)$ صامدة بينما تُعكس متعامدتها: أي
انعكاس. (و $R_\alpha R_\beta = R_{\alpha+\beta}$: ومنه
\omterm{def:b1:structures:group}{فزمرة} الدورانات هي \omterm{def:b1:structures:group}{زمرة} الزوايا --- قارن
\cref{thm:b1:complex:funceq}.)
\end{proof}

\begin{remark}[مزالق شائعة]
\label{rem:b1:euclid:pitfalls}
\emph{صيغة \omterm{def:b1:linmaps:projection}{الإسقاط} تحتاج إلى \omterm{def:b1:vspaces:free}{أساس} متعامد ممنظم}: فمن أجل
عائلة مولِّدة فحسب $(v_i)$ للمقدار $F$، لا يكون المجموع $\sum_i\langle
x, v_i\rangle v_i$ هو $p_F(x)$ (اختبر $F = \R^2$ مع $v_1
= e_1$ و $v_2 = e_1 + e_2$)؛ ومع عائلة غير \omterm{def:b1:euclid:orthogonal}{متعامدة ممنظمة}، حُلَّ
المعادلات الناظمية بدل ذلك
(\cref{met:b1:euclid:distance}~(2)).
\emph{على العائلات \omterm{def:b1:euclid:orthogonal}{المتعامدة} أن تتجنّب $0$ لتكون \omterm{def:b1:vspaces:free}{حرة}}: فالمتجهة
المعدومة \omterm{def:b1:euclid:orthogonal}{متعامدة} مع كل شيء، بما فيه نفسها ---
والحرية في \cref{prop:b1:euclid:orthfree} تقتضي متجهات
غير معدومة.
\emph{المقدار $F^\perp$ يتعلق \omterm{def:b1:euclid:def}{بالجداء السلّمي}}: ففي $\R_1[X]$، ليس
متمم $\operatorname{Vect}(X)$ من أجل $\int_0^1 PQ$
هو الثوابت بل $\operatorname{Vect}(1 - \frac32 X)$ ---
احسب $\int_0^1 x(1 - \frac32 x) = \frac12 - \frac12 = 0$؛
فكلمة «العمودي» بلا معنى حتى يُسمّى الجداء.
\emph{لا تنشر $\norm{x + y}$ نشرًا \omterm{def:b1:linmaps:def}{خطيًا}}: فالمتطابقة
الصحيحة هي $\norm{x+y}^2 = \norm x^2 + 2\langle x, y\rangle +
\norm y^2$؛ ولا ينعدم الحدّ المتقاطع إلا بالتعامد
(فيثاغورس)، ومتراجحة المثلث
\emph{متراجحة}.
\emph{إرسال المتجهات الواحدية إلى متجهات واحدية لا يكفي}: \omterm{def:b1:logic:map}{فالتطبيق} $u(x,
y) = (x + y,\ 0)$ يرسل متجهتَي \omterm{def:b1:vspaces:free}{الأساس} القانوني كلتيهما إلى المتجهة
الواحدية $(1, 0)$، ومع ذلك $\norm{u(1,1)} = 2 \neq \sqrt2$: فلا
\omterm{def:b1:euclid:isometry}{تقايس}. ويطلب التعريف $\norm{u(x)} = \norm x$ من أجل
\emph{كل} $x$؛ وبلغة المصفوفات $A^{\mathsf T}A = I$، أي
أعمدة واحدية \emph{\omterm{def:b1:euclid:orthogonal}{ومتعامدة} مثنى مثنى} --- الشرطان
معًا، متحقَّقًا منهما معًا.
\end{remark}

\begin{remark}[إلى أين يذهب الجداء السلّمي]
\label{rem:b1:euclid:whereused}
\omterm{thm:b1:euclid:projection}{الإسقاط المتعامد} أكثر مبرهنات الفصل تطبيقًا:
فهو يقوم تحت المربعات الصغرى (وتبني مسألة نهاية الأسبوع في
\cref{ch:b1:multivar} مستقيمات الانحدار عليه)، ومعاملات
فورييه (\cref{ex:b1:euclid:trigortho})، والمعادلات
الناظمية في \cref{exo:b1:euclid:5}، التي يحلّها التحليل العددي
على نطاق واسع. ويكتمل التصنيف $R_\theta / S_\theta$
أدناه: فتصنّف مسألة نهاية الأسبوع \emph{كل}
التحويلات الحافظة للمسافات في المستوي، خطيةً كانت أو لا،
وزمرها المنتهية --- وهي رياضيات الورديات والمضلعات
المنتظمة. وفي مجلّد السنة~الثانية يلتقي \omterm{def:b1:euclid:def}{الجداء السلّمي} بنظرية
القيم الذاتية (المصفوفات المتماثلة والأشكال التربيعية)؛ وفي
السنة~الثالثة تصير الهندسة الإقليدية غير المنتهية البُعد نظريةَ فضاءات
هيلبرت.
\end{remark}

\begin{remark}[منظورات داخل الكتاب 3]
\label{rem:b1:euclid:perspectives}
جسران يغادران هذا الفصل. \emph{إلى الوراء}، إلى الجبر
الخطي: تعلّب مصفوفة غرام في \cref{exo:b1:euclid:11}
الجداءاتِ السلّمية في آلة المحددات في
\cref{ch:b1:det}، ويكون \omterm{thm:b1:euclid:projection}{الإسقاط المتعامد} المسقطَ الخاص
في \cref{ch:b1:linmaps} الذي نواته
$F^\perp$ --- وينطبق كل جبره ($p^2 = p$ و $s = 2p -
\mathrm{id}$) حرفيًا، مع مكافأة أن $\norm{x
- p(x)}$ \emph{مسافة}. \emph{وإلى الأمام}، إلى التحليل:
يقيس \cref{ch:b1:curves} طول القوس بمعيار هذا الفصل
ولا يصنّف شيئًا دون تقايساته؛ و
يقرأ \cref{ch:b1:multivar} التدرجَ عبر
كوشي--شوارتز (أشدّ صعود) ويختم المجلّد بالمربعات
الصغرى، وهي \cref{thm:b1:euclid:projection} مطبَّقًا
على متجهة معطيات في $\R^n$. \omterm{def:b1:euclid:def}{والجداء السلّمي} هو الموضع
الذي يلتقي فيه جبر الكتاب وتحليله أخيرًا.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:euclid:2}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:euclid:3}
طبّق غرام--شميت على $\bigl((1,1,0), (1,0,1), (0,1,1)\bigr)$ في

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:euclid:4}
في $\R^3$، لتكن $F = \operatorname{Vect}\bigl((1,1,1)\bigr)$.
حدّد $F^\perp$ (معادلةً \omterm{def:b1:vspaces:free}{وأساسًا})، ومصفوفة $p_F$ في

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:euclid:5}
(المعادلات الناظمية) لتكن $F =
\operatorname{Vect}\bigl((1,0,1),(0,1,1)\bigr) \subseteq \R^3$ و

\rangle = 0$ من أجل المولّدين ($p = \alpha(1,0,1) +
\beta(0,1,1)$)، ثم $d(x, F)$. ولماذا لا يلزم غرام--شميت
هنا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:euclid:6}
من أجل $f$ \omterm{def:b1:continuity:continuous}{متصلة} على $\intcc{0}{1}$، برهن على

\Bigl(\int_0^1 f\Bigr)^{\!2} \leq \int_0^1 f^2 ,
\]
مع حالة المساواة، مثالًا على كوشي--شوارتز في
$C(\intcc{0}{1})$. ثم برهن على $\bigl(\sum_{i=1}^n a_i\bigr)^2 \leq

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:euclid:7}

$(F^{\perp})^{\perp} = F$ و $\dim F^{\perp} = \dim E - \dim F$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:euclid:9}
(القيمة الصغرى إسقاطًا) احسب

\]
باستعمال \cref{ex:b1:euclid:bestapprox}: فالقيمة الصغرى هي $\norm{f -

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:euclid:10}
ليكن $u$ \omterm{def:b1:euclid:isometry}{تقايسًا} \omterm{def:b1:euclid:def}{لفضاء إقليدي} $E$. برهن على أن
$\ker(u - \mathrm{id}) \perp \operatorname{im}(u - \mathrm{id})$،
واستنتج $E = \ker(u - \mathrm{id}) \oplus \operatorname{im}(u -
\mathrm{id})$. \emph{(احسب $\langle x - u(x), y\rangle$ من أجل
$u(y) = y$، مستعملًا حفظ الجداء.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:euclid:11}
(مصفوفة غرام) من أجل متجهات $v_1, \dots, v_k$ من \omterm{def:b1:euclid:def}{فضاء
إقليدي}، لتكن $G = \bigl(\langle v_i, v_j\rangle\bigr)_{1 \leq i, j
\leq k}$ \emph{مصفوفة غرام} لها.
\begin{enumerate}
  \item برهن على أن $(v_1, \dots, v_k)$ \omterm{def:b1:vspaces:free}{حرة} إذا وفقط إذا كانت
        $G$ قابلة للقلب. \emph{(إذا كان $Gc = 0$، فاحسب
        $\norm{\sum_i c_i v_i}^2$.)}
  \item احسب مصفوفة غرام للعائلة $(1, X, X^2)$ في $\R_2[X]$
        بالجداء $\langle P, Q\rangle = \int_0^1 PQ$، وتعرّف على
        \omterm{pb:b1:det:1}{مصفوفة هيلبرت} $H_3$ في مسألة نهاية الأسبوع في
        \cref{ch:b1:det}، واستنتج الحرية من $\det H_3 =
        \frac1{2160} \neq 0$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:euclid:12}
ليكن $u$ \omterm{def:b1:structures:morphism}{تشاكلًا} ذاتيًا \omterm{def:b1:euclid:def}{لفضاء إقليدي} $E$ مصفوفته
$A$ في \omterm{def:b1:vspaces:free}{أساس} متعامد ممنظم متعامدةً ($A^{\mathsf T}A
= I$) ومتماثلة ($A^{\mathsf T} = A$) معًا.
\begin{enumerate}
  \item بيّن $A^2 = I$، واستنتج (عبر
        \cref{thm:b1:linmaps:projchar}) أن $E = \ker(u -
        \mathrm{id}) \oplus \ker(u + \mathrm{id})$.
  \item بيّن أن الفضاءين الجزئيين متعامدان، ومنه \omterm{def:b1:logic:map}{فالتطبيق} $u$
        هو \emph{التماثل المتعامد} بالنسبة إلى $F =
        \ker(u - \mathrm{id})$: أي الانعكاس عبر $F$.
        \emph{(من أجل $u(x) = x$ و $u(y) = -y$، احسب $\langle
        x, y\rangle$ بكيفيتين.)}
  \item صنّف حالة المستوي: أيّ مصفوفات
        \cref{thm:b1:euclid:plane} متماثلة، وما التطبيقات
        المقابلة؟
\end{enumerate}
\end{exercise}

\section{مسألة: تقايسات المستوي ومبرهنة ليوناردو}

\begin{problem}\label{pb:b1:euclid:1}
\emph{\omterm{def:b1:euclid:isometry}{تقايس} المستوي} هو أيّ \omterm{def:b1:logic:map}{تطبيق} $f \colon \R^2 \to
\R^2$ يحفظ المسافات: $\norm{f(x) - f(y)} = \norm{x - y}$
من أجل كل $x, y$ --- دون افتراض خطية. وتبرهن هذه المسألة
على أن مثل هذه التطبيقات هي بالضبط الانسحابات والدورانات
والانعكاسات والانعكاسات المنزلقة (وهو تصنيف تقايسات
المستوي)، وتحسب تراكيبها، وتحدّد كل
زمرها \emph{المنتهية}: أي مبرهنة ليوناردو، وهي رياضيات
ما وراء أنماط الورديات. ومن الجزء 2 فصاعدًا نطابق $\R^2$ مع
$\C$ (\cref{ch:b1:complex}): \omterm{def:b1:euclid:def}{فالجداء السلّمي} القانوني هو
$\langle z, w\rangle = \operatorname{Re}(z\conj w)$ والمعيار
هو \omterm{def:b1:complex:field}{المقياس}.\index{تقايس}\index{انعكاس منزلق}
\index{مبرهنة ليوناردو}\index{زمرة ثنائية السطح}

\medskip
\noindent\textbf{الجزء 1 --- كل \omterm{def:b1:euclid:isometry}{تقايس} أفيني.}
\begin{enumerate}
  \item تحقق من أن الانسحابات $t_a(x) = x + a$ والتقايسات
        الخطية وكل تراكيبها تقايساتٌ،
        ومن أن التقايسات تكوّن زمرةً بالتركيب.
  \item ليكن $f$ \omterm{def:b1:euclid:isometry}{تقايسًا} مع $f(0) = 0$. بيّن أن $f$
        يحفظ المعايير، ثم --- بالاستقطاب،
        \cref{thm:b1:euclid:cs} --- أن $\langle f(x),
        f(y)\rangle = \langle x, y\rangle$ من أجل كل $x, y$.
  \item ومع $f(0) = 0$ دائمًا: انشر $\norm{f(x + y) - f(x) -
        f(y)}^2$ و $\norm{f(\lambda x) - \lambda f(x)}^2$
        باستعمال السؤال 2، واستنتج أن $f$
        \emph{خطي}: أي $f \in O(\R^2)$.
  \item استنتج أن كل \omterm{def:b1:euclid:isometry}{تقايس} $f$ يُكتب \emph{كتابة وحيدة}
        $f = t_a \circ g$ مع $a = f(0)$ و $g$ \omterm{def:b1:euclid:isometry}{تقايسًا}
        \omterm{def:b1:linmaps:def}{خطيًا} (وهو \emph{الجزء الخطي} للمقدار $f$).
  \item سمِّ $f$ \emph{مباشرًا} إذا كان $\det g = 1$، و\emph{غير مباشر}
        إذا كان $\det g = -1$. بيّن أن الجزء الخطي لتركيب
        هو تركيب الجزأين الخطيين، واذكر
        قاعدة الإشارة الناتجة
        (المباشر/غير المباشر يتركبان مثل $+1/-1$).
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- الأنواع الأربعة.} عبر
\cref{thm:b1:euclid:plane}، تكون التقايسات الخطية للمقدار $\C$ هي $z
\mapsto az$ و $z \mapsto a\conj z$ مع $\abs a = 1$؛ ومنه فكل
\omterm{def:b1:euclid:isometry}{تقايس} مستوٍ هو
\[
f(z) = a z + b \quad (\text{مباشر}) \qquad\text{أو}\qquad
f(z) = a\conj z + b \quad (\text{غير مباشر}), \qquad \abs a = 1 .
\]
\begin{enumerate}[resume]
  \item تحقق من القاموس: $R_\theta$ هو $z \mapsto
        \eu^{\iu\theta}z$ و $S_\theta$ هو $z \mapsto
        \eu^{\iu\theta}\conj z$ (تحقق من كليهما على $1$ و $\iu$).
  \item (الحالة المباشرة) ليكن $f(z) = az + b$ مع $\abs a = 1$. بيّن:
        أنه إذا كان $a = 1$ فإن $f$ انسحاب؛ وإذا كان $a \neq 1$ فللمقدار $f$
        نقطة صامدة وحيدة $z_0 = b/(1 - a)$ و $f(z) - z_0
        = a(z - z_0)$: أي \emph{دوران} مركزه $z_0$ و
        زاويته $\arg a$.
  \item (الحالة غير المباشرة) ليكن $f(z) = a\conj z + b$، ولتكن $v =
        a\conj b + b$. بيّن $f \circ f = t_v$ و $f \circ t_v =
        t_v \circ f$. وإذا كان $v = 0$: فبيّن أن منتصف $z$
        و $f(z)$ نقطة صامدة، وأن $f$
        \emph{انعكاس} في مستقيم. وإذا كان $v \neq 0$: فبيّن أن $r
        = t_{-v/2}\circ f$ انعكاسٌ محوره
        موازٍ للمقدار $v$، بحيث يكون $f = t_{v/2} \circ r$
        \emph{انعكاسًا منزلقًا}. واختم: كل \omterm{def:b1:euclid:isometry}{تقايس} مستوٍ
        انسحابٌ أو دوران أو انعكاس أو \omterm{pb:b1:euclid:1}{انعكاس
        منزلق} (وهو تصنيف تقايسات المستوي).
  \item (التراكيب) بيّن: أن تركيب دورانين بزاويتين
        $\alpha$ و $\beta$ دورانٌ بالزاوية
        $\alpha + \beta$ (وانسحابٌ إذا كان $\alpha + \beta \in
        2\pi\Z$)؛ وأن تركيب انعكاسين
        دورانٌ بضِعف الزاوية بين المحورين (و
        انسحابٌ إذا كان المحوران متوازيين).
  \item استنتج أن كل \omterm{def:b1:euclid:isometry}{تقايس} مستوٍ تركيبُ ثلاثة
        انعكاسات على الأكثر.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- ثلاثة تعيينات.}
\begin{enumerate}[resume]
  \item صنّف تصنيفًا كاملًا $f(z) = \iu\conj z + 1 + \iu$:
        النوع والمحور ومتجهة الانزلاق.
  \item ليكن $f$ الدوران بالزاوية $\frac\pi2$ حول $0$
        وليكن $g$ الدوران بالزاوية $\frac\pi2$ حول $1$.
        احسب $g \circ f$ في الصورة $z \mapsto az + b$ و
        عيّنه (النوع والمركز والزاوية).
  \item ليكن $r_1(z) = \conj z$ (الانعكاس في المحور الحقيقي) و
        $r_2(z) = \iu\conj z$ (الانعكاس في المستقيم $y = x$).
        احسب $r_2 \circ r_1$ وتحقق من السؤال 9 على هذا
        المثال.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- الزمر المنتهية: مبرهنة ليوناردو.}
لتكن $G$ زمرةً \emph{منتهية} من تقايسات المستوي.
\begin{enumerate}[resume]
  \item بيّن أن التقايسات تحفظ المرجّحات: فإذا كان
        $\lambda_1 + \dots + \lambda_m = 1$ و $f = t_a \circ
        g$ ($g$ خطي)، فإن $f\bigl(\sum_i \lambda_i
        x_i\bigr) = \sum_i \lambda_i f(x_i)$.
  \item ضع $c = \frac{1}{\abs G}\sum_{g \in G} g(x_0)$ من أجل أيّ
        $x_0$ مختار. بيّن أن كل $h \in G$ يثبّت $c$: أي إن
        \omterm{def:b1:structures:group}{للزمرة} المنتهية من التقايسات نقطة صامدة مشتركة.
  \item استنتج أنه، بعد الاقتران بالمقدار $t_{-c}$، يمكن
        افتراض $G \subseteq O(\R^2)$. ولتكن $G^{+} = \{g \in G :
        \det g = 1\}$؛ وبيّن أنه إمّا $G = G^{+}$ وإمّا أن $G^{+}$
        دليله $2$ بالضبط في $G$ (بإظهار تقابل $G^+
        \to G \setminus G^+$).
  \item بيّن أن \omterm{def:b1:structures:group}{الزمرة} المنتهية من \emph{الدورانات} حول $c$
        دوريةٌ: فبين عناصرها اختر الدوران
        $R_{\theta_0}$ ذا أصغر زاوية $\theta_0 \in
        \intoo{0}{2\pi}$، واستعمل القسمة الإقليدية للزوايا
        لتبرهن على أنه يولّد؛ واختم بأن $\theta_0 =
        \frac{2\pi}n$ و $G^{+} = \{R_{\theta_0}^{\,k}\} \cong
        C_n$.
  \item افترض $G \neq G^{+}$ واختر انعكاسًا $s \in G$.
        بيّن $G = G^{+} \cup sG^{+}$، وأن $s r s = r^{-1}$ من أجل
        كل دوران $r \in G^{+}$، وأن كل العناصر $n$ في
        $sG^{+}$ انعكاسات: ومنه \omterm{def:b1:structures:group}{فالزمرة} $G$ هي \emph{\omterm{pb:b1:euclid:1}{الزمرة
        ثنائية السطح}} $D_n$ من الرتبة $2n$.
  \item اختم (\emph{بمبرهنة ليوناردو}): كل \omterm{def:b1:structures:group}{زمرة} منتهية
        من تقايسات المستوي دورية $C_n$ أو ثنائية السطح
        $D_n$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- الأرباح، والتوليفة.}
\begin{enumerate}[resume]
  \item بيّن مباشرةً أن \omterm{def:b1:structures:group}{الزمرة} المنتهية من التقايسات لا يمكن
        أن تحتوي أيّ انسحاب ولا أيّ \omterm{pb:b1:euclid:1}{انعكاس منزلق} غير
        \omterm{def:b1:logic:map}{التطبيق} المطابق (انظر في قوى مثل هذا
        العنصر).
  \item ليكن $P_n$ المضلع المنتظم ذا $n$ ضلعًا برؤوسه
        \omterm{def:b1:complex:unity}{جذور الوحدة} من الرتبة $n$ ($n \geq 3$). بيّن أن \omterm{def:b1:structures:group}{زمرة}
        تماثله هي $D_n$ بالضبط: أي الدورانات $n$ $z
        \mapsto \omega^k z$ والانعكاسات $n$ $z \mapsto
        \omega^k \conj z$ مع $\omega = \eu^{2\iu\pi/n}$، ولا
        غير ذلك.
  \item أظهر أشكالًا مستوية زمر تماثلها $C_1$ و
        $D_1$ و $D_2$ و $C_3$ على الترتيب.
  \item اذكر العناصر الثمانية \omterm{def:b1:structures:group}{لزمرة} تماثل
        المربع ذي الرؤوس $\pm1, \pm\iu$ تطبيقاتٍ $z \mapsto
        \omega^k z$ أو $z \mapsto \omega^k\conj z$، وأعطِ
        محور كلٍّ من الانعكاسات الأربعة.
  \item ليكن $f, g$ دورانين بالزاوية نفسها $\theta
        \notin 2\pi\Z$ حول مركزين \emph{متمايزين} $c_1 \neq
        c_2$. احسب $f\circ g - g\circ f$ نقطةً نقطة وبيّن
        $f\circ g \neq g\circ f$؛ وبيّن فوق ذلك أن $(f \circ
        g)\circ(g\circ f)^{-1}$ انسحابٌ غير بديهي، ومنه
        تكون كل \omterm{def:b1:structures:group}{زمرة} تحتوي $f$ و $g$ غير منتهية ---
        وهو تفسير ثانٍ للمركز الوحيد في مبرهنة
        ليوناردو.
  \item توليفة، في أربع جمل: أيّ نتيجتين بنيويتين
        تردّان التقايسات الكيفية إلى الجبر الخطي
        (السؤالان 3--4) والزمر المنتهية الكيفية إلى
        زمر جزئية من $O(2)$ (السؤال 15)؛ وما القائمة
        الكاملة لتقايسات المستوي وأيّ قرائن
        (مباشر/غير مباشر، النقاط الصامدة) تفصل الأنواع الأربعة؛
        ولماذا تجعل قواعد التركيب في السؤال 9 الانعكاسات
        مولّدات كل شيء؛ وماذا تضيف مبرهنة ليوناردو
        على السلّم المنتهي. وسمِّ المبرهنتين
        المبرهن عليهما في الجزأين 2 و 4.
\end{enumerate}
\end{problem}
