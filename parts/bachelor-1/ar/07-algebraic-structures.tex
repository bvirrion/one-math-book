\chapter{البنى الجبرية}\label{ch:b1:structures}

تعود قواعد الحساب نفسها إلى الظهور مرارًا: في الأعداد الصحيحة والأعداد
الحقيقية والأعداد العقدية وصفوف \omterm{def:b1:arith:congruence}{التوافق}، وقريبًا في كثيرات الحدود
(\cref{ch:b1:poly}) والمتجهات والمصفوفات
(\cref{ch:b1:vspaces,ch:b1:matrices}). ويستخرج الجبر الأنماط
المشتركة ويسمّيها: \emph{\omterm{def:b1:structures:group}{الزمرة}} و\emph{\omterm{def:b1:structures:ring}{الحلقة}} و\emph{\omterm{def:b1:structures:field}{الحقل}}.
والبرهان على واقعة مرة واحدة، على مستوى البنية، يبرهن عليها من أجل
كل مثال دفعةً واحدة.

\section{قوانين التركيب}

\begin{definition}\label{def:b1:structures:law}
\emph{قانون التركيب}\index{قانون تركيب} على \omterm{def:b1:logic:sets}{مجموعة} $E$ هو
\omterm{def:b1:logic:map}{تطبيق} $E \times E \to E$، يُكتب $(x, y) \mapsto x * y$. ويكون
\emph{تجميعيًا} عندما يكون $(x*y)*z = x*(y*z)$ دائمًا، و\emph{تبديليًا}
عندما يكون $x * y = y * x$ دائمًا. ويكون العنصر $e$ \emph{محايدًا} عندما يكون
$e * x = x * e = x$ لكل $x$؛ وعندئذ يكون $x'$ \emph{مقلوبًا} للعنصر $x$
عندما يكون $x * x' = x' * x = e$.
\end{definition}

\begin{proposition}[الوحدانية]\label{prop:b1:structures:unique}
للقانون عنصر محايد واحد على الأكثر؛ ومن أجل قانون تجميعي ذي عنصر محايد،
يكون لكل عنصر مقلوب واحد على الأكثر.
\end{proposition}

\begin{proof}
إذا كان $e$ و $e'$ محايدين: $e = e * e' = e'$. وإذا قلب $x'$ و $x''$
العنصرَ $x$: $x' = x' * e = x' * (x * x'') = (x' * x) * x'' = e * x''
= x''$.
\end{proof}

\section{الزمر}

\begin{definition}[الزمرة]\label{def:b1:structures:group}
\emph{الزمرة}\index{زمرة} $(G, *)$ مجموعةٌ ذات قانون تجميعي
يقبل عنصرًا محايدًا ولكل عنصر فيها مقلوب. وتكون
الزمرة \emph{تبديلية}\index{زمرة تبديلية} عندما يكون القانون
تبديليًا.
\end{definition}

\begin{example}\label{ex:b1:structures:groups}
$(\Z, +)$ و $(\Q, +)$ و $(\R, +)$ و $(\C, +)$؛ و $(\Q^*, \times)$
و $(\R^*, \times)$ و $(\C^*, \times)$ و$(\mathbb{U}_n, \times)$ (\omterm{def:b1:complex:unity}{جذور
الوحدة}، \cref{def:b1:complex:unity})؛ \omterm{def:b1:logic:sets}{ومجموعة} $\mathfrak{S}(E)$ المكوَّنة من
تقابلات \omterm{def:b1:logic:sets}{مجموعة} $E$ على نفسها، بالتركيب --- وهي
\emph{الزمرة المتناظرة}\index{زمرة متناظرة} \omterm{def:b1:logic:sets}{للمجموعة} $E$، وهي غير \omterm{def:b1:structures:group}{تبديلية} بمجرد
أن يكون $\abs E \geq 3$. وليست زمرًا: $(\N, +)$ (فلا مقلوبات)،
و $(\Z, \times)$ (فلا يقبل القلب إلا $\pm 1$).
\end{example}

\begin{proposition}[قواعد الحساب]\label{prop:b1:structures:rules}
في \omterm{def:b1:structures:group}{زمرة} $G$ (مكتوبة ضربيًا، وعنصرها المحايد $e$):
\begin{enumerate}
  \item الاختصار: $ax = ay \implies x = y$ و$xa = ya \implies
        x = y$؛
  \item $(ab)^{-1} = b^{-1} a^{-1}$ و $(a^{-1})^{-1} = a$؛
  \item من أجل $a, b \in G$، لكلٍّ من المعادلتين $ax = b$ و $xa = b$
        حلٌّ وحيد ($x = a^{-1}b$ و $x = b a^{-1}$ على التوالي).
\end{enumerate}
\end{proposition}

\begin{proof}
(1) اضرب في $a^{-1}$ من الجهة المناسبة، مستعملًا
التجميعية. (2) $(b^{-1}a^{-1})(ab) = b^{-1}(a^{-1}a)b = b^{-1}b =
e$ وبالتناظر كذلك؛ وتُتمّ وحدانية المقلوب البرهان؛ والنقطة
الثانية هي \cref{prop:b1:structures:unique} مطبَّقة على $a^{-1}$. (3)
عوّض واستعمل (1) من أجل الوحدانية.
\end{proof}

\begin{example}[تناظرات المستطيل]\label{ex:b1:structures:klein}
يقبل المستطيل (غير المربع) أربعة تقايسات على
نفسه بالضبط: \omterm{def:b1:logic:map}{التطبيق} المطابق $e$، والتناظر بالنسبة إلى المحور الأفقي $h$،
والتناظر بالنسبة إلى المحور العمودي $v$، ونصف الدورة $r$ حول
المركز. ويجعل التركيب هذه \omterm{def:b1:logic:sets}{المجموعة} الرباعية \omterm{def:b1:structures:group}{زمرة}: فكل
عنصر مقلوب نفسه ($h^2 = v^2 = r^2 = e$)، ويكون
جداء أيّ عنصرين متمايزين غير محايدين هو الثالث
($hv = vh = r$: فالتناظر بالنسبة إلى المحورين هو نصف الدورة). و
الجدول الكامل متناظر، \omterm{def:b1:structures:group}{فالزمرة تبديلية} --- ومع ذلك فهي
\emph{ليست} \omterm{def:b1:structures:group}{الزمرة} نفسها التي هي \omterm{def:b1:structures:group}{زمرة} الدورانات $\mathbb U_4$ في
\cref{ex:b1:structures:order}: فهناك \omterm{def:b1:structures:order}{رتبة} $\iu$ تساوي $4$، بينما
هنا \omterm{def:b1:structures:order}{رتبة} كل عنصر $\leq 2$. فزمرتان لهما الحجم نفسه
قد يكون لهما بنيتا ضرب مختلفتان حقًّا
--- ويعرض الرسم أدناه الجدولين جنبًا إلى جنب. وتعود هذه
\omterm{def:b1:structures:group}{الزمرة} الرباعية في صورة $\{\pm1\} \times \{\pm1\}$، ويشرح
\cref{exo:b1:structures:7} لماذا يجب على أيّ \omterm{def:b1:structures:group}{زمرة} كل مربعاتها
محايدة أن تكون، مثل هذه، \omterm{def:b1:structures:group}{تبديلية}.
\end{example}

\begin{definition}[الزمرة الجزئية]\label{def:b1:structures:subgroup}
الجزء $H$ من \omterm{def:b1:structures:group}{زمرة} $G$ يكون \emph{زمرة جزئية}\index{زمرة جزئية}
(ويُكتب $H \leq G$) عندما يحتوي $e$ ويكون مستقرًّا بالقانون
وبالقلب. وعندئذ تكون $H$ زمرةً بنفسها.

\emph{المعيار:} الجزء غير الخالي $H \subseteq G$ يكون \omterm{def:b1:structures:group}{زمرة} جزئية إذا وفقط
إذا كان
\[
\forall x, y \in H, \quad x y^{-1} \in H .
\]
\end{definition}

\begin{proof}[برهان المعيار]
\omterm{def:b1:structures:subgroup}{الزمرة الجزئية} تحققه بداهةً. وبالعكس، لتكن $H \neq \emptyset$
تحققه، ولنأخذ $x_0 \in H$. عندئذ $e = x_0 x_0^{-1} \in H$؛ ومن أجل
$y \in H$ لدينا $y^{-1} = e\,y^{-1} \in H$؛ ومن أجل $x, y \in H$ لدينا $xy =
x (y^{-1})^{-1} \in H$.
\end{proof}

\begin{example}\label{ex:b1:structures:subgroups}
$\mathbb{U}_n \leq (\C^*, \times)$: فهي غير خالية، ومن أجل $z, w \in
\mathbb{U}_n$ لدينا $(zw^{-1})^n = z^n (w^n)^{-1} = 1$. والزمر الجزئية
\omterm{def:b1:structures:group}{للزمرة} $(\Z, +)$ هي بالضبط $n\Z$ (المبرهن عليه في
\cref{thm:b1:arith:gcd}). وتقاطع زمر جزئية زمرةٌ جزئية دائمًا،
لكن اتحادها لا يكون كذلك في الغالب الأعمّ
(\cref{exo:b1:structures:6}).
\end{example}

\begin{remark}[مزالق شائعة مع البنى]\label{rem:b1:structures:pitfalls}
\begin{enumerate}
  \item \emph{الاستقرار بالقانون لا يكفي.} \omterm{def:b1:logic:sets}{المجموعة} $\N$
        مستقرة بالجمع داخل $\Z$، وتحتوي $0$، ومع ذلك ليست
        \omterm{def:b1:structures:subgroup}{زمرة جزئية}: فالمقلوبات ناقصة. والمعيار
        $xy^{-1} \in H$ يختبر كل شيء دفعة واحدة --- لكن بعد
        التحقق من أن $H \neq \emptyset$.
  \item \emph{المنعكسات غير \omterm{def:b1:structures:group}{التبديلية}.} في \omterm{def:b1:structures:group}{زمرة} عامة، $(ab)^2 =
        abab$، وهو $a^2b^2$ عندما يتبادل $a$ و $b$ لا غير؛
        وبالمثل $(ab)^{-1} = b^{-1}a^{-1}$، بترتيب معكوس.
        فكل متطابقة مستوردة من جبر المدرسة يجب
        أن تُستخرج من جديد من البديهيات أو يُنبَّه إلى أنها \omterm{def:b1:structures:group}{تبديلية}.
  \item \emph{\omterm{def:b1:structures:morphism}{النواة} في مقابل الصورة.} تحيا $\ker f$ في
        \emph{\omterm{def:b1:logic:sets}{مجموعة} التعريف}، ويحيا $\operatorname{im} f$ في \omterm{def:b1:logic:sets}{مجموعة} الوصول؛
        \omterm{def:b1:logic:statement}{وعبارة} «$f$ متباين إذا وفقط إذا كانت $\ker f$ محايدة»
        (\cref{prop:b1:structures:kernel}) لا نظير لها مع الصورة (إذ إن $\operatorname{im} f = G'$ هو الشمول).
  \item \emph{الحلقات ليست زمرًا بالنسبة إلى $\times$.} في \omterm{def:b1:structures:ring}{حلقة}، لا يلزم أن
        يكون معظم العناصر قابلًا للقلب، ويقتضي الاختصار بالعنصر $a$ أن يكون $a$ قابلًا للقلب أو أن تكون \omterm{def:b1:structures:field}{الحلقة تامة}:
        ففي $\Z/12\Z$، $\overline3\,\overline2 =
        \overline3\,\overline6$ ومع ذلك $\overline2 \neq \overline6$
        (\cref{ex:b1:structures:zncomputation}).
\end{enumerate}
\end{remark}

\begin{definition}[تشاكل الزمر]\label{def:b1:structures:morphism}
لتكن $(G, *)$ و $(G', \star)$ زمرتين. \omterm{def:b1:logic:map}{التطبيق} $f \colon G \to G'$
يكون \emph{تشاكلًا}\index{تشاكل!الزمر} عندما يكون
\[
\forall x, y \in G, \qquad f(x * y) = f(x) \star f(y).
\]
وعندئذ $f(e_G) = e_{G'}$ و$f(x^{-1}) = f(x)^{-1}$. و
\emph{نواة}\index{نواة} $f$ و\emph{صورتها} هما
\[
\ker f = f^{-1}(\{e_{G'}\}) \leq G,
\qquad
\operatorname{im} f = f(G) \leq G' .
\]
والتشاكل التقابلي يُسمّى \emph{تماثلًا}؛ ويكون تطبيقه العكسي عندئذ
تشاكلًا تلقائيًا.
\end{definition}

\begin{proof}[برهان الدعاوى]
$f(e) = f(e * e) = f(e)\star f(e)$، وباختصار $f(e)$ نجد
$e_{G'} = f(e)$. ثم يعيّن $f(x)\star f(x^{-1}) = f(x x^{-1}) = e_{G'}$
المقدارَ $f(x^{-1})$ بوصفه المقلوب. وأمّا \omterm{def:b1:structures:morphism}{النواة}: فلدينا $e \in \ker f$؛ وإذا كان
$x, y \in \ker f$ فإن $f(xy^{-1}) = f(x)f(y)^{-1} = e$؛ فينطبق المعيار.
وأمّا الصورة: فالمعيار نفسه مع $f(x)f(y)^{-1} = f(xy^{-1})$.
ومقلوب التماثل: من أجل $u, v \in G'$، اكتب $u = f(x)$ و$v =
f(y)$؛ عندئذ $f^{-1}(u \star v) = f^{-1}(f(xy)) = xy =
f^{-1}(u) f^{-1}(v)$.
\end{proof}

\begin{proposition}[التباين عبر النواة]\label{prop:b1:structures:kernel}
تشاكل الزمر $f$ متباين إذا وفقط إذا كان $\ker f =
\{e\}$.
\end{proposition}

\begin{proof}
إذا كان $f$ \omterm{def:b1:logic:inj}{متباينًا} فلا يمكن أن تحتوي $\ker f$ إلا سابقة $e_{G'}$
الوحيدة، وهي $e$. وبالعكس، إذا كان $\ker f = \{e\}$ و$f(x) =
f(y)$ فإن $f(xy^{-1}) = f(x) f(y)^{-1} = e_{G'}$، ومنه $xy^{-1} = e$،
أي $x = y$.
\end{proof}

\begin{example}\label{ex:b1:structures:expmorphism}
\omterm{def:b1:logic:map}{التطبيق} $\exp \colon (\R, +) \to (\R_+^*, \times)$ تشاكل
($\eu^{x+y} = \eu^x \eu^y$)، وهو تقابليّ
(\cref{prop:b1:functions:expln}): فالبنيتان الجمعية والضربية
متماثلتان --- وهذا هو مبرر وجود اللوغاريتمات تاريخيًا.
وتشاكل آخر: $\theta \mapsto \eu^{\iu\theta}$ من
$(\R, +)$ على دائرة الوحدة $(\mathbb{U}, \times)$، ونواته
$2\pi\Z$.
\end{example}

\begin{example}[تشاكل الإشارة]\label{ex:b1:structures:signmorphism}
\omterm{def:b1:logic:map}{التطبيق} $s \colon (\R^*, \times) \to (\{\pm1\}, \times)$ الذي يرسل
$x$ إلى إشارته تشاكلٌ: فإشارة الجداء هي
جداء الإشارتين. ونواته $\intoo0{+\infty}$ (وهي
\omterm{def:b1:structures:subgroup}{زمرة جزئية}، كما يعد \cref{def:b1:structures:morphism})، وصورته
كل $\{\pm1\}$: فهو شامل، وغير متباين بإفراط.
وفي ذلك درسان عامّان مصغَّران. أولًا، قد يسحق التشاكل المعلومة: \omterm{def:b1:logic:map}{فالتطبيق} $s$ لا يتذكر من $x$ إلا بتّة واحدة، وتلك
فضيلته --- إذ إن حجج الإشارة هي بالضبط الحسابات
التي تمرّ عبر $s$. وثانيًا، التشاكلات نحو $\{\pm1\}$ هي
أبسط «الثوابت»: فتوقيع التبديلات، المبنيّ في
مسألة نهاية الأسبوع من هذا الفصل، هو الظاهرة نفسها على
\omterm{def:b1:structures:group}{الزمرة} $\mathfrak S_n$، وكل حجج الزوجية التي يشغّلها
تنحدر عبر تشاكل ثنائي القيمة كهذا.
\end{example}

\begin{definition}[القوى، رتبة عنصر]\label{def:b1:structures:order}
في \omterm{def:b1:structures:group}{زمرة} $G$ (بالترميز الضربي)، نضع $x^0 = e$ و$x^{k+1} =
x^k x$ و $x^{-k} = (x^k)^{-1}$ من أجل $k \in \N$؛ عندئذ $x^{k+l} = x^k
x^l$ لكل $k, l \in \Z$، فيكون $k \mapsto x^k$ \omterm{def:b1:structures:morphism}{تشاكلًا} $(\Z,
+) \to G$ صورته $\langle x \rangle = \{x^k : k \in \Z\}$ زمرةٌ
جزئية، هي \omterm{def:b1:structures:subgroup}{الزمرة الجزئية} \emph{المولَّدة} بالعنصر $x$. و
\emph{رتبة}\index{رتبة عنصر} $x$ هي أصغر $m \geq
1$ يحقق $x^m = e$ إن وُجد (وعندئذ يكون \omterm{def:b1:logic:sets}{للمجموعة} $\langle x\rangle = \{e, x,
\dots, x^{m-1}\}$ بالضبط $m$ عنصرًا، و$x^k = e \iff m
\mid k$)، وتكون $\infty$ فيما عدا ذلك.
\end{definition}

\begin{example}\label{ex:b1:structures:order}
في $(\C^*, \times)$: \omterm{def:b1:structures:order}{رتبة} $\iu$ هي $4$، مع $\langle \iu \rangle
= \{1, \iu, -1, -\iu\} = \mathbb{U}_4$؛ وبعمومية أكبر، \omterm{def:b1:structures:order}{رتبة} $\omega =
\eu^{2\iu\pi/n}$ هي $n$ و$\langle\omega\rangle =
\mathbb{U}_n$. وفي $(\Z, +)$، لكل $x \neq 0$ \omterm{def:b1:structures:order}{رتبة} لا نهائية.
ولماذا تصحّ الدعاوى الواردة في التعريف: إذا كانت \omterm{def:b1:structures:order}{رتبة} $x$ هي $m$، فاقسم
أيّ $k$ على $m$ (أي $k = mq + r$ حيث $0 \leq r < m$،
\cref{thm:b1:arith:division}): عندئذ $x^k = (x^m)^q x^r = x^r$، فتدور
القوى بدور $m$، وتكون العناصر المسرودة متمايزة مثنى
مثنى بأصغرية $m$، ويفرض $x^k = e$ أن $r = 0$. وتُحسب رتب
التبديلات في مسألة نهاية الأسبوع أدناه.
\end{example}

\begin{example}[الرتب داخل $\mathbb U_{12}$]\label{ex:b1:structures:orderU12}
ما \omterm{def:b1:structures:order}{رتبة} $\omega^k$ في $\mathbb U_n$، من أجل $\omega =
\eu^{2\iu\pi/n}$؟ لدينا $(\omega^k)^m = 1$ إذا وفقط إذا كان $n \mid km$، و
بكتابة $d = \gcd(n, k)$ و $n = dn'$ و $k = dk'$ حيث $\gcd(n', k')
= 1$: يكون $n \mid km \iff n' \mid k'm \iff n' \mid m$ (بمبرهنة غاوس
المساعدة، \cref{thm:b1:arith:gauss}). وأصغر $m \geq 1$ كهذا هو
$n' = \frac{n}{\gcd(n,k)}$. ففي $\mathbb U_{12}$ مثلًا، \omterm{def:b1:structures:order}{رتبة}
$\omega^8$ هي $\frac{12}{\gcd(12,8)} = 3$ (وبالفعل
$\omega^8 = \eu^{4\iu\pi/3} \in \mathbb U_3$)، بينما \omterm{def:b1:structures:order}{رتبة} $\omega^5$
هي $12$: فهو يولّد \omterm{def:b1:structures:group}{الزمرة} كلها، وإن لم يكن
المولّد «المعياري». وعدّ المولّدات --- أي الأعداد $k$
التي تحقق $\gcd(k, n) = 1$ --- يستعيد عدّ الأعداد \omterm{cor:b1:arith:bezout}{الأولية فيما بينها} في
\cref{ex:b1:counting:coprime}: فتلتقي نظرية الزمر بالعدّ.
\end{example}

\section{الحلقات والحقول}

\begin{definition}[الحلقة]\label{def:b1:structures:ring}
\emph{الحلقة}\index{حلقة} $(A, +, \times)$ مجموعةٌ ذات قانونين بحيث
تكون: $(A, +)$ زمرةً \omterm{def:b1:structures:group}{تبديلية} (عنصرها المحايد $0$)؛ ويكون $\times$
تجميعيًا ذا عنصر محايد $1$؛ ويكون $\times$ توزيعيًا على $+$ من
الجهتين. وتكون الحلقة \emph{\omterm{def:b1:structures:group}{تبديلية}} عندما يكون $\times$ كذلك. ويكون
العنصر $a$ \emph{قابلًا للقلب} عندما يكون $ab = ba = 1$
من أجل $b$ ما؛ وتكوّن العناصر القابلة للقلب زمرةً $(A^\times, \times)$.
\end{definition}

\begin{proof}[برهان أن العناصر القابلة للقلب تكوّن زمرة]
الاستقرار: إذا كان $a, a'$ قابلين للقلب ومقلوباهما $b, b'$ فإن
\[
(aa')(b'b) = a(a'b')b = a\,1\,b = ab = 1,
\qquad (b'b)(aa') = 1
\]
وبالتناظر كذلك، ومنه فإن $aa'$ قابل للقلب. والعنصر $1$ قابل للقلب (وهو
مقلوب نفسه)، والتجميعية موروثة من $A$، ومقلوب
العنصر القابل للقلب $a$، وهو $b$، قابل للقلب بنفسه (ومقلوبه $a$).
ومنه فإن $(A^\times, \times)$ يحقق كل بديهيات \omterm{def:b1:structures:group}{الزمرة}. وكل \omterm{def:b1:structures:group}{زمرة}
في هذا الكتاب غير مبنيّة من التبديلات تنشأ على هذا النحو:
$\Q^* = \Q^\times$ و $\R^*$ و $\C^*$، والعناصر القابلة للقلب في $\Z/n\Z$ أدناه،
ولاحقًا المصفوفات القابلة للقلب (\cref{ch:b1:matrices}).
\end{proof}

\begin{example}\label{ex:b1:structures:rings}
المجموعات $\Z, \Q, \R, \C$ حلقات \omterm{def:b1:structures:group}{تبديلية}؛ و$\Z^\times = \{1, -1\}$
و $\Q^\times = \Q^*$. ولاحقًا: حلقات كثيرات الحدود $K[X]$
(\cref{ch:b1:poly})، وحلقات المصفوفات (وهي غير \omterm{def:b1:structures:group}{تبديلية}،
\cref{ch:b1:matrices})، و $\Z/n\Z$ أدناه. وفي كل \omterm{def:b1:structures:ring}{حلقة}، $0 \times
a = 0$ (من التوزيعية: $0a = (0+0)a = 0a + 0a$)، و
$(-1)a = -a$.
\end{example}

\begin{example}[العناصر الجامدة: ظواهر جديدة في حلقات جديدة]\label{ex:b1:structures:idempotents}
في $\Z$، ليس للمعادلة $x^2 = x$، أي $x(x - 1) = 0$، إلا
الحلان $0$ و $1$. وفي $\Z/6\Z$، بتجريب كل الصفوف:
$\overline0^2 = \overline0$ و$\overline1^2 = \overline1$ و
$\overline3^2 = \overline9 = \overline3$ و$\overline4^2 =
\overline{16} = \overline4$ --- أي \emph{أربعة} عناصر جامدة. والعنصران
الغريبان يأتيان من قواسم الصفر: $\overline3\,(\overline3 -
\overline1) = \overline3 \times \overline2 = \overline6 =
\overline0$ دون أن ينعدم أيّ من العاملين. وحسابات كهذه تعاير
الحدس: فالوقائع المألوفة عن المعادلات تصمد في
الحلقات التامة والحقول، لكن \omterm{def:b1:structures:ring}{الحلقة} العامة يمكن أن تسلك
سلوكًا مختلفًا وهي تفعل --- وانظر كذلك الحلقات البوليانية في
\cref{exo:b1:structures:10}، حيث يكون \emph{كل} عنصر
جامدًا.
\end{example}

\begin{proposition}[مبرهنة ثنائي الحدّ في حلقة تبديلية]\label{prop:b1:structures:binomial}
إذا كان $a, b$ عنصرين من \omterm{def:b1:structures:ring}{حلقة} \omterm{def:b1:structures:group}{تبديلية} (وبعمومية أكبر، إذا كان
$ab = ba$)، فإنه من أجل $n \in \N$:
\[
(a+b)^n = \sum_{k=0}^n \binom nk a^k b^{n-k},
\qquad
a^n - b^n = (a - b) \sum_{k=0}^{n-1} a^k b^{\,n-1-k} .
\]
\end{proposition}

\begin{proof}
لا يستعمل برهانا \cref{thm:b1:counting:binomial} والمتطابقة
الهندسية إلا التجميعية وتبديل العنصرين
والتوزيعية --- فهما ينطبقان حرفيًا.
\end{proof}

\begin{example}[مبرهنة ثنائي الحدّ في حلقة غير مألوفة]\label{ex:b1:structures:binomialring}
فائدتان سريعتان من هذه العمومية. ففي $\Z/p\Z$ (حيث $p$ أوليّ)، تنعدم
المعاملات الثنائية الوسطى
(وهي الخطوة الأولى من \cref{thm:b1:arith:fermat})، فتنهار المبرهنة
إلى \emph{حلم المبتدئ}
\[
(a + b)^p = a^p + b^p \qquad \text{في } \Z/p\Z ,
\]
وهي متطابقة حقيقية هناك، مهما بدت إجرامية فوق $\R$.
وفي أيّ \omterm{def:b1:structures:ring}{حلقة} \omterm{def:b1:structures:group}{تبديلية} تحوي عنصرًا $\varepsilon$
يحقق $\varepsilon^2 = 0$، تُبتر المبرهنة:
$(a + \varepsilon)^n = a^n + n\,a^{n-1}\varepsilon$، إذ تحمل كل الحدود
الأعلى عاملًا $\varepsilon^2 = 0$. والمعامل
$n\,a^{n-1}$ للمقدار $\varepsilon$ هو مشتقة $x^n$ ---
وليس هذا مصادفة، وهو أول إشارة إلى أن المشتقات جبرٌ بقدر
ما هي تحليل (قارن المشتقة الصورية في
\cref{ch:b1:poly}).
\end{example}

\begin{definition}[الحلقة التامة، الحقل]\label{def:b1:structures:field}
\omterm{def:b1:structures:ring}{الحلقة} \omterm{def:b1:structures:group}{التبديلية} $A \neq \{0\}$ تكون \emph{حلقة
تامة}\index{حلقة تامة} عندما لا تكون فيها قواسم للصفر: أي $ab = 0
\implies a = 0$ أو $b = 0$. وتكون \emph{حقلًا}\index{حقل} عندما يكون
كل عنصر غير معدوم قابلًا للقلب. وكل حقل حلقةٌ تامة (إذ $ab = 0$ مع $a \neq 0$ يعطيان $b = a^{-1}ab = 0$).
\end{definition}

\begin{example}
المجموعات $\Q$ و $\R$ و $\C$ حقول؛ و $\Z$ \omterm{def:b1:structures:field}{حلقة تامة} لكنها ليست
\omterm{def:b1:structures:field}{حقلًا}. وفي \omterm{def:b1:structures:field}{حلقة تامة} يصحّ الاختصار بالنسبة إلى $\times$: إذ $ab =
ac$ و $a \neq 0$ يستلزمان $b = c$.
\end{example}

\section{الحلقة $\Z/n\Z$}

\begin{definition}\label{def:b1:structures:zn}
ثبّت $n \in \N^*$. تكوّن صفوف \omterm{def:b1:arith:congruence}{التوافق} بترديد $n$
(\cref{ex:b1:logic:congruence}) مجموعةً
$\Z/n\Z$\index{Z/nZ@$\Z/n\Z$} ذات $n$ عنصر، وتُكتب $\overline 0,
\overline 1, \dots, \overline{n-1}$. والعمليتان
\[
\overline a + \overline b = \overline{a + b},
\qquad
\overline a \times \overline b = \overline{ab}
\]
معرَّفتان تعريفًا سليمًا --- فصفوف النتائج لا تتعلق
بالممثلين، وذلك بالضبط لأن \omterm{def:b1:arith:congruence}{التوافق} متوافق مع $+$
و $\times$ (\cref{def:b1:arith:congruence}) --- وتجعلان $\Z/n\Z$ \omterm{def:b1:structures:ring}{حلقة}
\omterm{def:b1:structures:group}{تبديلية}.
\end{definition}

\begin{theorem}[العناصر القابلة للقلب في $\Z/n\Z$؛ الحقول $\Z/p\Z$]\label{thm:b1:structures:zp}
\begin{enumerate}
  \item الصف $\overline a$ قابل للقلب في $\Z/n\Z$ إذا وفقط إذا كان
        $\gcd(a, n) = 1$.
  \item \omterm{def:b1:structures:ring}{الحلقة} $\Z/n\Z$ حقلٌ إذا وفقط إذا كان $n$ \omterm{def:b1:arith:prime}{أوليًا}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) هي \cref{prop:b1:arith:invmod} معادة الصياغة بالصفوف.

(2) إذا كان $n = p$ \omterm{def:b1:arith:prime}{أوليًا} فإن كل $\overline a \neq \overline 0$ يحقق $p
\nmid a$، ومنه $\gcd(a, p) = 1$: فهو قابل للقلب حسب (1) --- أي \omterm{def:b1:structures:field}{حقل}. وإذا كان $n =
ab$ مع $1 < a, b < n$ فإن $\overline a\, \overline b = \overline
n = \overline 0$ مع $\overline a, \overline b \neq \overline 0$:
أي قواسم للصفر، فلا تكون حتى \omterm{def:b1:structures:field}{حلقة تامة}؛ و $n = 1$ يعطي
\omterm{def:b1:structures:ring}{الحلقة} الصفرية، وهي مستبعدة.
\end{proof}

\begin{example}[كم جذرًا تربيعيًا للعدد $1$؟]\label{ex:b1:structures:sqrtone}
حُلَّ $x^2 = \overline 1$ في $\Z/8\Z$ وفي $\Z/7\Z$. وبتجريب
الصفوف الثمانية بترديد $8$: $1^2 = 1$ و $3^2 = 9 \equiv 1$ و$5^2 = 25
\equiv 1$ و $7^2 = 49 \equiv 1$ --- أي \emph{أربعة} حلول
$\{\overline1, \overline3, \overline5, \overline7\}$، وإن كانت
درجة كثير الحدود $X^2 - 1$ هي $2$. وعلى النقيض، في \omterm{def:b1:structures:field}{الحقل} $\Z/7\Z$
تعني $x^2 = \overline1$ أن $(x - \overline1)(x +
\overline1) = \overline0$، وليس في \omterm{def:b1:structures:field}{الحقل} قواسم للصفر: ومنه $x =
\pm\overline1$، أي حلّان لا غير. والإخفاق بترديد $8$ قابل
للتعقب: $(3-1)(3+1) = 2 \times 4 = 8 \equiv 0$ دون أن ينعدم أيّ من
العاملين. والعبرة: أن القاعدة المألوفة «للمعادلة من الدرجة $d$
$d$ جذرًا على الأكثر» مبرهنةٌ عن
\emph{الحلقات التامة} (ويبرهن عليها \cref{cor:b1:poly:nroots} فوق
الحقول)؛ وهي تخفق ضمنًا في الحلقات ذات قواسم الصفر --- وهذا
بالضبط سبب حاجة برهان الازدواج لمبرهنة ويلسون
(\cref{exo:b1:arith:11}) إلى $p$ الأوليّ.
\end{example}

\begin{example}[الحساب في $\Z/n\Z$]\label{ex:b1:structures:zncomputation}
في $\Z/12\Z$: العناصر القابلة للقلب هي $\overline 1, \overline 5, \overline 7,
\overline{11}$ (أي الصفوف الأولية مع $12$)، وكلٌّ منها مقلوب
نفسه ($5^2 = 25 \equiv 1$ و $7^2 = 49 \equiv 1$ و$11^2 = 121 \equiv
1$). وللمعادلة $\overline 3\, x = \overline 6$ \emph{ثلاثة}
حلول ($x \in \{\overline 2, \overline 6, \overline{10}\}$):
فبلا قابلية للقلب لا اختصار. وعلى النقيض، في $\Z/11\Z$
يكون لكل معادلة $\overline a x = \overline b$ حيث $\overline a \neq
\overline 0$ حلٌّ واحد بالضبط.
\end{example}

\begin{example}[بديهيات الزمرة رخصةً للحلّ]\label{ex:b1:structures:solving}
في \omterm{def:b1:structures:group}{الزمرة} $\bigl((\Z/7\Z)^*, \times\bigr)$، حُلَّ $\overline
3\,x = \overline 5$. حسب \cref{prop:b1:structures:rules}~(3) يكون
الحلّ موجودًا ووحيدًا ويساوي $\overline3^{-1}\,
\overline5$؛ وبما أن $\overline3 \times \overline5 = \overline{15}
= \overline1$، يكون مقلوب $\overline 3$ هو $\overline 5$، ومنه
\[
x = \overline5 \times \overline5 = \overline{25} = \overline4,
\qquad\text{وللتحقق: } \overline3 \times \overline4 =
\overline{12} = \overline5 .
\]
والمهم هنا أقلّ من الجواب: إنه الضمان. ففي \omterm{def:b1:structures:group}{زمرة}،
تكون كل معادلة كهذه قابلة للحل حلًّا وحيدًا \emph{قبل} أيّ
حساب، فلا يمكن لإجراء الحلّ أن يصطدم \omterm{def:b1:logic:statement}{بعبارة} «لا حلّ» ولا \omterm{def:b1:logic:statement}{بعبارة} «عدة حلول». وقارن ذلك بالمعادلة $\overline3\,x = \overline6$
في $\Z/12\Z$ أعلاه، حيث يخفق الضمان --- فمعرفة أيّ
بنية نحن فيها هي معرفة ما يجوز التسليم به.
\end{example}

\begin{example}[الجداءات المباشرة]\label{ex:b1:structures:product}
إذا كانت $G$ و $H$ زمرتين، فإن \omterm{def:b1:logic:sets}{المجموعة} الجدائية $G \times H$ مع
القانون المركّبة مركّبةً $(g, h)(g', h') = (gg', hh')$ زمرةٌ:
إذ تُتحقَّق البديهيات إحداثيةً إحداثية، وعنصرها المحايد
$(e_G, e_H)$ ومقلوباتها $(g^{-1}, h^{-1})$. وتتركّب الرتب \omterm{def:b1:arith:lcm}{بالمضاعف المشترك الأصغر}: فالمقدار $(g, h)^m = (g^m, h^m)$ هو العنصر المحايد إذا وفقط إذا قسمت \omterm{def:b1:structures:order}{رتبة}
$g$ \omterm{def:b1:structures:order}{ورتبة} $h$ كلتاهما $m$. ومنه ففي $\Z/2\Z
\times \Z/2\Z$ (الجمعية) لكل عنصر غير معدوم رتبةٌ $2$
--- وهذه بالضبط \omterm{def:b1:structures:group}{زمرة} المستطيل في
\cref{ex:b1:structures:klein} بالإحداثيات --- بينما في $\Z/4\Z$
عنصر رتبته $4$: وهو برهان ثانٍ، بلا حساب،
على أن الزمرتين ذواتي الحجم $4$ غير متماثلتين (فالتماثل
يحفظ الرتب). والجداءات أسهل سبيل إلى
تصنيع زمر جديدة من قديمة، والمستوي $\R^2 = \R \times
\R$ في \cref{ch:b1:vspaces} هو أهمّ حالات هذا الإنشاء.
\end{example}

\begin{remark}[فيرما، بنيويًا]
في \omterm{def:b1:structures:field}{الحقل} $\Z/p\Z$، تكوّن الصفوف غير المعدومة زمرةً ضربية
ذات $p - 1$ عنصر، وتقول مبرهنة فيرما الصغرى
(\cref{thm:b1:arith:fermat}): إن كل عنصر $x$ من هذه \omterm{def:b1:structures:group}{الزمرة}
يحقق $x^{p-1} = \overline 1$. وهذه حالة خاصة من واقعة عامة
عن الزمر المنتهية (مبرهنة لاغرانج)، مبرهن عليها في السنة
الثانية؛ وكان لبرهان الازدواج لمبرهنة ويلسون
(\cref{exo:b1:arith:11}) هذا النكهة الزمرية أصلًا.
\end{remark}

\begin{remark}[استراحة: ماذا يشتري التجريد]\label{rem:b1:structures:interlude}
من المنصف أن يُسأل عمّا رُبح بالبرهان على
\cref{prop:b1:structures:unique} من أجل قانون مجرد بدل
البرهان عليها من أجل الأعداد. والجواب هو الرافعة. فتلك الحجة ذات السطرين تغطي
الآن، دفعةً واحدة: مقلوبات الدوال بالتركيب
(\cref{thm:b1:logic:inverse}، الذي يكرر برهان وحدانيتها
حرفًا بحرف)، والمقلوبات بترديد $n$
(\cref{prop:b1:arith:invmod})، ومقلوبات الأعداد الحقيقية غير المعدومة، و
مقلوبات العناصر القابلة للقلب في أيّ \omterm{def:b1:structures:ring}{حلقة}، و--- من غير أن نراها --- مقلوبات
المصفوفات القابلة للقلب في \cref{ch:b1:matrices}، حيث لن تحتاج وحدانية $A^{-1}$
إلى سطر برهان واحد. والاقتصاد نفسه يصحّ من أجل
\cref{prop:b1:structures:kernel} (معيار تباين واحد،
يُعاد استعماله من أجل التطبيقات الخطية في \cref{ch:b1:linmaps}) ومن أجل
معيار \omterm{def:b1:structures:subgroup}{الزمرة الجزئية}. فالتجريد هنا ليس عمومية لذاتها:
بل هو رفض البرهان على المبرهنة المساعدة نفسها خمس مرات
بخمسة أسماء. والثمن --- أي تتبّع أيّ البديهيات
استعملتها كل \omterm{def:b1:logic:statement}{عبارة} فعلًا --- هو بالضبط ما تدرّبه تمارين
هذا الفصل.
\end{remark}

\begin{remark}[أين يُستعمل هذا الفصل]\label{rem:b1:structures:whereused}
مفردات هذا الفصل هي نحو بقية
المجلد. فالحلقات والحقول تنظّم \cref{ch:b1:poly} (\omterm{def:b1:structures:ring}{فالحلقة} $K[X]$ تحاكي $\Z$) و\cref{ch:b1:fractions} (\omterm{def:b1:logic:sets}{فالمجموعة} $K(X)$ هي \omterm{def:b1:structures:field}{حقل} كسورها)؛ والفضاءات المتجهية (\cref{ch:b1:vspaces})
زمرٌ \omterm{def:b1:structures:group}{تبديلية} يؤثر فيها \omterm{def:b1:structures:field}{حقل}؛ والمصفوفات
(\cref{ch:b1:matrices}) تكوّن أول \omterm{def:b1:structures:ring}{حلقة} غير \omterm{def:b1:structures:group}{تبديلية}
جدّيًا في المجلد، وتكوّن عناصرها القابلة للقلب زمرةً دراستها
هي الجبر الخطي نفسه. وتعود التشاكلات والنوى في صورة
تطبيقات خطية وفضاءات \omterm{def:b1:structures:morphism}{نواة} في \cref{ch:b1:linmaps} --- إذ إن \cref{prop:b1:structures:kernel} \emph{هي} معيار التباين
في ذلك الفصل، المبرهن عليه مرة واحدة إلى الأبد هنا. و
\omterm{ex:b1:structures:groups}{الزمرة المتناظرة}، نجمة مسألة نهاية الأسبوع أدناه، توفّر
التوقيع الذي تُبنى عليه المحددات في \cref{ch:b1:det}.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:structures:1}
على $E = \R \setminus \{1\}$، عرّف $x * y = x + y - xy$. برهن على أن
$(E, *)$ \omterm{def:b1:structures:group}{زمرة تبديلية}. \emph{(عيّن العنصر المحايد ومقلوب
$x$؛ وتحقق من الاستقرار: لماذا يكون $x * y \neq 1$؟)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:2}
أيٌّ ممّا يلي \omterm{def:b1:structures:group}{زمرة}؟
\begin{enumerate}
  \item $(\intoo{0}{+\infty}, \times)$؛
  \item $(\{-1, 0, 1\}, +)$؛
  \item $(\Q^*, \times)$؛
  \item \omterm{def:b1:logic:sets}{مجموعة} الأعداد الصحيحة الفردية بالجمع.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:3}
اكتب جدول التركيب \omterm{ex:b1:structures:groups}{للزمرة المتناظرة} $\mathfrak{S}_3$
\omterm{def:b1:logic:sets}{للمجموعة} $\{1,2,3\}$ (وفيها ستة تقابلات: \omterm{def:b1:logic:map}{التطبيق} المطابق، وثلاث منقولات، ودورتان
من \omterm{def:b1:structures:order}{الرتبة} $3$)، وأبرز عنصرين لا يتبادلان.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:4}
برهن على أن $H = \{z \in \C^* : \abs z = 1\}$ \omterm{def:b1:structures:subgroup}{زمرة جزئية} من $(\C^*,
\times)$، وأن $\R_+^*$ \omterm{def:b1:structures:subgroup}{زمرة جزئية} أخرى؛ وهل $H \cup \R_+^*$ \omterm{def:b1:structures:subgroup}{زمرة
جزئية}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:5}
لتكن $f \colon (\R, +) \to (\C^*, \times)$ و$\theta \mapsto
\eu^{\iu\theta}$. برهن على أن $f$ تشاكل، واحسب $\ker f$ و
$\operatorname{im} f$، واستنتج من
\cref{prop:b1:structures:kernel} أن $f$ غير متباين. وقيّد
\omterm{def:b1:logic:sets}{مجموعة} التعريف لجعله \omterm{def:b1:logic:inj}{متباينًا} على أوسع مجال ممكن.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:6}
لتكن $H, K$ زمرتين جزئيتين من $G$. برهن على أن $H \cap K$ \omterm{def:b1:structures:subgroup}{زمرة جزئية}،
وأن $H \cup K$ لا تكون \omterm{def:b1:structures:subgroup}{زمرة جزئية} \emph{إلا} عندما $H \subseteq K$ أو
$K \subseteq H$. \emph{(إذا كان $h \in H \setminus K$ و$k \in K
\setminus H$، فأين يمكن أن يحيا $hk$؟)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:7}
\omterm{def:b1:structures:group}{زمرة} $G$ تحقق $x^2 = e$ لكل $x \in G$. برهن على أن $G$
\omterm{def:b1:structures:group}{تبديلية}. \emph{(انشر $(xy)^2$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:8}
في $\Z/18\Z$: اسرد العناصر القابلة للقلب وجد مقلوب $\overline 5$؛
وحُلَّ $\overline 5\, x = \overline 7$؛ وحُلَّ $\overline 6\, x =
\overline 3$ و$\overline 6\, x = \overline{12}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:9}
برهن على أن \omterm{def:b1:logic:sets}{المجموعة} $\Z[\sqrt 2] = \{a + b\sqrt 2 : a, b \in \Z\}$
\omterm{def:b1:structures:ring}{حلقة} (\omterm{def:b1:structures:ring}{حلقة} جزئية من $\R$)، وأن $1 + \sqrt 2$ عنصر قابل للقلب فيها
وله قوى متمايزة لا نهائية العدد --- ومنه فإن $\Z[\sqrt 2]^\times$
لا نهائية، بخلاف $\Z^\times$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:structures:10}
(الحلقات البوليانية) لتكن $A$ \omterm{def:b1:structures:ring}{حلقة} تحقق $x^2 = x$ من أجل كل $x$.
برهن على أن $x + x = 0$ لكل $x$، وأن $A$ \omterm{def:b1:structures:group}{تبديلية}.
\emph{(انشر $(x+x)^2$ و $(x+y)^2$.)} وأعط مثالًا على \omterm{def:b1:structures:ring}{حلقة}
كهذه \omterm{def:b1:logic:sets}{بالمجموعة} $\mathcal{P}(E)$، بأخذ الفرق المتناظر
جمعًا والتقاطع ضربًا.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:structures:11}
لتكن $G$ \omterm{def:b1:structures:group}{زمرة} يتحقق فيها، من أجل $n \geq 1$ مثبَّت، $(xy)^n =
x^n y^n$ و$(xy)^{n+1} = x^{n+1}y^{n+1}$ و$(xy)^{n+2} =
x^{n+2}y^{n+2}$ لكل $x, y$. برهن على أن $G$ \omterm{def:b1:structures:group}{تبديلية}.
\emph{(انطلاقًا من المتطابقات الثلاث، استخرج أولًا $y^n x = x y^n$، ثم
$y^{n+1} x = x y^{n+1}$، واستنتج.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:12}
\begin{enumerate}
  \item عيّن كل تشاكلات الزمر من $(\Z, +)$ إلى $(\Z, +)$.
  \item برهن على أن التشاكل الوحيد من $(\Q, +)$ إلى
        $(\Z, +)$ هو التشاكل المعدوم. \emph{(من أجل $x \in \Q$ و
        $n \in \N^*$، قارن $f(x)$ مع $n\,f(x/n)$.)}
\end{enumerate}
\end{exercise}

\section{مسألة: الزمرة المتناظرة ولغز الثمانية}

\begin{problem}\label{pb:b1:structures:1}
\omterm{def:b1:structures:group}{الزمرة} $\mathfrak S_n$ المكوَّنة من تبديلات $\intint1n$ أقدم
\omterm{def:b1:structures:group}{زمرة} في الرياضيات وأكثرها إفادةً إلى اليوم. وتبني هذه
المسألة نظرية بنيتها من الصفر --- الدورات، والتوليد
بالمنقولات، وتشاكل \emph{التوقيع}
$\varepsilon \colon \mathfrak S_n \to \{\pm1\}$ (الذي يكون وجوده
غير بديهي حقًّا)، \omterm{pb:b1:structures:1}{والزمرة المتناوبة} $\mathfrak A_n$
المولَّدة بالدورات من \omterm{def:b1:structures:order}{الرتبة} $3$ --- ثم تصرف ذلك على لغز
كلاسيكي: ففي لعبة البلاطات المنزلقة $3 \times 3$، لا يمكن لأيّ متتالية حركات
أن تبادل بلاطتين وتترك كل شيء آخر في مكانه.
\index{زمرة متناظرة}\index{توقيع}\index{زمرة متناوبة}
تؤثر التبديلات في $\intint1n$؛ والجداءات $\sigma\tau$ تعني
«طبّق $\tau$ أولًا»؛ ويرمز $[\,v_1, \dots, v_n]$ إلى
\omterm{def:b1:counting:objects}{التبديلة} التي ترسل $i$ إلى $v_i$.

\medskip
\noindent\textbf{الجزء 1 --- الدورات والمنقولات.}
\begin{enumerate}
  \item برّر أن $\abs{\mathfrak S_n} = n!$
        (\cref{thm:b1:counting:counts}). وفي $\mathfrak S_3$،
        احسب جداءَي $\sigma = [2, 3, 1]$ و$\tau =
        [1, 3, 2]$ بالترتيبين، واستنتج أن $\mathfrak S_3$ غير
        \omterm{def:b1:structures:group}{تبديلية}.
  \item \emph{الدورة من \omterm{def:b1:structures:order}{الرتبة} $k$} $(a_1\ a_2\ \dots\ a_k)$ (حيث $k \geq 2$،
        والعناصر $a_i$ متمايزة مثنى مثنى) ترسل $a_1 \mapsto a_2
        \mapsto \dots \mapsto a_k \mapsto a_1$ وتصمد عند
        كل ما عدا ذلك؛ و\emph{حاملها} هو $\{a_1, \dots,
        a_k\}$. برهن على أن دورتين ذواتي حاملين منفصلين
        تتبادلان.
  \item برهن على أن كل $\sigma \in \mathfrak S_n$ جداءُ
        دورات ذوات حوامل منفصلة مثنى مثنى، وأن هذا
        التفكيك وحيد بغضّ النظر عن ترتيب العوامل.
        \emph{(تأمّل، من أجل كل $i$، المتتالية $i, \sigma(i),
        \sigma^2(i), \dots$: فلا بدّ أن تعود إلى $i$؛ وتجزّئ
        \emph{المدارات} الناتجة المجموعةَ $\intint1n$، وتؤثر $\sigma$ في
        كلٍّ منها بوصفها دورة.)}
  \item فكّك $\sigma = [4, 1, 5, 2, 3, 7, 8, 6] \in
        \mathfrak S_8$ إلى دورات منفصلة. وبتعريف
        \emph{\omterm{def:b1:structures:order}{رتبة}} $\sigma$ كما في
        \cref{def:b1:structures:order}، برهن على أن \omterm{def:b1:structures:order}{رتبة} جداء
        دورات منفصلة هي \omterm{def:b1:arith:lcm}{المضاعف المشترك الأصغر} لأطوالها،
        واحسب \omterm{def:b1:structures:order}{رتبة} هذه $\sigma$.
  \item برهن على المتطابقة التلسكوبية
        \[
        (a_1\ a_2\ \dots\ a_k)
        = (a_1\ a_k)(a_1\ a_{k-1})\cdots(a_1\ a_2) ,
        \]
        واستنتج أن كل \omterm{def:b1:counting:objects}{تبديلة} جداءُ
        منقولات. واكتب $\sigma$ من السؤال 4 في صورة جداء
        كهذا.
  \item بيّن كذلك أن المنقولات \emph{المتجاورة}
        $(i\ \ i{+}1)$ تكفي: فمن أجل $a < b$،
        \[
        (a\ b) = (a\ \ a{+}1)(a{+}1\ \ a{+}2)\cdots(b{-}1\ \ b)
        \cdots(a{+}1\ \ a{+}2)(a\ \ a{+}1),
        \]
        وهو جداء $2(b - a) - 1$ منقولة متجاورة ---
        أي عدد \emph{فرديّ} (وستهمّ هذه الزوجية مرتين
        أدناه).
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- التوقيع موجود.} من أجل $\sigma
\in \mathfrak S_n$، ليكن
\[
N(\sigma) = \#\bigl\{(i, j) : i < j,\ \sigma(i) >
\sigma(j)\bigr\}
\]
عدد \emph{انقلاباتها}، ولنضع $\varepsilon(\sigma) =
(-1)^{N(\sigma)}$.
\begin{enumerate}[resume]
  \item احسب $N$ و $\varepsilon$ من أجل \omterm{def:b1:logic:map}{التطبيق} المطابق، ومن أجل
        منقولة $(i\ \ i{+}1)$، ومن أجل $[2, 3, 1]$.
  \item برهن على أنه من أجل كل $\sigma$ وكل منقولة
        متجاورة $\tau = (i\ \ i{+}1)$:
        $N(\sigma\tau) = N(\sigma) \pm 1$. \emph{(فالتركيب مع
        $\tau$ من اليمين يبادل القيمتين في الموضعين $i$ و
        $i + 1$؛ فيتغير زوج واحد بالضبط في حالة
        انقلابه.)}
  \item استنتج، باستعمال السؤال 6، أنه من أجل \emph{أيّ}
        منقولة $\tau$ يكون $\varepsilon(\sigma\tau) =
        -\varepsilon(\sigma)$؛ واستنتج أنه إذا كانت $\sigma$
        جداءَ $p$ منقولة فإن $\varepsilon(\sigma)
        = (-1)^p$ --- وبوجه خاص لا تتعلق زوجية $p$
        إلا \omterm{def:b1:counting:objects}{بالتبديلة} $\sigma$، لا بالتعميل المختار --- وأن
        $\varepsilon \colon \mathfrak S_n \to \{\pm 1\}$ تشاكل زمر.
  \item بيّن أن توقيع الدورة من \omterm{def:b1:structures:order}{الرتبة} $k$ هو $(-1)^{k-1}$، وأنه
        في الحالة العامة $\varepsilon(\sigma) = (-1)^{n -
        c(\sigma)}$، حيث $c(\sigma)$ عدد مدارات
        $\sigma$ (بما فيها النقاط الصامدة).
  \item \emph{\omterm{pb:b1:structures:1}{الزمرة المتناوبة}} هي $\mathfrak A_n =
        \ker\varepsilon$. برّر أنها \omterm{def:b1:structures:subgroup}{زمرة جزئية}، وبرهن على أن
        $\abs{\mathfrak A_n} = \frac{n!}2$ من أجل $n \geq 2$.
        \emph{(ثبّت منقولة $\tau_0$ وتأمّل $\sigma
        \mapsto \sigma\tau_0$.)}
  \item تحقق اتساق على $\sigma = [4, 1, 5, 2, 3, 7, 8, 6]$:
        احسب $\varepsilon(\sigma)$ بثلاث طرق --- بعدّ
        الانقلابات، ومن نمط الدورات عبر السؤال 10، ومن
        عدّ منقولاتك في السؤال 5.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- \omterm{def:b1:structures:group}{الزمرة} $\mathfrak A_n$ مولَّدة
بالدورات من \omterm{def:b1:structures:order}{الرتبة} $3$.}
\begin{enumerate}[resume]
  \item ليكن $a, b, c, d$ متمايزة مثنى مثنى. تحقق من المتطابقتين
        \[
        (a\ b)(a\ c) = (a\ c\ b),
        \qquad
        (a\ b)(c\ d) = (a\ c\ b)(a\ c\ d) .
        \]
  \item برهن على أنه من أجل $n \geq 3$، يكون كل عنصر من $\mathfrak
        A_n$ جداءَ دورات من \omterm{def:b1:structures:order}{الرتبة} $3$. \emph{(\omterm{def:b1:counting:objects}{فالتبديلة}
        الزوجية جداءُ عدد زوجيّ من
        المنقولات؛ فامتصّها اثنتين اثنتين.)}
  \item اكتب $(1\ 2)(3\ 4)$ والدورة من \omterm{def:b1:structures:order}{الرتبة} $5$ التي هي $(1\ 2\ 3\ 4\ 5)$
        صراحةً في صورة جداءات دورات من \omterm{def:b1:structures:order}{الرتبة} $3$.
  \item برهن على صيغة المرافقة: من أجل كل $\sigma \in
        \mathfrak S_n$،
        \[
        \sigma\,(a_1\ \dots\ a_k)\,\sigma^{-1}
        = \bigl(\sigma(a_1)\ \dots\ \sigma(a_k)\bigr) .
        \]
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- لغز الثمانية.} تنزلق البلاطات $1, \dots, 8$
في إطار $3 \times 3$ فيه خانة خالية واحدة؛ و\emph{الحركة}
تُزلق بلاطة مجاورة للخانة الخالية إليها. رقّم الخانات
$1, \dots, 9$ (سطرًا بعد سطر؛ والوضع المحلول تكون فيه البلاطة $i$ في
الخانة $i$ والخانة الخالية في الخانة $9$). وعامل الخانة الخالية بوصفها
بلاطةً تاسعة، فيكون الوضع تبديلةً $\sigma \in \mathfrak
S_9$ (فالبلاطة $\sigma(i)$ تقع في الخانة $i$).
\begin{enumerate}[resume]
  \item بيّن أن الحركة تضع مكان $\sigma$ التبديلةَ $\sigma \circ \tau$
        حيث $\tau$ منقولة الخانتين المعنيتين؛ واستنتج أن كل حركة تقلب
        $\varepsilon(\sigma)$.
  \item لتكن $d(\sigma)$ المسافة (عدد الأسطر زائد عدد
        الأعمدة) بين خانة الخانة الخالية الحالية و
        خانتها الأصلية $9$. بيّن أن كل حركة تغيّر $d$ بمقدار $\pm1$،
        فتقلب كل حركة كذلك $(-1)^{d(\sigma)}$. واستنتج أن
        \[
        I(\sigma) = \varepsilon(\sigma)\cdot(-1)^{d(\sigma)}
        \]
        \emph{لا يتغير} بأيّ حركة.
  \item برهن على استحالة اللغز الكلاسيكية: أي أن الوضع
        الذي يبادل البلاطتين $7$ و $8$ ويترك كل شيء آخر
        (بما فيه الخانة الخالية) في مكانه لا يمكن بلوغه
        من الوضع المحلول.
  \item ونقبل العكس (وبرهانه استقراء مفيد لكن
        طويل): أي أن كل وضع يحقق $I = +1$ قابل للبلوغ. استنتج
        أن نصف الأوضاع $8!$ التي تكون فيها الخانة الخالية في
        مكانها بالضبط قابل للحلّ، أي
        $\frac{8!}2 = 20\,160$.
  \item استنتج من السؤال 20 أن ترتيبات البلاطات
        القابلة للبلوغ والتي تكون فيها الخانة الخالية في مكانها تكوّن بالضبط
        \omterm{def:b1:structures:subgroup}{الزمرة الجزئية} $\mathfrak A_8 \leq \mathfrak S_8$.
  \item تطبيقات الثابت: هل يمكن بلوغ (أ)
        الوضع الذي تُبدَّل فيه البلاطات $1, 2, 3$ تبديلًا دائريًا
        ويكون كل شيء آخر، بما فيه الخانة الخالية، في مكانه؟
        و (ب) الوضع الذي تبادلت فيه البلاطة $5$ والخانة الخالية
        موضعيهما وتكون كل البلاطات الأخرى في مكانها؟ برّر
        الجوابين بالمقدار $I$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- توليفة ختامية.}
\begin{enumerate}[resume]
  \item برهن على أنه من أجل $n \geq 3$ لا تكون تشاكلات الزمر $f
        \colon \mathfrak S_n \to \{\pm 1\}$ إلا التشاكلَ
        الثابت و $\varepsilon$. \emph{(باستعمال السؤال 16 و
        \omterm{def:b1:structures:group}{تبديلية} $\{\pm1\}$، بيّن أن $f$ يأخذ القيمة نفسها
        على كل المنقولات.)}
  \item أين استعملت المسألة بالضبط: (أ) مفهوم التشاكل
        و\cref{prop:b1:structures:kernel}؛ (ب) مبادئ
        العدّ في \cref{ch:b1:counting}؛ (ج) مسألة سلامة
        التعريف التي يحسمها السؤالان 8--9؟ جملة
        واحدة لكلٍّ منها.
  \item توليفة، في فقرة قصيرة: دالة زوجية واحدة،
        مبرهن على سلامة تعريفها مرة واحدة، تنظّم في آن واحد
        البنية الداخلية \omterm{def:b1:structures:group}{للزمرة} $\mathfrak S_n$ (أي \omterm{def:b1:structures:subgroup}{الزمرة الجزئية}
        $\mathfrak A_n$)، وتحسم لغزًا ماديًا، و---
        عبر الصيغة $\det = \sum_\sigma
        \varepsilon(\sigma)\cdots$ --- ستعرّف المحددات
        في \cref{ch:b1:det}. علّق على النمط المتكرر:
        فالثوابت تحوّل «جرّب كل متتاليات الحركات» إلى حساب
        واحد.
\end{enumerate}
\end{problem}
